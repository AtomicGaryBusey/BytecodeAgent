.globl _start

.text
_start:
    xor ebx, ebx
.Lread_loop:
    cmp rbx, 65536
    jae .Lread_done
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz .Lread_done
    add rbx, rax
    jmp .Lread_loop
.Lread_done:
    lea r12, [rip + input_buf]
    lea r13, [r12 + rbx]
.Lline_loop:
    cmp r12, r13
    jae .Ldone
    mov rcx, r12
.Lscan_line:
    cmp rcx, r13
    jae .Lline_end_no_newline
    cmp byte ptr [rcx], 10
    je .Lline_end_newline
    inc rcx
    jmp .Lscan_line
.Lline_end_newline:
    mov rdi, r12
    mov rsi, rcx
    push rcx
    call eval_line
    pop rcx
    lea r12, [rcx + 1]
    jmp .Lline_loop
.Lline_end_no_newline:
    mov rdi, r12
    mov rsi, rcx
    call eval_line
    mov r12, r13
    jmp .Lline_loop
.Ldone:
    mov eax, 60
    xor edi, edi
    syscall

eval_line:
    mov rbp, rsi
    mov r15, rdi
    xor r14d, r14d
    lea r10, [rip + val_stack]
.Ltok_loop:
    cmp r15, rbp
    jae .Lend_tokens
    movzx eax, byte ptr [r15]
    cmp al, 32
    je .Lskip_sep
    cmp al, 9
    je .Lskip_sep
    jmp .Ltoken_start
.Lskip_sep:
    inc r15
    jmp .Ltok_loop
.Ltoken_start:
    mov rbx, r15
.Lscan_token:
    inc r15
    cmp r15, rbp
    jae .Ltoken_done
    movzx eax, byte ptr [r15]
    cmp al, 32
    je .Ltoken_done
    cmp al, 9
    je .Ltoken_done
    jmp .Lscan_token
.Ltoken_done:
    mov rcx, r15
    sub rcx, rbx
    cmp rcx, 1
    jne .Lcheck_number
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod
.Lcheck_number:
    mov rsi, rbx
    mov rdi, r15
    xor r8d, r8d
    cmp rsi, rdi
    je .Lerror
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne .Lno_sign
    mov r8d, 1
    inc rsi
    cmp rsi, rdi
    je .Lerror
.Lno_sign:
    mov rcx, rdi
    sub rcx, rsi
    cmp rcx, 1
    jb .Lerror
    cmp rcx, 18
    ja .Lerror
    xor eax, eax
    mov r9, 10
.Ldigit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja .Lerror
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    cmp rsi, rdi
    jb .Ldigit_loop
    test r8d, r8d
    jz .Lpush
    neg rax
.Lpush:
    mov [r10 + r14*8], rax
    inc r14
    jmp .Ltok_loop

.Lop_add:
    cmp r14, 2
    jb .Lerror
    mov rdx, [r10 + r14*8 - 8]
    mov rax, [r10 + r14*8 - 16]
    add rax, rdx
    sub r14, 2
    mov [r10 + r14*8], rax
    inc r14
    jmp .Ltok_loop
.Lop_sub:
    cmp r14, 2
    jb .Lerror
    mov rdx, [r10 + r14*8 - 8]
    mov rax, [r10 + r14*8 - 16]
    sub rax, rdx
    sub r14, 2
    mov [r10 + r14*8], rax
    inc r14
    jmp .Ltok_loop
.Lop_mul:
    cmp r14, 2
    jb .Lerror
    mov rdx, [r10 + r14*8 - 8]
    mov rax, [r10 + r14*8 - 16]
    imul rax, rdx
    sub r14, 2
    mov [r10 + r14*8], rax
    inc r14
    jmp .Ltok_loop
.Lop_div:
    cmp r14, 2
    jb .Lerror
    mov rcx, [r10 + r14*8 - 8]
    mov rax, [r10 + r14*8 - 16]
    test rcx, rcx
    jz .Lerror
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    jne .Ldiv_ok
    cmp rcx, -1
    je .Lerror
.Ldiv_ok:
    cqo
    idiv rcx
    sub r14, 2
    mov [r10 + r14*8], rax
    inc r14
    jmp .Ltok_loop
.Lop_mod:
    cmp r14, 2
    jb .Lerror
    mov rcx, [r10 + r14*8 - 8]
    mov rax, [r10 + r14*8 - 16]
    test rcx, rcx
    jz .Lerror
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    jne .Lmod_ok
    cmp rcx, -1
    je .Lerror
.Lmod_ok:
    cqo
    idiv rcx
    sub r14, 2
    mov [r10 + r14*8], rdx
    inc r14
    jmp .Ltok_loop

.Lend_tokens:
    cmp r14, 1
    jne .Lerror
    mov rax, [r10]
    call print_value
    ret

.Lerror:
    lea rsi, [rip + err_msg]
    mov edx, 6
    mov edi, 1
    call write_all
    ret

print_value:
    xor r8d, r8d
    test rax, rax
    jns .Lpv_pos
    mov r8d, 1
    neg rax
.Lpv_pos:
    lea rbx, [rip + out_buf + 32]
    mov byte ptr [rbx-1], 10
    lea rcx, [rbx-1]
    mov r9, 10
.Lpv_digit_loop:
    xor edx, edx
    div r9
    dec rcx
    add dl, 48
    mov [rcx], dl
    test rax, rax
    jnz .Lpv_digit_loop
    test r8d, r8d
    jz .Lpv_write
    dec rcx
    mov byte ptr [rcx], 45
.Lpv_write:
    mov rsi, rcx
    mov rdx, rbx
    sub rdx, rcx
    mov edi, 1
    call write_all
    ret

write_all:
    test rdx, rdx
    jz .Lwa_done
.Lwa_loop:
    mov eax, 1
    syscall
    test rax, rax
    jle .Lwa_done
    add rsi, rax
    sub rdx, rax
    jnz .Lwa_loop
.Lwa_done:
    ret

.section .rodata
err_msg:
    .ascii "error\n"

.bss
.align 8
input_buf:
    .zero 65536
val_stack:
    .zero 8192
out_buf:
    .zero 32
