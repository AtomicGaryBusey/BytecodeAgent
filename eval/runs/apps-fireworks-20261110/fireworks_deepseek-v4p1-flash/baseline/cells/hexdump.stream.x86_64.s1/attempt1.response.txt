.globl _start
.text
_start:
    lea r8, [rip + hex_table]
    xor r12, r12
.read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    cmp r12, 65536
    jb .read_loop
.read_done:
    xor r13, r13
    lea r14, [rip + inbuf]
    lea r15, [rip + linebuf]
.row_loop:
    cmp r13, r12
    jae .done_rows
    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jbe .n_ok
    mov eax, 16
.n_ok:
    mov rbx, rax
    mov rsi, r15
    mov eax, r13d
    mov ecx, 28
.offset_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    sub ecx, 4
    jns .offset_loop
    mov word ptr [rsi], 0x2020
    add rsi, 2
    lea r10, [r14 + r13]
    xor r9d, r9d
.hex_loop:
    cmp r9d, 16
    jae .hex_done
    cmp r9d, ebx
    jae .hex_pad
    movzx eax, byte ptr [r10 + r9]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    and eax, 0xF
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    jmp .hex_next
.hex_pad:
    mov byte ptr [rsi], ' '
    mov byte ptr [rsi+1], ' '
    mov byte ptr [rsi+2], ' '
    add rsi, 3
.hex_next:
    cmp r9d, 7
    jne .no_extra
    mov byte ptr [rsi], ' '
    inc rsi
.no_extra:
    inc r9d
    jmp .hex_loop
.hex_done:
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], '|'
    inc rsi
    xor r9d, r9d
.ascii_loop:
    cmp r9d, ebx
    jae .ascii_done
    movzx eax, byte ptr [r10 + r9]
    cmp al, 0x20
    jb .ascii_dot
    cmp al, 0x7E
    ja .ascii_dot
    mov byte ptr [rsi], al
    inc rsi
    jmp .ascii_next
.ascii_dot:
    mov byte ptr [rsi], '.'
    inc rsi
.ascii_next:
    inc r9d
    jmp .ascii_loop
.ascii_done:
    mov byte ptr [rsi], '|'
    inc rsi
    mov byte ptr [rsi], 10
    inc rsi
    mov rdx, rsi
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall
    add r13, 16
    jmp .row_loop
.done_rows:
    mov rsi, r15
    mov eax, r12d
    mov ecx, 28
.final_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    sub ecx, 4
    jns .final_loop
    mov byte ptr [rsi], 10
    inc rsi
    mov rdx, rsi
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
