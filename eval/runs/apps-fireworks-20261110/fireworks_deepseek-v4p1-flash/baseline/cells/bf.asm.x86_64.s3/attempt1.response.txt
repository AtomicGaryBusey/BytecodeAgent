.text
.globl _start
_start:
    # read stdin into inbuf
    xor r12, r12
    lea r13, [rip + inbuf]
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = total length
    xor rcx, rcx
find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [r13 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r14, rcx
    lea r15, [rcx + 1]
    mov rbx, r12
    sub rbx, r15
    jmp validate
no_bang:
    mov r14, r12
    mov r15, r12
    xor rbx, rbx
validate:
    xor r8, r8
    xor rcx, rcx
validate_loop:
    cmp rcx, r14
    jae validate_end
    mov al, byte ptr [r13 + rcx]
    cmp al, 0x5B
    je open_bracket
    cmp al, 0x5D
    je close_bracket
    jmp validate_next
open_bracket:
    cmp r8, 65536
    jae invalid
    lea rdx, [rip + bracket_stack]
    mov dword ptr [rdx + r8*4], ecx
    inc r8
    jmp validate_next
close_bracket:
    test r8, r8
    jz invalid
    dec r8
    lea rdx, [rip + bracket_stack]
    mov eax, dword ptr [rdx + r8*4]
    lea rdx, [rip + match_table]
    mov dword ptr [rdx + rax*4], ecx
    mov dword ptr [rdx + rcx*4], eax
validate_next:
    inc rcx
    jmp validate_loop
validate_end:
    test r8, r8
    jnz invalid
    jmp run
invalid:
    mov rax, 60
    mov rdi, 1
    syscall

run:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor rbp, rbp
    lea r12, [rip + tape]
run_loop:
    cmp r9, r14
    jae run_done
    movzx eax, byte ptr [r13 + r9]
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_dot
    cmp al, 0x2C
    je cmd_comma
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    jmp run_next
cmd_gt:
    inc r8
    jmp run_next
cmd_lt:
    dec r8
    jmp run_next
cmd_plus:
    inc byte ptr [r12 + r8]
    jmp run_next
cmd_minus:
    dec byte ptr [r12 + r8]
    jmp run_next
cmd_dot:
    mov al, byte ptr [r12 + r8]
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + rbp], al
    inc rbp
    jmp run_next
cmd_comma:
    cmp r10, rbx
    jae comma_zero
    lea rdx, [r15 + r10]
    mov al, byte ptr [r13 + rdx]
    mov byte ptr [r12 + r8], al
    inc r10
    jmp run_next
comma_zero:
    mov byte ptr [r12 + r8], 0
    jmp run_next
cmd_open:
    cmp byte ptr [r12 + r8], 0
    jne run_next
    lea rdx, [rip + match_table]
    mov eax, dword ptr [rdx + r9*4]
    lea r9, [rax + 1]
    jmp run_loop
cmd_close:
    cmp byte ptr [r12 + r8], 0
    je run_next
    lea rdx, [rip + match_table]
    mov eax, dword ptr [rdx + r9*4]
    lea r9, [rax + 1]
    jmp run_loop
run_next:
    inc r9
    jmp run_loop
run_done:
    mov rdx, rbp
    test rdx, rdx
    jz exit_prog
    lea rsi, [rip + outbuf]
    mov rdi, 1
flush_loop:
    mov rax, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jnz flush_loop
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.align 16
inbuf:      .zero 65536
tape:       .zero 30000
match_table: .zero 262144
bracket_stack: .zero 262144
outbuf:     .zero 65536
