.section .text

    add x20, x20, :lo12:inbuf
    mov x19, #0

    add x1, x20, x19
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

    mov x2, #0

    bge no_bang
    ldrb w3, [x20, x2]
    cmp w3, #0x21
    beq found_bang
    add x2, x2, #1
    b find_bang

    mov x22, x19
    mov x23, #0

    adrp x27, match
    add x27, x27, :lo12:match
    adrp x28, bstack
    add x28, x28, :lo12:bstack
    mov x25, #0
    mov x2, #0
match_loop:
    cmp x2, x21
    bge match_done
    ldrb w3, [x20, x2]
    cmp w3, #0x5B
    beq match_open
    cmp w3, #0x5D
    beq match_close
    add x2, x2, #1
    b match_loop
match_open:
    strh w2, [x28, x25, lsl #1]
    add x25, x25, #1
    add x2, x2, #1
    b match_loop
match_close:
    cbz x25, unbalanced
    sub x25, x25, #1
    ldrh w4, [x28, x25, lsl #1]
    strh w4, [x27, x2, lsl #1]
    strh w2, [x27, x4, lsl #1]
    add x2, x2, #1
    b match_loop
match_done:
    cbnz x25, unbalanced

    adrp x22, tape
    add x22, x22, :lo12:tape
    mov x29, x20
    mov x19, #0
    mov x20, #0
    mov x26, #0
run_loop:
    cmp x19, x21
    bge run_done
    ldrb w2, [x29, x19]
    cmp w2, #0x3E
    beq cmd_right
    cmp w2, #0x3C
    beq cmd_left
    cmp w2, #0x2B
    beq cmd_inc
    cmp w2, #0x2D
    beq cmd_dec
    cmp w2, #0x2E
    beq cmd_out
    cmp w2, #0x2C
    beq cmd_in
    cmp w2, #0x5B
    beq cmd_open
    cmp w2, #0x5D
    beq cmd_close
    add x19, x19, #1
    b run_loop

cmd_right:
    add x20, x20, #1
    add x19, x19, #1
    b run_loop

cmd_left:

    add x19, x19, #1
    b run_loop

cmd_inc:
    ldrb w3, [x22, x20]
    add w3, w3, #1
    strb w3, [x22, x20]
    add x19, x19, #1
    b run_loop

cmd_dec:
    ldrb w3, [x22, x20]
    sub w3, w3, #1
    strb w3, [x22, x20]
    add x19, x19, #1
    b run_loop

    add x1, x22, x20
    mov x2, #1
    mov x8, #64
    svc #0
    add x19, x19, #1

    bge cmd_in_zero
    ldrb w3, [x24, x26]
    add x26, x26, #1
    strb w3, [x22, x20]
    add x19, x19, #1
    b run_loop

    add x19, x19, #1
    b run_loop

    cbnz w3, cmd_open_skip
    ldrh w4, [x27, x19, lsl #1]
    add x19, x4, #1
    b run_loop
cmd_open_skip:

    b run_loop

cmd_close:

    cbz w3, cmd_close_skip
    ldrh w4, [x27, x19, lsl #1]
    add x19, x4, #1
    b run_loop
cmd_close_skip:
    add x19, x19, #1
    b run_loop

run_done:
    mov x0, #0
    mov x8, #93
    svc #0

unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0
