.section .bss
.align 4
inbuf:  .zero 65536
tape:   .zero 30000
jump:   .zero 131072
outbuf: .zero 65536
bstack: .zero 4000

.section .text
.globl _start
_start:
    // Read all of stdin into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cmp x2, #0
    b.le read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    // Find first '!'
    mov x1, #0
find_bang:
    cmp x1, x20
    b.ge no_bang
    ldrb w2, [x19, x1]
    cmp w2, #0x21
    b.eq found_bang
    add x1, x1, #1
    b find_bang
found_bang:
    mov x22, x1
    add x23, x1, #1
    mov x24, x20
    b done_bang
no_bang:
    mov x22, x20
    mov x23, x20
    mov x24, x20
done_bang:

    // Check bracket balance and build jump table
    mov x1, #0
    mov x3, #0
check_loop:
    cmp x1, x22
    b.ge check_done
    ldrb w4, [x19, x1]
    cmp w4, #0x5B
    b.eq open_bracket
    cmp w4, #0x5D
    b.eq close_bracket
    b next_check
open_bracket:
    adrp x5, bstack
    add x5, x5, :lo12:bstack
    str w1, [x5, x3, lsl #2]
    add x3, x3, #1
    b next_check
close_bracket:
    cmp x3, #0
    b.eq unbalanced
    sub x3, x3, #1
    adrp x5, bstack
    add x5, x5, :lo12:bstack
    ldr w6, [x5, x3, lsl #2]
    adrp x7, jump
    add x7, x7, :lo12:jump
    strh w1, [x7, x6, lsl #1]
    strh w6, [x7, x1, lsl #1]
    b next_check
next_check:
    add x1, x1, #1
    b check_loop
check_done:
    cmp x3, #0
    b.ne unbalanced

    // Run the program
    adrp x25, tape
    add x25, x25, :lo12:tape
    mov x26, #0
    mov x27, #0
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
    mov x29, #0

run_loop:
    cmp x27, x22
    b.ge run_done
    ldrb w4, [x19, x27]
    cmp w4, #0x3E
    b.eq cmd_gt
    cmp w4, #0x3C
    b.eq cmd_lt
    cmp w4, #0x2B
    b.eq cmd_plus
    cmp w4, #0x2D
    b.eq cmd_minus
    cmp w4, #0x2E
    b.eq cmd_dot
    cmp w4, #0x2C
    b.eq cmd_comma
    cmp w4, #0x5B
    b.eq cmd_open
    cmp w4, #0x5D
    b.eq cmd_close
    add x27, x27, #1
    b run_loop

cmd_gt:
    add x26, x26, #1
    add x27, x27, #1
    b run_loop
cmd_lt:
    sub x26, x26, #1
    add x27, x27, #1
    b run_loop
cmd_plus:
    ldrb w6, [x25, x26]
    add w6, w6, #1
    and w6, w6, #0xFF
    strb w6, [x25, x26]
    add x27, x27, #1
    b run_loop
cmd_minus:
    ldrb w6, [x25, x26]
    sub w6, w6, #1
    and w6, w6, #0xFF
    strb w6, [x25, x26]
    add x27, x27, #1
    b run_loop
cmd_dot:
    ldrb w6, [x25, x26]
    strb w6, [x28, x29]
    add x29, x29, #1
    add x27, x27, #1
    b run_loop
cmd_comma:
    cmp x23, x24
    b.ge comma_zero
    ldrb w6, [x19, x23]
    add x23, x23, #1
    b comma_store
comma_zero:
    mov w6, #0
comma_store:
    strb w6, [x25, x26]
    add x27, x27, #1
    b run_loop
cmd_open:
    ldrb w6, [x25, x26]
    cbnz w6, open_nonzero
    adrp x7, jump
    add x7, x7, :lo12:jump
    ldrh w8, [x7, x27, lsl #1]
    add x27, x8, #1
    b run_loop
open_nonzero:
    add x27, x27, #1
    b run_loop
cmd_close:
    ldrb w6, [x25, x26]
    cbz w6, close_zero
    adrp x7, jump
    add x7, x7, :lo12:jump
    ldrh w8, [x7, x27, lsl #1]
    add x27, x8, #1
    b run_loop
close_zero:
    add x27, x27, #1
    b run_loop

run_done:
    cmp x29, #0
    b.eq exit_ok
    mov x0, #1
    mov x1, x28
    mov x2, x29
    mov x8, #64
    svc #0
exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0
