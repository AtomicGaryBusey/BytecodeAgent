.section .bss
.align 16
input_buf:
    .zero 65536
out_buf:
    .zero 400000
stack:
    .zero 8192
digit_buf:
    .zero 32

.section .text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    lea r13, [rip + input_buf]
    lea r14, [rip + input_buf]
    add r14, r12
    lea r15, [rip + out_buf]
    test r12, r12
    jz exit_prog

line_loop:
    cmp r13, r14
    jae exit_prog
    mov rbx, r13
    mov rcx, r13
find_line_end:
    cmp rcx, r14
    jae line_end_found
    mov al, byte ptr [rcx]
    cmp al, 0x0A
    je line_end_found
    inc rcx
    jmp find_line_end
line_end_found:
    mov rdi, rcx
    mov rsi, rbx
    xor r8, r8
    lea r9, [rip + stack]

token_loop:
    cmp rsi, rdi
    jae check_end
    mov al, byte ptr [rsi]
    cmp al, ' '
    je skip_sep
    cmp al, 0x09
    je skip_sep
    jmp token_start
skip_sep:
    inc rsi
    jmp token_loop

token_start:
    mov r10, rsi
find_token_end:
    inc rsi
    cmp rsi, rdi
    jae token_end_found
    mov al, byte ptr [rsi]
    cmp al, ' '
    je token_end_found
    cmp al, 0x09
    je token_end_found
    jmp find_token_end
token_end_found:
    mov r11, rsi
    sub r11, r10

    cmp r11, 1
    jne check_multi
    mov al, byte ptr [r10]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    movzx eax, byte ptr [r10]
    sub eax, '0'
    mov qword ptr [r9 + r8*8], rax
    inc r8
    jmp token_loop

check_multi:
    mov al, byte ptr [r10]
    cmp al, '-'
    je check_negative
    cmp al, '0'
    jb invalid_token
    cmp al, '9'
    ja invalid_token
    cmp r11, 18
    ja invalid_token
    xor eax, eax
    mov rcx, r10
    mov rdx, r11
parse_pos_loop:
    movzx ebx, byte ptr [rcx]
    cmp bl, '0'
    jb invalid_token
    cmp bl, '9'
    ja invalid_token
    imul rax, rax, 10
    sub bl, '0'
    add rax, rbx
    inc rcx
    dec rdx
    jnz parse_pos_loop
    mov qword ptr [r9 + r8*8], rax
    inc r8
    jmp token_loop

check_negative:
    dec r11
    cmp r11, 1
    jb invalid_token
    cmp r11, 18
    ja invalid_token
    lea rcx, [r10 + 1]
    xor eax, eax
    mov rdx, r11
parse_neg_loop:
    movzx ebx, byte ptr [rcx]
    cmp bl, '0'
    jb invalid_token
    cmp bl, '9'
    ja invalid_token
    imul rax, rax, 10
    sub bl, '0'
    add rax, rbx
    inc rcx
    dec rdx
    jnz parse_neg_loop
    neg rax
    mov qword ptr [r9 + r8*8], rax
    inc r8
    jmp token_loop

op_add:
    cmp r8, 2
    jb invalid_token
    dec r8
    mov rax, qword ptr [r9 + r8*8]
    dec r8
    add qword ptr [r9 + r8*8], rax
    inc r8
    jmp token_loop

op_sub:
    cmp r8, 2
    jb invalid_token
    dec r8
    mov rax, qword ptr [r9 + r8*8]
    dec r8
    sub qword ptr [r9 + r8*8], rax
    inc r8
    jmp token_loop

op_mul:
    cmp r8, 2
    jb invalid_token
    dec r8
    mov rax, qword ptr [r9 + r8*8]
    dec r8
    mov rdx, qword ptr [r9 + r8*8]
    imul rdx, rax
    mov qword ptr [r9 + r8*8], rdx
    inc r8
    jmp token_loop

op_div:
    cmp r8, 2
    jb invalid_token
    dec r8
    mov rbx, qword ptr [r9 + r8*8]
    dec r8
    mov rax, qword ptr [r9 + r8*8]
    test rbx, rbx
    jz invalid_token
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne do_div
    cmp rbx, -1
    je invalid_token
do_div:
    cqo
    idiv rbx
    mov qword ptr [r9 + r8*8], rax
    inc r8
    jmp token_loop

op_mod:
    cmp r8, 2
    jb invalid_token
    dec r8
    mov rbx, qword ptr [r9 + r8*8]
    dec r8
    mov rax, qword ptr [r9 + r8*8]
    test rbx, rbx
    jz invalid_token
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne do_mod
    cmp rbx, -1
    je invalid_token
do_mod:
    cqo
    idiv rbx
    mov qword ptr [r9 + r8*8], rdx
    inc r8
    jmp token_loop

invalid_token:
    mov rsi, rdi
    mov byte ptr [r15], 'e'
    mov byte ptr [r15+1], 'r'
    mov byte ptr [r15+2], 'r'
    mov byte ptr [r15+3], 'o'
    mov byte ptr [r15+4], 'r'
    mov byte ptr [r15+5], 0x0A
    add r15, 6
    jmp next_line

check_end:
    cmp r8, 1
    jne invalid_token
    mov rax, qword ptr [r9]
    test rax, rax
    jns positive
    mov byte ptr [r15], '-'
    inc r15
    neg rax
positive:
    lea rbx, [rip + digit_buf + 31]
    mov rcx, 10
convert_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz convert_loop
    lea rsi, [rip + digit_buf + 31]
    sub rsi, rbx
copy_loop:
    mov al, byte ptr [rbx]
    mov byte ptr [r15], al
    inc rbx
    inc r15
    dec rsi
    jnz copy_loop
    mov byte ptr [r15], 0x0A
    inc r15
    jmp next_line

next_line:
    cmp rdi, r14
    jae exit_prog
    lea r13, [rdi + 1]
    jmp line_loop

exit_prog:
    lea rsi, [rip + out_buf]
    mov rdx, r15
    sub rdx, rsi
    test rdx, rdx
    jz exit_zero
write_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle exit_zero
    add rsi, rax
    sub rdx, rax
    jnz write_loop
exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
