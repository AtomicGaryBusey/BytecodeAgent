```asm
    .text
    .globl _start

_start:
    // Read all stdin into input_buf
    adrp x27, input_buf
    add x27, x27, :lo12:input_buf
    mov x19, #0          // total len = 0
read_loop:
    mov x0, #0           // fd = 0
    add x1, x27, x19     // buf = input_buf + len
    mov x2, #65536
    sub x2, x2, x19      // count = 65536 - len
    cbz x2, read_done
    mov x8, #63          // read
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop
read_done:
    // x19 = total input length

    // Find first '!'
    mov x20, x19         // prog_len = len initially
    mov x21, x19         // data_start = len (empty)
    mov x9, #0           // i = 0
    mov w10, #0x21       // '!'
find_bang:
    cmp x9, x19
    b.hs no_bang
    ldrb w11, [x27, x9]
    cmp w11, w10
    b.eq found_bang
    add x9, x9, #1
    b find_bang
found_bang:
    mov x20, x9          // prog_len = i
    add x21, x9, #1      // data_start = i+1
no_bang:
    // x20 = prog_len, x21 = data_start
    mov x22, x21         // data_pos = data_start

    // Bracket check and jump table
    mov x9, #0           // i = 0
    adrp x26, jumptab
    add x26, x26, :lo12:jumptab
    adrp x11, bracket_stack
    add x11, x11, :lo12:bracket_stack
    mov x12, #0          // sp_idx = 0
parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w13, [x27, x9]
    cmp w13, #'['
    b.eq push_bracket
    cmp w13, #']'
    b.eq pop_bracket
parse_next:
    add x9, x9, #1
    b parse_loop
push_bracket:
    str w9, [x11, x12, lsl #2]
    add x12, x12, #1
    b parse_next
pop_bracket:
    cbz x12, bracket_error
    sub x12, x12, #1
    ldr w14, [x11, x12, lsl #2]
    strh w9, [x26, x14, lsl #1]
    strh w14, [x26, x9, lsl #1]
    b parse_next
parse_done:
    cbnz x12, bracket_error

    // Execute
    adrp x23, tape
    add x23, x23, :lo12:tape
    mov x24, x23         // tape pointer
    adrp x28, out_buf
    add x28, x28, :lo12:out_buf
    mov x25, #0          // out_len = 0
    mov x23, #0          // pc = 0

exec_loop:
    cmp x23, x20
    b.hs exec_done
    ldrb w9, [x27, x23]
    cmp w9, #'>'
    b.eq cmd_right
    cmp w9, #'<'
    b.eq cmd_left
    cmp w9, #'+'
    b.eq cmd_inc
    cmp w9, #'-'
    b.eq cmd_dec
    cmp w9, #'.'
    b.eq cmd_out
    cmp w9, #','
    b.eq cmd_in
    cmp w9, #'['
    b.eq cmd_open
    cmp w9, #']'
    b.eq cmd_close
    // default: ignore
    add x23, x23, #1
    b exec_loop

cmd_right:
    add x24, x24, #1
    add x23, x23, #1
    b exec_loop

cmd_left:
    sub x24, x24, #1
    add x23, x23, #1
    b exec_loop

cmd_inc:
    ldrb w10, [x24]
    add w10, w10, #1
    strb w10, [x24]
    add x23, x23, #1
    b exec_loop

cmd_dec:
    ldrb w10, [x24]
    sub w10, w10, #1
    strb w10, [x24]
    add x23, x23, #1
    b exec_loop

cmd_out:
    ldrb w10, [x24]
    strb w10, [x28, x25]
    add x25, x25, #1
    add x23, x23, #1
    b exec_loop

cmd_in:
    cmp x22, x19
    b.hs in_zero
    ldrb w10, [x27, x22]
    add x22, x22, #1
    b in_store
in_zero:
    mov w10, #0
in_store:
    strb w10, [x24]
    add x23, x23, #1
    b exec_loop

cmd_open:
    ldrb w10, [x24]
    cbnz w10, open_skip
    ldrh w10, [x26, x23, lsl #1]
    add x23, x10, #1
    b exec_loop
open_skip:
    add x23, x23, #1
    b exec_loop

cmd_close:
    ldrb w10, [x24]
    cbz w10, close_skip
    ldrh w10, [x26, x23, lsl #1]
    add x23, x10, #1
    b exec_loop
close_skip:
    add x23, x23, #1
    b exec_loop

exec_done:
    cbz x25, exit_zero
    mov x0, #1
    mov x1, x28
    mov x2, x25
    mov x8, #64
    svc #0

exit_zero:
    mov x0, #0
    mov x8, #93
    svc #0

bracket_error:
    mov x0, #1
    mov x8, #93
    svc #0

    .bss
    .balign 4
input_buf:
    .zero 65536
    .balign 4
tape:
    .zero 30000
    .balign 4
jumptab:
    .zero 131072
    .balign 4
bracket_stack:
    .zero 4000
    .balign 4
out_buf:
    .zero 65536
```
