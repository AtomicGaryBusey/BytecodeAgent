.section .rodata
newline: .byte 10

.section .bss
.align 16
input_buf: .zero 32768
.align 16
lines: .zero 1024 * 16
.align 16
temp_key: .zero 16

.section .text
.globl _start
_start:
    # read input
    xor r12d, r12d          # total bytes read
.read_loop:
    mov eax, 0              # read
    mov edi, 0              # fd 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    cmp r12, 32768
    jb .read_loop
.read_done:

    # parse lines
    lea rbx, [rip + input_buf]      # scan pointer
    lea rbp, [rip + input_buf]
    add rbp, r12                    # end pointer
    xor r14d, r14d                  # line count
    lea r15, [rip + lines]          # entry pointer
    mov r8, rbx                     # line_start
.scan_loop:
    cmp rbx, rbp
    jae .scan_done
    mov al, byte ptr [rbx]
    cmp al, 10
    jne .not_newline
    # found newline
    mov r9, rbx
    sub r9, r8                      # length
    mov qword ptr [r15], r8
    mov qword ptr [r15+8], r9
    add r14, 1
    add r15, 16
    lea r8, [rbx+1]
.not_newline:
    add rbx, 1
    jmp .scan_loop
.scan_done:
    cmp r8, rbp
    jae .no_final
    mov r9, rbp
    sub r9, r8
    mov qword ptr [r15], r8
    mov qword ptr [r15+8], r9
    add r14, 1
    add r15, 16
.no_final:

    # insertion sort
    lea rbx, [rip + lines]
    mov r12, 1                      # i = 1
.outer_loop:
    cmp r12, r14
    jae .sort_done
    # copy arr[i] to temp_key
    mov rax, r12
    shl rax, 4
    lea rsi, [rbx + rax]
    lea rdi, [rip + temp_key]
    mov rcx, [rsi]
    mov rdx, [rsi+8]
    mov qword ptr [rdi], rcx
    mov qword ptr [rdi+8], rdx
    # j = i-1
    mov r13, r12
    dec r13
.inner_loop:
    test r13, r13
    js .insert_key
    # compare arr[j] with temp_key
    mov rax, r13
    shl rax, 4
    lea rdi, [rbx + rax]
    lea rsi, [rip + temp_key]
    call compare_lines
    cmp eax, 0
    jle .insert_key
    # shift arr[j] to arr[j+1]
    mov rax, r13
    shl rax, 4
    lea rdi, [rbx + rax]
    lea rsi, [rdi + 16]
    mov rcx, [rdi]
    mov rdx, [rdi+8]
    mov qword ptr [rsi], rcx
    mov qword ptr [rsi+8], rdx
    dec r13
    jmp .inner_loop
.insert_key:
    mov rax, r13
    inc rax
    shl rax, 4
    lea rdi, [rbx + rax]
    lea rsi, [rip + temp_key]
    mov rcx, [rsi]
    mov rdx, [rsi+8]
    mov qword ptr [rdi], rcx
    mov qword ptr [rdi+8], rdx
    inc r12
    jmp .outer_loop
.sort_done:

    # output
    xor r12d, r12d
.output_loop:
    cmp r12, r14
    jae .exit
    mov rax, r12
    shl rax, 4
    lea rsi, [rbx + rax]        # entry
    mov rdi, 1                  # fd 1
    mov rdx, [rsi+8]            # len
    mov rsi, [rsi]              # ptr
    mov eax, 1                  # write
    syscall
    # write newline
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    syscall
    inc r12
    jmp .output_loop
.exit:
    mov eax, 60
    xor edi, edi
    syscall

# compare_lines: rdi = ptr to entry A, rsi = ptr to entry B
# returns eax: negative if A < B, 0 if equal, positive if A > B
compare_lines:
    mov r8, [rdi]        # ptr A
    mov r9, [rdi+8]      # len A
    mov r10, [rsi]       # ptr B
    mov r11, [rsi+8]     # len B
    # min_len = min(r9, r11)
    mov rcx, r9
    cmp rcx, r11
    cmovg rcx, r11       # rcx = min_len
    xor rax, rax         # index
.compare_loop:
    cmp rax, rcx
    jae .compare_lengths
    movzx edx, byte ptr [r8+rax]
    movzx esi, byte ptr [r10+rax]
    cmp edx, esi
    jne .different
    inc rax
    jmp .compare_loop
.different:
    mov eax, edx
    sub eax, esi
    ret
.compare_lengths:
    mov rax, r9
    sub rax, r11
    ret
