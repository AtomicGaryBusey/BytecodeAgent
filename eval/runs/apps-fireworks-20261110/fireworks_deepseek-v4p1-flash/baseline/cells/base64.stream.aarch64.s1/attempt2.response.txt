.globl _start

.equ INBUF_SIZE, 49152

.text
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #INBUF_SIZE
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf
    adrp x21, inbuf
    add x21, x21, :lo12:inbuf
    mov x22, x19
    mov w23, #0

encode_loop:
    cmp x22, #0
    b.eq encode_done

    cmp x22, #3
    b.lo tail

    ldrb w3, [x21], #1
    ldrb w4, [x21], #1
    ldrb w5, [x21], #1
    sub x22, x22, #3

    lsl w6, w3, #16
    lsl w4, w4, #8
    orr w6, w6, w4
    orr w6, w6, w5

    lsr w0, w6, #18
    and w0, w0, #0x3F
    bl emit_char

    lsr w0, w6, #12
    and w0, w0, #0x3F
    bl emit_char

    lsr w0, w6, #6
    and w0, w0, #0x3F
    bl emit_char

    and w0, w6, #0x3F
    bl emit_char

    b encode_loop

tail:
    cmp x22, #2
    b.eq tail2

    ldrb w3, [x21], #1
    lsl w6, w3, #16

    lsr w0, w6, #18
    and w0, w0, #0x3F
    bl emit_char

    lsr w0, w6, #12
    and w0, w0, #0x3F
    bl emit_char

    mov w0, #0x3D
    bl emit_byte

    mov w0, #0x3D
    bl emit_byte

    b encode_done

tail2:
    ldrb w3, [x21], #1
    ldrb w4, [x21], #1
    lsl w6, w3, #16
    lsl w4, w4, #8
    orr w6, w6, w4

    lsr w0, w6, #18
    and w0, w0, #0x3F
    bl emit_char

    lsr w0, w6, #12
    and w0, w0, #0x3F
    bl emit_char

    lsr w0, w6, #6
    and w0, w0, #0x3F
    bl emit_char

    mov w0, #0x3D
    bl emit_byte

    b encode_done

encode_done:
    cbz w23, write_output
    mov w1, #10
    strb w1, [x20], #1

write_output:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x20, x1
    cbz x2, exit

write_loop:
    mov x0, #1
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_loop

exit:
    mov x0, #0
    mov x8, #93
    svc #0

emit_char:
    adrp x2, b64_table
    add x2, x2, :lo12:b64_table
    ldrb w0, [x2, x0]
emit_byte:
    strb w0, [x20], #1
    add w23, w23, #1
    cmp w23, #76
    b.ne emit_done
    mov w1, #10
    strb w1, [x20], #1
    mov w23, #0
emit_done:
    ret
