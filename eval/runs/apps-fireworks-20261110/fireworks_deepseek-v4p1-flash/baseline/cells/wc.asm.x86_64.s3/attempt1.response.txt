.globl _start

.section .bss
.align 16
buffer:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    xor r12, r12
.Lread_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .Ldone_read
    add r12, rax
    jmp .Lread_loop
.Ldone_read:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor rcx, rcx
    lea rsi, [rip + buffer]
.Lscan:
    cmp rcx, r12
    jae .Lprint
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp eax, 10
    jne .Lnot_nl
    inc r8
.Lnot_nl:
    cmp eax, 32
    je .Lis_ws
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe .Lis_ws
    test r10, r10
    jnz .Lscan
    mov r10, 1
    inc r9
    jmp .Lscan
.Lis_ws:
    xor r10, r10
    jmp .Lscan
.Lprint:
    lea rdi, [rip + outbuf]
    add rdi, 64
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r12
    call .Lputnum
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r9
    call .Lputnum
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r8
    call .Lputnum
    lea rsi, [rip + outbuf]
    add rsi, 64
    mov rdx, rsi
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

.Lputnum:
    mov rcx, 10
1:
    xor rdx, rdx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret
