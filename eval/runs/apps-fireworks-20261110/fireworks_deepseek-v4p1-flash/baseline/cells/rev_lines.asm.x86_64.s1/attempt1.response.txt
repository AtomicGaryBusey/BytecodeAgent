.text
.globl _start
_start:
    lea r15, [rip + inbuf]
    xor r12, r12
.read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    cmp r12, 65536
    jb .read_loop
.read_done:
    xor rbx, rbx
.process_loop:
    cmp rbx, r12
    jae .done
    mov rcx, rbx
.find_nl:
    cmp rcx, r12
    jae .found_end
    cmp byte ptr [r15 + rcx], 10
    je .found_nl
    inc rcx
    jmp .find_nl
.found_nl:
    mov r13, rcx
    cmp rbx, r13
    je .rev1_done
    mov r8, rbx
    lea r9, [r13 - 1]
.reverse1:
    cmp r8, r9
    jae .rev1_done
    mov al, byte ptr [r15 + r8]
    mov dl, byte ptr [r15 + r9]
    mov byte ptr [r15 + r8], dl
    mov byte ptr [r15 + r9], al
    inc r8
    dec r9
    jmp .reverse1
.rev1_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + rbx]
    mov rdx, r13
    sub rdx, rbx
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    lea rbx, [r13 + 1]
    jmp .process_loop
.found_end:
    mov r13, r12
    mov r8, rbx
    lea r9, [r13 - 1]
.reverse2:
    cmp r8, r9
    jae .rev2_done
    mov al, byte ptr [r15 + r8]
    mov dl, byte ptr [r15 + r9]
    mov byte ptr [r15 + r8], dl
    mov byte ptr [r15 + r9], al
    inc r8
    dec r9
    jmp .reverse2
.rev2_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + rbx]
    mov rdx, r13
    sub rdx, rbx
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    mov rbx, r13
    jmp .process_loop
.done:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
.align 16
inbuf:
    .zero 65536
