.globl _start

.equ INPUT_BUF_SIZE, 65536
.equ LINE_BUF_ALLOC, 65552
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1

.text
_start:
    movz x0, #0x10, lsl #0
    movk x0, #0x1, lsl #16
    sub sp, sp, x0
    mov x19, sp

    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    mov x21, #0

read_loop:
    mov x0, #STDIN
    add x1, x20, x21
    movz x2, #0x1, lsl #16
    sub x2, x2, x21
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.eq read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0

process_loop:
    cmp x22, x21
    b.hs exit

    mov x23, x22
find_newline:
    cmp x23, x21
    b.hs process_line
    ldrb w24, [x20, x23]
    cmp w24, #10
    b.eq process_line
    add x23, x23, #1
    b find_newline

process_line:
    sub x24, x23, x22
    add x25, x20, x22
    mov x27, x19
    mov x28, x24

    cbz x24, 1f
    add x26, x25, x24
    sub x26, x26, #1

copy_loop:
    ldrb w29, [x26], #-1
    strb w29, [x27], #1
    subs x24, x24, #1
    b.ne copy_loop

1:
    mov w29, #10
    strb w29, [x27], #1
    add x28, x28, #1

    mov x29, x19
    mov x30, x28
write_loop:
    mov x0, #STDOUT
    mov x1, x29
    mov x2, x30
    mov x8, #SYS_WRITE
    svc #0
    add x29, x29, x0
    subs x30, x30, x0
    b.ne write_loop

    cmp x23, x21
    b.eq set_x22
    add x23, x23, #1
set_x22:
    mov x22, x23
    b process_loop

exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.bss
input_buffer: .space INPUT_BUF_SIZE
