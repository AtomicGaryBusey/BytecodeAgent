.globl _start

.equ stdin, 0
.equ stdout, 1
.equ sys_read, 63
.equ sys_write, 64
.equ sys_exit, 93

.bss
input_buffer: .zero 65536
number_buffer: .zero 20

.section .rodata
ok_msg: .ascii "ok\n"
error_msg: .ascii "error at "
newline: .ascii "\n"

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #stdin
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #sys_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, #65536
    b.lt read_loop

read_done:
    mov x21, #0
    mov x22, #-1
    mov x23, #0

    bl skip_whitespace
    cmp x22, #0
    b.ge output_error

    bl parse_value
    cmp x22, #0
    b.ge output_error

    bl skip_whitespace
    cmp x22, #0
    b.ge output_error

    cmp x21, x20
    b.eq parse_success
    mov x22, x21
    b output_error

skip_whitespace:
    cmp x21, x20
    b.ge skip_ws_done
    ldrb w4, [x19, x21]
    cmp w4, #0x20
    beq skip_ws_next
    cmp w4, #0x09
    beq skip_ws_next
    cmp w4, #0x0A
    beq skip_ws_next
    cmp w4, #0x0D
    beq skip_ws_next
    b skip_ws_done
skip_ws_next:
    add x21, x21, #1
    b skip_whitespace
skip_ws_done:
    ret

parse_value:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]

    cmp w4, #'{'
    beq parse_object
    cmp w4, #'['
    beq parse_array
    cmp w4, #'"'
    beq parse_string
    cmp w4, #'-'
    beq parse_number
    cmp w4, #'0'
    beq parse_number
    cmp w4, #'1'
    b.ge check_digit
    cmp w4, #'t'
    beq parse_true
    cmp w4, #'f'
    beq parse_false
    cmp w4, #'n'
    beq parse_null
    b parse_error

check_digit:
    cmp w4, #'9'
    b.le parse_number
    b parse_error

parse_error:
    mov x22, x21
    ret

parse_object:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'{'
    b.ne parse_error
    add x21, x21, #1

    add x23, x23, #1
    cmp x23, #1000
    b.hi parse_error

    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'}'
    beq parse_object_end

parse_object_member:
    bl parse_string
    cmp x22, #0
    b.ge parse_error

    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #':'
    b.ne parse_error
    add x21, x21, #1

    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error

    bl parse_value
    cmp x22, #0
    b.ge parse_error

    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #','
    beq parse_object_comma
    cmp w4, #'}'
    beq parse_object_end
    b parse_error

parse_object_comma:
    add x21, x21, #1
    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error
    b parse_object_member

parse_object_end:
    add x21, x21, #1
    sub x23, x23, #1
    ret

parse_array:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'['
    b.ne parse_error
    add x21, x21, #1

    add x23, x23, #1
    cmp x23, #1000
    b.hi parse_error

    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #']'
    beq parse_array_end

parse_array_value:
    bl parse_value
    cmp x22, #0
    b.ge parse_error

    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #','
    beq parse_array_comma
    cmp w4, #']'
    beq parse_array_end
    b parse_error

parse_array_comma:
    add x21, x21, #1
    bl skip_whitespace
    cmp x22, #0
    b.ge parse_error
    b parse_array_value

parse_array_end:
    add x21, x21, #1
    sub x23, x23, #1
    ret

parse_string:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'"'
    b.ne parse_error
    add x21, x21, #1

parse_string_loop:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]

    cmp w4, #'"'
    beq parse_string_end
    cmp w4, #'\\'
    beq parse_string_escape

    cmp w4, #0x20
    b.lt parse_error

    add x21, x21, #1
    b parse_string_loop

parse_string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]

    cmp w4, #'"'
    beq parse_string_escape_done
    cmp w4, #'\\'
    beq parse_string_escape_done
    cmp w4, #'/'
    beq parse_string_escape_done
    cmp w4, #'b'
    beq parse_string_escape_done
    cmp w4, #'f'
    beq parse_string_escape_done
    cmp w4, #'n'
    beq parse_string_escape_done
    cmp w4, #'r'
    beq parse_string_escape_done
    cmp w4, #'t'
    beq parse_string_escape_done
    cmp w4, #'u'
    beq parse_string_unicode
    b parse_error

parse_string_escape_done:
    add x21, x21, #1
    b parse_string_loop

parse_string_unicode:
    add x21, x21, #1
    mov x5, #4
parse_string_unicode_loop:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]

    cmp w4, #'0'
    b.lt parse_error
    cmp w4, #'9'
    b.le parse_string_unicode_hex
    cmp w4, #'a'
    b.lt parse_string_unicode_check_A
    cmp w4, #'f'
    b.le parse_string_unicode_hex
parse_string_unicode_check_A:
    cmp w4, #'A'
    b.lt parse_error
    cmp w4, #'F'
    b.le parse_string_unicode_hex
    b parse_error

parse_string_unicode_hex:
    add x21, x21, #1
    subs x5, x5, #1
    b.ne parse_string_unicode_loop
    b parse_string_loop

parse_string_end:
    add x21, x21, #1
    ret

parse_number:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'-'
    b.ne 1f
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_error
1:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.ne parse_number_nonzero

    add x21, x21, #1
    cmp x21, x20
    b.ge parse_number_done
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.lt parse_number_frac_exp
    cmp w4, #'9'
    b.le parse_error
    b parse_number_frac_exp

parse_number_nonzero:
    cmp w4, #'1'
    b.lt parse_error
    cmp w4, #'9'
    b.gt parse_error

parse_number_int_digits:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_number_frac_exp
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.lt parse_number_frac_exp
    cmp w4, #'9'
    b.le parse_number_int_digits

parse_number_frac_exp:
    cmp x21, x20
    b.ge parse_number_done
    ldrb w4, [x19, x21]
    cmp w4, #'.'
    beq parse_number_frac
    cmp w4, #'e'
    beq parse_number_exp
    cmp w4, #'E'
    beq parse_number_exp
    b parse_number_done

parse_number_frac:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.lt parse_error
    cmp w4, #'9'
    b.gt parse_error

parse_number_frac_digits:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_number_check_exp
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.lt parse_number_check_exp
    cmp w4, #'9'
    b.le parse_number_frac_digits

parse_number_check_exp:
    cmp x21, x20
    b.ge parse_number_done
    ldrb w4, [x19, x21]
    cmp w4, #'e'
    beq parse_number_exp
    cmp w4, #'E'
    beq parse_number_exp
    b parse_number_done

parse_number_exp:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'+'
    beq parse_number_exp_sign
    cmp w4, #'-'
    beq parse_number_exp_sign
    cmp w4, #'0'
    b.lt parse_error
    cmp w4, #'9'
    b.gt parse_error
    b parse_number_exp_digits

parse_number_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.lt parse_error
    cmp w4, #'9'
    b.gt parse_error

parse_number_exp_digits:
    add x21, x21, #1
    cmp x21, x20
    b.ge parse_number_done
    ldrb w4, [x19, x21]
    cmp w4, #'0'
    b.lt parse_number_done
    cmp w4, #'9'
    b.le parse_number_exp_digits

parse_number_done:
    ret

parse_true:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'t'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'r'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'u'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'e'
    b.ne parse_error
    add x21, x21, #1
    ret

parse_false:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'f'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'a'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'l'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'s'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'e'
    b.ne parse_error
    add x21, x21, #1
    ret

parse_null:
    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'n'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'u'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'l'
    b.ne parse_error
    add x21, x21, #1

    cmp x21, x20
    b.ge parse_error
    ldrb w4, [x19, x21]
    cmp w4, #'l'
    b.ne parse_error
    add x21, x21, #1
    ret

parse_success:
    mov x0, #stdout
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #sys_write
    svc #0
    mov x0, #0
    mov x8, #sys_exit
    svc #0

convert_number:
    adrp x5, number_buffer
    add x5, x5, :lo12:number_buffer
    mov x6, #0
    mov x7, #10

    cmp x4, #0
    beq convert_zero

convert_loop:
    udiv x8, x4, x7
    msub x9, x8, x7, x4
    add x9, x9, #'0'
    strb w9, [x5, x6]
    add x6, x6, #1
    mov x4, x8
    cbnz x4, convert_loop
    b convert_reverse

convert_zero:
    mov w9, #'0'
    strb w9, [x5]
    mov x6, #1

convert_reverse:
    mov x8, #0
    sub x7, x6, #1
convert_reverse_loop:
    cmp x8, x7
    b.ge convert_done
    ldrb w9, [x5, x8]
    ldrb w10, [x5, x7]
    strb w10, [x5, x8]
    strb w9, [x5, x7]
    add x8, x8, #1
    sub x7, x7, #1
    b convert_reverse_loop

convert_done:
    mov x0, x6
    ret

output_error:
    mov x0, #stdout
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #9
    mov x8, #sys_write
    svc #0

    mov x4, x22
    bl convert_number
    mov x2, x0
    mov x0, #stdout
    adrp x1, number_buffer
    add x1, x1, :lo12:number_buffer
    mov x8, #sys_write
    svc #0

    mov x0, #stdout
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #sys_write
    svc #0

    mov x0, #1
    mov x8, #sys_exit
    svc #0
