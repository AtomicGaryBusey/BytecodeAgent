```asm
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
input_buf:
    .zero MAX_INPUT
line_buf:
    .zero 80

.text
_start:
    xor r8, r8
    lea r10, [input_buf]

read_loop:
    cmp r8, MAX_INPUT
    jge read_done
    mov rdi, STDIN
    lea rsi, [r10 + r8]
    mov rdx, MAX_INPUT
    sub rdx, r8
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    xor r9, r9
    test r8, r8
    jz print_total

line_loop:
    mov r11, r8
    sub r11, r9
    jle line_loop_done

    mov r12, 16
    cmp r11, r12
    cmovg r11, r12

    lea r13, [line_buf]
    mov rsi, r13
    mov rax, r9
    mov rcx, 8
    mov r14, hex_digits
    mov r15, 28
write_offset_loop:
    mov r12, rax
    shr r12, r15
    and r12, 0xf
    mov al, byte ptr [r14 + r12]
    mov byte ptr [rsi], al
    inc rsi
    sub r15, 4
    loop write_offset_loop

    mov word ptr [rsi], 0x2020
    add rsi, 2

    lea rdi, [r10 + r9]
    xor rcx, rcx
hex_loop:
    cmp rcx, r11
    jae hex_space
    movzx eax, byte ptr [rdi + rcx]
    mov r12, rax
    shr r12, 4
    and r12, 0xf
    mov al, byte ptr [r14 + r12]
    mov byte ptr [rsi], al
    inc rsi
    and rax, 0xf
    mov al, byte ptr [r14 + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
    jmp hex_next

hex_space:
    mov word ptr [rsi], 0x2020
    mov byte ptr [rsi + 2], 0x20
    add rsi, 3

hex_next:
    inc rcx
    cmp rcx, 16
    je hex_done
    cmp rcx, 8
    jne hex_loop
    mov byte ptr [rsi], 0x20
    inc rsi
    jmp hex_loop

hex_done:
    mov word ptr [rsi], 0x207c
    add rsi, 2

    xor rcx, rcx
ascii_loop:
    cmp rcx, r11
    jge ascii_done
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7E
    ja ascii_dot
    mov byte ptr [rsi], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rsi], 0x2e
ascii_next:
    inc rsi
    inc rcx
    jmp ascii_loop

ascii_done:
    mov word ptr [rsi], 0x0a7c
    add rsi, 2

    mov r14, rsi
    mov rdi, STDOUT
    mov rsi, r13
    lea rdx, [r14 - r13]
    mov rax, SYS_WRITE
    syscall

    add r9, r11
    jmp line_loop

line_loop_done:
print_total:
    lea rsi, [line_buf]
    mov rax, r8
    mov rcx, 8
    mov r14, hex_digits
    mov r15, 28
write_total_loop:
    mov r12, rax
    shr r12, r15
    and r12, 0xf
    mov al, byte ptr [r14 + r12]
    mov byte ptr [rsi], al
    inc rsi
    sub r15, 4
    loop write_total_loop

    mov byte ptr [rsi], 0xa

    mov rdi, STDOUT
    lea rsi, [line_buf]
    mov rdx, 9
    mov rax, SYS_WRITE
    syscall

    xor rdi, rdi
    mov rax, SYS_EXIT
    syscall
```
