```asm
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
input_buf:
    .zero MAX_INPUT
line_buf:
    .zero 80

.text
_start:
    xor r8, r8
    lea r10, [input_buf]

read_loop:
    cmp r8, MAX_INPUT
    jge read_done
    mov rdi, STDIN
    lea rsi, [r10 + r8]
    mov rdx, MAX_INPUT
    sub rdx, r8
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    xor r9, r9
    test r8, r8
    jz print_total

line_loop:
    lea rsi, [line_buf]
    mov rax, r9
    call write_hex_8
    mov word ptr [rsi], 0x2020
    add rsi, 2

    lea rdi, [r10 + r9]
    mov r11, r8
    sub r11, r9
    cmp r11, 16
    cmovg r11, 16

    xor rcx, rcx
hex_loop:
    cmp rcx, 16
    jge hex_done
    cmp rcx, 8
    jne no_extra
    mov byte ptr [rsi], ' '
    inc rsi
no_extra:
    cmp rcx, r11
    jge write_pad
    movzx eax, byte ptr [rdi + rcx]
    call write_hex_byte
    mov byte ptr [rsi], ' '
    inc rsi
    jmp next_hex
write_pad:
    mov word ptr [rsi], 0x2020
    mov byte ptr [rsi + 2], ' '
    add rsi, 3
next_hex:
    inc rcx
    jmp hex_loop

hex_done:
    mov word ptr [rsi], 0x7c20
    add rsi, 2

    xor rcx, rcx
ascii_loop:
    cmp rcx, r11
    jge ascii_done
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7E
    jbe ok
dot:
    mov al, '.'
ok:
    mov byte ptr [rsi], al
    inc rsi
    inc rcx
    jmp ascii_loop

ascii_done:
    mov word ptr [rsi], 0x0a7c
    add rsi, 2

    mov rdx, rsi
    lea rsi, [line_buf]
    sub rdx, rsi
    mov rdi, STDOUT
    mov rax, SYS_WRITE
    syscall

    add r9, 16
    cmp r9, r8
    jb line_loop

print_total:
    lea rsi, [line_buf]
    mov rax, r8
    call write_hex_8
    mov byte ptr [rsi], 0xa
    mov rdi, STDOUT
    mov rdx, 9
    mov rax, SYS_WRITE
    syscall

    xor rdi, rdi
    mov rax, SYS_EXIT
    syscall

write_hex_8:
    push rcx
    push rdi
    lea rdi, [rsi + 8]
    mov rcx, 8
1:
    dec rdi
    mov dl, al
    and dl, 0xf
    mov dl, byte ptr [hex_digits + rdx]
    mov byte ptr [rdi], dl
    shr rax, 4
    loop 1b
    mov rsi, rdi
    pop rdi
    pop rcx
    ret

write_hex_byte:
    push rax
    push rdx
    mov ah, al
    shr al, 4
    and ah, 0xf
    lea rdx, [rip + hex_digits]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rsi], al
    mov al, ah
    mov al, byte ptr [rdx + rax]
    mov byte ptr [rsi + 1], al
    add rsi, 2
    pop rdx
    pop rax
    ret
```
