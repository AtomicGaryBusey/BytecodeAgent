.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536

_start:
    xor r8, r8
    lea r10, [rip + input_buf]

read_loop:
    cmp r8, MAX_INPUT
    jge read_done
    mov rdi, STDIN
    lea rsi, [r10 + r8]
    mov rdx, MAX_INPUT
    sub rdx, r8
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    js error_exit
    add r8, rax
    jmp read_loop

read_done:
    xor r9, r9

process_loop:
    cmp r9, r8
    jge process_done
    lea r12, [rip + line_buf]
    mov rax, r9
    mov rcx, 0
    lea r13, [rip + hex_chars]
write_offset_loop:
    mov r14d, 7
    sub r14d, ecx
    shl r14d, 2
    mov r15, rax
    shr r15, r14
    and r15, 0xF
    mov r15b, byte ptr [r13 + r15]
    mov byte ptr [r12 + rcx], r15b
    inc rcx
    cmp rcx, 8
    jb write_offset_loop
    add r12, 8
    mov word ptr [r12], 0x2020
    add r12, 2
    xor rcx, rcx
hex_byte_loop:
    cmp rcx, 16
    jge hex_byte_done
    mov r14, r9
    add r14, rcx
    cmp r14, r8
    jae print_spaces
    movzx eax, byte ptr [r10 + r14]
    mov r15, rax
    shr r15, 4
    and r15, 0xF
    mov r15b, byte ptr [r13 + r15]
    mov byte ptr [r12], r15b
    inc r12
    and eax, 0xF
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r12], al
    inc r12
    mov byte ptr [r12], 0x20
    inc r12
    jmp check_extra
print_spaces:
    mov byte ptr [r12], 0x20
    inc r12
    mov byte ptr [r12], 0x20
    inc r12
    mov byte ptr [r12], 0x20
    inc r12
check_extra:
    cmp rcx, 7
    jne next_hex
    mov byte ptr [r12], 0x20
    inc r12
next_hex:
    inc rcx
    jmp hex_byte_loop
hex_byte_done:
    mov byte ptr [r12], 0x20
    inc r12
    mov byte ptr [r12], '|'
    inc r12
    xor rcx, rcx
ascii_loop:
    cmp rcx, 16
    jge ascii_done
    mov r14, r9
    add r14, rcx
    cmp r14, r8
    jae ascii_done
    movzx eax, byte ptr [r10 + r14]
    cmp al, 0x20
    jb print_dot
    cmp al, 0x7E
    ja print_dot
    mov byte ptr [r12], al
    inc r12
    jmp next_ascii
print_dot:
    mov byte ptr [r12], '.'
    inc r12
next_ascii:
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r12], '|'
    inc r12
    mov byte ptr [r12], '\n'
    inc r12
    lea rsi, [rip + line_buf]
    mov rdx, r12
    sub rdx, rsi
    mov rdi, STDOUT
    mov rax, SYS_WRITE
    syscall
    add r9, 16
    jmp process_loop

process_done:
    lea r12, [rip + line_buf]
    mov rax, r8
    mov rcx, 0
    lea r13, [rip + hex_chars]
write_total_loop:
    mov r14d, 7
    sub r14d, ecx
    shl r14d, 2
    mov r15, rax
    shr r15, r14
    and r15, 0xF
    mov r15b, byte ptr [r13 + r15]
    mov byte ptr [r12 + rcx], r15b
    inc rcx
    cmp rcx, 8
    jb write_total_loop
    mov byte ptr [r12 + 8], '\n'
    mov rdi, STDOUT
    lea rsi, [rip + line_buf]
    mov rdx, 9
    mov rax, SYS_WRITE
    syscall
    xor rdi, rdi
    mov rax, SYS_EXIT
    syscall

error_exit:
    mov rdi, 1
    mov rax, SYS_EXIT
    syscall

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
input_buf:
    .zero MAX_INPUT
line_buf:
    .zero 80
