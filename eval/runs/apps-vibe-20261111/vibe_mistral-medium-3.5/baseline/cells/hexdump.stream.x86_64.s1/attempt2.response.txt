```asm
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
input_buf:
    .zero MAX_INPUT
line_buf:
    .zero 80

.text
_start:
    xor r8, r8
    lea r10, [input_buf]

read_loop:
    cmp r8, MAX_INPUT
    jge read_done
    mov rdi, STDIN
    lea rsi, [r10 + r8]
    mov rdx, MAX_INPUT
    sub rdx, r8
    mov rax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    js exit_error
    add r8, rax
    jmp read_loop

read_done:
    xor r9, r9
    test r8, r8
    jz print_total

line_loop:
    lea rsi, [line_buf]
    mov rax, r9
    call write_hex_8
    mov word ptr [rsi], '  '
    add rsi, 2
    xor rbx, rbx

byte_loop:
    mov rax, r9
    add rax, rbx
    cmp rax, r8
    jge byte_pad
    movzx rax, byte ptr [r10 + r9 + rbx]
    mov rdi, rax
    shr rdi, 4
    and edi, 0xf
    mov al, byte ptr [hex_digits + rdi]
    mov byte ptr [rsi], al
    inc rsi
    and rax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    jmp check_extra

byte_pad:
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi

check_extra:
    cmp rbx, 7
    jne no_extra
    mov byte ptr [rsi], ' '
    inc rsi
no_extra:
    inc rbx
    cmp rbx, 16
    jl byte_loop

    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], '|'
    inc rsi
    xor rbx, rbx

ascii_loop:
    mov rax, r9
    add rax, rbx
    cmp rax, r8
    jge ascii_done
    movzx rax, byte ptr [r10 + r9 + rbx]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7e
    jg ascii_dot
    mov byte ptr [rsi], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rsi], '.'
ascii_next:
    inc rsi
    inc rbx
    cmp rbx, 16
    jl ascii_loop

ascii_done:
    mov byte ptr [rsi], '|'
    inc rsi
    mov byte ptr [rsi], '\n'
    inc rsi
    lea rdx, [rsi - line_buf]
    mov rdi, STDOUT
    lea rsi, [line_buf]
    mov rax, SYS_WRITE
    syscall
    add r9, 16
    cmp r9, r8
    jle line_loop

print_total:
    lea rsi, [line_buf]
    mov rax, r8
    call write_hex_8
    mov byte ptr [rsi], '\n'
    inc rsi
    lea rdx, [rsi - line_buf]
    mov rdi, STDOUT
    lea rsi, [line_buf]
    mov rax, SYS_WRITE
    syscall
    xor rdi, rdi
    mov rax, SYS_EXIT
    syscall

exit_error:
    mov rdi, 1
    mov rax, SYS_EXIT
    syscall

write_hex_8:
    push rax
    push rbx
    mov rbx, rax
    mov rax, rbx
    shr rax, 28
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    shr rax, 24
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    shr rax, 20
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    shr rax, 16
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    shr rax, 12
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    shr rax, 8
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    shr rax, 4
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    mov rax, rbx
    and eax, 0xf
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rsi], al
    inc rsi
    pop rbx
    pop rax
    ret
```
