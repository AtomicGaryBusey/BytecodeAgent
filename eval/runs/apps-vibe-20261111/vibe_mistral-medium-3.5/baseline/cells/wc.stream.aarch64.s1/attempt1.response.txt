.globl _start

.text
_start:
  adrp x0, lines
  add x0, x0, :lo12:lines
  str xzr, [x0]
  adrp x0, words
  add x0, x0, :lo12:words
  str xzr, [x0]
  adrp x0, bytes
  add x0, x0, :lo12:bytes
  str xzr, [x0]
  adrp x0, in_word
  add x0, x0, :lo12:in_word
  strb wzr, [x0]

read_loop:
  mov x8, #63
  mov x0, #0
  adrp x1, buffer
  add x1, x1, :lo12:buffer
  mov x2, #65536
  svc #0
  cbz x0, read_done

  mov x3, x0
  mov x4, #0

  adrp x5, lines
  add x5, x5, :lo12:lines
  ldr x5, [x5]
  adrp x6, words
  add x6, x6, :lo12:words
  ldr x6, [x6]
  adrp x7, bytes
  add x7, x7, :lo12:bytes
  ldr x7, [x7]
  adrp x8, in_word
  add x8, x8, :lo12:in_word
  ldrb w8, [x8]

process_byte:
  cmp x4, x3
  b.ge update_counters
  ldrb w9, [x1, x4]

  add x7, x7, #1

  cmp w9, #0x0A
  b.ne not_newline
  add x5, x5, #1
not_newline:

  cmp w9, #0x09
  blo not_whitespace
  cmp w9, #0x0D
  bls is_whitespace
  cmp w9, #0x20
  bne not_whitespace
is_whitespace:
  mov w8, #0
  b next_byte
not_whitespace:
  cmp w8, #0
  b.ne next_byte
  mov w8, #1
  add x6, x6, #1
next_byte:
  add x4, x4, #1
  b process_byte

update_counters:
  adrp x9, lines
  add x9, x9, :lo12:lines
  str x5, [x9]
  adrp x9, words
  add x9, x9, :lo12:words
  str x6, [x9]
  adrp x9, bytes
  add x9, x9, :lo12:bytes
  str x7, [x9]
  adrp x9, in_word
  add x9, x9, :lo12:in_word
  strb w8, [x9]

  b read_loop

read_done:
  adrp x0, lines
  add x0, x0, :lo12:lines
  ldr x0, [x0]
  adrp x1, conv_buf
  add x1, x1, :lo12:conv_buf
  bl convert_number

  mov x2, x0
  mov x0, #1
  mov x8, #64
  svc #0

  adrp x0, space
  add x0, x0, :lo12:space
  mov x1, x0
  mov x2, #1
  mov x0, #1
  mov x8, #64
  svc #0

  adrp x0, words
  add x0, x0, :lo12:words
  ldr x0, [x0]
  adrp x1, conv_buf
  add x1, x1, :lo12:conv_buf
  bl convert_number

  mov x2, x0
  mov x0, #1
  mov x8, #64
  svc #0

  adrp x0, space
  add x0, x0, :lo12:space
  mov x1, x0
  mov x2, #1
  mov x0, #1
  mov x8, #64
  svc #0

  adrp x0, bytes
  add x0, x0, :lo12:bytes
  ldr x0, [x0]
  adrp x1, conv_buf
  add x1, x1, :lo12:conv_buf
  bl convert_number

  mov x2, x0
  mov x0, #1
  mov x8, #64
  svc #0

  adrp x0, newline
  add x0, x0, :lo12:newline
  mov x1, x0
  mov x2, #1
  mov x0, #1
  mov x8, #64
  svc #0

  mov x8, #93
  mov x0, #0
  svc #0

convert_number:
  mov x2, x1
  add x1, x1, #20
  mov x3, #0
  mov x4, #10

  cmp x0, #0
  beq .Lzero

.Lloop:
  udiv x5, x0, x4
  msub x6, x5, x4, x0
  add w6, w6, #'0'
  strb w6, [x1, #-1]!
  mov x0, x5
  add x3, x3, #1
  cmp x0, #0
  b.ne .Lloop

  mov x0, x3
  ret

.Lzero:
  mov w10, #'0'
  strb w10, [x2]
  mov x0, #1
  mov x1, x2
  ret

.section .rodata
space: .ascii " "
newline: .ascii "\n"

.bss
.align 3
buffer: .space 65536
lines: .space 8
words: .space 8
bytes: .space 8
in_word: .space 1
.align 3
conv_buf: .space 20
