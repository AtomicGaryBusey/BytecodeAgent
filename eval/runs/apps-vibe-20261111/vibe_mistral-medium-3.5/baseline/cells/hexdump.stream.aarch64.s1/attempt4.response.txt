```asm
.globl _start

.section .rodata
hex_digits: .ascii "0123456789abcdef"
space_char: .ascii " "
pipe_char: .ascii "|"
newline_char: .ascii "\n"
dot_char: .ascii "."

.bss
input_buf: .zero 65536
line_buf: .zero 80

.text
_start:
    adrp x19, hex_digits
    add x19, x19, :lo12:hex_digits
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x26, line_buf
    add x26, x26, :lo12:line_buf
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x20, x21
    mov x2, #65536
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0

line_loop:
    cmp x22, x21
    b.ge print_total
    sub x23, x21, x22
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, ls
    mov x25, x26
    mov x0, x22
    mov x1, x25
    bl print_hex_64
    adrp x2, space_char
    add x2, x2, :lo12:space_char
    ldrb w3, [x2]
    strb w3, [x25], #1
    strb w3, [x25], #1
    mov x4, #0
byte_pos_loop:
    cmp x4, x23
    b.lt print_byte_hex
    strb w3, [x25], #1
    strb w3, [x25], #1
    strb w3, [x25], #1
    b check_extra_space
print_byte_hex:
    add x5, x20, x22
    add x5, x5, x4
    ldrb w6, [x5]
    ubfx x7, x6, #4, #4
    ldrb w7, [x19, x7]
    strb w7, [x25], #1
    and x7, x6, #0xf
    ldrb w7, [x19, x7]
    strb w7, [x25], #1
    strb w3, [x25], #1
check_extra_space:
    cmp x4, #7
    b.ne next_byte_pos
    strb w3, [x25], #1
next_byte_pos:
    add x4, x4, #1
    cmp x4, #16
    b.lt byte_pos_loop
    strb w3, [x25], #1
    adrp x2, pipe_char
    add x2, x2, :lo12:pipe_char
    ldrb w2, [x2]
    strb w2, [x25], #1
    mov x4, #0
char_loop:
    cmp x4, x23
    b.ge print_closing_pipe
    add x5, x20, x22
    add x5, x5, x4
    ldrb w6, [x5]
    cmp w6, #0x20
    b.lt print_dot
    cmp w6, #0x7e
    b.gt print_dot
    strb w6, [x25], #1
    b next_char
print_dot:
    adrp x7, dot_char
    add x7, x7, :lo12:dot_char
    ldrb w7, [x7]
    strb w7, [x25], #1
next_char:
    add x4, x4, #1
    b char_loop
print_closing_pipe:
    adrp x2, pipe_char
    add x2, x2, :lo12:pipe_char
    ldrb w2, [x2]
    strb w2, [x25], #1
    adrp x2, newline_char
    add x2, x2, :lo12:newline_char
    ldrb w2, [x2]
    strb w2, [x25], #1
    sub x2, x25, x26
    mov x0, #1
    mov x1, x26
    mov x8, #64
    svc #0
    add x22, x22, x23
    b line_loop

print_total:
    mov x25, x26
    mov x0, x21
    mov x1, x25
    bl print_hex_64
    adrp x2, newline_char
    add x2, x2, :lo12:newline_char
    ldrb w2, [x2]
    strb w2, [x25], #1
    sub x2, x25, x26
    mov x0, #1
    mov x1, x26
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_hex_64:
    stp x0, x1, [sp, #-16]!
    stp x2, x3, [sp, #-16]!
    mov x2, #28
1:
    lsr x3, x0, x2
    and x3, x3, #0xf
    ldrb w3, [x19, x3]
    strb w3, [x1], #1
    sub x2, x2, #4
    cmp x2, #0
    b.ge 1b
    ldp x2, x3, [sp], #16
    ldp x0, x1, [sp], #16
    ret
```
