.globl _start

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.bss
input_buf: .zero 65536
line_buf: .zero 80

.text
_start:
    adrp x19, hex_digits
    add x19, x19, :lo12:hex_digits
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x20, x21
    mov x2, #65536
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0

line_loop:
    cmp x22, x21
    b.ge line_done
    sub x23, x21, x22
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, ls
    adrp x25, line_buf
    add x25, x25, :lo12:line_buf
    mov x26, x25
    mov x28, #8
offset_loop:
    sub x28, x28, #1
    lsl x29, x28, #2
    lsr x30, x22, x29
    and x30, x30, #0xf
    ldrb w30, [x19, x30]
    strb w30, [x25], #1
    cbnz x28, offset_loop
    mov x30, #0x20
    strb w30, [x25], #1
    strb w30, [x25], #1
    mov x28, #0
hex_loop1:
    cmp x28, #8
    b.eq hex_loop1_done
    cmp x28, x23
    b.lt byte_exists1
    mov x30, #0x20
    strb w30, [x25], #1
    strb w30, [x25], #1
    strb w30, [x25], #1
    b hex_loop1_next
byte_exists1:
    add x31, x20, x22
    add x31, x31, x28
    ldrb w30, [x31]
    lsr x31, x30, #4
    and x31, x31, #0xf
    ldrb w31, [x19, x31]
    strb w31, [x25], #1
    and x31, x30, #0xf
    ldrb w31, [x19, x31]
    strb w31, [x25], #1
    mov x31, #0x20
    strb w31, [x25], #1
hex_loop1_next:
    add x28, x28, #1
    b hex_loop1
hex_loop1_done:
    mov x30, #0x20
    strb w30, [x25], #1
    mov x28, #8
hex_loop2:
    cmp x28, #16
    b.eq hex_loop2_done
    cmp x28, x23
    b.lt byte_exists2
    mov x30, #0x20
    strb w30, [x25], #1
    strb w30, [x25], #1
    strb w30, [x25], #1
    b hex_loop2_next
byte_exists2:
    add x31, x20, x22
    add x31, x31, x28
    ldrb w30, [x31]
    lsr x31, x30, #4
    and x31, x31, #0xf
    ldrb w31, [x19, x31]
    strb w31, [x25], #1
    and x31, x30, #0xf
    ldrb w31, [x19, x31]
    strb w31, [x25], #1
    mov x31, #0x20
    strb w31, [x25], #1
hex_loop2_next:
    add x28, x28, #1
    b hex_loop2
hex_loop2_done:
    mov x30, #0x20
    strb w30, [x25], #1
    mov x30, #0x7c
    strb w30, [x25], #1
    mov x28, #0
ascii_loop:
    cmp x28, x23
    b.ge ascii_done
    add x31, x20, x22
    add x31, x31, x28
    ldrb w30, [x31]
    cmp x30, #0x20
    b.lt ascii_dot
    cmp x30, #0x7e
    b.gt ascii_dot
    strb w30, [x25], #1
    b ascii_next
ascii_dot:
    mov x30, #0x2e
    strb w30, [x25], #1
ascii_next:
    add x28, x28, #1
    b ascii_loop
ascii_done:
    mov x30, #0x7c
    strb w30, [x25], #1
    mov x30, #0x0a
    strb w30, [x25], #1
    mov x0, #1
    mov x1, x26
    sub x2, x25, x26
    mov x8, #64
    svc #0
    add x22, x22, #16
    b line_loop

line_done:
    adrp x25, line_buf
    add x25, x25, :lo12:line_buf
    mov x26, x21
    mov x28, #8
total_loop:
    sub x28, x28, #1
    lsl x29, x28, #2
    lsr x30, x26, x29
    and x30, x30, #0xf
    ldrb w30, [x19, x30]
    strb w30, [x25], #1
    cbnz x28, total_loop
    mov x30, #0x0a
    strb w30, [x25], #1
    adrp x0, line_buf
    add x0, x0, :lo12:line_buf
    mov x1, x0
    sub x2, x25, x0
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
