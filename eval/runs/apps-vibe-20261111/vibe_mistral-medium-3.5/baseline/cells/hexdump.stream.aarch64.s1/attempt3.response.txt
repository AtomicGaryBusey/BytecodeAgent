```asm
.globl _start

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.bss
input_buf: .zero 65536
line_buf: .zero 80

.text
_start:
    adrp x19, hex_digits
    add x19, x19, :lo12:hex_digits
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x20, x21
    mov x2, #65536
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0

line_loop:
    cmp x22, x21
    b.ge print_total
    sub x23, x21, x22
    cmp x23, #16
    csel x23, x23, #16, ls
    adrp x24, line_buf
    add x24, x24, :lo12:line_buf
    mov x25, x24
    mov x26, x22
    mov x27, #8

offset_loop:
    sub x27, x27, #1
    and x28, x26, #0xf
    lsr x26, x26, #4
    ldrb w28, [x19, x28]
    strb w28, [x24], #1
    cbnz x27, offset_loop

    mov x0, #0x20
    strb w0, [x24], #1
    strb w0, [x24], #1
    mov x27, #0

hex_loop:
    cmp x27, x23
    b.lt byte_exists
    mov x0, #0x20
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1
    b check_extra

byte_exists:
    add x0, x20, x22
    add x0, x0, x27
    ldrb w1, [x0]
    lsr x2, w1, #4
    and x2, x2, #0xf
    ldrb w2, [x19, x2]
    strb w2, [x24], #1
    and x2, w1, #0xf
    ldrb w2, [x19, x2]
    strb w2, [x24], #1
    mov x0, #0x20
    strb w0, [x24], #1

check_extra:
    cmp x27, #7
    b.ne hex_next
    mov x0, #0x20
    strb w0, [x24], #1

hex_next:
    add x27, x27, #1
    cmp x27, #16
    b.lt hex_loop

    mov x0, #0x20
    strb w0, [x24], #1
    mov x0, #0x7c
    strb w0, [x24], #1
    mov x27, #0

ascii_loop:
    cmp x27, x23
    b.ge ascii_done
    add x0, x20, x22
    add x0, x0, x27
    ldrb w1, [x0]
    cmp w1, #0x20
    b.lt ascii_dot
    cmp w1, #0x7e
    b.gt ascii_dot
    strb w1, [x24], #1
    b ascii_next

ascii_dot:
    mov x0, #0x2e
    strb w0, [x24], #1

ascii_next:
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov x0, #0x7c
    strb w0, [x24], #1
    mov x0, #0x0a
    strb w0, [x24], #1
    mov x0, #1
    mov x1, x25
    sub x2, x24, x25
    mov x8, #64
    svc #0
    add x22, x22, #16
    b line_loop

print_total:
    adrp x0, line_buf
    add x0, x0, :lo12:line_buf
    mov x1, x21
    mov x2, #8

total_loop:
    sub x2, x2, #1
    and x3, x1, #0xf
    lsr x1, x1, #4
    ldrb w3, [x19, x3]
    strb w3, [x0], #1
    cbnz x2, total_loop

    mov x3, #0x0a
    strb w3, [x0], #1
    mov x0, #1
    adrp x1, line_buf
    add x1, x1, :lo12:line_buf
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
```
