```assembly
.section .rodata
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg
error_msg: .ascii "error at "
error_msg_len = . - error_msg
newline: .ascii "\n"
newline_len = . - newline
true_str: .ascii "true"
false_str: .ascii "false"
null_str: .ascii "null"

.section .bss
input_buffer: .space 65536
input_len: .quad 0
pos: .quad 0
error: .byte 0
error_offset: .quad 0
num_buf: .space 20

.text
.globl _start
_start:
	xor rbx, rbx
.read_loop:
	mov rax, 0
	mov rdi, 0
	lea rsi, [input_buffer + rbx]
	mov rdx, 65536
	sub rdx, rbx
	syscall
	test rax, rax
	jz .read_done
	js .read_done
	add rbx, rax
	cmp rbx, 65536
	jl .read_loop
.read_done:
	mov [input_len], rbx

	xor rax, rax
	mov [pos], rax
	mov byte [error], 0

	call parse_document

	cmp byte [error], 0
	jne output_error

	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + ok_msg]
	mov rdx, ok_msg_len
	syscall

	mov rax, 60
	xor rdi, rdi
	syscall

output_error:
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + error_msg]
	mov rdx, error_msg_len
	syscall

	mov rax, [error_offset]
	lea rdi, [rip + num_buf + 19]
	mov byte [rdi], 0
	dec rdi
	mov rbx, 10
.convert_loop:
	xor rdx, rdx
	div rbx
	add dl, '0'
	mov [rdi], dl
	dec rdi
	test rax, rax
	jnz .convert_loop
	inc rdi

	lea rsi, [rip + num_buf + 19]
	sub rsi, rdi
	mov rdx, rsi
	mov rax, 1
	mov rdi, 1
	syscall

	lea rsi, [rip + newline]
	mov rdx, newline_len
	mov rax, 1
	mov rdi, 1
	syscall

	mov rax, 60
	mov rdi, 1
	syscall

skip_ws:
	mov rsi, [pos]
	mov rdi, input_buffer
	mov rcx, [input_len]
.skip_ws_loop:
	cmp rsi, rcx
	je .skip_ws_done
	mov al, [rdi + rsi]
	cmp al, ' '
	je .skip_ws_next
	cmp al, 9
	je .skip_ws_next
	cmp al, 10
	je .skip_ws_next
	cmp al, 13
	je .skip_ws_next
	jmp .skip_ws_done
.skip_ws_next:
	inc rsi
	jmp .skip_ws_loop
.skip_ws_done:
	mov [pos], rsi
	ret

parse_document:
	call skip_ws
	cmp byte [error], 0
	jne .ret
	call parse_value
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .ok
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.ok:
	ret

parse_value:
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	mov al, [input_buffer + rax]
	cmp al, '{'
	je parse_object
	cmp al, '['
	je parse_array
	cmp al, '"'
	je parse_string
	cmp al, 't'
	je parse_true
	cmp al, 'f'
	je parse_false
	cmp al, 'n'
	je parse_null
	cmp al, '-'
	je parse_number
	cmp al, '0'
	jl .invalid
	cmp al, '9'
	jle parse_number
.invalid:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_object:
	mov rax, [pos]
	cmp byte [input_buffer + rax], '{'
	jne .error
	inc qword [pos]
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], '}'
	je .empty
	call parse_members
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], '}'
	jne .error
	inc qword [pos]
	ret
.empty:
	inc qword [pos]
	ret
.error:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_members:
	call parse_member
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
.parse_members_loop:
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], ','
	jne .parse_members_done
	inc qword [pos]
	call skip_ws
	cmp byte [error], 0
	jne .ret
	call parse_member
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
	jmp .parse_members_loop
.parse_members_done:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret

parse_member:
	call parse_string
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], ':'
	jne .error
	inc qword [pos]
	call skip_ws
	cmp byte [error], 0
	jne .ret
	call parse_value
	cmp byte [error], 0
	jne .ret
	ret
.error:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_array:
	mov rax, [pos]
	cmp byte [input_buffer + rax], '['
	jne .error
	inc qword [pos]
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], ']'
	je .empty
	call parse_elements
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], ']'
	jne .error
	inc qword [pos]
	ret
.empty:
	inc qword [pos]
	ret
.error:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_elements:
	call parse_value
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
.parse_elements_loop:
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	cmp byte [input_buffer + rax], ','
	jne .parse_elements_done
	inc qword [pos]
	call skip_ws
	cmp byte [error], 0
	jne .ret
	call parse_value
	cmp byte [error], 0
	jne .ret
	call skip_ws
	cmp byte [error], 0
	jne .ret
	jmp .parse_elements_loop
.parse_elements_done:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret

parse_string:
	mov rax, [pos]
	cmp byte [input_buffer + rax], '"'
	jne .error
	inc qword [pos]
.parse_string_loop:
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	mov al, [input_buffer + rax]
	cmp al, '"'
	je .end_string
	cmp al, '\'
	je parse_escape
	cmp al, 0x20
	jl .control
	inc qword [pos]
	jmp .parse_string_loop
.end_string:
	inc qword [pos]
	ret
.error:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
	ret
.control:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_escape:
	mov rax, [pos]
	cmp rax, [input_len]
	je .incomplete
	inc rax
	cmp rax, [input_len]
	je .incomplete
	mov al, [input_buffer + rax]
	cmp al, '"'
	je .valid
	cmp al, '\'
	je .valid
	cmp al, '/'
	je .valid
	cmp al, 'b'
	je .valid
	cmp al, 'f'
	je .valid
	cmp al, 'n'
	je .valid
	cmp al, 'r'
	je .valid
	cmp al, 't'
	je .valid
	cmp al, 'u'
	je parse_u_escape
	jmp .invalid
.valid:
	inc rax
	mov [pos], rax
	jmp parse_string.parse_string_loop
.invalid:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_u_escape:
	inc rax
	mov rcx, 4
.check_hex:
	cmp rax, [input_len]
	je .incomplete
	mov al, [input_buffer + rax]
	call is_hex_digit
	jnc .invalid
	inc rax
	loop .check_hex
	mov [pos], rax
	jmp parse_string.parse_string_loop
.incomplete:
	mov byte [error], 1
	mov [error_offset], rax
.ret:
	ret
.invalid:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_true:
	mov rax, [pos]
	add rax, 4
	cmp rax, [input_len]
	jg .incomplete
	mov rsi, [pos]
	lea rdi, [rip + true_str]
	mov rcx, 4
	repe cmpsb
	jne .mismatch
	add [pos], 4
	mov rax, [pos]
	cmp rax, [input_len]
	je .ok
	mov al, [input_buffer + rax]
	call is_alphanum
	jc .extra
.ok:
	ret
.incomplete:
	mov byte [error], 1
	mov rax, [input_len]
	mov [error_offset], rax
.ret:
	ret
.mismatch:
	mov byte [error], 1
	mov [error_offset], rsi
	ret
.extra:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_false:
	mov rax, [pos]
	add rax, 5
	cmp rax, [input_len]
	jg .incomplete
	mov rsi, [pos]
	lea rdi, [rip + false_str]
	mov rcx, 5
	repe cmpsb
	jne .mismatch
	add [pos], 5
	mov rax, [pos]
	cmp rax, [input_len]
	je .ok
	mov al, [input_buffer + rax]
	call is_alphanum
	jc .extra
.ok:
	ret
.incomplete:
	mov byte [error], 1
	mov rax, [input_len]
	mov [error_offset], rax
.ret:
	ret
.mismatch:
	mov byte [error], 1
	mov [error_offset], rsi
	ret
.extra:
	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_null:
	mov rax, [pos]
	add rax, 4
	cmp rax, [input_len]
	jg .incomplete
	mov rsi, [pos]
	lea rdi, [rip + null_str]
	mov rcx, 4
	repe cmpsb
	jne .mismatch
	add [pos], 4
	mov rax, [pos]
	cmp rax, [input_len]
	je .ok
	mov al, [input_buffer + rax]
	call is_alphanum
	jc .extra
.ok:
	ret
.incomplete:
	mov byte [error], 1
	mov rax, [input_len]
	mov [error_offset], rax
.ret:
	ret
.mismatch:
	mov byte [error], 1
	mov [error_offset], rsi
	ret

	mov byte [error], 1
	mov [error_offset], rax
	ret

parse_number:
	mov rax, [pos]
	mov r8, rax
	cmp byte [input_buffer + rax], '-'
	jne .no_sign
	inc rax
.no_sign:
	cmp rax, [input_len]
	je .incomplete
	mov al, [input_buffer + rax]
	cmp al, '0'
	jne .digit19
	inc rax
	cmp rax, [input_len]
	je .after_int
	mov al, [input_buffer + rax]
	call is_digit
	jc .invalid_int
	jmp .after_int
.digit19:
	call is_digit_19
	jnc .invalid_int
	inc rax
1:
	cmp rax, [input_len]
	je .after_int
	mov al, [input_buffer + rax]
	call is_digit
	jnc .after_int
	inc rax
	jmp 1b
.after_int:
	mov [pos], rax
	cmp rax, [input_len]
	je .after_frac
	mov al, [input_buffer + rax]
	cmp al, '.'
	jne .after_frac
	inc rax
	cmp rax, [input_len]
	je .incomplete_frac
	mov al, [input_buffer + rax]
	call is_digit
	jnc .incomplete_frac
1:
	inc rax
	cmp rax, [input_len]
	je .after_frac
	mov al, [input_buffer + rax]
	call is_digit
	jnc .after_frac
	jmp 1b
.after_frac:
	mov [pos], rax
	cmp rax, [input_len]
	je .after_exp
	mov al, [input_buffer + rax]
	cmp al, 'e'
	je .exp
	cmp al, 'E'
	jne .after_exp
.exp:
	inc rax
	cmp rax, [input_len]
	je .incomplete_exp
	mov al, [input_buffer + rax]
	cmp al, '+'
	je .exp_sign
	cmp al, '-'
	jne .exp_digits
.exp_sign:
	inc rax
	cmp rax, [input_len]
	je .incomplete_exp
	mov al, [input_buffer + rax]
.exp_digits:
	call is_digit
	jnc .incomplete_exp
1:
	inc rax
	cmp rax, [input_len]
	je .after_exp
	mov al, [input_buffer + rax]
	call is_digit
	jnc .after_exp
	jmp 1b
.after_exp:
	mov [pos], rax
	cmp rax, [input_len]
	je .ok
	mov al, [input_buffer + rax]
	cmp al, '0'
	jl .ok
	cmp al, '9'
	jle .extra
	cmp al, '.'
	je .extra
	cmp al, 'e'
	je .extra
	cmp al, 'E'
	je .extra
	cmp al, '+'
	je .extra
	cmp al, '-'
	je .extra
.ok:
	ret
.incomplete:
	mov byte [error], 1
	mov [error_offset], r8
	cmp byte [input_buffer + r8], '-'
	je .incomplete_after_sign
	ret
.incomplete_after_sign:
	inc qword [error_offset]
	ret
.invalid_int:
	mov byte [error], 1
	mov [error_offset], rax
	ret
.incomplete_frac:
	mov byte [error], 1
	mov [error_offset], rax
	ret
.incomplete_exp:
	mov byte [error], 1
	mov [error_offset], rax
	ret
.extra:
	mov byte [error], 1
	mov [error_offset], rax
	ret

is_hex_digit:
	cmp al, '0'
	jl .no
	cmp al, '9'
	jle .yes
	cmp al, 'a'
	jl .no
	cmp al, 'f'
	jle .yes
	cmp al, 'A'
	jl .no
	cmp al, 'F'
	jle .yes
.no:
	clc
	ret
.yes:
	stc
	ret

is_digit:
	cmp al, '0'
	jl .no
	cmp al, '9'
	jle .yes
.no:
	clc
	ret
.yes:
	stc
	ret

is_digit_19:
	cmp al, '1'
	jl .no
	cmp al, '9'
	jle .yes
.no:
	clc
	ret
.yes:
	stc
	ret

is_alphanum:
	cmp al, 'a'
	jl .check_digit
	cmp al, 'z'
	jle .yes
	cmp al, 'A'
	jl .check_digit
	cmp al, 'Z'
	jle .yes
.check_digit:
	cmp al, '0'
	jl .no
	cmp al, '9'
	jle .yes
.no:
	clc
	ret
.yes:
	stc
	ret
```
