```asm
.section .bss
.buffer:
    .space 65536
.lines:
    .space 8
.words:
    .space 8
.bytes:
    .space 8
.in_word:
    .space 1
.output:
    .space 20

.section .text
.globl _start

_start:
    lea rsi, [.buffer]
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    syscall

    test rax, rax
    jz end_of_input

    mov rcx, rax
    lea rsi, [.buffer]
    add qword ptr [.bytes], rcx

process_byte:
    movzx r8, byte ptr [rsi]
    inc rsi
    dec rcx
    js process_byte_done

    cmp r8, 0x0A
    jne not_newline
    inc qword ptr [.lines]
not_newline:

    cmp r8, 0x09
    jb not_whitespace
    cmp r8, 0x0D
    jbe is_whitespace
    cmp r8, 0x20
    jne not_whitespace
is_whitespace:
    cmp byte ptr [.in_word], 1
    jne process_byte
    mov byte ptr [.in_word], 0
    jmp process_byte
not_whitespace:
    cmp byte ptr [.in_word], 0
    jne process_byte
    mov byte ptr [.in_word], 1
    inc qword ptr [.words]
    jmp process_byte

process_byte_done:
    jmp read_loop

end_of_input:
    lea r8, [.output]

    mov r9, qword ptr [.lines]
    call convert_and_write

    mov byte ptr [r8], ' '
    inc r8

    mov r9, qword ptr [.words]
    call convert_and_write

    mov byte ptr [r8], ' '
    inc r8

    mov r9, qword ptr [.bytes]
    call convert_and_write

    mov byte ptr [r8], '\n'
    inc r8

    lea rsi, [.output]
    mov rdx, r8
    sub rdx, rsi

    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

convert_and_write:
    sub rsp, 20
    mov rax, r9
    mov rcx, 10
    lea rdi, [rsp + 20]
    mov rsi, rdi

    test rax, rax
    jz zero_case

digit_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz digit_loop
    jmp copy

zero_case:
    dec rsi
    mov byte ptr [rsi], '0'

copy:
    mov rdi, rsi
    lea r10, [rsp + 20]
copy_loop:
    mov al, byte ptr [rdi]
    mov byte ptr [r8], al
    inc r8
    inc rdi
    cmp rdi, r10
    jne copy_loop

    add rsp, 20
    ret
```
