.section .bss
.align 8
input_buf: .space 32768
input_size: .quad 0
buffer_end: .quad 0
line_starts: .space 8000
line_count: .quad 0

.section .rodata
newline_char: .byte 0x0A

.section .text
.globl _start

_start:
    lea rdi, [rip + input_buf]
    mov rsi, 32768
    call read_all
    mov [rip + input_size], rax
    lea rcx, [rip + input_buf]
    add rcx, rax
    mov [rip + buffer_end], rcx
    test rax, rax
    jz exit_zero
    lea rdi, [rip + input_buf]
    mov rsi, rax
    call split_lines
    lea rdi, [rip + line_starts]
    mov rsi, [rip + line_count]
    test rsi, rsi
    jz exit_zero
    call sort_lines
    lea rdi, [rip + line_starts]
    mov rsi, [rip + line_count]
    call output_lines
exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall

read_all:
    push rbx
    push rbp
    push r12
    mov rbx, rdi
    mov rbp, rsi
    xor r12, r12
.read_loop:
    test rbp, rbp
    jz .read_done
    mov rdi, 0
    mov rsi, rbx
    mov rdx, rbp
    mov rax, 0
    syscall
    test rax, rax
    jz .read_done
    js .read_error
    add r12, rax
    add rbx, rax
    sub rbp, rax
    jmp .read_loop
.read_error:
    pop r12
    pop rbp
    pop rbx
    mov rax, 60
    mov rdi, 1
    syscall
.read_done:
    mov rax, r12
    pop r12
    pop rbp
    pop rbx
    ret

split_lines:
    push rbx
    push rbp
    push r12
    mov rbx, rdi
    mov rbp, rsi
    lea r12, [rbx + rbp]
    xor rax, rax
    mov rcx, rbx
    test rbp, rbp
    jz .split_done
.split_loop:
    cmp rbx, r12
    je .split_check_last
    movzx rdx, byte ptr [rbx]
    inc rbx
    cmp rdx, 0x0A
    jne .split_loop
    mov [rip + line_starts + rax*8], rcx
    inc rax
    mov rcx, rbx
    jmp .split_loop
.split_check_last:
    cmp rcx, r12
    je .split_done
    mov [rip + line_starts + rax*8], rcx
    inc rax
.split_done:
    mov [rip + line_count], rax
    pop r12
    pop rbp
    pop rbx
    ret

sort_lines:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov rbp, rsi
    dec rbp
    jz .sort_done
    mov r12, 0
.sort_outer:
    mov r13, 0
    mov r14, 0
.sort_inner:
    mov r15, [rbx + r13*8]
    mov rax, [rbx + r13*8 + 8]
    push r13
    push rbx
    call compare_lines
    pop rbx
    pop r13
    cmp rax, 1
    jne .no_swap
    mov rcx, [rbx + r13*8]
    mov rdx, [rbx + r13*8 + 8]
    mov [rbx + r13*8], rdx
    mov [rbx + r13*8 + 8], rcx
    mov r14, 1
.no_swap:
    inc r13
    cmp r13, rbp
    jb .sort_inner
    test r14, r14
    jz .sort_done
    inc r12
    cmp r12, rbp
    jb .sort_outer
.sort_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

compare_lines:
    push rbx
    push rbp
    push rsi
    push rdi
    mov rsi, r15
    mov rdi, rax
    mov rbx, [rip + buffer_end]
.compare_loop:
    cmp rsi, rbx
    je .line1_end
    movzx r8, byte ptr [rsi]
    cmp r8, 0x0A
    je .line1_end
    cmp rdi, rbx
    je .line2_end
    movzx r9, byte ptr [rdi]
    cmp r9, 0x0A
    je .line2_end
    cmp r8, r9
    jb .line1_less
    ja .line1_greater
    inc rsi
    inc rdi
    jmp .compare_loop
.line1_end:
    cmp rdi, rbx
    je .lines_equal
    movzx r9, byte ptr [rdi]
    cmp r9, 0x0A
    je .lines_equal
    mov rax, -1
    jmp .compare_done
.line2_end:
    cmp rsi, rbx
    je .lines_equal
    movzx r8, byte ptr [rsi]
    cmp r8, 0x0A
    je .lines_equal
    mov rax, 1
    jmp .compare_done
.line1_less:
    mov rax, -1
    jmp .compare_done
.line1_greater:
    mov rax, 1
.lines_equal:
    xor rax, rax
.compare_done:
    pop rdi
    pop rsi
    pop rbp
    pop rbx
    ret

output_lines:
    push rbx
    push rbp
    push r12
    push r13
    mov rbx, rdi
    mov rbp, rsi
    xor r12, r12
    mov r13, [rip + buffer_end]
.output_loop:
    cmp r12, rbp
    jge .output_done
    mov rsi, [rbx + r12*8]
    mov rdi, rsi
.find_end:
    cmp rdi, r13
    je .end_found
    movzx rax, byte ptr [rdi]
    cmp rax, 0x0A
    je .end_found
    inc rdi
    jmp .find_end
.end_found:
    sub rdi, rsi
    mov rdx, rdi
    mov rdi, 1
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r12
    jmp .output_loop
.output_done:
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
