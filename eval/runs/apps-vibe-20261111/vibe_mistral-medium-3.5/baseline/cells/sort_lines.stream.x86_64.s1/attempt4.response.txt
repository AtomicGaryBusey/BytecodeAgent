```asm
.globl _start

.text
_start:
    call read_all
    mov [rip + input_size], rax

    lea rsi, [rip + input_buf]
    lea rdi, [rip + input_buf]
    add rdi, rax
    lea rbx, [rip + line_ptrs]
    xor rcx, rcx
    mov r8, rsi

split_loop:
    cmp rsi, rdi
    jge split_done
    mov al, byte ptr [rsi]
    cmp al, 0x0A
    je found_newline
    inc rsi
    jmp split_loop

found_newline:
    mov [rbx + rcx*8], r8
    inc rcx
    inc rsi
    mov r8, rsi
    jmp split_loop

split_done:
    cmp r8, rdi
    jge no_last_line
    mov [rbx + rcx*8], r8
    inc rcx
no_last_line:
    mov [rip + line_count], rcx

    mov r12, rcx
    cmp r12, 1
    jle no_sort
    dec r12

    lea rbx, [rip + line_ptrs]

bubble_outer:
    xor rdi, rdi
    xor r9, r9
    mov r8, r12

bubble_inner:
    cmp rdi, r8
    jge bubble_check_swap

    mov r10, [rbx + rdi*8]
    mov r11, [rbx + rdi*8 + 8]
    lea rdx, [rip + input_buf]
    add rdx, [rip + input_size]

    push rdi
    push rsi
    push rdx
    mov rdi, r10
    mov rsi, r11
    call compare_lines
    pop rdx
    pop rsi
    pop rdi

    cmp rax, 0
    jle bubble_next

    mov rax, [rbx + rdi*8]
    mov rcx, [rbx + rdi*8 + 8]
    mov [rbx + rdi*8], rcx
    mov [rbx + rdi*8 + 8], rax
    mov r9, 1

bubble_next:
    inc rdi
    jmp bubble_inner

bubble_check_swap:
    test r9, r9
    jz bubble_done
    dec r8
    jmp bubble_outer

bubble_done:
no_sort:
    lea rbx, [rip + line_ptrs]
    xor rcx, rcx

output_loop:
    cmp rcx, [rip + line_count]
    jge output_done

    mov rsi, [rbx + rcx*8]
    lea rdi, [rip + input_buf]
    add rdi, [rip + input_size]

    mov r8, rsi

find_end:
    cmp r8, rdi
    jge found_end
    mov al, byte ptr [r8]
    cmp al, 0x0A
    je found_end
    inc r8
    jmp find_end

found_end:
    mov rdx, r8
    sub rdx, rsi

    push rax
    push rdi
    push rcx
    mov rax, 1
    mov rdi, 1
    syscall
    pop rcx
    pop rdi
    pop rax

    push rax
    push rdi
    push rsi
    push rcx
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    pop rcx
    pop rsi
    pop rdi
    pop rax

    inc rcx
    jmp output_loop

output_done:
    mov rax, 60
    xor rdi, rdi
    syscall

read_all:
    push rbx
    push rsi

    xor rbx, rbx

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 32768
    sub rdx, rbx
    jle read_done
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov rax, rbx
    pop rsi
    pop rbx
    ret

compare_lines:
    push rbx
    push rcx
    push r8
    push r9

    lea rdx, [rip + input_buf]
    add rdx, [rip + input_size]

compare_loop:
    cmp rdi, rdx
    je line1_ended
    mov al, byte ptr [rdi]
    cmp al, 0x0A
    je line1_ended

    cmp rsi, rdx
    je line2_ended
    mov bl, byte ptr [rsi]
    cmp bl, 0x0A
    je line2_ended

    cmp al, bl
    jb less
    ja greater

    inc rdi
    inc rsi
    jmp compare_loop

line1_ended:
    cmp rsi, rdx
    je equal
    mov bl, byte ptr [rsi]
    cmp bl, 0x0A
    je equal
    mov rax, -1
    jmp compare_done

line2_ended:
    cmp rdi, rdx
    je equal
    mov al, byte ptr [rdi]
    cmp al, 0x0A
    je equal
    mov rax, 1
    jmp compare_done

less:
    mov rax, -1
    jmp compare_done

greater:
    mov rax, 1
    jmp compare_done

equal:
    xor rax, rax

compare_done:
    pop r9
    pop r8
    pop rcx
    pop rbx
    ret

.section .rodata
newline: .byte 0x0A

.section .bss
.align 8
input_buf: .space 32768
line_ptrs: .space 8000
input_size: .space 8
line_count: .space 8
```
