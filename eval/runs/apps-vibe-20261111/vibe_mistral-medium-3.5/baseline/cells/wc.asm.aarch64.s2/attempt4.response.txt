.section .rodata
space: .ascii " "
newline: .ascii "\n"

.section .bss
.align 4
input_buf: .space 4096
output_buf: .space 64
lines: .space 4
words: .space 4
bytes: .space 4
in_word: .space 1

.section .text
.globl _start
_start:
    adrp x0, in_word
    add x0, x0, :lo12:in_word
    strb wzr, [x0]

read_loop:
    mov w8, #63
    mov w0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov w2, #4096
    svc #0

    cbz x0, done_reading

    mov x19, x0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

process_byte:
    ldrb w1, [x20], #1
    subs x19, x19, #1

    adrp x2, bytes
    add x2, x2, :lo12:bytes
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]

    cmp w1, #0x0A
    bne check_whitespace
    adrp x2, lines
    add x2, x2, :lo12:lines
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]

check_whitespace:
    cmp w1, #0x09
    beq is_whitespace
    cmp w1, #0x0A
    beq is_whitespace
    cmp w1, #0x0B
    beq is_whitespace
    cmp w1, #0x0C
    beq is_whitespace
    cmp w1, #0x0D
    beq is_whitespace
    cmp w1, #0x20
    bne not_whitespace

is_whitespace:
    adrp x2, in_word
    add x2, x2, :lo12:in_word
    strb wzr, [x2]
    cbnz x19, process_byte
    b read_loop

not_whitespace:
    adrp x2, in_word
    add x2, x2, :lo12:in_word
    ldrb w3, [x2]
    cbnz w3, process_byte
    adrp x2, words
    add x2, x2, :lo12:words
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]
    adrp x2, in_word
    add x2, x2, :lo12:in_word
    mov w3, #1
    strb w3, [x2]
    cbnz x19, process_byte
    b read_loop

done_reading:
    adrp x0, lines
    add x0, x0, :lo12:lines
    ldr w0, [x0]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    bl uint_to_string
    mov x2, x0
    mov w8, #64
    mov w0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    svc #0

    mov w8, #64
    mov w0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov w2, #1
    svc #0

    adrp x0, words
    add x0, x0, :lo12:words
    ldr w0, [x0]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    bl uint_to_string
    mov x2, x0
    mov w8, #64
    mov w0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    svc #0

    mov w8, #64
    mov w0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov w2, #1
    svc #0

    adrp x0, bytes
    add x0, x0, :lo12:bytes
    ldr w0, [x0]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    bl uint_to_string
    mov x2, x0
    mov w8, #64
    mov w0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    svc #0

    mov w8, #64
    mov w0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov w2, #1
    svc #0

    mov w8, #93
    mov w0, #0
    svc #0

uint_to_string:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, x20
    cbz x19, zero_case
    mov w22, #10

1:
    udiv w2, w19, w22
    mul w3, w2, w22
    sub w3, w19, w3
    add w3, w3, #'0'
    strb w3, [x21], #1
    mov w19, w2
    cbnz w19, 1b

    mov x22, x20
    sub x23, x21, #1
2:
    cmp x22, x23
    b.hs 3f
    ldrb w2, [x22]
    ldrb w3, [x23]
    strb w3, [x22], #1
    strb w2, [x23], #-1
    b 2b
3:
    sub x0, x21, x20
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

zero_case:
    mov w0, #'0'
    strb w0, [x20]
    mov x0, #1
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
