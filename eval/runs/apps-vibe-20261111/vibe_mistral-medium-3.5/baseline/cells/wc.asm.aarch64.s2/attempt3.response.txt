.section .rodata
space: .ascii " "
newline: .ascii "\n"

.section .bss
.align 4
input_buf: .space 4096
output_buf: .space 64
lines: .space 4
words: .space 4
bytes: .space 4
in_word: .space 1

.section .text
.globl _start
_start:
    // Initialize counters to 0 (bss is zero-initialized, but be safe)
    adrp x0, lines
    add x0, x0, :lo12:lines
    str wzr, [x0]
    adrp x0, words
    add x0, x0, :lo12:words
    str wzr, [x0]
    adrp x0, bytes
    add x0, x0, :lo12:bytes
    str wzr, [x0]
    adrp x0, in_word
    add x0, x0, :lo12:in_word
    strb wzr, [x0]

read_loop:
    mov w8, #63
    mov w0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov w2, #4096
    svc #0

    cbz x0, done_reading

    mov x19, x0                 // x19 = bytes read
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

process_byte:
    ldrb w1, [x20], #1
    sub x19, x19, #1

    // Increment bytes
    adrp x2, bytes
    add x2, x2, :lo12:bytes
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]

    // Check for newline (0x0A)
    cmp w1, #0x0A
    bne check_word_boundary
    adrp x2, lines
    add x2, x2, :lo12:lines
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]

check_word_boundary:
    // Check if whitespace: 0x20, 0x09, 0x0A, 0x0B, 0x0C, 0x0D
    cmp w1, #0x20
    beq is_whitespace
    cmp w1, #0x09
    beq is_whitespace
    cmp w1, #0x0A
    beq is_whitespace
    cmp w1, #0x0B
    beq is_whitespace
    cmp w1, #0x0C
    beq is_whitespace
    cmp w1, #0x0D
    bne not_whitespace
is_whitespace:
    adrp x2, in_word
    add x2, x2, :lo12:in_word
    strb wzr, [x2]
    cbnz x19, process_byte
    b read_loop

not_whitespace:
    adrp x2, in_word
    add x2, x2, :lo12:in_word
    ldrb w3, [x2]
    cbnz w3, process_byte
    // Entering a word - increment word count
    adrp x2, words
    add x2, x2, :lo12:words
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]
    adrp x2, in_word
    add x2, x2, :lo12:in_word
    mov w3, #1
    strb w3, [x2]
    cbnz x19, process_byte
    b read_loop

done_reading:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!

    // Print lines
    adrp x0, lines
    add x0, x0, :lo12:lines
    ldr w0, [x0]
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, #4096
    bl uint_to_string
    mov x22, x0
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    add x21, x21, #4096

    mov w8, #64
    mov w0, #1
    mov x1, x21
    mov x2, x22
    svc #0

    // Print space
    mov w8, #64
    mov w0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov w2, #1
    svc #0

    // Print words
    adrp x0, words
    add x0, x0, :lo12:words
    ldr w0, [x0]
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, #4096
    bl uint_to_string
    mov x22, x0
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    add x21, x21, #4096

    mov w8, #64
    mov w0, #1
    mov x1, x21
    mov x2, x22
    svc #0

    // Print space
    mov w8, #64
    mov w0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov w2, #1
    svc #0

    // Print bytes
    adrp x0, bytes
    add x0, x0, :lo12:bytes
    ldr w0, [x0]
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, #4096
    bl uint_to_string
    mov x22, x0
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    add x21, x21, #4096

    mov w8, #64
    mov w0, #1
    mov x1, x21
    mov x2, x22
    svc #0

    // Print newline
    mov w8, #64
    mov w0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov w2, #1
    svc #0

    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16

    mov w8, #93
    mov w0, #0
    svc #0

uint_to_string:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    mov x29, sp

    mov x19, x1          // buffer pointer
    mov x20, x0          // value

    cbz x20, zero_case

    mov x2, x19
    add x2, x2, #10
    mov w3, #10

1:
    udiv w4, w20, w3
    mul w5, w4, w3
    sub w5, w20, w5
    add w5, w5, #'0'
    strb w5, [x2, #-1]!
    mov w20, w4
    cbnz w20, 1b

    sub x0, x2, x19      // length
    add x3, x19, x0
2:
    ldrb w4, [x19], #1
    ldrb w5, [x3, #-1]!
    strb w4, [x3]
    strb w5, [x19, #-1]
    cmp x19, x3
    b.lt 2b

    add x0, x0, #1

    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

zero_case:
    mov w0, #'0'
    strb w0, [x19]
    mov x0, #1
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
