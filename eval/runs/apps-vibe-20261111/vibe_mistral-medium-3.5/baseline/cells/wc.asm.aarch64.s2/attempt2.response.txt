.section .rodata
space: .ascii " "
newline: .ascii "\n"

.section .bss
.align 4
input_buf: .space 4096
output_buf: .space 64
lines: .space 4
words: .space 4
bytes: .space 4
in_word: .space 1

.section .text
.globl _start
_start:
read_loop:
    mov w8, #63
    mov w0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov w2, #4096
    svc #0

    cmp x0, #0
    beq done_reading

    mov x20, x0
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf

process_loop:
    ldrb w21, [x19], #1
    sub x20, x20, #1

    adrp x0, bytes
    add x0, x0, :lo12:bytes
    ldr w1, [x0]
    add w1, w1, #1
    str w1, [x0]

    cmp w21, #0x0A
    bne not_newline
    adrp x0, lines
    add x0, x0, :lo12:lines
    ldr w1, [x0]
    add w1, w1, #1
    str w1, [x0]
not_newline:

    cmp w21, #0x20
    beq is_whitespace
    cmp w21, #0x09
    beq is_whitespace
    cmp w21, #0x0B
    beq is_whitespace
    cmp w21, #0x0C
    beq is_whitespace
    cmp w21, #0x0D
    beq is_whitespace
    cmp w21, #0x0A
    beq is_whitespace
    b not_whitespace

is_whitespace:
    adrp x0, in_word
    add x0, x0, :lo12:in_word
    ldrb w1, [x0]
    cbz w1, 1f
    strb wzr, [x0]
1:
    cbnz x20, process_loop
    b read_loop

not_whitespace:
    adrp x0, in_word
    add x0, x0, :lo12:in_word
    ldrb w1, [x0]
    cbnz w1, 1f
    adrp x2, words
    add x2, x2, :lo12:words
    ldr w3, [x2]
    add w3, w3, #1
    str w3, [x2]
    mov w1, #1
    strb w1, [x0]
1:
    cbnz x20, process_loop
    b read_loop

done_reading:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!

    adrp x0, lines
    add x0, x0, :lo12:lines
    ldr w0, [x0]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    bl uint_to_string
    mov x22, x0
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf

    mov w8, #64
    mov w0, #1
    mov x1, x21
    mov x2, x22
    svc #0

    mov w8, #64
    mov w0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov w2, #1
    svc #0

    adrp x0, words
    add x0, x0, :lo12:words
    ldr w0, [x0]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    bl uint_to_string
    mov x22, x0

    mov w8, #64
    mov w0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x2, x22
    svc #0

    mov w8, #64
    mov w0, #1
    adrp x1, space
    add x1, x1, :lo12:space
    mov w2, #1
    svc #0

    adrp x0, bytes
    add x0, x0, :lo12:bytes
    ldr w0, [x0]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    bl uint_to_string
    mov x22, x0

    mov w8, #64
    mov w0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x2, x22
    svc #0

    mov w8, #64
    mov w0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov w2, #1
    svc #0

    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16

    mov w8, #93
    mov w0, #0
    svc #0

uint_to_string:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    cmp w0, #0
    bne 1f
    mov w1, #'0'
    strb w1, [x1]
    mov x0, #1
    b 9f

1:
    mov x2, x1
    add x1, x1, #10
    mov w3, #10
    mov w4, w0

2:
    udiv w5, w4, w3
    mul w6, w5, w3
    sub w6, w4, w6
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    mov w4, w5
    cbnz w4, 2b

    add x3, x2, #10
    sub x7, x3, x1

    mov x0, x2
    mov x5, x1
    mov x6, x7
3:
    ldrb w4, [x5], #1
    strb w4, [x0], #1
    subs x6, x6, #1
    b.ne 3b

    mov x0, x7

9:
    ldp x29, x30, [sp], #16
    ret
