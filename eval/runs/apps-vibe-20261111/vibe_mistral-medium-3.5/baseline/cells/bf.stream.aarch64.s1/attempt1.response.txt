// Constants for system calls and file descriptors
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 65536
.equ TAPE_SIZE, 30000
.equ MAX_NESTING, 1000

.bss
stdin_buf: .space MAX_INPUT
tape: .space TAPE_SIZE
output_buf: .space MAX_INPUT
jump_table: .space MAX_INPUT * 8
bracket_stack: .space MAX_NESTING * 8

.text
.globl _start
_start:
// Set up global addresses
adrp x19, stdin_buf
add x19, x19, :lo12:stdin_buf
adrp x13, tape
add x13, x13, :lo12:tape
adrp x15, output_buf
add x15, x15, :lo12:output_buf
adrp x16, jump_table
add x16, x16, :lo12:jump_table
adrp x29, bracket_stack
add x29, x29, :lo12:bracket_stack

// Read entire stdin into stdin_buf
mov x20, #0
read_loop:
mov x0, #STDIN
add x1, x19, x20
mov x2, #MAX_INPUT
sub x2, x2, x20
mov x8, #SYS_read
svc #0
cmp x0, #0
beq read_done
add x20, x20, x0
cmp x20, #MAX_INPUT
bge read_done
b read_loop
read_done:

// Find first '!' in stdin_buf
mov x22, #0
find_excl_loop:
cmp x22, x20
bge no_excl
ldrb w23, [x19, x22]
cmp w23, #'!'
beq found_excl
add x22, x22, #1
b find_excl_loop
found_excl:
b after_find
no_excl:
mov x22, x20
after_find:

// Compute program_len (x24) and input_len (x25)
mov x24, x22
cmp x22, x20
beq no_input
sub x25, x20, x22
sub x25, x25, #1
b have_input
no_input:
mov x25, #0
have_input:

// Build jump table and validate bracket balance
mov x26, x29
mov x27, #0
jump_loop:
cmp x27, x24
bge jump_done
ldrb w30, [x19, x27]
cmp w30, #'['
beq jump_push
cmp w30, #']'
beq jump_pop
b jump_next
jump_push:
str x27, [x26], #8
b jump_next
jump_pop:
cmp x26, x29
beq unbalanced
sub x26, x26, #8
ldr x30, [x26]
str x27, [x16, x30, lsl #3]
str x30, [x16, x27, lsl #3]
b jump_next
jump_next:
add x27, x27, #1
b jump_loop
jump_done:
cmp x26, x29
bne unbalanced

// Set up input buffer start (after '!')
add x14, x19, x24
add x14, x14, #1

// Initialize interpreter state
mov x9, #0
mov x10, #0
mov x11, #0
mov x12, #0

// Interpreter main loop
interp_loop:
cmp x9, x24
bge interp_done
ldrb w17, [x19, x9]
add x9, x9, #1
cmp w17, #'>'
beq cmd_right
cmp w17, #'<'
beq cmd_left
cmp w17, #'+'
beq cmd_plus
cmp w17, #'-'
beq cmd_minus
cmp w17, #'.'
beq cmd_period
cmp w17, #','
beq cmd_comma
cmp w17, #'['
beq cmd_open
cmp w17, #']'
beq cmd_close
b interp_loop

cmd_right:
add x10, x10, #1
b interp_loop
cmd_left:
sub x10, x10, #1
b interp_loop
cmd_plus:
ldrb w18, [x13, x10]
add w18, w18, #1
strb w18, [x13, x10]
b interp_loop
cmd_minus:
ldrb w18, [x13, x10]
sub w18, w18, #1
strb w18, [x13, x10]
b interp_loop
cmd_period:
ldrb w18, [x13, x10]
strb w18, [x15, x12]
add x12, x12, #1
b interp_loop
cmd_comma:
cmp x11, x25
bge comma_eof
ldrb w18, [x14, x11]
add x11, x11, #1
strb w18, [x13, x10]
b interp_loop
comma_eof:
strb wzr, [x13, x10]
b interp_loop
cmd_open:
ldrb w18, [x13, x10]
cbnz w18, interp_loop
sub x17, x9, #1
ldr x18, [x16, x17, lsl #3]
add x9, x18, #1
b interp_loop
cmd_close:
ldrb w18, [x13, x10]
cbz w18, interp_loop
sub x17, x9, #1
ldr x18, [x16, x17, lsl #3]
add x9, x18, #1
b interp_loop

interp_done:
// Write output buffer
mov x0, #STDOUT
mov x1, x15
mov x2, x12
mov x8, #SYS_write
svc #0
// Exit with status 0
mov x0, #0
mov x8, #SYS_exit
svc #0

unbalanced:
// Exit with status 1
mov x0, #1
mov x8, #SYS_exit
svc #0
