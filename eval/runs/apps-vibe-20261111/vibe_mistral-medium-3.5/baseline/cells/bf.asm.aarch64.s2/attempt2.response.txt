.globl _start
_start:
  adrp x3, input_buf
  add x3, x3, :lo12:input_buf
  adrp x4, tape
  add x4, x4, :lo12:tape
  adrp x5, match_table
  add x5, x5, :lo12:match_table
  adrp x20, bracket_stack
  add x20, x20, :lo12:bracket_stack
  mov x15, x20

  mov x14, #0
read_loop:

  add x1, x3, x14
  mov x2, #65536
  sub x2, x2, x14

  mov x8, #63

  mov x17, x3
  mov x18, x14
find_loop:
  cmp x16, x18
  b.ge not_found
  ldrb w19, [x17, x16]
  cmp w19, #0x21
  beq found_excl
  add x16, x16, #1
  b find_loop

  add x7, x3, x14
  mov x9, #0

found_excl:

  add x7, x3, x16
  add x7, x7, #1
  sub x9, x14, x16
  sub x9, x9, #1
after_find:

  mov x16, #0
bracket_loop:
  cmp x16, x6
  b.ge bracket_loop_done
  ldrb w18, [x3, x16]
  cmp w18, #'['
  beq push_bracket
  cmp w18, #']'
  beq pop_bracket
  b next_bracket
push_bracket:
  str w16, [x15]
  add x15, x15, #4
  b next_bracket
pop_bracket:
  cmp x15, x20
  beq unbalanced
  sub x15, x15, #4
  ldr w19, [x15]
  str w19, [x5, x16, lsl #2]
  str w16, [x5, x19, lsl #2]
next_bracket:
  add x16, x16, #1
  b bracket_loop
bracket_loop_done:
  cmp x15, x20
  b.ne unbalanced

  mov x10, #0
  mov x11, #0
  mov x12, #0

exec_loop:
  cmp x10, x6
  b.ge exit_ok
  ldrb w13, [x3, x10]
  cmp w13, #'>'
  beq cmd_gt
  cmp w13, #'<'
  beq cmd_lt
  cmp w13, #'+'
  beq cmd_plus
  cmp w13, #'-'
  beq cmd_minus
  cmp w13, #'.'
  beq cmd_dot
  cmp w13, #','
  beq cmd_comma
  cmp w13, #'['
  beq cmd_open
  cmp w13, #']'
  beq cmd_close
  add x10, x10, #1
  b exec_loop

cmd_gt:
  add x11, x11, #1
  add x10, x10, #1
  b exec_loop

cmd_lt:
  sub x11, x11, #1
  add x10, x10, #1
  b exec_loop

cmd_plus:

  add w14, w14, #1
  and w14, w14, #0xFF
  strb w14, [x4, x11]
  add x10, x10, #1
  b exec_loop

cmd_minus:

  sub w14, w14, #1
  and w14, w14, #0xFF
  strb w14, [x4, x11]
  add x10, x10, #1
  b exec_loop

  mov x0, #1
  mov x2, #1
  mov x8, #64

  b.ge comma_zero

  strb w14, [x4, x11]

  b exec_loop

cmd_open:

  cmp w14, #0
  b.ne open_nonzero
  ldr w15, [x5, x10, lsl #2]

  b exec_loop

cmd_close:

  cmp w14, #0
  b.eq close_zero
  ldr w15, [x5, x10, lsl #2]
  add x10, x15, #1
  b exec_loop
close_zero:

  b exec_loop

  mov x8, #93
