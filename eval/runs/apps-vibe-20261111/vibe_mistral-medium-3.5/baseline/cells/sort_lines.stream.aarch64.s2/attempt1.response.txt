.globl _start

.section .rodata
newline: .byte 0x0A

.bss
input_buffer: .space 32768
total_bytes: .space 8
line_starts: .space 8000
line_lens: .space 4000
line_count: .space 4

.text
_start:
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    adrp x21, total_bytes
    add x21, x21, :lo12:total_bytes
    adrp x22, line_starts
    add x22, x22, :lo12:line_starts
    adrp x23, line_lens
    add x23, x23, :lo12:line_lens
    adrp x24, line_count
    add x24, x24, :lo12:line_count
    str xzr, [x21]

read_loop:
    mov w0, #0
    ldr x3, [x21]
    add x1, x20, x3
    mov x2, #32768
    sub x2, x2, x3
    cbz x2, read_done
    mov w8, #63
    svc #0
    cbz x0, read_done
    ldr x4, [x21]
    add x4, x4, x0
    str x4, [x21]
    b read_loop

read_done:
    ldr x3, [x21]
    mov x4, xzr
    mov x5, xzr
    str xzr, [x24]
    cbz x3, sort_lines

split_loop:
    cmp x4, x3
    beq check_last_line
    ldrb w6, [x20, x4]
    add x4, x4, #1
    cmp w6, #0x0A
    b.ne split_loop
    ldr x7, [x24]
    add x8, x20, x5
    str x8, [x22, x7, lsl #3]
    sub x8, x4, x5
    sub x8, x8, #1
    str w8, [x23, x7, lsl #2]
    add x7, x7, #1
    str x7, [x24]
    mov x5, x4
    b split_loop

check_last_line:
    cmp x5, x3
    b.hs sort_lines
    ldr x7, [x24]
    add x8, x20, x5
    str x8, [x22, x7, lsl #3]
    sub x8, x3, x5
    str w8, [x23, x7, lsl #2]
    add x7, x7, #1
    str x7, [x24]

sort_lines:
    ldr x3, [x24]
    cbz x3, print_lines
    mov x4, xzr

outer_loop:
    mov x5, xzr
    sub x6, x3, x4
    sub x6, x6, #1

inner_loop:
    cmp x5, x6
    b.hs outer_next
    ldr x7, [x22, x5, lsl #3]
    add x9, x5, #1
    ldr x8, [x22, x9, lsl #3]
    ldr w10, [x23, x5, lsl #2]
    ldr w11, [x23, x9, lsl #2]
    mov x12, xzr
    mov x13, xzr
    cmp w10, w11
    csel w14, w10, w11, ls

compare_loop:
    cmp x13, x14
    b.hs compare_done
    ldrb w15, [x7, x13]
    ldrb w16, [x8, x13]
    cmp w15, w16
    beq compare_next
    sub x12, x15, x16
    b compare_done
compare_next:
    add x13, x13, #1
    b compare_loop

compare_done:
    cbz x12, compare_by_length
    cmp x12, #0
    b.gt needs_swap
    b no_swap
compare_by_length:
    sub x12, x10, x11
    cmp x12, #0
    b.gt needs_swap
    b no_swap
needs_swap:
    ldr x14, [x22, x5, lsl #3]
    ldr x15, [x22, x9, lsl #3]
    str x14, [x22, x9, lsl #3]
    str x15, [x22, x5, lsl #3]
    ldr w14, [x23, x5, lsl #2]
    ldr w15, [x23, x9, lsl #2]
    str w14, [x23, x9, lsl #2]
    str w15, [x23, x5, lsl #2]
no_swap:
    add x5, x5, #1
    b inner_loop

outer_next:
    add x4, x4, #1
    cmp x4, x3
    b.lt outer_loop

print_lines:
    ldr x3, [x24]
    mov x4, xzr

print_loop:
    cmp x4, x3
    b.hs exit
    ldr x5, [x22, x4, lsl #3]
    ldr w6, [x23, x4, lsl #2]
    mov w0, #1
    mov x1, x5
    mov x2, x6
    mov w8, #64
    svc #0
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov w0, #1
    mov w8, #64
    svc #0
    add x4, x4, #1
    b print_loop

exit:
    mov w0, #0
    mov w8, #93
    svc #0
