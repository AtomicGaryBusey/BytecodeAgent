.globl _start

SYS_READ = 0
SYS_WRITE = 1
SYS_EXIT = 60
STDIN = 0
STDOUT = 1

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.bss
input_buf: .space 65536
line_buf: .space 80

.text
_start:
    xor r8, r8
    lea r9, [input_buf]

read_loop:
    mov rax, SYS_READ
    mov rdi, STDIN
    mov rsi, r9
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add r9, rax
    jmp read_loop

read_done:
    xor r10, r10

row_loop:
    cmp r10, r8
    jge print_total

    mov r11, r8
    sub r11, r10
    cmp r11, 16
    jle 1f
    mov r11, 16
1:
    lea rsi, [line_buf]
    mov r12, rsi

    mov rax, r10
    mov rdx, rax
    shr rdx, 28
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 24
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 20
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 16
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 12
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 8
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 4
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov word ptr [rsi], 0x2020
    add rsi, 2

    xor rcx, rcx
hex_byte_loop:
    cmp rcx, 16
    jge ascii_part

    cmp rcx, r11
    jae print_spaces

    movzx eax, byte ptr [input_buf + r10 + rcx]
    mov edx, eax
    shr edx, 4
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    mov edx, eax
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
    jmp check_extra

print_spaces:
    mov word ptr [rsi], 0x2020
    mov byte ptr [rsi + 2], 0x20
    add rsi, 3

check_extra:
    cmp rcx, 7
    jne next_hex
    mov byte ptr [rsi], 0x20
    inc rsi
next_hex:
    inc rcx
    jmp hex_byte_loop

ascii_part:
    mov word ptr [rsi], 0x7c20
    add rsi, 2

    xor rcx, rcx
ascii_loop:
    cmp rcx, r11
    jge end_ascii
    movzx eax, byte ptr [input_buf + r10 + rcx]
    cmp al, 0x20
    jb not_printable
    cmp al, 0x7e
    ja not_printable
    mov byte ptr [rsi], al
    jmp next_ascii
not_printable:
    mov byte ptr [rsi], 0x2e
next_ascii:
    inc rsi
    inc rcx
    jmp ascii_loop

end_ascii:
    mov word ptr [rsi], 0x0a7c
    add rsi, 2

    mov rdx, rsi
    sub rdx, r12
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    lea rsi, [line_buf]
    syscall

    add r10, 16
    jmp row_loop

print_total:
    mov rax, r8
    lea rsi, [line_buf]
    mov r12, rsi

    mov rdx, rax
    shr rdx, 28
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 24
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 20
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 16
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 12
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 8
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    shr rdx, 4
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov rdx, rax
    and edx, 0xf
    movzx edx, byte ptr [hex_digits + rdx]
    mov byte ptr [rsi], dl
    inc rsi

    mov byte ptr [rsi], 0x0a
    inc rsi

    mov rdx, rsi
    sub rdx, r12
    mov rax, SYS_WRITE
    mov rdi, STDOUT
    lea rsi, [line_buf]
    syscall

    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
