```assembly
.globl _start

.section .rodata
hex_digits: .asciz "0123456789abcdef"

.section .bss
input_buffer: .space 65536
line_buffer: .space 80

.section .text

format_hex_dword:
    mov   eax, edi
    mov   ecx, 28
    lea   r8, [rip + hex_digits]
.format_hex_dword_loop:
    mov   edx, eax
    shr   edx, cl
    and   edx, 0x0F
    mov   bl, [r8 + rdx]
    mov   [rsi], bl
    inc   rsi
    sub   cl, 4
    jns   .format_hex_dword_loop
    ret

format_hex_byte:
    lea   r8, [rip + hex_digits]
    mov   bl, al
    shr   bl, 4
    mov   bl, [r8 + rbx]
    mov   [rsi], bl
    inc   rsi
    and   al, 0x0F
    mov   bl, al
    mov   bl, [r8 + rbx]
    mov   [rsi], bl
    inc   rsi
    ret

_start:
    lea   r12, [rip + input_buffer]
    xor   r13, r13

read_loop:
    mov   rax, 0
    mov   rdi, 0
    lea   rsi, [r12 + r13]
    mov   rdx, 65536
    sub   rdx, r13
    syscall
    test  rax, rax
    jz    read_done
    add   r13, rax
    cmp   r13, 65536
    jb    read_loop
read_done:

    lea   r14, [rip + line_buffer]
    xor   r15, r15
process_line:
    cmp   r15, r13
    jge   print_total_length

    lea   rsi, [r14]
    mov   rdi, r15
    call  format_hex_dword

    mov   byte ptr [rsi], ' '
    mov   byte ptr [rsi + 1], ' '
    add   rsi, 2

    xor   rcx, rcx
print_hex_loop:
    mov   rax, r15
    add   rax, rcx
    cmp   rax, r13
    jae   print_hex_spaces

    movzx eax, byte ptr [r12 + rax]
    call  format_hex_byte

    mov   byte ptr [rsi], ' '
    inc   rsi
    jmp   check_extra_space

print_hex_spaces:
    mov   byte ptr [rsi], ' '
    mov   byte ptr [rsi + 1], ' '
    mov   byte ptr [rsi + 2], ' '
    add   rsi, 3

check_extra_space:
    cmp   rcx, 7
    jne   next_hex
    mov   byte ptr [rsi], ' '
    inc   rsi

next_hex:
    inc   rcx
    cmp   rcx, 16
    jb    print_hex_loop

    mov   byte ptr [rsi], ' '
    mov   byte ptr [rsi + 1], '|'
    add   rsi, 2

    xor   rcx, rcx
print_ascii_loop:
    mov   rax, r15
    add   rax, rcx
    cmp   rax, r13
    jae   ascii_done

    movzx eax, byte ptr [r12 + rax]

    cmp   al, 0x20
    jb    ascii_dot
    cmp   al, 0x7E
    ja    ascii_dot

    mov   [rsi], al
    inc   rsi
    jmp   next_ascii

ascii_dot:
    mov   byte ptr [rsi], '.'
    inc   rsi

next_ascii:
    inc   rcx
    cmp   rcx, 16
    jb    print_ascii_loop

ascii_done:
    mov   byte ptr [rsi], '|'
    inc   rsi
    mov   byte ptr [rsi], '\n'
    inc   rsi

    mov   rdx, rsi
    sub   rdx, r14
    mov   rax, 1
    mov   rdi, 1
    lea   rsi, [r14]
    syscall

    add   r15, 16
    jmp   process_line

print_total_length:
    lea   rsi, [r14]
    mov   rdi, r13
    call  format_hex_dword

    mov   byte ptr [rsi], '\n'
    inc   rsi

    mov   rdx, rsi
    sub   rdx, r14
    mov   rax, 1
    mov   rdi, 1
    lea   rsi, [r14]
    syscall

    mov   rax, 60
    xor   rdi, rdi
    syscall
```
