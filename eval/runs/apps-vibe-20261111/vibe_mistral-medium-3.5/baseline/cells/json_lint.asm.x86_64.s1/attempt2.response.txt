```asm
.globl _start

.section .rodata
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg
error_msg_prefix: .ascii "error at "
error_msg_prefix_len = . - error_msg_prefix
newline: .ascii "\n"

.section .bss
input_buffer: .space 65536
decimal_buf: .space 20

.text
_start:
    xor r8, r8
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buffer]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jle .read_done
    add r8, rax
    cmp r8, 65536
    jl .read_loop
.read_done:
    lea rsi, [input_buffer]
    lea rdi, [input_buffer + r8]
    call parse_document
    cmp rax, rdi
    je .success
    mov rdi, 1
    mov rax, 1
    lea rsi, [error_msg_prefix]
    mov rdx, error_msg_prefix_len
    syscall
    mov rsi, rax
    call print_decimal
    lea rsi, [newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall

.success:
    mov rdi, 1
    mov rax, 1
    lea rsi, [ok_msg]
    mov rdx, ok_msg_len
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

skip_ws:
    mov rcx, rsi
1:
    cmp rcx, rdi
    je 2f
    movzx eax, byte ptr [rcx]
    cmp al, ' '
    je 3f
    cmp al, '\t'
    je 3f
    cmp al, '\n'
    je 3f
    cmp al, '\r'
    je 3f
    jmp 2f
3:
    inc rcx
    jmp 1b
2:
    mov rax, rcx
    ret

parse_document:
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je 9f
    mov rsi, rax
    call parse_value
    cmp rax, r12
    je 9f
    mov rsi, rax
    call skip_ws
    cmp rax, rdi
    jne 9f
    mov rax, rdi
    ret
9:
    mov rax, r12
    ret

parse_value:
    cmp rsi, rdi
    jbe 9f
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je 1f
    cmp al, '['
    je 2f
    cmp al, '"'
    je 3f
    cmp al, 't'
    je 4f
    cmp al, 'f'
    je 5f
    cmp al, 'n'
    je 6f
    cmp al, '-'
    je 7f
    cmp al, '0'
    je 7f
    cmp al, '1'
    jb 9f
    cmp al, '9'
    jbe 7f
9:
    mov rax, rsi
    ret
1:
    call parse_object
    ret
2:
    call parse_array
    ret
3:
    call parse_string
    ret
4:
    call parse_true
    ret
5:
    call parse_false
    ret
6:
    call parse_null
    ret
7:
    call parse_number
    ret

parse_object:
    cmp rsi, rdi
    jbe 9f
    cmp byte ptr [rsi], '{'
    jne 9f
    mov rcx, rsi
    inc rcx
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je 9f
    cmp byte ptr [r12], '}'
    je 1f
2:
    mov rsi, r12
    call parse_string
    cmp rax, r12
    je 9f
    mov rsi, rax
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je 9f
    cmp byte ptr [r12], ':'
    jne 9f
    mov rcx, r12
    inc rcx
    call skip_ws
    mov r12, rax
    mov rsi, r12
    call parse_value
    cmp rax, r12
    je 9f
    mov rsi, rax
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je 9f
    cmp byte ptr [r12], '}'
    je 1f
    cmp byte ptr [r12], ','
    jne 9f
    mov rcx, r12
    inc rcx
    call skip_ws
    mov r12, rax
    jmp 2b
1:
    mov rcx, r12
    inc rcx
    call skip_ws
    mov rax, rcx
    ret
9:
    mov rax, r12
    ret

parse_array:
    cmp rsi, rdi
    jbe 9f
    cmp byte ptr [rsi], '['
    jne 9f
    mov rcx, rsi
    inc rcx
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je 9f
    cmp byte ptr [r12], ']'
    je 1f
2:
    mov rsi, r12
    call parse_value
    cmp rax, r12
    je 9f
    mov rsi, rax
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je 9f
    cmp byte ptr [r12], ']'
    je 1f
    cmp byte ptr [r12], ','
    jne 9f
    mov rcx, r12
    inc rcx
    call skip_ws
    mov r12, rax
    jmp 2b
1:
    mov rcx, r12
    inc rcx
    call skip_ws
    mov rax, rcx
    ret
9:
    mov rax, r12
    ret

parse_string:
    cmp rsi, rdi
    jbe 9f
    cmp byte ptr [rsi], '"'
    jne 9f
    mov rcx, rsi
    inc rcx
1:
    cmp rcx, rdi
    je 9f
    movzx eax, byte ptr [rcx]
    cmp al, '"'
    je 2f
    cmp al, '\'
    je 3f
    test al, al
    jz 9f
    cmp al, 0x1F
    jbe 9f
    inc rcx
    jmp 1b
3:
    inc rcx
    cmp rcx, rdi
    je 9f
    movzx eax, byte ptr [rcx]
    cmp al, '"'
    je 4f
    cmp al, '\'
    je 4f
    cmp al, '/'
    je 4f
    cmp al, 'b'
    je 4f
    cmp al, 'f'
    je 4f
    cmp al, 'n'
    je 4f
    cmp al, 'r'
    je 4f
    cmp al, 't'
    je 4f
    cmp al, 'u'
    jne 9f
    inc rcx
    mov r13, 4
5:
    test r13, r13
    jz 1b
    cmp rcx, rdi
    jbe 9f
    movzx eax, byte ptr [rcx]
    call is_hex_digit
    test rax, rax
    jz 9f
    inc rcx
    dec r13
    jmp 5b
4:
    inc rcx
    jmp 1b
2:
    inc rcx
    mov rax, rcx
    ret
9:
    mov rax, rcx
    ret

is_hex_digit:
    cmp al, '0'
    jb 1f
    cmp al, '9'
    jbe 2f
    cmp al, 'a'
    jb 1f
    cmp al, 'f'
    jbe 2f
    cmp al, 'A'
    jb 1f
    cmp al, 'F'
    jbe 2f
1:
    xor rax, rax
    ret
2:
    mov rax, 1
    ret

parse_number:
    cmp rsi, rdi
    jbe 9f
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne 1f
    mov rcx, rsi
    inc rcx
    jmp 2f
1:
    mov rcx, rsi
2:
    cmp rcx, rdi
    jbe 9f
    movzx eax, byte ptr [rcx]
    cmp al, '0'
    je 3f
    cmp al, '1'
    jb 9f
    cmp al, '9'
    jbe 4f
    jmp 9f
3:
    inc rcx
    cmp rcx, rdi
    jbe 5f
    movzx eax, byte ptr [rcx]
    call is_digit
    test rax, rax
    jnz 9f
5:
    jmp 6f
4:
    inc rcx
    call parse_digits
    mov rcx, rax
6:
    cmp rcx, rdi
    jbe 7f
    movzx eax, byte ptr [rcx]
    cmp al, '.'
    je 8f
    cmp al, 'e'
    je 10f
    cmp al, 'E'
    jne 7f
10f:
    inc rcx
    cmp rcx, rdi
    jbe 9f
    movzx eax, byte ptr [rcx]
    cmp al, '+'
    je 11f
    cmp al, '-'
    jne 12f
11f:
    inc rcx
12f:
    call parse_digits
    cmp rax, rcx
    je 9f
    mov rcx, rax
    jmp 7f
8f:
    inc rcx
    call parse_digits
    cmp rax, rcx
    je 9f
    mov rcx, rax
    jmp 6b
7f:
    mov rax, rcx
    ret
9f:
    mov rax, rcx
    ret

parse_digits:
1:
    cmp rcx, rdi
    jbe 2f
    movzx eax, byte ptr [rcx]
    call is_digit
    test rax, rax
    jz 2f
    inc rcx
    jmp 1b
2:
    mov rax, rcx
    ret

is_digit:
    cmp al, '0'
    jb 1f
    cmp al, '9'
    jbe 2f
1:
    xor rax, rax
    ret
2:
    mov rax, 1
    ret

parse_true:
    cmp rdi, rsi
    jbe 9f
    cmp byte ptr [rsi], 't'
    jne 9f
    cmp rdi, rsi + 1
    jbe 9f
    cmp byte ptr [rsi + 1], 'r'
    jne 9f
    cmp rdi, rsi + 2
    jbe 9f
    cmp byte ptr [rsi + 2], 'u'
    jne 9f
    cmp rdi, rsi + 3
    jbe 9f
    cmp byte ptr [rsi + 3], 'e'
    jne 9f
    lea rax, [rsi + 4]
    ret
9:
    mov rax, rsi
    ret

parse_false:
    cmp rdi, rsi
    jbe 9f
    cmp byte ptr [rsi], 'f'
    jne 9f
    cmp rdi, rsi + 1
    jbe 9f
    cmp byte ptr [rsi + 1], 'a'
    jne 9f
    cmp rdi, rsi + 2
    jbe 9f
    cmp byte ptr [rsi + 2], 'l'
    jne 9f
    cmp rdi, rsi + 3
    jbe 9f
    cmp byte ptr [rsi + 3], 's'
    jne 9f
    cmp rdi, rsi + 4
    jbe 9f
    cmp byte ptr [rsi + 4], 'e'
    jne 9f
    lea rax, [rsi + 5]
    ret
9:
    mov rax, rsi
    ret

parse_null:
    cmp rdi, rsi
    jbe 9f
    cmp byte ptr [rsi], 'n'
    jne 9f
    cmp rdi, rsi + 1
    jbe 9f
    cmp byte ptr [rsi + 1], 'u'
    jne 9f
    cmp rdi, rsi + 2
    jbe 9f
    cmp byte ptr [rsi + 2], 'l'
    jne 9f
    cmp rdi, rsi + 3
    jbe 9f
    cmp byte ptr [rsi + 3], 'l'
    jne 9f
    lea rax, [rsi + 4]
    ret
9:
    mov rax, rsi
    ret

print_decimal:
    mov rax, rsi
    lea rdi, [decimal_buf + 20]
    mov byte ptr [rdi - 1], 0
    mov rcx, 10
    dec rdi
    test rax, rax
    jz 1f
2:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz 2b
    jmp 3f
1:
    mov byte ptr [rdi], '0'
    dec rdi
3:
    inc rdi
    mov rsi, rdi
    lea rdx, [decimal_buf + 20]
    sub rdx, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    ret
```
