```assembly
.globl _start

.section .rodata
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg
error_msg_prefix: .ascii "error at "
error_msg_prefix_len = . - error_msg_prefix
newline: .ascii "\n"

.section .bss
input_buffer: .space 65536
decimal_buf: .space 20

.text
_start:
    xor r8, r8
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buffer]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jle .read_done
    add r8, rax
    cmp r8, 65536
    jl .read_loop
.read_done:
    lea rsi, [input_buffer]
    lea rdi, [input_buffer + r8]
    call parse_document
    cmp rax, rdi
    je .success
    lea rsi, [error_msg_prefix]
    mov rdx, error_msg_prefix_len
    mov rax, 1
    mov rdi, 1
    syscall
    mov rsi, rax
    call print_decimal
    lea rsi, [newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall

.success:
    lea rsi, [ok_msg]
    mov rdx, ok_msg_len
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

skip_ws:
.loop:
    cmp rsi, rdi
    je .done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je .ws
    cmp al, 0x09
    je .ws
    cmp al, 0x0A
    je .ws
    cmp al, 0x0D
    je .ws
    jmp .done
.ws:
    inc rsi
    jmp .loop
.done:
    mov rax, rsi
    ret

parse_document:
    call skip_ws
    mov rsi, rax
    call parse_value
    cmp rax, rsi
    je .error
    mov rsi, rax
    call skip_ws
    cmp rax, rdi
    jne .error
    mov rax, rdi
    ret
.error:
    ret

parse_value:
    cmp rsi, rdi
    jbe .error
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je .object
    cmp al, '['
    je .array
    cmp al, '"'
    je .string
    cmp al, 't'
    je .true
    cmp al, 'f'
    je .false
    cmp al, 'n'
    je .null
    cmp al, '-'
    je .number
    cmp al, '0'
    je .number
    cmp al, '1'
    jb .error
    cmp al, '9'
    jbe .number
.error:
    mov rax, rsi
    ret
.object:
    call parse_object
    ret
.array:
    call parse_array
    ret
.string:
    call parse_string
    ret
.true:
    call parse_true
    ret
.false:
    call parse_false
    ret
.null:
    call parse_null
    ret
.number:
    call parse_number
    ret

parse_object:
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], '{'
    jne .error
    inc rsi
    call skip_ws
    mov rsi, rax
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], '}'
    je .empty
.member:
    call parse_string
    cmp rax, rsi
    je .error
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], ':'
    jne .error
    inc rsi
    call skip_ws
    mov rsi, rax
    call parse_value
    cmp rax, rsi
    je .error
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], '}'
    je .end
    cmp byte ptr [rsi], ','
    jne .error
    inc rsi
    call skip_ws
    mov rsi, rax
    jmp .member
.empty:
    inc rsi
    call skip_ws
    mov rax, rsi
    ret
.end:
    inc rsi
    call skip_ws
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

parse_array:
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], '['
    jne .error
    inc rsi
    call skip_ws
    mov rsi, rax
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], ']'
    je .empty
.value:
    call parse_value
    cmp rax, rsi
    je .error
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], ']'
    je .end
    cmp byte ptr [rsi], ','
    jne .error
    inc rsi
    call skip_ws
    mov rsi, rax
    jmp .value
.empty:
    inc rsi
    call skip_ws
    mov rax, rsi
    ret
.end:
    inc rsi
    call skip_ws
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

parse_string:
    cmp rsi, rdi
    jbe .error
    cmp byte ptr [rsi], '"'
    jne .error
    inc rsi
.loop:
    cmp rsi, rdi
    je .error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je .end
    cmp al, '\'
    je .escape
    test al, al
    jz .error
    cmp al, 0x1F
    jbe .error
    inc rsi
    jmp .loop
.escape:
    inc rsi
    cmp rsi, rdi
    je .error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je .escape_ok
    cmp al, '\'
    je .escape_ok
    cmp al, '/'
    je .escape_ok
    cmp al, 'b'
    je .escape_ok
    cmp al, 'f'
    je .escape_ok
    cmp al, 'n'
    je .escape_ok
    cmp al, 'r'
    je .escape_ok
    cmp al, 't'
    je .escape_ok
    cmp al, 'u'
    je .unicode
    jmp .error
.escape_ok:
    inc rsi
    jmp .loop
.unicode:
    inc rsi
    mov rcx, 4
.unicode_loop:
    test rcx, rcx
    jz .loop
    cmp rsi, rdi
    jbe .error
    movzx eax, byte ptr [rsi]
    call is_hex_digit
    test rax, rax
    jz .error
    inc rsi
    dec rcx
    jmp .unicode_loop
.end:
    inc rsi
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

is_hex_digit:
    cmp al, '0'
    jb .no
    cmp al, '9'
    jbe .yes
    cmp al, 'a'
    jb .no
    cmp al, 'f'
    jbe .yes
    cmp al, 'A'
    jb .no
    cmp al, 'F'
    jbe .yes
.no:
    xor rax, rax
    ret
.yes:
    mov rax, 1
    ret

parse_number:
    cmp rsi, rdi
    jbe .error
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne .int
    inc rsi
.int:
    cmp rsi, rdi
    jbe .error
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je .zero
    cmp al, '1'
    jb .error
    cmp al, '9'
    jbe .digits
    jmp .error
.zero:
    inc rsi
    cmp rsi, rdi
    jbe .frac_exp
    movzx eax, byte ptr [rsi]
    call is_digit
    test rax, rax
    jnz .error
    jmp .frac_exp
.digits:
    inc rsi
    call parse_digits
    mov rsi, rax
.frac_exp:
    cmp rsi, rdi
    jbe .done
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    je .frac
    cmp al, 'e'
    je .exp
    cmp al, 'E'
    jne .done
.exp:
    inc rsi
    cmp rsi, rdi
    jbe .error
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .exp_sign
    cmp al, '-'
    jne .exp_digits
.exp_sign:
    inc rsi
.exp_digits:
    call parse_digits
    cmp rax, rsi
    je .error
    mov rsi, rax
    jmp .done
.frac:
    inc rsi
    call parse_digits
    cmp rax, rsi
    je .error
    mov rsi, rax
    jmp .frac_exp
.done:
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

parse_digits:
.loop:
    cmp rsi, rdi
    jbe .done
    movzx eax, byte ptr [rsi]
    call is_digit
    test rax, rax
    jz .done
    inc rsi
    jmp .loop
.done:
    mov rax, rsi
    ret

is_digit:
    cmp al, '0'
    jb .no
    cmp al, '9'
    jbe .yes
.no:
    xor rax, rax
    ret
.yes:
    mov rax, 1
    ret

parse_true:
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 't'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'r'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'u'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'e'
    jne .error
    inc rsi
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

parse_false:
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'f'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'a'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'l'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 's'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'e'
    jne .error
    inc rsi
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

parse_null:
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'n'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'u'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'l'
    jne .error
    inc rsi
    cmp rdi, rsi
    jbe .error
    cmp byte ptr [rsi], 'l'
    jne .error
    inc rsi
    mov rax, rsi
    ret
.error:
    mov rax, rsi
    ret

print_decimal:
    mov rax, rsi
    lea rdi, [decimal_buf + 20]
    mov byte ptr [rdi - 1], 0
    mov rcx, 10
    dec rdi
    test rax, rax
    jz .zero
.loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz .loop
    jmp .print
.zero:
    mov byte ptr [rdi], '0'
    dec rdi
.print:
    inc rdi
    mov rsi, rdi
    lea rdx, [decimal_buf + 20]
    sub rdx, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    ret
```
