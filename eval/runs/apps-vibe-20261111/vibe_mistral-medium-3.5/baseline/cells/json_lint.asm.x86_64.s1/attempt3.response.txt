```asm
.globl _start

.section .rodata
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg
error_msg_prefix: .ascii "error at "
error_msg_prefix_len = . - error_msg_prefix
newline: .ascii "\n"

.section .bss
input_buffer: .space 65536
decimal_buf: .space 20

.text
_start:
    xor r8, r8
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buffer]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jle .read_done
    add r8, rax
    cmp r8, 65536
    jl .read_loop
.read_done:
    lea rsi, [input_buffer]
    lea rdi, [input_buffer + r8]
    call parse_document
    cmp rax, rdi
    je .success
    mov rdi, 1
    mov rax, 1
    lea rsi, [error_msg_prefix]
    mov rdx, error_msg_prefix_len
    syscall
    mov rsi, rax
    call print_decimal
    lea rsi, [newline]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall

.success:
    mov rdi, 1
    mov rax, 1
    lea rsi, [ok_msg]
    mov rdx, ok_msg_len
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

skip_ws:
    mov rcx, rsi
.skip_ws_loop:
    cmp rcx, rdi
    je .skip_ws_done
    movzx eax, byte ptr [rcx]
    cmp al, ' '
    je .skip_ws_next
    cmp al, '\t'
    je .skip_ws_next
    cmp al, '\n'
    je .skip_ws_next
    cmp al, '\r'
    je .skip_ws_next
    jmp .skip_ws_done
.skip_ws_next:
    inc rcx
    jmp .skip_ws_loop
.skip_ws_done:
    mov rax, rcx
    ret

parse_document:
    call skip_ws
    mov r13, rax
    cmp rax, rdi
    je .parse_doc_fail
    mov rsi, rax
    call parse_value
    cmp rax, r13
    je .parse_doc_fail
    mov rsi, rax
    call skip_ws
    cmp rax, rdi
    jne .parse_doc_fail
    mov rax, rdi
    ret
.parse_doc_fail:
    mov rax, r13
    ret

parse_value:
    cmp rsi, rdi
    jbe .parse_value_fail
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je .parse_value_object
    cmp al, '['
    je .parse_value_array
    cmp al, '"'
    je .parse_value_string
    cmp al, 't'
    je .parse_value_true
    cmp al, 'f'
    je .parse_value_false
    cmp al, 'n'
    je .parse_value_null
    cmp al, '-'
    je .parse_value_number
    cmp al, '0'
    je .parse_value_number
    cmp al, '1'
    jb .parse_value_fail
    cmp al, '9'
    jbe .parse_value_number
.parse_value_fail:
    mov rax, rsi
    ret
.parse_value_object:
    jmp parse_object
.parse_value_array:
    jmp parse_array
.parse_value_string:
    jmp parse_string
.parse_value_true:
    jmp parse_true
.parse_value_false:
    jmp parse_false
.parse_value_null:
    jmp parse_null
.parse_value_number:
    jmp parse_number

parse_object:
    cmp rsi, rdi
    jbe .parse_obj_fail
    cmp byte ptr [rsi], '{'
    jne .parse_obj_fail
    mov rcx, rsi
    inc rcx
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je .parse_obj_fail
    cmp byte ptr [r12], '}'
    je .parse_obj_end
.parse_obj_members:
    mov rsi, r12
    call parse_string
    cmp rax, r12
    je .parse_obj_fail
    mov rsi, rax
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je .parse_obj_fail
    cmp byte ptr [r12], ':'
    jne .parse_obj_fail
    mov rcx, r12
    inc rcx
    call skip_ws
    mov r12, rax
    mov rsi, r12
    call parse_value
    cmp rax, r12
    je .parse_obj_fail
    mov rsi, rax
    call skip_ws
    mov r12, rax
    cmp rax, rdi
    je .parse_obj_fail
    cmp byte ptr [r12], '}'
    je .parse_obj_end
    cmp byte ptr [r12], ','
    jne .parse_obj_fail
    mov rcx, r12
    inc rcx
    call skip_ws
    mov r12, rax
    jmp .parse_obj_members
.parse_obj_end:
    mov rcx, r12
    inc rcx
    call skip_ws
    mov rax, rcx
    ret
.parse_obj_fail:
    mov rax, r12
    ret

parse_array:
    cmp rsi, rdi
    jbe .parse_arr_fail
    cmp byte ptr [rsi], '['
    jne .parse_arr_fail
    mov rcx, rsi
    inc rcx
    call skip_ws
    mov r14, rax
    cmp rax, rdi
    je .parse_arr_fail
    cmp byte ptr [r14], ']'
    je .parse_arr_end
.parse_arr_values:
    mov rsi, r14
    call parse_value
    cmp rax, r14
    je .parse_arr_fail
    mov rsi, rax
    call skip_ws
    mov r14, rax
    cmp rax, rdi
    je .parse_arr_fail
    cmp byte ptr [r14], ']'
    je .parse_arr_end
    cmp byte ptr [r14], ','
    jne .parse_arr_fail
    mov rcx, r14
    inc rcx
    call skip_ws
    mov r14, rax
    jmp .parse_arr_values
.parse_arr_end:
    mov rcx, r14
    inc rcx
    call skip_ws
    mov rax, rcx
    ret
.parse_arr_fail:
    mov rax, r14
    ret

parse_string:
    cmp rsi, rdi
    jbe .parse_str_fail
    cmp byte ptr [rsi], '"'
    jne .parse_str_fail
    mov rcx, rsi
    inc rcx
.parse_str_loop:
    cmp rcx, rdi
    je .parse_str_fail
    movzx eax, byte ptr [rcx]
    cmp al, '"'
    je .parse_str_done
    cmp al, '\'
    je .parse_str_escape
    test al, al
    jz .parse_str_fail
    cmp al, 0x1F
    jbe .parse_str_fail
    inc rcx
    jmp .parse_str_loop
.parse_str_escape:
    inc rcx
    cmp rcx, rdi
    je .parse_str_fail
    movzx eax, byte ptr [rcx]
    cmp al, '"'
    je .parse_str_after_escape
    cmp al, '\'
    je .parse_str_after_escape
    cmp al, '/'
    je .parse_str_after_escape
    cmp al, 'b'
    je .parse_str_after_escape
    cmp al, 'f'
    je .parse_str_after_escape
    cmp al, 'n'
    je .parse_str_after_escape
    cmp al, 'r'
    je .parse_str_after_escape
    cmp al, 't'
    je .parse_str_after_escape
    cmp al, 'u'
    jne .parse_str_fail
    inc rcx
    mov r15, 4
.parse_str_u_loop:
    test r15, r15
    jz .parse_str_loop
    cmp rcx, rdi
    jbe .parse_str_fail
    movzx eax, byte ptr [rcx]
    call is_hex_digit
    test rax, rax
    jz .parse_str_fail
    inc rcx
    dec r15
    jmp .parse_str_u_loop
.parse_str_after_escape:
    inc rcx
    jmp .parse_str_loop
.parse_str_done:
    inc rcx
    mov rax, rcx
    ret
.parse_str_fail:
    mov rax, rcx
    ret

is_hex_digit:
    cmp al, '0'
    jb .is_hex_no
    cmp al, '9'
    jbe .is_hex_yes
    cmp al, 'a'
    jb .is_hex_no
    cmp al, 'f'
    jbe .is_hex_yes
    cmp al, 'A'
    jb .is_hex_no
    cmp al, 'F'
    jbe .is_hex_yes
.is_hex_no:
    xor rax, rax
    ret
.is_hex_yes:
    mov rax, 1
    ret

parse_number:
    cmp rsi, rdi
    jbe .parse_num_fail
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne .parse_num_no_neg
    mov rcx, rsi
    inc rcx
    jmp .parse_num_int
.parse_num_no_neg:
    mov rcx, rsi
.parse_num_int:
    cmp rcx, rdi
    jbe .parse_num_fail
    movzx eax, byte ptr [rcx]
    cmp al, '0'
    je .parse_num_zero
    cmp al, '1'
    jb .parse_num_fail
    cmp al, '9'
    jbe .parse_num_digits
    jmp .parse_num_fail
.parse_num_zero:
    inc rcx
    cmp rcx, rdi
    jbe .parse_num_after_int
    movzx eax, byte ptr [rcx]
    call is_digit
    test rax, rax
    jnz .parse_num_fail
.parse_num_after_int:
.parse_num_digits:
    call parse_digits
    mov rcx, rax
.parse_num_frac_exp:
    cmp rcx, rdi
    jbe .parse_num_done
    movzx eax, byte ptr [rcx]
    cmp al, '.'
    je .parse_num_frac
    cmp al, 'e'
    je .parse_num_exp
    cmp al, 'E'
    je .parse_num_exp
    jmp .parse_num_done
.parse_num_frac:
    inc rcx
    call parse_digits
    cmp rax, rcx
    je .parse_num_fail
    mov rcx, rax
    jmp .parse_num_frac_exp
.parse_num_exp:
    inc rcx
    cmp rcx, rdi
    jbe .parse_num_fail
    movzx eax, byte ptr [rcx]
    cmp al, '+'
    je .parse_num_exp_sign
    cmp al, '-'
    je .parse_num_exp_sign
    jmp .parse_num_exp_digits
.parse_num_exp_sign:
    inc rcx
.parse_num_exp_digits:
    call parse_digits
    cmp rax, rcx
    je .parse_num_fail
    mov rcx, rax
    jmp .parse_num_frac_exp
.parse_num_done:
    mov rax, rcx
    ret
.parse_num_fail:
    mov rax, rcx
    ret

parse_digits:
.parse_digits_loop:
    cmp rcx, rdi
    jbe .parse_digits_done
    movzx eax, byte ptr [rcx]
    call is_digit
    test rax, rax
    jz .parse_digits_done
    inc rcx
    jmp .parse_digits_loop
.parse_digits_done:
    mov rax, rcx
    ret

is_digit:
    cmp al, '0'
    jb .is_digit_no
    cmp al, '9'
    jbe .is_digit_yes
.is_digit_no:
    xor rax, rax
    ret
.is_digit_yes:
    mov rax, 1
    ret

parse_true:
    mov rcx, rsi
    cmp rdi, rcx
    jbe .parse_true_fail
    cmp byte ptr [rcx], 't'
    jne .parse_true_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_true_fail
    cmp byte ptr [rcx], 'r'
    jne .parse_true_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_true_fail
    cmp byte ptr [rcx], 'u'
    jne .parse_true_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_true_fail
    cmp byte ptr [rcx], 'e'
    jne .parse_true_fail
    inc rcx
    mov rax, rcx
    ret
.parse_true_fail:
    mov rax, rcx
    ret

parse_false:
    mov rcx, rsi
    cmp rdi, rcx
    jbe .parse_false_fail
    cmp byte ptr [rcx], 'f'
    jne .parse_false_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_false_fail
    cmp byte ptr [rcx], 'a'
    jne .parse_false_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_false_fail
    cmp byte ptr [rcx], 'l'
    jne .parse_false_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_false_fail
    cmp byte ptr [rcx], 's'
    jne .parse_false_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_false_fail
    cmp byte ptr [rcx], 'e'
    jne .parse_false_fail
    inc rcx
    mov rax, rcx
    ret
.parse_false_fail:
    mov rax, rcx
    ret

parse_null:
    mov rcx, rsi
    cmp rdi, rcx
    jbe .parse_null_fail
    cmp byte ptr [rcx], 'n'
    jne .parse_null_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_null_fail
    cmp byte ptr [rcx], 'u'
    jne .parse_null_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_null_fail
    cmp byte ptr [rcx], 'l'
    jne .parse_null_fail
    inc rcx
    cmp rdi, rcx
    jbe .parse_null_fail
    cmp byte ptr [rcx], 'l'
    jne .parse_null_fail
    inc rcx
    mov rax, rcx
    ret
.parse_null_fail:
    mov rax, rcx
    ret

print_decimal:
    mov rax, rsi
    lea rdi, [decimal_buf + 20]
    mov byte ptr [rdi - 1], 0
    mov rcx, 10
    dec rdi
    test rax, rax
    jz .print_dec_zero
.print_dec_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz .print_dec_loop
    jmp .print_dec_done
.print_dec_zero:
    mov byte ptr [rdi], '0'
    dec rdi
.print_dec_done:
    inc rdi
    mov rsi, rdi
    lea rdx, [decimal_buf + 20]
    sub rdx, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    ret
```
