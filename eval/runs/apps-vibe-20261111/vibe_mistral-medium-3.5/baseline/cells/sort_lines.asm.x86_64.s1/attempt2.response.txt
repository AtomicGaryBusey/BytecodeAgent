```asm
.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.text
_start:
	xor rbx, rbx
read_loop:
	mov rax, SYS_READ
	mov rdi, STDIN
	lea rsi, [rip + input_buffer]
	add rsi, rbx
	mov rdx, MAX_INPUT
	sub rdx, rbx
	syscall
	test rax, rax
	jz read_done
	add rbx, rax
	cmp rbx, MAX_INPUT
	jb read_loop
read_done:

	lea rsi, [rip + input_buffer]
	xor rdi, rdi
	xor rcx, rcx
	xor r8, r8
	mov r9, rbx

	test r9, r9
	jz no_lines
split_loop:
	cmp r8, r9
	jge after_split
	mov al, byte ptr [rsi + r8]
	cmp al, 0x0A
	jne not_newline
	mov [line_starts + rcx*8], rdi
	mov [line_lengths + rcx*8], r8
	sub qword ptr [line_lengths + rcx*8], rdi
	inc rcx
	mov rdi, r8
	inc rdi
	jmp next_char
not_newline:
next_char:
	inc r8
	jmp split_loop
after_split:
	cmp rdi, r9
	jge no_final_line
	mov [line_starts + rcx*8], rdi
	mov rax, r9
	sub rax, rdi
	mov [line_lengths + rcx*8], rax
	inc rcx
no_final_line:
	mov [line_count], rcx
no_lines:

	mov r10, [line_count]
	test r10, r10
	jz print_done
	dec r10
	jz print_lines

	xor r11, r11
outer_loop:
	xor r12, r12
	mov r13, r10
	sub r13, r11
inner_loop:
	mov r14, [line_starts + r12*8]
	mov r15, [line_lengths + r12*8]
	mov rax, [line_starts + r12*8 + 8]
	mov rbx, [line_lengths + r12*8 + 8]
	lea r14, [input_buffer + r14]
	lea rax, [input_buffer + rax]
	xor rcx, rcx
	mov rdi, r15
	mov rsi, rbx
	cmp rdi, rsi
	cmovg rdi, rsi
	test rdi, rdi
	jz compare_lengths
compare_bytes:
	mov dl, byte ptr [r14 + rcx]
	mov dh, byte ptr [rax + rcx]
	cmp dl, dh
	jl no_swap_needed
	jg need_swap
	inc rcx
	cmp rcx, rdi
	jb compare_bytes
compare_lengths:
	cmp r15, rbx
	jl no_swap_needed
	jg need_swap
	jmp no_swap_needed
need_swap:
	mov r8, [line_starts + r12*8]
	mov r9, [line_starts + r12*8 + 8]
	mov [line_starts + r12*8], r9
	mov [line_starts + r12*8 + 8], r8
	mov r8, [line_lengths + r12*8]
	mov r9, [line_lengths + r12*8 + 8]
	mov [line_lengths + r12*8], r9
	mov [line_lengths + r12*8 + 8], r8
no_swap_needed:
	inc r12
	cmp r12, r13
	jb inner_loop
	inc r11
	cmp r11, r10
	jb outer_loop

print_lines:
	xor r12, r12
	mov r13, [line_count]
print_loop:
	cmp r12, r13
	jge print_done
	mov r14, [line_starts + r12*8]
	mov r15, [line_lengths + r12*8]
	mov rax, SYS_WRITE
	mov rdi, STDOUT
	lea rsi, [input_buffer + r14]
	mov rdx, r15
	syscall
	mov rax, SYS_WRITE
	mov rdi, STDOUT
	lea rsi, [rip + newline]
	mov rdx, 1
	syscall
	inc r12
	jmp print_loop

print_done:
	mov rax, SYS_EXIT
	xor rdi, rdi
	syscall

.section .rodata
newline:
	.byte 0x0A

.bss
.align 8
input_buffer: .space MAX_INPUT
line_starts: .space MAX_LINES * 8
line_lengths: .space MAX_LINES * 8
line_count: .space 8
```
