```asm
.global _start

.text
_start:
    lea r12, [buffer]
    xor r13, r13
    xor r14, r14
    xor r15, r15

main_loop:
    cmp r15, r13
    jge need_read

    movzx eax, byte ptr [r12 + r15]
    cmp al, 10
    jne not_newline

    call process_line
    mov r14, r15
    inc r14
    inc r15
    jmp main_loop

not_newline:
    inc r15
    jmp main_loop

need_read:
    mov rax, 65536
    sub rax, r13
    jle read_done

    mov rdx, rax
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r13]
    syscall

    cmp rax, 0
    jle read_done

    add r13, rax
    jmp main_loop

read_done:
    cmp r14, r13
    jge exit
    mov r15, r13
    call process_line
    jmp exit

process_line:
    mov r8, r14
    mov r9, r15
    dec r9
reverse_loop:
    cmp r8, r9
    jge reverse_done
    mov al, byte ptr [r12 + r8]
    mov bl, byte ptr [r12 + r9]
    mov byte ptr [r12 + r8], bl
    mov byte ptr [r12 + r9], al
    inc r8
    dec r9
    jmp reverse_loop
reverse_done:
    mov rdx, r15
    sub rdx, r14
    mov rax, 1
    mov rdi, 1
    lea rsi, [r12 + r14]
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    ret

exit:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
newline: .byte 10

.bss
buffer: .space 65536
```
