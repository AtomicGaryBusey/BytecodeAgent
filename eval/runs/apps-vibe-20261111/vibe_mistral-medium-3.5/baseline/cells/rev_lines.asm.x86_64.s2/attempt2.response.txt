  jz read_done

  jmp read_loop
read_done:
  test rbx, rbx
  jz exit_zero
  xor r8, r8
  xor r9, r9
  lea r10, [output_buf]

  cmp r9, rbx
  jae check_final
  mov al, byte ptr [input_buf + r9]

  mov rcx, r9

  dec rsi
  mov rax, rcx

  dec rsi

  dec rax

  mov byte ptr [r11], 0x0A
  inc r11
  mov r8, r9
  inc r8

  inc r9

  cmp r8, rbx
  jae write_output
  mov rcx, rbx
  sub rcx, r8
  mov rsi, input_buf

  dec rsi
  mov rax, rcx

  dec rsi
  inc r11
  dec rax

  inc r11

  mov rdx, r11
  sub rdx, r10
  mov rax, 1
  mov rdi, 1
  mov rsi, r10
