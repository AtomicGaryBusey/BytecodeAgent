.globl _start

.section .rodata
msg_ok: .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_error_prefix: .ascii "error at "
msg_error_prefix_len = . - msg_error_prefix
msg_newline: .ascii "\n"

.section .bss
input_buffer: .space 65536
num_buf: .space 20

.section .text
_start:
    mov rdi, 0
    mov rsi, input_buffer
    mov rdx, 65536
    call read_all
    mov r12, input_buffer
    mov r13, input_buffer
    lea r14, [r12 + rax]
    call skip_ws
    test rax, rax
    jnz error
    call parse_value
    test rax, rax
    jnz error
    call skip_ws
    test rax, rax
    jnz error
    cmp r13, r14
    jne error
    mov rdi, 1
    mov rsi, msg_ok
    mov rdx, msg_ok_len
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

error:
    mov rax, r13
    sub rax, r12
    mov rdi, num_buf
    call itoa
    mov rdi, 1
    mov rsi, msg_error_prefix
    mov rdx, msg_error_prefix_len
    mov rax, 1
    syscall
    mov rdi, 1
    mov rsi, num_buf
    mov rdx, rax
    mov rax, 1
    syscall
    mov rdi, 1
    mov rsi, msg_newline
    mov rdx, 1
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

read_all:
    xor r10, r10
    mov r8, rsi
    mov r9, rdx
.read_loop:
    test r9, r9
    jz .done
    mov rsi, r8
    mov rdx, r9
    mov rax, 0
    syscall
    test rax, rax
    jz .done
    js .done
    add r8, rax
    add r10, rax
    sub r9, rax
    jmp .read_loop
.done:
    mov rax, r10
    ret

skip_ws:
.skip_loop:
    cmp r13, r14
    je .end
    mov al, byte ptr [r13]
    call is_ws
    test rax, rax
    jz .end
    inc r13
    jmp .skip_loop
.end:
    xor rax, rax
    ret

is_ws:
    cmp al, 32
    je .yes
    cmp al, 9
    je .yes
    cmp al, 10
    je .yes
    cmp al, 13
    je .yes
    xor rax, rax
    ret
.yes:
    mov rax, 1
    ret

is_digit:
    cmp al, '0'
    jb .no
    cmp al, '9'
    ja .no
    mov rax, 1
    ret
.no:
    xor rax, rax
    ret

is_hex_digit:
    call is_digit
    test rax, rax
    jnz .yes
    or al, 0x20
    cmp al, 'a'
    jb .no
    cmp al, 'f'
    ja .no
.yes:
    mov rax, 1
    ret
.no:
    xor rax, rax
    ret

parse_string:
    cmp r13, r14
    jle .fail
    cmp byte ptr [r13], '"'
    jne .fail
    inc r13
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], '"'
    je .string_end
.string_loop:
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    cmp al, '"'
    je .string_end
    cmp al, '\'
    jne .check_normal
    inc r13
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    inc r13
    cmp al, '"'
    je .string_loop
    cmp al, '\'
    je .string_loop
    cmp al, '/'
    je .string_loop
    cmp al, 'b'
    je .string_loop
    cmp al, 'f'
    je .string_loop
    cmp al, 'n'
    je .string_loop
    cmp al, 'r'
    je .string_loop
    cmp al, 't'
    je .string_loop
    cmp al, 'u'
    jne .fail
    mov rcx, 4
.hex_loop:
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    inc r13
    call is_hex_digit
    test rax, rax
    jz .fail
    loop .hex_loop
    jmp .string_loop
.check_normal:
    cmp al, 0x20
    jb .fail
    inc r13
    jmp .string_loop
.string_end:
    inc r13
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_number:
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    cmp al, '-'
    jne .no_minus
    inc r13
    cmp r13, r14
    je .fail
.no_minus:
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    cmp al, '0'
    jne .non_zero
    inc r13
    jmp .after_int
.non_zero:
    call is_digit
    test rax, rax
    jz .fail
    inc r13
.non_zero_loop:
    cmp r13, r14
    je .after_int
    mov al, byte ptr [r13]
    call is_digit
    test rax, rax
    jz .after_int
    inc r13
    jmp .non_zero_loop
.after_int:
    cmp r13, r14
    je .no_frac
    cmp byte ptr [r13], '.'
    jne .no_frac
    inc r13
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    call is_digit
    test rax, rax
    jz .fail
    inc r13
.digit_loop:
    cmp r13, r14
    je .after_frac
    mov al, byte ptr [r13]
    call is_digit
    test rax, rax
    jz .after_frac
    inc r13
    jmp .digit_loop
.after_frac:
.no_frac:
    cmp r13, r14
    je .done
    mov al, byte ptr [r13]
    cmp al, 'e'
    je .exp
    cmp al, 'E'
    jne .done
.exp:
    inc r13
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    cmp al, '+'
    je .exp_sign
    cmp al, '-'
    jne .exp_digits
.exp_sign:
    inc r13
    cmp r13, r14
    je .fail
.exp_digits:
    mov al, byte ptr [r13]
    call is_digit
    test rax, rax
    jz .fail
    inc r13
.exp_digit_loop:
    cmp r13, r14
    je .done
    mov al, byte ptr [r13]
    call is_digit
    test rax, rax
    jz .done
    inc r13
    jmp .exp_digit_loop
.done:
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_true:
    cmp r13, r14
    jle .fail
    cmp byte ptr [r13], 't'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'r'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'u'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'e'
    jne .fail
    inc r13
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_false:
    cmp r13, r14
    jle .fail
    cmp byte ptr [r13], 'f'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'a'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'l'
    jne .fail
    inc r13
    cmp byte ptr [r13], 's'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'e'
    jne .fail
    inc r13
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_null:
    cmp r13, r14
    jle .fail
    cmp byte ptr [r13], 'n'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'u'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'l'
    jne .fail
    inc r13
    cmp byte ptr [r13], 'l'
    jne .fail
    inc r13
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_object:
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], '{'
    jne .fail
    inc r13
    call skip_ws
    test rax, rax
    jnz .fail
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], '}'
    je .object_end
.object_loop:
    call parse_string
    test rax, rax
    jnz .fail
    call skip_ws
    test rax, rax
    jnz .fail
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], ':'
    jne .fail
    inc r13
    call skip_ws
    test rax, rax
    jnz .fail
    call parse_value
    test rax, rax
    jnz .fail
    call skip_ws
    test rax, rax
    jnz .fail
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], ','
    jne .check_end
    inc r13
    call skip_ws
    test rax, rax
    jnz .fail
    jmp .object_loop
.check_end:
    cmp byte ptr [r13], '}'
    jne .fail
.object_end:
    inc r13
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_array:
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], '['
    jne .fail
    inc r13
    call skip_ws
    test rax, rax
    jnz .fail
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], ']'
    je .array_end
.array_loop:
    call parse_value
    test rax, rax
    jnz .fail
    call skip_ws
    test rax, rax
    jnz .fail
    cmp r13, r14
    je .fail
    cmp byte ptr [r13], ','
    jne .check_end
    inc r13
    call skip_ws
    test rax, rax
    jnz .fail
    jmp .array_loop
.check_end:
    cmp byte ptr [r13], ']'
    jne .fail
.array_end:
    inc r13
    xor rax, rax
    ret
.fail:
    mov rax, 1
    ret

parse_value:
    cmp r13, r14
    je .fail
    mov al, byte ptr [r13]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    call is_digit
    test rax, rax
    jnz parse_number
.fail:
    mov rax, 1
    ret

itoa:
    mov rsi, rdi
    xor r8, r8
    test rax, rax
    jz .zero
.extract_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    push rdx
    inc r8
    test rax, rax
    jnz .extract_loop
.pop_loop:
    pop rax
    mov [rsi], al
    inc rsi
    dec r8
    jnz .pop_loop
    mov byte ptr [rsi], 0
    mov rax, rsi
    sub rax, rdi
    ret
.zero:
    mov byte ptr [rdi], '0'
    mov byte ptr [rdi+1], 0
    mov rax, 1
    ret
