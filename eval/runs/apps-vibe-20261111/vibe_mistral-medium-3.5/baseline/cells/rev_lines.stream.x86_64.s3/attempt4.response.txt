```asm
.global _start

.section .bss
input_buffer: .space 65536
output_buffer: .space 65537

.section .text
_start:
    lea rbx, [rip + input_buffer]
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13

process_loop:
    cmp r13, r12
    jge exit

    lea rdi, [rbx + r13]
    mov rcx, r12
    sub rcx, r13
    mov al, 0x0A
    repne scasb
    jne end_of_input

    mov r14, rdi
    sub r14, rbx
    sub r14, r13
    dec r14
    jmp reverse

end_of_input:
    mov r14, r12
    sub r14, r13

reverse:
    lea rsi, [rbx + r13]
    add rsi, r14
    dec rsi
    lea rdi, [rip + output_buffer]
    xor rcx, rcx

reverse_loop:
    cmp rcx, r14
    jge reverse_done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi + rcx], al
    dec rsi
    inc rcx
    jmp reverse_loop

reverse_done:
    mov byte ptr [rdi + r14], 0x0A

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    inc rdx
    syscall

    add r13, r14
    cmp r13, r12
    jge 1f
    inc r13
1:
    jmp process_loop

exit:
    mov rax, 60
    xor rdi, rdi
    syscall
```
