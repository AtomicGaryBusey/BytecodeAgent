.globl _start

sys_read = 0
sys_write = 1
sys_exit = 60
stdin = 0
stdout = 1
newline = 10
buffer_size = 65536

.section .rodata
newline_char:
    .byte newline

.bss
    .align 8
buffer:
    .zero buffer_size

.text
_start:
    xor r8, r8
1:
    lea rsi, [buffer + r8]
    mov rdx, buffer_size
    sub rdx, r8
    jz 2f
    mov rax, sys_read
    mov rdi, stdin
    syscall
    cmp rax, 0
    jl error
    test rax, rax
    jz 2f
    add r8, rax
    jmp 1b
2:
    xor r9, r9
3:
    cmp r9, r8
    jge 9f
    mov r10, r9
4:
    cmp r10, r8
    jge 8f
    cmp byte ptr [buffer + r10], newline
    je 5f
    inc r10
    jmp 4b
5:
    mov r11, r10
    sub r11, r9
    mov r12, r9
    lea r13, [r10 - 1]
10:
    cmp r12, r13
    jge 6f
    mov al, byte ptr [buffer + r12]
    mov bl, byte ptr [buffer + r13]
    mov byte ptr [buffer + r12], bl
    mov byte ptr [buffer + r13], al
    inc r12
    dec r13
    jmp 10b
6:
    mov rax, sys_write
    mov rdi, stdout
    lea rsi, [buffer + r9]
    lea rdx, [r11 + 1]
    syscall
    lea r9, [r10 + 1]
    jmp 3b
8:
    mov r11, r8
    sub r11, r9
    jz 9f
    mov r12, r9
    lea r13, [r8 - 1]
11:
    cmp r12, r13
    jge 12f
    mov al, byte ptr [buffer + r12]
    mov bl, byte ptr [buffer + r13]
    mov byte ptr [buffer + r12], bl
    mov byte ptr [buffer + r13], al
    inc r12
    dec r13
    jmp 11b
12:
    mov rax, sys_write
    mov rdi, stdout
    lea rsi, [buffer + r9]
    mov rdx, r11
    syscall
    mov rax, sys_write
    mov rdi, stdout
    lea rsi, [newline_char]
    mov rdx, 1
    syscall
9:
    mov rax, sys_exit
    xor rdi, rdi
    syscall
error:
    mov rax, sys_exit
    mov rdi, 1
    syscall
