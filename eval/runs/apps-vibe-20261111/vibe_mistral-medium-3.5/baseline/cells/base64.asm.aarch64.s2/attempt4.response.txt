.globl _start
.text

_start:
	b main_loop

write_char:
	adr x1, out_byte
	strb w0, [x1]
	mov x0, #1
	mov x2, #1
	mov w8, #64
	svc #0
	adr x2, line_pos
	ldr w3, [x2]
	add w3, w3, #1
	str w3, [x2]
	cmp w3, #76
	b.ne 1f
	mov w0, #0x0a
	adr x1, out_byte
	strb w0, [x1]
	mov x0, #1
	mov x2, #1
	mov w8, #64
	svc #0
	str wzr, [x2]
1:
	ret

encode_group:
	lsr x3, x0, #2
	and x4, x0, #3
	lsl x4, x4, #4
	lsr x5, x1, #4
	orr x4, x4, x5
	and x5, x1, #0x0F
	lsl x5, x5, #2
	lsr x6, x2, #6
	orr x5, x5, x6
	and x6, x2, #0x3F
	adr x7, base64_table
	ldrb w3, [x7, x3]
	ldrb w4, [x7, x4]
	ldrb w5, [x7, x5]
	ldrb w6, [x7, x6]
	mov w0, w3
	bl write_char
	mov w0, w4
	bl write_char
	mov w0, w5
	bl write_char
	mov w0, w6
	bl write_char
	ret

encode_leftover1:
	lsr x3, x0, #2
	and x4, x0, #0x03
	lsl x4, x4, #4
	adr x7, base64_table
	ldrb w3, [x7, x3]
	ldrb w4, [x7, x4]
	mov w0, w3
	bl write_char
	mov w0, w4
	bl write_char
	mov w0, #0x3d
	bl write_char
	mov w0, #0x3d
	bl write_char
	ret

encode_leftover2:
	lsr x3, x0, #2
	and x4, x0, #3
	lsl x4, x4, #4
	lsr x5, x1, #4
	orr x4, x4, x5
	and x5, x1, #0x0F
	lsl x5, x5, #2
	adr x7, base64_table
	ldrb w3, [x7, x3]
	ldrb w4, [x7, x4]
	ldrb w5, [x7, x5]
	mov w0, w3
	bl write_char
	mov w0, w4
	bl write_char
	mov w0, w5
	bl write_char
	mov w0, #0x3d
	bl write_char
	ret

main_loop:
	adr x21, leftover_count
	ldr w3, [x21]
	cbz w3, no_leftover_copy
	adr x22, leftover_bytes
	adr x23, read_buf
	cmp w3, #1
	b.eq copy_leftover_1
	ldrb w4, [x22]
	strb w4, [x23]
	ldrb w4, [x22, #1]
	strb w4, [x23, #1]
	b copy_leftover_done
copy_leftover_1:
	ldrb w4, [x22]
	strb w4, [x23]
copy_leftover_done:
no_leftover_copy:
	adr x0, read_buf
	add x0, x0, x3, uxtw
	mov x1, x0
	mov w2, #4098
	sub w2, w2, w3
	mov w0, #0
	mov w8, #63
	svc #0
	mov w4, w0
	add w5, w3, w4
	cbz w5, done_encoding
	mov w6, #0
group_loop:
	add w7, w6, #3
	cmp w7, w5
	b.hi group_loop_done
	adr x8, read_buf
	add x8, x8, w6, uxtw
	ldrb w0, [x8]
	ldrb w1, [x8, #1]
	ldrb w2, [x8, #2]
	bl encode_group
	add w6, w6, #3
	b group_loop
group_loop_done:
	sub w3, w5, w6
	adr x21, leftover_count
	str w3, [x21]
	cbz w3, no_copy_leftover_final
	adr x8, read_buf
	adr x1, leftover_bytes
	cmp w3, #1
	b.eq copy_leftover_final_1
	add x8, x8, w6, uxtw
	ldrb w4, [x8]
	strb w4, [x1]
	ldrb w4, [x8, #1]
	strb w4, [x1, #1]
	b no_copy_leftover_final
copy_leftover_final_1:
	add x8, x8, w6, uxtw
	ldrb w4, [x8]
	strb w4, [x1]
no_copy_leftover_final:
	cbz w4, done_encoding
	b main_loop

done_encoding:
	adr x21, leftover_count
	ldr w3, [x21]
	cbz w3, no_final_leftover
	cmp w3, #1
	b.eq final_leftover1
	adr x2, leftover_bytes
	ldrb w0, [x2]
	ldrb w1, [x2, #1]
	bl encode_leftover2
	b no_final_leftover
final_leftover1:
	adr x2, leftover_bytes
	ldrb w0, [x2]
	bl encode_leftover1
no_final_leftover:
	adr x0, line_pos
	ldr w0, [x0]
	cbz w0, no_final_newline
	mov w1, #0x0a
	adr x0, out_byte
	strb w1, [x0]
	mov x0, #1
	adr x1, out_byte
	mov x2, #1
	mov w8, #64
	svc #0
no_final_newline:
	mov x0, #0
	mov w8, #93
	svc #0

.section .rodata
base64_table:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
read_buf: .space 4098
leftover_bytes: .space 3
leftover_count: .space 4
line_pos: .space 4
out_byte: .space 1
