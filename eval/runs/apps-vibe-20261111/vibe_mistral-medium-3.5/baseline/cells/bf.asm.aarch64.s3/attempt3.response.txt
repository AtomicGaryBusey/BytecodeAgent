.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start

_start:
	adrp x4, input_buffer
	add x4, x4, :lo12:input_buffer
	mov x3, #0
read_loop:
	mov x0, #0
	add x1, x4, x3
	movz x2, #1, lsl #16
	sub x2, x2, x3
	mov x8, #SYS_READ
	svc #0
	cbz x0, read_done
	add x3, x3, x0
	b read_loop
read_done:
	mov x5, #0
	mov x6, x4
find_bang:
	cmp x5, x3
	b.ge no_bang
	ldrb w7, [x6, x5]
	cmp w7, #0x21
	b.eq found_bang
	add x5, x5, #1
	b find_bang
found_bang:
	mov x7, x5
	add x20, x6, x5
	add x20, x20, #1
	sub x9, x3, x5
	sub x9, x9, #1
	b validate_brackets
no_bang:
	mov x7, x3
	mov x20, xzr
	mov x9, xzr

validate_brackets:
	adrp x17, bracket_stack
	add x17, x17, :lo12:bracket_stack
	mov x18, #0
	adrp x16, bracket_match
	add x16, x16, :lo12:bracket_match
	mov x13, #0
val_loop:
	cmp x13, x7
	b.ge val_done
	ldrb w14, [x6, x13]
	cmp w14, #'['
	beq open_bracket
	cmp w14, #']'
	beq close_bracket
	add x13, x13, #1
	b val_loop
open_bracket:
	lsr x19, x18, #3
	cmp x19, #1000
	b.hs val_error
	str x13, [x17, x18]
	add x18, x18, #8
	add x13, x13, #1
	b val_loop
close_bracket:
	cmp x18, #0
	beq val_error
	sub x18, x18, #8
	ldr x15, [x17, x18]
	str x15, [x16, x13, lsl #3]
	str x13, [x16, x15, lsl #3]
	add x13, x13, #1
	b val_loop
val_done:
	cmp x18, #0
	b.ne val_error
	b interpret
val_error:
	mov x0, #1
	mov x8, #SYS_EXIT
	svc #0

interpret:
	adrp x10, tape
	add x10, x10, :lo12:tape
	mov x11, #0
	mov x12, #0
	mov x13, #0
interp_loop:
	cmp x13, x7
	b.ge interp_done
	ldrb w0, [x6, x13]
	cmp w0, #'['
	beq cmd_open
	cmp w0, #']'
	beq cmd_close
	cmp w0, #'>'
	beq cmd_gt
	cmp w0, #'<'
	beq cmd_lt
	cmp w0, #'+'
	beq cmd_plus
	cmp w0, #'-'
	beq cmd_minus
	cmp w0, #'.'
	beq cmd_dot
	cmp w0, #','
	beq cmd_comma
	add x13, x13, #1
	b interp_loop
cmd_open:
	ldrb w1, [x10, x11]
	cbnz w1, cmd_open_skip
	ldr x14, [x16, x13, lsl #3]
	add x13, x14, #1
	b interp_loop
cmd_open_skip:
	add x13, x13, #1
	b interp_loop
cmd_close:
	ldrb w1, [x10, x11]
	cbz w1, cmd_close_skip
	ldr x14, [x16, x13, lsl #3]
	add x13, x14, #1
	b interp_loop
cmd_close_skip:
	add x13, x13, #1
	b interp_loop
cmd_gt:
	add x11, x11, #1
	add x13, x13, #1
	b interp_loop
cmd_lt:
	sub x11, x11, #1
	add x13, x13, #1
	b interp_loop
cmd_plus:
	ldrb w1, [x10, x11]
	add w1, w1, #1
	and w1, w1, #0xff
	strb w1, [x10, x11]
	add x13, x13, #1
	b interp_loop
cmd_minus:
	ldrb w1, [x10, x11]
	sub w1, w1, #1
	and w1, w1, #0xff
	strb w1, [x10, x11]
	add x13, x13, #1
	b interp_loop
cmd_dot:
	mov x0, #1
	add x1, x10, x11
	mov x2, #1
	mov x8, #SYS_WRITE
	svc #0
	add x13, x13, #1
	b interp_loop
cmd_comma:
	cmp x12, x9
	b.hs comma_zero
	ldrb w1, [x20, x12]
	strb w1, [x10, x11]
	add x12, x12, #1
	b comma_done
comma_zero:
	strb wzr, [x10, x11]
comma_done:
	add x13, x13, #1
	b interp_loop
interp_done:
	mov x0, #0
	mov x8, #SYS_EXIT
	svc #0

.bss
.align 3
input_buffer: .zero 65536
tape: .zero 30000
bracket_stack: .zero 8000
bracket_match: .zero 524288
