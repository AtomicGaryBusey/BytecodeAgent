```asm
.globl _start

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    sub rsp, 128
    lea r14, [rsp]
    lea r13, [rsp + 77]
    lea r12, [rsp + 125]
    xor rbx, rbx
    xor r15, r15

read_loop:
    test rbx, rbx
    jz no_copy
    mov rcx, rbx
    mov rsi, r12
    mov rdi, r13
    rep movsb
no_copy:
    mov rdi, 0
    lea rsi, [r13 + rbx]
    mov rdx, 48
    sub rdx, rbx
    mov rax, 0
    syscall
    test rax, rax
    jz end_of_input
    add rax, rbx
    mov rbp, rax
    xor rbx, rbx
    xor r8, r8
process_loop:
    mov rcx, rbp
    sub rcx, r8
    jle read_loop
    cmp rcx, 3
    jl handle_remainder

    movzx r9d, byte ptr [r13 + r8]
    movzx r10d, byte ptr [r13 + r8 + 1]
    movzx r11d, byte ptr [r13 + r8 + 2]

    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [base64_table + rax]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne skip1
    call write_line
skip1:

    and r9d, 3
    shl r9d, 4
    mov eax, r10d
    shr eax, 4
    or r9d, eax
    mov al, byte ptr [base64_table + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne skip2
    call write_line
skip2:

    and r10d, 15
    shl r10d, 2
    mov eax, r11d
    shr eax, 6
    or r10d, eax
    mov al, byte ptr [base64_table + r10]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne skip3
    call write_line
skip3:

    and r11d, 63
    mov al, byte ptr [base64_table + r11]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne skip4
    call write_line
skip4:

    add r8, 3
    jmp process_loop

handle_remainder:
    mov rbx, rcx
    mov rsi, r13
    add rsi, r8
    mov rdi, r12
    cmp rcx, 1
    je copy_one
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    mov al, byte ptr [rsi + 1]
    mov byte ptr [rdi + 1], al
    jmp read_loop
copy_one:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    jmp read_loop

end_of_input:
    test rbx, rbx
    jz check_output
    cmp rbx, 1
    je one_byte
    cmp rbx, 2
    je two_bytes
    jmp check_output

one_byte:
    movzx r9d, byte ptr [r12]
    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [base64_table + rax]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne o1
    call write_line
o1:
    and r9d, 3
    shl r9d, 4
    mov al, byte ptr [base64_table + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne o2
    call write_line
o2:
    mov byte ptr [r14 + r15], '='
    inc r15
    cmp r15, 76
    jne o3
    call write_line
o3:
    mov byte ptr [r14 + r15], '='
    inc r15
    cmp r15, 76
    jne o_done
    call write_line
o_done:
    jmp check_output

two_bytes:
    movzx r9d, byte ptr [r12]
    movzx r10d, byte ptr [r12 + 1]
    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [base64_table + rax]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne t1
    call write_line
t1:
    and r9d, 3
    shl r9d, 4
    mov eax, r10d
    shr eax, 4
    or r9d, eax
    mov al, byte ptr [base64_table + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne t2
    call write_line
t2:
    and r10d, 15
    shl r10d, 2
    mov al, byte ptr [base64_table + r10]
    mov byte ptr [r14 + r15], al
    inc r15
    cmp r15, 76
    jne t3
    call write_line
t3:
    mov byte ptr [r14 + r15], '='
    inc r15
    cmp r15, 76
    jne t_done
    call write_line
t_done:
    jmp check_output

check_output:
    test r15, r15
    jz exit
    mov byte ptr [r14 + r15], 0x0A
    mov rdi, 1
    mov rsi, r14
    lea rdx, [r15 + 1]
    mov rax, 1
    syscall

exit:
    xor rdi, rdi
    mov rax, 60
    syscall

write_line:
    mov byte ptr [r14 + 76], 0x0A
    mov rdi, 1
    mov rsi, r14
    mov rdx, 77
    mov rax, 1
    syscall
    xor r15, r15
    ret
```
