```asm
.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline: .byte 0x0A

.bss
input_buf: .zero 49152
line_buf: .zero 76

.text
_start:
    xor r8, r8
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, 49152
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:
    test r8, r8
    jz exit_zero

    lea r12, [rip + alphabet]
    lea r9, [rip + input_buf]
    lea r10, [rip + line_buf]
    xor r13, r13

    lea r14, [rip + input_buf]
    add r14, r8

main_loop:
    cmp r9, r14
    jae final_line

    lea rax, [r9 + 3]
    cmp r14, rax
    jb partial_group

    movzx r8d, byte ptr [r9]
    movzx r11d, byte ptr [r9 + 1]
    movzx r15d, byte ptr [r9 + 2]
    add r9, 3

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [r12 + rax]
    call emit_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r11d
    shr ecx, 4
    or al, cl
    mov al, byte ptr [r12 + rax]
    call emit_char

    mov eax, r11d
    and eax, 0x0F
    shl eax, 2
    mov ecx, r15d
    shr ecx, 6
    or al, cl
    mov al, byte ptr [r12 + rax]
    call emit_char

    and r15d, 0x3F
    mov al, byte ptr [r12 + r15]
    call emit_char
    jmp main_loop

partial_group:
    lea rax, [r9 + 1]
    cmp r14, rax
    je one_byte

    movzx r8d, byte ptr [r9]
    movzx r11d, byte ptr [r9 + 1]
    add r9, 2

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [r12 + rax]
    call emit_char

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r11d
    shr ecx, 4
    or al, cl
    mov al, byte ptr [r12 + rax]
    call emit_char

    mov eax, r11d
    and eax, 0x0F
    shl eax, 2
    mov al, byte ptr [r12 + rax]
    call emit_char

    mov al, '='
    call emit_char
    jmp main_loop

one_byte:
    movzx r8d, byte ptr [r9]
    inc r9

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [r12 + rax]
    call emit_char

    and r8d, 3
    shl r8d, 4
    mov al, byte ptr [r12 + r8]
    call emit_char

    mov al, '='
    call emit_char
    mov al, '='
    call emit_char
    jmp main_loop

emit_char:
    cmp r13, 76
    jne store_char

    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, 76
    syscall
    mov rax, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    xor r13, r13

store_char:
    mov byte ptr [r10 + r13], al
    inc r13
    ret

final_line:
    test r13, r13
    jz exit_zero
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, r13
    syscall
    mov rax, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
```
