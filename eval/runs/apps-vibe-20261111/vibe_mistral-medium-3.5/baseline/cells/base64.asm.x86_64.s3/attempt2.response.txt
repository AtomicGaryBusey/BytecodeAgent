.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline: .byte 0x0A

.bss
input_buf: .zero 49152
line_buf: .zero 76

.text
_start:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    mov rdx, 49152
    syscall
    mov r8, rax
    test r8, r8
    jz exit_zero

    lea r14, [rip + input_buf]
    add r14, r8

    lea r9, [rip + input_buf]
    lea r10, [rip + line_buf]
    xor r13, r13
    lea r12, [rip + alphabet]

main_loop:
    cmp r9, r14
    je final_line

    lea rax, [r9 + 3]
    cmp r14, rax
    jb partial_group

    movzx eax, byte ptr [r9]
    movzx ecx, byte ptr [r9 + 1]
    movzx edx, byte ptr [r9 + 2]
    add r9, 3

    mov r15d, eax
    shr r15d, 2
    mov al, byte ptr [r12 + r15]
    call emit_char

    and eax, 3
    shl eax, 4
    shr ecx, 4
    or al, cl
    mov al, byte ptr [r12 + rax]
    call emit_char

    and ecx, 0x0F
    shl ecx, 2
    shr edx, 6
    or cl, dl
    mov al, byte ptr [r12 + rcx]
    call emit_char

    and edx, 0x3F
    mov al, byte ptr [r12 + rdx]
    call emit_char

    jmp main_loop

partial_group:
    movzx eax, byte ptr [r9]
    inc r9
    cmp r9, r14
    je one_byte

    movzx ecx, byte ptr [r9]
    inc r9

    mov r15d, eax
    shr r15d, 2
    mov al, byte ptr [r12 + r15]
    call emit_char

    and eax, 3
    shl eax, 4
    shr ecx, 4
    or al, cl
    mov al, byte ptr [r12 + rax]
    call emit_char

    and ecx, 0x0F
    shl ecx, 2
    mov al, byte ptr [r12 + rcx]
    call emit_char

    mov al, '='
    call emit_char
    jmp main_loop

one_byte:
    mov r15d, eax
    shr r15d, 2
    mov al, byte ptr [r12 + r15]
    call emit_char

    and eax, 3
    shl eax, 4
    mov al, byte ptr [r12 + rax]
    call emit_char

    mov al, '='
    call emit_char

    mov al, '='
    call emit_char
    jmp main_loop

emit_char:
    cmp r13, 76
    jne store_char
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, 76
    syscall
    mov rax, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    xor r13, r13
store_char:
    mov byte ptr [r10 + r13], al
    inc r13
    ret

final_line:
    test r13, r13
    jz exit_zero
    mov rax, 1
    mov rdi, 1
    mov rsi, r10
    mov rdx, r13
    syscall
    mov rax, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
