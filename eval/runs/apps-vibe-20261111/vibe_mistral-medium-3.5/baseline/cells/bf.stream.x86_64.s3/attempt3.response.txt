_start:

    mov rdx, 65536
read_loop:

    mov rax, 0
    syscall

    jz read_done
    add rsi, rax
    add r8, rax
    sub rdx, rax
    jg read_loop

    jz no_split

    mov rcx, r8
    mov al, '!'
    repne scasb

    dec r9
    jmp have_split
no_split:
    mov r9, r8
have_split:

    mov r12, r9
    cmp r9, r8
    jae no_input
    lea r14, [stdin_buf + r9 + 1]
    mov r15, r8
    sub r15, r9
    dec r15
    jmp setup
no_input:
    xor r14, r14
    xor r15, r15
setup:
    lea r11, [jump_table]
    lea rbx, [bracket_stack]
    xor rcx, rcx
    xor rdi, rdi
validate_loop:
    cmp rdi, r12
    jge validate_end
    movzx rax, byte ptr [r13 + rdi]
    cmp al, '['
    je open_bracket
    cmp al, ']'
    je close_bracket
    inc rdi
    jmp validate_loop
open_bracket:
    cmp rcx, 1000
    jge bracket_error
    mov [rbx + rcx*4], edi
    inc rcx
    inc rdi
    jmp validate_loop
close_bracket:
    test rcx, rcx
    jz bracket_error
    dec rcx
    mov r8d, [rbx + rcx*4]
    mov [r11 + rdi*4], r8d
    mov [r11 + r8*4], edi
    inc rdi
    jmp validate_loop
validate_end:
    test rcx, rcx
    jnz bracket_error
    xor r8, r8
    xor r9, r9
    xor r10, r10
    jmp exec_loop
bracket_error:
    mov rax, 60
    mov rdi, 1
    syscall

exec_loop:
    cmp r9, r12
    jge exec_done
    movzx rax, byte ptr [r13 + r9]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    inc r9
    jmp exec_loop
cmd_gt:
    inc r8
    inc r9
    jmp exec_loop
cmd_lt:
    dec r8
    inc r9
    jmp exec_loop
cmd_plus:
    inc byte ptr [tape + r8]
    inc r9
    jmp exec_loop
cmd_minus:
    dec byte ptr [tape + r8]
    inc r9
    jmp exec_loop
cmd_dot:
    mov rax, 1
    mov rdi, 1
    lea rsi, [tape + r8]
    mov rdx, 1
    syscall
    inc r9
    jmp exec_loop
cmd_comma:
    cmp r10, r15
    jae comma_eof
    mov al, byte ptr [r14 + r10]
    mov byte ptr [tape + r8], al
    inc r10
    jmp comma_done
comma_eof:
    mov byte ptr [tape + r8], 0
comma_done:
    inc r9
    jmp exec_loop
cmd_open:

    test al, al
    jnz open_skip
    mov r9d, [r11 + r9*4]
    inc r9
    jmp exec_loop
open_skip:
    inc r9
    jmp exec_loop
cmd_close:

    test al, al
    jz close_skip
    mov r9d, [r11 + r9*4]
    inc r9
    jmp exec_loop
close_skip:
    inc r9
    jmp exec_loop

    mov rax, 60
    xor rdi, rdi
    syscall
