.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

_start:
  adrp x19, input_buffer
  add x19, x19, :lo12:input_buffer
  mov x20, #0
  mov x21, #65536

read_loop:
  mov x0, #0
  add x1, x19, x20
  sub x2, x21, x20
  mov x8, #SYS_read
  svc #0
  cmp x0, #0
  b.le read_done
  add x20, x20, x0
  cmp x20, x21
  b.lt read_loop

read_done:
  add x22, x19, x20
  mov x23, x19

line_loop:
  cmp x23, x22
  b.ge line_done

  mov x24, x23
  mov x25, x23
find_nl:
  cmp x25, x22
  b.ge no_nl
  ldrb w26, [x25]
  cmp w26, #0x0A
  b.eq found_nl
  add x25, x25, #1
  b find_nl

found_nl:
  sub x27, x25, x24
  mov x0, x24
  mov x1, x27
  bl process_line
  add x23, x25, #1
  b line_loop

no_nl:
  sub x27, x22, x24
  cmp x27, #0
  b.eq line_done
  mov x0, x24
  mov x1, x27
  bl process_line
  mov x23, x22
  b line_loop

line_done:
  mov x0, #0
  mov x8, #SYS_exit
  svc #0

process_line:
  sub sp, sp, #112
  stp x19, x20, [sp, #0]
  stp x21, x22, [sp, #16]
  stp x23, x24, [sp, #32]
  stp x25, x26, [sp, #48]
  stp x27, x28, [sp, #64]

  mov x2, x0
  add x3, x0, x1
  adrp x28, rpn_stack
  add x28, x28, :lo12:rpn_stack
  mov x7, #0

  mov x4, x2

token_loop:
  cmp x4, x3
  b.ge token_done

skip_sep:
  cmp x4, x3
  b.ge token_done
  ldrb w5, [x4]
  cmp w5, #0x20
  b.eq sep_found
  cmp w5, #0x09
  b.eq sep_found
  b no_sep
sep_found:
  add x4, x4, #1
  b skip_sep
no_sep:

  cmp x4, x3
  b.ge token_done
  mov x6, x4

token_scan:
  cmp x4, x3
  b.ge token_end
  ldrb w5, [x4]
  cmp w5, #0x20
  b.eq token_end
  cmp w5, #0x09
  b.eq token_end
  add x4, x4, #1
  b token_scan
token_end:
  sub x5, x4, x6

  cmp x5, #1
  b.gt not_len1

  ldrb w8, [x6]
  cmp w8, #'+'
  b.eq handle_plus
  cmp w8, #'-'
  b.eq handle_minus
  cmp w8, #'*'
  b.eq handle_mul
  cmp w8, '/'
  b.eq handle_div
  cmp w8, '%'
  b.eq handle_mod
  b invalid_token

not_len1:
  cmp x5, #19
  b.hi invalid_token

  ldrb w8, [x6]
  cmp w8, #'-'
  b.ne check_digits_no_minus

  cmp x5, #1
  b.eq invalid_token
  sub x9, x5, #1
  cmp x9, #18
  b.hi invalid_token
  add x10, x6, #1
  mov x11, x10
  add x12, x10, x9
check_digits_minus:
  cmp x11, x12
  b.ge digit_check_done
  ldrb w13, [x11]
  cmp w13, #'0'
  b.lt invalid_token
  cmp w13, #'9'
  b.gt invalid_token
  add x11, x11, #1
  b check_digits_minus
digit_check_done:
  b parse_literal

check_digits_no_minus:
  cmp x5, #18
  b.hi invalid_token
  mov x10, x6
  mov x11, x6
  add x12, x6, x5
check_digits_loop:
  cmp x11, x12
  b.ge digit_check_done2
  ldrb w13, [x11]
  cmp w13, #'0'
  b.lt invalid_token
  cmp w13, #'9'
  b.gt invalid_token
  add x11, x11, #1
  b check_digits_loop
digit_check_done2:
  b parse_literal

invalid_token:
  mov x7, #1
  b token_loop_end

parse_literal:
  ldrb w8, [x6]
  cmp w8, #'-'
  b.ne positive_literal
  mov x14, #1
  add x10, x6, #1
  mov x11, x5
  sub x11, x11, #1
  b start_parse
positive_literal:
  mov x14, #0
  mov x10, x6
  mov x11, x5
start_parse:
  mov x13, #0
  cmp x11, #0
  b.eq literal_done
parse_digit_loop:
  ldrb w15, [x10]
  sub w15, w15, #'0'
  mul x13, x13, #10
  add x13, x13, x15
  add x10, x10, #1
  subs x11, x11, #1
  b.gt parse_digit_loop
literal_done:
  cmp x14, #0
  b.eq push_literal
  neg x13, x13
push_literal:
  str x13, [x28], #8
  b token_loop

handle_plus:
  adrp x15, rpn_stack
  add x15, x15, :lo12:rpn_stack
  sub x16, x28, x15
  cmp x16, #16
  b.lt op_error
  ldr x17, [x28, #-8]!
  ldr x18, [x28, #-8]!
  add x16, x18, x17
  str x16, [x28], #8
  b token_loop

handle_minus:
  adrp x15, rpn_stack
  add x15, x15, :lo12:rpn_stack
  sub x16, x28, x15
  cmp x16, #16
  b.lt op_error
  ldr x17, [x28, #-8]!
  ldr x18, [x28, #-8]!
  sub x16, x18, x17
  str x16, [x28], #8
  b token_loop

handle_mul:
  adrp x15, rpn_stack
  add x15, x15, :lo12:rpn_stack
  sub x16, x28, x15
  cmp x16, #16
  b.lt op_error
  ldr x17, [x28, #-8]!
  ldr x18, [x28, #-8]!
  mul x16, x18, x17
  str x16, [x28], #8
  b token_loop

handle_div:
  adrp x15, rpn_stack
  add x15, x15, :lo12:rpn_stack
  sub x16, x28, x15
  cmp x16, #16
  b.lt op_error
  ldr x17, [x28, #-8]!
  ldr x18, [x28, #-8]!
  cmp x17, #0
  b.eq div_error
  mvn x15, xzr
  lsl x15, x15, #63
  cmp x18, x15
  b.ne div_ok
  cmp x17, #-1
  b.eq div_error
div_ok:
  sdiv x16, x18, x17
  str x16, [x28], #8
  b token_loop

handle_mod:
  adrp x15, rpn_stack
  add x15, x15, :lo12:rpn_stack
  sub x16, x28, x15
  cmp x16, #16
  b.lt op_error
  ldr x17, [x28, #-8]!
  ldr x18, [x28, #-8]!
  cmp x17, #0
  b.eq div_error
  mvn x15, xzr
  lsl x15, x15, #63
  cmp x18, x15
  b.ne mod_ok
  cmp x17, #-1
  b.eq div_error
mod_ok:
  sdiv x16, x18, x17
  mul x16, x16, x17
  sub x16, x18, x16
  str x16, [x28], #8
  b token_loop

div_error:
  mov x7, #1
  b token_loop_end

op_error:
  mov x7, #1

token_loop_end:
  cmp x7, #0
  b.ne output_error

token_done:
  adrp x15, rpn_stack
  add x15, x15, :lo12:rpn_stack
  sub x16, x28, x15
  cmp x16, #8
  b.ne output_error

  ldr x0, [x28, #-8]!
  bl int_to_string
  adrp x1, output_buf
  add x1, x1, :lo12:output_buf
  mov x2, x0
  mov x0, #1
  mov x8, #SYS_write
  svc #0
  b process_line_return

  adrp x1, error_msg
  add x1, x1, :lo12:error_msg
  mov x2, #6
  mov x0, #1
  mov x8, #SYS_write
  svc #0

  ldp x19, x20, [sp, #0]
  ldp x21, x22, [sp, #16]
  ldp x23, x24, [sp, #32]
  ldp x25, x26, [sp, #48]
  ldp x27, x28, [sp, #64]
  add sp, sp, #112

  b.ne not_zero
  adrp x1, output_buf
  add x1, x1, :lo12:output_buf
  mov w2, #'0'
  strb w2, [x1]
  mov w2, #0x0A
  strb w2, [x1, #1]
  mov x0, #2
  ret

  cmp x0, #0
  b.ge positive
  sub x1, xzr, x0
  mov x2, #1
  b convert

  mov x1, x0
  mov x2, #0

  adrp x3, output_buf
  add x3, x3, :lo12:output_buf
  add x4, x3, #20
  mov x5, x1

  b.eq digit_done
  udiv x6, x5, #10
  mul x7, x6, #10
  sub x7, x5, x7
  add w7, w7, #'0'
  strb w7, [x4, #-1]!

  b digit_loop

  mov w7, #'-'

  sub x0, x4, x3
  add x5, x3, x0
  mov w6, #0x0A
  strb w6, [x5]
  add x0, x0, #1
  ret
