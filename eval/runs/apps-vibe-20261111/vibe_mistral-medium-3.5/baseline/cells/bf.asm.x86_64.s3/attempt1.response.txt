.section .bss
.align 8
tape: .space 30000
input_buf: .space 65536
jump_table: .space 131072
bracket_stack: .space 2000

.globl _start
.text
_start:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    syscall
    test rax, rax
    js exit_error
    mov r8, rax

    lea rsi, [rip + input_buf]
    xor rcx, rcx
find_sep:
    cmp rcx, r8
    jge no_separator
    cmp byte ptr [rsi + rcx], '!'
    je found_sep
    inc rcx
    jmp find_sep
found_sep:
    mov r9, rcx
    jmp after_sep
no_separator:
    mov r9, r8
after_sep:
    lea r10, [rip + input_buf]
    lea r11, [r10 + r9]
    lea r12, [r10 + r9 + 1]
    lea r13, [r10 + r8]

    lea rbx, [rip + bracket_stack]
    xor r14, r14
    lea r15, [rip + jump_table]
    mov rsi, r10
    mov rcx, r11
    sub rcx, r10
check_brackets:
    test rcx, rcx
    jz check_brackets_done
    movzx edx, byte ptr [rsi]
    cmp dl, '['
    je is_open
    cmp dl, ']'
    je is_close
    inc rsi
    dec rcx
    jmp check_brackets
is_open:
    mov rax, rsi
    sub rax, r10
    mov [rbx + r14*2], ax
    inc r14
    cmp r14, 1000
    jg unbalanced
    inc rsi
    dec rcx
    jmp check_brackets
is_close:
    test r14, r14
    jz unbalanced
    dec r14
    movzx edx, word ptr [rbx + r14*2]
    mov rax, rsi
    sub rax, r10
    mov [r15 + rdx*2], ax
    inc word ptr [r15 + rdx*2]
    mov [r15 + rax*2], dx
    inc word ptr [r15 + rax*2]
    inc rsi
    dec rcx
    jmp check_brackets
check_brackets_done:
    test r14, r14
    jnz unbalanced

    lea r8, [rip + tape]
    mov r9, r10

interpreter_loop:
    cmp r9, r11
    jge exit_success
    movzx eax, byte ptr [r9]
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    inc r9
    jmp interpreter_loop

cmd_right:
    inc r8
    inc r9
    jmp interpreter_loop
cmd_left:
    dec r8
    inc r9
    jmp interpreter_loop
cmd_plus:
    movzx eax, byte ptr [r8]
    inc eax
    and eax, 0xff
    mov byte ptr [r8], al
    inc r9
    jmp interpreter_loop
cmd_minus:
    movzx eax, byte ptr [r8]
    dec eax
    and eax, 0xff
    mov byte ptr [r8], al
    inc r9
    jmp interpreter_loop
cmd_dot:
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    syscall
    inc r9
    jmp interpreter_loop
cmd_comma:
    cmp r12, r13
    jae comma_eof
    movzx eax, byte ptr [r12]
    mov byte ptr [r8], al
    inc r12
    jmp comma_done
comma_eof:
    mov byte ptr [r8], 0
comma_done:
    inc r9
    jmp interpreter_loop
cmd_open:
    movzx eax, byte ptr [r8]
    test al, al
    jnz open_not_zero
    mov rax, r9
    sub rax, r10
    movzx edx, word ptr [r15 + rax*2]
    lea r9, [r10 + rdx]
    jmp interpreter_loop
open_not_zero:
    inc r9
    jmp interpreter_loop
cmd_close:
    movzx eax, byte ptr [r8]
    test al, al
    jz close_zero
    mov rax, r9
    sub rax, r10
    movzx edx, word ptr [r15 + rax*2]
    lea r9, [r10 + rdx]
    jmp interpreter_loop
close_zero:
    inc r9
    jmp interpreter_loop

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
