```asm
.globl _start

.section .rodata
newline: .byte 0x0A

.equ INPUT_BUF_SIZE, 32768
.equ MAX_LINES, 1000

.bss
input_buf: .space INPUT_BUF_SIZE
line_starts: .space MAX_LINES * 8
line_lengths: .space MAX_LINES * 4

.text
_start:
    xor rbx, rbx
    lea rbp, [rip + input_buf]
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbp + rbx]
    mov rdx, INPUT_BUF_SIZE
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r13, [rip + line_starts]
    lea r14, [rip + line_lengths]
    xor r12, r12
    xor rsi, rsi
    test rbx, rbx
    jz print_lines
parse_loop:
    mov rdi, rsi
find_newline:
    cmp rdi, rbx
    jge end_of_input
    movzx eax, byte ptr [rbp + rdi]
    cmp al, 0x0A
    je found_newline
    inc rdi
    jmp find_newline
found_newline:
    mov rax, rbp
    add rax, rsi
    mov qword ptr [r13 + r12*8], rax
    mov ecx, edi
    sub ecx, esi
    mov dword ptr [r14 + r12*4], ecx
    inc r12
    mov rsi, rdi
    inc rsi
    jmp parse_loop
end_of_input:
    cmp rsi, rbx
    jge print_lines
    mov rax, rbp
    add rax, rsi
    mov qword ptr [r13 + r12*8], rax
    mov ecx, ebx
    sub ecx, esi
    mov dword ptr [r14 + r12*4], ecx
    inc r12
print_lines:
    test r12, r12
    jz exit
    xor r8, r8
sort_outer:
    mov r9, r8
    mov r10, r8
    inc r10
    cmp r10, r12
    jge sort_next_i
sort_inner:
    mov rax, qword ptr [r13 + r9*8]
    mov ecx, dword ptr [r14 + r9*4]
    mov rdx, qword ptr [r13 + r10*8]
    mov esi, dword ptr [r14 + r10*4]
    xor rdi, rdi
compare_loop:
    cmp edi, ecx
    jge compare_lengths
    cmp edi, esi
    jge compare_lengths
    movzx r11d, byte ptr [rax + rdi]
    movzx ebx, byte ptr [rdx + rdi]
    cmp r11d, ebx
    jl less
    jg greater
    inc rdi
    jmp compare_loop
compare_lengths:
    cmp ecx, esi
    jl less
    jg greater
equal:
    inc r10
    cmp r10, r12
    jl sort_inner
    jmp sort_swap
greater:
    mov r9, r10
    jmp equal
less:
    inc r10
    cmp r10, r12
    jl sort_inner
sort_swap:
    cmp r9, r8
    je sort_next_i
    mov rax, qword ptr [r13 + r8*8]
    mov rbx, qword ptr [r13 + r9*8]
    mov qword ptr [r13 + r8*8], rbx
    mov qword ptr [r13 + r9*8], rax
    mov eax, dword ptr [r14 + r8*4]
    mov ebx, dword ptr [r14 + r9*4]
    mov dword ptr [r14 + r8*4], ebx
    mov dword ptr [r14 + r9*4], eax
sort_next_i:
    inc r8
    cmp r8, r12
    jl sort_outer
    xor r8, r8
print_loop:
    cmp r8, r12
    jge exit
    mov rsi, qword ptr [r13 + r8*8]
    mov edx, dword ptr [r14 + r8*4]
    test edx, edx
    jz write_newline
    mov rdi, 1
    mov rax, 1
    syscall
write_newline:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r8
    jmp print_loop
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
```
