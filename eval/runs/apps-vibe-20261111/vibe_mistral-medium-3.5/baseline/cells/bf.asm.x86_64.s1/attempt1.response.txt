```asm
.globl _start

.text
_start:
    xor r8, r8
.Lread_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buffer + r8]
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz .Lread_done
    add r8, rax
    cmp r8, 65536
    jb .Lread_loop
.Lread_done:

    xor r9, r9
    mov r10, r8
    xor r11, r11
    xor r12, r12
.Lsplit_loop:
    cmp r9, r10
    je .Lsplit_not_found
    mov al, byte ptr [input_buffer + r9]
    cmp al, '!'
    je .Lsplit_found
    inc r9
    jmp .Lsplit_loop
.Lsplit_not_found:
    mov r11, r8
    xor r12, r12
    jmp .Lsplit_done
.Lsplit_found:
    mov r11, r9
    mov r12, r8
    sub r12, r9
    dec r12
.Lsplit_done:

    xor rsi, rsi
    xor rdi, rdi
.Lcopy_program_loop:
    cmp rsi, r11
    je .Lcopy_program_done
    mov al, byte ptr [input_buffer + rsi]
    mov byte ptr [program + rdi], al
    inc rsi
    inc rdi
    jmp .Lcopy_program_loop
.Lcopy_program_done:

    lea rsi, [input_buffer + r9 + 1]
    xor rdi, rdi
.Lcopy_bf_input_loop:
    cmp rdi, r12
    je .Lcopy_bf_input_done
    mov al, byte ptr [rsi + rdi]
    mov byte ptr [bf_input + rdi], al
    inc rdi
    jmp .Lcopy_bf_input_loop
.Lcopy_bf_input_done:

    mov r15, rsp
    xor r13, r13
.Lbracket_scan_loop:
    cmp r13, r11
    je .Lbracket_check_stack
    mov al, byte ptr [program + r13]
    cmp al, '['
    je .Lbracket_push
    cmp al, ']'
    je .Lbracket_pop
    inc r13
    jmp .Lbracket_scan_loop
.Lbracket_push:
    push r13
    inc r13
    jmp .Lbracket_scan_loop
.Lbracket_pop:
    cmp rsp, r15
    je .Lunbalanced
    pop r14
    mov dword ptr [jump_table + r13 * 4], r14d
    mov dword ptr [jump_table + r14 * 4], r13d
    inc r13
    jmp .Lbracket_scan_loop
.Lbracket_check_stack:
    cmp rsp, r15
    jne .Lunbalanced

    lea r13, [rip + tape]
    lea r14, [rip + program]
    lea r15, [rip + jump_table]
    xor r8, r8
    xor r9, r9
    xor r10, r10
.Lexecute_loop:
    cmp r9, r11
    je .Lexecute_done
    mov al, byte ptr [r14 + r9]
    cmp al, '>'
    je .Lcmd_gt
    cmp al, '<'
    je .Lcmd_lt
    cmp al, '+'
    je .Lcmd_plus
    cmp al, '-'
    je .Lcmd_minus
    cmp al, '.'
    je .Lcmd_dot
    cmp al, ','
    je .Lcmd_comma
    cmp al, '['
    je .Lcmd_lb
    cmp al, ']'
    je .Lcmd_rb
    inc r9
    jmp .Lexecute_loop
.Lcmd_gt:
    inc r8
    inc r9
    jmp .Lexecute_loop
.Lcmd_lt:
    dec r8
    inc r9
    jmp .Lexecute_loop
.Lcmd_plus:
    movzx eax, byte ptr [r13 + r8]
    inc eax
    and eax, 0xff
    mov byte ptr [r13 + r8], al
    inc r9
    jmp .Lexecute_loop
.Lcmd_minus:
    movzx eax, byte ptr [r13 + r8]
    dec eax
    and eax, 0xff
    mov byte ptr [r13 + r8], al
    inc r9
    jmp .Lexecute_loop
.Lcmd_dot:
    mov rdi, 1
    lea rsi, [r13 + r8]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r9
    jmp .Lexecute_loop
.Lcmd_comma:
    cmp r10, r12
    jae .Linput_exhausted
    mov al, byte ptr [bf_input + r10]
    inc r10
    mov byte ptr [r13 + r8], al
    inc r9
    jmp .Lexecute_loop
.Linput_exhausted:
    mov byte ptr [r13 + r8], 0
    inc r9
    jmp .Lexecute_loop
.Lcmd_lb:
    movzx eax, byte ptr [r13 + r8]
    test eax, eax
    jnz .Lcmd_lb_skip
    mov r9d, dword ptr [r15 + r9 * 4]
    inc r9
    jmp .Lexecute_loop
.Lcmd_lb_skip:
    inc r9
    jmp .Lexecute_loop
.Lcmd_rb:
    movzx eax, byte ptr [r13 + r8]
    test eax, eax
    jz .Lcmd_rb_skip
    mov r9d, dword ptr [r15 + r9 * 4]
    inc r9
    jmp .Lexecute_loop
.Lcmd_rb_skip:
    inc r9
    jmp .Lexecute_loop
.Lexecute_done:
    xor rdi, rdi
    mov rax, 60
    syscall
.Lunbalanced:
    mov rdi, 1
    mov rax, 60
    syscall

.section .bss
input_buffer:
    .space 65536
program:
    .space 65536
bf_input:
    .space 65536
tape:
    .space 30000
jump_table:
    .space 65536 * 4
```
