.globl _start
_start:
    sub rsp, 65536
    mov r10, rsp
    mov r12, offset buffer
    xor r13, r13

read_loop:
    mov rdi, 0
    lea rsi, [r12 + r13]
    mov rdx, 65536
    sub rdx, r13
    jle end_of_read
    mov rax, 0
    syscall
    test rax, rax
    jz end_of_read
    js exit_error
    add r13, rax
    jmp read_loop

end_of_read:
    xor r14, r14
    xor r15, r15

process_loop:
    cmp r15, r13
    je check_last_line
    mov al, byte ptr [r12 + r15]
    cmp al, 0x0A
    jne not_newline
    mov r8, r15
    sub r8, r14
    jz write_newline
    lea rsi, [r12 + r15 - 1]
    mov rdi, r10
    mov rcx, r8
copy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    loop copy_loop
    mov rdi, 1
    mov rsi, r10
    mov rdx, r8
    mov rax, 1
    syscall
write_newline:
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    mov r14, r15
    inc r14
    inc r15
    jmp process_loop

not_newline:
    inc r15
    jmp process_loop

check_last_line:
    cmp r14, r13
    je done
    mov r8, r13
    sub r8, r14
    jz done
    lea rsi, [r12 + r13 - 1]
    mov rdi, r10
    mov rcx, r8
copy_last_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    loop copy_last_loop
    mov rdi, 1
    mov rsi, r10
    mov rdx, r8
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall

done:
    add rsp, 65536
    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall

.section .rodata
newline: .byte 0x0A

.section .bss
buffer: .space 65536
