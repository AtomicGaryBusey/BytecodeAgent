```assembly
.section .rodata
alphabet:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.p2align 2
input_buf:	.space 4096
output_buf:	.space 4096
output_pos:	.space 4
line_pos:	.space 4
shift_reg:	.space 4
shift_count:	.space 4

.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ BUF_SIZE, 4096
.equ LINE_LEN, 76

_start:
	adrp x0, output_pos
	add x0, x0, :lo12:output_pos
	str wzr, [x0]
	adrp x0, line_pos
	add x0, x0, :lo12:line_pos
	str wzr, [x0]
	adrp x0, shift_reg
	add x0, x0, :lo12:shift_reg
	str wzr, [x0]
	adrp x0, shift_count
	add x0, x0, :lo12:shift_count
	str wzr, [x0]

read_loop:
	mov x0, #STDIN
	adrp x1, input_buf
	add x1, x1, :lo12:input_buf
	mov x2, #BUF_SIZE
	mov x8, #SYS_READ
	svc #0
	cbz x0, finish_input
	mov x3, x0
	mov x4, #0
process_byte:
	cmp x4, x3
	b.ge read_loop
	adrp x5, input_buf
	add x5, x5, :lo12:input_buf
	ldrb w6, [x5, x4]
	add x4, x4, #1
	adrp x7, shift_reg
	add x7, x7, :lo12:shift_reg
	ldr w8, [x7]
	adrp x9, shift_count
	add x9, x9, :lo12:shift_count
	ldr w10, [x9]
	lsl w8, w8, #8
	orr w8, w8, w6
	str w8, [x7]
	add w10, w10, #1
	str w10, [x9]
	cmp w10, #3
	b.ne process_byte
	lsr w0, w8, #18
	bl output_b64_char
	lsr w0, w8, #12
	and w0, w0, #63
	bl output_b64_char
	lsr w0, w8, #6
	and w0, w0, #63
	bl output_b64_char
	and w0, w8, #63
	bl output_b64_char
	str wzr, [x7]
	str wzr, [x9]
	b process_byte

finish_input:
	adrp x9, shift_count
	add x9, x9, :lo12:shift_count
	ldr w10, [x9]
	cbz w10, flush_and_exit
	adrp x7, shift_reg
	add x7, x7, :lo12:shift_reg
	ldr w8, [x7]
	cmp w10, #1
	b.eq one_byte_left
two_bytes_left:
	lsr w0, w8, #10
	bl output_b64_char
	lsr w0, w8, #4
	and w0, w0, #63
	bl output_b64_char
	and w0, w8, #15
	lsl w0, w0, #2
	bl output_b64_char
	mov w0, #64
	bl output_padding
	b flush_and_exit
one_byte_left:
	lsr w0, w8, #2
	bl output_b64_char
	and w0, w8, #3
	lsl w0, w0, #4
	bl output_b64_char
	mov w0, #64
	bl output_padding
	mov w0, #64
	bl output_padding

flush_and_exit:
	adrp x0, line_pos
	add x0, x0, :lo12:line_pos
	ldr w1, [x0]
	cbz w1, flush_output
	mov w0, #0x0A
	bl output_char
flush_output:
	adrp x0, output_pos
	add x0, x0, :lo12:output_pos
	ldr w1, [x0]
	cbz w1, exit
	mov x0, #STDOUT
	adrp x1, output_buf
	add x1, x1, :lo12:output_buf
	mov x2, w1
	mov x8, #SYS_WRITE
	svc #0
exit:
	mov x0, #0
	mov x8, #SYS_EXIT
	svc #0

output_b64_char:
	adrp x1, alphabet
	add x1, x1, :lo12:alphabet
	ldrb w0, [x1, w0, uxtw]
	b output_char

output_padding:
	mov w0, #0x3D
	b output_char

output_char:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	adrp x1, output_buf
	add x1, x1, :lo12:output_buf
	adrp x2, output_pos
	add x2, x2, :lo12:output_pos
	ldr w3, [x2]
	strb w0, [x1, w3, uxtw]
	add w3, w3, #1
	str w3, [x2]
	cmp w3, #BUF_SIZE
	b.ne check_line
	mov x0, #STDOUT
	adrp x1, output_buf
	add x1, x1, :lo12:output_buf
	mov x2, #BUF_SIZE
	mov x8, #SYS_WRITE
	svc #0
	str wzr, [x2]
check_line:
	adrp x4, line_pos
	add x4, x4, :lo12:line_pos
	ldr w5, [x4]
	cmp w0, #0x0A
	b.eq newline_reset
	add w5, w5, #1
	str w5, [x4]
	cmp w5, #LINE_LEN
	b.ne check_line_done
	mov w0, #0x0A
	bl output_char
	str wzr, [x4]
	b check_line_done
newline_reset:
	str wzr, [x4]
check_line_done:
	ldp x29, x30, [sp], #32
	ret
```
