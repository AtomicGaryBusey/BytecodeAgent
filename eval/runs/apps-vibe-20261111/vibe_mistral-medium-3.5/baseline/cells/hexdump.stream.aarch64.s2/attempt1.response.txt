.text

  add x16, x16, :lo12:input_buffer

  add x1, x16, x17
  mov x2, #65536

  mov w8, #63

  adrp x27, hex_table
  add x27, x27, :lo12:hex_table
  mov x18, #0

  cmp x18, x17
  b.ge print_total
  sub x19, x17, x18
  cmp x19, #16
  mov x20, #16
  csel x20, x20, x19, hi

  mov x22, x18
  mov x23, #28

  and x25, x25, #0xf
  ldrb w25, [x27, x25]
  strb w25, [x21], #1
  sub x23, x23, #4
  subs x24, x24, #1
  b.ne print_offset_loop
  mov w25, #' '
  strb w25, [x21], #1
  strb w25, [x21], #1
  mov x23, #0
  add x24, x16, x18
row_hex_loop:

  b.lt 1f
  mov w25, #' '
  strb w25, [x21], #1
  strb w25, [x21], #1
  strb w25, [x21], #1
  b 2f

  lsr w26, w25, #4
  and w26, w26, #0xf
  ldrb w26, [x27, w26]
  strb w26, [x21], #1
  and w25, w25, #0xf
  ldrb w25, [x27, w25]
  strb w25, [x21], #1
  mov w25, #' '
  strb w25, [x21], #1

  b.ne 3f
  mov w25, #' '
  strb w25, [x21], #1

  cmp x23, #16
  b.lt row_hex_loop
  mov w25, #' '
  strb w25, [x21], #1
  mov w25, #'|'
  strb w25, [x21], #1
  mov x23, #0
ascii_loop:

  b.ge ascii_done
  ldrb w25, [x24, x23]
  cmp w25, #0x20
  b.lt 1f
  cmp w25, #0x7e

  strb w25, [x21], #1

  strb w25, [x21], #1
  mov w25, #'\n'
  strb w25, [x21], #1
  sub x22, x21, sp
  mov w0, #1
  mov x1, sp
  mov x2, x22
  mov w8, #64

  add sp, sp, #80
  add x18, x18, #16
  b process_loop

  mov x21, sp
  mov x22, x17
  mov x23, #28
  mov x24, #8

  lsr x25, x22, x23
  and x25, x25, #0xf

  sub x23, x23, #4

  mov x1, sp
  mov x2, x22
