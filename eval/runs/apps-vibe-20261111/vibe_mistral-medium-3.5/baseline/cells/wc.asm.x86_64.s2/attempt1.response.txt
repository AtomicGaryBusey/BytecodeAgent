.section .bss
buffer: .space 65536

.text
.globl _start
_start:
	xor rbx, rbx
	xor rbp, rbp
	xor r12, r12
	xor r13, r13

read_loop:
	mov rax, 0
	mov rdi, 0
	lea rsi, [rip + buffer]
	mov rdx, 65536
	syscall
	test rax, rax
	jz done_reading
	mov r14, rsi
	mov r15, rax

process_byte:
	movzx eax, byte ptr [r14]
	inc rbx
	inc r14
	cmp al, 0x0A
	jne not_newline
	inc rbp
not_newline:
	cmp al, 0x20
	je is_ws
	cmp al, 0x09
	je is_ws
	cmp al, 0x0A
	je is_ws
	cmp al, 0x0B
	je is_ws
	cmp al, 0x0C
	je is_ws
	cmp al, 0x0D
	je is_ws
	jmp not_ws
is_ws:
	xor r13, r13
	jmp next_byte
not_ws:
	test r13, r13
	jnz next_byte
	inc r12
	mov r13, 1
next_byte:
	dec r15
	jnz process_byte
	jmp read_loop

done_reading:
	sub rsp, 60
	lea r14, [rsp + 40]

	mov rax, rbp
	lea rsi, [rsp]
	add rsi, 19
	mov byte ptr [rsi], 0
	dec rsi
	xor r8, r8
	mov rcx, 10
	test rax, rax
	jz lines_zero
lines_loop:
	xor rdx, rdx
	div rcx
	add dl, '0'
	mov [rsi], dl
	dec rsi
	inc r8
	test rax, rax
	jnz lines_loop
	jmp lines_done
lines_zero:
	mov byte ptr [rsi], '0'
	dec rsi
	mov r8, 1
lines_done:
	inc rsi
	mov rcx, r8
	mov rdi, r14
	rep movsb
	mov r14, rdi
	mov byte ptr [r14], ' '
	inc r14

	mov rax, r12
	lea rsi, [rsp]
	add rsi, 19
	mov byte ptr [rsi], 0
	dec rsi
	xor r8, r8
	mov rcx, 10
	test rax, rax
	jz words_zero
words_loop:
	xor rdx, rdx
	div rcx
	add dl, '0'
	mov [rsi], dl
	dec rsi
	inc r8
	test rax, rax
	jnz words_loop
	jmp words_done
words_zero:
	mov byte ptr [rsi], '0'
	dec rsi
	mov r8, 1
words_done:
	inc rsi
	mov rcx, r8
	mov rdi, r14
	rep movsb
	mov r14, rdi
	mov byte ptr [r14], ' '
	inc r14

	mov rax, rbx
	lea rsi, [rsp]
	add rsi, 19
	mov byte ptr [rsi], 0
	dec rsi
	xor r8, r8
	mov rcx, 10
	test rax, rax
	jz bytes_zero
bytes_loop:
	xor rdx, rdx
	div rcx
	add dl, '0'
	mov [rsi], dl
	dec rsi
	inc r8
	test rax, rax
	jnz bytes_loop
	jmp bytes_done
bytes_zero:
	mov byte ptr [rsi], '0'
	dec rsi
	mov r8, 1
bytes_done:
	inc rsi
	mov rcx, r8
	mov rdi, r14
	rep movsb
	mov r14, rdi
	mov byte ptr [r14], 0x0A
	inc r14

	mov rdi, 1
	lea rsi, [rsp + 40]
	mov rdx, r14
	sub rdx, rsp
	sub rdx, 40

	mov rax, 60
	xor rdi, rdi
	syscall
