// BUFFER_SIZE = 65536

.text
.globl _start
_start:
	adrp x19, buffer
	add x19, x19, :lo12:buffer
	movz x21, #1, lsl #16
	mov x20, #0

read_loop:
	mov x0, #0
	add x1, x19, x20
	sub x2, x21, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	beq read_done
	add x20, x20, x0
	b read_loop

read_done:
	mov x21, #0
	mov x22, #0
process_loop:
	cmp x22, x20
	beq check_final
	ldrb w0, [x19, x22]
	cmp w0, #10
	bne next_byte
	mov x0, x19
	mov x1, x21
	sub x2, x22, #1
	bl reverse
	mov x0, #1
	add x1, x19, x21
	sub x2, x22, x21
	mov x8, #64
	svc #0
	mov x0, #1
	adr x1, newline
	mov x2, #1
	mov x8, #64
	svc #0
	add x21, x22, #1
next_byte:
	add x22, x22, #1
	b process_loop

check_final:
	cmp x21, x20
	beq exit
	mov x0, x19
	mov x1, x21
	sub x2, x20, #1
	bl reverse
	mov x0, #1
	add x1, x19, x21
	sub x2, x20, x21
	mov x8, #64
	svc #0
	mov x0, #1
	adr x1, newline
	mov x2, #1
	mov x8, #64
	svc #0

exit:
	mov x0, #0
	mov x8, #93
	svc #0

reverse:
	cmp x1, x2
	bge rev_done
rev_loop:
	ldrb w3, [x0, x1]
	ldrb w4, [x0, x2]
	strb w4, [x0, x1]
	strb w3, [x0, x2]
	add x1, x1, #1
	sub x2, x2, #1
	cmp x1, x2
	blt rev_loop
rev_done:
	ret

.section .rodata
newline: .byte 10

.bss
.align 4
buffer: .space 65536
