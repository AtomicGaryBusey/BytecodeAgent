_start:
  // Read all of stdin into input_buf
  adr x0, input_buf
  mov x1, xzr
  mov x2, #65536

1:
  mov x8, #63
  mov x0, xzr
  add x1, x1, x0
  cmp x1, #65536
  bge 1f
  mov x0, x1
  add x0, x0, x2
  sub x0, x2, x1
  mov x1, x2
  add x1, x1, x0
  mov x2, x0
  svc #0
  cmp x0, #0
  ble 1f
  b 1b
1:

  // Set up for parsing: x0 = buffer, x1 = 0, x2 = length
  adr x0, input_buf
  mov x1, xzr
  mov x2, x3
  bl parse_document

  cmp x0, #1
  beq 2f

  // Error case: x1 = error offset
  mov x0, x1
  adr x1, error_msg
  mov x2, #9
  mov x8, #64
  svc #0

  // Convert error offset (x0) to decimal string
  sub sp, sp, #6
  mov x3, sp
  add x3, x3, #5
  cmp x0, xzr
  bne 3f
  strb #0x30, [x3, #-1]
  mov x3, sp
  add x3, x3, #5
  sub x3, x3, #1
  mov x4, #1
  b 4f
3:
  mov x4, #10
  mov x5, xzr
5:
  udiv x6, x0, x4
  msub x7, x6, x4, x0
  add w7, w7, #0x30
  strb w7, [x3], #-1
  add x5, x5, #1
  mov x0, x6
  cbnz x0, 5b
  add x3, x3, #1
  mov x4, x5
4:
  mov x0, #1
  mov x1, x3
  mov x2, x4
  mov x8, #64
  svc #0

  adr x1, newline
  mov x2, #1
  mov x0, #1
  mov x8, #64
  svc #0

  mov x0, #1
  mov x8, #93
  svc #0

2:
  // Valid case
  adr x1, ok_msg
  mov x2, #3
  mov x0, #1
  mov x8, #64
  svc #0

  mov x0, xzr
  mov x8, #93
  svc #0

// ---- Parser functions ----

parse_document:
  stp x29, x30, [sp, #-16]!
  stp x1, x2, [sp, #-16]!
  mov x29, sp

  bl parse_ws
  cmp x0, #0
  bne 1f
  mov x0, xzr
  ldr x1, [x29, #16]
  b 9f
1:
  mov x3, x1
  bl parse_value
  cmp x0, #0
  beq 9f
  mov x3, x1
  bl parse_ws
  cmp x0, #0
  beq 9f
  ldr x2, [x29, #24]
  cmp x1, x2
  bne 9f
  mov x0, #1
  mov x1, x1
  b 9f
9:
  ldp x1, x2, [x29, #16]
  ldp x29, x30, [sp], #32
  ret

parse_ws:
  stp x29, x30, [sp, #-16]!
  mov x29, sp

1:
  cmp x1, x2
  bge 2f
  ldrb w3, [x0, x1]
  cmp w3, #0x20
  beq 3f
  cmp w3, #0x09
  beq 3f
  cmp w3, #0x0A
  beq 3f
  cmp w3, #0x0D
  beq 3f
  b 2f
3:
  add x1, x1, #1
  b 1b
2:
  mov x0, #1
  ldp x29, x30, [sp], #16
  ret

parse_value:
  stp x29, x30, [sp, #-16]!
  mov x29, sp

  cmp x1, x2
  bge 9f
  ldrb w3, [x0, x1]
  cmp w3, '#{'
  beq parse_value_object
  cmp w3, '#['
  beq parse_value_array
  cmp w3, '#"'
  beq parse_value_string
  cmp w3, '#-'
  beq parse_value_number
  cmp w3, '#t'
  beq parse_value_true
  cmp w3, '#f'
  beq parse_value_false
  cmp w3, '#n'
  beq parse_value_null
  cmp w3, '#0'
  blt 9f
  cmp w3, '#9'
  bgt 9f
  b parse_value_number

parse_value_object:
  bl parse_object
  b 9f
parse_value_array:
  bl parse_array
  b 9f
parse_value_string:
  bl parse_string
  b 9f
parse_value_number:
  bl parse_number
  b 9f
parse_value_true:

parse_value_false:

parse_value_null:

9:
  ldp x29, x30, [sp], #16
  ret

  beq 9f

  bl parse_value
  cmp x0, #0
  beq 9f
  bl parse_ws
  cmp x0, #0
  beq 9f
  cmp x1, x2
  bge 9f
  ldrb w3, [x0, x1]
  cmp w3, '#,'
  bne 2f
  add x1, x1, #1
  bl parse_ws
  cmp x0, #0
  beq 9f
  b 1b
2:
  cmp w3, '#}'
  bne 9f
8:
  add x1, x1, #1
