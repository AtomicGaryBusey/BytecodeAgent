.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input_buf: .space 49152
output_buf: .space 66400

.text
_start:
    mov rdi, 0
    lea rsi, [input_buf]
    mov rdx, 49152
    xor r8, r8

read_loop:
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add rsi, rax
    sub rdx, rax
    jnz read_loop

read_done:
    test r8, r8
    jz exit

    lea rsi, [input_buf]
    mov rcx, r8
    lea rdi, [output_buf]
    xor r9, r9
    xor r10, r10
    lea r11, [alphabet]

main_loop:
    cmp rcx, 3
    jb handle_remainder

    movzx r12, byte ptr [rsi]
    movzx r13, byte ptr [rsi+1]
    movzx r14, byte ptr [rsi+2]

    mov rax, r12
    shr rax, 2
    mov al, byte ptr [r11 + rax]
    call write_char

    mov rax, r12
    and rax, 3
    shl rax, 4
    mov rbx, r13
    shr rbx, 4
    or rax, rbx
    mov al, byte ptr [r11 + rax]
    call write_char

    mov rax, r13
    and rax, 0xF
    shl rax, 2
    mov rbx, r14
    shr rbx, 6
    or rax, rbx
    mov al, byte ptr [r11 + rax]
    call write_char

    mov rax, r14
    and rax, 0x3F
    mov al, byte ptr [r11 + rax]
    call write_char

    add rsi, 3
    sub rcx, 3
    jmp main_loop

handle_remainder:
    test rcx, rcx
    jz finish

    cmp rcx, 1
    je one_byte

    movzx r12, byte ptr [rsi]
    movzx r13, byte ptr [rsi+1]

    mov rax, r12
    shr rax, 2
    mov al, byte ptr [r11 + rax]
    call write_char

    mov rax, r12
    and rax, 3
    shl rax, 4
    mov rbx, r13
    shr rbx, 4
    or rax, rbx
    mov al, byte ptr [r11 + rax]
    call write_char

    mov rax, r13
    and rax, 0xF
    shl rax, 2
    mov al, byte ptr [r11 + rax]
    call write_char

    mov al, '='
    call write_char
    jmp finish

one_byte:
    movzx r12, byte ptr [rsi]

    mov rax, r12
    shr rax, 2
    mov al, byte ptr [r11 + rax]
    call write_char

    mov rax, r12
    and rax, 3
    shl rax, 4
    mov al, byte ptr [r11 + rax]
    call write_char

    mov al, '='
    call write_char
    call write_char

finish:
    test r9, r9
    jz write_output
    mov byte ptr [rdi], 0x0A
    inc rdi
    inc r10
    xor r9, r9

write_output:
    mov rdi, 1
    lea rsi, [output_buf]
    mov rdx, r10
    mov rax, 1
    syscall

exit:
    mov rax, 60
    xor rdi, rdi
    syscall

write_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r9
    inc r10
    cmp r9, 76
    jne write_char_done
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9, r9
    inc r10
write_char_done:
    ret
