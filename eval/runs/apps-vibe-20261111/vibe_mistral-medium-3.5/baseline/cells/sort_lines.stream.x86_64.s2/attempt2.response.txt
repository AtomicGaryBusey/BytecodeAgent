```asm
.globl _start

.section .rodata
newline: .byte 0x0A

.section .bss
input_buffer: .space 32768
line_starts: .space 8000
line_lengths: .space 4000
line_count: .space 4

.text
_start:
    xor r8, r8
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buffer + r8]
    mov rdx, 32768
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:

    xor rcx, rcx
    xor r9, r9
    lea r10, [input_buffer]
    lea r11, [line_starts]
    lea r12, [line_lengths]
parse_loop:
    cmp rcx, r8
    jge parse_done
    mov al, byte ptr [input_buffer + rcx]
    cmp al, 0x0A
    jne parse_no_newline
    lea rax, [input_buffer + rcx]
    sub rax, r10
    mov [r11], r10
    mov [r12], eax
    add r11, 8
    add r12, 4
    inc r9
    lea r10, [input_buffer + rcx + 1]
parse_no_newline:
    inc rcx
    jmp parse_loop
parse_done:
    lea rax, [input_buffer + r8]
    cmp r10, rax
    jge parse_final_done
    lea rax, [input_buffer + r8]
    sub rax, r10
    mov [r11], r10
    mov [r12], eax
    inc r9
parse_final_done:
    mov [line_count], r9d

    mov ecx, [line_count]
    dec ecx
    jle sort_done
    mov r15d, ecx
    xor r12, r12
bubble_outer:
    xor r13, r13
    xor r14, r14
bubble_inner:
    mov rdi, [line_starts + r14*8]
    mov esi, [line_lengths + r14*4]
    mov rbx, [line_starts + r14*8 + 8]
    mov edx, [line_lengths + r14*4 + 4]
    xor r8, r8
    xor r9, r9
    mov r10d, esi
    mov r11d, edx
    cmp r10d, r11d
    cmovg r10d, r11d
    test r10d, r10d
    jz compare_lengths
compare_bytes:
    movzx r8, byte ptr [rdi + r9]
    movzx r11, byte ptr [rbx + r9]
    cmp r8, r11
    jl no_swap
    jg do_swap
    inc r9
    cmp r9, r10
    jl compare_bytes
compare_lengths:
    cmp esi, edx
    jl no_swap
    jg do_swap
    jmp no_swap
do_swap:
    mov rax, [line_starts + r14*8]
    mov rcx, [line_starts + r14*8 + 8]
    mov [line_starts + r14*8], rcx
    mov [line_starts + r14*8 + 8], rax
    mov eax, [line_lengths + r14*4]
    mov ecx, [line_lengths + r14*4 + 4]
    mov [line_lengths + r14*4], ecx
    mov [line_lengths + r14*4 + 4], eax
    mov r13, 1
no_swap:
    inc r14
    mov eax, r15d
    sub eax, r12d
    dec eax
    cmp r14d, eax
    jle bubble_inner
    test r13, r13
    jz sort_done
    inc r12
    cmp r12d, r15d
    jl bubble_outer
sort_done:

    xor r14, r14
    mov ecx, [line_count]
    test ecx, ecx
    jz output_done
output_loop:
    mov rax, 1
    mov rdi, 1
    mov rsi, [line_starts + r14*8]
    mov edx, [line_lengths + r14*4]
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r14
    cmp r14d, ecx
    jl output_loop
output_done:

    mov rax, 60
    xor rdi, rdi
    syscall
```
