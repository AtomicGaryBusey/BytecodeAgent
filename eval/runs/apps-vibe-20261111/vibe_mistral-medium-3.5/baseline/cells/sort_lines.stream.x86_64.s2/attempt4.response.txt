```asm
.globl _start

.section .rodata
newline: .byte 0x0A

.section .bss
input_buffer: .space 32768
line_starts: .space 8000
line_lengths: .space 4000
line_count: .space 4

.text
_start:
    xor r8, r8
    lea r13, [rip + input_buffer]
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r8]
    mov rdx, 32768
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:

    xor rcx, rcx
    lea r10, [r13]
    lea r11, [rip + line_starts]
    lea r12, [rip + line_lengths]
    xor r9, r9
parse_loop:
    cmp rcx, r8
    jge parse_final
    mov al, byte ptr [r13 + rcx]
    cmp al, 0x0A
    jne parse_next
    lea rax, [r13 + rcx]
    sub rax, r10
    mov [r11], r10
    mov [r12], eax
    add r11, 8
    add r12, 4
    inc r9
    lea r10, [r13 + rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_final:
    lea rax, [r13 + r8]
    cmp r10, rax
    jae parse_done
    lea rax, [r13 + r8]
    sub rax, r10
    mov [r11], r10
    mov [r12], eax
    inc r9
parse_done:
    lea rax, [rip + line_count]
    mov [rax], r9d

    lea rax, [rip + line_count]
    mov ecx, [rax]
    dec ecx
    jle sort_done
    mov r15d, ecx
    xor r12d, r12d
bubble_outer:
    xor r13d, r13d
    xor r14d, r14d
bubble_inner:
    lea rax, [rip + line_starts]
    mov rdi, [rax + r14*8]
    mov rsi, [rax + r14*8 + 8]
    lea rbx, [rip + line_lengths]
    mov edx, [rbx + r14*4]
    mov ecx, [rbx + r14*4 + 4]
    xor r8, r8
    mov r9d, edx
    cmp r9d, ecx
    cmovg r9d, ecx
    test r9d, r9d
    jz compare_length
compare_bytes:
    mov al, byte ptr [rdi + r8]
    mov bl, byte ptr [rsi + r8]
    cmp al, bl
    jl no_swap
    jg do_swap
    inc r8
    cmp r8d, r9d
    jl compare_bytes
compare_length:
    cmp edx, ecx
    jl no_swap
    jg do_swap
    jmp no_swap
do_swap:
    lea rax, [rip + line_starts]
    mov r8, [rax + r14*8]
    mov r9, [rax + r14*8 + 8]
    mov [rax + r14*8], r9
    mov [rax + r14*8 + 8], r8
    lea rax, [rip + line_lengths]
    mov r8d, [rax + r14*4]
    mov r9d, [rax + r14*4 + 4]
    mov [rax + r14*4], r9d
    mov [rax + r14*4 + 4], r8d
    mov r13d, 1
no_swap:
    inc r14d
    cmp r14d, r15d
    jl bubble_inner
    test r13d, r13d
    jz sort_done
    inc r12d
    cmp r12d, r15d
    jl bubble_outer
sort_done:

    lea rax, [rip + line_count]
    mov ecx, [rax]
    xor r14d, r14d
output_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + line_starts]
    mov rsi, [rsi + r14*8]
    lea rdx, [rip + line_lengths]
    mov edx, [rdx + r14*4]
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r14d
    cmp r14d, ecx
    jl output_loop

    mov rax, 60
    xor rdi, rdi
    syscall
```
