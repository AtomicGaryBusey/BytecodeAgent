```asm
.globl _start

.section .rodata
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg
error_msg: .ascii "error at "
error_msg_len = . - error_msg
newline: .ascii "\n"

.section .bss
buffer: .space 65536
num_buf: .space 20

.text
_start:
    mov rdi, 0
    lea rbx, [rip + buffer]
    xor r12, r12
.read_loop:
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    jle .read_done
    mov rax, 0
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    lea rcx, [rbx + r12]
    mov rdi, rbx
    call parse_document
    test rax, rax
    jz .error
    cmp rdi, rcx
    jne .error
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

.error:
    lea rax, [rip + buffer]
    sub rdi, rax
    mov r10, rdi
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_msg_len
    mov rax, 1
    syscall
    mov rax, r10
    lea r9, [rip + num_buf + 19]
    mov byte ptr [r9], 0
    dec r9
    mov r8, 10
1:
    xor rdx, rdx
    div r8
    add dl, '0'
    mov [r9], dl
    dec r9
    test rax, rax
    jnz 1b
    inc r9
    lea rsi, [r9]
    lea rdx, [rip + num_buf + 19]
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

skip_ws:
    mov rax, 1
1:
    cmp rdi, rcx
    jge 2f
    mov al, byte ptr [rdi]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
    ret
3:
    inc rdi
    jmp 1b
2:
    ret

parse_document:
    call skip_ws
    call parse_value
    test rax, rax
    jz 1f
    call skip_ws
    test rax, rax
    jz 1f
    mov rax, 1
    ret
1:
    xor rax, rax
    ret

parse_value:
    cmp rdi, rcx
    jge 1f
    mov al, byte ptr [rdi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    jl 1f
    cmp al, '9'
    jg 1f
    jmp parse_number
1:
    xor rax, rax
    ret

parse_true:
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 't'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'r'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'u'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'e'
    jne 1f
    inc rdi
    mov rax, 1
    ret
1:
    xor rax, rax
    ret

parse_false:
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'f'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'a'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'l'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 's'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'e'
    jne 1f
    inc rdi
    mov rax, 1
    ret
1:

    ret

parse_null:
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'n'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'u'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'l'
    jne 1f
    inc rdi
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], 'l'
    jne 1f
    inc rdi
    mov rax, 1
    ret
1:
    xor rax, rax
    ret

parse_string:
    cmp rdi, rcx
    jge 1f
    cmp byte ptr [rdi], '"'
    jne 1f
    inc rdi
10:
    cmp rdi, rcx
    jge 1f
    mov al, byte ptr [rdi]
    cmp al, '"'
    je 11f
    cmp al, '\'
    je parse_escape
    cmp al, 0x20
    jl 1f
    inc rdi
    jmp 10b
11:
    inc rdi
    mov rax, 1
    ret
1:
    xor rax, rax
    ret

parse_escape:
    inc rdi
    cmp rdi, rcx
    jge 1f
    mov al, byte ptr [rdi]
    cmp al, '"'
    je 12f
    cmp al, '\'
    je 12f
    cmp al, '/'
    je 12f
    cmp al, 'b'
    je 12f
    cmp al, 'f'
    je 12f
    cmp al, 'n'
    je 12f
    cmp al, 'r'
    je 12f
    cmp al, 't'
    je 12f
    cmp al, 'u'
    je parse_unicode
    jmp 1f
12:
    inc rdi
    mov rax, 1
    ret
1:
    xor rax, rax
    ret

parse_unicode:
    inc rdi
    mov r8, 4
10:
    cmp rdi, rcx
    jge 1f
    mov al, byte ptr [rdi]
    cmp al, '0'
    jl 1f
    cmp al, '9'
    jle 11f
    cmp al, 'a'
    jl 12f
    cmp al, 'f'
    jle 11f
12:
    cmp al, 'A'
    jl 1f
    cmp al, 'F'
    jg 1f
11:
    inc rdi
    dec r8
    jnz 10b
    mov rax, 1
    ret
1:
    xor rax, rax
    ret

parse_number:

    cmp rdi, rcx
    jge 1f

    cmp al, '+'
    je 140f
    cmp al, '-'
    je 140f
    jmp 141f
140:
    inc rdi
    cmp rdi, rcx
    jge 1f
141:
    mov al, byte ptr [rdi]
    cmp al, '0'
    jl 1f
    cmp al, '9'
    jg 1f

142:
    cmp rdi, rcx
    jge 14f

    cmp al, '0'

    jg 14f

    mov rax, 1

1:

parse_object:
```
