.text
.globl _start

itoa:
	test rdi, rdi
	jz .zero
	xor rcx, rcx
	mov rax, rdi
	mov r8, 10
.div_loop:
	xor rdx, rdx
	div r8
	add dl, '0'
	push rdx
	inc rcx
	test rax, rax
	jnz .div_loop
	mov rax, rcx
.pop_loop:
	pop rdx
	mov [rsi], dl
	inc rsi
	dec rcx
	jnz .pop_loop
	ret
.zero:
	mov byte ptr [rsi], '0'
	mov rax, 1
	inc rsi
	ret

_start:
	mov qword ptr [lines], 0
	mov qword ptr [words], 0
	mov qword ptr [bytes], 0
	mov byte ptr [in_whitespace], 1
.read_loop:
	lea rdi, [input_buffer]
	mov rsi, 65536
	mov rax, 0
	syscall
	test rax, rax
	jz .read_done
	mov rcx, rax
	lea rsi, [input_buffer]
	lea r9, [rip + is_whitespace]
.process_loop:
	mov al, byte ptr [rsi]
	inc rsi
	inc qword ptr [bytes]
	movzx edx, byte ptr [r9 + rax]
	test dl, dl
	jnz .is_ws
	cmp byte ptr [in_whitespace], 1
	jne .next_byte
	mov byte ptr [in_whitespace], 0
	inc qword ptr [words]
	jmp .next_byte
.is_ws:
	mov byte ptr [in_whitespace], 1
	cmp al, 0x0A
	jne .next_byte
	inc qword ptr [lines]
.next_byte:
	dec rcx
	jnz .process_loop
	jmp .read_loop
.read_done:
	mov rdi, [lines]
	lea rsi, [output_buffer]
	call itoa
	mov rdx, rax
	mov rdi, 1
	lea rsi, [output_buffer]
	mov rax, 1
	syscall
	mov rdi, 1
	lea rsi, [space]
	mov rdx, 1
	mov rax, 1
	syscall
	mov rdi, [words]
	lea rsi, [output_buffer]
	call itoa
	mov rdx, rax
	mov rdi, 1
	lea rsi, [output_buffer]
	mov rax, 1
	syscall
	mov rdi, 1
	lea rsi, [space]
	mov rdx, 1
	mov rax, 1
	syscall
	mov rdi, [bytes]
	lea rsi, [output_buffer]
	call itoa
	mov rdx, rax
	mov rdi, 1
	lea rsi, [output_buffer]
	mov rax, 1
	syscall
	mov rdi, 1
	lea rsi, [newline]
	mov rdx, 1
	mov rax, 1
	syscall
	mov rdi, 0
	mov rax, 60
	syscall

.section .rodata
is_whitespace:
	.zero 9
	.byte 1
	.byte 1
	.byte 1
	.byte 1
	.byte 1
	.zero 18
	.byte 1
	.zero 223
space:
	.byte ' '
newline:
	.byte 0x0A

.bss
input_buffer:
	.zero 65536
lines:
	.zero 8
words:
	.zero 8
bytes:
	.zero 8
in_whitespace:
	.zero 1
output_buffer:
	.zero 20
