.text
.globl _start

_start:
  mov qword ptr [error_offset], -1
  xor r8, r8
read_loop:
  mov rdi, 0
  lea rsi, [input_buf + r8]
  mov rdx, 65536
  sub rdx, r8
  mov rax, 0
  syscall
  test rax, rax
  jz read_done
  add r8, rax
  cmp r8, 65536
  jb read_loop
read_done:
  mov [input_len], r8
  lea rax, [input_buf + r8]
  mov [input_end], rax
  lea rdi, [input_buf]
  call skip_ws
  mov rdi, rax
  call parse_value
  cmp rax, -1
  je error_detected
  mov rdi, rax
  call skip_ws
  mov rdi, rax
  cmp rax, [input_end]
  jne extra_bytes
  mov rdi, 1
  lea rsi, [ok_msg]
  mov rdx, 3
  mov rax, 1
  syscall
  mov rdi, 0
  mov rax, 60
  syscall
extra_bytes:
  mov r8, rax
  sub r8, input_buf
  mov [error_offset], r8
error_detected:
  mov rax, [error_offset]
  lea rsi, [num_buf]
  xor rcx, rcx
convert_loop:
  xor rdx, rdx
  mov rbx, 10
  div rbx
  add dl, '0'
  mov [rsi + rcx], dl
  inc rcx
  test rax, rax
  jnz convert_loop
  lea rdi, [rsi + rcx - 1]
  lea rsi, [rsi]
  mov r8, rcx
  shr r8, 1
  jz no_reverse
reverse_loop:
  mov al, [rsi]
  mov bl, [rdi]
  mov [rsi], bl
  mov [rdi], al
  inc rsi
  dec rdi
  dec r8
  jnz reverse_loop
no_reverse:
  mov r8, rcx
  mov rdi, 1
  lea rsi, [error_prefix]
  mov rdx, 9
  mov rax, 1
  syscall
  mov rdi, 1
  lea rsi, [num_buf]
  mov rdx, r8
  mov rax, 1
  syscall
  mov rdi, 1
  lea rsi, [newline]
  mov rdx, 1
  mov rax, 1
  syscall
  mov rdi, 1
  mov rax, 60
  syscall

skip_ws:
  mov rax, rdi
skip_ws_loop:
  cmp rax, [input_end]
  je skip_ws_done
  movzx ecx, byte ptr [rax]
  cmp cl, ' '
  je skip_ws_is_ws
  cmp cl, 0x09
  je skip_ws_is_ws
  cmp cl, 0x0A
  je skip_ws_is_ws
  cmp cl, 0x0D
  je skip_ws_is_ws
  jmp skip_ws_done
skip_ws_is_ws:
  inc rax
  jmp skip_ws_loop
skip_ws_done:
  ret

parse_value:
  cmp rdi, [input_end]
  je parse_value_error
  movzx eax, byte ptr [rdi]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, '"'
  je parse_string
  cmp al, 't'
  je parse_true
  cmp al, 'f'
  je parse_false
  cmp al, 'n'
  je parse_null
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jl parse_value_error
  cmp al, '9'
  jg parse_value_error
  jmp parse_number
parse_value_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_object:
  cmp rdi, [input_end]
  je parse_object_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '{'
  jne parse_object_expected
  inc rdi
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_object_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '}'
  je parse_object_empty
  call parse_string
  cmp rax, -1
  je parse_object_error
  mov rdi, rax
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_object_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ':'
  jne parse_object_expected_colon
  inc rdi
  call skip_ws
  mov rdi, rax
  call parse_value
  cmp rax, -1
  je parse_object_error
  mov rdi, rax
parse_object_members:
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_object_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ','
  jne parse_object_check_end
  inc rdi
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_object_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '}'
  je parse_object_trailing_comma
  call parse_string
  cmp rax, -1
  je parse_object_error
  mov rdi, rax
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_object_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ':'
  jne parse_object_expected_colon
  inc rdi
  call skip_ws
  mov rdi, rax
  call parse_value
  cmp rax, -1
  je parse_object_error
  mov rdi, rax
  jmp parse_object_members
parse_object_check_end:
  cmp al, '}'
  jne parse_object_expected
parse_object_empty:
  inc rdi
  mov rax, rdi
  ret
parse_object_incomplete:
parse_object_expected:
parse_object_expected_colon:
parse_object_trailing_comma:
parse_object_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_array:
  cmp rdi, [input_end]
  je parse_array_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '['
  jne parse_array_expected
  inc rdi
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_array_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ']'
  je parse_array_empty
  call parse_value
  cmp rax, -1
  je parse_array_error
  mov rdi, rax
parse_array_values:
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_array_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ','
  jne parse_array_check_end
  inc rdi
  call skip_ws
  mov rdi, rax
  cmp rdi, [input_end]
  je parse_array_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ']'
  je parse_array_trailing_comma
  call parse_value
  cmp rax, -1
  je parse_array_error
  mov rdi, rax
  jmp parse_array_values
parse_array_check_end:
  cmp al, ']'
  jne parse_array_expected
parse_array_empty:
  inc rdi
  mov rax, rdi
  ret
parse_array_incomplete:
parse_array_expected:
parse_array_trailing_comma:
parse_array_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

is_hex_digit:
  cmp al, '0'
  jl is_hex_digit_not_hex
  cmp al, '9'
  jle is_hex_digit_is_hex
  cmp al, 'a'
  jl is_hex_digit_not_hex
  cmp al, 'f'
  jle is_hex_digit_is_hex
  cmp al, 'A'
  jl is_hex_digit_not_hex
  cmp al, 'F'
  jle is_hex_digit_is_hex
is_hex_digit_not_hex:
  xor rax, rax
  ret
is_hex_digit_is_hex:
  mov rax, 1
  ret

parse_string:
  cmp rdi, [input_end]
  je parse_string_error
  movzx eax, byte ptr [rdi]
  cmp al, '"'
  jne parse_string_expected
  inc rdi
parse_string_loop:
  cmp rdi, [input_end]
  je parse_string_error
  movzx eax, byte ptr [rdi]
  cmp al, '"'
  je parse_string_end
  cmp al, '\'
  je parse_string_escape
  cmp al, 0x20
  jl parse_string_invalid_char
  inc rdi
  jmp parse_string_loop
parse_string_escape:
  inc rdi
  cmp rdi, [input_end]
  je parse_string_error
  movzx eax, byte ptr [rdi]
  cmp al, '"'
  je parse_string_escaped
  cmp al, '\'
  je parse_string_escaped
  cmp al, '/'
  je parse_string_escaped
  cmp al, 'b'
  je parse_string_escaped
  cmp al, 'f'
  je parse_string_escaped
  cmp al, 'n'
  je parse_string_escaped
  cmp al, 'r'
  je parse_string_escaped
  cmp al, 't'
  je parse_string_escaped
  cmp al, 'u'
  jne parse_string_invalid_escape
  inc rdi
  mov rcx, 4
parse_string_unicode:
  cmp rdi, [input_end]
  je parse_string_error
  movzx eax, byte ptr [rdi]
  call is_hex_digit
  test rax, rax
  jz parse_string_error
  inc rdi
  dec rcx
  jnz parse_string_unicode
  jmp parse_string_loop
parse_string_escaped:
  inc rdi
  jmp parse_string_loop
parse_string_end:
  inc rdi
  mov rax, rdi
  ret
parse_string_error:
parse_string_expected:
parse_string_invalid_char:
parse_string_invalid_escape:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_number:
  cmp rdi, [input_end]
  je parse_number_error
  movzx eax, byte ptr [rdi]
  cmp al, '-'
  jne parse_number_no_sign
  inc rdi
  cmp rdi, [input_end]
  je parse_number_error
parse_number_no_sign:
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  je parse_number_zero
  cmp al, '1'
  jl parse_number_error
  cmp al, '9'
  jg parse_number_error
  inc rdi
parse_number_int_loop:
  cmp rdi, [input_end]
  je parse_number_after_int
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jl parse_number_after_int
  cmp al, '9'
  jg parse_number_after_int
  inc rdi
  jmp parse_number_int_loop
parse_number_zero:
  inc rdi
  cmp rdi, [input_end]
  je parse_number_after_int
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jl parse_number_after_int
  cmp al, '9'
  jle parse_number_error
parse_number_after_int:
  cmp rdi, [input_end]
  je parse_number_end
  movzx eax, byte ptr [rdi]
  cmp al, '.'
  je parse_number_frac
  cmp al, 'e'
  je parse_number_exp
  cmp al, 'E'
  je parse_number_exp
  jmp parse_number_end
parse_number_frac:
  inc rdi
  cmp rdi, [input_end]
  je parse_number_error
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jl parse_number_error
  cmp al, '9'
  jg parse_number_error
  inc rdi
parse_number_frac_loop:
  cmp rdi, [input_end]
  je parse_number_after_frac
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jl parse_number_after_frac
  cmp al, '9'
  jg parse_number_after_frac
  inc rdi
  jmp parse_number_frac_loop
parse_number_after_frac:
  cmp rdi, [input_end]
  je parse_number_end
  movzx eax, byte ptr [rdi]
  cmp al, 'e'
  je parse_number_exp
  cmp al, 'E'
  je parse_number_exp
  jmp parse_number_end
parse_number_exp:
  inc rdi
  cmp rdi, [input_end]
  je parse_number_error
  movzx eax, byte ptr [rdi]
  cmp al, '+'
  je parse_number_exp_sign
  cmp al, '-'
  je parse_number_exp_sign
  cmp al, '0'
  jl parse_number_error
  cmp al, '9'
  jg parse_number_error
  jmp parse_number_exp_digits
parse_number_exp_sign:
  inc rdi
  cmp rdi, [input_end]
  je parse_number_error
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jl parse_number_error
  cmp al, '9'
  jg parse_number_error
parse_number_exp_digits:
  inc rdi
parse_number_exp_loop:
  cmp rdi, [input_end]
  je parse_number_after_exp
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jl parse_number_after_exp
  cmp al, '9'
  jg parse_number_after_exp
  inc rdi
  jmp parse_number_exp_loop
parse_number_after_exp:
  cmp rdi, [input_end]
  jne parse_number_error
parse_number_end:
  mov rax, rdi
  ret
parse_number_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_true:
  cmp rdi, [input_end]
  je parse_true_error
  movzx eax, byte ptr [rdi]
  cmp al, 't'
  jne parse_true_error
  inc rdi
  cmp rdi, [input_end]
  je parse_true_error
  movzx eax, byte ptr [rdi]
  cmp al, 'r'
  jne parse_true_error
  inc rdi
  cmp rdi, [input_end]
  je parse_true_error
  movzx eax, byte ptr [rdi]
  cmp al, 'u'
  jne parse_true_error
  inc rdi
  cmp rdi, [input_end]
  je parse_true_error
  movzx eax, byte ptr [rdi]
  cmp al, 'e'
  jne parse_true_error
  inc rdi
  mov rax, rdi
  ret
parse_true_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_false:
  cmp rdi, [input_end]
  je parse_false_error
  movzx eax, byte ptr [rdi]
  cmp al, 'f'
  jne parse_false_error
  inc rdi
  cmp rdi, [input_end]
  je parse_false_error
  movzx eax, byte ptr [rdi]
  cmp al, 'a'
  jne parse_false_error
  inc rdi
  cmp rdi, [input_end]
  je parse_false_error
  movzx eax, byte ptr [rdi]
  cmp al, 'l'
  jne parse_false_error
  inc rdi
  cmp rdi, [input_end]
  je parse_false_error
  movzx eax, byte ptr [rdi]
  cmp al, 's'
  jne parse_false_error
  inc rdi
  cmp rdi, [input_end]
  je parse_false_error
  movzx eax, byte ptr [rdi]
  cmp al, 'e'
  jne parse_false_error
  inc rdi
  mov rax, rdi
  ret
parse_false_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_null:
  cmp rdi, [input_end]
  je parse_null_error
  movzx eax, byte ptr [rdi]
  cmp al, 'n'
  jne parse_null_error
  inc rdi
  cmp rdi, [input_end]
  je parse_null_error
  movzx eax, byte ptr [rdi]
  cmp al, 'u'
  jne parse_null_error
  inc rdi
  cmp rdi, [input_end]
  je parse_null_error
  movzx eax, byte ptr [rdi]
  cmp al, 'l'
  jne parse_null_error
  inc rdi
  cmp rdi, [input_end]
  je parse_null_error
  movzx eax, byte ptr [rdi]
  cmp al, 'l'
  jne parse_null_error
  inc rdi
  mov rax, rdi
  ret
parse_null_error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

.section .rodata
ok_msg: .ascii "ok\n"
error_prefix: .ascii "error at "
newline: .byte 0x0A

.bss
input_buf: .space 65536
input_len: .space 8
input_end: .space 8
error_offset: .space 8
num_buf: .space 24
