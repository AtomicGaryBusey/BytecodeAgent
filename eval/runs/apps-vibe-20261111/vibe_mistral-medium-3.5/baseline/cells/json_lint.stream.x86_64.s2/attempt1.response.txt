.text
.globl _start

_start:
  mov qword ptr [error_offset], -1
  xor r8, r8
read_loop:
  mov rdi, 0
  lea rsi, [input_buf + r8]
  mov rdx, 65536
  sub rdx, r8
  mov rax, 0
  syscall
  test rax, rax
  jz read_done
  add r8, rax
  cmp r8, 65536
  jb read_loop
read_done:
  mov [input_len], r8
  lea rax, [input_buf + r8]
  mov [input_end], rax
  lea rdi, [input_buf]
  call skip_ws
  cmp rax, -1
  je error_detected
  mov rdi, rax
  call parse_value
  cmp rax, -1
  je error_detected
  mov rdi, rax
  call skip_ws
  cmp rax, -1
  je error_detected
  cmp rax, [input_end]
  jne extra_bytes
  mov rdi, 1
  lea rsi, [ok_msg]
  mov rdx, ok_msg_len
  mov rax, 1
  syscall
  mov rdi, 0
  mov rax, 60
  syscall
extra_bytes:
  mov r8, rax
  sub r8, input_buf
  mov [error_offset], r8
error_detected:
  mov rax, [error_offset]
  lea rsi, [num_buf]
  xor rcx, rcx
.convert_loop:
  xor rdx, rdx
  mov rbx, 10
  div rbx
  add dl, '0'
  mov [rsi + rcx], dl
  inc rcx
  test rax, rax
  jnz .convert_loop
  lea rdi, [rsi + rcx - 1]
  lea rsi, [rsi]
  mov r8, rcx
  shr r8, 1
  jz .no_reverse
.reverse_loop:
  mov al, [rsi]
  mov bl, [rdi]
  mov [rsi], bl
  mov [rdi], al
  inc rsi
  dec rdi
  dec r8
  jnz .reverse_loop
.no_reverse:
  mov rdi, 1
  lea rsi, [error_prefix]
  mov rdx, error_prefix_len
  mov rax, 1
  syscall
  mov rdi, 1
  lea rsi, [num_buf]
  mov rdx, rcx
  mov rax, 1
  syscall
  mov rdi, 1
  lea rsi, [newline]
  mov rdx, 1
  mov rax, 1
  syscall
  mov rdi, 1
  mov rax, 60
  syscall

skip_ws:
  mov rax, rdi
.loop:
  cmp rax, [input_end]
  je .done
  movzx ecx, byte ptr [rax]
  cmp cl, ' '
  je .is_ws
  cmp cl, 0x09
  je .is_ws
  cmp cl, 0x0A
  je .is_ws
  cmp cl, 0x0D
  je .is_ws
  jmp .done
.is_ws:
  inc rax
  jmp .loop
.done:
  ret

parse_value:
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, '"'
  je parse_string
  cmp al, 't'
  je parse_true
  cmp al, 'f'
  je parse_false
  cmp al, 'n'
  je parse_null
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jl .error
  cmp al, '9'
  jg .error
  jmp parse_number
.error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_object:
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '{'
  jne .error_expected
  inc rdi
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '}'
  je .end_object
.parse_member:
  call parse_string
  cmp rax, -1
  je .error
  mov rdi, rax
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ':'
  jne .error_expected_colon
  inc rdi
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  call parse_value
  cmp rax, -1
  je .error
  mov rdi, rax
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ','
  jne .not_comma
  mov r8, rdi
  inc rdi
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '}'
  je .error_trailing_comma
  jmp .parse_member
.not_comma:
  cmp al, '}'
  jne .error_expected
.end_object:
  inc rdi
  mov rax, rdi
  ret
.error:
  mov rax, -1
  ret
.error_incomplete:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_expected:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_expected_colon:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_trailing_comma:
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_array:
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '['
  jne .error_expected
  inc rdi
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ']'
  je .end_array
.parse_value:
  call parse_value
  cmp rax, -1
  je .error
  mov rdi, rax
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ','
  jne .not_comma
  mov r8, rdi
  inc rdi
  call skip_ws
  cmp rax, -1
  je .error
  mov rdi, rax
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, ']'
  je .error_trailing_comma
  jmp .parse_value
.not_comma:
  cmp al, ']'
  jne .error_expected
.end_array:
  inc rdi
  mov rax, rdi
  ret
.error:
  mov rax, -1
  ret
.error_incomplete:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_expected:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_trailing_comma:
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_string:
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '"'
  jne .error_expected
  inc rdi
.start:
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '"'
  je .end
  cmp al, '\'
  je .escape
  cmp al, 0x20
  jb .error_control
  inc rdi
  jmp .start
.escape:
  inc rdi
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '"'
  je .valid_escape
  cmp al, '\'
  je .valid_escape
  cmp al, '/'
  je .valid_escape
  cmp al, 'b'
  je .valid_escape
  cmp al, 'f'
  je .valid_escape
  cmp al, 'n'
  je .valid_escape
  cmp al, 'r'
  je .valid_escape
  cmp al, 't'
  je .valid_escape
  cmp al, 'u'
  je .u_escape
  jmp .error_invalid_escape
.valid_escape:
  inc rdi
  jmp .start
.u_escape:
  inc rdi
  mov ecx, 4
.hex_loop:
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .not_hex
  cmp al, '9'
  jbe .is_hex
  cmp al, 'a'
  jb .not_hex
  cmp al, 'f'
  jbe .is_hex
  cmp al, 'A'
  jb .not_hex
  cmp al, 'F'
  jbe .is_hex
.not_hex:
  jmp .error_invalid_hex
.is_hex:
  inc rdi
  dec ecx
  jnz .hex_loop
  jmp .start
.error_incomplete:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_control:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_invalid_escape:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_invalid_hex:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_expected:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.end:
  inc rdi
  mov rax, rdi
  ret

parse_number:
  cmp rdi, [input_end]
  je .error_incomplete
  movzx eax, byte ptr [rdi]
  cmp al, '-'
  jne .no_sign
  inc rdi
.no_sign:
  cmp rdi, [input_end]
  je .error_no_int
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jne .non_zero
  inc rdi
  cmp rdi, [input_end]
  je .end_int
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .end_int
  cmp al, '9'
  ja .end_int
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.non_zero:
  cmp al, '1'
  jb .error_no_int
  cmp al, '9'
  ja .error_no_int
.digits_loop:
  inc rdi
  cmp rdi, [input_end]
  je .end_int
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .end_int
  cmp al, '9'
  ja .end_int
  jmp .digits_loop
.end_int:
  cmp rdi, [input_end]
  je .end_number
  movzx eax, byte ptr [rdi]
  cmp al, '.'
  jne .no_frac
  inc rdi
  cmp rdi, [input_end]
  je .error_no_digit_after_dot
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .error_no_digit_after_dot
  cmp al, '9'
  ja .error_no_digit_after_dot
  inc rdi
.frac_digits:
  cmp rdi, [input_end]
  je .end_number
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .end_number
  cmp al, '9'
  ja .end_number
  inc rdi
  jmp .frac_digits
.no_frac:
  cmp al, 'e'
  je .exp
  cmp al, 'E'
  je .exp
  jmp .end_number
.exp:
  inc rdi
  cmp rdi, [input_end]
  je .error_no_digit_after_exp
  movzx eax, byte ptr [rdi]
  cmp al, '+'
  je .exp_sign
  cmp al, '-'
  je .exp_sign
  cmp al, '0'
  jb .error_no_digit_after_exp
  cmp al, '9'
  ja .error_no_digit_after_exp
  jmp .exp_digits
.exp_sign:
  inc rdi
  cmp rdi, [input_end]
  je .error_no_digit_after_exp
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .error_no_digit_after_exp
  cmp al, '9'
  ja .error_no_digit_after_exp
.exp_digits:
  inc rdi
  cmp rdi, [input_end]
  je .end_number
  movzx eax, byte ptr [rdi]
  cmp al, '0'
  jb .end_number
  cmp al, '9'
  ja .end_number
  inc rdi
  jmp .exp_digits
.end_number:
  mov rax, rdi
  ret
.error_incomplete:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_no_int:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_no_digit_after_dot:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret
.error_no_digit_after_exp:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_true:
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 't'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'r'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'u'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'e'
  jne .error
  inc rdi
  mov rax, rdi
  ret
.error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_false:
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'f'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'a'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'l'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 's'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'e'
  jne .error
  inc rdi
  mov rax, rdi
  ret
.error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

parse_null:
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'n'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'u'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'l'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'l'
  jne .error
  inc rdi
  cmp rdi, [input_end]
  je .error
  movzx eax, byte ptr [rdi]
  cmp al, 'n'
  jne .error
  inc rdi
  mov rax, rdi
  ret
.error:
  mov r8, rdi
  sub r8, input_buf
  mov [error_offset], r8
  mov rax, -1
  ret

.section .rodata
error_prefix: .ascii "error at "
error_prefix_len = . - error_prefix
newline: .ascii "\n"
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg

.bss
.align 8
input_buf: .space 65536
input_len: .quad 0
input_end: .quad 0
error_offset: .quad 0
num_buf: .space 20
