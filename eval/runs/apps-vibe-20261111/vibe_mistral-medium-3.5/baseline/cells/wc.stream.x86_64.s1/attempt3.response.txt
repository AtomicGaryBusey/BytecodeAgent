_start:
read_loop:
	mov rax, 0
	mov rdi, 0
	lea rsi, [rip + buf]
	mov rdx, 65536
	syscall
	test rax, rax
	jz read_done
	mov rcx, rax
	lea rsi, [rip + buf]
process_loop:
	movzx r9d, byte ptr [rsi]
	inc rsi
	dec rcx
	inc qword ptr [rip + bytes]
	cmp r9d, 0x0A
	jne not_newline
	inc qword ptr [rip + lines]
not_newline:
	movzx r10, byte ptr [rip + whitespace_table + r9]
	test r10, r10
	jnz is_whitespace
	cmp byte ptr [rip + in_word], 0
	jne after_process
	inc qword ptr [rip + words]
	mov byte ptr [rip + in_word], 1
	jmp after_process
is_whitespace:
	cmp byte ptr [rip + in_word], 0
	je after_process
	mov byte ptr [rip + in_word], 0
after_process:
	test rcx, rcx
	jnz process_loop
	jmp read_loop

read_done:
	lea rdi, [rip + out_buf]
	mov rsi, [rip + lines]
	call append_decimal
	mov byte ptr [rdi], ' '
	inc rdi
	mov rsi, [rip + words]
	call append_decimal
	mov byte ptr [rdi], ' '
	inc rdi
	mov rsi, [rip + bytes]
	call append_decimal
	mov byte ptr [rdi], 0x0A
	inc rdi
	lea rsi, [rip + out_buf]
	mov rdx, rdi
	sub rdx, rsi
	mov rax, 1
	mov rdi, 1
	syscall
	mov rax, 60
	xor rdi, rdi
	syscall

append_decimal:
	test rsi, rsi
	jnz non_zero
	mov byte ptr [rdi], '0'
	inc rdi
	ret
non_zero:
	mov r10, rsp
	mov r8, 10
	mov rax, rsi
digit_loop:
	xor rdx, rdx
	div r8
	push rdx
	test rax, rax
	jnz digit_loop
pop_loop:
	pop rax
	add al, '0'
	mov byte ptr [rdi], al
	inc rdi
	cmp rsp, r10
	jne pop_loop
	ret
