.bss
buffer: .space 65536
output_buffer: .space 32
num_buffer: .space 16

.text
.global _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x23, buffer
    add x23, x23, :lo12:buffer
    mov x24, #65536

read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x23
    mov x2, x24
    svc #0

    cbz x0, end_of_input

    add x21, x21, x0

    mov x26, x23
    mov x27, x0

process_loop:
    cbz x27, read_loop

    ldrb w25, [x26], #1
    sub x27, x27, #1

    cmp w25, #0x0A
    beq is_newline

    cmp w25, #0x20
    beq is_whitespace
    cmp w25, #0x09
    beq is_whitespace
    cmp w25, #0x0B
    beq is_whitespace
    cmp w25, #0x0C
    beq is_whitespace
    cmp w25, #0x0D
    beq is_whitespace

    cmp x22, #0
    b.ne process_loop

    add x20, x20, #1
    mov x22, #1
    b process_loop

is_newline:
    add x19, x19, #1
    mov x22, #0
    b process_loop

is_whitespace:
    mov x22, #0
    b process_loop

end_of_input:
    adrp x0, output_buffer
    add x0, x0, :lo12:output_buffer
    mov x1, x0

    adrp x2, num_buffer
    add x2, x2, :lo12:num_buffer
    add x2, x2, #15
    mov x4, #0

    mov x3, x19
    cmp x3, #0
    beq lines_zero

lines_convert:
    mov x5, #10
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #'0'
    strb w7, [x2]
    sub x2, x2, #1
    add x4, x4, #1
    mov x3, x6
    cbnz x3, lines_convert
    b lines_copy_start

lines_zero:
    mov w7, #'0'
    strb w7, [x2]
    sub x2, x2, #1
    mov x4, #1

lines_copy_start:
    add x2, x2, #1
lines_copy_loop:
    ldrb w3, [x2], #1
    strb w3, [x1], #1
    sub x4, x4, #1
    cbnz x4, lines_copy_loop

    mov w3, #' '
    strb w3, [x1], #1

    adrp x2, num_buffer
    add x2, x2, :lo12:num_buffer
    add x2, x2, #15
    mov x4, #0

    mov x3, x20
    cmp x3, #0
    beq words_zero

words_convert:
    mov x5, #10
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #'0'
    strb w7, [x2]
    sub x2, x2, #1
    add x4, x4, #1
    mov x3, x6
    cbnz x3, words_convert
    b words_copy_start

words_zero:
    mov w7, #'0'
    strb w7, [x2]
    sub x2, x2, #1
    mov x4, #1

words_copy_start:
    add x2, x2, #1
words_copy_loop:
    ldrb w3, [x2], #1
    strb w3, [x1], #1
    sub x4, x4, #1
    cbnz x4, words_copy_loop

    mov w3, #' '
    strb w3, [x1], #1

    adrp x2, num_buffer
    add x2, x2, :lo12:num_buffer
    add x2, x2, #15
    mov x4, #0

    mov x3, x21
    cmp x3, #0
    beq bytes_zero

bytes_convert:
    mov x5, #10
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #'0'
    strb w7, [x2]
    sub x2, x2, #1
    add x4, x4, #1
    mov x3, x6
    cbnz x3, bytes_convert
    b bytes_copy_start

bytes_zero:
    mov w7, #'0'
    strb w7, [x2]
    sub x2, x2, #1
    mov x4, #1

bytes_copy_start:
    add x2, x2, #1
bytes_copy_loop:
    ldrb w3, [x2], #1
    strb w3, [x1], #1
    sub x4, x4, #1
    cbnz x4, bytes_copy_loop

    mov w3, #0x0A
    strb w3, [x1], #1

    sub x2, x1, x0
    mov x8, #64
    mov x1, x0
    mov x0, #1
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0
