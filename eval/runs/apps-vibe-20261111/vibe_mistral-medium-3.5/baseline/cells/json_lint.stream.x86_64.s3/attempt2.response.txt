.globl _start

.text
_start:
    mov dword ptr [current_depth], 0
    mov rax, 0
    mov rdi, 0
    mov rsi, input_buffer
    mov rdx, 65536
    syscall
    cmp rax, 0
    jl error_exit
    mov r9, input_buffer
    lea r10, [r9 + rax]
    mov rsi, r9
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je empty_input
    call parse_value
    test rdx, rdx
    jz parse_failed
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    jne extra_data
    call print_ok
    xor rdi, rdi
    jmp do_exit

empty_input:
    xor rdi, rdi
    jmp print_error_and_exit

parse_failed:
    sub rax, r9
    mov rdi, rax
    jmp print_error_and_exit

extra_data:
    sub rsi, r9
    mov rdi, rsi
    jmp print_error_and_exit

print_error_and_exit:
    call print_error
    mov rax, 1
    jmp do_exit

error_exit:
    mov rax, 1

do_exit:
    mov rdi, rax
    mov rax, 60
    syscall
    ud2

skip_ws:
1:
    cmp rsi, r10
    je 2f
    movzx eax, byte ptr [rsi]
    cmp al, ' '
    je 3f
    cmp al, 9
    je 3f
    cmp al, 10
    je 3f
    cmp al, 13
    je 3f
    jmp 2f
3:
    inc rsi
    jmp 1b
2:
    mov rax, rsi
    mov rdx, 1
    ret

parse_value:
    cmp rsi, r10
    je parse_value_fail
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_value_true
    cmp al, 'f'
    je parse_value_false
    cmp al, 'n'
    je parse_value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_value_fail
    cmp al, '9'
    ja parse_value_fail
    jmp parse_number

parse_value_true:
    lea rcx, [rsi + 4]
    cmp rcx, r10
    jg parse_value_fail
    cmp dword ptr [rsi], 0x65757274
    jne parse_value_fail
    add rsi, 4
    mov rax, rsi
    mov rdx, 1
    ret

parse_value_false:
    lea rcx, [rsi + 5]
    cmp rcx, r10
    jg parse_value_fail
    cmp dword ptr [rsi], 0x656c7366
    jne parse_value_fail
    cmp byte ptr [rsi + 4], 0x65
    jne parse_value_fail
    add rsi, 5
    mov rax, rsi
    mov rdx, 1
    ret

parse_value_null:
    lea rcx, [rsi + 4]
    cmp rcx, r10
    jg parse_value_fail
    cmp dword ptr [rsi], 0x6c6c756e
    jne parse_value_fail
    add rsi, 4
    mov rax, rsi
    mov rdx, 1
    ret

parse_value_fail:
    mov rax, rsi
    mov rdx, 0
    ret

parse_object:
    cmp dword ptr [current_depth], 999
    jg parse_object_fail
    inc dword ptr [current_depth]
    cmp rsi, r10
    je parse_object_fail_after_inc
    cmp byte ptr [rsi], '{'
    jne parse_object_fail_after_inc
    inc rsi
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je parse_object_fail_after_inc
    cmp byte ptr [rsi], '}'
    je parse_object_empty
parse_object_loop:
    call parse_string
    test rdx, rdx
    jz parse_object_fail_after_inc
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je parse_object_fail_after_inc
    cmp byte ptr [rsi], ':'
    jne parse_object_fail_after_inc
    inc rsi
    call skip_ws
    mov rsi, rax
    call parse_value
    test rdx, rdx
    jz parse_object_fail_after_inc
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je parse_object_fail_after_inc
    cmp byte ptr [rsi], ','
    jne parse_object_end_check
    inc rsi
    call skip_ws
    mov rsi, rax
    jmp parse_object_loop
parse_object_end_check:
    cmp byte ptr [rsi], '}'
    jne parse_object_fail_after_inc
parse_object_empty:
    inc rsi
    dec dword ptr [current_depth]
    mov rax, rsi
    mov rdx, 1
    ret
parse_object_fail_after_inc:
parse_object_fail:
    dec dword ptr [current_depth]
    mov rax, rsi
    mov rdx, 0
    ret

parse_array:
    cmp dword ptr [current_depth], 999
    jg parse_array_fail
    inc dword ptr [current_depth]
    cmp rsi, r10
    je parse_array_fail_after_inc
    cmp byte ptr [rsi], '['
    jne parse_array_fail_after_inc
    inc rsi
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je parse_array_fail_after_inc
    cmp byte ptr [rsi], ']'
    je parse_array_empty
parse_array_loop:
    call parse_value
    test rdx, rdx
    jz parse_array_fail_after_inc
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je parse_array_fail_after_inc
    cmp byte ptr [rsi], ','
    jne parse_array_end_check
    inc rsi
    call skip_ws
    mov rsi, rax
    jmp parse_array_loop
parse_array_end_check:
    cmp byte ptr [rsi], ']'
    jne parse_array_fail_after_inc
parse_array_empty:
    inc rsi
    dec dword ptr [current_depth]
    mov rax, rsi
    mov rdx, 1
    ret
parse_array_fail_after_inc:
parse_array_fail:
    dec dword ptr [current_depth]
    mov rax, rsi
    mov rdx, 0
    ret

parse_string:
    cmp rsi, r10
    je parse_string_fail
    cmp byte ptr [rsi], '"'
    jne parse_string_fail
    inc rsi
parse_string_loop:
    cmp rsi, r10
    je parse_string_fail
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je parse_string_end
    cmp al, '\'
    je parse_string_escape
    cmp al, 0x20
    jb parse_string_fail
    inc rsi
    jmp parse_string_loop
parse_string_escape:
    inc rsi
    cmp rsi, r10
    je parse_string_fail
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je parse_string_escape_ok
    cmp al, '\'
    je parse_string_escape_ok
    cmp al, '/'
    je parse_string_escape_ok
    cmp al, 'b'
    je parse_string_escape_ok
    cmp al, 'f'
    je parse_string_escape_ok
    cmp al, 'n'
    je parse_string_escape_ok
    cmp al, 'r'
    je parse_string_escape_ok
    cmp al, 't'
    je parse_string_escape_ok
    cmp al, 'u'
    jne parse_string_fail
    inc rsi
    call parse_string_check_hex
    test rdx, rdx
    jz parse_string_fail
    mov rsi, rax
    call parse_string_check_hex
    test rdx, rdx
    jz parse_string_fail
    mov rsi, rax
    call parse_string_check_hex
    test rdx, rdx
    jz parse_string_fail
    mov rsi, rax
    call parse_string_check_hex
    test rdx, rdx
    jz parse_string_fail
    mov rsi, rax
    jmp parse_string_loop
parse_string_check_hex:
    cmp rsi, r10
    je parse_string_hex_fail
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_string_hex_fail
    cmp al, '9'
    jbe parse_string_hex_ok
    cmp al, 'a'
    jb parse_string_hex_fail
    cmp al, 'f'
    jbe parse_string_hex_ok
    cmp al, 'A'
    jb parse_string_hex_fail
    cmp al, 'F'
    jbe parse_string_hex_ok
parse_string_hex_fail:
    mov rax, rsi
    mov rdx, 0
    ret
parse_string_hex_ok:
    inc rsi
    mov rax, rsi
    mov rdx, 1
    ret
parse_string_escape_ok:
    inc rsi
    jmp parse_string_loop
parse_string_end:
    inc rsi
    mov rax, rsi
    mov rdx, 1
    ret
parse_string_fail:
    mov rax, rsi
    mov rdx, 0
    ret

parse_number:
    cmp rsi, r10
    je parse_number_fail
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne parse_number_no_minus
    inc rsi
    cmp rsi, r10
    je parse_number_fail
parse_number_no_minus:
    cmp rsi, r10
    je parse_number_fail
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je parse_number_int_zero
    cmp al, '1'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
    jmp parse_number_int_digits
parse_number_int_zero:
    inc rsi
    cmp rsi, r10
    je parse_number_after_int
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je parse_number_fail
    cmp al, '9'
    jbe parse_number_fail
    jmp parse_number_after_int
parse_number_int_digits:
    inc rsi
parse_number_int_digits_loop:
    cmp rsi, r10
    je parse_number_after_int
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_number_after_int
    cmp al, '9'
    ja parse_number_after_int
    inc rsi
    jmp parse_number_int_digits_loop
parse_number_after_int:
    cmp rsi, r10
    je parse_number_done
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne parse_number_no_frac
    inc rsi
    cmp rsi, r10
    je parse_number_fail
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
    inc rsi
parse_number_frac_loop:
    cmp rsi, r10
    je parse_number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_number_done
    cmp al, '9'
    ja parse_number_done
    inc rsi
    jmp parse_number_frac_loop
parse_number_no_frac:
    cmp rsi, r10
    je parse_number_done
    movzx eax, byte ptr [rsi]
    cmp al, 'e'
    je parse_number_exp
    cmp al, 'E'
    jne parse_number_done
parse_number_exp:
    inc rsi
    cmp rsi, r10
    je parse_number_fail
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je parse_number_exp_sign
    cmp al, '-'
    je parse_number_exp_sign
    cmp al, '0'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
    jmp parse_number_exp_digits
parse_number_exp_sign:
    inc rsi
    cmp rsi, r10
    je parse_number_fail
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_number_fail
    cmp al, '9'
    ja parse_number_fail
parse_number_exp_digits:
    inc rsi
parse_number_exp_digits_loop:
    cmp rsi, r10
    je parse_number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb parse_number_done
    cmp al, '9'
    ja parse_number_done
    inc rsi
    jmp parse_number_exp_digits_loop
parse_number_done:
    mov rax, rsi
    mov rdx, 1
    ret
parse_number_fail:
    mov rax, rsi
    mov rdx, 0
    ret

print_ok:
    mov rax, 1
    mov rdi, 1
    mov rsi, offset ok_msg
    mov rdx, ok_msg_len
    syscall
    ret

print_error:
    mov r12, rdi
    mov rax, 1
    mov rdi, 1
    mov rsi, offset error_prefix
    mov rdx, error_prefix_len
    syscall
    mov rdi, r12
    call print_uint
    mov rax, 1
    mov rdi, 1
    mov rsi, offset newline
    mov rdx, 1
    syscall
    ret

print_uint:
    sub rsp, 20
    mov rax, rdi
    lea rbx, [rsp + 19]
    mov byte ptr [rbx], 0
    mov rcx, 10
.print_uint_loop:
    xor rdx, rdx
    div rcx
    dec rbx
    add dl, '0'
    mov byte ptr [rbx], dl
    test rax, rax
    jnz .print_uint_loop
    lea rdx, [rsp + 19]
    sub rdx, rbx
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    syscall
    add rsp, 20
    ret
