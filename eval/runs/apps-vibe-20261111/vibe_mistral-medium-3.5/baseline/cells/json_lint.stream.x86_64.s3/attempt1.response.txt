```assembly
.globl _start

.text
_start:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    syscall
    cmp rax, 0
    jl error_exit
    lea r9, [rip + input_buffer]
    lea r10, [r9 + rax]
    mov rsi, r9
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je error_at_0
    call parse_value
    test rdx, rdx
    jz parse_error
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    jne extra_data
    call print_ok
    xor rdi, rdi
    jmp do_exit

error_at_0:
    xor rdi, rdi
    jmp print_error

parse_error:
    sub rax, r9
    mov rdi, rax
    jmp print_error

extra_data:
    sub rsi, r9
    mov rdi, rsi
    jmp print_error

print_error:
    mov r12, rdi
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_msg_len
    syscall
    mov rdi, r12
    call print_uint
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    mov rax, 1
    jmp do_exit

error_exit:
    mov rax, 1

do_exit:
    mov rdi, rax
    mov rax, 60
    syscall
    ud2

skip_ws:
1:
    cmp rsi, r10
    je 2f
    movzx eax, byte ptr [rsi]
    cmp al, ' '
    je 3f
    cmp al, 9
    je 3f
    cmp al, 10
    je 3f
    cmp al, 13
    je 3f
    jmp 2f
3:
    inc rsi
    jmp 1b
2:
    mov rax, rsi
    mov rdx, 1
    ret

parse_value:
    mov r8, rsi
    call parse_object
    test rdx, rdx
    jnz .value_done
    mov rsi, r8
    mov r8, rsi
    call parse_array
    test rdx, rdx
    jnz .value_done
    mov rsi, r8
    mov r8, rsi
    call parse_string
    test rdx, rdx
    jnz .value_done
    mov rsi, r8
    mov r8, rsi
    call parse_number
    test rdx, rdx
    jnz .value_done
    mov rsi, r8
    mov r8, rsi
    call parse_keywords
    test rdx, rdx
    jnz .value_done
    mov rsi, r8
    mov rax, rsi
    mov rdx, 0
    ret
.value_done:
    ret

parse_object:
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], '{'
    jne .fail
    inc rsi
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], '}'
    je .object_end
.object_loop:
    call parse_string
    test rdx, rdx
    jz .fail
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], ':'
    jne .fail
    inc rsi
    call skip_ws
    mov rsi, rax
    call parse_value
    test rdx, rdx
    jz .fail
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], ','
    jne .check_end
    inc rsi
    call skip_ws
    mov rsi, rax
    jmp .object_loop
.check_end:
    cmp byte ptr [rsi], '}'
    jne .fail
.object_end:
    inc rsi
    mov rax, rsi
    mov rdx, 1
    ret
.fail:
    mov rax, rsi
    mov rdx, 0
    ret

parse_array:
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], '['
    jne .fail
    inc rsi
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], ']'
    je .array_end
.array_loop:
    call parse_value
    test rdx, rdx
    jz .fail
    mov rsi, rax
    call skip_ws
    mov rsi, rax
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], ','
    jne .check_end
    inc rsi
    call skip_ws
    mov rsi, rax
    jmp .array_loop
.check_end:
    cmp byte ptr [rsi], ']'
    jne .fail
.array_end:
    inc rsi
    mov rax, rsi
    mov rdx, 1
    ret
.fail:
    mov rax, rsi
    mov rdx, 0
    ret

parse_string:
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], '"'
    jne .fail
    inc rsi
.string_loop:
    cmp rsi, r10
    je .fail
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je .string_end
    cmp al, '\'
    je .escape
    cmp al, 0x20
    jb .invalid_char
    inc rsi
    jmp .string_loop
.escape:
    mov r8, rsi
    inc rsi
    cmp rsi, r10
    je .escape_fail
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, '"'
    je .string_loop
    cmp al, '\'
    je .string_loop
    cmp al, '/'
    je .string_loop
    cmp al, 'b'
    je .string_loop
    cmp al, 'f'
    je .string_loop
    cmp al, 'n'
    je .string_loop
    cmp al, 'r'
    je .string_loop
    cmp al, 't'
    je .string_loop
    cmp al, 'u'
    jne .invalid_escape
    mov ecx, 4
.hex_loop:
    cmp rsi, r10
    je .u_fail
    movzx eax, byte ptr [rsi]
    call is_hex_digit
    test rax, rax
    jz .u_fail
    inc rsi
    loop .hex_loop
    jmp .string_loop
.u_fail:
    mov rax, r8
    mov rdx, 0
    ret
.escape_fail:
    mov rax, r8
    mov rdx, 0
    ret
.invalid_char:
    mov rax, rsi
    mov rdx, 0
    ret
.invalid_escape:
    mov rax, r8
    mov rdx, 0
    ret
.string_end:
    inc rsi
    mov rax, rsi
    mov rdx, 1
    ret
.fail:
    mov rax, rsi
    mov rdx, 0
    ret

is_hex_digit:
    cmp al, '0'
    jb .no
    cmp al, '9'
    jbe .yes
    cmp al, 'a'
    jb .no
    cmp al, 'f'
    jbe .yes
    cmp al, 'A'
    jb .no
    cmp al, 'F'
    jbe .yes
.no:
    xor eax, eax
    ret
.yes:
    mov eax, 1
    ret

parse_number:
    mov r8, rsi
    cmp rsi, r10
    je .fail
    cmp byte ptr [rsi], '-'
    jne .no_minus
    inc rsi
.no_minus:
    cmp rsi, r10
    je .fail_digit_expected
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je .int_zero
    cmp al, '1'
    jb .fail_digit_expected
    cmp al, '9'
    ja .fail_digit_expected
    inc rsi
.int_loop:
    cmp rsi, r10
    je .after_int
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .after_int
    cmp al, '9'
    ja .after_int
    inc rsi
    jmp .int_loop
.int_zero:
    inc rsi
    cmp rsi, r10
    je .after_int
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .after_int
    cmp al, '9'
    jbe .zero_fail
    jmp .after_int
.zero_fail:
    mov rax, rsi
    mov rdx, 0
    ret
.fail_digit_expected:
    cmp byte ptr [r8], '-'
    je .fail_at_minus
    mov rax, rsi
    mov rdx, 0
    ret
.fail_at_minus:
    mov rax, r8
    mov rdx, 0
    ret
.after_int:
    cmp rsi, r10
    je .after_frac
    cmp byte ptr [rsi], '.'
    jne .after_frac
    mov r9, rsi
    inc rsi
    cmp rsi, r10
    je .frac_fail
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .frac_fail
    cmp al, '9'
    ja .frac_fail
    inc rsi
.frac_loop:
    cmp rsi, r10
    je .after_frac
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .after_frac
    cmp al, '9'
    ja .after_frac
    inc rsi
    jmp .frac_loop
.frac_fail:
    mov rax, r9
    mov rdx, 0
    ret
.after_frac:
    cmp rsi, r10
    je .after_exp
    movzx eax, byte ptr [rsi]
    cmp al, 'e'
    je .exp_found
    cmp al, 'E'
    jne .after_exp
.exp_found:
    mov r9, rsi
    inc rsi
    cmp rsi, r10
    je .exp_fail
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .exp_sign
    cmp al, '-'
    je .exp_sign
    jmp .exp_digits
.exp_sign:
    inc rsi
.exp_digits:
    cmp rsi, r10
    je .exp_fail
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .exp_fail
    cmp al, '9'
    ja .exp_fail
    inc rsi
.exp_loop:
    cmp rsi, r10
    je .after_exp
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb .after_exp
    cmp al, '9'
    ja .after_exp
    inc rsi
    jmp .exp_loop
.exp_fail:
    mov rax, r9
    mov rdx, 0
    ret
.after_exp:
    mov rax, rsi
    mov rdx, 1
    ret
.fail:
    mov rax, r8
    mov rdx, 0
    ret

parse_keywords:
    cmp rsi, r10
    je .fail
    movzx eax, byte ptr [rsi]
    cmp al, 't'
    jne .check_false
    mov rax, r10
    sub rax, rsi
    cmp rax, 4
    jb .fail
    cmp dword ptr [rsi], 0x65757274
    jne .fail
    add rsi, 4
    mov rax, rsi
    mov rdx, 1
    ret
.check_false:
    cmp al, 'f'
    jne .check_null
    mov rax, r10
    sub rax, rsi
    cmp rax, 5
    jb .fail
    cmp dword ptr [rsi], 0x736c6166
    jne .fail
    cmp byte ptr [rsi + 4], 'e'
    jne .fail
    add rsi, 5
    mov rax, rsi
    mov rdx, 1
    ret
.check_null:
    cmp al, 'n'
    jne .fail
    mov rax, r10
    sub rax, rsi
    cmp rax, 4
    jb .fail
    cmp dword ptr [rsi], 0x6c6c756e
    jne .fail
    add rsi, 4
    mov rax, rsi
    mov rdx, 1
    ret
.fail:
    mov rax, rsi
    mov rdx, 0
    ret

print_ok:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    ret

print_uint:
    mov rax, rdi
    lea rsi, [rip + num_buffer + 19]
    mov byte ptr [rsi], 0
    dec rsi
    mov rcx, 10
    test rax, rax
    jz .zero
.convert_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rsi], dl
    dec rsi
    test rax, rax
    jnz .convert_loop
    inc rsi
    jmp .write
.zero:
    mov byte ptr [rsi], '0'
.write:
    lea rdi, [rip + num_buffer + 19]
    sub rdi, rsi
    mov rax, 1
    mov rdi, 1
    mov rdx, rdi
    syscall
    ret
```
