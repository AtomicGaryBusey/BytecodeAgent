.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.globl _start
.text
_start:
	adrp x20, input_buffer
	add x20, x20, :lo12:input_buffer
	mov x21, #0
	mov x22, #32768

read_loop:
	mov w0, #STDIN
	add x1, x20, x21
	sub x2, x22, x21
	mov w8, #SYS_READ
	svc #0
	cmp x0, #0
	b.eq read_done
	add x21, x21, x0
	b read_loop

read_done:
	adrp x10, line_descriptors
	add x10, x10, :lo12:line_descriptors
	cbz x21, parse_done
	mov x0, #0
	mov x1, #0
	mov x2, x21
	mov x5, #0

parse_loop:
	cmp x1, x2
	b.ge scan_done
	ldrb w6, [x20, x1]
	cmp w6, #0x0A
	b.ne next_byte
	add x8, x10, x5, lsl #3
	str w0, [x8]
	sub w6, w1, w0
	str w6, [x8, #4]
	add x5, x5, #1
	add x0, x1, #1
next_byte:
	add x1, x1, #1
	b parse_loop

scan_done:
	cmp x0, x2
	b.ge parse_done
	add x8, x10, x5, lsl #3
	str w0, [x8]
	sub w6, w2, w0
	str w6, [x8, #4]
	add x5, x5, #1

parse_done:
	cbz x5, output_done
	mov x6, #1

insertion_sort_loop:
	cmp x6, x5
	b.ge sort_done
	add x12, x10, x6, lsl #3
	ldr w8, [x12]
	ldr w9, [x12, #4]
	sub x7, x6, #1

inner_loop:
	cmp x7, #0
	b.lt insert_key
	add x13, x10, x7, lsl #3
	ldr w14, [x13]
	ldr w15, [x13, #4]
	cmp w15, w9
	b.lt min_j
	mov w16, w9
	b compare_bytes
min_j:
	mov w16, w15
compare_bytes:
	cbz w16, check_lengths
	mov x0, #0
compare_loop:
	add x17, x14, x0
	ldrb w1, [x20, x17]
	add x17, x8, x0
	ldrb w2, [x20, x17]
	cmp w1, w2
	b.ne compare_done
	add x0, x0, #1
	cmp x0, x16
	b.lt compare_loop
	b check_lengths

compare_done:
	cmp w1, w2
	b.gt need_shift
	b end_compare

check_lengths:
	cmp w15, w9
	b.gt need_shift
	b end_compare

need_shift:
	add x12, x10, x7, lsl #3
	add x13, x12, #8
	ldr w14, [x12]
	ldr w15, [x12, #4]
	str w14, [x13]
	str w15, [x13, #4]
	sub x7, x7, #1
	b inner_loop

end_compare:
insert_key:
	add x12, x10, x7, lsl #3
	add x12, x12, #8
	str w8, [x12]
	str w9, [x12, #4]
	add x6, x6, #1
	b insertion_sort_loop

sort_done:
	mov x6, #0
	adrp x7, newline
	add x7, x7, :lo12:newline

output_loop:
	cmp x6, x5
	b.ge output_done
	add x8, x10, x6, lsl #3
	ldr w9, [x8]
	ldr w10, [x8, #4]
	mov w0, #STDOUT
	add x1, x20, x9
	mov w2, w10
	mov w8, #SYS_WRITE
	svc #0
	mov w0, #STDOUT
	mov x1, x7
	mov w2, #1
	mov w8, #SYS_WRITE
	svc #0
	add x6, x6, #1
	b output_loop

output_done:
	mov w0, #0
	mov w8, #SYS_EXIT
	svc #0

.section .rodata
newline: .byte 0x0A

.bss
input_buffer: .space 32768
line_descriptors: .space 8000
line_count: .space 4
