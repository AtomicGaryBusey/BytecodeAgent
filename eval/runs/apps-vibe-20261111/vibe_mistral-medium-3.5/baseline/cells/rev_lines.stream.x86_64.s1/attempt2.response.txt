```asm
.global _start

.section .rodata
newline:
    .byte 0x0A

.section .bss
input_buf:
    .space 65536

.text
_start:
    lea rbx, [input_buf]
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz process_input
    add r12, rax
    jmp read_loop

process_input:
    xor r13, r13
    test r12, r12
    jz exit_ok

process_loop:
    cmp r13, r12
    jge exit_ok

    mov r14, r13
find_newline:
    cmp r14, r12
    jge no_newline
    mov al, byte ptr [rbx + r14]
    cmp al, 0x0A
    je found_newline
    inc r14
    jmp find_newline

found_newline:
    lea rsi, [rbx + r13]
    lea rdi, [rbx + r14 - 1]
    cmp rsi, rdi
    jae write_line

reverse_loop:
    mov al, byte ptr [rsi]
    mov r15b, byte ptr [rdi]
    mov byte ptr [rsi], r15b
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    cmp rsi, rdi
    jl reverse_loop

write_line:
    mov rdx, r14
    sub rdx, r13
    mov rax, 1
    mov rdi, 1
    lea rsi, [rbx + r13]
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [newline]
    mov rdx, 1
    syscall

    lea r13, [r14 + 1]
    jmp process_loop

no_newline:
    lea rsi, [rbx + r13]
    lea rdi, [rbx + r12 - 1]
    cmp rsi, rdi
    jae write_final_line

reverse_final:
    mov al, byte ptr [rsi]
    mov r15b, byte ptr [rdi]
    mov byte ptr [rsi], r15b
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    cmp rsi, rdi
    jl reverse_final

write_final_line:
    mov rdx, r12
    sub rdx, r13
    mov rax, 1
    mov rdi, 1
    lea rsi, [rbx + r13]
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [newline]
    mov rdx, 1
    syscall

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
```
