```assembly
.bss
input_buffer: .space 65536

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.text
.globl _start

_start:
    xor r8, r8
    lea r10, [input_buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r10 + r8]
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    xor r9, r9

dump_loop:
    cmp r9, r8
    jge print_total

    sub rsp, 80
    lea r12, [rsp]

    mov rax, r9
    mov cl, 28
    mov r10, 8
print_offset:
    mov r11, rax
    shr r11, cl
    and r11, 0xf
    movzx r11, byte ptr [hex_digits + r11]
    mov byte ptr [r12], r11b
    inc r12
    sub cl, 4
    dec r10
    jnz print_offset

    mov byte ptr [r12], ' '
    mov byte ptr [r12 + 1], ' '
    add r12, 2

    xor r10, r10
print_hex_loop:
    cmp r10, 16
    jge print_ascii_section

    mov rax, r9
    add rax, r10
    cmp rax, r8
    jb has_byte

    mov byte ptr [r12], ' '
    mov byte ptr [r12 + 1], ' '
    mov byte ptr [r12 + 2], ' '
    add r12, 3
    inc r10
    cmp r10, 8
    jne print_hex_loop
    mov byte ptr [r12], ' '
    inc r12
    jmp print_hex_loop

has_byte:
    lea r11, [input_buffer + rax]
    movzx rax, byte ptr [r11]

    mov r11, rax
    shr r11, 4
    and r11, 0xf
    movzx r11, byte ptr [hex_digits + r11]
    mov byte ptr [r12], r11b
    inc r12

    mov r11, rax
    and r11, 0xf
    movzx r11, byte ptr [hex_digits + r11]
    mov byte ptr [r12], r11b
    inc r12

    mov byte ptr [r12], ' '
    inc r12

    inc r10
    cmp r10, 8
    jne print_hex_loop
    mov byte ptr [r12], ' '
    inc r12
    jmp print_hex_loop

print_ascii_section:
    mov byte ptr [r12], ' '
    mov byte ptr [r12 + 1], '|'
    add r12, 2

    xor r10, r10
print_ascii_loop:
    cmp r10, 16
    jge finish_line

    mov rax, r9
    add rax, r10
    cmp rax, r8
    jge finish_line

    lea r11, [input_buffer + rax]
    movzx rax, byte ptr [r11]

    cmp al, 0x20
    jb is_dot
    cmp al, 0x7e
    jbe is_char
is_dot:
    mov al, '.'
is_char:
    mov byte ptr [r12], al
    inc r12
    inc r10
    jmp print_ascii_loop

finish_line:
    mov byte ptr [r12], '|'
    mov byte ptr [r12 + 1], '\n'
    add r12, 2

    mov rdx, r12
    sub rdx, rsp

    mov rax, 1
    mov rdi, 1
    mov rsi, rsp
    syscall

    add rsp, 80
    add r9, 16
    jmp dump_loop

print_total:
    sub rsp, 9
    mov rax, r8
    mov cl, 28
    lea r10, [rsp]
    mov r11, 8
print_total_loop:
    mov r12, rax
    shr r12, cl
    and r12, 0xf
    movzx r12, byte ptr [hex_digits + r12]
    mov byte ptr [r10], r12b
    inc r10
    sub cl, 4
    dec r11
    jnz print_total_loop

    mov byte ptr [r10], '\n'

    mov rax, 1
    mov rdi, 1
    mov rsi, rsp
    mov rdx, 9
    syscall

    add rsp, 9

    mov rax, 60
    xor rdi, rdi
    syscall
```
