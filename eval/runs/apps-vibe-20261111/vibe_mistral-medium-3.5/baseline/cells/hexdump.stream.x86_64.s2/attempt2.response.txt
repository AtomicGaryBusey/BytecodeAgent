```assembly
.bss
input_buffer: .space 65536

.section .rodata
hex_digits: .ascii "0123456789abcdef"

.text
.globl _start

_start:
    xor r8, r8
    lea r10, [input_buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r10 + r8]
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    xor r9, r9

dump_loop:
    cmp r9, r8
    jge print_total

    sub rsp, 80
    lea r12, [rsp]

    mov rax, r9
    mov r11, 28
print_offset_loop:
    mov r13, rax
    shr r13, r11
    and r13, 0xF
    mov cl, byte ptr [hex_digits + r13]
    mov [r12], cl
    inc r12
    sub r11, 4
    jns print_offset_loop

    mov word ptr [r12], '  '
    add r12, 2

    xor r11, r11
hex_byte_loop:
    mov rax, r9
    add rax, r11
    cmp rax, r8
    jl byte_exists

    mov word ptr [r12], '  '
    mov byte ptr [r12 + 2], ' '
    add r12, 3
    jmp check_extra_space

byte_exists:
    movzx r13, byte ptr [r10 + r9 + r11]
    mov r14, r13
    shr r14, 4
    and r14, 0xF
    mov cl, byte ptr [hex_digits + r14]
    mov [r12], cl
    and r13, 0xF
    mov cl, byte ptr [hex_digits + r13]
    mov [r12 + 1], cl
    mov byte ptr [r12 + 2], ' '
    add r12, 3

check_extra_space:
    cmp r11, 7
    jne next_hex_byte
    mov byte ptr [r12], ' '
    inc r12
next_hex_byte:
    inc r11
    cmp r11, 16
    jl hex_byte_loop

    mov byte ptr [r12], ' '
    inc r12
    mov byte ptr [r12], '|'
    inc r12

    xor r11, r11
ascii_loop:
    mov rax, r9
    add rax, r11
    cmp rax, r8
    jge ascii_loop_done

    movzx rax, byte ptr [r10 + r9 + r11]
    cmp al, 0x20
    jl not_printable
    cmp al, 0x7E
    jg not_printable
    mov [r12], al
    inc r12
    inc r11
    jmp ascii_loop
not_printable:
    mov byte ptr [r12], '.'
    inc r12
    inc r11
    jmp ascii_loop

ascii_loop_done:
    mov byte ptr [r12], '|'
    inc r12
    mov byte ptr [r12], '\n'
    inc r12

    mov rax, 1
    mov rdi, 1
    lea rsi, [rsp]
    mov rdx, r12
    sub rdx, rsp
    syscall

    add rsp, 80

    add r9, 16
    jmp dump_loop

print_total:
    sub rsp, 16
    lea r12, [rsp]

    mov rax, r8
    mov r11, 28
print_total_loop:
    mov r13, rax
    shr r13, r11
    and r13, 0xF
    mov cl, byte ptr [hex_digits + r13]
    mov [r12], cl
    inc r12
    sub r11, 4
    jns print_total_loop

    mov byte ptr [r12], '\n'
    inc r12

    mov rax, 1
    mov rdi, 1
    lea rsi, [rsp]
    mov rdx, r12
    sub rdx, rsp
    syscall

    add rsp, 16

    mov rax, 60
    xor rdi, rdi
    syscall
```
