.global _start

.text
_start:
  xor r8, r8
read_loop:
  mov rdx, 32768
  sub rdx, r8
  jz read_done
  mov rax, 0
  mov rdi, 0
  lea rsi, [input_buffer]
  add rsi, r8
  syscall
  test rax, rax
  jz read_done
  add r8, rax
  jmp read_loop
read_done:

  lea rsi, [input_buffer]
  xor rcx, rcx
  xor rdi, rdi
parse_loop:
  cmp rdi, r8
  jge parse_done
  lea rax, [rsi + rdi]
  mov [line_starts + rcx*8], rax
  mov rbx, rdi
parse_inner:
  cmp rbx, r8
  jge parse_inner_done
  movzx edx, byte ptr [rsi + rbx]
  cmp dl, 0x0A
  je parse_inner_done
  inc rbx
  jmp parse_inner
parse_inner_done:
  mov rax, rbx
  sub rax, rdi
  mov [line_lengths + rcx*4], eax
  inc rcx
  cmp rbx, r8
  jge parse_done
  cmp byte ptr [rsi + rbx], 0x0A
  jne parse_done
  inc rbx
  mov rdi, rbx
  jmp parse_loop
parse_done:
  mov [line_count], ecx

  mov r12d, [line_count]
  test r12d, r12d
  jz write_output

  mov r13, r12
  dec r13
outer_loop:
  mov r14, r13
  dec r14
inner_loop:
  mov rsi, [line_starts + r14*8]
  mov rdi, [line_lengths + r14*4]
  mov r8, [line_starts + r14*8 + 8]
  mov r9, [line_lengths + r14*4 + 4]
  mov rax, rdi
  mov rbx, r9
  cmp rax, rbx
  cmovg rax, rbx
  xor rcx, rcx
compare_loop:
  cmp rcx, rax
  jge compare_done
  movzx r10d, byte ptr [rsi + rcx]
  movzx r11d, byte ptr [r8 + rcx]
  cmp r10d, r11d
  jl no_swap_needed
  jg swap_needed
  inc rcx
  jmp compare_loop
compare_done:
  cmp rdi, r9
  jl no_swap_needed
  jg swap_needed
  jmp no_swap_needed
swap_needed:
  mov rax, [line_starts + r14*8]
  mov rbx, [line_starts + r14*8 + 8]
  mov [line_starts + r14*8], rbx
  mov [line_starts + r14*8 + 8], rax
  mov eax, [line_lengths + r14*4]
  mov ebx, [line_lengths + r14*4 + 4]
  mov [line_lengths + r14*4], ebx
  mov [line_lengths + r14*4 + 4], eax
no_swap_needed:
  dec r14
  jns inner_loop
  dec r13
  jns outer_loop

write_output:
  mov r12d, [line_count]
  xor r13, r13
write_loop:
  cmp r13d, r12d
  jge exit
  mov rsi, [line_starts + r13*8]
  mov edx, [line_lengths + r13*4]
  test edx, edx
  jz write_newline
  mov rdi, 1
  mov rax, 1
  syscall
write_newline:
  mov rdi, 1
  lea rsi, [newline]
  mov rdx, 1
  mov rax, 1
  syscall
  inc r13
  jmp write_loop

exit:
  mov rax, 60
  xor rdi, rdi
  syscall

.section .rodata
newline: .byte 0x0A

.section .bss
.align 8
input_buffer: .space 32768
line_starts: .space 8000
line_lengths: .space 4000
line_count: .long 0
