.global _start

.text
_start:
  xor rbx, rbx
read_loop:
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + input_buffer]
  add rsi, rbx
  mov rdx, 32768
  sub rdx, rbx
  syscall
  test rax, rax
  jz read_done
  add rbx, rax
  jmp read_loop
read_done:

  xor rcx, rcx
  xor r8, r8
  xor r9, r9
parse_loop:
  cmp r8, rbx
  jge parse_done
  mov al, byte ptr [rip + input_buffer + r8]
  cmp al, 0x0A
  jne parse_not_newline
  mov r10d, r8d
  sub r10d, r9d
  lea r11, [rip + line_pointers]
  mov [r11 + rcx*8], r9
  lea r11, [rip + line_lengths]
  mov [r11 + rcx*4], r10d
  inc rcx
  lea r9, [r8 + 1]
parse_not_newline:
  inc r8
  jmp parse_loop
parse_done:
  cmp r9, rbx
  jge no_final_line
  mov r10d, ebx
  sub r10d, r9d
  lea r11, [rip + line_pointers]
  mov [r11 + rcx*8], r9
  lea r11, [rip + line_lengths]
  mov [r11 + rcx*4], r10d
  inc rcx
no_final_line:

  test rcx, rcx
  jz exit

  xor r8, r8
sort_outer:
  mov r9, r8
  mov r10, r8
  inc r10
sort_inner_start:
  cmp r10, rcx
  jge sort_inner_done

  lea r11, [rip + line_pointers]
  mov r12, [r11 + r10*8]
  mov r13, [r11 + r9*8]

  lea r11, [rip + line_lengths]
  mov r14d, [r11 + r10*4]
  mov r15d, [r11 + r9*4]

  lea rbx, [rip + input_buffer]
  add rbx, r12
  lea rcx, [rip + input_buffer]
  add rcx, r13

  xor r11d, r11d
compare_loop:
  cmp r11d, r14d
  jge compare_j_done
  cmp r11d, r15d
  jge compare_min_done
  movzx edx, byte ptr [rbx + r11]
  movzx esi, byte ptr [rcx + r11]
  cmp dl, sil
  jl j_less
  jg j_greater
  inc r11d
  jmp compare_loop

compare_j_done:
  cmp r14d, r15d
  jl j_less
  jmp compare_next
compare_min_done:
  cmp r15d, r14d
  jl j_greater
  jmp compare_next

j_less:
  mov r9, r10
  jmp compare_next
j_greater:
  jmp compare_next
compare_next:
  inc r10
  jmp sort_inner_start

sort_inner_done:
  cmp r9, r8
  je sort_no_swap
  lea r11, [rip + line_pointers]
  mov r12, [r11 + r8*8]
  mov r13, [r11 + r9*8]
  mov [r11 + r8*8], r13
  mov [r11 + r9*8], r12
  lea r11, [rip + line_lengths]
  mov r12d, [r11 + r8*4]
  mov r13d, [r11 + r9*4]
  mov [r11 + r8*4], r13d
  mov [r11 + r9*4], r12d
sort_no_swap:
  inc r8
  cmp r8, rcx
  jl sort_outer

  xor r8, r8
print_loop:
  cmp r8, rcx
  jge exit

  lea r11, [rip + line_pointers]
  mov r12, [r11 + r8*8]
  lea r11, [rip + line_lengths]
  mov r13d, [r11 + r8*4]

  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + input_buffer]
  add rsi, r12
  mov rdx, r13
  syscall

  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + newline]
  mov rdx, 1
  syscall

  inc r8
  jmp print_loop

exit:
  mov rax, 60
  xor rdi, rdi
  syscall

.section .rodata
newline: .byte 0x0A

.bss
input_buffer: .space 32768
line_pointers: .space 8000
line_lengths: .space 4000
