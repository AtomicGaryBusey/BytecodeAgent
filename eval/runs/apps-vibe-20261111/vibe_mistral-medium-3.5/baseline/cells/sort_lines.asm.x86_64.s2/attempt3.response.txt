  mov rdi, 0

  mov rdx, 32768
  sub rdx, r12
  syscall
  test rax, rax
  jz read_done
  add r12, rax
  jmp read_loop

  xor r8, r8
  xor r9, r9

  jge parse_done

  cmp al, 0x0A

  mov r10d, r8d
  sub r10d, r9d
  mov [line_pointers + rcx*8], r9
  mov [line_lengths + rcx*4], r10d
  inc rcx
  lea r9, [r8 + 1]

  inc r8
  jmp parse_loop

  cmp r9, r12
  jge no_final_line
  mov r10d, r12d
  sub r10d, r9d
  mov [line_pointers + rcx*8], r9
  mov [line_lengths + rcx*4], r10d
  inc rcx
no_final_line:

  jz exit

  mov r9, r8
  mov r10, r8
  inc r10

  mov r15, [line_pointers + r9*8]

  movzx esi, byte ptr [input_buffer + r15 + r11]

  jg j_greater

  jmp compare_loop

compare_min_done:

  jmp compare_next

  mov r13d, [line_lengths + r9*4]
  mov [line_lengths + r8*4], r13d
  mov [line_lengths + r9*4], r12d

  inc r8
  cmp r8, rcx
  jl sort_outer

  xor r8, r8

  jge exit

  mov r14, [line_pointers + r8*8]
  mov r12d, [line_lengths + r8*4]

  mov rdx, r12

  mov rdi, 1
  lea rsi, [newline]

  syscall

  jmp print_loop
