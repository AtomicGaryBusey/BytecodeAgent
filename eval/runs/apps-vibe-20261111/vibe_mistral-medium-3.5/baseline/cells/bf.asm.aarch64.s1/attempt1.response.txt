```assembly
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.equ STDIN, 0
.equ STDOUT, 1

.equ TAPE_SIZE, 30000
.equ MAX_BRACKET_DEPTH, 1000

.bss
.align 8
tape: .space TAPE_SIZE
stdin_buffer: .space 65536
jump_table: .space 65536 * 8
bracket_stack: .space MAX_BRACKET_DEPTH * 8
output_byte: .space 1

.text
.globl _start

_start:
    mov x21, #1
    lsl x21, x21, #16
    adrp x19, stdin_buffer
    add x19, x19, :lo12:stdin_buffer
    mov x20, #0

read_again:
    mov x0, #STDIN
    add x1, x19, x20
    mov x2, x21
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0

    cmp x0, #0
    b.lt read_error
    b.eq read_finish

    add x20, x20, x0
    cmp x20, x21
    b.lt read_again

read_finish:
    mov x21, #0

find_loop:
    cmp x21, x20
    b.hs no_exclam

    ldrb w22, [x19, x21]
    cmp w22, #0x21
    b.eq found_exclam

    add x21, x21, #1
    b find_loop

found_exclam:
    mov x25, x21
    add x26, x21, #1
    sub x27, x20, x21
    sub x27, x27, #1
    b validate_brackets

no_exclam:
    mov x25, x20
    mov x26, x20
    mov x27, #0

validate_brackets:
    adrp x28, bracket_stack
    add x28, x28, :lo12:bracket_stack
    mov x29, x28
    mov x30, #0

val_loop:
    cmp x30, x25
    b.hs val_done

    ldrb w0, [x19, x30]
    cmp w0, #'['
    b.eq val_open
    cmp w0, #']'
    b.eq val_close
    b val_next

val_open:
    str x30, [x29], #8
    b val_next

val_close:
    cmp x29, x28
    b.eq unbalanced

    sub x29, x29, #8
    ldr x1, [x29]
    adrp x2, jump_table
    add x2, x2, :lo12:jump_table
    str x1, [x2, x30, lsl #3]
    str x30, [x2, x1, lsl #3]
    b val_next

val_next:
    add x30, x30, #1
    b val_loop

val_done:
    cmp x29, x28
    b.ne unbalanced
    b setup_execution

unbalanced:
read_error:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

setup_execution:
    adrp x19, tape
    add x19, x19, :lo12:tape
    adrp x20, stdin_buffer
    add x20, x20, :lo12:stdin_buffer
    mov x21, x25
    mov x22, #0
    mov x23, #0
    adrp x24, stdin_buffer
    add x24, x24, :lo12:stdin_buffer
    add x24, x24, x26
    mov x25, x27
    mov x26, #0
    adrp x27, jump_table
    add x27, x27, :lo12:jump_table

execute_loop:
    cmp x22, x21
    b.hs execute_done

    ldrb w0, [x20, x22]
    cmp w0, #'>'
    b.eq cmd_right
    cmp w0, #'<'
    b.eq cmd_left
    cmp w0, #'+'
    b.eq cmd_plus
    cmp w0, #'-'
    b.eq cmd_minus
    cmp w0, #'.'
    b.eq cmd_dot
    cmp w0, #','
    b.eq cmd_comma
    cmp w0, #'['
    b.eq cmd_open
    cmp w0, #']'
    b.eq cmd_close
    add x22, x22, #1
    b execute_loop

cmd_right:
    add x23, x23, #1
    add x22, x22, #1
    b execute_loop

cmd_left:
    sub x23, x23, #1
    add x22, x22, #1
    b execute_loop

cmd_plus:
    ldrb w1, [x19, x23]
    add w1, w1, #1
    and w1, w1, #0xff
    strb w1, [x19, x23]
    add x22, x22, #1
    b execute_loop

cmd_minus:
    ldrb w1, [x19, x23]
    sub w1, w1, #1
    and w1, w1, #0xff
    strb w1, [x19, x23]
    add x22, x22, #1
    b execute_loop

cmd_dot:
    ldrb w3, [x19, x23]
    adrp x1, output_byte
    add x1, x1, :lo12:output_byte
    strb w3, [x1]
    mov x0, #STDOUT
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    add x22, x22, #1
    b execute_loop

cmd_comma:
    cmp x26, x25
    b.hs comma_eof
    ldrb w1, [x24, x26]
    strb w1, [x19, x23]
    add x26, x26, #1
    b comma_done
comma_eof:
    strb wzr, [x19, x23]
comma_done:
    add x22, x22, #1
    b execute_loop

cmd_open:
    ldrb w0, [x19, x23]
    cbnz w0, open_skip
    ldr x1, [x27, x22, lsl #3]
    mov x22, x1
    add x22, x22, #1
    b execute_loop
open_skip:
    add x22, x22, #1
    b execute_loop

cmd_close:
    ldrb w0, [x19, x23]
    cbz w0, close_skip
    ldr x1, [x27, x22, lsl #3]
    mov x22, x1
    add x22, x22, #1
    b execute_loop
close_skip:
    add x22, x22, #1
    b execute_loop

execute_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
```
