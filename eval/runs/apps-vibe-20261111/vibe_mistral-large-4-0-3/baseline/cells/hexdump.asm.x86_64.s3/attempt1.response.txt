.equ INPUT_MAX, 65536
.equ OUT_MAX, 323593

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero INPUT_MAX
out_buf:
    .zero OUT_MAX

.text
.globl _start
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]
read_loop:
    mov eax, 0
    xor edi, edi
    mov rsi, r13
    mov edx, INPUT_MAX
    sub edx, r12d
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    add r13, rax
    cmp r12d, INPUT_MAX
    jb read_loop
read_done:
    lea r8, [rip + hexdigits]
    lea rsi, [rip + out_buf]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae rows_done
    mov eax, r13d
    call store_hex8
    mov byte ptr [rsi], 0x20
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
    lea r14, [rip + input_buf]
    add r14, r13
    mov rcx, r12
    sub rcx, r13
    cmp rcx, 16
    jbe row_n_ready
    mov ecx, 16
row_n_ready:
    xor edx, edx
slot_loop:
    cmp edx, 16
    jae slots_done
    cmp edx, ecx
    jae slot_empty
    movzx eax, byte ptr [r14 + rdx]
    mov r9d, eax
    shr r9d, 4
    movzx r10d, byte ptr [r8 + r9]
    mov byte ptr [rsi], r10b
    inc rsi
    mov r9d, eax
    and r9d, 0xf
    movzx r10d, byte ptr [r8 + r9]
    mov byte ptr [rsi], r10b
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
    jmp slot_next
slot_empty:
    mov byte ptr [rsi], 0x20
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
    mov byte ptr [rsi], 0x20
    inc rsi
slot_next:
    cmp edx, 7
    jne slot_advance
    mov byte ptr [rsi], 0x20
    inc rsi
slot_advance:
    inc edx
    jmp slot_loop
slots_done:
    mov byte ptr [rsi], 0x20
    inc rsi
    mov byte ptr [rsi], 0x7c
    inc rsi
    xor edx, edx
ascii_loop:
    cmp edx, ecx
    jae ascii_done
    movzx eax, byte ptr [r14 + rdx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [rsi], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rsi], 0x2e
ascii_next:
    inc rsi
    inc edx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rsi], 0x7c
    inc rsi
    mov byte ptr [rsi], 0x0a
    inc rsi
    add r13, 16
    jmp row_loop
rows_done:
    mov eax, r12d
    call store_hex8
    mov byte ptr [rsi], 0x0a
    inc rsi
    mov r15, rsi
    lea r14, [rip + out_buf]
    sub r15, r14
write_loop:
    test r15, r15
    jz write_done
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall
    test rax, rax
    jle write_done
    add r14, rax
    sub r15, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall

store_hex8:
    mov ecx, 8
store_hex8_loop:
    rol eax, 4
    mov r9d, eax
    and r9d, 0xf
    movzx r10d, byte ptr [r8 + r9]
    mov byte ptr [rsi], r10b
    inc rsi
    dec ecx
    jnz store_hex8_loop
    ret
