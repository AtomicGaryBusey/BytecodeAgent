.equ BUFSZ, 65536
.equ OUTSZ, 64

.text
.globl _start
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
read_loop:
    mov x0, #0               // fd 0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #BUFSZ
    mov x8, #63              // read
    svc #0
    cmp x0, #0
    b.le input_done          // 0 (or error) = end of input
    add x21, x21, x0         // bytes += n
    adrp x10, buf
    add x10, x10, :lo12:buf
    mov x9, #0               // index
proc_loop:
    cmp x9, x0
    b.hs proc_done
    ldrb w12, [x10, x9]
    cmp w12, #0x0A
    b.eq byte_nl
    cmp w12, #0x20
    b.eq byte_ws
    sub w13, w12, #0x09
    cmp w13, #5              // 0x09..0x0D
    b.lo byte_ws
    cbnz x22, 1f
    add x20, x20, #1         // new word
1:
    mov x22, #1
    b next_byte
byte_ws:
    mov x22, #0
    b next_byte
byte_nl:
    add x19, x19, #1         // newline
    mov x22, #0
next_byte:
    add x9, x9, #1
    b proc_loop
proc_done:
    b read_loop
input_done:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #OUTSZ       // p = one past end
    sub x1, x1, #1
    mov w2, #0x0A
    strb w2, [x1]            // trailing newline at the very end
    mov x0, x21
    bl u64dec                // bytes (rightmost)
    sub x1, x1, #1
    mov w2, #0x20
    strb w2, [x1]
    mov x0, x20
    bl u64dec                // words
    sub x1, x1, #1
    mov w2, #0x20
    strb w2, [x1]
    mov x0, x19
    bl u64dec                // lines (leftmost)
    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    add x2, x2, #OUTSZ
    sub x2, x2, x1           // length
write_loop:
    cmp x2, #0
    b.eq write_done
    mov x0, #1               // fd 1
    mov x8, #64              // write
    svc #0
    cmp x0, #0
    b.le write_done
    sub x2, x2, x0
    add x1, x1, x0
    b write_loop
write_done:
    mov x0, #0
    mov x8, #93              // exit(0)
    svc #0

// u64dec: x0 = unsigned value, x1 = one-past-end pointer; stores decimal
// digits backwards, returns x1 = start of digits. Clobbers x0, x2, x3, x4.
u64dec:
    mov x2, #10
1:
    udiv x3, x0, x2
    msub x4, x3, x2, x0      // remainder
    add w4, w4, #48
    sub x1, x1, #1
    strb w4, [x1]
    mov x0, x3
    cbnz x0, 1b
    ret

.bss
.balign 8
buf:
    .zero BUFSZ
outbuf:
    .zero OUTSZ
