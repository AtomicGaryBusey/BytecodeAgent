.equ INPUT_SIZE, 65536

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, INPUT_SIZE
    syscall
    test rax, rax
    jle read_done
    mov rcx, rax
    lea rsi, [rip + input_buf]
process_loop:
    test rcx, rcx
    jz chunk_done
    movzx r9d, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp r9b, 0x0A
    jne 1f
    inc r12
1:
    mov r10d, r9d
    sub r10d, 0x09
    cmp r10d, 5
    jb 2f
    cmp r9b, 0x20
    je 2f
    test r15d, r15d
    jnz process_loop
    inc r13
    mov r15d, 1
    jmp process_loop
2:
    xor r15d, r15d
    jmp process_loop
chunk_done:
    add r14, rax
    jmp read_loop
read_done:
    lea rsi, [rip + counts_arr]
    mov qword ptr [rsi], r12
    mov qword ptr [rsi + 8], r13
    mov qword ptr [rsi + 16], r14
    lea rdi, [rip + out_end]
    dec rdi
    mov byte ptr [rdi], 0x0A
    lea rsi, [rip + counts_arr]
    mov ebx, 2
    mov r9, 10
build_loop:
    mov rax, qword ptr [rsi + rbx*8]
3:
    xor edx, edx
    div r9
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 3b
    test ebx, ebx
    jz build_done
    dec rdi
    mov byte ptr [rdi], 0x20
    dec ebx
    jmp build_loop
build_done:
    lea rsi, [rip + out_end]
    sub rsi, rdi
    mov rdx, rsi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE
counts_arr:
    .zero 24
out_buf:
    .zero 32
out_end:
