.equ BUFSIZE, 49152
.equ OUTSIZE, 66432
.equ LINE_LEN, 76

.text
.balign 4
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUFSIZE
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    mov x9, #BUFSIZE
    cmp x20, x9
    b.eq read_done
    b read_loop
read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x22, alphabet
    add x22, x22, :lo12:alphabet
    mov x23, #0
encode_loop:
    cmp x20, #3
    b.lt final_group
    ldrb w0, [x19], #1
    ldrb w1, [x19], #1
    ldrb w2, [x19], #1
    sub x20, x20, #3
    lsr w4, w0, #2
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    and w4, w1, #15
    lsl w4, w4, #2
    lsr w5, w2, #6
    orr w4, w4, w5
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    and w4, w2, #63
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    b encode_loop
final_group:
    cmp x20, #0
    b.eq encode_done
    ldrb w0, [x19], #1
    sub x20, x20, #1
    cmp x20, #0
    b.eq one_byte_left
    ldrb w1, [x19], #1
    lsr w4, w0, #2
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    and w4, w1, #15
    lsl w4, w4, #2
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    mov w9, #61
    bl emit
    b encode_done
one_byte_left:
    lsr w4, w0, #2
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    and w4, w0, #3
    lsl w4, w4, #4
    add x6, x22, x4
    ldrb w9, [x6]
    bl emit
    mov w9, #61
    bl emit
    mov w9, #61
    bl emit
encode_done:
    cmp x23, #0
    b.eq write_setup
    mov w9, #10
    strb w9, [x21], #1
write_setup:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x21, x1
write_loop:
    cmp x2, #0
    b.eq exit_prog
    mov x0, #1
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop
exit_prog:
    mov x0, #0
    mov x8, #93
    svc #0

emit:
    strb w9, [x21], #1
    add x23, x23, #1
    cmp x23, #LINE_LEN
    b.lt emit_ret
    mov w9, #10
    strb w9, [x21], #1
    mov x23, #0
emit_ret:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    .ascii "abcdefghijklmnopqrstuvwxyz"
    .ascii "0123456789"
    .ascii "+/"

.bss
.balign 8
inbuf:
    .zero BUFSIZE
outbuf:
    .zero OUTSIZE
