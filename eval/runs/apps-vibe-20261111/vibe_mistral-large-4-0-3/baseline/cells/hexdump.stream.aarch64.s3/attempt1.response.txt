.text
.globl _start
_start:
  adrp x10, inbuf
  add x10, x10, :lo12:inbuf
  mov x9, #0
read_loop:
  movz x2, #1, lsl #16
  sub x2, x2, x9
  cmp x2, #0
  b.eq read_done
  mov x0, #0
  mov x1, x10
  add x1, x1, x9
  mov x8, #63
  svc #0
  cmp x0, #0
  b.eq read_done
  b.mi read_error
  add x9, x9, x0
  b read_loop
read_done:
  adrp x17, hexdigits
  add x17, x17, :lo12:hexdigits
  mov x12, #0
  mov x13, x9
row_loop:
  cmp x13, #0
  b.eq final_line
  mov x14, x13
  cmp x14, #16
  b.ls row_n_done
  mov x14, #16
row_n_done:
  adrp x15, rowbuf
  add x15, x15, :lo12:rowbuf
  mov x0, x12
  mov x1, x15
  bl store_hex8
  mov x15, x1
  mov w2, #32
  strb w2, [x15], #1
  strb w2, [x15], #1
  mov x5, #0
pos_loop:
  cmp x5, x14
  b.hs missing_slot
  add x6, x10, x12
  ldrb w0, [x6, x5]
  mov x1, x15
  bl store_hex2
  mov x15, x1
  mov w2, #32
  strb w2, [x15], #1
  b slot_done
missing_slot:
  mov w2, #32
  strb w2, [x15], #1
  strb w2, [x15], #1
  strb w2, [x15], #1
slot_done:
  cmp x5, #7
  b.ne no_extra_space
  mov w2, #32
  strb w2, [x15], #1
no_extra_space:
  add x5, x5, #1
  cmp x5, #16
  b.lo pos_loop
  mov w2, #32
  strb w2, [x15], #1
  mov w2, #124
  strb w2, [x15], #1
  mov x5, #0
char_loop:
  cmp x5, x14
  b.hs char_done
  add x6, x10, x12
  ldrb w0, [x6, x5]
  cmp w0, #32
  b.lo char_dot
  cmp w0, #126
  b.hi char_dot
  b char_store
char_dot:
  mov w0, #46
char_store:
  strb w0, [x15], #1
  add x5, x5, #1
  b char_loop
char_done:
  mov w2, #124
  strb w2, [x15], #1
  mov w2, #10
  strb w2, [x15], #1
  adrp x1, rowbuf
  add x1, x1, :lo12:rowbuf
  sub x2, x15, x1
  mov x0, #1
  mov x8, #64
  svc #0
  add x12, x12, x14
  sub x13, x13, x14
  b row_loop
final_line:
  adrp x15, rowbuf
  add x15, x15, :lo12:rowbuf
  mov x1, x15
  mov x0, x9
  bl store_hex8
  mov w2, #10
  strb w2, [x1], #1
  sub x2, x1, x15
  mov x0, #1
  mov x1, x15
  mov x8, #64
  svc #0
  mov x0, #0
  mov x8, #93
  svc #0
read_error:
  mov x0, #1
  mov x8, #93
  svc #0

store_hex8:
  add x2, x1, #8
  mov x3, #8
store_hex8_loop:
  sub x2, x2, #1
  and w4, w0, #0xf
  ldrb w4, [x17, x4]
  strb w4, [x2]
  lsr x0, x0, #4
  subs x3, x3, #1
  b.ne store_hex8_loop
  add x1, x1, #8
  ret

store_hex2:
  ubfx w2, w0, #4, #4
  ldrb w3, [x17, x2]
  strb w3, [x1], #1
  and w2, w0, #0xf
  ldrb w3, [x17, x2]
  strb w3, [x1], #1
  ret

.section .rodata
.balign 16
hexdigits:
  .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
  .zero 65536
rowbuf:
  .zero 96
