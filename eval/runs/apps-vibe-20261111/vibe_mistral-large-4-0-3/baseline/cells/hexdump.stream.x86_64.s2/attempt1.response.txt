.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT_GROUP, 231
.equ IN_CAP, 65536
.equ OUT_CAP, 327680

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rax, IN_CAP
    sub rax, r12
    test rax, rax
    jz read_done
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, rax
    mov eax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + outbuf]
    xor r14, r14
row_loop:
    cmp r14, r12
    jae rows_done
    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_have_len
    mov r15d, 16
row_have_len:
    mov rax, r14
    call hex_qword
    mov word ptr [r13], 0x2020
    add r13, 2
    lea rsi, [rip + inbuf]
    add rsi, r14
    xor ebx, ebx
slot_loop:
    cmp ebx, 16
    jae slots_done
    cmp ebx, r15d
    jae slot_pad
    movzx eax, byte ptr [rsi + rbx]
    call hex_byte
    mov byte ptr [r13], 32
    inc r13
    jmp slot_after_byte
slot_pad:
    mov word ptr [r13], 0x2020
    mov byte ptr [r13 + 2], 32
    add r13, 3
slot_after_byte:
    cmp ebx, 7
    jne slot_next
    mov byte ptr [r13], 32
    inc r13
slot_next:
    inc ebx
    jmp slot_loop
slots_done:
    mov byte ptr [r13], 32
    inc r13
    mov byte ptr [r13], 124
    inc r13
    xor ebx, ebx
ascii_loop:
    cmp ebx, r15d
    jae ascii_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [r13], al
    jmp ascii_store
ascii_dot:
    mov byte ptr [r13], 46
ascii_store:
    inc r13
    inc ebx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r13], 124
    inc r13
    mov byte ptr [r13], 10
    inc r13
    add r14, r15
    jmp row_loop
rows_done:
    mov rax, r12
    call hex_qword
    mov byte ptr [r13], 10
    inc r13
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
write_loop:
    mov eax, SYS_WRITE
    syscall
    cmp rax, rdx
    jae write_done
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov edi, 0
    mov eax, SYS_EXIT_GROUP
    syscall

hex_byte:
    movzx eax, al
    mov ecx, eax
    shr eax, 4
    and ecx, 0xf
    cmp al, 10
    jb 1f
    add al, 87
    jmp 2f
1:
    add al, 48
2:
    mov byte ptr [r13], al
    inc r13
    movzx eax, cl
    cmp al, 10
    jb 3f
    add al, 87
    jmp 4f
3:
    add al, 48
4:
    mov byte ptr [r13], al
    inc r13
    ret

hex_qword:
    mov r9d, eax
    mov r8d, 4
1:
    rol r9d, 8
    movzx eax, r9b
    call hex_byte
    dec r8d
    jnz 1b
    ret

.bss
.balign 16
inbuf:
    .zero IN_CAP
outbuf:
    .zero OUT_CAP
