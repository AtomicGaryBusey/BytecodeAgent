.equ BUFSZ, 65600

.text
.globl _start

# json_lint: stdin must be exactly one valid JSON document (RFC 8259).
# valid   -> "ok\n" on stdout, exit 0
# invalid -> "error at N\n" on stdout, exit 1 (N = offset of first bad byte)
# register conventions: rbx = input buffer, r12 = current offset, r13 = length
_start:
    lea rbx, [rip + buf]
    xor r12d, r12d
read_loop:
    mov rdx, BUFSZ
    sub rdx, r12
    test rdx, rdx
    jz have_input
    xor edi, edi
    lea rsi, [rbx + r12]
    xor eax, eax
    syscall
    cmp rax, 0
    jle have_input
    add r12, rax
    jmp read_loop
have_input:
    mov r13, r12
    xor r12d, r12d
    call ws
    call parse_value
    call ws
    cmp r12, r13
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

# ws: skip whitespace bytes 0x20 0x09 0x0A 0x0D at r12
ws:
ws_loop:
    cmp r12, r13
    jae ws_ret
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x20
    je ws_adv
    cmp al, 0x09
    je ws_adv
    cmp al, 0x0A
    je ws_adv
    cmp al, 0x0D
    je ws_adv
    ret
ws_adv:
    inc r12
    jmp ws_loop
ws_ret:
    ret

# fail: print "error at <r12>\n" and exit(1); r12 is the offending offset
fail:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov rax, r12
    lea rdi, [rip + numtmp_end]
    mov ecx, 10
fail_conv:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz fail_conv
    mov rsi, rdi
    lea rax, [rip + numtmp_end]
    sub rax, rdi
    mov rdx, rax
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_nl]
    mov edx, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

# parse_value: parse one value at r12
parse_value:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2D
    je parse_number
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6E
    je pv_null
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    jmp parse_number
pv_true:
    lea rsi, [rip + lit_true]
    jmp match_lit
pv_false:
    lea rsi, [rip + lit_false]
    jmp match_lit
pv_null:
    lea rsi, [rip + lit_null]
match_lit:
ml_loop:
    movzx eax, byte ptr [rsi]
    test al, al
    jz ml_done
    cmp r12, r13
    jae fail
    cmp byte ptr [rbx + r12], al
    jne fail
    inc r12
    inc rsi
    jmp ml_loop
ml_done:
    ret

# parse_string: byte at r12 is the opening double quote
parse_string:
    inc r12
ps_loop:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x22
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb fail
    inc r12
    jmp ps_loop
ps_end:
    inc r12
    ret
ps_esc:
    inc r12
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x22
    je ps_esc1
    cmp al, 0x5C
    je ps_esc1
    cmp al, 0x2F
    je ps_esc1
    cmp al, 0x62
    je ps_esc1
    cmp al, 0x66
    je ps_esc1
    cmp al, 0x6E
    je ps_esc1
    cmp al, 0x72
    je ps_esc1
    cmp al, 0x74
    je ps_esc1
    cmp al, 0x75
    je ps_uesc
    jmp fail
ps_esc1:
    inc r12
    jmp ps_loop
ps_uesc:
    inc r12
    mov ecx, 4
ps_hex:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja ps_hex_alpha
    jmp ps_hex_ok
ps_hex_alpha:
    cmp al, 0x61
    jb ps_hex_upper
    cmp al, 0x66
    ja fail
    jmp ps_hex_ok
ps_hex_upper:
    cmp al, 0x41
    jb fail
    cmp al, 0x46
    ja fail
ps_hex_ok:
    inc r12
    dec ecx
    jnz ps_hex
    jmp ps_loop

# parse_number: byte at r12 is minus or a digit
parse_number:
    cmp byte ptr [rbx + r12], 0x2D
    jne pn_int
    inc r12
pn_int:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    je pn_zero
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
pn_digits:
    inc r12
    cmp r12, r13
    jae pn_frac
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    ja pn_frac
    jmp pn_digits
pn_zero:
    inc r12
pn_frac:
    cmp r12, r13
    jae pn_done
    cmp byte ptr [rbx + r12], 0x2E
    jne pn_exp
    inc r12
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
pn_fdigits:
    inc r12
    cmp r12, r13
    jae pn_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    ja pn_exp
    jmp pn_fdigits
pn_exp:
    cmp r12, r13
    jae pn_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x65
    je pn_expc
    cmp al, 0x45
    je pn_expc
    jmp pn_done
pn_expc:
    inc r12
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x2B
    je pn_expsign
    cmp al, 0x2D
    je pn_expsign
    jmp pn_expdigits
pn_expsign:
    inc r12
    cmp r12, r13
    jae fail
pn_expdigits:
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
pn_edigits:
    inc r12
    cmp r12, r13
    jae pn_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x30
    jb pn_done
    cmp al, 0x39
    ja pn_done
    jmp pn_edigits
pn_done:
    ret

# parse_array: byte at r12 is open bracket
parse_array:
    inc r12
    call ws
    cmp r12, r13
    jae fail
    cmp byte ptr [rbx + r12], 0x5D
    je pa_end
pa_elem:
    call parse_value
    call ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    je pa_end
    jmp fail
pa_comma:
    inc r12
    call ws
    jmp pa_elem
pa_end:
    inc r12
    ret

# parse_object: byte at r12 is open brace
parse_object:
    inc r12
    call ws
    cmp r12, r13
    jae fail
    cmp byte ptr [rbx + r12], 0x7D
    je po_end
po_member:
    cmp r12, r13
    jae fail
    cmp byte ptr [rbx + r12], 0x22
    jne fail
    call parse_string
    call ws
    cmp r12, r13
    jae fail
    cmp byte ptr [rbx + r12], 0x3A
    jne fail
    inc r12
    call ws
    call parse_value
    call ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    je po_end
    jmp fail
po_comma:
    inc r12
    call ws
    jmp po_member
po_end:
    inc r12
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
msg_nl:
    .ascii "\n"
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
buf:
    .zero BUFSZ
numtmp:
    .zero 24
numtmp_end:
