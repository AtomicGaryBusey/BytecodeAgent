.equ INPUT_MAX, 65536
.equ TAPE_SIZE, 30000
.equ MATCH_SIZE, 262144
.equ OPEN_SIZE, 4096

.equ C_BANG, 33
.equ C_GT, 62
.equ C_LT, 60
.equ C_PLUS, 43
.equ C_MINUS, 45
.equ C_DOT, 46
.equ C_COMMA, 44
.equ C_LB, 91
.equ C_RB, 93

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12, r12
read_loop:
    cmp r12, INPUT_MAX
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, INPUT_MAX
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rcx, rcx
scan_bang:
    cmp rcx, r12
    jae no_bang
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, C_BANG
    je found_bang
    inc rcx
    jmp scan_bang
no_bang:
    mov r14, r12
    mov r15, r12
    jmp validate
found_bang:
    mov r14, rcx
    lea r15, [rcx + 1]
validate:
    xor rcx, rcx
    xor r8, r8
validate_loop:
    cmp rcx, r14
    jae validate_end
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, C_LB
    je val_open
    cmp eax, C_RB
    je val_close
    jmp val_next
val_open:
    inc r8
    jmp val_next
val_close:
    test r8, r8
    jz invalid
    dec r8
    jmp val_next
val_next:
    inc rcx
    jmp validate_loop
validate_end:
    test r8, r8
    jnz invalid
    jmp build
invalid:
    mov rax, 60
    mov rdi, 1
    syscall
build:
    lea r9, [rip + match_tab]
    lea r10, [rip + open_stack]
    xor rcx, rcx
    xor r8, r8
build_loop:
    cmp rcx, r14
    jae build_done
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, C_LB
    je build_open
    cmp eax, C_RB
    je build_close
    jmp build_next
build_open:
    mov dword ptr [r10 + r8*4], ecx
    inc r8
    jmp build_next
build_close:
    dec r8
    mov eax, dword ptr [r10 + r8*4]
    mov dword ptr [r9 + rcx*4], eax
    mov dword ptr [r9 + rax*4], ecx
    jmp build_next
build_next:
    inc rcx
    jmp build_loop
build_done:
    lea rsi, [rip + tape]
    xor r8, r8
run_loop:
    cmp r8, r14
    jae run_done
    movzx eax, byte ptr [rbx + r8]
    cmp eax, C_GT
    je cmd_gt
    cmp eax, C_LT
    je cmd_lt
    cmp eax, C_PLUS
    je cmd_plus
    cmp eax, C_MINUS
    je cmd_minus
    cmp eax, C_DOT
    je cmd_dot
    cmp eax, C_COMMA
    je cmd_comma
    cmp eax, C_LB
    je cmd_open
    cmp eax, C_RB
    je cmd_close
    jmp cmd_next
cmd_gt:
    inc rsi
    jmp cmd_next
cmd_lt:
    dec rsi
    jmp cmd_next
cmd_plus:
    inc byte ptr [rsi]
    jmp cmd_next
cmd_minus:
    dec byte ptr [rsi]
    jmp cmd_next
cmd_dot:
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    jmp cmd_next
cmd_comma:
    cmp r15, r12
    jae comma_eof
    movzx eax, byte ptr [rbx + r15]
    mov byte ptr [rsi], al
    inc r15
    jmp cmd_next
comma_eof:
    mov byte ptr [rsi], 0
    jmp cmd_next
cmd_open:
    movzx eax, byte ptr [rsi]
    test eax, eax
    jnz open_enter
    mov eax, dword ptr [r9 + r8*4]
    lea r8, [rax + 1]
    jmp run_loop
open_enter:
    inc r8
    jmp run_loop
cmd_close:
    movzx eax, byte ptr [rsi]
    test eax, eax
    je close_exit
    mov eax, dword ptr [r9 + r8*4]
    lea r8, [rax + 1]
    jmp run_loop
close_exit:
    inc r8
    jmp run_loop
cmd_next:
    inc r8
    jmp run_loop
run_done:
    mov rax, 60
    mov rdi, 0
    syscall

.bss
.balign 16
input_buf:
    .zero INPUT_MAX
tape:
    .zero TAPE_SIZE
match_tab:
    .zero MATCH_SIZE
open_stack:
    .zero OPEN_SIZE
