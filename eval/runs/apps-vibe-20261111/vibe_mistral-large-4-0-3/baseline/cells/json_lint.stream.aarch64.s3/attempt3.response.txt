.equ BUF_SIZE, 131072

.globl _start

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
rd_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, rd_done
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x20, x20, x0
    b rd_loop
rd_done:
    mov x0, #0
    bl skip_ws
    bl parse_value
    cmn x0, #1
    b.ne top_ok
    mov x0, x21
    b report_err
top_ok:
    bl skip_ws
    cmp x0, x20
    b.hs all_ok
    b report_err
all_ok:
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// report_err: x0 = byte offset of first error; prints "error at N\n", exit 1
report_err:
    mov x21, x0
    adrp x1, msgbuf
    add x1, x1, :lo12:msgbuf
    adrp x2, prefix
    add x2, x2, :lo12:prefix
    mov x6, #0
re_pfx:
    cmp x6, #9
    b.hs re_pfx_done
    ldrb w3, [x2, x6]
    strb w3, [x1, x6]
    add x6, x6, #1
    b re_pfx
re_pfx_done:
    add x4, x1, #9
    adrp x1, digbuf
    add x1, x1, :lo12:digbuf
    add x1, x1, #8
    mov x5, #0
re_cvt:
    mov x9, #10
    udiv x7, x21, x9
    msub x8, x7, x9, x21
    add w8, w8, #0x30
    sub x1, x1, #1
    strb w8, [x1]
    add x5, x5, #1
    mov x21, x7
    cbnz x21, re_cvt
    mov x6, #0
re_cpy:
    cmp x6, x5
    b.hs re_cpy_done
    ldrb w3, [x1, x6]
    strb w3, [x4, x6]
    add x6, x6, #1
    b re_cpy
re_cpy_done:
    add x6, x4, x5
    mov w3, #0x0A
    strb w3, [x6]
    mov x0, #1
    adrp x1, msgbuf
    add x1, x1, :lo12:msgbuf
    add x2, x5, #10
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip_ws: x0 = index -> x0 = index after run of ws (leaf)
skip_ws:
sw_loop:
    cmp x0, x20
    b.hs sw_done
    ldrb w1, [x19, x0]
    cmp w1, #0x20
    b.eq sw_next
    cmp w1, #0x09
    b.eq sw_next
    cmp w1, #0x0A
    b.eq sw_next
    cmp w1, #0x0D
    b.eq sw_next
    b sw_done
sw_next:
    add x0, x0, #1
    b sw_loop
sw_done:
    ret

// parse_value: x0 = index of value start -> x0 = next index, or -1 with err in x21
parse_value:
    str x30, [sp, #-16]!
    cmp x0, x20
    b.hs pv_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x7B
    b.eq pv_obj
    cmp w1, #0x5B
    b.eq pv_arr
    cmp w1, #0x22
    b.eq pv_str
    cmp w1, #0x2D
    b.eq pv_num
    cmp w1, #0x74
    b.eq pv_true
    cmp w1, #0x66
    b.eq pv_false
    cmp w1, #0x6E
    b.eq pv_null
    sub w2, w1, #0x30
    cmp w2, #9
    b.ls pv_num
    b pv_err
pv_obj:
    bl parse_object
    b pv_ret
pv_arr:
    bl parse_array
    b pv_ret
pv_str:
    bl parse_string
    b pv_ret
pv_num:
    bl parse_number
    b pv_ret
pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    mov x2, #4
    bl match_lit
    b pv_ret
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    mov x2, #5
    bl match_lit
    b pv_ret
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    mov x2, #4
    bl match_lit
    b pv_ret
pv_eof:
    mov x21, x20
    mov x0, #-1
    b pv_ret
pv_err:
    mov x21, x0
    mov x0, #-1
pv_ret:
    ldr x30, [sp], #16
    ret

// match_lit: x0 = index, x1 = literal bytes, x2 = literal length (leaf)
match_lit:
    mov x4, #0
ml_loop:
    cmp x4, x2
    b.eq ml_ok
    add x5, x0, x4
    cmp x5, x20
    b.hs ml_eof
    ldrb w6, [x19, x5]
    ldrb w7, [x1, x4]
    cmp w6, w7
    b.ne ml_err
    add x4, x4, #1
    b ml_loop
ml_ok:
    add x0, x0, x2
    ret
ml_eof:
    mov x21, x20
    b ml_fail
ml_err:
    mov x21, x5
ml_fail:
    mov x0, #-1
    ret

// parse_object: x0 = index of '{'
parse_object:
    str x30, [sp, #-16]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.hs po_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x7D
    b.eq po_done
po_loop:
    cmp x0, x20
    b.hs po_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x22
    b.ne po_err
    bl parse_string
    cmn x0, #1
    b.eq po_ret
    bl skip_ws
    cmp x0, x20
    b.hs po_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x3A
    b.ne po_err
    add x0, x0, #1
    bl skip_ws
    bl parse_value
    cmn x0, #1
    b.eq po_ret
    bl skip_ws
    cmp x0, x20
    b.hs po_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x2C
    b.eq po_comma
    cmp w1, #0x7D
    b.eq po_done
    b po_err
po_comma:
    add x0, x0, #1
    bl skip_ws
    b po_loop
po_done:
    add x0, x0, #1
po_ret:
    ldr x30, [sp], #16
    ret
po_eof:
    mov x21, x20
    mov x0, #-1
    b po_ret
po_err:
    mov x21, x0
    mov x0, #-1
    b po_ret

// parse_array: x0 = index of '['
parse_array:
    str x30, [sp, #-16]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.hs pa_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x5D
    b.eq pa_done
pa_loop:
    bl parse_value
    cmn x0, #1
    b.eq pa_ret
    bl skip_ws
    cmp x0, x20
    b.hs pa_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x2C
    b.eq pa_comma
    cmp w1, #0x5D
    b.eq pa_done
    b pa_err
pa_comma:
    add x0, x0, #1
    bl skip_ws
    b pa_loop
pa_done:
    add x0, x0, #1
pa_ret:
    ldr x30, [sp], #16
    ret
pa_eof:
    mov x21, x20
    mov x0, #-1
    b pa_ret
pa_err:
    mov x21, x0
    mov x0, #-1
    b pa_ret

// parse_string: x0 = index of '"' (leaf)
parse_string:
    add x0, x0, #1
ps_loop:
    cmp x0, x20
    b.hs ps_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x22
    b.eq ps_done
    cmp w1, #0x5C
    b.eq ps_esc
    cmp w1, #0x20
    b.lo ps_err
    add x0, x0, #1
    b ps_loop
ps_esc:
    add x0, x0, #1
    cmp x0, x20
    b.hs ps_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x22
    b.eq ps_esc1
    cmp w1, #0x5C
    b.eq ps_esc1
    cmp w1, #0x2F
    b.eq ps_esc1
    cmp w1, #0x62
    b.eq ps_esc1
    cmp w1, #0x66
    b.eq ps_esc1
    cmp w1, #0x6E
    b.eq ps_esc1
    cmp w1, #0x72
    b.eq ps_esc1
    cmp w1, #0x74
    b.eq ps_esc1
    cmp w1, #0x75
    b.eq ps_u
    b ps_err
ps_esc1:
    add x0, x0, #1
    b ps_loop
ps_u:
    add x0, x0, #1
    mov x2, #4
ps_hex:
    cmp x0, x20
    b.hs ps_eof
    ldrb w1, [x19, x0]
    sub w3, w1, #0x30
    cmp w3, #9
    b.ls ps_hexok
    sub w3, w1, #0x61
    cmp w3, #5
    b.ls ps_hexok
    sub w3, w1, #0x41
    cmp w3, #5
    b.ls ps_hexok
    b ps_err
ps_hexok:
    add x0, x0, #1
    sub x2, x2, #1
    cbnz x2, ps_hex
    b ps_loop
ps_done:
    add x0, x0, #1
    ret
ps_eof:
    mov x21, x20
    mov x0, #-1
    ret
ps_err:
    mov x21, x0
    mov x0, #-1
    ret

// parse_number: x0 = index of '-' or digit (leaf)
parse_number:
    ldrb w1, [x19, x0]
    cmp w1, #0x2D
    b.ne pn_int
    add x0, x0, #1
pn_int:
    cmp x0, x20
    b.hs pn_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x30
    b.eq pn_zero
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi pn_err
pn_dloop:
    add x0, x0, #1
pn_dchk:
    cmp x0, x20
    b.hs pn_frac
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.ls pn_dloop
    b pn_frac
pn_zero:
    add x0, x0, #1
pn_frac:
    cmp x0, x20
    b.hs pn_ok
    ldrb w1, [x19, x0]
    cmp w1, #0x2E
    b.ne pn_exp
    add x0, x0, #1
    cmp x0, x20
    b.hs pn_eof
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi pn_err
pn_floop:
    add x0, x0, #1
pn_fchk:
    cmp x0, x20
    b.hs pn_ok
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.ls pn_floop
    b pn_exp
pn_exp:
    cmp x0, x20
    b.hs pn_ok
    ldrb w1, [x19, x0]
    cmp w1, #0x65
    b.eq pn_estar
    cmp w1, #0x45
    b.eq pn_estar
    b pn_ok
pn_estar:
    add x0, x0, #1
    cmp x0, x20
    b.hs pn_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x2B
    b.eq pn_esign
    cmp w1, #0x2D
    b.eq pn_esign
    b pn_edig
pn_esign:
    add x0, x0, #1
pn_edig:
    cmp x0, x20
    b.hs pn_eof
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi pn_err
pn_eloop:
    add x0, x0, #1
pn_echk:
    cmp x0, x20
    b.hs pn_ok
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.ls pn_eloop
pn_ok:
    ret
pn_eof:
    mov x21, x20
    mov x0, #-1
    ret
pn_err:
    mov x21, x0
    mov x0, #-1
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
prefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
buf:
    .zero BUF_SIZE
digbuf:
    .zero 8
msgbuf:
    .zero 24
