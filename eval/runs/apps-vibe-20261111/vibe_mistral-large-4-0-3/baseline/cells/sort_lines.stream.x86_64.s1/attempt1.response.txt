.text
.equ MAXBYTES, 32768
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, MAXBYTES
    sub edx, ebx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
scan_loop:
    cmp r14, rbx
    jae scan_done
    movzx edx, byte ptr [input_buf + r14]
    cmp dl, 0x0A
    jne 1f
    mov eax, r12d
    mov dword ptr [line_off + rax*4], r13d
    mov edx, r14d
    sub edx, r13d
    mov dword ptr [line_len + rax*4], edx
    inc r12d
    mov r13, r14
    inc r13
1:
    inc r14
    jmp scan_loop
scan_done:
    cmp r13, rbx
    jae no_final
    mov eax, r12d
    mov dword ptr [line_off + rax*4], r13d
    mov edx, ebx
    sub edx, r13d
    mov dword ptr [line_len + rax*4], edx
    inc r12d
no_final:
    xor ecx, ecx
init_order:
    cmp rcx, r12
    jae init_done
    mov dword ptr [order + rcx*4], ecx
    inc rcx
    jmp init_order
init_done:
    cmp r12, 1
    jbe sort_done
    mov r13, 1
i_loop:
    cmp r13, r12
    jae sort_done
    mov eax, dword ptr [order + r13*4]
    mov r15d, eax
    mov r14, r13
    dec r14
inner:
    cmp r14, 0
    jl inner_done
    mov eax, dword ptr [order + r14*4]
    mov edi, eax
    mov esi, r15d
    call cmp_lines
    cmp eax, 0
    jle inner_done
    mov eax, dword ptr [order + r14*4]
    mov ecx, r14
    inc ecx
    mov dword ptr [order + rcx*4], eax
    dec r14
    jmp inner
inner_done:
    mov ecx, r14
    inc ecx
    mov dword ptr [order + rcx*4], r15d
    inc r13
    jmp i_loop
sort_done:
    xor r13d, r13d
out_loop:
    cmp r13, r12
    jae out_done
    mov eax, dword ptr [order + r13*4]
    mov edx, dword ptr [line_len + rax*4]
    test edx, edx
    jz skip_line
    mov ecx, dword ptr [line_off + rax*4]
    lea rsi, [rip + input_buf]
    add rsi, rcx
    mov edi, 1
    mov eax, 1
    syscall
skip_line:
    lea rsi, [rip + newline]
    mov edi, 1
    mov edx, 1
    mov eax, 1
    syscall
    inc r13
    jmp out_loop
out_done:
    mov edi, 0
    mov eax, 60
    syscall

cmp_lines:
    mov r9d, dword ptr [line_len + rdi*4]
    mov r10d, dword ptr [line_len + rsi*4]
    mov ecx, dword ptr [line_off + rdi*4]
    mov edx, dword ptr [line_off + rsi*4]
    lea rsi, [rip + input_buf]
    add rsi, rcx
    lea rdi, [rip + input_buf]
    add rdi, rdx
    mov rcx, r9
    cmp r10, rcx
    cmovb rcx, r10
    xor r8d, r8d
1:
    mov rax, rcx
    sub rax, r8
    cmp rax, 8
    jb 2f
    mov rax, qword ptr [rsi + r8]
    bswap rax
    mov rdx, qword ptr [rdi + r8]
    bswap rdx
    cmp rax, rdx
    jb 3f
    ja 4f
    add r8, 8
    jmp 1b
2:
    cmp r8, rcx
    jae 5f
    movzx eax, byte ptr [rsi + r8]
    movzx edx, byte ptr [rdi + r8]
    cmp al, dl
    jb 3f
    ja 4f
    inc r8
    jmp 2b
5:
    cmp r9, r10
    jb 3f
    ja 4f
    xor eax, eax
    ret
3:
    mov eax, -1
    ret
4:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 8
input_buf:
    .zero 32769
.balign 8
line_off:
    .zero 131072
.balign 8
line_len:
    .zero 131072
.balign 8
order:
    .zero 131072
