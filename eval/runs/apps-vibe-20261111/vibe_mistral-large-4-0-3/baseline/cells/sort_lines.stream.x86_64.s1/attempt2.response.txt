.equ MAXBYTES, 32768
.equ MAXLINES, 1000

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, MAXBYTES
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, MAXBYTES
    sub edx, ebx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    lea r15, [rip + input_buf]
scan_loop:
    cmp r14, rbx
    jae scan_done
    movzx edx, byte ptr [r15 + r14]
    cmp dl, 0x0A
    jne 1f
    mov eax, r12d
    mov dword ptr [line_off + rax*4], r13d
    mov edx, r14d
    sub edx, r13d
    mov dword ptr [line_len + rax*4], edx
    inc r12d
    mov r13, r14
    inc r13
1:
    inc r14
    jmp scan_loop
scan_done:
    cmp r13, rbx
    jae no_final
    mov eax, r12d
    mov dword ptr [line_off + rax*4], r13d
    mov edx, ebx
    sub edx, r13d
    mov dword ptr [line_len + rax*4], edx
    inc r12d
no_final:
    lea rbx, [rip + input_buf]
    xor ecx, ecx
init_order:
    cmp rcx, r12
    jae init_done
    mov dword ptr [order + rcx*4], ecx
    inc rcx
    jmp init_order
init_done:
    cmp r12, 1
    jbe sort_done
    mov r13, 1
i_loop:
    cmp r13, r12
    jae sort_done
    mov eax, dword ptr [order + r13*4]
    mov r15d, eax
    mov r14, r13
    dec r14
inner:
    cmp r14, 0
    jl inner_done
    mov eax, dword ptr [order + r14*4]
    mov edi, eax
    mov esi, r15d
    mov eax, dword ptr [line_off + rdi*4]
    mov r8d, dword ptr [line_len + rdi*4]
    mov ecx, dword ptr [line_off + rsi*4]
    mov r9d, dword ptr [line_len + rsi*4]
    mov r10, r8
    cmp r8, r9
    jbe 2f
    mov r10, r9
2:
    xor edx, edx
3:
    mov r11, rdx
    add r11, 8
    cmp r11, r10
    ja 4f
    mov r11, rax
    add r11, rdx
    movbe rsi, qword ptr [rbx + r11]
    mov rdi, rcx
    add rdi, rdx
    movbe r11, qword ptr [rbx + rdi]
    cmp rsi, r11
    jb a_less
    ja a_greater
    add rdx, 8
    jmp 3b
4:
    cmp rdx, r10
    jae cmp_len
    mov r11, rax
    add r11, rdx
    movzx esi, byte ptr [rbx + r11]
    mov rdi, rcx
    add rdi, rdx
    movzx r11d, byte ptr [rbx + rdi]
    cmp esi, r11d
    jb a_less
    ja a_greater
    inc rdx
    jmp 4b
cmp_len:
    cmp r8, r9
    jb a_less
    ja a_greater
    xor eax, eax
    jmp cmp_done
a_less:
    mov eax, 0xFFFFFFFF
    jmp cmp_done
a_greater:
    mov eax, 1
cmp_done:
    cmp eax, 0
    jle inner_done
    mov eax, dword ptr [order + r14*4]
    mov ecx, r14d
    inc ecx
    mov dword ptr [order + rcx*4], eax
    dec r14
    jmp inner
inner_done:
    mov rcx, r14
    inc rcx
    mov eax, r15d
    mov dword ptr [order + rcx*4], eax
    inc r13
    jmp i_loop
sort_done:
    xor ecx, ecx
out_loop:
    cmp rcx, r12
    jae out_done
    mov eax, dword ptr [order + rcx*4]
    mov r8d, dword ptr [line_off + rax*4]
    mov r9d, dword ptr [line_len + rax*4]
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, r9
    mov edi, 1
    mov eax, 1
write_line:
    test rdx, rdx
    jz line_written
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_line
line_written:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rcx
    jmp out_loop
out_done:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 8
input_buf:
    .zero MAXBYTES
line_off:
    .zero MAXLINES*4
line_len:
    .zero MAXLINES*4
order:
    .zero MAXLINES*4
