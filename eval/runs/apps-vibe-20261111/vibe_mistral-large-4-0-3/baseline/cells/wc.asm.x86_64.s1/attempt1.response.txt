.text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor r14, r14
read_loop:
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    xor eax, eax
    syscall
    test rax, rax
    jz done
    mov rcx, rax
    lea rsi, [rip + buf]
process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r10
    cmp al, 0x0A
    je newline_byte
    cmp al, 0x09
    jb not_space
    cmp al, 0x0D
    ja not_space
    xor r14, r14
    jmp process_next
not_space:
    cmp al, 0x20
    je space_byte
    test r14, r14
    jnz process_next
    inc r9
    mov r14, 1
    jmp process_next
space_byte:
    xor r14, r14
    jmp process_next
newline_byte:
    inc r8
    xor r14, r14
process_next:
    jmp process_loop
done:
    mov qword ptr [rip + counts], r8
    mov qword ptr [rip + counts + 8], r9
    mov qword ptr [rip + counts + 16], r10
    lea r15, [rip + outbuf]
    lea rbx, [rip + counts]
    xor r12, r12
    mov r13, 10
out_item_loop:
    mov rax, qword ptr [rbx + r12*8]
    lea rsi, [rip + numend]
conv_loop:
    xor edx, edx
    div r13
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz conv_loop
    lea rdx, [rip + numend]
copy_loop:
    cmp rsi, rdx
    jae copy_done
    movzx eax, byte ptr [rsi]
    mov byte ptr [r15], al
    inc rsi
    inc r15
    jmp copy_loop
copy_done:
    cmp r12, 2
    jae final_sep
    mov byte ptr [r15], 32
    inc r15
    jmp next_item
final_sep:
    mov byte ptr [r15], 10
    inc r15
next_item:
    inc r12
    cmp r12, 3
    jb out_item_loop
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_prog
    mov eax, 1
    syscall
    cmp rax, 0
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    xor edi, edi
    mov eax, 231
    syscall

.bss
outbuf:
    .zero 64
numbuf:
    .zero 32
numend:
counts:
    .zero 24
buf:
    .zero 4096
