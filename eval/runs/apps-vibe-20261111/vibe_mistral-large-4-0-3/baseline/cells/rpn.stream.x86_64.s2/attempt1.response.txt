.equ INPUT_CAP, 65536
.equ ERROR_LEN, 6

.section .rodata
error_msg:
    .byte 101,114,114,111,114,10

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rax, INPUT_CAP
    sub rax, r12
    test rax, rax
    jz read_done
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, rax
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + input_buf]
    xor rbx, rbx
line_loop:
    cmp rbx, r12
    jae all_done
    mov r14, rbx
    mov rcx, rbx
scan_nl:
    cmp rcx, r12
    jae no_nl
    movzx eax, byte ptr [r13 + rcx]
    cmp al, 10
    je found_nl
    inc rcx
    jmp scan_nl
no_nl:
    mov r15, r12
    mov rbx, r12
    jmp process_line
found_nl:
    mov r15, rcx
    lea rbx, [rcx + 1]
process_line:
    xor r8d, r8d
    mov r9, r14
token_loop:
    cmp r9, r15
    jae line_finish
    movzx eax, byte ptr [r13 + r9]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    mov r10, r9
tok_scan:
    inc r9
    cmp r9, r15
    jae tok_done
    movzx eax, byte ptr [r13 + r9]
    cmp al, 32
    je tok_done
    cmp al, 9
    je tok_done
    jmp tok_scan
tok_done:
    mov rsi, r10
    mov rdi, r9
    call do_token
    test al, al
    jnz line_error
    jmp token_loop
skip_sep:
    inc r9
    jmp token_loop
line_finish:
    cmp r8, 1
    jne line_error
    lea r11, [val_stack]
    mov rax, qword ptr [r11]
    call print_number
    jmp line_loop
line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, ERROR_LEN
    syscall
    jmp line_loop
all_done:
    mov eax, 60
    xor edi, edi
    syscall

do_token:
    mov rcx, rdi
    sub rcx, rsi
    cmp rcx, 1
    jne try_literal
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod
    jmp try_literal
try_literal:
    mov rcx, rsi
    xor r11d, r11d
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 45
    jne lit_digits
    mov r11d, 1
    inc rcx
lit_digits:
    mov rdx, rdi
    sub rdx, rcx
    test rdx, rdx
    jz invalid
    cmp rdx, 18
    ja invalid
    xor eax, eax
lit_loop:
    movzx edx, byte ptr [r13 + rcx]
    sub dl, 48
    cmp dl, 9
    ja invalid
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    cmp rcx, rdi
    jb lit_loop
    test r11d, r11d
    jz lit_push
    neg rax
lit_push:
    lea r11, [val_stack + r8*8]
    mov qword ptr [r11], rax
    inc r8
    xor al, al
    ret
invalid:
    mov al, 1
    ret
op_add:
    cmp r8, 2
    jb invalid
    dec r8
    lea r11, [val_stack + r8*8]
    mov rcx, qword ptr [r11]
    dec r8
    lea r11, [val_stack + r8*8]
    mov rax, qword ptr [r11]
    add rax, rcx
    lea r11, [val_stack + r8*8]
    mov qword ptr [r11], rax
    inc r8
    xor al, al
    ret
op_sub:
    cmp r8, 2
    jb invalid
    dec r8
    lea r11, [val_stack + r8*8]
    mov rcx, qword ptr [r11]
    dec r8
    lea r11, [val_stack + r8*8]
    mov rax, qword ptr [r11]
    sub rax, rcx
    lea r11, [val_stack + r8*8]
    mov qword ptr [r11], rax
    inc r8
    xor al, al
    ret
op_mul:
    cmp r8, 2
    jb invalid
    dec r8
    lea r11, [val_stack + r8*8]
    mov rcx, qword ptr [r11]
    dec r8
    lea r11, [val_stack + r8*8]
    mov rax, qword ptr [r11]
    imul rax, rcx
    lea r11, [val_stack + r8*8]
    mov qword ptr [r11], rax
    inc r8
    xor al, al
    ret
op_div:
    cmp r8, 2
    jb invalid
    dec r8
    lea r11, [val_stack + r8*8]
    mov rcx, qword ptr [r11]
    dec r8
    lea r11, [val_stack + r8*8]
    mov rax, qword ptr [r11]
    test rcx, rcx
    jz invalid
    mov r11, 0x8000000000000000
    cmp rax, r11
    jne div_safe
    cmp rcx, -1
    je invalid
div_safe:
    cqo
    idiv rcx
    lea r11, [val_stack + r8*8]
    mov qword ptr [r11], rax
    inc r8
    xor al, al
    ret
op_mod:
    cmp r8, 2
    jb invalid
    dec r8
    lea r11, [val_stack + r8*8]
    mov rcx, qword ptr [r11]
    dec r8
    lea r11, [val_stack + r8*8]
    mov rax, qword ptr [r11]
    test rcx, rcx
    jz invalid
    mov r11, 0x8000000000000000
    cmp rax, r11
    jne mod_safe
    cmp rcx, -1
    je invalid
mod_safe:
    cqo
    idiv rcx
    mov rax, rdx
    lea r11, [val_stack + r8*8]
    mov qword ptr [r11], rax
    inc r8
    xor al, al
    ret

print_number:
    lea rdi, [rip + out_buf + 32]
    dec rdi
    mov byte ptr [rdi], 10
    test rax, rax
    jns pn_nonneg
    dec rdi
    mov byte ptr [rdi], 45
    neg rax
pn_nonneg:
    mov ecx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pn_loop
    mov rsi, rdi
    lea rdx, [rip + out_buf + 32]
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    ret

.bss
.balign 8
input_buf:
    .space INPUT_CAP
val_stack:
    .space 524288
out_buf:
    .space 64
