.text
.equ INBUF_SIZE, 4096
.globl _start
_start:
    xor ebx, ebx
    xor ebp, ebp
    lea r15, [rip + rowdata]
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, INBUF_SIZE
    syscall
    test rax, rax
    jle eof
    mov r12, rax
    lea r13, [rip + inbuf]
byte_loop:
    movzx r14d, byte ptr [r13]
    mov byte ptr [r15 + rbp], r14b
    inc rbp
    inc rbx
    cmp rbp, 16
    jne byte_loop_next
    call emit_row
    xor ebp, ebp
byte_loop_next:
    inc r13
    dec r12
    jnz byte_loop
    jmp read_loop
eof:
    test rbp, rbp
    jz final_line
    call emit_row
final_line:
    lea rsi, [rip + rowbuf]
    lea r8, [rip + rowbuf]
    lea r10, [rip + hexchars]
    mov rax, rbx
    mov cl, 28
final_digit:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xf
    movzx r11d, byte ptr [r10 + rdx]
    mov byte ptr [r8], r11b
    inc r8
    sub cl, 4
    jns final_digit
    mov byte ptr [r8], 0x0a
    inc r8
    mov edi, 1
    mov rdx, r8
    sub rdx, rsi
    call write_all
    xor edi, edi
    mov eax, 60
    syscall

write_all:
    test rdx, rdx
    jz write_all_done
    mov eax, 1
    syscall
    cmp rax, 0
    jle write_all_done
    add rsi, rax
    sub rdx, rax
    jmp write_all
write_all_done:
    ret

emit_row:
    lea r8, [rip + rowbuf]
    lea r10, [rip + hexchars]
    mov rax, rbx
    sub rax, rbp
    mov cl, 28
emit_off_digit:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xf
    movzx r11d, byte ptr [r10 + rdx]
    mov byte ptr [r8], r11b
    inc r8
    sub cl, 4
    jns emit_off_digit
    mov byte ptr [r8], 0x20
    inc r8
    mov byte ptr [r8], 0x20
    inc r8
    xor r9d, r9d
emit_pos:
    cmp r9, rbp
    jae emit_pos_pad
    movzx eax, byte ptr [r15 + r9]
    mov rdx, rax
    shr rdx, 4
    and rdx, 0xf
    movzx r11d, byte ptr [r10 + rdx]
    mov byte ptr [r8], r11b
    inc r8
    mov rdx, rax
    and rdx, 0xf
    movzx r11d, byte ptr [r10 + rdx]
    mov byte ptr [r8], r11b
    inc r8
    mov byte ptr [r8], 0x20
    inc r8
    jmp emit_pos_after
emit_pos_pad:
    mov byte ptr [r8], 0x20
    inc r8
    mov byte ptr [r8], 0x20
    inc r8
    mov byte ptr [r8], 0x20
    inc r8
emit_pos_after:
    cmp r9, 7
    jne emit_pos_next
    mov byte ptr [r8], 0x20
    inc r8
emit_pos_next:
    inc r9
    cmp r9, 16
    jb emit_pos
    mov byte ptr [r8], 0x20
    inc r8
    mov byte ptr [r8], 0x7c
    inc r8
    xor r9d, r9d
emit_ascii:
    cmp r9, rbp
    jae emit_ascii_done
    movzx eax, byte ptr [r15 + r9]
    cmp al, 0x20
    jb emit_dot
    cmp al, 0x7e
    ja emit_dot
    mov byte ptr [r8], al
    jmp emit_ascii_next
emit_dot:
    mov byte ptr [r8], 0x2e
emit_ascii_next:
    inc r8
    inc r9
    jmp emit_ascii
emit_ascii_done:
    mov byte ptr [r8], 0x7c
    inc r8
    mov byte ptr [r8], 0x0a
    inc r8
    lea rsi, [rip + rowbuf]
    mov edi, 1
    mov rdx, rbp
    add rdx, 63
    call write_all
    ret

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
rowdata:
    .space 16
rowbuf:
    .space 80
