.text
.globl _start
_start:
    // Read all of standard input into inbuf (at most 49152 bytes).
    adrp    x9, inbuf
    add     x9, x9, :lo12:inbuf        // x9 = inbuf base
    mov     x20, #0                    // x20 = bytes read so far
read_loop:
    mov     x0, #0                     // fd 0
    mov     x1, x9
    add     x1, x1, x20                // buf = inbuf + bytes read
    mov     x2, #49152
    sub     x2, x2, x20                // count = 49152 - bytes read
    mov     x8, #63                    // read
    svc     #0
    cmp     x0, #0
    b.le    read_done                  // 0 = end of input
    add     x20, x20, x0
    cmp     x20, #49152
    b.lt    read_loop
read_done:
    mov     x19, x9                    // x19 = input cursor
    add     x20, x9, x20               // x20 = input end
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf     // x21 = output cursor
    mov     x22, #0                    // x22 = chars in current line
    adrp    x23, alphabet
    add     x23, x23, :lo12:alphabet   // x23 = base64 alphabet table
encode_loop:
    sub     x24, x20, x19              // bytes remaining
    cmp     x24, #3
    b.lt    encode_tail
    ldrb    w0, [x19]
    ldrb    w1, [x19, #1]
    ldrb    w2, [x19, #2]
    lsl     w0, w0, #16
    orr     w0, w0, w1, lsl #8
    orr     w0, w0, w2                 // w0 = b0<<16 | b1<<8 | b2
    lsr     w3, w0, #18                // char 0
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    lsr     w3, w0, #12
    and     w3, w3, #63                // char 1
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    lsr     w3, w0, #6
    and     w3, w3, #63                // char 2
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    and     w3, w0, #63                // char 3
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    add     x22, x22, #4
    cmp     x22, #76
    b.ne    no_wrap
    mov     w3, #10                    // newline after a full 76-char line
    strb    w3, [x21], #1
    mov     x22, #0
no_wrap:
    add     x19, x19, #3
    b       encode_loop
encode_tail:
    sub     x24, x20, x19
    cmp     x24, #0
    b.eq    final_nl
    ldrb    w0, [x19]
    cmp     x24, #1
    b.eq    tail1
    // two bytes left: 3 chars + '='
    ldrb    w1, [x19, #1]
    lsl     w0, w0, #16
    orr     w0, w0, w1, lsl #8
    lsr     w3, w0, #18
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    lsr     w3, w0, #12
    and     w3, w3, #63
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    lsr     w3, w0, #6
    and     w3, w3, #63
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    mov     w3, #61                    // '='
    strb    w3, [x21], #1
    b       tail_done
tail1:
    // one byte left: 2 chars + "=="
    lsl     w0, w0, #16
    lsr     w3, w0, #18
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    lsr     w3, w0, #12
    and     w3, w3, #63
    ldrb    w3, [x23, w3, uxtw]
    strb    w3, [x21], #1
    mov     w3, #61                    // '='
    strb    w3, [x21], #1
    mov     w3, #61                    // '='
    strb    w3, [x21], #1
tail_done:
    add     x22, x22, #4
final_nl:
    cmp     x22, #0
    b.eq    write_setup
    mov     w3, #10                    // newline at end of last (short) line
    strb    w3, [x21], #1
write_setup:
    adrp    x5, outbuf
    add     x5, x5, :lo12:outbuf
    sub     x6, x21, x5                // x6 = total output length
    mov     x7, x5                     // x7 = write cursor
write_loop:
    cmp     x6, #0
    b.eq    exit0
    mov     x0, #1                     // fd 1
    mov     x1, x7
    mov     x2, x6
    mov     x8, #64                    // write
    svc     #0
    cmp     x0, #0
    b.le    exit0
    add     x7, x7, x0
    sub     x6, x6, x0
    b       write_loop
exit0:
    mov     x0, #0
    mov     x8, #93                    // exit
    svc     #0

.section .rodata
.balign 64
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 66560
