.globl _start

.section .rodata
b64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 8
input_buf:
    .space 49152
output_buf:
    .space 66560

.text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]
read_loop:
    mov rax, 49152
    sub rax, r12
    test rax, rax
    jz read_done
    xor edi, edi
    mov rsi, r13
    add rsi, r12
    mov rdx, rax
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit0
    mov rsi, r13
    mov rcx, r12
    lea r8, [rip + b64_alphabet]
    lea r14, [rip + output_buf]
    xor r15d, r15d
encode_loop:
    cmp rcx, 3
    jb final_group
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    mov r9d, eax
    shr r9d, 2
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    mov r11d, ebx
    shr r11d, 4
    or r10d, r11d
    mov r11d, ebx
    and r11d, 15
    shl r11d, 2
    mov ebp, edx
    shr ebp, 6
    or r11d, ebp
    and edx, 63
    movzx eax, byte ptr [r8 + r9]
    mov byte ptr [r14], al
    movzx eax, byte ptr [r8 + r10]
    mov byte ptr [r14 + 1], al
    movzx eax, byte ptr [r8 + r11]
    mov byte ptr [r14 + 2], al
    movzx eax, byte ptr [r8 + rdx]
    mov byte ptr [r14 + 3], al
    add rsi, 3
    sub rcx, 3
    jmp finish_group
final_group:
    test rcx, rcx
    jz encode_done
    cmp rcx, 1
    je one_byte
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi + 1]
    mov r9d, eax
    shr r9d, 2
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    mov r11d, ebx
    shr r11d, 4
    or r10d, r11d
    mov r11d, ebx
    and r11d, 15
    shl r11d, 2
    movzx eax, byte ptr [r8 + r9]
    mov byte ptr [r14], al
    movzx eax, byte ptr [r8 + r10]
    mov byte ptr [r14 + 1], al
    movzx eax, byte ptr [r8 + r11]
    mov byte ptr [r14 + 2], al
    mov byte ptr [r14 + 3], 61
    jmp final_done
one_byte:
    movzx eax, byte ptr [rsi]
    mov r9d, eax
    shr r9d, 2
    mov r10d, eax
    and r10d, 3
    shl r10d, 4
    movzx eax, byte ptr [r8 + r9]
    mov byte ptr [r14], al
    movzx eax, byte ptr [r8 + r10]
    mov byte ptr [r14 + 1], al
    mov byte ptr [r14 + 2], 61
    mov byte ptr [r14 + 3], 61
final_done:
    xor ecx, ecx
    jmp finish_group
finish_group:
    add r14, 4
    add r15, 4
    cmp r15, 76
    jb encode_loop
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d
    jmp encode_loop
encode_done:
    test r15, r15
    jz write_out
    mov byte ptr [r14], 10
    inc r14
write_out:
    lea rsi, [rip + output_buf]
    mov rdx, r14
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
exit0:
    xor edi, edi
    mov eax, 60
    syscall
