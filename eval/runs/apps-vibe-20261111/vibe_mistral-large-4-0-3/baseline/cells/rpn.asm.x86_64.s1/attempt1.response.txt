.globl _start

.text
_start:
    lea rbx, [rip + inbuf]
    xor r12, r12
read_loop:
    mov rax, 65536
    sub rax, r12
    test rax, rax
    jz read_done
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, rax
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
line_loop:
    cmp r13, r12
    jae prog_done
    mov r14, r13
scan_newline:
    cmp r14, r12
    jae line_have_end
    movzx eax, byte ptr [rbx + r14]
    cmp al, 10
    je line_have_end
    inc r14
    jmp scan_newline
line_have_end:
    mov r15, r13
    xor ebp, ebp
    xor r9d, r9d
token_loop:
    cmp r15, r14
    jae tokens_done
    movzx eax, byte ptr [rbx + r15]
    cmp al, 32
    je tok_skip
    cmp al, 9
    je tok_skip
    jmp token_have
tok_skip:
    inc r15
    jmp token_loop
token_have:
    mov rsi, r15
tok_find_end:
    cmp r15, r14
    jae tok_end
    movzx eax, byte ptr [rbx + r15]
    cmp al, 32
    je tok_end
    cmp al, 9
    je tok_end
    inc r15
    jmp tok_find_end
tok_end:
    mov rcx, r15
    sub rcx, rsi
    movzx eax, byte ptr [rsi]
    cmp rcx, 1
    jne tok_long
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod
    cmp al, 48
    jb invalid
    cmp al, 57
    ja invalid
    jmp tok_literal
tok_long:
    cmp al, 45
    je tok_literal
    cmp al, 48
    jb invalid
    cmp al, 57
    ja invalid
tok_literal:
    xor edi, edi
    mov r8, rsi
    cmp al, 45
    jne lit_start
    mov edi, 1
    inc r8
lit_start:
    mov rcx, r15
    sub rcx, r8
    test rcx, rcx
    jz invalid
    cmp rcx, 18
    ja invalid
    xor eax, eax
    mov r11, 10
lit_digit:
    cmp r8, r15
    jae lit_done
    movzx r10d, byte ptr [rbx + r8]
    sub r10b, 48
    cmp r10b, 9
    ja invalid
    imul rax, r11
    add rax, r10
    inc r8
    jmp lit_digit
lit_done:
    test edi, edi
    jz lit_push
    neg rax
lit_push:
    lea rdx, [rip + valstack]
    mov [rdx + rbp*8], rax
    inc rbp
    jmp token_loop
op_add:
    cmp rbp, 2
    jb underflow
    lea r8, [rip + valstack]
    lea r8, [r8 + rbp*8]
    mov rax, [r8 - 16]
    mov r10, [r8 - 8]
    add rax, r10
    mov [r8 - 16], rax
    dec rbp
    jmp token_loop
op_sub:
    cmp rbp, 2
    jb underflow
    lea r8, [rip + valstack]
    lea r8, [r8 + rbp*8]
    mov rax, [r8 - 16]
    mov r10, [r8 - 8]
    sub rax, r10
    mov [r8 - 16], rax
    dec rbp
    jmp token_loop
op_mul:
    cmp rbp, 2
    jb underflow
    lea r8, [rip + valstack]
    lea r8, [r8 + rbp*8]
    mov rax, [r8 - 16]
    mov r10, [r8 - 8]
    imul rax, r10
    mov [r8 - 16], rax
    dec rbp
    jmp token_loop
op_div:
    cmp rbp, 2
    jb underflow
    lea r8, [rip + valstack]
    lea r8, [r8 + rbp*8]
    mov rax, [r8 - 16]
    mov r10, [r8 - 8]
    test r10, r10
    jz div_zero
    mov r11, 1
    shl r11, 63
    cmp rax, r11
    jne div_ok
    cmp r10, -1
    je div_overflow
div_ok:
    mov rdx, rax
    sar rdx, 63
    idiv r10
    mov [r8 - 16], rax
    dec rbp
    jmp token_loop
op_mod:
    cmp rbp, 2
    jb underflow
    lea r8, [rip + valstack]
    lea r8, [r8 + rbp*8]
    mov rax, [r8 - 16]
    mov r10, [r8 - 8]
    test r10, r10
    jz div_zero
    mov r11, 1
    shl r11, 63
    cmp rax, r11
    jne mod_ok
    cmp r10, -1
    je div_overflow
mod_ok:
    mov rdx, rax
    sar rdx, 63
    idiv r10
    mov [r8 - 16], rdx
    dec rbp
    jmp token_loop
underflow:
div_zero:
div_overflow:
invalid:
    mov r9d, 1
    jmp line_result
tokens_done:
    jmp line_result
line_result:
    test r9d, r9d
    jnz print_error
    cmp rbp, 1
    jne print_error
    lea r8, [rip + valstack]
    mov rax, [r8]
    lea rdi, [rip + outbuf]
    test rax, rax
    jns pv_nosign
    mov byte ptr [rdi], 45
    inc rdi
    neg rax
pv_nosign:
    lea rsi, [rip + numbuf]
    add rsi, 20
    mov r10, 10
pv_divloop:
    xor edx, edx
    div r10
    add dl, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz pv_divloop
    lea rcx, [rip + numbuf]
    add rcx, 20
    sub rcx, rsi
pv_copy:
    test rcx, rcx
    jz pv_copied
    movzx eax, byte ptr [rsi]
    mov [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp pv_copy
pv_copied:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    jmp write_line
print_error:
    lea rsi, [rip + err_msg]
    mov edx, 6
    jmp write_line
write_line:
    test rdx, rdx
    jz next_line
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle next_line
    add rsi, rax
    sub rdx, rax
    jmp write_line
next_line:
    mov r13, r14
    cmp r13, r12
    jae prog_done
    inc r13
    jmp line_loop
prog_done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata

    .byte 101, 114, 114, 111, 114, 10

    .zero 262144
.balign 8
