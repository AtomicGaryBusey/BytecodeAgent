.equ INPUT_CAP, 49153
.equ OUTPUT_CAP, 66560

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, INPUT_CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, INPUT_CAP
    jb read_loop
read_done:
    test r12d, r12d
    jz exit0
    lea rsi, [rip + input_buf]
    lea rdi, [rip + out_buf]
    lea rbx, [rip + chars]
    xor ecx, ecx
    xor r9d, r9d
encode_loop:
    cmp rcx, r12
    jae encode_done
    mov rdx, r12
    sub rdx, rcx
    cmp rdx, 1
    je one_byte
    cmp rdx, 2
    je two_bytes
    movzx eax, byte ptr [rsi + rcx]
    shl eax, 16
    movzx edx, byte ptr [rsi + rcx + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + rcx + 2]
    or eax, edx
    mov r8d, eax
    add rcx, 3
    mov edx, r8d
    shr edx, 18
    call emit_char
    mov edx, r8d
    shr edx, 12
    and edx, 63
    call emit_char
    mov edx, r8d
    shr edx, 6
    and edx, 63
    call emit_char
    mov edx, r8d
    and edx, 63
    call emit_char
    jmp encode_loop
one_byte:
    movzx eax, byte ptr [rsi + rcx]
    shl eax, 16
    mov r8d, eax
    add rcx, 1
    mov edx, r8d
    shr edx, 18
    call emit_char
    mov edx, r8d
    shr edx, 12
    and edx, 63
    call emit_char
    mov edx, 64
    call emit_char
    mov edx, 64
    call emit_char
    jmp encode_loop
two_bytes:
    movzx eax, byte ptr [rsi + rcx]
    shl eax, 16
    movzx edx, byte ptr [rsi + rcx + 1]
    shl edx, 8
    or eax, edx
    mov r8d, eax
    add rcx, 2
    mov edx, r8d
    shr edx, 18
    call emit_char
    mov edx, r8d
    shr edx, 12
    and edx, 63
    call emit_char
    mov edx, r8d
    shr edx, 6
    and edx, 63
    call emit_char
    mov edx, 64
    call emit_char
    jmp encode_loop
encode_done:
    test r9d, r9d
    jz write_setup
    mov byte ptr [rdi], 10
    inc rdi
write_setup:
    lea rax, [rip + out_buf]
    sub rdi, rax
    mov r12, rdi
    lea rbx, [rip + out_buf]
    xor r13d, r13d
write_loop:
    cmp r13, r12
    jae exit0
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r13]
    mov rdx, r12
    sub rdx, r13
    syscall
    test rax, rax
    jz exit0
    add r13, rax
    jmp write_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall
emit_char:
    movzx r10d, byte ptr [rbx + rdx]
    mov byte ptr [rdi], r10b
    inc rdi
    inc r9
    cmp r9, 76
    jb emit_char_no_newline
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
emit_char_no_newline:
    ret

.section .rodata
chars:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    .byte 61

.bss
.align 16
input_buf:
    .zero INPUT_CAP
out_buf:
    .zero OUTPUT_CAP
