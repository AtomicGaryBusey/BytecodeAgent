.equ INPUT_CAP, 49152
.equ OUT_CAP, 66560

.text
.globl _start
_start:
    lea r14, [rip + input_buf]
    mov r12, INPUT_CAP
read_loop:
    test r12, r12
    jz read_done
    xor edi, edi
    mov rsi, r14
    mov rdx, r12
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    sub r12, rax
    jmp read_loop
read_done:
    mov r15, INPUT_CAP
    sub r15, r12
    lea r14, [rip + input_buf]
    lea rbx, [rip + out_buf]
    xor r13, r13
encode_loop:
    cmp r15, 3
    jb tail
    movzx eax, byte ptr [r14]
    movzx ecx, byte ptr [r14 + 1]
    movzx edx, byte ptr [r14 + 2]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ecx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ecx
    and r10d, 15
    shl r10d, 2
    mov r11d, edx
    and r11d, 63
    shr edx, 6
    or r10d, edx
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + r8]
    call putc
    movzx eax, byte ptr [rdx + r9]
    call putc
    movzx eax, byte ptr [rdx + r10]
    call putc
    movzx eax, byte ptr [rdx + r11]
    call putc
    add r14, 3
    sub r15, 3
    jmp encode_loop
tail:
    test r15, r15
    jz finish
    cmp r15, 1
    je tail1
    movzx eax, byte ptr [r14]
    movzx ecx, byte ptr [r14 + 1]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ecx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ecx
    and r10d, 15
    shl r10d, 2
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + r8]
    call putc
    movzx eax, byte ptr [rdx + r9]
    call putc
    movzx eax, byte ptr [rdx + r10]
    call putc
    mov al, 61
    call putc
    jmp finish
tail1:
    movzx eax, byte ptr [r14]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + r8]
    call putc
    movzx eax, byte ptr [rdx + r9]
    call putc
    mov al, 61
    call putc
    mov al, 61
    call putc
finish:
    test r13, r13
    jz write_out
    mov byte ptr [rbx], 0x0A
    inc rbx
write_out:
    lea rax, [rip + out_buf]
    sub rbx, rax
    test rbx, rbx
    jz exit_prog
    mov rdx, rbx
    lea rsi, [rip + out_buf]
    mov edi, 1
    mov eax, 1
    syscall
exit_prog:
    mov edi, 0
    mov eax, 60
    syscall

putc:
    mov byte ptr [rbx], al
    inc rbx
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [rbx], 0x0A
    inc rbx
    xor r13, r13
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.p2align 4
input_buf:
    .zero INPUT_CAP
out_buf:
    .zero OUT_CAP
