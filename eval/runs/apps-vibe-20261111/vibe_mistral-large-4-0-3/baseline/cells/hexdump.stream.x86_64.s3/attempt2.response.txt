.text
.equ IN_MAX, 65536
.equ OUT_MAX, 323593

.globl _start
_start:
    xor r12, r12
    mov r14, IN_MAX
read_loop:
    test r14, r14
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, r14
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    sub r14, rax
    jmp read_loop
read_done:
    lea r13, [rip + outbuf]
    xor r14, r14
row_loop:
    cmp r14, r12
    jae rows_done
    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe have_n
    mov r15, 16
have_n:
    lea rbx, [rip + inbuf]
    add rbx, r14
    mov rax, r14
    call put_hex64
    mov al, 0x20
    call put
    call put
    xor rcx, rcx
byte_loop:
    cmp rcx, 16
    jae byte_loop_done
    cmp rcx, r15
    jae pad_slot
    movzx eax, byte ptr [rbx + rcx]
    call put_hex_byte
    mov al, 0x20
    call put
    jmp slot_done
pad_slot:
    mov al, 0x20
    call put
    call put
    call put
slot_done:
    cmp rcx, 7
    jne after_extra
    mov al, 0x20
    call put
after_extra:
    inc rcx
    jmp byte_loop
byte_loop_done:
    mov al, 0x20
    call put
    mov al, 0x7c
    call put
    xor rcx, rcx
char_loop:
    cmp rcx, r15
    jae char_loop_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x20
    jb print_dot
    cmp al, 0x7e
    ja print_dot
    call put
    jmp char_next
print_dot:
    mov al, 0x2e
    call put
char_next:
    inc rcx
    jmp char_loop
char_loop_done:
    mov al, 0x7c
    call put
    mov al, 0x0a
    call put
    add r14, 16
    jmp row_loop
rows_done:
    mov rax, r12
    call put_hex64
    mov al, 0x0a
    call put
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
    mov edi, 1
    call write_all
    xor edi, edi
    mov eax, 60
    syscall

put:
    mov byte ptr [r13], al
    inc r13
    ret

put_hex_nibble:
    and eax, 0xf
    lea rsi, [rip + hexdigits]
    movzx r10d, byte ptr [rsi + rax]
    mov al, r10b
    jmp put

put_hex_byte:
    mov dl, al
    mov al, dl
    shr al, 4
    call put_hex_nibble
    mov al, dl
    and al, 0xf
    call put_hex_nibble
    ret

put_hex64:
    mov r8, rax
    mov r9, 8
hex64_loop:
    mov rcx, r9
    dec rcx
    shl rcx, 2
    mov rax, r8
    shr rax, cl
    call put_hex_nibble
    dec r9
    jnz hex64_loop
    ret

write_all:
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    syscall
    cmp rax, 0
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero IN_MAX
outbuf:
    .zero OUT_MAX
