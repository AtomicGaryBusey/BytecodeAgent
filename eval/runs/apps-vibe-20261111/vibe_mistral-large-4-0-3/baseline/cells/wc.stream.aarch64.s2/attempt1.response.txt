.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.equ OUT_SIZE, 32

.text
.globl _start

    add x5, x19, x20
    mov x1, #0
    mov x2, #0
    mov x3, #0
count_loop:

    cmp w6, #0x0A
    b.ne 1f
    add x1, x1, #1

    mov x12, x2
    mov x13, x20
    adrp x10, outbuf
    add x10, x10, :lo12:outbuf
    mov x0, x11

    strb w14, [x10], #1
    mov x0, x12
    mov x1, x10
    bl emit_num
    mov x10, x1
    mov w14, #0x20
    strb w14, [x10], #1
    mov x0, x13
    mov x1, x10
    bl emit_num
    mov x10, x1
    mov w14, #0x0A
    strb w14, [x10], #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x10, x1
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT

    add x7, sp, #16
    mov x3, x7
    mov x2, #10

    msub x5, x4, x2, x0

    sub x6, x7, x3

    strb w8, [x1], #1
    subs x6, x6, #1
    b.ne 2b
    add sp, sp, #16
    ret
