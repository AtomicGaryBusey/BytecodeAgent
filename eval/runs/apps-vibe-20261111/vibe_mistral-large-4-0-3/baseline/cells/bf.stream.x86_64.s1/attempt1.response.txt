.text
.globl _start
_start:
    xor r15d, r15d
read_loop:
    cmp r15, 65536
    jae read_probe
    mov eax, 65536
    sub rax, r15
    xor edi, edi
    lea rsi, [inbuf + r15]
    mov rdx, rax
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add r15, rax
    jmp read_loop
read_probe:
    xor edi, edi
    lea rsi, [inbuf + r15]
    mov edx, 1
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    jmp read_probe
read_done:
    xor ecx, ecx
find_loop:
    cmp rcx, r15
    jae no_bang
    cmp byte ptr [inbuf + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r14, rcx
    lea rbp, [inbuf + r15]
    lea r12, [inbuf + rcx]
    inc r12
    jmp build
no_bang:
    mov r14, r15
    lea rbp, [inbuf + r15]
    mov r12, rbp
build:
    xor ecx, ecx
    xor r9d, r9d
build_loop:
    cmp rcx, r14
    jae build_done
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, 0x5B
    jne build_not_open
    cmp r9, 1000
    jae unbalanced
    mov dword ptr [openstack + r9*4], ecx
    inc r9
    jmp build_next
build_not_open:
    cmp al, 0x5D
    jne build_next
    test r9d, r9d
    jz unbalanced
    dec r9
    mov eax, dword ptr [openstack + r9*4]
    mov dword ptr [matchtab + rax*4], ecx
    mov dword ptr [matchtab + rcx*4], eax
build_next:
    inc rcx
    jmp build_loop
build_done:
    test r9d, r9d
    jnz unbalanced
    xor r13d, r13d
    lea rbx, [rip + tape]
    xor r8d, r8d
cmd_loop:
    cmp r13, r14
    jae run_done
    movzx eax, byte ptr [inbuf + r13]
    inc r13
    cmp al, 0x3E
    je do_gt
    cmp al, 0x3C
    je do_lt
    cmp al, 0x2B
    je do_plus
    cmp al, 0x2D
    je do_minus
    cmp al, 0x2E
    je do_dot
    cmp al, 0x2C
    je do_comma
    cmp al, 0x5B
    je do_open
    cmp al, 0x5D
    je do_close
    jmp cmd_loop
do_gt:
    inc rbx
    jmp cmd_loop
do_lt:
    dec rbx
    jmp cmd_loop
do_plus:
    inc byte ptr [rbx]
    jmp cmd_loop
do_minus:
    dec byte ptr [rbx]
    jmp cmd_loop
do_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [outbuf + r8], al
    inc r8
    jmp cmd_loop
do_comma:
    cmp r12, rbp
    jae comma_zero
    movzx eax, byte ptr [r12]
    inc r12
    mov byte ptr [rbx], al
    jmp cmd_loop
comma_zero:
    mov byte ptr [rbx], 0
    jmp cmd_loop
do_open:
    cmp byte ptr [rbx], 0
    jne cmd_loop
    mov r10, r13
    dec r10
    mov eax, dword ptr [matchtab + r10*4]
    inc rax
    mov r13, rax
    jmp cmd_loop
do_close:
    cmp byte ptr [rbx], 0
    je cmd_loop
    mov r10, r13
    dec r10
    mov eax, dword ptr [matchtab + r10*4]
    inc rax
    mov r13, rax
    jmp cmd_loop
run_done:
    test r8, r8
    jz exit0
    xor r9d, r9d
write_loop:
    mov eax, 1
    mov edi, 1
    lea rsi, [outbuf + r9]
    mov rdx, r8
    sub rdx, r9
    syscall
    test rax, rax
    jle exit0
    add r9, rax
    cmp r9, r8
    jb write_loop
exit0:
    xor edi, edi
    mov eax, 60
    syscall
unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.balign 16
inbuf:
    .zero 65537
tape:
    .zero 30000
outbuf:
    .zero 65536
.balign 4
matchtab:
    .zero 262144
openstack:
    .zero 4096
