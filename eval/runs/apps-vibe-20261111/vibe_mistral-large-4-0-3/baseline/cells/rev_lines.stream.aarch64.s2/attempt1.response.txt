.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT_GROUP, 94
.equ BUFSZ, 65536

.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    movz x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.eq read_done
    b.lt read_err
    add x20, x20, x0
    movz x3, #1, lsl #16
    cmp x20, x3
    b.eq read_done
    b read_loop
read_done:
    mov x21, x19
    add x22, x21, x20
line_loop:
    cmp x21, x22
    b.hs done
    mov x23, x21
scan_loop:
    cmp x23, x22
    b.hs no_nl
    ldrb w24, [x23]
    cmp w24, #0x0A
    b.eq found_nl
    add x23, x23, #1
    b scan_loop
found_nl:
    sub x25, x23, x21
    mov x0, x21
    mov x1, x25
    bl reverse
    mov x0, #1
    mov x1, x21
    add x2, x25, #1
    mov x8, #SYS_WRITE
    svc #0
    add x21, x23, #1
    b line_loop
no_nl:
    sub x25, x23, x21
    mov x0, x21
    mov x1, x25
    bl reverse
    mov x0, #1
    mov x1, x21
    mov x2, x25
    mov x8, #SYS_WRITE
    svc #0
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    mov x21, x23
    b line_loop
done:
    mov x0, #0
    mov x8, #SYS_EXIT_GROUP
    svc #0
read_err:
    mov x0, #1
    mov x8, #SYS_EXIT_GROUP
    svc #0

reverse:
    cbz x1, rev_done
    add x2, x0, x1
    sub x2, x2, #1
    mov x3, x0
rev_loop:
    cmp x3, x2
    b.hs rev_done
    ldrb w4, [x3]
    ldrb w5, [x2]
    strb w4, [x2]
    strb w5, [x3]
    add x3, x3, #1
    sub x2, x2, #1
    b rev_loop
rev_done:
    ret

.section .rodata
.balign 8
nl:
    .byte 0x0A

.bss
.balign 16
inbuf:
    .space BUFSZ
