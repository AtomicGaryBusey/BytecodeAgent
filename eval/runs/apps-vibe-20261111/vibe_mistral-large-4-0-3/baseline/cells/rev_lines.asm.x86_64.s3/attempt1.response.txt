.equ INPUT_MAX, 65536
.equ BUF_SIZE, 65537

.text
.globl _start
_start:
    lea rbx, [rip + buf]
    xor r12, r12
read_loop:
    mov eax, 0
    xor edi, edi
    mov rsi, rbx
    add rsi, r12
    mov edx, INPUT_MAX
    sub edx, r12d
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, INPUT_MAX
    jb read_loop
read_done:
    test r12, r12
    jz lines_done
    mov r9, r12
    dec r9
    movzx eax, byte ptr [rbx + r9*1]
    cmp al, 0x0A
    je no_append
    mov byte ptr [rbx + r12*1], 0x0A
    inc r12
no_append:
    xor r8, r8
line_loop:
    cmp r8, r12
    jae lines_done
    mov r9, r8
scan_loop:
    cmp r9, r12
    jae lines_done
    movzx eax, byte ptr [rbx + r9*1]
    cmp al, 0x0A
    je found_nl
    inc r9
    jmp scan_loop
found_nl:
    mov r10, r9
    sub r10, r8
    cmp r10, 2
    jb after_reverse
    mov r10, r8
    mov r11, r9
    dec r11
rev_swap:
    cmp r10, r11
    jae after_reverse
    movzx eax, byte ptr [rbx + r10*1]
    movzx ecx, byte ptr [rbx + r11*1]
    mov byte ptr [rbx + r11*1], al
    mov byte ptr [rbx + r10*1], cl
    inc r10
    dec r11
    jmp rev_swap
after_reverse:
    mov r8, r9
    inc r8
    jmp line_loop
lines_done:
    lea rsi, [rip + buf]
    mov rdx, r12
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
buf:
    .zero BUF_SIZE
