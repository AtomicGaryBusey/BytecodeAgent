.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT_GROUP, 94

.equ BUFSIZE, 32768
.equ MAXLINES, 1000
.equ LINES_BYTES, 16000
.equ NEWLINE, 0x0A

.section .rodata
nl_byte:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .zero BUFSIZE
.balign 16
lines:
    .zero LINES_BYTES
.balign 16

    .zero LINES_BYTES

.text

    mov x19, #0
    mov x9, #BUFSIZE

    mov x8, #SYS_READ
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, x9
    sub x2, x2, x19
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    cmp x19, x9
    b.lt read_loop
read_done:
    mov x28, #NEWLINE
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    adrp x25, lines
    add x25, x25, :lo12:lines
    mov x21, #0
    mov x22, #0
    mov x23, #0

    b.ge split_done
    ldrb w26, [x24, x22]
    cmp w26, w28
    b.ne split_next
    add x9, x25, x21, lsl #4
    add x10, x24, x23
    str x10, [x9]
    sub x11, x22, x23
    str x11, [x9, #8]
    add x21, x21, #1
    add x23, x22, #1

    b split_loop

    b.ge no_final_line
    add x9, x25, x21, lsl #4
    add x10, x24, x23
    str x10, [x9]
    sub x11, x19, x23
    str x11, [x9, #8]
    add x21, x21, #1
no_final_line:
    adrp x28, tmp
    add x28, x28, :lo12:tmp
    mov x26, #1
ms_outer:
    cmp x26, x21
    b.ge ms_done
    mov x27, #0
ms_inner:
    cmp x27, x21
    b.ge ms_next_pass
    add x9, x27, x26
    cmp x9, x21
    csel x9, x9, x21, le
    add x10, x27, x26, lsl #1
    cmp x10, x21
    csel x10, x10, x21, le
    add x11, x25, x27, lsl #4
    add x12, x25, x9, lsl #4
    add x14, x28, x27, lsl #4
    sub x15, x9, x27
    sub x16, x10, x9

    b.eq ms_l_exhausted
    cmp x16, #0
    b.eq ms_r_exhausted
    mov x0, x11
    mov x1, x12
    bl compare
    cmp w0, #0
    b.le ms_take_left
    b ms_take_right

    ldp x5, x6, [x11], #16
    stp x5, x6, [x14], #16
    sub x15, x15, #1
    b ms_merge
ms_take_right:
    ldp x5, x6, [x12], #16
    stp x5, x6, [x14], #16
    sub x16, x16, #1
    b ms_merge

    cmp x16, #0
    b.eq ms_copied
    ldp x5, x6, [x12], #16
    stp x5, x6, [x14], #16
    sub x16, x16, #1
    b ms_l_exhausted

    b.eq ms_copied
    ldp x5, x6, [x11], #16
    stp x5, x6, [x14], #16
    sub x15, x15, #1
    b ms_r_exhausted

    add x11, x25, x27, lsl #4
    add x12, x28, x27, lsl #4

    b.eq ms_advance
    ldp x5, x6, [x12], #16
    stp x5, x6, [x11], #16
    sub x9, x9, #1
    b ms_copyback

    add x27, x27, x26, lsl #1
    b ms_inner

    mov x26, x26, lsl #1
    b ms_outer

    mov x27, #0

    cmp x27, x21
    b.ge print_done
    add x9, x25, x27, lsl #4
    ldp x10, x11, [x9]
    cmp x11, #0
    b.eq print_nl

    mov x1, x10
    mov x2, x11

    ldp x2, x3, [x0]
    ldp x4, x5, [x1]
    cmp x3, x5

    mov x7, #0

    b.ge 2f
    ldrb w8, [x2, x7]
    ldrb w0, [x4, x7]
    cmp w8, w0
    b.ne 3f
    add x7, x7, #1
    b 1b

    sub w0, w3, w5
    ret

    sub w0, w8, w0
    ret
