.globl _start

.equ SYS_READ,       63
.equ SYS_WRITE,      64
.equ SYS_EXIT,       93
.equ SYS_EXIT_GROUP, 94

.equ BUFSIZE,  32768
.equ MAXLINES, 1000
.equ NEWLINE,  0x0A

.section .rodata
nl_byte:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .zero BUFSIZE
.balign 16
lines:
    .zero MAXLINES * 16
.balign 16
tmp:
    .zero MAXLINES * 16

.text
_start:
    // ---------- read all of standard input ----------
    mov x19, #0                         // x19 = total bytes read so far
    mov x9, #BUFSIZE
read_loop:
    mov x8, #SYS_READ
    mov x0, #0                          // fd 0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19                     // buf = input_buf + total
    mov x2, x9
    sub x2, x2, x19                     // count = BUFSIZE - total
    svc #0
    cmp x0, #0
    b.le read_done                      // 0 => end of input
    add x19, x19, x0
    cmp x19, x9
    b.lt read_loop
read_done:
    // x19 = total number of input bytes

    // ---------- split input into lines ----------
    // line struct: 16 bytes { ptr (8), len (8) }; len includes the newline when present
    mov x17, #MAXLINES
    mov x28, #NEWLINE
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf       // x24 = input base
    adrp x25, lines
    add x25, x25, :lo12:lines           // x25 = lines base
    mov x21, #0                         // n = 0
    mov x22, #0                         // i = 0
    mov x23, #0                         // start = 0
split_loop:
    cmp x22, x19
    b.ge split_done
    ldrb w26, [x24, x22]
    cmp w26, w28
    b.ne split_next
    cmp x21, x17
    b.hs split_next                     // safety cap (input never exceeds it)
    add x9, x25, x21, lsl #4
    add x10, x24, x23
    str x10, [x9]                       // ptr = base + start
    sub x11, x22, x23
    add x11, x11, #1
    str x11, [x9, #8]                   // len = i - start + 1
    add x21, x21, #1
    add x23, x22, #1                    // start = i + 1
split_next:
    add x22, x22, #1
    b split_loop
split_done:
    cmp x23, x19
    b.ge no_final_line
    cmp x21, x17
    b.hs no_final_line
    add x9, x25, x21, lsl #4
    add x10, x24, x23
    str x10, [x9]                       // final line without newline
    sub x11, x19, x23
    str x11, [x9, #8]
    add x21, x21, #1
no_final_line:
    // x21 = n lines, x25 = lines base

    // ---------- bottom-up merge sort ----------
    adrp x28, tmp
    add x28, x28, :lo12:tmp             // x28 = tmp base
    mov x26, #1                         // width
ms_outer:
    cmp x26, x21
    b.ge ms_done
    mov x27, #0                         // i
ms_inner:
    cmp x27, x21
    b.ge ms_next_pass
    add x9, x27, x26
    cmp x9, x21
    csel x9, x9, x21, le                // mid = min(i + width, n)
    add x10, x27, x26, lsl #1
    cmp x10, x21
    csel x10, x10, x21, le              // end = min(i + 2*width, n)
    add x11, x25, x27, lsl #4           // left cursor
    add x12, x25, x9, lsl #4            // right cursor
    add x14, x28, x27, lsl #4           // out cursor in tmp
    sub x15, x9, x27                    // left count
    sub x16, x10, x9                    // right count
ms_merge:
    cmp x15, #0
    b.eq ms_l_exhausted
    cmp x16, #0
    b.eq ms_r_exhausted
    mov x0, x11
    mov x1, x12
    bl compare
    cmp w0, #0
    b.le ms_take_left                   // left <= right: stable
    b ms_take_right
ms_take_left:
    ldp x5, x6, [x11], #16
    stp x5, x6, [x14], #16
    sub x15, x15, #1
    b ms_merge
ms_take_right:
    ldp x5, x6, [x12], #16
    stp x5, x6, [x14], #16
    sub x16, x16, #1
    b ms_merge
ms_l_exhausted:
    cmp x16, #0
    b.eq ms_copied
    ldp x5, x6, [x12], #16
    stp x5, x6, [x14], #16
    sub x16, x16, #1
    b ms_l_exhausted
ms_r_exhausted:
    cmp x15, #0
    b.eq ms_copied
    ldp x5, x6, [x11], #16
    stp x5, x6, [x14], #16
    sub x15, x15, #1
    b ms_r_exhausted
ms_copied:
    sub x9, x10, x27                    // count = end - i
    add x11, x25, x27, lsl #4           // dst = lines + i*16
    add x12, x28, x27, lsl #4           // src = tmp + i*16
ms_copyback:
    cmp x9, #0
    b.eq ms_advance
    ldp x5, x6, [x12], #16
    stp x5, x6, [x11], #16
    sub x9, x9, #1
    b ms_copyback
ms_advance:
    add x27, x27, x26, lsl #1           // i += 2*width
    b ms_inner
ms_next_pass:
    mov x26, x26, lsl #1                // width *= 2
    b ms_outer
ms_done:

    // ---------- print sorted lines ----------
    mov x13, #NEWLINE
    mov x27, #0
print_loop:
    cmp x27, x21
    b.ge print_done
    add x9, x25, x27, lsl #4
    ldp x10, x11, [x9]                  // ptr, len
    mov x8, #SYS_WRITE
    mov x0, #1                          // fd 1
    mov x1, x10
    mov x2, x11
    svc #0
    add x12, x10, x11
    ldrb w12, [x12, #-1]                // last byte of the line
    cmp w12, w13
    b.eq skip_nl                        // already newline-terminated
    mov x8, #SYS_WRITE
    mov x0, #1
    adrp x1, nl_byte
    add x1, x1, :lo12:nl_byte
    mov x2, #1
    svc #0
skip_nl:
    add x27, x27, #1
    b print_loop
print_done:
    mov x8, #SYS_EXIT
    mov x0, #0
    svc #0
    mov x8, #SYS_EXIT_GROUP
    mov x0, #0
    svc #0

// compare: x0 -> line a, x1 -> line b (16-byte structs {ptr,len})
// returns w0 < 0 if a < b, 0 if equal, > 0 if a > b (unsigned byte order, shorter prefix first)
compare:
    ldp x2, x3, [x0]                    // x2 = a.ptr, x3 = a.len
    ldp x4, x5, [x1]                    // x4 = b.ptr, x5 = b.len
    cmp x3, x5
    csel x6, x3, x5, le                 // x6 = min(a.len, b.len)
    mov x7, #0                          // i = 0
1:
    cmp x7, x6
    b.ge 2f
    ldrb w8, [x2, x7]
    ldrb w0, [x4, x7]
    cmp w8, w0
    b.ne 3f
    add x7, x7, #1
    b 1b
2:
    sub w0, w3, w5                      // common prefix: shorter line first
    ret
3:
    sub w0, w8, w0                      // first differing byte, unsigned
    ret
