// rev_lines: print every line of standard input with its bytes in reverse order
.globl _start

.text
_start:
    adrp x9, buf
    add  x9, x9, :lo12:buf      // x9 = input buffer
    mov  x10, #0                // x10 = total bytes read so far

read_loop:
    mov  x2, #65536
    sub  x2, x2, x10            // remaining capacity
    cbz  x2, read_done
    mov  x0, #0                 // fd 0 = stdin
    mov  x1, x9
    add  x1, x1, x10            // buf + offset
    mov  x8, #63                // read
    svc  #0
    cbz  x0, read_done          // 0 = end of input
    add  x10, x10, x0
    b    read_loop

read_done:
    mov  x11, x9                // x11 = start of current line
    mov  x12, x9                // x12 = scan cursor
    add  x13, x9, x10           // x13 = end of input

scan_loop:
    cmp  x12, x13
    b.hs scan_done
    ldrb w14, [x12], #1
    cmp  w14, #0x0A
    b.ne scan_loop
    // newline found; x12 now points just past it
    mov  x0, x11
    sub  x1, x12, #1
    bl   reverse                // reverse the line's bytes in place
    mov  x0, #1                 // fd 1 = stdout
    mov  x1, x11
    mov  x2, x12
    sub  x2, x2, x11            // length including the newline already at the end
    mov  x8, #64                // write
    svc  #0
    mov  x11, x12               // next line starts after the newline
    b    scan_loop

scan_done:
    cmp  x11, x13
    b.hs all_done               // no final unterminated line
    mov  x0, x11
    mov  x1, x13
    bl   reverse
    mov  x0, #1
    mov  x1, x11
    mov  x2, x13
    sub  x2, x2, x11
    mov  x8, #64
    svc  #0
    adrp x1, nl
    add  x1, x1, :lo12:nl
    mov  x0, #1
    mov  x2, #1
    mov  x8, #64
    svc  #0

all_done:
    mov  x0, #0
    mov  x8, #93                // exit
    svc  #0

// reverse the bytes in [x0, x1) in place
reverse:
    cmp  x0, x1
    b.hs rev_done
rev_loop:
    sub  x1, x1, #1
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w3, [x0]
    strb w2, [x1]
    add  x0, x0, #1
    cmp  x0, x1
    b.lo rev_loop
rev_done:
    ret

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 16
buf:
    .zero 65536
