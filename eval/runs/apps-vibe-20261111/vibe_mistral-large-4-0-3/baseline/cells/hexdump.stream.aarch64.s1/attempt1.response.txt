.globl _start
.text
_start:
    // read all of standard input into buf
    adrp x6, buf
    add  x6, x6, :lo12:buf
    mov  x9, #0
1:  mov  x0, #0
    mov  x2, #65536
    sub  x2, x2, x9
    add  x1, x6, x9
    mov  x8, #63
    svc  #0
    cbz  x0, 2f
    add  x9, x9, x0
    b    1b
2:
    adrp x21, hexdig
    add  x21, x21, :lo12:hexdig
    adrp x13, line
    add  x13, x13, :lo12:line
    mov  x10, #0
3:  cmp  x10, x9
    b.hs 10f
    sub  x12, x9, x10
    cmp  x12, #16
    b.lo 4f
    mov  x12, #16
4:  add  x11, x6, x10
    mov  x14, x13
    // 8 hex digits of the row offset
    mov  x15, #8
    mov  x18, #60
5:  lsr  x16, x10, x18
    and  x16, x16, #0xf
    ldrb w16, [x21, x16]
    strb w16, [x14], #1
    sub  x18, x18, #4
    subs x15, x15, #1
    b.ne 5b
    // two spaces
    mov  w16, #0x20
    strb w16, [x14], #1
    strb w16, [x14], #1
    // 16 byte slots
    mov  x15, #0
6:  cmp  x15, x12
    b.hs 7f
    ldrb w16, [x11, x15]
    lsr  w17, w16, #4
    ldrb w17, [x21, x17]
    strb w17, [x14], #1
    and  w18, w16, #0xf
    ldrb w18, [x21, x18]
    strb w18, [x14], #1
    mov  w17, #0x20
    strb w17, [x14], #1
    b    8f
7:  mov  w17, #0x20
    strb w17, [x14], #1
    strb w17, [x14], #1
    strb w17, [x14], #1
8:  cmp  x15, #7
    b.ne 9f
    mov  w17, #0x20
    strb w17, [x14], #1
9:  add  x15, x15, #1
    cmp  x15, #16
    b.lt 6b
    // space, '|', ascii column, '|', newline
    mov  w17, #0x20
    strb w17, [x14], #1
    mov  w17, #0x7c
    strb w17, [x14], #1
    mov  x15, #0
11: cmp  x15, x12
    b.hs 12f
    ldrb w16, [x11, x15]
    cmp  w16, #0x20
    b.lo 13f
    cmp  w16, #0x7e
    b.hi 13f
    strb w16, [x14], #1
    b    14f
13: mov  w17, #0x2e
    strb w17, [x14], #1
14: add  x15, x15, #1
    b    11b
12: mov  w17, #0x7c
    strb w17, [x14], #1
    mov  w17, #0x0a
    strb w17, [x14], #1
    // write the finished line
    mov  x2, x14
    sub  x2, x2, x13
    mov  x0, #1
    mov  x1, x13
    mov  x8, #64
    svc  #0
    add  x10, x10, #16
    b    3b
10: // final line: total length as 8 hex digits + newline
    mov  x14, x13
    mov  x15, #8
    mov  x18, #60
15: lsr  x16, x9, x18
    and  x16, x16, #0xf
    ldrb w16, [x21, x16]
    strb w16, [x14], #1
    sub  x18, x18, #4
    subs x15, x15, #1
    b.ne 15b
    mov  w17, #0x0a
    strb w17, [x14], #1
    mov  x2, x14
    sub  x2, x2, x13
    mov  x0, #1
    mov  x1, x13
    mov  x8, #64
    svc  #0
    // exit(0)
    mov  x0, #0
    mov  x8, #93
    svc  #0

.section .rodata
hexdig: .ascii "0123456789abcdef"

.section .bss
.balign 16
buf:  .space 65536
line: .space 80
