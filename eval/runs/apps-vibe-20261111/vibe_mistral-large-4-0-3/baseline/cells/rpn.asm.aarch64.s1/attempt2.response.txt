.text

  adrp x0, input_buf
  add x0, x0, :lo12:input_buf
  adrp x1, rd_ptr
  str x0, [x1, :lo12:rd_ptr]
  mov x2, #65536
  adrp x1, rd_rem
  str x2, [x1, :lo12:rd_rem]

read_loop:
  adrp x0, rd_rem
  ldr x2, [x0, :lo12:rd_rem]
  cmp x2, #0
  b.eq read_done
  adrp x0, rd_ptr
  ldr x1, [x0, :lo12:rd_ptr]
  adrp x0, rd_rem
  ldr x2, [x0, :lo12:rd_rem]
  mov x0, #0
  mov x8, #63
  svc #0
  cmp x0, #0
  b.eq read_done
  adrp x1, rd_ptr
  ldr x2, [x1, :lo12:rd_ptr]
  add x2, x2, x0
  str x2, [x1, :lo12:rd_ptr]
  adrp x1, rd_rem
  ldr x2, [x1, :lo12:rd_rem]
  sub x2, x2, x0
  str x2, [x1, :lo12:rd_rem]
  b read_loop

read_done:
  adrp x0, input_buf
  add x0, x0, :lo12:input_buf
  adrp x1, pos
  str x0, [x1, :lo12:pos]
  adrp x1, rd_ptr
  ldr x2, [x1, :lo12:rd_ptr]
  adrp x1, in_end
  str x2, [x1, :lo12:in_end]

line_loop:
  adrp x0, pos
  ldr x1, [x0, :lo12:pos]
  adrp x0, in_end
  ldr x2, [x0, :lo12:in_end]
  cmp x1, x2
  b.hs exit_prog
  adrp x0, line_start
  str x1, [x0, :lo12:line_start]
  mov x3, x1
scan_newline:
  cmp x3, x2
  b.hs line_found
  ldrb w4, [x3]
  cmp w4, #10
  b.eq line_found
  add x3, x3, #1
  b scan_newline
line_found:
  adrp x0, line_end
  str x3, [x0, :lo12:line_end]
  cmp x3, x2
  b.hs no_newline_skip
  add x3, x3, #1
no_newline_skip:
  adrp x0, pos
  str x3, [x0, :lo12:pos]
  bl process_line
  b line_loop

exit_prog:
  mov x0, #0
  mov x8, #93
  svc #0

process_line:
  stp x29, x30, [sp, #-16]!
  adrp x0, depth
  str xzr, [x0, :lo12:depth]
  adrp x0, line_start
  ldr x9, [x0, :lo12:line_start]
  adrp x0, cur
  str x9, [x0, :lo12:cur]

pline_loop:
  adrp x0, cur
  ldr x9, [x0, :lo12:cur]
  adrp x0, line_end
  ldr x10, [x0, :lo12:line_end]
  mov x12, x9
skip_sep:
  cmp x12, x10
  b.hs pline_end
  ldrb w13, [x12]
  cmp w13, #32
  b.eq skip_next
  cmp w13, #9
  b.eq skip_next
  b have_token
skip_next:
  add x12, x12, #1
  b skip_sep
have_token:
  mov x11, x12
scan_token:
  cmp x12, x10
  b.hs token_done
  ldrb w13, [x12]
  cmp w13, #32
  b.eq token_done
  cmp w13, #9
  b.eq token_done
  add x12, x12, #1
  b scan_token
token_done:
  adrp x0, cur
  str x12, [x0, :lo12:cur]
  sub x13, x12, x11
  ldrb w14, [x11]
  cmp x13, #1
  b.ne parse_int
  cmp w14, #43
  b.eq op_add
  cmp w14, #45
  b.eq op_sub
  cmp w14, #42
  b.eq op_mul
  cmp w14, #47
  b.eq op_div
  cmp w14, #37
  b.eq op_mod

parse_int:
  mov x15, x11
  mov x16, #0
  ldrb w17, [x15]
  cmp w17, #45
  b.ne parse_nosign
  mov x16, #1
  add x15, x15, #1
parse_nosign:
  sub x17, x12, x15
  cmp x17, #0
  b.eq pline_error
  cmp x17, #18
  b.hi pline_error
  mov x5, #0
parse_digit_loop:
  cmp x15, x12
  b.hs parse_digits_done
  ldrb w6, [x15]
  sub w7, w6, #48
  cmp w7, #9
  b.hi pline_error
  mov x8, #10
  mul x5, x5, x8
  add x5, x5, x7
  add x15, x15, #1
  b parse_digit_loop
parse_digits_done:
  cmp x16, #0
  b.eq push_literal
  neg x5, x5
push_literal:
  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  adrp x0, stk
  add x0, x0, :lo12:stk
  str x5, [x0, x4, lsl #3]
  add x4, x4, #1
  adrp x0, depth
  str x4, [x0, :lo12:depth]
  b pline_loop

op_add:
  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  cmp x4, #2
  b.lo pline_error
  sub x4, x4, #1
  adrp x0, stk
  add x0, x0, :lo12:stk
  ldr x5, [x0, x4, lsl #3]
  sub x4, x4, #1
  ldr x6, [x0, x4, lsl #3]
  add x7, x6, x5
  str x7, [x0, x4, lsl #3]
  add x4, x4, #1
  adrp x0, depth
  str x4, [x0, :lo12:depth]
  b pline_loop

op_sub:
  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  cmp x4, #2
  b.lo pline_error
  sub x4, x4, #1
  adrp x0, stk
  add x0, x0, :lo12:stk
  ldr x5, [x0, x4, lsl #3]
  sub x4, x4, #1
  ldr x6, [x0, x4, lsl #3]
  sub x7, x6, x5
  str x7, [x0, x4, lsl #3]
  add x4, x4, #1
  adrp x0, depth
  str x4, [x0, :lo12:depth]
  b pline_loop

op_mul:
  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  cmp x4, #2
  b.lo pline_error
  sub x4, x4, #1
  adrp x0, stk
  add x0, x0, :lo12:stk
  ldr x5, [x0, x4, lsl #3]
  sub x4, x4, #1
  ldr x6, [x0, x4, lsl #3]
  mul x7, x6, x5
  str x7, [x0, x4, lsl #3]
  add x4, x4, #1
  adrp x0, depth
  str x4, [x0, :lo12:depth]

op_div:
  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  cmp x4, #2
  b.lo pline_error
  sub x4, x4, #1
  adrp x0, stk
  add x0, x0, :lo12:stk
  ldr x5, [x0, x4, lsl #3]
  sub x4, x4, #1
  ldr x6, [x0, x4, lsl #3]
  cmp x5, #0
  b.eq pline_error
  cmn x5, #1

  movz x7, #0x8000, lsl #48
  cmp x6, x7

  str x7, [x0, x4, lsl #3]
  add x4, x4, #1
  adrp x0, depth
  str x4, [x0, :lo12:depth]

op_mod:
  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  cmp x4, #2
  b.lo pline_error
  sub x4, x4, #1
  adrp x0, stk
  add x0, x0, :lo12:stk
  ldr x5, [x0, x4, lsl #3]
  sub x4, x4, #1
  ldr x6, [x0, x4, lsl #3]
  cmp x5, #0
  b.eq pline_error

  msub x7, x8, x5, x6
  str x7, [x0, x4, lsl #3]
  add x4, x4, #1
  adrp x0, depth
  str x4, [x0, :lo12:depth]

  adrp x0, depth
  ldr x4, [x0, :lo12:depth]
  cmp x4, #1
  b.ne pline_error
  adrp x0, stk
  add x0, x0, :lo12:stk
  ldr x0, [x0]
