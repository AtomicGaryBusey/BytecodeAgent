.equ INPUT_MAX, 65536

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INPUT_MAX
    sub edx, r12d
    test edx, edx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r14, [rip + inbuf]
    lea r15, [rip + inbuf]
    add r15, r12
    lea rbx, [rip + outbuf]
line_loop:
    cmp r14, r15
    jae all_done
    mov r9, r14
scan_nl:
    cmp r9, r15
    jae line_end_found
    movzx eax, byte ptr [r9]
    cmp eax, 10
    je line_end_found
    inc r9
    jmp scan_nl
line_end_found:
    mov r8, r9
    cmp r8, r15
    jae next_ready
    inc r8
next_ready:
    mov rsi, r14
    xor r13d, r13d
    xor ebp, ebp
token_loop:
    cmp rsi, r9
    jae line_done
    movzx eax, byte ptr [rsi]
    cmp eax, 32
    je sep_skip
    cmp eax, 9
    je sep_skip
    jmp token_have
sep_skip:
    inc rsi
    jmp token_loop
token_have:
    mov rdx, rsi
scan_tok:
    cmp rdx, r9
    jae token_end
    movzx eax, byte ptr [rdx]
    cmp eax, 32
    je token_end
    cmp eax, 9
    je token_end
    inc rdx
    jmp scan_tok
token_end:
    mov r14, rdx
    mov rcx, rdx
    sub rcx, rsi
    cmp rcx, 1
    jne token_multi
    movzx eax, byte ptr [rsi]
    cmp eax, 43
    je op_add
    cmp eax, 45
    je op_sub
    cmp eax, 42
    je op_mul
    cmp eax, 47
    je op_div
    cmp eax, 37
    je op_mod
token_multi:
    movzx eax, byte ptr [rsi]
    xor r10d, r10d
    lea rdi, [rsi]
    mov r11, rcx
    cmp eax, 45
    jne lit_first
    cmp rcx, 1
    je op_sub
    mov r10d, 1
    inc rdi
    dec r11
    jmp lit_check
lit_first:
    cmp eax, 48
    jb bad_line
    cmp eax, 57
    ja bad_line
lit_check:
    cmp r11, 18
    ja bad_line
    mov r12, r11
lit_digits:
    test r12, r12
    jz lit_parsed_digits
    movzx eax, byte ptr [rdi]
    sub eax, 48
    cmp eax, 9
    ja bad_line
    inc rdi
    dec r12
    jmp lit_digits
lit_parsed_digits:
    mov rdi, rsi
    test r10d, r10d
    jz lit_ptr_ready
    inc rdi
lit_ptr_ready:
    xor eax, eax
lit_parse:
    test r11, r11
    jz lit_value_ready
    movzx edx, byte ptr [rdi]
    sub edx, 48
    imul rax, rax, 10
    add rax, rdx
    inc rdi
    dec r11
    jmp lit_parse
lit_value_ready:
    test r10d, r10d
    jz lit_push
    neg rax
lit_push:
    lea r12, [rip + valstack]
    mov [r12 + r13*8], rax
    inc r13
    jmp token_advance
op_add:
    cmp r13, 2
    jb bad_line
    lea r12, [rip + valstack]
    dec r13
    mov rdi, [r12 + r13*8]
    dec r13
    mov rax, [r12 + r13*8]
    add rax, rdi
    mov [r12 + r13*8], rax
    inc r13
    jmp token_advance
op_sub:
    cmp r13, 2
    jb bad_line
    lea r12, [rip + valstack]
    dec r13
    mov rdi, [r12 + r13*8]
    dec r13
    mov rax, [r12 + r13*8]
    sub rax, rdi
    mov [r12 + r13*8], rax
    inc r13
    jmp token_advance
op_mul:
    cmp r13, 2
    jb bad_line
    lea r12, [rip + valstack]
    dec r13
    mov rdi, [r12 + r13*8]
    dec r13
    mov rax, [r12 + r13*8]
    imul rax, rdi
    mov [r12 + r13*8], rax
    inc r13
    jmp token_advance
op_div:
    cmp r13, 2
    jb bad_line
    lea r12, [rip + valstack]
    dec r13
    mov rdi, [r12 + r13*8]
    dec r13
    mov rax, [r12 + r13*8]
    test rdi, rdi
    jz bad_line
    mov r11d, 0x80000000
    shl r11, 32
    cmp rax, r11
    jne div_safe
    cmp rdi, -1
    je bad_line
div_safe:
    cqo
    idiv rdi
    mov [r12 + r13*8], rax
    inc r13
    jmp token_advance
op_mod:
    cmp r13, 2
    jb bad_line
    lea r12, [rip + valstack]
    dec r13
    mov rdi, [r12 + r13*8]
    dec r13
    mov rax, [r12 + r13*8]
    test rdi, rdi
    jz bad_line
    mov r11d, 0x80000000
    shl r11, 32
    cmp rax, r11
    jne mod_safe
    cmp rdi, -1
    je bad_line
mod_safe:
    cqo
    idiv rdi
    mov [r12 + r13*8], rdx
    inc r13
    jmp token_advance
bad_line:
    mov ebp, 1
    jmp line_done
token_advance:
    mov rsi, r14
    jmp token_loop
line_done:
    test rbp, rbp
    jnz emit_error
    cmp r13, 1
    jne emit_error
    lea r12, [rip + valstack]
    mov rax, [r12]
    test rax, rax
    jns conv_unsigned
    mov byte ptr [rbx], 45
    inc rbx
    mov r11d, 0x80000000
    shl r11, 32
    cmp rax, r11
    jne conv_neg
    mov rax, r11
    jmp conv_unsigned
conv_neg:
    neg rax
conv_unsigned:
    lea rdi, [rip + convbuf]
    add rdi, 32
    mov rsi, rdi
    mov r9, 10
conv_loop:
    xor edx, edx
    div r9
    add edx, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz conv_loop
    mov rcx, rdi
    sub rcx, rsi
copy_digits:
    test rcx, rcx
    jz copy_done
    movzx eax, byte ptr [rsi]
    mov byte ptr [rbx], al
    inc rsi
    inc rbx
    dec rcx
    jmp copy_digits
copy_done:
    mov byte ptr [rbx], 10
    inc rbx
    jmp line_advance
emit_error:
    lea rsi, [rip + errormsg]
    mov rcx, 6
err_copy:
    movzx eax, byte ptr [rsi]
    mov byte ptr [rbx], al
    inc rsi
    inc rbx
    dec rcx
    jnz err_copy
line_advance:
    mov r14, r8
    jmp line_loop
all_done:
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_now
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_now:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
errormsg: .ascii "error\n"

.section .bss
.balign 32
inbuf: .space 65536
.balign 32
outbuf: .space 393216
.balign 32
valstack: .space 524288
.balign 32
convbuf: .space 32
