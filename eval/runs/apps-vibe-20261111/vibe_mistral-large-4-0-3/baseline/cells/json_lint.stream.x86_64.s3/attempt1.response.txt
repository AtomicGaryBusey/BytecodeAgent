.equ input_cap, 65537

.section .rodata
ok_msg:
    .ascii "ok\n"
.equ ok_len, . - ok_msg
err_prefix:
    .ascii "error at "
.equ err_prefix_len, . - err_prefix
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r15, r15
read_loop:
    mov eax, 0
    xor edi, edi
    mov rsi, rbx
    add rsi, r15
    mov rdx, input_cap
    sub rdx, r15
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    mov r14, r15
    lea r13, [rip + input_buf]
    xor r12, r12
    call parse_document
    cmp eax, 0
    je print_ok
    cmp eax, 1
    jne err_use_len
    mov r15, r12
    jmp have_error_n
err_use_len:
    mov r15, r14
have_error_n:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, err_prefix_len
    syscall
    sub rsp, 32
    mov rax, r15
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 10
    mov ebx, 10
dec_loop:
    xor edx, edx
    div ebx
    dec rdi
    add dl, 48
    mov byte ptr [rdi], dl
    test rax, rax
    jnz dec_loop
    lea rdx, [rsp + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
print_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
    cmp r12, r14
    jae skip_ws_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x20
    je skip_ws_next
    cmp al, 0x09
    je skip_ws_next
    cmp al, 0x0A
    je skip_ws_next
    cmp al, 0x0D
    je skip_ws_next
    ret
skip_ws_next:
    inc r12
    jmp skip_ws
skip_ws_done:
    ret

parse_document:
    call skip_ws
    call parse_value
    test eax, eax
    jnz pd_ret
    call skip_ws
    cmp r12, r14
    jb pd_err
    xor eax, eax
    ret
pd_err:
    mov eax, 1
    ret
pd_ret:
    ret

parse_value:
    call skip_ws
    cmp r12, r14
    jae pv_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x7B
    jne pv_not_obj
    call parse_object
    ret
pv_not_obj:
    cmp al, 0x5B
    jne pv_not_arr
    call parse_array
    ret
pv_not_arr:
    cmp al, 0x22
    jne pv_not_str
    call parse_string
    ret
pv_not_str:
    cmp al, 0x74
    jne pv_not_true
    lea rsi, [rip + lit_true]
    mov edx, 4
    call parse_lit
    ret
pv_not_true:
    cmp al, 0x66
    jne pv_not_false
    lea rsi, [rip + lit_false]
    mov edx, 5
    call parse_lit
    ret
pv_not_false:
    cmp al, 0x6E
    jne pv_not_null
    lea rsi, [rip + lit_null]
    mov edx, 4
    call parse_lit
    ret
pv_not_null:
    cmp al, 0x2D
    jne pv_not_minus
    call parse_number
    ret
pv_not_minus:
    cmp al, 0x30
    jb pv_err
    cmp al, 0x39
    ja pv_err
    call parse_number
    ret
pv_err:
    mov eax, 1
    ret
pv_eof:
    mov eax, 2
    ret

parse_lit:
    xor ecx, ecx
pl_loop:
    cmp ecx, edx
    jae pl_ok
    cmp r12, r14
    jae pl_eof
    movzx eax, byte ptr [r13 + r12]
    movzx r8d, byte ptr [rsi + rcx]
    cmp al, r8b
    jne pl_err
    inc r12
    inc ecx
    jmp pl_loop
pl_ok:
    xor eax, eax
    ret
pl_eof:
    mov eax, 2
    ret
pl_err:
    mov eax, 1
    ret

parse_number:
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2D
    jne pn_int
    inc r12
pn_int:
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    je pn_zero
    cmp al, 0x31
    jb pn_err
    cmp al, 0x39
    ja pn_err
    inc r12
pn_digits:
    cmp r12, r14
    jae pn_frac
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    ja pn_frac
    inc r12
    jmp pn_digits
pn_zero:
    inc r12
    jmp pn_frac
pn_frac:
    cmp r12, r14
    jae pn_ok
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2E
    jne pn_exp
    inc r12
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_err
    cmp al, 0x39
    ja pn_err
    inc r12
pn_frac_digits:
    cmp r12, r14
    jae pn_exp
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    ja pn_exp
    inc r12
    jmp pn_frac_digits
pn_exp:
    cmp r12, r14
    jae pn_ok
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x65
    je pn_exp_body
    cmp al, 0x45
    je pn_exp_body
    jmp pn_ok
pn_exp_body:
    inc r12
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2B
    je pn_exp_sign
    cmp al, 0x2D
    je pn_exp_sign
    jmp pn_exp_digit
pn_exp_sign:
    inc r12
pn_exp_digit:
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_err
    cmp al, 0x39
    ja pn_err
    inc r12
pn_exp_digits:
    cmp r12, r14
    jae pn_ok
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_ok
    cmp al, 0x39
    ja pn_ok
    inc r12
    jmp pn_exp_digits
pn_ok:
    xor eax, eax
    ret
pn_eof:
    mov eax, 2
    ret
pn_err:
    mov eax, 1
    ret

parse_string:
    inc r12
ps_loop:
    cmp r12, r14
    jae ps_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x22
    je ps_close
    cmp al, 0x5C
    jne ps_char
    inc r12
    cmp r12, r14
    jae ps_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x22
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, 0x2F
    je ps_esc_ok
    cmp al, 0x62
    je ps_esc_ok
    cmp al, 0x66
    je ps_esc_ok
    cmp al, 0x6E
    je ps_esc_ok
    cmp al, 0x72
    je ps_esc_ok
    cmp al, 0x74
    je ps_esc_ok
    cmp al, 0x75
    jne ps_err
    inc r12
    mov ecx, 4
ps_hex:
    cmp r12, r14
    jae ps_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb ps_err
    cmp al, 0x39
    jbe ps_hex_next
    cmp al, 0x41
    jb ps_err
    cmp al, 0x46
    jbe ps_hex_next
    cmp al, 0x61
    jb ps_err
    cmp al, 0x66
    jbe ps_hex_next
    jmp ps_err
ps_hex_next:
    inc r12
    dec ecx
    jnz ps_hex
    jmp ps_loop
ps_esc_ok:
    inc r12
    jmp ps_loop
ps_char:
    cmp al, 0x20
    jb ps_err
    inc r12
    jmp ps_loop
ps_close:
    inc r12
    xor eax, eax
    ret
ps_eof:
    mov eax, 2
    ret
ps_err:
    mov eax, 1
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r14
    jae pa_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x5D
    je pa_empty
pa_loop:
    call parse_value
    test eax, eax
    jnz pa_ret
    call skip_ws
    cmp r12, r14
    jae pa_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    je pa_close
    mov eax, 1
    ret
pa_comma:
    inc r12
    jmp pa_loop
pa_empty:
    inc r12
    xor eax, eax
    ret
pa_close:
    inc r12
    xor eax, eax
    ret
pa_eof:
    mov eax, 2
    ret
pa_ret:
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r14
    jae po_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x7D
    je po_empty
po_loop:
    call parse_member
    test eax, eax
    jnz po_ret
    call skip_ws
    cmp r12, r14
    jae po_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    je po_close
    mov eax, 1
    ret
po_comma:
    inc r12
    jmp po_loop
po_empty:
    inc r12
    xor eax, eax
    ret
po_close:
    inc r12
    xor eax, eax
    ret
po_eof:
    mov eax, 2
    ret
po_ret:
    ret

parse_member:
    call skip_ws
    cmp r12, r14
    jae pm_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x22
    jne pm_err
    call parse_string
    test eax, eax
    jnz pm_ret
    call skip_ws
    cmp r12, r14
    jae pm_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x3A
    jne pm_err
    inc r12
    call parse_value
    ret
pm_err:
    mov eax, 1
    ret
pm_eof:
    mov eax, 2
    ret
pm_ret:
    ret

.section .bss
.balign 16
input_buf:
    .zero input_cap
