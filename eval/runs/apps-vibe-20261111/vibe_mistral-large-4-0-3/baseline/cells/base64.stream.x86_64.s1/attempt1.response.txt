.globl _start
.equ IN_BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 70000
.equ LINE_LEN, 76

.text
_start:
    xor r12d, r12d
    lea r15, [rip + inbuf]
read_more:
    mov eax, IN_BUF_SIZE
    sub eax, r12d
    test eax, eax
    jz input_done
    mov edx, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more
input_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r11, [rip + b64tab]
    xor ebx, ebx
    mov rcx, r12
enc_loop:
    cmp rcx, 3
    jb enc_rem
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r11 + rax]
    call put
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [r11 + rax]
    call put
    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    movzx eax, byte ptr [r11 + rax]
    call put
    mov eax, r10d
    and eax, 63
    movzx eax, byte ptr [r11 + rax]
    call put
    add rsi, 3
    sub rcx, 3
    jmp enc_loop
enc_rem:
    test rcx, rcx
    jz enc_done
    movzx r8d, byte ptr [rsi]
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r11 + rax]
    call put
    mov eax, r8d
    and eax, 3
    shl eax, 4
    cmp rcx, 1
    je enc_rem1
    movzx r9d, byte ptr [rsi + 1]
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [r11 + rax]
    call put
    mov eax, r9d
    and eax, 15
    shl eax, 2
    movzx eax, byte ptr [r11 + rax]
    call put
    mov al, 61
    call put
    jmp enc_done
enc_rem1:
    movzx eax, byte ptr [r11 + rax]
    call put
    mov al, 61
    call put
    mov al, 61
    call put
enc_done:
    test ebx, ebx
    jz out_write
    mov byte ptr [rdi], 10
    inc rdi
out_write:
    lea r12, [rip + outbuf]
    mov r13, rdi
wr_loop:
    cmp r12, r13
    jae wr_done
    mov rsi, r12
    mov rdx, r13
    sub rdx, r12
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jz wr_done
    add r12, rax
    jmp wr_loop
wr_done:
    mov eax, 60
    xor edi, edi
    syscall

put:
    mov byte ptr [rdi], al
    inc rdi
    inc rbx
    cmp rbx, LINE_LEN
    jb 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor ebx, ebx
1:
    ret

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero IN_BUF_SIZE
outbuf:
    .zero OUT_BUF_SIZE
