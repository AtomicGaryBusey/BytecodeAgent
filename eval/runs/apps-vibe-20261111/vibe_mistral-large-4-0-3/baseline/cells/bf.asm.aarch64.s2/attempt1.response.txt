.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INBUF_BYTES, 65536
.equ OUTBUF_BYTES, 65536
.equ TAPE_BYTES, 30000
.equ CMDS_BYTES, 65536
.equ JTAB_BYTES, 262144
.equ BSTACK_BYTES, 262144

.globl _start
_start:
    adrp x26, inbuf
    add x26, x26, :lo12:inbuf
    mov x19, #0
read_loop:
    mov x0, #0
    add x1, x26, x19
    mov x2, #INBUF_BYTES
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop
read_done:
    mov x20, #0
find_bang:
    cmp x20, x19
    b.hs no_bang
    ldrb w8, [x26, x20]
    cmp w8, #33
    b.eq found_bang
    add x20, x20, #1
    b find_bang
found_bang:
    add x21, x20, #1
    mov x22, x19
    b have_parts
no_bang:
    mov x20, x19
    mov x21, x19
    mov x22, x19
have_parts:
    adrp x27, cmds
    add x27, x27, :lo12:cmds
    adrp x29, jtab
    add x29, x29, :lo12:jtab
    adrp x9, bstack
    add x9, x9, :lo12:bstack
    mov x5, x9
    mov x6, #0
    mov x7, #0
comp_loop:
    cmp x6, x20
    b.hs comp_done
    ldrb w8, [x26, x6]
    cmp w8, #91
    b.eq comp_open
    cmp w8, #93
    b.eq comp_close
    cmp w8, #62
    b.eq comp_cmd
    cmp w8, #60
    b.eq comp_cmd
    cmp w8, #43
    b.eq comp_cmd
    cmp w8, #45
    b.eq comp_cmd
    cmp w8, #46
    b.eq comp_cmd
    cmp w8, #44
    b.eq comp_cmd
    b comp_next
comp_cmd:
    strb w8, [x27, x7]
    add x7, x7, #1
    b comp_next
comp_open:
    strb w8, [x27, x7]
    str w7, [x5], #4
    add x7, x7, #1
    b comp_next
comp_close:
    cmp x5, x9
    b.eq unbalanced
    sub x5, x5, #4
    ldr w10, [x5]
    str w10, [x29, x7, lsl #2]
    str w7, [x29, x10, lsl #2]
    strb w8, [x27, x7]
    add x7, x7, #1
    b comp_next
comp_next:
    add x6, x6, #1
    b comp_loop
comp_done:
    cmp x5, x9
    b.ne unbalanced
    mov x20, x7
    b run
run:
    adrp x23, tape
    add x23, x23, :lo12:tape
    mov x24, #0
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
    mov x25, #0
    mov x17, #OUTBUF_BYTES
cmd_loop:
    cmp x24, x20
    b.hs finish
    ldrb w8, [x27, x24]
    cmp w8, #91
    b.eq c_open
    cmp w8, #93
    b.eq c_close
    cmp w8, #43
    b.eq c_plus
    cmp w8, #45
    b.eq c_minus
    cmp w8, #62
    b.eq c_gt
    cmp w8, #60
    b.eq c_lt
    cmp w8, #46
    b.eq c_dot
    cmp w8, #44
    b.eq c_comma
    b cmd_next
c_gt:
    add x23, x23, #1
    b cmd_next
c_lt:
    sub x23, x23, #1
    b cmd_next
c_plus:
    ldrb w9, [x23]
    add w9, w9, #1
    strb w9, [x23]
    b cmd_next
c_minus:
    ldrb w9, [x23]
    sub w9, w9, #1
    strb w9, [x23]
    b cmd_next
c_dot:
    ldrb w9, [x23]
    strb w9, [x28, x25]
    add x25, x25, #1
    cmp x25, x17
    b.hs flush_out
    b cmd_next
flush_out:
    mov x0, #1
    mov x1, x28
    mov x2, x25
    mov x8, #SYS_WRITE
    svc #0
    mov x25, #0
    b cmd_next
c_comma:
    cmp x21, x22
    b.hs comma_zero
    ldrb w9, [x26, x21]
    add x21, x21, #1
    strb w9, [x23]
    b cmd_next
comma_zero:
    strb wzr, [x23]
    b cmd_next
c_open:
    ldrb w9, [x23]
    cbnz w9, cmd_next
    ldr w10, [x29, x24, lsl #2]
    add x24, x10, #1
    b cmd_loop
c_close:
    ldrb w9, [x23]
    cbz w9, cmd_next
    ldr w10, [x29, x24, lsl #2]
    add x24, x10, #1
    b cmd_loop
cmd_next:
    add x24, x24, #1
    b cmd_loop
finish:
    cbz x25, do_exit
    mov x0, #1
    mov x1, x28
    mov x2, x25
    mov x8, #SYS_WRITE
    svc #0
do_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
unbalanced:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 8
inbuf: .zero INBUF_BYTES
outbuf: .zero OUTBUF_BYTES
tape: .zero TAPE_BYTES
cmds: .zero CMDS_BYTES
.balign 8
jtab: .zero JTAB_BYTES
bstack: .zero BSTACK_BYTES
