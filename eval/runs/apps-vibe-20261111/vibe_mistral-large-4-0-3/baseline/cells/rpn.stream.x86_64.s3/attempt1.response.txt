.section .rodata
errmsg: .byte 101, 114, 114, 111, 114, 10
.equ ERRMSG_LEN, . - errmsg

.text
.equ BUF_CAP, 65537
.globl _start
_start:
    xor r15d, r15d
read_loop:
    xor eax, eax
    mov edi, 0
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov edx, BUF_CAP
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    cmp r15, BUF_CAP
    jae read_done
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [rip + inbuf]
    add r13, r15
line_loop:
    cmp r12, r13
    jae all_done
    mov r14, r12
scan_nl:
    cmp r14, r13
    jae no_nl
    movzx eax, byte ptr [r14]
    cmp al, 10
    je found_nl
    inc r14
    jmp scan_nl
found_nl:
    mov rsi, r14
    lea rdi, [r14 + 1]
    jmp process_line
no_nl:
    mov rsi, r13
    mov rdi, r13
process_line:
    mov rbp, r12
    mov r12, rdi
    lea r14, [rip + val_stack]
    xor r15d, r15d
    xor ebx, ebx
token_loop:
    cmp rbp, rsi
    jae line_end
    movzx eax, byte ptr [rbp]
    cmp al, 32
    je sep
    cmp al, 9
    je sep
    jmp token_begin
sep:
    inc rbp
    jmp token_loop
token_begin:
    mov r8, rbp
token_scan:
    cmp rbp, rsi
    jae token_done
    movzx eax, byte ptr [rbp]
    cmp al, 32
    je token_done
    cmp al, 9
    je token_done
    inc rbp
    jmp token_scan
token_done:
    mov r9, rbp
    sub r9, r8
    cmp r9, 1
    jne try_literal
    movzx eax, byte ptr [r8]
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod
try_literal:
    xor eax, eax
    mov rcx, r8
    mov rdx, r9
    movzx r10d, byte ptr [rcx]
    cmp r10b, 45
    jne lit_digits
    dec rdx
    jz lit_invalid
    inc rcx
lit_digits:
    cmp rdx, 1
    jb lit_invalid
    cmp rdx, 18
    ja lit_invalid
lit_digit_loop:
    movzx r10d, byte ptr [rcx]
    sub r10b, 48
    cmp r10b, 9
    ja lit_invalid
    lea rax, [rax + rax*4]
    add rax, rax
    add rax, r10
    inc rcx
    dec rdx
    jnz lit_digit_loop
    movzx r10d, byte ptr [r8]
    cmp r10b, 45
    jne lit_push
    neg rax
lit_push:
    mov [r14], rax
    add r14, 8
    inc r15
    jmp token_loop
lit_invalid:
    mov ebx, 1
    jmp line_end
op_add:
    cmp r15, 2
    jb eval_error
    sub r14, 8
    mov r8, [r14]
    sub r14, 8
    mov r9, [r14]
    add r9, r8
    mov [r14], r9
    add r14, 8
    dec r15
    jmp token_loop
op_sub:
    cmp r15, 2
    jb eval_error
    sub r14, 8
    mov r8, [r14]
    sub r14, 8
    mov r9, [r14]
    sub r9, r8
    mov [r14], r9
    add r14, 8
    dec r15
    jmp token_loop
op_mul:
    cmp r15, 2
    jb eval_error
    sub r14, 8
    mov r8, [r14]
    sub r14, 8
    mov r9, [r14]
    imul r9, r8
    mov [r14], r9
    add r14, 8
    dec r15
    jmp token_loop
op_div:
    cmp r15, 2
    jb eval_error
    sub r14, 8
    mov r8, [r14]
    sub r14, 8
    mov r9, [r14]
    test r8, r8
    jz eval_error
    mov r10d, 1
    shl r10, 63
    cmp r9, r10
    jne op_div_ok
    cmp r8, -1
    je eval_error
op_div_ok:
    mov rax, r9
    cqo
    idiv r8
    mov [r14], rax
    add r14, 8
    dec r15
    jmp token_loop
op_mod:
    cmp r15, 2
    jb eval_error
    sub r14, 8
    mov r8, [r14]
    sub r14, 8
    mov r9, [r14]
    test r8, r8
    jz eval_error
    mov r10d, 1
    shl r10, 63
    cmp r9, r10
    jne op_mod_ok
    cmp r8, -1
    je eval_error
op_mod_ok:
    mov rax, r9
    cqo
    idiv r8
    mov [r14], rdx
    add r14, 8
    dec r15
    jmp token_loop
eval_error:
    mov ebx, 1
    jmp line_end
line_end:
    test ebx, ebx
    jnz print_err
    cmp r15, 1
    jne print_err
    sub r14, 8
    mov rax, [r14]
    call print_value
    jmp line_loop
print_err:
    call print_error
    jmp line_loop
all_done:
    mov edi, 0
    mov eax, 60
    syscall
print_value:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    xor r8d, r8d
    test rax, rax
    jns pv_conv
    mov r8b, 1
    mov r10d, 1
    shl r10, 63
    cmp rax, r10
    jne pv_neg
    mov rax, r10
    jmp pv_conv
pv_neg:
    neg rax
pv_conv:
    mov r9, 10
    xor ecx, ecx
pv_loop:
    xor edx, edx
    div r9
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz pv_loop
    lea rdi, [rip + outbuf]
    test r8b, r8b
    jz pv_pop_loop
    mov byte ptr [rdi], 45
    inc rdi
pv_pop_loop:
    test rcx, rcx
    jz pv_nl
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jmp pv_pop_loop
pv_nl:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
print_error:
    lea rsi, [rip + errmsg]
    mov edx, ERRMSG_LEN
    mov edi, 1
    mov eax, 1
    syscall
    ret

.bss
.balign 16
inbuf: .zero BUF_CAP
.balign 8
outbuf: .zero 32
.balign 8
val_stack: .zero 262152
