.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start
_start:
    adrp x27, input_buf
    add x27, x27, :lo12:input_buf
    mov x21, x27
    mov x22, #0
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x22
    cbz x2, read_done
    mov x0, #0
    mov x1, x21
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    tbnz x0, #63, fail
    add x22, x22, x0
    add x21, x21, x0
    b read_loop
read_done:
    mov x19, #0
find_bang:
    cmp x19, x22
    b.eq no_bang
    ldrb w10, [x27, x19]
    cmp w10, #0x21
    b.eq bang_found
    add x19, x19, #1
    b find_bang
no_bang:
    mov x20, x22
    add x21, x27, x22
    mov x23, #0
    b scan_setup
bang_found:
    mov x20, x19
    add x21, x27, x19
    add x21, x21, #1
    sub x23, x22, x19
    sub x23, x23, #1
scan_setup:
    adrp x26, match_tab
    add x26, x26, :lo12:match_tab
    adrp x28, open_stack
    add x28, x28, :lo12:open_stack
    adrp x24, tape
    add x24, x24, :lo12:tape
    mov x25, x24
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x19, #0
    mov x9, #0
scan_loop:
    cmp x19, x20
    b.eq scan_done
    ldrb w10, [x27, x19]
    cmp w10, #0x5B
    b.eq scan_open
    cmp w10, #0x5D
    b.eq scan_close
    add x19, x19, #1
    b scan_loop
scan_open:
    str w19, [x28, x9, lsl #2]
    add x9, x9, #1
    add x19, x19, #1
    b scan_loop
scan_close:
    cbz x9, fail
    sub x9, x9, #1
    ldr w10, [x28, x9, lsl #2]
    str w10, [x26, x19, lsl #2]
    str w19, [x26, x10, lsl #2]
    add x19, x19, #1
    b scan_loop
scan_done:
    cbnz x9, fail
    mov x28, #0
    b run
fail:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0
    b fail
run:
    mov x19, #0
run_loop:
    cmp x19, x20
    b.eq run_done
    ldrb w10, [x27, x19]
    cmp w10, #0x2B
    b.eq cmd_plus
    cmp w10, #0x2D
    b.eq cmd_minus
    cmp w10, #0x5B
    b.eq cmd_open
    cmp w10, #0x5D
    b.eq cmd_close
    cmp w10, #0x3E
    b.eq cmd_gt
    cmp w10, #0x3C
    b.eq cmd_lt
    cmp w10, #0x2E
    b.eq cmd_dot
    cmp w10, #0x2C
    b.eq cmd_comma
    add x19, x19, #1
    b run_loop
cmd_plus:
    ldrb w11, [x25]
    add w11, w11, #1
    strb w11, [x25]
    add x19, x19, #1
    b run_loop
cmd_minus:
    ldrb w11, [x25]
    sub w11, w11, #1
    strb w11, [x25]
    add x19, x19, #1
    b run_loop
cmd_open:
    ldrb w11, [x25]
    cbnz w11, open_nonzero
    ldr w12, [x26, x19, lsl #2]
    add x19, x12, #1
    b run_loop
open_nonzero:
    add x19, x19, #1
    b run_loop
cmd_close:
    ldrb w11, [x25]
    cbz w11, close_zero
    ldr w12, [x26, x19, lsl #2]
    add x19, x12, #1
    b run_loop
close_zero:
    add x19, x19, #1
    b run_loop
cmd_gt:
    add x25, x25, #1
    add x19, x19, #1
    b run_loop
cmd_lt:
    sub x25, x25, #1
    add x19, x19, #1
    b run_loop
cmd_dot:
    ldrb w11, [x25]
    strb w11, [x24, x28]
    add x28, x28, #1
    add x19, x19, #1
    b run_loop
cmd_comma:
    cbz x23, comma_eof
    ldrb w11, [x21]
    add x21, x21, #1
    sub x23, x23, #1
    strb w11, [x25]
    add x19, x19, #1
    b run_loop
comma_eof:
    mov w11, #0
    strb w11, [x25]
    add x19, x19, #1
    b run_loop
run_done:
    cbz x28, do_exit
    mov x0, #1
    mov x1, x24
    mov x2, x28
    mov x8, #SYS_WRITE
    svc #0
do_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    b do_exit

.bss
.balign 8
input_buf:
    .zero 65536
out_buf:
    .zero 65536
tape:
    .zero 30000
match_tab:
    .zero 262144
open_stack:
    .zero 262144
