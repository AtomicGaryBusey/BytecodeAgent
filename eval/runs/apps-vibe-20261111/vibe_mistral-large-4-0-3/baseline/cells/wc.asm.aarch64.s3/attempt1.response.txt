.globl _start
    .text
_start:
    // Read all of standard input into buf.
    adrp x24, buf
    add  x24, x24, :lo12:buf
    mov  x19, #0                     // x19 = total bytes read
read_loop:
    mov  x0, #0                      // fd 0
    mov  x1, x24
    add  x1, x1, x19                 // buf + total
    movz x2, #1, lsl #16             // 65536
    sub  x2, x2, x19                 // remaining space in buf
    mov  x8, #63                     // read
    svc  #0
    cmp  x0, #0
    b.le read_done                   // 0 = EOF, negative = error
    add  x19, x19, x0
    b    read_loop
read_done:
    // Count newlines (lines) and words.
    add  x27, x24, x19               // one past last byte
    mov  x20, #0                     // lines
    mov  x21, #0                     // words
    mov  w28, #0                     // in-word flag
count_loop:
    cmp  x24, x27
    b.hs count_done
    ldrb w26, [x24], #1
    cmp  w26, #10
    b.ne 1f
    add  x20, x20, #1                // newline byte
1:
    sub  w27, w26, #9
    cmp  w27, #4
    b.ls 2f                          // 0x09..0x0D is whitespace
    cmp  w26, #32
    b.ne 3f                          // 0x20 check; else word byte
2:
    mov  w28, #0
    b    4f
3:
    cbnz w28, 4f
    add  x21, x21, #1
    mov  w28, #1
4:
    b    count_loop
count_done:
    // Build "<lines> <words> <bytes>\n" backwards at the end of outbuf.
    adrp x22, outbuf
    add  x22, x22, :lo12:outbuf
    add  x22, x22, #64               // one past end of outbuf
    mov  x23, x22                    // x23 = end of output
    mov  w4, #10
    strb w4, [x22, #-1]!             // trailing newline
    mov  x5, x19                     // bytes
    bl   emit_num
    mov  w4, #32
    strb w4, [x22, #-1]!             // space
    mov  x5, x21                     // words
    bl   emit_num
    mov  w4, #32
    strb w4, [x22, #-1]!             // space
    mov  x5, x20                     // lines
    bl   emit_num
    // x22 = start of output, x23 = end of output
write_loop:
    cmp  x22, x23
    b.hs write_done
    mov  x0, #1                      // fd 1
    mov  x1, x22
    sub  x2, x23, x22
    mov  x8, #64                     // write
    svc  #0
    cmp  x0, #0
    b.le write_done
    add  x22, x22, x0
    b    write_loop
write_done:
    mov  x0, #0
    mov  x8, #93                     // exit
    svc  #0

// emit_num: convert unsigned value in x5 to decimal ASCII, storing digits
// backwards from [x22] with pre-decrement. Clobbers x2-x6 and x5.
emit_num:
    movz x2, #0xCCCD
    movk x2, #0xCCCC, lsl #16
    movk x2, #0xCCCC, lsl #32
    movk x2, #0xCCCC, lsl #48        // x2 = 0xCCCCCCCCCCCCCCCD
    mov  x6, #10
1:
    umulh x3, x5, x2
    lsr  x3, x3, #3                  // x3 = x5 / 10
    msub x4, x3, x6, x5              // x4 = x5 % 10
    add  w4, w4, #48                 // '0' + digit
    strb w4, [x22, #-1]!
    mov  x5, x3
    cbnz x5, 1b
    ret

    .bss
    .p2align 4
buf:
    .zero 65536
outbuf:
    .zero 64
