.section .rodata
.balign 8
nl:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .space 32768
line_tab:
    .space 262144
tmp_tab:
    .space 262144

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12, 32768
    jae read_done
    mov eax, 32768
    sub eax, r12d
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor edi, edi
    mov edx, eax
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    test r12, r12
    jz parse_done
scan:
    cmp r15, r12
    jae scan_end
    lea rax, [rip + input_buf]
    movzx ecx, byte ptr [rax + r15]
    cmp cl, 0x0A
    jne next_byte
    mov eax, r14d
    mov ecx, r15d
    sub ecx, eax
    shl ecx, 16
    or eax, ecx
    lea rdx, [rip + line_tab]
    mov qword ptr [rdx + r13*8], rax
    inc r13
    mov r14, r15
    inc r14
next_byte:
    inc r15
    jmp scan
scan_end:
    cmp r14, r12
    jae parse_done
    mov eax, r14d
    mov ecx, r12d
    sub ecx, eax
    shl ecx, 16
    or eax, ecx
    lea rdx, [rip + line_tab]
    mov qword ptr [rdx + r13*8], rax
    inc r13
parse_done:
    mov rbp, r12
    lea rbx, [rip + line_tab]
    lea r12, [rip + tmp_tab]
    call sort_lines
    xor r15d, r15d
out_loop:
    cmp r15, r13
    jae out_done
    mov eax, dword ptr [rbx + r15*8]
    mov edx, eax
    and eax, 0xFFFF
    shr edx, 16
    lea rcx, [rax + rdx]
    lea rsi, [rip + input_buf]
    lea rsi, [rsi + rax]
    cmp rcx, rbp
    jae final_line
    lea rdx, [rdx + 1]
    mov edi, 1
    call write_all
    jmp next_out
final_line:
    mov edi, 1
    call write_all
    lea rsi, [rip + nl]
    mov edx, 1
    mov edi, 1
    call write_all
next_out:
    inc r15
    jmp out_loop
out_done:
    xor edi, edi
    mov eax, 60
    syscall

compare_packed:
    mov r8d, edi
    mov r9d, esi
    mov eax, r8d
    and eax, 0xFFFF
    mov ecx, r9d
    and ecx, 0xFFFF
    shr r8d, 16
    shr r9d, 16
    mov r10d, r8d
    cmp r10d, r9d
    cmova r10d, r9d
    lea rdi, [rip + input_buf]
    lea r11, [rdi + rax]
    lea rsi, [rdi + rcx]
    xor edx, edx
cmp_loop:
    cmp rdx, r10
    jae cmp_end
    movzx eax, byte ptr [r11 + rdx]
    movzx ecx, byte ptr [rsi + rdx]
    cmp al, cl
    jb cmp_less
    ja cmp_greater
    inc rdx
    jmp cmp_loop
cmp_end:
    cmp r8d, r9d
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, 0xFFFFFFFF
    ret
cmp_greater:
    mov eax, 1
    ret

sort_lines:
    cmp r13, 2
    jb sort_done
    sub rsp, 48
    mov r14, 1
width_loop:
    cmp r14, r13
    jae sort_done_pop
    mov qword ptr [rsp + 40], 0
left_loop:
    mov rax, qword ptr [rsp + 40]
    cmp rax, r13
    jae width_done
    lea rax, [rax + r14]
    cmp rax, r13
    jbe mid_ok
    mov rax, r13
mid_ok:
    mov qword ptr [rsp + 0], rax
    mov rax, qword ptr [rsp + 40]
    lea rax, [rax + r14*2]
    cmp rax, r13
    jbe right_ok
    mov rax, r13
right_ok:
    mov qword ptr [rsp + 8], rax
    mov rax, qword ptr [rsp + 40]
    mov qword ptr [rsp + 16], rax
    mov qword ptr [rsp + 32], rax
    mov rax, qword ptr [rsp + 0]
    mov qword ptr [rsp + 24], rax
merge_loop:
    mov rax, qword ptr [rsp + 16]
    cmp rax, qword ptr [rsp + 0]
    jae left_exhausted
    mov rax, qword ptr [rsp + 24]
    cmp rax, qword ptr [rsp + 8]
    jae take_left
    mov rdx, qword ptr [rsp + 16]
    mov rcx, qword ptr [rsp + 24]
    mov rdi, qword ptr [rbx + rdx*8]
    mov rsi, qword ptr [rbx + rcx*8]
    call compare_packed
    test eax, eax
    jg take_right
    jmp take_left
left_exhausted:
    mov rax, qword ptr [rsp + 24]
    cmp rax, qword ptr [rsp + 8]
    jae merge_done
    jmp take_right
take_left:
    mov rdx, qword ptr [rsp + 16]
    mov rax, qword ptr [rbx + rdx*8]
    mov rcx, qword ptr [rsp + 32]
    mov qword ptr [r12 + rcx*8], rax
    add qword ptr [rsp + 16], 1
    add qword ptr [rsp + 32], 1
    jmp merge_loop
take_right:
    mov rdx, qword ptr [rsp + 24]
    mov rax, qword ptr [rbx + rdx*8]
    mov rcx, qword ptr [rsp + 32]
    mov qword ptr [r12 + rcx*8], rax
    add qword ptr [rsp + 24], 1
    add qword ptr [rsp + 32], 1
    jmp merge_loop
merge_done:
    mov rax, qword ptr [rsp + 40]
    mov qword ptr [rsp + 16], rax
copy_loop:
    mov rax, qword ptr [rsp + 16]
    cmp rax, qword ptr [rsp + 8]
    jae copy_done
    mov rdx, qword ptr [rsp + 16]
    mov rcx, qword ptr [r12 + rdx*8]
    mov qword ptr [rbx + rdx*8], rcx
    add qword ptr [rsp + 16], 1
    jmp copy_loop
copy_done:
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [rsp + 40], rax
    jmp left_loop
width_done:
    shl r14, 1
    jmp width_loop
sort_done_pop:
    add rsp, 48
sort_done:
    ret

write_all:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
wa_loop:
    test r10, r10
    jz wa_done
    mov eax, 1
    mov rdi, r8
    mov rsi, r9
    mov rdx, r10
    syscall
    cmp rax, 0
    jle wa_done
    add r9, rax
    sub r10, rax
    jmp wa_loop
wa_done:
    ret
