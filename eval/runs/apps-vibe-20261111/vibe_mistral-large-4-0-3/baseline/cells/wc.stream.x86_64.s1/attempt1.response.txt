.globl _start
.equ BUF_SIZE, 65536

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, BUF_SIZE
    sub rdx, r12
    test rdx, rdx
    jz read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + buf]
    add rsi, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + buf]
    mov rcx, r12
    xor r8, r8
    xor r9, r9
    xor r10, r10
scan_loop:
    test rcx, rcx
    jz scan_done
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne no_newline
    inc r8
no_newline:
    cmp al, 32
    je is_ws
    cmp al, 13
    ja not_ws
    cmp al, 9
    jb not_ws
is_ws:
    xor r10, r10
    jmp word_done
not_ws:
    test r10, r10
    jnz word_done
    inc r9
    mov r10, 1
word_done:
    jmp scan_loop
scan_done:
    mov qword ptr [rip + counts], r8
    mov qword ptr [rip + counts + 8], r9
    mov qword ptr [rip + counts + 16], r12
    lea rbx, [rip + counts]
    lea rdi, [rip + outbuf]
    xor r15, r15
out_loop:
    mov rax, qword ptr [rbx + r15*8]
    lea rsi, [rip + tmpbuf_end]
    xor r9, r9
conv_loop:
    xor edx, edx
    mov r11, 10
    div r11
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc r9
    test rax, rax
    jnz conv_loop
copy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec r9
    jnz copy_loop
    cmp r15, 2
    je out_newline
    mov byte ptr [rdi], 32
    jmp out_sep_done
out_newline:
    mov byte ptr [rdi], 10
out_sep_done:
    inc rdi
    inc r15
    cmp r15, 3
    jb out_loop
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

.bss
.balign 16
buf:
    .zero BUF_SIZE
counts:
    .zero 24
outbuf:
    .zero 64
tmpbuf:
    .zero 32
tmpbuf_end:
