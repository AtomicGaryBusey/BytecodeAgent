.globl _start

.text
_start:
    // ---- read all of standard input into inbuf (at most 49152 bytes) ----
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf          // x20 = inbuf base
    mov x19, #0                        // x19 = total bytes read so far
read_loop:
    mov x2, #49152
    sub x2, x2, x19                    // space left in inbuf
    cbz x2, read_done
    mov x0, #0                         // fd 0
    add x1, x20, x19                   // buf = inbuf + total
    mov x8, #63                        // read
    svc #0
    cmp x0, #0
    b.le read_done                     // 0 = end of input, negative = error
    add x19, x19, x0
    b read_loop
read_done:
    add x26, x20, x19                  // x26 = end of input
    mov x23, x20                       // x23 = input cursor
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf         // x21 = output cursor
    mov x22, #0                        // x22 = column (chars in current line)
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet       // x24 = base64 alphabet table
encode_loop:
    cmp x23, x26
    b.hs encode_done
    sub x9, x26, x23                   // x9 = bytes remaining in this group (1..3)
    ldrb w10, [x23], #1                // b0
    mov w11, #0                        // b1 (zero padded)
    mov w12, #0                        // b2 (zero padded)
    cmp x9, #2
    b.lo 1f
    ldrb w11, [x23], #1                // b1
    cmp x9, #3
    b.lo 1f
    ldrb w12, [x23], #1                // b2
1:
    lsr w13, w10, #2                   // c0 = b0 >> 2
    and w14, w10, #3
    lsl w14, w14, #4
    lsr w15, w11, #4
    orr w14, w14, w15                  // c1 = ((b0 & 3) << 4) | (b1 >> 4)
    and w15, w11, #0xF
    lsl w15, w15, #2
    lsr w16, w12, #6
    orr w15, w15, w16                  // c2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and w16, w12, #0x3F                // c3 = b2 & 0x3F
    ldrb w13, [x24, w13, uxtw #0]      // map sextets through the alphabet
    ldrb w14, [x24, w14, uxtw #0]
    ldrb w15, [x24, w15, uxtw #0]
    ldrb w16, [x24, w16, uxtw #0]
    cmp x9, #1
    b.ne 2f
    mov w15, #0x3D                     // '='
    mov w16, #0x3D                     // '='
    b 3f
2:
    cmp x9, #2
    b.ne 3f
    mov w16, #0x3D                     // '='
3:
    strb w13, [x21], #1
    strb w14, [x21], #1
    strb w15, [x21], #1
    strb w16, [x21], #1
    add x22, x22, #4
    cmp x22, #76
    b.lo 4f
    mov w9, #0x0A                      // line is full: newline
    strb w9, [x21], #1
    sub x22, x22, #76
4:
    b encode_loop
encode_done:
    cbz x22, 5f
    mov w9, #0x0A
    strb w9, [x21], #1                 // newline at end of last (short) line
5:
    // ---- write the whole output buffer to standard output ----
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x21, x1                    // length
    mov x0, #1                         // fd 1
    mov x8, #64                        // write
write_loop:
    cbz x2, do_exit
    svc #0
    cmp x0, #0
    b.le do_exit
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop
do_exit:
    mov x0, #0
    mov x8, #94                        // exit_group
    svc #0

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 66560
