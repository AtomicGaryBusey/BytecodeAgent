.text
.globl _start
_start:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r12d, r12d
read_loop:
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jz finish
    mov r14, rax
    xor r13d, r13d
scan_loop:
    cmp r13, r14
    jae read_loop
    movzx eax, byte ptr [rsi + r13]
    inc r13
    add r10, 1
    cmp al, 10
    je newline
    cmp al, 32
    je whitespace
    lea edx, [rax - 9]
    cmp edx, 4
    jbe whitespace
    test r12d, r12d
    jnz scan_loop
    inc r9
    mov r12d, 1
    jmp scan_loop
newline:
    inc r8
whitespace:
    xor r12d, r12d
    jmp scan_loop
finish:
    lea rdi, [rip + out_end]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r10
    call emit_dec
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r9
    call emit_dec
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r8
    call emit_dec
    mov rbx, rdi
    lea rsi, [rip + out_end]
    sub rsi, rbx
    mov r14, rsi
write_loop:
    test r14, r14
    jz program_exit
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r14
    syscall
    test rax, rax
    jz program_exit
    js program_exit
    add rbx, rax
    sub r14, rax
    jmp write_loop
program_exit:
    mov eax, 60
    xor edi, edi
    syscall
emit_dec:
1:
    xor edx, edx
    mov rsi, 10
    div rsi
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 96
out_end:
