# sort_lines: read stdin, sort lines byte-wise ascending, write stdout
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .zero 32768
.balign 16
records:
    .zero 16000
.balign 16
aux_recs:
    .zero 16000

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov rax, 32768
    sub rax, r12
    test rax, rax
    jz read_done
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, rax
    mov eax, SYS_READ
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
split_loop:
    cmp r15, r12
    jae split_done
    lea rax, [rip + input_buf]
    movzx ebx, byte ptr [rax + r15]
    cmp ebx, 0x0A
    jne split_next
    lea rdx, [rip + records]
    mov rax, r13
    shl rax, 4
    add rdx, rax
    lea rcx, [rip + input_buf]
    add rcx, r14
    mov qword ptr [rdx], rcx
    mov rax, r15
    sub rax, r14
    mov qword ptr [rdx + 8], rax
    inc r13
    inc r15
    mov r14, r15
    jmp split_loop
split_next:
    inc r15
    jmp split_loop
split_done:
    cmp r14, r12
    jae no_final_line
    lea rdx, [rip + records]
    mov rax, r13
    shl rax, 4
    add rdx, rax
    lea rcx, [rip + input_buf]
    add rcx, r14
    mov qword ptr [rdx], rcx
    mov rax, r12
    sub rax, r14
    mov qword ptr [rdx + 8], rax
    inc r13
no_final_line:
    cmp r13, 1
    jbe sort_done
    xor edi, edi
    mov rsi, r13
    call merge_sort
sort_done:
    xor r14d, r14d
out_loop:
    cmp r14, r13
    jae out_done
    lea rax, [rip + records]
    mov rcx, r14
    shl rcx, 4
    add rax, rcx
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    mov edi, 1
    mov eax, SYS_WRITE
    syscall
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, SYS_WRITE
    syscall
    inc r14
    jmp out_loop
out_done:
    xor edi, edi
    mov eax, SYS_EXIT
    syscall

compare:
    mov rax, qword ptr [rdi]
    mov rcx, qword ptr [rsi]
    mov r8, qword ptr [rdi + 8]
    mov r9, qword ptr [rsi + 8]
    mov r10, r8
    cmp r8, r9
    jbe cmp_have_min
    mov r10, r9
cmp_have_min:
    xor edx, edx
cmp_loop:
    cmp rdx, r10
    jae cmp_prefix
    movzx r11d, byte ptr [rax + rdx]
    movzx esi, byte ptr [rcx + rdx]
    cmp r11d, esi
    jb cmp_less
    ja cmp_greater
    inc rdx
    jmp cmp_loop
cmp_prefix:
    cmp r8, r9
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret

merge_sort:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
    mov rax, r12
    sub rax, rbx
    cmp rax, 1
    jbe ms_done
    mov r13, rbx
    add r13, r12
    shr r13, 1
    mov rdi, rbx
    mov rsi, r13
    call merge_sort
    mov rdi, r13
    mov rsi, r12
    call merge_sort
    mov r14, rbx
    mov r15, r13
    mov rbp, rbx
ms_loop:
    cmp r14, r13
    jae ms_copy_right
    cmp r15, r12
    jae ms_copy_left
    lea rdi, [rip + records]
    mov rax, r14
    shl rax, 4
    add rdi, rax
    lea rsi, [rip + records]
    mov rax, r15
    shl rax, 4
    add rsi, rax
    call compare
    test eax, eax
    jle ms_take_left
    lea rdi, [rip + aux_recs]
    mov rax, rbp
    shl rax, 4
    add rdi, rax
    lea rsi, [rip + records]
    mov rax, r15
    shl rax, 4
    add rsi, rax
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc r15
    inc rbp
    jmp ms_loop
ms_take_left:
    lea rdi, [rip + aux_recs]
    mov rax, rbp
    shl rax, 4
    add rdi, rax
    lea rsi, [rip + records]
    mov rax, r14
    shl rax, 4
    add rsi, rax
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc r14
    inc rbp
    jmp ms_loop
ms_copy_left:
    cmp r14, r13
    jae ms_copy_back
    lea rdi, [rip + aux_recs]
    mov rax, rbp
    shl rax, 4
    add rdi, rax
    lea rsi, [rip + records]
    mov rax, r14
    shl rax, 4
    add rsi, rax
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc r14
    inc rbp
    jmp ms_copy_left
ms_copy_right:
    cmp r15, r12
    jae ms_copy_back
    lea rdi, [rip + aux_recs]
    mov rax, rbp
    shl rax, 4
    add rdi, rax
    lea rsi, [rip + records]
    mov rax, r15
    shl rax, 4
    add rsi, rax
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc r15
    inc rbp
    jmp ms_copy_right
ms_copy_back:
    mov r14, rbx
ms_copy_back_loop:
    cmp r14, r12
    jae ms_done
    lea rdi, [rip + records]
    mov rax, r14
    shl rax, 4
    add rdi, rax
    lea rsi, [rip + aux_recs]
    mov rax, r14
    shl rax, 4
    add rsi, rax
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc r14
    jmp ms_copy_back_loop
ms_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
