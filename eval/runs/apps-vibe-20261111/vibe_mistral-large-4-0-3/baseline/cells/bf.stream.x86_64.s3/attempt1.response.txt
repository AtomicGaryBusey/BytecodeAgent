.globl _start
.text
_start:
    xor r8d, r8d
    lea rsi, [rip + inbuf]
    mov r9d, 65536
read_loop:
    xor edi, edi
    mov rdx, r9
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add rsi, rax
    sub r9, rax
    jz read_done
    jmp read_loop
read_done:
    lea rbx, [rip + inbuf]
    xor ecx, ecx
scan_bang:
    cmp rcx, r8
    jae no_bang
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x21
    je found_bang
    inc rcx
    jmp scan_bang
found_bang:
    mov r9, r8
    sub r9, rcx
    dec r9
    mov r8, rcx
    lea r10, [rbx + rcx + 1]
    jmp parse_done
no_bang:
    xor r9d, r9d
    lea r10, [rbx + r8]
parse_done:
    lea rbp, [rip + jtab]
    lea rdx, [rip + openstk]
    mov rsi, rdx
    xor ecx, ecx
bal_loop:
    cmp rcx, r8
    jae bal_end
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x5B
    je bal_open
    cmp al, 0x5D
    je bal_close
    jmp bal_next
bal_open:
    mov dword ptr [rsi], ecx
    add rsi, 4
    jmp bal_next
bal_close:
    cmp rsi, rdx
    je unbalanced
    sub rsi, 4
    mov eax, dword ptr [rsi]
    mov edi, eax
    inc edi
    mov dword ptr [rbp + rcx*4], edi
    mov edi, ecx
    inc edi
    mov dword ptr [rbp + rax*4], edi
    jmp bal_next
bal_next:
    inc rcx
    jmp bal_loop
bal_end:
    cmp rsi, rdx
    jne unbalanced
    jmp run_init
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall
run_init:
    xor r12d, r12d
    lea r13, [rip + tape]
    xor r14d, r14d
    xor r15d, r15d
run_loop:
    cmp r12, r8
    jae run_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_dot
    cmp al, 0x2C
    je cmd_comma
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    inc r12
    jmp run_loop
cmd_gt:
    inc r13
    inc r12
    jmp run_loop
cmd_lt:
    dec r13
    inc r12
    jmp run_loop
cmd_plus:
    add byte ptr [r13], 1
    inc r12
    jmp run_loop
cmd_minus:
    sub byte ptr [r13], 1
    inc r12
    jmp run_loop
cmd_dot:
    lea rsi, [rip + outbuf]
    mov al, byte ptr [r13]
    mov byte ptr [rsi + r15], al
    inc r15
    cmp r15, 65536
    jne dot_next
    lea rsi, [rip + outbuf]
    mov edx, 65536
dot_flush_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz dot_flush_loop
    xor r15d, r15d
dot_next:
    inc r12
    jmp run_loop
cmd_comma:
    cmp r14, r9
    jae comma_zero
    mov al, byte ptr [r10 + r14]
    inc r14
    mov byte ptr [r13], al
    jmp comma_next
comma_zero:
    mov byte ptr [r13], 0
comma_next:
    inc r12
    jmp run_loop
cmd_open:
    cmp byte ptr [r13], 0
    jne open_fall
    mov eax, dword ptr [rbp + r12*4]
    mov r12, rax
    jmp run_loop
open_fall:
    inc r12
    jmp run_loop
cmd_close:
    cmp byte ptr [r13], 0
    je close_fall
    mov eax, dword ptr [rbp + r12*4]
    mov r12, rax
    jmp run_loop
close_fall:
    inc r12
    jmp run_loop
run_done:
    test r15, r15
    jz exit_ok
    lea rsi, [rip + outbuf]
    mov rdx, r15
final_flush_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz final_flush_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
.bss
.balign 8
inbuf:
    .zero 65536
.balign 8
tape:
    .zero 30000
.balign 8
outbuf:
    .zero 65536
.balign 8
jtab:
    .zero 262144
.balign 8
openstk:
    .zero 262144
