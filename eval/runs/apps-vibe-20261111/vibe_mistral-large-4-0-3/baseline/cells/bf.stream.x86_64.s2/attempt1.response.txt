.text
.equ BUFSZ, 65536
.equ CODESZ, 65536
.equ TAPESZ, 30000
.equ MATCHSZ, 262144
.globl _start
_start:
    lea r9, [rip + input_buf]
    xor r8, r8
    mov r10, BUFSZ
read_loop:
    test r10, r10
    jz read_done
    xor edi, edi
    mov rsi, r9
    add rsi, r8
    mov rdx, r10
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    sub r10, rax
    jmp read_loop
read_done:
    mov r13, r8
    xor r12, r12
scan_loop:
    cmp r12, r13
    jae no_bang
    movzx eax, byte ptr [r9 + r12]
    cmp al, 0x21
    je found_bang
    inc r12
    jmp scan_loop
found_bang:
    mov r13, r12
    lea r14, [r12 + 1]
    mov r15, r8
    sub r15, r12
    dec r15
    jmp have_parts
no_bang:
    mov r13, r8
    mov r14, r8
    xor r15, r15
have_parts:
    xor r12, r12
    xor r8, r8
    xor rbx, rbx
build_loop:
    cmp r12, r13
    jae build_done
    movzx eax, byte ptr [r9 + r12]
    cmp al, 0x3E
    je b_cmd
    cmp al, 0x3C
    je b_cmd
    cmp al, 0x2B
    je b_cmd
    cmp al, 0x2D
    je b_cmd
    cmp al, 0x2E
    je b_cmd
    cmp al, 0x2C
    je b_cmd
    cmp al, 0x5B
    je b_open
    cmp al, 0x5D
    je b_close
    jmp build_next
b_cmd:
    mov byte ptr [code + r8], al
    inc r8
    jmp build_next
b_open:
    mov dword ptr [bstack + rbx*4], r8d
    inc rbx
    mov byte ptr [code + r8], al
    inc r8
    jmp build_next
b_close:
    test rbx, rbx
    jz build_bad
    mov byte ptr [code + r8], al
    dec rbx
    mov eax, dword ptr [bstack + rbx*4]
    mov dword ptr [match + r8*4], eax
    mov dword ptr [match + rax*4], r8d
    inc r8
    jmp build_next
build_next:
    inc r12
    jmp build_loop
build_done:
    test rbx, rbx
    jnz build_bad
    mov r13, r8
    jmp exec_init
build_bad:
    mov eax, 60
    mov edi, 1
    syscall

exec_init:
    lea rbx, [rip + tape]
    lea r10, [rip + code]
    lea r11, [rip + match]
    lea r8, [rip + out_buf]
    xor r12, r12
    xor rbp, rbp
exec_loop:
    cmp r12, r13
    jae exec_done
    movzx eax, byte ptr [r10 + r12]
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_out
    cmp al, 0x2C
    je cmd_in
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    jmp cmd_next
cmd_gt:
    inc rbx
    jmp cmd_next
cmd_lt:
    dec rbx
    jmp cmd_next
cmd_plus:
    inc byte ptr [rbx]
    jmp cmd_next
cmd_minus:
    dec byte ptr [rbx]
    jmp cmd_next
cmd_out:
    mov al, byte ptr [rbx]
    mov byte ptr [r8 + rbp], al
    inc rbp
    jmp cmd_next
cmd_in:
    test r15, r15
    jz cmd_in_eof
    movzx eax, byte ptr [r9 + r14]
    mov byte ptr [rbx], al
    inc r14
    dec r15
    jmp cmd_next
cmd_in_eof:
    mov byte ptr [rbx], 0
    jmp cmd_next
cmd_open:
    cmp byte ptr [rbx], 0
    jne cmd_next
    mov eax, dword ptr [r11 + r12*4]
    mov r12, rax
    jmp cmd_next
cmd_close:
    cmp byte ptr [rbx], 0
    je cmd_next
    mov eax, dword ptr [r11 + r12*4]
    mov r12, rax
    jmp cmd_next
cmd_next:
    inc r12
    jmp exec_loop
exec_done:
    test rbp, rbp
    jz exit0
    mov rsi, r8
    mov rdx, rbp
write_loop:
    test rdx, rdx
    jz exit0
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buf:
    .zero BUFSZ
.balign 16
code:
    .zero CODESZ
.balign 16
match:
    .zero MATCHSZ
.balign 16
bstack:
    .zero MATCHSZ
.balign 16
tape:
    .zero TAPESZ
.balign 16
out_buf:
    .zero BUFSZ
