.equ BUFSZ, 65536
.equ STATUS_OK, 0
.equ STATUS_ERR, 1
.equ STATUS_INCOMPLETE, 2
.equ OK_LEN, 3
.equ ERR_PREFIX_LEN, 9

.text

.globl _start

    mov x0, #1
    mov x8, #64

    ldrb w0, [x21, x19]

    cmp w0, #0x09
    b.eq sw_next
    cmp w0, #0x0A
    b.eq sw_next
    cmp w0, #0x0D

    add x19, x19, #1
    b sw_loop

    cmp x19, x20
    b.hs pv_incomplete
    ldrb w0, [x21, x19]

    cmp w0, #0x5B
    b.eq pv_array
    cmp w0, #0x22
    b.eq pv_string
    cmp w0, #0x74
    b.eq pv_true
    cmp w0, #0x66
    b.eq pv_false
    cmp w0, #0x6E
    b.eq pv_null
    cmp w0, #0x2D
    b.eq pv_number

    cmp w1, #9

    b pv_done

    b pv_done

    add x19, x19, #1

    cmp x19, x20
    b.hs ps_incomplete
    ldrb w0, [x21, x19]
    cmp w0, #0x22
    b.eq ps_end
    cmp w0, #0x5C
    b.eq ps_esc
    cmp w0, #0x20
    b.lo ps_invalid
    add x19, x19, #1
    b ps_loop

    add x19, x19, #1

    cmp x19, x20
    b.hs ps_incomplete
    ldrb w0, [x21, x19]
    cmp w0, #0x22
    b.eq ps_esc_ok
    cmp w0, #0x5C
    b.eq ps_esc_ok
    cmp w0, #0x2F
    b.eq ps_esc_ok
    cmp w0, #0x62
    b.eq ps_esc_ok
    cmp w0, #0x66
    b.eq ps_esc_ok
    cmp w0, #0x6E
    b.eq ps_esc_ok
    cmp w0, #0x72

    cmp w0, #0x74
