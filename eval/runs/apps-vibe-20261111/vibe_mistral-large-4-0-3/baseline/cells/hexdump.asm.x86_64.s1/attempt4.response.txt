.text
.globl _start

put_hex8:
    mov r8d, 8
    mov cl, 28
ph8_loop:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    lea rsi, [rip + hexdigits]
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    sub cl, 4
    dec r8d
    jnz ph8_loop
    ret

put_hex2:
    movzx eax, al
    mov rdx, rax
    shr rdx, 4
    lea rsi, [rip + hexdigits]
    movzx ecx, byte ptr [rsi + rdx]
    mov byte ptr [rbx], cl
    inc rbx
    and eax, 15
    movzx ecx, byte ptr [rsi + rax]
    mov byte ptr [rbx], cl
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
    ret

_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rbp, [rip + inbuf]
    lea rbx, [rip + outbuf]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae rows_done
    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe row_n_ok
    mov r15d, 16
row_n_ok:
    mov rax, r13
    call put_hex8
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
    xor r14d, r14d
slot_loop:
    cmp r14, r15
    jae pad_slot
    mov rdx, r13
    add rdx, r14
    movzx eax, byte ptr [rbp + rdx]
    call put_hex2
    jmp after_slot
pad_slot:
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
after_slot:
    cmp r14, 7
    jne no_extra
    mov byte ptr [rbx], 0x20
    inc rbx
no_extra:
    inc r14
    cmp r14, 16
    jb slot_loop
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x7c
    inc rbx
    xor r14d, r14d
char_loop:
    cmp r14, r15
    jae char_done
    mov rdx, r13
    add rdx, r14
    movzx eax, byte ptr [rbp + rdx]
    cmp al, 0x20
    jb dot_char
    cmp al, 0x7e
    ja dot_char
    mov byte ptr [rbx], al
    jmp stored_char
dot_char:
    mov byte ptr [rbx], 0x2e
stored_char:
    inc rbx
    inc r14
    jmp char_loop
char_done:
    mov byte ptr [rbx], 0x7c
    inc rbx
    mov byte ptr [rbx], 0x0a
    inc rbx
    add r13, 16
    jmp row_loop
rows_done:
    mov rax, r12
    call put_hex8
    mov byte ptr [rbx], 0x0a
    inc rbx
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
write_loop:
    mov edi, 1
    mov eax, 1
    syscall
    cmp rax, rdx
    je write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 65536
outbuf:
    .zero 323600
