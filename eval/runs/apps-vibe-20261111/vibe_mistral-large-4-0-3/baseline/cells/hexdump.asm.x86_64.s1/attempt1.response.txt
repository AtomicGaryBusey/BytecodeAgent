.equ INPUT_MAX, 65536
.equ OUT_MAX, 323600
.equ ROW_BYTES, 16

.text
.globl _start
_start:
    xor r12d, r12d
    lea rsi, [rip + input_buf]
    mov r13d, INPUT_MAX
read_loop:
    test r13, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rdx, r13
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    add rsi, rax
    sub r13, rax
    jmp read_loop
read_done:
    mov rbp, r12
    lea rsi, [rip + input_buf]
    lea rbx, [rip + out_buf]
    lea r9, [rip + hexdigits]
    xor r13d, r13d
row_loop:
    cmp r13, rbp
    jae rows_done
    mov r14, rbp
    sub r14, r13
    cmp r14, ROW_BYTES
    jbe have_n
    mov r14d, ROW_BYTES
have_n:
    mov rax, r13
    mov rcx, 56
    mov r8d, 8
off_loop:
    mov rdx, rax
    shr rdx, cl
    and edx, 0xf
    movzx r10d, byte ptr [r9 + rdx]
    mov byte ptr [rbx], r10b
    inc rbx
    sub rcx, 8
    dec r8d
    jnz off_loop
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    add rbx, 2
    xor r15d, r15d
slot_loop:
    cmp r15, ROW_BYTES
    jae slots_done
    cmp r15, r14
    jae pad_slot
    movzx eax, byte ptr [rsi + r15]
    mov rdx, rax
    shr rdx, 4
    movzx r10d, byte ptr [r9 + rdx]
    mov byte ptr [rbx], r10b
    mov rdx, rax
    and edx, 0xf
    movzx r10d, byte ptr [r9 + rdx]
    mov byte ptr [rbx + 1], r10b
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
    jmp next_slot
pad_slot:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x20
    mov byte ptr [rbx + 2], 0x20
    add rbx, 3
next_slot:
    cmp r15, 7
    jne no_extra
    mov byte ptr [rbx], 0x20
    inc rbx
no_extra:
    inc r15
    jmp slot_loop
slots_done:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx + 1], 0x7c
    add rbx, 2
    xor r15d, r15d
ascii_loop:
    cmp r15, r14
    jae ascii_done
    movzx eax, byte ptr [rsi + r15]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7e
    ja ascii_dot
    mov byte ptr [rbx], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rbx], 0x2e
ascii_next:
    inc rbx
    inc r15
    jmp ascii_loop
ascii_done:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx + 1], 0x0a
    add rbx, 2
    add rsi, r14
    add r13, r14
    jmp row_loop
rows_done:
    mov rax, rbp
    mov rcx, 56
    mov r8d, 8
fin_loop:
    mov rdx, rax
    shr rdx, cl
    and edx, 0xf
    movzx r10d, byte ptr [r9 + rdx]
    mov byte ptr [rbx], r10b
    inc rbx
    sub rcx, 8
    dec r8d
    jnz fin_loop
    mov byte ptr [rbx], 0x0a
    inc rbx
    lea rsi, [rip + out_buf]
    mov rdx, rbx
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
.balign 16
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero INPUT_MAX
out_buf:
    .zero OUT_MAX
