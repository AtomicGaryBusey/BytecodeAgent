.equ INBUF_SIZE, 65536
.text
.globl _start
_start:
    lea r15, [rip + inbuf]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INBUF_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, INBUF_SIZE
    jae read_done
    jmp read_loop
read_done:
    add r15, r12
    lea rsi, [rip + inbuf]
    jmp process_loop
process_loop:
    cmp rsi, r15
    jae all_done
    mov r14, rsi
scan_nl:
    cmp r14, r15
    jae line_have
    cmp byte ptr [r14], 0x0A
    je line_have
    inc r14
    jmp scan_nl
line_have:
    lea rbx, [rip + stk]
    lea r8, [rip + stk]
line_tokens:
    cmp rsi, r14
    jae line_finish
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je sep_adv
    cmp al, 0x09
    je sep_adv
    jmp token_begin
sep_adv:
    inc rsi
    jmp line_tokens
token_begin:
    mov r9, rsi
tok_scan:
    cmp rsi, r14
    jae tok_have
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je tok_have
    cmp al, 0x09
    je tok_have
    inc rsi
    jmp tok_scan
tok_have:
    mov r10, rsi
    sub r10, r9
    movzx eax, byte ptr [r9]
    cmp al, 0x2D
    je tok_minus
    cmp al, 0x2B
    je tok_plus
    cmp al, 0x2A
    je tok_mul
    cmp al, 0x2F
    je tok_div
    cmp al, 0x25
    je tok_mod
    sub al, 0x30
    cmp al, 9
    ja line_error
    cmp r10, 18
    ja line_error
    call parse_digits
    test r11d, r11d
    jnz line_error
    jmp push_rax
tok_minus:
    cmp r10, 1
    je op_sub
    inc r9
    dec r10
    cmp r10, 18
    ja line_error
    call parse_digits
    test r11d, r11d
    jnz line_error
    neg rax
    jmp push_rax
tok_plus:
    cmp r10, 1
    jne line_error
    jmp op_add
tok_mul:
    cmp r10, 1
    jne line_error
    jmp op_mul
tok_div:
    cmp r10, 1
    jne line_error
    jmp op_div
tok_mod:
    cmp r10, 1
    jne line_error
    jmp op_mod
push_rax:
    mov [r8], rax
    add r8, 8
    jmp line_tokens
op_add:
    lea rcx, [rbx + 16]
    cmp r8, rcx
    jb line_error
    sub r8, 8
    mov rdx, [r8]
    sub r8, 8
    mov rax, [r8]
    add rax, rdx
    jmp push_rax
op_sub:
    lea rcx, [rbx + 16]
    cmp r8, rcx
    jb line_error
    sub r8, 8
    mov rdx, [r8]
    sub r8, 8
    mov rax, [r8]
    sub rax, rdx
    jmp push_rax
op_mul:
    lea rcx, [rbx + 16]
    cmp r8, rcx
    jb line_error
    sub r8, 8
    mov rdx, [r8]
    sub r8, 8
    mov rax, [r8]
    imul rax, rdx
    jmp push_rax
op_div:
    lea rcx, [rbx + 16]
    cmp r8, rcx
    jb line_error
    sub r8, 8
    mov r11, [r8]
    sub r8, 8
    mov rax, [r8]
    test r11, r11
    jz line_error
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne div_ok
    cmp r11, -1
    je line_error
div_ok:
    cqo
    idiv r11
    jmp push_rax
op_mod:
    lea rcx, [rbx + 16]
    cmp r8, rcx
    jb line_error
    sub r8, 8
    mov r11, [r8]
    sub r8, 8
    mov rax, [r8]
    test r11, r11
    jz line_error
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne mod_ok
    cmp r11, -1
    je line_error
mod_ok:
    cqo
    idiv r11
    mov rax, rdx
    jmp push_rax
line_finish:
    lea rcx, [rbx + 8]
    cmp r8, rcx
    jne line_error
    sub r8, 8
    mov rax, [r8]
    call print_num
    jmp line_advance
line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 6
    syscall
line_advance:
    cmp r14, r15
    jae all_done
    lea rsi, [r14 + 1]
    jmp process_loop
all_done:
    mov edi, 0
    mov eax, 60
    syscall
parse_digits:
    xor eax, eax
    mov rcx, r10
    xor r11d, r11d
pd_loop:
    movzx edx, byte ptr [r9]
    sub edx, 0x30
    cmp edx, 9
    ja pd_bad
    imul rax, rax, 10
    add rax, rdx
    inc r9
    dec rcx
    jnz pd_loop
    ret
pd_bad:
    mov r11d, 1
    ret
print_num:
    push rdi
    lea r10, [rip + numbuf + 32]
    mov byte ptr [r10 - 1], 0x0A
    dec r10
    xor r11d, r11d
    test rax, rax
    jns pn_nosign
    neg rax
    mov r11d, 1
pn_nosign:
    mov ecx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, 0x30
    mov byte ptr [r10 - 1], dl
    dec r10
    test rax, rax
    jnz pn_loop
    test r11d, r11d
    jz pn_write
    mov byte ptr [r10 - 1], 0x2D
    dec r10
pn_write:
    mov rsi, r10
    lea rdx, [rip + numbuf + 32]
    sub rdx, r10
    mov edi, 1
    mov eax, 1
    syscall
    pop rdi
    ret
.section .rodata
errmsg:
    .ascii "error\n"
.bss
.balign 8
inbuf:
    .zero INBUF_SIZE
.balign 8
stk:
    .zero 524288
.balign 8
numbuf:
    .zero 32
