.equ INPUT_MAX, 49152
.equ OUT_MAX, 66560
.equ LINE_LEN, 76

.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, INPUT_MAX
    sub edx, ebx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, INPUT_MAX
    jae read_done
    jmp read_loop
read_done:
    test rbx, rbx
    jz exit_program
    lea r13, [r12 + rbx]
    lea r14, [rip + outbuf]
    lea rbp, [rip + alphabet]
    xor r15d, r15d
full_loop:
    mov rax, r13
    sub rax, r12
    cmp rax, 3
    jb tail
    movzx r8d, byte ptr [r12]
    movzx r9d, byte ptr [r12 + 1]
    movzx r10d, byte ptr [r12 + 2]
    mov esi, r8d
    shr esi, 2
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov esi, r8d
    and esi, 3
    shl esi, 4
    mov edi, r9d
    shr edi, 4
    or esi, edi
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov esi, r9d
    and esi, 0x0f
    shl esi, 2
    mov edi, r10d
    shr edi, 6
    or esi, edi
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov esi, r10d
    and esi, 0x3f
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    add r12, 3
    jmp full_loop
tail:
    mov rax, r13
    sub rax, r12
    test rax, rax
    jz final_newline
    cmp rax, 1
    je tail1
    movzx r8d, byte ptr [r12]
    movzx r9d, byte ptr [r12 + 1]
    mov esi, r8d
    shr esi, 2
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov esi, r8d
    and esi, 3
    shl esi, 4
    mov edi, r9d
    shr edi, 4
    or esi, edi
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov esi, r9d
    and esi, 0x0f
    shl esi, 2
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov al, 61
    call emit_char
    jmp final_newline
tail1:
    movzx r8d, byte ptr [r12]
    mov esi, r8d
    shr esi, 2
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov esi, r8d
    and esi, 3
    shl esi, 4
    movzx eax, byte ptr [rbp + rsi]
    call emit_char
    mov al, 61
    call emit_char
    mov al, 61
    call emit_char
final_newline:
    test r15, r15
    jz write_output
    mov byte ptr [r14], 10
    inc r14
write_output:
    lea r12, [rip + outbuf]
    mov r13, r14
    sub r13, r12
write_loop:
    test r13, r13
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jz exit_program
    add r12, rax
    sub r13, rax
    jmp write_loop
exit_program:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov byte ptr [r14], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d
1:
    ret

.section .rodata
.balign 8
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space INPUT_MAX
outbuf:
    .space OUT_MAX
