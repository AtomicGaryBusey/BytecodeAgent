.text
.globl _start
_start:
    xor r12, r12
read_loop:
    cmp r12, 32768
    jae read_done
    xor eax, eax
    mov edi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rbp, [rip + line_rec]
    lea r15, [rip + input_buf]
    xor r13, r13
    xor r14, r14
parse_loop:
    cmp r14, r12
    jae parse_done
    mov rbx, r14
scan_loop:
    cmp rbx, r12
    jae scan_no_newline
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 0x0A
    je scan_found
    inc rbx
    jmp scan_loop
scan_found:
    mov rax, rbx
    sub rax, r14
    mov rcx, rax
    shl rcx, 32
    mov rdx, r14
    or rcx, rdx
    mov [rbp + r13*8], rcx
    inc r13
    lea r14, [rbx + 1]
    jmp parse_loop
scan_no_newline:
    mov rax, r12
    sub rax, r14
    mov rcx, rax
    shl rcx, 32
    mov rdx, r14
    or rcx, rdx
    mov [rbp + r13*8], rcx
    inc r13
    mov r14, r12
    jmp parse_loop
parse_done:
    mov r12, r13
    mov r13, 1
sort_outer:
    cmp r13, r12
    jae sort_done
    mov r15, [rbp + r13*8]
    mov r14, r13
    dec r14
sort_inner:
    test r14, r14
    js sort_insert
    mov rbx, [rbp + r14*8]
    mov rdi, rbx
    mov rsi, r15
    call cmp_records
    test eax, eax
    jle sort_insert
    mov [rbp + r14*8 + 8], rbx
    dec r14
    jmp sort_inner
sort_insert:
    mov [rbp + r14*8 + 8], r15
    inc r13
    jmp sort_outer
sort_done:
    xor r13, r13
out_loop:
    cmp r13, r12
    jae out_done
    mov rax, [rbp + r13*8]
    mov r15d, eax
    lea rdx, [rip + input_buf]
    add r15, rdx
    mov r14, rax
    shr r14, 32
    test r14, r14
    jz out_newline
    mov edi, 1
    mov rsi, r15
    mov rdx, r14
    call write_all
out_newline:
    lea rsi, [rip + newline]
    mov edi, 1
    mov edx, 1
    call write_all
    inc r13
    jmp out_loop
out_done:
    mov edi, 0
    mov eax, 60
    syscall

cmp_records:
    mov eax, edi
    lea rdx, [rip + input_buf]
    add rax, rdx
    mov ecx, esi
    lea rdx, [rip + input_buf]
    add rcx, rdx
    mov rdx, rdi
    shr rdx, 32
    mov r8, rsi
    shr r8, 32
    mov r9, r8
    cmp rdx, r8
    cmovb r9, rdx
    xor esi, esi
cmp_loop:
    cmp rsi, r9
    jae cmp_after_loop
    movzx r10d, byte ptr [rax + rsi]
    movzx r11d, byte ptr [rcx + rsi]
    cmp r10d, r11d
    jb cmp_less
    ja cmp_greater
    inc rsi
    jmp cmp_loop
cmp_after_loop:
    cmp rdx, r8
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret

write_all:
    test rdx, rdx
    jz write_done
write_loop:
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_done:
    ret

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .zero 32768
line_rec:
    .zero 8000
