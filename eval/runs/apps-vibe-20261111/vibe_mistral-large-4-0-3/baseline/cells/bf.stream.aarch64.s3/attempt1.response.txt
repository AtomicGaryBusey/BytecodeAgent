.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_CAP, 65536
.equ OUT_CAP, 65536
.equ TAPE_CELLS, 30000
.equ JMP_BYTES, 262144
.equ BSTACK_BYTES, 4096
.equ CMDTAB_BYTES, 256
.equ CH_BANG, 0x21
.equ CH_PLUS, 0x2B
.equ CH_COMMA, 0x2C
.equ CH_MINUS, 0x2D
.equ CH_DOT, 0x2E
.equ CH_LT, 0x3C
.equ CH_GT, 0x3E
.equ CH_OPEN, 0x5B
.equ CH_CLOSE, 0x5D
.equ ID_COMMENT, 0
.equ ID_GT, 1
.equ ID_LT, 2
.equ ID_PLUS, 3
.equ ID_MINUS, 4
.equ ID_DOT, 5
.equ ID_COMMA, 6
.equ ID_OPEN, 7
.equ ID_CLOSE, 8

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #IN_CAP
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, x21
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    subs x21, x21, x0
    b.ne read_loop
read_done:
    mov x22, #0
scan_loop:
    cmp x22, x20
    b.hs no_bang
    ldrb w23, [x19, x22]
    cmp w23, #CH_BANG
    b.eq bang_found
    add x22, x22, #1
    b scan_loop
bang_found:
    mov x27, #1
    b split
no_bang:
    mov x27, #0
split:
    mov x21, x22
    sub x25, x20, x22
    sub x25, x25, x27
    add x24, x19, x22
    add x24, x24, x27
    mov x26, #0

    adrp x28, jmptab
    add x28, x28, :lo12:jmptab
    adrp x29, bstack
    add x29, x29, :lo12:bstack
    mov x20, #0
    mov x27, #0
bal_loop:
    cmp x20, x21
    b.hs bal_done
    ldrb w23, [x19, x20]
    cmp w23, #CH_OPEN
    b.eq bal_open
    cmp w23, #CH_CLOSE
    b.eq bal_close
    b bal_next
bal_open:
    str w20, [x29, x27, lsl #2]
    add x27, x27, #1
    b bal_next
bal_close:
    cbz x27, unbalanced
    sub x27, x27, #1
    ldr w22, [x29, x27, lsl #2]
    str w22, [x28, x20, lsl #2]
    str w20, [x28, x22, lsl #2]
    b bal_next
bal_next:
    add x20, x20, #1
    b bal_loop
bal_done:
    cbnz x27, unbalanced

    adrp x9, cmdtab
    add x9, x9, :lo12:cmdtab
    mov w15, #ID_GT
    strb w15, [x9, #CH_GT]
    mov w15, #ID_LT
    strb w15, [x9, #CH_LT]
    mov w15, #ID_PLUS
    strb w15, [x9, #CH_PLUS]
    mov w15, #ID_MINUS
    strb w15, [x9, #CH_MINUS]
    mov w15, #ID_DOT
    strb w15, [x9, #CH_DOT]
    mov w15, #ID_COMMA
    strb w15, [x9, #CH_COMMA]
    mov w15, #ID_OPEN
    strb w15, [x9, #CH_OPEN]
    mov w15, #ID_CLOSE
    strb w15, [x9, #CH_CLOSE]

    adrp x10, tape
    add x10, x10, :lo12:tape
    mov x11, x10
    adrp x16, outbuf
    add x16, x16, :lo12:outbuf
    mov x17, #0
    adrp x18, handlers
    add x18, x18, :lo12:handlers
    mov x20, #0
exec_loop:
    cmp x20, x21
    b.hs exec_done
    ldrb w14, [x19, x20]
    ldrb w15, [x9, x14]
    ldr x15, [x18, x15, lsl #3]
    br x15
exec_comment:
    add x20, x20, #1
    b exec_loop
cmd_gt:
    add x11, x11, #1
    add x20, x20, #1
    b exec_loop
cmd_lt:
    sub x11, x11, #1
    add x20, x20, #1
    b exec_loop
cmd_plus:
    ldrb w12, [x11]
    add w12, w12, #1
    strb w12, [x11]
    add x20, x20, #1
    b exec_loop
cmd_minus:
    ldrb w12, [x11]
    sub w12, w12, #1
    strb w12, [x11]
    add x20, x20, #1
    b exec_loop
cmd_dot:
    ldrb w12, [x11]
    strb w12, [x16, x17]
    add x17, x17, #1
    add x20, x20, #1
    b exec_loop
cmd_comma:
    cmp x26, x25
    b.hs comma_eof
    ldrb w12, [x24, x26]
    add x26, x26, #1
    strb w12, [x11]
    add x20, x20, #1
    b exec_loop
comma_eof:
    mov w12, #0
    strb w12, [x11]
    add x20, x20, #1
    b exec_loop
cmd_open:
    ldrb w12, [x11]
    cbnz w12, open_cont
    ldr w15, [x28, x20, lsl #2]
    add x20, x15, #1
    b exec_loop
open_cont:
    add x20, x20, #1
    b exec_loop
cmd_close:
    ldrb w12, [x11]
    cbz w12, close_cont
    ldr w15, [x28, x20, lsl #2]
    add x20, x15, #1
    b exec_loop
close_cont:
    add x20, x20, #1
    b exec_loop
exec_done:
    mov x20, #0
flush_loop:
    cmp x20, x17
    b.hs flush_done
    mov x0, #1
    add x1, x16, x20
    sub x2, x17, x20
    mov x8, #SYS_write
    svc #0
    add x20, x20, x0
    b flush_loop
flush_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
unbalanced:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

.section .rodata
.balign 8
handlers:
    .quad exec_comment
    .quad cmd_gt
    .quad cmd_lt
    .quad cmd_plus
    .quad cmd_minus
    .quad cmd_dot
    .quad cmd_comma
    .quad cmd_open
    .quad cmd_close

.bss
.balign 8
inbuf:
    .space IN_CAP
.balign 8
outbuf:
    .space OUT_CAP
.balign 8
tape:
    .space TAPE_CELLS
.balign 8
jmptab:
    .space JMP_BYTES
.balign 8
bstack:
    .space BSTACK_BYTES
.balign 8
cmdtab:
    .space CMDTAB_BYTES
