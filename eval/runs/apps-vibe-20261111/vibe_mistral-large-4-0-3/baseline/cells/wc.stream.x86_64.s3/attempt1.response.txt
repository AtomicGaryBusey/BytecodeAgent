.globl _start
.text
_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + input_buf]
    mov rcx, r12
    test rcx, rcx
    jz count_done
count_loop:
    movzx eax, byte ptr [rbx]
    cmp al, 0x0A
    jne not_nl
    inc r13
not_nl:
    cmp al, 0x20
    je ws
    cmp al, 0x09
    je ws
    cmp al, 0x0A
    je ws
    cmp al, 0x0B
    je ws
    cmp al, 0x0C
    je ws
    cmp al, 0x0D
    je ws
    test r15, r15
    jnz next_byte
    inc r14
    mov r15, 1
    jmp next_byte
ws:
    xor r15, r15
next_byte:
    inc rbx
    dec rcx
    jnz count_loop
count_done:
    mov qword ptr [rip + counts_arr], r13
    mov qword ptr [rip + counts_arr + 8], r14
    mov qword ptr [rip + counts_arr + 16], r12
    lea r15, [rip + counts_arr]
    lea rbx, [rip + outbuf]
    mov r8, 3
out_num_loop:
    mov rax, qword ptr [r15]
    lea r9, [rip + digitbuf + 24]
conv_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 0x30
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz conv_loop
    lea r10, [rip + digitbuf + 24]
    mov rdx, r10
    sub rdx, r9
    mov rsi, r9
    mov rdi, rbx
    mov rcx, rdx
    rep movsb
    add rbx, rdx
    cmp r8, 1
    je out_last
    mov byte ptr [rbx], 0x20
    inc rbx
    jmp out_advance
out_last:
    mov byte ptr [rbx], 0x0A
    inc rbx
out_advance:
    add r15, 8
    dec r8
    jnz out_num_loop
    lea rsi, [rip + outbuf]
    mov r12, rbx
    sub r12, rsi
write_loop:
    test r12, r12
    jz write_done
    mov rax, 1
    mov edi, 1
    mov rdx, r12
    syscall
    add rsi, rax
    sub r12, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall
.bss
.balign 16
input_buf:
    .zero 65536
counts_arr:
    .zero 24
digitbuf:
    .zero 32
outbuf:
    .zero 64
