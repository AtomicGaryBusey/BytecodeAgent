.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BANG, 0x21
.equ INPUT_CAP, 65536
.equ OUT_CAP, 65536
.equ C_GT, 0x3E
.equ C_LT, 0x3C
.equ C_PLUS, 0x2B
.equ C_MINUS, 0x2D
.equ C_DOT, 0x2E
.equ C_COMMA, 0x2C
.equ C_OPEN, 0x5B
.equ C_CLOSE, 0x5D

.text
.balign 16
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, INPUT_CAP
    sub edx, ebx
    lea rsi, [rip + input_buf]
    add rsi, rbx
    xor edi, edi
    mov eax, SYS_READ
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    xor ecx, ecx
find_loop:
    cmp rcx, rbx
    jae no_bang
    movzx eax, byte ptr [rsi + rcx]
    cmp al, BANG
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r12, rcx
    lea r14, [rsi + rcx + 1]
    mov rax, rbx
    sub rax, rcx
    dec rax
    mov r15, rax
    jmp validate
no_bang:
    mov r12, rbx
    lea r14, [rsi + rbx]
    xor r15d, r15d
validate:
    mov rbp, rsi
    xor r8d, r8d
    xor r9d, r9d
val_loop:
    cmp r8, r12
    jae val_end
    movzx eax, byte ptr [rbp + r8]
    cmp al, C_OPEN
    je val_open
    cmp al, C_CLOSE
    je val_close
val_next:
    inc r8
    jmp val_loop
val_open:
    mov word ptr [bstack + r9*2], r8w
    inc r9
    jmp val_next
val_close:
    test r9, r9
    jz invalid
    dec r9
    movzx ecx, word ptr [bstack + r9*2]
    mov word ptr [match + r8*2], cx
    mov word ptr [match + rcx*2], r8w
    jmp val_next
val_end:
    test r9, r9
    jnz invalid
    jmp run
invalid:
    mov edi, 1
    mov eax, SYS_EXIT
    syscall
run:
    mov r9, r12
    xor r12d, r12d
    lea r8, [rip + tape]
    xor r13d, r13d
    xor ebx, ebx
exec_loop:
    cmp r12, r9
    jae finish
    movzx eax, byte ptr [rbp + r12]
    cmp al, C_GT
    je cmd_gt
    cmp al, C_LT
    je cmd_lt
    cmp al, C_PLUS
    je cmd_plus
    cmp al, C_MINUS
    je cmd_minus
    cmp al, C_DOT
    je cmd_dot
    cmp al, C_COMMA
    je cmd_comma
    cmp al, C_OPEN
    je cmd_open
    cmp al, C_CLOSE
    je cmd_close
    inc r12
    jmp exec_loop
advance:
    inc r12
    jmp exec_loop
cmd_gt:
    inc r13
    jmp advance
cmd_lt:
    dec r13
    jmp advance
cmd_plus:
    inc byte ptr [r8 + r13]
    jmp advance
cmd_minus:
    dec byte ptr [r8 + r13]
    jmp advance
cmd_dot:
    movzx eax, byte ptr [r8 + r13]
    mov byte ptr [out_buf + rbx], al
    inc rbx
    cmp rbx, OUT_CAP
    jb advance
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, ebx
    mov eax, SYS_WRITE
    syscall
    xor ebx, ebx
    jmp advance
cmd_comma:
    test r15, r15
    jz comma_zero
    movzx eax, byte ptr [r14]
    inc r14
    dec r15
    mov byte ptr [r8 + r13], al
    jmp advance
comma_zero:
    mov byte ptr [r8 + r13], 0
    jmp advance
cmd_open:
    cmp byte ptr [r8 + r13], 0
    jne advance
    movzx ecx, word ptr [match + r12*2]
    lea r12, [rcx + 1]
    jmp exec_loop
cmd_close:
    cmp byte ptr [r8 + r13], 0
    je advance
    movzx ecx, word ptr [match + r12*2]
    lea r12, [rcx + 1]
    jmp exec_loop
finish:
    test rbx, rbx
    jz exit_ok
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov edx, ebx
    mov eax, SYS_WRITE
    syscall
exit_ok:
    xor edi, edi
    mov eax, SYS_EXIT
    syscall

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
out_buf:
    .zero 65536
.balign 16
tape:
    .zero 30000
.balign 16
match:
    .zero 131072
.balign 16
bstack:
    .zero 131072
