.equ BUFSIZE, 65536

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, BUFSIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, BUFSIZE
    jb read_loop
read_done:
    lea rbx, [rip + input_buffer]
    lea r14, [rip + input_buffer]
    add r14, r12
    lea r13, [rip + output_buffer]
line_loop:
    cmp rbx, r14
    jae all_done
    mov rsi, rbx
scan_loop:
    cmp rsi, r14
    jae scan_done
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je scan_done
    inc rsi
    jmp scan_loop
scan_done:
    mov r15, rsi
    mov rdi, r15
    dec rdi
copy_loop:
    cmp rdi, rbx
    jb copy_done
    movzx eax, byte ptr [rdi]
    mov byte ptr [r13], al
    inc r13
    dec rdi
    jmp copy_loop
copy_done:
    mov byte ptr [r13], 10
    inc r13
    lea rbx, [r15 + 1]
    jmp line_loop
all_done:
    lea rsi, [rip + output_buffer]
    mov rdx, r13
    sub rdx, rsi
    mov rax, 1
    mov edi, 1
    syscall
    mov rax, 231
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .space BUFSIZE
.balign 16
output_buffer:
    .space BUFSIZE + 1
