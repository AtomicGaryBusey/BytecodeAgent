.text
	.globl _start
_start:
	stp x19, x20, [sp, #-16]!
	stp x21, x22, [sp, #-16]!
	stp x23, x24, [sp, #-16]!

	// read all of stdin into inbuf
	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	mov x20, x19
read_loop:
	adrp x0, inbuf
	add x0, x0, :lo12:inbuf
	mov x1, #65536
	add x0, x0, x1
	subs x2, x0, x20
	b.eq read_done
	mov x0, #0
	mov x1, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	b.le read_done
	add x20, x20, x0
	b read_loop
read_done:

	// process one line per iteration: bytes [x24, x1)
line_loop:
	cmp x19, x20
	b.ge all_done
	mov x24, x19
	mov x0, x19
scan_nl:
	cmp x0, x20
	b.ge eof_line
	ldrb w1, [x0], #1
	cmp w1, #10
	b.ne scan_nl
	sub x1, x0, #1
	mov x19, x0
	b process_line
eof_line:
	mov x1, x20
	mov x19, x20
process_line:
	mov x22, #0
	adrp x21, stk
	add x21, x21, :lo12:stk
	mov x0, x24
tok_loop:
	cmp x0, x1
	b.ge tok_done
	ldrb w2, [x0]
	cmp w2, #32
	b.eq skip_sep
	cmp w2, #9
	b.eq skip_sep
	b have_tok
skip_sep:
	add x0, x0, #1
	b tok_loop
have_tok:
	mov x3, x0
tok_scan:
	cmp x0, x1
	b.ge tok_end
	ldrb w2, [x0]
	cmp w2, #32
	b.eq tok_end
	cmp w2, #9
	b.eq tok_end
	add x0, x0, #1
	b tok_scan
tok_end:
	sub x4, x0, x3
	ldrb w5, [x3]
	cmp x4, #1
	b.ne lit_path
	// single-character token
	cmp w5, #43
	b.eq op_add
	cmp w5, #45
	b.eq op_sub
	cmp w5, #42
	b.eq op_mul
	cmp w5, #47
	b.eq op_div
	cmp w5, #37
	b.eq op_mod
	sub w6, w5, #48
	cmp w6, #9
	b.ls lit_single
	b invalid
lit_single:
	sub x7, x5, #48
	b lit_push
lit_path:
	sub w6, w5, #48
	cmp w6, #9
	b.ls lit_pos
	cmp w5, #45
	b.ne invalid
	ldrb w6, [x3, #1]
	sub w6, w6, #48
	cmp w6, #9
	b.hi invalid
	cmp x4, #19
	b.hi invalid
	mov x6, #1
	add x3, x3, #1
	b lit_parse
lit_pos:
	cmp x4, #18
	b.hi invalid
	mov x6, #0
lit_parse:
	mov x7, #0
lit_loop:
	cmp x3, x0
	b.ge lit_done
	ldrb w2, [x3], #1
	sub w2, w2, #48
	cmp w2, #9
	b.hi invalid
	mov x9, #10
	mul x7, x7, x9
	add x7, x7, x2
	b lit_loop
lit_done:
	cbz x6, lit_push
	neg x7, x7
lit_push:
	str x7, [x21], #8
	b tok_loop
invalid:
	mov x22, #1
	b tok_done
op_add:
	mov x23, #43
	b op_do
op_sub:
	mov x23, #45
	b op_do
op_mul:
	mov x23, #42
	b op_do
op_div:
	mov x23, #47
	b op_do
op_mod:
	mov x23, #37
op_do:
	adrp x9, stk
	add x9, x9, :lo12:stk
	sub x10, x21, x9
	cmp x10, #16
	b.lt err_op
	ldr x11, [x21, #-8]!
	ldr x12, [x21, #-8]!
	cmp x23, #43
	b.eq do_add
	cmp x23, #45
	b.eq do_sub
	cmp x23, #42
	b.eq do_mul
	cmp x23, #47
	b.eq do_div
	b do_mod
do_add:
	add x13, x12, x11
	b op_push
do_sub:
	sub x13, x12, x11
	b op_push
do_mul:
	mul x13, x12, x11
	b op_push
do_div:
	cmp x11, #0
	b.eq err_op
	movz x9, #0x8000, lsl #48
	cmp x12, x9
	b.ne do_sdiv
	cmn x11, #1
	b.eq err_op
do_sdiv:
	sdiv x13, x12, x11
	b op_push
do_mod:
	cmp x11, #0
	b.eq err_op
	movz x9, #0x8000, lsl #48
	cmp x12, x9
	b.ne do_smod
	cmn x11, #1
	b.eq err_op
do_smod:
	sdiv x14, x12, x11
	msub x13, x14, x11, x12
op_push:
	str x13, [x21], #8
	b tok_loop
err_op:
	mov x22, #1
	b tok_done
tok_done:
	cbnz x22, out_error
	adrp x9, stk
	add x9, x9, :lo12:stk
	sub x10, x21, x9
	cmp x10, #8
	b.ne out_error
	ldr x13, [x9]
	// convert x13 to decimal, backwards into outbuf
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	add x1, x1, #64
	mov x2, #10
	mov x5, x13
	cmp x13, #0
	b.ge num_digits
	neg x5, x13
	sub x1, x1, #1
	mov w6, #45
	strb w6, [x1]
num_digits:
	udiv x6, x5, x2
	msub x7, x6, x2, x5
	add w7, w7, #48
	sub x1, x1, #1
	strb w7, [x1]
	mov x5, x6
	cbnz x5, num_digits
	sub x1, x1, #1
	mov w6, #10
	strb w6, [x1]
	adrp x0, outbuf
	add x0, x0, :lo12:outbuf
	add x0, x0, #64
	sub x2, x0, x1
	mov x0, #1
num_write:
	cmp x2, #0
	b.eq next_line
	mov x8, #64
	svc #0
	cmp x0, #0
	b.le next_line
	add x1, x1, x0
	sub x2, x2, x0
	b num_write
out_error:
	adrp x1, errmsg
	add x1, x1, :lo12:errmsg
	mov x2, #6
	mov x0, #1
err_write:
	cmp x2, #0
	b.eq next_line
	mov x8, #64
	svc #0
	cmp x0, #0
	b.le next_line
	add x1, x1, x0
	sub x2, x2, x0
	b err_write
next_line:
	b line_loop
all_done:
	mov x0, #0
	mov x8, #93
	svc #0

	.section .rodata
errmsg:
	.ascii "error\n"

	.bss
	.balign 8
inbuf:
	.zero 65536
	.balign 8
outbuf:
	.zero 64
	.balign 8
stk:
	.zero 524288
