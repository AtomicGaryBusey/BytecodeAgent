.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 0x8000

.text
.balign 4
.globl _start
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
read_loop:
    mov x7, #MAX_INPUT
    cmp x20, x7
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, x7
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.eq read_done
    b.mi fail
    add x20, x20, x0
    b read_loop
read_done:
    adrp x21, lptr
    add x21, x21, :lo12:lptr
    adrp x22, llen
    add x22, x22, :lo12:llen
    mov x23, #0
    add x24, x19, x20
    mov x5, x19
    mov x6, x19
split_loop:
    cmp x6, x24
    b.hs split_done
    ldrb w7, [x6]
    cmp w7, #0x0A
    b.ne split_next
    str x5, [x21, x23, lsl #3]
    sub x8, x6, x5
    str x8, [x22, x23, lsl #3]
    add x23, x23, #1
    add x5, x6, #1
split_next:
    add x6, x6, #1
    b split_loop
split_done:
    cmp x5, x24
    b.hs sort_init
    str x5, [x21, x23, lsl #3]
    sub x8, x24, x5
    str x8, [x22, x23, lsl #3]
    add x23, x23, #1
sort_init:
    mov x9, #1
sort_outer:
    cmp x9, x23
    b.hs output_init
    ldr x10, [x21, x9, lsl #3]
    ldr x11, [x22, x9, lsl #3]
    sub x12, x9, #1
sort_inner:
    cmp x12, #0
    b.lt sort_insert
    ldr x0, [x21, x12, lsl #3]
    ldr x1, [x22, x12, lsl #3]
    mov x2, x10
    mov x3, x11
    bl cmp_lines
    cmp x0, #0
    b.le sort_insert
    add x13, x12, #1
    ldr x14, [x21, x12, lsl #3]
    str x14, [x21, x13, lsl #3]
    ldr x14, [x22, x12, lsl #3]
    str x14, [x22, x13, lsl #3]
    sub x12, x12, #1
    b sort_inner
sort_insert:
    add x13, x12, #1
    str x10, [x21, x13, lsl #3]
    str x11, [x22, x13, lsl #3]
    add x9, x9, #1
    b sort_outer
output_init:
    mov x9, #0
output_loop:
    cmp x9, x23
    b.hs exit0
    ldr x1, [x21, x9, lsl #3]
    ldr x2, [x22, x9, lsl #3]
    mov x0, #1
    bl write_all
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    bl write_all
    add x9, x9, #1
    b output_loop
exit0:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
fail:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, ls
    mov x5, #0
cmp_loop:
    cmp x5, x4
    b.hs cmp_min_done
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.ne cmp_diff
    add x5, x5, #1
    b cmp_loop
cmp_diff:
    b.hi cmp_a_greater
    mov x0, #-1
    ret
cmp_a_greater:
    mov x0, #1
    ret
cmp_min_done:
    cmp x1, x3
    b.eq cmp_equal
    b.lo cmp_a_shorter
    mov x0, #1
    ret
cmp_a_shorter:
    mov x0, #-1
    ret
cmp_equal:
    mov x0, #0
    ret

write_all:
    cmp x2, #0
    b.eq write_done
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le fail
    add x1, x1, x0
    sub x2, x2, x0
    b write_all
write_done:
    ret

.section .rodata
.balign 1
nl:
    .byte 0x0A

.bss
.balign 8
buf:
    .zero MAX_INPUT
.balign 8
lptr:
    .zero 8192
.balign 8
llen:
    .zero 8192
