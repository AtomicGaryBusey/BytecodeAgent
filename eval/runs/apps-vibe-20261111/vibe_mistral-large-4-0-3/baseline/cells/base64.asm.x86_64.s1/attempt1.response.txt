# base64 - RFC 4648 encoder, static x86-64 Linux, no libc
.globl _start

.equ INBUF_SIZE, 49152
.equ OUTBUF_SIZE, 66560
.equ LINE_LEN, 76

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero INBUF_SIZE
outbuf:
    .zero OUTBUF_SIZE

.text
_start:
    # read all of stdin into inbuf
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, INBUF_SIZE
read_loop:
    xor eax, eax                    # rax = 0 = read
    syscall
    test rax, rax
    jz read_done                    # 0 = end of input
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    mov r8, INBUF_SIZE
    sub r8, rdx                     # r8 = total input bytes

    lea rsi, [rip + inbuf]          # src
    lea rdi, [rip + outbuf]         # dst
    lea r11, [rip + alphabet]
    xor r9d, r9d                    # column counter

encode_loop:
    test r8, r8
    jz finish
    cmp r8, 3
    jb partial
    # full group: 3 bytes -> 4 chars
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx ecx, byte ptr [rsi + 2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ecx                     # eax = 24-bit group
    mov ebx, eax
    mov edx, ebx
    shr edx, 18
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    mov edx, ebx
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    mov edx, ebx
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    mov edx, ebx
    and edx, 63
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    add rsi, 3
    sub r8, 3
    jmp encode_loop

partial:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r8, 2
    jb partial_one
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
partial_one:
    mov ebx, eax
    mov edx, ebx
    shr edx, 18
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    mov edx, ebx
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    cmp r8, 2
    jb partial_last_one
    mov edx, ebx
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [r11 + rdx]
    mov al, dl
    call emit
    mov al, 61                      # '='
    call emit
    jmp finish
partial_last_one:
    mov al, 61                      # '='
    call emit
    mov al, 61                      # '='
    call emit

finish:
    test r9, r9
    jz write_out
    mov byte ptr [rdi], 10          # newline after last line
    inc rdi
write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                    # rdx = output length
    mov edi, 1
write_loop:
    test rdx, rdx
    jz exit_prog
    mov eax, 1                      # rax = 1 = write
    syscall
    test rax, rax
    jz exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    xor edi, edi
    mov eax, 60                     # rax = 60 = exit
    syscall

# emit: append al to output, wrap line at LINE_LEN columns
# updates rdi, r9; preserves all other registers
emit:
    mov byte ptr [rdi], al
    inc rdi
    inc r9
    cmp r9, LINE_LEN
    jb emit_ret
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
emit_ret:
    ret
