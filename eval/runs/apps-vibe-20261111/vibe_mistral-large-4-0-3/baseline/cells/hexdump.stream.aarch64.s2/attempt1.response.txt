.globl _start

.text
_start:
    // read all of stdin into input_buf
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf      // x19 = current fill pointer
    mov  x20, #0                        // x20 = total bytes read
read_loop:
    mov  x0, #0                         // fd 0
    mov  x1, x19
    mov  x2, #65536
    sub  x2, x2, x20                    // remaining capacity
    mov  x8, #63                        // read
    svc  #0
    cmp  x0, #0
    b.eq read_done                      // 0 = end of input
    b.mi read_fail                      // negative = error
    add  x20, x20, x0
    add  x19, x19, x0
    b    read_loop
read_fail:
    mov  x0, #1
    mov  x8, #93                        // exit(1)
    svc  #0
read_done:
    mov  x21, #0                        // x21 = row offset
row_loop:
    cmp  x21, x20
    b.hs dump_done
    mov  x12, x20
    sub  x12, x12, x21                  // bytes remaining
    cmp  x12, #16
    b.le 1f
    mov  x12, #16
1:                                      // x12 = n (bytes in this row)
    adrp x13, input_buf
    add  x13, x13, :lo12:input_buf
    add  x13, x13, x21                  // x13 = &input_buf[offset]
    adrp x22, row_buf
    add  x22, x22, :lo12:row_buf        // x22 = output pointer
    // 8 hex digits of the offset
    mov  w9, w21
    mov  x10, #8
2:
    and  w11, w9, #0xf
    cmp  w11, #10
    b.lo 3f
    add  w11, w11, #87                  // 'a' - 10
    b    4f
3:
    add  w11, w11, #48                  // '0'
4:
    strb w11, [x22], #1
    lsr  w9, w9, #4
    subs x10, x10, #1
    b.ne 2b
    // two spaces
    mov  w11, #32
    strb w11, [x22], #1
    strb w11, [x22], #1
    // 16 hex slots
    mov  x14, #0
5:
    cmp  x14, #16
    b.hs 9f
    cmp  x14, #8
    b.ne 24f
    mov  w16, #32                       // extra space after slot 7
    strb w16, [x22], #1
24:
    cmp  x14, x12
    b.hs 7f                             // past end of row: pad
    ldrb w15, [x13, x14]
    lsr  w16, w15, #4
    cmp  w16, #10
    b.lo 6f
    add  w16, w16, #87
    b    8f
6:
    add  w16, w16, #48
8:
    strb w16, [x22], #1
    and  w16, w15, #0xf
    cmp  w16, #10
    b.lo 10f
    add  w16, w16, #87
    b    11f
10:
    add  w16, w16, #48
11:
    strb w16, [x22], #1
    mov  w16, #32
    strb w16, [x22], #1
    b    25f
7:
    mov  w16, #32
    strb w16, [x22], #1
    strb w16, [x22], #1
    strb w16, [x22], #1
25:
    add  x14, x14, #1
    b    5b
9:
    // ASCII column
    mov  w16, #32
    strb w16, [x22], #1
    mov  w16, #124                      // '|'
    strb w16, [x22], #1
    mov  x14, #0
13:
    cmp  x14, x12
    b.hs 16f
    ldrb w15, [x13, x14]
    cmp  w15, #0x20
    b.lo 14f
    cmp  w15, #0x7e
    b.hi 14f
    b    15f
14:
    mov  w15, #46                       // '.'
15:
    strb w15, [x22], #1
    add  x14, x14, #1
    b    13b
16:
    mov  w16, #124                      // '|'
    strb w16, [x22], #1
    mov  w16, #10                       // '\n'
    strb w16, [x22], #1
    // write the row to stdout (handle partial writes)
    adrp x1, row_buf
    add  x1, x1, :lo12:row_buf
    sub  x2, x22, x1
17:
    cmp  x2, #0
    b.le 18f
    mov  x0, #1
    mov  x8, #64                        // write
    svc  #0
    cmp  x0, #0
    b.le 18f
    add  x1, x1, x0
    sub  x2, x2, x0
    b    17b
18:
    add  x21, x21, #16
    b    row_loop
dump_done:
    // final line: total length as 8 hex digits + newline
    adrp x22, row_buf
    add  x22, x22, :lo12:row_buf
    mov  w9, w20
    mov  x10, #8
19:
    and  w11, w9, #0xf
    cmp  w11, #10
    b.lo 20f
    add  w11, w11, #87
    b    21f
20:
    add  w11, w11, #48
21:
    strb w11, [x22], #1
    lsr  w9, w9, #4
    subs x10, x10, #1
    b.ne 19b
    mov  w11, #10
    strb w11, [x22], #1
    adrp x1, row_buf
    add  x1, x1, :lo12:row_buf
    sub  x2, x22, x1
22:
    cmp  x2, #0
    b.le 23f
    mov  x0, #1
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le 23f
    add  x1, x1, x0
    sub  x2, x2, x0
    b    22b
23:
    mov  x0, #0
    mov  x8, #93                        // exit(0)
    svc  #0

.bss
.balign 8
input_buf:
    .space 65536
row_buf:
    .space 96
