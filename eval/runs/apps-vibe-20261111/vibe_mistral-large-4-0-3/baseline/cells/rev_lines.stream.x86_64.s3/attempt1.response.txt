.globl _start
_start:
    lea rbx, [rip + buf]
    xor r12, r12
read_loop:
    mov rdx, 65537
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    cmp r12, 65537
    jae set_end
    test r12, r12
    jz set_end
    movzx eax, byte ptr [rbx + r12 - 1]
    cmp al, 10
    je set_end
    mov byte ptr [rbx + r12], 10
    inc r12
set_end:
    lea r13, [rbx + r12]
    mov rsi, rbx
line_loop:
    cmp rsi, r13
    jae all_done
    mov rdi, rsi
scan_loop:
    cmp rdi, r13
    jae line_end
    movzx eax, byte ptr [rdi]
    cmp al, 10
    je line_end
    inc rdi
    jmp scan_loop
line_end:
    mov r10, rdi
    sub r10, rsi
    shr r10, 1
    mov r8, rsi
    lea r9, [rdi - 1]
rev_loop:
    test r10, r10
    jz rev_done
    mov al, byte ptr [r8]
    mov cl, byte ptr [r9]
    mov byte ptr [r8], cl
    mov byte ptr [r9], al
    inc r8
    dec r9
    dec r10
    jmp rev_loop
rev_done:
    lea rdx, [rdi + 1]
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    jmp line_loop
all_done:
    mov eax, 60
    xor edi, edi
    syscall
