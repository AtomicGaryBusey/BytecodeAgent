.equ INPUT_MAX, 65536
.equ OUT_MAX, 65538

.globl _start
.text
_start:
    lea rbx, [rip + input_buf]
    mov r12, INPUT_MAX
read_loop:
    test r12, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov rdx, r12
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    sub r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    mov r8, rsi
    lea rdi, [rip + out_buf]
scan_loop:
    cmp rsi, rbx
    jae scan_done
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je line_found
    inc rsi
    jmp scan_loop
line_found:
    mov r10, rsi
rev_loop:
    cmp r10, r8
    jbe line_rev_done
    dec r10
    movzx eax, byte ptr [r10]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev_loop
line_rev_done:
    mov byte ptr [rdi], 10
    inc rdi
    inc rsi
    mov r8, rsi
    jmp scan_loop
scan_done:
    cmp r8, rbx
    jae write_out
    mov r10, rbx
final_rev_loop:
    cmp r10, r8
    jbe final_rev_done
    dec r10
    movzx eax, byte ptr [r10]
    mov byte ptr [rdi], al
    inc rdi
    jmp final_rev_loop
final_rev_done:
    mov byte ptr [rdi], 10
    inc rdi
write_out:
    mov rdx, rdi
    lea rsi, [rip + out_buf]
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_prog
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buf:
    .zero INPUT_MAX
out_buf:
    .zero OUT_MAX
