.text
.equ INPUT_MAX, 65536
.equ OUTPUT_MAX, 65537
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov edx, INPUT_MAX
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    lea rbx, [rip + input_buf]
    add rbx, r12
    lea rdi, [rip + output_buf]
    mov r8, rsi
scan_loop:
    cmp r8, rbx
    jae scan_done
    movzx eax, byte ptr [r8]
    cmp al, 0x0A
    je found_nl
    inc r8
    jmp scan_loop
found_nl:
    mov r9, r8
rev_loop:
    cmp rsi, r9
    jae rev_done
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev_loop
rev_done:
    mov al, 0x0A
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    mov rsi, r8
    jmp scan_loop
scan_done:
    cmp rsi, rbx
    jae build_done
    mov r9, rbx
fin_loop:
    cmp rsi, r9
    jae fin_done
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    jmp fin_loop
fin_done:
    mov al, 0x0A
    mov byte ptr [rdi], al
    inc rdi
build_done:
    lea rsi, [rip + output_buf]
    mov r13, rdi
    lea rdx, [rip + output_buf]
    sub r13, rdx
write_loop:
    test r13, r13
    jz write_done
    mov rdx, r13
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jz write_done
    add rsi, rax
    sub r13, rax
    jmp write_loop
write_done:
    xor edi, edi
    mov eax, 60
    syscall

.bss
.balign 32
input_buf:
    .zero INPUT_MAX
output_buf:
    .zero OUTPUT_MAX
