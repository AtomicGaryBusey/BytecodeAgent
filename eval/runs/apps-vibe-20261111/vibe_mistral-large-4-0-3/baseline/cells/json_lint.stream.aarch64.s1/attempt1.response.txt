.equ BUFSIZE, 65536

.text
.globl _start

// register conventions: x19 = input buffer, x20 = input length, x21 = parse position
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
read_loop:
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cmp x2, #0
    b.eq read_done
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    bl skip_ws
    cmp x21, x20
    b.hs err_at_p
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs print_ok
    b err_at_p

// write_all: x0 = fd, x1 = buf, x2 = count
write_all:
    cmp x2, #0
    b.eq write_all_ret
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_all_ret
    add x1, x1, x0
    sub x2, x2, x0
    b write_all
write_all_ret:
    ret

print_ok:
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    bl write_all
    mov x0, #0
    mov x8, #93
    svc #0

// err_at_p: x21 = error offset; print "error at N\n", exit 1
err_at_p:
    adrp x1, errbuf
    add x1, x1, :lo12:errbuf
    adrp x0, msg_err
    add x0, x0, :lo12:msg_err
    mov x3, #9
err_copy:
    ldrb w4, [x0], #1
    strb w4, [x1], #1
    subs x3, x3, #1
    b.ne err_copy
    mov x5, x21
    add x6, x1, #16
    mov x7, #10
err_digit:
    udiv x8, x5, x7
    msub x9, x8, x7, x5
    add w9, w9, #0x30
    sub x6, x6, #1
    strb w9, [x6]
    mov x5, x8
    cbnz x5, err_digit
    add x10, x1, #16
    sub x10, x10, x6
err_copy2:
    ldrb w4, [x6], #1
    strb w4, [x1], #1
    subs x10, x10, #1
    b.ne err_copy2
    mov w4, #0x0A
    strb w4, [x1], #1
    mov x2, x1
    adrp x1, errbuf
    add x1, x1, :lo12:errbuf
    sub x2, x2, x1
    mov x0, #1
    bl write_all
    mov x0, #1
    mov x8, #93
    svc #0

// skip whitespace: 0x20 0x09 0x0A 0x0D
skip_ws:
    cmp x21, x20
    b.hs skip_ws_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq skip_ws_next
    cmp w0, #0x09
    b.eq skip_ws_next
    cmp w0, #0x0A
    b.eq skip_ws_next
    cmp w0, #0x0D
    b.eq skip_ws_next
    b skip_ws_done
skip_ws_next:
    add x21, x21, #1
    b skip_ws
skip_ws_done:
    ret

// parse one value at x21
parse_value:
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x7B
    b.eq parse_object
    cmp w0, #0x5B
    b.eq parse_array
    cmp w0, #0x22
    b.eq parse_string
    cmp w0, #0x2D
    b.eq parse_number
    cmp w0, #0x74
    b.eq lit_true
    cmp w0, #0x66
    b.eq lit_false
    cmp w0, #0x6E
    b.eq lit_null
    cmp w0, #0x30
    b.lo err_at_p
    cmp w0, #0x39
    b.hi err_at_p
    b parse_number

lit_true:
    adrp x0, str_true
    add x0, x0, :lo12:str_true
    mov x3, #4
    b match_lit
lit_false:
    adrp x0, str_false
    add x0, x0, :lo12:str_false
    mov x3, #5
    b match_lit
lit_null:
    adrp x0, str_null
    add x0, x0, :lo12:str_null
    mov x3, #4
    b match_lit

// match_lit: x0 = literal, x3 = its length
match_lit:
    cmp x21, x20
    b.hs err_at_p
    ldrb w1, [x19, x21]
    ldrb w2, [x0], #1
    cmp w1, w2
    b.ne err_at_p
    add x21, x21, #1
    subs x3, x3, #1
    b.ne match_lit
    ret

// parse_string: buf[x21] == '"'
parse_string:
    add x21, x21, #1
str_loop:
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq str_end
    cmp w0, #0x5C
    b.eq str_esc
    cmp w0, #0x20
    b.lo err_at_p
    add x21, x21, #1
    b str_loop
str_esc:
    add x21, x21, #1
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq str_esc_done
    cmp w0, #0x5C
    b.eq str_esc_done
    cmp w0, #0x2F
    b.eq str_esc_done
    cmp w0, #0x62
    b.eq str_esc_done
    cmp w0, #0x66
    b.eq str_esc_done
    cmp w0, #0x6E
    b.eq str_esc_done
    cmp w0, #0x72
    b.eq str_esc_done
    cmp w0, #0x74
    b.eq str_esc_done
    cmp w0, #0x75
    b.ne err_at_p
    add x21, x21, #1
    mov w2, #4
    b str_hex
str_hex:
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls str_hex_ok
    sub w1, w0, #0x61
    cmp w1, #5
    b.ls str_hex_ok
    sub w1, w0, #0x41
    cmp w1, #5
    b.hi err_at_p
str_hex_ok:
    add x21, x21, #1
    subs w2, w2, #1
    b.ne str_hex
    b str_loop
str_esc_done:
    add x21, x21, #1
    b str_loop
str_end:
    add x21, x21, #1
    ret

// parse_number: buf[x21] is '-' or a digit
parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #0x2D
    b.ne num_int
    add x21, x21, #1
num_int:
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.eq num_zero
    sub w1, w0, #0x31
    cmp w1, #8
    b.hi err_at_p
    add x21, x21, #1
num_int_digits:
    cmp x21, x20
    b.hs num_frac
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi num_frac
    add x21, x21, #1
    b num_int_digits
num_zero:
    add x21, x21, #1
num_frac:
    cmp x21, x20
    b.hs num_exp
    ldrb w0, [x19, x21]
    cmp w0, #0x2E
    b.ne num_exp
    add x21, x21, #1
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi err_at_p
num_frac_digits:
    cmp x21, x20
    b.hs num_exp
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi num_exp
    add x21, x21, #1
    b num_frac_digits
num_exp:
    cmp x21, x20
    b.hs num_done
    ldrb w0, [x19, x21]
    cmp w0, #0x65
    b.eq num_exp_body
    cmp w0, #0x45
    b.ne num_done
num_exp_body:
    add x21, x21, #1
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x2B
    b.eq num_exp_sign
    cmp w0, #0x2D
    b.ne num_exp_digits
num_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs err_at_p
num_exp_digits:
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi err_at_p
num_exp_digits_loop:
    cmp x21, x20
    b.hs num_done
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi num_done
    add x21, x21, #1
    b num_exp_digits_loop
num_done:
    ret

// parse_object: buf[x21] == '{'
parse_object:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x7D
    b.eq obj_empty
obj_loop:
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne err_at_p
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x3A
    b.ne err_at_p
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.eq obj_comma
    cmp w0, #0x7D
    b.eq obj_end
    b err_at_p
obj_comma:
    add x21, x21, #1
    bl skip_ws
    b obj_loop
obj_end:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret
obj_empty:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

// parse_array: buf[x21] == '['
parse_array:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x5D
    b.eq arr_empty
arr_loop:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs err_at_p
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.eq arr_comma
    cmp w0, #0x5D
    b.eq arr_end
    b err_at_p
arr_comma:
    add x21, x21, #1
    bl skip_ws
    b arr_loop
arr_end:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret
arr_empty:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
str_true:
    .ascii "true"
str_false:
    .ascii "false"
str_null:
    .ascii "null"

.bss
.balign 16
buf:
    .zero 65552
errbuf:
    .zero 64
