.text
    .globl _start

_start:
    adrp x21, inbuf
    add  x21, x21, :lo12:inbuf
    adrp x27, hextab
    add  x27, x27, :lo12:hextab
    mov  x19, #0

read_loop:
    mov  x0, #0
    mov  x1, x21
    add  x1, x1, x19
    mov  x2, #65536
    sub  x2, x2, x19
    cmp  x2, #0
    b.eq read_done
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x19, x19, x0
    b    read_loop

read_done:
    mov  x20, #0

row_loop:
    cmp  x20, x19
    b.hs rows_done
    sub  x23, x19, x20
    cmp  x23, #16
    b.lo 1f
    mov  x23, #16
1:
    adrp x22, outbuf
    add  x22, x22, :lo12:outbuf

    lsr  w26, w20, #24
    bl   hex2
    lsr  w26, w20, #16
    bl   hex2
    lsr  w26, w20, #8
    bl   hex2
    mov  w26, w20
    bl   hex2

    mov  w28, #0x20
    strb w28, [x22], #1
    strb w28, [x22], #1

    mov  x24, #0
slot_loop:
    cmp  x24, #16
    b.hs slots_done
    cmp  x24, x23
    b.hs slot_pad
    add  x25, x21, x20
    add  x25, x25, x24
    ldrb w26, [x25]
    bl   hex2
    mov  w28, #0x20
    strb w28, [x22], #1
    b    slot_next
slot_pad:
    mov  w28, #0x20
    strb w28, [x22], #1
    strb w28, [x22], #1
    strb w28, [x22], #1
slot_next:
    add  x24, x24, #1
    cmp  x24, #8
    b.ne slot_loop
    mov  w28, #0x20
    strb w28, [x22], #1
    b    slot_loop
slots_done:

    mov  w28, #0x20
    strb w28, [x22], #1
    mov  w28, #0x7c
    strb w28, [x22], #1

    mov  x24, #0
char_loop:
    cmp  x24, x23
    b.hs chars_done
    add  x25, x21, x20
    add  x25, x25, x24
    ldrb w26, [x25]
    cmp  w26, #0x20
    b.lo char_dot
    cmp  w26, #0x7e
    b.hi char_dot
    b    char_put
char_dot:
    mov  w26, #0x2e
char_put:
    strb w26, [x22], #1
    add  x24, x24, #1
    b    char_loop
chars_done:
    mov  w28, #0x7c
    strb w28, [x22], #1
    mov  w28, #0x0a
    strb w28, [x22], #1

    adrp x1, outbuf
    add  x1, x1, :lo12:outbuf
    sub  x2, x22, x1
    bl   emit

    add  x20, x20, x23
    b    row_loop

rows_done:
    adrp x22, outbuf
    add  x22, x22, :lo12:outbuf
    lsr  w26, w19, #24
    bl   hex2
    lsr  w26, w19, #16
    bl   hex2
    lsr  w26, w19, #8
    bl   hex2
    mov  w26, w19
    bl   hex2
    mov  w28, #0x0a
    strb w28, [x22], #1

    adrp x1, outbuf
    add  x1, x1, :lo12:outbuf
    sub  x2, x22, x1
    bl   emit

    mov  x0, #0
    mov  x8, #93
    svc  #0

hex2:
    lsr  w28, w26, #4
    and  w28, w28, #0xf
    ldrb w28, [x27, x28]
    strb w28, [x22], #1
    and  w28, w26, #0xf
    ldrb w28, [x27, x28]
    strb w28, [x22], #1
    ret

emit:
    cmp  x2, #0
    b.eq emit_done
    mov  x0, #1
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le emit_done
    add  x1, x1, x0
    sub  x2, x2, x0
    b    emit
emit_done:
    ret

    .section .rodata
    .balign 16
hextab:
    .ascii "0123456789abcdef"

    .bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 128
