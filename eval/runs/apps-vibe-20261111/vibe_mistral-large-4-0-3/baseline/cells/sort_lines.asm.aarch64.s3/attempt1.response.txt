.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #MAX_INPUT
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    cmp x20, #MAX_INPUT
    b.ne read_loop
read_done:
    mov x21, #0
    mov x22, #0
    mov x23, #0
    adrp x24, line_starts
    add x24, x24, :lo12:line_starts
    adrp x25, line_lens
    add x25, x25, :lo12:line_lens
    cbz x20, parse_done
scan_loop:
    cmp x22, x20
    b.hs scan_end
    ldrb w26, [x19, x22]
    cmp w26, #10
    b.ne next_byte
    add x27, x19, x23
    str x27, [x24, x21, lsl #3]
    sub x27, x22, x23
    str x27, [x25, x21, lsl #3]
    add x21, x21, #1
    add x23, x22, #1
next_byte:
    add x22, x22, #1
    b scan_loop
scan_end:
    cmp x23, x20
    b.hs parse_done
    add x27, x19, x23
    str x27, [x24, x21, lsl #3]
    sub x27, x20, x23
    str x27, [x25, x21, lsl #3]
    add x21, x21, #1
parse_done:
    mov x19, x21
    mov x20, #1
sort_outer:
    cmp x20, x19
    b.hs sort_done
    ldr x22, [x24, x20, lsl #3]
    ldr x23, [x25, x20, lsl #3]
    sub x21, x20, #1
sort_inner:
    tst x21, x21
    b.mi sort_insert
    ldr x0, [x24, x21, lsl #3]
    ldr x1, [x25, x21, lsl #3]
    mov x2, x22
    mov x3, x23
    bl compare_lines
    cmp x0, #0
    b.gt sort_shift
    b sort_insert
sort_shift:
    add x26, x21, #1
    ldr x27, [x24, x21, lsl #3]
    str x27, [x24, x26, lsl #3]
    ldr x27, [x25, x21, lsl #3]
    str x27, [x25, x26, lsl #3]
    sub x21, x21, #1
    b sort_inner
sort_insert:
    add x26, x21, #1
    str x22, [x24, x26, lsl #3]
    str x23, [x25, x26, lsl #3]
    add x20, x20, #1
    b sort_outer
sort_done:
    adrp x26, newline
    add x26, x26, :lo12:newline
    mov x20, #0
output_loop:
    cmp x20, x19
    b.hs output_done
    ldr x1, [x24, x20, lsl #3]
    ldr x2, [x25, x20, lsl #3]
    mov x0, #1
    bl write_all
    mov x1, x26
    mov x2, #1
    mov x0, #1
    bl write_all
    add x20, x20, #1
    b output_loop
output_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

compare_lines:
    mov x8, x1
    mov x9, x3
    cmp x8, x9
    csel x10, x8, x9, lo
    mov x11, x0
    mov x12, x2
    mov x13, #0
1:
    cmp x13, x10
    b.hs 2f
    ldrb w14, [x11, x13]
    ldrb w15, [x12, x13]
    cmp w14, w15
    b.lo 3f
    b.hi 4f
    add x13, x13, #1
    b 1b
2:
    cmp x8, x9
    b.lo 3f
    b.hi 4f
    mov x0, #0
    ret
3:
    mov x0, #0
    sub x0, x0, #1
    ret
4:
    mov x0, #1
    ret

write_all:
    mov x3, x0
    mov x4, x1
    mov x5, x2
1:
    cbz x5, 2f
    mov x0, x3
    mov x1, x4
    mov x2, x5
    mov x8, #SYS_WRITE
    svc #0
    add x4, x4, x0
    subs x5, x5, x0
    b.ne 1b
2:
    ret

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 8
input_buf:
    .zero MAX_INPUT
line_starts:
    .zero MAX_LINES * 8
line_lens:
    .zero MAX_LINES * 8
