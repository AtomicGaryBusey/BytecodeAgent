.text
.balign 16
.globl _start
_start:
    lea rbx, [rip + input]
    mov r14, 65536
read_loop:
    test r14, r14
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov rdx, r14
    syscall
    test rax, rax
    jle read_done
    cmp rax, r14
    ja read_done
    add rbx, rax
    sub r14, rax
    jmp read_loop
read_done:
    lea r13, [rip + input]
    mov r14, rbx
line_loop:
    cmp r13, r14
    jae prog_exit
    mov r15, r13
scan_nl:
    cmp r15, r14
    jae nl_found
    movzx eax, byte ptr [r15]
    cmp al, 10
    je nl_found
    inc r15
    jmp scan_nl
nl_found:
    mov rsi, r13
    mov rdx, r15
    call eval_line
    mov r13, r15
    cmp r13, r14
    jae line_loop
    inc r13
    jmp line_loop
prog_exit:
    xor edi, edi
    mov eax, 60
    syscall

eval_line:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbp, rsi
    mov r15, rdx
    xor r12d, r12d
ev_skip:
    cmp rbp, r15
    jae ev_after_tokens
    movzx eax, byte ptr [rbp]
    cmp al, 32
    je ev_skip_adv
    cmp al, 9
    je ev_skip_adv
    jmp ev_token
ev_skip_adv:
    inc rbp
    jmp ev_skip
ev_token:
    mov r13, rbp
    mov r14, rbp
ev_tok_scan:
    cmp r14, r15
    jae ev_tok_done
    movzx eax, byte ptr [r14]
    cmp al, 32
    je ev_tok_done
    cmp al, 9
    je ev_tok_done
    inc r14
    jmp ev_tok_scan
ev_tok_done:
    mov rbp, r14
    mov rax, r14
    sub rax, r13
    cmp rax, 1
    jne ev_literal
    movzx eax, byte ptr [r13]
    cmp al, 43
    je ev_op
    cmp al, 45
    je ev_op
    cmp al, 42
    je ev_op
    cmp al, 47
    je ev_op
    cmp al, 37
    je ev_op
    cmp al, 48
    jb ev_invalid
    cmp al, 57
    ja ev_invalid
    sub eax, 48
    jmp ev_push
ev_literal:
    cmp rax, 19
    ja ev_invalid
    mov r10d, eax
    movzx eax, byte ptr [r13]
    xor r8d, r8d
    cmp al, 45
    jne ev_lit_nosign
    mov r8d, 1
    lea rbx, [r13 + 1]
    jmp ev_parse
ev_lit_nosign:
    cmp r10d, 18
    ja ev_invalid
    lea rbx, [r13]
ev_parse:
    xor eax, eax
ev_parse_loop:
    cmp rbx, r14
    jae ev_parse_done
    movzx ecx, byte ptr [rbx]
    cmp cl, 48
    jb ev_invalid
    cmp cl, 57
    ja ev_invalid
    imul rax, rax, 10
    sub ecx, 48
    add rax, rcx
    inc rbx
    jmp ev_parse_loop
ev_parse_done:
    test r8d, r8d
    jz ev_push
    neg rax
    jmp ev_push
ev_op:
    mov r10d, eax
    cmp r12, 2
    jb ev_invalid
    lea r9, [rip + eval_stack]
    dec r12
    mov rbx, qword ptr [r9 + r12*8]
    dec r12
    mov rax, qword ptr [r9 + r12*8]
    cmp r10b, 43
    jne ev_not_plus
    add rax, rbx
    jmp ev_push
ev_not_plus:
    cmp r10b, 45
    jne ev_not_minus
    sub rax, rbx
    jmp ev_push
ev_not_minus:
    cmp r10b, 42
    jne ev_not_mul
    imul rax, rbx
    jmp ev_push
ev_not_mul:
    cmp r10b, 47
    je ev_divmod
    cmp r10b, 37
    je ev_divmod
    jmp ev_invalid
ev_divmod:
    test rbx, rbx
    jz ev_invalid
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne ev_divmod_ok
    cmp rbx, -1
    je ev_invalid
ev_divmod_ok:
    cqo
    idiv rbx
    cmp r10b, 47
    jne ev_mod
    jmp ev_push
ev_mod:
    mov rax, rdx
    jmp ev_push
ev_push:
    lea r9, [rip + eval_stack]
    mov qword ptr [r9 + r12*8], rax
    inc r12
    jmp ev_skip
ev_after_tokens:
    cmp r12, 1
    jne ev_invalid
    lea r9, [rip + eval_stack]
    mov rax, qword ptr [r9]
    call print_value
    jmp ev_line_done
ev_invalid:
    call print_error
ev_line_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

print_value:
    lea rdi, [rip + numbuf]
    add rdi, 31
    mov byte ptr [rdi], 10
    xor r8d, r8d
    test rax, rax
    jns pv_loop_entry
    mov r8d, 1
    neg rax
pv_loop_entry:
    mov ecx, 10
pv_digit_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pv_digit_loop
    test r8d, r8d
    jz pv_write
    dec rdi
    mov byte ptr [rdi], 45
pv_write:
    lea rsi, [rip + numbuf]
    add rsi, 32
    sub rsi, rdi
    mov rdx, rsi
    mov rsi, rdi
    call write_out
    ret

print_error:
    lea rsi, [rip + errmsg]
    mov edx, 6
    call write_out
    ret

write_out:
wo_loop:
    test rdx, rdx
    jz wo_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wo_done
    cmp rax, rdx
    jae wo_done
    add rsi, rax
    sub rdx, rax
    jmp wo_loop
wo_done:
    ret

.section .rodata
errmsg:
    .byte 101, 114, 114, 111, 114, 10

.bss
.balign 16
input:
    .zero 65536
eval_stack:
    .zero 262144
numbuf:
    .zero 32
