// rev_lines: read all of standard input, then print each line with its
// bytes in reverse order, one line per output line, each ending in 0x0A.
.equ BUFSIZE, 65536

.globl _start
.text
_start:
    // Input buffer on the stack: 64 KiB + 16 bytes of slack for a possible
    // appended newline; sp stays 16-byte aligned (entry sp is aligned).
    sub  sp, sp, #65536
    sub  sp, sp, #16
    mov  x19, sp                  // x19 = buffer base
    mov  x20, #0                  // x20 = total bytes stored so far

    // read(0, buf + x20, BUFSIZE - x20) until EOF (return 0) or buffer full
read_loop:
    mov  x0, #0
    mov  x1, x19
    add  x1, x1, x20
    mov  x2, #BUFSIZE
    sub  x2, x2, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    cmp  x20, #BUFSIZE
    b.lo read_loop

read_done:
    // If the input is non-empty and does not end with 0x0A, append one
    // so that every line in the buffer is newline-terminated.
    cmp  x20, #0
    b.eq line_init
    sub  x10, x20, #1
    ldrb w9, [x19, x10]
    cmp  w9, #0x0A
    b.eq line_init
    mov  w10, #0x0A
    strb w10, [x19, x20]
    add  x20, x20, #1

    // Reverse every line in place; each newline stays at its line's end.
line_init:
    mov  x21, #0                  // x21 = start offset of current line
line_loop:
    cmp  x21, x20
    b.hs write_init               // no more lines
    mov  x22, x21                 // x22 = scan offset
scan_loop:
    ldrb w9, [x19, x22]
    cmp  w9, #0x0A
    b.eq scan_done
    add  x22, x22, #1
    b    scan_loop
scan_done:
    // Reverse the bytes in [x21, x22). x24 is the exclusive right end and
    // never underflows, so an empty line (x21 == x22) exits immediately.
    mov  x23, x21                 // x23 = left index
    mov  x24, x22                 // x24 = right index (exclusive)
rev_loop:
    cmp  x23, x24
    b.hs rev_done
    sub  x25, x24, #1             // x25 = rightmost not-yet-placed index
    ldrb w9,  [x19, x23]
    ldrb w10, [x19, x25]
    strb w10, [x19, x23]
    strb w9,  [x19, x25]
    add  x23, x23, #1
    mov  x24, x25
    b    rev_loop
rev_done:
    add  x21, x22, #1             // next line starts after the newline
    b    line_loop

    // write(1, buf + written, x20 - written) until all output is out
write_init:
    mov  x25, #0                  // x25 = bytes written so far
write_loop:
    cmp  x25, x20
    b.hs exit_prog
    mov  x0, #1
    mov  x1, x19
    add  x1, x1, x25
    sub  x2, x20, x25
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit_prog
    add  x25, x25, x0
    b    write_loop

exit_prog:
    mov  x0, #0
    mov  x8, #93
    svc  #0
1:  b    1b                       // unreachable: exit never returns
