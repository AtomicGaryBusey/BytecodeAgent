// rev_lines: read all of standard input, then print each line with its
// bytes in reverse order, one line per output line, each ending in 0x0A.
.equ BUFSIZE, 65536

.globl _start
.text
_start:
    // x19 = input buffer base, x20 = total bytes stored so far
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0

    // read(0, inbuf + x20, BUFSIZE - x20) until EOF (0) or buffer full
read_loop:
    mov  x0, #0
    mov  x1, x19
    add  x1, x1, x20
    mov  x2, #BUFSIZE
    sub  x2, x2, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    cmp  x20, #BUFSIZE
    b.lo read_loop

read_done:
    // If input is non-empty and does not end with a newline, append one
    // so the final line is also newline-terminated in the buffer.
    cmp  x20, #0
    b.eq line_init
    sub  x10, x20, #1
    ldrb w9, [x19, x10]
    cmp  w9, #0x0A
    b.eq line_init
    mov  w10, #0x0A
    strb w10, [x19, x20]
    add  x20, x20, #1

    // Process every line in place: reverse its bytes, keep the newline.
line_init:
    mov  x21, #0                 // x21 = start offset of current line
line_loop:
    cmp  x21, x20
    b.hs write_init              // no more lines
    mov  x22, x21                // x22 = scan offset
scan_loop:
    ldrb w9, [x19, x22]
    cmp  w9, #0x0A
    b.eq scan_done
    add  x22, x22, #1
    b    scan_loop
scan_done:
    // reverse bytes in [x21, x22)
    mov  x23, x21
    sub  x24, x22, #1
rev_loop:
    cmp  x23, x24
    b.hs rev_done
    ldrb w9,  [x19, x23]
    ldrb w10, [x19, x24]
    strb w10, [x19, x23]
    strb w9,  [x19, x24]
    add  x23, x23, #1
    sub  x24, x24, #1
    b    rev_loop
rev_done:
    add  x21, x22, #1            // next line starts after the newline
    b    line_loop

    // write(1, inbuf + written, x20 - written) until everything is out
write_init:
    mov  x25, #0                 // x25 = bytes written so far
write_loop:
    cmp  x25, x20
    b.hs exit_prog
    mov  x0, #1
    mov  x1, x19
    add  x1, x1, x25
    sub  x2, x20, x25
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit_prog
    add  x25, x25, x0
    b    write_loop

exit_prog:
    mov  x0, #0
    mov  x8, #93
    svc  #0
1:  b    1b                      // unreachable: exit never returns

.bss
.balign 16
inbuf:
    .zero BUFSIZE+16
