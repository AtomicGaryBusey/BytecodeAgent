.section .rodata
okmsg: .ascii "ok\n"
okmsg_len = . - okmsg
errmsg: .ascii "error at "
errmsg_len = . - errmsg
lit_true: .ascii "true"
lit_false: .ascii "false"
lit_null: .ascii "null"

.text
.globl _start

set_err:
    mov qword ptr [rip + err], 1
    mov qword ptr [rip + errpos], rax
    mov eax, 1
    ret

skip_ws:
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
skip_ws_loop:
    cmp rcx, rdx
    jae skip_ws_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x20
    je skip_ws_next
    cmp al, 0x09
    je skip_ws_next
    cmp al, 0x0A
    je skip_ws_next
    cmp al, 0x0D
    je skip_ws_next
    jmp skip_ws_done
skip_ws_next:
    inc rcx
    jmp skip_ws_loop
skip_ws_done:
    mov qword ptr [rip + pos], rcx
    ret

parse_literal:
    mov rcx, qword ptr [rip + pos]
    mov r8, qword ptr [rip + len]
    lea rbx, [rip + buf]
    xor r9, r9
parse_literal_loop:
    cmp r9, rdx
    jae parse_literal_done
    cmp rcx, r8
    jae parse_literal_eof
    movzx eax, byte ptr [rbx + rcx]
    movzx r10d, byte ptr [rsi + r9]
    cmp al, r10b
    jne parse_literal_err
    inc rcx
    inc r9
    jmp parse_literal_loop
parse_literal_done:
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_literal_eof:
    mov rax, r8
    jmp set_err
parse_literal_err:
    mov rax, rcx
    jmp set_err

parse_string:
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    inc rcx
parse_string_loop:
    cmp rcx, rdx
    jae parse_string_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x22
    je parse_string_close
    cmp al, 0x5C
    je parse_string_esc
    cmp al, 0x20
    jb parse_string_err
    inc rcx
    jmp parse_string_loop
parse_string_close:
    inc rcx
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_string_esc:
    inc rcx
    cmp rcx, rdx
    jae parse_string_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x22
    je parse_string_esc_ok
    cmp al, 0x5C
    je parse_string_esc_ok
    cmp al, 0x2F
    je parse_string_esc_ok
    cmp al, 0x62
    je parse_string_esc_ok
    cmp al, 0x66
    je parse_string_esc_ok
    cmp al, 0x6E
    je parse_string_esc_ok
    cmp al, 0x72
    je parse_string_esc_ok
    cmp al, 0x74
    je parse_string_esc_ok
    cmp al, 0x75
    je parse_string_u
    mov rax, rcx
    jmp set_err
parse_string_esc_ok:
    inc rcx
    jmp parse_string_loop
parse_string_u:
    lea r8, [rcx + 1]
    mov r9, 4
parse_string_u_loop:
    cmp r8, rdx
    jae parse_string_eof
    movzx eax, byte ptr [rbx + r8]
    cmp al, 0x30
    jb parse_string_u_err
    cmp al, 0x39
    jbe parse_string_u_hex
    cmp al, 0x41
    jb parse_string_u_err
    cmp al, 0x46
    jbe parse_string_u_hex
    cmp al, 0x61
    jb parse_string_u_err
    cmp al, 0x66
    jbe parse_string_u_hex
    jmp parse_string_u_err
parse_string_u_hex:
    inc r8
    dec r9
    jnz parse_string_u_loop
    mov rcx, r8
    jmp parse_string_loop
parse_string_u_err:
    mov rax, r8
    jmp set_err
parse_string_eof:
    mov rax, rdx
    jmp set_err
parse_string_err:
    mov rax, rcx
    jmp set_err

parse_number:
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x2D
    jne parse_number_nominus
    inc rcx
    cmp rcx, rdx
    jae parse_number_eof
parse_number_nominus:
    cmp rcx, rdx
    jae parse_number_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x30
    je parse_number_zero
    cmp al, 0x31
    jb parse_number_err
    cmp al, 0x39
    ja parse_number_err
    inc rcx
parse_number_int_digits:
    cmp rcx, rdx
    jae parse_number_after_int
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x30
    jb parse_number_after_int
    cmp al, 0x39
    ja parse_number_after_int
    inc rcx
    jmp parse_number_int_digits
parse_number_zero:
    inc rcx
    jmp parse_number_after_int
parse_number_after_int:
    cmp rcx, rdx
    jae parse_number_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x2E
    jne parse_number_check_exp
    inc rcx
    cmp rcx, rdx
    jae parse_number_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x30
    jb parse_number_err
    cmp al, 0x39
    ja parse_number_err
parse_number_frac_digits:
    inc rcx
    cmp rcx, rdx
    jae parse_number_check_exp
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x30
    jb parse_number_check_exp
    cmp al, 0x39
    ja parse_number_check_exp
    jmp parse_number_frac_digits
parse_number_check_exp:
    cmp rcx, rdx
    jae parse_number_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x65
    je parse_number_exp
    cmp al, 0x45
    je parse_number_exp
    jmp parse_number_done
parse_number_exp:
    inc rcx
    cmp rcx, rdx
    jae parse_number_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x2B
    je parse_number_exp_sign
    cmp al, 0x2D
    je parse_number_exp_sign
    jmp parse_number_exp_digit_check
parse_number_exp_sign:
    inc rcx
    cmp rcx, rdx
    jae parse_number_eof
parse_number_exp_digit_check:
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x30
    jb parse_number_err
    cmp al, 0x39
    ja parse_number_err
parse_number_exp_digits:
    inc rcx
    cmp rcx, rdx
    jae parse_number_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x30
    jb parse_number_done
    cmp al, 0x39
    ja parse_number_done
    jmp parse_number_exp_digits
parse_number_done:
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_number_eof:
    mov rax, rdx
    jmp set_err
parse_number_err:
    mov rax, rcx
    jmp set_err

parse_value:
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rcx, rdx
    jae parse_value_eof
    lea rbx, [rip + buf]
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x7B
    je parse_value_object
    cmp al, 0x5B
    je parse_value_array
    cmp al, 0x22
    je parse_value_string
    cmp al, 0x74
    je parse_value_true
    cmp al, 0x66
    je parse_value_false
    cmp al, 0x6E
    je parse_value_null
    cmp al, 0x2D
    je parse_value_number
    cmp al, 0x30
    jb parse_value_err
    cmp al, 0x39
    ja parse_value_err
    jmp parse_value_number
parse_value_object:
    call parse_object
    ret
parse_value_array:
    call parse_array
    ret
parse_value_string:
    call parse_string
    ret
parse_value_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    call parse_literal
    ret
parse_value_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    call parse_literal
    ret
parse_value_null:
    lea rsi, [rip + lit_null]
    mov edx, 4
    call parse_literal
    ret
parse_value_number:
    call parse_number
    ret
parse_value_eof:
    mov rax, rdx
    jmp set_err
parse_value_err:
    mov rax, rcx
    jmp set_err

parse_array:
    mov rcx, qword ptr [rip + pos]
    inc rcx
    mov qword ptr [rip + pos], rcx
    call skip_ws
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    cmp rcx, rdx
    jae parse_array_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x5D
    je parse_array_close_empty
parse_array_value:
    call parse_value
    test rax, rax
    jnz parse_array_ret
parse_array_after_value:
    call skip_ws
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    cmp rcx, rdx
    jae parse_array_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x2C
    je parse_array_comma
    cmp al, 0x5D
    je parse_array_close
    mov rax, rcx
    jmp set_err
parse_array_comma:
    inc rcx
    mov qword ptr [rip + pos], rcx
    call skip_ws
    jmp parse_array_value
parse_array_close_empty:
    inc rcx
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_array_close:
    inc rcx
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_array_eof:
    mov rax, rdx
    jmp set_err
parse_array_ret:
    ret

parse_object:
    mov rcx, qword ptr [rip + pos]
    inc rcx
    mov qword ptr [rip + pos], rcx
    call skip_ws
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    cmp rcx, rdx
    jae parse_object_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x7D
    je parse_object_close_empty
parse_object_member:
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    cmp rcx, rdx
    jae parse_object_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x22
    jne parse_object_err
    call parse_string
    test rax, rax
    jnz parse_object_ret
    call skip_ws
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    cmp rcx, rdx
    jae parse_object_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x3A
    jne parse_object_err
    inc rcx
    mov qword ptr [rip + pos], rcx
    call skip_ws
    call parse_value
    test rax, rax
    jnz parse_object_ret
    call skip_ws
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rbx, [rip + buf]
    cmp rcx, rdx
    jae parse_object_eof
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x2C
    je parse_object_comma
    cmp al, 0x7D
    je parse_object_close
    mov rax, rcx
    jmp set_err
parse_object_comma:
    inc rcx
    mov qword ptr [rip + pos], rcx
    call skip_ws
    jmp parse_object_member
parse_object_close_empty:
    inc rcx
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_object_close:
    inc rcx
    mov qword ptr [rip + pos], rcx
    xor eax, eax
    ret
parse_object_eof:
    mov rax, rdx
    jmp set_err
parse_object_err:
    mov rax, rcx
    jmp set_err
parse_object_ret:
    ret

_start:
    sub rsp, 96
    lea r12, [rip + buf]
    xor r13, r13
read_loop:
    mov edi, 0
    lea rsi, [r12 + r13]
    mov rdx, 65537
    sub rdx, r13
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r13, rax
    cmp r13, 65537
    jb read_loop
read_done:
    mov qword ptr [rip + len], r13
    mov qword ptr [rip + pos], 0
    mov qword ptr [rip + err], 0
    mov qword ptr [rip + errpos], 0
    call skip_ws
    call parse_value
    test rax, rax
    jnz start_err
    call skip_ws
    mov rcx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rcx, rdx
    jae start_ok
    mov qword ptr [rip + err], 1
    mov qword ptr [rip + errpos], rcx
start_err:
    lea rsi, [rip + errmsg]
    mov edx, errmsg_len
    mov edi, 1
    mov eax, 1
    syscall
    mov rax, qword ptr [rip + errpos]
    lea r9, [rsp + 95]
    mov byte ptr [r9], 0x0A
    dec r9
    mov rbx, 10
convert_loop:
    xor edx, edx
    div rbx
    add dl, 0x30
    mov byte ptr [r9], dl
    dec r9
    test rax, rax
    jnz convert_loop
    lea rsi, [r9 + 1]
    lea rdx, [rsp + 96]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
start_ok:
    lea rsi, [rip + okmsg]
    mov edx, okmsg_len
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

.bss
.balign 8
buf: .zero 65537
.balign 8
len: .zero 8
pos: .zero 8
err: .zero 8
errpos: .zero 8
