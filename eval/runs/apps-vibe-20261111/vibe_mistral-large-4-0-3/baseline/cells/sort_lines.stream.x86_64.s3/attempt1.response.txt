.globl _start

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buf]
rd_loop:
    mov rax, 32768
    sub rax, r12
    test rax, rax
    jz rd_done
    mov rdx, rax
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    syscall
    test rax, rax
    jz rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    xor r14, r14
    xor r15, r15
    xor rbx, rbx
bl_scan:
    cmp rbx, r12
    jae bl_scan_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 10
    jne bl_next
    lea rax, [rip + line_off]
    mov qword ptr [rax + r15*8], r14
    lea rax, [rip + line_len]
    mov rcx, rbx
    sub rcx, r14
    mov qword ptr [rax + r15*8], rcx
    inc r15
    lea r14, [rbx + 1]
bl_next:
    inc rbx
    jmp bl_scan
bl_scan_done:
    cmp r14, r12
    jae bl_done
    lea rax, [rip + line_off]
    mov qword ptr [rax + r15*8], r14
    lea rax, [rip + line_len]
    mov rcx, r12
    sub rcx, r14
    mov qword ptr [rax + r15*8], rcx
    inc r15
bl_done:
    lea rax, [rip + n_lines]
    mov qword ptr [rax], r15
    xor rcx, rcx
init_loop:
    cmp rcx, r15
    jae init_done
    lea rax, [rip + idx]
    mov qword ptr [rax + rcx*8], rcx
    inc rcx
    jmp init_loop
init_done:
    sub rsp, 64
    mov qword ptr [rsp + 0], 1
ms_outer:
    mov rax, qword ptr [rsp + 0]
    mov rcx, qword ptr [rip + n_lines]
    cmp rax, rcx
    jae ms_done
    mov qword ptr [rsp + 8], 0
ms_inner:
    mov rax, qword ptr [rsp + 8]
    mov rcx, qword ptr [rip + n_lines]
    cmp rax, rcx
    jae ms_next_width
    mov rax, qword ptr [rsp + 8]
    add rax, qword ptr [rsp + 0]
    mov rcx, qword ptr [rip + n_lines]
    cmp rax, rcx
    jbe ms_mid_ok
    mov rax, rcx
ms_mid_ok:
    mov qword ptr [rsp + 16], rax
    mov rax, qword ptr [rsp + 8]
    add rax, qword ptr [rsp + 0]
    add rax, qword ptr [rsp + 0]
    mov rcx, qword ptr [rip + n_lines]
    cmp rax, rcx
    jbe ms_end_ok
    mov rax, rcx
ms_end_ok:
    mov qword ptr [rsp + 24], rax
    mov rax, qword ptr [rsp + 16]
    cmp rax, qword ptr [rsp + 24]
    jae ms_next_seg
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [rsp + 32], rax
    mov rax, qword ptr [rsp + 16]
    mov qword ptr [rsp + 40], rax
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [rsp + 48], rax
ms_merge_loop:
    mov rax, qword ptr [rsp + 16]
    cmp qword ptr [rsp + 32], rax
    jae ms_left_done
    mov rax, qword ptr [rsp + 24]
    cmp qword ptr [rsp + 40], rax
    jae ms_right_done
    lea r14, [rip + idx]
    mov rax, qword ptr [rsp + 32]
    mov rdi, qword ptr [r14 + rax*8]
    mov rax, qword ptr [rsp + 40]
    mov rsi, qword ptr [r14 + rax*8]
    call compare_lines
    test eax, eax
    jle ms_take_left
    mov rax, qword ptr [rsp + 40]
    lea r14, [rip + idx]
    mov rax, qword ptr [r14 + rax*8]
    lea rbx, [rip + tmp]
    mov rcx, qword ptr [rsp + 48]
    mov qword ptr [rbx + rcx*8], rax
    inc qword ptr [rsp + 40]
    inc qword ptr [rsp + 48]
    jmp ms_merge_loop
ms_take_left:
    mov rax, qword ptr [rsp + 32]
    lea r14, [rip + idx]
    mov rax, qword ptr [r14 + rax*8]
    lea rbx, [rip + tmp]
    mov rcx, qword ptr [rsp + 48]
    mov qword ptr [rbx + rcx*8], rax
    inc qword ptr [rsp + 32]
    inc qword ptr [rsp + 48]
    jmp ms_merge_loop
ms_left_done:
    mov rax, qword ptr [rsp + 24]
    cmp qword ptr [rsp + 40], rax
    jae ms_copyback
    mov rax, qword ptr [rsp + 40]
    lea r14, [rip + idx]
    mov rax, qword ptr [r14 + rax*8]
    lea rbx, [rip + tmp]
    mov rcx, qword ptr [rsp + 48]
    mov qword ptr [rbx + rcx*8], rax
    inc qword ptr [rsp + 40]
    inc qword ptr [rsp + 48]
    jmp ms_left_done
ms_right_done:
    mov rax, qword ptr [rsp + 16]
    cmp qword ptr [rsp + 32], rax
    jae ms_copyback
    mov rax, qword ptr [rsp + 32]
    lea r14, [rip + idx]
    mov rax, qword ptr [r14 + rax*8]
    lea rbx, [rip + tmp]
    mov rcx, qword ptr [rsp + 48]
    mov qword ptr [rbx + rcx*8], rax
    inc qword ptr [rsp + 32]
    inc qword ptr [rsp + 48]
    jmp ms_right_done
ms_copyback:
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [rsp + 32], rax
ms_cb_loop:
    mov rax, qword ptr [rsp + 24]
    cmp qword ptr [rsp + 32], rax
    jae ms_cb_done
    lea rbx, [rip + tmp]
    mov rcx, qword ptr [rsp + 32]
    mov rdx, qword ptr [rbx + rcx*8]
    lea r14, [rip + idx]
    mov qword ptr [r14 + rcx*8], rdx
    inc qword ptr [rsp + 32]
    jmp ms_cb_loop
ms_cb_done:
ms_next_seg:
    mov rax, qword ptr [rsp + 8]
    add rax, qword ptr [rsp + 0]
    add rax, qword ptr [rsp + 0]
    mov qword ptr [rsp + 8], rax
    jmp ms_inner
ms_next_width:
    mov rax, qword ptr [rsp + 0]
    shl rax, 1
    mov qword ptr [rsp + 0], rax
    jmp ms_outer
ms_done:
    add rsp, 64
    xor r12, r12
out_loop:
    mov rcx, qword ptr [rip + n_lines]
    cmp r12, rcx
    jae out_done
    lea r14, [rip + idx]
    mov rax, qword ptr [r14 + r12*8]
    lea rbx, [rip + line_off]
    mov rsi, qword ptr [rbx + rax*8]
    lea rbx, [rip + line_len]
    mov rdx, qword ptr [rbx + rax*8]
    lea r13, [rip + input_buf]
    add rsi, r13
    test rdx, rdx
    jz out_nl
    mov eax, 1
    mov edi, 1
    syscall
out_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    inc r12
    jmp out_loop
out_done:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r10, rdi
    mov r11, rsi
    lea rax, [rip + line_off]
    mov rdi, qword ptr [rax + r10*8]
    lea rax, [rip + line_len]
    mov rsi, qword ptr [rax + r10*8]
    lea rax, [rip + line_off]
    mov rcx, qword ptr [rax + r11*8]
    lea rax, [rip + line_len]
    mov rdx, qword ptr [rax + r11*8]
    lea r13, [rip + input_buf]
    add rdi, r13
    add rcx, r13
    xor eax, eax
cl_loop:
    cmp rax, rsi
    jae cl_len
    cmp rax, rdx
    jae cl_len
    movzx r8, byte ptr [rdi + rax]
    movzx r9, byte ptr [rcx + rax]
    cmp r8, r9
    jb cl_less
    ja cl_greater
    inc rax
    jmp cl_loop
cl_len:
    cmp rsi, rdx
    jb cl_less
    ja cl_greater
    xor eax, eax
    ret
cl_less:
    mov eax, -1
    ret
cl_greater:
    mov eax, 1
    ret
