.text
.equ IN_CAP, 65536
.equ OUT_CAP, 65537
.equ NL, 10
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.globl _start
_start:
    lea r13, [rip + in_buf]
    xor r12, r12
read_loop:
    mov eax, IN_CAP
    sub eax, r12d
    test eax, eax
    jz read_done
    mov rdx, rax
    lea rsi, [r13 + r12]
    mov edi, 0
    mov eax, SYS_READ
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rbx, [rip + in_buf]
    lea r14, [rip + out_buf]
    xor r15, r15
    xor r8, r8
    xor r9, r9
    cmp r8, r12
    jae build_final
build_scan:
    movzx eax, byte ptr [rbx + r8]
    cmp al, NL
    je build_emit
    inc r8
    jmp build_check
build_emit:
    mov rdx, r8
    sub rdx, r9
    lea rsi, [rbx + r9]
    test rdx, rdx
    jz build_nl
build_copy:
    dec rdx
    movzx eax, byte ptr [rsi + rdx]
    mov byte ptr [r14 + r15], al
    inc r15
    test rdx, rdx
    jnz build_copy
build_nl:
    mov byte ptr [r14 + r15], NL
    inc r15
    inc r8
    mov r9, r8
build_check:
    cmp r8, r12
    jb build_scan
build_final:
    cmp r9, r12
    jae build_done
    mov rdx, r12
    sub rdx, r9
    lea rsi, [rbx + r9]
    test rdx, rdx
    jz build_final_nl
build_final_copy:
    dec rdx
    movzx eax, byte ptr [rsi + rdx]
    mov byte ptr [r14 + r15], al
    inc r15
    test rdx, rdx
    jnz build_final_copy
build_final_nl:
    mov byte ptr [r14 + r15], NL
    inc r15
build_done:
    test r15, r15
    jz exit_ok
    lea rsi, [rip + out_buf]
    mov rdx, r15
write_loop:
    mov edi, 1
    mov eax, SYS_WRITE
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop
exit_ok:
    mov edi, 0
    mov eax, SYS_EXIT
    syscall

.bss
.balign 16
in_buf:
    .zero IN_CAP
out_buf:
    .zero OUT_CAP
