.equ INPUT_CAP, 32768
.equ ARR_BYTES, 8000

.globl _start

.text
_start:
    bl run
    mov x0, #0
    mov x8, #93
    svc #0

run:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    mov x29, sp
    mov x19, #0
read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #INPUT_CAP
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    b.lt read_err
    add x19, x19, x0
    b read_loop
read_done:
    cmp x19, #0
    b.eq empty_input
    adrp x0, line_start
    add x0, x0, :lo12:line_start
    adrp x1, line_len
    add x1, x1, :lo12:line_len
    adrp x2, input_buf
    add x2, x2, :lo12:input_buf
    mov x3, #0
    mov x4, x2
    mov x5, #0
line_loop:
    str x4, [x0, x3, lsl #3]
line_scan:
    cmp x5, x19
    b.hs line_finish
    ldrb w6, [x2, x5]
    cmp w6, #10
    b.eq line_finish
    add x5, x5, #1
    b line_scan
line_finish:
    add x7, x2, x5
    sub x7, x7, x4
    str x7, [x1, x3, lsl #3]
    add x3, x3, #1
    cmp x5, x19
    b.hs parse_done
    add x5, x5, #1
    cmp x5, x19
    b.hs parse_done
    add x4, x2, x5
    b line_loop
parse_done:
    mov x20, x3
    b init_idx
empty_input:
    mov x20, #0
init_idx:
    cmp x20, #0
    b.eq do_sort
    adrp x0, idx
    add x0, x0, :lo12:idx
    mov x1, #0
init_loop:
    cmp x1, x20
    b.hs init_done
    str x1, [x0, x1, lsl #3]
    add x1, x1, #1
    b init_loop
init_done:
do_sort:
    mov x0, x20
    bl merge_sort
    mov x9, #0
out_loop:
    cmp x9, x20
    b.hs out_done
    adrp x4, idx
    add x4, x4, :lo12:idx
    ldr x5, [x4, x9, lsl #3]
    adrp x6, line_start
    add x6, x6, :lo12:line_start
    ldr x1, [x6, x5, lsl #3]
    adrp x7, line_len
    add x7, x7, :lo12:line_len
    ldr x2, [x7, x5, lsl #3]
    mov x0, #1
    bl write_all
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    bl write_all
    add x9, x9, #1
    b out_loop
out_done:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret
read_err:
    mov x0, #1
    mov x8, #93
    svc #0

merge_sort:
    str x30, [sp, #-16]!
    cmp x0, #1
    b.le ms_done
    mov x18, x0
    adrp x14, idx
    add x14, x14, :lo12:idx
    adrp x15, tmp
    add x15, x15, :lo12:tmp
    mov x9, #1
ms_outer:
    cmp x9, x18
    b.hs ms_done
    lsl x16, x9, #1
    mov x10, #0
ms_seg:
    cmp x10, x18
    b.hs ms_seg_done
    add x12, x10, x9
    cmp x12, x18
    csel x12, x12, x18, lo
    add x13, x10, x16
    cmp x13, x18
    csel x13, x13, x18, lo
    mov x11, x10
    mov x17, x10
    mov x8, x12
ms_merge:
    cmp x17, x12
    b.hs ms_copy_right
    cmp x8, x13
    b.hs ms_copy_left
    ldr x0, [x14, x17, lsl #3]
    ldr x1, [x14, x8, lsl #3]
    bl compare
    cmp x0, #0
    b.le ms_take_left
    ldr x2, [x14, x8, lsl #3]
    str x2, [x15, x11, lsl #3]
    add x8, x8, #1
    add x11, x11, #1
    b ms_merge
ms_take_left:
    ldr x2, [x14, x17, lsl #3]
    str x2, [x15, x11, lsl #3]
    add x17, x17, #1
    add x11, x11, #1
    b ms_merge
ms_copy_left:
    cmp x17, x12
    b.hs ms_seg_next
    ldr x2, [x14, x17, lsl #3]
    str x2, [x15, x11, lsl #3]
    add x17, x17, #1
    add x11, x11, #1
    b ms_copy_left
ms_copy_right:
    cmp x8, x13
    b.hs ms_seg_next
    ldr x2, [x14, x8, lsl #3]
    str x2, [x15, x11, lsl #3]
    add x8, x8, #1
    add x11, x11, #1
    b ms_copy_right
ms_seg_next:
    add x10, x10, x16
    b ms_seg
ms_seg_done:
    mov x5, #0
ms_copy_back:
    cmp x5, x18
    b.hs ms_copy_back_done
    ldr x6, [x15, x5, lsl #3]
    str x6, [x14, x5, lsl #3]
    add x5, x5, #1
    b ms_copy_back
ms_copy_back_done:
    lsl x9, x9, #1
    b ms_outer
ms_done:
    ldr x30, [sp], #16
    ret

compare:
    adrp x6, line_start
    add x6, x6, :lo12:line_start
    ldr x2, [x6, x0, lsl #3]
    adrp x7, line_len
    add x7, x7, :lo12:line_len
    ldr x3, [x7, x0, lsl #3]
    ldr x4, [x6, x1, lsl #3]
    ldr x5, [x7, x1, lsl #3]
    cmp x3, x5
    csel x6, x3, x5, le
cmp_loop:
    cbz x6, cmp_len
    ldrb w7, [x2], #1
    ldrb w0, [x4], #1
    cmp w7, w0
    b.lo cmp_neg
    b.hi cmp_pos
    sub x6, x6, #1
    b cmp_loop
cmp_len:
    cmp x3, x5
    b.lo cmp_neg
    b.hi cmp_pos
    mov x0, #0
    ret
cmp_neg:
    mov x0, #0
    sub x0, x0, #1
    ret
cmp_pos:
    mov x0, #1
    ret

write_all:
    str x30, [sp, #-16]!
    mov x3, x0
wa_loop:
    cbz x2, wa_done
    mov x0, x3
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le wa_err
    sub x2, x2, x0
    add x1, x1, x0
    b wa_loop
wa_done:
    ldr x30, [sp], #16
    ret
wa_err:
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata

.bss

    .zero ARR_BYTES

    .zero ARR_BYTES

    .zero ARR_BYTES
