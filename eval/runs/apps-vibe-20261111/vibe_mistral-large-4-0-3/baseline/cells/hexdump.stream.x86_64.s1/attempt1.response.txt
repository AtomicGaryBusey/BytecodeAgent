.equ IN_CAP, 65537
.equ OUT_MAX, 324000

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space IN_CAP
.balign 16
outbuf:
    .space OUT_MAX

.text
.globl _start
_start:
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov rdx, IN_CAP
    xor r12, r12
.Lread:
    test rdx, rdx
    jz .Lread_done
    xor eax, eax
    syscall
    test rax, rax
    jz .Lread_done
    add rsi, rax
    sub rdx, rax
    add r12, rax
    jmp .Lread
.Lread_done:
    lea r9, [rip + hexdigits]
    lea r14, [rip + outbuf]
    xor rbx, rbx
.Lrow:
    cmp rbx, r12
    jae .Lrows_done
    mov eax, ebx
    mov ecx, 8
.Loff_hex:
    mov r8d, eax
    shr r8d, 28
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14], r8b
    inc r14
    shl eax, 4
    dec ecx
    jnz .Loff_hex
    mov byte ptr [r14], 32
    mov byte ptr [r14 + 1], 32
    add r14, 2
    mov r15, r12
    sub r15, rbx
    cmp r15, 16
    jbe .Ln_set
    mov r15, 16
.Ln_set:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edx, edx
.Lslot:
    cmp edx, r15d
    jae .Lpad
    movzx eax, byte ptr [rsi + rdx]
    mov r8d, eax
    shr r8d, 4
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14], r8b
    mov r8d, eax
    and r8d, 15
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14 + 1], r8b
    mov byte ptr [r14 + 2], 32
    add r14, 3
    jmp .Lslot_next
.Lpad:
    mov byte ptr [r14], 32
    mov byte ptr [r14 + 1], 32
    mov byte ptr [r14 + 2], 32
    add r14, 3
.Lslot_next:
    cmp edx, 7
    jne .Lno_extra
    mov byte ptr [r14], 32
    inc r14
.Lno_extra:
    inc edx
    cmp edx, 16
    jb .Lslot
    mov byte ptr [r14], 32
    inc r14
    mov byte ptr [r14], 124
    inc r14
    xor edx, edx
.Lascii:
    cmp edx, r15d
    jae .Lascii_done
    movzx eax, byte ptr [rsi + rdx]
    cmp eax, 32
    jb .Ldot
    cmp eax, 126
    ja .Ldot
    mov byte ptr [r14], al
    jmp .Lstore
.Ldot:
    mov byte ptr [r14], 46
.Lstore:
    inc r14
    inc edx
    jmp .Lascii
.Lascii_done:
    mov byte ptr [r14], 124
    inc r14
    mov byte ptr [r14], 10
    inc r14
    add rbx, 16
    jmp .Lrow
.Lrows_done:
    mov eax, r12d
    mov ecx, 8
.Lfin_hex:
    mov r8d, eax
    shr r8d, 28
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14], r8b
    inc r14
    shl eax, 4
    dec ecx
    jnz .Lfin_hex
    mov byte ptr [r14], 10
    inc r14
    lea rsi, [rip + outbuf]
    mov rdi, 1
    mov rdx, r14
    sub rdx, rsi
.Lwrite:
    test rdx, rdx
    jz .Lwrite_done
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite
.Lwrite_done:
    mov eax, 60
    xor edi, edi
    syscall
