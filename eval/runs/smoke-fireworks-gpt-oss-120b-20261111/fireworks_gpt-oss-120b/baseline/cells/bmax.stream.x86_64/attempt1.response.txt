bmax:
    mov rax, qword ptr [rdi]          # load first element as current max
    lea rdx, [rdi + 8]                # pointer to second element
    dec rsi                           # n-1 remaining elements
    jz .Ldone                         # if only one element, finish
.Lloop:
    mov rcx, qword ptr [rdx]          # load next element
    cmp rcx, rax
    cmovg rax, rcx                    # if rcx > rax, update max
    add rdx, 8                        # advance pointer
    dec rsi
    jnz .Lloop
.Ldone:
    ret
