gcd:
    mov rax, rdi            # a = first argument
    mov r8, rsi             # b = second argument
.Lloop:
    test r8, r8
    jz .Ldone
    xor rdx, rdx            # clear high 64 bits for div
    mov rcx, r8             # divisor = b
    div rcx                 # rdx = a % b, rax = a / b (ignored)
    mov rax, r8             # a = b
    mov r8, rdx             # b = remainder
    jmp .Lloop
.Ldone:
    ret
