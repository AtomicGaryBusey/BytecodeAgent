export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%c =l copy 0
	jmp @cond
@cond
	%i1 =l phi @start %i, @body %i2
	%c1 =l phi @start %c, @body %c2
	%more =w csltl %i1, %n
	jnz %more, @body, @done
@body
	%off =l mul %i1, 8
	%p =l add %a, %off
	%v =l loadl %p
	%g =w cugtl %v, %t
	%gl =l extuw %g
	%c2 =l add %c1, %gl
	%i2 =l add %i1, 1
	jmp @cond
@done
	ret %c1
}
