max_byte:
    xor eax, eax
    cmp rsi, 16
    jb .Lsmall
    pxor xmm0, xmm0
    xor ecx, ecx
.Lvec:
    movdqu xmm1, xmmword ptr [rdi + rcx]
    pmaxub xmm0, xmm1
    add rcx, 16
    lea rdx, [rcx + 16]
    cmp rdx, rsi
    jbe .Lvec
    movdqu xmm1, xmmword ptr [rdi + rsi - 16]
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
    ret
.Lsmall:
    test rsi, rsi
    jz .Ldone
    xor ecx, ecx
.Lloop:
    movzx edx, byte ptr [rdi + rcx]
    cmp edx, eax
    cmova eax, edx
    inc rcx
    cmp rcx, rsi
    jb .Lloop
.Ldone:
    ret
