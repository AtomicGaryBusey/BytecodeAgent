export function l $count_desc(l %a, l %n) {
@start
	%c =l copy 0
	%lt =w csltl %n, 2
	jnz %lt, @done, @init
@init
	%m =l sub %n, 1
	%i =l copy 0
	%p =l copy %a
	jmp @loop
@loop
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%gt =w csgtl %x, %y
	%gtl =l extuw %gt
	%c =l add %c, %gtl
	%p =l copy %q
	%i =l add %i, 1
	%more =w csltl %i, %m
	jnz %more, @loop, @done
@done
	ret %c
}
