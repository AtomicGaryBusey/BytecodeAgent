export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%mn =l copy %y0
	%mx =l copy %y0
	%sum =l copy %y0
	%cnt =l and %y0, 1
	%i =l copy 1
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum =l add %sum, %y
	%odd =l and %y, 1
	%cnt =l add %cnt, %odd
	%lt =w csltl %y, %mn
	jnz %lt, @setmin, @chkmax
@setmin
	%mn =l copy %y
@chkmax
	%gt =w csgtl %y, %mx
	jnz %gt, @setmax, @next
@setmax
	%mx =l copy %y
@next
	%i =l add %i, 1
	jmp @loop
@done
	storel %mn, %out
	%o1 =l add %out, 8
	storel %mx, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
