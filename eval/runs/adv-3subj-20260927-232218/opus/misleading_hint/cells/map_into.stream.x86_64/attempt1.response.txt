map_into:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
    mov r12, rdx
    mov r13, rcx
    xor r14d, r14d
    xor r15d, r15d
    test r12, r12
    jle .Ldone
.Lloop:
    mov rdi, qword ptr [rbp + r14*8]
    call r13
    mov qword ptr [rbx + r14*8], rax
    xor ecx, ecx
    cmp rax, qword ptr [rbp + r14*8]
    setl cl
    add r15, rcx
    inc r14
    cmp r14, r12
    jl .Lloop
.Ldone:
    mov rax, r15
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
