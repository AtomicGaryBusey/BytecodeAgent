export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	%dp =l add %dst, %off
	storel %r, %dp
	%lt =w csltl %r, %x
	%ltl =l extuw %lt
	%cnt =l add %cnt, %ltl
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
