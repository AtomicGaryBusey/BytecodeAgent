export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%p0 =l copy %a
	%x0 =l loadl %p0
	%y0 =l call %f(l %x0)
	%o0 =l and %y0, 1
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%mn =l phi @start %y0, @body %mn1
	%mx =l phi @start %y0, @body %mx1
	%s =l phi @start %y0, @body %s1
	%c =l phi @start %o0, @body %c1
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%s1 =l add %s, %y
	%od =l and %y, 1
	%c1 =l add %c, %od
	%isl =w csltl %y, %mn
	%mn1 =l phi_placeholder
	%i1 =l add %i, 1
	jmp @loop
@done
	%q0 =l add %out, 0
	storel %mn, %q0
	%q1 =l add %out, 8
	storel %mx, %q1
	%q2 =l add %out, 16
	storel %s, %q2
	ret %c
}
