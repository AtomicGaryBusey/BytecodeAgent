export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%o0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @j2 %i1
	%mn =l phi @start %y0, @j2 %mn1
	%mx =l phi @start %y0, @j2 %mx1
	%s =l phi @start %y0, @j2 %s1
	%c =l phi @start %o0, @j2 %c1
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%s1 =l add %s, %y
	%od =l and %y, 1
	%c1 =l add %c, %od
	%islt =w csltl %y, %mn
	jnz %islt, @a1, @a2
@a1
	jmp @j1
@a2
	jmp @j1
@j1
	%mn1 =l phi @a1 %y, @a2 %mn
	%isgt =w csgtl %y, %mx
	jnz %isgt, @b1, @b2
@b1
	jmp @j2
@b2
	jmp @j2
@j2
	%mx1 =l phi @b1 %y, @b2 %mx
	%i1 =l add %i, 1
	jmp @loop
@done
	storel %mn, %out
	%q1 =l add %out, 8
	storel %mx, %q1
	%q2 =l add %out, 16
	storel %s, %q2
	ret %c
}
