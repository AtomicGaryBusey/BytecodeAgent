export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @zero, @init
@zero
	ret 0
@init
	%lim =l sub %n, 1
	%cnt0 =l copy 0
	%i0 =l copy 0
	%p0 =l copy %a
	%prev0 =l loadl %a
	jmp @loop
@loop
	%cnt =l phi @init %cnt0, @body %cnt2
	%i =l phi @init %i0, @body %i2
	%p =l phi @init %p0, @body %p2
	%prev =l phi @init %prev0, @body %cur
	%more =w csltl %i, %lim
	jnz %more, @body, @done
@body
	%p2 =l add %p, 8
	%cur =l loadl %p2
	%gt =w csgtl %prev, %cur
	%gtl =l extuw %gt
	%cnt2 =l add %cnt, %gtl
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
