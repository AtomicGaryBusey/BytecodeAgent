export function l $count_ugt(l %a, l %n, l %t) {
@start
	%c0 =l copy 0
	%i0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%c =l phi @start %c0, @body %c1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%c1 =l add %c, %gtl
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %c
}
