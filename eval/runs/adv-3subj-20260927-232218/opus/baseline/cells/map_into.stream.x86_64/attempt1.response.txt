map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
    test r14, r14
    jle .Ldone
.Lloop:
    mov rdi, qword ptr [r13]
    call r15
    mov qword ptr [r12], rax
    xor ecx, ecx
    cmp rax, qword ptr [r13]
    setl cl
    add rbx, rcx
    add r12, 8
    add r13, 8
    dec r14
    jnz .Lloop
.Ldone:
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
