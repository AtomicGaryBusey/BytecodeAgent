export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @zero, @init
@zero
	ret 0
@init
	%lim =l sub %n, 1
	%prev0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @init 0, @body %i1
	%cnt =l phi @init 0, @body %cnt1
	%prev =l phi @init %prev0, @body %cur
	%more =w csltl %i, %lim
	jnz %more, @body, @done
@body
	%i1 =l add %i, 1
	%off =l mul %i1, 8
	%p =l add %a, %off
	%cur =l loadl %p
	%gt =w csgtl %prev, %cur
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	jmp @loop
@done
	ret %cnt
}
