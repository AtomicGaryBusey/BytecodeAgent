export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @zero, @init
@zero
	ret 0
@init
	%prev0 =l loadl %a
	%p0 =l add %a, 8
	%end =l mul %n, 8
	%endp =l add %a, %end
	jmp @loop
@loop
	%cnt =l phi @init 0, @body %cnt2
	%prev =l phi @init %prev0, @body %cur
	%p =l phi @init %p0, @body %p2
	%done =w cugel %p, %endp
	jnz %done, @exit, @body
@body
	%cur =l loadl %p
	%gt =w csgtl %prev, %cur
	%gtl =l extuw %gt
	%cnt2 =l add %cnt, %gtl
	%p2 =l add %p, 8
	jmp @loop
@exit
	ret %cnt
}
