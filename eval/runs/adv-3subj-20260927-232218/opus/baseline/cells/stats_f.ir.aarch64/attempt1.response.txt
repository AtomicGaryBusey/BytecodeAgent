export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%i =l copy 0
	%min =l copy 0
	%max =l copy 0
	%sum =l copy 0
	%cnt =l copy 0
@loop
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum =l add %sum, %y
	%b =l and %y, 1
	%cnt =l add %cnt, %b
	%first =w ceql %i, 0
	jnz %first, @init, @cmp
@init
	%min =l copy %y
	%max =l copy %y
	jmp @next
@cmp
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	%min =l copy %y
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @next
@setmax
	%max =l copy %y
@next
	%i =l add %i, 1
	%c =w csltl %i, %n
	jnz %c, @loop, @done
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
