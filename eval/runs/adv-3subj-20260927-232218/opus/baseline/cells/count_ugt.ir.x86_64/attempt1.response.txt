export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%c =l copy 0
	jmp @loop
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%g =w cugtl %v, %t
	%ge =l extuw %g
	%c =l add %c, %ge
	%i =l add %i, 1
	jmp @loop
@done
	ret %c
}
