export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%c =l copy 0
@loop
	%t =w csltl %i, %n
	jnz %t, @body, @end
@body
	%o =l mul %i, 8
	%ps =l add %src, %o
	%x =l loadl %ps
	%r =l call %f(l %x)
	%pd =l add %dst, %o
	storel %r, %pd
	%lt =w csltl %r, %x
	%le =l extuw %lt
	%c =l add %c, %le
	%i =l add %i, 1
	jmp @loop
@end
	ret %c
}
