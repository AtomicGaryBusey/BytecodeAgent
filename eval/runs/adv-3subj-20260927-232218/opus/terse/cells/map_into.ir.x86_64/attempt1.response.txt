export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%c =l copy 0
@loop
	%t =w cnel %n, 0
	jnz %t, @body, @end
@body
	%x =l loadl %src
	%r =l call %f(l %x)
	storel %r, %dst
	%lt =w csltl %r, %x
	%e =l extuw %lt
	%c =l add %c, %e
	%src =l add %src, 8
	%dst =l add %dst, 8
	%n =l sub %n, 1
	jmp @loop
@end
	ret %c
}
