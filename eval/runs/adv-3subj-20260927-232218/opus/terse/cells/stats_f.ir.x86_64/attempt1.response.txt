export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%mn =l copy 9223372036854775807
	%mx =l sub -1, %mn
	%s =l copy 0
	%c =l copy 0
	%e =l mul %n, 8
	%e =l add %a, %e
	%p =l copy %a
@loop
	%x =l loadl %p
	%y =l call %f(l %x)
	%s =l add %s, %y
	%o =l and %y, 1
	%c =l add %c, %o
	%t =w csltl %y, %mn
	jnz %t, @setmn, @b
@setmn
	%mn =l copy %y
@b
	%u =w csgtl %y, %mx
	jnz %u, @setmx, @next
@setmx
	%mx =l copy %y
@next
	%p =l add %p, 8
	%k =w cnel %p, %e
	jnz %k, @loop, @done
@done
	storel %mn, %out
	%q =l add %out, 8
	storel %mx, %q
	%r =l add %out, 16
	storel %s, %r
	ret %c
}
