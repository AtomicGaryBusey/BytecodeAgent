map_into:
push rbx
push rbp
push r12
push r13
push r14
mov r13, rcx
xor r14d, r14d
lea rbx, [rdi + rdx*8]
lea rbp, [rsi + rdx*8]
neg rdx
mov r12, rdx
jz .Ldone
.Lloop:
mov rdi, qword ptr [rbp + r12*8]
call r13
mov qword ptr [rbx + r12*8], rax
cmp rax, qword ptr [rbp + r12*8]
jge .Lskip
inc r14
.Lskip:
inc r12
jnz .Lloop
.Ldone:
mov rax, r14
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
