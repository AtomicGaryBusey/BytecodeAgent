export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%p =l copy %a
	%mn =l copy 9223372036854775807
	%mx =l copy -9223372036854775808
	%s =l copy 0
	%c =l copy 0
	%e =l mul %n, 8
	%e =l add %p, %e
@loop
	%x =l loadl %p
	%y =l call %f(l %x)
	%s =l add %s, %y
	%o =l and %y, 1
	%c =l add %c, %o
	%t =w csltl %y, %mn
	jnz %t, @setmin, @m2
@setmin
	%mn =l copy %y
@m2
	%u =w csgtl %y, %mx
	jnz %u, @setmax, @next
@setmax
	%mx =l copy %y
@next
	%p =l add %p, 8
	%v =w cultl %p, %e
	jnz %v, @loop, @done
@done
	storel %mn, %out
	%q =l add %out, 8
	storel %mx, %q
	%q =l add %out, 16
	storel %s, %q
	ret %c
}
