export function l $count_desc(l %a, l %n) {
@start
	%c =l copy 0
	%i =l copy 1
@loop
	%r =l phi @start 0, @body %c2
	%j =l phi @start 1, @body %j2
	%t =w csltl %j, %n
	jnz %t, @body, @end
@body
	%p =l mul %j, 8
	%q =l add %a, %p
	%y =l loadl %q
	%q0 =l sub %q, 8
	%x =l loadl %q0
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%c2 =l add %r, %ge
	%j2 =l add %j, 1
	jmp @loop
@end
	ret %r
}
