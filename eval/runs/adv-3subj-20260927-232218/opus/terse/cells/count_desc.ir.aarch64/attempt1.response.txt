export function l $count_desc(l %a, l %n) {
@start
	%m =l sub %n, 1
	%e =l mul %m, 8
	%end =l add %a, %e
	jmp @loop
@loop
	%p =l phi @start %a, @body %q
	%c =l phi @start 0, @body %c2
	%t =w csltl %p, %end
	jnz %t, @body, @done
@body
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%c2 =l add %c, %gl
	jmp @loop
@done
	ret %c
}
