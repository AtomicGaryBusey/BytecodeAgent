export function l $fill_down(l %a, l %n, l %start) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%v =l phi @start %start, @body %v2
	%p =l phi @start %a, @body %p2
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	storel %v, %p
	%p2 =l add %p, 8
	%v2 =l sub %v, 1
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %n
}
