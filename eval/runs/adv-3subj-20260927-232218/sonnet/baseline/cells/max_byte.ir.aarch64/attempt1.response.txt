export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @loop
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %max
	jnz %gt, @upd, @next
@upd
	%max =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
