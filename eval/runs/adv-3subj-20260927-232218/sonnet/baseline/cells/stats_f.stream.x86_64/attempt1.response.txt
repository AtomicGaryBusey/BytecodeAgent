stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24
    cld

    mov r12, rdi        # a
    mov r13, rsi        # n
    mov r14, rdx        # f
    mov r15, rcx         # out

    movabs rax, 0x7fffffffffffffff
    mov [rsp+0], rax    # min init
    movabs rax, 0x8000000000000000
    mov [rsp+8], rax    # max init
    mov qword ptr [rsp+16], 0   # sum init

    xor rbx, rbx        # i = 0
    xor rbp, rbp        # odd count = 0

.Lloop:
    mov rdi, qword ptr [r12 + rbx*8]
    call r14

    add [rsp+16], rax

    cmp rax, [rsp+0]
    jge .Lskip_min
    mov [rsp+0], rax
.Lskip_min:

    cmp rax, [rsp+8]
    jle .Lskip_max
    mov [rsp+8], rax
.Lskip_max:

    test al, 1
    jz .Lskip_odd
    inc rbp
.Lskip_odd:

    inc rbx
    cmp rbx, r13
    jl .Lloop

    mov rax, [rsp+0]
    mov [r15+0], rax
    mov rax, [rsp+8]
    mov [r15+8], rax
    mov rax, [rsp+16]
    mov [r15+16], rax

    mov rax, rbp

    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
