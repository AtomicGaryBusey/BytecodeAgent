map_into:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    mov rbx, rdi          # dst
    mov r12, rsi          # src
    mov r13, rdx          # n
    mov r14, rcx          # f
    xor r15, r15          # i = 0
    xor rbp, rbp          # count = 0

.Lloop:
    cmp r15, r13
    jge .Ldone

    mov rdi, qword ptr [r12 + r15*8]
    call r14
    mov qword ptr [rbx + r15*8], rax

    mov rdx, qword ptr [r12 + r15*8]
    cmp rax, rdx
    setl cl
    movzx ecx, cl
    add rbp, rcx

    inc r15
    jmp .Lloop

.Ldone:
    mov rax, rbp

    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
