sdivrem:
    test rsi, rsi
    jz .Lzero
    mov rax, rdi
    mov r10, rdx
    mov r8, 0x8000000000000000
    cmp rdi, r8
    jne .Ldiv
    cmp rsi, -1
    jne .Ldiv
    mov qword ptr [r10], 0
    mov rax, r8
    ret
.Ldiv:
    cqo
    idiv rsi
    mov qword ptr [r10], rdx
    ret
.Lzero:
    mov qword ptr [rdx], rdi
    xor rax, rax
    ret
