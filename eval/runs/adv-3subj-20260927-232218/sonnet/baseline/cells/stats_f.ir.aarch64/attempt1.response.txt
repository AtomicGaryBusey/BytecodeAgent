export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%v0 =l loadl %a
	%y0 =l call %f(l %v0)
	%odd0 =l and %y0, 1
	%isodd0 =w cnel %odd0, 0
	%cnt0 =l extuw %isodd0
	jmp @loop
@loop
	%i =l phi @start 1, @next %i2
	%min =l phi @start %y0, @next %min2
	%max =l phi @start %y0, @next %max2
	%sum =l phi @start %y0, @next %sum2
	%cnt =l phi @start %cnt0, @next %cnt2
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%pa =l add %a, %off
	%v =l loadl %pa
	%y =l call %f(l %v)
	%sum2 =l add %sum, %y
	%odd =l and %y, 1
	%isodd =w cnel %odd, 0
	%addcnt =l extuw %isodd
	%cnt2 =l add %cnt, %addcnt
	%ltmin =w csltl %y, %min
	jnz %ltmin, @newmin, @afterminupdate
@newmin
	jmp @afterminupdate
@afterminupdate
	%min2 =l phi @body %min, @newmin %y
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @newmax, @next
@newmax
	jmp @next
@next
	%max2 =l phi @afterminupdate %max, @newmax %y
	%i2 =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%outp1 =l add %out, 8
	storel %max, %outp1
	%outp2 =l add %out, 16
	storel %sum, %outp2
	ret %cnt
}
