export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%count =l copy 0
	jmp @loop
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%sp =l add %src, %i
	%dp =l add %dst, %i
	%x =l loadl 8
	%sv =l mul %i, 8
	%sp2 =l add %src, %sv
	%dp2 =l add %dst, %sv
	%xv =l loadl %sp2
	%r =l call %f(l %xv)
	storel %r, %dp2
	%lt =w csltl %r, %xv
	jnz %lt, @inc, @noinc
@inc
	%count2 =l add %count, 1
	%count =l copy %count2
	jmp @noinc
@noinc
	%i2 =l add %i, 1
	%i =l copy %i2
	jmp @loop
@end
	ret %count
}
