export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%iszero =w ceql %b, 0
	jnz %iszero, @zero, @checkovf
@checkovf
	%isminb =w ceql %b, -1
	%isminaw =w ceql %a, -9223372036854775808
	%bothw =w and %isminb, %isminaw
	jnz %bothw, @overflow, @normal
@zero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret -9223372036854775808
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
