export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%iszero = ceql %b, 0
	jnz %iszero, @zero, @checkovf
@checkovf
	%isminb = ceql %b, -1
	%isminaw = ceql %a, -9223372036854775808
	%bothw = and %isminb, %isminaw
	jnz %bothw, @overflow, @normal
@zero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret -9223372036854775808
@normal
	%q = div %a, %b
	%r = rem %a, %b
	storel %r, %rem
	ret %q
}
