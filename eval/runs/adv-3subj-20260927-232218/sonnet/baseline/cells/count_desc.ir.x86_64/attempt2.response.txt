export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%cond0 =w csltl 0, %nm1
	jnz %cond0, @loop, @end
@loop
	%i =l phi @start 0, @next %i2
	%count =l phi @start 0, @next %count2
	%pi =l mul %i, 8
	%addr1 =l add %a, %pi
	%v1 =l loadl %addr1
	%ip1 =l add %i, 1
	%pi2 =l mul %ip1, 8
	%addr2 =l add %a, %pi2
	%v2 =l loadl %addr2
	%cmp =w csgtl %v1, %v2
	jnz %cmp, @inc, @next
@inc
	%countinc =l add %count, 1
	jmp @next
@next
	%count2 =l phi @loop %count, @inc %countinc
	%i2 =l add %i, 1
	%cond =w csltl %i2, %nm1
	jnz %cond, @loop, @end
@end
	%result =l phi @start 0, @next %count2
	ret %result
}
