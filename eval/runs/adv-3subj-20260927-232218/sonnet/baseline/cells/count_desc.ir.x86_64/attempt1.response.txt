export function l $count_desc(l %a, l %n) {
@start
	%count =l copy 0
	%i =l copy 0
	%nm1 =l sub %n, 1
	%cond0 =w csltl %i, %nm1
	jnz %cond0, %loop, @end
@loop
	%pi =l mul %i, 8
	%addr1 =l add %a, %pi
	%v1 =l loadl %addr1
	%ip1 =l add %i, 1
	%pi2 =l mul %ip1, 8
	%addr2 =l add %a, %pi2
	%v2 =l loadl %addr2
	%cmp =w csgtl %v1, %v2
	jnz %cmp, @inc, @next
@inc
	%count =l add %count, 1
	jmp @next
@next
	%i =l add %i, 1
	%cond =w csltl %i, %nm1
	jnz %cond, @loop, @end
@end
	ret %count
}
