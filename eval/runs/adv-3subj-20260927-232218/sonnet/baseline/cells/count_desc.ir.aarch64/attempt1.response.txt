export function l $count_desc(l %a, l %n) {
@start
	%count =l copy 0
	%i =l copy 0
	%nm1 =l sub %n, 1
	%cond0 =w csltl %i, %nm1
	jnz %cond0, @loop, @end
@loop
	%off1 =l mul %i, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%ip1 =l add %i, 1
	%off2 =l mul %ip1, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	%gt =w csgtl %v1, %v2
	jnz %gt, @inc, @next
@inc
	%count =l add %count, 1
	jmp @next
@next
	%i =l add %i, 1
	%cond =w csltl %i, %nm1
	jnz %cond, @loop, @end
@end
	ret %count
}
