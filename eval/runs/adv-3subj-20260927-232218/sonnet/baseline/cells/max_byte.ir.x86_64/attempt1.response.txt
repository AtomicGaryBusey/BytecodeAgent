export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l add %p, %i
	%b =l loadub %off
	%cmp =w csgtl %b, %max
	jnz %cmp, @update, @next
@update
	%max =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
