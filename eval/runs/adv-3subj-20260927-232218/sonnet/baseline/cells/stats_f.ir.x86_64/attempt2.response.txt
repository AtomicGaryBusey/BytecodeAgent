export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%p0 =l add %a, 0
	%x0 =l loadl %p0
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @merge %i2
	%min =l phi @start %y0, @merge %maxv2
	%max =l phi @start %y0, @merge %minv2
	%sum =l phi @start %y0, @merge %sum2
	%cnt =l phi @start %odd0, @merge %cnt2
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%ltmin =w csltl %y, %min
	jnz %ltmin, @newmin, @keepmin
@newmin
	jmp @afterMin
@keepmin
	jmp @afterMin
@afterMin
	%minv =l phi @newmin %y, @keepmin %min
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @newmax, @keepmax
@newmax
	jmp @merge
@keepmax
	jmp @merge
@merge
	%maxv2 =l phi @newmax %y, @keepmax %max
	%minv2 =l phi @afterMin %minv, @afterMin %minv
	%sum2 =l add %sum, %y
	%odd =l and %y, 1
	%cnt2 =l add %cnt, %odd
	%i2 =l add %i, 1
	jmp @loop
@done
	%po0 =l add %out, 0
	storel %min, %po0
	%po1 =l add %out, 8
	storel %max, %po1
	%po2 =l add %out, 16
	storel %sum, %po2
	ret %cnt
}
