export function l $shift_sat(l %x, l %n) {
@start
	%ge64 =w csgel %n, 64
	jnz %ge64, @big, @small
@small
	%left1 =l shl %x, %n
	%right1 =l sar %x, %n
	%sum1 =l add %left1, %right1
	jmp @end
@big
	%neg =w csltl %x, 0
	jnz %neg, @negcase, @poscase
@negcase
	%right2 =l copy -1
	jmp @join
@poscase
	%right2b =l copy 0
	jmp @join
@join
	%right3 =l phi @negcase %right2, @poscase %right2b
	%sum2 =l add 0, %right3
	jmp @end
@end
	%result =l phi @small %sum1, @join %sum2
	ret %result
}
