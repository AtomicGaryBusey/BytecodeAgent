export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%count =l copy 0
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%lt =w csltl %r, %x
	jnz %lt, @inc_count, @next
@inc_count
	%count =l add %count, 1
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %count
}
