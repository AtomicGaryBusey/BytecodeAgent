export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @latch %i2
	%min =l phi @start 0, @latch %min2
	%max =l phi @start 0, @latch %max2
	%sum =l phi @start 0, @latch %sum2
	%odd =l phi @start 0, @latch %odd2
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%idx8 =l mul %i, 8
	%p =l add %a, %idx8
	%x =l loadl %p
	%y =l call %f(l %x)
	%isfirst =w ceql %i, 0
	jnz %isfirst, @first, @notfirst
@first
	jmp @afterfirst
@notfirst
	%ltmin =w csltl %y, %min
	jnz %ltmin, @updmin, @keepmin
@updmin
	jmp @minmerge
@keepmin
	jmp @minmerge
@minmerge
	%minc =l phi @updmin %y, @keepmin %min
	%gtmax =w csgtl %y, %max
	jnz %gtmax, @updmax, @keepmax
@updmax
	jmp @maxmerge
@keepmax
	jmp @maxmerge
@maxmerge
	%maxc =l phi @updmax %y, @keepmax %max
	jmp @afterfirst
@afterfirst
	%min2 =l phi @first %y, @maxmerge %minc
	%max2 =l phi @first %y, @maxmerge %maxc
	%sum2 =l add %sum, %y
	%ybit =l and %y, 1
	%isodd =w cnel %ybit, 0
	jnz %isodd, @incodd, @noincodd
@incodd
	%odda =l add %odd, 1
	jmp @oddmerge
@noincodd
	jmp @oddmerge
@oddmerge
	%odd2 =l phi @incodd %odda, @noincodd %odd
	jmp @latch
@latch
	%i2 =l add %i, 1
	jmp @loop
@end
	%p1 =l add %out, 8
	%p2 =l add %out, 16
	storel %min, %out
	storel %max, %p1
	storel %sum, %p2
	ret %odd
}
