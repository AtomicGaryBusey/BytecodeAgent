export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%cnt =l copy 0
	jmp @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	jnz %gt, @inc, @next
@inc
	%cnt =l add %cnt, 1
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
