export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%o0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @maxmerge %inext
	%min =l phi @start %y0, @maxmerge %minu
	%max =l phi @start %y0, @maxmerge %maxu
	%sum =l phi @start %y0, @maxmerge %sumu
	%odd =l phi @start %o0, @maxmerge %oddu
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%xv =l loadl %addr
	%yv =l call %f(l %xv)
	%sumu =l add %sum, %yv
	%ybit =l and %yv, 1
	%oddu =l add %odd, %ybit
	%lt =w csltl %yv, %min
	jnz %lt, @newmin, @keepmin
@newmin
	jmp @minmerge
@keepmin
	jmp @minmerge
@minmerge
	%minu =l phi @newmin %yv, @keepmin %min
	%gt =w csgtl %yv, %max
	jnz %gt, @newmax, @keepmax
@newmax
	jmp @maxmerge
@keepmax
	jmp @maxmerge
@maxmerge
	%maxu =l phi @newmax %yv, @keepmax %max
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%outoff1 =l add %out, 8
	storel %max, %outoff1
	%outoff2 =l add %out, 16
	storel %sum, %outoff2
	ret %odd
}
