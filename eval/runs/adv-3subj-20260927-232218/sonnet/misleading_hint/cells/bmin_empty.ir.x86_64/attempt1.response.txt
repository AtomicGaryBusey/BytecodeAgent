export function l $bmin_empty(l %a, l %n) {
@start
	%isz =w ceql %n, 0
	jnz %isz, @zero, @init
@zero
	ret 9223372036854775807
@init
	%m =l loadl %a
	%i =l copy 1
	jmp @loop
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%lt =w csltl %v, %m
	jnz %lt, @upd, @next
@upd

	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %m
}
