@start
	%iszero =w ceql %b, 0
	jnz %iszero, @zero, @checkovf
@checkovf
	%aismin =w ceql %a, -9223372036854775808
	%bisneg1 =w ceql %b, -1
	%ovf =w and %aismin, %bisneg1
	jnz %ovf, @ovfcase, @normal
@normal

@zero
