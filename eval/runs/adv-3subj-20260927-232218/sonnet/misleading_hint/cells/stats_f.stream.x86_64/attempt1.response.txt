stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24

    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rcx

    xor r15, r15

    mov rdi, qword ptr [rbx]
    call r13
    mov rbp, rax
    mov qword ptr [rsp], rax
    mov qword ptr [rsp+8], rax
    mov qword ptr [rsp+16], 0
    test rax, 1
    jz .Lskip_odd0
    mov qword ptr [rsp+16], 1
.Lskip_odd0:
    inc r15

.Lloop:
    cmp r15, r12
    jge .Ldone
    mov rdi, qword ptr [rbx + r15*8]
    call r13
    cmp rax, rbp
    jge .Lnotmin
    mov rbp, rax
.Lnotmin:
    cmp rax, qword ptr [rsp]
    jle .Lnotmax
    mov qword ptr [rsp], rax
.Lnotmax:
    add qword ptr [rsp+8], rax
    test rax, 1
    jz .Lnotodd
    inc qword ptr [rsp+16]
.Lnotodd:
    inc r15
    jmp .Lloop

.Ldone:
    mov qword ptr [r14], rbp
    mov rax, qword ptr [rsp]
    mov qword ptr [r14+8], rax
    mov rax, qword ptr [rsp+8]
    mov qword ptr [r14+16], rax
    mov rax, qword ptr [rsp+16]

    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
