export function l $fill_down(l %a, l %n, l %start) {
@start
	%i =l copy 0
	%v =l copy %start
	%cond0 =w ceql %n, 0
	jnz %cond0, @end, @loop
@loop
	%p =l add %a, %i
	%p8 =l mul %i, 8
	%p =l add %a, %p8
	storel %v, %p

	%v =l sub %v, 1
	%cond =w csltl %i, %n

@end

}
