export function l $count_desc(l %a, l %n) {
@start
	%count =l copy 0
	%i =l copy 0
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%cond =w csltl %i, %nm1
	jnz %cond, @body, @end
@body
	%off0 =l mul %i, 8
	%p0 =l add %a, %off0
	%v0 =l loadl %p0
	%i1 =l add %i, 1
	%off1 =l mul %i1, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	jnz %gt, @inc, @next
@inc
	%count =l add %count, 1
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %count
}
