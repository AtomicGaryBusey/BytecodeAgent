    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    mov x19, x0        // dst
    mov x20, x1        // src
    mov x21, x2        // n
    mov x23, x3        // f
    mov x22, #0        // i
    mov x24, #0        // count
.Lloop:
    cmp x22, x21
    b.ge .Ldone
    ldr x0, [x20, x22, lsl #3]
    blr x23
    str x0, [x19, x22, lsl #3]
    ldr x9, [x20, x22, lsl #3]
    cmp x0, x9
    b.ge .Lskip
    add x24, x24, #1
.Lskip:
    add x22, x22, #1
    b .Lloop
.Ldone:
    mov x0, x24
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #64
