export function l $fill_down(l %a, l %n, l %start) {
@start
	%i =l copy 0
	%cond0 =w ceql %n, 0
	jnz %cond0, @end, @loop
@loop
	%off =l mul %i, 8
	%p =l add %a, %off
	%idx =l sub %start, %i
	storel %idx, %p
	%i1 =l add %i, 1
	%i =l copy %i1
	%cond =w csltl %i, %n
	jnz %cond, @loop, @end
@end
	ret %n
}
