map_into:
    cld
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov rbx, rdx
    mov r12, rdi
    mov r13, rsi
    mov r15, rcx
    xor r14, r14
    xor rbp, rbp
.Lloop:
    cmp r14, rbx
    jge .Ldone
    mov rdi, qword ptr [r13 + r14*8]
    call r15
    mov qword ptr [r12 + r14*8], rax
    cmp rax, qword ptr [r13 + r14*8]
    jge .Lskip
    inc rbp
.Lskip:
    inc r14
    jmp .Lloop
.Ldone:
    mov rax, rbp
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
