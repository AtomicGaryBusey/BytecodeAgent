export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	%r =l call %f(l %v)
	storel %r, %dp
	%lt =w csltl %r, %v
	%inc =l extsw %lt
	%cnt =l add %cnt, %inc
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
