export function l $count_desc(l %a, l %n) {
@start
	%count =l copy 0
	%i =l copy 0
	%nm1 =l sub %n, 1
	%cond0 =w cslel %n, 1
	jnz %cond0, @end, @loop
@loop
	%cmp =w csltl %i, %nm1
	jnz %cmp, @body, @end
@body
	%off1 =l mul %i, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%i2 =l add %i, 1
	%off2 =l mul %i2, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	%gt =w csgtl %v1, %v2
	jnz %gt, @inc, @next
@inc
	%count1 =l add %count, 1
	%count =l copy %count1
	jmp @next
@next
	%i =l copy %i2
	jmp @loop
@end
	ret %count
}
