sdivrem:
    test rsi, rsi
    je .Lzero
    mov rax, rdi
    mov r10, 0x8000000000000000
    cmp rax, r10
    jne .Ldiv
    cmp rsi, -1
    jne .Ldiv
    mov qword ptr [rdx], 0
    mov rax, r10
    ret
.Ldiv:
    cqo
    idiv rsi
    mov qword ptr [rdx], rdx
    ret
.Lzero:
    mov qword ptr [rdx], rdi
    xor rax, rax
    ret
