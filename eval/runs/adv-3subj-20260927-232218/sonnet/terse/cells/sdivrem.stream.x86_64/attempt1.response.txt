sdivrem:
    mov r8, rdx
    test rsi, rsi
    je .Lzero
    mov r9, 0x8000000000000000
    cmp rdi, r9
    jne .Lnormal
    cmp rsi, -1
    jne .Lnormal
    mov rax, rdi
    xor edx, edx
    mov qword ptr [r8], rdx
    ret
.Lnormal:
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [r8], rdx
    ret
.Lzero:
    xor eax, eax
    mov rdx, rdi
    mov qword ptr [r8], rdx
    ret
