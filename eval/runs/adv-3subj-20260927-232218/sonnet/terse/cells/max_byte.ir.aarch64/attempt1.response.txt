export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	jmp @test
@test
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%a =l add %p, %i
	%b =l loadub %a
	%g =w csgtl %b, %max
	jnz %g, @take, @next
@take
	%max =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	jmp @test
@end
	ret %max
}
