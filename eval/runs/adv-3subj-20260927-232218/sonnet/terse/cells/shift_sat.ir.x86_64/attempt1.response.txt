export function l $shift_sat(l %x, l %n) {
@start
    %big =w csgel %n, 64
    jnz %big, @big, @small
@small
    %shl =l shl %x, %n
    %shr =l sar %x, %n
    jmp @done
@big
    %sfill =l sar %x, 63
    jmp @done
@done
    %shlv =l phi @small %shl, @big 0
    %shrv =l phi @small %shr, @big %sfill
    %r =l add %shlv, %shrv
    ret %r
}
