export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%sp =l add %src, %i
	%dp =l add %dst, %i
	%x =l loadl 0, %sp
	%r =l call %f(l %x)
	storel %r, 0, %dp
	%lt =w csltl %r, %x
	%cnt =l add %cnt, %lt
	%i =l add %i, 8
	jmp @loop
@end
	ret %cnt
}
