function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%lim =l sub %n, 1
@loop
	%c1 =w csltl %i, %lim
	jnz %c1, @body, @end
@body
	%off =l mul %i, 8
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%p1 =l add %p0, 8
	%v1 =l loadl %p1
	%c2 =w csgtl %v0, %v1
	%c2l =l extuw %c2
	%cnt =l add %cnt, %c2l
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
