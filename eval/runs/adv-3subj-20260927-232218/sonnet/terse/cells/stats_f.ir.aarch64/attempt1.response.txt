export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%min =l copy 9223372036854775807
	%max =l copy -9223372036854775808
	%sum =l copy 0
	%cnt =l copy 0
	%i =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	%min =l copy %y
	jmp @chkmax
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @chkodd
@setmax
	%max =l copy %y
	jmp @chkodd
@chkodd
	%sum =l add %sum, %y
	%odd =l and %y, 1
	%isodd =w cnel %odd, 0
	jnz %isodd, @incodd, @next
@incodd
	%cnt =l add %cnt, 1
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
