export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%pi =l alloc8 8
	%pc =l alloc8 8
	storel 0, %pi
	storel 0, %pc
@loop
	%i =l loadl %pi
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%lt =w csltl %r, %x
	%ni =l add %i, 1
	storel %ni, %pi
	jnz %lt, @inc, @loop
@inc
	%c =l loadl %pc
	%nc =l add %c, 1
	storel %nc, %pc
	jmp @loop
@end
	%res =l loadl %pc
	ret %res
}
