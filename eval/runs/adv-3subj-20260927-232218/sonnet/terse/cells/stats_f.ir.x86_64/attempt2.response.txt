export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%cnt =l phi @start 0, @body %cnt2
	%sum =l phi @start 0, @body %sum2
	%min =l phi @start 9223372036854775807, @body %min2
	%max =l phi @start -9223372036854775808, @body %max2
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%idx =l mul %i, 8
	%p =l add %a, %idx
	%v =l loadl %p
	%y =l call %f(l %v)
	%sum2 =l add %sum, %y
	%odd =l and %y, 1
	%cnt2 =l add %cnt, %odd
	%ltw =w csltl %y, %min
	%maskmin =l extsw %ltw
	%diffmin =l sub %y, %min
	%termmin =l mul %diffmin, %maskmin
	%min2 =l add %min, %termmin
	%gtw =w csgtl %y, %max
	%maskmax =l extsw %gtw
	%diffmax =l sub %y, %max
	%termmax =l mul %diffmax, %maskmax
	%max2 =l add %max, %termmax
	%i2 =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
