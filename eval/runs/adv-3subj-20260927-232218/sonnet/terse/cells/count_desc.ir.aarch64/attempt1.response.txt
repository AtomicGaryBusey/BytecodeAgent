export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%go =w csgtl %nm1, 0
	jnz %go, @loop, @done
@loop
	%i0 =l copy 0
	%cnt0 =l copy 0
@body
	%i =l phi @loop %i0, @body %i1
	%cnt =l phi @loop %cnt0, @body %cnt2
	%off =l mul %i, 8
	%p =l add %a, %off
	%cur =l loadl %p
	%np =l add %p, 8
	%next =l loadl %np
	%gtw =w csgtl %cur, %next
	%gt =l extsw %gtw
	%cnt2 =l add %cnt, %gt
	%i1 =l add %i, 1
	%more =w csltl %i1, %nm1
	jnz %more, @body, @done
@done
	%res =l phi @start 0, @body %cnt2
	ret %res
}
