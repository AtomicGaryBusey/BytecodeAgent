map_into:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    mov rbx, rdi
    mov r12, rsi
    mov r14, rcx
    xor r15, r15
    test rdx, rdx
    jz .Ldone
    lea r13, [r12 + rdx*8]

.Lloop:
    mov rbp, qword ptr [r12]
    mov rdi, rbp
    call r14
    mov qword ptr [rbx], rax
    cmp rax, rbp
    setl al
    movzx eax, al
    add r15, rax
    add r12, 8
    add rbx, 8
    cmp r12, r13
    jne .Lloop

.Ldone:
    mov rax, r15
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
