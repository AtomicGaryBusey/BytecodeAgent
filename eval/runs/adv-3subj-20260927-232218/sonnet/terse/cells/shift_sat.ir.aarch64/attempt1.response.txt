export function l $shift_sat(l %x, l %n) {
@start
	%c =w csltl %n, 64
	jnz %c, @small, @large
@small
	%ls =l shl %x, %n
	%rs =l sar %x, %n
	jmp @done
@large
	%ls2 =l copy 0
	%rs2 =l sar %x, 63
	jmp @done
@done
	%l =l phi @small %ls, @large %ls2
	%r =l phi @small %rs, @large %rs2
	%res =l add %l, %r
	ret %res
}
