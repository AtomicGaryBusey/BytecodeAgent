stats_f:
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 32

    mov rbx, rdi
    lea r12, [rdi + rsi*8]
    mov r13, rdx
    mov r14, rcx
    xor r15d, r15d

    mov rax, 0x7fffffffffffffff
    mov [rsp], rax
    mov rax, 0x8000000000000000
    mov [rsp+8], rax
    mov qword ptr [rsp+16], 0

.Lloop:
    mov rdi, [rbx]
    call r13

    cmp rax, [rsp]
    jge .Lskipmin
    mov [rsp], rax
.Lskipmin:
    cmp rax, [rsp+8]
    jle .Lskipmax
    mov [rsp+8], rax
.Lskipmax:
    add [rsp+16], rax

    test al, 1
    jz .Lskipodd
    inc r15
.Lskipodd:

    add rbx, 8
    cmp rbx, r12
    jl .Lloop

    mov rax, [rsp]
    mov [r14], rax
    mov rax, [rsp+8]
    mov [r14+8], rax
    mov rax, [rsp+16]
    mov [r14+16], rax

    mov rax, r15

    add rsp, 32
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
