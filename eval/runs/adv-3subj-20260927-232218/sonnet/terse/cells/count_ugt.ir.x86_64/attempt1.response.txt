export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
@loop
	%i =l phi @start %i0, @inc %i1
	%c =l phi @start %c0, @inc %c1
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%adr =l add %a, %off
	%v =l loadl %adr
	%gt =w cugtl %v, %t
	%c2 =l add %c, %gt
	jnz %gt, @take, @skip
@take
	jmp @inc
@skip
	%c2 =l copy %c
	jmp @inc
@inc
	%i1 =l add %i, 1
	%c1 =l phi @take %c2, @skip %c2
	jmp @loop
@end
	ret %c
}
