sdivrem:
    test rsi, rsi
    jnz .Lnot_zero
    mov qword ptr [rdx], rdi
    xor rax, rax
    ret
.Lnot_zero:
    cmp rsi, -1
    jne .Ldiv_normal
    movabs rax, 0x8000000000000000
    cmp rdi, rax
    jne .Ldiv_normal
    mov qword ptr [rdx], 0
    ret
.Ldiv_normal:
    push rbx
    mov rbx, rdx
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rbx], rdx
    pop rbx
    ret
