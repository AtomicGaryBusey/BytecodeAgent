export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop

@loop
    %i = phi @entry 0, @body %i_next
    %cond = csltl %i, %n
    jnz %cond @body
    jmp @done

@body
    %value = subl %start, %i
    %offset = mull %i, 8
    %addr = addl %a, %offset
    storel %value, %addr
    %i_next = addl %i, 1
    jmp @loop

@done
    ret %n
}
