export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop

@loop
    %i =l phi @entry 0, @body %i_next
    %cond =w csltl %i, %n
    jnz %cond @body
    jmp @done

@body
    %value =l subl %start, %i
    %offset =l mull %i, 8
    %addr =l addl %a, %offset
    storel %value, %addr
    %i_next =l addl %i, 1
    jmp @loop

@done
    ret %n
}
