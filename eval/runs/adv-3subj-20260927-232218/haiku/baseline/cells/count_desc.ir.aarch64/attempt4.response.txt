export function l $count_desc(l %a, l %n) {
@start
    %is_small = sltl %n, 2
    jnz %is_small, @return_zero
    %count_init = addl 0, 0
    %i_init = addl 0, 0
    jmp @loop_cond

@loop_cond
    %count = phi @start %count_init, @loop_continue %count_new
    %i = phi @start %i_init, @loop_continue %i_new
    %n_minus_1 = subl %n, 1
    %cond = sltl %i, %n_minus_1
    jnz %cond, @loop_body
    ret %count

@loop_body
    %offset_i = mull %i, 8
    %addr_i = addl %a, %offset_i
    %ai = loadl %addr_i

    %i_plus_1 = addl %i, 1
    %offset_i_plus_1 = mull %i_plus_1, 8
    %addr_i_plus_1 = addl %a, %offset_i_plus_1
    %ai_plus_1 = loadl %addr_i_plus_1

    %is_greater = sgtl %ai, %ai_plus_1
    jnz %is_greater, @increment_count
    jmp @loop_continue

@increment_count
    %count_inc = addl %count, 1
    jmp @loop_continue

@loop_continue
    %count_new = phi @loop_body %count, @increment_count %count_inc
    %i_val = phi @loop_body %i, @increment_count %i
    %i_new = addl %i_val, 1
    jmp @loop_cond

@return_zero
    ret 0
}
