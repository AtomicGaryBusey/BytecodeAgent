@start
    %is_small = csltl %n, 2
    jnz %is_small, @return_zero
    
    %count_init = copy 0
    %i_init = copy 0
    jmp @loop_cond
    
@loop_cond
    %count = phi @start %count_init, @loop_update %count_new
    %i = phi @start %i_init, @loop_update %i_new
    %n_minus_1 = subl %n, 1
    %cond = csltl %i, %n_minus_1
    jnz %cond, @loop_body
    jmp @return_count
    
@loop_body
    %offset_i = mull %i, 8

    %ai = loadl %addr_i

@increment_count

    jmp @loop_update
    
@loop_update

@return_count
