export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %first =l loadl %a
    %y0 =l call %f(l %first)
    
    %min =l copy %y0
    %max =l copy %y0
    %sum =l copy %y0
    
    %y0_low =l and %y0, 1
    %y0_odd =w ceql %y0_low, 1
    %count =l zext %y0_odd
    
    %n_is_1 =w ceql %n, 1
    jnz %n_is_1, @done
    
    %i =l copy 1
    jmp @loop_check
    
@loop_check
    %loop_cond =w csltl %i, %n
    jnz %loop_cond, @loop_body
    jmp @done
    
@loop_body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %ai =l loadl %ptr
    
    %yi =l call %f(l %ai)
    
    %yi_low =l and %yi, 1
    %yi_odd =w ceql %yi_low, 1
    %yi_odd_ext =l zext %yi_odd
    %count =l add %count, %yi_odd_ext
    
    %yi_lt_min =w csltl %yi, %min
    jnz %yi_lt_min, @update_min
    jmp @after_min
    
@update_min
    %min =l copy %yi
    jmp @after_min
    
@after_min
    %yi_gt_max =w csgtl %yi, %max
    jnz %yi_gt_max, @update_max
    jmp @after_max
    
@update_max
    %max =l copy %yi
    jmp @after_max
    
@after_max
    %sum =l add %sum, %yi
    
    %i =l add %i, 1
    jmp @loop_check
    
@done
    storel %min, %out
    
    %out_8 =l add %out, 8
    storel %max, %out_8
    
    %out_16 =l add %out, 16
    storel %sum, %out_16
    
    ret %count
}
