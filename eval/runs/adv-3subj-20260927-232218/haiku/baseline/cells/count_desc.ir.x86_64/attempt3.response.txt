export function l $count_desc(l %a, l %n) {
@entry
    %c1 =l copy 1
    %c2 =l copy 2
    %c8 =l copy 8
    %is_small =w csltl %n, %c2
    jnz %is_small, @ret_zero, @setup

@ret_zero
    ret 0

@setup
    %i0 =l copy 0
    %count0 =l copy 0
    jmp @loop

@loop
    %i =l phi %i0, %i_new
    %count =l phi %count0, %count_new
    %limit =l subl %n, %c1
    %cond =w csltl %i, %limit
    jnz %cond, @body, @return

@body
    %off_i =l mull %i, %c8
    %addr_i =l addl %a, %off_i
    %val_i =l loadl %addr_i
    %i1 =l addl %i, %c1
    %off_i1 =l mull %i1, %c8
    %addr_i1 =l addl %a, %off_i1
    %val_i1 =l loadl %addr_i1
    %is_desc =w csgtl %val_i, %val_i1
    jnz %is_desc, @inc, @noinc

@inc
    %count_new_inc =l addl %count, %c1
    %i_new_inc =l addl %i, %c1
    jmp @merge

@noinc
    %count_new_noinc =l copy %count
    %i_new_noinc =l addl %i, %c1
    jmp @merge

@merge
    %count_new =l phi %count_new_inc, %count_new_noinc
    %i_new =l phi %i_new_inc, %i_new_noinc
    jmp @loop

@return
    ret %count
}
