export function l $count_desc(l %a, l %n) {
@entry
    %cmp = csltl %n, 2
    jnz %cmp, @return_zero, @loop_init

@return_zero
    ret 0

@loop_init
    jmp @loop_check(0, 0)

@loop_check(l %count, l %i)
    %limit = subl %n, 1
    %continue = csltl %i, %limit
    jnz %continue, @loop_body, @return_count

@loop_body
    %offset_i = mull %i, 8
    %addr_i = addl %a, %offset_i
    %a_i = loadl %addr_i
    %i_plus_1 = addl %i, 1
    %offset_i1 = mull %i_plus_1, 8
    %addr_i1 = addl %a, %offset_i1
    %a_i1 = loadl %addr_i1
    %is_greater = csgtl %a_i, %a_i1
    %new_i = addl %i, 1
    jnz %is_greater, @increment, @no_increment

@increment
    %new_count = addl %count, 1
    jmp @loop_check(%new_count, %new_i)

@no_increment
    jmp @loop_check(%count, %new_i)

@return_count
    ret %count
}
