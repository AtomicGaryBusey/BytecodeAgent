export function l $count_desc(l %a, l %n) {
@entry
    %c1 = copy 1
    %c2 = copy 2
    %c8 = copy 8
    %is_small = csltl %n, %c2
    jnz %is_small, @ret_zero, @setup

@ret_zero
    ret 0

@setup
    %i0 = copy 0
    %count0 = copy 0
    jmp @loop

@loop
    %i = phi %i0, %i_new
    %count = phi %count0, %count_new
    %limit = subl %n, %c1
    %cond = csltl %i, %limit
    jnz %cond, @body, @return

@body
    %off_i = mull %i, %c8
    %addr_i = addl %a, %off_i
    %val_i = loadl %addr_i
    %i1 = addl %i, %c1
    %off_i1 = mull %i1, %c8
    %addr_i1 = addl %a, %off_i1
    %val_i1 = loadl %addr_i1
    %is_desc = csgtl %val_i, %val_i1
    jnz %is_desc, @inc, @noinc

@inc
    %count_new_inc = addl %count, %c1
    %i_new_inc = addl %i, %c1
    jmp @merge

@noinc
    %count_new_noinc = copy %count
    %i_new_noinc = addl %i, %c1
    jmp @merge

@merge
    %count_new = phi %count_new_inc, %count_new_noinc
    %i_new = phi %i_new_inc, %i_new_noinc
    jmp @loop

@return
    ret %count
}
