export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop

@loop
    %i = phi @entry $0l, @merge %i_next
    %count = phi @entry $0l, @merge %count_next
    %cond = cultl %i, %n
    jnz %cond @body, @end

@body
    %offset = mul %i, $8l
    %addr = add %a, %offset
    %elem = loadl %addr
    %gt = cugtl %elem, %t
    jnz %gt @inc, @noinc

@inc
    %count_plus_1 = add %count, $1l
    jmp @merge

@noinc
    jmp @merge

@merge
    %count_next = phi @inc %count_plus_1, @noinc %count
    %i_next = add %i, $1l
    jmp @loop

@end
    ret %count
}
