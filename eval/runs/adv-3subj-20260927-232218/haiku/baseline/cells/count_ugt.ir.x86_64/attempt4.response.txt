export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop

@loop
    %i = phi @entry $0, @merge %i_next
    %count = phi @entry $0, @merge %count_next
    %cond = cultl %i, %n
    jnz %cond @body, @end

@body
    %offset = mull %i, $8
    %addr = addl %a, %offset
    %elem = loadl %addr
    %gt = cugtl %elem, %t
    jnz %gt @inc, @noinc

@inc
    %count_plus_1 = addl %count, $1
    jmp @merge

@noinc
    jmp @merge

@merge
    %count_next = phi @inc %count_plus_1, @noinc %count
    %i_next = addl %i, $1
    jmp @loop

@end
    ret %count
}
