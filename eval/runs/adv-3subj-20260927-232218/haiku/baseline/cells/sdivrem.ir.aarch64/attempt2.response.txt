export function l $sdivrem(l %a, l %b, l %rem) {
@entry
    %b_is_zero =w ceql %b, 0
    jnz %b_is_zero @div_by_zero
    
    %a_is_min =w ceql %a, -9223372036854775808
    jz %a_is_min @normal_div
    
    %b_is_neg_one =w ceql %b, -1
    jnz %b_is_neg_one @min_overflow
    jmp @normal_div
    
@normal_div
    %q =l divl %a, %b
    %r =l reml %a, %b
    storel %r, %rem
    ret %q
    
@min_overflow
    storel 0, %rem
    ret -9223372036854775808
    
@div_by_zero
    storel %a, %rem
    ret 0
}
