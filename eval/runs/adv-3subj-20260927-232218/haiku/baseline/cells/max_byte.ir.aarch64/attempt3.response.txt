export function l $max_byte(l %p, l %n) {
@start
%cond =w ceql %n, 0
jnz %cond, @ret_zero
%first =l loadub %p
jmp @loop_header
@loop_header
%i =l phi 1 @start, %next_i @loop_foot
%max =l phi %first @start, %max_next @loop_foot
%cont =w cultl %i, %n
jnz %cont, @loop_body
ret %max
@loop_body
%ptr =l add %p, %i
%byte =l loadub %ptr
%gt =w cugtl %byte, %max
jnz %gt, @set_max
jmp @loop_foot
@set_max
jmp @loop_foot
@loop_foot
%max_next =l phi %byte @set_max, %max @loop_body
%next_i =l add %i, 1
jmp @loop_header
@ret_zero
ret 0
}
