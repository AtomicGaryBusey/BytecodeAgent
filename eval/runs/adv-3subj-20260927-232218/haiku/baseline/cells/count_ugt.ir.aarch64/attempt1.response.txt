export function l $count_ugt(l %a, l %n, l %t) {
@start
	%count = copy 0
	%i = copy 0
	jmp @loop

@loop
	%cond = csltl %i, %n
	jnz %cond, @body, @end

@body
	%offset = mul %i, 8
	%addr = add %a, %offset
	%elem = loadl %addr
	%gt = cugtl %elem, %t
	jnz %gt, @incr, @next

@incr
	%count = add %count, 1
	jmp @next

@next
	%i = add %i, 1
	jmp @loop

@end
	ret %count
}
