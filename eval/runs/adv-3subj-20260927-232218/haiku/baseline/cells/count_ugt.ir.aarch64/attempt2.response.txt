export function l $count_ugt(l %a, l %n, l %t) {
@start
	%zero = const l 0
	%one = const l 1
	%eight = const l 8
	jmp @loop

@loop
	%count = phi @start %zero, @skip %count_next
	%i = phi @start %zero, @skip %i_next
	%cond = csltl %i, %n
	jnz %cond, @body, @end

@body
	%offset = mul %i, %eight
	%addr = add %a, %offset
	%elem = loadl %addr
	%gt = cugtl %elem, %t
	jnz %gt, @incr, @skip

@incr
	%count_incr = add %count, %one
	jmp @skip

@skip
	%count_next = phi @body %count, @incr %count_incr
	%i_next = add %i, %one
	jmp @loop

@end
	ret %count
}
