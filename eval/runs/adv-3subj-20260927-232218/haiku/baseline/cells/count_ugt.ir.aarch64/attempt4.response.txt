export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%zero = copy 0
	jmp @loop

@loop
	%count = phi @entry %zero, @skip %count_next
	%i = phi @entry %zero, @skip %i_next
	%cond = csltl %i, %n
	jnz %cond, @body, @end

@body
	%offset = mul %i, 8
	%addr = add %a, %offset
	%elem = loadl %addr
	%gt = cugtl %elem, %t
	jnz %gt, @incr, @skip

@incr
	%count_incr = add %count, 1
	jmp @skip

@skip
	%count_next = phi @body %count, @incr %count_incr
	%i_next = add %i, 1
	jmp @loop

@end
	ret %count
}
