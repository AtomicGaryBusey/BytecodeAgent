export function l $bmin_empty(l %a, l %n) {
@start
    %is_empty = ceql %n 0l
    jnz %is_empty @empty
    %first = loadl %a
    %idx0 = 1l
    jmp @loop_cond

@loop_cond
    %idx = phi %idx0, %idx_new
    %min = phi %first, %min_new
    %cond = csltl %idx %n
    jnz %cond @loop_body
    jmp @done

@loop_body
    %offset = mull %idx 8l
    %addr = addl %a %offset
    %val = loadl %addr
    %smaller = csltl %val %min
    jnz %smaller @update_min
    jmp @no_update

@update_min
    %min_new = %val
    jmp @next_iter

@no_update
    %min_new = %min
    jmp @next_iter

@next_iter
    %idx_new = addl %idx 1l
    jmp @loop_cond

@done
    ret %min

@empty
    ret 9223372036854775807l
}
