export function l $map_into(l %dst, l %src, l %n, l %f) {
    %junk =l alloc8 8
    @start
        %count =l copy 0
        %i =l copy 0
        jmp @loop_cond

    @loop_body
        %offset =l mul %i, 8
        %src_addr =l add %src, %offset
        %src_val =l loadl %src_addr
        %dst_val =l call %f(l %src_val)
        %dst_addr =l add %dst, %offset
        storel %dst_addr, %dst_val
        %cmp =w csltl %dst_val, %src_val
        jnz %cmp, @increment_count, @next_iter

    @increment_count
        %count =l add %count, 1
        jmp @next_iter

    @next_iter
        %i =l add %i, 1
        jmp @loop_cond

    @loop_cond
        %continue =w csltl %i, %n
        jnz %continue, @loop_body, @end

    @end
        ret %count
}
