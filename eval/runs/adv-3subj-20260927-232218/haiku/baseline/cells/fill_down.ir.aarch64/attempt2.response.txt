export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop_check
    
@loop_check
    %i = phi l 0, %next_i
    %cond = csltl %i %n
    jnz %cond @loop_body @return
    
@loop_body
    %offset = mull 8 %i
    %addr = addl %a %offset
    %val = subl %start %i
    storel %val, %addr
    %next_i = addl %i 1
    jmp @loop_check
    
@return
    ret %n
}
