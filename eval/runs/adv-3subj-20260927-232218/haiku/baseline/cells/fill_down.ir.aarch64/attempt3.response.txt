export function l $fill_down(l %a, l %n, l %start) {
@entry
    %zero = copy l 0
    jmp @loop_check
    
@loop_check
    %i = phi l %zero %next_i
    %cond = csltl %i %n
    jnz w %cond @loop_body @return
    
@loop_body
    %offset = mull %i 8
    %addr = addl %a %offset
    %val = subl %start %i
    storel %val %addr
    %next_i = addl %i 1
    jmp @loop_check
    
@return
    ret %n
}
