bmin_empty:
    test rsi, rsi
    jz .Lempty
    mov rax, qword ptr [rdi]
    cmp rsi, 1
    je .Ldone
    mov rcx, 1
.Lloop:
    mov r8, qword ptr [rdi + rcx*8]
    cmp r8, rax
    cmovl rax, r8
    inc rcx
    cmp rcx, rsi
    jl .Lloop
.Ldone:
    ret
.Lempty:
    mov rax, 0x7FFFFFFFFFFFFFFF
    ret
