export function l $map_into(l %dst, l %src, l %n, l %f) {
@init
    %count =l copy 0
    %i =l copy 0
    jmp @loop_cond

@loop_cond
    %count_loop =l phi @init %count, @loop_end %count_next
    %i_loop =l phi @init %i, @loop_end %i_next
    
    %cond =w csltl %i_loop, %n
    jnz %cond, @loop_body, @done

@loop_body
    %offset =l mul %i_loop, 8
    %src_addr =l add %src, %offset
    %src_val =l loadl %src_addr
    
    %dst_val =l call %f(l %src_val)
    
    %dst_addr =l add %dst, %offset
    storel %dst_val, %dst_addr
    
    %cmp =w csltl %dst_val, %src_val
    jnz %cmp, @inc_count, @no_inc

@inc_count
    %count_inc =l add %count_loop, 1
    jmp @loop_end

@no_inc
    jmp @loop_end

@loop_end
    %count_next =l phi @inc_count %count_inc, @no_inc %count_loop
    %i_next =l add %i_loop, 1
    jmp @loop_cond

@done
    ret %count_loop
}
