export function l $shift_sat(l %x, l %n) {
@start
    %cmp =w csgtl %n, 63
    jnz %cmp, @large, @normal

@normal
    %left =l shl %x, %n
    %right =l sar %x, %n
    %result =l add %left, %right
    ret %result

@large
    %sign =w csltl %x, 0
    jnz %sign, @x_neg, @x_pos

@x_pos
    ret 0

@x_neg
    ret -1
}
