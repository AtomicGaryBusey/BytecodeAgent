export function l $max_byte(l %p, l %n) {
@entry
    %n_gt =w csgtl %n, 0
    jnz %n_gt, @loop, @return_zero
    
@loop
    %max =l phi @entry 0, @body_end %max_new
    %i =l phi @entry 0, @body_end %i_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @return_max
    
@body
    %addr =l add %p, %i
    %byte =l loadub %addr
    %greater =w cugtl %byte, %max
    %i_next =l add %i, 1
    jnz %greater, @update, @skip
    
@update
    jmp @body_end
    
@skip
    jmp @body_end
    
@body_end
    %max_new =l phi @update %byte, @skip %max
    jmp @loop
    
@return_max
    ret %max
    
@return_zero
    ret 0
}
