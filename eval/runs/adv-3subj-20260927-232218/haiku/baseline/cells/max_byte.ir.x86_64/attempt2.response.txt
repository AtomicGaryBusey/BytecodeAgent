export function l $max_byte(l %p, l %n) {
@entry
    %n_gt =w csgtl %n, 0
    jnz %n_gt, @start, @return_zero
    
@start
    %max_init =l copy 0
    %i_init =l copy 0
    jmp @loop
    
@loop
    %max =l phi @start %max_init, @iter_end %max_next
    %i =l phi @start %i_init, @iter_end %i_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @return_max
    
@body
    %addr =l add %p, %i
    %byte =l loadub %addr
    %greater =w cugtl %byte, %max
    jnz %greater, @update, @skip_update
    
@update
    %max_next =l copy %byte
    jmp @iter_end
    
@skip_update
    %max_next =l copy %max
    jmp @iter_end
    
@iter_end
    %i_next =l add %i, 1
    jmp @loop
    
@return_max
    ret %max
    
@return_zero
    ret 0
}
