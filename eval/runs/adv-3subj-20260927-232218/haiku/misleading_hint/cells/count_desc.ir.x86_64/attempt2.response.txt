function l $count_desc(l %a, l %n) {
@entry
    %cond = csltl %n, 2
    jnz %cond, @ret0
    jmp @loop

@loop
    %count = phi l 0, @entry, %count_new, @loop_cont
    %i = phi l 0, @entry, %i_next, @loop_cont
    %i_plus_one = add l %i, 1
    %done = csgeql %i_plus_one, %n
    jnz %done, @ret_count
    jmp @body

@body
    %off_i = mul l %i, 8
    %ptr_i = add l %a, %off_i
    %v_i = loadl %ptr_i
    %off_i1 = mul l %i_plus_one, 8
    %ptr_i1 = add l %a, %off_i1
    %v_i1 = loadl %ptr_i1
    %cmp = csgtl %v_i, %v_i1
    jnz %cmp, @inc_cnt
    jmp @no_inc

@inc_cnt
    %count_inc = add l %count, 1
    jmp @loop_cont

@no_inc
    jmp @loop_cont

@loop_cont
    %count_new = phi l %count_inc, @inc_cnt, %count, @no_inc
    %i_next = add l %i, 1
    jmp @loop

@ret_count
    ret %count

@ret0
    %zero = const l 0
    ret %zero
}
