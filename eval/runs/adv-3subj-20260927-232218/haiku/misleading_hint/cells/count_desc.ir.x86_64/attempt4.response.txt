export function l $count_desc(l %a, l %n) {
@entry
    %zero = const l 0
    %one = const l 1
    %two = const l 2
    %eight = const l 8
    %cond = csltl %n, %two
    jnz %cond, @ret0
    jmp @loop

@loop
    %count = phi l %zero @entry, %count_new @loop_cont
    %i = phi l %zero @entry, %i_next @loop_cont
    %i_plus_one = add l %i, %one
    %done = csgel %i_plus_one, %n
    jnz %done, @ret_count
    jmp @body

@body
    %off_i = mul l %i, %eight
    %ptr_i = add l %a, %off_i
    %v_i = loadl %ptr_i
    %off_i1 = mul l %i_plus_one, %eight
    %ptr_i1 = add l %a, %off_i1
    %v_i1 = loadl %ptr_i1
    %cmp = csgtl %v_i, %v_i1
    jnz %cmp, @inc_cnt
    jmp @no_inc

@inc_cnt
    %count_inc = add l %count, %one
    jmp @loop_cont

@no_inc
    jmp @loop_cont

@loop_cont
    %count_new = phi l %count_inc @inc_cnt, %count @no_inc
    %i_next = add l %i, %one
    jmp @loop

@ret_count
    ret %count

@ret0
    ret %zero
}
