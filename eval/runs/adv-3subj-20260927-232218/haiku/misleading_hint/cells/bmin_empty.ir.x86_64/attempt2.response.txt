export function l $bmin_empty(l %a, l %n) {
@entry
    %first = loadl %a
    %one = const l 1
    jmp @cond

@cond
    %min = phi %first @entry, %min_next @cont
    %i = phi %one @entry, %i_next @cont
    
    %lt = cltl %i, %n
    jnz %lt, @loop
    ret %min

@loop
    %eight = const l 8
    %offset = mull %i, %eight
    %addr = addl %a, %offset
    %elem = loadl %addr
    %less = cltl %elem, %min
    jnz %less, @update

@cont
    %min_next = phi %elem @update, %min @loop
    %i_next = addl %i, %one
    jmp @cond

@update
    jmp @cont
}
