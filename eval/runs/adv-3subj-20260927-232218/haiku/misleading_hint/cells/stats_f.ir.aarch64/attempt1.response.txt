export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    
    %m0 =l copy %y0
    %M0 =l copy %y0
    %s0 =l copy %y0
    %odd0_l =l and %y0, 1
    %odd0 =w ceql %odd0_l, 1
    %c0 =l extsw %odd0
    
    jmp @loop_header(%m0, %M0, %s0, %c0, 1)

@loop_header(l %min, l %max, l %sum, l %oddcount, l %i)
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @loop_exit(%min, %max, %sum, %oddcount)

@loop_body
    %offset =l mul %i, 8
    %ai_addr =l add %a, %offset
    %ai =l loadl %ai_addr
    %yi =l call %f(l %ai)
    
    %cmp_min =w csltl %yi, %min
    jnz %cmp_min, @update_min, @check_max(%yi, %min, %max, %sum, %oddcount, %i)
    
@update_min
    %min_new =l copy %yi
    jmp @check_max(%yi, %min_new, %max, %sum, %oddcount, %i)

@check_max(l %yi, l %min_now, l %max, l %sum, l %oddcount, l %i)
    %cmp_max =w csgtl %yi, %max
    jnz %cmp_max, @update_max, @update_sum(%yi, %min_now, %max, %sum, %oddcount, %i)
    
@update_max
    %max_new =l copy %yi
    jmp @update_sum(%yi, %min_now, %max_new, %sum, %oddcount, %i)

@update_sum(l %yi, l %min_now, l %max_now, l %sum, l %oddcount, l %i)
    %sum_new =l add %sum, %yi
    
    %odd_l =l and %yi, 1
    %odd_w =w ceql %odd_l, 1
    %odd_inc =l extsw %odd_w
    %oddcount_new =l add %oddcount, %odd_inc
    
    %i_next =l add %i, 1
    
    jmp @loop_header(%min_now, %max_now, %sum_new, %oddcount_new, %i_next)

@loop_exit(l %min, l %max, l %sum, l %oddcount)
    storel %min, %out
    %addr_max =l add %out, 8
    storel %max, %addr_max
    %addr_sum =l add %out, 16
    storel %sum, %addr_sum
    
    ret %oddcount
}
