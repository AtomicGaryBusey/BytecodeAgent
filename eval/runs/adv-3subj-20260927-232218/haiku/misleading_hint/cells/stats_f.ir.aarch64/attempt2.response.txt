export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    
    %m0 =l copy %y0
    %M0 =l copy %y0
    %s0 =l copy %y0
    %odd0_l =l and %y0, 1
    %odd0 =w ceql %odd0_l, 1
    %c0 =l extsw %odd0
    %i0 =l copy 1
    
    jmp @loop_header

@loop_header
    %min =l phi @entry %m0, @finish_iter %min_new
    %max =l phi @entry %M0, @finish_iter %max_new
    %sum =l phi @entry %s0, @finish_iter %sum_new
    %oddcount =l phi @entry %c0, @finish_iter %oddcount_new
    %i =l phi @entry %i0, @finish_iter %i_new
    
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @loop_exit

@loop_body
    %offset =l mul %i, 8
    %ai_addr =l add %a, %offset
    %ai =l loadl %ai_addr
    %yi =l call %f(l %ai)
    
    %cmp_min =w csltl %yi, %min
    jnz %cmp_min, @update_min, @keep_min

@update_min
    %min_cand =l copy %yi
    jmp @check_max

@keep_min
    %min_cand =l copy %min
    jmp @check_max

@check_max
    %min_tmp =l phi @update_min %min_cand, @keep_min %min_cand
    %cmp_max =w csgtl %yi, %max
    jnz %cmp_max, @update_max, @keep_max

@update_max
    %max_cand =l copy %yi
    jmp @finish_iter

@keep_max
    %max_cand =l copy %max
    jmp @finish_iter

@finish_iter
    %max_tmp =l phi @update_max %max_cand, @keep_max %max_cand
    %sum_new =l add %sum, %yi
    
    %odd_l =l and %yi, 1
    %odd_w =w ceql %odd_l, 1
    %odd_inc =l extsw %odd_w
    %oddcount_new =l add %oddcount, %odd_inc
    
    %i_new =l add %i, 1
    %min_new =l copy %min_tmp
    %max_new =l copy %max_tmp
    
    jmp @loop_header

@loop_exit
    storel %min, %out
    %addr_max =l add %out, 8
    storel %max, %addr_max
    %addr_sum =l add %out, 16
    storel %sum, %addr_sum
    
    ret %oddcount
}
