export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %min0 =l copy %y0
    %max0 =l copy %y0
    %sum0 =l copy %y0
    %odd0 =l and %y0, 1
    %init_i =l copy 1
    jmp @loop_check

@loop_check
    %i =l phi @start %init_i, @loop_end %i_next
    %min =l phi @start %min0, @loop_end %new_min
    %max =l phi @start %max0, @loop_end %new_max
    %sum =l phi @start %sum0, @loop_end %new_sum
    %count =l phi @start %odd0, @loop_end %new_count
    %cond =w csltl %i, %n
    jnz %cond @loop_body
    jmp @return

@loop_body
    %idx =l mul %i, 8
    %ptr =l add %a, %idx
    %elem =l loadl %ptr
    %y =l call %f(l %elem)
    %lt =w csltl %y, %min
    jnz %lt @min_update
    jmp @min_done

@min_update
    jmp @min_done

@min_done
    %new_min =l phi @loop_body %min, @min_update %y
    %gt =w csgtl %y, %max
    jnz %gt @max_update
    jmp @max_done

@max_update
    jmp @max_done

@max_done
    %new_max =l phi @min_done %max, @max_update %y
    %new_sum =l add %sum, %y
    %y_and =l and %y, 1
    %new_count =l add %count, %y_and
    jmp @loop_end

@loop_end
    %i_next =l add %i, 1
    jmp @loop_check

@return
    storel %min, %out
    %out_ptr_1 =l add %out, 8
    storel %max, %out_ptr_1
    %out_ptr_2 =l add %out, 16
    storel %sum, %out_ptr_2
    ret %count
}
