export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %min0 =l copy %y0
    %max0 =l copy %y0
    %sum0 =l copy %y0
    %odd0 =l and %y0, 1
    %i_init =l copy 1
    jmp @loop_check

@loop_check
    %i =l phi @start %i_init, @skip_max_update %i_next
    %min =l phi @start %min0, @skip_max_update %final_min
    %max =l phi @start %max0, @skip_max_update %final_max
    %sum =l phi @start %sum0, @skip_max_update %new_sum
    %count =l phi @start %odd0, @skip_max_update %new_count
    %cond =w csltl %i, %n
    jnz %cond @loop_body
    jmp @return

@loop_body
    %idx =l mul %i, 8
    %ptr =l add %a, %idx
    %elem =l loadl %ptr
    %y =l call %f(l %elem)
    
    %lt =w csltl %y, %min
    %gt =w csgtl %y, %max
    %is_odd =l and %y, 1
    %new_sum =l add %sum, %y
    %new_count =l add %count, %is_odd
    
    %new_min =l copy %min
    jnz %lt @update_min
    jmp @skip_min_update

@update_min
    %new_min =l copy %y
    jmp @skip_min_update

@skip_min_update
    %final_min =l phi @update_min %new_min, @loop_body %new_min
    
    %new_max =l copy %max
    jnz %gt @update_max
    jmp @skip_max_update

@update_max
    %new_max =l copy %y
    jmp @skip_max_update

@skip_max_update
    %final_max =l phi @update_max %new_max, @loop_body %new_max
    %i_next =l add %i, 1
    jmp @loop_check

@return
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %count
}
