export function l $max_byte(l %p, l %n) {
@entry
    %cmp =w csgtl %n, 0
    jnz %cmp, @init, @ret0

@ret0
    ret 0

@init
    %i =l copy 0
    %m =l copy 0
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @ret

@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %gt =w csgtl %b, %m
    jnz %gt, @upd, @next

@upd
    %m =l copy %b
    jmp @next

@next
    %i =l add %i, 1
    jmp @loop

@ret
    ret %m
}
