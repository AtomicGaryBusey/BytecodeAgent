export function l $max_byte(l %p, l %n)
{
@entry
    %idx =l sub %n %n
    %max =l sub %n %n
    jmp @loop

@loop
    %more_to_do =w csgtl %n %idx
    jnz %more_to_do @process
    jmp @return

@process
    %addr =l add %p %idx
    %byte =l loadub %addr
    
    %is_gt =w csgtl %byte %max
    jnz %is_gt @update_max
    jmp @next_iter

@update_max
    %max =l add %byte 0
    jmp @next_iter

@next_iter
    %idx =l add %idx 1
    jmp @loop

@return
    ret %max
}
