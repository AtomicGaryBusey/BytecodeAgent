export function l $max_byte(l %p, l %n) {
@entry
    %idx =l add 0 0
    %max =l add 0 0
    jmp @loop

@loop
    %cond =w csgtl %n %idx
    jnz %cond @body
    jmp @return

@body
    %addr =l add %p %idx
    %byte =l loadub %addr
    %cmp =w csgtl %byte %max
    jnz %cmp @update
    jmp @next

@update
    %max =l add %byte 0
    jmp @next

@next
    %idx =l add %idx 1
    jmp @loop

@return
    ret %max
}
