export function l $max_byte(l %p, l %n)
{
@entry
    %cond =w csltl %n 1
    jnz %cond @ret_zero
    
    %idx =l copy 0
    %max =l copy 0
    jmp @loop
    
@loop
    %done =w csgel %idx %n
    jnz %done @return
    
    %addr =l add %p %idx
    %byte =l loadub %addr
    
    %is_gt =w csgtl %byte %max
    %max =l sell %is_gt %byte %max
    
    %idx =l add %idx 1
    
    jmp @loop
    
@ret_zero
    ret 0
    
@return
    ret %max
}
