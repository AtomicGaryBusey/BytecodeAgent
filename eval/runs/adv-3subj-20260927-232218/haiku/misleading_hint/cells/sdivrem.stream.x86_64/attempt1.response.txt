sdivrem:
    mov r8, rdx
    test rsi, rsi
    jz .Ldiv_by_zero
    mov rax, rdi
    mov r9, 0x8000000000000000
    cmp rax, r9
    jne .Lnormal_div
    cmp rsi, -1
    jne .Lnormal_div
    mov qword ptr [r8], 0
    mov rax, r9
    ret
.Ldiv_by_zero:
    mov qword ptr [r8], rdi
    xor rax, rax
    ret
.Lnormal_div:
    cqo
    idiv rsi
    mov qword ptr [r8], rdx
    ret
