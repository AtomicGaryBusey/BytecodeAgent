export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop

@loop
    %count = phi l 0 @entry %count_next @loopend
    %i = phi l 0 @entry %i_next @loopend
    %cond = csltl %i %n
    jnz %cond @body
    ret %count

@body
    %offset = mull %i 8
    %addr = addl %a %offset
    %elem = loadl %addr
    %is_gt = cultl %t %elem
    jnz %is_gt @inc_count
    jmp @loopend

@inc_count
    %count_inc = addl %count 1
    jmp @loopend

@loopend
    %count_next = phi l %count @body %count_inc @inc_count
    %i_next = addl %i 1
    jmp @loop
}
