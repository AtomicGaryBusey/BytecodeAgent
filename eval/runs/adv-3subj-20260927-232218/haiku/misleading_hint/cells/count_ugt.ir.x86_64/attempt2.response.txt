export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop

@loop
    %count = phi l 0, %count_out
    %i = phi l 0, %i_out
    %cond = csltl %i %n
    jnz %cond @body
    ret %count

@body
    %offset = mull %i 8
    %addr = addl %a %offset
    %elem = loadl %addr
    %is_gt = cultl %t %elem
    jnz %is_gt @inc
    jmp @next

@inc
    %count_new = addl %count 1
    jmp @next

@next
    %count_out = phi l %count, %count_new
    %i_out = addl %i 1
    jmp @loop
}
