export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop(0, 0)

@loop(l %count, l %i)
    %cond = csltl %i %n
    jnz %cond @body(%count, %i)
    ret %count

@body(l %count, l %i)
    %offset = mull %i 8
    %addr = addl %a %offset
    %elem = loadl %addr
    %is_gt = cultl %t %elem
    jnz %is_gt @inc(%count, %i)
    jmp @next(%count, %i)

@inc(l %count, l %i)
    %new_count = addl %count 1
    jmp @next(%new_count, %i)

@next(l %count, l %i)
    %new_i = addl %i 1
    jmp @loop(%count, %new_i)
}
