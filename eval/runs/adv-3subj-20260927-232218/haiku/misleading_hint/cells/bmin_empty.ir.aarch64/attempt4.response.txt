@loop_cond

    %i = phi @start 1, @merge %i_new

@body

    %addr = add %a, %off
    %val = loadl %addr
    %test = csltl %val, %m
    jnz %test, @update, @skip

@update

@skip

@merge

    jmp @loop_cond

@done

}
