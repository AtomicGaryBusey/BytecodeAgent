export function l $sdivrem(l %a, l %b, l %rem_ptr) {
@entry
    %b_is_zero =w ceql %b, 0
    jnz %b_is_zero, @div_by_zero
    
    %a_is_min =w ceql %a, -9223372036854775808
    jnz %a_is_min, @check_overflow
    jmp @normal_case

@check_overflow
    %b_is_neg_one =w ceql %b, -1
    jnz %b_is_neg_one, @overflow_case
    jmp @normal_case

@normal_case
    %q =l divl %a, %b
    %r =l reml %a, %b
    storel %r, %rem_ptr
    ret %q

@overflow_case
    storel 0, %rem_ptr
    ret %a

@div_by_zero
    storel %a, %rem_ptr
    ret 0
}
