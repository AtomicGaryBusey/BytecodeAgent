export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    jmp @loop(l 0, l 0)

@loop(l %i, l %count)
    %cond =w csltl %i, %n
    jnz %cond, @body, @exit

@body
    %src_offset =l mul %i, 8
    %src_ptr =l add %src, %src_offset
    %src_val =l loadl %src_ptr
    
    %f_result =l call %f(l %src_val)
    
    %dst_offset =l mul %i, 8
    %dst_ptr =l add %dst, %dst_offset
    storel %f_result, %dst_ptr
    
    %less =w csltl %f_result, %src_val
    %less_ext =l exts %less
    %new_count =l add %count, %less_ext
    
    %i_next =l add %i, 1
    
    jmp @loop(l %i_next, l %new_count)

@exit
    ret %count
}
