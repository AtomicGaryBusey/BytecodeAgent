export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @loop_cond

@loop_cond
    %count = phi l @entry 0, @loop_end %count_next
    %i = phi l @entry 0, @loop_end %i_next
    %cond = csltl %i, %n
    jnz %cond, @loop_body, @return

@loop_body
    %offset = mul l %i, 8
    %addr = add l %a, %offset
    %elem = loadl %addr
    %is_greater = cultl %t, %elem
    jnz %is_greater, @increment, @no_increment

@increment
    %count_new = add l %count, 1
    jmp @loop_end

@no_increment
    jmp @loop_end

@loop_end
    %count_next = phi l @increment %count_new, @no_increment %count
    %i_next = add l %i, 1
    jmp @loop_cond

@return
    ret %count
}
