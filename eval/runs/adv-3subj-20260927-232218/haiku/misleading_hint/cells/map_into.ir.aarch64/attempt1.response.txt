export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %count =l copy 0
    %idx =l copy 0
    jmp @loop_check

@loop_check
    %cond =w csltl %idx, %n
    jnz %cond @loop_body @loop_end

@loop_body
    %offset =l mul %idx, 8
    %src_ptr =l add %src, %offset
    %src_val =l loadl %src_ptr
    %result =l call %f(l %src_val)
    %dst_ptr =l add %dst, %offset
    storel %result, %dst_ptr
    %is_less =w csltl %result, %src_val
    jnz %is_less @increment_count @next_iteration

@increment_count
    %count =l add %count, 1
    jmp @next_iteration

@next_iteration
    %idx =l add %idx, 1
    jmp @loop_check

@loop_end
    ret %count
}
