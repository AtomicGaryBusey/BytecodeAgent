export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop(0, %a)

@loop(l %i, l %cur_a)
    %val = sub l %start, %i
    storel %val, %cur_a
    %i_next = add l %i, 1
    %cur_a_next = add l %cur_a, 8
    %cond = csltl %i_next, %n
    jnz %cond, @loop(%i_next, %cur_a_next), @done

@done
    ret l %n
}
