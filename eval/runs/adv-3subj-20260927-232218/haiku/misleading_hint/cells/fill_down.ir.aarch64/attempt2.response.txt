export function l $fill_down(l %a, l %n, l %start) {
@entry
    %i = copy l 0
    jmp @loop

@loop
    %val = sub l %start, %i
    %scaled = mul l %i, 8
    %ptr = add l %a, %scaled
    storel %val, %ptr
    %i_next = add l %i, 1
    %cmp = csltl %i_next, %n
    jnz %cmp, @loop, @done

@done
    ret l %n
}
