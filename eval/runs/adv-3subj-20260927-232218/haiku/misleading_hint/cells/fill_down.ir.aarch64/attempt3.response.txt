export function l $fill_down(l %a, l %n, l %start) {
@entry
    %i = const l 0
    jmp @loop

@loop
    %val = subl %start, %i
    %scaled = mull %i, 8
    %ptr = addl %a, %scaled
    storel %val, %ptr
    %i_next = addl %i, 1
    %cmp = csltl %i_next, %n
    jnz %cmp, @loop, @done

@done
    ret l %n
}
