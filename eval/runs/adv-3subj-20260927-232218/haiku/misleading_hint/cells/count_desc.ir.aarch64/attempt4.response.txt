export function l $count_desc(l %a, l %n) {
@entry
    %n_lt_2 = cltl %n, 2
    jnz %n_lt_2, @return_zero
    jmp @loop

@loop
    %i = phi @entry 0, @loop_body %i_next
    %count = phi @entry 0, @loop_body %count_next
    
    %i_plus_1 = add %i, 1
    %should_continue = cltl %i_plus_1, %n
    jnz %should_continue, @loop_body
    ret %count

@loop_body
    %offset_i = mul %i, 8
    %addr_i = add %a, %offset_i
    %val_i = loadl %addr_i
    
    %offset_i1 = mul %i_plus_1, 8
    %addr_i1 = add %a, %offset_i1
    %val_i1 = loadl %addr_i1
    
    %is_gt = cgtl %val_i, %val_i1
    %is_gt_l = extuw %is_gt
    %count_next = add %count, %is_gt_l
    
    %i_next = add %i, 1
    jmp @loop

@return_zero
    ret 0
}
