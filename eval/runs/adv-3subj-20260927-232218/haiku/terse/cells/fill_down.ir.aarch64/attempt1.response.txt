export function l $fill_down(l %a, l %n, l %start) {
@entry
  jmp @loop_check

@loop_check
  %offset =l phi @entry 0, @loop_body %offset_next
  %cond =w csltl %offset, %n
  jnz %cond, @loop_body, @done

@loop_body
  %idx =l mul %offset, 8
  %addr =l add %a, %idx
  %val =l sub %start, %offset
  storel %val, %addr
  %offset_next =l add %offset, 1
  jmp @loop_check

@done
  ret %n
}
