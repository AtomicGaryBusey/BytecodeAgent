export function l $max_byte(l %p, l %n) {
@start
    %n_pos =w csgt %n, 0
    jz %n_pos, @zero
    jmp @loop

@loop
    %p_in =l phi @start %p, @end %p_next
    %n_in =l phi @start %n, @end %n_next
    %max =l phi @start 0, @up %byte, @end %max

    %done =w csle %n_in, 0
    jnz %done, @return

    %byte =l loadub %p_in
    %is_gt =w csgt %byte, %max

    jz %is_gt, @end
    jmp @up

@up
    jmp @end

@end
    %p_next =l add %p_in, 1
    %n_next =l sub %n_in, 1
    jmp @loop

@return
    ret %max

@zero
    ret 0
}
