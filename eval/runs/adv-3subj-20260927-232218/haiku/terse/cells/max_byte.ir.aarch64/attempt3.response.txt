export function l $max_byte(l %p, l %n) {
@entry
    %n_zero =w ceql %n, 0
    jnz %n_zero, @ret_zero
    jmp @loop

@loop
    %p_in =l phi @entry %p, @merge %p_next
    %n_in =l phi @entry %n, @merge %n_next
    %max =l phi @entry 0, @merge %max_new

    %byte =l loadub %p_in
    %cmp =w sgtl %byte, %max
    jnz %cmp, @upd

@no_upd
    %max_new =l add %max, 0
    jmp @merge

@upd
    %max_new =l add %byte, 0
    jmp @merge

@merge
    %p_next =l add %p_in, 1
    %n_next =l sub %n_in, 1
    %continue =w sgtl %n_next, 0
    jnz %continue, @loop

    ret %max_new

@ret_zero
    ret 0
}
