export function l $max_byte(l %p, l %n) {
@start
    jz %n, @ret_zero
    jmp @loop

@loop
    %p_in =l phi @start %p, @merge %p_next
    %n_in =l phi @start %n, @merge %n_next
    %max =l phi @start 0, @merge %max_new

    %done =w sle %n_in, 0
    jnz %done, @ret_max

    %byte =l loadub %p_in
    %cmp =w sgt %byte, %max
    jz %cmp, @no_upd

    %max_new =l add %byte, 0
    jmp @merge

@no_upd
    %max_new =l add %max, 0
    jmp @merge

@merge
    %p_next =l add %p_in, 1
    %n_next =l sub %n_in, 1
    jmp @loop

@ret_max
    ret %max

@ret_zero
    ret 0
}
