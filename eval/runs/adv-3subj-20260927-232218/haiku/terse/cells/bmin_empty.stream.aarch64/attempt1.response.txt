bmin_empty:
    cbz x1, .Lempty
    ldr x2, [x0]
    cmp x1, #1
    b.le .Ldone
    mov x3, #1
.Lloop:
    ldr x4, [x0, x3, lsl #3]
    cmp x4, x2
    csel x2, x4, x2, lt
    add x3, x3, #1
    cmp x3, x1
    b.lt .Lloop
.Ldone:
    mov x0, x2
    ret
.Lempty:
    mov x0, #-1
    asr x0, x0, #1
    ret
