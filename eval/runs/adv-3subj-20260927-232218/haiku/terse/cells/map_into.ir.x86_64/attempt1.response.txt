export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %count =l copy 0
    %i =l copy 0
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @end

@body
    %offset =l mul %i, 8
    %src_ptr =l add %src, %offset
    %src_val =l loadl %src_ptr
    %dst_val =l call %f(l %src_val)
    %dst_ptr =l add %dst, %offset
    storel %dst_val, %dst_ptr
    %is_less =w csltl %dst_val, %src_val
    %is_less_l =l extuw %is_less
    %count =l add %count, %is_less_l
    %i =l add %i, 1
    jmp @loop

@end
    ret %count
}
