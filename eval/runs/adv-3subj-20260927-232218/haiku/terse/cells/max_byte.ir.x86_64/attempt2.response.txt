export function l $max_byte(l %p, l %n) {
@init:
    %max =l copy 0
    %i =l copy 0
    jmp @loop
@loop:
    %cond =w cltl %i, %n
    jnz %cond @body
    ret %max
@body:
    %addr =l add %p, %i
    %byte =l loadub %addr
    %gt =w cltl %max, %byte
    jnz %gt @update
    %i =l add %i, 1
    jmp @loop
@update:
    %max =l copy %byte
    %i =l add %i, 1
    jmp @loop
}
