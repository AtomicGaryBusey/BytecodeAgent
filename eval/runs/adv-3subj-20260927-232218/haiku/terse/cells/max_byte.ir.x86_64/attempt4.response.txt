export function l $max_byte(l %p, l %n) {
@start:
    %max =l copy 0
    %i =l copy 0
    jmp @loop
@loop:
    %cond =w cltl %i, %n
    jnz %cond @body
    ret %max
@body:
    %addr =l add %p, %i
    %byte =l loadub %addr
    %gt =w cltl %max, %byte
    %max =l select %gt, %byte, %max
    %i =l add %i, 1
    jmp @loop
}
