export function l $stats_f(l %a, l %n, l %f, l %out)
@start
    %i =l copy 0
    %min =l copy 0
    %max =l copy 0
    %sum =l copy 0
    %odd_count =l copy 0
@loop
    %done =w ceql %i, %n
    jnz %done @store_results
    %offset =l mul %i, 8
    %a_ptr =l add %a, %offset
    %elem =l loadl %a_ptr
    %y =l call %f(l %elem)
    %sum =l add %sum, %y
    %is_first =w ceql %i, 0
    jnz %is_first @first_iter
    jmp @not_first
@first_iter
    %min =l copy %y
    %max =l copy %y
    jmp @check_odd
@not_first
    %cmp_min =w csltl %y, %min
    jnz %cmp_min @update_min
    jmp @check_max
@update_min
    %min =l copy %y
    jmp @check_max
@check_max
    %cmp_max =w csgtl %y, %max
    jnz %cmp_max @update_max
    jmp @check_odd
@update_max
    %max =l copy %y
    jmp @check_odd
@check_odd
    %is_odd =w and %y, 1
    jnz %is_odd @inc_odd
    jmp @next_iter
@inc_odd
    %odd_count =l add %odd_count, 1
@next_iter
    %i =l add %i, 1
    jmp @loop
@store_results
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd_count
