export function l $count_desc(l %a, l %n) {
@start
    %zero = copy 0
    %one = copy 1
    %cmp = csgtl %n, %one
    jnz %cmp, @loop %zero %zero, @fin %zero
    
@fin(l %res)
    ret %res
    
@loop(l %c, l %i)
    %lim = sub %n, 1
    %ok = csltl %i, %lim
    jnz %ok, @body %c %i, @fin %c
    
@body(l %c, l %i)
    %off = mul %i, 8
    %addr1 = add %a, %off
    %addr2 = add %addr1, 8
    %val1 = loadl %addr1
    %val2 = loadl %addr2
    %gt = csgtl %val1, %val2
    %ni = add %i, 1
    jnz %gt, @inc %c %ni, @loop %c %ni
    
@inc(l %c, l %i)
    %nc = add %c, 1
    jmp @loop %nc %i
}
