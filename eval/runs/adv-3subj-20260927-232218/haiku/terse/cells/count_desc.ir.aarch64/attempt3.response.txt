export function l $count_desc(l %a, l %n) {
@start
    %zero = copy 0
    %one = copy 1
    %two = copy 2
    %eight = copy 8
    
    %cmp = csltl %n, %two
    jnz %cmp, @ret0, @loop %zero %zero
    
@ret0
    ret %zero
    
@loop(l %c, l %i)
    %lim = sub %n, %one
    %ok = csltl %i, %lim
    jnz %ok, @body %c %i, @ret %c
    
@body(l %c, l %i)
    %off = mul %i, %eight
    %a1 = add %a, %off
    %a2 = add %a1, %eight
    %v1 = loadl %a1
    %v2 = loadl %a2
    %gt = csgtl %v1, %v2
    %ni = add %i, %one
    jnz %gt, @inc %c %ni, @loop %c %ni
    
@inc(l %c, l %i)
    %nc = add %c, %one
    jmp @loop %nc %i
    
@ret(l %c)
    ret %c
}
