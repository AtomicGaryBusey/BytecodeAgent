export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop

@loop
    %i = phi @start 0, @inc %i1
    %count = phi @start 0, @inc %count1
    %test = csltl %i, %n
    jnz %test @body
    ret %count

@body
    %offset = shll %i, 3
    %addr = addl %a, %offset
    %val = loadl %addr
    %gt = cugtl %val, %t
    jnz %gt @bump
    jmp @inc

@bump
    %c_b = addl %count, 1
    jmp @inc

@inc
    %count1 = phi @body %count, @bump %c_b
    %i1 = addl %i, 1
    jmp @loop
}
