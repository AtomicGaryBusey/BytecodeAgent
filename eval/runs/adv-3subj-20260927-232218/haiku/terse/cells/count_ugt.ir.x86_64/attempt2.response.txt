export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop(0, 0)

@loop(l %i, l %count)
    %cond = csltl %i, %n
    jnz %cond @body(%i, %count)
    ret %count

@body(l %i, l %count)
    %offset = shll %i, 3
    %addr = addl %a, %offset
    %val = loadl %addr
    %gt = cugtl %val, %t
    %gt_l = extuw %gt
    %new_count = addl %count, %gt_l
    %new_i = addl %i, 1
    jmp @loop(%new_i, %new_count)
}
