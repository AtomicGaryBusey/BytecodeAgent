export function l $count_ugt(l %a, l %n, l %t) {
@start
    %zero = addl 0, 0
    jmp @loop

@loop
    %i = phi @start %zero, @inc %i_1
    %count = phi @start %zero, @inc %count_1
    %cond = csltl %i, %n
    jnz %cond @body
    ret %count

@body
    %off = shll %i, 3
    %p = addl %a, %off
    %v = loadl %p
    %gt = cugtl %v, %t
    jnz %gt @bump
    jmp @inc

@bump
    %count_b = addl %count, 1
    jmp @inc

@inc
    %count_1 = phi @body %count, @bump %count_b
    %i_1 = addl %i, 1
    jmp @loop
}
