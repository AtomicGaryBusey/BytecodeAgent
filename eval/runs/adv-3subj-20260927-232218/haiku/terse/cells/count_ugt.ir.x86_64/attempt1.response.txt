export function l $count_ugt(l %a, l %n, l %t) {
@start
    %end_offset = mull %n, 8
    jmp @loop(0, 0, %end_offset)

@loop(l %offset, l %count, l %end_offset)
    %cond = csltl %offset, %end_offset
    jnz %cond @body(%offset, %count, %end_offset)
    ret %count

@body(l %offset, l %count, l %end_offset)
    %addr = addl %a, %offset
    %val = loadl %addr
    %gt = cugtl %val, %t
    jnz %gt @inc(%offset, %count, %end_offset)
    %new_offset = addl %offset, 8
    jmp @loop(%new_offset, %count, %end_offset)

@inc(l %offset, l %count, l %end_offset)
    %new_count = addl %count, 1
    %new_offset = addl %offset, 8
    jmp @loop(%new_offset, %new_count, %end_offset)
}
