@entry
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%odd =l copy 0
	%bit =l and %y0, 1
	%is_odd =w cnel %bit, 0
	jnz %is_odd @s1
	jmp @c1
@s1
	%odd =l copy 1
@c1
	%more =w cnel %n, 1
	jnz %more @li
	jmp @st
@li
	%i =l copy 1
@lp

	jnz %cond @bd
	jmp @st
@bd

	jnz %lt @sm
	jmp @cm
@sm
	%min =l copy %res
@cm

	jnz %gt @sx
	jmp @ad
@sx
	%max =l copy %res
@ad
	%sum =l addl %sum, %res
	%b =l and %res, 1
	%o =w cnel %b, 0
	jnz %o @io
	jmp @nx
@io
	%odd =l addl %odd, 1
@nx

	jmp @lp
@st

	%p1 =l addl %out, 8
	storel %max, %p1
	%p2 =l addl %out, 16

	ret %odd
}
