export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%v0 =l loadl %a
	%y0 =l call %f(l %v0)
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%odd =l copy 0
	%test =l and %y0, 1
	%is_odd =w cnel %test, 0
	jnz %is_odd @have_odd
	jmp @check_loop
@have_odd
	%odd =l copy 1
@check_loop
	%more =w cnel %n, 1
	jnz %more @loop
	jmp @store
@loop
	%i =l copy 1
@lb
	%cond =w csltl %i, %n
	jnz %cond @bd
	jmp @store
@bd
	%off =l mull %i, 8
	%ptr =l addl %a, %off
	%v =l loadl %ptr
	%y =l call %f(l %v)
	%lt =w csltl %y, %min
	jnz %lt @sm
	jmp @cm
@sm
	%min =l copy %y
@cm
	%gt =w csgtl %y, %max
	jnz %gt @sx
	jmp @ad
@sx
	%max =l copy %y
@ad
	%sum =l addl %sum, %y
	%b =l and %y, 1
	%o =w cnel %b, 0
	jnz %o @io
	jmp @nx
@io
	%odd =l addl %odd, 1
@nx
	%i =l addl %i, 1
	jmp @lb
@store
	storel %min, %out
	%p1 =l addl %out, 8
	storel %max, %p1
	%p2 =l addl %out, 16
	storel %sum, %p2
	ret %odd
}
