export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%odd =l copy 0
	
	%bit =l and %y0, 1
	%is_odd =w cnel %bit, 0
	jnz %is_odd @set_odd
	jmp @check_n

@set_odd
	%odd =l copy 1

@check_n
	%more =w cnel %n, 1
	jnz %more @loop_init
	jmp @store

@loop_init
	%i =l copy 1

@loop
	%cond =w csltl %i, %n
	jnz %cond @body
	jmp @store

@body
	%idx =l mull %i, 8
	%ptr =l addl %a, %idx
	%val =l loadl %ptr
	%res =l call %f(l %val)
	
	%lt =w csltl %res, %min
	jnz %lt @set_min
	jmp @check_max

@set_min
	%min =l copy %res

@check_max
	%gt =w csgtl %res, %max
	jnz %gt @set_max
	jmp @add_sum

@set_max
	%max =l copy %res

@add_sum
	%sum =l addl %sum, %res
	%b =l and %res, 1
	%o =w cnel %b, 0
	jnz %o @inc_odd
	jmp @next

@inc_odd
	%odd =l addl %odd, 1

@next
	%i =l addl %i, 1
	jmp @loop

@store
	storel %min, %out
	%p1 =l addl %out, 8
	storel %max, %p1
	%p2 =l addl %out, 16
	storel %sum, %p2
	ret %odd
}
