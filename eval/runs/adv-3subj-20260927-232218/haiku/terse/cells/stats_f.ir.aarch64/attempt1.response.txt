export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%odd =l copy 0
	
	%odd0 =w and (trunc %y0), 1
	jnz %odd0 @odd1

@after_odd
	%ne1 =w cnel %n, 1
	jnz %ne1 @loop_init
	jmp @store

@odd1
	%odd =l copy 1
	jmp @after_odd

@loop_init
	%i =l copy 1

@loop_cond
	%clt =w csltl %i, %n
	jnz %clt @loop_body
	jmp @store

@loop_body
	%off =l mull %i, 8
	%addr =l addl %a, %off
	%ai =l loadl %addr
	%yi =l call %f(l %ai)
	
	%cmin =w csltl %yi, %min
	jnz %cmin @upd_min

@after_min
	%cmax =w csgtl %yi, %max
	jnz %cmax @upd_max

@after_max
	%sum =l addl %sum, %yi
	
	%codd =w and (trunc %yi), 1
	jnz %codd @inc_odd

@after_odd_check
	%i =l addl %i, 1
	jmp @loop_cond

@upd_min
	%min =l copy %yi
	jmp @after_min

@upd_max
	%max =l copy %yi
	jmp @after_max

@inc_odd
	%odd =l addl %odd, 1
	jmp @after_odd_check

@store
	storel %min, %out
	%o1 =l addl %out, 8
	storel %max, %o1
	%o2 =l addl %out, 16
	storel %sum, %o2
	ret %odd
}
