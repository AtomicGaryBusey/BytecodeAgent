export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%count = copy 0
	%idx = copy 0
	jnz %n @loop @done
	
@loop
	%offset = mul l %idx, 8
	%addr = add l %a, %offset
	%elem = loadl %addr
	%is_gt = cugtl %elem, %t
	%is_gt_l = extwl %is_gt
	%count = add l %count, %is_gt_l
	%idx = add l %idx, 1
	%cond = csltl %idx, %n
	jnz %cond @loop @done
	
@done
	ret %count
}
