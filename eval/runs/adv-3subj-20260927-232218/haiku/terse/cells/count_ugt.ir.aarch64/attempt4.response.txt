export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%count = add 0, 0
	%idx = add 0, 0
	jnz %n @loop @done

@loop
	%offset = mul %idx, 8
	%addr = add %a, %offset
	%elem = loadl %addr
	%is_gt = cugtl %elem, %t
	%is_gt_l = extuw %is_gt
	%count = add %count, %is_gt_l
	%idx = add %idx, 1
	%cond = cultl %idx, %n
	jnz %cond @loop @done

@done
	ret %count
}
