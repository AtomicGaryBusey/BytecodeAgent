export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%count = addl 0, 0
	%idx = addl 0, 0
	jnz %n @loop @done

@loop
	%offset = mull %idx, 8
	%addr = addl %a, %offset
	%elem = loadl %addr
	%is_gt = cugtl %elem, %t
	%is_gt_l = extwl %is_gt
	%count = addl %count, %is_gt_l
	%idx = addl %idx, 1
	%cond = csltl %idx, %n
	jnz %cond @loop @done

@done
	ret %count
}
