export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    jmp @loop

@loop
    %i =l phi @entry 0, @body %i1
    %cnt =l phi @entry 0, @body %cnt1
    
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %off =l mul %i, 8
    %sa =l add %src, %off
    %sv =l loadl %sa
    
    %res =l call %f(l %sv)
    
    %da =l add %dst, %off
    storel %res, %da
    
    %cmp =w csltl %res, %sv
    %cmp_ext =l extzl %cmp
    %cnt1 =l add %cnt, %cmp_ext
    %i1 =l add %i, 1
    
    jmp @loop

@done
    ret %cnt
}
