export function l $count_desc(l %a, l %n) {
@start
    %check = csltl %n, 2
    jnz %check, @ret_zero
    jmp @loop

@loop
    %c = phi @start 0, @back_prep %c_new
    %i = phi @start 0, @back_prep %i_new
    
    %end = subl %n, 1
    %done = csgtl %i, %end
    jnz %done, @exit
    
    %shift = shl %i, 3
    %addr = addl %a, %shift
    %v0 = loadl %addr
    %addr1 = addl %addr, 8
    %v1 = loadl %addr1
    %gt = csgtl %v0, %v1
    
    jnz %gt, @incr_c
    jmp @back_prep

@incr_c
    %c_incr = addl %c, 1
    jmp @back_prep

@back_prep
    %c_new = phi @loop %c, @incr_c %c_incr
    %i_new = addl %i, 1
    jmp @loop

@exit
    ret %c

@ret_zero
    ret 0
}
