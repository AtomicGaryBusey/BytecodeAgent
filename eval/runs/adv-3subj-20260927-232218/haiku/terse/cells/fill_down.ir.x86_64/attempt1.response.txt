export function l $fill_down(l %a, l %n, l %start) {
@entry
    %i = copy 0
    jmp @loop
@loop
    %i_phi = phi %i @entry %i_next @body
    %cond = csltl %i_phi %n
    jnz %cond @body @end
@body
    %offset = mull %i_phi 8
    %ptr = addl %a %offset
    %val = subl %start %i_phi
    storel %val %ptr
    %i_next = addl %i_phi 1
    jmp @loop
@end
    ret %n
}
