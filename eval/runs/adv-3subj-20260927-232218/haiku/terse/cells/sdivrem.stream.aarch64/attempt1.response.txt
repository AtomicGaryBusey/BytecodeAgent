sdivrem:
    cbz x1, .Ldiv_by_zero
    movz x3, 0x8000, lsl 48
    cmp x0, x3
    b.ne .Lnormal_div
    cmn x1, 1
    b.ne .Lnormal_div
    str xzr, [x2]
    ret
.Ldiv_by_zero:
    str x0, [x2]
    mov x0, 0
    ret
.Lnormal_div:
    sdiv x3, x0, x1
    msub x4, x3, x1, x0
    str x4, [x2]
    mov x0, x3
    ret
