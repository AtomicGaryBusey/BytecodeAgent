bmin_empty:
    test rsi, rsi
    jz .Lempty
    mov rax, qword ptr [rdi]
    mov rcx, 1
    jmp .Ltest
.Lloop:
    mov r8, qword ptr [rdi + rcx*8]
    cmp r8, rax
    cmovl rax, r8
    inc rcx
.Ltest:
    cmp rcx, rsi
    jl .Lloop
    ret
.Lempty:
    mov rax, 0x7FFFFFFFFFFFFFFF
    ret
