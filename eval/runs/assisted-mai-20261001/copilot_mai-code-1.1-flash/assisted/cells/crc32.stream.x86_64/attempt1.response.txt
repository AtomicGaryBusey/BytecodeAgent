crc32:
    mov eax, 0xffffffff
    xor ecx, ecx
.Louter:
    cmp rcx, rsi
    jae .Ldone
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit:
    test eax, 1
    jz .Leven
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lbitcont
.Leven:
    shr eax, 1
.Lbitcont:
    dec r8d
    jnz .Lbit
    inc rcx
    jmp .Louter
.Ldone:
    xor eax, 0xffffffff
    ret
