dot_i32_simd:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x22, xzr
    cmp x21, #0
    ble .Ldone
.Lvec:
    cmp x21, #8
    b.lt .Ltail
    ld1 {v0.4s}, [x19], #16
    ld1 {v1.4s}, [x20], #16
    sxtl v2.2d, v0.2s
    sxtl2 v3.2d, v0.4s
    sxtl v4.2d, v1.2s
    sxtl2 v5.2d, v1.4s
    mul v2.2d, v2.2d, v4.2d
    mul v3.2d, v3.2d, v5.2d
    addv d2, v2.2d
    addv d3, v3.2d
    mov x9, v2.d[0]
    mov x10, v3.d[0]
    add x22, x22, x9
    add x22, x22, x10
    sub x21, x21, #8
    b .Lvec
.Ltail:
    cbz x21, .Ldone
.Ltail_loop:
    ldr w9, [x19], #4
    ldr w10, [x20], #4
    sxtw x9, w9
    sxtw x10, w10
    madd x22, x9, x10, x22
    sub x21, x21, #1
    cbnz x21, .Ltail_loop
.Ldone:
    mov x0, x22
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
