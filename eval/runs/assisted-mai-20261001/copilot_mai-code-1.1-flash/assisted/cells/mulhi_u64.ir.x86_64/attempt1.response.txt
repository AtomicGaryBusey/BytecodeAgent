export function l $mulhi_u64(l %a, l %b) {
@entry
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%sum_mid =l add %p01, %p10
	%carry_mid =w ultl %sum_mid, %p01
	%carry_mid64 =l extuw %carry_mid
	%carry_mid_shift =l shl %carry_mid64, 32

	%p00_hi =l shr %p00, 32
	%sum_mid_plus_c =l add %sum_mid, %p00_hi
	%u_hi =l shr %sum_mid_plus_c, 32
	%u =l add %u_hi, %carry_mid_shift

	%res_hi =l add %p11, %u
	ret %res_hi
}
