export function l $isqrt(l %n) {
@entry
	%lo =l copy 0
	%hi =l copy 4294967295
	jmp @loop
@loop
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @done
@body
	%diff =l sub %hi, %lo
	%plus =l add %diff, 1
	%shift =l shr %plus, 1
	%mid =l add %lo, %shift
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @take_lo, @take_hi
@take_lo
	%lo =l copy %mid
	jmp @loop
@take_hi
	%hi =l sub %mid, 1
	jmp @loop
@done
	ret %lo
}
