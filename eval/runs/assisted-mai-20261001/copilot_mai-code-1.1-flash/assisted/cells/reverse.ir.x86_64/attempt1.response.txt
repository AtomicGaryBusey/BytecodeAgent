export function l $reverse(l %a, l %n) {
@start
	%left =l copy 0
	%right =l sub %n, 1
@loop
	%cond =w csltl %left, %right
	jnz %cond, @body, @done
@body
	%l_off =l mul %left, 8
	%r_off =l mul %right, 8
	%l_ptr =l add %a, %l_off
	%r_ptr =l add %a, %r_off
	%lv =l loadl %l_ptr
	%rv =l loadl %r_ptr
	storel %rv, %l_ptr
	storel %lv, %r_ptr
	%left =l add %left, 1
	%right =l sub %right, 1
	jmp @loop
@done
	%zero =l copy 0
	ret %zero
}
