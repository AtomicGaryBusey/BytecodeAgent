export function l $crc32(l %p, l %n) {
@start
	%crc =w copy -1
	%i =l copy 0
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l add %p, %i
	%b =w loadub %off
	%crc =w xor %crc, %b
	%j =w copy 8
@bitloop
	%lsb =w and %crc, 1
	jnz %lsb, @odd, @even
@odd
	%crc =w shr %crc, 1
	%crc =w xor %crc, -306674912
	jmp @nextbit
@even
	%crc =w shr %crc, 1
@nextbit
	%j =w sub %j, 1
	jnz %j, @bitloop, @afterbyte
@afterbyte
	%i =l add %i, 1
	jmp @loop
@done
	%crc =w xor %crc, -1
	%res =l extuw %crc
	ret %res
}
