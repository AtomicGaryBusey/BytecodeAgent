export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	%i =l copy 0
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%iszero =w ceql %v, 0
	jnz %iszero, @found_zero, @continue
@continue
	%i =l add %i, 1
	jmp @loop
@found_zero
	ret %i
@done
	ret %n
}
