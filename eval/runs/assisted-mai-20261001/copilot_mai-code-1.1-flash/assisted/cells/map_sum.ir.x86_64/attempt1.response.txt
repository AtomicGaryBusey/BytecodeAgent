export function l $map_sum(l %a, l %n, l %f) {
@entry
	%sum =l copy 0
	%neg =w csltl %n, 0
	jnz %neg, @done, @init
@init
	%i =l copy 0
	jmp @loop
@loop
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%r =l call %f(l %v)
	%sum =l add %sum, %r
	%i =l add %i, 1
	jmp @loop
@done
	ret %sum
}
