export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%b0 =l and %b, %mask
	%a1 =l shr %a, 32
	%b1 =l shr %b, 32
	%p0 =l mul %a0, %b0
	%p0_hi =l shr %p0, 32
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1
	%low1 =l and %p1, %mask
	%low2 =l and %p2, %mask
	%low0 =l and %p0_hi, %mask
	%sum_lo =l add %low1, %low2
	%sum_lo =l add %sum_lo, %low0
	%carry0 =l shr %sum_lo, 32
	%hi1 =l shr %p1, 32
	%hi2 =l shr %p2, 32
	%mid_hi =l add %hi1, %hi2
	%mid_hi =l add %mid_hi, %carry0
	%hi =l add %p3, %mid_hi
	ret %hi
}
