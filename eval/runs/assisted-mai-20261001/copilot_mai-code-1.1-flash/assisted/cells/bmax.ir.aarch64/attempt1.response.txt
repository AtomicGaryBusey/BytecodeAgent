export function l $bmax(l %a, l %n) {
@start
	%i =l copy 1
	%max =l loadl %a
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%cmp =w csgtl %v, %max
	jnz %cmp, @update, @next
@update
	%max =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
