export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
@loop
	%cond =w csltl %i, %j
	jnz %cond, @body, @done
@body
	%off1 =l mul %i, 8
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%off2 =l mul %j, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	storel %v2, %p1
	storel %v1, %p2
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@done
	%ret =l copy 0
	ret %ret
}
