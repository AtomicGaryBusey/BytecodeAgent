add_overflow:
    mov r8, rdi
    add r8, rsi
    mov qword ptr [rdx], r8
    mov r9, rdi
    sar r9, 63
    mov r10, rsi
    sar r10, 63
    cmp r9, r10
    jne .Lno_overflow
    mov r11, r8
    sar r11, 63
    cmp r9, r11
    jne .Loverflow
.Lno_overflow:
    xor eax, eax
    ret
.Loverflow:
    mov eax, 1
    ret
