export function l $bmax(l %a, l %n) {
@entry
	%max =l loadl %a
	%i =l copy 1
	jmp @loop

@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%gt =w csgtl %v, %max
	jnz %gt, @update, @next

@update
	%max =l copy %v
	jmp @next

@next
	%i =l add %i, 1
	jmp @loop

@done
	ret %max
}
