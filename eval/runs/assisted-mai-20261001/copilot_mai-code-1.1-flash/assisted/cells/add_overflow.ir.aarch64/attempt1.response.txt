export function l $add_overflow(l %a, l %b, l %out) {
@entry
	%sum =l add %a, %b
	storel %sum, %out

	%a_nonneg =w csgel %a, 0
	%b_nonneg =w csgel %b, 0
	%a_neg =w csltl %a, 0
	%b_neg =w csltl %b, 0
	%sum_neg =w csltl %sum, 0
	%sum_nonneg =w csgel %sum, 0

	%t1 =w and %a_nonneg, %b_nonneg
	%t2 =w and %t1, %sum_neg
	%t3 =w and %a_neg, %b_neg
	%t4 =w and %t3, %sum_nonneg
	%ovf =w or %t2, %t4
	%ovf_l =l extuw %ovf
	ret %ovf_l
}
