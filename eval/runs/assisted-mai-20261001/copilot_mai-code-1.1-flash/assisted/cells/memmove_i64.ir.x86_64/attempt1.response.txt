export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%eq =w ceql %dst, %src
	jnz %eq, @done, @cmp
@cmp
	%fwd =w cultl %dst, %src
	jnz %fwd, @forward, @backward
@forward
	%i =l copy 0
@forwloop
	%cond =w csltl %i, %n
	jnz %cond, @forwbody, @done
@forwbody
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l add %i, 1
	jmp @forwloop
@backward
	%i =l copy %n
@backloop
	%cond =w csgtl %i, 0
	jnz %cond, @backbody, @done
@backbody
	%i =l sub %i, 1
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	jmp @backloop
@done
	ret 0
}
