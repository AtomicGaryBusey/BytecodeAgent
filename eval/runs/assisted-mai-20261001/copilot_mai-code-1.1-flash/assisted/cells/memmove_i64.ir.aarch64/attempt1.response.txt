export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%zero =w ceql %n, 0
	jnz %zero, @ret0, @cmp
@cmp
	%dst_lt_src =w cultl %dst, %src
	jnz %dst_lt_src, @forward, @check_src_lt_dst
@check_src_lt_dst
	%src_lt_dst =w cultl %src, %dst
	jnz %src_lt_dst, @backward, @ret0
@forward
	%i =l copy 0
@floop
	%cond =w cultl %i, %n
	jnz %cond, @fbody, @ret0
@fbody
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l add %i, 1
	jmp @floop
@backward
	%i =l sub %n, 1
@bloop
	%neg =w csltl %i, 0
	jnz %neg, @ret0, @bbody
@bbody
	%off =l mul %i, 8
	%srcp =l add %src, %off
	%dstp =l add %dst, %off
	%v =l loadl %srcp
	storel %v, %dstp
	%i =l sub %i, 1
	jmp @bloop
@ret0
	%r =l copy 0
	ret %r
}
