bubble_sort:
	xor eax, eax
	cmp rsi, 1
	jle .Ldone
	mov r8, rsi
	dec r8
	xor ecx, ecx
.Louter:
	cmp rcx, r8
	jge .Ldone
	xor r9d, r9d
	mov r10, rsi
	sub r10, rcx
	dec r10
.LInner:
	cmp r9, r10
	jge .LNextOuter
	lea r11, [rdi + r9*8]
	mov rax, qword ptr [r11]
	mov rdx, qword ptr [r11 + 8]
	cmp rax, rdx
	jle .LNoSwap
	mov qword ptr [r11], rdx
	mov qword ptr [r11 + 8], rax
.LNoSwap:
	inc r9
	jmp .LInner
.LNextOuter:
	inc rcx
	jmp .LOuter
.Ldone:
	xor eax, eax
	ret
