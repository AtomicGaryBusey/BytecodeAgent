bubble_sort:
    cbz x1, .Ldone
    mov x2, #0
.Louter:
    cmp x2, x1
    b.hs .Ldone
    sub x3, x1, x2
    sub x3, x3, #1
    mov x4, #0
.Linner:
    cmp x4, x3
    b.hs .LafterInner
    add x5, x0, x4, lsl #3
    ldr x6, [x5]
    ldr x7, [x5, #8]
    cmp x6, x7
    b.le .Lskip
    str x7, [x5]
    str x6, [x5, #8]
.Lskip:
    add x4, x4, #1
    b .Linner
.LafterInner:
    add x2, x2, #1
    b .Louter
.Ldone:
    mov x0, #0
    ret
