export function l $add_overflow(l %a, l %b, l %out) {
@entry
	%sum =l add %a, %b
	storel %sum, %out
	%zero =l copy 0
	%a_neg =w csltl %a, %zero
	%b_neg =w csltl %b, %zero
	%same =w ceqw %a_neg, %b_neg
	%sum_neg =w csltl %sum, %zero
	%diff =w cnew %sum_neg, %a_neg
	%ovf =w and %same, %diff
	%ovf64 =l extuw %ovf
	ret %ovf64
}
