export function l $add_overflow(l %a, l %b, l %out) {
@entry
	%sum =l add %a, %b
	storel %sum, %out
	%zero =l copy 0
	%as =w csltl %a, %zero
	%bs =w csltl %b, %zero
	%same =w ceqw %as, %bs
	%ss =w csltl %sum, %zero
	%diff =w cneqw %ss, %as
	%ovf =w and %same, %diff
	%ovf64 =l extuw %ovf
	ret %ovf64
}
