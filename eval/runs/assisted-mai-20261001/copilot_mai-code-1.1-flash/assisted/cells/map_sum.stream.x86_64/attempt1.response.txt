map_sum:
    cmp rsi, 0
    jle .Lzero
    push rbx
    push r12
    push r13
    push r14
    sub rsp, 8
    mov rbx, 0
    mov r12, rdx
    mov r14, rdi
    mov r13, rsi
.Lloop:
    cmp r13, 0
    jle .Lend
    mov rdi, qword ptr [r14]
    cld
    call r12
    add rbx, rax
    add r14, 8
    sub r13, 1
    jmp .Lloop
.Lend:
    mov rax, rbx
    add rsp, 8
    pop r14
    pop r13
    pop r12
    pop rbx
    cld
    ret
.Lzero:
    cld
    xor eax, eax
    ret
