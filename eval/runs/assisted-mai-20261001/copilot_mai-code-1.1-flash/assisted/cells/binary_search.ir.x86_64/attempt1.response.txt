export function l $binary_search(l %a, l %n, l %key) {
@entry
	%is_empty =w cslel %n, 0
	jnz %is_empty, @not_found, @init
@init
	%lo =l copy 0
	%hi =l sub %n, 1
@loop
	%go_on =w cslel %lo, %hi
	jnz %go_on, @probe, @not_found
@probe
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp_lt
@cmp_lt
	%lt =w csltl %v, %key
	jnz %lt, @move_right, @move_left
@move_right
	%lo =l add %mid, 1
	jmp @loop
@move_left
	%hi =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@not_found
	%neg_one =l copy -1
	ret %neg_one
}
