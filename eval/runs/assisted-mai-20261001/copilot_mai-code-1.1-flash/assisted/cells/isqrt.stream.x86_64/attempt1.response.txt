isqrt:
    xor eax, eax
    mov rdx, 1
    shl rdx, 62
.Linit:
    cmp rdx, rdi
    jbe .Lloop
    shr rdx, 2
    jmp .Linit
.Lloop:
    test rdx, rdx
    jz .Ldone
    mov r8, rax
    add r8, rdx
    cmp rdi, r8
    jb .Lelse
    sub rdi, r8
    mov r9, rax
    shr r9, 1
    add r9, rdx
    mov rax, r9
    jmp .Lafter
.Lelse:
    shr rax, 1
.Lafter:
    shr rdx, 2
    jmp .Lloop
.Ldone:
    ret
