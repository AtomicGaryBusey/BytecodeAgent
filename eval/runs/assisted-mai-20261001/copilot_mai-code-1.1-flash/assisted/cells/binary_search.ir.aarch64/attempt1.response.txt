export function l $binary_search(l %a, l %n, l %key) {
@entry
	%lo =l copy 0
	%hi =l sub %n, 1
	jmp @loop
@loop
	%cond =w cslel %lo, %hi
	jnz %cond, @body, @done
@body
	%diff =l sub %hi, %lo
	%half =l sar %diff, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @check
@check
	%lt =w csltl %val, %key
	jnz %lt, @move_right, @move_left
@move_right
	%lo =l add %mid, 1
	jmp @loop
@move_left
	%hi =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@done
	ret -1
}
