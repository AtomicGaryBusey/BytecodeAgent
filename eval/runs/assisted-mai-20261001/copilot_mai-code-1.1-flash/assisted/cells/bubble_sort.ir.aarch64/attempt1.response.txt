export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
	jmp @outer_loop
@outer_loop
	%cond_i =w csltl %i, %n
	jnz %cond_i, @outer_body, @done
@outer_body
	%limit =l sub %n, %i
	%limit1 =l sub %limit, 1
	%j =l copy 0
	jmp @inner_loop
@inner_loop
	%cond_j =w csltl %j, %limit1
	jnz %cond_j, @inner_body, @outer_inc
@inner_body
	%off_j =l mul %j, 8
	%p_j =l add %a, %off_j
	%v_j =l loadl %p_j
	%p_next =l add %p_j, 8
	%v_next =l loadl %p_next
	%swap_needed =w csgtl %v_j, %v_next
	jnz %swap_needed, @swap, @inner_inc
@swap
	storel %v_next, %p_j
	storel %v_j, %p_next
	jmp @inner_inc
@inner_inc
	%j =l add %j, 1
	jmp @inner_loop
@outer_inc
	%i =l add %i, 1
	jmp @outer_loop
@done
	%zero =l copy 0
	ret %zero
}
