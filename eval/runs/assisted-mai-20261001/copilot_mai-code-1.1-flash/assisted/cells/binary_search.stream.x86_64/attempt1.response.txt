binary_search:
    cmp rsi, 0
    jle .Lnot_found
    mov r8, 0
    lea r9, [rsi - 1]
.Lloop:
    cmp r8, r9
    jg .Lnot_found
    mov rax, r8
    add rax, r9
    shr rax, 1
    cmp rdx, qword ptr [rdi + rax*8]
    je .Lfound
    jl .Lleft
    lea r8, [rax + 1]
    jmp .Lloop
.Lleft:
    lea r9, [rax - 1]
    jmp .Lloop
.Lfound:
    ret
.Lnot_found:
    mov rax, -1
    ret
