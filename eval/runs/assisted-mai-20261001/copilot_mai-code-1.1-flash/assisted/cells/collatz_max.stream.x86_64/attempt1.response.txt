collatz_max:
    push rbx
    push r12
    mov rbx, 1
    xor r12, r12
    cmp rbx, rdi
    jg .Ldone
.Louter:
    mov rax, rbx
    xor rcx, rcx
.Lcount:
    cmp rax, 1
    je .Lafter_count
    test rax, 1
    jnz .Lodd
    shr rax, 1
    jmp .Lstep
.Lodd:
    imul rax, rax, 3
    inc rax
.Lstep:
    inc rcx
    jmp .Lcount
.Lafter_count:
    cmp rcx, r12
    jle .Lnext_k
    mov r12, rcx
.Lnext_k:
    inc rbx
    cmp rbx, rdi
    jle .Louter
.Ldone:
    mov rax, r12
    pop r12
    pop rbx
    ret
