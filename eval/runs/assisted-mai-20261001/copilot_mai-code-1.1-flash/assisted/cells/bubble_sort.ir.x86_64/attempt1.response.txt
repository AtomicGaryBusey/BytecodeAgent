export function l $bubble_sort(l %a, l %n) {
@entry
	%i =l copy 0
@outer_loop
	%outer_cond =w csltl %i, %n
	jnz %outer_cond, @outer_body, @outer_done
@outer_body
	%j =l copy 0
@inner_loop
	%n_minus_i =l sub %n, %i
	%limit =l sub %n_minus_i, 1
	%inner_cond =w csltl %j, %limit
	jnz %inner_cond, @inner_body, @inner_done
@inner_body
	%j_off =l mul %j, 8
	%a_j =l add %a, %j_off
	%v_j =l loadl %a_j
	%j_plus_1 =l add %j, 1
	%j1_off =l mul %j_plus_1, 8
	%a_j1 =l add %a, %j1_off
	%v_j1 =l loadl %a_j1
	%need_swap =w csgtl %v_j, %v_j1
	jnz %need_swap, @swap, @inner_continue
@swap
	storel %v_j1, %a_j
	storel %v_j, %a_j1
@inner_continue
	%j =l add %j, 1
	jmp @inner_loop
@inner_done
	%i =l add %i, 1
	jmp @outer_loop
@outer_done
	%ret =l copy 0
	ret %ret
}
