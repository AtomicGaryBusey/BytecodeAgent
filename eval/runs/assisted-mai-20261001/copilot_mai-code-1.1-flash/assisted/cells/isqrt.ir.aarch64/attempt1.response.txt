export function l $isqrt(l %n) {
@entry
	%lo =l copy 0
	%hi =l copy 4294967295
@loop
	%cond =w culel %lo, %hi
	jnz %cond, @body, @done
@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%sq =l mul %mid, %mid
	%cmp =w culel %sq, %n
	jnz %cmp, @inc_low, @dec_high
@inc_low
	%lo =l add %mid, 1
	jmp @loop
@dec_high
	%hi =l sub %mid, 1
	jmp @loop
@done
	ret %hi
}
