export function l $collatz_max(l %n) {
@entry
	%i =l copy 1
	%best =l copy 0
	jmp @loop_k

@loop_k
	%cond =w csgtl %i, %n
	jnz %cond, @done, @body_k

@body_k
	%x =l copy %i
	%steps =l copy 0
	jmp @collatz_loop

@collatz_loop
	%done_x =w ceql %x, 1
	jnz %done_x, @after_steps, @step_choice

@step_choice
	%rem =l rem %x, 2
	%is_even =w ceql %rem, 0
	jnz %is_even, @even_branch, @odd_branch

@even_branch
	%x =l div %x, 2
	%steps =l add %steps, 1
	jmp @collatz_loop

@odd_branch
	%tmp =l mul %x, 3
	%x =l add %tmp, 1
	%steps =l add %steps, 1
	jmp @collatz_loop

@after_steps
	%better =w csgtl %steps, %best
	jnz %better, @update_best, @next

@update_best
	%best =l copy %steps
	jmp @next

@next
	%i =l add %i, 1
	jmp @loop_k

@done
	ret %best
}
