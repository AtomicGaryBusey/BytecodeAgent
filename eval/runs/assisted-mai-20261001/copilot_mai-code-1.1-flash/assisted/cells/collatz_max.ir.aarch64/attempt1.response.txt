export function l $collatz_max(l %n) {
@entry
	%max =l copy 0
	%k =l copy 1
	jmp @outer
@outer
	%limit =w csgtl %k, %n
	jnz %limit, @done, @kbody
@kbody
	%x =l copy %k
	%steps =l copy 0
@inner
	%is_one =w ceql %x, 1
	jnz %is_one, @after_inner, @step
@step
	%r =l rem %x, 2
	%even =w ceql %r, 0
	jnz %even, @even_case, @odd_case
@even_case
	%x =l div %x, 2
	jmp @after_step
@odd_case
	%x =l mul %x, 3
	%x =l add %x, 1
@after_step
	%steps =l add %steps, 1
	jmp @inner
@after_inner
	%bigger =w csgtl %steps, %max
	jnz %bigger, @update_max, @next
@update_max
	%max =l copy %steps
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %max
}
