binary_search:
    cmp x1, #0
    b.le .Lnot_found
    mov x3, #0
    sub x4, x1, #1
.Lloop:
    cmp x3, x4
    b.gt .Lnot_found
    add x5, x3, x4
    lsr x5, x5, #1
    add x6, x0, x5, lsl #3
    ldr x7, [x6]
    cmp x7, x2
    b.eq .Lfound
    b.lt .Lright
    sub x4, x5, #1
    b .Lloop
.Lright:
    add x3, x5, #1
    b .Lloop
.Lfound:
    mov x0, x5
    ret
.Lnot_found:
    movn x0, #0
    ret
