memmove_i64:
    cmp rdx, 0
    jle .Lret0
    cmp rdi, rsi
    je .Lret0
    jb .Lforward
    mov rax, rdx
    dec rax
.Lbackward:
    mov r8, qword ptr [rsi + rax*8]
    mov qword ptr [rdi + rax*8], r8
    test rax, rax
    jz .Lret0
    dec rax
    jmp .Lbackward
.Lforward:
    xor rax, rax
.Lforward_loop:
    mov r8, qword ptr [rsi + rax*8]
    mov qword ptr [rdi + rax*8], r8
    inc rax
    cmp rax, rdx
    jb .Lforward_loop
.Lret0:
    xor eax, eax
    ret
