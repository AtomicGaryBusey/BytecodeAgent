dot_i32_simd:
    cld
    xor rax, rax
    mov rcx, rdx
    shr rcx, 1
    jz .Lscalar
.Lpair:
    test rcx, rcx
    jz .Lscalar
    pmovsxdq xmm0, xmmword ptr [rdi]
    pmovsxdq xmm1, xmmword ptr [rsi]
    pmuldq xmm0, xmm1
    movq r8, xmm0
    psrldq xmm0, 8
    movq r9, xmm0
    add rax, r8
    add rax, r9
    add rdi, 8
    add rsi, 8
    dec rcx
    jmp .Lpair
.Lscalar:
    test rdx, 1
    jz .Ldone
    movsx r8, dword ptr [rdi]
    movsx r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
.Ldone:
    ret
