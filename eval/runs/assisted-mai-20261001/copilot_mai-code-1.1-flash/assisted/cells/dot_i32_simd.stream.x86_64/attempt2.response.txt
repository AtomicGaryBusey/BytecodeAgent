dot_i32_simd:
    cld
    xor rax, rax
    mov rcx, rdx
    shr rcx, 2
    jz .Lscalar
    pxor xmm2, xmm2
.Lvec:
    movq xmm0, qword ptr [rdi]
    movq xmm1, qword ptr [rsi]
    pmovsxdq xmm0, xmm0
    pmovsxdq xmm1, xmm1
    pmuldq xmm0, xmm1
    paddq xmm2, xmm0
    movq xmm0, qword ptr [rdi+8]
    movq xmm1, qword ptr [rsi+8]
    pmovsxdq xmm0, xmm0
    pmovsxdq xmm1, xmm1
    pmuldq xmm0, xmm1
    paddq xmm2, xmm0
    add rdi, 16
    add rsi, 16
    dec rcx
    jnz .Lvec
    movdqa xmm0, xmm2
    psrldq xmm0, 8
    paddq xmm2, xmm0
    movq rax, xmm2
.Lscalar:
    mov rcx, rdx
    and rcx, 3
    jz .Ldone
.Lrem:
    movsxd r8, dword ptr [rdi]
    movsxd r9, dword ptr [rsi]
    imul r8, r9
    add rax, r8
    add rdi, 4
    add rsi, 4
    dec rcx
    jnz .Lrem
.Ldone:
    ret
