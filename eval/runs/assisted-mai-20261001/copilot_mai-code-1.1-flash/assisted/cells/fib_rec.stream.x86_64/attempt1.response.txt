fib_rec:
    push r12
    push rbx
    sub rsp, 8
    mov rbx, rdi
    cmp rbx, 1
    jg .Lrecurse
    mov rax, rbx
    jmp .Ldone
.Lrecurse:
    mov rdi, rbx
    dec rdi
    call fib_rec
    mov r12, rax
    mov rdi, rbx
    sub rdi, 2
    call fib_rec
    add rax, r12
.Ldone:
    add rsp, 8
    pop rbx
    pop r12
    ret
