export function l $crc32(l %p, l %n) {
@entry
	%crc =w copy 4294967295
	%i =l copy 0
	jmp @loop

@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%k =w copy 0
	jmp @bitloop

@bitloop
	%bit =w and %crc, 1
	jnz %bit, @bitodd, @biteven

@bitodd
	%shr =w shr %crc, 1
	%crc =w xor %shr, 3988292384
	jmp @bitnext

@biteven
	%crc =w shr %crc, 1
	jmp @bitnext

@bitnext
	%k =w add %k, 1
	%done8 =w ceqw %k, 8
	jnz %done8, @nextbyte, @bitloop

@nextbyte
	%i =l add %i, 1
	jmp @loop

@done
	%final =w xor %crc, 4294967295
	%out =l extuw %final
	ret %out
}
