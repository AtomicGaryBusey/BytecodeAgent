export function l $sum_to(l n) {
    @entry:
        %cond = csltl l n, l 0
        jnz %cond, @false
        %result = addl l 0, l 0
        ret l %result
    @false:
        %n_val = loadl l n
        %one = addl l 0, l 1
        %n_plus_one = addl l %n_val, l %one
        %prod = mulw l %n_val, l %n_plus_one
        %shifted = shrw l %prod, l 1
        ret l %shifted
}
