sum_to:
    cmp rdi, 1       # Compare n with 1
    jle .Lzero      # Jump if less or equal to zero

    mov rcx, rdi    # Move n into rcx
    xor rax, rax    # Clear RAX (accumulator)
    inc rcx         # Increment counter (rcx)

.Lloop:
    add rax, rdi    # Add current value of n to accumulator
    dec rcx         # Decrement counter
    jg .Lloop       # Jump if greater than zero

    ret             # Return result in RAX

.Lzero:
    xor eax, eax    # Set EAX (RAX) to 0 for return
    ret             # Return
