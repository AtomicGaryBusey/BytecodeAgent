export function l $sum_to(l n) {
    %zero = constl 0
    @start
    loadl %result, %zero

    csltl %cmp, n, %zero
    jnz %cmp, @positive
    ret %result

    @positive
    %one = constl 1
    storel %result+8, %one

    @loop_start
    loadl %prev_result, %result
    addl %next_val, n, -%one
    csltl %cmp, %next_val, %zero
    jnz %cmp, @addition

    ret %result

    @addition
    addl %new_sum, %prev_result, %next_val
    storel %result+8, %new_sum
    subl n, n, -%one
    jmp @loop_start
}
