export function l $sum_to(l n) {
    %zero = constl 0
    %one = constl 1
    @start
    storel %result, %zero

    csltl %cmp, n, %zero
    jnz %cmp, @positive
    ret %result

    @positive
    loadl %prev_result, %result
    subl %next_val, n, -%one
    csltl %cmp, %next_val, %zero
    jnz %cmp, @addition

    ret %result

    @addition
    addl %new_sum, %prev_result, %next_val
    storel %result+8, %new_sum
    subl n, n, -%one
    jmp @loop_start

    @loop_start
    loadl %prev_n, n
    csltl %cmp, %prev_n, %zero
    jnz %cmp, @addition

    ret %result
}
