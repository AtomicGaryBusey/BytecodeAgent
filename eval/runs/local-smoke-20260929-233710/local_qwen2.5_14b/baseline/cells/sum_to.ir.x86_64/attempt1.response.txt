export function l $sum_to(l n) {
    %zero = const l 0
    %one = const l 1
    %two = const l 2

    @entry:
    csltl w $nonpos, %zero, n
    jnz @done, $nonpos

    @initloop:
    storel n, -8(%rbp)
    subq l %one, n
    storel %one, -16(%rbp)
    jmp @loopentry

    @loopentry:
    loadl n, -8(%rbp)
    csltl w $zeroloopvar, %zero, n
    jnz @addloop, $zeroloopvar

    @done:
    loadl %zero, rax

    @return:
    addq l 16, %rsp
    ret

    @addloop:
    loadl -8(%rbp), n
    loadl -16(%rbp), %two
    csltl w $zeroloopvar, %zero, %two
    jnz @endloop, $zeroloopvar

    addq l n, rax
    loadl -16(%rbp), %two
    subq l %one, %two
    storel %two, -16(%rbp)
    csltl w $exitcond, %zero, %two
    jnz @initloop, $exitcond

    jmp @return

    @endloop:
    loadl -8(%rbp), rax
}

export function l $sum_to(l n) {
    @entry:
    cmpq n, const l 0
    jle @done
    movq l rax, %one
    subq l rax, n
    je @done
    xorq l rcx, rcx

    @loopstart:
    incq rcx
    addq rax, rcx
    decq n
    jnz @loopstart

    movq rax, %one

    @done:
}
