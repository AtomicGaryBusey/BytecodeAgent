    cmp x0, #1               // if (n <= 0)
    b.le .Lzero              //   goto .Lzero

.Linit:
    mov x1, xzr              // sum = 0
    mov x2, x0               // i = n

.Lloop:
    subs x3, x2, #1          // i--
    cbnz x3, .Ladd           // if (i != 0) goto Ladd

.Ldone:
    lsl x0, x1, #1           // sum * 2
    add x0, x0, x2           // sum = sum + n
    udiv x0, x0, #2          // sum / 2
