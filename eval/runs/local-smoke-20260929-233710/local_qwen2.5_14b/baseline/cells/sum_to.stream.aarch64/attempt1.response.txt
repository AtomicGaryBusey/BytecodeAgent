sum_to:
    cmp x0, #0
    b.le .Lzero
    add x2, x0, #1           // x2 = n + 1
    lsl x3, x2, #2           // x3 = (n + 1) * 4
    mov x5, xzr              // sum = 0

.Lloop:
    subs x6, x0, #1          // i--
    cbnz x6, .Ladd           // if (i != -1) goto Ladd
    lsr x0, x3, #2           // return (sum + n) / 4
    ret

.Ladd:
    add x5, x5, x0           // sum += i
    subs x6, x2, #1          // j--
    cbnz x6, .Lloop          // if (j != 0) goto Lloop

.Lzero:
    mov x0, xzr              // return 0
    ret
