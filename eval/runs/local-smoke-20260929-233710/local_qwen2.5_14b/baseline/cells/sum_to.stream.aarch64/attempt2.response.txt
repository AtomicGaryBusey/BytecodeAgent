sum_to:
    cmp x0, #0
    b.le .Lzero
    mov x1, xzr              // sum = 0
    add x2, x0, #1           // n + 1

.Lloop:
    subs x3, x2, #1          // i--
    cbnz x3, .Ladd           // if (i != 0) goto Ladd

.Ldone:
    lsr x0, x2, #2           // return sum / 4
    ret

.Ladd:
    add x1, x1, x2, lsl #3   // sum += i * 4
    subs x2, x2, #1          // n--
    b .Lloop                 // goto Lloop

.Lzero:
    mov x0, xzr              // return 0
    ret
