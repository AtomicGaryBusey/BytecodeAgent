import sys
d=sys.stdin.buffer.read()
lines=d.split(b'\n')
if d.endswith(b'\n') or d==b'': lines=lines[:-1]
def wrap(x):
    x&=(1<<64)-1
    return x-(1<<64) if x>>63 else x
o=[]
for ln in lines:
    toks=[t for t in ln.replace(b'\t',b' ').split(b' ') if t]
    st=[];err=False
    for t in toks:
        if t in (b'+',b'-',b'*',b'/',b'%'):
            if len(st)<2: err=True;break
            b=st.pop();a=st.pop()
            if t==b'+':st.append(wrap(a+b))
            elif t==b'-':st.append(wrap(a-b))
            elif t==b'*':st.append(wrap(a*b))
            else:
                if b==0 or (a==-(1<<63) and b==-1): err=True;break
                q=abs(a)//abs(b); 
                if (a<0)!=(b<0): q=-q
                st.append(q if t==b'/' else a-q*b)
        else:
            s=t[1:] if t.startswith(b'-') else t
            if not (1<=len(s)<=18 and all(48<=c<=57 for c in s)): err=True;break
            st.append(int(t))
    if err or len(st)!=1: o.append(b'error')
    else: o.append(str(st[0]).encode())
sys.stdout.buffer.write(b''.join(x+b'\n' for x in o))
