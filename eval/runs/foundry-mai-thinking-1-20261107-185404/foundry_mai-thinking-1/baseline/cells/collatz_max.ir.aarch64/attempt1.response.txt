export function l $collatz_max(l %n) {
@start
	%max = copy 0l
	%k = copy 1l
	jmp @loop_k

@loop_k
	%le = csleql %k, %n
	jnz %le, @do_k, @ret

@do_k
	%x = copy %k
	%s = copy 0l
	jmp @collatz

@collatz
	%is1 = ceql %x, 1l
	jnz %is1, @end_k, @step

@step
	%s2 = addl %s, 1l
	%s = copy %s2
	%odd_bit = andl %x, 1l
	%evenw = ceql %odd_bit, 0l
	jnz %evenw, @even, @oddp

@even
	%x2 = sdivl %x, 2l
	%x = copy %x2
	jmp @collatz

@oddp
	%x3 = mull %x, 3l
	%x4 = addl %x3, 1l
	%x = copy %x4
	jmp @collatz

@end_k
	%better = csgtl %s, %max
	jnz %better, @upd, @nextk

@upd
	%max = copy %s
	jmp @nextk

@nextk
	%k2 = addl %k, 1l
	%k = copy %k2
	jmp @loop_k

@ret
	ret %max
}
