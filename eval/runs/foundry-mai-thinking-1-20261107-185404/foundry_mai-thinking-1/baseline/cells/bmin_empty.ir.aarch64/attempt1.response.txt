export function l $bmin_empty(l %a, l %n) {
@start
	%iszero = ceql %n, 0
	jnz %iszero, @empty, @nonempty

@empty
	ret 0x7FFFFFFFFFFFFFFF

@nonempty
	%min = loadl %a
	%i = addl 1, 0

@loop
	%cont = csltl %i, %n
	jnz %cont, @body, @done

@body
	%off = shll %i, 3
	%p = addl %a, %off
	%v = loadl %p
	%smaller = csltl %v, %min
	jnz %smaller, @setmin, @inc

@setmin
	%min = %v

@inc
	%i = addl %i, 1
	jmp @loop

@done
	ret %min
}
