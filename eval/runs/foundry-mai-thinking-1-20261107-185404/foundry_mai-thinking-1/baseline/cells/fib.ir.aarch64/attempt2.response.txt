export function l $fib(l %n) {
@entry
	%is0 = ceql %n, 0
	jnz %is0, @return0, @check1
@check1
	%is1 = ceql %n, 1
	jnz %is1, @return1, @init
@return0
	ret 0
@return1
	ret 1
@init
	%prev = copy 0
	%curr = copy 1
	%i = copy 2
	jmp @loop
@loop
	%le = cslel %i, %n
	jnz %le, @body, @end
@body
	%next = addl %prev, %curr
	%prev = copy %curr
	%curr = copy %next
	%i = addl %i, 1
	jmp @loop
@end
	ret %curr
}
