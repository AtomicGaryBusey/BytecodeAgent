export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	%zero =l add 0, 0
	%one  =l add 0, 1
	%j0   =l add 0, 0
@init_loop
	%c0  =w csltl %j0, %n
	jnz %c0, @init_body, @init_done
@init_body
	%addr0 =l add %row, %j0
	%addr0s=l shl %addr0, 3
	%addr0f=l add %row, %addr0s
	storel %zero, %addr0f
	%j0n  =l add %j0, %one
	%j0   =l mov %j0n
	jmp @init_loop
@init_done
	%i    =l add 0, 0
@outer_loop
	%c1   =w csltl %i, %m
	jnz %c1, @outer_body, @outer_end
@outer_body
	%diag  =l add 0, 0
	%j     =l add 0, 1
@inner_loop
	%c2    =w csltl %j, %n
	%jn_eq =w ceql %j, %n
	%c2_or =w or %c2, %jn_eq
	jnz %c2_or, @inner_body, @outer_inc
@inner_body
	%rj_addr =l add %row, %j
	%rj_sh   =l shl %rj_addr, 3
	%rj_ptr  =l add %row, %rj_sh
	%up      =l loadl %rj_ptr
	%j1      =l add %j, -1
	%rj1_addr=l add %row, %j1
	%rj1_sh  =l shl %rj1_addr, 3
	%rj1_ptr =l add %row, %rj1_sh
	%left    =l loadl %rj1_ptr
	%xi_addr =l add %x, %i
	%xi_b    =w loadub %xi_addr
	%yj1_addr=l add %y, %j1
	%yj1_b   =w loadub %yj1_addr
	%eq_b    =w ceqw %xi_b, %yj1_b
	jnz %eq_b, @equal_case, @max_case
@equal_case
	%new_eq  =l add %diag, %one
	storel %new_eq, %rj_ptr
	%diag_new=l mov %up
	%diag    =l mov %diag_new
	jmp @inner_inc
@max_case
	%c3      =w csltl %up, %left
	jnz %c3, @take_left, @take_up
@take_left
	storel %left, %rj_ptr
	%diag_l  =l mov %up
	%diag    =l mov %diag_l
	jmp @inner_inc
@take_up
	storel %up, %rj_ptr
	%diag_u  =l mov %up
	%diag    =l mov %diag_u
	jmp @inner_inc
@inner_inc
	%jn      =l add %j, %one
	%j       =l mov %jn
	jmp @inner_loop
@outer_inc
	%in      =l add %i, %one
	%i       =l mov %in
	jmp @outer_loop
@outer_end
	%rn_addr =l add %row, %n
	%rn_sh   =l shl %rn_addr, 3
	%rn_ptr  =l add %row, %rn_sh
	%res     =l loadl %rn_ptr
	ret %res
}
