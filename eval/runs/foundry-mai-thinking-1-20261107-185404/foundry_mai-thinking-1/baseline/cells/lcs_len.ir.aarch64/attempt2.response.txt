export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	%zero =l add 0, 0
	%one  =l add 0, 1
	%j0   =l add 0, 0
@init_loop
	%c0   =w csltl %j0, %n
	jnz   %c0, @init_body, @init_done
@init_body
	%j0sh =l shl %j0, 3
	%p0   =l add %row, %j0sh
	storel %zero, %p0
	%j0n  =l add %j0, %one
	%j0   =l copy %j0n
	jmp   @init_loop
@init_done
	%i    =l add 0, 0
@outer_loop
	%c1   =w csltl %i, %m
	jnz   %c1, @outer_body, @outer_end
@outer_body
	%diag =l add 0, 0
	%j    =l add 0, 1
@inner_loop
	%c2   =w csltl %j, %n
	%ceq  =w ceql %j, %n
	%cor  =w or %c2, %ceq
	jnz   %cor, @inner_body, @outer_inc
@inner_body
	%jsh  =l shl %j, 3
	%rj   =l add %row, %jsh
	%up   =l loadl %rj
	%j1   =l add %j, -1
	%j1sh =l shl %j1, 3
	%rj1  =l add %row, %j1sh
	%left =l loadl %rj1
	%xis  =l add %x, %i
	%xib  =w loadub %xis
	%yjs  =l add %y, %j1
	%yjb  =w loadub %yjs
	%eqb  =w ceqw %xib, %yjb
	jnz   %eqb, @equal_case, @max_case
@equal_case
	%neq  =l add %diag, %one
	storel %neq, %rj
	%diag =l copy %up
	jmp   @inner_inc
@max_case
	%c3   =w csltl %up, %left
	jnz   %c3, @take_left, @take_up
@take_left
	storel %left, %rj
	%diag =l copy %up
	jmp   @inner_inc
@take_up
	storel %up, %rj
	%diag =l copy %up
	jmp   @inner_inc
@inner_inc
	%jn   =l add %j, %one
	%j    =l copy %jn
	jmp   @inner_loop
@outer_inc
	%in   =l add %i, %one
	%i    =l copy %in
	jmp   @outer_loop
@outer_end
	%nsh  =l shl %n, 3
	%rn   =l add %row, %nsh
	%res  =l loadl %rn
	ret   %res
}
