export function l $bubble_sort(l %a, l %n) {
@start
	%ret = copy 0
	%is_small = csltl %n, 2
	jnz %is_small, @end, @outer_init

@outer_init
	%i = copy 0
	jmp @outer_check

@outer_check
	%limit_i_pass = subl %n, 1
	%outer_done = csltl %i, %limit_i_pass
	jnz %outer_done, @inner_init, @end

@inner_init
	%j = copy 0
	%limit_j_base = subl %n, 1
	%limit_j = subl %limit_j_base, %i
	jmp @inner_check

@inner_check
	%inner_done = csltl %j, %limit_j
	jnz %inner_done, @compare, @outer_inc

@compare
	%j1 = addl %j, 1
	%off1 = shl %j, 3
	%off2 = shl %j1, 3
	%p1 = addl %a, %off1
	%p2 = addl %a, %off2
	%v1 = loadl %p1
	%v2 = loadl %p2
	%need_swap = csgtl %v1, %v2
	jnz %need_swap, @swap, @inner_inc

@swap
	storel %p1, %v2
	storel %p2, %v1
	jmp @inner_inc

@inner_inc
	%j_new = addl %j, 1
	%j = copy %j_new
	jmp @inner_check

@outer_inc
	%i_new = addl %i, 1
	%i = copy %i_new
	jmp @outer_check

@end
	ret %ret
}
