export function l $bubble_sort(l %a, l %n) {
@start
	%ret =l 0
	%is_small =w csltl %n, 2
	jnz %is_small, @end, @outer_init

@outer_init
	%i =l 0
	jmp @outer_check

@outer_check
	%limit_i_pass =l sub %n, 1
	%outer_done =w csltl %i, %limit_i_pass
	jnz %outer_done, @inner_init, @end

@inner_init
	%j =l 0
	%limit_j_base =l sub %n, 1
	%limit_j =l sub %limit_j_base, %i
	jmp @inner_check

@inner_check
	%inner_done =w csltl %j, %limit_j
	jnz %inner_done, @compare, @outer_inc

@compare
	%j1 =l add %j, 1
	%off1 =l shl %j, 3
	%off2 =l shl %j1, 3
	%p1 =l add %a, %off1
	%p2 =l add %a, %off2
	%v1 =l loadl %p1
	%v2 =l loadl %p2
	%need_swap =w csgtl %v1, %v2
	jnz %need_swap, @swap, @inner_inc

@swap
	storel %p1, %v2
	storel %p2, %v1
	jmp @inner_inc

@inner_inc
	%j_new =l add %j, 1
	%j =l copy %j_new
	jmp @inner_check

@outer_inc
	%i_new =l add %i, 1
	%i =l copy %i_new
	jmp @outer_check

@end
	ret %ret
}
