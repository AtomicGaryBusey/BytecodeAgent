	jnz %zb, @return_a, @loop

@loop

	jnz %za, @return_b, @next

@next
	%a1 = copy %b
	%b1 = copy %r
	jmp @loop2

@loop2
	%r2 = reml %a1, %b1
	%zr = ceql %r2, 0
	jnz %zr, @return_b1, @next2

@next2
	%a2 = copy %b1
	%b2 = copy %r2
	%a1 = copy %a2
	%b1 = copy %b2
	jmp @loop2

	ret %b1
}
