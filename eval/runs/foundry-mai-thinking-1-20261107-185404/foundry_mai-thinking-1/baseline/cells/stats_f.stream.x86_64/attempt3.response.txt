push rbx
push r12
push r13
push r14
push r15
sub rsp, 32
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor rbx, rbx
cmp r13, 0
jle .Ldone_init
mov rdi, qword ptr [r12 + rbx*8]
call r14
mov rbp, rax
mov qword ptr [rsp + 0], rax
mov qword ptr [rsp + 8], rax
xor rcx, rcx
test al, 1
jz .Lfirst_even
inc rcx
.Lfirst_even:
mov qword ptr [rsp + 16], rcx
inc rbx
cmp rbx, r13
jge .Lstore_result
.Lloop:
mov rdi, qword ptr [r12 + rbx*8]
call r14
mov rcx, rax
cmp rcx, rbp
jge .Lno_min
mov rbp, rcx
.Lno_min:
cmp rcx, qword ptr [rsp + 0]
jle .Lno_max
mov qword ptr [rsp + 0], rcx
.Lno_max:
add qword ptr [rsp + 8], rcx
test cl, 1
jz .Lno_odd
inc qword ptr [rsp + 16]
.Lno_odd:
inc rbx
cmp rbx, r13
jl .Lloop
.Lstore_result:
mov rax, rbp
mov qword ptr [r15 + 0], rax
mov rax, qword ptr [rsp + 0]
mov qword ptr [r15 + 8], rax
mov rax, qword ptr [rsp + 8]
mov qword ptr [r15 + 16], rax
mov rax, qword ptr [rsp + 16]
jmp .Lexit
.Ldone_init:
xor rax, rax
mov qword ptr [r15 + 0], rax
mov qword ptr [r15 + 8], rax
mov qword ptr [r15 + 16], rax
.Lexit:
add rsp, 32
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
add rsp, 8
