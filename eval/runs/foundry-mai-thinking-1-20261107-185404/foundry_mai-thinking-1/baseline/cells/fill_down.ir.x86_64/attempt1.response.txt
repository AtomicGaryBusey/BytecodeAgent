export function l $fill_down(l %a, l %n, l %start) {
@entry
	%zero = addl 0, 0
	%iszero = ceql %n, %zero
	jnz %iszero, @return
	%i = addl 0, 0
@loop
	%cond = csltl %i, %n
	jnz %cond, @body, @return
@body
	%off = mull %i, 8
	%ptr = addl %a, %off
	%val = subl %start, %i
	storel %val, %ptr
	%inext = addl %i, 1
	%i = %inext
	jmp @loop
@return
	ret %n
}
