packbits_decode:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov r12, rdi
  mov r13, rdi
  add r13, rsi
  mov r14, rdx
  mov r15, rdx
  add r15, rcx

.Lloop:
  cmp r12, r13
  jae .Ldone_ok
  movsx eax, byte ptr [r12]
  inc r12
  cmp eax, -128
  je .Lloop
  cmp eax, 0
  jl .Lrepeat
  mov ecx, eax
  inc ecx
  mov rax, r12
  add rax, rcx
  cmp rax, r13
  ja .Lerr_src
  mov rax, r14
  add rax, rcx
  cmp rax, r15
  ja .Lliteral_cap
  xor ebp, ebp

.Lliteral_copy:
  cmp rbp, rcx
  je .Lloop
  mov al, byte ptr [r12 + rbp]
  mov byte ptr [r14 + rbp], al
  inc rbp
  jmp .Lliteral_copy

.Lliteral_cap:
  sub rcx, rax
  add rcx, r15
  xor ebp, ebp

.Lliteral_cap_copy:
  cmp rbp, rcx
  je .Lerr_cap
  mov al, byte ptr [r12 + rbp]
  mov byte ptr [r14 + rbp], al
  inc rbp
  jmp .Lliteral_cap_copy

.Lrepeat:
  mov ecx, eax
  neg ecx
  inc ecx
  cmp r12, r13
  ja .Lerr_src
  mov al, byte ptr [r12]
  inc r12
  mov rbx, r14
  add rbx, rcx
  cmp rbx, r15
  ja .Lrepeat_cap
  xor ebp, ebp

.Lrepeat_fill:
  cmp rbp, rcx
  je .Lloop
  mov byte ptr [r14 + rbp], al
  inc rbp
  jmp .Lrepeat_fill

.Lrepeat_cap:
  sub rcx, rbx
  add rcx, r15
  xor ebp, ebp

.Lrepeat_cap_fill:
  cmp rbp, rcx
  je .Lerr_cap
  mov byte ptr [r14 + rbp], al
  inc rbp
  jmp .Lrepeat_cap_fill

.Ldone_ok:
  mov rax, r14
  sub rax, rdx
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret

.Lerr_src:
  mov rax, -2
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret

.Lerr_cap:
  mov rax, -1
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
