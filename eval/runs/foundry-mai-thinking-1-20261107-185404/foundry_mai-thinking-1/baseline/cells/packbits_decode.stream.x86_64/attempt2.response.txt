packbits_decode:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov rbx, rdx
  mov r12, rdi
  mov r13, rdi
  add r13, rsi
  mov r14, rdx
  mov r15, rdx
  add r15, rcx

.Lloop:
  cmp r12, r13
  jae .Ldone_ok
  movsx eax, byte ptr [r12]
  inc r12
  cmp eax, -128
  je .Lloop
  cmp eax, 0
  jge .Lliteral

  mov ecx, eax
  neg ecx
  inc ecx
  cmp r12, r13
  jae .Lerr_src
  mov al, byte ptr [r12]
  inc r12
  lea r10, [r14, rcx]
