packbits_decode:
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  mov rbx, rdx
  mov r12, rdi
  mov r13, rdi
  add r13, rsi
  mov r14, rdx
  mov r15, rdx
  add r15, rcx

.Lloop:
  cmp r12, r13
  jae .Ldone_ok
  movzx eax, byte ptr [r12]
  inc r12
  cmp al, 0x80
  je .Lloop
  js .Lrepeat

  movzx rcx, al
  inc rcx
  lea r10, [r12 + rcx]
  cmp r10, r13
  ja .Lerr_src
  lea r10, [r14 + rcx]
  cmp r10, r15
  ja .Llit_partial
  mov rsi, r12
  mov rdi, r14
  rep movsb
  mov r12, rsi
  mov r14, rdi
  jmp .Lloop

.Llit_partial:
  mov rcx, r15
  sub rcx, r14
  jz .Lerr_out
  mov rsi, r12
  mov rdi, r14
  rep movsb
  mov r12, rsi
  mov r14, rdi
  jmp .Lerr_out

.Lrepeat:
  neg al
  movzx rcx, al
  inc rcx
  cmp r12, r13
  jae .Lerr_src
  mov al, byte ptr [r12]
  inc r12
  lea r10, [r14 + rcx]
  cmp r10, r15
  ja .Lrep_partial
  mov rdi, r14
  rep stosb
  mov r14, rdi
  jmp .Lloop

.Lrep_partial:
  mov rcx, r15
  sub rcx, r14
  jz .Lerr_out
  mov rdi, r14
  rep stosb
  mov r14, rdi
  jmp .Lerr_out

.Ldone_ok:
  mov rax, r14
  sub rax, rbx
  jmp .Lexit

.Lerr_src:
  mov rax, -2
  jmp .Lexit

.Lerr_out:
  mov rax, -1

.Lexit:
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
