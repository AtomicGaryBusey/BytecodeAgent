export function l $bmax(l %a, l %n) {
@start
	%maxv = loadl %a
	%j = 1l
@loop
	%cmp = csltl %j, %n
	jnz %cmp, @body, @end
@body
	%idx = mul %j, 8l
	%p = add %a, %idx
	%v = loadl %p
	%isgt = csgtl %v, %maxv
	jnz %isgt, @update, @next
@update
	%maxv = %v
@next
	%j = add %j, 1l
	jmp @loop
@end
	ret %maxv
}
