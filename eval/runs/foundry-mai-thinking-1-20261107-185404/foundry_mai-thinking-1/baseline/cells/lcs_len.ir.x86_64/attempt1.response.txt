export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j_init =l copy 0
@init_loop
	%cj =w csltl %j_init, %n
	jnz %cj, @init_body, @init_end
@init_body
	%off_init =l mul %j_init, 8
	%p_init =l add %row, %off_init
	storel 0, %p_init
	%j_init_next =l add %j_init, 1
	%j_init =l copy %j_init_next
	jmp @init_loop
@init_end
	%i =l copy 0
@outer_loop
	%ci =w csltl %i, %m
	jnz %ci, @outer_body, @outer_end
@outer_body
	%off_xi =l add %x, %i
	%xi_b =w loadub %off_xi
	%diag =l copy 0
	%j =l copy 1
@inner_loop
	%cj_in =w csleql %j, %n
	jnz %cj_in, @inner_body, @inner_end
@inner_body
	%off_up =l mul %j, 8
	%p_up =l add %row, %off_up
	%up_val =l loadl %p_up
	%j1 =l sub %j, 1
	%off_yj1 =l add %y, %j1
	%yj1_b =w loadub %off_yj1
	%cmp_eq =w ceql %xi_b, %yj1_b
	%off_left =l mul %j1, 8
	%p_left =l add %row, %off_left
	%left_val =l loadl %p_left
	jnz %cmp_eq, @match, @no_match
@match
	%val_match =l add %diag, 1
	%val_res =l copy %val_match
	jmp @store_res
@no_match
	%c_left_geq_up =w csltl %up_val, %left_val
	jnz %c_left_geq_up, @take_left, @take_up
@take_left
	%val_res =l copy %left_val
	jmp @store_res
@take_up
	%val_res =l copy %up_val
	jmp @store_res
@store_res
	storel %val_res, %p_up
	%diag_new =l copy %up_val
	%diag =l copy %diag_new
	%j_next =l add %j, 1
	%j =l copy %j_next
	jmp @inner_loop
@inner_end
	%i_next =l add %i, 1
	%i =l copy %i_next
	jmp @outer_loop
@outer_end
	%off_res =l mul %n, 8
	%p_res =l add %row, %off_res
	%res_final =l loadl %p_res
	ret %res_final
}
