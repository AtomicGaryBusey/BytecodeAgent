mp_mul:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	mov r12, rdi
	mov r13, rsi
	mov r14, rdx
	mov r15, rcx
	mov rbx, r8
	mov rax, r13
	add rax, r15
	test rax, rax
	jle .Lzero_result
	mov rcx, rax
	xor rax, rax
.Lclear_loop:
	mov qword ptr [rbx + rcx*8 - 8], rax
	loop .Lclear_loop
	xor rbp, rbp
.Louter_loop:
	cmp rbp, r13
	jge .Lcount
	mov rdi, qword ptr [r12 + rbp*8]
	xor rcx, rcx
	xor rsi, rsi
.Linner_loop:
	cmp rsi, r15
	jge .Louter_next
	mov rax, rdi
	mul qword ptr [r14 + rsi*8]
	mov r8, rbx
	add r8, rbp
	add r8, rsi
	add rax, qword ptr [r8*8]
	adc rdx, 0
	add rax, rcx
	adc rdx, 0
	mov qword ptr [r8*8], rax
	mov rcx, rdx
	inc rsi
	jmp .Linner_loop
.Louter_next:
	mov r8, rbx
	add r8, rbp
	add r8, r15
	add qword ptr [r8*8], rcx
	inc rbp
	jmp .Louter_loop
.Lcount:
	mov rax, r13
	add rax, r15
	test rax, rax
	je .Lzero_result
.Lcount_loop:
	dec rax
	mov rdx, qword ptr [rbx + rax*8]
	test rdx, rdx
	jne .Lfound
	cmp rax, 0
	jne .Lcount_loop
.Lzero_result:
	xor rax, rax
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
.Lfound:
	inc rax
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
