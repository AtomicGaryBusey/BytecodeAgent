mp_mul:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	mov r12, rdi
	mov r13, rsi
	mov r14, rdx
	mov r15, rcx
	mov rbx, r8
	mov rax, r13
	add rax, r15
	cmp r13, 0
	jle .Lzero_all
	cmp r15, 0
	jle .Lzero_all
	mov rcx, rax
	xor rax, rax
.Lclear_mul:
	cmp rcx, 0
	je .Lstart_mul
	dec rcx
	mov qword ptr [rbx + rcx*8], rax
	jmp .Lclear_mul
.Lstart_mul:
	xor rbp, rbp
.Louter_loop:
	cmp rbp, r13
	jge .Lcount_high
	lea r9, [rbx + rbp*8]
	mov rdi, qword ptr [r12 + rbp*8]
	xor rcx, rcx
	xor rsi, rsi
.Linner_loop:
	cmp rsi, r15
	jge .Louter_next
	mov rax, rdi
	mul qword ptr [r14 + rsi*8]
	add rax, qword ptr [r9 + rsi*8]
	adc rdx, 0
	add rax, rcx
	adc rdx, 0
	mov qword ptr [r9 + rsi*8], rax
	mov rcx, rdx
	inc rsi
	jmp .Linner_loop
.Louter_next:
	add qword ptr [r9 + r15*8], rcx
	inc rbp
	jmp .Louter_loop
.Lcount_high:
	mov rax, r13
	add rax, r15
	dec rax
.Lfind_nonzero:
	cmp rax, 0
	jl .Lret_zero_res
	cmp qword ptr [rbx + rax*8], 0
	jne .Lfound
	dec rax
	jmp .Lfind_nonzero
.Lfound:
	inc rax
	jmp .Lend
.Lzero_all:
	mov rcx, rax
	xor rax, rax
	test rcx, rcx
	jle .Lret_zero_res
.Lclr_zero:
	dec rcx
	mov qword ptr [rbx + rcx*8], rax
	cmp rcx, 0
	jne .Lclr_zero
.Lret_zero_res:
	xor rax, rax
.Lend:
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
