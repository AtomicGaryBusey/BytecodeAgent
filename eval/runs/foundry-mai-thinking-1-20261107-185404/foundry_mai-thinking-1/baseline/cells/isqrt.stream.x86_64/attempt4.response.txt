isqrt:
  test rdi, rdi
  jnz .Lnonzero
  xor rax, rax
  ret

.Lnonzero:
  mov r8, rdi
  bsr rcx, rdi
  inc rcx
  shr rcx, 1
  mov rax, 1
  shl rax, cl
  mov r9, rax

.Lnewton_loop:
  mov rax, r8
  xor rdx, rdx
  div r9
  add rax, r9
  shr rax, 1
  mov r10, rax
  cmp r10, r9
  jae .Lconverged
  mov r9, r10
  jmp .Lnewton_loop

.Lconverged:
  cmp r10, r9
  je .Ladjust
  mov r9, r10

.Ladjust:
  mov rax, r9
  inc rax
  mul rax
  cmp rdx, 0
  jne .Lcheck_down
  cmp rax, r8
  jbe .Ladjust_up

.Lcheck_down:
  mov rax, r9
  mul rax
  cmp rdx, 0
  jne .Lfinal
  cmp rax, r8
  jbe .Lfinal
  dec r9
  jmp .Lfinal

.Ladjust_up:
  inc r9
  jmp .Ladjust

.Lfinal:
  mov rax, r9
  ret
