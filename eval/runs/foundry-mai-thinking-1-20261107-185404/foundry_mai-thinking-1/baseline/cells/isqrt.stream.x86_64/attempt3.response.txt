isqrt:
  test rdi, rdi
  jnz .Lnonzero
  xor rax, rax
  ret

.Lnonzero:
  mov r8, rdi
  bsr rax, rdi
  inc rax
  shr rax, 1
  mov rcx, rax
  mov rax, 1
  shl rax, cl
  mov r9, rax

.Lnewton:
  mov rax, r8
  xor rdx, rdx
  div r9
  add rax, r9
  shr rax, 1
  mov r10, rax
  cmp r10, r9
  jae .Lcheck
  mov r9, r10
  jmp .Lnewton

.Lcheck:
  mov rax, r9
  mul rax
  cmp rdx, 0
  jz .Lcheck_le
  dec r9
  mov rax, r9
  jmp .Lret

.Lcheck_le:
  cmp rax, r8
  jbe .Lret
  dec r9
  mov rax, r9

.Lret:
  mov rax, r9
  ret
