isqrt:
  test rdi, rdi
  jnz .Lnonzero
  xor rax, rax
  ret

.Lnonzero:
  bsr rax, rdi
  inc rax
  shr rax, 1
  mov rcx, rax
  mov rax, 1
  shl rax, cl
  mov r8, rdi
  mov r9, rax

.Lnewton_loop:
  cmp r9, 1
  jbe .Ladjust
  mov rax, r8
  xor rdx, rdx
  div r9
  add rax, r9
  shr rax, 1
  mov r10, rax
  cmp r10, r9
  jae .Lconverged
  mov r9, r10
  jmp .Lnewton_loop

.Lconverged:
  mov rax, r9

.Lmul_check:
  mov rdx, rax
  mul rdx
  cmp rdx, 0
  jnz .Ltoobig
  cmp rax, r8
  jbe .Ldone
  jmp .Ltoobig

.Ltoobig:
  dec rax
  jmp .Lmul_check

.Ladjust:
  mov rax, r9
  jmp .Lmul_check

.Ldone:
  ret
