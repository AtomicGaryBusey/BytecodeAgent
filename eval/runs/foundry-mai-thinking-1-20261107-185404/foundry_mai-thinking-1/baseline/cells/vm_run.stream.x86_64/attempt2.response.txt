vm_run:
	push	rbx
	push	rbp
	push	r12
	push	r13
	push	r14
	push	r15
	sub	rsp, 8

	mov	r13, rdi
	mov	r14, rsi
	mov	r15, rdx
	mov	rbx, rcx
	xor	r12, r12
	xor	rbp, rbp
	xor	r11, r11

.Lloop_start:
	cmp	r11, 5000
	jae	.Lstop_exec
	cmp	r12, r14
	jae	.Lstop_exec

	mov	rax, r12
	shl	rax, 3
	add	rax, r13
	mov	rax, qword ptr [rax]
	cmp	rax, 0
	je	.Lhalt
	cmp	rax, 1
	je	.Lpush
	cmp	rax, 2
	je	.Ladd
	cmp	rax, 3
	je	.Lsub
	cmp	rax, 4
	je	.Lmul
	cmp	rax, 5
	je	.Ldup
	cmp	rax, 6
	je	.Lswap
	cmp	rax, 7
	je	.Ldrop
	cmp	rax, 8
	je	.Lover
	cmp	rax, 9
	je	.Llt
	cmp	rax, 10
	je	.Ljz
	cmp	rax, 11
	je	.Ljmp
	jmp	.Linc_pc1

.Lhalt:
	jmp	.Lstop_exec

.Lpush:
	inc	r11
	cmp	rbp, rbx
	jae	.Lpush_skip
	mov	rax, r12
	add	rax, 1
	shl	rax, 3
	add	rax, r13
	mov	rax, qword ptr [rax]
	mov	rcx, rbp
	shl	rcx, 3
	add	rcx, r15
	mov	qword ptr [rcx], rax
	inc	rbp
.Lpush_skip:
	add	r12, 2
	jmp	.Lloop_start

.Ladd:
	inc	r11
	cmp	rbp, 2
	jb	.Ladd_skip
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rdx, qword ptr [rax]
	add	rdx, rcx
	mov	qword ptr [rax], rdx
	inc	rbp
.Ladd_skip:
	inc	r12
	jmp	.Lloop_start

.Lsub:
	inc	r11
	cmp	rbp, 2
	jb	.Lsub_skip
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rdx, qword ptr [rax]
	sub	rdx, rcx
	mov	qword ptr [rax], rdx
	inc	rbp
.Lsub_skip:
	inc	r12
	jmp	.Lloop_start

.Lmul:
	inc	r11
	cmp	rbp, 2
	jb	.Lmul_skip
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rdx, qword ptr [rax]
	mov	rax, rdx
	mul	rcx
	mov	qword ptr [rbp*8 + r15], rax
	inc	rbp
.Lmul_skip:
	inc	r12
	jmp	.Lloop_start

.Ldup:
	inc	r11
	cmp	rbp, 0
	je	.Ldup_skip
	cmp	rbp, rbx
	jae	.Ldup_skip
	mov	rax, rbp
	sub	rax, 1
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	qword ptr [rax], rcx
	inc	rbp
.Ldup_skip:
	inc	r12
	jmp	.Lloop_start

.Lswap:
	inc	r11
	cmp	rbp, 2
	jb	.Lswap_skip
	mov	rax, rbp
	sub	rax, 1
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	mov	rdx, rbp
	sub	rdx, 2
	shl	rdx, 3
	add	rdx, r15
	mov	r8, qword ptr [rdx]
	mov	qword ptr [rax], r8
	mov	qword ptr [rdx], rcx
.Lswap_skip:
	inc	r12
	jmp	.Lloop_start

.Ldrop:
	inc	r11
	cmp	rbp, 0
	je	.Ldrop_skip
	dec	rbp
.Ldrop_skip:
	inc	r12
	jmp	.Lloop_start

.Lover:
	inc	r11
	cmp	rbp, 2
	jb	.Lover_skip
	cmp	rbp, rbx
	jae	.Lover_skip
	mov	rax, rbp
	sub	rax, 2
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	qword ptr [rax], rcx
	inc	rbp
.Lover_skip:
	inc	r12
	jmp	.Lloop_start

.Llt:
	inc	r11
	cmp	rbp, 2
	jb	.Llt_skip
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rdx, qword ptr [rax]
	xor	r8, r8
	cmp	rdx, rcx
	setl	r8b
	mov	qword ptr [rax], r8
	inc	rbp
.Llt_skip:
	inc	r12
	jmp	.Lloop_start

.Ljz:
	inc	r11
	cmp	rbp, 0
	je	.Ljz_inc
	dec	rbp
	mov	rax, rbp
	shl	rax, 3
	add	rax, r15
	mov	rcx, qword ptr [rax]
	cmp	rcx, 0
	jne	.Ljz_nz
	mov	rax, r12
	add	rax, 1
	shl	rax, 3
	add	rax, r13
	mov	r12, qword ptr [rax]
	jmp	.Lloop_start
.Ljz_nz:
	inc	rbp
.Ljz_inc:
	add	r12, 2
	jmp	.Lloop_start

.Ljmp:
	inc	r11
	mov	rax, r12
	add	rax, 1
	shl	rax, 3
	add	rax, r13
	mov	r12, qword ptr [rax]
	jmp	.Lloop_start

.Linc_pc1:
	inc	r11
	inc	r12
	jmp	.Lloop_start

.Lstop_exec:
	xor	rax, rax
	cmp	rbp, 0
	je	.Lclear_stack
	mov	rax, rbp
	sub	rax, 1
	shl	rax, 3
	add	rax, r15
	mov	rax, qword ptr [rax]

.Lclear_stack:
	xor	rcx, rcx
.Lclear_loop:
	cmp	rcx, rbx
	jae	.Ldone_clear
	mov	rdx, rcx
	shl	rdx, 3
	add	rdx, r15
	mov	qword ptr [rdx], 0
	inc	rcx
	jmp	.Lclear_loop

.Ldone_clear:
	add	rsp, 8
	pop	r15
	pop	r14
	pop	r13
	pop	r12
	pop	rbp
	pop	rbx
	ret
