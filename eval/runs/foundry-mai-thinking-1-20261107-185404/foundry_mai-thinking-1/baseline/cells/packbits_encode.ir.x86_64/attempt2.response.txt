export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_i_n =w csltl %i, %n
	jnz %cond_i_n, @compute_run, @done

@compute_run
	%j_run =l copy %i
	jmp @run_entry

@run_entry
	%cond_j_n =w csltl %j_run, %n
	jnz %cond_j_n, @run_has_byte, @run_zero

@run_has_byte
	%ptr_b =l add %src, %j_run
	%b_byte_w =w loadub %ptr_b
	%limit1_run =l sub %n, %j_run
	%limit_run =l copy %limit1_run
	%cond_lim128 =w csgtl %limit_run, 128
	jnz %cond_lim128, @run_set128, @run_loop_init

@run_set128
	%limit_run =l copy 128
	jmp @run_loop_init

@run_loop_init
	%k_run =l copy 1
	jmp @run_loop

@run_loop
	%cond_k_lim =w csltl %k_run, %limit_run
	jnz %cond_k_lim, @run_check_eq, @run_end

@run_check_eq
	%pjk =l add %j_run, %k_run
	%ptr_c =l add %src, %pjk
	%c_byte_w =w loadub %ptr_c
	%eq_bc =w ceql %c_byte_w, %b_byte_w
	jnz %eq_bc, @run_inc_k, @run_end

@run_inc_k
	%k_run =l add %k_run, 1
	jmp @run_loop

@run_zero
	%r_res =l copy 0
	jmp @after_run

@run_end
	%r_res =l copy %k_run
	jmp @after_run

@after_run
	%r_cur =l copy %r_res
	%cond_r_ge3 =w csgtl %r_cur, 2
	jnz %cond_r_ge3, @repeat_check, @literal_check

@repeat_check
	%need_rep =l add %o, 2
	%cond_need_cap =w csgtl %need_rep, %cap
	jnz %cond_need_cap, @fail, @emit_repeat

@emit_repeat
	%hdr_rep_l =l sub 257, %r_cur
	%ptr_out_o_r =l add %out, %o
	storeb %hdr_rep_l, %ptr_out_o_r
	%o1_rep =l add %o, 1
	%ptr_src_i_r =l add %src, %i
	%b_src_w_r =w loadub %ptr_src_i_r
	%b_src_l_r =l extub %b_src_w_r
	%ptr_out_o1_r =l add %out, %o1_rep
	storeb %b_src_l_r, %ptr_out_o1_r
	%o2_rep =l add %o, 2
	%i2_rep =l add %i, %r_cur
	%o =l copy %o2_rep
	%i =l copy %i2_rep
	jmp @loop

@literal_check
	%L_lit =l copy 0
	%p_lit =l copy %i
	jmp @lit_build

@lit_build
	%cond_p_n_lb =w csltl %p_lit, %n
	jnz %cond_p_n_lb, @lit_check_L, @lit_found

@lit_check_L
	%cond_L_128_lb =w csltl %L_lit, 128
	jnz %cond_L_128_lb, @lit_run_prep, @lit_found

@lit_run_prep
	%j_rp =l copy %p_lit
	jmp @lit_run_entry

@lit_run_entry
	%cond_jrp_n_lr =w csltl %j_rp, %n
	jnz %cond_jrp_n_lr, @lit_run_has, @lit_run_zero

@lit_run_has
	%ptr_bp_lr =l add %src, %j_rp
	%b_bp_w_lr =w loadub %ptr_bp_lr
	%lim1_rp_lr =l sub %n, %j_rp
	%lim_rp_lr =l copy %lim1_rp_lr
	%cond_lr128_lr =w csgtl %lim_rp_lr, 128
	jnz %cond_lr128_lr, @lit_run_set128, @lit_run_loop_init

@lit_run_set128
	%lim_rp_lr =l copy 128
	jmp @lit_run_loop_init

@lit_run_loop_init
	%k_rp_lr =l copy 1
	jmp @lit_run_loop

@lit_run_loop
	%cond_kr_lr_ll =w csltl %k_rp_lr, %lim_rp_lr
	jnz %cond_kr_lr_ll, @lit_run_ceq, @lit_run_end

@lit_run_ceq
	%pjkr_lr =l add %j_rp, %k_rp_lr
	%ptr_cp_lr =l add %src, %pjkr_lr
	%c_bp_w_lr =w loadub %ptr_cp_lr
	%eq_bcp_lr =w ceql %c_bp_w_lr, %b_bp_w_lr
	jnz %eq_bcp_lr, @lit_run_inc, @lit_run_end

@lit_run_inc
	%k_rp_lr =l add %k_rp_lr, 1
	jmp @lit_run_loop

@lit_run_zero
	%rr_res_lr =l copy 0
	jmp @lit_run_after

@lit_run_end
	%rr_res_lr =l copy %k_rp_lr
	jmp @lit_run_after

@lit_run_after
	%rr_cur_lr =l copy %rr_res_lr
	%cond_rr_ge3_lr =w csgtl %rr_cur_lr, 2
	jnz %cond_rr_ge3_lr, @lit_found, @lit_add

@lit_add
	%L_new_la =l add %L_lit, %rr_cur_lr
	%p_new_la =l add %p_lit, %rr_cur_lr
	%L_lit =l copy %L_new_la
	%p_lit =l copy %p_new_la
	jmp @lit_build

@lit_found
	%L_f_lf =l copy %L_lit
	%need1_l_lf =l add %o, 1
	%need2_l_lf =l add %need1_l_lf, %L_f_lf
	%cond_nl_cap_lf =w csgtl %need2_l_lf, %cap
	jnz %cond_nl_cap_lf, @fail, @emit_literal

@emit_literal
	%hdr_l_el =l sub %L_f_lf, 1
	%ptr_ol0_el =l add %out, %o
	storeb %hdr_l_el, %ptr_ol0_el
	%o1_l_el =l add %o, 1
	%k_cpy_el =l copy 0
	jmp @lit_copy

@lit_copy
	%cond_kc_L_lc =w csltl %k_cpy_el, %L_f_lf
	jnz %cond_kc_L_lc, @lit_copy_byte, @lit_copy_done

@lit_copy_byte
	%psrc_c_lcb =l add %i, %k_cpy_el
	%ptr_sb_lcb =l add %src, %psrc_c_lcb
	%b_c_w_lcb =w loadub %ptr_sb_lcb
	%b_c_l_lcb =l extub %b_c_w_lcb
	%pout_c_lcb =l add %o1_l_el, %k_cpy_el
	%ptr_ob_lcb =l add %out, %pout_c_lcb
	storeb %b_c_l_lcb, %ptr_ob_lcb
	%k_cpy_new_lcb =l add %k_cpy_el, 1
	%k_cpy_el =l copy %k_cpy_new_lcb
	jmp @lit_copy

@lit_copy_done
	%o3_l_ld =l add %o1_l_el, %L_f_lf
	%i3_l_ld =l add %i, %L_f_lf
	%o =l copy %o3_l_ld
	%i =l copy %i3_l_ld
	jmp @loop

@done
	%ret_res_d =l copy %o
	ret %ret_res_d

@fail
	%ret_fail_f =l copy -1
	ret %ret_fail_f
}
