export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_i_n =w csltl %i, %n
	jnz %cond_i_n, @compute_run, @done

@compute_run
	%j_run =l copy %i
	jmp @run_entry

@run_entry
	%cond_j_n =w csltl %j_run, %n
	jnz %cond_j_n, @run_has_byte, @run_zero

@run_has_byte
	%ptr_b =l add %src, %j_run
	%b_byte =w loadub %ptr_b
	%k_run =l copy 1
	%limit1_run =l sub %n, %j_run
	%limit_run =l copy %limit1_run
	%cond_lim128 =w csgtl %limit_run, 128
	jnz %cond_lim128, @run_set128, @run_loop

@run_set128
	%limit_run =l copy 128
	jmp @run_loop

@run_loop
	%cond_k_lim =w csltl %k_run, %limit_run
	jnz %cond_k_lim, @run_check_eq, @run_end

@run_check_eq
	%pjk =l add %j_run, %k_run
	%ptr_c =l add %src, %pjk
	%c_byte =w loadub %ptr_c
	%eq_bc =w ceql %c_byte, %b_byte
	jnz %eq_bc, @run_inc_k, @run_end

@run_inc_k
	%k_run =l add %k_run, 1
	jmp @run_loop

@run_zero
	%r_res =l copy 0
	jmp @after_run

@run_end
	%r_res =l copy %k_run
	jmp @after_run

@after_run
	%r_cur =l copy %r_res
	%cond_r_ge3 =w csgtl %r_cur, 2
	jnz %cond_r_ge3, @repeat_check, @literal_check

@repeat_check
	%need_rep =l add %o, 2
	%cond_need_cap =w csgtl %need_rep, %cap
	jnz %cond_need_cap, @fail, @emit_repeat

@emit_repeat
	%hdr_rep =l sub 257, %r_cur
	%ptr_out_o =l add %out, %o
	storeb %hdr_rep, %ptr_out_o
	%o1_rep =l add %o, 1
	%ptr_src_i =l add %src, %i
	%b_src =w loadub %ptr_src_i
	%b_src_l =l extub %b_src
	%ptr_out_o1 =l add %out, %o1_rep
	storeb %b_src_l, %ptr_out_o1
	%o2_rep =l add %o, 2
	%i2_rep =l add %i, %r_cur
	%o =l copy %o2_rep
	%i =l copy %i2_rep
	jmp @loop

@literal_check
	%L_lit =l copy 0
	%p_lit =l copy %i
	jmp @lit_build

@lit_build
	%cond_p_n =w csltl %p_lit, %n
	jnz %cond_p_n, @lit_check_L, @lit_found

@lit_check_L
	%cond_L_128 =w csltl %L_lit, 128
	jnz %cond_L_128, @lit_run_p, @lit_found

@lit_run_p
	%j_rp =l copy %p_lit
	jmp @lit_run_entry

@lit_run_entry
	%cond_jrp_n =w csltl %j_rp, %n
	jnz %cond_jrp_n, @lit_run_has, @lit_run_zero

@lit_run_has
	%ptr_bp =l add %src, %j_rp
	%b_bp =w loadub %ptr_bp
	%k_rp =l copy 1
	%lim1_rp =l sub %n, %j_rp
	%lim_rp =l copy %lim1_rp
	%cond_lr128 =w csgtl %lim_rp, 128
	jnz %cond_lr128, @lit_run_set128, @lit_run_loop

@lit_run_set128
	%lim_rp =l copy 128
	jmp @lit_run_loop

@lit_run_loop
	%cond_kr_lr =w csltl %k_rp, %lim_rp
	jnz %cond_kr_lr, @lit_run_ceq, @lit_run_end

@lit_run_ceq
	%pjkr =l add %j_rp, %k_rp
	%ptr_cp =l add %src, %pjkr
	%c_bp =w loadub %ptr_cp
	%eq_bcp =w ceql %c_bp, %b_bp
	jnz %eq_bcp, @lit_run_inc, @lit_run_end

@lit_run_inc
	%k_rp =l add %k_rp, 1
	jmp @lit_run_loop

@lit_run_zero
	%rr_res =l copy 0
	jmp @lit_run_after

@lit_run_end
	%rr_res =l copy %k_rp
	jmp @lit_run_after

@lit_run_after
	%rr_cur =l copy %rr_res
	%cond_rr_ge3 =w csgtl %rr_cur, 2
	jnz %cond_rr_ge3, @lit_found, @lit_add

@lit_add
	%L_new =l add %L_lit, %rr_cur
	%p_new =l add %p_lit, %rr_cur
	%L_lit =l copy %L_new
	%p_lit =l copy %p_new
	jmp @lit_build

@lit_found
	%L_f =l copy %L_lit
	%need1_l =l add %o, 1
	%need2_l =l add %need1_l, %L_f
	%cond_nl_cap =w csgtl %need2_l, %cap
	jnz %cond_nl_cap, @fail, @emit_literal

@emit_literal
	%hdr_l =l sub %L_f, 1
	%ptr_ol0 =l add %out, %o
	storeb %hdr_l, %ptr_ol0
	%o1_l =l add %o, 1
	%k_cpy =l copy 0
	jmp @lit_copy

@lit_copy
	%cond_kc_L =w csltl %k_cpy, %L_f
	jnz %cond_kc_L, @lit_copy_byte, @lit_copy_done

@lit_copy_byte
	%psrc_c =l add %i, %k_cpy
	%ptr_sb =l add %src, %psrc_c
	%b_c =w loadub %ptr_sb
	%b_cl =l extub %b_c
	%pout_c =l add %o1_l, %k_cpy
	%ptr_ob =l add %out, %pout_c
	storeb %b_cl, %ptr_ob
	%k_cpy_new =l add %k_cpy, 1
	%k_cpy =l copy %k_cpy_new
	jmp @lit_copy

@lit_copy_done
	%o3_l =l add %o1_l, %L_f
	%i3_l =l add %i, %L_f
	%o =l copy %o3_l
	%i =l copy %i3_l
	jmp @loop

@done
	%ret_res =l copy %o
	ret %ret_res

@fail
	%ret_fail =l copy -1
	ret %ret_fail
}
