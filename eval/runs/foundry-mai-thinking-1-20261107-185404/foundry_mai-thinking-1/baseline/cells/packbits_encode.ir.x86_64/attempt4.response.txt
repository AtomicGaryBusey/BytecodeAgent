export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_i_n =w csltl %i, %n
	jnz %cond_i_n, @compute_run, @done

@compute_run
	%j_run =l copy %i
	%ptr_b0 =l add %src, %j_run
	%b0_w =w loadub %ptr_b0
	%limit_run_base =l sub %n, %j_run
	%limit_run_cur =l copy %limit_run_base
	%cond_lim_gt128 =w csgtl %limit_run_cur, 128
	jnz %cond_lim_gt128, @set_lim128, @run_loop_init

@set_lim128
	%limit_run_cur =l copy 128
	jmp @run_loop_init

@run_loop_init
	%k_run =l copy 1
	jmp @run_loop

@run_loop
	%cond_k_lt_lim =w csltl %k_run, %limit_run_cur
	jnz %cond_k_lt_lim, @run_check_eq, @run_end

@run_check_eq
	%pjk =l add %j_run, %k_run
	%ptr_ck =l add %src, %pjk
	%ck_w =w loadub %ptr_ck
	%eq_bc =w ceqw %ck_w, %b0_w
	jnz %eq_bc, @run_inc, @run_end

@run_inc
	%k_run =l add %k_run, 1
	jmp @run_loop

@run_end
	%r_cur_run =l copy %k_run
	%cond_r_ge3 =w csgtl %r_cur_run, 2
	jnz %cond_r_ge3, @repeat_check, @literal_start

@repeat_check
	%need_rep =l add %o, 2
	%cond_rep_fit =w csleql %need_rep, %cap
	jnz %cond_rep_fit, @emit_repeat, @fail

@emit_repeat
	%hdr_rep =l sub 257, %r_cur_run
	%ptr_or0 =l add %out, %o
	storeb %hdr_rep, %ptr_or0
	%o1_r =l add %o, 1
	%ptr_si =l add %src, %i
	%bi_w =w loadub %ptr_si
	%bi_l =l extub %bi_w
	%ptr_or1 =l add %out, %o1_r
	storeb %bi_l, %ptr_or1
	%o2_r =l add %o, 2
	%i2_r =l add %i, %r_cur_run
	%o =l copy %o2_r
	%i =l copy %i2_r
	jmp @loop

@literal_start
	%L_lit =l copy 0
	%p_lit =l copy %i
	jmp @lit_build

@lit_build
	%cond_p_lt_n =w csltl %p_lit, %n
	jnz %cond_p_lt_n, @lit_check_l128, @lit_found

@lit_check_l128
	%cond_l_lt128 =w csltl %L_lit, 128
	jnz %cond_l_lt128, @lit_run_prep, @lit_found

@lit_run_prep
	%j_lr =l copy %p_lit
	%ptr_bl0 =l add %src, %j_lr
	%bl0_w =w loadub %ptr_bl0
	%lim_lr_base =l sub %n, %j_lr
	%lim_lr_cur =l copy %lim_lr_base
	%cond_lr_gt128 =w csgtl %lim_lr_cur, 128
	jnz %cond_lr_gt128, @set_lr128, @lit_run_init

@set_lr128
	%lim_lr_cur =l copy 128
	jmp @lit_run_init

@lit_run_init
	%k_lr =l copy 1
	jmp @lit_run_loop

@lit_run_loop
	%cond_kl_lt_lim =w csltl %k_lr, %lim_lr_cur
	jnz %cond_kl_lt_lim, @lit_run_check, @lit_run_end

@lit_run_check
	%pjkl =l add %j_lr, %k_lr
	%ptr_cl =l add %src, %pjkl
	%cl_w =w loadub %ptr_cl
	%eq_bl_cl =w ceqw %cl_w, %bl0_w
	jnz %eq_bl_cl, @lit_run_inc, @lit_run_end

@lit_run_inc
	%k_lr =l add %k_lr, 1
	jmp @lit_run_loop

@lit_run_end
	%rr_cur =l copy %k_lr
	%cond_rr_ge3 =w csgtl %rr_cur, 2
	jnz %cond_rr_ge3, @lit_found, @lit_add

@lit_add
	%L_new =l add %L_lit, 1
	%p_new =l add %p_lit, 1
	%L_lit =l copy %L_new
	%p_lit =l copy %p_new
	jmp @lit_build

@lit_found
	%L_f =l copy %L_lit
	%need1_l =l add %o, 1
	%need2_l =l add %need1_l, %L_f
	%cond_l_fit =w csleql %need2_l, %cap
	jnz %cond_l_fit, @emit_literal, @fail

@emit_literal
	%hdr_l =l sub %L_f, 1
	%ptr_ol0 =l add %out, %o
	storeb %hdr_l, %ptr_ol0
	%o1_l =l add %o, 1
	%k_cpy =l copy 0
	jmp @lit_copy

@lit_copy
	%cond_kc_lt_l =w csltl %k_cpy, %L_f
	jnz %cond_kc_lt_l, @lit_copy_byte, @lit_copy_done

@lit_copy_byte
	%psrc_c =l add %i, %k_cpy
	%ptr_sb =l add %src, %psrc_c
	%sb_w =w loadub %ptr_sb
	%sb_lv =l extub %sb_w
	%pout_c =l add %o1_l, %k_cpy
	%ptr_ob =l add %out, %pout_c
	storeb %sb_lv, %ptr_ob
	%k_cpy_new =l add %k_cpy, 1
	%k_cpy =l copy %k_cpy_new
	jmp @lit_copy

@lit_copy_done
	%o_new_l =l add %o1_l, %L_f
	%i_new_l =l add %i, %L_f
	%o =l copy %o_new_l
	%i =l copy %i_new_l
	jmp @loop

@done
	%ret_ok =l copy %o
	ret %ret_ok

@fail
	%ret_fail =l copy -1
	ret %ret_fail
}
