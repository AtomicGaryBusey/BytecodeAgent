export function l $fill_down(l %a, l %n, l %start) {
@entry
	%iszero =w ceql %n, 0
	jnz %iszero, @ret, @loop_init

@loop_init
	%i =l copy 0
	jmp @loop_cond

@loop_cond
	%lt =w csltl %i, %n
	jnz %lt, @loop_body, @ret

@loop_body
	%val =l sub %start, %i
	%off =l mul %i, 8
	%ptr =l add %a, %off
	storel %val, %ptr
	%inext =l add %i, 1
	%i =l copy %inext
	jmp @loop_cond

@ret
	ret %n
}
