export function l $fill_down(l %a, l %n, l %start) {
@entry
	%iszero = ceql %n, 0
	jnz %iszero, @ret, @loop_init

@loop_init
	%i = copy 0
	jmp @loop_cond

@loop_cond
	%lt = csltl %i, %n
	jnz %lt, @loop_body, @ret

@loop_body
	%val = sub %start, %i
	%off = mul %i, 8
	%ptr = add %a, %off
	storel %val, %ptr
	%inext = add %i, 1
	%i = copy %inext
	jmp @loop_cond

@ret
	ret %n
}
