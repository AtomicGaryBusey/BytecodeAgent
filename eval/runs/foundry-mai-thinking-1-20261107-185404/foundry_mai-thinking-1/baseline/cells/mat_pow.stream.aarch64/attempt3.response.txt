mat_pow:
cbz x2, .Lk_zero

// Save callee-saved registers
stp x29, x30, [sp, #-16]!
mov x29, sp
stp x19, x20, [sp, #-16]!
stp x21, x22, [sp, #-16]!
stp x23, x24, [sp, #-16]!

// Arguments: x0=a, x1=n, x2=k, x3=out, x4=tmp
mov x19, x0           // a
mov x20, x1           // n
mov x21, x2           // k
mov x22, x3           // out
mov x23, x4           // tmp

// Compute n*n elements
mul x9, x20, x20      // x9 = n*n
mov x24, x9           // elements count

// Compute sizes in bytes
lsl x10, x9, #3       // x10 = n*n * 8

// Copy a to tmp[0] (M_buf)
mov x11, x19          // src a
mov x12, x23          // dst tmp base
mov x13, x24          // count elements
cbz x13, .Lafter_copy

.Lcopy_a_loop:
sub x13, x13, #1
ldr x14, [x11, x13, lsl #3]
str x14, [x12, x13, lsl #3]
cbnz x13, .Lcopy_a_loop

.Lafter_copy:

// Set identity matrix in out (R)
mov x11, x22          // dst out
mov x13, x24          // count elements
cbz x13, .Lid_diag

.Lzero_out_loop:
sub x13, x13, #1
str xzr, [x11, x13, lsl #3]
cbnz x13, .Lzero_out_loop

.Lid_diag:
mov x11, x20          // n
mov x12, x22          // out
mov x13, xzr          // i
cmp x13, x11
b.ge .Lexp_start

.Lid_diag_loop:
mul x14, x13, x11     // i*n
add x14, x14, x13     // i*n + i
mov x15, #1
str x15, [x12, x14, lsl #3]
add x13, x13, #1
cmp x13, x11
b.lt .Lid_diag_loop

.Lexp_start:
mov x9, x21           // k
mov x5, x23           // M_buf
add x6, x23, x10      // P_buf
mov x7, x22           // R

.Lexp_loop:
cbz x9, .Lexp_done

// Check lowest bit
tbz x9, #0, .Lno_mul_r

// R = R * M_buf -> P_buf
mov x10, x7           // A = R
mov x11, x5           // B = M_buf
mov x12, x6           // C = P_buf
mov x13, x20          // n
bl .Lmat_mul_inline

// Copy P_buf to R
mov x10, x7           // dst
mov x11, x6           // src
mov x12, x24          // elements
cbz x12, .Lno_mul_r

.Lcopy_r_loop:
sub x12, x12, #1
ldr x14, [x11, x12, lsl #3]
str x14, [x10, x12, lsl #3]
cbnz x12, .Lcopy_r_loop

.Lno_mul_r:

// Square M_buf = M_buf * M_buf -> P_buf
mov x10, x5           // A
mov x11, x5           // B
mov x12, x6           // C
mov x13, x20          // n
bl .Lmat_mul_inline

// Copy P_buf to M_buf
mov x10, x5           // dst
mov x11, x6           // src
mov x12, x24          // elements
cbz x12, .Lcopied_m

.Lcopy_m_loop:
sub x12, x12, #1
ldr x14, [x11, x12, lsl #3]
str x14, [x10, x12, lsl #3]
cbnz x12, .Lcopy_m_loop

.Lcopied_m:

// Right shift k
lsr x9, x9, #1
b .Lexp_loop

.Lexp_done:

// Zero tmp: 2*n*n elements
mov x10, x23          // tmp base
mov x11, x24          // n*n
lsl x12, x11, #1      // 2*n*n
cbz x12, .Lzero_tmp_done

.Lzero_tmp_loop:
sub x12, x12, #1
str xzr, [x10, x12, lsl #3]
cbnz x12, .Lzero_tmp_loop

.Lzero_tmp_done:

mov x0, xzr
ldp x23, x24, [sp], #16
ldp x21, x22, [sp], #16
ldp x19, x20, [sp], #16
ldp x29, x30, [sp], #16
ret

.Lk_zero:

// k == 0
stp x29, x30, [sp, #-16]!
mov x29, sp
stp x19, x20, [sp, #-16]!
stp x21, x22, [sp, #-16]!

mov x19, x1           // n
mov x20, x3           // out
mov x21, x4           // tmp

mul x9, x19, x19      // n*n
mov x11, x9           // elements

// Zero out
mov x10, x20
cbz x11, .Lid_k0_diag

.Lzero_out_k0_loop:
sub x11, x11, #1
str xzr, [x10, x11, lsl #3]
cbnz x11, .Lzero_out_k0_loop

.Lid_k0_diag:
mov x10, x19          // n
mov x12, x20          // out
mov x11, xzr          // i
cmp x11, x10
b.ge .Lzero_tmp_k0

.Lid_k0_diag_loop:
mul x13, x11, x10
add x13, x13, x11
mov x14, #1
str x14, [x12, x13, lsl #3]
add x11, x11, #1
cmp x11, x10
b.lt .Lid_k0_diag_loop

.Lzero_tmp_k0:
mov x10, x21          // tmp
mov x11, x9           // n*n
lsl x12, x11, #1      // 2*n*n
cbz x12, .Lret0_k0

.Lzero_tmp_k0_loop:
sub x12, x12, #1
str xzr, [x10, x12, lsl #3]
cbnz x12, .Lzero_tmp_k0_loop

.Lret0_k0:
mov x0, xzr
ldp x21, x22, [sp], #16
ldp x19, x20, [sp], #16
ldp x29, x30, [sp], #16
ret

// Matrix multiply: C = A * B, n in x13
// Inputs: x10=A, x11=B, x12=C, x13=n
.Lmat_mul_inline:
stp x29, x30, [sp, #-16]!
mov x29, sp
stp x19, x20, [sp, #-16]!
stp x21, x22, [sp, #-16]!
stp x23, x24, [sp, #-16]!

mov x19, x10          // A
mov x20, x11          // B
mov x21, x12          // C
mov x22, x13          // n

mov x23, xzr          // i
cmp x23, x22
b.ge .Lmul_done

.Lmul_i_loop:
mov x24, xzr          // j
cmp x24, x22
b.ge .Lmul_i_next

.Lmul_j_loop:
mov x9, xzr           // sum
mov x11, xzr          // t
cmp x11, x22
b.ge .Lmul_store

.Lmul_t_loop:
mul x12, x23, x22     // i*n
add x12, x12, x11     // i*n + t
ldr x10, [x19, x12, lsl #3]

mul x14, x11, x22     // t*n
add x14, x14, x24     // t*n + j
ldr x15, [x20, x14, lsl #3]

mul x10, x10, x15
add x9, x9, x10

add x11, x11, #1
cmp x11, x22
b.lt .Lmul_t_loop

.Lmul_store:
mul x12, x23, x22
add x12, x12, x24
str x9, [x21, x12, lsl #3]

add x24, x24, #1
cmp x24, x22
b.lt .Lmul_j_loop

.Lmul_i_next:
add x23, x23, #1
cmp x23, x22
b.lt .Lmul_i_loop

.Lmul_done:
ldp x23, x24, [sp], #16
ldp x21, x22, [sp], #16
ldp x19, x20, [sp], #16
ldp x29, x30, [sp], #16
ret
