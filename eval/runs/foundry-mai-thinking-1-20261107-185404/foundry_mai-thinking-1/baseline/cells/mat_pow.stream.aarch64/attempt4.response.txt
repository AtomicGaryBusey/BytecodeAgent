mat_pow:
cbz x2, .Lk_zero

// Save callee-saved registers
stp x29, x30, [sp, #-16]!
mov x29, sp
stp x19, x20, [sp, #-16]!
stp x21, x22, [sp, #-16]!
stp x23, x24, [sp, #-16]!
stp x25, x26, [sp, #-16]!

// Arguments: x0=a, x1=n, x2=k, x3=out, x4=tmp
mov x19, x0           // a
mov x20, x1           // n
mov x21, x2           // k
mov x22, x3           // out
mov x23, x4           // tmp

// Compute n*n
mul x24, x20, x20     // n*n elements

// Compute size in bytes
lsl x25, x24, #3      // n*n * 8

// Copy a to tmp[0] (M)
mov x0, x19           // src
mov x1, x23           // dst
mov x2, x24           // count
cbz x2, .Lafter_copy_m

.Lcopy_m_base_loop:
sub x2, x2, #1
ldr x3, [x0, x2, lsl #3]
str x3, [x1, x2, lsl #3]
cbnz x2, .Lcopy_m_base_loop

.Lafter_copy_m:

// Set identity in out (R)
mov x0, x22           // dst out
mov x2, x24           // count
cbz x2, .Lset_id_diag

.Lzero_out_exp_loop:
sub x2, x2, #1
str xzr, [x0, x2, lsl #3]
cbnz x2, .Lzero_out_exp_loop

.Lset_id_diag:
mov x0, x22           // out
mov x1, x20           // n
mov x2, xzr           // i
cmp x2, x1
b.ge .Lexp_start

.Lset_id_diag_loop:
mul x3, x2, x1
add x3, x3, x2
mov x4, #1
str x4, [x0, x3, lsl #3]
add x2, x2, #1
cmp x2, x1
b.lt .Lset_id_diag_loop

.Lexp_start:
mov x26, x21          // k
mov x19, x23          // M base
add x20_ptr, x23, x25 // P base (use different name)
