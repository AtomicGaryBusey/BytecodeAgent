export function l $mulhi_u64(l %a, l %b) {
@start
	%a_lo = and %a, 0xffffffffl
	%a_hi = shr %a, 32
	%b_lo = and %b, 0xffffffffl
	%b_hi = shr %b, 32
	%p0 = mul %a_lo, %b_lo
	%p1 = mul %a_lo, %b_hi
	%p2 = mul %a_hi, %b_lo
	%p3 = mul %a_hi, %b_hi
	%t0 = shr %p0, 32
	%t1 = and %p1, 0xffffffffl
	%t2 = and %p2, 0xffffffffl
	%m0 = add %t0, %t1
	%m1 = add %m0, %t2
	%c = shr %m1, 32
	%h0 = shr %p1, 32
	%h1 = shr %p2, 32
	%r0 = add %p3, %h0
	%r1 = add %r0, %h1
	%res = add %r1, %c
	ret %res
}
