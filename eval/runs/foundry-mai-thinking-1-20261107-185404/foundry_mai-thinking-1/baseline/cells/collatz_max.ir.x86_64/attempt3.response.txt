export function l $collatz_max(l %n) {
@start
	%max = 0l
	%k = 1l
	jmp @loop_k
@loop_k
	%c_le = csltl %k, %n
	%c_eq = ceql %k, %n
	%ok = orw %c_le, %c_eq
	jnz %ok, @do_k, @end
@do_k
	%x = %k
	%steps = 0l
	jmp @loop_c
@loop_c
	%is_one = ceql %x, 1l
	jnz %is_one, @end_c, @body_c
@body_c
	%rem = reml %x, 2l
	%is_even = ceql %rem, 0l
	jnz %is_even, @even, @odd
@even
	%x = divl %x, 2l
	jmp @next_c
@odd
	%x3 = mull %x, 3l
	%x = addl %x3, 1l
	jmp @next_c
@next_c
	%steps = addl %steps, 1l
	jmp @loop_c
@end_c
	%better = csgtl %steps, %max
	jnz %better, @update_max, @no_update
@update_max
	%max = %steps
@no_update
	%k = addl %k, 1l
	jmp @loop_k
@end
	ret %max
}
