export function l $collatz_max(l %n) {
@start
	%max = 0l
	%k = 1l
	jmp @loop_k
@loop_k
	%le = csltl %k, %n
	%eq = ceql %k, %n
	%ok = orw %le, %eq
	jnz %ok, @do_k, @end
@do_k
	%x = %k
	%steps = 0l
	jmp @loop_c
@loop_c
	%one = ceql %x, 1l
	jnz %one, @end_c, @body_c
@body_c
	%r = reml %x, 2l
	%even = ceql %r, 0l
	jnz %even, @even, @odd
@even
	%x = divl %x, 2l
	jmp @next_c
@odd
	%x3 = mull %x, 3l
	%x = addl %x3, 1l
	jmp @next_c
@next_c
	%steps = addl %steps, 1l
	jmp @loop_c
@end_c
	%gt = csgtl %steps, %max
	jnz %gt, @upd, @noupd
@upd
	%max = %steps
@noupd
	%k = addl %k, 1l
	jmp @loop_k
@end
	ret %max
}
