export function l $collatz_max(l %n) {
@start
	%max = copyl 0
	%k = copyl 1
@loop_k
	%c_le = csltl %k, %n
	%c_eq = ceql %k, %n
	%ok = or %c_le, %c_eq
	jnz %ok, @do_k, @end
@do_k
	%x = copyl %k
	%steps = copyl 0
@loop_c
	%one = copyl 1
	%is_one = ceql %x, %one
	jnz %is_one, @end_c, @body_c
@body_c
	%two = copyl 2
	%rem = reml %x, %two
	%is_even = ceql %rem, 0
	jnz %is_even, @even, @odd
@even
	%x_new = divl %x, %two
	%x = copyl %x_new
	jmp @next_c
@odd
	%x3 = mull %x, 3
	%x_new2 = addl %x3, 1
	%x = copyl %x_new2
	jmp @next_c
@next_c
	%steps_new = addl %steps, 1
	%steps = copyl %steps_new
	jmp @loop_c
@end_c
	%better = csltl %max, %steps
	jnz %better, @update_max, @no_update
@update_max
	%max = copyl %steps
@no_update
	%k_new = addl %k, 1
	%k = copyl %k_new
	jmp @loop_k
@end
	ret %max
}
