export function l $fib_rec(l %n) {
@entry
	%le0 =w cslel %n, 0
	jnz %le0, @ret0, @check1

@ret0
	ret 0

@check1
	%eq1 =w ceql %n, 1
	jnz %eq1, @ret1, @rec

@ret1
	ret 1

@rec
	%n1 =l sub %n, 1
	%n2 =l sub %n, 2
	%f1 =l call $fib_rec(l %n1)
	%f2 =l call $fib_rec(l %n2)
	%ans =l add %f1, %f2
	ret %ans
}
