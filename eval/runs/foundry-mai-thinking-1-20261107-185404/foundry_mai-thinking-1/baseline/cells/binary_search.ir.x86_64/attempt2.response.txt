export function l $binary_search(l %a, l %n, l %key) {
@start
	%iszero =w csle l %n, 0
	jnz %iszero, @notfound, @init

@init
	%left =l copy 0
	%right =l sub l %n, 1
	jmp @loop

@loop
	%cond =w csle l %left, %right
	jnz %cond, @body, @notfound

@body
	%sum =l add l %left, %right
	%mid =l sarl l %sum, 1
	%off =l mul l %mid, 8
	%ptr =l add l %a, %off
	%val =l loadl %ptr
	%eq =w ceq l %val, %key
	jnz %eq, @found, @check

@found
	ret %mid

@check
	%less =w cslt l %val, %key
	jnz %less, @leftinc, @rightdec

@leftinc
	%left =l add l %mid, 1
	jmp @loop

@rightdec
	%right =l sub l %mid, 1
	jmp @loop

@notfound
	ret -1
}
