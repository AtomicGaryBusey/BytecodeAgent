export function l $binary_search(l %a, l %n, l %key) {
@start
	%iszero =w cslel %n, 0
	jnz %iszero, @notfound, @init

@init
	%left =l copy 0
	%right =l sub %n, 1
	jmp @loop

@loop
	%cond =w cslel %left, %right
	jnz %cond, @body, @notfound

@body
	%sum =l add %left, %right
	%mid =l sar %sum, 1
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @check

@found
	ret %mid

@check
	%less =w csltl %val, %key
	jnz %less, @leftinc, @rightdec

@leftinc
	%left =l add %mid, 1
	jmp @loop

@rightdec
	%right =l sub %mid, 1
	jmp @loop

@notfound
	ret -1
}
