packbits_encode:
	push r12
	push r13
	push r14
	push r15
	cmp rsi, 0
	jne .Lnot_zero
	xor rax, rax
	jmp .Lexit
.Lnot_zero:
	mov r12, rdi
	mov r13, rsi
	mov r14, rdx
	mov r15, rcx
	xor r8, r8
	xor r9, r9
	cld
.Lloop:
	cmp r8, r13
	jl .Lprocess
	jmp .Lend_ok
.Lprocess:
	movzx r10, byte ptr [r12 + r8]
	xor r11, r11
.Lrun_loop:
	lea rax, [r8 + r11]
	cmp rax, r13
	jge .Lrun_done
	cmp r11, 128
	jge .Lrun_done
	movzx rdx, byte ptr [r12 + rax]
	cmp dl, r10b
	jne .Lrun_done
	inc r11
	jmp .Lrun_loop
.Lrun_done:
	cmp r11, 3
	jge .Lrepeat
.Lliteral:
	xor r11, r11
	mov rax, r8
.Llit_scan:
	cmp rax, r13
	jge .Llit_found
	cmp r11, 128
	jge .Llit_found
	lea rdx, [rax + 2]
	cmp rdx, r13
	jge .Llit_no_break
	mov cl, byte ptr [r12 + rax]
	mov dl, byte ptr [r12 + rax + 1]
	cmp cl, dl
	jne .Llit_no_break
	mov dl, byte ptr [r12 + rax + 2]
	cmp cl, dl
	je .Llit_found
.Llit_no_break:
	inc rax
	inc r11
	jmp .Llit_scan
.Llit_found:
	lea rax, [r9 + r11]
	inc rax
	cmp rax, r15
	jg .Lret_minus1
	mov rax, r11
	dec rax
	mov byte ptr [r14 + r9], al
	mov rsi, r12
	add rsi, r8
	lea rdi, [r14 + r9 + 1]
	mov rcx, r11
	rep movsb
	add r9, r11
	inc r9
	add r8, r11
	jmp .Lloop
.Lrepeat:
	lea rax, [r9 + 2]
	cmp rax, r15
	jg .Lret_minus1
	mov rax, 257
	sub rax, r11
	mov byte ptr [r14 + r9], al
	mov al, byte ptr [r12 + r8]
	mov byte ptr [r14 + r9 + 1], al
	add r9, 2
	add r8, r11
	jmp .Lloop
.Lend_ok:
	mov rax, r9
	jmp .Lexit
.Lret_minus1:
	mov rax, -1
.Lexit:
	pop r15
	pop r14
	pop r13
	pop r12
	ret
