export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%zero  = l add 0, 0
	%one   = l add 0, 1
	%two   = l add 0, 2
	%three = l add 0, 3
	%four  = l add 0, 4
	%six   = l add 0, 6
	%sixty = l add 0, 63

	%addn  = l add %n, %two
	%grp   = l div %addn, %three
	%rc    = l mul %grp, %four

	%i     = l add 0, 0
	%o     = l add 0, 0

@loop
	%lim3  = l sub %n, %three
	%c3g   = w csltl %i, %lim3
	jnz    %c3g, @loop_body, @remainder

@loop_body
	%p0    = l add %src, %i
	%b0    = l loadub %p0

	%i1    = l add %i, %one
	%p1    = l add %src, %i1
	%b1    = l loadub %p1

	%i2    = l add %i, %two
	%p2    = l add %src, %i2
	%b2    = l loadub %p2

	%v0s   = l shr %b0, %two
	%v0    = l and %v0s, %sixty

	%t1a   = l and %b0, %three
	%t1b   = l shl %t1a, %four
	%t2a   = l shr %b1, %four
	%t2b   = l and %t2a, 15
	%v1    = l or  %t1b, %t2b

	%t3a   = l and %b1, 15
	%t3b   = l shl %t3a, %two
	%t4a   = l shr %b2, %six
	%t4b   = l and %t4a, %three
	%v2    = l or  %t3b, %t4b

	%v3    = l and %b2, %sixty

	%po0   = l add %out, %o
	%ch0   = l call $map6(%v0)
	storeb %ch0, %po0
	%o1    = l add %o, %one

	%po1   = l add %out, %o1
	%ch1   = l call $map6(%v1)
	storeb %ch1, %po1
	%o2    = l add %o1, %one

	%po2   = l add %out, %o2
	%ch2   = l call $map6(%v2)
	storeb %ch2, %po2
	%o3    = l add %o2, %one

	%po3   = l add %out, %o3
	%ch3   = l call $map6(%v3)
	storeb %ch3, %po3
	%o4    = l add %o3, %one

	%i     = l add %i, %three
	%o     = l add %o4, 0
	jmp    @loop

@remainder
	%rem   = l sub %n, %i
	%c1    = w ceql %rem, %one
	jnz    %c1, @rem1, @check2

@rem1
	%p0r   = l add %src, %i
	%b0r   = l loadub %p0r

	%v0rs  = l shr %b0r, %two
	%v0r   = l and %v0rs, %sixty

	%t1ar  = l and %b0r, %three
	%v1r   = l shl %t1ar, %four

	%po0r  = l add %out, %o
	%ch0r  = l call $map6(%v0r)
	storeb %ch0r, %po0r
	%o1r   = l add %o, %one

	%po1r  = l add %out, %o1r
	%ch1r  = l call $map6(%v1r)
	storeb %ch1r, %po1r
	%o2r   = l add %o1r, %one

	%peq   = l add 0, 61
	%po2r  = l add %out, %o2r
	storeb %peq, %po2r
	%o3r   = l add %o2r, %one

	%po3r  = l add %out, %o3r
	storeb %peq, %po3r
	%o4r   = l add %o3r, %one

	%o     = l add %o4r, 0
	jmp    @done

@check2
	%c2    = w ceql %rem, %two
	jnz    %c2, @rem2, @done

@rem2
	%p0r2  = l add %src, %i
	%b0r2  = l loadub %p0r2

	%i1r2  = l add %i, %one
	%p1r2  = l add %src, %i1r2
	%b1r2  = l loadub %p1r2

	%v0r2s = l shr %b0r2, %two
	%v0r2  = l and %v0r2s, %sixty

	%t1ar2 = l and %b0r2, %three
	%t1br2 = l shl %t1ar2, %four
	%t2ar2 = l shr %b1r2, %four
	%t2br2 = l and %t2ar2, 15
	%v1r2  = l or  %t1br2, %t2br2

	%t3ar2 = l and %b1r2, 15
	%v2r2  = l shl %t3ar2, %two

	%po0r2 = l add %out, %o
	%ch0r2 = l call $map6(%v0r2)
	storeb %ch0r2, %po0r2
	%o1r2  = l add %o, %one

	%po1r2 = l add %out, %o1r2
	%ch1r2 = l call $map6(%v1r2)
	storeb %ch1r2, %po1r2
	%o2r2  = l add %o1r2, %one

	%po2r2 = l add %out, %o2r2
	%ch2r2 = l call $map6(%v2r2)
	storeb %ch2r2, %po2r2
	%o3r2  = l add %o2r2, %one

	%po3r2 = l add %out, %o3r2
	storeb %peq, %po3r2
	%o4r2  = l add %o3r2, %one

	%o     = l add %o4r2, 0

@done
	ret    %rc
}

function l $map6(l %v) {
@entry
	%c26   = w csltl %v, 26
	jnz    %c26, @upper, @check_lower

@upper
	%resu  = l add 65, %v
	ret    %resu

@check_lower
	%c52   = w csltl %v, 52
	jnz    %c52, @lower, @check_digit

@lower
	%vl    = l sub %v, 26
	%resl  = l add 97, %vl
	ret    %resl

@check_digit
	%c62   = w csltl %v, 62
	jnz    %c62, @digit, @check_plus

@digit
	%vd    = l sub %v, 52
	%resd  = l add 48, %vd
	ret    %resd

@check_plus
	%cplus = w ceql %v, 62
	jnz    %cplus, @plus, @slash

@plus
	%resp  = l add 0, 43
	ret    %resp

@slash
	%ress  = l add 0, 47
	ret    %ress
}
