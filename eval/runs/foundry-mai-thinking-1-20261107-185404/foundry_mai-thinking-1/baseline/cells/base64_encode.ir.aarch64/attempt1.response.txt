export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%zero =l add 0, 0
	%one  =l add 0, 1
	%two  =l add 0, 2
	%three= l add 0, 3
	%four = l add 0, 4
	%rem1 = l add 0, 1
	%rem2 = l add 0, 2

	%retcount =l mul %four, %n
	%retcount =l add %retcount, %two
	%retcount =l div %retcount, %three
	%retcount =l mul %retcount, %four

	%i =l add 0, 0
	%o =l add 0, 0

@loop
	%lim =l sub %n, %three
	%c  =w csltl %i, %lim
	jnz %c, @loop_body, @remainder

@loop_body
	%p0 =l add %src, %i
	%b0 =l loadub %p0

	%i1 =l add %i, %one
	%p1 =l add %src, %i1
	%b1 =l loadub %p1

	%i2 =l add %i, %two
	%p2 =l add %src, %i2
	%b2 =l loadub %p2

	%v0 =l shrl %b0, %two
	%v0 =l and  %v0, 63

	%t1 =l and  %b0, 3
	%t1 =l shll %t1, %four
	%t2 =l shrl %b1, %four
	%t2 =l and  %t2, 15
	%v1 =l or   %t1, %t2

	%t3 =l and  %b1, 15
	%t3 =l shll %t3, %two
	%t4 =l shrl %b2, %six
	%six= l add 0, 6
	%t4 =l and  %t4, 3
	%v2 =l or   %t3, %t4

	%v3 =l and  %b2, 63

	%ch0 =l call $map_base64(%v0)
	%po0 =l add %out, %o
	storeb %ch0, %po0
	%o1  =l add %o, %one

	%ch1 =l call $map_base64(%v1)
	%po1 =l add %out, %o1
	storeb %ch1, %po1
	%o2  =l add %o1, %one

	%ch2 =l call $map_base64(%v2)
	%po2 =l add %out, %o2
	storeb %ch2, %po2
	%o3  =l add %o2, %one

	%ch3 =l call $map_base64(%v3)
	%po3 =l add %out, %o3
	storeb %ch3, %po3
	%o4  =l add %o3, %one

	%i   =l add %i, %three
	%o   =l add %o4, 0
	jmp @loop

@remainder
	%rem =l sub %n, %i
	%c1  =w ceql %rem, %one
	jnz  %c1, @rem1, @check2

@rem1
	%p0r =l add %src, %i
	%b0r =l loadub %p0r

	%v0r =l shrl %b0r, %two
	%v0r =l and  %v0r, 63

	%t1r =l and  %b0r, 3
	%t1r =l shll %t1r, %four
	%v1r =l add  %t1r, 0

	%ch0r =l call $map_base64(%v0r)
	%po0r =l add %out, %o
	storeb %ch0r, %po0r
	%o1r  =l add %o, %one

	%ch1r =l call $map_base64(%v1r)
	%po1r =l add %out, %o1r
	storeb %ch1r, %po1r
	%o2r  =l add %o1r, %one

	%peq  =l add 0, 61
	%po2r =l add %out, %o2r
	storeb %peq, %po2r
	%o3r  =l add %o2r, %one

	%po3r =l add %out, %o3r
	storeb %peq, %po3r
	%o4r  =l add %o3r, %one

	%o    =l add %o4r, 0
	jmp @done

@check2
	%c2  =w ceql %rem, %two
	jnz  %c2, @rem2, @done

@rem2
	%p0r2 =l add %src, %i
	%b0r2 =l loadub %p0r2

	%i1r2 =l add %i, %one
	%p1r2 =l add %src, %i1r2
	%b1r2 =l loadub %p1r2

	%v0r2 =l shrl %b0r2, %two
	%v0r2 =l and  %v0r2, 63

	%t1r2 =l and  %b0r2, 3
	%t1r2 =l shll %t1r2, %four
	%t2r2 =l shrl %b1r2, %four
	%t2r2 =l and  %t2r2, 15
	%v1r2 =l or   %t1r2, %t2r2

	%t3r2 =l and  %b1r2, 15
	%t3r2 =l shll %t3r2, %two
	%v2r2 =l add  %t3r2, 0

	%ch0r2 =l call $map_base64(%v0r2)
	%po0r2 =l add %out, %o
	storeb %ch0r2, %po0r2
	%o1r2  =l add %o, %one

	%ch1r2 =l call $map_base64(%v1r2)
	%po1r2 =l add %out, %o1r2
	storeb %ch1r2, %po1r2
	%o2r2  =l add %o1r2, %one

	%ch2r2 =l call $map_base64(%v2r2)
	%po2r2 =l add %out, %o2r2
	storeb %ch2r2, %po2r2
	%o3r2  =l add %o2r2, %one

	%peq2  =l add 0, 61
	%po3r2 =l add %out, %o3r2
	storeb %peq2, %po3r2
	%o4r2  =l add %o3r2, %one

	%o     =l add %o4r2, 0

@done
	ret %retcount
}

function l $map_base64(l %v) {
@map_entry
	%a_cap =l add 0, 65
	%a_low =l add 0, 97
	%zero  =l add 0, 48
	%plus  =l add 0, 43
	%slash =l add 0, 47

	%c1 =w csltl %v, 26
	jnz %c1, @case_upper, @check_lower

@case_upper
	%r1 =l add %a_cap, %v
	ret %r1

@check_lower
	%c2 =w csltl %v, 52
	jnz %c2, @case_lower, @check_digit

@case_lower
	%v2m =l sub %v, 26
	%r2  =l add %a_low, %v2m
	ret %r2

@check_digit
	%c3 =w csltl %v, 62
	jnz %c3, @case_digit, @check_plus

@case_digit
	%v3m =l sub %v, 52
	%r3  =l add %zero, %v3m
	ret %r3

@check_plus
	%c4 =w ceql %v, 62
	jnz %c4, @case_plus, @case_slash

@case_plus
	ret %plus

@case_slash
	ret %slash
}
