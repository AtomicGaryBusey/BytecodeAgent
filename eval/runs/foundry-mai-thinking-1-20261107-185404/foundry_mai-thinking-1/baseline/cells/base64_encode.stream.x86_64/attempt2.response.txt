base64_encode:
  xor rax, rax
  xor r8, r8
  jmp .Lcheck_full

.Lfull_loop:
  movzx r9, byte ptr [rdi + r8]
  movzx r10, byte ptr [rdi + r8 + 1]
  movzx r11, byte ptr [rdi + r8 + 2]
  shl r9, 16
  shl r10, 8
  or r9, r10
  or r9, r11

  mov rcx, r9
  shr rcx, 18
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  mov rcx, r9
  shr rcx, 12
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  mov rcx, r9
  shr rcx, 6
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  mov rcx, r9
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  add r8, 3

.Lcheck_full:
  mov r10, rsi
  sub r10, r8
  cmp r10, 3
  jge .Lfull_loop

  cmp r10, 0
  je .Ldone

  xor r9, r9
  movzx r11, byte ptr [rdi + r8]
  shl r11, 16
  or r9, r11

  cmp r10, 2
  jl .Lone_byte

  movzx r11, byte ptr [rdi + r8 + 1]
  shl r11, 8
  or r9, r11

.Lone_byte:
  mov rcx, r9
  shr rcx, 18
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  mov rcx, r9
  shr rcx, 12
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  cmp r10, 2
  je .Ltwo_bytes_final

  mov byte ptr [rdx + rax], 61
  inc rax
  mov byte ptr [rdx + rax], 61
  inc rax
  jmp .Ldone

.Ltwo_bytes_final:
  mov rcx, r9
  shr rcx, 6
  and rcx, 63
  call .Lencode_char
  mov byte ptr [rdx + rax], cl
  inc rax

  mov byte ptr [rdx + rax], 61
  inc rax

.Ldone:
  ret

.Lencode_char:
  cmp rcx, 25
  ja .Lge26
  add rcx, 65
  ret

.Lge26:
  cmp rcx, 51
  ja .Lge52
  sub rcx, 26
  add rcx, 97
  ret

.Lge52:
  cmp rcx, 61
  ja .Lge62
  sub rcx, 52
  add rcx, 48
  ret

.Lge62:
  cmp rcx, 62
  ja .Lslash
  mov rcx, 43
  ret

.Lslash:
  mov rcx, 47
  ret
