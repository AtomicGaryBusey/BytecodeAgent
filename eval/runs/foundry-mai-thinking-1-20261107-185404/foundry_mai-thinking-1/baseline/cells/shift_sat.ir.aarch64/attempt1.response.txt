export function l $shift_sat(l %x, l %n) {
@start
	%nge = csgel %n, 64
	jnz %nge, @large, @small

@small
	%L = shll %x, %n
	%R = shrl %x, %n
	%res = addl %L, %R
	ret %res

@large
	%xneg = csltl %x, 0
	jnz %xneg, @neg, @pos

@neg
	%R1 = subl 0, 1
	jmp @done

@pos
	%R1 = mov 0
	jmp @done

@done
	%L1 = mov 0
	%res1 = addl %L1, %R1
	ret %res1
}
