export function l $shift_sat(l %x, l %n) {
@start
	%smallcond = csltl %n, $64
	jnz %smallcond, @small, @large

@small
	%L = shll %x, %n
	%R = sarl %x, %n
	%res = addl %L, %R
	ret %res

@large
	%xneg = csltl %x, $0
	jnz %xneg, @neg, @pos

@neg
	%R1 = mov $-1
	jmp @done

@pos
	%R1 = mov $0
	jmp @done

@done
	%L1 = mov $0
	%res1 = addl %L1, %R1
	ret %res1
}
