export function l $count_desc(l %a, l %n) {
@entry
  %two =l 2
  %geq2 = csgel %n, %two
  jnz %geq2, @init, @zero

@init
  %count =l 0
  %i =l 0
  %limit = sub %n, %two
  jmp @cond

@cond
  %isle = cslel %i, %limit
  jnz %isle, @body, @done

@body
  %off1 = mul %i, 8
  %p1 = add %a, %off1
  %v1 = loadl %p1
  %i1p = add %i, 1
  %off2 = mul %i1p, 8
  %p2 = add %a, %off2
  %v2 = loadl %p2
  %desc = csgtl %v1, %v2
  jnz %desc, @incr, @next

@incr
  %count = add %count, 1
  jmp @next

@next
  %i = add %i, 1
  jmp @cond

@done
  ret %count

@zero
  ret 0
}
