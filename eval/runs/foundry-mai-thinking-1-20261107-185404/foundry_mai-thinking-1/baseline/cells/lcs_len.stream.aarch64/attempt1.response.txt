lcs_len:
	// Initialize row[0..n] to 0
	mov	x5, #0
	cbz	x3, .Linit_done
.Linit_loop:
	str	xzr, [x4, x5, lsl #3]
	add	x5, x5, #1
	cmp	x5, x3
	ble	.Linit_loop
	str	xzr, [x4, x3, lsl #3]
.Linit_done:
	// Outer loop: i from 0 to m-1
	mov	x5, #0
	cmp	x5, x1
	bge	.Lend
.Louter_loop:
	// diag = 0
	mov	x7, #0
	// load x[i]
	ldrb	w12, [x0, x5]
	// Inner loop: j from 1 to n
	mov	x6, #1
	cmp	x6, x3
	bgt	.Lnext_i
.Linner_loop:
	// up = row[j]
	ldr	x10, [x4, x6, lsl #3]
	// load y[j-1]
	sub	x9, x6, #1
	ldrb	w13, [x2, x9]
	// compare x[i] == y[j-1]
	cmp	w12, w13
	bne	.Lnot_equal
	// equal: row[j] = diag + 1
	add	x11, x7, #1
	b	.Lstore
.Lnot_equal:
	// get row[j-1]
	sub	x9, x6, #1
	ldr	x11, [x4, x9, lsl #3]
	// compare row[j] (current, still in x10? up is x10 = row[j]) vs row[j-1]
	cmp	x10, x11
	csel	x11, x10, x11, ge
.Lstore:
	// store row[j] = x11, diag = up (x10)
	str	x11, [x4, x6, lsl #3]
	mov	x7, x10
	// j++
	add	x6, x6, #1
	cmp	x6, x3
	ble	.Linner_loop
.Lnext_i:
	// i++
	add	x5, x5, #1
	cmp	x5, x1
	blt	.Louter_loop
.Lend:
	// return row[n]
	ldr	x0, [x4, x3, lsl #3]
	ret
