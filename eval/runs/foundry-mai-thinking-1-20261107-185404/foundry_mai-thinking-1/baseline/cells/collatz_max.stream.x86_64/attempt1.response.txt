collatz_max:
  push rbx
  cmp rdi, 1
  jl .Lret0
  mov rax, 0
  mov rbx, 1
.Lk_loop:
  cmp rbx, rdi
  jg .Ldone
  mov rcx, rbx
  mov rdx, 0
.Lc_loop:
  cmp rcx, 1
  je .Lnext_k
  test rcx, 1
  jz .Leven
  lea rcx, [rcx + rcx*2 + 1]
  inc rdx
  jmp .Lc_loop
.Leven:
  shr rcx, 1
  inc rdx
  jmp .Lc_loop
.Lnext_k:
  cmp rdx, rax
  cmovg rax, rdx
  inc rbx
  jmp .Lk_loop
.Lret0:
  xor rax, rax
.Ldone:
  pop rbx
  ret
