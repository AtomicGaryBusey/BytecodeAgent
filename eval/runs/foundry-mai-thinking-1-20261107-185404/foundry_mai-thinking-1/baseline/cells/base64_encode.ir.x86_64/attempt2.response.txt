export function l $base64_encode(l %src, l %n, l %out) {
@start
	%zero  =l copy 0
	%one   =l copy 1
	%two   =l copy 2
	%three =l copy 3
	%four  =l copy 4
	%six3  =l copy 63
	%eqch  =l copy 61

	%fullg =l div %n, %three
	%remg  =l rem %n, %three

	%outcnt =l mul %fullg, %four
	%add4b  =l copy 0
	%isr1   =w ceql %remg, %one
	jnz %isr1, @set4_r1, @check_r2
@set4_r1
	%add4b  =l copy %four
	jmp @count_done
@check_r2
	%isr2   =w ceql %remg, %two
	jnz %isr2, @set4_r2, @count_done
@set4_r2
	%add4b  =l copy %four
@count_done
	%outcntf =l add %outcnt, %add4b

	%si =l copy 0
	%oi =l copy 0

@loop_full
	%limf =l mul %fullg, %three
	%cf   =w csltl %si, %limf
	jnz %cf, @do_full, @after_full

@do_full
	%p0  =l add %src, %si
	%b0  =l loadub %p0

	%s1  =l add %si, %one
	%p1  =l add %src, %s1
	%b1  =l loadub %p1

	%s2  =l add %si, %two
	%p2  =l add %src, %s2
	%b2  =l loadub %p2

	%v24 =l shl %b0, 16
	%b18 =l shl %b1, 8
	%v24 =l or  %v24, %b18
	%v24 =l or  %v24, %b2

	%c0v =l shr %v24, 18
	%c0v =l and %c0v, %six3

	%c1v =l shr %v24, 12
	%c1v =l and %c1v, %six3

	%c2v =l shr %v24, 6
	%c2v =l and %c2v, %six3

	%c3v =l and %v24, %six3

	%ch0 =l copy 0
	%lt0 =w csltl %c0v, 26
	jnz %lt0, @c0_upper, @c0_mid
@c0_upper
	%ch0 =l add %c0v, 65
	jmp @c0_done
@c0_mid
	%lt0b =w csltl %c0v, 52
	jnz %lt0b, @c0_lower, @c0_mid2
@c0_lower
	%ch0 =l add %c0v, 71
	jmp @c0_done
@c0_mid2
	%lt0c =w csltl %c0v, 62
	jnz %lt0c, @c0_digit, @c0_sym
@c0_digit
	%ch0 =l add %c0v, 252
	jmp @c0_done
@c0_sym
	%eq0  =w ceql %c0v, 62
	jnz %eq0, @c0_plus, @c0_slash
@c0_plus
	%ch0  =l copy 43
	jmp @c0_done
@c0_slash
	%ch0  =l copy 47
@c0_done

	%ch1  =l copy 0
	%lt1  =w csltl %c1v, 26
	jnz %lt1, @c1_upper, @c1_mid
@c1_upper
	%ch1  =l add %c1v, 65
	jmp @c1_done
@c1_mid
	%lt1b =w csltl %c1v, 52
	jnz %lt1b, @c1_lower, @c1_mid2
@c1_lower
	%ch1  =l add %c1v, 71
	jmp @c1_done
@c1_mid2
	%lt1c =w csltl %c1v, 62
	jnz %lt1c, @c1_digit, @c1_sym
@c1_digit
	%ch1  =l add %c1v, 252
	jmp @c1_done
@c1_sym
	%eq1  =w ceql %c1v, 62
	jnz %eq1, @c1_plus, @c1_slash
@c1_plus
	%ch1  =l copy 43
	jmp @c1_done
@c1_slash
	%ch1  =l copy 47
@c1_done

	%ch2  =l copy 0
	%lt2  =w csltl %c2v, 26
	jnz %lt2, @c2_upper, @c2_mid
@c2_upper
	%ch2  =l add %c2v, 65
	jmp @c2_done
@c2_mid
	%lt2b =w csltl %c2v, 52
	jnz %lt2b, @c2_lower, @c2_mid2
@c2_lower
	%ch2  =l add %c2v, 71
	jmp @c2_done
@c2_mid2
	%lt2c =w csltl %c2v, 62
	jnz %lt2c, @c2_digit, @c2_sym
@c2_digit
	%ch2  =l add %c2v, 252
	jmp @c2_done
@c2_sym
	%eq2  =w ceql %c2v, 62
	jnz %eq2, @c2_plus, @c2_slash
@c2_plus
	%ch2  =l copy 43
	jmp @c2_done
@c2_slash
	%ch2  =l copy 47
@c2_done

	%ch3  =l copy 0
	%lt3  =w csltl %c3v, 26
	jnz %lt3, @c3_upper, @c3_mid
@c3_upper
	%ch3  =l add %c3v, 65
	jmp @c3_done
@c3_mid
	%lt3b =w csltl %c3v, 52
	jnz %lt3b, @c3_lower, @c3_mid2
@c3_lower
	%ch3  =l add %c3v, 71
	jmp @c3_done
@c3_mid2
	%lt3c =w csltl %c3v, 62
	jnz %lt3c, @c3_digit, @c3_sym
@c3_digit
	%ch3  =l add %c3v, 252
	jmp @c3_done
@c3_sym
	%eq3  =w ceql %c3v, 62
	jnz %eq3, @c3_plus, @c3_slash
@c3_plus
	%ch3  =l copy 43
	jmp @c3_done
@c3_slash
	%ch3  =l copy 47
@c3_done

	%po0  =l add %out, %oi
	storeb %ch0, %po0
	%o1   =l add %oi, %one
	%po1  =l add %out, %o1
	storeb %ch1, %po1
	%o2   =l add %oi, %two
	%po2  =l add %out, %o2
	storeb %ch2, %po2
	%o3   =l add %oi, %three
	%po3  =l add %out, %o3
	storeb %ch3, %po3

	%si   =l add %si, %three
	%oi   =l add %oi, %four
	jmp @loop_full

@after_full
	jnz %isr1, @rem1, @chk_rem2

@rem1
	%p0r  =l add %src, %si
	%b0r  =l loadub %p0r
	%v24r =l shl %b0r, 16

	%c0r  =l shr %v24r, 18
	%c0r  =l and %c0r, %six3
	%c1r  =l shr %v24r, 12
	%c1r  =l and %c1r, %six3

	%ch0r =l copy 0
	%lr0  =w csltl %c0r, 26
	jnz %lr0, @r0_upper, @r0_mid
@r0_upper
	%ch0r =l add %c0r, 65
	jmp @r0_done
@r0_mid
	%lr0b =w csltl %c0r, 52
	jnz %lr0b, @r0_lower, @r0_mid2
@r0_lower
	%ch0r =l add %c0r, 71
	jmp @r0_done
@r0_mid2
	%lr0c =w csltl %c0r, 62
	jnz %lr0c, @r0_digit, @r0_sym
@r0_digit
	%ch0r =l add %c0r, 252
	jmp @r0_done
@r0_sym
	%er0  =w ceql %c0r, 62
	jnz %er0, @r0_plus, @r0_slash
@r0_plus
	%ch0r =l copy 43
	jmp @r0_done
@r0_slash
	%ch0r =l copy 47
@r0_done

	%ch1r =l copy 0
	%lr1  =w csltl %c1r, 26
	jnz %lr1, @r1_upper, @r1_mid
@r1_upper
	%ch1r =l add %c1r, 65
	jmp @r1_done
@r1_mid
	%lr1b =w csltl %c1r, 52
	jnz %lr1b, @r1_lower, @r1_mid2
@r1_lower
	%ch1r =l add %c1r, 71
	jmp @r1_done
@r1_mid2
	%lr1c =w csltl %c1r, 62
	jnz %lr1c, @r1_digit, @r1_sym
@r1_digit
	%ch1r =l add %c1r, 252
	jmp @r1_done
@r1_sym
	%er1  =w ceql %c1r, 62
	jnz %er1, @r1_plus, @r1_slash
@r1_plus
	%ch1r =l copy 43
	jmp @r1_done
@r1_slash
	%ch1r =l copy 47
@r1_done

	%po0r =l add %out, %oi
	storeb %ch0r, %po0r
	%o1r  =l add %oi, %one
	%po1r =l add %out, %o1r
	storeb %ch1r, %po1r
	%o2r  =l add %oi, %two
	%po2r =l add %out, %o2r
	storeb %eqch, %po2r
	%o3r  =l add %oi, %three
	%po3r =l add %out, %o3r
	storeb %eqch, %po3r
	jmp @done

@chk_rem2
	jnz %isr2, @rem2, @done

@rem2
	%p0r2 =l add %src, %si
	%b0r2 =l loadub %p0r2
	%s1r2 =l add %si, %one
	%p1r2 =l add %src, %s1r2
	%b1r2 =l loadub %p1r2

	%v24r2 =l shl %b0r2, 16
	%b18r2 =l shl %b1r2, 8
	%v24r2 =l or  %v24r2, %b18r2

	%c0r2  =l shr %v24r2, 18
	%c0r2  =l and %c0r2, %six3
	%c1r2  =l shr %v24r2, 12
	%c1r2  =l and %c1r2, %six3
	%c2r2  =l shr %v24r2, 6
	%c2r2  =l and %c2r2, %six3

	%ch0r2 =l copy 0
	%lr20  =w csltl %c0r2, 26
	jnz %lr20, @r20_upper, @r20_mid
@r20_upper
	%ch0r2 =l add %c0r2, 65
	jmp @r20_done
@r20_mid
	%lr20b =w csltl %c0r2, 52
	jnz %lr20b, @r20_lower, @r20_mid2
@r20_lower
	%ch0r2 =l add %c0r2, 71
	jmp @r20_done
@r20_mid2
	%lr20c =w csltl %c0r2, 62
	jnz %lr20c, @r20_digit, @r20_sym
@r20_digit
	%ch0r2 =l add %c0r2, 252
	jmp @r20_done
@r20_sym
	%er20  =w ceql %c0r2, 62
	jnz %er20, @r20_plus, @r20_slash
@r20_plus
	%ch0r2 =l copy 43
	jmp @r20_done
@r20_slash
	%ch0r2 =l copy 47
@r20_done

	%ch1r2 =l copy 0
	%lr21  =w csltl %c1r2, 26
	jnz %lr21, @r21_upper, @r21_mid
@r21_upper
	%ch1r2 =l add %c1r2, 65
	jmp @r21_done
@r21_mid
	%lr21b =w csltl %c1r2, 52
	jnz %lr21b, @r21_lower, @r21_mid2
@r21_lower
	%ch1r2 =l add %c1r2, 71
	jmp @r21_done
@r21_mid2
	%lr21c =w csltl %c1r2, 62
	jnz %lr21c, @r21_digit, @r21_sym
@r21_digit
	%ch1r2 =l add %c1r2, 252
	jmp @r21_done
@r21_sym
	%er21  =w ceql %c1r2, 62
	jnz %er21, @r21_plus, @r21_slash
@r21_plus
	%ch1r2 =l copy 43
	jmp @r21_done
@r21_slash
	%ch1r2 =l copy 47
@r21_done

	%ch2r2 =l copy 0
	%lr22  =w csltl %c2r2, 26
	jnz %lr22, @r22_upper, @r22_mid
@r22_upper
	%ch2r2 =l add %c2r2, 65
	jmp @r22_done
@r22_mid
	%lr22b =w csltl %c2r2, 52
	jnz %lr22b, @r22_lower, @r22_mid2
@r22_lower
	%ch2r2 =l add %c2r2, 71
	jmp @r22_done
@r22_mid2
	%lr22c =w csltl %c2r2, 62
	jnz %lr22c, @r22_digit, @r22_sym
@r22_digit
	%ch2r2 =l add %c2r2, 252
	jmp @r22_done
@r22_sym
	%er22  =w ceql %c2r2, 62
	jnz %er22, @r22_plus, @r22_slash
@r22_plus
	%ch2r2 =l copy 43
	jmp @r22_done
@r22_slash
	%ch2r2 =l copy 47
@r22_done

	%po0r2 =l add %out, %oi
	storeb %ch0r2, %po0r2
	%o1r2  =l add %oi, %one
	%po1r2 =l add %out, %o1r2
	storeb %ch1r2, %po1r2
	%o2r2  =l add %oi, %two
	%po2r2 =l add %out, %o2r2
	storeb %ch2r2, %po2r2
	%o3r2  =l add %oi, %three
	%po3r2 =l add %out, %o3r2
	storeb %eqch, %po3r2
	jmp @done

@done
	ret %outcntf
}
