export function l $base64_encode(l %src, l %n, l %out) {
@start
	%zero =l copy 0
	%one  =l copy 1
	%two  =l copy 2
	%three= l copy 3
	%four = l copy 4

	%remfull = l copy %n
	%fullg   = l div %remfull, %three
	%remg    = l rem %remfull, %three

	%outcnt  = l mul %fullg, %four
	%isr1    =w ceql %remg, %one
	%isr2    =w ceql %remg, %two
	%add4    =l csel %isr1, %four, %zero
	%add4b   =l csel %isr2, %four, %add4
	%outcntf =l add  %outcnt, %add4b

	%si =l copy 0
	%oi =l copy 0

@loop_full
	%limf =l mul %fullg, %three
	%cf   =w csltl %si, %limf
	jnz %cf, @do_full, @after_full

@do_full
	%p0  =l add %src, %si
	%b0  =l loadub %p0

	%s1  =l add %si, %one
	%p1  =l add %src, %s1
	%b1  =l loadub %p1

	%s2  =l add %si, %two
	%p2  =l add %src, %s2
	%b2  =l loadub %p2

	%v24 =l shl %b0, 16
	%b18 =l shl %b1, 8
	%v24 =l or  %v24, %b18
	%v24 =l or  %v24, %b2

	%c0v =l shr %v24, 18
	%c0v =l and %c0v, 63

	%c1v =l shr %v24, 12
	%c1v =l and %c1v, 63

	%c2v =l shr %v24, 6
	%c2v =l and %c2v, 63

	%c3v =l and %v24, 63

	%ch0 =l call $b64_char(l %c0v)
	%ch1 =l call $b64_char(l %c1v)
	%ch2 =l call $b64_char(l %c2v)
	%ch3 =l call $b64_char(l %c3v)

	%po0 =l add %out, %oi
	storeb %ch0, %po0
	%o1  =l add %oi, %one
	%po1 =l add %out, %o1
	storeb %ch1, %po1
	%o2  =l add %oi, %two
	%po2 =l add %out, %o2
	storeb %ch2, %po2
	%o3  =l add %oi, %three
	%po3 =l add %out, %o3
	storeb %ch3, %po3

	%si  =l add %si, %three
	%oi  =l add %oi, %four
	jmp @loop_full

@after_full
	jnz %isr1, @rem1, @chk_rem2

@rem1
	%p0r =l add %src, %si
	%b0r =l loadub %p0r
	%v24r=l shl %b0r, 16

	%c0r =l shr %v24r, 18
	%c0r =l and %c0r, 63
	%c1r =l shr %v24r, 12
	%c1r =l and %c1r, 63

	%ch0r=l call $b64_char(l %c0r)
	%ch1r=l call $b64_char(l %c1r)

	%po0r=l add %out, %oi
	storeb %ch0r, %po0r
	%o1r =l add %oi, %one
	%po1r=l add %out, %o1r
	storeb %ch1r, %po1r

	%eqc =l copy 61
	%o2r =l add %oi, %two
	%po2r=l add %out, %o2r
	storeb %eqc, %po2r
	%o3r =l add %oi, %three
	%po3r=l add %out, %o3r
	storeb %eqc, %po3r
	jmp @done

@chk_rem2
	jnz %isr2, @rem2, @done

@rem2
	%p0r2=l add %src, %si
	%b0r2=l loadub %p0r2
	%s1r2=l add %si, %one
	%p1r2=l add %src, %s1r2
	%b1r2=l loadub %p1r2
	%v24r2=l shl %b0r2, 16
	%b18r2=l shl %b1r2, 8
	%v24r2=l or %v24r2, %b18r2

	%c0r2=l shr %v24r2, 18
	%c0r2=l and %c0r2, 63
	%c1r2=l shr %v24r2, 12
	%c1r2=l and %c1r2, 63
	%c2r2=l shr %v24r2, 6
	%c2r2=l and %c2r2, 63

	%ch0r2=l call $b64_char(l %c0r2)
	%ch1r2=l call $b64_char(l %c1r2)
	%ch2r2=l call $b64_char(l %c2r2)

	%po0r2=l add %out, %oi
	storeb %ch0r2, %po0r2
	%o1r2 =l add %oi, %one
	%po1r2=l add %out, %o1r2
	storeb %ch1r2, %po1r2
	%o2r2 =l add %oi, %two
	%po2r2=l add %out, %o2r2
	storeb %ch2r2, %po2r2

	%eqc2 =l copy 61
	%o3r2 =l add %oi, %three
	%po3r2=l add %out, %o3r2
	storeb %eqc2, %po3r2
	jmp @done

@done
	ret %outcntf
}

function l $b64_char(l %v) {
@cstart
	%lt26 =w csltl %v, 26
	jnz %lt26, @upper, @ca

@upper
	%r =l add %v, 65
	ret %r

@ca
	%v2 =l copy %v
	%lt52=w csltl %v2, 52
	jnz %lt52, @lower, @cz

@lower
	%r2 =l add %v2, 71
	ret %r2

@cz
	%lt62=w csltl %v2, 62
	jnz %lt62, @digit, @csym

@digit
	%r3 =l add %v2, 252
	ret %r3

@csym
	%eq62=w ceql %v2, 62
	jnz %eq62, @plus, @slash

@plus
	%rp =l copy 43
	ret %rp

@slash
	%rs =l copy 47
	ret %rs
}
