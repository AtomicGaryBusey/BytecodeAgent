export function l $reverse(l %a, l %n) {
@entry
	%left = mov 0l
	%right = subl %n, 1l
	jmp @loop

@loop
	%cond = csltl %left, %right
	jnz %cond, @swap, @done

@swap
	%off1 = shll %left, 3l
	%off2 = shll %right, 3l
	%p1 = addl %a, %off1
	%p2 = addl %a, %off2
	%val1 = loadl %p1
	%val2 = loadl %p2
	storel %val2, %p1
	storel %val1, %p2
	%left = addl %left, 1l
	%right = subl %right, 1l
	jmp @loop

@done
	ret 0l
}
