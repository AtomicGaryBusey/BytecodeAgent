export function l $reverse(l %a, l %n) {
@entry
	%left = 0l
	%right = sub l %n, 1l
	jmp @loop

@loop
	%cond = cslt l %left, %right
	jnz %cond, @swap, @done

@swap
	%off1 = shl l %left, 3l
	%off2 = shl l %right, 3l
	%p1 = add l %a, %off1
	%p2 = add l %a, %off2
	%val1 = load l %p1
	%val2 = load l %p2
	store l %val2, %p1
	store l %val1, %p2
	%left = add l %left, 1l
	%right = sub l %right, 1l
	jmp @loop

@done
	ret 0l
}
