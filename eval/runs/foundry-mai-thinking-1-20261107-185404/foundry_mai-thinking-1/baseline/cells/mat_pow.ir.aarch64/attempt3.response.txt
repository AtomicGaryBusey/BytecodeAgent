export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%sz = mull %n, %n
	%kzero = ceql %k, 0
	jnz %kzero, @handle_kzero

@handle_kgt0
	%zero = addl 0, 0
	%one  = addl 0, 1
	%sz8  = mull %sz, 8
	%base = addl %tmp, 0
	%prod = addl %tmp, %sz8

@copy_base
	%i_cb = addl 0, 0
@cb_loop
	%cb_done = ceql %i_cb, %sz
	jnz %cb_done, @set_identity
	%off_cb  = mull %i_cb, 8
	%pa_cb   = addl %a, %off_cb
	%pb_cb   = addl %base, %off_cb
	%val_cb  = loadl %pa_cb
	storel %val_cb, %pb_cb
	%i_cb_n  = addl %i_cb, 1
	%i_cb    = addl %i_cb_n, 0
	jmp @cb_loop

@set_identity
	%i_si = addl 0, 0
@si_loop
	%si_done = ceql %i_si, %sz
	jnz %si_done, @pow_loop_init
	%off_si  = mull %i_si, 8
	%po_si   = addl %out, %off_si
	%row_si  = divl %i_si, %n
	%col_si  = reml %i_si, %n
	%diag    = ceql %row_si, %col_si
	jnz %diag, @si_set_one
	storel %zero, %po_si
	jmp @si_next
@si_set_one
	storel %one, %po_si
@si_next
	%i_si_n  = addl %i_si, 1
	%i_si    = addl %i_si_n, 0
	jmp @si_loop

@pow_loop_init
	%kk_cur = addl %k, 0

@pow_loop
	%kk_end = ceql %kk_cur, 0
	jnz %kk_end, @clear_tmp_all

@check_bit
	%bbit   = andl %kk_cur, 1
	%bzero  = ceql %bbit, 0
	jnz %bzero, @square_base

@multiply_r_by_b
	%i_mrb = addl 0, 0
@mr_i_loop
	%mr_i_end = ceql %i_mrb, %n
	jnz %mr_i_end, @copy_r_to_prod
	%j_mrb    = addl 0, 0
@mr_j_loop
	%mr_j_end = ceql %j_mrb, %n
	jnz %mr_j_end, @mr_i_next
	%sum_mrb  = addl 0, 0
	%t_mrb    = addl 0, 0
@mr_t_loop
	%mr_t_end = ceql %t_mrb, %n
	jnz %mr_t_end, @mr_store
	%idx_r1   = mull %i_mrb, %n
	%idx_r    = addl %idx_r1, %t_mrb
	%off_r    = mull %idx_r, 8
	%pr_r     = addl %out, %off_r
	%vr       = loadl %pr_r
	%idx_b1   = mull %t_mrb, %n
	%idx_b    = addl %idx_b1, %j_mrb
	%off_b    = mull %idx_b, 8
	%pb_b     = addl %base, %off_b
	%vb       = loadl %pb_b
	%term     = mull %vr, %vb
	%sum_new  = addl %sum_mrb, %term
	%sum_mrb  = addl %sum_new, 0
	%t_mrb_n  = addl %t_mrb, 1
	%t_mrb    = addl %t_mrb_n, 0
	jmp @mr_t_loop
@mr_store
	%idx_p1   = mull %i_mrb, %n
	%idx_p    = addl %idx_p1, %j_mrb
	%off_p    = mull %idx_p, 8
	%pp_p     = addl %prod, %off_p
	storel %sum_mrb, %pp_p
	%j_mrb_n  = addl %j_mrb, 1
	%j_mrb    = addl %j_mrb_n, 0
	jmp @mr_j_loop
@mr_i_next
	%i_mrb_n  = addl %i_mrb, 1
	%i_mrb    = addl %i_mrb_n, 0
	jmp @mr_i_loop

@copy_r_to_prod
	%i_crp = addl 0, 0
@crp_loop
	%crp_end = ceql %i_crp, %sz
	jnz %crp_end, @square_base
	%off_crp = mull %i_crp, 8
	%pp_crp  = addl %prod, %off_crp
	%po_crp  = addl %out, %off_crp
	%val_crp = loadl %pp_crp
	storel %val_crp, %po_crp
	%i_crp_n = addl %i_crp, 1
	%i_crp   = addl %i_crp_n, 0
	jmp @crp_loop

@square_base
	%i_mbb = addl 0, 0
@mbb_i_loop
	%mbb_i_end = ceql %i_mbb, %n
	jnz %mbb_i_end, @copy_b_to_prod
	%j_mbb     = addl 0, 0
@mbb_j_loop
	%mbb_j_end = ceql %j_mbb, %n
	jnz %mbb_j_end, @mbb_i_next
	%sum_mbb   = addl 0, 0
	%t_mbb     = addl 0, 0
@mbb_t_loop
	%mbb_t_end = ceql %t_mbb, %n
	jnz %mbb_t_end, @mbb_store
	%idx_b11   = mull %i_mbb, %n
	%idx_b1    = addl %idx_b11, %t_mbb
	%off_b1    = mull %idx_b1, 8
	%pb_b1     = addl %base, %off_b1
	%vb1       = loadl %pb_b1
	%idx_b21   = mull %t_mbb, %n
	%idx_b2    = addl %idx_b21, %j_mbb
	%off_b2    = mull %idx_b2, 8
	%pb_b2     = addl %base, %off_b2
	%vb2       = loadl %pb_b2
	%term2     = mull %vb1, %vb2
	%sum_n2    = addl %sum_mbb, %term2
	%sum_mbb   = addl %sum_n2, 0
	%t_mbb_n   = addl %t_mbb, 1
	%t_mbb     = addl %t_mbb_n, 0
	jmp @mbb_t_loop
@mbb_store
	%idx_pp1   = mull %i_mbb, %n
	%idx_pp    = addl %idx_pp1, %j_mbb
	%off_pp    = mull %idx_pp, 8
	%pp_pp     = addl %prod, %off_pp
	storel %sum_mbb, %pp_pp
	%j_mbb_n   = addl %j_mbb, 1
	%j_mbb     = addl %j_mbb_n, 0
	jmp @mbb_j_loop
@mbb_i_next
	%i_mbb_n   = addl %i_mbb, 1
	%i_mbb     = addl %i_mbb_n, 0
	jmp @mbb_i_loop

@copy_b_to_prod
	%i_cbp = addl 0, 0
@cbp_loop
	%cbp_end = ceql %i_cbp, %sz
	jnz %cbp_end, @shift_kk
	%off_cbp = mull %i_cbp, 8
	%pp_cbp  = addl %prod, %off_cbp
	%pb_cbp  = addl %base, %off_cbp
	%val_cbp = loadl %pp_cbp
	storel %val_cbp, %pb_cbp
	%i_cbp_n = addl %i_cbp, 1
	%i_cbp   = addl %i_cbp_n, 0
	jmp @cbp_loop

@shift_kk
	%kk_new  = shrl %kk_cur, 1
	%kk_cur  = addl %kk_new, 0
	jmp @pow_loop

@handle_kzero
	%zero_k  = addl 0, 0
	%one_k   = addl 0, 1
	%sz8_k   = mull %sz, 8
	%sz2_k   = mull %sz, 2
	%sz2_8_k = mull %sz2_k, 8
	%i_sk    = addl 0, 0
@sk_loop
	%sk_end  = ceql %i_sk, %sz
	jnz %sk_end, @clear_kzero_tmp
	%off_sk  = mull %i_sk, 8
	%po_sk   = addl %out, %off_sk
	%row_sk  = divl %i_sk, %n
	%col_sk  = reml %i_sk, %n
	%diag_sk = ceql %row_sk, %col_sk
	jnz %diag_sk, @sk_one
	storel %zero_k, %po_sk
	jmp @sk_next
@sk_one
	storel %one_k, %po_sk
@sk_next
	%i_sk_n  = addl %i_sk, 1
	%i_sk    = addl %i_sk_n, 0
	jmp @sk_loop

@clear_kzero_tmp
	%i_ck    = addl 0, 0
@ck_loop
	%ck_end  = ceql %i_ck, %sz2_k
	jnz %ck_end, @return_zero
	%off_ck  = mull %i_ck, 8
	%pt_ck   = addl %tmp, %off_ck
	storel %zero_k, %pt_ck
	%i_ck_n  = addl %i_ck, 1
	%i_ck    = addl %i_ck_n, 0
	jmp @ck_loop

@clear_tmp_all
	%zero_c  = addl 0, 0
	%sz2_c   = mull %sz, 2
	%i_ct    = addl 0, 0
@ct_loop
	%ct_end  = ceql %i_ct, %sz2_c
	jnz %ct_end, @return_zero
	%off_ct  = mull %i_ct, 8
	%pt_ct   = addl %tmp, %off_ct
	storel %zero_c, %pt_ct
	%i_ct_n  = addl %i_ct, 1
	%i_ct    = addl %i_ct_n, 0
	jmp @ct_loop

@return_zero
	%retv    = addl 0, 0
	ret %retv
}
