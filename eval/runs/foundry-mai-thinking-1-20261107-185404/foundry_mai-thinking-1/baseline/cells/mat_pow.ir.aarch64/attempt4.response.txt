export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%sz = mull %n, %n
	%k_is_zero = ceql %k, 0
	jnz %k_is_zero, @handle_kzero

@handle_kgt0
	%zero = addl 0, 0
	%one = addl 0, 1
	%sz8 = mull %sz, 8
	%sz2 = mull %sz, 2
	%base_ptr = addl %tmp, 0
	%prod_ptr = addl %tmp, %sz8

@copy_base
	%i_cb = addl 0, 0
@cb_loop
	%cb_done = ceql %i_cb, %sz
	jnz %cb_done, @set_identity
	%off_cb = mull %i_cb, 8
	%pa_cb = addl %a, %off_cb
	%pb_cb = addl %base_ptr, %off_cb
	%val_cb = loadl %pa_cb
	storel %val_cb, %pb_cb
	%i_cb_next = addl %i_cb, 1
	%i_cb = addl %i_cb_next, 0
	jmp @cb_loop

@set_identity
	%i_si = addl 0, 0
@si_loop
	%si_done = ceql %i_si, %sz
	jnz %si_done, @pow_init
	%off_si = mull %i_si, 8
	%po_si = addl %out, %off_si
	%row_si = divl %i_si, %n
	%col_si = reml %i_si, %n
	%is_diag = ceql %row_si, %col_si
	jnz %is_diag, @si_set_one
	storel %zero, %po_si
	jmp @si_next
@si_set_one
	storel %one, %po_si
@si_next
	%i_si_next = addl %i_si, 1
	%i_si = addl %i_si_next, 0
	jmp @si_loop

@pow_init
	%kk_cur = addl %k, 0

@pow_loop
	%kk_is_zero = ceql %kk_cur, 0
	jnz %kk_is_zero, @clear_tmp_all

@check_bit
	%bbit = andl %kk_cur, 1
	%bbit_zero = ceql %bbit, 0
	jnz %bbit_zero, @square_base

@mul_res_by_base
	%i_mrb = addl 0, 0
@mr_i_loop
	%mr_i_done = ceql %i_mrb, %n
	jnz %mr_i_done, @copy_prod_to_res
	%j_mrb = addl 0, 0
@mr_j_loop
	%mr_j_done = ceql %j_mrb, %n
	jnz %mr_j_done, @mr_i_next
	%sum_mrb = addl 0, 0
	%t_mrb = addl 0, 0
@mr_t_loop
	%mr_t_done = ceql %t_mrb, %n
	jnz %mr_t_done, @mr_store
	%idx_r1 = mull %i_mrb, %n
	%idx_r = addl %idx_r1, %t_mrb
	%off_r = mull %idx_r, 8
	%pr_r = addl %out, %off_r
	%vr = loadl %pr_r
	%idx_b1 = mull %t_mrb, %n
	%idx_b = addl %idx_b1, %j_mrb
	%off_b = mull %idx_b, 8
	%pb_b = addl %base_ptr, %off_b
	%vb = loadl %pb_b
	%term_mrb = mull %vr, %vb
	%sum_mrb_new = addl %sum_mrb, %term_mrb
	%sum_mrb = addl %sum_mrb_new, 0
	%t_mrb_next = addl %t_mrb, 1
	%t_mrb = addl %t_mrb_next, 0
	jmp @mr_t_loop
@mr_store
	%idx_p1 = mull %i_mrb, %n
	%idx_p = addl %idx_p1, %j_mrb
	%off_p = mull %idx_p, 8
	%pp_p = addl %prod_ptr, %off_p
	storel %sum_mrb, %pp_p
	%j_mrb_next = addl %j_mrb, 1
	%j_mrb = addl %j_mrb_next, 0
	jmp @mr_j_loop
@mr_i_next
	%i_mrb_next = addl %i_mrb, 1
	%i_mrb = addl %i_mrb_next, 0
	jmp @mr_i_loop

@copy_prod_to_res
	%i_cpr = addl 0, 0
@cpr_loop
	%cpr_done = ceql %i_cpr, %sz
	jnz %cpr_done, @square_base
	%off_cpr = mull %i_cpr, 8
	%pp_cpr = addl %prod_ptr, %off_cpr
	%po_cpr = addl %out, %off_cpr
	%val_cpr = loadl %pp_cpr
	storel %val_cpr, %po_cpr
	%i_cpr_next = addl %i_cpr, 1
	%i_cpr = addl %i_cpr_next, 0
	jmp @cpr_loop

@square_base
	%i_sqb = addl 0, 0
@sqb_i_loop
	%sqb_i_done = ceql %i_sqb, %n
	jnz %sqb_i_done, @copy_prod_to_base
	%j_sqb = addl 0, 0
@sqb_j_loop
	%sqb_j_done = ceql %j_sqb, %n
	jnz %sqb_j_done, @sqb_i_next
	%sum_sqb = addl 0, 0
	%t_sqb = addl 0, 0
@sqb_t_loop
	%sqb_t_done = ceql %t_sqb, %n
	jnz %sqb_t_done, @sqb_store
	%idx_b11 = mull %i_sqb, %n
	%idx_b1s = addl %idx_b11, %t_sqb
	%off_b1s = mull %idx_b1s, 8
	%pb_b1s = addl %base_ptr, %off_b1s
	%vb1s = loadl %pb_b1s
	%idx_b21 = mull %t_sqb, %n
	%idx_b2s = addl %idx_b21, %j_sqb
	%off_b2s = mull %idx_b2s, 8
	%pb_b2s = addl %base_ptr, %off_b2s
	%vb2s = loadl %pb_b2s
	%term_sqb = mull %vb1s, %vb2s
	%sum_sqb_new = addl %sum_sqb, %term_sqb
	%sum_sqb = addl %sum_sqb_new, 0
	%t_sqb_next = addl %t_sqb, 1
	%t_sqb = addl %t_sqb_next, 0
	jmp @sqb_t_loop
@sqb_store
	%idx_pp1s = mull %i_sqb, %n
	%idx_pps = addl %idx_pp1s, %j_sqb
	%off_pps = mull %idx_pps, 8
	%pp_pps = addl %prod_ptr, %off_pps
	storel %sum_sqb, %pp_pps
	%j_sqb_next = addl %j_sqb, 1
	%j_sqb = addl %j_sqb_next, 0
	jmp @sqb_j_loop
@sqb_i_next
	%i_sqb_next = addl %i_sqb, 1
	%i_sqb = addl %i_sqb_next, 0
	jmp @sqb_i_loop

@copy_prod_to_base
	%i_cpb = addl 0, 0
@cpb_loop
	%cpb_done = ceql %i_cpb, %sz
	jnz %cpb_done, @shift_kk
	%off_cpb = mull %i_cpb, 8
	%pp_cpb = addl %prod_ptr, %off_cpb
	%pb_cpb = addl %base_ptr, %off_cpb
	%val_cpb = loadl %pp_cpb
	storel %val_cpb, %pb_cpb
	%i_cpb_next = addl %i_cpb, 1
	%i_cpb = addl %i_cpb_next, 0
	jmp @cpb_loop

@shift_kk
	%kk_new = shrl %kk_cur, 1
	%kk_cur = addl %kk_new, 0
	jmp @pow_loop

@handle_kzero
	%zero_k = addl 0, 0
	%one_k = addl 0, 1
	%sz2_k = mull %sz, 2
	%i_id = addl 0, 0
@id_loop
	%id_done = ceql %i_id, %sz
	jnz %id_done, @clear_tmp_kzero
	%off_id = mull %i_id, 8
	%po_id = addl %out, %off_id
	%row_id = divl %i_id, %n
	%col_id = reml %i_id, %n
	%diag_id = ceql %row_id, %col_id
	jnz %diag_id, @id_set_one
	storel %zero_k, %po_id
	jmp @id_next
@id_set_one
	storel %one_k, %po_id
@id_next
	%i_id_next = addl %i_id, 1
	%i_id = addl %i_id_next, 0
	jmp @id_loop

@clear_tmp_kzero
	%i_ctk = addl 0, 0
@ctk_loop
	%ctk_done = ceql %i_ctk, %sz2_k
	jnz %ctk_done, @return_zero
	%off_ctk = mull %i_ctk, 8
	%pt_ctk = addl %tmp, %off_ctk
	storel %zero_k, %pt_ctk
	%i_ctk_next = addl %i_ctk, 1
	%i_ctk = addl %i_ctk_next, 0
	jmp @ctk_loop

@clear_tmp_all
	%zero_c = addl 0, 0
	%sz2_c = mull %sz, 2
	%i_cta = addl 0, 0
@cta_loop
	%cta_done = ceql %i_cta, %sz2_c
	jnz %cta_done, @return_zero
	%off_cta = mull %i_cta, 8
	%pt_cta = addl %tmp, %off_cta
	storel %zero_c, %pt_cta
	%i_cta_next = addl %i_cta, 1
	%i_cta = addl %i_cta_next, 0
	jmp @cta_loop

@return_zero
	%retv = addl 0, 0
	ret %retv
}
