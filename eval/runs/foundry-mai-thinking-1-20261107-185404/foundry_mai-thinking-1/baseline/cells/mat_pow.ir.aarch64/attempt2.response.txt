export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%sz = mull %n, %n
	%kisz = ceql %k, 0
	jnz %kisz, @kzero

@kgt0
	%zero = addl 0, 0
	%one  = addl 0, 1
	%sz2  = mull %sz, 2
	%boff = mull %sz, 8
	%base = addl %tmp, 0
	%prod = addl %tmp, %boff

@copy_base
	%i_cb = addl 0, 0
@cb_loop
	%cb_done = ceql %i_cb, %sz
	jnz %cb_done, @set_id
	%ea_cb = mull %i_cb, 8
	%pa_cb = addl %a, %ea_cb
	%vb    = loadl %pa_cb
	%pb_cb = addl %base, %ea_cb
	storel %vb, %pb_cb
	%i_cb1 = addl %i_cb, 1
	%i_cb  = addl %i_cb1, 0
	jmp @cb_loop

@set_id
	%i_si = addl 0, 0
@si_loop
	%si_done = ceql %i_si, %sz
	jnz %si_done, @pow_init
	%eo_si   = mull %i_si, 8
	%po_si   = addl %out, %eo_si
	%row_si  = divl %i_si, %n
	%col_si  = reml %i_si, %n
	%is_diag = ceql %row_si, %col_si
	jnz %is_diag, @si_one
	storel %zero, %po_si
	jmp @si_next
@si_one
	storel %one, %po_si
@si_next
	%i_si1 = addl %i_si, 1
	%i_si  = addl %i_si1, 0
	jmp @si_loop

@pow_init
	%kk = addl %k, 0

@pow_loop
	%kk_done = ceql %kk, 0
	jnz %kk_done, @clear_tmp
	%bbit  = andl %kk, 1
	%b0    = ceql %bbit, 0
	jnz %b0, @no_mul_r_b

@mul_r_b
	%i_mrb = addl 0, 0
@mr_i
	%mr_i_done = ceql %i_mrb, %n
	jnz %mr_i_done, @cp_r_prod
	%j_mrb = addl 0, 0
@mr_j
	%mr_j_done = ceql %j_mrb, %n
	jnz %mr_j_done, @mr_i_inc
	%s_mrb = addl 0, 0
	%t_mrb = addl 0, 0
@mr_t
	%mr_t_done = ceql %t_mrb, %n
	jnz %mr_t_done, @mr_st
	%idx_r1 = mull %i_mrb, %n
	%idx_r  = addl %idx_r1, %t_mrb
	%er     = mull %idx_r, 8
	%prm    = addl %out, %er
	%vr     = loadl %prm
	%idx_b1 = mull %t_mrb, %n
	%idx_b  = addl %idx_b1, %j_mrb
	%eb     = mull %idx_b, 8
	%pbm    = addl %base, %eb
	%vb2    = loadl %pbm
	%mterm  = mull %vr, %vb2
	%snew   = addl %s_mrb, %mterm
	%s_mrb  = addl %snew, 0
	%t_mrb1 = addl %t_mrb, 1
	%t_mrb  = addl %t_mrb1, 0
	jmp @mr_t
@mr_st
	%idx_p1 = mull %i_mrb, %n
	%idx_p  = addl %idx_p1, %j_mrb
	%ep     = mull %idx_p, 8
	%pp_st  = addl %prod, %ep
	storel %s_mrb, %pp_st
	%j_mrb1 = addl %j_mrb, 1
	%j_mrb  = addl %j_mrb1, 0
	jmp @mr_j
@mr_i_inc
	%i_mrb1 = addl %i_mrb, 1
	%i_mrb  = addl %i_mrb1, 0
	jmp @mr_i

@cp_r_prod
	%i_crp = addl 0, 0
@crp_loop
	%crp_done = ceql %i_crp, %sz
	jnz %crp_done, @no_mul_r_b
	%ecr   = mull %i_crp, 8
	%ppr   = addl %prod, %ecr
	%vpr   = loadl %ppr
	%por   = addl %out, %ecr
	storel %vpr, %por
	%i_crp1 = addl %i_crp, 1
	%i_crp  = addl %i_crp1, 0
	jmp @crp_loop

@no_mul_r_b
@mul_b_b
	%i_mbb = addl 0, 0
@mb_i
	%mb_i_done = ceql %i_mbb, %n
	jnz %mb_i_done, @cp_b_prod
	%j_mbb = addl 0, 0
@mb_j
	%mb_j_done = ceql %j_mbb, %n
	jnz %mb_j_done, @mb_i_inc
	%s_mbb = addl 0, 0
	%t_mbb = addl 0, 0
@mb_t
	%mb_t_done = ceql %t_mbb, %n
	jnz %mb_t_done, @mb_st
	%idx_b11 = mull %i_mbb, %n
	%idx_b1m = addl %idx_b11, %t_mbb
	%eb1     = mull %idx_b1m, 8
	%pbb1    = addl %base, %eb1
	%vb1m    = loadl %pbb1
	%idx_b21 = mull %t_mbb, %n
	%idx_b2m = addl %idx_b21, %j_mbb
	%eb2     = mull %idx_b2m, 8
	%pbb2    = addl %base, %eb2
	%vb2m    = loadl %pbb2
	%m2      = mull %vb1m, %vb2m
	%s2n     = addl %s_mbb, %m2
	%s_mbb   = addl %s2n, 0
	%t_mbb1  = addl %t_mbb, 1
	%t_mbb   = addl %t_mbb1, 0
	jmp @mb_t
@mb_st
	%idx_pp1 = mull %i_mbb, %n
	%idx_pp  = addl %idx_pp1, %j_mbb
	%epp     = mull %idx_pp, 8
	%ppp_st  = addl %prod, %epp
	storel %s_mbb, %ppp_st
	%j_mbb1  = addl %j_mbb, 1
	%j_mbb   = addl %j_mbb1, 0
	jmp @mb_j
@mb_i_inc
	%i_mbb1 = addl %i_mbb, 1
	%i_mbb  = addl %i_mbb1, 0
	jmp @mb_i

@cp_b_prod
	%i_cbp = addl 0, 0
@cbp_loop
	%cbp_done = ceql %i_cbp, %sz
	jnz %cbp_done, @kk_shift
	%ecb   = mull %i_cbp, 8
	%ppb   = addl %prod, %ecb
	%vpb   = loadl %ppb
	%pbbd  = addl %base, %ecb
	storel %vpb, %pbbd
	%i_cbp1 = addl %i_cbp, 1
	%i_cbp  = addl %i_cbp1, 0
	jmp @cbp_loop

@kk_shift
	%kk1 = shrl %kk, 1
	%kk  = addl %kk1, 0
	jmp @pow_loop

@clear_tmp
	%i_ct = addl 0, 0
@ct_loop
	%ct_done = ceql %i_ct, %sz2
	jnz %ct_done, @ret0
	%ect  = mull %i_ct, 8
	%pt   = addl %tmp, %ect
	storel %zero, %pt
	%i_ct1 = addl %i_ct, 1
	%i_ct  = addl %i_ct1, 0
	jmp @ct_loop

@kzero
	%zero2 = addl 0, 0
	%one2  = addl 0, 1
	%sz2k  = mull %sz, 2
	%i_si2 = addl 0, 0
@si2_loop
	%si2_done = ceql %i_si2, %sz
	jnz %si2_done, @ct2
	%eo2   = mull %i_si2, 8
	%po2   = addl %out, %eo2
	%row2  = divl %i_si2, %n
	%col2  = reml %i_si2, %n
	%d2    = ceql %row2, %col2
	jnz %d2, @si2_one
	storel %zero2, %po2
	jmp @si2_next
@si2_one
	storel %one2, %po2
@si2_next
	%i_si21 = addl %i_si2, 1
	%i_si2  = addl %i_si21, 0
	jmp @si2_loop

@ct2
	%i_ct2 = addl 0, 0
@ct2_loop
	%ct2_done = ceql %i_ct2, %sz2k
	jnz %ct2_done, @ret0
	%ect2 = mull %i_ct2, 8
	%pt2  = addl %tmp, %ect2
	storel %zero2, %pt2
	%i_ct21 = addl %i_ct2, 1
	%i_ct2  = addl %i_ct21, 0
	jmp @ct2_loop

@ret0
	%r = addl 0, 0
	ret %r
}
