export function l $count_desc(l %a, l %n) {
@start
	%lt2 = csltl %n, 2l
	jnz %lt2, @ret_zero
	%count = movl 0l
	%i = movl 0l
	%max_i = subl %n, 2l
@loop
	%is_done = csgtl %i, %max_i
	jnz %is_done, @ret_count
	%off1 = shl %i, 3l
	%v1 = loadl %a, %off1
	%i1 = addl %i, 1l
	%off2 = shl %i1, 3l
	%v2 = loadl %a, %off2
	%is_desc = csgtl %v1, %v2
	jnz %is_desc, @inc_count
@next
	%i = addl %i, 1l
	jmp @loop
@inc_count
	%count = addl %count, 1l
	jmp @next
@ret_count
	ret %count
@ret_zero
	ret 0l
}
