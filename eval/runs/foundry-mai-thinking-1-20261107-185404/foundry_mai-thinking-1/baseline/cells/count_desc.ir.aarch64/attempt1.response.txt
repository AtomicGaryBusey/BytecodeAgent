export function l $count_desc(l %a, l %n) {
@start
	%lt2 = csltl %n, 2
	jnz %lt2, @ret_zero
	%count = copyl 0
	%i = copyl 0
	%max_i = subl %n, 2
@loop
	%is_done = csgtl %i, %max_i
	jnz %is_done, @ret_count
	%off1 = shll %i, 3
	%v1 = loadl %a, %off1
	%i1 = addl %i, 1
	%off2 = shll %i1, 3
	%v2 = loadl %a, %off2
	%is_desc = csgtl %v1, %v2
	jnz %is_desc, @inc_count
@next
	%i = addl %i, 1
	jmp @loop
@inc_count
	%count_new = addl %count, 1
	%count = copyl %count_new
	jmp @next
@ret_count
	ret %count
@ret_zero
	%zero_res = copyl 0
	ret %zero_res
}
