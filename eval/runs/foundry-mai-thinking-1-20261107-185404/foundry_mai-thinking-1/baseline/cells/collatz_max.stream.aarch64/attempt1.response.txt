collatz_max:
  mov x1, #1
  mov x2, #0
.Louter_loop:
  cmp x1, x0
  b.gt .Lend
  mov x3, x1
  mov x4, #0
  cmp x3, #1
  b.eq .Lupdate_max
.Lcollatz_step:
  and x5, x3, #1
  cbnz x5, .Lodd_case
  lsr x3, x3, #1
  b .Lstep_count
.Lodd_case:
  add x3, x3, x3, lsl #1
  add x3, x3, #1
.Lstep_count:
  add x4, x4, #1
  cmp x3, #1
  b.ne .Lcollatz_step
.Lupdate_max:
  cmp x4, x2
  csel x2, x4, x2, hi
  add x1, x1, #1
  b .Louter_loop
.Lend:
  mov x0, x2
  ret
