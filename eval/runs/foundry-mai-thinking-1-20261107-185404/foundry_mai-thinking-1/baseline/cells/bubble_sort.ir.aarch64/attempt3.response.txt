export function l $bubble_sort(l %a, l %n) {
@start
	%res = copy 0l
	%i = copy %n
	%zero_check = ceql %n, 0l
	jnz %zero_check, @end, @outer_check

@outer_check
	%outer_cond = csgtl %i, 1l
	jnz %outer_cond, @outer_body, @end

@outer_body
	%j = copy 0l
	%limit = sub %i, 1l
	jmp @inner_check

@inner_check
	%inner_cond = csltl %j, %limit
	jnz %inner_cond, @compare, @outer_dec

@compare
	%off1 = shl %j, 3l
	%j2 = add %j, 1l
	%off2 = shl %j2, 3l
	%v1 = loadl %a + %off1
	%v2 = loadl %a + %off2
	%swap_cond = csgtl %v1, %v2
	jnz %swap_cond, @swap, @inner_inc

@swap
	storel %v2, %a + %off1
	storel %v1, %a + %off2
	jmp @inner_inc

@inner_inc
	%j_new = add %j, 1l
	%j = copy %j_new
	jmp @inner_check

@outer_dec
	%i_new = sub %i, 1l
	%i = copy %i_new
	jmp @outer_check

@end
	ret %res
}
