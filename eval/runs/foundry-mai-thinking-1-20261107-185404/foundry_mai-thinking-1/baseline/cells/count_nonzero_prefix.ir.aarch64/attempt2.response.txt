export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%is_zero_n = ceql %n, 0
	jnz %is_zero_n, @ret_zero, @loop_init

@loop_init
	%i = 0
	jmp @loop_check

@loop_check
	%in_range = csltl %i, %n
	jnz %in_range, @loop_body, @ret_n

@loop_body
	%off = shll %i, 3
	%addr = addl %a, %off
	%val = loadl %addr
	%is_zero_val = ceql %val, 0
	jnz %is_zero_val, @ret_i, @loop_inc

@loop_inc
	%i = addl %i, 1
	jmp @loop_check

@ret_zero
	ret 0

@ret_i
	ret %i

@ret_n
	ret %n
}
