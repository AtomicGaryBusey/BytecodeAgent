export function l $qsort_pairs(l %a, l %n) {
@entry
	%le1 =w csltl %n, 2
	jnz %le1, @ret_zero
	%last =l sub %n, 1
	%plo =l mul %last, 16
	%pptr =l add %a, %plo
	%pk =l loadl %pptr
	%pv =l loadl %pptr, 8
	%i =l sub 0, 1
	%j =l 0
	jmp @loop_j

@loop_j
	%j_eq_last =w ceql %j, %last
	jnz %j_eq_last, @part_end
	%jlo =l mul %j, 16
	%jptr =l add %a, %jlo
	%jk =l loadl %jptr
	%jv =l loadl %jptr, 8
	%jk_lt_pk =w csltl %jk, %pk
	jnz %jk_lt_pk, @do_swap
	%jk_eq_pk =w ceql %jk, %pk
	jnz %jk_eq_pk, @check_v
	jmp @j_next

@check_v
	%jv_le_pv =w csleql %jv, %pv
	jnz %jv_le_pv, @do_swap
	jmp @j_next

@do_swap
	%inew =l add %i, 1
	%ilo =l mul %inew, 16
	%jlo2 =l mul %j, 16
	%iptr =l add %a, %ilo
	%jptr2 =l add %a, %jlo2
	%tk1 =l loadl %iptr
	%tv1 =l loadl %iptr, 8
	%tk2 =l loadl %jptr2
	%tv2 =l loadl %jptr2, 8
	storel %tk2, %iptr
	storel %tv2, %iptr, 8
	storel %tk1, %jptr2
	storel %tv1, %jptr2, 8
	%i =l add %i, 1
	jmp @j_next

@j_next
	%j =l add %j, 1
	jmp @loop_j

@part_end
	%p1 =l add %i, 1
	%p1lo =l mul %p1, 16
	%plastlo =l mul %last, 16
	%p1ptr =l add %a, %p1lo
	%plastptr =l add %a, %plastlo
	%tkp1 =l loadl %p1ptr
	%tvp1 =l loadl %p1ptr, 8
	%tkpl =l loadl %plastptr
	%tvpl =l loadl %plastptr, 8
	storel %tkpl, %p1ptr
	storel %tvpl, %p1ptr, 8
	storel %tkp1, %plastptr
	storel %tvp1, %plastptr, 8
	%nleft =l copy %p1
	%pleft =l copy %a
	%mright_cnt =l sub %n, %p1
	%mright_cnt =l sub %mright_cnt, 1
	%pright_off =l mul %p1, 16
	%pright_off =l add %pright_off, 16
	%pright =l add %a, %pright_off
	%cleft =w csgtl %nleft, 1
	jnz %cleft, @call_left
	jmp @check_right

@call_left
	call $qsort_pairs(l %pleft, l %nleft)

@check_right
	%cright =w csgtl %mright_cnt, 1
	jnz %cright, @call_right
	jmp @ret_zero

@call_right
	call $qsort_pairs(l %pright, l %mright_cnt)
	jmp @ret_zero

@ret_zero
	ret 0
}
