export function l $qsort_pairs(l %a, l %n) {
@entry
	%lt2 =w csltl %n, 2
	jnz %lt2, @ret_zero
	%last =l sub %n, 1
	%lastoff =l mul %last, 16
	%pptr =l add %a, %lastoff
	%pk =l loadl %pptr
	%pv =l loadl %pptr, 8
	%i =l sub 0, 1
	%j =l 0
	jmp @loop_j

@loop_j
	%jeqlast =w ceql %j, %last
	jnz %jeqlast, @part_end
	%joff =l mul %j, 16
	%jptr =l add %a, %joff
	%jk =l loadl %jptr
	%jv =l loadl %jptr, 8
	%jltpk =w csltl %jk, %pk
	jnz %jltpk, @do_swap
	%jkeqpk =w ceql %jk, %pk
	jnz %jkeqpk, @check_jv
	jmp @j_next

@check_jv
	%jvlepv =w csleql %jv, %pv
	jnz %jvlepv, @do_swap
	jmp @j_next

@do_swap
	%inew =l add %i, 1
	%ioff =l mul %inew, 16
	%joff2 =l mul %j, 16
	%iptr =l add %a, %ioff
	%jptr2 =l add %a, %joff2
	%tik =l loadl %iptr
	%tiv =l loadl %iptr, 8
	%tjk =l loadl %jptr2
	%tjv =l loadl %jptr2, 8
	storel %tjk, %iptr
	storel %tjv, %iptr, 8
	storel %tik, %jptr2
	storel %tiv, %jptr2, 8
	%i =l add %inew, 0
	jmp @j_next

@j_next
	%j =l add %j, 1
	jmp @loop_j

@part_end
	%p1 =l add %i, 1
	%p1off =l mul %p1, 16
	%lastoff2 =l mul %last, 16
	%p1ptr =l add %a, %p1off
	%plastptr =l add %a, %lastoff2
	%tp1k =l loadl %p1ptr
	%tp1v =l loadl %p1ptr, 8
	%tplk =l loadl %plastptr
	%tplv =l loadl %plastptr, 8
	storel %tplk, %p1ptr
	storel %tplv, %p1ptr, 8
	storel %tp1k, %plastptr
	storel %tp1v, %plastptr, 8
	%nleft =l copy %p1
	%pleftptr =l copy %a
	%mright =l sub %n, %p1
	%mright =l sub %mright, 1
	%prightoff =l mul %p1, 16
	%prightptr =l add %a, %prightoff
	%prightptr2 =l add %prightptr, 16
	%cleft =w csgtl %nleft, 1
	jnz %cleft, @call_left
	jmp @check_right

@call_left
	call $qsort_pairs(l %pleftptr, l %nleft)

@check_right
	%cright =w csgtl %mright, 1
	jnz %cright, @call_right
	jmp @ret_zero

@call_right
	call $qsort_pairs(l %prightptr2, l %mright)
	jmp @ret_zero

@ret_zero
	ret 0
}
