export function l $binary_search(l %a, l %n, l %key) {
@start
	%iszero = ceql %n, $0
	jnz %iszero, @notfound, @init

@init
	%left = $0
	%right = subl %n, $1
	jmp @loopcond

@loopcond
	%cond = cslel %left, %right
	jnz %cond, @loopbody, @notfound

@loopbody
	%mid = addl %left, %right
	%mid = sarl %mid, $1
	%off = shll %mid, $3
	%ptr = addl %a, %off
	%val = loadl %ptr
	%eq = ceql %val, %key
	jnz %eq, @found, @checklt

@checklt
	%lt = csltl %val, %key
	jnz %lt, @goright, @goleft

@goleft
	%left = addl %mid, $1
	jmp @loopcond

@goright
	%right = subl %mid, $1
	jmp @loopcond

@found
	ret %mid

@notfound
	ret $-1
}
