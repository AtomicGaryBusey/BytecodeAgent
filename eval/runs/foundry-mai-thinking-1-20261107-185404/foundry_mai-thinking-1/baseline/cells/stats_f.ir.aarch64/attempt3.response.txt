export function l $stats_f(l %a, l %n, l %fptr, l %out) {
@start
	%zero =l add 0, 0
	%one  =l add 0, 1
	%eight =l add 0, 8

	%is_n_zero =w ceql %n, %zero
	jnz %is_n_zero, @store_zero

	%first_elem =l loadl %a
	%y0 =l call %fptr(l %first_elem)

	%min_init =l copy %y0
	%max_init =l copy %y0
	%sum_init =l copy %y0

	%y0_and1 =w and %y0, %one
	%y0_is_odd =w ceql %y0_and1, %one
	jnz %y0_is_odd, @odd0
	%oddcount_init =l add 0, 0
	jmp @after0

@odd0
	%oddcount_init =l add 0, 1
	jmp @after0

@after0
	%idx =l add 0, 1
	%min_cur =l copy %min_init
	%max_cur =l copy %max_init
	%sum_cur =l copy %sum_init
	%odd_cur =l copy %oddcount_init
	jmp @loop_check

@loop_check
	%cmp =w csltl %idx, %n
	jnz %cmp, @loop_body
	jmp @store_results

@loop_body
	%off =l mul %idx, %eight
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%y =l call %fptr(l %val)

	%cmp_min =w csltl %y, %min_cur
	jnz %cmp_min, @set_min
	jmp @check_max

@set_min
	%min_cur2 =l copy %y
	jmp @check_max2

@check_max
	%min_cur2 =l copy %min_cur
	jmp @check_max2

@check_max2
	%cmp_max =w csltl %max_cur, %y
	jnz %cmp_max, @set_max
	jmp @check_sum

@set_max
	%max_cur2 =l copy %y
	jmp @check_sum2

@check_sum
	%max_cur2 =l copy %max_cur
	jmp @check_sum2

@check_sum2
	%sum_cur2 =l add %sum_cur, %y

	%y_and1b =w and %y, %one
	%y_is_oddb =w ceql %y_and1b, %one
	jnz %y_is_oddb, @inc_odd
	%odd_cur2 =l copy %odd_cur
	jmp @loop_next

@inc_odd
	%odd_cur2 =l add %odd_cur, %one
	jmp @loop_next

@loop_next
	%min_cur =l copy %min_cur2
	%max_cur =l copy %max_cur2
	%sum_cur =l copy %sum_cur2
	%odd_cur =l copy %odd_cur2
	%idx2 =l add %idx, %one
	%idx =l copy %idx2
	jmp @loop_check

@store_zero
	%min_res =l add 0, 0
	%max_res =l add 0, 0
	%sum_res =l add 0, 0
	%ret_zero =l add 0, 0
	jmp @store_final

@store_results
	%min_res =l copy %min_cur
	%max_res =l copy %max_cur
	%sum_res =l copy %sum_cur
	%ret_res =l copy %odd_cur
	jmp @store_final

@store_final
	%out0 =l add %out, %zero
	storel %min_res, %out0
	%out1 =l add %out, %eight
	storel %max_res, %out1
	%out2 =l add %out, %eight
	%out2p =l add %out2, %eight
	storel %sum_res, %out2p
	%retv =l phi @store_zero %ret_zero, @store_results %ret_res
	ret %retv
}
