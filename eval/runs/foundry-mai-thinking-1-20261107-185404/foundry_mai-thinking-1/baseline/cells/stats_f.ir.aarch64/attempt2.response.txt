export function l $stats_f(l %a, l %n, l %fptr, l %out) {
@start
	%zero =l add 0, 0
	%one  =l add 0, 1
	%eight= l add 0, 8
	%idx  =l add 0, 0
	%oddcount =l add 0, 0
	%sum  =l add 0, 0
	%min  =l add 0, 0
	%max  =l add 0, 0

	%is_n_zero = w ceql %n, %zero
	jnz %is_n_zero, @store_results

	%first_val =l loadl %a
	%y0 =l call %fptr(l %first_val)

	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0

	%y0_and1 = w and %y0, %one
	%y0_is_odd = w ceql %y0_and1, %one
	jnz %y0_is_odd, @set_odd0

	%oddcount =l add 0, 0
	jmp @after_first

@set_odd0
	%oddcount =l add 0, 1
	jmp @after_first

@after_first
	%idx =l add 0, 1

@loop
	%cmp_idx_n = w csltl %idx, %n
	jnz %cmp_idx_n, @loop_body
	jmp @store_results

@loop_body
	%offset =l mul %idx, %eight
	%elem_ptr =l add %a, %offset
	%elem_val =l loadl %elem_ptr
	%y =l call %fptr(l %elem_val)

	%cmp_min = w csltl %y, %min
	jnz %cmp_min, @update_min
	jmp @check_max

@update_min
	%min =l copy %y
	jmp @check_max

@check_max
	%cmp_max = w csltl %max, %y
	jnz %cmp_max, @update_max
	jmp @update_sum

@update_max
	%max =l copy %y
	jmp @update_sum

@update_sum
	%sum =l add %sum, %y

	%y_and1 = w and %y, %one
	%y_is_odd = w ceql %y_and1, %one
	jnz %y_is_odd, @inc_odd
	jmp @loop_next

@inc_odd
	%oddcount =l add %oddcount, %one
	jmp @loop_next

@loop_next
	%idx =l add %idx, %one
	jmp @loop

@store_results
	%out0_ptr =l add %out, %zero
	storel %min, %out0_ptr

	%out1_ptr =l add %out, %eight
	storel %max, %out1_ptr

	%out2_ptr =l add %out, %eight
	%out2_ptr2 =l add %out2_ptr, %eight
	storel %sum, %out2_ptr2

	ret %oddcount
}
