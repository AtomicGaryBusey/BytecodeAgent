export function l $stats_f(l %a, l %n, l %fptr, l %out) {
@start
	%zero =l add 0, 0
	%one  =l add 0, 1
	%eight =l add 0, 8

	%n_is_zero =w ceql %n, %zero
	jnz %n_is_zero, @n_zero

	%idx =l add 0, 0
	%first_ptr =l add %a, %zero
	%first_val =l loadl %first_ptr
	%first_y =l call %fptr(l %first_val)

	%min_cur =l copy %first_y
	%max_cur =l copy %first_y
	%sum_cur =l copy %first_y

	%first_and1 =w and %first_y, %one
	%first_odd =w ceql %first_and1, %one
	jnz %first_odd, @first_is_odd
	%odd_cur =l add 0, 0
	jmp @loop_init

@first_is_odd
	%odd_cur =l add 0, 1
	jmp @loop_init

@loop_init
	%idx =l add 0, 1
	jmp @loop_check

@loop_check
	%in_range =w csltl %idx, %n
	jnz %in_range, @loop_body
	jmp @store_res

@loop_body
	%off =l mul %idx, %eight
	%elem_ptr =l add %a, %off
	%elem_val =l loadl %elem_ptr
	%y =l call %fptr(l %elem_val)

	%is_less =w csltl %y, %min_cur
	jnz %is_less, @set_min
	%min_new =l copy %min_cur
	jmp @check_max

@set_min
	%min_new =l copy %y
	jmp @check_max

@check_max
	%is_greater =w csltl %max_cur, %y
	jnz %is_greater, @set_max
	%max_new =l copy %max_cur
	jmp @check_sum

@set_max
	%max_new =l copy %y
	jmp @check_sum

@check_sum
	%sum_new =l add %sum_cur, %y

	%y_and1b =w and %y, %one
	%y_odd =w ceql %y_and1b, %one
	jnz %y_odd, @inc_odd
	%odd_new =l copy %odd_cur
	jmp @loop_next

@inc_odd
	%odd_new =l add %odd_cur, %one
	jmp @loop_next

@loop_next
	%min_cur =l copy %min_new
	%max_cur =l copy %max_new
	%sum_cur =l copy %sum_new
	%odd_cur =l copy %odd_new
	%idx_next =l add %idx, %one
	%idx =l copy %idx_next
	jmp @loop_check

@n_zero
	%min_res_z =l add 0, 0
	%max_res_z =l add 0, 0
	%sum_res_z =l add 0, 0
	%ret_res_z =l add 0, 0
	jmp @store_final

@store_res
	%min_res_r =l copy %min_cur
	%max_res_r =l copy %max_cur
	%sum_res_r =l copy %sum_cur
	%ret_res_r =l copy %odd_cur
	jmp @store_final

@store_final
	storel %min_res_z, %out
	%out1 =l add %out, %eight
	storel %max_res_z, %out1
	%out2 =l add %out, %eight
	%out2p =l add %out2, %eight
	storel %sum_res_z, %out2p
	%retv =l phi @n_zero %ret_res_z, @store_res %ret_res_r
	ret %retv
}
