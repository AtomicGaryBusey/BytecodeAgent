export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_in =w csltl %i, %n
	jnz %cond_in, @process, @end_ok

@process
	%r_run =l copy 1
	%p_run =l add %i, 1
	jmp @run_loop

@run_loop
	%p_lt_n =w csltl %p_run, %n
	jnz %p_lt_n, @run_chk_eq, @run_done

@run_chk_eq
	%bcur =w loadub %src +l %i
	%bnext =w loadub %src +l %p_run
	%eq_bn =w ceql %bcur, %bnext
	jnz %eq_bn, @run_inc, @run_done

@run_inc
	%rnew =l add %r_run, 1
	%pnew =l add %p_run, 1
	%rmax_c =w ceql %rnew, 128
	jnz %rmax_c, @run_done
	%r_run =l copy %rnew
	%p_run =l copy %pnew
	jmp @run_loop

@run_done
	%r_ge3 =w csgel %r_run, 3
	jnz %r_ge3, @emit_repeat, @emit_literal

@emit_repeat
	%need_rep =l add %o, 2
	%fits_rep =w cslel %need_rep, %cap
	jnz %fits_rep, @do_repeat, @ret_neg1

@do_repeat
	%hdr_rep =l sub 257, %r_run
	%outp_hr =l add %out, %o
	storeb %hdr_rep, %outp_hr
	%o1_r =l add %o, 1
	%outp_br =l add %out, %o1_r
	%bsrc_r =w loadub %src +l %i
	storeb %bsrc_r, %outp_br
	%o2_r =l add %o1_r, 1
	%i2_r =l add %i, %r_run
	%o =l copy %o2_r
	%i =l copy %i2_r
	jmp @loop

@emit_literal
	%j_lit =l copy %i
	jmp @lit_loop

@lit_loop
	%j_lt_nl =w csltl %j_lit, %n
	jnz %j_lt_nl, @lit_chk_cnt, @lit_done_scan

@lit_chk_cnt
	%cnt_cur_l =l sub %j_lit, %i
	%cnt_lt128l =w csltl %cnt_cur_l, 128
	jnz %cnt_lt128l, @lit_run_at_j, @lit_done_scan

@lit_run_at_j
	%rj =l copy 1
	%pj =l add %j_lit, 1
	jmp @lit_run_loop

@lit_run_loop
	%pj_lt_n =w csltl %pj, %n
	jnz %pj_lt_n, @lit_run_chk, @lit_run_done

@lit_run_chk
	%bcj =w loadub %src +l %j_lit
	%bnj =w loadub %src +l %pj
	%eq_j =w ceql %bcj, %bnj
	jnz %eq_j, @lit_run_inc, @lit_run_done

@lit_run_inc
	%rjnew =l add %rj, 1
	%pjnew =l add %pj, 1
	%rjmax =w ceql %rjnew, 128
	jnz %rjmax, @lit_run_done
	%rj =l copy %rjnew
	%pj =l copy %pjnew
	jmp @lit_run_loop

@lit_run_done
	%rj_ge3 =w csgel %rj, 3
	jnz %rj_ge3, @lit_done_scan, @lit_add_rj

@lit_add_rj
	%jnew_l =l add %j_lit, %rj
	%j_lit =l copy %jnew_l
	jmp @lit_loop

@lit_done_scan
	%L_lit =l sub %j_lit, %i
	%need_l1 =l add %o, 1
	%need_l2 =l add %need_l1, %L_lit
	%fits_lit =w cslel %need_l2, %cap
	jnz %fits_lit, @do_literal, @ret_neg1

@do_literal
	%hdr_lit =l sub %L_lit, 1
	%op_hl =l add %out, %o
	storeb %hdr_lit, %op_hl
	%o_cur_l =l add %o, 1
	%k_cp =l copy 0
	jmp @copy_loop

@copy_loop
	%k_lt_l =w csltl %k_cp, %L_lit
	jnz %k_lt_l, @copy_byte, @copy_done

@copy_byte
	%sp_cp =l add %src, %i
	%spk_cp =l add %sp_cp, %k_cp
	%bcp_v =w loadub %spk_cp
	%dp_cp =l add %out, %o_cur_l
	%dpk_cp =l add %dp_cp, %k_cp
	storeb %bcp_v, %dpk_cp
	%k1_cp =l add %k_cp, 1
	%k_cp =l copy %k1_cp
	jmp @copy_loop

@copy_done
	%o_new_l =l add %o_cur_l, %L_lit
	%i_new_l =l copy %j_lit
	%o =l copy %o_new_l
	%i =l copy %i_new_l
	jmp @loop

@end_ok
	ret %o

@ret_neg1
	%neg1v =l sub 0, 1
	ret %neg1v
}
