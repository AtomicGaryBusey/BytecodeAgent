export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pos =l copy 0
	%outidx =l copy 0
	jmp @loop

@loop
	%pos_lt_n =w csltl %pos, %n
	jnz %pos_lt_n, @process, @end_ok

@process
	%run_len =l copy 1
	%pnext =l add %pos, 1
	jmp @run_loop

@run_loop
	%pnext_lt_n =w csltl %pnext, %n
	jnz %pnext_lt_n, @run_check_eq, @run_done

@run_check_eq
	%srcp_cur =l add %src, %pos
	%bcur =w loadub %srcp_cur
	%srcp_nxt =l add %src, %pnext
	%bnxt =w loadub %srcp_nxt
	%beq =w ceql %bcur, %bnxt
	jnz %beq, @run_inc, @run_done

@run_inc
	%run_new =l add %run_len, 1
	%pnew =l add %pnext, 1
	%run_max128 =w ceql %run_new, 128
	jnz %run_max128, @run_done
	%run_len =l copy %run_new
	%pnext =l copy %pnew
	jmp @run_loop

@run_done
	%run_ge3 =w csgel %run_len, 3
	jnz %run_ge3, @emit_repeat, @emit_literal

@emit_repeat
	%need_r1 =l add %outidx, 1
	%need_r2 =l add %need_r1, 1
	%fits_r =w cslel %need_r2, %cap
	jnz %fits_r, @do_repeat, @ret_neg1

@do_repeat
	%hdr_r_l =l sub 257, %run_len
	%hdr_r_w =w extrl %hdr_r_l, 0
	%outp_hr =l add %out, %outidx
	storeb %hdr_r_w, %outp_hr
	%outidx1_r =l add %outidx, 1
	%srcp_br =l add %src, %pos
	%bval_r_w =w loadub %srcp_br
	%outp_br =l add %out, %outidx1_r
	storeb %bval_r_w, %outp_br
	%outidx2_r =l add %outidx1_r, 1
	%pos_r_new =l add %pos, %run_len
	%outidx =l copy %outidx2_r
	%pos =l copy %pos_r_new
	jmp @loop

@emit_literal
	%jlit =l copy %pos
	jmp @lit_scan_loop

@lit_scan_loop
	%jlit_lt_n =w csltl %jlit, %n
	jnz %jlit_lt_n, @lit_check_cnt, @lit_scan_done

@lit_check_cnt
	%cnt_l_cur =l sub %jlit, %pos
	%cnt_lt128 =w csltl %cnt_l_cur, 128
	jnz %cnt_lt128, @lit_check_run, @lit_scan_done

@lit_check_run
	%run_j_cur =l copy 1
	%pj_cur =l add %jlit, 1
	jmp @lit_run_loop

@lit_run_loop
	%pj_cur_lt_n =w csltl %pj_cur, %n
	jnz %pj_cur_lt_n, @lit_run_check_eq, @lit_run_done

@lit_run_check_eq
	%srcp_cj =l add %src, %jlit
	%bcj_w =w loadub %srcp_cj
	%srcp_nj =l add %src, %pj_cur
	%bnj_w =w loadub %srcp_nj
	%eqcj =w ceql %bcj_w, %bnj_w
	jnz %eqcj, @lit_run_inc, @lit_run_done

@lit_run_inc
	%run_j_new =l add %run_j_cur, 1
	%pj_new =l add %pj_cur, 1
	%run_j_max128 =w ceql %run_j_new, 128
	jnz %run_j_max128, @lit_run_done
	%run_j_cur =l copy %run_j_new
	%pj_cur =l copy %pj_new
	jmp @lit_run_loop

@lit_run_done
	%run_j_ge3 =w csgel %run_j_cur, 3
	jnz %run_j_ge3, @lit_scan_done, @lit_advance

@lit_advance
	%jlit_new_adv =l add %jlit, %run_j_cur
	%jlit =l copy %jlit_new_adv
	jmp @lit_scan_loop

@lit_scan_done
	%Llit =l sub %jlit, %pos
	%need_l1 =l add %outidx, 1
	%need_l2 =l add %need_l1, %Llit
	%fits_l =w cslel %need_l2, %cap
	jnz %fits_l, @do_literal, @ret_neg1

@do_literal
	%hdr_l_l =l sub %Llit, 1
	%hdr_l_w =w extrl %hdr_l_l, 0
	%outp_hl =l add %out, %outidx
	storeb %hdr_l_w, %outp_hl
	%outidx_l1 =l add %outidx, 1
	%k =l copy 0
	jmp @copy_loop

@copy_loop
	%k_lt_l =w csltl %k, %Llit
	jnz %k_lt_l, @copy_byte, @copy_done

@copy_byte
	%srcp_ck_base =l add %src, %pos
	%srcp_ck =l add %srcp_ck_base, %k
	%bcp_w =w loadub %srcp_ck
	%outp_ck_base =l add %out, %outidx_l1
	%outp_ck =l add %outp_ck_base, %k
	storeb %bcp_w, %outp_ck
	%k_new =l add %k, 1
	%k =l copy %k_new
	jmp @copy_loop

@copy_done
	%outidx_l_new =l add %outidx_l1, %Llit
	%pos_l_new =l copy %jlit
	%outidx =l copy %outidx_l_new
	%pos =l copy %pos_l_new
	jmp @loop

@end_ok
	ret %outidx

@ret_neg1
	%neg1 =l sub 0, 1
	ret %neg1
}
