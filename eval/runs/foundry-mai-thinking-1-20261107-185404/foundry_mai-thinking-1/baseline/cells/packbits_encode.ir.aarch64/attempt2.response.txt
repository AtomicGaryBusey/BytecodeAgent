export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pos =l copy 0
	%outidx =l copy 0
	jmp @loop

@loop
	%pos_lt_n =w csltl %pos, %n
	jnz %pos_lt_n, @process, @end_ok

@process
	%run_len =l copy 1
	%pnext =l add %pos, 1
	jmp @run_loop

@run_loop
	%pnext_lt_n =w csltl %pnext, %n
	jnz %pnext_lt_n, @run_check_eq, @run_done

@run_check_eq
	%srcp_cur =l add %src, %pos
	%bcur =w loadub %srcp_cur
	%srcp_nxt =l add %src, %pnext
	%bnxt =w loadub %srcp_nxt
	%beq =w ceql %bcur, %bnxt
	jnz %beq, @run_inc, @run_done

@run_inc
	%run_new =l add %run_len, 1
	%pnew =l add %pnext, 1
	%run_max =w ceql %run_new, 128
	jnz %run_max, @run_done
	%run_len =l copy %run_new
	%pnext =l copy %pnew
	jmp @run_loop

@run_done
	%run_ge3 =w csgel %run_len, 3
	jnz %run_ge3, @emit_repeat, @emit_literal

@emit_repeat
	%need_r =l add %outidx, 2
	%fits_r =w cslel %need_r, %cap
	jnz %fits_r, @do_repeat, @ret_neg1

@do_repeat
	%hdr_r =l sub 257, %run_len
	%outp_hr =l add %out, %outidx
	storeb %hdr_r, %outp_hr
	%outidx1 =l add %outidx, 1
	%srcp_br =l add %src, %pos
	%bval_r =w loadub %srcp_br
	%outp_br =l add %out, %outidx1
	storeb %bval_r, %outp_br
	%outidx2 =l add %outidx1, 1
	%pos_r =l add %pos, %run_len
	%outidx =l copy %outidx2
	%pos =l copy %pos_r
	jmp @loop

@emit_literal
	%jlit =l copy %pos
	jmp @lit_scan_loop

@lit_scan_loop
	%jlit_lt_n =w csltl %jlit, %n
	jnz %jlit_lt_n, @lit_check_cnt, @lit_scan_done

@lit_check_cnt
	%cnt_lit_cur =l sub %jlit, %pos
	%cnt_lt128 =w csltl %cnt_lit_cur, 128
	jnz %cnt_lt128, @lit_check_run, @lit_scan_done

@lit_check_run
	%run_j =l copy 1
	%pj =l add %jlit, 1
	jmp @lit_run_loop

@lit_run_loop
	%pj_lt_n =w csltl %pj, %n
	jnz %pj_lt_n, @lit_run_check_eq, @lit_run_done

@lit_run_check_eq
	%srcp_cj =l add %src, %jlit
	%bcj =w loadub %srcp_cj
	%srcp_nj =l add %src, %pj
	%bnj =w loadub %srcp_nj
	%eqcj =w ceql %bcj, %bnj
	jnz %eqcj, @lit_run_inc, @lit_run_done

@lit_run_inc
	%run_jn =l add %run_j, 1
	%pjn =l add %pj, 1
	%run_jmax =w ceql %run_jn, 128
	jnz %run_jmax, @lit_run_done
	%run_j =l copy %run_jn
	%pj =l copy %pjn
	jmp @lit_run_loop

@lit_run_done
	%run_jge3 =w csgel %run_j, 3
	jnz %run_jge3, @lit_scan_done, @lit_advance_j

@lit_advance_j
	%jlit_new =l add %jlit, %run_j
	%jlit =l copy %jlit_new
	jmp @lit_scan_loop

@lit_scan_done
	%Llit =l sub %jlit, %pos
	%need_l1 =l add %outidx, 1
	%need_l2 =l add %need_l1, %Llit
	%fits_l =w cslel %need_l2, %cap
	jnz %fits_l, @do_literal, @ret_neg1

@do_literal
	%hdr_l =l sub %Llit, 1
	%outp_hl =l add %out, %outidx
	storeb %hdr_l, %outp_hl
	%outidx_l =l add %outidx, 1
	%k =l copy 0
	jmp @copy_loop

@copy_loop
	%k_lt_l =w csltl %k, %Llit
	jnz %k_lt_l, @copy_byte, @copy_done

@copy_byte
	%srcp_ck =l add %src, %pos
	%srcp_ckk =l add %srcp_ck, %k
	%bcp =w loadub %srcp_ckk
	%outp_ck =l add %out, %outidx_l
	%outp_ckk =l add %outp_ck, %k
	storeb %bcp, %outp_ckk
	%k1 =l add %k, 1
	%k =l copy %k1
	jmp @copy_loop

@copy_done
	%outidx_ln =l add %outidx_l, %Llit
	%pos_ln =l copy %jlit
	%outidx =l copy %outidx_ln
	%pos =l copy %pos_ln
	jmp @loop

@end_ok
	ret %outidx

@ret_neg1
	%neg1 =l sub 0, 1
	ret %neg1
}
