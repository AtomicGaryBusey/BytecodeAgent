  add x5, x0, x1
  mov x6, x2
  add x7, x2, x3

.Lloop_header:

  ldrb w9, [x4], #1
  sxtb x10, w9
  cmp x10, #-128
  b.eq .Lloop_header
  cmp x10, #0
  b.lt .Lrepeat
  b .Lliteral

.Lrepeat:
  mov x11, #-1
  sub x11, x11, x10
  add x11, x11, #1
  cmp x4, x5
  b.ge .Lerr_src
  ldrb w12, [x4]
  add x13, x6, x11
  cmp x13, x7

  mov x14, x11
.Lrepeat_full:

  strb w12, [x6], #1
  sub x14, x14, #1
  b .Lrepeat_full

.Lrepeat_partial:

  cbz x14, .Lrepeat_cap_zero
  add x4, x4, #1
.Lrepeat_partial_loop:

  sub x14, x14, #1
  cbnz x14, .Lrepeat_partial_loop

.Lrepeat_cap_zero:

  b .Lerr_cap

.Lliteral:

  add x13, x4, x11
  cmp x13, x5
  b.gt .Lerr_src
  add x13, x6, x11
  cmp x13, x7
  b.gt .Lliteral_partial
  mov x14, x11
.Lliteral_full:
  cbz x14, .Lloop_header
  ldrb w12, [x4], #1
  strb w12, [x6], #1
  sub x14, x14, #1
  b .Lliteral_full

.Lliteral_partial:

  cbz x14, .Lliteral_cap_zero
.Lliteral_partial_loop:

  strb w12, [x6], #1
  sub x14, x14, #1
  cbnz x14, .Lliteral_partial_loop
  add x4, x4, x11
  b .Lerr_cap

.Lliteral_cap_zero:
  add x4, x4, x11
  b .Lerr_cap

.Lend_ok:

  ret

.Lerr_src:

.Lerr_cap:
