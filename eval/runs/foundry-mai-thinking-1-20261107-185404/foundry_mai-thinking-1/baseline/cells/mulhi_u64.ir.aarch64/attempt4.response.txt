export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 = and %a, 0xffffffff
	%a1 = shr %a, 32
	%b0 = and %b, 0xffffffff
	%b1 = shr %b, 32
	%p00 = mul %a0, %b0
	%p01 = mul %a0, %b1
	%p10 = mul %a1, %b0
	%p11 = mul %a1, %b1
	%m = add %p01, %p10
	%c1w = cultl %m, %p01
	%c1 = extuw %c1w, l
	%lmid = shl %m, 32
	%hmid = shr %m, 32
	%low = add %p00, %lmid
	%c2w = cultl %low, %p00
	%c2 = extuw %c2w, l
	%h1 = add %p11, %hmid
	%h2 = add %h1, %c1
	%hres = add %h2, %c2
	ret %hres
}
