bubble_sort:
  cmp rsi, 0
  jle .Lend_zero
  mov rcx, rsi
  dec rcx
.Louter:
  cmp rcx, 0
  jle .Lend_zero
  mov rdx, 0
.Linner:
  cmp rdx, rcx
  jge .Lnext_outer
  mov rax, qword ptr [rdi + rdx*8]
  mov r8, qword ptr [rdi + rdx*8 + 8]
  cmp rax, r8
  jle .Lno_swap
  mov qword ptr [rdi + rdx*8], r8
  mov qword ptr [rdi + rdx*8 + 8], rax
.Lno_swap:
  inc rdx
  jmp .Linner
.Lnext_outer:
  dec rcx
  jmp .Louter
.Lend_zero:
  mov rax, 0
  ret
