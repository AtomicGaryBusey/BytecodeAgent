memmove_i64:
xor rax, rax
cmp rdx, 0
je .Ldone
mov rcx, rsi
add rcx, rdx
cmp rdi, rcx
jae .Lforward
mov rcx, rdi
add rcx, rdx
cmp rdi, rsi
jbe .Lforward
lea rcx, [rdx - 1]
.Lbackward:
cmp rcx, 0
jl .Ldone
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
dec rcx
jmp .Lbackward
.Lforward:
xor rcx, rcx
.Lfloop:
cmp rcx, rdx
jge .Ldone
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
jmp .Lfloop
.Ldone:
ret
