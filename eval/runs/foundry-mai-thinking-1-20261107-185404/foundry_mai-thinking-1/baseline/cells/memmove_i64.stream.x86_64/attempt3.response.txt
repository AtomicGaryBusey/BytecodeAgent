memmove_i64:
xor rax, rax
cmp rdx, 0
jle .Ldone
cmp rdi, rsi
jle .Lforward
lea r8, [rsi + rdx*8]
cmp rdi, r8
jl .Lbackward
.Lforward:
xor rcx, rcx
.Lfloop:
cmp rcx, rdx
jge .Ldone
mov r9, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r9
inc rcx
jmp .Lfloop
.Lbackward:
lea rcx, [rdx - 1]
.Lbloop:
cmp rcx, 0
jl .Ldone
mov r9, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r9
dec rcx
jmp .Lbloop
.Ldone:
ret
