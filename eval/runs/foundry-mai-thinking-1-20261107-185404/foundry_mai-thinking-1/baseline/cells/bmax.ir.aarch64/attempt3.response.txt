export function l $bmax(l %a, l %n) {
@start
	%max0 = loadl %a
	%i0 = 1l
	jmp @loop
@loop
	%max = phi @start %max0, @body_next %max_new
	%i = phi @start %i0, @body_next %i_new
	%cond = csltl %i, %n
	jnz %cond, @body, @done
@body
	%off = mul %i, 8l
	%ptr = add %a, %off
	%val = loadl %ptr
	%gt = csgtl %val, %max
	jnz %gt, @set, @noset
@set
	jmp @body_next
@noset
	jmp @body_next
@body_next
	%max_new = phi @set %val, @noset %max
	%i_new = add %i, 1l
	jmp @loop
@done
	ret %max
}
