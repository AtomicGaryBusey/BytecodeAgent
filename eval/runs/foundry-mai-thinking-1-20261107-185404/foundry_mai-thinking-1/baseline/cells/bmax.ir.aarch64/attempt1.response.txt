export function l $bmax(l %a, l %n) {
@start
	%max = loadl %a
	%i = 1l
@loop
	%cond = csltl %i, %n
	jnz %cond, @body, @done
@body
	%off = mul %i, 8l
	%ptr = add %a, %off
	%val = loadl %ptr
	%gt = csgtl %val, %max
	jnz %gt, @set, @inc
@set
	%max = %val
@inc
	%i = add %i, 1l
	jmp @loop
@done
	ret %max
}
