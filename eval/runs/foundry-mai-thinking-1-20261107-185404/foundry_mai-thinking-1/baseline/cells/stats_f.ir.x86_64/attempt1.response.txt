export function l $stats_f(l %a, l %n, l %fptr, l %out) {
@start
	%one =l copy 1
	%two =l copy 2
	%eight =l copy 8
	%sixteen =l copy 16
	%idx =l copy 0
	%nw =l copy %n
	%iszero =w ceql %nw, 0
	jnz %iszero, @retzero
	# load first element
	%a0p =l shl %idx, 3
	%a0p2 =l add %a, %a0p
	%v0 =l loadl %a0p2
	# call f on first
	%y0 =l call %fptr(l %v0)
	# initialize min, max, sum, count
	%minv =l copy %y0
	%maxv =l copy %y0
	%sumv =l copy %y0
	# count odd for y0
	%y0and =l and %y0, %one
	%c0odd =w ceql %y0and, %one
	%cntv =l extw %c0odd
	%idxn =l add %idx, %one
	jmp @loop
@loop
	# check if idxn < nw
	%cmpn =w csltl %idxn, %nw
	jnz %cmpn, @body, @done
@body
	# load a[idxn]
	%offb =l shl %idxn, 3
	%apb =l add %a, %offb
	%vab =l loadl %apb
	# call f
	%yab =l call %fptr(l %vab)
	# min
	%cmin =w csltl %yab, %minv
	jnz %cmin, @setmin, @afmin
@setmin
	%minvnew =l copy %yab
	jmp @mincont
@afmin
	%minvnew =l copy %minv
@mincont
	# max
	%cmax =w csgtl %yab, %maxv
	jnz %cmax, @setmax, @afmax
@setmax
	%maxvnew =l copy %yab
	jmp @maxcont
@afmax
	%maxvnew =l copy %maxv
@maxcont
	# sum
	%sumvnew =l add %sumv, %yab
	# odd
	%yaband =l and %yab, %one
	%codd =w ceql %yaband, %one
	jnz %codd, @incodd, @noodd
@incodd
	%cntvnew =l add %cntv, %one
	jmp @oddcont
@noodd
	%cntvnew =l copy %cntv
@oddcont
	# update
	%minv =l copy %minvnew
	%maxv =l copy %maxvnew
	%sumv =l copy %sumvnew
	%cntv =l copy %cntvnew
	%idxn2 =l add %idxn, %one
	%idxn =l copy %idxn2
	jmp @loop
@done
	# store out[0], out[1], out[2]
	%o0p =l add %out, 0
	storel %minv, %o0p
	%o1p =l add %out, %eight
	storel %maxv, %o1p
	%o2p =l add %out, %sixteen
	storel %sumv, %o2p
	# return cntv
	ret %cntv
@retzero
	ret 0
}
