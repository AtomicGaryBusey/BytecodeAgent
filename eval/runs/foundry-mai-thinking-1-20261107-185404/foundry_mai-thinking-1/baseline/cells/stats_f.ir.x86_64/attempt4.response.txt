export function l $stats_f(l %a, l %n, l %fptr, l %out) {
@start
	%one =l copy 1
	%eight =l copy 8
	%sixteen =l copy 16
	%iszero =w ceql %n, 0
	jnz %iszero, @retzero, @first
@first
	%v0 =l loadl %a
	%y0 =l call %fptr(l %v0)
	%minv =l copy %y0
	%maxv =l copy %y0
	%sumv =l copy %y0
	%y0and =l and %y0, %one
	%c0odd =w ceql %y0and, %one
	%cntv =l extul %c0odd
	%idx =l copy 1
	jmp @loop
@loop
	%cmpidx =w csltl %idx, %n
	jnz %cmpidx, @body, @done
@body
	%off =l shl %idx, 3
	%aptr =l add %a, %off
	%va =l loadl %aptr
	%ya =l call %fptr(l %va)
	%cmin =w csltl %ya, %minv
	jnz %cmin, @setmin, @mincont
@setmin
	%minv =l copy %ya
@mincont
	%cmax =w csgtl %ya, %maxv
	jnz %cmax, @setmax, @maxcont
@setmax
	%maxv =l copy %ya
@maxcont
	%sumv =l add %sumv, %ya
	%yaand =l and %ya, %one
	%codd =w ceql %yaand, %one
	jnz %codd, @incodd, @oddcont
@incodd
	%cntv =l add %cntv, %one
@oddcont
	%idx =l add %idx, %one
	jmp @loop
@done
	storel %minv, %out
	%o1p =l add %out, %eight
	storel %maxv, %o1p
	%o2p =l add %out, %sixteen
	storel %sumv, %o2p
	ret %cntv
@retzero
	ret 0
}
