export function l $crc32(l %p, l %n) {
@entry
	%crc =w copy 0xffffffff
	%i =l copy 0
@loop_bytes
	%cmp_bytes =w csltl %i, %n
	jnz %cmp_bytes, @body_bytes, @done
@body_bytes
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc_xor_b =w xor %crc, %b
	%crc =w copy %crc_xor_b
	%bit =l copy 0
@loop_bits
	%cmp_bits =w csltl %bit, 8
	jnz %cmp_bits, @body_bits, @next_byte
@body_bits
	%lowbit =w and %crc, 1
	%crc_shr =w shr %crc, 1
	%is_one =w ceqw %lowbit, 1
	jnz %is_one, @xor_poly, @no_xor
@xor_poly
	%crc_poly =w xor %crc_shr, 0xedb88320
	%crc =w copy %crc_poly
	jmp @inc_bit
@no_xor
	%crc =w copy %crc_shr
	jmp @inc_bit
@inc_bit
	%bit_new =l add %bit, 1
	%bit =l copy %bit_new
	jmp @loop_bits
@next_byte
	%i_new =l add %i, 1
	%i =l copy %i_new
	jmp @loop_bytes
@done
	%crc_final_xor =w xor %crc, 0xffffffff
	%res =l extuw %crc_final_xor
	ret %res
}
