export function l $count_ugt(l %a, l %n, l %t) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%iszero =w cslel %n, 0
	jnz %iszero, @end
@loop
	%lt =w csltl %i, %n
	jnz %lt, @body, @end
@body
	%idx =l mul %i, 8
	%p =l add %a, %idx
	%v =l loadl %p
	%gtu =w cugtl %v, %t
	jnz %gtu, @inc, @next
@inc
	%cnt =l add %cnt, 1
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
