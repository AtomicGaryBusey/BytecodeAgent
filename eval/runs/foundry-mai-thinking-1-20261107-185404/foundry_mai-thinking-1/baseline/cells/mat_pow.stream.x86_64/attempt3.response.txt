mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15

mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbp, r8

mov rax, r13
imul rax, r13
mov rbx, rax

cmp r14, 0
je .Lk_zero

mov rdi, rbp
mov rsi, r12
mov rcx, rbx
xor r8, r8
.Lcopy_base:
cmp r8, rcx
je .Lbase_done
mov rax, qword ptr [rsi + r8*8]
mov qword ptr [rdi + r8*8], rax
inc r8
jmp .Lcopy_base
.Lbase_done:

mov rdi, r15
xor rcx, rcx
.Lidentity_loop:
cmp rcx, rbx
je .Lidentity_done
mov rax, rcx
xor rdx, rdx
div r13
cmp rdx, rax
jne .Lnot_diag
mov qword ptr [rdi + rcx*8], 1
jmp .Lid_next
.Lnot_diag:
mov qword ptr [rdi + rcx*8], 0
.Lid_next:
inc rcx
jmp .Lidentity_loop
.Lidentity_done:

mov r11, 63
.Lpower_loop:
cmp r11, 0
jl .Lclear_and_ret
mov rax, r14
mov rcx, r11
shr rax, cl
and rax, 1
test rax, rax
jz .Lno_bit

lea rdi, [rbp + rbx*8]
mov rsi, r15
mov rdx, rbp
mov rcx, r13
call .Lmat_mul_inline

mov r8, 0
.Lcopy_res_to_r:
cmp r8, rbx
je .Lno_bit
mov rax, qword ptr [rbp + rbx*8 + r8*8]
