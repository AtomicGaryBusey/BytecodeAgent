mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15

mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbp, r8

mov rax, r13
imul rax, r13
mov rbx, rax

cmp r14, 0
jne .Lk_not_zero

xor r10, r10
.Lid_fill_out:
cmp r10, rbx
je .Ltmp_zero_k0
mov rax, r10
xor rdx, rdx
mov rsi, r13
div rsi
cmp rdx, rax
jne .Lid_out_not_diag
mov qword ptr [r15 + r10*8], 1
jmp .Lid_out_next
.Lid_out_not_diag:
mov qword ptr [r15 + r10*8], 0
.Lid_out_next:
inc r10
jmp .Lid_fill_out

.Ltmp_zero_k0:
xor r10, r10
mov rax, rbx
add rax, rax
.Ltmp_zero_loop_k0:
cmp r10, rax
je .Lret_zero
mov qword ptr [rbp + r10*8], 0
inc r10
jmp .Ltmp_zero_loop_k0

.Lk_not_zero:
xor r10, r10
.Lcopy_base:
cmp r10, rbx
je .Lbase_copied
mov rax, qword ptr [r12 + r10*8]
mov qword ptr [rbp + r10*8], rax
inc r10
jmp .Lcopy_base
.Lbase_copied:

xor r10, r10
.Lid_fill_res:
cmp r10, rbx
je .Lres_ready
mov rax, r10
xor rdx, rdx
mov rsi, r13
div rsi
cmp rdx, rax
jne .Lid_res_not_diag
mov qword ptr [r15 + r10*8], 1
jmp .Lid_res_next
.Lid_res_not_diag:
mov qword ptr [r15 + r10*8], 0
.Lid_res_next:
inc r10
jmp .Lid_res_fill

.Lres_ready:
mov r11, 63
.Lpower_loop:
cmp r11, 0
jl .Lclear_tmp_and_ret
mov rax, r14
mov rcx, r11
shr rax, cl
and rax, 1
test rax, rax
jz .Lno_mul_res

lea rdi, [rbp + rbx*8]
mov rsi, r15
mov rdx, rbp
mov rcx, r13
call .Lmat_mul
mov r10, 0
.Lcopy_prod_to_res:
cmp r10, rbx
je .Lno_mul_res
mov rax, qword ptr [rbp + rbx*8 + r10*8]
