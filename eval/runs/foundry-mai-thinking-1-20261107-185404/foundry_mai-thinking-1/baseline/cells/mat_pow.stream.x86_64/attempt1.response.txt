mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15

mov r12, rdi
mov r13, rsi
mov r14, rdx
mov rcx, rcx
mov r15, r8
mov rbp, rcx

mov rax, r13
imul rax, r13
mov rbx, rax

cmp r14, 0
je .Lk_zero

mov rdi, r15
mov rsi, r12
mov rcx, rbx
.Lcopy_base:
test rcx, rcx
jz .Lbase_copied
dec rcx
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
jmp .Lcopy_base
.Lbase_copied:

mov rdi, rbp
xor rcx, rcx
.Linit_res:
cmp rcx, rbx
je .Lres_inited
mov rax, rcx
xor rdx, rdx
div r13
cmp rdx, rax
jne .Lnot_diag
mov qword ptr [rdi + rcx*8], 1
jmp .Lnext_res
.Lnot_diag:
mov qword ptr [rdi + rcx*8], 0
.Lnext_res:
inc rcx
jmp .Linit_res
.Lres_inited:

mov r11, 63
.Lpower_loop:
cmp r11, 0
jl .Lpower_done
mov rax, r14
mov rcx, r11
shr rax, cl
and rax, 1
test rax, rax
jz .Lno_mult_res

lea rdi, [r15 + rbx*8]
mov rsi, rbp
mov r9, r15
call .Lmat_mul

mov rcx, rbx
.Lcopy_res_back:
test rcx, rcx
jz .Lno_mult_res
dec rcx
mov rax, qword ptr [rdi + rcx*8]
mov qword ptr [rsi + rcx*8], rax
jmp .Lcopy_res_back

.Lno_mult_res:
lea rdi, [r15 + rbx*8]
mov rsi, r15
mov r9, r15
call .Lmat_mul

mov rcx, rbx
.Lcopy_base_back:
test rcx, rcx
jz .Lnext_bit
dec rcx
mov rax, qword ptr [rdi + rcx*8]
mov qword ptr [rsi + rcx*8], rax
jmp .Lcopy_base_back

.Lnext_bit:
dec r11
jmp .Lpower_loop

.Lpower_done:
jmp .Lclear_tmp

.Lk_zero:
mov rdi, rbp
xor rcx, rcx
.Lid_loop:
cmp rcx, rbx
je .Lclear_tmp
mov rax, rcx
xor rdx, rdx
div r13
cmp rdx, rax
jne .Lid_not_diag
mov qword ptr [rdi + rcx*8], 1
jmp .Lid_next
.Lid_not_diag:
mov qword ptr [rdi + rcx*8], 0
.Lid_next:
inc rcx
jmp .Lid_loop

.Lclear_tmp:
mov rdi, r15
mov rcx, rbx
shl rcx, 1
.Lzero_loop:
test rcx, rcx
jz .Lend
dec rcx
mov qword ptr [rdi + rcx*8], 0
jmp .Lzero_loop
.Lend:
xor rax, rax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret

.Lmat_mul:
push rbx
push rbp
push r12
push r13
push r14
push r15

mov r10, rdi
mov r11, rsi
mov r12, r9
mov r13, r13

xor r8, r8
.Li_loop:
cmp r8, r13
je .Lmul_done
xor r9, r9
.Lj_loop:
cmp r9, r13
je .Lnext_i
xor r14, r14
mov qword ptr [r10 + r8*r13 + r9*8], 0
.L
