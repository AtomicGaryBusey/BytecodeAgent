export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%msize = mul %n, %n
	%kzero = ceql %k, 0
	jnz %kzero, @identity_case, @nonzero_case

@identity_case
	%outi = 0
@id_out_loop_i
	%cidi = csltl %outi, %n
	jnz %cidi, @id_out_loop_j_init, @zero_tmp_id
@id_out_loop_j_init
	%outj = 0
@id_out_loop_j
	%cijd = csltl %outj, %n
	jnz %cijd, @id_set_elem, @id_out_inc_i
@id_set_elem
	%idxid = mul %outi, %n
	%idxid = add %idxid, %outj
	%isdiag = ceql %outi, %outj
	jnz %isdiag, @id_one, @id_zero
@id_one
	storel 1, %out, %idxid
	jmp @id_out_inc_j
@id_zero
	storel 0, %out, %idxid
	jmp @id_out_inc_j
@id_out_inc_j
	%outj = add %outj, 1
	jmp @id_out_loop_j
@id_out_inc_i
	%outi = add %outi, 1
	jmp @id_out_loop_i

@zero_tmp_id
	%tmpcountid = mul %msize, 2
	%tiid = 0
@zt_loop_id
	%ctiid = csltl %tiid, %tmpcountid
	jnz %ctiid, @zt_store_id, @ret_zero
@zt_store_id
	storel 0, %tmp, %tiid
	%tiid = add %tiid, 1
	jmp @zt_loop_id

@ret_zero
	ret 0

@nonzero_case
	%tmpb = %tmp
	%tmpp = add %tmp, %msize

@copy_a_to_b
	%cia = 0
@ca_loop
	%ccia = csltl %cia, %msize
	jnz %ccia, @ca_load, @init_r_identity
@ca_load
	%vaa = loadl %a, %cia
	storel %vaa, %tmpb, %cia
	%cia = add %cia, 1
	jmp @ca_loop

@init_r_identity
	%rir = 0
@ir_loop
	%crir = csltl %rir, %msize
	jnz %crir, @ir_set, @find_highbit
@ir_set
	%rrow = div %rir, %n
	%rcol = rem %rir, %n
	%rdiag = ceql %rrow, %rcol
	jnz %rdiag, @ir_one_r, @ir_zero_r
@ir_one_r
	storel 1, %out, %rir
	jmp @ir_inc
@ir_zero_r
	storel 0, %out, %rir
	jmp @ir_inc
@ir_inc
	%rir = add %rir, 1
	jmp @ir_loop

@find_highbit
	%kkhb = %k
	%hbit = 63
@fh_loop
	%chbit = csltl 0, %hbit
	jnz %chbit, @fh_check, @pow_loop_init
@fh_check
	%shhb = shll %kkhb, %hbit
	%bset = csltl 0, %shhb
	jnz %bset, @pow_loop_init, @fh_dec
@fh_dec
	%hbit = sub %hbit, 1
	jmp @fh_loop

@pow_loop_init
	%bcur = %hbit
@pow_loop
	%cbcur = csltl 0, %bcur
	jnz %cbcur, @pow_step, @final_zero_b
@pow_step
	%shkb = shll %k, %bcur
	%msbb = csltl 0, %shkb
	jnz %msbb, @case_r_mul_b, @case_b_mul_b

@case_r_mul_b
	%mir1 = 0
@r1_mm_i
	%cmir1 = csltl %mir1, %n
	jnz %cmir1, @r1_mm_j_init, @r1_done_mul
@r1_mm_j_init
	%mjr1 = 0
@r1_mm_j
	%cmjr1 = csltl %mjr1, %n
	jnz %cmjr1, @r1_mm_sum_init, @r1_mm_i_inc
@r1_mm_sum_init
	%msumr1 = 0
	%mtr1 = 0
@r1_mm_t
	%cmtr1 = csltl %mtr1, %n
	jnz %cmtr1, @r1_mm_term, @r1_mm_store
@r1_mm_term
	%mxr1 = mul %mir1, %n
	%mxr1 = add %mxr1, %mtr1
	%vxr1 = loadl %out, %mxr1
	%myr1 = mul %mtr1, %n
	%myr1 = add %myr1, %mjr1
	%vyr1 = loadl %tmpb, %myr1
	%vtermr1 = mull %vxr1, %vyr1
	%msumr1 = add %msumr1, %vtermr1
	%mtr1 = add %mtr1, 1
	jmp @r1_mm_t
@r1_mm_store
	%mpr1 = mul %mir1, %n
	%mpr1 = add %mpr1, %mjr1
	storel %msumr1, %tmpp, %mpr1
	%mjr1 = add %mjr1, 1
	jmp @r1_mm_j
@r1_mm_i_inc
	%mir1 = add %mir1, 1
	jmp @r1_mm_i
@r1_done_mul
	%mcr1 = 0
@r1_cp_loop
	%cmcr1 = csltl %mcr1, %msize
	jnz %cmcr1, @r1_cp_load, @r2_mul_b
@r1_cp_load
	%vsr1 = loadl %tmpp, %mcr1
	storel %vsr1, %out, %mcr1
	%mcr1 = add %mcr1, 1
	jmp @r1_cp_loop

@r2_mul_b
	%mir2 = 0
@r2_mm_i
	%cmir2 = csltl %mir2, %n
	jnz %cmir2, @r2_mm_j_init, @r2_done_mul
@r2_mm_j_init
	%mjr2 = 0
@r2_mm_j
	%cmjr2 = csltl %mjr2, %n
	jnz %cmjr2, @r2_mm_sum_init, @r2_mm_i_inc
@r2_mm_sum_init
	%msumr2 = 0
	%mtr2 = 0
@r2_mm_t
	%cmtr2 = csltl %mtr2, %n
	jnz %cmtr2, @r2_mm_term, @r2_mm_store
@r2_mm_term
	%mxr2 = mul %mir2, %n
	%mxr2 = add %mxr2, %mtr2
	%vxr2 = loadl %tmpb, %mxr2
	%myr2 = mul %mtr2, %n
	%myr2 = add %myr2, %mjr2
	%vyr2 = loadl %tmpb, %myr2
	%vtermr2 = mull %vxr2, %vyr2
	%msumr2 = add %msumr2, %vtermr2
	%mtr2 = add %mtr2, 1
	jmp @r2_mm_t
@r2_mm_store
	%mpr2 = mul %mir2, %n
	%mpr2 = add %mpr2, %mjr2
	storel %msumr2, %tmpp, %mpr2
	%mjr2 = add %mjr2, 1
	jmp @r2_mm_j
@r2_mm_i_inc
	%mir2 = add %mir2, 1
	jmp @r2_mm_i
@r2_done_mul
	%mcr2 = 0
@r2_cp_loop
	%cmcr2 = csltl %mcr2, %msize
	jnz %cmcr2, @r2_cp_load, @pow_dec
@r2_cp_load
	%vsr2 = loadl %tmpp, %mcr2
	storel %vsr2, %tmpb, %mcr2
	%mcr2 = add %mcr2, 1
	jmp @r2_cp_loop

@case_b_mul_b
	%mib1 = 0
@b1_mm_i
	%cmib1 = csltl %mib1, %n
	jnz %cmib1, @b1_mm_j_init, @b1_done_mul
@b1_mm_j_init
	%mjb1 = 0
@b1_mm_j
	%cmjb1 = csltl %mjb1, %n
	jnz %cmjb1, @b1_mm_sum_init, @b1_mm_i_inc
@b1_mm_sum_init
	%msumb1 = 0
	%mtb1 = 0
@b1_mm_t
	%cmtb1 = csltl %mtb1, %n
	jnz %cmtb1, @b1_mm_term, @b1_mm_store
@b1_mm_term
	%mxb1 = mul %mib1, %n
	%mxb1 = add %mxb1, %mtb1
	%vxb1 = loadl %tmpb, %mxb1
	%myb1 = mul %mtb1, %n
	%myb1 = add %myb1, %mjb1
	%vyb1 = loadl %tmpb, %myb1
	%vtermb1 = mull %vxb1, %vyb1
	%msumb1 = add %msumb1, %vtermb1
	%mtb1 = add %mtb1, 1
	jmp @b1_mm_t
@b1_mm_store
	%mpb1 = mul %mib1, %n
	%mpb1 = add %mpb1, %mjb1
	storel %msumb1, %tmpp, %mpb1
	%mjb1 = add %mjb1, 1
	jmp @b1_mm_j
@b1_mm_i_inc
	%mib1 = add %mib1, 1
	jmp @b1_mm_i
@b1_done_mul
	%mcb1 = 0
@b1_cp_loop
	%cmcb1 = csltl %mcb1, %msize
	jnz %cmcb1, @b1_cp_load, @pow_dec
@b1_cp_load
	%vsb1 = loadl %tmpp, %mcb1
	storel %vsb1, %tmpb, %mcb1
	%mcb1 = add %mcb1, 1
	jmp @b1_cp_loop

@pow_dec
	%bcur = sub %bcur, 1
	jmp @pow_loop

@final_zero_b
	%zib = 0
@zb_loop
	%czib = csltl %zib, %msize
	jnz %czib, @zb_store, @final_zero_p
@zb_store
	storel 0, %tmpb, %zib
	%zib = add %zib, 1
	jmp @zb_loop

@final_zero_p
	%zjp = 0
@zp_loop
	%czjp = csltl %zjp, %msize
	jnz %czjp, @zp_store, @end_ret
@zp_store
	storel 0, %tmpp, %zjp
	%zjp = add %zjp, 1
	jmp @zp_loop

@end_ret
	ret 0
}
