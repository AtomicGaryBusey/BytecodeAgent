export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%msize = mul %n, %n
	%kzero = ceql %k, 0
	jnz %kzero, @identity_case, @nonzero_case

@identity_case
	%outi = 0
@id_out_loop_i
	%cidi = csltl %outi, %n
	jnz %cidi, @id_out_loop_j_init, @zero_tmp
@id_out_loop_j_init
	%outj = 0
@id_out_loop_j
	%cijd = csltl %outj, %n
	jnz %cijd, @id_set_elem, @id_out_inc_i
@id_set_elem
	%idx = mul %outi, %n
	%idx = add %idx, %outj
	%isdiag = ceql %outi, %outj
	jnz %isdiag, @id_one, @id_zero
@id_one
	storel 1, %out, %idx
	jmp @id_out_inc_j
@id_zero
	storel 0, %out, %idx
	jmp @id_out_inc_j
@id_out_inc_j
	%outj = add %outj, 1
	jmp @id_out_loop_j
@id_out_inc_i
	%outi = add %outi, 1
	jmp @id_out_loop_i

@zero_tmp
	%tmpcount = mul %msize, 2
	%ti = 0
@zt_loop
	%cti = csltl %ti, %tmpcount
	jnz %cti, @zt_store, @ret_zero
@zt_store
	storel 0, %tmp, %ti
	%ti = add %ti, 1
	jmp @zt_loop

@ret_zero
	ret 0

@nonzero_case
	%tmpb = %tmp
	%tmpp = add %tmp, %msize

@copy_a_to_b
	%ci = 0
@ca_loop
	%cci = csltl %ci, %msize
	jnz %cci, @ca_load, @init_r_identity
@ca_load
	%va = loadl %a, %ci
	storel %va, %tmpb, %ci
	%ci = add %ci, 1
	jmp @ca_loop

@init_r_identity
	%ri = 0
@ir_loop
	%cri = csltl %ri, %msize
	jnz %cri, @ir_set, @find_highbit
@ir_set
	%rrow = div %ri, %n
	%rcol = rem %ri, %n
	%rdiag = ceql %rrow, %rcol
	jnz %rdiag, @ir_one, @ir_zero
@ir_one
	storel 1, %out, %ri
	jmp @ir_inc
@ir_zero
	storel 0, %out, %ri
	jmp @ir_inc
@ir_inc
	%ri = add %ri, 1
	jmp @ir_loop

@find_highbit
	%kk = %k
	%hb = 63
@fh_loop
	%chb = csltl 0, %hb
	jnz %chb, @fh_check, @pow_loop_init
@fh_check
	%sh = shll %kk, %hb
	%bitset = csltl 0, %sh
	jnz %bitset, @pow_loop_init, @fh_dec
@fh_dec
	%hb = sub %hb, 1
	jmp @fh_loop

@pow_loop_init
	%bit = %hb
@pow_loop
	%cbit = csltl 0, %bit
	jnz %cbit, @pow_step, @final_zero_b

@pow_step
	%shk = shll %k, %bit
	%msb = csltl 0, %shk
	jnz %msb, @r_mul_b, @b_mul_b

@r_mul_b
	call @mat_mul(%out, %tmpb, %tmpp, %n)
	call @copy_mat(%tmpp, %out, %n)
	call @mat_mul(%tmpb, %tmpb, %tmpp, %n)
	call @copy_mat(%tmpp, %tmpb, %n)
	jmp @pow_dec

@b_mul_b
	call @mat_mul(%tmpb, %tmpb, %tmpp, %n)
	call @copy_mat(%tmpp, %tmpb, %n)
	jmp @pow_dec

@pow_dec
	%bit = sub %bit, 1
	jmp @pow_loop

@final_zero_b
	%zi = 0
@zb_loop
	%czi = csltl %zi, %msize
	jnz %czi, @zb_store, @final_zero_p
@zb_store
	storel 0, %tmpb, %zi
	%zi = add %zi, 1
	jmp @zb_loop

@final_zero_p
	%zj = 0
@zp_loop
	%czj = csltl %zj, %msize
	jnz %czj, @zp_store, @end_ret
@zp_store
	storel 0, %tmpp, %zj
	%zj = add %zj, 1
	jmp @zp_loop

@end_ret
	ret 0

@mat_mul
	%mx = %0
	%my = %1
	%mp = %2
	%mn = %3
	%mi = 0
@mm_i
	%cmi = csltl %mi, %mn
	jnz %cmi, @mm_j_init, @mm_ret
@mm_j_init
	%mj = 0
@mm_j
	%cmj = csltl %mj, %mn
	jnz %cmj, @mm_sum_init, @mm_i_inc
@mm_sum_init
	%msum = 0
	%mt = 0
@mm_t
	%cmt = csltl %mt, %mn
	jnz %cmt, @mm_term, @mm_store
@mm_term
	%mxidx = mul %mi, %mn
	%mxidx = add %mxidx, %mt
	%vx = loadl %mx, %mxidx
	%myidx = mul %mt, %mn
	%myidx = add %myidx, %mj
	%vy = loadl %my, %myidx
	%vterm = mull %vx, %vy
	%msum = add %msum, %vterm
	%mt = add %mt, 1
	jmp @mm_t
@mm_store
	%mpidx = mul %mi, %mn
	%mpidx = add %mpidx, %mj
	storel %msum, %mp, %mpidx
	%mj = add %mj, 1
	jmp @mm_j
@mm_i_inc
	%mi = add %mi, 1
	jmp @mm_i
@mm_ret
	ret

@copy_mat
	%msrc = %0
	%mdst = %1
	%mn2 = %2
	%m2 = mul %mn2, %mn2
	%mc = 0
@cm_loop
	%cmc = csltl %mc, %m2
	jnz %cmc, @cm_load, @cm_ret
@cm_load
	%vsrc = loadl %msrc, %mc
	storel %vsrc, %mdst, %mc
	%mc = add %mc, 1
	jmp @cm_loop
@cm_ret
	ret
}
