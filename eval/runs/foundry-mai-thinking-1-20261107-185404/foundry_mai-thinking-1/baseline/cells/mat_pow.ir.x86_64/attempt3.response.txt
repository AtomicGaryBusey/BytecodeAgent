export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%msize = mull %n, %n
	%kisz = ceql %k, 0
	jnz %kisz, @identity_case, @nonzero_case

@identity_case
	%ii = 0
@id_i_loop
	%cii = csltl %ii, %n
	jnz %cii, @id_j_init, @id_zero_tmp
@id_j_init
	%ij = 0
@id_j_loop
	%cij = csltl %ij, %n
	jnz %cij, @id_elem, @id_i_inc
@id_elem
	%ididx = mull %ii, %n
	%ididx = addl %ididx, %ij
	%isdiag = ceql %ii, %ij
	jnz %isdiag, @id_set_one, @id_set_zero
@id_set_one
	storel 1, %out, %ididx
	jmp @id_j_inc
@id_set_zero
	storel 0, %out, %ididx
	jmp @id_j_inc
@id_j_inc
	%ij = addl %ij, 1
	jmp @id_j_loop
@id_i_inc
	%ii = addl %ii, 1
	jmp @id_i_loop

@id_zero_tmp
	%tidcnt = mull %msize, 2
	%tz = 0
@id_zt_loop
	%ctz = csltl %tz, %tidcnt
	jnz %ctz, @id_zt_store, @ret_zero
@id_zt_store
	storel 0, %tmp, %tz
	%tz = addl %tz, 1
	jmp @id_zt_loop

@ret_zero
	ret 0

@nonzero_case
	%tmpb = %tmp
	%tmpp = addl %tmp, %msize

@copy_a
	%ca = 0
@ca_loop
	%cca = csltl %ca, %msize
	jnz %cca, @ca_do, @init_r
@ca_do
	%va = loadl %a, %ca
	storel %va, %tmpb, %ca
	%ca = addl %ca, 1
	jmp @ca_loop

@init_r
	%ir = 0
@ir_loop
	%cir = csltl %ir, %msize
	jnz %cir, @ir_do, @find_hb
@ir_do
	%irr = divl %ir, %n
	%irc = reml %ir, %n
	%irdiag = ceql %irr, %irc
	jnz %irdiag, @ir_one, @ir_zero
@ir_one
	storel 1, %out, %ir
	jmp @ir_inc
@ir_zero
	storel 0, %out, %ir
	jmp @ir_inc
@ir_inc
	%ir = addl %ir, 1
	jmp @ir_loop

@find_hb
	%khb = %k
	%hb = 63
@fh_loop
	%chbpos = csltl 0, %hb
	jnz %chbpos, @fh_chk, @pow_init
@fh_chk
	%shkhb = shll %khb, %hb
	%bisset = csltl 0, %shkhb
	jnz %bisset, @pow_init, @fh_dec
@fh_dec
	%hb = subl %hb, 1
	jmp @fh_loop

@pow_init
	%bcur = %hb
@pow_loop
	%cbcur = csltl 0, %bcur
	jnz %cbcur, @pow_step, @final_clear
@pow_step
	%skbit = shll %k, %bcur
	%kbitset = csltl 0, %skbit
	jnz %kbitset, @r_mul_b_then_bb, @b_mul_b

@r_mul_b_then_bb
	%mi1 = 0
@r1_i
	%cmi1 = csltl %mi1, %n
	jnz %cmi1, @r1_jinit, @r1_done
@r1_jinit
	%mj1 = 0
@r1_j
	%cmj1 = csltl %mj1, %n
	jnz %cmj1, @r1_sinit, @r1_iinc
@r1_sinit
	%sr1 = 0
	%mt1 = 0
@r1_t
	%cmt1 = csltl %mt1, %n
	jnz %cmt1, @r1_term, @r1_store
@r1_term
	%mx1 = mull %mi1, %n
	%mx1 = addl %mx1, %mt1
	%vx1 = loadl %out, %mx1
	%my1 = mull %mt1, %n
	%my1 = addl %my1, %mj1
	%vy1 = loadl %tmpb, %my1
	%vtr1 = mull %vx1, %vy1
	%sr1 = addl %sr1, %vtr1
	%mt1 = addl %mt1, 1
	jmp @r1_t
@r1_store
	%mp1 = mull %mi1, %n
	%mp1 = addl %mp1, %mj1
	storel %sr1, %tmpp, %mp1
	%mj1 = addl %mj1, 1
	jmp @r1_j
@r1_iinc
	%mi1 = addl %mi1, 1
	jmp @r1_i
@r1_done
	%mc1 = 0
@r1_c
	%cmc1 = csltl %mc1, %msize
	jnz %cmc1, @r1_cld, @b_sq
@r1_cld
	%vsc1 = loadl %tmpp, %mc1
	storel %vsc1, %out, %mc1
	%mc1 = addl %mc1, 1
	jmp @r1_c

@b_sq
	%mi2 = 0
@r2_i
	%cmi2 = csltl %mi2, %n
	jnz %cmi2, @r2_jinit, @r2_done
@r2_jinit
	%mj2 = 0
@r2_j
	%cmj2 = csltl %mj2, %n
	jnz %cmj2, @r2_sinit, @r2_iinc
@r2_sinit
	%sr2 = 0
	%mt2 = 0
@r2_t
	%cmt2 = csltl %mt2, %n
	jnz %cmt2, @r2_term, @r2_store
@r2_term
	%mx2 = mull %mi2, %n
	%mx2 = addl %mx2, %mt2
	%vx2 = loadl %tmpb, %mx2
	%my2 = mull %mt2, %n
	%my2 = addl %my2, %mj2
	%vy2 = loadl %tmpb, %my2
	%vtr2 = mull %vx2, %vy2
	%sr2 = addl %sr2, %vtr2
	%mt2 = addl %mt2, 1
	jmp @r2_t
@r2_store
	%mp2 = mull %mi2, %n
	%mp2 = addl %mp2, %mj2
	storel %sr2, %tmpp, %mp2
	%mj2 = addl %mj2, 1
	jmp @r2_j
@r2_iinc
	%mi2 = addl %mi2, 1
	jmp @r2_i
@r2_done
	%mc2 = 0
@r2_c
	%cmc2 = csltl %mc2, %msize
	jnz %cmc2, @r2_cld, @pow_dec
@r2_cld
	%vsc2 = loadl %tmpp, %mc2
	storel %vsc2, %tmpb, %mc2
	%mc2 = addl %mc2, 1
	jmp @r2_c

@b_mul_b
	%mi3 = 0
@b1_i
	%cmi3 = csltl %mi3, %n
	jnz %cmi3, @b1_jinit, @b1_done
@b1_jinit
	%mj3 = 0
@b1_j
	%cmj3 = csltl %mj3, %n
	jnz %cmj3, @b1_sinit, @b1_iinc
@b1_sinit
	%sb1 = 0
	%mt3 = 0
@b1_t
	%cmt3 = csltl %mt3, %n
	jnz %cmt3, @b1_term, @b1_store
@b1_term
	%mx3 = mull %mi3, %n
	%mx3 = addl %mx3, %mt3
	%vx3 = loadl %tmpb, %mx3
	%my3 = mull %mt3, %n
	%my3 = addl %my3, %mj3
	%vy3 = loadl %tmpb, %my3
	%vtr3 = mull %vx3, %vy3
	%sb1 = addl %sb1, %vtr3
	%mt3 = addl %mt3, 1
	jmp @b1_t
@b1_store
	%mp3 = mull %mi3, %n
	%mp3 = addl %mp3, %mj3
	storel %sb1, %tmpp, %mp3
	%mj3 = addl %mj3, 1
	jmp @b1_j
@b1_iinc
	%mi3 = addl %mi3, 1
	jmp @b1_i
@b1_done
	%mc3 = 0
@b1_c
	%cmc3 = csltl %mc3, %msize
	jnz %cmc3, @b1_cld, @pow_dec
@b1_cld
	%vsc3 = loadl %tmpp, %mc3
	storel %vsc3, %tmpb, %mc3
	%mc3 = addl %mc3, 1
	jmp @b1_c

@pow_dec
	%bcur = subl %bcur, 1
	jmp @pow_loop

@final_clear
	%zc = 0
@zc_b
	%czcb = csltl %zc, %msize
	jnz %czcb, @zc_bs, @zc_pinit
@zc_bs
	storel 0, %tmpb, %zc
	%zc = addl %zc, 1
	jmp @zc_b

@zc_pinit
	%zp = 0
@zc_p
	%czcp = csltl %zp, %msize
	jnz %czcp, @zc_ps, @end
@zc_ps
	storel 0, %tmpp, %zp
	%zp = addl %zp, 1
	jmp @zc_p

@end
	ret 0
}
