export function l $max_byte(l %p, l %n) {
@start
	%iszero =w ceql %n, $0
	jnz %iszero, @retzero
	%maxv =l copy $0
	%i =l copy $0
@loop
	%ok =w csltl %i, %n
	jnz %ok, @body
	jmp @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%better =w csgtl %b, %maxv
	jnz %better, @update
	jmp @inc
@update
	%maxv =l copy %b
@inc
	%i =l add %i, $1
	jmp @loop
@end
	ret %maxv
@retzero
	ret $0
}
