export function l $max_byte(l %p, l %n) {
@start
	%iszero =w ceql %n, $0l
	jnz %iszero, @retzero
	%maxv =l loadub %p
	%i =l copy $1l
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body
	jmp @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%better =w csgtl %b, %maxv
	jnz %better, @update
	jmp @inc
@update
	%maxv =l copy %b
@inc
	%i =l add %i, $1l
	jmp @loop
@end
	ret %maxv
@retzero
	ret $0l
}
