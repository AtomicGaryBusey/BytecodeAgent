export function l $isqrt(l %n) {
@start
	%zero = copy 0
	%one  = copy 1
	%two  = copy 2

	# Check if n == 0
	%is_zero = ceql %n, %zero
	jnz %is_zero, @return_zero, @nonzero

@return_zero
	ret %zero

@nonzero
	# Initial approximation: x0 = n
	%x0 = copy %n

	# Compute x1 = (x0 + n/x0) / 2
	%div0 = udiv %n, %x0
	%sum0 = add %x0, %div0
	%x1   = udiv %sum0, %two

	# Newton iteration: while x1 < x0
	jmp @loop_check

@loop_check
	%go_down = cultl %x1, %x0
	jnz %go_down, @loop_body, @adjust_down

@loop_body
	# x0 = x1; x1 = (x0 + n/x0) / 2
	%x0_new = copy %x1
	%div1   = udiv %n, %x1
	%sum1   = add %x1, %div1
	%x1_new = udiv %sum1, %two
	%x0     = copy %x0_new
	%x1     = copy %x1_new
	jmp @loop_check

@adjust_down
	# Adjust down if x0*x0 > n: while (n/x0 < x0)
	jmp @adj1_check

@adj1_check
	%nx0  = udiv %n, %x0
	%bad1 = cultl %nx0, %x0
	jnz %bad1, @adj1_dec, @adjust_up

@adj1_dec
	%x0_dec = sub %x0, %one
	%x0     = copy %x0_dec
	jmp @adj1_check

@adjust_up
	# Adjust up if (x0+1)*(x0+1) <= n: while ((x0+1) <= n/(x0+1))
	jmp @adj2_check

@adj2_check
	%x0p1   = add %x0, %one
	%nx0p1  = udiv %n, %x0p1
	# Check if x0p1 <= n/x0p1
	%good2  = culel %x0p1, %nx0p1
	jnz %good2, @adj2_inc, @return_result

@adj2_inc
	%x0_inc = add %x0, %one
	%x0     = copy %x0_inc
	jmp @adj2_check

@return_result
	ret %x0
}
