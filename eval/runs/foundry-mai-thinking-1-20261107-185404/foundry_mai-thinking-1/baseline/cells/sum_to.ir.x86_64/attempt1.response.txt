export function l $sum_to(l %n) {
@start
	%le0 = cslel %n, 0
	jnz %le0, @ret_zero, @loop_init
@loop_init
	%sum = copy 0
	%i = copy 1
	jmp @loop_cond
@loop_cond
	%cond = cslel %i, %n
	jnz %cond, @loop_body, @ret_sum
@loop_body
	%sum_new = addl %sum, %i
	%sum = copy %sum_new
	%i_new = addl %i, 1
	%i = copy %i_new
	jmp @loop_cond
@ret_sum
	ret %sum
@ret_zero
	ret 0
}
