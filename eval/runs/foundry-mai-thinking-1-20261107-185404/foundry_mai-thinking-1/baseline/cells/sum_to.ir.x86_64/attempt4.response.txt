export function l $sum_to(l %n) {
@start
	w %le0 = cslel %n, 0
	jnz %le0, @ret_zero, @loop_init
@loop_init
	l %sum = 0
	l %i = 1
	jmp @loop_cond
@loop_cond
	w %cond = cslel %i, %n
	jnz %cond, @loop_body, @ret_sum
@loop_body
	%sum = addl %sum, %i
	%i = addl %i, 1
	jmp @loop_cond
@ret_sum
	ret %sum
@ret_zero
	ret 0
}
