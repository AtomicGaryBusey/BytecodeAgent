export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%iszero = ceql %n, $0
	jnz %iszero, @return_zero, @check_same
@check_same
	%issame = ceql %dst, %src
	jnz %issame, @return_zero, @compute
@compute
	%elsize = $8
	%total = mul %n, %elsize
	%src_end = add %src, %total
	%src_lt_dst = csltl %src, %dst
	%dst_lt_srcend = csltl %dst, %src_end
	%need_backward = and %src_lt_dst, %dst_lt_srcend
	jnz %need_backward, @backward, @forward
@forward
	%i = $0
@forward_loop
	%less = csltl %i, %n
	jnz %less, @forward_body, @return_zero
@forward_body
	%off = mul %i, %elsize
	%psrc = add %src, %off
	%pdst = add %dst, %off
	%val = loadl %psrc
	storel %val, %pdst
	%inext = add %i, $1
	%i = %inext
	jmp @forward_loop
@backward
	%i = %n
@backward_loop
	%done = ceql %i, $0
	jnz %done, @return_zero, @backward_body
@backward_body
	%iprev = sub %i, $1
	%i = %iprev
	%off = mul %i, %elsize
	%psrc = add %src, %off
	%pdst = add %dst, %off
	%val = loadl %psrc
	storel %val, %pdst
	jmp @backward_loop
@return_zero
	ret $0
}
