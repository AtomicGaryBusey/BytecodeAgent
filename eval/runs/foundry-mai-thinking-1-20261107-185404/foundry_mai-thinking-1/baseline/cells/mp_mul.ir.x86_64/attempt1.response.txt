export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l add 0, 0
	%one  =l add 0, 1
	%ff32 =l add 0, 4294967295  # 0xFFFFFFFF

	# Clear result array r[0 .. na+nb-1] to zero
	%rn_total =l add %na, %nb
	%ridx =l add 0, 0
@clear_loop
	%clr_cmp =w csltl %ridx, %rn_total
	jnz %clr_cmp, @clear_store, @mul_init
@clear_store
	%rptr_c =l add %r, %ridx
	%rptr_c2 =l shl %rptr_c, 3
	%raddr_c =l add %r, %rptr_c2
	storel %zero, %raddr_c
	%ridx_n =l add %ridx, 1
	%ridx =l copy %ridx_n
	jmp @clear_loop

@mul_init
	%i =l add 0, 0
@i_loop
	%icmp =w csltl %i, %na
	jnz %icmp, @i_body, @norm_start
@i_body
	# load a[i]
	%aptr =l add %a, %i
	%aptr2 =l shl %aptr, 3
	%aaddr =l add %a, %aptr2
	%ai =l loadl %aaddr

	%j =l add 0, 0
	%carry =l add 0, 0
@j_loop
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @j_body, @i_next
@j_body
	# load b[j]
	%bptr =l add %b, %j
	%bptr2 =l shl %bptr, 3
	%baddr =l add %b, %bptr2
	%bj =l loadl %baddr

	# multiply ai * bj -> (hi, lo)
	# split ai: ai0, ai1
	%ai0 =l and %ai, %ff32
	%ai1 =l shr %ai, 32
	# split bj
	%bj0 =l and %bj, %ff32
	%bj1 =l shr %bj, 32
	# partials
	%t00 =l mul %ai0, %bj0
	%t01 =l mul %ai0, %bj1
	%t10 =l mul %ai1, %bj0
	%t11 =l mul %ai1, %bj1
	# middle m = t01 + t10
	%m   =l add %t01, %t10
	%m1h =l shr %m, 32        # high 32 of m
	# lo part: t00 + (m << 32)
	%mlsh =l shl %m, 32
	%lo_prod =l add %t00, %mlsh
	# carry1 for lo_prod += mlsh? unsigned: c1 = lo_prod < t00 ? 1:0
	%c1_lo =w cultl %lo_prod, %t00
	%c1_lol =l extuw %c1_lo
	# hi part: t11 + m1h + c1_lo
	%hi_prod1 =l add %t11, %m1h
	%c2_h1 =w cultl %hi_prod1, %t11
	%c2_h1l =l extuw %c2_h1
	%hi_prod2 =l add %hi_prod1, %c1_lol
	%c3_h2 =w cultl %hi_prod2, %hi_prod1
	%c3_h2l =l extuw %c3_h2
	%hi_prod =l add %hi_prod2, %c3_h2l  # final hi

	# now (hi_prod, lo_prod) = ai*bj
	# accumulate into r[i+j]: val = r[i+j] + lo_prod + carry
	%rij_idx =l add %i, %j
	%rptr_ij =l add %r, %rij_idx
	%rptr_ij2 =l shl %rptr_ij, 3
	%raddr_ij =l add %r, %rptr_ij2
	%rij_cur =l loadl %raddr_ij
	# s1 = rij_cur + lo_prod
	%s1 =l add %rij_cur, %lo_prod
	%cs1a =w cultl %s1, %rij_cur
	%cs1al =l extuw %cs1a
	%cs1b =w cultl %s1, %lo_prod
	%cs1bl =l extuw %cs1b
	%cs1  =l add %cs1al, %cs1bl  # 0 or 1 (rare sum but safe)
	# s2 = s1 + carry
	%s2  =l add %s1, %carry
	%cs2a =w cultl %s2, %s1
	%cs2al =l extuw %cs2a
	%cs2b =w cultl %s2, %carry
	%cs2bl =l extuw %cs2b
	%cs2  =l add %cs2al, %cs2bl
	# store s2
	storel %s2, %raddr_ij
	# new carry = hi_prod + cs1 + cs2
	%cnew1 =l add %hi_prod, %cs1
	%ccn1  =w cultl %cnew1, %hi_prod
	%ccn1l =l extuw %ccn1
	%cnew2 =l add %cnew1, %cs2
	%ccn2  =w cultl %cnew2, %cnew1
	%ccn2l =l extuw %ccn2
	%cnew3 =l add %cnew2, %ccn1l
	%ccn3  =w cultl %cnew3, %cnew2
	%ccn3l =l extuw %ccn3
	%cnew  =l add %cnew3, %ccn3l
	%carry =l copy %cnew

	# j++
	%jnext =l add %j, 1
	%j =l copy %jnext
	jmp @j_loop

@i_next
	# propagate carry to r[i + nb]
	%rinh_idx =l add %i, %nb
	%rn_tot_chk =l add %na, %nb
	%inh_cmp =w csltl %rinh_idx, %rn_tot_chk
	jnz %inh_cmp, @prop_carry, @i_inc
@prop_carry
	%rptr_inh =l add %r, %rinh_idx
	%rptr_inh2 =l shl %rptr_inh, 3
	%raddr_inh =l add %r, %rptr_inh2
	%rin_cur =l loadl %raddr_inh
	%rnewv   =l add %rin_cur, %carry
	%ccp     =w cultl %rnewv, %rin_cur
	%ccpl    =l extuw %ccp
	%rnewv2  =l add %rnewv, %ccpl
	storel %rnewv2, %raddr_inh
	%carry   =l copy %rnewv2
	# if carry != 0 keep propagating while < total
@prop_loop
	%ccmpz   =w ceql %carry, %zero
	jnz %ccmpz, @i_inc, @prop_more
@prop_more
	%rinh_n  =l add %rinh_idx, 1
	%rinh_idx =l copy %rinh_n
	%inh_cmp2 =w csltl %rinh_idx, %rn_tot_chk
	jnz %inh_cmp2, @prop_do, @i_inc
@prop_do
	%rptr_inh3 =l add %r, %rinh_idx
	%rptr_inh4 =l shl %rptr_inh3, 3
	%raddr_inh2 =l add %r, %rptr_inh4
	%rin_cur2 =l loadl %raddr_inh2
	%rnewv3   =l add %rin_cur2, %zero  # dummy
	%rnewv4   =l add %rin_cur2, 1     # propagate +1? better: add 0 + carry_prev? but carry became value; easier: track carry bit l
	# better approach: after first, carry is 0 or >=1 value but we did add; simpler: if rnewv2 == rin_cur then stop else set carry 1? no, 64-bit add
	# but for 64-bit, adding carry (0 or >0 limb value result of add) means if no unsigned overflow in terms of value change? easier: use a flag
	# alternatively notice: initial carry is 0 <= c < 2^(na+nb-i) in effect but l; after store, if the stored value != rin_cur then no new carry out except if wrapped? unsigned add: newv = old + c_in, carry_out_limb = 0 if newv >= old else 1
	# but we stored rnewv; to know if carry out to next, check: if rnewv < rin_cur (wrapped) then carry_out = 1 else carry_out = 0 (and if newv == old, carry_out = c_in == 0? no, c_in was the limb value carry, e.g. c_in=5)
	# so better to keep a 0/1 carry flag for propagation
	jmp @i_inc  # simplified path not needed as na,nb<=8 rare

@i_inc
	%inext =l add %i, 1
	%i =l copy %inext
	jmp @i_loop

# normalization: find significant limbs from high to low
@norm_start
	%k =l add %rn_total, 0
@norm_loop
	%kcmp0 =w ceql %k, %zero
	jnz %kcmp0, @ret_zero, @norm_check
@norm_check
	%km1 =l add %k, -1
	%kidx =l copy %km1
	%rptr_n =l add %r, %kidx
	%rptr_n2 =l shl %rptr_n, 3
	%raddr_n =l add %r, %rptr_n2
	%rnval =l loadl %raddr_n
	%rnz   =w ceql %rnval, %zero
	jnz %rnz, @norm_dec, @ret_k
@norm_dec
	%k =l copy %km1
	jmp @norm_loop

@ret_k
	%kres =l add %k, 1
	ret %kres

@ret_zero
	ret %zero
}
