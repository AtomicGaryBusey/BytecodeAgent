export function l $isqrt(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @return_zero, @find_bound

@return_zero
	ret 0

@find_bound
	%n_tmp =l copy %n
	%bitlen =l copy 0

@count_bits
	%has_bit =w csgtu %n_tmp, 0
	jnz %has_bit, @compute_bound, @shift_n

@shift_n
	%n_tmp =l shr %n_tmp, 1
	%bitlen =l add %bitlen, 1
	jmp @count_bits

@compute_bound
	%exp =l shr %bitlen, 1
	%exp =l add %exp, 1
	%bound =l copy 1

@shift_bound
	%exp_done =w ceql %exp, 0
	jnz %exp_done, @binary_search, @do_shift

@do_shift
	%bound =l shl %bound, 1
	%exp =l sub %exp, 1
	jmp @shift_bound

@binary_search
	%left =l copy 1
	%right =l copy %bound
	%ans =l copy 0

@loop
	%can_search =w csleu %left, %right
	jnz %can_search, @mid_calc, @done

@mid_calc
	%mid =l add %left, %right
	%mid =l shr %mid, 1
	%quo =l divu %n, %mid
	%mid_ok =w csleu %mid, %quo
	jnz %mid_ok, @update_ans, @update_right

@update_ans
	%ans =l copy %mid
	%left =l add %mid, 1
	jmp @loop

@update_right
	%right =l sub %mid, 1
	jmp @loop

@done
	ret %ans
}
