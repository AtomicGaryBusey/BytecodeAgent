export function l $isqrt(l %n) {
@start
	%zero =w ceql %n, 0
	jnz %zero, @return_zero, @find_bound

@return_zero
	ret 0

@find_bound
	%bound =l copy 1
	%bitlen =l copy 0
	%n_tmp =l copy %n

@count_bits
	%hasbit =w csgtl %n_tmp, 0
	jnz %hasbit, @compute_bound, @shift_n

@shift_n
	%n_tmp =l shrl %n_tmp, 1
	%bitlen =l add %bitlen, 1
	jmp @count_bits

@compute_bound
	%half =l shrl %bitlen, 1
	%exp =l add %half, 1
	%bound =l copy 1
	jmp @shift_bound

@shift_bound
	%exp_done =w ceql %exp, 0
	jnz %exp_done, @binary_search, @do_shift

@do_shift
	%bound =l shll %bound, 1
	%exp =l sub %exp, 1
	jmp @shift_bound

@binary_search
	%left =l copy 1
	%right =l copy %bound
	%ans =l copy 0

@loop
	%cond_le =w cslel %left, %right
	jnz %cond_le, @mid_calc, @done

@mid_calc
	%mid =l add %left, %right
	%mid =l shrl %mid, 1
	%prod_test =l divu %n, %mid
	%ok =w cslel %mid, %prod_test
	jnz %ok, @update_ans, @update_right

@update_ans
	%ans =l copy %mid
	%left =l add %mid, 1
	jmp @loop

@update_right
	%right =l sub %mid, 1
	jmp @loop

@done
	ret %ans
}
