export function l $map_sum(l %a, l %n, l %f) {
@entry
	%is_zero =w ceql %n, 0
	jnz %is_zero, @done_zero, @loop_init

@loop_init
	%sum_init =l add 0, 0
	%i_init =l add 0, 0
	jmp @loop

@loop
	%i =l phi @loop_init %i_init, @body %i_next
	%sum =l phi @loop_init %sum_init, @body %sum_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done

@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%fv =l call %f(l %val)
	%sum_next =l add %sum, %fv
	%i_next =l add %i, 1
	jmp @loop

@done_zero
	ret 0

@done
	ret %sum
}
