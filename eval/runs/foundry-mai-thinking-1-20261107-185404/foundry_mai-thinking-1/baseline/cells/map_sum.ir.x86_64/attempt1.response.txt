export function l $map_sum(l %a, l %n, l %f) {
@entry
	%sum_init =l add 0, 0
	%i_init =l add 0, 0
	%n_is_zero =w ceql %n, 0
	jnz %n_is_zero, @done_zero, @loop

@loop
	%i =l phi @entry %i_init, @loop %i_next
	%sum =l phi @entry %sum_init, @loop %sum_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done

@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%fv =l call %f(l %val)
	%sum_next =l add %sum, %fv
	%i_next =l add %i, 1
	jmp @loop

@done_zero
	ret 0

@done
	%res =l phi @loop %sum
	ret %res
}
