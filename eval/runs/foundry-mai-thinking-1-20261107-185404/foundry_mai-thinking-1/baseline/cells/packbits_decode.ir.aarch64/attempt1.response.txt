export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%si =l copy 0
	%oi =l copy 0
@loop
	%cond_si_n =w csltl %si, %n
	jnz %cond_si_n, @read_header, @end_ok

@read_header
	%src_h =l add %src, %si
	%hsb =w loadsb %src_h
	%si1 =l add %si, 1
	%si =l copy %si1

	%h_eq_m128 =w ceqw %hsb, -128
	jnz %h_eq_m128, @loop, @check_type

@check_type
	%h_geq_0 =w csltw 0, %hsb
	jnz %h_geq_0, @literal, @repeat

@literal
	%lit_count_w =w add %hsb, 1
	%lit_count =l extsw %lit_count_w

	%need_src_end =l add %si, %lit_count
	%src_over =w csltl %n, %need_src_end
	jnz %src_over, @ret_m2, @lit_check_out

@lit_check_out
	%need_out_end =l add %oi, %lit_count
	%out_over =w csltl %cap, %need_out_end
	jnz %out_over, @lit_copy_partial, @lit_copy_full

@lit_copy_full
	%src_ptr =l add %src, %si
	%out_ptr =l add %out, %oi
	call $memcpy_small(l %src_ptr, l %out_ptr, l %lit_count)
	%si_new =l add %si, %lit_count
	%oi_new =l add %oi, %lit_count
	%si =l copy %si_new
	%oi =l copy %oi_new
	jmp @loop

@lit_copy_partial
	%space =l sub %cap, %oi
	%src_ptr_p =l add %src, %si
	%out_ptr_p =l add %out, %oi
	call $memcpy_small(l %src_ptr_p, l %out_ptr_p, l %space)
	jmp @ret_m1

@repeat
	%rep_count_w =w sub 1, %hsb
	%rep_count =l extsw %rep_count_w

	%src_has_byte =w csltl %si, %n
	jnz %src_has_byte, @repeat_read_byte, @ret_m2

@repeat_read_byte
	%src_b_ptr =l add %src, %si
	%b_val =w loadub %src_b_ptr
	%si_r1 =l add %si, 1
	%si =l copy %si_r1

	%need_out_end_r =l add %oi, %rep_count
	%out_over_r =w csltl %cap, %need_out_end_r
	jnz %out_over_r, @repeat_fill_partial, @repeat_fill_full

@repeat_fill_full
	%out_ptr_r =l add %out, %oi
	call $memset_small(l %out_ptr_r, w %b_val, l %rep_count)
	%oi_r_new =l add %oi, %rep_count
	%oi =l copy %oi_r_new
	jmp @loop

@repeat_fill_partial
	%space_r =l sub %cap, %oi
	%out_ptr_rp =l add %out, %oi
	call $memset_small(l %out_ptr_rp, w %b_val, l %space_r)
	jmp @ret_m1

@end_ok
	%ret_ok =l copy %oi
	ret %ret_ok

@ret_m1
	%ret_m1v =l copy -1
	ret %ret_m1v

@ret_m2
	%ret_m2v =l copy -2
	ret %ret_m2v
}

function $memcpy_small(l %srcp, l %dstp, l %cnt) {
@mc_start
	%done_c =w ceql %cnt, 0
	jnz %done_c, @mc_end, @mc_loop

@mc_loop
	%byte =w loadub %srcp
	storeb %byte, %dstp
	%srcp1 =l add %srcp, 1
	%dstp1 =l add %dstp, 1
	%cnt1 =l sub %cnt, 1
	%srcp =l copy %srcp1
	%dstp =l copy %dstp1
	%cnt =l copy %cnt1
	%more =w csltl 0, %cnt
	jnz %more, @mc_loop, @mc_end

@mc_end
	ret
}

function $memset_small(l %dstp, w %valb, l %cnt) {
@ms_start
	%done_s =w ceql %cnt, 0
	jnz %done_s, @ms_end, @ms_loop

@ms_loop
	storeb %valb, %dstp
	%dstp1 =l add %dstp, 1
	%cnt1 =l sub %cnt, 1
	%dstp =l copy %dstp1
	%cnt =l copy %cnt1
	%more_s =w csltl 0, %cnt
	jnz %more_s, @ms_loop, @ms_end

@ms_end
	ret
}
