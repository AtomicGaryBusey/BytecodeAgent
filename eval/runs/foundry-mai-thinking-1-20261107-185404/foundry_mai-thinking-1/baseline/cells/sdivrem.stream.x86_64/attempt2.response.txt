sdivrem:
  push rbx
  mov rbx, rdx
  cmp rsi, 0
  je .Lb_zero
  mov rax, -9223372036854775808
  cmp rdi, rax
  jne .Lnormal
  cmp rsi, -1
  jne .Lnormal
  mov rax, rdi
  xor rdx, rdx
  jmp .Lstore
.Lb_zero:
  xor rax, rax
  mov rdx, rdi
  jmp .Lstore
.Lnormal:
  mov rax, rdi
  cqo
  idiv rsi
.Lstore:
  mov qword ptr [rbx], rdx
  pop rbx
  ret
