binary_search:
  test rsi, rsi
  jle .Lnot_found
  xor r10, r10
  mov r11, rsi
  dec r11
.Lloop:
  cmp r10, r11
  jg .Lnot_found
  mov r9, r10
  add r9, r11
  shr r9, 1
  mov r8, qword ptr [rdi + r9*8]
  cmp r8, rdx
  je .Lfound
  jl .Lgo_right
  dec r11
  jmp .Lloop
.Lgo_right:
  lea r10, [r9 + 1]
  jmp .Lloop
.Lfound:
  mov rax, r9
  ret
.Lnot_found:
  mov rax, -1
  ret
