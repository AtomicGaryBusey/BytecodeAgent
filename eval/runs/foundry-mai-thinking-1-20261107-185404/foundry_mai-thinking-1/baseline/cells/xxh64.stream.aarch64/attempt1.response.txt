xxh64:
	// x0 = p, x1 = n (int64_t), x2 = seed
	mov	x9, x0			// q = p
	mov	x10, x1			// n
	mov	x11, x2			// seed
	// constants
	movz	x12, #0xCA87
	movk	x12, #0xEB85, lsl #16
	movk	x12, #0x779B, lsl #32
	movk	x12, #0x9E37, lsl #48	// P1 = 0x9E3779B185EBCA87
	movz	x13, #0xEB4F
	movk	x13, #0xD4EB, lsl #16
	movk	x13, #0xB2AE, lsl #32
	movk	x13, #0xC2B2, lsl #48	// P2 = 0xC2B2AE3D27D4EB4F
	movz	x14, #0x779F
	movk	x14, #0xE377, lsl #16
	movk	x14, #0x5667, lsl #32
	movk	x14, #0x1656, lsl #48	// P3 = 0x165667B19E3779F9
	movz	x15, #0xAE63
	movk	x15, #0xB2AE, lsl #16
	movk	x15, #0xEBCA, lsl #32
	movk	x15, #0x85EB, lsl #48	// P4 = 0x85EBCA77C2B2AE63
	movz	x16, #0x67C5
	movk	x16, #0x5666, lsl #16
	movk	x16, #0xD4EB, lsl #32
	movk	x16, #0x27D4, lsl #48	// P5 = 0x27D4EB2F165667C5
	// (1) check n >= 32
	cmp	x10, #32
	b.lt	.Lsmall
	// v1 = seed + P1 + P2
	add	x17, x11, x12		// seed + P1
	add	x17, x17, x13		// +P2 -> v1
	// v2 = seed + P2
	add	x19, x11, x13		// v2
	// v3 = seed
	mov	x20, x11		// v3
	// v4 = seed - P1
	sub	x21, x11, x12		// v4
	// stripe loop: while q + 31 < p + n? n>=32, process full 32s
	mov	x22, x9			// q
	add	x23, x9, x10		// end = p + n
	add	x24, x22, #32
	cmp	x24, x23
	b.gt	.Lstripe_end
.Lstripe_loop:
	// v1 = round(v1, read64(q))
	ldr	x25, [x22]
	mul	x26, x25, x13		// x * P2
	add	x26, x17, x26		// acc + x*P2
	ror	x26, x26, #31		// rotl by 31
	mul	x17, x26, x12		// *P1 -> v1
	// v2
	ldr	x25, [x22, #8]
	mul	x26, x25, x13
	add	x26, x19, x26
	ror	x26, x26, #31
	mul	x19, x26, x12		// v2
	// v3
	ldr	x25, [x22, #16]
	mul	x26, x25, x13
	add	x26, x20, x26
	ror	x26, x26, #31
	mul	x20, x26, x12		// v3
	// v4
	ldr	x25, [x22, #24]
	mul	x26, x25, x13
	add	x26, x21, x26
	ror	x26, x26, #31
	mul	x21, x26, x12		// v4
	add	x22, x22, #32
	add	x24, x22, #32
	cmp	x24, x23
	b.le	.Lstripe_loop
.Lstripe_end:
	// h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18)
	ror	x0, x17, #63		// rotl(v1,1)
	ror	x1, x19, #57		// rotl(v2,7)
	add	x0, x0, x1
	ror	x1, x20, #52		// rotl(v3,12)
	add	x0, x0, x1
	ror	x1, x21, #46		// rotl(v4,18)
	add	x0, x0, x1		// h
	// merge(h,v1)
	mov	x1, xzr
	mul	x2, x17, x13		// round(0,v1): x*v2? round(acc,x)=rotl(acc+x*P2,31)*P1, acc=0 -> rotl(x*P2,31)*P1
	ror	x2, x2, #31
	mul	x2, x2, x12
	eor	x0, x0, x2		// h ^ round(0,v1)
	mul	x0, x0, x12
	add	x0, x0, x15		// +P4 -> h
	// merge(h,v2)
	mul	x2, x19, x13
	ror	x2, x2, #31
	mul	x2, x2, x12
	eor	x0, x0, x2
	mul	x0, x0, x12
	add	x0, x0, x15
	// merge(h,v3)
	mul	x2, x20, x13
	ror	x2, x2, #31
	mul	x2, x2, x12
	eor	x0, x0, x2
	mul	x0, x0, x12
	add	x0, x0, x15
	// merge(h,v4)
	mul	x2, x21, x13
	ror	x2, x2, #31
	mul	x2, x2, x12
	eor	x0, x0, x2
	mul	x0, x0, x12
	add	x0, x0, x15		// h done
	b	.Lafter_init
.Lsmall:
	// h = seed + P5
	add	x0, x11, x16		// h
.Lafter_init:
	// (2) h += n
	add	x0, x0, x10
	// (3) remaining: q is x9 (p) if small, else x22 after stripe
	cmp	x10, #32
	b.lt	.Lq_init_small
	mov	x9, x22			// q after stripes
.Lq_init_small:
	add	x1, x9, x10		// end
	// while at least 8 remain: q + 7 < end?
.Lrem8_loop:
	add	x2, x9, #8
	cmp	x2, x1
	b.gt	.Lrem8_end
	// h ^= round(0, read64(q))
	ldr	x2, [x9]
	mul	x3, x2, x13		// x*P2
	ror	x3, x3, #31		// rotl by 31
	mul	x3, x3, x12		// *P1
	eor	x0, x0, x3		// h ^= 
	// h = rotl(h,27)*P1 + P4
	ror	x0, x0, #37		// rotl(h,27) (64-27=37)
	mul	x0, x0, x12
	add	x0, x0, x15
	add	x9, x9, #8
	b	.Lrem8_loop
.Lrem8_end:
	// if at least 4 remain
	add	x2, x9, #4
	cmp	x2, x1
	b.gt	.Lrem4_end
	// h ^= read32(q) * P1
	ldr	w2, [x9]
	// zero extend 32->64
	uxtw	x2, w2
	mul	x2, x2, x12		// *P1
	eor	x0, x0, x2
	// rotl(h,23)*P2 + P3
	ror	x0, x0, #41		// rotl(h,23)
	mul	x0, x0, x13
	add	x0, x0, x14
	add	x9, x9, #4
.Lrem4_end:
	// remaining bytes
	cmp	x9, x1
	b.eq	.Lfinalize
.Lrem1_loop:
	ldrb	w2, [x9]
	uxtw	x2, w2
	mul	x2, x2, x16		// *P5
	eor	x0, x0, x2
	// rotl(h,11)*P1
	ror	x0, x0, #53		// rotl(h,11)
	mul	x0, x0, x12
	add	x9, x9, #1
	cmp	x9, x1
	b.ne	.Lrem1_loop
.Lfinalize:
	// (4) final mix
	// h ^= h >> 33
	mov	x2, x0
	lsr	x2, x2, #33
	eor	x0, x0, x2
	// h *= P2
	mul	x0, x0, x13
	// h ^= h >> 29
	mov	x2, x0
	lsr	x2, x2, #29
	eor	x0, x0, x2
	// h *= P3
	mul	x0, x0, x14
	// h ^= h >> 32
	mov	x2, x0
	lsr	x2, x2, #32
	eor	x0, x0, x2
	// return x0
	ret
