xxh64:
	// x0 = p, x1 = n (int64_t), x2 = seed
	mov	x19, x0			// p
	mov	x20, x1			// n
	mov	x21, x2			// seed

	// constants
	movz	x9, #0xCA87
	movk	x9, #0xEB85, lsl #16
	movk	x9, #0x779B, lsl #32
	movk	x9, #0x9E37, lsl #48	// P1 = 0x9E3779B185EBCA87

	movz	x10, #0xEB4F
	movk	x10, #0xD4EB, lsl #16
	movk	x10, #0xB2AE, lsl #32
	movk	x10, #0xC2B2, lsl #48	// P2 = 0xC2B2AE3D27D4EB4F

	movz	x11, #0x779F
	movk	x11, #0xE377, lsl #16
	movk	x11, #0x5667, lsl #32
	movk	x11, #0x1656, lsl #48	// P3 = 0x165667B19E3779F9

	movz	x12, #0xAE63
	movk	x12, #0xB2AE, lsl #16
	movk	x12, #0xEBCA, lsl #32
	movk	x12, #0x85EB, lsl #48	// P4 = 0x85EBCA77C2B2AE63

	movz	x13, #0x67C5
	movk	x13, #0x5666, lsl #16
	movk	x13, #0xD4EB, lsl #32
	movk	x13, #0x27D4, lsl #48	// P5 = 0x27D4EB2F165667C5

	// (1) check n >= 32
	cmp	x20, #32
	b.lt	.Lsmall

	// v1 = seed + P1 + P2
	add	x14, x21, x9		// seed + P1
	add	x14, x14, x10		// +P2 -> v1

	// v2 = seed + P2
	add	x15, x21, x10		// v2

	// v3 = seed
	mov	x22, x21		// v3

	// v4 = seed - P1
	sub	x23, x21, x9		// v4

	// stripe loop
	mov	x24, x19		// q = p
	add	x25, x19, x20		// end = p + n

.Lstripe_loop:
	add	x26, x24, #32
	cmp	x26, x25
	b.gt	.Lstripe_end

	// v1 = round(v1, read64(q))
	ldr	x0, [x24]
	mul	x1, x0, x10		// x * P2
	add	x1, x14, x1		// acc + x*P2
	ror	x1, x1, #31		// rotl by 31
	mul	x14, x1, x9		// *P1 -> v1

	// v2 = round(v2, read64(q+8))
	ldr	x0, [x24, #8]
	mul	x1, x0, x10
	add	x1, x15, x1
	ror	x1, x1, #31
	mul	x15, x1, x9		// v2

	// v3 = round(v3, read64(q+16))
	ldr	x0, [x24, #16]
	mul	x1, x0, x10
	add	x1, x22, x1
	ror	x1, x1, #31
	mul	x22, x1, x9		// v3

	// v4 = round(v4, read64(q+24))
	ldr	x0, [x24, #24]
	mul	x1, x0, x10
	add	x1, x23, x1
	ror	x1, x1, #31
	mul	x23, x1, x9		// v4

	add	x24, x24, #32
	b	.Lstripe_loop

.Lstripe_end:
	// h = rotl(v1,1)+rotl(v2,7)+rotl(v3,12)+rotl(v4,18)
	ror	x0, x14, #63		// rotl(v1,1)
	ror	x1, x15, #57		// rotl(v2,7)
	add	x0, x0, x1
	ror	x1, x22, #52		// rotl(v3,12)
	add	x0, x0, x1
	ror	x1, x23, #46		// rotl(v4,18)
	add	x0, x0, x1		// h

	// merge(h,v1)
	mul	x1, x14, x10		// x*P2
	ror	x1, x1, #31		// rotl
	mul	x1, x1, x9		// *P1
	eor	x0, x0, x1		// h ^
	mul	x0, x0, x9		// *P1
	add	x0, x0, x12		// +P4

	// merge(h,v2)
	mul	x1, x15, x10
	ror	x1, x1, #31
	mul	x1, x1, x9
	eor	x0, x0, x1
	mul	x0, x0, x9
	add	x0, x0, x12

	// merge(h,v3)
	mul	x1, x22, x10
	ror	x1, x1, #31
	mul	x1, x1, x9
	eor	x0, x0, x1
	mul	x0, x0, x9
	add	x0, x0, x12

	// merge(h,v4)
	mul	x1, x23, x10
	ror	x1, x1, #31
	mul	x1, x1, x9
	eor	x0, x0, x1
	mul	x0, x0, x9
	add	x0, x0, x12		// h

	mov	x8, x24			// q after stripes
	b	.Lafter_init

.Lsmall:
	// h = seed + P5
	add	x0, x21, x13		// h
	mov	x8, x19			// q = p

.Lafter_init:
	// (2) h += n
	add	x0, x0, x20

	// (3) remaining: x8 = q, end = p + n
	add	x1, x19, x20		// end

.Lrem8_loop:
	sub	x2, x1, x8
	cmp	x2, #8
	b.lt	.Lrem4_check

	// h ^= round(0, read64(q))
	ldr	x2, [x8]
	mul	x3, x2, x10		// x*P2
	ror	x3, x3, #31		// rotl
	mul	x3, x3, x9		// *P1
	eor	x0, x0, x3		// h ^

	// h = rotl(h, 27) * P1 + P4
	ror	x0, x0, #37		// rotl(h, 27)
	mul	x0, x0, x9		// *P1
	add	x0, x0, x12		// +P4

	add	x8, x8, #8
	b	.Lrem8_loop

.Lrem4_check:
	sub	x2, x1, x8
	cmp	x2, #4
	b.lt	.Lrem1_loop

	// h ^= read32(q) * P1
	ldr	w2, [x8]
	umull	x2, w2, w2
	mul	x2, x2, x9		// *P1
	eor	x0, x0, x2		// h ^

	// h = rotl(h, 23) * P2 + P3
	ror	x0, x0, #41		// rotl(h, 23)
	mul	x0, x0, x10		// *P2
	add	x0, x0, x11		// +P3

	add	x8, x8, #4

.Lrem1_loop:
	cmp	x8, x1
	b.ge	.Lfinalize

	// h ^= c * P5
	ldrb	w2, [x8]
	mul	x2, x2, x13		// c * P5
	eor	x0, x0, x2		// h ^

	// h = rotl(h, 11) * P1
	ror	x0, x0, #53		// rotl(h, 11)
	mul	x0, x0, x9		// *P1

	add	x8, x8, #1
	b	.Lrem1_loop

.Lfinalize:
	// (4) avalanche
	// h ^= h >> 33
	lsr	x1, x0, #33
	eor	x0, x0, x1

	// h *= P2
	mul	x0, x0, x10

	// h ^= h >> 29
	lsr	x1, x0, #29
	eor	x0, x0, x1

	// h *= P3
	mul	x0, x0, x11

	// h ^= h >> 32
	lsr	x1, x0, #32
	eor	x0, x0, x1

	ret
