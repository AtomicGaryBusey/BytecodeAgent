export function l $sum_to(l %n) {
@entry
	%le0 = cslel %n, 0l
	jnz %le0, @ret_zero, @init
@init
	%sum = copy 0l
	%i = copy 1l
	jmp @cond
@cond
	%ok = cslel %i, %n
	jnz %ok, @body, @ret_sum
@body
	%new_sum = addl %sum, %i
	%new_i = addl %i, 1l
	%sum = copy %new_sum
	%i = copy %new_i
	jmp @cond
@ret_sum
	ret %sum
@ret_zero
	ret 0l
}
