export function l $add_overflow(l %a, l %b, l %out) {
@start
  %sum = addl %a, %b
  storel %sum, %out
  %sa = csltl %a, $0
  %sb = csltl %b, $0
  %ssum = csltl %sum, $0
  %samesign_ab = ceql %sa, %sb
  %same_sign_as_a = ceql %sa, %ssum
  %sign_changed = xorw %same_sign_as_a, $1
  %overflow_w = andw %samesign_ab, %sign_changed
  ret %overflow_w
}
