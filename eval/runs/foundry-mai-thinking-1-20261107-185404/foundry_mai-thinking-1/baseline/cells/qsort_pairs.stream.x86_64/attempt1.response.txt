qsort_pairs:
cmp rsi, 1
jle .Lret_zero
cmp rsi, 0
je .Lret_zero
push rbp
mov rbp, rsp
push rbx
push r12
push r13
push r14
push r15
sub rsp, 16
mov rbx, rdi
mov r12, rsi
mov rax, r12
dec rax
mov rcx, 2
mul rcx
mov r13, rax
mov rax, r12
shr rax, 1
mov rcx, 2
mul rcx
mov r14, rax
mov r15, r13
cmp r14, r15
jge .Lpivot_done
mov rdi, rbx
mov rsi, r14
mov rdx, r15
call .Lswap_pair
.Lpivot_done:
xor r8, r8
mov r9, 0
mov r10, r13
sub r10, 2
jmp .Lpart_loop_cond
.Lpart_loop:
mov rax, r9
mov rcx, 2
mul rcx
mov r11, rax
mov rdx, qword ptr [rbx + r11]
mov rax, qword ptr [rbx + r11 + 8]
mov rsi, qword ptr [rbx + r13]
mov rdi, qword ptr [rbx + r13 + 8]
cmp rdx, rsi
jl .Lless
jg .Lnot_less
cmp rax, rdi
jle .Lless
jmp .Lnot_less
.Lless:
cmp r8, r9
je .Lno_swap
mov rdi, rbx
mov rsi, r8
mov rdx, r9
call .Lswap_pair
.Lno_swap:
add r8, 2
.Lnot_less:
add r9, 2
.Lpart_loop_cond:
cmp r9, r10
jle .Lpart_loop
mov rdi, rbx
mov rsi, r8
mov rdx, r13
call .Lswap_pair
mov r14, r8
shr r14, 1
mov r15, r12
sub r15, r14
dec r15
cmp r14, 1
jle .Lrecurse_right
mov rdi, rbx
mov rsi, r14
call qsort_pairs
.Lrecurse_right:
cmp r15, 1
jle .Lend
mov rax, r8
add rax, 2
mov rcx, 2
mul rcx
add rax, rbx
mov rdi, rax
mov rsi, r15
call qsort_pairs
.Lend:
xor rax, rax
add rsp, 16
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
ret
.Lret_zero:
xor rax, rax
ret
.Lswap_pair:
mov rax, qword ptr [rdi + rsi]
mov rcx, qword ptr [rdi + rdx]
mov qword ptr [rdi + rsi], rcx
mov qword ptr [rdi + rdx], rax
add rsi, 8
add rdx, 8
mov rax, qword ptr [rdi + rsi]
mov rcx, qword ptr [rdi + rdx]
mov qword ptr [rdi + rsi], rcx
mov qword ptr [rdi + rdx], rax
ret
