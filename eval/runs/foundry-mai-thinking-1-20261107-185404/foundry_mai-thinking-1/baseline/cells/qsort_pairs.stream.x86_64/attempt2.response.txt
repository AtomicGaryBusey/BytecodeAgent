qsort_pairs:
cmp rsi, 1
jle .Lret_zero
cmp rsi, 0
je .Lret_zero
push rbp
mov rbp, rsp
push rbx
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
mov rax, r12
dec rax
shl rax, 1
mov r13, rax
mov rax, r12
shr rax, 1
shl rax, 1
mov r14, rax
cmp r14, r13
jge .Lpivot_done
mov rdi, rbx
mov rsi, r14
mov rdx, r13
call .Lswap_pair
.Lpivot_done:
xor r8, r8
xor r9, r9
mov r10, r13
sub r10, 2
.Lpart_loop_cond:
cmp r9, r10
jg .Lpart_loop_end
.Lpart_loop:
mov rax, r9
mov r11, qword ptr [rbx + rax]
mov rcx, qword ptr [rbx + rax + 8]
mov rdx, qword ptr [rbx + r13]
mov rax, qword ptr [rbx + r13 + 8]
cmp r11, rdx
jl .Lless
jg .Lnot_less
cmp rcx, rax
jg .Lnot_less
.Lless:
cmp r8, r9
je .Lno_swap
mov rdi, rbx
mov rsi, r8
mov rdx, r9
call .Lswap_pair
.Lno_swap:
add r8, 2
.Lnot_less:
add r9, 2
jmp .Lpart_loop_cond
.Lpart_loop_end:
mov rdi, rbx
mov rsi, r8
mov rdx, r13
call .Lswap_pair
mov r14, r8
shr r14, 1
mov r15, r12
sub r15, r14
dec r15
cmp r14, 1
jle .Lrecurse_right
mov rdi, rbx
mov rsi, r14
call qsort_pairs
.Lrecurse_right:
cmp r15, 1
jle .Lend
mov rax, r8
add rax, 2
mov rdi, rbx
add rdi, rax
mov rsi, r15
call qsort_pairs
.Lend:
xor rax, rax
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
ret
.Lret_zero:
xor rax, rax
ret
.Lswap_pair:
mov rax, qword ptr [rdi + rsi]
mov rcx, qword ptr [rdi + rdx]
mov qword ptr [rdi + rsi], rcx
mov qword ptr [rdi + rdx], rax
add rsi, 8
add rdx, 8
mov rax, qword ptr [rdi + rsi]
mov rcx, qword ptr [rdi + rdx]
mov qword ptr [rdi + rsi], rcx
mov qword ptr [rdi + rdx], rax
ret
