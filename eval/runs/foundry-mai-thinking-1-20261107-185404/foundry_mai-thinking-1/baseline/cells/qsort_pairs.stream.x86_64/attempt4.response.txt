qsort_pairs:
cmp rsi, 1
jle .Lret_zero
cmp rsi, 0
je .Lret_zero
push rbp
mov rbp, rsp
push rbx
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
mov r13, r12
dec r13
shl r13, 1
mov r14, r12
shr r14, 1
shl r14, 1
cmp r14, r13
jge .Lpivot_selected
mov rdi, rbx
mov rsi, r14
mov rdx, r13
call .Lswap_pair
.Lpivot_selected:
mov r8, -2
xor r9, r9
.Lpartition_loop:
cmp r9, r13
jge .Lpartition_end
mov rax, qword ptr [rbx + r9]
mov rcx, qword ptr [rbx + r9 + 8]
mov r10, qword ptr [rbx + r13]
mov r11, qword ptr [rbx + r13 + 8]
cmp rax, r10
jl .Lcurrent_less_equal
jg .Lcurrent_greater
cmp rcx, r11
jle .Lcurrent_less_equal
jmp .Lcurrent_greater
.Lcurrent_less_equal:
add r8, 2
cmp r8, r9
je .Lno_swap_current
mov rdi, rbx
mov rsi, r8
mov rdx, r9
call .Lswap_pair
.Lno_swap_current:
.Lcurrent_greater:
add r9, 2
jmp .Lpartition_loop
.Lpartition_end:
lea r15, [r8 + 2]
mov rdi, rbx
mov rsi, r15
mov rdx, r13
call .Lswap_pair
mov r13, r15
shr r13, 1
mov r14, r12
sub r14, r13
dec r14
cmp r13, 1
jle .Lrecurse_right
mov rdi, rbx
mov rsi, r13
call qsort_pairs
.Lrecurse_right:
cmp r14, 1
jle .Lfunction_end
mov rax, r15
add rax, 2
mov rdi, rbx
add rdi, rax
mov rsi, r14
call qsort_pairs
.Lfunction_end:
xor rax, rax
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
ret
.Lret_zero:
xor rax, rax
ret
.Lswap_pair:
mov rax, qword ptr [rdi + rsi]
mov rcx, qword ptr [rdi + rdx]
mov qword ptr [rdi + rsi], rcx
mov qword ptr [rdi + rdx], rax
add rsi, 8
add rdx, 8
mov rax, qword ptr [rdi + rsi]
mov rcx, qword ptr [rdi + rdx]
mov qword ptr [rdi + rsi], rcx
mov qword ptr [rdi + rdx], rax
ret
