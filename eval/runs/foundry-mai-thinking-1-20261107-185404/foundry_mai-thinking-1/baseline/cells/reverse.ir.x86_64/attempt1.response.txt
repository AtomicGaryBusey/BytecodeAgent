export function l $reverse(l %a, l %n) {
@entry
	%le1 = csltl %n, 2
	jnz %le1, @done, @init

@init
	%l = addl 0, 0
	%r = subl %n, 1
	jmp @cond

@cond
	%c = csltl %l, %r
	jnz %c, @body, @done

@body
	%sl = mull %l, 8
	%sr = mull %r, 8
	%pl = addl %a, %sl
	%pr = addl %a, %sr
	%v1 = loadl %pl
	%v2 = loadl %pr
	storel %v2, %pl
	storel %v1, %pr
	%l2 = addl %l, 1
	%r2 = subl %r, 1
	%l = %l2
	%r = %r2
	jmp @cond

@done
	ret 0
}
