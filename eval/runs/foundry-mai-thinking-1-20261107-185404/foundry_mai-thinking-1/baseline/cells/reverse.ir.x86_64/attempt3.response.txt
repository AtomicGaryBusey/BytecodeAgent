export function l $reverse(l %a, l %n) {
@entry
	%small = csltq %n, 2
	jnz %small, @done, @init

@init
	%l0 = 0
	%r0 = subq %n, 1
	jmp @cond

@cond
	%l = phi @init %l0, @body %l1
	%r = phi @init %r0, @body %r1
	%cont = csltq %l, %r
	jnz %cont, @body, @done

@body
	%offl = mulq %l, 8
	%offr = mulq %r, 8
	%pl = addq %a, %offl
	%pr = addq %a, %offr
	%v1 = loadq %pl
	%v2 = loadq %pr
	storeq %v2, %pl
	storeq %v1, %pr
	%l1 = addq %l, 1
	%r1 = subq %r, 1
	jmp @cond

@done
	ret 0
}
