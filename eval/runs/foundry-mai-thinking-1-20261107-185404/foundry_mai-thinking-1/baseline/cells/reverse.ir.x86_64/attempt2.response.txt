export function l $reverse(l %a, l %n) {
@entry
	%small = csltl %n, 2
	jnz %small, @done, @init

@init
	%l0 = 0
	%r0 = subl %n, 1
	jmp @cond

@cond
	%l = phi @init %l0, @body %l1
	%r = phi @init %r0, @body %r1
	%cont = csltl %l, %r
	jnz %cont, @body, @done

@body
	%offl = mull %l, 8
	%offr = mull %r, 8
	%pl = addl %a, %offl
	%pr = addl %a, %offr
	%v1 = loadl %pl
	%v2 = loadl %pr
	storel %v2, %pl
	storel %v1, %pr
	%l1 = addl %l, 1
	%r1 = subl %r, 1
	jmp @cond

@done
	ret 0
}
