export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%in_rem =w csltl %i, %n
	jnz %in_rem, @read_header, @return_o

@read_header
	%hptr =l add %src, %i
	%hb_w =w loadsb %hptr
	%i_new =l add %i, 1
	%i =l copy %i_new
	%hb_is_nop =w ceqw %hb_w, -128
	jnz %hb_is_nop, @loop, @check_type

@check_type
	%hb_geq0 =w cslew 0, %hb_w
	jnz %hb_geq0, @literal, @repeat

@literal
	%hb_l =l extw %hb_w
	%len_add1 =l add %hb_l, 1
	%need_src =l add %i, %len_add1
	%src_ok =w cslel %need_src, %n
	jnz %src_ok, @literal_check_out, @return_neg2

@literal_check_out
	%new_o_l =l add %o, %len_add1
	%out_ok =w cslel %new_o_l, %cap
	jnz %out_ok, @literal_full, @literal_partial

@literal_full
	%j =l copy 0
	jmp @literal_full_loop

@literal_full_loop
	%j_lt_len =w csltl %j, %len_add1
	jnz %j_lt_len, @literal_full_body, @literal_full_done

@literal_full_body
	%sptr =l add %src, %i
	%sptr_j =l add %sptr, %j
	%b_w =w loadub %sptr_j
	%optr =l add %out, %o
	%optr_j =l add %optr, %j
	storeb %b_w, %optr_j
	%j_inc =l add %j, 1
	%j =l copy %j_inc
	jmp @literal_full_loop

@literal_full_done
	%i_new_l =l add %i, %len_add1
	%i =l copy %i_new_l
	%o_new_l =l add %o, %len_add1
	%o =l copy %o_new_l
	jmp @loop

@literal_partial
	%rem =l sub %cap, %o
	%j_p =l copy 0
	jmp @literal_partial_loop

@literal_partial_loop
	%j_lt_rem =w csltl %j_p, %rem
	jnz %j_lt_rem, @literal_partial_body, @return_neg1

@literal_partial_body
	%sptr_p =l add %src, %i
	%sptr_pj =l add %sptr_p, %j_p
	%b_p_w =w loadub %sptr_pj
	%optr_p =l add %out, %o
	%optr_pj =l add %optr_p, %j_p
	storeb %b_p_w, %optr_pj
	%j_p_inc =l add %j_p, 1
	%j_p =l copy %j_p_inc
	jmp @literal_partial_loop

@repeat
	%hb_l_r =l extw %hb_w
	%count =l sub 1, %hb_l_r
	%has_b =w csltl %i, %n
	jnz %has_b, @repeat_read, @return_neg2

@repeat_read
	%bptr =l add %src, %i
	%br_w =w loadub %bptr
	%i_r =l add %i, 1
	%i =l copy %i_r
	%new_o_r =l add %o, %count
	%out_ok_r =w cslel %new_o_r, %cap
	jnz %out_ok_r, @repeat_full, @repeat_partial

@repeat_full
	%j_r =l copy 0
	jmp @repeat_full_loop

@repeat_full_loop
	%j_lt_c =w csltl %j_r, %count
	jnz %j_lt_c, @repeat_full_body, @repeat_full_done

@repeat_full_body
	%optr_r =l add %out, %o
	%optr_rj =l add %optr_r, %j_r
	storeb %br_w, %optr_rj
	%j_r_inc =l add %j_r, 1
	%j_r =l copy %j_r_inc
	jmp @repeat_full_loop

@repeat_full_done
	%o_r_new =l add %o, %count
	%o =l copy %o_r_new
	jmp @loop

@repeat_partial
	%rem_r =l sub %cap, %o
	%j_rp =l copy 0
	jmp @repeat_partial_loop

@repeat_partial_loop
	%j_lt_rem_r =w csltl %j_rp, %rem_r
	jnz %j_lt_rem_r, @repeat_partial_body, @return_neg1

@repeat_partial_body
	%optr_rp =l add %out, %o
	%optr_rpj =l add %optr_rp, %j_rp
	storeb %br_w, %optr_rpj
	%j_rp_inc =l add %j_rp, 1
	%j_rp =l copy %j_rp_inc
	jmp @repeat_partial_loop

@return_o
	%ret_o =l copy %o
	ret %ret_o

@return_neg1
	ret -1

@return_neg2
	ret -2
}
