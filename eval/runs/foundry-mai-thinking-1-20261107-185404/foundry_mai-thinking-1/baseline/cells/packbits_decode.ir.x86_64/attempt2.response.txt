export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_in =w csltl %i, %n
	jnz %cond_in, @read_header, @return_o

@read_header
	%hptr =l add %src, %i
	%hb_w =w loadsb %hptr
	%i1 =l add %i, 1
	%i =l copy %i1
	%nop_w =w copy -128
	%is_nop =w ceql %hb_w, %nop_w
	jnz %is_nop, @loop, @check_type

@check_type
	%zero_w =w copy 0
	%is_literal =w cslew %zero_w, %hb_w
	jnz %is_literal, @literal, @repeat

@literal
	%len_w_add =w add %hb_w, 1
	%len_l =l extsw %len_w_add
	%need_src =l add %i, %len_l
	%src_ok_lit =w cslel %need_src, %n
	jnz %src_ok_lit, @literal_check_out, @return_neg2

@literal_check_out
	%new_o_lit =l add %o, %len_l
	%out_ok_lit =w cslel %new_o_lit, %cap
	jnz %out_ok_lit, @literal_full, @literal_partial

@literal_full
	%j =l copy 0
	jmp @literal_full_loop

@literal_full_loop
	%jf_lt =w csltl %j, %len_l
	jnz %jf_lt, @literal_full_body, @literal_full_done

@literal_full_body
	%sptr_f =l add %src, %i
	%sptr_fj =l add %sptr_f, %j
	%b_f =w loadub %sptr_fj
	%optr_f =l add %out, %o
	%optr_fj =l add %optr_f, %j
	storeb %b_f, %optr_fj
	%j1f =l add %j, 1
	%j =l copy %j1f
	jmp @literal_full_loop

@literal_full_done
	%i_lfd =l add %i, %len_l
	%i =l copy %i_lfd
	%o_lfd =l add %o, %len_l
	%o =l copy %o_lfd
	jmp @loop

@literal_partial
	%rem_lp =l sub %cap, %o
	%j_p =l copy 0
	jmp @literal_partial_loop

@literal_partial_loop
	%jp_lt =w csltl %j_p, %rem_lp
	jnz %jp_lt, @literal_partial_body, @return_neg1

@literal_partial_body
	%sptr_p =l add %src, %i
	%sptr_pj =l add %sptr_p, %j_p
	%b_p =w loadub %sptr_pj
	%optr_p =l add %out, %o
	%optr_pj =l add %optr_p, %j_p
	storeb %b_p, %optr_pj
	%j1p =l add %j_p, 1
	%j_p =l copy %j1p
	jmp @literal_partial_loop

@repeat
	%count_w =w sub 1, %hb_w
	%count_l =l extsw %count_w
	%has_byte =w csltl %i, %n
	jnz %has_byte, @repeat_read, @return_neg2

@repeat_read
	%bptr_r =l add %src, %i
	%b_r =w loadub %bptr_r
	%i1r =l add %i, 1
	%i =l copy %i1r
	%new_o_r =l add %o, %count_l
	%out_ok_r =w cslel %new_o_r, %cap
	jnz %out_ok_r, @repeat_full, @repeat_partial

@repeat_full
	%j_r =l copy 0
	jmp @repeat_full_loop

@repeat_full_loop
	%jr_lt =w csltl %j_r, %count_l
	jnz %jr_lt, @repeat_full_body, @repeat_full_done

@repeat_full_body
	%optr_rf =l add %out, %o
	%optr_rfj =l add %optr_rf, %j_r
	storeb %b_r, %optr_rfj
	%j1rf =l add %j_r, 1
	%j_r =l copy %j1rf
	jmp @repeat_full_loop

@repeat_full_done
	%o_rfd =l add %o, %count_l
	%o =l copy %o_rfd
	jmp @loop

@repeat_partial
	%rem_rp =l sub %cap, %o
	%j_rp =l copy 0
	jmp @repeat_partial_loop

@repeat_partial_loop
	%jrp_lt =w csltl %j_rp, %rem_rp
	jnz %jrp_lt, @repeat_partial_body, @return_neg1

@repeat_partial_body
	%optr_rp =l add %out, %o
	%optr_rpj =l add %optr_rp, %j_rp
	storeb %b_r, %optr_rpj
	%j1rp =l add %j_rp, 1
	%j_rp =l copy %j1rp
	jmp @repeat_partial_loop

@return_o
	%retv =l copy %o
	ret %retv

@return_neg1
	ret -1

@return_neg2
	ret -2
}
