export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_i_n =w csltl %i, %n
	jnz %cond_i_n, @read_header, @return_o

@read_header
	%hptr =l add %src, %i
	%hb_w =w loadsb %hptr
	%i_new =l add %i, 1
	%i =l copy %i_new
	%hb_is_nop =w ceql %hb_w, -128
	jnz %hb_is_nop, @loop, @check_literal

@check_literal
	%hb_geq0 =w csltw 0, %hb_w
	jnz %hb_geq0, @literal, @repeat

@literal
	%len_w =w add %hb_w, 1
	%len =l extsw %len_w
	%need_src_end =l add %i, %len
	%src_ok =w cslel %need_src_end, %n
	jnz %src_ok, @literal_check_out, @return_neg2

@literal_check_out
	%new_o1 =l add %o, %len
	%out_ok_full =w cslel %new_o1, %cap
	jnz %out_ok_full, @literal_copy_full, @literal_copy_partial

@literal_copy_full
	%j =l copy 0
	jmp @literal_full_loop

@literal_full_loop
	%jf_lt =w csltl %j, %len
	jnz %jf_lt, @literal_full_body, @literal_full_done

@literal_full_body
	%sptr_f =l add %src, %i
	%sptr_fj =l add %sptr_f, %j
	%b_f =w loadub %sptr_fj
	%optr_f =l add %out, %o
	%optr_fj =l add %optr_f, %j
	storeb %b_f, %optr_fj
	%j_f1 =l add %j, 1
	%j =l copy %j_f1
	jmp @literal_full_loop

@literal_full_done
	%i_lf =l add %i, %len
	%i =l copy %i_lf
	%o_lf =l add %o, %len
	%o =l copy %o_lf
	jmp @loop

@literal_copy_partial
	%rem_lp =l sub %cap, %o
	%j_p =l copy 0
	jmp @literal_part_loop

@literal_part_loop
	%jp_lt =w csltl %j_p, %rem_lp
	jnz %jp_lt, @literal_part_body, @return_neg1

@literal_part_body
	%sptr_p =l add %src, %i
	%sptr_pj =l add %sptr_p, %j_p
	%b_p =w loadub %sptr_pj
	%optr_p =l add %out, %o
	%optr_pj =l add %optr_p, %j_p
	storeb %b_p, %optr_pj
	%j_p1 =l add %j_p, 1
	%j_p =l copy %j_p1
	jmp @literal_part_loop

@repeat
	%c_r =w copy %hb_w
	%count_w =w sub 1, %c_r
	%count =l extsw %count_w
	%need_r_byte =l add %i, 0
	%r_byte_ok =w csltl %need_r_byte, %n
	jnz %r_byte_ok, @repeat_read_byte, @return_neg2

@repeat_read_byte
	%bptr_r =l add %src, %i
	%b_r =w loadub %bptr_r
	%i_r1 =l add %i, 1
	%i =l copy %i_r1
	%new_o_r =l add %o, %count
	%r_full_ok =w cslel %new_o_r, %cap
	jnz %r_full_ok, @repeat_copy_full, @repeat_copy_partial

@repeat_copy_full
	%j_r =l copy 0
	jmp @repeat_full_loop

@repeat_full_loop
	%jr_lt =w csltl %j_r, %count
	jnz %jr_lt, @repeat_full_body, @repeat_full_done

@repeat_full_body
	%optr_rf =l add %out, %o
	%optr_rfj =l add %optr_rf, %j_r
	storeb %b_r, %optr_rfj
	%j_r1 =l add %j_r, 1
	%j_r =l copy %j_r1
	jmp @repeat_full_loop

@repeat_full_done
	%o_rf =l add %o, %count
	%o =l copy %o_rf
	jmp @loop

@repeat_copy_partial
	%rem_rp =l sub %cap, %o
	%j_rp =l copy 0
	jmp @repeat_part_loop

@repeat_part_loop
	%jrp_lt =w csltl %j_rp, %rem_rp
	jnz %jrp_lt, @repeat_part_body, @return_neg1

@repeat_part_body
	%optr_rp =l add %out, %o
	%optr_rpj =l add %optr_rp, %j_rp
	storeb %b_r, %optr_rpj
	%j_rp1 =l add %j_rp, 1
	%j_rp =l copy %j_rp1
	jmp @repeat_part_loop

@return_o
	%ret_o =l copy %o
	ret %ret_o

@return_neg1
	ret -1

@return_neg2
	ret -2
}
