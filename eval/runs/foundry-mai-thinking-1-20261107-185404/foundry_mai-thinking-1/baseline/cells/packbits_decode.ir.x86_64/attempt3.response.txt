export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	%i =l copy 0
	%o =l copy 0
	jmp @loop

@loop
	%cond_in =w csltl %i, %n
	jnz %cond_in, @read_header, @return_o

@read_header
	%hptr =l add %src, %i
	%hb_w =w loadsb %hptr
	%hb_l =l extsw %hb_w
	%i1 =l add %i, 1
	%i =l copy %i1
	%is_nop =w ceql %hb_l, -128l
	jnz %is_nop, @loop, @check_type

@check_type
	%is_literal =w cslel 0l, %hb_l
	jnz %is_literal, @literal, @repeat

@literal
	%len_l_add1 =l add %hb_l, 1
	%need_src_lit =l add %i, %len_l_add1
	%src_ok_lit =w cslel %need_src_lit, %n
	jnz %src_ok_lit, @literal_check_out, @return_neg2

@literal_check_out
	%new_o_lit =l add %o, %len_l_add1
	%out_ok_lit =w cslel %new_o_lit, %cap
	jnz %out_ok_lit, @literal_full, @literal_partial

@literal_full
	%j_lf =l copy 0
	jmp @literal_full_loop

@literal_full_loop
	%jf_lt =w csltl %j_lf, %len_l_add1
	jnz %jf_lt, @literal_full_body, @literal_full_done

@literal_full_body
	%sptr_lf =l add %src, %i
	%sptr_lfj =l add %sptr_lf, %j_lf
	%b_lf_w =w loadub %sptr_lfj
	%optr_lf =l add %out, %o
	%optr_lfj =l add %optr_lf, %j_lf
	storeb %b_lf_w, %optr_lfj
	%j1_lf =l add %j_lf, 1
	%j_lf =l copy %j1_lf
	jmp @literal_full_loop

@literal_full_done
	%i_lfd =l add %i, %len_l_add1
	%i =l copy %i_lfd
	%o_lfd =l add %o, %len_l_add1
	%o =l copy %o_lfd
	jmp @loop

@literal_partial
	%rem_lp =l sub %cap, %o
	%j_lp =l copy 0
	jmp @literal_partial_loop

@literal_partial_loop
	%jp_lt =w csltl %j_lp, %rem_lp
	jnz %jp_lt, @literal_partial_body, @return_neg1

@literal_partial_body
	%sptr_lp =l add %src, %i
	%sptr_lpj =l add %sptr_lp, %j_lp
	%b_lp_w =w loadub %sptr_lpj
	%optr_lp =l add %out, %o
	%optr_lpj =l add %optr_lp, %j_lp
	storeb %b_lp_w, %optr_lpj
	%j1_lp =l add %j_lp, 1
	%j_lp =l copy %j1_lp
	jmp @literal_partial_loop

@repeat
	%count_r =l sub 1l, %hb_l
	%has_byte_r =w csltl %i, %n
	jnz %has_byte_r, @repeat_read, @return_neg2

@repeat_read
	%bptr_rr =l add %src, %i
	%b_rr_w =w loadub %bptr_rr
	%i1_rr =l add %i, 1
	%i =l copy %i1_rr
	%new_o_rr =l add %o, %count_r
	%out_ok_rr =w cslel %new_o_rr, %cap
	jnz %out_ok_rr, @repeat_full, @repeat_partial

@repeat_full
	%j_rf =l copy 0
	jmp @repeat_full_loop

@repeat_full_loop
	%jr_lt =w csltl %j_rf, %count_r
	jnz %jr_lt, @repeat_full_body, @repeat_full_done

@repeat_full_body
	%optr_rf =l add %out, %o
	%optr_rfj =l add %optr_rf, %j_rf
	storeb %b_rr_w, %optr_rfj
	%j1_rf =l add %j_rf, 1
	%j_rf =l copy %j1_rf
	jmp @repeat_full_loop

@repeat_full_done
	%o_rfd =l add %o, %count_r
	%o =l copy %o_rfd
	jmp @loop

@repeat_partial
	%rem_rp =l sub %cap, %o
	%j_rp =l copy 0
	jmp @repeat_partial_loop

@repeat_partial_loop
	%jrp_lt =w csltl %j_rp, %rem_rp
	jnz %jrp_lt, @repeat_partial_body, @return_neg1

@repeat_partial_body
	%optr_rp =l add %out, %o
	%optr_rpj =l add %optr_rp, %j_rp
	storeb %b_rr_w, %optr_rpj
	%j1_rp =l add %j_rp, 1
	%j_rp =l copy %j1_rp
	jmp @repeat_partial_loop

@return_o
	%retv_o =l copy %o
	ret %retv_o

@return_neg1
	ret -1

@return_neg2
	ret -2
}
