map_into:
  cmp rdx, 0
  jle .Lzero
  push rbx
  push rbp
  push r12
  push r13
  push r14
  push r15
  sub rsp, 8
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  xor rbx, rbx
  xor rbp, rbp
.Lloop:
  mov rdi, qword ptr [r13 + rbp*8]
  call r15
  mov qword ptr [r12 + rbp*8], rax
  mov rcx, qword ptr [r13 + rbp*8]
  cmp rax, rcx
  jge .Lnoinc
  inc rbx
.Lnoinc:
  inc rbp
  cmp rbp, r14
  jl .Lloop
  mov rax, rbx
  add rsp, 8
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbp
  pop rbx
  ret
.Lzero:
  xor rax, rax
  ret
