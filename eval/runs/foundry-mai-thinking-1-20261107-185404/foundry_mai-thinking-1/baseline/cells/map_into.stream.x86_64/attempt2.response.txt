map_into:
  cmp rdx, 0
  jle .Lzero
  push rbx
  push r12
  push r13
  push r14
  push r15
  mov r12, rdi
  mov r13, rsi
  mov r14, rdx
  mov r15, rcx
  xor rbx, rbx
  xor r11, r11
.Lloop:
  mov r10, qword ptr [r13 + r11*8]
  mov rdi, r10
  call r15
  mov qword ptr [r12 + r11*8], rax
  cmp rax, r10
  jge .Lnoinc
  inc rbx
.Lnoinc:
  inc r11
  cmp r11, r14
  jl .Lloop
  mov rax, rbx
  pop r15
  pop r14
  pop r13
  pop r12
  pop rbx
  ret
.Lzero:
  xor rax, rax
  ret
