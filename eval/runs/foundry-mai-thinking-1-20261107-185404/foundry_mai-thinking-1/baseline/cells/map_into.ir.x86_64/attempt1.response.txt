export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%is_zero =w ceql %n, 0
	jnz %is_zero, @ret_zero, @init

@init
	%i =l copy 0
	%count =l copy 0
	jmp @loop_cond

@loop_cond
	%cond_loop =w csltl %i, %n
	jnz %cond_loop, @loop_body, @end

@loop_body
	%off =l mul %i, 8
	%src_ptr =l add %src, %off
	%dst_ptr =l add %dst, %off
	%val =l loadl %src_ptr
	%res =l call %f(l %val)
	storel %res, %dst_ptr
	%is_lt =w csltl %res, %val
	jnz %is_lt, @inc_count, @next_iter

@inc_count
	%count_new =l add %count, 1
	%count =l copy %count_new
	jmp @next_iter

@next_iter
	%i_new =l add %i, 1
	%i =l copy %i_new
	jmp @loop_cond

@end
	ret %count

@ret_zero
	ret 0
}
