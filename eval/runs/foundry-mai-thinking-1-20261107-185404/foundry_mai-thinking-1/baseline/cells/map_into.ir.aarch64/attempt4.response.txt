export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%count =l copy 0
	%i =l copy 0
	%nzero =w ceql %n, 0
	jnz %nzero, @exit, @loop

@loop
	%off =l shl %i, 3
	%ps =l add %src, %off
	%pd =l add %dst, %off
	%val =l loadl %ps
	%res =l call %f(l %val)
	storel %res, %pd
	%lt =w csltl %res, %val
	jnz %lt, @inc, @next

@inc
	%count =l add %count, 1

@next
	%i =l add %i, 1
	%more =w csltl %i, %n
	jnz %more, @loop, @exit

@exit
	ret %count
}
