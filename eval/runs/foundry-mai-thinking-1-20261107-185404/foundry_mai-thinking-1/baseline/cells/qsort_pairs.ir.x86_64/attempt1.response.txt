export function l $qsort_pairs(l %a, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @return_zero
	%isone =w ceql %n, 1
	jnz %isone, @return_zero
	%pidx =l sub %n, 1
	%pi2 =l mul %pidx, 2
	%pptr_key =l add %a, %pi2
	%pptr_val =l add %pptr_key, 8
	%pkey =l loadl %pptr_key
	%pval =l loadl %pptr_val
	%i =l sub 0, 1
	%j =l 0
@loop_partition
	%jge =w csltl %j, %n
	jnz %jge, @check_record
	jmp @end_partition
@check_record
	%j2 =l mul %j, 2
	%jptr_key =l add %a, %j2
	%jptr_val =l add %jptr_key, 8
	%jkey =l loadl %jptr_key
	%jval =l loadl %jptr_val
	%jlt_pkey =w csltl %jkey, %pkey
	jnz %jlt_pkey, @do_swap
	%jeq_pkey =w ceql %jkey, %pkey
	jnz %jeq_pkey, @check_jval
	jmp @next_j
@check_jval
	%jlt_pval =w csltl %jval, %pval
	jnz %jlt_pval, @do_swap
	jmp @next_j
@do_swap
	%i =l add %i, 1
	%ii2 =l mul %i, 2
	%iptr_key =l add %a, %ii2
	%iptr_val =l add %iptr_key, 8
	%ikey =l loadl %iptr_key
	%ival =l loadl %iptr_val
	storel %iptr_key, %jkey
	storel %iptr_val, %jval
	storel %jptr_key, %ikey
	storel %jptr_val, %ival
@next_j
	%j =l add %j, 1
	jmp @loop_partition
@end_partition
	%ip1 =l add %i, 1
	%ip12 =l mul %ip1, 2
	%ip1ptr_key =l add %a, %ip12
	%ip1ptr_val =l add %ip1ptr_key, 8
	%ip1key =l loadl %ip1ptr_key
	%ip1val =l loadl %ip1ptr_val
	storel %ip1ptr_key, %pkey
	storel %ip1ptr_val, %pval
	storel %pptr_key, %ip1key
	storel %pptr_val, %ip1val
	%left_n =l %ip1
	%left_ptr =l %a
	%call_left =l call $qsort_pairs(l %left_ptr, l %left_n)
	%right_n =l sub %n, %ip1
	%right_n =l sub %right_n, 1
	%right_idx =l add %ip1, 1
	%right_i2 =l mul %right_idx, 2
	%right_ptr =l add %a, %right_i2
	%right_iszero =w ceql %right_n, 0
	jnz %right_iszero, @return_zero
	%right_isone =w ceql %right_n, 1
	jnz %right_isone, @return_zero
	%call_right =l call $qsort_pairs(l %right_ptr, l %right_n)
@return_zero
	ret 0
}
