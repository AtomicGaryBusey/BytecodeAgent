export function l $qsort_pairs(l %a, l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @return_zero
	%is_one =w ceql %n, 1
	jnz %is_one, @return_zero
	%p_idx =l sub %n, 1
	%p_i2 =l mul %p_idx, 2
	%p_key_ptr =l add %a, %p_i2
	%p_val_ptr =l add %p_key_ptr, 8
	%p_key =l loadl %p_key_ptr
	%p_val =l loadl %p_val_ptr
	%i =l sub 0, 1
	%j =l 0
@loop_part
	%j_lt_n1 =w csltl %j, %p_idx
	jnz %j_lt_n1, @check_elem
	jmp @end_part
@check_elem
	%j_i2 =l mul %j, 2
	%j_key_ptr =l add %a, %j_i2
	%j_val_ptr =l add %j_key_ptr, 8
	%j_key =l loadl %j_key_ptr
	%j_val =l loadl %j_val_ptr
	%j_lt_pk =w csltl %j_key, %p_key
	jnz %j_lt_pk, @do_swap
	%j_eq_pk =w ceql %j_key, %p_key
	jnz %j_eq_pk, @check_jv
	jmp @next_j
@check_jv
	%j_lt_pv =w csltl %j_val, %p_val
	jnz %j_lt_pv, @do_swap
	jmp @next_j
@do_swap
	%ip1_swap =l add %i, 1
	%i_i2 =l mul %ip1_swap, 2
	%i_key_ptr =l add %a, %i_i2
	%i_val_ptr =l add %i_key_ptr, 8
	%i_key =l loadl %i_key_ptr
	%i_val =l loadl %i_val_ptr
	storel %i_key_ptr, %j_key
	storel %i_val_ptr, %j_val
	storel %j_key_ptr, %i_key
	storel %j_val_ptr, %i_val
	%i =l %ip1_swap
@next_j
	%j =l add %j, 1
	jmp @loop_part
@end_part
	%ip1_final =l add %i, 1
	%ip1_i2_final =l mul %ip1_final, 2
	%ip1_key_ptr_final =l add %a, %ip1_i2_final
	%ip1_val_ptr_final =l add %ip1_key_ptr_final, 8
	%ip1_key_final =l loadl %ip1_key_ptr_final
	%ip1_val_final =l loadl %ip1_val_ptr_final
	storel %ip1_key_ptr_final, %p_key
	storel %ip1_val_ptr_final, %p_val
	storel %p_key_ptr, %ip1_key_final
	storel %p_val_ptr, %ip1_val_final
	%left_n =l %ip1_final
	%left_ptr =l %a
	%left_is_zero =w ceql %left_n, 0
	jnz %left_is_zero, @right_part
	%left_is_one =w ceql %left_n, 1
	jnz %left_is_one, @right_part
	%left_res =l call $qsort_pairs(l %left_ptr, l %left_n)
@right_part
	%right_idx_start =l add %ip1_final, 1
	%right_n =l sub %n, %right_idx_start
	%right_is_zero =w ceql %right_n, 0
	jnz %right_is_zero, @return_zero
	%right_is_one =w ceql %right_n, 1
	jnz %right_is_one, @return_zero
	%right_i2_start =l mul %right_idx_start, 2
	%right_ptr =l add %a, %right_i2_start
	%right_res =l call $qsort_pairs(l %right_ptr, l %right_n)
@return_zero
	ret 0
}
