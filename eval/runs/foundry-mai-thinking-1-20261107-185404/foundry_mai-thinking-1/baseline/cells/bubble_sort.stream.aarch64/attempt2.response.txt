bubble_sort:
  cbz x1, .Lend
  cmp x1, #1
  b.eq .Lend
  mov x9, x0
  sub x10, x1, #1
.Louter:
  mov x11, #0
.Linner:
  cmp x11, x10
  b.ge .Ldec
  lsl x12, x11, #3
  add x12, x9, x12
  ldr x13, [x12]
  ldr x14, [x12, #8]
  cmp x13, x14
  b.le .Lnoswap
  str x14, [x12]
  str x13, [x12, #8]
.Lnoswap:
  add x11, x11, #1
  b .Linner
.Ldec:
  subs x10, x10, #1
  b.gt .Louter
.Lend:
  mov x0, #0
  ret
