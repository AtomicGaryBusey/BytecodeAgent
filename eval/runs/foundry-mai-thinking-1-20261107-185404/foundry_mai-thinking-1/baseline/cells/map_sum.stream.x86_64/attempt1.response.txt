map_sum:
  test rsi, rsi
  jz .Lzero
  push rbx
  push r12
  push r13
  push r14
  mov rbx, rdi
  mov r12, rdx
  mov r13, rsi
  xor r14, r14
  xor rcx, rcx
.Lloop:
  cmp rcx, r13
  jge .Ldone
  mov rdi, qword ptr [rbx + rcx*8]
  sub rsp, 8
  call r12
  add rsp, 8
  add r14, rax
  inc rcx
  jmp .Lloop
.Ldone:
  mov rax, r14
  pop r14
  pop r13
  pop r12
  pop rbx
  ret
.Lzero:
  xor rax, rax
  ret
