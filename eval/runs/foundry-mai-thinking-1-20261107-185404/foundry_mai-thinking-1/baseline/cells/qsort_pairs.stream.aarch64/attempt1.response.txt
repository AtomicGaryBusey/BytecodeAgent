qsort_pairs:
cbz x1, .Lret_zero
cmp x1, #1
b.le .Lret_zero

stp x29, x30, [sp, #-16]!
mov x29, sp

stp x19, x20, [sp, #-16]!
stp x21, x22, [sp, #-16]!
stp x23, x24, [sp, #-16]!

mov x19, x0
mov x20, x1

sub x10, x20, #1
lsl x11, x10, #1
add x11, x19, x11
ldr x12, [x11]
ldr x13, [x11, #8]

mov x21, #-1
mov x22, #0

.Lpart_loop:
cmp x22, x10
b.ge .Lpart_done

lsl x14, x22, #1
add x14, x19, x14
ldr x15, [x14]
ldr x16, [x14, #8]

cmp x15, x12
b.lt .Lswap_candidate
b.ne .Lnext_j
cmp x16, x13
b.lt .Lswap_candidate
b .Lnext_j

.Lswap_candidate:
add x21, x21, #1
mov x23, x21
mov x24, x22
bl .Lswap_records

.Lnext_j:
add x22, x22, #1
b .Lpart_loop

.Lpart_done:
add x21, x21, #1
mov x23, x21
mov x24, x10
bl .Lswap_records

mov x0, x19
mov x1, x21
bl qsort_pairs

add x9, x21, #1
sub x1, x20, x9
lsl x9, x21, #1
add x0, x19, x9
bl qsort_pairs

.Lend:
ldp x23, x24, [sp], #16
ldp x21, x22, [sp], #16
ldp x19, x20, [sp], #16
ldp x29, x30, [sp], #16
mov x0, #0
ret

.Lret_zero:
mov x0, #0
ret

.Lswap_records:
lsl x23, x23, #1
lsl x24, x24, #1
add x5, x19, x23
add x6, x19, x24
ldp x9, x10, [x5]
ldp x11, x12, [x6]
stp x11, x12, [x5]
stp x9, x10, [x6]
ret
