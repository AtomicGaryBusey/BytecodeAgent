export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%count =l copy 0
	%i =l copy 0
	%isneg =w csltl %n, 0
	jnz %isneg, @done, @loop

@loop
	%lt =w csltl %i, %n
	jnz %lt, @body, @done

@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%gt =w cugtl %val, %t
	jnz %gt, @inc, @next

@inc
	%count =l add %count, 1

@next
	%i =l add %i, 1
	jmp @loop

@done
	ret %count
}
