export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero      = l copy 0
	%one       = l copy 1
	%three     = l copy 3
	%thirtytwo = l copy 32
	%mask32    = l copy 4294967295

	%na_zero   = w ceql %na, %zero
	%nb_zero   = w ceql %nb, %zero
	%any_zero  = w or %na_zero, %nb_zero
	jnz %any_zero, @return_zero, @compute

@return_zero
	ret %zero

@compute
	%reslen    = l add %na, %nb
	%ci        = l copy 0
	jmp @clear_cond

@clear_cond
	%ccond     = w csltl %ci, %reslen
	jnz %ccond, @clear_body, @mul_init

@clear_body
	%coff      = l shl %ci, %three
	%cptr      = l add %r, %coff
	storel %zero, %cptr
	%cnext     = l add %ci, %one
	%ci        = l copy %cnext
	jmp @clear_cond

@mul_init
	%i         = l copy 0
	jmp @i_cond

@i_cond
	%icond     = w csltl %i, %na
	jnz %icond, @i_body, @find_sig

@i_body
	%ioff      = l shl %i, %three
	%aiptr     = l add %a, %ioff
	%ai        = l loadl %aiptr
	%carry     = l copy 0
	%j         = l copy 0
	jmp @j_cond

@j_cond
	%jcond     = w csltl %j, %nb
	jnz %jcond, @j_body, @i_final

@j_body
	%joff      = l shl %j, %three
	%biptr     = l add %b, %joff
	%bj        = l loadl %biptr

	%rij       = l add %i, %j
	%roff      = l shl %rij, %three
	%rptr      = l add %r, %roff
	%rold      = l loadl %rptr

	%ai0       = l and %ai, %mask32
	%ai1       = l shr %ai, %thirtytwo
	%bj0       = l and %bj, %mask32
	%bj1       = l shr %bj, %thirtytwo

	%m00       = l mul %ai0, %bj0
	%m01       = l mul %ai0, %bj1
	%m10       = l mul %ai1, %bj0
	%m11       = l mul %ai1, %bj1

	%m01shl    = l shl %m01, %thirtytwo
	%m10shl    = l shl %m10, %thirtytwo

	%s_low1    = l add %m00, %m01shl
	%c_low1_w  = w cultl %s_low1, %m00
	%c_low1    = l extsw %c_low1_w

	%s_low2    = l add %s_low1, %m10shl
	%c_low2_w  = w cultl %s_low2, %s_low1
	%c_low2    = l extsw %c_low2_w

	%mid_carry = l add %c_low1, %c_low2

	%m01shr    = l shr %m01, %thirtytwo
	%m10shr    = l shr %m10, %thirtytwo

	%mid_sum1  = l add %m01shr, %m10shr
	%c_ms1_w   = w cultl %mid_sum1, %m01shr
	%c_ms1     = l extsw %c_ms1_w

	%mid_sum2  = l add %mid_sum1, %mid_carry
	%c_ms2_w   = w cultl %mid_sum2, %mid_sum1
	%c_ms2     = l extsw %c_ms2_w

	%mid_h_c   = l add %c_ms1, %c_ms2

	%prod_high = l add %m11, %mid_h_c
	%prod_low  = l copy %s_low2

	%s_acc1    = l add %prod_low, %rold
	%c_a1_w    = w cultl %s_acc1, %prod_low
	%c_a1      = l extsw %c_a1_w

	%s_acc2    = l add %s_acc1, %carry
	%c_a2_w    = w cultl %s_acc2, %s_acc1
	%c_a2      = l extsw %c_a2_w

	%acc_c     = l add %c_a1, %c_a2

	%new_carry = l add %prod_high, %acc_c

	storel %s_acc2, %rptr

	%carry     = l copy %new_carry

	%jnext     = l add %j, %one
	%j         = l copy %jnext
	jmp @j_cond

@i_final
	%rifin     = l add %i, %nb
	%rifoff    = l shl %rifin, %three
	%rpfptr    = l add %r, %rifoff
	storel %carry, %rpfptr

	%inext     = l add %i, %one
	%i         = l copy %inext
	jmp @i_cond

@find_sig
	%k         = l copy %reslen
	jmp @sig_cond

@sig_cond
	%kgt0      = w csgtl %k, %zero
	jnz %kgt0, @sig_check, @return_zero

@sig_check
	%kidx      = l sub %k, %one
	%koff      = l shl %kidx, %three
	%rkptr     = l add %r, %koff
	%rkh       = l loadl %rkptr
	%rkz       = w ceql %rkh, %zero
	jnz %rkz, @sig_dec, @sig_ret

@sig_dec
	%knext     = l sub %k, %one
	%k         = l copy %knext
	jmp @sig_cond

@sig_ret
	ret %k
}
