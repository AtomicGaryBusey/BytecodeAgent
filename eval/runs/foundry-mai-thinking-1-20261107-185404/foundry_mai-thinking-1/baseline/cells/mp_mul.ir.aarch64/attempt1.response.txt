export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l copy 0
	%one  =l copy 1
	%two  =l copy 2
	%four =l copy 4
	%eight= l copy 8
	%thirtytwo = l copy 32
	%mask32    = l copy 4294967295

	%is_na_zero = w ceql %na, %zero
	%is_nb_zero = w ceql %nb, %zero
	%or_zero    = w or %is_na_zero, %is_nb_zero
	jnz %or_zero, @zero_result, @alloc_reslen

@alloc_reslen
	%reslen = l add %na, %nb
	jnz %reslen, @zero_loop_init, @zero_result

@zero_loop_init
	%zi = l copy 0
	jmp @zero_loop_cond

@zero_loop_cond
	%zcond = w csltl %zi, %reslen
	jnz %zcond, @zero_loop_body, @mul_init

@zero_loop_body
	%zptr = l add %r, %zi
	%zptrb= l shl %zptr, %zero
	storel %zero, %zptrb
	%zinext = l add %zi, %one
	%zi = l copy %zinext
	jmp @zero_loop_cond

@mul_init
	%i = l copy 0
	jmp @i_cond

@i_cond
	%icond = w csltl %i, %na
	jnz %icond, @i_body, @find_sig

@i_body
	%aiptr = l add %a, %i
	%ai    = l loadl %aiptr
	%carry = l copy 0
	%j     = l copy 0
	jmp @j_cond

@j_cond
	%jcond = w csltl %j, %nb
	jnz %jcond, @j_body, @i_final_carry

@j_body
	%biptr = l add %b, %j
	%bj    = l loadl %biptr
	%rij   = l add %i, %j
	%rptr  = l add %r, %rij
	%rold  = l loadl %rptr

	%ai0   = l and %ai, %mask32
	%ai1   = l shr %ai, %thirtytwo
	%bj0   = l and %bj, %mask32
	%bj1   = l shr %bj, %thirtytwo

	%m00   = l mul %ai0, %bj0
	%m01   = l mul %ai0, %bj1
	%m10   = l mul %ai1, %bj0
	%m11   = l mul %ai1, %bj1

	%m01s  = l shl %m01, %thirtytwo
	%m10s  = l shl %m10, %thirtytwo

	%s_low1= l add %m00, %m01s
	%clow1 = w cultl %s_low1, %m00
	%clow1l= l extsw %clow1

	%s_low2= l add %s_low1, %m10s
	%clow2 = w cultl %s_low2, %s_low1
	%clow2l= l extsw %clow2

	%mid_carry = l add %clow1l, %clow2l

	%m01sh = l shr %m01, %thirtytwo
	%m10sh = l shr %m10, %thirtytwo
	%m11h  = l add %m01sh, %m10sh
	%cmid1 = w cultl %m11h, %m01sh
	%cmid1l= l extsw %cmid1
	%m11h2 = l add %m11h, %mid_carry
	%cmid2 = w cultl %m11h2, %m11h
	%cmid2l= l extsw %cmid2
	%mid_h = l add %cmid1l, %cmid2l
	%hi128 = l add %m11, %mid_h

	%prod_low  = l copy %s_low2
	%prod_high = l copy %hi128

	%s1    = l add %prod_low, %rold
	%c1    = w cultl %s1, %prod_low
	%c1l   = l extsw %c1

	%s2    = l add %s1, %carry
	%c2    = w cultl %s2, %s1
	%c2l   = l extsw %c2

	%new_carry_mid = l add %c1l, %c2l
	%new_carry_hi  = l add %prod_high, %new_carry_mid

	%rnew  = l copy %s2
	storel %rnew, %rptr

	%carry_next = l copy %new_carry_hi
	%carry = l copy %carry_next

	%jnext = l add %j, %one
	%j     = l copy %jnext
	jmp @j_cond

@i_final_carry
	%rifin = l add %i, %nb
	%rpf   = l add %r, %rifin
	storel %carry, %rpf
	%inext = l add %i, %one
	%i     = l copy %inext
	jmp @i_cond

@zero_result
	%ret0  = l copy 0
	ret %ret0

@find_sig
	%k     = l copy %reslen
	jmp @sig_cond

@sig_cond
	%kgt0  = w csgtl %k, %zero
	jnz %kgt0, @sig_check, @ret_one

@sig_check
	%kidx  = l sub %k, %one
	%rkptr = l add %r, %kidx
	%rkh   = l loadl %rkptr
	%rkz   = w ceql %rkh, %zero
	jnz %rkz, @sig_dec, @sig_ret_k

@sig_dec
	%knext = l sub %k, %one
	%k     = l copy %knext
	jmp @sig_cond

@sig_ret_k
	ret %k

@ret_one
	%ret1  = l copy 1
	ret %ret1
}
