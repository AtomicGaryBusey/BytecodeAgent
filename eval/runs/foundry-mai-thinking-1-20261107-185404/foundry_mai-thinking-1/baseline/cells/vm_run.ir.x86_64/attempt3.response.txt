export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	%cnt =l copy 0
	jmp @loop

@loop
	%pclt =w csltl %pc, %len
	jnz %pclt, @fetch, @halt

@fetch
	%pcoff =l shl %pc, 3
	%codeptr =l add %code, %pcoff
	%op =l loadl %codeptr
	%cntnew =l add %cnt, 1
	%cnt =l copy %cntnew
	%cntlim =w ceql %cnt, 5000
	jnz %cntlim, @halt, @dispatch

@dispatch
	%op0 =w ceql %op, 0
	jnz %op0, @halt, @d1
@d1
	%op1 =w ceql %op, 1
	jnz %op1, @push, @d2
@d2
	%op2 =w ceql %op, 2
	jnz %op2, @add, @d3
@d3
	%op3 =w ceql %op, 3
	jnz %op3, @sub, @d4
@d4
	%op4 =w ceql %op, 4
	jnz %op4, @mul, @d5
@d5
	%op5 =w ceql %op, 5
	jnz %op5, @dup, @d6
@d6
	%op6 =w ceql %op, 6
	jnz %op6, @swap, @d7
@d7
	%op7 =w ceql %op, 7
	jnz %op7, @drop, @d8
@d8
	%op8 =w ceql %op, 8
	jnz %op8, @over, @d9
@d9
	%op9 =w ceql %op, 9
	jnz %op9, @lt, @d10
@d10
	%op10 =w ceql %op, 10
	jnz %op10, @jz, @d11
@d11
	%op11 =w ceql %op, 11
	jnz %op11, @jmp, @default

@push
	%pc1 =l add %pc, 1
	%p1off =l shl %pc1, 3
	%argptr =l add %code, %p1off
	%v =l loadl %argptr
	%spnew =l add %sp, 1
	%sp =l copy %spnew
	%spoff =l shl %sp, 3
	%stoptr =l add %stack, %spoff
	storel %v, %stoptr
	%pcnew =l add %pc, 2
	%pc =l copy %pcnew
	jmp @loop

@add
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%spboff =l shl %spb, 3
	%spaoff =l shl %spa, 3
	%bptr =l add %stack, %spboff
	%aptr =l add %stack, %spaoff
	%b =l loadl %bptr
	%a =l loadl %aptr
	%res =l add %a, %b
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@sub
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%spboff =l shl %spb, 3
	%spaoff =l shl %spa, 3
	%bptr =l add %stack, %spboff
	%aptr =l add %stack, %spaoff
	%b =l loadl %bptr
	%a =l loadl %aptr
	%res =l sub %a, %b
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@mul
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%spboff =l shl %spb, 3
	%spaoff =l shl %spa, 3
	%bptr =l add %stack, %spboff
	%aptr =l add %stack, %spaoff
	%b =l loadl %bptr
	%a =l loadl %aptr
	%res =l mul %a, %b
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@dup
	%sp0 =l add %sp, -1
	%sp0off =l shl %sp0, 3
	%tptr =l add %stack, %sp0off
	%vtop =l loadl %tptr
	%spnew =l add %sp, 1
	%sp =l copy %spnew
	%spnoff =l shl %sp, 3
	%nptr =l add %stack, %spnoff
	storel %vtop, %nptr
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@swap
	%sp0 =l add %sp, -1
	%sp1 =l add %sp, -2
	%sp0off =l shl %sp0, 3
	%sp1off =l shl %sp1, 3
	%p0 =l add %stack, %sp0off
	%p1 =l add %stack, %sp1off
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	storel %v0, %p1
	storel %v1, %p0
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@drop
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@over
	%spb =l add %sp, -2
	%spboff =l shl %spb, 3
	%bptr =l add %stack, %spboff
	%v =l loadl %bptr
	%spnew =l add %sp, 1
	%sp =l copy %spnew
	%spnoff =l shl %sp, 3
	%nptr =l add %stack, %spnoff
	storel %v, %nptr
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@lt
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%spboff =l shl %spb, 3
	%spaoff =l shl %spa, 3
	%bptr =l add %stack, %spboff
	%aptr =l add %stack, %spaoff
	%b =l loadl %bptr
	%a =l loadl %aptr
	%c =w csltl %a, %b
	%res =l extuw %c
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@jz
	%sp0 =l add %sp, -1
	%sp0off =l shl %sp0, 3
	%vptr =l add %stack, %sp0off
	%v =l loadl %vptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%iszero =w ceql %v, 0
	jnz %iszero, @jzt, @jzf
@jzt
	%pc1 =l add %pc, 1
	%p1off =l shl %pc1, 3
	%tptr =l add %code, %p1off
	%t =l loadl %tptr
	%pc =l copy %t
	jmp @loop
@jzf
	%pcnew =l add %pc, 2
	%pc =l copy %pcnew
	jmp @loop

@jmp
	%pc1 =l add %pc, 1
	%p1off =l shl %pc1, 3
	%tptr =l add %code, %p1off
	%t =l loadl %tptr
	%pc =l copy %t
	jmp @loop

@default
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@halt
	jmp @end_clear

@end_clear
	%d =l copy %sp
	%j =l copy %d
@clear_loop
	%jge =w csltl %j, %cap
	jnz %jge, @clear_body, @return_top
@clear_body
	%joff =l shl %j, 3
	%zptr =l add %stack, %joff
	storel 0, %zptr
	%jnew =l add %j, 1
	%j =l copy %jnew
	jmp @clear_loop

@return_top
	%has =w csltl 0, %sp
	jnz %has, @ret_topv, @ret_zero
@ret_topv
	%sp0r =l add %sp, -1
	%sp0roff =l shl %sp0r, 3
	%tpr =l add %stack, %sp0roff
	%rv =l loadl %tpr
	ret %rv
@ret_zero
	ret 0
}
