export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	%cnt =l copy 0
	jmp @loop

@loop
	%pclt =l csltl %pc, %len
	jnz %pclt, @fetch, @halt

@fetch
	%codeptr =l add %code, %pc
	%codeptr =l shl %codeptr, 3
	%codeptr =l add %code, %codeptr
	%op =l loadl %codeptr
	%cntnew =l add %cnt, 1
	%cnt =l copy %cntnew
	%cntlim =l ceql %cnt, 5000
	jnz %cntlim, @halt, @dispatch

@dispatch
	%op0 =l ceql %op, 0
	jnz %op0, @halt, @d1
@d1
	%op1 =l ceql %op, 1
	jnz %op1, @push, @d2
@d2
	%op2 =l ceql %op, 2
	jnz %op2, @add, @d3
@d3
	%op3 =l ceql %op, 3
	jnz %op3, @sub, @d4
@d4
	%op4 =l ceql %op, 4
	jnz %op4, @mul, @d5
@d5
	%op5 =l ceql %op, 5
	jnz %op5, @dup, @d6
@d6
	%op6 =l ceql %op, 6
	jnz %op6, @swap, @d7
@d7
	%op7 =l ceql %op, 7
	jnz %op7, @drop, @d8
@d8
	%op8 =l ceql %op, 8
	jnz %op8, @over, @d9
@d9
	%op9 =l ceql %op, 9
	jnz %op9, @lt, @d10
@d10
	%op10 =l ceql %op, 10
	jnz %op10, @jz, @d11
@d11
	%op11 =l ceql %op, 11
	jnz %op11, @jmp, @default

@push
	%pc1 =l add %pc, 1
	%argptr =l add %code, %pc1
	%argptr =l shl %argptr, 3
	%argptr =l add %code, %argptr
	%v =l loadl %argptr
	%spnew =l add %sp, 1
	%sp =l copy %spnew
	%stoptr =l add %stack, %sp
	%stoptr =l shl %stoptr, 3
	%stoptr =l add %stack, %stoptr
	storel %v, %stoptr
	%pcnew =l add %pc, 2
	%pc =l copy %pcnew
	jmp @loop

@add
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%bptr =l add %stack, %spb
	%bptr =l shl %bptr, 3
	%bptr =l add %stack, %bptr
	%aptr =l add %stack, %spa
	%aptr =l shl %aptr, 3
	%aptr =l add %stack, %aptr
	%b =l loadl %bptr
	%a =l loadl %aptr
	%res =l add %a, %b
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@sub
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%bptr =l add %stack, %spb
	%bptr =l shl %bptr, 3
	%bptr =l add %stack, %bptr
	%aptr =l add %stack, %spa
	%aptr =l shl %aptr, 3
	%aptr =l add %stack, %aptr
	%b =l loadl %bptr
	%a =l loadl %aptr
	%res =l sub %a, %b
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@mul
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%bptr =l add %stack, %spb
	%bptr =l shl %bptr, 3
	%bptr =l add %stack, %bptr
	%aptr =l add %stack, %spa
	%aptr =l shl %aptr, 3
	%aptr =l add %stack, %aptr
	%b =l loadl %bptr
	%a =l loadl %aptr
	%res =l mull %a, %b
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@dup
	%sp0 =l add %sp, -1
	%tptr =l add %stack, %sp0
	%tptr =l shl %tptr, 3
	%tptr =l add %stack, %tptr
	%vtop =l loadl %tptr
	%spnew =l add %sp, 1
	%sp =l copy %spnew
	%nptr =l add %stack, %sp
	%nptr =l shl %nptr, 3
	%nptr =l add %stack, %nptr
	storel %vtop, %nptr
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@swap
	%sp0 =l add %sp, -1
	%sp1 =l add %sp, -2
	%p0 =l add %stack, %sp0
	%p0 =l shl %p0, 3
	%p0 =l add %stack, %p0
	%p1 =l add %stack, %sp1
	%p1 =l shl %p1, 3
	%p1 =l add %stack, %p1
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	storel %v0, %p1
	storel %v1, %p0
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@drop
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@over
	%spb =l add %sp, -2
	%bptr =l add %stack, %spb
	%bptr =l shl %bptr, 3
	%bptr =l add %stack, %bptr
	%v =l loadl %bptr
	%spnew =l add %sp, 1
	%sp =l copy %spnew
	%nptr =l add %stack, %sp
	%nptr =l shl %nptr, 3
	%nptr =l add %stack, %nptr
	storel %v, %nptr
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@lt
	%spb =l add %sp, -1
	%spa =l add %sp, -2
	%bptr =l add %stack, %spb
	%bptr =l shl %bptr, 3
	%bptr =l add %stack, %bptr
	%aptr =l add %stack, %spa
	%aptr =l shl %aptr, 3
	%aptr =l add %stack, %aptr
	%b =l loadl %bptr
	%a =l loadl %aptr
	%c =w csltl %a, %b
	%res =l extw %c
	storel %res, %aptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@jz
	%sp0 =l add %sp, -1
	%vptr =l add %stack, %sp0
	%vptr =l shl %vptr, 3
	%vptr =l add %stack, %vptr
	%v =l loadl %vptr
	%spnew =l add %sp, -1
	%sp =l copy %spnew
	%iszero =l ceql %v, 0
	jnz %iszero, @jzt, @jzf
@jzt
	%tptr =l add %pc, 1
	%tptr =l add %code, %tptr
	%tptr =l shl %tptr, 3
	%tptr =l add %code, %tptr
	%t =l loadl %tptr
	%pc =l copy %t
	jmp @loop
@jzf
	%pcnew =l add %pc, 2
	%pc =l copy %pcnew
	jmp @loop

@jmp
	%tptr =l add %pc, 1
	%tptr =l add %code, %tptr
	%tptr =l shl %tptr, 3
	%tptr =l add %code, %tptr
	%t =l loadl %tptr
	%pc =l copy %t
	jmp @loop

@default
	%pcnew =l add %pc, 1
	%pc =l copy %pcnew
	jmp @loop

@halt
	jmp @end_clear

@end_clear
	%d =l copy %sp
	%j =l copy %d
@clear_loop
	%jge =l csltl %j, %cap
	jnz %jge, @clear_body, @return_top
@clear_body
	%zptr =l add %stack, %j
	%zptr =l shl %zptr, 3
	%zptr =l add %stack, %zptr
	storel 0, %zptr
	%jnew =l add %j, 1
	%j =l copy %jnew
	jmp @clear_loop

@return_top
	%has =l csltl 0, %sp
	jnz %has, @ret_topv, @ret_zero
@ret_topv
	%sp0r =l add %sp, -1
	%tpr =l add %stack, %sp0r
	%tpr =l shl %tpr, 3
	%tpr =l add %stack, %tpr
	%rv =l loadl %tpr
	ret %rv
@ret_zero
	ret 0
}
