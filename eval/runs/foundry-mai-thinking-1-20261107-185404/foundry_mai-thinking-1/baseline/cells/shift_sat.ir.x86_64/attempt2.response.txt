export function l $shift_sat(l %x, l %n) {
@start
	%isnlt64 = csltl %n, $64
	jnz %isnlt64, @left_small, @left_zero
@left_small
	%leftv = shll %x, %n
	jmp @right_check
@left_zero
	%leftv = $0
@right_check
	jnz %isnlt64, @right_small, @right_large
@right_small
	%rightv = sarl %x, %n
	jmp @add_res
@right_large
	%isxneg = csltl %x, $0
	jnz %isxneg, @right_m1, @right_zero
@right_m1
	%rightv = $-1
	jmp @add_res
@right_zero
	%rightv = $0
@add_res
	%res = addl %leftv, %rightv
	ret %res
}
