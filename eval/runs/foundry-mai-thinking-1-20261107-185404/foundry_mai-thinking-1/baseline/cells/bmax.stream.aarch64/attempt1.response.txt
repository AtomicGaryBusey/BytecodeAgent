bmax:
  ldr x2, [x0]        // load first element as max
  mov x3, #1          // initialize index to 1
.Lloop:
  cmp x3, x1
  b.ge .Lend          // if index >= n, exit
  ldr x4, [x0, x3, lsl #3]  // load current element
  cmp x4, x2
  b.le .Lnext         // if not greater, skip update
  mov x2, x4          // update max
.Lnext:
  add x3, x3, #1      // increment index
  b .Lloop
.Lend:
  mov x0, x2          // return max
  ret
