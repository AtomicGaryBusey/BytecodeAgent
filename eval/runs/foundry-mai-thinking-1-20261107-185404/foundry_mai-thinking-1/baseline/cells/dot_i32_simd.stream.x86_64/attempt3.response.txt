dot_i32_simd:
  xor rax, rax
  test rdx, rdx
  jle .Lend
  mov rcx, rdx
  shr rcx, 2
  jz .Lscalar
  pxor xmm0, xmm0
.Lvec4:
  movdqu xmm1, xmmword ptr [rdi]
  movdqu xmm2, xmmword ptr [rsi]
  movdqa xmm3, xmm1
  movdqa xmm4, xmm2
  psrldq xmm3, 8
  psrldq xmm4, 8
  pmovsxdq xmm1, xmm1
  pmovsxdq xmm3, xmm3
  pmovsxdq xmm2, xmm2
  pmovsxdq xmm4, xmm4
  pmuldq xmm1, xmm2
  pmuldq xmm3, xmm4
  paddq xmm0, xmm1
  paddq xmm0, xmm3
  add rdi, 16
  add rsi, 16
  dec rcx
  jnz .Lvec4
  movhlps xmm1, xmm0
  paddq xmm0, xmm1
  movq rax, xmm0
.Lscalar:
  mov rcx, rdx
  and rcx, 3
  jz .Lend
.Lsloop:
  movsxd r8, dword ptr [rdi]
  movsxd r9, dword ptr [rsi]
  imul r8, r9
  add rax, r8
  add rdi, 4
  add rsi, 4
  dec rcx
  jnz .Lsloop
.Lend:
  ret
