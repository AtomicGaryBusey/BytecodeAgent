export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 0xFFFFFFFF
	%i =l copy 0
	jmp @outer_check

@outer_check
	%cmp_n =w csltl %i, %n
	jnz %cmp_n, @outer_body, @done_outer

@outer_body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc1 =w xor %crc, %b
	%crc =w copy %crc1

	# bit 0
	%bit0 =w and %crc, 1
	%sh0 =w shr %crc, 1
	%is1_0 =w ceqw %bit0, 1
	%xorv0 =w xor %sh0, 0xEDB88320
	%crc_b0 =w csel %is1_0, %xorv0, %sh0
	%crc =w copy %crc_b0

	# bit 1
	%bit1 =w and %crc, 1
	%sh1 =w shr %crc, 1
	%is1_1 =w ceqw %bit1, 1
	%xorv1 =w xor %sh1, 0xEDB88320
	%crc_b1 =w csel %is1_1, %xorv1, %sh1
	%crc =w copy %crc_b1

	# bit 2
	%bit2 =w and %crc, 1
	%sh2 =w shr %crc, 1
	%is1_2 =w ceqw %bit2, 1
	%xorv2 =w xor %sh2, 0xEDB88320
	%crc_b2 =w csel %is1_2, %xorv2, %sh2
	%crc =w copy %crc_b2

	# bit 3
	%bit3 =w and %crc, 1
	%sh3 =w shr %crc, 1
	%is1_3 =w ceqw %bit3, 1
	%xorv3 =w xor %sh3, 0xEDB88320
	%crc_b3 =w csel %is1_3, %xorv3, %sh3
	%crc =w copy %crc_b3

	# bit 4
	%bit4 =w and %crc, 1
	%sh4 =w shr %crc, 1
	%is1_4 =w ceqw %bit4, 1
	%xorv4 =w xor %sh4, 0xEDB88320
	%crc_b4 =w csel %is1_4, %xorv4, %sh4
	%crc =w copy %crc_b4

	# bit 5
	%bit5 =w and %crc, 1
	%sh5 =w shr %crc, 1
	%is1_5 =w ceqw %bit5, 1
	%xorv5 =w xor %sh5, 0xEDB88320
	%crc_b5 =w csel %is1_5, %xorv5, %sh5
	%crc =w copy %crc_b5

	# bit 6
	%bit6 =w and %crc, 1
	%sh6 =w shr %crc, 1
	%is1_6 =w ceqw %bit6, 1
	%xorv6 =w xor %sh6, 0xEDB88320
	%crc_b6 =w csel %is1_6, %xorv6, %sh6
	%crc =w copy %crc_b6

	# bit 7
	%bit7 =w and %crc, 1
	%sh7 =w shr %crc, 1
	%is1_7 =w ceqw %bit7, 1
	%xorv7 =w xor %sh7, 0xEDB88320
	%crc_b7 =w csel %is1_7, %xorv7, %sh7
	%crc =w copy %crc_b7

	%i_next =l add %i, 1
	%i =l copy %i_next
	jmp @outer_check

@done_outer
	%crc_res =w xor %crc, 0xFFFFFFFF
	%ret =l extuw %crc_res
	ret %ret
}
