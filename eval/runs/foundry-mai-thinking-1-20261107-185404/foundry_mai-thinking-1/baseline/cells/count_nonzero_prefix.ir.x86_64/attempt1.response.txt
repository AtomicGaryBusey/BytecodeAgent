export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
	%check =w csltl %i, %n
	jnz %check, @loop, @return_n

@loop
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%iszero =w ceql %val, 0
	jnz %iszero, @return_i, @next

@next
	%i =l add %i, 1
	%continue =w csltl %i, %n
	jnz %continue, @loop, @return_n

@return_i
	ret %i

@return_n
	ret %n
}
