# rdi = p, rsi = n, rdx = seed
mov rax, rsi
cmp rax, 32
jl .Lsmall_n
# v1 = seed + P1 + P2
mov r8, rdx
mov r9, 0x9E3779B185EBCA87
add r8, r9
mov r9, 0xC2B2AE3D27D4EB4F
add r8, r9
# v2 = seed + P2
mov r10, rdx
mov r9, 0xC2B2AE3D27D4EB4F
add r10, r9
# v3 = seed
mov r11, rdx
# v4 = seed - P1
mov r12, rdx
mov r9, 0x9E3779B185EBCA87
sub r12, r9
# q = p
mov rbx, rdi
# end = p + n
lea r13, [rdi + rax]
# loop over 32-byte stripes
.Lstripe_loop:
lea r14, [rbx + 32]
cmp r14, r13
jg .Lstripe_end
