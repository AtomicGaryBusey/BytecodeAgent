reverse:
  xor rax, rax
  cmp rsi, 1
  jle .Ldone
  mov rcx, 0
  lea rdx, [rsi - 1]
.Lloop:
  cmp rcx, rdx
  jge .Ldone
  mov r8, qword ptr [rdi + rcx*8]
  mov r9, qword ptr [rdi + rdx*8]
  mov qword ptr [rdi + rcx*8], r9
  mov qword ptr [rdi + rdx*8], r8
  inc rcx
  dec rdx
  jmp .Lloop
.Ldone:
  ret
