export function l $map_sum(l %a, l %n, l %f) {
@entry
	%zero =l copy 0
	%iszero =w ceql %n, %zero
	jnz %iszero, @retzero, @init

@init
	%sum =l copy 0
	%i =l copy 0
	jmp @cond

@cond
	%less =w csltl %i, %n
	jnz %less, @body, @ret

@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%res =l call %f(l %val)
	%sum_new =l add %sum, %res
	%i_new =l add %i, 1
	%sum =l copy %sum_new
	%i =l copy %i_new
	jmp @cond

@retzero
	ret %zero

@ret
	ret %sum
}
