export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%iszero = ceql %n, 0
	jnz %iszero, @done
	%isless = csltl %dst, %src
	jnz %isless, @forward
	%iseq = ceql %dst, %src
	jnz %iseq, @done
	jmp @backward

@forward
	%i = 0
	jmp @fcond

@fcond
	%okf = csltl %i, %n
	jnz %okf, @fbody
	jmp @done

@fbody
	%off = mul %i, 8
	%sptr = add %src, %off
	%dptr = add %dst, %off
	%val = loadl %sptr
	storel %val, %dptr
	%inext = add %i, 1
	%i = %inext
	jmp @fcond

@backward
	%i = sub %n, 1
	jmp @bcond

@bcond
	%okb = csgeql %i, 0
	jnz %okb, @bbody
	jmp @done

@bbody
	%off = mul %i, 8
	%sptr = add %src, %off
	%dptr = add %dst, %off
	%val = loadl %sptr
	storel %val, %dptr
	%inext = sub %i, 1
	%i = %inext
	jmp @bcond

@done
	ret 0
}
