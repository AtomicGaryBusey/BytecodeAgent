export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%isz = ceql %n, 0
	jnz %isz, @done
	%c1 = csltl %dst, %src
	jnz %c1, @forward
	%c2 = ceql %dst, %src
	jnz %c2, @done
	jmp @backward

@forward
	%i = 0
	jmp @fcond

@fcond
	%ok = csltl %i, %n
	jnz %ok, @fbody
	jmp @done

@fbody
	%off = mul %i, 8
	%sp = add %src, %off
	%dp = add %dst, %off
	%v = loadl %sp
	storel %v, %dp
	%i2 = add %i, 1
	%i = %i2
	jmp @fcond

@backward
	%i = sub %n, 1
	jmp @bcond

@bcond
	%neg = csltl %i, 0
	jnz %neg, @done
	jmp @bbody

@bbody
	%off = mul %i, 8
	%sp = add %src, %off
	%dp = add %dst, %off
	%v = loadl %sp
	storel %v, %dp
	%i2 = sub %i, 1
	%i = %i2
	jmp @bcond

@done
	ret 0
}
