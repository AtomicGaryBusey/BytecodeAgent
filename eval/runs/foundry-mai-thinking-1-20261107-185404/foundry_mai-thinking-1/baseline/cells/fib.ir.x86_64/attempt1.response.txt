export function l $fib(l %n) {
@entry
	%is_zero =w ceql %n, 0
	jnz %is_zero, @ret_zero, @check_one
@ret_zero
	ret 0
@check_one
	%is_one =w ceql %n, 1
	jnz %is_one, @ret_one, @loop_init
@ret_one
	ret 1
@loop_init
	%prev2 =l copy 0
	%prev1 =l copy 1
	%i =l copy 2
@loop_body
	%cur =l add %prev1, %prev2
	%prev2 =l copy %prev1
	%prev1 =l copy %cur
	%i =l add %i, 1
	%cont =w csleql %i, %n
	jnz %cont, @loop_body, @end
@end
	ret %prev1
}
