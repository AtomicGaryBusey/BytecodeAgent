export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc = copy 0
	%cnt = copy 0
	jmp @loop

@loop
	%pcl = copy %pc
	%cndloop = csltl %pcl, %len
	jnz %cndloop, @fetch, @exit

@fetch
	%ci = mul %pc, 8
	%insl = loadl %code + %ci
	%op0 = ceql %insl, 0
	jnz %op0, @halt, @check1

@check1
	%op1 = ceql %insl, 1
	jnz %op1, @push, @check2

@check2
	%op2 = ceql %insl, 2
	jnz %op2, @add, @check3

@check3
	%op3 = ceql %insl, 3
	jnz %op3, @sub, @check4

@check4
	%op4 = ceql %insl, 4
	jnz %op4, @mul, @check5

@check5
	%op5 = ceql %insl, 5
	jnz %op5, @dup, @check6

@check6
	%op6 = ceql %insl, 6
	jnz %op6, @swap, @check7

@check7
	%op7 = ceql %insl, 7
	jnz %op7, @drop, @check8

@check8
	%op8 = ceql %insl, 8
	jnz %op8, @over, @check9

@check9
	%op9 = ceql %insl, 9
	jnz %op9, @lt, @check10

@check10
	%op10 = ceql %insl, 10
	jnz %op10, @jz, @check11

@check11
	%op11 = ceql %insl, 11
	jnz %op11, @jmp, @inc1

@halt
	jmp @exit

@push
	%vaddr = add %ci, 8
	%vpush = loadl %code + %vaddr
	%pidxpush = mul %cnt, 8
	storel %vpush, %stack + %pidxpush
	%ncpush = add %cnt, 1
	%npcpush = add %pc, 2
	%cnt = copy %ncpush
	%pc = copy %npcpush
	jmp @loop

@add
	%c1add = sub %cnt, 1
	%c2add = sub %cnt, 2
	%i1add = mul %c1add, 8
	%i2add = mul %c2add, 8
	%badd = loadl %stack + %i1add
	%aadd = loadl %stack + %i2add
	%radd = add %aadd, %badd
	storel %radd, %stack + %i2add
	%ncadd = copy %c2add
	%npcadd = add %pc, 1
	%cnt = copy %ncadd
	%pc = copy %npcadd
	jmp @loop

@sub
	%c1sub = sub %cnt, 1
	%c2sub = sub %cnt, 2
	%i1sub = mul %c1sub, 8
	%i2sub = mul %c2sub, 8
	%bsub = loadl %stack + %i1sub
	%asub = loadl %stack + %i2sub
	%rsub = sub %asub, %bsub
	storel %rsub, %stack + %i2sub
	%ncsub = copy %c2sub
	%npcsub = add %pc, 1
	%cnt = copy %ncsub
	%pc = copy %npcsub
	jmp @loop

@mul
	%c1mul = sub %cnt, 1
	%c2mul = sub %cnt, 2
	%i1mul = mul %c1mul, 8
	%i2mul = mul %c2mul, 8
	%bmul = loadl %stack + %i1mul
	%amul = loadl %stack + %i2mul
	%rmul = mul %amul, %bmul
	storel %rmul, %stack + %i2mul
	%ncmul = copy %c2mul
	%npcmul = add %pc, 1
	%cnt = copy %ncmul
	%pc = copy %npcmul
	jmp @loop

@dup
	%c1dup = sub %cnt, 1
	%i1dup = mul %c1dup, 8
	%topvdup = loadl %stack + %i1dup
	%i2ndup = mul %cnt, 8
	storel %topvdup, %stack + %i2ndup
	%ncdup = add %cnt, 1
	%npcdup = add %pc, 1
	%cnt = copy %ncdup
	%pc = copy %npcdup
	jmp @loop

@swap
	%c1sw = sub %cnt, 1
	%c2sw = sub %cnt, 2
	%i1sw = mul %c1sw, 8
	%i2sw = mul %c2sw, 8
	%tsw = loadl %stack + %i1sw
	%ssw = loadl %stack + %i2sw
	storel %tsw, %stack + %i2sw
	storel %ssw, %stack + %i1sw
	%npcsw = add %pc, 1
	%pc = copy %npcsw
	jmp @loop

@drop
	%ncdrop = sub %cnt, 1
	%npcdrop = add %pc, 1
	%cnt = copy %ncdrop
	%pc = copy %npcdrop
	jmp @loop

@over
	%c1ov = sub %cnt, 1
	%c2ov = sub %cnt, 2
	%i2ov = mul %c2ov, 8
	%ovv = loadl %stack + %i2ov
	%inov = mul %cnt, 8
	storel %ovv, %stack + %inov
	%ncov = add %cnt, 1
	%npcov = add %pc, 1
	%cnt = copy %ncov
	%pc = copy %npcov
	jmp @loop

@lt
	%c1lt = sub %cnt, 1
	%c2lt = sub %cnt, 2
	%i1lt = mul %c1lt, 8
	%i2lt = mul %c2lt, 8
	%blt = loadl %stack + %i1lt
	%alt = loadl %stack + %i2lt
	%clt = csltl %alt, %blt
	%rlt = extl %clt
	storel %rlt, %stack + %i2lt
	%nclt = copy %c2lt
	%npclt = add %pc, 1
	%cnt = copy %nclt
	%pc = copy %npclt
	jmp @loop

@jz
	%c1jz = sub %cnt, 1
	%i1jz = mul %c1jz, 8
	%vjz = loadl %stack + %i1jz
	%ncjz = copy %c1jz
	%v0jz = ceql %vjz, 0
	jnz %v0jz, @jzt, @jzf

@jzt
	%taddrjz = add %ci, 8
	%tjz = loadl %code + %taddrjz
	%pc = copy %tjz
	%cnt = copy %ncjz
	jmp @loop

@jzf
	%npcjz = add %pc, 2
	%pc = copy %npcjz
	%cnt = copy %ncjz
	jmp @loop

@jmp
	%taddrjmp = add %ci, 8
	%tjmp = loadl %code + %taddrjmp
	%pc = copy %tjmp
	jmp @loop

@inc1
	%npcinc = add %pc, 1
	%pc = copy %npcinc
	jmp @loop

@exit
	%czero = ceql %cnt, 0
	jnz %czero, @ret0, @rettop

@rettop
	%ctop = sub %cnt, 1
	%itop = mul %ctop, 8
	%topvret = loadl %stack + %itop
	jmp @clear

@ret0
	%topvret = copy 0
	jmp @clear

@clear
	%iclear = copy %cnt
	jmp @cloop

@cloop
	%cclr = csltl %iclear, %cap
	jnz %cclr, @cdo, @cdone

@cdo
	%iiclo = mul %iclear, 8
	storel 0, %stack + %iiclo
	%niclo = add %iclear, 1
	%iclear = copy %niclo
	jmp @cloop

@cdone
	%retres = copy %topvret
	ret %retres
}
