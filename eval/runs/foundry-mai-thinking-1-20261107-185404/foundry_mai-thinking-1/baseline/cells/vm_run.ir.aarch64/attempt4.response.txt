export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc0 = copy 0
	%cnt0 = copy 0
	jmp @loop

@loop
	%pc_cur = phi @start %pc0, @push %pc1, @add %pc2, @sub %pc3, @mul %pc4, @dup %pc5, @swap %pc6, @drop %pc7, @over %pc8, @lt %pc9, @jzt %pc10, @jzf %pc11, @jmp %pc12, @inc1 %pc13
	%cnt_cur = phi @start %cnt0, @push %cnt1, @add %cnt2, @sub %cnt3, @mul %cnt4, @dup %cnt5, @swap %cnt6, @drop %cnt7, @over %cnt8, @lt %cnt9, @jzt %cnt10, @jzf %cnt11, @jmp %cnt12, @inc1 %cnt13
	%cnd_loop = csltl %pc_cur, %len
	jnz %cnd_loop, @fetch, @exit

@fetch
	%ci_off = mul %pc_cur, 8
	%insl_val = loadl %code + %ci_off
	%op0_c = ceql %insl_val, 0
	jnz %op0_c, @halt, @check1

@check1
	%op1_c = ceql %insl_val, 1
	jnz %op1_c, @push, @check2

@check2
	%op2_c = ceql %insl_val, 2
	jnz %op2_c, @add, @check3

@check3
	%op3_c = ceql %insl_val, 3
	jnz %op3_c, @sub, @check4

@check4
	%op4_c = ceql %insl_val, 4
	jnz %op4_c, @mul, @check5

@check5
	%op5_c = ceql %insl_val, 5
	jnz %op5_c, @dup, @check6

@check6
	%op6_c = ceql %insl_val, 6
	jnz %op6_c, @swap, @check7

@check7
	%op7_c = ceql %insl_val, 7
	jnz %op7_c, @drop, @check8

@check8
	%op8_c = ceql %insl_val, 8
	jnz %op8_c, @over, @check9

@check9
	%op9_c = ceql %insl_val, 9
	jnz %op9_c, @lt, @check10

@check10
	%op10_c = ceql %insl_val, 10
	jnz %op10_c, @jz, @check11

@check11
	%op11_c = ceql %insl_val, 11
	jnz %op11_c, @jmp, @inc1

@halt
	jmp @exit

@push
	%vaddr_p = add %ci_off, 8
	%vpush_val = loadl %code + %vaddr_p
	%pidx_p = mul %cnt_cur, 8
	storel %vpush_val, %stack + %pidx_p
	%cnt1 = add %cnt_cur, 1
	%pc1 = add %pc_cur, 2
	jmp @loop

@add
	%c1a = sub %cnt_cur, 1
	%c2a = sub %cnt_cur, 2
	%i1a = mul %c1a, 8
	%i2a = mul %c2a, 8
	%ba = loadl %stack + %i1a
	%aa = loadl %stack + %i2a
	%ra = add %aa, %ba
	storel %ra, %stack + %i2a
	%cnt2 = copy %c2a
	%pc2 = add %pc_cur, 1
	jmp @loop

@sub
	%c1s = sub %cnt_cur, 1
	%c2s = sub %cnt_cur, 2
	%i1s = mul %c1s, 8
	%i2s = mul %c2s, 8
	%bs = loadl %stack + %i1s
	%asv = loadl %stack + %i2s
	%rs = sub %asv, %bs
	storel %rs, %stack + %i2s
	%cnt3 = copy %c2s
	%pc3 = add %pc_cur, 1
	jmp @loop

@mul
	%c1m = sub %cnt_cur, 1
	%c2m = sub %cnt_cur, 2
	%i1m = mul %c1m, 8
	%i2m = mul %c2m, 8
	%bm = loadl %stack + %i1m
	%amv = loadl %stack + %i2m
	%rm = mul %amv, %bm
	storel %rm, %stack + %i2m
	%cnt4 = copy %c2m
	%pc4 = add %pc_cur, 1
	jmp @loop

@dup
	%c1d = sub %cnt_cur, 1
	%i1d = mul %c1d, 8
	%topd = loadl %stack + %i1d
	%i2d = mul %cnt_cur, 8
	storel %topd, %stack + %i2d
	%cnt5 = add %cnt_cur, 1
	%pc5 = add %pc_cur, 1
	jmp @loop

@swap
	%c1sw = sub %cnt_cur, 1
	%c2sw = sub %cnt_cur, 2
	%i1sw = mul %c1sw, 8
	%i2sw = mul %c2sw, 8
	%tsw = loadl %stack + %i1sw
	%ssw = loadl %stack + %i2sw
	storel %tsw, %stack + %i2sw
	storel %ssw, %stack + %i1sw
	%cnt6 = copy %cnt_cur
	%pc6 = add %pc_cur, 1
	jmp @loop

@drop
	%cnt7 = sub %cnt_cur, 1
	%pc7 = add %pc_cur, 1
	jmp @loop

@over
	%c1ov = sub %cnt_cur, 1
	%c2ov = sub %cnt_cur, 2
	%i2ov = mul %c2ov, 8
	%ovv = loadl %stack + %i2ov
	%inov = mul %cnt_cur, 8
	storel %ovv, %stack + %inov
	%cnt8 = add %cnt_cur, 1
	%pc8 = add %pc_cur, 1
	jmp @loop

@lt
	%c1lt = sub %cnt_cur, 1
	%c2lt = sub %cnt_cur, 2
	%i1lt = mul %c1lt, 8
	%i2lt = mul %c2lt, 8
	%blt = loadl %stack + %i1lt
	%altv = loadl %stack + %i2lt
	%cltw = csltl %altv, %blt
	%rltv = zextl %cltw
	storel %rltv, %stack + %i2lt
	%cnt9 = copy %c2lt
	%pc9 = add %pc_cur, 1
	jmp @loop

@jz
	%c1jz = sub %cnt_cur, 1
	%i1jz = mul %c1jz, 8
	%vjzv = loadl %stack + %i1jz
	%ncjzv = copy %c1jz
	%v0jzw = ceql %vjzv, 0
	jnz %v0jzw, @jzt, @jzf

@jzt
	%taddrjz = add %ci_off, 8
	%tjzv = loadl %code + %taddrjz
	%cnt10 = copy %ncjzv
	%pc10 = copy %tjzv
	jmp @loop

@jzf
	%cnt11 = copy %ncjzv
	%pc11 = add %pc_cur, 2
	jmp @loop

@jmp
	%taddrjmp = add %ci_off, 8
	%tjmpv = loadl %code + %taddrjmp
	%cnt12 = copy %cnt_cur
	%pc12 = copy %tjmpv
	jmp @loop

@inc1
	%cnt13 = copy %cnt_cur
	%pc13 = add %pc_cur, 1
	jmp @loop

@exit
	%czero_e = ceql %cnt_cur, 0
	jnz %czero_e, @ret0, @rettop

@rettop
	%ctop_e = sub %cnt_cur, 1
	%itop_e = mul %ctop_e, 8
	%topv_r = loadl %stack + %itop_e
	jmp @clear

@ret0
	%topv_r = copy 0
	jmp @clear

@clear
	%iclear0 = copy %cnt_cur
	jmp @cloop

@cloop
	%iclear_cur = phi @clear %iclear0, @cdo %iclear_n
	%cclr_w = csltl %iclear_cur, %cap
	jnz %cclr_w, @cdo, @cdone

@cdo
	%iiclo = mul %iclear_cur, 8
	storel 0, %stack + %iiclo
	%iclear_n = add %iclear_cur, 1
	jmp @cloop

@cdone
	ret %topv_r
}
