export function l $max_byte(l %p, l %n) {
@start
	%zero =l copy 0
	%iszero =w ceql %n, %zero
	jnz %iszero, @retzero, @loopinit
@retzero
	ret %zero
@loopinit
	%maxv =l copy 0
	%i =l copy 0
	jmp @loopcond
@loopcond
	%cmpi =w csltl %i, %n
	jnz %cmpi, @loopbody, @done
@loopbody
	%addr =l add %p, %i
	%v =l loadub %addr
	%vgt =w csgtl %v, %maxv
	jnz %vgt, @updatemax, @incri
@updatemax
	%maxv =l copy %v
	jmp @incri
@incri
	%i =l add %i, 1
	jmp @loopcond
@done
	ret %maxv
}
