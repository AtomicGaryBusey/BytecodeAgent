export function l $fib_rec(l %n) {
@start
	%cmp0 =w ceql %n, 0
	jnz %cmp0, @ret0, @check1
@ret0
	ret 0
@check1
	%cmp1 =w ceql %n, 1
	jnz %cmp1, @ret1, @recur
@ret1
	ret 1
@recur
	%n1 =l sub %n, 1
	%f1 =l call $fib_rec(l %n1)
	%n2 =l sub %n, 2
	%f2 =l call $fib_rec(l %n2)
	%res =l add %f1, %f2
	ret %res
}
