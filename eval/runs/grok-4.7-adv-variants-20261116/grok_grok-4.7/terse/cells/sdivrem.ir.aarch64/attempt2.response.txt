export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @z, @n
@z
	storel %a, %rem
	ret 0
@n
	%ae =w ceql %a, 0x8000000000000000
	jnz %ae, @c, @d
@c
	%be =w ceql %b, -1
	jnz %be, @o, @d
@o
	storel 0, %rem
	ret 0x8000000000000000
@d
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
