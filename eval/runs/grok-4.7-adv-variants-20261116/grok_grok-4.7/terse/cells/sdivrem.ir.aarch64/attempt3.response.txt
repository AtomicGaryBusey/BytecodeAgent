export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @z, @n
@z
	storel %a, %rem
	ret 0
@n
	%one =l copy 1
	%min =l shl %one, 63
	%ae =w ceql %a, %min
	jnz %ae, @c, @d
@c
	%be =w ceql %b, -1
	jnz %be, @o, @d
@o
	storel 0, %rem
	ret %min
@d
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
