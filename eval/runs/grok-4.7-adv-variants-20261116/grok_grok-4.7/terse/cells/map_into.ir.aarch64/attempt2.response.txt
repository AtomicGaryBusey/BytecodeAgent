export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%k =l phi @start %n, @b %k1
	%c =l phi @start 0, @b %c1
	%s =l phi @start %src, @b %s1
	%d =l phi @start %dst, @b %d1
	%p =w csgtl %k, 0
	jnz %p, @b, @e
@b
	%v =l loadl %s
	%r =l call %f(l %v)
	storel %r, %d
	%t =w csltl %r, %v
	%u =l extsw %t
	%c1 =l add %c, %u
	%s1 =l add %s, 8
	%d1 =l add %d, 8
	%k1 =l sub %k, 1
	jmp @loop
@e
	ret %c
}
