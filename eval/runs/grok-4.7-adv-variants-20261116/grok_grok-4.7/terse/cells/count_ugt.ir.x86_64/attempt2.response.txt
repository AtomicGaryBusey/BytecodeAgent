export function l $count_ugt(l %a, l %n, l %t) {
@start
	%b =l mul %n, 8
	%e =l add %a, %b
	jmp @loop
@loop
	%p =l phi @start %a, @body %p1
	%c =l phi @start 0, @body %c1
	%k =w cnel %p, %e
	jnz %k, @body, @end
@body
	%v =l loadl %p
	%g =w cugtl %v, %t
	%h =l extuw %g
	%c1 =l add %c, %h
	%p1 =l add %p, 8
	jmp @loop
@end
	ret %c
}
