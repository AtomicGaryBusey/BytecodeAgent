export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%p =l phi @start %a, @body %p2
	%c =w phi @start 0, @body %c1
	%cont =w csltl %i, %nm1
	jnz %cont, @body, @end
@body
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%i1 =l add %i, 1
	%gt =w csgtl %x, %y
	%c1 =w add %c, %gt
	jmp @loop
@end
	%cl =l extuw %c
	ret %cl
}
