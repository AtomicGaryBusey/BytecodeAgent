export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%nb =l shl %n, 3
	%end =l add %a, %nb
	%max0 =l shl 1, 63
	%min0 =l sub %max0, 1
	jmp @loop
@loop
	%p =l phi @start %a, @body %p1
	%min =l phi @start %min0, @body %min1
	%max =l phi @start %max0, @body %max1
	%sum =l phi @start 0, @body %sum1
	%odd =l phi @start 0, @body %odd1
	%c =w ceql %p, %end
	jnz %c, @done, @body
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%b =l and %y, 1
	%odd1 =l add %odd, %b
	%sum1 =l add %sum, %y
	%lt =w csltl %y, %min
	%ls =l extsw %lt
	%d =l sub %y, %min
	%dm =l mul %d, %ls
	%min1 =l add %min, %dm
	%gt =w csgtl %y, %max
	%gs =l extsw %gt
	%e =l sub %y, %max
	%em =l mul %e, %gs
	%max1 =l add %max, %em
	%p1 =l add %p, 8
	jmp @loop
@done
	storel %min, %out
	%op =l add %out, 8
	storel %max, %op
	%op2 =l add %op, 8
	storel %sum, %op2
	ret %odd
}
