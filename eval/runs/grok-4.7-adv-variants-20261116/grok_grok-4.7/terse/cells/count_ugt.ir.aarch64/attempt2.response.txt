export function l $count_ugt(l %a, l %n, l %t) {
@start
	%b =l shl %n, 3
	%e =l add %a, %b
	jmp @loop
@loop
	%c =w phi @start 0, @body %c1
	%p =l phi @start %a, @body %p1
	%k =w cultl %p, %e
	jnz %k, @body, @end
@body
	%v =l loadl %p
	%g =w cugtl %v, %t
	%c1 =w add %c, %g
	%p1 =l add %p, 8
	jmp @loop
@end
	%r =l extuw %c
	ret %r
}
