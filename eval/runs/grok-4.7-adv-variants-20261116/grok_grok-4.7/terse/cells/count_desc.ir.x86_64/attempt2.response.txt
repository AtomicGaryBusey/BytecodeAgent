export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	jmp @h
@h
	%i =l phi @start 0, @b %ni
	%c =l phi @start 0, @b %nc
	%go =w csltl %i, %lim
	jnz %go, @b, @e
@b
	%ni =l add %i, 1
	%off =l shl %i, 3
	%pa =l add %a, %off
	%x =l loadl %pa
	%pb =l add %pa, 8
	%y =l loadl %pb
	%gw =w csgtl %x, %y
	%g =l extsw %gw
	%nc =l add %c, %g
	jmp @h
@e
	ret %c
}
