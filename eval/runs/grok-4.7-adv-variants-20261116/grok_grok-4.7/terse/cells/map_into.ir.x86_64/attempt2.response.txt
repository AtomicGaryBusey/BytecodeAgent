export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%c =l phi @start 0, @body %c1
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%sv =l loadl %sa
	%rv =l call %f(l %sv)
	%da =l add %dst, %off
	storel %rv, %da
	%lt =w csltl %rv, %sv
	%lte =l extuw %lt
	%c1 =l add %c, %lte
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %c
}
