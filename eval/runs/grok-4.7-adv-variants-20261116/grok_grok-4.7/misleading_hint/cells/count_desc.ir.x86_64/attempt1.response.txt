export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%small =w csltl %n, 2
	jnz %small, @zero, @loop
@zero
	%r0 =l copy 0
	ret %r0
@loop
	%i =l phi @start 0, @body %i1
	%count =l phi @start 0, @body %count1
	%more =w csltl %i, %nm1
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%i1 =l add %i, 1
	%off2 =l mul %i1, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	%gt =w csgtl %v, %v2
	%gtl =l extuw %gt
	%count1 =l add %count, %gtl
	jmp @loop
@done
	ret %count
}
