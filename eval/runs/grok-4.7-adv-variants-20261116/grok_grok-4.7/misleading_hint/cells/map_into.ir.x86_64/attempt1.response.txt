export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @start %zero, @body %inext
	%count =l phi @start %zero, @body %countnext
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%y =l call %f(l %x)
	storel %y, %dp
	%lt =w csltl %y, %x
	%ltl =l extuw %lt
	%countnext =l add %count, %ltl
	%inext =l add %i, 1
	jmp @loop
@end
	ret %count
}
