export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @head
@head
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%pdst =l phi @start %dst, @body %pdst1
	%psrc =l phi @start %src, @body %psrc1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @end
@body
	%val =l loadl %psrc
	%res =l call %f(l %val)
	storel %res, %pdst
	%lt =w csltl %res, %val
	%ltl =l extuw %lt
	%cnt1 =l add %cnt, %ltl
	%i1 =l add %i, 1
	%pdst1 =l add %pdst, 8
	%psrc1 =l add %psrc, 8
	jmp @head
@end
	ret %cnt
}
