export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%c =l phi @start 0, @body %cnext
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gtext =l extuw %gt
	%cnext =l add %c, %gtext
	%inext =l add %i, 1
	jmp @loop
@done
	ret %c
}
