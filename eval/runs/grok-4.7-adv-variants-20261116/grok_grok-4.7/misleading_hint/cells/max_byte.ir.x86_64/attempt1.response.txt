export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%max =l phi @start 0, @next %maxn
	%i =l phi @start 0, @next %in
	%cont =w csltl %i, %n
	jnz %cont, @body, @end
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %max
	jnz %gt, @update, @keep
@update
	jmp @next
@keep
	jmp @next
@next
	%maxn =l phi @update %b, @keep %max
	%in =l add %i, 1
	jmp @loop
@end
	ret %max
}
