export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @retz, @pre
@retz
	%z =l copy 0
	ret %z
@pre
	%nm1 =l sub %n, 1
	jmp @head
@head
	%i =l phi @pre 0, @body %inext
	%cnt =l phi @pre 0, @body %cntnext
	%end =w ceql %i, %nm1
	jnz %end, @done, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%ai =l loadl %p
	%inext =l add %i, 1
	%off1 =l mul %inext, 8
	%p1 =l add %a, %off1
	%ai1 =l loadl %p1
	%gt =w csgtl %ai, %ai1
	%gtl =l extuw %gt
	%cntnext =l add %cnt, %gtl
	jmp @head
@done
	ret %cnt
}
