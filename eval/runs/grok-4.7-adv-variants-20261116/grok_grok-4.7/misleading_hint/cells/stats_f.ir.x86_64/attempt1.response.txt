export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%od0 =l and %y0, 1
	%eq1 =w ceql %n, 1
	jnz %eq1, @merge, @loop
@loop
	%i =l phi @start 1, @body %inext
	%mn =l phi @start %y0, @body %mn2
	%mx =l phi @start %y0, @body %mx2
	%sm =l phi @start %y0, @body %sm2
	%od =l phi @start %od0, @body %od2
	%lt =w csltl %i, %n
	jnz %lt, @body, @merge
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%ai =l loadl %ptr
	%yi =l call %f(l %ai)
	%cl =w csltl %yi, %mn
	%cll =l extuw %cl
	%dmn =l sub %yi, %mn
	%amn =l mul %dmn, %cll
	%mn2 =l add %mn, %amn
	%cg =w csgtl %yi, %mx
	%cgl =l extuw %cg
	%dmx =l sub %yi, %mx
	%amx =l mul %dmx, %cgl
	%mx2 =l add %mx, %amx
	%sm2 =l add %sm, %yi
	%bit =l and %yi, 1
	%od2 =l add %od, %bit
	%inext =l add %i, 1
	jmp @loop
@merge
	%minf =l phi @start %y0, @loop %mn
	%maxf =l phi @start %y0, @loop %mx
	%sumf =l phi @start %y0, @loop %sm
	%oddf =l phi @start %od0, @loop %od
	storel %minf, %out
	%o1 =l add %out, 8
	storel %maxf, %o1
	%o2 =l add %out, 16
	storel %sumf, %o2
	ret %oddf
}
