export function l $max_byte(l %p, l %n) {
@start
	%isz =w ceql %n, 0
	jnz %isz, @empty, @init
@empty
	ret 0
@init
	jmp @loop
@loop
	%i =l phi @init 0, @cont %i1
	%m =l phi @init 0, @cont %m1
	%done =w ceql %i, %n
	jnz %done, @done, @body
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w cugtl %b, %m
	jnz %gt, @newmax, @samemax
@newmax
	jmp @cont
@samemax
	jmp @cont
@cont
	%m1 =l phi @newmax %b, @samemax %m
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %m
}
