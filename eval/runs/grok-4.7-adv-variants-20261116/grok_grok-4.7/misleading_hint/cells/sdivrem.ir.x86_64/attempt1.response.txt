export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%zero =l copy 0
	%bz =w ceql %b, %zero
	jnz %bz, @divzero, @chkmin
@divzero
	storel %a, %rem
	ret %zero
@chkmin
	%one =l copy 1
	%sh =w copy 63
	%min =l shl %one, %sh
	%ismin =w ceql %a, %min
	jnz %ismin, @chkneg, @div
@chkneg
	%negone =l copy -1
	%isn1 =w ceql %b, %negone
	jnz %isn1, @ovf, @div
@ovf
	storel %zero, %rem
	ret %min
@div
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
