export function l $fill_down(l %a, l %n, l %start) {
@start
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @start %zero, @body %inext
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l sub %start, %i
	storel %v, %p
	%inext =l add %i, 1
	jmp @loop
@end
	ret %n
}
