export function l $bmin_empty(l %a, l %n) {
@start
	%isz =w ceql %n, 0
	jnz %isz, @empty, @init
@empty
	%max =l copy 9223372036854775807
	ret %max
@init
	%m0 =l loadl %a
	%i0 =l copy 1
	jmp @cond
@cond
	%i =l phi @init %i0, @next %i1
	%m =l phi @init %m0, @next %m1
	%ge =w csgel %i, %n
	jnz %ge, @done, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%lt =w csltl %v, %m
	jnz %lt, @take, @keep
@take

	jmp @next
@keep
	%mk =l copy %m
	jmp @next
@next
	%m1 =l phi @take %mt, @keep %mk

	jmp @cond
@done
	ret %m
}
