export function l $bmin_empty(l %a, l %n) {
@start
	%isempty =w ceql %n, 0
	jnz %isempty, @empty, @init
@empty
	%max =l copy 9223372036854775807
	ret %max
@init
	%m0 =l loadl %a
	%i0 =l copy 1
	jmp @head
@head
	%m =l phi @init %m0, @latch %m1
	%i =l phi @init %i0, @latch %i1
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%less =w csltl %v, %m
	jnz %less, @take, @latch
@take
	jmp @latch
@latch
	%m1 =l phi @body %m, @take %v
	%i1 =l add %i, 1
	jmp @head
@done
	ret %m
}
