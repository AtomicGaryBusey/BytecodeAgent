export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%sixteen =l copy 16
	%sixtythree =l copy 63
	%allones =l copy -1
	%imax0 =l shl %one, %sixtythree
	%imin0 =l xor %imax0, %allones
	jmp @head
@head
	%i =l phi @start %zero, @latch %inext
	%min =l phi @start %imin0, @latch %minl
	%max =l phi @start %imax0, @latch %maxl
	%sum =l phi @start %zero, @latch %suml
	%odds =l phi @start %zero, @latch %oddsn
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, %eight
	%ptr =l add %a, %off
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @keepmin
@setmin
	%miny =l copy %y
	jmp @mergemin
@keepmin
	%minm =l copy %min
	jmp @mergemin
@mergemin
	%minn =l phi @setmin %miny, @keepmin %minm
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @keepmax
@setmax
	%maxy =l copy %y
	jmp @mergemax
@keepmax
	%maxm =l copy %max
	jmp @mergemax
@mergemax
	%maxn =l phi @setmax %maxy, @keepmax %maxm
	%sumn =l add %sum, %y
	%bit =l and %y, %one
	%isodd =w cnel %bit, %zero
	jnz %isodd, @isodd, @iseven
@isodd
	%oddsn1 =l add %odds, %one
	jmp @latch
@iseven
	%oddsn0 =l copy %odds
	jmp @latch
@latch
	%oddsn =l phi @isodd %oddsn1, @iseven %oddsn0
	%minl =l copy %minn
	%maxl =l copy %maxn
	%suml =l copy %sumn
	%inext =l add %i, %one
	jmp @head
@done
	%out1 =l add %out, %eight
	%out2 =l add %out, %sixteen
	storel %min, %out
	storel %max, %out1
	storel %sum, %out2
	ret %odds
}
