export function l $fill_down(l %a, l %n, l %start) {
@start
	%z =w ceql %n, 0
	jnz %z, @done, @loop
@loop
	%p =l phi @start %a, @loop %pnext
	%v =l phi @start %start, @loop %vnext
	%c =l phi @start %n, @loop %cnext
	storel %v, %p
	%pnext =l add %p, 8
	%vnext =l sub %v, 1
	%cnext =l sub %c, 1
	%more =w cnel %cnext, 0
	jnz %more, @loop, @done
@done
	ret %n
}
