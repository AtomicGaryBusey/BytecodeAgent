export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @byzero, @nz
@byzero
	storel %a, %rem
	%z =l copy 0
	ret %z
@nz
	%m1 =l copy -1
	%bn =w ceql %b, %m1
	jnz %bn, @maybeov, @dok
@maybeov
	%one =l copy 1
	%sh =l copy 63
	%min =l shl %one, %sh
	%ismin =w ceql %a, %min
	jnz %ismin, @ovf, @dok
@ovf
	%zz =l copy 0
	storel %zz, %rem
	ret %a
@dok
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
