popcount:
    movz x1, #0x0101
    movk x1, #0x0101, lsl #16
    movk x1, #0x0101, lsl #32
    movk x1, #0x0101, lsl #48          // x1 = 0x0101010101010101
    lsl x5, x1, #4
    sub x5, x5, x1                     // x5 = 0x0f0f0f0f0f0f0f0f
    lsl x4, x5, #2
    eor x4, x4, x5                     // x4 = 0x3333333333333333
    lsl x3, x4, #1
    eor x3, x3, x4                     // x3 = 0x5555555555555555
    lsr x2, x0, #1
    and x2, x2, x3
    sub x0, x0, x2                     // 2-bit fields hold pair counts
    and x2, x0, x4
    lsr x6, x0, #2
    and x6, x6, x4
    add x0, x2, x6                     // 4-bit fields hold nibble counts
    add x0, x0, x0, lsr #4
    and x0, x0, x5                     // bytes hold byte counts
    mul x0, x0, x1
    lsr x0, x0, #56                    // sum of all byte counts
    ret
