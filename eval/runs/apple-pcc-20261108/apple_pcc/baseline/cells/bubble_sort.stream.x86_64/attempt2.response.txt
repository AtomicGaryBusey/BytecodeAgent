bubble_sort:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rsi
    cmp r12, 1
    jle .Ldone
    mov r13, r12
    sub r13, 1
    .Louter:
        mov r14, 0
        .Linner:
            mov rax, qword ptr [rdi + r14*8]
            mov rbx, qword ptr [rdi + r14*8 + 8]
            cmp rax, rbx
            jle .Lnext
            mov qword ptr [rdi + r14*8], rbx
            mov qword ptr [rdi + r14*8 + 8], rax
        .Lnext:
            inc r14
            cmp r14, r13
            jl .Linner
        dec r13
        cmp r13, 1
        jg .Louter
    .Ldone:
        xor rax, rax
        pop r15
        pop r14
        pop r13
        pop r12
        pop rbp
        pop rbx
        ret
