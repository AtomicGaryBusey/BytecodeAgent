export function l $fib(l n) {
@entry
    %a = 0
    %b = 1
    %c = n
    %cmp = ceql %c 0
    jnz %cmp @loop
    %res = %a
    ret %res
@loop
    %cmp = ceql %c 1
    jnz %cmp @iter
    %res = %b
    ret %res
@iter
    %tmp = addl %a %b
    %a = %b
    %b = %tmp
    %c = subl %c 1
    %cmp = csltl %c 0
    jnz %cmp @iter
    %res = %b
    ret %res
}
