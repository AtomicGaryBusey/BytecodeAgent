export function l $bubble_sort(l a, l n) {
@entry
    %i = 0
@outer
    %cmp = csltl %i %n
    jnz %cmp @inner
    %ret = 0
    jump @return
@inner
    %j = 0
    %limit = %n
    %sub = %i
    %limit = %limit - %sub
@loop
    %cmpj = csltl %j %limit
    jnz %cmpj @body
    jump @next_outer
@body
    %addr = %a + %j
    %addr1 = %addr + 8
    %val = loadl %addr
    %val1 = loadl %addr1
    %cmpv = csltl %val1 %val
    jnz %cmpv @swap
    jump @incj
@swap
    storel %val %addr1
    storel %val1 %addr
@incj
    %j = %j + 1
    jump @loop
@next_outer
    %i = %i + 1
    jump @outer
@return
    return %ret
}
