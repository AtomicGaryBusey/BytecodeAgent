export function l $gcd(l $a, l $b) {
@entry
    %t0 = ceql $a, $b
    jnz %t0 @equal
    %t1 = csltl $a, $b
    jnz %t1 @swap
@loop
    %t2 = csltl $b, $a
    jnz %t2 @swap
    %t3 = ceql $b, $0
    jnz %t3 @done
    %t4 = modl $a, $b
    %a = %t4
    jnz @loop
@swap
    %t5 = %a
    %a = %b
    %b = %t5
    jnz @loop
@equal
    %a = %a
    jnz @done
@done
    ret
}
