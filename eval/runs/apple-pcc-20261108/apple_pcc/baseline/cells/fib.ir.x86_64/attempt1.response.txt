export function l $fib(l $n) {
@entry
    %a = 0
    %b = 1
    %c = $n
    %cond = ceql %c 0
    jnz %cond @loop
    %r = %a
    ret %r

@loop
    %cond = ceql %c 1
    jnz %cond @iter
    %r = %b
    ret %r

@iter
    %next = addl %a %b
    %a = %b
    %b = %next
    %c = subl %c 1
    %cond = csltl %c 0
    jnz %cond @exit
    jmp @loop

@exit
    %r = %b
    ret %r
}
