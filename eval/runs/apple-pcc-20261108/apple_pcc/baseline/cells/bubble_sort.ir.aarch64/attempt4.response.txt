export function l $bubble_sort(l a, l n) {
@entry
    l %i, %j, %temp, %val_i, %val_j
    w %cond
    %i = n
    %i = %i - 1
    %cond = %i <= 0
    jnz %cond, @return, %cond, 0

@outer_loop
    %j = 0
@inner_loop
    %cond = %j >= %i
    jnz %cond, @next_outer, %cond, 0
    %temp = %j * 8
    %val_i = loadl [a + %temp]
    %temp = %temp + 8
    %val_j = loadl [a + %temp]
    %cond = %val_i > %val_j
    jnz %cond, @swap, %cond, 0
    %j = %j + 1
    jnz %cond, @inner_loop, %cond, 0

@swap
    storel %val_j, [a + %temp - 8]
    storel %val_i, [a + %temp]
    %j = %j + 1
    jnz %cond, @inner_loop, %cond, 0

@next_outer
    %i = %i - 1
    %cond = %i <= 0
    jnz %cond, @return, %cond, 0
    %j = 0
    jnz %cond, @outer_loop, %cond, 0

@return
    %cond = 0
    ret %cond
}
