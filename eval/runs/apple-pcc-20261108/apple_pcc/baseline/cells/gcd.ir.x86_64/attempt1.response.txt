export function l $gcd(l $a, l $b) {
@entry
    %t1 = ceql $b, 0
    jnz %t1, @loop
    %r = $a
    jmp @exit
@loop
    %t2 = ceql $a, 0
    jnz %t2, @exit
    %t3 = csltl $a, $b
    jnz %t3, @swap
    %t4 = modl $a, $b
    %a = %t4
    jmp @loop
@swap
    %t5 = modl $b, $a
    %b = %t5
    jmp @loop
@exit
    %r = $b
    ret
}
