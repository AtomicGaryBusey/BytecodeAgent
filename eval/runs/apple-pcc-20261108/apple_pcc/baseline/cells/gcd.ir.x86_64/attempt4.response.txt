export function l $gcd(l $a, l $b) {
@entry
    %t1 = ceql $b, 0
    jnz %t1, @exit
    %t2 = ceql $a, 0
    jnz %t2, @swap_entry
    %t3 = csltl $a, $b
    jnz %t3, @swap_loop_start
    %t4 = modl $a, $b
    %a = %t4
    jmp @loop
@swap_entry
    %r = $b
    jmp @exit
@swap_loop_start
    %t5 = modl $b, $a
    %b = %t5
@loop
    %t6 = ceql $b, 0
    jnz %t6, @exit_loop
    %t7 = ceql $a, 0
    jnz %t7, @swap_loop
    %t8 = csltl $a, $b
    jnz %t8, @swap_loop
    %t9 = modl $a, $b
    %a = %t9
    jmp @loop
@swap_loop
    %t10 = modl $b, $a
    %b = %t10
    jmp @loop
@exit_loop
    %r = $b
@exit
    ret
}
