export function l $popcount(l %x) {
@start
	%c55 = or 0l, 0x5555555555555555l
	%t1 = and %x, %c55
	%t2 = shr %x, 1
	%t3 = and %t2, %c55
	%t4 = add %t1, %t3
	%c33 = or 0l, 0x3333333333333333l
	%t5 = and %t4, %c33
	%t6 = shr %t4, 2
	%t7 = and %t6, %c33
	%t8 = add %t5, %t7
	%t9 = shr %t8, 4
	%t10 = add %t8, %t9
	%c0f = or 0l, 0x0f0f0f0f0f0f0f0fl
	%t11 = and %t10, %c0f
	%c01 = or 0l, 0x0101010101010101l
	%t12 = mul %t11, %c01
	%t13 = shr %t12, 56
	ret %t13
}
