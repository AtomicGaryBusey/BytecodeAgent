add_overflow:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Compute sum = a + b (mod 2^64)
    add x8, x0, x1
    str x8, [x2]
    // Check for signed overflow: overflow occurs if a and b have same sign,
    // but sum has different sign from both.
    // Compute xor of inputs to check signs: if (a ^ b) < 0 then signs differ -> no overflow possible.
    eor x9, x0, x1
    // If signs differ (x9 negative), no overflow possible
    neg x10, x9       // x10 = ~(x9) (bitwise NOT)
    and x10, x10, x9  // x10 = (a^b) & ~(a^b) = 0 if signs differ? No—use different approach.
    // Instead: Check if a and b same sign: (a ^ b) >= 0 means same sign.
    cmp x9, #0
    b.ge .Lno_overflow_same_sign
    // Signs differ: overflow impossible
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret

.Lno_overflow_same_sign:
    // Now a and b have same sign. Check if sum has opposite sign.
    // If (a ^ sum) < 0 then overflow occurred.
    eor x10, x8, x0   // x10 = sum ^ a
    neg x11, x10      // x11 = ~x10
    and x12, x11, x10 // x12 = (sum^a) & ~(sum^a) = 0? Use cmp instead.
    cmp x10, #0
    b.lt .Loverflow
    mov x0, #0
    ldp x29, x30, [sp], #16
    ret

.Loverflow:
    mov x0, #1
    ldp x29, x30, [sp], #16
    ret
