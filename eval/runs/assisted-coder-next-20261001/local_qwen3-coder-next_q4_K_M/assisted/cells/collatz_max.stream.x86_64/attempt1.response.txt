collatz_max:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor eax, eax
    mov r15, rax
    cmp rdi, 1
    jle .Ldone
    mov rbp, 1
.Louter_loop:
    mov rax, rbp
    xor rcx, rcx
.Linner_loop:
    cmp rax, 1
    je .Ldone_inner
    test al, 1
    jnz .Lodd
    xor edx, edx
    shr rax, 1
    inc rcx
    jmp .Linner_loop
.Lodd:
    lea rdx, [rax + rax*2]
    inc rdx
    inc rcx
    mov rax, rdx
    jmp .Linner_loop
.Ldone_inner:
    cmp rcx, r15
    jle .Lnext_k
    mov r15, rcx
.Lnext_k:
    inc rbp
    cmp rbp, rdi
    jbe .Louter_loop
    mov rax, r15
.Ldone:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
