export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@outer_loop
	%more_bytes =w csltl %i, %n
	jnz %more_bytes, @byte_loop, @finish
@byte_loop
	%off =l add %p, %i
	%b =w loadub %off
	%crc =w xor %crc, %b
	%j =w copy 0
@bit_loop
	%test_bit =w and %crc, 1
	%branch_on_bit =w ceqw %test_bit, 1
	jnz %branch_on_bit, @xord_step, @shift_step
@xord_step
	%crc_shr =w shr %crc, 1
	%xor_val =w copy 2564026856
	%crc =w xor %crc_shr, %xor_val
	jmp @next_bit
@shift_step
	%crc =w shr %crc, 1
@next_bit
	%j =w add %j, 1
	%continue_bits =w ceqw %j, 8
	jnz %continue_bits, @done_byte, @bit_loop
@done_byte
	%i =l add %i, 1
	jmp @outer_loop
@finish
	%xor_final =w xor %crc, 4294967295
	%result =l extuw %xor_final
	ret %result
}
