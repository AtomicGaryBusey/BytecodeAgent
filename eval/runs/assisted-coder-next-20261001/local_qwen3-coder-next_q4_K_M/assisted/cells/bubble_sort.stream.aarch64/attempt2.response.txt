bubble_sort:
    stp x19, x20, [sp, #-16]!
    mov x19, x0                 // save array pointer
    mov x20, x1                 // save n
    cbz x20, .Ldone             // if n == 0, done
    sub x21, x20, #1            // i = n-1 (outer loop counter)
.Louter:
    cmp x21, #0
    b.le .Ldone                 // if i <= 0, done
    movz x22, #1                // swapped flag = 1
    sub x23, x21, #1            // j limit = i-1
    movz x24, #0                // j = 0
.Linner:
    cmp x24, x23
    b.gt .Lcheck_swapped        // if j > i-1, inner loop done
    lsl x9, x24, #3             // j * 8
    add x10, x19, x9            // &a[j]
    ldr x11, [x10]
    ldr x12, [x10, #8]
    cmp x11, x12
    b.le .Lnext                 // if a[j] <= a[j+1], no swap
    str x12, [x10]
    str x11, [x10, #8]
    movz x22, #0                // set swapped flag = 0
.Lnext:
    add x24, x24, #1            // j++
    b .Linner
.Lcheck_swapped:
    cbz x22, .Ldone             // if no swaps, done
    sub x21, x21, #1            // i--
    b .Louter
.Ldone:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ret
