bubble_sort:
    mov x19, x0                 // save array pointer
    mov x20, x1                 // save n
    cbz x20, .Ldone             // if n == 0, done
    sub x21, x20, #1            // i = n-1 (outer loop counter)
.Louter:
    cmp x21, #0
    b.le .Ldone                 // if i <= 0, done
    mov x22, #1                 // swapped flag = 1
    sub x23, x21, #1            // j = i-1 (inner loop limit)
    mov x24, #0                 // j = 0
.Linner:
    cmp x24, x23
    b.gt .Lcheck_swapped      // if j > i-1, inner loop done
    // compute addresses: a[j] and a[j+1]
    lsl x9, x24, #3             // j * 8
    add x10, x19, x9            // &a[j]
    ldr x11, [x10]              // load a[j]
    ldr x12, [x10, #8]          // load a[j+1]
    cmp x11, x12
    b.le .Lnext                 // if a[j] <= a[j+1], no swap
    // swap
    str x12, [x10]
    str x11, [x10, #8]
    mov x22, #0                 // set swapped flag
.Lnext:
    add x24, x24, #1            // j++
    b .Linner
.Lcheck_swapped:
    cbz x22, .Ldone             // if no swaps, done
    sub x21, x21, #1            // i--
    b .Louter
.Ldone:
    mov x0, #0
    ret
