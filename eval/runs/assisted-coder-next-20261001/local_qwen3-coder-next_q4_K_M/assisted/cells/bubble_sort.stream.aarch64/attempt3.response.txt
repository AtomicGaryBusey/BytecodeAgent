bubble_sort:
    stp x19, x20, [sp, #-16]!
    mov x19, x0                 // save array pointer
    mov x20, x1                 // save n
    cbz x20, .Ldone             // if n == 0, done
    sub x9, x20, #1             // i = n-1 (outer loop counter)
.Louter:
    cmp x9, #0
    b.le .Ldone                 // if i <= 0, done
    movz w8, #1                 // swapped flag = 1
    sub x10, x9, #1             // j limit = i-1
    movz w11, #0                // j = 0
.Linner:
    cmp w11, w10
    b.gt .Lcheck_swapped        // if j > i-1, inner loop done
    lsl x12, x11, #3            // j * 8
    add x13, x19, x12           // &a[j]
    ldr x14, [x13]
    ldr x15, [x13, #8]
    cmp x14, x15
    b.le .Lnext                 // if a[j] <= a[j+1], no swap
    str x15, [x13]
    str x14, [x13, #8]
    movz w8, #0                 // set swapped flag = 0
.Lnext:
    add w11, w11, #1            // j++
    b .Linner
.Lcheck_swapped:
    cbz w8, .Ldone              // if no swaps, done
    sub x9, x9, #1              // i--
    b .Louter
.Ldone:
    mov x0, #0
    ldp x19, x20, [sp], #16
    ret
