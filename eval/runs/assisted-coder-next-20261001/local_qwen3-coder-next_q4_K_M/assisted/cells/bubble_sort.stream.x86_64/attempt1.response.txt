bubble_sort:
    xor eax, eax
    cmp rsi, 1
    jle .Ldone

    mov rbx, rsi        # n
    dec rbx             # outer loop bound: n-1

.Louter_loop:
    cmp rbx, 1
    jl .Ldone
    lea rcx, [rsi - 1]
    xor rdx, rdx        # i = 0

.Linner_loop:
    cmp rdx, rbx
    jge .Louter_done
    mov rax, qword ptr [rdi + rdx*8]
    mov rsi, qword ptr [rdi + rdx*8 + 8]
    cmp rax, rsi
    jle .Lno_swap

    # swap
    mov qword ptr [rdi + rdx*8], rsi
    mov qword ptr [rdi + rdx*8 + 8], rax

.Lno_swap:
    inc rdx
    jmp .Linner_loop

.Louter_done:
    dec rbx
    jmp .Louter_loop

.Ldone:
    xor eax, eax
    ret
