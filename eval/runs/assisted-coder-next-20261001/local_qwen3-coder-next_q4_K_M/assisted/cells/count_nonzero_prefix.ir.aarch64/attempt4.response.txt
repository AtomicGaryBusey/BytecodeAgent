export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%count =l copy 0
	%i =l copy 0
@loop
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%val =l loadl %p
	%zero =w ceql %val, 0
	jnz %zero, @exit, @next
@next
	%i =l add %i, 1
	jmp @loop
@exit
	ret %count
@done
	ret %n
}
