map_sum:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    eor x8, x8, x8          // zero accumulator (sum = 0)
    cbz x1, .Ldone          // if n == 0, return 0
    mov x9, x0              // save pointer to array in callee-saved reg
    eor x10, x10, x10       // index i = 0
.Lloop:
    ldr x3, [x9, x10, lsl #3]   // load a[i]
    blr x2                     // call f(a[i]) — result in x0
    add x8, x8, x0             // sum += f(a[i])
    add x10, x10, x1           // increment index i
    cmp x10, x1                // compare i with n
    b.ne .Lloop               // loop if not equal
.Ldone:
    mov x0, x8                // return sum (in case of early exit or zero case)
    ldp x29, x30, [sp], #16
    ret
