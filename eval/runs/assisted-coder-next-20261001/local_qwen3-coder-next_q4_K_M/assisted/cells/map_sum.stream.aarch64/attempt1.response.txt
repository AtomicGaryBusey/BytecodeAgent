map_sum:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    eor x0, x0, x0          // zero accumulator (sum = 0)
    cbz x1, .Ldone          // if n == 0, return 0
    mov x19, x0             // x19 = sum (callee-saved)
    mov x20, x0             // x20 = counter initialized to 0 (callee-saved)
    mov x21, x0             // x21 = base pointer of array (callee-saved)
    add x21, x0, x2         // x21 = a + 0 (x2 is the address of a[0])
.Lloop:
    ldr x3, [x21, x20, lsl #3]   // load a[i]
    blr x2                        // call f(a[i]) — f in x2, result in x0
    add x19, x19, x0              // sum += f(a[i])
    add x20, x20, #1             // i++
    cmp x20, x1                  // compare i with n
    b.lt .Lloop                  // if i < n, continue loop
.Ldone:
    mov x0, x19                 // return sum
    ldp x29, x30, [sp], #16
    ret
