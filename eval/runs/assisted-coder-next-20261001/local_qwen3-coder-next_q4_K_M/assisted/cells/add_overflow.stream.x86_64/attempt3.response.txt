add_overflow:
    push rbx
    mov rbx, rsi
    lea rax, [rdi + rsi]
    mov qword ptr [rbx], rax
    # Check overflow: (a ^ result) & (b ^ result) & sign_bit
    mov rcx, rdi
    xor rcx, rax        # rcx = a ^ result
    mov rdx, rsi
    xor rdx, rax        # rdx = b ^ result
    and rcx, rdx
    test rcx, 0x8000000000000000
    set
