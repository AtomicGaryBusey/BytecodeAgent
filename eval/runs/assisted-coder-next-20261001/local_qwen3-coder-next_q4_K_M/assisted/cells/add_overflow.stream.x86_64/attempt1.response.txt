add_overflow:
    push rbx
    mov rbx, rsi
    lea rax, [rdi + rsi]
    mov qword ptr [rbx], rax
    xor edx, edx
    mov eax, edi
    idiv esi
    cmp esi, 0
    jg .Lpositive_divisor
    js .Lnegative_divisor
    # divisor is zero, but spec doesn't require checking for that case
    xor eax, eax
    pop rbx
    ret
.Lpositive_divisor:
    cmp rax, 9223372036854775807
