add_overflow:
    push rbx
    mov rbx, rsi
    lea rax, [rdi + rsi]
    mov qword ptr [rbx], rax
    # Check overflow: if signs of a and b are same and sign of result differs
    mov rcx, rdi
    xor rcx, rsi          # rcx = a ^ b (if different signs, no overflow possible)
    test rcx, 0x8000000000000000
    j
