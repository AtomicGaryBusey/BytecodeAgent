export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%cmp =w ceql %n, %zero
	jnz %cmp, @done, @check_direction
@check_direction
	%off_dst =l add %dst, %n
	%off_src =l add %src, %n
	%forward =w csltl %dst, %src
	jnz %forward, @copy_forward, @copy_backward
@copy_forward
	%i =l copy 0
@loop_f
	%cond_f =w csltl %i, %n
	jnz %cond_f, @body_f, @done
@body_f
	%off_f =l mul %i, 8
	%p_src_f =l add %src, %off_f
	%val_f =l loadl %p_src_f
	%p_dst_f =l add %dst, %off_f
	storel %val_f, %p_dst_f
	%i =l add %i, 1
	jmp @loop_f
@copy_backward
	%i =l copy 0
@loop_b
	%cond_b =w csltl %i, %n
	jnz %cond_b, @body_b, @done
@body_b
	%idx_b =l sub %n, %i
	%idx_b =l add %idx_b, -1
	%off_b =l mul %idx_b, 8
	%p_src_b =l add %src, %off_b
	%val_b =l loadl %p_src_b
	%p_dst_b =l add %dst, %off_b
	storel %val_b, %p_dst_b
	%i =l add %i, 1
	jmp @loop_b
@done
	ret %zero
}
