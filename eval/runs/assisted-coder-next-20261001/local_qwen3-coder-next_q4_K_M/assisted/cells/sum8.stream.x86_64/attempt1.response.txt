sum8:
    push rbx
    push rbp
    sub rsp, 16
    mov rbx, rdi           # a
    mov rbp, rsi           # b
    mov rcx, rdx           # c
    mov rdx, rcx           # d (will use in imul)
    lea rax, [rbp + rbp]   # 2*b
    sub rbx, rax           # a - 2*b
    xor rax, rax
    lea rax, [rcx + rcx*4] # 5*c
    add rbx, rax           # a - 2*b + 3*c (since 5*c - 2*c = 3*c? Wait)
    # Let's recompute carefully:
    # We need: a - 2b + 3c - 4d + 5e - 6f + 7g - 8h
    # Instead, compute term by term in registers

    # Reset and recompute properly
    mov rax, rbx           # a
    mov rbx, rsi           # b (restore)
    imul rbp, rsi, 2       # 2*b
    sub rax, rbp           # a - 2*b

    mov rbp, rdx           # c
    imul rcx, rbp, 3       # 3*c
    add rax, rcx           # + 3*c

    mov rbp, rcx           # d (original d in rdx, but rdx was c; fix)
    # Let's reassign registers more carefully:
    # At entry: a=rdi, b=rsi, c=rdx, d=rcx, e=r8, f=r9, g=[rsp+8], h=[rsp+16]
    # We already saved rdi (a) in rbx earlier? No — let's start over with clean register usage.

    mov rax, rdi           # a
    imul rcx, rsi, 2       # 2*b
    sub rax, rcx           # a - 2*b

    imul rcx, rdx, 3       # 3*c
    add rax, rcx           # + 3*c

    mov rbx, rcx           # save c*3 temporarily? No, reload d=rcx
    mov r10, rcx           # temporary for reuse
    mov rcx, [rsp + 8]     # d (but wait: d is in rcx at entry! rsi=2nd arg, rdx=3rd, rcx=4th)
    # Correct argument mapping:
    # arg1: rdi -> a
    # arg2: rsi -> b
    # arg3: rdx -> c
    # arg4: rcx -> d
    # arg5: r8  -> e
    # arg6: r9  -> f
    # arg7: [rsp+8]  -> g (since after return addr, so rsp+8)
    # arg8: [rsp+16] -> h

    imul rcx, rcx, 4       # 4*d? No: need -4*d; do later
    sub rax, rcx           # a -2b +3c -4d (if we compute 4*d now)
    # But above rcx was overwritten by d's value — correct.

    mov rcx, r8            # e
    imul rcx, rcx, 5       # 5*e
    add rax, rcx           # +5*e

    mov rcx, r9            # f
    imul rcx, rcx, 6       # 6*f
    sub rax, rcx           # -6*f

    mov rcx, [rsp + 8]     # g
    imul rcx, rcx, 7       # 7*g
    add rax, rcx           # +7*g

    mov rcx, [rsp + 16]    # h
    imul rcx, rcx, 8       # 8*h
    sub rax, rcx           # -8*h

    add rsp, 16
    pop rbx
    pop rbp
    ret
