    movsxd rbx, esi          # sign-extend b (arg2) to 64-bit
    imul rbx, rbx, 2         # rbx = 2*b
    sub rax, rbx             # rax = a - 2*b

    movsxd rcx, edx          # sign-extend c (arg3) to 64-bit
    imul rcx, rcx, 3         # rcx = 3*c
    add rax, rcx             # rax += 3*c

    movsxd rdx, ecx          # sign-extend d (arg4) to 64-bit
    imul rdx, rdx, 4         # rdx = 4*d
    sub rax, rdx             # rax -= 4*d

    movsxd rcx, r8d          # sign-extend e (arg5) to 64-bit
    imul rcx, rcx, 5         # rcx = 5*e
    add rax, rcx             # rax += 5*e

    movsxd rdx, r9d          # sign-extend f (arg6) to 64-bit
    imul rdx, rdx, 6         # rdx = 6*f
    sub rax, rdx             # rax -= 6*f

    mov r8, qword ptr [rsp+8]   # load g (7th arg)
    movsxd r8, r8d              # sign-extend g to 64-bit
    imul r8, r8, 7              # r8 = 7*g
    add rax, r8                 # rax += 7*g
