sum8:
    push rbp
    push rbx
    sub rsp, 16
    mov eax, edi           # a in lower 32 bits (zero-extended)
    imul esi, esi, 2       # 2*b
    sub eax, esi           # a - 2*b

    movsxd rsi, edx        # sign-extend c to 64-bit in rsi
    imul si, si, 3         # 3*c (32-bit multiply, then sign-extend)
    add eax, si            # + 3*c

    mov
