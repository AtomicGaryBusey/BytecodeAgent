sum8:
    push rbp
    mov rbp, rsp
    sub rsp, 16              # allocate space for saving registers
    mov [rbp-8], rbx         # save rbx
    mov [rbp-16], r12        # save r12

    movsxd rax, edi          # a (sign-extend 32-bit to 64)
    movsxd rbx, esi          # b
    imul rbx, rbx, 2         # 2*b
    sub rax, rbx             # a - 2*b

    movsxd rcx, edx          # c
    imul rcx, rcx, 3         # 3*c
    add rax, rcx             # += 3*c

    movsxd rdx, ecx          # d
    imul rdx, rdx, 4         # 4*d
    sub rax, rdx             # -= 4*d

    movsxd rcx, r8d          # e
    imul rcx, rcx, 5         # 5*e
    add rax, rcx             # += 5*e

    movsxd rdx, r9d          # f
    imul rdx, rdx, 6         # 6*f
    sub rax, rdx             # -= 6*f

    mov r12, qword ptr [rbp+24]   # g (7th arg: rbp+8 + 16 = rbp+24)
    movsxd r12, r12d              # sign-extend
    imul r12, r12, 7              # 7*g
    add rax, r12                  # += 7*g

    mov r12, qword ptr [rbp+32]   # h (8th arg: rbp+8 + 24 = rbp+32)
    movsxd r12, r12d              # sign-extend
    imul r12, r12, 8              # 8*h
    sub rax, r12                  # -= 8*h

    mov rbx, [rbp-8]         # restore rbx
    mov r12, [rbp-16]        # restore r12
    leave                    # mov rsp, rbp; pop rbp
    ret
