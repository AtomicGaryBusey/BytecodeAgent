export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%count =l copy 0
	jmp @loop
@loop
	%cond =w csltl %count, %n
	jnz %cond, @body, @done
@body
	%off =l mul %count, 8
	%p =l add %a, %off
	%val =l loadl %p
	%zero =w ceql %val, 0
	jnz %zero, @exit, @next
@next
	%count =l add %count, 1
	jmp @loop
@exit
	ret %count
@done
	ret %n
}
