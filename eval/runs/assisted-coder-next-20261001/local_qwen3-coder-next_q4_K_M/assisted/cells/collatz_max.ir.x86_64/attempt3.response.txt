export function l $collatz_max(l %n) {
@start
	%max_steps =l copy 0
	%i =l copy 1
@outer_loop
	%cond =w csltl %i, %n
	jnz %cond, @inner_body, @done
@inner_body
	%x =l copy %i
	%steps =l copy 0
@collatz_loop
	%eq_one =w ceql %x, 1
	jnz %eq_one, @end_collatz, @compute_next
@end_collatz
	%more_steps =w csltl %max_steps, %steps
	jnz %more_steps, @update_max, @skip_update
@update_max
	%max_steps =l copy %steps
@skip_update
	%i =l add %i, 1
	jmp @outer_loop
@compute_next
	%rem =l rem %x, 2
	%is_zero =w ceql %rem, 0
	jnz %is_zero, @even_branch, @odd_branch
@even_branch
	%x_new =l udiv %x, 2
	jmp @after_compute
@odd_branch
	%tmp =l mul %x, 3
	%x_new =l add %tmp, 1
@after_compute
	%x =l copy %x_new
	%steps =l add %steps, 1
	jmp @collatz_loop
@done
	ret %max_steps
}
