isqrt:
    # Handle special cases for small n
    cmp rdi, 1
    jbe .Lsmall_result
    mov rbx, rdi          # save n in callee-saved reg
    # Initial bounds: low = 0, high = min(n, 2^32)
    xor eax, eax          # low = 0
    mov rcx, 1
    shl rcx, 32           # high = 2^32 (upper bound for sqrt(2^64-1))
    cmp rdi, rcx
    ja .Luse_upper_bound
    mov rcx, rdi          # high = n if n < 2^32
.Luse_upper_bound:

.Lloop:
    # Binary search: find mid = (low + high) / 2 without overflow
    mov rdx, rcx
    sub rdx, rax          # high - low
    cmp rdx, 1
    jbe .Ldone            # if high - low <= 1, exit loop

    shr rdx, 1            # (high - low) / 2
    add rdx, rax          # mid = low + (high - low)/2
    
    # Compute mid * mid safely using 64-bit multiplication
    mov rsi, rdx          # save mid in callee-saved reg
    xor rdx, rdx
    mul rsi               # rdx:rax = mid * mid
    cmp rdx, rbx          # compare high parts
    ja .Lmid_too_large    # if product > n (high part larger)
    jb .Lmid_ok           # if product < n in high part
    # High parts equal, check low part
    cmp rax, rbx
    jbe .Lmid_ok          # mid*mid <= n

.Lmid_too_large:
    mov rcx, rsi          # high = mid (since mid is too large)
    jmp .Lloop
    
.Lmid_ok:
    mov rax, rsi          # low = mid (since mid*mid <= n)
    jmp .Lloop

.Ldone:
    # At exit: low + 1 >= high, and high is the answer
    mov rax, rcx
    dec rax               # floor(sqrt(n)) = high - 1
    cmp rbx, 1            # special case: if n == 1, result should be 1
    je .Lfix_one
    jmp .Lfinal_ret

.Lsmall_result:
    mov rax, rdi          # for n = 0 or 1, return n
    ret

.Lfix_one:
    inc rax               # adjust for n == 1 case
.Lfinal_ret:
    ret
