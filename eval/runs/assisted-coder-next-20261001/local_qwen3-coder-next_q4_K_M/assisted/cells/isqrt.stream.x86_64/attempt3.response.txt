isqrt:
    # Handle special cases for small n
    cmp rdi, 1
    ja .Lcheck_large
    mov rax, rdi
    ret

.Lcheck_large:
    # Binary search: low = 0, high = min(n, 2^32)
    xor eax, eax          # low = 0 (rax)
    mov rbx, rdi          # save n in rbx (callee-saved)
    mov rcx, 1
    shl rcx, 32           # high = 2^32 (upper bound for sqrt(2^64-1))
    cmp rdi, rcx
    ja .Luse_2pow32       # if n >= 2^32, keep high = 2^32
    mov rcx, rdi          # otherwise high = n
.Luse_2pow32:

.Lloop:
    cmp rax, rcx
    jge .Ldone            # low >= high, done
    # mid = (low + high) / 2 without overflow
    mov rdx, rax
    sub rdx, rcx          # low - high
    sar rdx, 1            # (low - high) / 2
    add rdx, rcx          # mid = high + (low - high)/2 = (low+high)/2
    mov rsi, rdx          # save mid

    # Compute mid * mid safely
    xor rdx, rdx          # clear rdx for mul
    mov rax, rsi          # mid in rax
    mul rsi               # rdx:rax = mid * mid (unsigned multiply)
    
    # Compare rdx:rax with rbx:0 (n)
    cmp rdx, rbx          # compare high 64 bits of product with n
    ja .Lmid_too_large    # if product > n (high part exceeds n)
    jb .Lmid_ok           # if product < n in high part
    # High parts equal, check low part
    cmp rax, rbx
    jbe .Lmid_ok          # mid*mid <= n

.Lmid_too_large:
    mov rcx, rdx          # high = mid - 1 (since mid is too large)
    dec rcx
    jmp .Lloop
    
.Lmid_ok:
    mov rax, rsi          # result = mid
    inc rcx               # low = mid + 1
    jmp .Lloop

.Ldone:
    # rax currently holds the last mid that was ok (or we need to adjust)
    # Actually, when loop exits, high is the answer since we want floor(sqrt(n))
    # But our logic sets result in rsi and advances low past high
    # Let's fix: when low >= high, high is the answer because we've been keeping high such that high*high <= n or adjusting accordingly
    
    # Revisit: In this binary search, when mid*mid <= n, we set low = mid+1,
    # so at exit low > high and high is the largest value with high*high <= n
    mov rax, rcx          # return high
    dec rax               # adjust since we might have overshoot
    cmp rcx, 1
    jg .Lfinal_check
    xor eax, eax
.Lfinal_check:
    ret
