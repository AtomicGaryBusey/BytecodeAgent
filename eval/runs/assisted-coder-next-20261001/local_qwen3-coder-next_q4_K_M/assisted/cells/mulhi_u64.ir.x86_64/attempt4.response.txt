export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l copy %a
	%bl =l copy %b

	%low32_a =l and %al, 4294967295
	%high32_a =l shr %al, 32
	%low32_b =l and %bl, 4294967295
	%high32_b =l shr %bl, 32

	%prod_low_low =l mul %low32_a, %low32_b
	%prod_low_high =l mul %low32_a, %high32_b
	%prod_high_low =l mul %high32_a, %low32_b
	%prod_high_high =l mul %high32_a, %high32_b

	%mid_sum1 =l add %prod_low_high, %prod_high_low
	%mid_carry =l and %mid_sum1, 4294967295
	%mid_shifted =l shr %mid_sum1, 32

	%low_result =l add %prod_low_low, %mid_carry
	%carry_flag =w ultl %low_result, %mid_carry
	%carry_ext =l extuw %carry_flag

	%final_high =l add %prod_high_high, %mid_shifted
	%result_high =l add %final_high, %carry_ext

	ret %result_high
}
