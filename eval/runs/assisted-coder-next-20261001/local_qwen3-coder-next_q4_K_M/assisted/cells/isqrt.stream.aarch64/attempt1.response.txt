isqrt:
    // Handle special case n == 0
    cbz x0, .Lzero

    // Initialize binary search bounds
    mov x1, #1          // low = 1
    mov x2, #30         // bit position for high bound start
.Lfind_high_bit:
    tst x0, x1
    cbnz x1, .Lfound_high_bit
    add x2, x2, #1
    lsl x1, x1, #1
    cmp x1, #0          // check if we've wrapped around (x1 overflowed)
    b.ne .Lfind_high_bit

.Lfound_high_bit:
    // Set high bound: 2^(bit/2 + 1) - 1, but safer to compute as 1 << ((bit+1)/2)
    // Since x0 has highest bit at position x2 (0-indexed), sqrt max is at most 2^((x2+1)/2)
    add x3, x2, #1      // x3 = x2 + 1
    lsr x4, x3, #1      // x4 = (x2+1)/2
    mov x5, #1          // high = 1 << x4, but need to handle large shifts safely
    cmp x4, #63
    csel x5, x5, xzr, gt // if x4 >= 64, set to 0 (but sqrt(2^64-1) = 2^32-1 so max x4=32)
    clz x6, x5          // x6 = leading zeros in x5 = 63 for x5=1
    sub x7, x6, x4      // shift amount = (63 - x4) -> need lsl by x4 means ror by (63-x4)? no.
    // Simpler: use movz/movk to build high bound as 1 << ((x2+1)/2)
    cbnz x4, .Lbuild_high
    b .Lbinary_search

.Lbuild_high:
    // Build high = 1 << x4 using shift instruction
    mov x6, #1
    lsl x5, x6, x4      // high = 1 << ((x2+1)/2)
    // Clamp high to at most 0x100000000 (but we want 2^32 which is too big for sqrt of uint64, so max high=0xFFFFFFFF)
    cmp x5, #0x100000000
    c
