isqrt:
    cbz x0, .Lzero

    // Use Newton's method with careful overflow handling
    mov x1, x0          // initial guess: n (will be too high for n>1)
    cmp x1, #2
    b.ls .Ldone         // if n <= 1, return n

.Lnewton:
    // r' = (r + n/r) / 2
    mov x3, x0          // save original n
    clz x4, x1          // count leading zeros of current guess
    lsl x5, x1, x4      // normalize to avoid division overflow: multiply n by 2^clz(x1)
    lsr x6, x5, #32     // high 32 bits of (n << clz)
    lsr x7, x1, #32     // high 32 bits of guess
    cbz x7, .Ldiv_small // if guess < 2^32, use simpler division

    // Compute n/r using 64-bit division: but we need to avoid overflow in n/r
    // Instead: use udiv which is safe when r <= n (true after first step)
    udiv x8, x0, x1     // q = n / r
    add x9, x1, x8      // r + q
    lsr x1, x9, #1      // new r = (r + q) / 2
    cmp x1, x8          // check if converged: r_new <= q means we've hit floor
    b.le .Ldone

    // Check for oscillation or convergence
    cmp x1, x0          // new guess should be less than initial n (except n=0,1 handled)
    b.hi .Lnewton       // if r_new > n, something wrong; but Newton should monotonically decrease after first step

    b .Lnewton

.Ldiv_small:
    // For small guesses (< 2^32), division is safe
    udiv x8, x0, x1     // q = n / r
    add x9, x1, x8      // r + q
    lsr x1, x9, #1      // new r = (r + q) / 2
    cmp x1, x8          // check if converged: r_new <= q
    b.le .Ldone
    cmp x1, x0
    b.hi .Lnewton
    b .Lnewton

.Ldone:
    // Verify result: if (r+1)^2 <= n then r++ (but careful of overflow)
    mov x2, x0          // save n for verification
    add x3, x1, #1      // r + 1
    mul x4, x3, x3      // (r+1)^2 - may overflow but we check after
    cmp x4, x2
    b.hi .Lfinal        // if (r+1)^2 > n, then r is final

    // (r+1)^2 <= n means r+1 is better; but for uint64_t, (r+1)^2 might wrap around and appear small
    // So check overflow: if x3 has bit 32 set, then (r+1) >= 2^32 => (r+1)^2 overflows 64 bits only if r+1 > 2^32
    clz x5, x3
    cmp x5, #32
    b.lo .Lfinal        // if leading zeros < 32 (i.e., value >= 2^32), then (r+1)^2 would overflow but we know n < 2^64 so r must be max

    // Since (r+1)^2 computed and <= n, try incrementing
    mov x0, x3
    ret

.Lfinal:
    // Final adjustment: ensure r*r <= n < (r+1)*(r+1)
    mul x2, x1, x1
    cmp x2, x0
    b.ls .Lreturn       // if r^2 <= n, we're good
    sub x0, x1, #1      // else decrement
    ret

.Lzero:
    mov x0, #0
.Lreturn:
    ret
