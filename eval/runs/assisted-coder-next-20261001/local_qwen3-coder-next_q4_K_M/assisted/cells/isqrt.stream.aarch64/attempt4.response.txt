isqrt:
    cbz x0, .Lzero

    // Handle small n <= 1
    cmp x0, #1
    b.ls .Ldone_small

    // Initial guess: 2^(32 - clz(n)/2) ≈ sqrt(n)
    clz x1, x0          // count leading zeros of n
    sub x2, x1, #32     // x2 = clz - 32 (negative if n >= 2^32)
    neg x2, x2          // x2 = 32 - clz
    lsrv x3, x0, x2     // shift n right by ~half its bits to get initial guess
    add x1, x3, #1      // ensure guess >= 1

.Lnewton:
    udiv x4, x0, x1     // q = n / r (safe since r <= n and r > 0)
    add x5, x1, x4      // r + q
    lsr x6, x5, #1      // new_r = (r + q) / 2

    // Check convergence: if new_r >= r then done (Newton's method decreases monotonically here)
    cmp x6, x1
    b.ge .Ldone_newton

    // Update r and iterate
    mov x1, x6
    b .Lnewton

.Ldone_newton:
    // Verify: if (r+1)^2 <= n then r++ is better
    add x5, x1, #1      // r+1
    mul x6, x5, x5      // (r+1)^2
    cmp x6, x0
    b.gt .Ldone_final   // if (r+1)^2 > n, current r is correct
    mov x1, x5          // otherwise increment r
.Ldone_final:
    mov x0, x1
    ret

.Lzero:
    mov x0, #0
    ret

.Ldone_small:
    mov x0, #1
    ret
