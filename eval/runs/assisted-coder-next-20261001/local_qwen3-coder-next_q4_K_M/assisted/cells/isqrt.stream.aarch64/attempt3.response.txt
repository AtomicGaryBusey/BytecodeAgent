isqrt:
    cbz x0, .Lzero

    // Handle small n <= 1
    cmp x0, #1
    b.ls .Ldone_small

    // Initial guess: 2^(32 - clz(n)/2) ≈ sqrt(n)
    clz x1, x0          // count leading zeros of n
    // Number of bits = 64 - clz
    sub x2, x1, #32     // x2 = clz - 32 (negative if n >= 2^32)
    neg x2, x2          // x2 = 32 - clz = half the bit position of MSB roughly
    lsrv x1, x0, x2     // shift n right by ~half its bits to get initial guess
    add x1, x1, #1      // ensure guess >= 1

.Lnewton:
    udiv x3, x0, x1     // q = n / r (safe since r <= n and r > 0)
    add x4, x1, x3      // r + q
    lsrv x5, x4, #1     // new_r = (r + q) / 2

    cmp
