memmove_i64:
    // Check if n is zero
    cbz x2, .Ldone

    // Save return address and frame pointer, set up stack frame
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Save callee-saved registers we'll use: x19-x22
    stp x19, x20, [sp, #-16]!
    str x21, [sp, #16]
    str x22, [sp, #24]

    mov x19, x0          // dst
    mov x20, x1          // src
    mov x21, x2          // n (count)

    // Check for identical ranges
    cmp x19, x20
    b.eq .Lcleanup_and_done

    // Compute total byte offset = n * 8
    lsl x22, x21, #3     // x22 = n*8

    // dst_end = dst + n*8 (address just past last element)
    add x22, x19, x22    // x22 = dst_end

    // Check if dst ends before or at src start: no overlap, copy forward
    cmp x22, x20
    b.le .Lforward_copy

    // Compute src_end = src + (n-1)*8 for backward overlap check
    sub x22, x21, #1     // n - 1
    lsl x22, x22, #3     // (n-1)*8
    add x22, x20, x22    // src_end = src + (n-1)*8

    // If src_end <= dst: safe to copy forward (no overlap or non-overlapping)
    cmp x22, x19
    b.le .Lforward_copy

    // Else, overlapping with dst after src: must copy backward
.Lbackward_copy:
    mov x22, x19         // dst
    lsl x19, x21, #3     // n*8 (recompute for backward start)
    add x22, x22, x19    // dst_end = dst + n*8

    mov x20, x1          // src
    add x20, x20, x19    // src_end = src + n*8

.Lbackward_loop:
    subs x21, x21, #1
    b.mi .Lcleanup_and_done  // done when n < 0

    sub x20, x20, #8       // move src pointer back
    ldr x9, [x20]          // load element
    sub x22, x22, #8       // move dst pointer back
    str x9, [x22]          // store element
    b .Lbackward_loop

.Lforward_copy:
    mov x22, x19         // dst pointer (working)
    mov x20, x1          // src pointer (working)

.Lforward_loop:
    ldr x9, [x20], #8    // load and increment src
    str x9, [x22], #8    // store and increment dst
    subs x21, x21, #1
    b.ne .Lforward_loop

.Lcleanup_and_done:
    ldr x22, [sp, #24]
    ldr x21, [sp, #16]
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16

.Ldone:
    mov x0, #0
    ret
