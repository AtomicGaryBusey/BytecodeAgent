memmove_i64:
    // Check if n is zero
    cbz x2, .Ldone

    // Save callee-saved registers we use (x19-x23)
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #16]
    mov x19, x0          // dst
    mov x20, x1          // src
    mov x21, x2          // n (count)

    // Compute src_end = src + (n-1)*8 for overlap check
    sub x22, x21, #1     // n-1
    lsl x22, x22, #3     // (n-1)*8
    add x22, x20, x22    // src_end = src + (n-1)*8

    cmp x19, x22
    b.ge .Lforward       // dst >= src_end: safe to copy forward

.Lbackward:
    // Backward copy: start from the end and move backwards
    lsl x23, x21, #3     // total bytes = n*8
    add x19, x19, x23    // dst + n*8 (one past last element)
    add x20, x20, x23    // src + n*8 (one past last element)

.Lbackward_loop:
    subs x21, x21, #1
    b.eq .Lbackward_last
    ldr x9, [x20, #-8]!
    str x9, [x19, #-8]!
    b .Lbackward_loop

.Lbackward_last:
    ldr x9, [x20, #-8]
    str x9, [x19, #-8]

    b .Ldone_cleanup

.Lforward:
    // Forward copy: start from beginning and move forwards
.Lforward_loop:
    cbz x21, .Ldone_cleanup
    ldr x9, [x20], #8
    str x9, [x19], #8
    subs x21, x21, #1
    b.ne .Lforward_loop

.Ldone_cleanup:
    // Restore callee-saved registers
    ldp x21, x22, [sp, #16]
    ldp x19, x20, [sp], #16

.Ldone:
    mov x0, #0
    ret
