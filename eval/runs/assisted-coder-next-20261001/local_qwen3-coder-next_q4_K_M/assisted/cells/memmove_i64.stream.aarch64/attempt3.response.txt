memmove_i64:
    // Check if n is zero
    cbz x2, .Ldone

    // Save callee-saved registers we'll use: x19, x20, x21, x22, x23
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #16]
    str x23, [sp, #32]

    mov x19, x0          // dst
    mov x20, x1          // src
    mov x21, x2          // n (count)

    // Compute offset for backward copy: total bytes = n * 8
    lsl x22, x21, #3     // x22 = n*8

    // For overlap check: dst_end = dst + n*8, src_start = src, src_end = src + (n-1)*8
    add x23, x19, x22    // dst_end = dst + n*8

    cmp x19, x20         // compare dst and src
    b.eq .Ldone          // identical: nothing to do

    cmp x23, x20         // dst_end <= src? then no overlap (dst before src)
    b.le .Lforward       // safe to copy forward

    // Check if src_end <= dst for backward copy safety
    sub x22, x21, #1     // n-1
    lsl x22, x22, #3     // (n-1)*8
    add x22, x20, x22    // src_end = src + (n-1)*8

    cmp x22, x19         // src_end <= dst?
    b.gt .Lbackward      // backward if src overlaps and starts after dst

.Lforward:
    mov x23, x0          // use x23 as working dst pointer
    mov x22, x1          // use x22 as working src pointer
    cbz x21, .Ldone_cleanup

.Lforward_loop:
    ldr x9, [x22], #8
    str x9, [x23], #8
    subs x21, x21, #1
    b.ne .Lforward_loop
    b .Ldone_cleanup

.Lbackward:
    mov x23, x0          // dst
    add x23, x23, x22    // dst_end (one past last element)
    mov x22, x1          // src
    add x22, x22, x22, lsl #0  // recompute total offset for backward start
    add x22, x22, x23, lsr #63   // this is redundant; use saved x22 (n*8)
    sub x22, x19, x22, lsr #63  // incorrect; fix:

.Lbackward_recalc:
    mov x22, x0          // dst
    add x22, x22, x21, lsl #3   // dst_end = dst + n*8
    mov x23, x1          // src
    add x23, x23, x21, lsl #3   // src_end = src + n*8

.Lbackward_loop:
    subs x21, x21, #1
    b.mi .Ldone_cleanup  // done when negative (after last element)
    ldr x9, [x23, #-8]!
    str x9, [x22, #-8]!
    b .Lbackward_loop

.Ldone_cleanup:
    ldp x21, x22, [sp, #16]
    ldr x23, [sp, #32]
    ldp x19, x20, [sp], #16

.Ldone:
    mov x0, #0
    ret
