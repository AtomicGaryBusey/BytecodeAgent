export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =l copy 0
	%cmp =w csltl %n, %zero
	jnz %cmp, @done, @main
@main
	%p_src_end =l add %src, %n
	%p_dst_end =l add %dst, %n
	%cmp_ptr =w csltl %dst, %src
	jnz %cmp_ptr, @backward, @forward
@forward
	%i =l copy 0
@f_loop
	%more =w csltl %i, %n
	jnz %more, @f_body, @f_done
@f_body
	%off_src_f =l mul %i, 8
	%off_dst_f =l mul %i, 8
	%p_src_f =l add %src, %off_src_f
	%p_dst_f =l add %dst, %off_dst_f
	%val =l loadl %p_src_f
	storel %val, %p_dst_f
	%i =l add %i, 1
	jmp @f_loop
@f_done
	ret %zero
@backward
	%i =l copy 0
@b_loop
	%more_b =w csltl %i, %n
	jnz %more_b, @b_body, @b_done
@b_body
	%n_minus_i =l sub %n, %i
	%off_src_b =l mul %n_minus_i, 8
	%off_dst_b =l mul %n_minus_i, 8
	%p_src_b =l add %src, %off_src_b
	%p_dst_b =l add %dst, %off_dst_b
	%val_b =l loadl %p_src_b
	storel %val_b, %p_dst_b
	%i =l add %i, 1
	jmp @b_loop
@b_done
	ret %zero
@done
	ret %zero
}
