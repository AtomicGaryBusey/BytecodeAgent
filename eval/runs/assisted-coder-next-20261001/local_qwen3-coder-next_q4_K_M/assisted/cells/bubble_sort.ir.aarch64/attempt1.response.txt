export function l $bubble_sort(l %a, l %n) {
@start
	%outer =l copy 0
	%n_val =l copy %n
	%n_dec =l sub %n_val, 1
@outer_loop
	%cond_outer =w csltl %outer, %n_dec
	jnz %cond_outer, @inner_init, @done
@inner_init
	%inner =l copy 0
	%n_dec_outer =l sub %n_dec, %outer
@inner_loop
	%cond_inner =w csltl %inner, %n_dec_outer
	jnz %cond_inner, @loop_body, @next_outer
@loop_body
	%off_i =l mul %inner, 8
	%addr_i =l add %a, %off_i
	%val_i =l loadl %addr_i
	%off_next =l add %addr_i, 8
	%val_next =l loadl %off_next
	%swap_needed =w csltl %val_next, %val_i
	jnz %swap_needed, @do_swap, @no_swap
@do_swap
	storel %val_next, %addr_i
	storel %val_i, %off_next
@no_swap
	%inner =l add %inner, 1
	jmp @inner_loop
@next_outer
	%outer =l add %outer, 1
	jmp @outer_loop
@done
	%zero =l copy 0
	ret %zero
}
