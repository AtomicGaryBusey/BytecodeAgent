export function l $sum_to(l %n) {
@start
	%result =l copy 0
	%i =l copy 1
	%is_non_positive =w csltl %n, 1
	jnz %is_non_positive, @done, @loop
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @after
@body
	%result =l add %result, %i
	%i =l add %i, 1
	jmp @loop
@after
	%result =l add %result, %i
@done
	ret %result
}
