export function l $sum_to(l %n) {
@start
	%result =l copy 0
	%i =l copy 1
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
%cmp_zero =w ceqw %n, 0
	jnz %cmp_zero, @done, @check_negative
@check_negative
	%is_neg =w csltl %n, 0
	jnz %is_neg, @done, @body
@body
	%result =l add %result, %i
	%i =l add %i, 1
	jmp @loop
@done
	ret %result
}
