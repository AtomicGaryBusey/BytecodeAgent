export function l $fib(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @ret0, @check1
@ret0
	ret 0
@check1
	%is_one =w ceql %n, 1
	jnz %is_one, @ret1, @compute
@ret1
	ret 1
@compute
	%a =l copy 0
	%b =l copy 1
	%i =l copy 2
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%new_val =l add %a, %b
	%a =l copy %b
	%b =l copy %new_val
	%i =l add %i, 1
	jmp @loop
@done
	ret %b
}
