fib_rec:
    cmp rdi, 1
    jle .Lbase
    sub rsp, 24                # align stack: entry rsp=8 mod 16, sub 24 -> rsp=8 mod 16 again (odd number of pushes/sub)
    mov [rsp+16], rdi          # save n at [rsp+16]
    
    mov rdi, [rsp+16]          # load n
    dec rdi                    # n-1 for first recursive call
    call fib_rec               # fib(n-1)
    mov rbx, rax               # save result of fib(n-1) in callee-saved register
    
    mov rdi, [rsp+16]          # load original n again
    dec rdi                    # n-2 for second recursive call
    call fib_rec               # fib(n-2)
    
    add rax, rbx               # fib(n-1) + fib(n-2)
    
    add rsp, 24                # restore stack
    ret

.Lbase:
    cmp rdi, 0
    jne .Lone
    xor eax, eax
    ret

.Lone:
    mov eax, 1
    ret
