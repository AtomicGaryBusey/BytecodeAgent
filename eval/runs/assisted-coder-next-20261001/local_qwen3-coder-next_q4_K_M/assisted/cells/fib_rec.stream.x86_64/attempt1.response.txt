fib_rec:
    cmp rdi, 1
    jle .Lbase
    # n >= 2: compute fib(n-1) + fib(n-2)
    push rbx
    sub rsp, 8              # align stack (rsp was 8 mod 16, now 0 mod 16)
    mov rbx, rdi            # save original n in rbx
    dec rdi                 # n-1
    call fib_rec            # fib(n-1)
    add rsp, 8              # restore stack (rsp = entry value + 8)
    push rax                # save result of fib(n-1) on stack
    mov rdi, rbx            # restore original n
    dec rdi                 # n-2
    call fib_rec            # fib(n-2)
    pop rbx                 # load fib(n-1) into rbx (callee-saved)
    add rax, rbx            # fib(n-1) + fib(n-2)
    ret

.Lbase:
    cmp rdi, 0
    jne .Lone
    xor eax, eax
    ret

.Lone:
    mov eax, 1
    ret
