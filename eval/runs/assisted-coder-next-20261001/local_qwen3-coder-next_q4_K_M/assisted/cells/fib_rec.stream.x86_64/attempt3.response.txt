fib_rec:
    cmp rdi, 1
    jle .Lbase
    # n >= 2: compute fib(n-1) + fib(n-2)
    push rbx
    push rbp
    mov rbp, rsp
    sub rsp, 32           # align stack to 16 bytes (rsp was 8 mod 16, now 0 mod 16)
    mov [rbp-8], rdi      # save original n
    
    mov rdi, [rbp-8]      # load n
    dec rdi               # n-1 for first recursive call
    call fib_rec          # fib(n-1), rsp is aligned at call (16-byte)
    mov rbx, rax          # save result of fib(n-1) in callee-saved register
    
    mov rdi, [rbp-8]      # load original n again
    dec rdi               # n-2 for second recursive call
    call fib_rec          # fib(n-2), rsp is aligned at call
    
    add rax, rbx          # fib(n-1) + fib(n-2)
    
    mov rsp, rbp
    pop rbp
    pop rbx
    ret

.Lbase:
    cmp rdi, 0
    jne .Lone
    xor eax, eax
    ret

.Lone:
    mov eax, 1
    ret
