fib_rec:
    cmp rdi, 1
    jle .Lbase
    # n >= 2: compute fib(n-1) + fib(n-2)
    push rbx
    sub rsp, 8              # align stack (rsp was 8 mod 16, now 0 mod 16 before call)
    mov rbx, rdi            # save original n in rbx
    dec rdi                 # n-1 for first recursive call
    call fib_rec            # fib(n-1), rsp is aligned at call (16-byte)
    add rsp, 8              # restore stack to entry + 8
    push rax                # save fib(n-1) result on stack
    mov rdi, rbx            # restore original n
    dec rdi                 # n-2 for second recursive call
    sub rsp, 8              # make space and re-align (rsp was entry+8, now entry+0 mod 16)
    call fib_rec            # fib(n-2), rsp is aligned at call
    add rsp, 8              # restore stack to entry + 8
    pop rbx                 # load fib(n-1) into rbx (callee-saved register)
    add rax, rbx            # fib(n-1) + fib(n-2)
    pop rbx                 # restore rbx from initial push
    ret

.Lbase:
    cmp rdi, 0
    jne .Lone
    xor eax, eax
    ret

.Lone:
    mov eax, 1
    ret
