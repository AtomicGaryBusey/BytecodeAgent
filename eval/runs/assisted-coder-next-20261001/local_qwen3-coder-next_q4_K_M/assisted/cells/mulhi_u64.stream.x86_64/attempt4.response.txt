mulhi_u64:
    push rbx
    push r12
    push r13
    sub rsp, 8              # align stack to 16-byte (rsp was 8 mod 16)
    xor eax, eax
    mov rbx, rdi            # save a in rbx
    mov r12, rsi            # save b in r12
    mov rax, rsi            # copy b
    shr rsi, 32             # b_high = b >> 32
    and eax, 0xffffffff     # b_low = b & 0xffffffff (clears upper half)
    mov rdx, rbx            # a in rdx for mul
    mul rsi                 # rdx:rax = a * b_high
    mov r13, rax            # store low part of a*b_high
    mov rdi, rdx            # high part of a*b_high
    xor eax, eax
    mov rdx, rbx            # a in rdx for mul
    mul rax                 # rdx:rax = a * b_low (b_low in lower 32 bits)
    add r13, rax            # add low parts: middle terms
    adc rdi, 0              # propagate carry into high part
    mov rax, rbx            # load a for final mul
    mov rcx, r12            # load b for final mul
    xor edx, edx
    mul rcx                 # rdx:rax = a * b (full product)
    add rdi, rdx            # add high part of low*low to accumulated high parts
    mov rax, rdi            # return high 64 bits
    add rsp, 8              # deallocate stack space
    pop rbx
    pop r12
    pop r13
    ret
