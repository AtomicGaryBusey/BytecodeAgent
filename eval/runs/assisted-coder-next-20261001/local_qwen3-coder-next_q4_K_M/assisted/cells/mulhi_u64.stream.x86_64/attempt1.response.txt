mulhi_u64:
    push rbx
    push rbp
    mov rbp, rsp
    sub rsp, 32             # allocate space for 256-bit product (4 qwords)
    xor eax, eax            # clear upper bits
    mul rdi                 # rdx:rax = a * low(b) [b is in rsi]
    mov rbx, rax            # save low part of low multiplication
    mov rcx, rdx            # save high part of low multiplication
    xor eax, eax            # clear for second multiply
    mul rsi                 # rdx:rax = a * high(b) [b is in rsi]
    add rax, rcx            # add high part from first mul to low part of this result
    adc rdx, 0              # propagate carry
    mov qword ptr [rbp-16], rax   # store low 64 bits of product (will be overwritten)
    mov qword ptr [rbp-8], rdx    # store high 64 bits so far
    xor eax, eax            # clear for third multiply
    mul rbx                 # rdx:rax = b_low * a_low -> need to add this to middle part
    add qword ptr [rbp-16], rax   # add low part to previous low
    adc rdx, 0              # propagate carry
    add rdx, qword ptr [rbp-8]    # add high part from b_low*a calculation
    mov rax, rdx            # final high 64 bits in rax
    leave
    pop rbx
    ret
