mulhi_u64:
    push rbx
    push rbp
    mov rbp, rsp
    sub rsp, 16             # allocate space for middle terms (2 qwords)
    xor eax, eax            # clear high bits of eax for mul
    mov rbx, rsi            # save b in rbx (callee-saved)
    mov rcx, rdi            # save a in rcx (callee-saved)
    mov rax, rbx            # copy b
    shr rbx, 32             # b_high = b >> 32
    and eax, 0xffffffff     # b_low = b & 0xffffffff (32-bit op clears upper half)
    mov rdx, rcx            # a in rdx for mul
    mul rbx                 # rdx:rax = a * b_high
    mov qword ptr [rbp-16], rax   # store low part of a*b_high
    mov rdi, rdx            # save high part of a*b_high
    xor eax, eax            # clear for next mul
    mov rdx, rcx            # a in rdx for mul
    mul rax                 # rdx:rax = a * b_low (b_low in lower 32 bits)
    add rdi, rdx            # add high part of a*b_low to saved a*b_high high part
    mov rsi, rax            # store low part of a*b_low for addition
    mov rax, qword ptr [rbp-16]   # load low part of a*b_high
    add rsi, rax            # add the two middle terms (low parts)
    adc rdi, 0              # propagate carry into high part
    mov rax, rdi            # high 64 bits of product
    leave
    pop rbx
    ret
