mulhi_u64:
    push rbx
    push rbp
    mov rbp, rsp
    sub rsp, 16             # allocate space for middle terms (2 qwords)
    xor eax, eax            # clear high bits of eax for mul
    mov rbx, rsi            # save b in rbx (callee-saved)
    mul rdi                 # rdx:rax = a * b_low (b is in rbx, but we need full b)
    mov qword ptr [rbp-16], rax   # store low 64 bits of a*b_low
    mov rcx, rdx            # save high part of a*b_low
    xor eax, eax            # clear for second multiply
    mov rdx, rbx            # move b into rdx for mul (rdx:rax = a * b)
    mov rdx, qword ptr [rbp+24]   # load full b from stack (7th arg area, but we passed only 2 args so it's not here; actually b is in rsi which we saved to rbx, but need to split b into high and low parts)
    mov rax, rdi            # a is in rdi
    mul rbx                 # rdx:rax = a * full_b (but this gives 128-bit result directly? No, mul only multiplies by one operand; we need to split b)
    # Let's restart with proper approach: b = b_low + b_high*2^32
    # Product = a*b = a*b_low + a*b_high*2^32
    # High 64 bits = high(a*b_low) + low(a*b_high) + carry from addition
    
    # Actually simpler: use the fact that we can get full product via schoolbook multiplication
    # Split b into high and low 32-bit parts
    xor rdx, rdx            # clear rdx for mul
    mov rbx, rsi            # b
    mov rcx, rdi            # a
    mov rax, rbx            # copy b
    shr rbx, 32             # b_high = b >> 32
    and rax, 0xffffffff     # b_low = b & 0xffffffff
    
    xor
