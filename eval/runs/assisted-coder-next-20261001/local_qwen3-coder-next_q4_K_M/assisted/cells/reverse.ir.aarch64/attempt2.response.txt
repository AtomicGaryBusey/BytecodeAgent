export function l $reverse(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	jmp @init

@loop
	%csltl_n_i =w csltl %i, %half_n
	jnz %csltl_n_i, @body, @done

@body
	%off_i =l mul %i, 8
	%ptr_i =l add %a, %off_i
	%val_i =l loadl %ptr_i

	%rev_idx =l sub %n, %one
	%offs_rev =l mul %rev_idx, 8
	%ptr_rev =l add %a, %offs_rev
	%val_rev =l loadl %ptr_rev

	storel %val_rev, %ptr_i
	storel %val_i, %ptr_rev

	%i =l add %i, %one
	jmp @loop

@init
	%i =l copy 0
	%half_n =l udiv %n, %two
	jmp @loop

@done
	ret %zero
}
