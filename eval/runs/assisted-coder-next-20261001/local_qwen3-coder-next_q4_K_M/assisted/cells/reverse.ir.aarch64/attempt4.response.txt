export function l $reverse(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%two =l copy 2
	jmp @init

@loop
	%cmp =w csltl %i, %half_n
	jnz %cmp, @body, @done

@body
	%off_i =l mul %i, 8
	%ptr_i =l add %a, %off_i
	%val_i =l loadl %ptr_i

	%j =l sub %n, %one
	%j =l sub %j, %i
	%off_j =l mul %j, 8
	%ptr_j =l add %a, %off_j
	%val_j =l loadl %ptr_j

	storel %val_j, %ptr_i
	storel %val_i, %ptr_j

	%i =l add %i, %one
	jmp @loop

@init
	%i =l copy 0
	%half_n =l udiv %n, %two
	jmp @loop

@done
	ret %zero
}
