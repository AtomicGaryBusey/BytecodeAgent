bmax:
    mov x1, x0                    // x1 = a (base pointer)
    ldr x2, [x1], #8              // load first element as initial max, advance ptr
    sub x3, x2, x2                // clear x3 (will hold current max)
    add x3, x3, x2                // copy first element to x3 (max so far)
    sub x4, x1, #8                // adjust back pointer (x1 points past first element)
    cmp x0, x4                    // n == 1? (if only one element, skip loop)
    b.eq .Ldone
.Lloop:
    ldr x2, [x1], #8              // load next element
    cmp x2, x3                    // compare with current max
    csel x3, x2, x3, gt           // update max if larger
    sub x0, x0, #1                // decrement counter (n--)
    cbnz x0, .Lloop               // loop until n == 0
.Ldone:
    mov x0, x3                    // return max in x0
    ret                           // return to caller
