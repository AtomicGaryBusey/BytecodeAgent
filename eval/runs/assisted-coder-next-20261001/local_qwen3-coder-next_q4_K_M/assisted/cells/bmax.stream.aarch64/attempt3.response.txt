bmax:
    mov x1, x0                    // x1 = base pointer a
    ldr x2, [x1], #8              // load first element, advance ptr
    mov x3, x2                    // initialize max with first element
    subs x0, x0, #2               // n-2 remaining elements to check
    b.lt .Ldone                   // if n < 2, done (n was 1)
.Lloop:
    ldr x2, [x1], #8              // load next element
    cmp x2, x3                    // compare with current max
    csel x3, x2, x3, gt           // update max if larger
    subs x0, x0, #1               // decrement counter
    b.ge .Lloop                   // loop while counter >= 0
.Ldone:
    mov x0, x3                    // return max in x0
    ret                           // return to caller
