bmax:
    ldr x1, [x0], #8              // load first element, advance ptr to second
    mov x2, x1                    // initialize max with first element
    subs x0, x0, #8               // adjust pointer back by 8 (undo post-increment)
    cbz x2, .Ldone                // if n==1, skip loop (we'll check count properly below)
.Lsetup:
    mov x3, x0                    // x3 = base pointer a
    mov x4, x2                    // x4 = max
    ldr x0, [x3], #8              // load first element again for proper counting
    sub x5, xzr, #1               // x5 = -1 (for comparing)
.L
