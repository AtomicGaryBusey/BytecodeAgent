memmove_i64:
    # Handle n == 0 case
    test rdx, rdx
    jz .Ldone

    # Save callee-saved registers we might use
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov rbx, rdi        # dst
    mov rsi, rsi        # src (already in rsi)
    mov r12, rdx        # n

    # Check for overlap and direction
    lea r13, [rbx + r12*8]  # end of dst region
    lea r14, [rsi + r12*8]  # end of src region
    
    # If dst >= src and they overlap, copy backwards; else forward
    cmp rbx, rsi
    ja .Lbackward        # if dst > src, might need backward
    je .Lcheck_overlap   # if equal, check regions
    
.Lforward:
    xor r15, r15         # i = 0
.Lf_loop:
    mov rax, [rsi + r15*8]
    mov [rbx + r15*8], rax
    inc r15
    cmp r15, r12
    jb .Lf_loop
    jmp .Lrestore_and_done

.Lcheck_overlap:
    # dst == src -> no overlap, skip backward handling
    jmp .Lforward
    
.Lbackward:
    # Check if dst is before src + n*8 (i.e., non-overlapping or safe to go forward)
    cmp r13, rsi
    ja .Lbackward_only   # if dst_end > src, need backward
    jmp .Lforward        # else forward is fine

.Lbackward_only:
    mov r15, r12         # i = n (start from end)
.Lb_loop:
    dec r15
    mov rax, [rsi + r15*8]
    mov [rbx + r15*8], rax
    cmp r15, 0
    jne .Lb_loop

.Lrestore_and_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx

.Ldone:
    xor eax, eax         # return 0
    ret
