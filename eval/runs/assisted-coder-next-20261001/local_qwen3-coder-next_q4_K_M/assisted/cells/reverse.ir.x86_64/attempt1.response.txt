export function l $reverse(l %a, l %n) {
@start
	%len =l copy %n
	%i =l copy 0
	%half =l div %len, 2
@loop
	%cmp =w csltl %i, %half
	jnz %cmp, @body, @done
@body
	%off_left =l mul %i, 8
	%p_left =l add %a, %off_left
	%left_val =l loadl %p_left
	%j =l sub %len, 1
	%j =l sub %j, %i
	%off_right =l mul %j, 8
	%p_right =l add %a, %off_right
	%right_val =l loadl %p_right
	storel %right_val, %p_left
	storel %left_val, %p_right
	%i =l add %i, 1
	jmp @loop
@done
	%zero =l copy 0
	ret %zero
}
