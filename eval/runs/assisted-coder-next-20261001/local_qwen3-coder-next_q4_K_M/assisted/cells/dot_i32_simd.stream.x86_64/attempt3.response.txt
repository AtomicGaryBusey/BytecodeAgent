dot_i32_simd:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor eax, eax
    mov r9, rcx
    test r9, r9
    jz .Lscalar_cleanup
    pxor xmm4, xmm4
    xor rdi, rdi

.Lsimd_loop:
    movdqu xmm0, XMMWORD PTR [rdi + rdx*4]
    movdqu xmm1, XMMWORD PTR [rdi + rcx*4]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm2
    add rdi, 4
    cmp rdi, r9
    jb .Lsimd_loop

.Lscalar_cleanup:
    test r9, r9
    jz .Ldone
    xor rbx, rbx
    pxor xmm0, xmm0
    movq xmm1, xmm4

.Lscalar_loop:
    movsxd rax, dword ptr [rdi + rbx*4]
    movsxd rcx, dword ptr [rsi + rbx*4]
    imul rax, rcx
    add qword ptr [rsp], rax
    inc rbx
    cmp rbx, r9
    jb .Lscalar_loop

.Ldone:
    movq xmm0, qword ptr [rsp]
    paddq xmm1, xmm0
    pmovsqd xmm0, xmm1
    mov
