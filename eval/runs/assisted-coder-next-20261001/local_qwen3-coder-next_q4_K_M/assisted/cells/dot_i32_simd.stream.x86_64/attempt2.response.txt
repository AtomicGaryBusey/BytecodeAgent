dot_i32_simd:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor eax, eax
    mov rbx, rdi
    mov rsi, rsi
    xor edi, edi
    lea r9, [rip + .Lsimd_loop_start]
    test rcx, rcx
    jz .Lscalar_cleanup

.Lsimd_loop:
    movdqu xmm0, XMMWORD PTR [rbp + rdi*4]
    movdqu xmm1, XMMWORD PTR [rbx + rdi*4]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm2
    add rdi, 4
    cmp rdi, rcx
    jb .Lsimd_loop

.Lscalar_cleanup:
    xor rax, rax
    mov rbx, rdi
    test rcx, rcx
    jz .Ldone

.Lscalar_loop:
    movsxd r8, dword ptr [rbp + rbx*4]
    movsxd r9, dword ptr [rsi + rbx*4]
    imul r8, r9
    add rax, r8
    inc rbx
    cmp rbx, rcx
    jb .Lscalar_loop

.Ldone:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
