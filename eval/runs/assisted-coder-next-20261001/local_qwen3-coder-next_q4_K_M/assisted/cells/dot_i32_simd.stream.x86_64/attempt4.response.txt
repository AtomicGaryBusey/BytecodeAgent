dot_i32_simd:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    xor eax, eax
    test rcx, rcx
    jz .Lscalar_cleanup
    pxor xmm4, xmm4
    xor rbx, rbx

.Lsimd_loop:
    movdqu xmm0, XMMWORD PTR [rdi + rbx*4]
    movdqu xmm1, XMMWORD PTR [rsi + rbx*4]
    pmovsxdq xmm2, xmm0
    pmovsxdq xmm3, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm2
    add rbx, 4
    cmp rbx, rcx
    jb .Lsimd_loop

.Lscalar_cleanup:
    mov rax, rcx
    sub rax, rbx
    test rax, rax
    jz .Ldone
    xor rcx, rcx

.Lscalar_loop:
    movsxd rdx, dword ptr [rdi + rbx*4]
    movsxd r8, dword ptr [rsi + rbx*4]
    imul rdx, r8
    add r9, rdx
    inc rbx
    dec rcx
    jnz .Lscalar_loop

.Ldone:
    movq xmm0, xmm4
    movd edx, xmm0
    movq xmm0, qword ptr [rsp + 8]
    movd eax, xmm0
    add rax, rdx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
