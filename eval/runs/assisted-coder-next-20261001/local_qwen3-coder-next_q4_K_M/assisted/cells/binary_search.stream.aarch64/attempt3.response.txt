binary_search:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cbz x1, .Lnot_found
    mov x4, x0
    mov x5, x1
    mov x6, #0
.Lloop:
    cmp x6, x5
    b.ge .Lnot_found
    sub x7, x5, x6
    cbz x7, .Lnot_found
    lsr x8, x7, #1
    add x8, x6, x8
    ldr x9, [x4, x8, lsl #3]
    cmp x9, x2
    b.eq .Lfound
    b.lt .Lright_half
    mov x5, x8
    b .Lloop
.Lright_half:
    add x6, x8, #1
    b .Lloop
.Lfound:
    mov x0, x8
    ldp x29, x30, [sp], #16
    ret
.Lnot_found:
    mov x0, #-1
    ldp x29, x30, [sp], #16
    ret
