export function l $binary_search(l %a, l %n, l %key) {
@start
	%low =l copy 0
	%high =l copy %n
@loop
	%not_empty =w csltl %low, %high
	jnz %not_empty, @body, @not_found
@body
	%mid =l add %low, %high
	%mid =l sar %mid, 1
	%off =l mul %mid, 8
	%p =l add %a, %off
	%val =l loadl %p
	%equal =w ceql %val, %key
	jnz %equal, @found, @check_mid
@check_mid
	%less =w csltl %val, %key
	jnz %less, @right_half, @left_half
@right_half
	%low =l add %mid, 1
	jmp @loop
@left_half
	%high =l copy %mid
	jmp @loop
@found
	ret %mid
@not_found
	%neg_one =l copy -1
	ret %neg_one
}
