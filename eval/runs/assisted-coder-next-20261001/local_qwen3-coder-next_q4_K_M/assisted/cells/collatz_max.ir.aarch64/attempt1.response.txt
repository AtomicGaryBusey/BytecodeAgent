export function l $collatz_max(l %n) {
@start
	%max_steps =l copy 0
	%k =l copy 1
@loop_k
	%more_k =w csltl %k, %n
	jnz %more_k, @body_k, @done
@body_k
	%x =l copy %k
	%steps =l copy 0
@loop_collatz
	%done =w ceql %x, 1
	jnz %done, @end_collatz, @collatz_body
@end_collatz
	%more_max =w csltl %max_steps, %steps
	jnz %more_max, @update_max, @next_k
@update_max
	%max_steps =l copy %steps
	jmp @next_k
@collatz_body
	%even_test =w rem %x, 2
	%is_even =w ceql %even_test, 0
	jnz %is_even, @even_path, @odd_path
@even_path
	%x =l udiv %x, 2
	jmp @next_step
@odd_path
	%x =l add %x, 1
	%x =l mul %x, 3
	jmp @next_step
@next_step
	%steps =l add %steps, 1
	jmp @loop_collatz
@next_k
	%k =l add %k, 1
	jmp @loop_k
@done
	ret %max_steps
}
