export function l $add_overflow(l %a, l %b) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%a_neg =w csltl %a, 0
	%b_neg =w csltl %b, 0
	%sum_neg =w csltl %sum, 0
	%pos_overflow =w andw %a_neg, %b_neg
	%pos_overflow =w xoriw %pos_overflow, 1
	%pos_overflow =w andw %pos_overflow, %sum_neg
	%neg_underflow =w andw %a_neg, %b_neg
	%result =w orw %pos_overflow, %neg_underflow
	ret %result
}
