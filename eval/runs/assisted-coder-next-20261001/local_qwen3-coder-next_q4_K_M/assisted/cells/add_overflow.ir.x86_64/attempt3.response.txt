export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%a_neg =w csltl %a, 0
	%b_neg =w csltl %b, 0
	%sum_neg =w csltl %sum, 0
	%both_pos =w and %a_neg, %b_neg
	%both_pos =w xor %both_pos, 1
	%pos_overflow =w and %both_pos, %sum_neg
	%neg_underflow =w and %a_neg, %b_neg
	%result =w or %pos_overflow, %neg_underflow
	ret %result
}
