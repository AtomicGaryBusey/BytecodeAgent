map_sum:
    push rbx
    push rbp
    push r12
    sub rsp, 8
    xor eax, eax
    test rsi, rsi
    jz .Ldone
    mov r12, rdx
    lea rbp, [rdi + rsi*8]
.Lloop:
    mov rbx, qword ptr [rdi]
    sub rsp, 8
    call r12
    add rsp, 8
    add rax, rbx
    add rdi, 8
    cmp rdi, rbp
    jne .Lloop
.Ldone:
    add rsp, 8
    pop r12
    pop rbp
    pop rbx
    ret
