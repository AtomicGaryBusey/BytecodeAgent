map_sum:
    push rbx
    push rbp
    push r12
    push r13
    mov rax, 0
    test rsi, rsi
    jz .Ldone
    lea rbp, [rdi + rsi*8]
    xor ebx, ebx
.Lloop:
    mov rbx, qword ptr [rdi + rbx*8]
    call r12
    add rax, rdx
    inc rbx
    cmp rbx, rsi
    jl .Lloop
.Ldone:
    pop r13
    pop rbp
    pop rbx
    ret
