export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%idx =l copy 0
@loop_byte
	%more_bytes =w csltl %idx, %n
	jnz %more_bytes @process_byte @finalize
@process_byte
	%byte_val =w loadub %p
	%crc =w xor %crc, %byte_val
	%i =w copy 0
@loop_bit
	%bit_test =w and %crc, 1
	%more_bits =w csltl %i, 8
	jnz %more_bits @process_bit @advance_byte
@process_bit
	%condition =w ceqw %bit_test, 1
	jnz %condition @reflect @shift_only
@reflect
	%shifted =w shr %crc, 1
	%xor_val =w copy 3728269056
	%crc =w xor %shifted, %xor_val
	jmp @next_bit
@shift_only
	%crc =w shr %crc, 1
@next_bit
	%i =w add %i, 1
	jmp @loop_bit
@advance_byte
	%p =l add %p, 1
	%idx =l add %idx, 1
	jmp @loop_byte
@finalize
	%final_val =w xor %crc, 4294967295
	%result =l extuw %final_val
	ret %result
}
