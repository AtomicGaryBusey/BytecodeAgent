export function l $sum_to(l %n) {
@start
	%cond =w csltl %n, 1
	jnz %cond, @zero, @loop_init
@zero
	%result =l copy 0
	jmp @done
@loop_init
	%result =l copy 0
	%i =l copy 1
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%result =l add %result, %i
	%i =l add %i, 1
	jmp @loop
@done
	ret %result
}
