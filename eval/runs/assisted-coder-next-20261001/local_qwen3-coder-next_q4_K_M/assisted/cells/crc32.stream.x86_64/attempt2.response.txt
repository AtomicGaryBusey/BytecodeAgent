crc32:
    push rbx
    push r12
    sub rsp, 8
    xor eax, eax
    mov ebx, 0xFFFFFFFF
    test rsi, rsi
    je .Ldone
    xor rcx, rcx
.Lloop:
    movzx edx, byte ptr [rdi + rcx]
    xor bl, dl
    mov r12d, 8
.Lbitloop:
    shr ebx, 1
    jnc .Lnoxor
    xor ebx, 0xEDB88320
.Lnoxor:
    dec r12d
    jnz .Lbitloop
    inc rcx
    cmp rcx, rsi
    jl .Lloop
.Ldone:
    xor eax, ebx
    add rsp, 8
    pop r12
    pop rbx
    ret
