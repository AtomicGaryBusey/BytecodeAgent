fib:
    // Handle n == 0 and n == 1
    cmp x0, #1
    b.le .Lbase_case
    
    // Initialize: a = fib(0) = 0, b = fib(1) = 1
    mov x19, #0          // a = 0
    mov x20, #1          // b = 1
    
    // Loop from i = 2 to n
    mov x21, #2          // i = 2
.Lloop:
    add x22, x19, x20    // c = a + b
    mov x19, x20         // a = b
    mov x20, x22         // b = c
    add x21, x21, #1     // i++
    cmp x21, x0
    b.le .Lloop
    
    // Return b (in x20)
    mov x0, x20
    ret

.Lbase_case:
    // If n == 0, return 0; if n == 1, return 1
    cset x0, eq          // x0 = (n == 1) ? 1 : 0
    ret
