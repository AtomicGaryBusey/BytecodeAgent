fib:
    // Handle n == 0 and n == 1 efficiently using branching
    cmp x0, #1
    b.eq .Lreturn_one
    cbz x0, .Lreturn_zero

    // Initialize: a = fib(0) = 0 (x9), b = fib(1) = 1 (x10)
    mov x9, #0          // a = 0
    mov x10, #1         // b = 1

    // Loop counter i = 2 (scratch register x11)
    mov x11, #2

.Lloop:
    add x12, x9, x10     // c = a + b
    mov x9, x10          // a = b
    mov x10, x12         // b = c
    add x11, x11, #1     // i++
    cmp x11, x0
    b.le .Lloop

    // Return b (in x10)
    mov x0, x10
    ret

.Lreturn_one:
    mov x0, #1
    ret

.Lreturn_zero:
    mov x0, #0
    ret
