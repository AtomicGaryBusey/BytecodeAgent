export function l $fib_rec(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @base0, @check1
@base0
	ret 0
@check1
	%is_one =w ceql %n, 1
	jnz %is_one, @base1, @recursive
@base1
	ret 1
@recursive
	%n_minus_1 =l sub %n, 1
	%r1 =l call $fib_rec(l %n_minus_1)
	%n_minus_2 =l sub %n, 2
	%r2 =l call $fib_rec(l %n_minus_2)
	%result =l add %r1, %r2
	ret %result
}
