export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%val =l loadl %p
	%is_greater =w csltl %max, %val
	jnz %is_greater, @update_max, @continue
@update_max
	%max =l copy %val
@continue
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
