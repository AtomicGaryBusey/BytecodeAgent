export function l $gcd(l %a, l %b) {
@start
	jnz %a, @nonzero_a, @check_b
@nonzero_a
	jnz %b, @loop, @done_a
@check_b
	jnz %b, @done_b, @both_zero
@loop
	%rem =l rem %a, %b
	jnz %rem, @next, @done_b
@next
	%tmp =l copy %b
	%b =l copy %rem
	%a =l copy %tmp
	jmp @loop
@done_a
	ret %a
@done_b
	ret %b
@both_zero
	ret 0
}
