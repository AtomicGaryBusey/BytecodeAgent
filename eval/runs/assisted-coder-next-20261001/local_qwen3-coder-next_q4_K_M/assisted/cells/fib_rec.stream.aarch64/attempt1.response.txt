fib_rec:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    str x19, [sp, #8]
    str x20, [sp, #16]
    str x21, [sp, #24]

    cmp x0, #1
    b.le .Lbase

    // n >= 2: compute fib(n-1) + fib(n-2)
    sub x19, x0, #1        // x19 = n-1
    mov x0, x19            // arg for first recursive call
    bl fib_rec             // fib(n-1), result in x0

    mov x20, x0            // save fib(n-1) in x20
    sub x0, x0, #1         // x0 = (n-1) - 1 = n-2? No, need to recompute n-2 properly
    mov x0, x19            // restore n-1 to x0 for second call准备 n-2
    sub x0, x0, #1         // x0 = n-2

    bl fib_rec             // fib(n-2), result in x0

    add x0, x0, x20        // fib(n) = fib(n-1) + fib(n-2)

    b .Lexit

.Lbase:
    cmp x0, #0
    cset x0, ne            // if n == 0 return 0, else (n==1) return 1

.Lexit:
    ldr x19, [sp, #8]
    ldr x20, [sp, #16]
    ldr x21, [sp, #24]
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret
