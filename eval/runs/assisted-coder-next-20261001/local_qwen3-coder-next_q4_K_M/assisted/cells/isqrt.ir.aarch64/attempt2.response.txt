export function l $isqrt(l %n) {
@start
	%low =l copy 0
	%high =l copy 4294967295
	%result =l copy 0

@loop
	%cmp =w csltl %low, %high
	jnz %cmp, @body, @done

@body
	%mid =l add %low, %high
	%mid =l udiv %mid, 2

	%sq =l mul %mid, %mid
	%leq =w cugel %sq, %n
	jnz %leq, @ok_or_eq, @too_big

@too_big
	%high =l sub %mid, 1
	jmp @loop

@ok_or_eq
	%result =l copy %mid
	%low =l add %mid, 1
	jmp @loop

@done
	ret %result
}
