export function l $add_overflow(l %a, l %b) {
@start
	%sum =l add %a, %b
	storel %sum, %out

	%a_pos =w csltl 0, %a
	%b_pos =w csltl 0, %b
	%sum_neg =w csltl %sum, 0

	%overflow1 =w and %a_pos, %b_pos
	%overflow2 =w and %overflow1, %sum_neg
	%a_neg =w csltl %a, 0
	%b_neg =w csltl %b, 0
	%sum_pos =w csltl 0, %sum

	%overflow3 =w and %a_neg, %b_neg
	%overflow4 =w and %overflow3, %sum_pos
	%overflow =w or %overflow2, %overflow4
	%ret =l extsw %overflow
	ret %ret
}
