export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out

	%zero =l copy 0
	%msb_a =w slti l %a, %zero
	%msb_b =w slti l %b, %zero
	%msb_sum =w slti l %sum, %zero

	%pos_overflow =w and %msb_a, %msb_b
	%pos_overflow =w and %pos_overflow, %msb_sum

	%neg_overflow =w andn %msb_a, %msb_b
	%neg_overflow =w andn %neg_overflow, %msb_sum
	%neg_overflow =w and %neg_overflow, 1

	%overflow =w or %pos_overflow, %neg_overflow
	%ret =l extsw %overflow
	ret %ret
}
