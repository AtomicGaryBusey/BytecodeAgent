export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out

	%zero =l copy 0
	%msb_a =w csltl %zero, %a
	%msb_b =w csltl %zero, %b
	%msb_sum =w csltl %zero, %sum

	%pos_overflow =w andn %msb_a, %msb_b
	%neg_overflow =w and %msb_a, %msb_b

	jnz %pos_overflow, @check_pos, @next1
@check_pos
	%pos_check =w csltl %zero, %sum
	storel %pos_check, %out
	jmp @done
@next1
	jnz %neg_overflow, @check_neg, @ret0
@check_neg
	%neg_check =w csltl %zero, %sum
	%neg_check =w xor %neg_check, 1
	jmp @store_ret
@ret0
	%res =l copy 0
	ret %res
@store_ret
	storel %neg_check, %out
	jmp @done
@done
	%final_val =l loadl %out
	ret %final_val
}
