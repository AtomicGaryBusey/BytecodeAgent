export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out

	%zero =l copy 0
	%msb_a =w csltl %zero, %a
	%msb_b =w csltl %zero, %b
	%msb_sum =w csltl %zero, %sum

	%pos_overflow =w andn %msb_a, %msb_b
	%pos_overflow =w andn %pos_overflow, %msb_sum

	%neg_overflow =w and %msb_a, %msb_b
	%neg_overflow =w andn %neg_overflow, %msb_sum

	%overflow =w or %pos_overflow, %neg_overflow
	%ret =l extsw %overflow
	ret %ret
}
