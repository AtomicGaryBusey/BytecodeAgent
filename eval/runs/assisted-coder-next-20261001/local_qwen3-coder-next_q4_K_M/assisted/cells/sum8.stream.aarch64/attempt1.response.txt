sum8:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Compute terms with multipliers: a - 2*b + 3*c - 4*d + 5*e - 6*f + 7*g - 8*h
    // Load arguments into registers for computation
    mov x9, x0           // x9 = a
    mov x10, x1          // x10 = b
    mov x11, x2          // x11 = c
    mov x12, x3          // x12 = d
    mov x13, x4          // x13 = e
    mov x14, x5          // x14 = f
    mov x15, x6          // x15 = g
    mov x19, x7          // x19 = h

    // Compute -2*b
    mov x20, #2
    mul x20, x10, x20    // x20 = 2*b
    neg x20, x20         // x20 = -2*b

    // Compute 3*c
    add x21, x11, x11    // x21 = 2*c
    add x21, x21, x11    // x21 = 3*c

    // Compute -4*d
    mov x22, #4
    mul x22, x12, x22    // x22 = 4*d
    neg x22, x22         // x22 = -4*d

    // Compute 5*e
    add x23, x13, x13    // x23 = 2*e
    add x23, x23, x23    // x23 = 4*e
    add x23, x23, x13    // x23 = 5*e

    // Compute -6*f
    mov x24, #6
    mul x24, x14, x24    // x24 = 6*f
    neg x24, x24         // x24 = -6*f

    // Compute 7*g
    add x25, x15, x15    // x25 = 2*g
    add x25, x25, x25    // x25 = 4*g
    add x25, x25, x15    // x25 = 5*g
    add x25, x25, x25    // x25 = 10*g (revised)
    sub x25, x25, x15    // x25 = 9*g (still wrong, let's fix: use 7 = 8-1)
    mov x26, #8
    mul x26, x15, x26    // x26 = 8*g
    sub x25, x26, x15    // x25 = 7*g

    // Compute -8*h
    mov x26, #8
    mul x26, x19, x26    // x26 = 8*h
    neg x26, x26         // x26 = -8*h

    // Sum all terms: x9 + x20 + x21 + x22 + x23 + x24 + x25 + x26
    add x0, x9, x20      // a + (-2*b)
    add x0, x0, x21      // + 3*c
    add x0, x0, x22      // + (-4*d)
    add x0, x0, x23      // + 5*e
    add x0, x0, x24      // + (-6*f)
    add x0, x0, x25      // + 7*g
    add x0, x0, x26      // + (-8*h)

    ldp x29, x30, [sp], #16
    ret
