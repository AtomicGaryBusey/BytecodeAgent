sum8:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // Compute each term using only caller-saved registers x9-x15
    mov x9, x0           // a

    // -2*b
    add x10, x1, x1      // 2*b
    neg x10, x10         // -2*b

    // 3*c
    add x11, x2, x2      // 2*c
    add x11, x11, x2     // 3*c

    // -4*d
    add x12, x3, x3      // 2*d
    add x12, x12, x12    // 4*d
    neg x12, x12         // -4*d

    // 5*e
    add x13, x4, x4      // 2*e
    add x13, x13, x13    // 4*e
    add x13, x13, x4     // 5*e

    // -6*f
    add x14, x5, x5      // 2*f
    add x14, x14, x14    // 4*f
    add x14, x14, x5     // 6*f
    neg x14, x14         // -6*f

    // 7*g: 8*g - g
    add x15, x6, x6      // 2*g
    add x15, x15, x15    // 4*g
    add x15, x15, x15    // 8*g
    sub x15, x15, x6     // 7*g

    // -8*h: 8*h then negate
    add x30, x7, x7      // 2*h
    add x30, x30, x30    // 4*h
    add x30, x30, x30    // 8*h
    neg x30, x30         // -8*h

    // Sum all terms into x9
    add x9, x9, x10      // a - 2*b
    add x9, x9, x11      // + 3*c
    add x9, x9, x12      // + (-4*d)
    add x9, x9, x13      // + 5*e
    add x9, x9, x14      // + (-6*f)
    add x9, x9, x15      // + 7*g
    add x0, x9, x30      // + (-8*h)

    ldp x29, x30, [sp], #16
    ret
