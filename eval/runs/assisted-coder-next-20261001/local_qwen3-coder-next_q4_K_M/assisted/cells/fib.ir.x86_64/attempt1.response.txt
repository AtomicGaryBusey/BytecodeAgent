export function l $fib(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @return_0, @check_one
@return_0
	%r =l copy 0
	ret %r
@check_one
	%is_one =w ceql %n, 1
	jnz %is_one, @return_1, @loop_init
@return_1
	%r =l copy 1
	ret %r
@loop_init
	%prev =l copy 0
	%curr =l copy 1
	%i =l copy 2
@loop_body
	%cond =w csltl %i, %n
	jnz %cond, @compute_next, @done
@compute_next
	%next =l add %prev, %curr
	%prev =l copy %curr
	%curr =l copy %next
	%i =l add %i, 1
	jmp @loop_body
@done
	ret %curr
}
