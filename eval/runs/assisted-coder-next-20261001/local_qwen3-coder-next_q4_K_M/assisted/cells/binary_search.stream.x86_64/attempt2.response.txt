binary_search:
    push rbx
    xor eax, eax
    test esi, esi
    jz .Lreturn_minus1
    mov rbx, rdi
    mov ecx, esi
.Lloop:
    cmp eax, ecx
    jge .Lnot_found
    lea edx, [eax + ecx]
    sar edx, 1
    cdqe
    mov rdx, [rbx + rax*8]
    cmp rdx, rsi
    je .Lfound
    jl .Lright_half
    mov ecx, edx
    dec ecx
    jmp .Lloop
.Lright_half:
    lea eax, [rdx + 1]
    jmp .Lloop
.Lfound:
    mov eax, edx
.Lnot_found:
    pop rbx
    ret
.Lreturn_minus1:
    mov eax, -1
    pop rbx
    ret
