binary_search:
    push rbx
    xor eax, eax
    test rsi, rsi
    jz .Lnot_found
    mov rbx, rdi
    mov rcx, rsi
.Lloop:
    cmp rax, rcx
    jge .Lnot_found
    lea rdx, [rax + rcx]
    shr rdx, 1
    mov rdx, [rbx + rdx*8]
    cmp rdx, r9
    je .Lfound
    jl .Lright_half
    mov rcx, rdx
    dec rcx
    jmp .Lloop
.Lright_half:
    lea rax, [rdx + 1]
    jmp .Lloop
.Lfound:
    mov eax, edx
.Lnot_found:
    pop rbx
    ret
