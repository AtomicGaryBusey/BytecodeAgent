binary_search:
    push rbp
    mov rbp, rsp
    sub rsp, 32
    xor eax, eax
    cmp esi, 0
    je .Lreturn_minus1
    mov rbx, rdi
    mov rsi, rax
    lea rdx, [rbp - 8]
    mov qword ptr [rdx], rsi
    mov rax, qword ptr [rbp + 16]
    mov rcx, qword ptr [rbp + 24]
    xor r9d, r9d
    mov r8d, esi
.Lloop:
    cmp r9d, r8d
    jge .Lnot_found
    mov eax, r9d
    add eax, r8d
    sar eax, 1
    cdqe
    movsxd rax, eax
    mov rdx, qword ptr [rbx + rax*8]
    cmp rdx, rcx
    je .Lfound
    jl .Lright_half
    mov r8d, eax
    sub r8d, 1
    jmp .Lloop
.Lright_half:
    mov r9d, eax
    add r9d, 1
    jmp .Lloop
.Lfound:
    mov eax, eax
    cdqe
    mov rsp, rbp
    pop rbp
    ret
.Lnot_found:
    mov eax, -1
    cdqe
.Lreturn_minus1:
    mov rsp, rbp
    pop rbp
    ret
