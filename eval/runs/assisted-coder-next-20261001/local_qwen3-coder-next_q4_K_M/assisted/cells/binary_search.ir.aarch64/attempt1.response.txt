export function l $binary_search(l %a, l %n, l %key) {
@start
	%low =l copy 0
	%high =l copy %n
@loop
	%empty =w ceql %low, %high
	jnz %empty, @not_found, @body
@body
	%mid =l add %low, %high
	%mid =l sar %mid, 1
	%off =l mul %mid, 8
	%p =l add %a, %off
	%val =l loadl %p
	%eq =w ceql %val, %key
	jnz %eq, @found, @check_mid
@check_mid
	%less =w csltl %val, %key
	jnz %less, @right_half, @left_half
@right_half
	%i =l add %mid, 1
	%low =l copy %i
	jmp @loop
@left_half
	%high =l copy %mid
	jmp @loop
@found
	ret %mid
@not_found
	%n1 =l copy -1
	ret %n1
}
