reverse:
    cmp rsi, 1
    jle .Ldone
    xor eax, eax
    mov rbx, rdi
    lea rcx, [rdi + rsi*8 - 8]
.Lloop:
    mov rdx, qword ptr [rbx]
    mov rax, qword ptr [rcx]
    mov qword ptr [rbx], rax
    mov qword ptr [rcx], rdx
    add rbx, 8
    sub rcx, 8
    cmp rbx, rcx
    jl .Lloop
.Ldone:
    xor eax, eax
    ret
