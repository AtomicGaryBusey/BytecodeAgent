reverse:
    cmp rsi, 1
    jle .Ldone
    xor eax, eax
    mov rbx, rdi
    lea r8, [rdi + rsi*8 - 8]
.Lloop:
    mov rdx, qword ptr [rbx]
    mov rcx, qword ptr [r8]
    mov qword ptr [rbx], rcx
    mov qword ptr [r8], rdx
    add rbx, 8
    sub r8, 8
    cmp rbx, r8
    jl .Lloop
.Ldone:
    xor eax, eax
    ret
