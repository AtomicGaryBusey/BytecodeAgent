reverse:
    cmp rsi, 1
    jle .Ldone
    xor eax, eax
    lea rbx, [rdi + rsi*8 - 8]
.Lloop:
    mov rcx, qword ptr [rdi]
    mov rdx, qword ptr [rbx]
    mov qword ptr [rdi], rdx
    mov qword ptr [rbx], rcx
    add rdi, 8
    sub rbx, 8
    cmp rdi, rbx
    jl .Lloop
.Ldone:
    xor eax, eax
    ret
