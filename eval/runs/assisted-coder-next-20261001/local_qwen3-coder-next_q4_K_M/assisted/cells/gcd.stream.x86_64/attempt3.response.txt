gcd:
    test rdi, rdi
    jz .Lreturn_b
    test rsi, rsi
    jz .Lreturn_a
.Lloop:
    xor rdx, rdx
    mov rax, rdi
    idiv rsi
    test rdx, rdx
    jz .Ldone
    mov rdi, rsi
    mov rsi, rdx
    jmp .Lloop
.Lreturn_a:
    mov rax, rdi
    ret
.Lreturn_b:
    mov rax, rsi
    ret
.Ldone:
    mov rax, rsi
    ret
