gcd:
    # Handle special cases where either operand is zero
    test rdi, rdi
    jnz .Lcheck_b
    mov rax, rsi
    ret

.Lcheck_b:
    test rsi, rsi
    jnz .Lloop
    mov rax, rdi
    ret

.Lloop:
    # Compute a % b using idiv
    cqo
    idiv rsi
    # Now rdx contains a % b
    test rdx, rdx
    jz .Ldone
    # Update: a = b, b = remainder (a % b)
    mov rdi, rsi
    mov rsi, rdx
    jmp .Lloop

.Ldone:
    mov rax, rsi
    ret
