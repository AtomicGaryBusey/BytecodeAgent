export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32

	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh

	%lh_low =l and %lh, 4294967295
	%shifted_lh_low =l shl %lh_low, 32
	%mid1 =l add %ll, %shifted_lh_low
	%carry1 =l shr %mid1, 32

	%hl_carry =l add %hl, %carry1
	%mid2 =l add %mid1, lshl(%hl_carry, 32)
	%carry2 =l shr %mid2, 32

	%high_result =l add %hh, %carry2

	ret %high_result
}
