export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32

	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh

	%mid1 =l add %ll, lshl(%lh, 32)
	%carry1 =l shr %mid1, 32
	%mid2 =l add %hl, carry1
	%carry2 =l shr %mid2, 32
	%low_part =l and %mid2, 4294967295

	%high_result =l add %hh, %carry2

	ret %high_result
}
