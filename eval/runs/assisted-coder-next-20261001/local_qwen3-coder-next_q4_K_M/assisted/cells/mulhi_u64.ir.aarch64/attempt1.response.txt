export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32

	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh

	%mid1 =l add %ll, %lh
	%mid2 =l shr %mid1, 32
	%carry =l and %mid1, 4294967295
	%mid3 =l add %hl, %mid2
	%carry2 =l shr %mid3, 32
	%low_part =l add %carry, %carry2

	%high_result =l add %hh, %low_part

	ret %high_result
}
