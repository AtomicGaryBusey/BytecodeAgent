crc32:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Initialize CRC to 0xFFFFFFFF
    movz x10, #0xFFFF
    movk x10, #0xFFFF, lsl #16
    // Handle n == 0 case
    cbz x1, .Lfinish_crc
    // Load constant for polynomial (0xEDB88320)
    movz x9, #0x8320
    movk x9, #0xEDB8, lsl #16
    // Loop over bytes
.Lloop_bytes:
    cbz x1, .Lfinish_crc
    ldrb w13, [x0]
    eor x10, x10, x13, lsl #0
    add x0, x0, #1
    sub x1, x1, #1
    mov x14, #8
.Lloop_bits:
    cbz x14, .Lnext_byte
    // Check LSB of crc
    and x15, x10, #1
    cbz x15, .Lshift_only
    // Apply polynomial: (crc >> 1) ^ 0xEDB88320
    lsr x10, x10, #1
    eor x10, x10, x9
    b .Lbit_done
.Lshift_only:
    lsr x10, x10, #1
.Lbit_done:
    sub x14, x14, #1
    b .Lloop_bits
.Lnext_byte:
    b .Lloop_bytes
.Lfinish_crc:
    eor x0, x10, #0xFFFFFFFF
    ldp x29, x30, [sp], #16
    ret
