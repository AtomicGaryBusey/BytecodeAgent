crc32:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Initialize CRC to 0xFFFFFFFF
    mov x0, #0xFFFFFFFF
    lsl x0, x0, #32
    orr x0, x0, #0xFFFFFFFF
    // Handle n == 0 case
    cbz x1, .Ldone
    // Load constant for polynomial (0xEDB88320)
    mov x9, #0x8320
    movk x9, #0xEDB8, lsl #16
    // Loop over bytes
    mov x10, x0          // crc in x10
    mov x11, x1          // n in x11
    xor x12, x12, x12    // clear byte index (unused)
.L
