export function l $bubble_sort(l %a, l %n) {
@start
	%outer =l copy 0
	%nval =l copy %n
	%one =l copy 1
	%nminus1 =l sub %nval, %one
@outer_loop
	%outer_cmp =w csltl %outer, %nminus1
	jnz %outer_cmp, @inner_start, @done
@inner_start
	%i =l copy 0
	%nminus2 =l sub %nval, %one
@inner_loop
	%inner_cmp =w csltl %i, %nminus2
	jnz %inner_cmp, @body, @inner_done
@body
	%off_i =l mul %i, 8
	%addr_i =l add %a, %off_i
	%val_i =l loadl %addr_i
	%next_i =l add %i, 1
	%off_next =l mul %next_i, 8
	%addr_next =l add %a, %off_next
	%val_next =l loadl %addr_next
	%cmp =w csltl %val_next, %val_i
	jnz %cmp, @swap, @skip_swap
@swap
	storel %val_next, %addr_i
	storel %val_i, %addr_next
	jmp @skip_swap
@skip_swap
	%i =l add %i, 1
	jmp @inner_loop
@inner_done
	%outer =l add %outer, 1
	jmp @outer_loop
@done
	%ret =l copy 0
	ret %ret
}
