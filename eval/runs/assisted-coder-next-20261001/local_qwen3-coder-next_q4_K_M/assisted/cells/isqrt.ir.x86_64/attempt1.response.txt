export function l $isqrt(l %n) {
@start
	%r =l copy 0
	%low =l copy 0
	%high =l copy 4294967295
	%mid =l copy 0
	%sq =l copy 0
@loop
	%cmp =w csltl %low, %high
	jnz %cmp, @body, @done
@body
	%mid =l add %low, 1
	%mid =l shr %mid, 1
	%sq =l mul %mid, %mid
	%le =w cugel %sq, %n
	jnz %le, @yes, @no
@yes
	%r =l copy %mid
	%low =l add %mid, 1
	jmp @loop
@no
	%high =l sub %mid, 1
	jmp @loop
@done
	ret %r
}
