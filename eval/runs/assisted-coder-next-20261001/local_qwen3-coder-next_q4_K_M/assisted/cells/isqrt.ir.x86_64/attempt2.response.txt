export function l $isqrt(l %n) {
@start
	%low =l copy 1
	%high =l copy 4294967295
	%r =l copy 0
	%max =l copy 4294967295
	jmp @check_zero
@loop
	%mid =l add %low, %high
	%mid =l shr %mid, 1
	%mid_sq_low =l mul %mid, %mid
	%mid_sq_high =l mul %mid, %mid
	%sq =l copy %mid_sq_low
	%cmp =w cugel %sq, %n
	jnz %cmp, @square_ok, @adjust_high
@square_ok
	%le =w culeg %sq, %n
	jnz %le, @update_r, @adjust_high
@update_r
	%r =l copy %mid
	%next_low =l add %mid, 1
	%cmp2 =w cugel %next_low, %high
	jnz %cmp2, @done, @loop
@adjust_high
	%new_high =l sub %mid, 1
	%cmp3 =w csltl %low, %new_high
	jnz %cmp3, @loop, @done
@check_zero
	%is_zero =w ceql %n, 0
	jnz %is_zero, @return_zero, @init_loop
@init_loop
	jmp @loop
@done
	ret %r
@return_zero
	%zero =l copy 0
	ret %zero
}
