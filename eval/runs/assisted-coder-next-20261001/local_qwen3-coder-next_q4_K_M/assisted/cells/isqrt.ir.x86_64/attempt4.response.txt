export function l $isqrt(l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @return_zero, @init
@init
	%low =l copy 0
	%high =l copy 4294967295
	%r =l copy 0
	jmp @loop
@loop
	%mid =l add %low, %high
	%mid =l shr %mid, 1
	%mid_plus_1 =l add %mid, 1
	%mid_sq_high =l mul %mid_plus_1, %mid
	%cmp =w cugel %mid_sq_high, %n
	jnz %cmp, @square_ok, @adjust_high
@square_ok
	%le =w culel %mid, %r
	jnz %le, @update_r, @adjust_high
@update_r
	%r =l copy %mid
	%next_low =l add %mid, 1
	%cmp2 =w cugel %next_low, %high
	jnz %cmp2, @done, @loop
@adjust_high
	%new_high =l sub %mid, 1
	%cmp3 =w csltl %low, %new_high
	jnz %cmp3, @loop, @done
@done
	ret %r
@return_zero
	%zero =l copy 0
	ret %zero
}
