collatz_max:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Load n into x1 for iteration
    mov x1, x0
    // Initialize max_steps = -1 (to track maximum)
    mov x2, #0
    neg x2, x2

    // Outer loop over k from 1 to n
    mov x3, #1        // current k

.Louter_loop:
    cmp x3, x1
    b.gt .Ldone_outer
    // Compute Collatz steps for current k (x3)
    mov x4, x3        // current value
    mov x5, #0        // step counter

.Linner_loop:
    cmp x4, #1
    beq .Ldone_inner
    // Check if even: test bit 0
    tbz x4, #0, .Leven_case
    // Odd: 3*x + 1
    add x7, x4, x4, lsl #1   // x4 * 3
    add x4, x7, #1           // +1
    b .Lstep_done

.Leven_case:
    lsr x4, x4, #1          // x4 / 2 (unsigned shift)

.Lstep_done:
    add x5, x5, #1        // increment step counter
    b .Linner_loop

.Ldone_inner:
    // Update max_steps if current steps > max_steps
    cmp x5, x2
    csel x2, x5, x2, gt  // max_steps = max(max_steps, steps)
    add x3, x3, #1       // next k
    b .Louter_loop

.Ldone_outer:
    mov x0, x2           // return max_steps
    ldp x29, x30, [sp], #16
    ret
