collatz_max:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    // Load n into x1 for iteration
    mov x1, x0
    // Initialize max_steps = -1 (to track maximum)
    mov x2, #0
    neg x2, x2

    // Outer loop over k from 1 to n
    mov x3, #1        // current k

.Louter_loop:
    cmp x3, x1
    b.gt .Ldone_outer
    // Compute Collatz steps for current k (x3)
    mov x4, x3        // current value
    mov x5, #0        // step counter

.Linner_loop:
    cmp x4, #1
    beq .Ldone_inner
    // Check if even
    and x6, x4, #1
    cbz x6, .Leven_case
    // Odd: 3*x + 1
    add x7, x4, x4, lsl #1   // x4 * 3
    add x7, x7, #1           // +1
    mov x4, x7
    b .Lstep_done

.Leven_case:
    lsrs x4, x4, #1          // x4 / 2 (unsigned shift)
