popcount:
    xor eax, eax        # clear result counter
    test rdi, rdi       # check if x is zero
    jz .Ldone           # early exit if zero
.Lloop:
    lea rcx, [rdi - 1]  # x-1
    and rcx, rdi        # (x-1) & x
    mov rdi, rcx        # x = x & (x-1)
    inc eax             # increment bit count
    test rdi, rdi       # check if remaining bits are zero
    jnz .Lloop          # continue until all bits cleared
.Ldone:
    ret                 # return with popcount in rax
