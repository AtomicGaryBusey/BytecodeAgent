.globl _start

.text
_start:
    lea r10, [rip + hex_digits]
    lea r12, [rip + in_buf]
    xor r13d, r13d

.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 131072
    sub edx, r13d
    jle .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r13, rax
    jmp .Lread_loop

.Lread_done:
    lea rdi, [rip + out_buf]
    xor ebx, ebx

.Lrow_loop:
    cmp rbx, r13
    jge .Lrows_done

    lea rax, [rip + out_buf + 65536]
    cmp rdi, rax
    jb .Lno_flush
    call flush_out
.Lno_flush:

    mov r14, r13
    sub r14, rbx
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:

    mov eax, ebx
    mov ecx, 8
.Lhex_off:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz .Lhex_off

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2

    xor r15d, r15d
.Lpos_loop:
    cmp r15, r14
    jae .Lpos_space

    lea rax, [rbx + r15]
    movzx eax, byte ptr [r12 + rax]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp .Lpos_check_gap

.Lpos_space:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

.Lpos_check_gap:
    cmp r15d, 7
    jne .Lpos_next
    mov byte ptr [rdi], 0x20
    inc rdi

.Lpos_next:
    inc r15d
    cmp r15d, 16
    jne .Lpos_loop

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2

    xor r15d, r15d
.Lchar_loop:
    cmp r15, r14
    jae .Lchar_done

    lea rax, [rbx + r15]
    movzx eax, byte ptr [r12 + rax]
    mov edx, 0x2e
    lea ecx, [rax - 0x20]
    cmp ecx, 0x5e
    cmova eax, edx
    mov byte ptr [rdi], al
    inc rdi
    inc r15d
    jmp .Lchar_loop

.Lchar_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2

    add rbx, 16
    jmp .Lrow_loop

.Lrows_done:
    mov eax, r13d
    mov ecx, 8
.Lhex_total:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz .Lhex_total

    mov byte ptr [rdi], 0x0a
    inc rdi

    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

flush_out:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
.Lwrite_loop:
    test rdx, rdx
    jle .Lflush_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lflush_done
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite_loop
.Lflush_done:
    lea rdi, [rip + out_buf]
    ret

.section .rodata
.balign 16
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
in_buf:
    .zero 131072
out_buf:
    .zero 131072
