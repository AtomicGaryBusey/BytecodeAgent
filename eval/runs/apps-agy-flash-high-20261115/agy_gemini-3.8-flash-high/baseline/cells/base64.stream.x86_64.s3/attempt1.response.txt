.globl _start

.text
_start:
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, rbx
    mov edx, 131072
    sub rdx, rbx
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_zero

    lea rsi, [rip + in_buf]
    lea rbp, [rsi + rbx]
    lea rdi, [rip + out_buf]
    lea r11, [rip + b64_table]
    xor r8d, r8d

main_loop:
    mov rax, rbp
    sub rax, rsi
    cmp rax, 3
    jb handle_remainder

    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 0x3F
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne main_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp main_loop

handle_remainder:
    cmp rax, 1
    je rem_one
    cmp rax, 2
    je rem_two
    jmp finish_line

rem_one:
    movzx eax, byte ptr [rsi]
    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    and edx, 3
    shl edx, 4
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], 0x3D
    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne finish_line
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp finish_line

rem_two:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]

    mov edx, eax
    shr edx, 10
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 4
    and edx, 0x3F
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    and edx, 0x0F
    shl edx, 2
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne finish_line
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

finish_line:
    test r8d, r8d
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi

do_write:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_zero

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_zero
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_zero:
    mov eax, 60
    xor edi, edi
    syscall
1:
    jmp 1b

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
in_buf:
    .space 131072
out_buf:
    .space 262144
