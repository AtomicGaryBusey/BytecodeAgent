.globl _start

.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 65536
    syscall

    test rax, rax
    jle done_reading

    add r14, rax
    lea rbx, [rsi + rax]

byte_loop:
    cmp rsi, rbx
    je read_loop
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    je is_newline
    cmp al, 32
    je is_space
    sub al, 9
    cmp al, 4
    jbe is_space

    test r15d, r15d
    jnz byte_loop
    inc r13
    mov r15d, 1
    jmp byte_loop

is_newline:
    inc r12
is_space:
    xor r15d, r15d
    jmp byte_loop

done_reading:
    lea rdi, [rip + out_buf]

    mov rax, r12
    call format_u64

    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call format_u64

    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call format_u64

    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jle exit_program
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

halt_loop:
    jmp halt_loop

format_u64:
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 48
    inc rdi
    ret

1:
    mov r8, rdi
    mov ecx, 10

2:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz 2b

    lea r9, [rdi - 1]

3:
    cmp r8, r9
    jae 4f
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp 3b

4:
    ret

.bss
.balign 16
in_buf:
    .zero 65536
out_buf:
    .zero 128
