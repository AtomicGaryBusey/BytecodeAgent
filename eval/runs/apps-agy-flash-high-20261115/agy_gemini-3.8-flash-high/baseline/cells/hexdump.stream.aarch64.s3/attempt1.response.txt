.globl _start

.text
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov x23, #0

    adrp x24, in_buf
    add x24, x24, :lo12:in_buf

    adrp x25, out_buf
    add x25, x25, :lo12:out_buf

    adrp x26, hex_chars
    add x26, x26, :lo12:hex_chars

.Lrow_loop:
    cbnz x22, .Lrefill_done
    sub x2, x21, x20
    cmp x2, #16
    b.ge .Lrefill_done

.Lrefill_start:
    sub x2, x21, x20
    cbz x20, .Lno_shift
    cbz x2, .Lzero_rem
    mov x0, x24
    add x1, x24, x20
    mov x3, x2
.Lcopy_loop:
    ldrb w4, [x1], #1
    strb w4, [x0], #1
    subs x3, x3, #1
    b.ne .Lcopy_loop
.Lzero_rem:
    mov x21, x2
    mov x20, #0
.Lno_shift:
    mov x0, #0
    add x1, x24, x21
    movz x2, #1, lsl #16
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le .Lrefill_eof
    add x21, x21, x0
    sub x2, x21, x20
    cmp x2, #16
    b.lt .Lrefill_start
    b .Lrefill_done

.Lrefill_eof:
    mov x22, #1

.Lrefill_done:
    sub x2, x21, x20
    cbz x2, .Ldone_input

    mov x10, x2
    cmp x10, #16
    b.lo .Lgot_n
    mov x10, #16
.Lgot_n:
    movz x0, #1, lsl #16
    sub x0, x0, #128
    cmp x23, x0
    b.lo .Lbuf_ok
    bl .Lflush_out
.Lbuf_ok:
    add x27, x25, x23
    add x28, x24, x20

    mov x0, #28
.Loff_loop:
    lsr x1, x19, x0
    and x1, x1, #0xf
    ldrb w2, [x26, x1]
    strb w2, [x27], #1
    subs x0, x0, #4
    b.ge .Loff_loop

    mov w0, #0x20
    strb w0, [x27], #1
    strb w0, [x27], #1

    mov x11, #0
.Lhex_loop:
    cmp x11, x10
    b.ge .Lhex_space

    ldrb w0, [x28, x11]
    lsr w1, w0, #4
    ldrb w2, [x26, x1]
    strb w2, [x27], #1

    and w1, w0, #0xf
    ldrb w2, [x26, x1]
    strb w2, [x27], #1

    mov w0, #0x20
    strb w0, [x27], #1
    b .Lhex_check_extra

.Lhex_space:
    mov w0, #0x20
    strb w0, [x27], #1
    strb w0, [x27], #1
    strb w0, [x27], #1

.Lhex_check_extra:
    cmp x11, #7
    b.ne .Lhex_next
    mov w0, #0x20
    strb w0, [x27], #1

.Lhex_next:
    add x11, x11, #1
    cmp x11, #16
    b.lt .Lhex_loop

    mov w0, #0x20
    strb w0, [x27], #1
    mov w0, #0x7c
    strb w0, [x27], #1

    mov x11, #0
.Lascii_loop:
    cmp x11, x10
    b.ge .Lascii_done
    ldrb w0, [x28, x11]
    sub w1, w0, #0x20
    cmp w1, #0x5e
    b.ls .Lchar_printable
    mov w0, #0x2e
.Lchar_printable:
    strb w0, [x27], #1
    add x11, x11, #1
    b .Lascii_loop
.Lascii_done:

    mov w0, #0x7c
    strb w0, [x27], #1
    mov w0, #0x0a
    strb w0, [x27], #1

    sub x23, x27, x25
    add x20, x20, x10
    add x19, x19, x10
    b .Lrow_loop

.Ldone_input:
    movz x0, #1, lsl #16
    sub x0, x0, #128
    cmp x23, x0
    b.lo .Ldone_buf_ok
    bl .Lflush_out
.Ldone_buf_ok:
    add x27, x25, x23

    mov x0, #28
.Lfinal_off_loop:
    lsr x1, x19, x0
    and x1, x1, #0xf
    ldrb w2, [x26, x1]
    strb w2, [x27], #1
    subs x0, x0, #4
    b.ge .Lfinal_off_loop

    mov w0, #0x0a
    strb w0, [x27], #1

    sub x23, x27, x25
    bl .Lflush_out

    mov x0, #0
    mov x8, #93
    svc #0

.Lflush_out:
    cbz x23, .Lflush_ret
    mov x0, #1
    mov x1, x25
    mov x2, x23
.Lwrite_loop:
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le .Lwrite_done
    subs x2, x2, x0
    cbz x2, .Lwrite_done
    add x1, x1, x0
    mov x0, #1
    b .Lwrite_loop
.Lwrite_done:
    mov x23, #0
.Lflush_ret:
    ret

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.section .bss
    .balign 16
in_buf:
    .zero 65536
out_buf:
    .zero 65536
