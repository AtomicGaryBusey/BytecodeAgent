.equ BUF_SIZE, 131072

.text
.globl _start
_start:
    xor ebp, ebp
    lea r12, [rip + in_buf]

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + rbp]
    mov edx, BUF_SIZE
    sub edx, ebp
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add rbp, rax
    jmp read_loop

read_done:
    test ebp, ebp
    jz exit_clean

    lea r8, [r12 + rbp]
    mov rsi, r12
    lea rdi, [rip + out_buf]

line_loop:
    cmp rsi, r8
    jnb flush_out

    mov r9, rsi

find_nl:
    cmp rsi, r8
    jnb found_nl_or_end
    cmp byte ptr [rsi], 10
    je found_nl_or_end
    inc rsi
    jmp find_nl

found_nl_or_end:
    mov rbx, rsi
    dec rbx

rev_loop:
    cmp rbx, r9
    jl rev_done
    mov al, byte ptr [rbx]
    mov byte ptr [rdi], al
    inc rdi
    dec rbx
    jmp rev_loop

rev_done:
    mov byte ptr [rdi], 10
    inc rdi

    cmp rsi, r8
    je flush_out
    inc rsi
    jmp line_loop

flush_out:
    lea rbx, [rip + out_buf]
    mov r13, rdi
    sub r13, rbx

write_loop:
    test r13, r13
    jz exit_clean
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r13
    syscall
    test rax, rax
    jle exit_clean
    add rbx, rax
    sub r13, rax
    jmp write_loop

exit_clean:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
in_buf:
    .space BUF_SIZE
out_buf:
    .space BUF_SIZE
