.globl _start

.text
_start:
    # Read all input from stdin (fd 0) into input_buffer
    xor ebp, ebp

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbp
    mov edx, 65536
    sub edx, ebp
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add rbp, rax
    jmp read_loop

read_done:
    test rbp, rbp
    jz exit_success

    # Parse lines from input_buffer
    lea rsi, [rip + input_buffer]
    mov rbx, rsi
    lea r12, [rsi + rbp]
    lea r14, [rip + lines]
    xor r13d, r13d

parse_loop:
    cmp rsi, r12
    jae parse_done
    mov al, byte ptr [rsi]
    cmp al, 10
    jne next_char

    mov rax, rsi
    sub rax, rbx
    mov qword ptr [r14], rbx
    mov qword ptr [r14 + 8], rax
    add r14, 16
    inc r13
    lea rbx, [rsi + 1]

next_char:
    inc rsi
    jmp parse_loop

parse_done:
    cmp rbx, r12
    jae lines_ready
    mov rax, r12
    sub rax, rbx
    mov qword ptr [r14], rbx
    mov qword ptr [r14 + 8], rax
    add r14, 16
    inc r13

lines_ready:
    test r13, r13
    jz exit_success

    cmp r13, 1
    jbe print_lines

    xor edi, edi
    mov rsi, r13
    dec rsi
    call quicksort

print_lines:
    lea rbx, [rip + lines]
    lea rdi, [rip + output_buffer]
    mov r12, r13

write_prep_loop:
    test r12, r12
    jz write_prep_done
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    dec r12
    jnz write_prep_loop

write_prep_done:
    lea rbx, [rip + output_buffer]
    sub rdi, rbx
    mov r12, rdi

write_loop:
    test r12, r12
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall
    test rax, rax
    jle exit_success
    add rbx, rax
    sub r12, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]

    mov rax, r9
    cmp rax, r11
    jbe cmp_len_min
    mov rax, r11

cmp_len_min:
    test rax, rax
    jz cmp_len_check

cmp_byte_loop:
    movzx ecx, byte ptr [r8]
    movzx edx, byte ptr [r10]
    cmp ecx, edx
    jb cmp_less
    ja cmp_greater
    inc r8
    inc r10
    dec rax
    jnz cmp_byte_loop

cmp_len_check:
    cmp r9, r11
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret

cmp_less:
    mov eax, -1
    ret

cmp_greater:
    mov eax, 1
    ret

quicksort:
    cmp rdi, rsi
    jge qs_return

    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24

    mov r12, rdi
    mov r13, rsi
    lea rbx, [rip + lines]

    mov rax, r12
    add rax, r13
    sar rax, 1
    shl rax, 4

    mov rcx, qword ptr [rbx + rax]
    mov rdx, qword ptr [rbx + rax + 8]
    mov qword ptr [rbp - 56], rcx
    mov qword ptr [rbp - 48], rdx

    mov r14, r12
    mov r15, r13

qs_partition_loop:
qs_i_loop:
    mov rax, r14
    shl rax, 4
    lea rdi, [rbx + rax]
    lea rsi, [rbp - 56]
    call compare_lines
    test eax, eax
    jns qs_j_loop
    inc r14
    jmp qs_i_loop

qs_j_loop:
    mov rax, r15
    shl rax, 4
    lea rdi, [rbx + rax]
    lea rsi, [rbp - 56]
    call compare_lines
    test eax, eax
    jle qs_check_swap
    dec r15
    jmp qs_j_loop

qs_check_swap:
    cmp r14, r15
    jg qs_partition_done

    mov rax, r14
    shl rax, 4
    mov rdx, r15
    shl rdx, 4

    mov r8, qword ptr [rbx + rax]
    mov r9, qword ptr [rbx + rax + 8]
    mov r10, qword ptr [rbx + rdx]
    mov r11, qword ptr [rbx + rdx + 8]

    mov qword ptr [rbx + rax], r10
    mov qword ptr [rbx + rax + 8], r11
    mov qword ptr [rbx + rdx], r8
    mov qword ptr [rbx + rdx + 8], r9

    inc r14
    dec r15

    cmp r14, r15
    jle qs_partition_loop

qs_partition_done:
    cmp r12, r15
    jge qs_skip_left
    mov rdi, r12
    mov rsi, r15
    call quicksort

qs_skip_left:
    cmp r14, r13
    jge qs_skip_right
    mov rdi, r14
    mov rsi, r13
    call quicksort

qs_skip_right:
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp

qs_return:
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
lines:
    .zero 65536
output_buffer:
    .zero 65536
