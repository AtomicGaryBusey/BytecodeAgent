.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
row_buf:
    .space 16
inbuf:
    .space 4096
outbuf:
    .space 8192

.text
_start:
    lea rbx, [rip + hex_digits]
    lea r10, [rip + row_buf]
    lea r12, [rip + outbuf]
    xor r13, r13
    xor r14, r14
    lea r15, [rip + inbuf]
    mov rbp, r15

main_loop:
    cmp r15, rbp
    jb has_input_bytes

    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jle input_eof
    lea r15, [rip + inbuf]
    lea rbp, [r15 + rax]

has_input_bytes:
    mov al, byte ptr [r15]
    inc r15
    mov byte ptr [r10 + r14], al
    inc r14
    cmp r14, 16
    jne main_loop

    call check_flush
    call format_row
    add r13, 16
    xor r14, r14
    jmp main_loop

input_eof:
    test r14, r14
    jz print_final_offset

    call check_flush
    call format_row
    add r13, r14

print_final_offset:
    call check_flush
    mov edx, r13d
    call format_hex8
    mov byte ptr [r12], 10
    inc r12
    call flush_outbuf

    mov eax, 60
    xor edi, edi
    syscall

check_flush:
    lea rax, [rip + outbuf]
    add rax, 8064
    cmp r12, rax
    jb 1f
    call flush_outbuf
1:
    ret

flush_outbuf:
    lea rsi, [rip + outbuf]
    mov rdx, r12
    sub rdx, rsi
    jz flush_done
flush_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle flush_done
    add rsi, rax
    sub rdx, rax
    jnz flush_loop
flush_done:
    lea r12, [rip + outbuf]
    ret

format_hex8:
    mov ecx, 8
1:
    rol edx, 4
    mov eax, edx
    and eax, 0xf
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r12], al
    inc r12
    dec ecx
    jnz 1b
    ret

format_row:
    mov edx, r13d
    call format_hex8
    mov byte ptr [r12], ' '
    mov byte ptr [r12 + 1], ' '
    add r12, 2

    xor r8d, r8d
slot_loop:
    cmp r8d, r14d
    jae slot_empty

    movzx eax, byte ptr [r10 + r8]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r12], dl
    and eax, 0xf
    mov al, byte ptr [rbx + rax]
    mov byte ptr [r12 + 1], al
    mov byte ptr [r12 + 2], ' '
    add r12, 3
    jmp slot_after

slot_empty:
    mov byte ptr [r12], ' '
    mov byte ptr [r12 + 1], ' '
    mov byte ptr [r12 + 2], ' '
    add r12, 3

slot_after:
    cmp r8d, 7
    jne 1f
    mov byte ptr [r12], ' '
    inc r12
1:
    inc r8d
    cmp r8d, 16
    jne slot_loop

    mov byte ptr [r12], ' '
    mov byte ptr [r12 + 1], '|'
    add r12, 2

    xor r8d, r8d
char_loop:
    cmp r8d, r14d
    jae char_done
    movzx eax, byte ptr [r10 + r8]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 2f
    mov al, '.'
2:
    mov byte ptr [r12], al
    inc r12
    inc r8d
    jmp char_loop

char_done:
    mov byte ptr [r12], '|'
    mov byte ptr [r12 + 1], 10
    add r12, 2
    ret
