.section .rodata
str_ok:
    .ascii "ok\n"
str_err_prefix:
    .ascii "error at "

.section .bss
.balign 16
input_buf:
    .zero 131072
err_buf:
    .zero 32

.text
.globl _start
_start:
    lea r12, [rip + input_buf]
    xor r13, r13
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 131072
    sub edx, r13d
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r14, r14
    call skip_ws
    cmp r14, r13
    jae error_pos

    call parse_value

    call skip_ws
    cmp r14, r13
    jne error_pos

write_ok:
    lea rsi, [rip + str_ok]
    mov edx, 3
write_ok_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_ok_done
    add rsi, rax
    sub rdx, rax
    jnz write_ok_loop
write_ok_done:
    mov eax, 60
    xor edi, edi
    syscall

error_pos:
    lea rsi, [rip + err_buf + 31]
    mov byte ptr [rsi], 10
    mov rax, r14
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b

    sub rsi, 9
    lea rdi, [rip + str_err_prefix]
    mov rax, qword ptr [rdi]
    mov qword ptr [rsi], rax
    mov al, byte ptr [rdi + 8]
    mov byte ptr [rsi + 8], al

    lea rdx, [rip + err_buf + 32]
    sub rdx, rsi
write_err_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_err_done
    add rsi, rax
    sub rdx, rax
    jnz write_err_loop
write_err_done:
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
ws_loop:
    cmp r14, r13
    jae ws_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, ' '
    je ws_next
    cmp al, 10
    je ws_next
    cmp al, 13
    je ws_next
    cmp al, 9
    jne ws_done
ws_next:
    inc r14
    jmp ws_loop
ws_done:
    ret

parse_value:
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    jne 1f
    inc r14
    jmp parse_object
1:
    cmp al, '['
    jne 2f
    inc r14
    jmp parse_array
2:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    jbe parse_number
    jmp error_pos

parse_object:
    call skip_ws
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '}'
    jne object_member_loop
    inc r14
    ret

object_member_loop:
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    jne error_pos
    call parse_string

    call skip_ws
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, ':'
    jne error_pos
    inc r14

    call skip_ws
    cmp r14, r13
    jae error_pos

    call parse_value

    call skip_ws
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '}'
    jne 1f
    inc r14
    ret
1:
    cmp al, ','
    jne error_pos
    inc r14
    call skip_ws
    cmp r14, r13
    jae error_pos
    jmp object_member_loop

parse_array:
    call skip_ws
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, ']'
    jne array_element_loop
    inc r14
    ret

array_element_loop:
    call parse_value
    call skip_ws
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, ']'
    jne 1f
    inc r14
    ret
1:
    cmp al, ','
    jne error_pos
    inc r14
    call skip_ws
    cmp r14, r13
    jae error_pos
    jmp array_element_loop

parse_string:
    inc r14
string_char_loop:
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je string_close
    cmp al, '\\'
    je string_escape
    cmp al, 0x20
    jb error_pos
    inc r14
    jmp string_char_loop

string_close:
    inc r14
    ret

string_escape:
    inc r14
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '"'
    je escape_single
    cmp al, '\\'
    je escape_single
    cmp al, '/'
    je escape_single
    cmp al, 'b'
    je escape_single
    cmp al, 'f'
    je escape_single
    cmp al, 'n'
    je escape_single
    cmp al, 'r'
    je escape_single
    cmp al, 't'
    je escape_single
    cmp al, 'u'
    je escape_u
    jmp error_pos

escape_single:
    inc r14
    jmp string_char_loop

escape_u:
    inc r14
    mov ecx, 4
escape_u_loop:
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    jbe 1f
    mov edx, eax
    sub edx, 'a'
    cmp edx, 5
    jbe 1f
    mov edx, eax
    sub edx, 'A'
    cmp edx, 5
    ja error_pos
1:
    inc r14
    dec ecx
    jnz escape_u_loop
    jmp string_char_loop

parse_true:
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'r'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'u'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'e'
    jne error_pos
    inc r14
    ret

parse_false:
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'a'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'l'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 's'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'e'
    jne error_pos
    inc r14
    ret

parse_null:
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'u'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'l'
    jne error_pos
    inc r14
    cmp r14, r13
    jae error_pos
    cmp byte ptr [r12 + r14], 'l'
    jne error_pos
    inc r14
    ret

parse_number:
    movzx eax, byte ptr [r12 + r14]
    cmp al, '-'
    jne num_int
    inc r14
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]

num_int:
    cmp al, '0'
    jne num_1_to_9
    inc r14
    cmp r14, r13
    jae num_frac_exp
    movzx eax, byte ptr [r12 + r14]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    jbe error_pos
    jmp num_frac_exp

num_1_to_9:
    mov edx, eax
    sub edx, '1'
    cmp edx, 8
    ja error_pos
    inc r14
num_int_digits:
    cmp r14, r13
    jae num_frac_exp
    movzx eax, byte ptr [r12 + r14]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    ja num_frac_exp
    inc r14
    jmp num_int_digits

num_frac_exp:
    cmp r14, r13
    jae num_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, '.'
    jne num_exp

    inc r14
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    ja error_pos
    inc r14
num_frac_digits:
    cmp r14, r13
    jae num_done
    movzx eax, byte ptr [r12 + r14]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    ja num_exp
    inc r14
    jmp num_frac_digits

num_exp:
    cmp r14, r13
    jae num_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 'e'
    je 1f
    cmp al, 'E'
    jne num_done
1:
    inc r14
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
    cmp al, '+'
    je 2f
    cmp al, '-'
    jne 3f
2:
    inc r14
    cmp r14, r13
    jae error_pos
    movzx eax, byte ptr [r12 + r14]
3:
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    ja error_pos
    inc r14
num_exp_digits:
    cmp r14, r13
    jae num_done
    movzx eax, byte ptr [r12 + r14]
    mov edx, eax
    sub edx, '0'
    cmp edx, 9
    ja num_done
    inc r14
    jmp num_exp_digits

num_done:
    ret
