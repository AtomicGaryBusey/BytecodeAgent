.text
.globl _start

_start:
    xor r12, r12            # lines = 0
    xor r13, r13            # words = 0
    xor r14, r14            # bytes = 0
    xor r15d, r15d          # in_word = 0
    mov rbx, 0x100003E00    # whitespace bitmask: bits 9..13 and bit 32

.Lread_loop:
    xor eax, eax            # sys_read = 0
    xor edi, edi            # fd = 0 (stdin)
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall

    test rax, rax
    jle .Lprint_and_exit

    add r14, rax            # bytes += bytes_read
    mov rcx, rax            # loop counter
    lea rsi, [rip + buf]

.Lbyte_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 32
    ja .Lnot_ws

    cmp al, 10
    jne 1f
    inc r12                 # lines++
1:
    bt rbx, rax
    jnc .Lnot_ws

.Lis_ws:
    xor r15d, r15d          # in_word = 0
    jmp .Lnext_byte

.Lnot_ws:
    test r15d, r15d
    jnz .Lnext_byte
    inc r13                 # words++
    mov r15d, 1             # in_word = 1

.Lnext_byte:
    inc rsi
    dec rcx
    jnz .Lbyte_loop

    jmp .Lread_loop

.Lprint_and_exit:
    lea rdi, [rip + out_buf]

    mov rax, r12            # lines
    call format_u64

    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r13            # words
    call format_u64

    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r14            # bytes
    call format_u64

    mov byte ptr [rdi], 10  # '\n'
    inc rdi

    # Write output to stdout
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi            # count

.Lwrite_loop:
    test rdx, rdx
    jle .Lexit
    mov eax, 1              # sys_write = 1
    mov edi, 1              # fd = 1 (stdout)
    syscall
    test rax, rax
    jle .Lexit
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite_loop

.Lexit:
    mov eax, 60             # sys_exit = 60
    xor edi, edi            # exit status 0
    syscall

# Helper: format unsigned 64-bit integer in rax to buffer at rdi.
# Advances rdi past written digits.
format_u64:
    test rax, rax
    jnz .Lpos
    mov byte ptr [rdi], '0'
    inc rdi
    ret

.Lpos:
    sub rsp, 24
    lea rcx, [rsp + 24]
    mov r8d, 10

.Ldiv_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz .Ldiv_loop

    lea rdx, [rsp + 24]

.Lcopy_loop:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    cmp rcx, rdx
    jne .Lcopy_loop

    add rsp, 24
    ret

.bss
.balign 16
buf:
    .space 65536
out_buf:
    .space 128
