.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
    .align 16
in_buf:
    .space 8192
out_buf:
    .space 8192
row_buf:
    .space 16
in_head:
    .space 8
in_tail:
    .space 8
out_pos:
    .space 8
total_offset:
    .space 8

.text
.globl _start
_start:
main_loop:
    xor r13d, r13d

fill_row:
    cmp r13d, 16
    je row_ready
    call getc
    jc row_eof
    lea rsi, [rip + row_buf]
    mov byte ptr [rsi + r13], al
    inc r13d
    jmp fill_row

row_eof:
    test r13d, r13d
    jz done
    call print_row
    jmp done

row_ready:
    call print_row
    jmp main_loop

done:
    cmp qword ptr [rip + out_pos], 8064
    jb done_buf_ok
    call flush_out
done_buf_ok:
    lea rdi, [rip + out_buf]
    add rdi, qword ptr [rip + out_pos]

    mov eax, dword ptr [rip + total_offset]
    call format_hex8

    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + out_buf]
    sub rdi, rax
    mov qword ptr [rip + out_pos], rdi

    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

print_row:
    cmp qword ptr [rip + out_pos], 8064
    jb print_buf_ok
    call flush_out
print_buf_ok:
    lea rdi, [rip + out_buf]
    add rdi, qword ptr [rip + out_pos]

    mov eax, dword ptr [rip + total_offset]
    call format_hex8
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2

    lea r8, [rip + hex_digits]
    lea rsi, [rip + row_buf]
    xor ebx, ebx

slot_loop:
    cmp ebx, r13d
    jae slot_empty

    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    and edx, 0x0f
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    jmp slot_sep

slot_empty:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
    add rdi, 3

slot_sep:
    cmp ebx, 7
    jne slot_next
    mov byte ptr [rdi], ' '
    inc rdi

slot_next:
    inc ebx
    cmp ebx, 16
    jne slot_loop

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2

    lea rsi, [rip + row_buf]
    xor ebx, ebx

char_loop:
    cmp ebx, r13d
    jae char_done
    movzx edx, byte ptr [rsi + rbx]
    cmp dl, ' '
    jb char_nonprint
    cmp dl, '~'
    jbe char_store
char_nonprint:
    mov dl, '.'
char_store:
    mov byte ptr [rdi], dl
    inc rdi
    inc ebx
    jmp char_loop

char_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2

    lea rax, [rip + out_buf]
    sub rdi, rax
    mov qword ptr [rip + out_pos], rdi

    movsxd rax, r13d
    add qword ptr [rip + total_offset], rax
    ret

format_hex8:
    lea r8, [rip + hex_digits]
    mov ecx, 28
format_hex8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0x0f
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns format_hex8_loop
    ret

getc:
    mov rax, qword ptr [rip + in_head]
    cmp rax, qword ptr [rip + in_tail]
    jb getc_have_byte

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 8192
    syscall
    test rax, rax
    jle getc_eof

    lea rcx, [rip + in_buf]
    mov qword ptr [rip + in_head], rcx
    add rcx, rax
    mov qword ptr [rip + in_tail], rcx
    mov rax, qword ptr [rip + in_head]

getc_have_byte:
    movzx edx, byte ptr [rax]
    inc rax
    mov qword ptr [rip + in_head], rax
    mov eax, edx
    clc
    ret

getc_eof:
    stc
    ret

flush_out:
    mov r8, qword ptr [rip + out_pos]
    test r8, r8
    jz flush_done
    xor r9, r9

flush_loop:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, r9
    mov rdx, r8
    sub rdx, r9
    syscall
    test rax, rax
    jle flush_done
    add r9, rax
    cmp r9, r8
    jb flush_loop
    mov qword ptr [rip + out_pos], 0

flush_done:
    ret
