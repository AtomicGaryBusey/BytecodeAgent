.globl _start

.section .rodata
str_error:
    .ascii "error\n"
str_error_len = . - str_error

.text
in_buf_size = 16384
line_max = 2048

_start:
.main_loop:
    call read_line
    jc .exit_prog

    mov r13, rax
    call eval_line
    jmp .main_loop

.exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

read_line:
    xor r12d, r12d
.rl_loop:
    call get_byte
    jc .rl_eof
    cmp al, 10
    je .rl_newline
    cmp r12, line_max
    jae .rl_loop
    lea rcx, [rip + line_buf]
    mov byte ptr [rcx + r12], al
    inc r12
    jmp .rl_loop

.rl_newline:
    mov rax, r12
    clc
    ret

.rl_eof:
    test r12, r12
    jz .rl_no_more
    mov rax, r12
    clc
    ret

.rl_no_more:
    stc
    ret

get_byte:
    mov rax, qword ptr [rip + in_ptr]
    cmp rax, qword ptr [rip + in_end]
    jb .gb_have_byte

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, in_buf_size
    syscall

    test rax, rax
    jle .gb_eof

    lea rcx, [rip + in_buf]
    mov qword ptr [rip + in_ptr], rcx
    add rax, rcx
    mov qword ptr [rip + in_end], rax

.gb_have_byte:
    mov rax, qword ptr [rip + in_ptr]
    movzx edx, byte ptr [rax]
    inc rax
    mov qword ptr [rip + in_ptr], rax
    mov al, dl
    clc
    ret

.gb_eof:
    stc
    ret

eval_line:
    lea rbp, [rip + eval_stack]
    xor r12d, r12d
    lea rsi, [rip + line_buf]
    lea r14, [rsi + r13]

.eval_next_token:
.eval_skip_sep:
    cmp rsi, r14
    jae .eval_line_done
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je .eval_inc_sep
    cmp al, 9
    je .eval_inc_sep
    jmp .eval_have_token
.eval_inc_sep:
    inc rsi
    jmp .eval_skip_sep

.eval_have_token:
    mov rdi, rsi
.eval_find_end:
    cmp rdi, r14
    jae .eval_token_found
    movzx eax, byte ptr [rdi]
    cmp al, 32
    je .eval_token_found
    cmp al, 9
    je .eval_token_found
    inc rdi
    jmp .eval_find_end

.eval_token_found:
    mov r15, rdi
    mov rdx, rdi
    sub rdx, rsi

    cmp rdx, 1
    jne .not_operator

    movzx eax, byte ptr [rsi]
    cmp al, 43
    je .op_add
    cmp al, 45
    je .op_sub
    cmp al, 42
    je .op_mul
    cmp al, 47
    je .op_div
    cmp al, 37
    je .op_mod

.not_operator:
    mov rcx, rsi
    mov r8, rdx
    xor r9d, r9d

    movzx eax, byte ptr [rcx]
    cmp al, 45
    jne .parse_digits_start

    mov r9d, 1
    inc rcx
    dec r8

.parse_digits_start:
    cmp r8, 1
    jl .eval_error
    cmp r8, 18
    jg .eval_error

    xor eax, eax

.parse_digit_loop:
    test r8, r8
    jz .parse_digits_done

    movzx ebx, byte ptr [rcx]
    sub ebx, 48
    cmp ebx, 9
    ja .eval_error

    lea rax, [rax + rax*4]
    add rax, rax
    add rax, rbx

    inc rcx
    dec r8
    jmp .parse_digit_loop

.parse_digits_done:
    test r9d, r9d
    jz .push_val
    neg rax

.push_val:
    mov qword ptr [rbp + r12*8], rax
    inc r12
    mov rsi, r15
    jmp .eval_next_token

.op_add:
    cmp r12, 2
    jl .eval_error
    dec r12
    mov rbx, qword ptr [rbp + r12*8]
    dec r12
    mov rax, qword ptr [rbp + r12*8]
    add rax, rbx
    mov qword ptr [rbp + r12*8], rax
    inc r12
    mov rsi, r15
    jmp .eval_next_token

.op_sub:
    cmp r12, 2
    jl .eval_error
    dec r12
    mov rbx, qword ptr [rbp + r12*8]
    dec r12
    mov rax, qword ptr [rbp + r12*8]
    sub rax, rbx
    mov qword ptr [rbp + r12*8], rax
    inc r12
    mov rsi, r15
    jmp .eval_next_token

.op_mul:
    cmp r12, 2
    jl .eval_error
    dec r12
    mov rbx, qword ptr [rbp + r12*8]
    dec r12
    mov rax, qword ptr [rbp + r12*8]
    imul rax, rbx
    mov qword ptr [rbp + r12*8], rax
    inc r12
    mov rsi, r15
    jmp .eval_next_token

.op_div:
    cmp r12, 2
    jl .eval_error
    dec r12
    mov rbx, qword ptr [rbp + r12*8]
    dec r12
    mov rax, qword ptr [rbp + r12*8]
    test rbx, rbx
    jz .eval_error
    cmp rbx, -1
    jne 1f
    mov r8, 1
    shl r8, 63
    cmp rax, r8
    je .eval_error
1:
    cqo
    idiv rbx
    mov qword ptr [rbp + r12*8], rax
    inc r12
    mov rsi, r15
    jmp .eval_next_token

.op_mod:
    cmp r12, 2
    jl .eval_error
    dec r12
    mov rbx, qword ptr [rbp + r12*8]
    dec r12
    mov rax, qword ptr [rbp + r12*8]
    test rbx, rbx
    jz .eval_error
    cmp rbx, -1
    jne 1f
    mov r8, 1
    shl r8, 63
    cmp rax, r8
    je .eval_error
1:
    cqo
    idiv rbx
    mov qword ptr [rbp + r12*8], rdx
    inc r12
    mov rsi, r15
    jmp .eval_next_token

.eval_line_done:
    cmp r12, 1
    jne .eval_error
    mov rax, qword ptr [rbp]
    call print_number
    ret

.eval_error:
    lea rsi, [rip + str_error]
    mov rdx, str_error_len
    call write_all
    ret

print_number:
    lea rdi, [rip + out_num_buf + 31]
    mov byte ptr [rdi], 10

    xor r8d, r8d
    test rax, rax
    jns .pn_check_zero
    mov r8d, 1
    neg rax

.pn_check_zero:
    test rax, rax
    jnz .pn_loop_start
    dec rdi
    mov byte ptr [rdi], 48
    jmp .pn_add_sign

.pn_loop_start:
    mov rbx, 10
.pn_div_loop:
    test rax, rax
    jz .pn_add_sign

    xor edx, edx
    div rbx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    jmp .pn_div_loop

.pn_add_sign:
    test r8d, r8d
    jz .pn_write
    dec rdi
    mov byte ptr [rdi], 45

.pn_write:
    lea rdx, [rip + out_num_buf + 32]
    sub rdx, rdi
    mov rsi, rdi
    call write_all
    ret

write_all:
    test rdx, rdx
    jle .write_done
.write_loop:
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle .write_done
    add rsi, rax
    sub rdx, rax
    jnz .write_loop
.write_done:
    ret

.bss
.balign 16
in_ptr:
    .space 8
in_end:
    .space 8
in_buf:
    .space 16384
line_buf:
    .space 2048
eval_stack:
    .space 16384
out_num_buf:
    .space 32
