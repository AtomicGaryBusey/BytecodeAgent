.globl _start

.text
_start:
    lea rbp, [rip + eval_stack]

.Lstart_line:
    mov r12, 1
    xor r13d, r13d

.Lscan_line:
    call getc
    cmp eax, -1
    je .Leof_on_line
    cmp eax, 10
    je .Lnewline_on_line
    cmp eax, ' '
    je .Lwhitespace
    cmp eax, 9
    je .Lwhitespace

    xor r12d, r12d
    lea rdi, [rip + token_buf]
    mov byte ptr [rdi], al
    mov r14, 1

.Lread_token:
    call getc
    cmp eax, -1
    je .Ltoken_delim_eof
    cmp eax, 10
    je .Ltoken_delim_nl
    cmp eax, ' '
    je .Ltoken_delim_sep
    cmp eax, 9
    je .Ltoken_delim_sep

    cmp r14, 1000
    jae .Lskip_store
    lea rdi, [rip + token_buf]
    mov byte ptr [rdi + r14], al
.Lskip_store:
    inc r14
    jmp .Lread_token

.Lwhitespace:
    xor r12d, r12d
    jmp .Lscan_line

.Lnewline_on_line:
    cmp r13, 1
    jne .Lnl_err
    mov rax, qword ptr [rbp]
    call print_int64
    jmp .Lstart_line
.Lnl_err:
    call print_error
    jmp .Lstart_line

.Leof_on_line:
    test r12, r12
    jnz .Lexit_program
    cmp r13, 1
    jne .Leof_err
    mov rax, qword ptr [rbp]
    call print_int64
    jmp .Lexit_program
.Leof_err:
    call print_error
    jmp .Lexit_program

.Ltoken_delim_sep:
    call process_token
    test eax, eax
    jnz .Lsep_err
    jmp .Lscan_line
.Lsep_err:
    call print_error
    jmp skip_to_eol

.Ltoken_delim_nl:
    call process_token
    test eax, eax
    jnz .Ldelim_nl_err
    cmp r13, 1
    jne .Ldelim_nl_err
    mov rax, qword ptr [rbp]
    call print_int64
    jmp .Lstart_line
.Ldelim_nl_err:
    call print_error
    jmp .Lstart_line

.Ltoken_delim_eof:
    call process_token
    test eax, eax
    jnz .Ldelim_eof_err
    cmp r13, 1
    jne .Ldelim_eof_err
    mov rax, qword ptr [rbp]
    call print_int64
    jmp .Lexit_program
.Ldelim_eof_err:
    call print_error
    jmp .Lexit_program

skip_to_eol:
    call getc
    cmp eax, -1
    je .Lexit_program
    cmp eax, 10
    je .Lstart_line
    jmp skip_to_eol

.Lexit_program:
    call flush_out
    mov eax, 60
    xor edi, edi
    syscall

process_token:
    cmp r14, 1
    jne .Lcheck_literal

    movzx ecx, byte ptr [rip + token_buf]
    cmp cl, '+'
    je .Lop_add
    cmp cl, '-'
    je .Lop_sub
    cmp cl, '*'
    je .Lop_mul
    cmp cl, '/'
    je .Lop_div
    cmp cl, '%'
    je .Lop_mod

.Lcheck_literal:
    lea rdi, [rip + token_buf]
    xor r8d, r8d
    xor ecx, ecx

    mov al, byte ptr [rdi]
    cmp al, '-'
    jne .Lnot_neg
    mov r8d, 1
    inc ecx
.Lnot_neg:
    mov rdx, r14
    sub rdx, rcx
    cmp rdx, 1
    jb .Ltoken_error
    cmp rdx, 18
    ja .Ltoken_error

    xor rax, rax
.Ldigit_parse_loop:
    movzx esi, byte ptr [rdi + rcx]
    sub esi, '0'
    cmp esi, 9
    ja .Ltoken_error
    imul rax, rax, 10
    add rax, rsi
    inc rcx
    cmp rcx, r14
    jb .Ldigit_parse_loop

    test r8d, r8d
    jz .Lpush_val
    neg rax

.Lpush_val:
    mov qword ptr [rbp + r13*8], rax
    inc r13
    xor eax, eax
    ret

.Lop_add:
    cmp r13, 2
    jb .Ltoken_error
    mov rcx, qword ptr [rbp + r13*8 - 8]
    mov rax, qword ptr [rbp + r13*8 - 16]
    add rax, rcx
    sub r13, 1
    mov qword ptr [rbp + r13*8 - 8], rax
    xor eax, eax
    ret

.Lop_sub:
    cmp r13, 2
    jb .Ltoken_error
    mov rcx, qword ptr [rbp + r13*8 - 8]
    mov rax, qword ptr [rbp + r13*8 - 16]
    sub rax, rcx
    sub r13, 1
    mov qword ptr [rbp + r13*8 - 8], rax
    xor eax, eax
    ret

.Lop_mul:
    cmp r13, 2
    jb .Ltoken_error
    mov rcx, qword ptr [rbp + r13*8 - 8]
    mov rax, qword ptr [rbp + r13*8 - 16]
    imul rax, rcx
    sub r13, 1
    mov qword ptr [rbp + r13*8 - 8], rax
    xor eax, eax
    ret

.Lop_div:
    cmp r13, 2
    jb .Ltoken_error
    mov rcx, qword ptr [rbp + r13*8 - 8]
    mov rax, qword ptr [rbp + r13*8 - 16]
    test rcx, rcx
    jz .Ltoken_error
    mov r10, 1
    shl r10, 63
    cmp rax, r10
    jne 1f
    cmp rcx, -1
    je .Ltoken_error
1:
    cqo
    idiv rcx
    sub r13, 1
    mov qword ptr [rbp + r13*8 - 8], rax
    xor eax, eax
    ret

.Lop_mod:
    cmp r13, 2
    jb .Ltoken_error
    mov rcx, qword ptr [rbp + r13*8 - 8]
    mov rax, qword ptr [rbp + r13*8 - 16]
    test rcx, rcx
    jz .Ltoken_error
    mov r10, 1
    shl r10, 63
    cmp rax, r10
    jne 1f
    cmp rcx, -1
    je .Ltoken_error
1:
    cqo
    idiv rcx
    sub r13, 1
    mov qword ptr [rbp + r13*8 - 8], rdx
    xor eax, eax
    ret

.Ltoken_error:
    mov eax, 1
    ret

print_int64:
    push rbx
    push r12
    push r13

    test rax, rax
    jnz .Lpi_not_zero

    mov al, '0'
    call putc
    mov al, 10
    call putc
    pop r13
    pop r12
    pop rbx
    ret

.Lpi_not_zero:
    mov r13, rax
    test r13, r13
    jns .Lpi_positive

    mov al, '-'
    call putc
    neg r13

.Lpi_positive:
    mov rax, r13
    sub rsp, 32
    lea rdi, [rsp + 30]
    mov byte ptr [rsp + 31], 10
    mov r10, 10

.Lpi_loop:
    xor edx, edx
    div r10
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    test rax, rax
    jnz .Lpi_loop

    inc rdi
    lea r12, [rsp + 32]
    sub r12, rdi

.Lpi_out_loop:
    test r12, r12
    jz .Lpi_done
    movzx eax, byte ptr [rdi]
    call putc
    inc rdi
    dec r12
    jmp .Lpi_out_loop

.Lpi_done:
    add rsp, 32
    pop r13
    pop r12
    pop rbx
    ret

print_error:
    push rbx
    lea rbx, [rip + str_error]
.Lpe_loop:
    movzx eax, byte ptr [rbx]
    test al, al
    jz .Lpe_done
    call putc
    inc rbx
    jmp .Lpe_loop
.Lpe_done:
    pop rbx
    ret

putc:
    push rcx
    push rsi
    mov rcx, [rip + out_pos]
    cmp rcx, 65536
    jb 1f
    push rax
    call flush_out
    pop rax
    mov rcx, [rip + out_pos]
1:
    lea rsi, [rip + out_buf]
    mov byte ptr [rsi + rcx], al
    inc rcx
    mov [rip + out_pos], rcx
    pop rsi
    pop rcx
    ret

flush_out:
    push rbx
    push r12
    push rdi
    push rsi
    push rdx
    mov r12, [rip + out_pos]
    test r12, r12
    jz .Lflush_done
    lea rbx, [rip + out_buf]
.Lwrite_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall
    test rax, rax
    jle .Lflush_done
    add rbx, rax
    sub r12, rax
    jnz .Lwrite_loop
    mov qword ptr [rip + out_pos], 0
.Lflush_done:
    pop rdx
    pop rsi
    pop rdi
    pop r12
    pop rbx
    ret

refill_in:
    push rbx
    push rdi
    push rsi
    push rdx
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle .Lin_eof
    lea rbx, [rip + in_buf]
    mov [rip + in_ptr], rbx
    add rbx, rax
    mov [rip + in_end], rbx
    pop rdx
    pop rsi
    pop rdi
    pop rbx
    ret
.Lin_eof:
    mov qword ptr [rip + in_ptr], 0
    mov qword ptr [rip + in_end], 0
    pop rdx
    pop rsi
    pop rdi
    pop rbx
    ret

getc:
    mov rax, [rip + in_ptr]
    cmp rax, [rip + in_end]
    jb .Lgetc_has_byte
    call refill_in
    mov rax, [rip + in_ptr]
    cmp rax, [rip + in_end]
    jae .Lgetc_eof
.Lgetc_has_byte:
    movzx edx, byte ptr [rax]
    inc rax
    mov [rip + in_ptr], rax
    mov eax, edx
    ret
.Lgetc_eof:
    mov eax, -1
    ret

.section .rodata
str_error:
    .asciz "error\n"

.section .bss
.balign 16
in_buf:
    .zero 65536
out_buf:
    .zero 65536
eval_stack:
    .zero 16384
token_buf:
    .zero 1024
.balign 8
in_ptr:
    .zero 8
in_end:
    .zero 8
out_pos:
    .zero 8
