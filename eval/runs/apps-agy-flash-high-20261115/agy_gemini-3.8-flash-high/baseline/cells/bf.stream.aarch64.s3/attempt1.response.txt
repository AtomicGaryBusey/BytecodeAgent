    .globl _start

    .text
_start:
    // Read entire standard input into stdin_buf
    mov x19, #0
read_loop:
    mov x0, #0
    adrp x1, stdin_buf
    add x1, x1, :lo12:stdin_buf
    add x1, x1, x19
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x19, x19, x0
    b read_loop

read_done:
    // Locate first '!' (0x21) to split program and input
    adrp x1, stdin_buf
    add x1, x1, :lo12:stdin_buf
    mov x2, #0

find_excl_loop:
    cmp x2, x19
    b.ge no_excl_found
    ldrb w3, [x1, x2]
    cmp w3, #0x21
    b.eq excl_found
    add x2, x2, #1
    b find_excl_loop

no_excl_found:
    mov x20, x19
    add x21, x1, x19
    mov x22, x21
    b split_done

excl_found:
    mov x20, x2
    add x3, x2, #1
    add x21, x1, x3
    add x22, x1, x19

split_done:
    // Validate bracket balance and populate jump table
    adrp x23, jump_table
    add x23, x23, :lo12:jump_table
    adrp x24, bracket_stack
    add x24, x24, :lo12:bracket_stack
    mov x25, #0
    mov x26, #0

bracket_loop:
    cmp x26, x20
    b.ge bracket_check_done
    ldrb w3, [x1, x26]
    cmp w3, #0x5B
    b.eq handle_open_bracket
    cmp w3, #0x5D
    b.eq handle_close_bracket
    add x26, x26, #1
    b bracket_loop

handle_open_bracket:
    str w26, [x24, x25, lsl #2]
    add x25, x25, #1
    add x26, x26, #1
    b bracket_loop

handle_close_bracket:
    cbz x25, bracket_error
    sub x25, x25, #1
    ldr w4, [x24, x25, lsl #2]
    str w26, [x23, x4, lsl #2]
    str w4, [x23, x26, lsl #2]
    add x26, x26, #1
    b bracket_loop

bracket_check_done:
    cbnz x25, bracket_error

    // Initialize execution state
    adrp x19, stdin_buf
    add x19, x19, :lo12:stdin_buf
    adrp x24, tape
    add x24, x24, :lo12:tape
    mov x25, #0
    adrp x26, stdout_buf
    add x26, x26, :lo12:stdout_buf
    mov x27, #0

exec_loop:
    cmp x25, x20
    b.ge exec_done

    ldrb w0, [x19, x25]

    cmp w0, #61
    b.gt is_above_61

    cmp w0, #45
    b.gt is_dot_or_less

    cmp w0, #43
    b.eq cmd_plus
    cmp w0, #45
    b.eq cmd_minus
    cmp w0, #44
    b.eq cmd_comma
    b cmd_comment

is_dot_or_less:
    cmp w0, #60
    b.eq cmd_less
    cmp w0, #46
    b.eq cmd_dot
    b cmd_comment

is_above_61:
    cmp w0, #62
    b.eq cmd_greater
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close
    b cmd_comment

cmd_greater:
    add x24, x24, #1
    add x25, x25, #1
    b exec_loop

cmd_less:
    sub x24, x24, #1
    add x25, x25, #1
    b exec_loop

cmd_plus:
    ldrb w0, [x24]
    add w0, w0, #1
    strb w0, [x24]
    add x25, x25, #1
    b exec_loop

cmd_minus:
    ldrb w0, [x24]
    sub w0, w0, #1
    strb w0, [x24]
    add x25, x25, #1
    b exec_loop

cmd_dot:
    ldrb w0, [x24]
    strb w0, [x26, x27]
    add x27, x27, #1
    cmp x27, #16, lsl #12
    b.lt 1f
    bl flush_stdout
1:
    add x25, x25, #1
    b exec_loop

cmd_comma:
    cmp x21, x22
    b.ge 1f
    ldrb w0, [x21], #1
    strb w0, [x24]
    b 2f
1:
    strb wzr, [x24]
2:
    add x25, x25, #1
    b exec_loop

cmd_open:
    ldrb w0, [x24]
    cbnz w0, 1f
    ldr w28, [x23, x25, lsl #2]
    mov x25, x28
1:
    add x25, x25, #1
    b exec_loop

cmd_close:
    ldrb w0, [x24]
    cbz w0, 1f
    ldr w28, [x23, x25, lsl #2]
    mov x25, x28
1:
    add x25, x25, #1
    b exec_loop

cmd_comment:
    add x25, x25, #1
    b exec_loop

exec_done:
    bl flush_stdout
    mov x0, #0
    mov x8, #93
    svc #0

bracket_error:
    mov x0, #1
    mov x8, #93
    svc #0

flush_stdout:
    cbz x27, flush_done
    mov x2, x27
    adrp x1, stdout_buf
    add x1, x1, :lo12:stdout_buf
flush_write_loop:
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    ble flush_done
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, flush_write_loop
    mov x27, #0
flush_done:
    ret

    .bss
    .balign 16
tape:
    .zero 30000

    .balign 16
stdin_buf:
    .zero 65536

    .balign 16
stdout_buf:
    .zero 65536

    .balign 16
jump_table:
    .zero 262144

    .balign 16
bracket_stack:
    .zero 4096
