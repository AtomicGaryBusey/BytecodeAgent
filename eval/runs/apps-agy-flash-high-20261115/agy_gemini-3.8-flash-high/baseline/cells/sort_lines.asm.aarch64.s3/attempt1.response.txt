.globl _start
.text

_start:
    // Read all input into input_buf
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, #0

read_loop:
    movz    x2, #1, lsl #16
    sub     x2, x2, x20
    cbz     x2, read_done

    mov     x8, #63
    mov     x0, #0
    mov     x1, x19
    svc     #0

    cmp     x0, #0
    b.le    read_done

    add     x19, x19, x0
    add     x20, x20, x0
    b       read_loop

read_done:
    cbz     x20, exit_success

    // Parse lines from input_buf
    adrp    x21, input_buf
    add     x21, x21, :lo12:input_buf
    add     x22, x21, x20
    adrp    x23, lines
    add     x23, x23, :lo12:lines
    mov     x24, #0
    mov     x25, x21
    mov     x26, x21

parse_loop:
    cmp     x25, x22
    b.ge    parse_end

    ldrb    w27, [x25], #1
    cmp     w27, #10
    b.ne    parse_loop

    sub     x28, x25, #1
    sub     x28, x28, x26
    add     x0, x23, x24, lsl #4
    stp     x26, x28, [x0]
    add     x24, x24, #1
    mov     x26, x25
    b       parse_loop

parse_end:
    cmp     x26, x22
    b.ge    lines_ready

    sub     x28, x22, x26
    add     x0, x23, x24, lsl #4
    stp     x26, x28, [x0]
    add     x24, x24, #1

lines_ready:
    mov     x0, x24
    bl      heapsort

    // Build output buffer
    adrp    x19, output_buf
    add     x19, x19, :lo12:output_buf
    mov     x20, x19
    adrp    x21, lines
    add     x21, x21, :lo12:lines
    mov     x22, #0

output_lines_loop:
    cmp     x22, x24
    b.ge    output_ready

    add     x0, x21, x22, lsl #4
    ldp     x1, x2, [x0]

copy_line_loop:
    cbz     x2, copy_line_done
    ldrb    w3, [x1], #1
    strb    w3, [x19], #1
    sub     x2, x2, #1
    b       copy_line_loop

copy_line_done:
    mov     w3, #10
    strb    w3, [x19], #1
    add     x22, x22, #1
    b       output_lines_loop

output_ready:
    sub     x23, x19, x20
    mov     x21, x20

write_loop:
    cbz     x23, exit_success
    mov     x8, #64
    mov     x0, #1
    mov     x1, x21
    mov     x2, x23
    svc     #0

    cmp     x0, #0
    b.le    exit_success

    add     x21, x21, x0
    sub     x23, x23, x0
    b       write_loop

exit_success:
    mov     x8, #93
    mov     x0, #0
    svc     #0

// heapsort(line_count [x0])
heapsort:
    stp     x29, x30, [sp, #-48]!
    mov     x29, sp
    stp     x19, x20, [sp, #16]
    stp     x21, x22, [sp, #32]

    mov     x19, x0
    cmp     x19, #1
    b.le    heapsort_done

    adrp    x22, lines
    add     x22, x22, :lo12:lines

    lsr     x21, x19, #1
    sub     x21, x21, #1

build_heap_loop:
    mov     x0, x21
    mov     x1, x19
    bl      sift_down
    subs    x21, x21, #1
    b.ge    build_heap_loop

    sub     x21, x19, #1

extract_loop:
    add     x1, x22, x21, lsl #4
    ldp     x2, x3, [x22]
    ldp     x4, x5, [x1]
    stp     x4, x5, [x22]
    stp     x2, x3, [x1]

    mov     x0, #0
    mov     x1, x21
    bl      sift_down

    subs    x21, x21, #1
    b.gt    extract_loop

heapsort_done:
    ldp     x21, x22, [sp, #32]
    ldp     x19, x20, [sp, #16]
    ldp     x29, x30, [sp], #48
    ret

// sift_down(r [x0], count [x1])
sift_down:
    stp     x29, x30, [sp, #-64]!
    mov     x29, sp
    stp     x19, x20, [sp, #16]
    stp     x21, x22, [sp, #32]
    stp     x23, x24, [sp, #48]

    mov     x19, x0
    mov     x20, x1
    adrp    x23, lines
    add     x23, x23, :lo12:lines

sift_loop:
    lsl     x21, x19, #1
    add     x21, x21, #1
    cmp     x21, x20
    b.ge    sift_done

    add     x22, x21, #1
    cmp     x22, x20
    b.ge    compare_root_child

    add     x0, x23, x21, lsl #4
    add     x1, x23, x22, lsl #4
    bl      compare_lines
    cmp     w0, #0
    b.ge    compare_root_child
    mov     x21, x22

compare_root_child:
    add     x0, x23, x19, lsl #4
    add     x1, x23, x21, lsl #4
    bl      compare_lines
    cmp     w0, #0
    b.ge    sift_done

    add     x0, x23, x19, lsl #4
    add     x1, x23, x21, lsl #4
    ldp     x2, x3, [x0]
    ldp     x4, x5, [x1]
    stp     x4, x5, [x0]
    stp     x2, x3, [x1]

    mov     x19, x21
    b       sift_loop

sift_done:
    ldp     x23, x24, [sp, #48]
    ldp     x21, x22, [sp, #32]
    ldp     x19, x20, [sp, #16]
    ldp     x29, x30, [sp], #64
    ret

// compare_lines(Line *A [x0], Line *B [x1]) -> w0
compare_lines:
    ldr     x2, [x0]
    ldr     x3, [x0, #8]
    ldr     x4, [x1]
    ldr     x5, [x1, #8]

    cmp     x3, x5
    csel    x6, x3, x5, lo

cmp_loop:
    cbz     x6, cmp_check_len
    ldrb    w7, [x2], #1
    ldrb    w8, [x4], #1
    sub     x6, x6, #1
    cmp     w7, w8
    b.eq    cmp_loop

    sub     w0, w7, w8
    ret

cmp_check_len:
    cmp     x3, x5
    b.eq    cmp_equal
    b.lo    cmp_less
    mov     w0, #1
    ret

cmp_less:
    mov     w0, #-1
    ret

cmp_equal:
    mov     w0, #0
    ret

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
lines:
    .zero 65536
.balign 16
output_buf:
    .zero 131072
