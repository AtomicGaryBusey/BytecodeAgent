.text
.globl _start
_start:
    // Read all input into in_buf
    mov x19, #0

.Lread_loop:
    movz x2, #2, lsl #16
    subs x2, x2, x19
    b.ls .Lread_done

    mov x0, #0
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    add x1, x1, x19
    mov x8, #63
    svc #0

    cmp x0, #0
    ble .Lread_done

    add x19, x19, x0
    b .Lread_loop

.Lread_done:
    // If input is empty, exit immediately
    cbz x19, .Lexit_success

    // Process lines
    adrp x20, in_buf
    add x20, x20, :lo12:in_buf
    add x21, x20, x19
    mov x22, x20

    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x25, x24

.Lline_start:
    cmp x22, x21
    b.hs .Lprocess_done

    mov x23, x22

.Lscan_nl:
    cmp x22, x21
    b.hs .Lfound_eof_line

    ldrb w0, [x22], #1
    cmp w0, #10
    b.ne .Lscan_nl

    // Found newline at (x22 - 1)
    sub x26, x22, #1
    mov w27, #0
    b .Lrev_loop

.Lfound_eof_line:
    mov x26, x21
    mov w27, #1

.Lrev_loop:
    cmp x26, x23
    b.ls .Lrev_done

    sub x26, x26, #1
    ldrb w1, [x26]
    strb w1, [x25], #1
    b .Lrev_loop

.Lrev_done:
    mov w1, #10
    strb w1, [x25], #1
    cbnz w27, .Lprocess_done
    b .Lline_start

.Lprocess_done:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    sub x20, x25, x24
    cbz x20, .Lexit_success

    mov x19, x24

.Lwrite_loop:
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov x8, #64
    svc #0

    cmp x0, #0
    ble .Lexit_success

    add x19, x19, x0
    sub x20, x20, x0
    cbnz x20, .Lwrite_loop

.Lexit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
in_buf:
    .zero 131072
out_buf:
    .zero 131072
