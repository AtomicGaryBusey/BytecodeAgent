.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_CAP, 131072

    .text
    .globl _start
    .balign 4

_start:
    adrp x22, in_buf
    add x22, x22, :lo12:in_buf

    adrp x23, out_buf
    add x23, x23, :lo12:out_buf

    mov x19, #0
    mov x20, #0
    mov x21, #0

read_more:
    movz x2, #2, lsl #16
    cmp x19, x2
    b.hs at_eof

    sub x2, x2, x19

    mov x0, #0
    add x1, x22, x19
    mov x8, #SYS_READ
    svc #0

    cmp x0, #0
    b.le at_eof

    add x19, x19, x0

scan_loop:
    cmp x21, x19
    b.hs read_more

    ldrb w24, [x22, x21]
    cmp w24, #10
    b.eq found_newline

    add x21, x21, #1
    b scan_loop

found_newline:
    sub x25, x21, x20

    add x1, x22, x21
    mov x2, x23
    mov x3, x25

rev_loop:
    cbz x3, rev_done
1:
    ldrb w4, [x1, #-1]!
    strb w4, [x2], #1
    subs x3, x3, #1
    b.ne 1b

rev_done:
    mov w4, #10
    strb w4, [x2], #1

    add x2, x25, #1
    mov x1, x23

write_loop:
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le exit_program
    subs x2, x2, x0
    add x1, x1, x0
    b.gt write_loop

    add x21, x21, #1
    mov x20, x21
    b scan_loop

at_eof:
    cmp x20, x19
    b.hs exit_program

    sub x25, x19, x20

    add x1, x22, x19
    mov x2, x23
    mov x3, x25

final_rev_loop:
    cbz x3, final_rev_done
1:
    ldrb w4, [x1, #-1]!
    strb w4, [x2], #1
    subs x3, x3, #1
    b.ne 1b

final_rev_done:
    mov w4, #10
    strb w4, [x2], #1

    add x2, x25, #1
    mov x1, x23

final_write_loop:
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le exit_program
    subs x2, x2, x0
    add x1, x1, x0
    b.gt final_write_loop

exit_program:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
exit_loop:
    b exit_loop

    .bss
    .balign 16
in_buf:
    .zero BUF_CAP
out_buf:
    .zero BUF_CAP
