// json_lint: static Linux executable for AArch64
// Checks if stdin is exactly one valid JSON document (RFC 8259)

    .globl  _start

    .section .rodata
    .align  3
str_ok:
    .ascii  "ok\n"

str_err_prefix:
    .ascii  "error at "

    .section .bss
    .align  4
input_buf:
    .space  131072

    .section .text
_start:
    // Read standard input completely into input_buf
    // x19 = input_buf pointer
    // x20 = total bytes read
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, #0

read_loop:
    mov     x0, #0                  // stdin
    add     x1, x19, x20            // current buffer pos
    movz    x2, #1, lsl #16         // 65536
    sub     x2, x2, x20             // count = 65536 - x20
    cbz     x2, read_done
    mov     x8, #63                 // read syscall
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    // x21 = current parsing position (offset from 0)
    mov     x21, #0

    // Top-level: ws value ws EOF
    bl      skip_ws

    // If input is empty or whitespace-only -> error at len (x21)
    cmp     x21, x20
    b.ge    err_at_pos

    // Parse the single JSON value
    bl      parse_value

    // Skip trailing whitespace
    bl      skip_ws

    // Must be at end of input
    cmp     x21, x20
    b.ne    err_at_pos

    // Valid JSON document!
    adrp    x1, str_ok
    add     x1, x1, :lo12:str_ok
    mov     x2, #3
    bl      write_stdout

    mov     x0, #0
    mov     x8, #93                 // exit(0)
    svc     #0

// -------------------------------------------------------------
// skip_ws: advance x21 over whitespace characters
// -------------------------------------------------------------
skip_ws:
    cmp     x21, x20
    b.ge    skip_ws_done
    ldrb    w0, [x19, x21]
    cmp     w0, #' '
    b.eq    skip_ws_advance
    cmp     w0, #'\t'
    b.eq    skip_ws_advance
    cmp     w0, #'\n'
    b.eq    skip_ws_advance
    cmp     w0, #'\r'
    b.eq    skip_ws_advance
    ret

skip_ws_advance:
    add     x21, x21, #1
    b       skip_ws

skip_ws_done:
    ret

// -------------------------------------------------------------
// parse_value: dispatches to appropriate value parser based on first byte
// -------------------------------------------------------------
parse_value:
    cmp     x21, x20
    b.ge    err_at_pos

    ldrb    w0, [x19, x21]
    cmp     w0, #'{'
    b.eq    parse_object
    cmp     w0, #'['
    b.eq    parse_array
    cmp     w0, #'"'
    b.eq    parse_string
    cmp     w0, #'t'
    b.eq    parse_true
    cmp     w0, #'f'
    b.eq    parse_false
    cmp     w0, #'n'
    b.eq    parse_null
    cmp     w0, #'-'
    b.eq    parse_number
    cmp     w0, #'0'
    b.lt    err_at_pos
    cmp     w0, #'9'
    b.le    parse_number

    b       err_at_pos

// -------------------------------------------------------------
// parse_true: parses 'true'
// -------------------------------------------------------------
parse_true:
    add     x21, x21, #1            // consume 't'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'r'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'r'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'u'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'u'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'e'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'e'
    ret

// -------------------------------------------------------------
// parse_false: parses 'false'
// -------------------------------------------------------------
parse_false:
    add     x21, x21, #1            // consume 'f'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'a'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'a'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'l'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'l'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'s'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 's'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'e'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'e'
    ret

// -------------------------------------------------------------
// parse_null: parses 'null'
// -------------------------------------------------------------
parse_null:
    add     x21, x21, #1            // consume 'n'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'u'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'u'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'l'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'l'
    cmp     x21, x20
    b.ge    err_at_pos
    ldrb    w0, [x19, x21]
    cmp     w0, #'l'
    b.ne    err_at_pos

    add     x21, x21, #1            // consume 'l'
    ret

// -------------------------------------------------------------
// parse_string: parses a JSON string
// -------------------------------------------------------------
parse_string:
    add     x21, x21, #1            // consume opening '"'

parse_string_loop:
    cmp     x21, x20
    b.ge    err_at_pos

    ldrb    w0, [x19, x21]
    cmp     w0, #'"'
    b.eq    parse_string_done

    cmp     w0, #'\\'
    b.eq    parse_string_escape

    // Control characters 0x00..0x1F are errors
    cmp     w0, #0x20
    b.lo    err_at_pos

    add     x21, x21, #1
    b       parse_string_loop

parse_string_done:
    add     x21, x21, #1            // consume closing '"'
    ret

parse_string_escape:
    add     x21, x21, #1            // consume '\\'
    cmp     x21, x20
    b.ge    err_at_pos

    ldrb    w0, [x19, x21]
    cmp     w0, #'"'
    b.eq    parse_string_esc_ok
    cmp     w0, #'\\'
    b.eq    parse_string_esc_ok
    cmp     w0, #'/'
    b.eq    parse_string_esc_ok
    cmp     w0, #'b'
    b.eq    parse_string_esc_ok
    cmp     w0, #'f'
    b.eq    parse_string_esc_ok
    cmp     w0, #'n'
    b.eq    parse_string_esc_ok
    cmp     w0, #'r'
    b.eq    parse_string_esc_ok
    cmp     w0, #'t'
    b.eq    parse_string_esc_ok
    cmp     w0, #'u'
    b.eq    parse_string_unicode

    // Invalid escape character
    b       err_at_pos

parse_string_esc_ok:
    add     x21, x21, #1
    b       parse_string_loop

parse_string_unicode:
    add     x21, x21, #1            // consume 'u'
    mov     w1, #4                  // 4 hex digits required

parse_string_hex_loop:
    cmp     x21, x20
    b.ge    err_at_pos

    ldrb    w0, [x
