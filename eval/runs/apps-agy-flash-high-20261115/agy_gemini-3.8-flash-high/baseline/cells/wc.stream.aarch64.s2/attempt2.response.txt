// Program wc: count lines, words, and bytes from stdin and print them.

    .text
    .globl _start

_start:
    // Initialize counters:
    // x19 = lines
    // x20 = words
    // x21 = bytes
    // x22 = in_word flag
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

.Lread_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    movz x2, #1, lsl #16
    mov x8, #63
    svc #0

    cmp x0, #0
    b.le .Lread_done

    add x21, x21, x0

    adrp x13, inbuf
    add x13, x13, :lo12:inbuf
    add x14, x13, x0

.Lbyte_loop:
    ldrb w15, [x13], #1

    // Newline check (0x0A = 10)
    cmp w15, #10
    cinc x19, x19, eq

    // Whitespace check:
    // space (0x20 = 32), tab..CR (0x09..0x0D = 9..13)
    cmp w15, #32
    b.eq .Lis_space
    sub w16, w15, #9
    cmp w16, #4
    b.ls .Lis_space

    // Non-whitespace character:
    // If not currently in a word, increment words
    cmp x22, #0
    cinc x20, x20, eq
    mov x22, #1
    b .Lcheck_end

.Lis_space:
    mov x22, #0

.Lcheck_end:
    cmp x13, x14
    b.ne .Lbyte_loop

    b .Lread_loop

.Lread_done:
    // Format the output into outbuf: "<lines> <words> <bytes>\n"
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf

    // 1. Format lines count
    mov x0, x19
    bl format_uint

    // Space separator
    mov w2, #32
    strb w2, [x1], #1

    // 2. Format words count
    mov x0, x20
    bl format_uint

    // Space separator
    mov w2, #32
    strb w2, [x1], #1

    // 3. Format bytes count
    mov x0, x21
    bl format_uint

    // Newline terminator
    mov w2, #10
    strb w2, [x1], #1

    // Write buffer to stdout (fd = 1)
    adrp x0, outbuf
    add x0, x0, :lo12:outbuf
    sub x21, x1, x0
    mov x20, x0
    mov x19, #1

.Lwrite_loop:
    cbz x21, .Lwrite_done
    mov x0, x19
    mov x1, x20
    mov x2, x21
    mov x8, #64
    svc #0

    cmp x0, #0
    b.le .Lwrite_done
    add x20, x20, x0
    sub x21, x21, x0
    b .Lwrite_loop

.Lwrite_done:
    // Exit with status 0
    mov x0, #0
    mov x8, #93
    svc #0

.Lhang:
    b .Lhang

// format_uint:
// Converts unsigned 64-bit integer in x0 to decimal ASCII.
// Writes digits to buffer at x1.
// Returns updated buffer pointer in x1.
format_uint:
    cbnz x0, .Lfu_nonzero
    mov w2, #48
    strb w2, [x1], #1
    ret

.Lfu_nonzero:
    sub sp, sp, #32
    add x2, sp, #32
    mov x4, #10

.Lfu_div_loop:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #48
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x3
    cbnz x0, .Lfu_div_loop

    add x3, sp, #32
.Lfu_copy_loop:
    ldrb w5, [x2], #1
    strb w5, [x1], #1
    cmp x2, x3
    b.ne .Lfu_copy_loop

    add sp, sp, #32
    ret

    .bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 128
