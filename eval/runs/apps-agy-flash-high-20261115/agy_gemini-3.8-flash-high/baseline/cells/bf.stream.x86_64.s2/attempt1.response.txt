.equ OP_END,   0
.equ OP_PLUS,  1
.equ OP_MINUS, 2
.equ OP_RIGHT, 3
.equ OP_LEFT,  4
.equ OP_OPEN,  5
.equ OP_CLOSE, 6
.equ OP_DOT,   7
.equ OP_COMMA, 8

.text
.globl _start

_start:
    xor r12d, r12d
.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + stdin_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jbe .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop
.Lread_done:

    lea rdi, [rip + stdin_buf]
    mov ecx, r12d
    test ecx, ecx
    jz .Lno_bang
    mov al, '!'
    repne scasb
    jne .Lno_bang

    lea rdx, [rip + stdin_buf]
    mov r8, rdi
    sub rdi, rdx
    dec rdi
    lea r9, [rdx + r12]
    jmp .Lsplit_done

.Lno_bang:
    mov rdi, r12
    lea rdx, [rip + stdin_buf]
    lea r8, [rdx + r12]
    mov r9, r8
.Lsplit_done:

    lea rsi, [rip + stdin_buf]
    mov rcx, rdi
    lea r15, [rip + code_buf]
    lea r10, [rip + jump_target]
    lea rbx, [rip + bracket_stack]
    xor edx, edx
    xor ebp, ebp

.Lfilter_loop:
    test rcx, rcx
    jz .Lfilter_done
    mov al, byte ptr [rsi]
    inc rsi
    dec rcx

    cmp al, '+'
    je .Lop_plus
    cmp al, '-'
    je .Lop_minus
    cmp al, '>'
    je .Lop_right
    cmp al, '<'
    je .Lop_left
    cmp al, '.'
    je .Lop_dot
    cmp al, ','
    je .Lop_comma
    cmp al, '['
    je .Lop_open
    cmp al, ']'
    je .Lop_close
    jmp .Lfilter_loop

.Lop_plus:
    mov byte ptr [r15 + rdx], OP_PLUS
    inc edx
    jmp .Lfilter_loop

.Lop_minus:
    mov byte ptr [r15 + rdx], OP_MINUS
    inc edx
    jmp .Lfilter_loop

.Lop_right:
    mov byte ptr [r15 + rdx], OP_RIGHT
    inc edx
    jmp .Lfilter_loop

.Lop_left:
    mov byte ptr [r15 + rdx], OP_LEFT
    inc edx
    jmp .Lfilter_loop

.Lop_dot:
    mov byte ptr [r15 + rdx], OP_DOT
    inc edx
    jmp .Lfilter_loop

.Lop_comma:
    mov byte ptr [r15 + rdx], OP_COMMA
    inc edx
    jmp .Lfilter_loop

.Lop_open:
    mov byte ptr [r15 + rdx], OP_OPEN
    mov dword ptr [rbx + rbp*4], edx
    inc ebp
    inc edx
    jmp .Lfilter_loop

.Lop_close:
    test ebp, ebp
    jz .Lbracket_err
    dec ebp
    mov eax, dword ptr [rbx + rbp*4]
    mov byte ptr [r15 + rdx], OP_CLOSE

    lea edi, [rdx + 1]
    mov dword ptr [r10 + rax*4], edi
    lea edi, [rax + 1]
    mov dword ptr [r10 + rdx*4], edi

    inc edx
    jmp .Lfilter_loop

.Lbracket_err:
    mov eax, 60
    mov edi, 1
    syscall

.Lfilter_done:
    test ebp, ebp
    jnz .Lbracket_err

    mov byte ptr [r15 + rdx], OP_END

    lea rbx, [rip + tape]
    xor r12d, r12d
    lea r13, [rip + out_buf]
    xor r14d, r14d
    lea rbp, [rip + dispatch_table]

    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_plus:
    inc byte ptr [rbx]
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_minus:
    dec byte ptr [rbx]
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_right:
    inc rbx
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_left:
    dec rbx
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_open:
    cmp byte ptr [rbx], 0
    jne .Lopen_not_zero
    mov r12d, dword ptr [r10 + r12*4]
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]
.Lopen_not_zero:
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_close:
    cmp byte ptr [rbx], 0
    je .Lclose_zero
    mov r12d, dword ptr [r10 + r12*4]
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]
.Lclose_zero:
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [r13 + r14], al
    inc r14d
    cmp r14d, 65536
    jne .Ldot_not_full
    call flush_out
.Ldot_not_full:
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

handle_comma:
    cmp r8, r9
    jae .Lcomma_eof
    mov al, byte ptr [r8]
    inc r8
    mov byte ptr [rbx], al
    jmp .Lcomma_next
.Lcomma_eof:
    mov byte ptr [rbx], 0
.Lcomma_next:
    inc r12
    movzx eax, byte ptr [r15 + r12]
    jmp qword ptr [rbp + rax*8]

exec_done:
    call flush_out
    mov eax, 60
    xor edi, edi
    syscall

flush_out:
    test r14, r14
    jz .Lflush_ret
    push r12
    xor r12d, r12d
.Lflush_loop:
    mov eax, 1
    mov edi, 1
    lea rsi, [r13 + r12]
    mov rdx, r14
    sub rdx, r12
    syscall
    test rax, rax
    jle .Lflush_done
    add r12, rax
    cmp r12, r14
    jb .Lflush_loop
.Lflush_done:
    pop r12
    xor r14d, r14d
.Lflush_ret:
    ret

.section .rodata
.align 8
dispatch_table:
    .quad exec_done
    .quad handle_plus
    .quad handle_minus
    .quad handle_right
    .quad handle_left
    .quad handle_open
    .quad handle_close
    .quad handle_dot
    .quad handle_comma

.bss
.align 16
stdin_buf:
    .zero 65540
code_buf:
    .zero 65540
out_buf:
    .zero 65540
tape:
    .zero 32768
jump_target:
    .zero 262160
bracket_stack:
    .zero 262160
