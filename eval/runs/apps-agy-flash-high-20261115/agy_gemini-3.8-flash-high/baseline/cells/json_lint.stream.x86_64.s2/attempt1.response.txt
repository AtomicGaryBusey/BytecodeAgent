.section .rodata
str_ok:
    .ascii "ok\n"

.section .bss
.align 16
input_buf:
    .zero 131072

.text
.globl _start

_start:
    lea r14, [rip + input_buf]
    xor r12, r12
.Lread_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [r14 + r12]
    mov edx, 65536
    syscall
    test rax, rax
    js .Lread_error
    jz .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_error:
    mov eax, 60
    mov edi, 1
    syscall

.Lread_done:
    mov r13, r12
    xor r12, r12

    call skip_ws
    cmp r12, r13
    jae error_at_r12

    call parse_value

    call skip_ws
    cmp r12, r13
    jne error_at_r12

    jmp report_ok

skip_ws:
    cmp r12, r13
    jae .Lws_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, 32
    ja .Lws_done
    mov rdx, 0x100002600
    bt rdx, rax
    jnc .Lws_done
    inc r12
    jmp skip_ws
.Lws_done:
    ret

parse_value:
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error_at_r12
    cmp al, '9'
    jbe parse_number
    jmp error_at_r12

parse_true:
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'r'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'u'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'e'
    jne error_at_r12
    inc r12
    ret

parse_false:
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'a'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'l'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 's'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'e'
    jne error_at_r12
    inc r12
    ret

parse_null:
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'u'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'l'
    jne error_at_r12
    inc r12
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], 'l'
    jne error_at_r12
    inc r12
    ret

parse_string:
    inc r12
.Lstr_loop:
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '"'
    je .Lstr_done
    cmp al, 0x20
    jb error_at_r12
    cmp al, '\\'
    je .Lstr_escape
    inc r12
    jmp .Lstr_loop

.Lstr_done:
    inc r12
    ret

.Lstr_escape:
    inc r12
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '"'
    je .Lesc_ok
    cmp al, '\\'
    je .Lesc_ok
    cmp al, '/'
    je .Lesc_ok
    cmp al, 'b'
    je .Lesc_ok
    cmp al, 'f'
    je .Lesc_ok
    cmp al, 'n'
    je .Lesc_ok
    cmp al, 'r'
    je .Lesc_ok
    cmp al, 't'
    je .Lesc_ok
    cmp al, 'u'
    je .Lesc_u
    jmp error_at_r12

.Lesc_ok:
    inc r12
    jmp .Lstr_loop

.Lesc_u:
    inc r12
    mov ecx, 4
.Lhex_loop:
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb error_at_r12
    cmp al, '9'
    jbe .Lhex_ok
    cmp al, 'A'
    jb error_at_r12
    cmp al, 'F'
    jbe .Lhex_ok
    cmp al, 'a'
    jb error_at_r12
    cmp al, 'f'
    jbe .Lhex_ok
    jmp error_at_r12

.Lhex_ok:
    inc r12
    dec ecx
    jnz .Lhex_loop
    jmp .Lstr_loop

parse_number:
    cmp byte ptr [r14 + r12], '-'
    jne .Lnum_int
    inc r12
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb error_at_r12
    cmp al, '9'
    ja error_at_r12

.Lnum_int:
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jne .Lnum_nonzero
    inc r12
    cmp r12, r13
    jae .Lnum_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb .Lnum_frac
    cmp al, '9'
    jbe error_at_r12
    jmp .Lnum_frac

.Lnum_nonzero:
    inc r12
.Lnum_int_loop:
    cmp r12, r13
    jae .Lnum_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb .Lnum_frac
    cmp al, '9'
    ja .Lnum_frac
    inc r12
    jmp .Lnum_int_loop

.Lnum_frac:
    cmp byte ptr [r14 + r12], '.'
    jne .Lnum_exp
    inc r12
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb error_at_r12
    cmp al, '9'
    ja error_at_r12
    inc r12
.Lnum_frac_loop:
    cmp r12, r13
    jae .Lnum_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb .Lnum_exp
    cmp al, '9'
    ja .Lnum_exp
    inc r12
    jmp .Lnum_frac_loop

.Lnum_exp:
    movzx eax, byte ptr [r14 + r12]
    cmp al, 'e'
    je .Lnum_has_exp
    cmp al, 'E'
    jne .Lnum_done

.Lnum_has_exp:
    inc r12
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '+'
    je .Lnum_exp_sign
    cmp al, '-'
    jne .Lnum_exp_digits

.Lnum_exp_sign:
    inc r12
    cmp r12, r13
    jae error_at_r12

.Lnum_exp_digits:
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb error_at_r12
    cmp al, '9'
    ja error_at_r12
    inc r12
.Lnum_exp_loop:
    cmp r12, r13
    jae .Lnum_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, '0'
    jb .Lnum_done
    cmp al, '9'
    ja .Lnum_done
    inc r12
    jmp .Lnum_exp_loop

.Lnum_done:
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], ']'
    je .Larr_empty

.Larr_loop:
    call parse_value
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, ']'
    je .Larr_close
    cmp al, ','
    jne error_at_r12
    inc r12
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], ']'
    je error_at_r12
    jmp .Larr_loop

.Larr_empty:
.Larr_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], '}'
    je .Lobj_empty

.Lobj_loop:
    cmp byte ptr [r14 + r12], '"'
    jne error_at_r12
    call parse_string
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    cmp byte ptr [r14 + r12], ':'
    jne error_at_r12
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    movzx eax, byte ptr [r14 + r12]
    cmp al, '}'
    je .Lobj_close
    cmp al, ','
    jne error_at_r12
    inc r12
    call skip_ws
    cmp r12, r13
    jae error_at_r12
    jmp .Lobj_loop

.Lobj_empty:
.Lobj_close:
    inc r12
    ret

report_ok:
    lea rsi, [rip + str_ok]
    mov edx, 3
.Lwrite_ok_loop:
    mov eax, 1
    mov edi, 1
    mov r8, rsi
    mov r9, rdx
    syscall
    sub r9, rax
    add r8, rax
    mov rsi, r8
    mov rdx, r9
    test rdx, rdx
    jnz .Lwrite_ok_loop

    mov eax, 60
    xor edi, edi
    syscall

error_at_r12:
    sub rsp, 32
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 10
    mov rax, r12
    mov ecx, 10
.Ldec_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .Ldec_loop

    sub rdi, 9
    mov byte ptr [rdi + 0], 'e'
    mov byte ptr [rdi + 1], 'r'
    mov byte ptr [rdi + 2], 'r'
    mov byte ptr [rdi + 3], 'o'
    mov byte ptr [rdi + 4], 'r'
    mov byte ptr [rdi + 5], ' '
    mov byte ptr [rdi + 6], 'a'
    mov byte ptr [rdi + 7], 't'
    mov byte ptr [rdi + 8], ' '

    lea rdx, [rsp + 32]
    sub rdx, rdi
    mov rsi, rdi

.Lwrite_err_loop:
    mov eax, 1
    mov edi, 1
    mov r8, rsi
    mov r9, rdx
    syscall
    sub r9, rax
    add r8, rax
    mov rsi, r8
    mov rdx, r9
    test rdx, rdx
    jnz .Lwrite_err_loop

    mov eax, 60
    mov edi, 1
    syscall
