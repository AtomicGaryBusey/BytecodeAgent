.globl _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.section .bss
.align 16
row_buf:
    .zero 16
in_buf:
    .zero 4096
out_buf:
    .zero 4096

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    lea r15, [rip + out_buf]

.Lmain_loop:
    call fill_row
    test ebx, ebx
    jz .Ldone

    # Ensure buffer has space for at least 128 bytes
    lea rax, [rip + out_buf + 3968]
    cmp r15, rax
    jb 1f
    call flush_out
1:
    # 1. Format offset as 8 lowercase hex digits, then two spaces
    mov edx, r12d
    mov ecx, 8
.Lhex_off:
    rol edx, 4
    mov eax, edx
    and eax, 0xf
    mov al, byte ptr [rip + hex_table + rax]
    mov byte ptr [r15], al
    inc r15
    dec ecx
    jnz .Lhex_off

    mov byte ptr [r15], ' '
    mov byte ptr [r15 + 1], ' '
    add r15, 2

    # 2. Format 16 hex byte positions
    xor ebp, ebp
.Lbyte_loop:
    cmp ebp, ebx
    jae .Lspace_slot

    movzx eax, byte ptr [rip + row_buf + rbp]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rip + hex_table + rdx]
    and eax, 0xf
    mov al, byte ptr [rip + hex_table + rax]
    mov byte ptr [r15], dl
    mov byte ptr [r15 + 1], al
    mov byte ptr [r15 + 2], ' '
    add r15, 3
    jmp .Lafter_slot

.Lspace_slot:
    mov byte ptr [r15], ' '
    mov byte ptr [r15 + 1], ' '
    mov byte ptr [r15 + 2], ' '
    add r15, 3

.Lafter_slot:
    cmp ebp, 7
    jne .Lnot_seven
    mov byte ptr [r15], ' '
    inc r15
.Lnot_seven:
    inc ebp
    cmp ebp, 16
    jne .Lbyte_loop

    # 3. Format ASCII column: one space, '|', n characters, '|', newline
    mov byte ptr [r15], ' '
    mov byte ptr [r15 + 1], '|'
    add r15, 2

    xor ebp, ebp
.Lascii_loop:
    cmp ebp, ebx
    jae .Lascii_done
    movzx eax, byte ptr [rip + row_buf + rbp]
    lea ecx, [rax - 0x20]
    cmp ecx, 0x5e
    mov edx, '.'
    cmovbe edx, eax
    mov byte ptr [r15], dl
    inc r15
    inc ebp
    jmp .Lascii_loop

.Lascii_done:
    mov byte ptr [r15], '|'
    mov byte ptr [r15 + 1], 10
    add r15, 2

    add r12, rbx
    cmp ebx, 16
    jb .Ldone
    jmp .Lmain_loop

.Ldone:
    lea rax, [rip + out_buf + 3968]
    cmp r15, rax
    jb 1f
    call flush_out
1:
    mov edx, r12d
    mov ecx, 8
.Lhex_final:
    rol edx, 4
    mov eax, edx
    and eax, 0xf
    mov al, byte ptr [rip + hex_table + rax]
    mov byte ptr [r15], al
    inc r15
    dec ecx
    jnz .Lhex_final

    mov byte ptr [r15], 10
    inc r15

    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

# fill_row: reads up to 16 bytes into row_buf
# Returns count in ebx (0 to 16)
fill_row:
    xor ebx, ebx
.Lfill_loop:
    cmp ebx, 16
    je .Lfill_done
    call read_byte
    jc .Lfill_done
    mov byte ptr [rip + row_buf + rbx], al
    inc ebx
    jmp .Lfill_loop
.Lfill_done:
    ret

# read_byte: returns next byte in al, CF=0; on EOF/error returns CF=1
read_byte:
    cmp r13, r14
    jb 1f
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 4096
    syscall
    test rax, rax
    jle 2f
    lea r13, [rip + in_buf]
    lea r14, [r13 + rax]
1:
    movzx eax, byte ptr [r13]
    inc r13
    clc
    ret
2:
    stc
    ret

# flush_out: writes buffered bytes in out_buf to stdout
flush_out:
    lea rsi, [rip + out_buf]
    mov rdx, r15
    sub rdx, rsi
    jle 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle 2f
    add rsi, rax
    sub rdx, rax
    ja 1b
2:
    lea r15, [rip + out_buf]
    ret
