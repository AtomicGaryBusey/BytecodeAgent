.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "

.section .bss
.balign 16
input_buf:
    .zero 131072
out_buf:
    .zero 64

.section .text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]

.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 65536
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    xor ebx, ebx
    call skip_ws
    cmp rbx, r12
    jae error_exit
    call parse_value
    call skip_ws
    cmp rbx, r12
    jne error_exit

success_exit:
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

error_exit:
    lea rdi, [rip + out_buf]
    mov rax, qword ptr [rip + err_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_prefix + 8]
    mov byte ptr [rdi + 8], al
    add rdi, 9

    test rbx, rbx
    jnz .Lconvert_digits
    mov byte ptr [rdi], '0'
    inc rdi
    jmp .Lappend_newline

.Lconvert_digits:
    mov rax, rbx
    mov r10, rsp
    mov rcx, 10
.Ldigit_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    push rdx
    test rax, rax
    jnz .Ldigit_loop

.Lpop_digits:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    cmp rsp, r10
    jne .Lpop_digits

.Lappend_newline:
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    call write_all

    mov eax, 60
    mov edi, 1
    syscall

write_all:
.Lwrite_loop:
    test rdx, rdx
    jle .Lwrite_done
    mov eax, 1
    syscall
    test rax, rax
    jle .Lwrite_done
    add rsi, rax
    sub rdx, rax
    mov edi, 1
    jmp .Lwrite_loop
.Lwrite_done:
    ret

skip_ws:
.Lws_loop:
    cmp rbx, r12
    jae .Lws_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x20
    je .Lws_next
    cmp al, 0x09
    je .Lws_next
    cmp al, 0x0A
    je .Lws_next
    cmp al, 0x0D
    je .Lws_next
    ret
.Lws_next:
    inc rbx
    jmp .Lws_loop
.Lws_done:
    ret

parse_value:
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error_exit
    cmp al, '9'
    jbe parse_number
    jmp error_exit

parse_string:
    inc rbx
.Lstr_loop:
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je .Lstr_done
    cmp al, '\\'
    je .Lstr_escape
    cmp al, 0x20
    jb error_exit
    inc rbx
    jmp .Lstr_loop

.Lstr_done:
    inc rbx
    ret

.Lstr_escape:
    inc rbx
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je .Lesc_single
    cmp al, '\\'
    je .Lesc_single
    cmp al, '/'
    je .Lesc_single
    cmp al, 'b'
    je .Lesc_single
    cmp al, 'f'
    je .Lesc_single
    cmp al, 'n'
    je .Lesc_single
    cmp al, 'r'
    je .Lesc_single
    cmp al, 't'
    je .Lesc_single
    cmp al, 'u'
    je .Lesc_u
    jmp error_exit

.Lesc_single:
    inc rbx
    jmp .Lstr_loop

.Lesc_u:
    inc rbx
    mov ecx, 4
.Lhex_loop:
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb error_exit
    cmp al, '9'
    jbe .Lhex_ok
    cmp al, 'A'
    jb error_exit
    cmp al, 'F'
    jbe .Lhex_ok
    cmp al, 'a'
    jb error_exit
    cmp al, 'f'
    ja error_exit
.Lhex_ok:
    inc rbx
    dec ecx
    jnz .Lhex_loop
    jmp .Lstr_loop

parse_true:
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'r'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'u'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'e'
    jne error_exit
    inc rbx
    ret

parse_false:
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'a'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'l'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 's'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'e'
    jne error_exit
    inc rbx
    ret

parse_null:
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'u'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'l'
    jne error_exit
    inc rbx
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], 'l'
    jne error_exit
    inc rbx
    ret

parse_number:
    cmp byte ptr [r13 + rbx], '-'
    jne .Lnum_int
    inc rbx
    cmp rbx, r12
    jae error_exit

.Lnum_int:
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    je .Lnum_zero
    cmp al, '1'
    jb error_exit
    cmp al, '9'
    ja error_exit
    inc rbx

.Lnum_digits:
    cmp rbx, r12
    jae .Lnum_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb .Lnum_frac_exp
    cmp al, '9'
    ja .Lnum_frac_exp
    inc rbx
    jmp .Lnum_digits

.Lnum_zero:
    inc rbx
    cmp rbx, r12
    jae .Lnum_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb .Lnum_frac_exp
    cmp al, '9'
    jbe error_exit
    jmp .Lnum_check_exp

.Lnum_frac_exp:
    cmp al, '.'
    jne .Lnum_check_exp
    inc rbx
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb error_exit
    cmp al, '9'
    ja error_exit
    inc rbx

.Lnum_frac_digits:
    cmp rbx, r12
    jae .Lnum_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb .Lnum_check_exp
    cmp al, '9'
    ja .Lnum_check_exp
    inc rbx
    jmp .Lnum_frac_digits

.Lnum_check_exp:
    cmp al, 'e'
    je .Lnum_exp
    cmp al, 'E'
    je .Lnum_exp
    jmp .Lnum_done

.Lnum_exp:
    inc rbx
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '+'
    je .Lnum_exp_sign
    cmp al, '-'
    je .Lnum_exp_sign
    jmp .Lnum_exp_digits

.Lnum_exp_sign:
    inc rbx
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]

.Lnum_exp_digits:
    cmp al, '0'
    jb error_exit
    cmp al, '9'
    ja error_exit
    inc rbx

.Lnum_exp_more:
    cmp rbx, r12
    jae .Lnum_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb .Lnum_done
    cmp al, '9'
    ja .Lnum_done
    inc rbx
    jmp .Lnum_exp_more

.Lnum_done:
    ret

parse_member:
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], '"'
    jne error_exit
    call parse_string
    call skip_ws
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], ':'
    jne error_exit
    inc rbx
    call skip_ws
    jmp parse_value

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], '}'
    je .Lobj_empty
    call parse_member

.Lobj_loop:
    call skip_ws
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '}'
    je .Lobj_close
    cmp al, ','
    jne error_exit
    inc rbx
    call skip_ws
    call parse_member
    jmp .Lobj_loop

.Lobj_empty:
.Lobj_close:
    inc rbx
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae error_exit
    cmp byte ptr [r13 + rbx], ']'
    je .Larr_empty
    call parse_value

.Larr_loop:
    call skip_ws
    cmp rbx, r12
    jae error_exit
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ']'
    je .Larr_close
    cmp al, ','
    jne error_exit
    inc rbx
    call skip_ws
    call parse_value
    jmp .Larr_loop

.Larr_empty:
.Larr_close:
    inc rbx
    ret
