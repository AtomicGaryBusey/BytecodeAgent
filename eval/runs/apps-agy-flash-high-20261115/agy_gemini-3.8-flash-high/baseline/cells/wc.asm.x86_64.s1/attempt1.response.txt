.globl _start
.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15d, r15d

.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 65536
    syscall

    test rax, rax
    jle .Lread_done

    add r14, rax

    lea rsi, [rip + in_buf]
    lea rbx, [rsi + rax]

.Lbyte_loop:
    cmp rsi, rbx
    je .Lread_loop

    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne .Lcheck_ws
    inc r12

.Lcheck_ws:
    cmp al, 32
    je .Lis_ws
    sub al, 9
    cmp al, 4
    jbe .Lis_ws

    test r15d, r15d
    jnz .Lbyte_loop
    inc r13
    mov r15d, 1
    jmp .Lbyte_loop

.Lis_ws:
    xor r15d, r15d
    jmp .Lbyte_loop

.Lread_done:
    lea rbx, [rip + counts]
    mov qword ptr [rbx], r12
    mov qword ptr [rbx + 8], r13
    mov qword ptr [rbx + 16], r14

    lea rdi, [rip + out_buf]
    xor ebp, ebp

.Lformat_loop:
    mov rax, qword ptr [rbx + rbp*8]

    test rax, rax
    jnz .Lformat_nonzero

    mov byte ptr [rdi], 48
    inc rdi
    jmp .Lformat_check_sep

.Lformat_nonzero:
    lea r8, [rip + num_buf + 32]
    mov r10d, 10

.Ldiv_loop:
    xor edx, edx
    div r10
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz .Ldiv_loop

    lea r9, [rip + num_buf + 32]

.Lcopy_loop:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jne .Lcopy_loop

.Lformat_check_sep:
    cmp ebp, 2
    je .Lsep_newline

    mov byte ptr [rdi], 32
    inc rdi
    inc ebp
    jmp .Lformat_loop

.Lsep_newline:
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi

    mov r12, rsi
    mov r13, rdx

.Lwrite_loop:
    test r13, r13
    jle .Lexit
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle .Lexit
    add r12, rax
    sub r13, rax
    jmp .Lwrite_loop

.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
in_buf:
    .zero 65536
counts:
    .zero 24
num_buf:
    .zero 32
out_buf:
    .zero 128
