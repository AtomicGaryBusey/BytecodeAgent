.globl _start

.text
_start:
    xor r12, r12            # lines = 0
    xor r13, r13            # words = 0
    xor r14, r14            # bytes = 0
    xor r15, r15            # in_word = 0

.Lread_loop:
    xor eax, eax            # sys_read = 0
    xor edi, edi            # fd = 0 (stdin)
    lea rsi, [rip + in_buf]
    mov edx, 65536          # buffer size
    syscall

    test rax, rax
    jle .Lread_done

    add r14, rax            # bytes += count
    mov rcx, rax            # loop counter
    lea rsi, [rip + in_buf]

.Lbyte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10              # '\n' (0x0A)
    jne .Lnot_newline
    inc r12                 # lines++

.Lnot_newline:
    cmp al, 32              # ' ' (0x20)
    je .Lis_whitespace

    sub eax, 9
    cmp eax, 4              # 0x09 .. 0x0D (tab, newline, vt, ff, cr)
    jbe .Lis_whitespace

    # Non-whitespace byte
    test r15b, r15b
    jnz .Lnext_byte
    inc r13                 # words++
    mov r15b, 1
    jmp .Lnext_byte

.Lis_whitespace:
    xor r15b, r15b

.Lnext_byte:
    dec rcx
    jnz .Lbyte_loop
    jmp .Lread_loop

.Lread_done:
    # Format: "<lines> <words> <bytes>\n"
    lea rdi, [rip + out_buf]

    mov rax, r12            # lines
    call format_u64

    mov byte ptr [rdi], 32  # ' '
    inc rdi

    mov rax, r13            # words
    call format_u64

    mov byte ptr [rdi], 32  # ' '
    inc rdi

    mov rax, r14            # bytes
    call format_u64

    mov byte ptr [rdi], 10  # '\n'
    inc rdi

    # Write output buffer to stdout
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi            # total length

.Lwrite_loop:
    test rdx, rdx
    jz .Lexit
    mov eax, 1              # sys_write = 1
    mov edi, 1              # fd = 1 (stdout)
    syscall
    test rax, rax
    jle .Lexit
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite_loop

.Lexit:
    mov eax, 60             # sys_exit = 60
    xor edi, edi            # exit status = 0
    syscall

# format_u64: converts unsigned 64-bit int in rax to decimal ASCII at [rdi]
# Advances rdi past the written characters.
format_u64:
    sub rsp, 32
    lea r8, [rsp + 32]
    mov rcx, r8
    mov r9, 10

.Ldiv_loop:
    xor edx, edx
    div r9
    dec rcx
    add dl, 48              # '0'
    mov byte ptr [rcx], dl
    test rax, rax
    jnz .Ldiv_loop

.Lcopy_loop:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    cmp rcx, r8
    jne .Lcopy_loop

    add rsp, 32
    ret

.section .bss
.align 16
in_buf:
    .space 65536
out_buf:
    .space 128
