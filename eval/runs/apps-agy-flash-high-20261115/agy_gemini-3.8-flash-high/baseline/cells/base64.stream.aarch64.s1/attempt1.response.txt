    .globl  _start
    .text

_start:
    // Read input from standard input (fd = 0) into in_buf
    adrp    x19, in_buf
    add     x19, x19, :lo12:in_buf
    mov     x20, #0
    movz    x22, #1, lsl #16        // buffer capacity = 65536

.Lread_loop:
    sub     x2, x22, x20            // remaining buffer capacity
    cbz     x2, .Lread_done

    mov     w0, #0                  // stdin
    mov     x1, x19
    mov     w8, #63                 // sys_read = 63
    svc     #0

    cmp     x0, #0
    ble     .Lread_done

    add     x19, x19, x0
    add     x20, x20, x0
    b       .Lread_loop

.Lread_done:
    // If empty input, exit immediately with status 0
    cbz     x20, .Lexit_success

    // Set up pointers for encoding
    adrp    x21, in_buf
    add     x21, x21, :lo12:in_buf   // x21 = src pointer
    add     x22, x21, x20           // x22 = src_end

    adrp    x23, out_buf
    add     x23, x23, :lo12:out_buf  // x23 = dst pointer

    adrp    x24, base64_table
    add     x24, x24, :lo12:base64_table

    mov     w25, #0                 // w25 = column counter (0..76)

.Lgroup3_loop:
    sub     x0, x22, x21
    cmp     x0, #3
    blo     .Lremainder

    ldrb    w1, [x21], #1
    ldrb    w2, [x21], #1
    ldrb    w3, [x21], #1

    lsl     w1, w1, #16
    orr     w4, w1, w2, lsl #8
    orr     w4, w4, w3

    ubfx    w5, w4, #18, #6
    ubfx    w6, w4, #12, #6
    ubfx    w7, w4, #6, #6
    ubfx    w8, w4, #0, #6

    ldrb    w5, [x24, w5, uxtw]
    ldrb    w6, [x24, w6, uxtw]
    ldrb    w7, [x24, w7, uxtw]
    ldrb    w8, [x24, w8, uxtw]

    strb    w5, [x23], #1
    strb    w6, [x23], #1
    strb    w7, [x23], #1
    strb    w8, [x23], #1

    add     w25, w25, #4
    cmp     w25, #76
    bne     .Lgroup3_loop

    mov     w0, #10                 // '\n'
    strb    w0, [x23], #1
    mov     w25, #0
    b       .Lgroup3_loop

.Lremainder:
    cbz     x0, .Lfinish_output
    cmp     x0, #1
    bne     .Lrem2

    // Remainder of 1 byte: produces 2 base64 chars + "=="
    ldrb    w1, [x21]
    lsl     w4, w1, #16

    ubfx    w5, w4, #18, #6
    ubfx    w6, w4, #12, #6
    ldrb    w5, [x24, w5, uxtw]
    ldrb    w6, [x24, w6, uxtw]
    mov     w7, #61                 // '='
    mov     w8, #61                 // '='

    strb    w5, [x23], #1
    strb    w6, [x23], #1
    strb    w7, [x23], #1
    strb    w8, [x23], #1

    add     w25, w25, #4
    b       .Lfinish_output

.Lrem2:
    // Remainder of 2 bytes: produces 3 base64 chars + "="
    ldrb    w1, [x21]
    ldrb    w2, [x21, #1]
    lsl     w1, w1, #16
    orr     w4, w1, w2, lsl #8

    ubfx    w5, w4, #18, #6
    ubfx    w6, w4, #12, #6
    ubfx    w7, w4, #6, #6
    ldrb    w5, [x24, w5, uxtw]
    ldrb    w6, [x24, w6, uxtw]
    ldrb    w7, [x24, w7, uxtw]
    mov     w8, #61                 // '='

    strb    w5, [x23], #1
    strb    w6, [x23], #1
    strb    w7, [x23], #1
    strb    w8, [x23], #1

    add     w25, w25, #4

.Lfinish_output:
    // If the line has characters that haven't been terminated with '\n', add '\n'
    cbz     w25, .Lwrite_all
    mov     w0, #10                 // '\n'
    strb    w0, [x23], #1

.Lwrite_all:
    adrp    x1, out_buf
    add     x1, x1, :lo12:out_buf
    sub     x2, x23, x1             // count of bytes in out_buf
    cbz     x2, .Lexit_success

    mov     x19, x1
    mov     x20, x2

.Lwrite_loop:
    mov     w0, #1                  // stdout
    mov     x1, x19
    mov     x2, x20
    mov     w8, #64                 // sys_write = 64
    svc     #0

    cmp     x0, #0
    ble     .Lexit_success

    add     x19, x19, x0
    sub     x20, x20, x0
    cbnz    x20, .Lwrite_loop

.Lexit_success:
    mov     w0, #0                  // status = 0
    mov     w8, #93                 // sys_exit = 93
    svc     #0

    .section .rodata
    .balign 16
base64_table:
    .ascii  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .bss
    .balign 16
in_buf:
    .zero   65536
out_buf:
    .zero   131072
