.globl _start

.text
_start:
    xor r12, r12            # lines = 0
    xor r13, r13            # words = 0
    xor r14, r14            # bytes = 0
    xor r15d, r15d          # in_word = 0

read_loop:
    xor eax, eax            # sys_read = 0
    xor edi, edi            # fd = 0 (stdin)
    lea rsi, [rip + in_buf]
    mov edx, 65536          # count
    syscall

    test rax, rax
    jle done_reading

    add r14, rax            # bytes += bytes_read
    mov rcx, rax            # byte loop counter
    lea rsi, [rip + in_buf]

byte_loop:
    movzx edx, byte ptr [rsi]
    inc rsi

    cmp dl, 10              # '\n'
    jne not_newline
    inc r12                 # lines++
not_newline:
    cmp dl, 32              # ' '
    je is_whitespace
    sub dl, 9
    cmp dl, 4               # 9 (tab) to 13 (CR)
    jbe is_whitespace

    # Non-whitespace byte
    test r15b, r15b
    jnz next_byte
    inc r13                 # words++
    mov r15b, 1
    jmp next_byte

is_whitespace:
    xor r15b, r15b

next_byte:
    dec rcx
    jnz byte_loop
    jmp read_loop

done_reading:
    lea rdi, [rip + out_buf]

    mov rax, r12            # lines
    call format_uint

    mov byte ptr [rdi], 32  # ' '
    inc rdi

    mov rax, r13            # words
    call format_uint

    mov byte ptr [rdi], 32  # ' '
    inc rdi

    mov rax, r14            # bytes
    call format_uint

    mov byte ptr [rdi], 10  # '\n'
    inc rdi

    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi            # total length to write

write_loop:
    test rdx, rdx
    jz exit_prog
    mov eax, 1              # sys_write = 1
    mov edi, 1              # fd = 1 (stdout)
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_prog:
    mov eax, 60             # sys_exit = 60
    xor edi, edi            # status = 0
    syscall

# format_uint: writes decimal representation of rax to [rdi], advancing rdi
format_uint:
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 48  # '0'
    inc rdi
    ret
1:
    sub rsp, 24
    lea r8, [rsp + 24]
    mov r9, r8
    mov rcx, 10
2:
    xor edx, edx
    div rcx
    add dl, 48              # '0'
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    cmp r9, r8
    jne 3b
    add rsp, 24
    ret

.section .bss
.p2align 4
in_buf:
    .space 65536
out_buf:
    .space 128
