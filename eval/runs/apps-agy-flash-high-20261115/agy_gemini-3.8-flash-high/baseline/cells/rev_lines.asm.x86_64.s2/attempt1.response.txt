.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 131072
    sub edx, r12d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_ok

    lea rsi, [rip + in_buf]
    mov rbx, rsi
    lea rbp, [rsi + r12]
    lea rdi, [rip + out_buf]

scan_loop:
    cmp rsi, rbp
    jae scan_done
    mov al, byte ptr [rsi]
    cmp al, 10
    jne next_char

    mov rcx, rsi
reverse_line:
    cmp rcx, rbx
    je reverse_done
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_line

reverse_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rbx, [rsi + 1]

next_char:
    inc rsi
    jmp scan_loop

scan_done:
    cmp rbx, rbp
    jae write_output

    mov rcx, rbp
reverse_last_line:
    cmp rcx, rbx
    je reverse_last_done
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_last_line

reverse_last_done:
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jle exit_ok

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
in_buf:
    .space 131072
out_buf:
    .space 131072
