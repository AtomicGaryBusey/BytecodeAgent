.equ BUF_SIZE, 131072

.globl _start

.section .rodata
ok_str:
    .ascii "ok\n"

.section .bss
.align 16
input_buf:
    .zero 131072
out_buf:
    .zero 64

.text

_start:
    xor ebx, ebx
.Lread_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, BUF_SIZE
    sub edx, ebx
    jle .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add ebx, eax
    cmp ebx, BUF_SIZE
    jb .Lread_loop

.Lread_done:
    lea rbp, [rip + input_buf]
    mov rsi, rbp
    lea r12, [rbp + rbx]

    call skip_ws
    cmp rsi, r12
    jae report_error
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne report_error

    jmp report_ok

skip_ws:
.Lws_loop:
    cmp rsi, r12
    jae .Lws_done
    mov al, byte ptr [rsi]
    cmp al, ' '
    je .Lws_adv
    cmp al, 0x09
    je .Lws_adv
    cmp al, 0x0A
    je .Lws_adv
    cmp al, 0x0D
    jne .Lws_done
.Lws_adv:
    inc rsi
    jmp .Lws_loop
.Lws_done:
    ret

parse_value:
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe parse_number
    jmp report_error

parse_true:
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'r'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'u'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'e'
    jne report_error
    inc rsi
    ret

parse_false:
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'a'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'l'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 's'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'e'
    jne report_error
    inc rsi
    ret

parse_null:
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'u'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'l'
    jne report_error
    inc rsi
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], 'l'
    jne report_error
    inc rsi
    ret

parse_string:
    inc rsi
.Lstr_loop:
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    cmp al, '"'
    je .Lstr_done
    cmp al, '\\'
    je .Lstr_esc
    cmp al, 0x20
    jb report_error
    inc rsi
    jmp .Lstr_loop

.Lstr_done:
    inc rsi
    ret

.Lstr_esc:
    inc rsi
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    cmp al, '"'
    je .Lstr_esc_ok
    cmp al, '\\'
    je .Lstr_esc_ok
    cmp al, '/'
    je .Lstr_esc_ok
    cmp al, 'b'
    je .Lstr_esc_ok
    cmp al, 'f'
    je .Lstr_esc_ok
    cmp al, 'n'
    je .Lstr_esc_ok
    cmp al, 'r'
    je .Lstr_esc_ok
    cmp al, 't'
    je .Lstr_esc_ok
    cmp al, 'u'
    je .Lstr_esc_u
    jmp report_error

.Lstr_esc_ok:
    inc rsi
    jmp .Lstr_loop

.Lstr_esc_u:
    inc rsi
    mov ecx, 4
.Lstr_u_loop:
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    mov dl, al
    sub dl, '0'
    cmp dl, 10
    jb .Lstr_u_hex_ok
    mov dl, al
    or dl, 0x20
    sub dl, 'a'
    cmp dl, 6
    jae report_error
.Lstr_u_hex_ok:
    inc rsi
    dec ecx
    jnz .Lstr_u_loop
    jmp .Lstr_loop

parse_number:
    cmp byte ptr [rsi], '-'
    jne .Lnum_pos
    inc rsi
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    cmp al, '0'
    je .Lnum_zero
    cmp al, '1'
    jb report_error
    cmp al, '9'
    ja report_error
    inc rsi
    jmp .Lnum_digits

.Lnum_pos:
    mov al, byte ptr [rsi]
    cmp al, '0'
    je .Lnum_zero
    inc rsi

.Lnum_digits:
    cmp rsi, r12
    jae .Lnum_frac
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae .Lnum_frac
    inc rsi
    jmp .Lnum_digits

.Lnum_zero:
    inc rsi
    cmp rsi, r12
    jae .Lnum_frac
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jb report_error
    jmp .Lnum_frac

.Lnum_frac:
    cmp rsi, r12
    jae .Lnum_done
    cmp byte ptr [rsi], '.'
    jne .Lnum_exp
    inc rsi
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae report_error
    inc rsi
.Lnum_frac_digits:
    cmp rsi, r12
    jae .Lnum_exp
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae .Lnum_exp
    inc rsi
    jmp .Lnum_frac_digits

.Lnum_exp:
    cmp rsi, r12
    jae .Lnum_done
    mov al, byte ptr [rsi]
    cmp al, 'e'
    je .Lnum_exp_start
    cmp al, 'E'
    jne .Lnum_done

.Lnum_exp_start:
    inc rsi
    cmp rsi, r12
    jae report_error
    mov al, byte ptr [rsi]
    cmp al, '+'
    je .Lnum_exp_sign
    cmp al, '-'
    je .Lnum_exp_sign
    jmp .Lnum_exp_check_digit

.Lnum_exp_sign:
    inc rsi
    cmp rsi, r12
    jae report_error

.Lnum_exp_check_digit:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae report_error
    inc rsi
.Lnum_exp_digits:
    cmp rsi, r12
    jae .Lnum_done
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae .Lnum_done
    inc rsi
    jmp .Lnum_exp_digits

.Lnum_done:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], ']'
    je .Larr_empty

.Larr_loop:
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], ']'
    je .Larr_close
    cmp byte ptr [rsi], ','
    jne report_error
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae report_error
    jmp .Larr_loop

.Larr_empty:
.Larr_close:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], '}'
    je .Lobj_empty

.Lobj_loop:
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], '"'
    jne report_error
    call parse_string
    call skip_ws
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], ':'
    jne report_error
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae report_error
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae report_error
    cmp byte ptr [rsi], '}'
    je .Lobj_close
    cmp byte ptr [rsi], ','
    jne report_error
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae report_error
    jmp .Lobj_loop

.Lobj_empty:
.Lobj_close:
    inc rsi
    ret

write_all:
.Lwrite_loop:
    test edx, edx
    jle .Lwrite_done
    mov eax, 1
    push rdi
    push rsi
    push rdx
    syscall
    pop rdx
    pop rsi
    pop rdi
    test rax, rax
    jle .Lwrite_done
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite_loop
.Lwrite_done:
    ret

report_ok:
    lea rsi, [rip + ok_str]
    mov edx, 3
    mov edi, 1
    call write_all

    mov eax, 60
    xor edi, edi
    syscall

report_error:
    sub rsi, rbp
    lea rdi, [rip + out_buf + 63]
    mov byte ptr [rdi], 0x0A
    mov rax, rsi

.Lfmt_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .Lfmt_loop

    sub rdi, 9
    mov byte ptr [rdi + 0], 'e'
    mov byte ptr [rdi + 1], 'r'
    mov byte ptr [rdi + 2], 'r'
    mov byte ptr [rdi + 3], 'o'
    mov byte ptr [rdi + 4], 'r'
    mov byte ptr [rdi + 5], ' '
    mov byte ptr [rdi + 6], 'a'
    mov byte ptr [rdi + 7], 't'
    mov byte ptr [rdi + 8], ' '

    lea rdx, [rip + out_buf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    call write_all

    mov eax, 60
    mov edi, 1
    syscall
