.globl _start

.section .rodata
.balign 8
b64_table:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
in_buf:
.zero 65536
out_buf:
.zero 131072

.text
.balign 16
_start:
    // Read all standard input into in_buf
    mov x19, #0

read_loop:
    mov x0, #0
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    add x1, x1, x19
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done

    mov x8, #63
    svc #0

    cmp x0, #0
    ble read_done

    add x19, x19, x0
    b read_loop

read_done:
    // Empty input prints nothing
    cbz x19, exit_zero

    adrp x20, in_buf
    add x20, x20, :lo12:in_buf
    add x19, x20, x19

    adrp x21, out_buf
    add x21, x21, :lo12:out_buf

    adrp x22, b64_table
    add x22, x22, :lo12:b64_table

    mov w23, #0

encode_loop:
    sub x0, x19, x20
    cmp x0, #3
    blt handle_remainder

    ldrb w0, [x20], #1
    ldrb w1, [x20], #1
    ldrb w2, [x20], #1

    lsl w0, w0, #16
    orr w3, w0, w1, lsl #8
    orr w3, w3, w2

    ubfx w4, w3, #18, #6
    ubfx w5, w3, #12, #6
    ubfx w6, w3, #6, #6
    and w7, w3, #0x3f

    ldrb w4, [x22, x4]
    ldrb w5, [x22, x5]
    ldrb w6, [x22, x6]
    ldrb w7, [x22, x7]

    orr w4, w4, w5, lsl #8
    orr w6, w6, w7, lsl #8
    orr w8, w4, w6, lsl #16
    str w8, [x21], #4

    add w23, w23, #4
    cmp w23, #76
    bne encode_loop

    mov w8, #10
    strb w8, [x21], #1
    mov w23, #0
    b encode_loop

handle_remainder:
    sub x0, x19, x20
    cbz x0, encoding_done
    cmp x0, #1
    bne rem_two

rem_one:
    ldrb w0, [x20]
    lsl w3, w0, #16
    ubfx w4, w3, #18, #6
    ubfx w5, w3, #12, #6
    ldrb w4, [x22, x4]
    ldrb w5, [x22, x5]
    mov w6, #61
    mov w7, #61
    orr w4, w4, w5, lsl #8
    orr w6, w6, w7, lsl #8
    orr w8, w4, w6, lsl #16
    str w8, [x21], #4
    add w23, w23, #4
    b encoding_done

rem_two:
    ldrb w0, [x20], #1
    ldrb w1, [x20]
    lsl w0, w0, #16
    orr w3, w0, w1, lsl #8
    ubfx w4, w3, #18, #6
    ubfx w5, w3, #12, #6
    ubfx w6, w3, #6, #6
    ldrb w4, [x22, x4]
    ldrb w5, [x22, x5]
    ldrb w6, [x22, x6]
    mov w7, #61
    orr w4, w4, w5, lsl #8
    orr w6, w6, w7, lsl #8
    orr w8, w4, w6, lsl #16
    str w8, [x21], #4
    add w23, w23, #4

encoding_done:
    cbz w23, flush_output
    mov w8, #10
    strb w8, [x21], #1

flush_output:
    adrp x20, out_buf
    add x20, x20, :lo12:out_buf
    sub x22, x21, x20
    mov x21, #0

write_loop:
    cmp x21, x22
    bge exit_zero

    mov x0, #1
    add x1, x20, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0

    cmp x0, #0
    ble exit_zero

    add x21, x21, x0
    b write_loop

exit_zero:
    mov x0, #0
    mov x8, #93
    svc #0
