.globl _start

.section .bss
.align 16
in_buf:
    .space 65536
out_buf:
    .space 128

.text
_start:
    xor r12, r12            # lines = 0
    xor r13, r13            # words = 0
    xor r14, r14            # bytes = 0
    xor r15d, r15d          # in_word = 0

read_loop:
    xor eax, eax            # sys_read = 0
    xor edi, edi            # fd = 0 (stdin)
    lea rsi, [rip + in_buf]
    mov edx, 65536
    syscall

    test rax, rax
    jle done_reading

    add r14, rax            # bytes += bytes_read
    mov rbx, rax            # rbx = byte counter
    lea rsi, [rip + in_buf]

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10              # newline (0x0A)
    jne 1f
    inc r12                 # lines++
    xor r15d, r15d          # in_word = 0
    jmp 3f

1:
    cmp al, 32              # space (0x20)
    je 2f
    lea ecx, [rax - 9]
    cmp ecx, 4              # tab (9), vertical tab (11), form feed (12), carriage return (13)
    jbe 2f

    # Non-whitespace (word byte):
    test r15d, r15d
    jnz 3f
    inc r13                 # words++
    mov r15d, 1             # in_word = 1
    jmp 3f

2:
    # Whitespace byte:
    xor r15d, r15d          # in_word = 0

3:
    dec rbx
    jnz process_loop
    jmp read_loop

done_reading:
    lea rsi, [rip + out_buf]

    mov rdi, r12            # format lines
    call format_u64

    mov byte ptr [rsi], 32  # ' '
    inc rsi

    mov rdi, r13            # format words
    call format_u64

    mov byte ptr [rsi], 32  # ' '
    inc rsi

    mov rdi, r14            # format bytes
    call format_u64

    mov byte ptr [rsi], 10  # '\n'
    inc rsi

    # Calculate output length
    lea rdx, [rip + out_buf]
    mov rbx, rsi
    sub rbx, rdx            # rbx = total bytes to write
    lea rsi, [rip + out_buf]

write_loop:
    test rbx, rbx
    jle exit_program
    mov eax, 1              # sys_write = 1
    mov edi, 1              # fd = 1 (stdout)
    mov rdx, rbx
    syscall

    test rax, rax
    jle exit_program
    add rsi, rax
    sub rbx, rax
    jmp write_loop

exit_program:
    mov eax, 60             # sys_exit = 60
    xor edi, edi            # exit status = 0
    syscall

format_u64:
    test rdi, rdi
    jnz 1f
    mov byte ptr [rsi], 48  # '0'
    inc rsi
    ret

1:
    sub rsp, 24
    lea r8, [rsp + 24]
    mov rax, rdi
    mov rcx, 10

2:
    xor edx, edx
    div rcx
    add dl, 48              # convert remainder to ASCII digit
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b

    lea rdx, [rsp + 24]

3:
    mov al, byte ptr [r8]
    mov byte ptr [rsi], al
    inc rsi
    inc r8
    cmp r8, rdx
    jb 3b

    add rsp, 24
    ret
