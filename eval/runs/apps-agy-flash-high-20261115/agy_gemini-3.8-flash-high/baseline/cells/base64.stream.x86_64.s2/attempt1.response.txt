.globl _start

.section .rodata
.balign 16
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
in_buf:
    .zero 65536
out_buf:
    .zero 131072

.text
_start:
    lea r12, [rip + in_buf]
    mov r13d, 65536

.Lread_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov edx, r13d
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    sub r13d, eax
    jnz .Lread_loop

.Lread_done:
    lea rbx, [rip + in_buf]
    cmp r12, rbx
    je .Lexit_success

    mov rsi, rbx
    mov r14, r12
    lea rdi, [rip + out_buf]
    lea r10, [rip + b64_table]
    xor ebp, ebp

.Ltriplet_loop:
    lea rax, [rsi + 3]
    cmp rax, r14
    ja .Ltriplet_done

    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 3], dl

    add rdi, 4

    add ebp, 4
    cmp ebp, 76
    jne .Ltriplet_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor ebp, ebp
    jmp .Ltriplet_loop

.Ltriplet_done:
    mov rax, r14
    sub rax, rsi
    jz .Lrem_zero
    cmp rax, 1
    je .Lrem_one

.Lrem_two:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 61
    mov byte ptr [rdi + 4], 10
    add rdi, 5
    jmp .Lwrite_output

.Lrem_one:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    mov byte ptr [rdi + 4], 10
    add rdi, 5
    jmp .Lwrite_output

.Lrem_zero:
    test ebp, ebp
    jz .Lwrite_output
    mov byte ptr [rdi], 10
    inc rdi

.Lwrite_output:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi

.Lwrite_loop:
    test rdx, rdx
    jz .Lexit_success
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lexit_success
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite_loop

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall
