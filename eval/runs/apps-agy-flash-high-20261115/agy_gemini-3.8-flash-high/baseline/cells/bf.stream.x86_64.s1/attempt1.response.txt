.text
.globl _start

_start:
    # Initialize jump table in .bss
    lea r9, [rip + jmp_table]
    lea r10, [rip + match_table]
    lea r12, [rip + clean_prog]
    lea r8, [rip + bracket_stack]

    mov rdi, r9
    lea rax, [rip + op_end]
    mov ecx, 256
    rep stosq

    lea rax, [rip + op_plus]
    mov qword ptr [r9 + 43*8], rax
    lea rax, [rip + op_minus]
    mov qword ptr [r9 + 45*8], rax
    lea rax, [rip + op_right]
    mov qword ptr [r9 + 62*8], rax
    lea rax, [rip + op_left]
    mov qword ptr [r9 + 60*8], rax
    lea rax, [rip + op_dot]
    mov qword ptr [r9 + 46*8], rax
    lea rax, [rip + op_comma]
    mov qword ptr [r9 + 44*8], rax
    lea rax, [rip + op_open]
    mov qword ptr [r9 + 91*8], rax
    lea rax, [rip + op_close]
    mov qword ptr [r9 + 93*8], rax

    # Read all of standard input into stdin_buf
    lea rbp, [rip + stdin_buf]
    xor r14d, r14d

.Lread_loop:
    mov edx, 65536
    sub edx, r14d
    jle .Lread_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rbp + r14]
    syscall
    test rax, rax
    jle .Lread_done
    add r14, rax
    cmp r14, 65536
    jb .Lread_loop

.Lread_done:
    # Find first '!' in stdin_buf
    xor ecx, ecx

.Lfind_excl:
    cmp rcx, r14
    jae .Lno_excl
    cmp byte ptr [rbp + rcx], 0x21
    je .Lfound_excl
    inc rcx
    jmp .Lfind_excl

.Lno_excl:
    mov rbx, r14
    lea r13, [rbp + r14]
    mov r14, r13
    jmp .Lsplit_done

.Lfound_excl:
    mov rbx, rcx
    lea r13, [rbp + rcx + 1]
    lea r14, [rbp + r14]

.Lsplit_done:
    # Parse program bytes (0 .. rbx - 1)
    xor esi, esi
    xor edi, edi
    xor ecx, ecx

.Lparse_loop:
    cmp rsi, rbx
    jae .Lparse_done
    movzx eax, byte ptr [rbp + rsi]
    inc rsi

    lea edx, [rax - 43]
    cmp edx, 4
    jb .Lis_bf_cmd
    cmp al, '<'
    je .Lis_bf_cmd
    cmp al, '>'
    je .Lis_bf_cmd
    cmp al, '['
    je .Lis_bf_cmd
    cmp al, ']'
    je .Lis_bf_cmd
    jmp .Lparse_loop

.Lis_bf_cmd:
    mov byte ptr [r12 + rdi], al
    cmp al, '['
    je .Lparse_open
    cmp al, ']'
    je .Lparse_close
    inc rdi
    jmp .Lparse_loop

.Lparse_open:
    cmp rcx, 1000
    jae .Lbracket_error
    mov qword ptr [r8 + rcx*8], rdi
    inc rcx
    inc rdi
    jmp .Lparse_loop

.Lparse_close:
    test rcx, rcx
    jz .Lbracket_error
    dec rcx
    mov rdx, qword ptr [r8 + rcx*8]
    lea rax, [r12 + rdi + 1]
    mov qword ptr [r10 + rdx*8], rax
    lea rax, [r12 + rdx + 1]
    mov qword ptr [r10 + rdi*8], rax
    inc rdi
    jmp .Lparse_loop

.Lparse_done:
    test rcx, rcx
    jnz .Lbracket_error

    mov byte ptr [r12 + rdi], 0

    # Initialize interpreter state
    mov rbp, r12
    lea rbx, [rip + tape]
    lea r15, [rip + out_buf]

    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

.Lbracket_error:
    mov eax, 60
    mov edi, 1
    syscall

op_plus:
    inc byte ptr [rbx]
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_minus:
    dec byte ptr [rbx]
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_right:
    inc rbx
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_left:
    dec rbx
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [r15], al
    inc r15
    lea rax, [rip + out_buf + 65536]
    cmp r15, rax
    jb 1f
    call flush_out
1:
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_comma:
    xor eax, eax
    cmp r13, r14
    jae 1f
    mov al, byte ptr [r13]
    inc r13
1:
    mov byte ptr [rbx], al
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_open:
    cmp byte ptr [rbx], 0
    jne 1f
    mov rdx, rbp
    sub rdx, r12
    mov rbp, qword ptr [r10 + rdx*8]
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]
1:
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_close:
    cmp byte ptr [rbx], 0
    je 1f
    mov rdx, rbp
    sub rdx, r12
    mov rbp, qword ptr [r10 + rdx*8]
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]
1:
    inc rbp
    movzx eax, byte ptr [rbp]
    jmp qword ptr [r9 + rax*8]

op_end:
    call flush_out
    mov eax, 60
    xor edi, edi
    syscall

flush_out:
    lea rsi, [rip + out_buf]
    mov rdx, r15
    sub rdx, rsi
    test rdx, rdx
    jle .Lflush_done

.Lwrite_loop:
    push rsi
    push rdx
    mov eax, 1
    mov edi, 1
    syscall
    pop rdx
    pop rsi
    test rax, rax
    jle .Lflush_done
    add rsi, rax
    sub rdx, rax
    jnz .Lwrite_loop

.Lflush_done:
    lea r15, [rip + out_buf]
    ret

.bss
.align 16
stdin_buf:
    .zero 65536
clean_prog:
    .zero 65552
.align 16
match_table:
    .zero 524288
.align 16
jmp_table:
    .zero 2048
.align 16
tape:
    .zero 30016
.align 16
out_buf:
    .zero 65536
.align 16
bracket_stack:
    .zero 8192
