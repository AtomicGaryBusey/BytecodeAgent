.globl _start

.section .rodata
.balign 16
hex_digits:
    .ascii "0123456789abcdef"

.section .text
_start:
    // Initialize hex table pointer
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits

    // Initialize output buffer pointer
    adrp x19, out_buf
    add x19, x19, :lo12:out_buf

    // out_limit = out_buf + 65408 (threshold to flush before 128-byte row)
    adrp x28, out_buf
    add x28, x28, :lo12:out_buf
    movz x0, #0xff80
    add x28, x28, x0

    // Initialize state
    mov x20, #0                 // total_offset = 0
    mov x22, #0                 // in_avail = 0
    adrp x21, in_buf
    add x21, x21, :lo12:in_buf  // in_pos = in_buf

.Lmain_loop:
    cmp x22, #16
    b.lt .Lrefill

    // Format one full 16-byte row
    cmp x19, x28
    b.lo 1f
    bl flush_out
1:
    mov x1, x21                 // in_row ptr
    mov x2, #16                 // n = 16
    bl format_row
    add x20, x20, #16
    add x21, x21, #16
    sub x22, x22, #16
    b .Lmain_loop

.Lrefill:
    // Check if leftover bytes need shifting to beginning of in_buf
    adrp x0, in_buf
    add x0, x0, :lo12:in_buf
    cmp x21, x0
    b.eq .Lno_shift
    cbz x22, .Lno_shift

    mov x1, x21                 // src = in_pos
    mov x2, x22                 // count = in_avail
.Lshift_loop:
    ldrb w3, [x1], #1
    strb w3, [x0], #1
    subs x2, x2, #1
    b.ne .Lshift_loop

.Lno_shift:
    adrp x21, in_buf
    add x21, x21, :lo12:in_buf  // in_pos = in_buf

    // Read into in_buf + in_avail
    mov x0, #0                  // stdin fd = 0
    add x1, x21, x22            // buf = in_buf + in_avail
    movz x2, #1, lsl #16        // 65536
    sub x2, x2, x22             // count = 65536 - in_avail
    mov x8, #63                 // sys_read
    svc #0
    cmp x0, #0
    ble .Leof                   // 0 = EOF, < 0 = error
    add x22, x22, x0            // in_avail += read_bytes
    b .Lmain_loop

.Leof:
    // If leftover bytes remain, format the final short row
    cbz x22, .Lfinal_off
    cmp x19, x28
    b.lo 1f
    bl flush_out
1:
    mov x1, x21                 // in_row ptr
    mov x2, x22                 // n = in_avail
    bl format_row
    add x20, x20, x22
    mov x22, #0

.Lfinal_off:
    // Flush if needed, then write final offset line
    cmp x19, x28
    b.lo 1f
    bl flush_out
1:
    mov x3, #28
.Lfinal_off_loop:
    lsr x4, x20, x3
    and x4, x4, #0xf
    ldrb w5, [x23, x4]
    strb w5, [x19], #1
    subs x3, x3, #4
    b.ge .Lfinal_off_loop

    mov w5, #10                 // '\n'
    strb w5, [x19], #1

    bl flush_out

    // Exit cleanly
    mov x0, #0
    mov x8, #93                 // sys_exit
    svc #0

// format_row: format row of n bytes
// Inputs:
//   x1: pointer to row bytes
//   x2: n (1..16)
//   x20: current offset
//   x19: out_ptr (updated)
//   x23: hex table pointer
format_row:
    // 1. Offset as 8 lowercase hex digits, then two spaces
    mov x3, #28
.Lfmt_off_loop:
    lsr x4, x20, x3
    and x4, x4, #0xf
    ldrb w5, [x23, x4]
    strb w5, [x19], #1
    subs x3, x3, #4
    b.ge .Lfmt_off_loop

    mov w5, #32                 // ' '
    strb w5, [x19], #1
    strb w5, [x19], #1

    // 2. 16 hex byte slots
    mov x3, #0                  // i = 0
.Lfmt_hex_loop:
    cmp x3, x2
    b.ge .Lfmt_hex_pad

    // i < n: 2 hex digits + 1 space
    ldrb w4, [x1, x3]
    lsr w5, w4, #4
    and w5, w5, #0xf
    ldrb w5, [x23, x5]
    strb w5, [x19], #1
    and w5, w4, #0xf
    ldrb w5, [x23, x5]
    strb w5, [x19], #1
    mov w5, #32                 // ' '
    strb w5, [x19], #1
    b .Lfmt_hex_slot_done

.Lfmt_hex_pad:
    // i >= n: 3 spaces
    mov w5, #32                 // ' '
    strb w5, [x19], #1
    strb w5, [x19], #1
    strb w5, [x19], #1

.Lfmt_hex_slot_done:
    cmp x3, #7
    b.ne .Lfmt_no_extra_space
    mov w5, #32                 // extra space after slot 7
    strb w5, [x19], #1

.Lfmt_no_extra_space:
    add x3, x3, #1
    cmp x3, #16
    b.lt .Lfmt_hex_loop

    // 3. Space, pipe, characters, pipe, newline
    mov w5, #32                 // ' '
    strb w5, [x19], #1
    mov w5, #124                // '|'
    strb w5, [x19], #1

    mov x3, #0                  // j = 0
.Lfmt_ascii_loop:
    cmp x3, x2
    b.ge .Lfmt_ascii_done
    ldrb w4, [x1, x3]
    sub w5, w4, #32
    cmp w5, #94                 // 0x7e - 0x20 = 94
    mov w6, #46                 // '.'
    csel w5, w4, w6, ls
    strb w5, [x19], #1
    add x3, x3, #1
    b .Lfmt_ascii_loop

.Lfmt_ascii_done:
    mov w5, #124                // '|'
    strb w5, [x19], #1
    mov w5, #10                 // '\n'
    strb w5, [x19], #1
    ret

// flush_out: write accumulated bytes in out_buf to stdout
flush_out:
    adrp x4, out_buf
    add x4, x4, :lo12:out_buf
    sub x2, x19, x4             // count = out_ptr - out_buf
    cbz x2, .Lflush_done
    mov x1, x4                  // buf = out_buf
.Lflush_write:
    mov x0, #1                  // stdout fd = 1
    mov x8, #64                 // sys_write
    svc #0
    cmp x0, #0
    ble .Lflush_reset
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, .Lflush_write
.Lflush_reset:
    adrp x19, out_buf
    add x19, x19, :lo12:out_buf
.Lflush_done:
    ret

.section .bss
.balign 16
in_buf:
    .space 65536
out_buf:
    .space 65536
