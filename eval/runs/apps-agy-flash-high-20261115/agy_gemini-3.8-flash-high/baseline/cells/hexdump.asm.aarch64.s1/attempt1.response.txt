.globl _start

.text
_start:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x23, x24

    adrp x25, in_buf
    add x25, x25, :lo12:in_buf
    mov x19, x25
    mov x20, x25

    mov x21, #0
    mov x22, #0

    adrp x26, hex_digits
    add x26, x26, :lo12:hex_digits

check_refill:
    sub x0, x20, x19
    cmp x0, #16
    b.ge ready_to_process

    cbnz x22, ready_to_process

    cbz x0, refill_read
    cmp x19, x25
    b.eq refill_read

    mov x1, x19
    mov x2, x25
    mov x3, x0
copy_rem_loop:
    ldrb w4, [x1], #1
    strb w4, [x2], #1
    subs x3, x3, #1
    b.ne copy_rem_loop

refill_read:
    mov x19, x25
    add x20, x25, x0

    sub x2, x20, x25
    mov x3, #16384
    sub x2, x3, x2
    mov x1, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le refill_eof
    add x20, x20, x0
    b check_refill

refill_eof:
    mov x22, #1
    b check_refill

ready_to_process:
    sub x0, x20, x19
    cbz x0, done_all_rows

    cmp x0, #16
    mov x27, #16
    csel x27, x0, x27, lo

    sub x1, x23, x24
    cmp x1, #2048
    b.lo skip_row_flush
    bl flush_out
skip_row_flush:

    // 1. Format offset (x21) as 8 lowercase hex digits + two spaces
    mov x0, x21
    mov x1, #28
format_off_loop:
    lsrv x2, x0, x1
    and x2, x2, #0xf
    ldrb w3, [x26, x2]
    strb w3, [x23], #1
    subs x1, x1, #4
    b.ge format_off_loop

    mov w3, #32
    strb w3, [x23], #1
    strb w3, [x23], #1

    // 2. Format 16 hex byte positions
    mov x28, #0
hex_pos_loop:
    cmp x28, x27
    b.hs hex_pos_empty

    ldrb w0, [x19, x28]
    lsr w1, w0, #4
    ldrb w3, [x26, x1]
    and w2, w0, #0xf
    ldrb w4, [x26, x2]
    strb w3, [x23], #1
    strb w4, [x23], #1
    mov w5, #32
    strb w5, [x23], #1
    b hex_pos_check_sep

hex_pos_empty:
    mov w3, #32
    strb w3, [x23], #1
    strb w3, [x23], #1
    strb w3, [x23], #1

hex_pos_check_sep:
    cmp x28, #7
    b.ne hex_pos_next
    mov w3, #32
    strb w3, [x23], #1

hex_pos_next:
    add x28, x28, #1
    cmp x28, #16
    b.ne hex_pos_loop

    // 3. ASCII representation
    mov w3, #32
    strb w3, [x23], #1
    mov w3, #124
    strb w3, [x23], #1

    mov x28, #0
    mov w2, #'.'
ascii_loop:
    ldrb w0, [x19, x28]
    sub w1, w0, #0x20
    cmp w1, #0x5e
    csel w0, w0, w2, ls
    strb w0, [x23], #1
    add x28, x28, #1
    cmp x28, x27
    b.ne ascii_loop

    mov w3, #124
    strb w3, [x23], #1
    mov w3, #10
    strb w3, [x23], #1

    add x19, x19, x27
    add x21, x21, x27
    b check_refill

done_all_rows:
    mov x0, x21
    mov x1, #28
format_len_loop:
    lsrv x2, x0, x1
    and x2, x2, #0xf
    ldrb w3, [x26, x2]
    strb w3, [x23], #1
    subs x1, x1, #4
    b.ge format_len_loop

    mov w3, #10
    strb w3, [x23], #1

    bl flush_out

    mov x0, #0
    mov x8, #93
    svc #0

flush_out:
    subs x2, x23, x24
    b.eq flush_out_ret
    mov x1, x24
flush_out_loop:
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le flush_out_err
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, flush_out_loop
flush_out_err:
    mov x23, x24
flush_out_ret:
    ret

.section .rodata
.balign 16
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
in_buf:
    .space 16384
out_buf:
    .space 16384
