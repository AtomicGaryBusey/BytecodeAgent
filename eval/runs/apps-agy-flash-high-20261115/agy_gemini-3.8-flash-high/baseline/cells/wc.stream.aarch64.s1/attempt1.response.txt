// wc: count lines, words, and bytes from standard input

.globl _start

.text
_start:
    // Initialize counters
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    movz x2, #1, lsl #16
    mov x8, #63
    svc #0

    cmp x0, #0
    b.le done_reading

    add x21, x21, x0

    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    add x3, x1, x0

byte_loop:
    ldrb w4, [x1], #1

    // Count newline (0x0A = 10)
    cmp w4, #10
    cinc x19, x19, eq

    // Check whitespace: space (32) or 9..13 (tab, newline, VT, FF, CR)
    cmp w4, #32
    b.eq is_ws
    sub w5, w4, #9
    cmp w5, #4
    b.ls is_ws

    // Non-whitespace byte
    cbz x22, new_word
    cmp x1, x3
    b.ne byte_loop
    b read_loop

new_word:
    mov x22, #1
    add x20, x20, #1
    cmp x1, x3
    b.ne byte_loop
    b read_loop

is_ws:
    mov x22, #0
    cmp x1, x3
    b.ne byte_loop
    b read_loop

done_reading:
    // Format lines, words, bytes into out_buf
    sub sp, sp, #64
    stp x19, x20, [sp]
    str x21, [sp, #16]

    mov x23, sp
    add x24, sp, #24

    adrp x25, out_buf
    add x25, x25, :lo12:out_buf

num_loop:
    ldr x0, [x23], #8

    cbnz x0, has_digits
    mov w5, #48
    strb w5, [x25], #1
    b append_sep

has_digits:
    add x2, sp, #64
    mov x3, #10
div_loop:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x4
    cbnz x0, div_loop

    add x6, sp, #64
copy_loop:
    ldrb w5, [x2], #1
    strb w5, [x25], #1
    cmp x2, x6
    b.ne copy_loop

append_sep:
    cmp x23, x24
    b.eq last_sep
    mov w5, #32
    strb w5, [x25], #1
    b num_loop

last_sep:
    mov w5, #10
    strb w5, [x25], #1

    add sp, sp, #64

    // Write formatted output from out_buf to stdout
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    sub x2, x25, x1

write_loop:
    cbz x2, exit_prog
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit_prog
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

exit_prog:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_prog

.bss
.balign 16
in_buf:
    .zero 65536

out_buf:
    .zero 128
