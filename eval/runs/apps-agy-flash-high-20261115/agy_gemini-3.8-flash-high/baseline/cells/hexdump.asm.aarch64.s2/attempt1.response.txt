.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.equ IN_BUF_SIZE, 16384
.equ OUT_BUF_SIZE, 65536
.equ FLUSH_THRESH, 60000

.section .rodata
.balign 8
hex_digits:
    .ascii "0123456789abcdef"

.text
.globl _start

_start:
    mov x19, #0             // in_head = 0
    mov x20, #0             // in_tail = 0
    mov x21, #0             // eof_flag = 0
    mov x22, #0             // total_offset = 0

    adrp x24, hex_digits
    add x24, x24, :lo12:hex_digits

    adrp x25, in_buf
    add x25, x25, :lo12:in_buf

    adrp x26, out_buf
    add x26, x26, :lo12:out_buf
    mov x23, x26            // x_out = out_buf

check_avail:
    sub x0, x20, x19        // avail = in_tail - in_head
    cmp x0, #16
    b.ge have_full_row

    cbnz x21, check_eof_avail

refill_input:
    cbz x19, 1f
    cbz x0, 2f

    mov x1, x25
    add x2, x25, x19
    mov x3, x0
copy_loop:
    ldrb w4, [x2], #1
    strb w4, [x1], #1
    subs x3, x3, #1
    b.ne copy_loop

2:
    mov x19, #0
    mov x20, x0

1:
    mov x0, #0
    add x1, x25, x20
    movz w2, #IN_BUF_SIZE
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le 3f
    add x20, x20, x0
    b check_avail

3:
    mov x21, #1
    b check_avail

check_eof_avail:
    cbz x0, done_rows
    mov x10, x0
    b format_row

have_full_row:
    mov x10, #16

format_row:
    sub x1, x23, x26
    movz w2, #FLUSH_THRESH
    cmp w1, w2
    b.lo 1f
    bl flush_out_buf

1:
    bl format_hex8

    mov w12, #' '
    strb w12, [x23], #1
    strb w12, [x23], #1

    mov x11, #0
hex_loop:
    cmp x11, x10
    b.ge hex_pad

    add x13, x25, x19
    ldrb w12, [x13, x11]

    lsr w13, w12, #4
    ldrb w13, [x24, x13]
    strb w13, [x23], #1

    and w13, w12, #0xf
    ldrb w13, [x24, x13]
    strb w13, [x23], #1

    mov w13, #' '
    strb w13, [x23], #1
    b hex_slot_done

hex_pad:
    mov w13, #' '
    strb w13, [x23], #1
    strb w13, [x23], #1
    strb w13, [x23], #1

hex_slot_done:
    cmp x11, #7
    b.ne 2f
    mov w13, #' '
    strb w13, [x23], #1

2:
    add x11, x11, #1
    cmp x11, #16
    b.ne hex_loop

    mov w13, #' '
    strb w13, [x23], #1
    mov w13, #'|'
    strb w13, [x23], #1

    cbz x10, ascii_done
    mov x11, #0

ascii_loop:
    add x13, x25, x19
    ldrb w12, [x13, x11]

    sub w14, w12, #0x20
    cmp w14, #0x5e
    b.ls 3f
    mov w12, #'.'

3:
    strb w12, [x23], #1

    add x11, x11, #1
    cmp x11, x10
    b.ne ascii_loop

ascii_done:
    mov w13, #'|'
    strb w13, [x23], #1
    mov w13, #'\n'
    strb w13, [x23], #1

    add x19, x19, x10
    add x22, x22, x10
    b check_avail

done_rows:
    bl format_hex8

    mov w12, #'\n'
    strb w12, [x23], #1

    bl flush_out_buf

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

format_hex8:
    ubfx w12, w22, #28, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    ubfx w12, w22, #24, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    ubfx w12, w22, #20, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    ubfx w12, w22, #16, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    ubfx w12, w22, #12, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    ubfx w12, w22, #8, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    ubfx w12, w22, #4, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1

    and w12, w22, #0xf
    ldrb w12, [x24, x12]
    strb w12, [x23], #1
    ret

flush_out_buf:
    mov x1, x26
    sub x2, x23, x1
    cbz x2, 2f

1:
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 2f
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, 1b
    mov x23, x26

2:
    ret

.bss
.balign 16
in_buf:
    .zero IN_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE
