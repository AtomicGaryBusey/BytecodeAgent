.globl _start

.text
_start:
    // Read all standard input into in_buf
    adrp    x19, in_buf
    add     x19, x19, :lo12:in_buf
    mov     x20, #0
    movz    x21, #2, lsl #16        // capacity: 131072 bytes

.Lread_loop:
    subs    x2, x21, x20
    b.le    .Lread_done
    mov     x0, #0                  // fd = 0 (stdin)
    add     x1, x19, x20
    mov     x8, #63                 // sys_read
    svc     #0
    cmp     x0, #0
    ble     .Lread_done
    add     x20, x20, x0
    b       .Lread_loop

.Lread_done:
    // If no bytes read, exit immediately
    cbz     x20, .Lexit

    // Set up output buffer and scan indices
    adrp    x23, out_buf
    add     x23, x23, :lo12:out_buf
    mov     x24, x23                // out_ptr
    mov     x21, #0                 // scan index
    mov     x22, #0                 // line_start index

.Lscan_loop:
    cmp     x21, x20
    b.ge    .Lscan_done
    ldrb    w25, [x19, x21]
    cmp     w25, #10                // newline (0x0A)
    b.ne    .Lnext_char

    // Found newline: reverse in_buf[x22 .. x21 - 1]
    sub     x25, x21, x22           // line length
    cbz     x25, .Lappend_nl

    add     x26, x19, x21           // pointer at newline
.Lrev_loop:
    ldrb    w27, [x26, #-1]!
    strb    w27, [x24], #1
    subs    x25, x25, #1
    b.ne    .Lrev_loop

.Lappend_nl:
    mov     w27, #10
    strb    w27, [x24], #1
    add     x22, x21, #1            // line_start = x21 + 1

.Lnext_char:
    add     x21, x21, #1
    b       .Lscan_loop

.Lscan_done:
    // Check for trailing line without newline
    cmp     x22, x20
    b.ge    .Lwrite_output

    sub     x25, x20, x22
    add     x26, x19, x20
.Lrev_tail_loop:
    ldrb    w27, [x26, #-1]!
    strb    w27, [x24], #1
    subs    x25, x25, #1
    b.ne    .Lrev_tail_loop

    mov     w27, #10
    strb    w27, [x24], #1

.Lwrite_output:
    sub     x20, x24, x23           // total bytes to write
    mov     x19, x23                // write buffer pointer

.Lwrite_loop:
    cbz     x20, .Lexit
    mov     x0, #1                  // fd = 1 (stdout)
    mov     x1, x19
    mov     x2, x20
    mov     x8, #64                 // sys_write
    svc     #0
    cmp     x0, #0
    ble     .Lexit
    add     x19, x19, x0
    sub     x20, x20, x0
    b       .Lwrite_loop

.Lexit:
    mov     x0, #0                  // exit status 0
    mov     x8, #93                 // sys_exit
    svc     #0

    .bss
    .balign 16
in_buf:
    .zero   131072
    .balign 16
out_buf:
    .zero   131072
