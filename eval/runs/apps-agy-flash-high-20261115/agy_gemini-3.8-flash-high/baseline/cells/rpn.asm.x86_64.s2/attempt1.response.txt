.globl _start

.section .text

_start:
    xor r13d, r13d
.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [in_buf + r13]
    mov edx, 131072
    sub edx, r13d
    jle .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r13, rax
    jmp .Lread_loop

.Lread_done:
    test r13, r13
    jz .Lexit_success

    xor r14d, r14d
    xor r12d, r12d

.Lline_loop:
    cmp r12, r13
    jae .Lall_done

    mov r8, r12
.Lscan_nl:
    cmp r12, r13
    jae .Lline_no_nl
    movzx eax, byte ptr [in_buf + r12]
    inc r12
    cmp al, 10
    je .Lline_has_nl
    jmp .Lscan_nl

.Lline_has_nl:
    lea r9, [r12 - 1]
    jmp .Lstart_line

.Lline_no_nl:
    mov r9, r13

.Lstart_line:
    xor ebp, ebp

.Ltoken_loop:
.Lskip_sep:
    cmp r8, r9
    jae .Ltokens_done
    movzx eax, byte ptr [in_buf + r8]
    cmp al, ' '
    je .Lis_sep
    cmp al, 9
    je .Lis_sep
    jmp .Ltoken_start
.Lis_sep:
    inc r8
    jmp .Lskip_sep

.Ltoken_start:
    mov rbx, r8
.Lscan_tok:
    inc r8
    cmp r8, r9
    jae .Ltok_end_found
    movzx eax, byte ptr [in_buf + r8]
    cmp al, ' '
    je .Ltok_end_found
    cmp al, 9
    je .Ltok_end_found
    jmp .Lscan_tok

.Ltok_end_found:
    mov rcx, r8
    sub rcx, rbx
    cmp rcx, 1
    jne .Ltok_len_gt_1

    movzx eax, byte ptr [in_buf + rbx]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod

    cmp al, '0'
    jb .Lline_error
    cmp al, '9'
    ja .Lline_error
    sub eax, '0'
    cdqe
    jmp .Lpush_val

.Ltok_len_gt_1:
    mov rsi, rbx
    xor edi, edi
    movzx eax, byte ptr [in_buf + rsi]
    cmp al, '-'
    jne .Lnot_neg
    mov edi, 1
    inc rsi
.Lnot_neg:
    mov rdx, r8
    sub rdx, rsi
    cmp rdx, 1
    jl .Lline_error
    cmp rdx, 18
    jg .Lline_error

    mov rcx, rdx
    mov r11, rsi
.Lcheck_digits:
    movzx eax, byte ptr [in_buf + r11]
    cmp al, '0'
    jb .Lline_error
    cmp al, '9'
    ja .Lline_error
    inc r11
    dec rcx
    jnz .Lcheck_digits

    xor eax, eax
    mov rcx, rdx
.Lparse_digits:
    movzx edx, byte ptr [in_buf + rsi]
    sub edx, '0'
    lea rax, [rax + rax * 4]
    lea rax, [rdx + rax * 2]
    inc rsi
    dec rcx
    jnz .Lparse_digits

    test edi, edi
    jz .Lpush_val
    neg rax

.Lpush_val:
    mov qword ptr [rpn_stack + rbp*8], rax
    inc ebp
    jmp .Ltoken_loop

.Lop_add:
    cmp ebp, 2
    jl .Lline_error
    sub ebp, 2
    mov rax, qword ptr [rpn_stack + rbp*8]
    mov rbx, qword ptr [rpn_stack + rbp*8 + 8]
    add rax, rbx
    mov qword ptr [rpn_stack + rbp*8], rax
    inc ebp
    jmp .Ltoken_loop

.Lop_sub:
    cmp ebp, 2
    jl .Lline_error
    sub ebp, 2
    mov rax, qword ptr [rpn_stack + rbp*8]
    mov rbx, qword ptr [rpn_stack + rbp*8 + 8]
    sub rax, rbx
    mov qword ptr [rpn_stack + rbp*8], rax
    inc ebp
    jmp .Ltoken_loop

.Lop_mul:
    cmp ebp, 2
    jl .Lline_error
    sub ebp, 2
    mov rax, qword ptr [rpn_stack + rbp*8]
    mov rbx, qword ptr [rpn_stack + rbp*8 + 8]
    imul rax, rbx
    mov qword ptr [rpn_stack + rbp*8], rax
    inc ebp
    jmp .Ltoken_loop

.Lop_div:
    cmp ebp, 2
    jl .Lline_error
    sub ebp, 2
    mov rax, qword ptr [rpn_stack + rbp*8]
    mov rbx, qword ptr [rpn_stack + rbp*8 + 8]
    test rbx, rbx
    jz .Lline_error
    mov r10, 1
    shl r10, 63
    cmp rax, r10
    jne 1f
    cmp rbx, -1
    je .Lline_error
1:
    cqo
    idiv rbx
    mov qword ptr [rpn_stack + rbp*8], rax
    inc ebp
    jmp .Ltoken_loop

.Lop_mod:
    cmp ebp, 2
    jl .Lline_error
    sub ebp, 2
    mov rax, qword ptr [rpn_stack + rbp*8]
    mov rbx, qword ptr [rpn_stack + rbp*8 + 8]
    test rbx, rbx
    jz .Lline_error
    mov r10, 1
    shl r10, 63
    cmp rax, r10
    jne 1f
    cmp rbx, -1
    je .Lline_error
1:
    cqo
    idiv rbx
    mov qword ptr [rpn_stack + rbp*8], rdx
    inc ebp
    jmp .Ltoken_loop

.Ltokens_done:
    cmp ebp, 1
    jne .Lline_error
    mov rax, qword ptr [rpn_stack]
    call print_int64
    jmp .Lline_loop

.Lline_error:
    call print_error
    jmp .Lline_loop

.Lall_done:
    call flush_out

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall

print_error:
    mov dword ptr [out_buf + r14], 0x6f727265
    mov word ptr [out_buf + r14 + 4], 0x0a72
    add r14, 6
    cmp r14, 60000
    jb 1f
    sub rsp, 8
    call flush_out
    add rsp, 8
1:
    ret

print_int64:
    sub rsp, 40
    lea rsi, [rsp + 39]
    xor edi, edi

    test rax, rax
    jns .Lpi_pos
    mov edi, 1
    neg rax
.Lpi_pos:
    test rax, rax
    jnz .Lpi_digits
    dec rsi
    mov byte ptr [rsi], '0'
    jmp .Lpi_check_neg

.Lpi_digits:
    mov rcx, 10
.Lpi_div_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .Lpi_div_loop

.Lpi_check_neg:
    test edi, edi
    jz .Lpi_copy
    dec rsi
    mov byte ptr [rsi], '-'

.Lpi_copy:
    lea rdx, [rsp + 39]
    sub rdx, rsi

.Lpi_copy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [out_buf + r14], al
    inc rsi
    inc r14
    dec edx
    jnz .Lpi_copy_loop

    mov byte ptr [out_buf + r14], 10
    inc r14

    cmp r14, 60000
    jb 1f
    call flush_out
1:
    add rsp, 40
    ret

flush_out:
    test r14, r14
    jz .Lflush_ret
    xor r10d, r10d
.Lwrite_loop:
    mov eax, 1
    mov edi, 1
    lea rsi, [out_buf + r10]
    mov edx, r14d
    sub edx, r10d
    syscall
    test rax, rax
    jle .Lflush_ret
    add r10, rax
    cmp r10, r14
    jb .Lwrite_loop
    xor r14, r14
.Lflush_ret:
    ret

.section .bss
.align 16
in_buf:
    .space 131072
out_buf:
    .space 65536
rpn_stack:
    .space 8192
