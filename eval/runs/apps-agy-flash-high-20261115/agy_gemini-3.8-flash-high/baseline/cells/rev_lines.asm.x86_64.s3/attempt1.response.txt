.text
.globl _start

_start:
    lea r12, [rip + in_buf]
    lea r15, [rip + out_buf]
    xor r13d, r13d
    xor r14d, r14d
    xor ebx, ebx

.Lread_more:
    call flush_out

    test r14, r14
    jz .Lno_shift
    mov rcx, r13
    sub rcx, r14
    jz .Lshift_zero
    mov r13, rcx
    lea rsi, [r12 + r14]
    mov rdi, r12
    rep movsb
    xor r14d, r14d
    jmp .Lno_shift

.Lshift_zero:
    xor r13d, r13d
    xor r14d, r14d

.Lno_shift:
    mov rbp, r13

    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 262144
    sub rdx, r13
    syscall

    test rax, rax
    jle .Leof

    add r13, rax

.Lscan_loop:
    cmp rbp, r13
    jae .Lread_more
    cmp byte ptr [r12 + rbp], 10
    je .Lfound_newline
    inc rbp
    jmp .Lscan_loop

.Lfound_newline:
    mov rcx, rbp
    sub rcx, r14
    lea rax, [rbx + rcx + 1]
    cmp rax, 262144
    jbe .Lout_ok1
    call flush_out
    mov rcx, rbp
    sub rcx, r14
.Lout_ok1:
    test rcx, rcx
    jz .Lcopy_nl
    lea rsi, [r12 + rbp - 1]
    lea rdi, [r15 + rbx]
    mov rdx, rcx
.Lrev_loop1:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    dec rdx
    jnz .Lrev_loop1
    add rbx, rcx

.Lcopy_nl:
    mov byte ptr [r15 + rbx], 10
    inc rbx
    lea r14, [rbp + 1]
    mov rbp, r14
    jmp .Lscan_loop

.Leof:
    cmp r14, r13
    jae .Lexit
    mov rcx, r13
    sub rcx, r14
    lea rax, [rbx + rcx + 1]
    cmp rax, 262144
    jbe .Lout_ok2
    call flush_out
    mov rcx, r13
    sub rcx, r14
.Lout_ok2:
    lea rsi, [r12 + r13 - 1]
    lea rdi, [r15 + rbx]
    mov rdx, rcx
.Lrev_loop2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    dec rdx
    jnz .Lrev_loop2
    add rbx, rcx
    mov byte ptr [r15 + rbx], 10
    inc rbx

.Lexit:
    call flush_out
    mov eax, 60
    xor edi, edi
    syscall

flush_out:
    test rbx, rbx
    jz .Lflush_ret
    mov r8, r15
.Lflush_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, rbx
    syscall
    test rax, rax
    jle .Lflush_err
    add r8, rax
    sub rbx, rax
    jnz .Lflush_loop
.Lflush_ret:
    ret

.Lflush_err:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.balign 16
in_buf:
    .zero 262144
out_buf:
    .zero 262144
