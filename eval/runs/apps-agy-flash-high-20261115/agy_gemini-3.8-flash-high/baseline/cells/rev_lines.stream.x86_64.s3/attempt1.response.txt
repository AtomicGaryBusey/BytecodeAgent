.globl _start

.text
_start:
    xor r12, r12

.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 262144
    sub rdx, r12
    jbe .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    test r12, r12
    jz .Lexit

    lea r8, [rip + in_buf]
    lea r9, [rip + in_buf]
    add r9, r12
    lea rdi, [rip + out_buf]

.Lline_loop:
    cmp r8, r9
    jae .Lprocess_done

    mov r10, r8
.Lscan:
    cmp r10, r9
    je .Lfound_no_nl
    cmp byte ptr [r10], 10
    je .Lfound_nl
    inc r10
    jmp .Lscan

.Lfound_nl:
    lea r11, [r10 + 1]
    jmp .Lreverse_line

.Lfound_no_nl:
    mov r11, r9

.Lreverse_line:
    mov rcx, r10
    sub rcx, r8
    lea rsi, [r10 - 1]
    test rcx, rcx
    jz .Lrev_done
.Lrev_copy:
    mov al, byte ptr [rsi]
    dec rsi
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz .Lrev_copy
.Lrev_done:
    mov byte ptr [rdi], 10
    inc rdi
    mov r8, r11
    jmp .Lline_loop

.Lprocess_done:
    lea r13, [rip + out_buf]
    mov r14, rdi
    sub r14, r13
    xor ebx, ebx

.Lwrite_loop:
    cmp rbx, r14
    jae .Lexit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    syscall
    test rax, rax
    jle .Lexit
    add rbx, rax
    jmp .Lwrite_loop

.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
in_buf:
    .zero 262144
out_buf:
    .zero 266240
