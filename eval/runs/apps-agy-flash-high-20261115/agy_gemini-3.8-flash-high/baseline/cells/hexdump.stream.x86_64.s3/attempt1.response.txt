.text
.globl _start

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15d, r15d
    xor ebx, ebx

.Lmain_loop:
.Lrefill_loop:
    mov rax, r13
    sub rax, r14
    cmp rax, 16
    jae .Lrefill_done
    test r15d, r15d
    jnz .Lrefill_done

    test r14, r14
    jz .Lno_shift
    test rax, rax
    jz .Lzero_avail
    lea rsi, [rip + in_buf]
    add rsi, r14
    lea rdi, [rip + in_buf]
    mov rcx, rax
    rep movsb
.Lzero_avail:
    mov r13, rax
    xor r14, r14
.Lno_shift:

    mov eax, 0
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r13
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle .Leof_reached
    add r13, rax
    jmp .Lrefill_loop

.Leof_reached:
    mov r15d, 1

.Lrefill_done:
    mov rax, r13
    sub rax, r14
    test rax, rax
    jz .Lall_done

    mov ebp, eax
    cmp ebp, 16
    jbe .Lhave_n
    mov ebp, 16
.Lhave_n:

    cmp rbx, 65536 - 128
    jb .Lno_flush
    call flush_out
.Lno_flush:

    lea rdi, [rip + out_buf]
    add rdi, rbx
    lea r9, [rip + hex_table]

    mov eax, r12d
    mov ecx, 8
.Lhex_off:
    rol eax, 4
    mov edx, eax
    and edx, 0x0f
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz .Lhex_off

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2

    lea rsi, [rip + in_buf]
    add rsi, r14
    xor r8d, r8d
.Lslots_loop:
    cmp r8d, ebp
    jae .Lslot_pad
    movzx eax, byte ptr [rsi + r8]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    and eax, 0x0f
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp .Lslot_next
.Lslot_pad:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
.Lslot_next:
    cmp r8d, 7
    jne .Lno_extra_space
    mov byte ptr [rdi], 32
    inc rdi
.Lno_extra_space:
    inc r8d
    cmp r8d, 16
    jne .Lslots_loop

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2

    xor r8d, r8d
.Lascii_loop:
    cmp r8d, ebp
    jae .Lascii_done
    mov al, byte ptr [rsi + r8]
    mov dl, al
    sub al, 0x20
    cmp al, 0x5e
    jbe 1f
    mov dl, 46
1:
    mov byte ptr [rdi], dl
    inc rdi
    inc r8d
    jmp .Lascii_loop
.Lascii_done:

    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2

    lea rax, [rip + out_buf]
    mov rbx, rdi
    sub rbx, rax

    mov eax, ebp
    add r12, rax
    add r14, rax

    jmp .Lmain_loop

.Lall_done:
    cmp rbx, 65536 - 128
    jb 1f
    call flush_out
1:
    lea rdi, [rip + out_buf]
    add rdi, rbx
    lea r9, [rip + hex_table]

    mov eax, r12d
    mov ecx, 8
.Lhex_final:
    rol eax, 4
    mov edx, eax
    and edx, 0x0f
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz .Lhex_final

    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + out_buf]
    mov rbx, rdi
    sub rbx, rax

    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

flush_out:
    test rbx, rbx
    jz .Lflush_ret
    xor r8d, r8d
.Lflush_write:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, r8
    mov rdx, rbx
    sub rdx, r8
    syscall
    test rax, rax
    jle .Lflush_done
    add r8, rax
    cmp r8, rbx
    jb .Lflush_write
.Lflush_done:
    xor ebx, ebx
.Lflush_ret:
    ret

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.balign 16
in_buf:
    .zero 65536
out_buf:
    .zero 65536
