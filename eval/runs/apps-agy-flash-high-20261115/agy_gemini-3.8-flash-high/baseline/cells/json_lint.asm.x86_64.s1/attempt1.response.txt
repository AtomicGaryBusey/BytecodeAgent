.globl _start

.section .rodata
.align 16
err_prefix:
    .ascii "error at "
.align 16
ok_msg:
    .ascii "ok\n"
.equ ok_msg_len, 3

.section .bss
.align 16
input_buf:
    .zero 131072
out_buf:
    .zero 64

.section .text

_start:
    lea r12, [rip + input_buf]
    xor r13d, r13d

read_loop:
    mov edx, 131072
    sub edx, r13d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor ebx, ebx
    call skip_ws
    cmp rbx, r13
    jae report_error
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne report_error

    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_msg_len
    call write_all

    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
1:
    cmp rbx, r13
    jae 2f
    mov al, byte ptr [r12 + rbx]
    cmp al, ' '
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
    ret
3:
    inc rbx
    jmp 1b
2:
    ret

parse_value:
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe parse_number
    jmp report_error

parse_true:
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'r'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'u'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'e'
    jne report_error
    inc rbx
    ret

parse_false:
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'a'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'l'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 's'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'e'
    jne report_error
    inc rbx
    ret

parse_null:
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'u'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'l'
    jne report_error
    inc rbx
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], 'l'
    jne report_error
    inc rbx
    ret

parse_string:
    inc rbx
string_loop:
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '"'
    je string_done
    cmp al, '\\'
    je string_escape
    cmp al, 0x20
    jb report_error
    inc rbx
    jmp string_loop

string_done:
    inc rbx
    ret

string_escape:
    inc rbx
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 'u'
    je string_u_escape
    cmp al, '"'
    je string_esc_ok
    cmp al, '\\'
    je string_esc_ok
    cmp al, '/'
    je string_esc_ok
    cmp al, 'b'
    je string_esc_ok
    cmp al, 'f'
    je string_esc_ok
    cmp al, 'n'
    je string_esc_ok
    cmp al, 'r'
    je string_esc_ok
    cmp al, 't'
    je string_esc_ok
    jmp report_error

string_esc_ok:
    inc rbx
    jmp string_loop

string_u_escape:
    inc rbx
    mov ecx, 4
hex_loop:
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb report_error
    cmp al, '9'
    jbe hex_digit_ok
    cmp al, 'A'
    jb report_error
    cmp al, 'F'
    jbe hex_digit_ok
    cmp al, 'a'
    jb report_error
    cmp al, 'f'
    jbe hex_digit_ok
    jmp report_error

hex_digit_ok:
    inc rbx
    dec ecx
    jnz hex_loop
    jmp string_loop

parse_number:
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '-'
    jne check_int_start
    inc rbx
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]

check_int_start:
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error

    cmp al, '0'
    jne int_non_zero
    inc rbx
    jmp check_frac

int_non_zero:
    inc rbx
int_digits_loop:
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb check_frac
    cmp al, '9'
    ja check_frac
    inc rbx
    jmp int_digits_loop

check_frac:
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '.'
    jne check_exp
    inc rbx
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error
    inc rbx

frac_digits_loop:
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb check_exp
    cmp al, '9'
    ja check_exp
    inc rbx
    jmp frac_digits_loop

check_exp:
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 'e'
    je exp_found
    cmp al, 'E'
    jne num_done

exp_found:
    inc rbx
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '+'
    je exp_sign
    cmp al, '-'
    jne exp_digit_check

exp_sign:
    inc rbx
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]

exp_digit_check:
    cmp al, '0'
    jb report_error
    cmp al, '9'
    ja report_error
    inc rbx

exp_digits_loop:
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '0'
    jb num_done
    cmp al, '9'
    ja num_done
    inc rbx
    jmp exp_digits_loop

num_done:
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], ']'
    jne array_value_loop
    inc rbx
    ret

array_value_loop:
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, ']'
    je array_done
    cmp al, ','
    jne report_error
    inc rbx
    call skip_ws
    jmp array_value_loop

array_done:
    inc rbx
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], '}'
    jne object_member_loop
    inc rbx
    ret

object_member_loop:
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], '"'
    jne report_error
    call parse_string
    call skip_ws
    cmp rbx, r13
    jae report_error
    cmp byte ptr [r12 + rbx], ':'
    jne report_error
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae report_error
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '}'
    je object_done
    cmp al, ','
    jne report_error
    inc rbx
    call skip_ws
    jmp object_member_loop

object_done:
    inc rbx
    ret

report_error:
    lea rdi, [rip + out_buf]
    lea rsi, [rip + err_prefix]
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rsi + 8]
    mov byte ptr [rdi + 8], al
    add rdi, 9

    mov rax, rbx
    lea rcx, [rip + out_buf + 63]
    mov byte ptr [rcx], 0x0A
    test rax, rax
    jnz 1f
    dec rcx
    mov byte ptr [rcx], '0'
    jmp 2f
1:
    mov r8, 10
3:
    xor edx, edx
    div r8
    add dl, '0'
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz 3b
2:
    lea rdx, [rip + out_buf + 64]
    sub rdx, rcx
    mov rsi, rcx
    mov r9, rdx
4:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec r9
    jnz 4b

    add rdx, 9
    mov edi, 1
    lea rsi, [rip + out_buf]
    call write_all

    mov edi, 1
    mov eax, 60
    syscall

write_all:
write_loop:
    test rdx, rdx
    jle write_done
    mov eax, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    ret
