.text
.globl _start
_start:
    sub rsp, 16

    # Read all input from stdin (fd = 0) into in_buf
    xor r12d, r12d
.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    test r12, r12
    jz .Lexit_clean

    # Parse lines into lines array: struct { void *ptr; size_t len; }
    lea rbx, [rip + in_buf]
    lea rdi, [rip + lines]
    xor r15d, r15d
    xor ecx, ecx
    xor edx, edx

.Lparse_loop:
    cmp rcx, r12
    jae .Lparse_end_loop

    mov al, byte ptr [rbx + rcx]
    cmp al, 10
    jne .Lparse_next

    lea rax, [rbx + rdx]
    mov r8, rcx
    sub r8, rdx
    mov qword ptr [rdi], rax
    mov qword ptr [rdi + 8], r8
    add rdi, 16
    inc r15

    inc rcx
    mov rdx, rcx
    jmp .Lparse_loop

.Lparse_next:
    inc rcx
    jmp .Lparse_loop

.Lparse_end_loop:
    cmp rdx, r12
    jae .Lparse_done

    lea rax, [rbx + rdx]
    mov r8, r12
    sub r8, rdx
    mov qword ptr [rdi], rax
    mov qword ptr [rdi + 8], r8
    add rdi, 16
    inc r15

.Lparse_done:
    test r15, r15
    jz .Lexit_clean

    # Shell sort on lines
    cmp r15, 1
    jbe .Lsort_done

    lea rbx, [rip + lines]
    mov r14, r15
    shr r14, 1

.Lgap_loop:
    test r14, r14
    jz .Lsort_done

    mov r13, r14
.Li_loop:
    cmp r13, r15
    jae .Lnext_gap

    mov rax, r13
    shl rax, 4
    mov r8, qword ptr [rbx + rax]
    mov r9, qword ptr [rbx + rax + 8]
    mov qword ptr [rsp], r8
    mov qword ptr [rsp + 8], r9

    mov r12, r13

.Lj_loop:
    cmp r12, r14
    jb .Lj_done

    mov rax, r12
    sub rax, r14
    shl rax, 4
    lea rsi, [rbx + rax]
    mov r8, qword ptr [rsp]
    mov r9, qword ptr [rsp + 8]
    call is_greater
    test eax, eax
    jz .Lj_done

    mov rax, r12
    sub rax, r14
    shl rax, 4
    mov rcx, qword ptr [rbx + rax]
    mov rdx, qword ptr [rbx + rax + 8]

    mov rax, r12
    shl rax, 4
    mov qword ptr [rbx + rax], rcx
    mov qword ptr [rbx + rax + 8], rdx

    sub r12, r14
    jmp .Lj_loop

.Lj_done:
    mov rcx, qword ptr [rsp]
    mov rdx, qword ptr [rsp + 8]
    mov rax, r12
    shl rax, 4
    mov qword ptr [rbx + rax], rcx
    mov qword ptr [rbx + rax + 8], rdx

    inc r13
    jmp .Li_loop

.Lnext_gap:
    shr r14, 1
    jmp .Lgap_loop

.Lsort_done:
    # Build output in out_buf
    lea rdi, [rip + out_buf]
    lea rbx, [rip + lines]
    xor r13d, r13d

.Lout_loop:
    cmp r13, r15
    jae .Lout_done

    mov rax, r13
    shl rax, 4
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb

    mov byte ptr [rdi], 10
    inc rdi

    inc r13
    jmp .Lout_loop

.Lout_done:
    lea rax, [rip + out_buf]
    sub rdi, rax
    mov rbp, rdi

    # Write output to stdout (fd = 1)
    xor r12d, r12d
.Lwrite_loop:
    cmp r12, rbp
    jae .Lexit_clean
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, r12
    mov rdx, rbp
    sub rdx, r12
    syscall
    test rax, rax
    jle .Lexit_clean
    add r12, rax
    jmp .Lwrite_loop

.Lexit_clean:
    mov eax, 60
    xor edi, edi
    syscall

# Returns 1 if Line A (*rsi) > Line B (r8, r9), else 0
is_greater:
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    mov rax, r8

    mov rcx, r11
    cmp rcx, r9
    cmova rcx, r9

    test rcx, rcx
    jz .Lcmp_len

.Lcmp_loop:
    mov dl, byte ptr [r10]
    mov dh, byte ptr [rax]
    cmp dl, dh
    jne .Lcmp_diff
    inc r10
    inc rax
    dec rcx
    jnz .Lcmp_loop

.Lcmp_len:
    cmp r11, r9
    ja .Ltrue
    xor eax, eax
    ret

.Lcmp_diff:
    ja .Ltrue
    xor eax, eax
    ret

.Ltrue:
    mov eax, 1
    ret

.bss
.balign 16
in_buf:
    .zero 65536
lines:
    .zero 32768
out_buf:
    .zero 65536
