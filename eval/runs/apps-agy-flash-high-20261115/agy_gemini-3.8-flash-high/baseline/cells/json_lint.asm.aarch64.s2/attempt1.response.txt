.globl _start

.section .rodata
.balign 8
msg_ok:
    .ascii "ok\n"
msg_error_prefix:
    .ascii "error at "

.text
.balign 4

_start:
    // x19 = pointer to input buffer
    // x21 = total bytes read
    mov x21, #0
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf

read_loop:
    mov x0, #0
    add x1, x19, x21
    movz x2, #1, lsl #16 // 65536 bytes
    mov x8, #63          // sys_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop

read_done:
    // x20 = current cursor position
    mov x20, #0

    bl skip_ws
    cmp x20, x21
    b.eq fail

    bl parse_value

    bl skip_ws
    cmp x20, x21
    b.ne fail

    // Valid document: write "ok\n" to stdout
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
write_ok_loop:
    mov x0, #1
    mov x8, #64          // sys_write
    svc #0
    cmp x0, #0
    b.le write_ok_done
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_ok_loop

write_ok_done:
    mov x0, #0           // exit status 0
    mov x8, #93          // sys_exit
    svc #0

// skip whitespace (0x20, 0x09, 0x0A, 0x0D)
skip_ws:
skip_ws_loop:
    cmp x20, x21
    b.eq skip_ws_done
    ldrb w0, [x19, x20]
    cmp w0, #0x20
    b.eq 1f
    cmp w0, #0x0A
    b.eq 1f
    cmp w0, #0x0D
    b.eq 1f
    cmp w0, #0x09
    b.ne skip_ws_done
1:
    add x20, x20, #1
    b skip_ws_loop
skip_ws_done:
    ret

// parse_value
parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #0x22        // '"'
    b.eq val_str
    cmp w0, #'{'
    b.eq val_obj
    cmp w0, #'['
    b.eq val_arr
    cmp w0, #'t'
    b.eq val_true
    cmp w0, #'f'
    b.eq val_false
    cmp w0, #'n'
    b.eq val_null
    cmp w0, #'-'
    b.eq val_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls val_num
    b fail

val_str:
    bl parse_string
    ldp x29, x30, [sp], #16
    ret
val_obj:
    bl parse_object
    ldp x29, x30, [sp], #16
    ret
val_arr:
    bl parse_array
    ldp x29, x30, [sp], #16
    ret
val_true:
    bl parse_true
    ldp x29, x30, [sp], #16
    ret
val_false:
    bl parse_false
    ldp x29, x30, [sp], #16
    ret
val_null:
    bl parse_null
    ldp x29, x30, [sp], #16
    ret
val_num:
    bl parse_number
    ldp x29, x30, [sp], #16
    ret

// parse_string
parse_string:
    add x20, x20, #1
str_loop:
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #0x22        // '"'
    b.eq str_done
    cmp w0, #0x5C        // '\\'
    b.eq str_escape
    cmp w0, #0x20
    b.lo fail
    add x20, x20, #1
    b str_loop

str_escape:
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #0x22        // '"'
    b.eq str_esc_ok
    cmp w0, #0x5C        // '\\'
    b.eq str_esc_ok
    cmp w0, #0x2F        // '/'
    b.eq str_esc_ok
    cmp w0, #'b'
    b.eq str_esc_ok
    cmp w0, #'f'
    b.eq str_esc_ok
    cmp w0, #'n'
    b.eq str_esc_ok
    cmp w0, #'r'
    b.eq str_esc_ok
    cmp w0, #'t'
    b.eq str_esc_ok
    cmp w0, #'u'
    b.eq str_unicode
    b fail

str_esc_ok:
    add x20, x20, #1
    b str_loop

str_unicode:
    add x20, x20, #1
    mov w3, #4
str_u_loop:
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 1f
    bic w2, w0, #0x20
    sub w2, w2, #'A'
    cmp w2, #5
    b.hi fail
1:
    add x20, x20, #1
    subs w3, w3, #1
    b.ne str_u_loop
    b str_loop

str_done:
    add x20, x20, #1
    ret

// parse_number
parse_number:
    ldrb w0, [x19, x20]
    cmp w0, #'-'
    b.ne num_int
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail

num_int:
    cmp w0, #'0'
    b.ne num_digits_1_9
    add x20, x20, #1
    b num_frac

num_digits_1_9:
    add x20, x20, #1
num_int_loop:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi num_frac
    add x20, x20, #1
    b num_int_loop

num_frac:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x19, x20]
    cmp w0, #'.'
    b.ne num_exp
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
    add x20, x20, #1
num_frac_loop:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi num_exp
    add x20, x20, #1
    b num_frac_loop

num_exp:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.eq 1f
    cmp w0, #'E'
    b.ne num_done
1:
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'+'
    b.eq 2f
    cmp w0, #'-'
    b.ne 3f
2:
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
3:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
    add x20, x20, #1
num_exp_loop:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi num_done
    add x20, x20, #1
    b num_exp_loop

num_done:
    ret

// parse_true
parse_true:
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'r'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'u'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.ne fail
    add x20, x20, #1
    ret

// parse_false
parse_false:
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'a'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'l'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'s'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.ne fail
    add x20, x20, #1
    ret

// parse_null
parse_null:
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'u'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'l'
    b.ne fail
    add x20, x20, #1
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'l'
    b.ne fail
    add x20, x20, #1
    ret

// parse_array
parse_array:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #']'
    b.eq arr_empty

    bl parse_value

arr_loop:
    bl skip_ws
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #']'
    b.eq arr_close
    cmp w0, #','
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    b arr_loop

arr_empty:
arr_close:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

// parse_object
parse_object:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'}'
    b.eq obj_empty

obj_member:
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #0x22        // '"'
    b.ne fail
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #':'
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value

obj_loop:
    bl skip_ws
    cmp x20, x21
    b.eq fail
    ldrb w0, [x19, x20]
    cmp w0, #'}'
    b.eq obj_close
    cmp w0, #','
    b.ne fail
    add x20, x20, #1
    bl skip_ws
    b obj_member

obj_empty:
obj_close:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

// fail handler: prints "error at N\n" and exits with status 1
fail:
    sub sp, sp, #64
    add x1, sp, #63
    mov w2, #0x0A        // '\n'
    strb w2, [x1]
    mov x3, x20

format_digits:
    mov x4, #10
    udiv x5, x3, x4
    msub x6, x5, x4, x3
    add w6, w6, #'0'
    sub x1, x1, #1
    strb w6, [x1]
    mov x3, x5
    cbnz x3, format_digits

    sub x1, x1, #9
    adrp x7, msg_error_prefix
    add x7, x7, :lo12:msg_error_prefix
    ldrb w8, [x7]
    strb w8, [x1]
    ldrb w8, [x7, #1]
    strb w8, [x1, #1]
    ldrb w8, [x7, #2]
    strb w8, [x1, #2]
    ldrb w8, [x7, #3]
    strb w8, [x1, #3]
    ldrb w8, [x7, #4]
    strb w8, [x1, #4]
    ldrb w8, [x7, #5]
    strb w8, [x1, #5]
    ldrb w8, [x7, #6]
    strb w8, [x1, #6]
    ldrb w8, [x7, #7]
    strb w8, [x1, #7]
    ldrb w8, [x7, #8]
    strb w8, [x1, #8]

    add x2, sp, #64
    sub x2, x2, x1

write_err_loop:
    mov x0, #1           // stdout
    mov x8, #64          // sys_write
    svc #0
    cmp x0, #0
    b.le write_err_done
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_err_loop

write_err_done:
    mov x0, #1           // exit status 1
    mov x8, #93          // sys_exit
    svc #0

.bss
.balign 16
input_buf:
    .zero 131072
