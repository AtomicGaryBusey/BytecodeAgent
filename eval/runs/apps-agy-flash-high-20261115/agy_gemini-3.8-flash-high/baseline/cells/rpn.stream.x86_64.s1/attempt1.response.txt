.equ str_error_len, 6

.section .rodata
str_error:
    .ascii "error\n"

.bss
.balign 16
in_head:
    .space 8
in_tail:
    .space 8
in_buf:
    .space 8192
line_buf:
    .space 2048
rpn_stack:
    .space 16384

.text
.globl _start
_start:
.Lmain_loop:
    call read_line
    cmp rax, -1
    je .Lexit_prog

    mov r12, rax
    xor r13d, r13d
    xor r14d, r14d
    lea rbp, [rip + line_buf]
    lea r8, [rip + rpn_stack]

.Ltoken_loop:
    cmp r13, r12
    jae .Lline_end

    movzx eax, byte ptr [rbp + r13]
    cmp al, ' '
    je .Lskip_sep
    cmp al, 9
    je .Lskip_sep

    mov r15, r13

.Lfind_token_end:
    inc r13
    cmp r13, r12
    jae .Ltoken_found
    movzx eax, byte ptr [rbp + r13]
    cmp al, ' '
    je .Ltoken_found
    cmp al, 9
    je .Ltoken_found
    jmp .Lfind_token_end

.Lskip_sep:
    inc r13
    jmp .Ltoken_loop

.Ltoken_found:
    mov rbx, r13
    sub rbx, r15
    lea rsi, [rbp + r15]

    cmp rbx, 1
    jne .Lnot_single_char

    mov al, byte ptr [rsi]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod

.Lnot_single_char:
    mov al, byte ptr [rsi]
    cmp al, '-'
    je .Lnegative_literal

.Lpositive_literal:
    cmp rbx, 18
    ja .Lline_error

    xor eax, eax
    xor ecx, ecx
.Lpos_digit_loop:
    movzx edx, byte ptr [rsi + rcx]
    sub edx, '0'
    cmp edx, 9
    ja .Lline_error
    lea rax, [rax + rax * 4]
    lea rax, [rdx + rax * 2]
    inc rcx
    cmp rcx, rbx
    jb .Lpos_digit_loop

    mov qword ptr [r8 + r14*8], rax
    inc r14
    jmp .Ltoken_loop

.Lnegative_literal:
    cmp rbx, 19
    ja .Lline_error

    xor eax, eax
    mov ecx, 1
.Lneg_digit_loop:
    movzx edx, byte ptr [rsi + rcx]
    sub edx, '0'
    cmp edx, 9
    ja .Lline_error
    lea rax, [rax + rax * 4]
    lea rax, [rdx + rax * 2]
    inc rcx
    cmp rcx, rbx
    jb .Lneg_digit_loop

    neg rax
    mov qword ptr [r8 + r14*8], rax
    inc r14
    jmp .Ltoken_loop

.Lop_add:
    cmp r14, 2
    jb .Lline_error
    mov rbx, qword ptr [r8 + r14*8 - 8]
    mov rax, qword ptr [r8 + r14*8 - 16]
    add rax, rbx
    dec r14
    mov qword ptr [r8 + r14*8 - 8], rax
    jmp .Ltoken_loop

.Lop_sub:
    cmp r14, 2
    jb .Lline_error
    mov rbx, qword ptr [r8 + r14*8 - 8]
    mov rax, qword ptr [r8 + r14*8 - 16]
    sub rax, rbx
    dec r14
    mov qword ptr [r8 + r14*8 - 8], rax
    jmp .Ltoken_loop

.Lop_mul:
    cmp r14, 2
    jb .Lline_error
    mov rbx, qword ptr [r8 + r14*8 - 8]
    mov rax, qword ptr [r8 + r14*8 - 16]
    imul rax, rbx
    dec r14
    mov qword ptr [r8 + r14*8 - 8], rax
    jmp .Ltoken_loop

.Lop_div:
    cmp r14, 2
    jb .Lline_error
    mov rbx, qword ptr [r8 + r14*8 - 8]
    mov rax, qword ptr [r8 + r14*8 - 16]
    test rbx, rbx
    jz .Lline_error
    cmp rbx, -1
    jne 1f
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je .Lline_error
1:
    cqo
    idiv rbx
    dec r14
    mov qword ptr [r8 + r14*8 - 8], rax
    jmp .Ltoken_loop

.Lop_mod:
    cmp r14, 2
    jb .Lline_error
    mov rbx, qword ptr [r8 + r14*8 - 8]
    mov rax, qword ptr [r8 + r14*8 - 16]
    test rbx, rbx
    jz .Lline_error
    cmp rbx, -1
    jne 1f
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je .Lline_error
1:
    cqo
    idiv rbx
    dec r14
    mov qword ptr [r8 + r14*8 - 8], rdx
    jmp .Ltoken_loop

.Lline_end:
    cmp r14, 1
    jne .Lline_error
    mov rax, qword ptr [r8]
    call print_int64
    jmp .Lmain_loop

.Lline_error:
    lea rsi, [rip + str_error]
    mov edx, str_error_len
    call write_all
    jmp .Lmain_loop

.Lexit_prog:
    mov eax, 60
    xor edi, edi
    syscall

read_line:
    push rbx
    push r12
    xor ebx, ebx
    lea r12, [rip + line_buf]

.Lread_line_loop:
    call get_byte
    cmp rax, -1
    je .Lread_line_eof
    cmp al, 10
    je .Lread_line_nl

    mov byte ptr [r12 + rbx], al
    inc rbx
    jmp .Lread_line_loop

.Lread_line_nl:
    mov rax, rbx
    pop r12
    pop rbx
    ret

.Lread_line_eof:
    test rbx, rbx
    jnz .Lread_line_nl
    mov rax, -1
    pop r12
    pop rbx
    ret

get_byte:
    mov rax, qword ptr [rip + in_head]
    cmp rax, qword ptr [rip + in_tail]
    jb 1f
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 8192
    syscall
    test rax, rax
    jle 2f
    mov qword ptr [rip + in_tail], rax
    mov qword ptr [rip + in_head], 1
    movzx eax, byte ptr [rip + in_buf]
    ret
1:
    lea rdx, [rip + in_buf]
    movzx edx, byte ptr [rdx + rax]
    inc rax
    mov qword ptr [rip + in_head], rax
    mov eax, edx
    ret
2:
    mov rax, -1
    ret

print_int64:
    sub rsp, 40
    lea rdi, [rsp + 39]
    mov byte ptr [rdi], 10

    xor r8d, r8d
    test rax, rax
    jns .Lpos_num
    inc r8d
    neg rax
.Lpos_num:
    mov rcx, 10
.Ldigit_loop:
    dec rdi
    xor edx, edx
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .Ldigit_loop

    test r8d, r8d
    jz .Lprint_out
    dec rdi
    mov byte ptr [rdi], '-'

.Lprint_out:
    lea rdx, [rsp + 40]
    sub rdx, rdi
    mov rsi, rdi
    call write_all

    add rsp, 40
    ret

write_all:
    test rdx, rdx
    jle .Lwrite_done
.Lwrite_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lwrite_done
    add rsi, rax
    sub rdx, rax
    jnz .Lwrite_loop
.Lwrite_done:
    ret
