.globl _start

.text
_start:
    xor r12, r12

.Lread_loop:
    cmp r12, 65536
    jae .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    test r12, r12
    jz .Lexit

    lea rsi, [rip + in_buf]
    lea rdi, [rip + out_buf]
    lea r8, [rip + b64_table]
    xor r13d, r13d

.Lmain_loop:
    cmp r12, 3
    jb .Lremainder

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 0x3F
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    sub r12, 3

    inc r13d
    cmp r13d, 19
    jne .Lmain_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r13d, r13d
    jmp .Lmain_loop

.Lremainder:
    test r12, r12
    jz .Lfinish

    cmp r12, 1
    je .Lrem1

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    shl eax, 16
    shl ecx, 8
    or eax, ecx

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 61
    add rdi, 4
    inc r13d
    jmp .Lfinish

.Lrem1:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    inc r13d

.Lfinish:
    test r13d, r13d
    jz .Lwrite_out
    mov byte ptr [rdi], 10
    inc rdi

.Lwrite_out:
    lea rax, [rip + out_buf]
    mov r15, rdi
    sub r15, rax
    xor ebx, ebx

.Lwrite_loop:
    cmp rbx, r15
    jae .Lexit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, rbx
    mov rdx, r15
    sub rdx, rbx
    syscall
    test rax, rax
    jle .Lexit
    add rbx, rax
    jmp .Lwrite_loop

.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata

    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
