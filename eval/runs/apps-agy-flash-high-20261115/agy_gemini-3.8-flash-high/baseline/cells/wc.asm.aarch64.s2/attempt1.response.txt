.text
.globl _start

_start:
    // Initialise counters and state:
    // x19 = line count
    // x20 = word count
    // x21 = byte count
    // x22 = in_word flag (0 = outside word, 1 = inside word)
    mov     x19, #0
    mov     x20, #0
    mov     x21, #0
    mov     x22, #0

.Lread_loop:
    // sys_read(fd=0, buf=inbuf, count=65536)
    mov     x0, #0
    adrp    x1, inbuf
    add     x1, x1, :lo12:inbuf
    movz    x2, #1, lsl #16
    mov     x8, #63
    svc     #0

    // If read returned <= 0, we reached EOF or error
    cmp     x0, #0
    b.le    .Lread_done

    // Update total byte count
    add     x21, x21, x0

    // Set up pointer and loop counter for the read chunk
    adrp    x1, inbuf
    add     x1, x1, :lo12:inbuf
    mov     x25, x0

.Lbyte_loop:
    ldrb    w2, [x1], #1

    // Check for newline (0x0A)
    cmp     w2, #10
    b.ne    .Lcheck_space
    add     x19, x19, #1        // lines++
    mov     x22, #0             // newline is whitespace
    b       .Lnext_byte

.Lcheck_space:
    // Check for space (0x20)
    cmp     w2, #32
    b.eq    .Lis_ws

    // Check for other whitespace: tab (9), VT (11), FF (12), CR (13)
    // Values 9..13 (10 was already checked above)
    sub     w3, w2, #9
    cmp     w3, #4
    b.ls    .Lis_ws

    // Non-whitespace character:
    cbnz    x22, .Lnext_byte    // already in word, continue
    add     x20, x20, #1        // new word started: words++
    mov     x22, #1             // in_word = 1
    b       .Lnext_byte

.Lis_ws:
    mov     x22, #0             // in_word = 0

.Lnext_byte:
    subs    x25, x25, #1
    b.ne    .Lbyte_loop

    b       .Lread_loop

.Lread_done:
    // Format output: "<lines> <words> <bytes>\n"
    adrp    x26, outbuf
    add     x26, x26, :lo12:outbuf
    mov     x1, x26             // x1 is write pointer in outbuf

    // Format line count
    mov     x0, x19
    bl      format_uint

    // Space separator
    mov     w2, #32
    strb    w2, [x1], #1

    // Format word count
    mov     x0, x20
    bl      format_uint

    // Space separator
    mov     w2, #32
    strb    w2, [x1], #1

    // Format byte count
    mov     x0, x21
    bl      format_uint

    // Trailing newline
    mov     w2, #10
    strb    w2, [x1], #1

    // Total bytes to write = current x1 - outbuf
    sub     x27, x1, x26

.Lwrite_loop:
    cbz     x27, .Lwrite_done
    mov     x0, #1              // fd = 1 (stdout)
    mov     x1, x26             // buf
    mov     x2, x27             // count
    mov     x8, #64             // sys_write = 64
    svc     #0

    cmp     x0, #0
    b.le    .Lwrite_done
    add     x26, x26, x0
    sub     x27, x27, x0
    b       .Lwrite_loop

.Lwrite_done:
    // Exit with status 0
    mov     x0, #0
    mov     x8, #93             // sys_exit = 93
    svc     #0

.Lhalt_loop:
    b       .Lhalt_loop

// Helper function: format uint64 in decimal
// Input:  x0 = uint64 value, x1 = output buffer pointer
// Output: x1 = updated output buffer pointer (points after last written char)
format_uint:
    cbnz    x0, .Lfu_nonzero
    mov     w2, #48             // '0'
    strb    w2, [x1], #1
    ret

.Lfu_nonzero:
    sub     sp, sp, #32
    add     x2, sp, #32         // pointer to end of temporary buffer
    mov     x3, #10

.Lfu_div_loop:
    udiv    x4, x0, x3          // x4 = x0 / 10
    msub    x5, x4, x3, x0      // x5 = x0 - x4 * 10 (remainder)
    add     w5, w5, #48         // convert remainder to ASCII digit
    sub     x2, x2, #1
    strb    w5, [x2]
    mov     x0, x4
    cbnz    x0, .Lfu_div_loop

    add     x4, sp, #32

.Lfu_copy_loop:
    ldrb    w5, [x2], #1
    strb    w5, [x1], #1
    cmp     x2, x4
    b.ne    .Lfu_copy_loop

    add     sp, sp, #32
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 128
