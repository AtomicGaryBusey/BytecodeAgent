.globl _start

.text
_start:
    cld
    sub rsp, 16

    xor r12, r12
.Lread_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jbe .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    test r12, r12
    jz .Lexit_success

    xor rbx, rbx
    xor rcx, rcx
    xor r13, r13
    lea r14, [rip + in_buf]
    lea r15, [rip + lines]

.Lparse_loop:
    cmp rcx, r12
    jae .Lparse_end
    cmp byte ptr [r14 + rcx], 10
    jne .Lnext_char

    mov rax, r14
    add rax, rbx
    mov rdx, rcx
    sub rdx, rbx

    mov r8, r13
    shl r8, 4
    mov [r15 + r8], rax
    mov [r15 + r8 + 8], rdx
    inc r13

    lea rbx, [rcx + 1]

.Lnext_char:
    inc rcx
    jmp .Lparse_loop

.Lparse_end:
    cmp rbx, r12
    jae .Llines_done
    mov rax, r14
    add rax, rbx
    mov rdx, r12
    sub rdx, rbx

    mov r8, r13
    shl r8, 4
    mov [r15 + r8], rax
    mov [r15 + r8 + 8], rdx
    inc r13

.Llines_done:
    test r13, r13
    jz .Lexit_success

    # Shell sort lines array
    mov r12, r13
    lea rbx, [rip + lines]

    cmp r12, 1
    jbe .Lsort_done

    mov r13, r12
    shr r13, 1

.Lgap_loop:
    test r13, r13
    jz .Lsort_done

    mov r14, r13
.Li_loop:
    cmp r14, r12
    jae .Lnext_gap

    mov rax, r14
    shl rax, 4
    mov r8, [rbx + rax]
    mov r9, [rbx + rax + 8]
    mov [rsp], r8
    mov [rsp + 8], r9

    mov r15, r14
.Lj_loop:
    cmp r15, r13
    jb .Lj_done

    mov rax, r15
    sub rax, r13
    shl rax, 4
    lea rsi, [rbx + rax]
    mov rdi, rsp
    call compare_lines
    jbe .Lj_done

    mov r8, [rsi]
    mov r9, [rsi + 8]
    mov rax, r15
    shl rax, 4
    mov [rbx + rax], r8
    mov [rbx + rax + 8], r9

    sub r15, r13
    jmp .Lj_loop

.Lj_done:
    mov r8, [rsp]
    mov r9, [rsp + 8]
    mov rax, r15
    shl rax, 4
    mov [rbx + rax], r8
    mov [rbx + rax + 8], r9

    inc r14
    jmp .Li_loop

.Lnext_gap:
    shr r13, 1
    jmp .Lgap_loop

.Lsort_done:
    # Populate out_buf
    lea rbx, [rip + lines]
    lea rdi, [rip + out_buf]
    xor r13, r13

.Lout_loop:
    cmp r13, r12
    jae .Lout_done

    mov rax, r13
    shl rax, 4
    mov rsi, [rbx + rax]
    mov rcx, [rbx + rax + 8]

    rep movsb

    mov byte ptr [rdi], 10
    inc rdi

    inc r13
    jmp .Lout_loop

.Lout_done:
    lea rax, [rip + out_buf]
    sub rdi, rax
    mov r14, rdi

    # Write output to stdout
    xor rbx, rbx
.Lwrite_loop:
    cmp rbx, r14
    jae .Lexit_success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    syscall
    test rax, rax
    jle .Lexit_success
    add rbx, rax
    jmp .Lwrite_loop

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r8, [rsi]
    mov r9, [rsi + 8]
    mov r10, [rdi]
    mov r11, [rdi + 8]

    mov rcx, r9
    cmp rcx, r11
    cmova rcx, r11

    xor rdx, rdx
.Lcmp_loop:
    cmp rdx, rcx
    je .Lcmp_len
    movzx eax, byte ptr [r8 + rdx]
    cmp al, byte ptr [r10 + rdx]
    jne .Lcmp_diff
    inc rdx
    jmp .Lcmp_loop

.Lcmp_diff:
    ret

.Lcmp_len:
    cmp r9, r11
    ret

.bss
.balign 16
in_buf:
    .zero 65536
.balign 16
lines:
    .zero 16384
.balign 16
out_buf:
    .zero 65536
