    .globl _start
    .text

_start:
    // Read all input from stdin into in_buf
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0                 // total bytes read

read_loop:
    movz x2, #1, lsl #16        // buffer capacity = 65536 bytes
    sub x2, x2, x20             // remaining space in buffer
    cbz x2, read_done

    mov x0, #0                  // stdin (fd = 0)
    add x1, x19, x20            // destination pointer
    mov x8, #63                 // read syscall
    svc #0

    cmp x0, #0
    ble read_done               // 0 = EOF, < 0 = error/stop
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_success       // empty input prints nothing

    // Set up pointers for base64 encoding
    mov x21, x19                // current input pointer
    add x22, x19, x20           // input end pointer

    adrp x23, out_buf
    add x23, x23, :lo12:out_buf // current output pointer

    adrp x26, b64_table
    add x26, x26, :lo12:b64_table

    mov w25, #0                 // line character count = 0

encode_loop:
    sub x0, x22, x21            // bytes remaining in input
    cmp x0, #3
    b.lt handle_remainder

    // Load 3 input bytes
    ldrb w0, [x21]
    ldrb w1, [x21, #1]
    ldrb w2, [x21, #2]
    add x21, x21, #3

    // Combine into 24-bit integer
    lsl w0, w0, #16
    orr w0, w0, w1, lsl #8
    orr w3, w0, w2

    // Extract four 6-bit values
    ubfx w4, w3, #18, #6
    ubfx w5, w3, #12, #6
    ubfx w6, w3, #6, #6
    ubfx w7, w3, #0, #6

    // Look up in base64 alphabet
    ldrb w4, [x26, x4]
    ldrb w5, [x26, x5]
    ldrb w6, [x26, x6]
    ldrb w7, [x26, x7]

    // Store 4 encoded characters
    strb w4, [x23], #1
    strb w5, [x23], #1
    strb w6, [x23], #1
    strb w7, [x23], #1

    // Update column counter; wrap line at 76 characters
    add w25, w25, #4
    cmp w25, #76
    b.ne encode_loop

    mov w8, #10                 // newline '\n'
    strb w8, [x23], #1
    mov w25, #0
    b encode_loop

handle_remainder:
    cbz x0, finish_output
    cmp x0, #1
    b.ne rem_two

    // Remainder of 1 input byte
    ldrb w0, [x21]
    lsl w3, w0, #16
    ubfx w4, w3, #18, #6
    ubfx w5, w3, #12, #6
    ldrb w4, [x26, x4]
    ldrb w5, [x26, x5]
    mov w6, #61                 // '='
    mov w7, #61                 // '='
    b emit_rem

rem_two:
    // Remainder of 2 input bytes
    ldrb w0, [x21]
    ldrb w1, [x21, #1]
    lsl w0, w0, #16
    orr w3, w0, w1, lsl #8
    ubfx w4, w3, #18, #6
    ubfx w5, w3, #12, #6
    ubfx w6, w3, #6, #6
    ldrb w4, [x26, x4]
    ldrb w5, [x26, x5]
    ldrb w6, [x26, x6]
    mov w7, #61                 // '='

emit_rem:
    strb w4, [x23], #1
    strb w5, [x23], #1
    strb w6, [x23], #1
    strb w7, [x23], #1

    add w25, w25, #4
    cmp w25, #76
    b.ne finish_output

    mov w8, #10                 // newline '\n'
    strb w8, [x23], #1
    mov w25, #0

finish_output:
    // If the last line has any characters, terminate it with a newline
    cbz w25, do_write
    mov w8, #10                 // newline '\n'
    strb w8, [x23], #1

do_write:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    sub x20, x23, x24           // total output bytes to write
    mov x19, x24                // current write buffer pointer

write_loop:
    cbz x20, exit_success
    mov x0, #1                  // stdout (fd = 1)
    mov x1, x19
    mov x2, x20
    mov x8, #64                 // write syscall
    svc #0

    cmp x0, #0
    ble exit_success
    add x19, x19, x0
    sub x20, x20, x0
    b write_loop

exit_success:
    mov x0, #0                  // exit status 0
    mov x8, #93                 // exit syscall
    svc #0

    .section .rodata
    .balign 8
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .bss
    .balign 16
in_buf:
    .space 65536
out_buf:
    .space 131072
