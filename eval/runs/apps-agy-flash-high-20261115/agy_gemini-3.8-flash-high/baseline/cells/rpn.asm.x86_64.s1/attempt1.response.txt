.globl _start

.section .rodata
err_msg:
    .ascii "error\n"
err_msg_len = . - err_msg

.text
_start:
    xor ebx, ebx
.Lread_loop:
    mov edx, 131072
    sub rdx, rbx
    jz .Lread_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle .Lread_done
    add rbx, rax
    jmp .Lread_loop

.Lread_done:
    test rbx, rbx
    jz .Lexit_success

    lea r12, [rip + inbuf]
    lea rbx, [r12 + rbx]
    lea rbp, [rip + eval_stack]

.Lline_loop:
    cmp r12, rbx
    jae .Lall_done

    mov r13, r12
.Lfind_nl:
    cmp r13, rbx
    jae .Lno_nl
    cmp byte ptr [r13], 10
    je .Lfound_nl
    inc r13
    jmp .Lfind_nl

.Lfound_nl:
    lea r15, [r13 + 1]
    jmp .Lprocess_line

.Lno_nl:
    mov r15, rbx

.Lprocess_line:
    xor r14d, r14d

.Ltoken_loop:
.Lskip_sep:
    cmp r12, r13
    jae .Lend_of_tokens
    mov al, byte ptr [r12]
    cmp al, ' '
    je .Lis_sep
    cmp al, 9
    je .Lis_sep
    jmp .Ltoken_start_found
.Lis_sep:
    inc r12
    jmp .Lskip_sep

.Ltoken_start_found:
    mov rsi, r12
.Lfind_token_end:
    cmp r12, r13
    jae .Ltoken_end_found
    mov al, byte ptr [r12]
    cmp al, ' '
    je .Ltoken_end_found
    cmp al, 9
    je .Ltoken_end_found
    inc r12
    jmp .Lfind_token_end

.Ltoken_end_found:
    mov rcx, r12
    sub rcx, rsi

    cmp rcx, 1
    jne .Ltok_len_gt_1

    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod

    cmp al, '0'
    jb .Lline_error
    cmp al, '9'
    ja .Lline_error
    sub eax, '0'
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp .Ltoken_loop

.Ltok_len_gt_1:
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    je .Ltok_negative

    cmp rcx, 18
    ja .Lline_error
    xor eax, eax
    xor edx, edx
.Lpos_digit_loop:
    cmp rdx, rcx
    jae .Lpos_digit_done
    movzx r8d, byte ptr [rsi + rdx]
    cmp r8b, '0'
    jb .Lline_error
    cmp r8b, '9'
    ja .Lline_error
    sub r8d, '0'
    lea rax, [rax + rax * 4]
    lea rax, [r8 + rax * 2]
    inc rdx
    jmp .Lpos_digit_loop
.Lpos_digit_done:
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp .Ltoken_loop

.Ltok_negative:
    cmp rcx, 19
    ja .Lline_error
    xor eax, eax
    mov edx, 1
.Lneg_digit_loop:
    cmp rdx, rcx
    jae .Lneg_digit_done
    movzx r8d, byte ptr [rsi + rdx]
    cmp r8b, '0'
    jb .Lline_error
    cmp r8b, '9'
    ja .Lline_error
    sub r8d, '0'
    lea rax, [rax + rax * 4]
    lea rax, [r8 + rax * 2]
    inc rdx
    jmp .Lneg_digit_loop
.Lneg_digit_done:
    neg rax
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp .Ltoken_loop

.Lop_add:
    cmp r14, 2
    jb .Lline_error
    sub r14, 2
    mov rdi, qword ptr [rbp + r14*8]
    mov r8,  qword ptr [rbp + r14*8 + 8]
    add rdi, r8
    mov qword ptr [rbp + r14*8], rdi
    inc r14
    jmp .Ltoken_loop

.Lop_sub:
    cmp r14, 2
    jb .Lline_error
    sub r14, 2
    mov rdi, qword ptr [rbp + r14*8]
    mov r8,  qword ptr [rbp + r14*8 + 8]
    sub rdi, r8
    mov qword ptr [rbp + r14*8], rdi
    inc r14
    jmp .Ltoken_loop

.Lop_mul:
    cmp r14, 2
    jb .Lline_error
    sub r14, 2
    mov rdi, qword ptr [rbp + r14*8]
    mov r8,  qword ptr [rbp + r14*8 + 8]
    imul rdi, r8
    mov qword ptr [rbp + r14*8], rdi
    inc r14
    jmp .Ltoken_loop

.Lop_div:
    cmp r14, 2
    jb .Lline_error
    sub r14, 2
    mov rdi, qword ptr [rbp + r14*8]
    mov r8,  qword ptr [rbp + r14*8 + 8]
    test r8, r8
    jz .Lline_error
    mov rax, 1
    shl rax, 63
    cmp rdi, rax
    jne .Ldiv_no_overflow
    cmp r8, -1
    je .Lline_error
.Ldiv_no_overflow:
    mov rax, rdi
    cqo
    idiv r8
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp .Ltoken_loop

.Lop_mod:
    cmp r14, 2
    jb .Lline_error
    sub r14, 2
    mov rdi, qword ptr [rbp + r14*8]
    mov r8,  qword ptr [rbp + r14*8 + 8]
    test r8, r8
    jz .Lline_error
    mov rax, 1
    shl rax, 63
    cmp rdi, rax
    jne .Lmod_no_overflow
    cmp r8, -1
    je .Lline_error
.Lmod_no_overflow:
    mov rax, rdi
    cqo
    idiv r8
    mov qword ptr [rbp + r14*8], rdx
    inc r14
    jmp .Ltoken_loop

.Lend_of_tokens:
    cmp r14, 1
    jne .Lline_error
    mov rax, qword ptr [rbp]
    call format_and_append_int
    jmp .Lline_finished

.Lline_error:
    lea rsi, [rip + err_msg]
    mov ecx, err_msg_len
    call append_output

.Lline_finished:
    mov r12, r15
    jmp .Lline_loop

.Lall_done:
    call flush_outbuf
.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall

format_and_append_int:
    sub rsp, 32
    xor r9d, r9d
    test rax, rax
    jns .Lfmt_pos
    mov r9b, 1
    neg rax
.Lfmt_pos:
    lea r10, [rsp + 31]
    mov byte ptr [r10], 10
    mov rdi, r10

    test rax, rax
    jnz .Lfmt_div_setup
    dec rdi
    mov byte ptr [rdi], '0'
    jmp .Lfmt_sign

.Lfmt_div_setup:
    mov r8, 10
.Lfmt_div_loop:
    test rax, rax
    jz .Lfmt_sign
    xor edx, edx
    div r8
    dec rdi
    add dl, '0'
    mov byte ptr [rdi], dl
    jmp .Lfmt_div_loop

.Lfmt_sign:
    test r9b, r9b
    jz .Lfmt_append
    dec rdi
    mov byte ptr [rdi], '-'

.Lfmt_append:
    mov rcx, r10
    sub rcx, rdi
    inc rcx
    mov rsi, rdi
    call append_output
    add rsp, 32
    ret

append_output:
    mov rax, [rip + outbuf_len]
    add rax, rcx
    cmp rax, 60000
    jb .Lno_flush
    push rsi
    push rcx
    call flush_outbuf
    pop rcx
    pop rsi
.Lno_flush:
    mov r8, [rip + outbuf_len]
    lea rdi, [rip + outbuf]
    add rdi, r8
    add r8, rcx
    mov [rip + outbuf_len], r8
    rep movsb
    ret

flush_outbuf:
    mov r9, [rip + outbuf_len]
    test r9, r9
    jz .Lflush_ret
    xor r8d, r8d
.Lflush_loop:
    cmp r8, r9
    jae .Lflush_complete
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r8
    mov rdx, r9
    sub rdx, r8
    syscall
    test rax, rax
    jle .Lflush_complete
    add r8, rax
    jmp .Lflush_loop
.Lflush_complete:
    mov qword ptr [rip + outbuf_len], 0
.Lflush_ret:
    ret

.bss
.balign 16
outbuf_len:
    .zero 8
inbuf:
    .zero 131072
outbuf:
    .zero 65536
eval_stack:
    .zero 16384
