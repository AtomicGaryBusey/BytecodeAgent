    add x0, x0, :lo12:line_buf

    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]

    mov x21, #0
    adrp x22, eval_stack

token_loop:

    cmp x19, x20
    b.hs tokens_done
    ldrb w0, [x19]
    cmp w0, #' '
    b.eq 1f
    cmp w0, #'\t'

    cmp x19, x20
    b.hs tok_end_found
    ldrb w0, [x19]
    cmp w0, #' '
    b.eq tok_end_found
    cmp w0, #'\t'
    b.eq tok_end_found

tok_end_found:

    cmp x24, #1
    b.ne parse_literal

    ldrb w0, [x23]
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    cmp w0, #'/'
    b.eq op_div
    cmp w0, #'%'
    b.eq op_mod

    cmp w0, #'-'
    b.ne 1f
    mov x2, #1
    add x0, x23, #1
    sub x1, x24, #1
    b 2f
1:
    mov x2, #0
    mov x0, x23
    mov x1, x24
2:
    cmp x1, #1
    b.lo eval_error
    cmp x1, #18
    b.hi eval_error

    mov x3, #0
    mov x4, #10
    mov x5, #0

    sub w6, w6, #'0'
    cmp w6, #9
    b.hi eval_error
    madd x3, x3, x4, x6
    add x5, x5, #1
    cmp x5, x1
    b.ne parse_digit_loop

    cbz x2, push_literal
    neg x3, x3

    add x21, x21, #1
    b token_loop

op_add:

    b.lo eval_error
    sub x21, x21, #2
    ldr x1, [x22, x21, lsl #3]
    add x3, x21, #1
    ldr x2, [x22, x3, lsl #3]

    str x0, [x22, x21, lsl #3]
    add x21, x21, #1

op_sub:

    b.lo eval_error
    sub x21, x21, #2
    ldr x1, [x22, x21, lsl #3]
    add x3, x21, #1
    ldr x2, [x22, x3, lsl #3]

    str x0, [x22, x21, lsl #3]

    b.lo eval_error
    sub x21, x21, #2
    ldr x1, [x22, x21, lsl #3]
    add x3, x21, #1
    ldr x2, [x22, x3, lsl #3]

    b.lo eval_error
    sub x21, x21, #2
    ldr x1, [x22, x21, lsl #3]
    add x3, x21, #1
    ldr x2, [x22, x3, lsl #3]
    cbz x2, eval_error

    b.ne 1f
    mov x4, #1
    lsl x4, x4, #63

    b.eq eval_error

    str x0, [x22, x21, lsl #3]

    b.lo eval_error
    sub x21, x21, #2
    ldr x1, [x22, x21, lsl #3]
    add x3, x21, #1
    ldr x2, [x22, x3, lsl #3]

    cmn x2, #1
    b.ne 1f

    cmp x1, x4
    b.eq eval_error

    msub x0, x4, x2, x1
