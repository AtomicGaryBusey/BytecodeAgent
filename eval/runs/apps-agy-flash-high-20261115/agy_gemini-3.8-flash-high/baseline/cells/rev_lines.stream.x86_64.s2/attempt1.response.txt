.globl _start

.text
_start:
    lea r12, [rip + in_buf]
    xor r13, r13

read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r12 + r13]
    mov rdx, 131072
    sub rdx, r13
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + out_buf]
    xor r15, r15

    test r13, r13
    jz write_phase

    xor rbx, rbx
    xor rsi, rsi

scan_loop:
    cmp rsi, r13
    jae scan_done
    mov al, byte ptr [r12 + rsi]
    cmp al, 10
    jne next_char

    mov rdx, rsi
reverse_line:
    cmp rdx, rbx
    je reverse_line_done
    dec rdx
    mov al, byte ptr [r12 + rdx]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_line

reverse_line_done:
    mov byte ptr [r14 + r15], 10
    inc r15
    lea rbx, [rsi + 1]

next_char:
    inc rsi
    jmp scan_loop

scan_done:
    cmp rbx, r13
    jae write_phase

    mov rdx, r13
reverse_final:
    cmp rdx, rbx
    je reverse_final_done
    dec rdx
    mov al, byte ptr [r12 + rdx]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_final

reverse_final_done:
    mov byte ptr [r14 + r15], 10
    inc r15

write_phase:
    xor rbx, rbx

write_loop:
    cmp rbx, r15
    jae exit_success
    mov rax, 1
    mov rdi, 1
    lea rsi, [r14 + rbx]
    mov rdx, r15
    sub rdx, rbx
    syscall
    test rax, rax
    jle exit_success
    add rbx, rax
    jmp write_loop

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

.bss
.balign 16
in_buf:
    .zero 131072
out_buf:
    .zero 131072
