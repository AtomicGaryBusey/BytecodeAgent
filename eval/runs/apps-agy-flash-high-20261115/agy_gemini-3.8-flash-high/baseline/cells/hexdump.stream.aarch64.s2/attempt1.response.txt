.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 8192
.equ FLUSH_THRESHOLD, 4000

.text
.globl _start

_start:
    mov x19, #0         // total_offset = 0
    mov x20, #0         // in_ptr = 0
    mov x21, #0         // in_end = 0
    mov x22, #0         // out_cnt = 0
    mov x23, #0         // eof_flag = 0

    adrp x26, row_data
    add x26, x26, :lo12:row_data

row_loop:
    mov x25, #0         // row_len = 0

fill_row_loop:
    cmp x25, #16
    beq format_row

    cmp x20, x21
    blo 1f

    cbnz x23, format_row

    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    mov x20, x1
    mov x2, #BUF_SIZE
    mov x0, #0          // fd = stdin
    mov x8, #SYS_READ
    svc #0

    cmp x0, #0
    ble 2f

    add x21, x20, x0
    b 1f

2:
    mov x23, #1
    mov x21, x20
    b format_row

1:
    ldrb w0, [x20], #1
    strb w0, [x26, x25]
    add x25, x25, #1
    b fill_row_loop

format_row:
    cbz x25, done_rows

    cmp x22, #FLUSH_THRESHOLD
    blo 1f
    bl flush_out
1:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    add x24, x24, x22

    // 1. Format offset (x19) as 8 lowercase hex digits
    mov w0, w19
    mov w2, #28
2:
    lsr w3, w0, w2
    and w3, w3, #0xf
    cmp w3, #10
    add w4, w3, #48
    add w5, w3, #87
    csel w3, w5, w4, ge
    strb w3, [x24], #1
    subs w2, w2, #4
    bge 2b

    // Two spaces
    mov w3, #32
    strb w3, [x24], #1
    strb w3, [x24], #1

    // 2. Format 16 hex positions
    mov x27, #0
hex_pos_loop:
    cmp x27, x25
    bhs hex_pad_spaces

    ldrb w0, [x26, x27]
    lsr w3, w0, #4
    cmp w3, #10
    add w4, w3, #48
    add w5, w3, #87
    csel w3, w5, w4, ge
    strb w3, [x24], #1

    and w3, w0, #0xf
    cmp w3, #10
    add w4, w3, #48
    add w5, w3, #87
    csel w3, w5, w4, ge
    strb w3, [x24], #1

    mov w3, #32
    strb w3, [x24], #1
    b hex_check_mid

hex_pad_spaces:
    mov w3, #32
    strb w3, [x24], #1
    strb w3, [x24], #1
    strb w3, [x24], #1

hex_check_mid:
    cmp x27, #7
    bne hex_next
    mov w3, #32
    strb w3, [x24], #1

hex_next:
    add x27, x27, #1
    cmp x27, #16
    blt hex_pos_loop

    // 3. Format ASCII representation
    mov w3, #32
    strb w3, [x24], #1
    mov w3, #124
    strb w3, [x24], #1

    mov x27, #0
ascii_char_loop:
    cmp x27, x25
    bge ascii_done
    ldrb w0, [x26, x27]
    sub w1, w0, #32
    cmp w1, #94
    mov w2, #46
    csel w0, w0, w2, ls
    strb w0, [x24], #1
    add x27, x27, #1
    b ascii_char_loop

ascii_done:
    mov w3, #124
    strb w3, [x24], #1
    mov w3, #10
    strb w3, [x24], #1

    add x19, x19, x25

    adrp x0, out_buf
    add x0, x0, :lo12:out_buf
    sub x22, x24, x0

    b row_loop

done_rows:
    cmp x22, #FLUSH_THRESHOLD
    blo 1f
    bl flush_out
1:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    add x24, x24, x22

    mov w0, w19
    mov w2, #28
2:
    lsr w3, w0, w2
    and w3, w3, #0xf
    cmp w3, #10
    add w4, w3, #48
    add w5, w3, #87
    csel w3, w5, w4, ge
    strb w3, [x24], #1
    subs w2, w2, #4
    bge 2b

    mov w3, #10
    strb w3, [x24], #1

    adrp x0, out_buf
    add x0, x0, :lo12:out_buf
    sub x22, x24, x0

    bl flush_out

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

flush_out:
    cbz x22, 2f
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    mov x2, x22
1:
    mov x0, #1          // fd = stdout
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    ble write_error
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, 1b
    mov x22, #0
2:
    ret

write_error:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
row_data:
    .space 16
in_buf:
    .space BUF_SIZE
out_buf:
    .space BUF_SIZE
