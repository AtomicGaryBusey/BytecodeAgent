.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .balign 16
in_buf:
    .zero 65536
out_buf:
    .zero 131072

.text
_start:
    lea r12, [rip + in_buf]
    mov rbx, 65536

.Lread_loop:
    test rbx, rbx
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, rbx
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    sub rbx, rax
    jmp .Lread_loop

.Lread_done:
    lea rax, [rip + in_buf]
    mov r13, r12
    sub r12, rax
    jz .Lexit_success

    mov rsi, rax
    lea rdi, [rip + out_buf]
    xor r14d, r14d
    lea r15, [rip + b64_table]

.Lencode_loop:
    mov rax, r13
    sub rax, rsi
    cmp rax, 3
    jb .Lremainder_check

    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3
    mov ebx, eax

    shr eax, 18
    call emit_b64

    mov eax, ebx
    shr eax, 12
    and eax, 0x3f
    call emit_b64

    mov eax, ebx
    shr eax, 6
    and eax, 0x3f
    call emit_b64

    mov eax, ebx
    and eax, 0x3f
    call emit_b64

    jmp .Lencode_loop

.Lremainder_check:
    mov rax, r13
    sub rax, rsi
    cmp rax, 1
    je .Lrem1
    cmp rax, 2
    je .Lrem2
    jmp .Lfinish

.Lrem1:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov ebx, eax

    shr eax, 18
    call emit_b64

    mov eax, ebx
    shr eax, 12
    and eax, 0x3f
    call emit_b64

    mov al, '='
    call emit_raw

    mov al, '='
    call emit_raw

    jmp .Lfinish

.Lrem2:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov ebx, eax

    shr eax, 18
    call emit_b64

    mov eax, ebx
    shr eax, 12
    and eax, 0x3f
    call emit_b64

    mov eax, ebx
    shr eax, 6
    and eax, 0x3f
    call emit_b64

    mov al, '='
    call emit_raw

.Lfinish:
    test r14d, r14d
    jz .Lwrite_out
    mov byte ptr [rdi], 10
    inc rdi

.Lwrite_out:
    lea rsi, [rip + out_buf]
    mov rbx, rdi
    sub rbx, rsi

.Lwrite_loop:
    test rbx, rbx
    jz .Lexit_success
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle .Lexit_success
    add rsi, rax
    sub rbx, rax
    jmp .Lwrite_loop

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall

emit_b64:
    movzx eax, al
    mov al, byte ptr [r15 + rax]
emit_raw:
    mov byte ptr [rdi], al
    inc rdi
    inc r14d
    cmp r14d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
1:
    ret
