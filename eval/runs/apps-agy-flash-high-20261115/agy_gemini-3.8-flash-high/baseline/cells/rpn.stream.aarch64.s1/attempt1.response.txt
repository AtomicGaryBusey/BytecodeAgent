.globl _start

.text
.balign 4

_start:
    // Read stdin until EOF (sys_read returns 0)
    mov x19, #0                 // x19 = total bytes read
.Lread_loop:
    mov x8, #63                 // sys_read = 63
    mov x0, #0                  // fd = 0 (stdin)
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    add x1, x1, x19
    movz x2, #2, lsl #16        // capacity: 131072 bytes
    sub x2, x2, x19
    cbz x2, .Lread_done
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x19, x19, x0
    b .Lread_loop

.Lread_done:
    cbz x19, .Lexit_program     // If empty input, exit immediately

    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    add x19, x1, x19            // x19 = in_end (pointer past end of input)
    mov x20, x1                 // x20 = cur_line_start

.Lline_loop:
    cmp x20, x19
    b.ge .Lexit_program

    // Scan for newline '\n' (ASCII 10) starting at cur_line_start
    mov x24, x20
.Lscan_nl:
    cmp x24, x19
    b.ge .Lline_no_nl
    ldrb w0, [x24], #1
    cmp w0, #10
    b.ne .Lscan_nl

    sub x21, x24, #1            // x21 = line_end (points to '\n')
    mov x22, x24                // x22 = next_line_start (after '\n')
    b .Lprocess_line

.Lline_no_nl:
    mov x21, x19                // line_end = in_end
    mov x22, x19                // next_line_start = in_end

.Lprocess_line:
    mov x23, #0                 // eval_sp = 0 (stack depth)
    mov x24, x20                // tok_p = cur_line_start

.Ltoken_loop:
    // Skip separators: space 0x20 (32) and tab 0x09 (9)
    cmp x24, x21
    b.ge .Lline_tokens_done
    ldrb w0, [x24]
    cmp w0, #32
    b.eq .Lskip_sep
    cmp w0, #9
    b.eq .Lskip_sep
    b .Lfound_tok_start

.Lskip_sep:
    add x24, x24, #1
    b .Ltoken_loop

.Lfound_tok_start:
    mov x25, x24                // tok_start = tok_p

.Lscan_tok_end:
    add x24, x24, #1
    cmp x24, x21
    b.ge .Ltok_end_found
    ldrb w0, [x24]
    cmp w0, #32
    b.eq .Ltok_end_found
    cmp w0, #9
    b.eq .Ltok_end_found
    b .Lscan_tok_end

.Ltok_end_found:
    sub x26, x24, x25           // tok_len = tok_p - tok_start

    // Check if token length is 1
    cmp x26, #1
    b.ne .Lcheck_multi_char_token

    ldrb w0, [x25]
    // Single-char operators: '+' (43), '-' (45), '*' (42), '/' (47), '%' (37)
    cmp w0, #43
    b.eq .Lop_common
    cmp w0, #45
    b.eq .Lop_common
    cmp w0, #42
    b.eq .Lop_common
    cmp w0, #47
    b.eq .Lop_common
    cmp w0, #37
    b.eq .Lop_common

    // Single digit '0'..'9'
    sub w1, w0, #48
    cmp w1, #9
    b.hi .Lline_error
    mov x0, x1
    b .Lpush_val

.Lcheck_multi_char_token:
    ldrb w0, [x25]
    cmp w0, #45                 // '-'
    b.ne .Lmulti_pos

    // Negative literal
    sub x1, x26, #1             // num_digits
    cmp x1, #18
    b.hi .Lline_error
    add x2, x25, #1             // pointer to digits
    mov x3, #1                  // is_neg = 1
    b .Lparse_digits

.Lmulti_pos:
    mov x1, x26                 // num_digits
    cmp x1, #18
    b.hi .Lline_error
    mov x2, x25                 // pointer to digits
    mov x3, #0                  // is_neg = 0

.Lparse_digits:
    mov x0, #0                  // accumulated integer value
    mov x4, #10
.Ldigit_loop_parse:
    ldrb w5, [x2], #1
    sub w6, w5, #48
    cmp w6, #9
    b.hi .Lline_error
    madd x0, x0, x4, x6
    subs x1, x1, #1
    b.ne .Ldigit_loop_parse

    cbz x3, .Lpush_val
    neg x0, x0

.Lpush_val:
    adrp x7, eval_stack
    add x7, x7, :lo12:eval_stack
    str x0, [x7, x23, lsl #3]
    add x23, x23, #1
    b .Ltoken_loop

.Lop_common:
    cmp x23, #2
    b.lt .Lline_error
    sub x23, x23, #2
    adrp x7, eval_stack
    add x7, x7, :lo12:eval_stack
    ldr x1, [x7, x23, lsl #3]   // a
    add x8, x23, #1
    ldr x2, [x7, x8, lsl #3]    // b

    cmp w0, #43                 // '+'
    b.eq .Ldo_add
    cmp w0, #45                 // '-'
    b.eq .Ldo_sub
    cmp w0, #42                 // '*'
    b.eq .Ldo_mul
    cmp w0, #47                 // '/'
    b.eq .Ldo_div

    // '%' operator: check b == 0 and overflow (a == INT64_MIN && b == -1)
    cbz x2, .Lline_error
    cmn x2, #1
    b.ne 1f
    movz x9, #0x8000, lsl #48
    cmp x1, x9
    b.eq .Lline_error
1:
    sdiv x0, x1, x2
    msub x0, x0, x2, x1
    b .Lop_store

.Ldo_add:
    add x0, x1, x2
    b .Lop_store

.Ldo_sub:
    sub x0, x1, x2
    b .Lop_store

.Ldo_mul:
    mul x0, x1, x2
    b .Lop_store

.Ldo_div:
    // '/' operator: check b == 0 and overflow (a == INT64_MIN && b == -1)
    cbz x2, .Lline_error
    cmn x2, #1
    b.ne 1f
    movz x9, #0x8000, lsl #48
    cmp x1, x9
    b.eq .Lline_error
1:
    sdiv x0, x1, x2

.Lop_store:
    str x0, [x7, x23, lsl #3]
    add x23, x23, #1
    b .Ltoken_loop

.Lline_tokens_done:
    cmp x23, #1
    b.ne .Lline_error

    adrp x7, eval_stack
    add x7, x7, :lo12:eval_stack
    ldr x0, [x7]
    bl print_int64
    b .Lfinish_line

.Lline_error:
    bl print_error

.Lfinish_line:
    adrp x0, out_len
    ldr w0, [x0, :lo12:out_len]
    mov w1, #32768
    cmp w0, w1
    b.lo 1f
    bl flush_out
1:
    mov x20, x22
    b .Lline_loop

.Lexit_program:
    bl flush_out
    mov x8, #93                 // sys_exit = 93
    mov x0, #0                  // exit status 0
    svc #0

flush_out:
    adrp x9, out_len
    ldr w10, [x9, :lo12:out_len]
    cbz w10, .Lflush_ret

    mov w11, #0
.Lflush_write_loop:
    mov x8, #64                 // sys_write = 64
    mov x0, #1                  // stdout = 1
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    add x1, x1, w11, uxtw
    mov w2, w10
    svc #0
    cmp x0, #0
    b.le .Lflush_clear
    add w11, w11, w0
    sub w10, w10, w0
    cbnz w10, .Lflush_write_loop

.Lflush_clear:
    adrp x9, out_len
    str wzr, [x9, :lo12:out_len]
.Lflush_ret:
    ret

print_error:
    adrp x0, out_len
    ldr w1, [x0, :lo12:out_len]
    adrp x2, out_buf
    add x2, x2, :lo12:out_buf
    add x2, x2, w1, uxtw

    adrp x3, msg_error
    add x3, x3, :lo12:msg_error

    ldrb w4, [x3]
    strb w4, [x2]
    ldrb w4, [x3, #1]
    strb w4, [x2, #1]
    ldrb w4, [x3, #2]
    strb w4, [x2, #2]
    ldrb w4, [x3, #3]
    strb w4, [x2, #3]
    ldrb w4, [x3, #4]
    strb w4, [x2, #4]
    ldrb w4, [x3, #5]
    strb w4, [x2, #5]

    add w1, w1, #6
    str w1, [x0, :lo12:out_len]
    ret

print_int64:
    sub sp, sp, #32
    add x1, sp, #31
    mov w2, #10                 // '\n'
    strb w2, [x1], #-1

    cbnz x0, .Lfmt_nonzero
    mov w2, #48                 // '0'
    strb w2, [x1], #-1
    b .Lfmt_done

.Lfmt_nonzero:
    mov x3, x0
    tbz x0, #63, .Lfmt_digits
    neg x0, x0

.Lfmt_digits:
    mov x4, #10
.Lfmt_loop:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #48
    strb w6, [x1], #-1
    mov x0, x5
    cbnz x0, .Lfmt_loop

    tbz x3, #63, .Lfmt_done
    mov w2, #45                 // '-'
    strb w2, [x1], #-1

.Lfmt_done:
    add x2, x1, #1
    add x3, sp, #32
    sub x3, x3, x2

    adrp x4, out_len
    ldr w5, [x4, :lo12:out_len]
    adrp x6, out_buf
    add x6, x6, :lo12:out_buf
    add x6, x6, w5, uxtw

    mov x7, #0
.Lcopy_loop:
    cmp x7, x3
    b.ge .Lcopy_done
    ldrb w8, [x2, x7]
    strb w8, [x6, x7]
    add x7, x7, #1
    b .Lcopy_loop

.Lcopy_done:
    add w5, w5, w3
    str w5, [x4, :lo12:out_len]

    add sp, sp, #32
    ret

.section .rodata
.balign 4
msg_error:
    .ascii "error\n"

.bss
.balign 16
in_buf:
    .zero 131072
out_buf:
    .zero 65536
eval_stack:
    .zero 8192
.balign 8
out_len:
    .zero 8
