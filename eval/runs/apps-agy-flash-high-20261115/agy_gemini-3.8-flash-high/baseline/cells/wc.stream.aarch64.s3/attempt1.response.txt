.text
.globl _start

_start:
    mov x19, #0                 // lines = 0
    mov x20, #0                 // words = 0
    mov x21, #0                 // bytes = 0
    mov x22, #0                 // in_word = 0

.Lread_loop:
    mov x0, #0                  // fd = 0 (stdin)
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    movz x2, #1, lsl #16        // count = 65536
    mov x8, #63                 // sys_read
    svc #0

    cmp x0, #0
    ble .Ldone_read

    add x21, x21, x0            // total bytes += count

    adrp x23, in_buf
    add x23, x23, :lo12:in_buf
    add x24, x23, x0            // end pointer = in_buf + count

.Lbyte_loop:
    ldrb w2, [x23], #1

    cmp w2, #10
    cinc x19, x19, eq           // if byte == '\n', lines++

    cmp w2, #32
    b.eq .Lspace
    sub w3, w2, #9
    cmp w3, #4
    b.ls .Lspace

    // Non-whitespace (word byte)
    eor x4, x22, #1             // x4 = (in_word == 0) ? 1 : 0
    add x20, x20, x4            // words += x4
    mov x22, #1                 // in_word = 1

.Lnext_byte:
    cmp x23, x24
    b.ne .Lbyte_loop
    b .Lread_loop

.Lspace:
    mov x22, #0                 // in_word = 0
    b .Lnext_byte

.Ldone_read:
    adrp x10, out_buf
    add x10, x10, :lo12:out_buf
    add x10, x10, #128
    mov x12, x10                // x12 = end of output buffer

    // Format bytes, words, lines backwards into out_buf
    // 1. Newline at the end of output
    mov w1, #10
    strb w1, [x10, #-1]!

    // 2. Format bytes (x21)
    mov x0, x21
    mov x4, #10
    cbnz x0, 1f
    mov w1, #48
    strb w1, [x10, #-1]!
    b 2f
1:
    udiv x1, x0, x4
    msub x2, x1, x4, x0
    add w2, w2, #48
    strb w2, [x10, #-1]!
    mov x0, x1
    cbnz x0, 1b
2:

    // 3. Space separator
    mov w1, #32
    strb w1, [x10, #-1]!

    // 4. Format words (x20)
    mov x0, x20
    cbnz x0, 3f
    mov w1, #48
    strb w1, [x10, #-1]!
    b 4f
3:
    udiv x1, x0, x4
    msub x2, x1, x4, x0
    add w2, w2, #48
    strb w2, [x10, #-1]!
    mov x0, x1
    cbnz x0, 3b
4:

    // 5. Space separator
    mov w1, #32
    strb w1, [x10, #-1]!

    // 6. Format lines (x19)
    mov x0, x19
    cbnz x0, 5f
    mov w1, #48
    strb w1, [x10, #-1]!
    b 6f
5:
    udiv x1, x0, x4
    msub x2, x1, x4, x0
    add w2, w2, #48
    strb w2, [x10, #-1]!
    mov x0, x1
    cbnz x0, 5b
6:

    // Write formatted output to stdout (fd 1)
    sub x2, x12, x10            // count = x12 - x10
    mov x1, x10                 // buffer = x10

.Lwrite_loop:
    cbz x2, .Lexit
    mov x0, #1                  // fd = 1 (stdout)
    mov x8, #64                 // sys_write
    svc #0
    cmp x0, #0
    ble .Lexit
    add x1, x1, x0
    sub x2, x2, x0
    b .Lwrite_loop

.Lexit:
    mov x0, #0                  // status = 0
    mov x8, #93                 // sys_exit
    svc #0

    .bss
    .balign 16
in_buf:
    .space 65536
out_buf:
    .space 128
