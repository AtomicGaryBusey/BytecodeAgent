.globl _start

.section .text
_start:
    lea r12, [rip + in_buf]
    mov r13d, 131000

.Lread_loop:
    test r13d, r13d
    jz .Lread_done

    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov edx, r13d
    syscall

    test rax, rax
    jle .Lread_done

    add r12, rax
    sub r13d, eax
    jmp .Lread_loop

.Lread_done:
    lea rsi, [rip + in_buf]
    mov rbx, r12
    lea rdi, [rip + out_buf]

    cmp rsi, rbx
    jae .Lprocess_done

.Lline_loop:
    cmp rsi, rbx
    jae .Lprocess_done

    mov rdx, rsi
.Lscan_nl:
    cmp rdx, rbx
    jae .Lfound_eof_line
    cmp byte ptr [rdx], 10
    je .Lfound_nl
    inc rdx
    jmp .Lscan_nl

.Lfound_nl:
    mov rcx, rdx
.Lrev_nl:
    cmp rcx, rsi
    jbe .Lrev_nl_done
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp .Lrev_nl

.Lrev_nl_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rdx + 1]
    jmp .Lline_loop

.Lfound_eof_line:
    mov rcx, rbx
.Lrev_eof:
    cmp rcx, rsi
    jbe .Lrev_eof_done
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp .Lrev_eof

.Lrev_eof_done:
    mov byte ptr [rdi], 10
    inc rdi
    jmp .Lprocess_done

.Lprocess_done:
    lea rsi, [rip + out_buf]
    sub rdi, rsi
    test rdi, rdi
    jz .Lexit_success

    mov r12, rsi
    mov r13, rdi

.Lwrite_loop:
    test r13, r13
    jz .Lexit_success

    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall

    test rax, rax
    jle .Lexit_success

    add r12, rax
    sub r13, rax
    jmp .Lwrite_loop

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.balign 16
in_buf:
    .zero 131072
out_buf:
    .zero 131072
