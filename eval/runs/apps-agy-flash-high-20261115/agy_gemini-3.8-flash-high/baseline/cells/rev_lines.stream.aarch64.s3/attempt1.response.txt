.globl _start
.text

_start:
    // x19 = in_buf base
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf

    mov x20, #0                 // x20 = total bytes read

read_loop:
    mov x0, #0                  // fd = 0 (stdin)
    add x1, x19, x20            // buf = in_buf + total_read
    movz x3, #2, lsl #16        // x3 = 131072 (buffer size)
    sub x2, x3, x20             // count = buffer remaining space
    cbz x2, read_done           // buffer full
    mov x8, #63                 // sys_read = 63
    svc #0
    cmp x0, #0
    b.le read_done              // <= 0 means EOF or error
    add x20, x20, x0
    b read_loop

read_done:
    // x21 = out_buf base
    adrp x21, out_buf
    add x21, x21, :lo12:out_buf

    mov x22, x21                // x22 = out_buf write pointer
    mov x23, x19                // x23 = line_start pointer
    mov x24, x19                // x24 = scan pointer
    add x25, x19, x20           // x25 = end of input pointer

scan_loop:
    cmp x24, x25
    b.hs scan_done
    ldrb w0, [x24], #1          // load byte, advance scan pointer
    cmp w0, #10                 // check for '\n'
    b.ne scan_loop

    // Found newline at (x24 - 1)
    sub x1, x24, #1             // x1 = pointer to newline
    sub x2, x1, x23             // x2 = length of line
    cbz x2, 2f                  // if line is empty, skip reversal

1:
    sub x2, x2, #1
    ldrb w0, [x23, x2]
    strb w0, [x22], #1
    cbnz x2, 1b

2:
    mov w0, #10
    strb w0, [x22], #1
    mov x23, x24                // next line starts after the newline
    b scan_loop

scan_done:
    // If there are bytes remaining after the last newline, output them as the final line
    cmp x23, x25
    b.hs write_prep

    sub x2, x25, x23            // x2 = length of final line
    cbz x2, write_prep

3:
    sub x2, x2, #1
    ldrb w0, [x23, x2]
    strb w0, [x22], #1
    cbnz x2, 3b

    mov w0, #10
    strb w0, [x22], #1

write_prep:
    sub x23, x22, x21           // x23 = total bytes to write
    mov x24, #0                 // x24 = bytes written so far

write_loop:
    cmp x24, x23
    b.hs exit_success
    mov x0, #1                  // fd = 1 (stdout)
    add x1, x21, x24            // buf = out_buf + bytes_written
    sub x2, x23, x24            // count = remaining bytes
    mov x8, #64                 // sys_write = 64
    svc #0
    cmp x0, #0
    b.le exit_success           // exit on write error
    add x24, x24, x0
    b write_loop

exit_success:
    mov x0, #0                  // exit status 0
    mov x8, #93                 // sys_exit = 93
    svc #0

.section .bss
.balign 16
in_buf:
    .space 131072
out_buf:
    .space 131072
