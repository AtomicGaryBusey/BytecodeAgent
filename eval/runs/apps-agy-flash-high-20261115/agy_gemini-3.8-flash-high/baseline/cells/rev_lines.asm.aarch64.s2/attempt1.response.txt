.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ FD_STDIN, 0
.equ FD_STDOUT, 1

    .text
    .globl _start

_start:
    adrp    x19, in_buf
    add     x19, x19, :lo12:in_buf

    adrp    x21, out_buf
    add     x21, x21, :lo12:out_buf

    mov     x20, #0

read_loop:
    movz    x2, #2, lsl #16
    sub     x2, x2, x20
    cbz     x2, read_done

    mov     x0, #FD_STDIN
    add     x1, x19, x20
    mov     x8, #SYS_read
    svc     #0

    cmp     x0, #0
    ble     read_done

    add     x20, x20, x0
    b       read_loop

read_done:
    mov     x22, #0
    mov     x23, #0

line_loop:
    cmp     x23, x20
    bge     process_done

    mov     x24, x23

scan_loop:
    cmp     x24, x20
    bge     found_eof

    ldrb    w26, [x19, x24]
    cmp     w26, #10
    beq     found_newline

    add     x24, x24, #1
    b       scan_loop

found_newline:
    mov     x25, x24
    add     x27, x24, #1
    b       rev_loop

found_eof:
    mov     x25, x24
    mov     x27, x24

rev_loop:
    cmp     x25, x23
    ble     rev_done
    sub     x25, x25, #1
    ldrb    w26, [x19, x25]
    strb    w26, [x21, x22]
    add     x22, x22, #1
    b       rev_loop

rev_done:
    mov     w26, #10
    strb    w26, [x21, x22]
    add     x22, x22, #1

    mov     x23, x27
    b       line_loop

process_done:
    mov     x28, #0

write_loop:
    cmp     x28, x22
    bge     exit_all

    mov     x0, #FD_STDOUT
    add     x1, x21, x28
    sub     x2, x22, x28
    mov     x8, #SYS_write
    svc     #0

    cmp     x0, #0
    ble     exit_all

    add     x28, x28, x0
    b       write_loop

exit_all:
    mov     x0, #0
    mov     x8, #SYS_exit
    svc     #0

    .bss
    .balign 16
in_buf:
    .zero 131072
out_buf:
    .zero 131072
