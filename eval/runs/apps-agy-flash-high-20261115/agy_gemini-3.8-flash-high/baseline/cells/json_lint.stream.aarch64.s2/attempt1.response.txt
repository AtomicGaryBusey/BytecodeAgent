.text
.globl _start

_start:
    adrp    x21, input_buf
    add     x21, x21, :lo12:input_buf
    mov     x22, #0

read_loop:
    mov     x0, #0
    add     x1, x21, x22
    movz    w2, #2, lsl #16
    sub     w2, w2, w22
    cbz     w2, read_done
    uxtw    x2, w2
    mov     x8, #63
    svc     #0

    cmp     x0, #0
    b.le    read_done

    add     x22, x22, x0
    b       read_loop

read_done:
    mov     x19, x21
    add     x20, x21, x22

    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    bl      parse_value

    bl      skip_ws
    cmp     x19, x20
    b.ne    err_cur

    b       success_exit

skip_ws:
1:
    cmp     x19, x20
    b.hs    2f
    ldrb    w0, [x19]
    cmp     w0, #0x20
    b.eq    3f
    cmp     w0, #0x0A
    b.eq    3f
    cmp     w0, #0x0D
    b.eq    3f
    cmp     w0, #0x09
    b.eq    3f
    ret
3:
    add     x19, x19, #1
    b       1b
2:
    ret

parse_value:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    cmp     x19, x20
    b.hs    err_eof

    ldrb    w0, [x19]
    cmp     w0, #'{'
    b.eq    val_obj
    cmp     w0, #'['
    b.eq    val_arr
    cmp     w0, #0x22
    b.eq    val_str
    cmp     w0, #'t'
    b.eq    val_true
    cmp     w0, #'f'
    b.eq    val_false
    cmp     w0, #'n'
    b.eq    val_null
    cmp     w0, #'-'
    b.eq    val_num
    cmp     w0, #'0'
    b.lo    err_cur
    cmp     w0, #'9'
    b.ls    val_num

    b       err_cur

val_obj:
    add     x19, x19, #1
    bl      parse_object
    ldp     x29, x30, [sp], #16
    ret

val_arr:
    add     x19, x19, #1
    bl      parse_array
    ldp     x29, x30, [sp], #16
    ret

val_str:
    bl      parse_string
    ldp     x29, x30, [sp], #16
    ret

val_num:
    bl      parse_number
    ldp     x29, x30, [sp], #16
    ret

val_true:
    bl      parse_true
    ldp     x29, x30, [sp], #16
    ret

val_false:
    bl      parse_false
    ldp     x29, x30, [sp], #16
    ret

val_null:
    bl      parse_null
    ldp     x29, x30, [sp], #16
    ret

parse_array:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    ldrb    w0, [x19]
    cmp     w0, #']'
    b.eq    arr_empty

arr_elem_loop:
    bl      parse_value
    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    ldrb    w0, [x19]
    cmp     w0, #']'
    b.eq    arr_close
    cmp     w0, #','
    b.ne    err_cur

    add     x19, x19, #1
    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    b       arr_elem_loop

arr_empty:
arr_close:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret

parse_object:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    ldrb    w0, [x19]
    cmp     w0, #'}'
    b.eq    obj_empty

obj_member_loop:
    ldrb    w0, [x19]
    cmp     w0, #0x22
    b.ne    err_cur

    bl      parse_string
    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    ldrb    w0, [x19]
    cmp     w0, #':'
    b.ne    err_cur
    add     x19, x19, #1

    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    bl      parse_value
    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    ldrb    w0, [x19]
    cmp     w0, #'}'
    b.eq    obj_close
    cmp     w0, #','
    b.ne    err_cur

    add     x19, x19, #1
    bl      skip_ws
    cmp     x19, x20
    b.hs    err_eof

    b       obj_member_loop

obj_empty:
obj_close:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret

parse_string:
    add     x19, x19, #1

str_loop:
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w0, [x19]
    cmp     w0, #0x22
    b.eq    str_done
    cmp     w0, #0x5C
    b.eq    str_esc
    cmp     w0, #0x20
    b.lo    err_cur
    add     x19, x19, #1
    b       str_loop

str_done:
    add     x19, x19, #1
    ret

str_esc:
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w1, [x19]
    cmp     w1, #0x22
    b.eq    esc_ok
    cmp     w1, #0x5C
    b.eq    esc_ok
    cmp     w1, #'/'
    b.eq    esc_ok
    cmp     w1, #'b'
    b.eq    esc_ok
    cmp     w1, #'f'
    b.eq    esc_ok
    cmp     w1, #'n'
    b.eq    esc_ok
    cmp     w1, #'r'
    b.eq    esc_ok
    cmp     w1, #'t'
    b.eq    esc_ok
    cmp     w1, #'u'
    b.eq    esc_u
    b       err_cur

esc_ok:
    add     x19, x19, #1
    b       str_loop

esc_u:
    add     x19, x19, #1
    mov     w3, #4

u_hex_loop:
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w2, [x19]
    cmp     w2, #'0'
    b.lo    err_cur
    cmp     w2, #'9'
    b.ls    hex_ok
    cmp     w2, #'A'
    b.lo    err_cur
    cmp     w2, #'F'
    b.ls    hex_ok
    cmp     w2, #'a'
    b.lo    err_cur
    cmp     w2, #'f'
    b.hi    err_cur

hex_ok:
    add     x19, x19, #1
    subs    w3, w3, #1
    b.ne    u_hex_loop
    b       str_loop

parse_number:
    ldrb    w0, [x19]
    cmp     w0, #'-'
    b.ne    num_int

    add     x19, x19, #1
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w0, [x19]
    cmp     w0, #'0'
    b.lo    err_cur
    cmp     w0, #'9'
    b.hi    err_cur

num_int:
    cmp     w0, #'0'
    b.ne    num_nonzero

    add     x19, x19, #1
    cmp     x19, x20
    b.hs    num_done_check_frac
    ldrb    w1, [x19]
    cmp     w1, #'0'
    b.lo    num_done_check_frac
    cmp     w1, #'9'
    b.ls    err_cur
    b       num_done_check_frac

num_nonzero:
    add     x19, x19, #1

int_digits_loop:
    cmp     x19, x20
    b.hs    num_done_check_frac
    ldrb    w1, [x19]
    cmp     w1, #'0'
    b.lo    num_done_check_frac
    cmp     w1, #'9'
    b.hi    num_done_check_frac
    add     x19, x19, #1
    b       int_digits_loop

num_done_check_frac:
    cmp     x19, x20
    b.hs    num_finished
    ldrb    w0, [x19]
    cmp     w0, #'.'
    b.ne    num_check_exp

    add     x19, x19, #1
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w0, [x19]
    cmp     w0, #'0'
    b.lo    err_cur
    cmp     w0, #'9'
    b.hi    err_cur
    add     x19, x19, #1

frac_digits_loop:
    cmp     x19, x20
    b.hs    num_finished
    ldrb    w1, [x19]
    cmp     w1, #'0'
    b.lo    num_check_exp
    cmp     w1, #'9'
    b.hi    num_check_exp
    add     x19, x19, #1
    b       frac_digits_loop

num_check_exp:
    cmp     x19, x20
    b.hs    num_finished
    ldrb    w0, [x19]
    cmp     w0, #'e'
    b.eq    1f
    cmp     w0, #'E'
    b.ne    num_finished
1:
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w0, [x19]
    cmp     w0, #'+'
    b.eq    2f
    cmp     w0, #'-'
    b.ne    3f
2:
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    err_eof
    ldrb    w0, [x19]
3:
    cmp     w0, #'0'
    b.lo    err_cur
    cmp     w0, #'9'
    b.hi    err_cur
    add     x19, x19, #1

exp_digits_loop:
    cmp     x19, x20
    b.hs    num_finished
    ldrb    w1, [x19]
    cmp     w1, #'0'
    b.lo    num_finished
    cmp     w1, #'9'
    b.hi    num_finished
    add     x19, x19, #1
    b       exp_digits_loop

num_finished:
    ret

parse_true:
    add     x19, x19, #1
