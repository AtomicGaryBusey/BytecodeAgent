.section .rodata
.balign 8
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "

.section .bss
.balign 16
input_buf:
    .space 131072
out_buf:
    .space 64

.text
.globl _start

_start:
    // Read stdin into input_buf
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x19, x21
    movz x2, #2, lsl #16
    sub x2, x2, x21
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    b.lt read_error
    add x21, x21, x0
    b read_loop

read_error:
    mov x0, #1
    mov x8, #93
    svc #0

read_done:
    mov x20, #0
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos

    bl parse_value

    bl skip_ws
    cmp x20, x21
    b.ne error_at_pos

success:
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    bl write_stdout
    mov x0, #0
    mov x8, #93
    svc #0

error_at_pos:
    adrp x10, out_buf
    add x10, x10, :lo12:out_buf

    adrp x11, msg_err
    add x11, x11, :lo12:msg_err
    ldr x12, [x11]
    str x12, [x10]
    ldrb w12, [x11, #8]
    strb w12, [x10, #8]
    add x10, x10, #9

    cbnz x20, 1f
    mov w12, #'0'
    strb w12, [x10], #1
    b 3f

1:
    sub sp, sp, #32
    add x13, sp, #32
    mov x14, #10

2:
    udiv x15, x20, x14
    msub x16, x15, x14, x20
    add w16, w16, #'0'
    sub x13, x13, #1
    strb w16, [x13]
    mov x20, x15
    cbnz x20, 2b

4:
    ldrb w16, [x13], #1
    strb w16, [x10], #1
    add x17, sp, #32
    cmp x13, x17
    b.ne 4b

    add sp, sp, #32

3:
    mov w12, #'\n'
    strb w12, [x10], #1

    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    sub x2, x10, x1
    bl write_stdout

    mov x0, #1
    mov x8, #93
    svc #0

write_stdout:
    cbz x2, 2f
1:
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, 1b
2:
    ret

skip_ws:
    cmp x20, x21
    b.ge 1f
    ldrb w0, [x19, x20]
    cmp w0, #0x20
    b.eq 2f
    cmp w0, #0x09
    b.eq 2f
    cmp w0, #0x0A
    b.eq 2f
    cmp w0, #0x0D
    b.eq 2f
1:
    ret
2:
    add x20, x20, #1
    b skip_ws

parse_value:
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq parse_string
    cmp w0, #'{'
    b.eq parse_object
    cmp w0, #'['
    b.eq parse_array
    cmp w0, #'t'
    b.eq parse_true
    cmp w0, #'f'
    b.eq parse_false
    cmp w0, #'n'
    b.eq parse_null
    cmp w0, #'-'
    b.eq parse_number
    sub w1, w0, #'0'
    cmp w1, #10
    b.lo parse_number
    b error_at_pos

parse_true:
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'r'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'u'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.ne error_at_pos

    add x20, x20, #1
    ret

parse_false:
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'a'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'l'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'s'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.ne error_at_pos

    add x20, x20, #1
    ret

parse_null:
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'u'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'l'
    b.ne error_at_pos

    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'l'
    b.ne error_at_pos

    add x20, x20, #1
    ret

parse_string:
    add x20, x20, #1
string_loop:
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq string_done
    cmp w0, #'\\'
    b.eq string_escape
    cmp w0, #0x20
    b.lo error_at_pos
    add x20, x20, #1
    b string_loop

string_done:
    add x20, x20, #1
    ret

string_escape:
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq esc_ok
    cmp w0, #'\\'
    b.eq esc_ok
    cmp w0, #'/'
    b.eq esc_ok
    cmp w0, #'b'
    b.eq esc_ok
    cmp w0, #'f'
    b.eq esc_ok
    cmp w0, #'n'
    b.eq esc_ok
    cmp w0, #'r'
    b.eq esc_ok
    cmp w0, #'t'
    b.eq esc_ok
    cmp w0, #'u'
    b.eq string_unicode
    b error_at_pos

esc_ok:
    add x20, x20, #1
    b string_loop

string_unicode:
    add x20, x20, #1
    mov w2, #4
hex_loop:
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.lo hex_ok
    sub w1, w0, #'a'
    cmp w1, #6
    b.lo hex_ok
    sub w1, w0, #'A'
    cmp w1, #6
    b.lo hex_ok
    b error_at_pos

hex_ok:
    add x20, x20, #1
    subs w2, w2, #1
    b.ne hex_loop
    b string_loop

parse_number:
    ldrb w0, [x19, x20]
    cmp w0, #'-'
    b.ne num_int
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs error_at_pos

num_int:
    cmp w0, #'0'
    b.ne num_nonzero_int
    add x20, x20, #1
    cmp x20, x21
    b.ge num_frac_exp
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.lo error_at_pos
    b num_frac_exp

num_nonzero_int:
    add x20, x20, #1
num_int_loop:
    cmp x20, x21
    b.ge num_frac_exp
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs num_frac_exp
    add x20, x20, #1
    b num_int_loop

num_frac_exp:
    cmp x20, x21
    b.ge num_done
    ldrb w0, [x19, x20]
    cmp w0, #'.'
    b.ne num_check_exp
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs error_at_pos
    add x20, x20, #1
num_frac_loop:
    cmp x20, x21
    b.ge num_done
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs num_check_exp
    add x20, x20, #1
    b num_frac_loop

num_check_exp:
    cmp x20, x21
    b.ge num_done
    ldrb w0, [x19, x20]
    cmp w0, #'e'
    b.eq 1f
    cmp w0, #'E'
    b.ne num_done

1:
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'+'
    b.eq 2f
    cmp w0, #'-'
    b.ne 3f
2:
    add x20, x20, #1
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
3:
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs error_at_pos
    add x20, x20, #1
num_exp_loop:
    cmp x20, x21
    b.ge num_done
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs num_done
    add x20, x20, #1
    b num_exp_loop

num_done:
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #']'
    b.ne 1f
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

1:
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #']'
    b.ne 2f
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

2:
    cmp w0, #','
    b.ne error_at_pos
    add x20, x20, #1
    bl skip_ws
    b 1b

parse_object:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'}'
    b.ne 1f
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

1:
    cmp w0, #'"'
    b.ne error_at_pos
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #':'
    b.ne error_at_pos
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    cmp w0, #'}'
    b.ne 2f
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

2:
    cmp w0, #','
    b.ne error_at_pos
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.ge error_at_pos
    ldrb w0, [x19, x20]
    b 1b
