.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.balign 16
in_buf:
    .space 65536
out_buf:
    .space 131072

.text
_start:
    xor ebx, ebx

read_loop:
    mov edx, 65536
    sub edx, ebx
    jz read_done
    lea rsi, [rip + in_buf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_clean

    lea rsi, [rip + in_buf]
    lea rdi, [rip + out_buf]
    lea r10, [rip + b64_table]
    xor r9d, r9d
    mov r8, rbx

enc_loop:
    cmp r8, 3
    jb enc_remainder

    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3f
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3f
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 0x3f
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    sub r8, 3

    inc r9d
    cmp r9d, 19
    jne enc_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp enc_loop

enc_remainder:
    test r8, r8
    jz check_final_nl

    cmp r8, 1
    jne rem_two_bytes

    movzx eax, byte ptr [rsi]
    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    and edx, 3
    shl edx, 4
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], 0x3d
    mov byte ptr [rdi + 3], 0x3d
    add rdi, 4
    inc r9d
    jmp handle_line_end

rem_two_bytes:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3f
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3f
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 0x3d
    add rdi, 4
    inc r9d

handle_line_end:
    cmp r9d, 19
    jne check_final_nl
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d

check_final_nl:
    test r9d, r9d
    jz encode_done
    mov byte ptr [rdi], 10
    inc rdi

encode_done:
    lea rdx, [rip + out_buf]
    mov r12, rdi
    sub r12, rdx
    xor ebx, ebx

write_loop:
    cmp ebx, r12d
    jae exit_clean
    mov edx, r12d
    sub edx, ebx
    lea rsi, [rip + out_buf]
    add rsi, rbx
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle exit_clean
    add rbx, rax
    jmp write_loop

exit_clean:
    xor edi, edi
    mov eax, 60
    syscall
