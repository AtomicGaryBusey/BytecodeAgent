.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
in_buf:
    .space 65536
out_buf:
    .space 131072

.text
_start:
    xor r12, r12

.Lread_loop:
    mov edx, 65536
    sub edx, r12d
    jle .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    test r12, r12
    jz .Lexit_success

    mov rax, r12
    xor edx, edx
    mov ecx, 3
    div rcx
    mov r13, rax
    mov r14, rdx

    lea rsi, [rip + in_buf]
    lea rdi, [rip + out_buf]
    lea r15, [rip + b64_table]
    xor ebx, ebx

    test r13, r13
    jz .Lblocks_done

.Lblock_loop:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3f
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3f
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 0x3f
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi + 3], dl

    add rdi, 4
    add ebx, 4
    cmp ebx, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor ebx, ebx
1:
    dec r13
    jnz .Lblock_loop

.Lblocks_done:
    test r14, r14
    jz .Lcheck_final_newline

    cmp r14, 1
    je .Lrem_1

.Lrem_2:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3f
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 0x3f
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 61
    jmp .Lrem_finish

.Lrem_1:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 0x3f
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61

.Lrem_finish:
    add rdi, 4
    add ebx, 4
    cmp ebx, 76
    jne .Lcheck_final_newline
    mov byte ptr [rdi], 10
    inc rdi
    xor ebx, ebx

.Lcheck_final_newline:
    test ebx, ebx
    jz .Lflush_output
    mov byte ptr [rdi], 10
    inc rdi

.Lflush_output:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jle .Lexit_success

.Lwrite_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lexit_success
    add rsi, rax
    sub rdx, rax
    jnz .Lwrite_loop

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall
