.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    movz x25, #0x3e00
    movk x25, #1, lsl #32

.Lread_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    movz x2, #1, lsl #16
    mov x8, #SYS_read
    svc #0

    cmp x0, #0
    b.le .Lread_done

    add x21, x21, x0

    adrp x23, buf
    add x23, x23, :lo12:buf
    add x24, x23, x0

.Lbyte_loop:
    ldrb w26, [x23], #1
    cmp w26, #10
    cinc x19, x19, eq

    cmp w26, #32
    b.hi .Lword_char
    lsr x27, x25, x26
    tbz x27, #0, .Lword_char

    mov x22, #0
    b .Lnext_byte

.Lword_char:
    cbnz x22, .Lnext_byte
    add x20, x20, #1
    mov x22, #1

.Lnext_byte:
    cmp x23, x24
    b.ne .Lbyte_loop
    b .Lread_loop

.Lread_done:
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    mov x23, x1

    mov x0, x19
    bl format_uint

    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl format_uint

    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl format_uint

    mov w2, #10
    strb w2, [x1], #1

    sub x2, x1, x23
    mov x1, x23

.Lwrite_loop:
    cbz x2, .Lexit
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le .Lexit
    add x1, x1, x0
    sub x2, x2, x0
    b .Lwrite_loop

.Lexit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

format_uint:
    cbnz x0, 1f
    mov w2, #48
    strb w2, [x1], #1
    ret
1:
    sub sp, sp, #32
    add x5, sp, #32
    mov x3, #10
2:
    udiv x2, x0, x3
    msub x4, x2, x3, x0
    add w4, w4, #48
    strb w4, [x5, #-1]!
    mov x0, x2
    cbnz x0, 2b

    add x6, sp, #32
3:
    ldrb w4, [x5], #1
    strb w4, [x1], #1
    cmp x5, x6
    b.ne 3b

    add sp, sp, #32
    ret

.section .bss
.balign 16
buf:
    .space 65536
out_buf:
    .space 128
