.globl _start

.section .text
_start:
    cld

    # Read standard input into input_buf
    xor r12, r12
.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz .Lread_done
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread_loop

.Lread_done:
    test r12, r12
    jz .Lexit_success

    # Parse lines from input_buf
    lea rsi, [rip + input_buf]
    lea r13, [rsi + r12]
    xor r14, r14
    lea r15, [rip + lines]
    mov rbx, rsi

.Lparse_loop:
    cmp rsi, r13
    je .Lparse_check_tail
    mov al, byte ptr [rsi]
    cmp al, 10
    jne .Lparse_next

    mov rdx, rsi
    sub rdx, rbx
    mov qword ptr [r15 + r14*16], rbx
    mov qword ptr [r15 + r14*16 + 8], rdx
    inc r14
    lea rbx, [rsi + 1]

.Lparse_next:
    inc rsi
    jmp .Lparse_loop

.Lparse_check_tail:
    cmp rbx, r13
    je .Lparse_done
    mov rdx, r13
    sub rdx, rbx
    mov qword ptr [r15 + r14*16], rbx
    mov qword ptr [r15 + r14*16 + 8], rdx
    inc r14

.Lparse_done:
    test r14, r14
    jz .Lexit_success

    # Shell sort lines
    mov rbx, r14
    shr rbx, 1

.Lgap_loop:
    test rbx, rbx
    jz .Lsort_done

    mov r12, rbx
.Li_loop:
    cmp r12, r14
    jge .Lnext_gap

    mov r8, qword ptr [r15 + r12*16]
    mov r9, qword ptr [r15 + r12*16 + 8]
    mov r13, r12

.Lj_loop:
    cmp r13, rbx
    jb .Lj_done

    mov rcx, r13
    sub rcx, rbx

    mov rsi, qword ptr [r15 + rcx*16]
    mov rdx, qword ptr [r15 + rcx*16 + 8]

    mov rax, rdx
    cmp rax, r9
    cmova rax, r9

    xor rcx, rcx
.Lcmp_bytes:
    cmp rcx, rax
    je .Lcmp_len
    movzx edi, byte ptr [rsi + rcx]
    movzx r10d, byte ptr [r8 + rcx]
    cmp edi, r10d
    jne .Lcmp_diff
    inc rcx
    jmp .Lcmp_bytes

.Lcmp_diff:
    ja .Lshift_element
    jmp .Lj_done

.Lcmp_len:
    cmp rdx, r9
    ja .Lshift_element
    jmp .Lj_done

.Lshift_element:
    mov rcx, r13
    sub rcx, rbx
    mov rax, qword ptr [r15 + rcx*16]
    mov rdx, qword ptr [r15 + rcx*16 + 8]
    mov qword ptr [r15 + r13*16], rax
    mov qword ptr [r15 + r13*16 + 8], rdx

    sub r13, rbx
    jmp .Lj_loop

.Lj_done:
    mov qword ptr [r15 + r13*16], r8
    mov qword ptr [r15 + r13*16 + 8], r9

    inc r12
    jmp .Li_loop

.Lnext_gap:
    shr rbx, 1
    jmp .Lgap_loop

.Lsort_done:
    # Build output buffer
    lea rdi, [rip + out_buf]
    xor r12, r12
.Lbuild_loop:
    cmp r12, r14
    jge .Lbuild_done

    mov rsi, qword ptr [r15 + r12*16]
    mov rcx, qword ptr [r15 + r12*16 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi

    inc r12
    jmp .Lbuild_loop

.Lbuild_done:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi

.Lwrite_loop:
    test rdx, rdx
    jle .Lexit_success
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lexit_success
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite_loop

.Lexit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 16
input_buf:
    .space 65536
out_buf:
    .space 65536
lines:
    .space 65536
