export function l $reverse(l %a, l %n) {
@start
    %cond0 =w csgtl %n, 1
    jnz %cond0, @init, @done
@init
    %n1 =l sub %n, 1
    %bytes =l mul %n1, 8
    %p2_init =l add %a, %bytes
    jmp @loop
@loop
    %p1 =l phi @init %a, @body %p1_next
    %p2 =l phi @init %p2_init, @body %p2_next
    %cond =w cultl %p1, %p2
    jnz %cond, @body, @done
@body
    %v1 =l loadl %p1
    %v2 =l loadl %p2
    storel %v2, %p1
    storel %v1, %p2
    %p1_next =l add %p1, 8
    %p2_next =l sub %p2, 8
    jmp @loop
@done
    ret 0
}
