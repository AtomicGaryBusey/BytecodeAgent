export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @zero_cond
@zero_cond
    %j_init =l phi @start 0, @zero_body %j_init_next
    %c_init =w cslel %j_init, %n
    jnz %c_init, @zero_body, @outer_cond
@zero_body
    %init_off =l mul %j_init, 8
    %init_ptr =l add %row, %init_off
    storel 0, %init_ptr
    %j_init_next =l add %j_init, 1
    jmp @zero_cond
@outer_cond
    %i =l phi @zero_cond 0, @inner_done %i_next
    %c_outer =w csltl %i, %m
    jnz %c_outer, @outer_body, @done
@outer_body
    %x_ptr =l add %x, %i
    %xi =w loadub %x_ptr
    jmp @inner_cond
@inner_cond
    %j =l phi @outer_body 1, @inner_store %j_next
    %diag =l phi @outer_body 0, @inner_store %up
    %left =l phi @outer_body 0, @inner_store %val
    %c_inner =w cslel %j, %n
    jnz %c_inner, @inner_body, @inner_done
@inner_body
    %j_off =l mul %j, 8
    %row_j =l add %row, %j_off
    %up =l loadl %row_j
    %jm1 =l sub %j, 1
    %y_ptr =l add %y, %jm1
    %yj =w loadub %y_ptr
    %match =w ceqw %xi, %yj
    jnz %match, @inner_match, @inner_nomatch
@inner_match
    %val_match =l add %diag, 1
    jmp @inner_store
@inner_nomatch
    %up_ge =w csgel %up, %left
    jnz %up_ge, @inner_up, @inner_left
@inner_up
    jmp @inner_store
@inner_left
    jmp @inner_store
@inner_store
    %val =l phi @inner_match %val_match, @inner_up %up, @inner_left %left
    storel %val, %row_j
    %j_next =l add %j, 1
    jmp @inner_cond
@inner_done
    %i_next =l add %i, 1
    jmp @outer_cond
@done
    %n_off =l mul %n, 8
    %row_n =l add %row, %n_off
    %res =l loadl %row_n
    ret %res
}
