export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %n2 =l mul %n, %n
    %n2x2 =l mul %n2, 2
    %np1 =l add %n, 1
    jmp @zero_out_cond
@zero_out_cond
    %z_i =l phi @start 0, @zero_out_body %z_next
    %z_cmp =w csltl %z_i, %n2
    jnz %z_cmp, @zero_out_body, @diag_cond
@zero_out_body
    %z_off =l mul %z_i, 8
    %z_ptr =l add %out, %z_off
    storel 0, %z_ptr
    %z_next =l add %z_i, 1
    jmp @zero_out_cond
@diag_cond
    %d_i =l phi @zero_out_cond 0, @diag_body %d_next
    %d_cmp =w csltl %d_i, %n
    jnz %d_cmp, @diag_body, @step_cond
@diag_body
    %d_idx =l mul %d_i, %np1
    %d_off =l mul %d_idx, 8
    %d_ptr =l add %out, %d_off
    storel 1, %d_ptr
    %d_next =l add %d_i, 1
    jmp @diag_cond
@step_cond
    %step =l phi @diag_cond 0, @step_inc %step_next
    %step_cmp =w csltl %step, %k
    jnz %step_cmp, @i_cond, @clear_tmp_cond
@i_cond
    %i =l phi @step_cond 0, @i_inc %i_next
    %i_cmp =w csltl %i, %n
    jnz %i_cmp, @i_body, @copy_cond
@i_body
    %in =l mul %i, %n
    jmp @j_cond
@j_cond
    %j =l phi @i_body 0, @store_tmp %j_next
    %j_cmp =w csltl %j, %n
    jnz %j_cmp, @t_cond, @i_inc
@t_cond
    %t =l phi @j_cond 0, @t_body %t_next
    %sum =l phi @j_cond 0, @t_body %sum_next
    %t_cmp =w csltl %t, %n
    jnz %t_cmp, @t_body, @store_tmp
@t_body
    %out_idx =l add %in, %t
    %out_off =l mul %out_idx, 8
    %out_ptr =l add %out, %out_off
    %out_val =l loadl %out_ptr
    %tn =l mul %t, %n
    %a_idx =l add %tn, %j
    %a_off =l mul %a_idx, 8
    %a_ptr =l add %a, %a_off
    %a_val =l loadl %a_ptr
    %prod =l mul %out_val, %a_val
    %sum_next =l add %sum, %prod
    %t_next =l add %t, 1
    jmp @t_cond
@store_tmp
    %tmp_idx =l add %in, %j
    %tmp_off =l mul %tmp_idx, 8
    %tmp_ptr =l add %tmp, %tmp_off
    storel %sum, %tmp_ptr
    %j_next =l add %j, 1
    jmp @j_cond
@i_inc
    %i_next =l add %i, 1
    jmp @i_cond
@copy_cond
    %c_idx =l phi @i_cond 0, @copy_body %c_next
    %c_cmp =w csltl %c_idx, %n2
    jnz %c_cmp, @copy_body, @step_inc
@copy_body
    %c_off =l mul %c_idx, 8
    %c_tptr =l add %tmp, %c_off
    %c_val =l loadl %c_tptr
    %c_optr =l add %out, %c_off
    storel %c_val, %c_optr
    %c_next =l add %c_idx, 1
    jmp @copy_cond
@step_inc
    %step_next =l add %step, 1
    jmp @step_cond
@clear_tmp_cond
    %cl_idx =l phi @step_cond 0, @clear_tmp_body %cl_next
    %cl_cmp =w csltl %cl_idx, %n2x2
    jnz %cl_cmp, @clear_tmp_body, @done
@clear_tmp_body
    %cl_off =l mul %cl_idx, 8
    %cl_ptr =l add %tmp, %cl_off
    storel 0, %cl_ptr
    %cl_next =l add %cl_idx, 1
    jmp @clear_tmp_cond
@done
    ret 0
}
