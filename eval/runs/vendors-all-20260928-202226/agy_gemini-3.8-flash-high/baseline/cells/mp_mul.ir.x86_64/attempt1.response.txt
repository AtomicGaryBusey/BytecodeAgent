export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%zi_0 =l copy 0
	jmp @zero_loop
@zero_loop
	%zi =l phi @start %zi_0, @zero_body %zi_next
	%zcond =w csltl %zi, %n
	jnz %zcond, @zero_body, @zero_done
@zero_body
	%zbytes =l shl %zi, 3
	%zr_ptr =l add %r, %zbytes
	storel 0, %zr_ptr
	%zi_next =l add %zi, 1
	jmp @zero_loop
@zero_done
	%na_zero =w ceql %na, 0
	jnz %na_zero, @ret_zero, @check_nb
@check_nb
	%nb_zero =w ceql %nb, 0
	jnz %nb_zero, @ret_zero, @mul_init
@ret_zero
	ret 0
@mul_init
	%i_0 =l copy 0
	jmp @outer_loop
@outer_loop
	%i =l phi @mul_init %i_0, @inner_done %i_next
	%cond_i =w csltl %i, %na
	jnz %cond_i, @outer_body, @outer_done
@outer_body
	%i_bytes =l shl %i, 3
	%a_ptr =l add %a, %i_bytes
	%ai =l loadl %a_ptr
	%u0 =l and %ai, 4294967295
	%u1 =l shr %ai, 32
	%j_0 =l copy 0
	%carry_0 =l copy 0
	jmp @inner_loop
@inner_loop
	%j =l phi @outer_body %j_0, @inner_body %j_next
	%carry =l phi @outer_body %carry_0, @inner_body %carry_next
	%cond_j =w csltl %j, %nb
	jnz %cond_j, @inner_body, @inner_done
@inner_body
	%j_bytes =l shl %j, 3
	%b_ptr =l add %b, %j_bytes
	%bj =l loadl %b_ptr
	%v0 =l and %bj, 4294967295
	%v1 =l shr %bj, 32
	%p0 =l mul %u0, %v0
	%p1 =l mul %u0, %v1
	%p2 =l mul %u1, %v0
	%p3 =l mul %u1, %v1
	%mid =l add %p1, %p2
	%c1 =w cultl %mid, %p1
	%c1_l =l extuw %c1
	%c1_sh =l shl %c1_l, 32
	%mid_lo =l shl %mid, 32
	%prod_lo =l add %p0, %mid_lo
	%c2 =w cultl %prod_lo, %p0
	%c2_l =l extuw %c2
	%mid_hi =l shr %mid, 32
	%hi1 =l add %p3, %mid_hi
	%hi2 =l add %hi1, %c1_sh
	%prod_hi =l add %hi2, %c2_l
	%ij =l add %i, %j
	%ij_bytes =l shl %ij, 3
	%rij_ptr =l add %r, %ij_bytes
	%rij =l loadl %rij_ptr
	%s1 =l add %prod_lo, %rij
	%ca1 =w cultl %s1, %prod_lo
	%ca1_l =l extuw %ca1
	%s2 =l add %s1, %carry
	%ca2 =w cultl %s2, %s1
	%ca2_l =l extuw %ca2
	storel %s2, %rij_ptr
	%c_sum1 =l add %prod_hi, %ca1_l
	%carry_next =l add %c_sum1, %ca2_l
	%j_next =l add %j, 1
	jmp @inner_loop
@inner_done
	%inb =l add %i, %nb
	%inb_bytes =l shl %inb, 3
	%rinb_ptr =l add %r, %inb_bytes
	storel %carry, %rinb_ptr
	%i_next =l add %i, 1
	jmp @outer_loop
@outer_done
	%k_0 =l add %na, %nb
	jmp @sig_loop
@sig_loop
	%k =l phi @outer_done %k_0, @sig_dec %k_next
	%cond_k =w csgtl %k, 0
	jnz %cond_k, @sig_check, @sig_zero
@sig_check
	%k_minus_1 =l sub %k, 1
	%k_bytes =l shl %k_minus_1, 3
	%rk_ptr =l add %r, %k_bytes
	%rk_val =l loadl %rk_ptr
	%is_zero =w ceql %rk_val, 0
	jnz %is_zero, @sig_dec, @sig_found
@sig_dec
	%k_next =l copy %k_minus_1
	jmp @sig_loop
@sig_found
	ret %k
@sig_zero
	ret 0
}
