export function l $base64_encode(l %src, l %n, l %out) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @ret_zero, @loop_3_header

@ret_zero
	%zero =l copy 0
	ret %zero

@loop_3_header
	%src_cur =l phi @start %src, @loop_3_body %src_next
	%out_cur =l phi @start %out, @loop_3_body %out_next
	%bytes_left =l phi @start %n, @loop_3_body %bytes_left_next
	%has_3 =w csgel %bytes_left, 3
	jnz %has_3, @loop_3_body, @loop_3_end

@loop_3_body
	%b0 =l loadub %src_cur
	%src_p1 =l add %src_cur, 1
	%b1 =l loadub %src_p1
	%src_p2 =l add %src_cur, 2
	%b2 =l loadub %src_p2

	%l_v0 =l shr %b0, 2
	%l_b0_shl =l shl %b0, 4
	%l_b1_shr =l shr %b1, 4
	%l_v1_comb =l or %l_b0_shl, %l_b1_shr

	%e0_c2 =l extuw %e0_cmp2
	%e0_m2 =l mul %e0_c2, 75
	%e0_t2 =l sub %e0_t1, %e0_m2
	%e0_cmp3 =w csgel %l_v0, 62
	%e0_c3 =l extuw %e0_cmp3
	%e0_m3 =l mul %e0_c3, 15
	%e0_t3 =l sub %e0_t2, %e0_m3
	%e0_cmp4 =w ceql %l_v0, 63
	%e0_c4 =l extuw %e0_cmp4
	%e0_m4 =l mul %e0_c4, 3
	%l_c0 =l add %e0_t3, %e0_m4

	%e1_t0 =l add %l_v1, 65
	%e1_cmp1 =w csgel %l_v1, 26
	%e1_c1 =l extuw %e1_cmp1
	%e1_m1 =l mul %e1_c1, 6
	%e1_t1 =l add %e1_t0, %e1_m1
	%e1_cmp2 =w csgel %l_v1, 52
	%e1_c2 =l extuw %e1_cmp2
	%e1_m2 =l mul %e1_c2, 75
	%e1_t2 =l sub %e1_t1, %e1_m2
	%e1_cmp3 =w csgel %l_v1, 62
	%e1_c3 =l extuw %e1_cmp3
	%e1_m3 =l mul %e1_c3, 15
	%e1_t3 =l sub %e1_t2, %e1_m3
	%e1_cmp4 =w ceql %l_v1, 63
	%e1_c4 =l extuw %e1_cmp4
	%e1_m4 =l mul %e1_c4, 3
	%l_c1 =l add %e1_t3, %e1_m4

	%e2_t0 =l add %l_v2, 65
	%e2_cmp1 =w csgel %l_v2, 26
	%e2_c1 =l extuw %e2_cmp1
	%e2_m1 =l mul %e2_c1, 6
	%e2_t1 =l add %e2_t0, %e2_m1
	%e2_cmp2 =w csgel %l_v2, 52
	%e2_c2 =l extuw %e2_cmp2
	%e2_m2 =l mul %e2_c2, 75
	%e2_t2 =l sub %e2_t1, %e2_m2
	%e2_cmp3 =w csgel %l_v2, 62
	%e2_c3 =l extuw %e2_cmp3

	%e2_cmp4 =w ceql %l_v2, 63
	%e2_c4 =l extuw %e2_cmp4
	%e2_m4 =l mul %e2_c4, 3
	%l_c2 =l add %e2_t3, %e2_m4

	%e3_t0 =l add %l_v3, 65
	%e3_cmp1 =w csgel %l_v3, 26
	%e3_c1 =l extuw %e3_cmp1
	%e3_m1 =l mul %e3_c1, 6
	%e3_t1 =l add %e3_t0, %e3_m1

	%e3_c2 =l extuw %e3_cmp2

	%e3_t2 =l sub %e3_t1, %e3_m2
	%e3_cmp3 =w csgel %l_v3, 62
	%e3_c3 =l extuw %e3_cmp3
	%e3_m3 =l mul %e3_c3, 15
	%e3_t3 =l sub %e3_t2, %e3_m3
	%e3_cmp4 =w ceql %l_v3, 63
	%e3_c4 =l extuw %e3_cmp4
	%e3_m4 =l mul %e3_c4, 3
	%l_c3 =l add %e3_t3, %e3_m4

	storeb %l_c0, %out_cur
	%out_p1 =l add %out_cur, 1
	storeb %l_c1, %out_p1
	%out_p2 =l add %out_cur, 2
	storeb %l_c2, %out_p2
	%out_p3 =l add %out_cur, 3
	storeb %l_c3, %out_p3

	%src_next =l add %src_cur, 3
	%out_next =l add %out_cur, 4
	%bytes_left_next =l sub %bytes_left, 3
	jmp @loop_3_header

@loop_3_end
	%is_rem_0 =w ceql %bytes_left, 0
	jnz %is_rem_0, @done, @check_rem_1

@check_rem_1

	jnz %is_rem_1, @rem_1, @rem_2

@rem_1

	%r1_v0 =l shr %r1_b0, 2
	%r1_b0_shl =l shl %r1_b0, 4
	%r1_v1 =l and %r1_b0_shl, 63

	%e4_t0 =l add %r1_v0, 65
	%e4_cmp1 =w csgel %r1_v0, 26
	%e4_c1 =l extuw %e4_cmp1

	%e4_t3 =l sub %e4_t2, %e4_m3
	%e4_cmp4 =w ceql %r1_v0, 63
	%e4_c4 =l extuw %e4_cmp4
	%e4_m4 =l mul %e4_c4, 3
	%r1_c0 =l add %e4_t3, %e4_m4

	%e5_t0 =l add %r1_v1, 65
	%e5_cmp1 =w csgel %r1_v1, 26

	%r1_c1 =l add %e5_t3, %e5_m4

	storeb %r1_c0, %out_cur
	%r1_out_p1 =l add %out_cur, 1
	storeb %r1_c1, %r1_out_p1
	%pad =l copy 61
	%r1_out_p2 =l add %out_cur, 2
	storeb %pad, %r1_out_p2
	%r1_out_p3 =l add %out_cur, 3
	storeb %pad, %r1_out_p3
	jmp @done

@rem_2
	%r2_b0 =l loadub %src_cur
	%r2_src_p1 =l add %src_cur, 1
	%r2_b1 =l loadub %r2_src_p1

	%r2_v0 =l shr %r2_b0, 2
	%r2_b0_shl =l shl %r2_b0, 4
	%r2_b1_shr =l shr %r2_b1, 4
	%r2_v1_comb =l or %r2_b0_shl, %r2_b1_shr
	%r2_v1 =l and %r2_v1_comb, 63
	%r2_b1_shl =l shl %r2_b1, 2
	%r2_v2 =l and %r2_b1_shl, 63

	%e6_t0 =l add %r2_v0, 65
	%e6_cmp1 =w csgel %r2_v0, 26
	%e6_c1 =l extuw %e6_cmp1
	%e6_m1 =l mul %e6_c1, 6
	%e6_t1 =l add %e6_t0, %e6_m1
	%e6_cmp2 =w csgel %r2_v0, 52
	%e6_c2 =l extuw %e6_cmp2
	%e6_m2 =l mul %e6_c2, 75
	%e6_t2 =l sub %e6_t1, %e6_m2
	%e6_cmp3 =w csgel %r2_v0, 62
	%e6_c3 =l extuw %e6_cmp3
	%e6_m3 =l mul %e6_c3, 15
	%e6_t3 =l sub %e6_t2, %e6_m3
	%e6_cmp4 =w ceql %r2_v0, 63
	%e6_c4 =l extuw %e6_cmp4
	%e6_m4 =l mul %e6_c4, 3
	%r2_c0 =l add %e6_t3, %e6_m4

	%e7_cmp1 =w csgel %r2_v1, 26

	%e7_t2 =l sub %e7_t1, %e7_m2
	%e7_cmp3 =w csgel %r2_v1, 62

	%e8_t0 =l add %r2_v2, 65
	%e8_cmp1 =w csgel %r2_v2, 26
	%e8_c1 =l extuw %e8_cmp1
	%e8_m1 =l mul %e8_c1, 6
	%e8_t1 =l add %e8_t0, %e8_m1
	%e8_cmp2 =w csgel %r2_v2, 52
	%e8_c2 =l extuw %e8_cmp2

	%e8_cmp3 =w csgel %r2_v2, 62
	%e8_c3 =l extuw %e8_cmp3
	%e8_m3 =l mul %e8_c3, 15
	%e8_t3 =l sub %e8_t2, %e8_m3
	%e8_cmp4 =w ceql %r2_v2, 63
	%e8_c4 =l extuw %e8_cmp4
	%e8_m4 =l mul %e8_c4, 3
	%r2_c2 =l add %e8_t3, %e8_m4

	storeb %r2_c0, %out_cur
	%r2_out_p1 =l add %out_cur, 1
	storeb %r2_c1, %r2_out_p1
	%r2_out_p2 =l add %out_cur, 2
	storeb %r2_c2, %r2_out_p2
	%pad2 =l copy 61
	%r2_out_p3 =l add %out_cur, 3
	storeb %pad2, %r2_out_p3
	jmp @done

@done
	%n_plus_2 =l add %n, 2
	%num_chunks =l udiv %n_plus_2, 3
	%ret_val =l mul %num_chunks, 4
	ret %ret_val
}
