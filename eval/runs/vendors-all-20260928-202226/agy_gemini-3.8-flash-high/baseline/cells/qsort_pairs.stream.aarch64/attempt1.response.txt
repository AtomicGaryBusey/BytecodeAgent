qsort_pairs:
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    mov x19, x0
    mov x20, x1
.Lloop:

    b.le .Ldone
    lsr x9, x20, #1
    lsl x9, x9, #4
    add x9, x19, x9
    ldp x10, x11, [x19]
    ldp x12, x13, [x9]
    stp x12, x13, [x19]
    stp x10, x11, [x9]
    sub x14, x19, #16
    lsl x15, x20, #4
    add x15, x19, x15
.Lloop_i:
    add x14, x14, #16
    ldp x2, x3, [x14]
    cmp x2, x12
    b.lt .Lloop_i
    b.gt .Li_done
    cmp x3, x13
    b.lt .Lloop_i
.Li_done:
.Lloop_j:

    ldp x4, x5, [x15]
    cmp x4, x12
    b.gt .Lloop_j
    b.lt .Lj_done
    cmp x5, x13
    b.gt .Lloop_j
.Lj_done:
    cmp x14, x15
    b.hs .Lpartition_done
    stp x4, x5, [x14]
    stp x2, x3, [x15]
    b .Lloop_i
.Lpartition_done:
    sub x9, x15, x19
    lsr x9, x9, #4
    add x9, x9, #1
    sub x10, x20, x9
    cmp x9, x10
    b.ge .Lrecurse_right
    mov x0, x19
    add x19, x19, x9, lsl #4
    mov x20, x10
    mov x1, x9
    bl qsort_pairs
    b .Lloop
.Lrecurse_right:
    add x0, x19, x9, lsl #4
    mov x1, x10
    mov x20, x9
    bl qsort_pairs
    b .Lloop
.Ldone:
    mov x0, #0
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
    ret
