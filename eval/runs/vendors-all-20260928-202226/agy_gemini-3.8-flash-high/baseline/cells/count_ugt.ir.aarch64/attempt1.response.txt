export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop
@loop
    %cnt =l phi @start 0, @body %cnt1
    %i =l phi @start 0, @body %i1
    %ptr =l phi @start %a, @body %ptr1
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %val =l loadl %ptr
    %gt =w cugtl %val, %t
    %inc =l extuw %gt
    %cnt1 =l add %cnt, %inc
    %i1 =l add %i, 1
    %ptr1 =l add %ptr, 8
    jmp @loop
@end
    ret %cnt
}
