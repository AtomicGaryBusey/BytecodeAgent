export function l $max_byte(l %p, l %n) {
@start
    %empty =w cslel %n, 0
    jnz %empty, @ret_zero, @loop

@loop
    %p_cur =l phi @start %p, @loop_step %p_next
    %rem =l phi @start %n, @loop_step %rem_next
    %max_cur =l phi @start 0, @loop_step %max_next
    %b =l loadub %p_cur
    %gt =w cugtl %b, %max_cur
    jnz %gt, @new_max, @keep_max

@new_max
    jmp @loop_step

@keep_max
    jmp @loop_step

@loop_step
    %max_next =l phi @new_max %b, @keep_max %max_cur
    %p_next =l add %p_cur, 1
    %rem_next =l sub %rem, 1
    %done =w ceql %rem_next, 0
    jnz %done, @done, @loop

@done
    ret %max_next

@ret_zero
    ret 0
}
