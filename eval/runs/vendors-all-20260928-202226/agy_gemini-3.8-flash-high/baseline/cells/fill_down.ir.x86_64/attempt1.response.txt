export function l $fill_down(l %a, l %n, l %start) {
@entry
    %zero =l copy 0
    jmp @loop
@loop
    %i =l phi @entry %zero, @body %i1
    %c =w csltl %i, %n
    jnz %c, @body, @end
@body
    %val =l sub %start, %i
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    storel %val, %ptr
    %i1 =l add %i, 1
    jmp @loop
@end
    ret %n
}
