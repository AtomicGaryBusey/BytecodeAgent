qsort_pairs:
    cmp rsi, 1
    jle .Lret_zero
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
    mov rax, rbp
    shr rax, 1
    shl rax, 4
    mov r12, qword ptr [rbx + rax]
    mov r13, qword ptr [rbx + rax + 8]
    xor r14, r14
    xor r8, r8
    lea r15, [rbp - 1]
.Lpart_loop:
    cmp r8, r15
    jg .Lpart_done
    mov rcx, r8
    shl rcx, 4
    mov rax, qword ptr [rbx + rcx]
    cmp rax, r12
    jl .Lcase_less
    jg .Lcase_greater
    mov rax, qword ptr [rbx + rcx + 8]
    cmp rax, r13
    jl .Lcase_less
    jg .Lcase_greater
    inc r8
    jmp .Lpart_loop
.Lcase_less:
    mov rdx, r14
    shl rdx, 4
    mov rax, qword ptr [rbx + rdx]
    mov r9, qword ptr [rbx + rcx]
    mov qword ptr [rbx + rdx], r9
    mov qword ptr [rbx + rcx], rax
    mov rax, qword ptr [rbx + rdx + 8]
    mov r9, qword ptr [rbx + rcx + 8]
    mov qword ptr [rbx + rdx + 8], r9
    mov qword ptr [rbx + rcx + 8], rax
    inc r14
    inc r8
    jmp .Lpart_loop
.Lcase_greater:
    mov rdx, r15
    shl rdx, 4
    mov rax, qword ptr [rbx + rdx]
    mov r9, qword ptr [rbx + rcx]
    mov qword ptr [rbx + rdx], r9
    mov qword ptr [rbx + rcx], rax
    mov rax, qword ptr [rbx + rdx + 8]
    mov r9, qword ptr [rbx + rcx + 8]
    mov qword ptr [rbx + rdx + 8], r9
    mov qword ptr [rbx + rcx + 8], rax
    dec r15
    jmp .Lpart_loop
.Lpart_done:
    test r14, r14
    jle .Lskip_left
    mov rdi, rbx
    mov rsi, r14
    call qsort_pairs
.Lskip_left:
    lea rdx, [r15 + 1]
    mov rsi, rbp
    sub rsi, rdx
    test rsi, rsi
    jle .Lskip_right
    shl rdx, 4
    lea rdi, [rbx + rdx]
    call qsort_pairs
.Lskip_right:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Lret_zero:
    xor eax, eax
    ret
