export function l $base64_encode(l %src, l %n, l %out) {
@start
	%n_plus_2 =l add %n, 2
	%ceil =l udiv %n_plus_2, 3
	%ret =l mul %ceil, 4
	%is_zero =w ceql %n, 0
	jnz %is_zero, @done, @loop_check
@loop_check
	%cur_src =l phi @start %src, @loop_body %next_src
	%cur_out =l phi @start %out, @loop_body %next_out
	%rem =l phi @start %n, @loop_body %next_rem
	%has_3 =w csgel %rem, 3
	jnz %has_3, @loop_body, @rem_check
@loop_body
	%b0_l =l loadub %cur_src
	%p1 =l add %cur_src, 1
	%b1_l =l loadub %p1
	%p2 =l add %cur_src, 2
	%b2_l =l loadub %p2
	%b0 =w copy %b0_l
	%b1 =w copy %b1_l
	%b2 =w copy %b2_l
	%b0_sh =w shl %b0, 16
	%b1_sh =w shl %b1, 8
	%v01 =w or %b0_sh, %b1_sh
	%v =w or %v01, %b2
	%v0 =w shr %v, 18
	%v1_s =w shr %v, 12
	%v1 =w and %v1_s, 63
	%v2_s =w shr %v, 6
	%v2 =w and %v2_s, 63
	%v3 =w and %v, 63
	%t1_0 =w csgew %v0, 26
	%m1_0 =w mul %t1_0, 6
	%d1_0 =w add 65, %m1_0
	%t2_0 =w csgew %v0, 52
	%m2_0 =w mul %t2_0, 75
	%d2_0 =w sub %d1_0, %m2_0
	%t3_0 =w csgew %v0, 62
	%m3_0 =w mul %t3_0, 15
	%d3_0 =w sub %d2_0, %m3_0
	%t4_0 =w ceqw %v0, 63
	%m4_0 =w mul %t4_0, 3
	%d4_0 =w add %d3_0, %m4_0
	%c0 =w add %v0, %d4_0
	%t1_1 =w csgew %v1, 26
	%m1_1 =w mul %t1_1, 6
	%d1_1 =w add 65, %m1_1
	%t2_1 =w csgew %v1, 52
	%m2_1 =w mul %t2_1, 75
	%d2_1 =w sub %d1_1, %m2_1
	%t3_1 =w csgew %v1, 62
	%m3_1 =w mul %t3_1, 15
	%d3_1 =w sub %d2_1, %m3_1
	%t4_1 =w ceqw %v1, 63
	%m4_1 =w mul %t4_1, 3
	%d4_1 =w add %d3_1, %m4_1
	%c1 =w add %v1, %d4_1
	%t1_2 =w csgew %v2, 26
	%m1_2 =w mul %t1_2, 6
	%d1_2 =w add 65, %m1_2
	%t2_2 =w csgew %v2, 52
	%m2_2 =w mul %t2_2, 75
	%d2_2 =w sub %d1_2, %m2_2
	%t3_2 =w csgew %v2, 62
	%m3_2 =w mul %t3_2, 15
	%d3_2 =w sub %d2_2, %m3_2
	%t4_2 =w ceqw %v2, 63
	%m4_2 =w mul %t4_2, 3
	%d4_2 =w add %d3_2, %m4_2
	%c2 =w add %v2, %d4_2
	%t1_3 =w csgew %v3, 26
	%m1_3 =w mul %t1_3, 6
	%d1_3 =w add 65, %m1_3
	%t2_3 =w csgew %v3, 52
	%m2_3 =w mul %t2_3, 75
	%d2_3 =w sub %d1_3, %m2_3
	%t3_3 =w csgew %v3, 62
	%m3_3 =w mul %t3_3, 15
	%d3_3 =w sub %d2_3, %m3_3
	%t4_3 =w ceqw %v3, 63
	%m4_3 =w mul %t4_3, 3
	%d4_3 =w add %d3_3, %m4_3
	%c3 =w add %v3, %d4_3
	storeb %c0, %cur_out
	%o1 =l add %cur_out, 1
	storeb %c1, %o1
	%o2 =l add %cur_out, 2
	storeb %c2, %o2
	%o3 =l add %cur_out, 3
	storeb %c3, %o3
	%next_src =l add %cur_src, 3
	%next_out =l add %cur_out, 4
	%next_rem =l sub %rem, 3
	jmp @loop_check
@rem_check
	%is_rem0 =w ceql %rem, 0
	jnz %is_rem0, @done, @check_rem1
@check_rem1
	%is_rem1 =w ceql %rem, 1
	jnz %is_rem1, @rem_one, @rem_two
@rem_one
	%r1_b0_l =l loadub %cur_src
	%r1_b0 =w copy %r1_b0_l
	%r1_v =w shl %r1_b0, 16
	%r1_v0 =w shr %r1_v, 18
	%r1_v1_s =w shr %r1_v, 12
	%r1_v1 =w and %r1_v1_s, 63
	%r1_t1_0 =w csgew %r1_v0, 26
	%r1_m1_0 =w mul %r1_t1_0, 6
	%r1_d1_0 =w add 65, %r1_m1_0
	%r1_t2_0 =w csgew %r1_v0, 52
	%r1_m2_0 =w mul %r1_t2_0, 75
	%r1_d2_0 =w sub %r1_d1_0, %r1_m2_0
	%r1_t3_0 =w csgew %r1_v0, 62
	%r1_m3_0 =w mul %r1_t3_0, 15
	%r1_d3_0 =w sub %r1_d2_0, %r1_m3_0
	%r1_t4_0 =w ceqw %r1_v0, 63
	%r1_m4_0 =w mul %r1_t4_0, 3
	%r1_d4_0 =w add %r1_d3_0, %r1_m4_0
	%r1_c0 =w add %r1_v0, %r1_d4_0
	%r1_t1_1 =w csgew %r1_v1, 26
	%r1_m1_1 =w mul %r1_t1_1, 6
	%r1_d1_1 =w add 65, %r1_m1_1
	%r1_t2_1 =w csgew %r1_v1, 52
	%r1_m2_1 =w mul %r1_t2_1, 75
	%r1_d2_1 =w sub %r1_d1_1, %r1_m2_1
	%r1_t3_1 =w csgew %r1_v1, 62
	%r1_m3_1 =w mul %r1_t3_1, 15
	%r1_d3_1 =w sub %r1_d2_1, %r1_m3_1
	%r1_t4_1 =w ceqw %r1_v1, 63
	%r1_m4_1 =w mul %r1_t4_1, 3
	%r1_d4_1 =w add %r1_d3_1, %r1_m4_1
	%r1_c1 =w add %r1_v1, %r1_d4_1
	storeb %r1_c0, %cur_out
	%r1_o1 =l add %cur_out, 1
	storeb %r1_c1, %r1_o1
	%eq_w =w copy 61
	%r1_o2 =l add %cur_out, 2
	storeb %eq_w, %r1_o2
	%r1_o3 =l add %cur_out, 3
	storeb %eq_w, %r1_o3
	ret %ret
@rem_two
	%r2_b0_l =l loadub %cur_src
	%r2_p1 =l add %cur_src, 1
	%r2_b1_l =l loadub %r2_p1
	%r2_b0 =w copy %r2_b0_l
	%r2_b1 =w copy %r2_b1_l
	%r2_b0_sh =w shl %r2_b0, 16
	%r2_b1_sh =w shl %r2_b1, 8
	%r2_v =w or %r2_b0_sh, %r2_b1_sh
	%r2_v0 =w shr %r2_v, 18
	%r2_v1_s =w shr %r2_v, 12
	%r2_v1 =w and %r2_v1_s, 63
	%r2_v2_s =w shr %r2_v, 6
	%r2_v2 =w and %r2_v2_s, 63
	%r2_t1_0 =w csgew %r2_v0, 26
	%r2_m1_0 =w mul %r2_t1_0, 6
	%r2_d1_0 =w add 65, %r2_m1_0
	%r2_t2_0 =w csgew %r2_v0, 52
	%r2_m2_0 =w mul %r2_t2_0, 75
	%r2_d2_0 =w sub %r2_d1_0, %r2_m2_0
	%r2_t3_0 =w csgew %r2_v0, 62
	%r2_m3_0 =w mul %r2_t3_0, 15
	%r2_d3_0 =w sub %r2_d2_0, %r2_m3_0
	%r2_t4_0 =w ceqw %r2_v0, 63
	%r2_m4_0 =w mul %r2_t4_0, 3
	%r2_d4_0 =w add %r2_d3_0, %r2_m4_0
	%r2_c0 =w add %r2_v0, %r2_d4_0
	%r2_t1_1 =w csgew %r2_v1, 26
	%r2_m1_1 =w mul %r2_t1_1, 6
	%r2_d1_1 =w add 65, %r2_m1_1
	%r2_t2_1 =w csgew %r2_v1, 52
	%r2_m2_1 =w mul %r2_t2_1, 75
	%r2_d2_1 =w sub %r2_d1_1, %r2_m2_1
	%r2_t3_1 =w csgew %r2_v1, 62
	%r2_m3_1 =w mul %r2_t3_1, 15
	%r2_d3_1 =w sub %r2_d2_1, %r2_m3_1
	%r2_t4_1 =w ceqw %r2_v1, 63
	%r2_m4_1 =w mul %r2_t4_1, 3
	%r2_d4_1 =w add %r2_d3_1, %r2_m4_1
	%r2_c1 =w add %r2_v1, %r2_d4_1
	%r2_t1_2 =w csgew %r2_v2, 26
	%r2_m1_2 =w mul %r2_t1_2, 6
	%r2_d1_2 =w add 65, %r2_m1_2
	%r2_t2_2 =w csgew %r2_v2, 52
	%r2_m2_2 =w mul %r2_t2_2, 75
	%r2_d2_2 =w sub %r2_d1_2, %r2_m2_2
	%r2_t3_2 =w csgew %r2_v2, 62
	%r2_m3_2 =w mul %r2_t3_2, 15
	%r2_d3_2 =w sub %r2_d2_2, %r2_m3_2
	%r2_t4_2 =w ceqw %r2_v2, 63
	%r2_m4_2 =w mul %r2_t4_2, 3
	%r2_d4_2 =w add %r2_d3_2, %r2_m4_2
	%r2_c2 =w add %r2_v2, %r2_d4_2
	storeb %r2_c0, %cur_out
	%r2_o1 =l add %cur_out, 1
	storeb %r2_c1, %r2_o1
	%r2_o2 =l add %cur_out, 2
	storeb %r2_c2, %r2_o2
	%r2_eq_w =w copy 61
	%r2_o3 =l add %cur_out, 3
	storeb %r2_eq_w, %r2_o3
	ret %ret
@done
	ret %ret
}
