export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%is_le_zero =w cslel %n, 0
	jnz %is_le_zero, @done, @check_same
@check_same
	%is_same =w ceql %dst, %src
	jnz %is_same, @done, @check_dir
@check_dir
	%is_fwd =w cultl %dst, %src
	jnz %is_fwd, @fwd_loop, @bwd_init
@fwd_loop
	%d_fwd =l phi @check_dir %dst, @fwd_loop %d_fwd_next
	%s_fwd =l phi @check_dir %src, @fwd_loop %s_fwd_next
	%i_fwd =l phi @check_dir %n, @fwd_loop %i_fwd_next
	%val_fwd =l loadl %s_fwd
	storel %val_fwd, %d_fwd
	%d_fwd_next =l add %d_fwd, 8
	%s_fwd_next =l add %s_fwd, 8
	%i_fwd_next =l sub %i_fwd, 1
	%fwd_more =w cnel %i_fwd_next, 0
	jnz %fwd_more, @fwd_loop, @done
@bwd_init
	%bytes =l mul %n, 8
	%d_end =l add %dst, %bytes
	%s_end =l add %src, %bytes
	jmp @bwd_loop
@bwd_loop
	%d_bwd =l phi @bwd_init %d_end, @bwd_loop %d_bwd_prev
	%s_bwd =l phi @bwd_init %s_end, @bwd_loop %s_bwd_prev
	%i_bwd =l phi @bwd_init %n, @bwd_loop %i_bwd_next
	%d_bwd_prev =l sub %d_bwd, 8
	%s_bwd_prev =l sub %s_bwd, 8
	%val_bwd =l loadl %s_bwd_prev
	storel %val_bwd, %d_bwd_prev
	%i_bwd_next =l sub %i_bwd, 1
	%bwd_more =w cnel %i_bwd_next, 0
	jnz %bwd_more, @bwd_loop, @done
@done
	ret 0
}
