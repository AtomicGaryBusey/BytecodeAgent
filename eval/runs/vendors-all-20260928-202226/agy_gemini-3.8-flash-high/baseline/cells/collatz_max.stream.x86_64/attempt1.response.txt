collatz_max:
    xor eax, eax
    mov rsi, 1
.Louter_loop:
    mov rdx, rsi
    xor ecx, ecx
.Linner_loop:
    cmp rdx, 1
    je .Linner_done
    inc rcx
    test dl, 1
    jnz .Lodd
    shr rdx, 1
    jmp .Linner_loop
.Lodd:
    lea rdx, [rdx + rdx*2 + 1]
    jmp .Linner_loop
.Linner_done:
    cmp rax, rcx
    cmovl rax, rcx
    inc rsi
    cmp rsi, rdi
    jle .Louter_loop
    ret
