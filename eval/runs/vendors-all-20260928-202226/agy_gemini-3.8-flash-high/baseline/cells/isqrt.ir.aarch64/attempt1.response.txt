export function l $isqrt(l %n) {
@start
    %b0 =l copy 2147483648
    %res0 =l copy 0
@loop
    %res =l phi @start %res0, @next %res_next
    %b =l phi @start %b0, @next %b_next
    %cand =l or %res, %b
    %sq =l mul %cand, %cand
    %cond =w culel %sq, %n
    jnz %cond, @take, @skip
@take
    jmp @next
@skip
    jmp @next
@next
    %res_next =l phi @take %cand, @skip %res
    %b_next =l shr %b, 1
    jnz %b_next, @loop, @done
@done
    ret %res_next
}
