export function l $isqrt(l %n) {
@start
    jmp @loop
@loop
    %ans =l phi @start 0, @loop_back %ans_next
    %bit =l phi @start 2147483648, @loop_back %bit_next
    %cand =l or %ans, %bit
    %sq =l mul %cand, %cand
    %cmp =w culel %sq, %n
    jnz %cmp, @set_bit, @keep_bit
@set_bit
    jmp @loop_cont
@keep_bit
    jmp @loop_cont
@loop_cont
    %ans_next =l phi @keep_bit %ans, @set_bit %cand
    %bit_next =l shr %bit, 1
    %continue =w cnel %bit_next, 0
    jnz %continue, @loop_back, @end
@loop_back
    jmp @loop
@end
    ret %ans_next
}
