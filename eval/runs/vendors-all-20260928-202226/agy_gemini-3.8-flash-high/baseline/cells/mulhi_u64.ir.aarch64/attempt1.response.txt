export function l $mulhi_u64(l %a, l %b) {
@start
    %mask =l copy 4294967295
    %a0 =l and %a, %mask
    %a1 =l shr %a, 32
    %b0 =l and %b, %mask
    %b1 =l shr %b, 32
    %w0 =l mul %a0, %b0
    %w0_hi =l shr %w0, 32
    %t0 =l mul %a1, %b0
    %t =l add %t0, %w0_hi
    %w1 =l and %t, %mask
    %w2 =l shr %t, 32
    %t1 =l mul %a0, %b1
    %k =l add %t1, %w1
    %k_hi =l shr %k, 32
    %p3 =l mul %a1, %b1
    %r0 =l add %p3, %w2
    %res =l add %r0, %k_hi
    ret %res
}
