export function l $bubble_sort(l %a, l %n) {
@start
	%cmp =w cslel %n, 1
	jnz %cmp, @ret, @outer_init
@outer_init
	%limit_init =l sub %n, 1
	jmp @outer_loop
@outer_loop
	%limit =l phi @outer_init %limit_init, @inner_exit %limit_next
	jmp @inner_loop
@inner_loop
	%j =l phi @outer_loop 0, @inner_step %j_next
	%j_cmp =w csltl %j, %limit
	jnz %j_cmp, @inner_body, @inner_exit
@inner_body
	%offset_j =l mul %j, 8
	%ptr_j =l add %a, %offset_j
	%ptr_j1 =l add %ptr_j, 8
	%val_j =l loadl %ptr_j
	%val_j1 =l loadl %ptr_j1
	%swap_cmp =w csgtl %val_j, %val_j1
	jnz %swap_cmp, @do_swap, @inner_step
@do_swap
	storel %val_j1, %ptr_j
	storel %val_j, %ptr_j1
	jmp @inner_step
@inner_step
	%j_next =l add %j, 1
	jmp @inner_loop
@inner_exit
	%limit_next =l sub %limit, 1
	%limit_cmp =w csgtl %limit_next, 0
	jnz %limit_cmp, @outer_loop, @ret
@ret
	ret 0
}
