map_into:
    test rdx, rdx
    jle .Lzero
    push r15
    push r14
    push r13
    push r12
    push rbp
    push rbx
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov rbp, rdx
    mov r14, rcx
    xor ebx, ebx
.Lloop:
    mov rdi, qword ptr [r13]
    mov r15, rdi
    call r14
    mov qword ptr [r12], rax
    cmp rax, r15
    setl al
    movzx eax, al
    add rbx, rax
    add r12, 8
    add r13, 8
    dec rbp
    jnz .Lloop
    mov rax, rbx
    add rsp, 8
    pop rbx
    pop rbp
    pop r12
    pop r13
    pop r14
    pop r15
    ret
.Lzero:
    xor eax, eax
    ret
