export function l $binary_search(l %a, l %n, l %key) {
@start
    %zero =l copy 0
    %h0 =l sub %n, 1
    jmp @loop
@loop
    %low =l phi @start %zero, @left %low, @right %mid_plus_1
    %high =l phi @start %h0, @left %mid_minus_1, @right %high
    %cond =w cslel %low, %high
    jnz %cond, @body, @not_found
@body
    %sum =l add %low, %high
    %mid =l div %sum, 2
    %shift =l mul %mid, 8
    %ptr =l add %a, %shift
    %val =l loadl %ptr
    %eq =w ceql %val, %key
    jnz %eq, @found, @check_less
@found
    ret %mid
@check_less
    %lt =w csltl %val, %key
    jnz %lt, @right, @left
@right
    %mid_plus_1 =l add %mid, 1
    jmp @loop
@left
    %mid_minus_1 =l sub %mid, 1
    jmp @loop
@not_found
    ret -1
}
