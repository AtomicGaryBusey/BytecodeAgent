export function l $map_sum(l %a, l %n, l %f) {
@start
    jmp @loop
@loop
    %sum =l phi @start 0, @body %sum1
    %i =l phi @start 0, @body %i1
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %fval =l call %f(l %val)
    %sum1 =l add %sum, %fval
    %i1 =l add %i, 1
    jmp @loop
@end
    ret %sum
}
