export function l $crc32(l %p, l %n) {
@start
    %poly =w copy 0xedb88320
    %mask =w copy 0xffffffff
    jmp @loop_cond
@loop_cond
    %crc =w phi @start %mask, @loop_body %c8
    %p_cur =l phi @start %p, @loop_body %p_next
    %i =l phi @start 0, @loop_body %i_next
    %cmp =w csltl %i, %n
    jnz %cmp, @loop_body, @done
@loop_body
    %b =w loadub %p_cur
    %c0 =w xor %crc, %b
    %b0 =w and %c0, 1
    %s0 =w shr %c0, 1
    %m0 =w sub 0, %b0
    %p0 =w and %m0, %poly
    %c1 =w xor %s0, %p0
    %b1 =w and %c1, 1
    %s1 =w shr %c1, 1
    %m1 =w sub 0, %b1
    %p1 =w and %m1, %poly
    %c2 =w xor %s1, %p1
    %b2 =w and %c2, 1
    %s2 =w shr %c2, 1
    %m2 =w sub 0, %b2
    %p2 =w and %m2, %poly
    %c3 =w xor %s2, %p2
    %b3 =w and %c3, 1
    %s3 =w shr %c3, 1
    %m3 =w sub 0, %b3
    %p3 =w and %m3, %poly
    %c4 =w xor %s3, %p3
    %b4 =w and %c4, 1
    %s4 =w shr %c4, 1
    %m4 =w sub 0, %b4
    %p4 =w and %m4, %poly
    %c5 =w xor %s4, %p4
    %b5 =w and %c5, 1
    %s5 =w shr %c5, 1
    %m5 =w sub 0, %b5
    %p5 =w and %m5, %poly
    %c6 =w xor %s5, %p5
    %b6 =w and %c6, 1
    %s6 =w shr %c6, 1
    %m6 =w sub 0, %b6
    %p6 =w and %m6, %poly
    %c7 =w xor %s6, %p6
    %b7 =w and %c7, 1
    %s7 =w shr %c7, 1
    %m7 =w sub 0, %b7
    %p7 =w and %m7, %poly
    %c8 =w xor %s7, %p7
    %p_next =l add %p_cur, 1
    %i_next =l add %i, 1
    jmp @loop_cond
@done
    %final =w xor %crc, %mask
    %res =l extuw %final
    ret %res
}
