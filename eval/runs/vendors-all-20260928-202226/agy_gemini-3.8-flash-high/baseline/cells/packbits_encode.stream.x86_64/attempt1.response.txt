packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Louter_loop:
    cmp r8, rsi
    jae .Louter_done
    lea rax, [r8 + 2]
    cmp rax, rsi
    jae .Lliteral
    mov al, byte ptr [rdi + r8]
    cmp al, byte ptr [rdi + r8 + 1]
    jne .Lliteral
    cmp al, byte ptr [rdi + r8 + 2]
    jne .Lliteral
    mov r10, rsi
    sub r10, r8
    cmp r10, 128
    jbe .Lrepeat_max_ok
    mov r10d, 128
.Lrepeat_max_ok:
    add r10, r8
    lea r11, [r8 + 3]
.Lrepeat_count_loop:
    cmp r11, r10
    jae .Lrepeat_count_done
    cmp byte ptr [rdi + r11], al
    jne .Lrepeat_count_done
    inc r11
    jmp .Lrepeat_count_loop
.Lrepeat_count_done:
    sub r11, r8
    lea rax, [r9 + 2]
    cmp rax, rcx
    jg .Lerr_capacity
    mov eax, 257
    sub eax, r11d
    mov r10b, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    mov byte ptr [rdx + r9 + 1], r10b
    add r9, 2
    add r8, r11
    jmp .Louter_loop
.Lliteral:
    lea r10, [r8 + 1]
.Lliteral_scan:
    cmp r10, rsi
    jae .Lliteral_scan_done
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jae .Lliteral_scan_done
    lea rax, [r10 + 2]
    cmp rax, rsi
    jae .Lliteral_advance
    mov al, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Lliteral_advance
    cmp al, byte ptr [rdi + r10 + 2]
    je .Lliteral_scan_done
.Lliteral_advance:
    inc r10
    jmp .Lliteral_scan
.Lliteral_scan_done:
    mov r11, r10
    sub r11, r8
    lea rax, [r9 + 1]
    add rax, r11
    cmp rax, rcx
    jg .Lerr_capacity
    lea eax, [r11 - 1]
    mov byte ptr [rdx + r9], al
    inc r9
.Lcopy_literal:
    cmp r8, r10
    jae .Louter_loop
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    jmp .Lcopy_literal
.Louter_done:
    mov rax, r9
    ret
.Lerr_capacity:
    mov rax, -1
    ret
