map_sum:
    test rsi, rsi
    jle .Lzero
    push rbx
    push r12
    push r13
    push r14
    sub rsp, 8
    mov r12, rdx
    mov r13, rdi
    lea r14, [rdi + rsi*8]
    xor ebx, ebx
.Lloop:
    mov rdi, qword ptr [r13]
    call r12
    add rbx, rax
    add r13, 8
    cmp r13, r14
    jne .Lloop
    mov rax, rbx
    add rsp, 8
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lzero:
    xor eax, eax
    ret
