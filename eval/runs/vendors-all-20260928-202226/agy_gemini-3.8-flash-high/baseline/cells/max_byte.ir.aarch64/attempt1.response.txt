export function l $max_byte(l %p, l %n) {
@start
	%is_zero =w cslel %n, 0
	jnz %is_zero, @ret_zero, @init
@ret_zero
	ret 0
@init
	%end =l add %p, %n
	jmp @loop
@loop
	%cur_p =l phi @init %p, @step %next_p
	%cur_max =l phi @init 0, @step %next_max
	%b =l loadub %cur_p
	%is_gt =w cugtl %b, %cur_max
	jnz %is_gt, @new_max, @old_max
@new_max
	jmp @step
@old_max
	jmp @step
@step
	%next_max =l phi @new_max %b, @old_max %cur_max
	%next_p =l add %cur_p, 1
	%done =w ceql %next_p, %end
	jnz %done, @exit, @loop
@exit
	ret %next_max
}
