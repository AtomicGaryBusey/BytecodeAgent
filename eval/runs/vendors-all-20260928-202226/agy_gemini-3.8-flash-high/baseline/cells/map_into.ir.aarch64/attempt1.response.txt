export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @loop_head
@loop_head
    %i =l phi @start 0, @loop_body %i_next
    %count =l phi @start 0, @loop_body %count_next
    %continue =w csltl %i, %n
    jnz %continue, @loop_body, @done
@loop_body
    %offset =l mul %i, 8
    %src_ptr =l add %src, %offset
    %src_val =l loadl %src_ptr
    %dst_val =l call %f(l %src_val)
    %dst_ptr =l add %dst, %offset
    storel %dst_val, %dst_ptr
    %cmp =w csltl %dst_val, %src_val
    %inc =l extuw %cmp
    %count_next =l add %count, %inc
    %i_next =l add %i, 1
    jmp @loop_head
@done
    ret %count
}
