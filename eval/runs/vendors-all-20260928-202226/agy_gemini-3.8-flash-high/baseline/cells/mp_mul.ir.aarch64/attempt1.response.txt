export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %n =l add %na, %nb
    jmp @clear_loop

@clear_loop
    %k_clear =l phi @start 0, @clear_body %k_clear1
    %clear_cond =w csltl %k_clear, %n
    jnz %clear_cond, @clear_body, @clear_done

@clear_body
    %k_clear_shift =l shl %k_clear, 3
    %rk_clear =l add %r, %k_clear_shift
    storel 0, %rk_clear
    %k_clear1 =l add %k_clear, 1
    jmp @clear_loop

@clear_done
    %na_zero =w ceql %na, 0
    jnz %na_zero, @sig_init, @check_nb

@check_nb
    %nb_zero =w ceql %nb, 0
    jnz %nb_zero, @sig_init, @outer_loop

@outer_loop
    %i =l phi @check_nb 0, @inner_done %i1
    %outer_cond =w csltl %i, %na
    jnz %outer_cond, @outer_body, @sig_init

@outer_body
    %i_shift =l shl %i, 3
    %ai_ptr =l add %a, %i_shift
    %ai =l loadl %ai_ptr
    jmp @inner_loop

@inner_loop
    %j =l phi @outer_body 0, @inner_body %j1
    %carry =l phi @outer_body 0, @inner_body %carry_next
    %inner_cond =w csltl %j, %nb
    jnz %inner_cond, @inner_body, @inner_done

@inner_body
    %j_shift =l shl %j, 3
    %bj_ptr =l add %b, %j_shift
    %bj =l loadl %bj_ptr

    %u0 =l and %ai, 4294967295
    %u1 =l shr %ai, 32
    %v0 =l and %bj, 4294967295
    %v1 =l shr %bj, 32

    %p0 =l mul %u0, %v0
    %p1 =l mul %u0, %v1
    %p2 =l mul %u1, %v0
    %p3 =l mul %u1, %v1

    %p0_hi =l shr %p0, 32
    %p1_lo =l and %p1, 4294967295
    %p1_hi =l shr %p1, 32
    %p2_lo =l and %p2, 4294967295
    %p2_hi =l shr %p2, 32

    %t0 =l add %p0_hi, %p1_lo
    %mid =l add %t0, %p2_lo
    %mid_hi =l shr %mid, 32

    %h0 =l add %p3, %p1_hi
    %h1 =l add %h0, %p2_hi
    %prod_hi =l add %h1, %mid_hi

    %prod_lo =l mul %ai, %bj

    %ij =l add %i, %j
    %ij_shift =l shl %ij, 3
    %rij_ptr =l add %r, %ij_shift
    %rij =l loadl %rij_ptr

    %s1 =l add %rij, %prod_lo
    %c1_w =w cultl %s1, %rij
    %c1 =l extuw %c1_w

    %s2 =l add %s1, %carry
    %c2_w =w cultl %s2, %s1
    %c2 =l extuw %c2_w

    storel %s2, %rij_ptr

    %carry1 =l add %prod_hi, %c1
    %carry_next =l add %carry1, %c2

    %j1 =l add %j, 1
    jmp @inner_loop

@inner_done
    %inb =l add %i, %nb
    %inb_shift =l shl %inb, 3
    %rinb_ptr =l add %r, %inb_shift
    storel %carry, %rinb_ptr
    %i1 =l add %i, 1
    jmp @outer_loop

@sig_init
    jmp @sig_loop

@sig_loop
    %k =l phi @sig_init %n, @sig_next %k_minus_1
    %k_zero =w ceql %k, 0
    jnz %k_zero, @sig_done, @sig_check

@sig_check
    %k_minus_1 =l sub %k, 1
    %k_shift =l shl %k_minus_1, 3
    %rk_ptr =l add %r, %k_shift
    %rk_val =l loadl %rk_ptr
    %val_nonzero =w cnel %rk_val, 0
    jnz %val_nonzero, @sig_done, @sig_next

@sig_next
    jmp @sig_loop

@sig_done
    ret %k
}
