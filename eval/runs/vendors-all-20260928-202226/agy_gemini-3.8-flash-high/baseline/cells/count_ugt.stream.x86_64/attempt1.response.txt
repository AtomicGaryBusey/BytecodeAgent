count_ugt:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    xor r8d, r8d
    mov rcx, rsi
    shr rcx, 1
    jz .Ltail
.Lloop:
    cmp rdx, qword ptr [rdi]
    adc rax, 0
    cmp rdx, qword ptr [rdi + 8]
    adc r8, 0
    add rdi, 16
    dec rcx
    jnz .Lloop
    add rax, r8
.Ltail:
    test esi, 1
    jz .Ldone
    cmp rdx, qword ptr [rdi]
    adc rax, 0
.Ldone:
    ret
