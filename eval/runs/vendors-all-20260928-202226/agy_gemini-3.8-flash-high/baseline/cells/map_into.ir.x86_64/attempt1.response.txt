export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %i0 =l copy 0
    %count0 =l copy 0
    jmp @loop
@loop
    %i =l phi @start %i0, @body %i1
    %count =l phi @start %count0, @body %count1
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %src_cur =l add %src, %offset
    %val =l loadl %src_cur
    %res =l call %f(l %val)
    %dst_cur =l add %dst, %offset
    storel %res, %dst_cur
    %cmp =w csltl %res, %val
    %inc =l extuw %cmp
    %count1 =l add %count, %inc
    %i1 =l add %i, 1
    jmp @loop
@end
    ret %count
}
