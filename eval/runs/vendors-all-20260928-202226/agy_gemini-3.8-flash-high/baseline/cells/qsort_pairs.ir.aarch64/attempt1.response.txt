export function l $qsort_pairs(l %a, l %n) {
@start
    %c_n =w csltl %n, 2
    jnz %c_n, @ret_0, @do_pivot
@ret_0
    ret 0
@do_pivot
    %mid =l shr %n, 1
    %mid_off =l shl %mid, 4
    %mid_ptr =l add %a, %mid_off
    %mid_vptr =l add %mid_ptr, 8
    %k0 =l loadl %a
    %v0_ptr =l add %a, 8
    %v0 =l loadl %v0_ptr
    %kmid =l loadl %mid_ptr
    %vmid =l loadl %mid_vptr
    storel %kmid, %a
    storel %vmid, %v0_ptr
    storel %k0, %mid_ptr
    storel %v0, %mid_vptr
    %minus1 =l sub 0, 1
    jmp @loop_head
@loop_head
    %i_in =l phi @do_pivot %minus1, @do_swap %i_next
    %j_in =l phi @do_pivot %n, @do_swap %j_next
    jmp @inc_i
@inc_i
    %i_cur =l phi @loop_head %i_in, @inc_i %i_next
    %i_next =l add %i_cur, 1
    %i_off =l shl %i_next, 4
    %i_ptr =l add %a, %i_off
    %i_vptr =l add %i_ptr, 8
    %ik =l loadl %i_ptr
    %iv =l loadl %i_vptr
    %c1 =w csltl %ik, %kmid
    %ceq1 =w ceql %ik, %kmid
    %cv1 =w csltl %iv, %vmid
    %c2 =w and %ceq1, %cv1
    %i_lt =w or %c1, %c2
    jnz %i_lt, @inc_i, @dec_j
@dec_j
    %j_cur =l phi @inc_i %j_in, @dec_j %j_next
    %j_next =l sub %j_cur, 1
    %j_off =l shl %j_next, 4
    %j_ptr =l add %a, %j_off
    %j_vptr =l add %j_ptr, 8
    %jk =l loadl %j_ptr
    %jv =l loadl %j_vptr
    %c3 =w csltl %kmid, %jk
    %ceq2 =w ceql %kmid, %jk
    %cv2 =w csltl %vmid, %jv
    %c4 =w and %ceq2, %cv2
    %j_gt =w or %c3, %c4
    jnz %j_gt, @dec_j, @check_cross
@check_cross
    %crossed =w csgel %i_next, %j_next
    jnz %crossed, @recurse, @do_swap
@do_swap
    storel %jk, %i_ptr
    storel %jv, %i_vptr
    storel %ik, %j_ptr
    storel %iv, %j_vptr
    jmp @loop_head
@recurse
    %len1 =l add %j_next, 1
    %len2 =l sub %n, %len1
    %off2 =l shl %len1, 4
    %a2 =l add %a, %off2
    %r1 =l call $qsort_pairs(l %a, l %len1)
    %r2 =l call $qsort_pairs(l %a2, l %len2)
    ret 0
}
