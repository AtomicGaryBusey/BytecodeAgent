mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rcx
    mov r14, r8
    mov r15, rdx
    mov rbx, rsi
    shl rbx, 3
    mov rbp, rsi
    imul rbp, rsi
    xor eax, eax
    xor ecx, ecx
.Lzero_out:
    cmp rcx, rbp
    jge .Lzero_out_done
    mov qword ptr [r13 + rcx*8], rax
    inc rcx
    jmp .Lzero_out
.Lzero_out_done:
    xor ecx, ecx
    xor eax, eax
    lea rdx, [rsi + 1]
.Lset_diag:
    cmp rcx, rsi
    jge .Lset_diag_done
    mov qword ptr [r13 + rax*8], 1
    add rax, rdx
    inc rcx
    jmp .Lset_diag
.Lset_diag_done:
    test r15, r15
    jz .Lzero_tmp
.Lpower_loop:
    mov r8, r14
    mov r9, r13
.Li_loop:
    mov r10, r12
.Lj_loop:
    mov rcx, r9
    mov rdx, r10
    lea rdi, [r9 + rbx]
    xor esi, esi
.Lt_loop:
    mov rax, qword ptr [rcx]
    imul rax, qword ptr [rdx]
    add rsi, rax
    add rcx, 8
    add rdx, rbx
    cmp rcx, rdi
    jne .Lt_loop
    mov qword ptr [r8], rsi
    add r8, 8
    add r10, 8
    lea rax, [r12 + rbx]
    cmp r10, rax
    jne .Lj_loop
    add r9, rbx
    lea rax, [r13 + rbp*8]
    cmp r9, rax
    jne .Li_loop
    xor ecx, ecx
.Lcopy_loop:
    cmp rcx, rbp
    jge .Lcopy_done
    mov rax, qword ptr [r14 + rcx*8]
    mov qword ptr [r13 + rcx*8], rax
    inc rcx
    jmp .Lcopy_loop
.Lcopy_done:
    dec r15
    jnz .Lpower_loop
.Lzero_tmp:
    lea r10, [rbp + rbp]
    xor eax, eax
    xor ecx, ecx
.Lzero_tmp_loop:
    cmp rcx, r10
    jge .Ldone
    mov qword ptr [r14 + rcx*8], rax
    inc rcx
    jmp .Lzero_tmp_loop
.Ldone:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
