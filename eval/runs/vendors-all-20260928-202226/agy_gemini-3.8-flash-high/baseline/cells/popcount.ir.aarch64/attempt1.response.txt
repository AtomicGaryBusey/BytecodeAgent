export function l $popcount(l %x) {
@start
	%x1 =l shr %x, 1
	%x2 =l and %x1, 0x5555555555555555
	%x3 =l sub %x, %x2
	%x4 =l and %x3, 0x3333333333333333
	%x5 =l shr %x3, 2
	%x6 =l and %x5, 0x3333333333333333
	%x7 =l add %x4, %x6
	%x8 =l shr %x7, 4
	%x9 =l add %x7, %x8
	%x10 =l and %x9, 0x0f0f0f0f0f0f0f0f
	%x11 =l mul %x10, 0x0101010101010101
	%res =l shr %x11, 56
	ret %res
}
