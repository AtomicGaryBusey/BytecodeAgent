    mov x2, #0
    mov x3, #1
.Louter_loop:
    cmp x3, x1
    b.gt .Ldone
    mov x4, x3
    mov x5, #0
.Linner_loop:
    cmp x4, #1
    b.eq .Linner_done
    tbz x4, #0, .Leven
    add x6, x4, x4, lsl #1
    add x4, x6, #1
    add x5, x5, #1
    b .Linner_loop
.Leven:
    lsr x4, x4, #1
    add x5, x5, #1
    b .Linner_loop
.Linner_done:
    cmp x5, x2
    csel x2, x5, x2, gt
    add x3, x3, #1
    b .Louter_loop
.Ldone:
    mov x0, x2
    ret
