export function l $crc32(l %p, l %n) {
@start
	%end =l add %p, %n
	%crc_init =w copy 4294967295
	%poly =w copy 3988292384
	jmp @loop_cond
@loop_cond
	%p_cur =l phi @start %p, @loop_body %p_next
	%crc_cur =w phi @start %crc_init, @loop_body %c8
	%done =w ceql %p_cur, %end
	jnz %done, @done, @loop_body
@loop_body
	%b =w loadub %p_cur
	%c0 =w xor %crc_cur, %b

	%lsb0 =w and %c0, 1
	%mask0 =w sub 0, %lsb0
	%poly0 =w and %mask0, %poly
	%shr0 =w shr %c0, 1
	%c1 =w xor %shr0, %poly0

	%lsb1 =w and %c1, 1
	%mask1 =w sub 0, %lsb1
	%poly1 =w and %mask1, %poly
	%shr1 =w shr %c1, 1
	%c2 =w xor %shr1, %poly1

	%lsb2 =w and %c2, 1
	%mask2 =w sub 0, %lsb2
	%poly2 =w and %mask2, %poly
	%shr2 =w shr %c2, 1
	%c3 =w xor %shr2, %poly2

	%lsb3 =w and %c3, 1
	%mask3 =w sub 0, %lsb3
	%poly3 =w and %mask3, %poly
	%shr3 =w shr %c3, 1
	%c4 =w xor %shr3, %poly4

	%lsb4 =w and %c4, 1
	%mask4 =w sub 0, %lsb4
	%poly4 =w and %mask4, %poly
	%shr4 =w shr %c4, 1
	%c5 =w xor %shr4, %poly4

	%lsb5 =w and %c5, 1
	%mask5 =w sub 0, %lsb5
	%poly5 =w and %mask5, %poly
	%shr5 =w shr %c5, 1
	%c6 =w xor %shr5, %poly5

	%lsb6 =w and %c6, 1
	%mask6 =w sub 0, %lsb6
	%poly6 =w and %mask6, %poly
	%shr6 =w shr %c6, 1
	%c7 =w xor %shr6, %poly6

	%lsb7 =w and %c7, 1
	%mask7 =w sub 0, %lsb7
	%poly7 =w and %mask7, %poly
	%shr7 =w shr %c7, 1
	%c8 =w xor %shr7, %poly7

	%p_next =l add %p_cur, 1
	jmp @loop_cond
@done
	%res_w =w xor %crc_cur, 4294967295
	%res =l extuw %res_w
	ret %res
}
