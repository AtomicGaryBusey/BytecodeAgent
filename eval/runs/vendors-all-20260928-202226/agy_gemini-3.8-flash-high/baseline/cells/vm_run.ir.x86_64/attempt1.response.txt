export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    jmp @loop
@loop
    %pc =l phi @start 0, @op_push %pc_push, @op_add %pc_add, @op_sub %pc_sub, @op_mul %pc_mul, @op_dup %pc_dup, @op_swap %pc_swap, @op_drop %pc_drop, @op_over %pc_over, @op_lt %pc_lt, @jz_end %pc_jz, @op_jmp %t_jmp
    %sp =l phi @start 0, @op_push %sp_push, @op_add %sp_add, @op_sub %sp_sub, @op_mul %sp_mul, @op_dup %sp_dup, @op_swap %sp_swap, @op_drop %sp_drop, @op_over %sp_over, @op_lt %sp_lt, @jz_end %sp_jz, @op_jmp %sp_jmp
    %at_end =w csgel %pc, %len
    jnz %at_end, @halt, @fetch
@fetch
    %pc8 =l shl %pc, 3
    %pc_ptr =l add %code, %pc8
    %op =l loadl %pc_ptr
    %c0 =w ceql %op, 0
    jnz %c0, @halt, @check_1
@check_1
    %c1 =w ceql %op, 1
    jnz %c1, @op_push, @check_2
@check_2
    %c2 =w ceql %op, 2
    jnz %c2, @op_add, @check_3
@check_3
    %c3 =w ceql %op, 3
    jnz %c3, @op_sub, @check_4
@check_4
    %c4 =w ceql %op, 4
    jnz %c4, @op_mul, @check_5
@check_5
    %c5 =w ceql %op, 5
    jnz %c5, @op_dup, @check_6
@check_6
    %c6 =w ceql %op, 6
    jnz %c6, @op_swap, @check_7
@check_7
    %c7 =w ceql %op, 7
    jnz %c7, @op_drop, @check_8
@check_8
    %c8 =w ceql %op, 8
    jnz %c8, @op_over, @check_9
@check_9
    %c9 =w ceql %op, 9
    jnz %c9, @op_lt, @check_10
@check_10
    %c10 =w ceql %op, 10
    jnz %c10, @op_jz, @op_jmp
@op_push
    %pc1_ptr =l add %pc_ptr, 8
    %val_push =l loadl %pc1_ptr
    %sp8_push =l shl %sp, 3
    %stk_push =l add %stack, %sp8_push
    storel %val_push, %stk_push
    %sp_push =l add %sp, 1
    %pc_push =l add %pc, 2
    jmp @loop
@op_add
    %sp_m1_add =l sub %sp, 1
    %sp_m2_add =l sub %sp, 2
    %sp_m1_8_add =l shl %sp_m1_add, 3
    %sp_m2_8_add =l shl %sp_m2_add, 3
    %ptr_b_add =l add %stack, %sp_m1_8_add
    %ptr_a_add =l add %stack, %sp_m2_8_add
    %b_add =l loadl %ptr_b_add
    %a_add =l loadl %ptr_a_add
    %res_add =l add %a_add, %b_add
    storel %res_add, %ptr_a_add
    %sp_add =l %sp_m1_add
    %pc_add =l add %pc, 1
    jmp @loop
@op_sub
    %sp_m1_sub =l sub %sp, 1
    %sp_m2_sub =l sub %sp, 2
    %sp_m1_8_sub =l shl %sp_m1_sub, 3
    %sp_m2_8_sub =l shl %sp_m2_sub, 3
    %ptr_b_sub =l add %stack, %sp_m1_8_sub
    %ptr_a_sub =l add %stack, %sp_m2_8_sub
    %b_sub =l loadl %ptr_b_sub
    %a_sub =l loadl %ptr_a_sub
    %res_sub =l sub %a_sub, %b_sub
    storel %res_sub, %ptr_a_sub
    %sp_sub =l %sp_m1_sub
    %pc_sub =l add %pc, 1
    jmp @loop
@op_mul
    %sp_m1_mul =l sub %sp, 1
    %sp_m2_mul =l sub %sp, 2
    %sp_m1_8_mul =l shl %sp_m1_mul, 3
    %sp_m2_8_mul =l shl %sp_m2_mul, 3
    %ptr_b_mul =l add %stack, %sp_m1_8_mul
    %ptr_a_mul =l add %stack, %sp_m2_8_mul
    %b_mul =l loadl %ptr_b_mul
    %a_mul =l loadl %ptr_a_mul
    %res_mul =l mul %a_mul, %b_mul
    storel %res_mul, %ptr_a_mul
    %sp_mul =l %sp_m1_mul
    %pc_mul =l add %pc, 1
    jmp @loop
@op_dup
    %sp_m1_dup =l sub %sp, 1
    %sp_m1_8_dup =l shl %sp_m1_dup, 3
    %ptr_top_dup =l add %stack, %sp_m1_8_dup
    %val_dup =l loadl %ptr_top_dup
    %sp8_dup =l shl %sp, 3
    %ptr_dst_dup =l add %stack, %sp8_dup
    storel %val_dup, %ptr_dst_dup
    %sp_dup =l add %sp, 1
    %pc_dup =l add %pc, 1
    jmp @loop
@op_swap
    %sp_m1_swap =l sub %sp, 1
    %sp_m2_swap =l sub %sp, 2
    %sp_m1_8_swap =l shl %sp_m1_swap, 3
    %sp_m2_8_swap =l shl %sp_m2_swap, 3
    %ptr_b_swap =l add %stack, %sp_m1_8_swap
    %ptr_a_swap =l add %stack, %sp_m2_8_swap
    %b_swap =l loadl %ptr_b_swap
    %a_swap =l loadl %ptr_a_swap
    storel %b_swap, %ptr_a_swap
    storel %a_swap, %ptr_b_swap
    %sp_swap =l copy %sp
    %pc_swap =l add %pc, 1
    jmp @loop
@op_drop
    %sp_drop =l sub %sp, 1
    %pc_drop =l add %pc, 1
    jmp @loop
@op_over
    %sp_m2_over =l sub %sp, 2
    %sp_m2_8_over =l shl %sp_m2_over, 3
    %ptr_a_over =l add %stack, %sp_m2_8_over
    %val_over =l loadl %ptr_a_over
    %sp8_over =l shl %sp, 3
    %ptr_dst_over =l add %stack, %sp8_over
    storel %val_over, %ptr_dst_over
    %sp_over =l add %sp, 1
    %pc_over =l add %pc, 1
    jmp @loop
@op_lt
    %sp_m1_lt =l sub %sp, 1
    %sp_m2_lt =l sub %sp, 2
    %sp_m1_8_lt =l shl %sp_m1_lt, 3
    %sp_m2_8_lt =l shl %sp_m2_lt, 3
    %ptr_b_lt =l add %stack, %sp_m1_8_lt
    %ptr_a_lt =l add %stack, %sp_m2_8_lt
    %b_lt =l loadl %ptr_b_lt
    %a_lt =l loadl %ptr_a_lt
    %cmp_lt =w csltl %a_lt, %b_lt
    %res_lt =l extuw %cmp_lt
    storel %res_lt, %ptr_a_lt
    %sp_lt =l %sp_m1_lt
    %pc_lt =l add %pc, 1
    jmp @loop
@op_jz
    %sp_m1_jz =l sub %sp, 1
    %sp_m1_8_jz =l shl %sp_m1_jz, 3
    %ptr_v_jz =l add %stack, %sp_m1_8_jz
    %v_jz =l loadl %ptr_v_jz
    %pc1_ptr_jz =l add %pc_ptr, 8
    %t_jz =l loadl %pc1_ptr_jz
    %is_zero =w ceql %v_jz, 0
    jnz %is_zero, @jz_is_zero, @jz_not_zero
@jz_is_zero
    jmp @jz_end
@jz_not_zero
    %pc_jz_step =l add %pc, 2
    jmp @jz_end
@jz_end
    %pc_jz =l phi @jz_is_zero %t_jz, @jz_not_zero %pc_jz_step
    %sp_jz =l copy %sp_m1_jz
    jmp @loop
@op_jmp
    %pc1_ptr_jmp =l add %pc_ptr, 8
    %t_jmp =l loadl %pc1_ptr_jmp
    %sp_jmp =l copy %sp
    jmp @loop
@halt
    %sp_is_zero =w ceql %sp, 0
    jnz %sp_is_zero, @ret_zero, @ret_top
@ret_zero
    jmp @start_zeroing
@ret_top
    %sp_top_idx =l sub %sp, 1
    %sp_top_idx8 =l shl %sp_top_idx, 3
    %top_val_ptr =l add %stack, %sp_top_idx8
    %res_top =l loadl %top_val_ptr
    jmp @start_zeroing
@start_zeroing
    %ret_val =l phi @ret_zero 0, @ret_top %res_top
    jmp @zero_loop_start
@zero_loop_start
    %i =l phi @start_zeroing %sp, @zero_loop_body %i_next
    %more_zero =w csltl %i, %cap
    jnz %more_zero, @zero_loop_body, @done
@zero_loop_body
    %i8 =l shl %i, 3
    %zero_ptr =l add %stack, %i8
    storel 0, %zero_ptr
    %i_next =l add %i, 1
    jmp @zero_loop_start
@done
    ret %ret_val
}
