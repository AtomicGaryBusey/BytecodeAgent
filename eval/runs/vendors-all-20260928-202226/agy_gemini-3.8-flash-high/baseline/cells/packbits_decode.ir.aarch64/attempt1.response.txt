export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop
@loop
    %s_idx =l phi @start 0, @handle_nop %s_idx1, @lit_fits_done %lit_s_end, @rep_fits_done %rep_s_end
    %o_idx =l phi @start 0, @handle_nop %o_idx, @lit_fits_done %lit_o_end, @rep_fits_done %rep_o_end
    %done =w csgel %s_idx, %n
    jnz %done, @ret_ok, @read_hdr
@ret_ok
    ret %o_idx
@read_hdr
    %p_hdr =l add %src, %s_idx
    %h =w loadsb %p_hdr
    %s_idx1 =l add %s_idx, 1
    %is_nop =w ceqw %h, -128
    jnz %is_nop, @handle_nop, @check_type
@handle_nop
    jmp @loop
@check_type
    %is_lit =w csgew %h, 0
    jnz %is_lit, @lit_start, @rep_start
@lit_start
    %h_l =l extsw %h
    %lit_cnt =l add %h_l, 1
    %lit_s_end =l add %s_idx1, %lit_cnt
    %lit_s_err =w csgtl %lit_s_end, %n
    jnz %lit_s_err, @err_src, @lit_start_ok
@lit_start_ok
    %p_src_lit =l add %src, %s_idx1
    %p_dst_lit =l add %out, %o_idx
    %lit_o_end =l add %o_idx, %lit_cnt
    %lit_cap_err =w csgtl %lit_o_end, %cap
    jnz %lit_cap_err, @lit_cap_exceeded, @lit_fits_prep
@lit_fits_prep
    jmp @lit_fits_loop
@lit_fits_loop
    %k_lf =l phi @lit_fits_prep 0, @lit_fits_step %k_lf_next
    %c_lf =w csltl %k_lf, %lit_cnt
    jnz %c_lf, @lit_fits_step, @lit_fits_done
@lit_fits_step
    %ps_lf =l add %p_src_lit, %k_lf
    %b_lf =w loadub %ps_lf
    %pd_lf =l add %p_dst_lit, %k_lf
    storeb %b_lf, %pd_lf
    %k_lf_next =l add %k_lf, 1
    jmp @lit_fits_loop
@lit_fits_done
    jmp @loop
@lit_cap_exceeded
    %lit_rem_cap =l sub %cap, %o_idx
    jmp @lit_exc_loop
@lit_exc_loop
    %k_le =l phi @lit_cap_exceeded 0, @lit_exc_step %k_le_next
    %c_le =w csltl %k_le, %lit_rem_cap
    jnz %c_le, @lit_exc_step, @lit_exc_done
@lit_exc_step
    %ps_le =l add %p_src_lit, %k_le
    %b_le =w loadub %ps_le
    %pd_le =l add %p_dst_lit, %k_le
    storeb %b_le, %pd_le
    %k_le_next =l add %k_le, 1
    jmp @lit_exc_loop
@lit_exc_done
    ret -1
@rep_start
    %has_byte =w csltl %s_idx1, %n
    jnz %has_byte, @rep_read_byte, @err_src
@rep_read_byte
    %p_byte =l add %src, %s_idx1
    %rep_val =w loadub %p_byte
    %rep_s_end =l add %s_idx1, 1
    %rep_h_l =l extsw %h
    %rep_cnt =l sub 1, %rep_h_l
    %p_dst_rep =l add %out, %o_idx
    %rep_o_end =l add %o_idx, %rep_cnt
    %rep_cap_err =w csgtl %rep_o_end, %cap
    jnz %rep_cap_err, @rep_cap_exceeded, @rep_fits_prep
@rep_fits_prep
    jmp @rep_fits_loop
@rep_fits_loop
    %k_rf =l phi @rep_fits_prep 0, @rep_fits_step %k_rf_next
    %c_rf =w csltl %k_rf, %rep_cnt
    jnz %c_rf, @rep_fits_step, @rep_fits_done
@rep_fits_step
    %pd_rf =l add %p_dst_rep, %k_rf
    storeb %rep_val, %pd_rf
    %k_rf_next =l add %k_rf, 1
    jmp @rep_fits_loop
@rep_fits_done
    jmp @loop
@rep_cap_exceeded
    %rep_rem_cap =l sub %cap, %o_idx
    jmp @rep_exc_loop
@rep_exc_loop
    %k_re =l phi @rep_cap_exceeded 0, @rep_exc_step %k_re_next
    %c_re =w csltl %k_re, %rep_rem_cap
    jnz %c_re, @rep_exc_step, @rep_exc_done
@rep_exc_step
    %pd_re =l add %p_dst_rep, %k_re
    storeb %rep_val, %pd_re
    %k_re_next =l add %k_re, 1
    jmp @rep_exc_loop
@rep_exc_done
    ret -1
@err_src
    ret -2
}
