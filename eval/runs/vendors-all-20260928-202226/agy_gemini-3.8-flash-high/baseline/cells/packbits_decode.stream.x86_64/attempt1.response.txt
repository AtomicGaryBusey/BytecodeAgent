packbits_decode:
    lea rsi, [rdi + rsi]
    lea rcx, [rdx + rcx]
    mov r8, rdx
.Lloop:
    cmp rdi, rsi
    je .Lsuccess
    movsx rax, byte ptr [rdi]
    inc rdi
    test rax, rax
    js .Lnegative
    lea r9, [rax + 1]
    mov r10, rsi
    sub r10, rdi
    cmp r10, r9
    jb .Lerr_source
    mov r10, rcx
    sub r10, rdx
    cmp r9, r10
    ja .Lliteral_exceeds
.Lliteral_loop:
    mov al, byte ptr [rdi]
    inc rdi
    mov byte ptr [rdx], al
    inc rdx
    dec r9
    jnz .Lliteral_loop
    jmp .Lloop
.Lliteral_exceeds:
    test r10, r10
    jz .Lerr_cap
.Lliteral_cap_loop:
    mov al, byte ptr [rdi]
    inc rdi
    mov byte ptr [rdx], al
    inc rdx
    dec r10
    jnz .Lliteral_cap_loop
    jmp .Lerr_cap
.Lnegative:
    cmp rax, -128
    je .Lloop
    cmp rdi, rsi
    jae .Lerr_source
    movzx r11d, byte ptr [rdi]
    inc rdi
    mov r9, 1
    sub r9, rax
    mov r10, rcx
    sub r10, rdx
    cmp r9, r10
    ja .Lrepeat_exceeds
.Lrepeat_loop:
    mov byte ptr [rdx], r11b
    inc rdx
    dec r9
    jnz .Lrepeat_loop
    jmp .Lloop
.Lrepeat_exceeds:
    test r10, r10
    jz .Lerr_cap
.Lrepeat_cap_loop:
    mov byte ptr [rdx], r11b
    inc rdx
    dec r10
    jnz .Lrepeat_cap_loop
    jmp .Lerr_cap
.Lerr_cap:
    mov rax, -1
    ret
.Lerr_source:
    mov rax, -2
    ret
.Lsuccess:
    mov rax, rdx
    sub rax, r8
    ret
