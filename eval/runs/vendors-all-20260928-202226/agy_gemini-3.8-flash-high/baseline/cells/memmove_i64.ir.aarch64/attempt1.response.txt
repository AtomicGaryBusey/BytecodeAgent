export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %is_zero =w cslel %n, 0
    jnz %is_zero, @ret, @check_same
@check_same
    %is_same =w ceql %dst, %src
    jnz %is_same, @ret, @check_dir
@check_dir
    %is_fwd =w cultl %dst, %src
    jnz %is_fwd, @fwd_loop, @bwd_loop
@fwd_loop
    %i_fwd =l phi @check_dir 0, @fwd_loop %i_fwd_next
    %off_fwd =l shl %i_fwd, 3
    %s_ptr_fwd =l add %src, %off_fwd
    %val_fwd =l loadl %s_ptr_fwd
    %d_ptr_fwd =l add %dst, %off_fwd
    storel %val_fwd, %d_ptr_fwd
    %i_fwd_next =l add %i_fwd, 1
    %fwd_done =w ceql %i_fwd_next, %n
    jnz %fwd_done, @ret, @fwd_loop
@bwd_loop
    %cur_bwd =l phi @check_dir %n, @bwd_loop %prev_bwd
    %prev_bwd =l sub %cur_bwd, 1
    %off_bwd =l shl %prev_bwd, 3
    %s_ptr_bwd =l add %src, %off_bwd
    %val_bwd =l loadl %s_ptr_bwd
    %d_ptr_bwd =l add %dst, %off_bwd
    storel %val_bwd, %d_ptr_bwd
    %bwd_done =w ceql %prev_bwd, 0
    jnz %bwd_done, @ret, @bwd_loop
@ret
    ret 0
}
