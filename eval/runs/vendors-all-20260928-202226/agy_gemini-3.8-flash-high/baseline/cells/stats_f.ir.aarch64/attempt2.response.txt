export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %val0 =l loadl %a
    %y0 =l call %f(l %val0)
    %odd0 =l and %y0, 1
    %a_next0 =l add %a, 8
    jmp @loop_cond
@loop_cond
    %i =l phi @start 1, @loop_inc %i_next
    %cur_a =l phi @start %a_next0, @loop_inc %a_next
    %min =l phi @start %y0, @loop_inc %min_next
    %max =l phi @start %y0, @loop_inc %max_next
    %sum =l phi @start %y0, @loop_inc %sum_next
    %cnt =l phi @start %odd0, @loop_inc %cnt_next
    %cmp =w csltl %i, %n
    jnz %cmp, @body, @done
@body
    %val =l loadl %cur_a
    %y =l call %f(l %val)
    %sum_next =l add %sum, %y
    %is_odd =l and %y, 1
    %cnt_next =l add %cnt, %is_odd
    %is_min =w csltl %y, %min
    jnz %is_min, @set_min, @chk_max
@set_min
    jmp @chk_max
@chk_max
    %min_next =l phi @body %min, @set_min %y
    %is_max =w csgtl %y, %max
    jnz %is_max, @set_max, @loop_inc
@set_max
    jmp @loop_inc
@loop_inc
    %max_next =l phi @chk_max %max, @set_max %y
    %i_next =l add %i, 1
    %a_next =l add %cur_a, 8
    jmp @loop_cond
@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %cnt
}
