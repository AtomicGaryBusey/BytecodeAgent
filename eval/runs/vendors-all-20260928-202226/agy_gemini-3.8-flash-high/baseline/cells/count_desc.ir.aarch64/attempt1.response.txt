export function l $count_desc(l %a, l %n) {
@start
    %zero =l copy 0
    %limit =l sub %n, 1
    jmp @loop
@loop
    %i =l phi @start %zero, @body %i_next
    %count =l phi @start %zero, @body %count_next
    %cond =w csltl %i, %limit
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %p0 =l add %a, %offset
    %v0 =l loadl %p0
    %p1 =l add %p0, 8
    %v1 =l loadl %p1
    %is_gt =w csgtl %v0, %v1
    %inc =l extuw %is_gt
    %count_next =l add %count, %inc
    %i_next =l add %i, 1
    jmp @loop
@end
    ret %count
}
