bubble_sort:
    subs x3, x1, #1
    b.le .Ldone
.Louter:
    mov x4, #0
.Linner:
    ldr x5, [x0, x4, lsl #3]
    add x7, x4, #1
    ldr x6, [x0, x7, lsl #3]
    cmp x5, x6
    b.le .Lnoswap
    str x6, [x0, x4, lsl #3]
    str x5, [x0, x7, lsl #3]
.Lnoswap:
    mov x4, x7
    cmp x4, x3
    b.lt .Linner
    subs x3, x3, #1
    b.gt .Louter
.Ldone:
    mov x0, #0
    ret
