export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	%p0 =l add %a, 8
	%rem0 =l sub %n, 1
	%cond0 =w csgtl %rem0, 0
	jnz %cond0, @loop, @done
@loop
	%p =l phi @start %p0, @loop_end %p_next
	%rem =l phi @start %rem0, @loop_end %rem_next
	%min_cur =l phi @start %y0, @loop_end %min_next
	%max_cur =l phi @start %y0, @loop_end %max_next
	%sum_cur =l phi @start %y0, @loop_end %sum_next
	%cnt_cur =l phi @start %odd0, @loop_end %cnt_next
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum_next =l add %sum_cur, %y
	%odd =l and %y, 1
	%cnt_next =l add %cnt_cur, %odd
	%lt =w csltl %y, %min_cur
	jnz %lt, @min_is_y, @check_max
@min_is_y
	jmp @check_max
@check_max
	%min_next =l phi @loop %min_cur, @min_is_y %y
	%gt =w csgtl %y, %max_cur
	jnz %gt, @max_is_y, @loop_end
@max_is_y
	jmp @loop_end
@loop_end
	%max_next =l phi @check_max %max_cur, @max_is_y %y
	%p_next =l add %p, 8
	%rem_next =l sub %rem, 1
	%cont =w csgtl %rem_next, 0
	jnz %cont, @loop, @done
@done
	%min_fin =l phi @start %y0, @loop_end %min_next
	%max_fin =l phi @start %y0, @loop_end %max_next
	%sum_fin =l phi @start %y0, @loop_end %sum_next
	%cnt_fin =l phi @start %odd0, @loop_end %cnt_next
	storel %min_fin, %out
	%out1 =l add %out, 8
	storel %max_fin, %out1
	%out2 =l add %out, 16
	storel %sum_fin, %out2
	ret %cnt_fin
}
