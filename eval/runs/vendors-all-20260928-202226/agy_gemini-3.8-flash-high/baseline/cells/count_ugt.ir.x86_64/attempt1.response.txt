export function l $count_ugt(l %a, l %n, l %t) {
@start
	%cnt0 =l copy 0
	jmp @loop
@loop
	%cur_a =l phi @start %a, @body %next_a
	%rem =l phi @start %n, @body %next_rem
	%cnt =l phi @start %cnt0, @body %next_cnt
	%cond =w csgtl %rem, 0
	jnz %cond, @body, @done
@body
	%val =l loadl %cur_a
	%cmp =w cugtl %val, %t
	%inc =l extuw %cmp
	%next_cnt =l add %cnt, %inc
	%next_a =l add %cur_a, 8
	%next_rem =l sub %rem, 1
	jmp @loop
@done
	ret %cnt
}
