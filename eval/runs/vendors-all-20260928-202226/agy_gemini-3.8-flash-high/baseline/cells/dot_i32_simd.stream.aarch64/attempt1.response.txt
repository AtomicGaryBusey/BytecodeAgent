dot_i32_simd:
    mov x3, #0
    lsr x4, x2, #2
    cbz x4, .Lscalar_check
    movi v2.2d, #0
    movi v3.2d, #0
.Lvector_loop:
    ldr q0, [x0], #16
    ldr q1, [x1], #16
    smlal v2.2d, v0.2s, v1.2s
    smlal2 v3.2d, v0.4s, v1.4s
    subs x4, x4, #1
    b.ne .Lvector_loop
    add v2.2d, v2.2d, v3.2d
    fmov x4, d2
    umov x5, v2.d[1]
    add x3, x4, x5
.Lscalar_check:
    ands x2, x2, #3
    b.eq .Ldone
.Lscalar_loop:
    ldr w4, [x0], #4
    ldr w5, [x1], #4
    smaddl x3, w4, w5, x3
    subs x2, x2, #1
    b.ne .Lscalar_loop
.Ldone:
    mov x0, x3
    ret
