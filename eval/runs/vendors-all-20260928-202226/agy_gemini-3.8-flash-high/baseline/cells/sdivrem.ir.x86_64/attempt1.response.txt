export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%is_zero =w ceql %b, 0
	jnz %is_zero, @b_zero, @check_neg1
@check_neg1
	%is_neg1 =w ceql %b, -1
	jnz %is_neg1, @b_neg1, @do_div
@b_zero
	storel %a, %rem
	ret 0
@b_neg1
	%q_neg1 =l sub 0, %a
	storel 0, %rem
	ret %q_neg1
@do_div
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
