memmove_i64:
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
.Lbackward:
    mov rcx, rdx
.Lback_loop:
    dec rcx
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    test rcx, rcx
    jnz .Lback_loop
    jmp .Ldone
.Lforward:
    xor ecx, ecx
.Lfwd_loop:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jne .Lfwd_loop
.Ldone:
    xor eax, eax
    ret
