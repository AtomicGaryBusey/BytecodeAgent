export function l $popcount(l %x) {
@start
    %a1 =l shr %x, 1
    %a2 =l and %a1, 0x5555555555555555
    %a3 =l sub %x, %a2
    %b1 =l and %a3, 0x3333333333333333
    %b2 =l shr %a3, 2
    %b3 =l and %b2, 0x3333333333333333
    %b4 =l add %b1, %b3
    %c1 =l shr %b4, 4
    %c2 =l add %b4, %c1
    %c3 =l and %c2, 0x0f0f0f0f0f0f0f0f
    %d1 =l mul %c3, 0x0101010101010101
    %res =l shr %d1, 56
    ret %res
}
