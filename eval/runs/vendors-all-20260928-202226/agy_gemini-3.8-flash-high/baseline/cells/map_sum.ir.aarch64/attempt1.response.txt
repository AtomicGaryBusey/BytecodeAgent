export function l $map_sum(l %a, l %n, l %f) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %i1
    %sum =l phi @start 0, @body %sum1
    %p =l phi @start %a, @body %p1
    %c =w csltl %i, %n
    jnz %c, @body, @done
@body
    %x =l loadl %p
    %fx =l call %f(l %x)
    %sum1 =l add %sum, %fx
    %p1 =l add %p, 8
    %i1 =l add %i, 1
    jmp @loop
@done
    ret %sum
}
