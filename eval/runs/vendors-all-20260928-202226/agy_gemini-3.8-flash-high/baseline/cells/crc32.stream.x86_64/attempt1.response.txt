crc32:
    mov eax, 0xffffffff
    test rsi, rsi
    jle .Ldone
    mov edx, 0xedb88320
.Lbyte_loop:
    xor al, byte ptr [rdi]
    inc rdi
    mov ecx, 8
.Lbit_loop:
    shr eax, 1
    jnc .Lno_xor
    xor eax, edx
.Lno_xor:
    dec ecx
    jnz .Lbit_loop
    dec rsi
    jnz .Lbyte_loop
.Ldone:
    not eax
    ret
