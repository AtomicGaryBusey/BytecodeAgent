export function l $binary_search(l %a, l %n, l %key) {
@start
    %low_init =l copy 0
    jmp @loop
@loop
    %low =l phi @start %low_init, @loop_next %low_next
    %high =l phi @start %n, @loop_next %high_next
    %cond =w csltl %low, %high
    jnz %cond, @body, @not_found
@body
    %diff =l sub %high, %low
    %half =l div %diff, 2
    %mid =l add %low, %half
    %off =l mul %mid, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %is_eq =w ceql %val, %key
    jnz %is_eq, @found, @check_lt
@found
    ret %mid
@check_lt
    %is_lt =w csltl %val, %key
    jnz %is_lt, @set_low, @set_high
@set_low
    %low_new =l add %mid, 1
    jmp @loop_next
@set_high
    jmp @loop_next
@loop_next
    %low_next =l phi @set_low %low_new, @set_high %low
    %high_next =l phi @set_low %high, @set_high %mid
    jmp @loop
@not_found
    %not_found_val =l copy -1
    ret %not_found_val
}
