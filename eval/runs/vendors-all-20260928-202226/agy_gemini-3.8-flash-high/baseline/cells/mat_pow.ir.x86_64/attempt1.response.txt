export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %zero =l copy 0
    %one =l copy 1
    %n2 =l mul %n, %n
    %n2x2 =l mul %n2, 2
    %np1 =l add %n, 1
    jmp @zero_out_loop

@zero_out_loop
    %z_idx =l phi @start 0, @zero_out_body %z_idx_next
    %z_cond =w csltl %z_idx, %n2
    jnz %z_cond, @zero_out_body, @diag_loop

@zero_out_body
    %z_byte =l mul %z_idx, 8
    %z_ptr =l add %out, %z_byte
    storel %zero, %z_ptr
    %z_idx_next =l add %z_idx, 1
    jmp @zero_out_loop

@diag_loop
    %d_i =l phi @zero_out_loop 0, @diag_body %d_i_next
    %d_cond =w csltl %d_i, %n
    jnz %d_cond, @diag_body, @pow_loop

@diag_body
    %d_idx =l mul %d_i, %np1
    %d_byte =l mul %d_idx, 8
    %d_ptr =l add %out, %d_byte
    storel %one, %d_ptr
    %d_i_next =l add %d_i, 1
    jmp @diag_loop

@pow_loop
    %step =l phi @diag_loop 0, @copy_done %step_next
    %pow_cond =w csltl %step, %k
    jnz %pow_cond, @loop_i, @zero_tmp_loop

@loop_i
    %i_val =l phi @pow_loop 0, @loop_i_latch %i_next
    %i_cond =w csltl %i_val, %n
    %i_mul_n =l mul %i_val, %n
    jnz %i_cond, @loop_j, @copy_loop

@loop_j
    %j_val =l phi @loop_i 0, @loop_j_latch %j_next
    %j_cond =w csltl %j_val, %n
    jnz %j_cond, @loop_t, @loop_i_latch

@loop_i_latch
    %i_next =l add %i_val, 1
    jmp @loop_i

@loop_t
    %t_val =l phi @loop_j 0, @loop_t_body %t_next
    %sum_val =l phi @loop_j 0, @loop_t_body %sum_next
    %t_cond =w csltl %t_val, %n
    jnz %t_cond, @loop_t_body, @loop_j_latch

@loop_t_body
    %out_idx =l add %i_mul_n, %t_val
    %out_off =l mul %out_idx, 8
    %out_ptr =l add %out, %out_off
    %out_elem =l loadl %out_ptr
    %t_mul_n =l mul %t_val, %n
    %a_idx =l add %t_mul_n, %j_val
    %a_off =l mul %a_idx, 8
    %a_ptr =l add %a, %a_off
    %a_elem =l loadl %a_ptr
    %prod =l mul %out_elem, %a_elem
    %sum_next =l add %sum_val, %prod
    %t_next =l add %t_val, 1
    jmp @loop_t

@loop_j_latch
    %tmp_idx =l add %i_mul_n, %j_val
    %tmp_off =l mul %tmp_idx, 8
    %tmp_ptr =l add %tmp, %tmp_off
    storel %sum_val, %tmp_ptr
    %j_next =l add %j_val, 1
    jmp @loop_j

@copy_loop
    %c_idx =l phi @loop_i 0, @copy_body %c_idx_next
    %c_cond =w csltl %c_idx, %n2
    jnz %c_cond, @copy_body, @copy_done

@copy_body
    %c_off =l mul %c_idx, 8
    %c_tmp_ptr =l add %tmp, %c_off
    %c_val =l loadl %c_tmp_ptr
    %c_out_ptr =l add %out, %c_off
    storel %c_val, %c_out_ptr
    %c_idx_next =l add %c_idx, 1
    jmp @copy_loop

@copy_done
    %step_next =l add %step, 1
    jmp @pow_loop

@zero_tmp_loop
    %zt_idx =l phi @pow_loop 0, @zero_tmp_body %zt_idx_next
    %zt_cond =w csltl %zt_idx, %n2x2
    jnz %zt_cond, @zero_tmp_body, @done

@zero_tmp_body
    %zt_off =l mul %zt_idx, 8
    %zt_ptr =l add %tmp, %zt_off
    storel %zero, %zt_ptr
    %zt_idx_next =l add %zt_idx, 1
    jmp @zero_tmp_loop

@done
    ret %zero
}
