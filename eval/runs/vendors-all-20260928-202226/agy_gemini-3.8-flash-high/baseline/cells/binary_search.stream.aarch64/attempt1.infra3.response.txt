binary_search:
    mov x3, #0
.Lloop:
    cmp x3, x1
    b.ge .Lnot_found
    add x4, x3, x1
    lsr x4, x4, #1
    ldr x5, [x0, x4, lsl #3]
    cmp x5, x2
    b.eq .Lfound
    b.lt .Lless
    mov x1, x4
    b .Lloop
.Lless:
    add x3, x4, #1
    b .Lloop
.Lfound:
    mov x0, x4
    ret
.Lnot_found:
    mov x0, #-1
    ret
