vm_run:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, 5
    jg .Lop_gt_5
    cmp rax, 2
    jg .Lop_3_4_5
    test rax, rax
    jz .Ldone
    cmp rax, 1
    je .Lop_push
.Lop_add:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    add qword ptr [rdx + r9*8 - 8], r10
    inc r8
    jmp .Lloop
.Lop_3_4_5:
    cmp rax, 4
    jl .Lop_sub
    je .Lop_mul
    jmp .Lop_dup
.Lop_gt_5:
    cmp rax, 8
    jg .Lop_gt_8
    cmp rax, 7
    jl .Lop_swap
    je .Lop_drop
    jmp .Lop_over
.Lop_gt_8:
    cmp rax, 10
    jl .Lop_lt
    je .Lop_jz
    jmp .Lop_jmp
.Lop_push:
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 2
    jmp .Lloop
.Lop_sub:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    sub qword ptr [rdx + r9*8 - 8], r10
    inc r8
    jmp .Lloop
.Lop_mul:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    imul r10, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8 - 8], r10
    inc r8
    jmp .Lloop
.Lop_dup:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Lop_swap:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 8], r11
    mov qword ptr [rdx + r9*8 - 16], r10
    inc r8
    jmp .Lloop
.Lop_drop:
    dec r9
    inc r8
    jmp .Lloop
.Lop_over:
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    inc r8
    jmp .Lloop
.Lop_lt:
    dec r9
    mov r10, qword ptr [rdx + r9*8 - 8]
    cmp r10, qword ptr [rdx + r9*8]
    setl r11b
    movzx r11d, r11b
    mov qword ptr [rdx + r9*8 - 8], r11
    inc r8
    jmp .Lloop
.Lop_jz:
    dec r9
    mov r10, qword ptr [rdi + r8*8 + 8]
    add r8, 2
    cmp qword ptr [rdx + r9*8], 0
    cmove r8, r10
    jmp .Lloop
.Lop_jmp:
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Ldone:
    xor eax, eax
    test r9, r9
    jz .Lzero_loop
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lzero_loop:
    cmp r9, rcx
    jge .Lexit
    mov qword ptr [rdx + r9*8], 0
    inc r9
    jmp .Lzero_loop
.Lexit:
    ret
