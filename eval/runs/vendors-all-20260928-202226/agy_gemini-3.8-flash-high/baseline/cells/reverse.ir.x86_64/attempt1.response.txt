export function l $reverse(l %a, l %n) {
@start
    %cmp =w cslel %n, 1
    jnz %cmp, @done, @init
@init
    %n1 =l sub %n, 1
    %bytes =l mul %n1, 8
    %end =l add %a, %bytes
    jmp @loop
@loop
    %left =l phi @init %a, @body %next_left
    %right =l phi @init %end, @body %prev_right
    %c =w cultl %left, %right
    jnz %c, @body, @done
@body
    %t1 =l loadl %left
    %t2 =l loadl %right
    storel %t2, %left
    storel %t1, %right
    %next_left =l add %left, 8
    %prev_right =l sub %right, 8
    jmp @loop
@done
    ret 0
}
