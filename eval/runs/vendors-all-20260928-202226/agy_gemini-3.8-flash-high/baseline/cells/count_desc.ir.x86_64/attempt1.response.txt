export function l $count_desc(l %a, l %n) {
@start
    %has_pairs =w csgtl %n, 1
    jnz %has_pairs, @init, @ret_zero
@ret_zero
    ret 0
@init
    %prev0 =l loadl %a
    %ptr0 =l add %a, 8
    %count0 =l copy 0
    %j0 =l copy 1
    jmp @loop
@loop
    %count =l phi @init %count0, @loop %count1
    %j =l phi @init %j0, @loop %j1
    %ptr =l phi @init %ptr0, @loop %ptr1
    %prev =l phi @init %prev0, @loop %curr
    %curr =l loadl %ptr
    %is_desc =w csgtl %prev, %curr
    %is_desc_l =l extuw %is_desc
    %count1 =l add %count, %is_desc_l
    %j1 =l add %j, 1
    %ptr1 =l add %ptr, 8
    %cont =w csltl %j1, %n
    jnz %cont, @loop, @end
@end
    ret %count1
}
