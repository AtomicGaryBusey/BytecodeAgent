export function l $qsort_pairs(l %a, l %n) {
@start
    %c_base =w csgtl %n, 1
    jnz %c_base, @init, @ret0
@ret0
    ret 0
@init
    %neg1 =l copy -1
    %t1 =l sub %n, 1
    %mid =l sar %t1, 1
    %mid16 =l shl %mid, 4
    %pmid =l add %a, %mid16
    %pkey =l loadl %pmid
    %pmid_v =l add %pmid, 8
    %pval =l loadl %pmid_v
    jmp @loop_start
@loop_start
    %i_curr =l phi @init %neg1, @swap %i_next
    %j_curr =l phi @init %n, @swap %j_next
    jmp @i_loop
@i_loop
    %i_in =l phi @loop_start %i_curr, @i_inc %i_next
    %i_next =l add %i_in, 1
    %i16 =l shl %i_next, 4
    %pi =l add %a, %i16
    %ki =l loadl %pi
    %pi8 =l add %pi, 8
    %vi =l loadl %pi8
    %c_klt =w csltl %ki, %pkey
    jnz %c_klt, @i_inc, @i_chk_eq
@i_chk_eq
    %c_keq =w ceql %ki, %pkey
    jnz %c_keq, @i_chk_v, @i_done
@i_chk_v
    %c_vlt =w csltl %vi, %pval
    jnz %c_vlt, @i_inc, @i_done
@i_inc
    jmp @i_loop
@i_done
    jmp @j_loop
@j_loop
    %j_in =l phi @i_done %j_curr, @j_dec %j_next
    %j_next =l sub %j_in, 1
    %j16 =l shl %j_next, 4
    %pj =l add %a, %j16
    %kj =l loadl %pj
    %pj8 =l add %pj, 8
    %vj =l loadl %pj8
    %c_kgt =w csgtl %kj, %pkey
    jnz %c_kgt, @j_dec, @j_chk_eq
@j_chk_eq
    %c_jkeq =w ceql %kj, %pkey
    jnz %c_jkeq, @j_chk_v, @j_done
@j_chk_v
    %c_vgt =w csgtl %vj, %pval
    jnz %c_vgt, @j_dec, @j_done
@j_dec
    jmp @j_loop
@j_done
    %c_lt =w csltl %i_next, %j_next
    jnz %c_lt, @swap, @partition_done
@swap
    %i16_s =l shl %i_next, 4
    %pi_s =l add %a, %i16_s
    %pi8_s =l add %pi_s, 8
    %j16_s =l shl %j_next, 4
    %pj_s =l add %a, %j16_s
    %pj8_s =l add %pj_s, 8
    %ki_s =l loadl %pi_s
    %vi_s =l loadl %pi8_s
    %kj_s =l loadl %pj_s
    %vj_s =l loadl %pj8_s
    storel %kj_s, %pi_s
    storel %vj_s, %pi8_s
    storel %ki_s, %pj_s
    storel %vi_s, %pj8_s
    jmp @loop_start
@partition_done
    %len1 =l add %j_next, 1
    %len1_16 =l shl %len1, 4
    %ptr2 =l add %a, %len1_16
    %len2 =l sub %n, %len1
    %r1 =l call $qsort_pairs(l %a, l %len1)
    %r2 =l call $qsort_pairs(l %ptr2, l %len2)
    ret 0
}
