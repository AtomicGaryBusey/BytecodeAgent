export function l $collatz_max(l %n) {
@start
    jmp @outer_loop
@outer_loop
    %k =l phi @start 1, @outer_cont %k_next
    %max_so_far =l phi @start 0, @outer_cont %max_next
    %cond_outer =w cslel %k, %n
    jnz %cond_outer, @inner_init, @outer_end
@inner_init
    jmp @inner_loop
@inner_loop
    %cur =l phi @inner_init %k, @inner_cont %cur_next
    %steps =l phi @inner_init 0, @inner_cont %steps_next
    %is_one =w ceql %cur, 1
    jnz %is_one, @inner_done, @inner_step
@inner_step
    %steps_next =l add %steps, 1
    %rem =l and %cur, 1
    %is_even =w ceql %rem, 0
    jnz %is_even, @inner_even, @inner_odd
@inner_even
    %cur_even =l shr %cur, 1
    jmp @inner_cont
@inner_odd
    %t1 =l mul %cur, 3
    %cur_odd =l add %t1, 1
    jmp @inner_cont
@inner_cont
    %cur_next =l phi @inner_even %cur_even, @inner_odd %cur_odd
    jmp @inner_loop
@inner_done
    %is_greater =w csgtl %steps, %max_so_far
    jnz %is_greater, @new_max, @keep_max
@new_max
    jmp @outer_cont
@keep_max
    jmp @outer_cont
@outer_cont
    %max_next =l phi @new_max %steps, @keep_max %max_so_far
    %k_next =l add %k, 1
    jmp @outer_loop
@outer_end
    ret %max_so_far
}
