export function l $bmin_empty(l %a, l %n) {
@start
	%is_zero =w ceql %n, 0
	jnz %is_zero, @ret_max, @init
@init
	%val0 =l loadl %a
	%rem0 =l sub %n, 1
	%ptr0 =l add %a, 8
	jmp @loop
@loop
	%min =l phi @init %val0, @loop_next %min_next
	%rem =l phi @init %rem0, @loop_next %rem_next
	%ptr =l phi @init %ptr0, @loop_next %ptr_next
	%cond =w ceql %rem, 0
	jnz %cond, @ret_min, @body
@body
	%elem =l loadl %ptr
	%rem_next =l sub %rem, 1
	%ptr_next =l add %ptr, 8
	%smaller =w csltl %elem, %min
	jnz %smaller, @take_elem, @keep_min
@take_elem
	jmp @loop_next
@keep_min
	jmp @loop_next
@loop_next
	%min_next =l phi @take_elem %elem, @keep_min %min
	jmp @loop
@ret_min
	ret %min
@ret_max
	ret 9223372036854775807
}
