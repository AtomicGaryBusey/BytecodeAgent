export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %p_i =l alloc8 8
    %p_out_len =l alloc8 8
    %p_max_k =l alloc8 8
    %p_k =l alloc8 8
    %p_j =l alloc8 8
    %p_copy_k =l alloc8 8
    storel 0, %p_i
    storel 0, %p_out_len
    jmp @outer_cond
@outer_cond
    %cur_i =l loadl %p_i
    %i_lt_n =w csltl %cur_i, %n
    jnz %i_lt_n, @calc_run_i, @return_ok
@calc_run_i
    %ri_pos =l loadl %p_i
    %ri_src_ptr =l add %src, %ri_pos
    %ri_b =w loadub %ri_src_ptr
    %ri_max1 =l add %ri_pos, 128
    %ri_cmp_n =w csgtl %ri_max1, %n
    jnz %ri_cmp_n, @ri_cap_n, @ri_max_ok
@ri_cap_n
    storel %n, %p_max_k
    jmp @ri_loop_init
@ri_max_ok
    storel %ri_max1, %p_max_k
    jmp @ri_loop_init
@ri_loop_init
    %ri_k_init =l add %ri_pos, 1
    storel %ri_k_init, %p_k
    jmp @ri_loop_cond
@ri_loop_cond
    %ri_cur_k =l loadl %p_k
    %ri_max_k =l loadl %p_max_k
    %ri_k_lt_max =w csltl %ri_cur_k, %ri_max_k
    jnz %ri_k_lt_max, @ri_check_byte, @ri_done
@ri_check_byte
    %ri_k_ptr =l add %src, %ri_cur_k
    %ri_k_byte =w loadub %ri_k_ptr
    %ri_match =w ceqw %ri_k_byte, %ri_b
    jnz %ri_match, @ri_inc_k, @ri_done
@ri_inc_k
    %ri_next_k =l add %ri_cur_k, 1
    storel %ri_next_k, %p_k
    jmp @ri_loop_cond
@ri_done
    %ri_final_k =l loadl %p_k
    %run_i =l sub %ri_final_k, %ri_pos
    %ge_3 =w csgel %run_i, 3
    jnz %ge_3, @emit_repeat, @start_literal
@emit_repeat
    %rep_out_len =l loadl %p_out_len
    %rep_need =l add %rep_out_len, 2
    %rep_fits =w cslel %rep_need, %cap
    jnz %rep_fits, @do_emit_repeat, @return_err
@do_emit_repeat
    %run_i_w =w copy %run_i
    %hdr_rep =w sub 257, %run_i_w
    %out_ptr0 =l add %out, %rep_out_len
    storeb %hdr_rep, %out_ptr0
    %out_ptr1 =l add %out_ptr0, 1
    storeb %ri_b, %out_ptr1
    storel %rep_need, %p_out_len
    %new_i =l add %ri_pos, %run_i
    storel %new_i, %p_i
    jmp @outer_cond
@start_literal
    %lit_i =l loadl %p_i
    %lit_j_init =l add %lit_i, 1
    storel %lit_j_init, %p_j
    jmp @lit_loop_cond
@lit_loop_cond
    %cur_j =l loadl %p_j
    %j_lt_n =w csltl %cur_j, %n
    jnz %j_lt_n, @lit_check_len, @lit_scan_done
@lit_check_len
    %cur_i_in_lit =l loadl %p_i
    %lit_len =l sub %cur_j, %cur_i_in_lit
    %len_lt_128 =w csltl %lit_len, 128
    jnz %len_lt_128, @calc_run_j, @lit_scan_done
@calc_run_j
    %rj_src_ptr =l add %src, %cur_j
    %rj_b =w loadub %rj_src_ptr
    %rj_max1 =l add %cur_j, 128
    %rj_cmp_n =w csgtl %rj_max1, %n
    jnz %rj_cmp_n, @rj_cap_n, @rj_max_ok
@rj_cap_n
    storel %n, %p_max_k
    jmp @rj_loop_init
@rj_max_ok
    storel %rj_max1, %p_max_k
    jmp @rj_loop_init
@rj_loop_init
    %rj_k_init =l add %cur_j, 1
    storel %rj_k_init, %p_k
    jmp @rj_loop_cond
@rj_loop_cond
    %rj_cur_k =l loadl %p_k
    %rj_max_k =l loadl %p_max_k
    %rj_k_lt_max =w csltl %rj_cur_k, %rj_max_k
    jnz %rj_k_lt_max, @rj_check_byte, @rj_done
@rj_check_byte
    %rj_k_ptr =l add %src, %rj_cur_k
    %rj_k_byte =w loadub %rj_k_ptr
    %rj_match =w ceqw %rj_k_byte, %rj_b
    jnz %rj_match, @rj_inc_k, @rj_done
@rj_inc_k
    %rj_next_k =l add %rj_cur_k, 1
    storel %rj_next_k, %p_k
    jmp @rj_loop_cond
@rj_done
    %rj_final_k =l loadl %p_k
    %run_j =l sub %rj_final_k, %cur_j
    %rj_ge_3 =w csgel %run_j, 3
    jnz %rj_ge_3, @lit_scan_done, @lit_inc_j
@lit_inc_j
    %next_j =l add %cur_j, 1
    storel %next_j, %p_j
    jmp @lit_loop_cond
@lit_scan_done
    %final_j =l loadl %p_j
    %lit_start_i =l loadl %p_i
    %lit_L =l sub %final_j, %lit_start_i
    %lit_pkt_size =l add %lit_L, 1
    %cur_out_len =l loadl %p_out_len
    %lit_need =l add %cur_out_len, %lit_pkt_size
    %lit_fits =w cslel %lit_need, %cap
    jnz %lit_fits, @do_emit_literal, @return_err
@do_emit_literal
    %lit_L_w =w copy %lit_L
    %hdr_lit =w sub %lit_L_w, 1
    %lit_out_ptr =l add %out, %cur_out_len
    storeb %hdr_lit, %lit_out_ptr
    storel 0, %p_copy_k
    jmp @copy_loop_cond
@copy_loop_cond
    %cur_copy_k =l loadl %p_copy_k
    %k_lt_L =w csltl %cur_copy_k, %lit_L
    jnz %k_lt_L, @copy_byte, @copy_done
@copy_byte
    %src_off1 =l add %lit_start_i, %cur_copy_k
    %src_addr =l add %src, %src_off1
    %byte_to_copy =w loadub %src_addr
    %dst_off1 =l add %cur_out_len, 1
    %dst_off2 =l add %dst_off1, %cur_copy_k
    %dst_addr =l add %out, %dst_off2
    storeb %byte_to_copy, %dst_addr
    %next_copy_k =l add %cur_copy_k, 1
    storel %next_copy_k, %p_copy_k
    jmp @copy_loop_cond
@copy_done
    storel %lit_need, %p_out_len
    storel %final_j, %p_i
    jmp @outer_cond
@return_ok
    %ret_len =l loadl %p_out_len
    ret %ret_len
@return_err
    %err =l sub 0, 1
    ret %err
}
