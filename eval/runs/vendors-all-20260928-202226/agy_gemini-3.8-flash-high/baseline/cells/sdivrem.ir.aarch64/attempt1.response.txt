export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %cz =w ceql %b, 0
    jnz %cz, @b_zero, @check_neg1
@check_neg1
    %cn =w ceql %b, -1
    jnz %cn, @b_neg1, @normal
@b_zero
    storel %a, %rem
    ret 0
@b_neg1
    %qn =l sub 0, %a
    %r0 =l copy 0
    storel %r0, %rem
    ret %qn
@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
