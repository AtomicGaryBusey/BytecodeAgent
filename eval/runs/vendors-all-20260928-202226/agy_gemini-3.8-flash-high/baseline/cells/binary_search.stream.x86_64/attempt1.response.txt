binary_search:
    xor ecx, ecx
.Lloop:
    cmp rcx, rsi
    jge .Lnot_found
    mov rax, rcx
    add rax, rsi
    shr rax, 1
    mov r8, qword ptr [rdi + rax*8]
    cmp r8, rdx
    je .Ldone
    jl .Lless
    mov rsi, rax
    jmp .Lloop
.Lless:
    lea rcx, [rax + 1]
    jmp .Lloop
.Lnot_found:
    mov rax, -1
.Ldone:
    ret
