export function l $bmax(l %a, l %n) {
@start
    %max0 =l loadl %a
    %ptr0 =l add %a, 8
    %rem0 =l sub %n, 1
@loop
    %max =l phi @start %max0, @next %max_next
    %ptr =l phi @start %ptr0, @next %ptr_next
    %rem =l phi @start %rem0, @next %rem_next
    %done =w ceql %rem, 0
    jnz %done, @end, @body
@body
    %val =l loadl %ptr
    %gt =w csgtl %val, %max
    jnz %gt, @take_new, @keep_old
@take_new
    jmp @next
@keep_old
    jmp @next
@next
    %max_next =l phi @take_new %val, @keep_old %max
    %ptr_next =l add %ptr, 8
    %rem_next =l sub %rem, 1
    jmp @loop
@end
    ret %max
}
