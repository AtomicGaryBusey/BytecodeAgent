export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pc_loc =l alloc8 8
    %sp_loc =l alloc8 8
    %clr_loc =l alloc8 8
    %ret_loc =l alloc8 8
    storel 0, %pc_loc
    storel 0, %sp_loc
    storel 0, %clr_loc
    storel 0, %ret_loc
    jmp @loop
@loop
    %pc =l loadl %pc_loc
    %sp =l loadl %sp_loc
    %stop =w csgel %pc, %len
    jnz %stop, @done, @decode
@decode
    %pc_off =l mul %pc, 8
    %pc_ptr =l add %code, %pc_off
    %op =l loadl %pc_ptr
    %c0 =w ceql %op, 0
    jnz %c0, @op_halt, @check_1
@check_1
    %c1 =w ceql %op, 1
    jnz %c1, @op_push, @check_2
@check_2
    %c2 =w ceql %op, 2
    jnz %c2, @op_add, @check_3
@check_3
    %c3 =w ceql %op, 3
    jnz %c3, @op_sub, @check_4
@check_4
    %c4 =w ceql %op, 4
    jnz %c4, @op_mul, @check_5
@check_5
    %c5 =w ceql %op, 5
    jnz %c5, @op_dup, @check_6
@check_6
    %c6 =w ceql %op, 6
    jnz %c6, @op_swap, @check_7
@check_7
    %c7 =w ceql %op, 7
    jnz %c7, @op_drop, @check_8
@check_8
    %c8 =w ceql %op, 8
    jnz %c8, @op_over, @check_9
@check_9
    %c9 =w ceql %op, 9
    jnz %c9, @op_lt, @check_10
@check_10
    %c10 =w ceql %op, 10
    jnz %c10, @op_jz, @op_jmp
@op_halt
    jmp @done
@op_push
    %pc1_1 =l add %pc, 1
    %pc1_off =l mul %pc1_1, 8
    %pc1_ptr =l add %code, %pc1_off
    %v1 =l loadl %pc1_ptr
    %sp1_off =l mul %sp, 8
    %sp1_ptr =l add %stack, %sp1_off
    storel %v1, %sp1_ptr
    %sp1_next =l add %sp, 1
    storel %sp1_next, %sp_loc
    %pc1_next =l add %pc, 2
    storel %pc1_next, %pc_loc
    jmp @loop
@op_add
    %sp2_1 =l sub %sp, 1
    %b2_off =l mul %sp2_1, 8
    %b2_ptr =l add %stack, %b2_off
    %b2 =l loadl %b2_ptr
    %sp2_2 =l sub %sp, 2
    %a2_off =l mul %sp2_2, 8
    %a2_ptr =l add %stack, %a2_off
    %a2 =l loadl %a2_ptr
    %res2 =l add %a2, %b2
    storel %res2, %a2_ptr
    storel %sp2_1, %sp_loc
    %pc2_next =l add %pc, 1
    storel %pc2_next, %pc_loc
    jmp @loop
@op_sub
    %sp3_1 =l sub %sp, 1
    %b3_off =l mul %sp3_1, 8
    %b3_ptr =l add %stack, %b3_off
    %b3 =l loadl %b3_ptr
    %sp3_2 =l sub %sp, 2
    %a3_off =l mul %sp3_2, 8
    %a3_ptr =l add %stack, %a3_off
    %a3 =l loadl %a3_ptr
    %res3 =l sub %a3, %b3
    storel %res3, %a3_ptr
    storel %sp3_1, %sp_loc
    %pc3_next =l add %pc, 1
    storel %pc3_next, %pc_loc
    jmp @loop
@op_mul
    %sp4_1 =l sub %sp, 1
    %b4_off =l mul %sp4_1, 8
    %b4_ptr =l add %stack, %b4_off
    %b4 =l loadl %b4_ptr
    %sp4_2 =l sub %sp, 2
    %a4_off =l mul %sp4_2, 8
    %a4_ptr =l add %stack, %a4_off
    %a4 =l loadl %a4_ptr
    %res4 =l mul %a4, %b4
    storel %res4, %a4_ptr
    storel %sp4_1, %sp_loc
    %pc4_next =l add %pc, 1
    storel %pc4_next, %pc_loc
    jmp @loop
@op_dup
    %sp5_1 =l sub %sp, 1
    %top5_off =l mul %sp5_1, 8
    %top5_ptr =l add %stack, %top5_off
    %top5 =l loadl %top5_ptr
    %sp5_off =l mul %sp, 8
    %sp5_ptr =l add %stack, %sp5_off
    storel %top5, %sp5_ptr
    %sp5_next =l add %sp, 1
    storel %sp5_next, %sp_loc
    %pc5_next =l add %pc, 1
    storel %pc5_next, %pc_loc
    jmp @loop
@op_swap
    %sp6_1 =l sub %sp, 1
    %b6_off =l mul %sp6_1, 8
    %b6_ptr =l add %stack, %b6_off
    %b6 =l loadl %b6_ptr
    %sp6_2 =l sub %sp, 2
    %a6_off =l mul %sp6_2, 8
    %a6_ptr =l add %stack, %a6_off
    %a6 =l loadl %a6_ptr
    storel %b6, %a6_ptr
    storel %a6, %b6_ptr
    %pc6_next =l add %pc, 1
    storel %pc6_next, %pc_loc
    jmp @loop
@op_drop
    %sp7_next =l sub %sp, 1
    storel %sp7_next, %sp_loc
    %pc7_next =l add %pc, 1
    storel %pc7_next, %pc_loc
    jmp @loop
@op_over
    %sp8_2 =l sub %sp, 2
    %a8_off =l mul %sp8_2, 8
    %a8_ptr =l add %stack, %a8_off
    %a8 =l loadl %a8_ptr
    %sp8_off =l mul %sp, 8
    %sp8_ptr =l add %stack, %sp8_off
    storel %a8, %sp8_ptr
    %sp8_next =l add %sp, 1
    storel %sp8_next, %sp_loc
    %pc8_next =l add %pc, 1
    storel %pc8_next, %pc_loc
    jmp @loop
@op_lt
    %sp9_1 =l sub %sp, 1
    %b9_off =l mul %sp9_1, 8
    %b9_ptr =l add %stack, %b9_off
    %b9 =l loadl %b9_ptr
    %sp9_2 =l sub %sp, 2
    %a9_off =l mul %sp9_2, 8
    %a9_ptr =l add %stack, %a9_off
    %a9 =l loadl %a9_ptr
    %cond9 =w csltl %a9, %b9
    %res9 =l extuw %cond9
    storel %res9, %a9_ptr
    storel %sp9_1, %sp_loc
    %pc9_next =l add %pc, 1
    storel %pc9_next, %pc_loc
    jmp @loop
@op_jz
    %sp10_1 =l sub %sp, 1
    storel %sp10_1, %sp_loc
    %v10_off =l mul %sp10_1, 8
    %v10_ptr =l add %stack, %v10_off
    %v10 =l loadl %v10_ptr
    %is_zero10 =w ceql %v10, 0
    jnz %is_zero10, @jz_taken, @jz_not_taken
@jz_taken
    %pc10_1 =l add %pc, 1
    %t10_off =l mul %pc10_1, 8
    %t10_ptr =l add %code, %t10_off
    %t10 =l loadl %t10_ptr
    storel %t10, %pc_loc
    jmp @loop
@jz_not_taken
    %pc10_next =l add %pc, 2
    storel %pc10_next, %pc_loc
    jmp @loop
@op_jmp
    %pc11_1 =l add %pc, 1
    %t11_off =l mul %pc11_1, 8
    %t11_ptr =l add %code, %t11_off
    %t11 =l loadl %t11_ptr
    storel %t11, %pc_loc
    jmp @loop
@done
    %sp_fin =l loadl %sp_loc
    %has_items =w csgtl %sp_fin, 0
    jnz %has_items, @get_top, @empty_stack
@get_top
    %top_idx =l sub %sp_fin, 1
    %top_off =l mul %top_idx, 8
    %top_ptr =l add %stack, %top_off
    %top_val =l loadl %top_ptr
    storel %top_val, %ret_loc
    jmp @zero_init
@empty_stack
    jmp @zero_init
@zero_init
    storel %sp_fin, %clr_loc
    jmp @zero_loop
@zero_loop
    %cur_idx =l loadl %clr_loc
    %need_zero =w csltl %cur_idx, %cap
    jnz %need_zero, @zero_step, @finish
@zero_step
    %zero_off =l mul %cur_idx, 8
    %zero_ptr =l add %stack, %zero_off
    storel 0, %zero_ptr
    %next_idx =l add %cur_idx, 1
    storel %next_idx, %clr_loc
    jmp @zero_loop
@finish
    %final_ret =l loadl %ret_loc
    ret %final_ret
}
