crc32:
    movn w2, #0
    cbz x1, .Ldone
    movz w3, #0x8320
    movk w3, #0xedb8, lsl #16
.Lbyte_loop:
    ldrb w4, [x0], #1
    eor w2, w2, w4
    mov w5, #8
.Lbit_loop:
    tst w2, #1
    lsr w4, w2, #1
    eor w6, w4, w3
    csel w2, w6, w4, ne
    subs w5, w5, #1
    b.ne .Lbit_loop
    sub x1, x1, #1
    cbnz x1, .Lbyte_loop
.Ldone:
    mvn w0, w2
    ret
