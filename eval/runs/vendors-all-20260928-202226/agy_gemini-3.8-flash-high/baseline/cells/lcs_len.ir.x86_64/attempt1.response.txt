export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %zero =l copy 0
    %init_j_start =l copy 0
    jmp @init_loop
@init_loop
    %init_j =l phi @start %init_j_start, @init_body %init_j_next
    %init_cond =w cslel %init_j, %n
    jnz %init_cond, @init_body, @outer_init
@init_body
    %init_off =l mul %init_j, 8
    %init_ptr =l add %row, %init_off
    storel %zero, %init_ptr
    %init_j_next =l add %init_j, 1
    jmp @init_loop
@outer_init
    %i_start =l copy 0
    jmp @outer_loop
@outer_loop
    %i =l phi @outer_init %i_start, @outer_next %i_next
    %i_cond =w csltl %i, %m
    jnz %i_cond, @outer_body, @done
@outer_body
    %p_xi =l add %x, %i
    %xi =w loadub %p_xi
    %j_start =l copy 1
    %diag_start =l copy 0
    jmp @inner_loop
@inner_loop
    %j =l phi @outer_body %j_start, @inner_store %j_next
    %diag =l phi @outer_body %diag_start, @inner_store %diag_next
    %j_cond =w cslel %j, %n
    jnz %j_cond, @inner_body, @outer_next
@inner_body
    %off_j =l mul %j, 8
    %p_row_j =l add %row, %off_j
    %up =l loadl %p_row_j
    %j_m1 =l sub %j, 1
    %p_yj =l add %y, %j_m1
    %yj =w loadub %p_yj
    %match =w ceqw %xi, %yj
    jnz %match, @match_true, @match_false
@match_true
    %diag_plus_1 =l add %diag, 1
    jmp @inner_store
@match_false
    %off_jm1 =l mul %j_m1, 8
    %p_row_jm1 =l add %row, %off_jm1
    %r_jm1 =l loadl %p_row_jm1
    %up_gt =w csgtl %up, %r_jm1
    jnz %up_gt, @pick_up, @pick_prev
@pick_up
    %val_up =l copy %up
    jmp @inner_store
@pick_prev
    %val_prev =l copy %r_jm1
    jmp @inner_store
@inner_store
    %new_val =l phi @match_true %diag_plus_1, @pick_up %val_up, @pick_prev %val_prev
    storel %new_val, %p_row_j
    %diag_next =l copy %up
    %j_next =l add %j, 1
    jmp @inner_loop
@outer_next
    %i_next =l add %i, 1
    jmp @outer_loop
@done
    %off_res =l mul %n, 8
    %p_res =l add %row, %off_res
    %res =l loadl %p_res
    ret %res
}
