dot_i32_simd:
    xor eax, eax
    pxor xmm0, xmm0
    cmp rdx, 4
    jl .Lscalar_check
.Lvector_loop:
    pmovsxdq xmm1, qword ptr [rdi]
    pmovsxdq xmm2, qword ptr [rsi]
    pmovsxdq xmm3, qword ptr [rdi + 8]
    pmovsxdq xmm4, qword ptr [rsi + 8]
    pmuldq xmm1, xmm2
    pmuldq xmm3, xmm4
    paddq xmm0, xmm1
    paddq xmm0, xmm3
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    cmp rdx, 4
    jge .Lvector_loop
.Lvector_done:
    movhlps xmm1, xmm0
    paddq xmm0, xmm1
    movq rax, xmm0
.Lscalar_check:
    test rdx, rdx
    jle .Ldone
.Lscalar_loop:
    movsxd rcx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rcx, r8
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jnz .Lscalar_loop
.Ldone:
    ret
