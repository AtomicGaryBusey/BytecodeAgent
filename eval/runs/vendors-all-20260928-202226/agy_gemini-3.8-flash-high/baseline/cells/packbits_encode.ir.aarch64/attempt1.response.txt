export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @outer_loop
@outer_loop
    %i =l phi @start 0, @after_packet %i_next
    %written =l phi @start 0, @after_packet %written_next
    %more =w csltl %i, %n
    jnz %more, @check_run, @return_success
@return_success
    ret %written
@return_fail
    ret -1
@check_run
    %rem =l sub %n, %i
    %c_rem =w csgel %rem, 3
    jnz %c_rem, @check_run_bytes, @is_literal
@check_run_bytes
    %p0 =l add %src, %i
    %b0 =w loadub %p0
    %p1 =l add %p0, 1
    %b1 =w loadub %p1
    %c01 =w ceqw %b0, %b1
    jnz %c01, @check_run_byte2, @is_literal
@check_run_byte2
    %p2 =l add %p0, 2
    %b2 =w loadub %p2
    %c02 =w ceqw %b0, %b2
    jnz %c02, @count_run_loop, @is_literal
@count_run_loop
    %r =w phi @check_run_byte2 3, @count_run_inc %r_next
    %r_l =l extuw %r
    %c_r_lt_rem =w csltl %r_l, %rem
    %c_r_lt_128 =w csltw %r, 128
    %c_can_cont =w and %c_r_lt_rem, %c_r_lt_128
    jnz %c_can_cont, @count_run_check_byte, @emit_repeat
@count_run_check_byte
    %p_curr =l add %p0, %r_l
    %b_curr =w loadub %p_curr
    %c_match =w ceqw %b_curr, %b0
    jnz %c_match, @count_run_inc, @emit_repeat
@count_run_inc
    %r_next =w add %r, 1
    jmp @count_run_loop
@emit_repeat
    %room_rep =l sub %cap, %written
    %c_room_rep =w csltl %room_rep, 2
    jnz %c_room_rep, @return_fail, @write_repeat
@write_repeat
    %hdr_rep =w sub 257, %r
    %out_rep_hdr_p =l add %out, %written
    storeb %hdr_rep, %out_rep_hdr_p
    %out_rep_b_p =l add %out_rep_hdr_p, 1
    storeb %b0, %out_rep_b_p
    %written_rep =l add %written, 2
    %r_l_rep =l extuw %r
    %i_rep =l add %i, %r_l_rep
    jmp @after_packet
@is_literal
    jmp @lit_loop
@lit_loop
    %L =w phi @is_literal 1, @lit_inc %L_next
    %L_l =l extuw %L
    %j =l add %i, %L_l
    %c_j_lt_n =w csltl %j, %n
    jnz %c_j_lt_n, @lit_check_len, @emit_literal
@lit_check_len
    %c_len =w csltw %L, 128
    jnz %c_len, @lit_check_run, @emit_literal
@lit_check_run
    %rem_j =l sub %n, %j
    %c_rem_j =w csgel %rem_j, 3
    jnz %c_rem_j, @lit_check_run_b01, @lit_inc
@lit_check_run_b01
    %pj0 =l add %src, %j
    %bj0 =w loadub %pj0
    %pj1 =l add %pj0, 1
    %bj1 =w loadub %pj1
    %c_j01 =w ceqw %bj0, %bj1
    jnz %c_j01, @lit_check_run_b02, @lit_inc
@lit_check_run_b02
    %pj2 =l add %pj0, 2
    %bj2 =w loadub %pj2
    %c_j02 =w ceqw %bj0, %bj2
    jnz %c_j02, @emit_literal, @lit_inc
@lit_inc
    %L_next =w add %L, 1
    jmp @lit_loop
@emit_literal
    %req_w =w add 1, %L
    %req =l extuw %req_w
    %room_lit =l sub %cap, %written
    %c_room_lit =w csltl %room_lit, %req
    jnz %c_room_lit, @return_fail, @write_lit_header
@write_lit_header
    %lit_hdr =w sub %L, 1
    %out_lit_hdr_p =l add %out, %written
    storeb %lit_hdr, %out_lit_hdr_p
    %src_i_base =l add %src, %i
    %out_lit_base =l add %out_lit_hdr_p, 1
    jmp @copy_lit_loop
@copy_lit_loop
    %k =w phi @write_lit_header 0, @copy_lit_step %k_next
    %c_k =w csltw %k, %L
    jnz %c_k, @copy_lit_step, @lit_done
@copy_lit_step
    %k_l =l extuw %k
    %src_k_p =l add %src_i_base, %k_l
    %byte_k =w loadub %src_k_p
    %out_k_p =l add %out_lit_base, %k_l
    storeb %byte_k, %out_k_p
    %k_next =w add %k, 1
    jmp @copy_lit_loop
@lit_done
    %written_lit =l add %written, %req
    %L_l_done =l extuw %L
    %i_lit =l add %i, %L_l_done
    jmp @after_packet
@after_packet
    %i_next =l phi @write_repeat %i_rep, @lit_done %i_lit
    %written_next =l phi @write_repeat %written_rep, @lit_done %written_lit
    jmp @outer_loop
}
