export function l $fill_down(l %a, l %n, l %start) {
@entry
	jmp @loop_cond
@loop_cond
	%i =l phi @entry 0, @loop_body %i_next
	%c =w csltl %i, %n
	jnz %c, @loop_body, @end
@loop_body
	%val =l sub %start, %i
	%off =l mul %i, 8
	%ptr =l add %a, %off
	storel %val, %ptr
	%i_next =l add %i, 1
	jmp @loop_cond
@end
	ret %n
}
