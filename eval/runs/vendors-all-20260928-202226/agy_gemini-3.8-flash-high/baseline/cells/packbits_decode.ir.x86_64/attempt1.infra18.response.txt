export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop
@loop
    %s =l phi @start 0, @pkt_done %s_next
    %o =l phi @start 0, @pkt_done %o_next
    %done =w csgel %s, %n
    jnz %done, @ret_ok, @read_hdr
@ret_ok
    ret %o
@read_hdr
    %hdr_ptr =l add %src, %s
    %h =w loadsb %hdr_ptr
    %s_plus_1 =l add %s, 1
    %rem =l sub %n, %s_plus_1
    %is_nop =w ceqw %h, -128
    jnz %is_nop, @handle_nop, @check_lit
@handle_nop
    %s_nop =l add %s, 1
    %o_nop =l add %o, 0
    jmp @pkt_done
@check_lit
    %is_lit =w csgew %h, 0
    jnz %is_lit, @handle_lit, @handle_rep
@handle_lit
    %h_long =l extsw %h
    %lit_cnt =l add %h_long, 1
    %src_err_lit =w csltl %rem, %lit_cnt
    jnz %src_err_lit, @ret_err_src, @lit_src_ok
@ret_err_src
    ret -2
@lit_src_ok
    %new_o_lit =l add %o, %lit_cnt
    %exceeds_lit =w csgtl %new_o_lit, %cap
    jnz %exceeds_lit, @lit_cap_exceeded, @lit_cap_ok
@lit_cap_exceeded
    %avail_lit =l sub %cap, %o
    %lit_exc_src =l add %src, %s_plus_1
    %lit_exc_dst =l add %out, %o
    jmp @lit_exc_loop
@lit_exc_loop
    %i_lexc =l phi @lit_cap_exceeded 0, @lit_exc_body %i_lexc_next
    %cond_lexc =w csltl %i_lexc, %avail_lit
    jnz %cond_lexc, @lit_exc_body, @ret_err_cap
@lit_exc_body
    %p_lsrc =l add %lit_exc_src, %i_lexc
    %b_lexc =w loadub %p_lsrc
    %p_ldst =l add %lit_exc_dst, %i_lexc
    storeb %b_lexc, %p_ldst
    %i_lexc_next =l add %i_lexc, 1
    jmp @lit_exc_loop
@ret_err_cap
    ret -1
@lit_cap_ok
    %lit_ok_src =l add %src, %s_plus_1
    %lit_ok_dst =l add %out, %o
    jmp @lit_ok_loop
@lit_ok_loop
    %i_lok =l phi @lit_cap_ok 0, @lit_ok_body %i_lok_next
    %cond_lok =w csltl %i_lok, %lit_cnt
    jnz %cond_lok, @lit_ok_body, @lit_ok_end
@lit_ok_body
    %p_lsrc2 =l add %lit_ok_src, %i_lok
    %b_lok =w loadub %p_lsrc2
    %p_ldst2 =l add %lit_ok_dst, %i_lok
    storeb %b_lok, %p_ldst2
    %i_lok_next =l add %i_lok, 1
    jmp @lit_ok_loop
@lit_ok_end
    %s_lit_next =l add %s_plus_1, %lit_cnt
    %o_lit_next =l add %o, %lit_cnt
    jmp @pkt_done
@handle_rep
    %h_long_rep =l extsw %h
    %rep_cnt =l sub 1, %h_long_rep
    %src_err_rep =w csltl %rem, 1
    jnz %src_err_rep, @ret_err_src, @rep_src_ok
@rep_src_ok
    %rep_byte_ptr =l add %src, %s_plus_1
    %rep_val =w loadub %rep_byte_ptr
    %new_o_rep =l add %o, %rep_cnt
    %exceeds_rep =w csgtl %new_o_rep, %cap
    jnz %exceeds_rep, @rep_cap_exceeded, @rep_cap_ok
@rep_cap_exceeded
    %avail_rep =l sub %cap, %o
    %rep_exc_dst =l add %out, %o
    jmp @rep_exc_loop
@rep_exc_loop
    %i_rexc =l phi @rep_cap_exceeded 0, @rep_exc_body %i_rexc_next
    %cond_rexc =w csltl %i_rexc, %avail_rep
    jnz %cond_rexc, @rep_exc_body, @ret_err_cap
@rep_exc_body
    %p_rdst =l add %rep_exc_dst, %i_rexc
    storeb %rep_val, %p_rdst
    %i_rexc_next =l add %i_rexc, 1
    jmp @rep_exc_loop
@rep_cap_ok
    %rep_ok_dst =l add %out, %o
    jmp @rep_ok_loop
@rep_ok_loop
    %i_rok =l phi @rep_cap_ok 0, @rep_ok_body %i_rok_next
    %cond_rok =w csltl %i_rok, %rep_cnt
    jnz %cond_rok, @rep_ok_body, @rep_ok_end
@rep_ok_body
    %p_rdst2 =l add %rep_ok_dst, %i_rok
    storeb %rep_val, %p_rdst2
    %i_rok_next =l add %i_rok, 1
    jmp @rep_ok_loop
@rep_ok_end
    %s_rep_next =l add %s_plus_1, 1
    %o_rep_next =l add %o, %rep_cnt
    jmp @pkt_done
@pkt_done
    %s_next =l phi @handle_nop %s_nop, @lit_ok_end %s_lit_next, @rep_ok_end %s_rep_next
    %o_next =l phi @handle_nop %o_nop, @lit_ok_end %o_lit_next, @rep_ok_end %o_rep_next
    jmp @loop
}
