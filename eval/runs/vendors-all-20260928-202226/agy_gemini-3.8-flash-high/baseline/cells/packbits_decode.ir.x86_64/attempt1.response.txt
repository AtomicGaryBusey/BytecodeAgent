export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %zero =l copy 0
    %err1 =l subl %zero, 1
    %err2 =l subl %zero, 2
    %c257 =l copy 257
    jmp @loop_head

@loop_head
    %s_idx =l phi @start %zero, @loop_next %s_next
    %o_idx =l phi @start %zero, @loop_next %o_next
    %done =w csgel %s_idx, %n
    jnz %done, @ret_ok, @read_hdr

@read_hdr
    %src_hdr_ptr =l add %src, %s_idx
    %hdr_b =w loadub %src_hdr_ptr
    %hdr =l extuw %hdr_b
    %s_idx1 =l add %s_idx, 1
    %is_lit =w cslel %hdr, 127
    jnz %is_lit, @lit_packet, @check_nop

@check_nop
    %is_nop =w ceql %hdr, 128
    jnz %is_nop, @nop_packet, @rep_packet

@nop_packet
    jmp @loop_next

@lit_packet
    %lit_cnt =l add %hdr, 1
    %src_lit_p =l add %src, %s_idx1
    %out_lit_p =l add %out, %o_idx
    %s_lit_end =l add %s_idx1, %lit_cnt
    %s_rem =l subl %n, %s_idx1
    %s_ok =w csgel %s_rem, %lit_cnt
    jnz %s_ok, @lit_check_cap, @ret_err2

@lit_check_cap
    %o_lit_end =l add %o_idx, %lit_cnt
    %o_exceed =w csgtl %o_lit_end, %cap
    jnz %o_exceed, @lit_exceed, @lit_copy_loop

@lit_copy_loop
    %i =l phi @lit_check_cap %zero, @lit_copy_body %i_next
    %copy_done =w ceql %i, %lit_cnt
    jnz %copy_done, @lit_done, @lit_copy_body

@lit_copy_body
    %src_p =l add %src_lit_p, %i
    %b =w loadub %src_p
    %out_p =l add %out_lit_p, %i
    storeb %b, %out_p
    %i_next =l add %i, 1
    jmp @lit_copy_loop

@lit_done
    jmp @loop_next

@lit_exceed
    %rem_cap =l subl %cap, %o_idx
    jmp @lit_exc_loop

@lit_exc_loop
    %i_exc =l phi @lit_exceed %zero, @lit_exc_body %i_exc_next
    %done_exc =w ceql %i_exc, %rem_cap
    jnz %done_exc, @ret_err1, @lit_exc_body

@lit_exc_body
    %src_p_exc =l add %src_lit_p, %i_exc
    %b_exc =w loadub %src_p_exc
    %out_p_exc =l add %out_lit_p, %i_exc
    storeb %b_exc, %out_p_exc
    %i_exc_next =l add %i_exc, 1
    jmp @lit_exc_loop

@rep_packet
    %rep_cnt =l subl %c257, %hdr
    %s_rem_rep =l subl %n, %s_idx1
    %s_rep_ok =w csgtl %s_rem_rep, 0
    jnz %s_rep_ok, @rep_read_val, @ret_err2

@rep_read_val
    %src_val_p =l add %src, %s_idx1
    %rep_val =w loadub %src_val_p
    %s_rep_end =l add %s_idx1, 1
    %out_rep_p0 =l add %out, %o_idx
    %o_rep_end =l add %o_idx, %rep_cnt
    %o_rep_exceed =w csgtl %o_rep_end, %cap
    jnz %o_rep_exceed, @rep_exceed, @rep_loop

@rep_loop
    %j =l phi @rep_read_val %zero, @rep_body %j_next
    %rep_is_done =w ceql %j, %rep_cnt
    jnz %rep_is_done, @rep_done, @rep_body

@rep_body
    %out_rep_p =l add %out_rep_p0, %j
    storeb %rep_val, %out_rep_p
    %j_next =l add %j, 1
    jmp @rep_loop

@rep_done
    jmp @loop_next

@rep_exceed
    %rem_cap_rep =l subl %cap, %o_idx
    jmp @rep_exc_loop

@rep_exc_loop
    %j_exc =l phi @rep_exceed %zero, @rep_exc_body %j_exc_next
    %rep_exc_done =w ceql %j_exc, %rem_cap_rep
    jnz %rep_exc_done, @ret_err1, @rep_exc_body

@rep_exc_body
    %out_rep_exc_p =l add %out_rep_p0, %j_exc
    storeb %rep_val, %out_rep_exc_p
    %j_exc_next =l add %j_exc, 1
    jmp @rep_exc_loop

@loop_next
    %s_next =l phi @nop_packet %s_idx1, @lit_done %s_lit_end, @rep_done %s_rep_end
    %o_next =l phi @nop_packet %o_idx, @lit_done %o_lit_end, @rep_done %o_rep_end
    jmp @loop_head

@ret_ok
    ret %o_idx

@ret_err1
    ret %err1

@ret_err2
    ret %err2
}
