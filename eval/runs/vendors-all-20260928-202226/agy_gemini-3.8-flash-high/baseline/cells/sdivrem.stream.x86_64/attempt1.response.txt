sdivrem:
    mov rcx, rdx
    test rsi, rsi
    jz .Ldiv_zero
    cmp rsi, -1
    je .Ldiv_minus_one
    mov rax, rdi
    cqo
    idiv rsi
    mov qword ptr [rcx], rdx
    ret
.Ldiv_minus_one:
    mov rax, rdi
    neg rax
    mov qword ptr [rcx], 0
    ret
.Ldiv_zero:
    xor eax, eax
    mov qword ptr [rcx], rdi
    ret
