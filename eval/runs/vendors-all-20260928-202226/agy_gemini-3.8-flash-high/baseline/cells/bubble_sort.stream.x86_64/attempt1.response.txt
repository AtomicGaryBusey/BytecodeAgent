bubble_sort:
    cmp rsi, 1
    jle .Ldone
    dec rsi
.Louter_loop:
    xor ecx, ecx
.Linner_loop:
    mov rdx, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rcx*8 + 8]
    cmp rdx, r8
    jle .Lno_swap
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rcx*8 + 8], rdx
.Lno_swap:
    inc rcx
    cmp rcx, rsi
    jl .Linner_loop
    dec rsi
    jnz .Louter_loop
.Ldone:
    xor eax, eax
    ret
