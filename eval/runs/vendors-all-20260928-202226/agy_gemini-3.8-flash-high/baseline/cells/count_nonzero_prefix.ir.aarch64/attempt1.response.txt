export function l $count_nonzero_prefix(l %a, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @next %i1
    %c =w csltl %i, %n
    jnz %c, @body, @done
@body
    %off =l shl %i, 3
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %zero =w ceql %val, 0
    jnz %zero, @done, @next
@next
    %i1 =l add %i, 1
    jmp @loop
@done
    ret %i
}
