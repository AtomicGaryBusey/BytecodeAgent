export function l $collatz_max(l %n) {
@start
    %max0 =l copy 0
    %k0 =l copy 1
    jmp @outer_cond
@outer_cond
    %max =l phi @start %max0, @outer_next %max_next
    %k =l phi @start %k0, @outer_next %k_next
    %cond_k =w csgtl %k, %n
    jnz %cond_k, @done, @inner_init
@inner_init
    %steps0 =l copy 0
    jmp @inner_cond
@inner_cond
    %cur =l phi @inner_init %k, @step_done %cur_next
    %steps =l phi @inner_init %steps0, @step_done %steps_next
    %is_one =w ceql %cur, 1
    jnz %is_one, @inner_done, @inner_step
@inner_step
    %steps_next =l add %steps, 1
    %rem =l and %cur, 1
    %is_odd =w cnel %rem, 0
    jnz %is_odd, @odd, @even
@even
    %cur_even =l shr %cur, 1
    jmp @step_done
@odd
    %t1 =l mul %cur, 3
    %cur_odd =l add %t1, 1
    jmp @step_done
@step_done
    %cur_next =l phi @even %cur_even, @odd %cur_odd
    jmp @inner_cond
@inner_done
    %is_greater =w csgtl %steps, %max
    jnz %is_greater, @update_max, @keep_max
@update_max
    jmp @outer_next
@keep_max
    jmp @outer_next
@outer_next
    %max_next =l phi @update_max %steps, @keep_max %max
    %k_next =l add %k, 1
    jmp @outer_cond
@done
    ret %max
}
