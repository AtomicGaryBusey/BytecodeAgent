export function l $bmax(l %a, l %n) {
@start
    %max0 =l loadl %a
    %ptr0 =l add %a, 8
    %cnt0 =l sub %n, 1
    jmp @loop
@loop
    %max =l phi @start %max0, @next %max1
    %ptr =l phi @start %ptr0, @next %ptr1
    %cnt =l phi @start %cnt0, @next %cnt1
    %done =w ceql %cnt, 0
    jnz %done, @end, @body
@body
    %val =l loadl %ptr
    %gt =w csgtl %val, %max
    jnz %gt, @update, @keep
@update
    jmp @next
@keep
    jmp @next
@next
    %max1 =l phi @update %val, @keep %max
    %ptr1 =l add %ptr, 8
    %cnt1 =l sub %cnt, 1
    jmp @loop
@end
    ret %max
}
