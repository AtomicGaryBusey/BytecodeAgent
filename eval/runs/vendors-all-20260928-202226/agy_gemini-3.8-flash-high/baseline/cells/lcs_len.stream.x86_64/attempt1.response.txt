lcs_len:
    push rbx
    push r12
    xor eax, eax
.Lzero_loop:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rcx
    jle .Lzero_loop
    test rsi, rsi
    jle .Ldone
    test rcx, rcx
    jle .Ldone
.Louter_loop:
    movzx r9d, byte ptr [rdi]
    inc rdi
    xor r10d, r10d
    xor r11d, r11d
    mov ebx, 1
.Linner_loop:
    mov r12, qword ptr [r8 + rbx*8]
    inc r10
    cmp r12, r11
    mov rax, r12
    cmovl rax, r11
    cmp r9b, byte ptr [rdx + rbx - 1]
    cmove rax, r10
    mov r10, r12
    mov r11, rax
    mov qword ptr [r8 + rbx*8], rax
    inc rbx
    cmp rbx, rcx
    jle .Linner_loop
    dec rsi
    jnz .Louter_loop
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    pop r12
    pop rbx
    ret
