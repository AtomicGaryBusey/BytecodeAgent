max_byte:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    cmp rsi, 16
    jb .Lscalar
    pxor xmm0, xmm0
.Lsse_loop:
    movdqu xmm1, xmmword ptr [rdi]
    pmaxub xmm0, xmm1
    add rdi, 16
    sub rsi, 16
    cmp rsi, 16
    jae .Lsse_loop
    test rsi, rsi
    jz .Lreduce
    movdqu xmm1, xmmword ptr [rdi + rsi - 16]
    pmaxub xmm0, xmm1
.Lreduce:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
    ret
.Lscalar:
    movzx edx, byte ptr [rdi]
    cmp eax, edx
    cmovb eax, edx
    inc rdi
    dec rsi
    jnz .Lscalar
.Ldone:
    ret
