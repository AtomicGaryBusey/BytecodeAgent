export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 =l and %a, 4294967295
    %a1 =l shr %a, 32
    %b0 =l and %b, 4294967295
    %b1 =l shr %b, 32
    %p0 =l mul %a0, %b0
    %p1 =l mul %a0, %b1
    %p2 =l mul %a1, %b0
    %p3 =l mul %a1, %b1
    %p0_hi =l shr %p0, 32
    %t =l add %p1, %p0_hi
    %t_lo =l and %t, 4294967295
    %t_hi =l shr %t, 32
    %u =l add %p2, %t_lo
    %u_hi =l shr %u, 32
    %res0 =l add %p3, %t_hi
    %res =l add %res0, %u_hi
    ret %res
}
