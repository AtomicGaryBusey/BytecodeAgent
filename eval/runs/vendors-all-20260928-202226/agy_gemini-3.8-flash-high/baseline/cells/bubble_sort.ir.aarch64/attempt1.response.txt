export function l $bubble_sort(l %a, l %n) {
@start
	%i0 =l sub %n, 1
	jmp @outer_cond
@outer_cond
	%i =l phi @start %i0, @outer_inc %i_next
	%cmp_outer =w csgtl %i, 0
	jnz %cmp_outer, @outer_body, @done
@outer_body
	%j0 =l copy 0
	jmp @inner_cond
@inner_cond
	%j =l phi @outer_body %j0, @inner_inc %j_next
	%cmp_inner =w csltl %j, %i
	jnz %cmp_inner, @inner_body, @outer_inc
@inner_body
	%j_bytes =l mul %j, 8
	%pj =l add %a, %j_bytes
	%pj1 =l add %pj, 8
	%vj =l loadl %pj
	%vj1 =l loadl %pj1
	%need_swap =w csgtl %vj, %vj1
	jnz %need_swap, @do_swap, @inner_inc
@do_swap
	storel %vj1, %pj
	storel %vj, %pj1
	jmp @inner_inc
@inner_inc
	%j_next =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i_next =l sub %i, 1
	jmp @outer_cond
@done
	ret 0
}
