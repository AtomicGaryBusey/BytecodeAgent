export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @test
@test
	%i =l phi @start 0, @body %inext
	%j =l phi @start %j0, @body %jnext
	%cond =w csltl %i, %j
	jnz %cond, @body, @done
@body
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%iptr =l add %a, %ioff
	%jptr =l add %a, %joff
	%iv =l loadl %iptr
	%jv =l loadl %jptr
	storel %jv, %iptr
	storel %iv, %jptr
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @test
@done
	ret 0
}
