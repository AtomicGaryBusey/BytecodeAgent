export function l $base64_encode(l %src, l %n, l %out) {
@start
    %offp =l alloc8 8
    %writtenp =l alloc8 8
    %kp =l alloc8 8
    %vp =l alloc8 8
    %bp0 =l alloc8 8
    %bp1 =l alloc8 8
    %bp2 =l alloc8 8
    %remp =l alloc8 8
    storel 0, %offp
    storel 0, %writtenp
    jmp @outer

@outer
    %off =l loadl %offp
    %done =w cslel %n, %off
    jnz %done, @finish, @group

@group
    %left =l sub %n, %off
    storel %left, %remp
    %p0 =l add %src, %off
    %b0 =l loadub %p0
    storel %b0, %bp0
    %two =w csgel %left, 2
    jnz %two, @load1, @zero1

@load1
    %p1 =l add %p0, 1
    %b1 =l loadub %p1
    storel %b1, %bp1
    jmp @check2

@zero1
    storel 0, %bp1
    jmp @zero2

@check2
    %three =w csgel %left, 3
    jnz %three, @load2, @zero2

@load2
    %p2 =l add %p0, 2
    %b2 =l loadub %p2
    storel %b2, %bp2
    jmp @indices

@zero2
    storel 0, %bp2

@indices
    %x0 =l loadl %bp0
    %x1 =l loadl %bp1
    %x2 =l loadl %bp2
    %i0 =l shr %x0, 2
    %x0low =l and %x0, 3
    %a =l shl %x0low, 4
    %b =l shr %x1, 4
    %i1 =l or %a, %b
    %x1low =l and %x1, 15
    %c =l shl %x1low, 2
    %d =l shr %x2, 6
    %i2 =l or %c, %d
    %i3 =l and %x2, 63
    storel 0, %kp
    jmp @inner

@inner
    %k =l loadl %kp
    %kend =w cslel 4, %k
    jnz %kend, @advance, @choose

@choose
    %k0 =w ceql %k, 0
    jnz %k0, @choose0, @choose1

@choose0
    storel %i0, %vp
    jmp @map

@choose1
    %k1 =w ceql %k, 1
    jnz %k1, @choosev1, @choose2

@choosev1
    storel %i1, %vp
    jmp @map

@choose2
    %k2 =w ceql %k, 2
    jnz %k2, @choosev2, @choosev3

@choosev2
    storel %i2, %vp
    jmp @map

@choosev3
    storel %i3, %vp
    jmp @map

@map
    %left2 =l loadl %remp
    %is1 =w ceql %left2, 1
    %is2 =w ceql %left2, 2
    %kcur =l loadl %kp
    %pad2 =w cslel 2, %kcur
    %pad3 =w ceql %kcur, 3
    %padone =w and %is1, %pad2
    %padtwo =w and %is2, %pad3
    %padding =w or %padone, %padtwo
    jnz %padding, @putpad, @alphabet

@putpad
    %wp =l loadl %writtenp
    %dest =l add %out, %wp
    storeb 61, %dest
    jmp @putdone

@alphabet
    %v =l loadl %vp
    %lt26 =w csltl %v, 26
    jnz %lt26, @upper, @test52

@upper
    %chupper =l add %v, 65
    %wpupper =l loadl %writtenp
    %destupper =l add %out, %wpupper
    storeb %chupper, %destupper
    jmp @putdone

@test52
    %lt52 =w csltl %v, 52
    jnz %lt52, @lower, @test62

@lower
    %chlower =l add %v, 71
    %wplower =l loadl %writtenp
    %destlower =l add %out, %wplower
    storeb %chlower, %destlower
    jmp @putdone

@test62
    %lt62 =w csltl %v, 62
    jnz %lt62, @digit, @testplus

@digit
    %chdigit =l add %v, -4
    %wpdigit =l loadl %writtenp
    %destdigit =l add %out, %wpdigit
    storeb %chdigit, %destdigit
    jmp @putdone

@testplus
    %is62 =w ceql %v, 62
    jnz %is62, @plus, @slash

@plus
    %wpplus =l loadl %writtenp
    %destplus =l add %out, %wpplus
    storeb 43, %destplus
    jmp @putdone

@slash
    %wpslash =l loadl %writtenp
    %destslash =l add %out, %wpslash
    storeb 47, %destslash
    jmp @putdone

@putdone
    %wp3 =l loadl %writtenp
    %wp4 =l add %wp3, 1
    storel %wp4, %writtenp
    %knext =l add %kcur, 1
    storel %knext, %kp
    jmp @inner

@advance
    %oldoff =l loadl %offp
    %new_off =l add %oldoff, 3
    storel %new_off, %offp
    jmp @outer

@finish
    %result =l loadl %writtenp
    ret %result
}
