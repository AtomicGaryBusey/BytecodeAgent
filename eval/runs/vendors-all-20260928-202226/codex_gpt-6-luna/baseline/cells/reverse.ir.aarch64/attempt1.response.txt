export function l $reverse(l %a, l %n) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %inext
	%r =l phi @entry %n, @body %rnext
	%cond =w csltl %i, %r
	jnz %cond, @body, @done
@body
	%rnext =l sub %r, 1
	%ioff =l mul %i, 8
	%roff =l mul %rnext, 8
	%iptr =l add %a, %ioff
	%rptr =l add %a, %roff
	%ival =l loadl %iptr
	%rval =l loadl %rptr
	storel %rval, %iptr
	storel %ival, %rptr
	%inext =l add %i, 1
	jmp @loop
@done
	ret 0
}
