export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@init
	%limit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @init 0, @step %nexti
	%count =l phi @init 0, @step %nextcount
	%more =w csltl %i, %limit
	jnz %more, @check, @done
@check
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%ptrnext =l add %ptr, 8
	%current =l loadl %ptr
	%next =l loadl %ptrnext
	%descending =w csltl %next, %current
	%inc =l extuw %descending
	%nextcount =l add %count, %inc
	%nexti =l add %i, 1
	jmp @step
@step
	jmp @loop
@done
	ret %count
@zero
	ret 0
}
