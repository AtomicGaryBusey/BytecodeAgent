export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @init

@init
    %j =l phi @start 0, @initbody %jn
    %initcond =w cslel %j, %n
    jnz %initcond, @initbody, @outer

@initbody
    %initoff =l shl %j, 3
    %initptr =l add %row, %initoff
    storel 0, %initptr
    %jn =l add %j, 1
    jmp @init

@outer
    %i =l phi @init 0, @outernext %nexti
    %outercond =w csltl %i, %m
    jnz %outercond, @innerstart, @finish

@innerstart
    jmp @inner

@inner
    %jinner =l phi @innerstart 1, @store %nextj
    %diag =l phi @innerstart 0, @store %up
    %innercond =w cslel %jinner, %n
    jnz %innercond, @innerbody, @outernext

@innerbody
    %off =l shl %jinner, 3
    %ptr =l add %row, %off
    %up =l loadl %ptr
    %prevj =l sub %jinner, 1
    %prevoff =l shl %prevj, 3
    %prevptr =l add %row, %prevoff
    %left =l loadl %prevptr
    %xoff =l add %x, %i
    %yoff =l add %y, %prevj
    %xb =w loadub %xoff
    %yb =w loadub %yoff
    %match =w ceqw %xb, %yb
    jnz %match, @matched, @unmatched

@matched
    %inc =l add %diag, 1
    jmp @store

@unmatched
    %less =w csltl %up, %left
    jnz %less, @takeleft, @takeup

@takeleft
    %leftval =l copy %left
    jmp @storeunmatched

@takeup
    %upval =l copy %up
    jmp @storeunmatched

@storeunmatched
    %maxval =l phi @takeleft %leftval, @takeup %upval
    jmp @store

@store
    %value =l phi @matched %inc, @storeunmatched %maxval
    storel %value, %ptr
    %nextj =l add %jinner, 1
    jmp @inner

@outernext
    %nexti =l add %i, 1
    jmp @outer

@finish
    %finaloff =l shl %n, 3
    %finalptr =l add %row, %finaloff
    %result =l loadl %finalptr
    ret %result
}
