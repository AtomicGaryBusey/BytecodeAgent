export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@init(%j =l 0)
    %initcond =w cslel %j, %n
    jnz %initcond, @initbody, @outerstart
@initbody
    %initoff =l shl %j, 3
    %initptr =l add %row, %initoff
    storel 0, %initptr
    %initnext =l add %j, 1
    jmp @init(%initnext)

@outerstart
    jmp @outer(0)

@outer(%i =l)
    %outercond =w csltl %i, %m
    jnz %outercond, @innerstart, @finish

@innerstart
    jmp @inner(1, 0)

@inner(%j =l, %diag =l)
    %innercond =w cslel %j, %n
    jnz %innercond, @innerbody, @outernext

@innerbody
    %off =l shl %j, 3
    %ptr =l add %row, %off
    %up =l loadl %ptr
    %prevj =l sub %j, 1
    %prevoff =l shl %prevj, 3
    %prevptr =l add %row, %prevoff
    %left =l loadl %prevptr
    %xoff =l add %x, %i
    %yoff =l add %y, %prevj
    %xb =w loadub %xoff
    %yb =w loadub %yoff
    %match =w ceqw %xb, %yb
    %less =w csltl %up, %left
    %max =l select %less, %left, %up
    %inc =l add %diag, 1
    %value =l select %match, %inc, %max
    storel %value, %ptr
    %nextj =l add %j, 1
    jmp @inner(%nextj, %up)

@outernext
    %nexti =l add %i, 1
    jmp @outer(%nexti)

@finish
    %finaloff =l shl %n, 3
    %finalptr =l add %row, %finaloff
    %result =l loadl %finalptr
    ret %result
}
