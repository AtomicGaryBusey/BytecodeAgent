export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%base =l copy %tmp
	%prod =l add %tmp, mul %nn, 8
	jmp @copybase(0)
@copybase(%cb =l)
	%cbend =w csltl %cb, %nn
	jnz %cbend, @copybasebody, @init(0)
@copybasebody
	%cboff =l mul %cb, 8
	%cbap =l add %a, %cboff
	%cbval =l loadl %cbap
	%cbbp =l add %base, %cboff
	storel %cbval, %cbbp
	%cbnext =l add %cb, 1
	jmp @copybase(%cbnext)
@init(%iq =l)
	%iend =w csltl %iq, %nn
	jnz %iend, @initbody, @power(0)
@initbody
	%modulus =l add %n, 1
	%irem =l reml %iq, %modulus
	%isdiag =w ceql %irem, 0
	jnz %isdiag, @diag, @nondiag
@diag
	%ione =l copy 1
	jmp @initstore(%iq, %ione)
@nondiag
	jmp @initstore(%iq, 0)
@initstore(%sq =l, %sv =l)
	%ioff =l mul %sq, 8
	%iptr =l add %out, %ioff
	storel %sv, %iptr
	%inext =l add %sq, 1
	jmp @init(%inext)
@power(%e =l)
	%more =w csltl %e, %k
	jnz %more, @multiply(0, 0, 0), @cleartmp(0)
@multiply(%q =l, %t =l, %acc =l)
	%qend =w csltl %q, %nn
	jnz %qend, @multq, @copyprod(0)
@multq
	%tend =w csltl %t, %n
	jnz %tend, @multbody, @storeprod
@multbody
	%row =l div %q, %n
	%col =l rem %q, %n
	%ri =l mul %row, %n
	%ri_t =l add %ri, %t
	%li =l mul %ri_t, 8
	%leftp =l add %out, %li
	%left =l loadl %leftp
	%tn =l mul %t, %n
	%ridx =l add %tn, %col
	%rbytes =l mul %ridx, 8
	%rightp =l add %base, %rbytes
	%right =l loadl %rightp
	%term =l mul %left, %right
	%sum =l add %acc, %term
	%tnext =l add %t, 1
	jmp @multiply(%q, %tnext, %sum)
@storeprod
	%poff =l mul %q, 8
	%pptr =l add %prod, %poff
	storel %acc, %pptr
	%qnext =l add %q, 1
	jmp @multiply(%qnext, 0, 0)
@copyprod(%pq =l)
	%pend =w csltl %pq, %nn
	jnz %pend, @copyprod_body, @power_next
@copyprod_body
	%pbytes =l mul %pq, 8
	%psrc =l add %prod, %pbytes
	%pval =l loadl %psrc
	%pdst =l add %out, %pbytes
	storel %pval, %pdst
	%pqnext =l add %pq, 1
	jmp @copyprod(%pqnext)
@power_next
	%enext =l add %e, 1
	jmp @power(%enext)
@cleartmp(%z =l)
	%two =l mul %nn, 2
	%zend =w csltl %z, %two
	jnz %zend, @clearbody, @done
@clearbody
	%zbytes =l mul %z, 8
	%zptr =l add %tmp, %zbytes
	storel 0, %zptr
	%znext =l add %z, 1
	jmp @cleartmp(%znext)
@done
	ret 0
}
