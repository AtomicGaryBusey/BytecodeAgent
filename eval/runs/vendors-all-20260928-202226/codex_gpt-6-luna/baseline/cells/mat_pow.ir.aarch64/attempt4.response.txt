export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%qslot =l alloc8 8
	%tslot =l alloc8 8
	%accslot =l alloc8 8
	%eslot =l alloc8 8
	%zslot =l alloc8 8
	storel 0, %qslot
	jmp @init
@init
	%iq =l loadl %qslot
	%iend =w csltl %iq, %nn
	jnz %iend, @initbody, @power
@initbody
	%modulus =l add %n, 1
	%irem =l rem %iq, %modulus
	%isdiag =w ceql %irem, 0
	jnz %isdiag, @diag, @nondiag
@diag
	%ioffd =l mul %iq, 8
	%iptrd =l add %out, %ioffd
	storel 1, %iptrd
	jmp @initnext
@nondiag
	%ioffn =l mul %iq, 8
	%iptrn =l add %out, %ioffn
	storel 0, %iptrn
	jmp @initnext
@initnext
	%iqnext =l add %iq, 1
	storel %iqnext, %qslot
	jmp @init
@power
	storel 0, %eslot
	jmp @powercheck
@powercheck
	%e =l loadl %eslot
	%more =w csltl %e, %k
	jnz %more, @mulbegin, @cleartmp
@mulbegin
	storel 0, %qslot
	jmp @mulcheck
@mulcheck
	%q =l loadl %qslot
	%qend =w csltl %q, %nn
	jnz %qend, @mulinit, @copybegin
@mulinit
	storel 0, %tslot
	storel 0, %accslot
	jmp @tcheck
@tcheck
	%t =l loadl %tslot
	%tend =w csltl %t, %n
	jnz %tend, @mulbody, @storeprod
@mulbody
	%row =l div %q, %n
	%col =l rem %q, %n
	%ri =l mul %row, %n
	%ri_t =l add %ri, %t
	%li =l mul %ri_t, 8
	%leftp =l add %out, %li
	%left =l loadl %leftp
	%tn =l mul %t, %n
	%ridx =l add %tn, %col
	%rbytes =l mul %ridx, 8
	%rightp =l add %a, %rbytes
	%right =l loadl %rightp
	%term =l mul %left, %right
	%oldacc =l loadl %accslot
	%sum =l add %oldacc, %term
	storel %sum, %accslot
	%tnext =l add %t, 1
	storel %tnext, %tslot
	jmp @tcheck
@storeprod
	%poff =l mul %q, 8
	%pptr =l add %tmp, %poff
	%prod =l loadl %accslot
	storel %prod, %pptr
	%qnext =l add %q, 1
	storel %qnext, %qslot
	jmp @mulcheck
@copybegin
	storel 0, %qslot
	jmp @copycheck
@copycheck
	%cq =l loadl %qslot
	%cend =w csltl %cq, %nn
	jnz %cend, @copybody, @powerinc
@copybody
	%cbytes =l mul %cq, 8
	%csrc =l add %tmp, %cbytes
	%cval =l loadl %csrc
	%cdst =l add %out, %cbytes
	storel %cval, %cdst
	%cnext =l add %cq, 1
	storel %cnext, %qslot
	jmp @copycheck
@powerinc
	%eold =l loadl %eslot
	%enext =l add %eold, 1
	storel %enext, %eslot
	jmp @powercheck
@cleartmp
	storel 0, %zslot
	jmp @clearcheck
@clearcheck
	%z =l loadl %zslot
	%two =l mul %nn, 2
	%zend =w csltl %z, %two
	jnz %zend, @clearbody, @done
@clearbody
	%zbytes =l mul %z, 8
	%zptr =l add %tmp, %zbytes
	storel 0, %zptr
	%znext =l add %z, 1
	storel %znext, %zslot
	jmp @clearcheck
@done
	ret 0
}
