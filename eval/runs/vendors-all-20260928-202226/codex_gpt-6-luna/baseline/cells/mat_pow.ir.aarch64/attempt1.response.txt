export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn = mul %n, %n
	%base = %tmp
	%prod = add %tmp, mul %nn, 8
	jmp @copybase(0)
@copybase(%cb =l)
	%cbend = csltl %cb, %nn
	jnz %cbend, @copybasebody, @init(0)
@copybasebody
	%cboff = mul %cb, 8
	%cbap = add %a, %cboff
	%cbval = loadl %cbap
	%cbbp = add %base, %cboff
	storel %cbval, %cbbp
	%cbnext = add %cb, 1
	jmp @copybase(%cbnext)
@init(%iq =l)
	%iend = csltl %iq, %nn
	jnz %iend, @initbody, @power(0)
@initbody
	%modulus = add %n, 1
	%irem = reml %iq, %modulus
	%isdiag = ceql %irem, 0
	jnz %isdiag, @diag, @nondiag
@diag
	%ione = add 0, 1
	jmp @initstore(%iq, %ione)
@nondiag
	jmp @initstore(%iq, 0)
@initstore(%sq =l, %sv =l)
	%ioff = mul %sq, 8
	%iptr = add %out, %ioff
	storel %sv, %iptr
	%inext = add %sq, 1
	jmp @init(%inext)
@power(%e =l)
	%more = csltl %e, %k
	jnz %more, @multiply(0, 0, 0, 0), @cleartmp(0)
@multiply(%q =l, %t =l, %acc =l, %dummy =l)
	%qend = csltl %q, %nn
	jnz %qend, @multq, @copyprod(0)
@multq
	%tend = csltl %t, %n
	jnz %tend, @multbody, @storeprod
@multbody
	%row = div %q, %n
	%col = rem %q, %n
	%ri = mul %row, %n
	%ri_t = add %ri, %t
	%li = mul %ri_t, 8
	%leftp = add %out, %li
	%left = loadl %leftp
	%tn = mul %t, %n
	%ridx = add %tn, %col
	%rbytes = mul %ridx, 8
	%rightp = add %base, %rbytes
	%right = loadl %rightp
	%term = mul %left, %right
	%sum = add %acc, %term
	%tnext = add %t, 1
	jmp @multiply(%q, %tnext, %sum, 0)
@storeprod
	%poff = mul %q, 8
	%pptr = add %prod, %poff
	storel %acc, %pptr
	%qnext = add %q, 1
	jmp @multiply(%qnext, 0, 0, 0)
@copyprod(%pq =l)
	%pend = csltl %pq, %nn
	jnz %pend, @copyprod_body, @power_next
@copyprod_body
	%pbytes = mul %pq, 8
	%psrc = add %prod, %pbytes
	%pval = loadl %psrc
	%pdst = add %out, %pbytes
	storel %pval, %pdst
	%pqnext = add %pq, 1
	jmp @copyprod(%pqnext)
@power_next
	%enext = add %e, 1
	jmp @power(%enext)
@cleartmp(%z =l)
	%two = mul %nn, 2
	%zend = csltl %z, %two
	jnz %zend, @clearbody, @done
@clearbody
	%zbytes = mul %z, 8
	%zptr = add %tmp, %zbytes
	storel 0, %zptr
	%znext = add %z, 1
	jmp @cleartmp(%znext)
@done
	ret 0
}
