export function l $bubble_sort(l %a, l %n) {
@start
    %i =l alloc8 8
    %j =l alloc8 8
    %bound =l alloc8 8
    storel 0, %i
    jmp @outer

@outer
    %oi =l loadl %i
    %more =w csltl %oi, %n
    jnz %more, @outer_body, @done

@outer_body
    %remaining =l sub %n, %oi
    %limit =l sub %remaining, 1
    storel %limit, %bound
    storel 0, %j
    jmp @inner

@inner
    %ij =l loadl %j
    %lim =l loadl %bound
    %inside =w csltl %ij, %lim
    jnz %inside, @inner_body, @outer_step

@inner_body
    %off =l mul %ij, 8
    %ptr =l add %a, %off
    %nextptr =l add %ptr, 8
    %left =l loadl %ptr
    %right =l loadl %nextptr
    %swap =w csltl %right, %left
    jnz %swap, @do_swap, @inner_step

@do_swap
    storel %right, %ptr
    storel %left, %nextptr
    jmp @inner_step

@inner_step
    %curj =l loadl %j
    %newj =l add %curj, 1
    storel %newj, %j
    jmp @inner

@outer_step
    %curi =l loadl %i
    %newi =l add %curi, 1
    storel %newi, %i
    jmp @outer

@done
    ret 0
}
