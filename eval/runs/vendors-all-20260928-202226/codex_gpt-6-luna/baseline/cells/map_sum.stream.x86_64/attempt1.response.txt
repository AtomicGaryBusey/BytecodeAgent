map_sum:
push rbx
push r12
push r13
sub rsp, 16
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov qword ptr [rsp], 0
.Lloop:
test r12, r12
jz .Ldone
mov rdi, qword ptr [rbx]
call r13
add qword ptr [rsp], rax
add rbx, 8
dec r12
jmp .Lloop
.Ldone:
mov rax, qword ptr [rsp]
add rsp, 16
pop r13
pop r12
pop rbx
ret
