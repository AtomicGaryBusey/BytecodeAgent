packbits_encode:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
.Louter:
cmp rbx, r13
jae .Ldone
mov r10b, byte ptr [r12 + rbx]
mov r8d, 1
.Lrun:
cmp r8d, 128
jae .Lrun_done
lea r9, [rbx + r8]
cmp r9, r13
jae .Lrun_done
cmp byte ptr [r12 + r9], r10b
jne .Lrun_done
inc r8d
jmp .Lrun
.Lrun_done:
cmp r8d, 3
jb .Lliteral_start
lea rax, [rbp + 2]
cmp rax, r15
jg .Lfail
mov eax, 257
sub eax, r8d
mov byte ptr [r14 + rbp], al
mov byte ptr [r14 + rbp + 1], r10b
add rbp, 2
add rbx, r8
jmp .Louter
.Lliteral_start:
mov r8d, 0
.Lliteral_scan:
cmp r8d, 128
jae .Lliteral_done
lea r9, [rbx + r8]
cmp r9, r13
jae .Lliteral_done
mov r11b, byte ptr [r12 + r9]
mov r10d, 1
.Llookahead:
cmp r10d, 3
jae .Llook_done
lea rax, [r9 + r10]
cmp rax, r13
jae .Llook_done
cmp byte ptr [r12 + rax], r11b
jne .Llook_done
inc r10d
jmp .Llookahead
.Llook_done:
cmp r10d, 3
jae .Lliteral_done
inc r8d
jmp .Lliteral_scan
.Lliteral_done:
lea rax, [rbp + r8 + 1]
cmp rax, r15
jg .Lfail
lea eax, [r8 - 1]
mov byte ptr [r14 + rbp], al
lea r9, [rbx]
lea r10, [rbp + 1]
xor ecx, ecx
.Lcopy:
cmp ecx, r8d
jae .Lcopied
mov al, byte ptr [r12 + r9]
mov byte ptr [r14 + r10], al
inc r9
inc r10
inc ecx
jmp .Lcopy
.Lcopied:
inc r8
add rbx, r8
add rbp, r8
jmp .Louter
.Ldone:
mov rax, rbp
jmp .Lexit
.Lfail:
mov rax, -1
.Lexit:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
