packbits_decode:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
xor r15d, r15d
xor ebp, ebp
.Lpacket:
cmp r15, r12
jge .Ldone
movsx rax, byte ptr [rbx + r15]
inc r15
cmp rax, -128
je .Lpacket
test rax, rax
js .Lrepeat
lea r8, [rax + 1]
mov r9, r12
sub r9, r15
cmp r9, r8
jb .Lerror
mov r9, r14
sub r9, rbp
cmp r9, r8
jb .Lliteral_truncated
.Lliteral_copy:
test r8, r8
je .Lpacket
mov dl, byte ptr [rbx + r15]
mov byte ptr [r13 + rbp], dl
inc r15
inc rbp
dec r8
jmp .Lliteral_copy
.Lliteral_truncated:
test r9, r9
je .Loverflow
.Lliteral_partial:
mov dl, byte ptr [rbx + r15]
mov byte ptr [r13 + rbp], dl
inc r15
inc rbp
dec r9
jnz .Lliteral_partial
.Loverflow:
mov rax, -1
jmp .Lreturn
.Lrepeat:
neg rax
inc rax
mov r8, rax
mov r9, r12
sub r9, r15
cmp r9, 1
jb .Lerror
mov dl, byte ptr [rbx + r15]
inc r15
mov r9, r14
sub r9, rbp
cmp r9, r8
jb .Lrepeat_truncated
.Lrepeat_fill:
test r8, r8
je .Lpacket
mov byte ptr [r13 + rbp], dl
inc rbp
dec r8
jmp .Lrepeat_fill
.Lrepeat_truncated:
test r9, r9
je .Loverflow
.Lrepeat_partial:
mov byte ptr [r13 + rbp], dl
inc rbp
dec r9
jnz .Lrepeat_partial
jmp .Loverflow
.Lerror:
mov rax, -2
jmp .Lreturn
.Ldone:
mov rax, rbp
.Lreturn:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
