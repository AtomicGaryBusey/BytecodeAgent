export function l $max_byte(l %p, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @next %ni
    %m =l phi @start 0, @next %nm
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %greater =w csltl %m, %b
    jnz %greater, @replace, @keep
@replace
    jmp @next
@keep
    jmp @next
@next
    %nm =l phi @replace %b, @keep %m
    %ni =l add %i, 1
    jmp @loop
@done
    ret %m
}
