export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	storel 0, %ip
	jmp @zero

@zero
	%i =l loadl %ip
	%done =w csgeul %i, %cap
	jnz %done, @run, @clear
@clear
	%off =l mul %i, 8
	%addr =l add %stack, %off
	storel 0, %addr
	%nexti =l add %i, 1
	storel %nexti, %ip
	jmp @zero

@run
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch

@fetch
	%coff =l mul %pc, 8
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%h =w ceql %op, 0
	jnz %h, @finish, @checkpush

@checkpush
	%ispush =w ceql %op, 1
	jnz %ispush, @push, @checkadd

@checkadd
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @checksub

@checksub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @checkmul

@checkmul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @checkdup

@checkdup
	%isdup =w ceql %op, 5
	jnz %isdup, @dup, @checkswap

@checkswap
	%isswap =w ceql %op, 6
	jnz %isswap, @swap, @checkdrop

@checkdrop
	%isdrop =w ceql %op, 7
	jnz %isdrop, @drop, @checkover

@checkover
	%isover =w ceql %op, 8
	jnz %isover, @over, @checklt

@checklt
	%islt =w ceql %op, 9
	jnz %islt, @lt, @checkjz

@checkjz
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @checkjmp

@checkjmp
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jmp, @other

@push
	%operandpc =l add %pc, 1
	%opoff =l mul %operandpc, 8
	%opaddr =l add %code, %opoff
	%value =l loadl %opaddr
	%d0 =l loadl %dp
	%doff =l mul %d0, 8
	%daddr =l add %stack, %doff
	storel %value, %daddr
	%d1 =l add %d0, 1
	storel %d1, %dp
	%pn =l add %pc, 2
	storel %pn, %pcp
	jmp @run

@add
	%d2 =l loadl %dp
	%ib =l sub %d2, 1
	%ia =l sub %d2, 2
	%boff =l mul %ib, 8
	%aoff =l mul %ia, 8
	%baddr =l add %stack, %boff
	%aaddr =l add %stack, %aoff
	%b =l loadl %baddr
	%a =l loadl %aaddr
	%r =l add %a, %b
	storel %r, %aaddr
	%dn =l sub %d2, 1
	storel %dn, %dp
	%pn =l add %pc, 1
	storel %pn, %pcp
	jmp @run

@sub
	%d3 =l loadl %dp
	%ib3 =l sub %d3, 1
	%ia3 =l sub %d3, 2
	%boff3 =l mul %ib3, 8
	%aoff3 =l mul %ia3, 8
	%baddr3 =l add %stack, %boff3
	%aaddr3 =l add %stack, %aoff3
	%b3 =l loadl %baddr3
	%a3 =l loadl %aaddr3
	%r3 =l sub %a3, %b3
	storel %r3, %aaddr3
	%dn3 =l sub %d3, 1
	storel %dn3, %dp
	%pn3 =l add %pc, 1
	storel %pn3, %pcp
	jmp @run

@mul
	%d4 =l loadl %dp
	%ib4 =l sub %d4, 1
	%ia4 =l sub %d4, 2
	%boff4 =l mul %ib4, 8
	%aoff4 =l mul %ia4, 8
	%baddr4 =l add %stack, %boff4
	%aaddr4 =l add %stack, %aoff4
	%b4 =l loadl %baddr4
	%a4 =l loadl %aaddr4
	%r4 =l mul %a4, %b4
	storel %r4, %aaddr4
	%dn4 =l sub %d4, 1
	storel %dn4, %dp
	%pn4 =l add %pc, 1
	storel %pn4, %pcp
	jmp @run

@dup
	%d5 =l loadl %dp
	%src5 =l sub %d5, 1
	%src_off5 =l mul %src5, 8
	%src5a =l add %stack, %src_off5
	%v5 =l loadl %src5a
	%d5off =l mul %d5, 8
	%d5addr =l add %stack, %d5off
	storel %v5, %d5addr
	%d5n =l add %d5, 1
	storel %d5n, %dp
	%p5 =l add %pc, 1
	storel %p5, %pcp
	jmp @run

@swap
	%d6 =l loadl %dp
	%ix6 =l sub %d6, 1
	%iy6 =l sub %d6, 2
	%ox6 =l mul %ix6, 8
	%oy6 =l mul %iy6, 8
	%x6a =l add %stack, %ox6
	%y6a =l add %stack, %oy6
	%x6 =l loadl %x6a
	%y6 =l loadl %y6a
	storel %x6, %y6a
	storel %y6, %x6a
	%p6 =l add %pc, 1
	storel %p6, %pcp
	jmp @run

@drop
	%d7 =l loadl %dp
	%d7n =l sub %d7, 1
	storel %d7n, %dp
	%p7 =l add %pc, 1
	storel %p7, %pcp
	jmp @run

@over
	%d8 =l loadl %dp
	%ix8 =l sub %d8, 2
	%ox8 =l mul %ix8, 8
	%x8a =l add %stack, %ox8
	%x8 =l loadl %x8a
	%od8 =l mul %d8, 8
	%da8 =l add %stack, %od8
	storel %x8, %da8
	%d8n =l add %d8, 1
	storel %d8n, %dp
	%p8 =l add %pc, 1
	storel %p8, %pcp
	jmp @run

@lt
	%d9 =l loadl %dp
	%ib9 =l sub %d9, 1
	%ia9 =l sub %d9, 2
	%boff9 =l mul %ib9, 8
	%aoff9 =l mul %ia9, 8
	%baddr9 =l add %stack, %boff9
	%aaddr9 =l add %stack, %aoff9
	%b9 =l loadl %baddr9
	%a9 =l loadl %aaddr9
	%cmp9 =w csltl %a9, %b9
	%res9 =l extuw %cmp9
	storel %res9, %aaddr9
	%dn9 =l sub %d9, 1
	storel %dn9, %dp
	%pn9 =l add %pc, 1
	storel %pn9, %pcp
	jmp @run

@jz
	%jpc =l add %pc, 1
	%joff =l mul %jpc, 8
	%jaddr =l add %code, %joff
	%target =l loadl %jaddr
	%dj =l loadl %dp
	%topj =l sub %dj, 1
	%topoff =l mul %topj, 8
	%topaddr =l add %stack, %topoff
	%v =l loadl %topaddr
	%djn =l sub %dj, 1
	storel %djn, %dp
	%zero =w ceql %v, 0
	jnz %zero, @jz_taken, @jz_not
@jz_taken
	storel %target, %pcp
	jmp @run
@jz_not
	%pnj =l add %pc, 2
	storel %pnj, %pcp
	jmp @run

@jmp
	%jpc11 =l add %pc, 1
	%joff11 =l mul %jpc11, 8
	%jaddr11 =l add %code, %joff11
	%target11 =l loadl %jaddr11
	storel %target11, %pcp
	jmp @run

@other
	%po =l add %pc, 1
	storel %po, %pcp
	jmp @run

@finish
	%df =l loadl %dp
	%empty =w ceql %df, 0
	jnz %empty, @emptyret, @valueret
@emptyret
	ret 0
@valueret
	%last =l sub %df, 1
	%lastoff =l mul %last, 8
	%lastaddr =l add %stack, %lastoff
	%answer =l loadl %lastaddr
	ret %answer
}
