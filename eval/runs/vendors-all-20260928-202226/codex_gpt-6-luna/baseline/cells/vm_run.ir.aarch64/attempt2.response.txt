export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	storel 0, %ip
	jmp @clearcheck

@clearcheck
	%i =l loadl %ip
	%more =w csltl %i, %cap
	jnz %more, @clear, @run
@clear
	%off =l mul %i, 8
	%addr =l add %stack, %off
	storel 0, %addr
	%nexti =l add %i, 1
	storel %nexti, %ip
	jmp @clearcheck

@run
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch

@fetch
	%coff =l mul %pc, 8
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%halt =w ceql %op, 0
	jnz %halt, @finish, @dispatch

@dispatch
	%push =w ceql %op, 1
	jnz %push, @pushop, @dispatch2
@dispatch2
	%addop =w ceql %op, 2
	jnz %addop, @addop, @dispatch3
@dispatch3
	%subop =w ceql %op, 3
	jnz %subop, @subop, @dispatch4
@dispatch4
	%mulop =w ceql %op, 4
	jnz %mulop, @mulop, @dispatch5
@dispatch5
	%dup =w ceql %op, 5
	jnz %dup, @dup, @dispatch6
@dispatch6
	%swap =w ceql %op, 6
	jnz %swap, @swap, @dispatch7
@dispatch7
	%drop =w ceql %op, 7
	jnz %drop, @drop, @dispatch8
@dispatch8
	%over =w ceql %op, 8
	jnz %over, @over, @dispatch9
@dispatch9
	%lt =w ceql %op, 9
	jnz %lt, @lt, @dispatch10
@dispatch10
	%jz =w ceql %op, 10
	jnz %jz, @jz, @dispatch11
@dispatch11
	%jmpop =w ceql %op, 11
	jnz %jmpop, @jmpop, @other

@pushop
	%pp =l loadl %pcp
	%argp =l add %pp, 1
	%arga =l mul %argp, 8
	%argaddr =l add %code, %arga
	%value =l loadl %argaddr
	%depth =l loadl %dp
	%doff =l mul %depth, 8
	%daddr =l add %stack, %doff
	storel %value, %daddr
	%depth1 =l add %depth, 1
	storel %depth1, %dp
	%pc1 =l add %pp, 2
	storel %pc1, %pcp
	jmp @run

@addop
	%pc2 =l loadl %pcp
	%d2 =l loadl %dp
	%i2 =l sub %d2, 2
	%off2 =l mul %i2, 8
	%addr2 =l add %stack, %off2
	%a2 =l loadl %addr2
	%j2 =l add %i2, 1
	%offb2 =l mul %j2, 8
	%addrb2 =l add %stack, %offb2
	%b2 =l loadl %addrb2
	%r2 =l add %a2, %b2
	storel %r2, %addr2
	storel 0, %addrb2
	%dn2 =l sub %d2, 1
	storel %dn2, %dp
	%pn2 =l add %pc2, 1
	storel %pn2, %pcp
	jmp @run

@subop
	%pc3 =l loadl %pcp
	%d3 =l loadl %dp
	%i3 =l sub %d3, 2
	%off3 =l mul %i3, 8
	%addr3 =l add %stack, %off3
	%a3 =l loadl %addr3
	%j3 =l add %i3, 1
	%offb3 =l mul %j3, 8
	%addrb3 =l add %stack, %offb3
	%b3 =l loadl %addrb3
	%r3 =l sub %a3, %b3
	storel %r3, %addr3
	storel 0, %addrb3
	%dn3 =l sub %d3, 1
	storel %dn3, %dp
	%pn3 =l add %pc3, 1
	storel %pn3, %pcp
	jmp @run

@mulop
	%pc4 =l loadl %pcp
	%d4 =l loadl %dp
	%i4 =l sub %d4, 2
	%off4 =l mul %i4, 8
	%addr4 =l add %stack, %off4
	%a4 =l loadl %addr4
	%j4 =l add %i4, 1
	%offb4 =l mul %j4, 8
	%addrb4 =l add %stack, %offb4
	%b4 =l loadl %addrb4
	%r4 =l mul %a4, %b4
	storel %r4, %addr4
	storel 0, %addrb4
	%dn4 =l sub %d4, 1
	storel %dn4, %dp
	%pn4 =l add %pc4, 1
	storel %pn4, %pcp
	jmp @run

@dup
	%pc5 =l loadl %pcp
	%d5 =l loadl %dp
	%i5 =l sub %d5, 1
	%off5 =l mul %i5, 8
	%addr5 =l add %stack, %off5
	%v5 =l loadl %addr5
	%offd5 =l mul %d5, 8
	%addrd5 =l add %stack, %offd5
	storel %v5, %addrd5
	%dn5 =l add %d5, 1
	storel %dn5, %dp
	%pn5 =l add %pc5, 1
	storel %pn5, %pcp
	jmp @run

@swap
	%pc6 =l loadl %pcp
	%d6 =l loadl %dp
	%i6 =l sub %d6, 1
	%j6 =l sub %d6, 2
	%offi6 =l mul %i6, 8
	%offj6 =l mul %j6, 8
	%ai6 =l add %stack, %offi6
	%aj6 =l add %stack, %offj6
	%vi6 =l loadl %ai6
	%vj6 =l loadl %aj6
	storel %vi6, %aj6
	storel %vj6, %ai6
	%pn6 =l add %pc6, 1
	storel %pn6, %pcp
	jmp @run

@drop
	%pc7 =l loadl %pcp
	%d7 =l loadl %dp
	%i7 =l sub %d7, 1
	%off7 =l mul %i7, 8
	%addr7 =l add %stack, %off7
	storel 0, %addr7
	storel %i7, %dp
	%pn7 =l add %pc7, 1
	storel %pn7, %pcp
	jmp @run

@over
	%pc8 =l loadl %pcp
	%d8 =l loadl %dp
	%i8 =l sub %d8, 2
	%off8 =l mul %i8, 8
	%addr8 =l add %stack, %off8
	%v8 =l loadl %addr8
	%offd8 =l mul %d8, 8
	%addrd8 =l add %stack, %offd8
	storel %v8, %addrd8
	%dn8 =l add %d8, 1
	storel %dn8, %dp
	%pn8 =l add %pc8, 1
	storel %pn8, %pcp
	jmp @run

@lt
	%pc9 =l loadl %pcp
	%d9 =l loadl %dp
	%i9 =l sub %d9, 2
	%off9 =l mul %i9, 8
	%addr9 =l add %stack, %off9
	%a9 =l loadl %addr9
	%j9 =l add %i9, 1
	%offb9 =l mul %j9, 8
	%addrb9 =l add %stack, %offb9
	%b9 =l loadl %addrb9
	%c9 =w csltl %a9, %b9
	%r9 =l extuw %c9
	storel %r9, %addr9
	storel 0, %addrb9
	%dn9 =l sub %d9, 1
	storel %dn9, %dp
	%pn9 =l add %pc9, 1
	storel %pn9, %pcp
	jmp @run

@jz
	%pc10 =l loadl %pcp
	%arg10 =l add %pc10, 1
	%offarg10 =l mul %arg10, 8
	%addrarg10 =l add %code, %offarg10
	%target10 =l loadl %addrarg10
	%d10 =l loadl %dp
	%i10 =l sub %d10, 1
	%off10 =l mul %i10, 8
	%addr10 =l add %stack, %off10
	%v10 =l loadl %addr10
	storel 0, %addr10
	storel %i10, %dp
	%zero10 =w ceql %v10, 0
	jnz %zero10, @jztaken, @jznot
@jztaken
	storel %target10, %pcp
	jmp @run
@jznot
	%pn10 =l add %pc10, 2
	storel %pn10, %pcp
	jmp @run

@jmpop
	%pc11 =l loadl %pcp
	%arg11 =l add %pc11, 1
	%off11 =l mul %arg11, 8
	%addr11 =l add %code, %off11
	%target11 =l loadl %addr11
	storel %target11, %pcp
	jmp @run

@other
	%pco =l loadl %pcp
	%pno =l add %pco, 1
	storel %pno, %pcp
	jmp @run

@finish
	%df =l loadl %dp
	storel %df, %ip
	jmp @tailcheck
@tailcheck
	%it =l loadl %ip
	%moret =w csltl %it, %cap
	jnz %moret, @tailclear, @return
@tailclear
	%offt =l mul %it, 8
	%addrt =l add %stack, %offt
	storel 0, %addrt
	%itn =l add %it, 1
	storel %itn, %ip
	jmp @tailcheck
@return
	%finald =l loadl %dp
	%empty =w ceql %finald, 0
	jnz %empty, @emptyret, @valueret
@emptyret
	ret 0
@valueret
	%last =l sub %finald, 1
	%lastoff =l mul %last, 8
	%lastaddr =l add %stack, %lastoff
	%answer =l loadl %lastaddr
	ret %answer
}
