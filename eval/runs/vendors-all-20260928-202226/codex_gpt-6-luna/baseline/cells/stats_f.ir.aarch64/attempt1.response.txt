export function l $stats_f(l %a, l %n, l %f, l %out) {
@init
	%v0 =l loadl %a
	%r0 =l call %f(l %v0)
	%b0 =l and %r0, 1
	%o0 =w cnel %b0, 0
	%ol0 =l extuw %o0
	jmp @test
@test
	%i =l phi @init 1, @body %next
	%min =l phi @init %r0, @body %newmin
	%max =l phi @init %r0, @body %newmax
	%sum =l phi @init %r0, @body %newsum
	%odd =l phi @init %ol0, @body %newodd
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%newmin =l sell %lt, %y, %min
	%gt =w csgtl %y, %max
	%newmax =l sell %gt, %y, %max
	%newsum =l add %sum, %y
	%bit =l and %y, 1
	%isodd =w cnel %bit, 0
	%odd64 =l extuw %isodd
	%newodd =l add %odd, %odd64
	%next =l add %i, 1
	jmp @test
@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %odd
}
