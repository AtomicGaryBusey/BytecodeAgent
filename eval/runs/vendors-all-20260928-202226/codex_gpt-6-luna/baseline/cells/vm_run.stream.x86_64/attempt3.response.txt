vm_run:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebp, ebp
    xor ebx, ebx
    xor edx, edx
.Lclear_init:
    cmp rdx, r15
    jae .Lrun
    mov qword ptr [r14 + rdx*8], 0
    inc rdx
    jmp .Lclear_init
.Lrun:
    cmp rbp, r13
    jae .Lfinish
    mov rax, qword ptr [r12 + rbp*8]
    cmp rax, 0
    je .Lfinish
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc rbp
    jmp .Lrun
.Lpush:
    mov rax, qword ptr [r12 + rbp*8 + 8]
    mov qword ptr [r14 + rbx*8], rax
    add rbx, 1
    add rbp, 2
    jmp .Lrun
.Ladd:
    mov rax, qword ptr [r14 + rbx*8 - 8]
    add qword ptr [r14 + rbx*8 - 16], rax
    dec rbx
    inc rbp
    jmp .Lrun
.Lsub:
    mov rax, qword ptr [r14 + rbx*8 - 16]
    sub rax, qword ptr [r14 + rbx*8 - 8]
    mov qword ptr [r14 + rbx*8 - 16], rax
    dec rbx
    inc rbp
    jmp .Lrun
.Lmul:
    mov rax, qword ptr [r14 + rbx*8 - 8]
    imul qword ptr [r14 + rbx*8 - 16]
    mov qword ptr [r14 + rbx*8 - 16], rax
    dec rbx
    inc rbp
    jmp .Lrun
.Ldup:
    mov rax, qword ptr [r14 + rbx*8 - 8]
    mov qword ptr [r14 + rbx*8], rax
    inc rbx
    inc rbp
    jmp .Lrun
.Lswap:
    mov rax, qword ptr [r14 + rbx*8 - 8]
    mov rcx, qword ptr [r14 + rbx*8 - 16]
    mov qword ptr [r14 + rbx*8 - 8], rcx
    mov qword ptr [r14 + rbx*8 - 16], rax
    inc rbp
    jmp .Lrun
.Ldrop:
    dec rbx
    inc rbp
    jmp .Lrun
.Lover:
    mov rax, qword ptr [r14 + rbx*8 - 16]
    mov qword ptr [r14 + rbx*8], rax
    inc rbx
    inc rbp
    jmp .Lrun
.Llt:
    mov rax, qword ptr [r14 + rbx*8 - 16]
    cmp rax, qword ptr [r14 + rbx*8 - 8]
    setl al
    movzx eax, al
    mov qword ptr [r14 + rbx*8 - 16], rax
    dec rbx
    inc rbp
    jmp .Lrun
.Ljz:
    mov rax, qword ptr [r12 + rbp*8 + 8]
    mov rcx, qword ptr [r14 + rbx*8 - 8]
    dec rbx
    test rcx, rcx
    jne .Ljz_no
    mov rbp, rax
    jmp .Lrun
.Ljz_no:
    add rbp, 2
    jmp .Lrun
.Ljmp:
    mov rbp, qword ptr [r12 + rbp*8 + 8]
    jmp .Lrun
.Lfinish:
    xor r10d, r10d
    test rbx, rbx
    jz .Lzero_tail
    mov r10, qword ptr [r14 + rbx*8 - 8]
.Lzero_tail:
    mov rdx, rbx
.Lzero_loop:
    cmp rdx, r15
    jae .Lreturn
    mov qword ptr [r14 + rdx*8], 0
    inc rdx
    jmp .Lzero_loop
.Lreturn:
    mov rax, r10
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
