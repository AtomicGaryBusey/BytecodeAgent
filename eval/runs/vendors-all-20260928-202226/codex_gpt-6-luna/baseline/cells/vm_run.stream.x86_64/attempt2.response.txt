vm_run:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebp, ebp
    xor ebx, ebx
    xor rcx, ecx
.Lzero:
    cmp rcx, r15
    jae .Lloop
    mov qword ptr [r14 + rcx*8], 0
    inc rcx
    jmp .Lzero
.Lloop:
    cmp rbx, r13
    jae .Lfinish
    mov rax, qword ptr [r12 + rbx*8]
    cmp rax, 0
    je .Lfinish
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc rbx
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [r12 + rbx*8 + 8]
    mov qword ptr [r14 + rbp*8], rax
    inc rbp
    add rbx, 2
    jmp .Lloop
.Ladd:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    add rax, qword ptr [r14 + rbp*8 - 8]
    mov qword ptr [r14 + rbp*8 - 16], rax
    mov qword ptr [r14 + rbp*8 - 8], 0
    dec rbp
    inc rbx
    jmp .Lloop
.Lsub:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    sub rax, qword ptr [r14 + rbp*8 - 8]
    mov qword ptr [r14 + rbp*8 - 16], rax
    mov qword ptr [r14 + rbp*8 - 8], 0
    dec rbp
    inc rbx
    jmp .Lloop
.Lmul:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    imul rax, qword ptr [r14 + rbp*8 - 8]
    mov qword ptr [r14 + rbp*8 - 16], rax
    mov qword ptr [r14 + rbp*8 - 8], 0
    dec rbp
    inc rbx
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    mov qword ptr [r14 + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    mov rcx, qword ptr [r14 + rbp*8 - 16]
    mov qword ptr [r14 + rbp*8 - 8], rcx
    mov qword ptr [r14 + rbp*8 - 16], rax
    inc rbx
    jmp .Lloop
.Ldrop:
    dec rbp
    mov qword ptr [r14 + rbp*8], 0
    inc rbx
    jmp .Lloop
.Lover:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    mov qword ptr [r14 + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Llt:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    cmp rax, qword ptr [r14 + rbp*8 - 8]
    setl al
    movzx rax, al
    mov qword ptr [r14 + rbp*8 - 16], rax
    mov qword ptr [r14 + rbp*8 - 8], 0
    dec rbp
    inc rbx
    jmp .Lloop
.Ljz:
    mov rdx, qword ptr [r12 + rbx*8 + 8]
    dec rbp
    mov rax, qword ptr [r14 + rbp*8]
    mov qword ptr [r14 + rbp*8], 0
    test rax, rax
    jne .Ljz_not_taken
    mov rbx, rdx
    jmp .Lloop
.Ljz_not_taken:
    add rbx, 2
    jmp .Lloop
.Ljmp:
    mov rbx, qword ptr [r12 + rbx*8 + 8]
    jmp .Lloop
.Lfinish:
    xor eax, eax
    test rbp, rbp
    jz .Lreturn
    mov rax, qword ptr [r14 + rbp*8 - 8]
.Lreturn:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
