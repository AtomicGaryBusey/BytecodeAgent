export function l $bubble_sort(l %a, l %n) {
@entry
    %small =cslel %n, 1
    jnz %small, @done, @outer

@outer
    %i =l phi @entry 0, @outer_next %inext
    %bound =l sub %n, 1
    %bound =l sub %bound, %i
    jmp @inner

@inner
    %j =l phi @outer 0, @inner_next %jnext
    %more =csltl %j, %bound
    jnz %more, @compare, @outer_next

@compare
    %next =l add %j, 1
    %off0 =l mul %j, 8
    %off1 =l mul %next, 8
    %ptr0 =l add %a, %off0
    %ptr1 =l add %a, %off1
    %v0 =l loadl %ptr0
    %v1 =l loadl %ptr1
    %swap =csgtl %v0, %v1
    jnz %swap, @do_swap, @inner_next

@do_swap
    storel %v1, %ptr0
    storel %v0, %ptr1
    jmp @inner_next

@inner_next
    %jnext =l add %j, 1
    jmp @inner

@outer_next
    %inext =l add %i, 1
    %again =csltl %inext, %bound
    jnz %again, @outer, @done

@done
    ret 0
}
