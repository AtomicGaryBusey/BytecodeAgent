export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%zeroa =w ceql %na, 0
	%zerob =w ceql %nb, 0
	%empty =w or %zeroa, %zerob
	jnz %empty, @empty, @init

@empty
	ret 0

@init
	%total =l add %na, %nb
	jmp @initloop

@initloop
	%i =l phi @init 0, @initnext %inext
	%done =w cslel %total, %i
	jnz %done, @outerstart, @initbody

@initbody
	%addr =l add %r, %i
	storel 0, %addr
	%inext =l add %i, 1
	jmp @initnext

@initnext
	jmp @initloop

@outerstart
	jmp @outer

@outer
	%oi =l phi @outerstart 0, @outerdone %oinext
	%odone =w cslel %na, %oi
	jnz %odone, @findstart, @innerstart

@innerstart
	jmp @inner

@inner
	%ji =l phi @innerstart 0, @innerdone %jnext
	%jdone =w cslel %nb, %ji
	jnz %jdone, @outerdone, @calc

@calc
	%ap =l add %a, %oi
	%bp =l add %b, %ji
	%av =l loadl %ap
	%bv =l loadl %bp
	%mask =l and %av, 4294967295
	%ah =l shr %av, 32
	%bmask =l and %bv, 4294967295
	%bh =l shr %bv, 32
	%p0 =l mul %mask, %bmask
	%p1 =l mul %mask, %bh
	%p2 =l mul %ah, %bmask
	%p3 =l mul %ah, %bh
	%mid0 =l shr %p0, 32
	%p1lo =l and %p1, 4294967295
	%mid1 =l add %mid0, %p1lo
	%p2lo =l and %p2, 4294967295
	%mid2 =l add %mid1, %p2lo
	%midhi =l shr %mid2, 32
	%plo =l and %p0, 4294967295
	%midlo =l shl %mid2, 32
	%lo =l or %plo, %midlo
	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%h0 =l add %p3, %p1hi
	%h1 =l add %h0, %p2hi
	%hi =l add %h1, %midhi
	%k =l add %oi, %ji
	%rp =l add %r, %k
	%old =l loadl %rp
	%s0 =l add %old, %lo
	%c0 =w cultl %s0, %old
	storel %s0, %rp
	%kp1 =l add %k, 1
	%rp1 =l add %r, %kp1
	%old1 =l loadl %rp1
	%carry =l extuw %c0
	%s1a =l add %old1, %hi
	%c1a =w cultl %s1a, %old1
	%s1 =l add %s1a, %carry
	%c1b =w cultl %s1, %s1a
	%carryout =w or %c1a, %c1b
	storel %s1, %rp1
	%kp2 =l add %k, 2
	jmp @prop

@prop
	%pi =l phi @calc %kp2, @propbody %pinext
	%pc =w phi @calc %carryout, @propbody %pcnext
	%pd =w ceqw %pc, 0
	%past =w cslel %total, %pi
	%stop =w or %pd, %past
	jnz %stop, @innerdone, @propbody

@propbody
	%rpi =l add %r, %pi
	%pv =l loadl %rpi
	%pvs =l add %pv, 1
	%pco =w cultl %pvs, %pv
	storel %pvs, %rpi
	%pinext =l add %pi, 1
	%pcnext =w copy %pco
	jmp @prop

@innerdone
	%jnext =l add %ji, 1
	jmp @inner

@outerdone
	%oinext =l add %oi, 1
	jmp @outer

@findstart
	%last0 =l sub %total, 1
	jmp @find

@find
	%fi =l phi @findstart %last0, @findstep %fprev
	%fzero =w csltl %fi, 0
	jnz %fzero, @retzero, @check

@check
	%fp =l add %r, %fi
	%fv =l loadl %fp
	%iszero =w ceql %fv, 0
	jnz %iszero, @findstep, @retcount

@findstep
	%fprev =l sub %fi, 1
	jmp @find

@retcount
	%result =l add %fi, 1
	ret %result

@retzero
	ret 0
}
