reverse:
    cmp rsi, 1
    jbe .Ldone
    xor rcx, rcx
    lea rdx, [rsi - 1]
.Lloop:
    cmp rcx, rdx
    jae .Ldone
    mov rax, qword ptr [rdi + rcx*8]
    mov r8, qword ptr [rdi + rdx*8]
    mov qword ptr [rdi + rcx*8], r8
    mov qword ptr [rdi + rdx*8], rax
    inc rcx
    dec rdx
    jmp .Lloop
.Ldone:
    xor eax, eax
    ret
