export function l $crc32(l %p, l %n) {
@start
    jmp @outer

@outer
    %pp =l phi @start %p, @byteend %nextp
    %rem =l phi @start %n, @byteend %nextrem
    %crc =w phi @start 4294967295, @byteend %finalcrc
    %has =w cnel %rem, 0
    jnz %has, @byte, @return

@byte
    %b =w loadub %pp
    %bc =w xor %crc, %b
    jmp @inner

@inner
    %j =w phi @byte 0, @mix %jnext
    %c =w phi @byte %bc, @mix %shifted
    %more =w csltw %j, 8
    jnz %more, @bit, @byteend

@bit
    %low =w and %c, 1
    %shifted =w shr %c, 1
    jnz %low, @xorpoly, @plain

@xorpoly
    %poly =w xor %shifted, 3988292384
    jmp @mix

@plain
    jmp @mix

@mix
    %shifted =w phi @xorpoly %poly, @plain %shifted
    %jnext =w add %j, 1
    jmp @inner

@byteend
    %finalcrc =w phi @inner %c
    %nextp =l add %pp, 1
    %nextrem =l sub %rem, 1
    jmp @outer

@return
    %out =w xor %crc, 4294967295
    %result =l extuw %out
    ret %result
}
