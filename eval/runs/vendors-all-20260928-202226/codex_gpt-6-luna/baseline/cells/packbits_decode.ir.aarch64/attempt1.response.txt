export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	%leftp =l alloc8 8
	%sp =l alloc8 8
	%valp =l alloc8 8
	%modep =l alloc8 8
	%zerol =l copy 0
	storel %zerol, %ip
	storel %zerol, %op
	jmp @packet

@packet
	%pi =l loadl %ip
	%more =w csltl %pi, %n
	jnz %more, @header, @done

@header
	%hp =l add %src, %pi
	%hw =w loadsb %hp
	%hl =l extsw %hw
	%isnoop =w ceqw %hw, -128
	jnz %isnoop, @noop, @classify

@noop
	%ni =l add %pi, 1
	storel %ni, %ip
	jmp @packet

@classify
	%lit =w csgel %hl, 0
	jnz %lit, @literal, @repeat

@literal
	%count =l add %hl, 1
	%start =l add %pi, 1
	%end =l add %start, %count
	%valid =w cslel %end, %n
	jnz %valid, @lit_valid, @bad

@lit_valid
	storel %end, %ip
	storel %start, %sp
	storel %count, %leftp
	storel 0, %modep
	jmp @prepare

@repeat
	%count2 =l sub 1, %hl
	%end2 =l add %pi, 2
	%valid2 =w cslel %end2, %n
	jnz %valid2, @rep_valid, @bad

@rep_valid
	%vp =l add %src, %pi
	%vaddr =l add %vp, 1
	%byte =w loadub %vaddr
	%byte64 =l extuw %byte
	%ni2 =l add %pi, 2
	storel %ni2, %ip
	storel %byte64, %valp
	storel %count2, %leftp
	storel 1, %modep
	jmp @prepare

@prepare
	%oi =l loadl %op
	%room =l sub %cap, %oi
	%plen =l loadl %leftp
	%overflow =w csgtl %plen, %room
	jnz %overflow, @truncate, @full_count

@full_count
	storel %plen, %leftp
	jmp @emit_init

@truncate
	storel %room, %leftp
	jmp @emit_init

@emit_init
	%srcpos =l loadl %sp
	storel %srcpos, %sp
	jmp @emit

@emit
	%left =l loadl %leftp
	%continue =w csgtl %left, 0
	jnz %continue, @emit_one, @packet_result

@emit_one
	%mode =l loadl %modep
	%isrep =w cnel %mode, 0
	jnz %isrep, @emit_repeat, @emit_literal

@emit_repeat
	%value =l loadl %valp
	%oi2 =l loadl %op
	%dst =l add %out, %oi2
	storeb %value, %dst
	%oi3 =l add %oi2, 1
	storel %oi3, %op
	%left2 =l loadl %leftp
	%left3 =l sub %left2, 1
	storel %left3, %leftp
	jmp @emit

@emit_literal
	%pos =l loadl %sp
	%from =l add %src, %pos
	%b =w loadub %from
	%oi4 =l loadl %op
	%to =l add %out, %oi4
	storeb %b, %to
	%oi5 =l add %oi4, 1
	storel %oi5, %op
	%pos2 =l add %pos, 1
	storel %pos2, %sp
	%left4 =l loadl %leftp
	%left5 =l sub %left4, 1
	storel %left5, %leftp
	jmp @emit

@packet_result
	%oi6 =l loadl %op
	%pi6 =l loadl %ip
	%used =l sub %oi6, %pi6
	%unused =l copy %used
	%origcount =l copy %unused
	%room2 =l sub %cap, %oi6
	%isfull =w ceql %room2, 0
	jnz %isfull, @check_trunc, @packet

@check_trunc
	%prior =l loadl %op
	%atcap =w ceql %prior, %cap
	jnz %atcap, @truncated_return, @packet

@truncated_return
	ret -1

@bad
	ret -2

@done
	%result =l loadl %op
	ret %result
}
