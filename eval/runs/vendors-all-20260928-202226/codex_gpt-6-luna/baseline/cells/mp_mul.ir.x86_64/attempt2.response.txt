export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
    %zero =l copy 0
    %mask =l copy 4294967295
    %n =l add %na, %nb
    jmp @zero_loop

@zero_loop
    %zi =l phi @entry %zero, @zero_body %zn
    %zc =w cultl %zi, %n
    jnz %zc, @zero_body, @outer_init

@zero_body
    %zoff =l shl %zi, 3
    %zp =l add %r, %zoff
    storel %zero, %zp
    %zn =l add %zi, 1
    jmp @zero_loop

@outer_init
    jmp @outer

@outer
    %i =l phi @outer_init %zero, @outer_next %inext
    %ic =w cultl %i, %na
    jnz %ic, @inner_init, @scan_init

@inner_init
    jmp @inner

@inner
    %j =l phi @inner_init %zero, @inner_next %jnext
    %carry =l phi @inner_init %zero, @inner_next %carryout
    %jc =w cultl %j, %nb
    jnz %jc, @mul, @prop_init

@mul
    %aoff =l shl %i, 3
    %ap =l add %a, %aoff
    %av =l loadl %ap
    %boff =l shl %j, 3
    %bp =l add %b, %boff
    %bv =l loadl %bp
    %a0 =l and %av, %mask
    %a1 =l shr %av, 32
    %b0 =l and %bv, %mask
    %b1 =l shr %bv, 32
    %p00 =l mul %a0, %b0
    %p01 =l mul %a0, %b1
    %p10 =l mul %a1, %b0
    %p11 =l mul %a1, %b1
    %m0 =l shr %p00, 32
    %m1 =l and %p01, %mask
    %ms1 =l add %m0, %m1
    %mc1 =w cultl %ms1, %m0
    %m2 =l and %p10, %mask
    %ms2 =l add %ms1, %m2
    %mc2 =w cultl %ms2, %ms1
    %mc =l extuw %mc1
    %mcx =l add %mc, %mc2
    %lohi =l shl %ms2, 32
    %lolow =l and %p00, %mask
    %lo =l or %lohi, %lolow
    %h0 =l shr %p01, 32
    %h1 =l shr %p10, 32
    %h2 =l add %p11, %h0
    %h3 =l add %h2, %h1
    %h4 =l add %h3, %mcx
    %k =l add %i, %j
    %koff =l shl %k, 3
    %rp =l add %r, %koff
    %old =l loadl %rp
    %s0 =l add %old, %lo
    %c0 =w cultl %s0, %old
    %s1 =l add %s0, %carry
    %c1 =w cultl %s1, %s0
    storel %s1, %rp
    %c0l =l extuw %c0
    %c1l =l extuw %c1
    %hc =l add %h4, %c0l
    %carryout =l add %hc, %c1l
    %jnext =l add %j, 1
    jmp @inner

@prop_init
    %pk0 =l add %i, %nb
    jmp @prop

@prop
    %pk =l phi @prop_init %pk0, @prop_body %pknext
    %pcarry =l phi @prop_init %carry, @prop_body %nextcarry
    %pc =w cultl %pk, %n
    jnz %pc, @prop_body, @outer_next

@prop_body
    %poff =l shl %pk, 3
    %pp =l add %r, %poff
    %pv =l loadl %pp
    %ps =l add %pv, %pcarry
    %pov =w cultl %ps, %pv
    storel %ps, %pp
    %nextcarry =l extuw %pov
    %pknext =l add %pk, 1
    jmp @prop

@outer_next
    %inext =l add %i, 1
    jmp @outer

@scan_init
    jmp @scan

@scan
    %si =l phi @scan_init %n, @scan_body %siprev
    %snz =w cnel %si, %zero
    jnz %snz, @scan_body, @done

@scan_body
    %siprev =l sub %si, 1
    %soff =l shl %siprev, 3
    %sp =l add %r, %soff
    %sv =l loadl %sp
    %svz =w cnel %sv, %zero
    jnz %svz, @done, @scan

@done
    %result =l phi @scan %zero, @scan_body %si
    ret %result
}
