export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @check
@check
    %i =l phi 0, @entry, %nexti, @body
    %p =l phi %a, @entry, %nextp, @body
    %v =l phi %start, @entry, %nextv, @body
    %more =w csltl %i, %n
    jnz %more, @body, @end
@body
    storel %v, %p
    %nexti =l add %i, 1
    %nextp =l add %p, 8
    %nextv =l sub %v, 1
    jmp @check
@end
    ret %n
}
