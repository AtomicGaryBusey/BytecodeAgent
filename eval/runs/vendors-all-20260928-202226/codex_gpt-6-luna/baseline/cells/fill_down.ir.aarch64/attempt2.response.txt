export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @check
@check
    %i =l phi @entry 0, @body %nexti
    %p =l phi @entry %a, @body %nextp
    %v =l phi @entry %start, @body %nextv
    %more =w csltl %i, %n
    jnz %more, @body, @end
@body
    storel %v, %p
    %nexti =l add %i, 1
    %nextp =l add %p, 8
    %nextv =l sub %v, 1
    jmp @check
@end
    ret %n
}
