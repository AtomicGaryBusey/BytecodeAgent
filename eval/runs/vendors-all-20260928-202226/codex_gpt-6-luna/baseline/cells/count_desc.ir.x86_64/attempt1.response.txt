export function l $count_desc(l %a, l %n) {
@start
	%small = cslel %n, 1
	jnz %small, @zero, @init
@zero
	ret 0
@init
	jmp @loop
@loop
	%i = phi @init 0, @body %inext
	%count = phi @init 0, @body %countnext
	%bound = sub %n, 1
	%more = csltl %i, %bound
	jnz %more, @body, @done
@body
	%offset = mul %i, 8
	%ptr = add %a, %offset
	%nextptr = add %ptr, 8
	%current = loadl %ptr
	%next = loadl %nextptr
	%desc = csltl %next, %current
	%descl = extuw %desc
	%countnext = add %count, %descl
	%inext = add %i, 1
	jmp @loop
@done
	ret %count
}
