export function l $count_desc(l %a, l %n) {
@start
	%bound =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%count =l phi @start 0, @body %countnext
	%more =w csltl %i, %bound
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%nextptr =l add %ptr, 8
	%current =l loadl %ptr
	%next =l loadl %nextptr
	%desc =w csltl %next, %current
	%descl =l extuw %desc
	%countnext =l add %count, %descl
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
}
