memmove_i64:
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    jbe .Lforward
    mov rax, rdx
    shl rax, 3
    mov rcx, rdi
    sub rcx, rsi
    cmp rcx, rax
    jae .Lforward
    lea rdi, [rdi + rax - 8]
    lea rsi, [rsi + rax - 8]
.Lbackward:
    mov rcx, qword ptr [rsi]
    mov qword ptr [rdi], rcx
    sub rsi, 8
    sub rdi, 8
    dec rdx
    jnz .Lbackward
    jmp .Ldone
.Lforward:
    mov rcx, qword ptr [rsi]
    mov qword ptr [rdi], rcx
    add rsi, 8
    add rdi, 8
    dec rdx
    jnz .Lforward
.Ldone:
    xor eax, eax
    ret
