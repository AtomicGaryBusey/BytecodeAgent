export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @outer(0, 0)

@outer
    %si =l phi @start 0, @next %sn
    %oi =l phi @start 0, @next %on
    %more =w csltl %si, %n
    jnz %more, @header, @done

@header
    %hp =l add %src, %si
    %h =w loadsb %hp
    %sn =l add %si, 1
    %noop =w ceqw %h, -128
    jnz %noop, @next, @classify

@classify
    %lit =w csgel %h, 0
    jnz %lit, @literal, @repeat_check

@literal
    %count =l add %h, 1
    %send =l add %sn, %count
    %srcok =w cslel %send, %n
    jnz %srcok, @literal_capacity, @error

@literal_capacity
    %oend =l add %oi, %count
    %fits =w cslel %oend, %cap
    jnz %fits, @literal_full, @literal_partial

@literal_full
    jmp @copy_loop(%sn, %oi, %count, %count)

@literal_partial
    %remain =l sub %cap, %oi
    jmp @copy_loop(%sn, %oi, %remain, 0)

@copy_loop
    %cp =l phi @literal_full 0, @literal_partial 0, @copy_body %cpnext
    %cs =l phi @literal_full %sn, @literal_partial %sn, @copy_body %csnext
    %co =l phi @literal_full %oi, @literal_partial %oi, @copy_body %conext
    %left =l phi @literal_full %count, @literal_partial %remain, @copy_body %leftnext
    %copydone =w ceql %left, 0
    jnz %copydone, @copy_finish, @copy_body

@copy_body
    %sp =l add %src, %cs
    %v =w loadub %sp
    %dp =l add %out, %co
    storeb %v, %dp
    %cpnext =l add %cp, 1
    %csnext =l add %cs, 1
    %conext =l add %co, 1
    %leftnext =l sub %left, 1
    jmp @copy_loop

@copy_finish
    %truncated =w ceql %left, 0
    %fullcount =w ceql %cp, %count
    jnz %fullcount, @next, @overflow
    jmp @overflow

@repeat_check
    %hasvalue =w csltl %sn, %n
    jnz %hasvalue, @repeat_packet, @error

@repeat_packet
    %count =l sub 1, %h
    %vp =l add %src, %sn
    %value =w loadub %vp
    %send =l add %sn, 1
    %oend =l add %oi, %count
    %fits =w cslel %oend, %cap
    jnz %fits, @repeat_full, @repeat_partial

@repeat_full
    jmp @repeat_loop(%oi, %count, %count)

@repeat_partial
    %remain =l sub %cap, %oi
    jmp @repeat_loop(%oi, %remain, 0)

@repeat_loop
    %ro =l phi @repeat_full %oi, @repeat_partial %oi, @repeat_body %ronext
    %left =l phi @repeat_full %count, @repeat_partial %remain, @repeat_body %rleftnext
    %written =l phi @repeat_full %count, @repeat_partial 0, @repeat_body %rwrittennext
    %rdone =w ceql %left, 0
    jnz %rdone, @repeat_finish, @repeat_body

@repeat_body
    %dp =l add %out, %ro
    storeb %value, %dp
    %ronext =l add %ro, 1
    %rleftnext =l sub %left, 1
    %rwrittennext =l add %written, 1
    jmp @repeat_loop

@repeat_finish
    %rfull =w ceql %written, %count
    jnz %rfull, @next, @overflow

@next
    %nextsi =l phi @outer %sn, @header %sn
    %nextoi =l phi @outer %oi, @header %oi
    jmp @outer(%nextsi, %nextoi)

@error
    ret -2

@overflow
    ret -1

@done
    ret %oi
}
