export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    jmp @outer

@outer
    %si =l phi @start 0, @advance %nextsi
    %oi =l phi @start 0, @advance %nextoi
    %more =w csltl %si, %n
    jnz %more, @header, @done

@header
    %hp =l add %src, %si
    %h =w loadsb %hp
    %sn =l add %si, 1
    %noop =w ceqw %h, -128
    jnz %noop, @noop, @classify

@noop
    jmp @advance

@classify
    %lit =w csgew %h, 0
    jnz %lit, @literal, @repeatcheck

@literal
    %hl =l extsw %h
    %count =l add %hl, 1
    %send =l add %sn, %count
    %srcok =w cslel %send, %n
    jnz %srcok, @literalcap, @error

@literalcap
    %oend =l add %oi, %count
    %fits =w cslel %oend, %cap
    jnz %fits, @litloop, @litpartialsetup

@litloop
    %lc =l phi @literalcap 0, @litloop %lcnext
    %lo =l phi @literalcap %oi, @litloop %lonext
    %ll =l phi @literalcap %count, @litloop %llnext
    %ldone =w ceql %ll, 0
    jnz %ldone, @litfinish, @litbody

@litbody
    %lsp =l add %src, %sn
    %lsp2 =l add %lsp, %lc
    %lv =w loadub %lsp2
    %ldp =l add %out, %lo
    storeb %lv, %ldp
    %lcnext =l add %lc, 1
    %lonext =l add %lo, 1
    %llnext =l sub %ll, 1
    jmp @litloop

@litfinish
    %litnextsi =l copy %send
    %litnextoi =l copy %oend
    jmp @advance

@litpartialsetup
    %remain =l sub %cap, %oi
    jmp @litpartialloop

@litpartialloop
    %pc =l phi @litpartialsetup 0, @litpartialloop %pcnext
    %po =l phi @litpartialsetup %oi, @litpartialloop %ponext
    %pl =l phi @litpartialsetup %remain, @litpartialloop %plnext
    %pdone =w ceql %pl, 0
    jnz %pdone, @overflow, @litpartialbody

@litpartialbody
    %psp =l add %src, %sn
    %psp2 =l add %psp, %pc
    %pv =w loadub %psp2
    %pdp =l add %out, %po
    storeb %pv, %pdp
    %pcnext =l add %pc, 1
    %ponext =l add %po, 1
    %plnext =l sub %pl, 1
    jmp @litpartialloop

@repeatcheck
    %hasvalue =w csltl %sn, %n
    jnz %hasvalue, @repeatpacket, @error

@repeatpacket
    %hlr =l extsw %h
    %count =l sub 1, %hlr
    %vp =l add %src, %sn
    %value =w loadub %vp
    %rsend =l add %sn, 1
    %rend =l add %oi, %count
    %rfits =w cslel %rend, %cap
    jnz %rfits, @reploop, @repeatpartial

@reploop
    %ro =l phi @repeatpacket %oi, @reploop %ronext
    %rl =l phi @repeatpacket %count, @reploop %rlnext
    %rdone =w ceql %rl, 0
    jnz %rdone, @repfinish, @repbody

@repbody
    %rdp =l add %out, %ro
    storeb %value, %rdp
    %ronext =l add %ro, 1
    %rlnext =l sub %rl, 1
    jmp @reploop

@repfinish
    %repnextsi =l copy %rsend
    %repnextoi =l copy %rend
    jmp @advance

@repeatpartial
    %rremain =l sub %cap, %oi
    jmp @reppartialloop

@reppartialloop
    %rpo =l phi @repeatpartial %oi, @reppartialloop %rponext
    %rpl =l phi @repeatpartial %rremain, @reppartialloop %rplnext
    %rpdone =w ceql %rpl, 0
    jnz %rpdone, @overflow, @reppartialbody

@reppartialbody
    %rpdp =l add %out, %rpo
    storeb %value, %rpdp
    %rponext =l add %rpo, 1
    %rplnext =l sub %rpl, 1
    jmp @reppartialloop

@advance
    %nextsi =l phi @noop %sn, @litfinish %litnextsi, @repfinish %repnextsi
    %nextoi =l phi @noop %oi, @litfinish %litnextoi, @repfinish %repnextoi
    jmp @outer

@error
    ret -2

@overflow
    ret -1

@done
    ret %oi
}
