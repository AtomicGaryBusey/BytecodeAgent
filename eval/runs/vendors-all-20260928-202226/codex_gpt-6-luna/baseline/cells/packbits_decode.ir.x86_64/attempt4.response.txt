export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@outer
    %si =l phi @start 0, @noopdone %noopsi, @litdone %litsend, @repdone %repsend
    %oi =l phi @start 0, @noopdone %oi, @litdone %litoend, @repdone %repoend
    %more =w csltl %si, %n
    jnz %more, @header, @done

@start
    jmp @outer

@header
    %hp =l add %src, %si
    %h =w loadsb %hp
    %sn =l add %si, 1
    %noop =w ceqw %h, -128
    jnz %noop, @nooppacket, @classify

@nooppacket
    %noopsi =l copy %sn
    jmp @noopdone

@noopdone
    jmp @outer

@classify
    %islt =w csgew %h, 0
    jnz %islt, @literal, @repeatcheck

@literal
    %hl =l extsw %h
    %count =l add %hl, 1
    %send =l add %sn, %count
    %srcok =w cslel %send, %n
    jnz %srcok, @literalcheck, @error

@literalcheck
    %oend =l add %oi, %count
    %fits =w cslel %oend, %cap
    jnz %fits, @litsetup, @litpartialsetup

@litsetup
    jmp @litloop

@litloop
    %lc =l phi @litsetup 0, @litbody %lcnext
    %lo =l phi @litsetup %oi, @litbody %lonext
    %ll =l phi @litsetup %count, @litbody %llnext
    %ldone =w ceql %ll, 0
    jnz %ldone, @litfinish, @litbody

@litbody
    %lsp =l add %src, %sn
    %lsp2 =l add %lsp, %lc
    %lv =w loadub %lsp2
    %ldp =l add %out, %lo
    storeb %lv, %ldp
    %lcnext =l add %lc, 1
    %lonext =l add %lo, 1
    %llnext =l sub %ll, 1
    jmp @litloop

@litfinish
    %litsend =l copy %send
    %litoend =l copy %oend
    jmp @litdone

@litdone
    jmp @outer

@litpartialsetup
    %lrem =l sub %cap, %oi
    jmp @litpartial

@litpartial
    %pc =l phi @litpartialsetup 0, @litpartialbody %pcnext
    %po =l phi @litpartialsetup %oi, @litpartialbody %ponext
    %pl =l phi @litpartialsetup %lrem, @litpartialbody %plnext
    %pdone =w ceql %pl, 0
    jnz %pdone, @overflow, @litpartialbody

@litpartialbody
    %psp =l add %src, %sn
    %psp2 =l add %psp, %pc
    %pv =w loadub %psp2
    %pdp =l add %out, %po
    storeb %pv, %pdp
    %pcnext =l add %pc, 1
    %ponext =l add %po, 1
    %plnext =l sub %pl, 1
    jmp @litpartial

@repeatcheck
    %hasvalue =w csltl %sn, %n
    jnz %hasvalue, @repeatpacket, @error

@repeatpacket
    %hr =l extsw %h
    %rcount =l sub 1, %hr
    %vp =l add %src, %sn
    %value =w loadub %vp
    %rsend =l add %sn, 1
    %rend =l add %oi, %rcount
    %rfits =w cslel %rend, %cap
    jnz %rfits, @reploopsetup, @reppartialsetup

@reploopsetup
    jmp @reploop

@reploop
    %ro =l phi @reploopsetup %oi, @repbody %ronext
    %rl =l phi @reploopsetup %rcount, @repbody %rlnext
    %rdone =w ceql %rl, 0
    jnz %rdone, @repfinish, @repbody

@repbody
    %rdp =l add %out, %ro
    storeb %value, %rdp
    %ronext =l add %ro, 1
    %rlnext =l sub %rl, 1
    jmp @reploop

@repfinish
    %repsend =l copy %rsend
    %repoend =l copy %rend
    jmp @repdone

@repdone
    jmp @outer

@reppartialsetup
    %rrem =l sub %cap, %oi
    jmp @reppartial

@reppartial
    %rpo =l phi @reppartialsetup %oi, @reppartialbody %rponext
    %rpl =l phi @reppartialsetup %rrem, @reppartialbody %rplnext
    %rpdone =w ceql %rpl, 0
    jnz %rpdone, @overflow, @reppartialbody

@reppartialbody
    %rpdp =l add %out, %rpo
    storeb %value, %rpdp
    %rponext =l add %rpo, 1
    %rplnext =l sub %rpl, 1
    jmp @reppartial

@error
    ret -2

@overflow
    ret -1

@done
    ret %oi
}
