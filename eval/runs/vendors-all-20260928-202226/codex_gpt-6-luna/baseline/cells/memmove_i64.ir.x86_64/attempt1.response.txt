export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%diff =l sub %dst, %src
	%bytes =l mul %n, 8
	%overlap =w cultl %diff, %bytes
	jnz %overlap, @back, @forward

@forward
	%fi =l phi @entry 0, @fbody %fnext
	%fmore =w csltl %fi, %n
	jnz %fmore, @fbody, @done

@fbody
	%foff =l mul %fi, 8
	%fsa =l add %src, %foff
	%fda =l add %dst, %foff
	%fval =l loadl %fsa
	storel %fval, %fda
	%fnext =l add %fi, 1
	jmp @forward

@back
	%bi =l phi @entry %n, @bbody %bnext
	%bmore =w csgtl %bi, 0
	jnz %bmore, @bbody, @done

@bbody
	%bnext =l sub %bi, 1
	%boff =l mul %bnext, 8
	%bsa =l add %src, %boff
	%bda =l add %dst, %boff
	%bval =l loadl %bsa
	storel %bval, %bda
	jmp @back

@done
	ret 0
}
