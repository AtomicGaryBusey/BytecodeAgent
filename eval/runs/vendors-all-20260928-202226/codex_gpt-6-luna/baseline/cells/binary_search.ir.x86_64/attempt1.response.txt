export function l $binary_search(l %a, l %n, l %key) {
@start
    jmp @loop
@loop
    %lo =l phi @start 0, @less %lo, @greater %next
    %hi =l phi @start %n, @less %mid, @greater %hi
    %valid =w csltl %lo, %hi
    jnz %valid, @search, @miss
@search
    %diff =l sub %hi, %lo
    %half =l shr %diff, 1
    %mid =l add %lo, %half
    %offset =l mul %mid, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %equal =w ceql %value, %key
    jnz %equal, @found, @compare
@compare
    %less =w csltl %key, %value
    jnz %less, @less, @greater
@less
    jmp @loop
@greater
    %next =l add %mid, 1
    jmp @loop
@found
    ret %mid
@miss
    %minusone =l copy -1
    ret %minusone
}
