export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @init

@init
    %j =l phi @start 0, @initbody %jnext
    %initdone =w csltl %n, %j
    jnz %initdone, @outerstart, @initbody

@initbody
    %roff =l mul %j, 8
    %rp =l add %row, %roff
    storel 0, %rp
    %jnext =l add %j, 1
    jmp @init

@outerstart
    jmp @outer

@outer
    %i =l phi @outerstart 0, @innerdone %inext
    %idone =w cslel %m, %i
    jnz %idone, @return, @innerstart

@innerstart
    %xp =l add %x, %i
    %xb =w loadub %xp
    jmp @inner

@inner
    %j2 =l phi @innerstart 1, @store %j2next
    %diag =l phi @innerstart 0, @store %up
    %jdone =w csltl %n, %j2
    jnz %jdone, @innerdone, @innerbody

@innerbody
    %yidx =l sub %j2, 1
    %yp =l add %y, %yidx
    %yb =w loadub %yp
    %roff2 =l mul %j2, 8
    %rp2 =l add %row, %roff2
    %up =l loadl %rp2
    %leftoff =l sub %roff2, 8
    %leftp =l add %row, %leftoff
    %left =l loadl %leftp
    %equal =w ceqw %xb, %yb
    jnz %equal, @match, @nomatch

@match
    %inc =l add %diag, 1
    jmp @store

@nomatch
    %less =w csltl %up, %left
    jnz %less, @leftcase, @upcase

@leftcase
    jmp @storeleft

@upcase
    jmp @storeup

@storeleft
    storel %left, %rp2
    jmp @advance

@storeup
    storel %up, %rp2
    jmp @advance

@store
    storel %inc, %rp2
    jmp @advance

@advance
    %j2next =l add %j2, 1
    jmp @inner

@innerdone
    %inext =l add %i, 1
    jmp @outer

@return
    %lastoff =l mul %n, 8
    %lastp =l add %row, %lastoff
    %result =l loadl %lastp
    ret %result
}
