export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @init

@init
    %j =l phi @start 0, @initbody %jnext
    %initmore =w cslel %j, %n
    jnz %initmore, @initbody, @outerstart

@initbody
    %initoff =l mul %j, 8
    %initp =l add %row, %initoff
    storel 0, %initp
    %jnext =l add %j, 1
    jmp @init

@outerstart
    jmp @outer

@outer
    %i =l phi @outerstart 0, @innerdone %inext
    %imore =w csltl %i, %m
    jnz %imore, @innerstart, @return

@innerstart
    %xp =l add %x, %i
    %xb =w loadub %xp
    jmp @inner

@inner
    %j2 =l phi @innerstart 1, @advance %j2next
    %diag =l phi @innerstart 0, @advance %up
    %jmore =w cslel %j2, %n
    jnz %jmore, @innerbody, @innerdone

@innerbody
    %yidx =l sub %j2, 1
    %yp =l add %y, %yidx
    %yb =w loadub %yp
    %roff =l mul %j2, 8
    %rp =l add %row, %roff
    %up =l loadl %rp
    %leftoff =l sub %roff, 8
    %leftp =l add %row, %leftoff
    %left =l loadl %leftp
    %equal =w ceqw %xb, %yb
    jnz %equal, @match, @nomatch

@match
    %inc =l add %diag, 1
    jmp @update

@nomatch
    %less =w csltl %up, %left
    jnz %less, @leftcase, @upcase

@leftcase
    jmp @update

@upcase
    jmp @update

@update
    %chosen =l phi @match %inc, @leftcase %left, @upcase %up
    storel %chosen, %rp
    jmp @advance

@advance
    %j2next =l add %j2, 1
    jmp @inner

@innerdone
    %inext =l add %i, 1
    jmp @outer

@return
    %lastoff =l mul %n, 8
    %lastp =l add %row, %lastoff
    %result =l loadl %lastp
    ret %result
}
