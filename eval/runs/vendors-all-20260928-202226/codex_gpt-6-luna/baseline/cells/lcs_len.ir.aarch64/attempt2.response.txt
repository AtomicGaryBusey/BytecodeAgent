export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %j0 =l copy 0
    jmp @init

@init
    %j =l phi @start %j0, @initbody %jnext
    %initdone =w csltl %n, %j
    jnz %initdone, @outerstart, @initbody

@initbody
    %roff =l mul %j, 8
    %rp =l add %row, %roff
    storel 0, %rp
    %jnext =l add %j, 1
    jmp @init

@outerstart
    %i0 =l copy 0
    jmp @outer

@outer
    %i =l phi @outerstart %i0, @innerdone %inext
    %idone =w cslel %m, %i
    jnz %idone, @return, @innerstart

@innerstart
    %xp =l add %x, %i
    %xb =w loadub %xp
    jmp @inner

@inner
    %j2 =l phi @innerstart 1, @innerbody %j2next
    %diag =l phi @innerstart 0, @innerbody %up
    %jdone =w csltl %n, %j2
    jnz %jdone, @innerdone, @innerbody

@innerbody
    %yidx =l sub %j2, 1
    %yp =l add %y, %yidx
    %yb =w loadub %yp
    %roff2 =l mul %j2, 8
    %rp2 =l add %row, %roff2
    %up =l loadl %rp2
    %leftoff =l sub %roff2, 8
    %leftp =l add %row, %leftoff
    %left =l loadl %leftp
    %equal =w ceqw %xb, %yb
    %inc =l add %diag, 1
    %less =w csltl %up, %left
    %best =l sell %less, %left, %up
    %value =l sell %equal, %inc, %best
    storel %value, %rp2
    %j2next =l add %j2, 1
    jmp @inner

@innerdone
    %inext =l add %i, 1
    jmp @outer

@return
    %lastoff =l mul %n, 8
    %lastp =l add %row, %lastoff
    %result =l loadl %lastp
    ret %result
}
