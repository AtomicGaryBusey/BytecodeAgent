export function l $popcount(l %x) {
@start
    %a = shr %x, 1
    %b = and %a, 0x5555555555555555
    %c = sub %x, %b
    %d = and %c, 0x3333333333333333
    %e = shr %c, 2
    %f = and %e, 0x3333333333333333
    %g = add %d, %f
    %h = shr %g, 4
    %i = add %g, %h
    %j = and %i, 0x0f0f0f0f0f0f0f0f
    %k = shr %j, 8
    %l = add %j, %k
    %m = shr %l, 16
    %n = add %l, %m
    %o = shr %n, 32
    %p = add %n, %o
    %q = and %p, 0x7f
    ret %q
}
