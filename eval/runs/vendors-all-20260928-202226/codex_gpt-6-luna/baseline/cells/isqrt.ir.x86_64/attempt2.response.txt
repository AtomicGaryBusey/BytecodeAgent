export function l $isqrt(l %n) {
@start
	%lo =l alloc8 8
	%hi =l alloc8 8
	storel 0, %lo
	storel 4294967296, %hi
	jmp @loop
@loop
	%l =l loadl %lo
	%h =l loadl %hi
	%gap =l sub %h, %l
	%done =w culel %gap, 1
	jnz %done, @exit, @step
@step
	%half =l udiv %gap, 2
	%mid =l add %l, %half
	%q =l udiv %n, %mid
	%fits =w culel %mid, %q
	jnz %fits, @raise, @lower
@raise
	storel %mid, %lo
	jmp @loop
@lower
	storel %mid, %hi
	jmp @loop
@exit
	%result =l loadl %lo
	ret %result
}
