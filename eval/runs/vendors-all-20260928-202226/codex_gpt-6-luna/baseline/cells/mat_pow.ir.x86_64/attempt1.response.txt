export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %i =l alloc8 8
    %j =l alloc8 8
    %t =l alloc8 8
    %iter =l alloc8 8
    %sum =l alloc8 8
    %idx =l alloc8 8
    %p =l alloc8 8
    %q =l alloc8 8
    %r =l alloc8 8
    storel 0, %i
    jmp @init

@init
    %iv =l loadl %i
    %initcond =w csltl %iv, %n
    jnz %initcond, @initbody, @initdone

@initbody
    storel 0, %j
    jmp @initj

@initj
    %jv =l loadl %j
    %jcond =w csltl %jv, %n
    jnz %jcond, @initjbody, @initnext

@initjbody
    %row =l mull %iv, %n
    %linear =l addl %row, %jv
    %outptr =l addl %out, mull %linear, 8
    %diag =w ceql %iv, %jv
    jnz %diag, @diagone, @diagzero

@diagone
    storel 1, %outptr
    jmp @initjnext

@diagzero
    storel 0, %outptr
    jmp @initjnext

@initjnext
    %jn =l addl %jv, 1
    storel %jn, %j
    jmp @initj

@initnext
    %in =l addl %iv, 1
    storel %in, %i
    jmp @init

@initdone
    storel 0, %i
    jmp @copybase

@copybase
    %ci =l loadl %i
    %cc =w csltl %ci, mull %n, %n
    jnz %cc, @copybody, @copydone

@copybody
    %ap =l addl %a, mull %ci, 8
    %av =l loadl %ap
    %inext =l addl %ci, 1
    storel %av, addl %tmp, mull %ci, 8
    storel %inext, %i
    jmp @copybase

@copydone
    storel 0, %iter
    jmp @outer

@outer
    %it =l loadl %iter
    %more =w csltl %it, %k
    jnz %more, @prodstart, @clearstart

@prodstart
    storel 0, %i
    jmp @prodi

@prodi
    %pi =l loadl %i
    %pic =w csltl %pi, %n
    jnz %pic, @prodjstart, @proddone

@prodjstart
    storel 0, %j
    jmp @prodj

@prodj
    %pj =l loadl %j
    %pjc =w csltl %pj, %n
    jnz %pjc, @prodtstart, @prodinext

@prodtstart
    storel 0, %t
    storel 0, %sum
    jmp @prodt

@prodt
    %pt =l loadl %t
    %ptc =w csltl %pt, %n
    jnz %ptc, @prodtbody, @prodtend

@prodtbody
    %oi =l addl mull %pi, %n, %pt
    %oiaddr =l addl %out, mull %oi, 8
    %x =l loadl %oiaddr
    %aj =l addl mull %pt, %n, %pj
    %ajaddr =l addl %a, mull %aj, 8
    %y =l loadl %ajaddr
    %prod =l mull %x, %y
    %oldsum =l loadl %sum
    %newsum =l addl %oldsum, %prod
    storel %newsum, %sum
    %ptnext =l addl %pt, 1
    storel %ptnext, %t
    jmp @prodt

@prodtend
    %dstidx =l addl mull %pi, %n, %pj
    storel loadl %sum, addl %tmp, mull %dstidx, 8
    %pjnext =l addl %pj, 1
    storel %pjnext, %j
    jmp @prodj

@prodinext
    %pinext =l addl %pi, 1
    storel %pinext, %i
    jmp @prodi

@proddone
    storel 0, %i
    jmp @copyprod

@copyprod
    %xi =l loadl %i
    %xc =w csltl %xi, mull %n, %n
    jnz %xc, @copyprodbody, @copyproddone

@copyprodbody
    %src =l loadl addl %tmp, mull %xi, 8
    storel %src, addl %out, mull %xi, 8
    %xinext =l addl %xi, 1
    storel %xinext, %i
    jmp @copyprod

@copyproddone
    %itnext =l addl %it, 1
    storel %itnext, %iter
    jmp @outer

@clearstart
    storel 0, %i
    jmp @clear

@clear
    %zi =l loadl %i
    %zcount =l mull 2, mull %n, %n
    %zc =w csltl %zi, %zcount
    jnz %zc, @clearbody, @done

@clearbody
    storel 0, addl %tmp, mull %zi, 8
    %zinext =l addl %zi, 1
    storel %zinext, %i
    jmp @clear

@done
    ret 0
}
