export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %i =l alloc8 8
    %j =l alloc8 8
    %t =l alloc8 8
    %iter =l alloc8 8
    %sum =l alloc8 8
    storel 0, %i
    jmp @init_i

@init_i
    %iv =l loadl %i
    %ic =w csltl %iv, %n
    jnz %ic, @init_j_start, @init_done

@init_j_start
    storel 0, %j
    jmp @init_j

@init_j
    %jv =l loadl %j
    %jc =w csltl %jv, %n
    jnz %jc, @init_body, @init_next_i

@init_body
    %row =l mul %iv, %n
    %idx =l add %row, %jv
    %off =l mul %idx, 8
    %dst =l add %out, %off
    %diag =w ceql %iv, %jv
    jnz %diag, @init_one, @init_zero

@init_one
    storel 1, %dst
    jmp @init_j_next

@init_zero
    storel 0, %dst
    jmp @init_j_next

@init_j_next
    %jn =l add %jv, 1
    storel %jn, %j
    jmp @init_j

@init_next_i
    %in =l add %iv, 1
    storel %in, %i
    jmp @init_i

@init_done
    storel 0, %iter
    jmp @outer

@outer
    %it =l loadl %iter
    %more =w csltl %it, %k
    jnz %more, @prod_i_start, @clear_start

@prod_i_start
    storel 0, %i
    jmp @prod_i

@prod_i
    %pi =l loadl %i
    %pic =w csltl %pi, %n
    jnz %pic, @prod_j_start, @prod_done

@prod_j_start
    storel 0, %j
    jmp @prod_j

@prod_j
    %pj =l loadl %j
    %pjc =w csltl %pj, %n
    jnz %pjc, @prod_t_start, @prod_i_next

@prod_t_start
    storel 0, %t
    storel 0, %sum
    jmp @prod_t

@prod_t
    %pt =l loadl %t
    %ptc =w csltl %pt, %n
    jnz %ptc, @prod_body, @prod_t_done

@prod_body
    %pir =l mul %pi, %n
    %oidx =l add %pir, %pt
    %ooff =l mul %oidx, 8
    %optr =l add %out, %ooff
    %x =l loadl %optr
    %ptr =l mul %pt, %n
    %aidx =l add %ptr, %pj
    %aoff =l mul %aidx, 8
    %aptr =l add %a, %aoff
    %y =l loadl %aptr
    %prod =l mul %x, %y
    %oldsum =l loadl %sum
    %newsum =l add %oldsum, %prod
    storel %newsum, %sum
    %ptnext =l add %pt, 1
    storel %ptnext, %t
    jmp @prod_t

@prod_t_done
    %pr =l mul %pi, %n
    %didx =l add %pr, %pj
    %doff =l mul %didx, 8
    %dptr =l add %tmp, %doff
    %sval =l loadl %sum
    storel %sval, %dptr
    %pjnext =l add %pj, 1
    storel %pjnext, %j
    jmp @prod_j

@prod_i_next
    %pinext =l add %pi, 1
    storel %pinext, %i
    jmp @prod_i

@prod_done
    storel 0, %i
    jmp @copy_start

@copy_start
    %xi =l loadl %i
    %count =l mul %n, %n
    %xc =w csltl %xi, %count
    jnz %xc, @copy_body, @copy_done

@copy_body
    %xoff =l mul %xi, 8
    %src =l add %tmp, %xoff
    %val =l loadl %src
    %dst =l add %out, %xoff
    storel %val, %dst
    %xinext =l add %xi, 1
    storel %xinext, %i
    jmp @copy_start

@copy_done
    %itnext =l add %it, 1
    storel %itnext, %iter
    jmp @outer

@clear_start
    storel 0, %i
    jmp @clear

@clear
    %zi =l loadl %i
    %n2 =l mul %n, %n
    %zcount =l mul 2, %n2
    %zc =w csltl %zi, %zcount
    jnz %zc, @clear_body, @done

@clear_body
    %zoff =l mul %zi, 8
    %zptr =l add %tmp, %zoff
    storel 0, %zptr
    %zinext =l add %zi, 1
    storel %zinext, %i
    jmp @clear

@done
    ret 0
}
