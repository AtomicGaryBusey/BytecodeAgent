mp_mul:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rdx
mov r14, r8
mov r15, rsi
mov rbp, rcx
lea r10, [r15 + rbp]
xor r8d, r8d
.Lzero:
cmp r8, r10
jge .Lmultiply_start
mov qword ptr [r14 + r8*8], 0
inc r8
jmp .Lzero
.Lmultiply_start:
xor ebx, ebx
.Louter:
cmp rbx, r15
jge .Lfind_length
xor r9d, r9d
xor r11d, r11d
.Linner:
cmp r9, rbp
jge .Lstore_carry
mov rax, qword ptr [r12 + rbx*8]
mul qword ptr [r13 + r9*8]
lea r8, [rbx + r9]
add rax, qword ptr [r14 + r8*8]
adc rdx, 0
add rax, r11
adc rdx, 0
mov qword ptr [r14 + r8*8], rax
mov r11, rdx
inc r9
jmp .Linner
.Lstore_carry:
lea r8, [rbx + rbp]
mov qword ptr [r14 + r8*8], r11
inc rbx
jmp .Louter
.Lfind_length:
mov rax, r10
.Ltrim:
test rax, rax
jz .Ldone
dec rax
cmp qword ptr [r14 + rax*8], 0
je .Ltrim
inc rax
.Ldone:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
