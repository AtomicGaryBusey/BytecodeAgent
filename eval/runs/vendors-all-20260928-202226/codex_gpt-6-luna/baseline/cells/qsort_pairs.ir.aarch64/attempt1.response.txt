export function l $qsort_pairs(l %a, l %n) {
@start
	%small =l cslel %n, 1
	jnz %small, @done, @init

@init
	%mid =l div %n, 2
	%midoff =l shli %mid, 4
	%midptr =l add %a, %midoff
	%pk =l loadl %midptr
	%pvptr =l add %midptr, 8
	%pv =l loadl %pvptr
	%j0 =l sub %n, 1
	jmp @part(0, %j0)

@part
	%i =l phi %i
	%j =l phi %j
	%active =w cslel %i, %j
	jnz %active, @leftscan, @recurse

@leftscan
	%ioff =l shli %i, 4
	%iptr =l add %a, %ioff
	%ik =l loadl %iptr
	%ivptr =l add %iptr, 8
	%iv =l loadl %ivptr
	%iklt =w csltl %ik, %pk
	%ikeq =w ceql %ik, %pk
	%ivlt =w csltl %iv, %pv
	%ivltkey =w and %ikeq, %ivlt
	%ilt =w or %iklt, %ivltkey
	jnz %ilt, @leftnext, @rightscan

@leftnext
	%inext =l add %i, 1
	jmp @part(%inext, %j)

@rightscan
	%joff =l shli %j, 4
	%jptr =l add %a, %joff
	%jk =l loadl %jptr
	%jvptr =l add %jptr, 8
	%jv =l loadl %jvptr
	%jgtkey =w csgtl %jk, %pk
	%jkeq =w ceql %jk, %pk
	%jvgt =w csgtl %jv, %pv
	%jvgtkey =w and %jkeq, %jvgt
	%jgt =w or %jgtkey, %jvgt
	jnz %jgt, @rightnext, @swapcheck

@rightnext
	%jprev =l sub %j, 1
	jmp @part(%i, %jprev)

@swapcheck
	%crossed =w cslel %i, %j
	jnz %crossed, @swap, @recurse

@swap
	%ik0 =l loadl %iptr
	%iv0 =l loadl %ivptr
	%jk0 =l loadl %jptr
	%jv0 =l loadl %jvptr
	storel %jk0, %iptr
	storel %jv0, %ivptr
	storel %ik0, %jptr
	storel %iv0, %jvptr
	%inext2 =l add %i, 1
	%jprev2 =l sub %j, 1
	jmp @part(%inext2, %jprev2)

@recurse
	%leftn =l add %j, 1
	%leftmore =w csgtl %leftn, 1
	jnz %leftmore, @callleft, @checkright

@callleft
	%leftret =l call $qsort_pairs(l %a, l %leftn)
	jmp @checkright

@checkright
	%rightn =l sub %n, %i
	%rightmore =w csgtl %rightn, 1
	jnz %rightmore, @callright, @done

@callright
	%rightoff =l shli %i, 4
	%rightptr =l add %a, %rightoff
	%rightret =l call $qsort_pairs(l %rightptr, l %rightn)
	jmp @done

@done
	ret 0
}
