export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @begin

@begin
	%last =l sub %n, 1
	%lastoff =l shl %last, 4
	%lastptr =l add %a, %lastoff
	%pv =l loadl %lastptr
	%pvptr =l add %lastptr, 8
	%pval =l loadl %pvptr
	jmp @loop

@loop
	%i =l phi @begin 0, @swap %inext, @noswap %i
	%j =l phi @begin 0, @swap %jnext, @noswap %jnext
	%more =w csltl %j, %last
	jnz %more, @check, @finish

@check
	%joff =l shl %j, 4
	%jptr =l add %a, %joff
	%jk =l loadl %jptr
	%jvptr =l add %jptr, 8
	%jv =l loadl %jvptr
	%keylt =w csltl %jk, %pv
	%keyeq =w ceql %jk, %pv
	%vle =w cslel %jv, %pval
	%tie =w and %keyeq, %vle
	%le =w or %keylt, %tie
	jnz %le, @swap, @noswap

@swap
	%ioff =l shl %i, 4
	%iptr =l add %a, %ioff
	%ik =l loadl %iptr
	%ivptr =l add %iptr, 8
	%iv =l loadl %ivptr
	storel %jk, %iptr
	storel %jv, %ivptr
	storel %ik, %jptr
	storel %iv, %jvptr
	%inext =l add %i, 1
	%jnext =l add %j, 1
	jmp @loop

@noswap
	%jnext =l add %j, 1
	jmp @loop

@finish
	%ioff2 =l shl %i, 4
	%iptr2 =l add %a, %ioff2
	%ik2 =l loadl %iptr2
	%ivptr2 =l add %iptr2, 8
	%iv2 =l loadl %ivptr2
	storel %pv, %iptr2
	storel %pval, %ivptr2
	storel %ik2, %lastptr
	storel %iv2, %pvptr
	%leftmore =w csltl 1, %i
	jnz %leftmore, @leftcall, @rightcheck

@leftcall
	%leftret =l call $qsort_pairs(l %a, l %i)
	jmp @rightcheck

@rightcheck
	%rightstart =l add %i, 1
	%rightn =l sub %n, %rightstart
	%rightmore =w csltl 1, %rightn
	jnz %rightmore, @rightcall, @done

@rightcall
	%rightoff =l shl %rightstart, 4
	%rightptr =l add %a, %rightoff
	%rightret =l call $qsort_pairs(l %rightptr, l %rightn)
	jmp @done

@done
	ret 0
}
