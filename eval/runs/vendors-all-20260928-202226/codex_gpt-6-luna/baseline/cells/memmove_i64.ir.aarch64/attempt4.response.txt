export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %dst_after =w csgtl %dst, %src
    jnz %dst_after, @check_overlap, @forward_init

@check_overlap
    %distance =l sub %dst, %src
    %bytes =l mul %n, 8
    %overlap =w csltl %distance, %bytes
    jnz %overlap, @back_init, @forward_init

@forward_init
    jmp @forward_loop

@forward_loop
    %fi =l phi @forward_init 0, @forward_body %fnext
    %fmore =w csltl %fi, %n
    jnz %fmore, @forward_body, @done

@forward_body
    %foff =l mul %fi, 8
    %fsa =l add %src, %foff
    %fda =l add %dst, %foff
    %fv =l loadl %fsa
    storel %fv, %fda
    %fnext =l add %fi, 1
    jmp @forward_loop

@back_init
    jmp @back_loop

@back_loop
    %bi =l phi @back_init %n, @back_body %bnext
    %bmore =w csgtl %bi, 0
    jnz %bmore, @back_body, @done

@back_body
    %bprev =l sub %bi, 1
    %boff =l mul %bprev, 8
    %bsa =l add %src, %boff
    %bda =l add %dst, %boff
    %bv =l loadl %bsa
    storel %bv, %bda
    %bnext =l sub %bi, 1
    jmp @back_loop

@done
    ret 0
}
