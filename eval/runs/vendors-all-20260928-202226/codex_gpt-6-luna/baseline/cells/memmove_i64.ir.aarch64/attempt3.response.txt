export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %dst_after = csltl %src, %dst
    jnz %dst_after, @check_overlap, @forward_init

@check_overlap
    %distance = sub %dst, %src
    %bytes = mul %n, 8
    %overlap = csltl %distance, %bytes
    jnz %overlap, @back_init, @forward_init

@forward_init
    jmp @forward_loop

@forward_loop
    %fi =l phi @forward_init 0, @forward_body %fnext
    %fmore = csltl %fi, %n
    jnz %fmore, @forward_body, @done

@forward_body
    %foff = mul %fi, 8
    %fsa = add %src, %foff
    %fda = add %dst, %foff
    %fv = loadl %fsa
    storel %fv, %fda
    %fnext = add %fi, 1
    jmp @forward_loop

@back_init
    jmp @back_loop

@back_loop
    %bi =l phi @back_init %n, @back_body %bnext
    %bmore = csgtl %bi, 0
    jnz %bmore, @back_body, @done

@back_body
    %bprev = sub %bi, 1
    %boff = mul %bprev, 8
    %bsa = add %src, %boff
    %bda = add %dst, %boff
    %bv = loadl %bsa
    storel %bv, %bda
    %bnext = sub %bi, 1
    jmp @back_loop

@done
    ret 0
}
