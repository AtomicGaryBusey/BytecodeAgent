binary_search:
    mov x6, x0
    mov x5, x2
    mov x3, xzr
    mov x4, x1
.Lloop:
    cmp x3, x4
    b.ge .Lnot_found
    sub x7, x4, x3
    lsr x7, x7, #1
    add x7, x3, x7
    ldr x8, [x6, x7, lsl #3]
    cmp x8, x5
    b.eq .Lfound
    b.lt .Lright
    mov x4, x7
    b .Lloop
.Lright:
    add x3, x7, #1
    b .Lloop
.Lfound:
    mov x0, x7
    ret
.Lnot_found:
    mov x0, #-1
    ret
