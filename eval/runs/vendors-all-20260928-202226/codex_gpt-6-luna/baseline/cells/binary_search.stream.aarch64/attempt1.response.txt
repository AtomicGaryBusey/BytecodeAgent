binary_search:
    mov x2, x1
    mov x3, xzr
.Lloop:
    cmp x3, x2
    b.ge .Lnot_found
    sub x4, x2, x3
    lsr x4, x4, #1
    add x4, x3, x4
    ldr x5, [x0, x4, lsl #3]
    cmp x5, x1
    b.eq .Lfound
    b.lt .Lright
    mov x2, x4
    b .Lloop
.Lright:
    add x3, x4, #1
    b .Lloop
.Lfound:
    mov x0, x4
    ret
.Lnot_found:
    mov x0, #-1
    ret
