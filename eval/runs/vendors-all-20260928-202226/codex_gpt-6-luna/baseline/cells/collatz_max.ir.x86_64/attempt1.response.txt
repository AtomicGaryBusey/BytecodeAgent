export function l $collatz_max(l %n) {
@start
    %kp =l alloc8 8
    %xp =l alloc8 8
    %sp =l alloc8 8
    %mp =l alloc8 8
    storel 1, %kp
    storel 0, %mp
    jmp @outer

@outer
    %k =l loadl %kp
    %more =w cslel %k, %n
    jnz %more, @begin, @done

@begin
    storel %k, %xp
    storel 0, %sp
    jmp @inner

@inner
    %x =l loadl %xp
    %active =w cnel %x, 1
    jnz %active, @step, @finish

@step
    %parity =l and %x, 1
    %even =w ceql %parity, 0
    jnz %even, @halve, @triple

@halve
    %half =l shr %x, 1
    storel %half, %xp
    jmp @count

@triple
    %triple =l mul %x, 3
    %next =l add %triple, 1
    storel %next, %xp
    jmp @count

@count
    %steps =l loadl %sp
    %steps1 =l add %steps, 1
    storel %steps1, %sp
    jmp @inner

@finish
    %steps2 =l loadl %sp
    %maximum =l loadl %mp
    %larger =w csgt %steps2, %maximum
    jnz %larger, @update, @advance

@update
    storel %steps2, %mp
    jmp @advance

@advance
    %k1 =l add %k, 1
    storel %k1, %kp
    jmp @outer

@done
    %result =l loadl %mp
    ret %result
}
