export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcslot =l alloc8 8
	%depthslot =l alloc8 8
	%islot =l alloc8 8
	%retslot =l alloc8 8
	storel 0, %islot
	jmp @zero

@zero
	%i =l loadl %islot
	%more =w csltl %i, %cap
	jnz %more, @zero_body, @begin

@zero_body
	%ioff =l mul %i, 8
	%iptr =l add %stack, %ioff
	storel 0, %iptr
	%inext =l add %i, 1
	storel %inext, %islot
	jmp @zero

@begin
	storel 0, %pcslot
	storel 0, %depthslot
	jmp @exec

@exec
	%pc =l loadl %pcslot
	%depth =l loadl %depthslot
	%running =w csltl %pc, %len
	jnz %running, @fetch, @finish

@fetch
	%codeoff =l mul %pc, 8
	%codeptr =l add %code, %codeoff
	%op =l loadl %codeptr
	%ishalt =w ceql %op, 0
	jnz %ishalt, @finish, @check_push

@check_push
	%ispush =w ceql %op, 1
	jnz %ispush, @push, @check_add

@check_add
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @check_sub

@check_sub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @check_mul

@check_mul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @check_dup

@check_dup
	%isdup =w ceql %op, 5
	jnz %isdup, @dup, @check_swap

@check_swap
	%isswap =w ceql %op, 6
	jnz %isswap, @swap, @check_drop

@check_drop
	%isdrop =w ceql %op, 7
	jnz %isdrop, @drop, @check_over

@check_over
	%isover =w ceql %op, 8
	jnz %isover, @over, @check_lt

@check_lt
	%islt =w ceql %op, 9
	jnz %islt, @lt, @check_jz

@check_jz
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @check_jmp

@check_jmp
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jmp, @other

@push
	%argpc =l add %pc, 1
	%argoff =l mul %argpc, 8
	%argptr =l add %code, %argoff
	%value =l loadl %argptr
	%pushoff =l mul %depth, 8
	%pushptr =l add %stack, %pushoff
	storel %value, %pushptr
	%pushdepth =l add %depth, 1
	storel %pushdepth, %depthslot
	%pushpc =l add %pc, 2
	storel %pushpc, %pcslot
	jmp @exec

@add
	%aidx =l sub %depth, 2
	%bidx =l sub %depth, 1
	%aoff =l mul %aidx, 8
	%boff =l mul %bidx, 8
	%aptr =l add %stack, %aoff
	%bptr =l add %stack, %boff
	%a =l loadl %aptr
	%b =l loadl %bptr
	%sum =l add %a, %b
	storel %sum, %aptr
	%newdepth =l sub %depth, 1
	storel %newdepth, %depthslot
	%newpc =l add %pc, 1
	storel %newpc, %pcslot
	jmp @exec

@sub
	%aidx2 =l sub %depth, 2
	%bidx2 =l sub %depth, 1
	%aoff2 =l mul %aidx2, 8
	%boff2 =l mul %bidx2, 8
	%aptr2 =l add %stack, %aoff2
	%bptr2 =l add %stack, %boff2
	%a2 =l loadl %aptr2
	%b2 =l loadl %bptr2
	%diff =l sub %a2, %b2
	storel %diff, %aptr2
	%depth2 =l sub %depth, 1
	storel %depth2, %depthslot
	%pc2 =l add %pc, 1
	storel %pc2, %pcslot
	jmp @exec

@mul
	%aidx3 =l sub %depth, 2
	%bidx3 =l sub %depth, 1
	%aoff3 =l mul %aidx3, 8
	%boff3 =l mul %bidx3, 8
	%aptr3 =l add %stack, %aoff3
	%bptr3 =l add %stack, %boff3
	%a3 =l loadl %aptr3
	%b3 =l loadl %bptr3
	%product =l mul %a3, %b3
	storel %product, %aptr3
	%depth3 =l sub %depth, 1
	storel %depth3, %depthslot
	%pc3 =l add %pc, 1
	storel %pc3, %pcslot
	jmp @exec

@dup
	%topidx =l sub %depth, 1
	%topoff =l mul %topidx, 8
	%topptr =l add %stack, %topoff
	%top =l loadl %topptr
	%dupoff =l mul %depth, 8
	%dupptr =l add %stack, %dupoff
	storel %top, %dupptr
	%dupdepth =l add %depth, 1
	storel %dupdepth, %depthslot
	%duppc =l add %pc, 1
	storel %duppc, %pcslot
	jmp @exec

@swap
	%swapidxa =l sub %depth, 2
	%swapidxb =l sub %depth, 1
	%swapaoff =l mul %swapidxa, 8
	%swapboff =l mul %swapidxb, 8
	%swapaptr =l add %stack, %swapaoff
	%swapbptr =l add %stack, %swapboff
	%swapav =l loadl %swapaptr
	%swapbv =l loadl %swapbptr
	storel %swapbv, %swapaptr
	storel %swapav, %swapbptr
	%swappc =l add %pc, 1
	storel %swappc, %pcslot
	jmp @exec

@drop
	%dropdepth =l sub %depth, 1
	storel %dropdepth, %depthslot
	%droppc =l add %pc, 1
	storel %droppc, %pcslot
	jmp @exec

@over
	%overidx =l sub %depth, 2
	%overoff =l mul %overidx, 8
	%overptr =l add %stack, %overoff
	%overval =l loadl %overptr
	%overdestoff =l mul %depth, 8
	%overdest =l add %stack, %overdestoff
	storel %overval, %overdest
	%overdepth =l add %depth, 1
	storel %overdepth, %depthslot
	%overpc =l add %pc, 1
	storel %overpc, %pcslot
	jmp @exec

@lt
	%ltidxa =l sub %depth, 2
	%ltidxb =l sub %depth, 1
	%ltaoff =l mul %ltidxa, 8
	%ltboff =l mul %ltidxb, 8
	%ltaptr =l add %stack, %ltaoff
	%ltbptr =l add %stack, %ltboff
	%ltav =l loadl %ltaptr
	%ltbv =l loadl %ltbptr
	%ltw =w csltl %ltav, %ltbv
	%ltvalue =l extuw %ltw
	storel %ltvalue, %ltaptr
	%ltdepth =l sub %depth, 1
	storel %ltdepth, %depthslot
	%ltpc =l add %pc, 1
	storel %ltpc, %pcslot
	jmp @exec

@jz
	%jzidx =l sub %depth, 1
	%jzoff =l mul %jzidx, 8
	%jzptr =l add %stack, %jzoff
	%jzval =l loadl %jzptr
	%jzcond =w ceql %jzval, 0
	%jzargpc =l add %pc, 1
	%jzargoff =l mul %jzargpc, 8
	%jzargptr =l add %code, %jzargoff
	%jztarget =l loadl %jzargptr
	%jzfall =l add %pc, 2
	%jzdepth =l sub %depth, 1
	storel %jzdepth, %depthslot
	jnz %jzcond, @jz_take, @jz_skip

@jz_take
	storel %jztarget, %pcslot
	jmp @exec

@jz_skip
	storel %jzfall, %pcslot
	jmp @exec

@jmp
	%jmpargpc =l add %pc, 1
	%jmpargoff =l mul %jmpargpc, 8
	%jmpargptr =l add %code, %jmpargoff
	%jmptarget =l loadl %jmpargptr
	storel %jmptarget, %pcslot
	jmp @exec

@other
	%otherpc =l add %pc, 1
	storel %otherpc, %pcslot
	jmp @exec

@finish
	%finaldepth =l loadl %depthslot
	%nonempty =w csgtl %finaldepth, 0
	jnz %nonempty, @get_result, @empty_result

@get_result
	%resultidx =l sub %finaldepth, 1
	%resultoff =l mul %resultidx, 8
	%resultptr =l add %stack, %resultoff
	%result =l loadl %resultptr
	storel %result, %retslot
	jmp @clear_begin

@empty_result
	storel 0, %retslot
	jmp @clear_begin

@clear_begin
	storel %finaldepth, %islot
	jmp @clear

@clear
	%clearidx =l loadl %islot
	%clearmore =w csltl %clearidx, %cap
	jnz %clearmore, @clear_body, @done

@clear_body
	%clearoff =l mul %clearidx, 8
	%clearptr =l add %stack, %clearoff
	storel 0, %clearptr
	%clearnext =l add %clearidx, 1
	storel %clearnext, %islot
	jmp @clear

@done
	%finalresult =l loadl %retslot
	ret %finalresult
}
