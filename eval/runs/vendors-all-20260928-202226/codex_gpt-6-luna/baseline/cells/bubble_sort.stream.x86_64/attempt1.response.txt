bubble_sort:
    xor eax, eax
    cmp rsi, 1
    jbe .Ldone
    lea rcx, [rsi - 1]
.Louter:
    xor edx, edx
    xor r10d, r10d
.Linner:
    mov r8, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp r8, r9
    jle .Lnext
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], r8
    mov r10d, 1
.Lnext:
    inc rdx
    cmp rdx, rcx
    jb .Linner
    test r10d, r10d
    jz .Ldone
    dec rcx
    jnz .Louter
.Ldone:
    ret
