export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @outer

@outer
    %i =l phi @start 0, @repeat_done %repeat_next, @literal_done %literal_next
    %o =l phi @start 0, @repeat_done %repeat_out, @literal_done %literal_out
    %finished =w csgel %i, %n
    jnz %finished, @return, @measure_start

@measure_start
    %src_i =l add %src, %i
    %first =w loadub %src_i
    jmp @measure

@measure
    %count =l phi @measure_start 1, @measure_continue %count_next
    %measure_pos =l add %i, %count
    %within_n =w csltl %measure_pos, %n
    %within_128 =w csltl %count, 128
    %can_measure =w and %within_n, %within_128
    jnz %can_measure, @measure_load, @measure_done

@measure_load
    %measure_ptr =l add %src, %measure_pos
    %measure_byte =w loadub %measure_ptr
    %same_byte =w ceqw %measure_byte, %first
    jnz %same_byte, @measure_continue, @measure_done

@measure_continue
    %count_next =l add %count, 1
    jmp @measure

@measure_done
    %is_repeat =w csgew %count, 3
    jnz %is_repeat, @repeat_packet, @literal_start

@repeat_packet
    %repeat_needed =l add 2, 0
    %repeat_end =l add %o, %repeat_needed
    %repeat_fits =w cslel %repeat_end, %cap
    jnz %repeat_fits, @repeat_write, @fail

@repeat_write
    %header_l =l sub 257, %count
    %header_w =w copy %header_l
    %header_ptr =l add %out, %o
    storeb %header_w, %header_ptr
    %value_pos =l add %o, 1
    %value_ptr =l add %out, %value_pos
    storeb %first, %value_ptr
    %repeat_next =l add %i, %count
    %repeat_out =l add %o, 2
    jmp @outer

@literal_start
    jmp @literal_scan

@literal_scan
    %j =l phi @literal_start %i, @literal_advance %j_next
    %literal_length =l sub %j, %i
    %length_room =w csltl %literal_length, 128
    %j_in_bounds =w csltl %j, %n
    %scan_allowed =w and %length_room, %j_in_bounds
    jnz %scan_allowed, @literal_check_run, @literal_ready

@literal_check_run
    %j1 =l add %j, 1
    %j2 =l add %j, 2
    %j1_bounds =w csltl %j1, %n
    %j2_bounds =w csltl %j2, %n
    %three_bounds =w and %j1_bounds, %j2_bounds
    jnz %three_bounds, @literal_compare, @literal_advance

@literal_compare
    %ptr0 =l add %src, %j
    %ptr1 =l add %src, %j1
    %ptr2 =l add %src, %j2
    %byte0 =w loadub %ptr0
    %byte1 =w loadub %ptr1
    %byte2 =w loadub %ptr2
    %equal01 =w ceqw %byte0, %byte1
    %equal12 =w ceqw %byte1, %byte2
    %triple_equal =w and %equal01, %equal12
    jnz %triple_equal, @literal_ready, @literal_advance

@literal_advance
    %j_next =l add %j, 1
    jmp @literal_scan

@literal_ready
    %literal_size =l sub %j, %i
    %packet_size =l add %literal_size, 1
    %literal_end =l add %o, %packet_size
    %literal_fits =w cslel %literal_end, %cap
    jnz %literal_fits, @literal_write_header, @fail

@literal_write_header
    %literal_header_l =l sub %literal_size, 1
    %literal_header_w =w copy %literal_header_l
    %literal_header_ptr =l add %out, %o
    storeb %literal_header_w, %literal_header_ptr
    %copy_start =l add %o, 1
    jmp @copy_literal

@copy_literal
    %copy_index =l phi @literal_write_header 0, @copy_continue %copy_next
    %copy_more =w csltl %copy_index, %literal_size
    jnz %copy_more, @copy_byte, @literal_done

@copy_byte
    %src_copy_pos =l add %i, %copy_index
    %src_copy_ptr =l add %src, %src_copy_pos
    %copy_byte_value =w loadub %src_copy_ptr
    %dst_copy_pos =l add %copy_start, %copy_index
    %dst_copy_ptr =l add %out, %dst_copy_pos
    storeb %copy_byte_value, %dst_copy_ptr
    jmp @copy_continue

@copy_continue
    %copy_next =l add %copy_index, 1
    jmp @copy_literal

@literal_done
    %literal_next =l copy %j
    %literal_out =l add %o, %packet_size
    jmp @outer

@fail
    ret -1

@return
    ret %o
}
