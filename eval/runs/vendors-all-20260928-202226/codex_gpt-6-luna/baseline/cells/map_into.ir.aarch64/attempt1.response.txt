export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %idx =l alloc8 8
    %count =l alloc8 8
    %saved =l alloc8 8
    storel 0, %idx
    storel 0, %count
    jmp @loop

@loop
    %i =l loadl %idx
    %more =w csltl %i, %n
    jnz %more, @body, @done

@body
    %offset =l shl %i, 3
    %srcaddr =l add %src, %offset
    %value =l loadl %srcaddr
    storel %value, %saved
    %result =l call %f(l %value)
    %dstaddr =l add %dst, %offset
    storel %result, %dstaddr
    %original =l loadl %saved
    %less =w csltl %result, %original
    jnz %less, @increment_count, @increment_index

@increment_count
    %oldcount =l loadl %count
    %newcount =l add %oldcount, 1
    storel %newcount, %count
    jmp @increment_index

@increment_index
    %oldidx =l loadl %idx
    %newidx =l add %oldidx, 1
    storel %newidx, %idx
    jmp @loop

@done
    %answer =l loadl %count
    ret %answer
}
