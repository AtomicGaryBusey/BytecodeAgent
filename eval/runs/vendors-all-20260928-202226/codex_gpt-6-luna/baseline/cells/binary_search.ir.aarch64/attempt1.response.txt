export function l $binary_search(l %a, l %n, l %key) {
@entry
    jmp @loop
@loop
    %lo =l phi @entry 0, @less %nextlo, @greater %lo
    %hi =l phi @entry %n, @less %hi, @greater %mid
    %done =w ceql %lo, %hi
    jnz %done, @miss, @search
@search
    %diff =l sub %hi, %lo
    %half =l shr %diff, 1
    %mid =l add %lo, %half
    %off =l mul %mid, 8
    %addr =l add %a, %off
    %value =l loadl %addr
    %lesskey =w csltl %value, %key
    jnz %lesskey, @less, @notless
@notless
    %greaterkey =w csltl %key, %value
    jnz %greaterkey, @greater, @found
@less
    %nextlo =l add %mid, 1
    jmp @loop
@greater
    jmp @loop
@found
    ret %mid
@miss
    %minusone =l sub 0, 1
    ret %minusone
}
