export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%ip =l alloc8 8
	%minp =l alloc8 8
	%maxp =l alloc8 8
	%sump =l alloc8 8
	%countp =l alloc8 8
	storel 0, %ip
	storel 0, %minp
	storel 0, %maxp
	storel 0, %sump
	storel 0, %countp
	jmp @loop
@loop
	%i =l loadl %ip
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%ap =l mul %i, 8
	%addr =l add %a, %ap
	%x =l loadl %addr
	%y =l call %f(l %x)
	%zero =w ceql %i, 0
	jnz %zero, @first, @update
@first
	storel %y, %minp
	storel %y, %maxp
	jmp @accumulate
@update
	%oldmin =l loadl %minp
	%ismin =w csltl %y, %oldmin
	jnz %ismin, @setmin, @checkmax
@setmin
	storel %y, %minp
	jmp @checkmax
@checkmax
	%oldmax =l loadl %maxp
	%ismax =w csgtl %y, %oldmax
	jnz %ismax, @setmax, @accumulate
@setmax
	storel %y, %maxp
	jmp @accumulate
@accumulate
	%sum =l loadl %sump
	%newsum =l add %sum, %y
	storel %newsum, %sump
	%odd =l and %y, 1
	%count =l loadl %countp
	%newcount =l add %count, %odd
	storel %newcount, %countp
	%next =l add %i, 1
	storel %next, %ip
	jmp @loop
@done
	%min =l loadl %minp
	%max =l loadl %maxp
	%sum =l loadl %sump
	%count =l loadl %countp
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
