export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%zero =l copy 0
	%three =l copy 3
	%four =l copy 4
	%groups =l add %n, 2
	%groups =l div %groups, %three
	%length =l mul %groups, %four
	%offsetslot =l alloc8 8
	%availSlot =l alloc8 8
	%packedSlot =l alloc8 8
	%validSlot =l alloc8 8
	%stageslot =l alloc8 8
	%idxslot =l alloc8 8
	%outidxslot =l alloc8 8
	%chslot =l alloc8 8
	storel %zero, %offsetslot
	jmp @group

@group
	%offset =l loadl %offsetslot
	%more =w csltl %offset, %n
	jnz %more, @readfirst, @done

@readfirst
	%sp =l add %src, %offset
	%a =l loadub %sp
	%avail =l sub %n, %offset
	storel %avail, %availSlot
	%hasb =w csltl 1, %avail
	jnz %hasb, @readsecond, @nob

@readsecond
	%sp1 =l add %sp, 1
	%b =l loadub %sp1
	jmp @aftersecond

@nob
	%b =l copy 0
	jmp @aftersecond

@aftersecond
	%hasc =w csltl 2, %avail
	jnz %hasc, @readthird, @noc

@readthird
	%sp2 =l add %sp, 2
	%c =l loadub %sp2
	jmp @afterthird

@noc
	%c =l copy 0
	jmp @afterthird

@afterthird
	%a16 =l shl %a, 16
	%b8 =l shl %b, 8
	%ab =l or %a16, %b8
	%packed =l or %ab, %c
	storel %packed, %packedSlot
	%valid2 =w ceql %avail, 1
	jnz %valid2, @twochars, @checktwo

@twochars
	storel 2, %validSlot
	jmp @startmap

@checktwo
	%valid3 =w ceql %avail, 2
	jnz %valid3, @threechars, @fourchars

@threechars
	storel 3, %validSlot
	jmp @startmap

@fourchars
	storel 4, %validSlot
	jmp @startmap

@startmap
	storel %zero, %stageslot
	jmp @map

@map
	%stage =l loadl %stageslot
	%valid =l loadl %validSlot
	%notpad =w csltl %stage, %valid
	jnz %notpad, @mapchar, @padding

@padding
	%ch =l copy 61
	jmp @emit

@mapchar
	%packedv =l loadl %packedSlot
	%twostage =l mul %stage, 2
	%shift =l sub 18, %twostage
	%vshift =l shr %packedv, %shift
	%idx =l and %vshift, 63
	storel %idx, %idxslot
	%lt26 =w csltl %idx, 26
	jnz %lt26, @upper, @test52

@upper
	%ch =l add %idx, 65
	jmp @emit

@test52
	%lt52 =w csltl %idx, 52
	jnz %lt52, @lower, @test62

@lower
	%u =l sub %idx, 26
	%ch =l add %u, 97
	jmp @emit

@test62
	%lt62 =w csltl %idx, 62
	jnz %lt62, @digit, @special

@digit
	%u2 =l sub %idx, 52
	%ch =l add %u2, 48
	jmp @emit

@special
	%is62 =w ceql %idx, 62
	jnz %is62, @plus, @slash

@plus
	%ch =l copy 43
	jmp @emit

@slash
	%ch =l copy 47
	jmp @emit

@emit
	%offset =l loadl %offsetslot
	%groupnum =l div %offset, %three
	%baseout =l mul %groupnum, %four
	%stage =l loadl %stageslot
	%outidx =l add %baseout, %stage
	%dst =l add %out, %outidx
	storeb %ch, %dst
	%nextstage =l add %stage, 1
	storel %nextstage, %stageslot
	%finished =w ceql %nextstage, 4
	jnz %finished, @nextgroup, @map

@nextgroup
	%offset =l loadl %offsetslot
	%nextoffset =l add %offset, 3
	storel %nextoffset, %offsetslot
	jmp @group

@done
	ret %length
}
