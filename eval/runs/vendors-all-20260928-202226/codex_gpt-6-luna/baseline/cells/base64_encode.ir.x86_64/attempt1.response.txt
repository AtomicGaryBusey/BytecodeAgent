export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%zero =l copy 0
	%four =l copy 4
	%three =l copy 3
	%groups =l add %n, 2
	%groups =l div %groups, %three
	%len =l mul %groups, %four
	%rem =l rem %n, %three
	%srcslot =l alloc8 8
	%outslot =l alloc8 8
	%groupslot =l alloc8 8
	%remslot =l alloc8 8
	%packedslot =l alloc8 8
	%stageslot =l alloc8 8
	%idxslot =l alloc8 8
	%chslot =l alloc8 8
	storel %src, %srcslot
	storel %out, %outslot
	storel %groups, %groupslot
	storel %rem, %remslot
	storel %zero, %packedslot
	storel %zero, %stageslot
	storel %zero, %idxslot
	jnz %groups, @group, @done
@group
	%sp =l loadl %srcslot
	%a =l loadub %sp
	%sp1 =l add %sp, 1
	%bp =l loadl %remslot
	%hasb =w cnel %bp, 1
	jnz %hasb, @readb, @nob
@readb
	%b =l loadub %sp1
	jmp @bready
@nob
	%b =l copy 0
	jmp @bready
@bready
	%sp2 =l add %sp, 2
	%hasc =w ceql %bp, 0
	jnz %hasc, @readc, @noc
@readc
	%c =l loadub %sp2
	jmp @cready
@noc
	%c =l copy 0
	jmp @cready
@cready
	%a16 =l shl %a, 16
	%b8 =l shl %b, 8
	%ab =l or %a16, %b8
	%packed =l or %ab, %c
	storel %packed, %packedslot
	storel %zero, %stageslot
	jmp @map
@map
	%packedv =l loadl %packedslot
	%stage =l loadl %stageslot
	%shift =l sub 18, %stage
	%shift =l mul %shift, 1
	%shiftbits =l shl %stage, 1
	%shiftbits =l sub 18, %shiftbits
	%shiftbits =l and %shiftbits, 63
	%vshift =l shr %packedv, %shiftbits
	%idx =l and %vshift, 63
	storel %idx, %idxslot
	%remv =l loadl %remslot
	%islast =w ceql %remv, 0
	%isstage =w ceql %stage, 2
	%pad2 =w and %islast, %isstage
	%isone =w ceql %remv, 1
	%isstage3 =w ceql %stage, 3
	%pad3a =w and %isone, %isstage
	%pad3b =w and %isone, %isstage3
	%pad =w or %pad2, %pad3a
	%pad =w or %pad, %pad3b
	jnz %pad, @padding, @mapchar
@padding
	%eqchar =l copy 61
	storel %eqchar, %chslot
	jmp @emit
@mapchar
	%v =l loadl %idxslot
	%lo26 =w csltl %v, 26
	jnz %lo26, @upper, @test52
@upper
	%ch =l add %v, 65
	storel %ch, %chslot
	jmp @emit
@test52
	%lt52 =w csltl %v, 52
	jnz %lt52, @lower, @test62
@lower
	%u =l sub %v, 26
	%ch =l add %u, 97
	storel %ch, %chslot
	jmp @emit
@test62
	%lt62 =w csltl %v, 62
	jnz %lt62, @digit, @test63
@digit
	%u2 =l sub %v, 52
	%ch2 =l add %u2, 48
	storel %ch2, %chslot
	jmp @emit
@test63
	%eq62 =w ceql %v, 62
	jnz %eq62, @plus, @slash
@plus
	%pluschar =l copy 43
	storel %pluschar, %chslot
	jmp @emit
@slash
	%slashchar =l copy 47
	storel %slashchar, %chslot
	jmp @emit
@emit
	%op =l loadl %outslot
	%stagev =l loadl %stageslot
	%dst =l add %op, %stagev
	%char =l loadl %chslot
	storeb %char, %dst
	%nextstage =l add %stagev, 1
	storel %nextstage, %stageslot
	%finished =w ceql %nextstage, 4
	jnz %finished, @nextgroup, @map
@nextgroup
	%spnow =l loadl %srcslot
	%remnow =l loadl %remslot
	%advance =l copy 3
	%partial =w cnel %remnow, 0
	jnz %partial, @advancepartial, @advancefull
@advancepartial
	%advancepartialv =l copy 0
	%remaininggroups =l loadl %groupslot
	%remaininggroups =l sub %remaininggroups, 1
	storel %remaininggroups, %groupslot
	storel %zero, %remslot
	jmp @advance_done
@advancefull
	%spnext =l add %spnow, %advance
	storel %spnext, %srcslot
	%remaininggroups2 =l loadl %groupslot
	%remaininggroups2 =l sub %remaininggroups2, 1
	storel %remaininggroups2, %groupslot
	jmp @advance_done
@advance_done
	%left =l loadl %groupslot
	jnz %left, @group, @done
@done
	ret %len
}
