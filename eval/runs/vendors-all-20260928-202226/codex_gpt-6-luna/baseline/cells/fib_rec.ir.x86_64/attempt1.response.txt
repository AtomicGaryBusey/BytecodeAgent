export function l $fib_rec(l %n) {
@entry
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @checkone
@zero
	ret 0
@checkone
	%isone =w ceql %n, 1
	jnz %isone, @one, @recurse
@one
	ret 1
@recurse
	%nm1 =l sub %n, 1
	%a =l call $fib_rec(l %nm1)
	%nm2 =l sub %n, 2
	%b =l call $fib_rec(l %nm2)
	%sum =l add %a, %b
	ret %sum
}
