bubble_sort:
cmp x1, #1
b.le .Ldone
mov x2, x1
.Lpass:
mov x3, #0
.Linner:
add x7, x3, #1
cmp x7, x2
b.hs .Lpass_done
ldr x4, [x0, x3, lsl #3]
ldr x5, [x0, x7, lsl #3]
cmp x4, x5
b.le .Lnext
str x5, [x0, x3, lsl #3]
str x4, [x0, x7, lsl #3]
.Lnext:
mov x3, x7
b .Linner
.Lpass_done:
sub x2, x2, #1
cmp x2, #1
b.hi .Lpass
.Ldone:
mov x0, #0
ret
