export function l $qsort_pairs(l %a, l %n) {
@start
    %lo =l alloc8
    %hi =l alloc8
    %i =l alloc8
    %j =l alloc8
    %pk =l alloc8
    %pv =l alloc8
    %split =l alloc8
    %len =l alloc8
    storel 0, %lo
    storel %n, %hi
    %small =w cslel %n, 1
    jnz %small, @done, @partition

@partition
    %midbase =l shr %n, 1
    %midbytes =l shl %midbase, 4
    %midptr =l add %a, %midbytes
    %pivotk =l loadl %midptr
    %pivotvp =l add %midptr, 8
    %pivotv =l loadl %pivotvp
    storel %pivotk, %pk
    storel %pivotv, %pv
    storel 0, %i
    %last =l sub %n, 1
    storel %last, %j
    jmp @scanleft

@scanleft
    %iv =l loadl %i
    %hv =l loadl %hi
    %ib =l shl %iv, 4
    %ip =l add %a, %ib
    %ik =l loadl %ip
    %pk1 =l loadl %pk
    %ikless =w csltl %ik, %pk1
    jnz %ikless, @incleft, @leftkeyeq

@leftkeyeq
    %ikeq =w ceql %ik, %pk1
    jnz %ikeq, @leftvalue, @leftstop

@leftvalue
    %ivp =l add %ip, 8
    %ival =l loadl %ivp
    %pv1 =l loadl %pv
    %ivless =w csltl %ival, %pv1
    jnz %ivless, @incleft, @leftstop

@incleft
    %inext =l add %iv, 1
    storel %inext, %i
    jmp @scanleft

@leftstop
    jmp @scanright

@scanright
    %jv =l loadl %j
    %jb =l shl %jv, 4
    %jp =l add %a, %jb
    %jk =l loadl %jp
    %pk2 =l loadl %pk
    %jkgreater =w csgtl %jk, %pk2
    jnz %jkgreater, @decright, @rightkeyeq

@rightkeyeq
    %jkeq =w ceql %jk, %pk2
    jnz %jkeq, @rightvalue, @rightstop

@rightvalue
    %jvp =l add %jp, 8
    %jval =l loadl %jvp
    %pv2 =l loadl %pv
    %jgreater =w csgtl %jval, %pv2
    jnz %jgreater, @decright, @rightstop

@decright
    %jprev =l sub %jv, 1
    storel %jprev, %j
    jmp @scanright

@rightstop
    %iv2 =l loadl %i
    %jv2 =l loadl %j
    %crossed =w csge l %iv2, %jv2
    jnz %crossed, @partitiondone, @swap

@swap
    %ib2 =l shl %iv2, 4
    %ip2 =l add %a, %ib2
    %jb2 =l shl %jv2, 4
    %jp2 =l add %a, %jb2
    %aki =l loadl %ip2
    %avp =l add %ip2, 8
    %avi =l loadl %avp
    %akj =l loadl %jp2
    %bvp =l add %jp2, 8
    %bvj =l loadl %bvp
    storel %akj, %ip2
    storel %bvj, %avp
    storel %aki, %jp2
    storel %avi, %bvp
    %inext2 =l add %iv2, 1
    %jprev2 =l sub %jv2, 1
    storel %inext2, %i
    storel %jprev2, %j
    jmp @scanleft

@partitiondone
    %cut =l loadl %iv2
    storel %cut, %split
    %leftn =l loadl %split
    %leftsmall =w cslel %leftn, 1
    jnz %leftsmall, @rightcall, @leftcall

@leftcall
    %rleft =l call $qsort_pairs(l %a, l %leftn)
    jmp @rightcall

@rightcall
    %cut2 =l loadl %split
    %rightn =l sub %n, %cut2
    %rightsmall =w cslel %rightn, 1
    jnz %rightsmall, @done, @callright

@callright
    %off =l shl %cut2, 4
    %rightp =l add %a, %off
    %rright =l call $qsort_pairs(l %rightp, l %rightn)
    jmp @done

@done
    ret 0
}
