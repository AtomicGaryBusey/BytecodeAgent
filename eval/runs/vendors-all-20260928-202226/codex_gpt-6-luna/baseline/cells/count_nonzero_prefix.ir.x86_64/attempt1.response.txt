export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	jmp @loop
@loop
	%i = phi @entry 0, @step %next
	%in_bounds = csltl %i, %n
	jnz %in_bounds, @check, @done
@check
	%offset = mull %i, 8
	%addr = addl %a, %offset
	%value = loadl %addr
	%is_zero = ceql %value, 0
	jnz %is_zero, @done, @step
@step
	%next = addl %i, 1
	jmp @loop
@done
	ret %i
}
