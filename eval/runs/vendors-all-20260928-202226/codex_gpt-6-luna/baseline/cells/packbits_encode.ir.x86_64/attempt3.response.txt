export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %islot =l alloc8 8
    %kslot =l alloc8 8
    %jslot =l alloc8 8
    %qslot =l alloc8 8
    %oslot =l alloc8 8
    %fslot =l alloc8 8
    %bslot =l alloc8 8
    storel 0, %islot
    storel 0, %oslot
    jmp @outer

@outer
    %i =l loadl %islot
    %done =w csgel %i, %n
    jnz %done, @finish, @runstart

@runstart
    %ip =l add %src, %i
    %first =w loadub %ip
    %firstl =l copy %first
    storel %firstl, %fslot
    storel %i, %kslot
    jmp @runscan

@runscan
    %k =l loadl %kslot
    %diff =l sub %k, %i
    %atend =w csgel %k, %n
    %full =w csgel %diff, 128
    %stop =w or %atend, %full
    jnz %stop, @rundone, @runcheck

@runcheck
    %kp =l add %src, %k
    %cur =w loadub %kp
    %f =l loadl %fslot
    %fw =w copy %f
    %same =w ceqw %cur, %fw
    jnz %same, @runadvance, @rundone

@runadvance
    %knext =l add %k, 1
    storel %knext, %kslot
    jmp @runscan

@rundone
    %kdone =l loadl %kslot
    %r =l sub %kdone, %i
    %isrun =w csge l %r, 3
    jnz %isrun, @repeat, @literalstart

@repeat
    %opos =l loadl %oslot
    %outend =l add %opos, 2
    %no_room =w csltl %cap, %outend
    jnz %no_room, @fail, @repeatwrite

@repeatwrite
    %oposr =l loadl %oslot
    %hval =l sub 257, %r
    %header =w copy %hval
    %hp =l add %out, %oposr
    storeb %header, %hp
    %bp =l add %hp, 1
    %fval =l loadl %fslot
    %fbyte =w copy %fval
    storeb %fbyte, %bp
    %opos2 =l add %oposr, 2
    storel %opos2, %oslot
    storel %kdone, %islot
    jmp @outer

@literalstart
    storel %i, %jslot
    jmp @litloop

@litloop
    %j =l loadl %jslot
    %length =l sub %j, %i
    %litfull =w csgel %length, 128
    %litend =w csgel %j, %n
    %litstop =w or %litfull, %litend
    jnz %litstop, @litdone, @probe

@probe
    %jp =l add %src, %j
    %jbyte =w loadub %jp
    %jbytel =l copy %jbyte
    storel %jbytel, %bslot
    %q =l add %j, 1
    storel %q, %qslot
    jmp @probe_scan

@probe_scan
    %qscan =l loadl %qslot
    %atend2 =w csgel %qscan, %n
    %count =l sub %qscan, %j
    %three =w csge l %count, 3
    %halt =w or %atend2, %three
    jnz %halt, @probe_done, @probe_check

@probe_check
    %qp =l add %src, %qscan
    %qbyte =w loadub %qp
    %byte =l loadl %bslot
    %bytew =w copy %byte
    %equal =w ceqw %qbyte, %bytew
    jnz %equal, @probe_inc, @probe_done

@probe_inc
    %qnext =l add %qscan, 1
    storel %qnext, %qslot
    jmp @probe_scan

@probe_done
    %countdone =l sub %qscan, %j
    %isrun2 =w csge l %countdone, 3
    jnz %isrun2, @litdone, @litadvance

@litadvance
    %jnext =l add %j, 1
    storel %jnext, %jslot
    jmp @litloop

@litdone
    %jdone =l loadl %jslot
    %llen =l sub %jdone, %i
    %litneed =l add %llen, 1
    %oposl =l loadl %oslot
    %litendout =l add %oposl, %litneed
    %lit_no_room =w csltl %cap, %litendout
    jnz %lit_no_room, @fail, @litwrite

@litwrite
    %lhval =l sub %llen, 1
    %lh =w copy %lhval
    %lp =l add %out, %oposl
    storeb %lh, %lp
    storel %i, %kslot
    jmp @copy

@copy
    %c =l loadl %kslot
    %copydone =w csgel %c, %jdone
    jnz %copydone, @copydone, @copybyte

@copybyte
    %sp =l add %src, %c
    %v =w loadub %sp
    %offset =l sub %c, %i
    %dp0 =l add %lp, 1
    %dp =l add %dp0, %offset
    storeb %v, %dp
    %cnext =l add %c, 1
    storel %cnext, %kslot
    jmp @copy

@copydone
    storel %litendout, %oslot
    storel %jdone, %islot
    jmp @outer

@fail
    ret -1

@finish
    %result =l loadl %oslot
    ret %result
}
