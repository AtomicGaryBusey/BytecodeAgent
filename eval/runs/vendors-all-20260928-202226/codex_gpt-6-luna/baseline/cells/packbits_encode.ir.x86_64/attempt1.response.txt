export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @outer(0)

@outer(l %i)
    %done =w csge l %i, %n
    jnz %done, @finish, @runstart

@runstart
    %ip =l add %src, %i
    %first =w loadub %ip
    jmp @runscan(%i, %i, %first)

@runscan(l %i, l %k, w %first)
    %diff =l sub %k, %i
    %atend =w csge l %k, %n
    %full =w csge l %diff, 128
    %stop =w or %atend, %full
    jnz %stop, @rundone(%i, %k), @runcheck

@runcheck
    %kp =l add %src, %k
    %cur =w loadub %kp
    %same =w ceq %cur, %first
    jnz %same, @runadvance, @rundone(%i, %k)

@runadvance
    %knext =l add %k, 1
    jmp @runscan(%i, %knext, %first)

@rundone(l %i, l %k)
    %r =l sub %k, %i
    %isrun =w csge l %r, 3
    jnz %isrun, @repeat, @literalstart

@repeat
    %need =l add %cap, 0
    %outend =l add %opos, 2
    %no_room =w csltl %cap, %outend
    jnz %no_room, @fail, @repeatwrite

@repeatwrite
    %header =l sub 257, %r
    %hp =l add %out, %opos
    storeb %header, %hp
    %bp =l add %hp, 1
    storeb %first, %bp
    %opos2 =l add %opos, 2
    jmp @outer2(%k, %opos2)

@outer2(l %i, l %opos)
    jmp @outer(%i)

@literalstart
    jmp @litloop(%i, %i)

@litloop(l %i, l %j)
    %length =l sub %j, %i
    %litfull =w csge l %length, 128
    %litend =w csge l %j, %n
    %litstop =w or %litfull, %litend
    jnz %litstop, @litdone(%i, %j), @probe

@probe
    %jp =l add %src, %j
    %jbyte =w loadub %jp
    %jnext =l add %j, 1
    jmp @probe_scan(%i, %j, %jnext, %jbyte, 0)

@probe_scan(l %i, l %j, l %q, w %byte, l %count)
    %atend2 =w csge l %q, %n
    %three =w csge l %count, 3
    %halt =w or %atend2, %three
    jnz %halt, @probe_done(%i, %j, %count), @probe_check

@probe_check
    %qp =l add %src, %q
    %qbyte =w loadub %qp
    %equal =w ceq %qbyte, %byte
    jnz %equal, @probe_inc, @probe_done(%i, %j, %count)

@probe_inc
    %count2 =l add %count, 1
    %q2 =l add %q, 1
    jmp @probe_scan(%i, %j, %q2, %byte, %count2)

@probe_done(l %i, l %j, l %count)
    %isrun2 =w csge l %count, 3
    jnz %isrun2, @litdone(%i, %j), @litadvance

@litadvance
    %j2 =l add %j, 1
    jmp @litloop(%i, %j2)

@litdone(l %i, l %j)
    %llen =l sub %j, %i
    %litneed =l add %llen, 1
    %litendout =l add %opos, %litneed
    %lit_no_room =w csltl %cap, %litendout
    jnz %lit_no_room, @fail, @litwrite

@litwrite
    %lh =l sub %llen, 1
    %lp =l add %out, %opos
    storeb %lh, %lp
    jmp @copy(%i, %j, %i, %lp)

@copy(l %i, l %j, l %c, l %lp)
    %copydone =w csge l %c, %j
    jnz %copydone, @copydone(%j), @copybyte

@copybyte
    %sp =l add %src, %c
    %v =w loadub %sp
    %offset =l sub %c, %i
    %dp0 =l add %lp, 1
    %dp =l add %dp0, %offset
    storeb %v, %dp
    %cnext =l add %c, 1
    jmp @copy(%i, %j, %cnext, %lp)

@copydone(l %j)
    %newopos =l add %opos, %litneed
    jmp @outer2(%j, %newopos)

@fail
    ret -1

@finish
    ret %opos
}
