export function l $crc32(l %p, l %n) {
@start
	%crc =l alloc4 4
	%i =l alloc4 4
	%j =l alloc4 4
	storew -1, %crc
	storel 0, %i
	jmp @outer

@outer
	%oi =l loadl %i
	%more =w csltl %oi, %n
	jnz %more, @byte, @done

@byte
	%bp =l add %p, %oi
	%b =w loadub %bp
	%c0 =w loadw %crc
	%c1 =w xor %c0, %b
	storew %c1, %crc
	storel 0, %j
	jmp @inner

@inner
	%ji =l loadl %j
	%bits =w csltl %ji, 8
	jnz %bits, @bit, @nextbyte

@bit
	%c2 =w loadw %crc
	%low =w and %c2, 1
	%odd =w cnel %low, 0
	jnz %odd, @one, @zero

@one
	%c3 =w shr %c2, 1
	%c4 =w xor %c3, -306674912
	storew %c4, %crc
	jmp @step

@zero
	%c5 =w shr %c2, 1
	storew %c5, %crc
	jmp @step

@step
	%jv =l loadl %j
	%jn =l add %jv, 1
	storel %jn, %j
	jmp @inner

@nextbyte
	%iv =l loadl %i
	%in =l add %iv, 1
	storel %in, %i
	jmp @outer

@done
	%cf =w loadw %crc
	%result =w xor %cf, -1
	%wide =l extuw %result
	ret %wide
}
