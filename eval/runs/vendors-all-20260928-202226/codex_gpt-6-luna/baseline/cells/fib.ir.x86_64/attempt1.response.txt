export function l $fib(l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %i.next
    %a =l phi @start 0, @body %b
    %b =l phi @start 1, @body %sum
    %done =w csltl %i, %n
    jnz %done, @body, @end
@body
    %sum =l add %a, %b
    %i.next =l add %i, 1
    jmp @loop
@end
    ret %a
}
