export function l $max_byte(l %p, l %n) {
@start
    %slot =l alloc8 8
    %idx =l alloc8 8
    storel 0, %slot
    storel 0, %idx
    jmp @loop
@loop
    %i =l loadl %idx
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %max =l loadl %slot
    %greater =w csgtl %b, %max
    jnz %greater, @update, @next
@update
    storel %b, %slot
    jmp @next
@next
    %cur =l loadl %idx
    %inc =l add %cur, 1
    storel %inc, %idx
    jmp @loop
@done
    %result =l loadl %slot
    ret %result
}
