export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @step %inext
	%p =l phi @start %a, @step %pnext
	%more =w csltl %i, %n
	jnz %more, @check, @done
@check
	%x =l loadl %p
	%nz =w cnel %x, 0
	jnz %nz, @step, @found
@step
	%inext =l add %i, 1
	%pnext =l add %p, 8
	jmp @loop
@found
	ret %i
@done
	ret %i
}
