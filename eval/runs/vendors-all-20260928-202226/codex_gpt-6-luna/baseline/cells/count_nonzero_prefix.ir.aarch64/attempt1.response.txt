export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =phi @start 0, @step %inext
	%p =phi @start %a, @step %pnext
	%more =csltl %i, %n
	jnz %more, @check, @done
@check
	%x =loadl %p
	%nz =cnel %x, 0
	jnz %nz, @step, @found
@step
	%inext =add %i, 1
	%pnext =add %p, 8
	jmp @loop
@found
	ret %i
@done
	ret %i
}
