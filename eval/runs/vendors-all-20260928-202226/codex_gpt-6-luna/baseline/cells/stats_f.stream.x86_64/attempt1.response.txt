stats_f:
push rbx
push rbp
push r12
push r13
push r14
sub rsp, 32
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov rbx, rcx
mov rbp, 0x8000000000000000
mov qword ptr [rsp], 0x7fffffffffffffff
mov qword ptr [rsp + 8], 0
mov qword ptr [rsp + 16], 0
mov qword ptr [rsp + 24], 0
.Lloop:
mov rcx, qword ptr [rsp + 24]
cmp rcx, r13
jge .Ldone
mov rdi, qword ptr [r12 + rcx*8]
call r14
cmp rax, rbp
jge .Lmin_done
mov rbp, rax
.Lmin_done:
cmp rax, qword ptr [rsp]
jle .Lmax_done
mov qword ptr [rsp], rax
.Lmax_done:
add qword ptr [rsp + 8], rax
test al, 1
jz .Leven
add qword ptr [rsp + 16], 1
.Leven:
add qword ptr [rsp + 24], 1
jmp .Lloop
.Ldone:
mov qword ptr [rbx], rbp
mov rax, qword ptr [rsp]
mov qword ptr [rbx + 8], rax
mov rax, qword ptr [rsp + 8]
mov qword ptr [rbx + 16], rax
mov rax, qword ptr [rsp + 16]
add rsp, 32
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
