mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbx, r8
mov rbp, r13
imul rbp, r13
xor r8d, r8d
.Lzero_out:
cmp r8, rbp
jae .Lidentity
mov qword ptr [r15 + r8*8], 0
inc r8
jmp .Lzero_out
.Lidentity:
xor r8d, r8d
.Ldiag:
cmp r8, r13
jae .Lpower_loop
mov qword ptr [r15 + r8*8], 1
lea rax, [r13 + 1]
add r8, rax
jmp .Ldiag
.Lpower_loop:
test r14, r14
jz .Lclear_tmp
xor r8d, r8d
.Lrow_loop:
cmp r8, r13
jae .Lcopy_product
xor r9d, r9d
.Lcol_loop:
cmp r9, r13
jae .Lnext_row
xor r11d, r11d
xor r10d, r10d
.Linner_loop:
cmp r10, r13
jae .Lstore_product
mov rax, r8
imul rax, r13
add rax, r10
mov rdx, r10
imul rdx, r13
add rdx, r9
mov rax, qword ptr [r12 + rax*8]
mov rcx, qword ptr [r15 + rdx*8]
imul rax, rcx
add r11, rax
inc r10
jmp .Linner_loop
.Lstore_product:
mov rax, r8
imul rax, r13
add rax, r9
mov qword ptr [rbx + rax*8], r11
inc r9
jmp .Lcol_loop
.Lnext_row:
inc r8
jmp .Lrow_loop
.Lcopy_product:
xor r8d, r8d
.Lcopy_loop:
cmp r8, rbp
jae .Ldecrement
mov rax, qword ptr [rbx + r8*8]
mov qword ptr [r15 + r8*8], rax
inc r8
jmp .Lcopy_loop
.Ldecrement:
dec r14
jmp .Lpower_loop
.Lclear_tmp:
lea rbp, [rbp*2]
xor r8d, r8d
.Lclear_loop:
cmp r8, rbp
jae .Lreturn
mov qword ptr [rbx + r8*8], 0
inc r8
jmp .Lclear_loop
.Lreturn:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
