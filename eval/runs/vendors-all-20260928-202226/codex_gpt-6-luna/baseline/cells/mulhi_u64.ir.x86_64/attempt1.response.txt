export function l $mulhi_u64(l %a, l %b) {
@start
    %mask =w copy 4294967295
    %a0 =l and %a, 4294967295
    %a1 =l shr %a, 32
    %b0 =l and %b, 4294967295
    %b1 =l shr %b, 32
    %p00 =l mul %a0, %b0
    %p01 =l mul %a0, %b1
    %p10 =l mul %a1, %b0
    %p11 =l mul %a1, %b1
    %p00h =l shr %p00, 32
    %p01l =l and %p01, 4294967295
    %p10l =l and %p10, 4294967295
    %m0 =l add %p00h, %p01l
    %m1 =l add %m0, %p10l
    %mhi =l shr %m1, 32
    %p01h =l shr %p01, 32
    %p10h =l shr %p10, 32
    %h0 =l add %p11, %p01h
    %h1 =l add %h0, %p10h
    %h2 =l add %h1, %mhi
    ret %h2
}
