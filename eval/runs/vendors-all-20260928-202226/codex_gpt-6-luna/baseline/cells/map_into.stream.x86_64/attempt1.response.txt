push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov rbx, rcx
xor ebp, ebp
xor r15d, r15d
.Lloop:
cmp rbp, r14
jge .Ldone
mov rdi, qword ptr [r13 + rbp*8]
mov qword ptr [rsp], rdi
call rbx
mov qword ptr [r12 + rbp*8], rax
cmp rax, qword ptr [rsp]
jge .Lnext
inc r15

inc rbp
jmp .Lloop

add rsp, 8

pop r14
pop r13
pop r12
pop rbp
pop rbx
