export function l $bmax(l %a, l %n) {
@start
	%bestp =l alloc8 8
	%idxp =l alloc8 8
	%first =l loadl %a
	storel %first, %bestp
	%one =l copy 1
	storel %one, %idxp
	jmp @loop
@loop
	%idx =l loadl %idxp
	%more =w csltl %idx, %n
	jnz %more, @body, @done
@body
	%ptr =l add %a, %idx
	%val =l loadl %ptr
	%best =l loadl %bestp
	%greater =w csgtl %val, %best
	jnz %greater, @update, @next
@update
	storel %val, %bestp
	jmp @next
@next
	%oldidx =l loadl %idxp
	%newidx =l add %oldidx, %one
	storel %newidx, %idxp
	jmp @loop
@done
	%result =l loadl %bestp
	ret %result
}
