export function l $bmax(l %a, l %n) {
@start
	%first = loadl %a
	%one =l copy 1
	%idx =l copy %one
	jmp @loop
@loop
	%more =w csltl %idx, %n
	jnz %more, @body, @done
@body
	%ptr =l add %a, %idx
	%val =l loadl %ptr
	%greater =w csgtl %val, %first
	jnz %greater, @update, @next
@update
	%first =l copy %val
	jmp @next
@next
	%idx =l add %idx, %one
	jmp @loop
@done
	ret %first
}
