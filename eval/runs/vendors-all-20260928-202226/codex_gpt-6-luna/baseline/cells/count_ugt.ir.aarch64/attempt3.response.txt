export function l $count_ugt(l %a, l %n, l %t) {
@entry
	jmp @loop

@loop
	%p =l phi @entry %a, @body %nextp
	%remaining =l phi @entry %n, @body %nextremaining
	%count =l phi @entry 0, @body %nextcount
	%more =w cnel %remaining, 0
	jnz %more, @body, @done

@body
	%value =l loadl %p
	%greater =w cugt %value, %t
	%increment =l extuw %greater
	%nextcount =l add %count, %increment
	%nextp =l add %p, 8
	%nextremaining =l sub %remaining, 1
	jmp @loop

@done
	ret %count
}
