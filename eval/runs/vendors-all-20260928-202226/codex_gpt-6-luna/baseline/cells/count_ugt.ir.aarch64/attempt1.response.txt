export function l $count_ugt(l %a, l %n, l %t) {
@entry
	jmp @loop(%a, %n, 0)

@loop(%p =l, %remaining =l, %count =l)
	%more =w cne %remaining, 0
	jnz %more, @body, @done

@body
	%value =l loadl %p
	%greater =w cugt %value, %t
	%increment =l extuw %greater
	%nextcount =l add %count, %increment
	%nextp =l add %p, 8
	%nextremaining =l sub %remaining, 1
	jmp @loop(%nextp, %nextremaining, %nextcount)

@done
	ret %count
}
