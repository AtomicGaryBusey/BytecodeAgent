qsort_pairs:
    test rsi, rsi
    jle .Lbase
    cmp rsi, 1
    jle .Lbase
    push rbx
    push rbp
    push r12
    push r13
    push r14
    sub rsp, 32
    mov rbx, rdi
    mov rbp, rsi
    mov rax, rbp
    dec rax
    shl rax, 4
    mov r12, qword ptr [rbx + rax]
    mov r13, qword ptr [rbx + rax + 8]
    xor r14d, r14d
    xor ecx, ecx
.Lpart:
    lea rdx, [rcx + 1]
    cmp rdx, rbp
    jge .Lpartition_done
    mov rdx, rcx
    shl rdx, 4
    mov r8, qword ptr [rbx + rdx]
    cmp r8, r12
    jl .Lswap_current
    jg .Lnext
    mov r8, qword ptr [rbx + rdx + 8]
    cmp r8, r13
    jg .Lnext
.Lswap_current:
    mov rdx, r14
    shl rdx, 4
    mov r8, qword ptr [rbx + rdx]
    mov r9, qword ptr [rbx + rdx + 8]
    mov r10, qword ptr [rbx + rcx*8]
    mov r11, qword ptr [rbx + rcx*8 + 8]
    mov qword ptr [rbx + rdx], r10
    mov qword ptr [rbx + rdx + 8], r11
    mov qword ptr [rbx + rcx*8], r8
    mov qword ptr [rbx + rcx*8 + 8], r9
    inc r14
.Lnext:
    inc rcx
    jmp .Lpart
.Lpartition_done:
    mov rax, rbp
    dec rax
    shl rax, 4
    mov rdx, r14
    shl rdx, 4
    mov r8, qword ptr [rbx + rdx]
    mov r9, qword ptr [rbx + rdx + 8]
    mov r10, qword ptr [rbx + rax]
    mov r11, qword ptr [rbx + rax + 8]
    mov qword ptr [rbx + rdx], r10
    mov qword ptr [rbx + rdx + 8], r11
    mov qword ptr [rbx + rax], r8
    mov qword ptr [rbx + rax + 8], r9
    mov qword ptr [rsp], r14
    mov rax, rbp
    sub rax, r14
    dec rax
    mov qword ptr [rsp + 8], rax
    test r14, r14
    jz .Lright
    mov rdi, rbx
    mov rsi, r14
    call qsort_pairs
.Lright:
    mov rsi, qword ptr [rsp + 8]
    test rsi, rsi
    jz .Ldone
    mov rdi, qword ptr [rsp]
    inc rdi
    shl rdi, 4
    add rdi, rbx
    call qsort_pairs
.Ldone:
    add rsp, 32
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Lbase:
    xor eax, eax
    ret
