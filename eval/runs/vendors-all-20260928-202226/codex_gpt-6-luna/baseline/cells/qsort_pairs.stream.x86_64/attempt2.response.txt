qsort_pairs:
    cmp rsi, 1
    jle .Lbase
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov rbx, rdi
    mov rbp, rsi
    mov rax, rbp
    dec rax
    shl rax, 4
    mov r12, qword ptr [rbx + rax]
    mov r13, qword ptr [rbx + rax + 8]
    xor r14d, r14d
    xor r15d, r15d
.Lpartition:
    lea rax, [r15 + 1]
    cmp rax, rbp
    jge .Lpartition_done
    mov rax, r15
    shl rax, 4
    mov rcx, qword ptr [rbx + rax]
    cmp rcx, r12
    jl .Linclude
    jg .Lnext
    mov rcx, qword ptr [rbx + rax + 8]
    cmp rcx, r13
    jg .Lnext
.Linclude:
    cmp r14, r15
    je .Lincluded
    mov rdx, r14
    shl rdx, 4
    mov rcx, qword ptr [rbx + rdx]
    mov r8, qword ptr [rbx + rdx + 8]
    mov r9, qword ptr [rbx + rax]
    mov r10, qword ptr [rbx + rax + 8]
    mov qword ptr [rbx + rdx], r9
    mov qword ptr [rbx + rdx + 8], r10
    mov qword ptr [rbx + rax], rcx
    mov qword ptr [rbx + rax + 8], r8
.Lincluded:
    inc r14
.Lnext:
    inc r15
    jmp .Lpartition
.Lpartition_done:
    mov rax, rbp
    dec rax
    shl rax, 4
    mov rdx, r14
    shl rdx, 4
    mov rcx, qword ptr [rbx + rdx]
    mov r8, qword ptr [rbx + rdx + 8]
    mov r9, qword ptr [rbx + rax]
    mov r10, qword ptr [rbx + rax + 8]
    mov qword ptr [rbx + rdx], r9
    mov qword ptr [rbx + rdx + 8], r10
    mov qword ptr [rbx + rax], rcx
    mov qword ptr [rbx + rax + 8], r8
    cmp r14, 1
    jle .Lright
    mov rdi, rbx
    mov rsi, r14
    call qsort_pairs
.Lright:
    mov rsi, rbp
    sub rsi, r14
    dec rsi
    cmp rsi, 1
    jle .Ldone
    lea rdi, [r14 + 1]
    shl rdi, 4
    add rdi, rbx
    call qsort_pairs
.Ldone:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Lbase:
    xor eax, eax
    ret
