export function l $collatz_max(l %n) {
@start
    jmp @outer

@outer
    %k =l phi @start 1, @advance %knext
    %best =l phi @start 0, @advance %bestnext
    %within =w cslel %k, %n
    jnz %within, @begin, @done

@begin
    jmp @inner

@inner
    %x =l phi @begin %k, @step %xnext
    %steps =l phi @begin 0, @step %stepsnext
    %finished =w ceql %x, 1
    jnz %finished, @check, @iterate

@iterate
    %parity =l and %x, 1
    %even =w ceql %parity, 0
    jnz %even, @halve, @triple

@halve
    %xhalf =l div %x, 2
    jmp @step(%xhalf)

@triple
    %threetimes =l mul %x, 3
    %xplusone =l add %threetimes, 1
    jmp @step(%xplusone)

@step(%xnext =l)
    %stepsnext =l add %steps, 1
    jmp @inner

@check
    %better =w csgtl %steps, %best
    jnz %better, @update, @advance

@update
    jmp @advance(%steps)

@advance(%bestnext =l)
    %knext =l add %k, 1
    jmp @outer

@done
    ret %best
}
