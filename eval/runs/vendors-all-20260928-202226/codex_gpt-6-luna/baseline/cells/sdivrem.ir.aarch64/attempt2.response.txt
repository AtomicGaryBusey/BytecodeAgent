export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @zero, @check
@zero
	storel %a, %rem
	ret 0
@check
	%amin =w ceql %a, -9223372036854775808
	%bneg =w ceql %b, -1
	%ovf =w and %amin, %bneg
	jnz %ovf, @overflow, @normal
@overflow
	storel 0, %rem
	ret %a
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
