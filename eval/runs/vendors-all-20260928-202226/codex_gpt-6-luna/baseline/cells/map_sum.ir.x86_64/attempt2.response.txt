export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %next
	%sum =l phi @start 0, @body %acc
	%ok =w csltl %i, %n
	jnz %ok, @body, @end
@body
	%off =l mull %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%acc =l add %sum, %r
	%next =l add %i, 1
	jmp @loop
@end
	ret %sum
}
