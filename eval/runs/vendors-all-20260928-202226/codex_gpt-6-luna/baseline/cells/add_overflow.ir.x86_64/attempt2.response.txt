export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum =l add %a, %b
    storel %sum, %out
    %nega =w csltl %a, 0
    %negb =w csltl %b, 0
    %negsum =w csltl %sum, 0
    %samesign =w ceqw %nega, %negb
    %changed =w cnew %negsum, %nega
    %overflow =w and %samesign, %changed
    %result =l extuw %overflow
    ret %result
}
