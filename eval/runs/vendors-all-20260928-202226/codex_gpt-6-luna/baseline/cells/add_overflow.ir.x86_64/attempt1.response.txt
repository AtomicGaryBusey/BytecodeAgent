export function l $add_overflow(l %a, l %b, l %out) {
@start
    %sum = add %a, %b
    storel %sum, %out
    %nega = csltl %a, 0
    %negb = csltl %b, 0
    %negsum = csltl %sum, 0
    %samesign = ceqw %nega, %negb
    %changed = cnew %negsum, %nega
    %overflow = andw %samesign, %changed
    %result = extuw %overflow
    ret %result
}
