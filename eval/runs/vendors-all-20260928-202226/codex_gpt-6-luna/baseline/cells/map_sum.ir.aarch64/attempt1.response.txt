export function l $map_sum(l %a, l %n, l %f) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %nexti
    %sum =l phi @entry 0, @body %nextsum
    %cond =w csltl %i, %n
    jnz %cond, @body, @exit
@body
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %x =l loadl %ptr
    %v =l call %f(l %x)
    %nextsum =l add %sum, %v
    %nexti =l add %i, 1
    jmp @loop
@exit
    ret %sum
}
