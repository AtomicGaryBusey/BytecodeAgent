export function l $shift_sat(l %x, l %n) {
@start
    %small =w csltl %n, 64
    jnz %small, @shift, @saturate
@shift
    %left =l shl %x, %n
    %right =l sar %x, %n
    %sum =l add %left, %right
    jmp @done
@saturate
    %sign =l sar %x, 63
    jmp @done
@done
    %result =l phi @shift %sum, @saturate %sign
    ret %result
}
