collatz_max:
    xor r9d, r9d
    mov r8d, 1
.Louter:
    mov rcx, r8
    xor edx, edx
.Linner:
    cmp rcx, 1
    je .Ldone_inner
    test rcx, 1
    jnz .Lodd
    shr rcx, 1
    jmp .Lstep
.Lodd:
    lea rcx, [rcx + rcx*2 + 1]
.Lstep:
    inc rdx
    jmp .Linner
.Ldone_inner:
    cmp rdx, r9
    cmovg r9, rdx
    cmp r8, rdi
    je .Ldone
    inc r8
    jmp .Louter
.Ldone:
    mov rax, r9
    ret
