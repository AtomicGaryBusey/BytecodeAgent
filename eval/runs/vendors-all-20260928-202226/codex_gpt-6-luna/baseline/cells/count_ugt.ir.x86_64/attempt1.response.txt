export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @head
@head
    %i = phi @entry 0, @body %nexti
    %count = phi @entry 0, @body %nextcount
    %more = csltl %i, %n
    jnz %more, @body, @exit
@body
    %offset = mul %i, 8
    %ptr = add %a, %offset
    %value = loadl %ptr
    %greater = cugtl %value, %t
    %increment = extuw %greater
    %nextcount = add %count, %increment
    %nexti = add %i, 1
    jmp @head
@exit
    ret %count
}
