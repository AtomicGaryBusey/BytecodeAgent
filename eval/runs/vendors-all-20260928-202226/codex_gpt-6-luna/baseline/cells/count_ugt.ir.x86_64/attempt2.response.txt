export function l $count_ugt(l %a, l %n, l %t) {
@entry
    jmp @head
@head
    %i =l phi @entry 0, @body %nexti
    %count =l phi @entry 0, @body %nextcount
    %more =w csltl %i, %n
    jnz %more, @body, @exit
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w cugtl %value, %t
    %increment =l extuw %greater
    %nextcount =l add %count, %increment
    %nexti =l add %i, 1
    jmp @head
@exit
    ret %count
}
