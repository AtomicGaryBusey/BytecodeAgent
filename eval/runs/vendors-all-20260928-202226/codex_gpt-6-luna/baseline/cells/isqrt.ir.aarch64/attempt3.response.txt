export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @raise %mid, @lower %lo
	%hi =l phi @start 4294967296, @raise %hi, @lower %mid
	%diff =l sub %hi, %lo
	%more =w cultl 1, %diff
	jnz %more, @search, @done
@search
	%rounded =l add %diff, 1
	%half =l udiv %rounded, 2
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%fits =w culel %mid, %quot
	jnz %fits, @raise, @lower
@raise
	jmp @loop
@lower
	jmp @loop
@done
	ret %lo
}
