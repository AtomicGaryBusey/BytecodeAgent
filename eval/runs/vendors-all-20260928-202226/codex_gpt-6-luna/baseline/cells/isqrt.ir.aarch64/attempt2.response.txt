export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l copy 4294967296
	jmp @loop
@loop
	%diff =l sub %hi, %lo
	%more =w cult 1, %diff
	jnz %more, @search, @done
@search
	%rounded =l add %diff, 1
	%half =l udiv %rounded, 2
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%fits =w cule %mid, %quot
	jnz %fits, @raise, @lower
@raise
	%lo =l copy %mid
	jmp @loop
@lower
	%hi =l copy %mid
	jmp @loop
@done
	ret %lo
}
