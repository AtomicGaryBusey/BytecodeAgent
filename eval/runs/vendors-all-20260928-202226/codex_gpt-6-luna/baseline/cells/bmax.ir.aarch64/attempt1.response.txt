export function l $bmax(l %a, l %n) {
@start
    %best0 =l loadl %a
    jmp @loop
@loop
    %i =l phi @start 1, @advance %inext
    %best =l phi @start %best0, @advance %bn
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %x =l loadl %ptr
    %larger =w csltl %best, %x
    jnz %larger, @replace, @keep
@replace
    jmp @advance
@keep
    jmp @advance
@advance
    %bn =l phi @replace %x, @keep %best
    %inext =l add %i, 1
    jmp @loop
@done
    ret %best
}
