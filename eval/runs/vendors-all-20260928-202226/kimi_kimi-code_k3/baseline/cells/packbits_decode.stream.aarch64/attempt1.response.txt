packbits_decode:
    mov x4, #0
    mov x5, #0
.Lloop:
    cmp x4, x1
    b.ge .Ldone
    ldrsb w6, [x0, x4]
    add x4, x4, #1
    cmp w6, #0
    b.lt .Lrep
    add x6, x6, #1
    sub x7, x1, x4
    cmp x7, x6
    b.lt .Lerr2
    sub x8, x3, x5
    cmp x8, x6
    b.lt .Llit_trunc
    mov x9, #0
.Llitcopy:
    cmp x9, x6
    b.ge .Lloop
    ldrb w10, [x0, x4]
    strb w10, [x2, x5]
    add x4, x4, #1
    add x5, x5, #1
    add x9, x9, #1
    b .Llitcopy
.Llit_trunc:
    cmp x8, #0
    b.le .Lerr1
    ldrb w10, [x0, x4]
    strb w10, [x2, x5]
    add x4, x4, #1
    add x5, x5, #1
    sub x8, x8, #1
    b .Llit_trunc
.Lrep:
    cmp w6, #-128
    b.eq .Lloop
    cmp x4, x1
    b.ge .Lerr2
    ldrb w7, [x0, x4]
    add x4, x4, #1
    neg w8, w6
    add x8, x8, #1
    sub x9, x3, x5
    cmp x9, x8
    b.lt .Lrep_trunc
.Lrepfill:
    cmp x8, #0
    b.le .Lloop
    strb w7, [x2, x5]
    add x5, x5, #1
    sub x8, x8, #1
    b .Lrepfill
.Lrep_trunc:
    cmp x9, #0
    b.le .Lerr1
    strb w7, [x2, x5]
    add x5, x5, #1
    sub x9, x9, #1
    b .Lrep_trunc
.Ldone:
    mov x0, x5
    ret
.Lerr1:
    mov x0, #-1
    ret
.Lerr2:
    mov x0, #-2
    ret
