export function l $isqrt(l %n) {
@start
	%lo_p =l alloc8 8
	%hi_p =l alloc8 8
	storel 0, %lo_p
	storel 4294967296, %hi_p
@loop
	%lo =l loadl %lo_p
	%hi =l loadl %hi_p
	%lo1 =l add %lo, 1
	%cont =w cultl %lo1, %hi
	jnz %cont, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @take, @drop
@take
	storel %mid, %lo_p
	jmp @loop
@drop
	storel %mid, %hi_p
	jmp @loop
@done
	%r =l loadl %lo_p
	ret %r
}
