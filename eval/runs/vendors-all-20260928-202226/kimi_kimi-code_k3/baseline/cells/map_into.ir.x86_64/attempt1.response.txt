export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%x =l loadl %sp
	%y =l call %f(l %x)
	%dp =l add %dst, %off
	storel %y, %dp
	%less =w csltl %y, %x
	%lesse =l extsw %less
	%cnt =l add %cnt, %lesse
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
