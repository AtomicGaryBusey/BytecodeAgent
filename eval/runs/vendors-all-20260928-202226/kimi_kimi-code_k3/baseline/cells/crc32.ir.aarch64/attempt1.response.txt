export function l $crc32(l %p, l %n) {
@entry
	%i =l copy 0
	%crc =w copy -1
@outer_test
	%odo =w csltl %i, %n
	jnz %odo, @byte, @done
@byte
	%ptr =l add %p, %i
	%b =w loadub %ptr
	%crc =w xor %crc, %b
	%k =w copy 0
@bit_test
	%kdo =w csltw %k, 8
	jnz %kdo, @bit, @next_byte
@bit
	%lsb =w and %crc, 1
	%sh =w shr %crc, 1
	jnz %lsb, @odd, @even
@odd
	%crc =w xor %sh, -306674912
	jmp @bit_next
@even
	%crc =w copy %sh
@bit_next
	%k =w add %k, 1
	jmp @bit_test
@next_byte
	%i =l add %i, 1
	jmp @outer_test
@done
	%outw =w xor %crc, -1
	%out =l extuw %outw
	ret %out
}
