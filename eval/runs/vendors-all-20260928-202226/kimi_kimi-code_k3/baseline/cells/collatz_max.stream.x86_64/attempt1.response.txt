collatz_max:
    xor eax, eax
    mov rcx, 1
.Louter:
    cmp rcx, rdi
    jg .Ldone
    mov rdx, rcx
    xor r8d, r8d
.Linner:
    cmp rdx, 1
    je .Linner_done
    test rdx, 1
    jz .Leven
    lea rdx, [rdx + rdx*2 + 1]
    jmp .Lstep
.Leven:
    shr rdx, 1
.Lstep:
    inc r8
    jmp .Linner
.Linner_done:
    cmp r8, rax
    cmovg rax, r8
    inc rcx
    jmp .Louter
.Ldone:
    ret
