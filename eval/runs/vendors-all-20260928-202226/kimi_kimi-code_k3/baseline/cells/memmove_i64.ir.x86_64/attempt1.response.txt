export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%n8 =l mul %n, 8
	%send =l add %src, %n8
	%lt =w csltl %dst, %src
	%ge =w cslel %send, %dst
	%fwd =w or %lt, %ge
	jnz %fwd, @fcheck, @bcheck
@fcheck
	%fi =l phi @start 0, @fbody %fi2
	%fc =w csltl %fi, %n8
	jnz %fc, @fbody, @done
@fbody
	%fp =l add %src, %fi
	%fv =l loadl %fp
	%fq =l add %dst, %fi
	storel %fv, %fq
	%fi2 =l add %fi, 8
	jmp @fcheck
@bcheck
	%bi =l phi @start %n8, @bbody %bi2
	%bc =w csgtl %bi, 0
	jnz %bc, @bbody, @done
@bbody
	%bi2 =l sub %bi, 8
	%bp =l add %src, %bi2
	%bv =l loadl %bp
	%bq =l add %dst, %bi2
	storel %bv, %bq
	jmp @bcheck
@done
	ret 0
}
