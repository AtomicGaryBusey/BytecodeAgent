export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %ip =l alloc8 8
    %jp =l alloc8 8
    %cp =l alloc8 8
    %kp =l alloc8 8
    %ntotal =l add %na, %nb
    storel 0, %ip
@zerocheck
    %zi =l loadl %ip
    %zinrange =w csltl %zi, %ntotal
    jnz %zinrange, @zerobody, @outerinit
@zerobody
    %zoff =l mul %zi, 8
    %zptr =l add %r, %zoff
    storel 0, %zptr
    %znext =l add %zi, 1
    storel %znext, %ip
    jmp @zerocheck
@outerinit
    storel 0, %ip
@outercheck
    %oi =l loadl %ip
    %oinrange =w csltl %oi, %na
    jnz %oinrange, @outerbody, @outerdone
@outerbody
    storel 0, %cp
    storel 0, %jp
@innercheck
    %ij =l loadl %jp
    %ijinrange =w csltl %ij, %nb
    jnz %ijinrange, @innerbody, @innerdone
@innerbody
    %irow =l loadl %ip
    %jcol =l loadl %jp
    %aoff =l mul %irow, 8
    %aptr =l add %a, %aoff
    %xv =l loadl %aptr
    %boff =l mul %jcol, 8
    %bptr =l add %b, %boff
    %yv =l loadl %bptr
    %x0 =l and %xv, 4294967295
    %x1 =l shr %xv, 32
    %y0 =l and %yv, 4294967295
    %y1 =l shr %yv, 32
    %p00 =l mul %x0, %y0
    %p01 =l mul %x0, %y1
    %p10 =l mul %x1, %y0
    %p11 =l mul %x1, %y1
    %p01lo =l and %p01, 4294967295
    %p10lo =l and %p10, 4294967295
    %mid =l add %p01lo, %p10lo
    %cmidl =l shr %mid, 32
    %midshift =l shl %mid, 32
    %prodlow =l add %p00, %midshift
    %clow =w cultl %prodlow, %p00
    %clowl =l extuw %clow
    %p01hi =l shr %p01, 32
    %p10hi =l shr %p10, 32
    %h1 =l add %p11, %p01hi
    %h2 =l add %h1, %p10hi
    %h3 =l add %h2, %cmidl
    %prodhigh =l add %h3, %clowl
    %kidx =l add %irow, %jcol
    %koff =l mul %kidx, 8
    %rptr =l add %r, %koff
    %oldr =l loadl %rptr
    %sum0 =l add %prodlow, %oldr
    %cold =w cultl %sum0, %prodlow
    %coldl =l extuw %cold
    %carryin =l loadl %cp
    %newlow =l add %sum0, %carryin
    %ccarry =w cultl %newlow, %sum0
    %ccarryl =l extuw %ccarry
    %newhigh0 =l add %prodhigh, %coldl
    %newhigh =l add %newhigh0, %ccarryl
    storel %newlow, %rptr
    storel %newhigh, %cp
    %jnext =l add %jcol, 1
    storel %jnext, %jp
    jmp @innercheck
@innerdone
    %ifinal =l loadl %ip
    %finalidx =l add %ifinal, %nb
    %finaloff =l mul %finalidx, 8
    %finalptr =l add %r, %finaloff
    %finalcarry =l loadl %cp
    storel %finalcarry, %finalptr
    %inext =l add %ifinal, 1
    storel %inext, %ip
    jmp @outercheck
@outerdone
    storel %ntotal, %kp
@scancheck
    %kval =l loadl %kp
    %kpositive =w csgtl %kval, 0
    jnz %kpositive, @scanbody, @done
@scanbody
    %kminus =l sub %kval, 1
    %scanoff =l mul %kminus, 8
    %scanptr =l add %r, %scanoff
    %scanlimb =l loadl %scanptr
    %scanzero =w ceql %scanlimb, 0
    jnz %scanzero, @scandec, @done
@scandec
    storel %kminus, %kp
    jmp @scancheck
@done
    %result =l loadl %kp
    ret %result
}
