export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%lt =w csltl %i, %n
	jnz %lt, @body, @ret
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cultl %t, %v
	%gtl =l extsw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	jmp @loop
@ret
	ret %cnt
}
