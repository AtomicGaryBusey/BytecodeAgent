export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%i =l copy 0
@init_loop
	%ic =w csltl %i, %n
	jnz %ic, @init_body, @init_done
@init_body
	%ioff =l mul %i, 8
	%ip =l add %row, %ioff
	storel 0, %ip
	%i =l add %i, 1
	jmp @init_loop
@init_done
	%i2 =l copy 0
@outer
	%oc =w csltl %i2, %m
	jnz %oc, @outer_body, @done
@outer_body
	%xp =l add %x, %i2
	%xi =w loadub %xp
	%diag =l copy 0
	%j =l copy 1
@inner
	%jc =w cslel %j, %n
	jnz %jc, @inner_body, @outer_next
@inner_body
	%joff =l mul %j, 8
	%pj =l add %row, %joff
	%up =l loadl %pj
	%jm1 =l sub %j, 1
	%jm1off =l mul %jm1, 8
	%pjm1 =l add %row, %jm1off
	%left =l loadl %pjm1
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%mv =l add %diag, 1
	storel %mv, %pj
	jmp @advance
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @choose_up, @choose_left
@choose_up
	storel %up, %pj
	jmp @advance
@choose_left
	storel %left, %pj
@advance
	%diag =l copy %up
	%j =l add %j, 1
	jmp @inner
@outer_next
	%i2 =l add %i2, 1
	jmp @outer
@done
	%noff =l mul %n, 8
	%pn =l add %row, %noff
	%res =l loadl %pn
	ret %res
}
