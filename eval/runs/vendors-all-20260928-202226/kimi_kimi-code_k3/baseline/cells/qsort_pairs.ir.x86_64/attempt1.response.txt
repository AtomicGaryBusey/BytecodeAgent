export function l $qsort_pairs(l %a, l %n) {
@entry
	%base =w csltl %n, 2
	jnz %base, @ret, @go
@ret
	%z =l copy 0
	ret %z
@go
	%last =l sub %n, 1
	%lastoff =l mul %last, 16
	%plast =l add %a, %lastoff
	%pk =l loadl %plast
	%pvaddr =l add %plast, 8
	%pv =l loadl %pvaddr
	%i =l copy 0
	%j =l copy 0
@loop
	%jok =w csltl %j, %last
	jnz %jok, @body, @done
@body
	%joff =l mul %j, 16
	%pj =l add %a, %joff
	%kj =l loadl %pj
	%vjaddr =l add %pj, 8
	%vj =l loadl %vjaddr
	%klt =w csltl %kj, %pk
	%keq =w ceql %kj, %pk
	%vle =w cslel %vj, %pv
	%eqvle =w and %keq, %vle
	%take =w or %klt, %eqvle
	jnz %take, @swap, @noswap
@swap
	%ioff =l mul %i, 16
	%pi =l add %a, %ioff
	%ki =l loadl %pi
	%viaddr =l add %pi, 8
	%vi =l loadl %viaddr
	storel %ki, %pj
	storel %vi, %vjaddr
	storel %kj, %pi
	storel %vj, %viaddr
	%i =l add %i, 1
	%j =l add %j, 1
	jmp @loop
@noswap
	%j =l add %j, 1
	jmp @loop
@done
	%ioff2 =l mul %i, 16
	%pi2 =l add %a, %ioff2
	%ki2 =l loadl %pi2
	%vi2addr =l add %pi2, 8
	%vi2 =l loadl %vi2addr
	storel %ki2, %plast
	storel %vi2, %pvaddr
	storel %pk, %pi2
	storel %pv, %vi2addr
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%rop =l mul %ip1, 16
	%rp =l add %a, %rop
	%rn =l sub %n, %ip1
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	%zero =l copy 0
	ret %zero
}
