export function l $max_byte(l %p, l %n) {
@entry
	%zero =l copy 0
	%has =w csgtl %n, %zero
	jnz %has, @loop, @zero
@loop
	%i =l phi @entry %zero, @step %inext
	%m =l phi @entry %zero, @step %mnext
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %m
	jnz %gt, @upd, @keep
@upd
	jmp @step
@keep
	jmp @step
@step
	%mnext =l phi @upd %b, @keep %m
	%inext =l add %i, 1
	%more =w csgtl %n, %inext
	jnz %more, @loop, @done
@done
	ret %mnext
@zero
	ret 0
}
