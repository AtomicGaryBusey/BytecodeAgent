export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%v =l loadl %ptr
	%z =w ceql %v, 0
	jnz %z, @done, @next
@next
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %i
}
