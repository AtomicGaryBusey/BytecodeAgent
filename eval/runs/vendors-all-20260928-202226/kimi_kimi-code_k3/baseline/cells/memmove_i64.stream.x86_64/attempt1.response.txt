memmove_i64:
cmp rdi, rsi
je .Ldone
jb .Lforward_setup
mov rcx, rdx
.Lbackward:
test rcx, rcx
jz .Ldone
dec rcx
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
jmp .Lbackward
.Lforward_setup:
xor ecx, ecx
.Lforward:
cmp rcx, rdx
jge .Ldone
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
inc rcx
jmp .Lforward
.Ldone:
xor eax, eax
ret
