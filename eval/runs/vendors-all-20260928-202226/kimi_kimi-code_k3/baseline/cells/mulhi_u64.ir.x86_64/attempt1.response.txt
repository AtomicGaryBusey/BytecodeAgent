export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32
	%p00 =l mul %al, %bl
	%p01 =l mul %al, %bh
	%p10 =l mul %ah, %bl
	%p11 =l mul %ah, %bh
	%s =l add %p01, %p10
	%cmidw =w cultl %s, %p01
	%cmid =l extsw %cmidw
	%slo =l shl %s, 32
	%t =l add %p00, %slo
	%cloww =w cultl %t, %p00
	%clow =l extsw %cloww
	%shi =l shr %s, 32
	%cmid32 =l shl %cmid, 32
	%h0 =l add %p11, %shi
	%h1 =l add %h0, %clow
	%h2 =l add %h1, %cmid32
	ret %h2
}
