mp_mul:

push rbp
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
mov r15, r8
lea rbp, [rsi + rcx]
xor eax, eax
test rbp, rbp
jz .Ldone
xor ecx, ecx
.Lzero:
mov qword ptr [r15 + rcx*8], 0
inc rcx
cmp rcx, rbp
jb .Lzero
xor edi, edi
test r12, r12
jz .Lscaninit
.Louter:
mov rdx, [rbx + rdi*8]
lea rcx, [r15 + rdi*8]
xor eax, eax
xor esi, esi
test r14, r14
jz .Lrowend
.Linner:
mulx r10, r11, qword ptr [r13 + rsi*8]
add r11, rax
adc r10, 0
add r11, [rcx + rsi*8]
mov [rcx + rsi*8], r11
adc r10, 0
mov rax, r10
inc rsi
cmp rsi, r14
jb .Linner
.Lrowend:
mov [rcx + rsi*8], rax
inc rdi
cmp rdi, r12
jb .Louter
.Lscaninit:
mov rax, rbp
.Lscan:
test rax, rax
jz .Ldone
cmp qword ptr [r15 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lscan
.Ldone:

pop r14
pop r13
pop r12
pop rbp
pop rbx
