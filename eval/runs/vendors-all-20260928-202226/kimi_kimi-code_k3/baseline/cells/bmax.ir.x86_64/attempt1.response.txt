export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%cond =w csltl %i, %n
	jnz %cond, @body, @end
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%gt =w csltl %max, %v
	jnz %gt, @upd, @next
@upd
	%max =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %max
}
