mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov rbx, rsi
mov r15, rdx
mov r13, rcx
mov r14, r8
mov rbp, rbx
imul rbp, rbx
xor ecx, ecx
.Lzero_out:
cmp rcx, rbp
jge .Lzero_out_done
mov qword ptr [r13 + rcx*8], 0
inc rcx
jmp .Lzero_out
.Lzero_out_done:
xor ecx, ecx
.Ldiag:
cmp rcx, rbx
jge .Ldiag_done
mov rdx, rcx
imul rdx, rbx
add rdx, rcx
mov qword ptr [r13 + rdx*8], 1
inc rcx
jmp .Ldiag
.Ldiag_done:
.Lkloop:
test r15, r15
jz .Ldone_mul
dec r15
xor r8, r8
.Li_loop:
cmp r8, rbx
jge .Li_done
xor r9, r9
.Lj_loop:
cmp r9, rbx
jge .Lj_done
mov rax, r8
imul rax, rbx
xor r11, r11
xor r10, r10
.Lt_loop:
cmp r10, rbx
jge .Lt_done
mov rdx, rax
add rdx, r10
mov rdx, qword ptr [r13 + rdx*8]
mov rcx, r10
imul rcx, rbx
add rcx, r9
mov rcx, qword ptr [r12 + rcx*8]
imul rdx, rcx
add r11, rdx
inc r10
jmp .Lt_loop
.Lt_done:
mov rdx, rax
add rdx, r9
mov qword ptr [r14 + rdx*8], r11
inc r9
jmp .Lj_loop
.Lj_done:
inc r8
jmp .Li_loop
.Li_done:
xor ecx, ecx
.Lcopy:
cmp rcx, rbp
jge .Lcopy_done
mov rax, qword ptr [r14 + rcx*8]
mov qword ptr [r13 + rcx*8], rax
inc rcx
jmp .Lcopy
.Lcopy_done:
jmp .Lkloop
.Ldone_mul:
lea rdx, [rbp + rbp]
xor ecx, ecx
.Lztmp:
cmp rcx, rdx
jge .Lztmp_done
mov qword ptr [r14 + rcx*8], 0
inc rcx
jmp .Lztmp
.Lztmp_done:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
