export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
@outer
	%inrange =w cslel %k, %n
	jnz %inrange, @body, @ret
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%isone =w ceql %x, 1
	jnz %isone, @innerdone, @step
@step
	%lowbit =l and %x, 1
	%isodd =w ceql %lowbit, 1
	jnz %isodd, @odd, @even
@even
	%x =l div %x, 2
	jmp @bump
@odd
	%triple =l mul %x, 3
	%x =l add %triple, 1
	jmp @bump
@bump
	%steps =l add %steps, 1
	jmp @inner
@innerdone
	%ismax =w csgtl %steps, %max
	jnz %ismax, @setmax, @next
@setmax
	%max =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@ret
	ret %max
}
