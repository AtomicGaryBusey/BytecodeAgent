export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%below =w cugtl %dst, %src
	jnz %below, @fwd, @bwd
@fwd
	%fi =l phi @start 0, @fwd_body %fi2
	%fc =w csltl %fi, %n
	jnz %fc, @fwd_body, @done
@fwd_body
	%fo =l mul %fi, 8
	%fa =l add %src, %fo
	%fv =l loadl %fa
	%fd =l add %dst, %fo
	storel %fv, %fd
	%fi2 =l add %fi, 1
	jmp @fwd
@bwd
	%bi =l phi @start %n, @bwd_body %bi2
	%bc =w csgtl %bi, 0
	jnz %bc, @bwd_body, @done
@bwd_body
	%bi2 =l sub %bi, 1
	%bo =l mul %bi2, 8
	%ba =l add %src, %bo
	%bv =l loadl %ba
	%bd =l add %dst, %bo
	storel %bv, %bd
	jmp @bwd
@done
	ret 0
}
