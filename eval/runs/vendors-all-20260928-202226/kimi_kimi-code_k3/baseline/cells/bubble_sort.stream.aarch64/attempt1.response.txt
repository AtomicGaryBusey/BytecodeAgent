bubble_sort:
	cmp	x1, #2
	b.lt	.Ldone
	sub	x2, x1, #1
.Louter:
	mov	x3, #0
.Linner:
	cmp	x3, x2
	b.ge	.Linner_done
	add	x6, x3, #1
	ldr	x4, [x0, x3, lsl #3]
	ldr	x5, [x0, x6, lsl #3]
	cmp	x4, x5
	b.le	.Lnoswap
	str	x5, [x0, x3, lsl #3]
	str	x4, [x0, x6, lsl #3]
.Lnoswap:
	add	x3, x3, #1
	b	.Linner
.Linner_done:
	subs	x2, x2, #1
	b.ne	.Louter
.Ldone:
	mov	x0, #0
	ret
