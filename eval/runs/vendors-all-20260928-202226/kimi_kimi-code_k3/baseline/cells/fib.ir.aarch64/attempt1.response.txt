export function l $fib(l %n) {
@start
	%pa =l alloc8 8
	%pb =l alloc8 8
	%pi =l alloc8 8
	storel 0, %pa
	storel 1, %pb
	storel 0, %pi
@loop
	%i =l loadl %pi
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%a =l loadl %pa
	%b =l loadl %pb
	%s =l add %a, %b
	storel %b, %pa
	storel %s, %pb
	%i2 =l add %i, 1
	storel %i2, %pi
	jmp @loop
@done
	%r =l loadl %pa
	ret %r
}
