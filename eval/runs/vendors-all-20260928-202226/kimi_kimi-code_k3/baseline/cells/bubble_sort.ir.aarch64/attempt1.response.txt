export function l $bubble_sort(l %a, l %n) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%nm1 =l sub %n, %one
	%small =w cslel %n, %one
	jnz %small, @ret, @outer
@outer
	%i =l phi @start %zero, @outer_next %i_next
	%oc =w csltl %i, %nm1
	%limit =l sub %nm1, %i
	jnz %oc, @inner, @ret
@inner
	%j =l phi @outer %zero, @inner_next %j_next
	%jc =w csltl %j, %limit
	jnz %jc, @body, @outer_next
@body
	%off =l mul %j, %eight
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%j1 =l add %j, %one
	%off1 =l mul %j1, %eight
	%p1 =l add %a, %off1
	%v1 =l loadl %p1
	%gt =w csgtl %v0, %v1
	jnz %gt, @swap, @inner_next
@swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_next
@inner_next
	%j_next =l add %j, %one
	jmp @inner
@outer_next
	%i_next =l add %i, %one
	jmp @outer
@ret
	%r =l copy 0
	ret %r
}
