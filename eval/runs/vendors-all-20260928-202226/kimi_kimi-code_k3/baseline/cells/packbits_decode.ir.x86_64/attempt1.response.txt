export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%go =w csltl %i, %n
	jnz %go, @body, @ret
@ret
	ret %o
@body
	%hp =l add %src, %i
	%hb =w loadsb %hp
	%hl =l extsw %hb
	%isnop =w ceqw %hb, -128
	jnz %isnop, @donop, @pkt
@donop
	%i =l add %i, 1
	jmp @loop
@pkt
	%lit =w csgew %hb, 0
	jnz %lit, @literal, @repeat
@literal
	%cnt =l add %hl, 1
	%need =l add %i, %cnt
	%need =l add %need, 1
	%ok =w csgel %n, %need
	jnz %ok, @litcap, @err2
@litcap
	%tot =l add %o, %cnt
	%fits =w csgel %cap, %tot
	jnz %fits, @litcopy, @litfill
@litcopy
	%k =l copy 0
@litcopyl
	%kd =w csltl %k, %cnt
	jnz %kd, @litcopyb, @litdone
@litcopyb
	%sp =l add %src, %i
	%sp =l add %sp, %k
	%sp =l add %sp, 1
	%b =w loadub %sp
	%op =l add %out, %o
	%op =l add %op, %k
	storeb %b, %op
	%k =l add %k, 1
	jmp @litcopyl
@litdone
	%o =l add %o, %cnt
	%i =l add %i, %cnt
	%i =l add %i, 1
	jmp @loop
@litfill
	%kf =l copy 0
@litfilll
	%fk =l add %o, %kf
	%fd =w csltl %fk, %cap
	jnz %fd, @litfillb, @err1
@litfillb
	%spf =l add %src, %i
	%spf =l add %spf, %kf
	%spf =l add %spf, 1
	%bf =w loadub %spf
	%opf =l add %out, %o
	%opf =l add %opf, %kf
	storeb %bf, %opf
	%kf =l add %kf, 1
	jmp @litfilll
@repeat
	%need2 =l add %i, 2
	%ok2 =w csgel %n, %need2
	jnz %ok2, @repok, @err2
@repok
	%rcnt =l sub 1, %hl
	%rbp =l add %src, %i
	%rbp =l add %rbp, 1
	%rb =w loadub %rbp
	%rtot =l add %o, %rcnt
	%rfit =w csgel %cap, %rtot
	jnz %rfit, @repcopy, @repfill
@repcopy
	%k2 =l copy 0
@repcopyl
	%k2d =w csltl %k2, %rcnt
	jnz %k2d, @repcopyb, @repdone
@repcopyb
	%op2 =l add %out, %o
	%op2 =l add %op2, %k2
	storeb %rb, %op2
	%k2 =l add %k2, 1
	jmp @repcopyl
@repdone
	%o =l add %o, %rcnt
	%i =l add %i, 2
	jmp @loop
@repfill
	%k3 =l copy 0
@repfilll
	%fk3 =l add %o, %k3
	%fd3 =w csltl %fk3, %cap
	jnz %fd3, @repfillb, @err1
@repfillb
	%op3 =l add %out, %o
	%op3 =l add %op3, %k3
	storeb %rb, %op3
	%k3 =l add %k3, 1
	jmp @repfilll
@err2
	ret -2
@err1
	ret -1
}
