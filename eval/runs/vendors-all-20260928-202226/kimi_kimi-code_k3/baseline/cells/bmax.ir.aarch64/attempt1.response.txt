export function l $bmax(l %a, l %n) {
@start
	%m =l loadl %a
	%i =l copy 1
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%t =w csltl %m, %v
	jnz %t, @set, @next
@set
	%m =l copy %v
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %m
}
