export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%i =l copy 0
@zero_out_test
	%zc =w csltl %i, %nn
	jnz %zc, @zero_out_body, @zero_out_done
@zero_out_body
	%off =l mul %i, 8
	%p =l add %out, %off
	storel 0, %p
	%i =l add %i, 1
	jmp @zero_out_test
@zero_out_done
	%nn2 =l mul %nn, 2
	%i2 =l copy 0
@zero_tmp_test
	%zc2 =w csltl %i2, %nn2
	jnz %zc2, @zero_tmp_body, @zero_tmp_done
@zero_tmp_body
	%off2 =l mul %i2, 8
	%p2 =l add %tmp, %off2
	storel 0, %p2
	%i2 =l add %i2, 1
	jmp @zero_tmp_test
@zero_tmp_done
	%id =l copy 0
@diag_test
	%dc =w csltl %id, %n
	jnz %dc, @diag_body, @diag_done
@diag_body
	%d1 =l mul %id, %n
	%d2 =l add %d1, %id
	%d3 =l mul %d2, 8
	%dp =l add %out, %d3
	storel 1, %dp
	%id =l add %id, 1
	jmp @diag_test
@diag_done
	%e =l copy 0
@pow_test
	%pc =w csltl %e, %k
	jnz %pc, @pow_body, @pow_done
@pow_body
	%ri =l copy 0
@row_test
	%rc =w csltl %ri, %n
	jnz %rc, @row_body, @row_done
@row_body
	%rj =l copy 0
@col_test
	%cc =w csltl %rj, %n
	jnz %cc, @col_body, @col_done
@col_body
	%sum =l copy 0
	%t =l copy 0
@inner_test
	%ic =w csltl %t, %n
	jnz %ic, @inner_body, @inner_done
@inner_body
	%x1 =l mul %ri, %n
	%x2 =l add %x1, %t
	%x3 =l mul %x2, 8
	%pa =l add %out, %x3
	%va =l loadl %pa
	%y1 =l mul %t, %n
	%y2 =l add %y1, %rj
	%y3 =l mul %y2, 8
	%pb =l add %a, %y3
	%vb =l loadl %pb
	%prod =l mul %va, %vb
	%sum =l add %sum, %prod
	%t =l add %t, 1
	jmp @inner_test
@inner_done
	%z1 =l mul %ri, %n
	%z2 =l add %z1, %rj
	%z3 =l mul %z2, 8
	%pz =l add %tmp, %z3
	storel %sum, %pz
	%rj =l add %rj, 1
	jmp @col_test
@col_done
	%ri =l add %ri, 1
	jmp @row_test
@row_done
	%ci =l copy 0
@copy_test
	%cpc =w csltl %ci, %nn
	jnz %cpc, @copy_body, @copy_done
@copy_body
	%co =l mul %ci, 8
	%pt =l add %tmp, %co
	%cv =l loadl %pt
	%po =l add %out, %co
	storel %cv, %po
	%ci =l add %ci, 1
	jmp @copy_test
@copy_done
	%e =l add %e, 1
	jmp @pow_test
@pow_done
	%fi =l copy 0
@fz_test
	%fzc =w csltl %fi, %nn2
	jnz %fzc, @fz_body, @fz_done
@fz_body
	%fo =l mul %fi, 8
	%fp =l add %tmp, %fo
	storel 0, %fp
	%fi =l add %fi, 1
	jmp @fz_test
@fz_done
	ret 0
}
