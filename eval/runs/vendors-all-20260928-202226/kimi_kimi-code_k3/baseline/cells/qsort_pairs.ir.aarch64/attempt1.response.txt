export function l $qsort_pairs(l %a, l %n) {
@start
	%pa =l alloc8 8
	%pn =l alloc8 8
	%pk =l alloc8 8
	%pv =l alloc8 8
	%si =l alloc8 8
	%sj =l alloc8 8
	storel %a, %pa
	storel %n, %pn
@loop
	%n0 =l loadl %pn
	%base =w csltl %n0, 2
	jnz %base, @done, @part
@part
	%a0 =l loadl %pa
	%n1 =l loadl %pn
	%nm1 =l sub %n1, 1
	%poff =l mul %nm1, 16
	%paddr =l add %a0, %poff
	%pvaddr =l add %paddr, 8
	%tkey =l loadl %paddr
	%tval =l loadl %pvaddr
	storel %tkey, %pk
	storel %tval, %pv
	storel 0, %si
	storel 0, %sj
@for
	%j =l loadl %sj
	%n2 =l loadl %pn
	%jm1 =l sub %n2, 1
	%jin =w csltl %j, %jm1
	jnz %jin, @fbody, @fend
@fbody
	%a1 =l loadl %pa
	%joff =l mul %j, 16
	%jaddr =l add %a1, %joff
	%jvaddr =l add %jaddr, 8
	%jk =l loadl %jaddr
	%jv =l loadl %jvaddr
	%pkey =l loadl %pk
	%pvalue =l loadl %pv
	%lt =w csltl %jk, %pkey
	jnz %lt, @swap, @feq
@feq
	%eq =w ceql %jk, %pkey
	jnz %eq, @fle, @fnext
@fle
	%le =w cslel %jv, %pvalue
	jnz %le, @swap, @fnext
@swap
	%i =l loadl %si
	%a2 =l loadl %pa
	%ioff =l mul %i, 16
	%iaddr =l add %a2, %ioff
	%ivaddr =l add %iaddr, 8
	%ik =l loadl %iaddr
	%iv =l loadl %ivaddr
	%j2 =l loadl %sj
	%joff2 =l mul %j2, 16
	%jaddr2 =l add %a2, %joff2
	%jvaddr2 =l add %jaddr2, 8
	%jk2 =l loadl %jaddr2
	%jv2 =l loadl %jvaddr2
	storel %jk2, %iaddr
	storel %jv2, %ivaddr
	storel %ik, %jaddr2
	storel %iv, %jvaddr2
	%i1 =l add %i, 1
	storel %i1, %si
	jmp @fnext
@fnext
	%j3 =l loadl %sj
	%j4 =l add %j3, 1
	storel %j4, %sj
	jmp @for
@fend
	%i3 =l loadl %si
	%a3 =l loadl %pa
	%n3 =l loadl %pn
	%nm1c =l sub %n3, 1
	%ioff3 =l mul %i3, 16
	%iaddr3 =l add %a3, %ioff3
	%ivaddr3 =l add %iaddr3, 8
	%ik3 =l loadl %iaddr3
	%iv3 =l loadl %ivaddr3
	%eoff =l mul %nm1c, 16
	%eaddr =l add %a3, %eoff
	%evaddr =l add %eaddr, 8
	%ek =l loadl %eaddr
	%ev =l loadl %evaddr
	storel %ek, %iaddr3
	storel %ev, %ivaddr3
	storel %ik3, %eaddr
	storel %iv3, %evaddr
	%rsz =l sub %nm1c, %i3
	%ltsz =w csltl %i3, %rsz
	jnz %ltsz, @recleft, @recright
@recleft
	%aL =l loadl %pa
	%lszL =l loadl %si
	%rL =l call $qsort_pairs(l %aL, l %lszL)
	%lp1 =l add %lszL, 1
	%loff =l mul %lp1, 16
	%newa =l add %aL, %loff
	storel %newa, %pa
	%nL =l loadl %pn
	%newn =l sub %nL, %lp1
	storel %newn, %pn
	jmp @loop
@recright
	%aR =l loadl %pa
	%lszR =l loadl %si
	%lp1R =l add %lszR, 1
	%loffR =l mul %lp1R, 16
	%raddr =l add %aR, %loffR
	%nR =l loadl %pn
	%rszR =l sub %nR, %lp1R
	%rR =l call $qsort_pairs(l %raddr, l %rszR)
	storel %lszR, %pn
	jmp @loop
@done
	ret 0
}
