export function l $base64_encode(l %src, l %n, l %out) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	%kp =l alloc8 8
	%xp =l alloc8 8
	%cntp =l alloc8 8
	%padp =l alloc8 8
	storel 0, %ip
	storel %out, %op
	jmp @outer
@outer
	%i =l loadl %ip
	%n3 =l sub %n, 2
	%full =w csltl %i, %n3
	jnz %full, @full, @tail
@full
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%i1 =l add %i, 1
	%p1 =l add %src, %i1
	%b1 =l loadub %p1
	%i2 =l add %i, 2
	%p2 =l add %src, %i2
	%b2 =l loadub %p2
	%x0 =l shl %b0, 16
	%y0 =l shl %b1, 8
	%xy =l or %x0, %y0
	%x =l or %xy, %b2
	storel %x, %xp
	storel 4, %cntp
	storel 0, %padp
	storel 0, %kp
	jmp @emitloop
@tail
	%rem =l sub %n, %i
	%is0 =w ceql %rem, 0
	jnz %is0, @finish, @tail_nonempty
@tail_nonempty
	%tp0 =l add %src, %i
	%tb0 =l loadub %tp0
	%tx0 =l shl %tb0, 16
	%is2 =w ceql %rem, 2
	jnz %is2, @tail2, @tail1
@tail1
	storel %tx0, %xp
	storel 2, %cntp
	storel 2, %padp
	storel 0, %kp
	jmp @emitloop
@tail2
	%ti1 =l add %i, 1
	%tp1 =l add %src, %ti1
	%tb1 =l loadub %tp1
	%ty0 =l shl %tb1, 8
	%tx =l or %tx0, %ty0
	storel %tx, %xp
	storel 3, %cntp
	storel 1, %padp
	storel 0, %kp
	jmp @emitloop
@emitloop
	%k =l loadl %kp
	%cnt =l loadl %cntp
	%more =w csltl %k, %cnt
	jnz %more, @doemit, @afteremit
@doemit
	%o =l loadl %op
	%xv =l loadl %xp
	%sixk =l mul %k, 6
	%sh =l sub 18, %sixk
	%raw =l shr %xv, %sh
	%v =l and %raw, 63
	%u26 =w csltl %v, 26
	jnz %u26, @m_upper, @m_notupper
@m_upper
	%ch =l add %v, 65
	storeb %ch, %o
	jmp @emit_next
@m_notupper
	%u52 =w csltl %v, 52
	jnz %u52, @m_lower, @m_notlower
@m_lower
	%ch =l add %v, 71
	storeb %ch, %o
	jmp @emit_next
@m_notlower
	%u62 =w csltl %v, 62
	jnz %u62, @m_digit, @m_sym
@m_digit
	%ch =l sub %v, 4
	storeb %ch, %o
	jmp @emit_next
@m_sym
	%is62 =w ceql %v, 62
	jnz %is62, @m_plus, @m_slash
@m_plus
	%ch =l copy 43
	storeb %ch, %o
	jmp @emit_next
@m_slash
	%ch =l copy 47
	storeb %ch, %o
	jmp @emit_next
@emit_next
	%o0 =l loadl %op
	%o1 =l add %o0, 1
	storel %o1, %op
	%k0 =l loadl %kp
	%k1 =l add %k0, 1
	storel %k1, %kp
	jmp @emitloop
@afteremit
	%pad =l loadl %padp
	%haspad =w csgtl %pad, 0
	jnz %haspad, @tailpad, @fullnext
@fullnext
	%fi =l loadl %ip
	%fi3 =l add %fi, 3
	storel %fi3, %ip
	jmp @outer
@tailpad
	%po =l loadl %op
	%eq =l copy 61
	%pad2 =w ceql %pad, 2
	jnz %pad2, @pad_two, @pad_one
@pad_one
	storeb %eq, %po
	%po1 =l add %po, 1
	storel %po1, %op
	jmp @finish
@pad_two
	storeb %eq, %po
	%pp1 =l add %po, 1
	storeb %eq, %pp1
	%po2 =l add %po, 2
	storel %po2, %op
	jmp @finish
@finish
	%fo =l loadl %op
	%len =l sub %fo, %out
	ret %len
}
