export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @goleft %lo1, @goright %lo_c
	%hi =l phi @start %n, @goleft %hi_c, @goright %mid
	%c =w csltl %lo, %hi
	jnz %c, @body, @check
@body
	%d =l sub %hi, %lo
	%h =l sar %d, 1
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%less =w csltl %v, %key
	jnz %less, @goleft, @goright
@goleft
	%lo1 =l add %mid, 1
	%hi_c =l copy %hi
	jmp @loop
@goright
	%lo_c =l copy %lo
	jmp @loop
@check
	%inb =w csltl %lo, %n
	jnz %inb, @fin, @fail
@fin
	%off2 =l mul %lo, 8
	%p2 =l add %a, %off2
	%v2 =l loadl %p2
	%eq =w ceql %v2, %key
	jnz %eq, @found, @fail
@found
	ret %lo
@fail
	ret -1
}
