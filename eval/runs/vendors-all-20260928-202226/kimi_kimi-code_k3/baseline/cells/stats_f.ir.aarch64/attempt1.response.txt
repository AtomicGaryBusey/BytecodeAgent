export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @next %inext
	%min =l phi @start %y0, @next %min1
	%max =l phi @start %y0, @next %max1
	%sum =l phi @start %y0, @next %sum1
	%cnt =l phi @start %odd0, @next %cnt1
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%x =l loadl %addr
	%y =l call %f(l %x)
	%clt =w csltl %y, %min
	jnz %clt, @setmin, @mrgmin
@setmin
	jmp @mrgmin
@mrgmin
	%min1 =l phi @body %min, @setmin %y
	%cgt =w csltl %max, %y
	jnz %cgt, @setmax, @next
@setmax
	jmp @next
@next
	%max1 =l phi @mrgmin %max, @setmax %y
	%sum1 =l add %sum, %y
	%odd =l and %y, 1
	%cnt1 =l add %cnt, %odd
	%inext =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
