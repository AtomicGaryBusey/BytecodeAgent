export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @next %nlo
	%hi =l phi @start %n, @next %nhi
	%cont =w csltl %lo, %hi
	jnz %cont, @body, @final
@body
	%sum =l add %lo, %hi
	%mid =l div %sum, 2
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%lt =w csltl %v, %key
	jnz %lt, @less, @ge
@less
	%mid1 =l add %mid, 1
	jmp @next
@ge
	jmp @next
@next
	%nlo =l phi @less %mid1, @ge %lo
	%nhi =l phi @less %hi, @ge %mid
	jmp @loop
@final
	%in =w csltl %lo, %n
	jnz %in, @check, @notfound
@check
	%off2 =l mul %lo, 8
	%addr2 =l add %a, %off2
	%v2 =l loadl %addr2
	%eq =w ceql %v2, %key
	jnz %eq, @found, @notfound
@found
	ret %lo
@notfound
	ret -1
}
