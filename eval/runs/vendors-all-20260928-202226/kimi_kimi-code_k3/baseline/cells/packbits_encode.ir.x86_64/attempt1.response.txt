export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%w =l copy 0
@loop
	%in =w csltl %i, %n
	jnz %in, @body, @retw
@body
	%pi =l add %src, %i
	%bi =w loadub %pi
	%r =l copy 1
@runcheck
	%ir =l add %i, %r
	%c1 =w csltl %ir, %n
	%c2 =w csltl %r, 128
	%cc =w and %c1, %c2
	jnz %cc, @runcmp, @rundone
@runcmp
	%pr =l add %src, %ir
	%br =w loadub %pr
	%ce =w ceqw %br, %bi
	jnz %ce, @runinc, @rundone
@runinc
	%r =l add %r, 1
	jmp @runcheck
@rundone
	%is3 =w csgel %r, 3
	jnz %is3, @repeat, @lit
@repeat
	%w2 =l add %w, 2
	%fit =w cslel %w2, %cap
	jnz %fit, @repok, @fail
@repok
	%hdr =l sub 257, %r
	%po =l add %out, %w
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %bi, %po1
	%w =l copy %w2
	%i =l add %i, %r
	jmp @loop
@lit
	%j =l copy %i
@litloop
	%jn =w csltl %j, %n
	%len =l sub %j, %i
	%lok =w csltl %len, 128
	%lc =w and %jn, %lok
	jnz %lc, @litcheck, @litdone
@litcheck
	%j2 =l add %j, 2
	%has3 =w csltl %j2, %n
	jnz %has3, @lit3cmp, @litinc
@lit3cmp
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %pj, 1
	%bj1 =w loadub %pj1
	%pj2 =l add %pj, 2
	%bj2 =w loadub %pj2
	%e1 =w ceqw %bj, %bj1
	%e2 =w ceqw %bj, %bj2
	%e3 =w and %e1, %e2
	jnz %e3, @litdone, @litinc
@litinc
	%j =l add %j, 1
	jmp @litloop
@litdone
	%L =l sub %j, %i
	%pkt =l add %L, 1
	%wnew =l add %w, %pkt
	%fit2 =w cslel %wnew, %cap
	jnz %fit2, @litok, @fail
@litok
	%hdrL =l sub %L, 1
	%poL =l add %out, %w
	storeb %hdrL, %poL
	%k =l copy 0
@copyloop
	%ck =w csltl %k, %L
	jnz %ck, @copybody, @copydone
@copybody
	%ps =l add %src, %i
	%psk =l add %ps, %k
	%bk =w loadub %psk
	%pok =l add %poL, 1
	%pokk =l add %pok, %k
	storeb %bk, %pokk
	%k =l add %k, 1
	jmp @copyloop
@copydone
	%w =l copy %wnew
	%i =l copy %j
	jmp @loop
@fail
	ret -1
@retw
	ret %w
}
