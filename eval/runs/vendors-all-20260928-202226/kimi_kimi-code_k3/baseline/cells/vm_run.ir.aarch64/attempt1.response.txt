export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
@loop
	%hit =w ceql %pc, %len
	jnz %hit, @end, @fetch
@fetch
	%pc8 =l mul %pc, 8
	%ca =l add %code, %pc8
	%op =l loadl %ca
	%is0 =w ceql %op, 0
	jnz %is0, @end, @c1
@c1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @c2
@c2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @c3
@c3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @c4
@c4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @c5
@c5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @c6
@c6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @c7
@c7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @c8
@c8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @c9
@c9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @c10
@c10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @c11
@c11
	%is11 =w ceql %op, 11
	jnz %is11, @jmp, @default
@push
	%p1 =l add %pc, 1
	%p18 =l mul %p1, 8
	%pa =l add %code, %p18
	%v =l loadl %pa
	%d8 =l mul %d, 8
	%sa =l add %stack, %d8
	storel %v, %sa
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ba =l add %stack, %dm1_8
	%b =l loadl %ba
	%dm2 =l sub %d, 2
	%dm2_8 =l mul %dm2, 8
	%aa =l add %stack, %dm2_8
	%a =l loadl %aa
	%s =l add %a, %b
	storel %s, %aa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@sub
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ba =l add %stack, %dm1_8
	%b =l loadl %ba
	%dm2 =l sub %d, 2
	%dm2_8 =l mul %dm2, 8
	%aa =l add %stack, %dm2_8
	%a =l loadl %aa
	%s =l sub %a, %b
	storel %s, %aa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@mul
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ba =l add %stack, %dm1_8
	%b =l loadl %ba
	%dm2 =l sub %d, 2
	%dm2_8 =l mul %dm2, 8
	%aa =l add %stack, %dm2_8
	%a =l loadl %aa
	%s =l mul %a, %b
	storel %s, %aa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@dup
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ta =l add %stack, %dm1_8
	%v =l loadl %ta
	%d8 =l mul %d, 8
	%na =l add %stack, %d8
	storel %v, %na
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ba =l add %stack, %dm1_8
	%b =l loadl %ba
	%dm2 =l sub %d, 2
	%dm2_8 =l mul %dm2, 8
	%aa =l add %stack, %dm2_8
	%a =l loadl %aa
	storel %b, %aa
	storel %a, %ba
	%pc =l add %pc, 1
	jmp @loop
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%dm2 =l sub %d, 2
	%dm2_8 =l mul %dm2, 8
	%ua =l add %stack, %dm2_8
	%v =l loadl %ua
	%d8 =l mul %d, 8
	%na =l add %stack, %d8
	storel %v, %na
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@lt
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ba =l add %stack, %dm1_8
	%b =l loadl %ba
	%dm2 =l sub %d, 2
	%dm2_8 =l mul %dm2, 8
	%aa =l add %stack, %dm2_8
	%a =l loadl %aa
	%c =w csltl %a, %b
	%cl =l extsw %c
	storel %cl, %aa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%va =l add %stack, %dm1_8
	%v =l loadl %va
	%d =l sub %d, 1
	%p1 =l add %pc, 1
	%p18 =l mul %p1, 8
	%ta =l add %code, %p18
	%t =l loadl %ta
	%vz =w ceql %v, 0
	jnz %vz, @jztake, @jzno
@jztake
	%pc =l copy %t
	jmp @loop
@jzno
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%p1 =l add %pc, 1
	%p18 =l mul %p1, 8
	%ta =l add %code, %p18
	%t =l loadl %ta
	%pc =l copy %t
	jmp @loop
@default
	%pc =l add %pc, 1
	jmp @loop
@end
	%r =l copy 0
	%empty =w ceql %d, 0
	jnz %empty, @zerofill, @gettop
@gettop
	%dm1 =l sub %d, 1
	%dm1_8 =l mul %dm1, 8
	%ta =l add %stack, %dm1_8
	%r =l loadl %ta
	jmp @zerofill
@zerofill
	%more =w csltl %d, %cap
	jnz %more, @zbody, @ret
@zbody
	%d8 =l mul %d, 8
	%za =l add %stack, %d8
	storel 0, %za
	%d =l add %d, 1
	jmp @zerofill
@ret
	ret %r
}
