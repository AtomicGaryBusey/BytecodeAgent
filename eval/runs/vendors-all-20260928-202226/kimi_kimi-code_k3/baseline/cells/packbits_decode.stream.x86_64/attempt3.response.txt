packbits_decode:
    lea r9, [rdi + rsi]
    lea r11, [rdx + rcx]
    mov r8, rdi
    mov r10, rdx
.Lloop:
    cmp r8, r9
    jae .Ldone
    movsx eax, byte ptr [r8]
    test eax, eax
    js .Lneg
    lea ecx, [eax + 1]
    mov rsi, r9
    sub rsi, r8
    dec rsi
    cmp rsi, rcx
    jl .Lerr2
    lea rsi, [r8 + 1]
    lea r8, [r8 + rcx + 1]
    mov rdi, r11
    sub rdi, r10
    cmp rdi, rcx
    jb .Llitpart
.Llitcopy:
    mov al, byte ptr [rsi]
    mov byte ptr [r10], al
    inc rsi
    inc r10
    dec ecx
    jnz .Llitcopy
    jmp .Lloop
.Llitpart:
    mov ecx, edi
    test ecx, ecx
    jz .Lerr1
.Llitpartcopy:
    mov al, byte ptr [rsi]
    mov byte ptr [r10], al
    inc rsi
    inc r10
    dec ecx
    jnz .Llitpartcopy
    mov rax, -1
    ret
.Lneg:
    cmp eax, -128
    je .Lnop
    lea rsi, [r8 + 1]
    cmp rsi, r9
    jae .Lerr2
    mov ecx, 1
    sub ecx, eax
    mov al, byte ptr [r8 + 1]
    add r8, 2
    mov rdi, r11
    sub rdi, r10
    cmp rdi, rcx
    jb .Lreppart
.Lrepcopy:
    mov byte ptr [r10], al
    inc r10
    dec ecx
    jnz .Lrepcopy
    jmp .Lloop
.Lreppart:
    mov ecx, edi
    test ecx, ecx
    jz .Lerr1
.Lreppartcopy:
    mov byte ptr [r10], al
    inc r10
    dec ecx
    jnz .Lreppartcopy
    mov rax, -1
    ret
.Lnop:
    inc r8
    jmp .Lloop
.Lerr1:
    mov rax, -1
    ret
.Lerr2:
    mov rax, -2
    ret
.Ldone:
    mov rax, r10
    sub rax, rdx
    ret
