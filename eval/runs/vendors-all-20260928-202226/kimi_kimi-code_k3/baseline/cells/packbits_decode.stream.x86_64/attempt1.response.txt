packbits_decode:
    lea r9, [rdi + rsi]
    lea r11, [rdx + rcx]
    mov r8, rdi
    mov r10, rdx
.Lloop:
    cmp r8, r9
    jae .Ldone
    movsx eax, byte [r8]
    inc r8
    test eax, eax
    js .Lneg
    lea ecx, [eax + 1]
    mov rax, r9
    sub rax, r8
    cmp rax, rcx
    jb .Lsrcerr
    mov rax, r11
    sub rax, r10
    cmp rcx, rax
    ja .Llit_over
.Llit_copy:
    mov al, byte [r8]
    mov byte [r10], al
    inc r8
    inc r10
    dec rcx
    jnz .Llit_copy
    jmp .Lloop
.Llit_over:
    test rax, rax
    jz .Loverflow
.Llit_over_copy:
    mov cl, byte [r8]
    mov byte [r10], cl
    inc r8
    inc r10
    dec rax
    jnz .Llit_over_copy
    jmp .Loverflow
.Lneg:
    cmp eax, -128
    je .Lloop
    cmp r8, r9
    jae .Lsrcerr
    mov ecx, 1
    sub ecx, eax
    movzx eax, byte [r8]
    inc r8
    mov rsi, r11
    sub rsi, r10
    cmp rcx, rsi
    ja .Lrep_over
.Lrep_fill:
    mov byte [r10], al
    inc r10
    dec rcx
    jnz .Lrep_fill
    jmp .Lloop
.Lrep_over:
    test rsi, rsi
    jz .Loverflow
.Lrep_over_fill:
    mov byte [r10], al
    inc r10
    dec rsi
    jnz .Lrep_over_fill
.Loverflow:
    mov rax, -1
    ret
.Lsrcerr:
    mov rax, -2
    ret
.Ldone:
    mov rax, r10
    sub rax, rdx
    ret
