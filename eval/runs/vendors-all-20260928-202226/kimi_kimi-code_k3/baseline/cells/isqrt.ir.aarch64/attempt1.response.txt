export function l $isqrt(l %n) {
@start
	%plo =l alloc8 8
	%phi =l alloc8 8
	storel 0, %plo
	storel 4294967296, %phi
@loop
	%lo =l loadl %plo
	%hi =l loadl %phi
	%diff =l sub %hi, %lo
	%cond =w cugtl %diff, 1
	jnz %cond, @body, @done
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	storel %mid, %plo
	jmp @loop
@sethi
	storel %mid, %phi
	jmp @loop
@done
	%r =l loadl %plo
	ret %r
}
