export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%outer0 =w csltl 0, %nm1
	jnz %outer0, @outer, @ret
@outer
	%i =l phi @start 0, @outer_next %inext
	%bound =l sub %nm1, %i
	%inner0 =w csltl 0, %bound
	jnz %inner0, @inner, @outer_next
@inner
	%j =l phi @outer 0, @inner_next %jnext
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csltl %y, %x
	jnz %swap, @swap, @inner_next
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner_next
@inner_next
	%jnext =l add %j, 1
	%innerc =w csltl %jnext, %bound
	jnz %innerc, @inner, @outer_next
@outer_next
	%inext =l add %i, 1
	%outerc =w csltl %inext, %nm1
	jnz %outerc, @outer, @ret
@ret
	ret 0
}
