export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p10lo =l and %p10, 4294967295
	%s0 =l add %p00hi, %p01lo
	%s1 =l add %s0, %p10lo
	%cy =l shr %s1, 32
	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%t0 =l add %p11, %p01hi
	%t1 =l add %t0, %p10hi
	%res =l add %t1, %cy
	ret %res
}
