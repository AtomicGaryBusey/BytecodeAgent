bubble_sort:
	xor eax, eax
	cmp rsi, 2
	jl .Ldone
	lea rcx, [rsi - 1]
.Louter:
	xor edx, edx
.Linner:
	mov r8, qword ptr [rdi + rdx*8]
	mov r9, qword ptr [rdi + rdx*8 + 8]
	cmp r8, r9
	jle .Lnoswap
	mov qword ptr [rdi + rdx*8], r9
	mov qword ptr [rdi + rdx*8 + 8], r8
.Lnoswap:
	inc rdx
	cmp rdx, rcx
	jl .Linner
	dec rcx
	jnz .Louter
.Ldone:
	ret
