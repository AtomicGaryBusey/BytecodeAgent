export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pt =l alloc8 8
	%pit =l alloc8 8
	%psum =l alloc8 8
	storel 0, %pi
@idloop
	%iv =l loadl %pi
	%cid =w csltl %iv, %nn
	jnz %cid, @idbody, @iddone
@idbody
	%row =l div %iv, %n
	%rn =l mul %row, %n
	%col =l sub %iv, %rn
	%eq =w ceql %row, %col
	%val =l extsw %eq
	%off =l mul %iv, 8
	%oa =l add %out, %off
	storel %val, %oa
	%iv1 =l add %iv, 1
	storel %iv1, %pi
	jmp @idloop
@iddone
	storel 0, %pit
@powloop
	%itv =l loadl %pit
	%cp =w csltl %itv, %k
	jnz %cp, @powbody, @powdone
@powbody
	storel 0, %pi
@iloop
	%i2 =l loadl %pi
	%ci =w csltl %i2, %n
	jnz %ci, @ibody, @idone
@ibody
	storel 0, %pj
@jloop
	%j2 =l loadl %pj
	%cj =w csltl %j2, %n
	jnz %cj, @jbody, @jdone
@jbody
	storel 0, %psum
	storel 0, %pt
@tloop
	%t2 =l loadl %pt
	%ct =w csltl %t2, %n
	jnz %ct, @tbody, @tdone
@tbody
	%i3 =l loadl %pi
	%in =l mul %i3, %n
	%t3 =l loadl %pt
	%idx1 =l add %in, %t3
	%o1 =l mul %idx1, 8
	%o2 =l add %out, %o1
	%xv =l loadl %o2
	%tn =l mul %t3, %n
	%j3 =l loadl %pj
	%idx2 =l add %tn, %j3
	%q1 =l mul %idx2, 8
	%q2 =l add %a, %q1
	%yv =l loadl %q2
	%pr =l mul %xv, %yv
	%s0 =l loadl %psum
	%s1 =l add %s0, %pr
	storel %s1, %psum
	%t4 =l add %t3, 1
	storel %t4, %pt
	jmp @tloop
@tdone
	%i4 =l loadl %pi
	%in2 =l mul %i4, %n
	%j4 =l loadl %pj
	%ij =l add %in2, %j4
	%r1 =l mul %ij, 8
	%r2 =l add %tmp, %r1
	%sv =l loadl %psum
	storel %sv, %r2
	%j5 =l add %j4, 1
	storel %j5, %pj
	jmp @jloop
@jdone
	%i5 =l loadl %pi
	%i6 =l add %i5, 1
	storel %i6, %pi
	jmp @iloop
@idone
	storel 0, %pi
@cloop
	%c0 =l loadl %pi
	%cc =w csltl %c0, %nn
	jnz %cc, @cbody, @cdone
@cbody
	%b1 =l mul %c0, 8
	%ta =l add %tmp, %b1
	%ob =l add %out, %b1
	%vv =l loadl %ta
	storel %vv, %ob
	%c1 =l add %c0, 1
	storel %c1, %pi
	jmp @cloop
@cdone
	%it2 =l loadl %pit
	%it3 =l add %it2, 1
	storel %it3, %pit
	jmp @powloop
@powdone
	%nn2 =l mul %nn, 2
	storel 0, %pi
@zloop
	%z0 =l loadl %pi
	%cz =w csltl %z0, %nn2
	jnz %cz, @zbody, @zdone
@zbody
	%z1 =l mul %z0, 8
	%z2 =l add %tmp, %z1
	storel 0, %z2
	%z3 =l add %z0, 1
	storel %z3, %pi
	jmp @zloop
@zdone
	ret 0
}
