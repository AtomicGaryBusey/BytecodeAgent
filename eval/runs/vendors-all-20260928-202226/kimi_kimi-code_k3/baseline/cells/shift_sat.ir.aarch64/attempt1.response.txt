export function l $shift_sat(l %x, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%left =l phi @start %x, @body %left1
	%right =l phi @start %x, @body %right1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%left1 =l shl %left, 1
	%right1 =l sar %right, 1
	%i1 =l add %i, 1
	jmp @loop
@done
	%res =l add %left, %right
	ret %res
}
