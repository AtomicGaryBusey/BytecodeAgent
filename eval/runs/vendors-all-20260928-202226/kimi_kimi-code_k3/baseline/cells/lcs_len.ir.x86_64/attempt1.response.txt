export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zloop
@zloop
	%jz =l phi @start 0, @zbody %jz1
	%zc =w cslel %jz, %n
	jnz %zc, @zbody, @zdone
@zbody
	%zoff =l mul %jz, 8
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%jz1 =l add %jz, 1
	jmp @zloop
@zdone
	jmp @iloop
@iloop
	%i =l phi @zdone 0, @jdone %i1
	%ic =w csltl %i, %m
	jnz %ic, @ibody, @idone
@ibody
	%iptr =l add %x, %i
	%xb =w loadub %iptr
	jmp @jloop
@jloop
	%j =l phi @ibody 1, @store %j1
	%diag =l phi @ibody 0, @store %up
	%jc =w cslel %j, %n
	jnz %jc, @jbody, @jdone
@jbody
	%joff =l mul %j, 8
	%uptr =l add %row, %joff
	%up =l loadl %uptr
	%jm1 =l sub %j, 1
	%yptr =l add %y, %jm1
	%yb =w loadub %yptr
	%match =w ceql %xb, %yb
	jnz %match, @match, @nomatch
@match
	%mv =l add %diag, 1
	jmp @store
@nomatch
	%lm1off =l mul %jm1, 8
	%lptr =l add %row, %lm1off
	%left =l loadl %lptr
	%gt =w csgtl %left, %up
	jnz %gt, @takeleft, @store
@takeleft
	jmp @store
@store
	%val =l phi @match %mv, @takeleft %left, @nomatch %up
	storel %val, %uptr
	%j1 =l add %j, 1
	jmp @jloop
@jdone
	%i1 =l add %i, 1
	jmp @iloop
@idone
	%noff =l mul %n, 8
	%nptr =l add %row, %noff
	%res =l loadl %nptr
	ret %res
}
