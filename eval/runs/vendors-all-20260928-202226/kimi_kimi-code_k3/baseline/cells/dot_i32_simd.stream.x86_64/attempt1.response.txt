dot_i32_simd:
xor eax, eax
test rdx, rdx
jle .Ldone
mov rcx, rdx
shr rcx, 2
jz .Ltail_setup
pxor xmm8, xmm8
pxor xmm9, xmm9
.Lvec_loop:
movdqu xmm0, xmmword ptr [rdi]
movdqu xmm1, xmmword ptr [rsi]
movdqa xmm2, xmm0
psrldq xmm2, 8
movdqa xmm3, xmm1
psrldq xmm3, 8
pmovsxdq xmm0, xmm0
pmovsxdq xmm2, xmm2
pmovsxdq xmm1, xmm1
pmovsxdq xmm3, xmm3
pmuldq xmm0, xmm1
pmuldq xmm2, xmm3
paddq xmm8, xmm0
paddq xmm9, xmm2
add rdi, 16
add rsi, 16
dec rcx
jnz .Lvec_loop
paddq xmm8, xmm9
movdqa xmm0, xmm8
psrldq xmm0, 8
paddq xmm8, xmm0
movq rax, xmm8
.Ltail_setup:
and rdx, 3
jz .Ldone
.Lscalar_loop:
movsxd r8, dword ptr [rdi]
movsxd r9, dword ptr [rsi]
imul r8, r9
add rax, r8
add rdi, 4
add rsi, 4
dec rdx
jnz .Lscalar_loop
.Ldone:
ret
