export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%zero =l copy 0
	%min =l copy 0x8000000000000000
	%mone =l copy -1
	%bz =w ceql %b, %zero
	jnz %bz, @bzero, @check
@bzero
	storel %a, %rem
	ret %zero
@check
	%amin =w ceql %a, %min
	%bmone =w ceql %b, %mone
	%ovf =w and %amin, %bmone
	jnz %ovf, @overflow, @normal
@overflow
	storel %zero, %rem
	ret %min
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
