map_into:
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 16
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rcx
    xor r15d, r15d
    test r13, r13
    jle .Ldone
.Lloop:
    mov rax, qword ptr [r12]
    mov qword ptr [rsp], rax
    mov rdi, rax
    call r14
    mov qword ptr [rbx], rax
    cmp rax, qword ptr [rsp]
    jge .Lskip_inc
    inc r15
.Lskip_inc:
    add rbx, 8
    add r12, 8
    dec r13
    jne .Lloop
.Ldone:
    mov rax, r15
    add rsp, 16
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
