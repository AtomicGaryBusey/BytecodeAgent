export function l $count_desc(l %a, l %n) {
@entry
	%lt2 =w csltl %n, 2
	jnz %lt2, @zero, @init
@zero
	ret 0
@init
	%last =l sub %n, 1
	%i =l copy 0
	%cnt =l copy 0
@loop
	%more =w csltl %i, %last
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%gt =w csltl %y, %x
	%gtl =l extsw %gt
	%cnt =l add %cnt, %gtl
	%i =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
