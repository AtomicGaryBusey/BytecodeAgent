export function l $count_desc(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	%r0 =l copy 0
	ret %r0
@init
	%limit =l sub %n, 1
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @cond
@cond
	%i =l phi @init %i0, @body %i1
	%cnt =l phi @init %c0, @body %c1
	%go =w csltl %i, %limit
	jnz %go, @body, @done
@body
	%off =l mul %i, 8
	%p0 =l add %a, %off
	%v0 =l loadl %p0
	%p1 =l add %p0, 8
	%v1 =l loadl %p1
	%gt =w csltl %v1, %v0
	%inc =l extsw %gt
	%c1 =l add %cnt, %inc
	%i1 =l add %i, 1
	jmp @cond
@done
	ret %cnt
}
