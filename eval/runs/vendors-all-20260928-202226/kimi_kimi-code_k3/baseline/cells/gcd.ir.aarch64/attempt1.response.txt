export function l $gcd(l %a0, l %b0) {
@entry
	%pa =l alloc8 16
	%pb =l add %pa, 8
	storel %a0, %pa
	storel %b0, %pb
@loop
	%a =l loadl %pa
	%b =l loadl %pb
	%z =w ceql %b, 0
	jnz %z, @done, @body
@body
	%r =l rem %a, %b
	storel %b, %pa
	storel %r, %pb
	jmp @loop
@done
	%res =l loadl %pa
	ret %res
}
