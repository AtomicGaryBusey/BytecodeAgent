export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%z =w ceql %b, 0
	jnz %z, @divzero, @chkneg
@chkneg
	%n =w ceql %b, -1
	jnz %n, @chkmin, @normal
@chkmin
	%m =w ceql %a, -9223372036854775808
	jnz %m, @overflow, @normal
@overflow
	storel 0, %rem
	ret %a
@divzero
	storel %a, %rem
	ret 0
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
