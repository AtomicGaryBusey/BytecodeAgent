export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%pi =l alloc8 8
	%pmin =l alloc8 8
	%pmax =l alloc8 8
	%psum =l alloc8 8
	%pcnt =l alloc8 8
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	storel 1, %pi
	storel %y0, %pmin
	storel %y0, %pmax
	storel %y0, %psum
	%b0 =l and %y0, 1
	storel %b0, %pcnt
	jmp @loop
@loop
	%i =l loadl %pi
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%s =l loadl %psum
	%ns =l add %s, %y
	storel %ns, %psum
	%mn =l loadl %pmin
	%ltm =w csltl %y, %mn
	jnz %ltm, @setmin, @chkmax
@setmin
	storel %y, %pmin
	jmp @chkmax
@chkmax
	%mx =l loadl %pmax
	%gtx =w csltl %mx, %y
	jnz %gtx, @setmax, @count
@setmax
	storel %y, %pmax
	jmp @count
@count
	%odd =l and %y, 1
	%c =l loadl %pcnt
	%nc =l add %c, %odd
	storel %nc, %pcnt
	%i2 =l loadl %pi
	%i3 =l add %i2, 1
	storel %i3, %pi
	jmp @loop
@end
	%fmin =l loadl %pmin
	storel %fmin, %out
	%fmax =l loadl %pmax
	%out1 =l add %out, 8
	storel %fmax, %out1
	%fsum =l loadl %psum
	%out2 =l add %out, 16
	storel %fsum, %out2
	%r =l loadl %pcnt
	ret %r
}
