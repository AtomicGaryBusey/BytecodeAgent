export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%frame =l alloc16 32
	%wrp =l add %frame, 8
	%idxp =l add %frame, 16
	%cntp =l add %frame, 24
	storel 0, %frame
	storel 0, %wrp
@loop
	%pos =l loadl %frame
	%done =w csgel %pos, %n
	jnz %done, @end, @read
@read
	%hp =l add %src, %pos
	%h =w loadsb %hp
	%pos1 =l add %pos, 1
	storel %pos1, %frame
	%nop =w ceqw %h, -128
	jnz %nop, @loop, @class
@class
	%neg =w csltw %h, 0
	jnz %neg, @repeat, @literal
@literal
	%cw =w add %h, 1
	%cnt =l extuw %cw
	%rem =l sub %n, %pos1
	%badsrc =w csltl %rem, %cnt
	jnz %badsrc, @errsrc, @litinit
@litinit
	storel %cnt, %cntp
	storel 0, %idxp
	jmp @litloop
@litloop
	%k =l loadl %idxp
	%c =l loadl %cntp
	%kdone =w csgel %k, %c
	jnz %kdone, @litend, @litbody
@litbody
	%wr =l loadl %wrp
	%ok =w csltl %wr, %cap
	jnz %ok, @litstore, @errcap
@litstore
	%base =l loadl %frame
	%sp1 =l add %src, %base
	%sp2 =l add %sp1, %k
	%b =w loadub %sp2
	%op =l add %out, %wr
	storeb %b, %op
	%wr1 =l add %wr, 1
	storel %wr1, %wrp
	%k1 =l add %k, 1
	storel %k1, %idxp
	jmp @litloop
@litend
	%pbase =l loadl %frame
	%cc =l loadl %cntp
	%pn =l add %pbase, %cc
	storel %pn, %frame
	jmp @loop
@repeat
	%remr =l sub %n, %pos1
	%badr =w csltl %remr, 1
	jnz %badr, @errsrc, @repinit
@repinit
	%negw =w sub 0, %h
	%runw =w add %negw, 1
	%run =l extuw %runw
	storel %run, %cntp
	storel 0, %idxp
	jmp @reploop
@reploop
	%rk =l loadl %idxp
	%rc =l loadl %cntp
	%rdone =w csgel %rk, %rc
	jnz %rdone, @repend, @repbody
@repbody
	%rwr =l loadl %wrp
	%rok =w csltl %rwr, %cap
	jnz %rok, @repstore, @errcap
@repstore
	%rbase =l loadl %frame
	%rsp =l add %src, %rbase
	%rb =w loadub %rsp
	%rop =l add %out, %rwr
	storeb %rb, %rop
	%rwr1 =l add %rwr, 1
	storel %rwr1, %wrp
	%rk1 =l add %rk, 1
	storel %rk1, %idxp
	jmp @reploop
@repend
	%rbase2 =l loadl %frame
	%rpn =l add %rbase2, 1
	storel %rpn, %frame
	jmp @loop
@end
	%ret =l loadl %wrp
	ret %ret
@errcap
	ret -1
@errsrc
	ret -2
}
