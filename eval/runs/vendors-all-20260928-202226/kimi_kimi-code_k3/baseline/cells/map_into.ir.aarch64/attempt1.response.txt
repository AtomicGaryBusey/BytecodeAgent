export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @loop
@loop
	%i =l phi @entry %i0, @body %i1
	%cnt =l phi @entry %c0, @body %c1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%v =l loadl %sa
	%r =l call %f(l %v)
	%da =l add %dst, %off
	storel %r, %da
	%lt =w csltl %r, %v
	%ltx =l extsw %lt
	%c1 =l add %cnt, %ltx
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
