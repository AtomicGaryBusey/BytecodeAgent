export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%atend =w ceql %pc, %len
	jnz %atend, @fin, @fetch
@fetch
	%pc8 =l mul %pc, 8
	%cp =l add %code, %pc8
	%op =l loadl %cp
	%eq0 =w ceql %op, 0
	jnz %eq0, @fin, @t1
@t1
	%eq1 =w ceql %op, 1
	jnz %eq1, @push, @t2
@t2
	%eq2 =w ceql %op, 2
	jnz %eq2, @add, @t3
@t3
	%eq3 =w ceql %op, 3
	jnz %eq3, @sub, @t4
@t4
	%eq4 =w ceql %op, 4
	jnz %eq4, @mul, @t5
@t5
	%eq5 =w ceql %op, 5
	jnz %eq5, @dup, @t6
@t6
	%eq6 =w ceql %op, 6
	jnz %eq6, @swap, @t7
@t7
	%eq7 =w ceql %op, 7
	jnz %eq7, @drop, @t8
@t8
	%eq8 =w ceql %op, 8
	jnz %eq8, @over, @t9
@t9
	%eq9 =w ceql %op, 9
	jnz %eq9, @lt, @t10
@t10
	%eq10 =w ceql %op, 10
	jnz %eq10, @jz, @t11
@t11
	%eq11 =w ceql %op, 11
	jnz %eq11, @jmpop, @other
@other
	%pc =l add %pc, 1
	jmp @loop
@push
	%pc1 =l add %pc, 1
	%pc18 =l mul %pc1, 8
	%vp =l add %code, %pc18
	%v =l loadl %vp
	%d8 =l mul %d, 8
	%sp =l add %stack, %d8
	storel %v, %sp
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%dm1 =l sub %d, 1
	%dm2 =l sub %d, 2
	%o1 =l mul %dm1, 8
	%pb =l add %stack, %o1
	%b =l loadl %pb
	%o2 =l mul %dm2, 8
	%pa =l add %stack, %o2
	%a =l loadl %pa
	%r =l add %a, %b
	storel %r, %pa
	%d =l copy %dm1
	%pc =l add %pc, 1
	jmp @loop
@sub
	%dm1 =l sub %d, 1
	%dm2 =l sub %d, 2
	%o1 =l mul %dm1, 8
	%pb =l add %stack, %o1
	%b =l loadl %pb
	%o2 =l mul %dm2, 8
	%pa =l add %stack, %o2
	%a =l loadl %pa
	%r =l sub %a, %b
	storel %r, %pa
	%d =l copy %dm1
	%pc =l add %pc, 1
	jmp @loop
@mul
	%dm1 =l sub %d, 1
	%dm2 =l sub %d, 2
	%o1 =l mul %dm1, 8
	%pb =l add %stack, %o1
	%b =l loadl %pb
	%o2 =l mul %dm2, 8
	%pa =l add %stack, %o2
	%a =l loadl %pa
	%r =l mul %a, %b
	storel %r, %pa
	%d =l copy %dm1
	%pc =l add %pc, 1
	jmp @loop
@dup
	%dm1 =l sub %d, 1
	%o =l mul %dm1, 8
	%pt =l add %stack, %o
	%v =l loadl %pt
	%d8 =l mul %d, 8
	%pd =l add %stack, %d8
	storel %v, %pd
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%dm1 =l sub %d, 1
	%dm2 =l sub %d, 2
	%o1 =l mul %dm1, 8
	%p1 =l add %stack, %o1
	%o2 =l mul %dm2, 8
	%p2 =l add %stack, %o2
	%b =l loadl %p1
	%a =l loadl %p2
	storel %b, %p2
	storel %a, %p1
	%pc =l add %pc, 1
	jmp @loop
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%dm2 =l sub %d, 2
	%o =l mul %dm2, 8
	%ps =l add %stack, %o
	%v =l loadl %ps
	%d8 =l mul %d, 8
	%pd =l add %stack, %d8
	storel %v, %pd
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@lt
	%dm1 =l sub %d, 1
	%dm2 =l sub %d, 2
	%o1 =l mul %dm1, 8
	%pb =l add %stack, %o1
	%b =l loadl %pb
	%o2 =l mul %dm2, 8
	%pa =l add %stack, %o2
	%a =l loadl %pa
	%rw =w csltl %a, %b
	%r =l extsw %rw
	storel %r, %pa
	%d =l copy %dm1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%pc1 =l add %pc, 1
	%pc18 =l mul %pc1, 8
	%tp =l add %code, %pc18
	%t =l loadl %tp
	%dm1 =l sub %d, 1
	%o =l mul %dm1, 8
	%pv =l add %stack, %o
	%v =l loadl %pv
	%d =l copy %dm1
	%is0 =w ceql %v, 0
	jnz %is0, @jzt, @jzf
@jzt
	%pc =l copy %t
	jmp @loop
@jzf
	%pc =l add %pc, 2
	jmp @loop
@jmpop
	%pc1 =l add %pc, 1
	%pc18 =l mul %pc1, 8
	%tp =l add %code, %pc18
	%pc =l loadl %tp
	jmp @loop
@fin
	%i =l copy %d
	jmp @zchk
@zchk
	%zd =w csgel %i, %cap
	jnz %zd, @ret, @zbody
@zbody
	%o =l mul %i, 8
	%zp =l add %stack, %o
	storel 0, %zp
	%i =l add %i, 1
	jmp @zchk
@ret
	%empty =w ceql %d, 0
	jnz %empty, @ret0, @retv
@ret0
	ret 0
@retv
	%dm1 =l sub %d, 1
	%o =l mul %dm1, 8
	%pv =l add %stack, %o
	%rv =l loadl %pv
	ret %rv
}
