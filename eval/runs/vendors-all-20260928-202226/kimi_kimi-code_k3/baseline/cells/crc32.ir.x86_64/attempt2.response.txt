export function l $crc32(l %p, l %n) {
@start
	%crc =w copy -1
	%i =l copy 0
@byteloop
	%go =w csltl %i, %n
	jnz %go, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%k =w copy 0
@bitloop
	%kgo =w csltw %k, 8
	jnz %kgo, @bitbody, @bytenext
@bitbody
	%lsb =w and %crc, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, -305914880
	%sh =w shr %crc, 1
	%crc =w xor %sh, %poly
	%k =w add %k, 1
	jmp @bitloop
@bytenext
	%i =l add %i, 1
	jmp @byteloop
@done
	%res =w xor %crc, -1
	%ret =l extuw %res
	ret %ret
}
