export function l $crc32(l %p, l %n) {
@start
	jmp @byteloop
@byteloop
	%i =l phi @start 0, @bytenext %inext
	%crc =w phi @start -1, @bytenext %crcnext
	%go =w csltl %i, %n
	jnz %go, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crcb =w xor %crc, %b
	jmp @bitloop
@bitloop
	%k =w phi @body 0, @bitbody %knext
	%crck =w phi @body %crcb, @bitbody %crcknext
	%kgo =w csltw %k, 8
	jnz %kgo, @bitbody, @bytenext
@bitbody
	%lsb =w and %crck, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, -306674912
	%sh =w shr %crck, 1
	%crcknext =w xor %sh, %poly
	%knext =w add %k, 1
	jmp @bitloop
@bytenext
	%crcnext =w copy %crck
	%inext =l add %i, 1
	jmp @byteloop
@done
	%res =w xor %crc, -1
	%ret =l extuw %res
	ret %ret
}
