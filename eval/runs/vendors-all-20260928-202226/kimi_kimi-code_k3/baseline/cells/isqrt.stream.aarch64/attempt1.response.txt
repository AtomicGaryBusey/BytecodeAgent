isqrt:
    mov x1, #0
    mov x2, #1
    lsl x2, x2, #62
    mov x4, #32
.Lloop:
    cmp x0, x2
    blo .Lskip
    sub x5, x0, x2
    cmp x5, x1
    blo .Lskip
    sub x0, x5, x1
    lsr x1, x1, #1
    add x1, x1, x2
    b .Lnext
.Lskip:
    lsr x1, x1, #1
.Lnext:
    lsr x2, x2, #2
    subs x4, x4, #1
    bne .Lloop
    mov x0, x1
    ret
