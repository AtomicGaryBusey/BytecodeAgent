export function l $fib(l %n) {
@start
	%i =l alloc8 8
	%a =l alloc8 8
	%b =l alloc8 8
	storel 0, %i
	storel 0, %a
	storel 1, %b
@loop
	%iv =l loadl %i
	%cond =w csltl %iv, %n
	jnz %cond, @body, @done
@body
	%av =l loadl %a
	%bv =l loadl %b
	%t =l add %av, %bv
	storel %bv, %a
	storel %t, %b
	%in =l add %iv, 1
	storel %in, %i
	jmp @loop
@done
	%r =l loadl %a
	ret %r
}
