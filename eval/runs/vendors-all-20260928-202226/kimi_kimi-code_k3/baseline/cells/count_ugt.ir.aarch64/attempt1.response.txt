export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%cnt =l copy 0
	%one =l copy 1
	%eight =l copy 8
@loop
	%in =w csltl %i, %n
	jnz %in, @body, @done
@body
	%off =l mul %i, %eight
	%addr =l add %a, %off
	%v =l loadl %addr
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%cnt =l add %cnt, %gtl
	%i =l add %i, %one
	jmp @loop
@done
	ret %cnt
}
