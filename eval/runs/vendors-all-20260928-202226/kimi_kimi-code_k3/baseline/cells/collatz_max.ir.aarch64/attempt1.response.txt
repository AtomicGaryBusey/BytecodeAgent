export function l $collatz_max(l %n) {
@start
	%maxp =l alloc8 8
	%kp =l alloc8 8
	%xp =l alloc8 8
	%stepp =l alloc8 8
	storel 0, %maxp
	storel 1, %kp
@outer
	%k =l loadl %kp
	%done =w csgtl %k, %n
	jnz %done, @ret, @init
@init
	storel %k, %xp
	storel 0, %stepp
@inner
	%x =l loadl %xp
	%one =w ceql %x, 1
	jnz %one, @checkmax, @step
@step
	%low =l and %x, 1
	%odd =w ceql %low, 1
	jnz %odd, @odd, @even
@even
	%nx0 =l div %x, 2
	storel %nx0, %xp
	jmp @inc
@odd
	%t3 =l mul %x, 3
	%nx1 =l add %t3, 1
	storel %nx1, %xp
	jmp @inc
@inc
	%s0 =l loadl %stepp
	%s1 =l add %s0, 1
	storel %s1, %stepp
	jmp @inner
@checkmax
	%steps =l loadl %stepp
	%max =l loadl %maxp
	%gt =w csgtl %steps, %max
	jnz %gt, @setmax, @next
@setmax
	storel %steps, %maxp
	jmp @next
@next
	%k2 =l add %k, 1
	storel %k2, %kp
	jmp @outer
@ret
	%res =l loadl %maxp
	ret %res
}
