export function l $map_sum(l %a, l %n, l %f) {
@start
	%ip =l alloc8 8
	%sp =l alloc8 8
	storel 0, %ip
	storel 0, %sp
@loop
	%i =l loadl %ip
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum =l loadl %sp
	%sum2 =l add %sum, %r
	storel %sum2, %sp
	%i2 =l add %i, 1
	storel %i2, %ip
	jmp @loop
@done
	%res =l loadl %sp
	ret %res
}
