export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	jmp @zloop
@zloop
	%zi =l phi @start 0, @zbody %zinext
	%zcond =w cultl %zi, %total
	jnz %zcond, @zbody, @oloop
@zbody
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zinext =l add %zi, 1
	jmp @zloop
@oloop
	%i =l phi @zloop 0, @jdone %inext
	%icond =w cultl %i, %na
	jnz %icond, @ibody, @kloop
@ibody
	%ioff =l mul %i, 8
	%aaddr =l add %a, %ioff
	%ai =l loadl %aaddr
	%ailo =l and %ai, 4294967295
	%aihi =l shr %ai, 32
	jmp @jloop
@jloop
	%j =l phi @ibody 0, @jbody %jnext
	%carry =l phi @ibody 0, @jbody %carrynext
	%jcond =w cultl %j, %nb
	jnz %jcond, @jbody, @jdone
@jbody
	%joff =l mul %j, 8
	%baddr =l add %b, %joff
	%bi =l loadl %baddr
	%bilo =l and %bi, 4294967295
	%bihi =l shr %bi, 32
	%p0 =l mul %ailo, %bilo
	%p1 =l mul %ailo, %bihi
	%p2 =l mul %aihi, %bilo
	%p3 =l mul %aihi, %bihi
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, 4294967295
	%p2lo =l and %p2, 4294967295
	%t0 =l add %p0hi, %p1lo
	%t =l add %t0, %p2lo
	%tsh =l shl %t, 32
	%p0lo =l and %p0, 4294967295
	%low =l or %p0lo, %tsh
	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%thi =l shr %t, 32
	%h0 =l add %p3, %p1hi
	%h1 =l add %h0, %p2hi
	%high =l add %h1, %thi
	%ij =l add %i, %j
	%ijoff =l mul %ij, 8
	%raddr =l add %r, %ijoff
	%rv =l loadl %raddr
	%s1 =l add %low, %rv
	%c1w =w cultl %s1, %low
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %raddr
	%hc =l add %high, %c1
	%carrynext =l add %hc, %c2
	%jnext =l add %j, 1
	jmp @jloop
@jdone
	%inb =l add %i, %nb
	%inboff =l mul %inb, 8
	%rend =l add %r, %inboff
	storel %carry, %rend
	%inext =l add %i, 1
	jmp @oloop
@kloop
	%k =l phi @oloop %total, @kbody %kdec
	%kpos =w cultl 0, %k
	jnz %kpos, @kcheck, @ret
@kcheck
	%km1 =l sub %k, 1
	%koff =l mul %km1, 8
	%kaddr =l add %r, %koff
	%kv =l loadl %kaddr
	%kz =w ceql %kv, 0
	jnz %kz, @kbody, @ret
@kbody
	%kdec =l sub %k, 1
	jmp @kloop
@ret
	ret %k
}
