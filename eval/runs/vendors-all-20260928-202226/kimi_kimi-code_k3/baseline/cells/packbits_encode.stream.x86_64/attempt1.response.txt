packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Louter:
    cmp r8, rsi
    jge .Lret_w
    movzx r11d, byte ptr [rdi + r8]
    lea r10, [r8 + 1]
    mov eax, 1
.Lrun_i_loop:
    cmp r10, rsi
    jge .Lrun_i_done
    cmp eax, 128
    jge .Lrun_i_done
    cmp byte ptr [rdi + r10], r11b
    jne .Lrun_i_done
    inc eax
    inc r10
    jmp .Lrun_i_loop
.Lrun_i_done:
    cmp eax, 3
    jl .Lliteral
    lea r10, [r9 + 2]
    cmp rcx, r10
    jl .Lfail
    mov r10d, 257
    sub r10d, eax
    mov byte ptr [rdx + r9], r10b
    mov byte ptr [rdx + r9 + 1], r11b
    add r9, 2
    mov r10d, eax
    add r8, r10
    jmp .Louter
.Lliteral:
    mov r10, r8
.Llit_scan:
    cmp r10, rsi
    jge .Llit_done
    mov rax, r10
    sub rax, r8
    cmp rax, 128
    jge .Llit_done
    lea r11, [r10 + 2]
    cmp r11, rsi
    jge .Llit_advance
    mov al, byte ptr [rdi + r10]
    cmp al, byte ptr [rdi + r10 + 1]
    jne .Llit_advance
    cmp al, byte ptr [rdi + r10 + 2]
    jne .Llit_advance
    jmp .Llit_done
.Llit_advance:
    inc r10
    jmp .Llit_scan
.Llit_done:
    mov rax, r10
    sub rax, r8
    lea r11, [r9 + rax + 1]
    cmp rcx, r11
    jl .Lfail
    mov r11d, eax
    dec r11d
    mov byte ptr [rdx + r9], r11b
    lea r11, [rdi + r8]
    mov r8, r10
    add r9, rax
    inc r9
    lea r10, [rdx + r9]
    sub r10, rax
    push rcx
.Llit_copy:
    test rax, rax
    jle .Llit_copy_done
    mov cl, byte ptr [r11]
    mov byte ptr [r10], cl
    inc r11
    inc r10
    dec rax
    jmp .Llit_copy
.Llit_copy_done:
    pop rcx
    jmp .Louter
.Lret_w:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
