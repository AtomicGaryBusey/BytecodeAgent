export function l $map_sum(l %a, l %n, l %f) {
@entry
	%sum0 =l copy 0
	%i0 =l copy 0
	jmp @loop
@loop
	%sum =l phi @entry %sum0, @body %sum2
	%i =l phi @entry %i0, @body %i2
	%done =w ceql %i, %n
	jnz %done, @ret, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum2 =l add %sum, %r
	%i2 =l add %i, 1
	jmp @loop
@ret
	ret %sum
}
