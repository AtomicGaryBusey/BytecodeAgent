qsort_pairs:
cmp rsi, 2
jl .Lret
push rbx
push rbp
push r12
push r13
push r14
mov rbx, rdi
mov r12, rsi
mov rax, r12
shr rax, 1
lea rax, [rax + rax]
mov r8, qword ptr [rbx + rax*8]
mov r9, qword ptr [rbx + rax*8 + 8]
xor r13d, r13d
lea r14, [r12 - 1]
.Lpart_outer:
lea rcx, [r13 + r13]
mov rax, qword ptr [rbx + rcx*8]
mov rdx, qword ptr [rbx + rcx*8 + 8]
cmp rax, r8
jl .Ladv_i
jg .Lscan_j
cmp rdx, r9
jge .Lscan_j
.Ladv_i:
inc r13
jmp .Lpart_outer
.Lscan_j:
lea rcx, [r14 + r14]
mov rax, qword ptr [rbx + rcx*8]
mov rdx, qword ptr [rbx + rcx*8 + 8]
cmp rax, r8
jg .Ladv_j
jl .Lcheck_swap
cmp rdx, r9
jle .Lcheck_swap
.Ladv_j:
dec r14
jmp .Lscan_j
.Lcheck_swap:
cmp r13, r14
jg .Lpart_done
lea rcx, [r13 + r13]
lea rdx, [r14 + r14]
mov rax, qword ptr [rbx + rcx*8]
mov r10, qword ptr [rbx + rcx*8 + 8]
mov r11, qword ptr [rbx + rdx*8]
mov rbp, qword ptr [rbx + rdx*8 + 8]
mov qword ptr [rbx + rcx*8], r11
mov qword ptr [rbx + rcx*8 + 8], rbp
mov qword ptr [rbx + rdx*8], rax
mov qword ptr [rbx + rdx*8 + 8], r10
inc r13
dec r14
cmp r13, r14
jle .Lpart_outer
.Lpart_done:
cmp r14, 1
jl .Lskip_left
mov rdi, rbx
lea rsi, [r14 + 1]
call qsort_pairs
.Lskip_left:
mov rsi, r12
sub rsi, r13
cmp rsi, 2
jl .Lskip_right
lea rdi, [r13 + r13]
lea rdi, [rbx + rdi*8]
call qsort_pairs
.Lskip_right:
pop r14
pop r13
pop r12
pop rbp
pop rbx
.Lret:
xor eax, eax
ret
