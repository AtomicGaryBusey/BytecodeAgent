export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%i_lt_n =w csltl %i, %n
	jnz %i_lt_n, @have, @done
@have
	%pi =l add %src, %i
	%bi =w loadub %pi
	%i2 =l add %i, 2
	%has3 =w csltl %i2, %n
	jnz %has3, @check3, @literal
@check3
	%p1 =l add %pi, 1
	%b1 =w loadub %p1
	%eq1 =w ceqw %b1, %bi
	jnz %eq1, @check3b, @literal
@check3b
	%p2 =l add %pi, 2
	%b2 =w loadub %p2
	%eq2 =w ceqw %b2, %bi
	jnz %eq2, @repeat, @literal
@repeat
	%r =w copy 1
@rloop
	%rl =l extsw %r
	%idx =l add %i, %rl
	%idxok =w csltl %idx, %n
	jnz %idxok, @rmax, @rdone
@rmax
	%rok =w csltw %r, 128
	jnz %rok, @rcmp, @rdone
@rcmp
	%pr =l add %src, %idx
	%br =w loadub %pr
	%eqr =w ceqw %br, %bi
	jnz %eqr, @rinc, @rdone
@rinc
	%r =w add %r, 1
	jmp @rloop
@rdone
	%rem =l sub %cap, %o
	%nofit =w csltl %rem, 2
	jnz %nofit, @fail, @rstore
@rstore
	%c257 =w copy 257
	%hdr =w sub %c257, %r
	%po =l add %out, %o
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %bi, %po1
	%o =l add %o, 2
	%irl =l extsw %r
	%i =l add %i, %irl
	jmp @loop
@literal
	%j =l copy %i
	%Lw =w copy 0
@lloop
	%jok =w csltl %j, %n
	jnz %jok, @llen, @ldone
@llen
	%lenok =w csltw %Lw, 128
	jnz %lenok, @ltriple, @ldone
@ltriple
	%j2 =l add %j, 2
	%jhas3 =w csltl %j2, %n
	jnz %jhas3, @lcmp, @laccept
@lcmp
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %pj, 1
	%bj1 =w loadub %pj1
	%jeq1 =w ceqw %bj1, %bj
	jnz %jeq1, @lcmp2, @laccept
@lcmp2
	%pj2 =l add %pj, 2
	%bj2 =w loadub %pj2
	%jeq2 =w ceqw %bj2, %bj
	jnz %jeq2, @ldone, @laccept
@laccept
	%j =l add %j, 1
	%Lw =w add %Lw, 1
	jmp @lloop
@ldone
	%Lwl =l extsw %Lw
	%need =l add %Lwl, 1
	%rem2 =l sub %cap, %o
	%nofit2 =w csltl %rem2, %need
	jnz %nofit2, @fail, @lstore
@lstore
	%lhdr =w sub %Lw, 1
	%po2 =l add %out, %o
	storeb %lhdr, %po2
	%dst =l add %po2, 1
	%k =w copy 0
@copyloop
	%klt =w csltw %k, %Lw
	jnz %klt, @copybody, @copydone
@copybody
	%kl =l extsw %k
	%sp =l add %pi, %kl
	%sb =w loadub %sp
	%dp =l add %dst, %kl
	storeb %sb, %dp
	%k =w add %k, 1
	jmp @copyloop
@copydone
	%adv =l add %Lwl, 1
	%o =l add %o, %adv
	%i =l copy %j
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
