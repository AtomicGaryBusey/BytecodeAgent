export function l $collatz_max(l %n) {
@start
	%best =l copy 0
	%k =l copy 1
@outer
	%c =w cslel %k, %n
	jnz %c, @body, @done
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%isone =w ceql %x, 1
	jnz %isone, @endk, @step
@step
	%r =l and %x, 1
	jnz %r, @odd, @even
@even
	%x =l shr %x, 1
	jmp @cont
@odd
	%t =l mul %x, 3
	%x =l add %t, 1
	jmp @cont
@cont
	%steps =l add %steps, 1
	jmp @inner
@endk
	%gt =w csgtl %steps, %best
	jnz %gt, @upd, @next
@upd
	%best =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %best
}
