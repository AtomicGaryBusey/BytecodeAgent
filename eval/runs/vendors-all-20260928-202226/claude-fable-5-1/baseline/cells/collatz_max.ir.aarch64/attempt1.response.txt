export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%best =l copy 0
@outer
	%c =w cslel %k, %n
	jnz %c, @body, @done
@body
	%x =l copy %k
	%steps =l copy 0
@inner
	%one =w ceql %x, 1
	jnz %one, @inner_done, @step
@step
	%odd =l and %x, 1
	jnz %odd, @isodd, @iseven
@isodd
	%t =l mul %x, 3
	%x =l add %t, 1
	jmp @cont
@iseven
	%x =l shr %x, 1
	jmp @cont
@cont
	%steps =l add %steps, 1
	jmp @inner
@inner_done
	%gt =w csgtl %steps, %best
	jnz %gt, @upd, @next
@upd
	%best =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %best
}
