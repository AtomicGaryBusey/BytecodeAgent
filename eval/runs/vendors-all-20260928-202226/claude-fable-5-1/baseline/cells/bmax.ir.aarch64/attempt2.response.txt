export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%i0 =l copy 1
@loop
	%m =l phi @start %m0, @upd %v, @skip %m
	%i =l phi @start %i0, @upd %i2a, @skip %i2b
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %m
	jnz %gt, @upd, @skip
@upd
	%i2a =l add %i, 1
	jmp @loop
@skip
	%i2b =l add %i, 1
	jmp @loop
@done
	ret %m
}
