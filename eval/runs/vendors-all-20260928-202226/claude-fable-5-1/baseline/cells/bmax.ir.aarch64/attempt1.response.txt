@start
	%m0 =l loadl %a
	%i0 =l copy 1
@loop
	%m =l phi @start %m0, @body %m2
	%i =l phi @start %i0, @body %i2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body

	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %m

@done
