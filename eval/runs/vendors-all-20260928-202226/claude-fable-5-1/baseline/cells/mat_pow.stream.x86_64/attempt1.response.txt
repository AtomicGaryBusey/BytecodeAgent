mat_pow:
        push rbx
        push r12
        push r13
        push r14
        push r15
        push rbp
        mov r9, rdx              # k
        mov r10, rsi
        imul r10, rsi            # n*n
        # out = identity
        xor eax, eax
.Lzero_out:
        mov qword ptr [rcx + rax*8], 0
        inc rax
        cmp rax, r10
        jb .Lzero_out
        lea rdx, [rsi + 1]       # n+1 stride
        mov rax, rcx
        mov rbx, rsi
.Ldiag:
        mov qword ptr [rax], 1
        lea rax, [rax + rdx*8]
        dec rbx
        jnz .Ldiag
        test r9, r9
        jz .Ldone
.Lpow:
        # tmp = out * a
        xor r12d, r12d           # i
        mov r11, rcx             # out row ptr
        mov r13, r8              # tmp row ptr
.Li:
        xor ebx, ebx             # j
.Lj:
        xor eax, eax             # sum
        mov r14, r11             # p = &out[i*n]
        lea r15, [rdi + rbx*8]   # q = &a[j]
        mov rdx, rsi             # t count
.Lt:
        mov rbp, qword ptr [r14]
        imul rbp, qword ptr [r15]
        add rax, rbp
        add r14, 8
        lea r15, [r15 + rsi*8]
        dec rdx
        jnz .Lt
        mov qword ptr [r13 + rbx*8], rax
        inc rbx
        cmp rbx, rsi
        jb .Lj
        lea r11, [r11 + rsi*8]
        lea r13, [r13 + rsi*8]
        inc r12
        cmp r12, rsi
        jb .Li
        # out = tmp
        xor eax, eax
.Lcopy:
        mov rdx, qword ptr [r8 + rax*8]
        mov qword ptr [rcx + rax*8], rdx
        inc rax
        cmp rax, r10
        jb .Lcopy
        dec r9
        jnz .Lpow
.Ldone:
        # zero tmp (2*n*n elements)
        lea rdx, [r10 + r10]
        xor eax, eax
.Lzt:
        mov qword ptr [r8 + rax*8], 0
        inc rax
        cmp rax, rdx
        jb .Lzt
        xor eax, eax
        pop rbp
        pop r15
        pop r14
        pop r13
        pop r12
        pop rbx
        ret
