export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @padend %i3
    %o =l phi @start 0, @padend %o3
    %c =w csltl %i, %n
    jnz %c, @group, @done
@group
    %p0 =l add %src, %i
    %b0 =w loadub %p0
    %v0 =w shl %b0, 16
    %rem =l sub %n, %i
    %c2 =w csgtl %rem, 1
    jnz %c2, @has2, @end0
@end0
    jmp @merge
@has2
    %p1 =l add %p0, 1
    %b1 =w loadub %p1
    %t1 =w shl %b1, 8
    %v1 =w or %v0, %t1
    %c3 =w csgtl %rem, 2
    jnz %c3, @has3, @end1
@end1
    jmp @merge
@has3
    %p2 =l add %p0, 2
    %b2 =w loadub %p2
    %v2 =w or %v1, %b2
    jmp @merge
@merge
    %v =w phi @end0 %v0, @end1 %v1, @has3 %v2
    %cnt =l phi @end0 2, @end1 3, @has3 4
    %i3 =l add %i, 3
    jmp @emit
@emit
    %k =l phi @merge 0, @emitbody %k1
    %sh =w phi @merge 18, @emitbody %sh1
    %ce =w csltl %k, %cnt
    jnz %ce, @emitbody, @pad
@emitbody
    %s =w shr %v, %sh
    %idx =w and %s, 63
    %ch0 =w add %idx, 65
    %g26 =w csgew %idx, 26
    %a26 =w mul %g26, 6
    %ch1 =w add %ch0, %a26
    %g52 =w csgew %idx, 52
    %a52 =w mul %g52, 75
    %ch2 =w sub %ch1, %a52
    %g62 =w csgew %idx, 62
    %a62 =w mul %g62, 15
    %ch3 =w sub %ch2, %a62
    %e63 =w ceqw %idx, 63
    %a63 =w mul %e63, 3
    %ch =w add %ch3, %a63
    %ok =l add %o, %k
    %po =l add %out, %ok
    storeb %ch, %po
    %k1 =l add %k, 1
    %sh1 =w sub %sh, 6
    jmp @emit
@pad
    %kp =l phi @emit %k, @padbody %kp1
    %cp =w csltl %kp, 4
    jnz %cp, @padbody, @padend
@padbody
    %okp =l add %o, %kp
    %pp =l add %out, %okp
    storeb 61, %pp
    %kp1 =l add %kp, 1
    jmp @pad
@padend
    %o3 =l add %o, 4
    jmp @loop
@done
    ret %o
}
