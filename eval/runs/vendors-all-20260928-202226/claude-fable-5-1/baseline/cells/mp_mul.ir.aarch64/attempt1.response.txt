	%k0 =l copy 0
@zloop

	%zc =w csltl %k, %n
	jnz %zc, @zbody, @outer0
@zbody
	%zo =l shl %k, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%k1 =l add %k, 1
	jmp @zloop
@outer0

@outer

	%carry =l phi @obody %c0, @ibody %carry1
