export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zloop
@zloop
	%i =l phi @start 0, @zbody %i1
	%c0 =w csltl %i, %n
	jnz %c0, @zbody, @outer
@zbody
	%zoff =l mul %i, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%i1 =l add %i, 1
	jmp @zloop
@outer
	%ii =l phi @zloop 0, @outer_end %ii1
	%c1 =w csltl %ii, %na
	jnz %c1, @outer_body, @count
@outer_body
	%aoff =l mul %ii, 8
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	jmp @inner
@inner
	%j =l phi @outer_body 0, @inner_body %j1
	%carry =l phi @outer_body 0, @inner_body %hi
	%c2 =w csltl %j, %nb
	jnz %c2, @inner_body, @outer_end
@inner_body
	%boff =l mul %j, 8
	%bp =l add %b, %boff
	%bj =l loadl %bp
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%p00h =l shr %p00, 32
	%mid =l add %p01, %p00h
	%mid2 =l add %mid, %p10
	%mc =w cultl %mid2, %mid
	%mcl =l extuw %mc
	%mch =l shl %mcl, 32
	%midlo =l shl %mid2, 32
	%p00l =l and %p00, 4294967295
	%lo =l or %midlo, %p00l
	%midhi =l shr %mid2, 32
	%hi0 =l add %p11, %midhi
	%hi1 =l add %hi0, %mch
	%k =l add %ii, %j
	%roff =l mul %k, 8
	%rp =l add %r, %roff
	%rv =l loadl %rp
	%s1 =l add %lo, %rv
	%c3 =w cultl %s1, %lo
	%c3l =l extuw %c3
	%s2 =l add %s1, %carry
	%c4 =w cultl %s2, %s1
	%c4l =l extuw %c4
	storel %s2, %rp
	%hi2 =l add %hi1, %c3l
	%hi =l add %hi2, %c4l
	%j1 =l add %j, 1
	jmp @inner
@outer_end
	%k2 =l add %ii, %nb
	%roff2 =l mul %k2, 8
	%rp2 =l add %r, %roff2
	storel %carry, %rp2
	%ii1 =l add %ii, 1
	jmp @outer
@count
	%cnt =l phi @outer %n, @count_body %cnt1
	%c5 =w csgtl %cnt, 0
	jnz %c5, @count_body, @done
@count_body
	%cnt1 =l sub %cnt, 1
	%roff3 =l mul %cnt1, 8
	%rp3 =l add %r, %roff3
	%v =l loadl %rp3
	%z =w ceql %v, 0
	jnz %z, @count, @done
@done
	ret %cnt
}
