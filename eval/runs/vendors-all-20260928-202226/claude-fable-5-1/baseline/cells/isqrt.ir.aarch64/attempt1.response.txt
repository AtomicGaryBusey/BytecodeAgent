export function l $isqrt(l %n0) {
@start
	%bit0 =l copy 4611686018427387904
	jmp @shrink
@shrink
	%bit =l phi @start %bit0, @shrinkbody %bitb
	%c =w cugtl %bit, %n0
	jnz %c, @shrinkbody, @loop
@shrinkbody
	%bitb =l shr %bit, 2
	jmp @shrink
@loop
	%n =l phi @shrink %n0, @join %n3
	%res =l phi @shrink 0, @join %res3
	%b =l phi @shrink %bit, @join %b3
	%z =w ceql %b, 0
	jnz %z, @done, @body
@body
	%s =l add %res, %b
	%ge =w cugel %n, %s
	jnz %ge, @yes, @no
@yes
	%n2 =l sub %n, %s
	%h =l shr %res, 1
	%res2 =l add %h, %b
	jmp @join
@no
	%res4 =l shr %res, 1
	jmp @join
@join
	%n3 =l phi @yes %n2, @no %n
	%res3 =l phi @yes %res2, @no %res4
	%b3 =l shr %b, 2
	jmp @loop
@done
	ret %res
}
