export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @merge %i2
	%o =l phi @start %out, @merge %o2
	%rem =l phi @start %n, @merge %rem2
	%z =w ceql %rem, 0
	jnz %z, @done, @body
@body
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%rem1 =l sub %rem, 1
	%one =w ceql %rem1, 0
	jnz %one, @k1, @more
@k1
	jmp @merge
@more
	%p1 =l add %p0, 1
	%b1m =w loadub %p1
	%two =w ceql %rem1, 1
	jnz %two, @k2, @k3
@k2
	jmp @merge
@k3
	%p2 =l add %p0, 2
	%b2m =w loadub %p2
	jmp @merge
@merge
	%b1 =w phi @k1 0, @k2 %b1m, @k3 %b1m
	%b2 =w phi @k1 0, @k2 0, @k3 %b2m
	%k =l phi @k1 1, @k2 2, @k3 3
	%t0 =w shl %b0, 16
	%t1 =w shl %b1, 8
	%t2 =w or %t0, %t1
	%v =w or %t2, %b2
	%v0 =w shr %v, 18
	%v1s =w shr %v, 12
	%v1 =w and %v1s, 63
	%v2s =w shr %v, 6
	%v2 =w and %v2s, 63
	%v3 =w and %v, 63
	%a00 =w add %v0, 65
	%g00 =w csgew %v0, 26
	%m00 =w mul %g00, 6
	%a01 =w add %a00, %m00
	%g01 =w csgew %v0, 52
	%m01 =w mul %g01, 75
	%a02 =w sub %a01, %m01
	%g02 =w csgew %v0, 62
	%m02 =w mul %g02, 15
	%a03 =w sub %a02, %m02
	%g03 =w ceqw %v0, 63
	%m03 =w mul %g03, 3
	%c0 =w add %a03, %m03
	%a10 =w add %v1, 65
	%g10 =w csgew %v1, 26
	%m10 =w mul %g10, 6
	%a11 =w add %a10, %m10
	%g11 =w csgew %v1, 52
	%m11 =w mul %g11, 75
	%a12 =w sub %a11, %m11
	%g12 =w csgew %v1, 62
	%m12 =w mul %g12, 15
	%a13 =w sub %a12, %m12
	%g13 =w ceqw %v1, 63
	%m13 =w mul %g13, 3
	%c1 =w add %a13, %m13
	%a20 =w add %v2, 65
	%g20 =w csgew %v2, 26
	%m20 =w mul %g20, 6
	%a21 =w add %a20, %m20
	%g21 =w csgew %v2, 52
	%m21 =w mul %g21, 75
	%a22 =w sub %a21, %m21
	%g22 =w csgew %v2, 62
	%m22 =w mul %g22, 15
	%a23 =w sub %a22, %m22
	%g23 =w ceqw %v2, 63
	%m23 =w mul %g23, 3
	%mp2 =w add %a23, %m23
	%d2 =w sub 61, %mp2
	%f2 =w csltl %k, 2
	%e2 =w mul %d2, %f2
	%c2 =w add %mp2, %e2
	%a30 =w add %v3, 65
	%g30 =w csgew %v3, 26
	%m30 =w mul %g30, 6
	%a31 =w add %a30, %m30
	%g31 =w csgew %v3, 52
	%m31 =w mul %g31, 75
	%a32 =w sub %a31, %m31
	%g32 =w csgew %v3, 62
	%m32 =w mul %g32, 15
	%a33 =w sub %a32, %m32
	%g33 =w ceqw %v3, 63
	%m33 =w mul %g33, 3
	%mp3 =w add %a33, %m33
	%d3 =w sub 61, %mp3
	%f3 =w csltl %k, 3
	%e3 =w mul %d3, %f3
	%c3 =w add %mp3, %e3
	storeb %c0, %o
	%oa =l add %o, 1
	storeb %c1, %oa
	%ob =l add %o, 2
	storeb %c2, %ob
	%oc =l add %o, 3
	storeb %c3, %oc
	%o2 =l add %o, 4
	%i2 =l add %i, %k
	%rem2 =l sub %rem, %k
	jmp @loop
@done
	%res =l sub %o, %out
	ret %res
}
