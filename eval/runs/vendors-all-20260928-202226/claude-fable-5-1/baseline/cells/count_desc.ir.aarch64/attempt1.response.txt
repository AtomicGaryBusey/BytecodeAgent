export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	%c0 =w csgtl %n, 1
	jnz %c0, @loop, @done0
@done0
	ret 0
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%p =l mul %i, 8
	%p =l add %a, %p
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%gt =w csgtl %x, %y
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	%more =w csltl %i1, %lim
	jnz %more, @body, @end
@body
	jmp @loop
@end
	ret %cnt1
}
