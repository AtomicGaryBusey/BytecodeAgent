export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%n1 =l add %n, 1
	%i =l copy 0
@id_loop
	%c =w csltl %i, %nn
	jnz %c, @id_body, @id_done
@id_body
	%r =l rem %i, %n1
	%z =w ceql %r, 0
	%v =l extuw %z
	%off =l mul %i, 8
	%p =l add %out, %off
	storel %v, %p
	%i =l add %i, 1
	jmp @id_loop
@id_done
	%cnt =l copy %k
@kloop
	%kc =w ceql %cnt, 0
	jnz %kc, @done, @mul
@mul
	%i =l copy 0
@iloop
	%ci =w csltl %i, %n
	jnz %ci, @ibody, @icopy
@ibody
	%j =l copy 0
@jloop
	%cj =w csltl %j, %n
	jnz %cj, @jbody, @jdone
@jbody
	%s =l copy 0
	%t =l copy 0
@tloop
	%ct =w csltl %t, %n
	jnz %ct, @tbody, @tdone
@tbody
	%idx1 =l mul %i, %n
	%idx1 =l add %idx1, %t
	%o1 =l mul %idx1, 8
	%p1 =l add %out, %o1
	%x =l loadl %p1
	%idx2 =l mul %t, %n
	%idx2 =l add %idx2, %j
	%o2 =l mul %idx2, 8
	%p2 =l add %a, %o2
	%y =l loadl %p2
	%prod =l mul %x, %y
	%s =l add %s, %prod
	%t =l add %t, 1
	jmp @tloop
@tdone
	%idx3 =l mul %i, %n
	%idx3 =l add %idx3, %j
	%o3 =l mul %idx3, 8
	%p3 =l add %tmp, %o3
	storel %s, %p3
	%j =l add %j, 1
	jmp @jloop
@jdone
	%i =l add %i, 1
	jmp @iloop
@icopy
	%i =l copy 0
@cloop
	%cc =w csltl %i, %nn
	jnz %cc, @cbody, @cdone
@cbody
	%o4 =l mul %i, 8
	%p4 =l add %tmp, %o4
	%v4 =l loadl %p4
	%p5 =l add %out, %o4
	storel %v4, %p5
	%i =l add %i, 1
	jmp @cloop
@cdone
	%cnt =l sub %cnt, 1
	jmp @kloop
@done
	%nn2 =l mul %nn, 2
	%i =l copy 0
@zloop
	%cz =w csltl %i, %nn2
	jnz %cz, @zbody, @zdone
@zbody
	%o6 =l mul %i, 8
	%p6 =l add %tmp, %o6
	storel 0, %p6
	%i =l add %i, 1
	jmp @zloop
@zdone
	ret 0
}
