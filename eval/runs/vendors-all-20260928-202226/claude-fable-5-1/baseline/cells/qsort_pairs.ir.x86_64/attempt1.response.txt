export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csgtl %n, 1
	jnz %c, @body, @ret
@body
	%m =l shr %n, 1
	%mo =l shl %m, 4
	%pm =l add %a, %mo
	%pm8 =l add %pm, 8
	%nl =l sub %n, 1
	%lo =l shl %nl, 4
	%pl =l add %a, %lo
	%pl8 =l add %pl, 8
	%pk =l loadl %pm
	%pv =l loadl %pm8
	%lk =l loadl %pl
	%lv =l loadl %pl8
	storel %lk, %pm
	storel %lv, %pm8
	storel %pk, %pl
	storel %pv, %pl8
	jmp @loop
@loop
	%j =l phi @body %a, @next %j1
	%i =l phi @body %a, @next %i1
	%c2 =w cnel %j, %pl
	jnz %c2, @cmp, @done
@cmp
	%jk =l loadl %j
	%jv8 =l add %j, 8
	%jv =l loadl %jv8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @eqchk
@eqchk
	%eq =w ceql %jk, %pk
	jnz %eq, @vchk, @noswap
@vchk
	%vlt =w csltl %jv, %pv
	jnz %vlt, @swap, @noswap
@swap
	%ik =l loadl %i
	%iv8 =l add %i, 8
	%iv =l loadl %iv8
	storel %jk, %i
	storel %jv, %iv8
	storel %ik, %j
	storel %iv, %jv8
	%i1a =l add %i, 16
	jmp @next
@noswap
	jmp @next
@next
	%i1 =l phi @swap %i1a, @noswap %i
	%j1 =l add %j, 16
	jmp @loop
@done
	%ik2 =l loadl %i
	%iv82 =l add %i, 8
	%iv2 =l loadl %iv82
	storel %pk, %i
	storel %pv, %iv82
	storel %ik2, %pl
	storel %iv2, %pl8
	%d =l sub %i, %a
	%cl =l sar %d, 4
	%r1 =l call $qsort_pairs(l %a, l %cl)
	%p2 =l add %i, 16
	%cr =l sub %nl, %cl
	%r2 =l call $qsort_pairs(l %p2, l %cr)
	jmp @ret
@ret
	ret 0
}
