memmove_i64:
	xor eax, eax
	test rdx, rdx
	jle .Ldone
	cmp rdi, rsi
	jbe .Lfwd
	# dst > src: copy backwards
	lea rcx, [rdx - 1]
.Lback:
	mov r8, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], r8
	dec rcx
	jns .Lback
	jmp .Ldone
.Lfwd:
	xor ecx, ecx
.Lfwdloop:
	mov r8, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], r8
	inc rcx
	cmp rcx, rdx
	jb .Lfwdloop
.Ldone:
	ret
