bubble_sort:
	cmp x1, #2
	b.lt .Ldone
	sub x9, x1, #1          // x9 = n-1 (number of comparisons in current pass)
.Louter:
	mov x10, #0             // x10 = i
	mov x11, #0             // x11 = swapped flag
.Linner:
	add x12, x10, #1
	ldr x13, [x0, x10, lsl #3]
	ldr x14, [x0, x12, lsl #3]
	cmp x13, x14
	b.le .Lnoswap
	str x14, [x0, x10, lsl #3]
	str x13, [x0, x12, lsl #3]
	mov x11, #1
.Lnoswap:
	mov x10, x12
	cmp x10, x9
	b.lt .Linner
	cbz x11, .Ldone
	subs x9, x9, #1
	b.gt .Louter
.Ldone:
	mov x0, #0
	ret
