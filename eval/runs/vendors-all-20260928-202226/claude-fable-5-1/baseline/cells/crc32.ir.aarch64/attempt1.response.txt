export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
@loop
	%crc =w phi @start %crc0, @bodyend %crc2
	%i =l phi @start %i0, @bodyend %i2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %crc, %b
	%k0 =w copy 0
@bit
	%cb =w phi @body %cx, @bitend %cb2
	%k =w phi @body %k0, @bitend %k2
	%lsb =w and %cb, 1
	%sh =w shr %cb, 1
	%xo =w xor %sh, 3988292384
	jnz %lsb, @odd, @even
@odd
	%r1 =w copy %xo
	jmp @bitend
@even
	%r2 =w copy %sh
	jmp @bitend
@bitend
	%cb2 =w phi @odd %r1, @even %r2
	%k2 =w add %k, 1
	%kc =w cnew %k2, 8
	jnz %kc, @bit, @bodyend
@bodyend
	%crc2 =w copy %cb2
	%i2 =l add %i, 1
	jmp @loop
@done
	%res =w xor %crc, 4294967295
	%ret =l extuw %res
	ret %ret
}
