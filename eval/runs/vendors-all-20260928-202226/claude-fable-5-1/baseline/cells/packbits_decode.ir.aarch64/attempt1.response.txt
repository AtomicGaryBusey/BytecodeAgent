export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i =l add %i, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @notnop
@notnop
	%neg =w csltw %h, 0
	jnz %neg, @repeat, @literal
@literal
	%len =w add %h, 1
	%lenl =l extsw %len
	%end =l add %i, %lenl
	%bad =w csgtl %end, %n
	jnz %bad, @err2, @litloop
@litloop
	%c2 =w csltl %i, %end
	jnz %c2, @litbody, @loop
@litbody
	%oc =w csltl %o, %cap
	jnz %oc, @litwrite, @err1
@litwrite
	%sp =l add %src, %i
	%b =w loadub %sp
	%dp =l add %out, %o
	storeb %b, %dp
	%i =l add %i, 1
	%o =l add %o, 1
	jmp @litloop
@repeat
	%c3 =w csltl %i, %n
	jnz %c3, @repok, @err2
@repok
	%sp2 =l add %src, %i
	%rb =w loadub %sp2
	%i =l add %i, 1
	%cnt =w sub 1, %h
	%cntl =l extsw %cnt
	%oend =l add %o, %cntl
@reploop
	%c4 =w csltl %o, %oend
	jnz %c4, @repbody, @loop
@repbody
	%oc2 =w csltl %o, %cap
	jnz %oc2, @repwrite, @err1
@repwrite
	%dp2 =l add %out, %o
	storeb %rb, %dp2
	%o =l add %o, 1
	jmp @reploop
@done
	ret %o
@err2
	ret -2
@err1
	ret -1
}
