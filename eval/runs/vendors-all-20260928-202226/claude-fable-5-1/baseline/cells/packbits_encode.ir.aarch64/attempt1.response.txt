export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%w =l copy 0
	%i =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%pi =l add %src, %i
	%b0 =w loadub %pi
	%k =l add %i, 1
	%lim =l add %i, 128
	%c =w csltl %n, %lim
	jnz %c, @limn, @runloop
@limn
	%lim =l copy %n
@runloop
	%c =w csltl %k, %lim
	jnz %c, @runchk, @runend
@runchk
	%pk =l add %src, %k
	%bk =w loadub %pk
	%c =w ceqw %bk, %b0
	jnz %c, @runinc, @runend
@runinc
	%k =l add %k, 1
	jmp @runloop
@runend
	%r =l sub %k, %i
	%c =w csltl %r, 3
	jnz %c, @literal, @repeat
@repeat
	%rem =l sub %cap, %w
	%c =w csltl %rem, 2
	jnz %c, @fail, @repeat2
@repeat2
	%h =w sub 257, %r
	%po =l add %out, %w
	storeb %h, %po
	%po1 =l add %po, 1
	storeb %b0, %po1
	%w =l add %w, 2
	%i =l copy %k
	jmp @loop
@literal
	%j =l add %i, 1
@litloop
	%c =w csltl %j, %n
	jnz %c, @lit2, @litend
@lit2
	%d =l sub %j, %i
	%c =w csltl %d, 128
	jnz %c, @lit3, @litend
@lit3
	%j2 =l add %j, 2
	%c =w csltl %j2, %n
	jnz %c, @lit4, @litinc
@lit4
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %pj, 1
	%bj1 =w loadub %pj1
	%c =w ceqw %bj, %bj1
	jnz %c, @lit5, @litinc
@lit5
	%pj2 =l add %pj, 2
	%bj2 =w loadub %pj2
	%c =w ceqw %bj, %bj2
	jnz %c, @litend, @litinc
@litinc
	%j =l add %j, 1
	jmp @litloop
@litend
	%len =l sub %j, %i
	%need =l add %len, 1
	%rem =l sub %cap, %w
	%c =w csltl %rem, %need
	jnz %c, @fail, @litemit
@litemit
	%h =w sub %len, 1
	%po =l add %out, %w
	storeb %h, %po
	%w =l add %w, 1
	%q =l copy %i
@copyloop
	%c =w csltl %q, %j
	jnz %c, @copy2, @copyend
@copy2
	%ps =l add %src, %q
	%bq =w loadub %ps
	%pd =l add %out, %w
	storeb %bq, %pd
	%q =l add %q, 1
	%w =l add %w, 1
	jmp @copyloop
@copyend
	%i =l copy %j
	jmp @loop
@fail
	%m =l copy -1
	ret %m
@done
	ret %w
}
