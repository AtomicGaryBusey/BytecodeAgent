export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
	jmp @zloop
@zloop
	%j =l phi @start %j0, @zbody %j1
	%cz =w csgtl %j, %n
	jnz %cz, @iinit, @zbody
@zbody
	%pz0 =l mul %j, 8
	%pz =l add %row, %pz0
	storel 0, %pz
	%j1 =l add %j, 1
	jmp @zloop
@iinit
	%i0 =l copy 0
	jmp @iloop
@iloop
	%i =l phi @iinit %i0, @iend %i1
	%ci =w csltl %i, %m
	jnz %ci, @ibody, @ret
@ibody
	%px =l add %x, %i
	%xi =w loadub %px
	%d0 =l copy 0
	%jj0 =l copy 1
	jmp @jloop
@jloop
	%jj =l phi @ibody %jj0, @jend %jj1
	%diag =l phi @ibody %d0, @jend %up
	%cj =w csgtl %jj, %n
	jnz %cj, @iend, @jbody
@jbody
	%q0 =l mul %jj, 8
	%q =l add %row, %q0
	%up =l loadl %q
	%q1 =l sub %q, 8
	%left =l loadl %q1
	%py0 =l add %y, %jj
	%py =l sub %py0, 1
	%yj =w loadub %py
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @jend
@nomatch
	%cm =w csgtl %up, %left
	%cml =l extuw %cm
	%mask =l sub 0, %cml
	%dif =l sub %up, %left
	%sel =l and %dif, %mask
	%v2 =l add %left, %sel
	jmp @jend
@jend
	%v =l phi @match %v1, @nomatch %v2
	storel %v, %q
	%jj1 =l add %jj, 1
	jmp @jloop
@iend
	%i1 =l add %i, 1
	jmp @iloop
@ret
	%pn0 =l mul %n, 8
	%pn =l add %row, %pn0
	%r =l loadl %pn
	ret %r
}
