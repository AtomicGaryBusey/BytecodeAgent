export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
@zloop
	%j =l phi @start %j0, @zbody %j1
	%c0 =w csgtl %j, %n
	jnz %c0, @iinit, @zbody
@zbody
	%off0 =l mul %j, 8
	%p0 =l add %row, %off0
	storel 0, %p0
	%j1 =l add %j, 1
	jmp @zloop
@iinit
	%i0 =l copy 0
@iloop
	%i =l phi @iinit %i0, @inext %i1
	%c1 =w csltl %i, %m
	jnz %c1, @ibody, @done
@ibody
	%xp =l add %x, %i
	%xi =w loadub %xp
	%jj0 =l copy 1
	%d0 =l copy 0
@jloop
	%jj =l phi @ibody %jj0, @jend %jj1
	%diag =l phi @ibody %d0, @jend %up
	%c2 =w csgtl %jj, %n
	jnz %c2, @inext, @jbody
@jbody
	%offj =l mul %jj, 8
	%pj =l add %row, %offj
	%up =l loadl %pj
	%pjm =l sub %pj, 8
	%left =l loadl %pjm
	%jm1 =l sub %jj, 1
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%v0 =l add %diag, 1
	jmp @jend
@nomatch
	%c3 =w csgtl %up, %left
	jnz %c3, @takeup, @takeleft
@takeup
	jmp @jend
@takeleft
	jmp @jend
@jend
	%v =l phi @match %v0, @takeup %up, @takeleft %left
	storel %v, %pj
	%jj1 =l add %jj, 1
	jmp @jloop
@inext
	%i1 =l add %i, 1
	jmp @iloop
@done
	%offn =l mul %n, 8
	%pn =l add %row, %offn
	%r =l loadl %pn
	ret %r
}
