map_sum:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    xor ebx, ebx
    xor ebp, ebp
    test r13, r13
    jle .Ldone
.Lloop:
    mov rdi, qword ptr [r12 + rbp*8]
    call r14
    add rbx, rax
    inc rbp
    cmp rbp, r13
    jl .Lloop
.Ldone:
    mov rax, rbx
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
