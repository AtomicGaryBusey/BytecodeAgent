export function l $reverse(l %a, l %n) {
@start
	%lo =l copy %a
	%nb =l mul %n, 8
	%hi0 =l add %a, %nb
	%hi =l sub %hi0, 8
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @done
@body
	%x =l loadl %lo
	%y =l loadl %hi
	storel %y, %lo
	storel %x, %hi
	%lo =l add %lo, 8
	%hi =l sub %hi, 8
	jmp @loop
@done
	ret 0
}
