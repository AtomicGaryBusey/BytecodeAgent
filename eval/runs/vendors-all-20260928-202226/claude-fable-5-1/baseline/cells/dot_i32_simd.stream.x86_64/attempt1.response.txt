dot_i32_simd:
	xor eax, eax
	xor r8d, r8d
	mov rcx, rdx
	and rcx, -4
	jz .Ltail
	pxor xmm0, xmm0
.Lloop:
	movdqu xmm1, xmmword ptr [rdi + r8*4]
	movdqu xmm2, xmmword ptr [rsi + r8*4]
	movdqa xmm3, xmm1
	movdqa xmm4, xmm2
	pmuldq xmm1, xmm2
	psrlq xmm3, 32
	psrlq xmm4, 32
	pmuldq xmm3, xmm4
	paddq xmm0, xmm1
	paddq xmm0, xmm3
	add r8, 4
	cmp r8, rcx
	jb .Lloop
	movdqa xmm1, xmm0
	psrldq xmm1, 8
	paddq xmm0, xmm1
	movq rax, xmm0
.Ltail:
	cmp r8, rdx
	jae .Ldone
.Lscalar:
	movsxd r9, dword ptr [rdi + r8*4]
	movsxd r10, dword ptr [rsi + r8*4]
	imul r9, r10
	add rax, r9
	inc r8
	cmp r8, rdx
	jb .Lscalar
.Ldone:
	ret
