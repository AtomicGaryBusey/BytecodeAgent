export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c0 =w csltl %i, %n
	jnz %c0, @hdr, @done
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i =l add %i, 1
	%c1 =w ceqw %h, -128
	jnz %c1, @loop, @classify
@classify
	%c2 =w csgew %h, 0
	jnz %c2, @lit, @rep
@lit
	%len =l extsw %h
	%len =l add %len, 1
	%end =l add %i, %len
	%c3 =w csgtl %end, %n
	jnz %c3, @err2, @litloop
@litloop
	%c4 =w csltl %i, %end
	jnz %c4, @litbody, @loop
@litbody
	%c5 =w csltl %o, %cap
	jnz %c5, @litw, @err1
@litw
	%sp =l add %src, %i
	%b =w loadub %sp
	%op =l add %out, %o
	storeb %b, %op
	%i =l add %i, 1
	%o =l add %o, 1
	jmp @litloop
@rep
	%c6 =w csltl %i, %n
	jnz %c6, @repok, @err2
@repok
	%sp2 =l add %src, %i
	%b2 =w loadub %sp2
	%i =l add %i, 1
	%hl =l extsw %h
	%cnt =l sub 1, %hl
@reploop
	%c7 =w cnel %cnt, 0
	jnz %c7, @repbody, @loop
@repbody
	%c8 =w csltl %o, %cap
	jnz %c8, @repw, @err1
@repw
	%op2 =l add %out, %o
	storeb %b2, %op2
	%o =l add %o, 1
	%cnt =l sub %cnt, 1
	jmp @reploop
@done
	ret %o
@err1
	ret -1
@err2
	ret -2
}
