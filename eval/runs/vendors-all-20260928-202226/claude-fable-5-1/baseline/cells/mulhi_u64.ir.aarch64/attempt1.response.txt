export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%alo =l and %a, %mask
	%ahi =l shr %a, 32
	%blo =l and %b, %mask
	%bhi =l shr %b, 32
	%ll =l mul %alo, %blo
	%lh =l mul %alo, %bhi
	%hl =l mul %ahi, %blo
	%hh =l mul %ahi, %bhi
	%llhi =l shr %ll, 32
	%lhlo =l and %lh, %mask
	%hllo =l and %hl, %mask
	%mid0 =l add %llhi, %lhlo
	%mid =l add %mid0, %hllo
	%midhi =l shr %mid, 32
	%lhhi =l shr %lh, 32
	%hlhi =l shr %hl, 32
	%r0 =l add %hh, %lhhi
	%r1 =l add %r0, %hlhi
	%r =l add %r1, %midhi
	ret %r
}
