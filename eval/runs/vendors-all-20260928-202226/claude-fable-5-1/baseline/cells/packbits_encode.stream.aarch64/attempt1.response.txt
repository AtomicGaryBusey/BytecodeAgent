packbits_encode:
    mov x9, #0                  // i
    mov x10, #0                 // w (bytes written)
.Lloop:
    cmp x9, x1
    b.ge .Ldone
    ldrb w13, [x0, x9]          // b = src[i]
    add x14, x9, #1             // k
    add x15, x9, #128
    cmp x15, x1
    csel x15, x15, x1, lt       // lim = min(n, i+128)
.Lrun:
    cmp x14, x15
    b.ge .Lrun_end
    ldrb w12, [x0, x14]
    cmp w12, w13
    b.ne .Lrun_end
    add x14, x14, #1
    b .Lrun
.Lrun_end:
    sub x12, x14, x9            // r = run(i)
    cmp x12, #3
    b.lt .Lliteral
    sub x15, x3, x10            // room
    cmp x15, #2
    b.lt .Lfail
    mov w14, #1
    sub w14, w14, w12           // header = 1 - r
    strb w14, [x2, x10]
    add x10, x10, #1
    strb w13, [x2, x10]
    add x10, x10, #1
    add x9, x9, x12             // i += r
    b .Lloop
.Lliteral:
    mov x11, x9                 // j = i
.Llit_loop:
    cmp x11, x1
    b.ge .Llit_end
    sub x15, x11, x9
    cmp x15, #128
    b.ge .Llit_end
    add x15, x11, #2
    cmp x15, x1
    b.ge .Llit_take             // j+2 >= n -> run(j) < 3
    add x15, x0, x11
    ldrb w13, [x15]
    ldrb w12, [x15, #1]
    cmp w12, w13
    b.ne .Llit_take
    ldrb w12, [x15, #2]
    cmp w12, w13
    b.eq .Llit_end              // run(j) >= 3
.Llit_take:
    add x11, x11, #1
    b .Llit_loop
.Llit_end:
    sub x12, x11, x9            // L
    add x15, x12, #1            // packet size
    sub x14, x3, x10            // room
    cmp x14, x15
    b.lt .Lfail
    sub w14, w12, #1
    strb w14, [x2, x10]
    add x10, x10, #1
.Lcopy:
    cmp x9, x11
    b.ge .Lloop
    ldrb w13, [x0, x9]
    strb w13, [x2, x10]
    add x9, x9, #1
    add x10, x10, #1
    b .Lcopy
.Lfail:
    mov x0, #-1
    ret
.Ldone:
    mov x0, x10
    ret
