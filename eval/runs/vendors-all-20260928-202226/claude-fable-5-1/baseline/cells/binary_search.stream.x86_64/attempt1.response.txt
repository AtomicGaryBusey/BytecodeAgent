binary_search:
    xor eax, eax          # lo = 0
    mov rcx, rsi          # hi = n
.Lloop:
    cmp rax, rcx
    jge .Lnotfound
    mov r8, rax
    add r8, rcx
    shr r8, 1             # mid = (lo+hi)/2 (no overflow, n <= 200)
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .Lfound
    jl .Lgo_right
    mov rcx, r8           # hi = mid
    jmp .Lloop
.Lgo_right:
    lea rax, [r8 + 1]     # lo = mid + 1
    jmp .Lloop
.Lfound:
    mov rax, r8
    ret
.Lnotfound:
    mov rax, -1
    ret
