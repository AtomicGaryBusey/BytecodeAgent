export function l $isqrt(l %n0) {
@start
	%n =l copy %n0
	%res =l copy 0
	%bit =l copy 4611686018427387904
	jmp @l1
@l1
	%c1 =w cugtl %bit, %n
	jnz %c1, @l1body, @l2
@l1body
	%bit =l shr %bit, 2
	jmp @l1
@l2
	%c2 =w cnel %bit, 0
	jnz %c2, @l2body, @end
@l2body
	%s =l add %res, %bit
	%c3 =w cugel %n, %s
	jnz %c3, @yes, @no
@yes
	%n =l sub %n, %s
	%r1 =l shr %res, 1
	%res =l add %r1, %bit
	jmp @l2next
@no
	%res =l shr %res, 1
	jmp @l2next
@l2next
	%bit =l shr %bit, 2
	jmp @l2
@end
	ret %res
}
