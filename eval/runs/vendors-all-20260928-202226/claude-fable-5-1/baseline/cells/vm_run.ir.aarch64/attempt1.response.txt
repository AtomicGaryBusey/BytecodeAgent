export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%end =w ceql %pc, %len
	jnz %end, @done, @fetch
@fetch
	%off =l mul %pc, 8
	%ia =l add %code, %off
	%op =l loadl %ia
	%oa =l add %ia, 8
	%doff =l mul %d, 8
	%na =l add %stack, %doff
	%ta =l sub %na, 8
	%sa =l sub %ta, 8
	%is0 =w ceql %op, 0
	jnz %is0, @done, @c1
@c1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @c2
@push
	%v =l loadl %oa
	storel %v, %na
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@c2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @c3
@add
	%a =l loadl %sa
	%b =l loadl %ta
	%r =l add %a, %b
	storel %r, %sa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @c4
@sub
	%a =l loadl %sa
	%b =l loadl %ta
	%r =l sub %a, %b
	storel %r, %sa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @c5
@mul
	%a =l loadl %sa
	%b =l loadl %ta
	%r =l mul %a, %b
	storel %r, %sa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @c6
@dup
	%b =l loadl %ta
	storel %b, %na
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @c7
@swap
	%a =l loadl %sa
	%b =l loadl %ta
	storel %b, %sa
	storel %a, %ta
	%pc =l add %pc, 1
	jmp @loop
@c7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @c8
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @c9
@over
	%a =l loadl %sa
	storel %a, %na
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @c10
@lt
	%a =l loadl %sa
	%b =l loadl %ta
	%rw =w csltl %a, %b
	%r =l extuw %rw
	storel %r, %sa
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@c10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @c11
@jz
	%v =l loadl %ta
	%d =l sub %d, 1
	%vz =w ceql %v, 0
	jnz %vz, @jztake, @jzskip
@jztake
	%pc =l loadl %oa
	jmp @loop
@jzskip
	%pc =l add %pc, 2
	jmp @loop
@c11
	%pc =l loadl %oa
	jmp @loop
@done
	%i =l copy %d
	jmp @zloop
@zloop
	%zc =w csltl %i, %cap
	jnz %zc, @zbody, @ret
@zbody
	%zoff =l mul %i, 8
	%za =l add %stack, %zoff
	storel 0, %za
	%i =l add %i, 1
	jmp @zloop
@ret
	%dz =w ceql %d, 0
	jnz %dz, @ret0, @rettop
@ret0
	ret 0
@rettop
	%tb =l sub %d, 1
	%toff =l mul %tb, 8
	%tad =l add %stack, %toff
	%res =l loadl %tad
	ret %res
}
