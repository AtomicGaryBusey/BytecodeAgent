isqrt:
    mov x2, #0                // r = 0
    mov x1, #1
    lsl x1, x1, #62           // bit = 1 << 62
.Lfind:
    cmp x1, x0
    b.ls .Lloop               // bit <= n: start
    lsr x1, x1, #2
    cbnz x1, .Lfind
    mov x0, #0                // n == 0
    ret
.Lloop:
    add x3, x2, x1            // t = r + bit
    cmp x0, x3
    sub x4, x0, x3            // n - t
    csel x0, x4, x0, hs       // if n >= t: n -= t
    add x5, x1, x2, lsr #1    // (r >> 1) + bit
    lsr x2, x2, #1            // r >> 1
    csel x2, x5, x2, hs       // r = (n>=t) ? (r>>1)+bit : r>>1
    lsr x1, x1, #2            // bit >>= 2
    cbnz x1, .Lloop
    mov x0, x2
    ret
