export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l call %f(l %a)
	%y0v =l loadl %a
	jmp @fix
@fix
	jmp @init
@init
	%p0 =l loadl %a
	%first =l call %f(l %p0)
	%odd0 =l and %first, 1
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @init %i0, @body %i1
	%min =l phi @init %first, @body %min1
	%max =l phi @init %first, @body %max1
	%sum =l phi @init %first, @body %sum1
	%cnt =l phi @init %odd0, @body %cnt1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%gt =w csgtl %y, %max
	%ltl =l extuw %lt
	%gtl =l extuw %gt
	%nlt =l sub 0, %ltl
	%ngt =l sub 0, %gtl
	%d1 =l sub %y, %min
	%d2 =l sub %y, %max
	%m1 =l and %d1, %nlt
	%m2 =l and %d2, %ngt
	%min1 =l add %min, %m1
	%max1 =l add %max, %m2
	%sum1 =l add %sum, %y
	%o =l and %y, 1
	%cnt1 =l add %cnt, %o
	%i1 =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
