export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%w =l copy 0
	jmp @loop
@loop
	%c0 =w csltl %i, %n
	jnz %c0, @body, @done
@done
	ret %w
@body
	%pi =l add %src, %i
	%bi =w loadub %pi
	%i2 =l add %i, 2
	%c1 =w csltl %i2, %n
	jnz %c1, @chk1, @lit
@chk1
	%p1 =l add %pi, 1
	%b1 =w loadub %p1
	%c2 =w ceqw %b1, %bi
	jnz %c2, @chk2, @lit
@chk2
	%p2 =l add %pi, 2
	%b2 =w loadub %p2
	%c3 =w ceqw %b2, %bi
	jnz %c3, @run, @lit
@run
	%k =l add %i, 3
	%lim0 =l add %i, 128
	%cl =w csltl %n, %lim0
	jnz %cl, @limn, @lim128
@limn
	%lim =l copy %n
	jmp @runloop
@lim128
	%lim =l copy %lim0
	jmp @runloop
@runloop
	%ck =w csltl %k, %lim
	jnz %ck, @runchk, @runend
@runchk
	%pk =l add %src, %k
	%bk =w loadub %pk
	%ce =w ceqw %bk, %bi
	jnz %ce, @runinc, @runend
@runinc
	%k =l add %k, 1
	jmp @runloop
@runend
	%r =l sub %k, %i
	%w2 =l add %w, 2
	%cf =w cslel %w2, %cap
	jnz %cf, @emitrun, @fail
@fail
	ret -1
@emitrun
	%hdr =l sub 257, %r
	%po =l add %out, %w
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %bi, %po1
	%w =l copy %w2
	%i =l add %i, %r
	jmp @loop
@lit
	%j =l add %i, 1
	%lim1 =l add %i, 128
	jmp @litloop
@litloop
	%cj =w csltl %j, %n
	jnz %cj, @lit2, @litend
@lit2
	%cj2 =w csltl %j, %lim1
	jnz %cj2, @lit3, @litend
@lit3
	%j2 =l add %j, 2
	%cj3 =w csltl %j2, %n
	jnz %cj3, @lit4, @litinc
@lit4
	%pj =l add %src, %j
	%bj =w loadub %pj
	%pj1 =l add %pj, 1
	%bj1 =w loadub %pj1
	%ce1 =w ceqw %bj1, %bj
	jnz %ce1, @lit5, @litinc
@lit5
	%pj2 =l add %pj, 2
	%bj2 =w loadub %pj2
	%ce2 =w ceqw %bj2, %bj
	jnz %ce2, @litend, @litinc
@litinc
	%j =l add %j, 1
	jmp @litloop
@litend
	%L =l sub %j, %i
	%need =l add %L, 1
	%wn =l add %w, %need
	%cf2 =w cslel %wn, %cap
	jnz %cf2, @emitlit, @fail
@emitlit
	%h2 =l sub %L, 1
	%po2 =l add %out, %w
	storeb %h2, %po2
	%d =l add %po2, 1
	%s =l add %src, %i
	%e =l add %src, %j
	jmp @cpy
@cpy
	%cc =w csltl %s, %e
	jnz %cc, @cpyb, @cpyend
@cpyb
	%bc =w loadub %s
	storeb %bc, %d
	%s =l add %s, 1
	%d =l add %d, 1
	jmp @cpy
@cpyend
	%w =l copy %wn
	%i =l copy %j
	jmp @loop
}
