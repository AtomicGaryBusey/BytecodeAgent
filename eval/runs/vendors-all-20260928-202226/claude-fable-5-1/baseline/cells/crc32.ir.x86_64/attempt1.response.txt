export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc1 =w xor %crc, %b
	%k =w copy 0
@bitloop
	%kc =w csltw %k, 8
	jnz %kc, @bitbody, @bitdone
@bitbody
	%lsb =w and %crc1, 1
	%sh =w shr %crc1, 1
	jnz %lsb, @odd, @even
@odd
	%crc2 =w xor %sh, 3988292384
	jmp @bitnext
@even
	%crc2 =w copy %sh
	jmp @bitnext
@bitnext
	%crc1 =w copy %crc2
	%k =w add %k, 1
	jmp @bitloop
@bitdone
	%crc =w copy %crc1
	%i =l add %i, 1
	jmp @loop
@done
	%res =w xor %crc, 4294967295
	%r =l extuw %res
	ret %r
}
