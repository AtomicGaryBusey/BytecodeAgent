stats_f:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	sub rsp, 8
	mov rbx, rdi                        # a (current element pointer)
	mov rbp, rsi                        # remaining count
	mov r12, rdx                        # f
	mov r13, rcx                        # out
	mov rdi, qword ptr [rbx]
	call r12
	mov r14, rax                        # min
	mov r15, rax                        # max
	mov qword ptr [r13 + 16], rax       # sum kept in out[2]
	and eax, 1
	mov qword ptr [rsp], rax            # odd count on stack
	add rbx, 8
	dec rbp
	jz .Ldone
.Lloop:
	mov rdi, qword ptr [rbx]
	call r12
	cmp rax, r14
	cmovl r14, rax
	cmp rax, r15
	cmovg r15, rax
	add qword ptr [r13 + 16], rax
	and eax, 1
	add qword ptr [rsp], rax
	add rbx, 8
	dec rbp
	jnz .Lloop
.Ldone:
	mov qword ptr [r13], r14
	mov qword ptr [r13 + 8], r15
	mov rax, qword ptr [rsp]
	add rsp, 8
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
