export function l $reverse(l %a, l %n) {
@start
	%lo =l copy %a
	%nb =l mul %n, 8
	%end =l add %a, %nb
	%hi =l sub %end, 8
@loop
	%lo1 =l phi @start %lo, @body %lo2
	%hi1 =l phi @start %hi, @body %hi2
	%c =w csltl %lo1, %hi1
	jnz %c, @body, @done
@body
	%x =l loadl %lo1
	%y =l loadl %hi1
	storel %y, %lo1
	storel %x, %hi1
	%lo2 =l add %lo1, 8
	%hi2 =l sub %hi1, 8
	jmp @loop
@done
	ret 0
}
