export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0h =l shr %p0, 32
	%p1l =l and %p1, 4294967295
	%p1h =l shr %p1, 32
	%p2l =l and %p2, 4294967295
	%p2h =l shr %p2, 32
	%m0 =l add %p0h, %p1l
	%mid =l add %m0, %p2l
	%midh =l shr %mid, 32
	%h0 =l add %p3, %p1h
	%h1 =l add %h0, %p2h
	%hi =l add %h1, %midh
	ret %hi
}
