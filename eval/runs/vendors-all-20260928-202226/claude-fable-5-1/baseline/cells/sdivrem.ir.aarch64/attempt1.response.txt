export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @zero, @chk
@chk
	%bm1 =w ceql %b, -1
	%min =l shl 1, 63
	%amin =w ceql %a, %min
	%both =w and %bm1, %amin
	jnz %both, @ovf, @norm
@norm
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@zero
	storel %a, %rem
	ret 0
@ovf
	storel 0, %rem
	ret %a
}
