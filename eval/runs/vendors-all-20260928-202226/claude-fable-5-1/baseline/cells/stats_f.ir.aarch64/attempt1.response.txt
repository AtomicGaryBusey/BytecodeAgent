export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%a0 =l loadl %a
	%y0 =l call %f(l %a0)
	%odd0 =l and %y0, 1
	%p1 =l add %a, 8
	jmp @loop
@loop
	%i =l phi @start 1, @body %i1
	%p =l phi @start %p1, @body %pn
	%min =l phi @start %y0, @body %minn
	%max =l phi @start %y0, @body %maxn
	%sum =l phi @start %y0, @body %sumn
	%cnt =l phi @start %odd0, @body %cntn
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%ltl =l extuw %lt
	%mlt =l sub 0, %ltl
	%d1 =l xor %min, %y
	%d2 =l and %d1, %mlt
	%minn =l xor %min, %d2
	%gt =w csgtl %y, %max
	%gtl =l extuw %gt
	%mgt =l sub 0, %gtl
	%e1 =l xor %max, %y
	%e2 =l and %e1, %mgt
	%maxn =l xor %max, %e2
	%sumn =l add %sum, %y
	%o =l and %y, 1
	%cntn =l add %cnt, %o
	%pn =l add %p, 8
	%i1 =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
