export function l $count_desc(l %a, l %n) {
@start
	%lim =l sub %n, 1
	%c0 =w csgtl %lim, 0
	jnz %c0, @loop, @done0
@done0
	ret 0
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%cnt1 =l add %cnt, %ge
	%i1 =l add %i, 1
	%c =w csltl %i1, %lim
	jnz %c, @body, @done
@body
	jmp @loop
@done
	ret %cnt1
}
