export function l $qsort_pairs(l %a, l %n) {
@start
	%c0 =w csltl %n, 2
	jnz %c0, @ret, @body
@body
	%nm1 =l sub %n, 1
	%last =l mul %nm1, 16
	%lp =l add %a, %last
	%lp8 =l add %lp, 8
	%mid =l shr %n, 1
	%moff =l mul %mid, 16
	%mp =l add %a, %moff
	%mp8 =l add %mp, 8
	%mk =l loadl %mp
	%mv =l loadl %mp8
	%lk =l loadl %lp
	%lv =l loadl %lp8
	storel %mk, %lp
	storel %mv, %lp8
	storel %lk, %mp
	storel %lv, %mp8
	%pk =l copy %mk
	%pv =l copy %mv
	%i =l copy 0
	%j =l copy 0
	jmp @loop
@loop
	%c1 =w csltl %j, %nm1
	jnz %c1, @check, @done
@check
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jp8 =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jp8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @eqk
@eqk
	%eq =w ceql %jk, %pk
	jnz %eq, @cmpv, @next
@cmpv
	%ltv =w csltl %jv, %pv
	jnz %ltv, @swap, @next
@swap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ip8 =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ip8
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	%i =l add %i, 1
	jmp @next
@next
	%j =l add %j, 1
	jmp @loop
@done
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ip28 =l add %ip2, 8
	%ik2 =l loadl %ip2
	%iv2 =l loadl %ip28
	storel %pk, %ip2
	storel %pv, %ip28
	storel %ik2, %lp
	storel %iv2, %lp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip3 =l add %ip2, 16
	%rest =l sub %nm1, %i
	%r2 =l call $qsort_pairs(l %ip3, l %rest)
	jmp @ret
@ret
	ret 0
}
