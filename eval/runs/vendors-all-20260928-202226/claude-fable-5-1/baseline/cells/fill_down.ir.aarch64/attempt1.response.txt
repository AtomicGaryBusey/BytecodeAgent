export function l $fill_down(l %a, l %n, l %start) {
@entry
	%c0 =w csltl %n, 1
	jnz %c0, @done, @loop
@loop
	%i =l phi @entry 0, @body %i1
	%v =l phi @entry %start, @body %v1
	%off =l mul %i, 8
	%p =l add %a, %off
	storel %v, %p
	%i1 =l add %i, 1
	%v1 =l sub %v, 1
	%c =w csltl %i1, %n
	jnz %c, @body, @done
@body
	jmp @loop
@done
	ret %n
}
