The phi in @loop names @body as a predecessor, but the back edge actually comes from @join. Fix by using @join in the phi.

export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%end =l mul %n, 8
	%end =l add %a, %end
	%p0 =l add %a, 8
@loop
	%m =l phi @start %m0, @join %m2
	%p =l phi @start %p0, @join %p2
	%done =w ceql %p, %end
	jnz %done, @ret, @body
@body
	%v =l loadl %p
	%gt =w csgtl %v, %m
	jnz %gt, @take, @keep
@take
	jmp @join
@keep
	jmp @join
@join
	%m2 =l phi @take %v, @keep %m
	%p2 =l add %p, 8
	jmp @loop
@ret
	ret %m
}
