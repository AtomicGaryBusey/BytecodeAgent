export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @next %i1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%nz =w cnel %v, 0
	jnz %nz, @next, @done
@next
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %i
}
