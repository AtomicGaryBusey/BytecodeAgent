export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%n2 =l mul %n, %n
	%np1 =l add %n, 1
	%i =l copy 0
	jmp @id_loop
@id_loop
	%c =w csltl %i, %n2
	jnz %c, @id_body, @pow_init
@id_body
	%r =l rem %i, %np1
	%z =w ceql %r, 0
	%v =l extuw %z
	%off =l mul %i, 8
	%p =l add %out, %off
	storel %v, %p
	%i =l add %i, 1
	jmp @id_loop
@pow_init
	%cnt =l copy 0
	jmp @pow_loop
@pow_loop
	%c =w csltl %cnt, %k
	jnz %c, @mul_init, @zero_init
@mul_init
	%i =l copy 0
	jmp @mul_i
@mul_i
	%c =w csltl %i, %n
	jnz %c, @mul_j_init, @copy_init
@mul_j_init
	%j =l copy 0
	jmp @mul_j
@mul_j
	%c =w csltl %j, %n
	jnz %c, @mul_t_init, @mul_i_next
@mul_t_init
	%t =l copy 0
	%s =l copy 0
	%rowbase =l mul %i, %n
	jmp @mul_t
@mul_t
	%c =w csltl %t, %n
	jnz %c, @mul_t_body, @mul_t_done
@mul_t_body
	%xi =l add %rowbase, %t
	%xo =l mul %xi, 8
	%xp =l add %out, %xo
	%x =l loadl %xp
	%yi =l mul %t, %n
	%yi =l add %yi, %j
	%yo =l mul %yi, 8
	%yp =l add %a, %yo
	%y =l loadl %yp
	%prod =l mul %x, %y
	%s =l add %s, %prod
	%t =l add %t, 1
	jmp @mul_t
@mul_t_done
	%di =l add %rowbase, %j
	%do =l mul %di, 8
	%dp =l add %tmp, %do
	storel %s, %dp
	%j =l add %j, 1
	jmp @mul_j
@mul_i_next
	%i =l add %i, 1
	jmp @mul_i
@copy_init
	%i =l copy 0
	jmp @copy_loop
@copy_loop
	%c =w csltl %i, %n2
	jnz %c, @copy_body, @pow_next
@copy_body
	%off =l mul %i, 8
	%sp =l add %tmp, %off
	%v =l loadl %sp
	%dp =l add %out, %off
	storel %v, %dp
	%i =l add %i, 1
	jmp @copy_loop
@pow_next
	%cnt =l add %cnt, 1
	jmp @pow_loop
@zero_init
	%n22 =l mul %n2, 2
	%i =l copy 0
	jmp @zero_loop
@zero_loop
	%c =w csltl %i, %n22
	jnz %c, @zero_body, @done
@zero_body
	%off =l mul %i, 8
	%p =l add %tmp, %off
	storel 0, %p
	%i =l add %i, 1
	jmp @zero_loop
@done
	ret 0
}
