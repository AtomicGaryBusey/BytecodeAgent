packbits_encode:
    xor eax, eax
    xor r8d, r8d
.Louter:
    cmp r8, rsi
    jge .Ldone
    movzx r11d, byte ptr [rdi + r8]
    mov r9d, 1
    lea r10, [r8 + 1]
.Lrun:
    cmp r10, rsi
    jge .Lrundone
    cmp r9d, 128
    jge .Lrundone
    cmp byte ptr [rdi + r10], r11b
    jne .Lrundone
    inc r9d
    inc r10
    jmp .Lrun
.Lrundone:
    cmp r9d, 3
    jb .Lliteral
    mov r10, rcx
    sub r10, rax
    cmp r10, 2
    jl .Lfail
    mov r10d, 1
    sub r10d, r9d
    mov byte ptr [rdx + rax], r10b
    mov byte ptr [rdx + rax + 1], r11b
    add rax, 2
    add r8, r9
    jmp .Louter
.Lliteral:
    mov r10, r8
.Llit:
    cmp r10, rsi
    jge .Llitdone
    lea r9, [r10 - 128]
    cmp r9, r8
    jge .Llitdone
    lea r9, [r10 + 2]
    cmp r9, rsi
    jge .Lnext
    movzx r9d, byte ptr [rdi + r10]
    cmp byte ptr [rdi + r10 + 1], r9b
    jne .Lnext
    cmp byte ptr [rdi + r10 + 2], r9b
    je .Llitdone
.Lnext:
    inc r10
    jmp .Llit
.Llitdone:
    mov r9, r10
    sub r9, r8
    mov r11, rcx
    sub r11, rax
    sub r11, r9
    cmp r11, 1
    jl .Lfail
    lea r11d, [r9 - 1]
    mov byte ptr [rdx + rax], r11b
    inc rax
.Lcopy:
    movzx r11d, byte ptr [rdi + r8]
    mov byte ptr [rdx + rax], r11b
    inc rax
    inc r8
    cmp r8, r10
    jb .Lcopy
    jmp .Louter
.Lfail:
    mov rax, -1
    ret
.Ldone:
    ret
