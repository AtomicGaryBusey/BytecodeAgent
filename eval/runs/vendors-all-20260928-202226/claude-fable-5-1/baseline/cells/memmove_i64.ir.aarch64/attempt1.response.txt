export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%le =w cslel %n, 0
	jnz %le, @done, @check
@check
	%lt =w cultl %dst, %src
	jnz %lt, @fwd_init, @bwd_init
@fwd_init
	%i =l copy 0
	jmp @fwd_loop
@fwd_loop
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i1 =l add %i, 1
	%i =l copy %i1
	%c =w csltl %i, %n
	jnz %c, @fwd_loop, @done
@bwd_init
	%j =l copy %n
	jmp @bwd_loop
@bwd_loop
	%j1 =l sub %j, 1
	%j =l copy %j1
	%boff =l mul %j, 8
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bv =l loadl %bsp
	storel %bv, %bdp
	%bc =w csgtl %j, 0
	jnz %bc, @bwd_loop, @done
@done
	ret 0
}
