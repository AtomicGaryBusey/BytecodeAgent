export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%end =w ceql %pc, %len
	jnz %end, @done, @fetch
@fetch
	%off =l mul %pc, 8
	%ip =l add %code, %off
	%op =l loadl %ip
	%t0 =w ceql %op, 0
	jnz %t0, @done, @c1
@c1
	%t1 =w ceql %op, 1
	jnz %t1, @push, @c2
@c2
	%t2 =w ceql %op, 2
	jnz %t2, @bin, @c3
@c3
	%t3 =w ceql %op, 3
	jnz %t3, @bin, @c4
@c4
	%t4 =w ceql %op, 4
	jnz %t4, @bin, @c5
@c5
	%t5 =w ceql %op, 5
	jnz %t5, @dup, @c6
@c6
	%t6 =w ceql %op, 6
	jnz %t6, @swap, @c7
@c7
	%t7 =w ceql %op, 7
	jnz %t7, @drop, @c8
@c8
	%t8 =w ceql %op, 8
	jnz %t8, @over, @c9
@c9
	%t9 =w ceql %op, 9
	jnz %t9, @bin, @c10
@c10
	%t10 =w ceql %op, 10
	jnz %t10, @jz, @c11
@c11
	%t11 =w ceql %op, 11
	jnz %t11, @jmp, @other
@other
	%pc =l add %pc, 1
	jmp @loop
@push
	%oaddr =l add %ip, 8
	%v =l loadl %oaddr
	%pa =l mul %d, 8
	%pa =l add %stack, %pa
	storel %v, %pa
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@bin
	%d =l sub %d, 1
	%bp =l mul %d, 8
	%bp =l add %stack, %bp
	%b =l loadl %bp
	%ap =l sub %bp, 8
	%a =l loadl %ap
	%pc =l add %pc, 1
	%u2 =w ceql %op, 2
	jnz %u2, @add, @b3
@add
	%r =l add %a, %b
	jmp @binstore
@b3
	%u3 =w ceql %op, 3
	jnz %u3, @sub, @b4
@sub
	%r =l sub %a, %b
	jmp @binstore
@b4
	%u4 =w ceql %op, 4
	jnz %u4, @mul, @lt
@mul
	%r =l mul %a, %b
	jmp @binstore
@lt
	%lc =w csltl %a, %b
	%r =l extuw %lc
	jmp @binstore
@binstore
	storel %r, %ap
	jmp @loop
@dup
	%tp =l sub %d, 1
	%tp =l mul %tp, 8
	%tp =l add %stack, %tp
	%tv =l loadl %tp
	%np =l add %tp, 8
	storel %tv, %np
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%tp =l sub %d, 1
	%tp =l mul %tp, 8
	%tp =l add %stack, %tp
	%sp2 =l sub %tp, 8
	%tv =l loadl %tp
	%sv =l loadl %sp2
	storel %sv, %tp
	storel %tv, %sp2
	%pc =l add %pc, 1
	jmp @loop
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%tp =l sub %d, 1
	%tp =l mul %tp, 8
	%tp =l add %stack, %tp
	%sp2 =l sub %tp, 8
	%sv =l loadl %sp2
	%np =l add %tp, 8
	storel %sv, %np
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%d =l sub %d, 1
	%vp =l mul %d, 8
	%vp =l add %stack, %vp
	%jv =l loadl %vp
	%jvz =w ceql %jv, 0
	jnz %jvz, @jmp, @jzskip
@jzskip
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%oaddr =l add %ip, 8
	%pc =l loadl %oaddr
	jmp @loop
@done
	%i =l copy %d
	%dz =w ceql %d, 0
	jnz %dz, @zero_init, @gettop
@gettop
	%tp =l sub %d, 1
	%tp =l mul %tp, 8
	%tp =l add %stack, %tp
	%res =l loadl %tp
	jmp @zloop
@zero_init
	%res =l copy 0
	jmp @zloop
@zloop
	%zc =w csltl %i, %cap
	jnz %zc, @zbody, @ret
@zbody
	%zp =l mul %i, 8
	%zp =l add %stack, %zp
	storel 0, %zp
	%i =l add %i, 1
	jmp @zloop
@ret
	ret %res
}
