export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%nm1 =l sub %n, 1
	%c0 =w csltl %i, %nm1
	jnz %c0, @inner_init, @done
@inner_init
	%j =l copy 0
@inner
	%lim =l sub %nm1, %i
	%c1 =w csltl %j, %lim
	jnz %c1, @body, @outer_next
@body
	%off =l mul %j, 8
	%p0 =l add %a, %off
	%p1 =l add %p0, 8
	%x =l loadl %p0
	%y =l loadl %p1
	%c2 =w csgtl %x, %y
	jnz %c2, @swap, @inner_next
@swap
	storel %y, %p0
	storel %x, %p1
@inner_next
	%j =l add %j, 1
	jmp @inner
@outer_next
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
