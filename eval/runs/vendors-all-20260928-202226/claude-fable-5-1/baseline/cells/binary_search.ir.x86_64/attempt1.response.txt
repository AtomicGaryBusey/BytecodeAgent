export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @start %lo0, @goright %lo1, @goleft %lo
	%hi =l phi @start %hi0, @goright %hi, @goleft %hi1
	%c =w csgtl %lo, %hi
	jnz %c, @notfound, @body
@body
	%d =l sub %hi, %lo
	%h =l shr %d, 1
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @goright, @goleft
@goright
	%lo1 =l add %mid, 1
	jmp @loop
@goleft
	%hi1 =l sub %mid, 1
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
