export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%z =w ceql %n, 0
	jnz %z, @done, @check
@check
	%lt =w cultl %dst, %src
	jnz %lt, @fwd_init, @bwd_init
@fwd_init
	%i =l copy 0
@fwd_loop
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i1 =l add %i, 1
	%i =l copy %i1
	%c =w cultl %i, %n
	jnz %c, @fwd_loop, @done
@bwd_init
	%j =l copy %n
@bwd_loop
	%j1 =l sub %j, 1
	%j =l copy %j1
	%off2 =l mul %j, 8
	%sp2 =l add %src, %off2
	%dp2 =l add %dst, %off2
	%v2 =l loadl %sp2
	storel %v2, %dp2
	%c2 =w cnel %j, 0
	jnz %c2, @bwd_loop, @done
@done
	ret 0
}
