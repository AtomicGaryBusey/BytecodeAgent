collatz_max:
	xor eax, eax
	mov rcx, 1
.Louter:
	cmp rcx, rdi
	jg .Ldone
	mov rdx, rcx
	xor r8d, r8d
.Linner:
	cmp rdx, 1
	je .Lend
	test dl, 1
	jnz .Lodd
	shr rdx, 1
	inc r8
	jmp .Linner
.Lodd:
	lea rdx, [rdx + rdx*2 + 1]
	inc r8
	jmp .Linner
.Lend:
	cmp r8, rax
	cmovg rax, r8
	inc rcx
	jmp .Louter
.Ldone:
	ret
