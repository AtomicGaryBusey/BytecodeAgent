export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
@outer
	%lim =l sub %n, 1
	%c0 =w csltl %i, %lim
	jnz %c0, @inner_init, @done
@inner_init
	%j =l copy 0
	%jlim =l sub %lim, %i
@inner
	%c1 =w csltl %j, %jlim
	jnz %c1, @body, @outer_inc
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%c2 =w csgtl %x, %y
	jnz %c2, @swap, @inner_inc
@swap
	storel %y, %p
	storel %x, %q
@inner_inc
	%j =l add %j, 1
	jmp @inner
@outer_inc
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
