packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movsx eax, byte ptr [rdi + r8]
    inc r8
    cmp eax, -128
    je .Lloop
    test eax, eax
    js .Lrepeat
    inc eax
    mov r10d, eax
    lea r11, [r8 + r10]
    cmp r11, rsi
    ja .Lerr2
.Lcopy:
    cmp r9, rcx
    jae .Lerr1
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r10
    jnz .Lcopy
    jmp .Lloop
.Lrepeat:
    cmp r8, rsi
    jae .Lerr2
    mov r10d, 1
    sub r10d, eax
    movzx eax, byte ptr [rdi + r8]
    inc r8
.Lrep:
    cmp r9, rcx
    jae .Lerr1
    mov byte ptr [rdx + r9], al
    inc r9
    dec r10
    jnz .Lrep
    jmp .Lloop
.Ldone:
    mov rax, r9
    ret
.Lerr1:
    mov rax, -1
    ret
.Lerr2:
    mov rax, -2
    ret
