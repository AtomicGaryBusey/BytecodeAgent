.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.align 4
inbuf:
    .zero 65536
linebuf:
    .zero 128

.text
.globl _start
_start:
    // Read all of stdin into inbuf; x20 = total bytes read
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
1:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUFSIZE
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    ble 2f
    add x20, x20, x0
    b 1b
2:
    // Dump rows; x21 = current offset
    mov x21, #0
3:
    cmp x21, x20
    bge 4f
    sub x2, x20, x21
    cmp x2, #16
    ble 5f
    mov x2, #16
5:
    mov x0, x21
    add x1, x19, x21
    bl fmtrow
    add x21, x21, #16
    b 3b
4:
    // Final line: total as 8 hex digits + newline
    adrp x3, linebuf
    add x3, x3, :lo12:linebuf
    adrp x5, hexdig
    add x5, x5, :lo12:hexdig
    mov w6, #28
6:
    lsr w7, w20, w6
    and w7, w7, #0xf
    ldrb w7, [x5, x7]
    strb w7, [x3], #1
    subs w6, w6, #4
    bge 6b
    mov w7, #'\n'
    strb w7, [x3], #1
    adrp x1, linebuf
    add x1, x1, :lo12:linebuf
    sub x2, x3, x1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// fmtrow(x0 = offset, x1 = data pointer, x2 = byte count n)
// Builds one output row in linebuf and writes it to stdout.
// Uses only caller-saved registers.
fmtrow:
    adrp x3, linebuf
    add x3, x3, :lo12:linebuf
    mov x4, x3
    adrp x5, hexdig
    add x5, x5, :lo12:hexdig
    // Offset: 8 lowercase hex digits
    mov w6, #28
1:
    lsr w7, w0, w6
    and w7, w7, #0xf
    ldrb w7, [x5, x7]
    strb w7, [x4], #1
    subs w6, w6, #4
    bge 1b
    // Two spaces
    mov w7, #' '
    strb w7, [x4], #1
    strb w7, [x4], #1
    // 16 hex slots, extra space before slot 8
    mov w6, #0
2:
    cmp w6, #16
    bge 3f
    cmp w6, #8
    bne 4f
    strb w7, [x4], #1
4:
    cmp w6, w2
    bge 5f
    ldrb w8, [x1, x6]
    lsr w9, w8, #4
    ldrb w9, [x5, x9]
    strb w9, [x4], #1
    and w9, w8, #0xf
    ldrb w9, [x5, x9]
    strb w9, [x4], #1
    mov w9, #' '
    strb w9, [x4], #1
    b 6f
5:
    mov w9, #' '
    strb w9, [x4], #1
    strb w9, [x4], #1
    strb w9, [x4], #1
6:
    add w6, w6, #1
    b 2b
3:
    // Space, pipe, ASCII column, pipe, newline
    mov w9, #' '
    strb w9, [x4], #1
    mov w9, #'|'
    strb w9, [x4], #1
    mov w6, #0
7:
    cmp w6, w2
    bge 8f
    ldrb w8, [x1, x6]
    cmp w8, #0x20
    blo 9f
    cmp w8, #0x7e
    bls 10f
9:
    mov w8, #'.'
10:
    strb w8, [x4], #1
    add w6, w6, #1
    b 7b
8:
    mov w9, #'|'
    strb w9, [x4], #1
    mov w9, #'\n'
    strb w9, [x4], #1
    // write(1, linebuf, length)
    sub x2, x4, x3
    mov x0, #1
    mov x1, x3
    mov x8, #SYS_write
    svc #0
    ret
