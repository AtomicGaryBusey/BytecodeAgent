.globl _start

.section .rodata
hexdigits: .ascii "0123456789abcdef"

.bss
.align 16
inbuf:   .zero 65536
linebuf: .zero 128

.text
_start:
    # ---- read all of stdin into inbuf ----
    xor r12, r12                 # r12 = total bytes read
read_loop:
    mov rdx, 65536
    sub rdx, r12                 # space left
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi                 # fd 0
    xor eax, eax                 # read = 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # ---- dump rows ----
    xor r13, r13                 # r13 = current offset
row_loop:
    cmp r13, r12
    jae final_line
    # r14 = n = min(16, total - offset)
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    lea rdx, [rip + inbuf]
    add rdx, r13                 # rdx = row base
    lea r15, [rip + linebuf]     # r15 = output cursor

    # offset: 8 lowercase hex digits
    mov ecx, 28
off_digits:
    mov rax, r13
    shr rax, cl
    and eax, 15
    lea r8, [rip + hexdigits]
    mov al, byte ptr [r8 + rax]
    mov byte ptr [r15], al
    inc r15
    sub ecx, 4
    jns off_digits
    mov byte ptr [r15], 0x20
    mov byte ptr [r15 + 1], 0x20
    add r15, 2

    # 16 hex slots
    xor ecx, ecx                 # i = 0..15
slot_loop:
    cmp rcx, r14
    jae slot_blank
    movzx eax, byte ptr [rdx + rcx]
    mov r8d, eax
    shr r8d, 4
    lea r9, [rip + hexdigits]
    mov r8b, byte ptr [r9 + r8]
    mov byte ptr [r15], r8b
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [r15 + 1], al
    mov byte ptr [r15 + 2], 0x20
    jmp slot_next
slot_blank:
    mov byte ptr [r15], 0x20
    mov byte ptr [r15 + 1], 0x20
    mov byte ptr [r15 + 2], 0x20
slot_next:
    add r15, 3
    cmp rcx, 7
    jne no_extra
    mov byte ptr [r15], 0x20
    inc r15
no_extra:
    inc rcx
    cmp rcx, 16
    jb slot_loop

    # separator and ASCII column
    mov byte ptr [r15], 0x20
    mov byte ptr [r15 + 1], 0x7c   # '|'
    add r15, 2
    xor ecx, ecx
ascii_loop:
    cmp rcx, r14
    jae ascii_done
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 0x20
    jb make_dot
    cmp al, 0x7e
    jbe ascii_store
make_dot:
    mov al, 0x2e                   # '.'
ascii_store:
    mov byte ptr [r15], al
    inc r15
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r15], 0x7c       # '|'
    mov byte ptr [r15 + 1], 0x0a   # '\n'
    add r15, 2

    # write the row
    lea rsi, [rip + linebuf]
    mov rdx, r15
    sub rdx, rsi
    mov edi, 1
    mov eax, 1                     # write = 1
    syscall

    add r13, r14
    jmp row_loop

final_line:
    # total length as 8 hex digits + newline
    lea r15, [rip + linebuf]
    mov ecx, 28
fin_digits:
    mov rax, r12
    shr rax, cl
    and eax, 15
    lea r8, [rip + hexdigits]
    mov al, byte ptr [r8 + rax]
    mov byte ptr [r15], al
    inc r15
    sub ecx, 4
    jns fin_digits
    mov byte ptr [r15], 0x0a
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov edx, 9
    mov eax, 1                     # write = 1
    syscall

    mov eax, 60                    # exit = 60
    xor edi, edi
    syscall
