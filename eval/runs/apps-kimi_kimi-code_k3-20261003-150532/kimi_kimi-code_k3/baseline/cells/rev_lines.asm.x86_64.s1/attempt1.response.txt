.globl _start

.equ INBUF_SIZE, 65536

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, INBUF_SIZE
    sub rdx, rbx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [r12 + rbx]
    lea rdi, [rip + outbuf]
    mov r14, r12
scan:
    cmp r12, r13
    jae tail
    cmp byte ptr [r12], 10
    je emit
    inc r12
    jmp scan
emit:
    mov rax, r12
rev_copy:
    cmp rax, r14
    jbe rev_done
    dec rax
    mov cl, byte ptr [rax]
    mov byte ptr [rdi], cl
    inc rdi
    jmp rev_copy
rev_done:
    mov byte ptr [rdi], 10
    inc rdi
    inc r12
    mov r14, r12
    jmp scan
tail:
    cmp r14, r13
    jae flush
    mov rax, r13
tail_copy:
    cmp rax, r14
    jbe tail_done
    dec rax
    mov cl, byte ptr [rax]
    mov byte ptr [rdi], cl
    inc rdi
    jmp tail_copy
tail_done:
    mov byte ptr [rdi], 10
    inc rdi
flush:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz quit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle quit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
quit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inbuf:
    .zero INBUF_SIZE
outbuf:
    .zero INBUF_SIZE + 1
