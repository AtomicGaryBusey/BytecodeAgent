// hexdump: canonical hex+ASCII dump of stdin, AArch64 Linux, no libc
// syscalls: read=63, write=64, exit=93

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.bss
.align 4
inbuf:
    .zero 65536
outbuf:
    .zero 128

.text
.globl _start
_start:
    // read all of stdin into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf     // x19 = input buffer base
    mov x20, #0                   // x20 = total bytes read
    mov x21, #65536               // x21 = capacity
1:
    cmp x20, x21
    b.hs 2f
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, 2f
    add x20, x20, x0
    b 1b
2:
    // dump rows
    mov x22, #0                   // x22 = current offset
3:
    cmp x22, x20
    b.hs 4f
    sub x23, x20, x22             // x23 = n = min(16, total - off)
    cmp x23, #16
    b.ls 5f
    mov x23, #16
5:
    bl print_row
    add x22, x22, x23
    b 3b
4:
    // final line: total as 8 hex digits + newline
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    mov x14, x20
    bl emit_hex8
    mov w11, #10
    strb w11, [x9], #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x9, x1
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// emit_hex8: write 8 lowercase hex digits of x14 at [x9]; x9 advances
// clobbers x10, w15
emit_hex8:
    mov x10, #28
1:
    lsr x15, x14, x10
    and w15, w15, #15
    cmp w15, #10
    b.lo 2f
    add w15, w15, #87             // 'a' - 10
    b 3f
2:
    add w15, w15, #48             // '0'
3:
    strb w15, [x9], #1
    subs x10, x10, #4
    b.hs 1b
    ret

// emit_byte_hex: write 2 lowercase hex digits of w14 at [x9]; x9 advances
// clobbers w15
emit_byte_hex:
    lsr w15, w14, #4
    cmp w15, #10
    b.lo 1f
    add w15, w15, #87
    b 2f
1:
    add w15, w15, #48
2:
    strb w15, [x9], #1
    and w15, w14, #15
    cmp w15, #10
    b.lo 3f
    add w15, w15, #87
    b 4f
3:
    add w15, w15, #48
4:
    strb w15, [x9], #1
    ret

// print_row: print one row
// x19 = input base, x22 = row offset, x23 = n (1..16)
print_row:
    str x30, [sp, #-16]!
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    mov x14, x22
    bl emit_hex8                  // 8-digit offset
    mov w11, #32                  // two spaces
    strb w11, [x9], #1
    strb w11, [x9], #1
    mov x12, #0                   // i = 0..15
1:
    cmp x12, #16
    b.hs 5f
    cmp x12, x23
    b.hs 2f
    add x13, x19, x22
    ldrb w14, [x13, x12]
    bl emit_byte_hex              // "xx "
    mov w11, #32
    strb w11, [x9], #1
    b 3f
2:
    mov w11, #32                  // "   " for missing byte
    strb w11, [x9], #1
    strb w11, [x9], #1
    strb w11, [x9], #1
3:
    cmp x12, #7                   // extra space after position 7
    b.ne 4f
    mov w11, #32
    strb w11, [x9], #1
4:
    add x12, x12, #1
    b 1b
5:
    mov w11, #32                  // " |"
    strb w11, [x9], #1
    mov w11, #124
    strb w11, [x9], #1
    mov x12, #0
6:
    cmp x12, x23
    b.hs 9f
    add x13, x19, x22
    ldrb w14, [x13, x12]
    cmp w14, #32
    b.lo 7f
    cmp w14, #126
    b.ls 8f
7:
    mov w14, #46                  // '.'
8:
    strb w14, [x9], #1
    add x12, x12, #1
    b 6b
9:
    mov w11, #124                 // "|"
    strb w11, [x9], #1
    mov w11, #10                  // "\n"
    strb w11, [x9], #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x9, x1
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    ldr x30, [sp], #16
    ret
