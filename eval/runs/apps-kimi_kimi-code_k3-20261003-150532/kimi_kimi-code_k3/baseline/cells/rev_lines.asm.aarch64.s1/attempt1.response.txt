    .text
    .globl _start

    .equ SYS_read, 63
    .equ SYS_write, 64
    .equ SYS_exit, 93
    .equ BUFSZ, 65536

_start:
    // x19 = input buffer base, x20 = total bytes read
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, 0
1:
    mov x0, 0
    add x1, x19, x20
    mov x2, BUFSZ
    sub x2, x2, x20
    cmp x2, 0
    ble 2f
    mov x8, SYS_read
    svc #0
    cmp x0, 0
    ble 2f
    add x20, x20, x0
    b 1b
2:
    // x23 = output buffer base, x24 = output length
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, 0
    // x21 = scan position, x22 = current line start
    mov x21, 0
    mov x22, 0
3:
    cmp x21, x20
    bhs 4f
    ldrb w3, [x19, x21]
    cmp w3, 0x0A
    beq 5f
    add x21, x21, 1
    b 3b
5:
    // emit line [x22, x21) reversed
    add x0, x19, x22
    sub x1, x21, x22
    bl emit_rev
    add x21, x21, 1
    mov x22, x21
    b 3b
4:
    // final line without trailing newline, if any
    cmp x22, x20
    bhs 6f
    add x0, x19, x22
    sub x1, x20, x22
    bl emit_rev
6:
    // write output buffer to stdout
    mov x25, 0
7:
    cmp x25, x24
    bhs 8f
    mov x0, 1
    add x1, x23, x25
    sub x2, x24, x25
    mov x8, SYS_write
    svc #0
    cmp x0, 0
    ble 8f
    add x25, x25, x0
    b 7b
8:
    mov x0, 0
    mov x8, SYS_exit
    svc #0

// emit_rev: append bytes [x0, x0+x1) reversed, then '\n', to outbuf
// clobbers x2, x3, x4, x5; advances x24
emit_rev:
    add x2, x23, x24
    mov x3, 0
1:
    cmp x3, x1
    bhs 2f
    sub x4, x1, x3
    sub x4, x4, 1
    ldrb w5, [x0, x4]
    strb w5, [x2, x3]
    add x3, x3, 1
    b 1b
2:
    mov w5, 0x0A
    strb w5, [x2, x1]
    add x24, x24, x1
    add x24, x24, 1
    ret

    .bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072
