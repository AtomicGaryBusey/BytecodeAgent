    .text
    .globl _start
_start:
    // ---- read all of stdin into inbuf ----
    mov x9, #0                      // total bytes read
1:
    mov x0, #49152
    sub x2, x0, x9                  // space remaining
    cbz x2, 2f
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x9
    mov x0, #0                      // fd 0
    mov x8, #63                     // read
    svc #0
    cbz x0, 2f                      // end of input
    add x9, x9, x0
    b 1b
2:
    // ---- encode ----
    adrp x10, inbuf
    add x10, x10, :lo12:inbuf       // input cursor
    mov x11, x9                     // bytes remaining
    adrp x12, outbuf
    add x12, x12, :lo12:outbuf      // output cursor
    mov x13, #0                     // column count
    adrp x14, b64tab
    add x14, x14, :lo12:b64tab

    cmp x11, #3
    b.lt 5f
3:                                  // full 3-byte groups
    ldrb w0, [x10]
    ldrb w1, [x10, #1]
    ldrb w2, [x10, #2]
    lsr w3, w0, #2                  // c0 = b0 >> 2
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12]
    and w3, w0, #3                  // c1 = ((b0 & 3) << 4) | (b1 >> 4)
    lsl w3, w3, #4
    orr w3, w3, w1, lsr #4
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12, #1]
    and w3, w1, #15                 // c2 = ((b1 & 15) << 2) | (b2 >> 6)
    lsl w3, w3, #2
    orr w3, w3, w2, lsr #6
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12, #2]
    and w3, w2, #63                 // c3 = b2 & 63
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12, #3]
    add x12, x12, #4
    add x13, x13, #4
    cmp x13, #76
    b.ne 4f
    mov w3, #10                     // wrap line at 76 chars
    strb w3, [x12], #1
    mov x13, #0
4:
    add x10, x10, #3
    sub x11, x11, #3
    cmp x11, #3
    b.ge 3b
5:                                  // trailing 1 or 2 bytes
    cbz x11, 7f
    ldrb w0, [x10]
    lsr w3, w0, #2
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12]
    and w3, w0, #3
    lsl w3, w3, #4
    cmp x11, #1
    b.eq 6f
    ldrb w1, [x10, #1]              // 2 bytes left: XX X=
    orr w3, w3, w1, lsr #4
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12, #1]
    and w3, w1, #15
    lsl w3, w3, #2
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12, #2]
    mov w3, #'='
    strb w3, [x12, #3]
    add x12, x12, #4
    add x13, x13, #4
    b 7f
6:                                  // 1 byte left: XX==
    ldrb w3, [x14, w3, uxtw]
    strb w3, [x12, #1]
    mov w3, #'='
    strb w3, [x12, #2]
    strb w3, [x12, #3]
    add x12, x12, #4
    add x13, x13, #4
7:
    cbz x13, 8f                     // final newline unless line already wrapped
    mov w3, #10
    strb w3, [x12], #1
8:
    // ---- write output ----
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x12, x1                 // total length
    cbz x2, 9f
10:
    mov x0, #1                      // fd 1
    mov x8, #64                     // write
    svc #0
    add x1, x1, x0
    subs x2, x2, x0
    b.ne 10b
9:
    mov x0, #0
    mov x8, #93                     // exit
    svc #0

    .section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .bss
    .align 4
inbuf:
    .zero 49152
outbuf:
    .zero 66560
