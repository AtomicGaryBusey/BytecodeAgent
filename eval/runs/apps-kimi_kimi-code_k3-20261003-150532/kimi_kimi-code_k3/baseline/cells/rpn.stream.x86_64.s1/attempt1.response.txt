.globl _start

.text
_start:
    lea rbx, [rip + inbuf]
    lea r13, [rip + inbuf + 65536]
read_loop:
    cmp rbx, r13
    jae read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov rdx, r13
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r13, rbx
    lea r12, [rip + inbuf]

line_loop:
    cmp r12, r13
    jae all_done
    mov r14, r12
find_nl:
    cmp r14, r13
    jae line_ready
    cmp byte ptr [r14], 10
    je line_ready
    inc r14
    jmp find_nl
line_ready:
    lea r15, [rip + vstack]
    mov rbp, r12

token_loop:
    cmp rbp, r14
    jae end_of_line
    mov al, byte ptr [rbp]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp have_token
skip_one:
    inc rbp
    jmp token_loop

have_token:
    lea rcx, [rbp + 1]
scan_tok:
    cmp rcx, r14
    jae tok_ready
    mov dl, byte ptr [rcx]
    cmp dl, 32
    je tok_ready
    cmp dl, 9
    je tok_ready
    inc rcx
    jmp scan_tok
tok_ready:
    mov r8, rcx
    mov rdx, rcx
    sub rdx, rbp
    cmp rdx, 1
    ja try_number
    mov al, byte ptr [rbp]
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod

try_number:
    mov r9, rbp
    xor r10d, r10d
    mov al, byte ptr [r9]
    cmp al, 45
    jne digits_init
    mov r10d, 1
    inc r9
digits_init:
    mov rcx, r8
    sub rcx, r9
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
digit_loop:
    cmp r9, r8
    jae number_done
    movzx edx, byte ptr [r9]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r9
    jmp digit_loop
number_done:
    test r10d, r10d
    jz do_push
    neg rax
do_push:
    lea rcx, [rip + vstack + 8192]
    cmp r15, rcx
    jae line_error
    mov qword ptr [r15], rax
    add r15, 8
    mov rbp, r8
    jmp token_loop

op_add:
    lea rcx, [rip + vstack + 16]
    cmp r15, rcx
    jb line_error
    mov rdx, [r15 - 8]
    mov rax, [r15 - 16]
    add rax, rdx
    jmp store_result
op_sub:
    lea rcx, [rip + vstack + 16]
    cmp r15, rcx
    jb line_error
    mov rdx, [r15 - 8]
    mov rax, [r15 - 16]
    sub rax, rdx
    jmp store_result
op_mul:
    lea rcx, [rip + vstack + 16]
    cmp r15, rcx
    jb line_error
    mov rdx, [r15 - 8]
    mov rax, [r15 - 16]
    imul rax, rdx
    jmp store_result
op_div:
    lea rcx, [rip + vstack + 16]
    cmp r15, rcx
    jb line_error
    mov rdx, [r15 - 8]
    mov rax, [r15 - 16]
    test rdx, rdx
    jz line_error
    cmp rdx, -1
    jne 1f
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    je line_error
1:
    mov rcx, rdx
    cqo
    idiv rcx
    jmp store_result
op_mod:
    lea rcx, [rip + vstack + 16]
    cmp r15, rcx
    jb line_error
    mov rdx, [r15 - 8]
    mov rax, [r15 - 16]
    test rdx, rdx
    jz line_error
    cmp rdx, -1
    jne 1f
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    je line_error
1:
    mov rcx, rdx
    cqo
    idiv rcx
    mov rax, rdx
store_result:
    mov [r15 - 16], rax
    sub r15, 8
    mov rbp, r8
    jmp token_loop

end_of_line:
    lea rax, [rip + vstack + 8]
    cmp r15, rax
    jne line_error
    mov rax, [rip + vstack]
    call print_number
    jmp next_line

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 6
    syscall
next_line:
    lea r12, [r14 + 1]
    jmp line_loop

all_done:
    mov eax, 60
    xor edi, edi
    syscall

print_number:
    lea rsi, [rip + numbuf + 32]
    dec rsi
    mov byte ptr [rsi], 10
    mov ecx, 10
    xor r8d, r8d
    test rax, rax
    jns 1f
    neg rax
    mov r8d, 1
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz 1b
    test r8d, r8d
    jz 2f
    dec rsi
    mov byte ptr [rsi], 45
2:
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.align 8
inbuf:
    .zero 65536
vstack:
    .zero 8192
numbuf:
    .zero 32
