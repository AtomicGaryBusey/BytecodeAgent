.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #1

read_more:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #8192
    mov x8, #63
    svc #0
    cbz x0, finish
    add x21, x21, x0
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf
    mov x24, x0

scan_loop:
    cbz x24, read_more
    ldrb w0, [x23], #1
    sub x24, x24, #1
    cmp w0, #10
    b.ne no_newline
    add x19, x19, #1
no_newline:
    cmp w0, #32
    b.eq is_space
    sub w2, w0, #9
    cmp w2, #4
    b.ls is_space
    cbz x22, no_word_start
    add x20, x20, #1
no_word_start:
    mov x22, #0
    b scan_loop
is_space:
    mov x22, #1
    b scan_loop

finish:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x0, x19
    mov x1, x23
    bl append_num
    mov x23, x1
    mov w0, #32
    strb w0, [x23], #1
    mov x0, x20
    mov x1, x23
    bl append_num
    mov x23, x1
    mov w0, #32
    strb w0, [x23], #1
    mov x0, x21
    mov x1, x23
    bl append_num
    mov x23, x1
    mov w0, #10
    strb w0, [x23], #1

    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    sub x25, x23, x24
write_loop:
    cbz x25, exit_ok
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x24, x24, x0
    sub x25, x25, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

append_num:
    mov x2, #10
    adrp x6, numtmp
    add x6, x6, :lo12:numtmp
    add x6, x6, #32
    mov x3, x6
    cbnz x0, an_loop
    mov w4, #48
    strb w4, [x1], #1
    ret
an_loop:
    udiv x4, x0, x2
    msub x5, x4, x2, x0
    add w5, w5, #48
    strb w5, [x3, #-1]!
    mov x0, x4
    cbnz x0, an_loop
an_copy:
    cmp x3, x6
    b.eq an_done
    ldrb w4, [x3], #1
    strb w4, [x1], #1
    b an_copy
an_done:
    ret

.bss
inbuf:  .space 8192
outbuf: .space 128
numtmp: .space 32
