// rev_lines: print every line of stdin with its bytes reversed.
// No libc, raw syscalls only. AArch64, GNU as syntax.

	.text
	.globl _start

.equ SYS_read,  63
.equ SYS_write, 64
.equ SYS_exit,  93
.equ INBUF_SIZE, 65536

_start:
	// ---- read all of stdin into inbuf ----
	adrp x19, inbuf
	add  x19, x19, :lo12:inbuf	// x19 = current read target
	mov  x20, #0			// x20 = total bytes read so far
1:
	mov  x2, #INBUF_SIZE
	sub  x2, x2, x20		// space remaining
	cbz  x2, 2f
	mov  x0, #0			// fd 0
	mov  x1, x19
	mov  x8, #SYS_read
	svc  #0
	cbz  x0, 2f			// 0 = EOF
	add  x19, x19, x0
	add  x20, x20, x0
	b    1b
2:
	// ---- scan lines, emit reversed copy + newline into outbuf ----
	adrp x21, inbuf
	add  x21, x21, :lo12:inbuf	// x21 = input base
	adrp x22, outbuf
	add  x22, x22, :lo12:outbuf	// x22 = output base
	mov  x23, x22			// x23 = output write pointer
	mov  x24, #0			// x24 = line start index
3:
	cmp  x24, x20
	b.ge 6f				// no more input
	mov  x25, x24			// x25 = scan index for line end
4:
	cmp  x25, x20
	b.ge 5f				// end of input ends the line
	ldrb w10, [x21, x25]
	cmp  w10, #0x0A
	b.eq 5f				// newline ends the line
	add  x25, x25, #1
	b    4b
5:
	// line = [x24, x25); copy it reversed to output
	mov  x9, x25
7:
	cmp  x9, x24
	b.le 8f
	sub  x9, x9, #1
	ldrb w10, [x21, x9]
	strb w10, [x23], #1
	b    7b
8:
	mov  w10, #0x0A
	strb w10, [x23], #1		// newline after every line
	cmp  x25, x20
	b.ge 6f				// final line had no newline
	add  x24, x25, #1		// skip the newline byte
	b    3b
6:
	// ---- write the whole output ----
	sub  x2, x23, x22
	cbz  x2, 9f
	mov  x0, #1			// fd 1
	mov  x1, x22
	mov  x8, #SYS_write
	svc  #0
9:
	mov  x0, #0
	mov  x8, #SYS_exit
	svc  #0

	.bss
	.balign 16
inbuf:
	.zero 65536
	.balign 16
outbuf:
	.zero 65537			// input plus at most one extra newline
