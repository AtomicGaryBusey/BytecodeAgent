.globl _start

.section .rodata
hexdig:
	.ascii "0123456789abcdef"

.text
_start:
	xor r12d, r12d
1:
	lea rsi, [rip + inbuf]
	add rsi, r12
	mov rdx, 65536
	sub rdx, r12
	jz 2f
	xor edi, edi
	xor eax, eax
	syscall
	test rax, rax
	jz 2f
	add r12, rax
	jmp 1b
2:
	xor r13d, r13d
row_loop:
	cmp r13, r12
	jae final
	mov r14, r12
	sub r14, r13
	cmp r14, 16
	jbe 3f
	mov r14, 16
3:
	lea r15, [rip + outbuf]
	mov ecx, 28
4:
	mov eax, r13d
	shr eax, cl
	and eax, 15
	lea rdx, [rip + hexdig]
	mov al, byte ptr [rdx + rax]
	mov byte ptr [r15], al
	inc r15
	sub ecx, 4
	jns 4b
	mov byte ptr [r15], 32
	mov byte ptr [r15 + 1], 32
	add r15, 2
	xor ebx, ebx
slot_loop:
	cmp rbx, r14
	jae pad_slot
	lea rdx, [r13 + rbx]
	lea rsi, [rip + inbuf]
	movzx eax, byte ptr [rsi + rdx]
	mov edx, eax
	shr edx, 4
	lea rsi, [rip + hexdig]
	mov dl, byte ptr [rsi + rdx]
	mov byte ptr [r15], dl
	and eax, 15
	mov al, byte ptr [rsi + rax]
	mov byte ptr [r15 + 1], al
	mov byte ptr [r15 + 2], 32
	jmp slot_done
pad_slot:
	mov byte ptr [r15], 32
	mov byte ptr [r15 + 1], 32
	mov byte ptr [r15 + 2], 32
slot_done:
	add r15, 3
	cmp ebx, 7
	jne 5f
	mov byte ptr [r15], 32
	inc r15
5:
	inc ebx
	cmp ebx, 16
	jb slot_loop
	mov byte ptr [r15], 32
	mov byte ptr [r15 + 1], 124
	add r15, 2
	xor ebx, ebx
ascii_loop:
	cmp rbx, r14
	jae ascii_done
	lea rdx, [r13 + rbx]
	lea rsi, [rip + inbuf]
	movzx eax, byte ptr [rsi + rdx]
	cmp al, 0x20
	jb put_dot
	cmp al, 0x7e
	jbe store_ch
put_dot:
	mov al, 46
store_ch:
	mov byte ptr [r15], al
	inc r15
	inc ebx
	jmp ascii_loop
ascii_done:
	mov byte ptr [r15], 124
	mov byte ptr [r15 + 1], 10
	add r15, 2
	lea rsi, [rip + outbuf]
	mov rdx, r15
	sub rdx, rsi
	mov edi, 1
	mov eax, 1
	syscall
	add r13, r14
	jmp row_loop
final:
	lea r15, [rip + outbuf]
	mov ecx, 28
6:
	mov eax, r12d
	shr eax, cl
	and eax, 15
	lea rdx, [rip + hexdig]
	mov al, byte ptr [rdx + rax]
	mov byte ptr [r15], al
	inc r15
	sub ecx, 4
	jns 6b
	mov byte ptr [r15], 10
	lea rsi, [rip + outbuf]
	mov edx, 9
	mov edi, 1
	mov eax, 1
	syscall
	xor edi, edi
	mov eax, 60
	syscall

.bss
.align 16
inbuf:
	.zero 65536
outbuf:
	.zero 256
