.equ BUF_SIZE, 65536

.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jle input_done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + inbuf]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne no_line
    inc r12
no_line:
    cmp al, 32
    je is_ws
    cmp al, 9
    je is_ws
    cmp al, 10
    je is_ws
    cmp al, 11
    je is_ws
    cmp al, 12
    je is_ws
    cmp al, 13
    je is_ws
    test r15d, r15d
    jnz scan_next
    inc r13
    mov r15d, 1
    jmp scan_next
is_ws:
    xor r15d, r15d
scan_next:
    dec rcx
    jnz scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call fmt_u64
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call fmt_u64
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call fmt_u64
    mov byte ptr [rdi], 10
    inc rdi

    lea rdx, [rip + outbuf]
    mov rsi, rdx
    sub rdi, rdx
    mov rdx, rdi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

fmt_u64:
    lea r8, [rip + numtmp + 32]
    mov ecx, 10
    test rax, rax
    jnz fmt_loop
    mov byte ptr [rdi], 48
    inc rdi
    ret
fmt_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz fmt_loop
    lea rdx, [rip + numtmp + 32]
    mov rsi, r8
    sub rdx, rsi
    mov rcx, rdx
    rep movsb
    ret

.bss
.align 16
inbuf:
    .space BUF_SIZE
numtmp:
    .space 32
outbuf:
    .space 80
