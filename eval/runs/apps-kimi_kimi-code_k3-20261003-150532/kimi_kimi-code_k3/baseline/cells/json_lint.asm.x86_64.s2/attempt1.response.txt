.globl _start

.section .rodata
msg_ok:    .ascii "ok\n"
msg_err:   .ascii "error at "
str_true:  .ascii "true"
str_false: .ascii "false"
str_null:  .ascii "null"

.bss
.align 16
inbuf:  .zero 131072
outbuf: .zero 32

.text
_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 131072
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rbx, [rip + inbuf]
    lea r13, [rip + inbuf]
    add r13, r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jne parse_error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
1:
    cmp rbx, r13
    jae 2f
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
2:
    ret
3:
    inc rbx
    jmp 1b

parse_value:
    call skip_ws
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je lit_true
    cmp al, 'f'
    je lit_false
    cmp al, 'n'
    je lit_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
    jmp parse_number

lit_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp lit_common
lit_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp lit_common
lit_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
lit_common:
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, byte ptr [rsi]
    jne parse_error
    inc rbx
    inc rsi
    dec ecx
    jnz lit_common
    ret

parse_string:
    inc rbx
1:
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je 4f
    cmp al, 0x5C
    je 5f
    cmp al, 0x20
    jb parse_error
    inc rbx
    jmp 1b
4:
    inc rbx
    ret
5:
    inc rbx
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, 'u'
    je 6f
    cmp al, '"'
    je 7f
    cmp al, 0x5C
    je 7f
    cmp al, '/'
    je 7f
    cmp al, 'b'
    je 7f
    cmp al, 'f'
    je 7f
    cmp al, 'n'
    je 7f
    cmp al, 'r'
    je 7f
    cmp al, 't'
    je 7f
    jmp parse_error
7:
    inc rbx
    jmp 1b
6:
    inc rbx
    mov ecx, 4
8:
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    jbe 9f
    cmp al, 'A'
    jb parse_error
    cmp al, 'F'
    jbe 9f
    cmp al, 'a'
    jb parse_error
    cmp al, 'f'
    ja parse_error
9:
    inc rbx
    dec ecx
    jnz 8b
    jmp 1b

parse_number:
    movzx eax, byte ptr [rbx]
    cmp al, '-'
    jne 1f
    inc rbx
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
1:
    cmp al, '0'
    je 2f
    cmp al, '1'
    jb parse_error
    cmp al, '9'
    ja parse_error
3:
    inc rbx
    cmp rbx, r13
    jae 10f
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb 10f
    cmp al, '9'
    jbe 3b
    jmp 10f
2:
    inc rbx
10:
    cmp rbx, r13
    jae 20f
    movzx eax, byte ptr [rbx]
    cmp al, '.'
    jne 20f
    inc rbx
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
11:
    inc rbx
    cmp rbx, r13
    jae 20f
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb 20f
    cmp al, '9'
    jbe 11b
20:
    cmp rbx, r13
    jae 30f
    movzx eax, byte ptr [rbx]
    cmp al, 'e'
    je 21f
    cmp al, 'E'
    je 21f
    jmp 30f
21:
    inc rbx
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je 22f
    cmp al, '-'
    je 22f
    jmp 23f
22:
    inc rbx
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
23:
    cmp al, '0'
    jb parse_error
    cmp al, '9'
    ja parse_error
24:
    inc rbx
    cmp rbx, r13
    jae 30f
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb 30f
    cmp al, '9'
    jbe 24b
30:
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, ']'
    jne 1f
    inc rbx
    ret
1:
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je 2f
    cmp al, ']'
    jne parse_error
    inc rbx
    ret
2:
    inc rbx
    jmp 1b

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '}'
    jne 1f
    inc rbx
    ret
1:
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, ':'
    jne parse_error
    inc rbx
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae parse_error
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je 2f
    cmp al, '}'
    jne parse_error
    inc rbx
    ret
2:
    inc rbx
    call skip_ws
    jmp 1b

parse_error:
    lea rax, [rip + inbuf]
    sub rbx, rax
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rsi, [rip + outbuf + 24]
    mov rax, rbx
    mov r8, 10
1:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    mov byte ptr [rip + outbuf + 24], 0x0A
    lea rdx, [rip + outbuf + 25]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
