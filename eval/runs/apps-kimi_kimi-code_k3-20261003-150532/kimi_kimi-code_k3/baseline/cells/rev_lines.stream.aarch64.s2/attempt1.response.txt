.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text

// rev_lines: print every line of stdin with its bytes reversed.
// x19 = input buffer base, x20 = total bytes read,
// x21 = scan index, x22 = current line start index, x23 = output buffer base.

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, xzr                // total bytes read so far

read_loop:
    mov x2, #65536
    sub x2, x2, x20             // space left in buffer
    cbz x2, read_done           // buffer full
    mov x0, xzr                 // fd 0 (stdin)
    add x1, x19, x20            // read position
    mov x8, #SYS_read
    svc #0
    cmp x0, xzr
    b.le read_done              // 0 = EOF, negative = error
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x21, xzr                // scan index i
    mov x22, xzr                // line start

scan_loop:
    cmp x21, x20
    b.hs scan_done              // reached end of input
    ldrb w0, [x19, x21]
    cmp w0, #10                 // newline byte
    b.eq 1f
    add x21, x21, #1
    b scan_loop
1:
    mov x0, x22                 // line = [x22, x21)
    mov x1, x21
    bl emit_line
    add x21, x21, #1            // skip the newline
    mov x22, x21
    b scan_loop

scan_done:
    cmp x22, x20
    b.hs finish                 // no trailing partial line
    mov x0, x22                 // final line without newline
    mov x1, x20
    bl emit_line

finish:
    mov x0, xzr                 // exit status 0
    mov x8, #SYS_exit
    svc #0

// emit_line(x0 = start index, x1 = end index):
// reverse inbuf[start..end) into outbuf, append '\n', write to stdout.
emit_line:
    sub x2, x1, x0              // line length
    sub x4, x1, #1              // index of last byte of line
    mov x3, xzr                 // j
1:
    cmp x3, x2
    b.hs 2f
    sub x5, x4, x3              // index of byte (end-1-j)
    ldrb w6, [x19, x5]
    strb w6, [x23, x3]
    add x3, x3, #1
    b 1b
2:
    mov w6, #10
    strb w6, [x23, x2]          // append newline
    add x2, x2, #1              // count = len + 1
    mov x0, #1                  // fd 1 (stdout)
    mov x1, x23
    mov x8, #SYS_write
    svc #0
    ret

.bss
.align 4
inbuf:
    .zero 65536
outbuf:
    .zero 65544
