// Brainfuck interpreter - static AArch64 Linux executable, no libc.
// read = 63, write = 64, exit = 93.

.text
.globl _start

_start:
    // ---- read all of stdin into inbuf ----
    adrp x11, inbuf
    add x11, x11, :lo12:inbuf      // x11 = buffer base
    mov x12, #65536
    add x13, x11, x12              // x13 = buffer end
    mov x10, x11                   // x10 = fill pointer
read_loop:
    cmp x10, x13
    b.hs read_done
    mov x0, #0                     // fd 0
    mov x1, x10
    sub x2, x13, x10
    mov x8, #63                    // read
    svc #0
    cmp x0, #0
    b.le read_done
    add x10, x10, x0
    b read_loop
read_done:

    // ---- split at first '!' (0x21) ----
    sub x2, x10, x11               // x2 = total bytes
    mov x1, #0                     // x1 = index
scan_loop:
    cmp x1, x2
    b.hs scan_done
    ldrb w3, [x11, x1]
    cmp w3, #33                    // '!'
    b.eq scan_done
    add x1, x1, #1
    b scan_loop
scan_done:
    mov x19, x11                   // x19 = program base
    mov x20, x1                    // x20 = program length
    add x21, x11, x1               // x21 = input cursor
    cmp x1, x2
    b.hs no_bang
    add x21, x21, #1               // skip the '!'
no_bang:
    mov x22, x10                   // x22 = input end

    // ---- build dispatch table (256 entries) ----
    adrp x27, disptab
    add x27, x27, :lo12:disptab
    adr x0, h_nop
    mov x1, #256
    mov x2, x27
init_disp:
    str x0, [x2], #8
    subs x1, x1, #1
    b.ne init_disp
    adr x0, h_plus
    str x0, [x27, #344]            // 43 '+'
    adr x0, h_comma
    str x0, [x27, #352]            // 44 ','
    adr x0, h_minus
    str x0, [x27, #360]            // 45 '-'
    adr x0, h_dot
    str x0, [x27, #368]            // 46 '.'
    adr x0, h_lt
    str x0, [x27, #480]            // 60 '<'
    adr x0, h_gt
    str x0, [x27, #496]            // 62 '>'
    adr x0, h_lbr
    str x0, [x27, #728]            // 91 '['
    adr x0, h_rbr
    str x0, [x27, #744]            // 93 ']'

    // ---- validate brackets, build jump table ----
    adrp x26, jumptab
    add x26, x26, :lo12:jumptab
    adrp x3, brstack
    add x3, x3, :lo12:brstack      // x3 = stack base
    mov x7, #16400
    add x6, x3, x7                 // x6 = stack limit
    mov x4, x3                     // x4 = stack pointer
    mov x5, #0                     // x5 = index i
val_loop:
    cmp x5, x20
    b.hs val_done
    ldrb w0, [x19, x5]
    cmp w0, #91                    // '['
    b.eq val_open
    cmp w0, #93                    // ']'
    b.eq val_close
val_next:
    add x5, x5, #1
    b val_loop
val_open:
    cmp x4, x6
    b.hs fail                      // nesting too deep
    str w5, [x4], #4               // push i
    b val_next
val_close:
    cmp x4, x3
    b.ls fail                      // ']' with no open '['
    ldr w7, [x4, #-4]!             // pop j
    str w5, [x26, x7, lsl #2]      // jump[j] = i
    str w7, [x26, x5, lsl #2]      // jump[i] = j
    b val_next
val_done:
    cmp x4, x3
    b.ne fail                      // unclosed '['

    // ---- execute ----
    adrp x23, tape
    add x23, x23, :lo12:tape       // x23 = data pointer
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf     // x24 = output cursor
    mov x25, #0                    // x25 = program counter
exec_loop:
    cmp x25, x20
    b.hs exec_done
    ldrb w0, [x19, x25]
    ldr x1, [x27, x0, lsl #3]
    br x1

h_nop:
    add x25, x25, #1
    b exec_loop
h_gt:
    add x23, x23, #1
    add x25, x25, #1
    b exec_loop
h_lt:
    sub x23, x23, #1
    add x25, x25, #1
    b exec_loop
h_plus:
    ldrb w0, [x23]
    add w0, w0, #1
    strb w0, [x23]
    add x25, x25, #1
    b exec_loop
h_minus:
    ldrb w0, [x23]
    sub w0, w0, #1
    strb w0, [x23]
    add x25, x25, #1
    b exec_loop
h_dot:
    ldrb w0, [x23]
    strb w0, [x24], #1
    add x25, x25, #1
    b exec_loop
h_comma:
    cmp x21, x22
    b.hs comma_eof
    ldrb w0, [x21], #1
    strb w0, [x23]
    add x25, x25, #1
    b exec_loop
comma_eof:
    strb wzr, [x23]
    add x25, x25, #1
    b exec_loop
h_lbr:
    ldrb w0, [x23]
    cbnz w0, lbr_no
    ldr w0, [x26, x25, lsl #2]     // matching ']'
    add x25, x0, #1
    b exec_loop
lbr_no:
    add x25, x25, #1
    b exec_loop
h_rbr:
    ldrb w0, [x23]
    cbz w0, rbr_no
    ldr w0, [x26, x25, lsl #2]     // matching '['
    add x25, x0, #1
    b exec_loop
rbr_no:
    add x25, x25, #1
    b exec_loop

exec_done:
    // ---- write all buffered output, exit 0 ----
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x24, x1
    mov x0, #1                     // fd 1
    mov x8, #64                    // write
    svc #0
    mov x0, #0
    mov x8, #93                    // exit
    svc #0

fail:
    mov x0, #1
    mov x8, #93                    // exit(1)
    svc #0

.bss
.balign 8
inbuf:   .zero 65536
.balign 8
jumptab: .zero 262144
.balign 8
disptab: .zero 2048
.balign 8
brstack: .zero 16400
.balign 8
tape:    .zero 30000
.balign 8
outbuf:  .zero 65536
