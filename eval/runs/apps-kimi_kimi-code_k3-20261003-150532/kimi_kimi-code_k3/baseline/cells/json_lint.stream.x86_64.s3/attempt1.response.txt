.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
str_true:
    .ascii "rue"
str_false:
    .ascii "alse"
str_null:
    .ascii "ull"

.bss
.align 8
inbuf:
    .zero 65536
numbuf:
    .zero 24

.text
_start:
    lea r13, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov edx, 65536
    sub rdx, r12
    jz read_done
    lea rsi, [r13 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    call skip_ws
    cmp rbx, r12
    je fail
    call parse_value
    call skip_ws
    cmp rbx, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov byte ptr [rip + numbuf + 23], 0x0A
    lea rdi, [rip + numbuf + 23]
    mov rax, rbx
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + numbuf + 24]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    cmp rbx, r12
    jae 1f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    je 2f
    cmp al, 0x0A
    je 2f
    cmp al, 0x0D
    je 2f
1:
    ret
2:
    inc rbx
    jmp skip_ws

parse_value:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je lit_true
    cmp al, 0x66
    je lit_false
    cmp al, 0x6E
    je lit_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe parse_number
    jmp fail

lit_true:
    lea rsi, [rip + str_true]
    mov ecx, 3
    jmp lit_common
lit_false:
    lea rsi, [rip + str_false]
    mov ecx, 4
    jmp lit_common
lit_null:
    lea rsi, [rip + str_null]
    mov ecx, 3
lit_common:
    inc rbx
1:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, byte ptr [rsi]
    jne fail
    inc rbx
    inc rsi
    dec ecx
    jnz 1b
    ret

parse_string:
    inc rbx
1:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x22
    je 2f
    cmp al, 0x5C
    je 3f
    cmp al, 0x20
    jb fail
    inc rbx
    jmp 1b
2:
    inc rbx
    ret
3:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x22
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, 0x2F
    je 4f
    cmp al, 0x62
    je 4f
    cmp al, 0x66
    je 4f
    cmp al, 0x6E
    je 4f
    cmp al, 0x72
    je 4f
    cmp al, 0x74
    je 4f
    cmp al, 0x75
    je 5f
    jmp fail
4:
    inc rbx
    jmp 1b
5:
    inc rbx
    mov ecx, 4
6:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe 7f
    or al, 0x20
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
7:
    inc rbx
    dec ecx
    jnz 6b
    jmp 1b

parse_number:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x2D
    jne 1f
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
1:
    cmp al, 0x30
    je 2f
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
3:
    inc rbx
    cmp rbx, r12
    jae 4f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x30
    jb 4f
    cmp al, 0x39
    jbe 3b
    jmp 4f
2:
    inc rbx
4:
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x2E
    jne 5f
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
6:
    inc rbx
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x30
    jb 5f
    cmp al, 0x39
    jbe 6b
5:
    cmp al, 0x65
    je 7f
    cmp al, 0x45
    je 7f
    ret
7:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x2B
    je 8f
    cmp al, 0x2D
    je 8f
    jmp 10f
8:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
10:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
11:
    inc rbx
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x30
    jb 9f
    cmp al, 0x39
    jbe 11b
9:
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x5D
    jne 1f
    inc rbx
    ret
1:
    call parse_value
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x2C
    je 2f
    cmp al, 0x5D
    jne fail
    inc rbx
    ret
2:
    inc rbx
    call skip_ws
    jmp 1b

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x7D
    jne 1f
    inc rbx
    ret
1:
    cmp al, 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x3A
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x2C
    je 2f
    cmp al, 0x7D
    jne fail
    inc rbx
    ret
2:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    jmp 1b
