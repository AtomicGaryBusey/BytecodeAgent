.globl _start

.text
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + raw]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + raw]
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [rsi + rcx], 33
    je bang_found
    inc rcx
    jmp find_bang

bang_found:
    mov r12, rcx
    lea r8, [rsi + rcx + 1]
    mov r9, rbx
    sub r9, rcx
    dec r9
    jmp have_parts

no_bang:
    mov r12, rbx
    lea r8, [rsi + rbx]
    xor r9d, r9d

have_parts:
    lea rdi, [rip + code]
    lea r10, [rip + jump]
    lea r11, [rip + bracket_stack]
    xor ecx, ecx
    xor edx, edx
    xor ebp, ebp

build_loop:
    cmp rcx, r12
    jae build_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 62
    jne not_gt
    mov bl, 1
    jmp store_cmd
not_gt:
    cmp al, 60
    jne not_lt
    mov bl, 2
    jmp store_cmd
not_lt:
    cmp al, 43
    jne not_plus
    mov bl, 3
    jmp store_cmd
not_plus:
    cmp al, 45
    jne not_minus
    mov bl, 4
    jmp store_cmd
not_minus:
    cmp al, 46
    jne not_dot
    mov bl, 5
    jmp store_cmd
not_dot:
    cmp al, 44
    jne not_comma
    mov bl, 6
    jmp store_cmd
not_comma:
    cmp al, 91
    jne not_open
    mov bl, 7
    jmp store_cmd
not_open:
    cmp al, 93
    jne build_next
    mov bl, 8

store_cmd:
    mov byte ptr [rdi + rdx], bl
    cmp bl, 7
    je store_open
    cmp bl, 8
    je store_close
    inc rdx
    jmp build_next

store_open:
    cmp rbp, 1000
    jae invalid
    mov dword ptr [r11 + rbp*4], edx
    inc rbp
    inc rdx
    jmp build_next

store_close:
    test rbp, rbp
    jz invalid
    dec rbp
    mov eax, dword ptr [r11 + rbp*4]
    mov dword ptr [r10 + rdx*4], eax
    mov dword ptr [r10 + rax*4], edx
    inc rdx

build_next:
    inc rcx
    jmp build_loop

build_done:
    test rbp, rbp
    jnz invalid
    mov r12, rdx

    lea rbx, [rip + code]
    lea rsi, [rip + outbuf]
    lea r15, [rip + tape]
    lea r11, [rip + jump]
    xor r13d, r13d
    xor r14d, r14d
    xor ebp, ebp
    xor r10d, r10d

exec_loop:
    cmp r13, r12
    jae run_done
    movzx eax, byte ptr [rbx + r13]
    cmp al, 1
    je op_gt
    cmp al, 2
    je op_lt
    cmp al, 3
    je op_plus
    cmp al, 4
    je op_minus
    cmp al, 5
    je op_dot
    cmp al, 6
    je op_comma
    cmp al, 7
    je op_open
    cmp al, 8
    je op_close
    inc r13
    jmp exec_loop

op_gt:
    inc r14
    inc r13
    jmp exec_loop

op_lt:
    dec r14
    inc r13
    jmp exec_loop

op_plus:
    inc byte ptr [r15 + r14]
    inc r13
    jmp exec_loop

op_minus:
    dec byte ptr [r15 + r14]
    inc r13
    jmp exec_loop

op_dot:
    mov al, byte ptr [r15 + r14]
    mov byte ptr [rsi + rbp], al
    inc rbp
    inc r13
    jmp exec_loop

op_comma:
    cmp r10, r9
    jae comma_eof
    mov al, byte ptr [r8 + r10]
    inc r10
    jmp comma_store
comma_eof:
    xor eax, eax
comma_store:
    mov byte ptr [r15 + r14], al
    inc r13
    jmp exec_loop

op_open:
    cmp byte ptr [r15 + r14], 0
    jne open_next
    mov eax, dword ptr [r11 + r13*4]
    lea r13, [rax + 1]
    jmp exec_loop
open_next:
    inc r13
    jmp exec_loop

op_close:
    cmp byte ptr [r15 + r14], 0
    je close_next
    mov eax, dword ptr [r11 + r13*4]
    lea r13, [rax + 1]
    jmp exec_loop
close_next:
    inc r13
    jmp exec_loop

run_done:
    test rbp, rbp
    jz exit_ok
    mov eax, 1
    mov edi, 1
    mov rdx, rbp
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall

.bss
raw:
    .zero 65536
code:
    .zero 65536
jump:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
outbuf:
    .zero 65536
