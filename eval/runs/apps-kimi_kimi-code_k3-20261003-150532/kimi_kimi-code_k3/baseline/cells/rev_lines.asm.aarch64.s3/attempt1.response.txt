.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 65537

.section .text

_start:
	// read all of stdin into inbuf
	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	mov x20, x19			// current fill pointer
	mov x21, INBUF_SIZE		// space remaining
1:
	cbz x21, 2f
	mov x0, 0
	mov x1, x20
	mov x2, x21
	mov x8, SYS_read
	svc #0
	cbz x0, 2f			// end of input
	add x20, x20, x0
	sub x21, x21, x0
	b 1b
2:
	// x20 = end of input data; build reversed output in outbuf
	adrp x22, outbuf
	add x22, x22, :lo12:outbuf	// output fill pointer
	mov x23, x19			// start of current line
3:
	cmp x23, x20
	b.hs 4f				// no more input
	// find end of line: x24 scans for newline or end of data
	mov x24, x23
5:
	cmp x24, x20
	b.hs 6f
	ldrb w0, [x24]
	cmp w0, 10
	b.eq 6f
	add x24, x24, 1
	b 5b
6:
	// copy bytes [x23, x24) in reverse order
	mov x0, x24
7:
	cmp x0, x23
	b.ls 8f
	sub x0, x0, 1
	ldrb w1, [x0]
	strb w1, [x22], #1
	b 7b
8:
	mov w1, 10
	strb w1, [x22], #1
	cmp x24, x20
	b.hs 4f				// line ended at EOF
	add x23, x24, 1			// skip the newline
	b 3b
4:
	// write the whole output buffer to stdout
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	sub x2, x22, x1
	cbz x2, 9f
	mov x0, 1
	mov x8, SYS_write
	svc #0
9:
	mov x0, 0
	mov x8, SYS_exit
	svc #0

.section .bss
.balign 16
inbuf:
	.zero INBUF_SIZE
outbuf:
	.zero OUTBUF_SIZE
