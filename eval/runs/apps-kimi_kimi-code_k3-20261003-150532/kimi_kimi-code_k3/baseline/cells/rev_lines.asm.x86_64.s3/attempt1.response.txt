.globl _start

.text
_start:
    xor ebx, ebx

read_loop:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r8, [rip + inbuf]
    lea r9, [rip + inbuf]
    lea r10, [rip + inbuf]
    add r10, rbx
    lea rdi, [rip + outbuf]

scan_loop:
    cmp r9, r10
    je final_line
    mov al, byte ptr [r9]
    cmp al, 10
    je have_line
    inc r9
    jmp scan_loop

have_line:
    mov rcx, r9
reverse_line:
    cmp rcx, r8
    je line_done
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_line
line_done:
    mov byte ptr [rdi], 10
    inc rdi
    inc r9
    mov r8, r9
    jmp scan_loop

final_line:
    cmp r8, r10
    je build_done
    mov rcx, r10
reverse_final:
    cmp rcx, r8
    je final_done
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_final
final_done:
    mov byte ptr [rdi], 10
    inc rdi

build_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
