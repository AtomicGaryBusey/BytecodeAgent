.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 70000

.text
_start:
    xor ebx, ebx
read_more:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    jz encode
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz encode
    add rbx, rax
    jmp read_more

encode:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + alphabet]
    xor r9d, r9d
    xor ecx, ecx

group_loop:
    lea rax, [rcx + 3]
    cmp rax, rbx
    ja tail
    movzx r10d, byte ptr [rsi + rcx]
    movzx r11d, byte ptr [rsi + rcx + 1]
    movzx r12d, byte ptr [rsi + rcx + 2]

    mov eax, r10d
    shr eax, 2
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi], dl

    mov eax, r10d
    and eax, 3
    shl eax, 4
    mov edx, r11d
    shr edx, 4
    or eax, edx
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], dl

    mov eax, r11d
    and eax, 15
    shl eax, 2
    mov edx, r12d
    shr edx, 6
    or eax, edx
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], dl

    mov eax, r12d
    and eax, 63
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], dl

    add rdi, 4
    add rcx, 3
    add r9d, 4
    cmp r9d, 76
    jne group_loop
    mov byte ptr [rdi], 10
    add rdi, 1
    xor r9d, r9d
    jmp group_loop

tail:
    mov rax, rbx
    sub rax, rcx
    test rax, rax
    jz finish
    cmp rax, 1
    je tail_one

    movzx r10d, byte ptr [rsi + rcx]
    movzx r11d, byte ptr [rsi + rcx + 1]

    mov eax, r10d
    shr eax, 2
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi], dl

    mov eax, r10d
    and eax, 3
    shl eax, 4
    mov edx, r11d
    shr edx, 4
    or eax, edx
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], dl

    mov eax, r11d
    and eax, 15
    shl eax, 2
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r9d, 4
    jmp finish

tail_one:
    movzx r10d, byte ptr [rsi + rcx]

    mov eax, r10d
    shr eax, 2
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi], dl

    mov eax, r10d
    and eax, 3
    shl eax, 4
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], dl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r9d, 4

finish:
    test r9d, r9d
    jz write_all
    mov byte ptr [rdi], 10
    add rdi, 1

write_all:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_ok
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
