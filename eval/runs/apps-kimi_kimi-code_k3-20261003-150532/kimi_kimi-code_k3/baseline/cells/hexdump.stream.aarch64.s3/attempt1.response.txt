.globl _start

.section .rodata
hexdigits:
	.ascii "0123456789abcdef"

.text
_start:
	adrp x20, inbuf
	add x20, x20, :lo12:inbuf
	adrp x28, outbuf
	add x28, x28, :lo12:outbuf
	mov x19, #0
	mov x9, #65536
read_loop:
	cmp x19, x9
	b.hs read_done
	mov x0, #0
	add x1, x20, x19
	mov x2, #65536
	sub x2, x2, x19
	mov x8, #63
	svc #0
	cmp x0, #0
	b.le read_done
	add x19, x19, x0
	b read_loop

read_done:
	mov x21, #0
row_loop:
	cmp x21, x19
	b.hs final_offset
	sub x22, x19, x21
	cmp x22, #16
	b.ls have_n
	mov x22, #16
have_n:
	add x25, x20, x21
	mov x3, x28
	adrp x2, hexdigits
	add x2, x2, :lo12:hexdigits

	mov x0, x21
	mov x1, #8
off_hex_loop:
	lsr x4, x0, #28
	ldrb w4, [x2, x4]
	strb w4, [x3], #1
	lsl x0, x0, #4
	subs x1, x1, #1
	b.ne off_hex_loop

	mov w1, #32
	strb w1, [x3], #1
	strb w1, [x3], #1

	mov x24, #0
hex_slots:
	cmp x24, x22
	b.hs blank_slot
	ldrb w0, [x25, x24]
	lsr x1, x0, #4
	ldrb w1, [x2, x1]
	strb w1, [x3], #1
	and x1, x0, #15
	ldrb w1, [x2, x1]
	strb w1, [x3], #1
	mov w1, #32
	strb w1, [x3], #1
	b slot_done
blank_slot:
	mov w1, #32
	strb w1, [x3], #1
	strb w1, [x3], #1
	strb w1, [x3], #1
slot_done:
	cmp x24, #7
	b.ne no_extra
	mov w1, #32
	strb w1, [x3], #1
no_extra:
	add x24, x24, #1
	cmp x24, #16
	b.lo hex_slots

	mov w1, #32
	strb w1, [x3], #1
	mov w1, #124
	strb w1, [x3], #1

	mov x24, #0
ascii_loop:
	cmp x24, x22
	b.hs ascii_done
	ldrb w0, [x25, x24]
	cmp x0, #32
	b.lo dot
	cmp x0, #126
	b.hi dot
	b store_ascii
dot:
	mov w0, #46
store_ascii:
	strb w0, [x3], #1
	add x24, x24, #1
	b ascii_loop
ascii_done:
	mov w1, #124
	strb w1, [x3], #1
	mov w1, #10
	strb w1, [x3], #1

	sub x27, x3, x28
	mov x26, x28
write_row:
	mov x0, #1
	mov x1, x26
	mov x2, x27
	mov x8, #64
	svc #0
	cmp x0, #0
	b.le write_fail
	add x26, x26, x0
	subs x27, x27, x0
	b.gt write_row

	add x21, x21, #16
	b row_loop

final_offset:
	mov x3, x28
	adrp x2, hexdigits
	add x2, x2, :lo12:hexdigits
	mov x0, x19
	mov x1, #8
final_hex_loop:
	lsr x4, x0, #28
	ldrb w4, [x2, x4]
	strb w4, [x3], #1
	lsl x0, x0, #4
	subs x1, x1, #1
	b.ne final_hex_loop
	mov w1, #10
	strb w1, [x3], #1

	sub x27, x3, x28
	mov x26, x28
write_final:
	mov x0, #1
	mov x1, x26
	mov x2, x27
	mov x8, #64
	svc #0
	cmp x0, #0
	b.le write_fail
	add x26, x26, x0
	subs x27, x27, x0
	b.gt write_final

	mov x0, #0
	mov x8, #93
	svc #0
	b .

write_fail:
	mov x0, #1
	mov x8, #93
	svc #0
	b .

.bss
	.align 4
inbuf:
	.zero 65536
outbuf:
	.zero 128
