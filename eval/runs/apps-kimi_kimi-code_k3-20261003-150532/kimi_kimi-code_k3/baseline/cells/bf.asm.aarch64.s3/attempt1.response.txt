.text
.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFCAP, 65536

_start:
    // Read all of standard input into inbuf
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf    // x19 = buffer base
    mov  x20, #0                  // x20 = total bytes read
read_loop:
    mov  x2, #BUFCAP
    sub  x2, x2, x20
    cbz  x2, read_done
    add  x1, x19, x20
    mov  x0, #0
    mov  x8, #SYS_READ
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop
read_done:

    // Find the first '!' (0x21); program = [0, x21), input = rest
    mov  x21, #0
find_bang:
    cmp  x21, x20
    b.hs find_bang_done
    ldrb w0, [x19, x21]
    cmp  w0, #0x21
    b.eq find_bang_done
    add  x21, x21, #1
    b    find_bang
find_bang_done:
    // x21 = program length

    // Check bracket balance and build jump table
    adrp x22, jtable
    add  x22, x22, :lo12:jtable   // x22 = jump table base
    adrp x23, bstack
    add  x23, x23, :lo12:bstack   // x23 = bracket stack base
    mov  x24, #0                  // depth
    mov  x25, #0                  // i
scan_loop:
    cmp  x25, x21
    b.hs scan_done
    ldrb w0, [x19, x25]
    cmp  w0, #'['
    b.eq scan_open
    cmp  w0, #']'
    b.eq scan_close
    b    scan_next
scan_open:
    str  w25, [x23, x24, lsl #2]
    add  x24, x24, #1
    b    scan_next
scan_close:
    cbz  x24, fail
    sub  x24, x24, #1
    ldr  w0, [x23, x24, lsl #2]   // w0 = matching open index
    add  w1, w25, #1
    str  w1, [x22, w0, uxtw #2]   // jtable[open] = close + 1
    add  w0, w0, #1
    str  w0, [x22, x25, lsl #2]   // jtable[close] = open + 1
scan_next:
    add  x25, x25, #1
    b    scan_loop
scan_done:
    cbnz x24, fail

    // Set up interpreter state
    adrp x25, tape
    add  x25, x25, :lo12:tape     // x25 = tape base
    adrp x23, outbuf
    add  x23, x23, :lo12:outbuf   // x23 = output buffer base
    mov  x24, #0                  // x24 = output count
    mov  x26, #0                  // x26 = pc
    mov  x27, #0                  // x27 = dp
    sub  x29, x20, x21            // bytes from '!' to end
    cbz  x29, no_input
    sub  x29, x29, #1             // x29 = input bytes remaining
    add  x28, x19, x21
    add  x28, x28, #1             // x28 = input read pointer
    b    run
no_input:
    mov  x29, #0
    mov  x28, x19

run:
    cmp  x26, x21
    b.hs finish
    ldrb w0, [x19, x26]
    cmp  w0, #'>'
    b.eq op_rgt
    cmp  w0, #'<'
    b.eq op_lft
    cmp  w0, #'+'
    b.eq op_inc
    cmp  w0, #'-'
    b.eq op_dec
    cmp  w0, #'.'
    b.eq op_out
    cmp  w0, #','
    b.eq op_inp
    cmp  w0, #'['
    b.eq op_lbr
    cmp  w0, #']'
    b.eq op_rbr
next:
    add  x26, x26, #1
    b    run

op_rgt:
    add  x27, x27, #1
    b    next
op_lft:
    sub  x27, x27, #1
    b    next
op_inc:
    ldrb w0, [x25, x27]
    add  w0, w0, #1
    strb w0, [x25, x27]
    b    next
op_dec:
    ldrb w0, [x25, x27]
    sub  w0, w0, #1
    strb w0, [x25, x27]
    b    next
op_out:
    ldrb w0, [x25, x27]
    strb w0, [x23, x24]
    add  x24, x24, #1
    b    next
op_inp:
    cbz  x29, inp_zero
    ldrb w0, [x28], #1
    sub  x29, x29, #1
    strb w0, [x25, x27]
    b    next
inp_zero:
    strb wzr, [x25, x27]
    b    next
op_lbr:
    ldrb w0, [x25, x27]
    cbnz w0, next
    ldr  w0, [x22, x26, lsl #2]
    mov  x26, x0
    b    run
op_rbr:
    ldrb w0, [x25, x27]
    cbz  w0, next
    ldr  w0, [x22, x26, lsl #2]
    mov  x26, x0
    b    run

finish:
    // Write output buffer to stdout
    mov  x20, #0                  // bytes written
write_loop:
    cmp  x20, x24
    b.hs write_done
    mov  x0, #1
    add  x1, x23, x20
    sub  x2, x24, x20
    mov  x8, #SYS_WRITE
    svc  #0
    add  x20, x20, x0
    b    write_loop
write_done:
    mov  x0, #0
    mov  x8, #SYS_EXIT
    svc  #0

fail:
    mov  x0, #1
    mov  x8, #SYS_EXIT
    svc  #0

.bss
.align 4
inbuf:  .zero 65536
tape:   .zero 30000
outbuf: .zero 65536
jtable: .zero 262144
bstack: .zero 131072
