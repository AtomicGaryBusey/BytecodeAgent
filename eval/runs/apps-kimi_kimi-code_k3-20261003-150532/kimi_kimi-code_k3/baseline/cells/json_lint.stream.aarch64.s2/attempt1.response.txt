.globl _start

.section .rodata
msg_ok:    .ascii "ok\n"
msg_err:   .ascii "error at "
lit_true:  .ascii "rue"
lit_false: .ascii "alse"
lit_null:  .ascii "ull"

.bss
.align 4
inbuf:  .zero 65536
outbuf: .zero 64

.text

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, x20
    mov x20, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne fail
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

fail:
    adrp x11, msg_err
    add x11, x11, :lo12:msg_err
    adrp x12, outbuf
    add x12, x12, :lo12:outbuf
    mov x13, #0
1:
    cmp x13, #9
    b.hs 2f
    ldrb w14, [x11, x13]
    strb w14, [x12, x13]
    add x13, x13, #1
    b 1b
2:
    mov x0, x20
    add x15, x12, #32
    mov x16, x15
    mov x1, #10
3:
    udiv x2, x0, x1
    msub x3, x2, x1, x0
    add w3, w3, #48
    sub x16, x16, #1
    strb w3, [x16]
    mov x0, x2
    cbnz x0, 3b
    add x13, x12, #9
4:
    cmp x16, x15
    b.hs 5f
    ldrb w14, [x16], #1
    strb w14, [x13], #1
    b 4b
5:
    mov w14, #10
    strb w14, [x13], #1
    mov x0, #1
    mov x1, x12
    sub x2, x13, x12
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
    cmp x20, x21
    b.hs 1f
    ldrb w9, [x19, x20]
    cmp w9, #0x20
    b.eq 2f
    cmp w9, #0x09
    b.eq 2f
    cmp w9, #0x0A
    b.eq 2f
    cmp w9, #0x0D
    b.eq 2f
1:
    ret
2:
    add x20, x20, #1
    b skip_ws

parse_value:
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x7B
    b.eq parse_object
    cmp w9, #0x5B
    b.eq parse_array
    cmp w9, #0x22
    b.eq parse_string
    cmp w9, #0x74
    b.eq parse_true
    cmp w9, #0x66
    b.eq parse_false
    cmp w9, #0x6E
    b.eq parse_null
    cmp w9, #0x2D
    b.eq parse_number
    sub w10, w9, #0x30
    cmp w10, #9
    b.ls parse_number
    b fail

parse_true:
    add x20, x20, #1
    adrp x0, lit_true
    add x0, x0, :lo12:lit_true
    mov x1, #3
    b expect_bytes

parse_false:
    add x20, x20, #1
    adrp x0, lit_false
    add x0, x0, :lo12:lit_false
    mov x1, #4
    b expect_bytes

parse_null:
    add x20, x20, #1
    adrp x0, lit_null
    add x0, x0, :lo12:lit_null
    mov x1, #3
    b expect_bytes

expect_bytes:
    cbz x1, 2f
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    ldrb w10, [x0], #1
    cmp w9, w10
    b.ne fail
    add x20, x20, #1
    sub x1, x1, #1
    b expect_bytes
2:
    ret

parse_string:
    add x20, x20, #1
str_loop:
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x22
    b.eq str_end
    cmp w9, #0x5C
    b.eq str_escape
    cmp w9, #0x20
    b.lo fail
    add x20, x20, #1
    b str_loop
str_end:
    add x20, x20, #1
    ret
str_escape:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x75
    b.eq str_unicode
    cmp w9, #0x22
    b.eq str_esc_ok
    cmp w9, #0x5C
    b.eq str_esc_ok
    cmp w9, #0x2F
    b.eq str_esc_ok
    cmp w9, #0x62
    b.eq str_esc_ok
    cmp w9, #0x66
    b.eq str_esc_ok
    cmp w9, #0x6E
    b.eq str_esc_ok
    cmp w9, #0x72
    b.eq str_esc_ok
    cmp w9, #0x74
    b.eq str_esc_ok
    b fail
str_esc_ok:
    add x20, x20, #1
    b str_loop
str_unicode:
    add x20, x20, #1
    mov x11, #4

    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.ls str_hex_ok
    sub w10, w9, #0x41
    cmp w10, #5
    b.ls str_hex_ok
    sub w10, w9, #0x61
    cmp w10, #5
    b.ls str_hex_ok
    b fail

    add x20, x20, #1
    subs x11, x11, #1
    b.ne str_hex
    b str_loop

parse_number:
    ldrb w9, [x19, x20]
    cmp w9, #0x2D
    b.ne num_int
    add x20, x20, #1
num_int:
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x30
    b.eq num_zero
    sub w10, w9, #0x31
    cmp w10, #8
    b.hi fail
    add x20, x20, #1

    cmp x20, x21
    b.hs num_frac
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi num_frac
    add x20, x20, #1
    b num_digits

    add x20, x20, #1

    cmp x20, x21
    b.hs num_done
    ldrb w9, [x19, x20]
    cmp w9, #0x2E
    b.ne num_exp
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi fail

    add x20, x20, #1
    cmp x20, x21
    b.hs num_done
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.ls num_frac_digits

    cmp x20, x21
    b.hs num_done
    ldrb w9, [x19, x20]
    cmp w9, #0x65
    b.eq num_exp_go
    cmp w9, #0x45
    b.eq num_exp_go
    b num_done

    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2B
    b.eq num_exp_sign
    cmp w9, #0x2D
    b.eq num_exp_sign
    b num_exp_first

    add x20, x20, #1

    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi fail
num_exp_digits:
    add x20, x20, #1
    cmp x20, x21
    b.hs num_done
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.ls num_exp_digits

    ret

    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x7D

    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x22
    b.ne fail
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x3A

    add x20, x20, #1

    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2C
    b.eq obj_comma
    cmp w9, #0x7D

    ldp x29, x30, [sp], #16

    cmp x20, x21
    b.hs fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2C
    b.eq arr_comma

    ldp x29, x30, [sp], #16
    ret
