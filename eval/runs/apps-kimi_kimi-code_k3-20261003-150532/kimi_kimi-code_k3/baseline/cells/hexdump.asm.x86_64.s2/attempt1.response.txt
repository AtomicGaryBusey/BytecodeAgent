.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 323600

.text
_start:
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + inbuf]
    lea r14, [rip + outbuf]
    xor r15d, r15d
row_loop:
    cmp r15, r12
    jae final_line
    mov eax, r15d
    call emit_hex8
    mov byte ptr [r14], 32
    inc r14
    mov byte ptr [r14], 32
    inc r14
    mov rbx, r12
    sub rbx, r15
    mov ecx, 16
    cmp rbx, 16
    jae 1f
    mov ecx, ebx
1:
    mov rbx, rcx
    lea rsi, [r13 + r15]
    xor r8d, r8d
slot_loop:
    cmp r8d, 16
    jae after_slots
    cmp r8d, ebx
    jae blank_slot
    movzx eax, byte ptr [rsi + r8]
    call emit_hex2
    mov byte ptr [r14], 32
    inc r14
    jmp slot_extra
blank_slot:
    mov byte ptr [r14], 32
    inc r14
    mov byte ptr [r14], 32
    inc r14
    mov byte ptr [r14], 32
    inc r14
slot_extra:
    cmp r8d, 7
    jne 2f
    mov byte ptr [r14], 32
    inc r14
2:
    inc r8d
    jmp slot_loop
after_slots:
    mov byte ptr [r14], 32
    inc r14
    mov byte ptr [r14], 124
    inc r14
    xor r8d, r8d
ascii_loop:
    cmp r8d, ebx
    jae ascii_done
    movzx eax, byte ptr [rsi + r8]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov al, 46
ascii_store:
    mov byte ptr [r14], al
    inc r14
    inc r8d
    jmp ascii_loop
ascii_done:
    mov byte ptr [r14], 124
    inc r14
    mov byte ptr [r14], 10
    inc r14
    add r15, rbx
    jmp row_loop
final_line:
    mov eax, r12d
    call emit_hex8
    mov byte ptr [r14], 10
    inc r14
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit0
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit0
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    lea r10, [rip + hexdigits]
    mov r8d, eax
    mov r9d, 28
1:
    mov edx, r8d
    mov ecx, r9d
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [r14], dl
    inc r14
    sub r9d, 4
    jns 1b
    ret

emit_hex2:
    lea r10, [rip + hexdigits]
    mov edx, eax
    shr edx, 4
    and edx, 15
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [r14], dl
    inc r14
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [r14], dl
    inc r14
    ret
