// rpn: evaluate one Reverse Polish Notation expression per input line
// AArch64 (ARMv8-A) static executable, no libc

.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    mov x21, #65536
read_loop:
    cbz x21, read_done
    mov x0, #0
    mov x1, x19
    mov x2, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    sub x21, x21, x0
    b read_loop

read_done:
    mov x22, x19
    mov x19, x20
    mov x20, x22
    adrp x24, vstack
    add x24, x24, :lo12:vstack

next_line:
    cmp x19, x20
    b.hs done_all
    mov x22, x19
find_nl:
    cmp x22, x20
    b.hs have_line
    ldrb w0, [x22]
    cmp w0, #10
    b.eq have_line
    add x22, x22, #1
    b find_nl
have_line:
    mov x23, x22
    mov x22, x19
    add x19, x23, #1
    mov x21, #0

tok_loop:
    cmp x22, x23
    b.hs end_tokens
    ldrb w0, [x22]
    cmp w0, #32
    b.eq skip_one
    cmp w0, #9
    b.eq skip_one
    b tok_start
skip_one:
    add x22, x22, #1
    b tok_loop

tok_start:
    mov x26, x22
scan_tok:
    cmp x22, x23
    b.hs tok_end
    ldrb w0, [x22]
    cmp w0, #32
    b.eq tok_end
    cmp w0, #9
    b.eq tok_end
    add x22, x22, #1
    b scan_tok

tok_end:
    sub x27, x22, x26
    cmp x27, #1
    b.ne try_number
    ldrb w0, [x26]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cmp w0, #47
    b.eq op_div
    cmp w0, #37
    b.eq op_mod

try_number:
    mov x1, x26
    mov x2, x27
    mov x3, #0
    ldrb w4, [x1]
    cmp w4, #45
    b.ne 2f
    mov x3, #1
    add x1, x1, #1
    sub x2, x2, #1
2:
    cbz x2, line_error
    cmp x2, #18
    b.hi line_error
    mov x5, #0
    mov x6, #10
digit_loop:
    ldrb w4, [x1], #1
    sub w4, w4, #48
    cmp w4, #9
    b.hi line_error
    mul x5, x5, x6
    add x5, x5, x4
    subs x2, x2, #1
    b.ne digit_loop
    cbz x3, 3f
    neg x5, x5
3:
    str x5, [x24, x21, lsl #3]
    add x21, x21, #1
    b tok_loop

op_add:
    cmp x21, #2
    b.lo line_error
    sub x2, x21, #1
    ldr x0, [x24, x2, lsl #3]
    sub x2, x21, #2
    ldr x1, [x24, x2, lsl #3]
    add x1, x1, x0
    str x1, [x24, x2, lsl #3]
    sub x21, x21, #1
    b tok_loop

op_sub:
    cmp x21, #2
    b.lo line_error
    sub x2, x21, #1
    ldr x0, [x24, x2, lsl #3]
    sub x2, x21, #2
    ldr x1, [x24, x2, lsl #3]
    sub x1, x1, x0
    str x1, [x24, x2, lsl #3]
    sub x21, x21, #1
    b tok_loop

op_mul:
    cmp x21, #2
    b.lo line_error
    sub x2, x21, #1
    ldr x0, [x24, x2, lsl #3]
    sub x2, x21, #2
    ldr x1, [x24, x2, lsl #3]
    mul x1, x1, x0
    str x1, [x24, x2, lsl #3]
    sub x21, x21, #1
    b tok_loop

op_div:
    cmp x21, #2
    b.lo line_error
    sub x2, x21, #1
    ldr x0, [x24, x2, lsl #3]
    sub x2, x21, #2
    ldr x1, [x24, x2, lsl #3]
    cbz x0, line_error
    mov x3, #1
    lsl x3, x3, #63
    cmp x1, x3
    b.ne 4f
    cmn x0, #1
    b.eq line_error
4:
    sdiv x1, x1, x0
    str x1, [x24, x2, lsl #3]
    sub x21, x21, #1
    b tok_loop

op_mod:
    cmp x21, #2
    b.lo line_error
    sub x2, x21, #1
    ldr x0, [x24, x2, lsl #3]
    sub x2, x21, #2
    ldr x1, [x24, x2, lsl #3]
    cbz x0, line_error
    mov x3, #1
    lsl x3, x3, #63
    cmp x1, x3
    b.ne 5f
    cmn x0, #1
    b.eq line_error
5:
    sdiv x4, x1, x0
    msub x1, x4, x0, x1
    str x1, [x24, x2, lsl #3]
    sub x21, x21, #1
    b tok_loop

end_tokens:
    cmp x21, #1
    b.ne line_error
    ldr x2, [x24]
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #32
    mov w0, #10
    sub x1, x1, #1
    strb w0, [x1]
    mov x3, #0
    tbz x2, #63, 6f
    mov x3, #1
    neg x2, x2
6:
    mov x4, #10
gen_loop:
    udiv x5, x2, x4
    msub x6, x5, x4, x2
    add w6, w6, #48
    sub x1, x1, #1
    strb w6, [x1]
    mov x2, x5
    cbnz x2, gen_loop
    cbz x3, 7f
    mov w6, #45
    sub x1, x1, #1
    strb w6, [x1]
7:
    adrp x2, numbuf
    add x2, x2, :lo12:numbuf
    add x2, x2, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    b next_line

line_error:
    adrp x1, errstr
    add x1, x1, :lo12:errstr
    mov x0, #1
    mov x2, #6
    mov x8, #64
    svc #0
    b next_line

done_all:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
errstr:
    .ascii "error\n"

.bss
    .align 3
inbuf:
    .zero 65600
    .align 3
vstack:
    .zero 8192
numbuf:
    .zero 32
