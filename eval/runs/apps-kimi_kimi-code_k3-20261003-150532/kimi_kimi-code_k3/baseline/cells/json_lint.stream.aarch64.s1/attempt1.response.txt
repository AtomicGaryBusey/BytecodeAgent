// json_lint: validate exactly one JSON document (RFC 8259) from stdin.
// AArch64 (ARMv8.0-A), static, no libc. syscalls: read=63, write=64, exit=93.

.text
.globl _start

// register conventions during the parse:
//   x19 = current position (pointer into buf)
//   x20 = end of input (buf + length)
//   x21 = buf base
// parse_error reports offset x19 - x21 and never returns.

_start:
    adrp x21, buf
    add  x21, x21, :lo12:buf
    mov  x22, 0                   // total bytes read
read_loop:
    mov  x0, 65536
    cmp  x22, x0
    b.hs read_done
    sub  x2, x0, x22              // count = 65536 - total
    add  x1, x21, x22             // buf + total
    mov  x0, 0                    // fd = stdin
    mov  x8, 63                   // read
    svc  #0
    cbz  x0, read_done            // 0 = end of input
    add  x22, x22, x0
    b    read_loop
read_done:
    mov  x19, x21                 // pos = buf
    add  x20, x21, x22            // end = buf + len

    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x19, x20
    b.ne parse_error              // trailing garbage

    // valid: print "ok\n", exit 0
    mov  x0, 1
    adrp x1, msg_ok
    add  x1, x1, :lo12:msg_ok
    mov  x2, 3
    mov  x8, 64
    svc  #0
    mov  x0, 0
    mov  x8, 93
    svc  #0

// ----------------------------------------------------------------
skip_ws:
1:
    cmp  x19, x20
    b.hs 2f
    ldrb w0, [x19]
    cmp  w0, 0x20                 // space
    b.eq 3f
    cmp  w0, 0x09                 // tab
    b.eq 3f
    cmp  w0, 0x0A                 // newline
    b.eq 3f
    cmp  w0, 0x0D                 // carriage return
    b.eq 3f
2:
    ret
3:
    add  x19, x19, 1
    b    1b

// ----------------------------------------------------------------
parse_value:
    stp  x29, x30, [sp, #-16]!
    cmp  x19, x20
    b.hs pv_error                 // EOF where a value is required
    ldrb w0, [x19]
    cmp  w0, 0x7B                 // '{'
    b.eq pv_obj
    cmp  w0, 0x5B                 // '['
    b.eq pv_arr
    cmp  w0, 0x22                 // '"'
    b.eq pv_str
    cmp  w0, 0x74                 // 't'
    b.eq pv_true
    cmp  w0, 0x66                 // 'f'
    b.eq pv_false
    cmp  w0, 0x6E                 // 'n'
    b.eq pv_null
    cmp  w0, 0x2D                 // '-'
    b.eq pv_num
    cmp  w0, 0x30                 // '0'
    b.lo pv_error
    cmp  w0, 0x39                 // '9'
    b.hi pv_error
pv_num:
    bl   parse_number
    b    pv_ret
pv_obj:
    bl   parse_object
    b    pv_ret
pv_arr:
    bl   parse_array
    b    pv_ret
pv_str:
    bl   parse_string
    b    pv_ret
pv_true:
    adrp x1, lit_true
    add  x1, x1, :lo12:lit_true
    mov  x2, 4
    bl   match_lit
    b    pv_ret
pv_false:
    adrp x1, lit_false
    add  x1, x1, :lo12:lit_false
    mov  x2, 5
    bl   match_lit
    b    pv_ret
pv_null:
    adrp x1, lit_null
    add  x1, x1, :lo12:lit_null
    mov  x2, 4
    bl   match_lit
pv_ret:
    ldp  x29, x30, [sp], #16
    ret
pv_error:
    b    parse_error

// match x2 literal bytes at x1 against input at x19
match_lit:
    cmp  x19, x20
    b.hs parse_error              // truncated literal
    ldrb w0, [x19]
    ldrb w3, [x1]
    cmp  w0, w3
    b.ne parse_error
    add  x19, x19, 1
    add  x1, x1, 1
    subs x2, x2, 1
    b.ne match_lit
    ret

// ----------------------------------------------------------------
parse_string:                     // [x19] == '"' on entry
    add  x19, x19, 1
1:
    cmp  x19, x20
    b.hs parse_error              // unterminated string
    ldrb w0, [x19]
    cmp  w0, 0x22                 // closing quote
    b.eq 2f
    cmp  w0, 0x5C                 // backslash
    b.eq 3f
    cmp  w0, 0x20                 // control bytes 0x00-0x1F are errors
    b.lo parse_error
    add  x19, x19, 1
    b    1b
2:
    add  x19, x19, 1
    ret
3:                                // escape sequence
    add  x19, x19, 1
    cmp  x19, x20
    b.hs parse_error
    ldrb w0, [x19]
    cmp  w0, 0x75                 // 'u'
    b.eq 4f
    cmp  w0, 0x22                 // '"'
    b.eq 8f
    cmp  w0, 0x5C                 // '\'
    b.eq 8f
    cmp  w0, 0x2F                 // '/'
    b.eq 8f
    cmp  w0, 0x62                 // 'b'
    b.eq 8f
    cmp  w0, 0x66                 // 'f'
    b.eq 8f
    cmp  w0, 0x6E                 // 'n'
    b.eq 8f
    cmp  w0, 0x72                 // 'r'
    b.eq 8f
    cmp  w0, 0x74                 // 't'
    b.eq 8f
    b    parse_error              // invalid escape
8:
    add  x19, x19, 1
    b    1b
4:                                // \uXXXX
    add  x19, x19, 1
    mov  x3, 4
5:
    cmp  x19, x20
    b.hs parse_error
    ldrb w0, [x19]
    cmp  w0, 0x30                 // '0'
    b.lo parse_error
    cmp  w0, 0x39                 // '9'
    b.ls 6f
    orr  w1, w0, 0x20             // fold A-F to a-f
    cmp  w1, 0x61                 // 'a'
    b.lo parse_error
    cmp  w1, 0x66                 // 'f'
    b.hi parse_error
6:
    add  x19, x19, 1
    subs x3, x3, 1
    b.ne 5b
    b    1b

// ----------------------------------------------------------------
parse_number:                     // [x19] is '-' or a digit on entry
    ldrb w0, [x19]
    cmp  w0, 0x2D                 // '-'
    b.ne 1f
    add  x19, x19, 1
1:                                // int part
    cmp  x19, x20
    b.hs parse_error
    ldrb w0, [x19]
    cmp  w0, 0x30                 // '0'
    b.eq 2f
    cmp  w0, 0x31                 // must be '1'-'9'
    b.lo parse_error
    cmp  w0, 0x39
    b.hi parse_error
    add  x19, x19, 1
3:                                // remaining digits
    cmp  x19, x20
    b.hs 4f
    ldrb w0, [x19]
    cmp  w0, 0x30
    b.lo 4f
    cmp  w0, 0x39
    b.hi 4f
    add  x19, x19, 1
    b    3b
2:                                // leading zero: no more int digits
    add  x19, x19, 1
4:                                // fraction
    cmp  x19, x20
    b.hs 6f
    ldrb w0, [x19]
    cmp  w0, 0x2E                 // '.'
    b.ne 6f
    add  x19, x19, 1
    cmp  x19, x20
    b.hs parse_error              // "1." then EOF
    ldrb w0, [x19]
    cmp  w0, 0x30
    b.lo parse_error              // '.' must be followed by a digit
    cmp  w0, 0x39
    b.hi parse_error
    add  x19, x19, 1
5:
    cmp  x19, x20
    b.hs 6f
    ldrb w0, [x19]
    cmp  w0, 0x30
    b.lo 6f
    cmp  w0, 0x39
    b.hi 6f
    add  x19, x19, 1
    b    5b
6:                                // exponent
    cmp  x19, x20
    b.hs 9f
    ldrb w0, [x19]
    cmp  w0, 0x65                 // 'e'
    b.eq 7f
    cmp  w0, 0x45                 // 'E'
    b.ne 9f
7:
    add  x19, x19, 1
    cmp  x19, x20
    b.hs parse_error
    ldrb w0, [x19]
    cmp  w0, 0x2B                 // '+'
    b.eq 8f
    cmp  w0, 0x2D                 // '-'
    b.ne 10f
8:
    add  x19, x19, 1
10:
    cmp  x19, x20
    b.hs parse_error
    ldrb w0, [x19]
    cmp  w0, 0x30
    b.lo parse_error              // exponent needs at least one digit
    cmp  w0, 0x39
    b.hi parse_error
    add  x19, x19, 1
11:
    cmp  x19, x20
    b.hs 9f
    ldrb w0, [x19]
    cmp  w0, 0x30
    b.lo 9f
    cmp  w0, 0x39
    b.hi 9f
    add  x19, x19, 1
    b    11b
9:
    ret

// ----------------------------------------------------------------
parse_object:                     // [x19] == '{' on entry
    stp  x29, x30, [sp, #-16]!
    add  x19, x19, 1
    bl   skip_ws
    cmp  x19, x20
    b.hs 7f
    ldrb w0, [x19]
    cmp  w0, 0x7D                 // '}' -> empty object
    b.eq 6f
1:                                // one member
    cmp  x19, x20
    b.hs 7f
    ldrb w0, [x19]
    cmp  w0, 0x22                 // member key must be a string
    b.ne 7f
    bl   parse_string
    bl   skip_ws
    cmp  x19, x20
    b.hs 7f
    ldrb w0, [x19]
    cmp  w0, 0x3A                 // ':'
    b.ne 7f
    add  x19, x19, 1
    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x19, x20
    b.hs 7f
    ldrb w0, [x19]
    cmp  w0, 0x2C                 // ','
    b.eq 2f
    cmp  w0, 0x7D                 // '}'
    b.eq 6f
    b    7f
2:
    add  x19, x19, 1
    bl   skip_ws
    b    1b
6:
    add  x19, x19, 1
    ldp  x29, x30, [sp], #16
    ret
7:
    b    parse_error

// ----------------------------------------------------------------
parse_array:                      // [x19] == '[' on entry
    stp  x29, x30, [sp, #-16]!
    add  x19, x19, 1
    bl   skip_ws
    cmp  x19, x20
    b.hs 7f
    ldrb w0, [x19]
    cmp  w0, 0x5D                 // ']' -> empty array
    b.eq 6f
1:                                // one element
    bl   parse_value
    bl   skip_ws
    cmp  x19, x20
    b.hs 7f
    ldrb w0, [x19]
    cmp  w0, 0x2C                 // ','
    b.eq 2f
    cmp  w0, 0x5D                 // ']'
    b.eq 6f
    b    7f
2:
    add  x19, x19, 1
    bl   skip_ws
    b    1b
6:
    add  x19, x19, 1
    ldp  x29, x30, [sp], #16
    ret
7:
    b    parse_error

// ----------------------------------------------------------------
// print "error at N\n" with N = x19 - x21, then exit(1)
parse_error:
    sub  x22, x19, x21
    mov  x0, 1
    adrp x1, msg_error
    add  x1, x1, :lo12:msg_error
    mov  x2, 9
    mov  x8, 64
    svc  #0
    // convert N to decimal, digits stored backwards before a '\n'
    adrp x2, numbuf
    add  x2, x2, :lo12:numbuf
    mov  w3, 0x0A
    strb w3, [x2, #23]
    add  x2, x2, 23
    mov  x0, x22
    mov  x3, 10
1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0           // remainder
    add  w5, w5, 0x30
    sub  x2, x2, 1
    strb w5, [x2]
    mov  x0, x4
    cbnz x0, 1b
    adrp x4, numbuf
    add  x4, x4, :lo12:numbuf
    add  x4, x4, 24               // one past the '\n'
    mov  x1, x2                   // first digit
    sub  x2, x4, x1               // count incl. '\n'
    mov  x0, 1
    mov  x8, 64
    svc  #0
    mov  x0, 1
    mov  x8, 93
    svc  #0

// ----------------------------------------------------------------
.section .rodata
msg_ok:
    .ascii "ok\n"
msg_error:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
    .balign 8
buf:
    .zero 65536
numbuf:
    .zero 32
