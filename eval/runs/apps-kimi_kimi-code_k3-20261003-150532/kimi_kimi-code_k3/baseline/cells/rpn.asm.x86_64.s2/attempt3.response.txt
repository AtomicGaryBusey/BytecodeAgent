.globl _start
.text
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [rip + inbuf]
    add r13, rbx
    lea r15, [rip + outbuf]

line_start:
    cmp r12, r13
    jae all_done
    xor r14d, r14d

token_loop:
    cmp r12, r13
    jae line_eof
    movzx eax, byte ptr [r12]
    cmp al, 0x0A
    je line_nl
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    jmp parse_token
skip_sep:
    inc r12
    jmp token_loop

parse_token:
    mov rsi, r12
scan_tok:
    cmp r12, r13
    jae tok_end
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je tok_end
    cmp al, 0x09
    je tok_end
    cmp al, 0x0A
    je tok_end
    inc r12
    jmp scan_tok
tok_end:
    mov rcx, r12
    sub rcx, rsi
    cmp rcx, 1
    jne try_literal
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je do_operator
    cmp al, '-'
    je do_operator
    cmp al, '*'
    je do_operator
    cmp al, '/'
    je do_operator
    cmp al, '%'
    je do_operator
try_literal:
    xor r8d, r8d
    xor r9d, r9d
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne lit_digits
    mov r9d, 1
    inc rsi
    dec rcx
lit_digits:
    cmp rcx, 18
    ja tok_invalid
    test rcx, rcx
    jz tok_invalid
digit_loop:
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 9
    ja tok_invalid
    imul r8, r8, 10
    add r8, rax
    inc rsi
    dec rcx
    jnz digit_loop
    test r9d, r9d
    jz lit_push
    neg r8
lit_push:
    mov qword ptr [valstack + r14*8], r8
    inc r14
    jmp token_loop

do_operator:
    mov r9b, al
    cmp r14, 2
    jb line_error
    mov rdx, qword ptr [valstack + r14*8 - 8]
    mov rax, qword ptr [valstack + r14*8 - 16]
    dec r14
    cmp r9b, '+'
    je op_add
    cmp r9b, '-'
    je op_sub
    cmp r9b, '*'
    je op_mul
    cmp r9b, '/'
    je op_div
    cmp r9b, '%'
    je op_mod
    jmp line_error
op_add:
    add rax, rdx
    jmp op_store
op_sub:
    sub rax, rdx
    jmp op_store
op_mul:
    imul rax, rdx
    jmp op_store
op_div:
    test rdx, rdx
    jz line_error
    cmp rdx, -1
    jne do_div
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    je line_error
do_div:
    mov rcx, rdx
    cqo
    idiv rcx
    jmp op_store
op_mod:
    test rdx, rdx
    jz line_error
    cmp rdx, -1
    jne do_mod
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    je line_error
do_mod:
    mov rcx, rdx
    cqo
    idiv rcx
    mov rax, rdx
op_store:
    mov qword ptr [valstack + r14*8 - 8], rax
    jmp token_loop

line_nl:
    inc r12
line_eof:
    cmp r14, 1
    jne print_error
    mov rax, qword ptr [valstack]
    call print_value
    jmp line_start

print_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6
    jmp line_start

tok_invalid:
line_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6
err_skip:
    cmp r12, r13
    jae line_start
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 0x0A
    jne err_skip
    jmp line_start

print_value:
    test rax, rax
    jns pv_pos
    mov byte ptr [r15], '-'
    inc r15
    neg rax
pv_pos:
    mov rcx, 10
    lea rdi, [rip + digitbuf + 32]
pv_div:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pv_div
    lea rcx, [rip + digitbuf + 32]
    sub rcx, rdi
pv_copy:
    mov dl, byte ptr [rdi]
    mov byte ptr [r15], dl
    inc rdi
    inc r15
    dec rcx
    jnz pv_copy
    mov byte ptr [r15], 0x0A
    inc r15
    ret

all_done:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz prog_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle prog_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
prog_exit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 8
valstack: .zero 4096
inbuf: .zero 65536
digitbuf: .zero 32
outbuf: .zero 400000
