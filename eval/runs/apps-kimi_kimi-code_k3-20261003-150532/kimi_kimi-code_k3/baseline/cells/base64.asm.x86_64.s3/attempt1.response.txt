.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:  .zero 49152
outbuf: .zero 66560

.text
_start:
    # read all of stdin into inbuf
    xor r12, r12                # total bytes read
1:
    mov rdx, 49152
    sub rdx, r12
    jz 2f
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jz 2f
    add r12, rax
    jmp 1b
2:
    # encode r12 bytes from inbuf into outbuf with 76-column wrapping
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alphabet]
    xor r13, r13                # column
    xor r14, r14                # input index
3:
    mov rax, r12
    sub rax, r14
    cmp rax, 3
    jb 5f
    movzx ecx, byte ptr [rsi + r14]
    movzx edx, byte ptr [rsi + r14 + 1]
    movzx r8d, byte ptr [rsi + r14 + 2]
    mov eax, ecx
    shr eax, 2                  # sextet 0
    mov r9d, ecx
    and r9d, 3
    shl r9d, 4
    mov r10d, edx
    shr r10d, 4
    or r9d, r10d                # sextet 1
    mov r10d, edx
    and r10d, 15
    shl r10d, 2
    mov r11d, r8d
    shr r11d, 6
    or r10d, r11d               # sextet 2
    and r8d, 63                 # sextet 3
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    mov al, byte ptr [rbx + r9]
    mov byte ptr [rdi + 1], al
    mov al, byte ptr [rbx + r10]
    mov byte ptr [rdi + 2], al
    mov al, byte ptr [rbx + r8]
    mov byte ptr [rdi + 3], al
    add rdi, 4
    add r14, 3
    add r13, 4
    cmp r13, 76
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    xor r13, r13
    jmp 3b
5:
    # final partial group: rax = 0, 1 or 2 leftover bytes
    test rax, rax
    jz 9f
    cmp rax, 1
    je 6f
    # 2 leftover bytes -> 3 chars + '='
    movzx ecx, byte ptr [rsi + r14]
    movzx edx, byte ptr [rsi + r14 + 1]
    mov eax, ecx
    shr eax, 2
    mov r9d, ecx
    and r9d, 3
    shl r9d, 4
    mov r10d, edx
    shr r10d, 4
    or r9d, r10d
    mov r10d, edx
    and r10d, 15
    shl r10d, 2
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    mov al, byte ptr [rbx + r9]
    mov byte ptr [rdi + 1], al
    mov al, byte ptr [rbx + r10]
    mov byte ptr [rdi + 2], al
    mov byte ptr [rdi + 3], '='
    add rdi, 4
    add r13, 4
    jmp 8f
6:
    # 1 leftover byte -> 2 chars + '=='
    movzx ecx, byte ptr [rsi + r14]
    mov eax, ecx
    shr eax, 2
    mov r9d, ecx
    and r9d, 3
    shl r9d, 4
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    mov al, byte ptr [rbx + r9]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], '='
    mov byte ptr [rdi + 3], '='
    add rdi, 4
    add r13, 4
8:
    cmp r13, 76
    jb 9f
    mov byte ptr [rdi], 10
    inc rdi
    xor r13, r13
9:
    # newline at end of last (short) line
    test r13, r13
    jz 10f
    mov byte ptr [rdi], 10
    inc rdi
10:
    # write outbuf to stdout
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                # total length
11:
    test rdx, rdx
    jz 12f
    mov eax, 1                  # write
    mov edi, 1                  # fd 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp 11b
12:
    mov eax, 60                 # exit
    xor edi, edi
    syscall
