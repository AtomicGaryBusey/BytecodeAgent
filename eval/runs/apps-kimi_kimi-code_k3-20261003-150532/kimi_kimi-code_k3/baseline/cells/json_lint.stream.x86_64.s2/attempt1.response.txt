.globl _start

.text

_start:
    lea r12, [rip + inbuf]
    xor r15d, r15d
read_loop:
    mov edx, 65536
    sub rdx, r15
    jz parse_start
    lea rsi, [r12 + r15]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle parse_start
    add r15, rax
    jmp read_loop
parse_start:
    mov r13, r12
    lea r14, [r12 + r15]
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov rax, r13
    sub rax, r12
    lea r8, [rip + numbuf + 32]
    mov byte ptr [r8], 0x0A
    mov rcx, 10
1:
    dec r8
    xor edx, edx
    div rcx
    add dl, '0'
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rdx, [rip + numbuf + 33]
    sub rdx, r8
    mov rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
1:
    cmp r13, r14
    jae 2f
    mov al, byte ptr [r13]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
2:
    ret
3:
    inc r13
    jmp 1b

parse_value:
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je lit_true
    cmp al, 'f'
    je lit_false
    cmp al, 'n'
    je lit_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp parse_number

parse_string:
    inc r13
str_loop:
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '"'
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb fail
    inc r13
    jmp str_loop
str_end:
    inc r13
    ret
str_esc:
    inc r13
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '"'
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    je str_hex
    jmp fail
str_esc_ok:
    inc r13
    jmp str_loop
str_hex:
    inc r13
    mov ecx, 4
str_hex_loop:
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe str_hex_ok
    or al, 0x20
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
str_hex_ok:
    inc r13
    dec ecx
    jnz str_hex_loop
    jmp str_loop

parse_number:
    mov al, byte ptr [r13]
    cmp al, '-'
    jne num_int
    inc r13
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
num_int:
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb fail
    cmp al, '9'
    ja fail
    inc r13
num_int_loop:
    cmp r13, r14
    jae num_frac
    mov al, byte ptr [r13]
    cmp al, '0'
    jb num_frac
    cmp al, '9'
    ja num_frac
    inc r13
    jmp num_int_loop
num_zero:
    inc r13
num_frac:
    cmp r13, r14
    jae num_ret
    mov al, byte ptr [r13]
    cmp al, '.'
    jne num_exp
    inc r13
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    inc r13
num_frac_loop:
    cmp r13, r14
    jae num_ret
    mov al, byte ptr [r13]
    cmp al, '0'
    jb num_exp
    cmp al, '9'
    ja num_exp
    inc r13
    jmp num_frac_loop
num_exp:
    cmp al, 'e'
    je num_exp_go
    cmp al, 'E'
    je num_exp_go
    jmp num_ret
num_exp_go:
    inc r13
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '+'
    je num_exp_sign
    cmp al, '-'
    je num_exp_sign
    jmp num_exp_first
num_exp_sign:
    inc r13
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
num_exp_first:
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    inc r13
num_exp_loop:
    cmp r13, r14
    jae num_ret
    mov al, byte ptr [r13]
    cmp al, '0'
    jb num_ret
    cmp al, '9'
    ja num_ret
    inc r13
    jmp num_exp_loop
num_ret:
    ret

lit_true:
    lea rsi, [rip + s_true]
    mov ecx, 4
    jmp lit_check
lit_false:
    lea rsi, [rip + s_false]
    mov ecx, 5
    jmp lit_check
lit_null:
    lea rsi, [rip + s_null]
    mov ecx, 4
lit_check:
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, byte ptr [rsi]
    jne fail
    inc r13
    inc rsi
    dec ecx
    jnz lit_check
    ret

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, '}'
    je obj_close
obj_member:
    cmp al, '"'
    jne fail
    call parse_string
    call skip_ws
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, ':'
    jne fail
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, ','
    je obj_next
    cmp al, '}'
    je obj_close
    jmp fail
obj_next:
    inc r13
    call skip_ws
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    jmp obj_member
obj_close:
    inc r13
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, ']'
    je arr_close
arr_elem:
    call parse_value
    call skip_ws
    cmp r13, r14
    jae fail
    mov al, byte ptr [r13]
    cmp al, ','
    je arr_next
    cmp al, ']'
    je arr_close
    jmp fail
arr_next:
    inc r13
    call skip_ws
    jmp arr_elem
arr_close:
    inc r13
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"

.bss
inbuf:
    .zero 65536
numbuf:
    .zero 33
