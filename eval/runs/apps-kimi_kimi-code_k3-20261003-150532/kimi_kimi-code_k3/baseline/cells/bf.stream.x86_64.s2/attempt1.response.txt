.globl _start

.text
_start:
    xor r15, r15
read_loop:
    mov rdx, 65536
    sub rdx, r15
    jz read_done
    lea rsi, [rip + buf]
    add rsi, r15
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop
read_done:
    lea r12, [rip + buf]
    lea r15, [r12 + r15]
    mov rax, r12
find_bang:
    cmp rax, r15
    jae no_bang
    cmp byte ptr [rax], 0x21
    je found_bang
    inc rax
    jmp find_bang
found_bang:
    mov rbp, rax
    lea r14, [rax + 1]
    jmp split_done
no_bang:
    mov rbp, r15
    mov r14, r15
split_done:
    lea r8, [rip + bstack]
    xor ecx, ecx
build_loop:
    lea rax, [r12 + rcx]
    cmp rax, rbp
    jae build_done
    movzx edx, byte ptr [rax]
    cmp dl, 0x5B
    je build_push
    cmp dl, 0x5D
    je build_pop
build_next:
    inc rcx
    jmp build_loop
build_push:
    mov [r8], rcx
    add r8, 8
    jmp build_next
build_pop:
    lea rax, [rip + bstack]
    cmp r8, rax
    je bad
    sub r8, 8
    mov rdx, [r8]
    lea rsi, [rip + jtab]
    lea rax, [r12 + rcx + 1]
    mov [rsi + rdx*8], rax
    lea rax, [r12 + rdx + 1]
    mov [rsi + rcx*8], rax
    jmp build_next
build_done:
    lea rax, [rip + bstack]
    cmp r8, rax
    jne bad
    mov rbx, r12
    lea r13, [rip + tape]
exec_loop:
    cmp rbx, rbp
    jae run_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_dot
    cmp al, 0x2C
    je cmd_comma
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    inc rbx
    jmp exec_loop
cmd_gt:
    inc r13
    inc rbx
    jmp exec_loop
cmd_lt:
    dec r13
    inc rbx
    jmp exec_loop
cmd_plus:
    inc byte ptr [r13]
    inc rbx
    jmp exec_loop
cmd_minus:
    dec byte ptr [r13]
    inc rbx
    jmp exec_loop
cmd_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov edx, 1
    syscall
    inc rbx
    jmp exec_loop
cmd_comma:
    cmp r14, r15
    jae comma_eof
    mov al, [r14]
    inc r14
    mov [r13], al
    inc rbx
    jmp exec_loop
comma_eof:
    mov byte ptr [r13], 0
    inc rbx
    jmp exec_loop
cmd_open:
    cmp byte ptr [r13], 0
    jne open_no
    mov rax, rbx
    sub rax, r12
    lea rsi, [rip + jtab]
    mov rbx, [rsi + rax*8]
    jmp exec_loop
open_no:
    inc rbx
    jmp exec_loop
cmd_close:
    cmp byte ptr [r13], 0
    je close_no
    mov rax, rbx
    sub rax, r12
    lea rsi, [rip + jtab]
    mov rbx, [rsi + rax*8]
    jmp exec_loop
close_no:
    inc rbx
    jmp exec_loop
run_done:
    xor edi, edi
    mov eax, 60
    syscall
bad:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 8
buf:
    .zero 65536
jtab:
    .zero 524288
bstack:
    .zero 320000
tape:
    .zero 30000
