.globl _start
.text
_start:
	lea rbx, [rip + inbuf]
read_loop:
	lea rdx, [rip + inbuf + 65536]
	sub rdx, rbx
	jz read_done
	xor eax, eax
	xor edi, edi
	mov rsi, rbx
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	jmp read_loop
read_done:
	lea rsi, [rip + inbuf]
	lea r15, [rip + outbuf]

# main loop over lines; rsi = cursor, rbx = end of input
line_loop:
	cmp rsi, rbx
	jae all_done
	mov r12, rsi
find_nl:
	cmp r12, rbx
	jae have_line
	cmp byte ptr [r12], 10
	je have_line
	inc r12
	jmp find_nl
have_line:
	# line is [rsi, r12); r13 = value stack depth
	xor r13d, r13d

token_loop:
	cmp rsi, r12
	jae line_done
	mov al, byte ptr [rsi]
	cmp al, ' '
	je skip_sep
	cmp al, 9
	je skip_sep
	# token starts at r8, ends at r9
	mov r8, rsi
	mov r9, rsi
tok_scan:
	cmp r9, r12
	jae tok_got
	mov al, byte ptr [r9]
	cmp al, ' '
	je tok_got
	cmp al, 9
	je tok_got
	inc r9
	jmp tok_scan
tok_got:
	mov rsi, r9
	mov rdx, r9
	sub rdx, r8
	cmp rdx, 1
	jne multi
	movzx eax, byte ptr [r8]
	cmp al, '0'
	jb maybe_op
	cmp al, '9'
	ja error
	sub eax, '0'
	cmp r13, 1024
	jae error
	lea rcx, [rip + valstack]
	mov [rcx + r13*8], rax
	inc r13
	jmp token_loop
maybe_op:
	cmp al, '+'
	je operator
	cmp al, '-'
	je operator
	cmp al, '*'
	je operator
	cmp al, '/'
	je operator
	cmp al, '%'
	je operator
	jmp error

# multi-byte token: must be optional '-' plus 1..18 digits
multi:
	mov r10, r8
	xor r11d, r11d
	cmp byte ptr [r10], '-'
	jne no_sign
	mov r11d, 1
	inc r10
no_sign:
	mov rax, r9
	sub rax, r10
	cmp rax, 18
	ja error
	xor edx, edx
digit_loop:
	cmp r10, r9
	jae digits_done
	movzx eax, byte ptr [r10]
	sub eax, '0'
	cmp eax, 9
	ja error
	imul rdx, rdx, 10
	add rdx, rax
	inc r10
	jmp digit_loop
digits_done:
	test r11d, r11d
	jz no_neg
	neg rdx
no_neg:
	cmp r13, 1024
	jae error
	lea rcx, [rip + valstack]
	mov [rcx + r13*8], rdx
	inc r13
	jmp token_loop

skip_sep:
	inc rsi
	jmp token_loop

# operator: pop b (rdx) then a (rax), push a op b
# cqo clobbers rdx, so the divisor is moved to r10 before idiv
operator:
	cmp r13, 2
	jb error
	lea rcx, [rip + valstack]
	mov rdx, [rcx + r13*8 - 8]
	mov rax, [rcx + r13*8 - 16]
	sub r13, 2
	movzx r10d, byte ptr [r8]
	cmp r10b, '+'
	je op_add
	cmp r10b, '-'
	je op_sub
	cmp r10b, '*'
	je op_mul
	cmp r10b, '/'
	je op_div
op_mod:
	test rdx, rdx
	jz error
	cmp rdx, -1
	jne do_mod
	mov r10, 0x8000000000000000
	cmp rax, r10
	je error
do_mod:
	mov r10, rdx
	cqo
	idiv r10
	mov rax, rdx
	jmp push_res
op_div:
	test rdx, rdx
	jz error
	cmp rdx, -1
	jne do_div
	mov r10, 0x8000000000000000
	cmp rax, r10
	je error
do_div:
	mov r10, rdx
	cqo
	idiv r10
	jmp push_res
op_add:
	add rax, rdx
	jmp push_res
op_sub:
	sub rax, rdx
	jmp push_res
op_mul:
	imul rax, rdx
push_res:
	mov [rcx + r13*8], rax
	inc r13
	jmp token_loop

line_done:
	cmp r13, 1
	jne error
	lea rcx, [rip + valstack]
	mov rax, [rcx]
	call print_value
	jmp next_line
error:
	mov dword ptr [r15], 0x6F727265
	mov byte ptr [r15 + 4], 'r'
	mov byte ptr [r15 + 5], 10
	add r15, 6
next_line:
	lea rsi, [r12 + 1]
	jmp line_loop

all_done:
	lea rsi, [rip + outbuf]
	mov rdx, r15
	sub rdx, rsi
write_loop:
	test rdx, rdx
	jz do_exit
	mov eax, 1
	mov edi, 1
	syscall
	test rax, rax
	jle do_exit
	add rsi, rax
	sub rdx, rax
	jmp write_loop
do_exit:
	mov eax, 60
	xor edi, edi
	syscall

# print_value: append rax in decimal plus newline to outbuf at r15
print_value:
	test rax, rax
	jns pv_pos
	mov byte ptr [r15], '-'
	inc r15
	neg rax
pv_pos:
	lea rdi, [rip + numbuf + 24]
	mov ecx, 10
pv_conv:
	xor edx, edx
	div rcx
	add dl, '0'
	dec rdi
	mov [rdi], dl
	test rax, rax
	jnz pv_conv
	lea rcx, [rip + numbuf + 24]
pv_copy:
	mov dl, [rdi]
	mov [r15], dl
	inc rdi
	inc r15
	cmp rdi, rcx
	jb pv_copy
	mov byte ptr [r15], 10
	inc r15
	ret

.bss
.align 8
inbuf:
	.zero 65536
.align 8
outbuf:
	.zero 400000
.align 8
valstack:
	.zero 8192
.align 8
numbuf:
	.zero 24
