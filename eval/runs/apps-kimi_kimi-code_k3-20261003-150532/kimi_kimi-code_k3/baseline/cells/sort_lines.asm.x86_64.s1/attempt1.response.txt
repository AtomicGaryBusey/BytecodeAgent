.globl _start

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 32768
    sub rdx, r12
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r14, r14
    xor rbx, rbx
    xor rcx, rcx
scan_loop:
    cmp rcx, r12
    jae scan_end
    lea rax, [rip + inbuf]
    cmp byte ptr [rax + rcx], 10
    jne scan_next
    lea rax, [rip + lines]
    mov rdx, r14
    shl rdx, 4
    mov [rax + rdx], rbx
    mov r8, rcx
    sub r8, rbx
    mov [rax + rdx + 8], r8
    inc r14
    lea rbx, [rcx + 1]
scan_next:
    inc rcx
    jmp scan_loop
scan_end:
    cmp rbx, r12
    jae split_done
    lea rax, [rip + lines]
    mov rdx, r14
    shl rdx, 4
    mov [rax + rdx], rbx
    mov r8, r12
    sub r8, rbx
    mov [rax + rdx + 8], r8
    inc r14
split_done:
    mov r15, 1
sort_outer:
    cmp r15, r14
    jae sort_done
    lea rax, [rip + lines]
    mov rdx, r15
    shl rdx, 4
    mov r8, [rax + rdx]
    mov r9, [rax + rdx + 8]
    push r8
    push r9
    xor rbx, rbx
    mov rbp, r15
bin_loop:
    cmp rbx, rbp
    jae bin_done
    lea r13, [rbx + rbp]
    shr r13, 1
    lea rax, [rip + lines]
    mov r11, r13
    shl r11, 4
    mov rdi, [rax + r11]
    mov rsi, [rax + r11 + 8]
    mov rdx, [rsp + 8]
    mov rcx, [rsp]
    call compare
    test eax, eax
    jle go_right
    mov rbp, r13
    jmp bin_loop
go_right:
    lea rbx, [r13 + 1]
    jmp bin_loop
bin_done:
    mov rcx, r15
shift_loop:
    cmp rcx, rbx
    jbe shift_done
    lea rax, [rip + lines]
    mov rdx, rcx
    shl rdx, 4
    mov r8, [rax + rdx - 16]
    mov r9, [rax + rdx - 8]
    mov [rax + rdx], r8
    mov [rax + rdx + 8], r9
    dec rcx
    jmp shift_loop
shift_done:
    lea rax, [rip + lines]
    mov rdx, rbx
    shl rdx, 4
    pop r9
    pop r8
    mov [rax + rdx], r8
    mov [rax + rdx + 8], r9
    inc r15
    jmp sort_outer
sort_done:
    lea rdi, [rip + outbuf]
    xor r15, r15
out_loop:
    cmp r15, r14
    jae out_done
    lea rax, [rip + lines]
    mov rdx, r15
    shl rdx, 4
    mov rsi, [rax + rdx]
    mov rcx, [rax + rdx + 8]
    lea r8, [rip + inbuf]
    add rsi, r8
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r15
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit0
    mov rax, 1
    mov rdi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit0:
    mov rax, 60
    xor rdi, rdi
    syscall

compare:
    lea r8, [rip + inbuf]
    add rdi, r8
    add rdx, r8
    mov r9, rsi
    cmp rcx, rsi
    cmovb r9, rcx
    xor r10, r10
cmp_loop:
    cmp r10, r9
    jae cmp_prefix
    movzx eax, byte ptr [rdi + r10]
    movzx r8d, byte ptr [rdx + r10]
    cmp eax, r8d
    jb cmp_less
    ja cmp_greater
    inc r10
    jmp cmp_loop
cmp_prefix:
    cmp rsi, rcx
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret

.bss
.align 16
inbuf:
    .zero 32768
lines:
    .zero 16384
outbuf:
    .zero 34816
