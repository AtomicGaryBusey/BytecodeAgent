.text
.globl _start
_start:
	// x21 = input buffer base, x19 = total bytes read
	adrp x21, inbuf
	add x21, x21, :lo12:inbuf
	mov x19, #0

read_loop:
	mov x0, #0				// fd 0
	add x1, x21, x19			// buf + total
	mov x2, #65536
	sub x2, x2, x19				// remaining space
	mov x8, #63				// read
	svc #0
	cbz x0, read_done
	add x19, x19, x0
	b read_loop

read_done:
	mov x20, #0				// current row offset

row_loop:
	cmp x20, x19
	b.hs final_line

	// build one output line in outbuf, x1 = write pointer
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	mov w2, w20				// offset
	bl hex8
	mov w4, #0x20
	strb w4, [x1], #1
	strb w4, [x1], #1			// two spaces

	// x6 = n = min(total - off, 16)
	sub x6, x19, x20
	mov x7, #16
	cmp x6, x7
	csel x6, x6, x7, lo

	add x10, x21, x20			// row data pointer
	adrp x5, hexdig
	add x5, x5, :lo12:hexdig

	mov x9, #0				// i = 0..15
byte_loop:
	cmp x9, x6
	b.hs pad_slot
	ldrb w4, [x10, x9]
	lsr w11, w4, #4
	ldrb w11, [x5, x11]
	strb w11, [x1], #1
	and w11, w4, #0xf
	ldrb w11, [x5, x11]
	strb w11, [x1], #1
	mov w11, #0x20
	strb w11, [x1], #1
	b slot_done
pad_slot:
	mov w11, #0x20
	strb w11, [x1], #1
	strb w11, [x1], #1
	strb w11, [x1], #1
slot_done:
	cmp x9, #7
	b.ne no_extra
	mov w11, #0x20
	strb w11, [x1], #1			// extra space after slot 7
no_extra:
	add x9, x9, #1
	cmp x9, #16
	b.lo byte_loop

	// " |"
	mov w11, #0x20
	strb w11, [x1], #1
	mov w11, #0x7C
	strb w11, [x1], #1

	mov x9, #0				// i = 0..n-1
char_loop:
	cmp x9, x6
	b.hs char_done
	ldrb w4, [x10, x9]
	cmp w4, #0x20
	b.lo dot
	cmp w4, #0x7E
	b.ls store_char
dot:
	mov w4, #0x2E				// '.'
store_char:
	strb w4, [x1], #1
	add x9, x9, #1
	b char_loop
char_done:
	mov w11, #0x7C
	strb w11, [x1], #1
	mov w11, #0x0A
	strb w11, [x1], #1

	// write(1, outbuf, x1 - outbuf)
	adrp x2, outbuf
	add x2, x2, :lo12:outbuf
	sub x2, x1, x2
	mov x0, #1
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	mov x8, #64				// write
	svc #0

	add x20, x20, #16
	b row_loop

final_line:
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	mov w2, w19				// total length
	bl hex8
	mov w11, #0x0A
	strb w11, [x1], #1
	mov x0, #1
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	mov x2, #9
	mov x8, #64				// write
	svc #0

	mov x0, #0
	mov x8, #93				// exit
	svc #0

// hex8: print w2 as 8 lowercase hex digits at x1; advances x1 by 8
hex8:
	mov x3, #28
hex8_loop:
	lsr w4, w2, w3
	and w4, w4, #0xf
	adrp x5, hexdig
	add x5, x5, :lo12:hexdig
	ldrb w4, [x5, x4]
	strb w4, [x1], #1
	subs x3, x3, #4
	b.pl hex8_loop
	ret

.section .rodata
hexdig: .ascii "0123456789abcdef"

.bss
.align 4
inbuf:	.zero 65536
outbuf:	.zero 128
