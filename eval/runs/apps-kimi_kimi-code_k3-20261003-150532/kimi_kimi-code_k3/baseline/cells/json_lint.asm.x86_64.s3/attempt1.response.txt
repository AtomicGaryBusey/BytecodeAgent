.text
.globl _start

_start:
    xor ebx, ebx
read_loop:
    mov eax, 65536
    sub rax, rbx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edi, edi
    mov rdx, rax
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov [rip + inlen], rbx
    mov qword ptr [rip + pos], 0
    call parse_value
    test eax, eax
    jz input_error
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jne input_error
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 2
    call write_all
    xor edi, edi
    jmp exit_now

input_error:
    mov rax, [rip + pos]
    lea rsi, [rip + outbuf]
    lea rdi, [rip + errprefix]
    mov ecx, 9
copy_prefix:
    mov bl, [rdi]
    mov [rsi], bl
    inc rdi
    inc rsi
    dec ecx
    jnz copy_prefix
    lea rdi, [rip + numbuf + 32]
    mov byte ptr [rdi], 10
    mov ebx, 10
convert_loop:
    dec rdi
    xor edx, edx
    div rbx
    add dl, 0x30
    mov [rdi], dl
    test rax, rax
    jnz convert_loop
    lea rdx, [rip + numbuf + 32]
    sub rdx, rdi
    inc rdx
    mov rcx, rdx
copy_num:
    mov bl, [rdi]
    mov [rsi], bl
    inc rdi
    inc rsi
    dec ecx
    jnz copy_num
    add rdx, 9
    mov edi, 1
    lea rsi, [rip + outbuf]
    call write_all
    mov edi, 1
exit_now:
    mov eax, 60
    syscall

write_all:
    test rdx, rdx
    jz write_done
    mov eax, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_all
write_done:
    ret

skip_ws:
    mov rax, [rip + pos]
    mov rcx, [rip + inlen]
    lea rdx, [rip + inbuf]
skip_ws_loop:
    cmp rax, rcx
    jae skip_ws_done
    mov bl, [rdx + rax]
    cmp bl, 0x20
    je skip_ws_inc
    cmp bl, 0x09
    je skip_ws_inc
    cmp bl, 0x0A
    je skip_ws_inc
    cmp bl, 0x0D
    jne skip_ws_done
skip_ws_inc:
    inc rax
    jmp skip_ws_loop
skip_ws_done:
    mov [rip + pos], rax
    ret

parse_value:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_value_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x7B
    je parse_object
    cmp bl, 0x5B
    je parse_array
    cmp bl, 0x22
    je parse_string
    cmp bl, 0x2D
    je parse_number
    cmp bl, 0x74
    je parse_true
    cmp bl, 0x66
    je parse_false
    cmp bl, 0x6E
    je parse_null
    cmp bl, 0x30
    jb parse_value_bad
    cmp bl, 0x39
    jbe parse_number
parse_value_bad:
    xor eax, eax
    ret
parse_value_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
    xor eax, eax
    ret

parse_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp expect_lit
parse_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp expect_lit
parse_null:
    lea rsi, [rip + lit_null]
    mov edx, 4
    jmp expect_lit

expect_lit:
    mov rax, [rip + pos]
    mov rcx, [rip + inlen]
    lea rdi, [rip + inbuf]
    xor r8d, r8d
expect_lit_loop:
    cmp r8, rdx
    je expect_lit_ok
    mov r9, rax
    add r9, r8
    cmp r9, rcx
    jae expect_lit_eof
    mov bl, [rdi + r9]
    cmp bl, [rsi + r8]
    jne expect_lit_bad
    inc r8
    jmp expect_lit_loop
expect_lit_ok:
    add [rip + pos], rdx
    mov eax, 1
    ret
expect_lit_bad:
    mov [rip + pos], r9
    xor eax, eax
    ret
expect_lit_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
    xor eax, eax
    ret

parse_string:
    inc qword ptr [rip + pos]
parse_string_loop:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_string_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x22
    je parse_string_close
    cmp bl, 0x5C
    je parse_string_escape
    cmp bl, 0x20
    jb parse_string_bad
    inc qword ptr [rip + pos]
    jmp parse_string_loop
parse_string_close:
    inc qword ptr [rip + pos]
    mov eax, 1
    ret
parse_string_bad:
    xor eax, eax
    ret
parse_string_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
    xor eax, eax
    ret
parse_string_escape:
    inc qword ptr [rip + pos]
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_string_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x22
    je parse_string_esc_one
    cmp bl, 0x5C
    je parse_string_esc_one
    cmp bl, 0x2F
    je parse_string_esc_one
    cmp bl, 0x62
    je parse_string_esc_one
    cmp bl, 0x66
    je parse_string_esc_one
    cmp bl, 0x6E
    je parse_string_esc_one
    cmp bl, 0x72
    je parse_string_esc_one
    cmp bl, 0x74
    je parse_string_esc_one
    cmp bl, 0x75
    jne parse_string_bad
    inc qword ptr [rip + pos]
    xor r8d, r8d
parse_string_hex_loop:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_string_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    jb parse_string_bad
    cmp bl, 0x39
    jbe parse_string_hex_ok
    cmp bl, 0x41
    jb parse_string_bad
    cmp bl, 0x46
    jbe parse_string_hex_ok
    cmp bl, 0x61
    jb parse_string_bad
    cmp bl, 0x66
    ja parse_string_bad
parse_string_hex_ok:
    inc qword ptr [rip + pos]
    inc r8d
    cmp r8d, 4
    jne parse_string_hex_loop
    jmp parse_string_loop
parse_string_esc_one:
    inc qword ptr [rip + pos]
    jmp parse_string_loop

parse_number:
    mov rax, [rip + pos]
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x2D
    jne parse_number_int
    inc qword ptr [rip + pos]
parse_number_int:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    je parse_number_zero
    cmp bl, 0x31
    jb parse_number_bad
    cmp bl, 0x39
    ja parse_number_bad
    inc qword ptr [rip + pos]
parse_number_digits:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_frac
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    jb parse_number_frac
    cmp bl, 0x39
    ja parse_number_frac
    inc qword ptr [rip + pos]
    jmp parse_number_digits
parse_number_zero:
    inc qword ptr [rip + pos]
parse_number_frac:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_exp
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x2E
    jne parse_number_exp
    inc qword ptr [rip + pos]
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    jb parse_number_bad
    cmp bl, 0x39
    ja parse_number_bad
    inc qword ptr [rip + pos]
parse_number_frac_digits:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_exp
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    jb parse_number_exp
    cmp bl, 0x39
    ja parse_number_exp
    inc qword ptr [rip + pos]
    jmp parse_number_frac_digits
parse_number_exp:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_ok
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x65
    je parse_number_exp_mark
    cmp bl, 0x45
    jne parse_number_ok
parse_number_exp_mark:
    inc qword ptr [rip + pos]
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_exp_need
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x2B
    je parse_number_exp_sign
    cmp bl, 0x2D
    jne parse_number_exp_need
parse_number_exp_sign:
    inc qword ptr [rip + pos]
parse_number_exp_need:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    jb parse_number_bad
    cmp bl, 0x39
    ja parse_number_bad
    inc qword ptr [rip + pos]
parse_number_exp_digits:
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_number_ok
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x30
    jb parse_number_ok
    cmp bl, 0x39
    ja parse_number_ok
    inc qword ptr [rip + pos]
    jmp parse_number_exp_digits
parse_number_ok:
    mov eax, 1
    ret
parse_number_bad:
    xor eax, eax
    ret
parse_number_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
    xor eax, eax
    ret

parse_member:
    call parse_string
    test eax, eax
    jz parse_member_fail
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_member_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x3A
    jne parse_member_bad
    inc qword ptr [rip + pos]
    jmp parse_value
parse_member_bad:
    xor eax, eax
    ret
parse_member_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
parse_member_fail:
    xor eax, eax
    ret

parse_object:
    inc qword ptr [rip + pos]
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_object_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x7D
    je parse_object_close
    cmp bl, 0x22
    jne parse_object_bad
    call parse_member
    test eax, eax
    jz parse_object_fail
parse_object_after:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_object_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x7D
    je parse_object_close
    cmp bl, 0x2C
    jne parse_object_bad
    inc qword ptr [rip + pos]
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_object_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x22
    jne parse_object_bad
    call parse_member
    test eax, eax
    jz parse_object_fail
    jmp parse_object_after
parse_object_close:
    inc qword ptr [rip + pos]
    mov eax, 1
    ret
parse_object_bad:
    xor eax, eax
    ret
parse_object_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
parse_object_fail:
    xor eax, eax
    ret

parse_array:
    inc qword ptr [rip + pos]
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_array_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x5D
    je parse_array_close
    call parse_value
    test eax, eax
    jz parse_array_fail
parse_array_after:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + inlen]
    jae parse_array_eof
    lea rdx, [rip + inbuf]
    mov bl, [rdx + rax]
    cmp bl, 0x5D
    je parse_array_close
    cmp bl, 0x2C
    jne parse_array_bad
    inc qword ptr [rip + pos]
    call parse_value
    test eax, eax
    jz parse_array_fail
    jmp parse_array_after
parse_array_close:
    inc qword ptr [rip + pos]
    mov eax, 1
    ret
parse_array_bad:
    xor eax, eax
    ret
parse_array_eof:
    mov rax, [rip + inlen]
    mov [rip + pos], rax
parse_array_fail:
    xor eax, eax
    ret

.section .rodata
okmsg:
    .ascii "ok\n"
errprefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
pos:
    .zero 8
inlen:
    .zero 8
outbuf:
    .zero 64
numbuf:
    .zero 40
inbuf:
    .zero 65536
