.globl _start

.text

_start:
    // read all of stdin into inbuf
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf   // x20 = buffer base
    mov x19, x20                // x19 = write pointer
1:
    sub x3, x19, x20            // bytes stored so far
    mov x2, #65536
    sub x2, x2, x3              // space remaining
    cbz x2, 2f
    mov x0, #0                  // fd 0 = stdin
    mov x1, x19
    mov x8, #63                 // read
    svc #0
    cbz x0, 2f                  // 0 = end of input
    add x19, x19, x0
    b 1b
2:
    sub x19, x19, x20           // x19 = total byte count

    // scan the buffer
    mov x9, x20                 // read pointer
    mov x10, x19                // bytes left
    mov x11, #0                 // line count
    mov x12, #0                 // word count
    mov x13, #0                 // inside-a-word flag
3:
    cbz x10, 6f
    ldrb w14, [x9], #1
    sub x10, x10, #1
    cmp w14, #10                // newline?
    b.ne 4f
    add x11, x11, #1
4:
    cmp w14, #32                // space?
    b.eq 5f
    sub w15, w14, #9            // tab..CR = 9..13
    cmp w15, #4
    b.ls 5f                     // unsigned <= 4 -> whitespace
    // word byte
    cbnz x13, 3b
    mov x13, #1
    add x12, x12, #1
    b 3b
5:
    mov x13, #0                 // whitespace ends any word
    b 3b
6:

    // build "<lines> <words> <bytes>\n" in outbuf
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x0, x11
    bl append_num
    mov w2, #32
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x12
    bl append_num
    mov w2, #32
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x19
    bl append_num
    mov w2, #10
    strb w2, [x0], #1

    // write(1, outbuf, len)
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x0, x1
    mov x0, #1
    mov x8, #64                 // write
    svc #0

    mov x0, #0
    mov x8, #93                 // exit
    svc #0

// append_num: x0 = value, x1 = output pointer; returns new pointer in x0
append_num:
    adrp x4, tmpbuf
    add x4, x4, :lo12:tmpbuf
    mov x2, #10
    mov x3, #0                  // digit count
1:
    udiv x5, x0, x2
    msub x6, x5, x2, x0         // remainder
    add w6, w6, #48             // '0'
    strb w6, [x4, x3]
    add x3, x3, #1
    mov x0, x5
    cbnz x0, 1b
2:
    cbz x3, 3f
    sub x3, x3, #1
    ldrb w6, [x4, x3]           // digits come out reversed
    strb w6, [x1], #1
    b 2b
3:
    mov x0, x1
    ret

.bss
.align 4
inbuf:
    .zero 65536
outbuf:
    .zero 32
tmpbuf:
    .zero 16
