.globl _start

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072

.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz input_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_loop

input_done:
    lea rbx, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r8, r8
    xor r9, r9
scan:
    cmp r8, r12
    jae final_line
    cmp byte ptr [rbx + r8], 10
    je line_end
    inc r8
    jmp scan
line_end:
    call reverse_copy
    inc r8
    mov r9, r8
    jmp scan

final_line:
    cmp r9, r12
    jae write_out
    mov r8, r12
    call reverse_copy

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

reverse_copy:
    mov rcx, r8
    sub rcx, r9
    xor r10, r10
1:
    cmp r10, rcx
    jae 2f
    mov rax, r8
    dec rax
    sub rax, r10
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al
    inc rdi
    inc r10
    jmp 1b
2:
    mov byte ptr [rdi], 10
    inc rdi
    ret
