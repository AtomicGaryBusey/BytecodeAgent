// hexdump -C style dump of stdin, AArch64 Linux, no libc
// syscalls: read=63, write=64, exit=93

    .globl _start

    .section .rodata
hexdigits:
    .ascii "0123456789abcdef"

    .text
_start:
    // sp points at argc; we ignore arguments entirely
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf      // x19 = input buffer base
    mov  x20, #0                    // x20 = total bytes read

// ---- read all of stdin into inbuf ----
read_loop:
    mov  x2, #65536
    sub  x2, x2, x20                // space remaining
    cbz  x2, read_done              // buffer full
    mov  x0, #0                     // fd 0
    add  x1, x19, x20               // write position
    mov  x8, #63                    // read
    svc  #0
    cbz  x0, read_done              // EOF
    add  x20, x20, x0
    b    read_loop
read_done:

    adrp x23, outbuf
    add  x23, x23, :lo12:outbuf     // x23 = output buffer base
    mov  x24, x23                   // x24 = output write pointer
    adrp x5, hexdigits
    add  x5, x5, :lo12:hexdigits    // x5 = hex digit table
    mov  x22, #0                    // x22 = current offset

// ---- per-row loop ----
row_loop:
    cmp  x22, x20
    b.hs rows_done
    sub  x25, x20, x22              // bytes left
    cmp  x25, #16
    b.lo 1f
    mov  x25, #16
1:                                  // x25 = n for this row (1..16)

    // offset: 8 lowercase hex digits, zero padded
    mov  w6, #28
off_loop:
    lsr  w7, w22, w6
    and  w7, w7, #15
    ldrb w8, [x5, x7]
    strb w8, [x24], #1
    subs w6, w6, #4
    b.ge off_loop
    mov  w8, #' '
    strb w8, [x24], #1
    strb w8, [x24], #1              // two spaces after offset

    // 16 hex byte slots
    add  x9, x19, x22               // row data pointer
    mov  x26, #0                    // i
hex_loop:
    cmp  x26, #16
    b.hs hex_done
    cmp  x26, x25
    b.hs hex_pad
    ldrb w7, [x9, x26]
    lsr  w8, w7, #4
    ldrb w8, [x5, x8]
    strb w8, [x24], #1
    and  w8, w7, #15
    ldrb w8, [x5, x8]
    strb w8, [x24], #1
    mov  w8, #' '
    strb w8, [x24], #1
    b    hex_next
hex_pad:
    mov  w8, #' '
    strb w8, [x24], #1
    strb w8, [x24], #1
    strb w8, [x24], #1
hex_next:
    cmp  x26, #7
    b.ne 2f
    mov  w8, #' '                   // extra space after slot 7
    strb w8, [x24], #1
2:
    add  x26, x26, #1
    b    hex_loop
hex_done:

    // " |" then ASCII column
    mov  w8, #' '
    strb w8, [x24], #1
    mov  w8, #'|'
    strb w8, [x24], #1
    mov  x26, #0
asc_loop:
    cmp  x26, x25
    b.hs asc_done
    ldrb w7, [x9, x26]
    cmp  w7, #0x20
    b.lo 3f
    cmp  w7, #0x7e
    b.ls 4f
3:
    mov  w7, #'.'
4:
    strb w7, [x24], #1
    add  x26, x26, #1
    b    asc_loop
asc_done:
    mov  w8, #'|'
    strb w8, [x24], #1
    mov  w8, #'\n'
    strb w8, [x24], #1

    add  x22, x22, x25
    b    row_loop
rows_done:

    // final line: total length as 8 hex digits + newline
    mov  w6, #28
fin_loop:
    lsr  w7, w20, w6
    and  w7, w7, #15
    ldrb w8, [x5, x7]
    strb w8, [x24], #1
    subs w6, w6, #4
    b.ge fin_loop
    mov  w8, #'\n'
    strb w8, [x24], #1

    // ---- write output ----
    mov  x1, x23                    // buffer
    sub  x2, x24, x23               // length
write_loop:
    cbz  x2, do_exit
    mov  x0, #1                     // fd 1
    mov  x8, #64                    // write
    svc  #0
    add  x1, x1, x0
    sub  x2, x2, x0
    b    write_loop

do_exit:
    mov  x0, #0
    mov  x8, #93                    // exit
    svc  #0

    .bss
    .balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 323600                    // 4096 rows * 79 bytes + final 9 bytes + slack
