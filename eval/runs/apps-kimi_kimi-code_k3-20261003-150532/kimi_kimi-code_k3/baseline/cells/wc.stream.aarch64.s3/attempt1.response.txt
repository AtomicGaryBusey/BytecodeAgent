.globl _start

.section .rodata
// (no read-only data needed)

.text
_start:
    mov x19, 0              // lines
    mov x20, 0              // words
    mov x21, 0              // bytes
    mov x22, 0              // in_word flag

read_loop:
    mov x0, 0               // fd = stdin
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, 65536
    mov x8, 63              // read
    svc #0
    cbz x0, done_reading
    add x21, x21, x0        // bytes += n

    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, x0              // remaining bytes in chunk

proc_loop:
    ldrb w3, [x1], #1
    cmp w3, 0x20            // space?
    b.eq is_ws
    sub w4, w3, 9
    cmp w4, 4               // 0x09..0x0D (tab, LF, VT, FF, CR)?
    b.ls is_ws
    // non-whitespace byte
    cbnz x22, 1f
    add x20, x20, 1         // start of new word
    mov x22, 1
1:
    b next_byte
is_ws:
    mov x22, 0
    cmp w3, 0x0A
    b.ne next_byte
    add x19, x19, 1         // newline
next_byte:
    subs x2, x2, 1
    b.ne proc_loop
    b read_loop

done_reading:
    adrp x5, outbuf
    add x5, x5, :lo12:outbuf
    mov x1, x5
    mov x0, x19
    bl utoa
    mov w12, 0x20
    strb w12, [x1], #1
    mov x0, x20
    bl utoa
    mov w12, 0x20
    strb w12, [x1], #1
    mov x0, x21
    bl utoa
    mov w12, 0x0A
    strb w12, [x1], #1

    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    sub x2, x1, x2          // length
    mov x0, 1               // fd = stdout
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x8, 64              // write
    svc #0

    mov x0, 0
    mov x8, 93              // exit
    svc #0

// utoa: x0 = value, x1 = output pointer; returns advanced x1
utoa:
    adrp x9, tmpbuf
    add x9, x9, :lo12:tmpbuf
    add x9, x9, 20          // write digits backwards from end
    mov x10, 10
2:
    udiv x11, x0, x10
    msub x12, x11, x10, x0  // remainder
    add x12, x12, 48        // '0'
    strb w12, [x9, #-1]!
    mov x0, x11
    cbnz x0, 2b
    adrp x13, tmpbuf
    add x13, x13, :lo12:tmpbuf
    add x13, x13, 20
    sub x14, x13, x9        // digit count
3:
    ldrb w12, [x9], #1
    strb w12, [x1], #1
    subs x14, x14, 1
    b.ne 3b
    ret

.bss
.align 4
buf:
    .zero 65536
tmpbuf:
    .zero 32
outbuf:
    .zero 64
