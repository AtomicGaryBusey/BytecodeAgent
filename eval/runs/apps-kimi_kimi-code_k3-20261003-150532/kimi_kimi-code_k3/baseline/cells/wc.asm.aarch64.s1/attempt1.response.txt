```asm
// wc: count lines, words and bytes of standard input (AArch64, no libc)

.bss
.align 4
inbuf:  .zero 65536
outbuf: .zero 64
tmpbuf: .zero 32

.text
.globl _start

// putnum: append decimal representation of x0 to buffer at x1
// clobbers x0, x2, x3, x4, x5; returns advanced x1
putnum:
    adrp x2, tmpbuf
    add x2, x2, :lo12:tmpbuf
    add x2, x2, #24
    cbnz x0, 1f
    mov w4, #'0'
    strb w4, [x1], #1
    ret
1:
    cbz x0, 2f
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #'0'
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x4
    b 1b
2:
    adrp x3, tmpbuf
    add x3, x3, :lo12:tmpbuf
    add x3, x3, #24
3:
    cmp x2, x3
    b.hs 4f
    ldrb w4, [x2], #1
    strb w4, [x1], #1
    b 3b
4:
    ret

_start:
    // ---- read all of stdin into inbuf ----
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
1:
    mov x2, #65536
    sub x2, x2, x20             // space left
    cbz x2, 2f
    mov x0, #0                  // fd 0
    add x1, x19, x20            // buf + offset
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:
    // ---- count ----
    mov x21, #0                 // lines
    mov x22, #0                 // words
    mov w23, #0                 // inside-word flag
    mov x9, #0                  // index
3:
    cmp x9, x20
    b.hs 4f
    ldrb w10, [x19, x9]
    add x9, x9, #1
    cmp w10, #0x0A
    b.ne 5f
    add x21, x21, #1
5:
    cmp w10, #0x20              // space
    b.eq 6f
    sub w11, w10, #9
    cmp w11, #4                 // 0x09..0x0D
    b.ls 6f
    // word byte
    cbnz w23, 3b
    mov w23, #1
    add x22, x22, #1
    b 3b
6:
    // whitespace byte
    mov w23, #0
    b 3b
4:
    // ---- format output ----
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x0, x21
    bl putnum
    mov w2, #' '
    strb w2, [x1], #1
    mov x0, x22
    bl putnum
    mov w2, #' '
    strb w2, [x1], #1
    mov x0, x20
    bl putnum
    mov w2, #0x0A
    strb w2, [x1], #1
    // ---- write to stdout ----
    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    sub x2, x1, x2              // length
    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x8, #64                 // write
    svc #0
    // ---- exit(0) ----
    mov x0, #0
    mov x8, #93
    svc #0
```
