.globl _start

.text
_start:
    xor ebx, ebx

1:
    mov edx, 65536
    sub rdx, rbx
    jz 2f
    lea rsi, [rip + inputbuf]
    add rsi, rbx
    jmp 3f

2:
    lea rsi, [rip + discardbuf]
    mov edx, 1

3:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz 4f
    cmp rbx, 65536
    jae 1b
    add rbx, rax
    jmp 1b

4:
    mov r13, rbx
    xor r14d, r14d
    lea r9, [rip + inputbuf]

line_loop:
    cmp r14, r13
    jae exit_success

    mov r15, r14
    mov rbx, r14

1:
    cmp rbx, r13
    jae 2f
    cmp byte ptr [r9 + rbx], 10
    je 2f
    inc rbx
    jmp 1b

2:
    mov rbp, rbx
    cmp rbx, r13
    jae 3f
    lea r14, [rbx + 1]
    jmp 4f

3:
    mov r14, r13

4:
    lea r10, [rip + evalstack]
    xor r12d, r12d
    mov rbx, r15

token_loop:
    cmp rbx, rbp
    jae line_done
    movzx eax, byte ptr [r9 + rbx]
    cmp al, 32
    je 1f
    cmp al, 9
    jne 2f

1:
    inc rbx
    jmp token_loop

2:
    lea rdi, [r9 + rbx]

3:
    cmp rbx, rbp
    jae 4f
    movzx eax, byte ptr [r9 + rbx]
    cmp al, 32
    je 4f
    cmp al, 9
    je 4f
    inc rbx
    jmp 3b

4:
    mov rdx, rbx
    sub rdx, rdi
    cmp rdx, 1
    jne parse_literal

    movzx eax, byte ptr [rdi]
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod

parse_literal:
    mov rsi, rdi
    mov rcx, rdx
    xor r8d, r8d
    cmp byte ptr [rsi], 45
    jne 1f
    mov r8d, 1
    inc rsi
    dec rcx

1:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error

    xor eax, eax

2:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz 2b

    test r8d, r8d
    jz 3f
    neg rax

3:
    cmp r12, 512
    jae line_error
    mov qword ptr [r10 + r12*8], rax
    inc r12
    jmp token_loop

op_add:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r10 + r12*8 - 16]
    mov rdx, qword ptr [r10 + r12*8 - 8]
    add rax, rdx
    dec r12
    mov qword ptr [r10 + r12*8 - 8], rax
    jmp token_loop

op_sub:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r10 + r12*8 - 16]
    mov rdx, qword ptr [r10 + r12*8 - 8]
    sub rax, rdx
    dec r12
    mov qword ptr [r10 + r12*8 - 8], rax
    jmp token_loop

op_mul:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r10 + r12*8 - 16]
    mov rdx, qword ptr [r10 + r12*8 - 8]
    imul rax, rdx
    dec r12
    mov qword ptr [r10 + r12*8 - 8], rax
    jmp token_loop

op_div:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r10 + r12*8 - 16]
    mov rdx, qword ptr [r10 + r12*8 - 8]
    test rdx, rdx
    jz line_error
    cmp rdx, -1
    jne 1f
    mov r8, 0x8000000000000000
    cmp rax, r8
    je line_error

1:
    mov r8, rdx
    cqo
    idiv r8
    dec r12
    mov qword ptr [r10 + r12*8 - 8], rax
    jmp token_loop

op_mod:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r10 + r12*8 - 16]
    mov rdx, qword ptr [r10 + r12*8 - 8]
    test rdx, rdx
    jz line_error
    cmp rdx, -1
    jne 1f
    mov r8, 0x8000000000000000
    cmp rax, r8
    je line_error

1:
    mov r8, rdx
    cqo
    idiv r8
    dec r12
    mov qword ptr [r10 + r12*8 - 8], rdx
    jmp token_loop

line_done:
    cmp r12, 1
    jne line_error
    mov rax, qword ptr [r10]
    call print_number
    jmp line_loop

line_error:
    call print_error
    jmp line_loop

print_number:
    lea rsi, [rip + numbuf + 31]
    mov byte ptr [rsi], 10
    xor r8d, r8d
    test rax, rax
    jns 1f
    mov r8d, 1
    neg rax

1:
    mov ecx, 10

2:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 2b

    test r8d, r8d
    jz 3f
    dec rsi
    mov byte ptr [rsi], 45

3:
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall
    ret

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
error_message:
    .ascii "error\n"

.bss
.balign 16
inputbuf:
    .zero 65536
discardbuf:
    .zero 1
.balign 8
evalstack:
    .zero 4096
numbuf:
    .zero 32
