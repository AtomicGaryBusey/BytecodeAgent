.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_MAX, 65536
.equ OUT_MAX, 131072

.globl _start

.text
_start:
    xor r13, r13
    lea r12, [rip + inbuf]
1:
    mov rdx, IN_MAX
    sub rdx, r13
    jz 2f
    lea rsi, [r12 + r13]
    xor edi, edi
    mov eax, SYS_read
    syscall
    test rax, rax
    jz 2f
    add r13, rax
    jmp 1b
2:
    mov rbx, r12
    mov r15, r12
    lea rbp, [r12 + r13]
    lea r14, [rip + outbuf]
scan_loop:
    cmp r15, rbp
    je final_line
    mov al, byte ptr [r15]
    cmp al, 10
    jne advance
    mov rcx, r15
    call emit_line
    inc r15
    mov rbx, r15
    jmp scan_loop
advance:
    inc r15
    jmp scan_loop
final_line:
    cmp rbx, rbp
    je output
    mov rcx, rbp
    call emit_line
output:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    jz exit0
    mov edi, 1
    mov eax, SYS_write
    syscall
exit0:
    xor edi, edi
    mov eax, SYS_exit
    syscall

emit_line:
    mov r8, rcx
    sub r8, rbx
    jz 2f
    mov r9, r8
1:
    mov dl, byte ptr [rbx + r9 - 1]
    mov byte ptr [r14], dl
    inc r14
    dec r9
    jnz 1b
2:
    mov byte ptr [r14], 10
    inc r14
    ret

.bss
.align 16
inbuf:
    .zero IN_MAX
.align 16
outbuf:
    .zero OUT_MAX
