```asm
.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 49152
outbuf:
    .zero 66560

.text
_start:
    # read all of stdin into inbuf; rbx = total bytes read
    xor ebx, ebx
1:
    mov rdx, 49152
    sub rdx, rbx
    jz 2f
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    # rsi = source, rdi = dest, rcx = alphabet, r8d = column, r9 = remaining
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rcx, [rip + alphabet]
    xor r8d, r8d
    mov r9, rbx

main_loop:
    cmp r9, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 63
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    sub r9, 3
    add r8d, 4
    cmp r8d, 76
    jne main_loop
    mov byte ptr [rdi], 10
    add rdi, 1
    xor r8d, r8d
    jmp main_loop

tail:
    cmp r9, 1
    je tail1
    cmp r9, 2
    je tail2
    # no tail: terminate last partial line if any
    test r8d, r8d
    jz write_out
    mov byte ptr [rdi], 10
    add rdi, 1
    jmp write_out

tail1:
    movzx eax, byte ptr [rsi]
    mov edx, eax
    shr edx, 2
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    and edx, 3
    shl edx, 4
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    add rdi, 1
    jmp write_out

tail2:
    movzx eax, byte ptr [rsi]
    movzx r10d, byte ptr [rsi + 1]
    mov edx, eax
    shr edx, 2
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov eax, r10d
    shr eax, 4
    or edx, eax
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, r10d
    and edx, 15
    shl edx, 2
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    add rdi, 1
    jmp write_out

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
3:
    test rdx, rdx
    jz 4f
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle 4f
    add rsi, rax
    sub rdx, rax
    jmp 3b
4:
    mov eax, 60
    xor edi, edi
    syscall
```
