.globl _start

.text

_start:
    xor r12d, r12d
    lea r13, [rip + inbuf]
read_loop:
    mov rdx, 65536
    sub rdx, r12
    lea rsi, [r13 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
1:
    cmp rbx, r12
    jae 2f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
    ret
3:
    inc rbx
    jmp 1b
2:
    ret

fail:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + errpfx]
    mov ecx, 9
1:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec ecx
    jne 1b
    mov rax, rbx
    mov r8d, 10
    lea r9, [rip + numtmp + 24]
2:
    xor edx, edx
    div r8
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jne 2b
    lea rcx, [rip + numtmp + 24]
    sub rcx, r9
3:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    dec cl
    jne 3b
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

parse_value:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je lit_true
    cmp al, 'f'
    je lit_false
    cmp al, 'n'
    je lit_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe parse_number
    jmp fail

parse_string:
    inc rbx
1:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je 2f
    cmp al, 0x5C
    je 3f
    cmp al, 0x20
    jb fail
    inc rbx
    jmp 1b
2:
    inc rbx
    ret
3:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, '/'
    je 4f
    cmp al, 'b'
    je 4f
    cmp al, 'f'
    je 4f
    cmp al, 'n'
    je 4f
    cmp al, 'r'
    je 4f
    cmp al, 't'
    je 4f
    cmp al, 'u'
    je 5f
    jmp fail
4:
    inc rbx
    jmp 1b
5:
    inc rbx
    mov ecx, 4
6:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe 7f
    or al, 0x20
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
7:
    inc rbx
    dec cl
    jne 6b
    jmp 1b

parse_number:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '-'
    jne 1f
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
1:
    cmp al, '0'
    je 2f
    cmp al, '1'
    jb fail
    cmp al, '9'
    ja fail
3:
    inc rbx
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb 5f
    cmp al, '9'
    jbe 3b
    jmp 5f
2:
    inc rbx
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
5:
    cmp al, '.'
    jne 6f
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
7:
    inc rbx
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb 6f
    cmp al, '9'
    jbe 7b
6:
    cmp al, 'e'
    je 8f
    cmp al, 'E'
    jne 9f
8:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '+'
    je 10f
    cmp al, '-'
    je 10f
    jmp 11f
10:
    inc rbx
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
11:
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
12:
    inc rbx
    cmp rbx, r12
    jae 9f
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb 9f
    cmp al, '9'
    jbe 12b
9:
    ret

lit_true:
    lea rsi, [rip + s_true]
    mov ecx, 4
    jmp lit_check
lit_false:
    lea rsi, [rip + s_false]
    mov ecx, 5
    jmp lit_check
lit_null:
    lea rsi, [rip + s_null]
    mov ecx, 4
lit_check:
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, byte ptr [rsi]
    jne fail
    inc rbx
    inc rsi
    dec cl
    jne lit_check
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '}'
    je obj_close
obj_member:
    cmp al, '"'
    jne fail
    call parse_string
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ':'
    jne fail
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ','
    je obj_next
    cmp al, '}'
    je obj_close
    jmp fail
obj_next:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    jmp obj_member
obj_close:
    inc rbx
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ']'
    je arr_close
arr_elem:
    call parse_value
    call skip_ws
    cmp rbx, r12
    jae fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ','
    je arr_next
    cmp al, ']'
    je arr_close
    jmp fail
arr_next:
    inc rbx
    call skip_ws
    jmp arr_elem
arr_close:
    inc rbx
    ret

.section .rodata
okmsg:
    .ascii "ok\n"
errpfx:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 64
numtmp:
    .zero 32
