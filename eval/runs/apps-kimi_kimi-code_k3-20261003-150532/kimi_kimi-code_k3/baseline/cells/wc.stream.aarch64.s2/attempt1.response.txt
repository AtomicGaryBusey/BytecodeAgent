// wc: count lines, words, bytes of stdin - static AArch64, no libc
        .text
        .globl _start

_start:
        // ---- read all of stdin into inbuf ----
        adrp    x20, inbuf
        add     x20, x20, :lo12:inbuf // x20 = buffer base
        mov     x21, 0                  // x21 = total bytes read
1:
        mov     x2, 65536
        sub     x2, x2, x21             // space remaining
        cbz     x2, 2f                  // buffer full -> stop
        mov     x0, 0                   // fd = stdin
        add     x1, x20, x21            // dst
        mov     x8, 63                  // read
        svc     #0
        cbz     x0, 2f                  // EOF
        add     x21, x21, x0
        b       1b
2:
        // ---- count ----
        mov     x22, 0                  // lines
        mov     x23, 0                  // words
        mov     x24, 0                  // in_word flag
        mov     x9, 0                   // index
1:
        cmp     x9, x21
        b.hs    2f
        ldrb    w10, [x20, x9]
        cmp     w10, 0x0A               // newline?
        b.ne    3f
        add     x22, x22, 1
3:
        // whitespace = 0x20 or 0x09..0x0D
        cmp     w10, 0x20
        b.eq    4f
        sub     w11, w10, 0x09
        cmp     w11, 4
        b.ls    4f
        // word byte
        cbnz    x24, 5f                 // already inside a word
        mov     x24, 1
        add     x23, x23, 1
        b       5f
4:
        mov     x24, 0                  // whitespace ends word
5:
        add     x9, x9, 1
        b       1b
2:
        // ---- build output: "<lines> <words> <bytes>\n" ----
        adrp    x25, outbuf
        add     x25, x25, :lo12:outbuf
        mov     x26, 0                  // output offset

        mov     x0, x22
        add     x1, x25, x26
        bl      itoa
        add     x26, x26, x2
        mov     w7, ' '
        strb    w7, [x25, x26]
        add     x26, x26, 1

        mov     x0, x23
        add     x1, x25, x26
        bl      itoa
        add     x26, x26, x2
        mov     w7, ' '
        strb    w7, [x25, x26]
        add     x26, x26, 1

        mov     x0, x21
        add     x1, x25, x26
        bl      itoa
        add     x26, x26, x2
        mov     w7, '\n'
        strb    w7, [x25, x26]
        add     x26, x26, 1

        // ---- write(1, outbuf, x26) ----
        mov     x0, 1
        mov     x1, x25
        mov     x2, x26
        mov     x8, 64
        svc     #0

        // ---- exit(0) ----
        mov     x0, 0
        mov     x8, 93
        svc     #0

// itoa: x0 = unsigned value, x1 = destination
// returns x2 = number of bytes written
itoa:
        stp     x29, x30, [sp, #-48]!
        mov     x29, sp
        add     x4, sp, 48              // x4 = one past end of temp (at [sp+16..sp+48))
        mov     x3, 10
1:
        udiv    x5, x0, x3              // quotient
        msub    x6, x5, x3, x0          // remainder
        add     x6, x6, '0'
        sub     x4, x4, 1
        strb    w6, [x4]
        mov     x0, x5
        cbnz    x0, 1b
        add     x2, sp, 48
        sub     x2, x2, x4              // length
        mov     x5, 0
2:
        cmp     x5, x2
        b.hs    3f
        ldrb    w6, [x4, x5]
        strb    w6, [x1, x5]
        add     x5, x5, 1
        b       2b
3:
        ldp     x29, x30, [sp], #48
        ret

        .bss
        .align 4
inbuf:
        .zero   65536
outbuf:
        .zero   64
