.globl _start

.text
_start:
    # ---- read all of stdin into buffer ----
    xor ebx, ebx                # rbx = total bytes read
1:
    mov edx, 32768
    sub rdx, rbx
    jz 2f
    lea rsi, [rip + buffer]
    add rsi, rbx
    xor edi, edi                # fd 0
    xor eax, eax                # read
    syscall
    test rax, rax
    jz 2f
    add rbx, rax
    jmp 1b
2:
    # ---- split buffer into lines ----
    lea r8, [rip + buffer]
    xor r9d, r9d                # r9 = pos
    xor r10d, r10d              # r10 = number of lines
3:
    cmp r9, rbx
    jae 5f
    mov r11, r9                 # r11 = line start
4:
    cmp r9, rbx
    jae 6f
    cmp byte ptr [r8 + r9], 10
    je 6f
    inc r9
    jmp 4b
6:
    lea rax, [rip + line_off]
    mov [rax + r10*8], r11
    mov rax, r9
    sub rax, r11
    lea rcx, [rip + line_len]
    mov [rcx + r10*8], rax
    inc r10
    cmp r9, rbx
    jae 3b
    inc r9                      # skip the newline
    jmp 3b
5:
    # ---- insertion sort of (off,len) pairs ----
    mov r12, r10                # r12 = nlines
    mov r13d, 1                 # r13 = i
7:
    cmp r13, r12
    jae 12f
    lea rax, [rip + line_off]
    mov r14, [rax + r13*8]      # r14 = key_off
    lea rax, [rip + line_len]
    mov r15, [rax + r13*8]      # r15 = key_len
    lea rbp, [r13 - 1]          # rbp = j
8:
    test rbp, rbp
    js 11f                      # j < 0: stop
    lea rax, [rip + line_off]
    mov rdi, [rax + rbp*8]
    lea rax, [rip + line_len]
    mov rsi, [rax + rbp*8]      # line[j] = (rdi, rsi)
    lea rdx, [rip + buffer]
    lea rcx, [rdx + r14]        # rcx = key ptr
    add rdi, rdx                # rdi = line[j] ptr
    xor r9d, r9d
    mov r10, rsi
    cmp r15, r10
    cmovb r10, r15              # r10 = min(len1, len2)
9:
    cmp r9, r10
    jae 10f
    mov al, byte ptr [rdi + r9]
    cmp al, byte ptr [rcx + r9]
    ja 13f                      # line[j] > key: shift
    jb 11f                      # line[j] < key: stop
    inc r9
    jmp 9b
10:
    cmp rsi, r15                # prefix case: longer one is bigger
    ja 13f
    jmp 11f
13:
    lea rax, [rip + line_off]
    mov rdx, [rax + rbp*8]
    mov [rax + rbp*8 + 8], rdx
    lea rax, [rip + line_len]
    mov rdx, [rax + rbp*8]
    mov [rax + rbp*8 + 8], rdx
    dec rbp
    jmp 8b
11:
    lea rax, [rip + line_off]
    mov [rax + rbp*8 + 8], r14
    lea rax, [rip + line_len]
    mov [rax + rbp*8 + 8], r15
    inc r13
    jmp 7b
12:
    # ---- print sorted lines ----
    xor r13d, r13d
14:
    cmp r13, r12
    jae 15f
    lea rax, [rip + line_off]
    mov rsi, [rax + r13*8]
    lea rdx, [rip + line_len]
    mov rdx, [rdx + r13*8]
    lea rax, [rip + buffer]
    add rsi, rax
    mov edi, 1
    mov eax, 1                  # write
    syscall
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    mov eax, 1
    syscall
    inc r13
    jmp 14b
15:
    xor edi, edi
    mov eax, 60                 # exit(0)
    syscall

.section .rodata
nl:
    .byte 10

.bss
.align 8
buffer:
    .zero 32768
line_off:
    .zero 8000
line_len:
    .zero 8000
