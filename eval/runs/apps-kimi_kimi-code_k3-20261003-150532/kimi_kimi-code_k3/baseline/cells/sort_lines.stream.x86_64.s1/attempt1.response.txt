.section .rodata
newline: .byte 10

.bss
.align 8
inbuf: .zero 32768
line_off: .zero 4000
line_len: .zero 4000

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    test rdx, rdx
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + inbuf]
parse_loop:
    cmp r15, r12
    jae parse_tail
    mov al, byte ptr [rbx + r15]
    cmp al, 10
    jne parse_next
    mov rax, r15
    sub rax, r14
    lea rcx, [rip + line_off]
    mov [rcx + r13*4], r14d
    lea rcx, [rip + line_len]
    mov [rcx + r13*4], eax
    inc r13d
    lea r14, [r15 + 1]
parse_next:
    inc r15
    jmp parse_loop
parse_tail:
    cmp r14, r12
    jae sort_begin
    mov rax, r12
    sub rax, r14
    lea rcx, [rip + line_off]
    mov [rcx + r13*4], r14d
    lea rcx, [rip + line_len]
    mov [rcx + r13*4], eax
    inc r13d

sort_begin:
    mov r14d, 1
outer:
    cmp r14d, r13d
    jae output
    lea rcx, [rip + line_off]
    mov r15d, [rcx + r14*4]
    lea rcx, [rip + line_len]
    mov ebx, [rcx + r14*4]
    lea ebp, [r14 - 1]
inner:
    test ebp, ebp
    js place
    lea rcx, [rip + line_off]
    mov edi, [rcx + rbp*4]
    lea rcx, [rip + line_len]
    mov esi, [rcx + rbp*4]
    mov edx, r15d
    mov ecx, ebx
    call cmp_lines
    test eax, eax
    jle place
    lea rcx, [rip + line_off]
    mov eax, [rcx + rbp*4]
    mov [rcx + rbp*4 + 4], eax
    lea rcx, [rip + line_len]
    mov eax, [rcx + rbp*4]
    mov [rcx + rbp*4 + 4], eax
    dec ebp
    jmp inner
place:
    lea rcx, [rip + line_off]
    mov [rcx + rbp*4 + 4], r15d
    lea rcx, [rip + line_len]
    mov [rcx + rbp*4 + 4], ebx
    inc r14d
    jmp outer

output:
    xor r14d, r14d
out_loop:
    cmp r14d, r13d
    jae exit0
    lea rcx, [rip + line_off]
    mov esi, [rcx + r14*4]
    lea rax, [rip + inbuf]
    add rsi, rax
    lea rcx, [rip + line_len]
    mov edx, [rcx + r14*4]
    call write_line
    inc r14d
    jmp out_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall

cmp_lines:
    lea r8, [rip + inbuf]
    add r8, rdi
    lea r9, [rip + inbuf]
    add r9, rdx
    mov r10d, esi
    cmp r10d, ecx
    cmova r10d, ecx
    xor r11d, r11d
cmp_loop:
    cmp r11d, r10d
    jae cmp_len
    mov al, byte ptr [r8 + r11]
    cmp al, byte ptr [r9 + r11]
    ja cmp_gt
    jb cmp_lt
    inc r11d
    jmp cmp_loop
cmp_gt:
    mov eax, 1
    ret
cmp_lt:
    mov eax, -1
    ret
cmp_len:
    cmp esi, ecx
    ja cmp_gt
    jb cmp_lt
    xor eax, eax
    ret

write_line:
    test rdx, rdx
    jz wl_nl
wl_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wl_nl
    add rsi, rax
    sub rdx, rax
    jnz wl_loop
wl_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    ret
