.text
.globl _start

_start:
    // read all of stdin into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf   // x19 = buffer base
    mov x20, #0                 // x20 = total bytes read
    mov x21, #65536             // capacity
read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x22, #0                 // x22 = current row offset
row_loop:
    cmp x22, x20
    b.hs final_line
    // n = min(16, total - offset)
    sub x25, x20, x22
    cmp x25, #16
    b.lo n_ok
    mov x25, #16
n_ok:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf  // x23 = outbuf base
    mov x24, x23                // x24 = write pointer
    adrp x26, hexdig
    add x26, x26, :lo12:hexdig
    // 8 hex digits of offset
    mov w27, w22
    mov x28, #28
off_hex:
    lsr w0, w27, w28
    and w0, w0, #15
    ldrb w0, [x26, x0]
    strb w0, [x24], #1
    subs x28, x28, #4
    b.ge off_hex
    // two spaces
    mov w0, #' '
    strb w0, [x24], #1
    strb w0, [x24], #1
    // 16 byte slots
    mov x29, #0
slot_loop:
    cmp x29, #16
    b.hs slots_done
    cmp x29, x25
    b.hs pad_slot
    add x2, x22, x29
    ldrb w0, [x19, x2]
    lsr w1, w0, #4
    ldrb w1, [x26, x1]
    and w2, w0, #15
    ldrb w2, [x26, x2]
    strb w1, [x24], #1
    strb w2, [x24], #1
    mov w0, #' '
    strb w0, [x24], #1
    b slot_sep
pad_slot:
    mov w0, #' '
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1
slot_sep:
    cmp x29, #7
    b.ne slot_inc
    mov w0, #' '
    strb w0, [x24], #1
slot_inc:
    add x29, x29, #1
    b slot_loop
slots_done:
    // " |"
    mov w0, #' '
    strb w0, [x24], #1
    mov w0, #'|'
    strb w0, [x24], #1
    // ascii column, n bytes
    mov x29, #0
ascii_loop:
    cmp x29, x25
    b.hs ascii_done
    add x2, x22, x29
    ldrb w0, [x19, x2]
    cmp w0, #0x20
    b.lo dot
    cmp w0, #0x7e
    b.ls store_ch
dot:
    mov w0, #'.'
store_ch:
    strb w0, [x24], #1
    add x29, x29, #1
    b ascii_loop
ascii_done:
    mov w0, #'|'
    strb w0, [x24], #1
    mov w0, #'\n'
    strb w0, [x24], #1
    // write the row
    sub x2, x24, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0
    add x22, x22, #16
    b row_loop

final_line:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    adrp x26, hexdig
    add x26, x26, :lo12:hexdig
    mov w27, w20
    mov x28, #28
fin_hex:
    lsr w0, w27, w28
    and w0, w0, #15
    ldrb w0, [x26, x0]
    strb w0, [x24], #1
    subs x28, x28, #4
    b.ge fin_hex
    mov w0, #'\n'
    strb w0, [x24], #1
    sub x2, x24, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 128
