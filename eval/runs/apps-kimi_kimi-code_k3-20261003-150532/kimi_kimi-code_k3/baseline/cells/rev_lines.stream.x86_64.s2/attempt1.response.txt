.globl _start

.text
_start:
    xor r12d, r12d
    lea r13, [rip + inbuf]

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor ebx, ebx
    xor r13d, r13d

scan_loop:
    cmp rbx, r12
    jae final_line
    cmp byte ptr [rsi + rbx], 10
    jne advance_char
    mov r8, rbx
    sub r8, r13
    call emit_line
    lea r13, [rbx + 1]
advance_char:
    inc rbx
    jmp scan_loop

final_line:
    cmp r13, r12
    jae prepare_write
    mov r8, r12
    sub r8, r13
    call emit_line

prepare_write:
    lea r14, [rip + outbuf]
    mov r15, rdi
    sub r15, r14

write_loop:
    test r15, r15
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall
    test rax, rax
    jle exit_failure
    add r14, rax
    sub r15, rax
    jmp write_loop

emit_line:
    test r8, r8
    jz emit_newline
    lea r9, [rsi + r13]
    lea r9, [r9 + r8 - 1]
    mov r10, r8
reverse_copy:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    dec r9
    dec r10
    jnz reverse_copy
emit_newline:
    mov byte ptr [rdi], 10
    inc rdi
    ret

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

exit_failure:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
