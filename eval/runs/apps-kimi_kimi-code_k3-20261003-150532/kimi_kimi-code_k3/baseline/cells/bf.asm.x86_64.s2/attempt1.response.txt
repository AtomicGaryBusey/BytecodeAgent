.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60

.section .bss
.align 16
inbuf:  .zero 65537
tape:   .zero 30000
outbuf: .zero 65536
jtab:   .zero 262144
stk:    .zero 4096

.text
_start:
        # read all of stdin into inbuf
        xor     r12, r12
1:
        mov     rdx, 65536
        sub     rdx, r12
        jz      2f
        mov     eax, SYS_read
        xor     edi, edi
        lea     rsi, [rip + inbuf]
        add     rsi, r12
        syscall
        test    rax, rax
        jz      2f
        add     r12, rax
        jmp     1b
2:
        # find first '!' : program = [0,rcx), input = rest
        lea     rsi, [rip + inbuf]
        xor     rcx, rcx
3:
        cmp     rcx, r12
        jae     4f
        cmp     byte ptr [rsi + rcx], 0x21
        je      4f
        inc     rcx
        jmp     3b
4:
        mov     r13, rcx                # program length
        lea     r14, [rsi + rcx]        # input pointer
        cmp     rcx, r12
        jae     5f
        inc     r14                     # skip '!'
5:
        lea     r15, [rsi + r12]        # end of input

        # validate brackets and build jump table
        xor     rcx, rcx
        xor     rdx, rdx
        lea     rdi, [rip + stk]
        lea     r8, [rip + jtab]
6:
        cmp     rcx, r13
        jae     7f
        movzx   eax, byte ptr [rsi + rcx]
        cmp     al, '['
        je      8f
        cmp     al, ']'
        je      9f
        jmp     10f
8:
        mov     dword ptr [rdi + rdx*4], ecx
        inc     rdx
        jmp     10f
9:
        test    rdx, rdx
        jz      badbr
        dec     rdx
        mov     eax, dword ptr [rdi + rdx*4]
        mov     dword ptr [r8 + rax*4], ecx
        mov     dword ptr [r8 + rcx*4], eax
10:
        inc     rcx
        jmp     6b
7:
        test    rdx, rdx
        jnz     badbr

        # interpret
        xor     rbx, rbx                # ip
        xor     r12, r12                # dp
        xor     ebp, ebp                # output count
        lea     rsi, [rip + inbuf]
        lea     rdi, [rip + tape]
        lea     r8, [rip + jtab]
        lea     r9, [rip + outbuf]
loop:
        cmp     rbx, r13
        jae     done
        movzx   eax, byte ptr [rsi + rbx]
        cmp     al, '>'
        je      inc_dp
        cmp     al, '<'
        je      dec_dp
        cmp     al, '+'
        je      inc_cell
        cmp     al, '-'
        je      dec_cell
        cmp     al, '.'
        je      put_cell
        cmp     al, ','
        je      get_cell
        cmp     al, '['
        je      open_br
        cmp     al, ']'
        je      close_br
next:
        inc     rbx
        jmp     loop
inc_dp:
        inc     r12
        jmp     next
dec_dp:
        dec     r12
        jmp     next
inc_cell:
        inc     byte ptr [rdi + r12]
        jmp     next
dec_cell:
        dec     byte ptr [rdi + r12]
        jmp     next
put_cell:
        mov     al, byte ptr [rdi + r12]
        mov     byte ptr [r9 + rbp], al
        inc     rbp
        jmp     next
get_cell:
        xor     eax, eax
        cmp     r14, r15
        jae     1f
        mov     al, byte ptr [r14]
        inc     r14
1:
        mov     byte ptr [rdi + r12], al
        jmp     next
open_br:
        cmp     byte ptr [rdi + r12], 0
        jne     next
        mov     ebx, dword ptr [r8 + rbx*4]
        jmp     next
close_br:
        cmp     byte ptr [rdi + r12], 0
        je      next
        mov     ebx, dword ptr [r8 + rbx*4]
        jmp     next
done:
        # flush output
        lea     rsi, [rip + outbuf]
        mov     rdx, rbp
        test    rdx, rdx
        jz      2f
1:
        mov     eax, SYS_write
        mov     edi, 1
        syscall
        sub     rdx, rax
        jz      2f
        add     rsi, rax
        jmp     1b
2:
        mov     eax, SYS_exit
        xor     edi, edi
        syscall
badbr:
        mov     eax, SYS_exit
        mov     edi, 1
        syscall
