.globl _start

.equ BUFSZ, 65536

.text
_start:
    xor ebx, ebx
.read_loop:
    mov rdx, BUFSZ
    sub rdx, rbx
    jz .done_reading
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, rbx
    syscall
    test rax, rax
    jle .done_reading
    add rbx, rax
    jmp .read_loop
.done_reading:
    lea r12, [rip + buf]
    lea r13, [rip + buf]
    add r13, rbx
    mov r14, r12
.scan:
    cmp r14, r13
    jae .maybe_final
    cmp byte ptr [r14], 10
    je .emit
    inc r14
    jmp .scan
.emit:
    mov rsi, r12
    lea rdi, [r14 - 1]
.rev:
    cmp rsi, rdi
    jae .revdone
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp .rev
.revdone:
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r14
    sub rdx, r12
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    lea r12, [r14 + 1]
    mov r14, r12
    jmp .scan
.maybe_final:
    cmp r12, r13
    jae .exit
    mov r14, r13
    jmp .emit
.exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
nl: .byte 10

.bss
.align 16
buf: .zero 65536
