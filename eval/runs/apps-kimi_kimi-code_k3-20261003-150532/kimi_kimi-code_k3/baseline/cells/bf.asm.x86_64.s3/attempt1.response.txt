.globl _start

.bss
.align 16
inbuf:   .zero 65536
tape:    .zero 30000
outbuf:  .zero 65536
jumptab: .zero 262144
bstack:  .zero 4096

.text
.globl _start
_start:
    # ---- read all of stdin into inbuf ----
    xor ebx, ebx                  # rbx = total bytes read
read_loop:
    mov edx, 65536
    sub rdx, rbx
    jle read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax                  # read
    xor edi, edi                  # fd 0
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:

    # ---- split at first '!' ----
    lea rsi, [rip + inbuf]        # rsi = program base (kept)
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je got_bang
    inc rcx
    jmp find_bang
got_bang:
    mov rbp, rbx
    sub rbp, rcx
    dec rbp                       # rbp = input length
    mov rbx, rcx                  # rbx = program length
    lea r9, [rsi + rcx + 1]       # r9 = input base
    jmp balance
no_bang:
    xor ebp, ebp                  # input length 0
    lea r9, [rip + inbuf]

    # ---- balance check + build jump table ----
balance:
    lea rdi, [rip + jumptab]
    lea r8, [rip + bstack]
    xor r12d, r12d                # i
    xor r13d, r13d                # depth
bal_loop:
    cmp r12, rbx
    jae bal_done
    movzx eax, byte ptr [rsi + r12]
    cmp al, 0x5B                  # '['
    je bal_push
    cmp al, 0x5D                  # ']'
    je bal_pop
bal_next:
    inc r12
    jmp bal_loop
bal_push:
    mov dword ptr [r8 + r13*4], r12d
    inc r13
    jmp bal_next
bal_pop:
    test r13, r13
    jz unbalanced
    dec r13
    mov eax, dword ptr [r8 + r13*4]     # j = matching '[' position
    lea ecx, [r12 + 1]
    mov dword ptr [rdi + rax*4], ecx    # jumptab[j] = i + 1
    inc eax
    mov dword ptr [rdi + r12*4], eax    # jumptab[i] = j + 1
    jmp bal_next
bal_done:
    test r13, r13
    jnz unbalanced

    # ---- interpret ----
    lea rdi, [rip + tape]         # rdi = tape base
    lea r8, [rip + outbuf]        # r8 = output buffer base
    lea r10, [rip + jumptab]      # r10 = jump table base
    xor r12d, r12d                # pc
    xor r13d, r13d                # dp
    xor r14d, r14d                # input index
    xor r15d, r15d                # output index

run_loop:
    cmp r12, rbx
    jae run_done
    movzx eax, byte ptr [rsi + r12]
    cmp al, 0x3E                  # '>'
    je c_gt
    cmp al, 0x3C                  # '<'
    je c_lt
    cmp al, 0x2B                  # '+'
    je c_plus
    cmp al, 0x2D                  # '-'
    je c_minus
    cmp al, 0x2E                  # '.'
    je c_dot
    cmp al, 0x2C                  # ','
    je c_comma
    cmp al, 0x5B                  # '['
    je c_lb
    cmp al, 0x5D                  # ']'
    je c_rb
    inc r12                       # comment byte
    jmp run_loop

c_gt:
    inc r13
    inc r12
    jmp run_loop
c_lt:
    dec r13
    inc r12
    jmp run_loop
c_plus:
    inc byte ptr [rdi + r13]
    inc r12
    jmp run_loop
c_minus:
    dec byte ptr [rdi + r13]
    inc r12
    jmp run_loop
c_dot:
    mov al, byte ptr [rdi + r13]
    mov byte ptr [r8 + r15], al
    inc r15
    cmp r15, 65536
    jb c_dot_done
    # buffer full: flush
    mov eax, 1                    # write
    mov edi, 1                    # fd 1
    mov rsi, r8
    mov edx, 65536
    syscall
    lea rsi, [rip + inbuf]
    lea rdi, [rip + tape]
    xor r15d, r15d
c_dot_done:
    inc r12
    jmp run_loop
c_comma:
    cmp r14, rbp
    jae c_comma_eof
    mov al, byte ptr [r9 + r14]
    inc r14
    jmp c_comma_store
c_comma_eof:
    xor eax, eax
c_comma_store:
    mov byte ptr [rdi + r13], al
    inc r12
    jmp run_loop
c_lb:
    cmp byte ptr [rdi + r13], 0
    jne c_lb_skip
    mov r12d, dword ptr [r10 + r12*4]
    jmp run_loop
c_lb_skip:
    inc r12
    jmp run_loop
c_rb:
    cmp byte ptr [rdi + r13], 0
    je c_rb_skip
    mov r12d, dword ptr [r10 + r12*4]
    jmp run_loop
c_rb_skip:
    inc r12
    jmp run_loop

run_done:
    test r15, r15
    jz exit_ok
    mov eax, 1                    # write
    mov edi, 1                    # fd 1
    mov rsi, r8
    mov rdx, r15
    syscall
exit_ok:
    mov eax, 60                   # exit
    xor edi, edi
    syscall

unbalanced:
    mov eax, 60                   # exit
    mov edi, 1
    syscall
