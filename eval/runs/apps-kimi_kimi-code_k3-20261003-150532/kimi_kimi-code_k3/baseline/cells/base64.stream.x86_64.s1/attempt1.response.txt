.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, 49152
    sub rdx, rbx
    je read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbp, [rip + alphabet]
    xor r9d, r9d
    xor ecx, ecx

main_loop:
    lea rax, [rcx + 3]
    cmp rax, rbx
    ja tail
    movzx eax, byte ptr [rsi + rcx]
    shl eax, 16
    movzx edx, byte ptr [rsi + rcx + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + rcx + 2]
    or eax, edx
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, eax
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    call emit
    add rcx, 3
    jmp main_loop

tail:
    mov rax, rbx
    sub rax, rcx
    test rax, rax
    je finish
    cmp rax, 1
    je one_byte
    movzx eax, byte ptr [rsi + rcx]
    movzx r10d, byte ptr [rsi + rcx + 1]
    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov r8d, r10d
    shr r8d, 4
    or edx, r8d
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, r10d
    and edx, 15
    shl edx, 2
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, 61
    call emit
    jmp finish
one_byte:
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov dl, byte ptr [rbp + rdx]
    call emit
    mov edx, 61
    call emit
    mov edx, 61
    call emit

finish:
    test r9, r9
    je no_newline
    mov byte ptr [rdi], 10
    inc rdi
no_newline:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    je exit0
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit0
    add rsi, rax
    sub rdx, rax
    jne write_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall

emit:
    mov byte ptr [rdi], dl
    inc rdi
    inc r9
    cmp r9, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
1:
    ret

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 66560
