.globl _start

.text
_start:
    xor rbx, rbx
    xor r12, r12
    xor r13, r13
    xor r14d, r14d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add rbx, rax
    lea rsi, [rip + buffer]
    mov rcx, rax
scan:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je is_ws
    lea edx, [rax - 9]
    cmp dl, 4
    jbe is_ws
    test r14d, r14d
    jnz next
    inc r13
    mov r14d, 1
    jmp next
is_ws:
    xor r14d, r14d
    cmp al, 0x0A
    jne next
    inc r12
next:
    inc rsi
    dec rcx
    jnz scan
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call itoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call itoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, rbx
    call itoa
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

itoa:
    lea rsi, [rip + tmpbuf + 32]
    mov r8, rsi
    mov rcx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    cmp rsi, r8
    je 3f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    jmp 2b
3:
    ret

.section .bss
.align 16
buffer:
    .zero 65536
outbuf:
    .zero 64
tmpbuf:
    .zero 32
