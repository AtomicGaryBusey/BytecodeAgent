.globl _start

.text
_start:
    # read all of stdin into inbuf
    xor ebx, ebx                # total bytes read so far
1:
    mov edx, 49152
    sub rdx, rbx                # space left
    jz 2f
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edi, edi                # fd 0
    xor eax, eax                # sys_read
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    # encode rbx bytes from inbuf into outbuf, 76 chars per line
    lea rbp, [rip + alphabet]
    lea rsi, [rip + inbuf]      # input cursor
    mov r9, rbx                 # bytes remaining
    lea rdi, [rip + outbuf]     # output cursor
    xor r8d, r8d                # column counter (0..76)
3:
    cmp r9, 3
    jb 4f
    movbe eax, dword ptr [rsi]  # b0<<24 | b1<<16 | b2<<8 | x
    mov edx, eax
    shr edx, 26
    movzx r10d, byte ptr [rbp + rdx]
    mov edx, eax
    shr edx, 20
    and edx, 63
    movzx r11d, byte ptr [rbp + rdx]
    shl r11d, 8
    or r10d, r11d
    mov edx, eax
    shr edx, 14
    and edx, 63
    movzx r11d, byte ptr [rbp + rdx]
    shl r11d, 16
    or r10d, r11d
    mov edx, eax
    shr edx, 8
    and edx, 63
    movzx r11d, byte ptr [rbp + rdx]
    shl r11d, 24
    or r10d, r11d
    mov dword ptr [rdi], r10d
    add rdi, 4
    add rsi, 3
    sub r9, 3
    add r8d, 4
    cmp r8d, 76
    jne 3b
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp 3b
4:
    # tail: 0, 1 or 2 bytes left
    test r9, r9
    jz 5f
    movbe eax, dword ptr [rsi]
    mov edx, eax
    shr edx, 26
    movzx r10d, byte ptr [rbp + rdx]
    mov edx, eax
    shr edx, 20
    and edx, 63
    movzx r11d, byte ptr [rbp + rdx]
    shl r11d, 8
    or r10d, r11d
    cmp r9, 1
    je tail_one
    # two leftover bytes: one more char then '='
    mov edx, eax
    shr edx, 14
    and edx, 63
    movzx r11d, byte ptr [rbp + rdx]
    shl r11d, 16
    or r10d, r11d
    or r10d, 0x3D000000         # '='
    jmp tail_store
tail_one:
    or r10d, 0x3D3D0000         # '=='
tail_store:
    mov dword ptr [rdi], r10d
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne 5f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
5:
    # final newline if the last line is not yet terminated
    test r8d, r8d
    jz 6f
    mov byte ptr [rdi], 10
    inc rdi
6:
    # write outbuf
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    jz 7f
    mov edi, 1                  # fd 1
    mov eax, 1                  # sys_write
    syscall
7:
    xor edi, edi
    mov eax, 60                 # sys_exit
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 49168
.align 16
outbuf:
    .zero 66560
