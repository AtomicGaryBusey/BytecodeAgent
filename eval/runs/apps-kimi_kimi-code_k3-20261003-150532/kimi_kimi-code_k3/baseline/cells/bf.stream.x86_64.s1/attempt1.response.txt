.globl _start

.text
_start:
    # ---- read all of stdin into inbuf ----
    xor ebx, ebx                # rbx = total bytes read
read_loop:
    mov edx, 65536
    sub edx, ebx                # remaining space
    test edx, edx
    jz read_done
    xor eax, eax                # sys_read
    xor edi, edi                # fd 0
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:

    # ---- split at first '!' ----
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [inbuf + rcx], 0x21
    je has_bang
    inc rcx
    jmp find_bang
no_bang:
    lea rbp, [rip + inbuf]      # input base (unused)
    xor r12d, r12d              # input length = 0
    jmp split_done              # proglen stays rbx (whole input)
has_bang:
    lea rbp, [rip + inbuf + 1]
    add rbp, rcx                # input base = inbuf + bang + 1
    mov r12, rbx
    sub r12, rcx
    dec r12                     # input length
    mov rbx, rcx                # proglen = bytes before '!'
split_done:

    # ---- check brackets, build jump table ----
    xor esi, esi                # pc index
    xor edi, edi                # bracket stack depth
build_loop:
    cmp rsi, rbx
    jae build_done
    movzx eax, byte ptr [inbuf + rsi]
    cmp al, '['
    je push_open
    cmp al, ']'
    je do_close
build_next:
    inc rsi
    jmp build_loop
push_open:
    mov dword ptr [bstack + rdi*4], esi
    inc rdi
    jmp build_next
do_close:
    test rdi, rdi
    jz bad_brackets
    dec rdi
    mov eax, dword ptr [bstack + rdi*4]     # matching '[' index
    mov dword ptr [jumptab + rsi*4], eax    # jump[']'] = '['
    mov dword ptr [jumptab + rax*4], esi    # jump['['] = ']'
    jmp build_next
build_done:
    test rdi, rdi
    jnz bad_brackets

    # ---- interpret ----
    # rbx = proglen, rbp = input base, r12 = input length
    xor r13d, r13d              # pc
    xor r14d, r14d              # data pointer
    xor r15d, r15d              # input index
run_loop:
    cmp r13, rbx
    jae run_done
    movzx eax, byte ptr [inbuf + r13]
    cmp al, '>'
    je op_right
    cmp al, '<'
    je op_left
    cmp al, '+'
    je op_inc
    cmp al, '-'
    je op_dec
    cmp al, '.'
    je op_out
    cmp al, ','
    je op_in
    cmp al, '['
    je op_open
    cmp al, ']'
    je op_close
    inc r13
    jmp run_loop

op_right:
    inc r14
    inc r13
    jmp run_loop

op_left:
    dec r14
    inc r13
    jmp run_loop

op_inc:
    inc byte ptr [tape + r14]
    inc r13
    jmp run_loop

op_dec:
    dec byte ptr [tape + r14]
    inc r13
    jmp run_loop

op_out:
    mov eax, 1                  # sys_write
    mov edi, 1                  # fd 1
    lea rsi, [rip + tape]
    add rsi, r14
    mov edx, 1
    syscall
    inc r13
    jmp run_loop

op_in:
    xor eax, eax
    cmp r15, r12
    jae in_store
    movzx eax, byte ptr [rbp + r15]
    inc r15
in_store:
    mov byte ptr [tape + r14], al
    inc r13
    jmp run_loop

op_open:
    cmp byte ptr [tape + r14], 0
    jne open_next
    mov r13d, dword ptr [jumptab + r13*4]   # matching ']'
open_next:
    inc r13
    jmp run_loop

op_close:
    cmp byte ptr [tape + r14], 0
    je close_next
    mov r13d, dword ptr [jumptab + r13*4]   # matching '['
close_next:
    inc r13
    jmp run_loop

run_done:
    mov eax, 60                 # sys_exit
    xor edi, edi                # status 0
    syscall

bad_brackets:
    mov eax, 60                 # sys_exit
    mov edi, 1                  # status 1
    syscall

.bss
.align 16
tape:
    .zero 30000
inbuf:
    .zero 65536
jumptab:
    .zero 262144
bstack:
    .zero 131072
