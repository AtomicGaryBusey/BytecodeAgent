.globl _start

.section .rodata
hexdigits: .ascii "0123456789abcdef"

.bss
.align 16
inbuf: .zero 65536
outbuf: .zero 400000

.text
_start:
    xor rbx, rbx
1:
    mov rdx, 65536
    sub rdx, rbx
    jz 2f
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    mov r12, rbx
    xor r13, r13
row_loop:
    mov rax, r12
    sub rax, r13
    jz done_rows
    mov r14, 16
    cmp rax, 16
    jae 1f
    mov r14, rax
1:
    mov eax, r13d
    mov ecx, 28
    lea r9, [rip + hexdigits]
2:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jnc 2b
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor r15, r15
slot_loop:
    cmp r15, r14
    jae blank_slot
    movzx eax, byte ptr [rsi + r15]
    mov edx, eax
    shr edx, 4
    lea r9, [rip + hexdigits]
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], ' '
    jmp slot_done
blank_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
slot_done:
    add rdi, 3
    cmp r15, 7
    jne 3f
    mov byte ptr [rdi], ' '
    inc rdi
3:
    inc r15
    cmp r15, 16
    jb slot_loop
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor r15, r15
asc_loop:
    cmp r15, r14
    jae asc_done
    movzx eax, byte ptr [rsi + r15]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    ja dot
    mov byte ptr [rdi], al
    jmp 4f
dot:
    mov byte ptr [rdi], '.'
4:
    inc rdi
    inc r15
    jmp asc_loop
asc_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add rsi, r14
    add r13, r14
    jmp row_loop
done_rows:
    mov eax, r12d
    mov ecx, 28
    lea r9, [rip + hexdigits]
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jnc 1b
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_ok
    mov rdi, 1
    mov rax, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
