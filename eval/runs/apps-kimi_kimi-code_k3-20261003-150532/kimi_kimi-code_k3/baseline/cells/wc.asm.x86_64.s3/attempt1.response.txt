.text
.globl _start

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax
process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne not_newline
    inc r12
not_newline:
    cmp al, 0x20
    je is_space
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe is_space
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
is_space:
    xor r15d, r15d
next_byte:
    dec rcx
    jnz process_loop
    jmp read_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_num:
    lea r9, [rip + tmpbuf + 31]
    mov r8, r9
    mov rbx, 10
convert_loop:
    xor edx, edx
    div rbx
    add dl, 0x30
    mov byte ptr [r9], dl
    dec r9
    test rax, rax
    jnz convert_loop
    lea rcx, [r9 + 1]
copy_loop:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    cmp rcx, r8
    jbe copy_loop
    ret

.bss
.align 16
buffer:
    .zero 65536
outbuf:
    .zero 128
tmpbuf:
    .zero 32
