// base64: encode stdin to base64 (RFC 4648), 76-char lines
// static AArch64, no libc
.text
.globl _start
_start:
    // ---- read all of stdin into inbuf ----
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf   // x19 = buffer base
    mov  x20, #0                 // x20 = total bytes read
    mov  x21, #49152             // capacity
1:
    sub  x2, x21, x20            // remaining space
    cbz  x2, 2f
    mov  x0, #0                  // fd = stdin
    add  x1, x19, x20
    mov  x8, #63                 // read
    svc  #0
    cmp  x0, #0
    ble  2f                      // 0 = EOF (treat error as EOF)
    add  x20, x20, x0
    b    1b
2:
    // ---- encode ----
    adrp x22, outbuf
    add  x22, x22, :lo12:outbuf
    mov  x23, x22                // x23 = output write ptr
    mov  x24, #0                 // x24 = column count
    adrp x25, b64tab
    add  x25, x25, :lo12:b64tab
    mov  x9, x19                 // x9 = input ptr
    mov  x10, #0                 // x10 = input index
    mov  w11, #10                // '\n'
    sub  x12, x20, #2            // loop while index < total-2
loop3:
    cmp  x10, x12
    b.ge tail
    ldrb w0, [x9], #1
    ldrb w1, [x9], #1
    ldrb w2, [x9], #1
    lsr  w3, w0, #2              // c0 = b0 >> 2
    and  w4, w0, #3
    lsl  w4, w4, #4
    orr  w4, w4, w1, lsr #4      // c1 = (b0&3)<<4 | b1>>4
    and  w5, w1, #15
    lsl  w5, w5, #2
    orr  w5, w5, w2, lsr #6      // c2 = (b1&15)<<2 | b2>>6
    and  w6, w2, #63             // c3 = b2 & 63
    ldrb w3, [x25, x3]
    ldrb w4, [x25, x4]
    ldrb w5, [x25, x5]
    ldrb w6, [x25, x6]
    strb w3, [x23], #1
    strb w4, [x23], #1
    strb w5, [x23], #1
    strb w6, [x23], #1
    add  x10, x10, #3
    add  x24, x24, #4
    cmp  x24, #76
    b.ne loop3
    strb w11, [x23], #1
    mov  x24, #0
    b    loop3
tail:
    sub  x13, x20, x10           // remaining bytes: 0, 1 or 2
    cbz  x13, finish
    cmp  x13, #1
    b.eq tail1
    // ---- 2 remaining bytes -> 3 chars + '=' ----
    ldrb w0, [x9], #1
    ldrb w1, [x9], #1
    lsr  w3, w0, #2
    and  w4, w0, #3
    lsl  w4, w4, #4
    orr  w4, w4, w1, lsr #4
    and  w5, w1, #15
    lsl  w5, w5, #2
    ldrb w3, [x25, x3]
    ldrb w4, [x25, x4]
    ldrb w5, [x25, x5]
    strb w3, [x23], #1
    strb w4, [x23], #1
    strb w5, [x23], #1
    mov  w6, #61                 // '='
    strb w6, [x23], #1
    add  x24, x24, #4
    b    finish
tail1:
    // ---- 1 remaining byte -> 2 chars + '==' ----
    ldrb w0, [x9], #1
    lsr  w3, w0, #2
    and  w4, w0, #3
    lsl  w4, w4, #4
    ldrb w3, [x25, x3]
    ldrb w4, [x25, x4]
    strb w3, [x23], #1
    strb w4, [x23], #1
    mov  w6, #61                 // '='
    strb w6, [x23], #1
    strb w6, [x23], #1
    add  x24, x24, #4
finish:
    cbz  x24, donl
    strb w11, [x23], #1          // newline after partial last line
donl:
    // ---- write output ----
    sub  x26, x23, x22           // total length
    cbz  x26, exit0
    mov  x27, #0                 // written so far
wloop:
    mov  x0, #1                  // fd = stdout
    add  x1, x22, x27
    sub  x2, x26, x27
    mov  x8, #64                 // write
    svc  #0
    add  x27, x27, x0
    cmp  x27, x26
    b.lt wloop
exit0:
    mov  x0, #0
    mov  x8, #93                 // exit
    svc  #0

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
inbuf:
    .zero 49152
outbuf:
    .zero 66560
