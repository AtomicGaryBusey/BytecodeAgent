.globl _start

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:  .zero 65536
outbuf: .zero 66560

.text
_start:
    # ---- read all of stdin into inbuf ----
    xor r12, r12                # total bytes read so far
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12                # space remaining
    jz read_done
    xor eax, eax                # read = 0
    xor edi, edi                # fd = 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # ---- encode ----
    lea rsi, [rip + inbuf]      # input cursor
    lea rdi, [rip + outbuf]     # output cursor
    mov r13, r12                # bytes remaining
    xor r8d, r8d                # column count (0..75)
    lea r9, [rip + alphabet]

main_loop:
    cmp r13, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx                 # eax = 24-bit group

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], dl
    mov edx, eax
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    sub r13, 3
    add r8d, 4
    cmp r8d, 76
    jne main_loop
    mov byte ptr [rdi], 10      # newline after 76 chars
    inc rdi
    xor r8d, r8d
    jmp main_loop

tail:
    test r13, r13
    jz write_out                # no leftover: output already newline-terminated
    cmp r13, 1
    je one_byte
    # ---- two leftover bytes ----
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], 61  # '='
    add rdi, 4
    jmp end_line
one_byte:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 61  # '='
    mov byte ptr [rdi + 3], 61  # '='
    add rdi, 4
end_line:
    mov byte ptr [rdi], 10      # newline ends the final short line
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                # total output length
write_loop:
    test rdx, rdx
    jz done
    mov eax, 1                  # write = 1
    mov edi, 1                  # fd = 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 60                 # exit = 60
    xor edi, edi                # status = 0
    syscall
