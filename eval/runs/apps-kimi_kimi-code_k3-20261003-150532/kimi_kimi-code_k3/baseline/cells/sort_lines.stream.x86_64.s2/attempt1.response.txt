.globl _start
.text
_start:
    lea rbx, [rip + inbuf]
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    lea rdx, [rip + inbuf_end]
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    mov rcx, rsi
    xor r13d, r13d
    lea r14, [rip + offs]
    lea r15, [rip + lens]
parse_loop:
    cmp rsi, rbx
    jae parse_end
    cmp byte ptr [rsi], 10
    je have_line
    inc rsi
    jmp parse_loop
have_line:
    mov rax, rcx
    lea rdx, [rip + inbuf]
    sub rax, rdx
    mov [r14 + r13*4], eax
    mov rdx, rsi
    sub rdx, rcx
    mov [r15 + r13*4], edx
    inc r13
    inc rsi
    mov rcx, rsi
    jmp parse_loop
parse_end:
    cmp rcx, rbx
    jae sort_start
    mov rax, rcx
    lea rdx, [rip + inbuf]
    sub rax, rdx
    mov [r14 + r13*4], eax
    mov rdx, rbx
    sub rdx, rcx
    mov [r15 + r13*4], edx
    inc r13
sort_start:
    mov r12, 1
sort_outer:
    cmp r12, r13
    jae output_start
    mov r8d, [r14 + r12*4]
    mov r9d, [r15 + r12*4]
    lea rbp, [r12 - 1]
sort_inner:
    test rbp, rbp
    js place_key
    mov eax, [r14 + rbp*4]
    lea rsi, [rip + inbuf]
    add rsi, rax
    mov ecx, [r15 + rbp*4]
    lea rdi, [rip + inbuf]
    add rdi, r8
    mov rdx, r9
    call cmp_gt
    test eax, eax
    jz place_key
    mov eax, [r14 + rbp*4]
    mov [r14 + rbp*4 + 4], eax
    mov eax, [r15 + rbp*4]
    mov [r15 + rbp*4 + 4], eax
    dec rbp
    jmp sort_inner
place_key:
    mov [r14 + rbp*4 + 4], r8d
    mov [r15 + rbp*4 + 4], r9d
    inc r12
    jmp sort_outer
cmp_gt:
    xor r10d, r10d
cmp_loop:
    cmp r10, rcx
    jae cmp_gt_no
    cmp r10, rdx
    jae cmp_gt_yes
    mov al, [rsi + r10]
    cmp al, [rdi + r10]
    ja cmp_gt_yes
    jb cmp_gt_no
    inc r10
    jmp cmp_loop
cmp_gt_no:
    xor eax, eax
    ret
cmp_gt_yes:
    mov eax, 1
    ret
output_start:
    lea rdi, [rip + outbuf]
    xor r12d, r12d
out_loop:
    cmp r12, r13
    jae out_flush
    mov eax, [r14 + r12*4]
    lea rsi, [rip + inbuf]
    add rsi, rax
    mov ecx, [r15 + r12*4]
    rep movsb
    mov al, 10
    stosb
    inc r12
    jmp out_loop
out_flush:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_prog
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
.bss
.align 8
inbuf:
    .zero 131072
inbuf_end:
offs:
    .zero 4096
lens:
    .zero 4096
outbuf:
    .zero 40960
