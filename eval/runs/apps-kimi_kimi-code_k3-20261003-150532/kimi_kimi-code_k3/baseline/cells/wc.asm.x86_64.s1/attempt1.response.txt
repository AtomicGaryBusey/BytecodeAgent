.globl _start

.text

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

.read_loop:
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jle .done_reading
    add r14, rax
    lea rsi, [rip + inbuf]
    mov rcx, rax
.byte_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je .is_ws
    lea edx, [eax - 9]
    cmp dl, 4
    jbe .is_ws
    test r15d, r15d
    jnz 1f
    inc r13
    mov r15d, 1
1:
    inc rsi
    dec rcx
    jnz .byte_loop
    jmp .read_loop
.is_ws:
    cmp al, 0x0A
    jne 2f
    inc r12
2:
    xor r15d, r15d
    inc rsi
    dec rcx
    jnz .byte_loop
    jmp .read_loop

.done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call .fmt_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call .fmt_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call .fmt_num
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

.fmt_num:
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], '0'
    inc rdi
    ret
1:
    lea rsi, [rip + numtmp + 32]
    mov r10, 10
2:
    xor edx, edx
    div r10
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 2b
    lea rdx, [rip + numtmp + 32]
    sub rdx, rsi
3:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rdx
    jnz 3b
    ret

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 64
numtmp:
    .zero 32
