.globl _start

.section .rodata
hexdigits:
	.ascii "0123456789abcdef"

.text
_start:
	xor ebx, ebx
1:
	mov rdx, 65536
	sub rdx, rbx
	jz 2f
	lea rsi, [rip + inbuf]
	add rsi, rbx
	xor edi, edi
	xor eax, eax
	syscall
	test rax, rax
	jle 2f
	add rbx, rax
	jmp 1b
2:
	lea r14, [rip + outbuf]
	lea r8, [rip + hexdigits]
	xor r12d, r12d
rowloop:
	mov r15, rbx
	sub r15, r12
	jz final
	mov ebp, 16
	cmp r15, 16
	jae 3f
	mov ebp, r15d
3:
	lea r13, [rip + inbuf]
	add r13, r12
	mov eax, r12d
	call write_hex8
	mov byte ptr [r14], 32
	mov byte ptr [r14+1], 32
	add r14, 2
	xor ecx, ecx
hexloop:
	cmp ecx, ebp
	jae 4f
	movzx eax, byte ptr [r13+rcx]
	mov edx, eax
	shr eax, 4
	and edx, 15
	mov al, byte ptr [r8+rax]
	mov dl, byte ptr [r8+rdx]
	mov byte ptr [r14], al
	mov byte ptr [r14+1], dl
	mov byte ptr [r14+2], 32
	jmp 5f
4:
	mov byte ptr [r14], 32
	mov byte ptr [r14+1], 32
	mov byte ptr [r14+2], 32
5:
	add r14, 3
	cmp ecx, 7
	jne 6f
	mov byte ptr [r14], 32
	inc r14
6:
	inc ecx
	cmp ecx, 16
	jb hexloop
	mov byte ptr [r14], 32
	mov byte ptr [r14+1], 124
	add r14, 2
	xor ecx, ecx
asciiloop:
	cmp ecx, ebp
	jae 7f
	movzx eax, byte ptr [r13+rcx]
	cmp al, 0x20
	jb 8f
	cmp al, 0x7e
	jbe 9f
8:
	mov al, 46
9:
	mov byte ptr [r14], al
	inc r14
	inc ecx
	jmp asciiloop
7:
	mov byte ptr [r14], 124
	mov byte ptr [r14+1], 10
	add r14, 2
	add r12, rbp
	jmp rowloop
final:
	mov eax, ebx
	call write_hex8
	mov byte ptr [r14], 10
	inc r14
	lea rsi, [rip + outbuf]
	mov rdx, r14
	sub rdx, rsi
writeloop:
	test rdx, rdx
	jz doexit
	mov edi, 1
	mov eax, 1
	syscall
	add rsi, rax
	sub rdx, rax
	jmp writeloop
doexit:
	xor edi, edi
	mov eax, 60
	syscall

write_hex8:
	mov ecx, 8
1:
	rol eax, 4
	mov edx, eax
	and edx, 15
	mov dl, byte ptr [r8+rdx]
	mov byte ptr [r14], dl
	inc r14
	dec ecx
	jnz 1b
	ret

.bss
.align 16
inbuf:
	.zero 65536
outbuf:
	.zero 330000
