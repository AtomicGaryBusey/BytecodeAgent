.globl _start

.text
_start:
    xor ebx, ebx
read_loop:
    mov rdx, 65536
    sub rdx, rbx
    jle read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    add r12, rbx
    lea rbx, [rip + inbuf]

line_loop:
    cmp rbx, r12
    jae do_exit
    mov r13, rbx
find_nl:
    cmp r13, r12
    jae have_line
    cmp byte ptr [r13], 10
    je have_line
    inc r13
    jmp find_nl
have_line:
    mov r14, rbx
    lea r15, [rip + val_stack]

tok_loop:
    cmp r14, r13
    jae line_done
    mov al, byte ptr [r14]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp token_start
skip_one:
    inc r14
    jmp tok_loop

token_start:
    mov r8, r14
scan_tok:
    cmp r14, r13
    jae tok_ready
    mov al, byte ptr [r14]
    cmp al, 32
    je tok_ready
    cmp al, 9
    je tok_ready
    inc r14
    jmp scan_tok
tok_ready:
    mov rax, r14
    sub rax, r8
    cmp rax, 1
    jne parse_literal
    mov al, byte ptr [r8]
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod

parse_literal:
    mov rsi, r8
    xor ecx, ecx
    cmp byte ptr [rsi], 45
    jne digits_init
    mov ecx, 1
    inc rsi
digits_init:
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jz eval_error
    cmp rdx, 18
    ja eval_error
    xor eax, eax
digit_loop:
    cmp rsi, r14
    jae digits_done
    movzx r10d, byte ptr [rsi]
    sub r10d, 48
    cmp r10d, 9
    ja eval_error
    imul rax, rax, 10
    add rax, r10
    inc rsi
    jmp digit_loop
digits_done:
    test ecx, ecx
    jz push_val
    neg rax
push_val:
    mov qword ptr [r15], rax
    add r15, 8
    jmp tok_loop

op_add:
    call need_two
    mov rax, [r15 - 16]
    add rax, [r15 - 8]
    mov [r15 - 16], rax
    sub r15, 8
    jmp tok_loop

op_sub:
    call need_two
    mov rax, [r15 - 16]
    sub rax, [r15 - 8]
    mov [r15 - 16], rax
    sub r15, 8
    jmp tok_loop

op_mul:
    call need_two
    mov rax, [r15 - 16]
    imul rax, qword ptr [r15 - 8]
    mov [r15 - 16], rax
    sub r15, 8
    jmp tok_loop

op_div:
    call need_two
    mov rax, [r15 - 16]
    mov r10, [r15 - 8]
    test r10, r10
    jz eval_error
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne do_div
    cmp r10, -1
    je eval_error
do_div:
    cqo
    idiv r10
    mov [r15 - 16], rax
    sub r15, 8
    jmp tok_loop

op_mod:
    call need_two
    mov rax, [r15 - 16]
    mov r10, [r15 - 8]
    test r10, r10
    jz eval_error
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne do_mod
    cmp r10, -1
    je eval_error
do_mod:
    cqo
    idiv r10
    mov [r15 - 16], rdx
    sub r15, 8
    jmp tok_loop

need_two:
    lea rax, [rip + val_stack]
    mov rcx, r15
    sub rcx, rax
    cmp rcx, 16
    jb eval_error
    ret

line_done:
    lea rax, [rip + val_stack]
    mov rcx, r15
    sub rcx, rax
    cmp rcx, 8
    jne eval_error
    mov rax, [r15 - 8]
    call print_value
    jmp next_line

eval_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errstr]
    mov edx, 6
    syscall

next_line:
    cmp r13, r12
    jae no_more
    lea rbx, [r13 + 1]
    jmp line_loop
no_more:
    mov rbx, r12
    jmp line_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

print_value:
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 10
    xor r9d, r9d
    test rax, rax
    jns pv_abs
    mov r9d, 1
    neg rax
pv_abs:
    mov rcx, 10
pv_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz pv_loop
    test r9d, r9d
    jz pv_write
    dec rdi
    mov byte ptr [rdi], 45
pv_write:
    lea rdx, [rip + numbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
errstr: .ascii "error\n"

.bss
.align 8
inbuf: .zero 65536
val_stack: .zero 8192
numbuf: .zero 32
