.globl _start

.equ INPUT_MAX, 32768
.equ MAX_LINES, 32768

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, INPUT_MAX
    sub rdx, rbx
    jz read_done
    lea rsi, [rip + buffer]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r14, [rip + lines]
    xor r13d, r13d
    xor r12d, r12d
    xor r15d, r15d
    lea rbp, [rip + buffer]
parse_loop:
    cmp r15, rbx
    jae parse_done
    cmp byte ptr [rbp + r15], 10
    jne parse_next
    mov rax, r13
    shl rax, 4
    mov [r14 + rax], r12
    mov rcx, r15
    sub rcx, r12
    mov [r14 + rax + 8], rcx
    inc r13
    inc r15
    mov r12, r15
    jmp parse_loop
parse_next:
    inc r15
    jmp parse_loop
parse_done:
    cmp r12, rbx
    jae sort_init
    mov rax, r13
    shl rax, 4
    mov [r14 + rax], r12
    mov rcx, rbx
    sub rcx, r12
    mov [r14 + rax + 8], rcx
    inc r13
sort_init:
    mov r15d, 1
sort_outer:
    cmp r15, r13
    jae output_init
    mov rax, r15
    shl rax, 4
    mov rbx, [r14 + rax]
    mov rbp, [r14 + rax + 8]
    lea r12, [r15 - 1]
sort_inner:
    test r12, r12
    js sort_insert
    mov rax, r12
    shl rax, 4
    mov rdi, [r14 + rax]
    mov rsi, [r14 + rax + 8]
    mov rdx, rbx
    mov rcx, rbp
    call compare
    test eax, eax
    jle sort_insert
    mov rax, r12
    shl rax, 4
    mov rdi, [r14 + rax]
    mov rsi, [r14 + rax + 8]
    mov [r14 + rax + 16], rdi
    mov [r14 + rax + 24], rsi
    dec r12
    jmp sort_inner
sort_insert:
    mov rax, r12
    shl rax, 4
    mov [r14 + rax + 16], rbx
    mov [r14 + rax + 24], rbp
    inc r15
    jmp sort_outer

compare:
    lea r8, [rip + buffer]
    add rdi, r8
    add rdx, r8
    mov rax, rsi
    cmp rcx, rax
    cmovb rax, rcx
    xor r8d, r8d
compare_loop:
    cmp r8, rax
    jae compare_prefix
    movzx r9d, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    sub r9d, r10d
    jnz compare_diff
    inc r8
    jmp compare_loop
compare_diff:
    mov eax, r9d
    ret
compare_prefix:
    xor eax, eax
    cmp rsi, rcx
    je compare_ret
    jb compare_less
    mov eax, 1
    ret
compare_less:
    mov eax, -1
compare_ret:
    ret

output_init:
    xor r15d, r15d
output_loop:
    cmp r15, r13
    jae finish
    mov rax, r15
    shl rax, 4
    mov rdx, [r14 + rax + 8]
    lea rsi, [rip + buffer]
    add rsi, [r14 + rax]
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r15
    jmp output_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
buffer:
    .zero 32768
lines:
    .zero 524288
