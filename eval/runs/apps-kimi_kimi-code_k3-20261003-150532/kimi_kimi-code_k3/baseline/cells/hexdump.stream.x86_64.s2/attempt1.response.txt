.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.text
_start:
    lea rbp, [rip + hexdigits]
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r13, [rip + outbuf]
    xor r14d, r14d
row_loop:
    cmp r14, rbx
    jae final_line
    mov r15, rbx
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    mov eax, r14d
    mov ecx, 8
2:
    mov edx, eax
    shr edx, 28
    movzx edx, byte ptr [rbp + rdx]
    mov byte ptr [r13], dl
    inc r13
    shl eax, 4
    dec ecx
    jnz 2b
    mov byte ptr [r13], 32
    mov byte ptr [r13 + 1], 32
    add r13, 2
    lea rsi, [r12 + r14]
    xor ecx, ecx
hex_loop:
    cmp rcx, r15
    jae pad_slot
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [rbp + rdx]
    mov byte ptr [r13], dl
    and eax, 15
    movzx eax, byte ptr [rbp + rax]
    mov byte ptr [r13 + 1], al
    jmp slot_space
pad_slot:
    mov byte ptr [r13], 32
    mov byte ptr [r13 + 1], 32
slot_space:
    mov byte ptr [r13 + 2], 32
    add r13, 3
    cmp ecx, 7
    jne 3f
    mov byte ptr [r13], 32
    inc r13
3:
    inc ecx
    cmp ecx, 16
    jb hex_loop
    mov byte ptr [r13], 32
    mov byte ptr [r13 + 1], 124
    add r13, 2
    xor ecx, ecx
ascii_loop:
    cmp rcx, r15
    jae ascii_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 32
    jb dot
    cmp al, 126
    jbe store_char
dot:
    mov al, 46
store_char:
    mov byte ptr [r13], al
    inc r13
    inc ecx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r13], 124
    mov byte ptr [r13 + 1], 10
    add r13, 2
    add r14, r15
    jmp row_loop
final_line:
    mov eax, ebx
    mov ecx, 8
4:
    mov edx, eax
    shr edx, 28
    movzx edx, byte ptr [rbp + rdx]
    mov byte ptr [r13], dl
    inc r13
    shl eax, 4
    dec ecx
    jnz 4b
    mov byte ptr [r13], 10
    inc r13
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 330000
