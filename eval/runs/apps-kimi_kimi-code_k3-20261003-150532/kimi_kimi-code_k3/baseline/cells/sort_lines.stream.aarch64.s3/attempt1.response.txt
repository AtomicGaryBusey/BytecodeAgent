.equ IN_CAP, 32768
.equ MAX_LINES, 1000
.equ OUT_CAP, 33768
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    movz x20, #32768
    mov x21, #0
read_loop:
    cmp x20, #0
    b.eq read_done
    mov x0, #0
    mov x1, x19
    mov x2, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    add x21, x21, x0
    sub x20, x20, x0
    b read_loop

read_done:
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    adrp x26, lineinfo
    add x26, x26, :lo12:lineinfo
    mov x23, #0
    mov x24, #0
    mov x25, #0
parse_loop:
    cmp x25, x21
    b.hs parse_done
    ldrb w0, [x22, x25]
    cmp w0, #10
    b.ne parse_next
    bl record_line
    add x24, x25, #1
parse_next:
    add x25, x25, #1
    b parse_loop
parse_done:
    cmp x24, x21
    b.hs sort_start
    mov x25, x21
    bl record_line

sort_start:
    mov x19, #1
sort_outer:
    cmp x19, x23
    b.hs build_output
    add x27, x26, x19, lsl #3
    ldr x21, [x27]
    mov x20, x19
sort_inner:
    cmp x20, #0
    b.eq sort_place
    add x27, x26, x20, lsl #3
    ldr x0, [x27, #-8]
    mov x1, x21
    mov x2, x22
    bl compare_lines
    cmp w0, #0
    b.le sort_place
    ldr x0, [x27, #-8]
    str x0, [x27]
    sub x20, x20, #1
    b sort_inner
sort_place:
    add x27, x26, x20, lsl #3
    str x21, [x27]
    add x19, x19, #1
    b sort_outer

build_output:
    mov x19, #0
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
output_loop:
    cmp x19, x23
    b.hs output_done
    add x27, x26, x19, lsl #3
    ldr x21, [x27]
    mov w0, w21
    lsr x1, x21, #32
    add x2, x22, x0
    mov x3, #0
copy_loop:
    cmp x3, x1
    b.hs copy_done
    ldrb w4, [x2, x3]
    strb w4, [x24, x3]
    add x3, x3, #1
    b copy_loop
copy_done:
    add x24, x24, x1
    mov w4, #10
    strb w4, [x24], #1
    add x19, x19, #1
    b output_loop
output_done:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    sub x2, x24, x25
    mov x1, x25
write_loop:
    cmp x2, #0
    b.eq exit0
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le exit0
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop
exit0:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

record_line:
    cmp x23, #MAX_LINES
    b.hs 1f
    add x27, x26, x23, lsl #3
    str w24, [x27]
    sub w0, w25, w24
    str w0, [x27, #4]
    add x23, x23, #1
1:
    ret

compare_lines:
    mov w3, w0
    lsr x4, x0, #32
    mov w5, w1
    lsr x6, x1, #32
    add x7, x2, x3
    add x11, x2, x5
    cmp x4, x6
    csel x12, x4, x6, ls
    mov x8, #0
1:
    cmp x8, x12
    b.hs 2f
    ldrb w9, [x7, x8]
    ldrb w10, [x11, x8]
    cmp w9, w10
    b.ne 3f
    add x8, x8, #1
    b 1b
3:
    b.lo 4f
    mov w0, #1
    ret
4:
    mov w0, #-1
    ret
2:
    cmp w4, w6
    b.lo 4b
    b.hi 3b
    mov w0, #0
    ret

.bss
.balign 8
inbuf:
    .zero IN_CAP
.balign 8
lineinfo:
    .zero MAX_LINES * 8
outbuf:
    .zero OUT_CAP
