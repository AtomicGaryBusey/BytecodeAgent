

bubble_sort:
sub x3, x1, #1
mov x2, x1
outer_loop:
cmp x2, #0
b.le outer_done
mov x4, #0
inner_loop_start:
cmp x4, x3
b.ge inner_done
add x5, x0, x4, lsl #3
ldr x6, [x5]
add x7, x5, #8
ldr x8, [x7]
cmp x6, x8
b.le no_swap
str x8, [x5]
str x6, [x7]
no_swap:
add x4, x4, #1
b inner_loop_start
inner_done:
sub x3, x3, #1
sub x2, x2, #1
b outer_loop
outer_done:
mov x0, #0
ret
