

export function l $count_nonzero_prefix(l %a, l %n) {
    %ptr = %a
    %counter = $0
    %end = addl %a, %n
    @loop:
    %lt = csltl %ptr, %end
    jnz @body
    %res = %n
    jmp @end
    @body:
    %val = loadl %ptr
    %zero = ceql %val, $0
    jnz @found_zero
    %counter = addl %counter, $1
    %ptr = addl %ptr, $1
    jmp @loop
    @found_zero:
    %res = %counter
    jmp @end
    @end:
    retl %res
}
