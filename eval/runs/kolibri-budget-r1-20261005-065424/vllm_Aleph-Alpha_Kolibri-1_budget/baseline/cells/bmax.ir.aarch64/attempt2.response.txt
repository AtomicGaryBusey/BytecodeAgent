

export function l $bmax(l ptr, l cnt) {
  @start
    loadl %max, [ptr]
    addl %p, ptr, 8
    movl %i, cnt
    subl %i, %i, 1
    csltl %t, 0, %i
    jnz @loop
    jmp @return

  @loop
    loadl %elem, [%p]
    csltl %t, %max, %elem
    jnz @update
    jmp @next

  @update
    movl %max, %elem
    jmp @next

  @next
    addl %p, %p, 8
    subl %i, %i, 1
    csltl %t, 0, %i
    jnz @loop
    jmp @return

  @return
    ret %max
}
