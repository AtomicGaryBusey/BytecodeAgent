

export function l $gcd(l a, l b) {
  @loop:
    %eq_a = ceql %a 0
    jz @exit_a
    %eq_b = ceql %b 0
    jz @exit_b
    %lt_ab = csltl %a %b
    jnz @b_ge_a
    %new_a = subl %a %b
    %a = %new_a
    jmp @loop
  @b_ge_a:
    %new_b = subl %b %a
    %b = %new_b
    jmp @loop
  @exit_a:
    ret l %b
  @exit_b:
    ret l %a
}
