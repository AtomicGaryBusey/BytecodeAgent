

export function l $gcd(l a, l b) {
  @loop:
    %a_val = loadl %a_val a
    %b_val = loadl %b_val b
    %eq_b = ceql %b_val 0
    jnz @exit_b
    %eq_a = ceql %a_val 0
    jnz @exit_a
    jmp @subtract
  @exit_b:
    ret l %a_val
  @exit_a:
    ret l %b_val
  @subtract:
    %lt = csltl %b_val %a_val
    jnz @a_ge_b
    jmp @b_ge_a
  @a_ge_b:
    %new_a = subl %a_val %b_val
    storel a %new_a
    jmp @loop
  @b_ge_a:
    %new_b = subl %b_val %a_val
    storel b %new_b
    jmp @loop
}
