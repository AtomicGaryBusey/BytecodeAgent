

export function l $reverse(l a, l n) {
  l i = 0;
  l j = 0;
  @loop:
    j = n - 1 - i;
    %cmp = cgeql i, j;
    jnz %cmp @done;
    %addr_i = a + i * 8;
    %addr_j = a + j * 8;
    loadl %val_i, %addr_i;
    loadl %val_j, %addr_j;
    storel %val_i, %addr_j;
    storel %val_j, %addr_i;
    i = i + 1;
    jmp @loop;
  @done:
    ret 0;
}
