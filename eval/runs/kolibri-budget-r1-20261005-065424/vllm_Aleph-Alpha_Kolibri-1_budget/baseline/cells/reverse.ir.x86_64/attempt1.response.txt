

export function l $reverse(l a, l n) {
  %cmp = csltl n, 2;
  jnz %cmp @done;
  l i = 0;
  l j;
  @loop:
    j = n - 1 - i;
    %cmp = csltl i, j;
    %exit_cond = 1 - %cmp;
    jnz %exit_cond @done;
    %addr_i = a + i * 8;
    %addr_j = a + j * 8;
    loadl %val_i, %addr_i;
    loadl %val_j, %addr_j;
    storel %val_i, %addr_j;
    storel %val_j, %addr_i;
    i = i + 1;
    jmp @loop;
  @done:
    ret 0;
}
