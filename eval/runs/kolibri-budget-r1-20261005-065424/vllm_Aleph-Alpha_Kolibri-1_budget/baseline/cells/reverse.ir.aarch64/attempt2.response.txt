

export function l $reverse(l %ptr, l %len) {
    @start:
    %flag_len_zero = ceql %w, %len, 0
    jnz %flag_len_zero, @compute_p2
    jmp @done
@compute_p2:
    %len1 = subl %len, 1
    %offset = mull %len1, 8
    %p1 = %ptr
    %p2 = addl %ptr, %offset
    jmp @loop
@loop:
    %flag_cmp = csltl %w, %p1, %p2
    jnz %flag_cmp, @swap
    jmp @done
@swap:
    %val1 = loadl %p1
    %val2 = loadl %p2
    storel %val1, %p2
    storel %val2, %p1
    %p1 = addl %p1, 8
    %p2 = subl %p2, 8
    jmp @loop
@done:
    %zero = subl %len, %len
    ret
}
