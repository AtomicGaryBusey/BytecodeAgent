

export function l $reverse(l %ptr, l %len) {
    @start
    %p1 = %ptr
    %len1 = subl %len, 1l
    %offset = mull %len1, 8l
    %p2 = addl %ptr, %offset
    jmp @loop
    @loop
    %flag_cmp = csltl %p1, %p2
    jnz %flag_cmp, @swap
    jmp @done
    @swap
    %val1 = loadl %p1
    %val2 = loadl %p2
    storel %val1, %p2
    storel %val2, %p1
    %p1 = addl %p1, 8l
    %p2 = subl %p2, 8l
    jmp @loop
    @done
    %zero = subl %len, %len
    ret
}
