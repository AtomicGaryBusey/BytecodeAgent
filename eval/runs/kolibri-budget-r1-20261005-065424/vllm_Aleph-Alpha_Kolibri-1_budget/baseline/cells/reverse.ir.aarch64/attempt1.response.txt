

export function l $reverse(l a, l n) {
    @start:
    %flag_n_zero = ceql %w, %n, 0
    jnz %flag_n_zero, @compute_p2
    jmp @done
@compute_p2:
    %n1 = subl %n, 1
    %offset = mull %n1, 8
    %p1 = %a
    %p2 = addl %a, %offset
    jmp @loop
@loop:
    %flag_cmp = csltl %w, %p1, %p2
    jnz %flag_cmp, @swap
    jmp @done
@swap:
    %val1 = loadl %p1
    %val2 = loadl %p2
    storel %val1, %p2
    storel %val2, %p1
    %p1 = addl %p1, 8
    %p2 = subl %p2, 8
    jmp @loop
@done:
    %zero = subl %n, %n
    ret
}
