

export function l $bubble_sort(l ptr, l len)
@bubble_sort:
    movl $0, %zero
    movl $1, %one
    movl $8, %eight
    movl ptr, %base
    movl len, %n
    subtl %limit, %n, %one
    jmp @outer
@outer:
    csltl %less, %zero, %limit
    jnz %less, @outer_body
    jmp @return_zero
@outer_body:
    movl $0, %j
    movl %base, %curr_addr
    addl %next_addr, %base, %eight
    jmp @inner
@inner:
    csltl %less, %j, %limit
    jnz %less, @inner_body
    jmp @inner_done
@inner_body:
    loadl %cur_val, %curr_addr
    loadl %next_val, %next_addr
    csltl %less, %next_val, %cur_val
    jnz %less, @swap
    jmp @increment
@swap:
    storel %next_val, %curr_addr
    storel %cur_val, %next_addr
    jmp @increment
@increment:
    addl %j, %j, %one
    addl %curr_addr, %curr_addr, %eight
    addl %next_addr, %next_addr, %eight
    jmp @inner
@inner_done:
    subtl %limit, %limit, %one
    jmp @outer
@return_zero:
    movl $0, %rax
    ret
