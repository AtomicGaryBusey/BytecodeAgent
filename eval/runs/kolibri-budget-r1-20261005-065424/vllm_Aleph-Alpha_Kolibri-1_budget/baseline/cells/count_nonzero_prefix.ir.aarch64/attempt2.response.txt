

export function l $count_nonzero_prefix(l a, l n)
@start:
    li %zero, 0
    li %one, 1
    li %eight, 8
    movl %ptr, %a
    movl %count, %zero
@loop:
    csltl %cmp, %count, %n
    jnz @body
    jmp @exit
@body:
    loadl %val, %ptr
    ceql %eq, %val, %zero
    jnz @exit
    addl %count, %count, %one
    addl %ptr, %ptr, %eight
    jmp @loop
@exit:
    retl
