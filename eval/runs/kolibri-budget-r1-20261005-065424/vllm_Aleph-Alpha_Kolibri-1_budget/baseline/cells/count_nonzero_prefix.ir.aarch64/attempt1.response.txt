

export function l $count_nonzero_prefix(l a, l n)
@start:
    movl %ptr, a
    movl %n, n
    li %zero, 0
    li %one, 1
    li %eight, 8
    movl %zero, %count
    jmp @loop
@loop:
    csltl %cmp, %count, %n
    jnz @load
    jmp @exit
@load:
    loadl %val, %ptr
    ceql %eq, %val, %zero
    jnz @zero
    addl %count, %count, %one
    addl %ptr, %ptr, %eight
    jmp @loop
@zero:
    jmp @exit
@exit:
    retl
