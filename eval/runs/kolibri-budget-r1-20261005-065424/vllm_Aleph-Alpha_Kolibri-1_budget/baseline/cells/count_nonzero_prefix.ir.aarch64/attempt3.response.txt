

export function l $count_nonzero_prefix(l ptr, l len)
@start:
    li %count, 0
    li %zero, 0
    li %one, 1
    li %eight, 8
@loop:
    csltl %cmp, %count, %len
    jnz @body
    jmp @exit
@body:
    loadl %val, %ptr
    ceql %eq, %val, %zero
    jnz @exit
    addl %count, %count, %one
    addl %ptr, %ptr, %eight
    jmp @loop
@exit:
    retl
