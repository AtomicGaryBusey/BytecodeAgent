

export function l $bubble_sort(l a, l n) {
@start:
  %i = 0
  %j = 0
  %one = 1
  %n1 = subl %n, %one
  jmp @outer_test
@outer_test:
  %cmp_o = csltl %i, %n1
  jnz %cmp_o, @inner_start
  jmp @outer_exit
@inner_start:
  %j = 0
  %limit = subl %n, %i
  %limit = subl %limit, %one
  %cmp_i = csltl %j, %limit
  jnz %cmp_i, @inner_body
  jmp @after_inner
@inner_body:
  loadl %val_j, a + %j*8
  loadl %val_j1, a + %j*8 + 8
  %cmp = csltl %val_j1, %val_j
  jnz %cmp, @swap
  jmp @increment_j
@swap:
  storel a + %j*8, %val_j1
  storel a + %j*8 + 8, %val_j
@increment_j:
  %j = addl %j, %one
  jmp @inner_start
@after_inner:
  %i = addl %i, %one
  jmp @outer_test
@outer_exit:
  retl 0
}
