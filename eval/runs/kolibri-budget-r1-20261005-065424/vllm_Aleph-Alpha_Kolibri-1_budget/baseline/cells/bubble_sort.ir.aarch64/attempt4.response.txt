

export function l $bubble_sort(l arr, l len) {
@start:
  %i = 0
  %one = 1
  %zero = 0
  %eight = 8
  %n1 = subl len, %one
  jmp @outer_test
@outer_test:
  %cmp_o = csltl %i, %n1
  jnz %cmp_o, @inner_start
  jmp @outer_exit
@inner_start:
  %j = 0
  %n_minus_i = subl len, %i
  %limit = subl %n_minus_i, %one
  jmp @inner_check
@inner_check:
  %cmp_i = csltl %j, %limit
  jnz %cmp_i, @inner_body
  jmp @after_inner
@inner_body:
  %mul_j = mulil %j, %eight
  %addr_j = addl arr, %mul_j
  %addr_j1 = addl %addr_j, %eight
  %val_j = loadl %val_j, %addr_j
  %val_j1 = loadl %val_j1, %addr_j1
  %cmp = csltl %val_j1, %val_j
  jnz %cmp, @swap
  jmp @increment_j
@swap:
  storel %addr_j, %val_j1
  storel %addr_j1, %val_j
  jmp @increment_j
@increment_j:
  %j = addl %j, %one
  jmp @inner_check
@after_inner:
  %i = addl %i, %one
  jmp @outer_test
@outer_exit:
  retl %zero
}
