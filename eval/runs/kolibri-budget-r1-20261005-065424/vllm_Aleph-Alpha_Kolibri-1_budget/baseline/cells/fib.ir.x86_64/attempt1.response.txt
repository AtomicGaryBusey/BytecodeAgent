

export function l $fib(l n)
@fib
%n = loadl n
%eq0 = ceql %n, 0
jnz %eq0, @ret0
%eq1 = ceql %n, 1
jnz %eq1, @ret1
%a = 0
%b = 1
%i = 2
@loop
%lt = csltl %n, %i
jnz %lt, @setret
%c = addl %a, %b
%a = %b
%b = %c
%i = addl %i, 1
jmp @loop
@setret
%rax = %b
jmp @done
@ret0
%rax = 0
jmp @done
@ret1
%rax = 1
jmp @done
@done
