

export function l $fib(l x)
@fib
w %eq0 = ceql x, 0l
jnz %eq0, @ret0
w %eq1 = ceql x, 1l
jnz %eq1, @ret1
l %a = 0l
l %b = 1l
l %i = 2l
l %limit = addl x, 1l
@loop
w %lt = csltl %i, %limit
jnz %lt, @iterate
jmp @setret
@iterate
l %c = addl %a, %b
l %a = %b
l %b = %c
l %i = addl %i, 1l
jmp @loop
@setret
l %rax = %b
jmp @done
@ret0
l %rax = 0l
jmp @done
@ret1
l %rax = 1l
jmp @done
@done
ret %rax
