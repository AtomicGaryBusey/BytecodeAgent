

export function l $fib(l n)
@fib
%val = n
%eq0 = ceql %val, 0l
jnz %eq0, @ret0
%eq1 = ceql %val, 1l
jnz %eq1, @ret1
%a = 0l
%b = 1l
%i = 2l
%limit = addl %val, 1l
@loop
%lt = csltl %i, %limit
jnz %lt, @iterate
jmp @setret
@iterate
%c = addl %a, %b
%a = %b
%b = %c
%i = addl %i, 1l
jmp @loop
@setret
%rax = %b
jmp @done
@ret0
%rax = 0l
jmp @done
@ret1
%rax = 1l
jmp @done
@done
ret %rax
