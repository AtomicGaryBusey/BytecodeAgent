

export function l $fib(l arg)
@fib
%val = loadl arg
%eq0 = ceql %val, 0l
jnz %eq0, @ret0
%eq1 = ceql %val, 1l
jnz %eq1, @ret1
%a = 0l
%b = 1l
%i = 2l
@loop
%lt = csltl %val, %i
jnz %lt, @setret
%c = addl %a, %b
%a = %b
%b = %c
%i = addl %i, 1l
jmp @loop
@setret
%rax = %b
jmp @done
@ret0
%rax = 0l
jmp @done
@ret1
%rax = 1l
jmp @done
@done
ret %rax
