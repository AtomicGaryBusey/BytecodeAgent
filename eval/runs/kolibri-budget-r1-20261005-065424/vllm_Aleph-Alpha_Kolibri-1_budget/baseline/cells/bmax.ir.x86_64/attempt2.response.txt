

export function l bmax(l $a, l $n) {
@entry
movl $1, %one
addl %end, %a, %n
loadl %max, %a
addl %ptr, %a, %one
jmp @loop
@loop
csltl %cmp, %ptr, %end
jnz @body
jmp @exit
@body
loadl %cur, %ptr
csltl %cmp2, %max, %cur
jnz @update
jmp @incr
@update
movl %cur, %max
@incr
addl %ptr, %ptr, %one
jmp @loop
@exit
retl %max
}
