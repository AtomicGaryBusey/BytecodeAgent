

bubble_sort:
test rsi, rsi
jz .Ldone
mov r8, rsi
dec r8
.Louter:
test r8, r8
jz .Ldone
mov rcx, 0
.Linner:
cmp rcx, r8
jge .Linner_end
mov rdx, qword ptr [rdi + rcx*8]
mov rax, qword ptr [rdi + rcx*8 + 8]
cmp rdx, rax
jle .Lnoswap
mov qword ptr [rdi + rcx*8], rax
mov qword ptr [rdi + rcx*8 + 8], rdx
.Lnoswap:
inc rcx
jmp .Linner
.Linner_end:
dec r8
jmp .Louter
.Ldone:
mov rax, 0
ret
