export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %n2 =l mul %n, %n
    %mat_bytes =l mul %n2, 8
    %scratch =l add %tmp, %mat_bytes
    %n2x2 =l mul %n2, 2
    jmp @id_i_enter
@id_i_enter
    jmp @id_i_cond
@id_i_cond
    %id_i =l phi @id_i_enter 0, @id_i_inc %id_i_nx
    %id_ci =w csltl %id_i, %n
    jnz %id_ci, @id_j_enter, @id_done
@id_j_enter
    jmp @id_j_cond
@id_j_cond
    %id_j =l phi @id_j_enter 0, @id_j_inc %id_j_nx
    %id_cj =w csltl %id_j, %n
    jnz %id_cj, @id_body, @id_i_inc
@id_body
    %id_in =l mul %id_i, %n
    %id_inj =l add %id_in, %id_j
    %id_off =l mul %id_inj, 8
    %id_ptr =l add %out, %id_off
    %id_eq =w ceql %id_i, %id_j
    %id_val =l extsw %id_eq
    storel %id_val, %id_ptr
    jmp @id_j_inc
@id_j_inc
    %id_j_nx =l add %id_j, 1
    jmp @id_j_cond
@id_i_inc
    %id_i_nx =l add %id_i, 1
    jmp @id_i_cond
@id_done
    jmp @kl_enter
@kl_enter
    jmp @kl_cond
@kl_cond
    %kl_s =l phi @kl_enter 0, @kl_inc %kl_s_nx
    %kl_c =w csltl %kl_s, %k
    jnz %kl_c, @mm_i_enter, @kl_done
@mm_i_enter
    jmp @mm_i_cond
@mm_i_cond
    %mm_i =l phi @mm_i_enter 0, @mm_i_inc %mm_i_nx
    %mm_ci =w csltl %mm_i, %n
    jnz %mm_ci, @mm_j_enter, @mm_done
@mm_j_enter
    jmp @mm_j_cond
@mm_j_cond
    %mm_j =l phi @mm_j_enter 0, @mm_j_inc %mm_j_nx
    %mm_cj =w csltl %mm_j, %n
    jnz %mm_cj, @mm_ij_pre, @mm_i_inc
@mm_ij_pre
    %mm_in =l mul %mm_i, %n
    %mm_inj =l add %mm_in, %mm_j
    jmp @mm_t_enter
@mm_t_enter
    jmp @mm_t_cond
@mm_t_cond
    %mm_t =l phi @mm_t_enter 0, @mm_t_inc %mm_t_nx
    %mm_sum =l phi @mm_t_enter 0, @mm_t_inc %mm_sum_nx
    %mm_ct =w csltl %mm_t, %n
    jnz %mm_ct, @mm_t_body, @mm_t_done
@mm_t_body
    %mm_int =l add %mm_in, %mm_t
    %mm_off_a =l mul %mm_int, 8
    %mm_ptr_a =l add %out, %mm_off_a
    %mm_val_a =l loadl %mm_ptr_a
    %mm_tn =l mul %mm_t, %n
    %mm_tnj =l add %mm_tn, %mm_j
    %mm_off_b =l mul %mm_tnj, 8
    %mm_ptr_b =l add %a, %mm_off_b
    %mm_val_b =l loadl %mm_ptr_b
    %mm_prod =l mul %mm_val_a, %mm_val_b
    %mm_sum_nx =l add %mm_sum, %mm_prod
    jmp @mm_t_inc
@mm_t_inc
    %mm_t_nx =l add %mm_t, 1
    jmp @mm_t_cond
@mm_t_done
    %mm_off_c =l mul %mm_inj, 8
    %mm_ptr_c =l add %scratch, %mm_off_c
    storel %mm_sum, %mm_ptr_c
    jmp @mm_j_inc
@mm_j_inc
    %mm_j_nx =l add %mm_j, 1
    jmp @mm_j_cond
@mm_i_inc
    %mm_i_nx =l add %mm_i, 1
    jmp @mm_i_cond
@mm_done
    jmp @cp_enter
@cp_enter
    jmp @cp_cond
@cp_cond
    %cp_i =l phi @cp_enter 0, @cp_inc %cp_i_nx
    %cp_c =w csltl %cp_i, %n2
    jnz %cp_c, @cp_body, @cp_done
@cp_body
    %cp_off =l mul %cp_i, 8
    %cp_src =l add %scratch, %cp_off
    %cp_val =l loadl %cp_src
    %cp_dst =l add %out, %cp_off
    storel %cp_val, %cp_dst
    jmp @cp_inc
@cp_inc
    %cp_i_nx =l add %cp_i, 1
    jmp @cp_cond
@cp_done
    jmp @kl_inc
@kl_inc
    %kl_s_nx =l add %kl_s, 1
    jmp @kl_cond
@kl_done
    jmp @zt_enter
@zt_enter
    jmp @zt_cond
@zt_cond
    %zt_i =l phi @zt_enter 0, @zt_inc %zt_i_nx
    %zt_c =w csltl %zt_i, %n2x2
    jnz %zt_c, @zt_body, @zt_done
@zt_body
    %zt_off =l mul %zt_i, 8
    %zt_ptr =l add %tmp, %zt_off
    storel 0, %zt_ptr
    jmp @zt_inc
@zt_inc
    %zt_i_nx =l add %zt_i, 1
    jmp @zt_cond
@zt_done
    ret 0
}
