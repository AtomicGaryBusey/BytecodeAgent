export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    jmp @main_loop

@main_loop
    %i =l phi @entry 0, @loop_next %i_next
    %op =l phi @entry 0, @loop_next %op_next
    %lc =w csltl %i, %n
    jnz %lc, @get_val, @done

@get_val
    %p_i =l add %src, %i
    %val =w loadub %p_i
    %i128 =l add %i, 128
    %n_lt =w csltl %n, %i128
    jnz %n_lt, @rmax_n, @rmax_128

@rmax_n
    jmp @run_start

@rmax_128
    jmp @run_start

@run_start
    %rmax =l phi @rmax_n %n, @rmax_128 %i128
    %rj0 =l add %i, 1
    jmp @run_loop

@run_loop
    %rj =l phi @run_start %rj0, @run_inc %rj_new
    %c1 =w csltl %rj, %rmax
    jnz %c1, @run_eq, @run_done

@run_eq
    %prj =l add %src, %rj
    %brj =w loadub %prj
    %eq =w ceqw %brj, %val
    jnz %eq, @run_inc, @run_done

@run_inc
    %rj_new =l add %rj, 1
    jmp @run_loop

@run_done
    %run =l sub %rj, %i
    %isr =w csltl 2, %run
    jnz %isr, @chk_rep, @lit_scan_loop

@chk_rep
    %sp_r =l sub %cap, %op
    %ok_r =w csltl 1, %sp_r
    jnz %ok_r, @do_rep, @ret_neg1

@do_rep
    %run_w =w truncl %run
    %hdr_r =w sub 257, %run_w
    %outp0 =l add %out, %op
    storeb %hdr_r, %outp0
    %op1 =l add %op, 1
    %outp1 =l add %out, %op1
    storeb %val, %outp1
    %op_r =l add %op, 2
    %i_r =l add %i, %run
    jmp @loop_next

@lit_scan_loop
    %lj =l phi @run_done %i, @lit_next %lj_new
    %c_jn =w csltl %lj, %n
    jnz %c_jn, @lit_c2, @lit_end

@lit_c2
    %lji =l sub %lj, %i
    %c_128 =w csltl %lji, 128
    jnz %c_128, @lit_c3, @lit_end

@lit_c3
    %ljp2 =l add %lj, 2
    %c_p2 =w csltl %ljp2, %n
    jnz %c_p2, @lit_run3, @lit_next

@lit_run3
    %plj =l add %src, %lj
    %b0 =w loadub %plj
    %plj1 =l add %plj, 1
    %b1 =w loadub %plj1
    %plj2 =l add %plj, 2
    %b2 =w loadub %plj2
    %e01 =w ceqw %b0, %b1
    %e12 =w ceqw %b1, %b2
    %r3 =w and %e01, %e12
    jnz %r3, @lit_end, @lit_next

@lit_next
    %lj_new =l add %lj, 1
    jmp @lit_scan_loop

@lit_end
    %L =l sub %lj, %i
    %need =l add %L, 1
    %sp_l =l sub %cap, %op
    %no_sp =w csltl %sp_l, %need
    jnz %no_sp, @ret_neg1, @do_lit

@do_lit
    %Lw =w truncl %L
    %lhdr =w sub %Lw, 1
    %outp_l =l add %out, %op
    storeb %lhdr, %outp_l
    %op_l1 =l add %op, 1
    jmp @copy_loop

@copy_loop
    %k =l phi @do_lit 0, @copy_inc %k_new
    %opc =l phi @do_lit %op_l1, @copy_inc %opc_new
    %ck =w csltl %k, %L
    jnz %ck, @copy_body, @copy_done

@copy_body
    %si =l add %i, %k
    %ps =l add %src, %si
    %bk =w loadub %ps
    %po =l add %out, %opc
    storeb %bk, %po
    jmp @copy_inc

@copy_inc
    %k_new =l add %k, 1
    %opc_new =l add %opc, 1
    jmp @copy_loop

@copy_done
    %op_l =l add %op, %need
    %i_l =l add %i, %L
    jmp @loop_next

@loop_next
    %i_next =l phi @do_rep %i_r, @copy_done %i_l
    %op_next =l phi @do_rep %op_r, @copy_done %op_l
    jmp @main_loop

@ret_neg1
    ret -1

@done
    ret %op
}
