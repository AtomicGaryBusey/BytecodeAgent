export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
  %cna =w ceql %na, 0
  jnz %cna, @ret0, @chk_nb
@chk_nb
  %cnb =w ceql %nb, 0
  jnz %cnb, @ret0, @begin
@begin
  %nab =l add %na, %nb
  jmp @zinit
@zinit
  %zi =l phi @begin 0, @zinit_body %zi_nxt
  %zi_lt =w csltl %zi, %nab
  jnz %zi_lt, @zinit_body, @outer_start
@zinit_body
  %zi_off =l mul %zi, 8
  %zi_ptr =l add %r, %zi_off
  storel 0, %zi_ptr
  %zi_nxt =l add %zi, 1
  jmp @zinit
@outer_start
  jmp @outer_hdr
@outer_hdr
  %oi =l phi @outer_start 0, @outer_inc %oi_nxt
  %oi_lt =w csltl %oi, %na
  jnz %oi_lt, @outer_body, @sig_start
@outer_body
  %ai_off =l mul %oi, 8
  %ai_ptr =l add %a, %ai_off
  %ai =l loadl %ai_ptr
  jmp @inner_hdr
@inner_hdr
  %ij =l phi @outer_body 0, @inner_inc %ij_nxt
  %ij_lt =w csltl %ij, %nb
  jnz %ij_lt, @inner_body, @outer_inc
@inner_body
  %bj_off =l mul %ij, 8
  %bj_ptr =l add %b, %bj_off
  %bj =l loadl %bj_ptr
  %ai_lo =l and %ai, 4294967295
  %ai_hi =l shr %ai, 32
  %bj_lo =l and %bj, 4294967295
  %bj_hi =l shr %bj, 32
  %p0 =l mul %ai_lo, %bj_lo
  %p1 =l mul %ai_lo, %bj_hi
  %p2 =l mul %ai_hi, %bj_lo
  %p3 =l mul %ai_hi, %bj_hi
  %mid =l add %p1, %p2
  %mid_ov_w =w cultl %mid, %p1
  %mid_ov =l extuw %mid_ov_w
  %mid_lo_part =l and %mid, 4294967295
  %lo_add =l shl %mid_lo_part, 32
  %prod_lo =l add %p0, %lo_add
  %lo_ov_w =w cultl %prod_lo, %p0
  %lo_ov =l extuw %lo_ov_w
  %mid_hi_part =l shr %mid, 32
  %mid_ov_sh =l shl %mid_ov, 32
  %prod_hi_1 =l add %p3, %mid_hi_part
  %prod_hi_2 =l add %prod_hi_1, %mid_ov_sh
  %prod_hi =l add %prod_hi_2, %lo_ov
  %rij_idx =l add %oi, %ij
  %rij_off =l mul %rij_idx, 8
  %rij_ptr =l add %r, %rij_off
  %rij_old =l loadl %rij_ptr
  %rij_new =l add %rij_old, %prod_lo
  storel %rij_new, %rij_ptr
  %c1_w =w cultl %rij_new, %rij_old
  %c1 =l extuw %c1_w
  %sum_hi =l add %prod_hi, %c1
  %rij1_idx =l add %rij_idx, 1
  %rij1_off =l mul %rij1_idx, 8
  %rij1_ptr =l add %r, %rij1_off
  %rij1_old =l loadl %rij1_ptr
  %rij1_new =l add %rij1_old, %sum_hi
  storel %rij1_new, %rij1_ptr
  %c2_w =w cultl %rij1_new, %rij1_old
  %c2 =l extuw %c2_w
  jmp @carry_hdr
@carry_hdr
  %ck =l phi @inner_body %rij1_idx, @carry_body %ck_nxt
  %cc =l phi @inner_body %c2, @carry_body %cc_nxt
  %cc_nz =w cnel %cc, 0
  jnz %cc_nz, @carry_body, @inner_inc
@carry_body
  %ck_nxt =l add %ck, 1
  %ck_off =l mul %ck_nxt, 8
  %ck_ptr =l add %r, %ck_off
  %ck_old =l loadl %ck_ptr
  %ck_new =l add %ck_old, %cc
  storel %ck_new, %ck_ptr
  %cc_nxt_w =w cultl %ck_new, %ck_old
  %cc_nxt =l extuw %cc_nxt_w
  jmp @carry_hdr
@inner_inc
  %ij_nxt =l add %ij, 1
  jmp @inner_hdr
@outer_inc
  %oi_nxt =l add %oi, 1
  jmp @outer_hdr
@sig_start
  %sk0 =l add %nab, -1
  jmp @sig_hdr
@sig_hdr
  %sk =l phi @sig_start %sk0, @sig_dec %sk_nxt
  %sk_neg =w csltl %sk, 0
  jnz %sk_neg, @ret0, @sig_chk
@sig_chk
  %sk_off =l mul %sk, 8
  %sk_ptr =l add %r, %sk_off
  %sk_val =l loadl %sk_ptr
  %sk_nz =w cnel %sk_val, 0
  jnz %sk_nz, @sig_found, @sig_dec
@sig_dec
  %sk_nxt =l add %sk, -1
  jmp @sig_hdr
@sig_found
  %sig =l add %sk, 1
  ret %sig
@ret0
  ret 0
}
