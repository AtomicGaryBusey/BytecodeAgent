lcs_len:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rbx, r8
    xor rcx, rcx
.Linit:
    cmp rcx, r15
    jg .Linit_done
    mov qword ptr [rbx + rcx*8], 0
    inc rcx
    jmp .Linit
.Linit_done:
    xor rbp, rbp
.Louter:
    cmp rbp, r13
    jge .Ldone
    xor r9, r9
    mov r10, 1
    movzx r8d, byte ptr [r12 + rbp]
.Linner:
    cmp r10, r15
    jg .Linner_done
    mov r11, qword ptr [rbx + r10*8]
    movzx eax, byte ptr [r14 + r10 - 1]
    cmp r8d, eax
    jne .Lno_match
    lea rax, [r9 + 1]
    mov qword ptr [rbx + r10*8], rax
    jmp .Lupdate_diag
.Lno_match:
    mov rax, qword ptr [rbx + r10*8 - 8]
    cmp r11, rax
    cmovge rax, r11
    mov qword ptr [rbx + r10*8], rax
.Lupdate_diag:
    mov r9, r11
    inc r10
    jmp .Linner
.Linner_done:
    inc rbp
    jmp .Louter
.Ldone:
    mov rax, qword ptr [rbx + r15*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
