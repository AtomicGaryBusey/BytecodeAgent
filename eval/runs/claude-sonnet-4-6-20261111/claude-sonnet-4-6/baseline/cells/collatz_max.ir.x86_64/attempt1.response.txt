export function l $collatz_max(l %n) {
@start
    jmp @outer_check
@outer_check
    %k =l phi @start 1, @outer_next %k_next
    %max =l phi @start 0, @outer_next %max_next
    %cond1 =w cslel %k, %n
    jnz %cond1, @inner_check, @done
@inner_check
    %x =l phi @outer_check %k, @step_done %xnew
    %steps =l phi @outer_check 0, @step_done %steps_inc
    %gt1 =w csgtl %x, 1
    jnz %gt1, @inner_body, @inner_done
@inner_body
    %rem =l rem %x, 2
    %is_even =w ceql %rem, 0
    jnz %is_even, @even_case, @odd_case
@even_case
    %xnew_e =l div %x, 2
    jmp @step_done
@odd_case
    %x3 =l mul %x, 3
    %xnew_o =l add %x3, 1
    jmp @step_done
@step_done
    %xnew =l phi @even_case %xnew_e, @odd_case %xnew_o
    %steps_inc =l add %steps, 1
    jmp @inner_check
@inner_done
    %gt_max =w csgtl %steps, %max
    jnz %gt_max, @update_max, @outer_next
@update_max
    jmp @outer_next
@outer_next
    %max_next =l phi @update_max %steps, @inner_done %max
    %k_next =l add %k, 1
    jmp @outer_check
@done
    ret %max
}
