packbits_encode:
    push rbp
    push rbx
    push r15
    push r14
    push r13
    push r12
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
    xor ebp, ebp
.Lmain:
    cmp rbx, r13
    jge .Ldone
    movzx eax, byte ptr [r12 + rbx]
    xor ecx, ecx
.Lrun:
    lea rdx, [rbx + rcx]
    cmp rdx, r13
    jge .Lrun_end
    cmp ecx, 128
    jge .Lrun_end
    movzx r8d, byte ptr [r12 + rdx]
    cmp r8b, al
    jne .Lrun_end
    inc ecx
    jmp .Lrun
.Lrun_end:
    cmp ecx, 3
    jge .Lrepeat
    mov r8, rbx
.Llit_scan:
    cmp r8, r13
    jge .Llit_scan_end
    mov r9, r8
    sub r9, rbx
    cmp r9, 128
    jge .Llit_scan_end
    lea r9, [r8 + 2]
    cmp r9, r13
    jge .Llit_inc
    movzx eax, byte ptr [r12 + r8]
    movzx r9d, byte ptr [r12 + r8 + 1]
    cmp r9b, al
    jne .Llit_inc
    movzx r9d, byte ptr [r12 + r8 + 2]
    cmp r9b, al
    jne .Llit_inc
    jmp .Llit_scan_end
.Llit_inc:
    inc r8
    jmp .Llit_scan
.Llit_scan_end:
    mov rcx, r8
    sub rcx, rbx
    lea rax, [rbp + rcx + 1]
    cmp rax, r15
    jg .Lneg1
    lea rax, [rcx - 1]
    mov byte ptr [r14 + rbp], al
    inc rbp
    xor r9, r9
.Llit_copy:
    cmp r9, rcx
    jge .Llit_copy_end
    movzx eax, byte ptr [r12 + rbx + r9]
    mov
