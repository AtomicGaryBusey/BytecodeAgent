packbits_encode:
    push rbp
    push rbx
    push r15
    push r14
    push r13
    push r12
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
    xor ebp, ebp
.Lmain_loop:
    cmp rbx, r13
    jge .Ldone
    movzx eax, byte ptr [r12 + rbx]
    mov ecx, 1
.Lrun_loop:
    cmp ecx, 128
    jge .Lrun_done
    lea rdx, [rbx + rcx]
    cmp rdx, r13
    jge .Lrun_done
    movzx r8d, byte ptr [r12 + rdx]
    cmp r8b, al
    jne .Lrun_done
    inc ecx
    jmp .Lrun_loop
.Lrun_done:
    cmp ecx, 3
    jge .Lrepeat_packet
    mov r8, rbx
.Llit_scan_loop:
    cmp r8, r13
    jge .Llit_scan_done
    mov r9, r8
    sub r9, rbx
    cmp r9, 128
    jge .Llit_scan_done
    lea r9, [r8 + 2]
    cmp r9, r13
    jge .Llit_inc
    movzx eax, byte ptr [r12 + r8]
    movzx r9d, byte ptr [r12 + r8 + 1]
    cmp r9b, al
    jne .Llit_inc
    movzx r9d, byte ptr [r12 + r8 + 2]
    cmp r9b, al
    jne .Llit_inc
    jmp .Llit_scan_done
.Llit_inc:
    inc r8
    jmp .Llit_scan_loop
.Llit_scan_done:
    mov rcx, r8
    sub rcx, rbx
    lea rax, [rbp + rcx + 1]
    cmp rax, r15
    jg .Lret_neg1
    lea rax, [rcx - 1]
    mov byte ptr [r14 + rbp], al
    inc rbp
    lea rax, [r12 + rbx]
    lea rdx, [r14 + rbp]
    xor r9, r9
.Llit_copy_loop:
    cmp r9, rcx
    jge .Llit_copy_done
    movzx r10d, byte ptr [rax + r9]
    mov byte ptr [rdx + r9], r10b
    inc r9
    jmp .Llit_copy_loop
.Llit_copy_done:
    add rbp, rcx
    mov rbx, r8
    jmp .Lmain_loop
.Lrepeat_packet:
    lea rax, [rbp + 2]
    cmp rax, r15
    jg .Lret_neg1
    mov rax, 257
    sub rax, rcx
    mov byte ptr [r14 + rbp], al
    inc rbp
    movzx eax, byte ptr [r12 + rbx]
    mov byte ptr [r14 + rbp], al
    inc rbp
    add rbx, rcx
    jmp .Lmain_loop
.Lret_neg1:
    mov rax, -1
    pop r12
    pop r13
    pop r14
    pop r15
    pop rbx
    pop rbp
    ret
.Ldone:
    mov rax, rbp
    pop r12
    pop r13
    pop r14
    pop r15
    pop rbx
    pop rbp
    ret
