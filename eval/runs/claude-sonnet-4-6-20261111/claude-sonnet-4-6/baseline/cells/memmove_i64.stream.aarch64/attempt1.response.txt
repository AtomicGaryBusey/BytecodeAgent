memmove_i64:
    cbz x2, .Ldone
    cmp x0, x1
    b.eq .Ldone
    b.hi .Lbackward
    mov x3, #0
.Lfwd_loop:
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    add x3, x3, #1
    cmp x3, x2
    b.lt .Lfwd_loop
    b .Ldone
.Lbackward:
    sub x3, x2, #1
.Lbwd_loop:
    ldr x4, [x1, x3, lsl #3]
    str x4, [x0, x3, lsl #3]
    subs x3, x3, #1
    b.ge .Lbwd_loop
.Ldone:
    mov x0, #0
    ret
