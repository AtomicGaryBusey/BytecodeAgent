export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    %ptr1 =l add %a, 8
    %nbytes =l mul %n, 8
    %endptr =l add %a, %nbytes
    %cmp0 =w csltl %ptr1, %endptr
    jnz %cmp0, @loop, @done
@loop
    %ptr =l phi @start %ptr1, @merge %ptr2
    %max =l phi @start %first, @merge %newmax
    %val =l loadl %ptr
    %gt =w csgtl %val, %max
    jnz %gt, @update, @keep
@update
    jmp @merge
@keep
    jmp @merge
@merge
    %newmax =l phi @update %val, @keep %max
    %ptr2 =l add %ptr, 8
    %cmp =w csltl %ptr2, %endptr
    jnz %cmp, @loop, @done
@done
    %result =l phi @start %first, @merge %newmax
    ret %result
}
