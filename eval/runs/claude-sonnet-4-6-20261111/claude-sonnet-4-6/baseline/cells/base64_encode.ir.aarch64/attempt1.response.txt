export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @loop
@loop
    %ip =l phi @start 0, @c3store %ip3
    %op =l phi @start %out, @c3store %op4
    %cond =w csltl %ip, %n
    jnz %cond, @process, @end
@process
    %pp0 =l add %src, %ip
    %b0 =l loadub %pp0
    %ip1 =l add %ip, 1
    %h1 =w csltl %ip1, %n
    jnz %h1, @getb1, @zerob1
@getb1
    %pp1 =l add %src, %ip1
    %b1a =l loadub %pp1
    jmp @haveb1
@zerob1
    jmp @haveb1
@haveb1
    %b1 =l phi @getb1 %b1a, @zerob1 0
    %ip2 =l add %ip, 2
    %h2 =w csltl %ip2, %n
    jnz %h2, @getb2, @zerob2
@getb2
    %pp2 =l add %src, %ip2
    %b2a =l loadub %pp2
    jmp @haveb2
@zerob2
    jmp @haveb2
@haveb2
    %b2 =l phi @getb2 %b2a, @zerob2 0
    %t0 =l shl %b0, 16
    %t1 =l shl %b1, 8
    %t01 =l or %t0, %t1
    %val =l or %t01, %b2
    %v0 =l shr %val, 18
    %r1 =l shr %val, 12
    %v1 =l and %r1, 63
    %r2 =l shr %val, 6
    %v2 =l and %r2, 63
    %v3 =l and %val, 63
    %v0lt26 =w csltl %v0, 26
    jnz %v0lt26, @c0r0, @c0r1
@c0r0
    %c0a =l add %v0, 65
    jmp @c0done
@c0r1
    %v0lt52 =w csltl %v0, 52
    jnz %v0lt52, @c0r2, @c0r3
@c0r2
    %c0b =l add %v0, 71
    jmp @c0done
@c0r3
    %v0lt62 =w csltl %v0, 62
    jnz %v0lt62, @c0r4, @c0r5
@c0r4
    %c0c =l sub %v0, 4
    jmp @c0done
@c0r5
    %v0eq62 =w ceql %v0, 62
    jnz %v0eq62, @c0r6, @c0r7
@c0r6
    jmp @c0done
@c0r7
    jmp @c0done
@c0done
    %c0 =l phi @c0r0 %c0a, @c0r2 %c0b, @c0r4 %c0c, @c0r6 43, @c0r7 47
    storeb %c0, %op
    %op1 =l add %op, 1
    %v1lt26 =w csltl %v1, 26
    jnz %v1lt26, @c1r0, @c1r1
@c1r0
    %c1a =l add %v1, 65
    jmp @c1done
@c1r1
    %v1lt52 =w csltl %v1, 52
    jnz %v1lt52, @c1r2, @c1r3
@c1r2
    %c1b =l add %v1, 71
    jmp @c1done
@c1r3
    %v1lt62 =w csltl %v1, 62
    jnz %v1lt62, @c1r4, @c1r5
@c1r4
    %c1c =l sub %v1, 4
    jmp @c1done
@c1r5
    %v1eq62 =w ceql %v1, 62
    jnz %v1eq62, @c1r6, @c1r7
@c1r6
    jmp @c1done
@c1r7
    jmp @c1done
@c1done
    %c1 =l phi @c1r0 %c1a, @c1r2 %c1b, @c1r4 %c1c, @c1r6 43, @c1r7 47
    storeb %c1, %op1
    %op2 =l add %op1, 1
    jnz %h1, @enc_c2, @pad_c2
@enc_c2
    %v2lt26 =w csltl %v2, 26
    jnz %v2lt26, @c2r0, @c2r1
@c2r0
    %c2a =l add %v2, 65
    jmp @c2done
@c2r1
    %v2lt52 =w csltl %v2, 52
    jnz %v2lt52, @c2r2, @c2r3
@c2r2
    %c2b =l add %v2, 71
    jmp @c2done
@c2r3
    %v2lt62 =w csltl %v2, 62
    jnz %v2lt62, @c2r4, @c2r5
@c2r4
    %c2c =l sub %v2, 4
    jmp @c2done
@c2r5
    %v2eq62 =w ceql %v2, 62
    jnz %v2eq62, @c2r6, @c2r7
@c2r6
    jmp @c2done
@c2r7
    jmp @c2done
@c2done
    %c2enc =l phi @c2r0 %c2a, @c2r2 %c2b, @c2r4 %c2c, @c2r6 43, @c2r7 47
    jmp @c2store
@pad_c2
    jmp @c2store
@c2store
    %c2 =l phi @c2done %c2enc, @pad_c2 61
    storeb %c2, %op2
    %op3 =l add %op2, 1
    jnz %h2, @enc_c3, @pad_c3
@enc_c3
    %v3lt26 =w csltl %v3, 26
    jnz %v3lt26, @c3r0, @c3r1
@c3r0
    %c3a =l add %v3, 65
    jmp @c3done
@c3r1
    %v3lt52 =w csltl %v3, 52
    jnz %v3lt52, @c3r2, @c3r3
@c3r2
    %c3b =l add %v3, 71
    jmp @c3done
@c3r3
    %v3lt62 =w csltl %v3, 62
    jnz %v3lt62, @c3r4, @c3r5
@c3r4
    %c3c =l sub %v3, 4
    jmp @c3done
@c3r5
    %v3eq62 =w ceql %v3, 62
    jnz %v3eq62, @c3r6, @c3r7
@c3r6
    jmp @c3done
@c3r7
    jmp @c3done
@c3done
    %c3enc =l phi @c3r0 %c3a, @c3r2 %c3b, @c3r4 %c3c, @c3r6 43, @c3r7 47
    jmp @c3store
@pad_c3
    jmp @c3store
@c3store
    %c3 =l phi @c3done %c3enc, @pad_c3 61
    storeb %c3, %op3
    %op4 =l add %op3, 1
    %ip3 =l add %ip, 3
    jmp @loop
@end
    %result =l sub %op, %out
    ret %result
}
