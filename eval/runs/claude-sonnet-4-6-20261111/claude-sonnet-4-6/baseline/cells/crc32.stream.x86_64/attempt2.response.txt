crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
    xor rcx, rcx
.Lbyte_loop:
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit_loop:
    shr eax, 1
    jnc .Lno_xor
    xor eax, 0xEDB88320
.Lno_xor:
    dec r8d
    jnz .Lbit_loop
    inc rcx
    cmp rcx, rsi
    jl .Lbyte_loop
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
