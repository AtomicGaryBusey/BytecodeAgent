export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %a0      =l loadl %a
    %y0      =l call %f(l %a0)
    %o0      =l and %y0, 1
    jmp @loop
@loop
    %i       =l phi @start 1,     @ama %inext
    %min     =l phi @start %y0,   @ama %minnext
    %max     =l phi @start %y0,   @ama %maxnext
    %sum     =l phi @start %y0,   @ama %sumnext
    %cnt     =l phi @start %o0,   @ama %cntnext
    %cmp     =w csltl %i, %n
    jnz %cmp, @body, @done
@body
    %off     =l mul %i, 8
    %ptr     =l add %a, %off
    %ai      =l loadl %ptr
    %yi      =l call %f(l %ai)
    %lt      =w csltl %yi, %min
    jnz %lt, @ymi, @nmi
@ymi
    jmp @ami
@nmi
    jmp @ami
@ami
    %minnext =l phi @ymi %yi, @nmi %min
    %gt      =w csgtl %yi, %max
    jnz %gt, @yma, @nma
@yma
    jmp @ama
@nma
    jmp @ama
@ama
    %maxnext =l phi @yma %yi, @nma %max
    %sumnext =l add %sum, %yi
    %yod     =l and %yi, 1
    %cntnext =l add %cnt, %yod
    %inext   =l add %i, 1
    jmp @loop
@done
    storel %min, %out
    %p1      =l add %out, 8
    storel %max, %p1
    %p2      =l add %out, 16
    storel %sum, %p2
    ret %cnt
}
