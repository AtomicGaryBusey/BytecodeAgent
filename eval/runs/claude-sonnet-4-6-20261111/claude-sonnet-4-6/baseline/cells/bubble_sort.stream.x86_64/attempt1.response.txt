bubble_sort:
    cmp rsi, 1
    jle .Ldone
    xor rcx, rcx
.Louter:
    mov rax, rsi
    dec rax
    cmp rcx, rax
    jge .Ldone
    xor rdx, rdx
    mov r9, rsi
    sub r9, rcx
    dec r9
.Linner:
    cmp rdx, r9
    jge .Lnext_outer
    mov rax, qword ptr [rdi + rdx*8]
    mov r8, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r8
    jle .Lno_swap
    mov qword ptr [rdi + rdx*8], r8
    mov qword ptr [rdi + rdx*8 + 8], rax
.Lno_swap:
    inc rdx
    jmp .Linner
.Lnext_outer:
    inc rcx
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
