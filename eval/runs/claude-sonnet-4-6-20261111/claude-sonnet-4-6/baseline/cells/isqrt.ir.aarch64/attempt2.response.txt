export function l $isqrt(l %n) {
@start
    jmp @check
@check
    %lo =l phi @start 0, @back_lo %nlo, @back_hi %ulo
    %hi =l phi @start 4294967295, @back_lo %uhi, @back_hi %nhi
    %cond =w cultl %lo, %hi
    jnz %cond, @body, @done
@body
    %sum =l add %lo, %hi
    %sum1 =l add %sum, 1
    %mid =l udiv %sum1, 2
    %sq =l mul %mid, %mid
    %cmp =w culel %sq, %n
    jnz %cmp, @back_lo, @back_hi
@back_lo
    %nlo =l copy %mid
    %uhi =l copy %hi
    jmp @check
@back_hi
    %ulo =l copy %lo
    %nhi =l sub %mid, 1
    jmp @check
@done
    ret %lo
}
