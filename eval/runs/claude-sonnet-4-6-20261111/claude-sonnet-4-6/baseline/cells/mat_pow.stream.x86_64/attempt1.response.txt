mat_pow:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rbp, r8
    xor rcx, rcx
.Linit_i:
    cmp rcx, r13
    jge .Linit_done
    xor r8, r8
.Linit_j:
    cmp r8, r13
    jge .Linit_j_end
    mov r9, rcx
    imul r9, r13
    add r9, r8
    xor r10, r10
    cmp rcx, r8
    sete r10b
    mov qword ptr [r15 + r9*8], r10
    inc r8
    jmp .Linit_j
.Linit_j_end:
    inc rcx
    jmp .Linit_i
.Linit_done:
    mov rax, r13
    imul rax, r13
    xor rcx, rcx
.Lcopy_a:
    cmp rcx, rax
    jge .Lcopy_a_done
    mov r9, qword ptr [r12 + rcx*8]
    mov qword ptr [rbp + rcx*8], r9
    inc rcx
    jmp .Lcopy_a
.Lcopy_a_done:
    xor rbx, rbx
.Lsam_loop:
    cmp rbx, 7
    jge .Lsam_done
    bt r14, rbx
    jnc .Lsam_no_mul
    mov rdi, r15
    mov rsi, rbp
    mov rax, r13
    imul rax, r13
    lea rdx, [rbp + rax*8]
    call .Lmatmul
    mov rax, r13
    imul rax, r13
    lea r9, [rbp + rax*8]
    xor rcx, rcx
.Lmul_cb:
    cmp rcx, rax
    jge .Lmul_cb_done
    mov r10, qword ptr [r9 + rcx*8]
    mov qword ptr [r15 + rcx*8], r10
    mov qword ptr [r9 + rcx*8], 0
    inc rcx
    jmp .Lmul_cb
.Lmul_cb_done:
.Lsam_no_mul:
    cmp rbx, 6
    jge .Lsam_no_sq
    mov rdi, rbp
    mov rsi, rbp
    mov rax, r13
    imul rax, r13
    lea rdx, [rbp + rax*8]
    call .Lmatmul
    mov rax, r13
    imul rax, r13
    lea r9, [rbp + rax*8]
    xor rcx, rcx
.Lsq_cb:
    cmp rcx, rax
    jge .Lsq_cb_done
    mov r10, qword ptr [r9 + rcx*8]
    mov qword ptr [rbp + rcx*8], r10
    mov qword ptr [r9 + rcx*8], 0
    inc rcx
    jmp .Lsq_cb
.Lsq_cb_done:
.Lsam_no_sq:
    inc rbx
    jmp .Lsam_loop
.Lsam_done:
    mov rax, r13
    imul rax, r13
    xor rcx, rcx
.Lzero_base:
    cmp rcx, rax
    jge .Lzero_done
    mov qword ptr [rbp + rcx*8], 0
    inc rcx
    jmp .Lzero_base
.Lzero_done:
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
.Lmatmul:
    xor rcx, rcx
.Lmm_i:
    cmp rcx, r13
    jge .Lmm_done
    xor r8, r8
.Lmm_j:
    cmp r8, r13
    jge .Lmm_j_end
    xor r10, r10
    xor r9, r9
.Lmm_t:
    cmp r9, r13
    jge .Lmm_t_end
    mov r11, rcx
    imul r11, r13
    add r11, r9
    mov rax, qword ptr [rdi + r11*8]
    mov r11, r9
    imul r11, r13
    add r11, r8
    imul rax, qword ptr [rsi + r11*8]
    add r10, rax
    inc r9
    jmp .Lmm_t
.Lmm_t_end:
    mov r11, rcx
    imul r11, r13
    add r11, r8
    mov qword ptr [rdx + r11*8], r10
    inc r8
    jmp .Lmm_j
.Lmm_j_end:
    inc rcx
    jmp .Lmm_i
.Lmm_done:
    ret
