export function l $map_sum(l %a, l %n, l %f) {
@start
    %sum =l copy 0
    %i =l copy 0
    jmp @loop
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %fval =l call %f(l %val)
    %sum =l add %sum, %fval
    %i =l add %i, 1
    jmp @loop
@end
    ret %sum
}
