export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %bit0 =l and %y0, 1
    %isodd0 =w cnel %bit0, 0
    %cnt0 =l extsw %isodd0
    jmp @loop
@loop
    %i =l phi @start 1, @maxmerge %inext
    %min =l phi @start %y0, @maxmerge %minnew
    %max =l phi @start %y0, @maxmerge %maxnew
    %sum =l phi @start %y0, @maxmerge %sumnew
    %cnt =l phi @start %cnt0, @maxmerge %cntnew
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %ai =l loadl %ptr
    %yi =l call %f(l %ai)
    %lt =w csltl %yi, %min
    jnz %lt, @minyes, @minno
@minyes
    jmp @minmerge
@minno
    jmp @minmerge
@minmerge
    %minnew =l phi @minyes %yi, @minno %min
    %gt =w csgtl %yi, %max
    jnz %gt, @maxyes, @maxno
@maxyes
    jmp @maxmerge
@maxno
    jmp @maxmerge
@maxmerge
    %maxnew =l phi @maxyes %yi, @maxno %max
    %sumnew =l add %sum, %yi
    %bit =l and %yi, 1
    %isodd =w cnel %bit, 0
    %addcnt =l extsw %isodd
    %cntnew =l add %cnt, %addcnt
    %inext =l add %i, 1
    jmp @loop
@end
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %cnt
}
