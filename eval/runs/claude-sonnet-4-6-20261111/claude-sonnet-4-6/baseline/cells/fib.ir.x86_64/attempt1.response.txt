export function l $fib(l %n) {
@start
    %cmp0 =w ceql %n, 0
    jnz %cmp0, @ret0, @loop_init
@ret0
    ret 0
@loop_init
    jmp @loop_cond
@loop_cond
    %a =l phi @loop_init 0, @loop_body %a_new
    %b =l phi @loop_init 1, @loop_body %b_new
    %i =l phi @loop_init 1, @loop_body %i_new
    %cmp =w csltl %i, %n
    jnz %cmp, @loop_body, @loop_end
@loop_body
    %a_new =l copy %b
    %b_new =l add %a, %b
    %i_new =l add %i, 1
    jmp @loop_cond
@loop_end
    ret %b
}
