export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %np1 =l add %n, 1
    jmp @init_check
@init_check
    %j_init =l phi @start 0, @init_body %j_init_next
    %cond_init =w csltl %j_init, %np1
    jnz %cond_init, @init_body, @outer_check
@init_body
    %rp_init =l mul %j_init, 8
    %rp_init2 =l add %row, %rp_init
    storel 0, %rp_init2
    %j_init_next =l add %j_init, 1
    jmp @init_check
@outer_check
    %i =l phi @init_check 0, @inner_done %i_next
    %cond_outer =w csltl %i, %m
    jnz %cond_outer, @outer_body, @outer_done
@outer_body
    %xip =l add %x, %i
    %xi =w loadub %xip
    jmp @inner_check
@inner_check
    %j =l phi @outer_body 1, @inner_store %j_next
    %diag =l phi @outer_body 0, @inner_store %diag_next
    %cond_inner =w csltl %j, %np1
    jnz %cond_inner, @inner_body, @inner_done
@inner_body
    %rjoff =l mul %j, 8
    %rjp =l add %row, %rjoff
    %up =l loadl %rjp
    %jm1 =l sub %j, 1
    %yjp =l add %y, %jm1
    %yj =w loadub %yjp
    %eq =w ceqw %xi, %yj
    jnz %eq, @inner_eq, @inner_neq
@inner_eq
    %new_val_eq =l add %diag, 1
    jmp @inner_store
@inner_neq
    %rj1off =l mul %jm1, 8
    %rj1p =l add %row, %rj1off
    %rj1 =l loadl %rj1p
    %gt =w csgtl %up, %rj1
    jnz %gt, @inner_gt, @inner_le
@inner_gt
    jmp @inner_max_done
@inner_le
    jmp @inner_max_done
@inner_max_done
    %new_val_neq =l phi @inner_gt %up, @inner_le %rj1
    jmp @inner_store
@inner_store
    %new_val =l phi @inner_eq %new_val_eq, @inner_max_done %new_val_neq
    storel %new_val, %rjp
    %diag_next =l copy %up
    %j_next =l add %j, 1
    jmp @inner_check
@inner_done
    %i_next =l add %i, 1
    jmp @outer_check
@outer_done
    %rnoff =l mul %n, 8
    %rnp =l add %row, %rnoff
    %result =l loadl %rnp
    ret %result
}
