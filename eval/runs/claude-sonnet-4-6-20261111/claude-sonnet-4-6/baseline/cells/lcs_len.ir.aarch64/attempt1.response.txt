export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    jmp @init_hd
@init_hd
    %ji =l phi @start 0, @init_body %ji_next
    %np1 =l add %n, 1
    %icond =w csltl %ji, %np1
    jnz %icond, @init_body, @init_done
@init_body
    %ji_off =l mul %ji, 8
    %ji_ptr =l add %row, %ji_off
    storel 0, %ji_ptr
    %ji_next =l add %ji, 1
    jmp @init_hd
@init_done
    jmp @outer_hd
@outer_hd
    %i =l phi @init_done 0, @inner_done %i_next
    %ocond =w csltl %i, %m
    jnz %ocond, @inner_hd, @outer_done
@inner_hd
    %j =l phi @outer_hd 1, @store %j_next
    %diag =l phi @outer_hd 0, @store %up
    %lim =l add %n, 1
    %jcond =w csltl %j, %lim
    jnz %jcond, @ibody, @inner_done
@ibody
    %rj_off =l mul %j, 8
    %rj_ptr =l add %row, %rj_off
    %up =l loadl %rj_ptr
    %xi_ptr =l add %x, %i
    %xi =w loadub %xi_ptr
    %jm1 =l sub %j, 1
    %yj_ptr =l add %y, %jm1
    %yj =w loadub %yj_ptr
    %eq =w ceqw %xi, %yj
    jnz %eq, @match, @nomatch
@match
    %val_m =l add %diag, 1
    jmp @store
@nomatch
    %rjm1_off =l mul %jm1, 8
    %rjm1_ptr =l add %row, %rjm1_off
    %rjm1 =l loadl %rjm1_ptr
    %gt =w csgtl %up, %rjm1
    jnz %gt, @useup, @userj
@useup
    jmp @store
@userj
    jmp @store
@store
    %val =l phi @match %val_m, @useup %up, @userj %rjm1
    storel %val, %rj_ptr
    %j_next =l add %j, 1
    jmp @inner_hd
@inner_done
    %i_next =l add %i, 1
    jmp @outer_hd
@outer_done
    %rn_off =l mul %n, 8
    %rn_ptr =l add %row, %rn_off
    %result =l loadl %rn_ptr
    ret %result
}
