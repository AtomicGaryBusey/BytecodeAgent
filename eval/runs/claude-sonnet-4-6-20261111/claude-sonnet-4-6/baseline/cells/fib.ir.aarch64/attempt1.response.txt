export function l $fib(l %n) {
@start
    jmp @loop
@loop
    %a =l phi @start 0, @body %a2
    %b =l phi @start 1, @body %b2
    %i =l phi @start 0, @body %i2
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %temp =l add %a, %b
    %a2 =l copy %b
    %b2 =l copy %temp
    %i2 =l add %i, 1
    jmp @loop
@end
    ret %a
}
