export function l $reverse(l %a, l %n) {
@start
    %right0 =l sub %n, 1
    jmp @loop
@loop
    %left =l phi @start 0, @body %left_next
    %right =l phi @start %right0, @body %right_next
    %cond =w csltl %left, %right
    jnz %cond, @body, @end
@body
    %left8 =l mul %left, 8
    %addr_left =l add %a, %left8
    %right8 =l mul %right, 8
    %addr_right =l add %a, %right8
    %tmp =l loadl %addr_left
    %val_right =l loadl %addr_right
    storel %val_right, %addr_left
    storel %tmp, %addr_right
    %left_next =l add %left, 1
    %right_next =l sub %right, 1
    jmp @loop
@end
    ret 0
}
