export function l $qsort_pairs(l %a, l %n) {
@start
    %lt2 =w csltl %n, 2
    jnz %lt2, @ret0, @setup

@ret0
    ret 0

@setup
    %half =l div %n, 2
    %halfx16 =l mul %half, 16
    %pivptr =l add %a, %halfx16
    %nm1 =l sub %n, 1
    %nm1x16 =l mul %nm1, 16
    %lastptr =l add %a, %nm1x16
    %pk0 =l loadl %pivptr
    %pvaddr0 =l add %pivptr, 8
    %pv0 =l loadl %pvaddr0
    %lk0 =l loadl %lastptr
    %lvaddr0 =l add %lastptr, 8
    %lv0 =l loadl %lvaddr0
    storel %pk0, %lastptr
    storel %pv0, %lvaddr0
    storel %lk0, %pivptr
    storel %lv0, %pvaddr0
    %pivk =l loadl %lastptr
    %pivvaddr =l add %lastptr, 8
    %pivv =l loadl %pivvaddr
    jmp @loophead

@loophead
    %i =l phi @setup 0, @do_swap %inew, @no_swap %i
    %j =l phi @setup 0, @do_swap %jnew_ds, @no_swap %jnew_ns
    %jlt =w csltl %j, %nm1
    jnz %jlt, @loop_body, @loop_done

@loop_body
    %jx16 =l mul %j, 16
    %jptr =l add %a, %jx16
    %jk =l loadl %jptr
    %jvaddr =l add %jptr, 8
    %jv =l loadl %jvaddr
    %jkgt =w csgtl %jk, %pivk
    %jkeq =w ceql %jk, %pivk
    %jvgt =w csgtl %jv, %pivv
    %eqkgt =w and %jkeq, %jvgt
    %jgtpiv =w or %jkgt, %eqkgt
    jnz %jgtpiv, @no_swap, @do_swap

@do_swap
    %ix16 =l mul %i, 16
    %iptr =l add %a, %ix16
    %ik =l loadl %iptr
    %ivaddr =l add %iptr, 8
    %iv =l loadl %ivaddr
    storel %jk, %iptr
    storel %jv, %ivaddr
    storel %ik, %jptr
    storel %iv, %jvaddr
    %inew =l add %i, 1
    %jnew_ds =l add %j, 1
    jmp @loophead

@no_swap
    %jnew_ns =l add %j, 1
    jmp @loophead

@loop_done
    %ix16f =l mul %i, 16
    %iptrf =l add %a, %ix16f
    %ikf =l loadl %iptrf
    %ivaddrf =l add %iptrf, 8
    %ivf =l loadl %ivaddrf
    %lastvaddr =l add %lastptr, 8
    storel %ikf, %lastptr
    storel %ivf, %lastvaddr
    storel %pivk, %iptrf
    storel %pivv, %ivaddrf
    %r1 =l call $qsort_pairs(l %a, l %i)
    %ip1 =l add %i, 1
    %roff =l mul %ip1, 16
    %rptr =l add %a, %roff
    %rn =l sub %n, %ip1
    %r2 =l call $qsort_pairs(l %rptr, l %rn)
    ret 0
}
