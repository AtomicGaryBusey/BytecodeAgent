export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %cmp0 =w ceql %n, 0
    jnz %cmp0, @done, @check_dir

@check_dir
    %fwd =w culel %dst, %src
    jnz %fwd, @fwd_init, @bwd_init

@fwd_init
    jmp @fwd_loop

@fwd_loop
    %i =l phi @fwd_init 0, @fwd_step %inext
    %cont =w csltl %i, %n
    jnz %cont, @fwd_body, @done

@fwd_body
    %off =l mul %i, 8
    %sp =l add %src, %off
    %dp =l add %dst, %off
    %v =l loadl %sp
    storel %v, %dp
    jmp @fwd_step

@fwd_step
    %inext =l add %i, 1
    jmp @fwd_loop

@bwd_init
    %j0 =l sub %n, 1
    jmp @bwd_loop

@bwd_loop
    %j =l phi @bwd_init %j0, @bwd_step %jnext
    %cont2 =w csgel %j, 0
    jnz %cont2, @bwd_body, @done

@bwd_body
    %off2 =l mul %j, 8
    %sp2 =l add %src, %off2
    %dp2 =l add %dst, %off2
    %v2 =l loadl %sp2
    storel %v2, %dp2
    jmp @bwd_step

@bwd_step
    %jnext =l sub %j, 1
    jmp @bwd_loop

@done
    ret 0
}
