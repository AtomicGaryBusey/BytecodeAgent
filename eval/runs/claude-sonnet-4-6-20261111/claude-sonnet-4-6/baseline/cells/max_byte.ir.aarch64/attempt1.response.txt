export function l $max_byte(l %p, l %n) {
@start
    jmp @check
@check
    %i =l phi @start 0, @next %i2
    %max =l phi @start 0, @next %max2
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %gt =w csgtl %b, %max
    jnz %gt, @update, @skip
@update
    jmp @next
@skip
    jmp @next
@next
    %max2 =l phi @update %b, @skip %max
    %i2 =l add %i, 1
    jmp @check
@done
    ret %max
}
