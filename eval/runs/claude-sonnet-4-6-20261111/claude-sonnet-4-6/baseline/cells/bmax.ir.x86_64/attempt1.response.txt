export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    jmp @check
@check
    %i =l phi @start 1, @next %i1
    %max =l phi @start %first, @next %max1
    %cond =w csltl %i, %n
    jnz %cond, @loop, @done
@loop
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %gt =w csgtl %val, %max
    jnz %gt, @update, @skip
@update
    jmp @next
@skip
    jmp @next
@next
    %max1 =l phi @update %val, @skip %max
    %i1 =l add %i, 1
    jmp @check
@done
    ret %max
}
