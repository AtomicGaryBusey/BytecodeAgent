export function l $map_sum(l %a, l %n, l %f) {
@start
    %sum =l copy 0
    %i =l copy 0
    jmp @loop
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %elem =l loadl %ptr
    %res =l call %f(l %elem)
    %sum =l add %sum, %res
    %i =l add %i, 1
    jmp @loop
@done
    ret %sum
}
