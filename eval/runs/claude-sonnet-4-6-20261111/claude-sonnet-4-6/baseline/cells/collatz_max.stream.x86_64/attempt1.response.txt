collatz_max:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    xor r14, r14
    mov r13, 1
.Lloop_k:
    cmp r13, r12
    jg .Ldone
    mov r15, r13
    xor rbx, rbx
.Lcollatz_loop:
    cmp r15, 1
    je .Lcollatz_done
    test r15, 1
    jz .Leven
    lea r15, [r15 + r15*2 + 1]
    inc rbx
    jmp .Lcollatz_loop
.Leven:
    sar r15, 1
    inc rbx
    jmp .Lcollatz_loop
.Lcollatz_done:
    cmp rbx, r14
    jle .Lnext_k
    mov r14, rbx
.Lnext_k:
    inc r13
    jmp .Lloop_k
.Ldone:
    mov rax, r14
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
