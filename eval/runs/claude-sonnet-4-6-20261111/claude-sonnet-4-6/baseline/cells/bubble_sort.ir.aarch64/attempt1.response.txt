export function l $bubble_sort(l %a, l %n) {
@start
    jmp @outer_loop
@outer_loop
    %i =l phi @start 0, @outer_inc %i_next
    %cond_outer =w csltl %i, %n
    jnz %cond_outer, @inner_init, @done
@inner_init
    %limit =l sub %n, %i
    %limit2 =l sub %limit, 1
    jmp @inner_loop
@inner_loop
    %j =l phi @inner_init 0, @after_body %j1
    %cond_inner =w csltl %j, %limit2
    jnz %cond_inner, @compare, @inner_done
@compare
    %off_j =l mul %j, 8
    %ptr_j =l add %a, %off_j
    %val_j =l loadl %ptr_j
    %j1 =l add %j, 1
    %off_j1 =l mul %j1, 8
    %ptr_j1 =l add %a, %off_j1
    %val_j1 =l loadl %ptr_j1
    %need_swap =w csgtl %val_j, %val_j1
    jnz %need_swap, @do_swap, @after_body
@do_swap
    storel %val_j1, %ptr_j
    storel %val_j, %ptr_j1
    jmp @after_body
@after_body
    jmp @inner_loop
@inner_done
    jmp @outer_inc
@outer_inc
    %i_next =l add %i, 1
    jmp @outer_loop
@done
    ret 0
}
