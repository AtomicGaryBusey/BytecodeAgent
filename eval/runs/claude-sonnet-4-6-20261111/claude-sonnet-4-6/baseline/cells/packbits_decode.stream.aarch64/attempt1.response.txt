packbits_decode:
    mov x4, #0
    mov x5, #0
.Lloop:
    cmp x4, x1
    b.ge .Ldone
    ldrb w6, [x0, x4]
    add x4, x4, #1
    sxtb x6, w6
    cmn x6, #128
    b.eq .Lloop
    tst x6, x6
    b.mi .Lrepeat
    add x7, x4, x6
    add x7, x7, #1
    cmp x7, x1
    b.hi .Lerror_src
    add x6, x6, #1
.Lliteral_copy:
    cbz x6, .Lloop
    cmp x5, x3
    b.ge .Lerror_cap
    ldrb w7, [x0, x4]
    strb w7, [x2, x5]
    add x4, x4, #1
    add x5, x5, #1
    sub x6, x6, #1
    b .Lliteral_copy
.Lrepeat:
    add x7, x4, #1
    cmp x7, x1
    b.hi .Lerror_src
    ldrb w7, [x0, x4]
    add x4, x4, #1
    neg x6, x6
    add x6, x6, #1
.Lrepeat_write:
    cbz x6, .Lloop
    cmp x5, x3
    b.ge .Lerror_cap
    strb w7, [x2, x5]
    add x5, x5, #1
    sub x6, x6, #1
    b .Lrepeat_write
.Ldone:
    mov x0, x5
    ret
.Lerror_src:
    mov x0, #-2
    ret
.Lerror_cap:
    mov x0, #-1
    ret
