export function l $qsort_pairs(l %a, l %n) {
@start
  %le1 =w csltl %n, 2
  jnz %le1, @ret0, @proceed
@ret0
  ret 0
@proceed
  %nm1 =l sub %n, 1
  %nm1x16 =l mul %nm1, 16
  %pvptr =l add %a, %nm1x16
  %pvkey =l loadl %pvptr
  %pvvptr =l add %pvptr, 8
  %pvval =l loadl %pvvptr
  jmp @loop
@loop
  %i =l phi @proceed 0, @swap %inew, @body %ibody
  %j =l phi @proceed 0, @swap %jnew, @body %jnext
  %jlt =w csltl %j, %nm1
  jnz %jlt, @body, @done
@body
  %joff =l mul %j, 16
  %jp =l add %a, %joff
  %jk =l loadl %jp
  %jvp =l add %jp, 8
  %jv =l loadl %jvp
  %klt =w csltl %jk, %pvkey
  %keq =w ceql %jk, %pvkey
  %vle =w cslel %jv, %pvval
  %kveq =w and %keq, %vle
  %le =w or %klt, %kveq
  %ibody =l copy %i
  %jnext =l add %j, 1
  jnz %le, @swap, @loop
@swap
  %ioff =l mul %i, 16
  %ip =l add %a, %ioff
  %ik =l loadl %ip
  %ivp =l add %ip, 8
  %iv =l loadl %ivp
  storel %jk, %ip
  storel %jv, %ivp
  storel %ik, %jp
  storel %iv, %jvp
  %inew =l add %i, 1
  %jnew =l copy %jnext
  jmp @loop
@done
  %ioff2 =l mul %i, 16
  %ip2 =l add %a, %ioff2
  %ik2 =l loadl %ip2
  %ivp2 =l add %ip2, 8
  %iv2 =l loadl %ivp2
  storel %pvkey, %ip2
  storel %pvval, %ivp2
  storel %ik2, %pvptr
  storel %iv2, %pvvptr
  %r1 =l call $qsort_pairs(l %a, l %i)
  %ip1 =l add %i, 1
  %ip1x16 =l mul %ip1, 16
  %rp =l add %a, %ip1x16
  %rn =l sub %n, %ip1
  %r2 =l call $qsort_pairs(l %rp, l %rn)
  ret 0
}
