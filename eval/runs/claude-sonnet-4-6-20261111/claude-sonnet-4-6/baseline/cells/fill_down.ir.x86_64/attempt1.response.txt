export function l $fill_down(l %a, l %n, l %start) {
@entry
    %i =l copy 0
    jmp @loop
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %val =l sub %start, %i
    %off =l mul %i, 8
    %ptr =l add %a, %off
    storel %val, %ptr
    %i =l add %i, 1
    jmp @loop
@done
    ret %n
}
