export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %si_p =l alloc8 8
    %oi_p =l alloc8 8
    %ri_p =l alloc8 8
    storel 0, %si_p
    storel 0, %oi_p
    jmp @pkt

@pkt
    %si =l loadl %si_p
    %c1 =w csltl %si, %n
    jnz %c1, @read_hdr, @done

@read_hdr
    %hp =l add %src, %si
    %h =w loadsb %hp
    %si2 =l add %si, 1
    storel %si2, %si_p
    %chknop =w ceqw %h, -128
    jnz %chknop, @pkt, @dispatch

@dispatch
    %chklit =w csgew %h, 0
    jnz %chklit, @lit_chk, @rep_chk

@lit_chk
    %hl =l extsw %h
    %cnt =l add %hl, 1
    %si3 =l loadl %si_p
    %sneed =l add %si3, %cnt
    %oklit =w cslel %sneed, %n
    jnz %oklit, @lit_run, @err_src

@lit_run
    %si4 =l loadl %si_p
    %send =l add %si4, %cnt
    jmp @lit_lp

@lit_lp
    %sic =l loadl %si_p
    %lcc =w csltl %sic, %send
    jnz %lcc, @lit_bdy, @lit_end

@lit_bdy
    %oic =l loadl %oi_p
    %owc =w csltl %oic, %cap
    jnz %owc, @lit_wr, @err_cap

@lit_wr
    %bp =l add %src, %sic
    %bv =w loadub %bp
    %op =l add %out, %oic
    storeb %bv, %op
    %sin =l add %sic, 1
    storel %sin, %si_p
    %oin =l add %oic, 1
    storel %oin, %oi_p
    jmp @lit_lp

@lit_end
    jmp @pkt

@rep_chk
    %si5 =l loadl %si_p
    %okrep =w csltl %si5, %n
    jnz %okrep, @rep_run, @err_src

@rep_run
    %si6 =l loadl %si_p
    %rp0 =l add %src, %si6
    %rb =w loadub %rp0
    %si7 =l add %si6, 1
    storel %si7, %si_p
    %hl2 =l extsw %h
    %cntr =l sub 1, %hl2
    storel 0, %ri_p
    jmp @rep_lp

@rep_lp
    %ric =l loadl %ri_p
    %rcc =w csltl %ric, %cntr
    jnz %rcc, @rep_bdy, @rep_end

@rep_bdy
    %oic2 =l loadl %oi_p
    %owc2 =w csltl %oic2, %cap
    jnz %owc2, @rep_wr, @err_cap

@rep_wr
    %op2 =l add %out, %oic2
    storeb %rb, %op2
    %oin2 =l add %oic2, 1
    storel %oin2, %oi_p
    %rin =l add %ric, 1
    storel %rin, %ri_p
    jmp @rep_lp

@rep_end
    jmp @pkt

@err_src
    ret -2

@err_cap
    ret -1

@done
    %oi_final =l loadl %oi_p
    ret %oi_final
}
