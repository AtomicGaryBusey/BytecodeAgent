qsort_pairs:
    push r15
    push r14
    push r13
    push r12
    push rbp
    push rbx
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
.Lqs_top:
    cmp r13, 1
    jle .Lqs_ret
    mov rax, r13
    sar rax, 1
    mov rcx, r13
    dec rcx
    cmp rax, rcx
    je .Lqs_pivot_at_end
    shl rax, 4
    shl rcx, 4
    mov r8,  qword ptr [r12 + rax]
    mov r9,  qword ptr [r12 + rax + 8]
    mov r10, qword ptr [r12 + rcx]
    mov r11, qword ptr [r12 + rcx + 8]
    mov qword ptr [r12 + rax],     r10
    mov qword ptr [r12 + rax + 8], r11
    mov qword ptr [r12 + rcx],     r8
    mov qword ptr [r12 + rcx + 8], r9
.Lqs_pivot_at_end:
    mov rax, r13
    dec rax
    shl rax, 4
    mov rbx, qword ptr [r12 + rax]
    mov rbp, qword ptr [r12 + rax + 8]
    mov r14, -1
    xor r15, r15
.Lqs_part_loop:
    mov rax, r13
    dec rax
    cmp r15, rax
    jge .Lqs_part_done
    mov rax, r15
    shl rax, 4
    mov rcx, qword ptr [r12 + rax]
    mov rdx, qword ptr [r12 + rax + 8]
    cmp rcx, rbx
    jg .Lqs_no_swap
    jl .Lqs_do_swap
    cmp rdx, rbp
    jg .Lqs_no_swap
.Lqs_do_swap:
    inc r14
    mov r8, r14
    shl r8, 4
    mov r9,  qword ptr [r12 + r8]
    mov r10, qword ptr [r12 + r8 + 8]
    mov qword ptr [r12 + r8],      rcx
    mov qword ptr [r12 + r8 + 8],  rdx
    mov qword ptr [r12 + rax],     r9
    mov qword ptr [r12 + rax + 8], r10
.Lqs_no_swap:
    inc r15
    jmp .Lqs_part_loop
.Lqs_part_done:
    inc r14
    mov rax, r14
    shl rax, 4
    mov rcx, r13
    dec rcx
    shl rcx, 4
    cmp rax, rcx
    je .Lqs_no_final_swap
    mov r8,  qword ptr [r12 + rax]
    mov r9,  qword ptr [r12 + rax + 8]
    mov r10, qword ptr [r12 + rcx]
    mov r11, qword ptr [r12 + rcx + 8]
    mov qword ptr [r12 + rax],     r10
    mov qword ptr [r12 + rax + 8], r11
    mov qword ptr [r12 + rcx],     r8
    mov qword ptr [r12 + rcx + 8], r9
.Lqs_no_final_swap:
    mov rdi, r12
    mov rsi, r14
    call qsort_pairs
    mov rax, r14
    inc rax
    shl rax, 4
    add r12, rax
    sub r13, r14
    dec r13
    jmp .Lqs_top
.Lqs_ret:
    xor eax, eax
    add rsp, 8
    pop rbx
    pop rbp
    pop r12
    pop r13
    pop r14
    pop r15
    ret
