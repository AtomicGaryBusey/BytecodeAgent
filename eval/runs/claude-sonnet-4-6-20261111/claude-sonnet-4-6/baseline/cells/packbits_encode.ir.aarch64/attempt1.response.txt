export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @main_loop
@main_loop
    %i =l phi @start 0, @do_repeat %i_rep, @lit_copy_done %i_lit
    %out_pos =l phi @start 0, @do_repeat %op_rep, @lit_copy_done %op_c
    %cond =w csltl %i, %n
    jnz %cond, @run_loop_start, @done
@run_loop_start
    %src_i_ptr =l add %src, %i
    %src_i =w loadub %src_i_ptr
    jmp @run_loop
@run_loop
    %run_j =l phi @run_loop_start %i, @run_cont %run_j_next
    %c_rj_lt_n =w csltl %run_j, %n
    jnz %c_rj_lt_n, @run_chk1, @run_done
@run_chk1
    %run_len =l sub %run_j, %i
    %c_rl_lt_128 =w csltl %run_len, 128
    jnz %c_rl_lt_128, @run_chk2, @run_done
@run_chk2
    %src_rj_ptr =l add %src, %run_j
    %src_rj =w loadub %src_rj_ptr
    %c_eq =w ceqw %src_rj, %src_i
    jnz %c_eq, @run_cont, @run_done
@run_cont
    %run_j_next =l add %run_j, 1
    jmp @run_loop
@run_done
    %run =l sub %run_j, %i
    %c_run_ge3 =w csltl 2, %run
    jnz %c_run_ge3, @rep_chk, @lit_start
@rep_chk
    %rem_rep =l sub %cap, %out_pos
    %c_no_rep =w csltl %rem_rep, 2
    jnz %c_no_rep, @fail, @do_repeat
@do_repeat
    %hdr_rep =l sub 257, %run
    %ptr_rep0 =l add %out, %out_pos
    storeb %hdr_rep, %ptr_rep0
    %op_rep1 =l add %out_pos, 1
    %ptr_rep1 =l add %out, %op_rep1
    storeb %src_i, %ptr_rep1
    %op_rep =l add %op_rep1, 1
    %i_rep =l add %i, %run
    jmp @main_loop
@lit_start
    jmp @lit_j_loop
@lit_j_loop
    %j =l phi @lit_start %i, @lit_j_cont %j_next
    %c_j_lt_n =w csltl %j, %n
    jnz %c_j_lt_n, @lit_chk1, @lit_j_done
@lit_chk1
    %jmi =l sub %j, %i
    %c_jmi_lt_128 =w csltl %jmi, 128
    jnz %c_jmi_lt_128, @run2_start, @lit_j_done
@run2_start
    %src_j_ptr =l add %src, %j
    %src_j =w loadub %src_j_ptr
    jmp @run2_loop
@run2_loop
    %run_j2 =l phi @run2_start %j, @run2_cont %run_j2_next
    %c_rj2_lt_n =w csltl %run_j2, %n
    jnz %c_rj2_lt_n, @run2_chk1, @run2_done
@run2_chk1
    %run2_len =l sub %run_j2, %j
    %c_r2l_lt_128 =w csltl %run2_len, 128
    jnz %c_r2l_lt_128, @run2_chk2, @run2_done
@run2_chk2
    %src_rj2_ptr =l add %src, %run_j2
    %src_rj2 =w loadub %src_rj2_ptr
    %c_eq2 =w ceqw %src_rj2, %src_j
    jnz %c_eq2, @run2_cont, @run2_done
@run2_cont
    %run_j2_next =l add %run_j2, 1
    jmp @run2_loop
@run2_done
    %run2 =l sub %run_j2, %j
    %c_run2_ge3 =w csltl 2, %run2
    jnz %c_run2_ge3, @lit_j_done, @lit_j_cont
@lit_j_cont
    %j_next =l add %j, 1
    jmp @lit_j_loop
@lit_j_done
    %L =l sub %j, %i
    %needed =l add %L, 1
    %rem_lit =l sub %cap, %out_pos
    %c_no_lit =w csltl %rem_lit, %needed
    jnz %c_no_lit, @fail, @lit_write
@lit_write
    %hdr_lit =l sub %L, 1
    %ptr_lit_hdr =l add %out, %out_pos
    storeb %hdr_lit, %ptr_lit_hdr
    %op_lit_start =l add %out_pos, 1
    jmp @lit_copy_loop
@lit_copy_loop
    %k =l phi @lit_write 0, @lit_copy_body %k_next
    %op_c =l phi @lit_write %op_lit_start, @lit_copy_body %op_c_next
    %c_k_lt_L =w csltl %k, %L
    jnz %c_k_lt_L, @lit_copy_body, @lit_copy_done
@lit_copy_body
    %src_ik_base =l add %src, %i
    %src_ik_ptr =l add %src_ik_base, %k
    %src_ik =w loadub %src_ik_ptr
    %out_c_ptr =l add %out, %op_c
    storeb %src_ik, %out_c_ptr
    %op_c_next =l add %op_c, 1
    %k_next =l add %k, 1
    jmp @lit_copy_loop
@lit_copy_done
    %i_lit =l add %i, %L
    jmp @main_loop
@fail
    ret -1
@done
    ret %out_pos
}
