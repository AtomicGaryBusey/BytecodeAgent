export function l $crc32(l %p, l %n) {
@start
    jmp @outer_cond
@outer_cond
    %crc =w phi @start 4294967295, @outer_next %crc1
    %i =l phi @start 0, @outer_next %i1
    %c0 =w csltl %i, %n
    jnz %c0, @outer_body, @done
@outer_body
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc0 =w xor %crc, %byte
    jmp @inner_cond
@inner_cond
    %crc1 =w phi @outer_body %crc0, @bit_end %crc2
    %j =w phi @outer_body 0, @bit_end %j1
    %c1 =w csltw %j, 8
    jnz %c1, @inner_body, @outer_next
@inner_body
    %lsb =w and %crc1, 1
    jnz %lsb, @bit1, @bit0
@bit1
    %crc1a =w shr %crc1, 1
    %crc1b =w xor %crc1a, 3988292384
    jmp @bit_end
@bit0
    %crc1c =w shr %crc1, 1
    jmp @bit_end
@bit_end
    %crc2 =w phi @bit1 %crc1b, @bit0 %crc1c
    %j1 =w add %j, 1
    jmp @inner_cond
@outer_next
    %i1 =l add %i, 1
    jmp @outer_cond
@done
    %result =w xor %crc, 4294967295
    %rl =l extuw %result
    ret %rl
}
