export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @after_rep %i_rep_next, @after_lit %i_lit_next
    %outpos =l phi @start 0, @after_rep %op_rep, @after_lit %op_lit_final
    %loop_cond =w csltl %i, %n
    jnz %loop_cond, @body, @end
@body
    %pi =l add %src, %i
    %bi =w loadub %pi
    %ri_init =l add %i, 1
    jmp @runloop
@runloop
    %run =l phi @body 1, @rlinc %run_next
    %ri =l phi @body %ri_init, @rlinc %ri_next
    %rc1 =w csltl %ri, %n
    jnz %rc1, @rl2, @rundone
@rl2
    %rc2 =w csltl %run, 128
    jnz %rc2, @rl3, @rundone
@rl3
    %pri =l add %src, %ri
    %bri =w loadub %pri
    %rc3 =w ceqw %bri, %bi
    jnz %rc3, @rlinc, @rundone
@rlinc
    %run_next =l add %run, 1
    %ri_next =l add %ri, 1
    jmp @runloop
@rundone
    %rge3 =w csgel %run, 3
    jnz %rge3, @repeat, @literal
@repeat
    %rem_rep =l sub %cap, %outpos
    %rep_fits =w csgel %rem_rep, 2
    jnz %rep_fits, @repeat_ok, @fail
@repeat_ok
    %rep_hdr =l sub 257, %run
    %po_rep1 =l add %out, %outpos
    storeb %rep_hdr, %po_rep1
    %op1 =l add %outpos, 1
    %po_rep2 =l add %out, %op1
    storeb %bi, %po_rep2
    %op_rep =l add %op1, 1
    %i_rep_next =l add %i, %run
    jmp @after_rep
@after_rep
    jmp @loop
@literal
    jmp @litloop
@litloop
    %j =l phi @literal %i, @litinc_blk %j_next
    %lc1 =w csltl %j, %n
    jnz %lc1, @ll2, @litdone
@ll2
    %jdiff =l sub %j, %i
    %lc2 =w csltl %jdiff, 128
    jnz %lc2, @ll3, @litdone
@ll3
    %jj1 =l add %j, 1
    %jj2 =l add %j, 2
    %lc3 =w csltl %jj2, %n
    jnz %lc3, @ll4, @litinc_blk
@ll4
    %pj =l add %src, %j
    %bj =w loadub %pj
    %pj1 =l add %src, %jj1
    %bj1 =w loadub %pj1
    %lc4 =w ceqw %bj, %bj1
    jnz %lc4, @ll5, @litinc_blk
@ll5
    %pj2 =l add %src, %jj2
    %bj2 =w loadub %pj2
    %lc5 =w ceqw %bj, %bj2
    jnz %lc5, @litdone, @litinc_blk
@litinc_blk
    %j_next =l add %j, 1
    jmp @litloop
@litdone
    %L =l sub %j, %i
    %need =l add %L, 1
    %rem_lit =l sub %cap, %outpos
    %lit_fits =w csgel %rem_lit, %need
    jnz %lit_fits, @lit_ok, @fail
@lit_ok
    %lit_hdr =l sub %L, 1
    %po_lit1 =l add %out, %outpos
    storeb %lit_hdr, %po_lit1
    %op_lit1 =l add %outpos, 1
    jmp @litcopy
@litcopy
    %k =l phi @lit_ok 0, @litcopy_inc %k_next
    %op_lc =l phi @lit_ok %op_lit1, @litcopy_inc %op_lc_next
    %lcc =w csltl %k, %L
    jnz %lcc, @litcopy_body, @after_lit
@litcopy_body
    %srcidx =l add %i, %k
    %psrc_lc =l add %src, %srcidx
    %bk =w loadub %psrc_lc
    %po_lc_addr =l add %out, %op_lc
    storeb %bk, %po_lc_addr
    %op_lc_next =l add %op_lc, 1
    %k_next =l add %k, 1
    jmp @litcopy_inc
@litcopy_inc
    jmp @litcopy
@after_lit
    %op_lit_final =l copy %op_lc
    %i_lit_next =l add %i, %L
    jmp @loop
@end
    ret %outpos
@fail
    ret -1
}
