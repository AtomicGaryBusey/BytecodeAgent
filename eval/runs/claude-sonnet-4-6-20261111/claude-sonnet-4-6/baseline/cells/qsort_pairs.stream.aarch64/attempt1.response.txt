qsort_pairs:
    cmp x1, #2
    b.lt .Lbase_case
    stp x29, x30, [sp, #-48]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    mov x19, x0
    mov x20, x1
    // choose pivot: middle element (index n/2), swap it to end
    lsr x9, x20, #1
    lsl x10, x9, #4
    add x10, x19, x10
    sub x11, x20, #1
    lsl x12, x11, #4
    add x12, x19, x12
    ldp x13, x14, [x10]
    ldp x15, x2, [x12]
    stp x15, x2, [x10]
    stp x13, x14, [x12]
    // lomuto partition: i=0, j=0..n-2, pivot key=x13 value=x14 at a[n-1]
    mov x9, #0
    mov x10, #0
.Llomuto_loop:
    cmp x10, x11
    b.ge .Llomuto_done
    lsl x2, x10, #4
    add x2, x19, x2
    ldr x3, [x2]
    ldr x4, [x2, #8]
    cmp x3, x13
    b.gt .Lno_swap
    b.lt .Ldo_swap
    cmp x4, x14
    b.gt .Lno_swap
.Ldo_swap:
    lsl x5, x9, #4
    add x5, x19, x5
    ldp x6, x7, [x5]
    stp x3, x4, [x5]
    stp x6, x7, [x2]
    add x9, x9, #1
.Lno_swap:
    add x10, x10, #1
    b .Llomuto_loop
.Llomuto_done:
    // place pivot at position i by swapping a[i] with a[n-1]
    lsl x2, x9, #4
    add x2, x19, x2
    lsl x3, x11, #4
    add x3, x19, x3
    ldp x4, x5, [x2]
    stp x13, x14, [x2]
    stp x4, x5, [x3]
    mov x21, x9
    // recurse on left part [0, pivot-1], size = pivot
    mov x0, x19
    mov x1, x21
    cmp x1, #2
    b.lt .Lskip_left
    bl qsort_pairs
.Lskip_left:
    // recurse on right part [pivot+1, n-1], size = n-pivot-1
    add x22, x21, #1
    lsl x2, x22, #4
    add x0, x19, x2
    sub x1, x20, x22
    cmp x1, #2
    b.lt .Lskip_right
    bl qsort_pairs
.Lskip_right:
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #48
    mov x0, #0
    ret
.Lbase_case:
    mov x0, #0
    ret
