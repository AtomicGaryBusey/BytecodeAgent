export function l $crc32(l %p, l %n) {
@start
    %crc =w copy 4294967295
    %i =l copy 0
@outer_check
    %cond =w csltl %i, %n
    jnz %cond, @outer_body, @done
@outer_body
    %addr =l add %p, %i
    %b =w loadub %addr
    %crc =w xor %crc, %b
    %j =w copy 0
@inner_check
    %inner_cond =w csltw %j, 8
    jnz %inner_cond, @inner_body, @inner_done
@inner_body
    %lsb =w and %crc, 1
    jnz %lsb, @bit_set, @bit_clear
@bit_set
    %crc =w shr %crc, 1
    %crc =w xor %crc, 3988292384
    jmp @inner_next
@bit_clear
    %crc =w shr %crc, 1
    jmp @inner_next
@inner_next
    %j =w add %j, 1
    jmp @inner_check
@inner_done
    %i =l add %i, 1
    jmp @outer_check
@done
    %result =w xor %crc, 4294967295
    %result_l =l extuw %result
    ret %result_l
}
