export function l $count_desc(l %a, l %n) {
@start
    %cmp = csltl %n, 2
    jnz %cmp, @ret0, @loop_init
@ret0
    ret 0
@loop_init
    %limit = subl %n, 1
    jmp @loop_check
@loop_check
    %i =l phi @loop_init 0, @loop_body %i_next
    %count =l phi @loop_init 0, @loop_body %count_next
    %cond = csltl %i, %limit
    jnz %cond, @loop_body, @loop_end
@loop_body
    %off_i = mull %i, 8
    %ptr_i = addl %a, %off_i
    %val_i = loadl %ptr_i
    %i1 = addl %i, 1
    %off_i1 = mull %i1, 8
    %ptr_i1 = addl %a, %off_i1
    %val_i1 = loadl %ptr_i1
    %gt = csgtl %val_i, %val_i1
    %gt_l = extuw %gt
    %count_next = addl %count, %gt_l
    %i_next = addl %i, 1
    jmp @loop_check
@loop_end
    ret %count
}
