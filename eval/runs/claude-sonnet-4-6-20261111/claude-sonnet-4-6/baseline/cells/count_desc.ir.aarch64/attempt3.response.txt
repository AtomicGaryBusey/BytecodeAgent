export function l $count_desc(l %a, l %n) {
@start
    %cmp =w csltl %n, 2
    jnz %cmp, @ret0, @loop_init
@ret0
    ret 0
@loop_init
    %limit =l sub %n, 1
    jmp @loop_check
@loop_check
    %i =l phi @loop_init 0, @loop_body %i_next
    %count =l phi @loop_init 0, @loop_body %count_next
    %cond =w csltl %i, %limit
    jnz %cond, @loop_body, @loop_end
@loop_body
    %off_i =l mul %i, 8
    %ptr_i =l add %a, %off_i
    %val_i =l loadl %ptr_i
    %i1 =l add %i, 1
    %off_i1 =l mul %i1, 8
    %ptr_i1 =l add %a, %off_i1
    %val_i1 =l loadl %ptr_i1
    %gt =w csgtl %val_i, %val_i1
    %gt_l =l extuw %gt
    %count_next =l add %count, %gt_l
    %i_next =l add %i, 1
    jmp @loop_check
@loop_end
    ret %count
}
