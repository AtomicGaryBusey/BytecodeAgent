export function l $shift_sat(l %x, l %n) {
@start
    %cmp_n64 =w csltl %n, 64
    jnz %cmp_n64, @small_n, @large_n
@small_n
    %left_s =l shl %x, %n
    %right_s =l sar %x, %n
    %result_s =l add %left_s, %right_s
    jmp @done
@large_n
    %cmp_x0 =w csltl %x, 0
    jnz %cmp_x0, @neg_x, @nonneg_x
@neg_x
    jmp @done_large
@nonneg_x
    jmp @done_large
@done_large
    %right_l =l phi @neg_x -1, @nonneg_x 0
    jmp @done
@done
    %result =l phi @small_n %result_s, @done_large %right_l
    ret %result
}
