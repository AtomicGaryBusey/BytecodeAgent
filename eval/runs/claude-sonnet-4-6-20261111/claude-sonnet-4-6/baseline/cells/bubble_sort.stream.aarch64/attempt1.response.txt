bubble_sort:
    cmp x1, #1
    b.le .Ldone
    mov x2, #0
.Louter:
    mov x3, #0
    sub x4, x1, #1
    sub x4, x4, x2
.Linner:
    cmp x3, x4
    b.ge .Linner_done
    lsl x5, x3, #3
    add x9, x0, x5
    ldr x6, [x9]
    ldr x7, [x9, #8]
    cmp x6, x7
    b.le .Lno_swap
    str x7, [x9]
    str x6, [x9, #8]
.Lno_swap:
    add x3, x3, #1
    b .Linner
.Linner_done:
    add x2, x2, #1
    cmp x2, x1
    b.lt .Louter
.Ldone:
    mov x0, #0
    ret
