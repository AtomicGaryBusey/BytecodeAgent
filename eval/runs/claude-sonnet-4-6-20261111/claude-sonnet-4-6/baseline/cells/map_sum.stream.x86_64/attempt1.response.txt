map_sum:
    push r15
    push r14
    push r13
    push r12
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    xor r15, r15
    test r13, r13
    jle .Ldone
.Lloop:
    mov rdi, qword ptr [r12]
    call r14
    add r15, rax
    add r12, 8
    dec r13
    jnz .Lloop
.Ldone:
    mov rax, r15
    add rsp, 8
    pop r12
    pop r13
    pop r14
    pop r15
    ret
