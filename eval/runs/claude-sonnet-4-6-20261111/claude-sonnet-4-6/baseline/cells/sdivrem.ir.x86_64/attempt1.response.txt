export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bzero =w ceql %b, 0
    jnz %bzero, @b_is_zero, @check_overflow
@check_overflow
    %a_is_min =w ceql %a, -9223372036854775808
    %b_is_neg1 =w ceql %b, -1
    %overflow =w and %a_is_min, %b_is_neg1
    jnz %overflow, @overflow_case, @normal
@b_is_zero
    storel %a, %rem
    ret 0
@overflow_case
    storel 0, %rem
    ret -9223372036854775808
@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
