export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %zerocmp =w ceql %n, 0
    jnz %zerocmp, @done, @check_dir
@check_dir
    %fwd =w culel %dst, %src
    jnz %fwd, @fwd_init, @bwd_init
@fwd_init
    jmp @fwd_loop
@fwd_loop
    %i =l phi @fwd_init 0, @fwd_body %i1
    %cmp =w csltl %i, %n
    jnz %cmp, @fwd_body, @done
@fwd_body
    %off =l mul %i, 8
    %sp =l add %src, %off
    %dp =l add %dst, %off
    %v =l loadl %sp
    storel %v, %dp
    %i1 =l add %i, 1
    jmp @fwd_loop
@bwd_init
    %nm1 =l sub %n, 1
    jmp @bwd_loop
@bwd_loop
    %j =l phi @bwd_init %nm1, @bwd_body %j1
    %jcmp =w csgel %j, 0
    jnz %jcmp, @bwd_body, @done
@bwd_body
    %joff =l mul %j, 8
    %jsp =l add %src, %joff
    %jdp =l add %dst, %joff
    %jv =l loadl %jsp
    storel %jv, %jdp
    %j1 =l sub %j, 1
    jmp @bwd_loop
@done
    ret 0
}
