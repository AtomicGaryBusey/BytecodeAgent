isqrt:
    push rbx
    push r12
    push r13
    mov r13, rdi
    xor r12d, r12d
    mov rbx, 0xFFFFFFFF
.Lloop:
    cmp r12, rbx
    jae .Ldone
    mov rax, r12
    add rax, rbx
    inc rax
    shr rax, 1
    mov rcx, rax
    mul rax
    test rdx, rdx
    jnz .Lhi_update
    cmp rax, r13
    ja .Lhi_update
    mov r12, rcx
    jmp .Lloop
.Lhi_update:
    lea rbx, [rcx - 1]
    jmp .Lloop
.Ldone:
    mov rax, r12
    pop r13
    pop r12
    pop rbx
    ret
