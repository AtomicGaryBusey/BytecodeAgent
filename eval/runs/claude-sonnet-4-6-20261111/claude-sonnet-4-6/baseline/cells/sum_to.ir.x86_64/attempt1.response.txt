export function l $sum_to(l %n) {
@start
    %cmp =w csltl %n, 1
    jnz %cmp, @ret_zero, @loop_init
@ret_zero
    ret 0
@loop_init
    %sum =l copy 0
    %i =l copy 1
    jmp @loop
@loop
    %done =w csgtl %i, %n
    jnz %done, @ret_sum, @body
@body
    %sum =l add %sum, %i
    %i =l add %i, 1
    jmp @loop
@ret_sum
    ret %sum
}
