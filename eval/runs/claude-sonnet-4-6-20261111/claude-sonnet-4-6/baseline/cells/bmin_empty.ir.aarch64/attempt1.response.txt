export function l $bmin_empty(l %a, l %n) {
@start
    %cmp =w csltl %n, 1
    jnz %cmp, @empty, @loop_init
@empty
    ret 9223372036854775807
@loop_init
    %first =l loadl %a
    %one =l copy 1
    jmp @loop_check
@loop_check
    %min =l phi @loop_init %first, @loop_next %next_min
    %i =l phi @loop_init %one, @loop_next %next_i
    %done =w csgel %i, %n
    jnz %done, @ret, @loop_body
@loop_body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %less =w csltl %val, %min
    jnz %less, @update, @no_update
@update
    jmp @loop_next
@no_update
    jmp @loop_next
@loop_next
    %next_min =l phi @update %val, @no_update %min
    %next_i =l add %i, 1
    jmp @loop_check
@ret
    ret %min
}
