export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @loop_check
@loop_check
    %i =l phi @start 0, @loop_body %i_next
    %count =l phi @start 0, @loop_body %count_next
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @loop_end
@loop_body
    %offset =l mul %i, 8
    %src_ptr =l add %src, %offset
    %src_val =l loadl %src_ptr
    %result =l call %f(l %src_val)
    %dst_ptr =l add %dst, %offset
    storel %result, %dst_ptr
    %less =w csltl %result, %src_val
    %less_l =l extsw %less
    %count_next =l add %count, %less_l
    %i_next =l add %i, 1
    jmp @loop_check
@loop_end
    ret %count
}
