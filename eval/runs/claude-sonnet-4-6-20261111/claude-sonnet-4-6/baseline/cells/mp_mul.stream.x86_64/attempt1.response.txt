mp_mul:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rbp, r8
    lea rbx, [r13 + r15]
    test r13, r13
    jz .Lzero_result
    test r15, r15
    jz .Lzero_result
    xor rcx, rcx
.Lzero_loop:
    mov qword ptr [rbp + rcx*8], 0
    inc rcx
    cmp rcx, rbx
    jl .Lzero_loop
    xor r10, r10
.Louter:
    cmp r10, r13
    jge .Ldone_mul
    xor r11, r11
.Linner:
    cmp r11, r15
    jge .Lnext_outer
    mov rax, qword ptr [r12 + r10*8]
    mul qword ptr [r14 + r11*8]
    lea rcx, [r10 + r11]
    add qword ptr [rbp + rcx*8], rax
    adc qword ptr [rbp + rcx*8 + 8], rdx
    lea rcx, [rcx + 2]
.Lcarry_prop:
    jnc .Lno_carry_prop
    adc qword ptr [rbp + rcx*8], 0
    inc rcx
    jmp .Lcarry_prop
.Lno_carry_prop:
    inc r11
    jmp .Linner
.Lnext_outer:
    inc r10
    jmp .Louter
.Ldone_mul:
    mov rax, rbx
.Lfind_sig:
    test rax, rax
    jz .Lclean_return
    dec rax
    cmp qword ptr [rbp + rax*8], 0
    je .Lfind_sig
    inc rax
    jmp .Lclean_return
.Lzero_result:
    test rbx, rbx
    jz .Lreturn_zero_val
    xor rcx, rcx
.Lzero_result_loop:
    mov qword ptr [rbp + rcx*8], 0
    inc rcx
    cmp rcx, rbx
    jl .Lzero_result_loop
.Lreturn_zero_val:
    xor rax, rax
.Lclean_return:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
