export function l $binary_search(l %a, l %n, l %key) {
@start
    %hi0 =l sub %n, 1
    jmp @loop
@loop
    %lo =l phi @start 0, @go_right %lo_new, @go_left %lo
    %hi =l phi @start %hi0, @go_right %hi, @go_left %hi_new
    %cond =w csltl %hi, %lo
    jnz %cond, @notfound, @body
@body
    %diff =l sub %hi, %lo
    %half =l div %diff, 2
    %mid =l add %lo, %half
    %offset =l mul %mid, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %eq =w ceql %val, %key
    jnz %eq, @found, @notfound_check
@notfound_check
    %lt =w csltl %val, %key
    jnz %lt, @go_right, @go_left
@go_right
    %lo_new =l add %mid, 1
    jmp @loop
@go_left
    %hi_new =l sub %mid, 1
    jmp @loop
@found
    ret %mid
@notfound
    ret -1
}
