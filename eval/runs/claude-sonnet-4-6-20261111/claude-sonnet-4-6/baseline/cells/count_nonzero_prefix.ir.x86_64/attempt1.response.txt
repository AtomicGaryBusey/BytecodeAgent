export function l $count_nonzero_prefix(l %a, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @next %i_next
    %cmp =w csltl %i, %n
    jnz %cmp, @check, @done_n
@check
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %val =l loadl %ptr
    %iszero =w ceql %val, 0
    jnz %iszero, @done_i, @next
@next
    %i_next =l add %i, 1
    jmp @loop
@done_i
    ret %i
@done_n
    ret %n
}
