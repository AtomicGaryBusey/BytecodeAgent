packbits_decode:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    lea r13, [rdi + rsi]
    mov rbx, rdx
    mov r14, rdx
    lea r15, [rdx + rcx]
.Lmain_loop:
    cmp r12, r13
    jae .Ldone
    movsx rax, byte ptr [r12]
    inc r12
    cmp rax, -128
    je .Lmain_loop
    test rax, rax
    js .Lrepeat
    lea rcx, [rax + 1]
    lea rax, [r12 + rcx]
    cmp rax, r13
    ja .Lerror_src
.Lcopy_loop:
    test rcx, rcx
    jz .Lmain_loop
    cmp r14, r15
    jae .Lcap_exceeded
    mov al, byte ptr [r12]
    mov byte ptr [r14], al
    inc r12
    inc r14
    dec rcx
    jmp .Lcopy_loop
.Lrepeat:
    cmp r12, r13
    jae .Lerror_src
    neg rax
    inc rax
    mov rdx, rax
    movzx rax, byte ptr [r12]
    inc r12
.Lrepeat_loop:
    test rdx, rdx
    jz .Lmain_loop
    cmp r14, r15
    jae .Lcap_exceeded
    mov byte ptr [r14], al
    inc r14
    dec rdx
    jmp .Lrepeat_loop
.Ldone:
    sub r14, rbx
    mov rax, r14
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lerror_src:
    mov rax, -2
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
.Lcap_exceeded:
    mov rax, -1
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
