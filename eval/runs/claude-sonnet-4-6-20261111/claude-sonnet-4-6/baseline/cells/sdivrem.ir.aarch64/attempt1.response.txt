export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %biszero =w ceql %b, 0
    jnz %biszero, @bzero, @bnonzero
@bzero
    storel %a, %rem
    ret 0
@bnonzero
    %intmin =l shl 1, 63
    %amin =w ceql %a, %intmin
    %bm1 =w ceql %b, -1
    %ovfw =w and %amin, %bm1
    jnz %ovfw, @ovf, @normal
@ovf
    storel 0, %rem
    ret %intmin
@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
