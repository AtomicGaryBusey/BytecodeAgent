export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pc_ptr =l alloc8 8
    %sp_ptr =l alloc8 8
    %zi_ptr =l alloc8 8
    storel 0, %pc_ptr
    storel 0, %sp_ptr
    jmp @loop

@loop
    %pc =l loadl %pc_ptr
    %sp =l loadl %sp_ptr
    %in_range =w csltl %pc, %len
    jnz %in_range, @fetch, @done

@fetch
    %pc_off =l mul %pc, 8
    %code_at_pc =l add %code, %pc_off
    %opcode =l loadl %code_at_pc
    %is0 =w ceql %opcode, 0
    jnz %is0, @op_halt, @c1

@c1
    %is1 =w ceql %opcode, 1
    jnz %is1, @op_push, @c2

@c2
    %is2 =w ceql %opcode, 2
    jnz %is2, @op_add, @c3

@c3
    %is3 =w ceql %opcode, 3
    jnz %is3, @op_sub, @c4

@c4
    %is4 =w ceql %opcode, 4
    jnz %is4, @op_mul, @c5

@c5
    %is5 =w ceql %opcode, 5
    jnz %is5, @op_dup, @c6

@c6
    %is6 =w ceql %opcode, 6
    jnz %is6, @op_swap, @c7

@c7
    %is7 =w ceql %opcode, 7
    jnz %is7, @op_drop, @c8

@c8
    %is8 =w ceql %opcode, 8
    jnz %is8, @op_over, @c9

@c9
    %is9 =w ceql %opcode, 9
    jnz %is9, @op_lt, @c10

@c10
    %is10 =w ceql %opcode, 10
    jnz %is10, @op_jz, @c11

@c11
    %is11 =w ceql %opcode, 11
    jnz %is11, @op_jmp, @op_other

@op_halt
    jmp @done

@op_push
    %pc1p =l add %pc, 1
    %pc1p_off =l mul %pc1p, 8
    %code_pc1p =l add %code, %pc1p_off
    %push_val =l loadl %code_pc1p
    %sp_off_push =l mul %sp, 8
    %stack_sp_push =l add %stack, %sp_off_push
    storel %push_val, %stack_sp_push
    %sp_push =l add %sp, 1
    %pc_push =l add %pc, 2
    storel %sp_push, %sp_ptr
    storel %pc_push, %pc_ptr
    jmp @loop

@op_add
    %sp_b_add =l sub %sp, 1
    %sp_a_add =l sub %sp, 2
    %off_b_add =l mul %sp_b_add, 8
    %off_a_add =l mul %sp_a_add, 8
    %ptr_b_add =l add %stack, %off_b_add
    %ptr_a_add =l add %stack, %off_a_add
    %b_add =l loadl %ptr_b_add
    %a_add =l loadl %ptr_a_add
    %res_add =l add %a_add, %b_add
    storel %res_add, %ptr_a_add
    %sp_new_add =l sub %sp, 1
    %pc_new_add =l add %pc, 1
    storel %sp_new_add, %sp_ptr
    storel %pc_new_add, %pc_ptr
    jmp @loop

@op_sub
    %sp_b_sub =l sub %sp, 1
    %sp_a_sub =l sub %sp, 2
    %off_b_sub =l mul %sp_b_sub, 8
    %off_a_sub =l mul %sp_a_sub, 8
    %ptr_b_sub =l add %stack, %off_b_sub
    %ptr_a_sub =l add %stack, %off_a_sub
    %b_sub =l loadl %ptr_b_sub
    %a_sub =l loadl %ptr_a_sub
    %res_sub =l sub %a_sub, %b_sub
    storel %res_sub, %ptr_a_sub
    %sp_new_sub =l sub %sp, 1
    %pc_new_sub =l add %pc, 1
    storel %sp_new_sub, %sp_ptr
    storel %pc_new_sub, %pc_ptr
    jmp @loop

@op_mul
    %sp_b_mul =l sub %sp, 1
    %sp_a_mul =l sub %sp, 2
    %off_b_mul =l mul %sp_b_mul, 8
    %off_a_mul =l mul %sp_a_mul, 8
    %ptr_b_mul =l add %stack, %off_b_mul
    %ptr_a_mul =l add %stack, %off_a_mul
    %b_mul =l loadl %ptr_b_mul
    %a_mul =l loadl %ptr_a_mul
    %res_mul =l mul %a_mul, %b_mul
    storel %res_mul, %ptr_a_mul
    %sp_new_mul =l sub %sp, 1
    %pc_new_mul =l add %pc, 1
    storel %sp_new_mul, %sp_ptr
    storel %pc_new_mul, %pc_ptr
    jmp @loop

@op_dup
    %sp_top_dup =l sub %sp, 1
    %off_top_dup =l mul %sp_top_dup, 8
    %ptr_top_dup =l add %stack, %off_top_dup
    %val_top_dup =l loadl %ptr_top_dup
    %off_new_dup =l mul %sp, 8
    %ptr_new_dup =l add %stack, %off_new_dup
    storel %val_top_dup, %ptr_new_dup
    %sp_new_dup =l add %sp, 1
    %pc_new_dup =l add %pc, 1
    storel %sp_new_dup, %sp_ptr
    storel %pc_new_dup, %pc_ptr
    jmp @loop

@op_swap
    %sp_top_sw =l sub %sp, 1
    %sp_bot_sw =l sub %sp, 2
    %off_top_sw =l mul %sp_top_sw, 8
    %off_bot_sw =l mul %sp_bot_sw, 8
    %ptr_top_sw =l add %stack, %off_top_sw
    %ptr_bot_sw =l add %stack, %off_bot_sw
    %val_top_sw =l loadl %ptr_top_sw
    %val_bot_sw =l loadl %ptr_bot_sw
    storel %val_top_sw, %ptr_bot_sw
    storel %val_bot_sw, %ptr_top_sw
    %pc_new_sw =l add %pc, 1
    storel %pc_new_sw, %pc_ptr
    jmp @loop

@op_drop
    %sp_new_dr =l sub %sp, 1
    %pc_new_dr =l add %pc, 1
    storel %sp_new_dr, %sp_ptr
    storel %pc_new_dr, %pc_ptr
    jmp @loop

@op_over
    %sp_bot_ov =l sub %sp, 2
    %off_bot_ov =l mul %sp_bot_ov, 8
    %ptr_bot_ov =l add %stack, %off_bot_ov
    %val_ov =l loadl %ptr_bot_ov
    %off_new_ov =l mul %sp, 8
    %ptr_new_ov =l add %stack, %off_new_ov
    storel %val_ov, %ptr_new_ov
    %sp_new_ov =l add %sp, 1
    %pc_new_ov =l add %pc, 1
    storel %sp_new_ov, %sp_ptr
    storel %pc_new_ov, %pc_ptr
    jmp @loop

@op_lt
    %sp_b_lt =l sub %sp, 1
    %sp_a_lt =l sub %sp, 2
    %off_b_lt =l mul %sp_b_lt, 8
    %off_a_lt =l mul %sp_a_lt, 8
    %ptr_b_lt =l add %stack, %off_b_lt
    %ptr_a_lt =l add %stack, %off_a_lt
    %b_lt =l loadl %ptr_b_lt
    %a_lt =l loadl %ptr_a_lt
    %cmp_lt =w csltl %a_lt, %b_lt
    %res_lt =l extsw %cmp_lt
    storel %res_lt, %ptr_a_lt
    %sp_new_lt =l sub %sp, 1
    %pc_new_lt =l add %pc, 1
    storel %sp_new_lt, %sp_ptr
    storel %pc_new_lt, %pc_ptr
    jmp @loop

@op_jz
    %sp_jz =l sub %sp, 1
    %off_jz =l mul %sp_jz, 8
    %ptr_jz =l add %stack, %off_jz
    %v_jz =l loadl %ptr_jz
    storel %sp_jz, %sp_ptr
    %pc1_jz =l add %pc, 1
    %off_t_jz =l mul %pc1_jz, 8
    %ptr_t_jz =l add %code, %off_t_jz
    %t_jz =l loadl %ptr_t_jz
    %is_zero_jz =w ceql %v_jz, 0
    jnz %is_zero_jz, @jz_taken, @jz_not

@jz_taken
    storel %t_jz, %pc_ptr
    jmp @loop

@jz_not
    %pc_new_jz =l add %pc, 2
    storel %pc_new_jz, %pc_ptr
    jmp @loop

@op_jmp
    %pc1_jmp =l add %pc, 1
    %off_t_jmp =l mul %pc1_jmp, 8
    %ptr_t_jmp =l add %code, %off_t_jmp
    %t_jmp =l loadl %ptr_t_jmp
    storel %t_jmp, %pc_ptr
    jmp @loop

@op_other
    %pc_new_ot =l add %pc, 1
    storel %pc_new_ot, %pc_ptr
    jmp @loop

@done
    %fsp =l loadl %sp_ptr
    storel %fsp, %zi_ptr
    jmp @zero_loop

@zero_loop
    %zi =l loadl %zi_ptr
    %zi_in_range =w csltl %zi, %cap
    jnz %zi_in_range, @zero_body, @finish

@zero_body
    %zi_off =l mul %zi, 8
    %zi_addr =l add %stack, %zi_off
    storel 0, %zi_addr
    %zi_next =l add %zi, 1
    storel %zi_next, %zi_ptr
    jmp @zero_loop

@finish
    %fsp2 =l loadl %sp_ptr
    %is_empty =w ceql %fsp2, 0
    jnz %is_empty, @ret_zero, @ret_top

@ret_zero
    ret 0

@ret_top
    %top_idx =l sub %fsp2, 1
    %top_off =l mul %top_idx, 8
    %top_addr =l add %stack, %top_off
    %result =l loadl %top_addr
    ret %result
}
