sdivrem:
    mov r10, rdx
    test rsi, rsi
    jz .Lbzero
    mov rax, rdi
    mov r11, 0x8000000000000000
    cmp rdi, r11
    jne .Lnormal
    cmp rsi, -1
    jne .Lnormal
    mov qword ptr [r10], 0
    ret
.Lnormal:
    cqo
    idiv rsi
    mov qword ptr [r10], rdx
    ret
.Lbzero:
    xor eax, eax
    mov qword ptr [r10], rdi
    ret
