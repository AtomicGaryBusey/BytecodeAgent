export function l $collatz_max(l %n) {
@start
    jmp @outer_cond
@outer_cond
    %k =l phi @start 1, @outer_next %k_next
    %max =l phi @start 0, @outer_next %max_next
    %cond =w cslel %k, %n
    jnz %cond, @inner_init, @done
@inner_init
    jmp @inner_cond
@inner_cond
    %x =l phi @inner_init %k, @inner_next %x_next
    %steps =l phi @inner_init 0, @inner_next %steps_next
    %eq1 =w ceql %x, 1
    jnz %eq1, @inner_done, @inner_body
@inner_body
    %lsb =l and %x, 1
    %is_even =w ceql %lsb, 0
    jnz %is_even, @even_branch, @odd_branch
@even_branch
    %x_even =l div %x, 2
    jmp @inner_next
@odd_branch
    %x_tmp =l mul %x, 3
    %x_odd =l add %x_tmp, 1
    jmp @inner_next
@inner_next
    %x_next =l phi @even_branch %x_even, @odd_branch %x_odd
    %steps_next =l add %steps, 1
    jmp @inner_cond
@inner_done
    %gt =w csgtl %steps, %max
    jnz %gt, @update_max, @keep_max
@update_max
    jmp @outer_next
@keep_max
    jmp @outer_next
@outer_next
    %max_next =l phi @update_max %steps, @keep_max %max
    %k_next =l add %k, 1
    jmp @outer_cond
@done
    ret %max
}
