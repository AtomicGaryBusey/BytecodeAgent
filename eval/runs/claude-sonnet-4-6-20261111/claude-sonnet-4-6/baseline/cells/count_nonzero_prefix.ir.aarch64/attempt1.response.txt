export function l $count_nonzero_prefix(l %a, l %n) {
@start
    jmp @check
@check
    %i =l phi @start 0, @continue %i1
    %cmp =w csltl %i, %n
    jnz %cmp, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %iszero =w ceql %val, 0
    jnz %iszero, @found, @continue
@continue
    %i1 =l add %i, 1
    jmp @check
@found
    ret %i
@done
    ret %n
}
