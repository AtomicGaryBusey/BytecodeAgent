export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %nn =l mul %n, %n
    %zero =l copy 0
    jmp @id_i_check
@id_i_check
    %id_i =l phi @start %zero, @id_i_inc %id_i_next
    %id_ci =w csltl %id_i, %n
    jnz %id_ci, @id_j_init, @id_done
@id_j_init
    jmp @id_j_check
@id_j_check
    %id_j =l phi @id_j_init %zero, @id_j_inc %id_j_next
    %id_cj =w csltl %id_j, %n
    jnz %id_cj, @id_j_body, @id_i_inc
@id_j_body
    %id_ij =l mul %id_i, %n
    %id_ij2 =l add %id_ij, %id_j
    %id_ij3 =l mul %id_ij2, 8
    %id_ptr =l add %out, %id_ij3
    %id_eq =w ceql %id_i, %id_j
    %id_val =l extsw %id_eq
    storel %id_val, %id_ptr
    jmp @id_j_inc
@id_j_inc
    %id_j_next =l add %id_j, 1
    jmp @id_j_check
@id_i_inc
    %id_i_next =l add %id_i, 1
    jmp @id_i_check
@id_done
    jmp @main_check
@main_check
    %iter =l phi @id_done %zero, @main_inc %iter_next
    %main_c =w csltl %iter, %k
    jnz %main_c, @mul_i_init, @main_done
@mul_i_init
    jmp @mul_i_check
@mul_i_check
    %mul_i =l phi @mul_i_init %zero, @mul_i_inc %mul_i_next
    %mul_ci =w csltl %mul_i, %n
    jnz %mul_ci, @mul_j_init, @mul_i_done
@mul_j_init
    jmp @mul_j_check
@mul_j_check
    %mul_j =l phi @mul_j_init %zero, @mul_j_inc %mul_j_next
    %mul_cj =w csltl %mul_j, %n
    jnz %mul_cj, @mul_t_init, @mul_j_done
@mul_t_init
    jmp @mul_t_check
@mul_t_check
    %mul_t =l phi @mul_t_init %zero, @mul_t_body %mul_t_next
    %mul_sum =l phi @mul_t_init %zero, @mul_t_body %mul_sum_next
    %mul_ct =w csltl %mul_t, %n
    jnz %mul_ct, @mul_t_body, @mul_t_done
@mul_t_body
    %mA_it =l mul %mul_i, %n
    %mA_it2 =l add %mA_it, %mul_t
    %mA_it3 =l mul %mA_it2, 8
    %mA_ptr =l add %out, %mA_it3
    %mA_val =l loadl %mA_ptr
    %mB_tj =l mul %mul_t, %n
    %mB_tj2 =l add %mB_tj, %mul_j
    %mB_tj3 =l mul %mB_tj2, 8
    %mB_ptr =l add %a, %mB_tj3
    %mB_val =l loadl %mB_ptr
    %mul_prod =l mul %mA_val, %mB_val
    %mul_sum_next =l add %mul_sum, %mul_prod
    %mul_t_next =l add %mul_t, 1
    jmp @mul_t_check
@mul_t_done
    %mC_ij =l mul %mul_i, %n
    %mC_ij2 =l add %mC_ij, %mul_j
    %mC_ij3 =l mul %mC_ij2, 8
    %mC_ptr =l add %tmp, %mC_ij3
    storel %mul_sum, %mC_ptr
    jmp @mul_j_inc
@mul_j_inc
    %mul_j_next =l add %mul_j, 1
    jmp @mul_j_check
@mul_j_done
    jmp @mul_i_inc
@mul_i_inc
    %mul_i_next =l add %mul_i, 1
    jmp @mul_i_check
@mul_i_done
    jmp @copy_check
@copy_check
    %cp_idx =l phi @mul_i_done %zero, @copy_inc %cp_idx_next
    %cp_c =w csltl %cp_idx, %nn
    jnz %cp_c, @copy_body, @copy_done
@copy_body
    %cp_offset =l mul %cp_idx, 8
    %cp_src =l add %tmp, %cp_offset
    %cp_val =l loadl %cp_src
    %cp_dst =l add %out, %cp_offset
    storel %cp_val, %cp_dst
    jmp @copy_inc
@copy_inc
    %cp_idx_next =l add %cp_idx, 1
    jmp @copy_check
@copy_done
    jmp @main_inc
@main_inc
    %iter_next =l add %iter, 1
    jmp @main_check
@main_done
    %two_nn =l mul %nn, 2
    jmp @zero_check
@zero_check
    %z_idx =l phi @main_done %zero, @zero_inc %z_idx_next
    %z_c =w csltl %z_idx, %two_nn
    jnz %z_c, @zero_body, @zero_done
@zero_body
    %z_offset =l mul %z_idx, 8
    %z_ptr =l add %tmp, %z_offset
    storel %zero, %z_ptr
    jmp @zero_inc
@zero_inc
    %z_idx_next =l add %z_idx, 1
    jmp @zero_check
@zero_done
    ret %zero
}
