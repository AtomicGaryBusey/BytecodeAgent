export function l $count_ugt(l %a, l %n, l %t) {
@start
    %count =l copy 0
    %i =l copy 0
@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @end
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %gt =w cugtl %val, %t
    %gt_l =l extsw %gt
    %count =l add %count, %gt_l
    %i =l add %i, 1
    jmp @loop
@end
    ret %count
}
