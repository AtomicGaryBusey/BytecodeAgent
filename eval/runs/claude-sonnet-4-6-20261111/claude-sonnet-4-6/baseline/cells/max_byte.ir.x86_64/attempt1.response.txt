export function l $max_byte(l %p, l %n) {
@start
    %cmp =w csltl %n, 1
    jnz %cmp, @ret_zero, @body_start
@ret_zero
    ret 0
@body_start
    jmp @check
@check
    %i =l phi @body_start 0, @next %i2
    %cur_max =l phi @body_start 0, @next %new_max
    %loop_cond =w csltl %i, %n
    jnz %loop_cond, @body, @done
@body
    %addr =l add %p, %i
    %b =l loadub %addr
    %gt =w csgtl %b, %cur_max
    jnz %gt, @do_update, @no_update
@do_update
    jmp @next
@no_update
    jmp @next
@next
    %new_max =l phi @do_update %b, @no_update %cur_max
    %i2 =l add %i, 1
    jmp @check
@done
    ret %cur_max
}
