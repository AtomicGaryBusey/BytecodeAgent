export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pcslot =l alloc8 8
    %dslot =l alloc8 8
    storel 0, %pcslot
    storel 0, %dslot
    jmp @loop
@loop
    %pc =l loadl %pcslot
    %depth =l loadl %dslot
    %cont =w csltl %pc, %len
    jnz %cont, @fetch, @halt
@fetch
    %pc8 =l mul %pc, 8
    %cptr =l add %code, %pc8
    %op =l loadl %cptr
    %is0 =w ceql %op, 0
    jnz %is0, @halt, @chk1
@chk1
    %is1 =w ceql %op, 1
    jnz %is1, @push, @chk2
@chk2
    %is2 =w ceql %op, 2
    jnz %is2, @add, @chk3
@chk3
    %is3 =w ceql %op, 3
    jnz %is3, @sub, @chk4
@chk4
    %is4 =w ceql %op, 4
    jnz %is4, @mul, @chk5
@chk5
    %is5 =w ceql %op, 5
    jnz %is5, @dup, @chk6
@chk6
    %is6 =w ceql %op, 6
    jnz %is6, @swap, @chk7
@chk7
    %is7 =w ceql %op, 7
    jnz %is7, @drop, @chk8
@chk8
    %is8 =w ceql %op, 8
    jnz %is8, @over, @chk9
@chk9
    %is9 =w ceql %op, 9
    jnz %is9, @lt, @chk10
@chk10
    %is10 =w ceql %op, 10
    jnz %is10, @jz, @chk11
@chk11
    %is11 =w ceql %op, 11
    jnz %is11, @jmp, @other
@other
    %pc_o =l add %pc, 1
    storel %pc_o, %pcslot
    jmp @loop
@push
    %pc1p =l add %pc, 1
    %pc1p8 =l mul %pc1p, 8
    %vptr =l add %code, %pc1p8
    %v =l loadl %vptr
    %doff_p =l mul %depth, 8
    %dstkp =l add %stack, %doff_p
    storel %v, %dstkp
    %depth1p =l add %depth, 1
    storel %depth1p, %dslot
    %pc2p =l add %pc, 2
    storel %pc2p, %pcslot
    jmp @loop
@add
    %db_a =l sub %depth, 1
    %da_a =l sub %depth, 2
    %boff_a =l mul %db_a, 8
    %aoff_a =l mul %da_a, 8
    %bptr_a =l add %stack, %boff_a
    %aptr_a =l add %stack, %aoff_a
    %bv_a =l loadl %bptr_a
    %av_a =l loadl %aptr_a
    %r_a =l add %av_a, %bv_a
    storel %r_a, %aptr_a
    storel %db_a, %dslot
    %pcn_a =l add %pc, 1
    storel %pcn_a, %pcslot
    jmp @loop
@sub
    %db_s =l sub %depth, 1
    %da_s =l sub %depth, 2
    %boff_s =l mul %db_s, 8
    %aoff_s =l mul %da_s, 8
    %bptr_s =l add %stack, %boff_s
    %aptr_s =l add %stack, %aoff_s
    %bv_s =l loadl %bptr_s
    %av_s =l loadl %aptr_s
    %r_s =l sub %av_s, %bv_s
    storel %r_s, %aptr_s
    storel %db_s, %dslot
    %pcn_s =l add %pc, 1
    storel %pcn_s, %pcslot
    jmp @loop
@mul
    %db_m =l sub %depth, 1
    %da_m =l sub %depth, 2
    %boff_m =l mul %db_m, 8
    %aoff_m =l mul %da_m, 8
    %bptr_m =l add %stack, %boff_m
    %aptr_m =l add %stack, %aoff_m
    %bv_m =l loadl %bptr_m
    %av_m =l loadl %aptr_m
    %r_m =l mul %av_m, %bv_m
    storel %r_m, %aptr_m
    storel %db_m, %dslot
    %pcn_m =l add %pc, 1
    storel %pcn_m, %pcslot
    jmp @loop
@dup
    %ti_d =l sub %depth, 1
    %toff_d =l mul %ti_d, 8
    %tptr_d =l add %stack, %toff_d
    %tv_d =l loadl %tptr_d
    %noff_d =l mul %depth, 8
    %nptr_d =l add %stack, %noff_d
    storel %tv_d, %nptr_d
    %depth1d =l add %depth, 1
    storel %depth1d, %dslot
    %pcn_d =l add %pc, 1
    storel %pcn_d, %pcslot
    jmp @loop
@swap
    %ti_sw =l sub %depth, 1
    %bi_sw =l sub %depth, 2
    %toff_sw =l mul %ti_sw, 8
    %boff_sw =l mul %bi_sw, 8
    %tptr_sw =l add %stack, %toff_sw
    %bptr_sw =l add %stack, %boff_sw
    %tv_sw =l loadl %tptr_sw
    %bv_sw =l loadl %bptr_sw
    storel %bv_sw, %tptr_sw
    storel %tv_sw, %bptr_sw
    %pcn_sw =l add %pc, 1
    storel %pcn_sw, %pcslot
    jmp @loop
@drop
    %depth1dr =l sub %depth, 1
    storel %depth1dr, %dslot
    %pcn_dr =l add %pc, 1
    storel %pcn_dr, %pcslot
    jmp @loop
@over
    %ui_ov =l sub %depth, 2
    %uoff_ov =l mul %ui_ov, 8
    %uptr_ov =l add %stack, %uoff_ov
    %uv_ov =l loadl %uptr_ov
    %noff_ov =l mul %depth, 8
    %nptr_ov =l add %stack, %noff_ov
    storel %uv_ov, %nptr_ov
    %depth1ov =l add %depth, 1
    storel %depth1ov, %dslot
    %pcn_ov =l add %pc, 1
    storel %pcn_ov, %pcslot
    jmp @loop
@lt
    %db_lt =l sub %depth, 1
    %da_lt =l sub %depth, 2
    %boff_lt =l mul %db_lt, 8
    %aoff_lt =l mul %da_lt, 8
    %bptr_lt =l add %stack, %boff_lt
    %aptr_lt =l add %stack, %aoff_lt
    %bv_lt =l loadl %bptr_lt
    %av_lt =l loadl %aptr_lt
    %cmp_w =w csltl %av_lt, %bv_lt
    %cmp_l =l extsw %cmp_w
    storel %cmp_l, %aptr_lt
    storel %db_lt, %dslot
    %pcn_lt =l add %pc, 1
    storel %pcn_lt, %pcslot
    jmp @loop
@jz
    %db_jz =l sub %depth, 1
    %voff_jz =l mul %db_jz, 8
    %vptr_jz =l add %stack, %voff_jz
    %vv_jz =l loadl %vptr_jz
    storel %db_jz, %dslot
    %pc1_jz =l add %pc, 1
    %pc1off_jz =l mul %pc1_jz, 8
    %tptr_jz =l add %code, %pc1off_jz
    %tv_jz =l loadl %tptr_jz
    %iz =w ceql %vv_jz, 0
    jnz %iz, @jztaken, @jzskip
@jztaken
    storel %tv_jz, %pcslot
    jmp @loop
@jzskip
    %pc2_jz =l add %pc, 2
    storel %pc2_jz, %pcslot
    jmp @loop
@jmp
    %pc1_jmp =l add %pc, 1
    %pc1off_jmp =l mul %pc1_jmp, 8
    %tptr_jmp =l add %code, %pc1off_jmp
    %tv_jmp =l loadl %tptr_jmp
    storel %tv_jmp, %pcslot
    jmp @loop
@halt
    %dfinal =l loadl %dslot
    jmp @zloop
@zloop
    %zi =l phi @halt %dfinal, @zbody %zi1
    %zdone =w ceql %zi, %cap
    jnz %zdone, @ret, @zbody
@zbody
    %zoff =l mul %zi, 8
    %zp =l add %stack, %zoff
    storel 0, %zp
    %zi1 =l add %zi, 1
    jmp @zloop
@ret
    %df =l loadl %dslot
    %emp =w ceql %df, 0
    jnz %emp, @ret0, @rettop
@ret0
    ret 0
@rettop
    %topi =l sub %df, 1
    %topoff =l mul %topi, 8
    %topptr =l add %stack, %topoff
    %topval =l loadl %topptr
    ret %topval
}
