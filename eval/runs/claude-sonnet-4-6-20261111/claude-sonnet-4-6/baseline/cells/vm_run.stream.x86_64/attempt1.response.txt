vm_run:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
    xor ebp, ebp
.Lloop:
    cmp rbx, r13
    jge .Ldone
    mov rax, qword ptr [r12 + rbx*8]
    test rax, rax
    jz .Ldone
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc rbx
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [r12 + rbx*8 + 8]
    mov qword ptr [r14 + rbp*8], rax
    inc rbp
    add rbx, 2
    jmp .Lloop
.Ladd:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    add qword ptr [r14 + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Lsub:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    sub qword ptr [r14 + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Lmul:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    imul rax, qword ptr [r14 + rbp*8 - 16]
    dec rbp
    mov qword ptr [r14 + rbp*8 - 8], rax
    inc rbx
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    mov qword ptr [r14 + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    mov rcx, qword ptr [r14 + rbp*8 - 16]
    mov qword ptr [r14 + rbp*8 - 8], rcx
    mov qword ptr [r14 + rbp*8 - 16], rax
    inc rbx
    jmp .Lloop
.Ldrop:
    dec rbp
    inc rbx
    jmp .Lloop
.Lover:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    mov qword ptr [r14 + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Llt:
    mov rax, qword ptr [r14 + rbp*8 - 16]
    cmp rax, qword ptr [r14 + rbp*8 - 8]
    setl al
    movzx rax, al
    dec rbp
    mov qword ptr [r14 + rbp*8 - 8], rax
    inc rbx
    jmp .Lloop
.Ljz:
    mov rax, qword ptr [r14 + rbp*8 - 8]
    dec rbp
    test rax, rax
    jnz .Ljz_next
    mov rbx, qword ptr [r12 + rbx*8 + 8]
    jmp .Lloop
.Ljz_next:
    add rbx, 2
    jmp .Lloop
.Ljmp:
    mov rbx, qword ptr [r12 + rbx*8 + 8]
    jmp .Lloop
.Ldone:
    mov rcx, r15
    sub rcx, rbp
    jle .Lskip_zero
    lea rdi, [r14 + rbp*8]
    xor eax, eax
    rep stosq
.Lskip_zero:
    test rbp, rbp
    jz .Lreturn_zero
    mov rax, qword ptr [r14 + rbp*8 - 8]
    jmp .Lexit
.Lreturn_zero:
    xor eax, eax
.Lexit:
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
