memmove_i64:
    xor eax, eax
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
.Lbackward:
    mov rcx, rdx
    dec rcx
.Lback_loop:
    test rcx, rcx
    js .Ldone
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    dec rcx
    jmp .Lback_loop
.Lforward:
    xor rcx, rcx
.Lfwd_loop:
    cmp rcx, rdx
    jge .Ldone
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    jmp .Lfwd_loop
.Ldone:
    ret
