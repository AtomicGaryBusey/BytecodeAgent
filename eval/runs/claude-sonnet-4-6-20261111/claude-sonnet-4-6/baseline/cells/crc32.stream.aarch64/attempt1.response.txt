crc32:
    mov w2, #0xFFFF
    movk w2, #0xFFFF, lsl #16
    cbz x1, .Ldone
    mov w5, #0x8320
    movk w5, #0xEDB8, lsl #16
.Lbyte_loop:
    ldrb w3, [x0], #1
    eor w2, w2, w3
    mov w4, #8
.Lbit_loop:
    tst w2, #1
    lsr w2, w2, #1
    b.eq .Lno_xor
    eor w2, w2, w5
.Lno_xor:
    subs w4, w4, #1
    b.ne .Lbit_loop
    subs x1, x1, #1
    b.ne .Lbyte_loop
.Ldone:
    mov w3, #0xFFFF
    movk w3, #0xFFFF, lsl #16
    eor w0, w2, w3
    ret
