dot_i32_simd:
    mov x3, #0
    cmp x2, #4
    b.lt .Lcheck_tail
    movi v4.2d, #0
    movi v5.2d, #0
.Lsimd_loop:
    ld1 {v0.4s}, [x0], #16
    ld1 {v1.4s}, [x1], #16
    smull v2.2d, v0.2s, v1.2s
    smull2 v3.2d, v0.4s, v1.4s
    add v4.2d, v4.2d, v2.2d
    add v5.2d, v5.2d, v3.2d
    sub x2, x2, #4
    cmp x2, #4
    b.ge .Lsimd_loop
    add v4.2d, v4.2d, v5.2d
    addp d4, v4.2d
    fmov x3, d4
.Lcheck_tail:
    cbz x2, .Ldone
.Lscalar_loop:
    ldr w4, [x0], #4
    ldr w5, [x1], #4
    sxtw x4, w4
    sxtw x5, w5
    mul x4, x4, x5
    add x3, x3, x4
    sub x2, x2, #1
    cbnz x2, .Lscalar_loop
.Ldone:
    mov x0, x3
    ret
