export function l $count_desc(l %a, l %n) {
@start
    %cmp =w csltl %n, 2
    jnz %cmp, @done_zero, @loop_init
@loop_init
    %nm1 =l sub %n, 1
    jmp @loop
@loop
    %i =l phi @loop_init 0, @loop_body %i_next
    %count =l phi @loop_init 0, @loop_body %count_next
    %cond =w csltl %i, %nm1
    jnz %cond, @loop_body, @done
@loop_body
    %off =l mul %i, 8
    %ptr =l add %a, %off
    %ai =l loadl %ptr
    %ptr2 =l add %ptr, 8
    %ai1 =l loadl %ptr2
    %desc_w =w csgtl %ai, %ai1
    %desc_l =l extuw %desc_w
    %count_next =l add %count, %desc_l
    %i_next =l add %i, 1
    jmp @loop
@done
    ret %count
@done_zero
    ret 0
}
