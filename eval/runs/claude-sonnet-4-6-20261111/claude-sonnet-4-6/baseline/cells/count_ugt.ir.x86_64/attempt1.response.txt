export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %i_next
    %count =l phi @start 0, @body %count_next
    %cond =w csltl %i, %n
    jnz %cond, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %val =l loadl %ptr
    %gt =w cugtl %val, %t
    %gt_ext =l extuw %gt
    %count_next =l add %count, %gt_ext
    %i_next =l add %i, 1
    jmp @loop
@done
    ret %count
}
