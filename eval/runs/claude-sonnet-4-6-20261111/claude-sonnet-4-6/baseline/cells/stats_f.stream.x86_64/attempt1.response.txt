stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor rbp, rbp
    mov rdi, qword ptr [r12]
    call r14
    mov qword ptr [rsp+0], rax
    mov qword ptr [rsp+8], rax
    mov qword ptr [rsp+16], rax
    and rax, 1
    add rbp, rax
    mov rbx, 1
.Lloop:
    cmp rbx, r13
    jge .Ldone
    mov rdi, qword ptr [r12 + rbx*8]
    call r14
    cmp rax, qword ptr [rsp+0]
    jge .Lskip_min
    mov qword ptr [rsp+0], rax
.Lskip_min:
    cmp rax, qword ptr [rsp+8]
    jle .Lskip_max
    mov qword ptr [rsp+8], rax
.Lskip_max:
    add qword ptr [rsp+16], rax
    mov rcx, rax
    and rcx, 1
    add rbp, rcx
    inc rbx
    jmp .Lloop
.Ldone:
    mov rax, qword ptr [rsp+0]
    mov qword ptr [r15+0], rax
    mov rax, qword ptr [rsp+8]
    mov qword ptr [r15+8], rax
    mov rax, qword ptr [rsp+16]
    mov qword ptr [r15+16], rax
    mov rax, rbp
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
