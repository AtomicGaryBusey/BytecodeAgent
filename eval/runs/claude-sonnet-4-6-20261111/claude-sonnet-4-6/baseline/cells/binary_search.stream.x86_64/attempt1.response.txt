binary_search:
  test rsi, rsi
  jle .Lnotfound
  xor rcx, rcx
  lea r8, [rsi - 1]
.Lloop:
  cmp rcx, r8
  jg .Lnotfound
  mov rax, r8
  sub rax, rcx
  sar rax, 1
  add rax, rcx
  cmp qword ptr [rdi + rax*8], rdx
  je .Lfound
  jl .Lgreater
  lea r8, [rax - 1]
  jmp .Lloop
.Lgreater:
  lea rcx, [rax + 1]
  jmp .Lloop
.Lfound:
  ret
.Lnotfound:
  mov rax, -1
  ret
