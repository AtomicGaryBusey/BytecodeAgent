export function l $isqrt(l %n) {
@start
    jmp @loop
@loop
    %lo =l phi @start 0, @take_mid %mid, @reject_mid %lo
    %hi =l phi @start 4294967296, @take_mid %hi, @reject_mid %mid
    %lo1 =l add %lo, 1
    %cond =w cultl %lo1, %hi
    jnz %cond, @body, @done
@body
    %diff =l sub %hi, %lo
    %half =l shr %diff, 1
    %mid =l add %lo, %half
    %sq =l mul %mid, %mid
    %cmp =w culel %sq, %n
    jnz %cmp, @take_mid, @reject_mid
@take_mid
    jmp @loop
@reject_mid
    jmp @loop
@done
    ret %lo
}
