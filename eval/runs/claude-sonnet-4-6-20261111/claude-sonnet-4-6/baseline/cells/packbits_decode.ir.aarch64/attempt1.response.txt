export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
  %src_end =l add %src, %n
  %out_end =l add %out, %cap
  jmp @outer_loop
@outer_loop
  %sp =l phi @start %src, @nop %sp_nop, @lit_copy_end %sp_lit, @rep_write_end %sp_rep
  %op =l phi @start %out, @nop %op_nop, @lit_copy_end %op_lit, @rep_write_end %op_rep
  %loop_done =w cugel %sp, %src_end
  jnz %loop_done, @return_written, @read_header
@read_header
  %h =w loadsb %sp
  %sp1 =l add %sp, 1
  %is_nop =w ceqw %h, -128
  jnz %is_nop, @nop, @check_sign
@nop
  %sp_nop =l copy %sp1
  %op_nop =l copy %op
  jmp @outer_loop
@check_sign
  %is_neg =w csltw %h, 0
  jnz %is_neg, @check_repeat, @literal_start
@literal_start
  %h_l =l extsw %h
  %count_lit =l add %h_l, 1
  %need_end_lit =l add %sp1, %count_lit
  %enough_lit =w culel %need_end_lit, %src_end
  jnz %enough_lit, @lit_copy, @return_error_src
@lit_copy
  %sp_lc =l phi @literal_start %sp1, @lit_copy_write %sp_lc_next
  %op_lc =l phi @literal_start %op, @lit_copy_write %op_lc_next
  %rem_lc =l phi @literal_start %count_lit, @lit_copy_write %rem_lc_next
  %lc_done =w ceql %rem_lc, 0
  jnz %lc_done, @lit_copy_end, @lit_cap_check
@lit_cap_check
  %cap_full_lc =w cugel %op_lc, %out_end
  jnz %cap_full_lc, @return_error_cap, @lit_copy_write
@lit_copy_write
  %b_lc =w loadub %sp_lc
  storeb %b_lc, %op_lc
  %sp_lc_next =l add %sp_lc, 1
  %op_lc_next =l add %op_lc, 1
  %rem_lc_next =l sub %rem_lc, 1
  jmp @lit_copy
@lit_copy_end
  %sp_lit =l copy %sp_lc
  %op_lit =l copy %op_lc
  jmp @outer_loop
@check_repeat
  %rep_need_end =l add %sp1, 1
  %enough_rep =w culel %rep_need_end, %src_end
  jnz %enough_rep, @repeat_body, @return_error_src
@repeat_body
  %rep_byte =w loadub %sp1
  %sp_after_rep =l add %sp1, 1
  %h_l_rep =l extsw %h
  %count_rep =l sub 1, %h_l_rep
  jmp @rep_write
@rep_write
  %sp_rw =l phi @repeat_body %sp_after_rep, @rep_write_write %sp_rw_same
  %op_rw =l phi @repeat_body %op, @rep_write_write %op_rw_next
  %rem_rw =l phi @repeat_body %count_rep, @rep_write_write %rem_rw_next
  %rw_done =w ceql %rem_rw, 0
  jnz %rw_done, @rep_write_end, @rep_cap_check
@rep_cap_check
  %cap_full_rw =w cugel %op_rw, %out_end
  jnz %cap_full_rw, @return_error_cap, @rep_write_write
@rep_write_write
  storeb %rep_byte, %op_rw
  %sp_rw_same =l copy %sp_rw
  %op_rw_next =l add %op_rw, 1
  %rem_rw_next =l sub %rem_rw, 1
  jmp @rep_write
@rep_write_end
  %sp_rep =l copy %sp_rw
  %op_rep =l copy %op_rw
  jmp @outer_loop
@return_written
  %written =l sub %op, %out
  ret %written
@return_error_src
  ret -2
@return_error_cap
  ret -1
}
