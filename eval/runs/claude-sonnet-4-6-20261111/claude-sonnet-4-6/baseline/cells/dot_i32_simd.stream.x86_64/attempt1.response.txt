dot_i32_simd:
    xor rax, rax
    test rdx, rdx
    jz .Lreturn
    pxor xmm4, xmm4
    mov r8, rdx
    shr r8, 2
    xor rcx, rcx
    test r8, r8
    jz .Lsimd_done
.Lsimd_loop:
    movdqu xmm0, [rdi + rcx*4]
    movdqu xmm1, [rsi + rcx*4]
    movdqa xmm2, xmm0
    movdqa xmm3, xmm1
    pmuldq xmm2, xmm3
    paddq xmm4, xmm2
    pshufd xmm0, xmm0, 0xF5
    pshufd xmm1, xmm1, 0xF5
    pmuldq xmm0, xmm1
    paddq xmm4, xmm0
    add rcx, 4
    dec r8
    jnz .Lsimd_loop
.Lsimd_done:
    movq rax, xmm4
    psrldq xmm4, 8
    movq r9, xmm4
    add rax, r9
.Lscalar_loop:
    cmp rcx, rdx
    jge .Lreturn
    movsxd r9, dword ptr [rdi + rcx*4]
    movsxd r10, dword ptr [rsi + rcx*4]
    imul r9, r10
    add rax, r9
    inc rcx
    jmp .Lscalar_loop
.Lreturn:
    ret
