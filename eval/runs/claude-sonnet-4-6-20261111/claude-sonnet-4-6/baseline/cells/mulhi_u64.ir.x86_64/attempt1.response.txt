export function l $mulhi_u64(l %a, l %b) {
@start
    %a_lo =l and %a, 4294967295
    %a_hi =l shr %a, 32
    %b_lo =l and %b, 4294967295
    %b_hi =l shr %b, 32
    %p0 =l mul %a_lo, %b_lo
    %p1 =l mul %a_lo, %b_hi
    %p2 =l mul %a_hi, %b_lo
    %p3 =l mul %a_hi, %b_hi
    %mid =l add %p1, %p2
    %cm_w =w cultl %mid, %p1
    %carry_mid =l extuw %cm_w
    %mid_lo =l shl %mid, 32
    %t =l add %p0, %mid_lo
    %ct_w =w cultl %t, %p0
    %carry_t =l extuw %ct_w
    %mid_hi =l shr %mid, 32
    %cm32 =l shl %carry_mid, 32
    %r1 =l add %p3, %mid_hi
    %r2 =l add %r1, %cm32
    %r3 =l add %r2, %carry_t
    ret %r3
}
