export function l $bubble_sort(l %a, l %n) {
@start
    %iptr =l alloc8 8
    %jptr =l alloc8 8
    storel 0, %iptr
    jmp @outer_check
@outer_check
    %i =l loadl %iptr
    %n1 =l sub %n, 1
    %c1 =w csltl %i, %n1
    jnz %c1, @outer_body, @done
@outer_body
    storel 0, %jptr
    %inner_limit =l sub %n1, %i
    jmp @inner_check
@inner_check
    %j =l loadl %jptr
    %c2 =w csltl %j, %inner_limit
    jnz %c2, @inner_body, @outer_inc
@inner_body
    %j8 =l mul %j, 8
    %addr_j =l add %a, %j8
    %j1 =l add %j, 1
    %j18 =l mul %j1, 8
    %addr_j1 =l add %a, %j18
    %vj =l loadl %addr_j
    %vj1 =l loadl %addr_j1
    %cmp =w csgtl %vj, %vj1
    jnz %cmp, @do_swap, @inner_inc
@do_swap
    storel %vj1, %addr_j
    storel %vj, %addr_j1
    jmp @inner_inc
@inner_inc
    %jcur =l loadl %jptr
    %jnxt =l add %jcur, 1
    storel %jnxt, %jptr
    jmp @inner_check
@outer_inc
    %icur =l loadl %iptr
    %inxt =l add %icur, 1
    storel %inxt, %iptr
    jmp @outer_check
@done
    ret 0
}
