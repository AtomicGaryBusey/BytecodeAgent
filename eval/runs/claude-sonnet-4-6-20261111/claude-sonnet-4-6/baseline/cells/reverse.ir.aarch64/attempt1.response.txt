export function l $reverse(l %a, l %n) {
@start
    %left =l copy 0
    %right =l sub %n, 1
    jmp @check
@check
    %cond =w csltl %left, %right
    jnz %cond, @body, @end
@body
    %left_off =l mul %left, 8
    %left_addr =l add %a, %left_off
    %right_off =l mul %right, 8
    %right_addr =l add %a, %right_off
    %lval =l loadl %left_addr
    %rval =l loadl %right_addr
    storel %rval, %left_addr
    storel %lval, %right_addr
    %left =l add %left, 1
    %right =l sub %right, 1
    jmp @check
@end
    ret 0
}
