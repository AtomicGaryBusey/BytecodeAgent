map_into:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov rbp, rdx
    mov r14, rcx
    xor r15d, r15d
    xor ebx, ebx
    test rbp, rbp
    jz .Ldone
.Lloop:
    mov rax, qword ptr [r13 + rbx*8]
    mov qword ptr [rsp], rax
    mov rdi, rax
    call r14
    mov qword ptr [r12 + rbx*8], rax
    cmp rax, qword ptr [rsp]
    jge .Lnext
    inc r15
.Lnext:
    inc rbx
    cmp rbx, rbp
    jl .Lloop
.Ldone:
    mov rax, r15
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret
