map_into:
  stp x29, x30, [sp, #-64]!
  mov x29, sp
  stp x19, x20, [sp, #16]
  stp x21, x22, [sp, #32]
  stp x23, x24, [sp, #48]
  mov x19, x0                 // dst pointer
  mov x20, x1                 // src pointer
  mov x21, x2                 // remaining count
  mov x22, x3                 // f
  mov x23, #0                 // result count
.Lloop:
  cbz x21, .Ldone
  ldr x24, [x20]              // src[i], kept across call
  mov x0, x24
  blr x22
  str x0, [x19]               // dst[i] = f(src[i])
  cmp x0, x24
  cset x9, lt
  add x23, x23, x9
  add x19, x19, #8
  add x20, x20, #8
  sub x21, x21, #1
  b .Lloop
.Ldone:
  mov x0, x23
  ldp x23, x24, [sp, #48]
  ldp x21, x22, [sp, #32]
  ldp x19, x20, [sp, #16]
  ldp x29, x30, [sp], #64
  ret
