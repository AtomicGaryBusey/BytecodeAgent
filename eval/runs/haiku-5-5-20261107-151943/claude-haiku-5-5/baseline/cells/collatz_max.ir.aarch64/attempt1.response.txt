export function l $collatz_max(l %n) {
@start
	%best =l alloc8 8
	%k =l alloc8 8
	%x =l alloc8 8
	%steps =l alloc8 8
	storel 0, %best
	storel 1, %k
@outer
	%kv =l loadl %k
	%c =w csgtl %kv, %n
	jnz %c, @done, @body
@body
	storel %kv, %x
	storel 0, %steps
@inner
	%xv =l loadl %x
	%one =w ceql %xv, 1
	jnz %one, @innerdone, @step
@step
	%par =l and %xv, 1
	%ev =w ceql %par, 0
	jnz %ev, @even, @odd
@even
	%h =l shr %xv, 1
	storel %h, %x
	jmp @cont
@odd
	%t =l mul %xv, 3
	%u =l add %t, 1
	storel %u, %x
	jmp @cont
@cont
	%sv =l loadl %steps
	%sn =l add %sv, 1
	storel %sn, %steps
	jmp @inner
@innerdone
	%sf =l loadl %steps
	%bv =l loadl %best
	%gt =w csgtl %sf, %bv
	jnz %gt, @upd, @next
@upd
	storel %sf, %best
	jmp @next
@next
	%kn =l loadl %k
	%kp =l add %kn, 1
	storel %kp, %k
	jmp @outer
@done
	%r =l loadl %best
	ret %r
}
