export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%t0 =l shr %p00, 32
	%t1 =l and %p01, 4294967295
	%t2 =l and %p10, 4294967295
	%s1 =l add %t0, %t1
	%mid =l add %s1, %t2
	%m1 =l shr %mid, 32
	%h1 =l shr %p01, 32
	%h2 =l shr %p10, 32
	%r1 =l add %p11, %h1
	%r2 =l add %r1, %h2
	%r3 =l add %r2, %m1
	ret %r3
}
