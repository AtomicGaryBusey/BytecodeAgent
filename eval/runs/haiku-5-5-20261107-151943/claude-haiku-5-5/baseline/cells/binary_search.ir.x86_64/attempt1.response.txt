export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l alloc8 8
	%hi =l alloc8 8
	storel 0, %lo
	storel %n, %hi
@loop
	%l =l loadl %lo
	%h =l loadl %hi
	%c =w csltl %l, %h
	jnz %c, @body, @notfound
@body
	%d =l sub %h, %l
	%half =l sar %d, 1
	%mid =l add %l, %half
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @check
@check
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%m1 =l add %mid, 1
	storel %m1, %lo
	jmp @loop
@left
	storel %mid, %hi
	jmp @loop
@found
	ret %mid
@notfound
	%neg =l copy -1
	ret %neg
}
