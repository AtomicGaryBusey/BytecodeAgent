export function l $qsort_pairs(l %a, l %n) {
@start
	%ip =l alloc8 8
	%jp =l alloc8 8
	%c =w csgel %n, 2
	jnz %c, @work, @ret0
@ret0
	ret 0
@work
	%nm1 =l sub %n, 1
	%off =l mul %nm1, 16
	%pp =l add %a, %off
	%pvp =l add %pp, 8
	%pk =l loadl %pp
	%pv =l loadl %pvp
	storel %a, %ip
	storel %a, %jp
	jmp @loop
@loop
	%j =l loadl %jp
	%c2 =w cultl %j, %pp
	jnz %c2, @body, @after
@body
	%jk =l loadl %j
	%jv8 =l add %j, 8
	%jv =l loadl %jv8
	%lt1 =w csltl %jk, %pk
	%eq =w ceql %jk, %pk
	%lt2 =w csltl %jv, %pv
	%t =w and %eq, %lt2
	%less =w or %lt1, %t
	jnz %less, @swap, @next
@swap
	%si =l loadl %ip
	%siv8 =l add %si, 8
	%sik =l loadl %si
	%siv =l loadl %siv8
	storel %jk, %si
	storel %jv, %siv8
	storel %sik, %j
	storel %siv, %jv8
	%si2 =l add %si, 16
	storel %si2, %ip
	jmp @next
@next
	%j2 =l add %j, 16
	storel %j2, %jp
	jmp @loop
@after
	%ai =l loadl %ip
	%aiv8 =l add %ai, 8
	%aik =l loadl %ai
	%aiv =l loadl %aiv8
	storel %pk, %ai
	storel %pv, %aiv8
	storel %aik, %pp
	storel %aiv, %pvp
	%d =l sub %ai, %a
	%cnt =l div %d, 16
	call $qsort_pairs(l %a, l %cnt)
	%r =l add %ai, 16
	%rem =l sub %n, %cnt
	%rem2 =l sub %rem, 1
	call $qsort_pairs(l %r, l %rem2)
	ret 0
}
