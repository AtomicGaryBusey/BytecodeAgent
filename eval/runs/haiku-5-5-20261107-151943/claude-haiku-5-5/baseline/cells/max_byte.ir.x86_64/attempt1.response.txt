export function l $max_byte(l %p, l %n) {
@start
	%best =l copy 0
	%i =l copy 0
@loop
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%addr =l add %p, %i
	%b =l loadub %addr
	%gt =w csgtl %b, %best
	jnz %gt, @update, @next
@update
	%best =l copy %b
@next
	%i =l add %i, 1
	jmp @loop
@end
	ret %best
}
