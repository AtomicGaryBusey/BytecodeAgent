push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov r12, rsi
mov r13, rdx
mov r14, rcx
xor r15d, r15d
xor ebp, ebp
.Lloop:
cmp rbp, r13
jge .Ldone
mov rdi, qword ptr [r12 + rbp*8]
call r14
