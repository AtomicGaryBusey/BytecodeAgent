export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @g3 %inext
	%j =l phi @start 0, @g3 %jnext
	%rem =l sub %n, %i
	%more =w csgtl %rem, 0
	jnz %more, @group, @done
@group
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%has1 =w csgel %rem, 2
	%has2 =w csgel %rem, 3
	jnz %has1, @ld1, @g2
@ld1
	%p1 =l add %p0, 1
	%b1l =l loadub %p1
	jmp @g2
@g2
	%b1 =l phi @group 0, @ld1 %b1l
	jnz %has2, @ld2, @g3
@ld2
	%p2 =l add %p0, 2
	%b2l =l loadub %p2
	jmp @g3
@g3
	%b2 =l phi @g2 0, @ld2 %b2l
	%hi =l shl %b0, 16
	%mid =l shl %b1, 8
	%v1 =l or %hi, %mid
	%v =l or %v1, %b2
	%s0 =l shr %v, 18
	%s1a =l shr %v, 12
	%s1 =l and %s1a, 63
	%s2a =l shr %v, 6
	%s2 =l and %s2a, 63
	%s3 =l and %v, 63
	%a0 =w csgel %s0, 26
	%b0c =w csgel %s0, 52
	%c0 =w csgel %s0, 62
	%d0 =w csgel %s0, 63
	%a0l =l extuw %a0
	%b0l2 =l extuw %b0c
	%c0l =l extuw %c0
	%d0l =l extuw %d0
	%ta0 =l mul %a0l, 6
	%tb0 =l mul %b0l2, -75
	%tc0 =l mul %c0l, -15
	%td0 =l mul %d0l, 3
	%oa0 =l add %ta0, %tb0
	%ob0 =l add %oa0, %tc0
	%oc0 =l add %ob0, %td0
	%od0 =l add %oc0, 65
	%e0 =l add %s0, %od0
	%a1 =w csgel %s1, 26
	%b1c =w csgel %s1, 52
	%c1 =w csgel %s1, 62
	%d1 =w csgel %s1, 63
	%a1l =l extuw %a1
	%b1l2 =l extuw %b1c
	%c1l =l extuw %c1
	%d1l =l extuw %d1
	%ta1 =l mul %a1l, 6
	%tb1 =l mul %b1l2, -75
	%tc1 =l mul %c1l, -15
	%td1 =l mul %d1l, 3
	%oa1 =l add %ta1, %tb1
	%ob1 =l add %oa1, %tc1
	%oc1 =l add %ob1, %td1
	%od1 =l add %oc1, 65
	%e1 =l add %s1, %od1
	%a2 =w csgel %s2, 26
	%b2c =w csgel %s2, 52
	%c2 =w csgel %s2, 62
	%d2 =w csgel %s2, 63
	%a2l =l extuw %a2
	%b2l2 =l extuw %b2c
	%c2l =l extuw %c2
	%d2l =l extuw %d2
	%ta2 =l mul %a2l, 6
	%tb2 =l mul %b2l2, -75
	%tc2 =l mul %c2l, -15
	%td2 =l mul %d2l, 3
	%oa2 =l add %ta2, %tb2
	%ob2 =l add %oa2, %tc2
	%oc2 =l add %ob2, %td2
	%od2 =l add %oc2, 65
	%e2 =l add %s2, %od2
	%a3 =w csgel %s3, 26
	%b3c =w csgel %s3, 52
	%c3 =w csgel %s3, 62
	%d3 =w csgel %s3, 63
	%a3l =l extuw %a3
	%b3l2 =l extuw %b3c
	%c3l =l extuw %c3
	%d3l =l extuw %d3
	%ta3 =l mul %a3l, 6
	%tb3 =l mul %b3l2, -75
	%tc3 =l mul %c3l, -15
	%td3 =l mul %d3l, 3
	%oa3 =l add %ta3, %tb3
	%ob3 =l add %oa3, %tc3
	%oc3 =l add %ob3, %td3
	%od3 =l add %oc3, 65
	%e3 =l add %s3, %od3
	%has1l =l extuw %has1
	%has2l =l extuw %has2
	%m2 =l sub %e2, 61
	%mm2 =l mul %has1l, %m2
	%c2v =l add %mm2, 61
	%m3 =l sub %e3, 61
	%mm3 =l mul %has2l, %m3
	%c3v =l add %mm3, 61
	%q =l add %out, %j
	%w0 =w copy %e0
	storeb %w0, %q
	%q1 =l add %q, 1
	%w1 =w copy %e1
	storeb %w1, %q1
	%q2 =l add %q, 2
	%w2 =w copy %c2v
	storeb %w2, %q2
	%q3 =l add %q, 3
	%w3 =w copy %c3v
	storeb %w3, %q3
	%jnext =l add %j, 4
	%inext =l add %i, 3
	jmp @loop
@done
	ret %j
}
