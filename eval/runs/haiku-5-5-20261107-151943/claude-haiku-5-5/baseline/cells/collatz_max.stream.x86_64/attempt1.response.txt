mov r8, 1
.Louter:
cmp r8, rdi
jg .Ldone
mov r10, r8
xor r11, r11
.Linner:
cmp r10, 1
je .Lcheck
test r10, 1
jnz .Lodd
shr r10, 1
jmp .Lstep
.Lodd:
lea r10, [r10 + r10*2 + 1]
.Lstep:
inc r11
jmp .Linner
.Lcheck:
cmp r11, r9
jle .Lnext
mov r9, r11
.Lnext:

jmp .Louter
.Ldone:
mov rax, r9
ret
