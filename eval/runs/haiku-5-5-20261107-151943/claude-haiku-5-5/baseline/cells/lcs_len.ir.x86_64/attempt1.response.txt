export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%np1 =l add %n, 1
	jmp @zloop
@zloop
	%zj =l phi @start 0, @zloop %zjn
	%zp =l mul %zj, 8
	%za =l add %row, %zp
	storel 0, %za
	%zjn =l add %zj, 1
	%zc =w csltl %zjn, %np1
	jnz %zc, @zloop, @zdone
@zdone
	jmp @oloop
@oloop
	%i =l phi @zdone 0, @onext %inext
	%ic =w csltl %i, %m
	jnz %ic, @obody, @done
@obody
	%xp =l add %x, %i
	%xi =w loadub %xp
	%hn =w csgtl %n, 0
	jnz %hn, @jloop, @onext
@jloop
	%j =l phi @obody 1, @store %jn
	%diag =l phi @obody 0, @store %up
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%lp =l mul %jm1, 8
	%lpa =l add %row, %lp
	%left =l loadl %lpa
	%jp =l mul %j, 8
	%ra =l add %row, %jp
	%up =l loadl %ra
	%eq =w ceqw %xi, %yb
	jnz %eq, @match, @nomatch
@match
	%mv =l add %diag, 1
	jmp @store
@nomatch
	%c2 =w csltl %up, %left
	jnz %c2, @useleft, @useup
@useleft
	jmp @store
@useup
	jmp @store
@store
	%nv =l phi @match %mv, @useleft %left, @useup %up
	storel %nv, %ra
	%jn =l add %j, 1
	%jc =w csltl %jn, %np1
	jnz %jc, @jloop, @onext
@onext
	%inext =l add %i, 1
	jmp @oloop
@done
	%nn8 =l mul %n, 8
	%ra2 =l add %row, %nn8
	%res =l loadl %ra2
	ret %res
}
