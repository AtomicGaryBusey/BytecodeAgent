memmove_i64:
    cmp rdi, rsi
    jbe .Lfwd
    lea r8, [rsi + rdx*8]
    cmp rdi, r8
    jae .Lfwd
    mov rcx, rdx
.Lback:
    test rcx, rcx
    jz .Ldone
    dec rcx
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    jmp .Lback
.Lfwd:
    xor ecx, ecx
.Lfloop:
    cmp rcx, rdx
    jge .Ldone
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    jmp .Lfloop
.Ldone:
    xor eax, eax
    ret
