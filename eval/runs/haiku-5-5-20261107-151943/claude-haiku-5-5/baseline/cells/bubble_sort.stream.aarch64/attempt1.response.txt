bubble_sort:
    mov x9, x1
.Louter:
    cmp x9, #1
    b.le .Ldone
    mov x10, #0
    sub x11, x9, #1
.Linner:
    cmp x10, x11
    b.ge .Lendinner
    ldr x12, [x0, x10, lsl #3]
    add x13, x10, #1
    ldr x14, [x0, x13, lsl #3]
    cmp x12, x14
    b.le .Lnoswap
    str x14, [x0, x10, lsl #3]
    str x12, [x0, x13, lsl #3]
.Lnoswap:
    mov x10, x13
    b .Linner
.Lendinner:
    sub x9, x9, #1
    b .Louter
.Ldone:
    mov x0, #0
    ret
