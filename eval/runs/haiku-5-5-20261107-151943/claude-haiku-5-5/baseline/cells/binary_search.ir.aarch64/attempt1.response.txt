export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	%hi0 =l copy %n
	jmp @loop
@loop
	%lo =l phi @start %lo0, @right %lo2, @left %lo
	%hi =l phi @start %hi0, @right %hi, @left %mid
	%c =w csltl %lo, %hi
	jnz %c, @body, @notfound
@body
	%d =l sub %hi, %lo
	%h =l sar %d, 1
	%mid =l add %lo, %h
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @check
@check
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%lo2 =l add %mid, 1
	jmp @loop
@left
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
