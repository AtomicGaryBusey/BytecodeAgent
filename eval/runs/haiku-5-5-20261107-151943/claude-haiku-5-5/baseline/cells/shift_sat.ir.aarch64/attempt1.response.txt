export function l $shift_sat(l %x, l %n) {
@start
	%big =w csgel %n, 64
	jnz %big, @large, @small
@small
	%s =l shl %x, %n
	%r =l sar %x, %n
	jmp @join
@large
	%z =l copy 0
	%c =l sar %x, 63
	jmp @join
@join
	%sl =l phi @small %s, @large %z
	%sr =l phi @small %r, @large %c
	%res =l add %sl, %sr
	ret %res
}
