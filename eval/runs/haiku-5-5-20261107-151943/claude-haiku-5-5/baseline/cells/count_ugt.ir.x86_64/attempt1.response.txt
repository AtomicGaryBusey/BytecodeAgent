export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @cond
@cond
	%i =l phi @start 0, @body %i2
	%cnt =l phi @start 0, @body %cnt2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%g =w cugtl %v, %t
	%ge =l extuw %g
	%cnt2 =l add %cnt, %ge
	%i2 =l add %i, 1
	jmp @cond
@done
	ret %cnt
}
