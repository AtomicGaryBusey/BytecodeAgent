export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %knext
	%m =l phi @start 0, @next %mnext
	%done =w csgtl %k, %n
	jnz %done, @end, @body
@body
	jmp @inner
@inner
	%x =l phi @body %k, @even %xe, @odd %xo
	%s =l phi @body 0, @even %se, @odd %so
	%one =w ceql %x, 1
	jnz %one, @count, @step
@step
	%par =l and %x, 1
	jnz %par, @odd, @even
@even
	%xe =l shr %x, 1
	%se =l add %s, 1
	jmp @inner
@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	%so =l add %s, 1
	jmp @inner
@count
	%gt =w csgtl %s, %m
	jnz %gt, @upd, @keep
@upd
	jmp @next
@keep
	jmp @next
@next
	%mnext =l phi @upd %s, @keep %m
	%knext =l add %k, 1
	jmp @outer
@end
	ret %m
}
