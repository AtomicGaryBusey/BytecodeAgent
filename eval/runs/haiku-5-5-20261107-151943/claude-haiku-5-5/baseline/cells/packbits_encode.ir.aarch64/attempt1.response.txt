export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	%pj =l alloc8 8
	%pr =l alloc8 8
	%pk =l alloc8 8
	storel 0, %pi
	storel 0, %po
	jmp @outer
@outer
	%ci =l loadl %pi
	%c =w csltl %ci, %n
	jnz %c, @o_body, @done_ok
@o_body
	%i =l loadl %pi
	%sa =l add %src, %i
	%b0 =w loadub %sa
	storel 1, %pr
	jmp @runloop
@runloop
	%rv =l loadl %pr
	%ii =l add %i, %rv
	%c1 =w csltl %ii, %n
	jnz %c1, @rl1, @rend
@rl1
	%c2 =w csltl %rv, 128
	jnz %c2, @rl2, @rend
@rl2
	%pa =l add %src, %ii
	%bb =w loadub %pa
	%c3 =w ceqw %bb, %b0
	jnz %c3, @rinc, @rend
@rinc
	%rn =l add %rv, 1
	storel %rn, %pr
	jmp @runloop
@rend
	%rf =l loadl %pr
	%c4 =w csgel %rf, 3
	jnz %c4, @rep, @lit
@rep
	%rr =l loadl %pr
	%o =l loadl %po
	%o2 =l add %o, 2
	%fits =w cslel %o2, %cap
	jnz %fits, @rep_w, @fail
@rep_w
	%rr2 =l loadl %pr
	%hv =l sub 257, %rr2
	%hw =w copy %hv
	%rp =l loadl %po
	%dst =l add %out, %rp
	storeb %hw, %dst
	%rp1 =l add %rp, 1
	%dst2 =l add %out, %rp1
	storeb %b0, %dst2
	%rp2 =l add %rp, 2
	storel %rp2, %po
	%ri =l loadl %pi
	%ni =l add %ri, %rr2
	storel %ni, %pi
	jmp @outer
@lit
	%ij =l add %i, 1
	storel %ij, %pj
	jmp @litloop
@litloop
	%j =l loadl %pj
	%cA =w csltl %j, %n
	jnz %cA, @la, @litend
@la
	%dj =l sub %j, %i
	%cB =w csltl %dj, 128
	jnz %cB, @lb, @litend
@lb
	%jp2 =l add %j, 2
	%cC =w csltl %jp2, %n
	jnz %cC, @lc, @ladv
@lc
	%q0 =l add %src, %j
	%v0 =w loadub %q0
	%q1 =l add %q0, 1
	%v1 =w loadub %q1
	%q2 =l add %q0, 2
	%v2 =w loadub %q2
	%e1 =w ceqw %v1, %v0
	%e2 =w ceqw %v2, %v0
	%e =w and %e1, %e2
	jnz %e, @litend, @ladv
@ladv
	%jn =l add %j, 1
	storel %jn, %pj
	jmp @litloop
@litend
	%jf =l loadl %pj
	%L =l sub %jf, %i
	%lo =l loadl %po
	%ob =l add %lo, 1
	%oe =l add %ob, %L
	%fit =w cslel %oe, %cap
	jnz %fit, @lcopy, @fail
@lcopy
	%Lm =l sub %L, 1
	%hlw =w copy %Lm
	%ldst =l add %out, %lo
	storeb %hlw, %ldst
	storel 0, %pk
	jmp @cp
@cp
	%k =l loadl %pk
	%ck =w csltl %k, %L
	jnz %ck, @cpb, @cpend
@cpb
	%k2 =l loadl %pk
	%sa2 =l add %src, %i
	%sk =l add %sa2, %k2
	%bv =w loadub %sk
	%dk =l add %out, %ob
	%dk2 =l add %dk, %k2
	storeb %bv, %dk2
	%kn =l add %k2, 1
	storel %kn, %pk
	jmp @cp
@cpend
	storel %oe, %po
	storel %jf, %pi
	jmp @outer
@fail
	ret -1
@done_ok
	%ofin =l loadl %po
	ret %ofin
}
