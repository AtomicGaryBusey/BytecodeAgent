export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%mnp =l alloc8 8
	%mxp =l alloc8 8
	%sp =l alloc8 8
	%cp =l alloc8 8
	%ip =l alloc8 8
	%m1 =l copy -9223372036854775807
	%mx0 =l sub %m1, 1
	storel 9223372036854775807, %mnp
	storel %mx0, %mxp
	storel 0, %sp
	storel 0, %cp
	storel 0, %ip
	jmp @head
@head
	%i =l loadl %ip
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%s =l loadl %sp
	%s2 =l add %s, %y
	storel %s2, %sp
	%od =l and %y, 1
	%cn =l loadl %cp
	%cn2 =l add %cn, %od
	storel %cn2, %cp
	%mn =l loadl %mnp
	%lt =w csltl %y, %mn
	jnz %lt, @setmn, @chkmx
@setmn
	storel %y, %mnp
	jmp @chkmx
@chkmx
	%mx =l loadl %mxp
	%gt =w csgtl %y, %mx
	jnz %gt, @setmx, @inc
@setmx
	storel %y, %mxp
	jmp @inc
@inc
	%i2 =l add %i, 1
	storel %i2, %ip
	jmp @head
@done
	%mnv =l loadl %mnp
	%mxv =l loadl %mxp
	%sv =l loadl %sp
	%cv =l loadl %cp
	storel %mnv, %out
	%o1 =l add %out, 8
	storel %mxv, %o1
	%o2 =l add %out, 16
	storel %sv, %o2
	ret %cv
}
