bubble_sort:
    mov r8, rsi
    sub r8, 1
.Louter:
    test r8, r8
    jle .Ldone
    xor ecx, ecx
.Linner:
    cmp rcx, r8
    jge .Lnext
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, qword ptr [rdi + rcx*8 + 8]
    cmp rax, rdx
    jle .Lskip
    mov qword ptr [rdi + rcx*8], rdx
    mov qword ptr [rdi + rcx*8 + 8], rax
.Lskip:
    add rcx, 1
    jmp .Linner
.Lnext:
    sub r8, 1
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
