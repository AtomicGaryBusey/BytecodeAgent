dot_i32_simd:

    movi v5.2d, #0
    lsr x3, x2, #2
    cbz x3, .Lreduce
.Lvloop:

    ldr q1, [x1], #16
    smlal v4.2d, v0.2s, v1.2s
    smlal2 v5.2d, v0.4s, v1.4s

.Lreduce:

    addp d4, v4.2d

    subs x5, x5, #1
    b.ne .Lsloop
.Ldone:
    mov x0, x4
