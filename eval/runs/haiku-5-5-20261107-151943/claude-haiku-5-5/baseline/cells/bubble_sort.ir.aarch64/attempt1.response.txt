export function l $bubble_sort(l %a, l %n) {
@start
	%ip =l alloc8 8
	%jp =l alloc8 8
	storel %n, %ip
@outer
	%i =l loadl %ip
	%oc =w csgtl %i, 1
	jnz %oc, @outer_body, @done
@outer_body
	storel 0, %jp
	jmp @inner
@inner
	%j =l loadl %jp
	%lim =l loadl %ip
	%lim2 =l sub %lim, 1
	%ic =w csltl %j, %lim2
	jnz %ic, @inner_body, @outer_next
@inner_body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%g =w csgtl %x, %y
	jnz %g, @swap, @inner_next
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner_next
@inner_next
	%j2 =l loadl %jp
	%j3 =l add %j2, 1
	storel %j3, %jp
	jmp @inner
@outer_next
	%i2 =l loadl %ip
	%i3 =l sub %i2, 1
	storel %i3, %ip
	jmp @outer
@done
	ret 0
}
