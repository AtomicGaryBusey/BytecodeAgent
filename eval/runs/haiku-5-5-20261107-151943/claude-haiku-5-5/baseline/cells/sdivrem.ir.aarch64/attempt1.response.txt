export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%z =w ceql %b, 0
	%m1 =w ceql %b, -1
	%mn =l shl 1, 63
	%am =w ceql %a, %mn
	%ov =w and %m1, %am
	%sp =w or %z, %ov
	jnz %sp, @special, @normal
@special
	jnz %z, @zcase, @ocase
@zcase
	storel %a, %rem
	ret 0
@ocase
	storel 0, %rem
	ret %mn
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
