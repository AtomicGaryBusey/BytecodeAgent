export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @latch %i2
	%m =l phi @start 0, @latch %m2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%a =l add %p, %i
	%b =l loadub %a
	%g =w cugtl %b, %m
	jnz %g, @upd, @latch
@upd
	jmp @latch
@latch
	%m2 =l phi @body %m, @upd %b
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %m
}
