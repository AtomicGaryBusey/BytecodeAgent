isqrt:
	xor eax, eax # r = 0
	mov rcx, 1
	shl rcx, 62 # bit = 1 << 62
.Lfind:
	cmp rcx, rdi
	jbe .Lloop # while bit > n: bit >>= 2
	shr rcx, 2
	jmp .Lfind
.Lloop:
	test rcx, rcx
	jz .Ldone
	lea rdx, [rax + rcx] # t = r + bit
	cmp rdi, rdx
	jb .Lelse # if n < t: r >>= 1
	sub rdi, rdx # n -= t
	shr rax, 1
	add rax, rcx # r = (r >> 1) + bit
	jmp .Lnext
.Lelse:
	shr rax, 1
.Lnext:
	shr rcx, 2
	jmp .Lloop
.Ldone:
	ret
