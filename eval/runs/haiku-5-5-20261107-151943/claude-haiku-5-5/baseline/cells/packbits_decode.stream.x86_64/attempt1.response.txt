packbits_decode:
  xor r8d, r8d
  xor r9d, r9d
.Lloop:
  cmp r8, rsi
  jge .Ldone
  movsx r10, byte ptr [rdi + r8]
  inc r8
  cmp r10, -128
  je .Lloop
  cmp r10, 0
  jl .Lrep
  lea r11, [r10 + 1]
  lea rax, [r8 + r11]
  cmp rax, rsi
  jg .Lerr2
  mov rax, rcx
  sub rax, r9
  cmp r11, rax
  jg .Llit_part
.Llit_full:
  movzx r10d, byte ptr [rdi + r8]
  mov byte ptr [rdx + r9], r10b
  inc r8
  inc r9
  dec r11
  jnz .Llit_full
  jmp .Lloop
.Llit_part:
  mov r11, rax
  test r11, r11
  jz .Lovf
.Llit_pl:
  movzx r10d, byte ptr [rdi + r8]
  mov byte ptr [rdx + r9], r10b
  inc r8
  inc r9
  dec r11
  jnz .Llit_pl
  jmp .Lovf
.Lrep:
  cmp r8, rsi
  jge .Lerr2
  mov r11, 1
  sub r11, r10
  movzx r10d, byte ptr [rdi + r8]
  inc r8
  mov rax, rcx
  sub rax, r9
  cmp r11, rax
  jg .Lrep_part
.Lrep_full:
  mov byte ptr [rdx + r9], r10b
  inc r9
  dec r11
  jnz .Lrep_full
  jmp .Lloop
.Lrep_part:
  mov r11, rax
  test r11, r11
  jz .Lovf
.Lrep_pl:
  mov byte ptr [rdx + r9], r10b
  inc r9
  dec r11
  jnz .Lrep_pl
  jmp .Lovf
.Ldone:
  mov rax, r9
  ret
.Lerr2:
  mov rax, -2
  ret
.Lovf:
  mov rax, -1
  ret
