export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i0 =l copy 0
	%c0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%c =l phi @start %c0, @body %c1
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%x =l loadl %sp
	%y =l call %f(l %x)
	%dp =l add %dst, %off
	storel %y, %dp
	%lt =w csltl %y, %x
	%ltl =l extuw %lt
	%c1 =l add %c, %ltl
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %c
}
