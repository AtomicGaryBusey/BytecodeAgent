isqrt:
    mov x1, x0
    mov x2, #0
    mov x3, #1
    lsl x3, x3, #62
.Lshrink:
    cmp x3, x1
    b.ls .Lloop
    lsr x3, x3, #2
    b .Lshrink
.Lloop:
    cbz x3, .Ldone
    add x4, x2, x3
    cmp x1, x4
    b.lo .Lelse
    sub x1, x1, x4
    lsr x2, x2, #1
    add x2, x2, x3
    b .Lnext
.Lelse:
    lsr x2, x2, #1
.Lnext:
    lsr x3, x3, #2
    b .Lloop
.Ldone:
    mov x0, x2
    ret
