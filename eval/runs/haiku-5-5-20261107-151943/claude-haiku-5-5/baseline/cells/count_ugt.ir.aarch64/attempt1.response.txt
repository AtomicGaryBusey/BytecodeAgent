export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%c =l phi @start 0, @body %cnext
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%g =w cugtl %v, %t
	%gl =l extuw %g
	%cnext =l add %c, %gl
	%inext =l add %i, 1
	jmp @loop
@end
	ret %c
}
