export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%fwd =w cultl %dst, %src
	jnz %fwd, @fwd_loop, @bwd_loop
@fwd_loop
	%i =l phi @start 0, @fwd_body %i1
	%c =w csltl %i, %n
	jnz %c, @fwd_body, @done
@fwd_body
	%off =l mul %i, 8
	%ps =l add %src, %off
	%v =l loadl %ps
	%pd =l add %dst, %off
	storel %v, %pd
	%i1 =l add %i, 1
	jmp @fwd_loop
@bwd_loop
	%k =l phi @start %n, @bwd_body %k1
	%c2 =w csgtl %k, 0
	jnz %c2, @bwd_body, @done
@bwd_body
	%k1 =l sub %k, 1
	%off2 =l mul %k1, 8
	%ps2 =l add %src, %off2
	%v2 =l loadl %ps2
	%pd2 =l add %dst, %off2
	storel %v2, %pd2
	jmp @bwd_loop
@done
	ret 0
}
