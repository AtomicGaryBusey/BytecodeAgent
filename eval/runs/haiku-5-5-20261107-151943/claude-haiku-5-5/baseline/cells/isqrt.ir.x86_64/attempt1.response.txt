export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%bit =l copy 4611686018427387904
	%res =l copy 0
@shrink
	%c =w cultl %x, %bit
	jnz %c, @shrinkbody, @loop
@shrinkbody
	%bit =l shr %bit, 2
	jmp @shrink
@loop
	%c =w cnel %bit, 0
	jnz %c, @step, @done
@step
	%t =l add %res, %bit
	%c2 =w cultl %x, %t
	jnz %c2, @less, @ge
@ge
	%x =l sub %x, %t
	%r1 =l shr %res, 1
	%res =l add %r1, %bit
	jmp @next
@less
	%res =l shr %res, 1
	jmp @next
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
