export function l $bubble_sort(l %a, l %n) {
@start
	%i =l alloc8 8
	%j =l alloc8 8
	%m =l sub %n, 1
	storel %m, %i
@outer
	%iv =l loadl %i
	%oc =w csgtl %iv, 0
	jnz %oc, @oinit, @done
@oinit
	storel 0, %j
@inner
	%jv =l loadl %j
	%iv2 =l loadl %i
	%ic =w csltl %jv, %iv2
	jnz %ic, @body, @oend
@body
	%off =l mul %jv, 8
	%p =l add %a, %off
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%sw =w csgtl %x, %y
	jnz %sw, @swap, @noswap
@swap
	storel %y, %p
	storel %x, %q
	jmp @noswap
@noswap
	%jv3 =l loadl %j
	%jn =l add %jv3, 1
	storel %jn, %j
	jmp @inner
@oend
	%iv3 =l loadl %i
	%in =l sub %iv3, 1
	storel %in, %i
	jmp @outer
@done
	ret 0
}
