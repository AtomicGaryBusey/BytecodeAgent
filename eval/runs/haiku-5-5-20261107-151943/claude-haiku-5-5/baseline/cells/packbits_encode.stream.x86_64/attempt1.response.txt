packbits_encode:
  push rbx
  push r12
  xor r8d, r8d
  xor r9d, r9d
.Lmain:
  cmp r8, rsi
  jge .Ldone
  movzx eax, byte ptr [rdi + r8]
  mov rbx, 1
.Lrun:
  lea r11, [r8 + rbx]
  cmp r11, rsi
  jge .Lrundone
  cmp rbx, 128
  jge .Lrundone
  cmp byte ptr [rdi + r11], al
  jne .Lrundone
  inc rbx
  jmp .Lrun
.Lrundone:
  cmp rbx, 3
  jl .Llit
  lea r11, [r9 + 2]
  cmp r11, rcx
  jg .Lfail
  mov r12, 257
  sub r12, rbx
  mov byte ptr [rdx + r9], r12b
  mov byte ptr [rdx + r9 + 1], al
  add r9, 2
  add r8, rbx
  jmp .Lmain
.Llit:
  lea r10, [r8 + 1]
.Lscan:
  cmp r10, rsi
  jge .Lscandone
  mov r11, r10
  sub r11, r8
  cmp r11, 128
  jge .Lscandone
  lea r11, [r10 + 2]
  cmp r11, rsi
  jge .Lscanadv
  movzx eax, byte ptr [rdi + r10]
  cmp byte ptr [rdi + r10 + 1], al
  jne .Lscanadv
  cmp byte ptr [rdi + r10 + 2], al
  je .Lscandone
.Lscanadv:
  inc r10
  jmp .Lscan
.Lscandone:
  mov rbx, r10
  sub rbx, r8
  lea r11, [r9 + rbx + 1]
  cmp r11, rcx
  jg .Lfail
  lea r12, [rbx - 1]
  mov byte ptr [rdx + r9], r12b
  lea r12, [rdi + r8]
  lea r10, [rdx + r9 + 1]
  xor r11d, r11d
.Lcopy:
  cmp r11, rbx
  jge .Lcopydone
  movzx eax, byte ptr [r12 + r11]
  mov byte ptr [r10 + r11], al
  inc r11
  jmp .Lcopy
.Lcopydone:
  lea r9, [r9 + rbx + 1]
  add r8, rbx
  jmp .Lmain
.Lfail:
  mov rax, -1
  jmp .Lret
.Ldone:
  mov rax, r9
.Lret:
  pop r12
  pop rbx
  ret
