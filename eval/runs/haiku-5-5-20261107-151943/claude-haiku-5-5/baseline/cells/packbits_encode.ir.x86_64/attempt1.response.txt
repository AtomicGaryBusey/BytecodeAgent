export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%po =l alloc8 8
	%pj =l alloc8 8
	%pk =l alloc8 8
	storel 0, %pi
	storel 0, %po
@outer
	%i =l loadl %pi
	%more =w csltl %i, %n
	jnz %more, @rinit, @done
@rinit
	storel 1, %pk
@rloop
	%k =l loadl %pk
	%kc =w csltl %k, 128
	jnz %kc, @rpos, @rdone
@rpos
	%pos =l add %i, %k
	%pc =w csltl %pos, %n
	jnz %pc, @rcmp, @rdone
@rcmp
	%pp =l add %src, %pos
	%sv =w loadub %pp
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%eq =w ceqw %sv, %b0
	jnz %eq, @rinc, @rdone
@rinc
	%k2 =l add %k, 1
	storel %k2, %pk
	jmp @rloop
@rdone
	%r =l loadl %pk
	%rc =w csgel %r, 3
	jnz %rc, @rep, @lit
@rep
	%ro =l loadl %po
	%need =l add %ro, 2
	%fit =w cslel %need, %cap
	jnz %fit, @rep2, @fail
@rep2
	%hl =l sub 257, %r
	%h =w copy %hl
	%dst =l add %out, %ro
	storeb %h, %dst
	%dst1 =l add %dst, 1
	%p1 =l add %src, %i
	%b =w loadub %p1
	storeb %b, %dst1
	storel %need, %po
	%i2 =l add %i, %r
	storel %i2, %pi
	jmp @outer
@lit
	%li =l loadl %pi
	storel %li, %pj
@lloop
	%j =l loadl %pj
	%jc =w csltl %j, %n
	jnz %jc, @lchk1, @lend
@lchk1
	%ld =l sub %j, %li
	%dc =w csltl %ld, 128
	jnz %dc, @lchk2, @lend
@lchk2
	%j1 =l add %j, 1
	%j2 =l add %j, 2
	%tc =w csltl %j2, %n
	jnz %tc, @lchk3, @lok
@lchk3
	%pa =l add %src, %j
	%va =w loadub %pa
	%pb =l add %src, %j1
	%vb =w loadub %pb
	%pc2 =l add %src, %j2
	%vc =w loadub %pc2
	%e1 =w ceqw %va, %vb
	%e2 =w ceqw %va, %vc
	%e =w and %e1, %e2
	jnz %e, @lend, @lok
@lok
	%jn =l add %j, 1
	storel %jn, %pj
	jmp @lloop
@lend
	%ej =l loadl %pj
	%L =l sub %ej, %li
	%lo =l loadl %po
	%endp =l add %lo, %L
	%need2 =l add %endp, 1
	%fit2 =w cslel %need2, %cap
	jnz %fit2, @lit2, @fail
@lit2
	%hl2 =l sub %L, 1
	%h2 =w copy %hl2
	%base =l add %out, %lo
	storeb %h2, %base
	storel 0, %pk
	jmp @cloop
@cloop
	%ck =l loadl %pk
	%cc =w csltl %ck, %L
	jnz %cc, @cbody, @cdone
@cbody
	%sp2 =l add %src, %li
	%sp3 =l add %sp2, %ck
	%cv =w loadub %sp3
	%bp =l add %base, 1
	%bp2 =l add %bp, %ck
	storeb %cv, %bp2
	%ck2 =l add %ck, 1
	storel %ck2, %pk
	jmp @cloop
@cdone
	storel %need2, %po
	storel %ej, %pi
	jmp @outer
@done
	%fo =l loadl %po
	ret %fo
@fail
	ret -1
}
