export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%resp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%fin =w csgel %pc, %len
	jnz %fin, @done, @fetch
@fetch
	%po =l mul %pc, 8
	%addr =l add %code, %po
	%op =l loadl %addr
	%isz =w ceql %op, 0
	jnz %isz, @done, @c1
@c1
	%k1 =w ceql %op, 1
	jnz %k1, @push, @c2
@c2
	%k2 =w ceql %op, 2
	jnz %k2, @add, @c3
@c3
	%k3 =w ceql %op, 3
	jnz %k3, @sub, @c4
@c4
	%k4 =w ceql %op, 4
	jnz %k4, @mul, @c5
@c5
	%k5 =w ceql %op, 5
	jnz %k5, @dup, @c6
@c6
	%k6 =w ceql %op, 6
	jnz %k6, @swap, @c7
@c7
	%k7 =w ceql %op, 7
	jnz %k7, @drop, @c8
@c8
	%k8 =w ceql %op, 8
	jnz %k8, @over, @c9
@c9
	%k9 =w ceql %op, 9
	jnz %k9, @lt, @c10
@c10
	%k10 =w ceql %op, 10
	jnz %k10, @jz, @c11
@c11
	%k11 =w ceql %op, 11
	jnz %k11, @jmp, @unk
@unk
	%pnu =l add %pc, 1
	storel %pnu, %pcp
	jmp @loop
@push
	%pa_ps =l add %addr, 8
	%pv_ps =l loadl %pa_ps
	%d_ps =l loadl %dp
	%so_ps =l mul %d_ps, 8
	%sa_ps =l add %stack, %so_ps
	storel %pv_ps, %sa_ps
	%d1_ps =l add %d_ps, 1
	storel %d1_ps, %dp
	%pn_ps =l add %pc, 2
	storel %pn_ps, %pcp
	jmp @loop
@add
	%d_ad =l loadl %dp
	%bi_ad =l sub %d_ad, 1
	%bo_ad =l mul %bi_ad, 8
	%ba_ad =l add %stack, %bo_ad
	%b_ad =l loadl %ba_ad
	%ai_ad =l sub %d_ad, 2
	%ao_ad =l mul %ai_ad, 8
	%aa_ad =l add %stack, %ao_ad
	%a_ad =l loadl %aa_ad
	%r_ad =l add %a_ad, %b_ad
	storel %r_ad, %aa_ad
	storel %bi_ad, %dp
	%pn_ad =l add %pc, 1
	storel %pn_ad, %pcp
	jmp @loop
@sub
	%d_sb =l loadl %dp
	%bi_sb =l sub %d_sb, 1
	%bo_sb =l mul %bi_sb, 8
	%ba_sb =l add %stack, %bo_sb
	%b_sb =l loadl %ba_sb
	%ai_sb =l sub %d_sb, 2
	%ao_sb =l mul %ai_sb, 8
	%aa_sb =l add %stack, %ao_sb
	%a_sb =l loadl %aa_sb
	%r_sb =l sub %a_sb, %b_sb
	storel %r_sb, %aa_sb
	storel %bi_sb, %dp
	%pn_sb =l add %pc, 1
	storel %pn_sb, %pcp
	jmp @loop
@mul
	%d_ml =l loadl %dp
	%bi_ml =l sub %d_ml, 1
	%bo_ml =l mul %bi_ml, 8
	%ba_ml =l add %stack, %bo_ml
	%b_ml =l loadl %ba_ml
	%ai_ml =l sub %d_ml, 2
	%ao_ml =l mul %ai_ml, 8
	%aa_ml =l add %stack, %ao_ml
	%a_ml =l loadl %aa_ml
	%r_ml =l mul %a_ml, %b_ml
	storel %r_ml, %aa_ml
	storel %bi_ml, %dp
	%pn_ml =l add %pc, 1
	storel %pn_ml, %pcp
	jmp @loop
@lt
	%d_lt =l loadl %dp
	%bi_lt =l sub %d_lt, 1
	%bo_lt =l mul %bi_lt, 8
	%ba_lt =l add %stack, %bo_lt
	%b_lt =l loadl %ba_lt
	%ai_lt =l sub %d_lt, 2
	%ao_lt =l mul %ai_lt, 8
	%aa_lt =l add %stack, %ao_lt
	%a_lt =l loadl %aa_lt
	%rw_lt =w csltl %a_lt, %b_lt
	%r_lt =l extuw %rw_lt
	storel %r_lt, %aa_lt
	storel %bi_lt, %dp
	%pn_lt =l add %pc, 1
	storel %pn_lt, %pcp
	jmp @loop
@dup
	%d_du =l loadl %dp
	%ti_du =l sub %d_du, 1
	%to_du =l mul %ti_du, 8
	%ta_du =l add %stack, %to_du
	%tv_du =l loadl %ta_du
	%no_du =l mul %d_du, 8
	%na_du =l add %stack, %no_du
	storel %tv_du, %na_du
	%d1_du =l add %d_du, 1
	storel %d1_du, %dp
	%pn_du =l add %pc, 1
	storel %pn_du, %pcp
	jmp @loop
@swap
	%d_sw =l loadl %dp
	%ai_sw =l sub %d_sw, 2
	%ao_sw =l mul %ai_sw, 8
	%aa_sw =l add %stack, %ao_sw
	%a_sw =l loadl %aa_sw
	%bi_sw =l sub %d_sw, 1
	%bo_sw =l mul %bi_sw, 8
	%ba_sw =l add %stack, %bo_sw
	%b_sw =l loadl %ba_sw
	storel %b_sw, %aa_sw
	storel %a_sw, %ba_sw
	%pn_sw =l add %pc, 1
	storel %pn_sw, %pcp
	jmp @loop
@drop
	%d_dr =l loadl %dp
	%d1_dr =l sub %d_dr, 1
	storel %d1_dr, %dp
	%pn_dr =l add %pc, 1
	storel %pn_dr, %pcp
	jmp @loop
@over
	%d_ov =l loadl %dp
	%ai_ov =l sub %d_ov, 2
	%ao_ov =l mul %ai_ov, 8
	%aa_ov =l add %stack, %ao_ov
	%a_ov =l loadl %aa_ov
	%no_ov =l mul %d_ov, 8
	%na_ov =l add %stack, %no_ov
	storel %a_ov, %na_ov
	%d1_ov =l add %d_ov, 1
	storel %d1_ov, %dp
	%pn_ov =l add %pc, 1
	storel %pn_ov, %pcp
	jmp @loop
@jz
	%d_jz =l loadl %dp
	%vi_jz =l sub %d_jz, 1
	%vo_jz =l mul %vi_jz, 8
	%va_jz =l add %stack, %vo_jz
	%v_jz =l loadl %va_jz
	storel %vi_jz, %dp
	%z_jz =w ceql %v_jz, 0
	jnz %z_jz, @jzt, @jzn
@jzt
	%ta_jzt =l add %addr, 8
	%t_jzt =l loadl %ta_jzt
	storel %t_jzt, %pcp
	jmp @loop
@jzn
	%pn_jzn =l add %pc, 2
	storel %pn_jzn, %pcp
	jmp @loop
@jmp
	%ta_jm =l add %addr, 8
	%t_jm =l loadl %ta_jm
	storel %t_jm, %pcp
	jmp @loop
@done
	%d_dn =l loadl %dp
	%dz_dn =w ceql %d_dn, 0
	jnz %dz_dn, @zero, @top
@zero
	storel 0, %resp
	jmp @zinit
@top
	%d_tp =l loadl %dp
	%dm_tp =l sub %d_tp, 1
	%so_tp =l mul %dm_tp, 8
	%sa_tp =l add %stack, %so_tp
	%tv_tp =l loadl %sa_tp
	storel %tv_tp, %resp
	jmp @zinit
@zinit
	%d_zi =l loadl %dp
	storel %d_zi, %ip
	jmp @zl
@zl
	%i_zl =l loadl %ip
	%lc_zl =w csltl %i_zl, %cap
	jnz %lc_zl, @zb, @ret
@zb
	%i_zb =l loadl %ip
	%zo_zb =l mul %i_zb, 8
	%zp_zb =l add %stack, %zo_zb
	storel 0, %zp_zb
	%i2_zb =l add %i_zb, 1
	storel %i2_zb, %ip
	jmp @zl
@ret
	%rv =l loadl %resp
	ret %rv
}
