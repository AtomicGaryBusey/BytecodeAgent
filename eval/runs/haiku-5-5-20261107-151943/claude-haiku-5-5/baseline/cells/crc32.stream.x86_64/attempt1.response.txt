crc32:
    mov eax, 0xFFFFFFFF
    xor r8d, r8d
.Lbyte:
    cmp r8, rsi
    jge .Ldone
    movzx ecx, byte ptr [rdi + r8]
    xor eax, ecx
    mov edx, 8
.Lbit:
    mov ecx, eax
    shr eax, 1
    test ecx, 1
    jz .Lnext
    xor eax, 0xEDB88320
.Lnext:
    dec edx
    jnz .Lbit
    inc r8
    jmp .Lbyte
.Ldone:
    not eax
    ret
