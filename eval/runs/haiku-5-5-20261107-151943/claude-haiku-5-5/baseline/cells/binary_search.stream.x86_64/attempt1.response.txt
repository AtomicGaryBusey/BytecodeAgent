binary_search:
    xor eax, eax
    mov rcx, rsi
.Lloop:
    cmp rax, rcx
    jge .Lnotfound
    mov r8, rax
    add r8, rcx
    shr r8, 1
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .Lfound
    jl .Lright
    mov rcx, r8
    jmp .Lloop
.Lright:
    lea rax, [r8 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r8
    ret
.Lnotfound:
    mov rax, -1
    ret
