export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
	jmp @cond
@cond
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%g =w csgtl %v, %max
	jnz %g, @upd, @next
@upd
	%max =l copy %v
	jmp @next
@next
	%i =l add %i, 1
	jmp @cond
@done
	ret %max
}
