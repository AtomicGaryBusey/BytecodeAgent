export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%mxa =l copy -9223372036854775807
	%mn0 =l copy 9223372036854775807
	%mx0 =l sub %mxa, 1
	jmp @loop
@loop
	%i =l phi @start 0, @afterMax %inext
	%mn =l phi @start %mn0, @afterMax %mn1
	%mx =l phi @start %mx0, @afterMax %mx1
	%s =l phi @start 0, @afterMax %s1
	%c =l phi @start 0, @afterMax %c1
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%bit =l and %y, 1
	%c1 =l add %c, %bit
	%s1 =l add %s, %y
	%lt =w csltl %y, %mn
	jnz %lt, @newmin, @afterMin
@newmin
	jmp @afterMin
@afterMin
	%mn1 =l phi @body %mn, @newmin %y
	%gt =w csgtl %y, %mx
	jnz %gt, @newmax, @afterMax
@newmax
	jmp @afterMax
@afterMax
	%mx1 =l phi @afterMin %mx, @newmax %y
	%inext =l add %i, 1
	jmp @loop
@end
	storel %mn, %out
	%o1 =l add %out, 8
	storel %mx, %o1
	%o2 =l add %out, 16
	storel %s, %o2
	ret %c
}
