export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %in
	%j =l phi @start %j0, @body %jn
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%oi =l mul %i, 8
	%pa =l add %a, %oi
	%oj =l mul %j, 8
	%pb =l add %a, %oj
	%x =l loadl %pa
	%y =l loadl %pb
	storel %y, %pa
	storel %x, %pb
	%in =l add %i, 1
	%jn =l sub %j, 1
	jmp @loop
@done
	ret 0
}
