export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l alloc8 8
	%c =l alloc8 8
	storel 0, %i
	storel 0, %c
	jmp @cond
@cond
	%iv =l loadl %i
	%t =w csltl %iv, %n
	jnz %t, @body, @done
@body
	%off =l mul %iv, 8
	%sp =l add %src, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	%dp =l add %dst, %off
	storel %r, %dp
	%lt =w csltl %r, %x
	%ltl =l extuw %lt
	%cv =l loadl %c
	%cn =l add %cv, %ltl
	storel %cn, %c
	%inc =l add %iv, 1
	storel %inc, %i
	jmp @cond
@done
	%res =l loadl %c
	ret %res
}
