export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	%sz =l mul %nn, 8
	%tp =l add %tmp, %sz
	%kv =l alloc8 8
	storel %k, %kv
	%ev =l alloc8 8
	%tv =l alloc8 8
	%acc =l alloc8 8
	storel 0, %ev
	jmp @id
@id
	%ie =l loadl %ev
	%ic =w csltl %ie, %nn
	jnz %ic, @idb, @ca0
@idb
	%irow =l div %ie, %n
	%icol =l rem %ie, %n
	%ieq =w ceql %irow, %icol
	%ieq8 =l extuw %ieq
	%ioff =l mul %ie, 8
	%iptr =l add %out, %ioff
	storel %ieq8, %iptr
	%ie1 =l add %ie, 1
	storel %ie1, %ev
	jmp @id
@ca0
	storel 0, %ev
	jmp @ca
@ca
	%ce =l loadl %ev
	%cc =w csltl %ce, %nn
	jnz %cc, @cab, @top
@cab
	%coff =l mul %ce, 8
	%capa =l add %a, %coff
	%cv =l loadl %capa
	%cpb =l add %tmp, %coff
	storel %cv, %cpb
	%ce1 =l add %ce, 1
	storel %ce1, %ev
	jmp @ca
@top
	%tk =l loadl %kv
	%tbit =l and %tk, 1
	%tc =w cnel %tbit, 0
	jnz %tc, @p0, @shift
@p0
	storel 0, %ev
	jmp @pe
@pe
	%pe =l loadl %ev
	%pc =w csltl %pe, %nn
	jnz %pc, @pb, @pz0
@pb
	%prow =l div %pe, %n
	%pcol =l rem %pe, %n
	%prn =l mul %prow, %n
	storel 0, %acc
	storel 0, %tv
	jmp @pt
@pt
	%pti =l loadl %tv
	%ptc =w csltl %pti, %n
	jnz %ptc, @ptb, @pst
@ptb
	%pix =l add %prn, %pti
	%pox =l mul %pix, 8
	%pxa =l add %out, %pox
	%pxv =l loadl %pxa
	%pty =l mul %pti, %n
	%piy =l add %pty, %pcol
	%poy =l mul %piy, 8
	%pyb =l add %tmp, %poy
	%pyv =l loadl %pyb
	%ppr =l mul %pxv, %pyv
	%pac =l loadl %acc
	%pac2 =l add %pac, %ppr
	storel %pac2, %acc
	%pt1 =l add %pti, 1
	storel %pt1, %tv
	jmp @pt
@pst
	%pres =l loadl %acc
	%poff =l mul %pe, 8
	%ptp =l add %tp, %poff
	storel %pres, %ptp
	%pe1 =l add %pe, 1
	storel %pe1, %ev
	jmp @pe
@pz0
	storel 0, %ev
	jmp @pz
@pz
	%pzl =l loadl %ev
	%pzc =w csltl %pzl, %nn
	jnz %pzc, @pzb, @shift
@pzb
	%pzo =l mul %pzl, 8
	%pzs =l add %tp, %pzo
	%pzv =l loadl %pzs
	%pzd =l add %out, %pzo
	storel %pzv, %pzd
	%pzn =l add %pzl, 1
	storel %pzn, %ev
	jmp @pz
@shift
	%sk =l loadl %kv
	%snk =l shr %sk, 1
	storel %snk, %kv
	%sc =w cnel %snk, 0
	jnz %sc, @sq0, @fin0
@sq0
	storel 0, %ev
	jmp @qe
@qe
	%qe =l loadl %ev
	%qc =w csltl %qe, %nn
	jnz %qc, @qb, @qz0
@qb
	%qrow =l div %qe, %n
	%qcol =l rem %qe, %n
	%qrn =l mul %qrow, %n
	storel 0, %acc
	storel 0, %tv
	jmp @qt
@qt
	%qti =l loadl %tv
	%qtc =w csltl %qti, %n
	jnz %qtc, @qtb, @qst
@qtb
	%qix =l add %qrn, %qti
	%qox =l mul %qix, 8
	%qxa =l add %tmp, %qox
	%qxv =l loadl %qxa
	%qty =l mul %qti, %n
	%qiy =l add %qty, %qcol
	%qoy =l mul %qiy, 8
	%qyb =l add %tmp, %qoy
	%qyv =l loadl %qyb
	%qpr =l mul %qxv, %qyv
	%qac =l loadl %acc
	%qac2 =l add %qac, %qpr
	storel %qac2, %acc
	%qt1 =l add %qti, 1
	storel %qt1, %tv
	jmp @qt
@qst
	%qres =l loadl %acc
	%qoff =l mul %qe, 8
	%qtp =l add %tp, %qoff
	storel %qres, %qtp
	%qe1 =l add %qe, 1
	storel %qe1, %ev
	jmp @qe
@qz0
	storel 0, %ev
	jmp @qz
@qz
	%qzl =l loadl %ev
	%qzc =w csltl %qzl, %nn
	jnz %qzc, @qzb, @top
@qzb
	%qzo =l mul %qzl, 8
	%qzs =l add %tp, %qzo
	%qzv =l loadl %qzs
	%qzd =l add %tmp, %qzo
	storel %qzv, %qzd
	%qzn =l add %qzl, 1
	storel %qzn, %ev
	jmp @qz
@fin0
	storel 0, %ev
	jmp @fz
@fz
	%fe =l loadl %ev
	%fc =w csltl %fe, %nn2
	jnz %fc, @fzb, @done
@fzb
	%fo =l mul %fe, 8
	%fp =l add %tmp, %fo
	storel 0, %fp
	%fe1 =l add %fe, 1
	storel %fe1, %ev
	jmp @fz
@done
	ret 0
}
