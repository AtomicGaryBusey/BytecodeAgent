qsort_pairs:
    cmp rsi, 2
    jl .Lret0
    push rbx
    push r12
    push r15
    mov rbx, rdi
    mov r12, rsi
    mov rax, r12
    shr rax, 1
    shl rax, 4
    lea rdi, [r12 - 1]
    shl rdi, 4
    mov rdx, qword ptr [rbx + rax]
    mov r8, qword ptr [rbx + rax + 8]
    mov r9, qword ptr [rbx + rdi]
    mov r10, qword ptr [rbx + rdi + 8]
    mov qword ptr [rbx + rax], r9
    mov qword ptr [rbx + rax + 8], r10
    mov qword ptr [rbx + rdi], rdx
    mov qword ptr [rbx + rdi + 8], r8
    xor r15d, r15d
    xor ecx, ecx
.Lloop:
    cmp rcx, rdi
    jge .Lpart_done
    mov r8, qword ptr [rbx + rcx]
    cmp r8, qword ptr [rbx + rdi]
    jl .Lless
    jg .Lnext
    mov r8, qword ptr [rbx + rcx + 8]
    cmp r8, qword ptr [rbx + rdi + 8]
    jge .Lnext
.Lless:
    mov rax, r15
    mov rdx, qword ptr [rbx + rax]
    mov r8, qword ptr [rbx + rax + 8]
    mov r9, qword ptr [rbx + rcx]
    mov r10, qword ptr [rbx + rcx + 8]
    mov qword ptr [rbx + rax], r9
    mov qword ptr [rbx + rax + 8], r10
    mov qword ptr [rbx + rcx], rdx
    mov qword ptr [rbx + rcx + 8], r8
    add r15, 16
.Lnext:
    add rcx, 16
    jmp .Lloop
.Lpart_done:
    mov rax, r15
    mov rdx, qword ptr [rbx + rax]
    mov r8, qword ptr [rbx + rax + 8]
    mov r9, qword ptr [rbx + rdi]
    mov r10, qword ptr [rbx + rdi + 8]
    mov qword ptr [rbx + rax], r9
    mov qword ptr [rbx + rax + 8], r10
    mov qword ptr [rbx + rdi], rdx
    mov qword ptr [rbx + rdi + 8], r8
    mov rdi, rbx
    mov rsi, r15
    shr rsi, 4
    call qsort_pairs
    lea rdi, [rbx + r15 + 16]
    mov rsi, r15
    shr rsi, 4
    mov rax, r12
    sub rax, rsi
    sub rax, 1
    mov rsi, rax
    call qsort_pairs
    xor eax, eax
    pop r15
    pop r12
    pop rbx
    ret
.Lret0:
    xor eax, eax
    ret
