export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%np1 =l add %n, 1
	%j0 =l copy 0
	jmp @zloop
@zloop
	%zj =l phi @start %j0, @zbody %zjn
	%zc =w csltl %zj, %np1
	jnz %zc, @zbody, @outer
@zbody
	%zoff =l mul %zj, 8
	%zp =l add %row, %zoff
	storel 0, %zp
	%zjn =l add %zj, 1
	jmp @zloop
@outer
	%i0 =l copy 0
	jmp @iloop
@iloop
	%i =l phi @outer %i0, @iend %in
	%ic =w csltl %i, %m
	jnz %ic, @ibody, @done
@ibody
	%px =l add %x, %i
	%xi =w loadub %px
	%d0 =l copy 0
	%j1 =l copy 1
	jmp @jloop
@jloop
	%j =l phi @ibody %j1, @jnext %jn
	%d =l phi @ibody %d0, @jnext %up
	%jc =w cslel %j, %n
	jnz %jc, @jbody, @iend
@jbody
	%joff =l mul %j, 8
	%pj =l add %row, %joff
	%up =l loadl %pj
	%jm1 =l sub %j, 1
	%py =l add %y, %jm1
	%yj =w loadub %py
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%dp =l add %d, 1
	storel %dp, %pj
	jmp @jnext
@nomatch
	%loff =l mul %jm1, 8
	%pl =l add %row, %loff
	%left =l loadl %pl
	%gt =w csgtl %left, %up
	jnz %gt, @takeleft, @takeup
@takeleft
	storel %left, %pj
	jmp @jnext
@takeup
	storel %up, %pj
	jmp @jnext
@jnext
	%jn =l add %j, 1
	jmp @jloop
@iend
	%in =l add %i, 1
	jmp @iloop
@done
	%roff =l mul %n, 8
	%rp =l add %row, %roff
	%res =l loadl %rp
	ret %res
}
