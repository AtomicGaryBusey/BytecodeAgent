export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l alloc8 8
	%i =l alloc8 8
	storel 0, %sum
	storel 0, %i
@loop
	%iv =l loadl %i
	%c =w csltl %iv, %n
	jnz %c, @body, @done
@body
	%ib =l loadl %i
	%off =l mul %ib, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%s =l loadl %sum
	%s2 =l add %s, %r
	storel %s2, %sum
	%ib2 =l loadl %i
	%i2 =l add %ib2, 1
	storel %i2, %i
	jmp @loop
@done
	%res =l loadl %sum
	ret %res
}
