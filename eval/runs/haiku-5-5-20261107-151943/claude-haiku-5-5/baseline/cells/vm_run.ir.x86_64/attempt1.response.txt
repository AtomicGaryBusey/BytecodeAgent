export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l alloc8 8
	%sp =l alloc8 8
	%res =l alloc8 8
	%i =l alloc8 8
	storel 0, %pc
	storel 0, %sp
	storel 0, %res
	jmp @loop
@loop
	%p0 =l loadl %pc
	%more =w csltl %p0, %len
	jnz %more, @fetch, @done
@fetch
	%p1 =l loadl %pc
	%ao0 =l mul %p1, 8
	%aa0 =l add %code, %ao0
	%op =l loadl %aa0
	%e0 =w ceql %op, 0
	jnz %e0, @halt, @d1
@d1
	%e1 =w ceql %op, 1
	jnz %e1, @push, @d2
@d2
	%e2 =w ceql %op, 2
	jnz %e2, @add, @d3
@d3
	%e3 =w ceql %op, 3
	jnz %e3, @sub, @d4
@d4
	%e4 =w ceql %op, 4
	jnz %e4, @mul, @d5
@d5
	%e5 =w ceql %op, 5
	jnz %e5, @dup, @d6
@d6
	%e6 =w ceql %op, 6
	jnz %e6, @swap, @d7
@d7
	%e7 =w ceql %op, 7
	jnz %e7, @drop, @d8
@d8
	%e8 =w ceql %op, 8
	jnz %e8, @over, @d9
@d9
	%e9 =w ceql %op, 9
	jnz %e9, @lt, @d10
@d10
	%e10 =w ceql %op, 10
	jnz %e10, @jz, @jmp
@halt
	jmp @done
@push
	%s0 =l loadl %sp
	%p2 =l loadl %pc
	%io0 =l add %p2, 1
	%ioff0 =l mul %io0, 8
	%ia0 =l add %code, %ioff0
	%v =l loadl %ia0
	%so0 =l mul %s0, 8
	%sa0 =l add %stack, %so0
	storel %v, %sa0
	%ns0 =l add %s0, 1
	storel %ns0, %sp
	%np0 =l add %p2, 2
	storel %np0, %pc
	jmp @loop
@add
	%s1 =l loadl %sp
	%bi1 =l sub %s1, 1
	%ai1 =l sub %s1, 2
	%bo1 =l mul %bi1, 8
	%ao1 =l mul %ai1, 8
	%ba1 =l add %stack, %bo1
	%aa1 =l add %stack, %ao1
	%b1 =l loadl %ba1
	%a1 =l loadl %aa1
	%r1 =l add %a1, %b1
	storel %r1, %aa1
	storel %bi1, %sp
	%p3 =l loadl %pc
	%np1 =l add %p3, 1
	storel %np1, %pc
	jmp @loop
@sub
	%s2 =l loadl %sp
	%bi2 =l sub %s2, 1
	%ai2 =l sub %s2, 2
	%bo2 =l mul %bi2, 8
	%ao2 =l mul %ai2, 8
	%ba2 =l add %stack, %bo2
	%aa2 =l add %stack, %ao2
	%b2 =l loadl %ba2
	%a2 =l loadl %aa2
	%r2 =l sub %a2, %b2
	storel %r2, %aa2
	storel %bi2, %sp
	%p4 =l loadl %pc
	%np2 =l add %p4, 1
	storel %np2, %pc
	jmp @loop
@mul
	%s3 =l loadl %sp
	%bi3 =l sub %s3, 1
	%ai3 =l sub %s3, 2
	%bo3 =l mul %bi3, 8
	%ao3 =l mul %ai3, 8
	%ba3 =l add %stack, %bo3
	%aa3 =l add %stack, %ao3
	%b3 =l loadl %ba3
	%a3 =l loadl %aa3
	%r3 =l mul %a3, %b3
	storel %r3, %aa3
	storel %bi3, %sp
	%p5 =l loadl %pc
	%np3 =l add %p5, 1
	storel %np3, %pc
	jmp @loop
@lt
	%s4 =l loadl %sp
	%bi4 =l sub %s4, 1
	%ai4 =l sub %s4, 2
	%bo4 =l mul %bi4, 8
	%ao4 =l mul %ai4, 8
	%ba4 =l add %stack, %bo4
	%aa4 =l add %stack, %ao4
	%b4 =l loadl %ba4
	%a4 =l loadl %aa4
	%c4 =w csltl %a4, %b4
	%r4 =l extuw %c4
	storel %r4, %aa4
	storel %bi4, %sp
	%p6 =l loadl %pc
	%np4 =l add %p6, 1
	storel %np4, %pc
	jmp @loop
@dup
	%s5 =l loadl %sp
	%ti5 =l sub %s5, 1
	%to5 =l mul %ti5, 8
	%ta5 =l add %stack, %to5
	%v5 =l loadl %ta5
	%so5 =l mul %s5, 8
	%sa5 =l add %stack, %so5
	storel %v5, %sa5
	%ns5 =l add %s5, 1
	storel %ns5, %sp
	%p7 =l loadl %pc
	%np5 =l add %p7, 1
	storel %np5, %pc
	jmp @loop
@over
	%s6 =l loadl %sp
	%ti6 =l sub %s6, 2
	%to6 =l mul %ti6, 8
	%ta6 =l add %stack, %to6
	%v6 =l loadl %ta6
	%so6 =l mul %s6, 8
	%sa6 =l add %stack, %so6
	storel %v6, %sa6
	%ns6 =l add %s6, 1
	storel %ns6, %sp
	%p8 =l loadl %pc
	%np6 =l add %p8, 1
	storel %np6, %pc
	jmp @loop
@swap
	%s7 =l loadl %sp
	%bi7 =l sub %s7, 1
	%ai7 =l sub %s7, 2
	%bo7 =l mul %bi7, 8
	%ao7 =l mul %ai7, 8
	%ba7 =l add %stack, %bo7
	%aa7 =l add %stack, %ao7
	%b7 =l loadl %ba7
	%a7 =l loadl %aa7
	storel %b7, %aa7
	storel %a7, %ba7
	%p9 =l loadl %pc
	%np7 =l add %p9, 1
	storel %np7, %pc
	jmp @loop
@drop
	%s8 =l loadl %sp
	%ns8 =l sub %s8, 1
	storel %ns8, %sp
	%p10 =l loadl %pc
	%np8 =l add %p10, 1
	storel %np8, %pc
	jmp @loop
@jz
	%p11 =l loadl %pc
	%s9 =l loadl %sp
	%ns9 =l sub %s9, 1
	storel %ns9, %sp
	%vo9 =l mul %ns9, 8
	%va9 =l add %stack, %vo9
	%v9 =l loadl %va9
	%io9 =l add %p11, 1
	%ioff9 =l mul %io9, 8
	%ia9 =l add %code, %ioff9
	%t9 =l loadl %ia9
	%z9 =w ceql %v9, 0
	jnz %z9, @jzt, @jzn
@jzt
	storel %t9, %pc
	jmp @loop
@jzn
	%p12 =l loadl %pc
	%np12 =l add %p12, 2
	storel %np12, %pc
	jmp @loop
@jmp
	%p13 =l loadl %pc
	%io13 =l add %p13, 1
	%ioff13 =l mul %io13, 8
	%ia13 =l add %code, %ioff13
	%t13 =l loadl %ia13
	storel %t13, %pc
	jmp @loop
@done
	%sd =l loadl %sp
	storel %sd, %i
	%emp =w ceql %sd, 0
	jnz %emp, @zero, @top
@top
	%ti14 =l sub %sd, 1
	%to14 =l mul %ti14, 8
	%ta14 =l add %stack, %to14
	%rv =l loadl %ta14
	storel %rv, %res
	jmp @zero
@zero
	%iv =l loadl %i
	%lt =w csltl %iv, %cap
	jnz %lt, @zbody, @ret
@zbody
	%iv2 =l loadl %i
	%zo =l mul %iv2, 8
	%za =l add %stack, %zo
	storel 0, %za
	%in =l add %iv2, 1
	storel %in, %i
	jmp @zero
@ret
	%out =l loadl %res
	ret %out
}
