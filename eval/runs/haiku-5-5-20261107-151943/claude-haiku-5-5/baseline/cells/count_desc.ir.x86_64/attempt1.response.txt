export function l $count_desc(l %a, l %n) {
@start
	%m =l sub %n, 1
	jmp @cond
@cond
	%i =l phi @start 0, @body %inext
	%cnt =l phi @start 0, @body %cnext
	%c =w csltl %i, %m
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%g =w csgtl %x, %y
	%gl =l extuw %g
	%cnext =l add %cnt, %gl
	%inext =l add %i, 1
	jmp @cond
@done
	ret %cnt
}
