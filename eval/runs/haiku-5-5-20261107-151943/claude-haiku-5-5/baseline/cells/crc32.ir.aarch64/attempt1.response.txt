export function l $crc32(l %p, l %n) {
@start
	%crc =w copy -1
	%i =l copy 0
	jmp @byteloop
@byteloop
	%more =w csltl %i, %n
	jnz %more, @bytebody, @done
@bytebody
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%k =w copy 8
	jmp @bitloop
@bitloop
	%m =w and %crc, 1
	%m =w sub 0, %m
	%pm =w and %m, -306674912
	%s =w shr %crc, 1
	%crc =w xor %s, %pm
	%k =w sub %k, 1
	%kc =w csgtw %k, 0
	jnz %kc, @bitloop, @bitdone
@bitdone
	%i =l add %i, 1
	jmp @byteloop
@done
	%crc =w xor %crc, -1
	%r =l extuw %crc
	ret %r
}
