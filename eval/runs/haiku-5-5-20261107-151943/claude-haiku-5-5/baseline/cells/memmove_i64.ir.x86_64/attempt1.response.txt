export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%fwd =w culel %dst, %src
	jnz %fwd, @fhead, @bhead
@fhead
	%i =l phi @start 0, @fbody %i1
	%fc =w csltl %i, %n
	jnz %fc, @fbody, @done
@fbody
	%foff =l mul %i, 8
	%fsp =l add %src, %foff
	%fdp =l add %dst, %foff
	%fv =l loadl %fsp
	storel %fv, %fdp
	%i1 =l add %i, 1
	jmp @fhead
@bhead
	%j =l phi @start %n, @bbody %j1
	%bc =w cnel %j, 0
	jnz %bc, @bbody, @done
@bbody
	%j1 =l sub %j, 1
	%boff =l mul %j1, 8
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bv =l loadl %bsp
	storel %bv, %bdp
	jmp @bhead
@done
	ret 0
}
