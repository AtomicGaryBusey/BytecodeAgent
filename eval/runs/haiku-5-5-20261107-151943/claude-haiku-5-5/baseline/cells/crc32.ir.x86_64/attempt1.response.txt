export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
	jmp @outer
@outer
	%crc =w phi @start %crc0, @bitdone %crcn
	%i =l phi @start %i0, @bitdone %inext
	%done =w csgel %i, %n
	jnz %done, @fin, @body
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%x =w xor %crc, %b
	jmp @bit
@bit
	%xv =w phi @body %x, @bit %xn
	%kv =w phi @body 0, @bit %kn
	%lsb =w and %xv, 1
	%sh =w shr %xv, 1
	%m =w neg %lsb
	%poly =w and %m, 3988292384
	%xn =w xor %sh, %poly
	%kn =w add %kv, 1
	%more =w csltw %kn, 8
	jnz %more, @bit, @bitdone
@bitdone
	%crcn =w copy %xn
	%inext =l add %i, 1
	jmp @outer
@fin
	%r =w xor %crc, 4294967295
	%rl =l extuw %r
	ret %rl
}
