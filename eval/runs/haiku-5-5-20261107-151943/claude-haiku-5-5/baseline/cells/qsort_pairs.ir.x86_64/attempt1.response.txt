export function l $qsort_pairs(l %p, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret0, @check
@ret0
	ret 0
@check
	%nm1 =l sub %n, 1
	%pa =l mul %nm1, 16
	%pivaddr =l add %p, %pa
	%pk =l loadl %pivaddr
	%pvaddr8 =l add %pivaddr, 8
	%pv =l loadl %pvaddr8
	%i0 =l copy 0
	%j0 =l copy 0
	jmp @loop
@loop
	%i =l phi @check %i0, @latch %i2
	%j =l phi @check %j0, @latch %j2
	%done =w csgel %j, %nm1
	jnz %done, @after, @body
@body
	%joff =l mul %j, 16
	%jp =l add %p, %joff
	%ioff =l mul %i, 16
	%ip =l add %p, %ioff
	%jk =l loadl %jp
	%jvp =l add %jp, 8
	%jv =l loadl %jvp
	%c1 =w csltl %jk, %pk
	%c2 =w ceql %jk, %pk
	%c3 =w csltl %jv, %pv
	%c4 =w and %c2, %c3
	%c =w or %c1, %c4
	jnz %c, @swap, @latch
@swap
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%i1 =l add %i, 1
	jmp @latch
@latch
	%i2 =l phi @body %i, @swap %i1
	%j2 =l add %j, 1
	jmp @loop
@after
	%ioff2 =l mul %i, 16
	%ip2 =l add %p, %ioff2
	%ik2 =l loadl %ip2
	%ivp2 =l add %ip2, 8
	%iv2 =l loadl %ivp2
	storel %pk, %ip2
	storel %pv, %ivp2
	storel %ik2, %pivaddr
	storel %iv2, %pvaddr8
	%rs1 =l add %i, 1
	%rs2 =l mul %rs1, 16
	%rp =l add %p, %rs2
	%rsz =l sub %nm1, %i
	call $qsort_pairs(l %p, l %i)
	call $qsort_pairs(l %rp, l %rsz)
	ret 0
}
