export function l $popcount(l %x) {
@start
	%v1 =l copy %x
	%shr1 =l shr %v1, 1
	%and1 =l and %shr1, 0x5555555555555555
	%sub1 =l sub %v1, %and1
	%shr2 =l shr %sub1, 2
	%and2 =l and %sub1, 0x3333333333333333
	%and3 =l and %shr2, 0x3333333333333333
	%add1 =l add %and2, %and3
	%shr3 =l shr %add1, 4
	%add2 =l add %add1, %shr3
	%and4 =l and %add2, 0x0f0f0f0f0f0f0f0f
	%mul1 =l mul %and4, 0x0101010101010101
	%shr4 =l shr %mul1, 56
	ret %shr4
}
