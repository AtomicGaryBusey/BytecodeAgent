export function l $popcount(l %x) {
@start
    %t1 =l shr %x, 1
    %m1 =l and %t1, 0x5555555555555555
    %x2 =l sub %x, %m1
    %t2 =l shr %x2, 2
    %m2 =l and %t2, 0x3333333333333333
    %a =l and %x2, 0x3333333333333333
    %x3 =l add %a, %m2
    %t3 =l shr %x3, 4
    %s =l add %x3, %t3
    %x4 =l and %s, 0x0F0F0F0F0F0F0F0F
    %m =l mul %x4, 0x0101010101010101
    %r =l shr %m, 56
    ret %r
}
