export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero_test

@zero_test
	%zi =l phi @start 0, @zero_next %zi1
	%zc =w csltl %zi, %n
	jnz %zc, @zero_body, @outer_test

@zero_body
	%zo =l mul %zi, 8
	%zp =l add %r, %zo
	storel 0, %zp
	%zi1 =l add %zi, 1
	jmp @zero_test

@outer_test
	%oi =l phi @zero_test 0, @outer_next %oi1
	%oc =w csltl %oi, %na
	jnz %oc, @outer_body, @trim_test

@outer_body
	%ao =l mul %oi, 8
	%ap =l add %a, %ao
	%av =l loadl %ap
	%a0 =l and %av, 4294967295
	%a1 =l shrl %av, 32
	jmp @inner_test

@inner_test
	%ij =l phi @outer_body 0, @inner_next %ij1
	%carry =l phi @outer_body 0, @inner_next %carry1
	%ic =w csltl %ij, %nb
	jnz %ic, @inner_body, @inner_done

@inner_body
	%bo =l mul %ij, 8
	%bp =l add %b, %bo
	%bv =l loadl %bp
	%b0 =l and %bv, 4294967295
	%b1 =l shrl %bv, 32

	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1

	%p0h =l shrl %p0, 32
	%p1l =l and %p1, 4294967295
	%p2l =l and %p2, 4294967295
	%m0 =l add %p0h, %p1l
	%m =l add %m0, %p2l

	%p0l =l and %p0, 4294967295
	%ms =l shl %m, 32
	%plo =l or %p0l, %ms

	%p1h =l shrl %p1, 32
	%p2h =l shrl %p2, 32
	%mh =l shrl %m, 32
	%h0 =l add %p3, %p1h
	%h1 =l add %h0, %p2h
	%phi =l add %h1, %mh

	%ri =l add %oi, %ij
	%ro =l mul %ri, 8
	%rp =l add %r, %ro
	%old =l loadl %rp

	%s0 =l add %old, %plo
	%c0w =w cultl %s0, %old
	%c0 =l extuw %c0w
	%s1 =l add %s0, %carry
	%c1w =w cultl %s1, %s0
	%c1 =l extuw %c1w
	storel %s1, %rp

	%ch0 =l add %phi, %c0
	%carry1 =l add %ch0, %c1
	%ij1 =l add %ij, 1
	jmp @inner_next

@inner_next
	jmp @inner_test

@inner_done
	%ci =l add %oi, %nb
	%co =l mul %ci, 8
	%cp =l add %r, %co
	storel %carry, %cp
	%oi1 =l add %oi, 1
	jmp @outer_next

@outer_next
	jmp @outer_test

@trim_test
	%tk =l phi @outer_test %n, @trim_zero %tk1
	%tez =w ceql %tk, 0
	jnz %tez, @return_zero, @trim_load

@trim_load
	%tk1 =l sub %tk, 1
	%to =l mul %tk1, 8
	%tp =l add %r, %to
	%tv =l loadl %tp
	%tvz =w ceql %tv, 0
	jnz %tvz, @trim_zero, @return_k

@trim_zero
	jmp @trim_test

@return_k
	ret %tk

@return_zero
	ret 0
}
