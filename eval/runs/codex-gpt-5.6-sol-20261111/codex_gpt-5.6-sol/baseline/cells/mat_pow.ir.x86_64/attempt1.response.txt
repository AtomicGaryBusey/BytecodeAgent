export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
	%nn =l mul %n, %n
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pt =l alloc8 8
	%ps =l alloc8 8
	%pr =l alloc8 8
	storel 0, %pi
	jmp @zero_out_test

@zero_out_test
	%zi =l loadl %pi
	%zc =w csltl %zi, %nn
	jnz %zc, @zero_out_body, @diag_init

@zero_out_body
	%zo =l mul %zi, 8
	%zp =l add %out, %zo
	storel 0, %zp
	%zin =l add %zi, 1
	storel %zin, %pi
	jmp @zero_out_test

@diag_init
	storel 0, %pi
	jmp @diag_test

@diag_test
	%di =l loadl %pi
	%dc =w csltl %di, %n
	jnz %dc, @diag_body, @power_init

@diag_body
	%din =l mul %di, %n
	%dix =l add %din, %di
	%dio =l mul %dix, 8
	%dip =l add %out, %dio
	storel 1, %dip
	%dinext =l add %di, 1
	storel %dinext, %pi
	jmp @diag_test

@power_init
	storel 0, %pr
	jmp @power_test

@power_test
	%ri =l loadl %pr
	%rc =w csltl %ri, %k
	jnz %rc, @mul_i_init, @clear_init

@mul_i_init
	storel 0, %pi
	jmp @mul_i_test

@mul_i_test
	%mi =l loadl %pi
	%mic =w csltl %mi, %n
	jnz %mic, @mul_j_init, @copy_init

@mul_j_init
	storel 0, %pj
	jmp @mul_j_test

@mul_j_test
	%mj =l loadl %pj
	%mjc =w csltl %mj, %n
	jnz %mjc, @mul_t_init, @mul_i_next

@mul_t_init
	storel 0, %pt
	storel 0, %ps
	jmp @mul_t_test

@mul_t_test
	%mt =l loadl %pt
	%mtc =w csltl %mt, %n
	jnz %mtc, @mul_t_body, @mul_store

@mul_t_body
	%rowbase =l mul %mi, %n
	%leftidx =l add %rowbase, %mt
	%leftoff =l mul %leftidx, 8
	%leftptr =l add %out, %leftoff
	%left =l loadl %leftptr
	%rightbase =l mul %mt, %n
	%rightidx =l add %rightbase, %mj
	%rightoff =l mul %rightidx, 8
	%rightptr =l add %a, %rightoff
	%right =l loadl %rightptr
	%prod =l mul %left, %right
	%sum =l loadl %ps
	%sumnew =l add %sum, %prod
	storel %sumnew, %ps
	%mtnext =l add %mt, 1
	storel %mtnext, %pt
	jmp @mul_t_test

@mul_store
	%mrowbase =l mul %mi, %n
	%midx =l add %mrowbase, %mj
	%moff =l mul %midx, 8
	%mptr =l add %tmp, %moff
	%msum =l loadl %ps
	storel %msum, %mptr
	%mjnext =l add %mj, 1
	storel %mjnext, %pj
	jmp @mul_j_test

@mul_i_next
	%minext =l add %mi, 1
	storel %minext, %pi
	jmp @mul_i_test

@copy_init
	storel 0, %pi
	jmp @copy_test

@copy_test
	%ci =l loadl %pi
	%cic =w csltl %ci, %nn
	jnz %cic, @copy_body, @power_next

@copy_body
	%coff =l mul %ci, 8
	%ctptr =l add %tmp, %coff
	%coptr =l add %out, %coff
	%cv =l loadl %ctptr
	storel %cv, %coptr
	%cinext =l add %ci, 1
	storel %cinext, %pi
	jmp @copy_test

@power_next
	%rinext =l add %ri, 1
	storel %rinext, %pr
	jmp @power_test

@clear_init
	%twonn =l mul %nn, 2
	storel 0, %pi
	jmp @clear_test

@clear_test
	%xi =l loadl %pi
	%xic =w csltl %xi, %twonn
	jnz %xic, @clear_body, @done

@clear_body
	%xoff =l mul %xi, 8
	%xptr =l add %tmp, %xoff
	storel 0, %xptr
	%xnext =l add %xi, 1
	storel %xnext, %pi
	jmp @clear_test

@done
	ret 0
}
