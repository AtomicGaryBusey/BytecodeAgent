binary_search:
    xor ecx, ecx
    mov r8, rsi
.Lloop:
    cmp rcx, r8
    jge .Lnot_found
    mov rax, rcx
    add rax, r8
    shr rax, 1
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    jl .Lmove_low
    jg .Lmove_high
    ret
.Lmove_low:
    lea rcx, [rax + 1]
    jmp .Lloop
.Lmove_high:
    mov r8, rax
    jmp .Lloop
.Lnot_found:
    mov rax, -1
    ret
