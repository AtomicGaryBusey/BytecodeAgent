export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jnz %n, @zero_loop, @zero_done

@zero_loop
	%zi =l phi @start 0, @zero_next %zi1
	%zoff =l shl %zi, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zi1 =l add %zi, 1
	%zmore =w csltl %zi1, %n
	jnz %zmore, @zero_next, @zero_done

@zero_next
	jmp @zero_loop

@zero_done
	%havea =w csltl 0, %na
	jnz %havea, @outer, @outer_done

@outer
	%i =l phi @zero_done 0, @outer_next %i1
	%aoff =l shl %i, 3
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	%haveb =w csltl 0, %nb
	jnz %haveb, @inner, @inner_done

@inner
	%j =l phi @outer 0, @inner_next %j1
	%carry =l phi @outer 0, @inner_next %carry1
	%boff =l shl %j, 3
	%bptr =l add %b, %boff
	%bv =l loadl %bptr

	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32
	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1

	%p1lo =l shl %p1, 32
	%lo1 =l add %p0, %p1lo
	%pc1w =w cultl %lo1, %p0
	%p2lo =l shl %p2, 32
	%plo =l add %lo1, %p2lo
	%pc2w =w cultl %plo, %lo1
	%pc1 =l extuw %pc1w
	%pc2 =l extuw %pc2w
	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%ph1 =l add %p3, %p1hi
	%ph2 =l add %ph1, %p2hi
	%ph3 =l add %ph2, %pc1
	%phi =l add %ph3, %pc2

	%k =l add %i, %j
	%koff =l shl %k, 3
	%rptr =l add %r, %koff
	%old =l loadl %rptr
	%s1 =l add %plo, %old
	%sc1w =w cultl %s1, %plo
	%s2 =l add %s1, %carry
	%sc2w =w cultl %s2, %s1
	storel %s2, %rptr
	%sc1 =l extuw %sc1w
	%sc2 =l extuw %sc2w
	%ch1 =l add %phi, %sc1
	%carry1 =l add %ch1, %sc2

	%j1 =l add %j, 1
	%jmore =w csltl %j1, %nb
	jnz %jmore, @inner_next, @inner_done

@inner_next
	jmp @inner

@inner_done
	%endk =l add %i, %nb
	%endoff =l shl %endk, 3
	%endptr =l add %r, %endoff
	%finalcarry =l phi @outer 0, @inner %carry1
	storel %finalcarry, %endptr
	%i1 =l add %i, 1
	%imore =w csltl %i1, %na
	jnz %imore, @outer_next, @outer_done

@outer_next
	jmp @outer

@outer_done
	jnz %n, @scan, @return_zero

@scan
	%count =l phi @outer_done %n, @scan_next %count1
	%last =l sub %count, 1
	%lastoff =l shl %last, 3
	%lastptr =l add %r, %lastoff
	%limb =l loadl %lastptr
	%iszero =w ceql %limb, 0
	jnz %iszero, @scan_dec, @return_count

@scan_dec
	%count1 =l sub %count, 1
	%more =w csltl 0, %count1
	jnz %more, @scan_next, @return_zero

@scan_next
	jmp @scan

@return_count
	ret %count

@return_zero
	ret 0
}
