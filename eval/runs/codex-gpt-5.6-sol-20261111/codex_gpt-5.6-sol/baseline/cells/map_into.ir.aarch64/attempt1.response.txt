export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done0, @loop

@loop
	%i =l phi @start 0, @next %inext
	%count =l phi @start 0, @next %countnext
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%less =w csltl %r, %x
	%inc =l extuw %less
	%countnext =l add %count, %inc
	%inext =l add %i, 1
	%more =w csltl %inext, %n
	jnz %more, @next, @done

@next
	jmp @loop

@done
	ret %countnext

@done0
	ret 0
}
