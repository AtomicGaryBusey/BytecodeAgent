export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
	jmp @test
@test
	%more =w csltl %i, %n
	jnz %more, @loop, @done
@loop
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%greater =w csltl %max, %val
	jnz %greater, @update, @next
@update
	%newmax =l copy %val
	jmp @next
@next
	%cur =l phi @loop %max, @update %newmax
	%next =l add %i, 1
	%i =l copy %next
	%max =l copy %cur
	jmp @test
@done
	ret %max
}
