export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%i0 =l copy 1
	jmp @test
@test
	%i =l phi @start %i0, @keep %inext, @update %inext
	%max =l phi @start %max0, @keep %max, @update %val
	%more =w csltl %i, %n
	jnz %more, @loop, @done
@loop
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%inext =l add %i, 1
	%greater =w csltl %max, %val
	jnz %greater, @update, @keep
@update
	jmp @test
@keep
	jmp @test
@done
	ret %max
}
