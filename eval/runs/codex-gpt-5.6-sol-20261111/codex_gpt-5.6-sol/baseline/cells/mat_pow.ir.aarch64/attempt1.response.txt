export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%n2 =l mul %n, %n
	%bytes =l mul %n2, 8
	%prod =l add %tmp, %bytes
	jmp @init_i

@init_i
	%ii =l phi @start 0, @init_i_next %ii_next
	%ii_ok =w csltl %ii, %n
	jnz %ii_ok, @init_i_body, @copy_base

@init_i_body
	jmp @init_j

@init_j
	%ij =l phi @init_i_body 0, @init_j_body %ij_next
	%ij_ok =w csltl %ij, %n
	jnz %ij_ok, @init_j_body, @init_i_next

@init_j_body
	%iidx0 =l mul %ii, %n
	%iidx =l add %iidx0, %ij
	%ioff =l mul %iidx, 8
	%iptr =l add %out, %ioff
	%ieq =w ceql %ii, %ij
	%ival =l extuw %ieq
	storel %ival, %iptr
	%ij_next =l add %ij, 1
	jmp @init_j

@init_i_next
	%ii_next =l add %ii, 1
	jmp @init_i

@copy_base
	jmp @copy_loop

@copy_loop
	%ci =l phi @copy_base 0, @copy_body %ci_next
	%ci_ok =w csltl %ci, %n2
	jnz %ci_ok, @copy_body, @pow_loop

@copy_body
	%coff =l mul %ci, 8
	%cas =l add %a, %coff
	%cts =l add %tmp, %coff
	%cval =l loadl %cas
	storel %cval, %cts
	%ci_next =l add %ci, 1
	jmp @copy_loop

@pow_loop
	%pe =l phi @copy_loop 0, @pow_next %pe_next
	%pe_ok =w csltl %pe, %k
	jnz %pe_ok, @mul_i_entry, @zero_entry

@mul_i_entry
	jmp @mul_i

@mul_i
	%mi =l phi @mul_i_entry 0, @mul_i_next %mi_next
	%mi_ok =w csltl %mi, %n
	jnz %mi_ok, @mul_j_entry, @copy_product_entry

@mul_j_entry
	jmp @mul_j

@mul_j
	%mj =l phi @mul_j_entry 0, @mul_j_next %mj_next
	%mj_ok =w csltl %mj, %n
	jnz %mj_ok, @mul_t_entry, @mul_i_next

@mul_t_entry
	jmp @mul_t

@mul_t
	%mt =l phi @mul_t_entry 0, @mul_t_body %mt_next
	%sum =l phi @mul_t_entry 0, @mul_t_body %sum_next
	%mt_ok =w csltl %mt, %n
	jnz %mt_ok, @mul_t_body, @mul_store

@mul_t_body
	%lidx0 =l mul %mi, %n
	%lidx =l add %lidx0, %mt
	%loff =l mul %lidx, 8
	%lptr =l add %out, %loff
	%lval =l loadl %lptr
	%ridx0 =l mul %mt, %n
	%ridx =l add %ridx0, %mj
	%roff =l mul %ridx, 8
	%rptr =l add %tmp, %roff
	%rval =l loadl %rptr
	%term =l mul %lval, %rval
	%sum_next =l add %sum, %term
	%mt_next =l add %mt, 1
	jmp @mul_t

@mul_store
	%sidx0 =l mul %mi, %n
	%sidx =l add %sidx0, %mj
	%soff =l mul %sidx, 8
	%sptr =l add %prod, %soff
	storel %sum, %sptr
	jmp @mul_j_next

@mul_j_next
	%mj_next =l add %mj, 1
	jmp @mul_j

@mul_i_next
	%mi_next =l add %mi, 1
	jmp @mul_i

@copy_product_entry
	jmp @copy_product

@copy_product
	%pi =l phi @copy_product_entry 0, @copy_product_body %pi_next
	%pi_ok =w csltl %pi, %n2
	jnz %pi_ok, @copy_product_body, @pow_next

@copy_product_body
	%poff =l mul %pi, 8
	%pps =l add %prod, %poff
	%pos =l add %out, %poff
	%pval =l loadl %pps
	storel %pval, %pos
	%pi_next =l add %pi, 1
	jmp @copy_product

@pow_next
	%pe_next =l add %pe, 1
	jmp @pow_loop

@zero_entry
	%zcount =l mul %n2, 2
	jmp @zero_loop

@zero_loop
	%zi =l phi @zero_entry 0, @zero_body %zi_next
	%zi_ok =w csltl %zi, %zcount
	jnz %zi_ok, @zero_body, @done

@zero_body
	%zoff =l mul %zi, 8
	%zptr =l add %tmp, %zoff
	storel 0, %zptr
	%zi_next =l add %zi, 1
	jmp @zero_loop

@done
	ret 0
}
