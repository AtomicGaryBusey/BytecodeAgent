export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%minp =l alloc8 8
	%maxp =l alloc8 8
	%sump =l alloc8 8
	%cntp =l alloc8 8
	%ip =l alloc8 8
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	storel %y0, %minp
	storel %y0, %maxp
	storel %y0, %sump
	storel %odd0, %cntp
	storel 1, %ip
	jmp @test
@test
	%i =l loadl %ip
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%min =l loadl %minp
	%ismin =w csltl %y, %min
	jnz %ismin, @setmin, @checkmax
@setmin
	storel %y, %minp
	jmp @checkmax
@checkmax
	%max =l loadl %maxp
	%ismax =w csltl %max, %y
	jnz %ismax, @setmax, @accumulate
@setmax
	storel %y, %maxp
	jmp @accumulate
@accumulate
	%sum =l loadl %sump
	%newsum =l add %sum, %y
	storel %newsum, %sump
	%cnt =l loadl %cntp
	%odd =l and %y, 1
	%newcnt =l add %cnt, %odd
	storel %newcnt, %cntp
	%next =l add %i, 1
	storel %next, %ip
	jmp @test
@done
	%finalmin =l loadl %minp
	%finalmax =l loadl %maxp
	%finalsum =l loadl %sump
	storel %finalmin, %out
	%out1 =l add %out, 8
	storel %finalmax, %out1
	%out2 =l add %out, 16
	storel %finalsum, %out2
	%result =l loadl %cntp
	ret %result
}
