export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet

@packet
	%sp =l phi @start %src, @lit_continue %lit_spend, @rep_continue %rep_spend, @nop %sph
	%srem =l phi @start %n, @lit_continue %lit_remend, @rep_continue %rep_remend, @nop %sremh
	%op =l phi @start %out, @lit_continue %lit_opend, @rep_continue %rep_opend, @nop %op
	%written =l phi @start 0, @lit_continue %lit_writtenend, @rep_continue %rep_writtenend, @nop %written
	%empty =w ceql %srem, 0
	jnz %empty, @success, @header

@header
	%h =w loadsb %sp
	%sph =l add %sp, 1
	%sremh =l sub %srem, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @classify

@nop
	jmp @packet

@classify
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat

@literal
	%lit_lenw =w add %h, 1
	%lit_len =l extuw %lit_lenw
	%lit_badsrc =w cultl %sremh, %lit_len
	jnz %lit_badsrc, @source_error, @lit_capacity

@lit_capacity
	%lit_avail =l sub %cap, %written
	%lit_over =w cultl %lit_avail, %lit_len
	jnz %lit_over, @lit_overflow_setup, @lit_normal_setup

@lit_overflow_setup
	%lit_count_over =l copy %lit_avail
	%lit_status_over =w copy 1
	jmp @lit_copy

@lit_normal_setup
	%lit_count_normal =l copy %lit_len
	%lit_status_normal =w copy 0
	jmp @lit_copy

@lit_copy
	%lit_count =l phi @lit_overflow_setup %lit_count_over, @lit_normal_setup %lit_count_normal
	%lit_status =w phi @lit_overflow_setup %lit_status_over, @lit_normal_setup %lit_status_normal
	%lit_i =l phi @lit_overflow_setup 0, @lit_normal_setup 0, @lit_body %lit_inext
	%lit_done =w ceql %lit_i, %lit_count
	jnz %lit_done, @lit_finish, @lit_body

@lit_body
	%lit_srcp =l add %sph, %lit_i
	%lit_dstp =l add %op, %lit_i
	%lit_byte =w loadub %lit_srcp
	storeb %lit_byte, %lit_dstp
	%lit_inext =l add %lit_i, 1
	jmp @lit_copy

@lit_finish
	jnz %lit_status, @capacity_error, @lit_continue

@lit_continue
	%lit_spend =l add %sph, %lit_len
	%lit_remend =l sub %sremh, %lit_len
	%lit_opend =l add %op, %lit_len
	%lit_writtenend =l add %written, %lit_len
	jmp @packet

@repeat
	%rep_badsrc =w ceql %sremh, 0
	jnz %rep_badsrc, @source_error, @rep_prepare

@rep_prepare
	%rep_byte =w loadub %sph
	%rep_spend =l add %sph, 1
	%rep_remend =l sub %sremh, 1
	%rep_lenw =w sub 1, %h
	%rep_len =l extuw %rep_lenw
	%rep_avail =l sub %cap, %written
	%rep_over =w cultl %rep_avail, %rep_len
	jnz %rep_over, @rep_overflow_setup, @rep_normal_setup

@rep_overflow_setup
	%rep_count_over =l copy %rep_avail
	%rep_status_over =w copy 1
	jmp @rep_copy

@rep_normal_setup
	%rep_count_normal =l copy %rep_len
	%rep_status_normal =w copy 0
	jmp @rep_copy

@rep_copy
	%rep_count =l phi @rep_overflow_setup %rep_count_over, @rep_normal_setup %rep_count_normal
	%rep_status =w phi @rep_overflow_setup %rep_status_over, @rep_normal_setup %rep_status_normal
	%rep_i =l phi @rep_overflow_setup 0, @rep_normal_setup 0, @rep_body %rep_inext
	%rep_done =w ceql %rep_i, %rep_count
	jnz %rep_done, @rep_finish, @rep_body

@rep_body
	%rep_dstp =l add %op, %rep_i
	storeb %rep_byte, %rep_dstp
	%rep_inext =l add %rep_i, 1
	jmp @rep_copy

@rep_finish
	jnz %rep_status, @capacity_error, @rep_continue

@rep_continue
	%rep_opend =l add %op, %rep_len
	%rep_writtenend =l add %written, %rep_len
	jmp @packet

@success
	ret %written

@source_error
	ret -2

@capacity_error
	ret -1
}
