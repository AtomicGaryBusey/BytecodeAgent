export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%rp =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	storel 0, %rp
	jmp @loop

@loop
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch

@fetch
	%coff =l shl %pc, 3
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @test1

@test1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @test2

@test2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @test3

@test3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @test4

@test4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @test5

@test5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @test6

@test6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @test7

@test7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @test8

@test8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @test9

@test9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @test10

@test10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @test11

@test11
	%is11 =w ceql %op, 11
	jnz %is11, @jmp, @unknown

@push
	%ppc =l loadl %pcp
	%vpc =l add %ppc, 1
	%voff =l shl %vpc, 3
	%vaddr =l add %code, %voff
	%v =l loadl %vaddr
	%pd =l loadl %dp
	%pdoff =l shl %pd, 3
	%pdaddr =l add %stack, %pdoff
	storel %v, %pdaddr
	%pd1 =l add %pd, 1
	storel %pd1, %dp
	%ppc2 =l add %ppc, 2
	storel %ppc2, %pcp
	jmp @loop

@add
	%ad =l loadl %dp
	%ai =l sub %ad, 2
	%aj =l sub %ad, 1
	%aio =l shl %ai, 3
	%ajo =l shl %aj, 3
	%aia =l add %stack, %aio
	%aja =l add %stack, %ajo
	%aa =l loadl %aia
	%ab =l loadl %aja
	%ar =l add %aa, %ab
	storel %ar, %aia
	%ad1 =l sub %ad, 1
	storel %ad1, %dp
	%apc =l loadl %pcp
	%apc1 =l add %apc, 1
	storel %apc1, %pcp
	jmp @loop

@sub
	%sd =l loadl %dp
	%si =l sub %sd, 2
	%sj =l sub %sd, 1
	%sio =l shl %si, 3
	%sjo =l shl %sj, 3
	%sia =l add %stack, %sio
	%sja =l add %stack, %sjo
	%sa =l loadl %sia
	%sb =l loadl %sja
	%sr =l sub %sa, %sb
	storel %sr, %sia
	%sd1 =l sub %sd, 1
	storel %sd1, %dp
	%spc =l loadl %pcp
	%spc1 =l add %spc, 1
	storel %spc1, %pcp
	jmp @loop

@mul
	%md =l loadl %dp
	%mi =l sub %md, 2
	%mj =l sub %md, 1
	%mio =l shl %mi, 3
	%mjo =l shl %mj, 3
	%mia =l add %stack, %mio
	%mja =l add %stack, %mjo
	%ma =l loadl %mia
	%mb =l loadl %mja
	%mr =l mul %ma, %mb
	storel %mr, %mia
	%md1 =l sub %md, 1
	storel %md1, %dp
	%mpc =l loadl %pcp
	%mpc1 =l add %mpc, 1
	storel %mpc1, %pcp
	jmp @loop

@dup
	%dd =l loadl %dp
	%di =l sub %dd, 1
	%dio =l shl %di, 3
	%ddo =l shl %dd, 3
	%dia =l add %stack, %dio
	%dda =l add %stack, %ddo
	%dv =l loadl %dia
	storel %dv, %dda
	%dd1 =l add %dd, 1
	storel %dd1, %dp
	%dpc =l loadl %pcp
	%dpc1 =l add %dpc, 1
	storel %dpc1, %pcp
	jmp @loop

@swap
	%swd =l loadl %dp
	%swi =l sub %swd, 2
	%swj =l sub %swd, 1
	%swio =l shl %swi, 3
	%swjo =l shl %swj, 3
	%swia =l add %stack, %swio
	%swja =l add %stack, %swjo
	%swa =l loadl %swia
	%swb =l loadl %swja
	storel %swb, %swia
	storel %swa, %swja
	%swpc =l loadl %pcp
	%swpc1 =l add %swpc, 1
	storel %swpc1, %pcp
	jmp @loop

@drop
	%drd =l loadl %dp
	%drd1 =l sub %drd, 1
	storel %drd1, %dp
	%drpc =l loadl %pcp
	%drpc1 =l add %drpc, 1
	storel %drpc1, %pcp
	jmp @loop

@over
	%od =l loadl %dp
	%oi =l sub %od, 2
	%oio =l shl %oi, 3
	%odo =l shl %od, 3
	%oia =l add %stack, %oio
	%oda =l add %stack, %odo
	%ov =l loadl %oia
	storel %ov, %oda
	%od1 =l add %od, 1
	storel %od1, %dp
	%opc =l loadl %pcp
	%opc1 =l add %opc, 1
	storel %opc1, %pcp
	jmp @loop

@lt
	%ld =l loadl %dp
	%li =l sub %ld, 2
	%lj =l sub %ld, 1
	%lio =l shl %li, 3
	%ljo =l shl %lj, 3
	%lia =l add %stack, %lio
	%lja =l add %stack, %ljo
	%la =l loadl %lia
	%lb =l loadl %lja
	%lc =w csltl %la, %lb
	%lr =l extuw %lc
	storel %lr, %lia
	%ld1 =l sub %ld, 1
	storel %ld1, %dp
	%lpc =l loadl %pcp
	%lpc1 =l add %lpc, 1
	storel %lpc1, %pcp
	jmp @loop

@jz
	%jzd =l loadl %dp
	%jzd1 =l sub %jzd, 1
	storel %jzd1, %dp
	%jzoff =l shl %jzd1, 3
	%jzaddr =l add %stack, %jzoff
	%jzv =l loadl %jzaddr
	%jzpc =l loadl %pcp
	%jznext =l add %jzpc, 1
	%jznoff =l shl %jznext, 3
	%jzna =l add %code, %jznoff
	%jzt =l loadl %jzna
	%jzz =w ceql %jzv, 0
	jnz %jzz, @jztaken, @jzfall

@jztaken
	storel %jzt, %pcp
	jmp @loop

@jzfall
	%jzpc2 =l add %jzpc, 2
	storel %jzpc2, %pcp
	jmp @loop

@jmp
	%jpc =l loadl %pcp
	%jnext =l add %jpc, 1
	%jnoff =l shl %jnext, 3
	%jna =l add %code, %jnoff
	%jt =l loadl %jna
	storel %jt, %pcp
	jmp @loop

@unknown
	%upc =l loadl %pcp
	%upc1 =l add %upc, 1
	storel %upc1, %pcp
	jmp @loop

@finish
	%fd =l loadl %dp
	%fempty =w ceql %fd, 0
	jnz %fempty, @clearinit, @saveresult

@saveresult
	%fti =l sub %fd, 1
	%fto =l shl %fti, 3
	%fta =l add %stack, %fto
	%fr =l loadl %fta
	storel %fr, %rp
	jmp @clearinit

@clearinit
	%ci =l loadl %dp
	storel %ci, %pcp
	jmp @clearloop

@clearloop
	%cl =l loadl %pcp
	%cdone =w ceql %cl, %cap
	jnz %cdone, @return, @clearone

@clearone
	%clo =l shl %cl, 3
	%cla =l add %stack, %clo
	storel 0, %cla
	%cl1 =l add %cl, 1
	storel %cl1, %pcp
	jmp @clearloop

@return
	%ret =l loadl %rp
	ret %ret
}
