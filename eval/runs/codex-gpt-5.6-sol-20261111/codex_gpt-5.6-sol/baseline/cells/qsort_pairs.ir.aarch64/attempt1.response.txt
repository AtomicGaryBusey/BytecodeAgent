export function l $qsort_pairs(l %a, l %n) {
@entry
	%more =w csltl 1, %n
	jnz %more, @start, @done

@start
	%last =l sub %n, 1
	%loff =l shl %last, 4
	%lp =l add %a, %loff
	%pk =l loadl %lp
	%lpv =l add %lp, 8
	%pv =l loadl %lpv
	jmp @loop

@loop
	%i =l phi @start 0, @next %inew
	%j =l phi @start 0, @next %jnext
	%cont =w csltl %j, %last
	jnz %cont, @inspect, @partitioned

@inspect
	%joff =l shl %j, 4
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jpv =l add %jp, 8
	%jv =l loadl %jpv
	%kless =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vless =w csltl %jv, %pv
	%tie =w and %keq, %vless
	%take =w or %kless, %tie
	jnz %take, @swap, @noswap

@swap
	%ioff =l shl %i, 4
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ipv =l add %ip, 8
	%iv =l loadl %ipv
	storel %jk, %ip
	storel %jv, %ipv
	storel %ik, %jp
	storel %iv, %jpv
	%iplus =l add %i, 1
	jmp @next

@noswap
	jmp @next

@next
	%inew =l phi @swap %iplus, @noswap %i
	%jnext =l add %j, 1
	jmp @loop

@partitioned
	%foff =l shl %i, 4
	%fp =l add %a, %foff
	%fk =l loadl %fp
	%fpv =l add %fp, 8
	%fv =l loadl %fpv
	storel %pk, %fp
	storel %pv, %fpv
	storel %fk, %lp
	storel %fv, %lpv
	%left =l call $qsort_pairs(l %a, l %i)
	%ri =l add %i, 1
	%roff =l shl %ri, 4
	%rp =l add %a, %roff
	%rn =l sub %last, %i
	%right =l call $qsort_pairs(l %rp, l %rn)
	jmp @done

@done
	ret 0
}
