export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@zero
	ret 0
@init
	%pairs =l sub %n, 1
	jmp @head
@head
	%p =l phi @init %a, @body %nextp
	%left =l phi @init %pairs, @body %nextleft
	%count =l phi @init 0, @body %nextcount
	%nextp =l add %p, 8
	%cur =l loadl %p
	%next =l loadl %nextp
	%desc =w csltl %next, %cur
	%inc =l extuw %desc
	%nextcount =l add %count, %inc
	%nextleft =l sub %left, 1
	%more =w cnel %nextleft, 0
	jnz %more, @body, @done
@body
	jmp @head
@done
	ret %nextcount
}
