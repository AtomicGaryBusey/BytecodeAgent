export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @head
@zero
	ret 0
@head
	%p =l phi @start %a, @body %nextp
	%left =l phi @start %pairs, @body %nextleft
	%count =l phi @start 0, @body %nextcount
	%pairs =l sub %n, 1
	%nextp =l add %p, 8
	%cur =l loadl %p
	%next =l loadl %nextp
	%desc =w csltl %next, %cur
	%inc =l extuw %desc
	%nextcount =l add %count, %inc
	%nextleft =l sub %left, 1
	%more =w cnel %nextleft, 0
	jnz %more, @body, @done
@body
	jmp @head
@done
	ret %nextcount
}
