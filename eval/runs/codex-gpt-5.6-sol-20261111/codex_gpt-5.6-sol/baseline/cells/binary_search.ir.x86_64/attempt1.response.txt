export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lower %nextlo, @upper %lo
	%hi =l phi @start %n, @lower %hi, @upper %mid
	%more =w csltl %lo, %hi
	jnz %more, @probe, @missing
@probe
	%span =l sub %hi, %lo
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%off =l shl %mid, 3
	%ptr =l add %a, %off
	%value =l loadl %ptr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @lower, @upper
@lower
	%nextlo =l add %mid, 1
	jmp @loop
@upper
	jmp @loop
@found
	ret %mid
@missing
	ret -1
}
