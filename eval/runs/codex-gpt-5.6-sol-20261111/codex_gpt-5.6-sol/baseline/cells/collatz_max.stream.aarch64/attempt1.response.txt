collatz_max:
    mov x1, #1
    mov x2, #0
.Louter:
    mov x3, x1
    mov x4, #0
.Lcollatz:
    cmp x3, #1
    b.eq .Lfinished
    add x4, x4, #1
    tbz x3, #0, .Leven
    add x3, x3, x3, lsl #1
    add x3, x3, #1
    b .Lcollatz
.Leven:
    lsr x3, x3, #1
    b .Lcollatz
.Lfinished:
    cmp x4, x2
    csel x2, x4, x2, hi
    cmp x1, x0
    b.eq .Lreturn
    add x1, x1, #1
    b .Louter
.Lreturn:
    mov x0, x2
    ret
