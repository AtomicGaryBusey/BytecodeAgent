export function l $collatz_max(l %n) {
@start
	%kp =l alloc8 8
	%xp =l alloc8 8
	%sp =l alloc8 8
	%mp =l alloc8 8
	storel 1, %kp
	storel 0, %mp
	jmp @next

@next
	%k =l loadl %kp
	storel %k, %xp
	storel 0, %sp
	jmp @collatz

@collatz
	%x =l loadl %xp
	%done =w ceql %x, 1
	jnz %done, @finished, @step

@step
	%bit =l and %x, 1
	%even =w ceql %bit, 0
	jnz %even, @even, @odd

@even
	%half =l div %x, 2
	storel %half, %xp
	jmp @count

@odd
	%triple =l mul %x, 3
	%oddnext =l add %triple, 1
	storel %oddnext, %xp
	jmp @count

@count
	%steps0 =l loadl %sp
	%steps1 =l add %steps0, 1
	storel %steps1, %sp
	jmp @collatz

@finished
	%steps =l loadl %sp
	%max =l loadl %mp
	%greater =w csltl %max, %steps
	jnz %greater, @update, @advance

@update
	storel %steps, %mp
	jmp @advance

@advance
	%current =l loadl %kp
	%last =w ceql %current, %n
	jnz %last, @return, @increment

@increment
	%nextk =l add %current, 1
	storel %nextk, %kp
	jmp @next

@return
	%result =l loadl %mp
	ret %result
}
