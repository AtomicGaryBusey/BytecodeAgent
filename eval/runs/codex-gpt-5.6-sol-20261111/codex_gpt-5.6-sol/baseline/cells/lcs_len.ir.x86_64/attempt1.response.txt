export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	storel 0, %row
	%bound =l add %n, 1
	jmp @zero_check

@zero_check
	%zj =l phi @start 1, @zero_body %zj_next
	%zmore =w csltl %zj, %bound
	jnz %zmore, @zero_body, @outer_check

@zero_body
	%zoff =l mul %zj, 8
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%zj_next =l add %zj, 1
	jmp @zero_check

@outer_check
	%i =l phi @zero_check 0, @inner_done %i_next
	%imore =w csltl %i, %m
	jnz %imore, @outer_body, @done

@outer_body
	%xptr =l add %x, %i
	%xb =w loadub %xptr
	jmp @inner_check

@inner_check
	%j =l phi @outer_body 1, @store %j_next
	%diag =l phi @outer_body 0, @store %up
	%jmore =w csltl %j, %bound
	jnz %jmore, @inner_body, @inner_done

@inner_body
	%off =l mul %j, 8
	%rptr =l add %row, %off
	%up =l loadl %rptr
	%leftptr =l sub %rptr, 8
	%left =l loadl %leftptr
	%yj =l sub %j, 1
	%yptr =l add %y, %yj
	%yb =w loadub %yptr
	%equal =w ceqw %xb, %yb
	jnz %equal, @match, @nomatch

@match
	%matchval =l add %diag, 1
	jmp @store

@nomatch
	%leftgreater =w csltl %up, %left
	jnz %leftgreater, @use_left, @use_up

@use_left
	jmp @store

@use_up
	jmp @store

@store
	%value =l phi @match %matchval, @use_left %left, @use_up %up
	storel %value, %rptr
	%j_next =l add %j, 1
	jmp @inner_check

@inner_done
	%i_next =l add %i, 1
	jmp @outer_check

@done
	%noff =l mul %n, 8
	%nptr =l add %row, %noff
	%result =l loadl %nptr
	ret %result
}
