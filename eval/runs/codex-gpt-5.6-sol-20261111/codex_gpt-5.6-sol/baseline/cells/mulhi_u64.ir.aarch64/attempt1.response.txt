export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00h =l shr %p00, 32
	%p01l =l and %p01, 4294967295
	%p10l =l and %p10, 4294967295
	%mid0 =l add %p00h, %p01l
	%mid =l add %mid0, %p10l
	%carry =l shr %mid, 32
	%p01h =l shr %p01, 32
	%p10h =l shr %p10, 32
	%hi0 =l add %p11, %p01h
	%hi1 =l add %hi0, %p10h
	%hi =l add %hi1, %carry
	ret %hi
}
