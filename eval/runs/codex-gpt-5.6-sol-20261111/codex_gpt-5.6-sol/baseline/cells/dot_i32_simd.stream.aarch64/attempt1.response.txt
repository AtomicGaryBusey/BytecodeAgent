dot_i32_simd:
    movi v0.2d, #0
    movi v1.2d, #0
.Lvector_loop:
    cmp x2, #4
    b.lt .Lvector_done
    ldr q2, [x0], #16
    ldr q3, [x1], #16
    smlal v0.2d, v2.2s, v3.2s
    smlal2 v1.2d, v2.4s, v3.4s
    sub x2, x2, #4
    b .Lvector_loop
.Lvector_done:
    add v0.2d, v0.2d, v1.2d
    umov x3, v0.d[0]
    umov x4, v0.d[1]
    add x3, x3, x4
.Lscalar_loop:
    cbz x2, .Ldone
    ldrsw x4, [x0], #4
    ldrsw x5, [x1], #4
    mul x4, x4, x5
    add x3, x3, x4
    sub x2, x2, #1
    b .Lscalar_loop
.Ldone:
    mov x0, x3
    ret
