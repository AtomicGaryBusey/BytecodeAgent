export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %kn
	%max =l phi @start 0, @next %maxnext
	%within =w cslel %k, %n
	jnz %within, @inner_entry, @done
@inner_entry
	jmp @inner
@inner
	%x =l phi @inner_entry %k, @even %xe, @odd %xo
	%steps =l phi @inner_entry 0, @even %se, @odd %so
	%finished =w ceql %x, 1
	jnz %finished, @check_max, @transform
@transform
	%bit =l and %x, 1
	%isodd =w cnel %bit, 0
	jnz %isodd, @odd, @even
@even
	%xe =l shr %x, 1
	%se =l add %steps, 1
	jmp @inner
@odd
	%triple =l mul %x, 3
	%xo =l add %triple, 1
	%so =l add %steps, 1
	jmp @inner
@check_max
	%greater =w csgtl %steps, %max
	jnz %greater, @new_max, @keep_max
@new_max
	jmp @next
@keep_max
	jmp @next
@next
	%maxnext =l phi @new_max %steps, @keep_max %max
	%kn =l add %k, 1
	jmp @outer
@done
	ret %max
}
