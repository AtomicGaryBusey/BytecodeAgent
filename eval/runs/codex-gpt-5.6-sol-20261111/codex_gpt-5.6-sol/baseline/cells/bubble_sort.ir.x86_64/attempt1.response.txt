export function l $bubble_sort(l %a, l %n) {
@start
	%many =w csltl 1, %n
	jnz %many, @init, @done
@init
	%end0 =l sub %n, 1
	jmp @outer
@outer
	%end =l phi @init %end0, @after %endnext
	%i0 =l copy 0
	jmp @inner
@inner
	%i =l phi @outer %i0, @step %inext
	%more =w csltl %i, %end
	jnz %more, @body, @after
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csltl %y, %x
	jnz %swap, @swap, @step
@swap
	storel %y, %p
	storel %x, %q
	jmp @step
@step
	%inext =l add %i, 1
	jmp @inner
@after
	%endnext =l sub %end, 1
	%again =w csltl 0, %endnext
	jnz %again, @outer, @done
@done
	ret 0
}
