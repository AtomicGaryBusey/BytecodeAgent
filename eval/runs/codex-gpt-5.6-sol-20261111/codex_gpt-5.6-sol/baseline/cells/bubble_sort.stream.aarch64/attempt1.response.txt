bubble_sort:

    b.lt .Ldone
    sub x9, x1, #1
.Louter:
    mov x10, #0
    mov x11, #0
.Linner:

    add x14, x10, #1
    ldr x13, [x0, x14, lsl #3]
    cmp x12, x13
    b.le .Lnoswap

.Lnoswap:

    cmp x10, x9
    b.lt .Linner

    sub x9, x9, #1
    cbnz x9, .Louter
.Ldone:

    ret
