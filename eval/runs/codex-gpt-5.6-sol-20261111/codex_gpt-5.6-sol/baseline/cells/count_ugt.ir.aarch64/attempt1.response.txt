export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%p =l phi @start %a, @body %pnext
	%i =l phi @start 0, @body %inext
	%count =l phi @start 0, @body %cnext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%value =l loadl %p
	%gt =w cultl %t, %value
	%inc =l extuw %gt
	%cnext =l add %count, %inc
	%pnext =l add %p, 8
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
}
