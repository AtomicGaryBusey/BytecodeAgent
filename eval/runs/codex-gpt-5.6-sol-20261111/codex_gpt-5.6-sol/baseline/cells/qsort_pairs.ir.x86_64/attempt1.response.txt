export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @init

@init
	%islot =l alloc8 8
	%jslot =l alloc8 8
	%mid =l div %n, 2
	%moff =l mul %mid, 16
	%mptr =l add %a, %moff
	%pkey =l loadl %mptr
	%mvptr =l add %mptr, 8
	%pval =l loadl %mvptr
	storel 0, %islot
	%nminus =l sub %n, 1
	storel %nminus, %jslot
	jmp @left_scan

@left_scan
	%i0 =l loadl %islot
	%ioff0 =l mul %i0, 16
	%iptr0 =l add %a, %ioff0
	%ikey0 =l loadl %iptr0
	%iklt =w csltl %ikey0, %pkey
	jnz %iklt, @left_inc, @left_key_ge

@left_key_ge
	%ikeq =w ceql %ikey0, %pkey
	jnz %ikeq, @left_value, @right_scan

@left_value
	%ivptr0 =l add %iptr0, 8
	%ival0 =l loadl %ivptr0
	%ivlt =w csltl %ival0, %pval
	jnz %ivlt, @left_inc, @right_scan

@left_inc
	%i1 =l add %i0, 1
	storel %i1, %islot
	jmp @left_scan

@right_scan
	%j0 =l loadl %jslot
	%joff0 =l mul %j0, 16
	%jptr0 =l add %a, %joff0
	%jkey0 =l loadl %jptr0
	%pklt =w csltl %pkey, %jkey0
	jnz %pklt, @right_dec, @right_key_le

@right_key_le
	%pkeq =w ceql %pkey, %jkey0
	jnz %pkeq, @right_value, @check_cross

@right_value
	%jvptr0 =l add %jptr0, 8
	%jval0 =l loadl %jvptr0
	%pvlt =w csltl %pval, %jval0
	jnz %pvlt, @right_dec, @check_cross

@right_dec
	%j1 =l sub %j0, 1
	storel %j1, %jslot
	jmp @right_scan

@check_cross
	%ic =l loadl %islot
	%jc =l loadl %jslot
	%crossed =w csltl %jc, %ic
	jnz %crossed, @partitioned, @swap

@swap
	%ioff =l mul %ic, 16
	%iptr =l add %a, %ioff
	%ivptr =l add %iptr, 8
	%joff =l mul %jc, 16
	%jptr =l add %a, %joff
	%jvptr =l add %jptr, 8
	%ik =l loadl %iptr
	%iv =l loadl %ivptr
	%jk =l loadl %jptr
	%jv =l loadl %jvptr
	storel %jk, %iptr
	storel %jv, %ivptr
	storel %ik, %jptr
	storel %iv, %jvptr
	%inext =l add %ic, 1
	%jnext =l sub %jc, 1
	storel %inext, %islot
	storel %jnext, %jslot
	jmp @left_scan

@partitioned
	%jf =l loadl %jslot
	%leftn =l add %jf, 1
	%leftbig =w csltl 1, %leftn
	jnz %leftbig, @call_left, @check_right

@call_left
	%lr =l call $qsort_pairs(l %a, l %leftn)
	jmp @check_right

@check_right
	%if =l loadl %islot
	%rightn =l sub %n, %if
	%rightbig =w csltl 1, %rightn
	jnz %rightbig, @call_right, @done

@call_right
	%rightoff =l mul %if, 16
	%rightptr =l add %a, %rightoff
	%rr =l call $qsort_pairs(l %rightptr, l %rightn)
	jmp @done

@done
	ret 0
}
