export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =w ceql %n, 0
	jnz %zero, @done, @direction

@direction
	%forward =w cultl %dst, %src
	jnz %forward, @forward, @backward_init

@forward
	%fi =l copy 0
	%fd =l copy %dst
	%fs =l copy %src
	jmp @forward_test

@forward_test
	%fmore =w csltl %fi, %n
	jnz %fmore, @forward_copy, @done

@forward_copy
	%fv =l loadl %fs
	storel %fv, %fd
	%fs1 =l add %fs, 8
	%fd1 =l add %fd, 8
	%fi1 =l add %fi, 1
	%fs =l copy %fs1
	%fd =l copy %fd1
	%fi =l copy %fi1
	jmp @forward_test

@backward_init
	%bytes =l mul %n, 8
	%bd0 =l add %dst, %bytes
	%bs0 =l add %src, %bytes
	%bi =l copy %n
	%bd =l copy %bd0
	%bs =l copy %bs0
	jmp @backward_test

@backward_test
	%bmore =w cnel %bi, 0
	jnz %bmore, @backward_copy, @done

@backward_copy
	%bs1 =l sub %bs, 8
	%bd1 =l sub %bd, 8
	%bv =l loadl %bs1
	storel %bv, %bd1
	%bi1 =l sub %bi, 1
	%bs =l copy %bs1
	%bd =l copy %bd1
	%bi =l copy %bi1
	jmp @backward_test

@done
	ret 0
}
