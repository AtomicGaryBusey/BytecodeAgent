packbits_encode:
mov r8, rdx
lea r9, [rdi + rsi]
lea r10, [rdx + rcx]
.Lmain:
cmp rdi, r9
jae .Lsuccess
movzx eax, byte ptr [rdi]
mov r11, 1
.Lrun_loop:
cmp r11, 128
jae .Lrun_done
lea rcx, [rdi + r11]
cmp rcx, r9
jae .Lrun_done
movzx ecx, byte ptr [rcx]
cmp cl, al
jne .Lrun_done
inc r11
jmp .Lrun_loop
.Lrun_done:
cmp r11, 3
jb .Lliteral
mov rsi, r10
sub rsi, rdx
cmp rsi, 2
jb .Lfailure
mov eax, 257
sub eax, r11d
mov byte ptr [rdx], al
movzx eax, byte ptr [rdi]
mov byte ptr [rdx + 1], al
add rdi, r11
add rdx, 2
jmp .Lmain
.Lliteral:
mov r11, rdi
.Lliteral_scan:
mov rcx, r11
sub rcx, rdi
cmp rcx, 128
jae .Lliteral_ready
cmp r11, r9
jae .Lliteral_ready
movzx eax, byte ptr [r11]
mov rsi, 1
.Lliteral_run:
cmp rsi, 128
jae .Lliteral_run_done
lea rcx, [r11 + rsi]
cmp rcx, r9
jae .Lliteral_run_done
movzx ecx, byte ptr [rcx]
cmp cl, al
jne .Lliteral_run_done
inc rsi
jmp .Lliteral_run
.Lliteral_run_done:
cmp rsi, 3
jae .Lliteral_ready
inc r11
jmp .Lliteral_scan
.Lliteral_ready:
mov rcx, r11
sub rcx, rdi
lea rax, [rcx + 1]
mov rsi, r10
sub rsi, rdx
cmp rsi, rax
jb .Lfailure
mov eax, ecx
dec eax
mov byte ptr [rdx], al
xor rsi, rsi
.Lcopy_literal:
cmp rsi, rcx
jae .Lcopy_done
movzx eax, byte ptr [rdi + rsi]
mov byte ptr [rdx + rsi + 1], al
inc rsi
jmp .Lcopy_literal
.Lcopy_done:
lea rdx, [rdx + rcx + 1]
mov rdi, r11
jmp .Lmain
.Lsuccess:
mov rax, rdx
sub rax, r8
ret
.Lfailure:
mov rax, -1
ret
