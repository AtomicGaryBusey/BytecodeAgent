export function l $bmin_empty(l %a, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @empty, @init
@empty
	ret 9223372036854775807
@init
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @init 1, @latch %next
	%min =l phi @init %first, @latch %newmin
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%less =w csltl %val, %min
	jnz %less, @take, @keep
@take
	jmp @latch
@keep
	jmp @latch
@latch
	%newmin =l phi @take %val, @keep %min
	%next =l add %i, 1
	jmp @loop
@done
	ret %min
}
