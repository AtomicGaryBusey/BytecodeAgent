export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @good %mid, @bad %lo
	%hi =l phi @start 4294967296, @good %hi, @bad %mid
	%delta =l sub %hi, %lo
	%more =w cugtl %delta, 1
	jnz %more, @test, @done
@test
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%square =l mul %mid, %mid
	%fits =w culel %square, %n
	jnz %fits, @good, @bad
@good
	jmp @loop
@bad
	jmp @loop
@done
	ret %lo
}
