export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @loop

@loop
	%si =l phi @entry 0, @nop %si_nop, @litfinish %si_lnext, @repfinish %si_rnext
	%oi =l phi @entry 0, @nop %oi, @litfinish %oi_lnext, @repfinish %oi_rnext
	%more =w csltl %si, %n
	jnz %more, @header, @success

@header
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si1 =l add %si, 1
	%neg =w csltw %h, 0
	jnz %neg, @negative, @literal

@negative
	%noptest =w ceqw %h, -128
	jnz %noptest, @nop, @repeat

@nop
	%si_nop =l copy %si1
	jmp @loop

@literal
	%hl =l extuw %h
	%llen =l add %hl, 1
	%rem =l sub %n, %si1
	%ltrunc =w csltl %rem, %llen
	jnz %ltrunc, @source_error, @litcap

@litcap
	%lavail =l sub %cap, %oi
	%lover =w csltl %lavail, %llen
	jnz %lover, @litoverflow, @litfull

@litfull
	%lcnt_full =l copy %llen
	%lstatus_full =w copy 0
	%lsi_full =l add %si1, %llen
	jmp @litsetup

@litoverflow
	%lcnt_over =l copy %lavail
	%lstatus_over =w copy 1
	%lsi_over =l copy %si1
	jmp @litsetup

@litsetup
	%lcnt =l phi @litfull %lcnt_full, @litoverflow %lcnt_over
	%lstatus =w phi @litfull %lstatus_full, @litoverflow %lstatus_over
	%lsi_next =l phi @litfull %lsi_full, @litoverflow %lsi_over
	jmp @litloop

@litloop
	%li =l phi @litsetup 0, @litbody %li_next
	%lmore =w csltl %li, %lcnt
	jnz %lmore, @litbody, @litfinish

@litbody
	%lsp =l add %src, %si1
	%lsp_i =l add %lsp, %li
	%lb =w loadub %lsp_i
	%ldp =l add %out, %oi
	%ldp_i =l add %ldp, %li
	storeb %lb, %ldp_i
	%li_next =l add %li, 1
	jmp @litloop

@litfinish
	%oi_lnext =l add %oi, %lcnt
	%si_lnext =l copy %lsi_next
	jnz %lstatus, @output_error, @loop

@repeat
	%rhas =w csltl %si1, %n
	jnz %rhas, @repload, @source_error

@repload
	%rp =l add %src, %si1
	%rv =w loadub %rp
	%hs =l extsw %h
	%rlen =l sub 1, %hs
	%ravail =l sub %cap, %oi
	%rover =w csltl %ravail, %rlen
	jnz %rover, @repoverflow, @repfull

@repfull
	%rcnt_full =l copy %rlen
	%rstatus_full =w copy 0
	%rsi_full =l add %si1, 1
	jmp @repsetup

@repoverflow
	%rcnt_over =l copy %ravail
	%rstatus_over =w copy 1
	%rsi_over =l copy %si1
	jmp @repsetup

@repsetup
	%rcnt =l phi @repfull %rcnt_full, @repoverflow %rcnt_over
	%rstatus =w phi @repfull %rstatus_full, @repoverflow %rstatus_over
	%rsi_next =l phi @repfull %rsi_full, @repoverflow %rsi_over
	jmp @reploop

@reploop
	%ri =l phi @repsetup 0, @repbody %ri_next
	%rmore =w csltl %ri, %rcnt
	jnz %rmore, @repbody, @repfinish

@repbody
	%rdp =l add %out, %oi
	%rdp_i =l add %rdp, %ri
	storeb %rv, %rdp_i
	%ri_next =l add %ri, 1
	jmp @reploop

@repfinish
	%oi_rnext =l add %oi, %rcnt
	%si_rnext =l copy %rsi_next
	jnz %rstatus, @output_error, @loop

@source_error
	ret -2

@output_error
	ret -1

@success
	ret %oi
}
