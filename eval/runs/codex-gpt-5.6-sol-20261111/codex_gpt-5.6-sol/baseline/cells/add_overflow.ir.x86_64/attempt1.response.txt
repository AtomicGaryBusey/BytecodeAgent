export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%pos =w csltl 0, %b
	jnz %pos, @check_pos, @check_neg
@check_pos
	%ov_pos =w csltl %sum, %a
	jnz %ov_pos, @overflow, @no_overflow
@check_neg
	%neg =w csltl %b, 0
	jnz %neg, @check_neg_overflow, @no_overflow
@check_neg_overflow
	%ov_neg =w csltl %a, %sum
	jnz %ov_neg, @overflow, @no_overflow
@overflow
	ret 1
@no_overflow
	ret 0
}
