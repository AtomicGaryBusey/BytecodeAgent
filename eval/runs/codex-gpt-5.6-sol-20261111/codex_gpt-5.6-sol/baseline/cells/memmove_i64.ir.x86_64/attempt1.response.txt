export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%forward =w culel %dst, %src
	jnz %forward, @floop, @bloop

@floop
	%i =l phi @start 0, @fbody %next
	%fmore =w cultl %i, %n
	jnz %fmore, @fbody, @done

@fbody
	%foff =l shl %i, 3
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fval =l loadl %fsrc
	storel %fval, %fdst
	%next =l add %i, 1
	jmp @floop

@bloop
	%j =l phi @start %n, @bbody %prev
	%bmore =w cnel %j, 0
	jnz %bmore, @bbody, @done

@bbody
	%prev =l sub %j, 1
	%boff =l shl %prev, 3
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bval =l loadl %bsrc
	storel %bval, %bdst
	jmp @bloop

@done
	ret 0
}
