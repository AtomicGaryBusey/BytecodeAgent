bubble_sort:
    cmp rsi, 2
    jb .Ldone
    lea rdx, [rsi - 1]
.Louter:
    xor ecx, ecx
    xor r10d, r10d
.Linner:
    mov r8, qword ptr [rdi + rcx*8]
    mov r9, qword ptr [rdi + rcx*8 + 8]
    cmp r8, r9
    jle .Lnext
    mov qword ptr [rdi + rcx*8], r9
    mov qword ptr [rdi + rcx*8 + 8], r8
    mov r10d, 1
.Lnext:
    inc rcx
    cmp rcx, rdx
    jb .Linner
    test r10d, r10d
    jz .Ldone
    dec rdx
    jnz .Louter
.Ldone:
    xor eax, eax
    ret
