export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%n1 =l add %n, 1
	jmp @zero_loop

@zero_loop
	%zj =l phi @start 0, @zero_body %znext
	%zc =w csltl %zj, %n1
	jnz %zc, @zero_body, @outer_start

@zero_body
	%zoff =l mul %zj, 8
	%zp =l add %row, %zoff
	storel 0, %zp
	%znext =l add %zj, 1
	jmp @zero_loop

@outer_start
	jmp @outer_loop

@outer_loop
	%i =l phi @outer_start 0, @outer_next %inext
	%ic =w csltl %i, %m
	jnz %ic, @outer_body, @done

@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner_loop

@inner_loop
	%j =l phi @outer_body 1, @inner_next %jnext
	%diag =l phi @outer_body 0, @inner_next %up
	%jc =w cslel %j, %n
	jnz %jc, @inner_body, @outer_next

@inner_body
	%off =l mul %j, 8
	%rp =l add %row, %off
	%up =l loadl %rp
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch

@match
	%matched =l add %diag, 1
	jmp @store

@mismatch
	%loff =l mul %jm1, 8
	%lp =l add %row, %loff
	%left =l loadl %lp
	%gt =w csgtl %up, %left
	jnz %gt, @take_up, @take_left

@take_up
	jmp @store_up

@take_left
	jmp @store_left

@store_up
	jmp @store_mismatch

@store_left
	jmp @store_mismatch

@store_mismatch
	%best =l phi @store_up %up, @store_left %left
	jmp @store2

@store
	storel %matched, %rp
	jmp @inner_next

@store2
	storel %best, %rp
	jmp @inner_next

@inner_next
	%jnext =l add %j, 1
	jmp @inner_loop

@outer_next
	%inext =l add %i, 1
	jmp @outer_loop

@done
	%roff =l mul %n, 8
	%resultp =l add %row, %roff
	%result =l loadl %resultp
	ret %result
}
