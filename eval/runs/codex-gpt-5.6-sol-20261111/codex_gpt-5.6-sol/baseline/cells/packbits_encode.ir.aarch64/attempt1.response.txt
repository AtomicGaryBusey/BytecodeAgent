export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @main

@main
	%i =l phi @entry 0, @next %nexti
	%o =l phi @entry 0, @next %nexto
	%done =w ceql %i, %n
	jnz %done, @success, @run

@run
	%k =l phi @main 1, @runinc %k2
	%kmax =w ceql %k, 128
	jnz %kmax, @rundone, @runbounds

@runbounds
	%ik =l add %i, %k
	%atend =w ceql %ik, %n
	jnz %atend, @rundone, @runcompare

@runcompare
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%pk =l add %src, %ik
	%bk =w loadub %pk
	%eq =w ceqw %b0, %bk
	jnz %eq, @runinc, @rundone

@runinc
	%k2 =l add %k, 1
	jmp @run

@rundone
	%isrep =w csgel %k, 3
	jnz %isrep, @repeat, @litstart

@repeat
	%ro2 =l add %o, 2
	%rfit =w cslel %ro2, %cap
	jnz %rfit, @repwrite, @failure

@repwrite
	%rh =l sub 257, %k
	%rhw =w copy %rh
	%rd0 =l add %out, %o
	storeb %rhw, %rd0
	%rd1 =l add %rd0, 1
	%rs =l add %src, %i
	%rb =w loadub %rs
	storeb %rb, %rd1
	%ri =l add %i, %k
	jmp @nextrep

@nextrep
	jmp @next

@litstart
	jmp @litcheck

@litcheck
	%j =l phi @litstart %i, @litadvance %j2
	%je =w ceql %j, %n
	jnz %je, @litdone, @litlencheck

@litlencheck
	%jl =l sub %j, %i
	%jmax =w ceql %jl, 128
	jnz %jmax, @litdone, @litruna

@litruna
	jmp @litrun

@litrun
	%rk =l phi @litruna 1, @litruninc %rk2
	%rkmax =w ceql %rk, 128
	jnz %rkmax, @litrundone, @litrunbounds

@litrunbounds
	%jr =l add %j, %rk
	%jrend =w ceql %jr, %n
	jnz %jrend, @litrundone, @litruncmp

@litruncmp
	%jp =l add %src, %j
	%jb =w loadub %jp
	%jrp =l add %src, %jr
	%jrb =w loadub %jrp
	%jreq =w ceqw %jb, %jrb
	jnz %jreq, @litruninc, @litrundone

@litruninc
	%rk2 =l add %rk, 1
	jmp @litrun

@litrundone
	%stoplit =w csgel %rk, 3
	jnz %stoplit, @litdone, @litadvance

@litadvance
	%j2 =l add %j, 1
	jmp @litcheck

@litdone
	%llen =l sub %j, %i
	%lneed0 =l add %o, 1
	%lneed =l add %lneed0, %llen
	%lfit =w cslel %lneed, %cap
	jnz %lfit, @litheader, @failure

@litheader
	%lh =l sub %llen, 1
	%lhw =w copy %lh
	%ldh =l add %out, %o
	storeb %lhw, %ldh
	jmp @copycheck

@copycheck
	%q =l phi @litheader 0, @copyinc %q2
	%qdone =w ceql %q, %llen
	jnz %qdone, @litcopydone, @copybyte

@copybyte
	%siq =l add %i, %q
	%spq =l add %src, %siq
	%cb =w loadub %spq
	%oq =l add %o, 1
	%doq =l add %oq, %q
	%dpq =l add %out, %doq
	storeb %cb, %dpq
	jmp @copyinc

@copyinc
	%q2 =l add %q, 1
	jmp @copycheck

@litcopydone
	jmp @nextlit

@nextlit
	jmp @next

@next
	%nexti =l phi @nextrep %ri, @nextlit %j
	%nexto =l phi @nextrep %ro2, @nextlit %lneed
	jmp @main

@success
	ret %o

@failure
	ret -1
}
