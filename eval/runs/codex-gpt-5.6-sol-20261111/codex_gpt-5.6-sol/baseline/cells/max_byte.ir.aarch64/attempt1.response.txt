export function l $max_byte(l %p, l %n) {
@start
	%zero =w ceql %n, 0
	jnz %zero, @empty, @loop
@loop
	%i =l phi @start 0, @cont %next
	%max =l phi @start 0, @cont %newmax
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w csltl %max, %b
	jnz %greater, @update, @keep
@update
	jmp @cont
@keep
	jmp @cont
@cont
	%newmax =l phi @update %b, @keep %max
	%next =l add %i, 1
	%done =w ceql %next, %n
	jnz %done, @exit, @loop
@exit
	ret %newmax
@empty
	ret 0
}
