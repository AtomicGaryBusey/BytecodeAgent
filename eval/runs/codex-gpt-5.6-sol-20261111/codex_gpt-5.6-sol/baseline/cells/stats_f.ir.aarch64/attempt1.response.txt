export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%bit0 =l and %y0, 1
	%odd0 =w cnel %bit0, 0
	%count0 =l extuw %odd0
	jmp @loop

@loop
	%i =l phi @start 1, @latch %ni
	%min =l phi @start %y0, @latch %nmin
	%max =l phi @start %y0, @latch %nmax
	%sum =l phi @start %y0, @latch %nsum
	%count =l phi @start %count0, @latch %ncount
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%less =w csltl %y, %min
	jnz %less, @minyes, @minno

@minyes
	jmp @minjoin

@minno
	jmp @minjoin

@minjoin
	%nmin =l phi @minyes %y, @minno %min
	%greater =w csltl %max, %y
	jnz %greater, @maxyes, @maxno

@maxyes
	jmp @maxjoin

@maxno
	jmp @maxjoin

@maxjoin
	%nmax =l phi @maxyes %y, @maxno %max
	%nsum =l add %sum, %y
	%bit =l and %y, 1
	%odd =w cnel %bit, 0
	%inc =l extuw %odd
	%ncount =l add %count, %inc
	%ni =l add %i, 1
	jmp @latch

@latch
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
