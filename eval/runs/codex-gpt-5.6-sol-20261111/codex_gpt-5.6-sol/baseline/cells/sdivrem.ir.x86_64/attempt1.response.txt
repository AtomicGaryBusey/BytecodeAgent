export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @divzero, @checkoverflow
@checkoverflow
	%bnegone =w ceql %b, -1
	jnz %bnegone, @checkmin, @divide
@checkmin
	%amin =w ceql %a, 0x8000000000000000
	jnz %amin, @overflow, @divide
@divzero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret 0x8000000000000000
@divide
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
