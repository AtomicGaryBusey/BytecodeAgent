export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @outer_init

@outer_init
	%initial_end =l sub %n, 1
	jmp @outer

@outer
	%end =l phi @outer_init %initial_end, @after_inner %next_end
	%active =w csgtl %end, 0
	jnz %active, @inner_init, @done

@inner_init
	jmp @inner

@inner
	%j =l phi @inner_init 0, @continue %next_j
	%more =w csltl %j, %end
	jnz %more, @compare, @after_inner

@compare
	%offset =l shl %j, 3
	%left_addr =l add %a, %offset
	%right_addr =l add %left_addr, 8
	%left =l loadl %left_addr
	%right =l loadl %right_addr
	%out_of_order =w csgtl %left, %right
	jnz %out_of_order, @swap, @continue

@swap
	storel %right, %left_addr
	storel %left, %right_addr
	jmp @continue

@continue
	%next_j =l add %j, 1
	jmp @inner

@after_inner
	%next_end =l sub %end, 1
	jmp @outer

@done
	ret 0
}
