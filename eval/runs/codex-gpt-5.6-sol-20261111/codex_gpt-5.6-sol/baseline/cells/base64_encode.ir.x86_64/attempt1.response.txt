export function l $base64_encode(l %src, l %n, l %out) {
@start
	%np2 =l add %n, 2
	%ng =l div %np2, 3
	%olen =l mul %ng, 4
	%full =l div %n, 3
	%rem =l rem %n, 3
	jmp @loop

@loop
	%i =l phi @start 0, @body %inext
	%sp =l phi @start %src, @body %spnext
	%op =l phi @start %out, @body %opnext
	%more =w csltl %i, %full
	jnz %more, @body, @tail

@body
	%b0 =l loadub %sp
	%p1 =l add %sp, 1
	%b1 =l loadub %p1
	%p2 =l add %sp, 2
	%b2 =l loadub %p2

	%x0 =l shr %b0, 2
	%x1a =l and %b0, 3
	%x1b =l shl %x1a, 4
	%x1c =l shr %b1, 4
	%x1 =l or %x1b, %x1c
	%x2a =l and %b1, 15
	%x2b =l shl %x2a, 2
	%x2c =l shr %b2, 6
	%x2 =l or %x2b, %x2c
	%x3 =l and %b2, 63

	%a0 =l add %x0, 65
	%q00 =w csltl 25, %x0
	%q01 =l extuw %q00
	%q02 =l mul %q01, 6
	%a01 =l add %a0, %q02
	%q03 =w csltl 51, %x0
	%q04 =l extuw %q03
	%q05 =l mul %q04, 75
	%a02 =l sub %a01, %q05
	%q06 =w csltl 61, %x0
	%q07 =l extuw %q06
	%q08 =l mul %q07, 15
	%a03 =l sub %a02, %q08
	%q09 =w csltl 62, %x0
	%q0a =l extuw %q09
	%q0b =l mul %q0a, 3
	%c0 =l add %a03, %q0b
	storeb %c0, %op

	%a1 =l add %x1, 65
	%q10 =w csltl 25, %x1
	%q11 =l extuw %q10
	%q12 =l mul %q11, 6
	%a11 =l add %a1, %q12
	%q13 =w csltl 51, %x1
	%q14 =l extuw %q13
	%q15 =l mul %q14, 75
	%a12 =l sub %a11, %q15
	%q16 =w csltl 61, %x1
	%q17 =l extuw %q16
	%q18 =l mul %q17, 15
	%a13 =l sub %a12, %q18
	%q19 =w csltl 62, %x1
	%q1a =l extuw %q19
	%q1b =l mul %q1a, 3
	%c1 =l add %a13, %q1b
	%o1 =l add %op, 1
	storeb %c1, %o1

	%a2 =l add %x2, 65
	%q20 =w csltl 25, %x2
	%q21 =l extuw %q20
	%q22 =l mul %q21, 6
	%a21 =l add %a2, %q22
	%q23 =w csltl 51, %x2
	%q24 =l extuw %q23
	%q25 =l mul %q24, 75
	%a22 =l sub %a21, %q25
	%q26 =w csltl 61, %x2
	%q27 =l extuw %q26
	%q28 =l mul %q27, 15
	%a23 =l sub %a22, %q28
	%q29 =w csltl 62, %x2
	%q2a =l extuw %q29
	%q2b =l mul %q2a, 3
	%c2 =l add %a23, %q2b
	%o2 =l add %op, 2
	storeb %c2, %o2

	%a3 =l add %x3, 65
	%q30 =w csltl 25, %x3
	%q31 =l extuw %q30
	%q32 =l mul %q31, 6
	%a31 =l add %a3, %q32
	%q33 =w csltl 51, %x3
	%q34 =l extuw %q33
	%q35 =l mul %q34, 75
	%a32 =l sub %a31, %q35
	%q36 =w csltl 61, %x3
	%q37 =l extuw %q36
	%q38 =l mul %q37, 15
	%a33 =l sub %a32, %q38
	%q39 =w csltl 62, %x3
	%q3a =l extuw %q39
	%q3b =l mul %q3a, 3
	%c3 =l add %a33, %q3b
	%o3 =l add %op, 3
	storeb %c3, %o3

	%inext =l add %i, 1
	%spnext =l add %sp, 3
	%opnext =l add %op, 4
	jmp @loop

@tail
	%hasrem =w cnel %rem, 0
	jnz %hasrem, @partial, @done

@partial
	%t0 =l loadub %sp
	%y0 =l shr %t0, 2
	%y1a =l and %t0, 3
	%y1 =l shl %y1a, 4

	%r00 =l add %y0, 65
	%r01w =w csltl 25, %y0
	%r01 =l extuw %r01w
	%r02 =l mul %r01, 6
	%r03 =l add %r00, %r02
	%r04w =w csltl 51, %y0
	%r04 =l extuw %r04w
	%r05 =l mul %r04, 75
	%r06 =l sub %r03, %r05
	%r07w =w csltl 61, %y0
	%r07 =l extuw %r07w
	%r08 =l mul %r07, 15
	%r09 =l sub %r06, %r08
	%r0aw =w csltl 62, %y0
	%r0a =l extuw %r0aw
	%r0b =l mul %r0a, 3
	%rc0 =l add %r09, %r0b
	storeb %rc0, %op

	%is2 =w ceql %rem, 2
	jnz %is2, @partial2, @partial1

@partial1
	%r10 =l add %y1, 65
	%r11w =w csltl 25, %y1
	%r11 =l extuw %r11w
	%r12 =l mul %r11, 6
	%r13 =l add %r10, %r12
	%r14w =w csltl 51, %y1
	%r14 =l extuw %r14w
	%r15 =l mul %r14, 75
	%r16 =l sub %r13, %r15
	%r17w =w csltl 61, %y1
	%r17 =l extuw %r17w
	%r18 =l mul %r17, 15
	%r19 =l sub %r16, %r18
	%r1aw =w csltl 62, %y1
	%r1a =l extuw %r1aw
	%r1b =l mul %r1a, 3
	%rc1 =l add %r19, %r1b
	%ro1 =l add %op, 1
	storeb %rc1, %ro1
	%ro2 =l add %op, 2
	storeb 61, %ro2
	%ro3 =l add %op, 3
	storeb 61, %ro3
	jmp @done

@partial2
	%tp1 =l add %sp, 1
	%t1 =l loadub %tp1
	%z1a =l shr %t1, 4
	%z1 =l or %y1, %z1a
	%z2a =l and %t1, 15
	%z2 =l shl %z2a, 2

	%s10 =l add %z1, 65
	%s11w =w csltl 25, %z1
	%s11 =l extuw %s11w
	%s12 =l mul %s11, 6
	%s13 =l add %s10, %s12
	%s14w =w csltl 51, %z1
	%s14 =l extuw %s14w
	%s15 =l mul %s14, 75
	%s16 =l sub %s13, %s15
	%s17w =w csltl 61, %z1
	%s17 =l extuw %s17w
	%s18 =l mul %s17, 15
	%s19 =l sub %s16, %s18
	%s1aw =w csltl 62, %z1
	%s1a =l extuw %s1aw
	%s1b =l mul %s1a, 3
	%sc1 =l add %s19, %s1b
	%so1 =l add %op, 1
	storeb %sc1, %so1

	%s20 =l add %z2, 65
	%s21w =w csltl 25, %z2
	%s21 =l extuw %s21w
	%s22 =l mul %s21, 6
	%s23 =l add %s20, %s22
	%s24w =w csltl 51, %z2
	%s24 =l extuw %s24w
	%s25 =l mul %s24, 75
	%s26 =l sub %s23, %s25
	%s27w =w csltl 61, %z2
	%s27 =l extuw %s27w
	%s28 =l mul %s27, 15
	%s29 =l sub %s26, %s28
	%s2aw =w csltl 62, %z2
	%s2a =l extuw %s2aw
	%s2b =l mul %s2a, 3
	%sc2 =l add %s29, %s2b
	%so2 =l add %op, 2
	storeb %sc2, %so2
	%so3 =l add %op, 3
	storeb 61, %so3
	jmp @done

@done
	ret %olen
}
