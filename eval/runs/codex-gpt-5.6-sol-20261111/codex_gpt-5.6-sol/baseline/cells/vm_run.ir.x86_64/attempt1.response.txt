export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pp =l alloc8 8
	%dp =l alloc8 8
	storel 0, %pp
	storel 0, %dp
	jmp @loop

@loop
	%pc =l loadl %pp
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch

@fetch
	%coff =l mul %pc, 8
	%caddr =l add %code, %coff
	%op =l loadl %caddr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @d1

@d1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @d2

@d2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @d3

@d3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @d4

@d4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @d5

@d5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @d6

@d6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @d7

@d7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @d8

@d8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @d9

@d9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @d10

@d10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @d11

@d11
	%is11 =w ceql %op, 11
	jnz %is11, @jmp, @unknown

@push
	%ppc =l loadl %pp
	%vpc =l add %ppc, 1
	%voff =l mul %vpc, 8
	%vaddr =l add %code, %voff
	%v =l loadl %vaddr
	%pd =l loadl %dp
	%pdoff =l mul %pd, 8
	%pdaddr =l add %stack, %pdoff
	storel %v, %pdaddr
	%pd1 =l add %pd, 1
	storel %pd1, %dp
	%ppc2 =l add %ppc, 2
	storel %ppc2, %pp
	jmp @loop

@add
	%ad =l loadl %dp
	%ai =l sub %ad, 2
	%bi =l sub %ad, 1
	%ao =l mul %ai, 8
	%bo =l mul %bi, 8
	%aa =l add %stack, %ao
	%ba =l add %stack, %bo
	%av =l loadl %aa
	%bv =l loadl %ba
	%rvadd =l add %av, %bv
	storel %rvadd, %aa
	storel %bi, %dp
	jmp @next1

@sub
	%sd =l loadl %dp
	%sai =l sub %sd, 2
	%sbi =l sub %sd, 1
	%sao =l mul %sai, 8
	%sbo =l mul %sbi, 8
	%saa =l add %stack, %sao
	%sba =l add %stack, %sbo
	%sav =l loadl %saa
	%sbv =l loadl %sba
	%rvsub =l sub %sav, %sbv
	storel %rvsub, %saa
	storel %sbi, %dp
	jmp @next1

@mul
	%md =l loadl %dp
	%mai =l sub %md, 2
	%mbi =l sub %md, 1
	%mao =l mul %mai, 8
	%mbo =l mul %mbi, 8
	%maa =l add %stack, %mao
	%mba =l add %stack, %mbo
	%mav =l loadl %maa
	%mbv =l loadl %mba
	%rvmul =l mul %mav, %mbv
	storel %rvmul, %maa
	storel %mbi, %dp
	jmp @next1

@dup
	%dd =l loadl %dp
	%di =l sub %dd, 1
	%dio =l mul %di, 8
	%dda =l add %stack, %dio
	%dv =l loadl %dda
	%ddo =l mul %dd, 8
	%ddst =l add %stack, %ddo
	storel %dv, %ddst
	%dd1 =l add %dd, 1
	storel %dd1, %dp
	jmp @next1

@swap
	%swd =l loadl %dp
	%swi0 =l sub %swd, 2
	%swi1 =l sub %swd, 1
	%swo0 =l mul %swi0, 8
	%swo1 =l mul %swi1, 8
	%swa0 =l add %stack, %swo0
	%swa1 =l add %stack, %swo1
	%swv0 =l loadl %swa0
	%swv1 =l loadl %swa1
	storel %swv1, %swa0
	storel %swv0, %swa1
	jmp @next1

@drop
	%drd =l loadl %dp
	%drd1 =l sub %drd, 1
	storel %drd1, %dp
	jmp @next1

@over
	%od =l loadl %dp
	%osi =l sub %od, 2
	%oso =l mul %osi, 8
	%osa =l add %stack, %oso
	%osv =l loadl %osa
	%odo =l mul %od, 8
	%oda =l add %stack, %odo
	storel %osv, %oda
	%od1 =l add %od, 1
	storel %od1, %dp
	jmp @next1

@lt
	%ld =l loadl %dp
	%lai =l sub %ld, 2
	%lbi =l sub %ld, 1
	%lao =l mul %lai, 8
	%lbo =l mul %lbi, 8
	%laa =l add %stack, %lao
	%lba =l add %stack, %lbo
	%lav =l loadl %laa
	%lbv =l loadl %lba
	%lc =w csltl %lav, %lbv
	%lr =l extuw %lc
	storel %lr, %laa
	storel %lbi, %dp
	jmp @next1

@jz
	%jzd =l loadl %dp
	%jzi =l sub %jzd, 1
	storel %jzi, %dp
	%jzo =l mul %jzi, 8
	%jza =l add %stack, %jzo
	%jzv =l loadl %jza
	%jzz =w ceql %jzv, 0
	jnz %jzz, @jztaken, @jznot

@jztaken
	%jzpc =l loadl %pp
	%jztpc =l add %jzpc, 1
	%jzto =l mul %jztpc, 8
	%jzta =l add %code, %jzto
	%jzt =l loadl %jzta
	storel %jzt, %pp
	jmp @loop

@jznot
	%jznpc =l loadl %pp
	%jznpc2 =l add %jznpc, 2
	storel %jznpc2, %pp
	jmp @loop

@jmp
	%jpc =l loadl %pp
	%jtpc =l add %jpc, 1
	%jto =l mul %jtpc, 8
	%jta =l add %code, %jto
	%jt =l loadl %jta
	storel %jt, %pp
	jmp @loop

@unknown
	jmp @next1

@next1
	%npc =l loadl %pp
	%npc1 =l add %npc, 1
	storel %npc1, %pp
	jmp @loop

@finish
	%fd =l loadl %dp
	%empty =w ceql %fd, 0
	jnz %empty, @emptyret, @nonempty

@emptyret
	%er =l copy 0
	jmp @clear

@nonempty
	%fi =l sub %fd, 1
	%fo =l mul %fi, 8
	%fa =l add %stack, %fo
	%nr =l loadl %fa
	jmp @clear

@clear
	%ret =l phi @emptyret %er, @nonempty %nr
	%zi =l copy %fd
	jmp @zloop

@zloop
	%more =w csltl %zi, %cap
	jnz %more, @zbody, @done

@zbody
	%zo =l mul %zi, 8
	%za =l add %stack, %zo
	storel 0, %za
	%zi1 =l add %zi, 1
	jmp @znext

@znext
	%zi =l copy %zi1
	jmp @zloop

@done
	ret %ret
}
