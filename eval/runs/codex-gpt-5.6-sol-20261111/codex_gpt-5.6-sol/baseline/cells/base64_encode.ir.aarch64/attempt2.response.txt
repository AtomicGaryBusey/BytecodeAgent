export function l $base64_encode(l %src, l %n, l %out) {
@start
	%nonzero =w csgtl %n, 0
	jnz %nonzero, @group, @done

@group
	%i =l phi @start 0, @nextgroup %inext
	%o =l phi @start 0, @nextgroup %onext
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%hasb1 =w csgtl %rem, 1
	jnz %hasb1, @loadb1, @zerob1

@loadb1
	%p1 =l add %p0, 1
	%b1loaded =l loadub %p1
	jmp @b1merge

@zerob1
	jmp @b1merge

@b1merge
	%b1 =l phi @loadb1 %b1loaded, @zerob1 0
	%hasb2 =w csgtl %rem, 2
	jnz %hasb2, @loadb2, @zerob2

@loadb2
	%p2 =l add %p0, 2
	%b2loaded =l loadub %p2
	jmp @b2merge

@zerob2
	jmp @b2merge

@b2merge
	%b2 =l phi @loadb2 %b2loaded, @zerob2 0
	%v0 =l shl %b0, 16
	%v1 =l shl %b1, 8
	%v01 =l or %v0, %v1
	%value =l or %v01, %b2
	jmp @emit

@emit
	%j =l phi @b2merge 0, @advance %jnext
	%one =w ceql %rem, 1
	%two =w ceql %rem, 2
	%jge2 =w csgel %j, 2
	%jge3 =w csgel %j, 3
	%padone =w and %one, %jge2
	%padtwo =w and %two, %jge3
	%ispad =w or %padone, %padtwo
	jnz %ispad, @padding, @index

@padding
	%opad0 =l add %out, %o
	%opad =l add %opad0, %j
	storeb 61, %opad
	jmp @advance

@index
	%isj0 =w ceql %j, 0
	jnz %isj0, @idx0, @testj1

@testj1
	%isj1 =w ceql %j, 1
	jnz %isj1, @idx1, @testj2

@testj2
	%isj2 =w ceql %j, 2
	jnz %isj2, @idx2, @idx3

@idx0
	%s0 =l shr %value, 18
	%a0 =l and %s0, 63
	jmp @map

@idx1
	%s1 =l shr %value, 12
	%a1 =l and %s1, 63
	jmp @map

@idx2
	%s2 =l shr %value, 6
	%a2 =l and %s2, 63
	jmp @map

@idx3
	%a3 =l and %value, 63
	jmp @map

@map
	%a =l phi @idx0 %a0, @idx1 %a1, @idx2 %a2, @idx3 %a3
	%cbase =l add %a, 65
	%lt26 =w csltl %a, 26
	%ge26w =w xor %lt26, 1
	%ge26 =l extuw %ge26w
	%adj26 =l mul %ge26, 6
	%c26 =l add %cbase, %adj26
	%lt52 =w csltl %a, 52
	%ge52w =w xor %lt52, 1
	%ge52 =l extuw %ge52w
	%adj52 =l mul %ge52, 75
	%c52 =l sub %c26, %adj52
	%lt62 =w csltl %a, 62
	%ge62w =w xor %lt62, 1
	%ge62 =l extuw %ge62w
	%adj62 =l mul %ge62, 15
	%c62 =l sub %c52, %adj62
	%lt63 =w csltl %a, 63
	%ge63w =w xor %lt63, 1
	%ge63 =l extuw %ge63w
	%adj63 =l mul %ge63, 3
	%c =l add %c62, %adj63
	%op0 =l add %out, %o
	%op =l add %op0, %j
	storeb %c, %op
	jmp @advance

@advance
	%jnext =l add %j, 1
	%morechars =w csltl %jnext, 4
	jnz %morechars, @emit, @nextgroup

@nextgroup
	%inext =l add %i, 3
	%onext =l add %o, 4
	%moregroups =w csltl %inext, %n
	jnz %moregroups, @group, @finish

@finish
	ret %onext

@done
	ret 0
}
