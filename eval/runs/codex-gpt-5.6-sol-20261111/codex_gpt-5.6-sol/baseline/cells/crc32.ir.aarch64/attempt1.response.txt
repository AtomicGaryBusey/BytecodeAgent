export function l $crc32(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%crc =w phi @start -1, @body %c7
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %crc, %b

	%l0 =w and %cx, 1
	%m0 =w sub 0, %l0
	%s0 =w shr %cx, 1
	%a0 =w and %m0, -306674912
	%c0 =w xor %s0, %a0

	%l1 =w and %c0, 1
	%m1 =w sub 0, %l1
	%s1 =w shr %c0, 1
	%a1 =w and %m1, -306674912
	%c1 =w xor %s1, %a1

	%l2 =w and %c1, 1
	%m2 =w sub 0, %l2
	%s2 =w shr %c1, 1
	%a2 =w and %m2, -306674912
	%c2 =w xor %s2, %a2

	%l3 =w and %c2, 1
	%m3 =w sub 0, %l3
	%s3 =w shr %c2, 1
	%a3 =w and %m3, -306674912
	%c3 =w xor %s3, %a3

	%l4 =w and %c3, 1
	%m4 =w sub 0, %l4
	%s4 =w shr %c3, 1
	%a4 =w and %m4, -306674912
	%c4 =w xor %s4, %a4

	%l5 =w and %c4, 1
	%m5 =w sub 0, %l5
	%s5 =w shr %c4, 1
	%a5 =w and %m5, -306674912
	%c5 =w xor %s5, %a5

	%l6 =w and %c5, 1
	%m6 =w sub 0, %l6
	%s6 =w shr %c5, 1
	%a6 =w and %m6, -306674912
	%c6 =w xor %s6, %a6

	%l7 =w and %c6, 1
	%m7 =w sub 0, %l7
	%s7 =w shr %c6, 1
	%a7 =w and %m7, -306674912
	%c7 =w xor %s7, %a7

	%inext =l add %i, 1
	jmp @loop
@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
