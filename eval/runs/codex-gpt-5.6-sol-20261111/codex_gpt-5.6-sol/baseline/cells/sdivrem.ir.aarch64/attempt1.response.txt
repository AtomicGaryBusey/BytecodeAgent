export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bz =w ceql %b, 0
	jnz %bz, @divzero, @checkoverflow
@checkoverflow
	%amin =w ceql %a, 0x8000000000000000
	%bneg1 =w ceql %b, -1
	%overflow =w and %amin, %bneg1
	jnz %overflow, @overflow, @divide
@divzero
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret 0x8000000000000000
@divide
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
