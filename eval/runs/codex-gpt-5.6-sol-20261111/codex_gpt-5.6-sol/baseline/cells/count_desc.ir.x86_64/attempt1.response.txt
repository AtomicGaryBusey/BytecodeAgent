export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init

@zero
	ret 0

@init
	%limit =l sub %n, 1
	jmp @loop

@loop
	%i =l phi @init 0, @body %nexti
	%count =l phi @init 0, @body %nextcount
	%more =w csltl %i, %limit
	jnz %more, @body, @done

@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%q =l add %p, 8
	%current =l loadl %p
	%next =l loadl %q
	%desc =w csltl %next, %current
	%increment =l extuw %desc
	%nextcount =l add %count, %increment
	%nexti =l add %i, 1
	jmp @loop

@done
	ret %count
}
