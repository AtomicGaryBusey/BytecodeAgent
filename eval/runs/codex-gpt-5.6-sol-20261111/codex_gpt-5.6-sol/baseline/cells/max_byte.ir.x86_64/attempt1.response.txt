export function l $max_byte(l %p, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @zero, @loop
@zero
	ret 0
@loop
	%i =l phi @start 0, @body %inext
	%max =l phi @start 0, @body %mnext
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w cugtl %b, %max
	jnz %greater, @set, @keep
@set
	jmp @body
@keep
	jmp @body
@body
	%mnext =l phi @set %b, @keep %max
	%inext =l add %i, 1
	%more =w csltl %inext, %n
	jnz %more, @loop, @done
@done
	ret %mnext
}
