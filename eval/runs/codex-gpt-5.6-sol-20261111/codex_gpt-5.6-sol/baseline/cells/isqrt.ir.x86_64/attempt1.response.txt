export function l $isqrt(l %n) {
@start
	jmp @align
@align
	%bit =l phi @start 4611686018427387904, @down %bit2
	%gt =w cultl %n, %bit
	jnz %gt, @down, @init
@down
	%bit2 =l shr %bit, 2
	jmp @align
@init
	jmp @loop
@loop
	%rem =l phi @init %n, @merge %rem2
	%res =l phi @init 0, @merge %res2
	%b =l phi @init %bit, @merge %b2
	%more =w cnel %b, 0
	jnz %more, @body, @done
@body
	%sum =l add %res, %b
	%less =w cultl %rem, %sum
	jnz %less, @no, @yes
@yes
	%rem_yes =l sub %rem, %sum
	%half_yes =l shr %res, 1
	%res_yes =l add %half_yes, %b
	jmp @merge
@no
	%res_no =l shr %res, 1
	jmp @merge
@merge
	%rem2 =l phi @yes %rem_yes, @no %rem
	%res2 =l phi @yes %res_yes, @no %res_no
	%b2 =l shr %b, 2
	jmp @loop
@done
	ret %res
}
