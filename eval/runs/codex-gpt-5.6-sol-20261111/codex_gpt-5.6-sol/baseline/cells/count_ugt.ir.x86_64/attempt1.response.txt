export function l $count_ugt(l %a, l %n, l %t) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @zero, @loop
@loop
	%i =l phi @start 0, @loop %nexti
	%count =l phi @start 0, @loop %nextcount
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w cugtl %value, %t
	%inc =l extuw %greater
	%nextcount =l add %count, %inc
	%nexti =l add %i, 1
	%more =w csltl %nexti, %n
	jnz %more, @loop, @done
@done
	ret %nextcount
@zero
	ret 0
}
