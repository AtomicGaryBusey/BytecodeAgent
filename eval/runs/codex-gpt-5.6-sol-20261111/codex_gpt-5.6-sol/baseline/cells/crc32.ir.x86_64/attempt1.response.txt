export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @next %in
	%crc =w phi @start -1, @next %crcn
	%more =w csltl %i, %n
	jnz %more, @byte, @done
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %crc, %b
	jmp @bits
@bits
	%j =w phi @byte 0, @join %jn
	%x =w phi @byte %cx, @join %xn
	%again =w csltw %j, 8
	jnz %again, @bit, @next
@bit
	%low =w and %x, 1
	%shifted =w shr %x, 1
	jnz %low, @odd, @even
@odd
	%xo =w xor %shifted, -306674912
	jmp @join
@even
	%xe =w copy %shifted
	jmp @join
@join
	%xn =w phi @odd %xo, @even %xe
	%jn =w add %j, 1
	jmp @bits
@next
	%crcn =w copy %x
	%in =l add %i, 1
	jmp @outer
@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
