export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%islot =l alloc8 8
	%oslot =l alloc8 8
	%rslot =l alloc8 8
	%jslot =l alloc8 8
	%rrslot =l alloc8 8
	%kslot =l alloc8 8
	storel 0, %islot
	storel 0, %oslot
	jmp @main

@main
	%i =l loadl %islot
	%more =w csltl %i, %n
	jnz %more, @measure_init, @done

@measure_init
	storel 1, %rslot
	jmp @measure

@measure
	%r =l loadl %rslot
	%rsmall =w csltl %r, 128
	jnz %rsmall, @measure_bound, @measure_done

@measure_bound
	%i1 =l loadl %islot
	%ir =l add %i1, %r
	%inrange =w csltl %ir, %n
	jnz %inrange, @measure_bytes, @measure_done

@measure_bytes
	%p0 =l add %src, %i1
	%pr =l add %src, %ir
	%b0 =w loadub %p0
	%br =w loadub %pr
	%equal =w ceqw %b0, %br
	jnz %equal, @measure_inc, @measure_done

@measure_inc
	%rn =l add %r, 1
	storel %rn, %rslot
	jmp @measure

@measure_done
	%rf =l loadl %rslot
	%isrepeat =w csgel %rf, 3
	jnz %isrepeat, @repeat_check, @literal_init

@repeat_check
	%o0 =l loadl %oslot
	%o2 =l add %o0, 2
	%repeatfits =w cslel %o2, %cap
	jnz %repeatfits, @repeat_emit, @fail

@repeat_emit
	%rh =l sub 257, %rf
	%ohp =l add %out, %o0
	storeb %rh, %ohp
	%odp =l add %ohp, 1
	%ri =l loadl %islot
	%rip =l add %src, %ri
	%rb =w loadub %rip
	storeb %rb, %odp
	%rin =l add %ri, %rf
	storel %rin, %islot
	storel %o2, %oslot
	jmp @main

@literal_init
	%li =l loadl %islot
	storel %li, %jslot
	jmp @literal_scan

@literal_scan
	%lj =l loadl %jslot
	%ljn =w csltl %lj, %n
	jnz %ljn, @literal_length_check, @literal_ready

@literal_length_check
	%li2 =l loadl %islot
	%llen0 =l sub %lj, %li2
	%lshort =w csltl %llen0, 128
	jnz %lshort, @literal_run_init, @literal_ready

@literal_run_init
	storel 1, %rrslot
	jmp @literal_run

@literal_run
	%rr =l loadl %rrslot
	%rrsmall =w csltl %rr, 128
	jnz %rrsmall, @literal_run_bound, @literal_run_done

@literal_run_bound
	%lj2 =l loadl %jslot
	%ljr =l add %lj2, %rr
	%ljrin =w csltl %ljr, %n
	jnz %ljrin, @literal_run_bytes, @literal_run_done

@literal_run_bytes
	%ljp =l add %src, %lj2
	%ljrp =l add %src, %ljr
	%ljb =w loadub %ljp
	%ljrb =w loadub %ljrp
	%ljeq =w ceqw %ljb, %ljrb
	jnz %ljeq, @literal_run_inc, @literal_run_done

@literal_run_inc
	%rrn =l add %rr, 1
	storel %rrn, %rrslot
	jmp @literal_run

@literal_run_done
	%rrf =l loadl %rrslot
	%rrrepeat =w csgel %rrf, 3
	jnz %rrrepeat, @literal_ready, @literal_advance

@literal_advance
	%lja =l loadl %jslot
	%ljan =l add %lja, 1
	storel %ljan, %jslot
	jmp @literal_scan

@literal_ready
	%lri =l loadl %islot
	%lrj =l loadl %jslot
	%llen =l sub %lrj, %lri
	%need =l add %llen, 1
	%lo =l loadl %oslot
	%lend =l add %lo, %need
	%literalfits =w cslel %lend, %cap
	jnz %literalfits, @literal_header, @fail

@literal_header
	%lh =l sub %llen, 1
	%lhp =l add %out, %lo
	storeb %lh, %lhp
	storel 0, %kslot
	jmp @literal_copy

@literal_copy
	%k =l loadl %kslot
	%kmore =w csltl %k, %llen
	jnz %kmore, @literal_copy_byte, @literal_finish

@literal_copy_byte
	%ci =l add %lri, %k
	%cp =l add %src, %ci
	%cb =w loadub %cp
	%co0 =l add %lo, 1
	%co =l add %co0, %k
	%cop =l add %out, %co
	storeb %cb, %cop
	%kn =l add %k, 1
	storel %kn, %kslot
	jmp @literal_copy

@literal_finish
	storel %lrj, %islot
	storel %lend, %oslot
	jmp @main

@done
	%result =l loadl %oslot
	ret %result

@fail
	ret -1
}
