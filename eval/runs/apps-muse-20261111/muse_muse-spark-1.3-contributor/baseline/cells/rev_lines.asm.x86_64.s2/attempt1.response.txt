.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit_zero
    xor rbx, rbx
    xor rbp, rbp
    xor r13, r13
    lea r14, [rip + inbuf]
    lea r15, [rip + outbuf]
scan_loop:
    cmp rbp, r12
    jae handle_tail
    mov al, byte ptr [r14 + rbp]
    cmp al, 10
    je found_nl
    inc rbp
    jmp scan_loop
found_nl:
    mov rsi, rbp
    sub rsi, rbx
    test rsi, rsi
    jz empty_line
    lea r8, [r15 + r13]
    lea r9, [r14 + rbp]
    dec r9
    mov rcx, rsi
rev_loop:
    mov al, byte ptr [r9]
    mov byte ptr [r8], al
    dec r9
    inc r8
    dec rcx
    jnz rev_loop
    mov byte ptr [r8], 10
    add r13, rsi
    inc r13
    jmp next_line
empty_line:
    mov byte ptr [r15 + r13], 10
    inc r13
next_line:
    inc rbp
    mov rbx, rbp
    jmp scan_loop
handle_tail:
    cmp rbx, r12
    jae write_out
    mov rsi, r12
    sub rsi, rbx
    lea r8, [r15 + r13]
    lea r9, [r14 + r12]
    dec r9
    mov rcx, rsi
rev_tail_loop:
    mov al, byte ptr [r9]
    mov byte ptr [r8], al
    dec r9
    inc r8
    dec rcx
    jnz rev_tail_loop
    mov byte ptr [r8], 10
    add r13, rsi
    inc r13
write_out:
    test r13, r13
    jz exit_zero
    xor rbx, rbx
write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [r15 + rbx]
    mov rdx, r13
    sub rdx, rbx
    syscall
    add rbx, rax
    cmp rbx, r13
    jb write_loop
exit_zero:
    mov rax, 60
    xor edi, edi
    syscall
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
