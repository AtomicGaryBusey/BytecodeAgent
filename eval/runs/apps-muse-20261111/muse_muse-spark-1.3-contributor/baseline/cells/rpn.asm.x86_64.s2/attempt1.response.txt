.text
.globl _start
_start:
  xor r14d, r14d
  lea r15, [rip + input_buf]
read_loop:
  mov rsi, r15
  add rsi, r14
  mov edx, 65536
  sub rdx, r14
  test rdx, rdx
  jz drain_loop
  mov edi, 0
  mov eax, 0
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r14, rax
  jmp read_loop
drain_loop:
  lea rsi, [rip + out_buf]
  mov edx, 32
  mov edi, 0
  mov eax, 0
  syscall
  test rax, rax
  jz read_done
  js read_done
  jmp drain_loop
read_done:
  test r14, r14
  jz exit_ok
  lea rbx, [rip + stack_vals]
  xor r13d, r13d
line_loop:
  cmp r13, r14
  jae exit_ok
  mov r12, r13
find_nl:
  cmp r12, r14
  jae process_line
  mov rsi, r15
  add rsi, r12
  movzx eax, byte ptr [rsi]
  cmp eax, 10
  je process_line
  inc r12
  jmp find_nl
process_line:
  mov r8, r15
  add r8, r13
  mov r9, r15
  add r9, r12
  xor r10d, r10d
token_loop:
  cmp r8, r9
  jae tokens_done
  movzx eax, byte ptr [r8]
  cmp eax, 32
  je skip_sep
  cmp eax, 9
  je skip_sep
  jmp token_start
skip_sep:
  inc r8
  jmp token_loop
token_start:
  mov rsi, r8
  mov rcx, r8
scan_tok:
  cmp rcx, r9
  jae tok_end
  movzx eax, byte ptr [rcx]
  cmp eax, 32
  je tok_end
  cmp eax, 9
  je tok_end
  inc rcx
  jmp scan_tok
tok_end:
  mov r8, rcx
  mov rdx, rcx
  sub rdx, rsi
  cmp rdx, 1
  jne try_literal
  movzx eax, byte ptr [rsi]
  cmp eax, 43
  je do_add
  cmp eax, 45
  je do_sub
  cmp eax, 42
  je do_mul
  cmp eax, 47
  je do_div
  cmp eax, 37
  je do_mod
try_literal:
  mov rdi, rsi
  xor r11d, r11d
  movzx eax, byte ptr [rdi]
  cmp eax, 45
  jne lit_check
  inc rdi
  mov r11d, 1
  cmp rdi, rcx
  jae line_error
lit_check:
  mov rdx, rcx
  sub rdx, rdi
  cmp rdx, 1
  jb line_error
  cmp rdx, 18
  ja line_error
  xor eax, eax
lit_loop:
  cmp rdi, rcx
  jae lit_done
  movzx edx, byte ptr [rdi]
  sub edx, 48
  cmp edx, 9
  ja line_error
  imul rax, rax, 10
  add rax, rdx
  inc rdi
  jmp lit_loop
lit_done:
  test r11d, r11d
  jz lit_push
  neg rax
lit_push:
  cmp r10, 1024
  jae line_error
  mov qword ptr [rbx + r10*8], rax
  inc r10
  jmp token_loop
do_add:
  cmp r10, 2
  jb line_error
  dec r10
  mov rcx, qword ptr [rbx + r10*8]
  dec r10
  mov rax, qword ptr [rbx + r10*8]
  add rax, rcx
  mov qword ptr [rbx + r10*8], rax
  inc r10
  jmp token_loop
do_sub:
  cmp r10, 2
  jb line_error
  dec r10
  mov rcx, qword ptr [rbx + r10*8]
  dec r10
  mov rax, qword ptr [rbx + r10*8]
  sub rax, rcx
  mov qword ptr [rbx + r10*8], rax
  inc r10
  jmp token_loop
do_mul:
  cmp r10, 2
  jb line_error
  dec r10
  mov rcx, qword ptr [rbx + r10*8]
  dec r10
  mov rax, qword ptr [rbx + r10*8]
  imul rax, rcx
  mov qword ptr [rbx + r10*8], rax
  inc r10
  jmp token_loop
do_div:
  cmp r10, 2
  jb line_error
  dec r10
  mov rcx, qword ptr [rbx + r10*8]
  dec r10
  mov rax, qword ptr [rbx + r10*8]
  test rcx, rcx
  jz line_error
  mov edx, 1
  shl rdx, 63
  cmp rax, rdx
  jne div_ok
  cmp rcx, -1
  je line_error
div_ok:
  cqo
  idiv rcx
  mov qword ptr [rbx + r10*8], rax
  inc r10
  jmp token_loop
do_mod:
  cmp r10, 2
  jb line_error
  dec r10
  mov rcx, qword ptr [rbx + r10*8]
  dec r10
  mov rax, qword ptr [rbx + r10*8]
  test rcx, rcx
  jz line_error
  mov edx, 1
  shl rdx, 63
  cmp rax, rdx
  jne mod_ok
  cmp rcx, -1
  je line_error
mod_ok:
  cqo
  idiv rcx
  mov rax, rdx
  mov qword ptr [rbx + r10*8], rax
  inc r10
  jmp token_loop
tokens_done:
  cmp r10, 1
  jne line_error
  mov rax, qword ptr [rbx]
  lea rdi, [rip + out_buf]
  add rdi, 32
  dec rdi
  mov byte ptr [rdi], 10
  mov r10d, 0
  test rax, rax
  jns conv_digits
  mov r10d, 1
  neg rax
conv_digits:
  mov ecx, 10
  test rax, rax
  jnz conv_loop
  dec rdi
  mov byte ptr [rdi], 48
  jmp conv_sign
conv_loop:
  xor edx, edx
  div rcx
  add edx, 48
  dec rdi
  mov byte ptr [rdi], dl
  test rax, rax
  jnz conv_loop
conv_sign:
  test r10d, r10d
  jz conv_write
  dec rdi
  mov byte ptr [rdi], 45
conv_write:
  lea rcx, [rip + out_buf]
  add rcx, 32
  sub rcx, rdi
  mov rsi, rdi
  mov rdx, rcx
  mov edi, 1
  mov eax, 1
  syscall
  jmp line_next
line_error:
  lea rsi, [rip + err_msg]
  mov edx, 6
  mov edi, 1
  mov eax, 1
  syscall
line_next:
  cmp r12, r14
  jae exit_ok
  mov r13, r12
  inc r13
  jmp line_loop
exit_ok:
  mov edi, 0
  mov eax, 60
  syscall
  jmp exit_ok
.section .rodata
err_msg:
.ascii "error"
.byte 10
.bss
input_buf:
.zero 65536
stack_vals:
.zero 8192
out_buf:
.zero 32
