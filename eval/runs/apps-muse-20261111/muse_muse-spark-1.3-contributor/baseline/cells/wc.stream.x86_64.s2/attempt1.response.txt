.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    syscall
    test rax, rax
    jle done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb is_word
    cmp al, 13
    jbe is_ws
    jmp is_word
is_ws:
    xor r15d, r15d
    jmp next_byte
is_word:
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
next_byte:
    dec rcx
    jnz byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call write_uint
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call write_uint
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call write_uint
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
write_uint:
    test rax, rax
    jnz write_uint_nz
    mov byte ptr [rdi], 48
    inc rdi
    ret
write_uint_nz:
    sub rsp, 32
    mov ecx, 10
    lea r8, [rsp + 32]
    mov rsi, r8
write_uint_div:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz write_uint_div
write_uint_copy:
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    inc rsi
    inc rdi
    cmp rsi, r8
    jb write_uint_copy
    add rsp, 32
    ret
.bss
.align 16
buf:
    .zero 4096
outbuf:
    .zero 64
