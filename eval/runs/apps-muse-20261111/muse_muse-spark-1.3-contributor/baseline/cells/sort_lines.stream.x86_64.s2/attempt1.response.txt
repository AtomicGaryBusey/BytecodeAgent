.text
.globl _start
_start:
  xor r12d, r12d # total = 0
read_loop:
  cmp r12d, 32768
  jae drain_read
  xor eax, eax # read
  xor edi, edi # fd 0
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov edx, 32768
  sub edx, r12d
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
drain_read:
  xor eax, eax
  xor edi, edi
  lea rsi, [rip + drain_buf]
  mov edx, 256
  syscall
  test rax, rax
  jz read_done
  js read_done
  jmp drain_read
read_done:
  test r12, r12
  jz exit_zero
  xor ebx, ebx # pos
  xor ebp, ebp # start
  xor r14d, r14d # nlines
parse_loop:
  cmp rbx, r12
  jae parse_end
  lea rax, [rip + input_buf]
  movzx ecx, byte ptr [rax + rbx]
  cmp cl, 10
  jne parse_next
  cmp r14d, 1000
  jae parse_store_skip
  mov eax, ebp
  mov edx, ebx
  sub edx, ebp
  lea rsi, [rip + entries]
  mov rcx, r14
  shl rcx, 3
  add rsi, rcx
  mov dword ptr [rsi], eax
  mov dword ptr [rsi + 4], edx
  inc r14d
parse_store_skip:
  lea rbp, [rbx + 1]
parse_next:
  inc rbx
  jmp parse_loop
parse_end:
  cmp rbp, r12
  jae parse_done
  cmp r14d, 1000
  jae parse_done
  mov eax, ebp
  mov edx, r12d
  sub edx, ebp
  lea rsi, [rip + entries]
  mov rcx, r14
  shl rcx, 3
  add rsi, rcx
  mov dword ptr [rsi], eax
  mov dword ptr [rsi + 4], edx
  inc r14d
parse_done:
  cmp r14d, 1
  jbe output_start
  xor r15d, r15d # i
outer_loop:
  mov eax, r15d
  inc eax
  cmp eax, r14d
  jae sort_done
  mov ebx, r15d # min_idx
  mov ebp, r15d
  inc ebp # j
inner_loop:
  cmp ebp, r14d
  jae inner_done
  lea r8, [rip + entries]
  mov rax, rbp
  shl rax, 3
  mov r9d, dword ptr [r8 + rax]
  mov r10d, dword ptr [r8 + rax + 4]
  mov rax, rbx
  shl rax, 3
  mov esi, dword ptr [r8 + rax]
  mov edi, dword ptr [r8 + rax + 4]
  lea r8, [rip + input_buf]
  lea r9, [r8 + r9]
  lea r11, [r8 + rsi]
  mov eax, r10d
  cmp eax, edi
  jbe min_done
  mov eax, edi
min_done:
  xor ecx, ecx
  test eax, eax
  jz cmp_len_check
cmp_byte_loop:
  movzx edx, byte ptr [r9 + rcx]
  movzx esi, byte ptr [r11 + rcx]
  cmp edx, esi
  jne cmp_diff
  inc ecx
  cmp ecx, eax
  jb cmp_byte_loop
  jmp cmp_len_check
cmp_diff:
  cmp edx, esi
  jb cmp_less
  jmp inner_next
cmp_len_check:
  cmp r10d, edi
  jae inner_next
cmp_less:
  mov ebx, ebp
inner_next:
  inc ebp
  jmp inner_loop
inner_done:
  cmp ebx, r15d
  je outer_next
  lea r8, [rip + entries]
  mov rax, r15
  shl rax, 3
  mov rcx, rbx
  shl rcx, 3
  mov r9, qword ptr [r8 + rax]
  mov r10, qword ptr [r8 + rcx]
  mov qword ptr [r8 + rax], r10
  mov qword ptr [r8 + rcx], r9
outer_next:
  inc r15d
  jmp outer_loop
sort_done:
output_start:
  xor r15d, r15d
out_loop:
  cmp r15d, r14d
  jae exit_zero
  lea r8, [rip + entries]
  mov rax, r15
  shl rax, 3
  mov esi, dword ptr [r8 + rax]
  mov edx, dword ptr [r8 + rax + 4]
  test edx, edx
  jz write_newline
  lea r8, [rip + input_buf]
  add r8, rsi
  mov rsi, r8
  mov eax, 1
  mov edi, 1
  syscall
write_newline:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + newline_byte]
  mov edx, 1
  syscall
  inc r15d
  jmp out_loop
exit_zero:
  mov eax, 60
  xor edi, edi
  syscall
  jmp exit_zero
.section .rodata
newline_byte:
  .byte 10
.bss
  .balign 8
input_buf:
  .zero 32768
entries:
  .zero 8000
drain_buf:
  .zero 256
