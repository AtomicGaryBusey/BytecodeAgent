.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz done_read
    mov rbx, rax
    add r14, rbx
    lea rsi, [rip + buffer]
proc_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb is_word
    cmp al, 13
    jbe is_ws
is_word:
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
is_ws:
    xor r15d, r15d
next_byte:
    dec rbx
    jnz proc_loop
    jmp read_loop
done_read:
    lea rbx, [rip + outbuf]
    mov rax, r12
    call write_num
    mov byte ptr [rbx], 32
    inc rbx
    mov rax, r13
    call write_num
    mov byte ptr [rbx], 32
    inc rbx
    mov rax, r14
    call write_num
    mov byte ptr [rbx], 10
    inc rbx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
write_num:
    test rax, rax
    jnz wn_nz
    mov byte ptr [rbx], 48
    inc rbx
    ret
wn_nz:
    xor r10d, r10d
    mov rcx, 10
wn_div:
    xor edx, edx
    div rcx
    push rdx
    inc r10
    test rax, rax
    jnz wn_div
wn_out:
    pop rax
    add al, 48
    mov byte ptr [rbx], al
    inc rbx
    dec r10
    jnz wn_out
    ret
.bss
buffer:
    .zero 4096
outbuf:
    .zero 64
