.text
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_SIZE, 65537
.globl _start
_start:
    lea rbx, [rip + buf]
    mov r12, 0
1:
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, BUF_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp 1b
read_done:
    test r12, r12
    jz do_exit
    mov r13, 0
    mov r14, 0
scan_loop:
    cmp r14, r12
    jae scan_done
    movzx eax, byte ptr [rbx + r14]
    cmp al, 10
    jne scan_next
    lea r8, [rbx + r13]
    lea r9, [rbx + r14]
    dec r9
    call reverse_range
    mov r13, r14
    inc r13
scan_next:
    inc r14
    jmp scan_loop
scan_done:
    cmp r13, r12
    jae write_setup
    lea r8, [rbx + r13]
    lea r9, [rbx + r12]
    dec r9
    call reverse_range
    mov byte ptr [rbx + r12], 10
    inc r12
write_setup:
    mov r13, 0
write_loop:
    cmp r13, r12
    jae do_exit
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rbx + r13]
    mov rdx, r12
    sub rdx, r13
    syscall
    add r13, rax
    jmp write_loop
do_exit:
    mov rax, SYS_EXIT
    mov rdi, 0
    syscall
reverse_range:
    cmp r8, r9
    jnb reverse_done
reverse_loop:
    mov al, byte ptr [r8]
    mov cl, byte ptr [r9]
    mov byte ptr [r8], cl
    mov byte ptr [r9], al
    inc r8
    dec r9
    cmp r8, r9
    jb reverse_loop
reverse_done:
    ret
.bss
.p2align 4
buf:
    .zero 65537
