.text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ BUFSIZE, 4096
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x23, buf
    add x23, x23, :lo12:buf
read_loop:
    mov x0, #FD_STDIN
    mov x1, x23
    mov x2, #BUFSIZE
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    mov x24, x23
    add x25, x23, x0
process_loop:
    cmp x24, x25
    b.hs read_loop
    ldrb w26, [x24], #1
    cmp w26, #10
    b.ne not_nl
    add x19, x19, #1
not_nl:
    cmp w26, #32
    b.eq is_ws
    cmp w26, #9
    b.lo is_word
    cmp w26, #13
    b.ls is_ws
    b is_word
is_ws:
    mov x22, #0
    b process_loop
is_word:
    cbnz x22, process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop
read_done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    mov x0, x19
    mov x1, x23
    bl format_uint
    mov x23, x0
    mov w0, #32
    strb w0, [x23], #1
    mov x0, x20
    mov x1, x23
    bl format_uint
    mov x23, x0
    mov w0, #32
    strb w0, [x23], #1
    mov x0, x21
    mov x1, x23
    bl format_uint
    mov x23, x0
    mov w0, #10
    strb w0, [x23], #1
    sub x2, x23, x24
    mov x0, #FD_STDOUT
    mov x1, x24
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
hang:
    b hang
format_uint:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    mov x2, #10
    cbnz x0, fmt_convert
    mov w3, #48
    strb w3, [x1], #1
    b fmt_done
fmt_convert:
    mov x4, sp
    mov x5, #0
fmt_div_loop:
    udiv x6, x0, x2
    msub x7, x6, x2, x0
    add x7, x7, #48
    strb w7, [x4, x5]
    add x5, x5, #1
    mov x0, x6
    cbnz x0, fmt_div_loop
fmt_copy_loop:
    sub x5, x5, #1
    ldrb w7, [x4, x5]
    strb w7, [x1], #1
    cbnz x5, fmt_copy_loop
fmt_done:
    mov x0, x1
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret
.bss
.align 4
buf:
    .zero 4096
outbuf:
    .zero 64
