.text
.globl _start
_start:
xor r12d, r12d
lea rbx, [rip + input_buf]
read_loop:
mov rax, 65536
sub rax, r12
test rax, rax
jz done_read
mov rdx, rax
lea rsi, [rbx + r12]
mov rdi, 0
mov rax, 0
syscall
test rax, rax
jz done_read
js done_read
add r12, rax
jmp read_loop
done_read:
test r12, r12
jz exit_zero
lea rbp, [rip + stack_vals]
xor r13d, r13d
line_loop:
cmp r13, r12
jae exit_zero
mov r14, r13
find_nl:
cmp r14, r12
jae no_newline
movzx eax, byte ptr [rbx + r14]
cmp al, 10
je found_newline
inc r14
jmp find_nl
found_newline:
mov r15, r14
inc r15
jmp eval_start
no_newline:
mov r15, r12
eval_start:
xor r8d, r8d
mov r10, r13
skip_sep:
cmp r10, r14
jae end_of_tokens
movzx eax, byte ptr [rbx + r10]
cmp al, 0x20
je adv_p
cmp al, 0x09
je adv_p
jmp tok_begin
adv_p:
inc r10
jmp skip_sep
tok_begin:
mov rsi, r10
scan_tok:
cmp r10, r14
jae tok_end
movzx eax, byte ptr [rbx + r10]
cmp al, 0x20
je tok_end
cmp al, 0x09
je tok_end
inc r10
jmp scan_tok
tok_end:
mov rcx, r10
sub rcx, rsi
cmp rcx, 1
jne parse_literal
movzx eax, byte ptr [rbx + rsi]
cmp al, 0x30
jb chk_op
cmp al, 0x39
jbe single_digit
chk_op:
cmp al, 0x2B
je do_op
cmp al, 0x2D
je do_op
cmp al, 0x2A
je do_op
cmp al, 0x2F
je do_op
cmp al, 0x25
je do_op
jmp out_error
single_digit:
sub al, 0x30
movzx eax, al
cmp r8, 32768
jae out_error
mov qword ptr [rbp + r8*8], rax
inc r8
jmp skip_sep
parse_literal:
mov rdi, rsi
xor r9d, r9d
movzx eax, byte ptr [rbx + rdi]
cmp al, 0x2D
jne lit_check_len
mov r9d, 1
inc rdi
dec rcx
lit_check_len:
test rcx, rcx
jz out_error
cmp rcx, 18
ja out_error
xor eax, eax
lit_loop:
test rcx, rcx
jz lit_done
movzx edx, byte ptr [rbx + rdi]
cmp dl, 0x30
jb out_error
cmp dl, 0x39
ja out_error
sub dl, 0x30
movzx edx, dl
imul rax, rax, 10
add rax, rdx
inc rdi
dec rcx
jmp lit_loop
lit_done:
test r9d, r9d
jz lit_push
neg rax
lit_push:
cmp r8, 32768
jae out_error
mov qword ptr [rbp + r8*8], rax
inc r8
jmp skip_sep
do_op:
mov r9b, al
cmp r8, 2
jb out_error
dec r8
mov rcx, qword ptr [rbp + r8*8]
dec r8
mov rax, qword ptr [rbp + r8*8]
cmp r9b, 0x2B
je op_add
cmp r9b, 0x2D
je op_sub
cmp r9b, 0x2A
je op_mul
cmp r9b, 0x2F
je op_div
jmp op_mod
op_add:
add rax, rcx
jmp push_res
op_sub:
sub rax, rcx
jmp push_res
op_mul:
imul rax, rcx
jmp push_res
op_div:
cmp rcx, 0
je out_error
cmp rcx, -1
jne div_ok
mov rdx, 1
shl rdx, 63
cmp rax, rdx
je out_error
div_ok:
cqo
idiv rcx
jmp push_res
op_mod:
cmp rcx, 0
je out_error
cmp rcx, -1
jne mod_ok
mov rdx, 1
shl rdx, 63
cmp rax, rdx
je out_error
mod_ok:
cqo
idiv rcx
mov rax, rdx
jmp push_res
push_res:
mov qword ptr [rbp + r8*8], rax
inc r8
jmp skip_sep
end_of_tokens:
cmp r8, 1
jne out_error
mov r9, qword ptr [rbp]
mov rax, r9
lea rsi, [rip + outbuf]
lea rdi, [rsi + 32]
dec rdi
mov byte ptr [rdi], 10
xor r10d, r10d
test rax, rax
jns conv_setup
mov r10d, 1
neg rax
conv_setup:
mov rcx, 10
conv_loop:
xor edx, edx
div rcx
add dl, 0x30
dec rdi
mov byte ptr [rdi], dl
test rax, rax
jnz conv_loop
test r10d, r10d
jz conv_done
dec rdi
mov byte ptr [rdi], 0x2D
conv_done:
lea rdx, [rsi + 32]
sub rdx, rdi
mov rsi, rdi
mov rdi, 1
mov rax, 1
syscall
jmp next_line
out_error:
lea rsi, [rip + err_msg]
mov rdx, 6
mov rdi, 1
mov rax, 1
syscall

mov r13, r15
jmp line_loop

mov rdi, 0
mov rax, 60
syscall
