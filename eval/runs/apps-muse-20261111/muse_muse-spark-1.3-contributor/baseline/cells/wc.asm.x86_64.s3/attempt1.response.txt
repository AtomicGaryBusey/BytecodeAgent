.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz done_read
    js done_read
    mov rcx, rax
    lea rsi, [rip + buffer]
next_byte:
    movzx eax, byte ptr [rsi]
    inc rsi
    inc r14
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jb 3f
    cmp al, 13
    ja 3f
2:
    xor r15d, r15d
    jmp 4f
3:
    test r15, r15
    jnz 4f
    inc r13
    mov r15d, 1
4:
    dec rcx
    jnz next_byte
    jmp read_loop
done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
hang:
    jmp hang
print_num:
    test rax, rax
    jnz 5f
    mov byte ptr [rdi], 48
    inc rdi
    ret
5:
    xor ecx, ecx
    mov esi, 10
6:
    xor edx, edx
    div rsi
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz 6b
7:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz 7b
    ret
.bss
.align 16
buffer:
    .space 4096
outbuf:
    .space 64
