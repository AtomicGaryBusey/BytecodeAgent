.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x23, buf
    add x23, x23, :lo12:buf
read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, done_read
    add x21, x21, x0
    mov x24, x23
    add x25, x23, x0
byte_loop:
    cmp x24, x25
    b.hs read_loop
    ldrb w26, [x24], #1
    cmp w26, #10
    b.ne check_space
    add x19, x19, #1
    mov x22, #0
    b byte_loop
check_space:
    cmp w26, #32
    b.eq is_ws
    cmp w26, #9
    b.eq is_ws
    cmp w26, #11
    b.eq is_ws
    cmp w26, #12
    b.eq is_ws
    cmp w26, #13
    b.eq is_ws
    cbnz x22, byte_loop
    add x20, x20, #1
    mov x22, #1
    b byte_loop
is_ws:
    mov x22, #0
    b byte_loop
done_read:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x1, x24
    mov x0, x19
    bl print_num
    mov w2, #32
    strb w2, [x1], #1
    mov x0, x20
    bl print_num
    mov w2, #32
    strb w2, [x1], #1
    mov x0, x21
    bl print_num
    mov w2, #10
    strb w2, [x1], #1
    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
print_num:
    adrp x2, digits
    add x2, x2, :lo12:digits
    mov x3, x2
    cbnz x0, div_loop
    mov w4, #48
    strb w4, [x1], #1
    ret
div_loop:
    mov x4, #10
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add x6, x6, #48
    strb w6, [x3], #1
    mov x0, x5
    cbnz x0, div_loop
copy_loop:
    cmp x3, x2
    b.eq copy_done
    sub x3, x3, #1
    ldrb w7, [x3]
    strb w7, [x1], #1
    b copy_loop
copy_done:
    ret
