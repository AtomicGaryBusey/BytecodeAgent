.text
.globl _start
_start:
  lea rbx, [rip + input_buf]
  xor r12d, r12d
read_loop:
  mov edx, 65536
  sub rdx, r12
  jz read_done
  xor eax, eax
  xor edi, edi
  lea rsi, [rbx + r12]
  syscall
  test rax, rax
  jz read_done
  add r12, rax
  jmp read_loop
read_done:
  lea r9, [rip + hexdigits]
  xor r14d, r14d
dump_loop:
  cmp r14, r12
  jae print_final
  mov rax, r12
  sub rax, r14
  mov r15d, 16
  cmp rax, 16
  jae have_n
  mov r15, rax
have_n:
  lea rdi, [rip + linebuf]
  mov eax, r14d
  mov r8d, 8
off_loop:
  rol eax, 4
  mov edx, eax
  and edx, 15
  movzx ecx, byte ptr [r9 + rdx]
  mov byte ptr [rdi], cl
  inc rdi
  dec r8d
  jnz off_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  add rdi, 2
  lea rsi, [rbx + r14]
  xor r8d, r8d
hex_loop:
  cmp r8d, 8
  jne no_extra
  mov byte ptr [rdi], 32
  inc rdi
no_extra:
  cmp r8d, r15d
  jae hex_blank
  movzx edx, byte ptr [rsi + r8]
  mov ecx, edx
  shr ecx, 4
  movzx eax, byte ptr [r9 + rcx]
  mov byte ptr [rdi], al
  and edx, 15
  movzx eax, byte ptr [r9 + rdx]
  mov byte ptr [rdi + 1], al
  mov byte ptr [rdi + 2], 32
  add rdi, 3
  jmp hex_next
hex_blank:
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  mov byte ptr [rdi + 2], 32
  add rdi, 3
hex_next:
  inc r8d
  cmp r8d, 16
  jb hex_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 124
  add rdi, 2
  xor r8d, r8d
ascii_loop:
  cmp r8d, r15d
  jae ascii_done
  movzx edx, byte ptr [rsi + r8]
  cmp edx, 32
  jb ascii_dot
  cmp edx, 126
  ja ascii_dot
  jmp ascii_store
ascii_dot:
  mov edx, 46
ascii_store:
  mov byte ptr [rdi], dl
  inc rdi
  inc r8d
  jmp ascii_loop
ascii_done:
  mov byte ptr [rdi], 124
  mov byte ptr [rdi + 1], 10
  add rdi, 2
  mov rdx, rdi
  lea rsi, [rip + linebuf]
  sub rdx, rsi
  mov eax, 1
  mov edi, 1
  syscall
  add r14, r15
  jmp dump_loop
print_final:
  lea rdi, [rip + linebuf]
  mov eax, r12d
  mov r8d, 8
final_loop:
  rol eax, 4
  mov edx, eax
  and edx, 15
  movzx ecx, byte ptr [r9 + rdx]
  mov byte ptr [rdi], cl
  inc rdi
  dec r8d
  jnz final_loop
  mov byte ptr [rdi], 10
  inc rdi
  mov rdx, rdi
  lea rsi, [rip + linebuf]
  sub rdx, rsi
  mov eax, 1
  mov edi, 1
  syscall
  mov eax, 60
  xor edi, edi
  syscall
.section .rodata
hexdigits:
  .ascii "0123456789abcdef"
.bss
input_buf:
  .zero 65536
linebuf:
  .zero 128
