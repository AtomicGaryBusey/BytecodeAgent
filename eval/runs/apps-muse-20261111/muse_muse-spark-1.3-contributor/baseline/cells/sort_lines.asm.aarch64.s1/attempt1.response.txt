.text
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
read_loop:
    movz x2, #0x8000
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x22, line_off
    add x22, x22, :lo12:line_off
    adrp x23, line_len
    add x23, x23, :lo12:line_len
    mov x21, #0
    mov x24, #0
    mov x25, #0
parse_loop:
    cmp x25, x20
    b.hs parse_tail
    add x9, x19, x25
    ldrb w10, [x9]
    cmp w10, #10
    b.ne parse_next
    sub x10, x25, x24
    lsl x11, x21, #2
    add x12, x22, x11
    str w24, [x12]
    add x12, x23, x11
    str w10, [x12]
    add x21, x21, #1
    add x24, x25, #1
parse_next:
    add x25, x25, #1
    b parse_loop
parse_tail:
    cmp x24, x20
    b.hs parse_done
    sub x10, x20, x24
    lsl x11, x21, #2
    add x12, x22, x11
    str w24, [x12]
    add x12, x23, x11
    str w10, [x12]
    add x21, x21, #1
parse_done:
    cmp x21, #2
    b.lo sort_done
    mov x24, #1
outer_loop:
    cmp x24, x21
    b.hs sort_done
    lsl x11, x24, #2
    add x12, x22, x11
    ldr w25, [x12]
    add x12, x23, x11
    ldr w26, [x12]
    sub x27, x24, #1
inner_loop:
    cmp x27, #0
    b.lt insert_key
    lsl x11, x27, #2
    add x12, x22, x11
    ldr w9, [x12]
    add x12, x23, x11
    ldr w10, [x12]
    cmp x10, x26
    b.lo cmp_min_j
    mov x11, x26
    b cmp_min_done
cmp_min_j:
    mov x11, x10
cmp_min_done:
    add x12, x19, x9
    add x13, x19, x25
    mov x14, #0
cmp_loop:
    cmp x14, x11
    b.hs cmp_len
    add x17, x12, x14
    ldrb w15, [x17]
    add x17, x13, x14
    ldrb w16, [x17]
    cmp w15, w16
    b.hi shift_line
    b.lo insert_key
    add x14, x14, #1
    b cmp_loop
cmp_len:
    cmp x10, x26
    b.hi shift_line
    b insert_key
shift_line:
    lsl x11, x27, #2
    add x12, x22, x11
    str w9, [x12, #4]
    add x12, x23, x11
    str w10, [x12, #4]
    sub x27, x27, #1
    b inner_loop
insert_key:
    add x11, x27, #1
    lsl x11, x11, #2
    add x12, x22, x11
    str w25, [x12]
    add x12, x23, x11
    str w26, [x12]
    add x24, x24, #1
    b outer_loop
sort_done:
    adrp x28, nl
    add x28, x28, :lo12:nl
    mov x24, #0
out_loop:
    cmp x24, x21
    b.hs out_done
    lsl x11, x24, #2
    add x12, x22, x11
    ldr w9, [x12]
    add x12, x23, x11
    ldr w10, [x12]
    cbz x10, write_nl
    mov x0, #1
    add x1, x19, x9
    mov x2, x10
    mov x8, #SYS_write
    svc #0
write_nl:
    mov x0, #1
    mov x1, x28
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x24, x24, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
.section .rodata
nl:
    .byte 10
.bss
.balign 8
input_buf:
    .space 32768
line_off:
    .space 4000
line_len:
    .space 4000
