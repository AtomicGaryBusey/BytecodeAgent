.text
.globl _start
_start:
lea rbx, [rip + input_buf]
lea r15, [rip + line_buf]
xor r12, r12
read_loop:
mov edx, 65536
sub rdx, r12
jnz read_have_space
xor edi, edi
mov rsi, r15
mov edx, 64
xor eax, eax
syscall
test rax, rax
jz read_done
js read_done
jmp read_loop
read_have_space:
lea rsi, [rbx + r12]
xor edi, edi
xor eax, eax
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_done:
xor r13, r13
row_loop:
cmp r13, r12
jae final_line
mov rcx, r12
sub rcx, r13
cmp rcx, 16
jbe row_n_ok
mov ecx, 16
row_n_ok:
mov r14, rcx
lea rsi, [rbx + r13]
mov rdi, r15
mov r8d, r13d
mov ecx, 8
off_loop:
mov eax, r8d
shr eax, 28
cmp al, 10
jb off_num
add al, 87
jmp off_store
off_num:
add al, 48
off_store:
mov byte ptr [rdi], al
inc rdi
shl r8d, 4
dec ecx
jnz off_loop
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
xor ecx, ecx
hex_loop:
cmp rcx, r14
jae hex_blank
movzx eax, byte ptr [rsi + rcx]
mov edx, eax
shr edx, 4
and eax, 15
cmp dl, 10
jb hex_high_num
add dl, 87
jmp hex_high_store
hex_high_num:
add dl, 48
hex_high_store:
mov byte ptr [rdi], dl
inc rdi
cmp al, 10
jb hex_low_num
add al, 87
jmp hex_low_store
hex_low_num:
add al, 48
hex_low_store:
mov byte ptr [rdi], al
inc rdi
mov byte ptr [rdi], 32
inc rdi
jmp hex_after
hex_blank:
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
hex_after:
cmp rcx, 7
jne hex_no_extra
mov byte ptr [rdi], 32
inc rdi
hex_no_extra:
inc rcx
cmp rcx, 16
jb hex_loop
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 124
inc rdi
xor ecx, ecx
ascii_loop:
cmp rcx, r14
jae ascii_done
movzx eax, byte ptr [rsi + rcx]
cmp al, 32
jb ascii_dot
cmp al, 126
ja ascii_dot
jmp ascii_store
ascii_dot:
mov al, 46
ascii_store:
mov byte ptr [rdi], al
inc rdi
inc rcx
jmp ascii_loop
ascii_done:
mov byte ptr [rdi], 124
inc rdi
mov byte ptr [rdi], 10
inc rdi
mov rdx, rdi
sub rdx, r15
mov rsi, r15
mov edi, 1
mov eax, 1
syscall
add r13, r14
jmp row_loop
final_line:
mov rdi, r15
mov r8d, r12d
mov ecx, 8
final_loop:
mov eax, r8d
shr eax, 28
cmp al, 10
jb final_num
add al, 87
jmp final_store
final_num:
add al, 48
final_store:
mov byte ptr [rdi], al
inc rdi
shl r8d, 4
dec ecx
jnz final_loop
mov byte ptr [rdi], 10
inc rdi
mov rdx, rdi
sub rdx, r15
mov rsi, r15
mov edi, 1
mov eax, 1
syscall
mov eax, 60
xor edi, edi
syscall
.bss
input_buf:
.zero 65536
line_buf:
.zero 128
