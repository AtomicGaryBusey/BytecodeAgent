.globl _start
.text
_start:
  lea rbx, [rip + inbuf]
  lea rbp, [rip + outbuf]
  xor r12d, r12d
read_loop:
  mov eax, 65536
  sub eax, r12d
  jz drain_loop
  mov edx, eax
  mov rsi, rbx
  add rsi, r12
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
drain_loop:
  mov edx, 65537
  mov rsi, rbp
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz read_done
  js read_done
  jmp drain_loop
read_done:
  test r12, r12
  jz exit_ok
  xor r13d, r13d
  xor r14d, r14d
line_loop:
  cmp r13, r12
  jae write_start
  mov r15, r13
scan_loop:
  cmp r15, r12
  jae scan_done
  movzx ecx, byte ptr [rbx + r15]
  cmp cl, 10
  je scan_done
  inc r15
  jmp scan_loop
scan_done:
  mov rcx, r15
  sub rcx, r13
  test rcx, rcx
  jz rev_done
  xor r8d, r8d
rev_loop:
  cmp r8, rcx
  jae rev_done
  mov rax, rcx
  dec rax
  sub rax, r8
  add rax, r13
  movzx edx, byte ptr [rbx + rax]
  mov rax, r14
  add rax, r8
  mov byte ptr [rbp + rax], dl
  inc r8
  jmp rev_loop
rev_done:
  mov rax, r14
  add rax, rcx
  mov byte ptr [rbp + rax], 10
  inc rcx
  add r14, rcx
  mov r13, r15
  cmp r13, r12
  jae line_loop
  inc r13
  jmp line_loop
write_start:
  test r14, r14
  jz exit_ok
  xor r13d, r13d
write_loop:
  cmp r13, r14
  jae exit_ok
  mov rdx, r14
  sub rdx, r13
  mov rsi, rbp
  add rsi, r13
  mov edi, 1
  mov eax, 1
  syscall
  test rax, rax
  jz exit_ok
  js exit_ok
  add r13, rax
  jmp write_loop
exit_ok:
  xor edi, edi
  mov eax, 60
  syscall
halt:
  jmp halt
.bss
.align 16
inbuf:
  .zero 65536
outbuf:
  .zero 65537
