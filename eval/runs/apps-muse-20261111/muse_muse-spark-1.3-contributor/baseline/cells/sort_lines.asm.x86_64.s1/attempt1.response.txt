.text
.globl _start
_start:
 lea r12, [rip + input_buf]
 lea r13, [rip + line_table]
 xor ebx, ebx
read_loop:
 cmp ebx, 32768
 jae read_discard
 lea rsi, [r12 + rbx]
 mov edx, 32768
 sub edx, ebx
 xor edi, edi
 xor eax, eax
 syscall
 test rax, rax
 jz read_done
 js read_done
 add rbx, rax
 jmp read_loop
read_discard:
 lea rsi, [rip + discard_buf]
 mov edx, 1024
 xor edi, edi
 xor eax, eax
 syscall
 test rax, rax
 jz read_done
 js read_done
 jmp read_discard
read_done:
 xor r14d, r14d
 xor r15d, r15d
 xor ecx, ecx
parse_loop:
 cmp ecx, ebx
 jae parse_tail
 mov al, byte ptr [r12 + rcx]
 cmp al, 10
 jne parse_next
 mov eax, ecx
 sub eax, r15d
 mov dword ptr [r13 + r14*8], r15d
 mov dword ptr [r13 + r14*8 + 4], eax
 inc r14d
 mov r15d, ecx
 inc r15d
parse_next:
 inc ecx
 jmp parse_loop
parse_tail:
 cmp r15d, ebx
 jae parse_done
 mov eax, ebx
 sub eax, r15d
 mov dword ptr [r13 + r14*8], r15d
 mov dword ptr [r13 + r14*8 + 4], eax
 inc r14d
parse_done:
 cmp r14d, 1
 jle output_start
 mov r15d, 1
sort_outer:
 cmp r15d, r14d
 jge output_start
 mov r8d, dword ptr [r13 + r15*8]
 mov r9d, dword ptr [r13 + r15*8 + 4]
 mov r10, r15
 dec r10
sort_inner:
 cmp r10, 0
 jl sort_insert
 mov r11d, dword ptr [r13 + r10*8]
 mov edx, dword ptr [r13 + r10*8 + 4]
 lea rsi, [r12 + r11]
 lea rdi, [r12 + r8]
 mov ecx, edx
 cmp ecx, r9d
 jbe sort_cmp
 mov ecx, r9d
sort_cmp:
 test ecx, ecx
 jz sort_cmplen
sort_byteloop:
 mov al, byte ptr [rsi]
 mov r11b, byte ptr [rdi]
 cmp al, r11b
 ja sort_shift
 jb sort_insert
 inc rsi
 inc rdi
 dec ecx
 jnz sort_byteloop
sort_cmplen:
 cmp edx, r9d
 ja sort_shift
 jmp sort_insert
sort_shift:
 mov eax, dword ptr [r13 + r10*8]
 mov edx, dword ptr [r13 + r10*8 + 4]
 mov dword ptr [r13 + r10*8 + 8], eax
 mov dword ptr [r13 + r10*8 + 12], edx
 dec r10
 jmp sort_inner
sort_insert:
 mov dword ptr [r13 + r10*8 + 8], r8d
 mov dword ptr [r13 + r10*8 + 12], r9d
 inc r15d
 jmp sort_outer
output_start:
 xor r15d, r15d
 test r14d, r14d
 jz exit_ok
output_loop:
 cmp r15d, r14d
 jge exit_ok
 mov r8d, dword ptr [r13 + r15*8]
 mov edx, dword ptr [r13 + r15*8 + 4]
 test edx, edx
 jz output_nl
 lea rsi, [r12 + r8]
 mov edi, 1
 mov eax, 1
 syscall
output_nl:
 lea rsi, [rip + newline]
 mov edi, 1
 mov edx, 1
 mov eax, 1
 syscall
 inc r15d
 jmp output_loop
exit_ok:
 mov eax, 60
 xor edi, edi
 syscall
.section .rodata
newline:
 .byte 10
.bss
.align 8
input_buf:
 .zero 32768
line_table:
 .zero 8000
discard_buf:
 .zero 1024
