.globl _start

    mov x22, #0
    mov x23, #0

parse_loop:

    cmp w3, #10
    b.ne parse_next
    sub x3, x22, x23
    lsl x4, x21, #2
    str w23, [x24, x4]
    str w3, [x25, x4]

    add x23, x22, #1

parse_check:

    b.hs sort_check
    sub x3, x19, x23
    lsl x4, x21, #2
    str w23, [x24, x4]
    str w3, [x25, x4]
    add x21, x21, #1

    b.lo write_setup
    mov x22, #1
outer_loop:
    cmp x22, x21
    b.hs write_setup
    lsl x4, x22, #2
    ldr w26, [x24, x4]
    ldr w27, [x25, x4]
    sub x23, x22, #1
inner_loop:
    cmp x23, #0
    b.lt inner_done
    lsl x4, x23, #2
    ldr w3, [x24, x4]
    ldr w4, [x25, x4]
    cmp x4, x27
    csel x7, x4, x27, lo
    add x5, x20, x3
    add x6, x20, x26
    mov x9, #0

    b.hs cmp_len
    ldrb w10, [x5, x9]
    ldrb w11, [x6, x9]
    cmp w10, w11
    b.ne cmp_diff
    add x9, x9, #1
    b cmp_loop
cmp_len:
    cmp x4, x27
    b.eq inner_done
    b.lo inner_done
    b cmp_greater
cmp_diff:

cmp_greater:
    add x12, x23, #1
    lsl x12, x12, #2
    str w3, [x24, x12]
    str w4, [x25, x12]
    sub x23, x23, #1
    b inner_loop

    lsl x4, x4, #2
    str w26, [x24, x4]
    str w27, [x25, x4]
    add x22, x22, #1

    adrp x28, newline
    add x28, x28, :lo12:newline
    mov x22, #0
write_loop:

    b.hs exit_ok
    lsl x4, x22, #2
    ldr w3, [x24, x4]
    ldr w4, [x25, x4]
    cbz x4, write_nl
    mov x0, #1
    add x1, x20, x3
    mov x2, x4
    mov x8, #64

    mov x1, x28
    mov x2, #1
    mov x8, #64
