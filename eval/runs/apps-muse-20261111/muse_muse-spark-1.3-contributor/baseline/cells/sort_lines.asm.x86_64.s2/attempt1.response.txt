.text
.equ BUFFER_SIZE, 32768
.equ MAX_LINES, 32768
.globl _start
_start:
    xor r12d, r12d
    lea rbx, [rip + buffer]
read_loop:
    cmp r12d, BUFFER_SIZE
    jae read_discard
    mov edi, 0
    mov rsi, rbx
    add rsi, r12
    mov edx, BUFFER_SIZE
    sub rdx, r12
    mov eax, 0
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_discard:
    mov edi, 0
    lea rsi, [rip + discard_buf]
    mov edx, 1024
    mov eax, 0
    syscall
    cmp rax, 0
    jle read_done
    jmp read_discard
read_done:
    test r12, r12
    jz exit_zero
    lea r8, [rip + offsets]
    lea r9, [rip + lengths]
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
parse_loop:
    cmp r15, r12
    jae parse_end
    movzx eax, byte ptr [rbx + r15]
    cmp al, 10
    jne parse_next
    cmp r13d, MAX_LINES
    jae parse_found_skip
    mov dword ptr [r8 + r13*4], r14d
    mov eax, r15d
    sub eax, r14d
    mov dword ptr [r9 + r13*4], eax
    inc r13
parse_found_skip:
    mov r14, r15
    inc r14
parse_next:
    inc r15
    jmp parse_loop
parse_end:
    cmp r14, r12
    jae parse_done
    cmp r13d, MAX_LINES
    jae parse_done
    mov dword ptr [r8 + r13*4], r14d
    mov eax, r12d
    sub eax, r14d
    mov dword ptr [r9 + r13*4], eax
    inc r13
parse_done:
    mov r12, r13
    cmp r12, 1
    jle output_start
    lea rbp, [rip + gaps]
gap_loop:
    mov eax, dword ptr [rbp]
    add rbp, 4
    test eax, eax
    jz sort_done
    cmp eax, r12d
    jae gap_loop
    mov r13d, eax
    mov r14, r13
i_loop:
    cmp r14, r12
    jae gap_loop
    mov r10d, dword ptr [r8 + r14*4]
    mov r11d, dword ptr [r9 + r14*4]
    mov r15, r14
j_loop:
    cmp r15, r13
    jb place_case
    mov rax, r15
    sub rax, r13
    mov esi, dword ptr [r8 + rax*4]
    mov edx, dword ptr [r9 + rax*4]
    mov ecx, edx
    cmp ecx, r11d
    jbe min_done
    mov ecx, r11d
min_done:
    test ecx, ecx
    jz len_decide
    add rsi, rbx
    mov rdi, rbx
    add rdi, r10
    xor eax, eax
byte_loop:
    cmp eax, ecx
    jae len_decide
    movzx edx, byte ptr [rsi + rax]
    cmp dl, byte ptr [rdi + rax]
    jne diff_decide
    inc eax
    jmp byte_loop
diff_decide:
    ja shift_case
    jmp place_case
len_decide:
    mov rax, r15
    sub rax, r13
    mov edx, dword ptr [r9 + rax*4]
    cmp edx, r11d
    ja shift_case
    jmp place_case
shift_case:
    mov rax, r15
    sub rax, r13
    mov esi, dword ptr [r8 + rax*4]
    mov dword ptr [r8 + r15*4], esi
    mov esi, dword ptr [r9 + rax*4]
    mov dword ptr [r9 + r15*4], esi
    sub r15, r13
    jmp j_loop
place_case:
    mov dword ptr [r8 + r15*4], r10d
    mov dword ptr [r9 + r15*4], r11d
    inc r14
    jmp i_loop
sort_done:
output_start:
    test r12d, r12d
    jz exit_zero
    xor r14d, r14d
output_loop:
    cmp r14, r12
    jae exit_zero
    mov esi, dword ptr [r8 + r14*4]
    mov edx, dword ptr [r9 + r14*4]
    test edx, edx
    jz output_newline
    mov r10, rbx
    add r10, rsi
    mov r15, rdx
output_data:
    mov edi, 1
    mov rsi, r10
    mov rdx, r15
    mov eax, 1
    syscall
    cmp rax, 0
    jle output_next
    sub r15, rax
    add r10, rax
    test r15, r15
    jnz output_data
output_newline:
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    mov eax, 1
    syscall
output_next:
    inc r14
    jmp output_loop
exit_zero:
    mov edi, 0
    mov eax, 60
    syscall
    mov edi, 0
    mov eax, 231
    syscall
    jmp exit_zero
.section .rodata
.balign 4
gaps:
    .long 20000
    .long 10000
    .long 5000
    .long 2000
    .long 1000
    .long 701
    .long 301
    .long 132
    .long 57
    .long 23
    .long 10
    .long 4
    .long 1
    .long 0
newline_byte:
    .byte 10
.bss
.balign 8
buffer:
    .zero 32768
.balign 4
offsets:
    .zero 131072
.balign 4
lengths:
    .zero 131072
.balign 8
discard_buf:
    .zero 1024
