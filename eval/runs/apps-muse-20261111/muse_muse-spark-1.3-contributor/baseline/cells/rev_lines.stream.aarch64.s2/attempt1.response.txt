.text
.globl _start
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x20, #0
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x22, #0
    mov x23, #0
    mov x24, #0
line_loop:
    cmp x24, x20
    b.hs lines_done
    add x0, x19, x24
    ldrb w0, [x0]
    cmp w0, #10
    b.ne line_next
    sub x25, x24, x23
    bl rev_copy
    add x23, x24, #1
line_next:
    add x24, x24, #1
    b line_loop
lines_done:
    cmp x23, x20
    b.hs write_phase
    sub x25, x20, x23
    bl rev_copy
write_phase:
    cbz x22, exit_ok
    mov x26, #0
write_loop:
    cmp x26, x22
    b.hs exit_ok
    mov x0, #1
    add x1, x21, x26
    sub x2, x22, x26
    mov x8, #SYS_write
    svc #0
    add x26, x26, x0
    b write_loop
exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
rev_copy:
    add x0, x19, x23
    add x1, x21, x22
    mov x2, x25
    add x0, x0, x2
    sub x0, x0, #1
rev_loop:
    cbz x2, rev_store_nl
    ldrb w3, [x0]
    strb w3, [x1]
    sub x0, x0, #1
    add x1, x1, #1
    sub x2, x2, #1
    b rev_loop
rev_store_nl:
    mov w3, #10
    strb w3, [x1]
    add x22, x22, x25
    add x22, x22, #1
    ret
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
