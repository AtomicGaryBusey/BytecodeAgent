.globl _start
.text
_start:
xor r12d, r12d
lea r13, [rip + input_buf]
lea r14, [rip + output_buf]
read_loop:
mov eax, 65536
sub eax, r12d
jz read_done
mov edx, eax
mov rsi, r13
add rsi, r12
xor edi, edi
xor eax, eax
syscall
test rax, rax
jle read_done
add r12, rax
jmp read_loop
read_done:
test r12, r12
jz exit_zero
mov r15, r14
mov rbx, r13
mov rsi, r13
mov rbp, r13
add rbp, r12
scan_loop:
cmp rsi, rbp
jae handle_tail
movzx eax, byte ptr [rsi]
cmp al, 10
je found_nl
inc rsi
jmp scan_loop
found_nl:
cmp rbx, rsi
jae emit_nl
mov rdx, rsi
dec rdx
rev_copy:
movzx eax, byte ptr [rdx]
mov byte ptr [r15], al
inc r15
cmp rdx, rbx
je emit_nl
dec rdx
jmp rev_copy
emit_nl:
mov byte ptr [r15], 10
inc r15
inc rsi
mov rbx, rsi
jmp scan_loop
handle_tail:
cmp rbx, rbp
jae write_out
mov rdx, rbp
dec rdx
tail_copy:
movzx eax, byte ptr [rdx]
mov byte ptr [r15], al
inc r15
cmp rdx, rbx
je tail_nl
dec rdx
jmp tail_copy
tail_nl:
mov byte ptr [r15], 10
inc r15
write_out:
sub r15, r14
jz exit_zero
mov rsi, r14
write_loop:
test r15, r15
jz exit_zero
mov rdx, r15
mov edi, 1
mov eax, 1
syscall
test rax, rax
jle exit_zero
add rsi, rax
sub r15, rax
jmp write_loop
exit_zero:
mov eax, 60
xor edi, edi
syscall
.bss
input_buf:
.zero 65536
output_buf:
.zero 65537
