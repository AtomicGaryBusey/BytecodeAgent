.text

    add x22, x22, :lo12:line_buf

    mov x8, #63

    cbz x0, read_done

    b.hs rows_done
    sub x0, x19, x21
    mov x1, #16
    cmp x0, #16
    csel x24, x0, x1, lo
    mov x25, x22
    mov x26, x21
    mov x27, #28

    and x0, x0, #15
    add x1, x23, x0
    ldrb w1, [x1]
    strb w1, [x25], #1
    cbz x27, off_done
    sub x27, x27, #4

    strb w0, [x25], #1
    strb w0, [x25], #1
    mov x26, #0
hex_loop:

    add x0, x0, x26

    ldrb w1, [x1]
    strb w1, [x25], #1
    and w1, w0, #15
    add x1, x23, x1
    ldrb w1, [x1]
    strb w1, [x25], #1
    mov w1, #32
    strb w1, [x25], #1

hex_blank:

    strb w0, [x25], #1
    strb w0, [x25], #1
    strb w0, [x25], #1

    mov w0, #32

    mov w0, #124

    add x0, x20, x21
    add x0, x0, x26
    ldrb w0, [x0]
    cmp w0, #32

    cmp w0, #126

    add x26, x26, #1

ascii_done:

    strb w0, [x25], #1
    mov w0, #10
    strb w0, [x25], #1
    sub x2, x25, x22

    and x0, x0, #15

    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
exit_hang:
