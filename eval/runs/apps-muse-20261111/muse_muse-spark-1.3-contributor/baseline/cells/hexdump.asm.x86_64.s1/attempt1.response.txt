.section .rodata
hexdigits:
.ascii "0123456789abcdef"
.bss
input_buf:
.zero 65536
line_buf:
.zero 128
.text
.globl _start
_start:
xor ebx, ebx
lea r10, [rip + hexdigits]
read_loop:
cmp ebx, 65536
jae read_done
lea rsi, [rip + input_buf]
add rsi, rbx
mov edx, 65536
sub rdx, rbx
mov eax, 0
mov edi, 0
syscall
test rax, rax
jle read_done
add rbx, rax
jmp read_loop
read_done:
xor r12d, r12d
row_loop:
cmp r12, rbx
jae final_line
mov r15, rbx
sub r15, r12
cmp r15, 16
jbe n_ok
mov r15d, 16
n_ok:
lea r9, [rip + input_buf]
add r9, r12
lea rsi, [rip + line_buf]
mov eax, r12d
mov ecx, 28
off_loop:
mov edx, eax
shr edx, cl
and edx, 15
mov dl, byte ptr [r10 + rdx]
mov byte ptr [rsi], dl
inc rsi
sub ecx, 4
jge off_loop
mov byte ptr [rsi], 32
mov byte ptr [rsi + 1], 32
add rsi, 2
xor ecx, ecx
hex_loop:
cmp ecx, r15d
jae hex_blank
movzx edx, byte ptr [r9 + rcx]
mov eax, edx
shr eax, 4
mov al, byte ptr [r10 + rax]
mov byte ptr [rsi], al
and edx, 15
mov dl, byte ptr [r10 + rdx]
mov byte ptr [rsi + 1], dl
mov byte ptr [rsi + 2], 32
add rsi, 3
jmp hex_next
hex_blank:
mov byte ptr [rsi], 32
mov byte ptr [rsi + 1], 32
mov byte ptr [rsi + 2], 32
add rsi, 3
hex_next:
cmp ecx, 7
jne hex_skip
mov byte ptr [rsi], 32
inc rsi
hex_skip:
inc ecx
cmp ecx, 16
jb hex_loop
mov byte ptr [rsi], 32
inc rsi
mov byte ptr [rsi], 124
inc rsi
xor ecx, ecx
asc_loop:
cmp ecx, r15d
jae asc_done
movzx edx, byte ptr [r9 + rcx]
cmp dl, 32
jb asc_dot
cmp dl, 126
ja asc_dot
mov byte ptr [rsi], dl
jmp asc_next
asc_dot:
mov byte ptr [rsi], 46
asc_next:
inc rsi
inc ecx
jmp asc_loop
asc_done:
mov byte ptr [rsi], 124
inc rsi
mov byte ptr [rsi], 10
inc rsi
mov rdx, rsi
lea rsi, [rip + line_buf]
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
add r12, r15
jmp row_loop
final_line:
lea rsi, [rip + line_buf]
mov eax, ebx
mov ecx, 28
fin_loop:
mov edx, eax
shr edx, cl
and edx, 15
mov dl, byte ptr [r10 + rdx]
mov byte ptr [rsi], dl
inc rsi
sub ecx, 4
jge fin_loop
mov byte ptr [rsi], 10
inc rsi
mov rdx, rsi
lea rsi, [rip + line_buf]
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
mov eax, 60
xor edi, edi
syscall
