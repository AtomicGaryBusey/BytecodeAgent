.text
.globl _start
_start:
  xor r12d, r12d
read_loop:
  mov eax, 65536
  sub eax, r12d
  jz dump_init
  mov edx, eax
  lea rsi, [rip + input_buf]
  add rsi, r12
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz dump_init
  add r12, rax
  jmp read_loop
dump_init:
  xor r13d, r13d
dump_loop:
  cmp r13, r12
  jae print_final
  mov r14, r12
  sub r14, r13
  cmp r14, 16
  jbe n_ok
  mov r14d, 16
n_ok:
  lea rdi, [rip + linebuf]
  lea rbx, [rip + hexdigits]
  lea rsi, [rip + input_buf]
  add rsi, r13
  mov eax, r13d
  mov r8d, 8
off_loop:
  rol eax, 4
  mov ecx, eax
  and ecx, 15
  mov dl, byte ptr [rbx + rcx]
  mov byte ptr [rdi], dl
  inc rdi
  dec r8d
  jnz off_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  add rdi, 2
  xor r8d, r8d
hex_loop:
  cmp r8d, r14d
  jae hex_blank
  movzx ecx, byte ptr [rsi + r8]
  mov edx, ecx
  shr edx, 4
  mov dl, byte ptr [rbx + rdx]
  mov byte ptr [rdi], dl
  and ecx, 15
  mov cl, byte ptr [rbx + rcx]
  mov byte ptr [rdi + 1], cl
  mov byte ptr [rdi + 2], 32
  add rdi, 3
  jmp hex_after
hex_blank:
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  mov byte ptr [rdi + 2], 32
  add rdi, 3
hex_after:
  cmp r8d, 7
  jne no_extra
  mov byte ptr [rdi], 32
  inc rdi
no_extra:
  inc r8d
  cmp r8d, 16
  jb hex_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 124
  add rdi, 2
  xor r8d, r8d
ascii_loop:
  cmp r8d, r14d
  jae ascii_done
  movzx ecx, byte ptr [rsi + r8]
  cmp ecx, 32
  jb ascii_dot
  cmp ecx, 126
  ja ascii_dot
  jmp ascii_store
ascii_dot:
  mov ecx, 46
ascii_store:
  mov byte ptr [rdi], cl
  inc rdi
  inc r8d
  jmp ascii_loop
ascii_done:
  mov byte ptr [rdi], 124
  mov byte ptr [rdi + 1], 10
  add rdi, 2
  mov rdx, rdi
  lea rsi, [rip + linebuf]
  sub rdx, rsi
  mov edi, 1
  mov eax, 1
  syscall
  add r13, r14
  jmp dump_loop
print_final:
  lea rdi, [rip + linebuf]
  lea rbx, [rip + hexdigits]
  mov eax, r12d
  mov r8d, 8
final_loop:
  rol eax, 4
  mov ecx, eax
  and ecx, 15
  mov dl, byte ptr [rbx + rcx]
  mov byte ptr [rdi], dl
  inc rdi
  dec r8d
  jnz final_loop
  mov byte ptr [rdi], 10
  inc rdi
  mov edx, 9
  lea rsi, [rip + linebuf]
  mov edi, 1
  mov eax, 1
  syscall
  xor edi, edi
  mov eax, 60
  syscall
.section .rodata
hexdigits:
  .ascii "0123456789abcdef"
.bss
linebuf:
  .zero 80
input_buf:
  .zero 65536
