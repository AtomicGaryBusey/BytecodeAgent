.text
.equ INPUT_MAX, 65536
.equ STACK_CAP, 1024
.equ OUT_SIZE, 32
.globl _start
_start:
xor r12d, r12d
read_loop:
cmp r12, INPUT_MAX
jae read_done
lea rsi, [rip + input_buf]
add rsi, r12
mov edx, INPUT_MAX
sub rdx, r12
mov edi, 0
mov eax, 0
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_done:
lea r13, [rip + input_buf]
xor r14d, r14d
line_loop:
cmp r14, r12
jae do_exit
mov rsi, r13
add rsi, r14
mov rdx, r13
add rdx, r12
mov rdi, rsi
scan_nl:
cmp rdi, rdx
jae scan_done
cmp byte ptr [rdi], 10
je scan_done
inc rdi
jmp scan_nl
scan_done:
mov r15, rdi
sub r15, r13
cmp rdi, rdx
jae after_nl
inc r15
after_nl:
lea r10, [rip + stack_vals]
xor r8d, r8d
eval_skip:
cmp rsi, rdi
jae eval_end
movzx eax, byte ptr [rsi]
cmp al, 32
je eval_skip_inc
cmp al, 9
je eval_skip_inc
jmp eval_token
eval_skip_inc:
inc rsi
jmp eval_skip
eval_token:
mov rcx, rsi
find_end:
cmp rsi, rdi
jae have_token
movzx eax, byte ptr [rsi]
cmp al, 32
je have_token
cmp al, 9
je have_token
inc rsi
jmp find_end
have_token:
mov rdx, rsi
sub rdx, rcx
cmp rdx, 1
jne parse_num
movzx eax, byte ptr [rcx]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47
je do_operator
cmp al, 37
je do_operator
jmp parse_num
parse_num:
mov r9, rcx
mov r11, rdx
movzx eax, byte ptr [r9]
xor r14d, r14d
cmp al, 45
jne num_digits_check
mov r14d, 1
inc r9
dec r11
num_digits_check:
cmp r11, 1
jb line_error
cmp r11, 18
ja line_error
xor ebx, ebx
num_loop:
movzx eax, byte ptr [r9]
sub al, 48
cmp al, 9
ja line_error
movzx eax, al
imul rbx, rbx, 10
add rbx, rax
inc r9
dec r11
jnz num_loop
test r14d, r14d
jz num_push
neg rbx
num_push:
cmp r8, STACK_CAP
jae line_error
mov qword ptr [r10 + r8*8], rbx
inc r8
jmp eval_skip
do_operator:
mov r11b, al
cmp r8, 2
jb line_error
dec r8
mov rbp, qword ptr [r10 + r8*8]
dec r8
mov rax, qword ptr [r10 + r8*8]
cmp r11b, 43
je op_add
cmp r11b, 45
je op_sub
cmp r11b, 42
je op_mul
cmp r11b, 47
je op_div
jmp op_mod
op_add:
add rax, rbp
mov qword ptr [r10 + r8*8], rax
inc r8
jmp eval_skip
op_sub:
sub rax, rbp
mov qword ptr [r10 + r8*8], rax
inc r8
jmp eval_skip
op_mul:
imul rax, rbp
mov qword ptr [r10 + r8*8], rax
inc r8
jmp eval_skip
op_div:
test rbp, rbp
jz line_error
cmp rbp, -1
jne div_ok
mov edx, 1
shl rdx, 63
cmp rax, rdx
je line_error
div_ok:
cqo
idiv rbp
mov qword ptr [r10 + r8*8], rax
inc r8
jmp eval_skip
op_mod:
test rbp, rbp
jz line_error
cmp rbp, -1
jne mod_ok
mov edx, 1
shl rdx, 63
cmp rax, rdx
je line_error
mod_ok:
cqo
idiv rbp
mov qword ptr [r10 + r8*8], rdx
inc r8
jmp eval_skip
eval_end:
cmp r8, 1
jne line_error
mov rax, qword ptr [r10]
lea r9, [rip + out_buf]
mov rsi, r9
add rsi, OUT_SIZE
dec rsi
mov byte ptr [rsi], 10
test rax, rax
jns conv_pos
neg rax
mov r11d, 1
jmp conv_digits
conv_pos:
mov r11d, 0
conv_digits:
mov r10d, 10
test rax, rax
jnz conv_loop
dec rsi
mov byte ptr [rsi], 48
jmp conv_sign
conv_loop:
xor edx, edx
div r10
add dl, 48
dec rsi
mov byte ptr [rsi], dl
test rax, rax
jnz conv_loop
conv_sign:
test r11d, r11d
jz conv_write
dec rsi
mov byte ptr [rsi], 45
conv_write:
mov rdx, r9
add rdx, OUT_SIZE
sub rdx, rsi
mov edi, 1
mov eax, 1
syscall
mov r14, r15
jmp line_loop
line_error:
lea rsi, [rip + err_msg]
mov edx, 6
mov edi, 1
mov eax, 1
syscall
mov r14, r15
jmp line_loop
do_exit:
mov edi, 0
mov eax, 60
syscall
jmp do_exit
.section .rodata
err_msg:
.ascii "error"
.byte 10
.bss
.balign 8
input_buf:
.zero 65536
.balign 8
stack_vals:
.zero 8192
.balign 8
out_buf:
.zero 32
