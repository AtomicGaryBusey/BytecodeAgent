.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    lea r15, [rip + buffer]
read_loop:
    mov eax, 65536
    sub eax, r12d
    jz read_done
    mov rsi, r15
    add rsi, r12
    mov edx, eax
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
row_loop:
    mov rax, r12
    sub rax, r13
    jz final_offset
    mov r14, rax
    cmp r14, 16
    jbe row_have_n
    mov r14, 16
row_have_n:
    lea rbp, [rip + linebuf]
    lea r10, [rip + hexdigits]
    mov eax, r13d
    mov ecx, 28
off_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rbp], dl
    inc rbp
    sub ecx, 4
    jns off_loop
    mov byte ptr [rbp], 32
    mov byte ptr [rbp + 1], 32
    add rbp, 2
    xor ecx, ecx
hex_loop:
    cmp rcx, r14
    jae hex_blank
    mov rax, r13
    add rax, rcx
    movzx edx, byte ptr [r15 + rax]
    mov r8d, edx
    shr edx, 4
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rbp], dl
    mov edx, r8d
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rbp + 1], dl
    mov byte ptr [rbp + 2], 32
    jmp hex_store_done
hex_blank:
    mov byte ptr [rbp], 32
    mov byte ptr [rbp + 1], 32
    mov byte ptr [rbp + 2], 32
hex_store_done:
    add rbp, 3
    cmp ecx, 7
    jne hex_next
    mov byte ptr [rbp], 32
    inc rbp
hex_next:
    inc rcx
    cmp ecx, 16
    jb hex_loop
    mov byte ptr [rbp], 32
    mov byte ptr [rbp + 1], 124
    add rbp, 2
    xor ecx, ecx
ascii_loop:
    cmp rcx, r14
    jae ascii_done
    mov rax, r13
    add rax, rcx
    movzx edx, byte ptr [r15 + rax]
    cmp dl, 32
    jb ascii_dot
    cmp dl, 126
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov dl, 46
ascii_store:
    mov byte ptr [rbp], dl
    inc rbp
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rbp], 124
    mov byte ptr [rbp + 1], 10
    add rbp, 2
    lea rsi, [rip + linebuf]
    mov rdx, rbp
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add r13, r14
    jmp row_loop
final_offset:
    lea rbp, [rip + linebuf]
    lea r10, [rip + hexdigits]
    mov eax, r12d
    mov ecx, 28
final_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rbp], dl
    inc rbp
    sub ecx, 4
    jns final_loop
    mov byte ptr [rbp], 10
    inc rbp
    lea rsi, [rip + linebuf]
    mov rdx, rbp
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
buffer:
    .zero 65536
linebuf:
    .zero 128
