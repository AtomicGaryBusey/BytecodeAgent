.text
.globl _start
_start:
  lea r12, [rip + input_buf]
  xor ebx, ebx
read_loop:
  mov eax, 32768
  sub eax, ebx
  jz read_done
  mov edx, eax
  mov rsi, r12
  add rsi, rbx
  mov edi, 0
  xor eax, eax
  syscall
  test rax, rax
  jle read_done
  add rbx, rax
  jmp read_loop
read_done:
  test rbx, rbx
  jz exit_zero
  lea r13, [rip + offs]
  lea r14, [rip + lens]
  xor r8d, r8d
  xor r9d, r9d
  xor r10d, r10d
parse_loop:
  cmp r10, rbx
  jae parse_end
  mov al, byte ptr [r12 + r10]
  cmp al, 10
  jne parse_next
  mov dword ptr [r13 + r8*4], r9d
  mov eax, r10d
  sub eax, r9d
  mov dword ptr [r14 + r8*4], eax
  inc r8
  mov r9, r10
  inc r9
parse_next:
  inc r10
  jmp parse_loop
parse_end:
  cmp r9, rbx
  jae parse_done
  mov dword ptr [r13 + r8*4], r9d
  mov eax, ebx
  sub eax, r9d
  mov dword ptr [r14 + r8*4], eax
  inc r8
parse_done:
  mov r15, r8
  test r15, r15
  jz exit_zero
  cmp r15, 1
  jle output_phase
  mov ebx, 1
sort_outer:
  cmp rbx, r15
  jge sort_done
  mov r8d, dword ptr [r13 + rbx*4]
  mov r9d, dword ptr [r14 + rbx*4]
  mov rbp, rbx
  dec rbp
sort_inner:
  cmp rbp, 0
  jl sort_insert
  mov eax, dword ptr [r13 + rbp*4]
  mov ecx, dword ptr [r14 + rbp*4]
  mov rsi, r12
  add rsi, rax
  mov rdi, r12
  add rdi, r8
  mov r10d, ecx
  cmp r10d, r9d
  cmova r10d, r9d
  xor r11d, r11d
cmp_loop:
  cmp r11, r10
  jae cmp_len
  mov al, byte ptr [rsi + r11]
  mov dl, byte ptr [rdi + r11]
  cmp al, dl
  jne cmp_diff
  inc r11
  jmp cmp_loop
cmp_diff:
  ja sort_shift
  jmp sort_insert
cmp_len:
  cmp ecx, r9d
  ja sort_shift
  jmp sort_insert
sort_shift:
  mov eax, dword ptr [r13 + rbp*4]
  mov dword ptr [r13 + rbp*4 + 4], eax
  mov eax, dword ptr [r14 + rbp*4]
  mov dword ptr [r14 + rbp*4 + 4], eax
  dec rbp
  jmp sort_inner
sort_insert:
  mov rax, rbp
  inc rax
  mov dword ptr [r13 + rax*4], r8d
  mov dword ptr [r14 + rax*4], r9d
  inc rbx
  jmp sort_outer
sort_done:
output_phase:
  lea rbx, [rip + output_buf]
  xor ebp, ebp
  xor r8d, r8d
out_loop:
  cmp r8, r15
  jge out_done
  mov eax, dword ptr [r13 + r8*4]
  mov ecx, dword ptr [r14 + r8*4]
  mov rsi, r12
  add rsi, rax
  mov rdi, rbx
  add rdi, rbp
  mov r10, rcx
  test r10, r10
  jz store_nl
  xor r11d, r11d
copy_loop:
  mov al, byte ptr [rsi + r11]
  mov byte ptr [rdi + r11], al
  inc r11
  cmp r11, r10
  jb copy_loop
store_nl:
  add rdi, r10
  mov byte ptr [rdi], 10
  add rbp, r10
  inc rbp
  inc r8
  jmp out_loop
out_done:
  mov eax, 1
  mov edi, 1
  mov rsi, rbx
  mov rdx, rbp
  syscall
exit_zero:
  mov eax, 60
  xor edi, edi
  syscall
.bss
.align 8
input_buf:
  .zero 32768
.align 8
output_buf:
  .zero 32784
.align 8
offs:
  .zero 4000
.align 8
lens:
  .zero 4000
