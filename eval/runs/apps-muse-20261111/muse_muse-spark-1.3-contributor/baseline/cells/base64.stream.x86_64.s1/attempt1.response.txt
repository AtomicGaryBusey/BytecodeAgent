.text
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ INPUT_MAX, 49152
.equ LINE_WIDTH, 76
.equ NEWLINE, 10
.equ PAD, 61
.globl _start
_start:
  lea r13, [rip + input_buf]
  xor r12d, r12d
read_loop:
  cmp r12, INPUT_MAX
  jae read_done
  lea rsi, [r13 + r12]
  mov rdx, INPUT_MAX
  sub rdx, r12
  mov eax, SYS_READ
  mov edi, STDIN
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_done:
  test r12, r12
  jz exit_zero
  mov rsi, r13
  mov rbx, r12
  lea rdi, [rip + output_buf]
  lea r9, [rip + alphabet]
  xor r8d, r8d
encode_loop:
  cmp rbx, 3
  jb encode_remainder
  movzx eax, byte ptr [rsi]
  movzx ecx, byte ptr [rsi + 1]
  movzx edx, byte ptr [rsi + 2]
  mov r10d, eax
  shr r10d, 2
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne emit0_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
emit0_done:
  mov r10d, eax
  and r10d, 3
  shl r10d, 4
  mov r11d, ecx
  shr r11d, 4
  or r10d, r11d
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne emit1_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
emit1_done:
  mov r10d, ecx
  and r10d, 15
  shl r10d, 2
  mov r11d, edx
  shr r11d, 6
  or r10d, r11d
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne emit2_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
emit2_done:
  mov r10d, edx
  and r10d, 63
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne emit3_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
emit3_done:
  add rsi, 3
  sub rbx, 3
  jmp encode_loop
encode_remainder:
  cmp rbx, 0
  je encode_finish
  cmp rbx, 1
  je encode_one
  movzx eax, byte ptr [rsi]
  movzx ecx, byte ptr [rsi + 1]
  mov r10d, eax
  shr r10d, 2
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem2_0_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem2_0_done:
  mov r10d, eax
  and r10d, 3
  shl r10d, 4
  mov r11d, ecx
  shr r11d, 4
  or r10d, r11d
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem2_1_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem2_1_done:
  mov r10d, ecx
  and r10d, 15
  shl r10d, 2
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem2_2_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem2_2_done:
  mov byte ptr [rdi], PAD
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem2_3_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem2_3_done:
  jmp encode_finish
encode_one:
  movzx eax, byte ptr [rsi]
  mov r10d, eax
  shr r10d, 2
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem1_0_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem1_0_done:
  mov r10d, eax
  and r10d, 3
  shl r10d, 4
  movzx r11d, byte ptr [r9 + r10]
  mov byte ptr [rdi], r11b
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem1_1_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem1_1_done:
  mov byte ptr [rdi], PAD
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem1_2_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem1_2_done:
  mov byte ptr [rdi], PAD
  add rdi, 1
  add r8d, 1
  cmp r8d, LINE_WIDTH
  jne rem1_3_done
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
  xor r8d, r8d
rem1_3_done:
encode_finish:
  test r8d, r8d
  jz no_final_newline
  mov byte ptr [rdi], NEWLINE
  add rdi, 1
no_final_newline:
  lea rsi, [rip + output_buf]
  mov rdx, rdi
  sub rdx, rsi
  mov r13, rsi
  mov r12, rdx
write_loop:
  test r12, r12
  jz exit_zero
  mov eax, SYS_WRITE
  mov edi, STDOUT
  mov rsi, r13
  mov rdx, r12
  syscall
  test rax, rax
  jle exit_zero
  add r13, rax
  sub r12, rax
  jmp write_loop
exit_zero:
  mov eax, SYS_EXIT
  xor edi, edi
  syscall
.section .rodata
.p2align 4
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.bss
.p2align 4
input_buf:
.zero 49152
.p2align 4
output_buf:
.zero 66560
