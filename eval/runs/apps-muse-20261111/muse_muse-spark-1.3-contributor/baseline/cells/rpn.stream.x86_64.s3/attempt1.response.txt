.globl _start
.text
_start:
  xor r12, r12
read_loop:
  cmp r12, 65536
  jae read_discard
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov rdx, 65536
  sub rdx, r12
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_discard:
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + eval_stack]
  mov rdx, 1024
  syscall
  test rax, rax
  jz read_done
  js read_done
  jmp read_discard
read_done:
  test r12, r12
  jz do_exit0
  lea rbx, [rip + input_buf]
  xor r13, r13
line_loop:
  cmp r13, r12
  jae do_exit0
  mov r14, r13
  mov r15, r14
find_nl:
  cmp r15, r12
  jae line_eof
  movzx eax, byte ptr [rbx + r15]
  cmp al, 10
  je line_nl
  inc r15
  jmp find_nl
line_nl:
  mov rbp, r15
  inc rbp
  jmp process_line
line_eof:
  mov rbp, r12
process_line:
  xor r10, r10
  mov r8, r14
tok_skip:
  cmp r8, r15
  jae tok_end
  movzx eax, byte ptr [rbx + r8]
  cmp al, 32
  je tok_skip_inc
  cmp al, 9
  je tok_skip_inc
  jmp tok_begin
tok_skip_inc:
  inc r8
  jmp tok_skip
tok_begin:
  mov r9, r8
tok_scan:
  inc r8
  cmp r8, r15
  jae tok_have
  movzx eax, byte ptr [rbx + r8]
  cmp al, 32
  je tok_have
  cmp al, 9
  je tok_have
  jmp tok_scan
tok_have:
  mov rcx, r8
  sub rcx, r9
  cmp rcx, 1
  jne try_literal
  movzx eax, byte ptr [rbx + r9]
  cmp al, 43
  je do_operator
  cmp al, 45
  je do_operator
  cmp al, 42
  je do_operator
  cmp al, 47
  je do_operator
  cmp al, 37
  je do_operator
try_literal:
  xor rdi, rdi
  xor rsi, rsi
  movzx eax, byte ptr [rbx + r9]
  cmp al, 45
  jne lit_check
  mov rdi, 1
  mov rsi, 1
lit_check:
  mov rax, rcx
  sub rax, rsi
  cmp rax, 1
  jb line_error
  cmp rax, 18
  ja line_error
  lea r11, [rbx + r9]
  mov rax, rsi
lit_digit_loop:
  cmp rax, rcx
  jae lit_parse
  movzx edx, byte ptr [r11 + rax]
  cmp dl, 48
  jb line_error
  cmp dl, 57
  ja line_error
  inc rax
  jmp lit_digit_loop
lit_parse:
  xor rax, rax
lit_parse_loop:
  cmp rsi, rcx
  jae lit_done
  mov rdx, rax
  shl rax, 3
  shl rdx, 1
  add rax, rdx
  movzx edx, byte ptr [r11 + rsi]
  sub edx, 48
  add rax, rdx
  inc rsi
  jmp lit_parse_loop
lit_done:
  test rdi, rdi
  jz lit_push
  neg rax
lit_push:
  cmp r10, 2048
  jae line_error
  lea r11, [rip + eval_stack]
  mov qword ptr [r11 + r10*8], rax
  inc r10
  jmp tok_skip
do_operator:
  cmp r10, 2
  jb line_error
  movzx eax, byte ptr [rbx + r9]
  mov rdi, rax
  lea r11, [rip + eval_stack]
  sub r10, 2
  mov rax, qword ptr [r11 + r10*8]
  mov rcx, qword ptr [r11 + r10*8 + 8]
  cmp dil, 43
  je op_add
  cmp dil, 45
  je op_sub
  cmp dil, 42
  je op_mul
  cmp dil, 47
  je op_div
  jmp op_mod
op_add:
  add rax, rcx
  jmp op_store
op_sub:
  sub rax, rcx
  jmp op_store
op_mul:
  imul rax, rcx
  jmp op_store
op_div:
  test rcx, rcx
  jz line_error
  cmp rcx, -1
  jne op_div_ok
  mov rdx, 1
  shl rdx, 63
  cmp rax, rdx
  je line_error
op_div_ok:
  cqo
  idiv rcx
  jmp op_store
op_mod:
  test rcx, rcx
  jz line_error
  cmp rcx, -1
  jne op_mod_ok
  mov rdx, 1
  shl rdx, 63
  cmp rax, rdx
  je line_error
op_mod_ok:
  cqo
  idiv rcx
  mov rax, rdx
op_store:
  mov qword ptr [r11 + r10*8], rax
  inc r10
  jmp tok_skip
tok_end:
  cmp r10, 1
  jne line_error
  lea r11, [rip + eval_stack]
  mov rax, qword ptr [r11]
  lea r11, [rip + outbuf]
  mov r8, rax
  xor r9, r9
  test rax, rax
  jns print_mag_ready
  mov byte ptr [r11], 45
  mov r9, 1
  neg r8
print_mag_ready:
  lea rdi, [rip + digit_tmp]
  mov rax, r8
  xor r10, r10
  test rax, rax
  jz print_zero
  mov rcx, 10
print_div_loop:
  xor rdx, rdx
  div rcx
  add dl, 48
  mov byte ptr [rdi + r10], dl
  inc r10
  test rax, rax
  jnz print_div_loop
  jmp print_copy
print_zero:
  mov byte ptr [rdi], 48
  mov r10, 1
print_copy:
  add r11, r9
  xor rsi, rsi
print_copy_loop:
  cmp rsi, r10
  jae print_copy_done
  mov rax, r10
  sub rax, rsi
  dec rax
  movzx edx, byte ptr [rdi + rax]
  mov byte ptr [r11 + rsi], dl
  inc rsi
  jmp print_copy_loop
print_copy_done:
  mov byte ptr [r11 + r10], 10
  mov rdx, r9
  add rdx, r10
  inc rdx
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + outbuf]
  syscall
  mov r13, rbp
  jmp line_loop
line_error:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + err_msg]
  mov rdx, 6
  syscall
  mov r13, rbp
  jmp line_loop
do_exit0:
  mov rax, 60
  mov rdi, 0
  syscall
  jmp do_exit0
.section .rodata
err_msg:
  .ascii "error"
  .byte 10
.bss
  .balign 8
input_buf:
  .zero 65536
eval_stack:
  .zero 16384
outbuf:
  .zero 32
digit_tmp:
  .zero 32
