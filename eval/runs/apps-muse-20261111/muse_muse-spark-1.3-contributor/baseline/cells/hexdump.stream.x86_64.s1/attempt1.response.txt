.text
.globl _start
_start:
  xor r12, r12
  lea r13, [rip + buffer]
  lea r10, [rip + hexdigits]
read_loop:
  cmp r12, 65536
  jae done_read
  mov rax, 0
  mov rdi, 0
  mov rsi, r13
  add rsi, r12
  mov rdx, 65536
  sub rdx, r12
  syscall
  test rax, rax
  jle done_read
  add r12, rax
  jmp read_loop
done_read:
  xor r14, r14
row_loop:
  cmp r14, r12
  jae print_final
  mov r15, r12
  sub r15, r14
  cmp r15, 16
  jbe n_ok
  mov r15, 16
n_ok:
  lea rbx, [rip + linebuf]
  mov eax, r14d
  mov r9d, 8
off_loop:
  rol eax, 4
  mov edx, eax
  and edx, 15
  movzx ecx, byte ptr [r10 + rdx]
  mov byte ptr [rbx], cl
  inc rbx
  dec r9d
  jnz off_loop
  mov byte ptr [rbx], 32
  inc rbx
  mov byte ptr [rbx], 32
  inc rbx
  xor r8d, r8d
hex_loop:
  cmp r8, r15
  jae hex_blank
  mov rsi, r14
  add rsi, r8
  movzx eax, byte ptr [r13 + rsi]
  mov edx, eax
  shr edx, 4
  movzx ecx, byte ptr [r10 + rdx]
  mov byte ptr [rbx], cl
  inc rbx
  and eax, 15
  movzx ecx, byte ptr [r10 + rax]
  mov byte ptr [rbx], cl
  inc rbx
  mov byte ptr [rbx], 32
  inc rbx
  jmp hex_after
hex_blank:
  mov byte ptr [rbx], 32
  inc rbx
  mov byte ptr [rbx], 32
  inc rbx
  mov byte ptr [rbx], 32
  inc rbx
hex_after:
  cmp r8, 7
  jne hex_noextra
  mov byte ptr [rbx], 32
  inc rbx
hex_noextra:
  inc r8
  cmp r8, 16
  jb hex_loop
  mov byte ptr [rbx], 32
  inc rbx
  mov byte ptr [rbx], 124
  inc rbx
  xor r8d, r8d
ascii_loop:
  cmp r8, r15
  jae ascii_done
  mov rsi, r14
  add rsi, r8
  movzx eax, byte ptr [r13 + rsi]
  cmp al, 32
  jb ascii_dot
  cmp al, 126
  ja ascii_dot
  jmp ascii_store
ascii_dot:
  mov al, 46
ascii_store:
  mov byte ptr [rbx], al
  inc rbx
  inc r8
  jmp ascii_loop
ascii_done:
  mov byte ptr [rbx], 124
  inc rbx
  mov byte ptr [rbx], 10
  inc rbx
  lea rsi, [rip + linebuf]
  mov rdx, rbx
  sub rdx, rsi
  mov rax, 1
  mov rdi, 1
  syscall
  add r14, r15
  jmp row_loop
print_final:
  lea rbx, [rip + linebuf]
  mov eax, r12d
  mov r9d, 8
final_loop:
  rol eax, 4
  mov edx, eax
  and edx, 15
  movzx ecx, byte ptr [r10 + rdx]
  mov byte ptr [rbx], cl
  inc rbx
  dec r9d
  jnz final_loop
  mov byte ptr [rbx], 10
  inc rbx
  lea rsi, [rip + linebuf]
  mov rdx, rbx
  sub rdx, rsi
  mov rax, 1
  mov rdi, 1
  syscall
  mov rax, 60
  xor rdi, rdi
  syscall
.section .rodata
hexdigits:
.ascii "0123456789abcdef"
.bss
.align 16
buffer:
.zero 65536
linebuf:
.zero 80
