.text
.globl _start
_start:
xor r12, r12
lea rbx, [rip + buffer]
read_loop:
mov rax, 32768
sub rax, r12
jz parse_init
mov rsi, rbx
add rsi, r12
mov rdx, rax
mov rax, 0
mov rdi, 0
syscall
test rax, rax
jz parse_init
js parse_init
add r12, rax
jmp read_loop
parse_init:
xor r13, r13
test r12, r12
jz exit_program
mov r14, rbx
mov r15, rbx
add r15, r12
mov rsi, rbx
lea rbx, [rip + line_ptr]
lea r12, [rip + line_len]
parse_loop:
cmp rsi, r15
jae parse_tail
movzx eax, byte ptr [rsi]
cmp eax, 10
je parse_newline
inc rsi
jmp parse_loop
parse_newline:
mov rax, rsi
sub rax, r14
mov qword ptr [rbx + r13*8], r14
mov qword ptr [r12 + r13*8], rax
inc r13
inc rsi
mov r14, rsi
jmp parse_loop
parse_tail:
cmp r14, r15
jae sort_check
mov rax, r15
sub rax, r14
mov qword ptr [rbx + r13*8], r14
mov qword ptr [r12 + r13*8], rax
inc r13
sort_check:
cmp r13, 2
jb write_init
mov r15, 1
outer_loop:
cmp r15, r13
jae write_init
mov r8, qword ptr [rbx + r15*8]
mov r9, qword ptr [r12 + r15*8]
mov r14, r15
dec r14
inner_loop:
cmp r14, 0
jl inner_done
mov r10, qword ptr [rbx + r14*8]
mov r11, qword ptr [r12 + r14*8]
mov rcx, r11
cmp rcx, r9
jbe cmp_min_done
mov rcx, r9
cmp_min_done:
xor rdx, rdx
cmp_byte_loop:
cmp rdx, rcx
jae cmp_len_decide
movzx eax, byte ptr [r10 + rdx]
movzx esi, byte ptr [r8 + rdx]
cmp eax, esi
jne cmp_diff
inc rdx
jmp cmp_byte_loop
cmp_diff:
ja shift_needed
jmp inner_done
cmp_len_decide:
cmp r11, r9
ja shift_needed
jmp inner_done
shift_needed:
mov rax, r14
inc rax
mov qword ptr [rbx + rax*8], r10
mov qword ptr [r12 + rax*8], r11
dec r14
jmp inner_loop
inner_done:
mov rax, r14
inc rax
mov qword ptr [rbx + rax*8], r8
mov qword ptr [r12 + rax*8], r9
inc r15
jmp outer_loop
write_init:
test r13, r13
jz exit_program
xor r15, r15
write_loop:
cmp r15, r13
jae exit_program
mov rsi, qword ptr [rbx + r15*8]
mov rdx, qword ptr [r12 + r15*8]
test rdx, rdx
jz write_newline
mov rax, 1
mov rdi, 1
syscall
write_newline:
lea rsi, [rip + nl]
mov rdx, 1
mov rax, 1
mov rdi, 1
syscall
inc r15
jmp write_loop
exit_program:
mov rax, 60
xor rdi, rdi
syscall
