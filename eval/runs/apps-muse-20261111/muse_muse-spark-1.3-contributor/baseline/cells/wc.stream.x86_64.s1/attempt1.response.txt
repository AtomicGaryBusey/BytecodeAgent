.text
.globl _start
_start:
  xor r12, r12
  xor r13, r13
  xor r14, r14
  xor r15, r15
read_loop:
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + buffer]
  mov rdx, 8192
  syscall
  cmp rax, 0
  jle done
  add r14, rax
  mov rcx, rax
  lea rsi, [rip + buffer]
byte_loop:
  movzx eax, byte ptr [rsi]
  inc rsi
  cmp al, 10
  jne not_nl
  inc r12
not_nl:
  cmp al, 32
  je is_ws
  cmp al, 9
  je is_ws
  cmp al, 10
  je is_ws
  cmp al, 11
  je is_ws
  cmp al, 12
  je is_ws
  cmp al, 13
  je is_ws
  test r15, r15
  jnz next_byte
  inc r13
  mov r15, 1
  jmp next_byte
is_ws:
  xor r15, r15
next_byte:
  dec rcx
  jnz byte_loop
  jmp read_loop
done:
  lea rdi, [rip + outbuf]
  mov rax, r12
  call print_num
  mov byte ptr [rdi], 32
  inc rdi
  mov rax, r13
  call print_num
  mov byte ptr [rdi], 32
  inc rdi
  mov rax, r14
  call print_num
  mov byte ptr [rdi], 10
  inc rdi
  mov rdx, rdi
  lea rsi, [rip + outbuf]
  sub rdx, rsi
  mov rax, 1
  mov rdi, 1
  syscall
  mov rax, 60
  xor edi, edi
  syscall
print_num:
  test rax, rax
  jnz pn_nonzero
  mov byte ptr [rdi], 48
  inc rdi
  ret
pn_nonzero:
  mov r8, 10
  sub rsp, 32
  mov rsi, rsp
  add rsi, 32
  xor ecx, ecx
pn_divloop:
  xor edx, edx
  div r8
  dec rsi
  add dl, 48
  mov byte ptr [rsi], dl
  inc ecx
  test rax, rax
  jnz pn_divloop
pn_copyloop:
  movzx eax, byte ptr [rsi]
  mov byte ptr [rdi], al
  inc rsi
  inc rdi
  dec ecx
  jnz pn_copyloop
  add rsp, 32
  ret
.bss
.align 16
buffer:
.zero 8192
outbuf:
.zero 96
