.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    cmp r12, 49152
    jae encode_start
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jle encode_start
    add r12, rax
    jmp read_loop

encode_start:
    lea rsi, [rip + input_buf]
    lea rdi, [rip + output_buf]
    mov r13, rdi
    lea rbp, [rip + b64_alphabet]
    xor r14d, r14d
    mov rbx, r12

encode_groups:
    cmp rbx, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    sub rbx, 3
    add r14d, 4
    cmp r14d, 76
    jne encode_groups

    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
    jmp encode_groups

encode_tail:
    test rbx, rbx
    jnz encode_partial
    test r14d, r14d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi
    jmp write_output

encode_partial:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rbx, 1
    je partial_value_ready
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx

partial_value_ready:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 1], dl

    cmp rbx, 1
    je partial_one_byte

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 2], dl
    jmp partial_padding

partial_one_byte:
    mov byte ptr [rdi + 2], 61

partial_padding:
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    mov rdx, rdi
    sub rdx, r13
    test rdx, rdx
    jz exit_ok
    mov rsi, r13
    mov edi, 1
    mov eax, 1
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
b64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buf:
    .zero 49152
.align 16
output_buf:
    .zero 66560
