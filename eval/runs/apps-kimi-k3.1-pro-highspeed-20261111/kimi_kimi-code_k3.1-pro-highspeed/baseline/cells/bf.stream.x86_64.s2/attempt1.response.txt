.globl _start

.text
_start:
    lea r12, [rip + input_buf]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    cmp r13, 65537
    jb read_loop

read_done:
    lea r14, [r12 + r13]
    mov r13, r14
    mov r15, r14
    mov rbx, r14
    mov rax, r12

find_separator:
    cmp rax, r14
    jae separator_done
    cmp byte ptr [rax], 0x21
    je separator_found
    inc rax
    jmp find_separator

separator_found:
    mov r13, rax
    lea r15, [rax + 1]

separator_done:
    mov rsi, r12
    lea rdi, [rip + bracket_stack]
    mov r9, rdi
    lea r8, [rip + jump_map]

validate_loop:
    cmp rsi, r13
    jae validate_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x5b
    je validate_open
    cmp al, 0x5d
    je validate_close
    inc rsi
    jmp validate_loop

validate_open:
    mov rax, rsi
    sub rax, r12
    mov dword ptr [rdi], eax
    add rdi, 4
    inc rsi
    jmp validate_loop

validate_close:
    cmp rdi, r9
    je invalid_program
    sub rdi, 4
    mov rax, rsi
    sub rax, r12
    mov edx, dword ptr [rdi]
    mov dword ptr [r8 + rdx*4], eax
    mov dword ptr [r8 + rax*4], edx
    inc rsi
    jmp validate_loop

validate_done:
    cmp rdi, r9
    jne invalid_program

    mov rsi, r12
    lea rbp, [rip + tape]
    lea rdi, [rip + output_buf]

interpret_loop:
    cmp rsi, r13
    jae program_done
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x3e
    je command_right
    cmp al, 0x3c
    je command_left
    cmp al, 0x2b
    je command_increment
    cmp al, 0x2d
    je command_decrement
    cmp al, 0x2e
    je command_output
    cmp al, 0x2c
    je command_input
    cmp al, 0x5b
    je command_open
    cmp al, 0x5d
    je command_close
    jmp interpret_loop

command_right:
    inc rbp
    jmp interpret_loop

command_left:
    dec rbp
    jmp interpret_loop

command_increment:
    inc byte ptr [rbp]
    jmp interpret_loop

command_decrement:
    dec byte ptr [rbp]
    jmp interpret_loop

command_output:
    mov al, byte ptr [rbp]
    mov byte ptr [rdi], al
    inc rdi
    jmp interpret_loop

command_input:
    cmp r15, rbx
    jae input_exhausted
    mov al, byte ptr [r15]
    inc r15
    mov byte ptr [rbp], al
    jmp interpret_loop

input_exhausted:
    mov byte ptr [rbp], 0
    jmp interpret_loop

command_open:
    cmp byte ptr [rbp], 0
    jne interpret_loop
    jmp bracket_jump

command_close:
    cmp byte ptr [rbp], 0
    je interpret_loop

bracket_jump:
    mov rdx, rsi
    sub rdx, r12
    dec rdx
    mov eax, dword ptr [r8 + rdx*4]
    lea rsi, [r12 + rax + 1]
    jmp interpret_loop

program_done:
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buf:
    .zero 65537
.balign 16
output_buf:
    .zero 65536
.balign 16
tape:
    .zero 30000
.balign 16
jump_map:
    .zero 262144
.balign 16
bracket_stack:
    .zero 262144
