.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    xor r12d, r12d

read_more:
    cmp r12, 49152
    jae input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jz input_ready
    add r12, rax
    jmp read_more

input_ready:
    lea rsi, [rip + input_buffer]
    mov rbx, r12
    lea r13, [rip + alphabet]
    lea r14, [rip + output_buffer]
    xor r15d, r15d

encode_groups:
    cmp rbx, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov ebp, eax

    add rsi, 3
    sub rbx, 3

    mov eax, ebp
    shr eax, 18
    mov al, byte ptr [r13 + rax]
    call put_character

    mov eax, ebp
    shr eax, 12
    and eax, 63
    mov al, byte ptr [r13 + rax]
    call put_character

    mov eax, ebp
    shr eax, 6
    and eax, 63
    mov al, byte ptr [r13 + rax]
    call put_character

    mov eax, ebp
    and eax, 63
    mov al, byte ptr [r13 + rax]
    call put_character

    jmp encode_groups

encode_tail:
    test rbx, rbx
    jz finish_output
    cmp rbx, 1
    jne tail_two

    movzx ebp, byte ptr [rsi]

    mov eax, ebp
    shr eax, 2
    mov al, byte ptr [r13 + rax]
    call put_character

    mov eax, ebp
    and eax, 3
    shl eax, 4
    mov al, byte ptr [r13 + rax]
    call put_character

    mov al, '='
    call put_character
    mov al, '='
    call put_character
    jmp finish_output

tail_two:
    movzx ebp, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    shl ebp, 8
    or ebp, ecx

    mov eax, ebp
    shr eax, 10
    mov al, byte ptr [r13 + rax]
    call put_character

    mov eax, ebp
    shr eax, 4
    and eax, 63
    mov al, byte ptr [r13 + rax]
    call put_character

    mov eax, ebp
    shl eax, 2
    and eax, 63
    mov al, byte ptr [r13 + rax]
    call put_character

    mov al, '='
    call put_character

finish_output:
    test r15, r15
    jz write_output
    mov byte ptr [r14], 10
    inc r14

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

put_character:
    mov byte ptr [r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne put_character_done
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d
put_character_done:
    ret

.bss
.align 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66400
