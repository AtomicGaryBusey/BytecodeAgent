.globl _start

.text
_start:
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    mov x19, #0

read_more:
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x19
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cbz x0, input_complete
    add x19, x19, x0
    b read_more

input_complete:
    mov x0, #0
find_separator:
    cmp x0, x19
    b.eq no_separator
    ldrb w1, [x20, x0]
    cmp w1, #0x21
    b.eq separator_found
    add x0, x0, #1
    b find_separator

separator_found:
    mov x21, x0
    add x23, x20, x0
    add x23, x23, #1
    sub x24, x19, x0
    sub x24, x24, #1
    b validation_setup

no_separator:
    mov x21, x19
    mov x23, x20
    mov x24, #0

validation_setup:
    adrp x25, jump_table
    add x25, x25, :lo12:jump_table
    adrp x5, bracket_stack
    add x5, x5, :lo12:bracket_stack
    mov x0, #0
    mov x3, #0

validate_loop:
    cmp x0, x21
    b.hs validation_finished
    ldrb w1, [x20, x0]
    cmp w1, #0x5b
    b.eq validate_open
    cmp w1, #0x5d
    b.eq validate_close

validate_next:
    add x0, x0, #1
    b validate_loop

validate_open:
    cmp x3, #1000
    b.hs invalid_program
    str w0, [x5, x3, lsl #2]
    add x3, x3, #1
    b validate_next

validate_close:
    cbz x3, invalid_program
    sub x3, x3, #1
    ldr w4, [x5, x3, lsl #2]
    add w6, w0, #1
    str w6, [x25, x4, lsl #2]
    add w6, w4, #1
    str w6, [x25, x0, lsl #2]
    b validate_next

validation_finished:
    cbnz x3, invalid_program

    adrp x26, tape
    add x26, x26, :lo12:tape
    mov x27, x26
    adrp x16, output_buffer
    add x16, x16, :lo12:output_buffer
    mov x15, x16
    mov x28, #0

execute_loop:
    cmp x28, x21
    b.hs execution_finished
    ldrb w0, [x20, x28]

    cmp w0, #0x3e
    b.eq command_right
    cmp w0, #0x3c
    b.eq command_left
    cmp w0, #0x2b
    b.eq command_increment
    cmp w0, #0x2d
    b.eq command_decrement
    cmp w0, #0x2e
    b.eq command_output
    cmp w0, #0x2c
    b.eq command_input
    cmp w0, #0x5b
    b.eq command_open
    cmp w0, #0x5d
    b.eq command_close

next_command:
    add x28, x28, #1
    b execute_loop

command_right:
    add x27, x27, #1
    b next_command

command_left:
    sub x27, x27, #1
    b next_command

command_increment:
    ldrb w1, [x27]
    add w1, w1, #1
    strb w1, [x27]
    b next_command

command_decrement:
    ldrb w1, [x27]
    sub w1, w1, #1
    strb w1, [x27]
    b next_command

command_output:
    ldrb w1, [x27]
    strb w1, [x15], #1
    b next_command

command_input:
    cbz x24, input_exhausted
    ldrb w1, [x23], #1
    sub x24, x24, #1
    strb w1, [x27]
    b next_command

input_exhausted:
    strb wzr, [x27]
    b next_command

command_open:
    ldrb w1, [x27]
    cbnz w1, next_command
    ldr w0, [x25, x28, lsl #2]
    mov x28, x0
    b execute_loop

command_close:
    ldrb w1, [x27]
    cbz w1, next_command
    ldr w0, [x25, x28, lsl #2]
    mov x28, x0
    b execute_loop

execution_finished:
    mov x0, #1
    mov x1, x16
    sub x2, x15, x16
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.align 4
input_buffer:
    .zero 65537

.align 4
jump_table:
    .zero 262144

.align 4
bracket_stack:
    .zero 4000

.align 4
tape:
    .zero 30000

.align 4
output_buffer:
    .zero 65536
