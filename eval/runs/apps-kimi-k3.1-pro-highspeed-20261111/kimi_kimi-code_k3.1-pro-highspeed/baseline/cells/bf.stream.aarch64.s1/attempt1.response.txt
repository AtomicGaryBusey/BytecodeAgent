.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #65536
    add x21, x21, #1

read_more:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_more

read_done:
    mov x0, #0

find_separator:
    cmp x0, x20
    b.hs no_separator
    ldrb w1, [x19, x0]
    cmp w1, #33
    b.eq separator_found
    add x0, x0, #1
    b find_separator

separator_found:
    mov x21, x0
    add x24, x19, x0
    add x24, x24, #1
    add x25, x19, x20
    b split_done

no_separator:
    mov x21, x20
    add x24, x19, x20
    mov x25, x24

split_done:
    adrp x28, jump_table
    add x28, x28, :lo12:jump_table
    adrp x29, bracket_stack
    add x29, x29, :lo12:bracket_stack
    mov x0, #0
    mov x1, #0

validate_loop:
    cmp x0, x21
    b.hs validate_done
    ldrb w3, [x19, x0]
    cmp w3, #91
    b.eq open_bracket
    cmp w3, #93
    b.eq close_bracket

validate_next:
    add x0, x0, #1
    b validate_loop

open_bracket:
    str w0, [x29, x1, lsl #2]
    add x1, x1, #1
    b validate_next

close_bracket:
    cbz x1, invalid_program
    sub x1, x1, #1
    ldr w3, [x29, x1, lsl #2]
    str w0, [x28, x3, lsl #2]
    str w3, [x28, x0, lsl #2]
    b validate_next

validate_done:
    cbnz x1, invalid_program

    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x20, #0
    mov x23, #0
    mov x27, #0

run_loop:
    cmp x20, x21
    b.hs program_done
    ldrb w0, [x19, x20]
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_increment
    cmp w0, #45
    b.eq cmd_decrement
    cmp w0, #46
    b.eq cmd_output
    cmp w0, #44
    b.eq cmd_input
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close

advance_pc:
    add x20, x20, #1
    b run_loop

cmd_right:
    add x23, x23, #1
    b advance_pc

cmd_left:
    sub x23, x23, #1
    b advance_pc

cmd_increment:
    ldrb w1, [x22, x23]
    add w1, w1, #1
    strb w1, [x22, x23]
    b advance_pc

cmd_decrement:
    ldrb w1, [x22, x23]
    sub w1, w1, #1
    strb w1, [x22, x23]
    b advance_pc

cmd_output:
    ldrb w1, [x22, x23]
    strb w1, [x26, x27]
    add x27, x27, #1
    b advance_pc

cmd_input:
    cmp x24, x25
    b.hs input_exhausted
    ldrb w1, [x24], #1
    b store_input

input_exhausted:
    mov w1, #0

store_input:
    strb w1, [x22, x23]
    b advance_pc

cmd_open:
    ldrb w1, [x22, x23]
    cbnz w1, advance_pc
    ldr w1, [x28, x20, lsl #2]
    add x20, x1, #1
    b run_loop

cmd_close:
    ldrb w1, [x22, x23]
    cbz w1, advance_pc
    ldr w1, [x28, x20, lsl #2]
    add x20, x1, #1
    b run_loop

program_done:
    cbz x27, exit_success
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0

exit_success:

    mov x8, #93
    svc #0

    mov x8, #93
    svc #0
