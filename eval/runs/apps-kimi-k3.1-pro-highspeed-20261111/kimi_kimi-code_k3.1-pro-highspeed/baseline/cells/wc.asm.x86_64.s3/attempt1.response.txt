.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall

    test rax, rax
    jle input_done

    add r14, rax
    mov rcx, rax

byte_loop:
    movzx eax, byte ptr [rsi]

    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    inc rsi
    dec rcx
    jnz byte_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

append_decimal:
    lea rsi, [rip + number_buffer + 32]
    mov r8, rsi
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    cmp rsi, r8
    je 3f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    jmp 2b
3:
    ret

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
number_buffer:
    .zero 32
