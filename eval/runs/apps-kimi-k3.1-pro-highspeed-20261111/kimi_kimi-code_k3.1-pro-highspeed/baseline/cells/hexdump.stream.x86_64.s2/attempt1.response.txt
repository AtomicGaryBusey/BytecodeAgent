.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .balign 16
input_buffer:
    .zero 65537
output_buffer:
    .zero 323600

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz input_complete
    add r12, rax
    jmp read_more

input_complete:
    xor r13d, r13d
    lea r14, [rip + input_buffer]
    lea r15, [rip + output_buffer]
    mov rbx, r15
    lea r8, [rip + hex_digits]

row_loop:
    cmp r13, r12
    jae final_offset

    mov rbp, r12
    sub rbp, r13
    cmp rbp, 16
    jbe 1f
    mov ebp, 16
1:
    mov eax, r13d
    mov ecx, 8
2:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r15], dl
    inc r15
    dec ecx
    jnz 2b

    mov word ptr [r15], 0x2020
    add r15, 2

    xor ecx, ecx
hex_slot_loop:
    cmp rcx, rbp
    jae blank_slot

    movzx eax, byte ptr [r14 + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r15], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [r15 + 1], al
    mov byte ptr [r15 + 2], 32
    add r15, 3
    jmp slot_done

blank_slot:
    mov word ptr [r15], 0x2020
    mov byte ptr [r15 + 2], 32
    add r15, 3

slot_done:
    cmp ecx, 7
    jne 3f
    mov byte ptr [r15], 32
    inc r15
3:
    inc ecx
    cmp ecx, 16
    jb hex_slot_loop

    mov word ptr [r15], 0x7c20
    add r15, 2

    xor ecx, ecx
ascii_loop:
    movzx eax, byte ptr [r14 + rcx]
    cmp eax, 32
    jb 4f
    cmp eax, 126
    jbe 5f
4:
    mov eax, 46
5:
    mov byte ptr [r15], al
    inc r15
    inc ecx
    cmp rcx, rbp
    jb ascii_loop

    mov word ptr [r15], 0x0a7c
    add r15, 2
    add r13, rbp
    add r14, rbp
    jmp row_loop

final_offset:
    mov eax, r12d
    mov ecx, 8
6:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r15], dl
    inc r15
    dec ecx
    jnz 6b

    mov byte ptr [r15], 10
    inc r15

    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r15
    sub rdx, rbx
    syscall

    mov eax, 60
    xor edi, edi
    syscall
