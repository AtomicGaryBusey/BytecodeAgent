.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    movz x21, #1, lsl #16
    add x21, x21, #1

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, input_done
    tbnz x0, #63, input_done
    add x20, x20, x0
    cmp x20, x21
    b.lo read_loop

input_done:
    add x20, x19, x20

line_loop:
    cmp x19, x20
    b.eq program_exit

    mov x21, x19
    mov x0, x19

find_line_end:
    cmp x0, x20
    b.eq final_line
    ldrb w1, [x0]
    cmp w1, #10
    b.eq newline_found
    add x0, x0, #1
    b find_line_end

newline_found:
    mov x22, x0
    add x19, x0, #1
    b initialise_line

final_line:
    mov x22, x20
    mov x19, x20

initialise_line:
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack
    mov x25, x24

skip_separators:
    cmp x21, x22
    b.eq line_finished
    ldrb w0, [x21]
    cmp w0, #32
    b.eq skip_one
    cmp w0, #9
    b.ne token_start
skip_one:
    add x21, x21, #1
    b skip_separators

token_start:
    mov x23, x21
    mov x26, x21

scan_token:
    cmp x26, x22
    b.eq token_ready
    ldrb w0, [x26]
    cmp w0, #32
    b.eq token_ready
    cmp w0, #9
    b.eq token_ready
    add x26, x26, #1
    b scan_token

token_ready:
    sub x9, x26, x23
    cmp x9, #1
    b.ne parse_literal

    ldrb w0, [x23]
    mov w10, w0
    cmp w0, #43
    b.eq have_operator
    cmp w0, #45
    b.eq have_operator
    cmp w0, #42
    b.eq have_operator
    cmp w0, #47
    b.eq have_operator
    cmp w0, #37
    b.eq have_operator

parse_literal:
    mov x0, x23
    mov w2, #0
    ldrb w1, [x0]
    cmp w1, #45
    b.ne literal_digits_setup
    mov w2, #1
    add x0, x0, #1

literal_digits_setup:
    sub x1, x26, x0
    cbz x1, line_error
    cmp x1, #18
    b.hi line_error
    mov x3, #0
    mov x6, #10

literal_digit_loop:
    ldrb w5, [x0], #1
    sub w5, w5, #48
    cmp w5, #9
    b.hi line_error
    madd x3, x3, x6, x5
    subs x1, x1, #1
    b.ne literal_digit_loop

    cbz w2, push_literal
    neg x3, x3

push_literal:
    str x3, [x24], #8
    mov x21, x26
    b skip_separators

have_operator:
    sub x0, x24, x25
    cmp x0, #16
    b.lo line_error
    ldr x1, [x24, #-8]
    ldr x2, [x24, #-16]

    cmp w10, #43
    b.eq operator_add
    cmp w10, #45
    b.eq operator_subtract
    cmp w10, #42
    b.eq operator_multiply
    cmp w10, #47
    b.eq operator_divide
    b operator_remainder

operator_add:
    add x3, x2, x1
    b store_operator_result

operator_subtract:
    sub x3, x2, x1
    b store_operator_result

operator_multiply:
    mul x3, x2, x1
    b store_operator_result

operator_divide:
    cbz x1, line_error
    movz x3, #0x8000, lsl #48
    cmp x2, x3
    b.ne divide_ready
    cmn x1, #1
    b.eq line_error
divide_ready:
    sdiv x3, x2, x1
    b store_operator_result

operator_remainder:
    cbz x1, line_error
    movz x3, #0x8000, lsl #48
    cmp x2, x3
    b.ne remainder_ready
    cmn x1, #1
    b.eq line_error
remainder_ready:
    sdiv x4, x2, x1
    msub x3, x4, x1, x2

store_operator_result:
    sub x24, x24, #8
    str x3, [x24, #-8]
    mov x21, x26
    b skip_separators

line_finished:
    sub x0, x24, x25
    cmp x0, #8
    b.ne line_error
    ldr x0, [x25]
    bl print_value
    b line_loop

line_error:
    mov x0, #1
    adrp x1, error_message
    add x1, x1, :lo12:error_message
    mov x2, #6
    mov x8, #64
    svc #0
    b line_loop

print_value:
    adrp x1, number_buffer
    add x1, x1, :lo12:number_buffer
    add x1, x1, #32
    sub x1, x1, #1
    mov w2, #10
    strb w2, [x1]

    mov x3, x0
    mov w4, #0
    cmp x3, #0
    b.pl magnitude_ready
    neg x3, x3
    mov w4, #1

magnitude_ready:
    mov x5, #10

decimal_loop:
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #48
    sub x1, x1, #1
    strb w7, [x1]
    mov x3, x6
    cbnz x3, decimal_loop

    cbz w4, decimal_ready
    sub x1, x1, #1
    mov w7, #45
    strb w7, [x1]

decimal_ready:
    adrp x2, number_buffer
    add x2, x2, :lo12:number_buffer
    add x2, x2, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_message:
    .ascii "error\n"

.bss
.balign 8
input_buffer:
    .zero 65537
value_stack:
    .zero 8192
number_buffer:
    .zero 32
