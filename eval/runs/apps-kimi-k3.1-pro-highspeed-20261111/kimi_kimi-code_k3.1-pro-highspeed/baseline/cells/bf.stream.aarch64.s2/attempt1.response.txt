.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #65536

read_loop:
    cmp x20, x21
    b.hs input_ready
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_ready
    add x20, x20, x0
    b read_loop

input_ready:
    adrp x22, code_buffer
    add x22, x22, :lo12:code_buffer
    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer
    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x29, jump_table
    add x29, x29, :lo12:jump_table
    adrp x15, bracket_stack
    add x15, x15, :lo12:bracket_stack

    mov x0, #0
    mov x21, #0
    mov x2, #0
    mov x27, #0
    mov x28, #0

parse_loop:
    cmp x0, x20
    b.hs parse_end
    ldrb w3, [x19, x0]
    add x0, x0, #1

    cmp w3, #33
    b.eq separator_found
    cmp w3, #62
    b.eq store_command
    cmp w3, #60
    b.eq store_command
    cmp w3, #43
    b.eq store_command
    cmp w3, #45
    b.eq store_command
    cmp w3, #46
    b.eq store_command
    cmp w3, #44
    b.eq store_command
    cmp w3, #91
    b.eq parse_open
    cmp w3, #93
    b.eq parse_close
    b parse_loop

parse_open:
    mov w4, #1000
    cmp x2, x4
    b.hs invalid_program
    str w21, [x15, x2, lsl #2]
    add x2, x2, #1
    b store_command

parse_close:
    cbz x2, invalid_program
    sub x2, x2, #1
    ldr w4, [x15, x2, lsl #2]
    str w21, [x29, x4, lsl #2]
    str w4, [x29, x21, lsl #2]
    b store_command

store_command:
    strb w3, [x22, x21]
    add x21, x21, #1
    b parse_loop

separator_found:
    add x27, x19, x0
    sub x28, x20, x0
    b validate_brackets

parse_end:
    mov x27, #0
    mov x28, #0

validate_brackets:
    cbnz x2, invalid_program

    mov x20, #0
    mov x24, #0
    mov x26, x25

run_loop:
    cmp x20, x21
    b.hs program_done
    ldrb w0, [x22, x20]
    add x20, x20, #1

    cmp w0, #43
    b.eq cmd_plus
    cmp w0, #45
    b.eq cmd_minus
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #46
    b.eq cmd_output
    cmp w0, #44
    b.eq cmd_input
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close
    b run_loop

cmd_plus:
    ldrb w0, [x26]
    add w0, w0, #1
    strb w0, [x26]
    b run_loop

cmd_minus:
    ldrb w0, [x26]
    sub w0, w0, #1
    strb w0, [x26]
    b run_loop

cmd_right:
    add x26, x26, #1
    b run_loop

cmd_left:
    sub x26, x26, #1
    b run_loop

cmd_output:
    ldrb w0, [x26]
    strb w0, [x23, x24]
    add x24, x24, #1
    b run_loop

cmd_input:
    cbz x28, input_exhausted
    ldrb w0, [x27], #1
    sub x28, x28, #1
    strb w0, [x26]
    b run_loop

input_exhausted:
    strb wzr, [x26]
    b run_loop

cmd_open:
    ldrb w0, [x26]
    cbnz w0, run_loop
    sub x1, x20, #1
    ldr w1, [x29, x1, lsl #2]
    add x20, x1, #1
    b run_loop

cmd_close:
    ldrb w0, [x26]
    cbz w0, run_loop
    sub x1, x20, #1
    ldr w1, [x29, x1, lsl #2]
    add x20, x1, #1
    b run_loop

program_done:
    cbz x24, exit_success
    mov x0, #1
    mov x1, x23
    mov x2, x24
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0
