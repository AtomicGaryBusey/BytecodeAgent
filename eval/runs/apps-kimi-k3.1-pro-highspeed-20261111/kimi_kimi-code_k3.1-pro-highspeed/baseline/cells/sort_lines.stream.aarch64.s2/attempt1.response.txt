.section .rodata
.balign 1
newline:
.byte 10

.bss
.balign 16
input_buffer:
.zero 32768
.balign 16
line_table:
.zero 16000

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #32768
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x23, x19
    add x24, x19, x20
    mov x25, x23
    adrp x26, line_table
    add x26, x26, :lo12:line_table
    mov x22, #0

parse_loop:
    cmp x23, x24
    b.hs parse_tail
    ldrb w0, [x23], #1
    cmp w0, #10
    b.ne parse_loop
    mov x0, x25
    sub x1, x23, x25
    sub x1, x1, #1
    stp x0, x1, [x26], #16
    add x22, x22, #1
    mov x25, x23
    b parse_loop

parse_tail:
    cmp x25, x24
    b.eq parse_done
    mov x0, x25
    sub x1, x24, x25
    stp x0, x1, [x26], #16
    add x22, x22, #1

parse_done:
    adrp x26, line_table
    add x26, x26, :lo12:line_table
    mov x3, #1

sort_outer:
    cmp x3, x22
    b.hs sort_done
    lsl x4, x3, #4
    add x5, x26, x4
    ldp x6, x7, [x5]
    mov x8, x3

sort_inner:
    cbz x8, insert_key
    sub x9, x8, #1
    lsl x4, x9, #4
    add x10, x26, x4
    ldp x11, x12, [x10]

    cmp x12, x7
    csel x13, x12, x7, ls
    mov x14, #0

compare_loop:
    cmp x14, x13
    b.hs compare_prefix
    ldrb w0, [x11, x14]
    ldrb w1, [x6, x14]
    cmp w0, w1
    b.ne compare_different
    add x14, x14, #1
    b compare_loop

compare_prefix:
    cmp x12, x7
    b.hi shift_previous
    b insert_key

compare_different:
    b.hi shift_previous
    b insert_key

shift_previous:
    ldp x0, x1, [x10]
    lsl x4, x8, #4
    add x5, x26, x4
    stp x0, x1, [x5]
    sub x8, x8, #1
    b sort_inner

insert_key:
    lsl x4, x8, #4
    add x5, x26, x4
    stp x6, x7, [x5]
    add x3, x3, #1
    b sort_outer

sort_done:
    adrp x27, newline
    add x27, x27, :lo12:newline
    mov x3, #0

output_loop:
    cmp x3, x22
    b.hs output_done
    lsl x4, x3, #4
    add x5, x26, x4
    ldp x1, x2, [x5]
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #1
    mov x1, x27
    mov x2, #1
    mov x8, #64
    svc #0

    add x3, x3, #1
    b output_loop

output_done:
    mov x0, #0
    mov x8, #93
    svc #0

halt_loop:
    b halt_loop
