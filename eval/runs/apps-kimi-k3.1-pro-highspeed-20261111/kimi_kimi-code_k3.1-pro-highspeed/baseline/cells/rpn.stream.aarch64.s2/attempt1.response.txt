    mov x10, x1
    mov x11, x2
    mov x12, #0

    ldrb w14, [x9]
    cmp w14, #32

    cmp w14, #9
    b.eq skip_separator

    cmp x9, x10
    b.eq token_ready
    ldrb w14, [x9]
    cmp w14, #32
    b.eq token_ready

    cmp x17, #1
    b.ne number_token
    ldrb w14, [x15]
    cmp w14, #43
    b.eq operator_token
    cmp w14, #45
    b.eq operator_token
    cmp w14, #42
    b.eq operator_token
    cmp w14, #47
    b.eq operator_token
    cmp w14, #37

    mov x4, x17
    mov x5, #0
    ldrb w14, [x3]
    cmp w14, #45
    b.ne number_sign_done

    cmp x4, #18

    sub w7, w7, #48

    sub x12, x12, #2
    add x6, x13, x12, lsl #3
    ldp x4, x5, [x6]

    cmp w14, #43
    b.eq do_add
    cmp w14, #45
    b.eq do_sub
    cmp w14, #42
    b.eq do_mul
    cmp w14, #47
    b.eq do_divmod
    cmp w14, #37
