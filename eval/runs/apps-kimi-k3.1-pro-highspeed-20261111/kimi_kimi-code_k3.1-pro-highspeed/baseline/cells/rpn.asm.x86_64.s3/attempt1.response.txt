.equ INPUT_CAP, 65536

.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_CAP + 1
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    lea r15, [rip + input_buffer]
    lea r14, [rip + input_buffer]
    add r14, r12

next_line:
    cmp r15, r14
    jae exit_success

    mov rbx, r15
    mov rsi, rbx

find_newline:
    cmp rsi, r14
    jae last_line
    cmp byte ptr [rsi], 10
    je newline_found
    inc rsi
    jmp find_newline

newline_found:
    lea r15, [rsi + 1]
    mov rdi, rsi
    jmp begin_evaluation

last_line:
    mov rdi, r14
    mov r15, r14

begin_evaluation:
    mov rsi, rbx
    lea rbx, [rip + value_stack]

token_next:
    cmp rsi, rdi
    jae tokens_done
    mov al, byte ptr [rsi]
    cmp al, 32
    je separator
    cmp al, 9
    je separator

    mov rdx, rsi

token_scan:
    inc rsi
    cmp rsi, rdi
    jae token_scanned
    mov al, byte ptr [rsi]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    jmp token_scan

token_scanned:
    mov rcx, rsi
    sub rcx, rdx
    cmp rcx, 1
    jne integer_check

    mov al, byte ptr [rdx]
    cmp al, 43
    je have_operator
    cmp al, 45
    je have_operator
    cmp al, 42
    je have_operator
    cmp al, 47
    je have_operator
    cmp al, 37
    je have_operator

integer_check:
    xor r8d, r8d
    mov r9, rdx
    cmp byte ptr [r9], 45
    jne digits_count_ready
    mov r8d, 1
    inc r9
    dec rcx

digits_count_ready:
    test rcx, rcx
    jz evaluation_error
    cmp rcx, 18
    ja evaluation_error

    xor r10d, r10d

digit_loop:
    movzx eax, byte ptr [r9]
    sub eax, 48
    cmp eax, 9
    ja evaluation_error
    imul r10, r10, 10
    add r10, rax
    inc r9
    dec rcx
    jnz digit_loop

    test r8d, r8d
    jz store_literal
    neg r10

store_literal:
    mov qword ptr [rbx], r10
    add rbx, 8
    jmp token_next

separator:
    inc rsi
    jmp token_next

have_operator:
    lea rcx, [rip + value_stack + 16]
    cmp rbx, rcx
    jb evaluation_error

    mov r8, qword ptr [rbx - 8]
    mov r9, qword ptr [rbx - 16]
    sub rbx, 8

    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div

    mov r10d, 1
    jmp op_div_common

op_add:
    mov rax, r9
    add rax, r8
    jmp store_result

op_sub:
    mov rax, r9
    sub rax, r8
    jmp store_result

op_mul:
    mov rax, r9
    imul rax, r8
    jmp store_result

op_div:
    xor r10d, r10d

op_div_common:
    test r8, r8
    jz evaluation_error
    cmp r8, -1
    jne division_safe
    mov rcx, 0x8000000000000000
    cmp r9, rcx
    je evaluation_error

division_safe:
    mov rax, r9
    cqo
    idiv r8
    test r10d, r10d
    jz store_result
    mov rax, rdx

store_result:
    mov qword ptr [rbx - 8], rax
    jmp token_next

tokens_done:
    lea rax, [rip + value_stack + 8]
    cmp rbx, rax
    jne evaluation_error

    mov rax, qword ptr [rip + value_stack]
    lea rdi, [rip + number_buffer + 32]
    dec rdi
    mov byte ptr [rdi], 10

    xor r9d, r9d
    test rax, rax
    jns conversion_check
    mov r9d, 1
    neg rax

conversion_check:
    test rax, rax
    jnz conversion_loop
    dec rdi
    mov byte ptr [rdi], 48
    jmp conversion_sign

conversion_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz conversion_loop

conversion_sign:
    test r9d, r9d
    jz conversion_done
    dec rdi
    mov byte ptr [rdi], 45

conversion_done:
    lea rdx, [rip + number_buffer + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    jmp line_advance

evaluation_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall

line_advance:
    jmp next_line

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
error_message:
    .ascii "error\n"

.bss
.align 8
input_buffer:
    .space INPUT_CAP + 1

.align 8
value_stack:
    .space 8192

.align 8
number_buffer:
    .space 32
