.globl _start

.section .rodata
.align 4
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.align 4
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, x19
    adrp x21, output_buffer
    add x21, x21, :lo12:output_buffer
    adrp x25, base64_alphabet
    add x25, x25, :lo12:base64_alphabet
    mov x23, #0

full_group_loop:
    cmp x20, #3
    b.lo final_group

    ldrb w0, [x22]
    ldrb w1, [x22, #1]
    ldrb w2, [x22, #2]

    lsr w3, w0, #2
    ldrb w3, [x25, w3, uxtw]
    strb w3, [x21], #1

    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    ldrb w4, [x25, w4, uxtw]
    strb w4, [x21], #1

    and w3, w1, #15
    lsl w3, w3, #2
    lsr w5, w2, #6
    orr w3, w3, w5
    ldrb w3, [x25, w3, uxtw]
    strb w3, [x21], #1

    and w3, w2, #63
    ldrb w3, [x25, w3, uxtw]
    strb w3, [x21], #1

    add x22, x22, #3
    sub x20, x20, #3
    add x23, x23, #1
    cmp x23, #19
    b.ne full_group_loop

    mov w0, #10
    strb w0, [x21], #1
    mov x23, #0
    b full_group_loop

final_group:
    cbz x20, finish_last_line

    ldrb w0, [x22]

    lsr w3, w0, #2
    ldrb w3, [x25, w3, uxtw]
    strb w3, [x21], #1

    and w4, w0, #3
    lsl w4, w4, #4
    cmp x20, #1
    b.eq one_byte_left

    ldrb w1, [x22, #1]
    lsr w5, w1, #4
    orr w4, w4, w5
    ldrb w4, [x25, w4, uxtw]
    strb w4, [x21], #1

    and w3, w1, #15
    lsl w3, w3, #2
    ldrb w3, [x25, w3, uxtw]
    strb w3, [x21], #1

    mov w3, #61
    strb w3, [x21], #1
    b append_final_newline

one_byte_left:
    ldrb w4, [x25, w4, uxtw]
    strb w4, [x21], #1
    mov w3, #61
    strb w3, [x21], #1
    strb w3, [x21], #1

append_final_newline:
    mov w0, #10
    strb w0, [x21], #1
    b write_output

finish_last_line:
    cbz x23, write_output
    mov w0, #10
    strb w0, [x21], #1

write_output:
    adrp x1, output_buffer
    add x1, x1, :lo12:output_buffer
    sub x2, x21, x1
    cbz x2, exit_success
    mov x0, #1
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.align 4
input_buffer:
    .space 65536
.align 4
output_buffer:
    .space 66560
