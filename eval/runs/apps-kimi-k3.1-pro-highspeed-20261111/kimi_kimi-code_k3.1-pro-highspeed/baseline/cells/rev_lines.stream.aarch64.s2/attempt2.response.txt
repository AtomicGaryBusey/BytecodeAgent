.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output_buffer
    add x21, x21, :lo12:output_buffer
    mov x22, #0
    mov x24, #0
    mov x25, #0

scan_loop:
    cmp x24, x20
    b.eq input_end
    ldrb w9, [x19, x24]
    cmp w9, #10
    b.eq found_newline
    add x24, x24, #1
    b scan_loop

found_newline:
    sub x10, x24, x25
    bl append_reversed_line
    add x24, x24, #1
    mov x25, x24
    b scan_loop

input_end:
    cmp x25, x20
    b.eq write_setup
    sub x10, x20, x25
    bl append_reversed_line

write_setup:
    cbz x22, exit_success
    mov x23, #0

write_loop:
    mov x0, #1
    add x1, x21, x23
    sub x2, x22, x23
    mov x8, #64
    svc #0
    add x23, x23, x0
    cmp x23, x22
    b.lo write_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0
1:
    b 1b

append_reversed_line:
    add x11, x21, x22
    mov x12, #0
1:
    cmp x12, x10
    b.eq 2f
    sub x13, x10, x12
    sub x13, x13, #1
    add x13, x13, x25
    ldrb w14, [x19, x13]
    strb w14, [x11, x12]
    add x12, x12, #1
    b 1b
2:
    add x22, x22, x10
    mov w15, #10
    strb w15, [x21, x22]
    add x22, x22, #1
    ret

.bss
.balign 16
input_buffer:
    .space 65536
.balign 16
output_buffer:
