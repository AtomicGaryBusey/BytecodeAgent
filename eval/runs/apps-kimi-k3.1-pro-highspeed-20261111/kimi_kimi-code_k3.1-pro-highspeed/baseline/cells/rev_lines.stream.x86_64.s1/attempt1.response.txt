.globl _start
.text
_start:
    xor r12d, r12d
    lea r15, [rip + input_buffer]

read_input:
    mov edx, 65536
    sub rdx, r12
    jz process_input
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    syscall
    test rax, rax
    jle process_input
    add r12, rax
    jmp read_input

process_input:
    xor r13d, r13d

next_line:
    cmp r13, r12
    jae finished
    mov r14, r13

find_line_end:
    cmp r13, r12
    jae reverse_line
    cmp byte ptr [r15 + r13], 10
    je reverse_line
    inc r13
    jmp find_line_end

reverse_line:
    mov rbx, r13
    sub rbx, r14
    lea rsi, [r15 + r14]
    lea rdi, [rip + output_buffer]
    mov rcx, rbx
    test rcx, rcx
    jz append_newline

reverse_byte:
    mov al, byte ptr [rsi + rcx - 1]
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz reverse_byte

append_newline:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + output_buffer]
    sub rdi, rsi
    mov rdx, rdi
    mov edi, 1
    mov eax, 1
    syscall
    inc r13
    jmp next_line

finished:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
