.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x21, #3, lsl #14

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, input_complete
    add x20, x20, x0
    cmp x20, x21
    b.lo read_loop

input_complete:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, #0
    mov x24, x20
    adrp x25, base64_alphabet
    add x25, x25, :lo12:base64_alphabet

encode_loop:
    cmp x24, #3
    b.lo encode_tail

    ldrb w0, [x19], #1
    ldrb w1, [x19], #1
    ldrb w2, [x19], #1

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    and w5, w1, #15
    lsl w5, w5, #2
    lsr w6, w2, #6
    orr w5, w5, w6
    and w6, w2, #63

    ldrb w3, [x25, x3]
    ldrb w4, [x25, x4]
    ldrb w5, [x25, x5]
    ldrb w6, [x25, x6]

    orr w3, w3, w4, lsl #8
    orr w3, w3, w5, lsl #16
    orr w3, w3, w6, lsl #24
    str w3, [x22], #4

    add x23, x23, #4
    cmp x23, #76
    b.ne no_line_break
    mov w0, #10
    strb w0, [x22], #1
    mov x23, #0

no_line_break:
    sub x24, x24, #3
    b encode_loop

encode_tail:
    cbz x24, finish_full_groups
    cmp x24, #1
    b.eq tail_one_byte

    ldrb w0, [x19]
    ldrb w1, [x19, #1]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    and w5, w1, #15
    lsl w5, w5, #2
    ldrb w3, [x25, x3]
    ldrb w4, [x25, x4]
    ldrb w5, [x25, x5]
    movz w6, #61
    orr w3, w3, w4, lsl #8
    orr w3, w3, w5, lsl #16
    orr w3, w3, w6, lsl #24
    str w3, [x22], #4
    b append_final_newline

tail_one_byte:
    ldrb w0, [x19]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    ldrb w3, [x25, x3]
    ldrb w4, [x25, x4]
    movz w5, #0x3d3d
    orr w3, w3, w4, lsl #8
    orr w3, w3, w5, lsl #16
    str w3, [x22], #4
    b append_final_newline

finish_full_groups:
    cbz x23, write_output

append_final_newline:
    mov w0, #10
    strb w0, [x22], #1

write_output:
    sub x2, x22, x21
    cbz x2, exit_program
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0
