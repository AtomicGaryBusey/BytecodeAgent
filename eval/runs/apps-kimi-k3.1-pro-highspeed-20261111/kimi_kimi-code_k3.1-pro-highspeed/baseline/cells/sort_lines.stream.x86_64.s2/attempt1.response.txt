.section .rodata
newline_byte:
    .byte 10

.bss
    .align 16
input_buffer:
    .zero 32769
    .align 16
line_array:
    .zero 16000
    .align 16
temp_array:
    .zero 16000

.text
    .globl _start
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    lea r13, [rip + input_buffer]
    lea r14, [r13 + r12]
    xor r15d, r15d
    lea rbx, [rip + line_array]

parse_loop:
    cmp r13, r14
    jae parse_done
    mov rbp, r13

scan_line:
    cmp r13, r14
    jae final_line
    cmp byte ptr [r13], 10
    je newline_found
    inc r13
    jmp scan_line

newline_found:
    mov [rbx], rbp
    mov rax, r13
    sub rax, rbp
    mov [rbx + 8], rax
    add rbx, 16
    inc r15
    inc r13
    jmp parse_loop

final_line:
    mov [rbx], rbp
    mov rax, r14
    sub rax, rbp
    mov [rbx + 8], rax
    inc r15
    mov r13, r14
    jmp parse_loop

parse_done:
    mov r14, r15
    lea rbx, [rip + line_array]
    cmp r14, 2
    jb output_ready

    mov r12d, 1
    lea rbp, [rip + temp_array]

sort_pass:
    xor r13d, r13d

sort_group:
    cmp r13, r14
    jae sort_pass_done

    mov rax, r13
    shl rax, 4
    lea r8, [rbx + rax]
    lea r15, [rbp + rax]

    lea rcx, [r13 + r12]
    cmp rcx, r14
    cmova rcx, r14
    mov rax, rcx
    shl rax, 4
    lea r9, [rbx + rax]
    mov r10, r9

    lea rcx, [r13 + r12*2]
    cmp rcx, r14
    cmova rcx, r14
    mov rax, rcx
    shl rax, 4
    lea r11, [rbx + rax]

merge_loop:
    cmp r8, r9
    je copy_right
    cmp r10, r11
    je copy_left

    mov rdi, r8
    mov rsi, r10
    call compare_lines
    test eax, eax
    jg take_right

    mov rax, [r8]
    mov [r15], rax
    mov rax, [r8 + 8]
    mov [r15 + 8], rax
    add r8, 16
    add r15, 16
    jmp merge_loop

take_right:
    mov rax, [r10]
    mov [r15], rax
    mov rax, [r10 + 8]
    mov [r15 + 8], rax
    add r10, 16
    add r15, 16
    jmp merge_loop

copy_left:
    cmp r8, r9
    je copy_right
    mov rax, [r8]
    mov [r15], rax
    mov rax, [r8 + 8]
    mov [r15 + 8], rax
    add r8, 16
    add r15, 16
    jmp copy_left

copy_right:
    cmp r10, r11
    je group_done
    mov rax, [r10]
    mov [r15], rax
    mov rax, [r10 + 8]
    mov [r15 + 8], rax
    add r10, 16
    add r15, 16
    jmp copy_right

group_done:
    lea rax, [r12 + r12]
    add r13, rax
    jmp sort_group

sort_pass_done:
    mov rax, rbx
    mov rbx, rbp
    mov rbp, rax
    shl r12, 1
    cmp r12, r14
    jb sort_pass

output_ready:
    xor r12d, r12d

output_loop:
    cmp r12, r14
    jae program_exit
    mov rax, r12
    shl rax, 4
    lea r13, [rbx + rax]

    mov eax, 1
    mov edi, 1
    mov rsi, [r13]
    mov rdx, [r13 + 8]
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall

    inc r12
    jmp output_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov rcx, [rdi + 8]
    mov rdx, [rsi + 8]
    mov rdi, [rdi]
    mov rsi, [rsi]

compare_loop:
    test rcx, rcx
    jz first_exhausted
    test rdx, rdx
    jz first_greater
    mov al, [rdi]
    cmp al, [rsi]
    jb first_less
    ja first_greater
    inc rdi
    inc rsi
    dec rcx
    dec rdx
    jmp compare_loop

first_exhausted:
    test rdx, rdx
    jnz first_less
    xor eax, eax
    ret

first_less:
    mov eax, -1
    ret

first_greater:
    mov eax, 1
    ret
