.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall

    test rax, rax
    jz input_done

    mov rbx, rax
    lea rsi, [rip + input_buffer]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    inc r12

    cmp al, 10
    jne check_whitespace
    inc r13

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz next_byte
    mov r15d, 1
    inc r14
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec rbx
    jnz scan_loop
    jmp read_more

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    mov r9, 10
    xor r8d, r8d

decimal_digits:
    xor edx, edx
    div r9
    push rdx
    inc r8
    test rax, rax
    jnz decimal_digits

decimal_output:
    pop rdx
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    dec r8
    jnz decimal_output
    ret

.bss
.p2align 4
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
