    cmp w0, #0x22

    cmp w0, #0x66
    b.eq .Lvalue_false
    cmp w0, #0x6E
    b.eq .Lvalue_null
    cmp w0, #0x2D

    cmp w0, #0x66
    b.eq .Lstring_simple_escape
    cmp w0, #0x6E
