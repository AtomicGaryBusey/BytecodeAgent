.globl _start

.text
_start:
    xor r12d, r12d

.Lread_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    cmp r12, 65536
    jb .Lread_loop

.Lread_done:
    mov qword ptr [rip + input_len], r12
    mov qword ptr [rip + pos], 0

    call skip_ws
    call parse_value
    test eax, eax
    jz .Lerror
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne .Lerror

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.Lerror:
    lea rdi, [rip + out_buf]
    lea rsi, [rip + error_prefix]
    mov ecx, 9
    rep movsb

    mov rax, qword ptr [rip + pos]
    test rax, rax
    jnz .Lconvert_number
    mov byte ptr [rdi], 0x30
    inc rdi
    jmp .Lnumber_done

.Lconvert_number:
    lea r8, [rip + reverse_digits]
    xor ecx, ecx
    mov r9d, 10

.Ldigit_loop:
    xor edx, edx
    div r9
    add dl, 0x30
    mov byte ptr [r8 + rcx], dl
    inc ecx
    test rax, rax
    jnz .Ldigit_loop

.Lcopy_digits:
    dec ecx
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], dl
    inc rdi
    test ecx, ecx
    jnz .Lcopy_digits

.Lnumber_done:
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall


skip_ws:
    mov rax, qword ptr [rip + pos]

.Lws_loop:
    cmp rax, qword ptr [rip + input_len]
    jae .Lws_done
    lea rdx, [rip + input_buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, 0x20
    je .Lws_advance
    cmp cl, 0x09
    je .Lws_advance
    cmp cl, 0x0A
    je .Lws_advance
    cmp cl, 0x0D
    jne .Lws_done

.Lws_advance:
    inc rax
    jmp .Lws_loop

.Lws_done:
    mov qword ptr [rip + pos], rax
    ret


parse_value:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lvalue_fail
    lea rdx, [rip + input_buf]
    movzx ecx, byte ptr [rdx + rax]

    cmp cl, 0x7B
    je .Lvalue_object
    cmp cl, 0x5B
    je .Lvalue_array
    cmp cl, 0x22
    je .Lvalue_string
    cmp cl, 0x2D
    je .Lvalue_number
    cmp cl, 0x30
    jb .Lvalue_literals
    cmp cl, 0x39
    jbe .Lvalue_number

.Lvalue_literals:
    cmp cl, 0x74
    je .Lvalue_true
    cmp cl, 0x66
    je .Lvalue_false
    cmp cl, 0x6E
    je .Lvalue_null

.Lvalue_fail:
    xor eax, eax
    ret

.Lvalue_object:
    jmp parse_object
.Lvalue_array:
    jmp parse_array
.Lvalue_string:
    jmp parse_string
.Lvalue_number:
    jmp parse_number
.Lvalue_true:
    lea rsi, [rip + literal_true]
    mov edx, 4
    jmp parse_literal
.Lvalue_false:
    lea rsi, [rip + literal_false]
    mov edx, 5
    jmp parse_literal
.Lvalue_null:
    lea rsi, [rip + literal_null]
    mov edx, 4
    jmp parse_literal


parse_literal:
    xor ecx, ecx

.Lliteral_loop:
    cmp ecx, edx
    jae .Lliteral_success
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lliteral_fail
    lea r8, [rip + input_buf]
    mov r9b, byte ptr [r8 + rax]
    cmp r9b, byte ptr [rsi + rcx]
    jne .Lliteral_fail
    inc rax
    mov qword ptr [rip + pos], rax
    inc ecx
    jmp .Lliteral_loop

.Lliteral_success:
    mov eax, 1
    ret
.Lliteral_fail:
    xor eax, eax
    ret


parse_string:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lstring_fail
    lea r8, [rip + input_buf]
    cmp byte ptr [r8 + rax], 0x22
    jne .Lstring_fail
    inc rax
    mov qword ptr [rip + pos], rax

.Lstring_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lstring_fail
    lea r8, [rip + input_buf]
    movzx ecx, byte ptr [r8 + rax]

    cmp cl, 0x22
    je .Lstring_close
    cmp cl, 0x5C
    je .Lstring_escape
    cmp cl, 0x20
    jb .Lstring_fail

    inc rax
    mov qword ptr [rip + pos], rax
    jmp .Lstring_loop

.Lstring_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret

.Lstring_escape:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lstring_fail
    movzx ecx, byte ptr [r8 + rax]

    cmp cl, 0x22
    je .Lsimple_escape
    cmp cl, 0x5C
    je .Lsimple_escape
    cmp cl, 0x2F
    je .Lsimple_escape
    cmp cl, 0x62
    je .Lsimple_escape
    cmp cl, 0x66
    je .Lsimple_escape
    cmp cl, 0x6E
    je .Lsimple_escape
    cmp cl, 0x72
    je .Lsimple_escape
    cmp cl, 0x74
    je .Lsimple_escape
    cmp cl, 0x75
    jne .Lstring_fail

    inc rax
    mov qword ptr [rip + pos], rax
    mov r10d, 4

.Lunicode_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lstring_fail
    lea r8, [rip + input_buf]
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x30
    jb .Lstring_fail
    cmp cl, 0x39
    jbe .Lhex_valid
    or cl, 0x20
    cmp cl, 0x61
    jb .Lstring_fail
    cmp cl, 0x66
    ja .Lstring_fail

.Lhex_valid:
    inc rax
    mov qword ptr [rip + pos], rax
    dec r10d
    jnz .Lunicode_loop
    jmp .Lstring_loop

.Lsimple_escape:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .Lstring_loop

.Lstring_fail:
    xor eax, eax
    ret


parse_number:
    mov rax, qword ptr [rip + pos]
    lea r8, [rip + input_buf]
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x2D
    jne .Linteger_part
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_fail
    movzx ecx, byte ptr [r8 + rax]

.Linteger_part:
    cmp cl, 0x30
    je .Lzero_integer
    cmp cl, 0x31
    jb .Lnumber_fail
    cmp cl, 0x39
    ja .Lnumber_fail

.Linteger_digits:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_success
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x30
    jb .Lfraction_part
    cmp cl, 0x39
    jbe .Linteger_digits
    jmp .Lfraction_part

.Lzero_integer:
    inc rax
    mov qword ptr [rip + pos], rax

.Lfraction_part:
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_success
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x2E
    jne .Lexponent_part

    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_fail
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x30
    jb .Lnumber_fail
    cmp cl, 0x39
    ja .Lnumber_fail

.Lfraction_digits:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_success
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x30
    jb .Lexponent_part
    cmp cl, 0x39
    jbe .Lfraction_digits

.Lexponent_part:
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_success
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x65
    je .Lexponent_marker
    cmp cl, 0x45
    jne .Lnumber_success

.Lexponent_marker:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_fail
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x2B
    je .Lexponent_sign
    cmp cl, 0x2D
    jne .Lexponent_require_digit

.Lexponent_sign:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_fail
    movzx ecx, byte ptr [r8 + rax]

.Lexponent_require_digit:
    cmp cl, 0x30
    jb .Lnumber_fail
    cmp cl, 0x39
    ja .Lnumber_fail

.Lexponent_digits:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae .Lnumber_success
    movzx ecx, byte ptr [r8 + rax]
    cmp cl, 0x30
    jb .Lnumber_success
    cmp cl, 0x39
    jbe .Lexponent_digits

.Lnumber_success:
    mov eax, 1
    ret
.Lnumber_fail:
    xor eax, eax
    ret


parse_array:
    mov rax, qword ptr [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws

    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Larray_fail
    lea rdx, [rip + input_buf]
    cmp byte ptr [rdx + rax], 0x5D
    je .Larray_close

.Larray_value:
    call parse_value
    test eax, eax
    jz .Larray_fail
    call skip_ws

    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Larray_fail
    lea rdx, [rip + input_buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, 0x5D
    je .Larray_close
    cmp cl, 0x2C
    jne .Larray_fail

    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    jmp .Larray_value

.Larray_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.Larray_fail:
    xor eax, eax
    ret


parse_object:
    mov rax, qword ptr [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws

    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lobject_fail
    lea rdx, [rip + input_buf]
    cmp byte ptr [rdx + rax], 0x7D
    je .Lobject_close

.Lobject_member:
    call parse_string
    test eax, eax
    jz .Lobject_fail
    call skip_ws

    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lobject_fail
    lea rdx, [rip + input_buf]
    cmp byte ptr [rdx + rax], 0x3A
    jne .Lobject_fail

    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    call parse_value
    test eax, eax
    jz .Lobject_fail
    call skip_ws

    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jae .Lobject_fail
    lea rdx, [rip + input_buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, 0x7D
    je .Lobject_close
    cmp cl, 0x2C
    jne .Lobject_fail

    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    jmp .Lobject_member

.Lobject_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.Lobject_fail:
    xor eax, eax
    ret


.section .rodata
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
.balign 16
input_len:
    .zero 8
pos:
    .zero 8
reverse_digits:
    .zero 16
out_buf:
    .zero 64
input_buf:
    .zero 65536
