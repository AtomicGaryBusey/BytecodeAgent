.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x22, #0

.Lread_loop:
    mov x0, #0
    add x1, x19, x22
    movz x2, #1, lsl #16
    sub x2, x2, x22
    cbz x2, .Lread_done
    mov x8, #63
    svc #0
    cbz x0, .Lread_done
    add x22, x22, x0
    b .Lread_loop

.Lread_done:
    add x20, x19, x22
    mov x21, x19

    bl skip_ws
    bl parse_value
    cbnz w0, .Lerror
    bl skip_ws
    cmp x21, x20
    b.ne .Lerror

    mov x0, #1
    adr x1, msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.Lerror:
    sub x22, x21, x19
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    adr x0, err_prefix
    ldr x5, [x0]
    str x5, [x1]
    ldrb w5, [x0, #8]
    strb w5, [x1, #8]

    add x2, x1, #9
    mov x3, x2
    mov x6, #10
    cbnz x22, .Ldecimal_loop
    mov w5, #48
    strb w5, [x2], #1
    b .Ldecimal_ready

.Ldecimal_loop:
    udiv x7, x22, x6
    msub x8, x7, x6, x22
    add w8, w8, #48
    strb w8, [x2], #1
    mov x22, x7
    cbnz x22, .Ldecimal_loop

    sub x4, x2, #1
.Lreverse_loop:
    cmp x3, x4
    b.hs .Ldecimal_ready
    ldrb w5, [x3]
    ldrb w7, [x4]
    strb w7, [x3]
    strb w5, [x4]
    add x3, x3, #1
    sub x4, x4, #1
    b .Lreverse_loop

.Ldecimal_ready:
    mov w5, #10
    strb w5, [x2], #1
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
.Lws_loop:
    cmp x21, x20
    b.eq .Lws_done
    ldrb w0, [x21]
    cmp w0, #32
    b.eq .Lws_consume
    cmp w0, #9
    b.eq .Lws_consume
    cmp w0, #10
    b.eq .Lws_consume
    cmp w0, #13
    b.ne .Lws_done
.Lws_consume:
    add x21, x21, #1
    b .Lws_loop
.Lws_done:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    bl skip_ws
    cmp x21, x20
    b.eq .Lpv_invalid
    ldrb w0, [x21]

    cmp w0, #123
    b.eq .Lpv_object
    cmp w0, #91
    b.eq .Lpv_array
    cmp w0, #34
    b.eq .Lpv_string
    cmp w0, #116
    b.eq .Lpv_true
    cmp w0, #102
    b.eq .Lpv_false
    cmp w0, #110
    b.eq .Lpv_null
    cmp w0, #45
    b.eq .Lpv_number
    cmp w0, #48
    b.lo .Lpv_invalid
    cmp w0, #57
    b.ls .Lpv_number
    b .Lpv_invalid

.Lpv_object:
    bl parse_object
    b .Lpv_done
.Lpv_array:
    bl parse_array
    b .Lpv_done
.Lpv_string:
    bl parse_string
    b .Lpv_done
.Lpv_true:
    adr x0, lit_true
    mov x1, #4
    bl parse_literal
    b .Lpv_done
.Lpv_false:
    adr x0, lit_false
    mov x1, #5
    bl parse_literal
    b .Lpv_done
.Lpv_null:
    adr x0, lit_null
    mov x1, #4
    bl parse_literal
    b .Lpv_done
.Lpv_number:
    bl parse_number
    b .Lpv_done
.Lpv_invalid:
    mov w0, #1
.Lpv_done:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
.Literal_loop:
    cbz x1, .Literal_success
    cmp x21, x20
    b.eq .Literal_fail
    ldrb w2, [x21]
    ldrb w3, [x0], #1
    cmp w2, w3
    b.ne .Literal_fail
    add x21, x21, #1
    sub x1, x1, #1
    b .Literal_loop
.Literal_success:
    mov w0, #0
    ret
.Literal_fail:
    mov w0, #1
    ret

parse_string:
    cmp x21, x20
    b.eq .Lstr_fail
    ldrb w0, [x21]
    cmp w0, #34
    b.ne .Lstr_fail
    add x21, x21, #1

.Lstr_loop:
    cmp x21, x20
    b.eq .Lstr_fail
    ldrb w0, [x21]
    cmp w0, #34
    b.eq .Lstr_close
    cmp w0, #92
    b.eq .Lstr_escape
    cmp w0, #32
    b.lo .Lstr_fail
    add x21, x21, #1
    b .Lstr_loop

.Lstr_close:
    add x21, x21, #1
    mov w0, #0
    ret

.Lstr_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lstr_fail
    ldrb w0, [x21]
    cmp w0, #117
    b.eq .Lstr_unicode
    cmp w0, #34
    b.eq .Lstr_escape_ok
    cmp w0, #92
    b.eq .Lstr_escape_ok
    cmp w0, #47
    b.eq .Lstr_escape_ok
    cmp w0, #98
    b.eq .Lstr_escape_ok
    cmp w0, #102
    b.eq .Lstr_escape_ok
    cmp w0, #110
    b.eq .Lstr_escape_ok
    cmp w0, #114
    b.eq .Lstr_escape_ok
    cmp w0, #116
    b.eq .Lstr_escape_ok
    b .Lstr_fail

.Lstr_escape_ok:
    add x21, x21, #1
    b .Lstr_loop

.Lstr_unicode:
    add x21, x21, #1
    mov w1, #4
.Lstr_hex_loop:
    cmp x21, x20
    b.eq .Lstr_fail
    ldrb w0, [x21]
    sub w2, w0, #48
    cmp w2, #9
    b.ls .Lstr_hex_ok
    sub w2, w0, #65
    cmp w2, #5
    b.ls .Lstr_hex_ok
    sub w2, w0, #97
    cmp w2, #5
    b.hi .Lstr_fail
.Lstr_hex_ok:
    add x21, x21, #1
    subs w1, w1, #1
    b.ne .Lstr_hex_loop
    b .Lstr_loop

.Lstr_fail:
    mov w0, #1
    ret

parse_number:
    ldrb w0, [x21]
    cmp w0, #45
    b.ne .Lnum_int
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_fail

.Lnum_int:
    cmp x21, x20
    b.eq .Lnum_fail
    ldrb w0, [x21]
    cmp w0, #48
    b.eq .Lnum_zero
    sub w1, w0, #49
    cmp w1, #8
    b.hi .Lnum_fail
    add x21, x21, #1

.Lnum_int_loop:
    cmp x21, x20
    b.eq .Lnum_success
    ldrb w0, [x21]
    sub w1, w0, #48
    cmp w1, #9
    b.hi .Lnum_after_int
    add x21, x21, #1
    b .Lnum_int_loop

.Lnum_zero:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_success
    ldrb w0, [x21]

.Lnum_after_int:
    cmp w0, #46
    b.eq .Lnum_fraction
    cmp w0, #101
    b.eq .Lnum_exponent
    cmp w0, #69
    b.eq .Lnum_exponent
    b .Lnum_success

.Lnum_fraction:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_fail
    ldrb w0, [x21]
    sub w1, w0, #48
    cmp w1, #9
    b.hi .Lnum_fail

.Lnum_fraction_loop:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_success
    ldrb w0, [x21]
    sub w1, w0, #48
    cmp w1, #9
    b.ls .Lnum_fraction_loop
    cmp w0, #101
    b.eq .Lnum_exponent
    cmp w0, #69
    b.eq .Lnum_exponent
    b .Lnum_success

.Lnum_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_fail
    ldrb w0, [x21]
    cmp w0, #43
    b.eq .Lnum_exp_sign
    cmp w0, #45
    b.ne .Lnum_exp_need_digit
.Lnum_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_fail
    ldrb w0, [x21]
.Lnum_exp_need_digit:
    sub w1, w0, #48
    cmp w1, #9
    b.hi .Lnum_fail

.Lnum_exp_loop:
    add x21, x21, #1
    cmp x21, x20
    b.eq .Lnum_success
    ldrb w0, [x21]
    sub w1, w0, #48
    cmp w1, #9
    b.ls .Lnum_exp_loop

.Lnum_success:
    mov w0, #0
    ret
.Lnum_fail:
    mov w0, #1
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.eq .Larray_fail
    ldrb w0, [x21]
    cmp w0, #93
    b.eq .Larray_empty

.Larray_value:
    bl parse_value
    cbnz w0, .Larray_fail
    bl skip_ws
    cmp x21, x20
    b.eq .Larray_fail
    ldrb w0, [x21]
    cmp w0, #93
    b.eq .Larray_close
    cmp w0, #44
    b.ne .Larray_fail
    add x21, x21, #1
    bl skip_ws
    b .Larray_value

.Larray_empty:
.Larray_close:
    add x21, x21, #1
    mov w0, #0
    b .Larray_done
.Larray_fail:
    mov w0, #1
.Larray_done:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.eq .Lobject_fail
    ldrb w0, [x21]
    cmp w0, #125
    b.eq .Lobject_empty

.Lobject_member:
    bl parse_string
    cbnz w0, .Lobject_fail
    bl skip_ws
    cmp x21, x20
    b.eq .Lobject_fail
    ldrb w0, [x21]
    cmp w0, #58
    b.ne .Lobject_fail
    add x21, x21, #1
    bl parse_value
    cbnz w0, .Lobject_fail
    bl skip_ws
    cmp x21, x20
    b.eq .Lobject_fail
    ldrb w0, [x21]
    cmp w0, #125
    b.eq .Lobject_close
    cmp w0, #44
    b.ne .Lobject_fail
    add x21, x21, #1
    bl skip_ws
    b .Lobject_member

.Lobject_empty:
.Lobject_close:
    add x21, x21, #1
    mov w0, #0
    b .Lobject_done
.Lobject_fail:
    mov w0, #1
.Lobject_done:
    ldp x29, x30, [sp], #16
    ret
