.globl _start

.text
_start:
    lea rbp, [rip + input_buffer]
    xor r13d, r13d

read_input:
    mov edx, 65536
    sub rdx, r13
    jz dump_begin

    xor eax, eax
    xor edi, edi
    lea rsi, [rbp + r13]
    syscall

    test rax, rax
    jz dump_begin
    add r13, rax
    jmp read_input

dump_begin:
    lea rbx, [rip + hex_digits]
    xor r12d, r12d

next_row:
    cmp r12, r13
    jae print_total

    mov r14, r13
    sub r14, r12
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16
row_size_ready:

    lea rdi, [rip + output_buffer]
    mov ecx, 80
    mov al, 32
    rep stosb
    mov byte ptr [rip + output_buffer + 60], 124

    mov eax, r12d
    lea rdi, [rip + output_buffer + 8]
    mov ecx, 8
offset_digits:
    dec rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    shr eax, 4
    dec ecx
    jnz offset_digits

    lea rsi, [rbp + r12]
    lea rdi, [rip + output_buffer + 10]
    xor r15d, r15d

hex_columns:
    cmp r15, r14
    jae hex_column_done

    movzx eax, byte ptr [rsi + r15]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl

hex_column_done:
    add rdi, 3
    inc r15
    cmp r15, 8
    jne no_middle_space
    add rdi, 1
no_middle_space:
    cmp r15, 16
    jne hex_columns

    lea rdi, [rip + output_buffer + 61]
    xor r15d, r15d

ascii_column:
    cmp r15, r14
    jae ascii_done
    mov al, byte ptr [rsi + r15]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    jbe ascii_store
ascii_dot:
    mov al, 46
ascii_store:
    mov byte ptr [rdi + r15], al
    inc r15
    jmp ascii_column

ascii_done:
    mov byte ptr [rdi + r15], 124
    mov byte ptr [rdi + r15 + 1], 10

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    lea rdx, [r14 + 63]
    syscall

    add r12, r14
    jmp next_row

print_total:
    mov eax, r13d
    lea rdi, [rip + output_buffer + 8]
    mov ecx, 8
total_digits:
    dec rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    shr eax, 4
    dec ecx
    jnz total_digits

    mov byte ptr [rip + output_buffer + 8], 10

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.p2align 4
input_buffer:
    .zero 65536
output_buffer:
    .zero 80
