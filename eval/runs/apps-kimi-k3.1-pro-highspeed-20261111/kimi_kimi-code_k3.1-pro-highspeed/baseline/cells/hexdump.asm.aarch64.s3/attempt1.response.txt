.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    cmp x20, x21
    b.hs 1f
    sub x2, x21, x20
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 1f
    add x20, x20, x0
    b read_loop

1:
    mov x22, #0
    mov x23, x19

row_loop:
    cmp x22, x20
    b.hs final_output

    sub x24, x20, x22
    mov x25, #16
    cmp x24, #16
    b.hs 2f
    mov x25, x24
2:
    mov x0, x23
    mov x1, x25
    mov x2, x22
    bl format_row

    mov x2, x1
    mov x1, x0
    mov x0, #1
    mov x8, #64
    svc #0

    add x23, x23, x25
    add x22, x22, x25
    b row_loop

final_output:
    adrp x3, row_buf
    add x3, x3, :lo12:row_buf
    adrp x6, hex_digits
    add x6, x6, :lo12:hex_digits
    mov x4, #28
    mov x5, x3

3:
    lsr x7, x19, x4
    and x7, x7, #0xf
    ldrb w7, [x6, x7]
    strb w7, [x5], #1
    subs x4, x4, #4
    b.pl 3b

    mov w7, #10
    strb w7, [x5]

    mov x0, #1
    mov x1, x3
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

format_row:
    adrp x3, row_buf
    add x3, x3, :lo12:row_buf
    adrp x6, hex_digits
    add x6, x6, :lo12:hex_digits

    mov x4, #28
    mov x5, x3
1:
    lsr x7, x2, x4
    and x7, x7, #0xf
    ldrb w7, [x6, x7]
    strb w7, [x5], #1
    subs x4, x4, #4
    b.pl 1b

    mov w7, #0x20
    strb w7, [x5], #1
    strb w7, [x5], #1

    add x8, x3, #61
    mov x9, #0

2:
    cmp x9, x1
    b.hs 3f

    ldrb w10, [x0, x9]
    lsr w11, w10, #4
    ldrb w11, [x6, x11]
    strb w11, [x5]
    and w11, w10, #0xf
    ldrb w11, [x6, x11]
    strb w11, [x5, #1]
    strb w7, [x5, #2]

    cmp w10, #0x20
    b.lo 4f
    cmp w10, #0x7e
    b.hi 4f
    strb w10, [x8], #1
    b 6f

4:
    mov w11, #0x2e
    strb w11, [x8], #1
    b 6f

3:
    strb w7, [x5]
    strb w7, [x5, #1]
    strb w7, [x5, #2]

6:
    add x5, x5, #3
    cmp x9, #7
    b.ne 7f
    strb w7, [x5], #1
7:
    add x9, x9, #1
    cmp x9, #16
    b.lo 2b

    strb w7, [x5], #1
    mov w10, #0x7c
    strb w10, [x5], #1
    strb w10, [x8], #1
    mov w10, #10
    strb w10, [x8], #1

    sub x1, x8, x3
    mov x0, x3
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
row_buf:
    .zero 80
