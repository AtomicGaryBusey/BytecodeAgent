.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae input_done

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall

    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    mov r15, r12
    xor r13d, r13d
    xor r14d, r14d

line_loop:
    cmp r13, r15
    jae output_ready

    mov rbx, r13

scan_line:
    cmp r13, r15
    jae reverse_line
    lea rsi, [rip + input_buf]
    cmp byte ptr [rsi + r13], 10
    je reverse_line
    inc r13
    jmp scan_line

reverse_line:
    mov rcx, r13
    sub rcx, rbx

    lea rdi, [rip + output_buf]
    add rdi, r14

    lea rsi, [rip + input_buf]
    add rsi, r13

    mov rdx, rcx

reverse_bytes:
    test rdx, rdx
    jz line_reversed
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rdx
    jmp reverse_bytes

line_reversed:
    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + output_buf]
    sub rdi, rax
    mov r14, rdi

    cmp r13, r15
    jae line_loop
    inc r13
    jmp line_loop

output_ready:
    xor r12d, r12d

write_more:
    cmp r12, r14
    jae exit_success

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    add rsi, r12
    mov rdx, r14
    sub rdx, r12
    syscall

    test rax, rax
    jle exit_success
    add r12, rax
    jmp write_more

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 65537
