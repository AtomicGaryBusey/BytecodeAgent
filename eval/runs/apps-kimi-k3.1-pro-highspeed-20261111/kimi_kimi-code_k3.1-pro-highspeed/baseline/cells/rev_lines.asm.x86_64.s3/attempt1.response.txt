.globl _start

.text
_start:
    lea r12, [rip + input_buffer]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    lea rdx, [rip + input_buffer + 65536]
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_loop

input_done:
    lea r13, [rip + input_buffer]

line_loop:
    cmp r13, r12
    je program_exit

    mov r14, r13
find_line_end:
    cmp r14, r12
    je reverse_line
    cmp byte ptr [r14], 10
    je reverse_line
    inc r14
    jmp find_line_end

reverse_line:
    mov r15, r14
    sub r15, r13
    mov rsi, r13
    lea rdi, [r14 - 1]

reverse_loop:
    cmp rsi, rdi
    jae write_line
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp reverse_loop

write_line:
    test r15, r15
    jz write_newline
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, r15
    syscall

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall

    cmp r14, r12
    je program_exit
    lea r13, [r14 + 1]
    jmp line_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline_byte:
    .byte 10

.bss
.align 16
input_buffer:
    .zero 65536
