    .text
    .globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    movz x21, #1, lsl #16
    mov x19, #0

read_loop:
    cmp x19, x21
    b.hs read_done
    mov x0, #0
    add x1, x20, x19
    sub x2, x21, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits
    mov x22, #0

row_loop:
    cmp x22, x19
    b.hs final_offset

    sub x24, x19, x22
    cmp x24, #16
    b.ls row_length_ready
    mov x24, #16
row_length_ready:
    add x25, x20, x22
    adrp x26, out_buf
    add x26, x26, :lo12:out_buf

    mov x3, x22
    mov x4, #28
    mov x5, #8
row_offset_hex:
    lsr x6, x3, x4
    and x6, x6, #0xf
    ldrb w6, [x23, x6]
    strb w6, [x26], #1
    subs x5, x5, #1
    sub x4, x4, #4
    b.ne row_offset_hex

    mov w0, #32
    strb w0, [x26], #1
    strb w0, [x26], #1
    mov x27, #0

slot_loop:
    cmp x27, x24
    b.hs blank_slot
    ldrb w0, [x25, x27]
    lsr w1, w0, #4
    ldrb w1, [x23, x1]
    strb w1, [x26], #1
    and w1, w0, #0xf
    ldrb w1, [x23, x1]
    strb w1, [x26], #1
    mov w1, #32
    strb w1, [x26], #1
    b slot_done

blank_slot:
    mov w0, #32
    strb w0, [x26], #1
    strb w0, [x26], #1
    strb w0, [x26], #1

slot_done:
    add x27, x27, #1
    cmp x27, #8
    b.ne no_extra_space
    mov w0, #32
    strb w0, [x26], #1
no_extra_space:
    cmp x27, #16
    b.lo slot_loop

    mov w0, #32
    strb w0, [x26], #1
    mov w0, #124
    strb w0, [x26], #1

    mov x27, #0
ascii_loop:
    cmp x27, x24
    b.hs ascii_done
    ldrb w0, [x25, x27]
    cmp w0, #0x20
    b.lo ascii_dot
    cmp w0, #0x7e
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w0, #46
ascii_store:
    strb w0, [x26], #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x26], #1
    mov w0, #10
    strb w0, [x26], #1

    mov x0, #1
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    sub x2, x26, x1
    mov x8, #64
    svc #0

    add x22, x22, x24
    b row_loop

final_offset:
    adrp x26, out_buf
    add x26, x26, :lo12:out_buf
    mov x3, x19
    mov x4, #28
    mov x5, #8
final_hex_loop:
    lsr x6, x3, x4
    and x6, x6, #0xf
    ldrb w6, [x23, x6]
    strb w6, [x26], #1
    subs x5, x5, #1
    sub x4, x4, #4
    b.ne final_hex_loop

    mov w0, #10
    strb w0, [x26], #1
    mov x0, #1
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    sub x2, x26, x1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

    .section .rodata
hex_digits:
    .ascii "0123456789abcdef"

    .bss
    .balign 16
input_buf:
    .zero 65536
out_buf:
    .zero 80
