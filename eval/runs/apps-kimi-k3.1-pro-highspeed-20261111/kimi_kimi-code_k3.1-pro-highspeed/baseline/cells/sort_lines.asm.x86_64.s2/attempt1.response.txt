.globl _start

.text
_start:
    lea r15, [rip + input_buf]
    xor r12d, r12d

read_loop:
    cmp r12, 32768
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success

    lea rbp, [rip + entries]
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d

parse_loop:
    cmp r13, r12
    jae parse_done
    cmp byte ptr [r15 + r13], 10
    jne parse_next

    lea rax, [r15 + r14]
    mov qword ptr [rbp], rax
    mov rcx, r13
    sub rcx, r14
    mov qword ptr [rbp + 8], rcx
    add rbp, 16
    inc rbx
    lea r14, [r13 + 1]

parse_next:
    inc r13
    jmp parse_loop

parse_done:
    cmp r14, r12
    jae parse_finish
    lea rax, [r15 + r14]
    mov qword ptr [rbp], rax
    mov rcx, r12
    sub rcx, r14
    mov qword ptr [rbp + 8], rcx
    inc rbx

parse_finish:
    test rbx, rbx
    jz exit_success

    lea r14, [rip + entries]
    mov r13d, 1

sort_outer:
    cmp r13, rbx
    jae sorted

    mov rax, r13
    shl rax, 4
    mov r12, qword ptr [r14 + rax]
    mov rbp, qword ptr [r14 + rax + 8]

    xor r8d, r8d
    mov r9, r13

search_loop:
    cmp r8, r9
    jae search_done
    lea r10, [r8 + r9]
    shr r10, 1

    mov rax, r10
    shl rax, 4
    mov rdi, qword ptr [r14 + rax]
    mov rsi, qword ptr [r14 + rax + 8]
    mov rdx, r12
    mov rcx, rbp
    call compare_lines

    test eax, eax
    jg key_before_mid
    lea r8, [r10 + 1]
    jmp search_loop

key_before_mid:
    mov r9, r10
    jmp search_loop

search_done:
    mov r10, r13

shift_loop:
    cmp r10, r8
    jbe shift_done
    mov rax, r10
    shl rax, 4
    lea rdx, [r14 + rax]
    mov rdi, qword ptr [rdx - 16]
    mov qword ptr [rdx], rdi
    mov rdi, qword ptr [rdx - 8]
    mov qword ptr [rdx + 8], rdi
    dec r10
    jmp shift_loop

shift_done:
    mov rax, r8
    shl rax, 4
    mov qword ptr [r14 + rax], r12
    mov qword ptr [r14 + rax + 8], rbp
    inc r13
    jmp sort_outer

sorted:
    mov r13, rbx
    lea r15, [rip + entries]

output_loop:
    test r13, r13
    jz exit_success
    mov rsi, qword ptr [r15]
    mov rdx, qword ptr [r15 + 8]
    test rdx, rdx
    jz output_newline

    mov eax, 1
    mov edi, 1
    syscall

output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    add r15, 16
    dec r13
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r11, rsi
    cmp r11, rcx
    jbe 1f
    mov r11, rcx

1:
    test r11, r11
    jz 3f

2:
    mov al, byte ptr [rdi]
    cmp al, byte ptr [rdx]
    jb 4f
    ja 5f
    inc rdi
    inc rdx
    dec r11
    jnz 2b

3:
    cmp rsi, rcx
    jb 4f
    ja 5f
    xor eax, eax
    ret

4:
    mov eax, -1
    ret

5:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buf:
    .zero 32768
.align 16
entries:
    .zero 16000
