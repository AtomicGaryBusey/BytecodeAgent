.text
.globl _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #65536
    add x21, x21, #1

read_loop:
    cmp x20, x21
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x0, #0

find_separator:
    cmp x0, x20
    b.hs no_separator
    ldrb w1, [x19, x0]
    cmp w1, #33
    b.eq separator_found
    add x0, x0, #1
    b find_separator

separator_found:
    add x21, x19, x0
    add x21, x21, #1
    sub x22, x20, x0
    sub x22, x22, #1
    b separator_ready

no_separator:
    mov x0, x20
    add x21, x19, x20
    mov x22, #0

separator_ready:
    mov x23, x0
    mov x7, x23
    adrp x5, bracket_stack
    add x5, x5, :lo12:bracket_stack
    adrp x6, jump_table
    add x6, x6, :lo12:jump_table
    mov x0, #0
    mov x1, #0
    mov x2, #0

scan_loop:
    cmp x0, x7
    b.hs scan_done
    ldrb w3, [x19, x0]
    add x0, x0, #1

    cmp w3, #43
    b.eq store_command
    cmp w3, #45
    b.eq store_command
    cmp w3, #62
    b.eq store_command
    cmp w3, #60
    b.eq store_command
    cmp w3, #46
    b.eq store_command
    cmp w3, #44
    b.eq store_command
    cmp w3, #91
    b.eq scan_open
    cmp w3, #93
    b.eq scan_close
    b scan_loop

store_command:
    strb w3, [x19, x1]
    add x1, x1, #1
    b scan_loop

scan_open:
    cmp x2, #1000
    b.hs invalid_program
    strb w3, [x19, x1]
    str w1, [x5, x2, lsl #2]
    add x2, x2, #1
    add x1, x1, #1
    b scan_loop

scan_close:
    cmp x2, #0
    b.eq invalid_program
    strb w3, [x19, x1]
    sub x2, x2, #1
    ldr w4, [x5, x2, lsl #2]
    add w3, w4, #1
    str w3, [x6, x1, lsl #2]
    add w3, w1, #1
    str w3, [x6, x4, lsl #2]
    add x1, x1, #1
    b scan_loop

scan_done:
    cbnz x2, invalid_program
    mov x20, x1

    adrp x23, tape
    add x23, x23, :lo12:tape
    mov x24, x23
    adrp x25, output_buffer
    add x25, x25, :lo12:output_buffer
    mov x26, x25
    adrp x27, jump_table
    add x27, x27, :lo12:jump_table
    mov x28, #0

execute_loop:
    cmp x28, x20
    b.hs program_done
    ldrb w0, [x19, x28]
    add x28, x28, #1

    cmp w0, #43
    b.eq exec_plus
    cmp w0, #45
    b.eq exec_minus
    cmp w0, #62
    b.eq exec_right
    cmp w0, #60
    b.eq exec_left
    cmp w0, #46
    b.eq exec_output
    cmp w0, #44
    b.eq exec_input
    cmp w0, #91
    b.eq exec_open

exec_close:
    ldrb w1, [x24]
    cbz w1, execute_loop
    sub x2, x28, #1
    ldr w1, [x27, x2, lsl #2]
    mov x28, x1
    b execute_loop

exec_plus:
    ldrb w1, [x24]
    add w1, w1, #1
    strb w1, [x24]
    b execute_loop

exec_minus:
    ldrb w1, [x24]
    sub w1, w1, #1
    strb w1, [x24]
    b execute_loop

exec_right:
    add x24, x24, #1
    b execute_loop

exec_left:
    sub x24, x24, #1
    b execute_loop

exec_output:
    ldrb w1, [x24]
    strb w1, [x26], #1
    b execute_loop

exec_input:
    cbz x22, input_exhausted
    ldrb w1, [x21], #1
    sub x22, x22, #1
    strb w1, [x24]
    b execute_loop

input_exhausted:
    strb wzr, [x24]
    b execute_loop

exec_open:
    ldrb w1, [x24]
    cbnz w1, execute_loop
    sub x2, x28, #1
    ldr w1, [x27, x2, lsl #2]
    mov x28, x1
    b execute_loop

program_done:
    sub x20, x26, x25

write_output:
    cbz x20, exit_success
    mov x0, #1
    mov x1, x25
    mov x2, x20
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit_success
    add x25, x25, x0
    sub x20, x20, x0
    b write_output

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_success

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0
    b invalid_program
