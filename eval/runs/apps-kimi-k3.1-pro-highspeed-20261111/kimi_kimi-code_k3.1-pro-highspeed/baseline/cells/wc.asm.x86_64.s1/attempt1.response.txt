.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + input_buffer]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne check_space
    inc r12

check_space:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec rcx
    jnz scan_loop
    jmp read_more

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_number
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_number:
    lea r8, [rip + digit_buffer + 32]
    mov r9, r8
    mov r10d, 10
    test rax, rax
    jnz convert_loop

    dec r8
    mov byte ptr [r8], 48
    jmp copy_digits

convert_loop:
    xor edx, edx
    div r10
    dec r8
    add dl, 48
    mov byte ptr [r8], dl
    test rax, rax
    jnz convert_loop

copy_digits:
    mov dl, byte ptr [r8]
    inc r8
    mov byte ptr [rdi], dl
    inc rdi
    cmp r8, r9
    jne copy_digits
    ret

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 128
digit_buffer:
    .zero 32
