.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #65536

read_loop:
    cbz x20, input_done
    mov x0, #0
    mov x1, x19
    mov x2, x20
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    sub x20, x20, x0
    b read_loop

input_done:
    mov x22, #65536
    sub x20, x22, x20
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits
    adrp x26, line_buf
    add x26, x26, :lo12:line_buf
    mov x21, #0

row_loop:
    cbz x20, print_final

    mov x23, #16
    cmp x20, x23
    csel x23, x20, x23, lo

    movz x3, #0x2020
    movk x3, #0x2020, lsl #16
    movk x3, #0x2020, lsl #32
    movk x3, #0x2020, lsl #48
    mov x4, x26
    mov x5, #5
clear_line:
    stp x3, x3, [x4], #16
    subs x5, x5, #1
    b.ne clear_line

    mov x24, #0
    mov x27, #28
offset_loop:
    lsr x3, x21, x27
    and x3, x3, #0xf
    ldrb w4, [x25, x3]
    strb w4, [x26, x24]
    add x24, x24, #1
    sub x27, x27, #4
    cmp x24, #8
    b.lo offset_loop

    add x27, x26, #10
    mov x24, #0
hex_loop:
    cmp x24, x23
    b.hs ascii_begin
    ldrb w3, [x19, x24]
    lsr w4, w3, #4
    and w4, w4, #0xf
    ldrb w4, [x25, x4]
    strb w4, [x27]
    and w4, w3, #0xf
    ldrb w4, [x25, x4]
    strb w4, [x27, #1]
    add x27, x27, #3
    cmp x24, #7
    b.ne no_extra_space
    add x27, x27, #1
no_extra_space:
    add x24, x24, #1
    b hex_loop

ascii_begin:
    mov w3, #124
    strb w3, [x26, #60]
    add x27, x26, #61
    mov x24, #0
ascii_loop:
    cmp x24, x23
    b.hs ascii_done
    ldrb w3, [x19, x24]
    cmp w3, #0x20
    b.lo ascii_dot
    cmp w3, #0x7e
    b.ls ascii_store
ascii_dot:
    mov w3, #46
ascii_store:
    strb w3, [x27], #1
    add x24, x24, #1
    b ascii_loop

ascii_done:
    mov w3, #124
    strb w3, [x27]
    mov w3, #10
    strb w3, [x27, #1]

    mov x0, #1
    mov x1, x26
    add x2, x23, #63
    mov x8, #64
    svc #0

    add x19, x19, x23
    sub x20, x20, x23
    add x21, x21, x23
    b row_loop

print_final:
    mov x24, #0
    mov x27, #28
final_offset_loop:
    lsr x3, x21, x27
    and x3, x3, #0xf
    ldrb w4, [x25, x3]
    strb w4, [x26, x24]
    add x24, x24, #1
    sub x27, x27, #4
    cmp x24, #8
    b.lo final_offset_loop

    mov w3, #10
    strb w3, [x26, #8]
    mov x0, #1
    mov x1, x26
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
    b .

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
input_buf:
    .zero 65536
line_buf:
    .zero 80
