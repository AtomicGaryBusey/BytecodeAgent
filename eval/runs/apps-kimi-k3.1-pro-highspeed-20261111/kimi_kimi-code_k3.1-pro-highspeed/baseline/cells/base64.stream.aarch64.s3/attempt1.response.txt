.globl _start

.text

    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #49152

    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0

    adrp x22, output_buffer
    add x22, x22, :lo12:output_buffer
    mov x21, x22
    mov w23, #0
    adrp x24, base64_alphabet
    add x24, x24, :lo12:base64_alphabet

    cmp x20, #3
    b.hs encode_full_group
    cbz x20, finish_output
    cmp x20, #1
    b.eq encode_one_byte
    b encode_two_bytes

    ldrb w0, [x19], #1
    ldrb w1, [x19], #1
    ldrb w2, [x19], #1
    lsl w0, w0, #16
    orr w0, w0, w1, lsl #8
    orr w0, w0, w2

    lsr w1, w0, #18
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    lsr w1, w0, #12
    and w1, w1, #63
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    lsr w1, w0, #6
    and w1, w1, #63
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    and w1, w0, #63
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    sub x20, x20, #3
    b update_line

encode_one_byte:

    lsr w1, w0, #2
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    and w1, w0, #3

    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    mov w1, #61
    strb w1, [x21], #1
    strb w1, [x21], #1

    mov x20, #0

    ldrb w2, [x19]

    lsr w1, w0, #2
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    and w1, w0, #3
    lsl w1, w1, #4
    lsr w3, w2, #4
    orr w1, w1, w3
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    and w1, w2, #15
    lsl w1, w1, #2
    ldrb w1, [x24, x1]
    strb w1, [x21], #1

    strb w1, [x21], #1

    cmp w23, #76
    b.ne encode_next
    mov w1, #10
    strb w1, [x21], #1
    mov w23, #0

    mov w1, #10
    strb w1, [x21], #1

    mov x0, #1
    mov x1, x22
