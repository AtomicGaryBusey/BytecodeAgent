.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x21, x21, x0
    mov x3, x0
    adrp x4, input_buf
    add x4, x4, :lo12:input_buf

byte_loop:
    ldrb w5, [x4], #1
    cmp w5, #10
    b.ne not_newline
    add x19, x19, #1

not_newline:
    cmp w5, #32
    b.eq whitespace
    cmp w5, #9
    b.eq whitespace
    cmp w5, #10
    b.eq whitespace
    cmp w5, #11
    b.eq whitespace
    cmp w5, #12
    b.eq whitespace
    cmp w5, #13
    b.eq whitespace

    cbnz w22, next_byte
    add x20, x20, #1
    mov w22, #1
    b next_byte

whitespace:
    mov w22, #0

next_byte:
    subs x3, x3, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf

    mov x0, x19
    bl append_u64
    mov w4, #32
    strb w4, [x1], #1

    mov x0, x20
    bl append_u64
    mov w4, #32
    strb w4, [x1], #1

    mov x0, x21
    bl append_u64
    mov w4, #10
    strb w4, [x1], #1

    adrp x2, output_buf
    add x2, x2, :lo12:output_buf
    sub x2, x1, x2
    mov x0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_u64:
    adrp x2, digit_buf
    add x2, x2, :lo12:digit_buf
    add x6, x2, #31
    mov x7, x6
    mov x3, #10
    cbnz x0, convert_loop

    mov w4, #48
    strb w4, [x1], #1
    ret

convert_loop:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    strb w5, [x6], #-1
    mov x0, x4
    cbnz x0, convert_loop

copy_loop:
    ldrb w4, [x6, #1]!
    strb w4, [x1], #1
    cmp x6, x7
    b.lo copy_loop
    ret

.bss
.p2align 4
input_buf:
    .zero 65536
digit_buf:
    .zero 32
output_buf:
    .zero 80
