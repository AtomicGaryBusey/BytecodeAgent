.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x2, #32768
    sub x2, x2, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_success

    adrp x24, line_offsets
    add x24, x24, :lo12:line_offsets
    adrp x25, line_lengths
    add x25, x25, :lo12:line_lengths

    mov x21, #0
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x23, x20
    b.hs scan_done
    ldrb w0, [x19, x23]
    cmp w0, #10
    b.eq found_newline
    add x23, x23, #1
    b scan_loop

found_newline:
    str w22, [x24, x21, lsl #2]
    sub w0, w23, w22
    str w0, [x25, x21, lsl #2]
    add x21, x21, #1
    add x22, x23, #1
    add x23, x23, #1
    b scan_loop

scan_done:
    cmp x22, x20
    b.eq sort_initialize
    str w22, [x24, x21, lsl #2]
    sub w0, w20, w22
    str w0, [x25, x21, lsl #2]
    add x21, x21, #1

sort_initialize:
    mov x26, #1

sort_outer:
    cmp x26, x21
    b.hs output_initialize
    ldr w28, [x24, x26, lsl #2]
    ldr w29, [x25, x26, lsl #2]
    mov x27, x26

sort_inner:
    cbz x27, sort_place
    sub x0, x27, #1
    ldr w9, [x24, x0, lsl #2]
    ldr w10, [x25, x0, lsl #2]
    add x11, x19, x9
    add x12, x19, x28
    mov x13, #0
    cmp x10, x29
    csel x14, x10, x29, ls

compare_loop:
    cmp x13, x14
    b.hs compare_prefix
    ldrb w15, [x11, x13]
    ldrb w16, [x12, x13]
    cmp w15, w16
    b.lo sort_place
    b.hi sort_shift
    add x13, x13, #1
    b compare_loop

compare_prefix:
    cmp x10, x29
    b.ls sort_place

sort_shift:
    str w9, [x24, x27, lsl #2]
    str w10, [x25, x27, lsl #2]
    sub x27, x27, #1
    b sort_inner

sort_place:
    str w28, [x24, x27, lsl #2]
    str w29, [x25, x27, lsl #2]
    add x26, x26, #1
    b sort_outer

output_initialize:
    mov x26, #0
    adrp x17, newline
    add x17, x17, :lo12:newline

output_loop:
    cmp x26, x21
    b.hs exit_success
    ldr w9, [x24, x26, lsl #2]
    ldr w10, [x25, x26, lsl #2]
    mov x0, #1
    add x1, x19, x9
    mov x2, x10
    mov x8, #64
    svc #0

    mov x0, #1
    mov x1, x17
    mov x2, #1
    mov x8, #64
    svc #0

    add x26, x26, #1
    b output_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.bss
    .align 4
input_buffer:
    .zero 32768
    .align 4
line_offsets:
    .zero 4000
    .align 4
line_lengths:
    .zero 4000
