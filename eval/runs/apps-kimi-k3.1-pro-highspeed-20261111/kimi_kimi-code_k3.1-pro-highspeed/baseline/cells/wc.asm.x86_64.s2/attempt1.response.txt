.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done

    add r12, rax
    lea rsi, [rip + input_buffer]
    lea rdi, [rsi + rax]

next_byte:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 32
    je whitespace
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe whitespace

    test r15d, r15d
    jnz byte_done
    mov r15d, 1
    inc r14
    jmp byte_done

whitespace:
    cmp al, 10
    jne not_newline
    inc r13
not_newline:
    xor r15d, r15d

byte_done:
    cmp rsi, rdi
    jb next_byte
    jmp read_more

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    mov r8, rdi
    mov r9d, 10
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 48
    inc rdi
    ret

1:
    xor edx, edx
    div r9
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz 1b

    mov rsi, r8
    lea rcx, [rdi - 1]
2:
    cmp rsi, rcx
    jae 3f
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rcx]
    mov byte ptr [rsi], dl
    mov byte ptr [rcx], al
    inc rsi
    dec rcx
    jmp 2b
3:
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 128
