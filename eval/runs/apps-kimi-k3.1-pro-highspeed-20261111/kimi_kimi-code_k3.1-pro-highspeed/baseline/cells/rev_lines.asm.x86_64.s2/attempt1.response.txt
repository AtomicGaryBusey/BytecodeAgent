.globl _start

.equ INPUT_SIZE, 65536

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_more:
    cmp r13, INPUT_SIZE
    jae process_input
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INPUT_SIZE
    sub rdx, r13
    syscall
    test rax, rax
    jle process_input
    add r13, rax
    jmp read_more

process_input:
    lea r14, [rip + output_buffer]
    mov r15, r14
    xor ebx, ebx

next_line:
    cmp rbx, r13
    jae output_done
    mov rcx, rbx

find_line_end:
    cmp rcx, r13
    jae reverse_line
    cmp byte ptr [r12 + rcx], 10
    je reverse_line
    inc rcx
    jmp find_line_end

reverse_line:
    mov rdx, rcx

reverse_byte:
    cmp rdx, rbx
    je emit_newline
    dec rdx
    mov al, byte ptr [r12 + rdx]
    mov byte ptr [r15], al
    inc r15
    jmp reverse_byte

emit_newline:
    mov byte ptr [r15], 10
    inc r15
    cmp rcx, r13
    jae output_done
    inc rcx
    mov rbx, rcx
    jmp next_line

output_done:
    mov rdx, r15
    sub rdx, r14
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero INPUT_SIZE + 1
