.globl _start

.text
_start:
    lea r8, [rip + hex_digits]
    lea rsi, [rip + input_buffer]
    xor r12d, r12d

read_more:
    mov edx, 65536
    sub rdx, r12
    jz dump_begin
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle dump_begin
    add r12, rax
    add rsi, rax
    jmp read_more

dump_begin:
    xor r13d, r13d

row_begin:
    cmp r13, r12
    jae final_offset

    mov rbp, r12
    sub rbp, r13
    cmp rbp, 16
    jbe row_size_ready
    mov ebp, 16
row_size_ready:

    lea rsi, [rip + input_buffer]
    add rsi, r13

    lea rdi, [rip + line_buffer + 8]
    mov eax, r13d
    mov ecx, 8
offset_digits:
    dec rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    shr eax, 4
    dec ecx
    jnz offset_digits

    add rdi, 8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx

hex_slots:
    cmp ecx, ebp
    jae blank_slot

    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp slot_done

blank_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

slot_done:
    inc ecx
    cmp ecx, 8
    jne no_extra_space
    mov byte ptr [rdi], 0x20
    inc rdi
no_extra_space:
    cmp ecx, 16
    jb hex_slots

    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx

ascii_bytes:
    cmp ecx, ebp
    jae ascii_done
    mov al, byte ptr [rsi + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, 0x2e
ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp ascii_bytes

ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 10
    add rdi, 2

    mov rdx, rdi
    lea rsi, [rip + line_buffer]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r13, rbp
    jmp row_begin

final_offset:
    lea rdi, [rip + line_buffer + 8]
    mov eax, r13d
    mov ecx, 8
final_digits:
    dec rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    shr eax, 4
    dec ecx
    jnz final_digits

    lea rsi, [rip + line_buffer]
    mov byte ptr [rsi + 8], 10
    mov edx, 9
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .space 65536
line_buffer:
    .space 80
