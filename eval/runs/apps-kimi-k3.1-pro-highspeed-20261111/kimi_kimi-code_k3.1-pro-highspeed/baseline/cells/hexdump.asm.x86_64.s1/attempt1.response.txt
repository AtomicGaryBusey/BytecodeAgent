.globl _start

.text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

input_done:
    xor r14d, r14d

row_loop:
    cmp r14, r12
    jae final_offset

    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_size_ready
    mov r15d, 16
row_size_ready:
    lea rbp, [r13 + r14]
    lea rdi, [rip + output_buf]
    lea rsi, [rip + hex_digits]

    mov eax, r14d
    mov ecx, 8
offset_hex_loop:
    rol eax, 4
    mov ebx, eax
    and ebx, 15
    movzx ebx, byte ptr [rsi + rbx]
    mov byte ptr [rdi], bl
    inc rdi
    dec ecx
    jnz offset_hex_loop

    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor ecx, ecx
hex_slot_loop:
    cmp rcx, r15
    jae blank_hex_slot

    movzx eax, byte ptr [rbp + rcx]
    mov edx, eax
    shr eax, 4
    and edx, 15
    mov al, byte ptr [rsi + rax]
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], al
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp hex_slot_done

blank_hex_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

hex_slot_done:
    inc ecx
    cmp ecx, 8
    jne no_extra_space
    mov byte ptr [rdi], 0x20
    inc rdi
no_extra_space:
    cmp ecx, 16
    jb hex_slot_loop

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2

    xor ecx, ecx
ascii_loop:
    cmp rcx, r15
    jae ascii_done
    mov al, byte ptr [rbp + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, 0x2e
ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2

    mov rdx, rdi
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r14, r15
    jmp row_loop

final_offset:
    lea rdi, [rip + output_buf]
    lea rsi, [rip + hex_digits]
    mov eax, r12d
    mov ecx, 8
final_hex_loop:
    rol eax, 4
    mov ebx, eax
    and ebx, 15
    movzx ebx, byte ptr [rsi + rbx]
    mov byte ptr [rdi], bl
    inc rdi
    dec ecx
    jnz final_hex_loop

    mov byte ptr [rdi], 0x0a
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 65536
output_buf:
    .space 80
