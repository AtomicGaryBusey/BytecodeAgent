.globl _start

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_more:
    cmp r13, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_more

read_done:
    xor r14d, r14d
    xor r15d, r15d
    call skip_ws
    call parse_value
    test eax, eax
    jnz report_error
    call skip_ws
    cmp r14, r13
    jne document_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

document_error:
    mov r15, r14

report_error:
    lea rdi, [rip + output_buffer]
    movabs rax, 0x746120726f727265
    mov qword ptr [rdi], rax
    mov byte ptr [rdi + 8], 0x20

    mov rax, r15
    lea rbx, [rip + digit_buffer + 32]
    mov ecx, 10

convert_digit:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz convert_digit

    lea rdi, [rip + output_buffer + 9]
    lea r8, [rip + digit_buffer + 32]

copy_digits:
    mov dl, byte ptr [rbx]
    mov byte ptr [rdi], dl
    inc rbx
    inc rdi
    cmp rbx, r8
    jb copy_digits

    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    mov rax, r14

skip_ws_loop:
    cmp rax, r13
    jae skip_ws_done
    movzx edx, byte ptr [r12 + rax]
    cmp dl, 0x20
    je skip_ws_advance
    cmp dl, 0x09
    je skip_ws_advance
    cmp dl, 0x0a
    je skip_ws_advance
    cmp dl, 0x0d
    jne skip_ws_done

skip_ws_advance:
    inc rax
    jmp skip_ws_loop

skip_ws_done:
    mov r14, rax
    ret

parse_value:
    call skip_ws
    cmp r14, r13
    jae value_error
    movzx eax, byte ptr [r12 + r14]

    cmp al, 0x7b
    je value_object
    cmp al, 0x5b
    je value_array
    cmp al, 0x22
    je value_string
    cmp al, 0x2d
    je value_number
    cmp al, 0x74
    je value_true
    cmp al, 0x66
    je value_false
    cmp al, 0x6e
    je value_null
    cmp al, 0x30
    jb value_error
    cmp al, 0x39
    jbe value_number

value_error:
    mov r15, r14
    mov eax, 1
    ret

value_object:
    jmp parse_object

value_array:
    jmp parse_array

value_string:
    jmp parse_string

value_number:
    jmp parse_number

value_true:
    lea rsi, [rip + literal_true]
    mov edx, 4
    jmp parse_literal

value_false:
    lea rsi, [rip + literal_false]
    mov edx, 5
    jmp parse_literal

value_null:
    lea rsi, [rip + literal_null]
    mov edx, 4
    jmp parse_literal

parse_literal:
    xor ecx, ecx

literal_loop:
    cmp rcx, rdx
    jae literal_success
    cmp r14, r13
    jae literal_error
    mov al, byte ptr [rsi + rcx]
    cmp al, byte ptr [r12 + r14]
    jne literal_error
    inc r14
    inc rcx
    jmp literal_loop

literal_success:
    xor eax, eax
    ret

literal_error:
    mov r15, r14
    mov eax, 1
    ret

parse_string:
    inc r14

string_loop:
    cmp r14, r13
    jae string_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x22
    je string_close
    cmp al, 0x5c
    je string_escape
    cmp al, 0x20
    jb string_error
    inc r14
    jmp string_loop

string_close:
    inc r14
    xor eax, eax
    ret

string_escape:
    inc r14
    cmp r14, r13
    jae string_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je string_simple_escape
    cmp al, 0x5c
    je string_simple_escape
    cmp al, 0x2f
    je string_simple_escape
    cmp al, 0x62
    je string_simple_escape
    cmp al, 0x66
    je string_simple_escape
    cmp al, 0x6e
    je string_simple_escape
    cmp al, 0x72
    je string_simple_escape
    cmp al, 0x74
    je string_simple_escape
    jmp string_error

string_simple_escape:
    inc r14
    jmp string_loop

string_unicode:
    inc r14
    mov ecx, 4

unicode_digit_loop:
    cmp r14, r13
    jae string_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb string_error
    cmp al, 0x39
    jbe unicode_digit_valid
    cmp al, 0x41
    jb string_error
    cmp al, 0x46
    jbe unicode_digit_valid
    cmp al, 0x61
    jb string_error
    cmp al, 0x66
    ja string_error

unicode_digit_valid:
    inc r14
    dec ecx
    jnz unicode_digit_loop
    jmp string_loop

string_error:
    mov r15, r14
    mov eax, 1
    ret

parse_number:
    cmp byte ptr [r12 + r14], 0x2d
    jne number_integer
    inc r14
    cmp r14, r13
    jae number_error

number_integer:
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb number_error
    cmp al, 0x39
    ja number_error

number_integer_digits:
    inc r14
    cmp r14, r13
    jae number_fraction
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_fraction
    cmp al, 0x39
    jbe number_integer_digits
    jmp number_fraction

number_zero:
    inc r14

number_fraction:
    cmp r14, r13
    jae number_exponent
    cmp byte ptr [r12 + r14], 0x2e
    jne number_exponent
    inc r14
    cmp r14, r13
    jae number_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_error
    cmp al, 0x39
    ja number_error

number_fraction_digits:
    inc r14
    cmp r14, r13
    jae number_exponent
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_exponent
    cmp al, 0x39
    jbe number_fraction_digits

number_exponent:
    cmp r14, r13
    jae number_success
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x65
    je number_exponent_start
    cmp al, 0x45
    jne number_success

number_exponent_start:
    inc r14
    cmp r14, r13
    jae number_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2b
    je number_exponent_sign
    cmp al, 0x2d
    jne number_exponent_first_digit

number_exponent_sign:
    inc r14
    cmp r14, r13
    jae number_error

number_exponent_first_digit:
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_error
    cmp al, 0x39
    ja number_error

number_exponent_digits:
    inc r14
    cmp r14, r13
    jae number_success
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_success
    cmp al, 0x39
    jbe number_exponent_digits

number_success:
    xor eax, eax
    ret

number_error:
    mov r15, r14
    mov eax, 1
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae array_error
    cmp byte ptr [r12 + r14], 0x5d
    je array_empty

array_element:
    call parse_value
    test eax, eax
    jnz array_return
    call skip_ws
    cmp r14, r13
    jae array_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x5d
    je array_close
    cmp al, 0x2c
    jne array_error
    inc r14
    jmp array_element

array_empty:
array_close:
    inc r14
    xor eax, eax
    ret

array_error:
    mov r15, r14
    mov eax, 1

array_return:
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae object_error
    cmp byte ptr [r12 + r14], 0x7d
    je object_empty

object_member:
    call skip_ws
    cmp r14, r13
    jae object_error
    cmp byte ptr [r12 + r14], 0x22
    jne object_error
    call parse_string
    test eax, eax
    jnz object_return
    call skip_ws
    cmp r14, r13
    jae object_error
    cmp byte ptr [r12 + r14], 0x3a
    jne object_error
    inc r14
    call parse_value
    test eax, eax
    jnz object_return
    call skip_ws
    cmp r14, r13
    jae object_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x7d
    je object_close
    cmp al, 0x2c
    jne object_error
    inc r14
    jmp object_member

object_empty:
object_close:
    inc r14
    xor eax, eax
    ret

object_error:
    mov r15, r14
    mov eax, 1

object_return:
    ret

.section .rodata
ok_message:
    .ascii "ok\n"
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
    .p2align 4
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
digit_buffer:
    .zero 32
