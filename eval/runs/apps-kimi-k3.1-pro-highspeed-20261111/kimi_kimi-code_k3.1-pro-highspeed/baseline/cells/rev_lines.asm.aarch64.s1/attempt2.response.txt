.text
.globl _start

.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 131072

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #INPUT_SIZE
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_success

    mov x21, x19
    add x22, x19, x20
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf

line_loop:
    cmp x21, x22
    b.hs output
    mov x24, x21

scan_line:
    cmp x21, x22
    b.hs final_line
    ldrb w0, [x21], #1
    cmp w0, #10
    b.ne scan_line
    sub x25, x21, #1
    b reverse_line

final_line:
    mov x25, x21

reverse_line:
    mov x0, x24
    mov x1, x25

reverse_loop:
    cmp x0, x1
    b.hs reverse_done
    ldrb w2, [x1, #-1]!
    strb w2, [x23], #1
    b reverse_loop

reverse_done:
    mov w2, #10
    strb w2, [x23], #1
    b line_loop

output:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x23, x1
    mov x0, #1
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE
output_buf:
    .zero OUTPUT_SIZE
