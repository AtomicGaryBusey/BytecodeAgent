.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle input_ready
    add r12, rax
    jmp read_more

input_ready:
    xor r13d, r13d

line_loop:
    cmp r13, r12
    jae program_exit

    lea r14, [rip + input_buffer]
    add r14, r13
    mov r15, r14
    lea rdx, [rip + input_buffer]
    add rdx, r12

find_line_end:
    cmp r15, rdx
    jae evaluate_line
    cmp byte ptr [r15], 10
    je evaluate_line
    inc r15
    jmp find_line_end

evaluate_line:
    lea rbx, [rip + value_stack]

skip_separators:
    cmp r14, r15
    jae evaluation_finished
    mov al, byte ptr [r14]
    cmp al, 32
    je skip_one_separator
    cmp al, 9
    je skip_one_separator
    jmp token_begin

skip_one_separator:
    inc r14
    jmp skip_separators

token_begin:
    mov rsi, r14

scan_token:
    cmp r14, r15
    jae token_scanned
    mov al, byte ptr [r14]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    inc r14
    jmp scan_token

token_scanned:
    mov rdx, r14
    sub rdx, rsi
    cmp rdx, 1
    jne parse_integer

    mov al, byte ptr [rsi]
    cmp al, 43
    je operator_add
    cmp al, 45
    je operator_subtract
    cmp al, 42
    je operator_multiply
    cmp al, 47
    je operator_divide
    cmp al, 37
    je operator_remainder

parse_integer:
    xor r8d, r8d
    cmp byte ptr [rsi], 45
    jne integer_sign_done
    mov r8d, 1
    inc rsi
    dec rdx

integer_sign_done:
    test rdx, rdx
    jz line_error
    cmp rdx, 18
    ja line_error

    xor eax, eax

parse_digit:
    movzx ecx, byte ptr [rsi]
    sub ecx, 48
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc rsi
    dec rdx
    jnz parse_digit

    test r8d, r8d
    jz push_integer
    neg rax

push_integer:
    mov qword ptr [rbx], rax
    add rbx, 8
    jmp skip_separators

operator_add:
    mov r9d, 1
    jmp evaluate_operator

operator_subtract:
    mov r9d, 2
    jmp evaluate_operator

operator_multiply:
    mov r9d, 3
    jmp evaluate_operator

operator_divide:
    mov r9d, 4
    jmp evaluate_operator

operator_remainder:
    mov r9d, 5

evaluate_operator:
    lea rax, [rip + value_stack + 16]
    cmp rbx, rax
    jb line_error

    sub rbx, 8
    mov rcx, qword ptr [rbx]
    sub rbx, 8
    mov rax, qword ptr [rbx]

    cmp r9d, 1
    je perform_add
    cmp r9d, 2
    je perform_subtract
    cmp r9d, 3
    je perform_multiply

    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne division_safe
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error

division_safe:
    cqo
    idiv rcx
    cmp r9d, 4
    je store_operator_result
    mov rax, rdx
    jmp store_operator_result

perform_add:
    add rax, rcx
    jmp store_operator_result

perform_subtract:
    sub rax, rcx
    jmp store_operator_result

perform_multiply:
    imul rax, rcx

store_operator_result:
    mov qword ptr [rbx], rax
    add rbx, 8
    jmp skip_separators

evaluation_finished:
    lea rax, [rip + value_stack + 8]
    cmp rbx, rax
    jne line_error

    mov rax, qword ptr [rip + value_stack]
    lea r10, [rip + output_buffer + 32]
    mov byte ptr [r10 - 1], 10
    lea rsi, [r10 - 1]

    xor r8d, r8d
    test rax, rax
    jns magnitude_ready
    mov r8d, 1
    neg rax

magnitude_ready:
    mov r9, 10
    test rax, rax
    jnz convert_digit

    dec rsi
    mov byte ptr [rsi], 48
    jmp add_result_sign

convert_digit:
    xor edx, edx
    div r9
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert_digit

add_result_sign:
    test r8d, r8d
    jz write_result
    dec rsi
    mov byte ptr [rsi], 45

write_result:
    mov rdx, r10
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp line_done

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall

line_done:
    mov r13, r15
    lea rax, [rip + input_buffer]
    sub r13, rax
    cmp r13, r12
    jae line_loop
    inc r13
    jmp line_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
error_message:
    .ascii "error\n"

.bss
.align 16
input_buffer:
    .zero 65536
.align 16
value_stack:
    .zero 8192
.align 16
output_buffer:
    .zero 32
