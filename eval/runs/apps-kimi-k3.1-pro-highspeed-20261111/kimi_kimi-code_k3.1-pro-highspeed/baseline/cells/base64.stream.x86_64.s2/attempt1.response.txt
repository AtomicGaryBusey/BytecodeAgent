.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    lea r13, [rip + input_buffer]
    lea r14, [rip + output_buffer]
    lea rbp, [rip + base64_alphabet]
    mov rbx, r12
    xor r15d, r15d

    cmp rbx, 3
    jb final_group

encode_full_group:
    movzx eax, byte ptr [r13]
    shl eax, 16
    movzx ecx, byte ptr [r13 + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [r13 + 2]
    or eax, ecx

    mov ecx, eax
    shr ecx, 18
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14], dl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14 + 1], dl

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14 + 2], dl

    and eax, 63
    mov dl, byte ptr [rbp + rax]
    mov byte ptr [r14 + 3], dl

    add r13, 3
    add r14, 4
    sub rbx, 3

    inc r15d
    cmp r15d, 19
    jne full_group_no_newline
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d

full_group_no_newline:
    cmp rbx, 3
    jae encode_full_group

final_group:
    test rbx, rbx
    jz finish_encoding
    cmp rbx, 1
    je final_one_byte

    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]

    mov r8d, eax
    shr r8d, 2
    mov dl, byte ptr [rbp + r8]
    mov byte ptr [r14], dl

    and eax, 3
    shl eax, 4
    mov r8d, ecx
    shr r8d, 4
    or eax, r8d
    mov dl, byte ptr [rbp + rax]
    mov byte ptr [r14 + 1], dl

    and ecx, 15
    shl ecx, 2
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14 + 2], dl
    mov byte ptr [r14 + 3], 61
    jmp final_group_done

final_one_byte:
    movzx eax, byte ptr [r13]

    mov ecx, eax
    shr ecx, 2
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14], dl

    and eax, 3
    shl eax, 4
    mov dl, byte ptr [rbp + rax]
    mov byte ptr [r14 + 1], dl
    mov byte ptr [r14 + 2], 61
    mov byte ptr [r14 + 3], 61

final_group_done:
    add r14, 4
    inc r15d

finish_encoding:
    test r15d, r15d
    jz prepare_output
    mov byte ptr [r14], 10
    inc r14

prepare_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    jz exit_success

    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero 49153
.align 16
output_buffer:
    .zero 66560
