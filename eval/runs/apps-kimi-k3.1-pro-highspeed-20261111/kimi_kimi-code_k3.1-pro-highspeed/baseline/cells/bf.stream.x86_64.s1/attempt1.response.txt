.globl _start
.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 65537
    jb read_more

read_done:
    lea rsi, [rip + input_buf]
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rsi + rcx], 33
    je separator_found
    inc rcx
    jmp find_separator

no_separator:
    mov r13, r12
    lea r14, [rsi + r12]
    mov r15, r14
    jmp validate_start

separator_found:
    mov r13, rcx
    lea r14, [rsi + rcx + 1]
    lea r15, [rsi + r12]

validate_start:
    xor ecx, ecx
    xor edx, edx
    lea rdi, [rip + jump_table]
    lea r8, [rip + bracket_stack]

validate_loop:
    cmp rcx, r13
    jae validate_end
    mov al, byte ptr [rsi + rcx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close

validate_next:
    inc rcx
    jmp validate_loop

validate_open:
    cmp edx, 1000
    jae invalid_program
    mov dword ptr [r8 + rdx*4], ecx
    inc edx
    inc rcx
    jmp validate_loop

validate_close:
    test edx, edx
    jz invalid_program
    dec edx
    mov eax, dword ptr [r8 + rdx*4]
    mov dword ptr [rdi + rax*4], ecx
    mov dword ptr [rdi + rcx*4], eax
    inc rcx
    jmp validate_loop

validate_end:
    test edx, edx
    jnz invalid_program

    xor ebx, ebx
    lea rbp, [rip + tape]
    lea r10, [rip + output_buf]

execute_loop:
    cmp rbx, r13
    jae execution_done
    movzx eax, byte ptr [rsi + rbx]
    inc rbx

    cmp al, 62
    je cmd_right
    cmp al, 60
    je cmd_left
    cmp al, 43
    je cmd_increment
    cmp al, 45
    je cmd_decrement
    cmp al, 46
    je cmd_output
    cmp al, 44
    je cmd_input
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    jmp execute_loop

cmd_right:
    inc rbp
    jmp execute_loop

cmd_left:
    dec rbp
    jmp execute_loop

cmd_increment:
    inc byte ptr [rbp]
    jmp execute_loop

cmd_decrement:
    dec byte ptr [rbp]
    jmp execute_loop

cmd_output:
    mov al, byte ptr [rbp]
    mov byte ptr [r10], al
    inc r10
    jmp execute_loop

cmd_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbp], al
    jmp execute_loop

input_exhausted:
    mov byte ptr [rbp], 0
    jmp execute_loop

cmd_open:
    cmp byte ptr [rbp], 0
    jne execute_loop
    mov eax, dword ptr [rdi + rbx*4 - 4]
    lea ebx, [rax + 1]
    jmp execute_loop

cmd_close:
    cmp byte ptr [rbp], 0
    je execute_loop
    mov eax, dword ptr [rdi + rbx*4 - 4]
    lea ebx, [rax + 1]
    jmp execute_loop

execution_done:
    lea rsi, [rip + output_buf]
    mov rdx, r10
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

.bss
input_buf:
    .zero 65537
output_buf:
    .zero 65536
tape:
    .zero 30000
    .p2align 4
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
