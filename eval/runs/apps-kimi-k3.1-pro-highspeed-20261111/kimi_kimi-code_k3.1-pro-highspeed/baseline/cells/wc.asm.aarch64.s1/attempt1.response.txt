.globl _start

.text
_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov w22, wzr
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf

read_loop:
    mov x0, xzr
    mov x1, x23
    mov x2, #65536
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x19, x19, x0
    mov x3, x23
    mov x4, x0

byte_loop:
    ldrb w5, [x3], #1
    cmp w5, #10
    b.ne check_whitespace
    add x20, x20, #1
    b whitespace

check_whitespace:
    cmp w5, #32
    b.eq whitespace
    cmp w5, #9
    b.eq whitespace
    cmp w5, #11
    b.eq whitespace
    cmp w5, #12
    b.eq whitespace
    cmp w5, #13
    b.eq whitespace

    cbnz w22, next_byte
    mov w22, #1
    add x21, x21, #1
    b next_byte

whitespace:
    mov w22, wzr

next_byte:
    subs x4, x4, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf

    mov x0, x19
    bl append_uint
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl append_uint
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl append_uint
    mov w2, #10
    strb w2, [x1], #1

    adrp x2, output_buf
    add x2, x2, :lo12:output_buf
    sub x2, x1, x2
    mov x0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x8, #64
    svc #0

    mov x0, xzr
    mov x8, #93
    svc #0
    b .

append_uint:
    adrp x3, number_tmp
    add x3, x3, :lo12:number_tmp
    mov x4, x3
    mov x5, #10

1:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    strb w7, [x4], #1
    mov x0, x6
    cbnz x0, 1b

2:
    sub x4, x4, #1
    ldrb w6, [x4]
    strb w6, [x1], #1
    cmp x4, x3
    b.ne 2b
    ret

.bss
.align 4
input_buf:
    .zero 65536
output_buf:
    .zero 80
number_tmp:
    .zero 32
