.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jle input_ready
    add r12, rax
    jmp read_more

input_ready:
    lea rdi, [rip + input_buf]
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rdi + rcx], 0x21
    je separator_found
    inc rcx
    jmp find_separator

separator_found:
    mov r13, rcx
    lea r14, [rdi + rcx + 1]
    lea r15, [rdi + r12]
    jmp begin_parse

no_separator:
    mov r13, r12
    lea r15, [rdi + r12]
    mov r14, r15

begin_parse:
    lea r8, [rip + code_buf]
    lea r9, [rip + jump_table]
    lea r10, [rip + bracket_stack]
    xor ebx, ebx
    xor ebp, ebp
    xor ecx, ecx

parse_loop:
    cmp rcx, r13
    jae parse_finished
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x5b
    je parse_open
    cmp al, 0x5d
    je parse_close
    cmp al, 0x3e
    je append_command
    cmp al, 0x3c
    je append_command
    cmp al, 0x2b
    je append_command
    cmp al, 0x2d
    je append_command
    cmp al, 0x2e
    je append_command
    cmp al, 0x2c
    je append_command

parse_next:
    inc rcx
    jmp parse_loop

append_command:
    mov byte ptr [r8 + rbx], al
    inc rbx
    jmp parse_next

parse_open:
    mov byte ptr [r8 + rbx], 0x5b
    mov dword ptr [r10 + rbp*4], ebx
    inc rbp
    inc rbx
    jmp parse_next

parse_close:
    test rbp, rbp
    jz invalid_program
    dec rbp
    mov edx, dword ptr [r10 + rbp*4]
    mov byte ptr [r8 + rbx], 0x5d
    lea eax, [rdx + 1]
    mov dword ptr [r9 + rbx*4], eax
    lea eax, [rbx + 1]
    mov dword ptr [r9 + rdx*4], eax
    inc rbx
    jmp parse_next

parse_finished:
    test rbp, rbp
    jnz invalid_program

    lea rbp, [rip + tape]
    lea r13, [rip + output_buf]
    xor r11d, r11d

execute_loop:
    cmp r11, rbx
    jae execution_finished
    movzx eax, byte ptr [r8 + r11]
    cmp al, 0x3e
    je cmd_right
    cmp al, 0x3c
    je cmd_left
    cmp al, 0x2b
    je cmd_increment
    cmp al, 0x2d
    je cmd_decrement
    cmp al, 0x2e
    je cmd_output
    cmp al, 0x2c
    je cmd_input
    cmp al, 0x5b
    je cmd_open
    jmp cmd_close

cmd_right:
    inc rbp
    inc r11
    jmp execute_loop

cmd_left:
    dec rbp
    inc r11
    jmp execute_loop

cmd_increment:
    inc byte ptr [rbp]
    inc r11
    jmp execute_loop

cmd_decrement:
    dec byte ptr [rbp]
    inc r11
    jmp execute_loop

cmd_output:
    mov al, byte ptr [rbp]
    mov byte ptr [r13], al
    inc r13
    inc r11
    jmp execute_loop

cmd_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbp], al
    inc r11
    jmp execute_loop

input_exhausted:
    mov byte ptr [rbp], 0
    inc r11
    jmp execute_loop

cmd_open:
    cmp byte ptr [rbp], 0
    jne open_advance
    mov r11d, dword ptr [r9 + r11*4]
    jmp execute_loop

open_advance:
    inc r11
    jmp execute_loop

cmd_close:
    cmp byte ptr [rbp], 0
    je close_advance
    mov r11d, dword ptr [r9 + r11*4]
    jmp execute_loop

close_advance:
    inc r11
    jmp execute_loop

execution_finished:
    lea rsi, [rip + output_buf]
    sub r13, rsi
    test r13, r13
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buf:
    .zero 65537

.balign 16
code_buf:
    .zero 65536

.balign 16
jump_table:
    .zero 262144

.balign 16
bracket_stack:
    .zero 4000

.balign 16
tape:
    .zero 30000

.balign 16
output_buf:
    .zero 65536
