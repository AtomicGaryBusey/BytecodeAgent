.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r15, [rip + output_buffer]
    lea r9, [rip + hex_digits]
    xor r14d, r14d

row_loop:
    test r13, r13
    jz final_offset

    mov ebx, 16
    cmp r13, rbx
    cmovb rbx, r13

    mov eax, r14d
    mov ecx, 8
offset_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r15], dl
    inc r15
    dec ecx
    jnz offset_loop

    mov word ptr [r15], 0x2020
    add r15, 2

    xor ecx, ecx
hex_slot_loop:
    cmp ecx, ebx
    jae blank_slot

    movzx eax, byte ptr [r12 + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r15], dl
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [r15 + 1], al
    mov byte ptr [r15 + 2], 0x20
    add r15, 3
    jmp slot_separator

blank_slot:
    mov word ptr [r15], 0x2020
    mov byte ptr [r15 + 2], 0x20
    add r15, 3

slot_separator:
    cmp ecx, 7
    jne no_extra_space
    mov byte ptr [r15], 0x20
    inc r15
no_extra_space:
    inc ecx
    cmp ecx, 16
    jne hex_slot_loop

    mov byte ptr [r15], 0x20
    mov byte ptr [r15 + 1], 0x7c
    add r15, 2

    xor ecx, ecx
ascii_loop:
    cmp ecx, ebx
    jae ascii_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, 0x2e
ascii_store:
    mov byte ptr [r15], al
    inc r15
    inc ecx
    jmp ascii_loop

ascii_done:
    mov word ptr [r15], 0x0a7c
    add r15, 2

    add r12, rbx
    sub r13, rbx
    add r14, rbx
    jmp row_loop

final_offset:
    mov eax, r14d
    mov ecx, 8
final_hex_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r15], dl
    inc r15
    dec ecx
    jnz final_hex_loop
    mov byte ptr [r15], 0x0a
    inc r15

    lea rsi, [rip + output_buffer]
    mov rdx, r15
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 323593
