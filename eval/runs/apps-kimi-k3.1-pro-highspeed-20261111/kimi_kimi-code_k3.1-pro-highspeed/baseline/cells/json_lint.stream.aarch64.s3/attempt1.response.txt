    ret

    b.hs .Lskip_ws_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq .Lskip_ws_advance
    cmp w0, #0x09
    b.eq .Lskip_ws_advance
    cmp w0, #0x0a
    b.eq .Lskip_ws_advance
    cmp w0, #0x0d

.Lskip_ws_advance:
    add x21, x21, #1

.Lskip_ws_done:

    ret

    ldrb w0, [x19, x21]
    cmp w0, #0x7b

    cmp w0, #0x5b
    b.eq parse_array
    cmp w0, #0x22
    b.eq parse_string
    cmp w0, #0x74
    b.eq .Lvalue_true
    cmp w0, #0x66
    b.eq .Lvalue_false
    cmp w0, #0x6e
    b.eq .Lvalue_null
    cmp w0, #0x2d
    b.eq parse_number
    sub w1, w0, #0x30
    cmp w1, #9

    b fail

.Lvalue_true:

    add x1, x1, :lo12:literal_true
    mov x2, #4
    b parse_literal

.Lvalue_false:

    add x1, x1, :lo12:literal_false
    mov x2, #5
    b parse_literal

.Lvalue_null:

    add x1, x1, :lo12:literal_null

.Lliteral_loop:

    b.hs fail

    ldrb w5, [x1, x3]

    b.ne fail

    add x3, x3, #1

    add x21, x21, #1
.Lstring_loop:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq .Lstring_close
    cmp w0, #0x5c
    b.eq .Lstring_escape
    cmp w0, #0x20

    add x21, x21, #1
    b .Lstring_loop

    ret

.Lstring_escape:

    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x75
    b.eq .Lstring_unicode
    cmp w0, #0x22
    b.eq .Lstring_simple_escape
    cmp w0, #0x5c
    b.eq .Lstring_simple_escape
    cmp w0, #0x2f
    b.eq .Lstring_simple_escape
    cmp w0, #0x62
    b.eq .Lstring_simple_escape
    cmp w0, #0x66
    b.eq .Lstring_simple_escape
    cmp w0, #0x6e
    b.eq .Lstring_simple_escape
    cmp w0, #0x72
    b.eq .Lstring_simple_escape
    cmp w0, #0x74

    cmp w0, #0x2d
