.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, 65536
    syscall

    test rax, rax
    jle input_done

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + input_buf]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne check_space
    inc r12

check_space:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jne next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec rcx
    jnz byte_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buf]

    mov rax, r12
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_number
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_number:
    lea rsi, [rip + digit_buf]
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 48
    inc rdi
    ret

1:
    xor ecx, ecx
    mov r8d, 10

2:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rsi + rcx], dl
    inc rcx
    test rax, rax
    jnz 2b

3:
    dec rcx
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [rdi], al
    inc rdi
    test rcx, rcx
    jnz 3b
    ret

.bss
.balign 16
input_buf:
    .zero 65536
digit_buf:
    .zero 32
output_buf:
    .zero 64
