.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    mov edx, 49152
    sub rdx, r12
    jz read_done

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_more

read_done:
    test r12, r12
    jz exit_success

    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea rbp, [rip + alphabet]
    mov r15, r12
    xor r14d, r14d

encode_loop:
    cmp r15, 3
    jb encode_tail

    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [rbp + rax]
    call put_character

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [rbp + rax]
    call put_character

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov ecx, r10d
    shr ecx, 6
    or eax, ecx
    mov al, byte ptr [rbp + rax]
    call put_character

    mov eax, r10d
    and eax, 63
    mov al, byte ptr [rbp + rax]
    call put_character

    add rsi, 3
    sub r15, 3
    jmp encode_loop

encode_tail:
    test r15, r15
    jz finish_encoding

    movzx r8d, byte ptr [rsi]
    cmp r15, 1
    je tail_one

    movzx r9d, byte ptr [rsi + 1]

    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [rbp + rax]
    call put_character

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [rbp + rax]
    call put_character

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [rbp + rax]
    call put_character

    mov al, 61
    call put_character
    jmp finish_encoding

tail_one:
    mov eax, r8d
    shr eax, 2
    mov al, byte ptr [rbp + rax]
    call put_character

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbp + rax]
    call put_character

    mov al, 61
    call put_character
    mov al, 61
    call put_character

finish_encoding:
    test r14d, r14d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    mov rdx, rdi
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

put_character:
    mov byte ptr [rdi], al
    inc rdi
    inc r14d
    cmp r14d, 76
    jne put_character_done
    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
put_character_done:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66400
