.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    mov r13, r12
    lea r12, [rip + input_buf]
    xor r14d, r14d

    call skip_ws
    call parse_value
    test eax, eax
    jnz report_error
    call skip_ws
    cmp r14, r13
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

report_error:
    lea rbx, [rip + num_buf + 32]
    mov rax, r14
    mov ecx, 10

convert_digit:
    xor edx, edx
    div rcx
    dec rbx
    add dl, 0x30
    mov byte ptr [rbx], dl
    test rax, rax
    jnz convert_digit

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, 9
    syscall

    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    lea rdx, [rip + num_buf + 32]
    sub rdx, rbx
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r14, r13
    jae skip_ws_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    jne skip_ws_done

skip_ws_advance:
    inc r14
    jmp skip_ws

skip_ws_done:
    ret

parse_value:
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je value_true
    cmp al, 0x66
    je value_false
    cmp al, 0x6e
    je value_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb parse_error
    cmp al, 0x39
    jbe parse_number
    jmp parse_error

value_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal

value_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal

value_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4
    jmp parse_literal

parse_literal:
    cmp r14, r13
    jae parse_error
    mov al, byte ptr [r12 + r14]
    cmp al, byte ptr [rsi]
    jne parse_error
    inc r14
    inc rsi
    dec ecx
    jnz parse_literal
    xor eax, eax
    ret

parse_string:
    inc r14

string_loop:
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x22
    je string_end
    cmp al, 0x5c
    je string_escape
    cmp al, 0x20
    jb parse_error
    inc r14
    jmp string_loop

string_end:
    inc r14
    xor eax, eax
    ret

string_escape:
    inc r14
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je string_simple_escape
    cmp al, 0x5c
    je string_simple_escape
    cmp al, 0x2f
    je string_simple_escape
    cmp al, 0x62
    je string_simple_escape
    cmp al, 0x66
    je string_simple_escape
    cmp al, 0x6e
    je string_simple_escape
    cmp al, 0x72
    je string_simple_escape
    cmp al, 0x74
    jne parse_error

string_simple_escape:
    inc r14
    jmp string_loop

string_unicode:
    inc r14
    mov ecx, 4

unicode_digit_loop:
    cmp r14, r13
    jae parse_error
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_error
    cmp al, 0x39
    jbe unicode_digit_ok
    mov dl, al
    or dl, 0x20
    cmp dl, 0x61
    jb parse_error
    cmp dl, 0x66
    ja parse_error

unicode_digit_ok:
    inc r14
    dec ecx
    jnz unicode_digit_loop
    jmp string_loop

parse_number:
    cmp byte ptr [r12 + r14], 0x2d
    jne number_integer
    inc r14
    cmp r14, r13
    jae parse_error

number_integer:
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    je number_leading_zero
    cmp al, 0x31
    jb parse_error
    cmp al, 0x39
    ja parse_error

number_integer_digits:
    inc r14
    cmp r14, r13
    jae parse_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_fraction
    cmp al, 0x39
    jbe number_integer_digits
    jmp number_fraction

number_leading_zero:
    inc r14

number_fraction:
    cmp r14, r13
    jae parse_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x2e
    jne number_exponent
    inc r14
    cmp r14, r13
    jae parse_error
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_error
    cmp al, 0x39
    ja parse_error

number_fraction_digits:
    inc r14
    cmp r14, r13
    jae parse_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_exponent
    cmp al, 0x39
    jbe number_fraction_digits

number_exponent:
    cmp r14, r13
    jae parse_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x65
    je number_exponent_start
    cmp al, 0x45
    jne parse_success

number_exponent_start:
    inc r14
    cmp r14, r13
    jae parse_error
    mov al, byte ptr [r12 + r14]
    cmp al, 0x2b
    je number_exponent_sign
    cmp al, 0x2d
    jne number_exponent_digit_check

number_exponent_sign:
    inc r14
    cmp r14, r13
    jae parse_error

number_exponent_digit_check:
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_error
    cmp al, 0x39
    ja parse_error

number_exponent_digits:
    inc r14
    cmp r14, r13
    jae parse_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_success
    cmp al, 0x39
    jbe number_exponent_digits
    jmp parse_success

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 0x5d
    je array_close

array_value:
    call parse_value
    test eax, eax
    jnz array_return_error
    call skip_ws
    cmp r14, r13
    jae parse_error
    mov al, byte ptr [r12 + r14]
    cmp al, 0x2c
    je array_comma
    cmp al, 0x5d
    je array_close
    jmp parse_error

array_comma:
    inc r14
    call skip_ws
    jmp array_value

array_close:
    inc r14
    xor eax, eax
    ret

array_return_error:
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 0x7d
    je object_close

object_member:
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 0x22
    jne parse_error
    call parse_string
    test eax, eax
    jnz object_return_error
    call skip_ws
    cmp r14, r13
    jae parse_error
    cmp byte ptr [r12 + r14], 0x3a
    jne parse_error
    inc r14
    call skip_ws
    call parse_value
    test eax, eax
    jnz object_return_error
    call skip_ws
    cmp r14, r13
    jae parse_error
    mov al, byte ptr [r12 + r14]
    cmp al, 0x2c
    je object_comma
    cmp al, 0x7d
    je object_close
    jmp parse_error

object_comma:
    inc r14
    call skip_ws
    jmp object_member

object_close:
    inc r14
    xor eax, eax
    ret

object_return_error:
    ret

parse_success:
    xor eax, eax
    ret

parse_error:
    mov eax, 1
    ret

.section .rodata
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
newline:
    .ascii "\n"

.bss
.align 16
input_buf:
    .zero 65537
num_buf:
    .zero 32
