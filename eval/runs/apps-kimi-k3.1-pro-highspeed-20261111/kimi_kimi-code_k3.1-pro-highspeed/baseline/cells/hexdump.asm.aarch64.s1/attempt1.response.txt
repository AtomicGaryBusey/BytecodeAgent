.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x15, hex_digits
    add x15, x15, :lo12:hex_digits
    mov x19, #0
    movz x21, #1, lsl #16

read_loop:
    cmp x19, x21
    b.hs input_done
    mov x0, #0
    add x1, x20, x19
    sub x2, x21, x19
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    b read_loop

input_done:
    adrp x25, line_buf
    add x25, x25, :lo12:line_buf
    mov x22, #0

row_loop:
    cmp x22, x19
    b.hs final_offset
    add x23, x20, x22
    sub x3, x19, x22
    mov x24, #16
    cmp x3, x24
    csel x24, x3, x24, ls

    mov x1, x25
    mov w0, w22
    mov x2, x15
    bl format_hex8
    mov w3, #32
    strb w3, [x1], #1
    strb w3, [x1], #1

    mov x26, #0
slot_loop:
    cmp x26, #16
    b.hs slots_done
    cmp x26, x24
    b.hs blank_slot

    ldrb w3, [x23, x26]
    lsr w4, w3, #4
    ldrb w4, [x15, x4]
    strb w4, [x1], #1
    and w5, w3, #15
    ldrb w5, [x15, x5]
    strb w5, [x1], #1
    mov w3, #32
    strb w3, [x1], #1
    b slot_separator

blank_slot:
    mov w3, #32
    strb w3, [x1], #1
    strb w3, [x1], #1
    strb w3, [x1], #1

slot_separator:
    cmp x26, #7
    b.ne no_extra_space
    mov w3, #32
    strb w3, [x1], #1
no_extra_space:
    add x26, x26, #1
    b slot_loop

slots_done:
    mov w3, #32
    strb w3, [x1], #1
    mov w3, #124
    strb w3, [x1], #1

    mov x26, #0
ascii_loop:
    cmp x26, x24
    b.hs ascii_done
    ldrb w3, [x23, x26]
    cmp w3, #32
    b.lo ascii_dot
    cmp w3, #126
    b.ls ascii_store
ascii_dot:
    mov w3, #46
ascii_store:
    strb w3, [x1], #1
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w3, #124
    strb w3, [x1], #1
    mov w3, #10
    strb w3, [x1], #1

    sub x2, x1, x25
    mov x0, #1
    mov x1, x25
    mov x8, #64
    svc #0

    add x22, x22, x24
    b row_loop

final_offset:
    mov x1, x25
    mov w0, w19
    mov x2, x15
    bl format_hex8
    mov w3, #10
    strb w3, [x1], #1
    sub x2, x1, x25
    mov x0, #1
    mov x1, x25
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

format_hex8:
    mov w3, w0
    mov w4, #8
1:
    lsr w5, w3, #28
    ldrb w5, [x2, x5]
    strb w5, [x1], #1
    lsl w3, w3, #4
    subs w4, w4, #1
    b.ne 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 65536
.balign 16
line_buf:
    .space 80
