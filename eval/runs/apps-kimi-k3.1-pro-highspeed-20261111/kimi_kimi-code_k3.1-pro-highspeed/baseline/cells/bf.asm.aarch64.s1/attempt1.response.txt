.equ INPUT_CAPACITY, 65537
.equ JUMP_TABLE_SIZE, 262144
.equ BRACKET_STACK_SIZE, 262144
.equ TAPE_SIZE, 30000
.equ OUTPUT_CAPACITY, 65536

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x28, #1
    movk x28, #1, lsl #16

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x28, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x22, x19, x20
    mov x27, #0

find_separator:
    cmp x27, x20
    b.hs no_separator
    ldrb w0, [x19, x27]
    cmp w0, #33
    b.eq separator_found
    add x27, x27, #1
    b find_separator

separator_found:
    mov x20, x27
    add x21, x19, x27
    add x21, x21, #1
    b split_done

no_separator:
    mov x21, x22

split_done:
    adrp x23, jump_table
    add x23, x23, :lo12:jump_table
    adrp x24, bracket_stack
    add x24, x24, :lo12:bracket_stack
    mov x25, x24
    mov x27, #0

bracket_loop:
    cmp x27, x20
    b.hs brackets_finished
    ldrb w0, [x19, x27]
    cmp w0, #91
    b.eq open_bracket
    cmp w0, #93
    b.eq close_bracket

bracket_next:
    add x27, x27, #1
    b bracket_loop

open_bracket:
    str w27, [x25], #4
    b bracket_next

close_bracket:
    cmp x25, x24
    b.eq invalid_program
    ldr w1, [x25, #-4]!
    add w2, w27, #1
    str w2, [x23, x1, lsl #2]
    add w1, w1, #1
    str w1, [x23, x27, lsl #2]
    b bracket_next

brackets_finished:
    cmp x25, x24
    b.ne invalid_program

    adrp x24, tape
    add x24, x24, :lo12:tape
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x25, x26
    mov x27, #0

execute_loop:
    cmp x27, x20
    b.hs execution_done
    ldrb w0, [x19, x27]
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_increment
    cmp w0, #45
    b.eq cmd_decrement
    cmp w0, #46
    b.eq cmd_output
    cmp w0, #44
    b.eq cmd_input
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close

advance_pc:
    add x27, x27, #1
    b execute_loop

cmd_right:
    add x24, x24, #1
    b advance_pc

cmd_left:
    sub x24, x24, #1
    b advance_pc

cmd_increment:
    ldrb w0, [x24]
    add w0, w0, #1
    strb w0, [x24]
    b advance_pc

cmd_decrement:
    ldrb w0, [x24]
    sub w0, w0, #1
    strb w0, [x24]
    b advance_pc

cmd_output:
    ldrb w0, [x24]
    strb w0, [x25], #1
    b advance_pc

cmd_input:
    cmp x21, x22
    b.hs input_exhausted
    ldrb w0, [x21], #1
    strb w0, [x24]
    b advance_pc

input_exhausted:
    strb wzr, [x24]
    b advance_pc

cmd_open:
    ldrb w0, [x24]
    cbnz w0, advance_pc
    ldr w0, [x23, x27, lsl #2]
    mov x27, x0
    b execute_loop

cmd_close:
    ldrb w0, [x24]
    cbz w0, advance_pc
    ldr w0, [x23, x27, lsl #2]
    mov x27, x0
    b execute_loop

execution_done:
    sub x2, x25, x26
    cbz x2, exit_success
    mov x0, #1
    mov x1, x26
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero INPUT_CAPACITY
.balign 16
jump_table:
    .zero JUMP_TABLE_SIZE
.balign 16
bracket_stack:
    .zero BRACKET_STACK_SIZE
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
output_buf:
    .zero OUTPUT_CAPACITY
