.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buf]
    xor r13d, r13d

find_separator:
    cmp r13, r12
    jae no_separator
    cmp byte ptr [rsi + r13], 0x21
    je separator_found
    inc r13
    jmp find_separator

separator_found:
    lea r14, [rsi + r13 + 1]
    mov r15, r12
    sub r15, r13
    dec r15
    jmp validate_start

no_separator:
    mov r13, r12
    lea r14, [rsi + r12]
    xor r15d, r15d

validate_start:
    lea rdi, [rip + jump_table]
    lea rdx, [rip + bracket_stack]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

validate_loop:
    cmp r8, r13
    jae validation_done
    mov al, byte ptr [rsi + r8]
    cmp al, 0x3e
    je store_command
    cmp al, 0x3c
    je store_command
    cmp al, 0x2b
    je store_command
    cmp al, 0x2d
    je store_command
    cmp al, 0x2e
    je store_command
    cmp al, 0x2c
    je store_command
    cmp al, 0x5b
    je open_bracket
    cmp al, 0x5d
    je close_bracket
    inc r8
    jmp validate_loop

store_command:
    mov byte ptr [rsi + r10], al
    inc r10
    inc r8
    jmp validate_loop

open_bracket:
    cmp r9, 1000
    jae invalid_program
    mov byte ptr [rsi + r10], al
    mov dword ptr [rdx + r9*4], r10d
    inc r9
    inc r10
    inc r8
    jmp validate_loop

close_bracket:
    test r9, r9
    jz invalid_program
    mov byte ptr [rsi + r10], al
    dec r9
    mov ecx, dword ptr [rdx + r9*4]
    lea eax, [r10 + 1]
    mov dword ptr [rdi + rcx*4], eax
    lea eax, [rcx + 1]
    mov dword ptr [rdi + r10*4], eax
    inc r10
    inc r8
    jmp validate_loop

validation_done:
    test r9, r9
    jnz invalid_program
    mov r13, r10
    xor ebx, ebx
    xor ebp, ebp
    lea r10, [rip + tape]
    lea rdx, [rip + out_buf]

run_loop:
    cmp rbx, r13
    jae program_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x3e
    je cmd_right
    cmp al, 0x3c
    je cmd_left
    cmp al, 0x2b
    je cmd_increment
    cmp al, 0x2d
    je cmd_decrement
    cmp al, 0x2e
    je cmd_output
    cmp al, 0x2c
    je cmd_input
    cmp al, 0x5b
    je cmd_open
    cmp al, 0x5d
    je cmd_close

run_next:
    inc rbx
    jmp run_loop

cmd_right:
    inc r10
    inc rbx
    jmp run_loop

cmd_left:
    dec r10
    inc rbx
    jmp run_loop

cmd_increment:
    inc byte ptr [r10]
    inc rbx
    jmp run_loop

cmd_decrement:
    dec byte ptr [r10]
    inc rbx
    jmp run_loop

cmd_output:
    mov al, byte ptr [r10]
    mov byte ptr [rdx + rbp], al
    inc rbp
    inc rbx
    jmp run_loop

cmd_input:
    test r15, r15
    jz input_exhausted
    mov al, byte ptr [r14]
    inc r14
    dec r15
    jmp store_input

input_exhausted:
    xor eax, eax

store_input:
    mov byte ptr [r10], al
    inc rbx
    jmp run_loop

cmd_open:
    cmp byte ptr [r10], 0
    jne run_next
    mov eax, dword ptr [rdi + rbx*4]
    mov rbx, rax
    jmp run_loop

cmd_close:
    cmp byte ptr [r10], 0
    je run_next
    mov eax, dword ptr [rdi + rbx*4]
    mov rbx, rax
    jmp run_loop

program_done:
    test rbp, rbp
    jz exit_success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov rdx, rbp
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.section .bss
.align 16
input_buf:
    .zero 65537
.align 16
jump_table:
    .zero 262144
.align 16
bracket_stack:
    .zero 4000
.align 16
tape:
    .zero 30000
.align 16
out_buf:
    .zero 65536
