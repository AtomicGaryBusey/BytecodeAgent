.globl _start

.text
_start:
    xor r12d, r12d
    lea rbx, [rip + input_buf]

read_more:
    cmp r12, 65537
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    lea r15, [rbx + r12]
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rbx + rcx], 33
    je separator_found
    inc rcx
    jmp find_separator

no_separator:
    mov r13, r12
    mov r14, r15
    jmp validate_start

separator_found:
    mov r13, rcx
    lea r14, [rbx + rcx + 1]

validate_start:
    xor ecx, ecx
    xor edx, edx
    lea r8, [rip + bracket_stack]
    lea r9, [rip + jump_table]

validate_loop:
    cmp rcx, r13
    jae validate_end
    mov al, byte ptr [rbx + rcx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
validate_next:
    inc rcx
    jmp validate_loop

open_bracket:
    cmp edx, 1000
    jae invalid
    mov dword ptr [r8 + rdx*4], ecx
    inc edx
    jmp validate_next

close_bracket:
    test edx, edx
    jz invalid
    dec edx
    mov eax, dword ptr [r8 + rdx*4]
    lea esi, [rcx + 1]
    mov dword ptr [r9 + rax*4], esi
    lea esi, [rax + 1]
    mov dword ptr [r9 + rcx*4], esi
    jmp validate_next

validate_end:
    test edx, edx
    jnz invalid

    xor ecx, ecx
    lea rbp, [rip + tape]
    lea r8, [rip + jump_table]
    lea r9, [rip + output_buf]
    xor r10d, r10d

run_loop:
    cmp rcx, r13
    jae program_done
    movzx eax, byte ptr [rbx + rcx]
    inc rcx

    cmp al, 62
    je cmd_right
    cmp al, 60
    je cmd_left
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_output
    cmp al, 44
    je cmd_input
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    jmp run_loop

cmd_right:
    inc rbp
    jmp run_loop

cmd_left:
    dec rbp
    jmp run_loop

cmd_plus:
    inc byte ptr [rbp]
    jmp run_loop

cmd_minus:
    dec byte ptr [rbp]
    jmp run_loop

cmd_output:
    mov al, byte ptr [rbp]
    mov byte ptr [r9 + r10], al
    inc r10
    jmp run_loop

cmd_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbp], al
    jmp run_loop

input_exhausted:
    mov byte ptr [rbp], 0
    jmp run_loop

cmd_open:
    cmp byte ptr [rbp], 0
    jne run_loop
    mov ecx, dword ptr [r8 + rcx*4 - 4]
    jmp run_loop

cmd_close:
    cmp byte ptr [rbp], 0
    je run_loop
    mov ecx, dword ptr [r8 + rcx*4 - 4]
    jmp run_loop

program_done:
    test r10, r10
    jz exit_success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov rdx, r10
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buf:
    .zero 65537
.balign 16
output_buf:
    .zero 65536
.balign 16
jump_table:
    .zero 262144
.balign 16
bracket_stack:
    .zero 4000
.balign 16
tape:
    .zero 30000
