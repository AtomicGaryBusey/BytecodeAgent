.globl _start
.text
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x19, #0

read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, read_loop

input_done:
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x20, #0
    mov x22, #0

line_loop:
    cmp x20, x19
    b.hs write_output
    mov x21, x20

scan_loop:
    cmp x21, x19
    b.eq final_line
    ldrb w3, [x23, x21]
    cmp w3, #10
    b.eq newline_line
    add x21, x21, #1
    b scan_loop

newline_line:
    mov x26, #0
    b reverse_line

final_line:
    mov x26, #1

reverse_line:
    mov x25, x21
1:
    cmp x25, x20
    b.ls 2f
    sub x25, x25, #1
    ldrb w3, [x23, x25]
    add x4, x24, x22
    strb w3, [x4]
    add x22, x22, #1
    b 1b
2:
    mov w3, #10
    add x4, x24, x22
    strb w3, [x4]
    add x22, x22, #1
    cbnz x26, write_output
    add x20, x21, #1
    b line_loop

write_output:
    cbz x22, exit_program
    mov x0, #1
    mov x1, x24
    mov x2, x22
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .space 65536
.balign 16
output_buf:
    .space 65537
