.globl _start

.section .text
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0

    cmp x0, #0
    b.le input_done

    add x21, x21, x0
    adrp x2, input_buf
    add x2, x2, :lo12:input_buf
    add x3, x2, x0

scan_loop:
    ldrb w4, [x2], #1

    cmp w4, #32
    b.eq whitespace
    sub w5, w4, #9
    cmp w5, #4
    b.ls whitespace

    cmp x22, #0
    b.ne scan_next
    mov x22, #1
    add x20, x20, #1
    b scan_next

whitespace:
    mov x22, #0
    cmp w4, #10
    b.ne scan_next
    add x19, x19, #1

scan_next:
    cmp x2, x3
    b.lo scan_loop
    b read_loop

input_done:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf

    mov x0, x19
    bl append_u64
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl append_u64
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl append_u64
    mov w2, #10
    strb w2, [x1], #1

    adrp x2, output_buf
    add x2, x2, :lo12:output_buf
    sub x2, x1, x2
    mov x0, #1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_u64:
    adrp x9, number_tmp
    add x9, x9, :lo12:number_tmp
    add x14, x9, #32
    mov x9, x14
    mov x10, x0
    mov x11, #10

    cbnz x10, 1f
    mov w12, #48
    strb w12, [x1], #1
    ret

1:
    udiv x12, x10, x11
    msub x13, x12, x11, x10
    add w13, w13, #48
    strb w13, [x9, #-1]!
    mov x10, x12
    cbnz x10, 1b

2:
    ldrb w12, [x9], #1
    strb w12, [x1], #1
    cmp x9, x14
    b.lo 2b
    ret

.section .bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 64
number_tmp:
    .zero 32
