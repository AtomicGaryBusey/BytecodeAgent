.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    mov x2, #1
    lsl x2, x2, #16
    cmp x19, x2
    b.hs read_full
    sub x2, x2, x19
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, input_ready
    add x19, x19, x0
    b read_loop

read_full:
    mov x0, #0
    adrp x1, read_scratch
    add x1, x1, :lo12:read_scratch
    mov x2, #1
    mov x8, #63
    svc #0
    cbz x0, input_ready
    b read_full

input_ready:
    mov x21, x20
    add x22, x20, x19

line_dispatch:
    cmp x21, x22
    b.hs exit_success
    mov x23, x21

find_line_end:
    cmp x21, x22
    b.hs line_at_eof
    ldrb w0, [x21]
    cmp w0, #10
    b.eq line_at_newline
    add x21, x21, #1
    b find_line_end

line_at_newline:
    mov x24, x21
    add x21, x21, #1
    b evaluate_line

line_at_eof:
    mov x24, x21

evaluate_line:
    mov x0, x23
    mov x1, x24
    bl eval_line
    cbnz w1, print_error
    bl print_number
    b line_dispatch

print_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, #64
    svc #0
    b line_dispatch

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

eval_line:
    mov x4, x0
    mov x5, x1
    adrp x3, value_stack
    add x3, x3, :lo12:value_stack
    mov x2, x3

next_token:
    cmp x4, x5
    b.hs tokens_done
    ldrb w6, [x4]
    cmp w6, #32
    b.eq skip_separator
    cmp w6, #9
    b.eq skip_separator
    mov x6, x4

find_token_end:
    cmp x4, x5
    b.hs token_ready
    ldrb w7, [x4]
    cmp w7, #32
    b.eq token_ready
    cmp w7, #9
    b.eq token_ready
    add x4, x4, #1
    b find_token_end

token_ready:
    sub x7, x4, x6
    cmp x7, #1
    b.ne parse_literal
    ldrb w8, [x6]
    cmp w8, #43
    b.eq have_operator
    cmp w8, #45
    b.eq have_operator
    cmp w8, #42
    b.eq have_operator
    cmp w8, #47
    b.eq have_operator
    cmp w8, #37
    b.eq have_operator

parse_literal:
    mov x9, x6
    mov w15, #0
    ldrb w8, [x9]
    cmp w8, #45
    b.ne digits_setup
    mov w15, #1
    add x9, x9, #1

digits_setup:
    sub x7, x4, x9
    cbz x7, line_error
    cmp x7, #18
    b.hi line_error
    mov x10, #0
    mov x11, #10

digit_loop:
    ldrb w8, [x9], #1
    sub w8, w8, #48
    cmp w8, #9
    b.hi line_error
    madd x10, x10, x11, x8
    subs x7, x7, #1
    b.ne digit_loop
    cbz w15, store_literal
    neg x10, x10

store_literal:
    str x10, [x2], #8
    b next_token

have_operator:
    sub x13, x2, x3
    cmp x13, #16
    b.lo line_error
    ldr x11, [x2, #-8]!
    ldr x12, [x2, #-8]!
    cmp w8, #43
    b.eq op_add
    cmp w8, #45
    b.eq op_sub
    cmp w8, #42
    b.eq op_mul
    b op_divmod

op_add:
    add x10, x12, x11
    b push_result

op_sub:
    sub x10, x12, x11
    b push_result

op_mul:
    mul x10, x12, x11
    b push_result

op_divmod:
    cbz x11, line_error
    mov x13, #1
    lsl x13, x13, #63
    cmp x12, x13
    b.ne division_safe
    cmn x11, #1
    b.eq line_error

division_safe:
    sdiv x14, x12, x11
    cmp w8, #47
    b.eq quotient_result
    msub x10, x14, x11, x12
    b push_result

quotient_result:
    mov x10, x14

push_result:
    str x10, [x2], #8
    b next_token

skip_separator:
    add x4, x4, #1
    b next_token

tokens_done:
    sub x13, x2, x3
    cmp x13, #8
    b.ne line_error
    ldr x0, [x3]
    mov w1, #0
    ret

line_error:
    mov w1, #1
    ret

print_number:
    adrp x1, number_buf
    add x1, x1, :lo12:number_buf
    add x1, x1, #32
    mov x2, x1
    mov w3, #10
    strb w3, [x1, #-1]!
    mov x4, #10
    mov x5, x0
    cmp x0, #0
    b.ge convert_digits
    neg x5, x0

convert_digits:
    udiv x6, x5, x4
    msub x7, x6, x4, x5
    add w7, w7, #48
    strb w7, [x1, #-1]!
    mov x5, x6
    cbnz x5, convert_digits
    cmp x0, #0
    b.ge number_ready
    mov w3, #45
    strb w3, [x1, #-1]!

number_ready:
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .rodata
error_msg:
    .ascii "error\n"

.bss
.align 3
input_buf:
    .zero 65536
read_scratch:
    .zero 1
.align 3
value_stack:
    .zero 4096
number_buf:
    .zero 32
