.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 49152
outbuf:
    .zero 66560

.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d

read_loop:
    mov edx, 49152
    sub rdx, r13
    lea rsi, [r12 + r13]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle input_done
    add r13, rax
    cmp r13, 49152
    jb read_loop

input_done:
    test r13, r13
    jz exit_success

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + alphabet]
    lea r14, [rsi + r13]
    xor r9d, r9d

encode_full:
    lea rax, [rsi + 3]
    cmp rax, r14
    ja encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ebx, byte ptr [rsi + 1]
    shl ebx, 8
    or eax, ebx
    movzx ebx, byte ptr [rsi + 2]
    or eax, ebx
    add rsi, 3

    mov ebx, eax
    shr ebx, 18
    mov bl, byte ptr [r8 + rbx]
    mov byte ptr [rdi], bl

    mov ebx, eax
    shr ebx, 12
    and ebx, 63
    mov bl, byte ptr [r8 + rbx]
    mov byte ptr [rdi + 1], bl

    mov ebx, eax
    shr ebx, 6
    and ebx, 63
    mov bl, byte ptr [r8 + rbx]
    mov byte ptr [rdi + 2], bl

    and eax, 63
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], al

    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne encode_full
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encode_full

encode_tail:
    mov rcx, r14
    sub rcx, rsi
    test rcx, rcx
    jz finish_lines

    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rcx, 2
    jb tail_chars
    movzx ebx, byte ptr [rsi + 1]
    shl ebx, 8
    or eax, ebx

tail_chars:
    mov ebx, eax
    shr ebx, 18
    mov bl, byte ptr [r8 + rbx]
    mov byte ptr [rdi], bl

    mov ebx, eax
    shr ebx, 12
    and ebx, 63
    mov bl, byte ptr [r8 + rbx]
    mov byte ptr [rdi + 1], bl

    cmp rcx, 2
    jb tail_one

    mov ebx, eax
    shr ebx, 6
    and ebx, 63
    mov bl, byte ptr [r8 + rbx]
    mov byte ptr [rdi + 2], bl
    mov byte ptr [rdi + 3], 61
    jmp tail_group_done

tail_one:
    mov word ptr [rdi + 2], 0x3d3d

tail_group_done:
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne finish_lines
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d

finish_lines:
    test r9d, r9d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_success
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
