.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0

read_loop:
    mov x0, #0
    adrp x1, buffer
    add x1, x1, :lo12:buffer
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_done

    add x21, x21, x0
    adrp x1, buffer
    add x1, x1, :lo12:buffer
    mov x2, x0

process_loop:
    ldrb w3, [x1], #1

    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace

    cbnz w22, next_byte
    add x20, x20, #1
    mov w22, #1
    b next_byte

whitespace:
    mov w22, #0

next_byte:
    subs x2, x2, #1
    b.ne process_loop
    b read_loop

input_done:
    mov x0, x19
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    bl append_u64

    mov w3, #32
    strb w3, [x1], #1
    mov x0, x20
    bl append_u64

    mov w3, #32
    strb w3, [x1], #1
    mov x0, x21
    bl append_u64

    mov w3, #10
    strb w3, [x1], #1

    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    sub x2, x1, x2
    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_u64:

    mov x5, #0
    mov x6, #10

1:

    msub x8, x7, x6, x0
    add w8, w8, #48
    strb w8, [sp, x5]
    add x5, x5, #1
    mov x0, x7
    cbnz x0, 1b

    ldrb w8, [sp, x5]
    strb w8, [x1], #1
    cbnz x5, 2b

    add sp, sp, #32
