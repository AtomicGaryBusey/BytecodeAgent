.globl _start

.text
_start:
    lea r10, [rip + input_buffer]
    xor r12d, r12d

read_more:
    mov edx, 65536
    sub rdx, r12
    jz input_complete
    xor eax, eax
    xor edi, edi
    lea rsi, [r10 + r12]
    syscall
    test rax, rax
    jz input_complete
    add r12, rax
    jmp read_more

input_complete:
    xor r13d, r13d

find_separator:
    cmp r13, r12
    jae no_separator
    cmp byte ptr [r10 + r13], 33
    je separator_found
    inc r13
    jmp find_separator

separator_found:
    lea r14, [r10 + r13 + 1]
    mov r15, r12
    sub r15, r13
    dec r15
    jmp validate_start

no_separator:
    lea r14, [r10 + r12]
    xor r15d, r15d

validate_start:
    xor ecx, ecx
    xor edx, edx
    lea rdi, [rip + jump_table]
    lea r8, [rip + bracket_stack]

validate_loop:
    cmp rcx, r13
    jae validate_end
    mov al, byte ptr [r10 + rcx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    inc rcx
    jmp validate_loop

open_bracket:
    cmp edx, 1000
    jae invalid_program
    mov dword ptr [r8 + rdx*4], ecx
    inc edx
    inc rcx
    jmp validate_loop

close_bracket:
    test edx, edx
    jz invalid_program
    dec edx
    mov eax, dword ptr [r8 + rdx*4]
    mov dword ptr [rdi + rcx*4], eax
    mov dword ptr [rdi + rax*4], ecx
    inc rcx
    jmp validate_loop

validate_end:
    test edx, edx
    jnz invalid_program

    xor ecx, ecx
    xor r8d, r8d
    lea rbx, [rip + tape]
    lea r9, [rip + jump_table]
    lea rsi, [rip + output_buffer]

execute_loop:
    cmp rcx, r13
    jae execution_complete
    movzx eax, byte ptr [r10 + rcx]
    inc rcx

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close
    jmp execute_loop

command_right:
    inc rbx
    jmp execute_loop

command_left:
    dec rbx
    jmp execute_loop

command_plus:
    inc byte ptr [rbx]
    jmp execute_loop

command_minus:
    dec byte ptr [rbx]
    jmp execute_loop

command_output:
    mov al, byte ptr [rbx]
    mov byte ptr [rsi + r8], al
    inc r8
    jmp execute_loop

command_input:
    test r15, r15
    jz input_exhausted
    mov al, byte ptr [r14]
    inc r14
    dec r15
    mov byte ptr [rbx], al
    jmp execute_loop

input_exhausted:
    mov byte ptr [rbx], 0
    jmp execute_loop

command_open:
    cmp byte ptr [rbx], 0
    jne execute_loop
    mov eax, dword ptr [r9 + rcx*4 - 4]
    lea ecx, [rax + 1]
    jmp execute_loop

command_close:
    cmp byte ptr [rbx], 0
    je execute_loop
    mov eax, dword ptr [r9 + rcx*4 - 4]
    lea ecx, [rax + 1]
    jmp execute_loop

execution_complete:
    test r8, r8
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rdx, r8
    syscall

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
tape:
    .zero 30000
output_buffer:
    .zero 65536
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
