.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #65536

read_loop:
    cmp x20, x21
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    adrp x28, hex_digits
    add x28, x28, :lo12:hex_digits
    mov x22, #0

row_loop:
    cmp x22, x20
    b.hs print_final

    sub x23, x20, x22
    cmp x23, #16
    b.ls row_size_ready
    mov x23, #16
row_size_ready:
    add x27, x19, x22
    mov x25, x24

    mov w0, w22
    mov x1, x25
    bl hex8
    mov x25, x1

    mov w4, #32
    strb w4, [x25], #1
    strb w4, [x25], #1

    mov x26, #0
slot_loop:
    cmp x26, x23
    b.hs slot_empty

    ldrb w3, [x27, x26]
    lsr w4, w3, #4
    and w5, w3, #15
    ldrb w4, [x28, x4]
    ldrb w5, [x28, x5]
    strb w4, [x25], #1
    strb w5, [x25], #1
    mov w4, #32
    strb w4, [x25], #1
    b slot_done

slot_empty:
    mov w4, #32
    strb w4, [x25], #1
    strb w4, [x25], #1
    strb w4, [x25], #1

slot_done:
    add x26, x26, #1
    cmp x26, #8
    b.ne no_extra_space
    mov w4, #32
    strb w4, [x25], #1
no_extra_space:
    cmp x26, #16
    b.lo slot_loop

    mov w4, #32
    strb w4, [x25], #1
    mov w4, #124
    strb w4, [x25], #1

    mov x26, #0
ascii_loop:
    cmp x26, x23
    b.hs ascii_done
    ldrb w3, [x27, x26]
    cmp w3, #32
    b.lo ascii_dot
    cmp w3, #126
    b.hi ascii_dot
    strb w3, [x25], #1
    b ascii_next
ascii_dot:
    mov w3, #46
    strb w3, [x25], #1
ascii_next:
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w3, #124
    strb w3, [x25], #1
    mov w3, #10
    strb w3, [x25], #1

    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0

    add x22, x22, x23
    b row_loop

print_final:
    mov w0, w20
    mov x1, x24
    bl hex8
    mov w3, #10
    strb w3, [x1], #1

    mov x0, #1
    sub x2, x1, x24
    mov x1, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

hex8:
    adrp x2, hex_digits
    add x2, x2, :lo12:hex_digits
    mov w3, #28
hex8_loop:
    lsr w4, w0, w3
    and w4, w4, #15
    ldrb w4, [x2, x4]
    strb w4, [x1], #1
    subs w3, w3, #4
    b.ge hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
out_buf:
    .zero 80
