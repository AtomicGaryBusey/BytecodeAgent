.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + inbuf]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15, r15
    jne next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec rcx
    jne byte_loop
    jmp read_loop

input_done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call put_decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call put_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

put_decimal:
    test rax, rax
    jne decimal_nonzero
    mov byte ptr [rdi], 48
    inc rdi
    ret

decimal_nonzero:
    lea r8, [rip + numtmp + 32]
    xor ecx, ecx
    mov r9d, 10
decimal_loop:
    xor edx, edx
    div r9
    dec r8
    add dl, 48
    mov byte ptr [r8], dl
    inc rcx
    test rax, rax
    jne decimal_loop

    mov rsi, r8
    rep movsb
    ret

.bss
.balign 16
inbuf:
    .zero 4096
numtmp:
    .zero 32
outbuf:
    .zero 128
