.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r13, [rip + input_buf]
    add r13, r12
    lea r12, [rip + input_buf]

line_loop:
    cmp r12, r13
    jae program_exit
    mov r14, r12

find_line_end:
    cmp r14, r13
    jae line_end_found
    cmp byte ptr [r14], 10
    je line_end_found
    inc r14
    jmp find_line_end

line_end_found:
    mov r15, r14
    cmp r14, r13
    jae have_next_line
    inc r15

have_next_line:
    mov rdi, r12
    mov rsi, r14
    call process_line
    mov r12, r15
    jmp line_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

process_line:
    xor r8d, r8d

next_token:
    cmp rdi, rsi
    jae line_finished
    mov al, byte ptr [rdi]
    cmp al, 32
    je skip_separator
    cmp al, 9
    jne token_start

skip_separator:
    inc rdi
    jmp next_token

token_start:
    mov rdx, rdi

scan_token:
    cmp rdi, rsi
    jae token_ready
    mov al, byte ptr [rdi]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc rdi
    jmp scan_token

token_ready:
    mov rcx, rdi
    sub rcx, rdx
    cmp rcx, 1
    jne parse_literal
    mov al, byte ptr [rdx]
    cmp al, 48
    jb check_single_operator
    cmp al, 57
    jbe parse_literal

check_single_operator:
    cmp al, 43
    je set_operator
    cmp al, 45
    je set_operator
    cmp al, 42
    je set_operator
    cmp al, 47
    je set_operator
    cmp al, 37
    je set_operator
    jmp line_error

set_operator:
    mov r10b, al
    jmp apply_operator

parse_literal:
    mov r11, rdx
    xor r10d, r10d
    cmp byte ptr [r11], 45
    jne count_digits
    cmp rcx, 1
    je line_error
    mov r10d, 1
    inc r11

count_digits:
    mov r9, rdi
    sub r9, r11
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor eax, eax

digit_loop:
    cmp r11, rdi
    jae literal_complete
    movzx ecx, byte ptr [r11]
    sub ecx, 48
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc r11
    jmp digit_loop

literal_complete:
    test r10d, r10d
    jz push_literal
    neg rax

push_literal:
    lea r11, [rip + value_stack]
    mov qword ptr [r11 + r8*8], rax
    inc r8
    jmp next_token

apply_operator:
    cmp r8, 2
    jb line_error
    lea r11, [rip + value_stack]
    mov rcx, qword ptr [r11 + r8*8 - 8]
    mov rax, qword ptr [r11 + r8*8 - 16]
    cmp r10b, 43
    je operator_add
    cmp r10b, 45
    je operator_subtract
    cmp r10b, 42
    je operator_multiply
    jmp operator_divide

operator_add:
    add rax, rcx
    jmp store_operator_result

operator_subtract:
    sub rax, rcx
    jmp store_operator_result

operator_multiply:
    imul rax, rcx
    jmp store_operator_result

operator_divide:
    test rcx, rcx
    jz line_error
    mov r9, 0x8000000000000000
    cmp rax, r9
    jne division_safe
    cmp rcx, -1
    je line_error

division_safe:
    cqo
    idiv rcx
    cmp r10b, 37
    jne store_operator_result
    mov rax, rdx

store_operator_result:
    mov qword ptr [r11 + r8*8 - 16], rax
    dec r8
    jmp next_token

line_finished:
    cmp r8, 1
    jne line_error
    mov rax, qword ptr [rip + value_stack]
    lea rsi, [rip + out_buf + 32]
    mov r10, rax
    mov rcx, rax
    test rcx, rcx
    jns magnitude_ready
    neg rcx

magnitude_ready:
    mov rax, rcx
    mov r9, 10

convert_digit:
    xor edx, edx
    div r9
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert_digit
    test r10, r10
    jns write_result
    dec rsi
    mov byte ptr [rsi], 45

write_result:
    lea rdx, [rip + out_buf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall
    ret

.section .rodata
error_message:
    .ascii "error\n"

.bss
.p2align 3
value_stack:
    .zero 8192
input_buf:
    .zero 65536
out_buf:
    .zero 32
