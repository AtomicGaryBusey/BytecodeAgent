.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    mov edx, 49152
    sub rdx, r12
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    cmp r12, 49152
    jb read_more

input_done:
    test r12, r12
    jz exit_success

    lea r13, [rip + input_buffer]
    lea r14, [rip + output_buffer]
    lea rbp, [rip + alphabet]
    mov rbx, r12
    xor r15d, r15d

encode_full:
    cmp rbx, 3
    jb encode_tail

    movzx eax, byte ptr [r13]
    shl eax, 16
    movzx ecx, byte ptr [r13 + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [r13 + 2]
    or eax, ecx

    mov ecx, eax
    shr ecx, 18
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14], dl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14 + 1], dl

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14 + 2], dl

    and eax, 63
    mov dl, byte ptr [rbp + rax]
    mov byte ptr [r14 + 3], dl

    add r13, 3
    add r14, 4
    sub rbx, 3
    add r15d, 4

    cmp r15d, 76
    jne encode_full
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d
    jmp encode_full

encode_tail:
    test rbx, rbx
    jz finish_output
    cmp rbx, 1
    jne tail_two

    movzx eax, byte ptr [r13]
    mov ecx, eax
    shr ecx, 2
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14], dl

    and eax, 3
    shl eax, 4
    mov dl, byte ptr [rbp + rax]
    mov byte ptr [r14 + 1], dl
    mov byte ptr [r14 + 2], 61
    mov byte ptr [r14 + 3], 61
    add r14, 4
    add r15d, 4
    jmp finish_output

tail_two:
    movzx eax, byte ptr [r13]
    shl eax, 8
    movzx ecx, byte ptr [r13 + 1]
    or eax, ecx

    mov ecx, eax
    shr ecx, 10
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14], dl

    mov ecx, eax
    shr ecx, 4
    and ecx, 63
    mov dl, byte ptr [rbp + rcx]
    mov byte ptr [r14 + 1], dl

    and eax, 15
    shl eax, 2
    mov dl, byte ptr [rbp + rax]
    mov byte ptr [r14 + 2], dl
    mov byte ptr [r14 + 3], 61
    add r14, 4
    add r15d, 4

finish_output:
    test r15d, r15d
    jz write_output
    mov byte ptr [r14], 10
    inc r14

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66560
