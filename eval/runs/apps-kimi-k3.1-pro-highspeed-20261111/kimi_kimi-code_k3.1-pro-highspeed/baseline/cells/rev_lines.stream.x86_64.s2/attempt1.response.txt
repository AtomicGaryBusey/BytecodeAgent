.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae read_done

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall

    test rax, rax
    jle read_done
    add r12, rax
    jmp read_more

read_done:
    lea r8, [rip + input_buffer]
    lea r9, [r8 + r12]
    mov r10, r8
    lea r13, [rip + output_buffer]

scan_input:
    cmp r8, r9
    jae input_done

    mov al, byte ptr [r8]
    inc r8
    cmp al, 10
    jne scan_input

    lea rsi, [r8 - 1]

reverse_terminated_line:
    cmp rsi, r10
    jbe terminated_line_done
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [r13], al
    inc r13
    jmp reverse_terminated_line

terminated_line_done:
    mov byte ptr [r13], 10
    inc r13
    mov r10, r8
    jmp scan_input

input_done:
    cmp r10, r9
    jae output_ready

    mov rsi, r9

reverse_final_line:
    cmp rsi, r10
    jbe final_line_done
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [r13], al
    inc r13
    jmp reverse_final_line

final_line_done:
    mov byte ptr [r13], 10
    inc r13

output_ready:
    lea rsi, [rip + output_buffer]
    mov r12, r13
    sub r12, rsi

write_more:
    test r12, r12
    jz exit_success

    mov eax, 1
    mov edi, 1
    mov rdx, r12
    syscall

    test rax, rax
    jle exit_success
    add rsi, rax
    sub r12, rax
    jmp write_more

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
