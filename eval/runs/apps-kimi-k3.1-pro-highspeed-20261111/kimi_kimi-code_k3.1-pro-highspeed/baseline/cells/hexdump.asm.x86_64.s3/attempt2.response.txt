.globl _start

.text
_start:
    xor r12d, r12d
    lea rbx, [rip + hex_digits]

read_input:
    cmp r12, 65536
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_input

input_done:
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    lea rdi, [rip + output_buffer]
    mov rax, r13
    mov ecx, 8

offset_digits:
    mov rdx, rax
    shr rdx, 28
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl rax, 4
    dec ecx
    jnz offset_digits

    mov word ptr [rdi], 0x2020
    add rdi, 2

    mov r8, r12
    sub r8, r13
    cmp r8, 16
    jbe row_size_ready
    mov r8d, 16
row_size_ready:

    lea r10, [rip + input_buffer]
    add r10, r13
    xor r9d, r9d

hex_slots:
    cmp r9, r8
    jae blank_slot

    movzx eax, byte ptr [r10 + r9]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp slot_done

blank_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

slot_done:
    inc r9
    cmp r9, 8
    jne no_extra_space
    mov byte ptr [rdi], 0x20
    inc rdi
no_extra_space:
    cmp r9, 16
    jb hex_slots

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2

    xor r9d, r9d
ascii_loop:
    cmp r9, r8
    jae ascii_done
    mov al, byte ptr [r10 + r9]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, 0x2e
ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc r9
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2

    mov eax, 1
    mov edx, edi
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov edi, 1
    syscall

    add r13, r8
    jmp row_loop

final_offset:
    lea rdi, [rip + output_buffer]
    mov rax, r12
    mov ecx, 8

final_digits:
    mov rdx, rax
    shr rdx, 28
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl rax, 4
    dec ecx
    jnz final_digits

    mov byte ptr [rdi], 0x0a
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 80
