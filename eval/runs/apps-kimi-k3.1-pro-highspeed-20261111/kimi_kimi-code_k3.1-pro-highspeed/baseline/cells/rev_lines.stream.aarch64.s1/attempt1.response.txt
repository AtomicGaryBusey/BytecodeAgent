.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    mov x20, #0

read_loop:
    mov x2, #1
    lsl x2, x2, #16
    cmp x20, x2
    b.eq process_input
    sub x2, x2, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, process_input
    add x20, x20, x0
    b read_loop

process_input:
    add x24, x19, x20
    mov x21, x19
    mov x22, x19

scan_loop:
    cmp x21, x24
    b.eq input_done
    ldrb w9, [x21], #1
    cmp w9, #10
    b.ne scan_loop
    sub x25, x21, #1
    bl emit_line
    mov x22, x21
    b scan_loop

input_done:
    cmp x22, x24
    b.eq exit_program
    mov x25, x24
    bl emit_line

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

emit_line:
    sub x10, x25, x22
    mov x11, x25
    mov x12, x23
    cbz x10, 1f

2:
    ldrb w13, [x11, #-1]!
    strb w13, [x12], #1
    subs x10, x10, #1
    b.ne 2b

1:
    mov w13, #10
    strb w13, [x12], #1
    sub x2, x12, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0
    ret

.bss
.balign 16

    .space 65536
output_buf:
