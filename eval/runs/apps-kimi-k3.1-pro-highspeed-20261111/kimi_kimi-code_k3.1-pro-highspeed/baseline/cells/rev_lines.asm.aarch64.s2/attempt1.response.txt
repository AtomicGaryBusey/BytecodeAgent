.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    movz x28, #1, lsl #16
    mov x20, #0

read_loop:
    cmp x20, x28
    b.hs input_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x28, x20
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x20, x20, x0
    b read_loop

input_done:
    mov x21, x19
    add x22, x19, x20
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf

next_line:
    cmp x21, x22
    b.hs finished
    mov x23, x21

find_line_end:
    cmp x21, x22
    b.hs line_ready
    ldrb w0, [x21]
    cmp w0, #10
    b.eq line_ready
    add x21, x21, #1
    b find_line_end

line_ready:
    mov x24, x21
    cmp x21, x22
    b.hs reverse_setup
    add x24, x21, #1

reverse_setup:
    mov x26, x21
    mov x27, x25

reverse_loop:
    cmp x26, x23
    b.ls append_newline
    sub x26, x26, #1
    ldrb w0, [x26]
    strb w0, [x27], #1
    b reverse_loop

append_newline:
    mov w0, #10
    strb w0, [x27], #1
    sub x2, x27, x25
    mov x0, #1
    mov x1, x25
    mov x8, #64
    svc #0
    mov x21, x24
    b next_line

finished:
    mov x0, #0
    mov x8, #93
    svc #0
    b finished

.bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 65537
