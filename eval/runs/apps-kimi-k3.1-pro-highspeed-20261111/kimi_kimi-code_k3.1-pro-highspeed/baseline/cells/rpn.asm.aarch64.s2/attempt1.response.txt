.globl _start

.section .rodata
err_msg:
    .ascii "error\n"
.equ err_msg_len, . - err_msg

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_more:
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x20
    cbz x2, input_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x20, x20, x0
    b read_more

input_done:
    mov x21, #0

next_line:
    cmp x21, x20
    b.hs program_exit
    mov x22, x21

find_line_end:
    cmp x22, x20
    b.hs line_end_found
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq line_end_found
    add x22, x22, #1
    b find_line_end

line_end_found:
    sub x1, x22, x21
    add x0, x19, x21
    cmp x22, x20
    b.hs last_line
    add x21, x22, #1
    b evaluate_current_line

last_line:
    mov x21, x22

evaluate_current_line:
    bl eval_line
    cbnz w0, print_error
    mov x0, x1
    bl print_value
    b next_line

print_error:
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #err_msg_len
    mov x8, #64
    svc #0
    b next_line

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

eval_line:
    mov x2, x0
    add x3, x0, x1
    mov x4, #0
    adrp x16, eval_stack
    add x16, x16, :lo12:eval_stack

eval_skip:
    cmp x2, x3
    b.hs eval_finish
    ldrb w7, [x2]
    cmp w7, #32
    b.eq eval_skip_one
    cmp w7, #9
    b.ne eval_token_start

eval_skip_one:
    add x2, x2, #1
    b eval_skip

eval_token_start:
    mov x5, x2

eval_token_scan:
    cmp x2, x3
    b.hs eval_token_ready
    ldrb w7, [x2]
    cmp w7, #32
    b.eq eval_token_ready
    cmp w7, #9
    b.eq eval_token_ready
    add x2, x2, #1
    b eval_token_scan

eval_token_ready:
    sub x6, x2, x5
    cmp x6, #1
    b.ne eval_literal
    ldrb w7, [x5]
    cmp w7, #43
    b.eq eval_operator
    cmp w7, #45
    b.eq eval_operator
    cmp w7, #42
    b.eq eval_operator
    cmp w7, #47
    b.eq eval_operator
    cmp w7, #37
    b.eq eval_operator

eval_literal:
    mov x9, #0
    mov x10, x5
    mov x11, #0
    ldrb w7, [x10]
    cmp w7, #45
    b.ne eval_digit_count
    mov x11, #1
    add x10, x10, #1

eval_digit_count:
    sub x8, x2, x10
    cbz x8, eval_error
    cmp x8, #18
    b.hi eval_error
    mov x12, #10

eval_digit_loop:
    ldrb w7, [x10], #1
    sub w17, w7, #48
    cmp w17, #9
    b.hi eval_error
    mul x9, x9, x12
    add x9, x9, x17
    subs x8, x8, #1
    b.ne eval_digit_loop
    cbz x11, eval_push_literal
    neg x9, x9

eval_push_literal:
    str x9, [x16, x4, lsl #3]
    add x4, x4, #1
    b eval_skip

eval_operator:
    cmp x4, #2
    b.lo eval_error
    sub x4, x4, #2
    ldr x13, [x16, x4, lsl #3]
    add x17, x4, #1
    ldr x14, [x16, x17, lsl #3]
    cmp w7, #43
    b.eq eval_do_add
    cmp w7, #45
    b.eq eval_do_sub
    cmp w7, #42
    b.eq eval_do_mul
    b eval_divmod

eval_do_add:
    add x15, x13, x14
    b eval_push_result

eval_do_sub:
    sub x15, x13, x14
    b eval_push_result

eval_do_mul:
    mul x15, x13, x14
    b eval_push_result

eval_divmod:
    cbz x14, eval_error
    cmn x14, #1
    b.ne eval_division_safe
    mov x17, #1
    lsl x17, x17, #63
    cmp x13, x17
    b.eq eval_error

eval_division_safe:
    sdiv x17, x13, x14
    cmp w7, #47
    b.eq eval_quotient
    msub x15, x17, x14, x13
    b eval_push_result

eval_quotient:
    mov x15, x17

eval_push_result:
    str x15, [x16, x4, lsl #3]
    add x4, x4, #1
    b eval_skip

eval_finish:
    cmp x4, #1
    b.ne eval_error
    ldr x1, [x16]
    mov w0, #0
    ret

eval_error:
    mov w0, #1
    ret

print_value:
    mov x4, #0
    cmp x0, #0
    b.ge print_value_abs_ready
    mov x4, #1
    neg x0, x0

print_value_abs_ready:
    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    add x2, x2, #32
    mov w5, #10
    strb w5, [x2, #-1]!
    mov x3, #10

print_value_digits:
    udiv x5, x0, x3
    msub x6, x5, x3, x0
    add w6, w6, #48
    strb w6, [x2, #-1]!
    mov x0, x5
    cbnz x0, print_value_digits
    cbz x4, print_value_write
    mov w5, #45
    strb w5, [x2, #-1]!

print_value_write:
    mov x0, x2
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #32
    sub x2, x1, x0
    mov x1, x0
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .bss
.balign 16
inbuf:
    .zero 65536
.balign 16
eval_stack:
    .zero 4096
.balign 16
outbuf:
    .zero 32
