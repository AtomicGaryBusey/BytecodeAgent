.globl _start

.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 393216

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, INPUT_SIZE
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    lea r13, [rip + input_buffer]
    lea r14, [rip + input_buffer]
    add r14, r12
    lea r12, [rip + output_buffer]

next_line:
    cmp r13, r14
    jae flush_output

    mov r15, r13
find_line_end:
    cmp r15, r14
    jae line_found
    cmp byte ptr [r15], 10
    je line_found
    inc r15
    jmp find_line_end

line_found:
    lea rbx, [rip + value_stack]

token_loop:
    cmp r13, r15
    jae finish_line
    mov al, byte ptr [r13]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    jmp token_start

skip_separator:
    inc r13
    jmp token_loop

token_start:
    mov rsi, r13

scan_token:
    cmp r13, r15
    jae token_scanned
    mov al, byte ptr [r13]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    inc r13
    jmp scan_token

token_scanned:
    mov rdx, r13
    sub rdx, rsi
    cmp rdx, 1
    jne try_literal

    mov al, byte ptr [rsi]
    cmp al, 43
    je operator_found
    cmp al, 45
    je operator_found
    cmp al, 42
    je operator_found
    cmp al, 47
    je operator_found
    cmp al, 37
    je operator_found

try_literal:
    xor r8d, r8d
    mov r9, rsi
    cmp byte ptr [r9], 45
    jne digits_ready
    mov r8d, 1
    inc r9
    dec rdx

digits_ready:
    test rdx, rdx
    jz line_error
    cmp rdx, 18
    ja line_error

    xor eax, eax

digit_loop:
    movzx ecx, byte ptr [r9]
    sub ecx, 48
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc r9
    dec rdx
    jnz digit_loop

    test r8d, r8d
    jz push_literal
    neg rax

push_literal:
    mov qword ptr [rbx], rax
    add rbx, 8
    jmp token_loop

operator_found:
    mov r8b, al
    lea rdx, [rip + value_stack + 16]
    cmp rbx, rdx
    jb line_error

    mov rax, qword ptr [rbx - 16]
    mov rcx, qword ptr [rbx - 8]
    sub rbx, 8

    cmp r8b, 43
    je do_add
    cmp r8b, 45
    je do_subtract
    cmp r8b, 42
    je do_multiply
    cmp r8b, 47
    je do_divide
    jmp do_remainder

do_add:
    add rax, rcx
    jmp store_result

do_subtract:
    sub rax, rcx
    jmp store_result

do_multiply:
    imul rax, rcx
    jmp store_result

do_divide:
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne divide_safe
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error

divide_safe:
    cqo
    idiv rcx
    jmp store_result

do_remainder:
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne remainder_safe
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error

remainder_safe:
    cqo
    idiv rcx
    mov rax, rdx

store_result:
    mov qword ptr [rbx - 8], rax
    jmp token_loop

finish_line:
    lea rax, [rip + value_stack + 8]
    cmp rbx, rax
    jne line_error

    mov rax, qword ptr [rip + value_stack]
    xor r8d, r8d
    test rax, rax
    jns convert_absolute
    mov r8d, 1
    neg rax

convert_absolute:
    lea r9, [rip + number_buffer + 32]
    mov rsi, r9
    mov ecx, 10

convert_digit:
    xor edx, edx
    div rcx
    dec rsi
    add dl, 48
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert_digit

    test r8d, r8d
    jz copy_number
    dec rsi
    mov byte ptr [rsi], 45

copy_number:
    mov rdi, r12
    mov rcx, r9
    sub rcx, rsi
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    mov r12, rdi
    jmp line_done

line_error:
    mov rdi, r12
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    mov r12, rdi

line_done:
    mov r13, r15
    cmp r13, r14
    jae next_line
    inc r13
    jmp next_line

flush_output:
    mov rdx, r12
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero INPUT_SIZE

.align 16
output_buffer:
    .zero OUTPUT_SIZE

.align 16
value_stack:
    .zero 8192

.align 16
number_buffer:
    .zero 32
