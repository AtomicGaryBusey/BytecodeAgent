.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x16, #1
    movk x16, #1, lsl #16

read_more:
    mov x0, #0
    add x1, x19, x20
    sub x2, x16, x20
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x20, x20, x0
    cmp x20, x16
    b.lo read_more

input_done:
    mov x21, #0

line_loop:
    cmp x21, x20
    b.hs program_exit
    add x22, x19, x21

find_line_end:
    cmp x21, x20
    b.hs final_line
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq newline_line
    add x21, x21, #1
    b find_line_end

newline_line:
    mov x0, x22
    add x1, x19, x21
    bl eval_line
    add x21, x21, #1
    b line_loop

final_line:
    mov x0, x22
    add x1, x19, x21
    bl eval_line
    b line_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

eval_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x2, x0
    mov x3, x1
    mov x4, #0
    adrp x5, value_stack
    add x5, x5, :lo12:value_stack

eval_loop:
    cmp x2, x3
    b.hs eval_finish
    ldrb w7, [x2]
    cmp w7, #32
    b.eq skip_separator
    cmp w7, #9
    b.eq skip_separator
    b token_begin

skip_separator:
    add x2, x2, #1
    b eval_loop

token_begin:
    mov x6, x2

token_scan:
    cmp x2, x3
    b.hs token_ready
    ldrb w7, [x2]
    cmp w7, #32
    b.eq token_ready
    cmp w7, #9
    b.eq token_ready
    add x2, x2, #1
    b token_scan

token_ready:
    sub x8, x2, x6
    cmp x8, #1
    b.ne integer_token
    ldrb w7, [x6]
    cmp w7, #43
    b.eq eval_operator
    cmp w7, #45
    b.eq eval_operator
    cmp w7, #42
    b.eq eval_operator
    cmp w7, #47
    b.eq eval_operator
    cmp w7, #37
    b.eq eval_operator

integer_token:
    mov x9, #0
    mov x11, x6
    ldrb w7, [x6]
    cmp w7, #45
    b.ne digits_setup
    mov x9, #1
    add x11, x11, #1
    sub x8, x8, #1

digits_setup:
    cmp x8, #1
    b.lo line_error
    cmp x8, #18
    b.hi line_error
    mov x12, #0

digit_loop:
    ldrb w7, [x11], #1
    sub w7, w7, #48
    cmp w7, #9
    b.hi line_error
    mov x13, #10
    mul x12, x12, x13
    add x12, x12, x7
    subs x8, x8, #1
    b.ne digit_loop
    cbz x9, push_integer
    neg x12, x12

push_integer:
    str x12, [x5, x4, lsl #3]
    add x4, x4, #1
    b eval_loop

eval_operator:
    cmp x4, #2
    b.lo line_error
    sub x4, x4, #2
    ldr x13, [x5, x4, lsl #3]
    add x12, x4, #1
    ldr x14, [x5, x12, lsl #3]
    cmp w7, #43
    b.eq calc_add
    cmp w7, #45
    b.eq calc_sub
    cmp w7, #42
    b.eq calc_mul
    cmp w7, #47
    b.eq calc_div
    b calc_rem

calc_add:
    add x15, x13, x14
    b store_operator_result

calc_sub:
    sub x15, x13, x14
    b store_operator_result

calc_mul:
    mul x15, x13, x14
    b store_operator_result

calc_div:
    cbz x14, line_error
    cmn x14, #1
    b.ne div_do
    mov x12, #1
    lsl x12, x12, #63
    cmp x13, x12
    b.eq line_error

div_do:
    sdiv x15, x13, x14
    b store_operator_result

calc_rem:
    cbz x14, line_error
    cmn x14, #1
    b.ne rem_do
    mov x12, #1
    lsl x12, x12, #63
    cmp x13, x12
    b.eq line_error

rem_do:
    sdiv x15, x13, x14
    msub x15, x15, x14, x13

store_operator_result:
    str x15, [x5, x4, lsl #3]
    add x4, x4, #1
    b eval_loop

eval_finish:
    cmp x4, #1
    b.ne line_error
    ldr x0, [x5]
    bl print_value
    b eval_return

line_error:
    bl print_error

eval_return:
    ldp x29, x30, [sp], #16
    ret

print_value:
    adrp x1, num_buf
    add x1, x1, :lo12:num_buf
    add x7, x1, #32
    mov x1, x7
    mov w2, #10
    strb w2, [x1, #-1]!
    mov x3, x0
    mov w4, #0
    mov x10, #10
    cmp x3, #0
    b.ge 1f
    mov w4, #1
    neg x3, x3

1:
    udiv x5, x3, x10
    msub x6, x5, x10, x3
    add w6, w6, #48
    strb w6, [x1, #-1]!
    mov x3, x5
    cbnz x3, 1b
    cbz w4, 2f
    mov w6, #45
    strb w6, [x1, #-1]!

2:
    sub x2, x7, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

print_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, #64
    svc #0
    ret
