.globl _start
.text
_start:
    xor r12d, r12d
.read_more:
    cmp r12, 65536
    jae .input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .input_ready
    add r12, rax
    jmp .read_more

.input_ready:
    lea rbx, [rip + input_buffer]
    lea rbp, [rbx + r12]
    lea r14, [rip + output_buffer]

.next_line:
    cmp rbx, rbp
    jae .all_done
    xor r13d, r13d

.skip_separators:
    cmp rbx, rbp
    jae .finish_line
    mov al, byte ptr [rbx]
    cmp al, 10
    je .line_newline
    cmp al, 32
    je .skip_one
    cmp al, 9
    je .skip_one

    mov rsi, rbx
.scan_token:
    cmp rbx, rbp
    jae .token_ready
    mov al, byte ptr [rbx]
    cmp al, 10
    je .token_ready
    cmp al, 32
    je .token_ready
    cmp al, 9
    je .token_ready
    inc rbx
    jmp .scan_token

.skip_one:
    inc rbx
    jmp .skip_separators

.token_ready:
    mov rdx, rbx
    sub rdx, rsi
    cmp rdx, 1
    jne .literal
    mov al, byte ptr [rsi]
    cmp al, 43
    je .operator
    cmp al, 45
    je .operator
    cmp al, 42
    je .operator
    cmp al, 47
    je .operator
    cmp al, 37
    je .operator

.literal:
    xor r8d, r8d
    xor r10d, r10d
    mov r9, rsi
    mov rcx, rdx
    cmp byte ptr [rsi], 45
    jne .digits_ready
    mov r10d, 1
    inc r9
    dec rcx
.digits_ready:
    cmp rcx, 18
    ja .line_error
.digit_loop:
    movzx eax, byte ptr [r9]
    sub eax, 48
    cmp eax, 9
    ja .line_error
    imul r8, r8, 10
    add r8, rax
    inc r9
    dec rcx
    jne .digit_loop
    test r10d, r10d
    jz .push_literal
    neg r8
.push_literal:
    lea r11, [rip + value_stack]
    mov qword ptr [r11 + r13*8], r8
    inc r13
    jmp .skip_separators

.operator:
    cmp r13, 2
    jb .line_error
    lea r11, [rip + value_stack]
    mov r8, qword ptr [r11 + r13*8 - 8]
    mov r9, qword ptr [r11 + r13*8 - 16]
    sub r13, 2
    cmp al, 43
    je .op_add
    cmp al, 45
    je .op_subtract
    cmp al, 42
    je .op_multiply
    cmp al, 47
    je .op_divide
    jmp .op_remainder

.op_add:
    add r8, r9
    jmp .push_operator
.op_subtract:
    sub r9, r8
    mov r8, r9
    jmp .push_operator
.op_multiply:
    imul r8, r9
    jmp .push_operator

.op_divide:
    test r8, r8
    jz .line_error
    mov rdx, 0x8000000000000000
    cmp r9, rdx
    jne .divide_now
    cmp r8, -1
    je .line_error
.divide_now:
    mov rax, r9
    cqo
    idiv r8
    mov r8, rax
    jmp .push_operator

.op_remainder:
    test r8, r8
    jz .line_error
    mov rdx, 0x8000000000000000
    cmp r9, rdx
    jne .remainder_now
    cmp r8, -1
    je .line_error
.remainder_now:
    mov rax, r9
    cqo
    idiv r8
    mov r8, rdx

.push_operator:
    mov qword ptr [r11 + r13*8], r8
    inc r13
    jmp .skip_separators

.line_newline:
    inc rbx
.finish_line:
    cmp r13, 1
    jne .line_error
    mov rax, qword ptr [rip + value_stack]
    xor r10d, r10d
    test rax, rax
    jns .magnitude_ready
    mov r10d, 1
    neg rax
.magnitude_ready:
    lea rdi, [rip + number_buffer + 32]
    xor ecx, ecx
    mov r8, 10
.convert_digit:
    xor edx, edx
    div r8
    dec rdi
    add dl, 48
    mov byte ptr [rdi], dl
    inc ecx
    test rax, rax
    jne .convert_digit
    test r10d, r10d
    jz .copy_number
    dec rdi
    mov byte ptr [rdi], 45
    inc ecx
.copy_number:
    mov edx, ecx
    mov rsi, rdi
    mov rdi, r14
    rep movsb
    add r14, rdx
    mov byte ptr [r14], 10
    inc r14
    jmp .next_line

.line_error:
    cmp rbx, rbp
    jae .emit_error
    cmp byte ptr [rbx], 10
    je .error_newline
    inc rbx
    jmp .line_error
.error_newline:
    inc rbx
.emit_error:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp .next_line

.all_done:
    lea rsi, [rip + output_buffer]
    mov rdx, r14
    sub rdx, rsi
    jz .exit_program
    mov eax, 1
    mov edi, 1
    syscall
.exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
.balign 16
output_buffer:
    .zero 393216
.balign 16
value_stack:
    .zero 8192
.balign 16
number_buffer:
    .zero 32
