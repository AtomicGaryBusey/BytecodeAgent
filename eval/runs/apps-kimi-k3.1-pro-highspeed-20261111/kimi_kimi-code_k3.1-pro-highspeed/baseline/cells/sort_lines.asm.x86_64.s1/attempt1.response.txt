.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    mov rdx, 32768
    sub rdx, r12
    jz input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall
    test rax, rax
    jz input_ready
    add r12, rax
    jmp read_more

input_ready:
    lea r13, [rip + input_buffer]
    mov rbx, r13
    lea r14, [r13 + r12]
    xor r15d, r15d
    lea r8, [rip + descriptors]

parse_loop:
    cmp rbx, r14
    je parsing_done
    mov r10, rbx

scan_line:
    cmp rbx, r14
    je line_ready
    cmp byte ptr [rbx], 10
    je line_ready
    inc rbx
    jmp scan_line

line_ready:
    mov rax, rbx
    sub rax, r10
    mov rcx, r15
    shl rcx, 4
    mov qword ptr [r8 + rcx], r10
    mov qword ptr [r8 + rcx + 8], rax
    inc r15
    cmp rbx, r14
    je parse_loop
    inc rbx
    jmp parse_loop

parsing_done:
    lea r14, [rip + descriptors]
    mov ebx, 1

sort_outer:
    cmp rbx, r15
    jae output_begin
    mov rax, rbx
    shl rax, 4
    mov r12, qword ptr [r14 + rax]
    mov r13, qword ptr [r14 + rax + 8]
    mov rbp, rbx

sort_inner:
    test rbp, rbp
    jz place_key
    mov rax, rbp
    dec rax
    shl rax, 4
    mov rdi, qword ptr [r14 + rax]
    mov rsi, qword ptr [r14 + rax + 8]
    mov rdx, r12
    mov rcx, r13
    call compare_lines
    test eax, eax
    jle place_key

    mov rax, rbp
    dec rax
    shl rax, 4
    mov rdx, qword ptr [r14 + rax]
    mov rcx, qword ptr [r14 + rax + 8]
    mov r9, rbp
    shl r9, 4
    mov qword ptr [r14 + r9], rdx
    mov qword ptr [r14 + r9 + 8], rcx
    dec rbp
    jmp sort_inner

place_key:
    mov rax, rbp
    shl rax, 4
    mov qword ptr [r14 + rax], r12
    mov qword ptr [r14 + rax + 8], r13
    inc rbx
    jmp sort_outer

compare_lines:
    mov r9, rsi
    cmp r9, rcx
    cmova r9, rcx
    xor r8d, r8d

compare_bytes:
    cmp r8, r9
    jae compare_lengths
    movzx eax, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp eax, r10d
    jb line_less
    ja line_greater
    inc r8
    jmp compare_bytes

compare_lengths:
    cmp rsi, rcx
    jb line_less
    ja line_greater
    xor eax, eax
    ret

line_less:
    mov eax, -1
    ret

line_greater:
    mov eax, 1
    ret

output_begin:
    xor ebx, ebx
    lea r14, [rip + descriptors]

output_loop:
    cmp rbx, r15
    jae finished
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r14 + rax]
    mov rdx, qword ptr [r14 + rax + 8]
    test rdx, rdx
    jz output_newline
    mov eax, 1
    mov edi, 1
    syscall

output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rbx
    jmp output_loop

finished:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
    .align 16
input_buffer:
    .zero 32768
    .align 16
descriptors:
    .zero 16000
