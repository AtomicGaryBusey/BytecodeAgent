.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_more:
    mov x0, #0
    add x1, x20, x19
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    b read_more

input_done:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, #0
    cbz x19, exit_success

next_line:
    mov x24, x23
    mov x25, x23

find_line_end:
    cmp x25, x19
    b.eq line_end_found
    ldrb w3, [x20, x25]
    cmp w3, #10
    b.eq line_end_found
    add x25, x25, #1
    b find_line_end

line_end_found:
    sub x26, x25, x24
    add x27, x20, x25
    sub x27, x27, #1
    cbz x26, reverse_done

reverse_byte:
    ldrb w3, [x27], #-1
    strb w3, [x22], #1
    subs x26, x26, #1
    b.ne reverse_byte

reverse_done:
    mov w3, #10
    strb w3, [x22], #1

    cmp x25, x19
    b.eq write_output
    add x23, x25, #1
    cmp x23, x19
    b.eq write_output
    b next_line

write_output:
    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 65537
