.equ INPUT_CAP, 32768

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #INPUT_CAP
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, #INPUT_CAP
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x24, line_ptrs
    add x24, x24, :lo12:line_ptrs
    adrp x25, line_lens
    add x25, x25, :lo12:line_lens
    mov x21, #0
    mov x22, #0

parse_loop:
    cmp x22, x20
    b.hs parse_done
    mov x23, x22

scan_line:
    cmp x22, x20
    b.hs final_line
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq newline_line
    add x22, x22, #1
    b scan_line

newline_line:
    add x0, x19, x23
    sub x3, x22, x23
    str x0, [x24, x21, lsl #3]
    str w3, [x25, x21, lsl #2]
    add x21, x21, #1
    add x22, x22, #1
    b parse_loop

final_line:
    add x0, x19, x23
    sub x3, x22, x23
    str x0, [x24, x21, lsl #3]
    str w3, [x25, x21, lsl #2]
    add x21, x21, #1

parse_done:
    cbz x21, exit_program
    mov x0, x21
    bl merge_sort
    mov x19, x0
    mov x20, x1
    mov x22, #0

output_loop:
    cmp x22, x21
    b.hs exit_program
    ldr x1, [x19, x22, lsl #3]
    ldr w2, [x20, x22, lsl #2]
    cbz x2, output_newline
    mov x0, #1
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x22, x22, #1
    b output_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

merge_sort:
    stp x29, x30, [sp, #-96]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    stp x27, x28, [sp, #80]

    mov x19, x0
    adrp x20, line_ptrs
    add x20, x20, :lo12:line_ptrs
    adrp x21, line_lens
    add x21, x21, :lo12:line_lens
    adrp x22, temp_ptrs
    add x22, x22, :lo12:temp_ptrs
    adrp x23, temp_lens
    add x23, x23, :lo12:temp_lens
    mov x24, #1

pass_loop:
    cmp x24, x19
    b.hs sort_done
    mov x25, #0

block_loop:
    cmp x25, x19
    b.hs pass_done
    add x28, x25, x24
    cmp x28, x19
    b.ls 1f
    mov x28, x19
1:
    add x29, x25, x24, lsl #1
    cmp x29, x19
    b.ls 2f
    mov x29, x19
2:
    mov x26, x25
    mov x27, x28

merge_loop:
    cmp x26, x28
    b.hs copy_right
    cmp x27, x29
    b.hs copy_left
    ldr x0, [x20, x26, lsl #3]
    ldr w1, [x21, x26, lsl #2]
    ldr x2, [x20, x27, lsl #3]
    ldr w3, [x21, x27, lsl #2]
    bl compare_lines
    cmp w0, #0
    b.gt take_right

take_left:
    ldr x0, [x20, x26, lsl #3]
    ldr w1, [x21, x26, lsl #2]
    str x0, [x22, x25, lsl #3]
    str w1, [x23, x25, lsl #2]
    add x26, x26, #1
    add x25, x25, #1
    b merge_loop

take_right:
    ldr x0, [x20, x27, lsl #3]
    ldr w1, [x21, x27, lsl #2]
    str x0, [x22, x25, lsl #3]
    str w1, [x23, x25, lsl #2]
    add x27, x27, #1
    add x25, x25, #1
    b merge_loop

copy_left:
    cmp x26, x28
    b.hs block_done
    ldr x0, [x20, x26, lsl #3]
    ldr w1, [x21, x26, lsl #2]
    str x0, [x22, x25, lsl #3]
    str w1, [x23, x25, lsl #2]
    add x26, x26, #1
    add x25, x25, #1
    b copy_left

copy_right:
    cmp x27, x29
    b.hs block_done
    ldr x0, [x20, x27, lsl #3]
    ldr w1, [x21, x27, lsl #2]
    str x0, [x22, x25, lsl #3]
    str w1, [x23, x25, lsl #2]
    add x27, x27, #1
    add x25, x25, #1
    b copy_right

block_done:
    mov x25, x29
    b block_loop

pass_done:
    mov x0, x20
    mov x20, x22
    mov x22, x0
    mov x0, x21
    mov x21, x23
    mov x23, x0
    lsl x24, x24, #1
    b pass_loop

sort_done:
    mov x0, x20
    mov x1, x21
    ldp x27, x28, [sp, #80]
    ldp x25, x26, [sp, #64]
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #96
    ret

compare_lines:
    cmp x1, x3
    csel x4, x1, x3, ls
    mov x5, #0

compare_loop:
    cmp x5, x4
    b.hs compare_prefix
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.lo compare_less
    b.hi compare_greater
    add x5, x5, #1
    b compare_loop

compare_prefix:
    cmp x1, x3
    b.lo compare_less
    b.hi compare_greater
    mov w0, #0
    ret

compare_less:
    mov w0, #-1
    ret

compare_greater:
    mov w0, #1
    ret

.section .rodata
newline:
    .byte 10

.bss
    .p2align 3
input_buf:
    .zero INPUT_CAP
line_ptrs:
    .zero 8000
line_lens:
    .zero 4000
temp_ptrs:
    .zero 8000
temp_lens:
    .zero 4000
