.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x18, #65536

read_loop:
    cmp x20, x18
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x18, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs print_total

    sub x22, x20, x21
    cmp x22, #16
    b.ls 1f
    mov x22, #16
1:
    add x23, x19, x21
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf

    mov x0, x21
    mov x1, x24
    bl put_hex8
    mov x24, x1

    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1

    adrp x2, hex_digits
    add x2, x2, :lo12:hex_digits
    mov w25, #0

slot_loop:
    cmp w25, w22
    b.hs blank_slot

    ldrb w0, [x23, x25]
    lsr w1, w0, #4
    ldrb w1, [x2, x1]
    strb w1, [x24], #1
    and w0, w0, #15
    ldrb w0, [x2, x0]
    strb w0, [x24], #1
    mov w0, #32
    strb w0, [x24], #1
    b slot_done

blank_slot:
    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1
    strb w0, [x24], #1

slot_done:
    cmp w25, #7
    b.ne 2f
    mov w0, #32
    strb w0, [x24], #1
2:
    add w25, w25, #1
    cmp w25, #16
    b.lo slot_loop

    mov w0, #32
    strb w0, [x24], #1
    mov w0, #124
    strb w0, [x24], #1

    mov x25, #0
ascii_loop:
    cmp x25, x22
    b.hs ascii_done
    ldrb w0, [x23, x25]
    cmp w0, #32
    b.lo ascii_dot
    cmp w0, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w0, #46
ascii_store:
    strb w0, [x24], #1
    add x25, x25, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x24, x1
    mov x0, #1
    mov x8, #64
    svc #0

    add x21, x21, x22
    b row_loop

print_total:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x0, x20
    bl put_hex8
    mov w0, #10
    strb w0, [x1], #1

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
1:
    b 1b

put_hex8:
    mov x3, #28
    adrp x6, hex_digits
    add x6, x6, :lo12:hex_digits
1:
    lsr x4, x0, x3
    and x4, x4, #15
    ldrb w5, [x6, x4]
    strb w5, [x1], #1
    subs x3, x3, #4
    b.pl 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 80
