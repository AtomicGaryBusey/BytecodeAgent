.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    je input_done

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall

    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    lea r13, [rip + input_buffer]
    lea r14, [r13 + r12]
    lea r15, [rip + output_buffer]

next_line:
    cmp r13, r14
    je write_output

    mov rbx, r13

find_end:
    cmp r13, r14
    je reverse_line
    cmp byte ptr [r13], 10
    je reverse_line
    inc r13
    jmp find_end

reverse_line:
    mov rsi, r13
    mov rcx, r13
    sub rcx, rbx
    test rcx, rcx
    jz append_newline

reverse_loop:
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [r15], al
    inc r15
    dec rcx
    jnz reverse_loop

append_newline:
    mov byte ptr [r15], 10
    inc r15

    cmp r13, r14
    je write_output
    inc r13
    jmp next_line

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r15
    sub rdx, rsi
    test rdx, rdx
    jz exit_success

    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
