.globl _start

.text
_start:
    xor ebx, ebx

read_loop:
    cmp rbx, 65536
    je process_input

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall

    test rax, rax
    jle process_input
    add rbx, rax
    jmp read_loop

process_input:
    lea rsi, [rip + input_buffer]
    lea r8, [rsi + rbx]
    lea rdi, [rip + output_buffer]

next_line:
    cmp rsi, r8
    je write_output

    mov rdx, rsi

find_line_end:
    cmp rsi, r8
    je reverse_line
    cmp byte ptr [rsi], 10
    je reverse_line
    inc rsi
    jmp find_line_end

reverse_line:
    mov rcx, rsi

reverse_loop:
    cmp rdx, rcx
    je append_newline
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    jmp reverse_loop

append_newline:
    mov byte ptr [rdi], 10
    inc rdi

    cmp rsi, r8
    je write_output
    inc rsi
    jmp next_line

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_success

    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
