.text
.globl _start
.p2align 2
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #32768

read_loop:
    cmp x20, x21
    b.hs read_full
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_full:
    mov x0, #0
    adrp x1, extra_byte
    add x1, x1, :lo12:extra_byte
    mov x2, #1
    mov x8, #63
    svc #0

read_done:
    mov x21, #0
    mov x22, #0
    adrp x23, line_desc
    add x23, x23, :lo12:line_desc

parse_loop:
    cmp x22, x20
    b.hs parse_done
    mov x24, x22
    mov x25, x22

scan_line:
    cmp x25, x20
    b.hs scan_done
    ldrb w0, [x19, x25]
    cmp w0, #10
    b.eq scan_done
    add x25, x25, #1
    b scan_line

scan_done:
    add x26, x23, x21, lsl #4
    add x27, x19, x24
    sub x28, x25, x24
    stp x27, x28, [x26]
    add x21, x21, #1
    cmp x25, x20
    b.hs 1f
    add x25, x25, #1
1:
    mov x22, x25
    b parse_loop

parse_done:
    mov x19, x23
    adrp x20, temp_desc
    add x20, x20, :lo12:temp_desc
    mov x22, #1
    cmp x21, #1
    b.ls output_setup

sort_outer:
    cmp x22, x21
    b.hs output_setup
    mov x23, #0

sort_runs:
    cmp x23, x21
    b.hs pass_done
    add x11, x23, x22
    cmp x11, x21
    b.ls 1f
    mov x11, x21
1:
    add x12, x11, x22
    cmp x12, x21
    b.ls 2f
    mov x12, x21
2:
    mov x24, x12
    add x25, x19, x23, lsl #4
    add x26, x19, x11, lsl #4
    mov x27, x26
    add x28, x19, x24, lsl #4
    add x10, x20, x23, lsl #4

merge_loop:
    cmp x25, x26
    b.hs copy_right
    cmp x27, x28
    b.hs copy_left
    mov x0, x25
    mov x1, x27
    bl compare_lines
    cmp w0, #0
    b.gt take_right
    ldp x0, x1, [x25], #16
    stp x0, x1, [x10], #16
    b merge_loop

take_right:
    ldp x0, x1, [x27], #16
    stp x0, x1, [x10], #16
    b merge_loop

copy_left:
    cmp x25, x26
    b.hs run_done
    ldp x0, x1, [x25], #16
    stp x0, x1, [x10], #16
    b copy_left

copy_right:
    cmp x27, x28
    b.hs run_done
    ldp x0, x1, [x27], #16
    stp x0, x1, [x10], #16
    b copy_right

run_done:
    mov x23, x24
    b sort_runs

pass_done:
    mov x0, x19
    mov x19, x20
    mov x20, x0
    lsl x22, x22, #1
    b sort_outer

compare_lines:
    ldp x2, x3, [x0]
    ldp x4, x5, [x1]
    cmp x3, x5
    csel x6, x3, x5, lo
    mov x7, #0

compare_bytes:
    cmp x7, x6
    b.hs compare_lengths
    ldrb w8, [x2, x7]
    ldrb w9, [x4, x7]
    cmp w8, w9
    b.lo line_less
    b.hi line_greater
    add x7, x7, #1
    b compare_bytes

compare_lengths:
    cmp x3, x5
    b.lo line_less
    b.hi line_greater
    mov w0, #0
    ret

line_less:
    mov w0, #-1
    ret

line_greater:
    mov w0, #1
    ret

output_setup:
    mov x23, x21
    adrp x20, newline
    add x20, x20, :lo12:newline
    cbz x23, exit_success

output_loop:
    ldp x1, x2, [x19], #16
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x1, x20
    mov x2, #1
    mov x8, #64
    svc #0
    subs x23, x23, #1
    b.ne output_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
