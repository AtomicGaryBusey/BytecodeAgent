.section .rodata
.balign 1
newline_char:
.byte 10

.bss
.balign 16
input_buffer:
.zero 32768
line_table:
.zero 16000

.text
.globl _start
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	mov x20, #0

read_input:
	movz x2, #0x8000
	sub x2, x2, x20
	cbz x2, input_complete
	mov x0, #0
	add x1, x19, x20
	mov x8, #63
	svc #0
	cbz x0, input_complete
	add x20, x20, x0
	b read_input

input_complete:
	cbz x20, exit_success

	adrp x21, line_table
	add x21, x21, :lo12:line_table
	mov x22, #0
	mov x23, x19
	add x24, x19, x20
	mov x25, x19

parse_lines:
	cmp x23, x24
	b.eq final_line
	ldrb w0, [x23], #1
	cmp w0, #10
	b.ne parse_lines

	sub x26, x23, x25
	sub x26, x26, #1
	add x2, x21, x22, lsl #4
	stp x25, x26, [x2]
	add x22, x22, #1
	mov x25, x23
	b parse_lines

final_line:
	cmp x25, x24
	b.eq sort_begin
	sub x26, x24, x25
	add x2, x21, x22, lsl #4
	stp x25, x26, [x2]
	add x22, x22, #1

sort_begin:
	cmp x22, #2
	b.lo output_begin
	mov x23, #1

sort_outer:
	cmp x23, x22
	b.hs output_begin
	add x4, x21, x23, lsl #4
	ldp x24, x25, [x4]
	mov x26, x23

sort_inner:
	cbz x26, sort_place
	sub x5, x26, #1
	add x6, x21, x5, lsl #4
	ldp x0, x1, [x6]
	mov x2, x24
	mov x3, x25
	bl compare_lines
	cmp w0, #0
	b.le sort_place

	sub x5, x26, #1
	add x6, x21, x5, lsl #4
	add x7, x21, x26, lsl #4
	ldp x8, x9, [x6]
	stp x8, x9, [x7]
	sub x26, x26, #1
	b sort_inner

sort_place:
	add x7, x21, x26, lsl #4
	stp x24, x25, [x7]
	add x23, x23, #1
	b sort_outer

compare_lines:
	cmp x1, x3
	csel x4, x1, x3, lo
	mov x5, #0

compare_words:
	sub x8, x4, x5
	cmp x8, #8
	b.lo compare_bytes
	ldr x6, [x0, x5]
	ldr x7, [x2, x5]
	cmp x6, x7
	b.ne compare_bytes
	add x5, x5, #8
	b compare_words

compare_bytes:
	cmp x5, x4
	b.eq compare_lengths
	ldrb w6, [x0, x5]
	ldrb w7, [x2, x5]
	cmp w6, w7
	b.ne compare_different
	add x5, x5, #1
	b compare_bytes

compare_different:
	sub w0, w6, w7
	ret

compare_lengths:
	cmp x1, x3
	b.lo compare_less
	b.hi compare_greater
	mov w0, #0
	ret

compare_less:
	mov w0, #-1
	ret

compare_greater:
	mov w0, #1
	ret

output_begin:
	mov x23, #0

output_loop:
	cmp x23, x22
	b.hs exit_success
	add x4, x21, x23, lsl #4
	ldp x1, x2, [x4]
	cbz x2, output_newline
	mov x0, #1
	mov x8, #64
	svc #0

output_newline:
	mov x0, #1
	adrp x1, newline_char
	add x1, x1, :lo12:newline_char
	mov x2, #1
	mov x8, #64
	svc #0
	add x23, x23, #1
	b output_loop

exit_success:
	mov x0, #0
	mov x8, #93
	svc #0
