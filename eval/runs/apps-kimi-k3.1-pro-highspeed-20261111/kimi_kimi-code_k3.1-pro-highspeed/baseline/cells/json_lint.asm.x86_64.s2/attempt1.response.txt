.globl _start

.text
_start:
    xor r12d, r12d
read_loop:
    cmp r12, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov r14, r12
    lea r15, [rip + input_buf]
    xor r13d, r13d
    call skip_ws
    call parse_value
    test eax, eax
    jz report_error
    call skip_ws
    cmp r13, r14
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

report_error:
    lea rdi, [rip + output_buf]
    mov rax, qword ptr [rip + error_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + error_prefix + 8]
    mov byte ptr [rdi + 8], al

    mov rax, r13
    mov r8d, 10
    lea rcx, [rip + output_buf + 32]
convert_digit:
    xor edx, edx
    div r8
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz convert_digit

    lea rsi, [rip + output_buf + 32]
    sub rsi, rcx
    lea rdi, [rip + output_buf + 9]
    mov r8, rsi
    mov rsi, rcx
    mov rcx, r8
    rep movsb
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rdx, [rip + output_buf]
    sub rdi, rdx
    mov rdx, rdi

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buf]
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r13, r14
    jae skip_ws_done
    mov al, byte ptr [r15 + r13]
    cmp al, 0x20
    je skip_ws_one
    cmp al, 0x09
    je skip_ws_one
    cmp al, 0x0a
    je skip_ws_one
    cmp al, 0x0d
    jne skip_ws_done
skip_ws_one:
    inc r13
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je value_true
    cmp al, 0x66
    je value_false
    cmp al, 0x6e
    je value_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    jbe parse_number
    jmp parse_fail

value_true:
    lea rsi, [rip + kw_true]
    mov edx, 4
    jmp parse_keyword
value_false:
    lea rsi, [rip + kw_false]
    mov edx, 5
    jmp parse_keyword
value_null:
    lea rsi, [rip + kw_null]
    mov edx, 4
    jmp parse_keyword

parse_keyword:
    xor ecx, ecx
keyword_loop:
    cmp ecx, edx
    jae parse_ok
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [rsi + rcx]
    cmp al, byte ptr [r15 + r13]
    jne parse_fail
    inc r13
    inc ecx
    jmp keyword_loop

parse_string:
    inc r13
string_loop:
    cmp r13, r14
    jae parse_fail
    movzx eax, byte ptr [r15 + r13]
    cmp al, 0x22
    je string_close
    cmp al, 0x5c
    je string_escape
    cmp al, 0x20
    jb parse_fail
    inc r13
    jmp string_loop

string_close:
    inc r13
    jmp parse_ok

string_escape:
    inc r13
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je string_escape_ok
    cmp al, 0x5c
    je string_escape_ok
    cmp al, 0x2f
    je string_escape_ok
    cmp al, 0x62
    je string_escape_ok
    cmp al, 0x66
    je string_escape_ok
    cmp al, 0x6e
    je string_escape_ok
    cmp al, 0x72
    je string_escape_ok
    cmp al, 0x74
    jne parse_fail
string_escape_ok:
    inc r13
    jmp string_loop

string_unicode:
    inc r13
    mov ecx, 4
string_hex_loop:
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    jbe string_hex_ok
    or al, 0x20
    cmp al, 0x61
    jb parse_fail
    cmp al, 0x66
    ja parse_fail
string_hex_ok:
    inc r13
    dec ecx
    jnz string_hex_loop
    jmp string_loop

parse_number:
    mov al, byte ptr [r15 + r13]
    cmp al, 0x2d
    jne number_int
    inc r13

number_int:
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb parse_fail
    cmp al, 0x39
    ja parse_fail
    inc r13

number_int_digits:
    cmp r13, r14
    jae parse_ok
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    jb number_after_int
    cmp al, 0x39
    ja number_after_int
    inc r13
    jmp number_int_digits

number_zero:
    inc r13
    cmp r13, r14
    jae parse_ok
    mov al, byte ptr [r15 + r13]

number_after_int:
    cmp al, 0x2e
    je number_frac
    cmp al, 0x65
    je number_exp
    cmp al, 0x45
    je number_exp
    jmp parse_ok

number_frac:
    inc r13
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    ja parse_fail

number_frac_digits:
    inc r13
    cmp r13, r14
    jae parse_ok
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    jb number_frac_end
    cmp al, 0x39
    jbe number_frac_digits
number_frac_end:
    cmp al, 0x65
    je number_exp
    cmp al, 0x45
    je number_exp
    jmp parse_ok

number_exp:
    inc r13
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x2b
    je number_exp_sign
    cmp al, 0x2d
    jne number_exp_require_digit
number_exp_sign:
    inc r13

number_exp_require_digit:
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    ja parse_fail

number_exp_digits:
    inc r13
    cmp r13, r14
    jae parse_ok
    mov al, byte ptr [r15 + r13]
    cmp al, 0x30
    jb parse_ok
    cmp al, 0x39
    jbe number_exp_digits
    jmp parse_ok

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae parse_fail
    cmp byte ptr [r15 + r13], 0x7d
    je object_close

object_member:
    cmp r13, r14
    jae parse_fail
    cmp byte ptr [r15 + r13], 0x22
    jne parse_fail
    call parse_string
    test eax, eax
    jz parse_fail
    call skip_ws
    cmp r13, r14
    jae parse_fail
    cmp byte ptr [r15 + r13], 0x3a
    jne parse_fail
    inc r13
    call skip_ws
    call parse_value
    test eax, eax
    jz parse_fail
    call skip_ws
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x7d
    je object_close
    cmp al, 0x2c
    jne parse_fail
    inc r13
    call skip_ws
    jmp object_member

object_close:
    inc r13
    jmp parse_ok

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae parse_fail
    cmp byte ptr [r15 + r13], 0x5d
    je array_close

array_value:
    call parse_value
    test eax, eax
    jz parse_fail
    call skip_ws
    cmp r13, r14
    jae parse_fail
    mov al, byte ptr [r15 + r13]
    cmp al, 0x5d
    je array_close
    cmp al, 0x2c
    jne parse_fail
    inc r13
    call skip_ws
    jmp array_value

array_close:
    inc r13
    jmp parse_ok

parse_ok:
    mov eax, 1
    ret

parse_fail:
    xor eax, eax
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
kw_true:
    .ascii "true"
kw_false:
    .ascii "false"
kw_null:
    .ascii "null"

.bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 32
