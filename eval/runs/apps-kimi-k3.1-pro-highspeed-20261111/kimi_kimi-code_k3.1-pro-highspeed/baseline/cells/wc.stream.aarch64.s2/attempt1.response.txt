.globl _start

.text
_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov w22, wzr

read_more:
    mov x0, xzr
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x21, x21, x0
    mov x2, x0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf

process_byte:
    ldrb w3, [x1], #1
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace

    cbnz w22, byte_done
    add x20, x20, #1
    mov w22, #1
    b byte_done

whitespace:
    cmp w3, #10
    b.ne clear_word
    add x19, x19, #1
clear_word:
    mov w22, wzr

byte_done:
    subs x2, x2, #1
    b.ne process_byte
    b read_more

input_done:
    adrp x23, output_buf
    add x23, x23, :lo12:output_buf
    mov x1, x23

    mov x0, x19
    bl append_decimal
    mov w3, #32
    strb w3, [x1], #1

    mov x0, x20
    bl append_decimal
    mov w3, #32
    strb w3, [x1], #1

    mov x0, x21
    bl append_decimal
    mov w3, #10
    strb w3, [x1], #1

    sub x2, x1, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0

    mov x0, xzr
    mov x8, #93
    svc #0

append_decimal:
    mov x2, #10
    mov x4, x1

decimal_digits:
    udiv x5, x0, x2
    msub x6, x5, x2, x0
    add w6, w6, #48
    strb w6, [x1], #1
    mov x0, x5
    cbnz x0, decimal_digits

    sub x5, x1, #1
reverse_digits:
    cmp x4, x5
    b.hs decimal_done
    ldrb w6, [x4]
    ldrb w7, [x5]
    strb w7, [x4], #1
    strb w6, [x5]
    sub x5, x5, #1
    b reverse_digits

decimal_done:
    ret

.bss
.balign 16
input_buf:
    .space 65536
output_buf:
    .space 64
