.section .rodata
newline:
    .byte 10

.bss
    .align 16
input_buffer:
    .zero 32768
    .align 16
line_array:
    .zero 16000
    .align 16
temp_array:
    .zero 16000
    .align 8
merge_mid:
    .zero 8
merge_right:
    .zero 8
merge_j:
    .zero 8
merge_k:
    .zero 8

.text
    .globl _start
_start:
    xor ebp, ebp

read_loop:
    cmp ebp, 32768
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbp
    mov edx, 32768
    sub edx, ebp
    syscall
    test rax, rax
    jle read_done
    add rbp, rax
    jmp read_loop

read_done:
    test rbp, rbp
    jz exit_program

    xor ebx, ebx
    xor r12d, r12d
    xor r14d, r14d

parse_loop:
    cmp rbx, rbp
    jae parse_end
    lea rax, [rip + input_buffer]
    cmp byte ptr [rax + rbx], 10
    jne parse_next

    mov rax, r12
    shl rax, 4
    lea rdi, [rip + line_array]
    mov qword ptr [rdi + rax], r14
    mov rdx, rbx
    sub rdx, r14
    mov qword ptr [rdi + rax + 8], rdx
    inc r12
    lea r14, [rbx + 1]

parse_next:
    inc rbx
    jmp parse_loop

parse_end:
    cmp r14, rbp
    jae parsed
    mov rax, r12
    shl rax, 4
    lea rdi, [rip + line_array]
    mov qword ptr [rdi + rax], r14
    mov rdx, rbp
    sub rdx, r14
    mov qword ptr [rdi + rax + 8], rdx
    inc r12

parsed:
    lea r14, [rip + line_array]
    lea r15, [rip + temp_array]
    mov r13d, 1

sort_pass:
    cmp r13, r12
    jae sorted
    xor ebx, ebx

merge_run:
    cmp rbx, r12
    jae pass_end

    mov rbp, rbx
    lea rax, [rbx + r13]
    cmp rax, r12
    cmova rax, r12
    mov qword ptr [rip + merge_mid], rax
    mov qword ptr [rip + merge_j], rax

    lea rcx, [rax + r13]
    cmp rcx, r12
    cmova rcx, r12
    mov qword ptr [rip + merge_right], rcx
    mov qword ptr [rip + merge_k], rbx

merge_loop:
    cmp rbp, qword ptr [rip + merge_mid]
    jae right_only
    mov rax, qword ptr [rip + merge_j]
    cmp rax, qword ptr [rip + merge_right]
    jae take_left

    mov rcx, rbp
    shl rcx, 4
    lea rdi, [r14 + rcx]
    mov rcx, rax
    shl rcx, 4
    lea rsi, [r14 + rcx]
    call compare_lines
    test eax, eax
    jle take_left
    jmp take_right

right_only:
    mov rax, qword ptr [rip + merge_j]
    cmp rax, qword ptr [rip + merge_right]
    jae run_done

take_right:
    mov rax, qword ptr [rip + merge_j]
    mov rcx, rax
    shl rcx, 4
    mov rdx, qword ptr [r14 + rcx]
    mov r8, qword ptr [r14 + rcx + 8]
    mov rcx, qword ptr [rip + merge_k]
    shl rcx, 4
    mov qword ptr [r15 + rcx], rdx
    mov qword ptr [r15 + rcx + 8], r8
    inc qword ptr [rip + merge_j]
    inc qword ptr [rip + merge_k]
    jmp merge_loop

take_left:
    mov rcx, rbp
    shl rcx, 4
    mov rdx, qword ptr [r14 + rcx]
    mov r8, qword ptr [r14 + rcx + 8]
    mov rcx, qword ptr [rip + merge_k]
    shl rcx, 4
    mov qword ptr [r15 + rcx], rdx
    mov qword ptr [r15 + rcx + 8], r8
    inc rbp
    inc qword ptr [rip + merge_k]
    jmp merge_loop

run_done:
    mov rbx, qword ptr [rip + merge_right]
    jmp merge_run

pass_end:
    xchg r14, r15
    shl r13, 1
    jmp sort_pass

sorted:
    xor ebx, ebx

output_loop:
    cmp rbx, r12
    jae exit_program
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r14 + rax]
    lea rcx, [rip + input_buffer]
    add rsi, rcx
    mov rdx, qword ptr [r14 + rax + 8]
    call write_all

    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    inc rbx
    jmp output_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rsi]
    mov rcx, qword ptr [rdi + 8]
    mov rdx, qword ptr [rsi + 8]
    mov r10, rcx
    cmp r10, rdx
    cmova r10, rdx
    lea r11, [rip + input_buffer]
    add r8, r11
    add r9, r11
    xor eax, eax

compare_bytes:
    cmp rax, r10
    jae compare_lengths
    movzx r11d, byte ptr [r8 + rax]
    movzx edi, byte ptr [r9 + rax]
    cmp r11d, edi
    jb line_less
    ja line_greater
    inc rax
    jmp compare_bytes

compare_lengths:
    cmp rcx, rdx
    jb line_less
    ja line_greater
    xor eax, eax
    ret

line_less:
    mov eax, -1
    ret

line_greater:
    mov eax, 1
    ret

write_all:
    test rdx, rdx
    jz write_done

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop

write_done:
    ret
