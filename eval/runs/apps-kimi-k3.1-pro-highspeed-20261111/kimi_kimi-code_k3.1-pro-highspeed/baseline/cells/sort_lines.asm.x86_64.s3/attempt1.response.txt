.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    cmp r12, 32768
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + input_buf]
    mov rsi, rbx
    lea r13, [rbx + r12]
    lea r15, [rip + lines]
    xor r14d, r14d

parse_loop:
    cmp rsi, r13
    jae parse_end
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne parse_loop

    mov qword ptr [r15], rbx
    mov rdx, rsi
    sub rdx, rbx
    dec rdx
    mov qword ptr [r15 + 8], rdx
    add r15, 16
    inc r14
    mov rbx, rsi
    jmp parse_loop

parse_end:
    cmp rbx, r13
    jae parsed
    mov qword ptr [r15], rbx
    mov rdx, r13
    sub rdx, rbx
    mov qword ptr [r15 + 8], rdx
    inc r14

parsed:
    cmp r14, 2
    jb output_begin
    xor edi, edi
    mov rsi, r14
    call sort_range

output_begin:
    lea rdi, [rip + output_buf]
    xor r12d, r12d
    test r14, r14
    jz output_done

output_lines:
    mov rax, r12
    shl rax, 4
    lea rbx, [rip + lines]
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb
    mov al, 10
    stosb
    inc r12
    cmp r12, r14
    jb output_lines

output_done:
    lea r12, [rip + output_buf]
    mov r13, rdi
    sub r13, r12

write_loop:
    test r13, r13
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle exit_program
    add r12, rax
    sub r13, rax
    jmp write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_program

# Sort records [rdi, rsi) in ascending line order.
sort_range:
    lea rax, [rdi + 1]
    cmp rax, rsi
    jae sort_return

    push rbp
    mov rbp, rsp
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    mov r12, rdi
    mov r13, rsi
    lea r14, [rdi + rsi]
    shr r14, 1

    mov rdi, r12
    mov rsi, r14
    call sort_range
    mov rdi, r14
    mov rsi, r13
    call sort_range

    mov r15, r12
    mov rbx, r14
    mov qword ptr [rbp - 48], r12

merge_loop:
    cmp r15, r14
    jae merge_left_done
    cmp rbx, r13
    jae merge_right_done

    mov rax, r15
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    mov rax, rbx
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    call compare_records
    test eax, eax
    jg take_right

take_left:
    mov rax, r15
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    mov rdx, qword ptr [rbp - 48]
    mov rcx, rdx
    shl rcx, 4
    lea rdi, [rip + temp_lines]
    add rdi, rcx
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc r15
    inc rdx
    mov qword ptr [rbp - 48], rdx
    jmp merge_loop

take_right:
    mov rax, rbx
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    mov rdx, qword ptr [rbp - 48]
    mov rcx, rdx
    shl rcx, 4
    lea rdi, [rip + temp_lines]
    add rdi, rcx
    mov rax, qword ptr [rsi]
    mov qword ptr [rdi], rax
    mov rax, qword ptr [rsi + 8]
    mov qword ptr [rdi + 8], rax
    inc rbx
    inc rdx
    mov qword ptr [rbp - 48], rdx
    jmp merge_loop

merge_left_done:
    cmp rbx, r13
    jae merge_done
    jmp take_right

merge_right_done:
    cmp r15, r14
    jae merge_done
    jmp take_left

merge_done:
    mov rcx, r13
    sub rcx, r12
    shl rcx, 1
    mov rax, r12
    shl rax, 4
    lea rsi, [rip + temp_lines]
    add rsi, rax
    lea rdi, [rip + lines]
    add rdi, rax
    rep movsq

    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp

sort_return:
    ret

# Compare the records at rdi and rsi. Return -1, 0, or 1 in eax.
compare_records:
    mov r8, qword ptr [rdi + 8]
    mov r9, qword ptr [rsi + 8]
    mov rcx, r8
    cmp rcx, r9
    cmova rcx, r9
    mov rdi, qword ptr [rdi]
    mov rsi, qword ptr [rsi]
    test rcx, rcx
    jz compare_prefix
    repe cmpsb
    jne compare_different

compare_prefix:
    xor eax, eax
    cmp r8, r9
    je compare_return
    jb compare_less
    mov eax, 1
    ret

compare_different:
    movzx eax, byte ptr [rdi - 1]
    movzx edx, byte ptr [rsi - 1]
    cmp eax, edx
    jb compare_less
    mov eax, 1
    ret

compare_less:
    mov eax, -1

compare_return:
    ret

.bss
.p2align 4
input_buf:
    .zero 32768
lines:
    .zero 16000
temp_lines:
    .zero 16000
output_buf:
    .zero 33768
