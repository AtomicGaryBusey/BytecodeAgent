.globl _start

.text
_start:
    xor r12d, r12d

.read_input:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .input_done
    add r12, rax
    cmp r12, 65536
    jb .read_input

.input_done:
    lea r13, [rip + input_buffer]
    mov r14, r12
    xor r15d, r15d

    call parse_value
    test eax, eax
    jz report_error

    call skip_ws
    cmp r15, r14
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall

    xor edi, edi
    mov eax, 60
    syscall

report_error:
    lea rbx, [rip + output_buffer + 32]
    dec rbx
    mov byte ptr [rbx], 10

    mov rax, r15
    mov ecx, 10

.convert_offset:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz .convert_offset

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, 9
    syscall

    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    lea rdx, [rip + output_buffer + 32]
    sub rdx, rbx
    syscall

    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    cmp r15, r14
    jae .sw_done
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x20
    je .sw_consume
    cmp al, 0x09
    je .sw_consume
    cmp al, 0x0a
    je .sw_consume
    cmp al, 0x0d
    jne .sw_done
.sw_consume:
    inc r15
    jmp skip_ws
.sw_done:
    ret

parse_value:
    call skip_ws
    cmp r15, r14
    jae .pv_fail

    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x7b
    je .pv_object
    cmp al, 0x5b
    je .pv_array
    cmp al, 0x22
    je .pv_string
    cmp al, 0x74
    je .pv_true
    cmp al, 0x66
    je .pv_false
    cmp al, 0x6e
    je .pv_null
    cmp al, 0x2d
    je .pv_number
    cmp al, 0x30
    jb .pv_fail
    cmp al, 0x39
    jbe .pv_number

.pv_fail:
    xor eax, eax
    ret

.pv_object:
    jmp parse_object
.pv_array:
    jmp parse_array
.pv_string:
    jmp parse_string
.pv_number:
    jmp parse_number
.pv_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal
.pv_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal
.pv_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4
    jmp parse_literal

parse_literal:
    xor edx, edx
.pl_loop:
    cmp r15, r14
    jae .pl_fail
    movzx eax, byte ptr [r13 + r15]
    cmp al, byte ptr [rsi + rdx]
    jne .pl_fail
    inc r15
    inc edx
    cmp edx, ecx
    jb .pl_loop
    mov eax, 1
    ret
.pl_fail:
    xor eax, eax
    ret

parse_string:
    cmp r15, r14
    jae .ps_fail
    cmp byte ptr [r13 + r15], 0x22
    jne .ps_fail
    inc r15

.ps_loop:
    cmp r15, r14
    jae .ps_fail
    movzx eax, byte ptr [r13 + r15]

    cmp al, 0x22
    je .ps_close
    cmp al, 0x5c
    je .ps_escape
    cmp al, 0x20
    jb .ps_fail

    inc r15
    jmp .ps_loop

.ps_close:
    inc r15
    mov eax, 1
    ret

.ps_escape:
    inc r15
    cmp r15, r14
    jae .ps_fail
    movzx eax, byte ptr [r13 + r15]

    cmp al, 0x22
    je .ps_simple_escape
    cmp al, 0x5c
    je .ps_simple_escape
    cmp al, 0x2f
    je .ps_simple_escape
    cmp al, 0x62
    je .ps_simple_escape
    cmp al, 0x66
    je .ps_simple_escape
    cmp al, 0x6e
    je .ps_simple_escape
    cmp al, 0x72
    je .ps_simple_escape
    cmp al, 0x74
    je .ps_simple_escape
    cmp al, 0x75
    je .ps_unicode_escape
    jmp .ps_fail

.ps_simple_escape:
    inc r15
    jmp .ps_loop

.ps_unicode_escape:
    inc r15
    mov ecx, 4

.ps_hex_loop:
    cmp r15, r14
    jae .ps_fail
    movzx eax, byte ptr [r13 + r15]

    cmp al, 0x30
    jb .ps_fail
    cmp al, 0x39
    jbe .ps_hex_valid
    cmp al, 0x41
    jb .ps_fail
    cmp al, 0x46
    jbe .ps_hex_valid
    cmp al, 0x61
    jb .ps_fail
    cmp al, 0x66
    ja .ps_fail

.ps_hex_valid:
    inc r15
    dec ecx
    jnz .ps_hex_loop
    jmp .ps_loop

.ps_fail:
    xor eax, eax
    ret

parse_number:
    cmp byte ptr [r13 + r15], 0x2d
    jne .pn_integer
    inc r15

.pn_integer:
    cmp r15, r14
    jae .pn_fail
    movzx eax, byte ptr [r13 + r15]

    cmp al, 0x30
    je .pn_zero
    cmp al, 0x31
    jb .pn_fail
    cmp al, 0x39
    ja .pn_fail

.pn_integer_digits:
    inc r15
    cmp r15, r14
    jae .pn_fraction
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x30
    jb .pn_fraction
    cmp al, 0x39
    jbe .pn_integer_digits
    jmp .pn_fraction

.pn_zero:
    inc r15

.pn_fraction:
    cmp r15, r14
    jae .pn_success
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x2e
    jne .pn_exponent

    inc r15
    cmp r15, r14
    jae .pn_fail
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x30
    jb .pn_fail
    cmp al, 0x39
    ja .pn_fail

.pn_fraction_digits:
    inc r15
    cmp r15, r14
    jae .pn_success
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x30
    jb .pn_exponent
    cmp al, 0x39
    jbe .pn_fraction_digits

.pn_exponent:
    cmp r15, r14
    jae .pn_success
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x65
    je .pn_have_exponent
    cmp al, 0x45
    jne .pn_success

.pn_have_exponent:
    inc r15
    cmp r15, r14
    jae .pn_fail
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x2b
    je .pn_exponent_sign
    cmp al, 0x2d
    jne .pn_first_exponent_digit

.pn_exponent_sign:
    inc r15
    cmp r15, r14
    jae .pn_fail

.pn_first_exponent_digit:
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x30
    jb .pn_fail
    cmp al, 0x39
    ja .pn_fail

.pn_exponent_digits:
    inc r15
    cmp r15, r14
    jae .pn_success
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x30
    jb .pn_success
    cmp al, 0x39
    jbe .pn_exponent_digits

.pn_success:
    mov eax, 1
    ret
.pn_fail:
    xor eax, eax
    ret

parse_array:
    inc r15
    call skip_ws
    cmp r15, r14
    jae .pa_fail
    cmp byte ptr [r13 + r15], 0x5d
    je .pa_empty

.pa_value:
    call parse_value
    test eax, eax
    jz .pa_fail

    call skip_ws
    cmp r15, r14
    jae .pa_fail
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x5d
    je .pa_close
    cmp al, 0x2c
    jne .pa_fail
    inc r15
    jmp .pa_value

.pa_empty:
.pa_close:
    inc r15
    mov eax, 1
    ret
.pa_fail:
    xor eax, eax
    ret

parse_object:
    inc r15
    call skip_ws
    cmp r15, r14
    jae .po_fail
    cmp byte ptr [r13 + r15], 0x7d
    je .po_empty

.po_member:
    call parse_string
    test eax, eax
    jz .po_fail

    call skip_ws
    cmp r15, r14
    jae .po_fail
    cmp byte ptr [r13 + r15], 0x3a
    jne .po_fail
    inc r15

    call parse_value
    test eax, eax
    jz .po_fail

    call skip_ws
    cmp r15, r14
    jae .po_fail
    movzx eax, byte ptr [r13 + r15]
    cmp al, 0x7d
    je .po_close
    cmp al, 0x2c
    jne .po_fail

    inc r15
    call skip_ws
    jmp .po_member

.po_empty:
.po_close:
    inc r15
    mov eax, 1
    ret
.po_fail:
    xor eax, eax
    ret

.section .rodata
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 32
