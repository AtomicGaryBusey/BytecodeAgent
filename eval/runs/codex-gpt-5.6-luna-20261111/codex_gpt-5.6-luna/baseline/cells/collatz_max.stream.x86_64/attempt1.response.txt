collatz_max:
    xor r8d, r8d
    mov ecx, 1
.Louter:
    cmp rcx, rdi
    jg .Ldone
    mov rdx, rcx
    xor esi, esi
.Linner:
    cmp rdx, 1
    je .Lchecked
    test rdx, 1
    jnz .Lodd
    shr rdx, 1
    jmp .Lstep
.Lodd:
    lea rdx, [rdx + rdx*2 + 1]
.Lstep:
    inc rsi
    jmp .Linner
.Lchecked:
    cmp rsi, r8
    jle .Lnext
    mov r8, rsi
.Lnext:
    inc rcx
    jmp .Louter
.Ldone:
    mov rax, r8
    ret
