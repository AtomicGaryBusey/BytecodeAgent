export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @entry %zero, @hit %inext_h, @miss %inext_m
	%p =l phi @entry %a, @hit %pnext_h, @miss %pnext_m
	%count =l phi @entry %zero, @hit %count_h, @miss %count_m
	%cond =w cultl %i, %n
	jnz %cond, @body, @done
@body
	%v =l loadl %p
	%gt =w cugtl %v, %t
	jnz %gt, @hit, @miss
@hit
	%inext_h =l add %i, 1
	%pnext_h =l add %p, 8
	%count_h =l add %count, 1
	jmp @loop
@miss
	%inext_m =l add %i, 1
	%pnext_m =l add %p, 8
	%count_m =l copy %count
	jmp @loop
@done
	ret %count
}
