export function l $binary_search(l %a, l %n, l %key) {
@entry
    jmp @loop

@loop
    %lo =l phi @entry 0, @increase %nextlo, @decrease %lo
    %hi =l phi @entry %n, @increase %hi, @decrease %mid
    %has =w csltl %lo, %hi
    jnz %has, @probe, @notfound

@probe
    %span =l sub %hi, %lo
    %half =l div %span, 2
    %mid =l add %lo, %half
    %offset =l mul %mid, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %equal =w ceql %value, %key
    jnz %equal, @found, @compare

@compare
    %less =w csltl %value, %key
    jnz %less, @increase, @decrease

@increase
    %nextlo =l add %mid, 1
    jmp @loop

@decrease
    jmp @loop

@found
    ret %mid

@notfound
    ret -1
}
