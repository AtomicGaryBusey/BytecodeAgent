export function l $sum_to(l %n) {
@start
	%nonpos =w cslel %n, 0
	jnz %nonpos, @zero, @iter
@zero
	ret 0
@iter
	%i =l phi @start 1, @body %nexti
	%sum =l phi @start 0, @body %nextsum
	%done =w csgtl %i, %n
	jnz %done, @ret, @body
@body
	%nextsum =l add %sum, %i
	%nexti =l add %i, 1
	jmp @iter
@ret
	ret %sum
}
