binary_search:
cbz x1, .Lnot_found
mov x3, #0
sub x4, x1, #1
.Lloop:
sub x5, x4, x3
lsr x5, x5, #1
add x5, x3, x5
ldr x6, [x0, x5, lsl #3]
cmp x6, x2
b.eq .Lfound
b.lt .Lless
sub x4, x5, #1
b .Lcheck
.Lless:
add x3, x5, #1
.Lcheck:
cmp x3, x4
b.le .Lloop
.Lnot_found:
mov x0, #-1
ret
.Lfound:
mov x0, x5
ret
