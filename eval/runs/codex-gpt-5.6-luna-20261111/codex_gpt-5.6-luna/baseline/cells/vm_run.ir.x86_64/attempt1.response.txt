export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	jmp @clear

@clear
	%ci = phi @entry 0, @clear_more %cin
	%cleft = csltl %ci, %cap
	jnz %cleft, @clear_more, @clear_done

@clear_more
	%coff = mul %ci, 8
	%caddr = add %stack, %coff
	storel 0, %caddr
	%cin = add %ci, 1
	jmp @clear

@clear_done
	jmp @exec

@exec
	%pc = phi @clear_done 0, @push %pushpc, @add %addpc, @sub %subpc, @mul %mulpc, @dup %duppc, @swap %swappc, @drop %droppc, @over %overpc, @lt %ltpc, @jz_taken %jztakenpc, @jz_not %jznotpc, @jmp %jmppc
	%d = phi @clear_done 0, @push %pushd, @add %addd, @sub %subd, @mul %muld, @dup %dupd, @swap %swapd, @drop %dropd, @over %overd, @lt %ltd, @jz_taken %jzd, @jz_not %jzd, @jmp %jmpd
	%running = csltl %pc, %len
	jnz %running, @fetch, @endrun

@fetch
	%pcoff = mul %pc, 8
	%pcaddr = add %code, %pcoff
	%op = loadl %pcaddr
	%e0 = ceql %op, 0
	jnz %e0, @op0, @check1

@check1
	%e1 = ceql %op, 1
	jnz %e1, @push, @check2

@check2
	%e2 = ceql %op, 2
	jnz %e2, @add, @check3

@check3
	%e3 = ceql %op, 3
	jnz %e3, @sub, @check4

@check4
	%e4 = ceql %op, 4
	jnz %e4, @mul, @check5

@check5
	%e5 = ceql %op, 5
	jnz %e5, @dup, @check6

@check6
	%e6 = ceql %op, 6
	jnz %e6, @swap, @check7

@check7
	%e7 = ceql %op, 7
	jnz %e7, @drop, @check8

@check8
	%e8 = ceql %op, 8
	jnz %e8, @over, @check9

@check9
	%e9 = ceql %op, 9
	jnz %e9, @lt, @check10

@check10
	%e10 = ceql %op, 10
	jnz %e10, @jz, @check11

@check11
	%e11 = ceql %op, 11
	jnz %e11, @jmp, @bad

@op0
	jmp @halt

@push
	%pushargaddr = add %pcaddr, 8
	%pushv = loadl %pushargaddr
	%pushoff = mul %d, 8
	%pushaddr = add %stack, %pushoff
	storel %pushv, %pushaddr
	%pushpc = add %pc, 2
	%pushd = add %d, 1
	jmp @exec

@add
	%addd1 = sub %d, 1
	%addd2 = sub %d, 2
	%addoff1 = mul %addd1, 8
	%addoff2 = mul %addd2, 8
	%addaddr1 = add %stack, %addoff1
	%addaddr2 = add %stack, %addoff2
	%addb = loadl %addaddr1
	%adda = loadl %addaddr2
	%addv = add %adda, %addb
	storel %addv, %addaddr2
	%addpc = add %pc, 1
	%addd = sub %d, 1
	jmp @exec

@sub
	%subd1 = sub %d, 1
	%subd2 = sub %d, 2
	%suboff1 = mul %subd1, 8
	%suboff2 = mul %subd2, 8
	%subaddr1 = add %stack, %suboff1
	%subaddr2 = add %stack, %suboff2
	%subb = loadl %subaddr1
	%suba = loadl %subaddr2
	%subv = sub %suba, %subb
	storel %subv, %subaddr2
	%subpc = add %pc, 1
	%subd = sub %d, 1
	jmp @exec

@mul
	%muld1 = sub %d, 1
	%muld2 = sub %d, 2
	%muloff1 = mul %muld1, 8
	%muloff2 = mul %muld2, 8
	%muladdr1 = add %stack, %muloff1
	%muladdr2 = add %stack, %muloff2
	%mulb = loadl %muladdr1
	%m ula = loadl %muladdr2
	%mulv = mul %mula, %mulb
	storel %mulv, %muladdr2
	%mulpc = add %pc, 1
	%muld = sub %d, 1
	jmp @exec

@dup
	%dupd1 = sub %d, 1
	%dupoff1 = mul %dupd1, 8
	%dupaddr1 = add %stack, %dupoff1
	%dupv = loadl %dupaddr1
	%dupoff2 = mul %d, 8
	%dupaddr2 = add %stack, %dupoff2
	storel %dupv, %dupaddr2
	%duppc = add %pc, 1
	%dupd = add %d, 1
	jmp @exec

@swap
	%swapd1 = sub %d, 1
	%swapd2 = sub %d, 2
	%swapoff1 = mul %swapd1, 8
	%swapoff2 = mul %swapd2, 8
	%swapaddr1 = add %stack, %swapoff1
	%swapaddr2 = add %stack, %swapoff2
	%swapb = loadl %swapaddr1
	%swapa = loadl %swapaddr2
	storel %swapb, %swapaddr2
	storel %swapa, %swapaddr1
	%swappc = add %pc, 1
	%swapd = %d
	jmp @exec

@drop
	%droppc = add %pc, 1
	%droppd = sub %d, 1
	jmp @exec

@over
	%overd2 = sub %d, 2
	%overoff2 = mul %overd2, 8
	%overaddr2 = add %stack, %overoff2
	%overv = loadl %overaddr2
	%overoff = mul %d, 8
	%overaddr = add %stack, %overoff
	storel %overv, %overaddr
	%overpc = add %pc, 1
	%overd = add %d, 1
	jmp @exec

@lt
	%ltd1 = sub %d, 1
	%ltd2 = sub %d, 2
	%ltoff1 = mul %ltd1, 8
	%ltoff2 = mul %ltd2, 8
	%ltaddr1 = add %stack, %ltoff1
	%ltaddr2 = add %stack, %ltoff2
	%ltb = loadl %ltaddr1
	%lta = loadl %ltaddr2
	%ltw = csltl %lta, %ltb
	%ltv = extsw %ltw
	storel %ltv, %ltaddr2
	%ltpc = add %pc, 1
	%ltd = sub %d, 1
	jmp @exec

@jz
	%jzargaddr = add %pcaddr, 8
	%jzt = loadl %jzargaddr
	%jzdtop = sub %d, 1
	%jzoff = mul %jzdtop, 8
	%jzaddr = add %stack, %jzoff
	%jzv = loadl %jzaddr
	%jzzero = ceql %jzv, 0
	jnz %jzzero, @jz_taken, @jz_not

@jz_taken
	%jztakenpc = %jzt
	%jzd = %jzdtop
	jmp @exec

@jz_not
	%jznotpc = add %pc, 2
	%jzd = %jzdtop
	jmp @exec

@jmp
	%jmpargaddr = add %pcaddr, 8
	%jmppc = loadl %jmpargaddr
	%jmpd = %d
	jmp @exec

@bad
	jmp @halt

@halt
	jmp @cleanup

@endrun
	jmp @cleanup

@cleanup
	%zi = phi @halt %d, @endrun %d
	%zrun = csltl %zi, %cap
	jnz %zrun, @zero_more, @result

@zero_more
	%zoff = mul %zi, 8
	%zaddr = add %stack, %zoff
	storel 0, %zaddr
	%zin = add %zi, 1
	jmp @cleanup

@result
	%has = csgtl %zi, 0
	jnz %has, @result_load, @result_zero

@result_load
	%ri = sub %zi, 1
	%roff = mul %ri, 8
	%raddr = add %stack, %roff
	%rv = loadl %raddr
	ret %rv

@result_zero
	ret 0
}
