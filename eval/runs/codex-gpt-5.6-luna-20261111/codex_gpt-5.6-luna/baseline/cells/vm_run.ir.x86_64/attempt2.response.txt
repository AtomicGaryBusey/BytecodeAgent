export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	%pm =l alloc8 8
	%dm =l alloc8 8
	%im =l alloc8 8
	storel 0, %pm
	storel 0, %dm
	storel 0, %im
	jmp @clear

@clear
	%ci =l loadl %im
	%cc =w csltl %ci, %cap
	jnz %cc, @clear_more, @exec

@clear_more
	%co =l mul %ci, 8
	%ca =l add %stack, %co
	storel 0, %ca
	%cn =l add %ci, 1
	storel %cn, %im
	jmp @clear

@exec
	%pc =l loadl %pm
	%d =l loadl %dm
	%run =w csltl %pc, %len
	jnz %run, @fetch, @cleanup

@fetch
	%po =l mul %pc, 8
	%pa =l add %code, %po
	%op =l loadl %pa
	%e0 =w ceql %op, 0
	jnz %e0, @cleanup, @check1

@check1
	%e1 =w ceql %op, 1
	jnz %e1, @push, @check2

@check2
	%e2 =w ceql %op, 2
	jnz %e2, @add, @check3

@check3
	%e3 =w ceql %op, 3
	jnz %e3, @sub, @check4

@check4
	%e4 =w ceql %op, 4
	jnz %e4, @mul, @check5

@check5
	%e5 =w ceql %op, 5
	jnz %e5, @dup, @check6

@check6
	%e6 =w ceql %op, 6
	jnz %e6, @swap, @check7

@check7
	%e7 =w ceql %op, 7
	jnz %e7, @drop, @check8

@check8
	%e8 =w ceql %op, 8
	jnz %e8, @over, @check9

@check9
	%e9 =w ceql %op, 9
	jnz %e9, @lt, @check10

@check10
	%e10 =w ceql %op, 10
	jnz %e10, @jz, @jmp

@push
	%pva =l add %pa, 8
	%pv =l loadl %pva
	%po2 =l mul %d, 8
	%paddr =l add %stack, %po2
	storel %pv, %paddr
	%ppc =l add %pc, 2
	%pd =l add %d, 1
	storel %ppc, %pm
	storel %pd, %dm
	jmp @exec

@add
	%ad1 =l sub %d, 1
	%ad2 =l sub %d, 2
	%ao1 =l mul %ad1, 8
	%ao2 =l mul %ad2, 8
	%aa1 =l add %stack, %ao1
	%aa2 =l add %stack, %ao2
	%ab =l loadl %aa1
	%aa =l loadl %aa2
	%av =l add %aa, %ab
	storel %av, %aa2
	%apc =l add %pc, 1
	%and =l sub %d, 1
	storel %apc, %pm
	storel %and, %dm
	jmp @exec

@sub
	%sd1 =l sub %d, 1
	%sd2 =l sub %d, 2
	%so1 =l mul %sd1, 8
	%so2 =l mul %sd2, 8
	%sa1 =l add %stack, %so1
	%sa2 =l add %stack, %so2
	%sb =l loadl %sa1
	%sa =l loadl %sa2
	%sv =l sub %sa, %sb
	storel %sv, %sa2
	%spc =l add %pc, 1
	%snd =l sub %d, 1
	storel %spc, %pm
	storel %snd, %dm
	jmp @exec

@mul
	%md1 =l sub %d, 1
	%md2 =l sub %d, 2
	%mo1 =l mul %md1, 8
	%mo2 =l mul %md2, 8
	%ma1 =l add %stack, %mo1
	%ma2 =l add %stack, %mo2
	%mb =l loadl %ma1
	%ma =l loadl %ma2
	%mv =l mul %ma, %mb
	storel %mv, %ma2
	%mpc =l add %pc, 1
	%mnd =l sub %d, 1
	storel %mpc, %pm
	storel %mnd, %dm
	jmp @exec

@dup
	%dd1 =l sub %d, 1
	%do1 =l mul %dd1, 8
	%da1 =l add %stack, %do1
	%dv =l loadl %da1
	%do2 =l mul %d, 8
	%da2 =l add %stack, %do2
	storel %dv, %da2
	%dpc =l add %pc, 1
	%dnd =l add %d, 1
	storel %dpc, %pm
	storel %dnd, %dm
	jmp @exec

@swap
	%wd1 =l sub %d, 1
	%wd2 =l sub %d, 2
	%wo1 =l mul %wd1, 8
	%wo2 =l mul %wd2, 8
	%wa1 =l add %stack, %wo1
	%wa2 =l add %stack, %wo2
	%wb =l loadl %wa1
	%wa =l loadl %wa2
	storel %wb, %wa2
	storel %wa, %wa1
	%wpc =l add %pc, 1
	storel %wpc, %pm
	jmp @exec

@drop
	%rpc =l add %pc, 1
	%rnd =l sub %d, 1
	storel %rpc, %pm
	storel %rnd, %dm
	jmp @exec

@over
	%od2 =l sub %d, 2
	%oo2 =l mul %od2, 8
	%oa2 =l add %stack, %oo2
	%ov =l loadl %oa2
	%oo =l mul %d, 8
	%oa =l add %stack, %oo
	storel %ov, %oa
	%opc =l add %pc, 1
	%ond =l add %d, 1
	storel %opc, %pm
	storel %ond, %dm
	jmp @exec

@lt
	%ld1 =l sub %d, 1
	%ld2 =l sub %d, 2
	%lo1 =l mul %ld1, 8
	%lo2 =l mul %ld2, 8
	%la1 =l add %stack, %lo1
	%la2 =l add %stack, %lo2
	%lb =l loadl %la1
	%la =l loadl %la2
	%lw =w csltl %la, %lb
	%lv =l extsw %lw
	storel %lv, %la2
	%lpc =l add %pc, 1
	%lnd =l sub %d, 1
	storel %lpc, %pm
	storel %lnd, %dm
	jmp @exec

@jz
	%jarg =l add %pa, 8
	%jt =l loadl %jarg
	%jd =l sub %d, 1
	%jo =l mul %jd, 8
	%ja =l add %stack, %jo
	%jv =l loadl %ja
	%jzv =w ceql %jv, 0
	jnz %jzv, @jzt, @jzn

@jzt
	storel %jt, %pm
	storel %jd, %dm
	jmp @exec

@jzn
	%jnpc =l add %pc, 2
	storel %jnpc, %pm
	storel %jd, %dm
	jmp @exec

@jmp
	%jparg =l add %pa, 8
	%jpt =l loadl %jparg
	storel %jpt, %pm
	jmp @exec

@cleanup
	%zi =l loadl %im
	%zd =l loadl %dm
	storel %zd, %im
	%zc =w csltl %zi, %cap
	jnz %zc, @zero, @result

@zero
	%zo =l mul %zi, 8
	%za =l add %stack, %zo
	storel 0, %za
	%zn =l add %zi, 1
	storel %zn, %im
	jmp @cleanup

@result
	%rd =l loadl %dm
	%rh =w csgtl %rd, 0
	jnz %rh, @result_load, @result_zero

@result_load
	%ri =l sub %rd, 1
	%ro =l mul %ri, 8
	%ra =l add %stack, %ro
	%rv =l loadl %ra
	ret %rv

@result_zero
	ret 0
}
