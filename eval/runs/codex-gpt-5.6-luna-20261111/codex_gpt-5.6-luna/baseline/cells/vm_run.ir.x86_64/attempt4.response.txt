export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	%pm =l alloc8 8
	%dm =l alloc8 8
	%ci =l alloc8 8
	storel 0, %pm
	storel 0, %dm
	jmp @exec

@exec
	%pc =l loadl %pm
	%d =l loadl %dm
	%ok =w csltl %pc, %len
	jnz %ok, @fetch, @finish

@fetch
	%off =l mul %pc, 8
	%addr =l add %code, %off
	%op =l loadl %addr
	%e0 =w ceql %op, 0
	jnz %e0, @finish, @c1

@c1
	%e1 =w ceql %op, 1
	jnz %e1, @push, @c2

@c2
	%e2 =w ceql %op, 2
	jnz %e2, @add, @c3

@c3
	%e3 =w ceql %op, 3
	jnz %e3, @sub, @c4

@c4
	%e4 =w ceql %op, 4
	jnz %e4, @mul, @c5

@c5
	%e5 =w ceql %op, 5
	jnz %e5, @dup, @c6

@c6
	%e6 =w ceql %op, 6
	jnz %e6, @swap, @c7

@c7
	%e7 =w ceql %op, 7
	jnz %e7, @drop, @c8

@c8
	%e8 =w ceql %op, 8
	jnz %e8, @over, @c9

@c9
	%e9 =w ceql %op, 9
	jnz %e9, @lt, @c10

@c10
	%e10 =w ceql %op, 10
	jnz %e10, @jz, @jmp

@push
	%va =l add %addr, 8
	%v =l loadl %va
	%do =l mul %d, 8
	%da =l add %stack, %do
	storel %v, %da
	%np =l add %pc, 2
	%nd =l add %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@add
	%i1 =l sub %d, 1
	%i2 =l sub %d, 2
	%o1 =l mul %i1, 8
	%o2 =l mul %i2, 8
	%a1 =l add %stack, %o1
	%a2 =l add %stack, %o2
	%b =l loadl %a1
	%a =l loadl %a2
	%v =l add %a, %b
	storel %v, %a2
	%np =l add %pc, 1
	%nd =l sub %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@sub
	%i1 =l sub %d, 1
	%i2 =l sub %d, 2
	%o1 =l mul %i1, 8
	%o2 =l mul %i2, 8
	%a1 =l add %stack, %o1
	%a2 =l add %stack, %o2
	%b =l loadl %a1
	%a =l loadl %a2
	%v =l sub %a, %b
	storel %v, %a2
	%np =l add %pc, 1
	%nd =l sub %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@mul
	%i1 =l sub %d, 1
	%i2 =l sub %d, 2
	%o1 =l mul %i1, 8
	%o2 =l mul %i2, 8
	%a1 =l add %stack, %o1
	%a2 =l add %stack, %o2
	%b =l loadl %a1
	%a =l loadl %a2
	%v =l mul %a, %b
	storel %v, %a2
	%np =l add %pc, 1
	%nd =l sub %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@dup
	%i =l sub %d, 1
	%o =l mul %i, 8
	%sa =l add %stack, %o
	%v =l loadl %sa
	%do =l mul %d, 8
	%da =l add %stack, %do
	storel %v, %da
	%np =l add %pc, 1
	%nd =l add %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@swap
	%i1 =l sub %d, 1
	%i2 =l sub %d, 2
	%o1 =l mul %i1, 8
	%o2 =l mul %i2, 8
	%a1 =l add %stack, %o1
	%a2 =l add %stack, %o2
	%v1 =l loadl %a1
	%v2 =l loadl %a2
	storel %v2, %a1
	storel %v1, %a2
	%np =l add %pc, 1
	storel %np, %pm
	jmp @exec

@drop
	%np =l add %pc, 1
	%nd =l sub %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@over
	%i =l sub %d, 2
	%o =l mul %i, 8
	%sa =l add %stack, %o
	%v =l loadl %sa
	%do =l mul %d, 8
	%da =l add %stack, %do
	storel %v, %da
	%np =l add %pc, 1
	%nd =l add %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@lt
	%i1 =l sub %d, 1
	%i2 =l sub %d, 2
	%o1 =l mul %i1, 8
	%o2 =l mul %i2, 8
	%a1 =l add %stack, %o1
	%a2 =l add %stack, %o2
	%b =l loadl %a1
	%a =l loadl %a2
	%c =w csltl %a, %b
	%v =l extsw %c
	storel %v, %a2
	%np =l add %pc, 1
	%nd =l sub %d, 1
	storel %np, %pm
	storel %nd, %dm
	jmp @exec

@jz
	%ta =l add %addr, 8
	%t =l loadl %ta
	%i =l sub %d, 1
	%o =l mul %i, 8
	%a =l add %stack, %o
	%v =l loadl %a
	%z =w ceql %v, 0
	jnz %z, @jzt, @jzn

@jzt
	storel %t, %pm
	storel %i, %dm
	jmp @exec

@jzn
	%np =l add %pc, 2
	storel %np, %pm
	storel %i, %dm
	jmp @exec

@jmp
	%ta =l add %addr, 8
	%t =l loadl %ta
	storel %t, %pm
	jmp @exec

@finish
	%fd =l loadl %dm
	storel %fd, %ci
	jmp @clear_check

@clear_check
	%i =l loadl %ci
	%more =w csltl %i, %cap
	jnz %more, @clear_one, @result

@clear_one
	%o =l mul %i, 8
	%a =l add %stack, %o
	storel 0, %a
	%ni =l add %i, 1
	storel %ni, %ci
	jmp @clear_check

@result
	%d =l loadl %dm
	%nonempty =w csltl 0, %d
	jnz %nonempty, @result_load, @result_zero

@result_load
	%i =l sub %d, 1
	%o =l mul %i, 8
	%a =l add %stack, %o
	%v =l loadl %a
	ret %v

@result_zero
	ret 0
}
