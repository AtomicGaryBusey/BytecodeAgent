export function l $bmin_empty(l %a, l %n) {
@entry
	%max =l copy 9223372036854775807
	%zero =w ceql %n, 0
	jnz %zero, @empty, @loop

@loop
	%i =l phi @entry 0, @cont %inext
	%min =l phi @entry %max, @cont %chosen
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%less =w csltl %v, %min
	jnz %less, @take, @keep

@take
	jmp @cont

@keep
	jmp @cont

@cont
	%chosen =l phi @take %v, @keep %min
	%inext =l add %i, 1
	%done =w ceql %inext, %n
	jnz %done, @result, @loop

@result
	ret %chosen

@empty
	ret %max
}
