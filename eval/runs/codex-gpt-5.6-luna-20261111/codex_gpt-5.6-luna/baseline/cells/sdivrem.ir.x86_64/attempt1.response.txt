export function l $sdivrem(l %a, l %b, l %rem) {
@entry
	%bz =w ceql %b, 0
	jnz %bz, @zero, @check

@zero
	storel %a, %rem
	ret 0

@check
	%amin =w ceql %a, -9223372036854775808
	jnz %amin, @min, @normal

@min
	%bneg =w ceql %b, -1
	jnz %bneg, @overflow, @normal

@overflow
	storel 0, %rem
	ret -9223372036854775808

@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
