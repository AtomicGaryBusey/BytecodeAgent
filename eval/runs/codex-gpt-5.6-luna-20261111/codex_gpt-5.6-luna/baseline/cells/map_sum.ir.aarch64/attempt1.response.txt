export function l $map_sum(l %a, l %n, l %f) {
@entry
	%zero =l copy 0
	jmp @loop
@loop
	%sum =l phi @entry %zero, @body %nextsum
	%p =l phi @entry %a, @body %nextp
	%i =l phi @entry 0, @body %nexti
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%x =l loadl %p
	%r =l call %f(l %x)
	%nextsum =l add %sum, %r
	%nextp =l add %p, 8
	%nexti =l add %i, 1
	jmp @loop
@done
	ret %sum
}
