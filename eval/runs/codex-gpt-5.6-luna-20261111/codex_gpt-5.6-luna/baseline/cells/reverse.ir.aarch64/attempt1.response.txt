export function l $reverse(l %a, l %n) {
@entry
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @entry 0, @body %inext
	%j =l phi @entry %j0, @body %jnext
	%cond =w csltl %i, %j
	jnz %cond, @body, @done
@body
	%io =l mul %i, 8
	%jo =l mul %j, 8
	%lp =l add %a, %io
	%rp =l add %a, %jo
	%lv =l loadl %lp
	%rv =l loadl %rp
	storel %rv, %lp
	storel %lv, %rp
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
