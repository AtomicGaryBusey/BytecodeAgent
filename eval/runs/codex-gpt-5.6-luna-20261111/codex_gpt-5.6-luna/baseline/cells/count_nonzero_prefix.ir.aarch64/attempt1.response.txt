export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @entry %zero, @body %next
	%in =w csltl %i, %n
	jnz %in, @check, @done
@check
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%iszero =w ceql %v, 0
	jnz %iszero, @donezero, @body
@body
	%next =l add %i, 1
	jmp @loop
@donezero
	ret %i
@done
	ret %i
}
