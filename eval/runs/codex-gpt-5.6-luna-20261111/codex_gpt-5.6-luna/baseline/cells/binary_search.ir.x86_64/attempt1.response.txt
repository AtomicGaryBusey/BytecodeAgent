export function l $binary_search(l %a, l %n, l %key) {
@entry
	%zero =l copy 0
	jmp @loop

@loop
	%lo =l phi @entry %zero, @advance %nextlo, @shrink %lo
	%hi =l phi @entry %n, @advance %hi, @shrink %mid
	%active =w csltl %lo, %hi
	jnz %active, @search, @notfound

@search
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%ptr =l add %a, %offset
	%val =l loadl %ptr
	%equal =w ceql %val, %key
	jnz %equal, @found, @compare

@compare
	%less =w csltl %val, %key
	jnz %less, @advance, @shrink

@advance
	%nextlo =l add %mid, 1
	jmp @loop

@shrink
	jmp @loop

@found
	ret %mid

@notfound
	ret -1
}
