export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w cslel %n, 1
	jnz %small, @done, @setup

@setup
	%last =l sub %n, 1
	%loff =l mul %last, 16
	%pp =l add %a, %loff
	%pk =l loadl %pp
	%pvoff =l add %loff, 8
	%pvp =l add %a, %pvoff
	%pv =l loadl %pvp
	jmp @loop

@loop
	%i =l phi @setup 0, @noswap %i, @doswap %inext
	%j =l phi @setup 0, @noswap %jn, @doswap %jn
	%more =w csltl %j, %last
	jnz %more, @scan, @partition

@scan
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jvoff =l add %joff, 8
	%jvp =l add %a, %jvoff
	%jv =l loadl %jvp
	%klt =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vle =w cslel %jv, %pv
	%eqtake =w and %keq, %vle
	%take =w or %klt, %eqtake
	jnz %take, @doswap, @noswap

@noswap
	%jn =l add %j, 1
	jmp @loop

@doswap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivoff =l add %ioff, 8
	%ivp =l add %a, %ivoff
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%inext =l add %i, 1
	%jn =l add %j, 1
	jmp @loop

@partition
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivoff =l add %ioff, 8
	%ivp =l add %a, %ivoff
	%iv =l loadl %ivp
	storel %pk, %ip
	storel %pv, %ivp
	storel %ik, %pp
	storel %iv, %pvp
	%rightn =l sub %n, %i
	%rightn2 =l sub %rightn, 1
	%one =l add %i, 1
	%roff =l mul %one, 16
	%rp =l add %a, %roff
	%left =l call $qsort_pairs(l %a, l %i)
	%right =l call $qsort_pairs(l %rp, l %rightn2)
	ret 0

@done
	ret 0
}
