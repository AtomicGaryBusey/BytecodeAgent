export function l $fib(l %n) {
@start
	%base =w csltl %n, 2
	jnz %base, @base, @loop
@base
	ret %n
@loop
	%a =l phi @start 0, @body %b
	%b =l phi @start 1, @body %sum
	%i =l phi @start 2, @body %next
	%done =w cslel %i, %n
	jnz %done, @body, @return
@body
	%sum =l add %a, %b
	%next =l add %i, 1
	jmp @loop
@return
	ret %b
}
