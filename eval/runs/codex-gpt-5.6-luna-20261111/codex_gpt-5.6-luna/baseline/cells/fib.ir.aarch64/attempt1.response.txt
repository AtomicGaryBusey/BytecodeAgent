export function l $fib(l %n) {
@start
	%base = csltl %n, 2
	jnz %base, @base, @loop
@base
	ret %n
@loop
	%a = phi @start 0, @body %b
	%b = phi @start 1, @body %sum
	%i = phi @start 2, @body %next
	%done = cslel %i, %n
	jnz %done, @body, @return
@body
	%sum = add %a, %b
	%next = add %i, 1
	jmp @loop
@return
	ret %b
}
