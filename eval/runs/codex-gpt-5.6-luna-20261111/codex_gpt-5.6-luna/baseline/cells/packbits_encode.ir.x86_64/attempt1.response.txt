export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    jump @loop

@loop
    %i =l phi @start 0, @repdone %inext, @litcopydone %jend
    %o =l phi @start 0, @repdone %onext, @litcopydone %olit
    %active =w csltl %i, %n
    jnz %active, @process, @finish

@process
    %sp =l add %src, %i
    %v =w loadub %sp
    jump @rcheck

@rcheck
    %k =l phi @process %i, @rinc %knext
    %has =w csltl %k, %n
    jnz %has, @rbound, @rstop

@rbound
    %rd =l sub %k, %i
    %under =w csltl %rd, 128
    jnz %under, @rbyte, @rstop

@rbyte
    %kp =l add %src, %k
    %x =w loadub %kp
    %same =w ceql %x, %v
    jnz %same, @rinc, @rstop

@rinc
    %knext =l add %k, 1
    jump @rcheck

@rstop
    %rend =l phi @rcheck %k, @rbound %k, @rbyte %k
    %r =l sub %rend, %i
    %isrun =w cslel 3, %r
    jnz %isrun, @repcheck, @lit

@repcheck
    %rem =l sub %cap, %o
    %fits =w cslel 2, %rem
    jnz %fits, @repwrite, @fail

@repwrite
    %head =l sub 257, %r
    %headw =w copy %head
    %op =l add %out, %o
    storeb %headw, %op
    %bp =l add %op, 1
    storeb %v, %bp
    %inext =l add %i, %r
    %onext =l add %o, 2
    jump @repdone

@repdone
    jump @loop

@lit
    %j =l phi @rstop %i, @litinc %jnext
    %jn =w csltl %j, %n
    jnz %jn, @litlen, @litdone

@litlen
    %llen =l sub %j, %i
    %lroom =w csltl %llen, 128
    jnz %lroom, @litv, @litdone

@litv
    %jp1 =l add %j, 1
    %has1 =w csltl %jp1, %n
    jnz %has1, @litv1, @litinc

@litv1
    %p0 =l add %src, %j
    %p1 =l add %src, %jp1
    %b0 =w loadub %p0
    %b1 =w loadub %p1
    %eq1 =w ceql %b0, %b1
    jnz %eq1, @litv2, @litinc

@litv2
    %jp2 =l add %j, 2
    %has2 =w csltl %jp2, %n
    jnz %has2, @litv2load, @litinc

@litv2load
    %p2 =l add %src, %jp2
    %b2 =w loadub %p2
    %eq2 =w ceql %b0, %b2
    jnz %eq2, @litdone, @litinc

@litinc
    %jnext =l add %j, 1
    jump @lit

@litdone
    %jend =l phi @lit %j, @litlen %j, @litv %j, @litv2 %j, @litv2load %j
    %llen2 =l sub %jend, %i
    %lsize =l add %llen2, 1
    %lrem =l sub %cap, %o
    %lfits =w cslel %lsize, %lrem
    jnz %lfits, @litwrite, @fail

@litwrite
    %lh =l sub %llen2, 1
    %lhw =w copy %lh
    %lop =l add %out, %o
    storeb %lhw, %lop
    %olit =l add %o, 1
    %qstart =l copy %i
    jump @litcopy

@litcopy
    %q =l phi @litwrite %qstart, @litcopyloop %qnext
    %co =l phi @litwrite %olit, @litcopyloop %conext
    %more =w csltl %q, %jend
    jnz %more, @litcopyloop, @litcopydone

@litcopyloop
    %qp =l add %src, %q
    %cb =w loadub %qp
    %cop =l add %out, %co
    storeb %cb, %cop
    %qnext =l add %q, 1
    %conext =l add %co, 1
    jump @litcopy

@litcopydone
    jump @loop

@finish
    ret %o

@fail
    ret -1
}
