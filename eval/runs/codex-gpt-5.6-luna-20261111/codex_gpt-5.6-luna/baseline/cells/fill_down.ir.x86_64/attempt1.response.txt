export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %inext
    %v =l phi @entry %start, @body %vnext
    %p =l phi @entry %a, @body %pnext
    %ok =w csltl %i, %n
    jnz %ok, @body, @done
@body
    storel %v, %p
    %inext =l add %i, 1
    %vnext =l sub %v, 1
    %pnext =l add %p, 8
    jmp @loop
@done
    ret %n
}
