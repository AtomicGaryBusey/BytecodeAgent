export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%aneg =w csltl %a, 0
	jnz %aneg, @aneg, @apos

@apos
	%bneg =w csltl %b, 0
	jnz %bneg, @done, @aposbpos

@aposbpos
	%sumneg =w csltl %sum, 0
	jnz %sumneg, @overflow, @done

@aneg
	%bneg2 =w csltl %b, 0
	jnz %bneg2, @anegbneg, @done

@anegbneg
	%sumneg2 =w csltl %sum, 0
	jnz %sumneg2, @done, @overflow

@overflow
	ret 1

@done
	ret 0
}
