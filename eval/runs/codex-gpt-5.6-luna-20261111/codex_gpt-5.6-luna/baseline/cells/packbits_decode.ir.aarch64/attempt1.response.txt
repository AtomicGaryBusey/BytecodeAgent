export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop

@loop
	%i =l phi @start 0, @lit_done %lin, @rep_done %rin, @noop %nin
	%o =l phi @start 0, @lit_done %lout, @rep_done %rout, @noop %on
	%more =w csltl %i, %n
	jnz %more, @have_header, @err

@have_header
	%hp =l add %src, %i
	%h =w loadsb %hp
	%ni =l add %i, 1
	%isnoop =w ceql %h, -128
	jnz %isnoop, @noop, @classify

@classify
	%isrep =w csltw %h, 0
	jnz %isrep, @repeat, @literal

@literal
	%cnt =l extuw %h
	%cnt1 =l add %cnt, 1
	%remain =l sub %n, %ni
	%bad =w csltl %remain, %cnt1
	jnz %bad, @err, @lit_check_output

@lit_check_output
	%outremain =l sub %cap, %o
	%trunc =w csltl %outremain, %cnt1
	jnz %trunc, @lit_trunc_start, @lit_start

@lit_start
	jmp @lit_copy

@lit_copy
	%k =l phi @lit_start 0, @lit_body %k2
	%go =w csltl %k, %cnt1
	jnz %go, @lit_body, @lit_done

@lit_body
	%si =l add %ni, %k
	%sp =l add %src, %si
	%b =w loadub %sp
	%oi =l add %o, %k
	%op =l add %out, %oi
	storeb %b, %op
	%k2 =l add %k, 1
	jmp @lit_copy

@lit_done
	%lin =l add %ni, %cnt1
	%lout =l add %o, %cnt1
	jmp @loop

@lit_trunc_start
	jmp @lit_trunc

@lit_trunc
	%tk =l phi @lit_trunc_start 0, @lit_trunc_body %tk2
	%tgo =w csltl %tk, %outremain
	jnz %tgo, @lit_trunc_body, @overflow

@lit_trunc_body
	%tsi =l add %ni, %tk
	%tsp =l add %src, %tsi
	%tb =w loadub %tsp
	%toi =l add %o, %tk
	%top =l add %out, %toi
	storeb %tb, %top
	%tk2 =l add %tk, 1
	jmp @lit_trunc

@repeat
	%cw =w sub 1, %h
	%cnt =l extuw %cw
	%has =w csltl %ni, %n
	jnz %has, @repeat_value, @err

@repeat_value
	%rp =l add %src, %ni
	%v =w loadub %rp
	%outremain =l sub %cap, %o
	%trunc =w csltl %outremain, %cnt
	jnz %trunc, @rep_trunc_start, @rep_start

@rep_start
	jmp @rep_copy

@rep_copy
	%k =l phi @rep_start 0, @rep_body %k2
	%go =w csltl %k, %cnt
	jnz %go, @rep_body, @rep_done

@rep_body
	%oi =l add %o, %k
	%op =l add %out, %oi
	storeb %v, %op
	%k2 =l add %k, 1
	jmp @rep_copy

@rep_done
	%rin =l add %ni, 1
	%rout =l add %o, %cnt
	jmp @loop

@rep_trunc_start
	jmp @rep_trunc

@rep_trunc
	%tk =l phi @rep_trunc_start 0, @rep_trunc_body %tk2
	%tgo =w csltl %tk, %outremain
	jnz %tgo, @rep_trunc_body, @overflow

@rep_trunc_body
	%toi =l add %o, %tk
	%top =l add %out, %toi
	storeb %v, %top
	%tk2 =l add %tk, 1
	jmp @rep_trunc

@noop
	%nin =l add %i, 1
	%on =l copy %o
	jmp @loop

@err
	ret -2

@overflow
	ret -1
}
