export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop

@loop
	%i =l phi @start 0, @lit_done %lit_next_i, @rep_done %rep_next_i, @noop %noop_next_i
	%o =l phi @start 0, @lit_done %lit_next_o, @rep_done %rep_next_o, @noop %noop_next_o
	%more =w csltl %i, %n
	jnz %more, @have_header, @err

@have_header
	%hp =l add %src, %i
	%h =w loadsb %hp
	%ni =l add %i, 1
	%isnoop =w ceqw %h, -128
	jnz %isnoop, @noop, @classify

@classify
	%isrep =w csltw %h, 0
	jnz %isrep, @repeat, @literal

@literal
	%lit_cnt_w =w extuw %h
	%lit_cnt =l add %lit_cnt_w, 1
	%lit_remain =l sub %n, %ni
	%lit_bad =w csltl %lit_remain, %lit_cnt
	jnz %lit_bad, @err, @lit_check_output

@lit_check_output
	%lit_outremain =l sub %cap, %o
	%lit_trunc =w csltl %lit_outremain, %lit_cnt
	jnz %lit_trunc, @lit_trunc_start, @lit_start

@lit_start
	jmp @lit_copy

@lit_copy
	%lit_k =l phi @lit_start 0, @lit_body %lit_k2
	%lit_go =w csltl %lit_k, %lit_cnt
	jnz %lit_go, @lit_body, @lit_done

@lit_body
	%lit_si =l add %ni, %lit_k
	%lit_sp =l add %src, %lit_si
	%lit_b =w loadub %lit_sp
	%lit_oi =l add %o, %lit_k
	%lit_op =l add %out, %lit_oi
	storeb %lit_b, %lit_op
	%lit_k2 =l add %lit_k, 1
	jmp @lit_copy

@lit_done
	%lit_next_i =l add %ni, %lit_cnt
	%lit_next_o =l add %o, %lit_cnt
	jmp @loop

@lit_trunc_start
	jmp @lit_trunc

@lit_trunc
	%lit_tk =l phi @lit_trunc_start 0, @lit_trunc_body %lit_tk2
	%lit_tgo =w csltl %lit_tk, %lit_outremain
	jnz %lit_tgo, @lit_trunc_body, @overflow

@lit_trunc_body
	%lit_tsi =l add %ni, %lit_tk
	%lit_tsp =l add %src, %lit_tsi
	%lit_tb =w loadub %lit_tsp
	%lit_toi =l add %o, %lit_tk
	%lit_top =l add %out, %lit_toi
	storeb %lit_tb, %lit_top
	%lit_tk2 =l add %lit_tk, 1
	jmp @lit_trunc

@repeat
	%rep_cnt_w =w sub 1, %h
	%rep_cnt =l extuw %rep_cnt_w
	%rep_has =w csltl %ni, %n
	jnz %rep_has, @repeat_value, @err

@repeat_value
	%rep_p =l add %src, %ni
	%rep_v =w loadub %rep_p
	%rep_outremain =l sub %cap, %o
	%rep_trunc =w csltl %rep_outremain, %rep_cnt
	jnz %rep_trunc, @rep_trunc_start, @rep_start

@rep_start
	jmp @rep_copy

@rep_copy
	%rep_k =l phi @rep_start 0, @rep_body %rep_k2
	%rep_go =w csltl %rep_k, %rep_cnt
	jnz %rep_go, @rep_body, @rep_done

@rep_body
	%rep_oi =l add %o, %rep_k
	%rep_op =l add %out, %rep_oi
	storeb %rep_v, %rep_op
	%rep_k2 =l add %rep_k, 1
	jmp @rep_copy

@rep_done
	%rep_next_i =l add %ni, 1
	%rep_next_o =l add %o, %rep_cnt
	jmp @loop

@rep_trunc_start
	jmp @rep_trunc

@rep_trunc
	%rep_tk =l phi @rep_trunc_start 0, @rep_trunc_body %rep_tk2
	%rep_tgo =w csltl %rep_tk, %rep_outremain
	jnz %rep_tgo, @rep_trunc_body, @overflow

@rep_trunc_body
	%rep_toi =l add %o, %rep_tk
	%rep_top =l add %out, %rep_toi
	storeb %rep_v, %rep_top
	%rep_tk2 =l add %rep_tk, 1
	jmp @rep_trunc

@noop
	%noop_next_i =l add %i, 1
	%noop_next_o =l copy %o
	jmp @loop

@err
	ret -2

@overflow
	ret -1
}
