export function l $collatz_max(l %n) {
@entry
	%np =l alloc8 8
	%kp =l alloc8 8
	%mp =l alloc8 8
	%xp =l alloc8 8
	%sp =l alloc8 8
	storel %n, %np
	storel 1, %kp
	storel 0, %mp
@outer
	%k =l loadl %kp
	%limit =l loadl %np
	%within =w cslel %k, %limit
	jnz %within, @start, @done
@start
	storel %k, %xp
	storel 0, %sp
@collatz
	%x =l loadl %xp
	%isone =w ceql %x, 1
	jnz %isone, @finished, @advance
@advance
	%parity =l and %x, 1
	%odd =w cnel %parity, 0
	jnz %odd, @odd_step, @even_step
@even_step
	%next_even =l div %x, 2
	storel %next_even, %xp
	jmp @count
@odd_step
	%triple =l mul %x, 3
	%next_odd =l add %triple, 1
	storel %next_odd, %xp
@count
	%steps =l loadl %sp
	%next_steps =l add %steps, 1
	storel %next_steps, %sp
	jmp @collatz
@finished
	%count =l loadl %sp
	%maximum =l loadl %mp
	%greater =w csltl %maximum, %count
	jnz %greater, @setmax, @next
@setmax
	storel %count, %mp
@next
	%next_k =l add %k, 1
	storel %next_k, %kp
	jmp @outer
@done
	%result =l loadl %mp
	ret %result
}
