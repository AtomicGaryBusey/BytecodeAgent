bubble_sort:
cbz x1, .Ldone
mov x2, x0
sub x3, x1, #1
.Louter:
cbz x3, .Ldone
mov x4, #0
.Linner:
cmp x4, x3
b.ge .Lnext
ldr x5, [x2, x4, lsl #3]
add x6, x4, #1
ldr x7, [x2, x6, lsl #3]
cmp x5, x7
b.le .Lnoswap
str x7, [x2, x4, lsl #3]
str x5, [x2, x6, lsl #3]
.Lnoswap:
add x4, x4, #1
b .Linner
.Lnext:
sub x3, x3, #1
b .Louter
.Ldone:
mov x0, #0
ret
