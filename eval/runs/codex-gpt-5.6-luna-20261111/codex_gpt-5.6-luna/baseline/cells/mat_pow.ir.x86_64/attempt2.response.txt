export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
	%nn =l mul %n, %n
	%bytes =l mul %nn, 8
	%prod =l add %tmp, %bytes
	%two =l add %nn, %nn
	%zero =l copy 0
	%one =l copy 1
	jmp @init

@init
	%ix =l phi @entry 0, @init_next %ix1
	%idone =w cslel %nn, %ix
	jnz %idone, @diag, @init_body

@init_body
	%off =l mul %ix, 8
	%op =l add %out, %off
	%bp =l add %tmp, %off
	%ap =l add %a, %off
	%av =l loadl %ap
	storel %zero, %op
	storel %av, %bp
	jmp @init_next

@init_next
	%ix1 =l add %ix, 1
	jmp @init

@diag
	%di =l phi @init 0, @diag_next %di1
	%ddone =w cslel %nn, %di
	jnz %ddone, @pow_loop, @diag_body

@diag_body
	%doff =l mul %di, 8
	%dptr =l add %out, %doff
	storel %one, %dptr
	%di1 =l add %di, %n
	%di1 =l add %di1, 1
	jmp @diag_next

@diag_next
	jmp @diag

@pow_loop
	%kk =l phi @diag %k, @copy_done %kk1
	%has =w csltl 0, %kk
	jnz %has, @mul_i, @clear

@mul_i
	%mi =l phi @pow_loop 0, @mul_inext %mi1
	%idone2 =w cslel %n, %mi
	jnz %idone2, @copy, @mul_i_body

@mul_i_body
	%mrow =l mul %mi, %n
	jmp @mul_j

@mul_j
	%mj =l phi @mul_i_body 0, @mul_tdone %mj1
	%jdone =w cslel %n, %mj
	jnz %jdone, @mul_inext, @mul_j_body

@mul_j_body
	jmp @mul_t

@mul_t
	%mt =l phi @mul_j_body 0, @mul_tbody %mt1
	%acc =l phi @mul_j_body 0, @mul_tbody %acc1
	%tdone =w cslel %n, %mt
	jnz %tdone, @mul_tdone, @mul_tbody

@mul_tbody
	%xidx =l add %mrow, %mt
	%xoff =l mul %xidx, 8
	%xptr =l add %out, %xoff
	%xval =l loadl %xptr

	%yrow =l mul %mt, %n
	%yidx =l add %yrow, %mj
	%yoff =l mul %yidx, 8
	%yptr =l add %tmp, %yoff
	%yval =l loadl %yptr

	%prodv =l mul %xval, %yval
	%acc1 =l add %acc, %prodv
	%mt1 =l add %mt, 1
	jmp @mul_tbody

@mul_tdone
	%zidx =l add %mrow, %mj
	%zoff =l mul %zidx, 8
	%zptr =l add %prod, %zoff
	storel %acc, %zptr
	%mj1 =l add %mj, 1
	jmp @mul_j

@mul_inext
	%mi1 =l add %mi, 1
	jmp @mul_i

@copy
	%ci =l phi @mul_i 0, @copy_next %ci1
	%cdone =w cslel %nn, %ci
	jnz %cdone, @copy_done, @copy_body

@copy_body
	%coff =l mul %ci, 8
	%csrc =l add %prod, %coff
	%cdst =l add %out, %coff
	%cv =l loadl %csrc
	storel %cv, %cdst
	jmp @copy_next

@copy_next
	%ci1 =l add %ci, 1
	jmp @copy

@copy_done
	%kk1 =l sub %kk, 1
	jmp @pow_loop

@clear
	%cl =l phi @pow_loop 0, @clear_next %cl1
	%cldone =w cslel %two, %cl
	jnz %cldone, @ret, @clear_body

@clear_body
	%cloff =l mul %cl, 8
	%clptr =l add %tmp, %cloff
	storel %zero, %clptr
	jmp @clear_next

@clear_next
	%cl1 =l add %cl, 1
	jmp @clear

@ret
	ret 0
}
