export function l $max_byte(l %p, l %n) {
@entry
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @entry %zero, @cont %next
	%m =l phi @entry %zero, @cont %mnext
	%active =w csltl %i, %n
	jnz %active, @body, @done
@body
	%q =l add %p, %i
	%b =l loadub %q
	%greater =w csltl %m, %b
	jnz %greater, @set, @keep
@set
	jmp @cont
@keep
	jmp @cont
@cont
	%mnext =l phi @set %b, @keep %m
	%next =l add %i, 1
	jmp @loop
@done
	ret %m
}
