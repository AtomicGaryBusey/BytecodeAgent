export function l $reverse(l %a, l %n) {
@entry
	%small =w cslel %n, 1
	jnz %small, @done, @init

@init
	%jm1 =l sub %n, 1
	jmp @loop

@loop
	%i =l phi @init 0, @body %inext
	%j =l phi @init %jm1, @body %jnext
	%active =w csltl %i, %j
	jnz %active, @body, @done

@body
	%ioff =l shl %i, 3
	%joff =l shl %j, 3
	%ip =l add %a, %ioff
	%jp =l add %a, %joff
	%iv =l loadl %ip
	%jv =l loadl %jp
	storel %jv, %ip
	storel %iv, %jp
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop

@done
	ret 0
}
