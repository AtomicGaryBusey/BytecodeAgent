export function l $count_nonzero_prefix(l %a, l %n) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %next
    %inbounds =w csltl %i, %n
    jnz %inbounds, @check, @done
@check
    %off =l mul %i, 8
    %p =l add %a, %off
    %v =l loadl %p
    %zero =w ceql %v, 0
    jnz %zero, @done, @body
@body
    %next =l add %i, 1
    jmp @loop
@done
    ret %i
}
