export function l $crc32(l %p, l %n) {
@start
	%init =w copy 0xFFFFFFFF
	jmp @outer

@outer
	%pc =l phi @start %p, @next %pn
	%rem =l phi @start %n, @next %rn
	%crc =w phi @start %init, @next %c8
	%done =w ceql %rem, 0
	jnz %done, @finish, @byte

@byte
	%b =w loadub %pc
	%c0 =w xor %crc, %b

	%l0 =w and %c0, 1
	%m0 =w neg %l0
	%r0 =w shr %c0, 1
	%x0 =w and 0xEDB88320, %m0
	%c1 =w xor %r0, %x0

	%l1 =w and %c1, 1
	%m1 =w neg %l1
	%r1 =w shr %c1, 1
	%x1 =w and 0xEDB88320, %m1
	%c2 =w xor %r1, %x1

	%l2 =w and %c2, 1
	%m2 =w neg %l2
	%r2 =w shr %c2, 1
	%x2 =w and 0xEDB88320, %m2
	%c3 =w xor %r2, %x2

	%l3 =w and %c3, 1
	%m3 =w neg %l3
	%r3 =w shr %c3, 1
	%x3 =w and 0xEDB88320, %m3
	%c4 =w xor %r3, %x3

	%l4 =w and %c4, 1
	%m4 =w neg %l4
	%r4 =w shr %c4, 1
	%x4 =w and 0xEDB88320, %m4
	%c5 =w xor %r4, %x4

	%l5 =w and %c5, 1
	%m5 =w neg %l5
	%r5 =w shr %c5, 1
	%x5 =w and 0xEDB88320, %m5
	%c6 =w xor %r5, %x5

	%l6 =w and %c6, 1
	%m6 =w neg %l6
	%r6 =w shr %c6, 1
	%x6 =w and 0xEDB88320, %m6
	%c7 =w xor %r6, %x6

	%l7 =w and %c7, 1
	%m7 =w neg %l7
	%r7 =w shr %c7, 1
	%x7 =w and 0xEDB88320, %m7
	%c8 =w xor %r7, %x7

@next
	%pn =l add %pc, 1
	%rn =l sub %rem, 1
	jmp @outer

@finish
	%out =w xor %crc, 0xFFFFFFFF
	%result =l extuw %out
	ret %result
}
