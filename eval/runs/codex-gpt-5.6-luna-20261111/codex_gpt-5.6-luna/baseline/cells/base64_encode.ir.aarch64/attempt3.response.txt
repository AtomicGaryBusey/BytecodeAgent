export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%tmp =l alloc8 32
	%t0 =l add %tmp, 0
	%t1 =l add %tmp, 8
	%t2 =l add %tmp, 16
	%t3 =l add %tmp, 24
	%z =w ceql %n, 0
	jnz %z, @retzero, @loop

@loop
	%ls =l phi @entry %src, @advance %lsn
	%ln =l phi @entry %n, @advance %lnn
	%lo =l phi @entry %out, @advance %lon
	%ltotal =l phi @entry 0, @advance %ltn
	%done =w ceql %ln, 0
	jnz %done, @retotal, @check

@check
	%small =w csltl %ln, 3
	jnz %small, @partial, @full

@full
	%p1 =l add %ls, 1
	%p2 =l add %ls, 2
	%b0 =l loadub %ls
	%b1 =l loadub %p1
	%b2 =l loadub %p2
	%i0 =l shr %b0, 2
	%a1 =l and %b0, 3
	%a2 =l shl %a1, 4
	%b1h =l shr %b1, 4
	%i1 =l or %a2, %b1h
	%b1l =l and %b1, 15
	%a3 =l shl %b1l, 2
	%b2h =l shr %b2, 6
	%i2 =l or %a3, %b2h
	%i3 =l and %b2, 63
	storel %i0, %t0
	storel %i1, %t1
	storel %i2, %t2
	storel %i3, %t3
	jmp @emitprep

@partial
	%one =w ceql %ln, 1
	jnz %one, @one, @two

@one
	%b0o =l loadub %ls
	%j0 =l shr %b0o, 2
	%j1a =l and %b0o, 3
	%j1 =l shl %j1a, 4
	storel %j0, %t0
	storel %j1, %t1
	jmp @emitprep

@two
	%p1t =l add %ls, 1
	%b0t =l loadub %ls
	%b1t =l loadub %p1t
	%j0t =l shr %b0t, 2
	%j1ta =l and %b0t, 3
	%j1tb =l shl %j1ta, 4
	%j1tc =l shr %b1t, 4
	%j1t =l or %j1tb, %j1tc
	%j2ta =l and %b1t, 15
	%j2t =l shl %j2ta, 2
	storel %j0t, %t0
	storel %j1t, %t1
	storel %j2t, %t2
	jmp @emitprep

@emitprep
	%ek =l phi @full 4, @one 2, @two 3
	%em =l phi @full 0, @one 1, @two 1
	%eo =l phi @full %lo, @one %lo, @two %lo
	jmp @emitloop

@emitloop
	%ec =l phi @emitprep 0, @emitagain %ecn
	%epos =l shl %ec, 3
	%eip =l add %tmp, %epos
	%ei =l loadl %eip
	%q26 =w csltl %ei, 26
	%q26l =l extuw %q26
	%m26 =l neg %q26l
	%q52 =w csltl %ei, 52
	%q52l =l extuw %q52
	%m52a =l neg %q52l
	%nm26 =l xor %m26, -1
	%m52 =l and %m52a, %nm26
	%q62 =w csltl %ei, 62
	%q62l =l extuw %q62
	%m62a =l neg %q62l
	%rms =l or %m26, %m52
	%nrms =l xor %rms, -1
	%m62 =l and %m62a, %nrms
	%qeq62 =w ceql %ei, 62
	%qeq62l =l extuw %qeq62
	%mm62 =l neg %qeq62l
	%qeq63 =w ceql %ei, 63
	%qeq63l =l extuw %qeq63
	%mm63 =l neg %qeq63l
	%c26a =l add %ei, 65
	%c52a =l add %ei, 71
	%c62a =l add %ei, -4
	%c26 =l and %c26a, %m26
	%c52 =l and %c52a, %m52
	%c62 =l and %c62a, %m62
	%cplus =l and 43, %mm62
	%cslash =l and 47, %mm63
	%c01 =l or %c26, %c52
	%c23 =l or %c62, %cplus
	%c45 =l or %c01, %c23
	%ch =l or %c45, %cslash
	%opos =l add %eo, %ec
	storeb %ch, %opos
	%ecn =l add %ec, 1
	%edone =w ceql %ecn, %ek
	jnz %edone, @emitdone, @emitagain

@emitagain
	jmp @emitloop

@emitdone
	jnz %em, @pad, @advance

@pad
	%padp =l add %eo, %ek
	storeb 61, %padp
	%twoeq =w ceql %ek, 2
	jnz %twoeq, @pad2, @retpartial

@pad2
	%padp2 =l add %padp, 1
	storeb 61, %padp2
	jmp @retpartial

@retpartial
	%rpartial =l add %ltotal, 4
	ret %rpartial

@advance
	%lsn =l add %ls, 3
	%lnn =l add %ln, -3
	%lon =l add %lo, 4
	%ltn =l add %ltotal, 4
	jmp @loop

@retotal
	ret %ltotal

@retzero
	ret 0
}
