export function l $base64_encode(l %src, l %n, l %out) {
@entry
	%tmp =l alloc8 32
	%z =w ceql %n, 0
	jnz %z, @retzero, @check
@check
	%small =w csltl %n, 3
	jnz %small, @partial, @full
@full
	%p1 =l add %src, 1
	%p2 =l add %src, 2
	%b0 =l loadub %src
	%b1 =l loadub %p1
	%b2 =l loadub %p2
	%i0 =l shr %b0, 2
	%a1 =l and %b0, 3
	%a2 =l shl %a1, 4
	%b1h =l shr %b1, 4
	%i1 =l or %a2, %b1h
	%b1l =l and %b1, 15
	%a3 =l shl %b1l, 2
	%b2h =l shr %b2, 6
	%i2 =l or %a3, %b2h
	%i3 =l and %b2, 63
	%t0 =l add %tmp, 0
	%t1 =l add %tmp, 8
	%t2 =l add %tmp, 16
	%t3 =l add %tmp, 24
	storel %i0, %t0
	storel %i1, %t1
	storel %i2, %t2
	storel %i3, %t3
	jmp @emitfull
@partial
	%one =w ceql %n, 1
	jnz %one, @one, @two
@one
	%b0o =l loadub %src
	%j0 =l shr %b0o, 2
	%j1a =l and %b0o, 3
	%j1 =l shl %j1a, 4
	storel %j0, %tmp
	storel %j1, %t1
	%k1 =l 2
	jmp @emitpartial
@two
	%p1t =l add %src, 1
	%b0t =l loadub %src
	%b1t =l loadub %p1t
	%j0t =l shr %b0t, 2
	%j1ta =l and %b0t, 3
	%j1tb =l shl %j1ta, 4
	%j1tc =l shr %b1t, 4
	%j1t =l or %j1tb, %j1tc
	%j2ta =l and %b1t, 15
	%j2t =l shl %j2ta, 2
	storel %j0t, %tmp
	storel %j1t, %t1
	storel %j2t, %t2
	%k2 =l 3
	jmp @emitpartial
@emitfull
	%efk =l 4
	%efm =l 0
	jmp @emitprep
@emitpartial
	%epk =l phi @one %k1, @two %k2
	%epm =l 1
	jmp @emitprep
@emitprep
	%ek =l phi @emitfull %efk, @emitpartial %epk
	%em =l phi @emitfull %efm, @emitpartial %epm
	%ec =l 0
	%eo =l %out
	jmp @emitloop
@emitloop
	%epos =l shl %ec, 3
	%eip =l add %tmp, %epos
	%ei =l loadl %eip
	%q26 =w csltl %ei, 26
	%q26l =l extuw %q26
	%m26 =l neg %q26l
	%q52 =w csltl %ei, 52
	%q52l =l extuw %q52
	%nm26 =l xor %m26, -1
	%m52a =l neg %q52l
	%m52 =l and %m52a, %nm26
	%q62 =w csltl %ei, 62
	%q62l =l extuw %q62
	%rms =l or %m26, %m52
	%nrms =l xor %rms, -1
	%m62a =l neg %q62l
	%m62 =l and %m62a, %nrms
	%qeq62 =w ceql %ei, 62
	%qeq62l =l extuw %qeq62
	%mm62 =l neg %qeq62l
	%qeq63 =w ceql %ei, 63
	%qeq63l =l extuw %qeq63
	%mm63 =l neg %qeq63l
	%c26a =l add %ei, 65
	%c52a =l add %ei, 71
	%c62a =l add %ei, -4
	%c26 =l and %c26a, %m26
	%c52 =l and %c52a, %m52
	%c62 =l and %c62a, %m62
	%cplus =l and 43, %mm62
	%cslash =l and 47, %mm63
	%c01 =l or %c26, %c52
	%c23 =l or %c62, %cplus
	%c45 =l or %c01, %c23
	%ch =l or %c45, %cslash
	%opos =l add %eo, %ec
	storeb %ch, %opos
	%ecnext =l add %ec, 1
	%eonext =l add %eo, 1
	%edone =w ceql %ecnext, %ek
	jnz %edone, @emitdone, @emitagain
@emitagain
	%ec =l %ecnext
	%eo =l %eonext
	jmp @emitloop
@emitdone
	jnz %em, @pad, @nextfull
@nextfull
	%srcnext =l add %src, 3
	%nnext =l add %n, -3
	%outnext =l add %out, 4
	%src =l %srcnext
	%n =l %nnext
	%out =l %outnext
	%znext =w ceql %nnext, 0
	jnz %znext, @retzero, @check
@pad
	%padoff =l sub %ek, 2
	%padp =l add %eo, %padoff
	storeb 61, %padp
	%twoeq =w ceql %ek, 2
	jnz %twoeq, @pad2, @retpartial
@pad2
	%padp2 =l add %padp, 1
	storeb 61, %padp2
	jmp @retpartial
@retpartial
	%resultp =l sub %eo, %out
	ret %resultp
@retzero
	ret 0
}
