export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%li =l alloc8 8
	%lj =l alloc8 8
	%lt =l alloc8 8
	%lk =l alloc8 8
	%acc =l alloc8 8
	%nn =l mul %n, %n
	%second =l mul %nn, 8
	storel %k, %lk
	storel 0, %li
	jmp @id_i

@id_i
	%ii =l loadl %li
	%ic =w csltl %ii, %n
	jnz %ic, @id_j_init, @copy_init

@id_j_init
	storel 0, %lj
	jmp @id_j

@id_j
	%jj =l loadl %lj
	%jc =w csltl %jj, %n
	jnz %jc, @id_val, @id_i_next

@id_val
	%ieq =w ceql %ii, %jj
	jnz %ieq, @id_one, @id_zero

@id_one
	%ioff =l mul %ii, %n
	%iidx =l add %ioff, %jj
	%ibyte =l shl %iidx, 3
	%iaddr =l add %out, %ibyte
	storel 1, %iaddr
	jmp @id_j_next

@id_zero
	%zoff =l mul %ii, %n
	%zidx =l add %zoff, %jj
	%zbyte =l shl %zidx, 3
	%zaddr =l add %out, %zbyte
	storel 0, %zaddr
	jmp @id_j_next

@id_j_next
	%jn =l add %jj, 1
	storel %jn, %lj
	jmp @id_j

@id_i_next
	%in =l add %ii, 1
	storel %in, %li
	jmp @id_i

@copy_init
	storel 0, %li
	jmp @copy_loop

@copy_loop
	%ci =l loadl %li
	%cc =w csltl %ci, %nn
	jnz %cc, @copy_do, @pow_check

@copy_do
	%cbyte =l shl %ci, 3
	%csrc =l add %a, %cbyte
	%cdst =l add %tmp, %cbyte
	%cv =l loadl %csrc
	storel %cv, %cdst
	%cin =l add %ci, 1
	storel %cin, %li
	jmp @copy_loop

@pow_check
	%kv =l loadl %lk
	%kc =w csgtl %kv, 0
	jnz %kc, @prod_i_init, @clear_init

@prod_i_init
	storel 0, %li
	jmp @prod_i

@prod_i
	%pi =l loadl %li
	%pic =w csltl %pi, %n
	jnz %pic, @prod_j_init, @copy_prod_init

@prod_j_init
	storel 0, %lj
	jmp @prod_j

@prod_j
	%pj =l loadl %lj
	%pjc =w csltl %pj, %n
	jnz %pjc, @prod_cell_init, @prod_i_next

@prod_cell_init
	storel 0, %lt
	storel 0, %acc
	jmp @prod_t

@prod_t
	%pt =l loadl %lt
	%ptc =w csltl %pt, %n
	jnz %ptc, @prod_term, @prod_store

@prod_term
	%pioff =l mul %pi, %n
	%piidx =l add %pioff, %pt
	%pibyte =l shl %piidx, 3
	%paddr =l add %out, %pibyte
	%px =l loadl %paddr

	%ptoff =l mul %pt, %n
	%ptidx =l add %ptoff, %pj
	%ptbyte =l shl %ptidx, 3
	%ptaddr =l add %tmp, %ptbyte
	%py =l loadl %ptaddr

	%pmul =l mul %px, %py
	%oldacc =l loadl %acc
	%newacc =l add %oldacc, %pmul
	storel %newacc, %acc

	%ptn =l add %pt, 1
	storel %ptn, %lt
	jmp @prod_t

@prod_store
	%poff =l mul %pi, %n
	%pidx =l add %poff, %pj
	%pbyte =l shl %pidx, 3
	%pbase =l add %tmp, %second
	%paddr2 =l add %pbase, %pbyte
	%pval =l loadl %acc
	storel %pval, %paddr2

	%pjn =l add %pj, 1
	storel %pjn, %lj
	jmp @prod_j

@prod_i_next
	%pin =l add %pi, 1
	storel %pin, %li
	jmp @prod_i

@copy_prod_init
	storel 0, %li
	jmp @copy_prod

@copy_prod
	%qi =l loadl %li
	%qic =w csltl %qi, %nn
	jnz %qic, @copy_prod_do, @pow_dec

@copy_prod_do
	%qbyte =l shl %qi, 3
	%qsrcbase =l add %tmp, %second
	%qsrc =l add %qsrcbase, %qbyte
	%qdst =l add %out, %qbyte
	%qv =l loadl %qsrc
	storel %qv, %qdst
	%qin =l add %qi, 1
	storel %qin, %li
	jmp @copy_prod

@pow_dec
	%qkv =l loadl %lk
	%qkn =l sub %qkv, 1
	storel %qkn, %lk
	jmp @pow_check

@clear_init
	storel 0, %li
	%twonn =l mul %nn, 2
	jmp @clear_loop

@clear_loop
	%xi =l loadl %li
	%xic =w csltl %xi, %twonn
	jnz %xic, @clear_do, @done

@clear_do
	%xbyte =l shl %xi, 3
	%xaddr =l add %tmp, %xbyte
	storel 0, %xaddr
	%xn =l add %xi, 1
	storel %xn, %li
	jmp @clear_loop

@done
	ret 0
}
