export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
    jmp @loop
@loop
    %i =l phi @entry 0, @body %i_next
    %count =l phi @entry 0, @body %count_next
    %d =l phi @entry %dst, @body %d_next
    %s =l phi @entry %src, @body %s_next
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %x =l loadl %s
    %y =l call %f(l %x)
    storel %y, %d
    %less =w csltl %y, %x
    %less_l =l extuw %less
    %i_next =l add %i, 1
    %count_next =l add %count, %less_l
    %d_next =l add %d, 8
    %s_next =l add %s, 8
    jmp @loop
@done
    ret %count
}
