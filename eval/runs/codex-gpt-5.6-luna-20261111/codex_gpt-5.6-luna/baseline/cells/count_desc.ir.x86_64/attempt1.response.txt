export function l $count_desc(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @zero, @init

@zero
	ret 0

@init
	%z =l copy 0
	jmp @loop

@loop
	%i =l phi @init %z, @advance %next
	%cnt =l phi @init %z, @advance %cntnext
	%next =l add %i, 1
	%hasnext =w csltl %next, %n
	jnz %hasnext, @read, @done

@read
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%off2 =l shl %next, 3
	%p2 =l add %a, %off2
	%y =l loadl %p2
	%gt =w csgtl %x, %y
	%inc =l extuw %gt
	%cntnext =l add %cnt, %inc
	jmp @advance

@advance
	jmp @loop

@done
	ret %cnt
}
