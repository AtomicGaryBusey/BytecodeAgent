export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    jmp @init

@init
    %j =l phi @entry 0, @init %jn
    %ok =w cslel %j, %n
    jnz %ok, @init_body, @outer

@init_body
    %off =l mul %j, 8
    %p =l add %row, %off
    storel 0, %p
    %jn =l add %j, 1
    jmp @init

@outer
    %i =l phi @init 0, @inner_done %in
    %ok_i =w csltl %i, %m
    jnz %ok_i, @outer_body, @done

@outer_body
    %xp =l add %x, %i
    %xb =w loadub %xp
    jmp @inner

@inner
    %j2 =l phi @outer_body 1, @write %jnext
    %diag =l phi @outer_body 0, @write %up
    %ok_j =w cslel %j2, %n
    jnz %ok_j, @inner_body, @inner_done

@inner_body
    %off2 =l mul %j2, 8
    %rp =l add %row, %off2
    %up =l loadl %rp
    %jm1 =l sub %j2, 1
    %yp =l add %y, %jm1
    %yb =w loadub %yp
    %eq =w ceql %xb, %yb
    jnz %eq, @match, @nomatch

@match
    %matched =l add %diag, 1
    jmp @write

@nomatch
    %less =w csltl %up, %left
    %left =l loadl %rp
    %less2 =w csltl %up, %left
    jnz %less2, @take_left, @take_up

@take_left
    jmp @write

@take_up
    jmp @write

@write
    %left =l sub %rp, 8
    %value =l phi @match %matched, @take_left %left, @take_up %up
    storel %value, %rp
    %jnext =l add %j2, 1
    jmp @inner

@inner_done
    %in =l add %i, 1
    jmp @outer

@done
    %offn =l mul %n, 8
    %end =l add %row, %offn
    %result =l loadl %end
    ret %result
}
