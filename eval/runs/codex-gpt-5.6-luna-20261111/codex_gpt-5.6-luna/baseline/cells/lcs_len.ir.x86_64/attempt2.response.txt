export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    jmp @init

@init
    %j =l phi @entry 0, @init %jnext
    %ok =w cslel %j, %n
    jnz %ok, @init_body, @outer

@init_body
    %off =l mul %j, 8
    %p =l add %row, %off
    storel 0, %p
    %jnext =l add %j, 1
    jmp @init

@outer
    %i =l phi @init 0, @inner_done %in
    %ok_i =w csltl %i, %m
    jnz %ok_i, @outer_body, @done

@outer_body
    %xp =l add %x, %i
    %xb =w loadub %xp
    jmp @inner

@inner
    %j2 =l phi @outer_body 1, @write_match %jnext_m, @write_left %jnext_l, @write_up %jnext_u
    %diag =l phi @outer_body 0, @write_match %up, @write_left %up, @write_up %up
    %ok_j =w cslel %j2, %n
    jnz %ok_j, @inner_body, @inner_done

@inner_body
    %off2 =l mul %j2, 8
    %rp =l add %row, %off2
    %up =l loadl %rp
    %jm1 =l sub %j2, 1
    %lp =l add %rp, -8
    %left =l loadl %lp
    %yp =l add %y, %jm1
    %yb =w loadub %yp
    %eq =w ceql %xb, %yb
    jnz %eq, @match, @nomatch

@match
    %value_m =l add %diag, 1
    storel %value_m, %rp
    %jnext_m =l add %j2, 1
    jmp @inner

@nomatch
    %less =w csltl %up, %left
    jnz %less, @take_left, @take_up

@take_left
    storel %left, %rp
    %jnext_l =l add %j2, 1
    jmp @inner

@take_up
    storel %up, %rp
    %jnext_u =l add %j2, 1
    jmp @inner

@inner_done
    %in =l add %i, 1
    jmp @outer

@done
    %offn =l mul %n, 8
    %end =l add %row, %offn
    %result =l loadl %end
    ret %result
}
