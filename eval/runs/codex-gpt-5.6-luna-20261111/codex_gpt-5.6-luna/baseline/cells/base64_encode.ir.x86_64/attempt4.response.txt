export function l $base64_encode(l %src, l %n, l %out) {
@entry
jmp @loop

@loop
%i =l phi @entry 0, @next %inext
%o =l phi @entry 0, @next %onext
%more =w csltl %i, %n
jnz %more, @body, @done

@body
%rem =l sub %n, %i
%p0 =l add %src, %i
%b0 =l loadub %p0
%has1 =w csgtl %rem, 1
jnz %has1, @load1, @no1

@load1
%p1 =l add %p0, 1
%b1x =l loadub %p1
jmp @join1

@no1
%b1z =l copy 0
jmp @join1

@join1
%b1 =l phi @load1 %b1x, @no1 %b1z
%has2 =w csgtl %rem, 2
jnz %has2, @load2, @no2

@load2
%p2 =l add %p0, 2
%b2x =l loadub %p2
jmp @join2

@no2
%b2z =l copy 0
jmp @join2

@join2
%b2 =l phi @load2 %b2x, @no2 %b2z

%v0a =l shr %b0, 2
%v1a =l and %b0, 3
%v1b =l shl %v1a, 4
%v1c =l shr %b1, 4
%v1 =l or %v1b, %v1c
%v2a =l and %b1, 15
%v2b =l shl %v2a, 2
%v2c =l shr %b2, 6
%v2 =l or %v2b, %v2c
%v3 =l and %b2, 63

%q0 =w csltl %v0a, 26
jnz %q0, @m0a, @m0b
@m0a
%m0av =l add %v0a, 65
jmp @m0j
@m0b
%q0b =w csltl %v0a, 52
jnz %q0b, @m0c, @m0d
@m0c
%m0cv =l add %v0a, 71
jmp @m0j
@m0d
%q0d =w csltl %v0a, 62
jnz %q0d, @m0e, @m0f
@m0e
%m0ev =l sub %v0a, 4
jmp @m0j
@m0f
%q0f =w ceql %v0a, 62
jnz %q0f, @m0g, @m0h
@m0g
%m0gv =l copy 43
jmp @m0j
@m0h
%m0hv =l copy 47
jmp @m0j
@m0j
%a0 =l phi @m0a %m0av, @m0c %m0cv, @m0e %m0ev, @m0g %m0gv, @m0h %m0hv

%q1 =w csltl %v1, 26
jnz %q1, @m1a, @m1b
@m1a
%m1av =l add %v1, 65
jmp @m1j
@m1b
%q1b =w csltl %v1, 52
jnz %q1b, @m1c, @m1d
@m1c
%m1cv =l add %v1, 71
jmp @m1j
@m1d
%q1d =w csltl %v1, 62

@m1e
%m1ev =l sub %v1, 4
jmp @m1j
@m1f
%q1f =w ceql %v1, 62
jnz %q1f, @m1g, @m1h
@m1g
%m1gv =l copy 43
jmp @m1j
@m1h
%m1hv =l copy 47
jmp @m1j
@m1j
%a1 =l phi @m1a %m1av, @m1c %m1cv, @m1e %m1ev, @m1g %m1gv, @m1h %m1hv

%q2 =w csltl %v2, 26
jnz %q2, @m2a, @m2b
@m2a
%m2av =l add %v2, 65
jmp @m2j
@m2b
%q2b =w csltl %v2, 52
jnz %q2b, @m2c, @m2d
@m2c
%m2cv =l add %v2, 71
jmp @m2j
@m2d
%q2d =w csltl %v2, 62
jnz %q2d, @m2e, @m2f
@m2e
%m2ev =l sub %v2, 4
jmp @m2j
@m2f
%q2f =w ceql %v2, 62
jnz %q2f, @m2g, @m2h
@m2g
%m2gv =l copy 43
jmp @m2j
@m2h
%m2hv =l copy 47
jmp @m2j
@m2j
%a2 =l phi @m2a %m2av, @m2c %m2cv, @m2e %m2ev, @m2g %m2gv, @m2h %m2hv

%q3 =w csltl %v3, 26
jnz %q3, @m3a, @m3b
@m3a
%m3av =l add %v3, 65
jmp @m3j
@m3b

@m3j

@pad2

@pad3

storeb 61, %op3b

@nextout
%inext =l add %i, 3
%onext =l add %o, 4
jmp @next

@next
jmp @loop

@done
ret %o
}
