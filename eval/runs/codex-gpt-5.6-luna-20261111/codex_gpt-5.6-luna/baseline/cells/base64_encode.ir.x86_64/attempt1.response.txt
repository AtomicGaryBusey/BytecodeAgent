export function l $base64_encode(l %src, l %n, l %out) {
@loop
%i =l phi @entry 0, @next %inext
%o =l phi @entry 0, @next %onext
%more =w csltl %i, %n
jnz %more, @body, @done

@body
%rem =l sub %n, %i
%p0 =l add %src, %i
%b0 =l loadub %p0
%has1 =w csgtl %rem, 1
jnz %has1, @load1, @no1

@load1
%p1 =l add %p0, 1
%x1 =l loadub %p1
jmp @join1

@no1
%x1z =l copy 0
jmp @join1

@join1
%b1 =l phi @load1 %x1, @no1 %x1z
%has2 =w csgtl %rem, 2
jnz %has2, @load2, @no2

@load2
%p2 =l add %p0, 2
%x2 =l loadub %p2
jmp @join2

@no2
%x2z =l copy 0
jmp @join2

@join2
%b2 =l phi @load2 %x2, @no2 %x2z
%v0a =l shr %b0, 2
%v1a =l and %b0, 3
%v1b =l shl %v1a, 4
%v1c =l shr %b1, 4
%v1 =l or %v1b, %v1c
%v2a =l and %b1, 15
%v2b =l shl %v2a, 2
%v2c =l shr %b2, 6
%v2 =l or %v2b, %v2c
%v3 =l and %b2, 63

%a0 =l add %v0a, 65
%a0b =l add %v0a, 71
%a0c =l sub %v0a, 4
%q0a =w csltl %v0a, 26
%q0b =w csltl %v0a, 52
%a0d =l select %q0a, %a0, %a0b
%a0e =l select %q0b, %a0d, %a0c
%q0c =w csltl %v0a, 62
%q0d =w ceql %v0a, 62
%a0f =l select %q0d, 43, 47
%a0g =l select %q0c, %a0e, %a0f

%a1 =l add %v1, 65
%a1b =l add %v1, 71
%a1c =l sub %v1, 4
%q1a =w csltl %v1, 26
%q1b =w csltl %v1, 52
%a1d =l select %q1a, %a1, %a1b
%a1e =l select %q1b, %a1d, %a1c
%q1c =w csltl %v1, 62
%q1d =w ceql %v1, 62
%a1f =l select %q1d, 43, 47
%a1g =l select %q1c, %a1e, %a1f

%a2 =l add %v2, 65
%a2b =l add %v2, 71
%a2c =l sub %v2, 4
%q2a =w csltl %v2, 26
%q2b =w csltl %v2, 52
%a2d =l select %q2a, %a2, %a2b
%a2e =l select %q2b, %a2d, %a2c
%q2c =w csltl %v2, 62
%q2d =w ceql %v2, 62
%a2f =l select %q2d, 43, 47
%a2g =l select %q2c, %a2e, %a2f

%a3 =l add %v3, 65
%a3b =l add %v3, 71
%a3c =l sub %v3, 4
%q3a =w csltl %v3, 26
%q3b =w csltl %v3, 52
%a3d =l select %q3a, %a3, %a3b
%a3e =l select %q3b, %a3d, %a3c
%q3c =w csltl %v3, 62
%q3d =w ceql %v3, 62
%a3f =l select %q3d, 43, 47
%a3g =l select %q3c, %a3e, %a3f

%op =l add %out, %o
storeb %a0g, %op
%op1 =l add %op, 1
storeb %a1g, %op1
%op2 =l add %op, 2
%has2w =w csgtl %rem, 2
%a2out =l select %has2w, %a2g, 61
storeb %a2out, %op2
%op3 =l add %op, 3
%a3out =l select %has2w, %a3g, 61
storeb %a3out, %op3
%inext =l add %i, 3
%onext =l add %o, 4
jmp @loop

@entry
jmp @loop

@done
ret %o
}
