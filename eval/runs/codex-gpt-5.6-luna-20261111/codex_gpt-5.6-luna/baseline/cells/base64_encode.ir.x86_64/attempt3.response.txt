export function l $base64_encode(l %src, l %n, l %out) {
@entry
jmp @loop

@loop
%i =l phi @entry 0, @next %inext
%o =l phi @entry 0, @next %onext
%more =w csltl %i, %n
jnz %more, @body, @done

@body
%rem =l sub %n, %i
%p0 =l add %src, %i
%b0 =l loadub %p0
%has1 =w csgtl %rem, 1
jnz %has1, @load1, @no1

@load1
%p1 =l add %p0, 1
%b1x =l loadub %p1
jmp @join1

@no1
%b1z =l copy 0
jmp @join1

@join1
%b1 =l phi @load1 %b1x, @no1 %b1z
%has2 =w csgtl %rem, 2
jnz %has2, @load2, @no2

@load2
%p2 =l add %p0, 2
%b2x =l loadub %p2
jmp @join2

@no2
%b2z =l copy 0
jmp @join2

@join2
%b2 =l phi @load2 %b2x, @no2 %b2z

%v0a =l shr %b0, 2
%v1a =l and %b0, 3
%v1b =l shl %v1a, 4
%v1c =l shr %b1, 4
%v1 =l or %v1b, %v1c
%v2a =l and %b1, 15
%v2b =l shl %v2a, 2
%v2c =l shr %b2, 6
%v2 =l or %v2b, %v2c
%v3 =l and %b2, 63

%a0a =l add %v0a, 65
%a0b =l add %v0a, 71
%a0c =l sub %v0a, 4
%q0a =w csltl %v0a, 26
%q0b =w csltl %v0a, 52
%q0c =w csltl %v0a, 62
%q0d =w ceql %v0a, 62
%q0al =l extuw %q0a
%q0bl =l extuw %q0b
%q0cl =l extuw %q0c
%q0dl =l extuw %q0d
%m0a =l sub 0, %q0al
%m0b =l sub 0, %q0bl
%m0c =l sub 0, %q0cl
%m0d =l sub 0, %q0dl
%nm0a =l xor %m0a, -1
%nm0b =l xor %m0b, -1
%nm0c =l xor %m0c, -1
%nm0d =l xor %m0d, -1
%t0a =l or (and %a0a, %m0a), (and %a0b, %nm0a)
%t0b =l or (and %t0a, %m0b), (and %a0c, %nm0b)
%t0c =l or (and 43, %m0d), (and 47, %nm0d)
%a0 =l or (and %t0b, %m0c), (and %t0c, %nm0c)

%a1a =l add %v1, 65
%a1b =l add %v1, 71
%a1c =l sub %v1, 4
%q1a =w csltl %v1, 26
%q1b =w csltl %v1, 52
%q1c =w csltl %v1, 62
%q1d =w ceql %v1, 62
%q1al =l extuw %q1a
%q1bl =l extuw %q1b
%q1cl =l extuw %q1c
%q1dl =l extuw %q1d
%m1a =l sub 0, %q1al
%m1b =l sub 0, %q1bl
%m1c =l sub 0, %q1cl
%m1d =l sub 0, %q1dl
%nm1a =l xor %m1a, -1
%nm1b =l xor %m1b, -1
%nm1c =l xor %m1c, -1
%nm1d =l xor %m1d, -1
%t1a =l or (and %a1a, %m1a), (and %a1b, %nm1a)
%t1b =l or (and %t1a, %m1b), (and %a1c, %nm1b)
%t1c =l or (and 43, %m1d), (and 47, %nm1d)
%a1 =l or (and %t1b, %m1c), (and %t1c, %nm1c)

%a2a =l add %v2, 65
%a2b =l add %v2, 71
%a2c =l sub %v2, 4
%q2a =w csltl %v2, 26
%q2b =w csltl %v2, 52
%q2c =w csltl %v2, 62
%q2d =w ceql %v2, 62
%q2al =l extuw %q2a
%q2bl =l extuw %q2b
%q2cl =l extuw %q2c
%q2dl =l extuw %q2d
%m2a =l sub 0, %q2al
%m2b =l sub 0, %q2bl
%m2c =l sub 0, %q2cl
%m2d =l sub 0, %q2dl
%nm2a =l xor %m2a, -1
%nm2b =l xor %m2b, -1
%nm2c =l xor %m2c, -1
%nm2d =l xor %m2d, -1
%t2a =l or (and %a2a, %m2a), (and %a2b, %nm2a)
%t2b =l or (and %t2a, %m2b), (and %a2c, %nm2b)
%t2c =l or (and 43, %m2d), (and 47, %nm2d)
%a2 =l or (and %t2b, %m2c), (and %t2c, %nm2c)

%a3a =l add %v3, 65
%a3b =l add %v3, 71
%a3c =l sub %v3, 4
%q3a =w csltl %v3, 26
%q3b =w csltl %v3, 52
%q3c =w csltl %v3, 62
%q3d =w ceql %v3, 62
%q3al =l extuw %q3a
%q3bl =l extuw %q3b
%q3cl =l extuw %q3c
%q3dl =l extuw %q3d
%m3a =l sub 0, %q3al
%m3b =l sub 0, %q3bl
%m3c =l sub 0, %q3cl
%m3d =l sub 0, %q3dl
%nm3a =l xor %m3a, -1
%nm3b =l xor %m3b, -1
%nm3c =l xor %m3c, -1
%nm3d =l xor %m3d, -1
%t3a =l or (and %a3a, %m3a), (and %a3b, %nm3a)
%t3b =l or (and %t3a, %m3b), (and %a3c, %nm3b)
%t3c =l or (and 43, %m3d), (and 47, %nm3d)
%a3 =l or (and %t3b, %m3c), (and %t3c, %nm3c)

%op =l add %out, %o
storeb %a0, %op
%op1 =l add %op, 1
storeb %a1, %op1

%h2 =l extuw %has2
%mh2 =l sub 0, %h2
%nmh2 =l xor %mh2, -1
%a2out =l or (and %a2, %mh2), (and 61, %nmh2)
%op2 =l add %op, 2
storeb %a2out, %op2

%a3out =l or (and %a3, %mh2), (and 61, %nmh2)
%op3 =l add %op, 3
storeb %a3out, %op3

%inext =l add %i, 3
%onext =l add %o, 4
jmp @next

@next
jmp @loop

@done
ret %o
}
