export function l $sum_to(l %n) {
@start
	%sum =l alloc8 8
	%i =l alloc8 8
	storel 0, %sum
	storel %n, %i
	jmp @check
@check
	%iv =l loadl %i
	%positive =w csgtl %iv, 0
	jnz %positive, @body, @done
@body
	%current =l loadl %sum
	%next =l add %current, %iv
	storel %next, %sum
	%incremented =l add %iv, 1
	storel %incremented, %i
	jmp @check
@done
	%result =l loadl %sum
	ret %result
}
