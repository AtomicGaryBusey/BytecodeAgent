packbits_encode:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebx, ebx
xor ebp, ebp
test r13, r13
jz .Lsuccess
.Louter:
cmp rbx, r13
jae .Lsuccess
mov r10, 1
movzx r11d, byte ptr [r12 + rbx]
.Lmeasure:
cmp r10, 128
jae .Lmeasure_done
lea rax, [rbx + r10]
cmp rax, r13
jae .Lmeasure_done
movzx edx, byte ptr [r12 + rax]
cmp dl, r11b
jne .Lmeasure_done
inc r10
jmp .Lmeasure
.Lmeasure_done:
cmp r10, 3
jae .Lrepeat
mov r8, rbx
mov r9, rbx
.Lliteral_scan:
mov rax, r9
sub rax, r8
cmp rax, 128
jae .Lliteral_done
cmp r9, r13
jae .Lliteral_done
lea rax, [r9 + 2]
cmp rax, r13
jae .Lliteral_continue
movzx eax, byte ptr [r12 + r9]
movzx edx, byte ptr [r12 + r9 + 1]
cmp al, dl
jne .Lliteral_continue
movzx edx, byte ptr [r12 + r9 + 2]
cmp al, dl
je .Lliteral_done
.Lliteral_continue:
inc r9
jmp .Lliteral_scan
.Lliteral_done:
mov r10, r9
sub r10, r8
lea rax, [rbp + r10 + 1]
cmp rax, r15
ja .Lfail
mov eax, r10d
dec eax
mov byte ptr [r14 + rbp], al
mov rdi, r12
add rdi, r8
mov rdx, r14
add rdx, rbp
inc rdx
xor eax, eax
.Lliteral_copy:
cmp rax, r10
jae .Lliteral_copied
mov cl, byte ptr [rdi]
mov byte ptr [rdx], cl
inc rdi
inc rdx
inc rax
jmp .Lliteral_copy
.Lliteral_copied:
lea rax, [r10 + 1]
add rbp, rax
mov rbx, r9
cmp rbx, r13
jb .Louter
jmp .Lsuccess
.Lrepeat:
lea rax, [rbp + 2]
cmp rax, r15
ja .Lfail
mov eax, 257
sub eax, r10d
mov byte ptr [r14 + rbp], al
mov byte ptr [r14 + rbp + 1], r11b
add rbp, 2
add rbx, r10
cmp rbx, r13
jb .Louter
.Lsuccess:
mov rax, rbp
cld
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
.Lfail:
mov rax, -1
cld
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
