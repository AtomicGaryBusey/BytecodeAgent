export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
%x0 =l loadl %a
%y0 =l call %f(l %x0)
%bit0 =l and %y0, 1
%odd0 =w cnel %bit0, 0
%cnt0 =l extuw %odd0
%off1 =l copy 8
%i0 =l copy 1
jmp @loop

@loop
%i =l phi @start %i0, @next %in
%mn =l phi @start %y0, @next %minn
%mx =l phi @start %y0, @next %maxn
%sum =l phi @start %y0, @next %sumn
%cnt =l phi @start %cnt0, @next %cntn
%more =w csltl %i, %n
jnz %more, @body, @done

@body
%off =l mul %i, 8
%p =l add %a, %off
%x =l loadl %p
%y =l call %f(l %x)
%ismin =w csltl %y, %mn
jnz %ismin, @min_yes, @min_no

@min_yes
jmp @min_merge

@min_no
jmp @min_merge

@min_merge
%minv =l phi @min_yes %y, @min_no %mn
%ismax =w csgtl %y, %mx
jnz %ismax, @max_yes, @max_no

@max_yes
jmp @max_merge

@max_no
jmp @max_merge

@max_merge
%maxv =l phi @max_yes %y, @max_no %mx
%bit =l and %y, 1
%odd =w cnel %bit, 0
%oddl =l extuw %odd
%in =l add %i, 1
%minn =l copy %minv
%maxn =l copy %maxv
%sumn =l add %sum, %y
%cntn =l add %cnt, %oddl
jmp @next

@next
jmp @loop

@done
storel %mn, %out
%out1 =l add %out, 8
storel %mx, %out1
%out2 =l add %out, 16
storel %sum, %out2
ret %cnt
}
