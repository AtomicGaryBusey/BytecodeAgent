export function l $add_overflow(l %a, l %b, l %out) {
@entry
%sum =l add %a, %b
storel %sum, %out
%na =w csltl %a, 0
jnz %na, @aneg, @apos

@aneg
%nb =w csltl %b, 0
jnz %nb, @bothneg, @done

@apos
%nb2 =w csltl %b, 0
jnz %nb2, @done, @bothpos

@bothneg
%ns =w csltl %sum, 0
jnz %ns, @done, @overflow

@bothpos
%ns2 =w csltl %sum, 0
jnz %ns2, @overflow, @done

@overflow
ret 1

@done
ret 0
}
