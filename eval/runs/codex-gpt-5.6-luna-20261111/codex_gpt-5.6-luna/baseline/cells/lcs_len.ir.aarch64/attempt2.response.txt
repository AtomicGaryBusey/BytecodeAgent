export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    jmp @init

@init
    %k =l phi @entry 0, @init_store %knext
    %kcond =w cslel %k, %n
    jnz %kcond, @init_store, @after_init

@init_store
    %koff =l mul %k, 8
    %kaddr =l add %row, %koff
    storel 0, %kaddr
    %knext =l add %k, 1
    jmp @init

@after_init
    jmp @outer

@outer
    %i =l phi @after_init 0, @outer_next %inext
    %icond =w csltl %i, %m
    jnz %icond, @outer_start, @finish

@outer_start
    jmp @inner

@inner
    %j =l phi @outer_start 1, @write %jnext
    %diag =l phi @outer_start 0, @write %upw
    %jcond =w cslel %j, %n
    jnz %jcond, @work, @outer_next

@work
    %joff =l mul %j, 8
    %jaddr =l add %row, %joff
    %up =l loadl %jaddr
    %jm1 =l sub %j, 1
    %jm1off =l mul %jm1, 8
    %leftaddr =l add %row, %jm1off
    %left =l loadl %leftaddr
    %xoff =l add %x, %i
    %xb =w loadub %xoff
    %yidx =l sub %j, 1
    %yoff =l add %y, %yidx
    %yb =w loadub %yoff
    %same =w ceqw %xb, %yb
    jnz %same, @match, @nomatch

@match
    %matchval =l add %diag, 1
    jmp @write

@nomatch
    %chooseup =w csgel %up, %left
    jnz %chooseup, @useup, @useleft

@useup
    %maxvalup =l copy %up
    jmp @nomatch_done

@useleft
    %maxvalleft =l copy %left
    jmp @nomatch_done

@nomatch_done
    %maxval =l phi @useup %maxvalup, @useleft %maxvalleft
    jmp @write_nomatch

@write_nomatch
    %newval =l copy %maxval
    jmp @write

@write
    %val =l phi @match %matchval, @write_nomatch %newval
    %upw =l phi @match %up, @write_nomatch %up
    storel %val, %jaddr
    %jnext =l add %j, 1
    jmp @inner

@outer_next
    %inext =l add %i, 1
    jmp @outer

@finish
    %finaloff =l mul %n, 8
    %finaladdr =l add %row, %finaloff
    %result =l loadl %finaladdr
    ret %result
}
