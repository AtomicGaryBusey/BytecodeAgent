binary_search:
test rsi, rsi
jz .Lnot_found
xor ecx, ecx
lea r8, [rsi - 1]
.Lloop:
cmp rcx, r8
jg .Lnot_found
mov r9, r8
sub r9, rcx
shr r9, 1
add r9, rcx
mov rax, qword ptr [rdi + r9*8]
cmp rax, rdx
je .Lfound
jl .Lgreater
lea r8, [r9 - 1]
jmp .Lloop
.Lgreater:
lea rcx, [r9 + 1]
jmp .Lloop
.Lfound:
mov rax, r9
ret
.Lnot_found:
mov rax, -1
ret
