export function l $gcd(l %a, l %b) {
@start
	%pa =l alloc8 8
	%pb =l alloc8 8
	storel %a, %pa
	storel %b, %pb
	jmp @loop
@loop
	%a0 =l loadl %pa
	%b0 =l loadl %pb
	%z =w ceql %b0, 0
	jnz %z, @done, @step
@step
	%r =l rem %a0, %b0
	storel %b0, %pa
	storel %r, %pb
	jmp @loop
@done
	%result =l loadl %pa
	ret %result
}
