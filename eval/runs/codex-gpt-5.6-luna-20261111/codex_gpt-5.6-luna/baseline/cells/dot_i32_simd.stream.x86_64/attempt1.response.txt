dot_i32_simd:
    pxor xmm0, xmm0
    xor rcx, rcx
    mov r8, rdx
    shr r8, 2
    test r8, r8
    jz .Lscalar_start
.Lvector:
    movdqu xmm1, xmmword ptr [rdi + rcx*4]
    movdqu xmm2, xmmword ptr [rsi + rcx*4]
    pshufd xmm3, xmm1, 0xee
    pshufd xmm4, xmm2, 0xee
    pmovsxdq xmm1, xmm1
    pmovsxdq xmm2, xmm2
    pmovsxdq xmm3, xmm3
    pmovsxdq xmm4, xmm4
    pmuldq xmm1, xmm2
    pmuldq xmm3, xmm4
    paddq xmm1, xmm3
    paddq xmm0, xmm1
    add rcx, 4
    dec r8
    jnz .Lvector
.Lscalar_start:
    movq rax, xmm0
    psrldq xmm0, 8
    movq r8, xmm0
    add rax, r8
.Lscalar:
    cmp rcx, rdx
    jae .Ldone
    movsxd r9, dword ptr [rdi + rcx*4]
    movsxd r8, dword ptr [rsi + rcx*4]
    imul r9, r8
    add rax, r9
    inc rcx
    jmp .Lscalar
.Ldone:
    ret
