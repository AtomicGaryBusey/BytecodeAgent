export function l $bubble_sort(l %a, l %n) {
@entry
%zero =l copy 0
%one =l copy 1
jmp @outer_init

@outer_init
%end0 =l sub %n, %one
jmp @outer_cond

@outer_cond
%end =l phi @outer_init %end0, @outer_dec %endnext
%active =w csgtl %end, %zero
jnz %active, @inner_init, @done

@inner_init
jmp @inner_cond

@inner_cond
%j =l phi @inner_init %zero, @inner_step %jnext
%inside =w csltl %j, %end
jnz %inside, @inner_body, @outer_dec

@inner_body
%jnext =l add %j, %one
%off =l mul %j, 8
%p =l add %a, %off
%j1 =l add %j, %one
%off1 =l mul %j1, 8
%p1 =l add %a, %off1
%x =l loadl %p
%y =l loadl %p1
%greater =w csgtl %x, %y
jnz %greater, @swap, @inner_step

@swap
storel %y, %p
storel %x, %p1
jmp @inner_step

@inner_step
jmp @inner_cond

@outer_dec
%endnext =l sub %end, %one
jmp @outer_cond

@done
ret %zero
}
