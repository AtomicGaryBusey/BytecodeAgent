export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1
	%t0 =l shr %p0, 32
	%t1 =l and %p1, %mask
	%t2 =l and %p2, %mask
	%t3 =l add %t0, %t1
	%t4 =l add %t3, %t2
	%carry =l shr %t4, 32
	%h1 =l shr %p1, 32
	%h2 =l shr %p2, 32
	%h3 =l add %p3, %h1
	%h4 =l add %h3, %h2
	%result =l add %h4, %carry
	ret %result
}
