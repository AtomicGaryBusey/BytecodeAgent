export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p0 =l mul %a0, %b0
	%p1 =l mul %a0, %b1
	%p2 =l mul %a1, %b0
	%p3 =l mul %a1, %b1
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, %mask
	%p2lo =l and %p2, %mask
	%t0 =l add %p0hi, %p1lo
	%t1 =l add %t0, %p2lo
	%carry =l shr %t1, 32
	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%t2 =l add %p3, %p1hi
	%t3 =l add %t2, %p2hi
	%result =l add %t3, %carry
	ret %result
}
