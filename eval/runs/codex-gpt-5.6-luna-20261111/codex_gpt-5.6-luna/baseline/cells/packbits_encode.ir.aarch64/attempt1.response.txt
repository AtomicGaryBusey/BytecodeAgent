export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	%si =l alloc8 8
	%so =l alloc8 8
	%sk =l alloc8 8
	%sr =l alloc8 8
	%sj =l alloc8 8
	%sbase =l alloc8 8
	storel 0, %si
	storel 0, %so
	jmp @outer

@outer
	%i =l loadl %si
	%has =w csltl %i, %n
	jnz %has, @run_init, @done

@run_init
	%k0 =l add %i, 1
	storel %k0, %sk
	storel 1, %sr
	%bp0 =l add %src, %i
	%b0 =w loadub %bp0
	storel %b0, %sbase
	jmp @run_loop

@run_loop
	%k1 =l loadl %sk
	%r1 =l loadl %sr
	%in1 =w csltl %k1, %n
	%diff1 =l sub %k1, %i
	%lt128_1 =w csltl %diff1, 128
	jnz %in1, @run_bound, @run_done

@run_bound
	jnz %lt128_1, @run_check, @run_done

@run_check
	%kp1 =l add %src, %k1
	%bk1 =w loadub %kp1
	%base1 =w loadw %sbase
	%eq1 =w ceql %bk1, %base1
	jnz %eq1, @run_inc, @run_done

@run_inc
	%kn1 =l add %k1, 1
	%rn1 =l add %r1, 1
	storel %kn1, %sk
	storel %rn1, %sr
	jmp @run_loop

@run_done
	%r2 =l loadl %sr
	%small =w csltl %r2, 3
	jnz %small, @literal_init, @repeat_check

@repeat_check
	%o1 =l loadl %so
	%end1 =l add %o1, 2
	%cap1 =l add %cap, 1
	%fits1 =w csltl %end1, %cap1
	jnz %fits1, @repeat_write, @fail

@repeat_write
	%h1 =l sub 257, %r2
	%hw1 =w castw %h1
	%op1 =l add %out, %o1
	storeb %hw1, %op1
	%bp1 =l add %op1, 1
	%b1 =w loadw %sbase
	storeb %b1, %bp1
	storel %end1, %so
	%k2 =l loadl %sk
	storel %k2, %si
	jmp @outer

@literal_init
	%i2 =l loadl %si
	storel %i2, %sj
	jmp @literal_scan

@literal_scan
	%j1 =l loadl %sj
	%in2 =w csltl %j1, %n
	%diff2 =l sub %j1, %i2
	%lt128_2 =w csltl %diff2, 128
	jnz %in2, @literal_bound, @literal_done

@literal_bound
	jnz %lt128_2, @literal_run_init, @literal_done

@literal_run_init
	storel %j1, %sk
	storel 1, %sr
	%bp2 =l add %src, %j1
	%b2 =w loadub %bp2
	storel %b2, %sbase
	jmp @literal_run_loop

@literal_run_loop
	%k3 =l loadl %sk
	%r3 =l loadl %sr
	%in3 =w csltl %k3, %n
	%diff3 =l sub %k3, %j1
	%lt128_3 =w csltl %diff3, 128
	jnz %in3, @literal_run_bound, @literal_run_done

@literal_run_bound
	jnz %lt128_3, @literal_run_check, @literal_run_done

@literal_run_check
	%kp3 =l add %src, %k3
	%bk3 =w loadub %kp3
	%base3 =w loadw %sbase
	%eq3 =w ceql %bk3, %base3
	jnz %eq3, @literal_run_inc, @literal_run_done

@literal_run_inc
	%kn3 =l add %k3, 1
	%rn3 =l add %r3, 1
	storel %kn3, %sk
	storel %rn3, %sr
	jmp @literal_run_loop

@literal_run_done
	%r4 =l loadl %sr
	%stop =w csltl %r4, 3
	jnz %stop, @literal_advance, @literal_done

@literal_advance
	%jn1 =l add %j1, 1
	storel %jn1, %sj
	jmp @literal_scan

@literal_done
	%j2 =l loadl %sj
	%len1 =l sub %j2, %i2
	%need1 =l add %len1, 1
	%o2 =l loadl %so
	%end2 =l add %o2, %need1
	%cap2 =l add %cap, 1
	%fits2 =w csltl %end2, %cap2
	jnz %fits2, @literal_write_header, @fail

@literal_write_header
	%lh1 =l sub %len1, 1
	%lhw1 =w castw %lh1
	%op2 =l add %out, %o2
	storeb %lhw1, %op2
	%oo1 =l add %o2, 1
	storel %oo1, %so
	storel %i2, %sk
	jmp @literal_copy

@literal_copy
	%cp1 =l loadl %sk
	%more1 =w csltl %cp1, %j2
	jnz %more1, @literal_copy_one, @literal_copy_done

@literal_copy_one
	%spp1 =l add %src, %cp1
	%cb1 =w loadub %spp1
	%co1 =l loadl %so
	%opp1 =l add %out, %co1
	storeb %cb1, %opp1
	%cp2 =l add %cp1, 1
	%co2 =l add %co1, 1
	storel %cp2, %sk
	storel %co2, %so
	jmp @literal_copy

@literal_copy_done
	storel %j2, %si
	jmp @outer

@done
	%res =l loadl %so
	ret %res

@fail
	ret -1
}
