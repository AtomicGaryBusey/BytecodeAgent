export function l $count_desc(l %a, l %n) {
@entry
%lt2 =w csltl %n, 2
jnz %lt2, @zero, @init

@zero
ret 0

@init
%nminus =l sub %n, 1
jmp @check

@check
%p =l phi @init %a, @body %pnext
%rem =l phi @init %nminus, @body %remnext
%cnt =l phi @init 0, @body %cntnext
%done =w ceql %rem, 0
jnz %done, @finish, @body

@body
%pnext =l add %p, 8
%x =l loadl %p
%y =l loadl %pnext
%isdesc =w csltl %y, %x
%inc =l extuw %isdesc
%cntnext =l add %cnt, %inc
%remnext =l sub %rem, 1
jmp @check

@finish
ret %cnt
}
