export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w cslel %n, 1
	jnz %small, @base, @prep

@base
	ret 0

@prep
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%pivot =l add %a, %poff
	%pk =l loadl %pivot
	%pvp =l add %pivot, 8
	%pv =l loadl %pvp
	jmp @loop

@loop
	%i =l phi @prep 0, @next %i, @swap %inext
	%j =l phi @prep 0, @next %jnext, @swap %jnext
	%done =w ceql %j, %nm1
	jnz %done, @finish, @scan

@scan
	%joff =l mul %j, 16
	%jptr =l add %a, %joff
	%jvptr =l add %jptr, 8
	%jk =l loadl %jptr
	%jv =l loadl %jvptr
	%kless =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vle =w cslel %jv, %pv
	%eqtake =w and %keq, %vle
	%take =w or %kless, %eqtake
	%jnext =l add %j, 1
	jnz %take, @swap, @next

@next
	jmp @loop

@swap
	%ioff =l mul %i, 16
	%iptr =l add %a, %ioff
	%ivptr =l add %iptr, 8
	%ik =l loadl %iptr
	%iv =l loadl %ivptr
	storel %ik, %jptr
	storel %iv, %jvptr
	storel %jk, %iptr
	storel %jv, %ivptr
	%inext =l add %i, 1
	jmp @loop

@finish
	%foff =l mul %i, 16
	%fptr =l add %a, %foff
	%fvptr =l add %fptr, 8
	%fk =l loadl %fptr
	%fv =l loadl %fvptr
	storel %pk, %fptr
	storel %pv, %fvptr
	storel %fk, %pivot
	storel %fv, %pvp
	%rnoff =l add %foff, 16
	%rptr =l add %a, %rnoff
	%rn =l sub %n, %i
	%rn1 =l sub %rn, 1
	%left =l call $qsort_pairs(l %a, l %i)
	%right =l call $qsort_pairs(l %rptr, l %rn1)
	ret 0
}
