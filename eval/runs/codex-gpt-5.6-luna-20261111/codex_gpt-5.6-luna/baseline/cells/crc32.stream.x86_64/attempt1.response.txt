crc32:
    mov eax, 0xffffffff
    xor ecx, ecx
    mov r9d, 0xedb88320
.Lbyte:
    cmp rcx, rsi
    jae .Ldone
    movzx edx, byte ptr [rdi + rcx]
    xor eax, edx
    mov r8d, 8
.Lbit:
    test eax, 1
    jz .Lshift
    shr eax, 1
    xor eax, r9d
    jmp .Lnext
.Lshift:
    shr eax, 1
.Lnext:
    dec r8d
    jnz .Lbit
    inc rcx
    jmp .Lbyte
.Ldone:
    xor eax, 0xffffffff
    ret
