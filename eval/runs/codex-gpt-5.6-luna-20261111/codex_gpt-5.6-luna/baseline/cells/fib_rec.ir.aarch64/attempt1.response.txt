export function l $fib_rec(l %n) {
@start
	%saved =l alloc8 8
	%first =l alloc8 8
	storel %n, %saved
	%z =w ceql %n, 0
	jnz %z, @zero, @check_one

@check_one
	%o =w ceql %n, 1
	jnz %o, @one, @recurse

@recurse
	%orig1 =l loadl %saved
	%nm1 =l sub %orig1, 1
	%r1 =l call $fib_rec(l %nm1)
	storel %r1, %first
	%orig2 =l loadl %saved
	%nm2 =l sub %orig2, 2
	%r2 =l call $fib_rec(l %nm2)
	%r1saved =l loadl %first
	%sum =l add %r1saved, %r2
	ret %sum

@zero
	ret 0

@one
	ret 1
}
