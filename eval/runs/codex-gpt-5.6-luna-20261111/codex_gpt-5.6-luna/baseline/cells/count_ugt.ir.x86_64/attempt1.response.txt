export function l $count_ugt(l %a, l %n, l %t) {
@entry
	%zero =l copy 0
	jmp @check
@check
	%i =l phi @entry %zero, @same %i_same, @up %i_up
	%c =l phi @entry %zero, @same %c_same, @up %c_up
	%done =w ceql %i, %n
	jnz %done, @ret, @body
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	jnz %gt, @up, @same
@same
	%i_same =l add %i, 1
	%c_same =l copy %c
	jmp @check
@up
	%i_up =l add %i, 1
	%c_up =l add %c, 1
	jmp @check
@ret
	ret %c
}
