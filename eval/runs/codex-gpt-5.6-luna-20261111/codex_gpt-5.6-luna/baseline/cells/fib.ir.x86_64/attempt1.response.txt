export function l $fib(l %n) {
@start
	%a =l alloc8 8
	%b =l alloc8 8
	%i =l alloc8 8
	storel 0, %a
	storel 1, %b
	storel 0, %i
	jmp @loop
@loop
	%iv =l loadl %i
	%cond =w csltl %iv, %n
	jnz %cond, @body, @done
@body
	%av =l loadl %a
	%bv =l loadl %b
	%sum =l add %av, %bv
	storel %bv, %a
	storel %sum, %b
	%next =l add %iv, 1
	storel %next, %i
	jmp @loop
@done
	%result =l loadl %a
	ret %result
}
