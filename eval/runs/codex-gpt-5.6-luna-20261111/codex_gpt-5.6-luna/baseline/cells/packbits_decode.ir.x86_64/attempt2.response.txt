export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%vp =l alloc8 8
	%vo =l alloc8 8
	%vi =l alloc8 8
	storel 0, %vp
	storel 0, %vo
	jmp @packet

@packet
	%p0 =l loadl %vp
	%has =w csltl %p0, %n
	jnz %has, @header, @done

@header
	%hp =l add %src, %p0
	%h =w loadsb %hp
	%p1 =l add %p0, 1
	storel %p1, %vp
	%noop =w ceqw %h, -128
	jnz %noop, @packet, @classify

@classify
	%literal =w csgew %h, 0
	jnz %literal, @literal, @repeat

@literal
	%hl =l extsw %h
	%cnt =l add %hl, 1
	%rem =l sub %n, %p1
	%srcok =w csgtl %cnt, %rem
	jnz %srcok, @source_error, @literal_output

@literal_output
	%op =l loadl %vo
	%avail =l sub %cap, %op
	%too =w csgtl %cnt, %avail
	jnz %too, @literal_partial, @literal_full

@literal_full
	storel 0, %vi
	jmp @literal_copy_full

@literal_copy_full
	%ii =l loadl %vi
	%more =w csltl %ii, %cnt
	jnz %more, @literal_copy_full_do, @literal_full_done

@literal_copy_full_do
	%si =l add %p1, %ii
	%sp =l add %src, %si
	%c =w loadub %sp
	%di =l add %op, %ii
	%dp =l add %out, %di
	storeb %c, %dp
	%ni =l add %ii, 1
	storel %ni, %vi
	jmp @literal_copy_full

@literal_full_done
	%np =l add %p1, %cnt
	%no =l add %op, %cnt
	storel %np, %vp
	storel %no, %vo
	jmp @packet

@literal_partial
	storel 0, %vi
	jmp @literal_copy_partial

@literal_copy_partial
	%ii2 =l loadl %vi
	%more2 =w csltl %ii2, %avail
	jnz %more2, @literal_copy_partial_do, @overflow

@literal_copy_partial_do
	%si2 =l add %p1, %ii2
	%sp2 =l add %src, %si2
	%c2 =w loadub %sp2
	%di2 =l add %op, %ii2
	%dp2 =l add %out, %di2
	storeb %c2, %dp2
	%ni2 =l add %ii2, 1
	storel %ni2, %vi
	jmp @literal_copy_partial

@repeat
	%rp =l loadl %vp
	%rok =w csltl %rp, %n
	jnz %rok, @repeat_value, @source_error

@repeat_value
	%raddr =l add %src, %rp
	%rv =w loadub %raddr
	%rp1 =l add %rp, 1
	storel %rp1, %vp
	%hl2 =l extsw %h
	%rcnt =l sub 1, %hl2
	%rop =l loadl %vo
	%ravail =l sub %cap, %rop
	%rtoo =w csgtl %rcnt, %ravail
	jnz %rtoo, @repeat_partial, @repeat_full

@repeat_full
	storel 0, %vi
	jmp @repeat_write_full

@repeat_write_full
	%ri =l loadl %vi
	%rmore =w csltl %ri, %rcnt
	jnz %rmore, @repeat_write_full_do, @repeat_full_done

@repeat_write_full_do
	%rdi =l add %rop, %ri
	%rdp =l add %out, %rdi
	storeb %rv, %rdp
	%rni =l add %ri, 1
	storel %rni, %vi
	jmp @repeat_write_full

@repeat_full_done
	%rno =l add %rop, %rcnt
	storel %rno, %vo
	jmp @packet

@repeat_partial
	storel 0, %vi
	jmp @repeat_write_partial

@repeat_write_partial
	%rj =l loadl %vi
	%rmore2 =w csltl %rj, %ravail
	jnz %rmore2, @repeat_write_partial_do, @overflow

@repeat_write_partial_do
	%rdj =l add %rop, %rj
	%rdp2 =l add %out, %rdj
	storeb %rv, %rdp2
	%rnj =l add %rj, 1
	storel %rnj, %vi
	jmp @repeat_write_partial

@done
	%result =l loadl %vo
	ret %result

@source_error
	ret -2

@overflow
	ret -1
}
