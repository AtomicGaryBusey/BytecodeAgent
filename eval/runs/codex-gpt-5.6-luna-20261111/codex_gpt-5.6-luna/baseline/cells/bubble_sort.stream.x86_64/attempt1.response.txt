bubble_sort:
    cmp rsi, 1
    jle .Ldone
    mov rcx, rsi
    dec rcx
.Louter:
    xor edx, edx
.Linner:
    mov rax, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp r9, rax
    jge .Lnoswap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
    inc rdx
    cmp rdx, rcx
    jl .Linner
    dec rcx
    jg .Louter
.Ldone:
    xor eax, eax
    ret
