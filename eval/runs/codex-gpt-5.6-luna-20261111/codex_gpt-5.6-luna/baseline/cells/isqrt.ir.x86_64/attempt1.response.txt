export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @raise %mid, @lower %lo
	%hi =l phi @start 4294967296, @raise %hi, @lower %mid
	%diff =l sub %hi, %lo
	%more =w cugt %diff, 1
	jnz %more, @search, @done
@search
	%sum =l add %lo, %hi
	%mid =l udiv %sum, 2
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @raise, @lower
@raise
	jmp @loop
@lower
	jmp @loop
@done
	ret %lo
}
