export function l $max_byte(l %p, l %n) {
@entry
    %zero =l copy 0
    jmp @loop
@loop
    %i =l phi @entry %zero, @cont %next
    %max =l phi @entry %zero, @cont %maxnext
    %inrange =w csltl %i, %n
    jnz %inrange, @read, @done
@read
    %addr =l add %p, %i
    %b =l loadub %addr
    %greater =w csgtl %b, %max
    jnz %greater, @newmax, @same
@newmax
    jmp @cont
@same
    jmp @cont
@cont
    %maxnext =l phi @newmax %b, @same %max
    %next =l add %i, 1
    jmp @loop
@done
    ret %max
}
