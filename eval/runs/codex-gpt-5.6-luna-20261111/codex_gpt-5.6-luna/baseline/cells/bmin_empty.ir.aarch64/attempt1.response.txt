export function l $bmin_empty(l %a, l %n) {
@entry
	%zero =w ceql %n, 0
	%initial =l copy 9223372036854775807
	jnz %zero, @empty, @loop

@empty
	ret 9223372036854775807

@loop
	%min =l phi @entry %initial, @advance %nextmin
	%ptr =l phi @entry %a, @advance %nextptr
	%cnt =l phi @entry %n, @advance %nextcnt
	%value =l loadl %ptr
	%less =w csltl %value, %min
	jnz %less, @setmin, @keepmin

@setmin
	%candidate =l copy %value
	jmp @advance

@keepmin
	%candidate2 =l copy %min
	jmp @advance

@advance
	%nextmin =l phi @setmin %candidate, @keepmin %candidate2
	%nextptr =l add %ptr, 8
	%nextcnt =l sub %cnt, 1
	%done =w ceql %nextcnt, 0
	jnz %done, @finish, @loop

@finish
	ret %nextmin
}
