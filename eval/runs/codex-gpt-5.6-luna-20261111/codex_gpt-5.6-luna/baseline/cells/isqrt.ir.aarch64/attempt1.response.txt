export function l $isqrt(l %n) {
@entry
	%lo =l alloc8 8
	%hi =l alloc8 8
	storel 0, %lo
	storel 4294967296, %hi
	jmp @loop

@loop
	%lov =l loadl %lo
	%hiv =l loadl %hi
	%diff =l sub %hiv, %lov
	%more =w csgtl %diff, 1
	jnz %more, @search, @done

@search
	%sum =l add %lov, %hiv
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%ok =w culel %sq, %n
	jnz %ok, @lower, @upper

@lower
	storel %mid, %lo
	jmp @loop

@upper
	storel %mid, %hi
	jmp @loop

@done
	%result =l loadl %lo
	ret %result
}
