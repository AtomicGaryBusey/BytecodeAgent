export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %bz =w ceql %b, 0
    jnz %bz, @divzero, @check

@check
    %amin =w ceql %a, -9223372036854775808
    jnz %amin, @check_overflow, @normal

@check_overflow
    %bminus =w ceql %b, -1
    jnz %bminus, @overflow, @normal

@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q

@divzero
    storel %a, %rem
    ret 0

@overflow
    storel 0, %rem
    ret -9223372036854775808
}
