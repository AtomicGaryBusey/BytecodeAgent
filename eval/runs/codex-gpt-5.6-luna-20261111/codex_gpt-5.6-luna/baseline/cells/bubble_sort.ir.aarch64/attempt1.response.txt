export function l $bubble_sort(l %a, l %n) {
@start
    %i =l alloc8 8
    %j =l alloc8 8
    storel 0, %i
    jmp @outer

@outer
    %iv =l loadl %i
    %more =w csltl %iv, %n
    jnz %more, @init_inner, @done

@init_inner
    storel 0, %j
    jmp @inner

@inner
    %iv2 =l loadl %i
    %jv =l loadl %j
    %rem =l sub %n, %iv2
    %jp =l add %jv, 1
    %has =w csltl %jp, %rem
    jnz %has, @compare, @next_outer

@compare
    %off =l mul %jv, 8
    %p =l add %a, %off
    %pnext =l add %p, 8
    %left =l loadl %p
    %right =l loadl %pnext
    %swap =w csgtl %left, %right
    jnz %swap, @do_swap, @advance

@do_swap
    storel %right, %p
    storel %left, %pnext
    jmp @advance

@advance
    %jn =l add %jv, 1
    storel %jn, %j
    jmp @inner

@next_outer
    %in =l add %iv2, 1
    storel %in, %i
    jmp @outer

@done
    ret 0
}
