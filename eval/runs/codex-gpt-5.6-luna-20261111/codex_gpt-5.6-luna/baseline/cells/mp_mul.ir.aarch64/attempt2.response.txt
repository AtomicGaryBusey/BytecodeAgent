export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%n =l add %na, %nb
	jmp @zloop

@zloop
	%zi =l phi @entry 0, @zbody %zin
	%zok =w csltl %zi, %n
	jnz %zok, @zbody, @zero_done

@zbody
	%zoff =l mul %zi, 8
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zin =l add %zi, 1
	jmp @zloop

@zero_done
	jmp @outer

@outer
	%ii =l phi @zero_done 0, @outer_done %inext
	%iok =w csltl %ii, %na
	jnz %iok, @outer_body, @scan

@outer_body
	jmp @inner

@inner
	%jj =l phi @outer_body 0, @prod_done %jnext
	%jok =w csltl %jj, %nb
	jnz %jok, @prod, @outer_done

@prod
	%ioff =l mul %ii, 8
	%joff =l mul %jj, 8
	%aptr =l add %a, %ioff
	%bptr =l add %b, %joff
	%av =l loadl %aptr
	%bv =l loadl %bptr

	%amask =l and %av, 4294967295
	%bmask =l and %bv, 4294967295
	%ahi =l shr %av, 32
	%bhi =l shr %bv, 32

	%p0 =l mul %amask, %bmask
	%p1 =l mul %amask, %bhi
	%p2 =l mul %ahi, %bmask
	%p3 =l mul %ahi, %bhi

	%p1lo =l shl %p1, 32
	%t1 =l add %p0, %p1lo
	%c1 =w cultl %t1, %p0
	%p2lo =l shl %p2, 32
	%t2 =l add %t1, %p2lo
	%c2 =w cultl %t2, %t1

	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%h0 =l add %p3, %p1hi
	%h1 =l add %h0, %p2hi
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%h2 =l add %h1, %c1l
	%high =l add %h2, %c2l

	%idx =l add %ii, %jj
	%roff =l mul %idx, 8
	%rptr =l add %r, %roff
	%old =l loadl %rptr
	%sum =l add %old, %t2
	%carry0 =w cultl %sum, %old
	storel %sum, %rptr

	%carry0l =l extuw %carry0
	%hsum =l add %high, %carry0l
	%carry1 =w cultl %hsum, %high
	%next =l add %idx, 1
	%nextoff =l mul %next, 8
	%nextptr =l add %r, %nextoff
	%nextold =l loadl %nextptr
	%nextsum =l add %nextold, %hsum
	%carry2 =w cultl %nextsum, %nextold
	storel %nextsum, %nextptr
	%carry =w or %carry1, %carry2
	%propidx =l add %next, 1
	jmp @prop

@prop
	%pk =l phi @prod %propidx, @prop_body %pnext
	%pc =w phi @prod %carry, @prop_body %pcnext
	%pkok =w csltl %pk, %n
	%pdo =w and %pkok, %pc
	jnz %pdo, @prop_body, @prod_done

@prop_body
	%poff =l mul %pk, 8
	%pptr =l add %r, %poff
	%pold =l loadl %pptr
	%psum =l add %pold, 1
	%pcnext =w cultl %psum, %pold
	storel %psum, %pptr
	%pnext =l add %pk, 1
	jmp @prop

@prod_done
	%jnext =l add %jj, 1
	jmp @inner

@outer_done
	%inext =l add %ii, 1
	jmp @outer

@scan
	jmp @scan_loop

@scan_loop
	%sk =l phi @scan %n, @scan_zero %sknext
	%sz =w ceql %sk, 0
	jnz %sz, @scan_ret, @scan_check

@scan_check
	%skm =l sub %sk, 1
	%sko =l mul %skm, 8
	%skp =l add %r, %sko
	%sv =l loadl %skp
	%svz =w ceql %sv, 0
	jnz %svz, @scan_zero, @scan_ret

@scan_zero
	%sknext =l sub %sk, 1
	jmp @scan_loop

@scan_ret
	ret %sk
}
