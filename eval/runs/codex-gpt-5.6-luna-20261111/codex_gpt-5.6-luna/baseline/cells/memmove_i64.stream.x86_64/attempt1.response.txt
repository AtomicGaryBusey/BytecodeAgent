memmove_i64:
cld
test rdx, rdx
jz .Ldone
cmp rdi, rsi
ja .Lbackward
xor rcx, rcx
.Lforward_loop:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
inc rcx
cmp rcx, rdx
jne .Lforward_loop
jmp .Ldone
.Lbackward:
mov rcx, rdx
.Lbackward_loop:
dec rcx
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
test rcx, rcx
jne .Lbackward_loop
.Ldone:
xor eax, eax
ret
