export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	jmp @loop

@loop
	%i =l phi @entry 0, @body %nexti
	%count =l phi @entry 0, @body %nextcount
	%p =l phi @entry %src, @body %nextp
	%cond =w csltl %i, %n
	jnz %cond, @body, @done

@body
	%x =l loadl %p
	%y =l call %f(l %x)
	storel %y, %p
	%less =w csltl %y, %x
	%inc =l extuw %less
	%nextcount =l add %count, %inc
	%nexti =l add %i, 1
	%nextp =l add %p, 8
	jmp @loop

@done
	ret %count
}
