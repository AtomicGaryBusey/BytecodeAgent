export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	jmp @loop

@loop
	%i =l phi @entry 0, @body %nexti
	%count =l phi @entry 0, @body %nextcount
	%sp =l phi @entry %src, @body %nextsp
	%dp =l phi @entry %dst, @body %nextdp
	%cond =w csltl %i, %n
	jnz %cond, @body, @done

@body
	%x =l loadl %sp
	%y =l call %f(l %x)
	storel %y, %dp
	%less =w csltl %y, %x
	%inc =l extuw %less
	%nextcount =l add %count, %inc
	%nexti =l add %i, 1
	%nextsp =l add %sp, 8
	%nextdp =l add %dp, 8
	jmp @loop

@done
	ret %count
}
