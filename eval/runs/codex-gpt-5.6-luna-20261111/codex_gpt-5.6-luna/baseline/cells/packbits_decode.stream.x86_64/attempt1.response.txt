packbits_decode:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    xor ebx, ebx
    xor ebp, ebp
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
.Lloop:
    cmp rbx, r13
    jae .Lsuccess
    movsx r8, byte ptr [r12 + rbx]
    inc rbx
    cmp r8, -128
    je .Lloop
    js .Lrepeat
    inc r8
    mov r9, r13
    sub r9, rbx
    cmp r9, r8
    jb .Lerror2
    mov r10, r15
    sub r10, rbp
    cmp r10, r8
    jb .Lliteral_partial
    mov r11, r8
.Lliteral_copy:
    test r11, r11
    je .Lloop
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + rbp], al
    inc rbx
    inc rbp
    dec r11
    jmp .Lliteral_copy
.Lliteral_partial:
    mov r11, r10
.Lliteral_partial_copy:
    test r11, r11
    je .Lerror1
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + rbp], al
    inc rbx
    inc rbp
    dec r11
    jmp .Lliteral_partial_copy
.Lrepeat:
    cmp rbx, r13
    jae .Lerror2
    mov al, byte ptr [r12 + rbx]
    inc rbx
    neg r8
    inc r8
    mov r10, r15
    sub r10, rbp
    cmp r10, r8
    jb .Lrepeat_partial
    mov r11, r8
.Lrepeat_copy:
    test r11, r11
    je .Lloop
    mov byte ptr [r14 + rbp], al
    inc rbp
    dec r11
    jmp .Lrepeat_copy
.Lrepeat_partial:
    mov r11, r10
.Lrepeat_partial_copy:
    test r11, r11
    je .Lerror1
    mov byte ptr [r14 + rbp], al
    inc rbp
    dec r11
    jmp .Lrepeat_partial_copy
.Lsuccess:
    mov rax, rbp
    jmp .Lreturn
.Lerror1:
    mov eax, -1
    jmp .Lreturn
.Lerror2:
    mov eax, -2
.Lreturn:
    cld
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
