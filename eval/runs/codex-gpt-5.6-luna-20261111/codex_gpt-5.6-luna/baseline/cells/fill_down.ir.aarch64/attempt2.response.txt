export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop

@loop
    %i =l phi @entry 0, @body %nexti
    %v =l phi @entry %start, @body %nextv
    %cond =w csltl %i, %n
    jnz %cond, @body, @done

@body
    %off =l shl %i, 3
    %p =l add %a, %off
    storel %v, %p
    %nexti =l add %i, 1
    %nextv =l sub %v, 1
    jmp @loop

@done
    ret %n
}
