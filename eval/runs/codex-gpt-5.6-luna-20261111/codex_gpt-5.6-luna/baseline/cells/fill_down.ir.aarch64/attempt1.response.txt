export function l $fill_down(l %a, l %n, l %start) {
@entry
    jmp @loop

@loop
    %i = phi @entry 0, @body %nexti
    %v = phi @entry %start, @body %nextv
    %cond = csltl %i, %n
    jnz %cond, @body, @done

@body
    %off = shl %i, 3
    %p = add %a, %off
    storel %v, %p
    %nexti = add %i, 1
    %nextv = sub %v, 1
    jmp @loop

@done
    ret %n
}
