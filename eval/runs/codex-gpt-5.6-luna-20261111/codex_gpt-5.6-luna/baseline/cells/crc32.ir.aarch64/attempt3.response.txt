export function l $crc32(l %p, l %n) {
@entry
	%crc0 =w copy -1
	%i0 =l copy 0
	jmp @byte

@byte
	%i =l phi @entry %i0, @next %inext
	%c =w phi @entry %crc0, @next %cout
	%done =w ceql %i, %n
	jnz %done, @finish, @load

@load
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %c, %b
	%bit0 =w copy 0
	jmp @bits

@bits
	%bit =w phi @load %bit0, @shift %bitnext, @mix %bitnext
	%cc =w phi @load %cx, @shift %cshift, @mix %cmix
	%bdone =w ceqw %bit, 8
	jnz %bdone, @next, @test

@test
	%lsb =w and %cc, 1
	%zero =w ceqw %lsb, 0
	%bitnext =w add %bit, 1
	jnz %zero, @shift, @mix

@shift
	%cshift =w shr %cc, 1
	jmp @bits

@mix
	%cshift2 =w shr %cc, 1
	%cmix =w xor %cshift2, -306674912
	jmp @bits

@next
	%inext =l add %i, 1
	%cout =w copy %cc
	jmp @byte

@finish
	%res =w xor %c, -1
	%out =l extuw %res
	ret %out
}
