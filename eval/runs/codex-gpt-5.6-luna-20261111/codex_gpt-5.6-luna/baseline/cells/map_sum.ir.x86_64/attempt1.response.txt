export function l $map_sum(l %a, l %n, l %f) {
@start
	%zero =l copy 0
	jmp @loop
@loop
	%p =l phi @start %a, @body %nextp
	%i =l phi @start %n, @body %nexti
	%sum =l phi @start %zero, @body %nextsum
	%done =w ceql %i, 0
	jnz %done, @return, @body
@body
	%x =l loadl %p
	%y =l call %f(l %x)
	%nextsum =l add %sum, %y
	%nextp =l add %p, 8
	%nexti =l sub %i, 1
	jmp @loop
@return
	ret %sum
}
