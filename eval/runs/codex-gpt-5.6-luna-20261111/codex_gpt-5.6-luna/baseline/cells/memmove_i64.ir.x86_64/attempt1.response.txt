export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%zero =w ceql %n, 0
	jnz %zero, @done, @direction

@direction
	%forward =w csltl %dst, %src
	jnz %forward, @floop, @binit

@floop
	%fd =l phi @direction %dst, @floop %fdn
	%fs =l phi @direction %src, @floop %fsn
	%fr =l phi @direction %n, @floop %frn
	%fc =w csgtl %fr, 0
	jnz %fc, @fbody, @done

@fbody
	%fv =l loadl %fs
	storel %fv, %fd
	%fdn =l add %fd, 8
	%fsn =l add %fs, 8
	%frn =l sub %fr, 1
	jmp @floop

@binit
	%bm1 =l sub %n, 1
	%boff =l mul %bm1, 8
	%bds =l add %dst, %boff
	%bss =l add %src, %boff
	jmp @bloop

@bloop
	%bd =l phi @binit %bds, @bloop %bdn
	%bs =l phi @binit %bss, @bloop %bsn
	%br =l phi @binit %n, @bloop %brn
	%bc =w csgtl %br, 0
	jnz %bc, @bbody, @done

@bbody
	%bv =l loadl %bs
	storel %bv, %bd
	%bdn =l sub %bd, 8
	%bsn =l sub %bs, 8
	%brn =l sub %br, 1
	jmp @bloop

@done
	ret 0
}
