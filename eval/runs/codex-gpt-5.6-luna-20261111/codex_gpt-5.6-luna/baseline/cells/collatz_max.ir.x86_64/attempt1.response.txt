export function l $collatz_max(l %n) {
@start
	%ip =l alloc8 8
	%mp =l alloc8 8
	%xp =l alloc8 8
	%sp =l alloc8 8
	storel 1, %ip
	storel 0, %mp
@outer
	%i =l loadl %ip
	%done =w csgtl %i, %n
	jnz %done, @finish, @begin
@begin
	storel %i, %xp
	storel 0, %sp
@collatz
	%x =l loadl %xp
	%isone =w ceql %x, 1
	jnz %isone, @check, @step
@step
	%rem =l rem %x, 2
	%even =w ceql %rem, 0
	jnz %even, @halve, @triple
@halve
	%nx =l div %x, 2
	storel %nx, %xp
	%s =l loadl %sp
	%snew =l add %s, 1
	storel %snew, %sp
	jmp @collatz
@triple
	%t =l mul %x, 3
	%nx2 =l add %t, 1
	storel %nx2, %xp
	%s2 =l loadl %sp
	%snew2 =l add %s2, 1
	storel %snew2, %sp
	jmp @collatz
@check
	%s3 =l loadl %sp
	%m =l loadl %mp
	%greater =w csgtl %s3, %m
	jnz %greater, @setmax, @next
@setmax
	storel %s3, %mp
	jmp @next
@next
	%i2 =l add %i, 1
	storel %i2, %ip
	jmp @outer
@finish
	%result =l loadl %mp
	ret %result
}
