export function l $stats_f(l %a, l %n, l %f, l %out) {
@entry
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%i0 =l copy 1
	%odd0 =l and %y0, 1
	jmp @loop

@loop
	%i =l phi @entry %i0, @max_done %inext
	%min =l phi @entry %y0, @max_done %mn
	%max =l phi @entry %y0, @max_done %mx
	%sum =l phi @entry %y0, @max_done %snext
	%count =l phi @entry %odd0, @max_done %cnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @newmin, @keepmin

@newmin
	jmp @min_done

@keepmin
	jmp @min_done

@min_done
	%mn =l phi @newmin %y, @keepmin %min
	%gt =w csgtl %y, %max
	jnz %gt, @newmax, @keepmax

@newmax
	jmp @max_done

@keepmax
	jmp @max_done

@max_done
	%mx =l phi @newmax %y, @keepmax %max
	%snext =l add %sum, %y
	%odd =l and %y, 1
	%cnext =l add %count, %odd
	%inext =l add %i, 1
	jmp @loop

@done
	storel %min, %out
	%out1 =l add %out, 8
	storel %max, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %count
}
