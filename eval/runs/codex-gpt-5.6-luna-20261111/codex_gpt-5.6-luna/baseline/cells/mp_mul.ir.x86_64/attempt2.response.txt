export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%total =l add %na, %nb
	jmp @init

@init
	%ii =l phi @entry 0, @init %iin
	%ic =w csltl %ii, %total
	jnz %ic, @initbody, @itest

@initbody
	%ioff =l mul %ii, 8
	%ip =l add %r, %ioff
	storel 0, %ip
	%iin =l add %ii, 1
	jmp @init

@itest
	%i =l phi @init %ii, @inext %inextval
	%icond =w csltl %i, %na
	jnz %icond, @jtest, @scaninit

@jtest
	%j =l phi @itest 0, @jbody %jnextval
	%carry =l phi @itest 0, @jbody %carryout
	%jcond =w csltl %j, %nb
	jnz %jcond, @jbody, @iend

@jbody
	%io =l mul %i, 8
	%jp =l mul %j, 8
	%ap0 =l add %a, %io
	%bp0 =l add %b, %jp
	%av =l loadl %ap0
	%bv =l loadl %bp0

	%mask =l copy 4294967295
	%a0 =l and %av, %mask
	%a1 =l shr %av, 32
	%b0 =l and %bv, %mask
	%b1 =l shr %bv, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%p01l =l shl %p01, 32
	%t0 =l add %p00, %p01l
	%ct0 =w cultl %t0, %p00
	%p10l =l shl %p10, 32
	%t1 =l add %t0, %p10l
	%ct1 =w cultl %t1, %t0

	%p01h =l shr %p01, 32
	%p10h =l shr %p10, 32
	%h0 =l add %p11, %p01h
	%h1 =l add %h0, %p10h
	%ct0l =l extuw %ct0
	%ct1l =l extuw %ct1
	%h2 =l add %h1, %ct0l
	%hi =l add %h2, %ct1l

	%k =l add %i, %j
	%ko =l mul %k, 8
	%kp =l add %r, %ko
	%rv =l loadl %kp
	%s0 =l add %rv, %t1
	%c0 =w cultl %s0, %rv
	%s1 =l add %s0, %carry
	%c1 =w cultl %s1, %s0
	storel %s1, %kp

	%c0l =l extuw %c0
	%c1l =l extuw %c1
	%hc0 =l add %hi, %c0l
	%carryout =l add %hc0, %c1l
	%jnextval =l add %j, 1
	jmp @jtest

@iend
	%endidx =l add %i, %nb
	%endoff =l mul %endidx, 8
	%endp =l add %r, %endoff
	%endv =l loadl %endp
	%ends =l add %endv, %carry
	%endc =w cultl %ends, %endv
	storel %ends, %endp
	%pidx0 =l add %endidx, 1
	jmp @pcheck

@pcheck
	%pidx =l phi @iend %pidx0, @pbody %pidxnext
	%pcarry =w phi @iend %endc, @pbody %pcarrynext
	%pvalid =w csltl %pidx, %total
	jnz %pvalid, @pnonzero, @inext

@pnonzero
	%pz =w ceql %pcarry, 0
	jnz %pz, @inext, @pbody

@pbody
	%poff =l mul %pidx, 8
	%pp =l add %r, %poff
	%pv =l loadl %pp
	%ps =l add %pv, %pcarry
	%pcarrynext =w cultl %ps, %pv
	storel %ps, %pp
	%pidxnext =l add %pidx, 1
	jmp @pcheck

@inext
	%inextval =l add %i, 1
	jmp @itest

@scaninit
	jmp @scan

@scan
	%sk =l phi @scaninit %total, @scanloop %sknext
	%sgt =w csgtl %sk, 0
	jnz %sgt, @scanbody, @zero

@scanbody
	%skm =l sub %sk, 1
	%sko =l mul %skm, 8
	%skp =l add %r, %sko
	%skv =l loadl %skp
	%skz =w ceql %skv, 0
	jnz %skz, @scanloop, @return

@scanloop
	%sknext =l sub %sk, 1
	jmp @scan

@return
	ret %sk

@zero
	ret 0
}
