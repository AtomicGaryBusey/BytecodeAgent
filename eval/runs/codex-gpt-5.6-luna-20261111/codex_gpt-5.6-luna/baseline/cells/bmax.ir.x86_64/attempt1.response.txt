export function l $bmax(l %a, l %n) {
@entry
	%first =l loadl %a
	%p0 =l add %a, 8
	%r0 =l sub %n, 1
	jmp @loop

@loop
	%m =l phi @entry %first, @step %newmax
	%p =l phi @entry %p0, @step %pnext
	%r =l phi @entry %r0, @step %rnext
	%done =w ceql %r, 0
	jnz %done, @ret, @scan

@scan
	%v =l loadl %p
	%gt =w csgtl %v, %m
	%pnext =l add %p, 8
	%rnext =l sub %r, 1
	jnz %gt, @greater, @keep

@greater
	jmp @step

@keep
	jmp @step

@step
	%newmax =l phi @greater %v, @keep %m
	jmp @loop

@ret
	ret %m
}
