export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%endoff =l shl %cap, 3
	jmp @clear
@clear
	%coff =l phi @start 0, @clear_store %coff_next
	%cc =w csltl %coff, %endoff
	jnz %cc, @clear_store, @run
@clear_store
	%caddr =l add %stack, %coff
	storel 0, %caddr
	%coff_next =l add %coff, 8
	jmp @clear

@run
	jmp @dispatch

@dispatch
	%pc =l phi @run 0, @push %push_pc, @add %add_pc, @sub %sub_pc, @mul %mul_pc, @dup %dup_pc, @swap %swap_pc, @drop %drop_pc, @over %over_pc, @lt %lt_pc, @jz_zero %jz_zero_pc, @jz_nonzero %jz_nonzero_pc, @jmp %jmp_pc, @other %other_pc
	%d =l phi @run 0, @push %push_d, @add %add_d, @sub %sub_d, @mul %mul_d, @dup %dup_d, @swap %swap_d, @drop %drop_d, @over %over_d, @lt %lt_d, @jz_zero %jz_d, @jz_nonzero %jz_d, @jmp %jmp_d, @other %other_d
	%at_end =w ceql %pc, %len
	jnz %at_end, @stop_check, @fetch

@fetch
	%pcoff =l shl %pc, 3
	%pcaddr =l add %code, %pcoff
	%op =l loadl %pcaddr
	jmp @test0

@test0
	%t0 =w ceql %op, 0
	jnz %t0, @halt, @test1
@test1
	%t1 =w ceql %op, 1
	jnz %t1, @push, @test2
@test2
	%t2 =w ceql %op, 2
	jnz %t2, @add, @test3
@test3
	%t3 =w ceql %op, 3
	jnz %t3, @sub, @test4
@test4
	%t4 =w ceql %op, 4
	jnz %t4, @mul, @test5
@test5
	%t5 =w ceql %op, 5
	jnz %t5, @dup, @test6
@test6
	%t6 =w ceql %op, 6
	jnz %t6, @swap, @test7
@test7
	%t7 =w ceql %op, 7
	jnz %t7, @drop, @test8
@test8
	%t8 =w ceql %op, 8
	jnz %t8, @over, @test9
@test9
	%t9 =w ceql %op, 9
	jnz %t9, @lt, @test10
@test10
	%t10 =w ceql %op, 10
	jnz %t10, @jz, @test11
@test11
	%t11 =w ceql %op, 11
	jnz %t11, @jmp, @other

@halt
	jmp @stop_check

@push
	%push_one =l add %pc, 1
	%push_off =l shl %push_one, 3
	%push_addr =l add %code, %push_off
	%push_val =l loadl %push_addr
	%push_stack_off =l shl %d, 3
	%push_stack_addr =l add %stack, %push_stack_off
	storel %push_val, %push_stack_addr
	%push_d =l add %d, 1
	%push_pc =l add %pc, 2
	jmp @dispatch

@add
	%add_top =l sub %d, 1
	%add_boff =l shl %add_top, 3
	%add_baddr =l add %stack, %add_boff
	%add_b =l loadl %add_baddr
	%add_aidx =l sub %d, 2
	%add_aoff =l shl %add_aidx, 3
	%add_aaddr =l add %stack, %add_aoff
	%add_a =l loadl %add_aaddr
	%add_r =l add %add_a, %add_b
	storel %add_r, %add_aaddr
	%add_d =l sub %d, 1
	%add_pc =l add %pc, 1
	jmp @dispatch

@sub
	%sub_top =l sub %d, 1
	%sub_boff =l shl %sub_top, 3
	%sub_baddr =l add %stack, %sub_boff
	%sub_b =l loadl %sub_baddr
	%sub_aidx =l sub %d, 2
	%sub_aoff =l shl %sub_aidx, 3
	%sub_aaddr =l add %stack, %sub_aoff
	%sub_a =l loadl %sub_aaddr
	%sub_r =l sub %sub_a, %sub_b
	storel %sub_r, %sub_aaddr
	%sub_d =l sub %d, 1
	%sub_pc =l add %pc, 1
	jmp @dispatch

@mul
	%mul_top =l sub %d, 1
	%mul_boff =l shl %mul_top, 3
	%mul_baddr =l add %stack, %mul_boff
	%mul_b =l loadl %mul_baddr
	%mul_aidx =l sub %d, 2
	%mul_aoff =l shl %mul_aidx, 3
	%mul_aaddr =l add %stack, %mul_aoff
	%mul_a =l loadl %mul_aaddr
	%mul_r =l mul %mul_a, %mul_b
	storel %mul_r, %mul_aaddr
	%mul_d =l sub %d, 1
	%mul_pc =l add %pc, 1
	jmp @dispatch

@dup
	%dup_top =l sub %d, 1
	%dup_off =l shl %dup_top, 3
	%dup_addr =l add %stack, %dup_off
	%dup_val =l loadl %dup_addr
	%dup_new_off =l shl %d, 3
	%dup_new_addr =l add %stack, %dup_new_off
	storel %dup_val, %dup_new_addr
	%dup_d =l add %d, 1
	%dup_pc =l add %pc, 1
	jmp @dispatch

@swap
	%swap_top =l sub %d, 1
	%swap_boff =l shl %swap_top, 3
	%swap_baddr =l add %stack, %swap_boff
	%swap_b =l loadl %swap_baddr
	%swap_aidx =l sub %d, 2
	%swap_aoff =l shl %swap_aidx, 3
	%swap_aaddr =l add %stack, %swap_aoff
	%swap_a =l loadl %swap_aaddr
	storel %swap_b, %swap_aaddr
	storel %swap_a, %swap_baddr
	%swap_d =l copy %d
	%swap_pc =l add %pc, 1
	jmp @dispatch

@drop
	%drop_d =l sub %d, 1
	%drop_pc =l add %pc, 1
	jmp @dispatch

@over
	%over_idx =l sub %d, 2
	%over_off =l shl %over_idx, 3
	%over_addr =l add %stack, %over_off
	%over_val =l loadl %over_addr
	%over_new_off =l shl %d, 3
	%over_new_addr =l add %stack, %over_new_off
	storel %over_val, %over_new_addr
	%over_d =l add %d, 1
	%over_pc =l add %pc, 1
	jmp @dispatch

@lt
	%lt_top =l sub %d, 1
	%lt_boff =l shl %lt_top, 3
	%lt_baddr =l add %stack, %lt_boff
	%lt_b =l loadl %lt_baddr
	%lt_aidx =l sub %d, 2
	%lt_aoff =l shl %lt_aidx, 3
	%lt_aaddr =l add %stack, %lt_aoff
	%lt_a =l loadl %lt_aaddr
	%lt_c =w csltl %lt_a, %lt_b
	%lt_r =l extuw %lt_c
	storel %lt_r, %lt_aaddr
	%lt_d =l sub %d, 1
	%lt_pc =l add %pc, 1
	jmp @dispatch

@jz
	%jz_top =l sub %d, 1
	%jz_off =l shl %jz_top, 3
	%jz_addr =l add %stack, %jz_off
	%jz_val =l loadl %jz_addr
	%jz_one =l add %pc, 1
	%jz_opoff =l shl %jz_one, 3
	%jz_opaddr =l add %code, %jz_opoff
	%jz_target =l loadl %jz_opaddr
	%jz_zero_c =w ceql %jz_val, 0
	jnz %jz_zero_c, @jz_zero, @jz_nonzero

@jz_zero
	%jz_zero_pc =l copy %jz_target
	%jz_d =l sub %d, 1
	jmp @dispatch

@jz_nonzero
	%jz_nonzero_pc =l add %pc, 2
	jmp @dispatch

@jmp
	%jmp_one =l add %pc, 1
	%jmp_off =l shl %jmp_one, 3
	%jmp_addr =l add %code, %jmp_off
	%jmp_pc =l loadl %jmp_addr
	%jmp_d =l copy %d
	jmp @dispatch

@other
	%other_pc =l add %pc, 1
	%other_d =l copy %d
	jmp @dispatch

@stop_check
	%stop_d =l phi @dispatch %d, @halt %d
	%stop_empty =w ceql %stop_d, 0
	jnz %stop_empty, @empty, @nonempty

@empty
	ret 0

@nonempty
	%stop_idx =l sub %stop_d, 1
	%stop_off =l shl %stop_idx, 3
	%stop_addr =l add %stack, %stop_off
	%stop_val =l loadl %stop_addr
	ret %stop_val
}
