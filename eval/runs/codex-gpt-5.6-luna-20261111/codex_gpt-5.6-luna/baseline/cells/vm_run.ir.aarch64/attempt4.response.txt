export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @dispatch

@dispatch
	%pc =l phi @start 0, @push %ppc, @add %apc, @sub %spc, @mul %mpc, @dup %dpc, @swap %wpc, @drop %rpc, @over %opc, @lt %lpc, @jz_zero %zpc, @jz_nonzero %nzpc, @jmp %jpc, @other %xpc
	%d =l phi @start 0, @push %pd, @add %ad, @sub %sd, @mul %md, @dup %dd, @swap %wd, @drop %rd, @over %od, @lt %ld, @jz_zero %jd, @jz_nonzero %jd, @jmp %kd, @other %xd
	%done =w ceql %pc, %len
	jnz %done, @stop_run, @fetch

@fetch
	%off =l shl %pc, 3
	%addr =l add %code, %off
	%op =l loadl %addr
	%c0 =w ceql %op, 0
	jnz %c0, @stop_run, @t1

@t1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @t2

@t2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @t3

@t3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @t4

@t4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @t5

@t5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @t6

@t6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @t7

@t7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @t8

@t8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @t9

@t9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @t10

@t10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @t11

@t11
	%c11 =w ceql %op, 11
	jnz %c11, @jmp, @other

@push
	%p1 =l add %pc, 1
	%p2 =l shl %p1, 3
	%p3 =l add %code, %p2
	%pv =l loadl %p3
	%p4 =l shl %d, 3
	%p5 =l add %stack, %p4
	storel %pv, %p5
	%pd =l add %d, 1
	%ppc =l add %pc, 2
	jmp @dispatch

@add
	%ai =l sub %d, 1
	%ao =l shl %ai, 3
	%aa =l add %stack, %ao
	%ab =l loadl %aa
	%aj =l sub %d, 2
	%ak =l shl %aj, 3
	%al =l add %stack, %ak
	%av =l loadl %al
	%ar =l add %av, %ab
	storel %ar, %al
	%ad =l sub %d, 1
	%apc =l add %pc, 1
	jmp @dispatch

@sub
	%si =l sub %d, 1
	%so =l shl %si, 3
	%sa =l add %stack, %so
	%sb =l loadl %sa
	%sj =l sub %d, 2
	%sk =l shl %sj, 3
	%sl =l add %stack, %sk
	%sv =l loadl %sl
	%sr =l sub %sv, %sb
	storel %sr, %sl
	%sd =l sub %d, 1
	%spc =l add %pc, 1
	jmp @dispatch

@mul
	%mi =l sub %d, 1
	%mo =l shl %mi, 3
	%ma =l add %stack, %mo
	%mb =l loadl %ma
	%mj =l sub %d, 2
	%mk =l shl %mj, 3
	%ml =l add %stack, %mk
	%mv =l loadl %ml
	%mr =l mul %mv, %mb
	storel %mr, %ml
	%md =l sub %d, 1
	%mpc =l add %pc, 1
	jmp @dispatch

@dup
	%di =l sub %d, 1
	%do =l shl %di, 3
	%da =l add %stack, %do
	%dv =l loadl %da
	%dn =l shl %d, 3
	%dx =l add %stack, %dn
	storel %dv, %dx
	%dd =l add %d, 1
	%dpc =l add %pc, 1
	jmp @dispatch

@swap
	%wi =l sub %d, 1
	%wo =l shl %wi, 3
	%wa =l add %stack, %wo
	%wb =l loadl %wa
	%wj =l sub %d, 2
	%wk =l shl %wj, 3
	%wl =l add %stack, %wk
	%wv =l loadl %wl
	storel %wb, %wl
	storel %wv, %wa
	%wd =l copy %d
	%wpc =l add %pc, 1
	jmp @dispatch

@drop
	%rd =l sub %d, 1
	%rpc =l add %pc, 1
	jmp @dispatch

@over
	%oi =l sub %d, 2
	%oo =l shl %oi, 3
	%oa =l add %stack, %oo
	%ov =l loadl %oa
	%on =l shl %d, 3
	%ox =l add %stack, %on
	storel %ov, %ox
	%od =l add %d, 1
	%opc =l add %pc, 1
	jmp @dispatch

@lt
	%li =l sub %d, 1
	%lo =l shl %li, 3
	%la =l add %stack, %lo
	%lb =l loadl %la
	%lj =l sub %d, 2
	%lk =l shl %lj, 3
	%ll =l add %stack, %lk
	%lv =l loadl %ll
	%lc =w csltl %lv, %lb
	%lr =l extuw %lc
	storel %lr, %ll
	%ld =l sub %d, 1
	%lpc =l add %pc, 1
	jmp @dispatch

@jz
	%zi =l sub %d, 1
	%zo =l shl %zi, 3
	%za =l add %stack, %zo
	%zv =l loadl %za
	%zq =l add %pc, 1
	%zr =l shl %zq, 3
	%zs =l add %code, %zr
	%zt =l loadl %zs
	%zc =w ceql %zv, 0
	%jd =l sub %d, 1
	%zn =l add %pc, 2
	jnz %zc, @jz_zero, @jz_nonzero

@jz_zero
	%zpc =l copy %zt
	jmp @dispatch

@jz_nonzero
	%nzpc =l copy %zn
	jmp @dispatch

@jmp
	%ji =l add %pc, 1
	%jo =l shl %ji, 3
	%ja =l add %code, %jo
	%jpc =l loadl %ja
	%kd =l copy %d
	jmp @dispatch

@other
	%xd =l copy %d
	%xpc =l add %pc, 1
	jmp @dispatch

@stop_run
	%empty =w ceql %d, 0
	jnz %empty, @run_empty, @run_nonempty

@run_empty
	jmp @finish

@run_nonempty
	%ri =l sub %d, 1
	%ro =l shl %ri, 3
	%ra =l add %stack, %ro
	%rvv =l loadl %ra
	jmp @finish

@finish
	%rv =l phi @run_empty 0, @run_nonempty %rvv
	jmp @clear

@clear
	%ci =l phi @finish %d, @clear_store %cn
	%cc =w csltl %ci, %cap
	jnz %cc, @clear_store, @return

@clear_store
	%co =l shl %ci, 3
	%ca =l add %stack, %co
	storel 0, %ca
	%cn =l add %ci, 1
	jmp @clear

@return
	ret %rv
}
