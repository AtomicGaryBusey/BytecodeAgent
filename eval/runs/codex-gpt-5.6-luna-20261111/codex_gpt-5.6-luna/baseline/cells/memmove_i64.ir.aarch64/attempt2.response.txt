export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%z =w ceql %n, 0
	jnz %z, @done, @nonzero

@nonzero
	%bytes =l mul %n, 8
	%send =l add %src, %bytes
	%before =w cultl %dst, %src
	jnz %before, @forward, @check

@check
	%inside =w cultl %dst, %send
	jnz %inside, @backward, @forward

@forward
	jmp @floop

@floop
	%fr =l phi @forward %n, @fbody %frn
	%fsp =l phi @forward %src, @fbody %fsn
	%fdp =l phi @forward %dst, @fbody %fdn
	%fz =w ceql %fr, 0
	jnz %fz, @done, @fbody

@fbody
	%fv =l loadl %fsp
	storel %fv, %fdp
	%frn =l sub %fr, 1
	%fsn =l add %fsp, 8
	%fdn =l add %fdp, 8
	jmp @floop

@backward
	%last =l sub %n, 1
	%lastbytes =l mul %last, 8
	%bs =l add %src, %lastbytes
	%bd =l add %dst, %lastbytes
	jmp @bloop

@bloop
	%br =l phi @backward %n, @bbody %brn
	%bsp =l phi @backward %bs, @bbody %bspn
	%bdp =l phi @backward %bd, @bbody %bdpn
	%bz =w ceql %br, 0
	jnz %bz, @done, @bbody

@bbody
	%bv =l loadl %bsp
	storel %bv, %bdp
	%brn =l sub %br, 1
	%bspn =l sub %bsp, 8
	%bdpn =l sub %bdp, 8
	jmp @bloop

@done
	ret 0
}
