export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
%z = ceql %n, 0
jnz %z, @done, @nonzero

@nonzero
%bytes = mul %n, 8
%send = add %src, %bytes
%before = cultl %dst, %src
jnz %before, @forward, @check

@check
%inside = cultl %dst, %send
jnz %inside, @backward, @forward

@forward
jmp @floop

@floop
%fr = phi @forward %n, @fbody %frn
%fsp = phi @forward %src, @fbody %fsn
%fdp = phi @forward %dst, @fbody %fdn
%fz = ceql %fr, 0
jnz %fz, @done, @fbody

@fbody
%fv = loadl %fsp
storel %fv, %fdp
%frn = sub %fr, 1
%fsn = add %fsp, 8
%fdn = add %fdp, 8
jmp @floop

@backward
%last = sub %n, 1
%lastbytes = mul %last, 8
%bs = add %src, %lastbytes
%bd = add %dst, %lastbytes
jmp @bloop

@bloop
%br = phi @backward %n, @bbody %brn
%bsp = phi @backward %bs, @bbody %bspn
%bdp = phi @backward %bd, @bbody %bdpn
%bz = ceql %br, 0
jnz %bz, @done, @bbody

@bbody
%bv = loadl %bsp
storel %bv, %bdp
%brn = sub %br, 1
%bspn = sub %bsp, 8
%bdpn = sub %bdp, 8
jmp @bloop

@done
ret 0
}
