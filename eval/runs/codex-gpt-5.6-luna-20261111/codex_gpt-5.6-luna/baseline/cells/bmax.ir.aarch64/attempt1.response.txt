export function l $bmax(l %a, l %n) {
@start
	%maxslot =l alloc8 8
	%cntslot =l alloc8 8
	%ptrslot =l alloc8 8
	%first =l loadl %a
	storel %first, %maxslot
	%count =l sub %n, 1
	storel %count, %cntslot
	%nextptr =l add %a, 8
	storel %nextptr, %ptrslot
	jmp @loop
@loop
	%remaining =l loadl %cntslot
	%done =w ceql %remaining, 0
	jnz %done, @return, @check
@check
	%ptr =l loadl %ptrslot
	%value =l loadl %ptr
	%current =l loadl %maxslot
	%larger =w csltl %current, %value
	jnz %larger, @update, @advance
@update
	storel %value, %maxslot
@advance
	%pnext =l add %ptr, 8
	storel %pnext, %ptrslot
	%rnext =l sub %remaining, 1
	storel %rnext, %cntslot
	jmp @loop
@return
	%result =l loadl %maxslot
	ret %result
}
