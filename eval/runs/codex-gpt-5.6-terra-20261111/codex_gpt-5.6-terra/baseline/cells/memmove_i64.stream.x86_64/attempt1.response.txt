memmove_i64:
test rdx, rdx
jz .Ldone
cmp rdi, rsi
jbe .Lforward
.Lbackward:
dec rdx
mov rcx, qword ptr [rsi + rdx*8]
mov qword ptr [rdi + rdx*8], rcx
test rdx, rdx
jnz .Lbackward
jmp .Ldone
.Lforward:
mov rcx, qword ptr [rsi]
mov qword ptr [rdi], rcx
add rsi, 8
add rdi, 8
dec rdx
jnz .Lforward
.Ldone:
xor eax, eax
ret
