export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	%p0 =l add %a, 8
	%i0 =l sub %n, 1
	jmp @loop

@loop
	%max =l phi @start %max0, @cont %max1
	%p =l phi @start %p0, @cont %p1
	%i =l phi @start %i0, @cont %i1
	%done =w ceql %i, 0
	jnz %done, @end, @body

@body
	%v =l loadl %p
	%p1 =l add %p, 8
	%i1 =l sub %i, 1
	%greater =w csltl %max, %v
	jnz %greater, @take, @keep

@take
	jmp @cont

@keep
	jmp @cont

@cont
	%max1 =l phi @take %v, @keep %max
	jmp @loop

@end
	ret %max
}
