export function l $bmax(l %a, l %n) {
@start
	%max0 = loadl %a
	%p0 = add %a, 8
	%i0 = sub %n, 1
	jmp @loop

@loop
	%max = phi @start %max0, @cont %max1
	%p = phi @start %p0, @cont %p1
	%i = phi @start %i0, @cont %i1
	%done = ceql %i, 0
	jnz %done, @end, @body

@body
	%v = loadl %p
	%p1 = add %p, 8
	%i1 = sub %i, 1
	%greater = csltl %max, %v
	jnz %greater, @take, @keep

@take
	jmp @cont

@keep
	jmp @cont

@cont
	%max1 = phi @take %v, @keep %max
	jmp @loop

@end
	ret %max
}
