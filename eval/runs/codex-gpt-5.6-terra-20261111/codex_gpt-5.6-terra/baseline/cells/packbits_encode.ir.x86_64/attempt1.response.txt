export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l alloc8 8
	%written =l alloc8 8
	%run =l alloc8 8
	%len =l alloc8 8
	%k =l alloc8 8
	%litbyte =l alloc8 8
	storel 0, %i
	storel 0, %written
	jmp @outer

@outer
	%iv =l loadl %i
	%more =w csltl %iv, %n
	jnz %more, @outer_run_start, @done

@outer_run_start
	%op =l add %src, %iv
	%ob =w loadub %op
	storel 1, %run
	jmp @outer_run_loop

@outer_run_loop
	%rv =l loadl %run
	%rroom =w csltl %rv, 128
	jnz %rroom, @outer_run_check_input, @outer_run_done

@outer_run_check_input
	%ix =l add %iv, %rv
	%inrange =w csltl %ix, %n
	jnz %inrange, @outer_run_compare, @outer_run_done

@outer_run_compare
	%xp =l add %src, %ix
	%xb =w loadub %xp
	%same =w ceqw %xb, %ob
	jnz %same, @outer_run_inc, @outer_run_done

@outer_run_inc
	%rn =l add %rv, 1
	storel %rn, %run
	jmp @outer_run_loop

@outer_run_done
	%rf =l loadl %run
	%isrep =w cslel 3, %rf
	jnz %isrep, @repeat_packet, @literal_start

@repeat_packet
	%wv =l loadl %written
	%wend =l add %wv, 2
	%fits =w cslel %wend, %cap
	jnz %fits, @repeat_write, @fail

@repeat_write
	%dp =l add %out, %wv
	%hdr =l sub 257, %rf
	storeb %hdr, %dp
	%dp1 =l add %dp, 1
	storeb %ob, %dp1
	storel %wend, %written
	%inext =l add %iv, %rf
	storel %inext, %i
	jmp @outer

@literal_start
	storel 0, %len
	jmp @literal_loop

@literal_loop
	%lv =l loadl %len
	%full =w cslel 128, %lv
	jnz %full, @literal_emit, @literal_check_input

@literal_check_input
	%ibase =l loadl %i
	%jv =l add %ibase, %lv
	%linrange =w csltl %jv, %n
	jnz %linrange, @literal_run_start, @literal_emit

@literal_run_start
	%lp =l add %src, %jv
	%lfirst =w loadub %lp
	storeb %lfirst, %litbyte
	storel 1, %run
	jmp @literal_run_loop

@literal_run_loop
	%lrv =l loadl %run
	%lrroom =w csltl %lrv, 128
	jnz %lrroom, @literal_run_check_input, @literal_run_done

@literal_run_check_input
	%lix =l add %jv, %lrv
	%lrange =w csltl %lix, %n
	jnz %lrange, @literal_run_compare, @literal_run_done

@literal_run_compare
	%lxp =l add %src, %lix
	%lxb =w loadub %lxp
	%lb =w loadub %litbyte
	%lsame =w ceqw %lxb, %lb
	jnz %lsame, @literal_run_inc, @literal_run_done

@literal_run_inc
	%lrn =l add %lrv, 1
	storel %lrn, %run
	jmp @literal_run_loop

@literal_run_done
	%lrf =l loadl %run
	%stoplit =w cslel 3, %lrf
	jnz %stoplit, @literal_emit, @literal_len_inc

@literal_len_inc
	%ln =l add %lv, 1
	storel %ln, %len
	jmp @literal_loop

@literal_emit
	%elen =l loadl %len
	%ew =l loadl %written
	%need =l add %elen, 1
	%end =l add %ew, %need
	%efits =w cslel %end, %cap
	jnz %efits, @literal_write_header, @fail

@literal_write_header
	%eh =l sub %elen, 1
	%edp =l add %out, %ew
	storeb %eh, %edp
	storel 0, %k
	jmp @literal_copy_loop

@literal_copy_loop
	%kv =l loadl %k
	%copy_more =w csltl %kv, %elen
	jnz %copy_more, @literal_copy_one, @literal_finish

@literal_copy_one
	%ci =l loadl %i
	%si =l add %ci, %kv
	%sp =l add %src, %si
	%cb =w loadub %sp
	%doff =l add %ew, 1
	%didx =l add %doff, %kv
	%cp =l add %out, %didx
	storeb %cb, %cp
	%kn =l add %kv, 1
	storel %kn, %k
	jmp @literal_copy_loop

@literal_finish
	storel %end, %written
	%fi =l loadl %i
	%fni =l add %fi, %elen
	storel %fni, %i
	jmp @outer

@done
	%result =l loadl %written
	ret %result

@fail
	ret -1
}
