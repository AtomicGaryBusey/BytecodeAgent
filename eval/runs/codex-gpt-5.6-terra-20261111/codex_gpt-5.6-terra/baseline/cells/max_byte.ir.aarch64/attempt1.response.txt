export function l $max_byte(l %p, l %n) {
@start
	%z =l copy 0
	%nonzero =w csltl %z, %n
	jnz %nonzero, @loop, @done

@loop
	%i =l phi @start %z, @next %in
	%m =l phi @start %z, @next %mn
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%a =l add %p, %i
	%b =l loadub %a
	%larger =w csltl %m, %b
	jnz %larger, @update, @keep

@update
	jmp @next

@keep
	jmp @next

@next
	%mn =l phi @update %b, @keep %m
	%in =l add %i, 1
	jmp @loop

@done
	%r =l phi @start %z, @loop %m
	ret %r
}
