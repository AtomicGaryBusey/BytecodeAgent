export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @packet

@packet
	%more =w csltl %i, %n
	jnz %more, @header, @done

@header
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i =l add %i, 1
	%noop =w ceqw %h, -128
	jnz %noop, @packet, @kind

@kind
	%negative =w csltw %h, 0
	jnz %negative, @repeat, @literal

@literal
	%hu =l extuw %h
	%llen =l add %hu, 1
	%remain =l sub %n, %i
	%valid =w cslel %llen, %remain
	jnz %valid, @literal_space, @source_error

@literal_space
	%space =l sub %cap, %o
	%fits =w cslel %llen, %space
	jnz %fits, @literal_full_init, @literal_part_init

@literal_full_init
	%k =l copy 0
	jmp @literal_full_loop

@literal_full_loop
	%cont =w csltl %k, %llen
	jnz %cont, @literal_full_copy, @literal_full_done

@literal_full_copy
	%sp =l add %src, %i
	%sp =l add %sp, %k
	%b =w loadub %sp
	%op =l add %out, %o
	%op =l add %op, %k
	storeb %b, %op
	%k =l add %k, 1
	jmp @literal_full_loop

@literal_full_done
	%i =l add %i, %llen
	%o =l add %o, %llen
	jmp @packet

@literal_part_init
	%kpart =l copy 0
	jmp @literal_part_loop

@literal_part_loop
	%pcont =w csltl %kpart, %space
	jnz %pcont, @literal_part_copy, @output_error

@literal_part_copy
	%sp2 =l add %src, %i
	%sp2 =l add %sp2, %kpart
	%b2 =w loadub %sp2
	%op2 =l add %out, %o
	%op2 =l add %op2, %kpart
	storeb %b2, %op2
	%kpart =l add %kpart, 1
	jmp @literal_part_loop

@repeat
	%hs =l extsw %h
	%rlen =l sub 1, %hs
	%remain2 =l sub %n, %i
	%valid2 =w cslel 1, %remain2
	jnz %valid2, @repeat_value, @source_error

@repeat_value
	%rp =l add %src, %i
	%rb =w loadub %rp
	%space2 =l sub %cap, %o
	%fits2 =w cslel %rlen, %space2
	jnz %fits2, @repeat_full_init, @repeat_part_init

@repeat_full_init
	%rk =l copy 0
	jmp @repeat_full_loop

@repeat_full_loop
	%rcont =w csltl %rk, %rlen
	jnz %rcont, @repeat_full_copy, @repeat_full_done

@repeat_full_copy
	%rop =l add %out, %o
	%rop =l add %rop, %rk
	storeb %rb, %rop
	%rk =l add %rk, 1
	jmp @repeat_full_loop

@repeat_full_done
	%i =l add %i, 1
	%o =l add %o, %rlen
	jmp @packet

@repeat_part_init
	%rpk =l copy 0
	jmp @repeat_part_loop

@repeat_part_loop
	%rpcont =w csltl %rpk, %space2
	jnz %rpcont, @repeat_part_copy, @output_error

@repeat_part_copy
	%rpop =l add %out, %o
	%rpop =l add %rpop, %rpk
	storeb %rb, %rpop
	%rpk =l add %rpk, 1
	jmp @repeat_part_loop

@source_error
	ret -2

@output_error
	ret -1

@done
	ret %o
}
