export function l $fib(l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @loop

@zero
	ret 0

@loop
	%a =l phi @start 0, @body %b
	%b =l phi @start 1, @body %sum
	%i =l phi @start 0, @body %next
	%done =w ceql %i, %n
	jnz %done, @return, @body

@body
	%sum =l add %a, %b
	%next =l add %i, 1
	jmp @loop

@return
	ret %a
}
