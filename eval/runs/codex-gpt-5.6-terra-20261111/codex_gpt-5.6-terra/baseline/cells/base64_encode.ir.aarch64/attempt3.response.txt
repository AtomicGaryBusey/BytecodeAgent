export function l $base64_encode(l %src, l %n, l %out) {
@entry
	jmp @loop

@loop
	%s =l phi @entry %src, @groupdone %sn
	%r =l phi @entry %n, @groupdone %rn
	%op =l phi @entry %out, @groupdone %opn
	%cnt =l phi @entry 0, @groupdone %cntn
	%more =w csltl 0, %r
	jnz %more, @group, @done

@group
	%b0 =l loadub %s
	%has1 =w csltl 1, %r
	jnz %has1, @load1, @no1

@load1
	%p1 =l add %s, 1
	%x1 =l loadub %p1
	jmp @join1

@no1
	jmp @join1

@join1
	%b1 =l phi @load1 %x1, @no1 0
	%has2 =w csltl 2, %r
	jnz %has2, @load2, @no2

@load2
	%p2 =l add %s, 2
	%x2 =l loadub %p2
	jmp @join2

@no2
	jmp @join2

@join2
	%b2 =l phi @load2 %x2, @no2 0
	%v0 =l shl %b0, 16
	%v1 =l shl %b1, 8
	%v2 =l or %v0, %v1
	%val =l or %v2, %b2
	jnz %has2, @full, @partial

@full
	jmp @ready

@partial
	jmp @ready

@ready
	%k =l phi @full 3, @partial %r
	%m =l add %k, 1
	jmp @emit

@emit
	%j =l phi @ready 0, @put %jn
	%cp =l phi @ready %op, @put %cpn
	%valid =w csltl %j, %m
	%sb =l sub 3, %j
	%sh =l mul %sb, 6
	%q =l shr %val, %sh
	%idx =l and %q, 63
	jnz %valid, @map, @pad

@pad
	jmp @put

@map
	%lt26 =w csltl %idx, 26
	jnz %lt26, @mapa, @maprest

@mapa
	%ca =l add %idx, 65
	jmp @put

@maprest
	%lt52 =w csltl %idx, 52
	jnz %lt52, @mapb, @maprest2

@mapb
	%cb =l add %idx, 71
	jmp @put

@maprest2
	%lt62 =w csltl %idx, 62
	jnz %lt62, @mapc, @maprest3

@mapc
	%cc =l sub %idx, 4
	jmp @put

@maprest3
	%is62 =w ceql %idx, 62
	jnz %is62, @mapplus, @mapslash

@mapplus
	jmp @put

@mapslash
	jmp @put

@put
	%ch =l phi @pad 61, @mapa %ca, @mapb %cb, @mapc %cc, @mapplus 43, @mapslash 47
	storeb %ch, %cp
	%cpn =l add %cp, 1
	%jn =l add %j, 1
	%again =w csltl %jn, 4
	jnz %again, @emit, @groupdone

@groupdone
	%sn =l add %s, %k
	%rn =l sub %r, %k
	%opn =l copy %cpn
	%cntn =l add %cnt, 4
	jmp @loop

@done
	ret %cnt
}
