export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%rp =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	%z0 =l copy 0
@clear
	%zc =w csltl %z0, %cap
	jnz %zc, @clearbody, @dispatch
@clearbody
	%zo =l shl %z0, 3
	%za =l add %stack, %zo
	storel 0, %za
	%z0 =l add %z0, 1
	jmp @clear
@dispatch
	%pc =l loadl %pcp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch
@fetch
	%fo =l shl %pc, 3
	%fa =l add %code, %fo
	%op =l loadl %fa
	%c0 =w ceql %op, 0
	jnz %c0, @finish, @test1
@test1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @test2
@test2
	%c2 =w ceql %op, 2
	jnz %c2, @addop, @test3
@test3
	%c3 =w ceql %op, 3
	jnz %c3, @subop, @test4
@test4
	%c4 =w ceql %op, 4
	jnz %c4, @mulop, @test5
@test5
	%c5 =w ceql %op, 5
	jnz %c5, @dupop, @test6
@test6
	%c6 =w ceql %op, 6
	jnz %c6, @swapop, @test7
@test7
	%c7 =w ceql %op, 7
	jnz %c7, @dropop, @test8
@test8
	%c8 =w ceql %op, 8
	jnz %c8, @overop, @test9
@test9
	%c9 =w ceql %op, 9
	jnz %c9, @ltop, @test10
@test10
	%c10 =w ceql %op, 10
	jnz %c10, @jzop, @test11
@test11
	%c11 =w ceql %op, 11
	jnz %c11, @jmpop, @next
@next
	%np =l add %pc, 1
	storel %np, %pcp
	jmp @dispatch
@push
	%pp =l add %pc, 1
	%po =l shl %pp, 3
	%pa =l add %code, %po
	%pv =l loadl %pa
	%pd =l loadl %dp
	%ps =l shl %pd, 3
	%psa =l add %stack, %ps
	storel %pv, %psa
	%pnd =l add %pd, 1
	%pnp =l add %pc, 2
	storel %pnd, %dp
	storel %pnp, %pcp
	jmp @dispatch
@addop
	%ad =l loadl %dp
	%ai1 =l sub %ad, 1
	%ai2 =l sub %ad, 2
	%ao1 =l shl %ai1, 3
	%ao2 =l shl %ai2, 3
	%aa1 =l add %stack, %ao1
	%aa2 =l add %stack, %ao2
	%ab =l loadl %aa1
	%aa =l loadl %aa2
	%ar =l add %aa, %ab
	storel %ar, %aa2
	storel %ai1, %dp
	%anp =l add %pc, 1
	storel %anp, %pcp
	jmp @dispatch
@subop
	%sd =l loadl %dp
	%si1 =l sub %sd, 1
	%si2 =l sub %sd, 2
	%so1 =l shl %si1, 3
	%so2 =l shl %si2, 3
	%sa1 =l add %stack, %so1
	%sa2 =l add %stack, %so2
	%sb =l loadl %sa1
	%sa =l loadl %sa2
	%sr =l sub %sa, %sb
	storel %sr, %sa2
	storel %si1, %dp
	%snp =l add %pc, 1
	storel %snp, %pcp
	jmp @dispatch
@mulop
	%md =l loadl %dp
	%mi1 =l sub %md, 1
	%mi2 =l sub %md, 2
	%mo1 =l shl %mi1, 3
	%mo2 =l shl %mi2, 3
	%ma1 =l add %stack, %mo1
	%ma2 =l add %stack, %mo2
	%mb =l loadl %ma1
	%ma =l loadl %ma2
	%mr =l mul %ma, %mb
	storel %mr, %ma2
	storel %mi1, %dp
	%mnp =l add %pc, 1
	storel %mnp, %pcp
	jmp @dispatch
@dupop
	%dd =l loadl %dp
	%di =l sub %dd, 1
	%do =l shl %di, 3
	%da =l add %stack, %do
	%dv =l loadl %da
	%dso =l shl %dd, 3
	%dsa =l add %stack, %dso
	storel %dv, %dsa
	%dnd =l add %dd, 1
	%dnp =l add %pc, 1
	storel %dnd, %dp
	storel %dnp, %pcp
	jmp @dispatch
@swapop
	%wd =l loadl %dp
	%wi1 =l sub %wd, 1
	%wi2 =l sub %wd, 2
	%wo1 =l shl %wi1, 3
	%wo2 =l shl %wi2, 3
	%wa1 =l add %stack, %wo1
	%wa2 =l add %stack, %wo2
	%wv1 =l loadl %wa1
	%wv2 =l loadl %wa2
	storel %wv2, %wa1
	storel %wv1, %wa2
	%wnp =l add %pc, 1
	storel %wnp, %pcp
	jmp @dispatch
@dropop
	%xd =l loadl %dp
	%xnd =l sub %xd, 1
	%xnp =l add %pc, 1
	storel %xnd, %dp
	storel %xnp, %pcp
	jmp @dispatch
@overop
	%od =l loadl %dp
	%oi =l sub %od, 2
	%oo =l shl %oi, 3
	%oa =l add %stack, %oo
	%ov =l loadl %oa
	%oso =l shl %od, 3
	%osa =l add %stack, %oso
	storel %ov, %osa
	%ond =l add %od, 1
	%onp =l add %pc, 1
	storel %ond, %dp
	storel %onp, %pcp
	jmp @dispatch
@ltop
	%ld =l loadl %dp
	%li1 =l sub %ld, 1
	%li2 =l sub %ld, 2
	%lo1 =l shl %li1, 3
	%lo2 =l shl %li2, 3
	%la1 =l add %stack, %lo1
	%la2 =l add %stack, %lo2
	%lb =l loadl %la1
	%la =l loadl %la2
	%lc =w csltl %la, %lb
	%lv =l extsw %lc
	storel %lv, %la2
	storel %li1, %dp
	%lnp =l add %pc, 1
	storel %lnp, %pcp
	jmp @dispatch
@jzop
	%jd =l loadl %dp
	%jnd =l sub %jd, 1
	%ji =l sub %jd, 1
	%jo =l shl %ji, 3
	%ja =l add %stack, %jo
	%jv =l loadl %ja
	%jp =l add %pc, 1
	%jpo =l shl %jp, 3
	%jpa =l add %code, %jpo
	%jt =l loadl %jpa
	storel %jnd, %dp
	%jzero =w ceql %jv, 0
	jnz %jzero, @jztake, @jzskip
@jztake
	storel %jt, %pcp
	jmp @dispatch
@jzskip
	%jsnp =l add %pc, 2
	storel %jsnp, %pcp
	jmp @dispatch
@jmpop
	%kp =l add %pc, 1
	%ko =l shl %kp, 3
	%ka =l add %code, %ko
	%kv =l loadl %ka
	storel %kv, %pcp
	jmp @dispatch
@finish
	%fd =l loadl %dp
	%fempty =w ceql %fd, 0
	jnz %fempty, @empty, @nonempty
@empty
	storel 0, %rp
	storel 0, %pcp
	jmp @tail
@nonempty
	%fi =l sub %fd, 1
	%foff =l shl %fi, 3
	%faddr =l add %stack, %foff
	%fv =l loadl %faddr
	storel %fv, %rp
	storel %fd, %pcp
	jmp @tail
@tail
	%ti =l loadl %pcp
	%tc =w csltl %ti, %cap
	jnz %tc, @tailbody, @return
@tailbody
	%to =l shl %ti, 3
	%ta =l add %stack, %to
	storel 0, %ta
	%tn =l add %ti, 1
	storel %tn, %pcp
	jmp @tail
@return
	%rv =l loadl %rp
	ret %rv
}
