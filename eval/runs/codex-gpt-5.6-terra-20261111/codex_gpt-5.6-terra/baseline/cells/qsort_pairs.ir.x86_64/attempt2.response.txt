export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret, @setup

@setup
	%mid =l shr %n, 1
	%moff =l shl %mid, 4
	%mp =l add %a, %moff
	%pk =l loadl %mp
	%mvp =l add %mp, 8
	%pv =l loadl %mvp
	jmp @part

@part
	%i =l phi @setup 0, @less %i_less, @greater %i_greater, @equal %i_equal
	%lt =l phi @setup 0, @less %lt_less, @greater %lt_greater, @equal %lt_equal
	%gt =l phi @setup %n, @less %gt_less, @greater %gt_greater, @equal %gt_equal
	%more =w csltl %i, %gt
	jnz %more, @inspect, @done

@inspect
	%ioff =l shl %i, 4
	%ip =l add %a, %ioff
	%ck =l loadl %ip
	%ivp =l add %ip, 8
	%cv =l loadl %ivp
	%kl =w csltl %ck, %pk
	jnz %kl, @less, @keyeq

@keyeq
	%ke =w ceql %ck, %pk
	jnz %ke, @valuecmp, @greater

@valuecmp
	%vl =w csltl %cv, %pv
	jnz %vl, @less, @valueeq

@valueeq
	%ve =w ceql %cv, %pv
	jnz %ve, @equal, @greater

@less
	%loff =l shl %lt, 4
	%lp =l add %a, %loff
	%lk =l loadl %lp
	%lvp =l add %lp, 8
	%lv =l loadl %lvp
	storel %ck, %lp
	storel %cv, %lvp
	storel %lk, %ip
	storel %lv, %ivp
	%i_less =l add %i, 1
	%lt_less =l add %lt, 1
	%gt_less =l copy %gt
	jmp @part

@greater
	%gt_greater =l sub %gt, 1
	%goff =l shl %gt_greater, 4
	%gp =l add %a, %goff
	%gk =l loadl %gp
	%gvp =l add %gp, 8
	%gv =l loadl %gvp
	storel %gk, %ip
	storel %gv, %ivp
	storel %ck, %gp
	storel %cv, %gvp
	%i_greater =l copy %i
	%lt_greater =l copy %lt
	jmp @part

@equal
	%i_equal =l add %i, 1
	%lt_equal =l copy %lt
	%gt_equal =l copy %gt
	jmp @part

@done
	%left =l call $qsort_pairs(l %a, l %lt)
	%roff =l shl %gt, 4
	%rp =l add %a, %roff
	%rn =l sub %n, %gt
	%right =l call $qsort_pairs(l %rp, l %rn)
	jmp @ret

@ret
	ret 0
}
