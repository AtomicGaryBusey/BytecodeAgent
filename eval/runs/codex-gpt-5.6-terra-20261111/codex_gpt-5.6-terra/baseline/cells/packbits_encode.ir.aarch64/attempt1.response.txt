export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	%ip =l alloc8 8
	%wp =l alloc8 8
	%rp =l alloc4 4
	%jp =l alloc8 8
	%lp =l alloc4 4
	%kp =l alloc4 4
	storel 0, %ip
	storel 0, %wp
	jmp @main

@main
	%i =l loadl %ip
	%more =w csltl %i, %n
	jnz %more, @run_start, @done

@run_start
	%sptr =l add %src, %i
	%v =w loadub %sptr
	storew 1, %rp
	jmp @run_loop

@run_loop
	%r =w loadw %rp
	%rsmall =w csltw %r, 128
	jnz %rsmall, @run_bounds, @run_done

@run_bounds
	%rl =l extuw %r
	%ri =l add %i, %rl
	%rin =w csltl %ri, %n
	jnz %rin, @run_compare, @run_done

@run_compare
	%rptr =l add %src, %ri
	%rv =w loadub %rptr
	%same =w ceqw %rv, %v
	jnz %same, @run_inc, @run_done

@run_inc
	%rnext =w add %r, 1
	storew %rnext, %rp
	jmp @run_loop

@run_done
	%rf =w loadw %rp
	%isrep =w cslew 3, %rf
	jnz %isrep, @repeat_packet, @literal_start

@repeat_packet
	%rfl =l extuw %rf
	%needr =l add 2, 0
	%writtenr =l loadl %wp
	%endr =l add %writtenr, %needr
	%fitsr =w cslel %endr, %cap
	jnz %fitsr, @repeat_write, @fail

@repeat_write
	%rh =w sub 257, %rf
	%rop =l add %out, %writtenr
	storeb %rh, %rop
	%ropp =l add %rop, 1
	storeb %v, %ropp
	%inextr =l add %i, %rfl
	%wnextr =l add %writtenr, 2
	storel %inextr, %ip
	storel %wnextr, %wp
	jmp @main

@literal_start
	storel %i, %jp
	storew 0, %lp
	jmp @literal_loop

@literal_loop
	%j =l loadl %jp
	%len =w loadw %lp
	%jin =w csltl %j, %n
	jnz %jin, @literal_lencheck, @literal_done

@literal_lencheck
	%lroom =w csltw %len, 128
	jnz %lroom, @literal_triple_bounds, @literal_done

@literal_triple_bounds
	%j2 =l add %j, 2
	%has3 =w csltl %j2, %n
	jnz %has3, @literal_triple_check, @literal_add

@literal_triple_check
	%p0 =l add %src, %j
	%j1 =l add %j, 1
	%p1 =l add %src, %j1
	%p2 =l add %src, %j2
	%b0 =w loadub %p0
	%b1 =w loadub %p1
	%b2 =w loadub %p2
	%e01 =w ceqw %b0, %b1
	jnz %e01, @literal_triple_second, @literal_add

@literal_triple_second
	%e12 =w ceqw %b1, %b2
	jnz %e12, @literal_done, @literal_add

@literal_add
	%jnext =l add %j, 1
	%lennext =w add %len, 1
	storel %jnext, %jp
	storew %lennext, %lp
	jmp @literal_loop

@literal_done
	%lf =w loadw %lp
	%lfl =l extuw %lf
	%needl =l add %lfl, 1
	%writtenl =l loadl %wp
	%endl =l add %writtenl, %needl
	%fitsl =w cslel %endl, %cap
	jnz %fitsl, @literal_write_header, @fail

@literal_write_header
	%lh =w sub %lf, 1
	%lop =l add %out, %writtenl
	storeb %lh, %lop
	storew 0, %kp
	jmp @literal_copy

@literal_copy
	%k =w loadw %kp
	%morecopy =w csltw %k, %lf
	jnz %morecopy, @literal_copy_one, @literal_finish

@literal_copy_one
	%kl =l extuw %k
	%cpin =l add %i, %kl
	%cpip =l add %src, %cpin
	%cb =w loadub %cpip
	%cpoff =l add %writtenl, 1
	%cpout0 =l add %out, %cpoff
	%cpout =l add %cpout0, %kl
	storeb %cb, %cpout
	%knext =w add %k, 1
	storew %knext, %kp
	jmp @literal_copy

@literal_finish
	%jfinal =l loadl %jp
	%wfinal =l add %writtenl, %needl
	storel %jfinal, %ip
	storel %wfinal, %wp
	jmp @main

@done
	%result =l loadl %wp
	ret %result

@fail
	ret -1
}
