export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	jmp @zero_loop

@zero_loop
	%zi =l phi @start 0, @zero_body %zinext
	%zmore =w csltl %zi, %total
	jnz %zmore, @zero_body, @outer_loop

@zero_body
	%zoff =l mul %zi, 8
	%zptr =l add %r, %zoff
	storel 0, %zptr
	%zinext =l add %zi, 1
	jmp @zero_loop

@outer_loop
	%oi =l phi @zero_loop 0, @outer_end %oinext
	%omore =w csltl %oi, %na
	jnz %omore, @outer_body, @scan_init

@outer_body
	%aoff =l mul %oi, 8
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	jmp @inner_loop

@inner_loop
	%ij =l phi @outer_body 0, @inner_body %ijnext
	%carry =l phi @outer_body 0, @inner_body %carrynext
	%imore =w csltl %ij, %nb
	jnz %imore, @inner_body, @outer_end

@inner_body
	%boff =l mul %ij, 8
	%bptr =l add %b, %boff
	%bv =l loadl %bptr

	%al =l and %av, 4294967295
	%ah =l shr %av, 32
	%bl =l and %bv, 4294967295
	%bh =l shr %bv, 32

	%p0 =l mul %al, %bl
	%p1 =l mul %al, %bh
	%p2 =l mul %ah, %bl
	%p3 =l mul %ah, %bh

	%p1lo =l and %p1, 4294967295
	%p1sh =l shl %p1lo, 32
	%p2lo =l and %p2, 4294967295
	%p2sh =l shl %p2lo, 32

	%ps1 =l add %p0, %p1sh
	%pc1 =w cultl %ps1, %p0
	%plo =l add %ps1, %p2sh
	%pc2 =w cultl %plo, %ps1

	%p1hi =l shr %p1, 32
	%p2hi =l shr %p2, 32
	%pc1l =l extuw %pc1
	%pc2l =l extuw %pc2
	%ph1 =l add %p3, %p1hi
	%ph2 =l add %ph1, %p2hi
	%ph3 =l add %ph2, %pc1l
	%phi =l add %ph3, %pc2l

	%rk =l add %oi, %ij
	%roff =l mul %rk, 8
	%rptr =l add %r, %roff
	%old =l loadl %rptr
	%sum1 =l add %old, %plo
	%rc1 =w cultl %sum1, %old
	%sum2 =l add %sum1, %carry
	%rc2 =w cultl %sum2, %sum1
	storel %sum2, %rptr

	%rc1l =l extuw %rc1
	%rc2l =l extuw %rc2
	%ch1 =l add %phi, %rc1l
	%carrynext =l add %ch1, %rc2l
	%ijnext =l add %ij, 1
	jmp @inner_loop

@outer_end
	%endidx =l add %oi, %nb
	%endoff =l mul %endidx, 8
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%oinext =l add %oi, 1
	jmp @outer_loop

@scan_init
	jmp @scan_loop

@scan_loop
	%si =l phi @scan_init %total, @scan_zero %sinext
	%smore =w csltl 0, %si
	jnz %smore, @scan_check, @ret_zero

@scan_check
	%sinext =l sub %si, 1
	%soff =l mul %sinext, 8
	%sptr =l add %r, %soff
	%sv =l loadl %sptr
	%sz =w ceql %sv, 0
	jnz %sz, @scan_zero, @ret_found

@scan_zero
	jmp @scan_loop

@ret_found
	ret %sinext

@ret_zero
	ret 0
}
