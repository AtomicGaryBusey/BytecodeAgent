export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @init

@init
	%z =l phi @start 0, @initnext %znext
	%iz =w csltl %z, %n
	jnz %iz, @initbody, @outer

@initbody
	%zoff =l shl %z, 3
	%zptr =l add %r, %zoff
	storel 0, %zptr
	jmp @initnext

@initnext
	%znext =l add %z, 1
	jmp @init

@outer
	%i =l phi @init 0, @outernext %inext
	%io =w csltl %i, %na
	jnz %io, @outerbody, @scanstart

@outerbody
	%aioff =l shl %i, 3
	%aiptr =l add %a, %aioff
	%ai =l loadl %aiptr
	jmp @inner

@inner
	%j =l phi @outerbody 0, @innernext %jnext
	%ij =w csltl %j, %nb
	jnz %ij, @pair, @outernext

@pair
	%bjoff =l shl %j, 3
	%bjptr =l add %b, %bjoff
	%bj =l loadl %bjptr

	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32

	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1

	%p01s =l shl %p01, 32
	%plo1 =l add %p00, %p01s
	%c01 =w cultl %plo1, %p00
	%c01l =l extuw %c01
	%phi1 =l shr %p01, 32
	%phi2 =l add %phi1, %c01l

	%p10s =l shl %p10, 32
	%plo =l add %plo1, %p10s
	%c10 =w cultl %plo, %plo1
	%c10l =l extuw %c10
	%phi3 =l shr %p10, 32
	%phi4 =l add %phi2, %phi3
	%phi5 =l add %phi4, %c10l
	%phi =l add %phi5, %p11

	%k =l add %i, %j
	%koff =l shl %k, 3
	%kptr =l add %r, %koff
	%oldlo =l loadl %kptr
	%sumlo =l add %oldlo, %plo
	%clo =w cultl %sumlo, %oldlo
	storel %sumlo, %kptr

	%k1 =l add %k, 1
	%k1off =l shl %k1, 3
	%k1ptr =l add %r, %k1off
	%oldhi =l loadl %k1ptr
	%sumhi1 =l add %oldhi, %phi
	%chi1 =w cultl %sumhi1, %oldhi
	%clol =l extuw %clo
	%sumhi =l add %sumhi1, %clol
	%chi2 =w cultl %sumhi, %sumhi1
	storel %sumhi, %k1ptr
	%cout =w or %chi1, %chi2
	%kp2 =l add %k, 2
	%jnext =l add %j, 1
	jmp @propcheck

@propcheck
	%pp =l phi @pair %kp2, @propnext %ppnext
	%pc =w phi @pair %cout, @propnext %pcnext
	jnz %pc, @prophave, @innernext

@prophave
	%pin =w csltl %pp, %n
	jnz %pin, @propdo, @innernext

@propdo
	%poff =l shl %pp, 3
	%pptr =l add %r, %poff
	%pv =l loadl %pptr
	%ps =l add %pv, 1
	%pcnext =w ceql %ps, 0
	storel %ps, %pptr
	jmp @propnext

@propnext
	%ppnext =l add %pp, 1
	jmp @propcheck

@innernext
	jmp @inner

@outernext
	%inext =l add %i, 1
	jmp @outer

@scanstart
	%nonempty =w csltl 0, %n
	jnz %nonempty, @scaninit, @retzero

@scaninit
	%slast =l sub %n, 1
	jmp @scan

@scan
	%si =l phi @scaninit %slast, @scanagain %sprevious
	%sioff =l shl %si, 3
	%siptr =l add %r, %sioff
	%sv =l loadl %siptr
	%szero =w ceql %sv, 0
	jnz %szero, @scannext, @found

@scannext
	%atzero =w ceql %si, 0
	jnz %atzero, @retzero, @scanagain

@scanagain
	%sprevious =l sub %si, 1
	jmp @scan

@found
	%result =l add %si, 1
	ret %result

@retzero
	ret 0
}
