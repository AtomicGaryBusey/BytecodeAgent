export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%twonn =l mul %nn, 2
	jmp @clearout

@clearout
	%co =l phi @start 0, @clearoutinc %con
	%cocond =w csltl %co, %nn
	jnz %cocond, @clearoutbody, @diag

@clearoutbody
	%coff =l mul %co, 8
	%cop =l add %out, %coff
	storel 0, %cop
	jmp @clearoutinc

@clearoutinc
	%con =l add %co, 1
	jmp @clearout

@diag
	%d =l phi @clearout 0, @diaginc %dn
	%dcond =w csltl %d, %n
	jnz %dcond, @diagbody, @power

@diagbody
	%dbase =l mul %d, %n
	%didx =l add %dbase, %d
	%doff =l mul %didx, 8
	%dp =l add %out, %doff
	storel 1, %dp
	jmp @diaginc

@diaginc
	%dn =l add %d, 1
	jmp @diag

@power
	%iter =l phi @diag 0, @powerinc %itern
	%itercond =w csltl %iter, %k
	jnz %itercond, @powrow, @cleanup

@powrow
	%r =l phi @power 0, @powrowinc %rn
	%rcond =w csltl %r, %n
	jnz %rcond, @powcol, @copy

@powcol
	%c =l phi @powrow 0, @powcolinc %cn
	%ccond =w csltl %c, %n
	jnz %ccond, @powt, @powrowinc

@powt
	%t =l phi @powcol 0, @powtbody %tn
	%s =l phi @powcol 0, @powtbody %sn
	%tcond =w csltl %t, %n
	jnz %tcond, @powtbody, @prodstore

@powtbody
	%lbase =l mul %r, %n
	%lidx =l add %lbase, %t
	%loff =l mul %lidx, 8
	%lp =l add %out, %loff
	%lv =l loadl %lp
	%rbase =l mul %t, %n
	%ridx =l add %rbase, %c
	%roff =l mul %ridx, 8
	%rp =l add %a, %roff
	%rv =l loadl %rp
	%prod =l mul %lv, %rv
	%sn =l add %s, %prod
	%tn =l add %t, 1
	jmp @powt

@prodstore
	%pbase =l mul %r, %n
	%pidx =l add %pbase, %c
	%poff =l mul %pidx, 8
	%pp =l add %tmp, %poff
	storel %s, %pp
	jmp @powcolinc

@powcolinc
	%cn =l add %c, 1
	jmp @powcol

@powrowinc
	%rn =l add %r, 1
	jmp @powrow

@copy
	%q =l phi @powrow 0, @copyinc %qn
	%qcond =w csltl %q, %nn
	jnz %qcond, @copybody, @powerinc

@copybody
	%qoff =l mul %q, 8
	%qp =l add %tmp, %qoff
	%qv =l loadl %qp
	%qout =l add %out, %qoff
	storel %qv, %qout
	jmp @copyinc

@copyinc
	%qn =l add %q, 1
	jmp @copy

@powerinc
	%itern =l add %iter, 1
	jmp @power

@cleanup
	%z =l phi @power 0, @cleanupinc %zn
	%zcond =w csltl %z, %twonn
	jnz %zcond, @cleanupbody, @done

@cleanupbody
	%zoff =l mul %z, 8
	%zp =l add %tmp, %zoff
	storel 0, %zp
	jmp @cleanupinc

@cleanupinc
	%zn =l add %z, 1
	jmp @cleanup

@done
	ret 0
}
