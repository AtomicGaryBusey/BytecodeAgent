export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%zero =l copy 0
	%go =w csltl %zero, %n
	jnz %go, @loop, @done

@loop
	%i =l phi @start %zero, @next %i1
	%dp =l phi @start %dst, @next %dp1
	%sp =l phi @start %src, @next %sp1
	%count =l phi @start %zero, @next %count1
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%less =w csltl %r, %x
	%inc =l extsw %less
	%count1 =l add %count, %inc
	%dp1 =l add %dp, 8
	%sp1 =l add %sp, 8
	%i1 =l add %i, 1

@next
	%more =w csltl %i1, %n
	jnz %more, @loop, @done

@done
	%out =l phi @start %zero, @next %count1
	ret %out
}
