export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @partition_init

@partition_init
	%zero =l copy 0
	%one =l copy 1
	%last =l sub %n, %one
	%lastoff =l mul %last, 16
	%lastp =l add %a, %lastoff
	%pk =l loadl %lastp
	%pvp =l add %lastp, 8
	%pv =l loadl %pvp
	jmp @loop

@loop
	%i =l phi @partition_init %zero, @no_swap %i, @do_swap %inext
	%j =l phi @partition_init %zero, @no_swap %jnext, @do_swap %jnext
	%more =w csltl %j, %last
	jnz %more, @loop_body, @partition_done

@loop_body
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jvp =l add %jp, 8
	%jv =l loadl %jvp
	%kless =w csltl %jk, %pk
	%keq =w ceql %jk, %pk
	%vless =w csltl %jv, %pv
	%tie =w and %keq, %vless
	%less =w or %kless, %tie
	%jnext =l add %j, %one
	jnz %less, @do_swap, @no_swap

@no_swap
	jmp @loop

@do_swap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%inext =l add %i, %one
	jmp @loop

@partition_done
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ivp2 =l add %ip2, 8
	%iv2 =l loadl %ivp2
	%lk =l loadl %lastp
	%lvp =l add %lastp, 8
	%lv =l loadl %lvp
	storel %lk, %ip2
	storel %lv, %ivp2
	storel %ik2, %lastp
	storel %iv2, %lvp
	%rightn =l sub %last, %i
	%ri =l add %i, %one
	%rioff =l mul %ri, 16
	%rightp =l add %a, %rioff
	%leftmore =w csltl %one, %i
	jnz %leftmore, @call_left, @after_left

@call_left
	%leftret =l call $qsort_pairs(l %a, l %i)
	jmp @after_left

@after_left
	%rightmore =w csltl %one, %rightn
	jnz %rightmore, @call_right, @done

@call_right
	%rightret =l call $qsort_pairs(l %rightp, l %rightn)
	jmp @done

@done
	ret 0
}
