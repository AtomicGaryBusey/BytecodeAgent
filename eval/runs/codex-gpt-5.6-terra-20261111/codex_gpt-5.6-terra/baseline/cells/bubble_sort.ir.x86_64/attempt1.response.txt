export function l $bubble_sort(l %a, l %n) {
@start
	%one =l copy 1
	%limit =l sub %n, %one
	jmp @outer

@outer
	%i =l phi @start 0, @innerdone %inext
	%outercond =w csltl %i, %limit
	jnz %outercond, @inner, @done

@inner
	%j =l phi @outer 0, @inc %jnext
	%bound =l sub %limit, %i
	%innercond =w csltl %j, %bound
	jnz %innercond, @compare, @innerdone

@compare
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csltl %y, %x
	jnz %swap, @doswap, @inc

@doswap
	storel %y, %p
	storel %x, %q
	jmp @inc

@inc
	%jnext =l add %j, 1
	jmp @inner

@innerdone
	%inext =l add %i, 1
	jmp @outer

@done
	ret 0
}
