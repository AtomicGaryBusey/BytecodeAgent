export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @outer

@outer
	%i =l phi @start 0, @outerinc %inext
	%last =l sub %n, 1
	%more =w csltl %i, %last
	jnz %more, @innerinit, @done

@innerinit
	%limit =l sub %last, %i
	jmp @inner

@inner
	%j =l phi @innerinit 0, @advance %jnext
	%inside =w csltl %j, %limit
	jnz %inside, @compare, @outerinc

@compare
	%off =l shl %j, 3
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csltl %y, %x
	jnz %swap, @doswap, @advance

@doswap
	storel %y, %p
	storel %x, %q
	jmp @advance

@advance
	%jnext =l add %j, 1
	jmp @inner

@outerinc
	%inext =l add %i, 1
	jmp @outer

@done
	ret 0
}
