export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%count =l phi @start 0, @body %countnext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%inc =l extuw %gt
	%countnext =l add %count, %inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
}
