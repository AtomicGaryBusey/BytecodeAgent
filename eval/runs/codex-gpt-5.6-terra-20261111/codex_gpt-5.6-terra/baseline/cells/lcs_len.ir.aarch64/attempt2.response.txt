export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%limit =l add %n, 1
	%zj =l copy 0
	jmp @zero

@zero
	%zok =w csltl %zj, %limit
	jnz %zok, @zerobody, @zerodone

@zerobody
	%zoff =l mul %zj, 8
	%zptr =l add %row, %zoff
	storel 0, %zptr
	%znext =l add %zj, 1
	%zj =l copy %znext
	jmp @zero

@zerodone
	jmp @outer

@outer
	%i =l phi @zerodone 0, @innerdone %inext
	%iok =w csltl %i, %m
	jnz %iok, @outerbody, @done

@outerbody
	%xptr =l add %x, %i
	%xb =w loadub %xptr
	jmp @inner

@inner
	%j =l phi @outerbody 1, @after %jnext
	%diag =l phi @outerbody 0, @after %up
	%jok =w csltl %j, %limit
	jnz %jok, @innerbody, @innerdone

@innerbody
	%roff =l mul %j, 8
	%rptr =l add %row, %roff
	%up =l loadl %rptr
	%jm1 =l sub %j, 1
	%yptr =l add %y, %jm1
	%yb =w loadub %yptr
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch

@match
	%inc =l add %diag, 1
	storel %inc, %rptr
	jmp @after

@nomatch
	%loff =l mul %jm1, 8
	%lptr =l add %row, %loff
	%left =l loadl %lptr
	%gt =w csgtl %up, %left
	jnz %gt, @useup, @useleft

@useup
	storel %up, %rptr
	jmp @after

@useleft
	storel %left, %rptr
	jmp @after

@after
	%jnext =l add %j, 1
	jmp @inner

@innerdone
	%inext =l add %i, 1
	jmp @outer

@done
	%aoff =l mul %n, 8
	%aptr =l add %row, %aoff
	%ans =l loadl %aptr
	ret %ans
}
