export function l $isqrt(l %n) {
@start
	%init =l copy 4611686018427387904
	jmp @align

@align
	%bit =l phi @start %init, @alignshift %bitnext
	%small =w cultl %n, %bit
	jnz %small, @alignshift, @mainentry

@alignshift
	%bitnext =l shr %bit, 2
	jmp @align

@mainentry
	%zero =l copy 0
	jmp @loop

@loop
	%b =l phi @mainentry %bit, @take %btake, @skip %bskip
	%r =l phi @mainentry %zero, @take %rtake, @skip %rskip
	%rem =l phi @mainentry %n, @take %remtake, @skip %remskip
	%done =w ceql %b, 0
	jnz %done, @finish, @body

@body
	%trial =l add %r, %b
	%less =w cultl %rem, %trial
	jnz %less, @skip, @take

@take
	%remtake =l sub %rem, %trial
	%rhalf =l shr %r, 1
	%rtake =l add %rhalf, %b
	%btake =l shr %b, 2
	jmp @loop

@skip
	%remskip =l copy %rem
	%rskip =l shr %r, 1
	%bskip =l shr %b, 2
	jmp @loop

@finish
	ret %r
}
