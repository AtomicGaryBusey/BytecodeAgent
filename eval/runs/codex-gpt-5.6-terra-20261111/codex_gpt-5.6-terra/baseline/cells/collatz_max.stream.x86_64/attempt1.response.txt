collatz_max:
xor eax, eax
mov esi, 1
.Louter:
cmp rsi, rdi
jg .Ldone
mov rdx, rsi
xor ecx, ecx
.Linner:
cmp rdx, 1
je .Lcount_done
test rdx, 1
jz .Leven
lea rdx, [rdx + rdx*2 + 1]
inc rcx
jmp .Linner
.Leven:
shr rdx, 1
inc rcx
jmp .Linner
.Lcount_done:
cmp rcx, rax
cmovg rax, rcx
inc rsi
jmp .Louter
.Ldone:
ret
