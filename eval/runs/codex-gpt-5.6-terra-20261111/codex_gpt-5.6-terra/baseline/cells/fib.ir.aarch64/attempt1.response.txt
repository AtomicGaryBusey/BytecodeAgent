export function l $fib(l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @loop

@zero
	ret 0

@loop
	%a =l phi @start 0, @body %b
	%b =l phi @start 1, @body %next
	%i =l phi @start 1, @body %inc
	%done =w ceql %i, %n
	jnz %done, @end, @body

@body
	%next =l add %a, %b
	%inc =l add %i, 1
	jmp @loop

@end
	ret %b
}
