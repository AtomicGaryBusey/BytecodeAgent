export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%si =l alloc8 8
	%sj =l alloc8 8
	%st =l alloc8 8
	%ssum =l alloc8 8
	%srem =l alloc8 8
	%sci =l alloc8 8
	%count =l mul %n, %n
	%bytes =l mul %count, 8
	%prod =l add %tmp, %bytes
	storel %k, %srem
	storel 0, %si
	jmp @zero_out_loop

@zero_out_loop
	%zoi =l loadl %si
	%zoc =w csltl %zoi, %count
	jnz %zoc, @zero_out_body, @diag_init

@zero_out_body
	%zoo =l mul %zoi, 8
	%zop =l add %out, %zoo
	storel 0, %zop
	%zoin =l add %zoi, 1
	storel %zoin, %si
	jmp @zero_out_loop

@diag_init
	storel 0, %si
	jmp @diag_loop

@diag_loop
	%di =l loadl %si
	%dc =w csltl %di, %n
	jnz %dc, @diag_body, @copy_base_init

@diag_body
	%dr =l mul %di, %n
	%didx =l add %dr, %di
	%doff =l mul %didx, 8
	%dp =l add %out, %doff
	storel 1, %dp
	%din =l add %di, 1
	storel %din, %si
	jmp @diag_loop

@copy_base_init
	storel 0, %si
	jmp @copy_base_loop

@copy_base_loop
	%cbi =l loadl %si
	%cbc =w csltl %cbi, %count
	jnz %cbc, @copy_base_body, @power_check

@copy_base_body
	%cbo =l mul %cbi, 8
	%cbap =l add %a, %cbo
	%cbv =l loadl %cbap
	%cbtp =l add %tmp, %cbo
	storel %cbv, %cbtp
	%cbin =l add %cbi, 1
	storel %cbin, %si
	jmp @copy_base_loop

@power_check
	%rem =l loadl %srem
	%done =w ceql %rem, 0
	jnz %done, @cleanup_init, @product_i_init

@product_i_init
	storel 0, %si
	jmp @product_i_loop

@product_i_loop
	%pi =l loadl %si
	%pic =w csltl %pi, %n
	jnz %pic, @product_j_init, @copy_product_init

@product_j_init
	storel 0, %sj
	jmp @product_j_loop

@product_j_loop
	%pj =l loadl %sj
	%pjc =w csltl %pj, %n
	jnz %pjc, @product_t_init, @product_i_next

@product_t_init
	storel 0, %st
	storel 0, %ssum
	jmp @product_t_loop

@product_t_loop
	%pt =l loadl %st
	%ptc =w csltl %pt, %n
	jnz %ptc, @product_t_body, @product_store

@product_t_body
	%plr =l mul %pi, %n
	%plidx =l add %plr, %pt
	%ploff =l mul %plidx, 8
	%plp =l add %out, %ploff
	%plv =l loadl %plp
	%prr =l mul %pt, %n
	%pridx =l add %prr, %pj
	%proff =l mul %pridx, 8
	%prp =l add %tmp, %proff
	%prv =l loadl %prp
	%pmul =l mul %plv, %prv
	%psum =l loadl %ssum
	%pnewsum =l add %psum, %pmul
	storel %pnewsum, %ssum
	%ptn =l add %pt, 1
	storel %ptn, %st
	jmp @product_t_loop

@product_store
	%psr =l mul %pi, %n
	%psidx =l add %psr, %pj
	%psoff =l mul %psidx, 8
	%psp =l add %prod, %psoff
	%psv =l loadl %ssum
	storel %psv, %psp
	%pjn =l add %pj, 1
	storel %pjn, %sj
	jmp @product_j_loop

@product_i_next
	%pin =l add %pi, 1
	storel %pin, %si
	jmp @product_i_loop

@copy_product_init
	storel 0, %si
	jmp @copy_product_loop

@copy_product_loop
	%cpi =l loadl %si
	%cpc =w csltl %cpi, %count
	jnz %cpc, @copy_product_body, @power_next

@copy_product_body
	%cpo =l mul %cpi, 8
	%cpp =l add %prod, %cpo
	%cpv =l loadl %cpp
	%cop =l add %out, %cpo
	storel %cpv, %cop
	%cpin =l add %cpi, 1
	storel %cpin, %si
	jmp @copy_product_loop

@power_next
	%oldrem =l loadl %srem
	%newrem =l sub %oldrem, 1
	storel %newrem, %srem
	jmp @power_check

@cleanup_init
	%twocount =l mul %count, 2
	storel 0, %sci
	jmp @cleanup_loop

@cleanup_loop
	%ci =l loadl %sci
	%cc =w csltl %ci, %twocount
	jnz %cc, @cleanup_body, @return

@cleanup_body
	%co =l mul %ci, 8
	%cp =l add %tmp, %co
	storel 0, %cp
	%cin =l add %ci, 1
	storel %cin, %sci
	jmp @cleanup_loop

@return
	ret 0
}
