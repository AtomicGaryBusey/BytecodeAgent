export function l $shift_sat(l %x, l %n) {
@start
	%small =w csltl %n, 64
	jnz %small, @smallshift, @largeshift
@smallshift
	%left =l shl %x, %n
	%right =l sar %x, %n
	%result =l add %left, %right
	ret %result
@largeshift
	%negative =w csltl %x, 0
	jnz %negative, @neg, @pos
@neg
	ret -1
@pos
	ret 0
}
