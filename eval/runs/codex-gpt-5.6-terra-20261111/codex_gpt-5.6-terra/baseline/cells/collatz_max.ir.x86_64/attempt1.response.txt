export function l $collatz_max(l %n) {
@start
	%pi =l alloc8 8
	%px =l alloc8 8
	%ps =l alloc8 8
	%pm =l alloc8 8
	storel 1, %pi
	storel 0, %pm
	jmp @outer

@outer
	%i =l loadl %pi
	%done =w csltl %n, %i
	jnz %done, @finish, @begin

@begin
	storel %i, %px
	storel 0, %ps
	jmp @inner

@inner
	%x =l loadl %px
	%atone =w ceql %x, 1
	jnz %atone, @innerdone, @step

@step
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @oddstep, @evenstep

@oddstep
	%triple =l mul %x, 3
	%nextodd =l add %triple, 1
	storel %nextodd, %px
	jmp @count

@evenstep
	%nexteven =l div %x, 2
	storel %nexteven, %px
	jmp @count

@count
	%s =l loadl %ps
	%snext =l add %s, 1
	storel %snext, %ps
	jmp @inner

@innerdone
	%steps =l loadl %ps
	%max =l loadl %pm
	%better =w csltl %max, %steps
	jnz %better, @setmax, @advance

@setmax
	storel %steps, %pm
	jmp @advance

@advance
	%inext =l add %i, 1
	storel %inext, %pi
	jmp @outer

@finish
	%result =l loadl %pm
	ret %result
}
