mat_pow:
push rbx
push r12
push r13
push r14
push r15
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbx, r8
xor r9d, r9d
.Linit_i:
cmp r9, r13
jge .Lpower_check
xor r10d, r10d
.Linit_j:
cmp r10, r13
jge .Linit_next_i
mov rax, r9
imul rax, r13
add rax, r10
cmp r9, r10
jne .Linit_zero
mov qword ptr [r15 + rax*8], 1
jmp .Linit_next_j
.Linit_zero:
mov qword ptr [r15 + rax*8], 0
.Linit_next_j:
inc r10
jmp .Linit_j
.Linit_next_i:
inc r9
jmp .Linit_i
.Lpower_check:
test r14, r14
jz .Lclear_tmp
xor r9d, r9d
.Lmul_i:
cmp r9, r13
jge .Lcopy_product
xor r10d, r10d
.Lmul_j:
cmp r10, r13
jge .Lmul_next_i
xor r11d, r11d
xor ecx, ecx
.Lmul_t:
cmp rcx, r13
jge .Lmul_store
mov rax, r9
imul rax, r13
add rax, rcx
mov rdx, qword ptr [r15 + rax*8]
mov rax, rcx
imul rax, r13
add rax, r10
imul rdx, qword ptr [r12 + rax*8]
add r11, rdx
inc rcx
jmp .Lmul_t
.Lmul_store:
mov rax, r9
imul rax, r13
add rax, r10
mov qword ptr [rbx + rax*8], r11
inc r10
jmp .Lmul_j
.Lmul_next_i:
inc r9
jmp .Lmul_i
.Lcopy_product:
mov rax, r13
imul rax, r13
xor r9d, r9d
.Lcopy_loop:
cmp r9, rax
jge .Lpower_next
mov rdx, qword ptr [rbx + r9*8]
mov qword ptr [r15 + r9*8], rdx
inc r9
jmp .Lcopy_loop
.Lpower_next:
dec r14
jmp .Lpower_check
.Lclear_tmp:
mov rax, r13
imul rax, r13
add rax, rax
xor ecx, ecx
.Lclear_loop:
cmp rcx, rax
jge .Ldone
mov qword ptr [rbx + rcx*8], 0
inc rcx
jmp .Lclear_loop
.Ldone:
xor eax, eax
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
