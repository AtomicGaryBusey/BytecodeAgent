export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%zero =w ceql %n, 0
	jnz %zero, @done, @direction

@direction
	%forward =w cultl %dst, %src
	jnz %forward, @forward_init, @backward_init

@forward_init
	%dp =l copy %dst
	%sp =l copy %src
	%rem =l copy %n
	jmp @forward_loop

@forward_loop
	%v =l loadl %sp
	storel %v, %dp
	%dp =l add %dp, 8
	%sp =l add %sp, 8
	%rem =l sub %rem, 1
	%more =w cnel %rem, 0
	jnz %more, @forward_loop, @done

@backward_init
	%off =l mul %n, 8
	%dpb =l add %dst, %off
	%spb =l add %src, %off
	%remb =l copy %n
	jmp @backward_loop

@backward_loop
	%dpb =l sub %dpb, 8
	%spb =l sub %spb, 8
	%vb =l loadl %spb
	storel %vb, %dpb
	%remb =l sub %remb, 1
	%moreb =w cnel %remb, 0
	jnz %moreb, @backward_loop, @done

@done
	ret 0
}
