export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
	jmp @loop
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%gt =w csltl %max, %x
	jnz %gt, @update, @next
@update
	%max1 =l copy %x
	jmp @next
@next
	%max2 =l phi @body %max, @update %max1
	%i1 =l add %i, 1
	%i =l copy %i1
	%max =l copy %max2
	jmp @loop
@done
	ret %max
}
