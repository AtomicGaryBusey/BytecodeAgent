export function l $bmax(l %a, l %n) {
@start
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %i1
	%max =l phi @start %first, @next %max1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%gt =w csltl %max, %x
	jnz %gt, @update, @next
@update
	jmp @next
@next
	%max1 =l phi @body %max, @update %x
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %max
}
