export function l $reverse(l %a, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @init
@init
	%last =l sub %n, 1
	%off =l mul %last, 8
	%left =l copy %a
	%right =l add %a, %off
	jmp @loop
@loop
	%go =w csltl %left, %right
	jnz %go, @swap, @done
@swap
	%x =l loadl %left
	%y =l loadl %right
	storel %y, %left
	storel %x, %right
	%left =l add %left, 8
	%right =l sub %right, 8
	jmp @loop
@done
	ret 0
}
