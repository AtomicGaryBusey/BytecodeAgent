export function l $map_sum(l %a, l %n, l %f) {
@start
	%z =l copy 0
	%c =w csltl %z, %n
	jnz %c, @loop, @done

@loop
	%p =l phi @start %a, @loop %pn
	%i =l phi @start 0, @loop %in
	%s =l phi @start 0, @loop %sn
	%x =l loadl %p
	%r =l call %f(l %x)
	%sn =l add %s, %r
	%pn =l add %p, 8
	%in =l add %i, 1
	%c =w csltl %in, %n
	jnz %c, @loop, @ret

@ret
	ret %sn

@done
	ret 0
}
