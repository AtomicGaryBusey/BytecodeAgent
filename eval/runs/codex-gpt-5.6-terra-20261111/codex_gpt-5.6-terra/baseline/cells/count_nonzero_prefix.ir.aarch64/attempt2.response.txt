export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %inext
	%more =w csltl %i, %n
	jnz %more, @load, @done
@load
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%nz =w cnel %v, 0
	jnz %nz, @next, @done
@next
	%inext =l add %i, 1
	jmp @loop
@done
	ret %i
}
