export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
@loop
	%more =w csltl %i, %n
	jnz %more, @load, @done
@load
	%p =l add %a, %i
	%p =l add %p, %i
	%p =l add %p, %i
	%p =l add %p, %i
	%v =l loadl %p
	%nz =w cnel %v, 0
	jnz %nz, @next, @done
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %i
}
