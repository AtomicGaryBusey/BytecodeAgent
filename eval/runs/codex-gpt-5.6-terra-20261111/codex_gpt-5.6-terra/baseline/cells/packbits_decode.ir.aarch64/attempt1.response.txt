export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%zero =l copy 0
	jmp @head

@head
	%si =l phi @start %zero, @noop %sih, @lit_full_done %lend, @rep_full_done %rend
	%oi =l phi @start %zero, @noop %nooi, @lit_full_done %oend, @rep_full_done %oend2
	%more =w csltl %si, %n
	jnz %more, @read_head, @success

@read_head
	%hp =l add %src, %si
	%h =w loadsb %hp
	%sih =l add %si, 1
	%isnoop =w ceqw %h, -128
	jnz %isnoop, @noop, @classify

@noop
	%nooi =l copy %oi
	jmp @head

@classify
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat

@literal
	%cntw =w add %h, 1
	%cnt =l extsw %cntw
	%lend =l add %sih, %cnt
	%badsrc =w csgtl %lend, %n
	jnz %badsrc, @src_error, @lit_space

@lit_space
	%oend =l add %oi, %cnt
	%over =w csgtl %oend, %cap
	jnz %over, @lit_partial, @lit_full

@lit_full
	jmp @lit_full_loop

@lit_full_loop
	%lk =l phi @lit_full %zero, @lit_full_body %lk1
	%lgo =w csltl %lk, %cnt
	jnz %lgo, @lit_full_body, @lit_full_done

@lit_full_body
	%lsp =l add %sih, %lk
	%lsp2 =l add %src, %lsp
	%lv =w loadub %lsp2
	%lop =l add %oi, %lk
	%lop2 =l add %out, %lop
	storeb %lv, %lop2
	%lk1 =l add %lk, 1
	jmp @lit_full_loop

@lit_full_done
	jmp @head

@lit_partial
	%avail =l sub %cap, %oi
	jmp @lit_part_loop

@lit_part_loop
	%pk =l phi @lit_partial %zero, @lit_part_body %pk1
	%pgo =w csltl %pk, %avail
	jnz %pgo, @lit_part_body, @out_error

@lit_part_body
	%ps =l add %sih, %pk
	%ps2 =l add %src, %ps
	%pv =w loadub %ps2
	%po =l add %oi, %pk
	%po2 =l add %out, %po
	storeb %pv, %po2
	%pk1 =l add %pk, 1
	jmp @lit_part_loop

@repeat
	%cntw2 =w sub 1, %h
	%cnt2 =l extsw %cntw2
	%rend =l add %sih, 1
	%badsrc2 =w csgtl %rend, %n
	jnz %badsrc2, @src_error, @rep_value

@rep_value
	%rp =l add %src, %sih
	%rv =w loadub %rp
	%oend2 =l add %oi, %cnt2
	%over2 =w csgtl %oend2, %cap
	jnz %over2, @rep_partial, @rep_full

@rep_full
	jmp @rep_full_loop

@rep_full_loop
	%rk =l phi @rep_full %zero, @rep_full_body %rk1
	%rgo =w csltl %rk, %cnt2
	jnz %rgo, @rep_full_body, @rep_full_done

@rep_full_body
	%ro =l add %oi, %rk
	%ro2 =l add %out, %ro
	storeb %rv, %ro2
	%rk1 =l add %rk, 1
	jmp @rep_full_loop

@rep_full_done
	jmp @head

@rep_partial
	%avail2 =l sub %cap, %oi
	jmp @rep_part_loop

@rep_part_loop
	%qk =l phi @rep_partial %zero, @rep_part_body %qk1
	%qgo =w csltl %qk, %avail2
	jnz %qgo, @rep_part_body, @out_error

@rep_part_body
	%qo =l add %oi, %qk
	%qo2 =l add %out, %qo
	storeb %rv, %qo2
	%qk1 =l add %qk, 1
	jmp @rep_part_loop

@src_error
	%err2 =l copy -2
	ret %err2

@out_error
	%err1 =l copy -1
	ret %err1

@success
	ret %oi
}
