bubble_sort:
xor eax, eax
cmp rsi, 1
jle .Ldone
.Louter:
xor ecx, ecx
mov rdx, 1
.Linner:
mov r8, qword ptr [rdi + rcx*8]
mov r9, qword ptr [rdi + rdx*8]
cmp r8, r9
jle .Lnoswap
mov qword ptr [rdi + rcx*8], r9
mov qword ptr [rdi + rdx*8], r8
.Lnoswap:
inc rcx
inc rdx
cmp rdx, rsi
jl .Linner
dec rsi
cmp rsi, 1
jg .Louter
.Ldone:
ret
