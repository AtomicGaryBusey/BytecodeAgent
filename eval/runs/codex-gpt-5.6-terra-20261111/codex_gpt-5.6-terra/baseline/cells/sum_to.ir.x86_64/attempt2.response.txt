export function l $sum_to(l %n) {
@start
	%ok =w csltl 0, %n
	jnz %ok, @loop, @zero

@zero
	ret 0

@loop
	%i =l phi 1, @start, %next, @body
	%sum =l phi 0, @start, %nextsum, @body
	%done =w csltl %n, %i
	jnz %done, @end, @body

@body
	%nextsum =l add %sum, %i
	%next =l add %i, 1
	jmp @loop

@end
	ret %sum
}
