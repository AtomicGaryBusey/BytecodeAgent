export function l $sum_to(l %n) {
@start
	%ok = csltl 0, %n
	jnz %ok, @loop, @zero

@zero
	ret 0

@loop
	%i = phi l 1, @start, %next, @body
	%sum = phi l 0, @start, %nextsum, @body
	%done = csltl %n, %i
	jnz %done, @end, @body

@body
	%nextsum = add %sum, %i
	%next = add %i, 1
	jmp @loop

@end
	ret %sum
}
