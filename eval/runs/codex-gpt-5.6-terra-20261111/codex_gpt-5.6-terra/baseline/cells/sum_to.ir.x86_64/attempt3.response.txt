export function l $sum_to(l %n) {
@start
	%ok =w csltl 0, %n
	jnz %ok, @loop, @zero

@zero
	ret 0

@loop
	%i =l phi @start 1, @body %next
	%sum =l phi @start 0, @body %nextsum
	%done =w csltl %n, %i
	jnz %done, @end, @body

@body
	%nextsum =l add %sum, %i
	%next =l add %i, 1
	jmp @loop

@end
	ret %sum
}
