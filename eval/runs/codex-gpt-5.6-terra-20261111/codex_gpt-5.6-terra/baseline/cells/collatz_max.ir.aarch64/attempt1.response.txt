export function l $collatz_max(l %n) {
@entry
	jmp @outer

@outer
	%i =l phi @entry 1, @outer_join %inext
	%max =l phi @entry 0, @outer_join %maxnext
	%more =w cslel %i, %n
	jnz %more, @inner, @done

@inner
	%x =l phi @outer %i, @step %xnext
	%steps =l phi @outer 0, @step %stepsnext
	%finished =w ceql %x, 1
	jnz %finished, @outer_inc, @calc

@calc
	%bit =l and %x, 1
	%even =w ceql %bit, 0
	jnz %even, @even, @odd

@even
	%half =l div %x, 2
	jmp @step

@odd
	%triple =l mul %x, 3
	%plus =l add %triple, 1
	jmp @step

@step
	%xnext =l phi @even %half, @odd %plus
	%stepsnext =l add %steps, 1
	jmp @inner

@outer_inc
	%better =w csltl %max, %steps
	jnz %better, @take, @keep

@take
	jmp @outer_join

@keep
	jmp @outer_join

@outer_join
	%maxnext =l phi @take %steps, @keep %max
	%inext =l add %i, 1
	jmp @outer

@done
	ret %max
}
