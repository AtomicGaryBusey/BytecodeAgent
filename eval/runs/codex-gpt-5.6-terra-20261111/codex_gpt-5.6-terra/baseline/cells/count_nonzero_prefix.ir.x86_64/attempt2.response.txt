export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop

@loop
	%i =l phi @start 0, @next %inext
	%p =l phi @start %a, @next %pnext
	%more =w csltl %i, %n
	jnz %more, @check, @ret

@check
	%x =l loadl %p
	%zero =w ceql %x, 0
	jnz %zero, @ret, @next

@next
	%inext =l add %i, 1
	%pnext =l add %p, 8
	jmp @loop

@ret
	ret %i
}
