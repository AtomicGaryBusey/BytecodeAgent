export function l $crc32(l %p, l %n) {
@start
	%init =w copy -1
	jmp @bytes

@bytes
	%q =l phi @start %p, @afterbits %qnext
	%left =l phi @start %n, @afterbits %leftnext
	%crc =w phi @start %init, @afterbits %cfin
	%has =w csltl 0, %left
	jnz %has, @body, @done

@body
	%b =w loadub %q
	%cx =w xor %crc, %b
	%qnext =l add %q, 1
	%leftnext =l sub %left, 1
	jmp @bits

@bits
	%cbit =w phi @body %cx, @odd %co, @even %ce
	%k =w phi @body 8, @odd %ko, @even %ke
	%lsb =w and %cbit, 1
	jnz %lsb, @odd, @even

@odd
	%so =w shr %cbit, 1
	%co =w xor %so, -306674912
	%ko =w sub %k, 1
	%moreo =w csltw 0, %ko
	jnz %moreo, @bits, @afterbits

@even
	%ce =w shr %cbit, 1
	%ke =w sub %k, 1
	%moree =w csltw 0, %ke
	jnz %moree, @bits, @afterbits

@afterbits
	%cfin =w phi @odd %co, @even %ce
	jmp @bytes

@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
