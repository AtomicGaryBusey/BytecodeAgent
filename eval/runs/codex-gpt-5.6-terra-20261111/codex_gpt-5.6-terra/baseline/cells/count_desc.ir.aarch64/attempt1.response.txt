export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init

@zero
	ret 0

@init
	%prev0 =l loadl %a
	jmp @loop

@loop
	%i =l phi @init 1, @body %inext
	%prev =l phi @init %prev0, @body %cur
	%cnt =l phi @init 0, @body %cntnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%cur =l loadl %p
	%desc =w csltl %cur, %prev
	%d =l extuw %desc
	%cntnext =l add %cnt, %d
	%inext =l add %i, 1
	jmp @loop

@done
	ret %cnt
}
