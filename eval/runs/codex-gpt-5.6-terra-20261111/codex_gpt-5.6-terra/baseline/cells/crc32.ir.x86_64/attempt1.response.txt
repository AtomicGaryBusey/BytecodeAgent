export function l $crc32(l %p, l %n) {
@start
	jmp @outer

@outer
	%count =l phi @start %n, @outer_after %next_count
	%curp =l phi @start %p, @outer_after %nextp
	%crc =w phi @start -1, @outer_after %icrc
	%more =w csltl 0, %count
	jnz %more, @byte, @done

@byte
	%b =w loadub %curp
	%mixed =w xor %crc, %b
	jmp @bit

@bit
	%i =w phi @byte 0, @bitnext %nexti
	%icrc =w phi @byte %mixed, @bitnext %updated
	%finished =w csgew %i, 8
	jnz %finished, @outer_after, @bitnext

@bitnext
	%lsb =w and %icrc, 1
	%mask =w sub 0, %lsb
	%poly =w and %mask, -306674912
	%shifted =w shr %icrc, 1
	%updated =w xor %shifted, %poly
	%nexti =w add %i, 1
	jmp @bit

@outer_after
	%next_count =l sub %count, 1
	%nextp =l add %curp, 1
	jmp @outer

@done
	%final =w xor %crc, -1
	%result =l extuw %final
	ret %result
}
