export function l $add_overflow(l %a, l %b, l %out) {
@start
	%sum =l add %a, %b
	storel %sum, %out
	%aneg =w csltl %a, 0
	jnz %aneg, @aneg, @apos

@aneg
	%bneg =w csltl %b, 0
	jnz %bneg, @bothneg, @no

@bothneg
	%sneg =w csltl %sum, 0
	jnz %sneg, @no, @yes

@apos
	%bneg2 =w csltl %b, 0
	jnz %bneg2, @no, @bothpos

@bothpos
	%sneg2 =w csltl %sum, 0
	jnz %sneg2, @yes, @no

@yes
	ret 1

@no
	ret 0
}
