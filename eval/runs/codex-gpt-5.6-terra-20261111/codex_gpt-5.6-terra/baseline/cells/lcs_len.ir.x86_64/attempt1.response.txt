export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%n1 =l add %n, 1
	jmp @zero

@zero
	%k =l phi @start, 0, @zerobody, %knext
	%zmore =w csltl %k, %n1
	jnz %zmore, @zerobody, @outer

@zerobody
	%koff =l shl %k, 3
	%kp =l add %row, %koff
	storel 0, %kp
	%knext =l add %k, 1
	jmp @zero

@outer
	%i =l phi @zero, 0, @afterinner, %inext
	%imore =w csltl %i, %m
	jnz %imore, @outerbody, @done

@outerbody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner

@inner
	%j =l phi @outerbody, 1, @inextinner, %jnext
	%diag =l phi @outerbody, 0, @inextinner, %up
	%jmore =w csltl %j, %n1
	jnz %jmore, @innerbody, @afterinner

@innerbody
	%joff =l shl %j, 3
	%rp =l add %row, %joff
	%up =l loadl %rp
	%lp =l sub %rp, 8
	%left =l loadl %lp
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch

@match
	%v =l add %diag, 1
	storel %v, %rp
	jmp @inextinner

@nomatch
	%usel =w csltl %up, %left
	jnz %usel, @useleft, @useup

@useleft
	storel %left, %rp
	jmp @inextinner

@useup
	storel %up, %rp
	jmp @inextinner

@inextinner
	%jnext =l add %j, 1
	jmp @inner

@afterinner
	%inext =l add %i, 1
	jmp @outer

@done
	%noff =l shl %n, 3
	%np =l add %row, %noff
	%result =l loadl %np
	ret %result
}
