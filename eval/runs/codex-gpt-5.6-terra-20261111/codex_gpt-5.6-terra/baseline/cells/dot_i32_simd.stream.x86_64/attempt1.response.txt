dot_i32_simd:
pxor xmm0, xmm0
mov rcx, rdx
shr rcx, 2
and rdx, 3
test rcx, rcx
jz .Lreduce
.Lvector:
movdqu xmm1, xmmword ptr [rdi]
movdqu xmm2, xmmword ptr [rsi]
pmovsxdq xmm3, xmm1
pmovsxdq xmm4, xmm2
pmuldq xmm3, xmm4
psrldq xmm1, 8
psrldq xmm2, 8
pmovsxdq xmm1, xmm1
pmovsxdq xmm2, xmm2
pmuldq xmm1, xmm2
paddq xmm0, xmm3
paddq xmm0, xmm1
add rdi, 16
add rsi, 16
dec rcx
jnz .Lvector
.Lreduce:
movq rax, xmm0
psrldq xmm0, 8
movq rcx, xmm0
add rax, rcx
test rdx, rdx
jz .Ldone
.Ltail:
movsxd r8, dword ptr [rdi]
movsxd r9, dword ptr [rsi]
imul r8, r9
add rax, r8
add rdi, 4
add rsi, 4
dec rdx
jnz .Ltail
.Ldone:
ret
