export function l $shift_sat(l %x, l %n) {
@start
	%small =w csltl %n, 64
	jnz %small, @small, @large

@large
	%sign =l sar %x, 63
	ret %sign

@small
	jmp @loop

@loop
	%i =l phi @small 0, @body %nexti
	%left =l phi @small %x, @body %nextleft
	%right =l phi @small %x, @body %nextright
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%nextleft =l shl %left, 1
	%nextright =l sar %right, 1
	%nexti =l add %i, 1
	jmp @loop

@done
	%result =l add %left, %right
	ret %result
}
