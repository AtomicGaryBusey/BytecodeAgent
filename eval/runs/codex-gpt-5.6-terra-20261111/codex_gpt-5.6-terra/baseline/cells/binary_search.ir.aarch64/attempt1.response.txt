export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop

@loop
	%more =w csltl %lo, %hi
	jnz %more, @body, @notfound

@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @compare

@compare
	%less =w csltl %val, %key
	jnz %less, @right, @left

@right
	%lo =l add %mid, 1
	jmp @loop

@left
	%hi =l copy %mid
	jmp @loop

@found
	ret %mid

@notfound
	ret -1
}
