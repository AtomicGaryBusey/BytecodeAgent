export function l $max_byte(l %p, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @zero, @init
@zero
	ret 0
@init
	%first =l loadub %p
	jmp @loop
@loop
	%i =l phi @init 1, @cont %nexti
	%max =l phi @init %first, @cont %nextmax
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%q =l add %p, %i
	%b =l loadub %q
	%greater =w csgtl %b, %max
	jnz %greater, @take, @keep
@take
	jmp @cont
@keep
	jmp @cont
@cont
	%nextmax =l phi @take %b, @keep %max
	%nexti =l add %i, 1
	jmp @loop
@done
	ret %max
}
