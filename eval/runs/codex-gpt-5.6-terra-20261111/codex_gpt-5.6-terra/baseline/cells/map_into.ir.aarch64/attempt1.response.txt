export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%count =l phi @start 0, @body %count1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%sptr =l add %src, %off
	%x =l loadl %sptr
	%r =l call %f(l %x)
	%dptr =l add %dst, %off
	storel %r, %dptr
	%less =w csltl %r, %x
	%inc =l extuw %less
	%count1 =l add %count, %inc
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %count
}
