export function l $sdivrem(l %a, l %b, l %rp) {
@start
	%z =l copy 0
	%bzero =w ceql %b, %z
	jnz %bzero, @divzero, @check

@check
	%one =l copy 1
	%min =l shl %one, 63
	%ismin =w ceql %a, %min
	jnz %ismin, @checknegone, @normal

@checknegone
	%negone =l sub %z, %one
	%isnegone =w ceql %b, %negone
	jnz %isnegone, @overflow, @normal

@divzero
	storel %a, %rp
	ret %z

@overflow
	storel %z, %rp
	ret %min

@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rp
	ret %q
}
