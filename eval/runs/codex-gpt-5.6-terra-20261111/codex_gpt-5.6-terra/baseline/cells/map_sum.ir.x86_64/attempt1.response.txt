export function l $map_sum(l %a, l %n, l %f) {
@entry
	%empty =w ceql %n, 0
	jnz %empty, @done, @loop
@loop
	%i =l phi @entry 0, @loop %nexti
	%sum =l phi @entry 0, @loop %nextsum
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%nextsum =l add %sum, %r
	%nexti =l add %i, 1
	%more =w csltl %nexti, %n
	jnz %more, @loop, @exit
@exit
	ret %nextsum
@done
	ret 0
}
