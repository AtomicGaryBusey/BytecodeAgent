export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%rp =l alloc8 8
	%ip =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @dispatch

@dispatch
	%pc =l loadl %pcp
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch

@fetch
	%coff =l shl %pc, 3
	%cp =l add %code, %coff
	%op =l loadl %cp
	%c0 =w ceql %op, 0
	jnz %c0, @finish, @test1

@test1
	%c1 =w ceql %op, 1
	jnz %c1, @push, @test2

@test2
	%c2 =w ceql %op, 2
	jnz %c2, @add, @test3

@test3
	%c3 =w ceql %op, 3
	jnz %c3, @sub, @test4

@test4
	%c4 =w ceql %op, 4
	jnz %c4, @mul, @test5

@test5
	%c5 =w ceql %op, 5
	jnz %c5, @dup, @test6

@test6
	%c6 =w ceql %op, 6
	jnz %c6, @swap, @test7

@test7
	%c7 =w ceql %op, 7
	jnz %c7, @drop, @test8

@test8
	%c8 =w ceql %op, 8
	jnz %c8, @over, @test9

@test9
	%c9 =w ceql %op, 9
	jnz %c9, @lt, @test10

@test10
	%c10 =w ceql %op, 10
	jnz %c10, @jz, @test11

@test11
	%c11 =w ceql %op, 11
	jnz %c11, @jmpop, @nop

@push
	%p1 =l add %pc, 1
	%p1o =l shl %p1, 3
	%p1p =l add %code, %p1o
	%v =l loadl %p1p
	%d =l loadl %dp
	%do =l shl %d, 3
	%sp =l add %stack, %do
	storel %v, %sp
	%nd =l add %d, 1
	%np =l add %pc, 2
	storel %nd, %dp
	storel %np, %pcp
	jmp @dispatch

@add
	%d1 =l loadl %dp
	%ib =l sub %d1, 1
	%ioa =l sub %d1, 2
	%bo =l shl %ib, 3
	%ao =l shl %ioa, 3
	%bp =l add %stack, %bo
	%ap =l add %stack, %ao
	%b =l loadl %bp
	%a =l loadl %ap
	%r =l add %a, %b
	storel %r, %ap
	%nd1 =l sub %d1, 1
	%np1 =l add %pc, 1
	storel %nd1, %dp
	storel %np1, %pcp
	jmp @dispatch

@sub
	%d2 =l loadl %dp
	%ib2 =l sub %d2, 1
	%ia2 =l sub %d2, 2
	%bo2 =l shl %ib2, 3
	%ao2 =l shl %ia2, 3
	%bp2 =l add %stack, %bo2
	%ap2 =l add %stack, %ao2
	%b2 =l loadl %bp2
	%a2 =l loadl %ap2
	%r2 =l sub %a2, %b2
	storel %r2, %ap2
	%nd2 =l sub %d2, 1
	%np2 =l add %pc, 1
	storel %nd2, %dp
	storel %np2, %pcp
	jmp @dispatch

@mul
	%d3 =l loadl %dp
	%ib3 =l sub %d3, 1
	%ia3 =l sub %d3, 2
	%bo3 =l shl %ib3, 3
	%ao3 =l shl %ia3, 3
	%bp3 =l add %stack, %bo3
	%ap3 =l add %stack, %ao3
	%b3 =l loadl %bp3
	%a3 =l loadl %ap3
	%r3 =l mul %a3, %b3
	storel %r3, %ap3
	%nd3 =l sub %d3, 1
	%np3 =l add %pc, 1
	storel %nd3, %dp
	storel %np3, %pcp
	jmp @dispatch

@dup
	%d4 =l loadl %dp
	%it4 =l sub %d4, 1
	%to4 =l shl %it4, 3
	%tp4 =l add %stack, %to4
	%v4 =l loadl %tp4
	%do4 =l shl %d4, 3
	%dp4 =l add %stack, %do4
	storel %v4, %dp4
	%nd4 =l add %d4, 1
	%np4 =l add %pc, 1
	storel %nd4, %dp
	storel %np4, %pcp
	jmp @dispatch

@swap
	%d5 =l loadl %dp
	%i51 =l sub %d5, 1
	%i52 =l sub %d5, 2
	%o51 =l shl %i51, 3
	%o52 =l shl %i52, 3
	%p51 =l add %stack, %o51
	%p52 =l add %stack, %o52
	%v51 =l loadl %p51
	%v52 =l loadl %p52
	storel %v52, %p51
	storel %v51, %p52
	%np5 =l add %pc, 1
	storel %np5, %pcp
	jmp @dispatch

@drop
	%d6 =l loadl %dp
	%nd6 =l sub %d6, 1
	%np6 =l add %pc, 1
	storel %nd6, %dp
	storel %np6, %pcp
	jmp @dispatch

@over
	%d7 =l loadl %dp
	%i7 =l sub %d7, 2
	%o7 =l shl %i7, 3
	%p7 =l add %stack, %o7
	%v7 =l loadl %p7
	%do7 =l shl %d7, 3
	%dp7 =l add %stack, %do7
	storel %v7, %dp7
	%nd7 =l add %d7, 1
	%np7 =l add %pc, 1
	storel %nd7, %dp
	storel %np7, %pcp
	jmp @dispatch

@lt
	%d8 =l loadl %dp
	%ib8 =l sub %d8, 1
	%ia8 =l sub %d8, 2
	%bo8 =l shl %ib8, 3
	%ao8 =l shl %ia8, 3
	%bp8 =l add %stack, %bo8
	%ap8 =l add %stack, %ao8
	%b8 =l loadl %bp8
	%a8 =l loadl %ap8
	%c8v =w csltl %a8, %b8
	%r8 =l extuw %c8v
	storel %r8, %ap8
	%nd8 =l sub %d8, 1
	%np8 =l add %pc, 1
	storel %nd8, %dp
	storel %np8, %pcp
	jmp @dispatch

@jz
	%d9 =l loadl %dp
	%iv9 =l sub %d9, 1
	%ov9 =l shl %iv9, 3
	%pv9 =l add %stack, %ov9
	%v9 =l loadl %pv9
	%nd9 =l sub %d9, 1
	storel %nd9, %dp
	%z9 =w ceql %v9, 0
	jnz %z9, @jztake, @jzskip

@jztake
	%jp9 =l add %pc, 1
	%jo9 =l shl %jp9, 3
	%ja9 =l add %code, %jo9
	%target9 =l loadl %ja9
	storel %target9, %pcp
	jmp @dispatch

@jzskip
	%np9 =l add %pc, 2
	storel %np9, %pcp
	jmp @dispatch

@jmpop
	%jp10 =l add %pc, 1
	%jo10 =l shl %jp10, 3
	%ja10 =l add %code, %jo10
	%target10 =l loadl %ja10
	storel %target10, %pcp
	jmp @dispatch

@nop
	%np11 =l add %pc, 1
	storel %np11, %pcp
	jmp @dispatch

@finish
	%fd =l loadl %dp
	%empty =w ceql %fd, 0
	jnz %empty, @zeroresult, @loadresult

@zeroresult
	storel 0, %rp
	jmp @cleaninit

@loadresult
	%ri =l sub %fd, 1
	%ro =l shl %ri, 3
	%ra =l add %stack, %ro
	%rv =l loadl %ra
	storel %rv, %rp
	jmp @cleaninit

@cleaninit
	%ci =l loadl %dp
	storel %ci, %ip
	jmp @cleancheck

@cleancheck
	%i =l loadl %ip
	%more =w cultl %i, %cap
	jnz %more, @cleanbody, @return

@cleanbody
	%io =l shl %i, 3
	%ia =l add %stack, %io
	storel 0, %ia
	%ni =l add %i, 1
	storel %ni, %ip
	jmp @cleancheck

@return
	%retv =l loadl %rp
	ret %retv
}
