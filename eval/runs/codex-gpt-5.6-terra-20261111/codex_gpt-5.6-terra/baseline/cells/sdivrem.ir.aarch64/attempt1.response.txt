export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%z =w ceql %b, 0
	jnz %z, @zero, @check

@check
	%one =l copy 1
	%min =l shl %one, 63
	%ismin =w ceql %a, %min
	jnz %ismin, @checkneg, @normal

@checkneg
	%isneg =w ceql %b, -1
	jnz %isneg, @overflow, @normal

@zero
	storel %a, %rem
	ret 0

@overflow
	storel 0, %rem
	ret %min

@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
