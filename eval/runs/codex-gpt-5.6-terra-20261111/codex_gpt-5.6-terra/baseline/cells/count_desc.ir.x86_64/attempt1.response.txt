export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@init
	%first =l loadl %a
	jmp @loop
@loop
	%i =l phi @init 1, @inc %nexti, @keep %nexti
	%prev =l phi @init %first, @inc %cur, @keep %cur
	%count =l phi @init 0, @inc %newcount, @keep %count
	%more =w csltl %i, %n
	jnz %more, @check, @done
@check
	%off =l shl %i, 3
	%ptr =l add %a, %off
	%cur =l loadl %ptr
	%desc =w csltl %cur, %prev
	%nexti =l add %i, 1
	jnz %desc, @inc, @keep
@inc
	%newcount =l add %count, 1
	jmp @loop
@keep
	jmp @loop
@done
	ret %count
@zero
	ret 0
}
