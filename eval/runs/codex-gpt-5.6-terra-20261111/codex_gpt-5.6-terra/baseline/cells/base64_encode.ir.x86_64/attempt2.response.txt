export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%p =l phi @start %src, @groupdone %ep
	%r =l phi @start %n, @groupdone %er
	%op =l phi @start %out, @groupdone %onext
	%cnt =l phi @start 0, @groupdone %cntnext
	%short =w csltl %r, 3
	jnz %short, @tail, @full
@full
	%b0 =l loadub %p
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%x0 =l shl %b0, 16
	%x1 =l shl %b1, 8
	%x2 =l or %x0, %x1
	%xfull =l or %x2, %b2
	%pfull =l add %p, 3
	%rfull =l sub %r, 3
	jmp @emitinit
@tail
	%empty =w ceql %r, 0
	jnz %empty, @done, @tailonecheck
@tailonecheck
	%one =w ceql %r, 1
	jnz %one, @tailone, @tailtwo
@tailone
	%tb0 =l loadub %p
	%xtailone =l shl %tb0, 16
	%ptailone =l add %p, 1
	jmp @emitinit
@tailtwo
	%ub0 =l loadub %p
	%up1 =l add %p, 1
	%ub1 =l loadub %up1
	%ux0 =l shl %ub0, 16
	%ux1 =l shl %ub1, 8
	%xtailtwo =l or %ux0, %ux1
	%ptailtwo =l add %p, 2
	jmp @emitinit
@emitinit
	%ex =l phi @full %xfull, @tailone %xtailone, @tailtwo %xtailtwo
	%ep =l phi @full %pfull, @tailone %ptailone, @tailtwo %ptailtwo
	%er =l phi @full %rfull, @tailone 0, @tailtwo 0
	%valid =w phi @full 4, @tailone 2, @tailtwo 3
	jmp @emitloop
@emitloop
	%cur =l phi @emitinit %op, @advance %onext
	%cc =l phi @emitinit %cnt, @advance %cntnext
	%shift =w phi @emitinit 18, @advance %shiftnext
	%real =w phi @emitinit %valid, @advance %realnext
	%slots =w phi @emitinit 4, @advance %slotsnext
	%hasreal =w csltw 0, %real
	jnz %hasreal, @encode, @padding
@encode
	%q =l shr %ex, %shift
	%idx =l and %q, 63
	%lt26 =w csltl %idx, 26
	jnz %lt26, @upper, @check52
@upper
	%chu =l add %idx, 65
	jmp @putchar
@check52
	%lt52 =w csltl %idx, 52
	jnz %lt52, @lower, @check62
@lower
	%chl =l add %idx, 71
	jmp @putchar
@check62
	%lt62 =w csltl %idx, 62
	jnz %lt62, @digit, @special
@digit
	%chd =l sub %idx, 4
	jmp @putchar
@special
	%is62 =w ceql %idx, 62
	jnz %is62, @plus, @slash
@plus
	%chp =l copy 43
	jmp @putchar
@slash
	%chs =l copy 47
	jmp @putchar
@padding
	%chpad =l copy 61
	jmp @putchar
@putchar
	%ch =l phi @upper %chu, @lower %chl, @digit %chd, @plus %chp, @slash %chs, @padding %chpad
	storeb %ch, %cur
	jmp @advance
@advance
	%onext =l add %cur, 1
	%cntnext =l add %cc, 1
	%shiftnext =w sub %shift, 6
	%realnext =w sub %real, 1
	%slotsnext =w sub %slots, 1
	%more =w csltw 0, %slotsnext
	jnz %more, @emitloop, @groupdone
@groupdone
	jmp @loop
@done
	ret %cnt
}
