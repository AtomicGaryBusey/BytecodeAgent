export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%p =l phi @start %src, @emitdone %ep
	%r =l phi @start %n, @emitdone %er
	%op =l phi @start %out, @emitdone %curNext
	%cnt =l phi @start 0, @emitdone %cntNext
	%short =w csltl %r, 3
	jnz %short, @tail, @full
@full
	%b0 =l loadub %p
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%x0 =l shl %b0, 16
	%x1 =l shl %b1, 8
	%x2 =l or %x0, %x1
	%xfull =l or %x2, %b2
	%pfull =l add %p, 3
	%rfull =l sub %r, 3
	jmp @emitinit
@tail
	%zero =w ceql %r, 0
	jnz %zero, @done, @tailcheck1
@tailcheck1
	%one =w ceql %r, 1
	jnz %one, @tail1, @tail2
@tail1
	%tb0 =l loadub %p
	%xtail1 =l shl %tb0, 16
	%ptail1 =l add %p, 1
	jmp @emitinit
@tail2
	%ub0 =l loadub %p
	%up1 =l add %p, 1
	%ub1 =l loadub %up1
	%ux0 =l shl %ub0, 16
	%ux1 =l shl %ub1, 8
	%xtail2 =l or %ux0, %ux1
	%ptail2 =l add %p, 2
	jmp @emitinit
@emitinit
	%ex =l phi @full %xfull, @tail1 %xtail1, @tail2 %xtail2
	%ep =l phi @full %pfull, @tail1 %ptail1, @tail2 %ptail2
	%er =l phi @full %rfull, @tail1 0, @tail2 0
	%chars =l phi @full 4, @tail1 2, @tail2 3
	jmp @emitloop
@emitloop
	%cur =l phi @emitinit %op, @put %curNext
	%cc =l phi @emitinit %cnt, @put %cntNext
	%shift =l phi @emitinit 18, @put %shiftNext
	%left =l phi @emitinit %chars, @put %leftNext
	%q =l shrl %ex, %shift
	%idx =l and %q, 63
	%lt26 =w csltl %idx, 26
	jnz %lt26, @mupper, @mcheck52
@mupper
	%chu =l add %idx, 65
	jmp @putchar
@mcheck52
	%lt52 =w csltl %idx, 52
	jnz %lt52, @mlower, @mcheck62
@mlower
	%chl =l add %idx, 71
	jmp @putchar
@mcheck62
	%lt62 =w csltl %idx, 62
	jnz %lt62, @mdigit, @mspecial
@mdigit
	%chd =l sub %idx, 4
	jmp @putchar
@mspecial
	%is62 =w ceql %idx, 62
	jnz %is62, @mplus, @mslash
@mplus
	%chp =l copy 43
	jmp @putchar
@mslash
	%chs =l copy 47
	jmp @putchar
@putchar
	%ch =l phi @mupper %chu, @mlower %chl, @mdigit %chd, @mplus %chp, @mslash %chs
	storeb %ch, %cur
	jmp @put
@put
	%curNext =l add %cur, 1
	%cntNext =l add %cc, 1
	%shiftNext =l sub %shift, 6
	%leftNext =l sub %left, 1
	%more =w csltl 0, %leftNext
	jnz %more, @emitloop, @emitdone
@emitdone
	jmp @loop
@done
	ret %cnt
}
