export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%a0 =l and %a, %mask
	%a1 =l shr %a, 32
	%b0 =l and %b, %mask
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%h00 =l shr %p00, 32
	%l01 =l and %p01, %mask
	%l10 =l and %p10, %mask
	%s0 =l add %h00, %l01
	%c0 =l shr %s0, 32
	%s0l =l and %s0, %mask
	%s1 =l add %s0l, %l10
	%c1 =l shr %s1, 32
	%carry =l add %c0, %c1
	%h01 =l shr %p01, 32
	%h10 =l shr %p10, 32
	%r0 =l add %p11, %h01
	%r1 =l add %r0, %h10
	%r2 =l add %r1, %carry
	ret %r2
}
