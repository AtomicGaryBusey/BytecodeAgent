export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l shl 1, 32
	jmp @loop

@loop
	%more =w cultl %lo, %hi
	jnz %more, @body, @done

@body
	%diff =l sub %hi, %lo
	%half =l shr %diff, 1
	%mid =l add %lo, %half
	%sq =l mul %mid, %mid
	%too_high =w cultl %n, %sq
	jnz %too_high, @lower, @higher

@lower
	%hi =l copy %mid
	jmp @loop

@higher
	%lo =l add %mid, 1
	jmp @loop

@done
	%result =l sub %lo, 1
	ret %result
}
