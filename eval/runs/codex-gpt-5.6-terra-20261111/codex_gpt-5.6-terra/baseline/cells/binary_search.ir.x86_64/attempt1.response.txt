export function l $binary_search(l %a, l %n, l %key) {
@start
	%hi0 =l sub %n, 1
	jmp @loop

@loop
	%lo =l phi @start 0, @right %lo1, @left %lo
	%hi =l phi @start %hi0, @right %hi, @left %hi1
	%ok =w cslel %lo, %hi
	jnz %ok, @body, @notfound

@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @compare

@compare
	%less =w csltl %val, %key
	jnz %less, @right, @left

@right
	%lo1 =l add %mid, 1
	jmp @loop

@left
	%hi1 =l sub %mid, 1
	jmp @loop

@found
	ret %mid

@notfound
	ret -1
}
