export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%c =l phi @start 0, @body %cnext
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%gt =w cultl %t, %x
	%g =l extuw %gt
	%cnext =l add %c, %g
	%inext =l add %i, 1
	jmp @loop
@end
	ret %c
}
