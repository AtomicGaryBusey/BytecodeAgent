export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @direction

@direction
	%backward =w cultl %src, %dst
	jnz %backward, @backinit, @fwdinit

@fwdinit
	jmp @fwdloop

@fwdloop
	%fi =l phi @fwdinit 0, @fwdloop %fnext
	%foff =l mul %fi, 8
	%fsp =l add %src, %foff
	%fdp =l add %dst, %foff
	%fv =l loadl %fsp
	storel %fv, %fdp
	%fnext =l add %fi, 1
	%fmore =w cultl %fnext, %n
	jnz %fmore, @fwdloop, @done

@backinit
	jmp @backloop

@backloop
	%bi =l phi @backinit %n, @backloop %bnext
	%bnext =l sub %bi, 1
	%boff =l mul %bnext, 8
	%bsp =l add %src, %boff
	%bdp =l add %dst, %boff
	%bv =l loadl %bsp
	storel %bv, %bdp
	%bmore =w cnel %bnext, 0
	jnz %bmore, @backloop, @done

@done
	ret 0
}
