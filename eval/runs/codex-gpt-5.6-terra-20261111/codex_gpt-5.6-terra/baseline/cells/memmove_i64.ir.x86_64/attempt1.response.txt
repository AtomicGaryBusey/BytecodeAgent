export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%empty =w ceql %n, 0
	jnz %empty, @done, @direction

@direction
	%backward =w cultl %src, %dst
	jnz %backward, @backinit, @fwdinit

@fwdinit
	jmp @fwdloop

@fwdloop
	%i =l phi @fwdinit 0, @fwdloop %next
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%next =l add %i, 1
	%more =w cultl %next, %n
	jnz %more, @fwdloop, @done

@backinit
	jmp @backloop

@backloop
	%i =l phi @backinit %n, @backloop %nextb
	%nextb =l sub %i, 1
	%offb =l mul %nextb, 8
	%spb =l add %src, %offb
	%dpb =l add %dst, %offb
	%vb =l loadl %spb
	storel %vb, %dpb
	%moreb =w cnel %nextb, 0
	jnz %moreb, @backloop, @done

@done
	ret 0
}
