export function l $fill_down(l %a, l %n, l %start) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %next
	%p =l phi @entry %a, @body %nextp
	%v =l phi @entry %start, @body %nextv
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	storel %v, %p
	%next =l add %i, 1
	%nextp =l add %p, 8
	%nextv =l sub %v, 1
	jmp @loop
@done
	ret %n
}
