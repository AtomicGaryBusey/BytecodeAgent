export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32

	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p01hi =l shr %p01, 32
	%p10lo =l and %p10, 4294967295
	%p10hi =l shr %p10, 32

	%t =l add %p00hi, %p01lo
	%c1 =l shr %t, 32
	%tlo =l and %t, 4294967295
	%u =l add %tlo, %p10lo
	%c2 =l shr %u, 32

	%r0 =l add %p11, %p01hi
	%r1 =l add %r0, %p10hi
	%r2 =l add %r1, %c1
	%r3 =l add %r2, %c2
	ret %r3
}
