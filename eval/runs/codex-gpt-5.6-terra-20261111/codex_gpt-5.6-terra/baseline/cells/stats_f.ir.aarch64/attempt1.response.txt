export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%b0 =l and %y0, 1
	%o0 =w cnel %b0, 0
	%c0 =l extuw %o0
	jmp @loop

@loop
	%i =l phi @start 1, @latch %inext
	%mn =l phi @start %y0, @latch %mnnext
	%mx =l phi @start %y0, @latch %mxnext
	%sum =l phi @start %y0, @latch %sumnext
	%cnt =l phi @start %c0, @latch %cntnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %mn
	jnz %lt, @setmin, @keepmin

@setmin
	jmp @minjoin

@keepmin
	jmp @minjoin

@minjoin
	%mnnew =l phi @setmin %y, @keepmin %mn
	%gt =w csltl %mx, %y
	jnz %gt, @setmax, @keepmax

@setmax
	jmp @maxjoin

@keepmax
	jmp @maxjoin

@maxjoin
	%mxnew =l phi @setmax %y, @keepmax %mx
	%sumnew =l add %sum, %y
	%bit =l and %y, 1
	%odd =w cnel %bit, 0
	%odd64 =l extuw %odd
	%cntnew =l add %cnt, %odd64
	jmp @latch

@latch
	%inext =l add %i, 1
	%mnnext =l copy %mnnew
	%mxnext =l copy %mxnew
	%sumnext =l copy %sumnew
	%cntnext =l copy %cntnew
	jmp @loop

@done
	storel %mn, %out
	%out1 =l add %out, 8
	storel %mx, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
