export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
@loop
	%more =w csltl %i, %j
	jnz %more, @swap, @done
@swap
	%ioff =l shl %i, 3
	%joff =l shl %j, 3
	%ip =l add %a, %ioff
	%jp =l add %a, %joff
	%x =l loadl %ip
	%y =l loadl %jp
	storel %y, %ip
	storel %x, %jp
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@done
	ret 0
}
