export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%b0 =l and %y0, 1
	%o0 =w cnel %b0, 0
	%c0 =l extuw %o0
	jmp @loop

@loop
	%i =l phi @start 1, @tail %inext
	%mn =l phi @start %y0, @tail %mnnext
	%mx =l phi @start %y0, @tail %mxnext
	%sm =l phi @start %y0, @tail %smnext
	%ct =l phi @start %c0, @tail %ctnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%lt =w csltl %y, %mn
	jnz %lt, @minyes, @minno

@minyes
	jmp @minjoin

@minno
	jmp @minjoin

@minjoin
	%mnnext =l phi @minyes %y, @minno %mn
	%gt =w csltl %mx, %y
	jnz %gt, @maxyes, @maxno

@maxyes
	jmp @maxjoin

@maxno
	jmp @maxjoin

@maxjoin
	%mxnext =l phi @maxyes %y, @maxno %mx
	%smnext =l add %sm, %y
	%b =l and %y, 1
	%o =w cnel %b, 0
	%ol =l extuw %o
	%ctnext =l add %ct, %ol
	%inext =l add %i, 1
	jmp @tail

@tail
	jmp @loop

@done
	storel %mn, %out
	%out1 =l add %out, 8
	storel %mx, %out1
	%out2 =l add %out, 16
	storel %sm, %out2
	ret %ct
}
