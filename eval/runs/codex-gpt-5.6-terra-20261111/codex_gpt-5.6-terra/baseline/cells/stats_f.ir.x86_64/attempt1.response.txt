export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%b0 =l and %y0, 1
	%o0 =w cnel %b0, 0
	%c0 =l extuw %o0
	jmp @loop

@loop
	%i =l phi @start 1, @body %inext
	%mn =l phi @start %y0, @body %mnnext
	%mx =l phi @start %y0, @body %mxnext
	%sm =l phi @start %y0, @body %smnext
	%ct =l phi @start %c0, @body %ctnext
	%more =w csltl %i, %n
	jnz %more, @body, @done

@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%lt =w csltl %y, %mn
	%mnnext =l sel %lt, %y, %mn
	%gt =w csltl %mx, %y
	%mxnext =l sel %gt, %y, %mx
	%smnext =l add %sm, %y
	%b =l and %y, 1
	%o =w cnel %b, 0
	%ol =l extuw %o
	%ctnext =l add %ct, %ol
	%inext =l add %i, 1
	jmp @loop

@done
	storel %mn, %out
	%out1 =l add %out, 8
	storel %mx, %out1
	%out2 =l add %out, 16
	storel %sm, %out2
	ret %ct
}
