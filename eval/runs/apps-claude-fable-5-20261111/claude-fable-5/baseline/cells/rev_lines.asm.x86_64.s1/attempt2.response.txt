.text
.globl _start
_start:
        xor     ebx, ebx                # total bytes read
        lea     r8, [rip + inbuf]
        lea     r9, [rip + outbuf]
.read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jle     .read_done
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [r8 + rbx]
        syscall
        test    rax, rax
        jle     .read_done
        add     rbx, rax
        jmp     .read_loop
.read_done:
        # process: rsi = input index, rdi = output index, rbx = total
        xor     esi, esi
        xor     edi, edi
.line_loop:
        cmp     rsi, rbx
        jae     .flush
        # find end of line: rcx scans from rsi
        mov     rcx, rsi
.find_nl:
        cmp     rcx, rbx
        jae     .got_end
        cmp     byte ptr [r8 + rcx], 0x0A
        je      .got_end
        inc     rcx
        jmp     .find_nl
.got_end:
        # copy [rsi, rcx) reversed into outbuf at rdi
        mov     rdx, rcx
.rev_loop:
        cmp     rdx, rsi
        jbe     .rev_done
        dec     rdx
        mov     al, byte ptr [r8 + rdx]
        mov     byte ptr [r9 + rdi], al
        inc     rdi
        jmp     .rev_loop
.rev_done:
        mov     byte ptr [r9 + rdi], 0x0A
        inc     rdi
        lea     rsi, [rcx + 1]          # skip the newline (or past end)
        jmp     .line_loop
.flush:
        # write rdi bytes from outbuf
        xor     ebx, ebx                # written so far
.write_loop:
        mov     rdx, rdi
        sub     rdx, rbx
        jle     .done
        lea     rsi, [r9 + rbx]
        push    rdi
        mov     eax, 1                  # write
        mov     edi, 1                  # fd 1
        syscall
        pop     rdi
        test    rax, rax
        jle     .done
        add     rbx, rax
        jmp     .write_loop
.done:
        mov     eax, 231                # exit_group
        xor     edi, edi
        syscall

.bss
.balign 16
inbuf:
        .zero   65536
outbuf:
        .zero   65544
