.text
.globl _start

_start:
    mov     x20, #0                     // total bytes read
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x23, hexchars
    add     x23, x23, :lo12:hexchars
    adrp    x22, linebuf
    add     x22, x22, :lo12:linebuf
    mov     x24, #65536

rdloop:
    subs    x2, x24, x20                // remaining capacity
    b.le    rddone
    mov     x0, #0                      // fd = stdin
    add     x1, x19, x20
    mov     x8, #63                     // read
    svc     #0
    cmp     x0, #0
    b.le    rddone
    add     x20, x20, x0
    b       rdloop

rddone:
    mov     x21, #0                     // current offset

rowloop:
    cmp     x21, x20
    b.hs    final
    sub     x25, x20, x21               // bytes left
    cmp     x25, #16
    b.lo    1f
    mov     x25, #16
1:
    mov     x9, x22                     // p = linebuf
    // 8 hex digits of offset
    mov     w10, #28
offloop:
    lsr     w11, w21, w10
    and     w11, w11, #15
    ldrb    w11, [x23, w11, uxtw]
    strb    w11, [x9], #1
    subs    w10, w10, #4
    b.ge    offloop
    mov     w11, #0x20
    strb    w11, [x9], #1
    strb    w11, [x9], #1
    // 16 hex slots
    mov     x12, #0
hexloop:
    cmp     x12, x25
    b.hs    hexpad
    add     x13, x21, x12
    ldrb    w14, [x19, x13]
    lsr     w15, w14, #4
    ldrb    w15, [x23, w15, uxtw]
    strb    w15, [x9], #1
    and     w15, w14, #15
    ldrb    w15, [x23, w15, uxtw]
    strb    w15, [x9], #1
    mov     w15, #0x20
    strb    w15, [x9], #1
    b       hexnext
hexpad:
    mov     w15, #0x20
    strb    w15, [x9], #1
    strb    w15, [x9], #1
    strb    w15, [x9], #1
hexnext:
    cmp     x12, #7
    b.ne    2f
    mov     w15, #0x20                  // extra space after 8th slot
    strb    w15, [x9], #1
2:
    add     x12, x12, #1
    cmp     x12, #16
    b.lo    hexloop
    // " |"
    mov     w15, #0x20
    strb    w15, [x9], #1
    mov     w15, #0x7c
    strb    w15, [x9], #1
    // ASCII column
    mov     x12, #0
ascloop:
    cmp     x12, x25
    b.hs    ascdone
    add     x13, x21, x12
    ldrb    w14, [x19, x13]
    cmp     w14, #0x20
    b.lo    3f
    cmp     w14, #0x7e
    b.ls    4f
3:
    mov     w14, #0x2e                  // '.'
4:
    strb    w14, [x9], #1
    add     x12, x12, #1
    b       ascloop
ascdone:
    mov     w15, #0x7c                  // '|'
    strb    w15, [x9], #1
    mov     w15, #0x0a                  // '\n'
    strb    w15, [x9], #1
    // write the line
    sub     x2, x9, x22
    mov     x1, x22
    bl      write_all
    add     x21, x21, x25
    b       rowloop

final:
    mov     x9, x22
    mov     w10, #28
5:
    lsr     w11, w20, w10
    and     w11, w11, #15
    ldrb    w11, [x23, w11, uxtw]
    strb    w11, [x9], #1
    subs    w10, w10, #4
    b.ge    5b
    mov     w11, #0x0a
    strb    w11, [x9], #1
    mov     x1, x22
    mov     x2, #9
    bl      write_all
    mov     x0, #0
    mov     x8, #93                     // exit
    svc     #0

// write_all: x1 = buf, x2 = len (clobbers x0, x1, x2, x8)
write_all:
6:
    mov     x0, #1                      // fd = stdout
    mov     x8, #64                     // write
    svc     #0
    cmp     x0, #0
    b.le    7f
    add     x1, x1, x0
    subs    x2, x2, x0
    b.gt    6b
7:
    ret

.section .rodata
hexchars:
    .ascii  "0123456789abcdef"

.bss
    .balign 8
inbuf:
    .space  65536
linebuf:
    .space  96
