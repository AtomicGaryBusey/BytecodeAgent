.text
.globl _start

_start:
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x2, #0x0400
    movk    x2, #1, lsl #16         // 66560 = buffer size
    sub     x2, x2, x20
    cmp     x2, #0
    b.le    read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    mov     x21, #0                 // pos = 0
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.ne    fail
    // success: print "ok\n"
    adrp    x1, okmsg
    add     x1, x1, :lo12:okmsg
    mov     x0, #1
    mov     x2, #3
    mov     x8, #64                 // write
    svc     #0
    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0

// ---------------------------------------------------------------
// skip whitespace (space, tab, LF, CR)
skip_ws:
1:  cmp     x21, x20
    b.hs    2f
    ldrb    w0, [x19, x21]
    cmp     w0, #0x20
    b.eq    3f
    cmp     w0, #0x09
    b.eq    3f
    cmp     w0, #0x0A
    b.eq    3f
    cmp     w0, #0x0D
    b.ne    2f
3:  add     x21, x21, #1
    b       1b
2:  ret

// ---------------------------------------------------------------
// match literal tail: x0 = nul-terminated expected bytes
match_lit:
1:  ldrb    w1, [x0], #1
    cbz     w1, 2f
    cmp     x21, x20
    b.hs    fail
    ldrb    w2, [x19, x21]
    cmp     w1, w2
    b.ne    fail
    add     x21, x21, #1
    b       1b
2:  ret

// ---------------------------------------------------------------
parse_value:
    stp     x29, x30, [sp, #-16]!
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7B               // '{'
    b.eq    pv_obj
    cmp     w0, #0x5B               // '['
    b.eq    pv_arr
    cmp     w0, #0x22               // '"'
    b.eq    pv_str
    cmp     w0, #0x2D               // '-'
    b.eq    pv_num
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pv_num
    cmp     w0, #0x74               // 't'
    b.eq    pv_true
    cmp     w0, #0x66               // 'f'
    b.eq    pv_false
    cmp     w0, #0x6E               // 'n'
    b.eq    pv_null
    b       fail
pv_obj:
    bl      parse_object
    b       pv_done
pv_arr:
    bl      parse_array
    b       pv_done
pv_str:
    bl      parse_string
    b       pv_done
pv_num:
    bl      parse_number
    b       pv_done
pv_true:
    add     x21, x21, #1
    adrp    x0, lit_true
    add     x0, x0, :lo12:lit_true
    bl      match_lit
    b       pv_done
pv_false:
    add     x21, x21, #1
    adrp    x0, lit_false
    add     x0, x0, :lo12:lit_false
    bl      match_lit
    b       pv_done
pv_null:
    add     x21, x21, #1
    adrp    x0, lit_null
    add     x0, x0, :lo12:lit_null
    bl      match_lit
pv_done:
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x21, x21, #1            // consume '{'
    bl      skip_ws
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7D               // '}'
    b.ne    po_member
    add     x21, x21, #1
    b       po_done
po_member:
    cmp     w0, #0x22               // member key must be a string
    b.ne    fail
    bl      parse_string
    bl      skip_ws
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x3A               // ':'
    b.ne    fail
    add     x21, x21, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x7D               // '}'
    b.eq    po_close
    cmp     w0, #0x2C               // ','
    b.ne    fail
    add     x21, x21, #1
    bl      skip_ws
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    b       po_member
po_close:
    add     x21, x21, #1
po_done:
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x21, x21, #1            // consume '['
    bl      skip_ws
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x5D               // ']'
    b.ne    pa_loop
    add     x21, x21, #1
    b       pa_done
pa_loop:
    bl      parse_value
    bl      skip_ws
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x5D               // ']'
    b.eq    pa_close
    cmp     w0, #0x2C               // ','
    b.ne    fail
    add     x21, x21, #1
    bl      skip_ws
    b       pa_loop
pa_close:
    add     x21, x21, #1
pa_done:
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
parse_string:
    add     x21, x21, #1            // consume opening '"'
ps_loop:
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x22               // '"'
    b.eq    ps_end
    cmp     w0, #0x5C               // '\'
    b.eq    ps_esc
    cmp     w0, #0x20
    b.lo    fail                    // control bytes forbidden
    add     x21, x21, #1
    b       ps_loop
ps_end:
    add     x21, x21, #1
    ret
ps_esc:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x75               // 'u'
    b.eq    ps_u
    cmp     w0, #0x22               // '"'
    b.eq    ps_ok1
    cmp     w0, #0x5C               // '\'
    b.eq    ps_ok1
    cmp     w0, #0x2F               // '/'
    b.eq    ps_ok1
    cmp     w0, #0x62               // 'b'
    b.eq    ps_ok1
    cmp     w0, #0x66               // 'f'
    b.eq    ps_ok1
    cmp     w0, #0x6E               // 'n'
    b.eq    ps_ok1
    cmp     w0, #0x72               // 'r'
    b.eq    ps_ok1
    cmp     w0, #0x74               // 't'
    b.eq    ps_ok1
    b       fail
ps_ok1:
    add     x21, x21, #1
    b       ps_loop
ps_u:
    add     x21, x21, #1
    mov     w3, #4
ps_hex:
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30           // '0'-'9'?
    cmp     w1, #9
    b.ls    ps_hexok
    orr     w1, w0, #0x20           // to lowercase
    sub     w1, w1, #0x61           // 'a'-'f'?
    cmp     w1, #5
    b.hi    fail
ps_hexok:
    add     x21, x21, #1
    subs    w3, w3, #1
    b.ne    ps_hex
    b       ps_loop

// ---------------------------------------------------------------
parse_number:
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2D               // '-'
    b.ne    pn_int
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
pn_int:
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail
    cmp     w0, #0x30               // '0'
    b.eq    pn_zero
pn_digits:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    pn_ret
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pn_digits
    b       pn_frac
pn_zero:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    pn_ret
    ldrb    w0, [x19, x21]
pn_frac:
    cmp     w0, #0x2E               // '.'
    b.ne    pn_expq
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail
pn_fdig:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    pn_ret
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pn_fdig
pn_expq:
    orr     w1, w0, #0x20
    cmp     w1, #0x65               // 'e' / 'E'
    b.ne    pn_ret
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
    cmp     w0, #0x2B               // '+'
    b.eq    pn_sign
    cmp     w0, #0x2D               // '-'
    b.ne    pn_edig0
pn_sign:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    fail
    ldrb    w0, [x19, x21]
pn_edig0:
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail
pn_edig:
    add     x21, x21, #1
    cmp     x21, x20
    b.hs    pn_ret
    ldrb    w0, [x19, x21]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pn_edig
pn_ret:
    ret

// ---------------------------------------------------------------
// error exit: print "error at N\n" (N = x21), exit 1. Never returns.
fail:
    sub     sp, sp, #32
    mov     x4, x21
    mov     x5, sp
    mov     x6, #10
1:  udiv    x7, x4, x6
    msub    x8, x7, x6, x4
    add     x8, x8, #0x30
    strb    w8, [x5], #1
    mov     x4, x7
    cbnz    x4, 1b
    adrp    x0, errbuf
    add     x0, x0, :lo12:errbuf
    adrp    x1, errmsg
    add     x1, x1, :lo12:errmsg
    mov     x2, #9
2:  ldrb    w3, [x1], #1
    strb    w3, [x0], #1
    subs    x2, x2, #1
    b.ne    2b
3:  ldrb    w3, [x5, #-1]!
    strb    w3, [x0], #1
    cmp     x5, sp
    b.ne    3
