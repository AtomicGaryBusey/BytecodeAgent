.text
.globl _start
_start:
	# read all of stdin into buf
	xor	ebx, ebx		# total bytes read
rdloop:
	mov	edx, 65536
	sub	edx, ebx
	jz	rddone
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + buf]
	add	rsi, rbx
	syscall
	test	rax, rax
	jle	rddone
	add	rbx, rax
	jmp	rdloop
rddone:
	# find first '!' (0x21): program = [0, r12), input = [r14ofs, rbx)
	xor	ecx, ecx
findbang:
	cmp	rcx, rbx
	jae	nobang
	cmp	byte ptr [buf + rcx], 33
	je	gotbang
	inc	rcx
	jmp	findbang
nobang:
	mov	r12, rbx		# program length = all input
	mov	r14, rbx		# input start = end (empty)
	jmp	brackets
gotbang:
	mov	r12, rcx		# program length
	lea	r14, [rcx + 1]		# input starts after '!'
brackets:
	# bracket matching: build jump table, fail if unbalanced
	xor	ecx, ecx		# index
	xor	edx, edx		# stack depth
brloop:
	cmp	rcx, r12
	jae	brdone
	mov	al, byte ptr [buf + rcx]
	cmp	al, 91			# '['
	je	brpush
	cmp	al, 93			# ']'
	je	brpop
brnext:
	inc	rcx
	jmp	brloop
brpush:
	mov	dword ptr [bstack + rdx*4], ecx
	inc	edx
	jmp	brnext
brpop:
	test	edx, edx
	jz	fail
	dec	edx
	mov	edi, dword ptr [bstack + rdx*4]
	mov	dword ptr [jmptab + rcx*4], edi
	mov	dword ptr [jmptab + rdi*4], ecx
	jmp	brnext
brdone:
	test	edx, edx
	jnz	fail

	# set up execution
	lea	r13, [rip + tape]	# tape pointer
	lea	rsi, [rip + buf]
	lea	rdi, [rsi + rbx]	# rdi = input end
	add	rsi, r14		# rsi = input pointer
	xor	ecx, ecx		# pc
	xor	ebp, ebp		# output count
execloop:
	cmp	rcx, r12
	jae	done
	movzx	eax, byte ptr [buf + rcx]
	cmp	al, 62			# '>'
	je	c_right
	cmp	al, 60			# '<'
	je	c_left
	cmp	al, 43			# '+'
	je	c_plus
	cmp	al, 45			# '-'
	je	c_minus
	cmp	al, 91			# '['
	je	c_open
	cmp	al, 93			# ']'
	je	c_close
	cmp	al, 46			# '.'
	je	c_dot
	cmp	al, 44			# ','
	je	c_comma
step:
	inc	rcx
	jmp	execloop
c_right:
	inc	r13
	jmp	step
c_left:
	dec	r13
	jmp	step
c_plus:
	inc	byte ptr [r13]
	jmp	step
c_minus:
	dec	byte ptr [r13]
	jmp	step
c_open:
	cmp	byte ptr [r13], 0
	jne	step
	mov	ecx, dword ptr [jmptab + rcx*4]
	jmp	step
c_close:
	cmp	byte ptr [r13], 0
	je	step
	mov	ecx, dword ptr [jmptab + rcx*4]
	jmp	step
c_dot:
	mov	al, byte ptr [r13]
	mov	byte ptr [outbuf + rbp], al
	inc	rbp
	jmp	step
c_comma:
	xor	eax, eax
	cmp	rsi, rdi
	jae	1f
	mov	al, byte ptr [rsi]
	inc	rsi
1:
	mov	byte ptr [r13], al
	jmp	step

done:
	# flush output buffer
	lea	rsi, [rip + outbuf]
	mov	rdx, rbp
wrloop:
	test	rdx, rdx
	jz	exit0
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	syscall
	test	rax, rax
	jle	exit0
	add	rsi, rax
	sub	rdx, rax
	jmp	wrloop
exit0:
	mov	eax, 60			# exit
	xor	edi, edi
	syscall
fail:
	mov	eax, 60			# exit
	mov	edi, 1
	syscall

.bss
.balign 8
buf:
	.zero	65536
outbuf:
	.zero	65536
.balign 4
jmptab:
	.zero	262144
bstack:
	.zero	4096
tape:
	.zero	30000
