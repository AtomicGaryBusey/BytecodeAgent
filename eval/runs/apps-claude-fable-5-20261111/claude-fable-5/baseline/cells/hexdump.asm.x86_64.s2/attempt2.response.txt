.text
.globl _start

_start:
	xor	r12, r12		# total bytes read
read_loop:
	mov	edx, 65536
	sub	edx, r12d
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + inbuf]
	add	rsi, r12
	syscall
	test	rax, rax
	jle	read_done
	add	r12, rax
	cmp	r12, 65536
	jb	read_loop
read_done:
	xor	r13, r13		# current offset

row_loop:
	cmp	r13, r12
	jae	rows_done

	lea	rdi, [rip + linebuf]
	mov	eax, r13d
	call	hex8
	mov	word ptr [rdi], 0x2020	# two spaces
	add	rdi, 2

	mov	r14, r12
	sub	r14, r13		# remaining bytes
	cmp	r14, 16
	jbe	1f
	mov	r14d, 16
1:
	lea	rsi, [rip + inbuf]
	add	rsi, r13		# base of this row
	xor	ebx, ebx		# i = 0
hex_loop:
	cmp	rbx, r14
	jae	pad_slot
	movzx	eax, byte ptr [rsi + rbx]
	mov	edx, eax
	shr	edx, 4
	mov	dl, byte ptr [hexd + rdx]
	mov	byte ptr [rdi], dl
	mov	edx, eax
	and	edx, 15
	mov	dl, byte ptr [hexd + rdx]
	mov	byte ptr [rdi + 1], dl
	mov	byte ptr [rdi + 2], 0x20
	jmp	slot_next
pad_slot:
	mov	byte ptr [rdi], 0x20
	mov	byte ptr [rdi + 1], 0x20
	mov	byte ptr [rdi + 2], 0x20
slot_next:
	add	rdi, 3
	cmp	ebx, 7
	jne	2f
	mov	byte ptr [rdi], 0x20	# extra space after 8th slot
	inc	rdi
2:
	inc	ebx
	cmp	ebx, 16
	jb	hex_loop

	mov	byte ptr [rdi], 0x20
	mov	byte ptr [rdi + 1], 0x7c	# '|'
	add	rdi, 2

	xor	ebx, ebx
asc_loop:
	cmp	rbx, r14
	jae	asc_done
	movzx	eax, byte ptr [rsi + rbx]
	cmp	al, 0x20
	jb	asc_dot
	cmp	al, 0x7e
	jbe	asc_store
asc_dot:
	mov	al, 0x2e		# '.'
asc_store:
	mov	byte ptr [rdi], al
	inc	rdi
	inc	ebx
	jmp	asc_loop
asc_done:
	mov	byte ptr [rdi], 0x7c	# '|'
	mov	byte ptr [rdi + 1], 0x0a
	add	rdi, 2

	lea	rsi, [rip + linebuf]
	mov	rdx, rdi
	sub	rdx, rsi
	mov	eax, 1			# write
	mov	edi, 1
	syscall

	add	r13, 16
	jmp	row_loop

rows_done:
	lea	rdi, [rip + linebuf]
	mov	eax, r12d
	call	hex8
	mov	byte ptr [rdi], 0x0a
	lea	rsi, [rip + linebuf]
	mov	edx, 9
	mov	eax, 1			# write
	mov	edi, 1
	syscall

	mov	eax, 231		# exit_group
	xor	edi, edi
	syscall

# hex8: write eax as 8 lowercase hex digits to [rdi], advance rdi by 8
hex8:
	mov	ecx, 8
1:
	rol	eax, 4
	mov	edx, eax
	and	edx, 15
	mov	dl, byte ptr [hexd + rdx]
	mov	byte ptr [rdi], dl
	inc	rdi
	dec	ecx
	jnz	1b
	ret

.section .rodata
hexd:
	.ascii	"0123456789abcdef"

.bss
	.balign	16
inbuf:
	.space	65536
linebuf:
	.space	96
