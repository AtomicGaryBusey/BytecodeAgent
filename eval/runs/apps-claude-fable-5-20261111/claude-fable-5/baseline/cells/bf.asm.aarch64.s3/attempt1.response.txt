.bss
.balign 8
inbuf:  .space 65536
jump:   .space 262144
bstack: .space 262144
tape:   .space 30016
outb:   .space 16

.text
.globl _start
_start:
        // ---- read all of stdin into inbuf ----
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, #0                 // total bytes read
read_loop:
        mov     x2, #65536
        sub     x2, x2, x20
        cbz     x2, read_done
        add     x1, x19, x20
        mov     x0, #0
        mov     x8, #63                 // read
        svc     #0
        cmp     x0, #0
        b.le    read_done
        add     x20, x20, x0
        b       read_loop
read_done:
        // ---- find first '!' ----
        mov     x21, #0
find_bang:
        cmp     x21, x20
        b.ge    no_bang
        ldrb    w0, [x19, x21]
        cmp     w0, #0x21
        b.eq    found_bang
        add     x21, x21, #1
        b       find_bang
found_bang:
        add     x23, x19, x21
        add     x23, x23, #1            // program input start
        add     x24, x19, x20           // program input end
        b       brackets
no_bang:
        add     x23, x19, x20           // empty input
        mov     x24, x23
brackets:
        // x21 = program length
        adrp    x25, jump
        add     x25, x25, :lo12:jump
        adrp    x4, bstack
        add     x4, x4, :lo12:bstack
        mov     x5, #0                  // stack depth
        mov     x6, #0                  // scan position
bm_loop:
        cmp     x6, x21
        b.ge    bm_done
        ldrb    w0, [x19, x6]
        cmp     w0, #91                 // '['
        b.eq    bm_open
        cmp     w0, #93                 // ']'
        b.eq    bm_close
bm_next:
        add     x6, x6, #1
        b       bm_loop
bm_open:
        str     w6, [x4, x5, lsl #2]
        add     x5, x5, #1
        b       bm_next
bm_close:
        cbz     x5, fail
        sub     x5, x5, #1
        ldr     w7, [x4, x5, lsl #2]
        str     w6, [x25, x7, lsl #2]   // jump[open]  = close
        str     w7, [x25, x6, lsl #2]   // jump[close] = open
        b       bm_next
bm_done:
        cbnz    x5, fail

        // ---- run ----
        adrp    x22, tape
        add     x22, x22, :lo12:tape    // data pointer
        mov     x20, #0                 // pc
loop:
        cmp     x20, x21
        b.ge    done
        ldrb    w0, [x19, x20]
        cmp     w0, #62                 // '>'
        b.eq    c_gt
        cmp     w0, #60                 // '<'
        b.eq    c_lt
        cmp     w0, #43                 // '+'
        b.eq    c_plus
        cmp     w0, #45                 // '-'
        b.eq    c_minus
        cmp     w0, #46                 // '.'
        b.eq    c_dot
        cmp     w0, #44                 // ','
        b.eq    c_comma
        cmp     w0, #91                 // '['
        b.eq    c_open
        cmp     w0, #93                 // ']'
        b.eq    c_close
next:
        add     x20, x20, #1
        b       loop
c_gt:
        add     x22, x22, #1
        b       next
c_lt:
        sub     x22, x22, #1
        b       next
c_plus:
        ldrb    w0, [x22]
        add     w0, w0, #1
        strb    w0, [x22]
        b       next
c_minus:
        ldrb    w0, [x22]
        sub     w0, w0, #1
        strb    w0, [x22]
        b       next
c_dot:
        adrp    x1, outb
        add     x1, x1, :lo12:outb
        ldrb    w0, [x22]
        strb    w0, [x1]
        mov     x0, #1
        mov     x2, #1
        mov     x8, #64                 // write
        svc     #0
        b       next
c_comma:
        cmp     x23, x24
        b.ge    1f
        ldrb    w0, [x23], #1
        strb    w0, [x22]
        b       next
1:
        strb    wzr, [x22]
        b       next
c_open:
        ldrb    w0, [x22]
        cbnz    w0, next
        ldr     w20, [x25, x20, lsl #2]
        b       next
c_close:
        ldrb    w0, [x22]
        cbz     w0, next
        ldr     w20, [x25, x20, lsl #2]
        b       next

done:
        mov     x0, #0
        mov     x8, #93                 // exit
        svc     #0
fail:
        mov     x0, #1
        mov     x8, #93                 // exit
        svc     #0
        b       fail
