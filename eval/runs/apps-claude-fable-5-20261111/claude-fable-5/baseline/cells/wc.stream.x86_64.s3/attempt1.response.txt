	.text
	.globl _start
_start:
	# r12 = in_word flag, r13 = lines, r14 = words, r15 = bytes
	xor	r12d, r12d
	xor	r13d, r13d
	xor	r14d, r14d
	xor	r15d, r15d

read_loop:
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + buf]
	mov	edx, 65536
	syscall
	test	rax, rax
	jle	done			# 0 = EOF, negative = error
	add	r15, rax
	lea	rsi, [rip + buf]
	xor	ebx, ebx
byte_loop:
	movzx	ecx, byte ptr [rsi + rbx]
	cmp	cl, 0x0a
	jne	1f
	inc	r13
1:
	cmp	cl, 0x20
	je	is_ws
	lea	edx, [ecx - 9]
	cmp	edx, 4			# 0x09..0x0d
	jbe	is_ws
	test	r12, r12
	jnz	next_byte
	mov	r12d, 1
	inc	r14
	jmp	next_byte
is_ws:
	xor	r12d, r12d
next_byte:
	inc	rbx
	cmp	rbx, rax
	jb	byte_loop
	jmp	read_loop

done:
	lea	rdi, [rip + outbuf]
	mov	rax, r13
	call	itoa
	mov	byte ptr [rdi], 0x20
	inc	rdi
	mov	rax, r14
	call	itoa
	mov	byte ptr [rdi], 0x20
	inc	rdi
	mov	rax, r15
	call	itoa
	mov	byte ptr [rdi], 0x0a
	inc	rdi

	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	syscall

	mov	eax, 231		# exit_group
	xor	edi, edi
	syscall

# itoa: rax = value, rdi = destination; advances rdi past digits
itoa:
	lea	rsi, [rip + tmp + 31]
	mov	r9, 10
1:
	xor	edx, edx
	div	r9
	add	dl, '0'
	mov	byte ptr [rsi], dl
	dec	rsi
	test	rax, rax
	jnz	1b
	inc	rsi
	lea	rdx, [rip + tmp + 32]
2:
	mov	al, byte ptr [rsi]
	mov	byte ptr [rdi], al
	inc	rsi
	inc	rdi
	cmp	rsi, rdx
	jb	2b
	ret

	.bss
	.balign 16
buf:
	.space	65536
outbuf:
	.space	80
tmp:
	.space	32
