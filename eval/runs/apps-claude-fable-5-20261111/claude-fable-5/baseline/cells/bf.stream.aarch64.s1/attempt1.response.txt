.text
.globl _start

_start:
	// read all of stdin into buf
	adrp	x19, buf
	add	x19, x19, :lo12:buf
	mov	x24, #0			// total bytes read
read_loop:
	mov	x2, #65536
	sub	x2, x2, x24		// remaining space
	cbz	x2, read_done
	mov	x0, #0			// fd 0
	add	x1, x19, x24
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x24, x24, x0
	b	read_loop
read_done:

	// find first '!' (0x21): program = [0, x20), input starts at x23
	mov	x20, #0
find_bang:
	cmp	x20, x24
	b.ge	no_bang
	ldrb	w0, [x19, x20]
	cmp	w0, #0x21
	b.eq	found_bang
	add	x20, x20, #1
	b	find_bang
found_bang:
	add	x23, x20, #1		// input start
	b	split_done
no_bang:
	mov	x23, x24		// empty input
split_done:

	// bracket matching: build jump table, check balance
	adrp	x21, jumptab
	add	x21, x21, :lo12:jumptab
	sub	sp, sp, #4096		// bracket position stack (1000 x 4 bytes)
	mov	x9, #0			// depth
	mov	x10, #0			// scan index
scan:
	cmp	x10, x20
	b.ge	scan_end
	ldrb	w0, [x19, x10]
	cmp	w0, #91			// '['
	b.eq	scan_open
	cmp	w0, #93			// ']'
	b.eq	scan_close
scan_next:
	add	x10, x10, #1
	b	scan
scan_open:
	str	w10, [sp, x9, lsl #2]
	add	x9, x9, #1
	b	scan_next
scan_close:
	cbz	x9, err
	sub	x9, x9, #1
	ldr	w11, [sp, x9, lsl #2]	// matching '[' position (zero-extended)
	str	w10, [x21, x11, lsl #2]	// jump[open] = close
	str	w11, [x21, x10, lsl #2]	// jump[close] = open
	b	scan_next
scan_end:
	cbnz	x9, err
	add	sp, sp, #4096

	// interpret
	adrp	x22, tape
	add	x22, x22, :lo12:tape
	mov	x25, #0			// pc
	mov	x26, #0			// data pointer index
	mov	x27, x23		// input index
iloop:
	cmp	x25, x20
	b.ge	done
	ldrb	w0, [x19, x25]
	cmp	w0, #62			// '>'
	b.eq	c_gt
	cmp	w0, #60			// '<'
	b.eq	c_lt
	cmp	w0, #43			// '+'
	b.eq	c_plus
	cmp	w0, #45			// '-'
	b.eq	c_minus
	cmp	w0, #91			// '['
	b.eq	c_lbr
	cmp	w0, #93			// ']'
	b.eq	c_rbr
	cmp	w0, #46			// '.'
	b.eq	c_dot
	cmp	w0, #44			// ','
	b.eq	c_comma
istep:
	add	x25, x25, #1
	b	iloop

c_gt:
	add	x26, x26, #1
	b	istep
c_lt:
	sub	x26, x26, #1
	b	istep
c_plus:
	ldrb	w1, [x22, x26]
	add	w1, w1, #1
	strb	w1, [x22, x26]
	b	istep
c_minus:
	ldrb	w1, [x22, x26]
	sub	w1, w1, #1
	strb	w1, [x22, x26]
	b	istep
c_dot:
	mov	x0, #1			// fd 1
	add	x1, x22, x26
	mov	x2, #1
	mov	x8, #64			// write
	svc	#0
	b	istep
c_comma:
	mov	w1, #0
	cmp	x27, x24
	b.ge	1f
	ldrb	w1, [x19, x27]
	add	x27, x27, #1
1:
	strb	w1, [x22, x26]
	b	istep
c_lbr:
	ldrb	w1, [x22, x26]
	cbnz	w1, istep
	ldr	w25, [x21, x25, lsl #2]	// jump to matching ']', then +1
	b	istep
c_rbr:
	ldrb	w1, [x22, x26]
	cbz	w1, istep
	ldr	w25, [x21, x25, lsl #2]	// jump to matching '[', then +1
	b	istep

done:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0
	b	done

err:
	mov	x0, #1
	mov	x8, #93			// exit
	svc	#0
	b	err

.bss
	.balign 8
buf:
	.space	65536
	.balign 4
jumptab:
	.space	262144
tape:
	.space	30000
