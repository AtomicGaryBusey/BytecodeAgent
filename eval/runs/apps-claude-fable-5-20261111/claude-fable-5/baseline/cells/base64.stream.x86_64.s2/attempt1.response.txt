.text
.globl _start
_start:
    xor ebx, ebx                # total bytes read
read_loop:
    mov edx, 49153
    sub edx, ebx
    jle read_done
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    # rbx = total input length
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r9, [rip + b64tab]
    xor r10d, r10d              # groups on current line
enc_loop:
    cmp rbx, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    mov edx, eax
    shr edx, 18
    mov cl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], cl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov cl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], cl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov cl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], cl
    and eax, 63
    mov cl, byte ptr [r9 + rax]
    mov byte ptr [rdi + 3], cl
    add rsi, 3
    add rdi, 4
    sub rbx, 3
    inc r10d
    cmp r10d, 19
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r10d, r10d
    jmp enc_loop
enc_tail:
    test rbx, rbx
    jnz tail12
    test r10d, r10d
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi
    jmp do_write
tail12:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rbx, 2
    jb tail_chars
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
tail_chars:
    mov edx, eax
    shr edx, 18
    mov cl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], cl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov cl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], cl
    cmp rbx, 2
    jb pad2
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov cl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], cl
    mov byte ptr [rdi + 3], 61
    jmp tail_nl
pad2:
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
tail_nl:
    mov byte ptr [rdi + 4], 10
    add rdi, 5
do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz done
    mov eax, 1                  # write
    mov edi, 1                  # fd 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 60                 # exit
    xor edi, edi
    syscall

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .balign 16
inbuf:
    .space 49160
outbuf:
    .space 70016
