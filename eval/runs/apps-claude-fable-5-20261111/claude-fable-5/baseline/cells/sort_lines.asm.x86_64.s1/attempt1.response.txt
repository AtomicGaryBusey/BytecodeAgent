.text
.globl _start

INBUF_SIZE = 32768

_start:
    xor     r12, r12                # total bytes read
read_loop:
    mov     edx, INBUF_SIZE
    sub     edx, r12d
    jz      parse_start
    xor     eax, eax                # read
    xor     edi, edi                # fd 0
    lea     rsi, [rip + inbuf]
    add     rsi, r12
    syscall
    test    rax, rax
    jle     parse_start
    add     r12, rax
    jmp     read_loop

parse_start:
    xor     rbx, rbx                # start index of current line
    xor     r9, r9                  # i
    xor     r13, r13                # line count
parse_loop:
    cmp     r9, r12
    jae     parse_end
    cmp     byte ptr [inbuf + r9], 10
    jne     1f
    lea     rax, [inbuf + rbx]
    mov     qword ptr [lineptr + r13*8], rax
    mov     rax, r9
    sub     rax, rbx
    mov     qword ptr [linelen + r13*8], rax
    inc     r13
    lea     rbx, [r9 + 1]
1:
    inc     r9
    jmp     parse_loop
parse_end:
    cmp     rbx, r12
    jae     sort_start
    lea     rax, [inbuf + rbx]
    mov     qword ptr [lineptr + r13*8], rax
    mov     rax, r12
    sub     rax, rbx
    mov     qword ptr [linelen + r13*8], rax
    inc     r13

sort_start:
    cmp     r13, 2
    jb      output
    mov     r14, 1                  # i
outer:
    cmp     r14, r13
    jae     output
    mov     r15, qword ptr [lineptr + r14*8]    # key ptr
    mov     rbp, qword ptr [linelen + r14*8]    # key len
    mov     rbx, r14                            # insertion position
inner:
    test    rbx, rbx
    jz      place
    mov     rdi, qword ptr [lineptr + rbx*8 - 8]
    mov     rsi, qword ptr [linelen + rbx*8 - 8]
    mov     rdx, r15
    mov     rcx, rbp
    call    cmp_lines
    cmp     rax, 0
    jle     place
    mov     rax, qword ptr [lineptr + rbx*8 - 8]
    mov     qword ptr [lineptr + rbx*8], rax
    mov     rax, qword ptr [linelen + rbx*8 - 8]
    mov     qword ptr [linelen + rbx*8], rax
    dec     rbx
    jmp     inner
place:
    mov     qword ptr [lineptr + rbx*8], r15
    mov     qword ptr [linelen + rbx*8], rbp
    inc     r14
    jmp     outer

# cmp_lines: rdi=ptrA, rsi=lenA, rdx=ptrB, rcx=lenB
# returns rax = -1 / 0 / 1; clobbers rax, r8, r9, r10
cmp_lines:
    mov     r8, rsi
    cmp     r8, rcx
    jbe     1f
    mov     r8, rcx
1:
    xor     r9, r9
2:
    cmp     r9, r8
    jae     3f
    mov     al, byte ptr [rdi + r9]
    mov     r10b, byte ptr [rdx + r9]
    cmp     al, r10b
    jb      ret_neg
    ja      ret_pos
    inc     r9
    jmp     2b
3:
    cmp     rsi, rcx
    jb      ret_neg
    ja      ret_pos
    xor     eax, eax
    ret
ret_neg:
    mov     rax, -1
    ret
ret_pos:
    mov     eax, 1
    ret

output:
    lea     rdi, [rip + outbuf]
    xor     rbx, rbx                # line index
out_loop:
    cmp     rbx, r13
    jae     out_done
    mov     rsi, qword ptr [lineptr + rbx*8]
    mov     rcx, qword ptr [linelen + rbx*8]
    rep movsb
    mov     byte ptr [rdi], 10
    inc     rdi
    inc     rbx
    jmp     out_loop
out_done:
    lea     rsi, [rip + outbuf]
    mov     rdx, rdi
    sub     rdx, rsi
    test    rdx, rdx
    jz      done
    mov     eax, 1                  # write
    mov     edi, 1                  # fd 1
    syscall
done:
    mov     eax, 60                 # exit
    xor     edi, edi
    syscall

.bss
.balign 8
inbuf:
    .space INBUF_SIZE
outbuf:
    .space INBUF_SIZE + 1008
lineptr:
    .space 8000
linelen:
    .space 8000
