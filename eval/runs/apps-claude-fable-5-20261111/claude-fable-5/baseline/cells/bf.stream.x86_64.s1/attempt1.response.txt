.text
.globl _start
_start:
    # ---- read all of stdin into inbuf ----
    xor r12, r12                # total bytes read
.Lread:
    mov edx, 65536
    sub edx, r12d
    jle .Lread_done             # buffer full
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle .Lread_done             # 0 = EOF (or error)
    add r12, rax
    jmp .Lread
.Lread_done:

    # ---- find first '!' ----
    xor rcx, rcx
.Lfind:
    cmp rcx, r12
    jae .Lnobang
    cmp byte ptr [inbuf + rcx], 0x21
    je .Lbang
    inc rcx
    jmp .Lfind
.Lbang:
    mov rbp, rcx                # program length
    lea r14, [rip + inbuf]
    lea r14, [r14 + rcx + 1]    # input start
    jmp .Lsetend
.Lnobang:
    mov rbp, r12                # whole input is program
    lea r14, [rip + inbuf]
    add r14, r12                # empty input
.Lsetend:
    lea r15, [rip + inbuf]
    add r15, r12                # input end

    # ---- bracket check + build jump table ----
    xor rbx, rbx                # index i
    xor r8, r8                  # bracket stack depth
.Lbr:
    cmp rbx, rbp
    jae .Lbr_done
    movzx eax, byte ptr [inbuf + rbx]
    cmp al, 0x5b                # '['
    je .Lopen
    cmp al, 0x5d                # ']'
    je .Lclose
.Lbr_next:
    inc rbx
    jmp .Lbr
.Lopen:
    mov dword ptr [bstack + r8*4], ebx
    inc r8
    jmp .Lbr_next
.Lclose:
    test r8, r8
    jz .Lerr                    # unmatched ']'
    dec r8
    mov eax, dword ptr [bstack + r8*4]
    mov dword ptr [jmptab + rbx*4], eax
    mov dword ptr [jmptab + rax*4], ebx
    jmp .Lbr_next
.Lbr_done:
    test r8, r8
    jnz .Lerr                   # unclosed '['

    # ---- interpret ----
    xor rbx, rbx                # pc
    lea r13, [rip + tape]       # data pointer
    xor r9, r9                  # output count
.Lrun:
    cmp rbx, rbp
    jae .Lflush
    movzx eax, byte ptr [inbuf + rbx]
    inc rbx
    cmp al, 0x3e                # '>'
    je .Lgt
    cmp al, 0x3c                # '<'
    je .Llt
    cmp al, 0x2b                # '+'
    je .Lplus
    cmp al, 0x2d                # '-'
    je .Lminus
    cmp al, 0x2e                # '.'
    je .Ldot
    cmp al, 0x2c                # ','
    je .Lcomma
    cmp al, 0x5b                # '['
    je .Llb
    cmp al, 0x5d                # ']'
    je .Lrb
    jmp .Lrun                   # comment byte
.Lgt:
    inc r13
    jmp .Lrun
.Llt:
    dec r13
    jmp .Lrun
.Lplus:
    inc byte ptr [r13]
    jmp .Lrun
.Lminus:
    dec byte ptr [r13]
    jmp .Lrun
.Ldot:
    mov al, byte ptr [r13]
    mov byte ptr [outbuf + r9], al
    inc r9
    jmp .Lrun
.Lcomma:
    xor eax, eax
    cmp r14, r15
    jae 1f
    mov al, byte ptr [r14]
    inc r14
1:
    mov byte ptr [r13], al
    jmp .Lrun
.Llb:
    cmp byte ptr [r13], 0
    jne .Lrun
    mov ebx, dword ptr [jmptab + rbx*4 - 4]
    inc rbx                     # just past matching ']'
    jmp .Lrun
.Lrb:
    cmp byte ptr [r13], 0
    je .Lrun
    mov ebx, dword ptr [jmptab + rbx*4 - 4]
    inc rbx                     # just past matching '['
    jmp .Lrun

    # ---- flush output buffer ----
.Lflush:
    xor rbx, rbx                # bytes written
.Lw:
    mov rdx, r9
    sub rdx, rbx
    jz .Lexit0
    mov eax, 1                  # write
    mov edi, 1                  # stdout
    lea rsi, [rip + outbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle .Lexit0
    add rbx, rax
    jmp .Lw
.Lexit0:
    xor edi, edi
    mov eax, 231                # exit_group(0)
    syscall
.Lerr:
    mov edi, 1
    mov eax, 231                # exit_group(1)
    syscall

.bss
.balign 8
inbuf:
    .space 65536
jmptab:
    .space 262144
bstack:
    .space 4096
tape:
    .space 30000
outbuf:
    .space 65536
