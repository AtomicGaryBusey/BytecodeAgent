	.text
	.globl _start

_start:
	// read all of stdin into inbuf
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0			// total bytes read
read_loop:
	mov	x2, #32769
	sub	x2, x2, x20
	cbz	x2, read_done
	mov	x0, #0
	add	x1, x19, x20
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	ble	read_done
	add	x20, x20, x0
	b	read_loop
read_done:

	// parse lines into (ptr,len) table
	adrp	x21, lines
	add	x21, x21, :lo12:lines
	mov	x22, #0			// line count
	mov	x3, #0			// position
	mov	x4, #0			// start of current line
parse_loop:
	cmp	x3, x20
	bge	parse_end
	ldrb	w5, [x19, x3]
	add	x3, x3, #1
	cmp	w5, #10
	bne	parse_loop
	add	x6, x19, x4		// line pointer
	sub	x7, x3, x4
	sub	x7, x7, #1		// line length (without newline)
	lsl	x9, x22, #4
	add	x9, x21, x9
	stp	x6, x7, [x9]
	add	x22, x22, #1
	mov	x4, x3
	b	parse_loop
parse_end:
	cmp	x4, x20
	bge	sort_start
	add	x6, x19, x4		// final line without newline
	sub	x7, x20, x4
	lsl	x9, x22, #4
	add	x9, x21, x9
	stp	x6, x7, [x9]
	add	x22, x22, #1

sort_start:
	// insertion sort on the line table
	mov	x23, #1			// i
sort_outer:
	cmp	x23, x22
	bge	output
	lsl	x9, x23, #4
	add	x9, x21, x9
	ldp	x24, x25, [x9]		// key (ptr,len)
	sub	x26, x23, #1		// j
sort_inner:
	tbnz	x26, #63, sort_place	// j < 0 -> place
	lsl	x9, x26, #4
	add	x9, x21, x9
	ldp	x0, x1, [x9]
	mov	x2, x24
	mov	x3, x25
	bl	cmp_lines
	cmp	x0, #0
	ble	sort_place
	// shift lines[j] to lines[j+1]
	lsl	x9, x26, #4
	add	x9, x21, x9
	ldp	x0, x1, [x9]
	stp	x0, x1, [x9, #16]
	sub	x26, x26, #1
	b	sort_inner
sort_place:
	add	x9, x26, #1
	lsl	x9, x9, #4
	add	x9, x21, x9
	stp	x24, x25, [x9]
	add	x23, x23, #1
	b	sort_outer

output:
	// build output buffer: each line followed by newline
	adrp	x27, outbuf
	add	x27, x27, :lo12:outbuf
	mov	x28, #0			// output length
	mov	x23, #0			// index
out_loop:
	cmp	x23, x22
	bge	do_write
	lsl	x9, x23, #4
	add	x9, x21, x9
	ldp	x0, x1, [x9]		// ptr, len
copy_loop:
	cbz	x1, copy_done
	ldrb	w2, [x0], #1
	strb	w2, [x27, x28]
	add	x28, x28, #1
	sub	x1, x1, #1
	b	copy_loop
copy_done:
	mov	w2, #10
	strb	w2, [x27, x28]
	add	x28, x28, #1
	add	x23, x23, #1
	b	out_loop

do_write:
	mov	x24, #0			// bytes written
write_loop:
	cmp	x24, x28
	bge	done
	mov	x0, #1
	add	x1, x27, x24
	sub	x2, x28, x24
	mov	x8, #64			// write
	svc	#0
	cmp	x0, #0
	ble	done
	add	x24, x24, x0
	b	write_loop

done:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

// cmp_lines: x0=aptr, x1=alen, x2=bptr, x3=blen
// returns x0 < 0, == 0, > 0 like memcmp with length tiebreak
cmp_lines:
	cmp	x1, x3
	csel	x4, x1, x3, lo		// min length
	mov	x5, #0			// index
cmp_loop:
	cmp	x5, x4
	bge	cmp_len
	ldrb	w6, [x0, x5]
	ldrb	w7, [x2, x5]
	cmp	w6, w7
	bne	cmp_diff
	add	x5, x5, #1
	b	cmp_loop
cmp_len:
	sub	x0, x1, x3
	ret
cmp_diff:
	sub	w0, w6, w7
	sxtw	x0, w0
	ret

	.bss
	.balign 16
inbuf:
	.space	32769
	.balign 16
lines:
	.space	16000
	.balign 16
outbuf:
	.space	33800
