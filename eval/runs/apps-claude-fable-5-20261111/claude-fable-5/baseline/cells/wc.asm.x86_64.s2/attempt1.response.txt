.text
.globl _start
_start:
    # r12 = lines, r13 = words, r14 = bytes, r15 = in_word flag
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading

    add r14, rax                # bytes += n
    lea rsi, [rip + buffer]
    mov rcx, rax                # count

scan_byte:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a
    jne 1f
    inc r12                     # lines++
1:
    # whitespace? 0x09..0x0d or 0x20
    cmp al, 0x20
    je is_ws
    sub al, 0x09
    cmp al, 4
    jbe is_ws
    # word byte
    test r15, r15
    jnz next_byte
    mov r15, 1
    inc r13                     # words++
    jmp next_byte
is_ws:
    xor r15, r15
next_byte:
    dec rcx
    jnz scan_byte
    jmp read_loop

done_reading:
    # build output string backwards in outbuf
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 0x0a
    mov rax, r14
    call put_num
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call put_num
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call put_num

    # write(1, rdi, len)
    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 231
    xor edi, edi
    syscall

# put_num: rax = value, rdi points just past where digits go;
# writes decimal digits before rdi, returns rdi at first digit.
# clobbers rax, rbx, rdx, rcx
put_num:
    mov ebx, 10
1:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.balign 16
buffer:
    .space 65536
outbuf:
    .space 64
