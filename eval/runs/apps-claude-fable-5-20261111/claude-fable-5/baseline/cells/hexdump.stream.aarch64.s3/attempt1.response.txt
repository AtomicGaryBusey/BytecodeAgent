.text
.globl _start

_start:
	// read entire input into inbuf
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	adrp	x22, linebuf
	add	x22, x22, :lo12:linebuf
	adrp	x23, hextab
	add	x23, x23, :lo12:hextab
	mov	x20, #0			// total bytes read
	mov	x8, #63			// read
rd_loop:
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #65536
	sub	x2, x2, x20
	svc	#0
	cmp	x0, #0
	b.le	rd_done
	add	x20, x20, x0
	b	rd_loop
rd_done:
	mov	x21, #0			// current offset

row_loop:
	cmp	x21, x20
	b.ge	finish
	sub	x24, x20, x21		// remaining
	mov	x25, #16
	cmp	x24, x25
	csel	x24, x24, x25, lo	// n = min(remaining, 16)
	// offset field
	mov	x0, x22
	mov	w1, w21
	bl	put8hex
	mov	w4, #0x20		// ' '
	strb	w4, [x0], #1
	strb	w4, [x0], #1
	// hex column
	mov	x5, #0
hexcol:
	cmp	x5, x24
	b.ge	pad3
	add	x6, x21, x5
	ldrb	w7, [x19, x6]
	lsr	w3, w7, #4
	ldrb	w9, [x23, w3, uxtw]
	strb	w9, [x0], #1
	and	w3, w7, #15
	ldrb	w9, [x23, w3, uxtw]
	strb	w9, [x0], #1
	strb	w4, [x0], #1
	b	hexnext
pad3:
	strb	w4, [x0], #1
	strb	w4, [x0], #1
	strb	w4, [x0], #1
hexnext:
	cmp	x5, #7
	b.ne	1f
	strb	w4, [x0], #1		// extra space after slot 8
1:
	add	x5, x5, #1
	cmp	x5, #16
	b.lt	hexcol
	// space + '|'
	strb	w4, [x0], #1
	mov	w9, #0x7c		// '|'
	strb	w9, [x0], #1
	// ASCII column
	mov	x5, #0
asccol:
	cmp	x5, x24
	b.ge	asc_done
	add	x6, x21, x5
	ldrb	w7, [x19, x6]
	sub	w10, w7, #0x20
	cmp	w10, #0x5e
	b.ls	2f
	mov	w7, #0x2e		// '.'
2:
	strb	w7, [x0], #1
	add	x5, x5, #1
	b	asccol
asc_done:
	strb	w9, [x0], #1		// '|'
	mov	w9, #10
	strb	w9, [x0], #1		// '\n'
	// write the line
	sub	x2, x0, x22
	mov	x1, x22
	bl	wrall
	add	x21, x21, #16
	b	row_loop

finish:
	// final offset line
	mov	x0, x22
	mov	w1, w20
	bl	put8hex
	mov	w9, #10
	strb	w9, [x0], #1
	sub	x2, x0, x22
	mov	x1, x22
	bl	wrall
	// exit(0)
	mov	x8, #93
	mov	x0, #0
	svc	#0

// put8hex: write w1 as 8 lowercase hex digits at [x0]; x0 advances.
// clobbers w2, w3, w4
put8hex:
	mov	w2, #28
1:
	lsr	w3, w1, w2
	and	w3, w3, #15
	ldrb	w4, [x23, w3, uxtw]
	strb	w4, [x0], #1
	subs	w2, w2, #4
	b.ge	1b
	ret

// wrall: write x2 bytes from [x1] to fd 1; clobbers x0, x1, x2, x8
wrall:
	mov	x8, #64
1:
	cmp	x2, #0
	b.le	2f
	mov	x0, #1
	svc	#0
	cmp	x0, #0
	b.le	2f
	add	x1, x1, x0
	sub	x2, x2, x0
	b	1b
2:
	ret

.section .rodata
hextab:
	.ascii	"0123456789abcdef"

.bss
	.balign	16
inbuf:
	.space	65536
linebuf:
	.space	96
