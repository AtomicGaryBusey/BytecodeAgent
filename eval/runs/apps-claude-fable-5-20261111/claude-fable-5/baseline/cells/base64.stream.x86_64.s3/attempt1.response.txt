.text
.globl _start
_start:
    xor ebx, ebx                      # total bytes read
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 49152
    sub edx, ebx
    xor eax, eax                      # read
    xor edi, edi                      # fd 0
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r10, [rip + b64tbl]
    xor ecx, ecx                      # chars on current line
    mov r8, rbx                       # remaining input bytes

enc_loop:
    cmp r8, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    add rsi, 3
    sub r8, 3

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl
    mov edx, eax
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 3], dl
    add rdi, 4
    add ecx, 4
    cmp ecx, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor ecx, ecx
    jmp enc_loop

enc_tail:
    test r8, r8
    jz enc_done
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r8, 2
    jb tail_have
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
tail_have:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl
    cmp r8, 2
    je tail_two
    mov byte ptr [rdi + 2], 61        # '='
    jmp tail_pad
tail_two:
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], dl
tail_pad:
    mov byte ptr [rdi + 3], 61        # '='
    add rdi, 4
    add ecx, 4

enc_done:
    test ecx, ecx
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi

do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                      # total output length
write_loop:
    test rdx, rdx
    jz exit0
    mov eax, 1                        # write
    mov edi, 1                        # fd 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit0:
    mov eax, 60                       # exit
    xor edi, edi
    syscall

.section .rodata
b64tbl:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space 49152
outbuf:
    .space 70016
