.text
.globl _start
_start:
    // read all input into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0              // total bytes read
1:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    add x2, x2, #1           // allow reading a bit; input <= 65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:
    // process: build reversed output in outbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, #0              // output length
    mov x23, #0              // line start index i
    mov x24, #0              // scan index j
scan:
    cmp x24, x20
    b.ge after
    ldrb w0, [x19, x24]
    cmp w0, #10
    b.ne next
    // line from x23 .. x24-1, reversed
    mov x25, x24             // k = j
rev1:
    cmp x25, x23
    b.le revdone1
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b rev1
revdone1:
    mov w0, #10
    strb w0, [x21, x22]
    add x22, x22, #1
    add x23, x24, #1
next:
    add x24, x24, #1
    b scan
after:
    // final line without newline, if any
    cmp x23, x20
    b.ge dowrite
    mov x25, x20
rev2:
    cmp x25, x23
    b.le revdone2
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b rev2
revdone2:
    mov w0, #10
    strb w0, [x21, x22]
    add x22, x22, #1
dowrite:
    // write output, looping on partial writes
    mov x26, #0              // written so far
wloop:
    cmp x26, x22
    b.ge done
    mov x0, #1
    add x1, x21, x26
    sub x2, x22, x26
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le done
    add x26, x26, x0
    b wloop
done:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 8
inbuf:
    .space 65544
outbuf:
    .space 65544
