.text
.globl _start
_start:
        # ---- read all input ----
        xor     ebx, ebx                # total bytes read
read_loop:
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, rbx
        mov     edx, 32768
        sub     edx, ebx
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        cmp     rbx, 32768
        jb      read_loop
read_done:
        # ---- split into lines: table of (ptr, len) pairs ----
        lea     rsi, [rip + buf]
        lea     rdi, [rip + table]
        xor     r8, r8                  # pos
        xor     r9, r9                  # line start
        xor     r10, r10                # line count
parse_loop:
        cmp     r8, rbx
        jae     parse_end
        cmp     byte ptr [rsi + r8], 0x0A
        jne     parse_next
        lea     rax, [rsi + r9]
        mov     qword ptr [rdi], rax
        mov     rax, r8
        sub     rax, r9
        mov     qword ptr [rdi + 8], rax
        add     rdi, 16
        inc     r10
        lea     r9, [r8 + 1]
parse_next:
        inc     r8
        jmp     parse_loop
parse_end:
        cmp     r9, rbx
        jae     parse_done
        lea     rax, [rsi + r9]
        mov     qword ptr [rdi], rax
        mov     rax, rbx
        sub     rax, r9
        mov     qword ptr [rdi + 8], rax
        inc     r10
parse_done:
        mov     r12, r10                # r12 = number of lines

        # ---- insertion sort of the (ptr,len) table ----
        cmp     r12, 2
        jb      sort_done
        mov     r13, 1                  # i
sort_outer:
        mov     rax, r13
        shl     rax, 4
        lea     rbx, [rip + table]
        add     rbx, rax                # rbx = &table[i] (current slot)
        mov     rbp, qword ptr [rbx]    # key ptr
        mov     r14, qword ptr [rbx + 8]# key len
sort_inner:
        lea     rax, [rip + table]
        cmp     rbx, rax
        je      sort_place
        mov     rsi, qword ptr [rbx - 16]
        mov     rdx, qword ptr [rbx - 8]
        mov     rdi, rbp
        mov     r8, r14
        call    cmp_lines
        test    rax, rax
        jle     sort_place
        mov     rax, qword ptr [rbx - 16]
        mov     qword ptr [rbx], rax
        mov     rax, qword ptr [rbx - 8]
        mov     qword ptr [rbx + 8], rax
        sub     rbx, 16
        jmp     sort_inner
sort_place:
        mov     qword ptr [rbx], rbp
        mov     qword ptr [rbx + 8], r14
        inc     r13
        cmp     r13, r12
        jb      sort_outer
sort_done:

        # ---- build output buffer ----
        lea     rdi, [rip + outbuf]
        lea     rbx, [rip + table]
        xor     r13, r13
out_loop:
        cmp     r13, r12
        jae     out_done
        mov     rsi, qword ptr [rbx]
        mov     rcx, qword ptr [rbx + 8]
        rep movsb
        mov     byte ptr [rdi], 0x0A
        inc     rdi
        add     rbx, 16
        inc     r13
        jmp     out_loop
out_done:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi

        # ---- write everything to stdout ----
write_loop:
        test    rdx, rdx
        jz      do_exit
        mov     eax, 1                  # sys_write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     do_exit
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop

do_exit:
        mov     eax, 60                 # sys_exit
        xor     edi, edi
        syscall

# ---- compare lines: rsi=p1, rdx=l1, rdi=p2, r8=l2 ----
# returns rax <0 / 0 / >0 (memcmp order, shorter prefix first)
cmp_lines:
        mov     rcx, rdx
        cmp     rcx, r8
        cmova   rcx, r8                 # rcx = min(l1, l2)
        test    rcx, rcx
        jz      1f
        repe cmpsb
        je      1f
        movzx   eax, byte ptr [rsi - 1]
        movzx   ecx, byte ptr [rdi - 1]
        sub     eax, ecx
        ret
1:
        mov     rax, rdx
        sub     rax, r8
        ret

.bss
.balign 8
buf:
        .space  32768
table:
        .space  16016                   # 1001 entries * 16 bytes
outbuf:
        .space  33776                   # 32768 bytes + 1000 newlines + slack
