.text
.globl _start
_start:
	// ---- read all of stdin into inbuf ----
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0			// total bytes read
read_loop:
	mov	x2, #65536
	sub	x2, x2, x20
	cbz	x2, read_done		// buffer full
	mov	x0, #0
	add	x1, x19, x20
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop
read_done:
	// ---- split at first '!' ----
	mov	x21, #0
find_bang:
	cmp	x21, x20
	b.ge	no_bang
	ldrb	w0, [x19, x21]
	cmp	w0, #0x21
	b.eq	found_bang
	add	x21, x21, #1
	b	find_bang
found_bang:
	add	x24, x19, x21
	add	x24, x24, #1		// input start
	add	x25, x19, x20		// input end
	mov	x20, x21		// program length
	b	after_split
no_bang:
	add	x24, x19, x20		// empty input
	mov	x25, x24
after_split:
	// ---- bracket check + jump table ----
	adrp	x27, jumptab
	add	x27, x27, :lo12:jumptab
	adrp	x9, bstack
	add	x9, x9, :lo12:bstack
	mov	x10, #0			// stack depth
	mov	x21, #0			// scan index
bscan:
	cmp	x21, x20
	b.ge	bscan_done
	ldrb	w0, [x19, x21]
	cmp	w0, #91			// '['
	b.eq	bpush
	cmp	w0, #93			// ']'
	b.eq	bpop
bnext:
	add	x21, x21, #1
	b	bscan
bpush:
	str	x21, [x9, x10, lsl #3]
	add	x10, x10, #1
	b	bnext
bpop:
	cbz	x10, fail
	sub	x10, x10, #1
	ldr	x11, [x9, x10, lsl #3]
	str	w21, [x27, x11, lsl #2]
	str	w11, [x27, x21, lsl #2]
	b	bnext
bscan_done:
	cbnz	x10, fail

	// ---- interpret ----
	adrp	x22, tape
	add	x22, x22, :lo12:tape
	mov	x23, x22		// data pointer
	adrp	x26, outbuf
	add	x26, x26, :lo12:outbuf
	mov	x28, x26		// output write pointer
	mov	x21, #0			// pc
run:
	cmp	x21, x20
	b.ge	done
	ldrb	w0, [x19, x21]
	cmp	w0, #62			// '>'
	b.eq	c_gt
	cmp	w0, #60			// '<'
	b.eq	c_lt
	cmp	w0, #43			// '+'
	b.eq	c_plus
	cmp	w0, #45			// '-'
	b.eq	c_minus
	cmp	w0, #46			// '.'
	b.eq	c_dot
	cmp	w0, #44			// ','
	b.eq	c_comma
	cmp	w0, #91			// '['
	b.eq	c_lb
	cmp	w0, #93			// ']'
	b.eq	c_rb
next:
	add	x21, x21, #1
	b	run
c_gt:
	add	x23, x23, #1
	b	next
c_lt:
	sub	x23, x23, #1
	b	next
c_plus:
	ldrb	w1, [x23]
	add	w1, w1, #1
	strb	w1, [x23]
	b	next
c_minus:
	ldrb	w1, [x23]
	sub	w1, w1, #1
	strb	w1, [x23]
	b	next
c_dot:
	ldrb	w1, [x23]
	strb	w1, [x28], #1
	b	next
c_comma:
	cmp	x24, x25
	b.ge	comma_zero
	ldrb	w1, [x24], #1
	strb	w1, [x23]
	b	next
comma_zero:
	strb	wzr, [x23]
	b	next
c_lb:
	ldrb	w1, [x23]
	cbnz	w1, next
	ldr	w21, [x27, x21, lsl #2]
	b	next
c_rb:
	ldrb	w1, [x23]
	cbz	w1, next
	ldr	w21, [x27, x21, lsl #2]
	b	next

done:
	sub	x2, x28, x26
	cbz	x2, exit0
	mov	x0, #1
	mov	x1, x26
	mov	x8, #64			// write
	svc	#0
exit0:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0
fail:
	mov	x0, #1
	mov	x8, #93			// exit
	svc	#0

.bss
.balign 8
inbuf:
	.space	65536
tape:
	.space	30000
.balign 8
bstack:
	.space	8000
.balign 4
jumptab:
	.space	262144
outbuf:
	.space	65536
