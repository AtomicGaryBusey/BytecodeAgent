.section .rodata
errmsg: .ascii "error at "
.equ errmsg_len, . - errmsg
okmsg:  .ascii "ok\n"

.bss
.balign 8
buf:    .space 65600
numbuf: .space 32

.text
.globl _start

_start:
        adrp    x19, buf
        add     x19, x19, :lo12:buf
        mov     x20, #0                 // len
.read_loop:
        movz    x2, #0x0001
        movk    x2, #0x1, lsl #16       // 65537
        sub     x2, x2, x20
        cmp     x2, #0
        b.le    .read_done
        mov     x0, #0
        add     x1, x19, x20
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    .read_done
        add     x20, x20, x0
        b       .read_loop
.read_done:
        mov     x21, #0                 // pos
        bl      skip_ws
        bl      parse_value
        bl      skip_ws
        cmp     x21, x20
        b.ne    fail
// ok
        mov     x0, #1
        adrp    x1, okmsg
        add     x1, x1, :lo12:okmsg
        mov     x2, #3
        mov     x8, #64
        svc     #0
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       .

// peek: w0 = buf[pos] or 0x100 at EOF
peek:
        cmp     x21, x20
        b.hs    1f
        ldrb    w0, [x19, x21]
        ret
1:      mov     w0, #0x100
        ret

skip_ws:
1:      cmp     x21, x20
        b.hs    2f
        ldrb    w0, [x19, x21]
        cmp     w0, #0x20
        b.eq    3f
        cmp     w0, #0x09
        b.eq    3f
        cmp     w0, #0x0a
        b.eq    3f
        cmp     w0, #0x0d
        b.ne    2f
3:      add     x21, x21, #1
        b       1b
2:      ret

// match_lit: x0 -> asciz literal, consume matching bytes or fail
match_lit:
1:      ldrb    w1, [x0], #1
        cbz     w1, 2f
        cmp     x21, x20
        b.hs    fail
        ldrb    w2, [x19, x21]
        cmp     w1, w2
        b.ne    fail
        add     x21, x21, #1
        b       1b
2:      ret

parse_value:
        stp     x29, x30, [sp, #-16]!
        bl      peek
        cmp     w0, #'{'
        b.ne    1f
        bl      parse_object
        b       .pv_done
1:      cmp     w0, #'['
        b.ne    2f
        bl      parse_array
        b       .pv_done
2:      cmp     w0, #0x22
        b.ne    3f
        bl      parse_string
        b       .pv_done
3:      cmp     w0, #'-'
        b.eq    .pv_num
        cmp     w0, #'0'
        b.lo    4f
        cmp     w0, #'9'
        b.ls    .pv_num
4:      cmp     w0, #'t'
        b.ne    5f
        adrp    x0, lit_true
        add     x0, x0, :lo12:lit_true
        bl      match_lit
        b       .pv_done
5:      cmp     w0, #'f'
        b.ne    6f
        adrp    x0, lit_false
        add     x0, x0, :lo12:lit_false
        bl      match_lit
        b       .pv_done
6:      cmp     w0, #'n'
        b.ne    fail
        adrp    x0, lit_null
        add     x0, x0, :lo12:lit_null
        bl      match_lit
        b       .pv_done
.pv_num:
        bl      parse_number
.pv_done:
        ldp     x29, x30, [sp], #16
        ret

parse_object:
        stp     x29, x30, [sp, #-16]!
        add     x21, x21, #1            // {
        bl      skip_ws
        bl      peek
        cmp     w0, #'}'
        b.eq    .po_close
.po_member:
        bl      peek
        cmp     w0, #0x22
        b.ne    fail
        bl      parse_string
        bl      skip_ws
        bl      peek
        cmp     w0, #':'
        b.ne    fail
        add     x21, x21, #1
        bl      skip_ws
        bl      parse_value
        bl      skip_ws
        bl      peek
        cmp     w0, #','
        b.eq    .po_comma
        cmp     w0, #'}'
        b.ne    fail
.po_close:
        add     x21, x21, #1
        ldp     x29, x30, [sp], #16
        ret
.po_comma:
        add     x21, x21, #1
        bl      skip_ws
        b       .po_member

parse_array:
        stp     x29, x30, [sp, #-16]!
        add     x21, x21, #1            // [
        bl      skip_ws
        bl      peek
        cmp     w0, #']'
        b.eq    .pa_close
.pa_elem:
        bl      parse_value
        bl      skip_ws
        bl      peek
        cmp     w0, #','
        b.eq    .pa_comma
        cmp     w0, #']'
        b.ne    fail
.pa_close:
        add     x21, x21, #1
        ldp     x29, x30, [sp], #16
        ret
.pa_comma:
        add     x21, x21, #1
        bl      skip_ws
        b       .pa_elem

parse_string:
        add     x21, x21, #1            // opening "
.ps_loop:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #0x22
        b.eq    .ps_end
        cmp     w0, #0x5c
        b.eq    .ps_esc
        cmp     w0, #0x20
        b.lo    fail
        add     x21, x21, #1
        b       .ps_loop
.ps_end:
        add     x21, x21, #1
        ret
.ps_esc:
        add     x21, x21, #1
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #'u'
        b.eq    .ps_u
        cmp     w0, #0x22
        b.eq    .ps_ok
        cmp     w0, #0x5c
        b.eq    .ps_ok
        cmp     w0, #'/'
        b.eq    .ps_ok
        cmp     w0, #'b'
        b.eq    .ps_ok
        cmp     w0, #'f'
        b.eq    .ps_ok
        cmp     w0, #'n'
        b.eq    .ps_ok
        cmp     w0, #'r'
        b.eq    .ps_ok
        cmp     w0, #'t'
        b.ne    fail
.ps_ok:
        add     x21, x21, #1
        b       .ps_loop
.ps_u:
        add     x21, x21, #1
        mov     w3, #4
.ps_hex:
        cmp     x21, x20
        b.hs    fail
        ldrb    w0, [x19, x21]
        cmp     w0, #'0'
        b.lo    fail
        cmp     w0, #'9'
        b.ls    .ps_hok
        orr     w1, w0, #0x20
        cmp     w1, #'a'
        b.lo    fail
        cmp     w1, #'f'
        b.hi    fail
.ps_hok:
        add     x21, x21, #1
        subs    w3, w3, #1
        b.ne    .ps_hex
        b       .ps_loop

parse_number:
        stp     x29, x30, [sp, #-16]!
        bl      peek
        cmp     w0, #'-'
        b.ne    1f
        add     x21, x21, #1
        bl      peek
1:      cmp     w0, #'0'
        b.eq    .pn_zero
        b.lo    fail
        cmp     w0, #'9'
        b.hi    fail
        add     x21, x21, #1
.pn_intloop:
        bl      peek
        cmp     w0, #'0'
        b.lo    .pn_frac
        cmp     w0, #'9'
        b.hi    .pn_frac
        add     x21, x21, #1
        b       .pn_intloop
.pn_zero:
        add     x21, x21, #1
        bl      peek
.pn_frac:
        cmp     w0, #'.'
        b.ne    .pn_exp
        add     x21, x21, #1
        bl      peek
        cmp     w0, #'0'
        b.lo    fail
        cmp     w0, #'9'
        b.hi    fail
        add     x21, x21, #1
.pn_fracloop:
        bl      peek
        cmp     w0, #'0'
        b.lo    .pn_exp
        cmp     w0, #'9'
        b.hi    .pn_exp
        add     x21, x21, #1
        b       .pn_fracloop
.pn_exp:
        cmp     w0, #'e'
        b.eq    2f
        cmp     w0, #'E'
        b.ne    .pn_done
2:      add     x21, x21, #1
        bl      peek
        cmp     w0, #'+'
        b.eq    3f
        cmp     w0, #'-'
        b.ne    4f
3:      add     x21, x21, #1
        bl      peek
4:      cmp     w0, #'0'
        b.lo    fail
        cmp     w0, #'9'
        b.hi    fail
        add     x21, x21, #1
.pn_exploop:
        bl      peek
        cmp     w0, #'0'
        b.lo    .pn_done
        cmp     w0, #'9'
        b.hi    .pn_done
        add     x21, x21, #1
        b       .pn_exploop
.pn_done:
        ldp     x29, x30, [sp], #16
        ret

// fail: print "error at N\n" with N = x21, exit 1
fail:
        mov     x0, #1
        adrp    x1, errmsg
        add     x1, x1, :lo12:errmsg
        mov     x2, #errmsg_len
        mov     x8, #64
        svc     #0
        adrp    x4, numbuf
        add     x4, x4, :lo12:numbuf
        add     x4, x4, #31
        mov     w0, #10                 // '\n'
        strb    w0, [x4]
        mov     x1, x21
        mov     x5, #10
1:      udiv    x2, x1, x5
        msub    x3, x2, x5, x1
        add     w3, w3, #'0'
        sub     x4, x4, #1
        strb    w3, [x4]
        mov     x1, x2
        cbnz    x1, 1b
        adrp    x2, numbuf
        add     x2, x2, :lo12:numbuf
        add     x2, x2, #32
        sub     x2, x2, x4
        mov     x1, x4
        mov     x0, #1
        mov     x8, #64
        svc     #0
        mov     x0, #1
        mov     x8, #93
        svc     #0
        b       .

.section .rodata
lit_true:  .asciz "true"
lit_false: .asciz "false"
lit_null:  .asciz "null"
