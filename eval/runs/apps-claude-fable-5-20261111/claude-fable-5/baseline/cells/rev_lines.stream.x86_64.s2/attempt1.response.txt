.text
.globl _start
_start:
	xor	r12, r12		# total bytes read

read_loop:
	mov	edx, 65536
	sub	edx, r12d
	jz	read_done
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + inbuf]
	add	rsi, r12
	syscall
	test	rax, rax
	jle	read_done
	add	r12, rax
	jmp	read_loop

read_done:
	lea	rsi, [rip + inbuf]
	lea	rdi, [rip + outbuf]
	xor	ebx, ebx		# current line start i

line_loop:
	cmp	rbx, r12
	jae	do_write
	mov	rcx, rbx		# scan for newline
scan:
	cmp	rcx, r12
	jae	found_end
	cmp	byte ptr [rsi + rcx], 10
	je	found_end
	inc	rcx
	jmp	scan
found_end:
	mov	rdx, rcx		# line is [rbx, rcx)
rev_loop:
	cmp	rdx, rbx
	je	rev_done
	dec	rdx
	mov	al, byte ptr [rsi + rdx]
	mov	byte ptr [rdi], al
	inc	rdi
	jmp	rev_loop
rev_done:
	mov	byte ptr [rdi], 10
	inc	rdi
	lea	rbx, [rcx + 1]
	jmp	line_loop

do_write:
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi		# output length
write_loop:
	test	rdx, rdx
	jz	done
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	syscall
	test	rax, rax
	jle	done
	add	rsi, rax
	sub	rdx, rax
	jmp	write_loop

done:
	mov	eax, 60			# exit
	xor	edi, edi
	syscall

.bss
.balign 16
inbuf:
	.zero	65536
outbuf:
	.zero	65600
