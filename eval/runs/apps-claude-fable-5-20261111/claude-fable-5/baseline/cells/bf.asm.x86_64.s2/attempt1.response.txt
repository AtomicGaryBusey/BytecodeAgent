.text
.globl _start

.equ BUFSZ, 65536

_start:
        # ---- read all of stdin into buf ----
        xor     ebx, ebx                # total bytes read
read_loop:
        mov     edx, BUFSZ
        sub     edx, ebx
        jz      read_done
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, rbx
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        # rbx = total input length

        # ---- find first '!' ----
        xor     ecx, ecx
find_bang:
        cmp     rcx, rbx
        jae     no_bang
        cmp     byte ptr [buf + rcx], 0x21
        je      found_bang
        inc     rcx
        jmp     find_bang
found_bang:
        mov     r12, rcx                # program length
        lea     r14, [rcx + 1]          # input start index
        jmp     split_done
no_bang:
        mov     r12, rbx
        mov     r14, rbx
split_done:
        # r12 = program length, r14 = input start index, rbx = total length
        lea     r15, [rip + buf]
        add     r15, rbx                # input end address
        lea     rax, [rip + buf]
        add     r14, rax                # input start address

        # ---- bracket check + jump table ----
        xor     edx, edx                # stack depth
        xor     ecx, ecx                # index
bc_loop:
        cmp     rcx, r12
        jae     bc_end
        mov     al, byte ptr [buf + rcx]
        cmp     al, 0x5b                # '['
        je      bc_open
        cmp     al, 0x5d                # ']'
        je      bc_close
bc_next:
        inc     rcx
        jmp     bc_loop
bc_open:
        mov     dword ptr [bstack + rdx*4], ecx
        inc     edx
        jmp     bc_next
bc_close:
        test    edx, edx
        jz      fail
        dec     edx
        mov     eax, dword ptr [bstack + rdx*4]   # eax = pos of matching '['
        lea     r9d, [rcx + 1]
        mov     dword ptr [jmptab + rax*4], r9d   # '[' jumps past ']'
        lea     r9d, [rax + 1]
        mov     dword ptr [jmptab + rcx*4], r9d   # ']' jumps past '['
        jmp     bc_next
bc_end:
        test    edx, edx
        jnz     fail

        # ---- interpreter ----
        xor     ebx, ebx                # pc
        lea     r13, [rip + tape]       # data pointer
        lea     rbp, [rip + outbuf]     # output pointer
run:
        cmp     rbx, r12
        jae     finish
        movzx   eax, byte ptr [buf + rbx]
        inc     rbx
        cmp     al, 0x3e                # '>'
        je      op_gt
        cmp     al, 0x3c                # '<'
        je      op_lt
        cmp     al, 0x2b                # '+'
        je      op_plus
        cmp     al, 0x2d                # '-'
        je      op_minus
        cmp     al, 0x2e                # '.'
        je      op_dot
        cmp     al, 0x2c                # ','
        je      op_comma
        cmp     al, 0x5b                # '['
        je      op_lbr
        cmp     al, 0x5d                # ']'
        je      op_rbr
        jmp     run
op_gt:
        inc     r13
        jmp     run
op_lt:
        dec     r13
        jmp     run
op_plus:
        inc     byte ptr [r13]
        jmp     run
op_minus:
        dec     byte ptr [r13]
        jmp     run
op_dot:
        mov     al, byte ptr [r13]
        mov     byte ptr [rbp], al
        inc     rbp
        jmp     run
op_comma:
        xor     eax, eax
        cmp     r14, r15
        jae     1f
        mov     al, byte ptr [r14]
        inc     r14
1:
        mov     byte ptr [r13], al
        jmp     run
op_lbr:
        cmp     byte ptr [r13], 0
        jne     run
        mov     ebx, dword ptr [jmptab + rbx*4 - 4]
        jmp     run
op_rbr:
        cmp     byte ptr [r13], 0
        je      run
        mov     ebx, dword ptr [jmptab + rbx*4 - 4]
        jmp     run

        # ---- flush output and exit 0 ----
finish:
        lea     rsi, [rip + outbuf]
        mov     rdx, rbp
        sub     rdx, rsi
flush_loop:
        test    rdx, rdx
        jz      exit0
        mov     eax, 1                  # write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     exit0
        add     rsi, rax
        sub     rdx, rax
        jmp     flush_loop
exit0:
        mov     eax, 60
        xor     edi, edi
        syscall

fail:
        mov     eax, 60
        mov     edi, 1
        syscall

.bss
.balign 8
buf:    .space 65536
tape:   .space 30000
outbuf: .space 65536
.balign 4
jmptab: .space 262144
bstack: .space 262144
