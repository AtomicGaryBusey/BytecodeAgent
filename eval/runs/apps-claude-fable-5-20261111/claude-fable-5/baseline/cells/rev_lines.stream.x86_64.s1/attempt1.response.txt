.text
.globl _start
_start:
	xor	ebx, ebx		# total bytes read
read_loop:
	mov	edx, 65536
	sub	edx, ebx
	jz	read_done
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + inbuf]
	add	rsi, rbx
	syscall
	test	rax, rax
	jle	read_done
	add	rbx, rax
	jmp	read_loop
read_done:
	xor	r8, r8			# input index
	xor	r9, r9			# output index
	lea	r10, [rip + inbuf]
	lea	r12, [rip + outbuf]
proc_loop:
	cmp	r8, rbx
	jae	proc_done
	mov	rcx, r8			# find newline or end
find_nl:
	cmp	rcx, rbx
	jae	got_end
	cmp	byte ptr [r10 + rcx], 10
	je	got_end
	inc	rcx
	jmp	find_nl
got_end:
	mov	rdx, rcx		# copy [r8, rcx) reversed
rev_loop:
	cmp	rdx, r8
	je	rev_done
	dec	rdx
	mov	al, byte ptr [r10 + rdx]
	mov	byte ptr [r12 + r9], al
	inc	r9
	jmp	rev_loop
rev_done:
	mov	byte ptr [r12 + r9], 10
	inc	r9
	lea	r8, [rcx + 1]		# skip newline
	jmp	proc_loop
proc_done:
	xor	rbx, rbx		# bytes written
write_loop:
	cmp	rbx, r9
	jae	exit
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	lea	rsi, [rip + outbuf]
	add	rsi, rbx
	mov	rdx, r9
	sub	rdx, rbx
	syscall
	test	rax, rax
	jle	exit
	add	rbx, rax
	jmp	write_loop
exit:
	xor	edi, edi
	mov	eax, 60			# exit
	syscall

.bss
.balign 16
inbuf:
	.zero	65536
outbuf:
	.zero	65537
