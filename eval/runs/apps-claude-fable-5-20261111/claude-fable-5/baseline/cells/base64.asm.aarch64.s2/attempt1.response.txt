	.text
	.globl _start

_start:
	// read all of stdin into inbuf
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0			// total bytes read
1:
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #49152
	sub	x2, x2, x20
	cbz	x2, 2f
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	b.le	2f
	add	x20, x20, x0
	b	1b
2:
	// set up for encoding
	adrp	x23, b64tab
	add	x23, x23, :lo12:b64tab
	adrp	x21, outbuf
	add	x21, x21, :lo12:outbuf	// output write pointer
	mov	x26, x21		// output start
	mov	x22, #0			// column counter
	mov	x24, x19		// source pointer
	mov	x25, x20		// remaining bytes

enc_loop:
	cmp	x25, #3
	b.lt	enc_tail
	ldrb	w2, [x24], #1
	ldrb	w3, [x24], #1
	ldrb	w4, [x24], #1
	lsl	w5, w2, #16
	orr	w5, w5, w3, lsl #8
	orr	w5, w5, w4
	lsr	w6, w5, #18
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	lsr	w6, w5, #12
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	lsr	w6, w5, #6
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	and	w6, w5, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	sub	x25, x25, #3
	b	enc_loop

enc_tail:
	cbz	x25, enc_done
	cmp	x25, #1
	b.eq	tail1
	// two leftover bytes -> 3 chars + '='
	ldrb	w2, [x24]
	ldrb	w3, [x24, #1]
	lsl	w5, w2, #16
	orr	w5, w5, w3, lsl #8
	lsr	w6, w5, #18
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	lsr	w6, w5, #12
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	lsr	w6, w5, #6
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	mov	w6, #61			// '='
	bl	emit
	b	enc_done

tail1:
	// one leftover byte -> 2 chars + "=="
	ldrb	w2, [x24]
	lsl	w5, w2, #16
	lsr	w6, w5, #18
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	lsr	w6, w5, #12
	and	w6, w6, #63
	ldrb	w6, [x23, w6, uxtw]
	bl	emit
	mov	w6, #61			// '='
	bl	emit
	mov	w6, #61			// '='
	bl	emit

enc_done:
	// final newline if last line not already terminated
	cbz	x22, do_write
	mov	w6, #10
	strb	w6, [x21], #1

do_write:
	sub	x2, x21, x26		// total output length
	mov	x1, x26
wr_loop:
	cbz	x2, done
	mov	x0, #1
	mov	x8, #64			// write
	svc	#0
	cmp	x0, #0
	b.le	done
	add	x1, x1, x0
	sub	x2, x2, x0
	b	wr_loop

done:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

// emit one base64 character in w6, handle 76-column line wrapping
emit:
	strb	w6, [x21], #1
	add	x22, x22, #1
	cmp	x22, #76
	b.ne	1f
	mov	w7, #10
	strb	w7, [x21], #1
	mov	x22, #0
1:
	ret

	.section .rodata
b64tab:
	.ascii	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

	.bss
	.balign	8
inbuf:
	.space	49152
outbuf:
	.space	69632
