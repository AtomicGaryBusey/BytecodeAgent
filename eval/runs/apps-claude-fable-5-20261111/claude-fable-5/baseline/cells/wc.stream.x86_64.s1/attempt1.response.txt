.text
.globl _start
_start:
	# r12 = lines, r13 = words, r14 = bytes, r15 = in_word flag
	xor r12d, r12d
	xor r13d, r13d
	xor r14d, r14d
	xor r15d, r15d

read_loop:
	xor eax, eax                    # read
	xor edi, edi                    # fd 0
	lea rsi, [rip + buffer]
	mov edx, 65536
	syscall
	test rax, rax
	jle done_reading

	add r14, rax                    # bytes += n
	lea rsi, [rip + buffer]
	mov rcx, rax                    # rcx = count

scan_byte:
	movzx eax, byte ptr [rsi]
	inc rsi
	cmp al, 0x0a
	jne 1f
	inc r12                         # lines++
1:
	# whitespace? 0x09..0x0d or 0x20
	cmp al, 0x20
	je is_space
	sub al, 0x09
	cmp al, 4
	jbe is_space
	# word byte
	test r15d, r15d
	jnz next_byte
	mov r15d, 1
	inc r13                         # words++
	jmp next_byte
is_space:
	xor r15d, r15d
next_byte:
	dec rcx
	jnz scan_byte
	jmp read_loop

done_reading:
	# build output string backwards in outbuf
	lea rdi, [rip + outbuf + 63]    # rdi points past end; fill backwards
	mov byte ptr [rdi], 0x0a
	mov rax, r14
	call put_num
	dec rdi
	mov byte ptr [rdi], 0x20
	mov rax, r13
	call put_num
	dec rdi
	mov byte ptr [rdi], 0x20
	mov rax, r12
	call put_num

	# write(1, rdi, len)
	mov rsi, rdi
	lea rdx, [rip + outbuf + 64]
	sub rdx, rsi
	mov eax, 1
	mov edi, 1
	syscall

	mov eax, 60
	xor edi, edi
	syscall

# put_num: writes decimal of rax backwards, ending just before [rdi];
# on return rdi points at the first digit
put_num:
	mov r8, 10
1:
	xor edx, edx
	div r8
	add dl, '0'
	dec rdi
	mov byte ptr [rdi], dl
	test rax, rax
	jnz 1b
	ret

.bss
.balign 16
buffer:
	.space 65536
outbuf:
	.space 64
