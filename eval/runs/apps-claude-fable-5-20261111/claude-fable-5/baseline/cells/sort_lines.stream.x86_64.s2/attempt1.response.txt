	.text
	.globl _start
_start:
	# ---- read all of stdin into inbuf ----
	xor	ebx, ebx		# rbx = total bytes read
read_loop:
	mov	edx, 32768
	sub	edx, ebx
	jle	read_done
	xor	eax, eax		# sys_read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + inbuf]
	add	rsi, rbx
	syscall
	test	rax, rax
	jle	read_done
	add	rbx, rax
	jmp	read_loop
read_done:

	# ---- parse lines into (ptr,len) records ----
	lea	rsi, [rip + inbuf]
	lea	rdi, [rip + lines]
	xor	r12, r12		# line count
	xor	rcx, rcx		# position
parse:
	cmp	rcx, rbx
	jae	parse_end
	mov	r8, rcx			# line start
find_nl:
	cmp	rcx, rbx
	jae	no_nl
	cmp	byte ptr [rsi + rcx], 10
	je	got_nl
	inc	rcx
	jmp	find_nl
got_nl:
	lea	rax, [rsi + r8]
	mov	qword ptr [rdi], rax
	mov	r9, rcx
	sub	r9, r8
	mov	qword ptr [rdi + 8], r9
	add	rdi, 16
	inc	r12
	inc	rcx
	jmp	parse
no_nl:
	lea	rax, [rsi + r8]
	mov	qword ptr [rdi], rax
	mov	r9, rbx
	sub	r9, r8
	mov	qword ptr [rdi + 8], r9
	inc	r12
parse_end:

	# ---- insertion sort on records ----
	lea	rbp, [rip + lines]
	mov	r13, 1			# i
outer:
	cmp	r13, r12
	jae	sort_done
	mov	rax, r13
	shl	rax, 4
	mov	r14, qword ptr [rbp + rax]	# key ptr
	mov	r15, qword ptr [rbp + rax + 8]	# key len
	mov	rbx, r13		# j
inner:
	test	rbx, rbx
	jz	place
	mov	rax, rbx
	shl	rax, 4
	mov	rsi, qword ptr [rbp + rax - 16]	# A ptr
	mov	r8,  qword ptr [rbp + rax - 8]	# A len
	# compare A=(rsi,r8) with key=(r14,r15)
	mov	rcx, r8
	cmp	rcx, r15
	cmova	rcx, r15		# rcx = min len
	xor	edx, edx
cmp_loop:
	cmp	rdx, rcx
	jae	cmp_len
	movzx	eax, byte ptr [rsi + rdx]
	movzx	edi, byte ptr [r14 + rdx]
	cmp	eax, edi
	ja	a_greater
	jb	place
	inc	rdx
	jmp	cmp_loop
cmp_len:
	cmp	r8, r15
	jbe	place			# A <= key: stop
a_greater:
	# shift lines[j] = lines[j-1]
	mov	rax, rbx
	shl	rax, 4
	mov	rdx, qword ptr [rbp + rax - 16]
	mov	qword ptr [rbp + rax], rdx
	mov	rdx, qword ptr [rbp + rax - 8]
	mov	qword ptr [rbp + rax + 8], rdx
	dec	rbx
	jmp	inner
place:
	mov	rax, rbx
	shl	rax, 4
	mov	qword ptr [rbp + rax], r14
	mov	qword ptr [rbp + rax + 8], r15
	inc	r13
	jmp	outer
sort_done:

	# ---- build output buffer ----
	lea	rdi, [rip + outbuf]
	xor	r13, r13
out_loop:
	cmp	r13, r12
	jae	out_done
	mov	rax, r13
	shl	rax, 4
	mov	rsi, qword ptr [rbp + rax]
	mov	rcx, qword ptr [rbp + rax + 8]
	rep movsb
	mov	byte ptr [rdi], 10
	inc	rdi
	inc	r13
	jmp	out_loop
out_done:
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi

	# ---- write it all out ----
write_loop:
	test	rdx, rdx
	jz	done
	mov	eax, 1			# sys_write
	mov	edi, 1			# fd 1
	syscall
	test	rax, rax
	jle	done
	add	rsi, rax
	sub	rdx, rax
	jmp	write_loop
done:
	mov	eax, 231		# exit_group
	xor	edi, edi
	syscall

	.bss
	.balign 16
inbuf:
	.space	32768
outbuf:
	.space	33768
	.balign 16
lines:
	.space	16000
