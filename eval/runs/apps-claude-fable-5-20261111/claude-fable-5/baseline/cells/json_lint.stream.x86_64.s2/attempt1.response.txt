.text
.globl _start

_start:
        xor     ebx, ebx                # total bytes read
        lea     r12, [rip + buf]
read_loop:
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        lea     rsi, [r12 + rbx]
        mov     edx, 65600
        sub     edx, ebx
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        mov     r13, rbx                # r13 = length
        xor     ebx, ebx                # rbx = current offset
        call    skip_ws
        call    parse_value
        call    skip_ws
        cmp     rbx, r13
        jne     fail
        lea     rsi, [rip + ok_msg]
        mov     edx, ok_len
        call    write_all
        xor     edi, edi
        mov     eax, 231
        syscall

fail:                                   # error at offset rbx
        lea     rsi, [rip + err_msg]
        mov     edx, err_len
        call    write_all
        lea     rdi, [rip + numbuf + 31]
        mov     byte ptr [rdi], 10      # newline
        mov     rax, rbx
        mov     ecx, 10
1:
        xor     edx, edx
        div     rcx
        add     dl, '0'
        dec     rdi
        mov     byte ptr [rdi], dl
        test    rax, rax
        jnz     1b
        mov     rsi, rdi
        lea     rdx, [rip + numbuf + 32]
        sub     rdx, rdi
        call    write_all
        mov     edi, 1
        mov     eax, 231
        syscall

# write_all: write rdx bytes at rsi to stdout
write_all:
1:
        test    rdx, rdx
        jle     2f
        mov     eax, 1
        mov     edi, 1
        syscall
        test    rax, rax
        jle     2f
        add     rsi, rax
        sub     rdx, rax
        jmp     1b
2:
        ret

# skip whitespace (space, tab, LF, CR)
skip_ws:
1:
        cmp     rbx, r13
        jae     2f
        mov     al, byte ptr [r12 + rbx]
        cmp     al, 0x20
        je      3f
        cmp     al, 0x09
        je      3f
        cmp     al, 0x0A
        je      3f
        cmp     al, 0x0D
        je      3f
2:
        ret
3:
        inc     rbx
        jmp     1b

parse_value:
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '{'
        je      parse_object
        cmp     al, '['
        je      parse_array
        cmp     al, '"'
        je      parse_string
        cmp     al, 't'
        je      lit_true
        cmp     al, 'f'
        je      lit_false
        cmp     al, 'n'
        je      lit_null
        cmp     al, '-'
        je      parse_number
        cmp     al, '0'
        jb      fail
        cmp     al, '9'
        ja      fail
        jmp     parse_number

lit_true:
        lea     rsi, [rip + str_true]
        jmp     match_lit
lit_false:
        lea     rsi, [rip + str_false]
        jmp     match_lit
lit_null:
        lea     rsi, [rip + str_null]
match_lit:
1:
        mov     cl, byte ptr [rsi]
        test    cl, cl
        je      2f
        cmp     rbx, r13
        jae     fail
        cmp     cl, byte ptr [r12 + rbx]
        jne     fail
        inc     rbx
        inc     rsi
        jmp     1b
2:
        ret

parse_string:                           # entered at the opening quote
        inc     rbx
str_loop:
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '"'
        je      str_done
        cmp     al, 0x5C
        je      str_esc
        cmp     al, 0x20
        jb      fail
        inc     rbx
        jmp     str_loop
str_done:
        inc     rbx
        ret
str_esc:
        inc     rbx
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, 'u'
        je      str_uni
        cmp     al, '"'
        je      str_esc_ok
        cmp     al, 0x5C
        je      str_esc_ok
        cmp     al, '/'
        je      str_esc_ok
        cmp     al, 'b'
        je      str_esc_ok
        cmp     al, 'f'
        je      str_esc_ok
        cmp     al, 'n'
        je      str_esc_ok
        cmp     al, 'r'
        je      str_esc_ok
        cmp     al, 't'
        je      str_esc_ok
        jmp     fail
str_esc_ok:
        inc     rbx
        jmp     str_loop
str_uni:
        inc     rbx
        mov     r8d, 4
str_hex:
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '0'
        jb      fail
        cmp     al, '9'
        jbe     str_hex_ok
        cmp     al, 'A'
        jb      fail
        cmp     al, 'F'
        jbe     str_hex_ok
        cmp     al, 'a'
        jb      fail
        cmp     al, 'f'
        ja      fail
str_hex_ok:
        inc     rbx
        dec     r8d
        jnz     str_hex
        jmp     str_loop

parse_number:                           # entered at '-' or a digit
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '-'
        jne     num_int
        inc     rbx
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
num_int:
        cmp     al, '0'
        jb      fail
        cmp     al, '9'
        ja      fail
        cmp     al, '0'
        je      num_zero
num_digits:
        inc     rbx
        cmp     rbx, r13
        jae     num_ret
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '0'
        jb      num_frac
        cmp     al, '9'
        ja      num_frac
        jmp     num_digits
num_zero:
        inc     rbx
        cmp     rbx, r13
        jae     num_ret
        mov     al, byte ptr [r12 + rbx]
num_frac:
        cmp     al, '.'
        jne     num_exp_check
        inc     rbx
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '0'
        jb      fail
        cmp     al, '9'
        ja      fail
num_frac_digits:
        inc     rbx
        cmp     rbx, r13
        jae     num_ret
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '0'
        jb      num_exp_check
        cmp     al, '9'
        ja      num_exp_check
        jmp     num_frac_digits
num_exp_check:
        cmp     al, 'e'
        je      num_exp
        cmp     al, 'E'
        jne     num_ret
num_exp:
        inc     rbx
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '+'
        je      num_exp_sign
        cmp     al, '-'
        jne     num_exp_dig1
num_exp_sign:
        inc     rbx
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
num_exp_dig1:
        cmp     al, '0'
        jb      fail
        cmp     al, '9'
        ja      fail
num_exp_digits:
        inc     rbx
        cmp     rbx, r13
        jae     num_ret
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '0'
        jb      num_ret
        cmp     al, '9'
        ja      num_ret
        jmp     num_exp_digits
num_ret:
        ret

parse_array:                            # entered at '['
        inc     rbx
        call    skip_ws
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, ']'
        jne     arr_elem
        inc     rbx
        ret
arr_elem:
        call    parse_value
arr_after:
        call    skip_ws
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, ']'
        je      arr_close
        cmp     al, ','
        jne     fail
        inc     rbx
        call    skip_ws
        call    parse_value
        jmp     arr_after
arr_close:
        inc     rbx
        ret

parse_object:                           # entered at '{'
        inc     rbx
        call    skip_ws
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '}'
        jne     obj_member
        inc     rbx
        ret
obj_member:                             # al = current byte, rbx < r13
        cmp     al, '"'
        jne     fail
        call    parse_string
        call    skip_ws
        cmp     rbx, r13
        jae     fail
        cmp     byte ptr [r12 + rbx], ':'
        jne     fail
        inc     rbx
        call    skip_ws
        call    parse_value
        call    skip_ws
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        cmp     al, '}'
        je      obj_close
        cmp     al, ','
        jne     fail
        inc     rbx
        call    skip_ws
        cmp     rbx, r13
        jae     fail
        mov     al, byte ptr [r12 + rbx]
        jmp     obj_member
obj_close:
        inc     rbx
        ret

.section .rodata
ok_msg:
        .ascii  "ok\n"
ok_len = . - ok_msg
err
