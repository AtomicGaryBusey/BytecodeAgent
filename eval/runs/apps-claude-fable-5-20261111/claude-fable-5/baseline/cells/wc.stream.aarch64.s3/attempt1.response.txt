// wc for AArch64 Linux, no libc
	.equ SYS_read, 63
	.equ SYS_write, 64
	.equ SYS_exit, 93

	.bss
	.balign 8
buf:
	.space 4096
outbuf:
	.space 80

	.text
	.globl _start
_start:
	mov x19, #0		// lines
	mov x20, #0		// words
	mov x21, #0		// bytes
	mov x22, #0		// in_word flag

read_loop:
	mov x0, #0		// fd 0
	adrp x1, buf
	add x1, x1, :lo12:buf
	mov x2, #4096
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	ble done_reading
	add x21, x21, x0	// bytes += n
	adrp x1, buf
	add x1, x1, :lo12:buf
	add x3, x1, x0		// end pointer
scan:
	ldrb w4, [x1], #1
	cmp w4, #0x0a
	b.ne 1f
	add x19, x19, #1	// newline
1:
	// whitespace test: 0x09..0x0d or 0x20
	cmp w4, #0x20
	b.eq is_ws
	sub w5, w4, #0x09
	cmp w5, #4		// 0x09..0x0d
	b.ls is_ws
	// word byte
	cbnz x22, next_byte
	mov x22, #1
	add x20, x20, #1
	b next_byte
is_ws:
	mov x22, #0
next_byte:
	cmp x1, x3
	b.lo scan
	b read_loop

done_reading:
	// build output string backwards into outbuf
	adrp x9, outbuf
	add x9, x9, :lo12:outbuf
	add x9, x9, #80		// end of buffer
	mov w4, #0x0a
	strb w4, [x9, #-1]!
	mov x0, x21
	bl putnum
	mov w4, #0x20
	strb w4, [x9, #-1]!
	mov x0, x20
	bl putnum
	mov w4, #0x20
	strb w4, [x9, #-1]!
	mov x0, x19
	bl putnum

	// write(1, x9, len)
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	add x1, x1, #80
	sub x2, x1, x9		// length
	mov x1, x9
	mov x0, #1
	mov x8, #SYS_write
	svc #0

	mov x0, #0
	mov x8, #SYS_exit
	svc #0

// putnum: write decimal of x0 backwards at x9 (predecrement). Clobbers x0,x5,x6,x7,w4.
putnum:
	mov x7, #10
1:
	udiv x5, x0, x7		// quotient
	msub x6, x5, x7, x0	// remainder
	add w4, w6, #0x30
	strb w4, [x9, #-1]!
	mov x0, x5
	cbnz x0, 1b
	ret
