.text
.globl _start
_start:
    xor r12, r12                 # total bytes read
read_loop:
    mov edx, 32768
    sub edx, r12d
    jle read_done
    lea rsi, [rip + buf]
    add rsi, r12
    xor edi, edi                 # fd 0
    xor eax, eax                 # read
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # parse lines into (ptr,len) table
    xor r13, r13                 # line count
    lea rbx, [rip + buf]         # current line start
    mov rsi, rbx                 # cursor
    lea rdi, [rbx + r12]         # end of input
    lea r14, [rip + lines]
parse_loop:
    cmp rsi, rdi
    jae parse_end
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne parse_loop
    lea rax, [rsi - 1]
    sub rax, rbx                 # length without newline
    mov rcx, r13
    shl rcx, 4
    mov qword ptr [r14 + rcx], rbx
    mov qword ptr [r14 + rcx + 8], rax
    inc r13
    mov rbx, rsi
    jmp parse_loop
parse_end:
    cmp rbx, rdi
    jae sort_begin
    mov rax, rdi
    sub rax, rbx                 # final line without newline
    mov rcx, r13
    shl rcx, 4
    mov qword ptr [r14 + rcx], rbx
    mov qword ptr [r14 + rcx + 8], rax
    inc r13

sort_begin:
    cmp r13, 2
    jb output
    mov rbx, 1                   # i
outer:
    cmp rbx, r13
    jae output
    mov rcx, rbx
    shl rcx, 4
    mov r8, qword ptr [r14 + rcx]      # key ptr
    mov r9, qword ptr [r14 + rcx + 8]  # key len
    mov rdx, rbx                 # j
inner:
    test rdx, rdx
    jz place
    lea rcx, [rdx - 1]
    shl rcx, 4
    mov r10, qword ptr [r14 + rcx]     # rec[j-1] ptr
    mov r11, qword ptr [r14 + rcx + 8] # rec[j-1] len
    # compare rec[j-1] (r10,r11) with key (r8,r9), memcmp order
    mov rsi, r10
    mov rdi, r8
    mov rax, r11
    cmp rax, r9
    cmova rax, r9                # rax = min(len)
2:  cmp rax, 8
    jb 4f
    mov rcx, qword ptr [rsi]
    mov rbp, qword ptr [rdi]
    bswap rcx
    bswap rbp
    cmp rcx, rbp
    jne 6f
    add rsi, 8
    add rdi, 8
    sub rax, 8
    jmp 2b
4:  test rax, rax
    jz 5f
    movzx ecx, byte ptr [rsi]
    movzx ebp, byte ptr [rdi]
    cmp ecx, ebp
    jne 6f
    inc rsi
    inc rdi
    dec rax
    jmp 4b
5:  cmp r11, r9                  # common prefix: shorter first
    jbe place
    jmp shift
6:  jb place                     # rec[j-1] < key: key belongs here
shift:
    mov rcx, rdx
    shl rcx, 4
    mov qword ptr [r14 + rcx], r10
    mov qword ptr [r14 + rcx + 8], r11
    dec rdx
    jmp inner
place:
    mov rcx, rdx
    shl rcx, 4
    mov qword ptr [r14 + rcx], r8
    mov qword ptr [r14 + rcx + 8], r9
    inc rbx
    jmp outer

output:
    lea rdi, [rip + outbuf]
    xor rbx, rbx
out_loop:
    cmp rbx, r13
    jae out_write
    mov rcx, rbx
    shl rcx, 4
    mov rsi, qword ptr [r14 + rcx]
    mov rcx, qword ptr [r14 + rcx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp out_loop
out_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                 # total output bytes
write_loop:
    test rdx, rdx
    jz done
    mov eax, 1                   # write
    mov edi, 1                   # fd 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 60                  # exit
    xor edi, edi
    syscall

.bss
.balign 16
buf:
    .space 32776                 # input + padding for 8-byte compares
lines:
    .space 16000                 # 1000 x (ptr,len)
outbuf:
    .space 33776                 # input + 1000 newlines + padding
