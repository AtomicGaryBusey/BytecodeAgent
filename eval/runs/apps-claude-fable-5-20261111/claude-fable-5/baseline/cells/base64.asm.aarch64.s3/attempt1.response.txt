.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 8
inbuf:
    .space 49160
outbuf:
    .space 70000

.text
.globl _start
_start:
    // x21 = inbuf base
    adrp x21, inbuf
    add x21, x21, :lo12:inbuf
    mov x20, #0                 // total bytes read
read_loop:
    mov x2, #49160
    sub x2, x2, x20             // space left
    cbz x2, read_done
    mov x8, #63                 // read
    mov x0, #0
    add x1, x21, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    // x22 = dst, x23 = table, w24 = column, x25 = remaining
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    adrp x23, b64tab
    add x23, x23, :lo12:b64tab
    mov w24, #0
    mov x25, x20
enc_loop:
    cmp x25, #3
    b.lt enc_tail
    ldrb w5, [x21], #1
    ldrb w6, [x21], #1
    ldrb w7, [x21], #1
    lsl w5, w5, #16
    orr w5, w5, w6, lsl #8
    orr w5, w5, w7
    lsr w6, w5, #18
    ldrb w9, [x23, w6, uxtw]
    bl emit
    ubfx w6, w5, #12, #6
    ldrb w9, [x23, w6, uxtw]
    bl emit
    ubfx w6, w5, #6, #6
    ldrb w9, [x23, w6, uxtw]
    bl emit
    and w6, w5, #63
    ldrb w9, [x23, w6, uxtw]
    bl emit
    sub x25, x25, #3
    b enc_loop
enc_tail:
    cbz x25, enc_end
    ldrb w5, [x21], #1
    lsr w6, w5, #2
    ldrb w9, [x23, w6, uxtw]
    bl emit
    cmp x25, #2
    b.eq tail2
    // one trailing byte: c1 + "=="
    and w6, w5, #3
    lsl w6, w6, #4
    ldrb w9, [x23, w6, uxtw]
    bl emit
    mov w9, #61
    bl emit
    mov w9, #61
    bl emit
    b enc_end
tail2:
    // two trailing bytes: c1, c2 + "="
    ldrb w7, [x21], #1
    and w6, w5, #3
    lsl w6, w6, #4
    orr w6, w6, w7, lsr #4
    ldrb w9, [x23, w6, uxtw]
    bl emit
    and w6, w7, #15
    lsl w6, w6, #2
    ldrb w9, [x23, w6, uxtw]
    bl emit
    mov w9, #61
    bl emit
enc_end:
    cbz w24, do_write
    mov w9, #10
    strb w9, [x22], #1
do_write:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    sub x25, x22, x21           // output length
wr_loop:
    cbz x25, done
    mov x8, #64                 // write
    mov x0, #1
    mov x1, x21
    mov x2, x25
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    sub x25, x25, x0
    b wr_loop
done:
    mov x8, #93                 // exit
    mov x0, #0
    svc #0

// emit: store char in w9, handle 76-column line wrap
emit:
    strb w9, [x22], #1
    add w24, w24, #1
    cmp w24, #76
    b.ne 1f
    mov w9, #10
    strb w9, [x22], #1
    mov w24, #0
1:
    ret
