.text
.globl _start

_start:
        # ---- read all input ----
        xor     ebx, ebx                # total bytes read
read_loop:
        mov     edx, 32768
        sub     edx, ebx
        jle     read_done
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, rbx
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:

        # ---- parse lines ----
        lea     rsi, [rip + buf]        # start of current line
        lea     rdi, [rip + lines]      # next record
        xor     r12, r12                # line count
        xor     rcx, rcx                # index
parse_loop:
        cmp     rcx, rbx
        jae     parse_end
        lea     rdx, [rip + buf]
        cmp     byte ptr [rdx + rcx], 10
        jne     parse_next
        lea     rdx, [rdx + rcx]
        mov     qword ptr [rdi], rsi
        mov     rax, rdx
        sub     rax, rsi
        mov     qword ptr [rdi + 8], rax
        add     rdi, 16
        inc     r12
        lea     rsi, [rdx + 1]
parse_next:
        inc     rcx
        jmp     parse_loop
parse_end:
        lea     rdx, [rip + buf]
        add     rdx, rbx
        cmp     rsi, rdx                # final line without newline?
        jae     sort_start
        mov     qword ptr [rdi], rsi
        mov     rax, rdx
        sub     rax, rsi
        mov     qword ptr [rdi + 8], rax
        inc     r12

        # ---- insertion sort ----
sort_start:
        cmp     r12, 2
        jb      output
        lea     rbp, [rip + lines]
        mov     r13, 1                  # i
outer:
        cmp     r13, r12
        jae     output
        mov     rax, r13
        shl     rax, 4
        mov     r14, qword ptr [rbp + rax]      # key ptr
        mov     r15, qword ptr [rbp + rax + 8]  # key len
        lea     rbx, [r13 - 1]          # j
inner:
        mov     rax, rbx
        shl     rax, 4
        mov     rdi, qword ptr [rbp + rax]
        mov     rsi, qword ptr [rbp + rax + 8]
        mov     rdx, r14
        mov     rcx, r15
        call    cmp_lines
        test    eax, eax
        jle     inner_done
        mov     rax, rbx
        shl     rax, 4
        mov     rdi, qword ptr [rbp + rax]
        mov     rsi, qword ptr [rbp + rax + 8]
        mov     qword ptr [rbp + rax + 16], rdi
        mov     qword ptr [rbp + rax + 24], rsi
        dec     rbx
        jns     inner
inner_done:
        lea     rax, [rbx + 1]
        shl     rax, 4
        mov     qword ptr [rbp + rax], r14
        mov     qword ptr [rbp + rax + 8], r15
        inc     r13
        jmp     outer

        # ---- build output ----
output:
        lea     rdi, [rip + outbuf]
        lea     rbp, [rip + lines]
        xor     rbx, rbx
out_loop:
        cmp     rbx, r12
        jae     out_write
        mov     rax, rbx
        shl     rax, 4
        mov     rsi, qword ptr [rbp + rax]
        mov     rcx, qword ptr [rbp + rax + 8]
        rep movsb
        mov     byte ptr [rdi], 10
        inc     rdi
        inc     rbx
        jmp     out_loop
out_write:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
write_loop:
        test    rdx, rdx
        jle     done
        mov     eax, 1                  # write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
done:
        mov     eax, 60                 # exit
        xor     edi, edi
        syscall
hang:
        jmp     hang

# compare line (rdi,len rsi) with line (rdx,len rcx)
# returns eax < 0, == 0, > 0 ; clobbers rax, r8, r9, r10
cmp_lines:
        mov     r8, rsi
        cmp     r8, rcx
        cmova   r8, rcx                 # r8 = min(len)
        xor     r9, r9
1:      lea     rax, [r9 + 8]
        cmp     rax, r8
        ja      3f
        mov     rax, qword ptr [rdi + r9]
        mov     r10, qword ptr [rdx + r9]
        cmp     rax, r10
        jne     2f
        add     r9, 8
        jmp     1b
2:      bswap   rax
        bswap   r10
        cmp     rax, r10
        mov     eax, 1
        ja      6f
        mov     eax, -1
        ret
3:      cmp     r9, r8
        jae     5f
        movzx   eax, byte ptr [rdi + r9]
        movzx   r10d, byte ptr [rdx + r9]
        sub     eax, r10d
        jnz     6f
        inc     r9
        jmp     3b
5:      mov     eax, esi
        sub     eax, ecx
6:      ret

.bss
.balign 8
buf:    .space 32768
outbuf: .space 33792
lines:  .space 16000
