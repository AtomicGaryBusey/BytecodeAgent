.text
.globl _start
_start:
    xor     ebx, ebx                # total bytes read
read_loop:
    xor     eax, eax                # read
    xor     edi, edi                # fd 0
    lea     rsi, [rip + buf]
    add     rsi, rbx
    mov     edx, 65536
    sub     edx, ebx
    syscall
    test    rax, rax
    jle     read_done
    add     ebx, eax
    cmp     ebx, 65536
    jb      read_loop
read_done:
    xor     r12d, r12d              # current offset
row_loop:
    cmp     r12d, ebx
    jae     final
    mov     r13d, ebx
    sub     r13d, r12d
    cmp     r13d, 16
    jbe     1f
    mov     r13d, 16
1:
    # build the line
    lea     rdi, [rip + line]
    mov     eax, r12d
    call    hex8
    mov     word ptr [rdi], 0x2020  # two spaces
    add     rdi, 2
    lea     rsi, [rip + buf]
    add     rsi, r12
    xor     ecx, ecx                # i = 0
hexslot:
    cmp     ecx, r13d
    jae     blanks
    movzx   eax, byte ptr [rsi + rcx]
    mov     edx, eax
    shr     edx, 4
    mov     dl, byte ptr [hexdig + rdx]
    mov     byte ptr [rdi], dl
    and     eax, 15
    mov     al, byte ptr [hexdig + rax]
    mov     byte ptr [rdi + 1], al
    mov     byte ptr [rdi + 2], 0x20
    add     rdi, 3
    jmp     slot_next
blanks:
    mov     byte ptr [rdi], 0x20
    mov     word ptr [rdi + 1], 0x2020
    add     rdi, 3
slot_next:
    cmp     ecx, 7
    jne     2f
    mov     byte ptr [rdi], 0x20    # extra space after 8th slot
    inc     rdi
2:
    inc     ecx
    cmp     ecx, 16
    jb      hexslot
    mov     word ptr [rdi], 0x7c20  # " |"
    add     rdi, 2
    xor     ecx, ecx
ascloop:
    cmp     ecx, r13d
    jae     ascdone
    mov     al, byte ptr [rsi + rcx]
    cmp     al, 0x20
    jb      makedot
    cmp     al, 0x7e
    jbe     storech
makedot:
    mov     al, 0x2e                # '.'
storech:
    mov     byte ptr [rdi], al
    inc     rdi
    inc     ecx
    jmp     ascloop
ascdone:
    mov     word ptr [rdi], 0x0a7c  # "|\n"
    add     rdi, 2
    lea     rsi, [rip + line]
    mov     rdx, rdi
    sub     rdx, rsi
    call    write_all
    add     r12d, 16
    jmp     row_loop
final:
    lea     rdi, [rip + line]
    mov     eax, ebx
    call    hex8
    mov     byte ptr [rdi], 0x0a
    lea     rsi, [rip + line]
    mov     edx, 9
    call    write_all
    mov     eax, 231                # exit_group
    xor     edi, edi
    syscall

# hex8: write eax as 8 lowercase hex digits at rdi, advance rdi by 8
hex8:
    mov     ecx, 8
1:
    rol     eax, 4
    mov     edx, eax
    and     edx, 15
    mov     dl, byte ptr [hexdig + rdx]
    mov     byte ptr [rdi], dl
    inc     rdi
    dec     ecx
    jnz     1b
    ret

# write_all: write rdx bytes at rsi to stdout
write_all:
    mov     eax, 1                  # write
    mov     edi, 1                  # fd 1
    syscall
    test    rax, rax
    jle     3f
    add     rsi, rax
    sub     rdx, rax
    jnz     write_all
3:
    ret

.section .rodata
hexdig:
    .ascii  "0123456789abcdef"

.bss
.balign 16
buf:
    .zero   65536
line:
    .zero   128
