.section .rodata
ok_msg:   .ascii "ok\n"
err_pre:  .ascii "error at "
lit_true: .ascii "true"
lit_false:.ascii "false"
lit_null: .ascii "null"

.bss
.balign 8
buf:    .space 65600
numbuf: .space 32

.text
.globl _start
_start:
        xor     r13, r13                # total bytes read
read_loop:
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, r13
        mov     edx, 65600
        sub     edx, r13d
        syscall
        test    rax, rax
        jle     read_done
        add     r13, rax
        jmp     read_loop
read_done:
        lea     rbx, [rip + buf]        # base pointer
        xor     r12, r12                # position
        call    skip_ws
        call    parse_value
        call    skip_ws
        cmp     r12, r13
        jne     error
        mov     eax, 1
        mov     edi, 1
        lea     rsi, [rip + ok_msg]
        mov     edx, 3
        syscall
        mov     eax, 60
        xor     edi, edi
        syscall

# ---------- skip whitespace ----------
skip_ws:
1:      cmp     r12, r13
        jae     2f
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, 0x20
        je      3f
        cmp     al, 0x09
        je      3f
        cmp     al, 0x0A
        je      3f
        cmp     al, 0x0D
        jne     2f
3:      inc     r12
        jmp     1b
2:      ret

# ---------- value dispatch ----------
parse_value:
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '{'
        je      parse_object
        cmp     al, '['
        je      parse_array
        cmp     al, '"'
        je      parse_string
        cmp     al, '-'
        je      parse_number
        cmp     al, '0'
        jb      1f
        cmp     al, '9'
        jbe     parse_number
1:      cmp     al, 't'
        je      2f
        cmp     al, 'f'
        je      3f
        cmp     al, 'n'
        je      4f
        jmp     error
2:      lea     rsi, [rip + lit_true]
        mov     ecx, 4
        jmp     parse_lit
3:      lea     rsi, [rip + lit_false]
        mov     ecx, 5
        jmp     parse_lit
4:      lea     rsi, [rip + lit_null]
        mov     ecx, 4
parse_lit:
1:      cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, byte ptr [rsi]
        jne     error
        inc     r12
        inc     rsi
        dec     ecx
        jnz     1b
        ret

# ---------- string (at opening quote) ----------
parse_string:
        inc     r12
ps_loop:
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '"'
        je      ps_done
        cmp     al, 0x5C
        je      ps_esc
        cmp     al, 0x20
        jb      error
        inc     r12
        jmp     ps_loop
ps_done:
        inc     r12
        ret
ps_esc:
        inc     r12
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, 'u'
        je      ps_uni
        cmp     al, '"'
        je      ps_ok
        cmp     al, 0x5C
        je      ps_ok
        cmp     al, '/'
        je      ps_ok
        cmp     al, 'b'
        je      ps_ok
        cmp     al, 'f'
        je      ps_ok
        cmp     al, 'n'
        je      ps_ok
        cmp     al, 'r'
        je      ps_ok
        cmp     al, 't'
        je      ps_ok
        jmp     error
ps_ok:
        inc     r12
        jmp     ps_loop
ps_uni:
        inc     r12
        mov     ecx, 4
ps_hex:
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '0'
        jb      error
        cmp     al, '9'
        jbe     ps_hok
        or      al, 0x20
        cmp     al, 'a'
        jb      error
        cmp     al, 'f'
        ja      error
ps_hok:
        inc     r12
        dec     ecx
        jnz     ps_hex
        jmp     ps_loop

# ---------- number (at '-' or digit) ----------
parse_number:
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '-'
        jne     1f
        inc     r12
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
1:      cmp     al, '0'
        jne     2f
        inc     r12
        jmp     pn_frac
2:      cmp     al, '1'
        jb      error
        cmp     al, '9'
        ja      error
        inc     r12
3:      cmp     r12, r13
        jae     pn_ret
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '0'
        jb      pn_frac2
        cmp     al, '9'
        ja      pn_frac2
        inc     r12
        jmp     3b
pn_frac:
        cmp     r12, r13
        jae     pn_ret
        movzx   eax, byte ptr [rbx + r12]
pn_frac2:                               # eax loaded, r12 < r13
        cmp     al, '.'
        jne     pn_exp
        inc     r12
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '0'
        jb      error
        cmp     al, '9'
        ja      error
        inc     r12
4:      cmp     r12, r13
        jae     pn_ret
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '0'
        jb      pn_exp
        cmp     al, '9'
        ja      pn_exp
        inc     r12
        jmp     4b
pn_exp:                                 # eax loaded, r12 < r13
        cmp     al, 'e'
        je      5f
        cmp     al, 'E'
        jne     pn_ret
5:      inc     r12
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '+'
        je      6f
        cmp     al, '-'
        jne     7f
6:      inc     r12
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
7:      cmp     al, '0'
        jb      error
        cmp     al, '9'
        ja      error
        inc     r12
8:      cmp     r12, r13
        jae     pn_ret
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '0'
        jb      pn_ret
        cmp     al, '9'
        ja      pn_ret
        inc     r12
        jmp     8b
pn_ret:
        ret

# ---------- object (at '{') ----------
parse_object:
        inc     r12
        call    skip_ws
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '}'
        jne     po_member
        inc     r12
        ret
po_member:                              # eax loaded, r12 < r13
        cmp     al, '"'
        jne     error
        call    parse_string
        call    skip_ws
        cmp     r12, r13
        jae     error
        cmp     byte ptr [rbx + r12], ':'
        jne     error
        inc     r12
        call    skip_ws
        call    parse_value
        call    skip_ws
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, '}'
        je      po_close
        cmp     al, ','
        jne     error
        inc     r12
        call    skip_ws
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        jmp     po_member
po_close:
        inc     r12
        ret

# ---------- array (at '[') ----------
parse_array:
        inc     r12
        call    skip_ws
        cmp     r12, r13
        jae     error
        cmp     byte ptr [rbx + r12], ']'
        jne     pa_elem
        inc     r12
        ret
pa_elem:
        call    parse_value
        call    skip_ws
        cmp     r12, r13
        jae     error
        movzx   eax, byte ptr [rbx + r12]
        cmp     al, ']'
        je      pa_close
        cmp     al, ','
        jne     error
        inc     r12
        call    skip_ws
        jmp     pa_elem
pa_close:
        inc     r12
        ret

# ---------- error: print "error at N\n", exit 1 ----------
error:
        mov     rax, r12
        lea     rsi, [rip + numbuf + 31]
        mov     byte ptr [rsi], 10      # '\n'
        mov     ecx, 10
e_div:
        dec     rsi
        xor     edx, edx
        div     rcx
        add     dl, '0'
        mov     byte ptr [rsi], dl
        test    rax, rax
        jnz     e_div
        push    rsi                     # save start of digits
        mov     eax, 1
        mov     edi, 1
        lea     rsi, [rip + err_pre]
        mov     edx, 9
        syscall
        pop     rsi
        lea     rdx, [rip + numbuf + 32]
        sub     rdx, rsi
        mov     eax, 1
        mov     edi, 1
        syscall
        mov     eax, 60
        mov     edi, 1
        syscall
