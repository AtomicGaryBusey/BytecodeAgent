.section .rodata
hexd:   .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:  .space 65537
outbuf: .space 400000

.text
.globl _start
_start:
        xor     r13, r13                # total bytes read
read_loop:
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + inbuf]
        add     rsi, r13
        mov     edx, 65537
        sub     edx, r13d
        syscall
        test    rax, rax
        jle     read_done
        add     r13, rax
        jmp     read_loop
read_done:
        lea     r14, [rip + outbuf]     # output write pointer
        lea     rbx, [rip + inbuf]
        xor     r12, r12                # current offset

line_loop:
        cmp     r12, r13
        jae     lines_done
        mov     r15, r13                # n = min(16, total - off)
        sub     r15, r12
        cmp     r15, 16
        jbe     1f
        mov     r15, 16
1:
        lea     rbp, [rbx + r12]        # pointer to this row
        mov     eax, r12d
        call    emit_hex8
        mov     word ptr [r14], 0x2020  # two spaces
        add     r14, 2

        xor     ecx, ecx                # i = 0..15
byte_loop:
        cmp     rcx, r15
        jae     byte_pad
        movzx   eax, byte ptr [rbp + rcx]
        lea     rsi, [rip + hexd]
        mov     edx, eax
        shr     edx, 4
        mov     dl, byte ptr [rsi + rdx]
        mov     byte ptr [r14], dl
        and     eax, 15
        mov     al, byte ptr [rsi + rax]
        mov     byte ptr [r14 + 1], al
        mov     byte ptr [r14 + 2], 0x20
        add     r14, 3
        jmp     byte_after
byte_pad:
        mov     byte ptr [r14], 0x20
        mov     byte ptr [r14 + 1], 0x20
        mov     byte ptr [r14 + 2], 0x20
        add     r14, 3
byte_after:
        cmp     rcx, 7                  # extra space after 8th slot
        jne     2f
        mov     byte ptr [r14], 0x20
        inc     r14
2:
        inc     rcx
        cmp     rcx, 16
        jb      byte_loop

        mov     byte ptr [r14], 0x20    # " |"
        mov     byte ptr [r14 + 1], 0x7c
        add     r14, 2

        xor     ecx, ecx                # ASCII column
char_loop:
        cmp     rcx, r15
        jae     chars_done
        movzx   eax, byte ptr [rbp + rcx]
        cmp     al, 0x20
        jb      char_dot
        cmp     al, 0x7e
        jbe     char_ok
char_dot:
        mov     al, 0x2e
char_ok:
        mov     byte ptr [r14], al
        inc     r14
        inc     rcx
        jmp     char_loop
chars_done:
        mov     byte ptr [r14], 0x7c    # "|\n"
        mov     byte ptr [r14 + 1], 10
        add     r14, 2

        add     r12, 16
        jmp     line_loop

lines_done:
        mov     eax, r13d               # final offset line
        call    emit_hex8
        mov     byte ptr [r14], 10
        inc     r14

        lea     rsi, [rip + outbuf]     # write everything out
        mov     rdx, r14
        sub     rdx, rsi
write_loop:
        test    rdx, rdx
        jle     done
        mov     eax, 1                  # write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop

done:
        mov     eax, 231                # exit_group
        xor     edi, edi
        syscall

# emit_hex8: write eax as 8 lowercase hex digits at [r14], advance r14.
emit_hex8:
        mov     ecx, 8
        lea     rsi, [rip + hexd]
1:
        rol     eax, 4
        mov     edx, eax
        and     edx, 15
        mov     dl, byte ptr [rsi + rdx]
        mov     byte ptr [r14], dl
        inc     r14
        dec     ecx
        jnz     1b
        ret
