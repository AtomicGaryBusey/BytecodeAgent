.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 65537

.text
.globl _start
_start:
    # read all input into inbuf
    xor r12, r12            # total bytes read
1:
    xor edi, edi            # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle 2f
    xor eax, eax            # read
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    # process lines: r13 = current pos in inbuf, r14 = out pos
    xor r13, r13
    xor r14, r14
3:
    cmp r13, r12
    jae 7f                  # done scanning
    # find newline from r13
    mov rbx, r13            # line start
4:
    cmp r13, r12
    jae 5f
    lea rsi, [rip + inbuf]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x0A
    je 5f
    inc r13
    jmp 4b
5:
    # line is [rbx, r13); copy reversed into outbuf at r14
    mov rcx, r13            # one past last byte
6:
    cmp rcx, rbx
    je 9f
    dec rcx
    lea rsi, [rip + inbuf]
    movzx eax, byte ptr [rsi + rcx]
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], al
    inc r14
    jmp 6b
9:
    # append newline
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], 0x0A
    inc r14
    # skip the newline in input if present
    cmp r13, r12
    jae 7f
    inc r13
    jmp 3b
7:
    # write outbuf[0..r14)
    xor r13, r13            # written so far
8:
    cmp r13, r14
    jae 10f
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r13
    mov rdx, r14
    sub rdx, r13
    mov eax, 1
    syscall
    test rax, rax
    jle 10f
    add r13, rax
    jmp 8b
10:
    xor edi, edi
    mov eax, 60
    syscall
