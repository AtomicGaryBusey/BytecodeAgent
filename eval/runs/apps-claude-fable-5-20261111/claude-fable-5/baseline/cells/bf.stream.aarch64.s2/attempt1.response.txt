// bf - Brainfuck interpreter for AArch64 Linux, freestanding

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 8
buf:
    .zero 65536
tape:
    .zero 30000
.balign 4
jumptab:
    .zero 262144

.text
.globl _start
_start:
    // x19 = buffer base
    adrp x19, buf
    add x19, x19, :lo12:buf

    // read all of stdin into buf; x26 = total bytes
    mov x26, #0
1:
    mov x2, #65536
    sub x2, x2, x26
    cbz x2, 2f
    mov x0, #0
    add x1, x19, x26
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le 2f
    add x26, x26, x0
    b 1b
2:
    // find first '!' -> x20 = program length
    mov x20, #0
3:
    cmp x20, x26
    b.ge 4f                     // no '!': program = everything
    ldrb w0, [x19, x20]
    cmp w0, #0x21
    b.eq 5f
    add x20, x20, #1
    b 3b
4:
    // input empty
    add x21, x19, x26           // input ptr
    mov x22, x21                // input end
    b 6f
5:
    add x21, x19, x20
    add x21, x21, #1            // input ptr = just after '!'
    add x22, x19, x26           // input end
6:
    // bracket check + build jump table
    adrp x24, jumptab
    add x24, x24, :lo12:jumptab
    mov x27, #0                 // bracket stack depth
    mov x25, #0                 // index
7:
    cmp x25, x20
    b.ge 11f
    ldrb w0, [x19, x25]
    cmp w0, #0x5B               // '['
    b.eq 9f
    cmp w0, #0x5D               // ']'
    b.eq 10f
8:
    add x25, x25, #1
    b 7b
9:
    str x25, [sp, #-16]!
    add x27, x27, #1
    b 8b
10:
    cbz x27, badprog
    ldr x0, [sp], #16
    sub x27, x27, #1
    str w25, [x24, x0, lsl #2]  // match['['] = ']'
    str w0, [x24, x25, lsl #2]  // match[']'] = '['
    b 8b
11:
    cbnz x27, badprog

    // execute
    adrp x23, tape
    add x23, x23, :lo12:tape    // data pointer
    mov x25, #0                 // pc
exec:
    cmp x25, x20
    b.ge done
    ldrb w0, [x19, x25]
    add x25, x25, #1
    cmp w0, #0x3E               // '>'
    b.eq c_right
    cmp w0, #0x3C               // '<'
    b.eq c_left
    cmp w0, #0x2B               // '+'
    b.eq c_plus
    cmp w0, #0x2D               // '-'
    b.eq c_minus
    cmp w0, #0x2E               // '.'
    b.eq c_out
    cmp w0, #0x2C               // ','
    b.eq c_in
    cmp w0, #0x5B               // '['
    b.eq c_open
    cmp w0, #0x5D               // ']'
    b.eq c_close
    b exec

c_right:
    add x23, x23, #1
    b exec
c_left:
    sub x23, x23, #1
    b exec
c_plus:
    ldrb w0, [x23]
    add w0, w0, #1
    strb w0, [x23]
    b exec
c_minus:
    ldrb w0, [x23]
    sub w0, w0, #1
    strb w0, [x23]
    b exec
c_out:
    mov x0, #1
    mov x1, x23
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    b exec
c_in:
    cmp x21, x22
    b.ge 1f
    ldrb w0, [x21], #1
    strb w0, [x23]
    b exec
1:
    strb wzr, [x23]
    b exec
c_open:
    ldrb w0, [x23]
    cbnz w0, exec
    sub x0, x25, #1             // index of '['
    ldr w0, [x24, x0, lsl #2]   // matching ']'
    add x25, x0, #1
    b exec
c_close:
    ldrb w0, [x23]
    cbz w0, exec
    sub x0, x25, #1             // index of ']'
    ldr w0, [x24, x0, lsl #2]   // matching '['
    add x25, x0, #1
    b exec

done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

badprog:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
