	.text
	.globl _start
_start:
	// buffers
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	adrp	x21, outbuf
	add	x21, x21, :lo12:outbuf
	mov	x20, #0			// total bytes read

read_loop:
	mov	x2, #65536
	subs	x2, x2, x20		// remaining space
	b.le	read_done
	mov	x0, #0			// fd 0
	add	x1, x19, x20
	mov	x8, #63			// read
	svc	#0
	cbz	x0, read_done
	tbnz	x0, #63, read_done	// error: treat as EOF
	add	x20, x20, x0
	b	read_loop

read_done:
	mov	x22, #0			// output length
	mov	x23, #0			// input position

line_loop:
	cmp	x23, x20
	b.ge	write_out
	mov	x24, x23		// line start
scan:
	ldrb	w0, [x19, x23]
	add	x23, x23, #1
	cmp	w0, #10
	b.eq	got_nl
	cmp	x23, x20
	b.lt	scan
	mov	x25, x20		// final line, no newline
	b	emit
got_nl:
	sub	x25, x23, #1		// end of line (exclusive)
emit:
	mov	x26, x25
rev_loop:
	cmp	x26, x24
	b.le	rev_done
	sub	x26, x26, #1
	ldrb	w0, [x19, x26]
	strb	w0, [x21, x22]
	add	x22, x22, #1
	b	rev_loop
rev_done:
	mov	w0, #10
	strb	w0, [x21, x22]
	add	x22, x22, #1
	b	line_loop

write_out:
	mov	x27, #0			// bytes written
write_loop:
	cmp	x27, x22
	b.ge	done
	mov	x0, #1			// fd 1
	add	x1, x21, x27
	sub	x2, x22, x27
	mov	x8, #64			// write
	svc	#0
	tbnz	x0, #63, done		// error: give up
	add	x27, x27, x0
	b	write_loop

done:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

	.bss
	.balign	8
inbuf:
	.space	65536
outbuf:
	.space	65537
