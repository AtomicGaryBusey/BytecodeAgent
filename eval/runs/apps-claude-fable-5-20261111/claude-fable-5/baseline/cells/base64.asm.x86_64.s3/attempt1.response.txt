.section .rodata
b64tab:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
	.balign 16
inbuf:
	.space 49152
outbuf:
	.space 70000

.text
.globl _start
_start:
	xor ebx, ebx			# total bytes read
read_loop:
	mov edx, 49152
	sub edx, ebx
	jz done_read
	lea rsi, [rip + inbuf]
	add rsi, rbx
	xor eax, eax			# read
	xor edi, edi			# fd 0
	syscall
	test rax, rax
	jle done_read
	add rbx, rax
	jmp read_loop

done_read:
	lea rsi, [rip + inbuf]
	lea rdi, [rip + outbuf]
	mov rdx, rbx			# remaining input bytes
	xor r8d, r8d			# chars on current line
	lea r9, [rip + b64tab]

enc_loop:
	cmp rdx, 3
	jb tail
	movzx eax, byte ptr [rsi]
	shl eax, 16
	movzx ecx, byte ptr [rsi + 1]
	shl ecx, 8
	or eax, ecx
	movzx ecx, byte ptr [rsi + 2]
	or eax, ecx
	add rsi, 3
	sub rdx, 3

	mov ecx, eax
	shr ecx, 18
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi], cl
	mov ecx, eax
	shr ecx, 12
	and ecx, 63
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi + 1], cl
	mov ecx, eax
	shr ecx, 6
	and ecx, 63
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi + 2], cl
	mov ecx, eax
	and ecx, 63
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi + 3], cl
	add rdi, 4

	add r8d, 4
	cmp r8d, 76
	jne enc_loop
	mov byte ptr [rdi], 10
	inc rdi
	xor r8d, r8d
	jmp enc_loop

tail:
	test rdx, rdx
	jz finish
	movzx eax, byte ptr [rsi]
	shl eax, 16
	cmp rdx, 2
	jb have_val
	movzx ecx, byte ptr [rsi + 1]
	shl ecx, 8
	or eax, ecx
have_val:
	mov ecx, eax
	shr ecx, 18
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi], cl
	mov ecx, eax
	shr ecx, 12
	and ecx, 63
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi + 1], cl
	cmp rdx, 2
	je two_tail
	mov word ptr [rdi + 2], 0x3D3D	# "=="
	jmp tail_done
two_tail:
	mov ecx, eax
	shr ecx, 6
	and ecx, 63
	mov cl, byte ptr [r9 + rcx]
	mov byte ptr [rdi + 2], cl
	mov byte ptr [rdi + 3], 0x3D	# "="
tail_done:
	add rdi, 4
	add r8d, 4

finish:
	test r8d, r8d
	jz do_write
	mov byte ptr [rdi], 10
	inc rdi

do_write:
	lea rsi, [rip + outbuf]
	sub rdi, rsi
	mov rbx, rdi			# bytes remaining to write
write_loop:
	test rbx, rbx
	jz exit0
	mov eax, 1			# write
	mov edi, 1			# fd 1
	mov rdx, rbx
	syscall
	test rax, rax
	jle exit0
	add rsi, rax
	sub rbx, rax
	jmp write_loop

exit0:
	mov eax, 60			# exit
	xor edi, edi
	syscall
