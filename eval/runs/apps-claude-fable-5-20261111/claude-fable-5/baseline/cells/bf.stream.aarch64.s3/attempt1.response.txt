.text
.globl _start
_start:
        // --- read all of stdin into buf ---
        adrp    x19, buf
        add     x19, x19, :lo12:buf
        mov     x20, #0                 // total bytes read
1:
        mov     x2, #65536
        subs    x2, x2, x20             // space remaining
        beq     2f
        mov     x0, #0
        add     x1, x19, x20
        mov     x8, #63                 // read
        svc     #0
        cmp     x0, #0
        ble     2f
        add     x20, x20, x0
        b       1b
2:
        // --- split at first '!' ---
        mov     x25, #0
3:
        cmp     x25, x20
        beq     4f                      // no '!'
        ldrb    w0, [x19, x25]
        cmp     w0, #0x21
        beq     5f
        add     x25, x25, #1
        b       3b
4:                                      // no bang: input empty
        add     x21, x19, x20           // input start
        mov     x22, x21                // input end
        b       6f
5:                                      // bang at x25
        add     x21, x19, x25
        add     x21, x21, #1            // input start
        add     x22, x19, x20           // input end
        mov     x20, x25                // program length
6:
        // --- bracket matching, build jump table ---
        adrp    x24, jtab
        add     x24, x24, :lo12:jtab
        adrp    x27, bstk
        add     x27, x27, :lo12:bstk
        mov     x26, #0                 // stack depth
        mov     x25, #0                 // position
7:
        cmp     x25, x20
        beq     11f
        ldrb    w0, [x19, x25]
        cmp     w0, #0x5B               // '['
        beq     9f
        cmp     w0, #0x5D               // ']'
        beq     10f
8:
        add     x25, x25, #1
        b       7b
9:                                      // push '[' position
        str     w25, [x27, x26, lsl #2]
        add     x26, x26, #1
        b       8b
10:                                     // pop and record match
        cbz     x26, bad
        sub     x26, x26, #1
        ldr     w1, [x27, x26, lsl #2]  // '[' position
        str     w25, [x24, x1, uxtw #2] // jtab['['] = ']'
        str     w1, [
