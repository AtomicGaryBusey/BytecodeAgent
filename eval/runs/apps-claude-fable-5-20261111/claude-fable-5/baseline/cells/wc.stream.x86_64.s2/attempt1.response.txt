.text
.globl _start
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # in_word flag (0 = not in word)

read_loop:
    xor edi, edi            # fd 0
    lea rsi, [rip + buffer]
    mov edx, 65536
    xor eax, eax            # read
    syscall
    test rax, rax
    jle done_reading

    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax            # count

scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a
    jne 1f
    inc r12
1:
    # is whitespace?
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0d
    jbe is_ws
not_ws:
    test r15, r15
    jnz 2f
    inc r13
    mov r15, 1
    jmp 2f
is_ws:
    xor r15, r15
2:
    dec rcx
    jnz scan
    jmp read_loop

done_reading:
    # build output string backwards in outbuf
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 0x0a
    mov rax, r14
    call put_num
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call put_num
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call put_num

    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rdi
    mov edi, 1
    mov eax, 1              # write
    syscall

    xor edi, edi
    mov eax, 60             # exit
    syscall

# put_num: rax = value, rdi points just past where digits go; writes digits backwards,
# returns rdi pointing at first digit
put_num:
    mov rbx, 10
1:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.balign 16
buffer:
    .space 65536
outbuf:
    .space 64
