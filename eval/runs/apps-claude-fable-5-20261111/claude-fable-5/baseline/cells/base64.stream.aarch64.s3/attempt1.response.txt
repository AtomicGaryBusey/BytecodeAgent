.section .rodata
b64tab:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
	.balign 8
inbuf:
	.space 49152
outbuf:
	.space 66560

.text
.globl _start

// putb: write byte in w0 to output buffer, handle 76-char line wrap
putb:
	strb w0, [x22], #1
	add x23, x23, #1
	cmp x23, #76
	b.ne 1f
	mov w0, #10
	strb w0, [x22], #1
	mov x23, #0
1:
	ret

// emit: w0 = 6-bit value, output its base64 character
emit:
	ldrb w0, [x24, w0, uxtw]
	b putb

_start:
	// read all of stdin into inbuf
	mov x19, #0                  // total bytes read
	adrp x25, inbuf
	add x25, x25, :lo12:inbuf
read_loop:
	mov x0, #0
	add x1, x25, x19
	mov x2, #49152
	sub x2, x2, x19
	mov x8, #63
	svc #0
	cbz x0, read_done
	add x19, x19, x0
	b read_loop
read_done:
	mov x20, x25                 // input cursor
	add x21, x25, x19            // input end
	adrp x22, outbuf
	add x22, x22, :lo12:outbuf   // output cursor
	mov x23, #0                  // chars on current line
	adrp x24, b64tab
	add x24, x24, :lo12:b64tab

enc_loop:
	sub x9, x21, x20
	cmp x9, #3
	b.lt tail
	ldrb w10, [x20], #1
	ldrb w11, [x20], #1
	ldrb w12, [x20], #1
	lsl w10, w10, #16
	orr w10, w10, w11, lsl #8
	orr w25, w10, w12            // 24-bit group
	lsr w0, w25, #18
	bl emit
	ubfx w0, w25, #12, #6
	bl emit
	ubfx w0, w25, #6, #6
	bl emit
	and w0, w25, #63
	bl emit
	b enc_loop

tail:
	cbz x9, flush
	ldrb w10, [x20], #1
	cmp x9, #2
	b.eq two
	// one leftover byte -> 2 chars + "=="
	lsr w0, w10, #2
	bl emit
	lsl w0, w10, #4
	and w0, w0, #63
	bl emit
	mov w0, #61
	bl putb
	mov w0, #61
	bl putb
	b flush
two:
	// two leftover bytes -> 3 chars + "="
	ldrb w11, [x20], #1
	lsr w0, w10, #2
	bl emit
	lsl w0, w10, #4
	orr w0, w0, w11, lsr #4
	and w0, w0, #63
	bl emit
	lsl w0, w11, #2
	and w0, w0, #63
	bl emit
	mov w0, #61
	bl putb

flush:
	// final newline if the last line is non-empty
	cbz x23, do_write
	mov w0, #10
	strb w0, [x22], #1

do_write:
	adrp x25, outbuf
	add x25, x25, :lo12:outbuf
	sub x26, x22, x25            // total output length
wr_loop:
	cbz x26, done
	mov x0, #1
	mov x1, x25
	mov x2, x26
	mov x8, #64
	svc #0
	add x25, x25, x0
	sub x26, x26, x0
	b wr_loop

done:
	mov x0, #0
	mov x8, #93
	svc #0
