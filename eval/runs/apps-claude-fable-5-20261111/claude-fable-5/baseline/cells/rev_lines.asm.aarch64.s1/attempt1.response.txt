	.text
	.globl _start
_start:
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf	// inbuf base
	mov	x20, #0			// total bytes read
read_loop:
	mov	x2, #65536
	sub	x2, x2, x20
	cbz	x2, read_done
	mov	x0, #0
	add	x1, x19, x20
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop
read_done:
	add	x21, x19, x20		// end of input
	mov	x22, x19		// current pointer
	adrp	x23, outbuf
	add	x23, x23, :lo12:outbuf
	mov	x24, x23		// output pointer
line_loop:
	cmp	x22, x21
	b.hs	output
	mov	x25, x22		// scan for newline
find_nl:
	cmp	x25, x21
	b.hs	found
	ldrb	w0, [x25]
	cmp	w0, #10
	b.eq	found
	add	x25, x25, #1
	b	find_nl
found:
	mov	x26, x25		// q = line end
rev:
	cmp	x26, x22
	b.ls	rev_done
	sub	x26, x26, #1
	ldrb	w0, [x26]
	strb	w0, [x24], #1
	b	rev
rev_done:
	mov	w0, #10
	strb	w0, [x24], #1
	mov	x22, x25
	cmp	x22, x21
	b.hs	line_loop
	add	x22, x22, #1		// skip the newline
	b	line_loop
output:
	mov	x25, x23		// write position
	sub	x26, x24, x23		// bytes remaining
write_loop:
	cbz	x26, done
	mov	x0, #1
	mov	x1, x25
	mov	x2, x26
	mov	x8, #64			// write
	svc	#0
	cmp	x0, #0
	b.le	done
	add	x25, x25, x0
	sub	x26, x26, x0
	b	write_loop
done:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

	.bss
	.balign	8
inbuf:
	.space	65536
outbuf:
	.space	65544
