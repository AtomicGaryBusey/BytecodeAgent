	.text
	.globl _start
_start:
	// read all input
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0			// total bytes read
1:
	mov	x8, #63
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #65536
	sub	x2, x2, x20
	cbz	x2, 2f
	svc	#0
	cmp	x0, #0
	ble	2f
	add	x20, x20, x0
	b	1b
2:
	// process lines
	adrp	x22, outbuf
	add	x22, x22, :lo12:outbuf
	mov	x23, #0			// output length
	mov	x21, #0			// input position
3:
	cmp	x21, x20
	bge	7f			// no more lines
	// find newline from x21
	mov	x4, x21			// scan index
4:
	cmp	x4, x20
	bge	5f
	ldrb	w5, [x19, x4]
	cmp	w5, #10
	beq	5f
	add	x4, x4, #1
	b	4b
5:
	// line is [x21, x4), copy reversed
	mov	x6, x4
6:
	cmp	x6, x21
	ble	8f
	sub	x6, x6, #1
	ldrb	w5, [x19, x6]
	strb	w5, [x22, x23]
	add	x23, x23, #1
	b	6b
8:
	mov	w5, #10
	strb	w5, [x22, x23]
	add	x23, x23, #1
	add	x21, x4, #1
	b	3b
7:
	// write output, handling short writes
	mov	x24, #0			// bytes written
9:
	cmp	x24, x23
	bge	10f
	mov	x8, #64
	mov	x0, #1
	add	x1, x22, x24
	sub	x2, x23, x24
	svc	#0
	cmp	x0, #0
	ble	10f
	add	x24, x24, x0
	b	9b
10:
	mov	x8, #93
	mov	x0, #0
	svc	#0

	.bss
	.balign	8
inbuf:
	.space	65536
outbuf:
	.space	65537
