# json_lint - validate JSON from stdin, report first error offset

.equ BUFSZ, 65600

.section .rodata
ok_msg:      .ascii "ok\n"
err_pre:     .ascii "error at "
.equ err_pre_len, . - err_pre
lit_true_s:  .ascii "rue"
lit_false_s: .ascii "alse"
lit_null_s:  .ascii "ull"

.bss
.balign 8
buf:    .space BUFSZ
numbuf: .space 32

.text
.globl _start
_start:
    # ---- read all of stdin into buf ----
    xor ebx, ebx                 # total bytes read
rd_loop:
    mov edx, BUFSZ
    sub edx, ebx
    jle rd_done
    xor eax, eax                 # sys_read
    xor edi, edi                 # fd 0
    lea rsi, [rip + buf]
    add rsi, rbx
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd_loop
rd_done:
    mov r12, rbx                 # r12 = input length
    lea rbp, [rip + buf]         # rbp = buffer base
    xor esi, esi                 # rsi = position

    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne err_here

    # ---- print "ok\n", exit 0 ----
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 231
    syscall

# ---- error: print "error at N\n", exit 1 (N = rsi) ----
err_here:
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 10       # '\n'
    mov rax, rsi
    mov r9, 10
eh_dig:
    dec rdi
    xor edx, edx
    div r9
    add dl, '0'
    mov byte ptr [rdi], dl
    test rax, rax
    jnz eh_dig
    mov r13, rdi                 # start of digits
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_pre]
    mov edx, err_pre_len
    syscall
    lea rdx, [rip + numbuf + 32]
    sub rdx, r13
    mov rsi, r13
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 231
    syscall

# ---- skip whitespace (space, tab, LF, CR) ----
skip_ws:
1:  cmp rsi, r12
    jae 2f
    movzx eax, byte ptr [rbp + rsi]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    jne 2f
3:  inc rsi
    jmp 1b
2:  ret

# ---- parse one value, rsi at its first byte ----
parse_value:
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je pv_true
    cmp al, 'f'
    je pv_false
    cmp al, 'n'
    je pv_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb err_here
    cmp al, '9'
    jbe parse_number
    jmp err_here

pv_true:
    lea rdx, [rip + lit_true_s]
    mov ecx, 3
    jmp match_lit
pv_false:
    lea rdx, [rip + lit_false_s]
    mov ecx, 4
    jmp match_lit
pv_null:
    lea rdx, [rip + lit_null_s]
    mov ecx, 3
match_lit:
    inc rsi
1:  cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, byte ptr [rdx]
    jne err_here
    inc rsi
    inc rdx
    dec ecx
    jnz 1b
    ret

# ---- string, rsi at opening quote ----
parse_string:
    inc rsi
ps_loop:
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '"'
    je ps_done
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb err_here
    inc rsi
    jmp ps_loop
ps_done:
    inc rsi
    ret
ps_esc:
    inc rsi
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, 'u'
    je ps_uni
    cmp al, '"'
    je ps_ok
    cmp al, 0x5C
    je ps_ok
    cmp al, '/'
    je ps_ok
    cmp al, 'b'
    je ps_ok
    cmp al, 'f'
    je ps_ok
    cmp al, 'n'
    je ps_ok
    cmp al, 'r'
    je ps_ok
    cmp al, 't'
    je ps_ok
    jmp err_here
ps_ok:
    inc rsi
    jmp ps_loop
ps_uni:
    inc rsi
    mov ecx, 4
ps_hex:
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '0'
    jb err_here
    cmp al, '9'
    jbe ps_hok
    or al, 0x20
    cmp al, 'a'
    jb err_here
    cmp al, 'f'
    ja err_here
ps_hok:
    inc rsi
    dec ecx
    jnz ps_hex
    jmp ps_loop

# ---- number, rsi at '-' or digit ----
parse_number:
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '-'
    jne pn_int
    inc rsi
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
pn_int:
    cmp al, '0'
    jb err_here
    cmp al, '9'
    ja err_here
    inc rsi
    cmp al, '0'
    je pn_fracpeek               # '0' takes no more int digits
pn_intdig:
    cmp rsi, r12
    jae pn_done
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '0'
    jb pn_fracchk
    cmp al, '9'
    ja pn_fracchk
    inc rsi
    jmp pn_intdig
pn_fracpeek:
    cmp rsi, r12
    jae pn_done
    movzx eax, byte ptr [rbp + rsi]
pn_fracchk:
    cmp al, '.'
    jne pn_expchk
    inc rsi
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '0'
    jb err_here
    cmp al, '9'
    ja err_here
    inc rsi
pn_fracdig:
    cmp rsi, r12
    jae pn_done
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '0'
    jb pn_expchk
    cmp al, '9'
    ja pn_expchk
    inc rsi
    jmp pn_fracdig
pn_expchk:
    cmp al, 'e'
    je pn_exp
    cmp al, 'E'
    jne pn_done
pn_exp:
    inc rsi
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '+'
    je pn_sign
    cmp al, '-'
    jne pn_expd1
pn_sign:
    inc rsi
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
pn_expd1:
    cmp al, '0'
    jb err_here
    cmp al, '9'
    ja err_here
    inc rsi
pn_expdig:
    cmp rsi, r12
    jae pn_done
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '0'
    jb pn_done
    cmp al, '9'
    ja pn_done
    inc rsi
    jmp pn_expdig
pn_done:
    ret

# ---- object, rsi at '{' ----
parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '}'
    jne po_member
    inc rsi
    ret
po_member:
    cmp al, '"'
    jne err_here
    call parse_string
    call skip_ws
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, ':'
    jne err_here
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, ','
    je po_next
    cmp al, '}'
    jne err_here
    inc rsi
    ret
po_next:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    jmp po_member

# ---- array, rsi at '[' ----
parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, ']'
    jne pa_elem
    inc rsi
    ret
pa_elem:
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae err_here
    movzx eax, byte ptr [rbp + rsi]
    cmp al, ','
    je pa_next
    cmp al, ']'
    jne err_here
    inc rsi
    ret
pa_next:
    inc rsi
    call skip_ws
    jmp pa_elem
