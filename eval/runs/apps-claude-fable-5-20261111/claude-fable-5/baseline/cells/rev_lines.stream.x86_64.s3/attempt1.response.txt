.text
.globl _start
_start:
    xor r12, r12                    # total bytes read
read_loop:
    mov edx, 65536
    sub edx, r12d
    jle read_done                   # buffer full
    xor eax, eax                    # read
    xor edi, edi                    # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done                   # 0 = EOF (treat error as EOF too)
    add r12, rax
    jmp read_loop
read_done:
    lea r8, [rip + inbuf]           # input base
    lea r9, [rip + outbuf]          # output write pointer
    xor rbx, rbx                    # i = current position
proc_loop:
    cmp rbx, r12
    jae proc_done
    mov rcx, rbx                    # scan for newline
find_nl:
    cmp rcx, r12
    jae got_end
    cmp byte ptr [r8 + rcx], 10
    je got_end
    inc rcx
    jmp find_nl
got_end:
    mov rdx, rcx                    # line = [rbx, rcx)
rev_loop:
    cmp rdx, rbx
    jbe rev_done
    dec rdx
    mov al, byte ptr [r8 + rdx]
    mov byte ptr [r9], al
    inc r9
    jmp rev_loop
rev_done:
    mov byte ptr [r9], 10
    inc r9
    lea rbx, [rcx + 1]              # skip the newline (or run past end)
    jmp proc_loop
proc_done:
    lea rsi, [rip + outbuf]
    mov r10, r9
    sub r10, rsi                    # output length
write_loop:
    test r10, r10
    jz do_exit
    mov eax, 1                      # write
    mov edi, 1                      # fd 1
    mov rdx, r10
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub r10, rax
    jmp write_loop
do_exit:
    mov eax, 231                    # exit_group
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:  .space 65536
outbuf: .space 65537
