.text
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0                 // total bytes read
read_loop:
    mov x0, #0                  // fd 0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    ble read_done
    add x19, x19, x0
    b read_loop
read_done:
    adrp x22, linebuf
    add x22, x22, :lo12:linebuf
    adrp x24, hexdig
    add x24, x24, :lo12:hexdig
    mov x21, #0                 // current offset
row_loop:
    cmp x21, x19
    bge final
    sub x25, x19, x21           // n = min(16, remaining)
    cmp x25, #16
    mov x0, #16
    csel x25, x0, x25, gt
    mov x23, x22                // line pointer
    bl emit_off
    mov w0, #32
    strb w0, [x23], #1
    strb w0, [x23], #1
    mov x26, #0                 // i
hex_loop:
    cmp x26, #16
    bge hex_done
    cmp x26, x25
    bge pad_slot
    add x1, x21, x26
    ldrb w0, [x20, x1]
    bl emit_hex
    mov w0, #32
    strb w0, [x23], #1
    b after_slot
pad_slot:
    mov w0, #32
    strb w0, [x23], #1
    strb w0, [x23], #1
    strb w0, [x23], #1
after_slot:
    cmp x26, #7
    bne 1f
    mov w0, #32                 // extra space after 8th slot
    strb w0, [x23], #1
1:
    add x26, x26, #1
    b hex_loop
hex_done:
    mov w0, #32
    strb w0, [x23], #1
    mov w0, #124                // '|'
    strb w0, [x23], #1
    mov x26, #0
chr_loop:
    cmp x26, x25
    bge chr_done
    add x1, x21, x26
    ldrb w0, [x20, x1]
    cmp w0, #0x20
    blo use_dot
    cmp w0, #0x7e
    bls emit_char
use_dot:
    mov w0, #46                 // '.'
emit_char:
    strb w0, [x23], #1
    add x26, x26, #1
    b chr_loop
chr_done:
    mov w0, #124                // '|'
    strb w0, [x23], #1
    mov w0, #10
    strb w0, [x23], #1
    bl write_line
    add x21, x21, x25
    b row_loop
final:
    mov x23, x22
    bl emit_off
    mov w0, #10
    strb w0, [x23], #1
    bl write_line
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

// emit 8 lowercase hex digits of w21 at x23
emit_off:
    mov x1, #28
2:
    lsr w0, w21, w1
    and w0, w0, #0xf
    ldrb w0, [x24, w0, uxtw]
    strb w0, [x23], #1
    subs x1, x1, #4
    bpl 2b
    ret

// emit 2 lowercase hex digits of byte in w0 at x23
emit_hex:
    lsr w1, w0, #4
    ldrb w1, [x24, w1, uxtw]
    strb w1, [x23], #1
    and w1, w0, #0xf
    ldrb w1, [x24, w1, uxtw]
    strb w1, [x23], #1
    ret

// write linebuf..x23 to stdout
write_line:
    mov x0, #1
    mov x1, x22
    sub x2, x23, x22
    mov x8, #64                 // write
    svc #0
    ret

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536
linebuf:
    .space 96
