	.text
	.globl _start
_start:
	# r12 = lines, r13 = words, r14 = bytes, r15 = in_word flag
	xor r12, r12
	xor r13, r13
	xor r14, r14
	xor r15, r15

read_loop:
	xor eax, eax            # read
	xor edi, edi            # fd 0
	lea rsi, [rip + buffer]
	mov edx, 65536
	syscall
	test rax, rax
	jle done_reading

	add r14, rax            # bytes += n
	lea rsi, [rip + buffer]
	mov rcx, rax            # rcx = count

scan:
	movzx eax, byte ptr [rsi]
	inc rsi
	cmp al, 0x0a
	jne 1f
	inc r12                 # newline
1:
	# whitespace test: 0x20 or 0x09..0x0d
	cmp al, 0x20
	je ws
	cmp al, 0x09
	jb nonws
	cmp al, 0x0d
	jbe ws
nonws:
	test r15, r15
	jnz next
	inc r13                 # word start
	mov r15, 1
	jmp next
ws:
	xor r15, r15
next:
	dec rcx
	jnz scan
	jmp read_loop

done_reading:
	# build output string backwards in outbuf
	lea rdi, [rip + outbuf + 63]
	mov byte ptr [rdi], 0x0a
	mov rax, r14
	call put_num
	dec rdi
	mov byte ptr [rdi], 0x20
	mov rax, r13
	call put_num
	dec rdi
	mov byte ptr [rdi], 0x20
	mov rax, r12
	call put_num

	# write(1, rdi, len)
	lea rdx, [rip + outbuf + 64]
	sub rdx, rdi
	mov rsi, rdi
	mov edi, 1
	mov eax, 1
	syscall

	mov eax, 60
	xor edi, edi
	syscall

# put_num: writes decimal of rax backwards, rdi points just past where to
# start; on return rdi points at the first digit. Clobbers rax, rdx, r8.
put_num:
	mov r8, 10
1:
	xor edx, edx
	div r8
	add dl, 0x30
	dec rdi
	mov byte ptr [rdi], dl
	test rax, rax
	jnz 1b
	ret

	.bss
	.balign 16
buffer:
	.space 65536
outbuf:
	.space 64
