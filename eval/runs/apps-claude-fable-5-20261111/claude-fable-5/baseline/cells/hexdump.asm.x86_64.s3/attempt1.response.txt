.text
.globl _start

_start:
	xor	ebx, ebx		# total bytes read
rdloop:
	mov	edx, 65536
	sub	edx, ebx
	jz	rddone
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	lea	rsi, [rip + inbuf]
	add	rsi, rbx
	syscall
	test	rax, rax
	jle	rddone
	add	ebx, eax
	jmp	rdloop
rddone:
	xor	r12d, r12d		# current offset

rowloop:
	cmp	r12d, ebx
	jae	finish
	mov	r13d, ebx
	sub	r13d, r12d
	cmp	r13d, 16
	jbe	1f
	mov	r13d, 16
1:
	lea	rdi, [rip + line]
	mov	eax, r12d
	call	hex8			# 8 hex digits of offset
	mov	word ptr [rdi], 0x2020	# two spaces
	add	rdi, 2

	lea	r8, [rip + hexdig]
	lea	r9, [rip + inbuf]
	add	r9, r12
	xor	ecx, ecx		# position i
hexloop:
	cmp	ecx, r13d
	jae	blank
	movzx	eax, byte ptr [r9 + rcx]
	mov	esi, eax
	shr	esi, 4
	mov	dl, byte ptr [r8 + rsi]
	mov	byte ptr [rdi], dl
	and	eax, 15
	mov	dl, byte ptr [r8 + rax]
	mov	byte ptr [rdi + 1], dl
	mov	byte ptr [rdi + 2], 0x20
	jmp	nextpos
blank:
	mov	word ptr [rdi], 0x2020
	mov	byte ptr [rdi + 2], 0x20
nextpos:
	add	rdi, 3
	cmp	ecx, 7
	jne	2f
	mov	byte ptr [rdi], 0x20	# extra space after 8th slot
	inc	rdi
2:
	inc	ecx
	cmp	ecx, 16
	jb	hexloop

	mov	byte ptr [rdi], 0x20
	mov	byte ptr [rdi + 1], 0x7c	# '|'
	add	rdi, 2

	xor	ecx, ecx
ascloop:
	cmp	ecx, r13d
	jae	ascdone
	mov	al, byte ptr [r9 + rcx]
	cmp	al, 0x20
	jb	dot
	cmp	al, 0x7e
	jbe	keep
dot:
	mov	al, 0x2e		# '.'
keep:
	mov	byte ptr [rdi], al
	inc	rdi
	inc	ecx
	jmp	ascloop
ascdone:
	mov	byte ptr [rdi], 0x7c	# '|'
	mov	byte ptr [rdi + 1], 10
	add	rdi, 2

	lea	rsi, [rip + line]
	mov	rdx, rdi
	sub	rdx, rsi
	call	writeall

	add	r12d, r13d
	jmp	rowloop

finish:
	lea	rdi, [rip + line]
	mov	eax, ebx
	call	hex8
	mov	byte ptr [rdi], 10
	lea	rsi, [rip + line]
	mov	edx, 9
	call	writeall
	mov	eax, 231		# exit_group
	xor	edi, edi
	syscall

# hex8: write eax as 8 lowercase hex digits at [rdi], advance rdi by 8.
# clobbers eax, ecx, edx, r8
hex8:
	lea	r8, [rip + hexdig]
	mov	ecx, 8
1:
	rol	eax, 4
	mov	edx, eax
	and	edx, 15
	mov	dl, byte ptr [r8 + rdx]
	mov	byte ptr [rdi], dl
	inc	rdi
	dec	ecx
	jnz	1b
	ret

# writeall: write rdx bytes at rsi to stdout, retrying on short writes.
# clobbers rax, rcx, rdx, rsi, rdi, r11
writeall:
1:
	test	rdx, rdx
	jz	2f
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	syscall
	test	rax, rax
	jle	2f
	add	rsi, rax
	sub	rdx, rax
	jmp	1b
2:
	ret

.section .rodata
hexdig:
	.ascii	"0123456789abcdef"

.bss
	.balign	16
inbuf:
	.zero	65536
line:
	.zero	128
