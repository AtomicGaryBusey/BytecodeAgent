.text
.globl _start

_start:
        # ---- read all of stdin into buf ----
        xor     ebp, ebp                # total bytes read
.Lread:
        mov     edx, 65536
        sub     edx, ebp
        jz      .Lrdone
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, rbp
        syscall
        test    rax, rax
        jle     .Lrdone
        add     rbp, rax
        jmp     .Lread
.Lrdone:
        lea     rsi, [rip + buf]        # rsi = buf base (kept)

        # ---- split at first '!' ----
        xor     ecx, ecx
.Lfind:
        cmp     rcx, rbp
        jae     .Lnobang
        cmp     byte ptr [rsi + rcx], 0x21
        je      .Lbang
        inc     rcx
        jmp     .Lfind
.Lbang:
        mov     r13, rcx                # program length
        lea     r14, [rsi + rcx + 1]    # input start
        jmp     .Lsetend
.Lnobang:
        mov     r13, rbp
        lea     r14, [rsi + rbp]
.Lsetend:
        lea     r15, [rsi + rbp]        # input end

        # ---- validate brackets, build match table ----
        xor     ecx, ecx                # i
        xor     edx, edx                # stack depth
        lea     r8, [rip + bstack]
        lea     r9, [rip + matchtab]
.Lv:
        cmp     rcx, r13
        jae     .Lvend
        movzx   eax, byte ptr [rsi + rcx]
        cmp     al, 91
        je      .Lvopen
        cmp     al, 93
        je      .Lvclose
.Lvnext:
        inc     rcx
        jmp     .Lv
.Lvopen:
        mov     dword ptr [r8 + rdx*4], ecx
        inc     rdx
        jmp     .Lvnext
.Lvclose:
        test    rdx, rdx
        jz      .Lerr
        dec     rdx
        mov     eax, dword ptr [r8 + rdx*4]
        mov     dword ptr [r9 + rax*4], ecx
        mov     dword ptr [r9 + rcx*4], eax
        jmp     .Lvnext
.Lvend:
        test    rdx, rdx
        jnz     .Lerr

        # ---- run ----
        lea     rbx, [rip + tape]       # data pointer
        xor     r12d, r12d              # pc
        lea     rbp, [rip + outbuf]     # output write pointer
        lea     r10, [rip + jtab]
.Lrun:
        cmp     r12, r13
        jae     .Ldone
        movzx   eax, byte ptr [rsi + r12]
        cmp     al, 93
        ja      .Lcnx
        jmp     qword ptr [r10 + rax*8]

.Lcpl:  inc     byte ptr [rbx]
        jmp     .Lcnx
.Lcmi:  dec     byte ptr [rbx]
        jmp     .Lcnx
.Lcgt:  inc     rbx
        jmp     .Lcnx
.Lclt:  dec     rbx
        jmp     .Lcnx
.Lcdt:  mov     al, byte ptr [rbx]
        mov     byte ptr [rbp], al
        inc     rbp
        jmp     .Lcnx
.Lccm:
        xor     eax, eax
        cmp     r14, r15
        jae     1f
        mov     al, byte ptr [r14]
        inc     r14
1:      mov     byte ptr [rbx], al
        jmp     .Lcnx
.Lclb:
        cmp     byte ptr [rbx], 0
        jne     .Lcnx
        mov     r12d, dword ptr [r9 + r12*4]
        jmp     .Lcnx
.Lcrb:
        cmp     byte ptr [rbx], 0
        je      .Lcnx
        mov     r12d, dword ptr [r9 + r12*4]
.Lcnx:
        inc     r12
        jmp     .Lrun

        # ---- flush output and exit 0 ----
.Ldone:
        lea     rsi, [rip + outbuf]
        mov     rdx, rbp
        sub     rdx, rsi
.Lwr:
        test    rdx, rdx
        jle     .Lexit0
        mov     eax, 1                  # sys_write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     .Lexit0
        add     rsi, rax
        sub     rdx, rax
        jmp     .Lwr
.Lexit0:
        mov     eax, 60
        xor     edi, edi
        syscall

.Lerr:
        mov     eax, 60
        mov     edi, 1
        syscall

        .section .rodata
        .balign 8
jtab:
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 0-7
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 8-15
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 16-23
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 24-31
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 32-39
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcpl,.Lccm,.Lcmi,.Lcdt,.Lcnx   # 40-47
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 48-55
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lclt,.Lcnx,.Lcgt,.Lcnx   # 56-63
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 64-71
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 72-79
        .quad .Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx,.Lcnx   # 80-87
        .quad .Lcnx,.Lcnx,.Lcnx,.Lclb,.Lcnx,.Lcrb               # 88-93

        .bss
        .balign 8
buf:
        .zero   65536
tape:
        .zero   30000
outbuf:
        .zero   65536
        .balign 4
matchtab:
        .zero   262144
bstack:
        .zero   262144
