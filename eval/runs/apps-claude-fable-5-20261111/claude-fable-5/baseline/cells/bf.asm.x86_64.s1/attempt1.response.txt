.bss
.balign 8
buf:    .space 65536
tape:   .space 30000
.balign 4
jmptab: .space 262144

.text
.globl _start
_start:
        xor     ebx, ebx                # total bytes read
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jz      read_done
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, rbx
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     r15, [rip + buf]
        lea     r9,  [rip + jmptab]
        # find first '!'
        xor     ecx, ecx
find_bang:
        cmp     rcx, rbx
        jae     no_bang
        cmp     byte ptr [r15 + rcx], 0x21
        je      found_bang
        inc     rcx
        jmp     find_bang
found_bang:
        mov     r14, rcx                # program length
        lea     r13, [r15 + rcx + 1]    # input start
        jmp     set_end
no_bang:
        mov     r14, rbx
        lea     r13, [r15 + rbx]
set_end:
        lea     r8, [r15 + rbx]         # input end

        # bracket check + jump table
        xor     ecx, ecx                # ip
        xor     edx, edx                # depth
bscan:
        cmp     rcx, r14
        jae     bscan_end
        movzx   eax, byte ptr [r15 + rcx]
        cmp     al, 0x5b                # '['
        je      bopen
        cmp     al, 0x5d                # ']'
        je      bclose
bnext:
        inc     rcx
        jmp     bscan
bopen:
        push    rcx
        inc     edx
        jmp     bnext
bclose:
        test    edx, edx
        jz      fail
        dec     edx
        pop     rax                     # matching '[' position
        lea     edi, [ecx + 1]
        mov     dword ptr [r9 + rax*4], edi
        lea     edi, [eax + 1]
        mov     dword ptr [r9 + rcx*4], edi
        jmp     bnext
bscan_end:
        test    edx, edx
        jnz     fail

        # run
        lea     r12, [rip + tape]       # data pointer
        xor     ebx, ebx                # ip
run:
        cmp     rbx, r14
        jae     done
        movzx   eax, byte ptr [r15 + rbx]
        inc     rbx
        cmp     al, 0x2b                # '+'
        je      c_plus
        cmp     al, 0x2d                # '-'
        je      c_minus
        cmp     al, 0x3e                # '>'
        je      c_right
        cmp     al, 0x3c                # '<'
        je      c_left
        cmp     al, 0x5b                # '['
        je      c_open
        cmp     al, 0x5d                # ']'
        je      c_close
        cmp     al, 0x2e                # '.'
        je      c_dot
        cmp     al, 0x2c                # ','
        je      c_comma
        jmp     run
c_plus:
        inc     byte ptr [r12]
        jmp     run
c_minus:
        dec     byte ptr [r12]
        jmp     run
c_right:
        inc     r12
        jmp     run
c_left:
        dec     r12
        jmp     run
c_open:
        cmp     byte ptr [r12], 0
        jne     run
        mov     ebx, dword ptr [r9 + rbx*4 - 4]
        jmp     run
c_close:
        cmp     byte ptr [r12], 0
        je      run
        mov     ebx, dword ptr [r9 + rbx*4 - 4]
        jmp     run
c_dot:
        mov     eax, 1                  # write
        mov     edi, 1
        mov     rsi, r12
        mov     edx, 1
        syscall
        jmp     run
c_comma:
        xor     eax, eax
        cmp     r13, r8
        jae     1f
        mov     al, byte ptr [r13]
        inc     r13
1:
        mov     byte ptr [r12], al
        jmp     run
done:
        mov     eax, 60                 # exit
        xor     edi, edi
        syscall
fail:
        mov     eax, 60
        mov     edi, 1
        syscall
