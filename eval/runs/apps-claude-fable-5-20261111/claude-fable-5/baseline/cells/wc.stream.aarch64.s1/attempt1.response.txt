// wc for AArch64 Linux, freestanding
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSZ, 65536

.bss
.balign 8
buf:
	.space BUFSZ
outbuf:
	.space 80

.text
.globl _start
_start:
	mov	x19, #0		// lines
	mov	x20, #0		// words
	mov	x21, #0		// bytes
	mov	x22, #0		// in-word flag
	adrp	x23, buf
	add	x23, x23, :lo12:buf

read_loop:
	mov	x0, #0
	mov	x1, x23
	mov	x2, #BUFSZ
	mov	x8, #SYS_read
	svc	#0
	cmp	x0, #0
	ble	done_reading

	add	x21, x21, x0	// bytes += n
	mov	x3, x23		// ptr
	add	x4, x23, x0	// end
scan:
	ldrb	w5, [x3], #1
	cmp	w5, #0x0a
	bne	1f
	add	x19, x19, #1	// newline
1:	// whitespace test: 0x20 or 0x09..0x0d
	cmp	w5, #0x20
	beq	is_ws
	sub	w6, w5, #0x09
	cmp	w6, #4		// 0x09..0x0d
	bls	is_ws
	// non-whitespace
	cbnz	x22, 2f
	mov	x22, #1
	add	x20, x20, #1
	b	2f
is_ws:
	mov	x22, #0
2:
	cmp	x3, x4
	blo	scan
	b	read_loop

done_reading:
	// build output string in outbuf
	adrp	x9, outbuf
	add	x9, x9, :lo12:outbuf
	mov	x10, x9		// write pointer

	mov	x0, x19
	bl	put_num
	mov	w0, #0x20
	strb	w0, [x10], #1
	mov	x0, x20
	bl	put_num
	mov	w0, #0x20
	strb	w0, [x10], #1
	mov	x0, x21
	bl	put_num
	mov	w0, #0x0a
	strb	w0, [x10], #1

	mov	x0, #1
	mov	x1, x9
	sub	x2, x10, x9
	mov	x8, #SYS_write
	svc	#0

	mov	x0, #0
	mov	x8, #SYS_exit
	svc	#0

// put_num: write decimal of x0 at x10, advance x10.
// clobbers x0,x5,x6,x7,x11,x12
put_num:
	add	x11, sp, #-32	// temp area on stack (below sp, safe: no calls/signals matter? use pre-decrement instead)
	sub	sp, sp, #32
	mov	x11, sp
	mov	x12, #0		// digit count
	mov	x7, #10
1:
	udiv	x5, x0, x7
	msub	x6, x5, x7, x0	// remainder
	add	w6, w6, #'0'
	strb	w6, [x11, x12]
	add	x12, x12, #1
	mov	x0, x5
	cbnz	x0, 1b
2:
	sub	x12, x12, #1
	ldrb	w6, [x11, x12]
	strb	w6, [x10], #1
	cbnz	x12, 2b
	add	sp, sp, #32
	ret
