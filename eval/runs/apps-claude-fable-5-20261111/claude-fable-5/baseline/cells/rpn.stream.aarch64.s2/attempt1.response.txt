.text
.globl _start

_start:
        // read all input
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, #0                 // bytes read so far
rd_loop:
        mov     x2, #65536
        sub     x2, x2, x20
        cbz     x2, rd_done
        add     x1, x19, x20
        mov     x0, #0
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        ble     rd_done
        add     x20, x20, x0
        b       rd_loop
rd_done:
        add     x20, x19, x20           // x20 = end of input, x19 = cursor
        adrp    x21, stack
        add     x21, x21, :lo12:stack

lineloop:
        cmp     x19, x20
        bhs     done
        // find end of line in x23
        mov     x23, x19
findnl:
        cmp     x23, x20
        bhs     fnl_done
        ldrb    w0, [x23]
        cmp     w0, #10
        beq     fnl_done
        add     x23, x23, #1
        b       findnl
fnl_done:
        mov     x22, #0                 // stack count

tokloop:
skipsep:
        cmp     x19, x23
        bhs     lineend
        ldrb    w0, [x19]
        cmp     w0, #0x20
        beq     sep_adv
        cmp     w0, #0x09
        bne     tokstart
sep_adv:
        add     x19, x19, #1
        b       skipsep

tokstart:
        // find token end in x24
        mov     x24, x19
tscan:
        cmp     x24, x23
        bhs     ts_done
        ldrb    w0, [x24]
        cmp     w0, #0x20
        beq     ts_done
        cmp     w0, #0x09
        beq     ts_done
        add     x24, x24, #1
        b       tscan
ts_done:
        sub     x9, x24, x19            // token length (>=1)
        ldrb    w0, [x19]
        cmp     x9, #1
        bne     try_lit
        cmp     w0, #'+'
        beq     op_add
        cmp     w0, #'-'
        beq     op_sub
        cmp     w0, #'*'
        beq     op_mul
        cmp     w0, #'/'
        beq     op_div
        cmp     w0, #'%'
        beq     op_mod
try_lit:
        mov     x10, x19
        mov     x11, #0                 // negative flag
        cmp     w0, #'-'
        bne     lit_digits
        mov     x11, #1
        add     x10, x10, #1
lit_digits:
        sub     x12, x24, x10           // digit count
        cmp     x12, #1
        blt     err
        cmp     x12, #18
        bgt     err
        mov     x13, #0
        mov     x14, #10
digloop:
        cmp     x10, x24
        bhs     dig_done
        ldrb    w0, [x10], #1
        sub     w0, w0, #'0'
        cmp     w0, #9
        bhi     err
        madd    x13, x13, x14, x0
        b       digloop
dig_done:
        cbz     x11, push_adv
        neg     x13, x13
        b       push_adv

op_add:
        cmp     x22, #2
        blt     err
        sub     x22, x22, #2
        ldr     x13, [x21, x22, lsl #3]
        add     x9, x22, #1
        ldr     x14, [x21, x9, lsl #3]
        add     x13, x13, x14
        b       push_adv
op_sub:
        cmp     x22, #2
        blt     err
        sub     x22, x22, #2
        ldr     x13, [x21, x22, lsl #3]
        add     x9, x22, #1
        ldr     x14, [x21, x9, lsl #3]
        sub     x13, x13, x14
        b       push_adv
op_mul:
        cmp     x22, #2
        blt     err
        sub     x22, x22, #2
        ldr     x13, [x21, x22, lsl #3]
        add     x9, x22, #1
        ldr     x14, [x21, x9, lsl #3]
        mul     x13, x13, x14
        b       push_adv
op_div:
        cmp     x22, #2
        blt     err
        sub     x22, x22, #2
        ldr     x13, [x21, x22, lsl #3]
        add     x9, x22, #1
        ldr     x14, [x21, x9, lsl #3]
        cbz     x14, err
        cmn     x14, #1
        bne     div_ok
        movz    x9, #0x8000, lsl #48
        cmp     x13, x9
        beq     err
div_ok:
        sdiv    x13, x13, x14
        b       push_adv
op_mod:
        cmp     x22, #2
        blt     err
        sub     x22, x22, #2
        ldr     x13, [x21, x22, lsl #3]
        add     x9, x22, #1
        ldr     x14, [x21, x9, lsl #3]
        cbz     x14, err
        cmn     x14, #1
        bne     mod_ok
        movz    x9, #0x8000, lsl #48
        cmp     x13, x9
        beq     err
mod_ok:
        sdiv    x9, x13, x14
        msub    x13, x9, x14, x13
        b       push_adv

push_adv:
        str     x13, [x21, x22, lsl #3]
        add     x22, x22, #1
        mov     x19, x24
        b       tokloop

lineend:
        cmp     x22, #1
        bne     err
        ldr     x0, [x21]
        bl      printnum
        b       nextline

err:
        adrp    x1, errmsg
        add     x1, x1, :lo12:errmsg
        mov     x2, #6
        mov     x0, #1
        mov     x8, #64
        svc     #0

nextline:
        cmp     x23, x20
        add     x9, x23, #1
        csel    x19, x9, x23, lo
        b       lineloop

done:
        mov     x0, #0
        mov     x8, #93
        svc     #0

// printnum: x0 = signed 64-bit value, prints decimal + newline to stdout
printnum:
        adrp    x1, numbuf
        add     x1, x1, :lo12:numbuf
        add     x1, x1, #31
        mov     w2, #10
        strb    w2, [x1]                // trailing newline
        mov     x3, #0                  // sign flag
        tbz     x0, #63, pn_conv
        mov     x3, #1
        neg     x0, x0
pn_conv:
        mov     x4, #10
pn_loop:
        udiv    x5, x0, x4
        msub    x6, x5, x4, x0
        add     w6, w6, #'0'
        strb    w6, [x1, #-1]!
        mov     x0, x5
        cbnz    x0, pn_loop
        cbz     x3, pn_write
        mov     w6, #'-'
        strb    w6, [x1, #-1]!
pn_write:
        adrp    x2, numbuf
        add     x2, x2, :lo12:numbuf
        add     x2, x2, #32
        sub     x2, x2, x1
        mov     x0, #1
        mov     x8, #64
        svc     #0
        ret

.section .rodata
errmsg:
        .ascii  "error\n"

.bss
.balign 8
stack:
        .space  8192
numbuf:
        .space  32
inbuf:
        .space  65537
