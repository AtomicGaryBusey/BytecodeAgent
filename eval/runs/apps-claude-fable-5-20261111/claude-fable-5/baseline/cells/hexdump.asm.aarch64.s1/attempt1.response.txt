.text
.globl _start

// nibble in w0 (0-15) -> ascii hex char in w0
nibchar:
	cmp	w0, #10
	add	w0, w0, #48		// '0'
	blt	1f
	add	w0, w0, #39		// 'a' - '0' - 10
1:
	ret

_start:
	// read all of stdin into inbuf
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0			// total bytes read
read_loop:
	mov	x0, #0			// fd 0
	add	x1, x19, x20
	mov	x2, #65536
	sub	x2, x2, x20
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	ble	read_done
	add	x20, x20, x0
	b	read_loop
read_done:
	adrp	x21, line
	add	x21, x21, :lo12:line
	mov	x23, #0			// current offset

row_loop:
	cmp	x23, x20
	bge	final
	sub	x24, x20, x23		// remaining
	mov	x0, #16
	cmp	x24, #16
	csel	x24, x24, x0, lt	// n = min(16, remaining)
	mov	x22, x21		// p = line

	// offset as 8 hex digits
	mov	x25, #32
off_hex:
	sub	x25, x25, #4
	lsr	x0, x23, x25
	and	x0, x0, #15
	bl	nibchar
	strb	w0, [x22], #1
	cbnz	x25, off_hex

	// two spaces
	mov	w0, #32
	strb	w0, [x22], #1
	strb	w0, [x22], #1

	// 16 byte slots
	mov	x26, #0
slot_loop:
	cmp	x26, x24
	bge	slot_pad
	add	x0, x19, x23
	ldrb	w27, [x0, x26]
	lsr	w0, w27, #4
	bl	nibchar
	strb	w0, [x22], #1
	and	w0, w27, #15
	bl	nibchar
	strb	w0, [x22], #1
	mov	w0, #32
	strb	w0, [x22], #1
	b	slot_after
slot_pad:
	mov	w0, #32
	strb	w0, [x22], #1
	strb	w0, [x22], #1
	strb	w0, [x22], #1
slot_after:
	cmp	x26, #7
	bne	slot_next
	mov	w0, #32			// extra space after eighth slot
	strb	w0, [x22], #1
slot_next:
	add	x26, x26, #1
	cmp	x26, #16
	blt	slot_loop

	// space and opening bar
	mov	w0, #32
	strb	w0, [x22], #1
	mov	w0, #124		// '|'
	strb	w0, [x22], #1

	// ascii column
	mov	x26, #0
char_loop:
	cmp	x26, x24
	bge	char_done
	add	x0, x19, x23
	ldrb	w0, [x0, x26]
	cmp	w0, #0x20
	blt	char_dot
	cmp	w0, #0x7e
	ble	char_store
char_dot:
	mov	w0, #46			// '.'
char_store:
	strb	w0, [x22], #1
	add	x26, x26, #1
	b	char_loop
char_done:
	mov	w0, #124		// '|'
	strb	w0, [x22], #1
	mov	w0, #10			// '\n'
	strb	w0, [x22], #1

	// write the line
	mov	x0, #1
	mov	x1, x21
	sub	x2, x22, x21
	mov	x8, #64			// write
	svc	#0

	add	x23, x23, x24
	b	row_loop

final:
	// total length as 8 hex digits + newline
	mov	x22, x21
	mov	x25, #32
fin_hex:
	sub	x25, x25, #4
	lsr	x0, x20, x25
	and	x0, x0, #15
	bl	nibchar
	strb	w0, [x22], #1
	cbnz	x25, fin_hex
	mov	w0, #10
	strb	w0, [x22], #1
	mov	x0, #1
	mov	x1, x21
	mov	x2, #9
	mov	x8, #64			// write
	svc	#0

	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

.bss
.balign 8
inbuf:
	.space	65536
line:
	.space	96
