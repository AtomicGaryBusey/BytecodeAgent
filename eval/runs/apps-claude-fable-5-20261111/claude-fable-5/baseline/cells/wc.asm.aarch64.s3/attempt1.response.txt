.text
.globl _start
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in_word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x8, #63              // read
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    svc #0
    cmp x0, #0
    b.le done

    mov x5, x23              // scan pointer
    mov x6, x0               // bytes remaining
    add x21, x21, x0         // total bytes

scan:
    ldrb w7, [x5], #1
    cmp w7, #0x0A
    b.ne 1f
    add x19, x19, #1         // newline -> lines++
1:
    cmp w7, #0x20
    b.eq is_ws
    sub w9, w7, #0x09
    cmp w9, #4               // 0x09..0x0D
    b.ls is_ws
    // word byte
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
next:
    subs x6, x6, #1
    b.ne scan
    b read_loop

done:
    // build output string from end of outbuf
    adrp x4, outbuf
    add x4, x4, :lo12:outbuf
    add x4, x4, #64          // one past end
    mov w7, #0x0A
    strb w7, [x4, #-1]!      // newline
    mov x0, x21
    bl itoa
    mov w7, #0x20
    strb w7, [x4, #-1]!      // space
    mov x0, x20
    bl itoa
    mov w7, #0x20
    strb w7, [x4, #-1]!      // space
    mov x0, x19
    bl itoa

    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    add x2, x2, #64
    sub x2, x2, x4           // length
    mov x1, x4
    mov x0, #1
    mov x8, #64              // write
    svc #0

    mov x0, #0
    mov x8, #93              // exit
    svc #0

// itoa: value in x0, stores digits backwards at [x4, #-1]! (x4 updated)
// clobbers x1, x2, x3, x7
itoa:
    mov x1, #10
itoa_loop:
    udiv x2, x0, x1
    msub x3, x2, x1, x0
    add w3, w3, #0x30
    strb w3, [x4, #-1]!
    mov x0, x2
    cbnz x0, itoa_loop
    ret

.bss
.balign 8
buf:
    .space 65536
outbuf:
    .space 64
