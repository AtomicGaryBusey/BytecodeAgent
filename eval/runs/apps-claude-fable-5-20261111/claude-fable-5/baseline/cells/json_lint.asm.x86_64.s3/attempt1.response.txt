.text
.globl _start

_start:
    lea rbx, [rip + buf]
    xor r12, r12
read_loop:
    mov edx, 65540
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor esi, esi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne fail
    lea rsi, [rip + str_ok]
    mov edx, 3
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 231
    syscall

# ---- skip whitespace ----
skip_ws:
    cmp rsi, r12
    jae 2f
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0a
    je 1f
    cmp al, 0x0d
    jne 2f
1:  inc rsi
    jmp skip_ws
2:  ret

# ---- value dispatch ----
parse_value:
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x7b            # {
    je parse_object
    cmp al, 0x5b            # [
    je parse_array
    cmp al, 0x22            # "
    je parse_string
    cmp al, 0x2d            # -
    je parse_number
    cmp al, 0x30
    jb 1f
    cmp al, 0x39
    jbe parse_number
1:  cmp al, 0x74            # t
    je lit_true
    cmp al, 0x66            # f
    je lit_false
    cmp al, 0x6e            # n
    je lit_null
    jmp fail

lit_true:
    lea rdi, [rip + str_true]
    mov ecx, 4
    jmp check_lit
lit_false:
    lea rdi, [rip + str_false]
    mov ecx, 5
    jmp check_lit
lit_null:
    lea rdi, [rip + str_null]
    mov ecx, 4
check_lit:
1:  cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, byte ptr [rdi]
    jne fail
    inc rsi
    inc rdi
    dec ecx
    jnz 1b
    ret

# ---- string (entered with byte at rsi == '"') ----
parse_string:
    inc rsi
ps_loop:
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x22            # closing "
    je ps_done
    cmp al, 0x20
    jb fail
    cmp al, 0x5c            # backslash
    jne ps_plain
    inc rsi
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x75            # u
    je ps_u
    cmp al, 0x22
    je ps_plain
    cmp al, 0x5c
    je ps_plain
    cmp al, 0x2f            # /
    je ps_plain
    cmp al, 0x62            # b
    je ps_plain
    cmp al, 0x66            # f
    je ps_plain
    cmp al, 0x6e            # n
    je ps_plain
    cmp al, 0x72            # r
    je ps_plain
    cmp al, 0x74            # t
    je ps_plain
    jmp fail
ps_u:
    inc rsi
    mov ecx, 4
1:  cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe 2f
    cmp al, 0x41
    jb fail
    cmp al, 0x46
    jbe 2f
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
2:  inc rsi
    dec ecx
    jnz 1b
    jmp ps_loop
ps_plain:
    inc rsi
    jmp ps_loop
ps_done:
    inc rsi
    ret

# ---- number (entered at '-' or digit) ----
parse_number:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2d
    jne pn_int
    inc rsi
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
pn_int:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
    cmp al, 0x30
    je pn_frac
pn_intloop:
    cmp rsi, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    ja pn_frac
    inc rsi
    jmp pn_intloop
pn_frac:
    cmp rsi, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2e            # .
    jne pn_exp
    inc rsi
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
1:  cmp rsi, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    ja pn_exp
    inc rsi
    jmp 1b
pn_exp:
    cmp rsi, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x65            # e
    je 1f
    cmp al, 0x45            # E
    jne pn_ret
1:  inc rsi
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2b            # +
    je 2f
    cmp al, 0x2d            # -
    jne 3f
2:  inc rsi
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
3:  cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
4:  cmp rsi, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb pn_ret
    cmp al, 0x39
    ja pn_ret
    inc rsi
    jmp 4b
pn_ret:
    ret

# ---- object (entered at '{') ----
parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x7d            # }
    jne po_member
    inc rsi
    ret
po_member:
    cmp al, 0x22            # "
    jne fail
    call parse_string
    call skip_ws
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x3a            # :
    jne fail
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x7d            # }
    jne 1f
    inc rsi
    ret
1:  cmp al, 0x2c            # ,
    jne fail
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    jmp po_member

# ---- array (entered at '[') ----
parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x5d            # ]
    jne pa_val
    inc rsi
    ret
pa_val:
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae fail_eof
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x5d            # ]
    jne 1f
    inc rsi
    ret
1:  cmp al, 0x2c            # ,
    jne fail
    inc rsi
    call skip_ws
    jmp pa_val

# ---- error reporting: error position in rsi ----
fail_eof:
    mov rsi, r12
fail:
    mov rax, rsi
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 0x0a
    mov r8, 10
1:  xor edx, edx
    div r8
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    lea rcx, [rip + str_err]
    mov r9, 9
2:  dec rdi
    movzx eax, byte ptr [rcx + r9 - 1]
    mov byte ptr [rdi], al
    dec r9
    jnz 2b
    lea rdx, [rip + numbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 231
    syscall

.section .rodata
str_ok:    .ascii "ok\n"
str_err:   .ascii "error at "
str_true:  .ascii "true"
str_false: .ascii "false"
str_null:  .ascii "null"

.bss
.balign 64
buf:    .space 65540
numbuf: .space 32
