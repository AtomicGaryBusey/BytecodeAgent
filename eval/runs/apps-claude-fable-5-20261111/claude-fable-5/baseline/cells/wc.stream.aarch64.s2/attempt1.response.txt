// wc for AArch64 Linux, freestanding

	.bss
	.balign 8
buf:
	.space 65536
outbuf:
	.space 64

	.text
	.globl _start
_start:
	mov	x19, #0			// lines
	mov	x20, #0			// words
	mov	x21, #0			// bytes
	mov	x22, #0			// in-word flag
	adrp	x23, buf
	add	x23, x23, :lo12:buf

read_loop:
	mov	x0, #0			// fd 0
	mov	x1, x23
	mov	x2, #65536
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	ble	done
	add	x21, x21, x0
	mov	x1, x23			// ptr
	add	x2, x23, x0		// end
byte_loop:
	ldrb	w3, [x1], #1
	cmp	w3, #0x0a
	bne	1f
	add	x19, x19, #1
1:
	cmp	w3, #0x20
	beq	is_ws
	sub	w4, w3, #9
	cmp	w4, #4
	bls	is_ws
	// non-whitespace byte
	cbnz	x22, next
	mov	x22, #1
	add	x20, x20, #1
	b	next
is_ws:
	mov	x22, #0
next:
	cmp	x1, x2
	blo	byte_loop
	b	read_loop

done:
	adrp	x24, outbuf
	add	x24, x24, :lo12:outbuf
	mov	x25, x24		// output write pointer
	mov	x0, x19
	bl	putnum
	mov	w0, #0x20
	strb	w0, [x25], #1
	mov	x0, x20
	bl	putnum
	mov	w0, #0x20
	strb	w0, [x25], #1
	mov	x0, x21
	bl	putnum
	mov	w0, #0x0a
	strb	w0, [x25], #1

	mov	x0, #1			// stdout
	mov	x1, x24
	sub	x2, x25, x24
	mov	x8, #64			// write
	svc	#0

	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

// putnum: print unsigned x0 in decimal to [x25], advancing x25.
// Clobbers x0, x3-x6.
putnum:
	sub	sp, sp, #32
	mov	x3, sp
	mov	x4, #10
1:
	udiv	x5, x0, x4
	msub	x6, x5, x4, x0
	add	w6, w6, #'0'
	strb	w6, [x3], #1
	mov	x0, x5
	cbnz	x0, 1b
2:
	ldrb	w6, [x3, #-1]!
	strb	w6, [x25], #1
	cmp	x3, sp
