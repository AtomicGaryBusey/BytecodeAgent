.bss
.balign 8
buf:    .space 65536
outbuf: .space 64

.text
.globl _start
_start:
    mov x19, #0            // lines
    mov x20, #0            // words
    mov x21, #0            // bytes
    mov x22, #0            // in_word flag

read_loop:
    mov x0, #0             // fd 0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    mov x8, #63            // read
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0       // bytes += n
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x3, x0             // remaining count

proc:
    ldrb w4, [x1], #1
    cmp w4, #0x0a
    b.ne 1f
    add x19, x19, #1       // lines++
1:
    cmp w4, #0x20          // space?
    b.eq ws
    sub w5, w4, #9
    cmp w5, #4             // 0x09..0x0d?
    b.ls ws
    // word byte
    cbnz x22, next
    mov x22, #1
    add x20, x20, #1       // words++
    b next
ws:
    mov x22, #0
next:
    subs x3, x3, #1
    b.ne proc
    b read_loop

done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x1, x23
    mov x0, x19
    bl itoa
    mov w2, #0x20
    strb w2, [x1], #1
    mov x0, x20
    bl itoa
    mov w2, #0x20
    strb w2, [x1], #1
    mov x0, x21
    bl itoa
    mov w2, #0x0a
    strb w2, [x1], #1
    sub x2, x1, x23        // length
    mov x0, #1             // stdout
    mov x1, x23
    mov x8, #64            // write
    svc #0
    mov x0, #0
    mov x8, #93            // exit
    svc #0

// itoa: x0 = value, x1 = dest pointer; advances x1 past digits.
// clobbers x0, x2-x6
itoa:
    sub sp, sp, #32
    add x2, sp, #32        // end of temp digit area
    mov x4, #10
1:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #0x30
    strb w5, [x2, #-1]!
    mov x0, x3
    cbnz x0, 1b
    add x6, sp, #32        // end marker
2:
    ldrb w5, [x2], #1
    strb w5, [x1], #1
    cmp x2, x6
    b.lo 2b
    add sp, sp, #32
    ret
