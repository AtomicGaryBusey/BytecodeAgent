.text
.globl _start

_start:
	mov	x19, #0			// lines
	mov	x20, #0			// words
	mov	x21, #0			// bytes
	mov	x22, #0			// in-word flag
	adrp	x23, buf
	add	x23, x23, :lo12:buf

read_loop:
	mov	x0, #0			// fd 0
	mov	x1, x23
	mov	x2, #65536
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	b.le	finished
	add	x21, x21, x0		// bytes += n
	mov	x1, x23			// cursor
	add	x2, x23, x0		// end

scan:
	ldrb	w3, [x1], #1
	cmp	w3, #0x0a
	b.ne	1f
	add	x19, x19, #1		// lines++
1:
	cmp	w3, #0x20		// space?
	b.eq	is_ws
	sub	w4, w3, #0x09		// 0x09..0x0d ?
	cmp	w4, #4
	b.ls	is_ws
	// non-whitespace byte
	cbnz	x22, next
	add	x20, x20, #1		// words++
	mov	x22, #1
	b	next
is_ws:
	mov	x22, #0
next:
	cmp	x1, x2
	b.lo	scan
	b	read_loop

finished:
	adrp	x24, obuf
	add	x24, x24, :lo12:obuf
	mov	x1, x24

	mov	x0, x19
	bl	itoa
	mov	w3, #0x20
	strb	w3, [x1], #1

	mov	x0, x20
	bl	itoa
	mov	w3, #0x20
	strb	w3, [x1], #1

	mov	x0, x21
	bl	itoa
	mov	w3, #0x0a
	strb	w3, [x1], #1

	sub	x2, x1, x24		// length
	mov	x0, #1			// stdout
	mov	x1, x24
	mov	x8, #64			// write
	svc	#0

	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

// itoa: x0 = value, x1 = dest pointer (advanced on return)
// clobbers x0, x2-x6
itoa:
	adrp	x5, num
	add	x5, x5, :lo12:num
	add	x5, x5, #32		// end of scratch, fill backwards
	mov	x6, x5
	mov	x2, #10
1:
	udiv	x3, x0, x2
	msub	x4, x3, x2, x0		// remainder
	add	w4, w4, #0x30
	strb	w4, [x5, #-1]!
	mov	x0, x3
	cbnz	x0, 1b
2:
	ldrb	w4, [x5], #1
	strb	w4, [x1], #1
	cmp	x5, x6
	b.lo	2b
	ret

.bss
.balign	16
buf:	.space	65536
num:	.space	32
obuf:	.space	80
