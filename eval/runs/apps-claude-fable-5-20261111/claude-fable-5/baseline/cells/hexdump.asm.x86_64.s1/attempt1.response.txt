.section .rodata
hexd: .ascii "0123456789abcdef"

.bss
.balign 16
inbuf: .space 65536
linebuf: .space 128

.text
.globl _start
_start:
    xor r12, r12                # total bytes read

rdloop:
    mov edx, 65536
    sub edx, r12d
    jz rddone
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax                # read
    xor edi, edi                # fd 0
    syscall
    test rax, rax
    jz rddone
    add r12, rax
    jmp rdloop

rddone:
    xor r13, r13                # current offset

mainloop:
    cmp r13, r12
    jae finish
    mov r14, r12
    sub r14, r13                # remaining
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    lea rdi, [rip + linebuf]
    mov eax, r13d
    call put_off8
    mov al, ' '
    stosb
    stosb

    xor ebx, ebx                # column index i

hexcol:
    cmp rbx, r14
    jae pad3
    lea rsi, [rip + inbuf]
    lea rdx, [rip + hexd]
    movzx eax, byte ptr [rsi + r13 + rbx]
    mov ecx, eax
    shr ecx, 4
    mov al, byte ptr [rdx + rcx]
    stosb
    movzx eax, byte ptr [rsi + r13 + rbx]
    and eax, 15
    mov al, byte ptr [rdx + rax]
    stosb
    mov al, ' '
    stosb
    jmp aftercol
pad3:
    mov al, ' '
    stosb
    stosb
    stosb
aftercol:
    cmp ebx, 7
    jne 2f
    mov al, ' '
    stosb
2:
    inc ebx
    cmp ebx, 16
    jb hexcol

    mov al, ' '
    stosb
    mov al, '|'
    stosb

    xor ebx, ebx
charcol:
    cmp rbx, r14
    jae chardone
    lea rsi, [rip + inbuf]
    movzx eax, byte ptr [rsi + r13 + rbx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    jbe keep
dot:
    mov al, '.'
keep:
    stosb
    inc rbx
    jmp charcol
chardone:
    mov al, '|'
    stosb
    mov al, 10
    stosb

    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r13, r14
    jmp mainloop

finish:
    lea rdi, [rip + linebuf]
    mov eax, r12d
    call put_off8
    mov al, 10
    stosb
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 231
    syscall

# put_off8: write eax as 8 lowercase hex digits at [rdi], advance rdi.
# clobbers eax, ebx, ecx, rdx
put_off8:
    mov ecx, 8
    lea rdx, [rip + hexd]
1:
    rol eax, 4
    mov ebx, eax
    and ebx, 15
    mov bl, byte ptr [rdx + rbx]
    mov byte ptr [rdi], bl
    inc rdi
    dec ecx
    jnz 1b
    ret
