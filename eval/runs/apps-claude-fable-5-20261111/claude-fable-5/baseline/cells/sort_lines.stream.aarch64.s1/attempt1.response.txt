	.text
	.globl _start
_start:
	// x19 = input buffer, x20 = total bytes read
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0

read_loop:
	mov	x2, #32768
	subs	x2, x2, x20
	b.eq	read_done
	mov	x0, #0
	add	x1, x19, x20
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop

read_done:
	// parse lines: x21 = line table, x22 = line count
	adrp	x21, lines
	add	x21, x21, :lo12:lines
	mov	x22, #0
	mov	x0, #0		// pos
	mov	x1, #0		// start of current line

parse_loop:
	cmp	x0, x20
	b.ge	parse_tail
	ldrb	w2, [x19, x0]
	cmp	w2, #10
	b.ne	parse_next
	// record line [start, pos)
	add	x3, x19, x1	// pointer
	sub	x4, x0, x1	// length
	add	x5, x21, x22, lsl #4
	stp	x3, x4, [x5]
	add	x22, x22, #1
	add	x1, x0, #1
parse_next:
	add	x0, x0, #1
	b	parse_loop

parse_tail:
	cmp	x1, x20
	b.ge	sort
	add	x3, x19, x1
	sub	x4, x20, x1
	add	x5, x21, x22, lsl #4
	stp	x3, x4, [x5]
	add	x22, x22, #1

sort:
	// insertion sort over x22 records of 16 bytes
	mov	x24, #1		// i
outer:
	cmp	x24, x22
	b.ge	output
	add	x3, x21, x24, lsl #4
	ldp	x9, x10, [x3]	// key: ptr, len
	sub	x25, x24, #1	// j
inner:
	tbnz	x25, #63, place	// j < 0
	add	x3, x21, x25, lsl #4
	ldp	x11, x12, [x3]	// rec[j]: ptr, len
	// compare rec[j] vs key (memcmp, shorter first on prefix)
	cmp	x12, x10
	csel	x4, x12, x10, lo	// min length
	mov	x5, #0
cmp_loop:
	cmp	x5, x4
	b.eq	cmp_len
	ldrb	w6, [x11, x5]
	ldrb	w7, [x9, x5]
	cmp	w6, w7
	b.hi	shift		// rec[j] > key
	b.lo	place		// rec[j] < key
	add	x5, x5, #1
	b	cmp_loop
cmp_len:
	cmp	x12, x10
	b.ls	place		// rec[j] <= key
shift:
	add	x6, x25, #1
	add	x3, x21, x6, lsl #4
	stp	x11, x12, [x3]
	sub	x25, x25, #1
	b	inner
place:
	add	x25, x25, #1
	add	x3, x21, x25, lsl #4
	stp	x9, x10, [x3]
	add	x24, x24, #1
	b	outer

output:
	// build output buffer: each line + '\n'
	adrp	x26, outbuf
	add	x26, x26, :lo12:outbuf
	mov	x27, #0		// output length
	mov	x24, #0		// line index
out_loop:
	cmp	x24, x22
	b.ge	out_write
	add	x3, x21, x24, lsl #4
	ldp	x9, x10, [x3]
	mov	x5, #0
copy_loop:
	cmp	x5, x10
	b.ge	copy_done
	ldrb	w6, [x9, x5]
	strb	w6, [x26, x27]
	add	x5, x5, #1
	add	x27, x27, #1
	b	copy_loop
copy_done:
	mov	w6, #10
	strb	w6, [x26, x27]
	add	x27, x27, #1
	add	x24, x24, #1
	b	out_loop

out_write:
	mov	x28, #0		// bytes written
write_loop:
	cmp	x28, x27
	b.ge	done
	mov	x0, #1
	add	x1, x26, x28
	sub	x2, x27, x28
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	b.le	done
	add	x28, x28, x0
	b	write_loop

done:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.bss
	.balign 8
inbuf:
	.space	32768
	.balign 8
lines:
	.space	16000
	.balign 8
outbuf:
	.space	33768
