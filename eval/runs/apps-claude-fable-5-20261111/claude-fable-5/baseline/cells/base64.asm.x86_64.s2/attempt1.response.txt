.text
.globl _start
_start:
    # ---- read all of stdin ----
    xor     r12d, r12d              # total bytes read
rloop:
    lea     rsi, [rip + inbuf]
    add     rsi, r12
    mov     edx, 49152
    sub     edx, r12d
    xor     edi, edi
    xor     eax, eax                # read
    syscall
    test    rax, rax
    jz      rdone
    add     r12, rax
    jmp     rloop
rdone:
    # ---- encode ----
    lea     rsi, [rip + inbuf]      # input pointer
    lea     rdi, [rip + outbuf]     # output pointer
    xor     r13d, r13d              # column counter
enc_loop:
    cmp     r12, 3
    jb      tail
    movzx   r14d, byte ptr [rsi]
    shl     r14d, 16
    movzx   eax, byte ptr [rsi + 1]
    shl     eax, 8
    or      r14d, eax
    movzx   eax, byte ptr [rsi + 2]
    or      r14d, eax
    mov     ecx, r14d
    shr     ecx, 18
    and     ecx, 63
    call    emit64
    mov     ecx, r14d
    shr     ecx, 12
    and     ecx, 63
    call    emit64
    mov     ecx, r14d
    shr     ecx, 6
    and     ecx, 63
    call    emit64
    mov     ecx, r14d
    and     ecx, 63
    call    emit64
    add     rsi, 3
    sub     r12, 3
    jmp     enc_loop
tail:
    test    r12, r12
    jz      finish
    cmp     r12, 1
    je      one_byte
    # two bytes remain
    movzx   r14d, byte ptr [rsi]
    shl     r14d, 8
    movzx   eax, byte ptr [rsi + 1]
    or      r14d, eax
    shl     r14d, 2                 # 18 bits
    mov     ecx, r14d
    shr     ecx, 12
    and     ecx, 63
    call    emit64
    mov     ecx, r14d
    shr     ecx, 6
    and     ecx, 63
    call    emit64
    mov     ecx, r14d
    and     ecx, 63
    call    emit64
    mov     al, 61                  # '='
    call    putc
    jmp     finish
one_byte:
    movzx   r14d, byte ptr [rsi]
    shl     r14d, 4                 # 12 bits
    mov     ecx, r14d
    shr     ecx, 6
    and     ecx, 63
    call    emit64
    mov     ecx, r14d
    and     ecx, 63
    call    emit64
    mov     al, 61
    call    putc
    mov     al, 61
    call    putc
finish:
    test    r13, r13
    jz      do_write
    mov     byte ptr [rdi], 10
    inc     rdi
do_write:
    lea     rsi, [rip + outbuf]
    mov     r12, rdi
    sub     r12, rsi                # total output length
wloop:
    test    r12, r12
    jz      done
    mov     eax, 1                  # write
    mov     edi, 1
    mov     rdx, r12
    syscall
    add     rsi, rax
    sub     r12, rax
    jmp     wloop
done:
    mov     eax, 60                 # exit
    xor     edi, edi
    syscall

# emit alphabet character for index in ecx (0..63)
emit64:
    mov     al, byte ptr [rip + alphabet + rcx]
    # fall through to putc
putc:
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r13
    cmp     r13, 76
    jne     1f
    mov     byte ptr [rdi], 10
    inc     rdi
    xor     r13d, r13d
1:
    ret

.section .rodata
alphabet:
    .ascii  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space  49152
outbuf:
    .space  67584
