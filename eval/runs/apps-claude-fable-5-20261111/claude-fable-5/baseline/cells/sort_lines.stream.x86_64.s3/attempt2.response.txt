.text
.globl _start

_start:
        # ---------- read all input ----------
        xor     r12, r12                # total bytes read
read_loop:
        mov     rdx, 32768
        sub     rdx, r12
        jz      read_done
        lea     rsi, [rip + inbuf]
        add     rsi, r12
        xor     edi, edi                # fd 0
        xor     eax, eax                # read
        syscall
        test    rax, rax
        jle     read_done
        add     r12, rax
        jmp     read_loop
read_done:

        # ---------- parse lines ----------
        xor     r13, r13                # line count
        xor     rbx, rbx                # pos
        lea     rbp, [rip + inbuf]
parse_loop:
        cmp     rbx, r12
        jae     parse_done
        mov     r14, rbx                # start
scan:
        cmp     rbx, r12
        jae     scan_done
        cmp     byte ptr [rbp + rbx], 0x0A
        je      scan_done
        inc     rbx
        jmp     scan
scan_done:
        # store line (ptr = inbuf+start, len = pos-start)
        lea     rax, [rbp + r14]
        mov     rdx, rbx
        sub     rdx, r14
        mov     rcx, r13
        shl     rcx, 4
        lea     rsi, [rip + lines]
        mov     qword ptr [rsi + rcx], rax
        mov     qword ptr [rsi + rcx + 8], rdx
        inc     r13
        cmp     rbx, r12
        jae     parse_done
        inc     rbx                     # skip the newline
        jmp     parse_loop
parse_done:

        # ---------- insertion sort ----------
        mov     rbx, 1                  # i
sort_outer:
        cmp     rbx, r13
        jae     sort_done
        mov     rcx, rbx
        shl     rcx, 4
        lea     rbp, [rip + lines]
        mov     r14, qword ptr [rbp + rcx]      # key ptr
        mov     r15, qword ptr [rbp + rcx + 8]  # key len
        mov     rdx, rbx                # slot index (j+1)
sort_inner:
        test    rdx, rdx
        jz      sort_insert
        mov     rcx, rdx
        dec     rcx
        shl     rcx, 4
        mov     rsi, qword ptr [rbp + rcx]      # lines[j].ptr
        mov     r8,  qword ptr [rbp + rcx + 8]  # lines[j].len
        mov     rdi, r14
        mov     r9,  r15
        call    cmp_lines
        test    rax, rax
        jle     sort_insert
        # shift lines[j] into lines[j+1]
        mov     rcx, rdx
        dec     rcx
        shl     rcx, 4
        mov     rax, qword ptr [rbp + rcx]
        mov     rsi, qword ptr [rbp + rcx + 8]
        mov     rcx, rdx
        shl     rcx, 4
        mov     qword ptr [rbp + rcx], rax
        mov     qword ptr [rbp + rcx + 8], rsi
        dec     rdx
        jmp     sort_inner
sort_insert:
        mov     rcx, rdx
        shl     rcx, 4
        mov     qword ptr [rbp + rcx], r14
        mov     qword ptr [rbp + rcx + 8], r15
        inc     rbx
        jmp     sort_outer
sort_done:

        # ---------- build output ----------
        lea     rdi, [rip + outbuf]
        lea     rbp, [rip + lines]
        xor     rbx, rbx                # index
out_loop:
        cmp     rbx, r13
        jae     out_done
        mov     rdx, rbx
        shl     rdx, 4
        mov     rsi, qword ptr [rbp + rdx]
        mov     rcx, qword ptr [rbp + rdx + 8]
        rep movsb
        mov     byte ptr [rdi], 0x0A
        inc     rdi
        inc     rbx
        jmp     out_loop
out_done:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi                # total output bytes
write_loop:
        test    rdx, rdx
        jz      done
        mov     rax, 1                  # write
        mov     rdi, 1                  # stdout
        push    rsi
        push    rdx
        syscall
        pop     rdx
        pop     rsi
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
done:
        xor     edi, edi
        mov     rax, 60                 # exit
        syscall

# cmp_lines: rsi=ptrA, r8=lenA, rdi=ptrB, r9=lenB
# compares unsigned bytes; returns rax <0 / 0 / >0 (sign-extended 64-bit)
cmp_lines:
        mov     r10, r8
        cmp     r10, r9
        cmova   r10, r9                 # r10 = min(lenA, lenB)
cl_loop:
        test    r10, r10
        jz      cl_len
        movzx   eax, byte ptr [rsi]
        movzx   r11d, byte ptr [rdi]
        sub     eax, r11d
        jnz     cl_diff
        inc     rsi
        inc     rdi
        dec     r10
        jmp     cl_loop
cl_diff:
        cdqe                            # sign-extend eax into rax
        ret
cl_len:
        mov     rax, r8
        sub     rax, r9
        ret

.bss
.balign 16
inbuf:
        .space  32768
outbuf:
        .space  33792
lines:
        .space  16000
