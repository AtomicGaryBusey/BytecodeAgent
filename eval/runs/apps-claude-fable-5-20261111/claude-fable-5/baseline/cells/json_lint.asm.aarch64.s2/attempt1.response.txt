.text
.globl _start

_start:
    // read entire input into buf
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x21, #0                 // length
rd_loop:
    mov     x8, #63                 // read
    mov     x0, #0
    add     x1, x19, x21
    mov     x2, #65536
    add     x2, x2, #64
    sub     x2, x2, x21
    svc     #0
    cmp     x0, #0
    b.le    rd_done
    add     x21, x21, x0
    b       rd_loop
rd_done:
    mov     x20, #0                 // position
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.ne    fail_pos
    mov     x8, #64                 // write
    mov     x0, #1
    adr     x1, msg_ok
    mov     x2, #3
    svc     #0
    mov     x8, #93                 // exit
    mov     x0, #0
    svc     #0

// -------- whitespace --------
skip_ws:
1:  cmp     x20, x21
    b.hs    2f
    ldrb    w0, [x19, x20]
    cmp     w0, #0x20
    b.eq    3f
    cmp     w0, #0x09
    b.eq    3f
    cmp     w0, #0x0a
    b.eq    3f
    cmp     w0, #0x0d
    b.ne    2f
3:  add     x20, x20, #1
    b       1b
2:  ret

// -------- value dispatch --------
parse_value:
    stp     x29, x30, [sp, #-16]!
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x7b               // {
    b.eq    pv_obj
    cmp     w0, #0x5b               // [
    b.eq    pv_arr
    cmp     w0, #0x22               // "
    b.eq    pv_str
    cmp     w0, #0x2d               // -
    b.eq    pv_num
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pv_num
    cmp     w0, #0x74               // t
    b.eq    pv_true
    cmp     w0, #0x66               // f
    b.eq    pv_false
    cmp     w0, #0x6e               // n
    b.eq    pv_null
    b       fail_pos
pv_obj:
    bl      parse_object
    b       pv_ret
pv_arr:
    bl      parse_array
    b       pv_ret
pv_str:
    bl      parse_string
    b       pv_ret
pv_num:
    bl      parse_number
    b       pv_ret
pv_true:
    adr     x0, lit_true
    mov     x1, #4
    bl      parse_lit
    b       pv_ret
pv_false:
    adr     x0, lit_false
    mov     x1, #5
    bl      parse_lit
    b       pv_ret
pv_null:
    adr     x0, lit_null
    mov     x1, #4
    bl      parse_lit
    b       pv_ret
pv_ret:
    ldp     x29, x30, [sp], #16
    ret

// -------- literal: x0 = ptr, x1 = count --------
parse_lit:
1:  cbz     x1, 2f
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w2, [x0], #1
    ldrb    w3, [x19, x20]
    cmp     w2, w3
    b.ne    fail_pos
    add     x20, x20, #1
    sub     x1, x1, #1
    b       1b
2:  ret

// -------- string: current byte is '"' --------
parse_string:
    add     x20, x20, #1
ps_loop:
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x22               // "
    b.eq    ps_end
    cmp     w0, #0x5c               // backslash
    b.eq    ps_esc
    cmp     w0, #0x20
    b.lo    fail_pos
    add     x20, x20, #1
    b       ps_loop
ps_end:
    add     x20, x20, #1
    ret
ps_esc:
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x75               // u
    b.eq    ps_u
    cmp     w0, #0x22               // "
    b.eq    ps_ok
    cmp     w0, #0x5c
    b.eq    ps_ok
    cmp     w0, #0x2f               // /
    b.eq    ps_ok
    cmp     w0, #0x62               // b
    b.eq    ps_ok
    cmp     w0, #0x66               // f
    b.eq    ps_ok
    cmp     w0, #0x6e               // n
    b.eq    ps_ok
    cmp     w0, #0x72               // r
    b.eq    ps_ok
    cmp     w0, #0x74               // t
    b.eq    ps_ok
    b       fail_pos
ps_ok:
    add     x20, x20, #1
    b       ps_loop
ps_u:
    add     x20, x20, #1
    mov     x4, #4
ps_hex:
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    1f
    orr     w1, w0, #0x20
    sub     w1, w1, #0x61           // a
    cmp     w1, #5
    b.hi    fail_pos
1:  add     x20, x20, #1
    subs    x4, x4, #1
    b.ne    ps_hex
    b       ps_loop

// -------- number: current byte is '-' or digit --------
parse_number:
    ldrb    w0, [x19, x20]
    cmp     w0, #0x2d               // -
    b.ne    1f
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
1:  sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail_pos
    add     x20, x20, #1
    cmp     w0, #0x30               // leading 0: no more int digits
    b.eq    pn_frac
pn_dloop:
    cmp     x20, x21
    b.hs    pn_done
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    pn_frac_chk
    add     x20, x20, #1
    b       pn_dloop
pn_frac:
    cmp     x20, x21
    b.hs    pn_done
    ldrb    w0, [x19, x20]
pn_frac_chk:                        // w0 = byte at x20
    cmp     w0, #0x2e               // .
    b.ne    pn_exp_chk
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail_pos
    add     x20, x20, #1
pn_floop:
    cmp     x20, x21
    b.hs    pn_done
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    pn_exp_chk
    add     x20, x20, #1
    b       pn_floop
pn_exp_chk:                         // w0 = byte at x20
    orr     w1, w0, #0x20
    cmp     w1, #0x65               // e / E
    b.ne    pn_done
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x2b               // +
    b.eq    2f
    cmp     w0, #0x2d               // -
    b.ne    3f
2:  add     x20, x20, #1
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
3:  sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail_pos
    add     x20, x20, #1
pn_eloop:
    cmp     x20, x21
    b.hs    pn_done
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    pn_done
    add     x20, x20, #1
    b       pn_eloop
pn_done:
    ret

// -------- object: current byte is '{' --------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x7d               // }
    b.ne    po_member
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
po_member:                          // w0 = byte at x20
    cmp     w0, #0x22               // "
    b.ne    fail_pos
    bl      parse_string
    bl      skip_ws
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x3a               // :
    b.ne    fail_pos
    add     x20, x20, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x7d               // }
    b.eq    po_end
    cmp     w0, #0x2c               // ,
    b.ne    fail_pos
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    b       po_member
po_end:
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret

// -------- array: current byte is '[' --------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.hs    pa_val                  // parse_value reports EOF
    ldrb    w0, [x19, x20]
    cmp     w0, #0x5d               // ]
    b.ne    pa_val
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
pa_val:
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.hs    fail_eof
    ldrb    w0, [x19, x20]
    cmp     w0, #0x5d               // ]
    b.eq    pa_end
    cmp     w0, #0x2c               // ,
    b.ne    fail_pos
    add     x20, x20, #1
    bl      skip_ws
    b       pa_val
pa_end:
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret

// -------- error reporting --------
fail_eof:
    mov     x0, x21
    b       fail
fail_pos:
    mov     x0, x20
fail:                               // x0 = offset N, never returns
    sub     sp, sp, #48
    add     x1, sp, #47
    mov     w2, #0x0a
    strb    w2, [x1]
    mov     x3, #10
1:  udiv    x4, x0, x3
    msub    x5, x4, x3, x0
    add     w5, w5, #0x30
    sub     x1, x1, #1
    strb    w5, [x1]
    mov     x0, x4
    cbnz    x0, 1b
    adr     x6, msg_err
    mov     x7, #9
2:  sub     x1, x1, #1
    sub     x7, x7, #1
    ldrb    w5, [x6, x7]
    strb    w5, [x1]
    cbnz    x7, 2b
    mov     x8, #64                 // write(1, x1, len)
    mov     x0, #1
    add     x2, sp, #48
    sub     x2, x2, x1
    svc     #0
    mov     x8, #93                 // exit(1)
    mov     x0, #1
    svc     #0

.section .rodata
msg_ok:
    .ascii  "ok\n"
msg_err:
    .ascii  "error at "
lit_true:
    .ascii  "true"
lit_false:
    .ascii  "false"
lit_null:
    .ascii  "null"

.bss
.balign 16
buf:
    .space  65600
