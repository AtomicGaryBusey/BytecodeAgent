// hexdump - canonical hex+ASCII dump of stdin (AArch64 Linux, freestanding)

.bss
.balign 8
inbuf:
    .space 65536
linebuf:
    .space 128

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.text
.globl _start
_start:
    // ---- read all of stdin into inbuf ----
    mov     x19, #0                 // total bytes read
    adrp    x20, inbuf
    add     x20, x20, :lo12:inbuf
rdloop:
    mov     x2, #65536
    sub     x2, x2, x19             // remaining space
    cbz     x2, rddone
    mov     x0, #0                  // fd 0
    add     x1, x20, x19
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    ble     rddone                  // 0 = EOF (errors treated as EOF)
    add     x19, x19, x0
    b       rdloop
rddone:

    adrp    x22, linebuf
    add     x22, x22, :lo12:linebuf
    adrp    x23, hexdig
    add     x23, x23, :lo12:hexdig

    mov     x21, #0                 // current offset
rowloop:
    cmp     x21, x19
    bge     rowsdone
    sub     x24, x19, x21           // bytes left
    mov     x3, #16
    cmp     x24, x3
    csel    x24, x24, x3, lo        // n = min(left, 16)

    mov     x25, x22                // line write pointer

    // offset as 8 hex digits
    mov     x4, #28
1:
    lsr     x5, x21, x4
    and     x5, x5, #0xf
    ldrb    w5, [x23, x5]
    strb    w5, [x25], #1
    subs    x4, x4, #4
    bge     1b

    // two spaces
    mov     w5, #0x20
    strb    w5, [x25], #1
    strb    w5, [x25], #1

    // 16 hex positions
    mov     x6, #0                  // i
hexloop:
    cmp     x6, x24
    bge     pad3
    add     x7, x21, x6
    ldrb    w7, [x20, x7]           // the byte
    lsr     x9, x7, #4
    ldrb    w9, [x23, x9]
    strb    w9, [x25], #1
    and     x9, x7, #0xf
    ldrb    w9, [x23, x9]
    strb    w9, [x25], #1
    mov     w9, #0x20
    strb    w9, [x25], #1
    b       afterpos
pad3:
    mov     w9, #0x20
    strb    w9, [x25], #1
    strb    w9, [x25], #1
    strb    w9, [x25], #1
afterpos:
    cmp     x6, #7
    bne     2f
    mov     w9, #0x20               // extra space after 8th slot
    strb    w9, [x25], #1
2:
    add     x6, x6, #1
    cmp     x6, #16
    blt     hexloop

    // " |"
    mov     w9, #0x20
    strb    w9, [x25], #1
    mov     w9, #0x7c               // '|'
    strb    w9, [x25], #1

    // ASCII column
    mov     x6, #0
chloop:
    cmp     x6, x24
    bge     chdone
    add     x7, x21, x6
    ldrb    w7, [x20, x7]
    cmp     w7, #0x20
    blo     dot
    cmp     w7, #0x7e
    bls     chok
dot:
    mov     w7, #0x2e               // '.'
chok:
    strb    w7, [x25], #1
    add     x6, x6, #1
    b       chloop
chdone:
    mov     w9, #0x7c               // '|'
    strb    w9, [x25], #1
    mov     w9, #10                 // '\n'
    strb    w9, [x25], #1

    // write the line
    mov     x0, #1
    mov     x1, x22
    sub     x2, x25, x22
    mov     x8, #64                 // write
    svc     #0

    add     x21, x21, #16
    b       rowloop

rowsdone:
    // final offset line: total length as 8 hex digits + newline
    mov     x25, x22
    mov     x4, #28
3:
    lsr     x5, x19, x4
    and     x5, x5, #0xf
    ldrb    w5, [x23, x5]
    strb    w5, [x25], #1
    subs    x4, x4, #4
    bge     3b
    mov     w5, #10
    strb    w5, [x25], #1

    mov     x0, #1
    mov     x1, x22
    mov     x2, #9
    mov     x8, #64                 // write
    svc     #0

    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0
