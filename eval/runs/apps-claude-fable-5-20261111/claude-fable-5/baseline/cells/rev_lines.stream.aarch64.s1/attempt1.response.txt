.text
.globl _start
_start:
	adrp	x19, buf
	add	x19, x19, :lo12:buf
	mov	x20, #0			// total bytes read
read_loop:
	mov	x2, #65536
	sub	x2, x2, x20
	cbz	x2, done_read
	mov	x0, #0
	add	x1, x19, x20
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	ble	done_read
	add	x20, x20, x0
	b	read_loop
done_read:
	cbz	x20, exit0
	// append newline if input does not end with one
	sub	x1, x20, #1
	ldrb	w2, [x19, x1]
	cmp	w2, #10
	beq	lines
	mov	w2, #10
	strb	w2, [x19, x20]
	add	x20, x20, #1
lines:
	mov	x21, #0			// current position
line_loop:
	cmp	x21, x20
	bge	exit0
	// find newline starting at x21
	mov	x22, x21
find_nl:
	ldrb	w2, [x19, x22]
	cmp	w2, #10
	beq	found_nl
	add	x22, x22, #1
	b	find_nl
found_nl:
	// reverse bytes [x21, x22-1] in place
	mov	x3, x21
	sub	x4, x22, #1
rev_loop:
	cmp	x3, x4
	bge	write_line
	ldrb	w5, [x19, x3]
	ldrb	w6, [x19, x4]
	strb	w6, [x19, x3]
	strb	w5, [x19, x4]
	add	x3, x3, #1
	sub	x4, x4, #1
	b	rev_loop
write_line:
	mov	x0, #1
	add	x1, x19, x21
	sub	x2, x22, x21
	add	x2, x2, #1		// include the newline
	mov	x8, #64			// write
	svc	#0
	add	x21, x22, #1
	b	line_loop
exit0:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

.bss
.balign 8
buf:
	.space	65537
