.text
.globl _start

_start:
        xor     r12d, r12d              # total bytes read
read_loop:
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + inbuf]
        add     rsi, r12
        mov     edx, 65536
        sub     edx, r12d
        syscall
        test    rax, rax
        jle     read_done
        add     r12, rax
        jmp     read_loop

read_done:
        xor     r13d, r13d              # current offset
row_loop:
        cmp     r13, r12
        jae     finish
        mov     r14, r12
        sub     r14, r13                # remaining
        cmp     r14, 16
        jbe     1f
        mov     r14, 16
1:
        lea     rdi, [rip + linebuf]
        mov     eax, r13d
        call    hex8
        mov     word ptr [rdi], 0x2020  # two spaces
        add     rdi, 2
        lea     rsi, [rip + inbuf]
        add     rsi, r13
        xor     ecx, ecx
col_loop:
        cmp     rcx, r14
        jae     blanks
        movzx   eax, byte ptr [rsi + rcx]
        mov     edx, eax
        shr     edx, 4
        mov     dl, byte ptr [hextab + rdx]
        mov     byte ptr [rdi], dl
        and     eax, 15
        mov     al, byte ptr [hextab + rax]
        mov     byte ptr [rdi + 1], al
        mov     byte ptr [rdi + 2], 0x20
        jmp     col_adv
blanks:
        mov     byte ptr [rdi], 0x20
        mov     byte ptr [rdi + 1], 0x20
        mov     byte ptr [rdi + 2], 0x20
col_adv:
        add     rdi, 3
        cmp     ecx, 7
        jne     2f
        mov     byte ptr [rdi], 0x20    # extra space after 8th slot
        inc     rdi
2:
        inc     ecx
        cmp     ecx, 16
        jb      col_loop

        mov     byte ptr [rdi], 0x20
        mov     byte ptr [rdi + 1], '|'
        add     rdi, 2
        xor     ecx, ecx
char_loop:
        cmp     rcx, r14
        jae     char_done
        mov     al, byte ptr [rsi + rcx]
        cmp     al, 0x20
        jb      dot
        cmp     al, 0x7e
        jbe     store_c
dot:
        mov     al, '.'
store_c:
        mov     byte ptr [rdi], al
        inc     rdi
        inc     ecx
        jmp     char_loop
char_done:
        mov     byte ptr [rdi], '|'
        mov     byte ptr [rdi + 1], 10
        add     rdi, 2

        lea     rsi, [rip + linebuf]
        mov     rdx, rdi
        sub     rdx, rsi
        mov     edi, 1
        mov     eax, 1                  # sys_write
        syscall

        add     r13, 16
        jmp     row_loop

finish:
        lea     rdi, [rip + linebuf]
        mov     eax, r12d
        call    hex8
        mov     byte ptr [rdi], 10
        lea     rsi, [rip + linebuf]
        mov     edx, 9
        mov     edi, 1
        mov     eax, 1                  # sys_write
        syscall

        xor     edi, edi
        mov     eax, 231                # exit_group(0)
        syscall

# hex8: write eax as 8 lowercase hex digits at rdi, advance rdi by 8.
# clobbers eax, ecx, edx
hex8:
        mov     ecx, 8
1:
        rol     eax, 4
        mov     edx, eax
        and     edx, 15
        mov     dl, byte ptr [hextab + rdx]
        mov     byte ptr [rdi], dl
        inc     rdi
        dec     ecx
        jnz     1b
        ret

.section .rodata
hextab:
        .ascii  "0123456789abcdef"

.bss
.balign 16
inbuf:
        .space  65536
linebuf:
        .space  128
