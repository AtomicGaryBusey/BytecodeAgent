.section .rodata
hextab: .ascii "0123456789abcdef"

.bss
.balign 8
buf:  .space 65536
line: .space 96

.text
.globl _start
_start:
    mov     x19, #0                 // total bytes read
    adrp    x20, buf
    add     x20, x20, :lo12:buf
read_loop:
    mov     x2, #65536
    subs    x2, x2, x19
    b.le    read_done
    mov     x0, #0
    add     x1, x20, x19
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x19, x19, x0
    b       read_loop
read_done:
    adrp    x25, hextab
    add     x25, x25, :lo12:hextab
    adrp    x23, line
    add     x23, x23, :lo12:line
    mov     x21, #0                 // current offset
row_loop:
    cmp     x21, x19
    b.hs    finish
    sub     x22, x19, x21           // n = min(16, remaining)
    cmp     x22, #16
    mov     x5, #16
    csel    x22, x22, x5, lo
    mov     x24, x23                // cursor
    mov     w5, #28                 // offset: 8 hex digits
off_loop:
    lsr     w6, w21, w5
    and     w6, w6, #0xf
    ldrb    w6, [x25, w6, uxtw]
    strb    w6, [x24], #1
    subs    w5, w5, #4
    b.ge    off_loop
    mov     w5, #0x20               // two spaces
    strb    w5, [x24], #1
    strb    w5, [x24], #1
    mov     x7, #0                  // hex column loop i = 0..15
hex_loop:
    cmp     x7, x22
    b.hs    hex_pad
    add     x6, x21, x7
    ldrb    w6, [x20, x6]
    lsr     w5, w6, #4
    ldrb    w5, [x25, w5, uxtw]
    strb    w5, [x24], #1
    and     w5, w6, #0xf
    ldrb    w5, [x25, w5, uxtw]
    strb    w5, [x24], #1
    mov     w5, #0x20
    strb    w5, [x24], #1
    b       hex_next
hex_pad:
    mov     w5, #0x20
    strb    w5, [x24], #1
    strb    w5, [x24], #1
    strb    w5, [x24], #1
hex_next:
    cmp     x7, #7
    b.ne    1f
    mov     w5, #0x20               // extra space after slot 7
    strb    w5, [x24], #1
1:
    add     x7, x7, #1
    cmp     x7, #16
    b.lo    hex_loop
    mov     w5, #0x20               // space, '|'
    strb    w5, [x24], #1
    mov     w5, #0x7c
    strb    w5, [x24], #1
    mov     x7, #0                  // ASCII column
asc_loop:
    cmp     x7, x22
    b.hs    asc_done
    add     x6, x21, x7
    ldrb    w6, [x20, x6]
    cmp     w6, #0x20
    b.lo    asc_dot
    cmp     w6, #0x7e
    b.ls    asc_store
asc_dot:
    mov     w6, #0x2e
asc_store:
    strb    w6, [x24], #1
    add     x7, x7, #1
    b       asc_loop
asc_done:
    mov     w5, #0x7c               // '|'
    strb    w5, [x24], #1
    mov     w5, #10                 // newline
    strb    w5, [x24], #1
    mov     x0, x23
    sub     x1, x24, x23
    bl      write_all
    add     x21, x21, x22
    b       row_loop
finish:
    mov     x24, x23                // final offset line
    mov     w5, #28
fin_off:
    lsr     w6, w19, w5
    and     w6, w6, #0xf
    ldrb    w6, [x25, w6, uxtw]
    strb    w6, [x24], #1
    subs    w5, w5, #4
    b.ge    fin_off
    mov     w5, #10
    strb    w5, [x24], #1
    mov     x0, x23
    mov     x1, #9
    bl      write_all
    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0

// write_all(x0 = ptr, x1 = len): write everything to stdout
write_all:
    mov     x3, x0
    mov     x4, x1
1:
    cbz     x4, 2f
    mov     x0, #1
    mov     x1, x3
    mov     x2, x4
    mov     x8, #64                 // write
    svc     #0
    cmp     x0, #0
    b.le    2f
    add     x3, x3, x0
    sub     x4, x4, x0
    b       1b
2:
    ret
