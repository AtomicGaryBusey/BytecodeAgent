.text
.globl _start

_start:
    // read all of stdin into buf
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x2, #32768
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    // parse lines into (ptr, len) pairs
    adrp    x22, lines
    add     x22, x22, :lo12:lines
    mov     x21, #0                 // line count
    mov     x4, #0                  // index i
    mov     x5, #0                  // line start
parse_loop:
    cmp     x4, x20
    b.ge    parse_end
    ldrb    w6, [x19, x4]
    cmp     w6, #10
    b.ne    1f
    add     x7, x19, x5             // line pointer
    sub     x6, x4, x5              // line length
    add     x9, x22, x21, lsl #4
    stp     x7, x6, [x9]
    add     x21, x21, #1
    add     x5, x4, #1
1:
    add     x4, x4, #1
    b       parse_loop
parse_end:
    cmp     x5, x20
    b.ge    sort
    add     x7, x19, x5             // final line without newline
    sub     x6, x20, x5
    add     x9, x22, x21, lsl #4
    stp     x7, x6, [x9]
    add     x21, x21, #1

sort:
    // insertion sort on the (ptr, len) table
    mov     x23, #1                 // i
sort_outer:
    cmp     x23, x21
    b.ge    output
    add     x9, x22, x23, lsl #4
    ldp     x24, x25, [x9]          // key ptr, len
    sub     x26, x23, #1            // j
sort_inner:
    tbnz    x26, #63, sort_place    // j < 0
    add     x9, x22, x26, lsl #4
    ldp     x27, x28, [x9]          // lines[j]
    mov     x0, x27
    mov     x1, x28
    mov     x2, x24
    mov     x3, x25
    bl      cmp_lines
    cmp     x0, #0
    b.le    sort_place
    add     x9, x22, x26, lsl #4
    stp     x27, x28, [x9, #16]     // shift up
    sub     x26, x26, #1
    b       sort_inner
sort_place:
    add     x9, x22, x26, lsl #4
    stp     x24, x25, [x9, #16]
    add     x23, x23, #1
    b       sort_outer

output:
    adrp    x24, nl
    add     x24, x24, :lo12:nl
    mov     x23, #0
out_loop:
    cmp     x23, x21
    b.ge    done
    add     x9, x22, x23, lsl #4
    ldp     x1, x2, [x9]
    mov     x0, #1
    mov     x8, #64                 // write line
    svc     #0
    mov     x0, #1
    mov     x1, x24
    mov     x2, #1
    mov     x8, #64                 // write newline
    svc     #0
    add     x23, x23, #1
    b       out_loop

done:
    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0
hang:
    b       hang

// cmp_lines: x0=ptrA, x1=lenA, x2=ptrB, x3=lenB
// returns x0 <0 / 0 / >0 ; clobbers x4-x7
cmp_lines:
    cmp     x1, x3
    csel    x4, x1, x3, lo          // min(lenA, lenB)
    mov     x5, #0
1:
    cmp     x5, x4
    b.eq    2f
    ldrb    w6, [x0, x5]
    ldrb    w7, [x2, x5]
    cmp     w6, w7
    b.ne    3f
    add     x5, x5, #1
    b       1b
3:
    sub     w0, w6, w7
    sxtw    x0, w0
    ret
2:
    subs    x0, x1, x3
    ret

.section .rodata
nl:
    .byte   10

.bss
.balign 8
buf:
    .space  32768
lines:
    .space  16016
