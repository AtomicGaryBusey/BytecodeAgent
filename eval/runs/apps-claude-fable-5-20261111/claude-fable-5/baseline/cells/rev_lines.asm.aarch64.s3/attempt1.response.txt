.text
.globl _start
_start:
    // read all input into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf      // x19 = inbuf base
    mov x20, #0                    // x20 = total bytes read
1:
    mov x0, #0                     // fd 0
    add x1, x19, x20               // buf + total
    mov x2, #65536
    sub x2, x2, x20                // remaining space
    cbz x2, 2f
    mov x8, #63                    // read
    svc #0
    cbz x0, 2f
    add x20, x20, x0
    b 1b
2:
    // build output in outbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf     // x21 = outbuf base
    mov x22, #0                    // output length
    mov x23, #0                    // line start index
    mov x24, #0                    // scan index
3:
    cmp x24, x20
    b.ge 6f                        // end of input
    ldrb w0, [x19, x24]
    cmp w0, #10
    b.ne 5f
    // found newline: emit reversed buf[x23 .. x24-1]
    mov x25, x24                   // one past last byte of line
4:
    cmp x25, x23
    b.le 41f
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b 4b
41:
    mov w0, #10
    strb w0, [x21, x22]
    add x22, x22, #1
    add x23, x24, #1               // next line start
5:
    add x24, x24, #1
    b 3b
6:
    // final partial line (no trailing newline)
    cmp x23, x20
    b.ge 8f
    mov x25, x20
7:
    cmp x25, x23
    b.le 71f
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b 7b
71:
    mov w0, #10
    strb w0, [x21, x22]
    add x22, x22, #1
8:
    // write outbuf, handling partial writes
    mov x20, #0                    // bytes written
9:
    cmp x20, x22
    b.ge 10f
    mov x0, #1                     // fd 1
    add x1, x21, x20
    sub x2, x22, x20
    mov x8, #64                    // write
    svc #0
    cmp x0, #0
    b.le 10f
    add x20, x20, x0
    b 9b
10:
    mov x0, #0
    mov x8, #93                    // exit
    svc #0

.bss
.balign 8
inbuf:
    .space 65536
outbuf:
    .space 65537
