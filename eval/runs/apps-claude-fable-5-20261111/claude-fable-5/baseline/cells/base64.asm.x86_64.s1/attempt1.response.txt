.text
.globl _start

_start:
        xor     r12d, r12d              # total bytes read
read_loop:
        lea     rsi, [rip + inbuf]
        add     rsi, r12
        mov     edx, 49160
        sub     edx, r12d
        jz      read_done
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        syscall
        test    rax, rax
        jle     read_done
        add     r12, rax
        jmp     read_loop
read_done:
        lea     rsi, [rip + inbuf]      # input pointer
        lea     rdi, [rip + outbuf]     # output pointer
        xor     r13d, r13d              # line length counter
        lea     rbx, [rip + b64tab]

enc_loop:
        cmp     r12, 3
        jb      tail
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        movzx   ecx, byte ptr [rsi + 1]
        shl     ecx, 8
        or      eax, ecx
        movzx   ecx, byte ptr [rsi + 2]
        or      eax, ecx
        add     rsi, 3
        sub     r12, 3
        mov     ebp, eax
        mov     eax, ebp
        shr     eax, 18
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        mov     eax, ebp
        shr     eax, 12
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        mov     eax, ebp
        shr     eax, 6
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        mov     eax, ebp
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        jmp     enc_loop

tail:
        test    r12, r12
        jz      finish
        movzx   ebp, byte ptr [rsi]
        shl     ebp, 16
        cmp     r12, 2
        jb      tail1
        movzx   ecx, byte ptr [rsi + 1]
        shl     ecx, 8
        or      ebp, ecx
tail1:
        mov     eax, ebp
        shr     eax, 18
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        mov     eax, ebp
        shr     eax, 12
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        cmp     r12, 2
        jb      pad2
        mov     eax, ebp
        shr     eax, 6
        and     eax, 63
        mov     al, byte ptr [rbx + rax]
        call    emit
        mov     al, 61                  # '='
        call    emit
        jmp     finish
pad2:
        mov     al, 61                  # '='
        call    emit
        mov     al, 61                  # '='
        call    emit

finish:
        test    r13, r13
        jz      write_out
        mov     byte ptr [rdi], 10
        inc     rdi
write_out:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
wr_loop:
        test    rdx, rdx
        jz      done
        mov     eax, 1                  # sys_write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     wr_loop
done:
        mov     eax, 60                 # sys_exit
        xor     edi, edi
        syscall

# emit: store character al at [rdi], advance, wrap lines at 76 chars
emit:
        mov     byte ptr [rdi], al
        inc     rdi
        inc     r13
        cmp     r13, 76
        jne     1f
        mov     byte ptr [rdi], 10
        inc     rdi
        xor     r13d, r13d
1:
        ret

.section .rodata
b64tab:
        .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 8
inbuf:
        .space 49160
outbuf:
        .space 67000
