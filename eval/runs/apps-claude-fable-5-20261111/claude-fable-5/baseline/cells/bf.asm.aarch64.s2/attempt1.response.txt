.text
.globl _start

_start:
    // ---- read all of stdin into inbuf ----
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
rd_loop:
    mov     x0, #65536
    sub     x2, x0, x20             // space left
    cbz     x2, rd_done
    add     x1, x19, x20
    mov     x0, #0                  // fd 0
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.le    rd_done
    add     x20, x20, x0
    b       rd_loop
rd_done:

    // ---- find first '!' to split program / input ----
    mov     x21, #0
find_loop:
    cmp     x21, x20
    b.ge    no_bang
    ldrb    w0, [x19, x21]
    cmp     w0, #0x21
    b.eq    got_bang
    add     x21, x21, #1
    b       find_loop
got_bang:
    add     x23, x19, x21
    add     x23, x23, #1            // input start
    add     x24, x19, x20           // input end
    mov     x20, x21                // program length
    b       have_split
no_bang:
    add     x23, x19, x20           // empty input
    mov     x24, x23
have_split:

    // ---- bracket check + build jump table ----
    adrp    x27, jtab
    add     x27, x27, :lo12:jtab
    adrp    x9, bstack
    add     x9, x9, :lo12:bstack
    mov     x10, #0                 // bracket stack depth
    mov     x11, #0                 // index i
bt_loop:
    cmp     x11, x20
    b.ge    bt_end
    ldrb    w0, [x19, x11]
    cmp     w0, #0x5B
    b.eq    bt_open
    cmp     w0, #0x5D
    b.eq    bt_close
bt_next:
    add     x11, x11, #1
    b       bt_loop
bt_open:
    str     w11, [x9, x10, lsl #2]
    add     x10, x10, #1
    b       bt_next
bt_close:
    cbz     x10, err
    sub     x10, x10, #1
    ldr     w12, [x9, x10, lsl #2]  // j = matching '[' index
    add     w13, w11, #1
    str     w13, [x27, w12, uxtw #2]    // jtab[j] = i+1
    add     w14, w12, #1
    str     w14, [x27, x11, lsl #2]     // jtab[i] = j+1
    b       bt_next
bt_end:
    cbnz    x10, err

    // ---- interpret ----
    adrp    x22, tape
    add     x22, x22, :lo12:tape    // data pointer
    adrp    x25, outbuf
    add     x25, x25, :lo12:outbuf
    mov     x26, #0                 // output count
    mov     x21, #0                 // pc
run:
    cmp     x21, x20
    b.ge    done
    ldrb    w0, [x19, x21]
    add     x21, x21, #1
    cmp     w0, #0x3E
    b.eq    c_right
    cmp     w0, #0x3C
    b.eq    c_left
    cmp     w0, #0x2B
    b.eq    c_plus
    cmp     w0, #0x2D
    b.eq    c_minus
    cmp     w0, #0x2E
    b.eq    c_dot
    cmp     w0, #0x2C
    b.eq    c_comma
    cmp     w0, #0x5B
    b.eq    c_open
    cmp     w0, #0x5D
    b.eq    c_close
    b       run
c_right:
    add     x22, x22, #1
    b       run
c_left:
    sub     x22, x22, #1
    b       run
c_plus:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       run
c_minus:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       run
c_dot:
    ldrb    w1, [x22]
    strb    w1, [x25, x26]
    add     x26, x26, #1
    b       run
c_comma:
    mov     w1, #0
    cmp     x23, x24
    b.ge    1f
    ldrb    w1, [x23], #1
1:
    strb    w1, [x22]
    b       run
c_open:
    ldrb    w1, [x22]
    cbnz    w1, run
    sub     x2, x21, #1
    ldr     w3, [x27, x2, lsl #2]
    mov     w21, w3
    b       run
c_close:
    ldrb    w1, [x22]
    cbz     w1, run
    sub     x2, x21, #1
    ldr     w3, [x27, x2, lsl #2]
    mov     w21, w3
    b       run

done:
    // ---- flush output ----
    mov     x1, x25
    mov     x2, x26
flush_loop:
    cbz     x2, exit0
    mov     x0, #1                  // fd 1
    mov     x8, #64                 // write
    svc     #0
    cmp     x0, #1
    b.lt    exit0
    add     x1, x1, x0
    sub     x2, x2, x0
    b       flush_loop

exit0:
    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0
err:
    mov     x0, #1
    mov     x8, #93                 // exit
    svc     #0

.bss
.balign 8
inbuf:
    .space 65536
tape:
    .space 30000
outbuf:
    .space 65536
.balign 4
jtab:
    .space 262144
bstack:
    .space 262144
