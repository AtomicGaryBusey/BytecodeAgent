.text
.globl _start
_start:
        # ---- read all of stdin into buf ----
        xor     ebx, ebx                # rbx = total bytes read
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jle     read_done               # buffer full
        xor     eax, eax                # sys_read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + buf]
        add     rsi, rbx
        syscall
        test    rax, rax
        jle     read_done               # 0 = EOF (or error)
        add     rbx, rax
        jmp     read_loop
read_done:
        # ---- split at first '!' ----
        lea     r12, [rip + buf]        # r12 = program base
        xor     ecx, ecx
find_bang:
        cmp     rcx, rbx
        je      no_bang
        cmp     byte ptr [r12 + rcx], 0x21
        je      found_bang
        inc     rcx
        jmp     find_bang
no_bang:
        mov     r13, rbx                # program length
        lea     r14, [r12 + rbx]        # input start (empty)
        jmp     after_split
found_bang:
        mov     r13, rcx                # program length
        lea     r14, [r12 + rcx + 1]    # input start
after_split:
        lea     r15, [r12 + rbx]        # input end

        # ---- bracket matching / validation ----
        mov     rbp, rsp
        xor     ecx, ecx
match_loop:
        cmp     rcx, r13
        je      match_done
        movzx   eax, byte ptr [r12 + rcx]
        cmp     al, 0x5B                # '['
        je      push_br
        cmp     al, 0x5D                # ']'
        je      pop_br
match_next:
        inc     rcx
        jmp     match_loop
push_br:
        push    rcx
        jmp     match_next
pop_br:
        cmp     rsp, rbp
        je      error                   # ']' with no open '['
        pop     rax
        lea     rdx, [rip + jmptab]
        mov     dword ptr [rdx + rax*4], ecx
        mov     dword ptr [rdx + rcx*4], eax
        jmp     match_next
match_done:
        cmp     rsp, rbp
        jne     error                   # unclosed '['

        # ---- interpret ----
        lea     rbx, [rip + tape]       # data pointer
        lea     rdi, [rip + outbuf]     # output write pointer
        xor     esi, esi                # instruction index
interp:
        cmp     rsi, r13
        jae     done
        movzx   eax, byte ptr [r12 + rsi]
        inc     rsi
        cmp     al, 0x3E                # '>'
        je      c_right
        cmp     al, 0x3C                # '<'
        je      c_left
        cmp     al, 0x2B                # '+'
        je      c_plus
        cmp     al, 0x2D                # '-'
        je      c_minus
        cmp     al, 0x2E                # '.'
        je      c_dot
        cmp     al, 0x2C                # ','
        je      c_comma
        cmp     al, 0x5B                # '['
        je      c_lbr
        cmp     al, 0x5D                # ']'
        je      c_rbr
        jmp     interp
c_right:
        inc     rbx
        jmp     interp
c_left:
        dec     rbx
        jmp     interp
c_plus:
        inc     byte ptr [rbx]
        jmp     interp
c_minus:
        dec     byte ptr [rbx]
        jmp     interp
c_dot:
        mov     al, byte ptr [rbx]
        mov     byte ptr [rdi], al
        inc     rdi
        jmp     interp
c_comma:
        xor     eax, eax
        cmp     r14, r15
        jae     1f
        mov     al, byte ptr [r14]
        inc     r14
1:
        mov     byte ptr [rbx], al
        jmp     interp
c_lbr:
        cmp     byte ptr [rbx], 0
        jne     interp
        jmp     do_jump
c_rbr:
        cmp     byte ptr [rbx], 0
        je      interp
do_jump:
        lea     rax, [rsi - 1]          # index of this bracket
        lea     rdx, [rip + jmptab]
        mov     esi, dword ptr [rdx + rax*4]
        inc     rsi                     # continue after the match
        jmp     interp

done:
        # ---- flush output ----
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
write_loop:
        test    rdx, rdx
        jle     exit0
        mov     eax, 1                  # sys_write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     exit0
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
exit0:
        xor     edi, edi
        mov     eax, 60                 # sys_exit
        syscall
error:
        mov     rsp, rbp
        mov     edi, 1
        mov     eax, 60
        syscall

.bss
.balign 8
buf:
        .space 65536
.balign 4
jmptab:
        .space 262144
tape:
        .space 30000
outbuf:
        .space 65536
