.bss
	.p2align 4
inbuf:
	.space 65536
outbuf:
	.space 65544

.text
	.globl _start
_start:
	xor	r12, r12		# total bytes read

read_loop:
	mov	edx, 65536
	sub	edx, r12d		# space left
	jz	read_done
	lea	rsi, [rip + inbuf]
	add	rsi, r12
	xor	eax, eax		# read
	xor	edi, edi		# fd 0
	syscall
	test	rax, rax
	jle	read_done
	add	r12, rax
	jmp	read_loop

read_done:
	lea	rsi, [rip + inbuf]	# current line start
	lea	rbp, [rsi + r12]	# end of input
	lea	rdi, [rip + outbuf]	# output cursor

line_loop:
	cmp	rsi, rbp
	jae	lines_done
	mov	rcx, rsi		# scan for newline
find_nl:
	cmp	rcx, rbp
	jae	got_end
	cmp	byte ptr [rcx], 0x0A
	je	got_end
	inc	rcx
	jmp	find_nl

got_end:
	mov	rdx, rcx		# copy [rsi, rcx) reversed
copy_rev:
	cmp	rdx, rsi
	jbe	copy_done
	dec	rdx
	mov	al, byte ptr [rdx]
	mov	byte ptr [rdi], al
	inc	rdi
	jmp	copy_rev

copy_done:
	mov	byte ptr [rdi], 0x0A
	inc	rdi
	lea	rsi, [rcx + 1]		# skip newline (or run past end)
	jmp	line_loop

lines_done:
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi		# total output length

write_loop:
	test	rdx, rdx
	jz	finish
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	syscall
	test	rax, rax
	jle	finish
	add	rsi, rax
	sub	rdx, rax
	jmp	write_loop

finish:
	mov	eax, 231		# exit_group
	xor	edi, edi
	syscall
