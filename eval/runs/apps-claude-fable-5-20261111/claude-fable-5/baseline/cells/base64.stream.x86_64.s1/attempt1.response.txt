.text
.globl _start

_start:
	# ---- read all of stdin into inbuf ----
	xor	r13, r13		# total bytes read
read_loop:
	mov	edx, 49152
	sub	rdx, r13
	jle	read_done
	lea	rsi, [rip + inbuf]
	add	rsi, r13
	xor	edi, edi		# fd 0
	xor	eax, eax		# read
	syscall
	test	rax, rax
	jle	read_done
	add	r13, rax
	jmp	read_loop
read_done:

	# ---- encode ----
	lea	rsi, [rip + inbuf]	# input pointer
	lea	rdi, [rip + outbuf]	# output pointer
	mov	rbx, r13		# remaining input bytes
	xor	r12, r12		# chars on current line

enc_loop:
	cmp	rbx, 3
	jb	enc_tail
	movzx	eax, byte ptr [rsi]
	shl	eax, 8
	mov	al, byte ptr [rsi + 1]
	shl	eax, 8
	mov	al, byte ptr [rsi + 2]
	mov	r8d, eax
	mov	ecx, r8d
	shr	ecx, 18
	call	emit6
	mov	ecx, r8d
	shr	ecx, 12
	and	ecx, 63
	call	emit6
	mov	ecx, r8d
	shr	ecx, 6
	and	ecx, 63
	call	emit6
	mov	ecx, r8d
	and	ecx, 63
	call	emit6
	add	rsi, 3
	sub	rbx, 3
	jmp	enc_loop

enc_tail:
	test	rbx, rbx
	jz	enc_done
	cmp	rbx, 1
	je	tail1
	# two bytes left
	movzx	eax, byte ptr [rsi]
	shl	eax, 8
	mov	al, byte ptr [rsi + 1]
	shl	eax, 8
	mov	r8d, eax
	mov	ecx, r8d
	shr	ecx, 18
	call	emit6
	mov	ecx, r8d
	shr	ecx, 12
	and	ecx, 63
	call	emit6
	mov	ecx, r8d
	shr	ecx, 6
	and	ecx, 63
	call	emit6
	mov	al, '='
	call	emit_byte
	jmp	enc_done
tail1:
	movzx	r8d, byte ptr [rsi]
	shl	r8d, 16
	mov	ecx, r8d
	shr	ecx, 18
	call	emit6
	mov	ecx, r8d
	shr	ecx, 12
	and	ecx, 63
	call	emit6
	mov	al, '='
	call	emit_byte
	mov	al, '='
	call	emit_byte

enc_done:
	# final newline if the last line is not empty
	test	r12, r12
	jz	do_write
	mov	byte ptr [rdi], 10
	inc	rdi

do_write:
	lea	rsi, [rip + outbuf]
	mov	rdx, rdi
	sub	rdx, rsi
write_loop:
	test	rdx, rdx
	jz	done
	mov	eax, 1			# write
	mov	edi, 1			# fd 1
	syscall
	test	rax, rax
	jle	done
	add	rsi, rax
	sub	rdx, rax
	jmp	write_loop

done:
	mov	eax, 60			# exit
	xor	edi, edi
	syscall

# ---- emit the base64 char for the 6-bit index in ecx ----
emit6:
	lea	rax, [rip + b64tab]
	mov	al, byte ptr [rax + rcx]
	# fall through / tail jump into emit_byte
	jmp	emit_byte

# ---- append the char in al to the output, wrapping lines at 76 ----
emit_byte:
	mov	byte ptr [rdi], al
	inc	rdi
	inc	r12
	cmp	r12, 76
	jne	1f
	mov	byte ptr [rdi], 10
	inc	rdi
	xor	r12, r12
1:	ret

.section .rodata
b64tab:
	.ascii	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
	.balign 16
inbuf:
	.space	49152
outbuf:
	.space	66560
