.text
.globl _start

_start:
	// x19 = input buffer, x24 = table, x25 = output buffer base
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	adrp	x24, b64tab
	add	x24, x24, :lo12:b64tab
	adrp	x25, outbuf
	add	x25, x25, :lo12:outbuf
	mov	x21, x25		// output pointer
	mov	x22, #0			// chars on current line
	mov	x20, #0			// total input bytes

	// read all input until read returns 0
read_loop:
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #49160
	sub	x2, x2, x20
	cbz	x2, read_done
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop
read_done:

	// encode
	mov	x23, #0			// input index
enc_loop:
	sub	x3, x20, x23
	cmp	x3, #3
	b.lt	enc_tail
	ldrb	w4, [x19, x23]
	add	x23, x23, #1
	ldrb	w5, [x19, x23]
	add	x23, x23, #1
	ldrb	w6, [x19, x23]
	add	x23, x23, #1
	lsl	w4, w4, #16
	orr	w4, w4, w5, lsl #8
	orr	w4, w4, w6
	ubfx	w0, w4, #18, #6
	bl	emiti
	ubfx	w0, w4, #12, #6
	bl	emiti
	ubfx	w0, w4, #6, #6
	bl	emiti
	and	w0, w4, #63
	bl	emiti
	b	enc_loop

enc_tail:
	cbz	x3, enc_done
	cmp	x3, #1
	b.eq	rem1
	// 2 bytes remain
	add	x6, x19, x23
	ldrb	w4, [x6]
	ldrb	w5, [x6, #1]
	lsl	w4, w4, #16
	orr	w4, w4, w5, lsl #8
	ubfx	w0, w4, #18, #6
	bl	emiti
	ubfx	w0, w4, #12, #6
	bl	emiti
	ubfx	w0, w4, #6, #6
	bl	emiti
	mov	w0, #61			// '='
	bl	emitc
	b	enc_done
rem1:
	// 1 byte remains
	ldrb	w4, [x19, x23]
	lsl	w4, w4, #16
	ubfx	w0, w4, #18, #6
	bl	emiti
	ubfx	w0, w4, #12, #6
	bl	emiti
	mov	w0, #61			// '='
	bl	emitc
	mov	w0, #61			// '='
	bl	emitc

enc_done:
	// final newline if current line is non-empty
	cbz	x22, write_out
	mov	w0, #10
	strb	w0, [x21], #1

write_out:
	sub	x26, x21, x25		// total output length
	mov	x27, #0			// bytes written
wloop:
	cmp	x27, x26
	b.ge	exit0
	mov	x0, #1
	add	x1, x25, x27
	sub	x2, x26, x27
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	b.le	exit0
	add	x27, x27, x0
	b	wloop

exit0:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// emiti: w0 = 6-bit value -> emit alphabet char
	// emitc: w0 = raw char to emit (counts toward line length)
emiti:
	ldrb	w0, [x24, w0, uxtw]
emitc:
	strb	w0, [x21], #1
	add	x22, x22, #1
	cmp	x22, #76
	b.ne	1f
	mov	w1, #10
	strb	w1, [x21], #1
	mov	x22, #0
1:	ret

.section .rodata
b64tab:
	.ascii	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
	.balign	8
inbuf:
	.space	49160
	.balign	8
outbuf:
	.space	70000
