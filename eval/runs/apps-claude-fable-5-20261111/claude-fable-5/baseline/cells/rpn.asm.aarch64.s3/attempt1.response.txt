	.section .rodata
errmsg:	.ascii "error\n"

	.bss
	.balign 8
inbuf:	.space 65544
stack:	.space 8192
outbuf:	.space 32

	.text
	.globl _start
_start:
	// read all input
	mov	x19, #0			// total bytes read
	adrp	x20, inbuf
	add	x20, x20, :lo12:inbuf
read_loop:
	mov	x2, #65536
	sub	x2, x2, x19
	cbz	x2, read_done
	mov	x8, #63			// read
	mov	x0, #0
	add	x1, x20, x19
	svc	#0
	cbz	x0, read_done
	tbnz	x0, #63, read_done
	add	x19, x19, x0
	b	read_loop
read_done:
	mov	x21, x20		// cur
	add	x22, x20, x19		// input end

line_loop:
	cmp	x21, x22
	b.ge	done
	mov	x23, x21		// line start
findnl:
	cmp	x21, x22
	b.ge	gotline
	ldrb	w0, [x21], #1
	cmp	w0, #10
	b.ne	findnl
	sub	x24, x21, #1		// line end (excl newline)
	b	evalline
gotline:
	mov	x24, x22		// final line without newline

evalline:
	mov	x25, #0			// stack count
tok_loop:
skipsep:
	cmp	x23, x24
	b.ge	line_end
	ldrb	w0, [x23]
	cmp	w0, #32
	b.eq	1f
	cmp	w0, #9
	b.ne	tok_start
1:	add	x23, x23, #1
	b	skipsep
tok_start:
	mov	x26, x23		// token start
scan:
	cmp	x23, x24
	b.ge	tok_end
	ldrb	w0, [x23]
	cmp	w0, #32
	b.eq	tok_end
	cmp	w0, #9
	b.eq	tok_end
	add	x23, x23, #1
	b	scan
tok_end:
	sub	x27, x23, x26		// token length
	ldrb	w0, [x26]
	cmp	x27, #1
	b.ne	try_num
	cmp	w0, #'+'
	b.eq	op_common
	cmp	w0, #'-'
	b.eq	op_common
	cmp	w0, #'*'
	b.eq	op_common
	cmp	w0, #'/'
	b.eq	op_common
	cmp	w0, #'%'
	b.eq	op_common
try_num:
	mov	x1, x26
	mov	w2, #0			// neg flag
	ldrb	w0, [x1]
	cmp	w0, #'-'
	b.ne	2f
	mov	w2, #1
	add	x1, x1, #1
2:
	sub	x3, x23, x1		// digit count
	cbz	x3, err
	cmp	x3, #18
	b.gt	err
	mov	x4, #0			// value
digloop:
	cmp	x1, x23
	b.ge	dig_done
	ldrb	w0, [x1], #1
	sub	w0, w0, #'0'
	cmp	w0, #9
	b.hi	err
	mov	x5, #10
	mul	x4, x4, x5
	add	x4, x4, x0
	b	digloop
dig_done:
	cbz	w2, push
	neg	x4, x4
push:
	adrp	x0, stack
	add	x0, x0, :lo12:stack
	str	x4, [x0, x25, lsl #3]
	add	x25, x25, #1
	b	tok_loop

op_common:				// w0 = operator char
	cmp	x25, #2
	b.lt	err
	sub	x25, x25, #2
	adrp	x1, stack
	add	x1, x1, :lo12:stack
	ldr	x4, [x1, x25, lsl #3]	// a
	add	x2, x25, #1
	ldr	x5, [x1, x2, lsl #3]	// b
	cmp	w0, #'+'
	b.eq	do_add
	cmp	w0, #'-'
	b.eq	do_sub
	cmp	w0, #'*'
	b.eq	do_mul
	// / or %
	cbz	x5, err
	cmn	x5, #1
	b.ne	3f
	mov	x6, #1
	lsl	x6, x6, #63		// INT64_MIN
	cmp	x4, x6
	b.eq	err
3:
	sdiv	x6, x4, x5
	cmp	w0, #'/'
	b.ne	do_mod
	mov	x4, x6
	b	push
do_mod:
	msub	x4, x6, x5, x4
	b	push
do_add:
	add	x4, x4, x5
	b	push
do_sub:
	sub	x4, x4, x5
	b	push
do_mul:
	mul	x4, x4, x5
	b	push

line_end:
	cmp	x25, #1
	b.ne	err
	adrp	x0, stack
	add	x0, x0, :lo12:stack
	ldr	x0, [x0]
	bl	print_num
	b	line_loop

err:
	mov	x0, #1
	adrp	x1, errmsg
	add	x1, x1, :lo12:errmsg
	mov	x2, #6
	mov	x8, #64
	svc	#0
	b	line_loop

done:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0
	b	done

// print_num: print signed x0 in decimal plus newline; clobbers x0-x8
print_num:
	adrp	x1, outbuf
	add	x1, x1, :lo12:outbuf
	add	x1, x1, #32
	mov	w2, #10
	strb	w2, [x1, #-1]!		// newline
	mov	w6, #0			// neg flag
	tbz	x0, #63, 1f
	mov	w6, #1
	neg	x0, x0
1:
	mov	x4, #10
2:
	udiv	x5, x0, x4
	msub	x3, x5, x4, x0
	add	w3, w3, #'0'
	strb	w3, [x1, #-1]!
	mov	x0, x5
	cbnz	x0, 2b
	cbz	w6, 3f
	mov	w3, #'-'
	strb	w3, [x1, #-1]!
3:
	adrp	x2, outbuf
	add	x2, x2, :lo12:outbuf
	add	x2, x2, #32
	sub	x2, x2, x1		// length
	mov	x0, #1
	mov	x8, #64			// write
	svc	#0
	ret
