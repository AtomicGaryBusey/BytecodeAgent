.text
.globl _start

// ---------------------------------------------------------------
// sort_lines: read stdin, sort lines bytewise ascending, print.
// ---------------------------------------------------------------

_start:
	adrp	x19, buf
	add	x19, x19, :lo12:buf	// x19 = input buffer
	adrp	x21, lines
	add	x21, x21, :lo12:lines	// x21 = line table (ptr,len pairs)
	mov	x20, #0			// x20 = total bytes read

// ---- read all of stdin -----------------------------------------
read_loop:
	mov	x2, #32768
	sub	x2, x2, x20		// remaining space
	cbz	x2, read_done
	mov	x0, #0			// fd 0
	add	x1, x19, x20
	mov	x8, #63			// read
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop
read_done:

// ---- split into lines ------------------------------------------
	mov	x22, #0			// x22 = number of lines
	mov	x23, #0			// x23 = current line start
	mov	x24, #0			// x24 = index
parse_loop:
	cmp	x24, x20
	b.ge	parse_end
	ldrb	w9, [x19, x24]
	cmp	w9, #10
	b.ne	1f
	add	x10, x19, x23		// line pointer
	sub	x11, x24, x23		// line length
	add	x12, x21, x22, lsl #4
	stp	x10, x11, [x12]
	add	x22, x22, #1
	add	x23, x24, #1
1:
	add	x24, x24, #1
	b	parse_loop
parse_end:
	cmp	x23, x20		// trailing line without newline?
	b.ge	sort_start
	add	x10, x19, x23
	sub	x11, x20, x23
	add	x12, x21, x22, lsl #4
	stp	x10, x11, [x12]
	add	x22, x22, #1

// ---- insertion sort of the line table ---------------------------
sort_start:
	mov	x24, #1			// i
sort_outer:
	cmp	x24, x22
	b.ge	sort_done
	add	x9, x21, x24, lsl #4
	ldp	x25, x26, [x9]		// key = (ptr, len)
	sub	x27, x24, #1		// j
sort_inner:
	tbnz	x27, #63, sort_place	// j < 0?
	add	x9, x21, x27, lsl #4
	ldp	x0, x1, [x9]
	mov	x2, x25
	mov	x3, x26
	bl	cmpfn
	cmp	x0, #0
	b.le	sort_place
	add	x9, x21, x27, lsl #4	// lines[j+1] = lines[j]
	ldp	x4, x5, [x9]
	stp	x4, x5, [x9, #16]
	sub	x27, x27, #1
	b	sort_inner
sort_place:
	add	x9, x21, x27, lsl #4
	stp	x25, x26, [x9, #16]	// lines[j+1] = key
	add	x24, x24, #1
	b	sort_outer
sort_done:

// ---- build the output buffer ------------------------------------
	adrp	x26, outbuf
	add	x26, x26, :lo12:outbuf
	mov	x24, #0			// line index
	mov	x25, #0			// output position
out_loop:
	cmp	x24, x22
	b.ge	out_done
	add	x9, x21, x24, lsl #4
	ldp	x4, x5, [x9]		// ptr, len
copy_loop:
	cbz	x5, add_nl
	ldrb	w7, [x4], #1
	strb	w7, [x26, x25]
	add	x25, x25, #1
	sub	x5, x5, #1
	b	copy_loop
add_nl:
	mov	w7, #10
	strb	w7, [x26, x25]
	add	x25, x25, #1
	add	x24, x24, #1
	b	out_loop
out_done:

// ---- write everything -------------------------------------------
	mov	x20, #0			// bytes written
write_loop:
	subs	x2, x25, x20
	b.le	do_exit
	mov	x0, #1			// fd 1
	add	x1, x26, x20
	mov	x8, #64			// write
	svc	#0
	cmp	x0, #0
	b.le	do_exit
	add	x20, x20, x0
	b	write_loop

do_exit:
	mov	x0, #0
	mov	x8, #93			// exit
	svc	#0

// ---- memcmp-style line compare ----------------------------------
// in:  x0 = p1, x1 = l1, x2 = p2, x3 = l2
// out: x0 < 0, == 0, > 0
cmpfn:
	cmp	x1, x3
	csel	x4, x1, x3, lo		// min length
	mov	x5, #0
1:
	cmp	x5, x4
	b.eq	2f
	ldrb	w6, [x0, x5]
	ldrb	w7, [x2, x5]
	cmp	w6, w7
	b.ne	3f
	add	x5, x5, #1
	b	1b
2:
	sub	x0, x1, x3		// shorter line first
	ret
3:
	sub	x0, x6, x7
	ret

// -----------------------------------------------------------------
.bss
.balign 16
buf:
	.space	32768			// input
lines:
	.space	16000			// 1000 x (ptr, len)
outbuf:
	.space	33776			// output (input + added newline)
