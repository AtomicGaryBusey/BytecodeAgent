.text
.globl _start
_start:
    xor ebx, ebx            # lines
    xor r12, r12            # words
    xor r13, r13            # bytes
    xor ebp, ebp            # in-word flag

read_loop:
    xor eax, eax            # read
    xor edi, edi            # fd 0
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r13, rax
    mov rdx, rax            # count
    lea rsi, [rip + buffer]
    xor ecx, ecx            # index
scan:
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x0a
    jne not_nl
    inc rbx
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0d
    jbe is_ws
is_word:
    test ebp, ebp
    jnz next
    mov ebp, 1
    inc r12
    jmp next
is_ws:
    xor ebp, ebp
next:
    inc rcx
    cmp rcx, rdx
    jb scan
    jmp read_loop

done:
    # build output string backwards in outbuf
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 0x0a
    mov rax, r13
    call put_num
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call put_num
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, rbx
    call put_num
    # write
    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

# put_num: rax = value, rdi points just after where digits go; writes digits backwards, rdi updated
put_num:
    mov r8, 10
1:
    xor edx, edx
    div r8
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.balign 16
buffer:
    .space 65536
outbuf:
    .space 64
