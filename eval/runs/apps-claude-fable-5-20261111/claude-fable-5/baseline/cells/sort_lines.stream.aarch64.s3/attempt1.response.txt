.text
.globl _start
_start:
	// load buffer addresses
	adrp	x20, inbuf
	add	x20, x20, :lo12:inbuf
	adrp	x21, lines
	add	x21, x21, :lo12:lines
	adrp	x27, outbuf
	add	x27, x27, :lo12:outbuf

	// ---- read all of stdin ----
	mov	x19, #0			// total bytes read
read_loop:
	mov	x2, #32768
	sub	x2, x2, x19		// remaining space
	cbz	x2, read_done
	mov	x8, #63			// read
	mov	x0, #0
	add	x1, x20, x19
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x19, x19, x0
	b	read_loop
read_done:

	// ---- split into lines ----
	mov	x22, #0			// line count
	mov	x4, x20			// current line start
	mov	x5, #0			// index
parse_loop:
	cmp	x5, x19
	b.ge	parse_end
	ldrb	w6, [x20, x5]
	add	x5, x5, #1
	cmp	w6, #10
	b.ne	parse_loop
	// record line [x4, len)
	add	x7, x20, x5
	sub	x7, x7, #1
	sub	x7, x7, x4		// length (excl newline)
	add	x9, x21, x22, lsl #4
	stp	x4, x7, [x9]
	add	x22, x22, #1
	add	x4, x20, x5		// next line start
	b	parse_loop
parse_end:
	add	x6, x20, x19
	cmp	x4, x6
	b.ge	sort_start		// no trailing partial line
	sub	x7, x6, x4
	add	x9, x21, x22, lsl #4
	stp	x4, x7, [x9]
	add	x22, x22, #1

	// ---- insertion sort ----
sort_start:
	mov	x23, #1			// i
sort_outer:
	cmp	x23, x22
	b.ge	sort_done
	add	x9, x21, x23, lsl #4
	ldp	x24, x25, [x9]		// key (ptr, len)
	sub	x26, x23, #1		// j
sort_inner:
	tbnz	x26, #63, sort_insert	// j < 0
	add	x9, x21, x26, lsl #4
	ldp	x0, x1, [x9]
	mov	x2, x24
	mov	x3, x25
	bl	cmp_lines
	cmp	w0, #0
	b.le	sort_insert
	// shift lines[j] -> lines[j+1]
	add	x9, x21, x26, lsl #4
	ldp	x0, x1, [x9]
	stp	x0, x1, [x9, #16]
	sub	x26, x26, #1
	b	sort_inner
sort_insert:
	add	x9, x21, x26, lsl #4
	stp	x24, x25, [x9, #16]
	add	x23, x23, #1
	b	sort_outer
sort_done:

	// ---- build output ----
	mov	x28, #0			// output length
	mov	x23, #0			// line index
out_loop:
	cmp	x23, x22
	b.ge	out_write
	add	x9, x21, x23, lsl #4
	ldp	x0, x1, [x9]		// ptr, len
	mov	x5, #0
copy_loop:
	cmp	x5, x1
	b.ge	copy_done
	ldrb	w6, [x0, x5]
	add	x5, x5, #1
	strb	w6, [x27, x28]
	add	x28, x28, #1
	b	copy_loop
copy_done:
	mov	w6, #10
	strb	w6, [x27, x28]
	add	x28, x28, #1
	add	x23, x23, #1
	b	out_loop

	// ---- write output ----
out_write:
	mov	x24, #0			// bytes written
write_loop:
	cmp	x24, x28
	b.ge	finish
	mov	x8, #64			// write
	mov	x0, #1
	add	x1, x27, x24
	sub	x2, x28, x24
	svc	#0
	cmp	x0, #0
	b.le	finish
	add	x24, x24, x0
	b	write_loop

finish:
	mov	x8, #93			// exit
	mov	x0, #0
	svc	#0

// cmp_lines: compare (x0=ptr1, x1=len1) with (x2=ptr2, x3=len2)
// returns w0: <0, 0, >0  (memcmp order, shorter prefix first)
cmp_lines:
	cmp	x1, x3
	csel	x4, x1, x3, lo		// min length
	mov	x5, #0
1:
	cmp	x5, x4
	b.ge	2f
	ldrb	w6, [x0, x5]
	ldrb	w7, [x2, x5]
	add	x5, x5, #1
	cmp	w6, w7
	b.eq	1b
	b.lo	3f
	mov	w0, #1
	ret
3:
	mov	w0, #-1
	ret
2:
	cmp	x1, x3
	b.eq	4f
	b.lo	3b
	mov	w0, #1
	ret
4:
	mov	w0, #0
	ret

.bss
.balign 8
inbuf:
	.space	32769
.balign 8
lines:
	.space	16000			// 1000 entries * 16 bytes (ptr, len)
.balign 8
outbuf:
	.space	33800			// max 32768 bytes + 1000 newlines
