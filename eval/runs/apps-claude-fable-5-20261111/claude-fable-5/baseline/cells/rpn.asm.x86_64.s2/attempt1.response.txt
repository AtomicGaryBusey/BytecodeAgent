.text
.globl _start

_start:
        # read all input
        xor     ebx, ebx                # total bytes read
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jz      read_done
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        xor     edi, edi
        xor     eax, eax
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     r12, [rip + inbuf]      # current position
        lea     r13, [r12 + rbx]        # end of input

main_loop:
        cmp     r12, r13
        jae     all_done
        # find end of line
        mov     r14, r12
1:
        cmp     r14, r13
        jae     2f
        cmp     byte ptr [r14], 10
        je      2f
        inc     r14
        jmp     1b
2:
        call    process_line
        mov     r12, r14
        cmp     r12, r13
        jae     main_loop
        inc     r12                     # skip the newline
        jmp     main_loop

all_done:
        mov     eax, 60
        xor     edi, edi
        syscall

# ---------------------------------------------------------------
# process_line: line is [r12, r14).  Preserves r12, r13, r14.
# ---------------------------------------------------------------
process_line:
        xor     r15d, r15d              # value stack count
        mov     rsi, r12
tok_loop:
1:      # skip separators
        cmp     rsi, r14
        jae     eol_check
        movzx   eax, byte ptr [rsi]
        cmp     al, 32
        je      2f
        cmp     al, 9
        jne     3f
2:
        inc     rsi
        jmp     1b
3:      # token start
        mov     rdi, rsi
4:
        cmp     rsi, r14
        jae     5f
        movzx   eax, byte ptr [rsi]
        cmp     al, 32
        je      5f
        cmp     al, 9
        je      5f
        inc     rsi
        jmp     4b
5:
        mov     rcx, rsi
        sub     rcx, rdi                # token length
        cmp     rcx, 1
        jne     literal
        movzx   eax, byte ptr [rdi]
        cmp     al, '+'
        je      op_add
        cmp     al, '-'
        je      op_sub
        cmp     al, '*'
        je      op_mul
        cmp     al, '/'
        je      op_div
        cmp     al, '%'
        je      op_mod
        # single char, not an operator: try literal

literal:
        xor     r9d, r9d                # sign flag
        cmp     byte ptr [rdi], '-'
        jne     6f
        inc     rdi
        dec     rcx
        mov     r9d, 1
6:
        test    rcx, rcx
        jz      line_error
        cmp     rcx, 18
        ja      line_error
        xor     eax, eax
7:
        movzx   edx, byte ptr [rdi]
        sub     edx, '0'
        cmp     edx, 9
        ja      line_error
        imul    rax, rax, 10
        add     rax, rdx
        inc     rdi
        dec     rcx
        jnz     7b
        test    r9d, r9d
        jz      8f
        neg     rax
8:      # push
        cmp     r15, 1024
        jae     line_error
        lea     r8, [rip + vstack]
        mov     qword ptr [r8 + r15*8], rax
        inc     r15
        jmp     tok_loop

op_add:
        cmp     r15, 2
        jb      line_error
        lea     r8, [rip + vstack]
        mov     rbx, qword ptr [r8 + r15*8 - 8]
        dec     r15
        mov     rax, qword ptr [r8 + r15*8 - 8]
        add     rax, rbx
        mov     qword ptr [r8 + r15*8 - 8], rax
        jmp     tok_loop

op_sub:
        cmp     r15, 2
        jb      line_error
        lea     r8, [rip + vstack]
        mov     rbx, qword ptr [r8 + r15*8 - 8]
        dec     r15
        mov     rax, qword ptr [r8 + r15*8 - 8]
        sub     rax, rbx
        mov     qword ptr [r8 + r15*8 - 8], rax
        jmp     tok_loop

op_mul:
        cmp     r15, 2
        jb      line_error
        lea     r8, [rip + vstack]
        mov     rbx, qword ptr [r8 + r15*8 - 8]
        dec     r15
        mov     rax, qword ptr [r8 + r15*8 - 8]
        imul    rax, rbx
        mov     qword ptr [r8 + r15*8 - 8], rax
        jmp     tok_loop

op_div:
        cmp     r15, 2
        jb      line_error
        lea     r8, [rip + vstack]
        mov     rbx, qword ptr [r8 + r15*8 - 8]
        dec     r15
        mov     rax, qword ptr [r8 + r15*8 - 8]
        test    rbx, rbx
        jz      line_error
        cmp     rbx, -1
        jne     1f
        movabs  rdx, 0x8000000000000000
        cmp     rax, rdx
        je      line_error
1:
        cqo
        idiv    rbx
        mov     qword ptr [r8 + r15*8 - 8], rax
        jmp     tok_loop

op_mod:
        cmp     r15, 2
        jb      line_error
        lea     r8, [rip + vstack]
        mov     rbx, qword ptr [r8 + r15*8 - 8]
        dec     r15
        mov     rax, qword ptr [r8 + r15*8 - 8]
        test    rbx, rbx
        jz      line_error
        cmp     rbx, -1
        jne     1f
        movabs  rdx, 0x8000000000000000
        cmp     rax, rdx
        je      line_error
1:
        cqo
        idiv    rbx
        mov     qword ptr [r8 + r15*8 - 8], rdx
        jmp     tok_loop

eol_check:
        cmp     r15, 1
        jne     line_error
        lea     r8, [rip + vstack]
        mov     rax, qword ptr [r8]
        call    print_num
        ret

line_error:
        lea     rsi, [rip + errmsg]
        mov     edx, 6
        call    do_write
        ret

# ---------------------------------------------------------------
# print_num: print signed value in rax, followed by newline
# ---------------------------------------------------------------
print_num:
        lea     rdi, [rip + numbuf + 31]
        mov     byte ptr [rdi], 10
        xor     r9d, r9d
        test    rax, rax
        jns     1f
        neg     rax
        mov     r9d, 1
1:
        mov     rbx, 10
2:
        xor     edx, edx
        div     rbx
        add     dl, '0'
        dec     rdi
        mov     byte ptr [rdi], dl
        test    rax, rax
        jnz     2b
        test    r9d, r9d
        jz      3f
        dec     rdi
        mov     byte ptr [rdi], '-'
3:
        lea     rdx, [rip + numbuf + 32]
        sub     rdx, rdi
        mov     rsi, rdi
        call    do_write
        ret

# do_write: write(1, rsi, rdx)
do_write:
        mov     eax, 1
        mov     edi, 1
        syscall
        ret

.section .rodata
errmsg:
        .ascii  "error\n"

.bss
.balign 8
vstack:
        .space  8192
numbuf:
        .space  32
inbuf:
        .space  65536
