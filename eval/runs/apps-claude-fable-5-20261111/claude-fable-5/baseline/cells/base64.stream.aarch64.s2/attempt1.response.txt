.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read

.Lrd:
    mov     x8, #63                 // read
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #49152
    sub     x2, x2, x20
    svc     #0
    cmp     x0, #0
    b.le    .Lrddone
    add     x20, x20, x0
    b       .Lrd

.Lrddone:
    adrp    x23, alpha
    add     x23, x23, :lo12:alpha
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x24, x21                // start of output
    mov     w22, #0                 // chars on current line

.Lloop:
    cbz     x20, .Ldone
    cmp     x20, #3
    b.lt    .Lpartial
    ldrb    w0, [x19], #1
    ldrb    w2, [x19], #1
    ldrb    w3, [x19], #1
    sub     x20, x20, #3
    orr     w0, w3, w0, lsl #16
    orr     w0, w0, w2, lsl #8
    lsr     w1, w0, #18
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    ubfx    w1, w0, #12, #6
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    ubfx    w1, w0, #6, #6
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    and     w1, w0, #63
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    b       .Lloop

.Lpartial:
    cmp     x20, #2
    b.eq    .Ltwo
    // one remaining byte
    ldrb    w0, [x19]
    lsr     w1, w0, #2
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    lsl     w1, w0, #4
    and     w1, w1, #63
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    mov     w1, #'='
    bl      emit
    mov     w1, #'='
    bl      emit
    b       .Ldone

.Ltwo:
    ldrb    w0, [x19]
    ldrb    w2, [x19, #1]
    orr     w0, w2, w0, lsl #8
    lsr     w1, w0, #10
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    ubfx    w1, w0, #4, #6
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    lsl     w1, w0, #2
    and     w1, w1, #63
    ldrb    w1, [x23, w1, uxtw]
    bl      emit
    mov     w1, #'='
    bl      emit

.Ldone:
    cbz     w22, .Lwrite            // empty or line just ended with \n
    mov     w1, #10
    strb    w1, [x21], #1

.Lwrite:
    sub     x2, x21, x24
    cbz     x2, .Lexit
    mov     x25, x24
.Lwr:
    mov     x8, #64                 // write
    mov     x0, #1
    mov     x1, x25
    svc     #0
    cmp     x0, #0
    b.le    .Lexit
    add     x25, x25, x0
    sub     x2, x2, x0
    cbnz    x2, .Lwr

.Lexit:
    mov     x8, #94                 // exit_group
    mov     x0, #0
    svc     #0

// emit one encoded character in w1, breaking lines at 76 chars
emit:
    strb    w1, [x21], #1
    add     w22, w22, #1
    cmp     w22, #76
    b.ne    1f
    mov     w1, #10
    strb    w1, [x21], #1
    mov     w22, #0
1:  ret

.section .rodata

.bss
.balign 8

    .space  49152
