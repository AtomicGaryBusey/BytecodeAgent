.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 16
inbuf:
    .space 65536
linebuf:
    .space 96

.text
.globl _start

_start:
    mov x19, xzr
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf

read_more:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, have_input
    mov x0, xzr
    add x1, x20, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le have_input
    add x19, x19, x0
    b read_more

have_input:
    mov x21, xzr
    adrp x23, linebuf
    add x23, x23, :lo12:linebuf

next_row:
    cmp x21, x19
    b.hs print_total
    sub x22, x19, x21
    cmp x22, #16
    b.ls row_len
    mov x22, #16
row_len:
    mov x24, x23
    mov x0, x21
    mov x1, x24
    bl put_hex32
    add x24, x24, #8

    mov w0, #32
    strb w0, [x24], #1
    strb w0, [x24], #1

    mov x25, xzr
hex_loop:
    cmp x25, #16
    b.eq hex_done
    cmp x25, x22
    b.hs hex_empty

    add x0, x20, x21
    ldrb w1, [x0, x25]

    lsr w2, w1, #4
    cmp w2, #10
    b.lo nib_hi_dec
    add w2, w2, #87
    b nib_hi_out
nib_hi_dec:
    add w2, w2, #48
nib_hi_out:
    strb w2, [x24], #1

    and w2, w1, #15
    cmp w2, #10
    b.lo nib_lo_dec
    add w2, w2, #87
    b nib_lo_out
nib_lo_dec:
    add w2, w2, #48
nib_lo_out:
    strb w2, [x24], #1

    mov w2, #32
    strb w2, [x24], #1
    b hex_after

hex_empty:
    mov w2, #32
    strb w2, [x24], #1
    strb w2, [x24], #1
    strb w2, [x24], #1

hex_after:
    cmp x25, #7
    b.ne hex_inc
    mov w2, #32
    strb w2, [x24], #1
hex_inc:
    add x25, x25, #1
    b hex_loop

hex_done:
    mov w2, #32
    strb w2, [x24], #1
    mov w2, #124
    strb w2, [x24], #1

    mov x25, xzr
asc_loop:
    cmp x25, x22
    b.eq asc_done
    add x0, x20, x21
    ldrb w1, [x0, x25]
    cmp w1, #32
    b.lo asc_dot
    cmp w1, #126
    b.hi asc_dot
    b asc_store
asc_dot:
    mov w1, #46
asc_store:
    strb w1, [x24], #1
    add x25, x25, #1
    b asc_loop

asc_done:
    mov w2, #124
    strb w2, [x24], #1
    mov w2, #10
    strb w2, [x24], #1

    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #SYS_write
    svc #0

    add x21, x21, x22
    b next_row

print_total:
    mov x0, x19
    mov x1, x23
    bl put_hex32
    mov w2, #10
    strb w2, [x23, #8]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #SYS_write
    svc #0

    mov x0, xzr
    mov x8, #SYS_exit
    svc #0

// x0 = value, x1 = destination. Writes 8 lowercase hex digits. Clobbers x2, x4, x5.
put_hex32:
    mov x4, #28
    mov x5, xzr
put_hex_loop:
    lsr x2, x0, x4
    and w2, w2, #15
    cmp w2, #10
    b.lo put_hex_dec
    add w2, w2, #87
    b put_hex_store
put_hex_dec:
    add w2, w2, #48
put_hex_store:
    strb w2, [x1, x5]
    add x5, x5, #1
    cmp x5, #8
    b.eq put_hex_done
    sub x4, x4, #4
    b put_hex_loop
put_hex_done:
    ret
