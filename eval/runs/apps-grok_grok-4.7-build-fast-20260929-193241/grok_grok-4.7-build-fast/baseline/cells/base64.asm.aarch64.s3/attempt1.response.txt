.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 49152
.equ WRAP, 76
.equ NL, 10
.equ PAD, 61

.section .rodata
.balign 16
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 131072

.text
.balign 4
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, xzr

read_loop:
    movz x2, #IN_MAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    mov x0, xzr
    add x1, x19, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_ok

    adrp x21, b64tab
    add x21, x21, :lo12:b64tab
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x24, x22
    mov x23, xzr

encode_loop:
    cmp x20, #3
    b.lo encode_tail
    ldrb w0, [x19]
    ldrb w1, [x19, #1]
    ldrb w2, [x19, #2]
    add x19, x19, #3
    sub x20, x20, #3

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    and w5, w1, #0xf
    lsl w5, w5, #2
    lsr w6, w2, #6
    orr w5, w5, w6
    and w6, w2, #0x3f

    ldrb w0, [x21, x3]
    bl emit
    ldrb w0, [x21, x4]
    bl emit
    ldrb w0, [x21, x5]
    bl emit
    ldrb w0, [x21, x6]
    bl emit
    b encode_loop

encode_tail:
    cbz x20, encode_finish
    cmp x20, #1
    b.eq tail_one

    ldrb w0, [x19]
    ldrb w1, [x19, #1]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    and w5, w1, #0xf
    lsl w5, w5, #2

    ldrb w0, [x21, x3]
    bl emit
    ldrb w0, [x21, x4]
    bl emit
    ldrb w0, [x21, x5]
    bl emit
    mov w0, #PAD
    bl emit
    b encode_finish

tail_one:
    ldrb w0, [x19]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    ldrb w0, [x21, x3]
    bl emit
    ldrb w0, [x21, x4]
    bl emit
    mov w0, #PAD
    bl emit
    mov w0, #PAD
    bl emit

encode_finish:
    cbz x23, write_out
    mov w0, #NL
    strb w0, [x22], #1

write_out:
    sub x20, x22, x24
    mov x21, x24

write_loop:
    cbz x20, exit_ok
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x21
    mov x2, x20
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x21, x21, x0
    sub x20, x20, x0
    b write_loop

exit_ok:
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0
    b exit_ok

// w0 = output byte. Advances x22. Inserts NL every WRAP encoded bytes.
emit:
    strb w0, [x22], #1
    add x23, x23, #1
    cmp x23, #WRAP
    b.ne emit_ret
    mov w0, #NL
    strb w0, [x22], #1
    mov x23, xzr
emit_ret:
    ret
