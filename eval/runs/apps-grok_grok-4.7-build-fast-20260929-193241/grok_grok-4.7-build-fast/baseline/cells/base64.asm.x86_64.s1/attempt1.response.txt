.globl _start

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
inbuf:
    .zero 49152
outbuf:
    .zero 66560

.text
emit:
    mov byte ptr [outbuf + r13], al
    inc r13
    inc r14
    cmp r14, 76
    jne emit_done
    mov byte ptr [outbuf + r13], 10
    inc r13
    xor r14d, r14d
emit_done:
    ret

_start:
    xor rbx, rbx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 49152
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, 49152
    jb read_loop
read_done:
    test rbx, rbx
    jz quit
    xor r12, r12
    xor r13, r13
    xor r14, r14
encode_loop:
    cmp r12, rbx
    jae encode_done
    xor r10d, r10d
    xor r11d, r11d
    mov r8d, 1
    movzx r9d, byte ptr [inbuf + r12]
    inc r12
    cmp r12, rbx
    jae encode_triplet
    movzx r10d, byte ptr [inbuf + r12]
    inc r12
    mov r8d, 2
    cmp r12, rbx
    jae encode_triplet
    movzx r11d, byte ptr [inbuf + r12]
    inc r12
    mov r8d, 3
encode_triplet:
    mov esi, r9d
    shr esi, 2
    mov al, byte ptr [alpha + rsi]
    call emit
    mov esi, r9d
    and esi, 3
    shl esi, 4
    mov edi, r10d
    shr edi, 4
    or esi, edi
    mov al, byte ptr [alpha + rsi]
    call emit
    cmp r8d, 1
    je encode_pad2
    mov esi, r10d
    and esi, 15
    shl esi, 2
    mov edi, r11d
    shr edi, 6
    or esi, edi
    mov al, byte ptr [alpha + rsi]
    call emit
    cmp r8d, 2
    je encode_pad1
    mov esi, r11d
    and esi, 63
    mov al, byte ptr [alpha + rsi]
    call emit
    jmp encode_loop
encode_pad2:
    mov al, 61
    call emit
    mov al, 61
    call emit
    jmp encode_loop
encode_pad1:
    mov al, 61
    call emit
    jmp encode_loop
encode_done:
    test r14, r14
    jz write_out
    mov byte ptr [outbuf + r13], 10
    inc r13
write_out:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall
quit:
    mov eax, 60
    xor edi, edi
    syscall
