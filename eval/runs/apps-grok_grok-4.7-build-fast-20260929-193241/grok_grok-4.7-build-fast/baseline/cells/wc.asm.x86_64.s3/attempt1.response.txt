.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_MAX, 65536

.globl _start

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 48

.text
_start:
    xor ebx, ebx
read_more:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, IN_MAX
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, IN_MAX
    jb read_more
read_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
count_loop:
    cmp rcx, rbx
    jae count_done
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x20
    je do_ws
    cmp al, 0x0A
    je do_nl
    cmp al, 0x09
    jb do_word
    cmp al, 0x0D
    jbe do_ws
do_word:
    test r14, r14
    jnz count_loop
    inc r13
    mov r14, 1
    jmp count_loop
do_nl:
    inc r12
do_ws:
    xor r14d, r14d
    jmp count_loop
count_done:
    lea rsi, [rip + outbuf]
    mov rdi, rsi
    mov rax, r12
    call format_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call format_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, rbx
    call format_num
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov rdx, rdi
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall
    mov eax, SYS_exit
    xor edi, edi
    syscall

format_num:
    test rax, rax
    jnz fn_nonzero
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
fn_nonzero:
    mov r8, rdi
    mov r9, 10
    xor r10d, r10d
fn_div:
    xor edx, edx
    div r9
    add dl, 0x30
    push rdx
    inc r10
    test rax, rax
    jnz fn_div
fn_pop:
    pop rdx
    mov byte ptr [r8], dl
    inc r8
    dec r10
    jnz fn_pop
    mov rdi, r8
    ret
