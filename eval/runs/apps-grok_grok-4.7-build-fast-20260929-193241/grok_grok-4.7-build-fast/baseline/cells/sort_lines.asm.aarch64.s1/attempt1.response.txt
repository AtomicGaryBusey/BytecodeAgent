.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.globl _start

.text

_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0

read_loop:
    mov x2, #32768
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x24, line_off
    add x24, x24, :lo12:line_off
    adrp x25, line_len
    add x25, x25, :lo12:line_len
    mov x21, #0
    mov x22, x19
    add x9, x19, x20
    mov x23, x19

parse_loop:
    cmp x22, x9
    b.hs parse_after
    ldrb w0, [x22], #1
    cmp w0, #0x0A
    b.ne parse_loop
    sub x1, x22, x23
    sub x1, x1, #1
    sub x2, x23, x19
    str x2, [x24, x21, lsl #3]
    str x1, [x25, x21, lsl #3]
    add x21, x21, #1
    mov x23, x22
    b parse_loop

parse_after:
    cmp x23, x9
    b.hs sort_begin
    sub x1, x9, x23
    sub x2, x23, x19
    str x2, [x24, x21, lsl #3]
    str x1, [x25, x21, lsl #3]
    add x21, x21, #1

sort_begin:
    cmp x21, #2
    b.lt print_begin
    mov x26, x21
    lsr x26, x26, #1

build_loop:
    cbz x26, heap_built
    sub x26, x26, #1
    mov x0, x26
    mov x1, x21
    bl siftdown
    b build_loop

heap_built:
    mov x27, x21

extract_loop:
    cmp x27, #1
    b.le print_begin
    sub x27, x27, #1
    ldr x0, [x24]
    ldr x1, [x25]
    ldr x2, [x24, x27, lsl #3]
    ldr x3, [x25, x27, lsl #3]
    str x2, [x24]
    str x3, [x25]
    str x0, [x24, x27, lsl #3]
    str x1, [x25, x27, lsl #3]
    mov x0, #0
    mov x1, x27
    bl siftdown
    b extract_loop

print_begin:
    mov x26, #0

print_loop:
    cmp x26, x21
    b.hs do_exit
    ldr x27, [x24, x26, lsl #3]
    ldr x28, [x25, x26, lsl #3]
    cbz x28, write_nl
    add x1, x19, x27
    mov x2, x28
    mov x0, #1
    mov x8, #SYS_write
    svc #0

write_nl:
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x26, x26, #1
    b print_loop

do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// compare line a (x0 off, x1 len) with line b (x2 off, x3 len)
// x0 = 1 if a > b, -1 if a < b, 0 if equal; unsigned bytes, shorter prefix first
compare:
    add x6, x19, x0
    add x7, x19, x2
    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, #0

compare_loop:
    cmp x5, x4
    b.hs compare_lens
    ldrb w0, [x6], #1
    ldrb w2, [x7], #1
    cmp w0, w2
    b.hi compare_gt
    b.lo compare_lt
    add x5, x5, #1
    b compare_loop

compare_lens:
    cmp x1, x3
    b.hi compare_gt
    b.lo compare_lt
    mov x0, #0
    ret

compare_gt:
    mov x0, #1
    ret

compare_lt:
    mov x0, #-1
    ret

// siftdown: x0 = root index, x1 = heap size (max-heap on line order)
siftdown:
    stp x29, x30, [sp, #-16]!
    mov x20, x0
    mov x22, x1

sift_loop:
    lsl x2, x20, #1
    add x2, x2, #1
    cmp x2, x22
    b.ge sift_done
    mov x23, x2
    add x28, x2, #1
    cmp x28, x22
    b.ge sift_cmp_root
    ldr x0, [x24, x23, lsl #3]
    ldr x1, [x25, x23, lsl #3]
    ldr x2, [x24, x28, lsl #3]
    ldr x3, [x25, x28, lsl #3]
    bl compare
    cmp x0, #0
    b.ge sift_cmp_root
    mov x23, x28

sift_cmp_root:
    ldr x0, [x24, x20, lsl #3]
    ldr x1, [x25, x20, lsl #3]
    ldr x2, [x24, x23, lsl #3]
    ldr x3, [x25, x23, lsl #3]
    bl compare
    cmp x0, #0
    b.ge sift_done
    ldr x0, [x24, x20, lsl #3]
    ldr x1, [x25, x20, lsl #3]
    ldr x2, [x24, x23, lsl #3]
    ldr x3, [x25, x23, lsl #3]
    str x2, [x24, x20, lsl #3]
    str x3, [x25, x20, lsl #3]
    str x0, [x24, x23, lsl #3]
    str x1, [x25, x23, lsl #3]
    mov x20, x23
    b sift_loop

sift_done:
    ldp x29, x30, [sp], #16
    ret

.section .rodata
nl:
    .byte 0x0A

.bss
.align 3
buf:
    .zero 32768
line_off:
    .zero 8000
line_len:
    .zero 8000
