.globl _start

.bss
.align 8
input_buf:
    .zero 65536
.align 8
tape:
    .zero 30000
.align 8
jump_tab:
    .zero 262144
.align 8
brack_stk:
    .zero 8000
.align 8
dispatch:
    .zero 2048

.text
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    xor ecx, ecx
    test r12, r12
    jz no_bang
find_loop:
    cmp byte ptr [rbx + rcx], 33
    je got_bang
    inc rcx
    cmp rcx, r12
    jb find_loop
no_bang:
    mov r13, r12
    lea r14, [rbx + r12]
    mov r15, r14
    jmp do_filter
got_bang:
    mov r13, rcx
    lea r14, [rbx + rcx + 1]
    lea r15, [rbx + r12]
do_filter:
    lea rsi, [rip + brack_stk]
    lea r8, [rip + jump_tab]
    xor edi, edi
    xor ecx, ecx
    xor edx, edx
filter_loop:
    cmp rcx, r13
    jae filter_done
    movzx eax, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 62
    je keep
    cmp al, 60
    je keep
    cmp al, 43
    je keep
    cmp al, 45
    je keep
    cmp al, 46
    je keep
    cmp al, 44
    je keep
    cmp al, 91
    je keep_lbr
    cmp al, 93
    je keep_rbr
    jmp filter_loop
keep:
    mov byte ptr [rbx + rdx], al
    inc rdx
    jmp filter_loop
keep_lbr:
    cmp rdi, 1000
    jae bad
    mov qword ptr [rsi + rdi*8], rdx
    inc rdi
    mov byte ptr [rbx + rdx], 91
    inc rdx
    jmp filter_loop
keep_rbr:
    test rdi, rdi
    jz bad
    dec rdi
    mov rax, qword ptr [rsi + rdi*8]
    lea r10, [rax + 1]
    mov dword ptr [r8 + rdx*4], r10d
    lea r10, [rdx + 1]
    mov dword ptr [r8 + rax*4], r10d
    mov byte ptr [rbx + rdx], 93
    inc rdx
    jmp filter_loop
filter_done:
    test rdi, rdi
    jnz bad
    mov r13, rdx
    lea rdi, [rip + dispatch]
    lea rax, [rip + op_next]
    mov ecx, 256
fill_disp:
    mov qword ptr [rdi], rax
    add rdi, 8
    dec ecx
    jnz fill_disp
    lea r9, [rip + dispatch]
    lea rax, [rip + op_plus]
    mov qword ptr [r9 + 344], rax
    lea rax, [rip + op_comma]
    mov qword ptr [r9 + 352], rax
    lea rax, [rip + op_minus]
    mov qword ptr [r9 + 360], rax
    lea rax, [rip + op_dot]
    mov qword ptr [r9 + 368], rax
    lea rax, [rip + op_left]
    mov qword ptr [r9 + 480], rax
    lea rax, [rip + op_right]
    mov qword ptr [r9 + 496], rax
    lea rax, [rip + op_lbr]
    mov qword ptr [r9 + 728], rax
    lea rax, [rip + op_rbr]
    mov qword ptr [r9 + 744], rax
    xor r12d, r12d
    lea rbp, [rip + tape]
exec_loop:
    cmp r12, r13
    jae ok
    movzx eax, byte ptr [rbx + r12]
    inc r12
    jmp qword ptr [r9 + rax*8]
op_plus:
    inc byte ptr [rbp]
    jmp exec_loop
op_minus:
    dec byte ptr [rbp]
    jmp exec_loop
op_right:
    inc rbp
    jmp exec_loop
op_left:
    dec rbp
    jmp exec_loop
op_comma:
    cmp r14, r15
    jae comma_eof
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbp], al
    jmp exec_loop
comma_eof:
    mov byte ptr [rbp], 0
    jmp exec_loop
op_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    mov edx, 1
    syscall
    jmp exec_loop
op_lbr:
    cmp byte ptr [rbp], 0
    jne exec_loop
    lea rax, [r12 - 1]
    mov eax, dword ptr [r8 + rax*4]
    mov r12d, eax
    jmp exec_loop
op_rbr:
    cmp byte ptr [rbp], 0
    je exec_loop
    lea rax, [r12 - 1]
    mov eax, dword ptr [r8 + rax*4]
    mov r12d, eax
    jmp exec_loop
op_next:
    jmp exec_loop
ok:
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall
