.globl _start
.text

_start:
    xor ebx, ebx
read_more:
    cmp ebx, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + stdin_buf]
    add rsi, rbx
    mov edx, 65536
    sub edx, ebx
    syscall
    test rax, rax
    jle read_done
    add ebx, eax
    jmp read_more
read_done:
    lea r15, [rip + stdin_buf]
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [r15 + rcx], 33
    je has_bang
    inc rcx
    jmp find_bang
has_bang:
    mov r12, rcx
    lea r13, [r15 + rcx]
    inc r13
    mov r14, rbx
    sub r14, rcx
    dec r14
    jmp br_init
no_bang:
    mov r12, rbx
    xor r13, r13
    xor r14, r14
br_init:
    lea rdi, [rip + jump_table]
    lea r8, [rip + brack_stack]
    xor r9d, r9d
    xor ecx, ecx
br_scan:
    cmp rcx, r12
    jae br_done
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 91
    je br_open
    cmp al, 93
    je br_close
    inc rcx
    jmp br_scan
br_open:
    cmp r9, 1000
    jae fail
    mov qword ptr [r8 + r9*8], rcx
    inc r9
    inc rcx
    jmp br_scan
br_close:
    test r9, r9
    jz fail
    dec r9
    mov rax, qword ptr [r8 + r9*8]
    mov qword ptr [rdi + rcx*8], rax
    mov qword ptr [rdi + rax*8], rcx
    inc rcx
    jmp br_scan
br_done:
    test r9, r9
    jnz fail
    lea r9, [rip + dispatch]
    lea rax, [rip + interp]
    mov ecx, 256
fill_disp:
    mov qword ptr [r9], rax
    add r9, 8
    dec ecx
    jnz fill_disp
    lea r9, [rip + dispatch]
    lea rax, [rip + op_gt]
    mov qword ptr [r9 + 496], rax
    lea rax, [rip + op_lt]
    mov qword ptr [r9 + 480], rax
    lea rax, [rip + op_plus]
    mov qword ptr [r9 + 344], rax
    lea rax, [rip + op_minus]
    mov qword ptr [r9 + 360], rax
    lea rax, [rip + op_dot]
    mov qword ptr [r9 + 368], rax
    lea rax, [rip + op_comma]
    mov qword ptr [r9 + 352], rax
    lea rax, [rip + op_lb]
    mov qword ptr [r9 + 728], rax
    lea rax, [rip + op_rb]
    mov qword ptr [r9 + 744], rax
    lea r8, [rip + jump_table]
    lea r10, [rip + tape]
    xor ebx, ebx
interp:
    cmp rbx, r12
    jae ok_exit
    movzx eax, byte ptr [r15 + rbx]
    inc rbx
    jmp qword ptr [r9 + rax*8]
op_gt:
    inc r10
    jmp interp
op_lt:
    dec r10
    jmp interp
op_plus:
    inc byte ptr [r10]
    jmp interp
op_minus:
    dec byte ptr [r10]
    jmp interp
op_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov edx, 1
    syscall
    jmp interp
op_comma:
    test r14, r14
    jz op_comma_eof
    mov al, byte ptr [r13]
    mov byte ptr [r10], al
    inc r13
    dec r14
    jmp interp
op_comma_eof:
    mov byte ptr [r10], 0
    jmp interp
op_lb:
    cmp byte ptr [r10], 0
    jne interp
    lea rax, [rbx - 1]
    mov rax, qword ptr [r8 + rax*8]
    lea rbx, [rax + 1]
    jmp interp
op_rb:
    cmp byte ptr [r10], 0
    je interp
    lea rax, [rbx - 1]
    mov rax, qword ptr [r8 + rax*8]
    lea rbx, [rax + 1]
    jmp interp
ok_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp ok_exit
fail:
    mov eax, 60
    mov edi, 1
    syscall
    jmp fail

.bss
.align 8
stdin_buf:
    .zero 65536
tape:
    .zero 30000
.align 8
jump_table:
    .zero 524288
brack_stack:
    .zero 8000
.align 8
dispatch:
    .zero 2048
