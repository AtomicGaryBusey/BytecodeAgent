.text
.globl _start

_start:
  lea r12, [rip + inbuf]
  xor r13, r13
read_loop:
  xor eax, eax
  xor edi, edi
  lea rsi, [r12 + r13]
  mov edx, 65536
  sub rdx, r13
  test rdx, rdx
  jz read_done
  syscall
  test rax, rax
  jle read_done
  add r13, rax
  jmp read_loop
read_done:
  xor r14, r14
  call skip_ws
  call parse_value
  call skip_ws
  cmp r14, r13
  jne fail
  lea rsi, [rip + ok_msg]
  mov edx, 3
  mov edi, 1
  mov eax, 1
  syscall
  xor edi, edi
  mov eax, 60
  syscall
  jmp die

skip_ws:
sw_loop:
  cmp r14, r13
  jae sw_ret
  movzx eax, byte ptr [r12 + r14]
  cmp al, 0x20
  je sw_adv
  cmp al, 0x09
  je sw_adv
  cmp al, 0x0A
  je sw_adv
  cmp al, 0x0D
  je sw_adv
sw_ret:
  ret
sw_adv:
  inc r14
  jmp sw_loop

parse_value:
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, 0x7B
  je pv_obj
  cmp al, 0x5B
  je pv_arr
  cmp al, 0x22
  je parse_string
  cmp al, 't'
  je pv_true
  cmp al, 'f'
  je pv_false
  cmp al, 'n'
  je pv_null
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jb fail
  cmp al, '9'
  ja fail
  jmp parse_number
pv_obj:
  inc r14
  jmp parse_object
pv_arr:
  inc r14
  jmp parse_array
pv_true:
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'r'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'u'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'e'
  jne fail
  inc r14
  ret
pv_false:
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'a'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'l'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 's'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'e'
  jne fail
  inc r14
  ret
pv_null:
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'u'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'l'
  jne fail
  inc r14
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 'l'
  jne fail
  inc r14
  ret

parse_object:
  call skip_ws
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 0x7D
  jne po_member
  inc r14
  ret
po_member:
  cmp byte ptr [r12 + r14], 0x22
  jne fail
  call parse_string
  call skip_ws
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 0x3A
  jne fail
  inc r14
  call skip_ws
  call parse_value
po_sep:
  call skip_ws
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, 0x7D
  je po_close
  cmp al, 0x2C
  jne fail
  inc r14
  call skip_ws
  cmp r14, r13
  jae fail
  jmp po_member
po_close:
  inc r14
  ret

parse_array:
  call skip_ws
  cmp r14, r13
  jae fail
  cmp byte ptr [r12 + r14], 0x5D
  jne pa_value
  inc r14
  ret
pa_value:
  call parse_value
pa_sep:
  call skip_ws
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, 0x5D
  je pa_close
  cmp al, 0x2C
  jne fail
  inc r14
  call skip_ws
  cmp r14, r13
  jae fail
  jmp pa_value
pa_close:
  inc r14
  ret

parse_string:
  inc r14
ps_loop:
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, 0x22
  je ps_end
  cmp al, 0x5C
  je ps_esc
  cmp al, 0x20
  jb fail
  inc r14
  jmp ps_loop
ps_end:
  inc r14
  ret
ps_esc:
  inc r14
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, 0x22
  je ps_ok
  cmp al, 0x5C
  je ps_ok
  cmp al, 0x2F
  je ps_ok
  cmp al, 'b'
  je ps_ok
  cmp al, 'f'
  je ps_ok
  cmp al, 'n'
  je ps_ok
  cmp al, 'r'
  je ps_ok
  cmp al, 't'
  je ps_ok
  cmp al, 'u'
  je ps_u
  jmp fail
ps_ok:
  inc r14
  jmp ps_loop
ps_u:
  inc r14
  mov ecx, 4
ps_hex:
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jb fail
  cmp al, '9'
  jbe ps_hex_ok
  cmp al, 'A'
  jb fail
  cmp al, 'F'
  jbe ps_hex_ok
  cmp al, 'a'
  jb fail
  cmp al, 'f'
  ja fail
ps_hex_ok:
  inc r14
  dec ecx
  jnz ps_hex
  jmp ps_loop

parse_number:
  movzx eax, byte ptr [r12 + r14]
  cmp al, '-'
  jne pn_int
  inc r14
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jb fail
  cmp al, '9'
  ja fail
pn_int:
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jne pn_nz
  inc r14
  jmp pn_frac
pn_nz:
  cmp al, '1'
  jb fail
  cmp al, '9'
  ja fail
pn_digs:
  inc r14
  cmp r14, r13
  jae pn_ret
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jb pn_frac
  cmp al, '9'
  ja pn_frac
  jmp pn_digs
pn_frac:
  cmp r14, r13
  jae pn_ret
  movzx eax, byte ptr [r12 + r14]
  cmp al, '.'
  jne pn_exp
  inc r14
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jb fail
  cmp al, '9'
  ja fail
pn_fdigs:
  inc r14
  cmp r14, r13
  jae pn_ret
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jb pn_exp
  cmp al, '9'
  ja pn_exp
  jmp pn_fdigs
pn_exp:
  cmp r14, r13
  jae pn_ret
  movzx eax, byte ptr [r12 + r14]
  cmp al, 'e'
  je pn_ego
  cmp al, 'E'
  jne pn_ret
pn_ego:
  inc r14
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
  cmp al, '+'
  je pn_esign
  cmp al, '-'
  jne pn_edig
pn_esign:
  inc r14
  cmp r14, r13
  jae fail
  movzx eax, byte ptr [r12 + r14]
pn_edig:
  cmp al, '0'
  jb fail
  cmp al, '9'
  ja fail
pn_edigs:
  inc r14
  cmp r14, r13
  jae pn_ret
  movzx eax, byte ptr [r12 + r14]
  cmp al, '0'
  jb pn_ret
  cmp al, '9'
  ja pn_ret
  jmp pn_edigs
pn_ret:
  ret

fail:
  lea rsi, [rip + err_prefix]
  mov edx, 9
  mov edi, 1
  mov eax, 1
  syscall
  mov rax, r14
  call print_u64
  lea rsi, [rip + nlbyte]
  mov edx, 1
  mov edi, 1
  mov eax, 1
  syscall
  mov edi, 1
  mov eax, 60
  syscall
  jmp die

print_u64:
  lea r10, [rip + numbuf_end]
  mov ecx, 10
  test rax, rax
  jnz pu_loop
  dec r10
  mov byte ptr [r10], '0'
  jmp pu_write
pu_loop:
  xor edx, edx
  div rcx
  add dl, '0'
  dec r10
  mov byte ptr [r10], dl
  test rax, rax
  jnz pu_loop
pu_write:
  lea rdx, [rip + numbuf_end]
  sub rdx, r10
  mov rsi, r10
  mov edi, 1
  mov eax, 1
  syscall
  ret

die:
  jmp die

.section .rodata
ok_msg:
  .ascii "ok"
  .byte 0x0A
err_prefix:
  .ascii "error at "
nlbyte:
  .byte 0x0A

.bss
.balign 16
inbuf:
  .zero 65536
numbuf:
  .zero 32
numbuf_end:
