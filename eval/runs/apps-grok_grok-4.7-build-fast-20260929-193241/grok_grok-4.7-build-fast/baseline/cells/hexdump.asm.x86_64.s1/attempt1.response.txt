.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 8
input_buf:
    .zero 65536
out_buf:
    .zero 80

.text
_start:
    cld
    xor r12, r12

read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    xor r13, r13

row_loop:
    cmp r13, r12
    jae final_line

    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe have_n
    mov r15, 16
have_n:
    lea rdi, [rip + out_buf]
    mov rcx, 79
    mov al, 0x20
    rep stosb

    mov eax, r13d
    lea rbx, [rip + hexdigits]
    lea rdi, [rip + out_buf]
    mov ecx, 7
off_loop:
    mov edx, eax
    and edx, 0xf
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + rcx], dl
    shr eax, 4
    dec ecx
    jns off_loop

    lea rsi, [rip + input_buf]
    add rsi, r13
    xor ecx, ecx

byte_loop:
    cmp rcx, r15
    jae bytes_done

    movzx eax, byte ptr [rsi + rcx]

    lea edx, [rcx + rcx*2]
    add edx, 10
    cmp ecx, 8
    jb pos_ok
    inc edx
pos_ok:
    mov r8d, eax
    shr r8d, 4
    movzx r8d, byte ptr [rbx + r8]
    mov byte ptr [rdi + rdx], r8b

    mov r8d, eax
    and r8d, 0xf
    movzx r8d, byte ptr [rbx + r8]
    mov byte ptr [rdi + rdx + 1], r8b

    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    jbe ascii_store
ascii_dot:
    mov al, 0x2e
ascii_store:
    mov byte ptr [rdi + rcx + 61], al

    inc ecx
    jmp byte_loop

bytes_done:
    mov byte ptr [rdi + 60], 0x7c
    lea r8, [r15 + 61]
    mov byte ptr [rdi + r8], 0x7c
    mov byte ptr [rdi + r8 + 1], 0x0a

    lea rdx, [r15 + 63]
    mov rsi, rdi
    mov rdi, 1
    mov rax, 1
    syscall

    add r13, r15
    jmp row_loop

final_line:
    mov eax, r12d
    lea rbx, [rip + hexdigits]
    lea rdi, [rip + out_buf]
    mov ecx, 7
final_hex:
    mov edx, eax
    and edx, 0xf
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + rcx], dl
    shr eax, 4
    dec ecx
    jns final_hex

    mov byte ptr [rdi + 8], 0x0a

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, 9
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
