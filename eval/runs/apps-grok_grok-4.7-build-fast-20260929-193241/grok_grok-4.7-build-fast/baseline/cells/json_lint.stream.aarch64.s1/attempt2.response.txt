.globl _start

// x19 = input base, x20 = length, x21 = index
_start:
  adrp x19, inbuf
  add x19, x19, :lo12:inbuf
  mov x20, #0
  movz x22, #1, lsl #16
read_loop:
  mov x8, #63
  mov x0, #0
  add x1, x19, x20
  sub x2, x22, x20
  svc #0
  cmp x0, #0
  b.le read_done
  add x20, x20, x0
  cmp x20, x22
  b.lo read_loop
read_done:
  mov x21, #0
  bl parse_document
  adrp x1, okmsg
  add x1, x1, :lo12:okmsg
  mov x0, #1
  mov x2, #3
  mov x8, #64
  svc #0
  mov x0, #0
  mov x8, #93
  svc #0
die0:
  b die0

parse_document:
  stp x29, x30, [sp, #-16]!
  bl skip_ws
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.lo error
  ldp x29, x30, [sp], #16
  ret

parse_value:
  stp x29, x30, [sp, #-16]!
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #123
  b.eq pv_obj
  cmp w0, #91
  b.eq pv_arr
  cmp w0, #34
  b.eq pv_str
  cmp w0, #116
  b.eq pv_true
  cmp w0, #102
  b.eq pv_false
  cmp w0, #110
  b.eq pv_null
  cmp w0, #45
  b.eq pv_num
  cmp w0, #48
  b.lo error
  cmp w0, #57
  b.hi error
  b pv_num
pv_obj:
  bl parse_object
  b pv_done
pv_arr:
  bl parse_array
  b pv_done
pv_str:
  bl parse_string
  b pv_done
pv_true:
  bl parse_true
  b pv_done
pv_false:
  bl parse_false
  b pv_done
pv_null:
  bl parse_null
  b pv_done
pv_num:
  bl parse_number
pv_done:
  ldp x29, x30, [sp], #16
  ret

parse_object:
  stp x29, x30, [sp, #-16]!
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #125
  b.eq po_end
po_member:
  bl parse_string
  bl skip_ws
  mov w0, #58
  bl expect_char
  bl skip_ws
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #125
  b.eq po_end
  cmp w0, #44
  b.ne error
  add x21, x21, #1
  bl skip_ws
  b po_member
po_end:
  add x21, x21, #1
  ldp x29, x30, [sp], #16
  ret

parse_array:
  stp x29, x30, [sp, #-16]!
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #93
  b.eq pa_end
pa_elem:
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #93
  b.eq pa_end
  cmp w0, #44
  b.ne error
  add x21, x21, #1
  bl skip_ws
  b pa_elem
pa_end:
  add x21, x21, #1
  ldp x29, x30, [sp], #16
  ret

parse_string:
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #34
  b.ne error
  add x21, x21, #1
ps_loop:
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #34
  b.eq ps_end
  cmp w0, #92
  b.eq ps_esc
  cmp w0, #32
  b.lo error
  add x21, x21, #1
  b ps_loop
ps_end:
  add x21, x21, #1
  ret
ps_esc:
  add x21, x21, #1
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #34
  b.eq ps_esc_ok
  cmp w0, #92
  b.eq ps_esc_ok
  cmp w0, #47
  b.eq ps_esc_ok
  cmp w0, #98
  b.eq ps_esc_ok
  cmp w0, #102
  b.eq ps_esc_ok
  cmp w0, #110
  b.eq ps_esc_ok
  cmp w0, #114
  b.eq ps_esc_ok
  cmp w0, #116
  b.eq ps_esc_ok
  cmp w0, #117
  b.eq ps_u
  b error
ps_esc_ok:
  add x21, x21, #1
  b ps_loop
ps_u:
  add x21, x21, #1
  mov w3, #4
ps_hex:
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  sub w1, w0, #48
  cmp w1, #10
  b.lo ps_hex_ok
  sub w1, w0, #65
  cmp w1, #6
  b.lo ps_hex_ok
  sub w1, w0, #97
  cmp w1, #6
  b.hs error
ps_hex_ok:
  add x21, x21, #1
  subs w3, w3, #1
  b.ne ps_hex
  b ps_loop

parse_number:
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #45
  b.ne pn_int
  add x21, x21, #1
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
pn_int:
  cmp w0, #48
  b.eq pn_zero
  cmp w0, #49
  b.lo error
  cmp w0, #57
  b.hi error
pn_digs:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_done
  ldrb w0, [x19, x21]
  cmp w0, #48
  b.lo pn_frac
  cmp w0, #57
  b.hi pn_frac
  b pn_digs
pn_zero:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_done
  ldrb w0, [x19, x21]
  cmp w0, #48
  b.lo pn_frac
  cmp w0, #57
  b.ls error
  b pn_frac
pn_frac:
  cmp w0, #46
  b.ne pn_exp
  add x21, x21, #1
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #48
  b.lo error
  cmp w0, #57
  b.hi error
pn_frac_digs:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_done
  ldrb w0, [x19, x21]
  cmp w0, #48
  b.lo pn_exp
  cmp w0, #57
  b.hi pn_exp
  b pn_frac_digs
pn_exp:
  cmp w0, #101
  b.eq pn_exp_go
  cmp w0, #69
  b.ne pn_done
pn_exp_go:
  add x21, x21, #1
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
  cmp w0, #43
  b.eq pn_exp_sign
  cmp w0, #45
  b.ne pn_exp_first
pn_exp_sign:
  add x21, x21, #1
  cmp x21, x20
  b.hs error
  ldrb w0, [x19, x21]
pn_exp_first:
  cmp w0, #48
  b.lo error
  cmp w0, #57
  b.hi error
pn_exp_digs:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_done
  ldrb w0, [x19, x21]
  cmp w0, #48
  b.lo pn_done
  cmp w0, #57
  b.hi pn_done
  b pn_exp_digs
pn_done:
  ret

parse_true:
  stp x29, x30, [sp, #-16]!
  mov w0, #116
  bl expect_char
  mov w0, #114
  bl expect_char
  mov w0, #117
  bl expect_char
  mov w0, #101
  bl expect_char
  ldp x29, x30, [sp], #16
  ret

parse_false:
  stp x29, x30, [sp, #-16]!
  mov w0, #102
  bl expect_char
  mov w0, #97
  bl expect_char
  mov w0, #108
  bl expect_char
  mov w0, #115
  bl expect_char
  mov w0, #101
  bl expect_char
  ldp x29, x30, [sp], #16
  ret

parse_null:
  stp x29, x30, [sp, #-16]!
  mov w0, #110
  bl expect_char
  mov w0, #117
  bl expect_char
  mov w0, #108
  bl expect_char
  mov w0, #108
  bl expect_char
  ldp x29, x30, [sp], #16
  ret

// w0 = expected byte; errors without consuming on mismatch or eof
expect_char:
  cmp x21, x20
  b.hs error
  ldrb w1, [x19, x21]
  cmp w1, w0
  b.ne error
  add x21, x21, #1
  ret

skip_ws:
sw_loop:
  cmp x21, x20
  b.hs sw_done
  ldrb w0, [x19, x21]
  cmp w0, #32
  b.eq sw_adv
  cmp w0, #9
  b.eq sw_adv
  cmp w0, #10
  b.eq sw_adv
  cmp w0, #13
  b.eq sw_adv
  ret
sw_adv:
  add x21, x21, #1
  b sw_loop
sw_done:
  ret

// print "error at <x21>\n" and exit 1
error:
  mov x15, x21
  adrp x10, prefix
  add x10, x10, :lo12:prefix
  adrp x11, outbuf
  add x11, x11, :lo12:outbuf
  mov x12, #9
copy_pre:
  ldrb w13, [x10], #1
  strb w13, [x11], #1
  subs x12, x12, #1
  b.ne copy_pre
  mov x14, #10
  adrp x16, digbuf
  add x16, x16, :lo12:digbuf
  mov x17, x16
  cbz x15, err_zero
err_div:
  udiv x4, x15, x14
  msub x5, x4, x14, x15
  add x5, x5, #48
  strb w5, [x17], #1
  mov x15, x4
  cbnz x15, err_div
err_rev:
  sub x17, x17, #1
  ldrb w5, [x17]
  strb w5, [x11], #1
  cmp x17, x16
  b.ne err_rev
  b err_nl
err_zero:
  mov w5, #48
  strb w5, [x11], #1
err_nl:
  mov w5, #10
  strb w5, [x11], #1
  adrp x1, outbuf
  add x1, x1, :lo12:outbuf
  sub x2, x11, x1
  mov x0, #1
  mov x8, #64
  svc #0
  mov x0, #1
  mov x8, #93
  svc #0
die1:
  b die1
