.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUFSIZE, 65536

.section .rodata
ten:
    .quad 10

.bss
.align 8
inbuf:
    .zero 65536
outbuf:
    .zero 64
digitbuf:
    .zero 32
digit_end:

.text
.globl _start
_start:
    xor ebx, ebx
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d

read_loop:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, BUFSIZE
    syscall
    test rax, rax
    jle finish
    add r13, rax
    mov rcx, rax
    lea rsi, [rip + inbuf]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne not_nl
    inc rbx
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    jbe is_ws
is_word:
    test r14d, r14d
    jnz next_byte
    inc r12
    mov r14d, 1
    jmp next_byte
is_ws:
    xor r14d, r14d
next_byte:
    dec rcx
    jnz scan_loop
    jmp read_loop

finish:
    lea rdi, [rip + outbuf]
    mov rax, rbx
    call append_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r12
    call append_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call append_uint
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall

exit_loop:
    mov eax, SYS_exit
    xor edi, edi
    syscall
    jmp exit_loop

append_uint:
    lea rsi, [rip + digit_end]
    mov rcx, rsi
    test rax, rax
    jnz au_loop
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
au_loop:
    xor edx, edx
    div qword ptr [rip + ten]
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz au_loop
au_copy:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    cmp rcx, rsi
    jne au_copy
    ret
