.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_NEST, 1000
.equ C_PLUS, 43
.equ C_COMMA, 44
.equ C_MINUS, 45
.equ C_DOT, 46
.equ C_LT, 60
.equ C_GT, 62
.equ C_LB, 91
.equ C_RB, 93
.equ C_BANG, 33

.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, x20
    mov x2, #1
    lsl x2, x2, #16
    add x21, x20, x2

read_loop:
    cmp x19, x21
    b.hs read_done
    mov x0, #0
    mov x1, x19
    sub x2, x21, x19
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    sub x19, x19, x20
    mov x21, #0

find_loop:
    cmp x21, x19
    b.hs no_bang
    ldrb w0, [x20, x21]
    cmp w0, #C_BANG
    b.eq found_bang
    add x21, x21, #1
    b find_loop

found_bang:
    mov x23, x21
    add x24, x21, #1
    b do_compact

no_bang:
    mov x23, x19
    mov x24, x19

do_compact:
    adrp x22, code_buf
    add x22, x22, :lo12:code_buf
    mov x26, #0
    mov x21, #0

compact_loop:
    cmp x26, x23
    b.hs compact_done
    ldrb w0, [x20, x26]
    add x26, x26, #1
    cmp w0, #C_PLUS
    b.lo compact_loop
    cmp w0, #C_DOT
    b.ls compact_keep
    cmp w0, #C_LT
    b.eq compact_keep
    cmp w0, #C_GT
    b.eq compact_keep
    cmp w0, #C_LB
    b.eq compact_keep
    cmp w0, #C_RB
    b.eq compact_keep
    b compact_loop

compact_keep:
    strb w0, [x22, x21]
    add x21, x21, #1
    b compact_loop

compact_done:
    mov x23, x21
    mov x25, #0
    mov x26, #0
    adrp x27, jumps
    add x27, x27, :lo12:jumps
    adrp x28, bstack
    add x28, x28, :lo12:bstack

match_loop:
    cmp x26, x23
    b.hs match_end
    ldrb w0, [x22, x26]
    cmp w0, #C_LB
    b.eq on_lb
    cmp w0, #C_RB
    b.eq on_rb
    add x26, x26, #1
    b match_loop

on_lb:
    cmp x25, #MAX_NEST
    b.hs fail
    str x26, [x28, x25, lsl #3]
    add x25, x25, #1
    add x26, x26, #1
    b match_loop

on_rb:
    cbz x25, fail
    sub x25, x25, #1
    ldr x1, [x28, x25, lsl #3]
    str w26, [x27, x1, lsl #2]
    str w1, [x27, x26, lsl #2]
    add x26, x26, #1
    b match_loop

match_end:
    cbnz x25, fail
    adrp x21, tape
    add x21, x21, :lo12:tape
    mov x26, #0

exec_loop:
    cmp x26, x23
    b.hs ok_exit
    ldrb w0, [x22, x26]
    cmp w0, #C_RB
    b.eq c_rb
    cmp w0, #C_LB
    b.eq c_lb
    cmp w0, #C_PLUS
    b.eq c_plus
    cmp w0, #C_MINUS
    b.eq c_minus
    cmp w0, #C_GT
    b.eq c_gt
    cmp w0, #C_LT
    b.eq c_lt
    cmp w0, #C_DOT
    b.eq c_dot
    cmp w0, #C_COMMA
    b.eq c_comma

exec_next:
    add x26, x26, #1
    b exec_loop

c_plus:
    ldrb w1, [x21]
    add w1, w1, #1
    strb w1, [x21]
    b exec_next

c_minus:
    ldrb w1, [x21]
    sub w1, w1, #1
    strb w1, [x21]
    b exec_next

c_gt:
    add x21, x21, #1
    b exec_next

c_lt:
    sub x21, x21, #1
    b exec_next

c_dot:
    mov x0, #1
    mov x1, x21
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    b exec_next

c_comma:
    cmp x24, x19
    b.hs c_comma_z
    ldrb w1, [x20, x24]
    add x24, x24, #1
    strb w1, [x21]
    b exec_next

c_comma_z:
    strb wzr, [x21]
    b exec_next

c_lb:
    ldrb w1, [x21]
    cbz w1, c_lb_jump
    b exec_next

c_lb_jump:
    ldr w1, [x27, x26, lsl #2]
    add x26, x1, #1
    b exec_loop

c_rb:
    ldrb w1, [x21]
    cbnz w1, c_rb_jump
    b exec_next

c_rb_jump:
    ldr w1, [x27, x26, lsl #2]
    add x26, x1, #1
    b exec_loop

fail:
    mov x0, #1
    b do_exit

ok_exit:
    mov x0, #0

do_exit:
    mov x8, #SYS_EXIT
    svc #0
    b do_exit

.bss
.balign 16
input_buf:
    .zero 65536
code_buf:
    .zero 65536
tape:
    .zero 30000
.balign 8
jumps:
    .zero 262144
bstack:
    .zero 8000
