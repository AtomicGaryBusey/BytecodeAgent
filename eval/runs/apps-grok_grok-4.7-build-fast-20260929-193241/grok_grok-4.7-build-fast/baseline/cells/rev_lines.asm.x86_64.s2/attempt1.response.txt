.section .rodata
newline:
    .byte 0x0A

.bss
.align 8
input:
    .zero 65536

.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_loop
read_done:
    xor r12, r12
next_line:
    cmp r12, rbx
    jae do_exit
    mov r13, r12
scan_line:
    cmp r13, rbx
    jae line_to_end
    mov al, byte ptr [input + r13]
    cmp al, 10
    je line_with_nl
    inc r13
    jmp scan_line
line_with_nl:
    mov r14, r13
    call reverse_write
    lea r12, [r13 + 1]
    jmp next_line
line_to_end:
    mov r14, rbx
    call reverse_write
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

reverse_write:
    mov r8, r12
    mov r9, r14
    cmp r8, r9
    jae rw_write
    dec r9
rw_loop:
    cmp r8, r9
    jge rw_write
    mov al, byte ptr [input + r8]
    mov cl, byte ptr [input + r9]
    mov byte ptr [input + r8], cl
    mov byte ptr [input + r9], al
    inc r8
    dec r9
    jmp rw_loop
rw_write:
    mov rdx, r14
    sub rdx, r12
    test rdx, rdx
    jz rw_nl
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + input]
    add rsi, r12
    syscall
rw_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    ret
