.globl _start

.equ IN_CAP, 49152
.equ OUT_CAP, 66560
.equ LINE_LEN, 76

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 8
inbuf:
    .zero IN_CAP
outbuf:
    .zero OUT_CAP

.text
_start:
    lea rbx, [rip + inbuf]
    xor r12, r12

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, IN_CAP
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, IN_CAP
    jb read_loop

read_done:
    test r12, r12
    jz exit_ok

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + alphabet]
    xor r9d, r9d

encode_loop:
    cmp r12, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx ecx, byte ptr [rsi + 2]
    add rsi, 3
    sub r12, 3

    mov r10d, eax
    shr r10d, 2

    mov r11d, eax
    and r11d, 3
    shl r11d, 4
    mov ebx, edx
    shr ebx, 4
    or r11d, ebx

    mov ebx, edx
    and ebx, 15
    shl ebx, 2
    mov r13d, ecx
    shr r13d, 6
    or ebx, r13d

    and ecx, 63

    movzx eax, byte ptr [r8 + r10]
    call emit
    movzx eax, byte ptr [r8 + r11]
    call emit
    movzx eax, byte ptr [r8 + rbx]
    call emit
    movzx eax, byte ptr [r8 + rcx]
    call emit
    jmp encode_loop

encode_tail:
    cmp r12, 1
    je tail_one
    cmp r12, 2
    jne encode_finish

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]

    mov r10d, eax
    shr r10d, 2

    mov r11d, eax
    and r11d, 3
    shl r11d, 4
    mov ebx, edx
    shr ebx, 4
    or r11d, ebx

    and edx, 15
    shl edx, 2

    movzx eax, byte ptr [r8 + r10]
    call emit
    movzx eax, byte ptr [r8 + r11]
    call emit
    movzx eax, byte ptr [r8 + rdx]
    call emit
    mov al, 0x3D
    call emit
    jmp encode_finish

tail_one:
    movzx eax, byte ptr [rsi]
    mov r10d, eax
    shr r10d, 2
    and eax, 3
    shl eax, 4
    mov r11d, eax

    movzx eax, byte ptr [r8 + r10]
    call emit
    movzx eax, byte ptr [r8 + r11]
    call emit
    mov al, 0x3D
    call emit
    mov al, 0x3D
    call emit

encode_finish:
    test r9d, r9d
    jz write_out
    mov byte ptr [rdi], 0x0A
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    mov rdx, rdi

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_ok

emit:
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, LINE_LEN
    jne 1f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
1:
    ret
