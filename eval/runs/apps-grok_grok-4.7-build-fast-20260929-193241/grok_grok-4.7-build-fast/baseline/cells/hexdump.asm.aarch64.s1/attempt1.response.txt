.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
.globl _start

_start:
    mov x19, #0

read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0

row_loop:
    cmp x20, x19
    b.hs final_off
    sub x21, x19, x20
    cmp x21, #16
    b.ls row_n_set
    mov x21, #16
row_n_set:
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    add x22, x22, x20
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x0, x20
    bl hex8
    mov w2, #32
    strb w2, [x1], #1
    strb w2, [x1], #1
    mov x23, #0
slot_loop:
    cmp x23, x21
    b.hs slot_pad
    ldrb w0, [x22, x23]
    bl hex2
    mov w2, #32
    strb w2, [x1], #1
    b slot_next
slot_pad:
    mov w2, #32
    strb w2, [x1], #1
    strb w2, [x1], #1
    strb w2, [x1], #1
slot_next:
    cmp x23, #7
    b.ne slot_no_gap
    mov w2, #32
    strb w2, [x1], #1
slot_no_gap:
    add x23, x23, #1
    cmp x23, #16
    b.lo slot_loop
    mov w2, #32
    strb w2, [x1], #1
    mov w2, #124
    strb w2, [x1], #1
    mov x23, #0
asc_loop:
    cmp x23, x21
    b.hs asc_done
    ldrb w0, [x22, x23]
    cmp w0, #32
    b.lo asc_dot
    cmp w0, #126
    b.hi asc_dot
    strb w0, [x1], #1
    b asc_advance
asc_dot:
    mov w0, #46
    strb w0, [x1], #1
asc_advance:
    add x23, x23, #1
    b asc_loop
asc_done:
    mov w2, #124
    strb w2, [x1], #1
    mov w2, #10
    strb w2, [x1], #1
    bl write_line
    add x20, x20, x21
    b row_loop

final_off:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x0, x19
    bl hex8
    mov w2, #10
    strb w2, [x1], #1
    bl write_line
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b final_off

hex8:
    mov x2, #32
hex8_loop:
    sub x2, x2, #4
    lsr x3, x0, x2
    and x3, x3, #15
    cmp x3, #10
    b.lo hex8_dec
    add x3, x3, #87
    b hex8_store
hex8_dec:
    add x3, x3, #48
hex8_store:
    strb w3, [x1], #1
    cbnz x2, hex8_loop
    ret

hex2:
    lsr w2, w0, #4
    and w3, w0, #15
    cmp w2, #10
    b.lo hex2_dec1
    add w2, w2, #87
    b hex2_put1
hex2_dec1:
    add w2, w2, #48
hex2_put1:
    strb w2, [x1], #1
    cmp w3, #10
    b.lo hex2_dec2
    add w3, w3, #87
    b hex2_put2
hex2_dec2:
    add w3, w3, #48
hex2_put2:
    strb w3, [x1], #1
    ret

write_line:
    mov x2, x1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x2, x1
write_loop:
    cbz x2, write_ret
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le write_ret
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop
write_ret:
    ret

.section .bss
.align 3
inbuf:
    .zero 65536
outbuf:
    .zero 128
