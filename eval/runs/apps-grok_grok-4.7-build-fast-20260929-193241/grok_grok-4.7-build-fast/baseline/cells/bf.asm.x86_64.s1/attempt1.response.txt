.bss
.align 8
input_buf:
    .zero 65536
tape:
    .zero 30000
jump_map:
    .zero 262144
bracket_stack:
    .zero 8000
dispatch:
    .zero 2048

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_more
read_done:
    mov r13, r12
    xor r14d, r14d
    xor r15d, r15d
    xor ecx, ecx
find_bang:
    cmp rcx, r12
    jae split_done
    cmp byte ptr [rbx + rcx], 33
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r13, rcx
    lea r14, [rcx + 1]
    mov r15, r12
    sub r15, r14
split_done:
    lea rdi, [rip + dispatch]
    lea rax, [rip + cmd_skip]
    mov ecx, 256
fill_loop:
    mov qword ptr [rdi], rax
    add rdi, 8
    dec ecx
    jnz fill_loop
    lea rax, [rip + cmd_plus]
    mov qword ptr [rip + dispatch + 344], rax
    lea rax, [rip + cmd_comma]
    mov qword ptr [rip + dispatch + 352], rax
    lea rax, [rip + cmd_minus]
    mov qword ptr [rip + dispatch + 360], rax
    lea rax, [rip + cmd_dot]
    mov qword ptr [rip + dispatch + 368], rax
    lea rax, [rip + cmd_lt]
    mov qword ptr [rip + dispatch + 480], rax
    lea rax, [rip + cmd_gt]
    mov qword ptr [rip + dispatch + 496], rax
    lea rax, [rip + cmd_lbr]
    mov qword ptr [rip + dispatch + 728], rax
    lea rax, [rip + cmd_rbr]
    mov qword ptr [rip + dispatch + 744], rax
    xor ecx, ecx
    xor edx, edx
    lea r8, [rip + bracket_stack]
    lea r9, [rip + jump_map]
scan_loop:
    cmp rcx, r13
    jae scan_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 91
    je scan_open
    cmp al, 93
    je scan_close
    inc rcx
    jmp scan_loop
scan_open:
    cmp rdx, 1000
    jae unbalanced
    mov qword ptr [r8 + rdx*8], rcx
    inc rdx
    inc rcx
    jmp scan_loop
scan_close:
    test rdx, rdx
    jz unbalanced
    dec rdx
    mov rax, qword ptr [r8 + rdx*8]
    mov dword ptr [r9 + rax*4], ecx
    mov dword ptr [r9 + rcx*4], eax
    inc rcx
    jmp scan_loop
scan_done:
    test rdx, rdx
    jnz unbalanced
    lea rsi, [rbx + r14]
    lea r12, [rip + tape]
    lea r10, [rip + dispatch]
    xor ecx, ecx
exec_loop:
    cmp rcx, r13
    jae done_ok
    movzx eax, byte ptr [rbx + rcx]
    jmp qword ptr [r10 + rax*8]
cmd_skip:
    inc rcx
    jmp exec_loop
cmd_gt:
    inc r12
    inc rcx
    jmp exec_loop
cmd_lt:
    dec r12
    inc rcx
    jmp exec_loop
cmd_plus:
    inc byte ptr [r12]
    inc rcx
    jmp exec_loop
cmd_minus:
    dec byte ptr [r12]
    inc rcx
    jmp exec_loop
cmd_dot:
    push rcx
    push rsi
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov edx, 1
    syscall
    pop rsi
    pop rcx
    inc rcx
    jmp exec_loop
cmd_comma:
    test r15, r15
    jz comma_eof
    mov al, byte ptr [rsi]
    mov byte ptr [r12], al
    inc rsi
    dec r15
    inc rcx
    jmp exec_loop
comma_eof:
    mov byte ptr [r12], 0
    inc rcx
    jmp exec_loop
cmd_lbr:
    cmp byte ptr [r12], 0
    jne lbr_nz
    mov ecx, dword ptr [r9 + rcx*4]
    inc rcx
    jmp exec_loop
lbr_nz:
    inc rcx
    jmp exec_loop
cmd_rbr:
    cmp byte ptr [r12], 0
    je rbr_z
    mov ecx, dword ptr [r9 + rcx*4]
    inc rcx
    jmp exec_loop
rbr_z:
    inc rcx
    jmp exec_loop
done_ok:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall
