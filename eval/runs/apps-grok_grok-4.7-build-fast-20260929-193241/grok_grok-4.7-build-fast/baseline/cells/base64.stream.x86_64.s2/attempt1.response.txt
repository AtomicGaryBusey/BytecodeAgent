.equ IN_MAX, 49152
.equ WRAP, 76
.equ PAD, 64

.section .rodata
alphabet:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="

.bss
inbuf:
	.zero 49152
outbuf:
	.zero 131072

.text
.globl _start
_start:
	xor ebx, ebx
read_loop:
	xor eax, eax
	xor edi, edi
	lea rsi, [rip + inbuf]
	add rsi, rbx
	mov edx, IN_MAX
	sub rdx, rbx
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	cmp rbx, IN_MAX
	jb read_loop
read_done:
	lea rsi, [rip + inbuf]
	mov rcx, rbx
	lea rdi, [rip + outbuf]
	xor r15d, r15d

encode_full:
	cmp rcx, 3
	jb encode_rem
	movzx eax, byte ptr [rsi]
	movzx edx, byte ptr [rsi + 1]
	movzx r8d, byte ptr [rsi + 2]
	add rsi, 3
	sub rcx, 3

	mov r9d, eax
	shr eax, 2
	call emit_one

	mov eax, r9d
	and eax, 3
	shl eax, 4
	mov r10d, edx
	shr r10d, 4
	or eax, r10d
	call emit_one

	mov eax, edx
	and eax, 15
	shl eax, 2
	mov r10d, r8d
	shr r10d, 6
	or eax, r10d
	call emit_one

	mov eax, r8d
	and eax, 63
	call emit_one
	jmp encode_full

encode_rem:
	cmp rcx, 1
	jb encode_finish
	je encode_one

	movzx r9d, byte ptr [rsi]
	movzx edx, byte ptr [rsi + 1]

	mov eax, r9d
	shr eax, 2
	call emit_one

	mov eax, r9d
	and eax, 3
	shl eax, 4
	mov r10d, edx
	shr r10d, 4
	or eax, r10d
	call emit_one

	mov eax, edx
	and eax, 15
	shl eax, 2
	call emit_one

	mov eax, PAD
	call emit_one
	jmp encode_finish

encode_one:
	movzx r9d, byte ptr [rsi]
	mov eax, r9d
	shr eax, 2
	call emit_one

	mov eax, r9d
	and eax, 3
	shl eax, 4
	call emit_one

	mov eax, PAD
	call emit_one
	mov eax, PAD
	call emit_one

encode_finish:
	test r15, r15
	jz no_nl
	mov byte ptr [rdi], 10
	inc rdi
no_nl:
	lea rsi, [rip + outbuf]
	sub rdi, rsi
	mov rdx, rdi

write_loop:
	test rdx, rdx
	jz do_exit
	mov eax, 1
	mov edi, 1
	syscall
	add rsi, rax
	sub rdx, rax
	jmp write_loop

do_exit:
	mov eax, 60
	xor edi, edi
	syscall

emit_one:
	mov r11d, eax
	lea rax, [rip + alphabet]
	movzx eax, byte ptr [rax + r11]
	mov byte ptr [rdi], al
	inc rdi
	inc r15
	cmp r15, WRAP
	jne 1f
	mov byte ptr [rdi], 10
	inc rdi
	xor r15d, r15d
1:
	ret
