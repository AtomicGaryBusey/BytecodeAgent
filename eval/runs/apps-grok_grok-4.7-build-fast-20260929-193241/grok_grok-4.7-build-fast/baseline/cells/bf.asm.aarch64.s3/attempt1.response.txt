.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536
.equ MAX_DEPTH, 1000

.globl _start

.text
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    mov x8, #SYS_read
    mov x0, #0
    add x1, x20, x19
    mov x2, #IN_MAX
    sub x2, x2, x19
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    cmp x19, #IN_MAX
    b.lo read_loop

read_done:
    // x21 = index while searching; becomes program length
    mov x21, #0
    mov w22, #33

find_bang:
    cmp x21, x19
    b.eq no_bang
    ldrb w0, [x20, x21]
    cmp w0, w22
    b.eq found_bang
    add x21, x21, #1
    b find_bang

found_bang:
    // program is [0, x21); input is [x21+1, x19)
    add x23, x21, #1
    mov x24, x19
    b brackets

no_bang:
    mov x21, x19
    mov x23, x19
    mov x24, x19

brackets:
    adrp x26, jumps
    add x26, x26, :lo12:jumps
    adrp x27, bstack
    add x27, x27, :lo12:bstack
    mov x25, #0
    mov x28, #0
    mov w4, #91
    mov w5, #93

scan_loop:
    cmp x28, x21
    b.eq scan_done
    ldrb w0, [x20, x28]
    cmp w0, w4
    b.eq scan_open
    cmp w0, w5
    b.eq scan_close
    add x28, x28, #1
    b scan_loop

scan_open:
    cmp x25, #MAX_DEPTH
    b.hs unbalanced
    str w28, [x27, x25, lsl #2]
    add x25, x25, #1
    add x28, x28, #1
    b scan_loop

scan_close:
    cbz x25, unbalanced
    sub x25, x25, #1
    ldr w0, [x27, x25, lsl #2]
    add x1, x28, #1
    str w1, [x26, x0, lsl #2]
    add w1, w0, #1
    str w1, [x26, x28, lsl #2]
    add x28, x28, #1
    b scan_loop

scan_done:
    cbnz x25, unbalanced

    adrp x9, tape
    add x9, x9, :lo12:tape
    mov x10, #0
    mov x11, #0

exec_loop:
    cmp x11, x21
    b.hs finish_ok
    ldrb w0, [x20, x11]
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_inc
    cmp w0, #45
    b.eq cmd_dec
    cmp w0, #46
    b.eq cmd_out
    cmp w0, #44
    b.eq cmd_in
    cmp w0, #91
    b.eq cmd_lbrk
    cmp w0, #93
    b.eq cmd_rbrk
    add x11, x11, #1
    b exec_loop

cmd_right:
    add x10, x10, #1
    add x11, x11, #1
    b exec_loop

cmd_left:
    sub x10, x10, #1
    add x11, x11, #1
    b exec_loop

cmd_inc:
    ldrb w0, [x9, x10]
    add w0, w0, #1
    strb w0, [x9, x10]
    add x11, x11, #1
    b exec_loop

cmd_dec:
    ldrb w0, [x9, x10]
    sub w0, w0, #1
    strb w0, [x9, x10]
    add x11, x11, #1
    b exec_loop

cmd_out:
    add x1, x9, x10
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x11, x11, #1
    b exec_loop

cmd_in:
    cmp x23, x24
    b.hs cmd_in_eof
    ldrb w0, [x20, x23]
    add x23, x23, #1
    strb w0, [x9, x10]
    add x11, x11, #1
    b exec_loop

cmd_in_eof:
    strb wzr, [x9, x10]
    add x11, x11, #1
    b exec_loop

cmd_lbrk:
    ldrb w0, [x9, x10]
    cbz w0, cmd_lbrk_jump
    add x11, x11, #1
    b exec_loop

cmd_lbrk_jump:
    ldr w0, [x26, x11, lsl #2]
    mov x11, x0
    b exec_loop

cmd_rbrk:
    ldrb w0, [x9, x10]
    cbnz w0, cmd_rbrk_jump
    add x11, x11, #1
    b exec_loop

cmd_rbrk_jump:
    ldr w0, [x26, x11, lsl #2]
    mov x11, x0
    b exec_loop

finish_ok:
    mov x0, #0
    b do_exit

unbalanced:
    mov x0, #1

do_exit:
    mov x8, #SYS_exit
    svc #0
    b do_exit

.bss
.align 3
input_buf:
    .zero 65536
tape:
    .zero 30000
.align 3
jumps:
    .zero 262144
bstack:
    .zero 4000
