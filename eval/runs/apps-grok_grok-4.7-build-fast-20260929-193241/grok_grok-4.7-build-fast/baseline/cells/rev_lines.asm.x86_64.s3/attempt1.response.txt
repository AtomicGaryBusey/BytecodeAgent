.equ BUF_SIZE, 65536

.globl _start

.text
_start:
    lea r14, [rip + buf]
    xor rbx, rbx

read_loop:
    cmp rbx, BUF_SIZE
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + rbx]
    mov rdx, BUF_SIZE
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    xor r12, r12

line_loop:
    cmp r12, rbx
    jae exit_ok
    mov r13, r12

scan_loop:
    cmp r12, rbx
    jae reverse_line
    movzx eax, byte ptr [r14 + r12]
    cmp al, 10
    je reverse_line
    inc r12
    jmp scan_loop

reverse_line:
    mov rsi, r13
    mov rdi, r12
    cmp rsi, rdi
    jae write_line
    dec rdi

rev_loop:
    cmp rsi, rdi
    jae write_line
    mov al, byte ptr [r14 + rsi]
    mov cl, byte ptr [r14 + rdi]
    mov byte ptr [r14 + rsi], cl
    mov byte ptr [r14 + rdi], al
    inc rsi
    dec rdi
    jmp rev_loop

write_line:
    mov rdx, r12
    sub rdx, r13
    test rdx, rdx
    jz write_nl
    mov rax, 1
    mov rdi, 1
    lea rsi, [r14 + r13]
    syscall

write_nl:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    cmp r12, rbx
    jae line_loop
    inc r12
    jmp line_loop

exit_ok:
    mov rax, 60
    xor edi, edi
    syscall

.section .rodata
nl:
    .byte 10

.bss
.align 8
buf:
    .zero 65536
