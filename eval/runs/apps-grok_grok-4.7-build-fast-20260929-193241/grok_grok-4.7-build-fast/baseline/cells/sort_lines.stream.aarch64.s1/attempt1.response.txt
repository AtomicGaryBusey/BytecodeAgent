.globl _start
.text
.balign 4

_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    adrp x23, line_off
    add x23, x23, :lo12:line_off
    adrp x24, line_len
    add x24, x24, :lo12:line_len
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #32768
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    mov x2, #32768
    cmp x19, x2
    b.lo read_loop

read_done:
    mov x21, #0
    mov x22, #0

parse_loop:
    cmp x22, x19
    b.hs parse_done
    mov x25, x22

scan_loop:
    cmp x22, x19
    b.hs eof_line
    ldrb w0, [x20, x22]
    add x22, x22, #1
    cmp w0, #10
    b.ne scan_loop
    sub x26, x22, x25
    sub x26, x26, #1
    str w25, [x23, x21, lsl #2]
    str w26, [x24, x21, lsl #2]
    add x21, x21, #1
    b parse_loop

eof_line:
    sub x26, x19, x25
    str w25, [x23, x21, lsl #2]
    str w26, [x24, x21, lsl #2]
    add x21, x21, #1

parse_done:
    cmp x21, #2
    b.lo sort_done
    lsr x19, x21, #1

build_loop:
    sub x19, x19, #1
    mov x0, x19
    mov x1, x21
    bl sift
    cbnz x19, build_loop
    mov x19, x21

extract_loop:
    subs x19, x19, #1
    b.eq sort_done
    mov x0, #0
    mov x1, x19
    bl swap_ij
    mov x0, #0
    mov x1, x19
    bl sift
    b extract_loop

sort_done:
    mov x19, #0
    adrp x22, nlbyte
    add x22, x22, :lo12:nlbyte

print_loop:
    cmp x19, x21
    b.hs do_exit
    ldr w2, [x23, x19, lsl #2]
    ldr w3, [x24, x19, lsl #2]
    cbz x3, print_nl
    mov x0, #1
    add x1, x20, x2
    mov x2, x3
    mov x8, #64
    svc #0

print_nl:
    mov x0, #1
    mov x1, x22
    mov x2, #1
    mov x8, #64
    svc #0
    add x19, x19, #1
    b print_loop

do_exit:
    mov x0, #0
    mov x8, #93
    svc #0
    b do_exit

cmp_gt:
    ldr w2, [x23, x0, lsl #2]
    ldr w3, [x24, x0, lsl #2]
    ldr w4, [x23, x1, lsl #2]
    ldr w5, [x24, x1, lsl #2]
    add x2, x20, x2
    add x4, x20, x4
    mov x6, #0
    cmp x3, x5
    csel x7, x3, x5, lo

cmp_gt_loop:
    cmp x6, x7
    b.hs cmp_gt_len
    ldrb w0, [x2, x6]
    ldrb w1, [x4, x6]
    cmp w0, w1
    b.hi cmp_gt_yes
    b.lo cmp_gt_no
    add x6, x6, #1
    b cmp_gt_loop

cmp_gt_len:
    cmp x3, x5
    b.hi cmp_gt_yes

cmp_gt_no:
    mov x0, #0
    ret

cmp_gt_yes:
    mov x0, #1
    ret

swap_ij:
    ldr w2, [x23, x0, lsl #2]
    ldr w3, [x24, x0, lsl #2]
    ldr w4, [x23, x1, lsl #2]
    ldr w5, [x24, x1, lsl #2]
    str w4, [x23, x0, lsl #2]
    str w5, [x24, x0, lsl #2]
    str w2, [x23, x1, lsl #2]
    str w3, [x24, x1, lsl #2]
    ret

sift:
    stp x29, x30, [sp, #-64]!
    stp x19, x25, [sp, #16]
    stp x26, x27, [sp, #32]
    str x28, [sp, #48]
    mov x19, x0
    mov x25, x1

sift_loop:
    lsl x26, x19, #1
    add x26, x26, #1
    cmp x26, x25
    b.hs sift_ret
    mov x27, x19
    mov x0, x26
    mov x1, x27
    bl cmp_gt
    cbz x0, sift_skip_left
    mov x27, x26

sift_skip_left:
    add x28, x26, #1
    cmp x28, x25
    b.hs sift_check
    mov x0, x28
    mov x1, x27
    bl cmp_gt
    cbz x0, sift_check
    mov x27, x28

sift_check:
    cmp x27, x19
    b.eq sift_ret
    mov x0, x19
    mov x1, x27
    bl swap_ij
    mov x19, x27
    b sift_loop

sift_ret:
    ldr x28, [sp, #48]
    ldp x26, x27, [sp, #32]
    ldp x19, x25, [sp, #16]
    ldp x29, x30, [sp], #64
    ret

.section .rodata
nlbyte:
    .byte 10

.bss
.balign 8
inbuf:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000
