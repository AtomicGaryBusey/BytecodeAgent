.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_CAP, 65536

.section .bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537

.text
_start:
    xor ebx, ebx

read_more:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, IN_CAP
    sub rdx, rbx
    test rdx, rdx
    jz have_input
    syscall
    cmp rax, 0
    jle have_input
    add rbx, rax
    jmp read_more

have_input:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d

next_byte:
    cmp r13, rbx
    jae end_of_input
    lea rsi, [rip + inbuf]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 10
    je got_newline
    inc r13
    jmp next_byte

got_newline:
    mov r15, r13
rev_line:
    cmp r15, r12
    jbe put_nl
    dec r15
    movzx eax, byte ptr [rsi + r15]
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], al
    inc r14
    jmp rev_line

put_nl:
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], 10
    inc r14
    inc r13
    mov r12, r13
    jmp next_byte

end_of_input:
    cmp r12, rbx
    jae write_all
    lea rsi, [rip + inbuf]
    mov r15, rbx
rev_tail:
    cmp r15, r12
    jbe put_tail_nl
    dec r15
    movzx eax, byte ptr [rsi + r15]
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], al
    inc r14
    jmp rev_tail

put_tail_nl:
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r14], 10
    inc r14

write_all:
    xor r12d, r12d
write_loop:
    cmp r12, r14
    jae do_exit
    mov eax, SYS_write
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r12
    mov rdx, r14
    sub rdx, r12
    syscall
    cmp rax, 0
    jle do_exit
    add r12, rax
    jmp write_loop

do_exit:
    mov eax, SYS_exit
    xor edi, edi
    syscall
