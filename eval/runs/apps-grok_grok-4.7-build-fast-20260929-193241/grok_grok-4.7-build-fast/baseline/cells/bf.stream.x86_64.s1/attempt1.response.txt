.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ecx, ecx
    mov r13, r12
    mov r14, r12
    mov r15, r12
find_bang:
    cmp rcx, r12
    jae have_parts
    cmp byte ptr [rbx + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r13, rcx
    lea r14, [rcx + 1]
have_parts:
    lea r8, [rip + brack_stack]
    lea r9, [rip + jump_table]
    xor edx, edx
    xor ecx, ecx
scan_loop:
    cmp rcx, r13
    jae scan_done
    mov al, byte ptr [rbx + rcx]
    cmp al, 0x5b
    je scan_open
    cmp al, 0x5d
    je scan_close
    inc rcx
    jmp scan_loop
scan_open:
    cmp edx, 1000
    jae fail
    mov dword ptr [r8 + rdx*4], ecx
    inc edx
    inc rcx
    jmp scan_loop
scan_close:
    test edx, edx
    jz fail
    dec edx
    mov eax, dword ptr [r8 + rdx*4]
    lea esi, [rcx + 1]
    mov dword ptr [r9 + rax*4], esi
    lea esi, [rax + 1]
    mov dword ptr [r9 + rcx*4], esi
    inc rcx
    jmp scan_loop
scan_done:
    test edx, edx
    jnz fail
    lea r8, [rip + dispatch]
    lea rax, [rip + cmd_next]
    mov rsi, r8
    mov ecx, 256
fill_dispatch:
    mov qword ptr [rsi], rax
    add rsi, 8
    dec ecx
    jnz fill_dispatch
    lea rdx, [rip + cmd_plus]
    mov eax, 43
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_comma]
    mov eax, 44
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_minus]
    mov eax, 45
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_dot]
    mov eax, 46
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_left]
    mov eax, 60
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_right]
    mov eax, 62
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_open]
    mov eax, 91
    mov qword ptr [r8 + rax*8], rdx
    lea rdx, [rip + cmd_close]
    mov eax, 93
    mov qword ptr [r8 + rax*8], rdx
    lea r10, [rip + tape]
    xor r12, r12
exec_loop:
    cmp r12, r13
    jae success
    movzx eax, byte ptr [rbx + r12]
    jmp qword ptr [r8 + rax*8]
cmd_plus:
    inc byte ptr [r10]
    jmp cmd_next
cmd_minus:
    dec byte ptr [r10]
    jmp cmd_next
cmd_right:
    inc r10
    jmp cmd_next
cmd_left:
    dec r10
    jmp cmd_next
cmd_open:
    cmp byte ptr [r10], 0
    jnz cmd_next
    mov r12d, dword ptr [r9 + r12*4]
    jmp exec_loop
cmd_close:
    cmp byte ptr [r10], 0
    jz cmd_next
    mov r12d, dword ptr [r9 + r12*4]
    jmp exec_loop
cmd_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov edx, 1
    syscall
    jmp cmd_next
cmd_comma:
    cmp r14, r15
    jae cmd_comma_eof
    mov al, byte ptr [rbx + r14]
    mov byte ptr [r10], al
    inc r14
    jmp cmd_next
cmd_comma_eof:
    mov byte ptr [r10], 0
    jmp cmd_next
cmd_next:
    inc r12
    jmp exec_loop
success:
    mov eax, 60
    xor edi, edi
    syscall
fail:
    mov eax, 60
    mov edi, 1
    syscall
.bss
.balign 8
input_buf:
    .zero 65536
tape:
    .zero 30000
jump_table:
    .zero 262144
brack_stack:
    .zero 4000
dispatch:
    .zero 2048
