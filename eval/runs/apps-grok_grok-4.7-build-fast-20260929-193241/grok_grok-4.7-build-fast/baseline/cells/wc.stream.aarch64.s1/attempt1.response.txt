.text
.align 2
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x26, buf
    add x26, x26, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x26
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le finish
    add x21, x21, x0
    mov x23, x0
    mov x24, x26

scan_loop:
    cbz x23, read_loop
    ldrb w0, [x24], #1
    sub x23, x23, #1
    cmp w0, #0x0A
    b.ne 1f
    add x19, x19, #1
1:
    cmp w0, #0x20
    b.eq is_space
    sub w3, w0, #0x09
    cmp w3, #4
    b.hi word_byte

is_space:
    mov x22, #0
    b scan_loop

word_byte:
    cbnz x22, scan_loop
    mov x22, #1
    add x20, x20, #1
    b scan_loop

finish:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x25, x1
    mov x0, x19
    bl append_u64
    mov w2, #0x20
    strb w2, [x1], #1
    mov x0, x20
    bl append_u64
    mov w2, #0x20
    strb w2, [x1], #1
    mov x0, x21
    bl append_u64
    mov w2, #0x0A
    strb w2, [x1], #1
    sub x2, x1, x25
    mov x0, #1
    mov x1, x25
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

hang:
    b hang

// x0 = unsigned value, x1 = destination; on return x1 points past the digits
append_u64:
    cbnz x0, append_digits
    mov w2, #0x30
    strb w2, [x1], #1
    ret

append_digits:
    mov x4, #10
    sub sp, sp, #32
    add x3, sp, #32

append_div:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #0x30
    sub x3, x3, #1
    strb w6, [x3]
    mov x0, x5
    cbnz x0, append_div
    add x4, sp, #32

append_copy:
    ldrb w6, [x3], #1
    strb w6, [x1], #1
    cmp x3, x4
    b.ne append_copy
    add sp, sp, #32
    ret

.bss
.align 3
buf:
    .zero 4096
outbuf:
    .zero 64
