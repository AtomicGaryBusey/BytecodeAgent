.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.globl _start

.text

_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0

read_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x23, x19
    b.hs scan_done
    ldrb w0, [x20, x23]
    add x23, x23, #1
    cmp w0, #10
    b.ne scan_ws
    add x21, x21, #1

scan_ws:
    cmp w0, #32
    b.eq scan_space
    sub w1, w0, #9
    cmp w1, #4
    b.ls scan_space
    cbnz x24, scan_loop
    mov x24, #1
    add x22, x22, #1
    b scan_loop

scan_space:
    mov x24, #0
    b scan_loop

scan_done:
    mov x25, #0
    mov x0, x21
    bl utoa
    mov w0, #32
    bl append_byte
    mov x0, x22
    bl utoa
    mov w0, #32
    bl append_byte
    mov x0, x19
    bl utoa
    mov w0, #10
    bl append_byte

    mov x0, #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x2, x25
    mov x8, #SYS_write
    svc #0

exit_loop:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b exit_loop

utoa:
    sub sp, sp, #32
    add x2, sp, #32
    cbnz x0, utoa_nz
    sub x2, x2, #1
    mov w3, #48
    strb w3, [x2]
    mov x4, #1
    b utoa_copy

utoa_nz:
    mov x5, #10
    mov x4, #0

utoa_div:
    cbz x0, utoa_copy
    udiv x6, x0, x5
    msub x3, x6, x5, x0
    add w3, w3, #48
    sub x2, x2, #1
    strb w3, [x2]
    mov x0, x6
    add x4, x4, #1
    b utoa_div

utoa_copy:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, x25

utoa_copy_loop:
    cbz x4, utoa_ret
    ldrb w3, [x2], #1
    strb w3, [x1], #1
    sub x4, x4, #1
    add x25, x25, #1
    b utoa_copy_loop

utoa_ret:
    add sp, sp, #32
    ret

append_byte:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    strb w0, [x1, x25]
    add x25, x25, #1
    ret

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 64
