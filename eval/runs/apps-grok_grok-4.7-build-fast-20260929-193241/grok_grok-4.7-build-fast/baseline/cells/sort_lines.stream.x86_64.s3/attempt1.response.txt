.text
.globl _start

compare_lines:
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9, r9
compare_bytes:
    cmp r9, r8
    jae compare_lengths
    lea rax, [rdi + r9]
    movzx eax, byte ptr [r12 + rax]
    lea r10, [rdx + r9]
    movzx r10d, byte ptr [r12 + r10]
    cmp eax, r10d
    ja compare_greater
    jb compare_less
    inc r9
    jmp compare_bytes
compare_lengths:
    cmp rsi, rcx
    ja compare_greater
    jb compare_less
    xor eax, eax
    ret
compare_greater:
    mov eax, 1
    ret
compare_less:
    mov eax, -1
    ret

compare_idx:
    mov eax, dword ptr [rbx + rdi*4]
    mov r8d, dword ptr [rbp + rdi*4]
    mov edx, dword ptr [rbx + rsi*4]
    mov ecx, dword ptr [rbp + rsi*4]
    mov rdi, rax
    mov rsi, r8
    jmp compare_lines

siftdown:
    mov r13, rsi
sd_loop:
    lea r15, [rdi + rdi + 1]
    cmp r15, r13
    jae sd_done
    push rdi
    push r15
    mov rdi, r15
    mov rsi, qword ptr [rsp + 8]
    call compare_idx
    pop r15
    pop rdi
    mov r11, rdi
    test eax, eax
    jle sd_right
    mov r11, r15
sd_right:
    lea r8, [r15 + 1]
    cmp r8, r13
    jae sd_swap_check
    push rdi
    push r11
    push r15
    push r8
    mov rdi, r8
    mov rsi, r11
    call compare_idx
    pop r8
    pop r15
    pop r11
    pop rdi
    test eax, eax
    jle sd_swap_check
    mov r11, r8
sd_swap_check:
    cmp r11, rdi
    je sd_done
    mov eax, dword ptr [rbx + rdi*4]
    mov edx, dword ptr [rbx + r11*4]
    mov dword ptr [rbx + rdi*4], edx
    mov dword ptr [rbx + r11*4], eax
    mov eax, dword ptr [rbp + rdi*4]
    mov edx, dword ptr [rbp + r11*4]
    mov dword ptr [rbp + rdi*4], edx
    mov dword ptr [rbp + r11*4], eax
    mov rdi, r11
    jmp sd_loop
sd_done:
    ret

_start:
    lea r12, [rip + input_buf]
    lea rbx, [rip + line_off]
    lea rbp, [rip + line_len]
    xor r13, r13
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov rdx, 32768
    sub rdx, r13
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14, r14
    xor r15, r15
parse_loop:
    cmp r15, r13
    jae parse_done
    mov r10, r15
scan_loop:
    cmp r15, r13
    jae final_line
    movzx eax, byte ptr [r12 + r15]
    inc r15
    cmp al, 10
    jne scan_loop
    mov rax, r15
    dec rax
    sub rax, r10
    mov dword ptr [rbx + r14*4], r10d
    mov dword ptr [rbp + r14*4], eax
    inc r14
    jmp parse_loop
final_line:
    mov rax, r13
    sub rax, r10
    mov dword ptr [rbx + r14*4], r10d
    mov dword ptr [rbp + r14*4], eax
    inc r14
parse_done:
    cmp r14, 2
    jb output_start
    mov r15, r14
    sub r15, 2
    shr r15, 1
build_loop:
    mov rdi, r15
    mov rsi, r14
    push r15
    call siftdown
    pop r15
    test r15, r15
    jz build_done
    dec r15
    jmp build_loop
build_done:
    mov r15, r14
    dec r15
extract_loop:
    test r15, r15
    jz output_start
    mov eax, dword ptr [rbx]
    mov edx, dword ptr [rbx + r15*4]
    mov dword ptr [rbx + r15*4], eax
    mov dword ptr [rbx], edx
    mov eax, dword ptr [rbp]
    mov edx, dword ptr [rbp + r15*4]
    mov dword ptr [rbp + r15*4], eax
    mov dword ptr [rbp], edx
    xor edi, edi
    mov rsi, r15
    push r15
    call siftdown
    pop r15
    dec r15
    jmp extract_loop
output_start:
    xor r15, r15
output_loop:
    cmp r15, r14
    jae do_exit
    mov r13d, dword ptr [rbx + r15*4]
    mov r10d, dword ptr [rbp + r15*4]
    test r10, r10
    jz write_newline
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + r13]
    mov rdx, r10
    syscall
write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_byte]
    mov edx, 1
    syscall
    inc r15
    jmp output_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline_byte:
    .byte 10

.bss
.align 8
input_buf:
    .zero 32768
.align 8
line_off:
    .zero 4000
.align 8
line_len:
    .zero 4000
