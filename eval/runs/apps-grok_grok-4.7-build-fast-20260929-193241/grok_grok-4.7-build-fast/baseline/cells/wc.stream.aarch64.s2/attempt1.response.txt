.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536
.equ STDIN, 0
.equ STDOUT, 1

.globl _start
.text

_start:
    mov x19, #0
    adrp x20, buf
    add x20, x20, :lo12:buf

read_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x19
    cbz x2, count_init
    mov x0, #STDIN
    add x1, x20, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le count_init
    add x19, x19, x0
    b read_loop

count_init:
    mov x21, #0
    mov x22, #0
    mov x24, #0
    mov x23, x20
    add x27, x20, x19

count_loop:
    cmp x23, x27
    b.eq format_out
    ldrb w0, [x23], #1
    cmp w0, #10
    b.ne 1f
    add x21, x21, #1
1:
    cmp w0, #32
    b.eq 2f
    sub w1, w0, #9
    cmp w1, #4
    b.ls 2f
    cbnz x24, count_loop
    mov x24, #1
    add x22, x22, #1
    b count_loop
2:
    mov x24, #0
    b count_loop

format_out:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x26, x25
    mov x0, x21
    bl utoa
    mov w0, #32
    strb w0, [x25], #1
    mov x0, x22
    bl utoa
    mov w0, #32
    strb w0, [x25], #1
    mov x0, x19
    bl utoa
    mov w0, #10
    strb w0, [x25], #1
    mov x0, #STDOUT
    mov x1, x26
    sub x2, x25, x26
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// Append unsigned decimal x0 at [x25]; x25 advanced. Clobbers x0-x5.
utoa:
    adrp x1, digs
    add x1, x1, :lo12:digs
    add x1, x1, #32
    mov x2, x1
1:
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    sub x1, x1, #1
    strb w5, [x1]
    mov x0, x4
    cbnz x0, 1b
2:
    ldrb w3, [x1], #1
    strb w3, [x25], #1
    cmp x1, x2
    b.ne 2b
    ret

.bss
.align 3
buf:
    .zero 65536
outbuf:
    .zero 64
digs:
    .zero 32
