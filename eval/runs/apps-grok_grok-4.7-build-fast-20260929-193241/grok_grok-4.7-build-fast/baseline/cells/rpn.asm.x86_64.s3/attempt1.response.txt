.text
.globl _start

_start:
  xor r15d, r15d

L_read:
  mov eax, 65536
  sub rax, r15
  jz L_ready
  mov rdx, rax
  xor eax, eax
  xor edi, edi
  lea rsi, [rip + input]
  add rsi, r15
  syscall
  test rax, rax
  jle L_ready
  add r15, rax
  jmp L_read

L_ready:
  lea rbx, [rip + input]
  mov r12, rbx
  add r12, r15
  lea r14, [rip + values]

L_line:
  cmp rbx, r12
  jae L_exit
  mov r13, r14

L_tok:
  cmp rbx, r12
  jae L_finish
  movzx eax, byte ptr [rbx]
  cmp al, 0x0a
  je L_finish
  cmp al, 0x20
  je L_skipsep
  cmp al, 0x09
  je L_skipsep
  jmp L_token

L_skipsep:
  inc rbx
  jmp L_tok

L_token:
  mov rsi, rbx

L_scan:
  cmp rbx, r12
  jae L_have
  movzx eax, byte ptr [rbx]
  cmp al, 0x20
  je L_have
  cmp al, 0x09
  je L_have
  cmp al, 0x0a
  je L_have
  inc rbx
  jmp L_scan

L_have:
  mov r8, rbx
  sub r8, rsi
  movzx eax, byte ptr [rsi]
  cmp r8, 1
  jne L_not1
  cmp al, '+'
  je L_op_add
  cmp al, '-'
  je L_op_sub
  cmp al, '*'
  je L_op_mul
  cmp al, '/'
  je L_op_div
  cmp al, '%'
  je L_op_mod
  jmp L_pos

L_not1:
  cmp al, '-'
  je L_neg
  jmp L_pos

L_pos:
  xor r10d, r10d
  cmp r8, 18
  ja L_bad
  mov rcx, rsi
  jmp L_chkdigs

L_neg:
  mov r10d, 1
  cmp r8, 2
  jb L_bad
  cmp r8, 19
  ja L_bad
  lea rcx, [rsi + 1]

L_chkdigs:
  mov r9, rcx

L_chk_loop:
  cmp r9, rbx
  jae L_chk_ok
  movzx eax, byte ptr [r9]
  sub al, '0'
  cmp al, 9
  ja L_bad
  inc r9
  jmp L_chk_loop

L_chk_ok:
  xor eax, eax

L_conv:
  cmp rcx, rbx
  jae L_conv_ok
  imul rax, rax, 10
  movzx edx, byte ptr [rcx]
  sub edx, '0'
  add rax, rdx
  inc rcx
  jmp L_conv

L_conv_ok:
  test r10d, r10d
  jz L_push
  neg rax

L_push:
  lea rdx, [rip + values_end]
  cmp r13, rdx
  jae L_bad
  mov qword ptr [r13], rax
  add r13, 8
  jmp L_tok

L_op_add:
  call L_pop2
  add rax, rcx
  jmp L_push

L_op_sub:
  call L_pop2
  sub rax, rcx
  jmp L_push

L_op_mul:
  call L_pop2
  imul rax, rcx
  jmp L_push

L_op_div:
  call L_pop2
  test rcx, rcx
  jz L_bad
  mov rdx, 1
  shl rdx, 63
  cmp rax, rdx
  jne L_div_do
  cmp rcx, -1
  je L_bad
L_div_do:
  cqo
  idiv rcx
  jmp L_push

L_op_mod:
  call L_pop2
  test rcx, rcx
  jz L_bad
  mov rdx, 1
  shl rdx, 63
  cmp rax, rdx
  jne L_mod_do
  cmp rcx, -1
  je L_bad
L_mod_do:
  cqo
  idiv rcx
  mov rax, rdx
  jmp L_push

L_finish:
  mov rax, r13
  sub rax, r14
  cmp rax, 8
  jne L_bad
  mov rax, qword ptr [r14]
  mov r10, 1
  shl r10, 63
  cmp rax, r10
  je L_print_min
  xor r8d, r8d
  test rax, rax
  jns L_print_abs
  neg rax
  mov r8d, 1

L_print_abs:
  lea rdi, [rip + outbuf + 31]
  mov byte ptr [rdi], 0x0a
  test rax, rax
  jnz L_print_loop
  dec rdi
  mov byte ptr [rdi], '0'
  jmp L_print_sign

L_print_loop:
  xor edx, edx
  mov ecx, 10
  div rcx
  add dl, '0'
  dec rdi
  mov byte ptr [rdi], dl
  test rax, rax
  jnz L_print_loop

L_print_sign:
  test r8d, r8d
  jz L_print_go
  dec rdi
  mov byte ptr [rdi], '-'

L_print_go:
  lea r9, [rip + outbuf + 32]
  mov rsi, rdi
  mov rdx, r9
  sub rdx, rsi
  mov eax, 1
  mov edi, 1
  syscall
  jmp L_nextline

L_print_min:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + min_msg]
  mov edx, 21
  syscall

L_nextline:
  cmp rbx, r12
  jae L_exit
  inc rbx
  jmp L_line

L_bad:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + err_msg]
  mov edx, 6
  syscall

L_skip_line:
  cmp rbx, r12
  jae L_exit
  movzx eax, byte ptr [rbx]
  inc rbx
  cmp al, 0x0a
  jne L_skip_line
  jmp L_line

L_exit:
  mov eax, 60
  xor edi, edi
  syscall
  jmp L_exit

L_pop2:
  mov rdx, r13
  sub rdx, r14
  cmp rdx, 16
  jb L_pop_fail
  sub r13, 16
  mov rax, qword ptr [r13]
  mov rcx, qword ptr [r13 + 8]
  ret

L_pop_fail:
  add rsp, 8
  jmp L_bad

.section .rodata
err_msg:
  .ascii "error"
  .byte 0x0a
min_msg:
  .ascii "-9223372036854775808"
  .byte 0x0a

.bss
.align 8
input:
  .zero 65536
values:
  .zero 16384
values_end:
outbuf:
  .zero 32
