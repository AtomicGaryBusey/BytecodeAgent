.equ IN_CAP, 49152
.equ OUT_CAP, 66560

.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, IN_CAP
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, IN_CAP
    jb read_loop
read_done:
    test r12, r12
    jz do_exit
    xor r13, r13
    xor r14, r14
    xor r15, r15
encode_loop:
    lea rax, [r13 + 3]
    cmp rax, r12
    ja encode_tail
    lea rsi, [rip + inbuf]
    movzx r8d, byte ptr [rsi + r13]
    movzx r9d, byte ptr [rsi + r13 + 1]
    movzx r10d, byte ptr [rsi + r13 + 2]
    add r13, 3
    mov edx, r8d
    shr edx, 2
    call emit_idx
    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov eax, r9d
    shr eax, 4
    or edx, eax
    call emit_idx
    mov edx, r9d
    and edx, 15
    shl edx, 2
    mov eax, r10d
    shr eax, 6
    or edx, eax
    call emit_idx
    mov edx, r10d
    and edx, 63
    call emit_idx
    jmp encode_loop
encode_tail:
    mov rax, r12
    sub rax, r13
    cmp rax, 1
    je tail1
    cmp rax, 2
    je tail2
    jmp finish_line
tail2:
    lea rsi, [rip + inbuf]
    movzx r8d, byte ptr [rsi + r13]
    movzx r9d, byte ptr [rsi + r13 + 1]
    mov edx, r8d
    shr edx, 2
    call emit_idx
    mov edx, r8d
    and edx, 3
    shl edx, 4
    mov eax, r9d
    shr eax, 4
    or edx, eax
    call emit_idx
    mov edx, r9d
    and edx, 15
    shl edx, 2
    call emit_idx
    mov al, 0x3D
    call emit_al
    jmp finish_line
tail1:
    lea rsi, [rip + inbuf]
    movzx r8d, byte ptr [rsi + r13]
    mov edx, r8d
    shr edx, 2
    call emit_idx
    mov edx, r8d
    and edx, 3
    shl edx, 4
    call emit_idx
    mov al, 0x3D
    call emit_al
    mov al, 0x3D
    call emit_al
finish_line:
    test r15, r15
    jz do_write
    lea rcx, [rip + outbuf]
    mov byte ptr [rcx + r14], 0x0A
    inc r14
do_write:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r14
    syscall
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit
emit_idx:
    lea rax, [rip + alphabet]
    movzx eax, byte ptr [rax + rdx]
emit_al:
    lea rcx, [rip + outbuf]
    mov byte ptr [rcx + r14], al
    inc r14
    inc r15
    cmp r15, 76
    jne emit_ret
    mov byte ptr [rcx + r14], 0x0A
    inc r14
    xor r15, r15
emit_ret:
    ret

.bss
.balign 16
inbuf:
    .zero IN_CAP
outbuf:
    .zero OUT_CAP
