.globl _start

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    cmp rax, 0
    jle finish

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne check_ws
    inc r12
check_ws:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    jbe is_ws
is_word:
    test r15, r15
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
is_ws:
    xor r15d, r15d
next_byte:
    dec rcx
    jnz byte_loop
    jmp read_loop

finish:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call append_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call append_uint
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call append_uint
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_uint:
    mov rbx, 10
    lea rcx, [rip + numtmp]
    add rcx, 24
    test rax, rax
    jnz au_conv
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
au_conv:
    mov r8, rcx
au_div:
    xor edx, edx
    div rbx
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz au_div
au_copy:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rcx
    inc rdi
    cmp rcx, r8
    jne au_copy
    ret

.bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 64
numtmp:
    .zero 32
