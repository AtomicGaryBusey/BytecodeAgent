.section .rodata
str_error_at:
.ascii "error at "
str_ok:
.ascii "ok\n"
str_nl:
.ascii "\n"
lit_null:
.ascii "null"
lit_true:
.ascii "true"
lit_false:
.ascii "false"

.text
.globl _start

_start:
    xor eax, eax
    mov qword ptr [rip + pos], rax
    call read_all
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jb fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + str_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
su_die:
    jmp su_die

read_all:
    xor ebx, ebx
ra_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz ra_done
    syscall
    cmp rax, 0
    jle ra_done
    add rbx, rax
    cmp rbx, 65536
    jb ra_loop
ra_done:
    mov qword ptr [rip + len], rbx
    ret

skip_ws:
    lea rdx, [rip + buf]
sw_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae sw_done
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, 0x20
    je sw_adv
    cmp cl, 0x09
    je sw_adv
    cmp cl, 0x0A
    je sw_adv
    cmp cl, 0x0D
    je sw_adv
    ret
sw_adv:
    inc qword ptr [rip + pos]
    jmp sw_loop
sw_done:
    ret

parse_value:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx eax, byte ptr [rdx + rax]
    cmp al, 0x22
    je parse_string
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 't'
    je pv_true
    cmp al, 'f'
    je pv_false
    cmp al, 'n'
    je pv_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp parse_number
pv_true:
    lea rdi, [rip + lit_true]
    mov esi, 4
    jmp parse_lit
pv_false:
    lea rdi, [rip + lit_false]
    mov esi, 5
    jmp parse_lit
pv_null:
    lea rdi, [rip + lit_null]
    mov esi, 4
    jmp parse_lit

parse_lit:
    xor ecx, ecx
pl_loop:
    cmp rcx, rsi
    jae pl_done
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx r11d, byte ptr [rdx + rax]
    cmp r11b, byte ptr [rdi + rcx]
    jne fail
    inc qword ptr [rip + pos]
    inc rcx
    jmp pl_loop
pl_done:
    ret

parse_string:
    inc qword ptr [rip + pos]
    lea r10, [rip + buf]
ps_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, 0x22
    je ps_end
    cmp cl, 0x5C
    je ps_esc
    cmp cl, 0x20
    jb fail
    inc qword ptr [rip + pos]
    jmp ps_loop
ps_end:
    inc qword ptr [rip + pos]
    ret
ps_esc:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, 0x22
    je ps_esc_ok
    cmp cl, 0x5C
    je ps_esc_ok
    cmp cl, 0x2F
    je ps_esc_ok
    cmp cl, 'b'
    je ps_esc_ok
    cmp cl, 'f'
    je ps_esc_ok
    cmp cl, 'n'
    je ps_esc_ok
    cmp cl, 'r'
    je ps_esc_ok
    cmp cl, 't'
    je ps_esc_ok
    cmp cl, 'u'
    je ps_u
    jmp fail
ps_esc_ok:
    inc qword ptr [rip + pos]
    jmp ps_loop
ps_u:
    inc qword ptr [rip + pos]
    mov esi, 4
ps_u_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    jbe ps_hex_ok
    cmp cl, 'A'
    jb fail
    cmp cl, 'F'
    jbe ps_hex_ok
    cmp cl, 'a'
    jb fail
    cmp cl, 'f'
    ja fail
ps_hex_ok:
    inc qword ptr [rip + pos]
    dec esi
    jnz ps_u_loop
    jmp ps_loop

parse_number:
    lea r10, [rip + buf]
    mov rax, qword ptr [rip + pos]
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '-'
    jne pn_int
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
pn_int:
    cmp cl, '0'
    jne pn_nz
    inc qword ptr [rip + pos]
    jmp pn_frac
pn_nz:
    inc qword ptr [rip + pos]
pn_nz_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae pn_done
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '0'
    jb pn_frac
    cmp cl, '9'
    ja pn_frac
    inc qword ptr [rip + pos]
    jmp pn_nz_loop
pn_frac:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae pn_done
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '.'
    jne pn_exp
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
    inc qword ptr [rip + pos]
pn_frac_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae pn_done
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '0'
    jb pn_exp
    cmp cl, '9'
    ja pn_exp
    inc qword ptr [rip + pos]
    jmp pn_frac_loop
pn_exp:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae pn_done
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, 'e'
    je pn_exp_body
    cmp cl, 'E'
    je pn_exp_body
    jmp pn_done
pn_exp_body:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '+'
    je pn_exp_sign
    cmp cl, '-'
    jne pn_exp_dig
pn_exp_sign:
    inc qword ptr [rip + pos]
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    movzx ecx, byte ptr [r10 + rax]
pn_exp_dig:
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
    inc qword ptr [rip + pos]
pn_exp_loop:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae pn_done
    movzx ecx, byte ptr [r10 + rax]
    cmp cl, '0'
    jb pn_done
    cmp cl, '9'
    ja pn_done
    inc qword ptr [rip + pos]
    jmp pn_exp_loop
pn_done:
    ret

parse_array:
    inc qword ptr [rip + pos]
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, ']'
    jne pa_elem
    inc qword ptr [rip + pos]
    ret
pa_elem:
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, ']'
    je pa_end
    cmp cl, ','
    jne fail
    inc qword ptr [rip + pos]
    call skip_ws
    jmp pa_elem
pa_end:
    inc qword ptr [rip + pos]
    ret

parse_object:
    inc qword ptr [rip + pos]
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, '}'
    jne po_member
    inc qword ptr [rip + pos]
    ret
po_member:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, 0x22
    jne fail
    call parse_string
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, ':'
    jne fail
    inc qword ptr [rip + pos]
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + len]
    jae fail
    lea rdx, [rip + buf]
    movzx ecx, byte ptr [rdx + rax]
    cmp cl, '}'
    je po_end
    cmp cl, ','
    jne fail
    inc qword ptr [rip + pos]
    call skip_ws
    jmp po_member
po_end:
    inc qword ptr [rip + pos]
    ret

fail:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + str_error_at]
    mov edx, 9
    syscall
    mov rax, qword ptr [rip + pos]
    lea rbx, [rip + numbuf]
    add rbx, 24
    mov r8, rbx
    mov ecx, 10
fl_digit:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz fl_digit
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r8
    sub rdx, rbx
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + str_nl]
    mov edx, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
fl_die:
    jmp fl_die

.bss
.balign 8
buf:
.zero 65536
len:
.zero 8
pos:
.zero 8
numbuf:
.zero 32
