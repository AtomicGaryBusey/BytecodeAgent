.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ BUF_SIZE, 65536

.section .rodata
nl:
	.byte 0x0A

.bss
.balign 8
inbuf:
	.zero BUF_SIZE

.text
.balign 4
.globl _start
_start:
	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	adrp x25, nl
	add x25, x25, :lo12:nl
	mov x20, x19
	mov x22, #BUF_SIZE
	add x22, x19, x22

read_loop:
	sub x2, x22, x20
	cbz x2, read_done
	mov x0, #STDIN
	mov x1, x20
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	b.le read_done
	add x20, x20, x0
	b read_loop

read_done:
	mov x21, x19

line_loop:
	cmp x21, x20
	b.hs exit_ok
	mov x23, x21

scan:
	cmp x21, x20
	b.hs reverse
	ldrb w0, [x21]
	cmp w0, #0x0A
	b.eq reverse
	add x21, x21, #1
	b scan

reverse:
	mov x24, x23
	mov x26, x21

rev_loop:
	cmp x24, x26
	b.hs rev_done
	sub x26, x26, #1
	cmp x24, x26
	b.hs rev_done
	ldrb w0, [x24]
	ldrb w1, [x26]
	strb w1, [x24]
	strb w0, [x26]
	add x24, x24, #1
	b rev_loop

rev_done:
	sub x2, x21, x23
	cbz x2, write_nl
	mov x0, #STDOUT
	mov x1, x23
	mov x8, #SYS_write
	svc #0

write_nl:
	mov x0, #STDOUT
	mov x1, x25
	mov x2, #1
	mov x8, #SYS_write
	svc #0
	cmp x21, x20
	b.hs exit_ok
	add x21, x21, #1
	b line_loop

exit_ok:
	mov x0, #0
	mov x8, #SYS_exit
	svc #0
	b exit_ok
