.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .rodata
.p2align 3
hextab:
    .ascii "0123456789abcdef"

.bss
.p2align 3
inbuf:
    .zero 65536
linebuf:
    .zero 128

.text
.p2align 2

// x0 = value, x1 = dest; advances x1 by 8. Clobbers x0, x2, x3, x4.
hex8:
    adrp x2, hextab
    add x2, x2, :lo12:hextab
    mov x4, #8
1:
    lsr x3, x0, #28
    and x3, x3, #15
    ldrb w3, [x2, x3]
    strb w3, [x1], #1
    lsl x0, x0, #4
    subs x4, x4, #1
    b.ne 1b
    ret

// w0 = byte, x1 = dest; advances x1 by 2. Clobbers x2, x3, x4.
hexb:
    adrp x2, hextab
    add x2, x2, :lo12:hextab
    lsr w3, w0, #4
    and w4, w0, #15
    ldrb w3, [x2, x3]
    strb w3, [x1], #1
    ldrb w4, [x2, x4]
    strb w4, [x1], #1
    ret

_start:
    mov x19, #0
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0
row_loop:
    cmp x20, x19
    b.hs final_line
    sub x21, x19, x20
    cmp x21, #16
    b.ls 2f
    mov x21, #16
2:
    adrp x1, linebuf
    add x1, x1, :lo12:linebuf
    mov x0, x20
    bl hex8
    mov w3, #32
    strb w3, [x1], #1
    strb w3, [x1], #1
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    add x22, x22, x20
    mov x23, #0
byte_loop:
    cmp x23, #8
    b.ne 3f
    mov w3, #32
    strb w3, [x1], #1
3:
    cmp x23, x21
    b.hs 4f
    ldrb w0, [x22, x23]
    bl hexb
    mov w3, #32
    strb w3, [x1], #1
    b 5f
4:
    mov w3, #32
    strb w3, [x1], #1
    strb w3, [x1], #1
    strb w3, [x1], #1
5:
    add x23, x23, #1
    cmp x23, #16
    b.ne byte_loop
    mov w3, #32
    strb w3, [x1], #1
    mov w3, #124
    strb w3, [x1], #1
    mov x23, #0
char_loop:
    cmp x23, x21
    b.hs 6f
    ldrb w0, [x22, x23]
    cmp w0, #32
    b.lo 7f
    cmp w0, #126
    b.hi 7f
    strb w0, [x1], #1
    b 8f
7:
    mov w3, #46
    strb w3, [x1], #1
8:
    add x23, x23, #1
    b char_loop
6:
    mov w3, #124
    strb w3, [x1], #1
    mov w3, #10
    strb w3, [x1], #1
    adrp x3, linebuf
    add x3, x3, :lo12:linebuf
    sub x2, x1, x3
    mov x1, x3
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    add x20, x20, x21
    b row_loop

final_line:
    adrp x1, linebuf
    add x1, x1, :lo12:linebuf
    mov x0, x19
    bl hex8
    mov w3, #10
    strb w3, [x1], #1
    adrp x3, linebuf
    add x3, x3, :lo12:linebuf
    sub x2, x1, x3
    mov x1, x3
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
