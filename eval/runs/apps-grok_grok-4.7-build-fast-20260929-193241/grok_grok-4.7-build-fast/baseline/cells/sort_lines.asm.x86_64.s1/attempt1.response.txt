.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_CAP, 32768
.equ NL, 10

.section .rodata
nlbyte:
	.byte NL

.bss
.balign 8
input_buf:
	.zero 32768
.balign 8
line_off:
	.zero 8000
line_len:
	.zero 4000
order:
	.zero 4000

.text
.globl _start

compare_lines:
	lea rcx, [rip + line_off]
	mov r8, qword ptr [rcx + rdi*8]
	mov r9, qword ptr [rcx + rsi*8]
	lea rcx, [rip + line_len]
	mov r10d, dword ptr [rcx + rdi*4]
	mov r11d, dword ptr [rcx + rsi*4]
	lea rcx, [rip + input_buf]
	add r8, rcx
	add r9, rcx
	mov ecx, r10d
	cmp ecx, r11d
	cmova ecx, r11d
	xor eax, eax
	test ecx, ecx
	jz cmp_by_len
cmp_bytes:
	movzx edx, byte ptr [r8 + rax]
	cmp dl, byte ptr [r9 + rax]
	ja cmp_greater
	jb cmp_less
	inc eax
	cmp eax, ecx
	jb cmp_bytes
cmp_by_len:
	cmp r10d, r11d
	ja cmp_greater
	jb cmp_less
	xor eax, eax
	ret
cmp_greater:
	mov eax, 1
	ret
cmp_less:
	mov eax, -1
	ret

sift_down:
sift_loop:
	mov r8, rdi
	shl r8, 1
	inc r8
	cmp r8, rsi
	jae sift_done
	mov r10, rdi
	push rdi
	push rsi
	push r8
	push r10
	lea rcx, [rip + order]
	mov edi, dword ptr [rcx + r8*4]
	mov esi, dword ptr [rcx + r10*4]
	call compare_lines
	pop r10
	pop r8
	pop rsi
	pop rdi
	cmp eax, 1
	jne sift_left_no
	mov r10, r8
sift_left_no:
	lea r9, [r8 + 1]
	cmp r9, rsi
	jae sift_right_no
	push rdi
	push rsi
	push r8
	push r9
	push r10
	lea rcx, [rip + order]
	mov edi, dword ptr [rcx + r9*4]
	mov esi, dword ptr [rcx + r10*4]
	call compare_lines
	pop r10
	pop r9
	pop r8
	pop rsi
	pop rdi
	cmp eax, 1
	jne sift_right_no
	mov r10, r9
sift_right_no:
	cmp r10, rdi
	je sift_done
	lea rcx, [rip + order]
	mov eax, dword ptr [rcx + rdi*4]
	mov edx, dword ptr [rcx + r10*4]
	mov dword ptr [rcx + rdi*4], edx
	mov dword ptr [rcx + r10*4], eax
	mov rdi, r10
	jmp sift_loop
sift_done:
	ret

_start:
	xor ebx, ebx
read_loop:
	mov eax, SYS_read
	xor edi, edi
	lea rsi, [rip + input_buf]
	add rsi, rbx
	mov edx, BUF_CAP
	sub rdx, rbx
	test rdx, rdx
	jz read_done
	syscall
	cmp rax, 0
	jle read_done
	add rbx, rax
	jmp read_loop
read_done:
	xor ecx, ecx
	xor r8d, r8d
	xor r9d, r9d
parse_loop:
	cmp rcx, rbx
	jae parse_after
	lea rdx, [rip + input_buf]
	movzx eax, byte ptr [rdx + rcx]
	cmp al, NL
	je parse_nl
	inc rcx
	jmp parse_loop
parse_nl:
	mov rax, rcx
	sub rax, r8
	lea rdx, [rip + line_off]
	mov qword ptr [rdx + r9*8], r8
	lea rdx, [rip + line_len]
	mov dword ptr [rdx + r9*4], eax
	lea rdx, [rip + order]
	mov dword ptr [rdx + r9*4], r9d
	inc r9
	inc rcx
	mov r8, rcx
	jmp parse_loop
parse_after:
	cmp r8, rbx
	jae parse_have_n
	mov rax, rbx
	sub rax, r8
	lea rdx, [rip + line_off]
	mov qword ptr [rdx + r9*8], r8
	lea rdx, [rip + line_len]
	mov dword ptr [rdx + r9*4], eax
	lea rdx, [rip + order]
	mov dword ptr [rdx + r9*4], r9d
	inc r9
parse_have_n:
	mov rbx, r9
	cmp rbx, 2
	jb print_start
	mov rdi, rbx
	shr rdi, 1
	dec rdi
build_loop:
	mov rsi, rbx
	push rdi
	call sift_down
	pop rdi
	test rdi, rdi
	jz extract_init
	dec rdi
	jmp build_loop
extract_init:
	mov r12, rbx
	dec r12
extract_loop:
	test r12, r12
	jz print_start
	lea rcx, [rip + order]
	mov eax, dword ptr [rcx]
	mov edx, dword ptr [rcx + r12*4]
	mov dword ptr [rcx], edx
	mov dword ptr [rcx + r12*4], eax
	xor edi, edi
	mov rsi, r12
	call sift_down
	dec r12
	jmp extract_loop
print_start:
	xor r13d, r13d
print_loop:
	cmp r13, rbx
	jae do_exit
	lea rcx, [rip + order]
	mov eax, dword ptr [rcx + r13*4]
	lea rcx, [rip + line_off]
	mov rsi, qword ptr [rcx + rax*8]
	lea rcx, [rip + input_buf]
	add rsi, rcx
	lea rcx, [rip + line_len]
	mov edx, dword ptr [rcx + rax*4]
	test edx, edx
	jz print_nl
	mov eax, SYS_write
	mov edi, 1
	syscall
print_nl:
	mov eax, SYS_write
	mov edi, 1
	lea rsi, [rip + nlbyte]
	mov edx, 1
	syscall
	inc r13
	jmp print_loop
do_exit:
	mov eax, SYS_exit
	xor edi, edi
	syscall
	jmp do_exit
