.section .rodata
.align 3
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.align 2
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #0xc000
    sub x2, x2, x20
    cbz x2, encode
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le encode
    add x20, x20, x0
    b read_loop

encode:
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x21, #0
    mov x23, #0
    mov x24, #0
    cbz x20, do_exit

enc_loop:
    cmp x21, x20
    b.hs finish_line

    sub x25, x20, x21
    add x9, x19, x21
    ldrb w0, [x9]
    mov w1, #0
    mov w2, #0

    cmp x25, #2
    b.lo one_byte
    ldrb w1, [x9, #1]
    cmp x25, #3
    b.lo two_bytes
    ldrb w2, [x9, #2]
    add x21, x21, #3
    mov w26, #3
    b have_bytes

two_bytes:
    add x21, x21, #2
    mov w26, #2
    b have_bytes

one_byte:
    add x21, x21, #1
    mov w26, #1

have_bytes:
    lsr w3, w0, #2

    and w4, w0, #3
    lsl w4, w4, #4
    lsr w7, w1, #4
    orr w4, w4, w7

    and w5, w1, #15
    lsl w5, w5, #2
    lsr w7, w2, #6
    orr w5, w5, w7

    and w6, w2, #63

    bl emit_idx
    mov w3, w4
    bl emit_idx

    cmp w26, #2
    b.lo pad2
    mov w3, w5
    bl emit_idx
    b third
pad2:
    mov w3, #61
    bl emit_char
third:
    cmp w26, #3
    b.lo pad3
    mov w3, w6
    bl emit_idx
    b enc_loop
pad3:
    mov w3, #61
    bl emit_char
    b enc_loop

finish_line:
    cbz x24, do_write
    mov w3, #10
    strb w3, [x22, x23]
    add x23, x23, #1

do_write:
    mov x25, #0
write_loop:
    cmp x25, x23
    b.hs do_exit
    mov x0, #1
    add x1, x22, x25
    sub x2, x23, x25
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le do_exit
    add x25, x25, x0
    b write_loop

do_exit:
    mov x0, #0
    mov x8, #93
    svc #0
    b do_exit

emit_idx:
    adrp x7, alpha
    add x7, x7, :lo12:alpha
    add x7, x7, x3
    ldrb w3, [x7]

emit_char:
    strb w3, [x22, x23]
    add x23, x23, #1
    add x24, x24, #1
    cmp x24, #76
    b.ne 1f
    mov w7, #10
    strb w7, [x22, x23]
    add x23, x23, #1
    mov x24, #0
1:
    ret

.bss
.align 3
inbuf:
    .zero 49152
outbuf:
    .zero 200000
