.equ BUFSIZE, 65536

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 16
inbuf:
    .zero BUFSIZE

.text
.globl _start
_start:
    lea r15, [rip + inbuf]
    xor ebx, ebx

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + rbx]
    mov edx, BUFSIZE
    sub rdx, rbx
    syscall
    test rax, rax
    jle have_input
    add rbx, rax
    cmp rbx, BUFSIZE
    jb read_more

have_input:
    xor r12, r12

next_line:
    cmp r12, rbx
    jae finish
    mov r13, r12
    mov r14, r12

scan:
    cmp r14, rbx
    jae got_line
    movzx eax, byte ptr [r15 + r14]
    cmp al, 0x0A
    je got_line
    inc r14
    jmp scan

got_line:
    cmp r13, r14
    je rev_done
    mov rsi, r13
    mov rdi, r14
    dec rdi

rev:
    cmp rsi, rdi
    jae rev_done
    mov al, byte ptr [r15 + rsi]
    mov cl, byte ptr [r15 + rdi]
    mov byte ptr [r15 + rsi], cl
    mov byte ptr [r15 + rdi], al
    inc rsi
    dec rdi
    jmp rev

rev_done:
    mov rdx, r14
    sub rdx, r13
    test rdx, rdx
    jz write_nl
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + r13]
    syscall

write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    cmp r14, rbx
    jae finish
    lea r12, [r14 + 1]
    jmp next_line

finish:
    mov eax, 60
    xor edi, edi
    syscall
