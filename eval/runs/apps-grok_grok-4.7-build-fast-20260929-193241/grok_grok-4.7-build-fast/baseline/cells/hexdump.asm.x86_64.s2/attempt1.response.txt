.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUFSIZE, 65536

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.section .bss
.balign 16
inbuf:
    .zero 65536
linebuf:
    .zero 128

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, BUFSIZE
    jae read_done
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, BUFSIZE
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12d, r12d
row_loop:
    cmp r12, rbx
    jae rows_done
    mov r14, rbx
    sub r14, r12
    cmp r14, 16
    jbe row_n
    mov r14, 16
row_n:
    lea r13, [rip + linebuf]
    mov rax, r12
    call append_hex32
    mov al, ' '
    call append_char
    mov al, ' '
    call append_char
    xor r15d, r15d
hex_loop:
    cmp r15, 16
    jae hex_done
    cmp r15, r14
    jae hex_pad
    lea rsi, [rip + inbuf]
    mov rax, r12
    add rax, r15
    mov al, byte ptr [rsi + rax]
    call append_hex_byte
    mov al, ' '
    call append_char
    jmp hex_mid
hex_pad:
    mov al, ' '
    call append_char
    mov al, ' '
    call append_char
    mov al, ' '
    call append_char
hex_mid:
    cmp r15, 7
    jne hex_next
    mov al, ' '
    call append_char
hex_next:
    inc r15
    jmp hex_loop
hex_done:
    mov al, ' '
    call append_char
    mov al, '|'
    call append_char
    xor r15d, r15d
asc_loop:
    cmp r15, r14
    jae asc_done
    lea rsi, [rip + inbuf]
    mov rax, r12
    add rax, r15
    mov al, byte ptr [rsi + rax]
    cmp al, 0x20
    jb asc_dot
    cmp al, 0x7e
    ja asc_dot
    call append_char
    jmp asc_next
asc_dot:
    mov al, '.'
    call append_char
asc_next:
    inc r15
    jmp asc_loop
asc_done:
    mov al, '|'
    call append_char
    mov al, 10
    call append_char
    call write_line
    add r12, r14
    jmp row_loop
rows_done:
    lea r13, [rip + linebuf]
    mov rax, rbx
    call append_hex32
    mov al, 10
    call append_char
    call write_line
    mov eax, SYS_exit
    xor edi, edi
    syscall

write_line:
    lea rsi, [rip + linebuf]
    mov rdx, r13
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall
    ret

append_char:
    mov byte ptr [r13], al
    inc r13
    ret

append_hex_byte:
    movzx eax, al
    mov esi, eax
    shr eax, 4
    lea rdx, [rip + hextab]
    mov al, byte ptr [rdx + rax]
    mov byte ptr [r13], al
    inc r13
    and esi, 15
    mov al, byte ptr [rdx + rsi]
    mov byte ptr [r13], al
    inc r13
    ret

append_hex32:
    mov rsi, rax
    shl rsi, 32
    lea rdx, [rip + hextab]
    mov ecx, 8
ah32_loop:
    rol rsi, 4
    mov eax, esi
    and eax, 15
    mov al, byte ptr [rdx + rax]
    mov byte ptr [r13], al
    inc r13
    dec ecx
    jnz ah32_loop
    ret
