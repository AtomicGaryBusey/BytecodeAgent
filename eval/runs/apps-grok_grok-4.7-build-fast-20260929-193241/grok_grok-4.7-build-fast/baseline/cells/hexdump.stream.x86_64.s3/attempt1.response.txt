.globl _start

.text

_start:
    xor r12, r12
    lea r15, [rip + inbuf]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r15
    mov edx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    add r15, rax
    jmp read_loop

read_done:
    lea rbx, [rip + inbuf]
    lea r13, [rip + hexdigits]
    xor r14, r14

row_loop:
    cmp r14, r12
    jae final_line

    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_n_ok
    mov r15, 16
row_n_ok:
    lea rdi, [rip + outbuf]
    mov eax, r14d
    call hex8

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2

    xor ecx, ecx
slot_loop:
    cmp ecx, 8
    jne slot_no_gap
    mov byte ptr [rdi], 32
    inc rdi
slot_no_gap:
    cmp rcx, r15
    jae slot_pad
    mov rax, r14
    add rax, rcx
    movzx eax, byte ptr [rbx + rax]
    mov edx, eax
    shr eax, 4
    and edx, 15
    movzx eax, byte ptr [r13 + rax]
    movzx edx, byte ptr [r13 + rdx]
    mov byte ptr [rdi], al
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp slot_advance
slot_pad:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
slot_advance:
    inc ecx
    cmp ecx, 16
    jb slot_loop

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2

    xor ecx, ecx
ascii_loop:
    cmp rcx, r15
    jae ascii_done
    mov rax, r14
    add rax, rcx
    movzx eax, byte ptr [rbx + rax]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_store
ascii_dot:
    mov byte ptr [rdi], 46
ascii_store:
    inc rdi
    inc ecx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r14, r15
    jmp row_loop

final_line:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10

    lea rsi, [rip + outbuf]
    mov edx, 9
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

# eax = value, rdi = dest; rdi advances by 8. Clobbers rax, rcx, rdx, rsi.
hex8:
    lea rsi, [rip + hexdigits]
    mov ecx, 8
hex8_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz hex8_loop
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 65536
outbuf:
    .zero 128
