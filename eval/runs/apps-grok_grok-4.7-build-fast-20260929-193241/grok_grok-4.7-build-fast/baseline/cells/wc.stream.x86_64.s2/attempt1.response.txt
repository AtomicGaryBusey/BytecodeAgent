.globl _start

.section .bss
.balign 8
buffer:
    .zero 65536
obuf:
    .zero 32

.section .rodata
spchr:
    .byte 0x20
nlchr:
    .byte 0x0A

.text
_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    xor ebx, ebx
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rsi, [rip + buffer]
count_loop:
    cmp rbx, r12
    jae count_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x0A
    jne no_nl
    inc r13
no_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    jbe is_ws
is_word:
    test r15, r15
    jnz next_byte
    inc r14
    mov r15d, 1
    jmp next_byte
is_ws:
    xor r15d, r15d
next_byte:
    inc rbx
    jmp count_loop
count_done:
    mov rax, r13
    call print_u64
    call print_sp
    mov rax, r14
    call print_u64
    call print_sp
    mov rax, r12
    call print_u64
    call print_nl
    mov eax, 60
    xor edi, edi
    syscall

print_sp:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + spchr]
    mov edx, 1
    syscall
    ret

print_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nlchr]
    mov edx, 1
    syscall
    ret

print_u64:
    lea rcx, [rip + obuf]
    add rcx, 32
    mov r8, rcx
    mov r9d, 10
digit_loop:
    xor edx, edx
    div r9
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz digit_loop
    mov eax, 1
    mov edi, 1
    mov rsi, rcx
    mov rdx, r8
    sub rdx, rcx
    syscall
    ret
