.globl _start

.bss
.align 16
inbuf:
.zero 4096
outbuf:
.zero 96
digbuf:
.zero 32

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov rdx, 4096
    syscall
    test rax, rax
    jle done_read

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + inbuf]

scan_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne check_ws
    inc r12
check_ws:
    cmp al, 32
    je mark_ws
    cmp al, 9
    jb mark_word
    cmp al, 13
    jbe mark_ws
mark_word:
    test r15, r15
    jnz scan_loop
    mov r15, 1
    inc r13
    jmp scan_loop
mark_ws:
    xor r15, r15
    jmp scan_loop

done_read:
    lea rbx, [rip + outbuf]
    mov rdi, r12
    call append_uint
    mov byte ptr [rbx], 32
    inc rbx
    mov rdi, r13
    call append_uint
    mov byte ptr [rbx], 32
    inc rbx
    mov rdi, r14
    call append_uint
    mov byte ptr [rbx], 10
    inc rbx

    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor edi, edi
    syscall

append_uint:
    lea rcx, [rip + digbuf]
    add rcx, 31
    mov rsi, rcx
    mov rax, rdi
    mov r8, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    cmp rsi, rcx
    jae 3f
    mov al, byte ptr [rsi]
    mov byte ptr [rbx], al
    inc rsi
    inc rbx
    jmp 2b
3:
    ret
