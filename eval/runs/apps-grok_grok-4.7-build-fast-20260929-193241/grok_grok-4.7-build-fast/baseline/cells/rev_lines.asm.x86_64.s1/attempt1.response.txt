.globl _start

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 16
inbuf:
    .zero 65536

.text
_start:
    xor rbx, rbx

read_loop:
    cmp rbx, 65536
    jae have_input
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle have_input
    add rbx, rax
    jmp read_loop

have_input:
    lea r15, [rip + inbuf]
    xor r12, r12

next_line:
    cmp r12, rbx
    jae quit
    mov r13, r12

scan:
    cmp r13, rbx
    jae last_line
    movzx eax, byte ptr [r15 + r13]
    cmp al, 0x0A
    je got_nl
    inc r13
    jmp scan

got_nl:
    mov r14, r13
    call emit_rev
    lea r12, [r13 + 1]
    jmp next_line

last_line:
    mov r14, rbx
    call emit_rev

quit:
    mov eax, 60
    xor edi, edi
    syscall

emit_rev:
    cmp r12, r14
    jae emit_nl
    mov rcx, r12
    mov rdx, r14
    dec rdx
rev:
    cmp rcx, rdx
    jae rev_write
    mov al, byte ptr [r15 + rcx]
    mov r8b, byte ptr [r15 + rdx]
    mov byte ptr [r15 + rcx], r8b
    mov byte ptr [r15 + rdx], al
    inc rcx
    dec rdx
    jmp rev
rev_write:
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + r12]
    mov rdx, r14
    sub rdx, r12
    syscall
emit_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    ret
