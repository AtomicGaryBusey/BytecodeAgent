.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_CAP, 32768

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 8
buf:
    .space 32768
.balign 8
line_off:
    .space 8000
.balign 8
line_len:
    .space 8000

.text
.globl _start
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
    mov x26, #BUF_CAP
read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x26, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, x26
    b.lo read_loop
read_done:
    adrp x24, line_off
    add x24, x24, :lo12:line_off
    adrp x25, line_len
    add x25, x25, :lo12:line_len
    mov x21, #0
    mov x22, #0
    mov x23, #0
parse_loop:
    cmp x22, x20
    b.hs parse_done
    ldrb w0, [x19, x22]
    add x22, x22, #1
    cmp w0, #10
    b.ne parse_loop
    sub x0, x22, x23
    sub x0, x0, #1
    str x23, [x24, x21, lsl #3]
    str x0, [x25, x21, lsl #3]
    add x21, x21, #1
    mov x23, x22
    b parse_loop
parse_done:
    cmp x23, x20
    b.hs parse_no_final
    sub x0, x20, x23
    str x23, [x24, x21, lsl #3]
    str x0, [x25, x21, lsl #3]
    add x21, x21, #1
parse_no_final:
    mov x0, x21
    bl heapsort
    mov x22, #0
out_loop:
    cmp x22, x21
    b.hs out_done
    ldr x3, [x24, x22, lsl #3]
    ldr x4, [x25, x22, lsl #3]
    add x1, x19, x3
    cbz x4, out_nl
    mov x0, #1
    mov x2, x4
    mov x8, #SYS_write
    svc #0
out_nl:
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x22, x22, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b out_done

heapsort:
    stp x30, x19, [sp, #-16]!
    str x20, [sp, #-16]!
    mov x19, x0
    cmp x19, #2
    b.lo hs_done
    lsr x20, x19, #1
    sub x20, x20, #1
hs_build:
    mov x0, x20
    mov x1, x19
    bl sift_down
    cbz x20, hs_extract_init
    sub x20, x20, #1
    b hs_build
hs_extract_init:
    sub x20, x19, #1
hs_extract:
    mov x0, #0
    mov x1, x20
    bl swap
    mov x0, #0
    mov x1, x20
    bl sift_down
    sub x20, x20, #1
    cbnz x20, hs_extract
hs_done:
    ldr x20, [sp], #16
    ldp x30, x19, [sp], #16
    ret

sift_down:
    stp x30, x19, [sp, #-16]!
    stp x20, x21, [sp, #-16]!
    stp x22, x23, [sp, #-16]!
    mov x19, x0
    mov x20, x1
sift_loop:
    lsl x21, x19, #1
    add x21, x21, #1
    cmp x21, x20
    b.hs sift_done
    mov x22, x19
    mov x0, x21
    mov x1, x22
    bl compare
    cmp x0, #0
    b.le sift_skip_left
    mov x22, x21
sift_skip_left:
    add x23, x21, #1
    cmp x23, x20
    b.hs sift_check
    mov x0, x23
    mov x1, x22
    bl compare
    cmp x0, #0
    b.le sift_check
    mov x22, x23
sift_check:
    cmp x22, x19
    b.eq sift_done
    mov x0, x19
    mov x1, x22
    bl swap
    mov x19, x22
    b sift_loop
sift_done:
    ldp x22, x23, [sp], #16
    ldp x20, x21, [sp], #16
    ldp x30, x19, [sp], #16
    ret

compare:
    adrp x2, line_off
    add x2, x2, :lo12:line_off
    adrp x3, line_len
    add x3, x3, :lo12:line_len
    ldr x4, [x2, x0, lsl #3]
    ldr x5, [x2, x1, lsl #3]
    ldr x6, [x3, x0, lsl #3]
    ldr x7, [x3, x1, lsl #3]
    adrp x2, buf
    add x2, x2, :lo12:buf
    add x4, x2, x4
    add x5, x2, x5
    cmp x6, x7
    csel x8, x6, x7, lo
    mov x9, #0
cmp_loop:
    cmp x9, x8
    b.hs cmp_lens
    ldrb w10, [x4, x9]
    ldrb w11, [x5, x9]
    cmp w10, w11
    b.lo cmp_less
    b.hi cmp_greater
    add x9, x9, #1
    b cmp_loop
cmp_lens:
    cmp x6, x7
    b.lo cmp_less
    b.hi cmp_greater
    mov x0, #0
    ret
cmp_less:
    mov x0, #-1
    ret
cmp_greater:
    mov x0, #1
    ret

swap:
    adrp x2, line_off
    add x2, x2, :lo12:line_off
    ldr x3, [x2, x0, lsl #3]
    ldr x4, [x2, x1, lsl #3]
    str x4, [x2, x0, lsl #3]
    str x3, [x2, x1, lsl #3]
    adrp x2, line_len
    add x2, x2, :lo12:line_len
    ldr x3, [x2, x0, lsl #3]
    ldr x4, [x2, x1, lsl #3]
    str x4, [x2, x0, lsl #3]
    str x3, [x2, x1, lsl #3]
    ret
