.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INSIZE, 65536

.section .rodata
.align 2
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 3
inbuf:
    .zero INSIZE
outbuf:
    .zero 96

.text
.align 2
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    adrp x27, outbuf
    add x27, x27, :lo12:outbuf
    adrp x25, hexdigits
    add x25, x25, :lo12:hexdigits
    movz x26, #1, lsl #16
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x20, x19
    sub x2, x26, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    cmp x19, x26
    b.lo read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x19
    b.hs final_line
    sub x22, x19, x21
    cmp x22, #16
    b.ls have_n
    mov x22, #16
have_n:
    mov x23, x27
    mov x0, x21
    bl put_u32_hex
    mov w0, #32
    strb w0, [x23], #1
    strb w0, [x23], #1

    mov x24, #0
hex_loop:
    cmp x24, x22
    b.hs hex_pad
    add x1, x20, x21
    ldrb w0, [x1, x24]
    bl put_hex2
    mov w0, #32
    strb w0, [x23], #1
    b hex_slot_done
hex_pad:
    mov w0, #32
    strb w0, [x23], #1
    strb w0, [x23], #1
    strb w0, [x23], #1
hex_slot_done:
    cmp x24, #7
    b.ne hex_no_gap
    mov w0, #32
    strb w0, [x23], #1
hex_no_gap:
    add x24, x24, #1
    cmp x24, #16
    b.lo hex_loop

    mov w0, #32
    strb w0, [x23], #1
    mov w0, #124
    strb w0, [x23], #1

    mov x24, #0
asc_loop:
    cmp x24, x22
    b.hs asc_done
    add x1, x20, x21
    ldrb w0, [x1, x24]
    cmp w0, #0x20
    b.lo asc_dot
    cmp w0, #0x7e
    b.hi asc_dot
    strb w0, [x23], #1
    b asc_next
asc_dot:
    mov w0, #46
    strb w0, [x23], #1
asc_next:
    add x24, x24, #1
    b asc_loop
asc_done:
    mov w0, #124
    strb w0, [x23], #1
    mov w0, #10
    strb w0, [x23], #1

    mov x1, x27
    sub x2, x23, x27
    bl sys_write
    add x21, x21, x22
    b row_loop

final_line:
    mov x23, x27
    mov x0, x19
    bl put_u32_hex
    mov w0, #10
    strb w0, [x23], #1
    mov x1, x27
    sub x2, x23, x27
    bl sys_write

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// x0 = value, x23 = destination (advanced), x25 = hexdigits
put_u32_hex:
    mov x1, #28
put_u32_loop:
    lsr x2, x0, x1
    and x2, x2, #15
    ldrb w2, [x25, x2]
    strb w2, [x23], #1
    subs x1, x1, #4
    b.ge put_u32_loop
    ret

// w0 = byte, x23 = destination (advanced), x25 = hexdigits
put_hex2:
    lsr w1, w0, #4
    and w2, w0, #15
    ldrb w1, [x25, x1]
    strb w1, [x23], #1
    ldrb w2, [x25, x2]
    strb w2, [x23], #1
    ret

// x1 = buffer, x2 = length
sys_write:
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    ret
