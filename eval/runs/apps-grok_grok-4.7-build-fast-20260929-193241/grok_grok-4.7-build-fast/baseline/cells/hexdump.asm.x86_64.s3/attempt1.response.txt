.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.section .bss
.balign 16
input_buf:
    .zero 65536
outbuf:
    .zero 524288

.text
_start:
    lea rbx, [rip + input_buf]
    xor r12, r12

read_loop:
    mov eax, 65536
    sub rax, r12
    jz read_done
    mov rdx, rax
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r8, [rip + outbuf]
    xor r13, r13

row_loop:
    cmp r13, r12
    jae final_off
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe have_n
    mov r14d, 16
have_n:
    lea rbp, [rbx + r13]
    mov rdi, r13
    mov rsi, r8
    call print_hex8
    mov r8, rsi
    mov byte ptr [r8], 32
    mov byte ptr [r8 + 1], 32
    add r8, 2
    xor r15, r15

slot_loop:
    cmp r15, r14
    jae empty_slot
    movzx edi, byte ptr [rbp + r15]
    mov rsi, r8
    call print_hex2
    mov r8, rsi
    mov byte ptr [r8], 32
    inc r8
    jmp after_slot
empty_slot:
    mov byte ptr [r8], 32
    mov byte ptr [r8 + 1], 32
    mov byte ptr [r8 + 2], 32
    add r8, 3
after_slot:
    inc r15
    cmp r15, 8
    jne no_mid
    mov byte ptr [r8], 32
    inc r8
no_mid:
    cmp r15, 16
    jb slot_loop

    mov byte ptr [r8], 32
    mov byte ptr [r8 + 1], 124
    add r8, 2
    xor r15, r15

ascii_loop:
    cmp r15, r14
    jae ascii_done
    movzx eax, byte ptr [rbp + r15]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    mov byte ptr [r8], al
    inc r8
    jmp ascii_next
ascii_dot:
    mov byte ptr [r8], 46
    inc r8
ascii_next:
    inc r15
    jmp ascii_loop

ascii_done:
    mov byte ptr [r8], 124
    mov byte ptr [r8 + 1], 10
    add r8, 2
    add r13, 16
    jmp row_loop

final_off:
    mov rdi, r12
    mov rsi, r8
    call print_hex8
    mov byte ptr [rsi], 10
    inc rsi
    lea rdx, [rip + outbuf]
    sub rsi, rdx
    mov r13, rsi
    mov r12, rdx

write_loop:
    test r13, r13
    jz do_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle do_exit
    add r12, rax
    sub r13, rax
    jmp write_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

print_hex8:
    mov ecx, 32
ph8_loop:
    sub ecx, 4
    mov rax, rdi
    shr rax, cl
    and eax, 15
    lea rdx, [rip + hexdigits]
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [rsi], al
    inc rsi
    test ecx, ecx
    jnz ph8_loop
    ret

print_hex2:
    movzx eax, dil
    mov ecx, eax
    shr ecx, 4
    lea rdx, [rip + hexdigits]
    movzx ecx, byte ptr [rdx + rcx]
    mov byte ptr [rsi], cl
    and eax, 15
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [rsi + 1], al
    add rsi, 2
    ret
