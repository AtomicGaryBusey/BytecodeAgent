.globl _start

.section .rodata
nlbyte:
    .byte 0x0A

.bss
.align 3
inbuf:
    .zero 65536

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    movz x21, #1, lsl #16

read_more:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    cmp x20, x21
    b.lt read_more

read_done:
    mov x21, #0

next_line:
    cmp x21, x20
    b.ge exit_ok
    mov x22, x21

scan:
    cmp x22, x20
    b.ge emit_last
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq emit_nl
    add x22, x22, #1
    b scan

emit_nl:
    bl rev_write
    add x21, x22, #1
    b next_line

emit_last:
    bl rev_write

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

// reverse inbuf[x21, x22) and write it followed by 0x0A
rev_write:
    mov x0, x21
    sub x1, x22, #1
rev_loop:
    cmp x0, x1
    b.ge rev_out
    ldrb w2, [x19, x0]
    ldrb w3, [x19, x1]
    strb w3, [x19, x0]
    strb w2, [x19, x1]
    add x0, x0, #1
    sub x1, x1, #1
    b rev_loop
rev_out:
    sub x2, x22, x21
    cbz x2, wr_nl
    mov x0, #1
    add x1, x19, x21
    mov x8, #64
    svc #0
wr_nl:
    mov x0, #1
    adrp x1, nlbyte
    add x1, x1, :lo12:nlbyte
    mov x2, #1
    mov x8, #64
    svc #0
    ret
