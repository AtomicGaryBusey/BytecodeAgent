.globl _start

.section .rodata
nl_char:
    .byte 10

.bss
.align 8
input_buf:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000

.text
_start:
    lea r15, [rip + input_buf]
    lea r14, [rip + line_off]
    lea rbp, [rip + line_len]
    xor r12, r12

read_more:
    mov rdx, 32768
    sub rdx, r12
    jz parse_start
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    syscall
    test rax, rax
    jle parse_start
    add r12, rax
    jmp read_more

parse_start:
    xor r13, r13
    xor rbx, rbx

parse_loop:
    cmp rbx, r12
    jae sort_begin
    mov r8, rbx

parse_scan:
    cmp rbx, r12
    jae parse_last
    movzx eax, byte ptr [r15 + rbx]
    inc rbx
    cmp al, 10
    jne parse_scan
    mov rdx, rbx
    dec rdx
    sub rdx, r8
    mov dword ptr [r14 + r13*4], r8d
    mov dword ptr [rbp + r13*4], edx
    inc r13
    jmp parse_loop

parse_last:
    mov rdx, r12
    sub rdx, r8
    mov dword ptr [r14 + r13*4], r8d
    mov dword ptr [rbp + r13*4], edx
    inc r13

sort_begin:
    mov rax, r13
    shr rax, 1
    test rax, rax
    jz heap_ready

build_heap:
    dec rax
    push rax
    mov rdi, rax
    mov rsi, r13
    call sift_down
    pop rax
    test rax, rax
    jnz build_heap

heap_ready:
    mov r12, r13

extract_loop:
    cmp r12, 1
    jbe write_begin
    dec r12
    mov eax, dword ptr [r14]
    mov ecx, dword ptr [r14 + r12*4]
    mov dword ptr [r14], ecx
    mov dword ptr [r14 + r12*4], eax
    mov eax, dword ptr [rbp]
    mov ecx, dword ptr [rbp + r12*4]
    mov dword ptr [rbp], ecx
    mov dword ptr [rbp + r12*4], eax
    xor edi, edi
    mov rsi, r12
    call sift_down
    jmp extract_loop

write_begin:
    xor rbx, rbx

write_loop:
    cmp rbx, r13
    jae do_exit
    mov eax, dword ptr [r14 + rbx*4]
    mov edx, dword ptr [rbp + rbx*4]
    lea rsi, [r15 + rax]
    mov r12, rbx
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl_char]
    mov edx, 1
    syscall
    mov rbx, r12
    inc rbx
    jmp write_loop

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

cmp_lines:
    push rbx
    mov r8d, dword ptr [r14 + rdi*4]
    mov r9d, dword ptr [rbp + rdi*4]
    mov r10d, dword ptr [r14 + rsi*4]
    mov r11d, dword ptr [rbp + rsi*4]
    mov ecx, r9d
    cmp ecx, r11d
    cmova ecx, r11d
    xor edx, edx

cmp_bytes:
    cmp edx, ecx
    jae cmp_by_len
    lea rax, [r15 + r8]
    movzx eax, byte ptr [rax + rdx]
    lea rbx, [r15 + r10]
    movzx ebx, byte ptr [rbx + rdx]
    cmp al, bl
    ja cmp_greater
    jb cmp_less
    inc edx
    jmp cmp_bytes

cmp_by_len:
    cmp r9d, r11d
    ja cmp_greater
    jb cmp_less
    xor eax, eax
    pop rbx
    ret

cmp_greater:
    mov eax, 1
    pop rbx
    ret

cmp_less:
    mov eax, -1
    pop rbx
    ret

sift_down:
    push rbx
    push r12
    mov rbx, rdi
    mov r12, rsi

sift_loop:
    mov rax, rbx
    shl rax, 1
    inc rax
    cmp rax, r12
    jae sift_done
    mov r8, rax
    lea r9, [rax + 1]
    cmp r9, r12
    jae sift_have_child
    push r8
    push r9
    mov rdi, r9
    mov rsi, r8
    call cmp_lines
    pop r9
    pop r8
    test eax, eax
    jle sift_have_child
    mov r8, r9

sift_have_child:
    push r8
    mov rdi, r8
    mov rsi, rbx
    call cmp_lines
    pop r8
    test eax, eax
    jle sift_done
    mov eax, dword ptr [r14 + rbx*4]
    mov ecx, dword ptr [r14 + r8*4]
    mov dword ptr [r14 + rbx*4], ecx
    mov dword ptr [r14 + r8*4], eax
    mov eax, dword ptr [rbp + rbx*4]
    mov ecx, dword ptr [rbp + r8*4]
    mov dword ptr [rbp + rbx*4], ecx
    mov dword ptr [rbp + r8*4], eax
    mov rbx, r8
    jmp sift_loop

sift_done:
    pop r12
    pop rbx
    ret
