.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.section .bss
.align 3
inbuf:
    .zero 65536
outbuf:
    .zero 96

.text
_start:
    adr x19, inbuf
    mov x20, xzr

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    add x1, x19, x20
    mov x0, xzr
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, xzr

row_loop:
    cmp x21, x20
    b.hs final_off
    sub x22, x20, x21
    cmp x22, #16
    b.ls have_n
    mov x22, #16
have_n:
    adr x27, outbuf
    mov x0, x21
    mov x1, x27
    bl put_hex32
    mov x0, #2
    bl put_spaces
    mov x27, x1

    mov x24, xzr
hex_i:
    cmp x24, x22
    b.hs hex_pad
    add x3, x19, x21
    ldrb w0, [x3, x24]
    mov x1, x27
    bl put_hex8
    mov x0, #1
    bl put_spaces
    b hex_next
hex_pad:
    mov x0, #3
    mov x1, x27
    bl put_spaces
hex_next:
    mov x27, x1
    add x24, x24, #1
    cmp x24, #8
    b.ne hex_check
    mov x0, #1
    mov x1, x27
    bl put_spaces
    mov x27, x1
hex_check:
    cmp x24, #16
    b.ne hex_i

    mov x0, #1
    mov x1, x27
    bl put_spaces
    mov w0, #124
    bl put_char
    mov x27, x1

    mov x24, xzr
asc_i:
    cmp x24, x22
    b.hs asc_done
    add x3, x19, x21
    ldrb w0, [x3, x24]
    cmp w0, #32
    b.lo asc_dot
    cmp w0, #126
    b.ls asc_put
asc_dot:
    mov w0, #46
asc_put:
    mov x1, x27
    bl put_char
    mov x27, x1
    add x24, x24, #1
    b asc_i
asc_done:
    mov w0, #124
    mov x1, x27
    bl put_char
    mov w0, #10
    bl put_char
    mov x27, x1

    adr x1, outbuf
    sub x2, x27, x1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    add x21, x21, x22
    b row_loop

final_off:
    mov x0, x20
    adr x1, outbuf
    bl put_hex32
    mov w0, #10
    bl put_char
    adr x2, outbuf
    sub x2, x1, x2
    adr x1, outbuf
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0

// x0 = value, x1 = dest; returns x1 advanced by 8; clobbers x2, x3, x4
put_hex32:
    adr x4, hexdig
    mov x2, #28
put_hex32_loop:
    lsr x3, x0, x2
    and x3, x3, #15
    ldrb w3, [x4, x3]
    strb w3, [x1], #1
    subs x2, x2, #4
    b.ge put_hex32_loop
    ret

// x0 = byte, x1 = dest; returns x1 advanced by 2; clobbers x2, x4
put_hex8:
    adr x4, hexdig
    lsr x2, x0, #4
    and x2, x2, #15
    ldrb w2, [x4, x2]
    strb w2, [x1], #1
    and x2, x0, #15
    ldrb w2, [x4, x2]
    strb w2, [x1], #1
    ret

// x0 = count, x1 = dest; returns x1 advanced; clobbers x0, x2
put_spaces:
    mov w2, #32
put_spaces_loop:
    cbz x0, put_spaces_ret
    strb w2, [x1], #1
    sub x0, x0, #1
    b put_spaces_loop
put_spaces_ret:
    ret

// x0 = char, x1 = dest; returns x1 advanced by 1
put_char:
    strb w0, [x1], #1
    ret
