.globl _start

.section .rodata
nl:
  .byte 0x0A

.bss
.balign 8
buffer:
  .zero 32768
line_off:
  .zero 4000
line_len:
  .zero 4000
tmp_off:
  .zero 4
tmp_len:
  .zero 4

.text
_start:
  xor ebx, ebx
read_loop:
  mov edx, 32768
  sub edx, ebx
  jz read_done
  xor eax, eax
  xor edi, edi
  lea rsi, [rip + buffer]
  add rsi, rbx
  syscall
  test rax, rax
  jle read_done
  add ebx, eax
  jmp read_loop
read_done:
  lea r12, [rip + buffer]
  lea r13, [rip + line_off]
  lea r14, [rip + line_len]
  xor ecx, ecx
  xor edx, edx
  xor r8d, r8d
parse_loop:
  cmp ecx, ebx
  jae parse_tail
  movzx eax, byte ptr [r12 + rcx]
  cmp al, 0x0A
  jne parse_advance
  mov dword ptr [r13 + r8*4], edx
  mov eax, ecx
  sub eax, edx
  mov dword ptr [r14 + r8*4], eax
  inc r8d
  mov edx, ecx
  inc edx
parse_advance:
  inc ecx
  jmp parse_loop
parse_tail:
  cmp edx, ebx
  jae have_lines
  mov dword ptr [r13 + r8*4], edx
  mov eax, ebx
  sub eax, edx
  mov dword ptr [r14 + r8*4], eax
  inc r8d
have_lines:
  test r8d, r8d
  jz do_exit
  mov r9d, 1
sort_outer:
  cmp r9d, r8d
  jae sort_done
  mov eax, dword ptr [r13 + r9*4]
  mov dword ptr [rip + tmp_off], eax
  mov eax, dword ptr [r14 + r9*4]
  mov dword ptr [rip + tmp_len], eax
  mov r10d, r9d
sort_inner:
  test r10d, r10d
  jz sort_insert
  mov eax, r10d
  dec eax
  mov esi, dword ptr [r13 + rax*4]
  mov ecx, dword ptr [r14 + rax*4]
  lea rsi, [r12 + rsi]
  mov eax, dword ptr [rip + tmp_off]
  lea rdx, [r12 + rax]
  mov r11d, dword ptr [rip + tmp_len]
  call cmp_line
  test eax, eax
  jle sort_insert
  mov eax, r10d
  dec eax
  mov ebx, dword ptr [r13 + rax*4]
  mov dword ptr [r13 + r10*4], ebx
  mov ebx, dword ptr [r14 + rax*4]
  mov dword ptr [r14 + r10*4], ebx
  dec r10d
  jmp sort_inner
sort_insert:
  mov eax, dword ptr [rip + tmp_off]
  mov dword ptr [r13 + r10*4], eax
  mov eax, dword ptr [rip + tmp_len]
  mov dword ptr [r14 + r10*4], eax
  inc r9d
  jmp sort_outer
sort_done:
  xor r9d, r9d
write_loop:
  cmp r9d, r8d
  jae do_exit
  mov edx, dword ptr [r14 + r9*4]
  test edx, edx
  jz write_nl
  mov esi, dword ptr [r13 + r9*4]
  lea rsi, [r12 + rsi]
  mov eax, 1
  mov edi, 1
  syscall
write_nl:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + nl]
  mov edx, 1
  syscall
  inc r9d
  jmp write_loop
do_exit:
  mov eax, 60
  xor edi, edi
  syscall

# Compare line A (rsi, ecx) with line B (rdx, r11d). eax < 0, 0, > 0.
# Clobbers rax, rbx, rsi, rdx, rdi, r15. Preserves rcx, r11 and callee-saved regs used above.
cmp_line:
  mov r15, rcx
  cmp r15, r11
  cmova r15, r11
cmp_qword_loop:
  cmp r15, 8
  jb cmp_byte_start
  mov rax, qword ptr [rsi]
  mov rbx, qword ptr [rdx]
  bswap rax
  bswap rbx
  cmp rax, rbx
  jne cmp_decide
  add rsi, 8
  add rdx, 8
  sub r15, 8
  jmp cmp_qword_loop
cmp_byte_start:
  test r15, r15
  jz cmp_by_length
cmp_byte_loop:
  mov al, byte ptr [rsi]
  cmp al, byte ptr [rdx]
  jne cmp_decide
  inc rsi
  inc rdx
  dec r15
  jnz cmp_byte_loop
cmp_by_length:
  mov eax, ecx
  sub eax, r11d
  ret
cmp_decide:
  mov eax, 1
  ja cmp_ret
  mov eax, -1
cmp_ret:
  ret
