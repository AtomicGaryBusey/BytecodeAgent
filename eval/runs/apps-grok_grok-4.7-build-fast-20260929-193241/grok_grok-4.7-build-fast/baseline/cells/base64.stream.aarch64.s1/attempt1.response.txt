.equ SYS_read, 63

.equ SYS_exit, 93
.equ IN_MAX, 49152

.equ PAD_MARK, 64

.globl _start
.text

    add x19, x19, :lo12:inbuf

    mov x21, #IN_MAX
1:

    mov x0, xzr
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    sub x21, x21, x0
    b 1b
2:
    sub x23, x20, x19
    mov x22, x19
    adrp x19, b64tab
    add x19, x19, :lo12:b64tab
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf
    mov x21, xzr

    b.lt 4f
    ldrb w4, [x22]
    ldrb w5, [x22, #1]
    ldrb w6, [x22, #2]
    add x22, x22, #3
    sub x23, x23, #3
    lsr w25, w4, #2
    and w26, w4, #3
    lsl w26, w26, #4
    lsr w7, w5, #4
    orr w26, w26, w7
    and w27, w5, #15
    lsl w27, w27, #2
    lsr w7, w6, #6
    orr w27, w27, w7
    and w28, w6, #63
    mov w0, w25
    bl emit
    mov w0, w26
    bl emit
    mov w0, w27
    bl emit
    mov w0, w28
    bl emit
    b 3b
4:
    cbz x23, 6f
    cmp x23, #1
    b.eq 5f
    ldrb w4, [x22]
    ldrb w5, [x22, #1]
    lsr w25, w4, #2
    and w26, w4, #3
    lsl w26, w26, #4
    lsr w7, w5, #4
    orr w26, w26, w7
    and w27, w5, #15
    lsl w27, w27, #2
    mov w0, w25
    bl emit
    mov w0, w26
    bl emit
    mov w0, w27
    bl emit
    mov w0, #PAD_MARK
    bl emit
    b 6f
5:
    ldrb w4, [x22]
    lsr w25, w4, #2
    and w26, w4, #3
    lsl w26, w26, #4
    mov w0, w25
    bl emit
    mov w0, w26
    bl emit
    mov w0, #PAD_MARK
    bl emit
    mov w0, #PAD_MARK

    sub x23, x23, x0

    mov x8, #SYS_exit
    svc #0

// w0 = 6-bit index, or PAD_MARK for '='
// x19 = alphabet, x20 = output cursor, x21 = column
emit:
    cmp w0, #PAD_MARK
    b.eq 2f
    ldrb w1, [x19, x0]
    b 3f
2:
    mov w1, #61
3:
    strb w1, [x20], #1
    add x21, x21, #1
    cmp x21, #LINE_LEN
    b.ne 1f
    mov w1, #10
    strb w1, [x20], #1
    mov x21, xzr
1:
    ret
