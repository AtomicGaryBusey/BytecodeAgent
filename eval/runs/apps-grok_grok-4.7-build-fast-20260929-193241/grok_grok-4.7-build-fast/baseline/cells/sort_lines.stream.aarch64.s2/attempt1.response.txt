.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 32768

.globl _start
.text

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x26, line_off
    add x26, x26, :lo12:line_off
    adrp x27, line_len
    add x27, x27, :lo12:line_len

    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, #BUF_SIZE
    b.lt read_loop
read_done:
    mov x21, #0
    mov x22, #0
parse_loop:
    cmp x22, x20
    b.hs parse_done
    mov x23, x22
find_nl:
    cmp x22, x20
    b.hs line_eof
    ldrb w0, [x19, x22]
    add x22, x22, #1
    cmp w0, #0x0A
    b.ne find_nl
    sub x24, x22, x23
    sub x24, x24, #1
    b save_line
line_eof:
    sub x24, x20, x23
save_line:
    str x23, [x26, x21, lsl #3]
    str x24, [x27, x21, lsl #3]
    add x21, x21, #1
    b parse_loop
parse_done:
    lsr x22, x21, #1
build_loop:
    cbz x22, build_done
    sub x22, x22, #1
    mov x0, x21
    mov x1, x22
    bl heapify
    cbnz x22, build_loop
build_done:
    mov x22, x21
extract_loop:
    cmp x22, #2
    b.lt print_init
    sub x22, x22, #1
    ldr x0, [x26]
    ldr x1, [x26, x22, lsl #3]
    str x1, [x26]
    str x0, [x26, x22, lsl #3]
    ldr x0, [x27]
    ldr x1, [x27, x22, lsl #3]
    str x1, [x27]
    str x0, [x27, x22, lsl #3]
    mov x0, x22
    mov x1, #0
    bl heapify
    b extract_loop
print_init:
    mov x22, #0
print_loop:
    cmp x22, x21
    b.hs do_exit
    ldr x23, [x26, x22, lsl #3]
    ldr x24, [x27, x22, lsl #3]
    cbz x24, write_nl
    mov x0, #1
    add x1, x19, x23
    mov x2, x24
    mov x8, #SYS_write
    svc #0
write_nl:
    mov x0, #1
    adrp x1, nl_char
    add x1, x1, :lo12:nl_char
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x22, x22, #1
    b print_loop
do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// compare(off_a, len_a, off_b, len_b) -> 1 if a>b, -1 if a<b, 0 if equal
// clobbers x0, x1, x2, x4, x5, x10, x11
compare:
    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, #0
cmp_byte:
    cmp x5, x4
    b.hs cmp_lens
    ldrb w10, [x19, x0]
    ldrb w11, [x19, x2]
    cmp w10, w11
    b.ne cmp_ne
    add x0, x0, #1
    add x2, x2, #1
    add x5, x5, #1
    b cmp_byte
cmp_ne:
    cset x0, hi
    cset x4, lo
    sub x0, x0, x4
    ret
cmp_lens:
    cmp x1, x3
    b.eq cmp_same
    cset x0, hi
    cset x4, lo
    sub x0, x0, x4
    ret
cmp_same:
    mov x0, #0
    ret

// heapify(heap_n = x0, i = x1). Preserves x19, x21, x22, x26, x27.
heapify:
    str x30, [sp, #-16]!
    mov x23, x0
    mov x24, x1
hf_loop:
    mov x25, x24
    add x28, x24, x24
    add x28, x28, #1
    add x29, x28, #1
    cmp x28, x23
    b.hs hf_right
    ldr x0, [x26, x28, lsl #3]
    ldr x1, [x27, x28, lsl #3]
    ldr x2, [x26, x25, lsl #3]
    ldr x3, [x27, x25, lsl #3]
    bl compare
    cmp x0, #0
    b.le hf_right
    mov x25, x28
hf_right:
    cmp x29, x23
    b.hs hf_check
    ldr x0, [x26, x29, lsl #3]
    ldr x1, [x27, x29, lsl #3]
    ldr x2, [x26, x25, lsl #3]
    ldr x3, [x27, x25, lsl #3]
    bl compare
    cmp x0, #0
    b.le hf_check
    mov x25, x29
hf_check:
    cmp x25, x24
    b.eq hf_done
    ldr x0, [x26, x24, lsl #3]
    ldr x1, [x26, x25, lsl #3]
    str x1, [x26, x24, lsl #3]
    str x0, [x26, x25, lsl #3]
    ldr x0, [x27, x24, lsl #3]
    ldr x1, [x27, x25, lsl #3]
    str x1, [x27, x24, lsl #3]
    str x0, [x27, x25, lsl #3]
    mov x24, x25
    b hf_loop
hf_done:
    ldr x30, [sp], #16
    ret

.section .rodata
nl_char:
    .byte 0x0A

.bss
.align 3
inbuf:
    .zero 32768
.align 3
line_off:
    .zero 8000
.align 3
line_len:
    .zero 8000
