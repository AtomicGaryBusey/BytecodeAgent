.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_CAP, 65536
.equ MAX_DEPTH, 1000

.bss
.align 8
buf:
    .zero 65536
tape:
    .zero 30000
jump_tab:
    .zero 524288
br_stack:
    .zero 8000
dispatch:
    .zero 2048

.text
.globl _start
_start:
    lea r8, [rip + buf]
    xor r12, r12

read_loop:
    cmp r12, BUF_CAP
    jae read_done
    mov rax, SYS_read
    xor rdi, rdi
    lea rsi, [r8 + r12]
    mov rdx, BUF_CAP
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov rbp, r12
    xor rcx, rcx

find_loop:
    cmp rcx, rbp
    jae no_bang
    cmp byte ptr [r8 + rcx], 33
    je yes_bang
    inc rcx
    jmp find_loop

yes_bang:
    mov r12, rcx
    lea r13, [r8 + rcx + 1]
    lea r14, [r8 + rbp]
    jmp compact

no_bang:
    mov r12, rbp
    lea r13, [r8 + rbp]
    mov r14, r13

compact:
    xor rdi, rdi
    xor rcx, rcx

compact_loop:
    cmp rcx, r12
    jae compact_done
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    cmp al, 62
    je keep
    cmp al, 60
    je keep
    cmp al, 43
    je keep
    cmp al, 45
    je keep
    cmp al, 46
    je keep
    cmp al, 44
    je keep
    cmp al, 91
    je keep
    cmp al, 93
    je keep
    jmp compact_loop

keep:
    mov byte ptr [r8 + rdi], al
    inc rdi
    jmp compact_loop

compact_done:
    mov r12, rdi
    lea r10, [rip + jump_tab]
    lea rsi, [rip + br_stack]
    xor rdi, rdi
    xor rcx, rcx

match_loop:
    cmp rcx, r12
    jae match_done
    movzx eax, byte ptr [r8 + rcx]
    cmp al, 91
    je match_open
    cmp al, 93
    je match_close
    inc rcx
    jmp match_loop

match_open:
    cmp rdi, MAX_DEPTH
    jae bad
    mov qword ptr [rsi + rdi*8], rcx
    inc rdi
    inc rcx
    jmp match_loop

match_close:
    test rdi, rdi
    jz bad
    dec rdi
    mov rax, qword ptr [rsi + rdi*8]
    mov qword ptr [r10 + rax*8], rcx
    mov qword ptr [r10 + rcx*8], rax
    inc rcx
    jmp match_loop

match_done:
    test rdi, rdi
    jnz bad

    lea r9, [rip + dispatch]
    lea rax, [rip + op_nop]
    mov rcx, 256
    mov rsi, r9

fill_disp:
    mov qword ptr [rsi], rax
    add rsi, 8
    dec rcx
    jnz fill_disp

    lea rax, [rip + op_gt]
    mov ecx, 62
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_lt]
    mov ecx, 60
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_plus]
    mov ecx, 43
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_minus]
    mov ecx, 45
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_dot]
    mov ecx, 46
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_comma]
    mov ecx, 44
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_lbr]
    mov ecx, 91
    mov qword ptr [r9 + rcx*8], rax
    lea rax, [rip + op_rbr]
    mov ecx, 93
    mov qword ptr [r9 + rcx*8], rax

    lea r15, [rip + tape]
    xor rbx, rbx
    xor rbp, rbp

exec:
    cmp rbx, r12
    jae ok
    movzx eax, byte ptr [r8 + rbx]
    jmp qword ptr [r9 + rax*8]

op_gt:
    inc rbp
    inc rbx
    jmp exec

op_lt:
    dec rbp
    inc rbx
    jmp exec

op_plus:
    inc byte ptr [r15 + rbp]
    inc rbx
    jmp exec

op_minus:
    dec byte ptr [r15 + rbp]
    inc rbx
    jmp exec

op_dot:
    mov rax, SYS_write
    mov rdi, 1
    lea rsi, [r15 + rbp]
    mov rdx, 1
    syscall
    inc rbx
    jmp exec

op_comma:
    cmp r13, r14
    jae comma_eof
    movzx eax, byte ptr [r13]
    mov byte ptr [r15 + rbp], al
    inc r13
    inc rbx
    jmp exec

comma_eof:
    mov byte ptr [r15 + rbp], 0
    inc rbx
    jmp exec

op_lbr:
    cmp byte ptr [r15 + rbp], 0
    jne lbr_cont
    mov rbx, qword ptr [r10 + rbx*8]
lbr_cont:
    inc rbx
    jmp exec

op_rbr:
    cmp byte ptr [r15 + rbp], 0
    je rbr_cont
    mov rbx, qword ptr [r10 + rbx*8]
rbr_cont:
    inc rbx
    jmp exec

op_nop:
    inc rbx
    jmp exec

bad:
    mov rax, SYS_exit
    mov rdi, 1
    syscall

ok:
    mov rax, SYS_exit
    xor rdi, rdi
    syscall
