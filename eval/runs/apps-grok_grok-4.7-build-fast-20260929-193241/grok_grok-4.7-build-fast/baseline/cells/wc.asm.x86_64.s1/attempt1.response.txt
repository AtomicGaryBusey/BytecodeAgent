.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_SIZE, 65536

.globl _start

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 64
numtmp:
    .zero 32

.section .text

_start:
    xor r12d, r12d

read_more:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, BUF_SIZE
    sub edx, r12d
    jz do_count
    syscall
    test rax, rax
    jle do_count
    add r12, rax
    cmp r12, BUF_SIZE
    jb read_more

do_count:
    xor r13, r13
    xor r14, r14
    xor r15d, r15d
    lea rsi, [rip + inbuf]
    mov rcx, r12
    test rcx, rcx
    jz do_format

scan:
    movzx eax, byte ptr [rsi]
    cmp al, 0x0A
    jne 1f
    inc r13
1:
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    je 2f
    cmp al, 0x0A
    je 2f
    cmp al, 0x0B
    je 2f
    cmp al, 0x0C
    je 2f
    cmp al, 0x0D
    je 2f
    test r15d, r15d
    jnz 3f
    inc r14
    mov r15d, 1
    jmp 3f
2:
    xor r15d, r15d
3:
    inc rsi
    dec rcx
    jnz scan

do_format:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call u64_to_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call u64_to_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r12
    call u64_to_dec
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall

    mov eax, SYS_exit
    xor edi, edi
    syscall

u64_to_dec:
    lea rbx, [rip + numtmp]
    add rbx, 32
    mov rsi, rbx
    mov ecx, 10
    test rax, rax
    jnz 1f
    dec rsi
    mov byte ptr [rsi], 0x30
    jmp 2f
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    cmp rsi, rbx
    jne 2b
    ret
