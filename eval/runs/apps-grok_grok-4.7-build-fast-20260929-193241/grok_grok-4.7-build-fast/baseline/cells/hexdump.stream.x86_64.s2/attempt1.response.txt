.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ INSIZE, 65536

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.align 8
inbuf:
    .zero 65536
outbuf:
    .zero 96

.text
.globl _start
_start:
    xor rbx, rbx
read_more:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, INSIZE
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, INSIZE
    jb read_more
read_done:
    xor r12, r12
row_loop:
    cmp r12, rbx
    jae rows_done
    mov r13, rbx
    sub r13, r12
    cmp r13, 16
    jbe row_n_ready
    mov r13, 16
row_n_ready:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor r8, r8
hex_slots:
    cmp r8, r13
    jae hex_pad
    mov al, byte ptr [rsi]
    inc rsi
    call hex2
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_slot_done
hex_pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
hex_slot_done:
    cmp r8, 7
    jne hex_no_gap
    mov byte ptr [rdi], 0x20
    inc rdi
hex_no_gap:
    inc r8
    cmp r8, 16
    jb hex_slots
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor r8, r8
ascii_loop:
    cmp r8, r13
    jae ascii_done
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rdi], 0x2e
ascii_next:
    inc rdi
    inc r8
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, SYS_write
    syscall
    add r12, r13
    jmp row_loop
rows_done:
    lea rdi, [rip + outbuf]
    mov eax, ebx
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, SYS_write
    syscall
    mov eax, SYS_exit
    xor edi, edi
    syscall

hex8:
    lea r11, [rip + hextab]
    mov ecx, 8
hex8_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz hex8_loop
    ret

hex2:
    movzx eax, al
    mov edx, eax
    shr edx, 4
    lea r11, [rip + hextab]
    movzx edx, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    movzx eax, byte ptr [r11 + rax]
    mov byte ptr [rdi], al
    inc rdi
    ret
