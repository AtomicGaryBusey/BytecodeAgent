.globl _start
.text

_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x21, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x21
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x0, #0
find_loop:
    cmp x0, x21
    b.eq no_bang
    ldrb w1, [x20, x0]
    cmp w1, #33
    b.eq found_bang
    add x0, x0, #1
    b find_loop

found_bang:
    mov x22, x0
    add x23, x20, x0
    add x23, x23, #1
    add x24, x20, x21
    b do_filter

no_bang:
    mov x22, x21
    add x23, x20, x21
    mov x24, x23

do_filter:
    adrp x19, match
    add x19, x19, :lo12:match
    adrp x10, brstack
    add x10, x10, :lo12:brstack
    mov x0, #0
    mov x1, #0
    mov x11, #0

filter_loop:
    cmp x0, x22
    b.eq filter_done
    ldrb w2, [x20, x0]
    add x0, x0, #1
    cmp w2, #43
    b.eq f_store
    cmp w2, #45
    b.eq f_store
    cmp w2, #62
    b.eq f_store
    cmp w2, #60
    b.eq f_store
    cmp w2, #46
    b.eq f_store
    cmp w2, #44
    b.eq f_store
    cmp w2, #91
    b.eq f_open
    cmp w2, #93
    b.eq f_close
    b filter_loop

f_open:
    cmp x11, #1000
    b.hs unbalanced
    str w1, [x10, x11, lsl #2]
    add x11, x11, #1
    b f_store

f_close:
    cbz x11, unbalanced
    sub x11, x11, #1
    ldr w3, [x10, x11, lsl #2]
    strh w3, [x19, x1, lsl #1]
    strh w1, [x19, x3, lsl #1]
    b f_store

f_store:
    strb w2, [x20, x1]
    add x1, x1, #1
    b filter_loop

filter_done:
    cbnz x11, unbalanced
    mov x22, x1

    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
    mov x27, #0
    mov x29, #0

    adrp x26, cmdtab
    add x26, x26, :lo12:cmdtab
    adr x1, exec_next
    mov x3, x26
    mov x2, #256
fill_tab:
    str x1, [x3], #8
    subs x2, x2, #1
    b.ne fill_tab

    adr x1, c_plus
    mov w2, #43
    str x1, [x26, x2, lsl #3]
    adr x1, c_comma
    mov w2, #44
    str x1, [x26, x2, lsl #3]
    adr x1, c_minus
    mov w2, #45
    str x1, [x26, x2, lsl #3]
    adr x1, c_dot
    mov w2, #46
    str x1, [x26, x2, lsl #3]
    adr x1, c_left
    mov w2, #60
    str x1, [x26, x2, lsl #3]
    adr x1, c_right
    mov w2, #62
    str x1, [x26, x2, lsl #3]
    adr x1, c_lbr
    mov w2, #91
    str x1, [x26, x2, lsl #3]
    adr x1, c_rbr
    mov w2, #93
    str x1, [x26, x2, lsl #3]

exec_loop:
    cmp x27, x22
    b.hs exec_done
    ldrb w2, [x20, x27]
    ldr x4, [x26, x2, lsl #3]
    br x4

c_plus:
    ldrb w2, [x25]
    add w2, w2, #1
    strb w2, [x25]
    b exec_next

c_minus:
    ldrb w2, [x25]
    sub w2, w2, #1
    strb w2, [x25]
    b exec_next

c_right:
    add x25, x25, #1
    b exec_next

c_left:
    sub x25, x25, #1
    b exec_next

c_lbr:
    ldrb w2, [x25]
    cbnz w2, exec_next
    ldrh w2, [x19, x27, lsl #1]
    mov x27, x2
    b exec_next

c_rbr:
    ldrb w2, [x25]
    cbz w2, exec_next
    ldrh w2, [x19, x27, lsl #1]
    mov x27, x2
    b exec_next

c_dot:
    ldrb w2, [x25]
    strb w2, [x28, x29]
    add x29, x29, #1
    b exec_next

c_comma:
    cmp x23, x24
    b.hs c_eof
    ldrb w2, [x23], #1
    strb w2, [x25]
    b exec_next

c_eof:
    strb wzr, [x25]
    b exec_next

exec_next:
    add x27, x27, #1
    b exec_loop

exec_done:
    cbz x29, exit_ok
    mov x0, #1
    mov x1, x28
    mov x2, x29
    mov x8, #64
    svc #0
exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_ok

unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0
    b unbalanced
