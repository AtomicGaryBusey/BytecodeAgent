.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536
.equ MAX_NEST, 1024
.equ BANG, 33
.equ CH_PLUS, 43
.equ CH_COMMA, 44
.equ CH_MINUS, 45
.equ CH_DOT, 46
.equ CH_LT, 60
.equ CH_GT, 62
.equ CH_LB, 91
.equ CH_RB, 93

.text
.align 2
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.eq read_done
    b.lt exit_fail
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

find_loop:
    cmp x21, x20
    b.eq no_bang
    ldrb w0, [x19, x21]
    cmp w0, #BANG
    b.eq found_bang
    add x21, x21, #1
    b find_loop

found_bang:
    add x23, x21, #1
    sub x24, x20, x23
    b split_done

no_bang:
    mov x23, x20
    mov x24, #0

split_done:
    adrp x27, brack_stk
    add x27, x27, :lo12:brack_stk
    adrp x28, jump_tab
    add x28, x28, :lo12:jump_tab
    mov x25, #0
    mov x26, #0
    mov x20, #0

filter_loop:
    cmp x25, x21
    b.eq filter_done
    ldrb w0, [x19, x25]
    add x25, x25, #1
    cmp w0, #CH_PLUS
    b.eq store_cmd
    cmp w0, #CH_COMMA
    b.eq store_cmd
    cmp w0, #CH_MINUS
    b.eq store_cmd
    cmp w0, #CH_DOT
    b.eq store_cmd
    cmp w0, #CH_LT
    b.eq store_cmd
    cmp w0, #CH_GT
    b.eq store_cmd
    cmp w0, #CH_LB
    b.eq store_lb
    cmp w0, #CH_RB
    b.eq store_rb
    b filter_loop

store_lb:
    cmp x20, #MAX_NEST
    b.hs exit_fail
    str w26, [x27, x20, lsl #2]
    add x20, x20, #1
    b store_cmd

store_rb:
    cbz x20, exit_fail
    sub x20, x20, #1
    ldr w1, [x27, x20, lsl #2]
    str w26, [x28, x1, lsl #2]
    str w1, [x28, x26, lsl #2]
    b store_cmd

store_cmd:
    strb w0, [x19, x26]
    add x26, x26, #1
    b filter_loop

filter_done:
    cbnz x20, exit_fail
    mov x21, x26
    adrp x22, tape
    add x22, x22, :lo12:tape
    add x23, x19, x23
    mov x25, #0
    mov x26, #0
    mov x27, #0

exec_loop:
    cmp x25, x21
    b.hs exec_done
    ldrb w0, [x19, x25]
    cmp w0, #CH_DOT
    b.hi high_cmd
    cmp w0, #CH_PLUS
    b.eq cmd_plus
    cmp w0, #CH_COMMA
    b.eq cmd_comma
    cmp w0, #CH_MINUS
    b.eq cmd_minus
    cmp w0, #CH_DOT
    b.eq cmd_dot
    b cmd_next

high_cmd:
    cmp w0, #CH_RB
    b.eq cmd_rbrack
    cmp w0, #CH_LB
    b.eq cmd_lbrack
    cmp w0, #CH_GT
    b.eq cmd_right
    cmp w0, #CH_LT
    b.eq cmd_left
    b cmd_next

cmd_right:
    add x26, x26, #1
    b cmd_next

cmd_left:
    sub x26, x26, #1
    b cmd_next

cmd_plus:
    ldrb w0, [x22, x26]
    add w0, w0, #1
    strb w0, [x22, x26]
    b cmd_next

cmd_minus:
    ldrb w0, [x22, x26]
    sub w0, w0, #1
    strb w0, [x22, x26]
    b cmd_next

cmd_dot:
    add x1, x22, x26
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    b cmd_next

cmd_comma:
    cmp x27, x24
    b.hs comma_eof
    ldrb w0, [x23, x27]
    add x27, x27, #1
    strb w0, [x22, x26]
    b cmd_next

comma_eof:
    mov w0, #0
    strb w0, [x22, x26]
    b cmd_next

cmd_lbrack:
    ldrb w0, [x22, x26]
    cbnz w0, cmd_next
    ldr w0, [x28, x25, lsl #2]
    mov x25, x0
    b cmd_next

cmd_rbrack:
    ldrb w0, [x22, x26]
    cbz w0, cmd_next
    ldr w0, [x28, x25, lsl #2]
    mov x25, x0
    b cmd_next

cmd_next:
    add x25, x25, #1
    b exec_loop

exec_done:
    mov x0, #0
    b do_exit

exit_fail:
    mov x0, #1

do_exit:
    mov x8, #SYS_exit
    svc #0
    b do_exit

.bss
.align 3
input_buf:
    .zero 65536
tape:
    .zero 30000
.align 3
jump_tab:
    .zero 262144
brack_stk:
    .zero 4096
