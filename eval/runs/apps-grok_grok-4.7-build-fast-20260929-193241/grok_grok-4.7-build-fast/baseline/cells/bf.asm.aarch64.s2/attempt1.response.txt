.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_CAP, 65536
.equ MAX_NEST, 1024

.globl _start
.text

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #IN_CAP
read_loop:
    cbz x21, read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, x21
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    sub x21, x21, x0
    b read_loop
read_done:
    mov x4, #0
find_loop:
    cmp x4, x20
    b.hs no_bang
    ldrb w0, [x19, x4]
    cmp w0, #33
    b.eq yes_bang
    add x4, x4, #1
    b find_loop
yes_bang:
    mov x21, x4
    add x22, x19, x4
    add x22, x22, #1
    sub x23, x20, x4
    sub x23, x23, #1
    b filter
no_bang:
    mov x21, x20
    mov x22, #0
    mov x23, #0
filter:
    mov x4, #0
    mov x5, #0
filt_loop:
    cmp x4, x21
    b.hs filt_done
    ldrb w0, [x19, x4]
    add x4, x4, #1
    cmp w0, #43
    b.eq k_plus
    cmp w0, #45
    b.eq k_minus
    cmp w0, #62
    b.eq k_gt
    cmp w0, #60
    b.eq k_lt
    cmp w0, #46
    b.eq k_dot
    cmp w0, #44
    b.eq k_comma
    cmp w0, #91
    b.eq k_lbr
    cmp w0, #93
    b.eq k_rbr
    b filt_loop
k_plus:
    mov w0, #0
    b k_store
k_minus:
    mov w0, #1
    b k_store
k_gt:
    mov w0, #2
    b k_store
k_lt:
    mov w0, #3
    b k_store
k_dot:
    mov w0, #4
    b k_store
k_comma:
    mov w0, #5
    b k_store
k_lbr:
    mov w0, #6
    b k_store
k_rbr:
    mov w0, #7
k_store:
    strb w0, [x19, x5]
    add x5, x5, #1
    b filt_loop
filt_done:
    mov x21, x5
    adrp x24, jumps
    add x24, x24, :lo12:jumps
    adrp x25, brstack
    add x25, x25, :lo12:brstack
    mov x26, #0
    mov x27, #0
match_loop:
    cmp x27, x21
    b.hs match_done
    ldrb w0, [x19, x27]
    cmp w0, #6
    b.eq do_push
    cmp w0, #7
    b.eq do_pop
    b match_next
do_push:
    cmp x26, #MAX_NEST
    b.hs bad_brackets
    str w27, [x25, x26, lsl #2]
    add x26, x26, #1
    b match_next
do_pop:
    cbz x26, bad_brackets
    sub x26, x26, #1
    ldr w0, [x25, x26, lsl #2]
    add w1, w27, #1
    str w1, [x24, x0, lsl #2]
    add w1, w0, #1
    str w1, [x24, x27, lsl #2]
    b match_next
match_next:
    add x27, x27, #1
    b match_loop
match_done:
    cbnz x26, bad_brackets
    adrp x9, tape
    add x9, x9, :lo12:tape
    adrp x14, outbuf
    add x14, x14, :lo12:outbuf
    adrp x15, htable
    add x15, x15, :lo12:htable
    adr x0, op_plus
    str x0, [x15]
    adr x0, op_minus
    str x0, [x15, #8]
    adr x0, op_gt
    str x0, [x15, #16]
    adr x0, op_lt
    str x0, [x15, #24]
    adr x0, op_dot
    str x0, [x15, #32]
    adr x0, op_comma
    str x0, [x15, #40]
    adr x0, op_lbr
    str x0, [x15, #48]
    adr x0, op_rbr
    str x0, [x15, #56]
    mov x11, #0
    mov x10, #0
    mov x25, #0
    mov x26, #0
exec_loop:
    cmp x11, x21
    b.hs exec_done
    ldrb w0, [x19, x11]
    add x11, x11, #1
    ldr x1, [x15, x0, lsl #3]
    br x1
op_plus:
    ldrb w0, [x9, x10]
    add w0, w0, #1
    strb w0, [x9, x10]
    b exec_loop
op_minus:
    ldrb w0, [x9, x10]
    sub w0, w0, #1
    strb w0, [x9, x10]
    b exec_loop
op_gt:
    add x10, x10, #1
    b exec_loop
op_lt:
    sub x10, x10, #1
    b exec_loop
op_dot:
    ldrb w0, [x9, x10]
    strb w0, [x14, x26]
    add x26, x26, #1
    b exec_loop
op_comma:
    cmp x25, x23
    b.hs comma_z
    ldrb w0, [x22, x25]
    add x25, x25, #1
    strb w0, [x9, x10]
    b exec_loop
comma_z:
    strb wzr, [x9, x10]
    b exec_loop
op_lbr:
    ldrb w0, [x9, x10]
    cbnz w0, exec_loop
    b bracket_jump
op_rbr:
    ldrb w0, [x9, x10]
    cbz w0, exec_loop
bracket_jump:
    sub x1, x11, #1
    ldr w11, [x24, x1, lsl #2]
    b exec_loop
exec_done:
    cbz x26, do_exit
    mov x0, #1
    mov x1, x14
    mov x2, x26
    mov x8, #SYS_write
    svc #0
do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b do_exit
bad_brackets:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
    b bad_brackets

.bss
.align 4
inbuf:
    .zero 65536
outbuf:
    .zero 65536
jumps:
    .zero 262144
brstack:
    .zero 4096
htable:
    .zero 64
tape:
    .zero 30000
