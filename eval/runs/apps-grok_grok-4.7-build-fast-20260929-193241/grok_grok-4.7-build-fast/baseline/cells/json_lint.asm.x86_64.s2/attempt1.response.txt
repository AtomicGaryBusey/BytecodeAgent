.globl _start

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
input:
    .zero 65536
outbuf:
    .zero 64
scratch:
    .zero 32

.text
_start:
    lea rbx, [rip + input]
    xor r12, r12
read_loop:
    mov rax, 65536
    sub rax, r12
    jz read_done
    mov rdx, rax
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
ok_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp ok_exit

skip_ws:
sw_loop:
    cmp r13, r12
    jae sw_ret
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x20
    je sw_inc
    cmp al, 0x09
    je sw_inc
    cmp al, 0x0A
    je sw_inc
    cmp al, 0x0D
    je sw_inc
sw_ret:
    ret
sw_inc:
    inc r13
    jmp sw_loop

parse_value:
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, 0x2D
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp parse_number

parse_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp match_lit
parse_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp match_lit
parse_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4
    jmp match_lit

match_lit:
ml_loop:
    test ecx, ecx
    jz ml_ret
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, byte ptr [rsi]
    jne fail
    inc r13
    inc rsi
    dec ecx
    jmp ml_loop
ml_ret:
    ret

parse_string:
    inc r13
ps_loop:
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x22
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb fail
    inc r13
    jmp ps_loop
ps_end:
    inc r13
    ret
ps_esc:
    inc r13
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x22
    je ps_simple
    cmp al, 0x5C
    je ps_simple
    cmp al, 0x2F
    je ps_simple
    cmp al, 'b'
    je ps_simple
    cmp al, 'f'
    je ps_simple
    cmp al, 'n'
    je ps_simple
    cmp al, 'r'
    je ps_simple
    cmp al, 't'
    je ps_simple
    cmp al, 'u'
    je ps_u
    jmp fail
ps_simple:
    inc r13
    jmp ps_loop
ps_u:
    inc r13
    mov ecx, 4
ps_hexloop:
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe ps_hexok
    cmp al, 'A'
    jb fail
    cmp al, 'F'
    jbe ps_hexok
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
ps_hexok:
    inc r13
    dec ecx
    jnz ps_hexloop
    jmp ps_loop

parse_number:
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x2D
    jne pn_first
    inc r13
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
pn_first:
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    cmp al, '0'
    je pn_zero
pn_intdig:
    inc r13
    cmp r13, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + r13]
    cmp al, '0'
    jb pn_dot
    cmp al, '9'
    ja pn_dot
    jmp pn_intdig
pn_zero:
    inc r13
    cmp r13, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + r13]
pn_dot:
    cmp al, 0x2E
    jne pn_exp
    inc r13
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
pn_fracdig:
    inc r13
    cmp r13, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + r13]
    cmp al, '0'
    jb pn_exp
    cmp al, '9'
    ja pn_exp
    jmp pn_fracdig
pn_exp:
    cmp al, 'e'
    je pn_exp1
    cmp al, 'E'
    jne pn_ret
pn_exp1:
    inc r13
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x2B
    je pn_expsign
    cmp al, 0x2D
    jne pn_expdig
pn_expsign:
    inc r13
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
pn_expdig:
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
pn_expdigs:
    inc r13
    cmp r13, r12
    jae pn_ret
    movzx eax, byte ptr [rbx + r13]
    cmp al, '0'
    jb pn_ret
    cmp al, '9'
    ja pn_ret
    jmp pn_expdigs
pn_ret:
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x5D
    je pa_end
pa_loop:
    call parse_value
    call skip_ws
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    jne fail
pa_end:
    inc r13
    ret
pa_comma:
    inc r13
    call skip_ws
    jmp pa_loop

parse_object:
    inc r13
    call skip_ws
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x7D
    je po_end
po_loop:
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x3A
    jne fail
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r12
    jae fail
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    jne fail
po_end:
    inc r13
    ret
po_comma:
    inc r13
    call skip_ws
    jmp po_loop

fail:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + err_msg]
    mov ecx, 9
fc_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec ecx
    jnz fc_loop
    mov rax, r13
    lea r14, [rip + scratch + 16]
    test rax, rax
    jnz fd_loop
    mov byte ptr [rdi], 0x30
    inc rdi
    jmp fd_nl
fd_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 0x30
    dec r14
    mov byte ptr [r14], dl
    test rax, rax
    jnz fd_loop
    lea rsi, [rip + scratch + 16]
fd_copy:
    cmp r14, rsi
    jae fd_nl
    mov al, byte ptr [r14]
    mov byte ptr [rdi], al
    inc r14
    inc rdi
    jmp fd_copy
fd_nl:
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
err_exit:
    mov eax, 60
    mov edi, 1
    syscall
    jmp err_exit
