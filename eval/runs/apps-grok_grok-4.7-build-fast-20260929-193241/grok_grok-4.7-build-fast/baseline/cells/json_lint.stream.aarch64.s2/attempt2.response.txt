.equ SYS_read, 63
.equ SYS_write, 64

.equ CH_LF, 10
.equ CH_CR, 13
.equ CH_SP, 32
.equ CH_QUOT, 34
.equ CH_PLUS, 43
.equ CH_COMMA, 44
.equ CH_MINUS, 45
.equ CH_DOT, 46
.equ CH_SLASH, 47
.equ CH_0, 48
.equ CH_9, 57
.equ CH_COLON, 58
.equ CH_A, 65
.equ CH_E, 69
.equ CH_F, 70
.equ CH_LBR, 91
.equ CH_BS, 92
.equ CH_RBR, 93
.equ CH_a, 97
.equ CH_b, 98
.equ CH_e, 101
.equ CH_f, 102
.equ CH_n, 110
.equ CH_r, 114
.equ CH_t, 116
.equ CH_u, 117
.equ CH_LBRACE, 123
.equ CH_RBRACE, 125

.text
.globl _start
_start:
  adrp x19, inbuf
  add x19, x19, :lo12:inbuf
  mov x20, #0
rd_loop:
  movz x2, #1, lsl #16
  sub x2, x2, x20
  cbz x2, rd_done
  mov x0, #0
  add x1, x19, x20
  mov x8, #SYS_read
  svc #0
  cmp x0, #0
  b.le rd_done
  add x20, x20, x0
  b rd_loop
rd_done:
  mov x21, #0
  mov x22, #0
  bl skip_ws
  bl parse_value
  cbnz x22, do_error
  bl skip_ws
  cmp x21, x20
  b.lo do_error
do_ok:
  mov x0, #1
  adrp x1, ok_msg
  add x1, x1, :lo12:ok_msg
  mov x2, #3
  mov x8, #SYS_write
  svc #0
  mov x0, #0
  mov x8, #SYS_exit
  svc #0
  b hang
do_error:
  adrp x0, err_pref
  add x0, x0, :lo12:err_pref
  adrp x1, outbuf
  add x1, x1, :lo12:outbuf
  mov x10, x1
  mov x2, #9
de_pref:
  ldrb w3, [x0], #1
  strb w3, [x1], #1
  subs x2, x2, #1
  b.ne de_pref
  mov x4, x21
  adrp x5, digs
  add x5, x5, :lo12:digs
  add x5, x5, #16
  mov x6, x5
  cbz x4, de_zero
  mov x7, #10
de_cd:
  udiv x8, x4, x7
  msub x9, x8, x7, x4
  add w9, w9, #CH_0
  sub x5, x5, #1
  strb w9, [x5]
  mov x4, x8
  cbnz x4, de_cd
  b de_app
de_zero:
  sub x5, x5, #1
  mov w3, #CH_0
  strb w3, [x5]
de_app:
  cmp x5, x6
  b.hs de_nl
  ldrb w3, [x5], #1
  strb w3, [x1], #1
  b de_app
de_nl:
  mov w3, #CH_LF
  strb w3, [x1], #1
  sub x2, x1, x10
  mov x0, #1
  mov x1, x10
  mov x8, #SYS_write
  svc #0
  mov x0, #1
  mov x8, #SYS_exit
  svc #0
hang:
  b hang

skip_ws:
sw_loop:
  cmp x21, x20
  b.hs sw_done
  ldrb w0, [x19, x21]
  cmp w0, #CH_SP
  b.eq sw_adv
  cmp w0, #CH_TAB
  b.eq sw_adv
  cmp w0, #CH_LF
  b.eq sw_adv
  cmp w0, #CH_CR
  b.eq sw_adv
sw_done:
  ret
sw_adv:
  add x21, x21, #1
  b sw_loop

parse_value:
  stp x29, x30, [sp, #-16]!
  cbnz x22, pv_ret
  cmp x21, x20
  b.hs pv_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_QUOT
  b.eq pv_str
  cmp w0, #CH_LBRACE
  b.eq pv_obj
  cmp w0, #CH_LBR
  b.eq pv_arr
  cmp w0, #CH_t
  b.eq pv_true
  cmp w0, #CH_f
  b.eq pv_false
  cmp w0, #CH_n
  b.eq pv_null
  cmp w0, #CH_MINUS
  b.eq pv_num
  cmp w0, #CH_0
  b.lo pv_err
  cmp w0, #CH_9
  b.ls pv_num
pv_err:
  mov x22, #1
  b pv_ret
pv_str:
  bl parse_string
  b pv_ret
pv_obj:
  bl parse_object
  b pv_ret
pv_arr:
  bl parse_array
  b pv_ret
pv_num:
  bl parse_number
  b pv_ret
pv_true:
  adrp x0, lit_true
  add x0, x0, :lo12:lit_true
  mov x1, #4
  bl parse_lit
  b pv_ret
pv_false:
  adrp x0, lit_false
  add x0, x0, :lo12:lit_false
  mov x1, #5
  bl parse_lit
  b pv_ret
pv_null:
  adrp x0, lit_null
  add x0, x0, :lo12:lit_null
  mov x1, #4
  bl parse_lit
pv_ret:
  ldp x29, x30, [sp], #16
  ret

parse_lit:
  mov x2, #0
pl_loop:
  cmp x2, x1
  b.hs pl_ok
  cmp x21, x20
  b.hs pl_err
  ldrb w3, [x19, x21]
  ldrb w4, [x0, x2]
  cmp w3, w4
  b.ne pl_err
  add x21, x21, #1
  add x2, x2, #1
  b pl_loop
pl_ok:
  ret
pl_err:
  mov x22, #1
  ret

parse_string:
  cmp x21, x20
  b.hs ps_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_QUOT
  b.ne ps_err
  add x21, x21, #1
ps_loop:
  cmp x21, x20
  b.hs ps_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_QUOT
  b.eq ps_end
  cmp w0, #CH_BS
  b.eq ps_esc
  cmp w0, #CH_SP
  b.lo ps_err
  add x21, x21, #1
  b ps_loop
ps_end:
  add x21, x21, #1
  ret
ps_esc:
  add x21, x21, #1
  cmp x21, x20
  b.hs ps_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_QUOT
  b.eq ps_escok
  cmp w0, #CH_BS
  b.eq ps_escok
  cmp w0, #CH_SLASH
  b.eq ps_escok
  cmp w0, #CH_b
  b.eq ps_escok
  cmp w0, #CH_f
  b.eq ps_escok
  cmp w0, #CH_n
  b.eq ps_escok
  cmp w0, #CH_r
  b.eq ps_escok
  cmp w0, #CH_t
  b.eq ps_escok
  cmp w0, #CH_u
  b.eq ps_u
  b ps_err
ps_escok:
  add x21, x21, #1
  b ps_loop
ps_u:
  add x21, x21, #1
  mov x1, #4
ps_uloop:
  cmp x21, x20
  b.hs ps_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_0
  b.lo ps_err
  cmp w0, #CH_9
  b.ls ps_hexok
  cmp w0, #CH_A
  b.lo ps_err
  cmp w0, #CH_F
  b.ls ps_hexok
  cmp w0, #CH_a
  b.lo ps_err
  cmp w0, #CH_f
  b.hi ps_err
ps_hexok:
  add x21, x21, #1
  subs x1, x1, #1
  b.ne ps_uloop
  b ps_loop
ps_err:
  mov x22, #1
  ret

parse_number:
  cmp x21, x20
  b.hs pn_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_MINUS
  b.ne pn_int
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_err
  ldrb w0, [x19, x21]
pn_int:
  cmp w0, #CH_0
  b.lo pn_err
  cmp w0, #CH_9
  b.hi pn_err
  add x21, x21, #1
  cmp w0, #CH_0
  b.eq pn_after_int
pn_more_digs:
  cmp x21, x20
  b.hs pn_ok
  ldrb w1, [x19, x21]
  cmp w1, #CH_0
  b.lo pn_after_int
  cmp w1, #CH_9
  b.hi pn_after_int
  add x21, x21, #1
  b pn_more_digs
pn_after_int:
  cmp x21, x20
  b.hs pn_ok
  ldrb w0, [x19, x21]
  cmp w0, #CH_DOT
  b.ne pn_exp
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_0
  b.lo pn_err
  cmp w0, #CH_9
  b.hi pn_err
pn_frac_digs:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_ok
  ldrb w0, [x19, x21]
  cmp w0, #CH_0
  b.lo pn_exp
  cmp w0, #CH_9
  b.hi pn_exp
  b pn_frac_digs
pn_exp:
  cmp x21, x20
  b.hs pn_ok
  ldrb w0, [x19, x21]
  cmp w0, #CH_e
  b.eq pn_exp2
  cmp w0, #CH_E
  b.ne pn_ok
pn_exp2:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_PLUS
  b.eq pn_expsign
  cmp w0, #CH_MINUS
  b.ne pn_expdig
pn_expsign:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_err
  ldrb w0, [x19, x21]
pn_expdig:
  cmp w0, #CH_0
  b.lo pn_err
  cmp w0, #CH_9
  b.hi pn_err
pn_exp_more:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_ok
  ldrb w0, [x19, x21]
  cmp w0, #CH_0
  b.lo pn_ok
  cmp w0, #CH_9
  b.hi pn_ok
  b pn_exp_more
pn_ok:
  ret
pn_err:
  mov x22, #1
  ret

parse_array:
  stp x29, x30, [sp, #-16]!
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.hs pa_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_RBR
  b.eq pa_close
pa_loop:
  bl parse_value
  cbnz x22, pa_ret
  bl skip_ws
  cmp x21, x20
  b.hs pa_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_RBR
  b.eq pa_close
  cmp w0, #CH_COMMA
  b.ne pa_err
  add x21, x21, #1
  bl skip_ws
  b pa_loop
pa_close:
  add x21, x21, #1
  b pa_ret
pa_err:
  mov x22, #1
pa_ret:
  ldp x29, x30, [sp], #16
  ret

parse_object:
  stp x29, x30, [sp, #-16]!
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.hs po_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_RBRACE
  b.eq po_close
po_loop:
  bl parse_string
  cbnz x22, po_ret
  bl skip_ws
  cmp x21, x20
  b.hs po_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_COLON
  b.ne po_err
  add x21, x21, #1
  bl skip_ws
  bl parse_value
  cbnz x22, po_ret
  bl skip_ws
  cmp x21, x20
  b.hs po_err
  ldrb w0, [x19, x21]
  cmp w0, #CH_RBRACE
  b.eq po_close
  cmp w0, #CH_COMMA
  b.ne po_err
  add x21, x21, #1
  bl skip_ws
  b po_loop
po_close:
  add x21, x21, #1
  b po_ret
po_err:
  mov x22, #1
po_ret:
  ldp x29, x30, [sp], #16
  ret
