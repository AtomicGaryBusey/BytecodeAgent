.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ ASCII_TAB, 9
.equ ASCII_LF, 10
.equ ASCII_SP, 32
.equ ASCII_PCT, 37
.equ ASCII_STAR, 42
.equ ASCII_PLUS, 43
.equ ASCII_MINUS, 45
.equ ASCII_SLASH, 47
.equ ASCII_0, 48
.equ DIGIT_MAX, 18
.equ STK_CAP, 1024

.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    movz x9, #1, lsl #16
    add x21, x19, x9
read_loop:
    cmp x20, x21
    b.hs read_done
    mov x0, #0
    mov x1, x20
    sub x2, x21, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x22, evalstk
    add x22, x22, :lo12:evalstk
    mov x21, x19
line_loop:
    cmp x21, x20
    b.hs exit_ok
    mov x0, x21
find_nl:
    cmp x21, x20
    b.hs have_line
    ldrb w2, [x21]
    cmp w2, #ASCII_LF
    b.eq have_nl
    add x21, x21, #1
    b find_nl
have_nl:
    mov x1, x21
    add x21, x21, #1
    bl process_line
    b line_loop
have_line:
    mov x1, x20
    bl process_line
exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b exit_ok

process_line:
    stp x29, x30, [sp, #-16]!
    mov x23, x0
    mov x24, x1
    mov x25, #0
tok_loop:
    cmp x23, x24
    b.hs finish_line
    ldrb w2, [x23]
    cmp w2, #ASCII_SP
    b.eq skip_one
    cmp w2, #ASCII_TAB
    b.eq skip_one
    b got_tok
skip_one:
    add x23, x23, #1
    b tok_loop
got_tok:
    mov x26, x23
scan_tok:
    add x23, x23, #1
    cmp x23, x24
    b.hs classify
    ldrb w2, [x23]
    cmp w2, #ASCII_SP
    b.eq classify
    cmp w2, #ASCII_TAB
    b.eq classify
    b scan_tok
classify:
    sub x3, x23, x26
    cmp x3, #1
    b.ne parse_num
    ldrb w2, [x26]
    cmp w2, #ASCII_PLUS
    b.eq op_add
    cmp w2, #ASCII_MINUS
    b.eq op_sub
    cmp w2, #ASCII_STAR
    b.eq op_mul
    cmp w2, #ASCII_SLASH
    b.eq op_div
    cmp w2, #ASCII_PCT
    b.eq op_mod
parse_num:
    mov x4, x26
    mov w5, #0
    ldrb w2, [x4]
    cmp w2, #ASCII_MINUS
    b.ne no_minus
    mov w5, #1
    add x4, x4, #1
no_minus:
    sub x3, x23, x4
    cbz x3, line_err
    cmp x3, #DIGIT_MAX
    b.hi line_err
    mov x6, #0
    mov x7, #10
dig_loop:
    cmp x4, x23
    b.hs dig_done
    ldrb w2, [x4]
    sub w8, w2, #ASCII_0
    cmp w8, #9
    b.hi line_err
    mul x6, x6, x7
    add x6, x6, x8
    add x4, x4, #1
    b dig_loop
dig_done:
    cbz w5, push_val
    neg x6, x6
    b push_val
push_val:
    cmp x25, #STK_CAP
    b.hs line_err
    str x6, [x22, x25, lsl #3]
    add x25, x25, #1
    b tok_loop
op_add:
    bl pop_two
    cbnz x0, line_err
    add x6, x10, x11
    b push_val
op_sub:
    bl pop_two
    cbnz x0, line_err
    sub x6, x10, x11
    b push_val
op_mul:
    bl pop_two
    cbnz x0, line_err
    mul x6, x10, x11
    b push_val
op_div:
    bl pop_two
    cbnz x0, line_err
    cbz x11, line_err
    mov x12, #1
    lsl x12, x12, #63
    cmp x10, x12
    b.ne div_ok
    cmn x11, #1
    b.eq line_err
div_ok:
    sdiv x6, x10, x11
    b push_val
op_mod:
    bl pop_two
    cbnz x0, line_err
    cbz x11, line_err
    mov x12, #1
    lsl x12, x12, #63
    cmp x10, x12
    b.ne mod_ok
    cmn x11, #1
    b.eq line_err
mod_ok:
    sdiv x6, x10, x11
    msub x6, x6, x11, x10
    b push_val
finish_line:
    cmp x25, #1
    b.ne line_err
    ldr x0, [x22]
    bl print_i64
    b line_done
line_err:
    bl print_err
line_done:
    ldp x29, x30, [sp], #16
    ret

pop_two:
    cmp x25, #2
    b.lo pop_fail
    sub x25, x25, #1
    ldr x11, [x22, x25, lsl #3]
    sub x25, x25, #1
    ldr x10, [x22, x25, lsl #3]
    mov x0, #0
    ret
pop_fail:
    mov x0, #1
    ret

print_i64:
    sub sp, sp, #64
    add x1, sp, #48
    cmp x0, #0
    b.ge p_nonneg
    mov w4, #1
    mov x12, #1
    lsl x12, x12, #63
    cmp x0, x12
    b.eq p_digits
    neg x0, x0
    b p_digits
p_nonneg:
    mov w4, #0
p_digits:
    mov x5, x1
    mov x7, #10
p_loop:
    udiv x6, x0, x7
    msub x8, x6, x7, x0
    add w8, w8, #ASCII_0
    sub x5, x5, #1
    strb w8, [x5]
    mov x0, x6
    cbnz x0, p_loop
    cbz w4, p_nl
    sub x5, x5, #1
    mov w8, #ASCII_MINUS
    strb w8, [x5]
p_nl:
    mov w8, #ASCII_LF
    strb w8, [x1]
    add x2, x1, #1
    sub x2, x2, x5
    mov x0, #1
    mov x1, x5
    mov x8, #SYS_write
    svc #0
    add sp, sp, #64
    ret

print_err:
    adrp x1, errmsg
    add x1, x1, :lo12:errmsg
    mov x0, #1
    mov x2, #6
    mov x8, #SYS_write
    svc #0
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 8
inbuf:
    .zero 65536
.balign 8
evalstk:
    .zero 8192
