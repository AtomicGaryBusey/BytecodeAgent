.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.data
nl:
    .byte 0x0A

.bss
.balign 8
inbuf:
    .zero 65536

.text
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, xzr

read_loop:
    mov x0, xzr
    add x1, x20, x19
    mov x2, #BUF_SIZE
    sub x2, x2, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    cmp x19, #BUF_SIZE
    b.lo read_loop

read_done:
    mov x21, xzr
    mov x22, xzr

scan_loop:
    cmp x22, x19
    b.hs scan_done
    ldrb w3, [x20, x22]
    add x22, x22, #1
    cmp w3, #0x0A
    b.ne scan_loop
    sub x23, x22, x21
    sub x23, x23, #1
    bl emit_line
    mov x21, x22
    b scan_loop

scan_done:
    cmp x21, x19
    b.hs do_exit
    sub x23, x19, x21
    bl emit_line

do_exit:
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0
    b do_exit

emit_line:
    cbz x23, emit_nl
    sub x25, x23, #1
    mov x24, xzr

rev_loop:
    cmp x24, x25
    b.hs emit_body
    add x0, x20, x21
    ldrb w1, [x0, x24]
    ldrb w2, [x0, x25]
    strb w2, [x0, x24]
    strb w1, [x0, x25]
    add x24, x24, #1
    sub x25, x25, #1
    b rev_loop

emit_body:
    mov x0, #1
    add x1, x20, x21
    mov x2, x23
    mov x8, #SYS_write
    svc #0

emit_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    ret
