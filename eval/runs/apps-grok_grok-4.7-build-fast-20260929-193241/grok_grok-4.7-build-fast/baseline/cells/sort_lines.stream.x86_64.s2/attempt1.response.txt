.globl _start

.text

_start:
  xor r12, r12
  lea r14, [rip + buf]

read_more:
  xor eax, eax
  xor edi, edi
  lea rsi, [r14 + r12]
  mov edx, 32768
  sub edx, r12d
  syscall
  test rax, rax
  jle read_done
  add r12, rax
  cmp r12, 32768
  jb read_more

read_done:
  xor r13, r13
  xor rcx, rcx
  xor r8, r8
  test r12, r12
  jz print_lines

parse_loop:
  cmp rcx, r12
  jae parse_end
  movzx eax, byte ptr [r14 + rcx]
  inc rcx
  cmp al, 10
  jne parse_loop
  mov rdx, rcx
  dec rdx
  sub rdx, r8
  lea rdi, [rip + starts]
  mov qword ptr [rdi + r13*8], r8
  lea rdi, [rip + lens]
  mov qword ptr [rdi + r13*8], rdx
  lea rdi, [rip + order]
  mov qword ptr [rdi + r13*8], r13
  inc r13
  mov r8, rcx
  jmp parse_loop

parse_end:
  cmp r8, r12
  jae do_sort
  mov rdx, r12
  sub rdx, r8
  lea rdi, [rip + starts]
  mov qword ptr [rdi + r13*8], r8
  lea rdi, [rip + lens]
  mov qword ptr [rdi + r13*8], rdx
  lea rdi, [rip + order]
  mov qword ptr [rdi + r13*8], r13
  inc r13

do_sort:
  cmp r13, 2
  jb print_lines
  mov r15, 1

sort_i:
  cmp r15, r13
  jae print_lines
  lea rax, [rip + order]
  mov rbx, qword ptr [rax + r15*8]
  mov r12, r15

sort_j:
  test r12, r12
  jz sort_put
  mov rdx, r12
  dec rdx
  lea rax, [rip + order]
  mov rdi, qword ptr [rax + rdx*8]
  mov rsi, rbx
  call cmp_lines
  test rax, rax
  jle sort_put
  mov rdx, r12
  dec rdx
  lea rax, [rip + order]
  mov rsi, qword ptr [rax + rdx*8]
  mov qword ptr [rax + r12*8], rsi
  dec r12
  jmp sort_j

sort_put:
  lea rax, [rip + order]
  mov qword ptr [rax + r12*8], rbx
  inc r15
  jmp sort_i

print_lines:
  xor r15, r15

print_loop:
  cmp r15, r13
  jae do_exit
  lea rax, [rip + order]
  mov rbx, qword ptr [rax + r15*8]
  lea rax, [rip + starts]
  mov r12, qword ptr [rax + rbx*8]
  lea rax, [rip + lens]
  mov rdx, qword ptr [rax + rbx*8]
  mov eax, 1
  mov edi, 1
  lea rsi, [r14 + r12]
  syscall
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + nl]
  mov edx, 1
  syscall
  inc r15
  jmp print_loop

do_exit:
  mov eax, 60
  xor edi, edi
  syscall

# rax > 0 if line rdi sorts after line rsi, < 0 if before, 0 if equal
cmp_lines:
  lea r8, [rip + starts]
  lea r9, [rip + lens]
  mov r10, qword ptr [r8 + rdi*8]
  mov rax, qword ptr [r9 + rdi*8]
  mov r11, qword ptr [r8 + rsi*8]
  mov rdx, qword ptr [r9 + rsi*8]
  mov rcx, rax
  cmp rcx, rdx
  cmova rcx, rdx
  lea rdi, [r14 + r10]
  lea rsi, [r14 + r11]
  test rcx, rcx
  jz cmp_len

cmp_loop:
  movzx r8d, byte ptr [rdi]
  movzx r9d, byte ptr [rsi]
  cmp r8d, r9d
  jne cmp_ne
  inc rdi
  inc rsi
  dec rcx
  jnz cmp_loop

cmp_len:
  sub rax, rdx
  ret

cmp_ne:
  mov rax, r8
  sub rax, r9
  ret

.section .rodata
nl:
  .byte 10

.bss
.balign 8
buf:
  .zero 32768
starts:
  .zero 8000
lens:
  .zero 8000
order:
  .zero 8000
