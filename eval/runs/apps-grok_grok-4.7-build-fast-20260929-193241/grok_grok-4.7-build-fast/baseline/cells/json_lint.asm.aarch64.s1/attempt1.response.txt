.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536

.text
.balign 4
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #IN_MAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    bl parse_document
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
ok_halt:
    b ok_halt

report_error:
    adrp x10, numbuf
    add x10, x10, :lo12:numbuf
    add x12, x10, #20
    mov x11, x12
    mov x4, x21
    cbnz x4, re_digits
    sub x11, x11, #1
    mov w5, #48
    strb w5, [x11]
    b re_ready
re_digits:
    mov x5, #10
    udiv x6, x4, x5
    msub x7, x6, x5, x4
    add w7, w7, #48
    sub x11, x11, #1
    strb w7, [x11]
    mov x4, x6
    cbnz x4, re_digits
re_ready:
    mov x19, x11
    sub x20, x12, x11
    mov x0, #1
    adrp x1, err_prefix
    add x1, x1, :lo12:err_prefix
    mov x2, #9
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    adrp x1, nl_char
    add x1, x1, :lo12:nl_char
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
err_halt:
    b err_halt

parse_document:
    str x30, [sp, #-16]!
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne report_error
    ldr x30, [sp], #16
    ret

skip_ws:
ws_loop:
    cmp x21, x20
    b.hs ws_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq ws_yes
    cmp w0, #0x09
    b.eq ws_yes
    cmp w0, #0x0A
    b.eq ws_yes
    cmp w0, #0x0D
    b.ne ws_done
ws_yes:
    add x21, x21, #1
    b ws_loop
ws_done:
    ret

parse_value:
    str x30, [sp, #-16]!
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x7B
    b.eq pv_object
    cmp w0, #0x5B
    b.eq pv_array
    cmp w0, #0x22
    b.eq pv_string
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    cmp w0, #0x2D
    b.eq pv_number
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
pv_number:
    bl parse_number
    b pv_done
pv_object:
    bl parse_object
    b pv_done
pv_array:
    bl parse_array
    b pv_done
pv_string:
    bl parse_string
    b pv_done
pv_true:
    bl parse_true
    b pv_done
pv_false:
    bl parse_false
    b pv_done
pv_null:
    bl parse_null
pv_done:
    ldr x30, [sp], #16
    ret

parse_object:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x7D
    b.eq obj_close
obj_loop:
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x3A
    b.ne report_error
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.eq obj_comma
    cmp w0, #0x7D
    b.ne report_error
obj_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret
obj_comma:
    add x21, x21, #1
    bl skip_ws
    b obj_loop

parse_array:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x5D
    b.eq arr_close
arr_loop:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.eq arr_comma
    cmp w0, #0x5D
    b.ne report_error
arr_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret
arr_comma:
    add x21, x21, #1
    bl skip_ws
    b arr_loop

parse_string:
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne report_error
    add x21, x21, #1
str_loop:
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.lo report_error
    cmp w0, #0x22
    b.eq str_close
    cmp w0, #0x5C
    b.eq str_esc
    add x21, x21, #1
    b str_loop
str_close:
    add x21, x21, #1
    ret
str_esc:
    add x21, x21, #1
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq str_esc_ok
    cmp w0, #0x5C
    b.eq str_esc_ok
    cmp w0, #0x2F
    b.eq str_esc_ok
    cmp w0, #'b'
    b.eq str_esc_ok
    cmp w0, #'f'
    b.eq str_esc_ok
    cmp w0, #'n'
    b.eq str_esc_ok
    cmp w0, #'r'
    b.eq str_esc_ok
    cmp w0, #'t'
    b.eq str_esc_ok
    cmp w0, #'u'
    b.eq str_u
    b report_error
str_esc_ok:
    add x21, x21, #1
    b str_loop
str_u:
    add x21, x21, #1
    mov x1, #4
str_u_loop:
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.le str_hex_ok
    cmp w0, #'A'
    b.lt report_error
    cmp w0, #'F'
    b.le str_hex_ok
    cmp w0, #'a'
    b.lt report_error
    cmp w0, #'f'
    b.gt report_error
str_hex_ok:
    add x21, x21, #1
    subs x1, x1, #1
    b.ne str_u_loop
    b str_loop

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #0x2D
    b.ne pn_int
    add x21, x21, #1
    cmp x21, x20
    b.hs report_error
pn_int:
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.eq pn_zero
    cmp w0, #'1'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
    add x21, x21, #1
pn_digits:
    cmp x21, x20
    b.hs pn_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt pn_frac
    cmp w0, #'9'
    b.gt pn_frac
    add x21, x21, #1
    b pn_digits
pn_zero:
    add x21, x21, #1
pn_frac:
    cmp x21, x20
    b.hs pn_done
    ldrb w0, [x19, x21]
    cmp w0, #0x2E
    b.ne pn_exp
    add x21, x21, #1
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
pn_frac_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs pn_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt pn_exp
    cmp w0, #'9'
    b.gt pn_exp
    b pn_frac_digits
pn_exp:
    cmp x21, x20
    b.hs pn_done
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.eq pn_exp_body
    cmp w0, #'E'
    b.ne pn_done
pn_exp_body:
    add x21, x21, #1
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, #0x2B
    b.eq pn_exp_sign
    cmp w0, #0x2D
    b.ne pn_exp_first
pn_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs report_error
pn_exp_first:
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
pn_exp_more:
    add x21, x21, #1
    cmp x21, x20
    b.hs pn_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt pn_done
    cmp w0, #'9'
    b.gt pn_done
    b pn_exp_more
pn_done:
    ret

parse_true:
    str x30, [sp, #-16]!
    mov w3, #'t'
    bl expect_byte
    mov w3, #'r'
    bl expect_byte
    mov w3, #'u'
    bl expect_byte
    mov w3, #'e'
    bl expect_byte
    ldr x30, [sp], #16
    ret

parse_false:
    str x30, [sp, #-16]!
    mov w3, #'f'
    bl expect_byte
    mov w3, #'a'
    bl expect_byte
    mov w3, #'l'
    bl expect_byte
    mov w3, #'s'
    bl expect_byte
    mov w3, #'e'
    bl expect_byte
    ldr x30, [sp], #16
    ret

parse_null:
    str x30, [sp, #-16]!
    mov w3, #'n'
    bl expect_byte
    mov w3, #'u'
    bl expect_byte
    mov w3, #'l'
    bl expect_byte
    mov w3, #'l'
    bl expect_byte
    ldr x30, [sp], #16
    ret

expect_byte:
    cmp x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp w0, w3
    b.ne report_error
    add x21, x21, #1
    ret

.section .rodata
.balign 4
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
nl_char:
    .ascii "\n"

.section .bss
.balign 8
inbuf:
    .zero 65536
numbuf:
    .zero 32
