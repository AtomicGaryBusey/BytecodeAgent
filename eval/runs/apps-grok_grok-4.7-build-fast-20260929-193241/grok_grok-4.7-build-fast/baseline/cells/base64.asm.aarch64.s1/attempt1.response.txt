.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ MAX_IN, 49152
.equ OUT_SIZE, 66560

.globl _start
.text

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #MAX_IN

read_loop:
    mov x8, #SYS_READ
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, x21
    b.lo read_loop

read_done:
    cbz x20, exit_ok

    adrp x23, alphabet
    add x23, x23, :lo12:alphabet
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, #0
    add x24, x19, x20

encode_loop:
    sub x0, x24, x19
    cmp x0, #3
    b.lo encode_tail

    ldrb w1, [x19]
    ldrb w2, [x19, #1]
    ldrb w3, [x19, #2]
    add x19, x19, #3

    lsr w0, w1, #2
    bl lookup_emit

    and w0, w1, #3
    lsl w0, w0, #4
    lsr w4, w2, #4
    orr w0, w0, w4
    bl lookup_emit

    and w0, w2, #15
    lsl w0, w0, #2
    lsr w4, w3, #6
    orr w0, w0, w4
    bl lookup_emit

    and w0, w3, #63
    bl lookup_emit
    b encode_loop

encode_tail:
    cbz x0, finish_line
    cmp x0, #1
    b.eq encode_one

    ldrb w1, [x19]
    ldrb w2, [x19, #1]

    lsr w0, w1, #2
    bl lookup_emit

    and w0, w1, #3
    lsl w0, w0, #4
    lsr w4, w2, #4
    orr w0, w0, w4
    bl lookup_emit

    and w0, w2, #15
    lsl w0, w0, #2
    bl lookup_emit

    mov w0, #61
    bl emit_byte
    b finish_line

encode_one:
    ldrb w1, [x19]

    lsr w0, w1, #2
    bl lookup_emit

    and w0, w1, #3
    lsl w0, w0, #4
    bl lookup_emit

    mov w0, #61
    bl emit_byte
    mov w0, #61
    bl emit_byte

finish_line:
    cbz x22, do_write
    mov w0, #10
    strb w0, [x21], #1

do_write:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x21, x1
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    b exit_ok

lookup_emit:
    ldrb w0, [x23, x0]
emit_byte:
    strb w0, [x21], #1
    add x22, x22, #1
    cmp x22, #76
    b.ne 1f
    mov w0, #10
    strb w0, [x21], #1
    mov x22, #0
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 8
inbuf:
    .zero MAX_IN
outbuf:
    .zero OUT_SIZE
