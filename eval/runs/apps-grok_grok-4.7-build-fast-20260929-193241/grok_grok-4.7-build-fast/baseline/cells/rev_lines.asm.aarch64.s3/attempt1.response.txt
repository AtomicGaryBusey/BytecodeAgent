.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .rodata
nl:
    .byte 0x0A

.bss
.align 3
inbuf:
    .zero 65536

.text
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, xzr
    movz x23, #1, lsl #16

read_more:
    mov x0, xzr
    add x1, x20, x19
    sub x2, x23, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le have_input
    add x19, x19, x0
    cmp x19, x23
    b.lo read_more

have_input:
    cbz x19, do_exit
    mov x21, xzr

next_line:
    mov x22, x21

scan:
    cmp x22, x19
    b.eq emit_final
    ldrb w0, [x20, x22]
    cmp w0, #0x0A
    b.eq emit_line
    add x22, x22, #1
    b scan

emit_line:
    bl rev_and_print
    add x21, x22, #1
    cmp x21, x19
    b.lo next_line
    b do_exit

emit_final:
    cmp x21, x19
    b.hs do_exit
    bl rev_and_print
    b do_exit

// Reverse inbuf[x21, x22) and write it, then write one newline.
rev_and_print:
    cmp x21, x22
    b.eq just_nl
    mov x3, x21
    sub x4, x22, #1
1:
    cmp x3, x4
    b.hs 2f
    ldrb w5, [x20, x3]
    ldrb w6, [x20, x4]
    strb w6, [x20, x3]
    strb w5, [x20, x4]
    add x3, x3, #1
    sub x4, x4, #1
    b 1b
2:
    mov x0, #1
    add x1, x20, x21
    sub x2, x22, x21
    mov x8, #SYS_write
    svc #0
just_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    ret

do_exit:
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0
