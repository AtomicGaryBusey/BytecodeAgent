.globl _start

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 66560

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 49152
    jb read_loop

read_done:
    test r12, r12
    jz do_exit

    lea rbx, [rip + inbuf]
    lea r14, [rip + outbuf]
    xor r13d, r13d

encode_loop:
    cmp r12, 3
    jb tail

    movzx eax, byte ptr [rbx]
    movzx ecx, byte ptr [rbx + 1]
    movzx edx, byte ptr [rbx + 2]
    add rbx, 3
    sub r12, 3

    mov esi, eax
    shr esi, 2
    call emit_idx

    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ecx
    shr edi, 4
    or esi, edi
    call emit_idx

    mov esi, ecx
    and esi, 15
    shl esi, 2
    mov edi, edx
    shr edi, 6
    or esi, edi
    call emit_idx

    mov esi, edx
    and esi, 63
    call emit_idx

    jmp encode_loop

tail:
    cmp r12, 1
    jb finish
    je one_byte

    movzx eax, byte ptr [rbx]
    movzx ecx, byte ptr [rbx + 1]

    mov esi, eax
    shr esi, 2
    call emit_idx

    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ecx
    shr edi, 4
    or esi, edi
    call emit_idx

    mov esi, ecx
    and esi, 15
    shl esi, 2
    call emit_idx

    mov esi, 0x3d
    call emit_raw
    jmp finish

one_byte:
    movzx eax, byte ptr [rbx]

    mov esi, eax
    shr esi, 2
    call emit_idx

    mov esi, eax
    and esi, 3
    shl esi, 4
    call emit_idx

    mov esi, 0x3d
    call emit_raw
    mov esi, 0x3d
    call emit_raw

finish:
    test r13d, r13d
    jz do_write
    mov byte ptr [r14], 0x0a
    inc r14

do_write:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit_idx:
    lea rax, [rip + alpha]
    movzx esi, byte ptr [rax + rsi]

emit_raw:
    mov byte ptr [r14], sil
    inc r14
    inc r13d
    cmp r13d, 76
    jne emit_done
    mov byte ptr [r14], 0x0a
    inc r14
    xor r13d, r13d

emit_done:
    ret
