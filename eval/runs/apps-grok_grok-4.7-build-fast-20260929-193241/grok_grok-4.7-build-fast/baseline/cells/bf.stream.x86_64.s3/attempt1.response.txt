.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_MAX, 65536
.equ TAPE_LEN, 30000
.equ MAX_DEPTH, 1000
.equ JUMP_BYTES, 262144
.equ STACK_BYTES, 8000
.equ DISP_BYTES, 2048

.globl _start
.text

_start:
  lea rdi, [rip + dispatch]
  lea rax, [rip + op_nop]
  mov ecx, 256
40:
  mov qword ptr [rdi], rax
  add rdi, 8
  dec ecx
  jnz 40b
  lea rcx, [rip + dispatch]
  lea rax, [rip + op_plus]
  mov edx, 0x2b
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_comma]
  mov edx, 0x2c
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_minus]
  mov edx, 0x2d
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_dot]
  mov edx, 0x2e
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_lt]
  mov edx, 0x3c
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_gt]
  mov edx, 0x3e
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_lbr]
  mov edx, 0x5b
  mov qword ptr [rcx + rdx*8], rax
  lea rax, [rip + op_rbr]
  mov edx, 0x5d
  mov qword ptr [rcx + rdx*8], rax
  mov rbp, rcx

  lea r8, [rip + inbuf]
  xor r12, r12
1:
  mov eax, SYS_read
  xor edi, edi
  lea rsi, [r8 + r12]
  mov edx, IN_MAX
  sub rdx, r12
  syscall
  test rax, rax
  jle 2f
  add r12, rax
  cmp r12, IN_MAX
  jb 1b
2:
  xor ecx, ecx
3:
  cmp rcx, r12
  jae 4f
  cmp byte ptr [r8 + rcx], 0x21
  je 5f
  inc rcx
  jmp 3b
5:
  mov r13, rcx
  lea r14, [r8 + rcx + 1]
  lea r15, [r8 + r12]
  jmp 6f
4:
  mov r13, r12
  lea r14, [r8 + r12]
  mov r15, r14
6:
  # Compact commands in place and check brackets. r13 is the source length.
  lea r9, [rip + jumps]
  lea rdi, [rip + brstack]
  xor esi, esi
  xor ecx, ecx
  xor r11, r11
7:
  cmp rcx, r13
  jae 8f
  movzx eax, byte ptr [r8 + rcx]
  inc rcx
  cmp al, 0x2b
  je 12f
  cmp al, 0x2d
  je 12f
  cmp al, 0x3c
  je 12f
  cmp al, 0x3e
  je 12f
  cmp al, 0x2e
  je 12f
  cmp al, 0x2c
  je 12f
  cmp al, 0x5b
  je 13f
  cmp al, 0x5d
  je 14f
  jmp 7b
12:
  mov byte ptr [r8 + r11], al
  inc r11
  jmp 7b
13:
  cmp esi, MAX_DEPTH
  jae 11f
  mov qword ptr [rdi + rsi*8], r11
  inc esi
  mov byte ptr [r8 + r11], al
  inc r11
  jmp 7b
14:
  test esi, esi
  jz 11f
  dec esi
  mov rdx, qword ptr [rdi + rsi*8]
  mov byte ptr [r8 + r11], al
  mov dword ptr [r9 + r11*4], edx
  mov dword ptr [r9 + rdx*4], r11d
  inc r11
  jmp 7b
8:
  test esi, esi
  jnz 11f
  mov r13, r11
  lea r10, [rip + tape]
  lea r11, [rip + outbuf]
  xor ebx, ebx
  xor r12, r12
20:
  cmp rbx, r13
  jae 21f
  movzx eax, byte ptr [r8 + rbx]
  inc rbx
  jmp qword ptr [rbp + rax*8]

op_nop:
  jmp 20b
op_gt:
  inc r10
  jmp 20b
op_lt:
  dec r10
  jmp 20b
op_plus:
  inc byte ptr [r10]
  jmp 20b
op_minus:
  dec byte ptr [r10]
  jmp 20b
op_dot:
  mov al, byte ptr [r10]
  mov byte ptr [r11 + r12], al
  inc r12
  jmp 20b
op_comma:
  cmp r14, r15
  jae 38f
  mov al, byte ptr [r14]
  mov byte ptr [r10], al
  inc r14
  jmp 20b
38:
  mov byte ptr [r10], 0
  jmp 20b
op_lbr:
  cmp byte ptr [r10], 0
  jne 20b
  dec rbx
  mov eax, dword ptr [r9 + rbx*4]
  lea rbx, [rax + 1]
  jmp 20b
op_rbr:
  cmp byte ptr [r10], 0
  je 20b
  dec rbx
  mov eax, dword ptr [r9 + rbx*4]
  lea rbx, [rax + 1]
  jmp 20b
21:
  mov eax, SYS_write
  mov edi, 1
  mov rsi, r11
  mov rdx, r12
  syscall
  mov eax, SYS_exit
  xor edi, edi
  syscall
11:
  mov eax, SYS_exit
  mov edi, 1
  syscall

.bss
.align 8
dispatch:
  .zero DISP_BYTES
inbuf:
  .zero IN_MAX
outbuf:
  .zero IN_MAX
tape:
  .zero TAPE_LEN
.align 8
jumps:
  .zero JUMP_BYTES
.align 8
brstack:
  .zero STACK_BYTES
