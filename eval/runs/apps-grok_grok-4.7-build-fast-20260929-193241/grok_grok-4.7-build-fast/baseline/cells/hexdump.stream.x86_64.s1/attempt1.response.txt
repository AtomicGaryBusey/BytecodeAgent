.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
input_buf:
    .zero 65536
out_buf:
    .zero 128

.text
_start:
    lea r15, [rip + input_buf]
    lea r8, [rip + hextab]
    xor r12, r12

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle have_input
    add r12, rax
    cmp r12, 65536
    jb read_more

have_input:
    xor r13, r13

next_row:
    cmp r13, r12
    jae print_total
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_n_ok
    mov r14, 16
row_n_ok:
    lea rbx, [r15 + r13]
    lea rdi, [rip + out_buf]
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2
    xor ecx, ecx

byte_loop:
    cmp ecx, 16
    je bytes_done
    cmp ecx, 8
    jne no_mid_space
    mov byte ptr [rdi], 0x20
    inc rdi
no_mid_space:
    cmp rcx, r14
    jae pad_slot
    movzx eax, byte ptr [rbx + rcx]
    call hex2
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp byte_advance
pad_slot:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
byte_advance:
    inc ecx
    jmp byte_loop

bytes_done:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx

asc_loop:
    cmp rcx, r14
    jae asc_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x20
    jb asc_dot
    cmp al, 0x7e
    ja asc_dot
    mov byte ptr [rdi], al
    jmp asc_advance
asc_dot:
    mov byte ptr [rdi], 0x2e
asc_advance:
    inc rdi
    inc ecx
    jmp asc_loop

asc_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    mov rdx, rdi
    lea rsi, [rip + out_buf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add r13, r14
    jmp next_row

print_total:
    lea rdi, [rip + out_buf]
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + out_buf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

# eax = value, write 8 lowercase hex digits at rdi
hex8:
    mov ecx, 8
hex8_loop:
    mov edx, eax
    shr edx, 28
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl eax, 4
    dec ecx
    jnz hex8_loop
    ret

# al = byte, write 2 lowercase hex digits at rdi
hex2:
    movzx eax, al
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    ret
