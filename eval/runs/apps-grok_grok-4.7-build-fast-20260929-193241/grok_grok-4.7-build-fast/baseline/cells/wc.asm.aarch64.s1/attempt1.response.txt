.text
.globl _start

_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr

read_loop:
    mov x0, xzr
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le finish
    cmp x0, #4096
    b.hi finish
    add x21, x21, x0
    mov x3, x0
    mov x4, xzr

scan_loop:
    cmp x4, x3
    b.eq read_loop
    ldrb w5, [x1, x4]
    add x4, x4, #1
    cmp w5, #10
    b.ne check_ws
    add x19, x19, #1
check_ws:
    cmp w5, #32
    b.eq mark_ws
    sub w6, w5, #9
    cmp w6, #4
    b.ls mark_ws
    cbnz x22, scan_loop
    mov x22, #1
    add x20, x20, #1
    b scan_loop
mark_ws:
    mov x22, xzr
    b scan_loop

finish:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x1, x23
    mov x0, x19
    bl u64_to_dec
    mov w2, #32
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x20
    bl u64_to_dec
    mov w2, #32
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x21
    bl u64_to_dec
    mov w2, #10
    strb w2, [x0], #1
    sub x2, x0, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0

exit_loop:
    mov x0, xzr
    mov x8, #93
    svc #0
    b exit_loop

// x0 = value, x1 = destination. Returns updated pointer in x0.
// Preserves x19-x23.
u64_to_dec:
    cbnz x0, u64_nz
    mov w2, #48
    strb w2, [x1], #1
    mov x0, x1
    ret
u64_nz:
    sub sp, sp, #32
    add x3, sp, #32
    mov x4, x3
u64_div:
    mov x5, #10
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add x7, x7, #48
    sub x4, x4, #1
    strb w7, [x4]
    mov x0, x6
    cbnz x0, u64_div
u64_copy:
    ldrb w7, [x4], #1
    strb w7, [x1], #1
    cmp x4, x3
    b.ne u64_copy
    mov x0, x1
    add sp, sp, #32
    ret

.bss
.align 3
inbuf:
    .zero 4096
outbuf:
    .zero 64
