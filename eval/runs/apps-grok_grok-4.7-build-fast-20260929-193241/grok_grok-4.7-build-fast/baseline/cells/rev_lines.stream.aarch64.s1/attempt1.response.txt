.globl _start

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 8
inbuf:
    .zero 65536

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    movz x26, #1, lsl #16
    mov x20, xzr

read_loop:
    mov x0, xzr
    add x1, x19, x20
    sub x2, x26, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, x26
    b.lo read_loop

read_done:
    mov x21, xzr

line_loop:
    cmp x21, x20
    b.hs quit
    mov x22, x21
    mov x23, x21

scan_loop:
    cmp x23, x20
    b.hs emit_last
    ldrb w0, [x19, x23]
    cmp w0, #0x0A
    b.eq emit_line
    add x23, x23, #1
    b scan_loop

emit_line:
    bl reverse_write
    bl write_nl
    add x21, x23, #1
    b line_loop

emit_last:
    bl reverse_write
    bl write_nl

quit:
    mov x0, xzr
    mov x8, #93
    svc #0

// Reverse inbuf[x22:x23) in place, then write those bytes to stdout.
reverse_write:
    cmp x22, x23
    b.hs rw_write
    mov x24, x22
    sub x25, x23, #1
rw_loop:
    cmp x24, x25
    b.hs rw_write
    ldrb w0, [x19, x24]
    ldrb w1, [x19, x25]
    strb w1, [x19, x24]
    strb w0, [x19, x25]
    add x24, x24, #1
    sub x25, x25, #1
    b rw_loop
rw_write:
    sub x2, x23, x22
    cbz x2, rw_ret
    mov x0, #1
    add x1, x19, x22
    mov x8, #64
    svc #0
rw_ret:
    ret

write_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #64
    svc #0
    ret
