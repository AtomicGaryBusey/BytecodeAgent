.text
.globl _start

// x0 = value, x1 = destination (advanced by 8). x26 = hex table.
put_hex8:
    mov w2, #28
1:
    lsr w3, w0, w2
    and w3, w3, #0xf
    ldrb w3, [x26, x3]
    strb w3, [x1], #1
    subs w2, w2, #4
    b.pl 1b
    ret

_start:
    adrp x26, hexdigits
    add x26, x26, :lo12:hexdigits
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, linebuf
    add x22, x22, :lo12:linebuf

    mov x20, #0
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
row_loop:
    cmp x21, x20
    b.hs rows_done
    sub x25, x20, x21
    cmp x25, #16
    b.ls have_n
    mov x25, #16
have_n:
    add x24, x19, x21
    mov x0, x21
    mov x1, x22
    bl put_hex8
    mov w3, #32
    strb w3, [x1], #1
    strb w3, [x1], #1
    mov x4, #0
hex_loop:
    cmp x4, #16
    b.eq hex_done
    cmp x4, x25
    b.hs hex_pad
    ldrb w0, [x24, x4]
    lsr w3, w0, #4
    ldrb w3, [x26, x3]
    strb w3, [x1], #1
    and w3, w0, #0xf
    ldrb w3, [x26, x3]
    strb w3, [x1], #1
    mov w3, #32
    strb w3, [x1], #1
    b hex_next
hex_pad:
    mov w3, #32
    strb w3, [x1], #1
    strb w3, [x1], #1
    strb w3, [x1], #1
hex_next:
    add x4, x4, #1
    cmp x4, #8
    b.ne hex_loop
    mov w3, #32
    strb w3, [x1], #1
    b hex_loop
hex_done:
    mov w3, #32
    strb w3, [x1], #1
    mov w3, #124
    strb w3, [x1], #1
    mov x4, #0
ascii_loop:
    cmp x4, x25
    b.eq ascii_done
    ldrb w0, [x24, x4]
    cmp w0, #32
    b.lo ascii_dot
    cmp w0, #126
    b.hi ascii_dot
    strb w0, [x1], #1
    b ascii_next
ascii_dot:
    mov w0, #46
    strb w0, [x1], #1
ascii_next:
    add x4, x4, #1
    b ascii_loop
ascii_done:
    mov w3, #124
    strb w3, [x1], #1
    mov w3, #10
    strb w3, [x1], #1
    sub x2, x1, x22
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0
    add x21, x21, x25
    b row_loop
rows_done:
    mov x0, x20
    mov x1, x22
    bl put_hex8
    mov w3, #10
    strb w3, [x1]
    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 8
inbuf:
    .zero 65536
linebuf:
    .zero 128
