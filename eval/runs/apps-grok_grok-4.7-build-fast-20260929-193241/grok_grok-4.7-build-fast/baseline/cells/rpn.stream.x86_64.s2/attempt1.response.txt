.globl _start

.section .rodata
err_str:
    .ascii "error\n"
min_str:
    .ascii "9223372036854775808"

.bss
.align 8
in_buf:
    .zero 65536
num_stk:
    .zero 8192
out_buf:
    .zero 32

.text
_start:
    lea rbx, [rip + in_buf]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    xor r13, r13
line_loop:
    cmp r13, r12
    jae exit_ok
    mov r14, r13
scan_line:
    cmp r13, r12
    jae last_line
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x0A
    je got_nl
    inc r13
    jmp scan_line
got_nl:
    mov rsi, r14
    mov rdi, r13
    call eval_line
    inc r13
    jmp line_loop
last_line:
    mov rsi, r14
    mov rdi, r12
    call eval_line
exit_ok:
    mov rax, 60
    xor edi, edi
    syscall

eval_line:
    lea r9, [rip + num_stk]
    xor r8, r8
eval_skip:
    cmp rsi, rdi
    jae eval_finish
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x20
    je eval_sep
    cmp al, 0x09
    je eval_sep
    jmp eval_tok
eval_sep:
    inc rsi
    jmp eval_skip
eval_tok:
    mov r10, rsi
eval_scan:
    cmp rsi, rdi
    jae eval_got_tok
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x20
    je eval_got_tok
    cmp al, 0x09
    je eval_got_tok
    inc rsi
    jmp eval_scan
eval_got_tok:
    mov rax, rsi
    sub rax, r10
    cmp rax, 1
    jne eval_number
    movzx eax, byte ptr [rbx + r10]
    cmp al, 0x2B
    je eval_add
    cmp al, 0x2D
    je eval_sub
    cmp al, 0x2A
    je eval_mul
    cmp al, 0x2F
    je eval_div
    cmp al, 0x25
    je eval_mod
eval_number:
    mov r14, r10
    xor r15d, r15d
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x2D
    jne eval_digits
    mov r15d, 1
    inc r14
eval_digits:
    mov rcx, rsi
    sub rcx, r14
    cmp rcx, 1
    jb eval_error
    cmp rcx, 18
    ja eval_error
    xor eax, eax
eval_dloop:
    cmp r14, rsi
    jae eval_num_done
    movzx ecx, byte ptr [rbx + r14]
    cmp cl, 0x30
    jb eval_error
    cmp cl, 0x39
    ja eval_error
    sub cl, 0x30
    imul rax, rax, 10
    movzx ecx, cl
    add rax, rcx
    inc r14
    jmp eval_dloop
eval_num_done:
    test r15d, r15d
    jz eval_push
    neg rax
eval_push:
    cmp r8, 1024
    jae eval_error
    mov qword ptr [r9 + r8*8], rax
    inc r8
    jmp eval_skip
eval_add:
    cmp r8, 2
    jb eval_error
    sub r8, 2
    mov rax, qword ptr [r9 + r8*8]
    mov rdx, qword ptr [r9 + r8*8 + 8]
    add rax, rdx
    jmp eval_store
eval_sub:
    cmp r8, 2
    jb eval_error
    sub r8, 2
    mov rax, qword ptr [r9 + r8*8]
    mov rdx, qword ptr [r9 + r8*8 + 8]
    sub rax, rdx
    jmp eval_store
eval_mul:
    cmp r8, 2
    jb eval_error
    sub r8, 2
    mov rax, qword ptr [r9 + r8*8]
    mov rdx, qword ptr [r9 + r8*8 + 8]
    imul rax, rdx
    jmp eval_store
eval_div:
    xor r15d, r15d
    jmp eval_divmod
eval_mod:
    mov r15d, 1
eval_divmod:
    cmp r8, 2
    jb eval_error
    sub r8, 2
    mov rax, qword ptr [r9 + r8*8]
    mov rcx, qword ptr [r9 + r8*8 + 8]
    test rcx, rcx
    jz eval_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne eval_do_idiv
    cmp rcx, -1
    je eval_error
eval_do_idiv:
    cqo
    idiv rcx
    test r15d, r15d
    jz eval_store
    mov rax, rdx
eval_store:
    mov qword ptr [r9 + r8*8], rax
    inc r8
    jmp eval_skip
eval_finish:
    cmp r8, 1
    jne eval_error
    mov rax, qword ptr [r9]
    call print_i64
    ret
eval_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_str]
    mov rdx, 6
    syscall
    ret

print_i64:
    lea r9, [rip + out_buf]
    mov r10, r9
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne print_regular
    mov byte ptr [r10], 0x2D
    inc r10
    lea rsi, [rip + min_str]
    mov ecx, 19
print_min_loop:
    movzx edx, byte ptr [rsi]
    mov byte ptr [r10], dl
    inc rsi
    inc r10
    dec ecx
    jnz print_min_loop
    jmp print_nl
print_regular:
    test rax, rax
    jns print_nonneg
    mov byte ptr [r10], 0x2D
    inc r10
    neg rax
print_nonneg:
    test rax, rax
    jnz print_conv
    mov byte ptr [r10], 0x30
    inc r10
    jmp print_nl
print_conv:
    xor ecx, ecx
print_conv_loop:
    xor edx, edx
    mov r11, 10
    div r11
    add dl, 0x30
    push rdx
    inc ecx
    test rax, rax
    jnz print_conv_loop
print_pop_loop:
    pop rdx
    mov byte ptr [r10], dl
    inc r10
    dec ecx
    jnz print_pop_loop
print_nl:
    mov byte ptr [r10], 0x0A
    inc r10
    mov rsi, r9
    mov rdx, r10
    sub rdx, r9
    mov rax, 1
    mov rdi, 1
    syscall
    ret
