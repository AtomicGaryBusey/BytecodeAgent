.globl _start
.text
_start:
    cld
    lea rbx, [rip + inbuf]
    xor r15, r15
.read_more:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    add rsi, r15
    mov edx, 32768
    sub rdx, r15
    syscall
    cmp rax, 0
    jle .read_done
    add r15, rax
    cmp r15, 32768
    jb .read_more
.read_done:
    xor r14, r14
    xor r13, r13
.parse:
    cmp r13, r15
    jae .parsed
    mov r12, r13
.findnl:
    cmp r13, r15
    jae .eol_no_nl
    movzx eax, byte ptr [rbx + r13]
    inc r13
    cmp al, 10
    jne .findnl
    mov rax, r13
    sub rax, r12
    dec rax
    jmp .store
.eol_no_nl:
    mov rax, r15
    sub rax, r12
.store:
    lea rcx, [rip + line_off]
    mov qword ptr [rcx + r14*8], r12
    lea rcx, [rip + line_len]
    mov qword ptr [rcx + r14*8], rax
    lea rcx, [rip + line_idx]
    mov qword ptr [rcx + r14*8], r14
    inc r14
    jmp .parse
.parsed:
    cmp r14, 2
    jb .output
    mov r12, r14
    shr r12, 1
.build:
    test r12, r12
    jz .extract_init
    dec r12
    mov rdi, r12
    mov rsi, r14
    call sift_down
    jmp .build
.extract_init:
    mov r12, r14
.extract:
    dec r12
    jz .output
    lea r8, [rip + line_idx]
    mov rax, qword ptr [r8]
    mov rcx, qword ptr [r8 + r12*8]
    mov qword ptr [r8], rcx
    mov qword ptr [r8 + r12*8], rax
    xor edi, edi
    mov rsi, r12
    call sift_down
    jmp .extract
.output:
    xor r12, r12
.out_loop:
    cmp r12, r14
    jae .do_exit
    lea r8, [rip + line_idx]
    mov r9, qword ptr [r8 + r12*8]
    lea r8, [rip + line_len]
    mov rdx, qword ptr [r8 + r9*8]
    test rdx, rdx
    jz .write_nl
    lea r8, [rip + line_off]
    mov rsi, qword ptr [r8 + r9*8]
    add rsi, rbx
    mov eax, 1
    mov edi, 1
    syscall
.write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    inc r12
    jmp .out_loop
.do_exit:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    lea r8, [rip + line_off]
    lea r9, [rip + line_len]
    mov r10, qword ptr [r9 + rdi*8]
    mov r11, qword ptr [r9 + rsi*8]
    mov rax, qword ptr [r8 + rdi*8]
    mov rcx, qword ptr [r8 + rsi*8]
    lea r8, [rip + inbuf]
    lea rsi, [r8 + rax]
    lea rdi, [r8 + rcx]
    mov rcx, r10
    cmp rcx, r11
    cmova rcx, r11
    test rcx, rcx
    jz .cmp_len
    repe cmpsb
    je .cmp_len
    mov eax, 1
    ja .cmp_ret
    mov eax, -1
    ret
.cmp_len:
    cmp r10, r11
    mov eax, 0
    je .cmp_ret
    mov eax, 1
    ja .cmp_ret
    mov eax, -1
.cmp_ret:
    ret

sift_down:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
.sloop:
    mov r14, r12
    shl r14, 1
    inc r14
    cmp r14, r13
    jae .sdone
    lea rbx, [rip + line_idx]
    mov rdi, qword ptr [rbx + r14*8]
    mov rsi, qword ptr [rbx + r12*8]
    call compare_lines
    mov r15, r12
    test eax, eax
    jle .check_right
    mov r15, r14
.check_right:
    mov rax, r14
    inc rax
    cmp rax, r13
    jae .maybe_swap
    mov rdi, qword ptr [rbx + rax*8]
    mov rsi, qword ptr [rbx + r15*8]
    push rax
    call compare_lines
    pop rcx
    test eax, eax
    jle .maybe_swap
    mov r15, rcx
.maybe_swap:
    cmp r15, r12
    je .sdone
    mov rax, qword ptr [rbx + r12*8]
    mov rcx, qword ptr [rbx + r15*8]
    mov qword ptr [rbx + r12*8], rcx
    mov qword ptr [rbx + r15*8], rax
    mov r12, r15
    jmp .sloop
.sdone:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

.section .rodata
nl:
    .byte 0x0A

.bss
.align 8
inbuf:
    .zero 32768
line_off:
    .zero 8000
line_len:
    .zero 8000
line_idx:
    .zero 8000
