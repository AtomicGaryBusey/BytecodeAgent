.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_CAP, 32768

.section .rodata
nlbuf:
    .byte 0x0A

.bss
.align 3
inbuf:
    .zero 32768
.align 3
line_ptr:
    .zero 8000
.align 3
line_len:
    .zero 4000
.align 3
tmp_ptr:
    .zero 8000
.align 3
tmp_len:
    .zero 4000

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x2, #BUF_CAP
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, do_exit
    adrp x22, line_ptr
    add x22, x22, :lo12:line_ptr
    adrp x23, line_len
    add x23, x23, :lo12:line_len
    mov x21, #0
    mov x24, #0

parse_loop:
    cmp x24, x20
    b.hs parse_done
    mov x25, x24

parse_scan:
    cmp x25, x20
    b.hs parse_last
    ldrb w3, [x19, x25]
    cmp w3, #0x0A
    b.eq parse_nl
    add x25, x25, #1
    b parse_scan

parse_nl:
    sub x4, x25, x24
    add x5, x19, x24
    str x5, [x22, x21, lsl #3]
    str w4, [x23, x21, lsl #2]
    add x21, x21, #1
    add x24, x25, #1
    b parse_loop

parse_last:
    sub x4, x20, x24
    add x5, x19, x24
    str x5, [x22, x21, lsl #3]
    str w4, [x23, x21, lsl #2]
    add x21, x21, #1

parse_done:
    adrp x26, tmp_ptr
    add x26, x26, :lo12:tmp_ptr
    adrp x27, tmp_len
    add x27, x27, :lo12:tmp_len
    mov x28, #1

sort_pass:
    cmp x28, x21
    b.hs print_start
    mov x9, #0

sort_runs:
    cmp x9, x21
    b.hs sort_swap
    add x12, x9, x28
    cmp x12, x21
    csel x12, x12, x21, lo
    add x13, x9, x28
    add x13, x13, x28
    cmp x13, x21
    csel x13, x13, x21, lo
    mov x10, x9
    mov x11, x12
    mov x14, x9

sort_merge:
    cmp x10, x12
    b.hs sort_tail_q
    cmp x11, x13
    b.hs sort_tail_p
    ldr x0, [x22, x10, lsl #3]
    ldr w1, [x23, x10, lsl #2]
    ldr x2, [x22, x11, lsl #3]
    ldr w3, [x23, x11, lsl #2]
    bl compare
    cmp w0, #0
    b.gt sort_take_q
    ldr x0, [x22, x10, lsl #3]
    ldr w1, [x23, x10, lsl #2]
    add x10, x10, #1
    b sort_store

sort_take_q:
    ldr x0, [x22, x11, lsl #3]
    ldr w1, [x23, x11, lsl #2]
    add x11, x11, #1

sort_store:
    str x0, [x26, x14, lsl #3]
    str w1, [x27, x14, lsl #2]
    add x14, x14, #1
    b sort_merge

sort_tail_p:
    cmp x10, x12
    b.hs sort_run_done
    ldr x0, [x22, x10, lsl #3]
    ldr w1, [x23, x10, lsl #2]
    str x0, [x26, x14, lsl #3]
    str w1, [x27, x14, lsl #2]
    add x10, x10, #1
    add x14, x14, #1
    b sort_tail_p

sort_tail_q:
    cmp x11, x13
    b.hs sort_run_done
    ldr x0, [x22, x11, lsl #3]
    ldr w1, [x23, x11, lsl #2]
    str x0, [x26, x14, lsl #3]
    str w1, [x27, x14, lsl #2]
    add x11, x11, #1
    add x14, x14, #1
    b sort_tail_q

sort_run_done:
    add x9, x9, x28
    add x9, x9, x28
    b sort_runs

sort_swap:
    mov x0, x22
    mov x22, x26
    mov x26, x0
    mov x0, x23
    mov x23, x27
    mov x27, x0
    add x28, x28, x28
    b sort_pass

print_start:
    adrp x29, nlbuf
    add x29, x29, :lo12:nlbuf
    mov x9, #0

print_loop:
    cmp x9, x21
    b.hs do_exit
    ldr x1, [x22, x9, lsl #3]
    ldr w2, [x23, x9, lsl #2]
    cbz w2, print_nl
    mov x0, #1
    mov x8, #SYS_write
    svc #0

print_nl:
    mov x0, #1
    mov x1, x29
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x9, x9, #1
    b print_loop

do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b do_exit

// x0=ptr_a x1=len_a x2=ptr_b x3=len_b -> x0 <0 if a<b, 0 if eq, >0 if a>b
compare:
    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, #0

compare_loop:
    cmp x5, x4
    b.hs compare_len
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.ne compare_done
    add x5, x5, #1
    b compare_loop

compare_len:
    cmp x1, x3

compare_done:
    cset w0, hi
    cset w1, lo
    sub w0, w0, w1
    ret
