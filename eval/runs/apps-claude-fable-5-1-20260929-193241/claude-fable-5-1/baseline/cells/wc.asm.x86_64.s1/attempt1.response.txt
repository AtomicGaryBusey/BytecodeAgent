.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea rdx, [rsi + rax]
scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    lea ecx, [rax - 9]
    cmp ecx, 4
    jbe ws
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    cmp rsi, rdx
    jb scan
    jmp read_loop
done:
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    mov rax, r14
    call putnum
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call putnum
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call putnum
    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp done

putnum:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 64
