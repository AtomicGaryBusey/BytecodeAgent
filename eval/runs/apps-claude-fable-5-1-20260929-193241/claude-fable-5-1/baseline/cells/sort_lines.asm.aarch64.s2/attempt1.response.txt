.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x2, #0x10000
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    // split into lines: (ptr,len) records in linesA
    adrp    x21, linesA
    add     x21, x21, :lo12:linesA
    adrp    x25, linesB
    add     x25, x25, :lo12:linesB
    mov     x22, #0                 // line count
    mov     x23, #0                 // line start offset
    mov     x24, #0                 // index
split_loop:
    cmp     x24, x20
    beq     split_end
    ldrb    w0, [x19, x24]
    add     x24, x24, #1
    cmp     w0, #10
    bne     split_loop
    add     x0, x19, x23
    sub     x1, x24, #1
    sub     x1, x1, x23
    add     x2, x21, x22, lsl #4
    stp     x0, x1, [x2]
    add     x22, x22, #1
    mov     x23, x24
    b       split_loop
split_end:
    cmp     x23, x20
    beq     no_final
    add     x0, x19, x23
    sub     x1, x20, x23
    add     x2, x21, x22, lsl #4
    stp     x0, x1, [x2]
    add     x22, x22, #1
no_final:

    // bottom-up merge sort, src = x21, dst = x25
    cmp     x22, #1
    ble     sort_done
    mov     x26, #1                 // width
sort_outer:
    cmp     x26, x22
    bge     sort_done
    mov     x27, #0                 // lo
merge_pass:
    cmp     x27, x22
    bge     pass_done
    add     x9, x27, x26            // mid
    cmp     x9, x22
    csel    x9, x9, x22, lt
    add     x10, x9, x26            // hi
    cmp     x10, x22
    csel    x10, x10, x22, lt
    mov     x11, x27                // i
    mov     x12, x9                 // j
    mov     x13, x27                // k
merge_loop:
    cmp     x11, x9
    bge     take_right_rest
    cmp     x12, x10
    bge     take_left
    add     x0, x21, x11, lsl #4
    ldp     x0, x1, [x0]
    add     x2, x21, x12, lsl #4
    ldp     x2, x3, [x2]
    bl      compare
    cmp     x0, #0
    bgt     take_right
take_left:
    add     x0, x21, x11, lsl #4
    ldp     x0, x1, [x0]
    add     x2, x25, x13, lsl #4
    stp     x0, x1, [x2]
    add     x11, x11, #1
    add     x13, x13, #1
    b       merge_loop
take_right:
    add     x0, x21, x12, lsl #4
    ldp     x0, x1, [x0]
    add     x2, x25, x13, lsl #4
    stp     x0, x1, [x2]
    add     x12, x12, #1
    add     x13, x13, #1
    b       merge_loop
take_right_rest:
    cmp     x12, x10
    bge     merge_end
    b       take_right
merge_end:
    add     x27, x27, x26, lsl #1
    b       merge_pass
pass_done:
    mov     x0, x21
    mov     x21, x25
    mov     x25, x0
    lsl     x26, x26, #1
    b       sort_outer
sort_done:

    // output
    adrp    x19, outbuf
    add     x19, x19, :lo12:outbuf
    mov     x20, #0                 // outpos
    mov     x24, #0
out_loop:
    cmp     x24, x22
    bge     out_flush
    add     x0, x21, x24, lsl #4
    ldp     x0, x1, [x0]
    add     x2, x20, x1
    add     x2, x2, #1
    mov     x3, #0x20000
    cmp     x2, x3
    bls     out_copy
    bl      flush
out_copy:
    add     x0, x21, x24, lsl #4
    ldp     x0, x1, [x0]
    add     x2, x19, x20
    mov     x3, #0
copy_loop:
    cmp     x3, x1
    beq     copy_done
    ldrb    w4, [x0, x3]
    strb    w4, [x2, x3]
    add     x3, x3, #1
    b       copy_loop
copy_done:
    add     x20, x20, x1
    mov     w4, #10
    strb    w4, [x19, x20]
    add     x20, x20, #1
    add     x24, x24, #1
    b       out_loop
out_flush:
    bl      flush
    mov     x8, #94
    mov     x0, #0
    svc     #0
hang:
    b       hang

// flush: write outbuf[0..x20) to stdout, reset x20
flush:
    mov     x9, #0
flush_loop:
    cmp     x9, x20
    bge     flush_done
    mov     x8, #64
    mov     x0, #1
    add     x1, x19, x9
    sub     x2, x20, x9
    svc     #0
    cmp     x0, #0
    ble     flush_done
    add     x9, x9, x0
    b       flush_loop
flush_done:
    mov     x20, #0
    ret

// compare(x0=p1, x1=l1, x2=p2, x3=l2) -> x0 = -1 / 0 / 1
compare:
    cmp     x1, x3
    csel    x4, x1, x3, lo
    mov     x5, #0
cmp_loop:
    cmp     x5, x4
    beq     cmp_len
    ldrb    w6, [x0, x5]
    ldrb    w7, [x2, x5]
    add     x5, x5, #1
    cmp     w6, w7
    beq     cmp_loop
    blo     cmp_less
    mov     x0, #1
    ret
cmp_less:
    mov     x0, #-1
    ret
cmp_len:
    cmp     x1, x3
    blo     cmp_less
    beq     cmp_eq
    mov     x0, #1
    ret
cmp_eq:
    mov     x0, #0
    ret
