.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    mov x28, #0x20000
    add x28, x28, x19
read_loop:
    sub x2, x28, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, x21
    adrp x23, estack
    add x23, x23, :lo12:estack

main_loop:
    cmp x19, x20
    b.hs done
    mov x25, x19
find_nl:
    cmp x25, x20
    b.hs found
    ldrb w0, [x25]
    cmp w0, #10
    b.eq found
    add x25, x25, #1
    b find_nl
found:
    mov x24, #0
    mov x26, x19
tok_loop:
    cmp x26, x25
    b.hs end_of_line
    ldrb w0, [x26]
    cmp w0, #32
    b.eq skip_sep
    cmp w0, #9
    b.ne tok_start
skip_sep:
    add x26, x26, #1
    b tok_loop
tok_start:
    mov x27, x26
tok_end_loop:
    add x27, x27, #1
    cmp x27, x25
    b.hs tok_end
    ldrb w1, [x27]
    cmp w1, #32
    b.eq tok_end
    cmp w1, #9
    b.ne tok_end_loop
tok_end:
    sub x1, x27, x26
    cmp x1, #1
    b.ne literal
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    cmp w0, #'/'
    b.eq op_div
    cmp w0, #'%'
    b.eq op_rem
literal:
    mov x2, x26
    mov x3, #0
    cmp w0, #'-'
    b.ne 1f
    mov x3, #1
    add x2, x2, #1
1:
    sub x4, x27, x2
    sub x4, x4, #1
    cmp x4, #17
    b.hi line_error
    mov x5, #0
    mov x6, #10
2:
    ldrb w7, [x2], #1
    sub w7, w7, #'0'
    cmp w7, #9
    b.hi line_error
    madd x5, x5, x6, x7
    cmp x2, x27
    b.lo 2b
    cbz x3, 3f
    neg x5, x5
3:
    str x5, [x23, x24, lsl #3]
    add x24, x24, #1
    mov x26, x27
    b tok_loop

op_add:
    bl pop2
    add x0, x1, x2
    b push_result
op_sub:
    bl pop2
    sub x0, x1, x2
    b push_result
op_mul:
    bl pop2
    mul x0, x1, x2
    b push_result
op_div:
    bl pop2
    bl div_check
    sdiv x0, x1, x2
    b push_result
op_rem:
    bl pop2
    bl div_check
    sdiv x0, x1, x2
    msub x0, x0, x2, x1
    b push_result

push_result:
    sub x3, x24, #1
    str x0, [x23, x3, lsl #3]
    mov x26, x27
    b tok_loop

pop2:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x2, [x23, x24, lsl #3]
    sub x3, x24, #1
    ldr x1, [x23, x3, lsl #3]
    ret

div_check:
    cbz x2, line_error
    cmn x2, #1
    b.ne 1f
    mov x4, #0x8000000000000000
    cmp x1, x4
    b.eq line_error
1:
    ret

end_of_line:
    cmp x24, #1
    b.ne line_error
    ldr x0, [x23]
    bl print_num
    b next_line
line_error:
    adrp x3, errmsg
    add x3, x3, :lo12:errmsg
    mov x4, #6
1:
    ldrb w5, [x3], #1
    strb w5, [x22], #1
    subs x4, x4, #1
    b.ne 1b
next_line:
    add x19, x25, #1
    sub x2, x22, x21
    mov x3, #0xFF00
    cmp x2, x3
    b.lo main_loop
    bl flush
    b main_loop

done:
    bl flush
    mov x0, #0
    mov x8, #94
    svc #0
hang:
    b hang

print_num:
    sub sp, sp, #32
    add x3, sp, #32
    mov x4, x0
    tbz x0, #63, 1f
    neg x4, x0
    mov w5, #'-'
    strb w5, [x22], #1
1:
    mov x6, #10
2:
    udiv x7, x4, x6
    msub x10, x7, x6, x4
    add x10, x10, #'0'
    strb w10, [x3, #-1]!
    mov x4, x7
    cbnz x4, 2b
    add x9, sp, #32
3:
    ldrb w5, [x3], #1
    strb w5, [x22], #1
    cmp x3, x9
    b.lo 3b
    mov w5, #10
    strb w5, [x22], #1
    add sp, sp, #32
    ret

flush:
    sub x2, x22, x21
    cbz x2, 1f
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0
    mov x22, x21
1:
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 16
inbuf:
    .space 0x20000
.balign 16
outbuf:
    .space 0x10000
.balign 16
estack:
    .space 0x8000
