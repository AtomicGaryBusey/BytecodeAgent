.bss
.balign 16
inbuf:  .space 65537
tape:   .space 30000
.balign 16
jt:     .space 65536*4
bstack: .space 65536*4
outbuf: .space 65536

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65537
    sub edx, ebx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    lea r9, [rsi + rcx + 1]
    lea r10, [rsi + rbx]
    mov rbx, rcx
    jmp matched
no_bang:
    lea r9, [rsi + rbx]
    mov r10, r9
matched:
    xor ecx, ecx
    xor edx, edx
    lea r8, [rip + jt]
    lea rdi, [rip + bstack]
bm_loop:
    cmp rcx, rbx
    jae bm_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5B
    je bm_open
    cmp al, 0x5D
    jne bm_next
    test edx, edx
    jz fail
    dec edx
    mov eax, dword ptr [rdi + rdx*4]
    mov dword ptr [r8 + rax*4], ecx
    mov dword ptr [r8 + rcx*4], eax
    jmp bm_next
bm_open:
    mov dword ptr [rdi + rdx*4], ecx
    inc edx
bm_next:
    inc rcx
    jmp bm_loop
bm_done:
    test edx, edx
    jnz fail

    lea rdx, [rip + tape]
    lea r12, [rip + outbuf]
    xor r13d, r13d
    xor ecx, ecx
run:
    cmp rcx, rbx
    jae finish
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x3E
    je op_r
    cmp al, 0x3C
    je op_l
    cmp al, 0x2B
    je op_p
    cmp al, 0x2D
    je op_m
    cmp al, 0x2E
    je op_dot
    cmp al, 0x2C
    je op_comma
    cmp al, 0x5B
    je op_ob
    cmp al, 0x5D
    je op_cb
next:
    inc rcx
    jmp run
op_r:
    inc rdx
    jmp next
op_l:
    dec rdx
    jmp next
op_p:
    inc byte ptr [rdx]
    jmp next
op_m:
    dec byte ptr [rdx]
    jmp next
op_dot:
    mov al, byte ptr [rdx]
    mov byte ptr [r12 + r13], al
    inc r13
    cmp r13, 65536
    jb next
    call flush
    jmp next
op_comma:
    cmp r9, r10
    jae store0
    mov al, byte ptr [r9]
    inc r9
    mov byte ptr [rdx], al
    jmp next
store0:
    mov byte ptr [rdx], 0
    jmp next
op_ob:
    cmp byte ptr [rdx], 0
    jne next
    mov ecx, dword ptr [r8 + rcx*4]
    jmp next
op_cb:
    cmp byte ptr [rdx], 0
    je next
    mov ecx, dword ptr [r8 + rcx*4]
    jmp next

flush:
    test r13, r13
    jz 1f
    push rcx
    push rsi
    push rdx
    push rdi
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    pop rdi
    pop rdx
    pop rsi
    pop rcx
    xor r13d, r13d
1:
    ret

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp finish
fail:
    mov eax, 231
    mov edi, 1
    syscall
    jmp fail
