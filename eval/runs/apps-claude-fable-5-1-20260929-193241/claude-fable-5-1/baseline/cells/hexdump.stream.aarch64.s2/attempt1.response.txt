.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x26, hextab
    add x26, x26, :lo12:hextab
    mov x20, #0                 // total input length
read_loop:
    mov x2, #0x20000
    subs x2, x2, x20
    ble done_read
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    svc #0
    cmp x0, #0
    ble done_read
    add x20, x20, x0
    b read_loop
done_read:
    sub sp, sp, #96
    mov x22, sp                 // line buffer
    mov x21, #0                 // current offset
row_loop:
    cmp x21, x20
    bge final
    sub x23, x20, x21           // remaining
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, lo      // n = min(remaining, 16)
    mov x25, x22
    mov x0, x21
    mov x1, x25
    bl hex8
    add x25, x25, #8
    mov w0, #0x20
    strb w0, [x25], #1
    strb w0, [x25], #1
    add x27, x19, x21
    mov x28, #0
byte_loop:
    cmp x28, x23
    bge pad3
    ldrb w0, [x27, x28]
    lsr x1, x0, #4
    ldrb w1, [x26, x1]
    strb w1, [x25], #1
    and x1, x0, #15
    ldrb w1, [x26, x1]
    strb w1, [x25], #1
    mov w0, #0x20
    strb w0, [x25], #1
    b after_slot
pad3:
    mov w0, #0x20
    strb w0, [x25], #1
    strb w0, [x25], #1
    strb w0, [x25], #1
after_slot:
    cmp x28, #7
    bne 1f
    mov w0, #0x20
    strb w0, [x25], #1
1:
    add x28, x28, #1
    cmp x28, #16
    blt byte_loop
    mov w0, #0x20
    strb w0, [x25], #1
    mov w0, #0x7c
    strb w0, [x25], #1
    mov x28, #0
char_loop:
    cmp x28, x23
    bge char_done
    ldrb w0, [x27, x28]
    sub w1, w0, #0x20
    cmp w1, #0x5e
    bls 2f
    mov w0, #0x2e
2:
    strb w0, [x25], #1
    add x28, x28, #1
    b char_loop
char_done:
    mov w0, #0x7c
    strb w0, [x25], #1
    mov w0, #0x0a
    strb w0, [x25], #1
    mov x8, #64
    mov x0, #1
    mov x1, x22
    sub x2, x25, x22
    svc #0
    add x21, x21, x23
    b row_loop
final:
    mov x0, x20
    mov x1, x22
    bl hex8
    mov w0, #0x0a
    strb w0, [x22, #8]
    mov x8, #64
    mov x0, #1
    mov x1, x22
    mov x2, #9
    svc #0
    mov x8, #94
    mov x0, #0
    svc #0

// hex8: write x0 as 8 lowercase hex digits at [x1]; uses x26 = hextab
hex8:
    mov x2, #7
1:
    and x3, x0, #15
    ldrb w3, [x26, x3]
    strb w3, [x1, x2]
    lsr x0, x0, #4
    subs x2, x2, #1
    bge 1b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 131072
