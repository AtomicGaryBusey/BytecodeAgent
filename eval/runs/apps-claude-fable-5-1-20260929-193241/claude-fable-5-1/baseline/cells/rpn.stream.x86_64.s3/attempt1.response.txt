.text
.globl _start
_start:
    lea r15, [rip + inbuf]
read_loop:
    lea rdx, [rip + inbuf_end]
    sub rdx, r15
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r15
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r13, [rip + vstack]
    lea r14, [rip + outbuf]

line_loop:
    cmp rsi, r15
    jae finish
    xor r12d, r12d
tok_loop:
    cmp rsi, r15
    jae end_line
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    cmp al, 0x0A
    je end_line
    mov rbx, rsi
find_end:
    inc rsi
    cmp rsi, r15
    jae got_end
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je got_end
    cmp al, 0x09
    je got_end
    cmp al, 0x0A
    jne find_end
got_end:
    mov rdx, rsi
    sub rdx, rbx
    movzx eax, byte ptr [rbx]
    cmp rdx, 1
    jne parse_num
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
parse_num:
    xor r8d, r8d
    mov rdi, rbx
    cmp al, '-'
    jne pn_nosign
    inc r8d
    inc rdi
    dec rdx
pn_nosign:
    test rdx, rdx
    jz line_error
    cmp rdx, 18
    ja line_error
    xor eax, eax
pn_digit:
    movzx ecx, byte ptr [rdi]
    sub ecx, '0'
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc rdi
    dec rdx
    jnz pn_digit
    test r8d, r8d
    jz pn_push
    neg rax
pn_push:
    mov qword ptr [r13 + r12*8], rax
    inc r12
    jmp tok_loop
skip_sep:
    inc rsi
    jmp tok_loop

op_add:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r13 + r12*8 - 16]
    add rax, qword ptr [r13 + r12*8 - 8]
    dec r12
    mov qword ptr [r13 + r12*8 - 8], rax
    jmp tok_loop
op_sub:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r13 + r12*8 - 16]
    sub rax, qword ptr [r13 + r12*8 - 8]
    dec r12
    mov qword ptr [r13 + r12*8 - 8], rax
    jmp tok_loop
op_mul:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r13 + r12*8 - 16]
    imul rax, qword ptr [r13 + r12*8 - 8]
    dec r12
    mov qword ptr [r13 + r12*8 - 8], rax
    jmp tok_loop
op_div:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r13 + r12*8 - 16]
    mov rcx, qword ptr [r13 + r12*8 - 8]
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne div_ok
    mov rdx, 1
    ror rdx, 1
    cmp rax, rdx
    je line_error
div_ok:
    cqo
    idiv rcx
    dec r12
    mov qword ptr [r13 + r12*8 - 8], rax
    jmp tok_loop
op_mod:
    cmp r12, 2
    jb line_error
    mov rax, qword ptr [r13 + r12*8 - 16]
    mov rcx, qword ptr [r13 + r12*8 - 8]
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne mod_ok
    mov rdx, 1
    ror rdx, 1
    cmp rax, rdx
    je line_error
mod_ok:
    cqo
    idiv rcx
    dec r12
    mov qword ptr [r13 + r12*8 - 8], rdx
    jmp tok_loop

line_error:
    cmp rsi, r15
    jae emit_error
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne line_error
emit_error:
    call ensure_space
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp line_loop

end_line:
    cmp rsi, r15
    jae el_noeat
    inc rsi
el_noeat:
    cmp r12, 1
    jne emit_error
    mov rax, qword ptr [r13]
    call ensure_space
    test rax, rax
    jns el_pos
    mov byte ptr [r14], '-'
    inc r14
    neg rax
el_pos:
    sub rsp, 32
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 0x0A
    mov ecx, 10
el_conv:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz el_conv
    lea rcx, [rsp + 32]
    sub rcx, rdi
el_copy:
    mov al, byte ptr [rdi]
    mov byte ptr [r14], al
    inc rdi
    inc r14
    dec rcx
    jnz el_copy
    add rsp, 32
    jmp line_loop

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
finish_hang:
    jmp finish_hang

ensure_space:
    lea rcx, [rip + outbuf + 65536 - 64]
    cmp r14, rcx
    jb es_ok
    call flush
es_ok:
    ret

flush:
    push rsi
    push rdi
    push rdx
    push rax
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    jz fl_done
    mov eax, 1
    mov edi, 1
    syscall
    lea r14, [rip + outbuf]
fl_done:
    pop rax
    pop rdx
    pop rdi
    pop rsi
    ret

.bss
.balign 16
inbuf:
    .space 65536 + 16
inbuf_end:
outbuf:
    .space 65536
vstack:
    .space 524288
