        .bss
        .balign 16
buf:    .space 65536
outbuf: .space 64

        .text
        .globl _start
_start:
        mov     x19, #0                 // lines
        mov     x20, #0                 // words
        mov     x21, #0                 // bytes
        mov     x22, #0                 // in_word flag
        adrp    x23, buf
        add     x23, x23, :lo12:buf

read_loop:
        mov     x0, #0
        mov     x1, x23
        mov     x2, #65536
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    done
        add     x21, x21, x0
        mov     x1, x23
        add     x2, x23, x0

byte_loop:
        ldrb    w3, [x1], #1
        cmp     w3, #0x0a
        b.ne    1f
        add     x19, x19, #1
        b       2f
1:
        cmp     w3, #0x20
        b.eq    2f
        sub     w4, w3, #9
        cmp     w4, #4
        b.ls    2f                      // 0x09..0x0d whitespace
        // word byte
        cbnz    x22, 3f
        mov     x22, #1
        add     x20, x20, #1
        b       3f
2:
        mov     x22, #0
3:
        cmp     x1, x2
        b.lo    byte_loop
        b       read_loop

done:
        adrp    x5, outbuf
        add     x5, x5, :lo12:outbuf
        add     x5, x5, #64
        mov     x24, x5                 // end of buffer
        mov     w6, #10
        strb    w6, [x5, #-1]!
        mov     x0, x21
        bl      print_num
        mov     w6, #0x20
        strb    w6, [x5, #-1]!
        mov     x0, x20
        bl      print_num
        mov     w6, #0x20
        strb    w6, [x5, #-1]!
        mov     x0, x19
        bl      print_num

        mov     x0, #1
        mov     x1, x5
        sub     x2, x24, x5
        mov     x8, #64
        svc     #0

        mov     x0, #0
        mov     x8, #93
        svc     #0

// print_num: x0 = value, x5 = pointer (writes backwards, pre-decrement)
print_num:
        mov     x6, #10
1:
        udiv    x7, x0, x6
        msub    x9, x7, x6, x0
        add     x9, x9, #0x30
        strb    w9, [x5, #-1]!
        mov     x0, x7
        cbnz    x0, 1b
        ret
