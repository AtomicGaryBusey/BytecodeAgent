.text
.globl _start
_start:
    mov     x19, #0                 // lines
    mov     x20, #0                 // words
    mov     x21, #0                 // bytes
    mov     x22, #0                 // in_word flag
    adrp    x23, buf
    add     x23, x23, :lo12:buf

read_loop:
    mov     x0, #0
    mov     x1, x23
    mov     x2, #65536
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.le    done
    add     x21, x21, x0
    mov     x1, x23
    add     x2, x23, x0             // end pointer

scan:
    ldrb    w3, [x1], #1
    cmp     w3, #0x0A
    cinc    x19, x19, eq
    cmp     w3, #0x20
    b.eq    ws
    sub     w4, w3, #9
    cmp     w4, #4
    b.ls    ws
    // word byte
    cbnz    x22, next
    add     x20, x20, #1
    mov     x22, #1
    b       next
ws:
    mov     x22, #0
next:
    cmp     x1, x2
    b.lo    scan
    b       read_loop

done:
    adrp    x8, outbuf
    add     x8, x8, :lo12:outbuf
    add     x8, x8, #64             // end of output buffer
    mov     w3, #10
    strb    w3, [x8, #-1]!
    mov     x4, x21
    bl      fmt
    mov     w3, #32
    strb    w3, [x8, #-1]!
    mov     x4, x20
    bl      fmt
    mov     w3, #32
    strb    w3, [x8, #-1]!
    mov     x4, x19
    bl      fmt

    adrp    x2, outbuf
    add     x2, x2, :lo12:outbuf
    add     x2, x2, #64
    sub     x2, x2, x8              // length
    mov     x1, x8
    mov     x0, #1
    mov     x8, #64                 // write
    svc     #0

    mov     x0, #0
    mov     x8, #93                 // exit
    svc     #0

// fmt: x4 = value, x8 = pointer (pre-decrement store), clobbers x5,x6,x7
fmt:
    mov     x6, #10
1:
    udiv    x5, x4, x6
    msub    x7, x5, x6, x4
    add     w7, w7, #48
    strb    w7, [x8, #-1]!
    mov     x4, x5
    cbnz    x4, 1b
    ret

.bss
.balign 16
buf:
    .space  65536
outbuf:
    .space  64
