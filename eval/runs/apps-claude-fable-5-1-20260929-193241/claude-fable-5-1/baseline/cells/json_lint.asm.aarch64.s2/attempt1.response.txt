.text
.globl _start

_start:
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x21, #0
read_loop:
        mov     x2, #65536
        sub     x2, x2, x21
        cbz     x2, read_done
        mov     x0, #0
        add     x1, x19, x21
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    read_done
        add     x21, x21, x0
        b       read_loop
read_done:
        mov     x20, #0
        bl      skip_ws
        bl      parse_value
        bl      skip_ws
        cmp     x20, x21
        b.ne    error
        mov     x0, #1
        adrp    x1, ok_msg
        add     x1, x1, :lo12:ok_msg
        mov     x2, #3
        mov     x8, #64
        svc     #0
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       .

// ---------------------------------------------------------------
// error: print "error at N\n" with N = x20, exit 1
error:
        adrp    x9, outbuf
        add     x9, x9, :lo12:outbuf
        mov     x1, x9
        adrp    x2, err_msg
        add     x2, x2, :lo12:err_msg
        mov     x3, #9
1:      ldrb    w4, [x2], #1
        strb    w4, [x1], #1
        subs    x3, x3, #1
        b.ne    1b
        mov     x0, x20
        mov     x10, #10
        sub     sp, sp, #32
        add     x5, sp, #32
        mov     x3, #0
2:      udiv    x6, x0, x10
        msub    x7, x6, x10, x0
        add     w7, w7, #0x30
        sub     x5, x5, #1
        strb    w7, [x5]
        add     x3, x3, #1
        mov     x0, x6
        cbnz    x0, 2b
3:      ldrb    w4, [x5], #1
        strb    w4, [x1], #1
        subs    x3, x3, #1
        b.ne    3b
        mov     w4, #10
        strb    w4, [x1], #1
        sub     x2, x1, x9
        mov     x1, x9
        mov     x0, #1
        mov     x8, #64
        svc     #0
        mov     x0, #1
        mov     x8, #93
        svc     #0
        b       .

// ---------------------------------------------------------------
// peek: w0 = current byte, or 0x100 at end of input
peek:
        cmp     x20, x21
        b.hs    1f
        ldrb    w0, [x19, x20]
        ret
1:      mov     w0, #0x100
        ret

// skip_ws: advance over space, tab, LF, CR
skip_ws:
        cmp     x20, x21
        b.hs    9f
        ldrb    w0, [x19, x20]
        cmp     w0, #0x20
        b.eq    1f
        cmp     w0, #0x09
        b.eq    1f
        cmp     w0, #0x0A
        b.eq    1f
        cmp     w0, #0x0D
        b.ne    9f
1:      add     x20, x20, #1
        b       skip_ws
9:      ret

// expect: current byte must equal w1, then advance
expect:
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        cmp     w0, w1
        b.ne    error
        add     x20, x20, #1
        ret

// match_lit: x1 -> NUL-terminated literal, match it byte by byte
match_lit:
1:      ldrb    w2, [x1], #1
        cbz     w2, 2f
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        cmp     w0, w2
        b.ne    error
        add     x20, x20, #1
        b       1b
2:      ret

// ---------------------------------------------------------------
parse_value:
        stp     x29, x30, [sp, #-16]!
        bl      peek
        cmp     w0, #0x7B
        b.eq    pv_obj
        cmp     w0, #0x5B
        b.eq    pv_arr
        cmp     w0, #0x22
        b.eq    pv_str
        cmp     w0, #0x2D
        b.eq    pv_num
        sub     w1, w0, #0x30
        cmp     w1, #9
        b.ls    pv_num
        cmp     w0, #0x74
        b.eq    pv_true
        cmp     w0, #0x66
        b.eq    pv_false
        cmp     w0, #0x6E
        b.eq    pv_null
        b       error
pv_obj:
        bl      parse_object
        b       pv_ret
pv_arr:
        bl      parse_array
        b       pv_ret
pv_str:
        bl      parse_string
        b       pv_ret
pv_num:
        bl      parse_number
        b       pv_ret
pv_true:
        adrp    x1, lit_true
        add     x1, x1, :lo12:lit_true
        bl      match_lit
        b       pv_ret
pv_false:
        adrp    x1, lit_false
        add     x1, x1, :lo12:lit_false
        bl      match_lit
        b       pv_ret
pv_null:
        adrp    x1, lit_null
        add     x1, x1, :lo12:lit_null
        bl      match_lit
pv_ret:
        ldp     x29, x30, [sp], #16
        ret

// ---------------------------------------------------------------
parse_object:
        stp     x29, x30, [sp, #-16]!
        add     x20, x20, #1
        bl      skip_ws
        bl      peek
        cmp     w0, #0x7D
        b.eq    po_end
po_member:
        bl      peek
        cmp     w0, #0x22
        b.ne    error
        bl      parse_string
        bl      skip_ws
        mov     w1, #0x3A
        bl      expect
        bl      skip_ws
        bl      parse_value
        bl      skip_ws
        bl      peek
        cmp     w0, #0x2C
        b.eq    po_comma
        cmp     w0, #0x7D
        b.ne    error
po_end:
        add     x20, x20, #1
        ldp     x29, x30, [sp], #16
        ret
po_comma:
        add     x20, x20, #1
        bl      skip_ws
        b       po_member

// ---------------------------------------------------------------
parse_array:
        stp     x29, x30, [sp, #-16]!
        add     x20, x20, #1
        bl      skip_ws
        bl      peek
        cmp     w0, #0x5D
        b.eq    pa_end
pa_elem:
        bl      parse_value
        bl      skip_ws
        bl      peek
        cmp     w0, #0x2C
        b.eq    pa_comma
        cmp     w0, #0x5D
        b.ne    error
pa_end:
        add     x20, x20, #1
        ldp     x29, x30, [sp], #16
        ret
pa_comma:
        add     x20, x20, #1
        bl      skip_ws
        b       pa_elem

// ---------------------------------------------------------------
parse_string:
        add     x20, x20, #1
ps_loop:
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        cmp     w0, #0x22
        b.eq    ps_end
        cmp     w0, #0x5C
        b.eq    ps_esc
        cmp     w0, #0x20
        b.lo    error
        add     x20, x20, #1
        b       ps_loop
ps_end:
        add     x20, x20, #1
        ret
ps_esc:
        add     x20, x20, #1
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        cmp     w0, #0x75
        b.eq    ps_u
        cmp     w0, #0x22
        b.eq    1f
        cmp     w0, #0x5C
        b.eq    1f
        cmp     w0, #0x2F
        b.eq    1f
        cmp     w0, #0x62
        b.eq    1f
        cmp     w0, #0x66
        b.eq    1f
        cmp     w0, #0x6E
        b.eq    1f
        cmp     w0, #0x72
        b.eq    1f
        cmp     w0, #0x74
        b.ne    error
1:      add     x20, x20, #1
        b       ps_loop
ps_u:
        add     x20, x20, #1
        mov     w3, #4
2:      cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        sub     w1, w0, #0x30
        cmp     w1, #9
        b.ls    3f
        sub     w1, w0, #0x41
        cmp     w1, #5
        b.ls    3f
        sub     w1, w0, #0x61
        cmp     w1, #5
        b.hi    error
3:      add     x20, x20, #1
        subs    w3, w3, #1
        b.ne    2b
        b       ps_loop

// ---------------------------------------------------------------
parse_number:
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        cmp     w0, #0x2D
        b.ne    1f
        add     x20, x20, #1
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
1:      cmp     w0, #0x30
        b.ne    2f
        add     x20, x20, #1
        b       pn_frac
2:      sub     w1, w0, #0x31
        cmp     w1, #8
        b.hi    error
        add     x20, x20, #1
pn_int:
        cmp     x20, x21
        b.hs    pn_ret
        ldrb    w0, [x19, x20]
        sub     w1, w0, #0x30
        cmp     w1, #9
        b.hi    pn_frac
        add     x20, x20, #1
        b       pn_int
pn_frac:
        cmp     x20, x21
        b.hs    pn_ret
        ldrb    w0, [x19, x20]
        cmp     w0, #0x2E
        b.ne    pn_exp
        add     x20, x20, #1
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        sub     w1, w0, #0x30
        cmp     w1, #9
        b.hi    error
        add     x20, x20, #1
pn_fracd:
        cmp     x20, x21
        b.hs    pn_ret
        ldrb    w0, [x19, x20]
        sub     w1, w0, #0x30
        cmp     w1, #9
        b.hi    pn_exp
        add     x20, x20, #1
        b       pn_fracd
pn_exp:
        cmp     w0, #0x65
        b.eq    4f
        cmp     w0, #0x45
        b.ne    pn_ret
4:      add     x20, x20, #1
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
        cmp     w0, #0x2B
        b.eq    5f
        cmp     w0, #0x2D
        b.ne    6f
5:      add     x20, x20, #1
        cmp     x20, x21
        b.hs    error
        ldrb    w0, [x19, x20]
6:      sub     w1, w0, #0x30
        cmp     w1, #9
        b.hi    error
        add     x20, x20, #1
pn_expd:
        cmp     x20, x21
        b.hs    pn_ret
        ldrb    w0, [x19, x20]
        sub     w1, w0, #0x30
        cmp     w1, #9
        b.hi    pn_ret
        add     x20, x20, #1
        b       pn_expd
pn_ret:
        ret

// ---------------------------------------------------------------
        .section .rodata
lit_true:
        .asciz  "true"
lit_false:
        .asciz  "false"
lit_null:
        .asciz  "null"
ok_msg:
        .ascii  "ok\n"
err_msg:
        .ascii  "error at "

        .bss
        .balign 16
inbuf:
        .space  65600
outbuf:
        .space  64
