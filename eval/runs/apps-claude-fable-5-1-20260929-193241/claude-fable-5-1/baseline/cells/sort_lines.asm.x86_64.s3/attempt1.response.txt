.text
.globl _start

.equ INBUF_SIZE, 65536
.equ LINES_MAX, 8192
.equ OUTBUF_SIZE, 73728

_start:
    xor r12d, r12d              # total bytes read
read_loop:
    mov eax, INBUF_SIZE
    sub eax, r12d
    jz read_done
    mov edx, eax
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # parse lines
    lea rsi, [rip + inbuf]
    xor ecx, ecx                # i
    xor ebx, ebx                # start
    xor r13d, r13d              # n
    lea rdi, [rip + lines]
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rsi + rcx], 10
    jne parse_next
    cmp r13, LINES_MAX
    jae parse_skip
    lea rax, [rsi + rbx]
    mov qword ptr [rdi], rax
    mov rax, rcx
    sub rax, rbx
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r13
parse_skip:
    lea rbx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rbx, r12
    jae sort
    cmp r13, LINES_MAX
    jae sort
    lea rax, [rsi + rbx]
    mov qword ptr [rdi], rax
    mov rax, r12
    sub rax, rbx
    mov qword ptr [rdi + 8], rax
    inc r13
sort:
    mov r12d, 1                 # width
sort_pass:
    cmp r12, r13
    jae sort_done
    xor r14d, r14d              # lo
merge_loop:
    cmp r14, r13
    jae pass_end
    lea r11, [r14 + r12]        # mid
    cmp r11, r13
    cmova r11, r13
    lea rbx, [r11 + r12]        # hi
    cmp rbx, r13
    cmova rbx, r13
    mov r8, r14                 # i
    mov r9, r11                 # j
    mov r10, r14                # k
merge_inner:
    cmp r8, r11
    jae take_right_rest
    cmp r9, rbx
    jae take_left_rest
    lea r15, [rip + lines]
    mov rax, r8
    shl rax, 4
    mov rdi, qword ptr [r15 + rax]
    mov rsi, qword ptr [r15 + rax + 8]
    mov rax, r9
    shl rax, 4
    mov rdx, qword ptr [r15 + rax]
    mov rcx, qword ptr [r15 + rax + 8]
    call cmp_le
    test eax, eax
    jz take_right
take_left:
    lea r15, [rip + lines]
    lea rdi, [rip + tmp]
    mov rax, r8
    shl rax, 4
    mov rcx, r10
    shl rcx, 4
    mov rdx, qword ptr [r15 + rax]
    mov qword ptr [rdi + rcx], rdx
    mov rdx, qword ptr [r15 + rax + 8]
    mov qword ptr [rdi + rcx + 8], rdx
    inc r8
    inc r10
    jmp merge_inner
take_right:
    lea r15, [rip + lines]
    lea rdi, [rip + tmp]
    mov rax, r9
    shl rax, 4
    mov rcx, r10
    shl rcx, 4
    mov rdx, qword ptr [r15 + rax]
    mov qword ptr [rdi + rcx], rdx
    mov rdx, qword ptr [r15 + rax + 8]
    mov qword ptr [rdi + rcx + 8], rdx
    inc r9
    inc r10
    jmp merge_inner
take_left_rest:
    cmp r8, r11
    jae merge_next
    lea r15, [rip + lines]
    lea rdi, [rip + tmp]
    mov rax, r8
    shl rax, 4
    mov rcx, r10
    shl rcx, 4
    mov rdx, qword ptr [r15 + rax]
    mov qword ptr [rdi + rcx], rdx
    mov rdx, qword ptr [r15 + rax + 8]
    mov qword ptr [rdi + rcx + 8], rdx
    inc r8
    inc r10
    jmp take_left_rest
take_right_rest:
    cmp r9, rbx
    jae merge_next
    lea r15, [rip + lines]
    lea rdi, [rip + tmp]
    mov rax, r9
    shl rax, 4
    mov rcx, r10
    shl rcx, 4
    mov rdx, qword ptr [r15 + rax]
    mov qword ptr [rdi + rcx], rdx
    mov rdx, qword ptr [r15 + rax + 8]
    mov qword ptr [rdi + rcx + 8], rdx
    inc r9
    inc r10
    jmp take_right_rest
merge_next:
    mov r14, rbx
    jmp merge_loop
pass_end:
    lea rsi, [rip + tmp]
    lea rdi, [rip + lines]
    mov rcx, r13
    add rcx, rcx
    rep movsq
    add r12, r12
    jmp sort_pass
sort_done:
    # build output
    lea rdi, [rip + outbuf]
    lea rbx, [rip + lines]
    xor r14d, r14d
out_loop:
    cmp r14, r13
    jae out_done
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    inc r14
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                # total length
write_loop:
    test rdx, rdx
    jz finish
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop
finish:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp finish

# cmp_le: rdi=ptr a, rsi=len a, rdx=ptr b, rcx=len b; returns eax=1 if a<=b else 0
cmp_le:
    mov r15, rsi
    cmp r15, rcx
    cmova r15, rcx              # m = min(la, lb)
    xor eax, eax
cmp_loop:
    cmp rax, r15
    jae cmp_prefix
    movzx ebp, byte ptr [rdi + rax]
    cmp bpl, byte ptr [rdx + rax]
    jne cmp_differ
    inc rax
    jmp cmp_loop
cmp_differ:
    setb al
    movzx eax, al
    ret
cmp_prefix:
    cmp rsi, rcx
    setbe al
    movzx eax, al
    ret

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
outbuf:
    .space OUTBUF_SIZE
.balign 16
lines:
    .space LINES_MAX * 16
tmp:
    .space LINES_MAX * 16
