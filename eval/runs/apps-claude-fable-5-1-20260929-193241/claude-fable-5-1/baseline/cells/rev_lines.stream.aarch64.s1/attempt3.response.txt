.text
.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
    mov     x25, #65536             // buffer capacity

read_loop:
    cmp     x20, x25
    b.hs    read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    sub     x2, x25, x20
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x22, #0                 // output length
    mov     x23, #0                 // line start index

line_loop:
    cmp     x23, x20
    b.hs    output
    mov     x24, x23                // scan index j
find_nl:
    cmp     x24, x20
    b.hs    found
    ldrb    w0, [x19, x24]
    cmp     w0, #10
    b.eq    found
    add     x24, x24, #1
    b       find_nl
found:
    mov     x26, x24                // k = j
rev_loop:
    cmp     x26, x23
    b.ls    rev_done
    sub     x26, x26, #1
    ldrb    w0, [x19, x26]
    strb    w0, [x21, x22]
    add     x22, x22, #1
    b       rev_loop
rev_done:
    mov     w0, #10
    strb    w0, [x21, x22]
    add     x22, x22, #1
    add     x23, x24, #1
    b       line_loop

output:
    mov     x23, #0                 // bytes written
write_loop:
    cmp     x23, x22
    b.hs    done
    mov     x8, #64
    mov     x0, #1
    add     x1, x21, x23
    sub     x2, x22, x23
    svc     #0
    cmp     x0, #0
    b.le    done
    add     x23, x23, x0
    b       write_loop

done:
    mov     x8, #93
    mov     x0, #0
    svc     #0
    b       done

.bss
.balign 16
inbuf:
    .space  65536
.balign 16
outbuf:
    .space  65552
