.equ IN_SIZE, 65552
.equ OUT_SIZE, 409600
.equ STK_SIZE, 8192

.bss
.balign 16
in_buf: .space IN_SIZE
.balign 16
out_buf: .space OUT_SIZE
.balign 16
stk: .space STK_SIZE
numtmp: .space 32

.section .rodata
err_msg: .ascii "error\n"

.text
.globl _start
_start:
    lea r13, [rip + in_buf]
read_loop:
    lea rdx, [rip + in_buf + IN_SIZE]
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rip + in_buf]
    lea r14, [rip + out_buf]
    lea rbp, [rip + stk]
line_loop:
    cmp r12, r13
    jae finish
    mov rbx, rbp
tok_loop:
    cmp r12, r13
    jae end_line
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    cmp al, 0x0A
    je end_line
    mov r8, r12
tok_scan:
    inc r12
    cmp r12, r13
    jae tok_end
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je tok_end
    cmp al, 0x09
    je tok_end
    cmp al, 0x0A
    jne tok_scan
tok_end:
    mov rcx, r12
    sub rcx, r8
    movzx eax, byte ptr [r8]
    cmp rcx, 1
    jne parse_num
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
parse_num:
    mov rsi, r8
    xor r9d, r9d
    cmp al, '-'
    jne 1f
    inc rsi
    mov r9d, 1
1:
    mov rcx, r12
    sub rcx, rsi
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
2:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    cmp rsi, r12
    jb 2b
    test r9d, r9d
    jz 3f
    neg rax
3:
    lea rdx, [rip + stk + STK_SIZE]
    cmp rbx, rdx
    jae line_error
    mov qword ptr [rbx], rax
    add rbx, 8
    jmp tok_loop
skip_sep:
    inc r12
    jmp tok_loop

op_add:
    call pop2
    add rax, rcx
    jmp push_res
op_sub:
    call pop2
    sub rax, rcx
    jmp push_res
op_mul:
    call pop2
    imul rax, rcx
    jmp push_res
op_div:
    call pop2
    call divchk
    cqo
    idiv rcx
    jmp push_res
op_mod:
    call pop2
    call divchk
    cqo
    idiv rcx
    mov rax, rdx
push_res:
    mov qword ptr [rbx - 16], rax
    sub rbx, 8
    jmp tok_loop

pop2:
    mov rax, rbx
    sub rax, rbp
    cmp rax, 16
    jb 1f
    mov rax, qword ptr [rbx - 16]
    mov rcx, qword ptr [rbx - 8]
    ret
1:
    add rsp, 8
    jmp line_error

divchk:
    test rcx, rcx
    jz 1f
    cmp rcx, -1
    jne 2f
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je 1f
2:
    ret
1:
    add rsp, 8
    jmp line_error

line_error:
    cmp r12, r13
    jae 1f
    cmp byte ptr [r12], 0x0A
    je 1f
    inc r12
    jmp line_error
1:
    lea rsi, [rip + err_msg]
    mov rdi, r14
    mov ecx, 6
    rep movsb
    mov r14, rdi
    jmp after_line

end_line:
    mov rax, rbx
    sub rax, rbp
    cmp rax, 8
    jne line_error
    mov rax, qword ptr [rbp]
    lea rdi, [rip + numtmp + 32]
    test rax, rax
    jns 1f
    mov byte ptr [r14], '-'
    inc r14
    neg rax
1:
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
    lea rcx, [rip + numtmp + 32]
    sub rcx, rdi
    mov rsi, rdi
    mov rdi, r14
    rep movsb
    mov r14, rdi
    mov byte ptr [r14], 0x0A
    inc r14

after_line:
    cmp r12, r13
    jae 1f
    inc r12
1:
    lea rax, [rip + out_buf + OUT_SIZE - 64]
    cmp r14, rax
    jb line_loop
    call flush
    jmp line_loop

flush:
    lea rsi, [rip + out_buf]
    mov rdx, r14
    sub rdx, rsi
    jz 1f
    mov eax, 1
    mov edi, 1
    syscall
    lea r14, [rip + out_buf]
1:
    ret

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp finish
