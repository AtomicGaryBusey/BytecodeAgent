.text
.globl _start

.equ INSIZE, 0x11000
.equ OUTSIZE, 0x40000
.equ STKSIZE, 0x40000

// registers:
//  x19 = input cursor        x20 = input end
//  x21 = value stack base    x22 = value stack count
//  x23 = output buffer base  x24 = output write pointer
//  x26 = current line end    x27 = start of next line

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, x19
    add     x9, x19, #0x11, lsl #12        // inbuf + INSIZE
read_loop:
    sub     x2, x9, x20
    cbz     x2, read_done
    mov     x0, #0
    mov     x1, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x21, vstack
    add     x21, x21, :lo12:vstack
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf
    mov     x24, x23

main_loop:
    cmp     x19, x20
    b.hs    all_done
    // find end of line
    mov     x26, x19
find_nl:
    cmp     x26, x20
    b.hs    found_end
    ldrb    w0, [x26]
    cmp     w0, #10
    b.eq    found_nl
    add     x26, x26, #1
    b       find_nl
found_nl:
    add     x27, x26, #1
    b       eval_line
found_end:
    mov     x27, x26

eval_line:
    mov     x22, #0
tok_loop:
    cmp     x19, x26
    b.hs    line_end
    ldrb    w0, [x19]
    cmp     w0, #32
    b.eq    skip_sep
    cmp     w0, #9
    b.eq    skip_sep
    // token starts at x19; find its end in x1
    mov     x1, x19
tok_end_loop:
    cmp     x1, x26
    b.hs    tok_end
    ldrb    w0, [x1]
    cmp     w0, #32
    b.eq    tok_end
    cmp     w0, #9
    b.eq    tok_end
    add     x1, x1, #1
    b       tok_end_loop
tok_end:
    ldrb    w0, [x19]
    sub     x2, x1, x19
    cmp     x2, #1
    b.ne    parse_literal
    cmp     w0, #'+'
    b.eq    op_add
    cmp     w0, #'-'
    b.eq    op_sub
    cmp     w0, #'*'
    b.eq    op_mul
    cmp     w0, #'/'
    b.eq    op_div
    cmp     w0, #'%'
    b.eq    op_mod
parse_literal:
    mov     x3, x19
    mov     x4, #0
    cmp     w0, #'-'
    b.ne    no_neg
    mov     x4, #1
    add     x3, x3, #1
no_neg:
    sub     x5, x1, x3
    cmp     x5, #1
    b.lo    line_error
    cmp     x5, #18
    b.hi    line_error
    mov     x6, #0
    mov     x7, #10
digit_loop:
    cmp     x3, x1
    b.hs    digits_done
    ldrb    w0, [x3], #1
    sub     w0, w0, #'0'
    cmp     w0, #9
    b.hi    line_error
    madd    x6, x6, x7, x0
    b       digit_loop
digits_done:
    cbz     x4, push_val
    neg     x6, x6
push_val:
    str     x6, [x21, x22, lsl #3]
    add     x22, x22, #1
    mov     x19, x1
    b       tok_loop

skip_sep:
    add     x19, x19, #1
    b       tok_loop

// operators: pop b (x10), a (x9); result stored in place of a
op_add:
    cmp     x22, #2
    b.lo    line_error
    sub     x22, x22, #1
    ldr     x10, [x21, x22, lsl #3]
    sub     x11, x22, #1
    ldr     x9, [x21, x11, lsl #3]
    add     x9, x9, x10
    str     x9, [x21, x11, lsl #3]
    mov     x19, x1
    b       tok_loop
op_sub:
    cmp     x22, #2
    b.lo    line_error
    sub     x22, x22, #1
    ldr     x10, [x21, x22, lsl #3]
    sub     x11, x22, #1
    ldr     x9, [x21, x11, lsl #3]
    sub     x9, x9, x10
    str     x9, [x21, x11, lsl #3]
    mov     x19, x1
    b       tok_loop
op_mul:
    cmp     x22, #2
    b.lo    line_error
    sub     x22, x22, #1
    ldr     x10, [x21, x22, lsl #3]
    sub     x11, x22, #1
    ldr     x9, [x21, x11, lsl #3]
    mul     x9, x9, x10
    str     x9, [x21, x11, lsl #3]
    mov     x19, x1
    b       tok_loop
op_div:
    cmp     x22, #2
    b.lo    line_error
    sub     x22, x22, #1
    ldr     x10, [x21, x22, lsl #3]
    sub     x11, x22, #1
    ldr     x9, [x21, x11, lsl #3]
    cbz     x10, line_error
    cmn     x10, #1
    b.ne    1f
    mov     x12, #1
    lsl     x12, x12, #63
    cmp     x9, x12
    b.eq    line_error
1:
    sdiv    x9, x9, x10
    str     x9, [x21, x11, lsl #3]
    mov     x19, x1
    b       tok_loop
op_mod:
    cmp     x22, #2
    b.lo    line_error
    sub     x22, x22, #1
    ldr     x10, [x21, x22, lsl #3]
    sub     x11, x22, #1
    ldr     x9, [x21, x11, lsl #3]
    cbz     x10, line_error
    cmn     x10, #1
    b.ne    1f
    mov     x12, #1
    lsl     x12, x12, #63
    cmp     x9, x12
    b.eq    line_error
1:
    sdiv    x12, x9, x10
    msub    x9, x12, x10, x9
    str     x9, [x21, x11, lsl #3]
    mov     x19, x1
    b       tok_loop

line_end:
    cmp     x22, #1
    b.ne    line_error
    ldr     x0, [x21]
    bl      emit_number
    mov     x19, x27
    b       main_loop

line_error:
    bl      emit_error
    mov     x19, x27
    b       main_loop

all_done:
    bl      flush
    mov     x0, #0
    mov     x8, #94
    svc     #0
    b       all_done

// flush output buffer: clobbers x0, x1, x2, x8; resets x24
flush:
    mov     x1, x23
flush_loop:
    sub     x2, x24, x1
    cbz     x2, flush_done
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    flush_done
    add     x1, x1, x0
    b       flush_loop
flush_done:
    mov     x24, x23
    ret

// ensure at least 4096 bytes free in output buffer
check_flush:
    sub     x0, x24, x23
    cmp     x0, #0x3F, lsl #12
    b.lo    1f
    b       flush
1:
    ret

// emit "error\n"
emit_error:
    str     x30, [sp, #-16]!
    bl      check_flush
    adr     x1, errstr
    mov     x2, #6
1:
    ldrb    w0, [x1], #1
    strb    w0, [x24], #1
    subs    x2, x2, #1
    b.ne    1b
    ldr     x30, [sp], #16
    ret

// emit signed decimal in x0 followed by newline
emit_number:
    sub     sp, sp, #48
    str     x30, [sp, #32]
    mov     x9, x0
    bl      check_flush
    add     x10, sp, #32
    mov     x11, x9
    cmp     x9, #0
    b.ge    1f
    mov     w0, #'-'
    strb    w0, [x24], #1
    neg     x11, x9
1:
    mov     x12, #10
2:
    udiv    x0, x11, x12
    msub    x1, x0, x12, x11
    add     w1, w1, #'0'
    strb    w1, [x10, #-1]!
    mov     x11, x0
    cbnz    x11, 2b
    add     x1, sp, #32
3:
    cmp     x10, x1
    b.hs    4f
    ldrb    w0, [x10], #1
    strb    w0, [x24], #1
    b       3b
4:
    mov     w0, #10
    strb    w0, [x24], #1
    ldr     x30, [sp, #32]
    add     sp, sp, #48
    ret

.section .rodata
errstr:
    .ascii  "error\n"

.bss
.balign 16
inbuf:
    .space  INSIZE
.balign 16
outbuf:
    .space  OUTSIZE
.balign 16
vstack:
    .space  STKSIZE
