.bss
.balign 16
inbuf:  .space 32800
outbuf: .space 34000
recs:   .space 16384
aux:    .space 16384

.text
.globl _start
_start:
    xor ebx, ebx
.Lread:
    mov edx, 32800
    sub edx, ebx
    jle .Lreaddone
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle .Lreaddone
    add rbx, rax
    jmp .Lread
.Lreaddone:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + inbuf]
    add rdi, rbx
    lea r8, [rip + recs]
    xor r9d, r9d
.Lparse:
    cmp rsi, rdi
    jae .Lparsedone
    cmp r9, 1024
    jae .Lparsedone
    mov r10, rsi
.Lscan:
    cmp rsi, rdi
    jae .Lnonl
    cmp byte ptr [rsi], 10
    je .Lnl
    inc rsi
    jmp .Lscan
.Lnl:
    mov qword ptr [r8], r10
    mov rax, rsi
    sub rax, r10
    mov qword ptr [r8 + 8], rax
    add r8, 16
    inc r9
    inc rsi
    jmp .Lparse
.Lnonl:
    mov qword ptr [r8], r10
    mov rax, rsi
    sub rax, r10
    mov qword ptr [r8 + 8], rax
    add r8, 16
    inc r9
.Lparsedone:
    mov r12, r9
    lea r13, [rip + recs]
    lea r14, [rip + aux]
    mov r15, 1
.Lwidth:
    cmp r15, r12
    jae .Lsorted
    xor ebx, ebx
.Lpass:
    cmp rbx, r12
    jae .Lpassdone
    lea rcx, [rbx + r15]
    cmp rcx, r12
    cmova rcx, r12
    lea rdx, [rcx + r15]
    cmp rdx, r12
    cmova rdx, r12
    mov r8, rbx
    shl r8, 4
    lea rdi, [r14 + r8]
    add r8, r13
    mov r9, rcx
    shl r9, 4
    add r9, r13
    mov r10, r9
    mov r11, rdx
    shl r11, 4
    add r11, r13
.Lmerge:
    cmp r8, r9
    jae .Lrestright
    cmp r10, r11
    jae .Lrestleft
    call cmp_recs
    test eax, eax
    jnz .Ltakeright
.Ltakeleft:
    movups xmm0, xmmword ptr [r8]
    movups xmmword ptr [rdi], xmm0
    add r8, 16
    add rdi, 16
    jmp .Lmerge
.Ltakeright:
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [rdi], xmm0
    add r10, 16
    add rdi, 16
    jmp .Lmerge
.Lrestleft:
    cmp r8, r9
    jae .Lmergedone
    movups xmm0, xmmword ptr [r8]
    movups xmmword ptr [rdi], xmm0
    add r8, 16
    add rdi, 16
    jmp .Lrestleft
.Lrestright:
    cmp r10, r11
    jae .Lmergedone
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [rdi], xmm0
    add r10, 16
    add rdi, 16
    jmp .Lrestright
.Lmergedone:
    lea rbx, [rbx + r15*2]
    jmp .Lpass
.Lpassdone:
    xchg r13, r14
    add r15, r15
    jmp .Lwidth
.Lsorted:
    lea rdi, [rip + outbuf]
.Lout:
    test r12, r12
    jz .Loutdone
    mov rsi, qword ptr [r13]
    mov rcx, qword ptr [r13 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add r13, 16
    dec r12
    jmp .Lout
.Loutdone:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
.Lwrite:
    test rdx, rdx
    jle .Lexit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lexit
    add rsi, rax
    sub rdx, rax
    jmp .Lwrite
.Lexit:
    mov eax, 231
    xor edi, edi
    syscall
    jmp .Lexit

# returns eax = 1 if record [r8] > record [r10], else 0
cmp_recs:
    mov rsi, qword ptr [r8]
    mov rdx, qword ptr [r10]
    mov rcx, qword ptr [r8 + 8]
    mov rax, qword ptr [r10 + 8]
    cmp rcx, rax
    cmova rcx, rax
1:
    test rcx, rcx
    jz 2f
    movzx eax, byte ptr [rsi]
    cmp al, byte ptr [rdx]
    ja 3f
    jb 4f
    inc rsi
    inc rdx
    dec rcx
    jmp 1b
2:
    mov rcx, qword ptr [r8 + 8]
    cmp rcx, qword ptr [r10 + 8]
    ja 3f
4:
    xor eax, eax
    ret
3:
    mov eax, 1
    ret
