.text
.globl _start
_start:
    xor r12d, r12d              # lines
    xor r13d, r13d              # words
    xor r14d, r14d              # bytes
    xor r15d, r15d              # in_word flag

read_loop:
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
process:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    dec rcx
    jnz process
    jmp read_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call fmt
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call fmt
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call fmt
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp done_reading

# fmt: rax = value, rdi = destination pointer; advances rdi past digits
fmt:
    lea rsi, [rip + tmp + 32]
    mov ebx, 10
1:
    xor edx, edx
    div rbx
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rcx, [rip + tmp + 32]
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    inc rsi
    cmp rsi, rcx
    jne 2b
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
tmp:
    .space 32
