.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x2, #65536
    add x2, x2, #16
    subs x2, x2, x20
    ble read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x23, hexdigits
    add x23, x23, :lo12:hexdigits
    mov x22, #0
row_loop:
    sub x24, x20, x22
    cbz x24, finish
    mov x25, #16
    cmp x24, x25
    csel x24, x24, x25, lo
    mov x0, x22
    bl emit_hex8
    mov w0, #0x20
    strb w0, [x21], #1
    strb w0, [x21], #1
    mov x25, #0
col_loop:
    cmp x25, x24
    bhs col_pad
    add x26, x22, x25
    ldrb w0, [x19, x26]
    lsr w1, w0, #4
    ldrb w1, [x23, x1]
    and w2, w0, #15
    ldrb w2, [x23, x2]
    strb w1, [x21], #1
    strb w2, [x21], #1
    mov w1, #0x20
    strb w1, [x21], #1
    b col_next
col_pad:
    mov w1, #0x20
    strb w1, [x21], #1
    strb w1, [x21], #1
    strb w1, [x21], #1
col_next:
    cmp x25, #7
    bne 1f
    mov w1, #0x20
    strb w1, [x21], #1
1:
    add x25, x25, #1
    cmp x25, #16
    blo col_loop
    mov w1, #0x20
    strb w1, [x21], #1
    mov w1, #0x7c
    strb w1, [x21], #1
    mov x25, #0
chr_loop:
    add x26, x22, x25
    ldrb w0, [x19, x26]
    sub w1, w0, #0x20
    cmp w1, #0x5e
    mov w2, #0x2e
    csel w0, w0, w2, ls
    strb w0, [x21], #1
    add x25, x25, #1
    cmp x25, x24
    blo chr_loop
    mov w1, #0x7c
    strb w1, [x21], #1
    mov w1, #0x0a
    strb w1, [x21], #1
    add x22, x22, x24
    b row_loop
finish:
    mov x0, x20
    bl emit_hex8
    mov w1, #0x0a
    strb w1, [x21], #1
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    sub x25, x21, x24
write_loop:
    cbz x25, do_exit
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    cmp x0, #0
    ble do_exit
    add x24, x24, x0
    sub x25, x25, x0
    b write_loop
do_exit:
    mov x0, #0
    mov x8, #94
    svc #0
    b do_exit

// emit 8 hex digits of x0 at x21, advancing x21; clobbers x1, x2
emit_hex8:
    mov x1, #28
1:
    lsr x2, x0, x1
    and x2, x2, #15
    ldrb w2, [x23, x2]
    strb w2, [x21], #1
    subs x1, x1, #4
    bpl 1b
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 323616
