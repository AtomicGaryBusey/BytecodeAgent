.text
.globl _start
_start:
    xor r12d, r12d            # lines
    xor r13d, r13d            # words
    xor r14d, r14d            # bytes
    xor r15d, r15d            # in-word flag
.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle .Ldone
    add r14, rax
    lea rsi, [rip + buf]
    lea rdx, [rsi + rax]
.Lscan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je 2f
    sub al, 9
    cmp al, 4
    jbe 2f
    test r15d, r15d
    jnz 3f
    inc r13
    mov r15d, 1
    jmp 3f
2:
    xor r15d, r15d
3:
    cmp rsi, rdx
    jb .Lscan
    jmp .Lread

.Ldone:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call .Lputnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call .Lputnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call .Lputnum
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall
.Lhang:
    jmp .Lhang

# rax = value, rdi = destination pointer; advances rdi
.Lputnum:
    sub rsp, 32
    lea rsi, [rsp + 32]
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    lea rax, [rsp + 32]
    cmp rsi, rax
    jb 2b
    add rsp, 32
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
