.section .rodata
hexdig: .ascii "0123456789abcdef"

.bss
.balign 16
inbuf: .space 65536
outbuf: .space 331776

.text
.globl _start
_start:
    xor ebx, ebx
1:
    mov edx, 65536
    sub edx, ebx
    jz 2f
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    lea r9, [rip + hexdig]
    lea r12, [rip + inbuf]
    lea r13, [rip + outbuf]
    xor r14d, r14d
    mov r15, rbx
row_loop:
    test r15, r15
    jz finish
    mov eax, r14d
    call put_hex32
    mov word ptr [r13], 0x2020
    add r13, 2
    mov rcx, r15
    cmp rcx, 16
    jbe 3f
    mov ecx, 16
3:
    mov rbp, rcx
    xor edx, edx
hex_loop:
    cmp rdx, rbp
    jae pad_slot
    movzx eax, byte ptr [r12 + rdx]
    mov r8d, eax
    shr r8d, 4
    mov r8b, byte ptr [r9 + r8]
    mov byte ptr [r13], r8b
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [r13 + 1], al
    mov byte ptr [r13 + 2], 0x20
    jmp next_slot
pad_slot:
    mov byte ptr [r13], 0x20
    mov byte ptr [r13 + 1], 0x20
    mov byte ptr [r13 + 2], 0x20
next_slot:
    add r13, 3
    cmp edx, 7
    jne 4f
    mov byte ptr [r13], 0x20
    inc r13
4:
    inc rdx
    cmp rdx, 16
    jb hex_loop
    mov word ptr [r13], 0x7c20
    add r13, 2
    xor edx, edx
asc_loop:
    movzx eax, byte ptr [r12 + rdx]
    lea ecx, [rax - 0x20]
    cmp ecx, 0x5e
    jbe 5f
    mov al, 0x2e
5:
    mov byte ptr [r13], al
    inc r13
    inc rdx
    cmp rdx, rbp
    jb asc_loop
    mov word ptr [r13], 0x0a7c
    add r13, 2
    add r12, rbp
    add r14, rbp
    sub r15, rbp
    jmp row_loop
finish:
    mov eax, r14d
    call put_hex32
    mov byte ptr [r13], 10
    inc r13
    lea rsi, [rip + outbuf]
    mov rdx, r13
    sub rdx, rsi
wr_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp wr_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall

put_hex32:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r13], dl
    inc r13
    dec ecx
    jnz 1b
    ret
