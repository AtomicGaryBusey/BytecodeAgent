```
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 458752

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov edx, INBUF_SIZE
    sub rdx, r12
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + inbuf]
    lea r14, [r13 + r12]
    lea r15, [rip + outbuf]
    mov r12, rsp

line_loop:
    cmp r13, r14
    jae finish
    xor ebx, ebx
token_loop:
    cmp r13, r14
    jae end_of_line
    movzx eax, byte ptr [r13]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    cmp al, 0x0A
    je end_of_line
    mov rdi, r13
find_end:
    inc r13
    cmp r13, r14
    jae token_found
    movzx eax, byte ptr [r13]
    cmp al, 0x20
    je token_found
    cmp al, 0x09
    je token_found
    cmp al, 0x0A
    jne find_end
token_found:
    mov rcx, r13
    sub rcx, rdi
    movzx eax, byte ptr [rdi]
    cmp rcx, 1
    jne parse_literal
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
parse_literal:
    xor ebp, ebp
    cmp al, '-'
    jne no_neg
    inc ebp
    inc rdi
    dec rcx
no_neg:
    test rcx, rcx
    jz error
    cmp rcx, 18
    ja error
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [rdi]
    sub edx, '0'
    cmp edx, 9
    ja error
    lea rax, [rax + rax*4]
    lea rax, [rdx + rax*2]
    inc rdi
    dec rcx
    jnz digit_loop
    test ebp, ebp
    jz push_val
    neg rax
push_val:
    lea rdx, [rip + stack]
    mov qword ptr [rdx + rbx*8], rax
    inc rbx
    jmp token_loop
skip_sep:
    inc r13
    jmp token_loop

op_add:
    call pop2
    add rax, rdx
    jmp push_val
op_sub:
    call pop2
    sub rax, rdx
    jmp push_val
op_mul:
    call pop2
    imul rax, rdx
    jmp push_val
op_div:
    call pop2
    call check_div
    cqo
    idiv rcx
    jmp push_val
op_mod:
    call pop2
    call check_div
    cqo
    idiv rcx
    mov rax, rdx
    jmp push_val

check_div:
    test rdx, rdx
    jz error
    mov rcx, rdx
    cmp rdx, -1
    jne cd_ok
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error
cd_ok:
    ret

pop2:
    cmp rbx, 2
    jb error
    dec rbx
    lea rcx, [rip + stack]
    mov rdx, qword ptr [rcx + rbx*8]
    dec rbx
    mov rax, qword ptr [rcx + rbx*8]
    ret

end_of_line:
    cmp rbx, 1
    jne error
    lea rdx, [rip + stack]
    mov rax, qword ptr [rdx]
    call print_num
    jmp consume_newline

error:
    mov rsp, r12
err_skip:
    cmp r13, r14
    jae err_emit
    cmp byte ptr [r13], 0x0A
    je err_emit
    inc r13
    jmp err_skip
err_emit:
    mov byte ptr [r15], 'e'
    mov byte ptr [r15 + 1], 'r'
    mov byte ptr [r15 + 2], 'r'
    mov byte ptr [r15 + 3], 'o'
    mov byte ptr [r15 + 4], 'r'
    mov byte ptr [r15 + 5], 0x0A
    add r15, 6
consume_newline:
    cmp r13, r14
    jae line_loop
    inc r13
    jmp line_loop

print_num:
    test rax, rax
    jns pn_pos
    mov byte ptr [r15], '-'
    inc r15
    neg rax
pn_pos:
    lea rsi, [rip + numbuf + 24]
    mov ecx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz pn_loop
    lea rdx, [rip + numbuf + 24]
    sub rdx, rsi
pn_copy:
    mov al, byte ptr [rsi]
    mov byte ptr [r15], al
    inc rsi
    inc r15
    dec rdx
    jnz pn_copy
    mov byte ptr [r15], 0x0A
    inc r15
    ret

finish:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

.bss
.balign 16
inbuf:
    .space INBUF_SIZE + 16
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 16
stack:
    .space 16384 * 8
.balign 16
numbuf:
    .space 32
```
