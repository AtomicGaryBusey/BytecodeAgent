.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    adrp x25, hexdigits
    add x25, x25, :lo12:hexdigits

read_loop:
    mov w2, #0x10000
    add w2, w2, #64
    subs x2, x2, x20
    b.le read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, #0

row_loop:
    sub x23, x20, x22
    cbz x23, finish
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, ls

    mov x0, x22
    bl hex8
    mov w0, #0x20
    strb w0, [x21], #1
    strb w0, [x21], #1

    mov x26, #0
byte_loop:
    cmp x26, x23
    b.hs pad
    add x1, x22, x26
    ldrb w0, [x19, x1]
    lsr x3, x0, #4
    ldrb w3, [x25, x3]
    strb w3, [x21], #1
    and x3, x0, #15
    ldrb w3, [x25, x3]
    strb w3, [x21], #1
    mov w0, #0x20
    strb w0, [x21], #1
    b next_byte
pad:
    mov w0, #0x20
    strb w0, [x21], #1
    strb w0, [x21], #1
    strb w0, [x21], #1
next_byte:
    cmp x26, #7
    b.ne 2f
    strb w0, [x21], #1
2:
    add x26, x26, #1
    cmp x26, #16
    b.lo byte_loop

    mov w0, #0x20
    strb w0, [x21], #1
    mov w0, #0x7c
    strb w0, [x21], #1

    mov x26, #0
char_loop:
    cmp x26, x23
    b.hs chars_done
    add x1, x22, x26
    ldrb w0, [x19, x1]
    cmp w0, #0x20
    b.lo 3f
    cmp w0, #0x7e
    b.ls 4f
3:
    mov w0, #0x2e
4:
    strb w0, [x21], #1
    add x26, x26, #1
    b char_loop
chars_done:
    mov w0, #0x7c
    strb w0, [x21], #1
    mov w0, #0x0a
    strb w0, [x21], #1

    add x22, x22, x23
    b row_loop

finish:
    mov x0, x20
    bl hex8
    mov w0, #0x0a
    strb w0, [x21], #1

    adrp x27, outbuf
    add x27, x27, :lo12:outbuf
    sub x28, x21, x27
write_loop:
    cbz x28, do_exit
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le do_exit
    add x27, x27, x0
    sub x28, x28, x0
    b write_loop

do_exit:
    mov x0, #0
    mov x8, #93
    svc #0
    b do_exit

// x0 = value; writes 8 lowercase hex digits at x21, advancing x21
hex8:
    mov x2, #28
1:
    lsr x3, x0, x2
    and x3, x3, #15
    ldrb w3, [x25, x3]
    strb w3, [x21], #1
    subs x2, x2, #4
    b.pl 1b
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536 + 128
.balign 16
outbuf:
    .space 4096 * 80 + 64
