INBUF_SIZE = 49216
OUTBUF_SIZE = 70000

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, INBUF_SIZE
    sub edx, ebx
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    mov r13, rbx
    lea rdi, [rip + outbuf]
    xor r14d, r14d
    lea r15, [rip + alphabet]
enc_loop:
    cmp r13, 3
    jb enc_tail
    movzx eax, byte ptr [r12]
    movzx ecx, byte ptr [r12 + 1]
    movzx edx, byte ptr [r12 + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov ecx, eax
    shr ecx, 18
    call emit
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    call emit
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    call emit
    mov ecx, eax
    and ecx, 63
    call emit
    add r12, 3
    sub r13, 3
    jmp enc_loop
enc_tail:
    test r13, r13
    jz enc_done
    movzx eax, byte ptr [r12]
    shl eax, 16
    cmp r13, 1
    je tail1
    movzx ecx, byte ptr [r12 + 1]
    shl ecx, 8
    or eax, ecx
    mov ecx, eax
    shr ecx, 18
    call emit
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    call emit
    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    call emit
    mov ecx, 64
    call emit
    jmp enc_done
tail1:
    mov ecx, eax
    shr ecx, 18
    call emit
    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    call emit
    mov ecx, 64
    call emit
    mov ecx, 64
    call emit
enc_done:
    test r14d, r14d
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi
do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall

# emit: ecx = symbol index (0-63, 64 = '='); rdi = out ptr; r14d = column. Preserves eax.
emit:
    mov cl, byte ptr [r15 + rcx]
    mov byte ptr [rdi], cl
    inc rdi
    inc r14d
    cmp r14d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="

.bss
.balign 16
inbuf:
    .zero INBUF_SIZE
.balign 16
outbuf:
    .zero OUTBUF_SIZE
