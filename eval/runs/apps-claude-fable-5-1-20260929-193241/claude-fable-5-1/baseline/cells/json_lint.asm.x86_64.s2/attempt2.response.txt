.text
.globl _start

.equ BUFCAP, 65600

_start:
    lea rbx, [rip + buf]
    xor r12, r12
read_loop:
    mov edx, BUFCAP
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov byte ptr [rbx + r12], 0
    xor esi, esi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_ok]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall

skip_ws:
1:  movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    je 2f
    cmp al, 0x0A
    je 2f
    cmp al, 0x0D
    jne 3f
2:  inc rsi
    jmp 1b
3:  ret

parse_value:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je lit_true
    cmp al, 0x66
    je lit_false
    cmp al, 0x6E
    je lit_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    jmp parse_number

lit_true:
    lea rdi, [rip + s_true]
    jmp lit
lit_false:
    lea rdi, [rip + s_false]
    jmp lit
lit_null:
    lea rdi, [rip + s_null]
lit:
1:  movzx eax, byte ptr [rdi]
    test al, al
    jz 2f
    cmp al, byte ptr [rbx + rsi]
    jne fail
    inc rsi
    inc rdi
    jmp 1b
2:  ret

parse_string:
    inc rsi
1:  movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x22
    je 9f
    cmp al, 0x5C
    je 5f
    cmp al, 0x20
    jb fail
    inc rsi
    jmp 1b
9:  inc rsi
    ret
5:  inc rsi
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x75
    je 7f
    cmp al, 0x22
    je 6f
    cmp al, 0x5C
    je 6f
    cmp al, 0x2F
    je 6f
    cmp al, 0x62
    je 6f
    cmp al, 0x66
    je 6f
    cmp al, 0x6E
    je 6f
    cmp al, 0x72
    je 6f
    cmp al, 0x74
    je 6f
    jmp fail
6:  inc rsi
    jmp 1b
7:  inc rsi
    mov ecx, 4
2:  movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe 3f
    or al, 0x20
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
3:  inc rsi
    dec ecx
    jnz 2b
    jmp 1b

parse_number:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2D
    jne 1f
    inc rsi
    movzx eax, byte ptr [rbx + rsi]
1:  cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
    cmp al, 0x30
    je 3f
2:  movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb 3f
    cmp al, 0x39
    ja 3f
    inc rsi
    jmp 2b
3:  movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2E
    jne 4f
    inc rsi
    call digits1
4:  movzx eax, byte ptr [rbx + rsi]
    or al, 0x20
    cmp al, 0x65
    jne 6f
    inc rsi
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2B
    je 5f
    cmp al, 0x2D
    jne 7f
5:  inc rsi
7:  call digits1
6:  ret

digits1:
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
1:  movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x30
    jb 2f
    cmp al, 0x39
    ja 2f
    inc rsi
    jmp 1b
2:  ret

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rbx + rsi], 0x5D
    jne 1f
    inc rsi
    ret
1:  call parse_value
    call skip_ws
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2C
    je 2f
    cmp al, 0x5D
    jne fail
    inc rsi
    ret
2:  inc rsi
    call skip_ws
    jmp 1b

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rbx + rsi], 0x7D
    jne 1f
    inc rsi
    ret
1:  cmp byte ptr [rbx + rsi], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [rbx + rsi], 0x3A
    jne fail
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 0x2C
    je 2f
    cmp al, 0x7D
    jne fail
    inc rsi
    ret
2:  inc rsi
    call skip_ws
    jmp 1b

fail:
    mov rax, rsi
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:  xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    mov r13, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_err]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    lea rdx, [rip + numbuf + 32]
    sub rdx, r13
    syscall
    mov eax, 231
    mov edi, 1
    syscall

.section .rodata
s_ok:    .ascii "ok\n"
s_err:   .ascii "error at "
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"

.bss
.balign 16
buf:    .space BUFCAP + 16
numbuf: .space 32
