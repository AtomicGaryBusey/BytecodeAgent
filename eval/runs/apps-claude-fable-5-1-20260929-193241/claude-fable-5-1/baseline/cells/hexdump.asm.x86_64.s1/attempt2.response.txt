.text
.globl _start
_start:
    xor r12d, r12d
1:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jle 2f
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    lea r14, [rip + outbuf]
    lea rbp, [rip + hextab]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae done
    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe 3f
    mov r15d, 16
3:
    mov edi, r13d
    call put_hex32
    mov word ptr [r14], 0x2020
    add r14, 2
    lea rbx, [rip + inbuf]
    add rbx, r13
    xor ecx, ecx
hexloop:
    cmp rcx, r15
    jae hexpad
    movzx eax, byte ptr [rbx + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r14], dl
    and eax, 15
    mov al, byte ptr [rbp + rax]
    mov byte ptr [r14 + 1], al
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
    jmp hexnext
hexpad:
    mov byte ptr [r14], 0x20
    mov word ptr [r14 + 1], 0x2020
    add r14, 3
hexnext:
    cmp ecx, 7
    jne 4f
    mov byte ptr [r14], 0x20
    inc r14
4:
    inc ecx
    cmp ecx, 16
    jb hexloop
    mov word ptr [r14], 0x7c20
    add r14, 2
    xor ecx, ecx
chrloop:
    cmp rcx, r15
    jae chrdone
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, 0x20
    jb 5f
    cmp eax, 0x7e
    jbe 6f
5:
    mov eax, 0x2e
6:
    mov byte ptr [r14], al
    inc r14
    inc ecx
    jmp chrloop
chrdone:
    mov word ptr [r14], 0x0a7c
    add r14, 2
    add r13, 16
    jmp row_loop
done:
    mov edi, r12d
    call put_hex32
    mov byte ptr [r14], 0x0a
    inc r14
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
wloop:
    test rdx, rdx
    jle wdone
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wdone
    add rsi, rax
    sub rdx, rax
    jmp wloop
wdone:
    mov eax, 231
    xor edi, edi
    syscall

put_hex32:
    mov ecx, 8
1:
    rol edi, 4
    mov eax, edi
    and eax, 15
    mov al, byte ptr [rbp + rax]
    mov byte ptr [r14], al
    inc r14
    dec ecx
    jnz 1b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .space 65552
outbuf:
    .space 400000
