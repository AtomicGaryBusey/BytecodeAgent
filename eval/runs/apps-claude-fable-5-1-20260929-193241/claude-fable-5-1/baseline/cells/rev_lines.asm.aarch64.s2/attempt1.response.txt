.globl _start
_start:
    // read all input into inbuf
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total length
    mov x21, #65536
    add x21, x21, #64           // capacity
1:
    cmp x20, x21
    b.hs 2f
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:
    // process lines
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x23, #0                 // out length
    mov x24, #0                 // line start
3:
    cmp x24, x20
    b.hs 6f
    // find end of line
    mov x25, x24
4:
    cmp x25, x20
    b.hs 5f
    ldrb w0, [x19, x25]
    cmp w0, #10
    b.eq 5f
    add x25, x25, #1
    b 4b
5:
    // reverse [x24, x25) into outbuf
    mov x26, x25
7:
    cmp x26, x24
    b.ls 8f
    sub x26, x26, #1
    ldrb w0, [x19, x26]
    strb w0, [x22, x23]
    add x23, x23, #1
    b 7b
8:
    mov w0, #10
    strb w0, [x22, x23]
    add x23, x23, #1
    add x24, x25, #1
    b 3b
6:
    // write output
    mov x24, #0
9:
    cmp x24, x23
    b.hs 10f
    mov x0, #1
    add x1, x22, x24
    sub x2, x23, x24
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 10f
    add x24, x24, x0
    b 9b
10:
    mov x0, #0
    mov x8, #93
    svc #0
