.bss
.balign 16
inbuf: .space 65536
outbuf: .space 65536
tabA: .space 32768
tabB: .space 32768

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    lea rsi, [rip + inbuf]
    lea rdi, [rip + tabA]
    xor ecx, ecx
    xor r8d, r8d
parse_loop:
    cmp rcx, rbx
    jae parse_end
    cmp byte ptr [rsi + rcx], 10
    jne parse_next
    lea rax, [rsi + r8]
    mov qword ptr [rdi], rax
    mov rax, rcx
    sub rax, r8
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r12
    lea r8, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp r8, rbx
    jae sort
    lea rax, [rsi + r8]
    mov qword ptr [rdi], rax
    mov rax, rbx
    sub rax, r8
    mov qword ptr [rdi + 8], rax
    inc r12
sort:
    lea r13, [rip + tabA]
    lea r14, [rip + tabB]
    mov r15d, 1
width_loop:
    cmp r15, r12
    jae output
    xor r9d, r9d
pass_loop:
    cmp r9, r12
    jae pass_done
    lea r10, [r9 + r15]
    cmp r10, r12
    cmova r10, r12
    lea r11, [r10 + r15]
    cmp r11, r12
    cmova r11, r12
    mov rsi, r9
    shl rsi, 4
    add rsi, r13
    mov rdi, r10
    shl rdi, 4
    add rdi, r13
    mov rbp, rdi
    mov r8, r11
    shl r8, 4
    add r8, r13
    mov rdx, r9
    shl rdx, 4
    add rdx, r14
    mov r9, r11
merge_loop:
    cmp rsi, rdi
    jae merge_rest_right
    cmp rbp, r8
    jae merge_rest_left
    call cmp_entries
    test eax, eax
    jg take_right
    movups xmm0, xmmword ptr [rsi]
    movups xmmword ptr [rdx], xmm0
    add rsi, 16
    add rdx, 16
    jmp merge_loop
take_right:
    movups xmm0, xmmword ptr [rbp]
    movups xmmword ptr [rdx], xmm0
    add rbp, 16
    add rdx, 16
    jmp merge_loop
merge_rest_right:
    cmp rbp, r8
    jae pass_loop
    movups xmm0, xmmword ptr [rbp]
    movups xmmword ptr [rdx], xmm0
    add rbp, 16
    add rdx, 16
    jmp merge_rest_right
merge_rest_left:
    cmp rsi, rdi
    jae pass_loop
    movups xmm0, xmmword ptr [rsi]
    movups xmmword ptr [rdx], xmm0
    add rsi, 16
    add rdx, 16
    jmp merge_rest_left
pass_done:
    xchg r13, r14
    shl r15, 1
    jmp width_loop

cmp_entries:
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rbp]
    mov rcx, qword ptr [rsi + 8]
    cmp rcx, qword ptr [rbp + 8]
    cmova rcx, qword ptr [rbp + 8]
cmp_loop:
    test rcx, rcx
    jz cmp_len
    movzx eax, byte ptr [r10]
    cmp al, byte ptr [r11]
    jne cmp_diff
    inc r10
    inc r11
    dec rcx
    jmp cmp_loop
cmp_diff:
    sbb eax, eax
    or eax, 1
    ret
cmp_len:
    mov rax, qword ptr [rsi + 8]
    cmp rax, qword ptr [rbp + 8]
    jne 1f
    xor eax, eax
    ret
1:
    sbb eax, eax
    or eax, 1
    ret

output:
    lea rdi, [rip + outbuf]
    mov r9, r12
out_loop:
    test r9, r9
    jz out_write
    mov rsi, qword ptr [r13]
    mov rcx, qword ptr [r13 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add r13, 16
    dec r9
    jmp out_loop
out_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit:
    mov eax, 231
    xor edi, edi
    syscall
