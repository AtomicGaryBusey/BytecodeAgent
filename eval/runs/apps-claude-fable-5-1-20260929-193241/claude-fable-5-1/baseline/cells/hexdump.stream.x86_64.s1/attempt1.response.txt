.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 69632
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r14, [rip + hexdigits]
    xor r13d, r13d
row_loop:
    mov rax, r12
    sub rax, r13
    jz final
    mov r15, rax
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    mov eax, r13d
    call emit_hex32
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
hex_loop:
    cmp rcx, r15
    jae pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r14 + rdx]
    and eax, 15
    mov al, byte ptr [r14 + rax]
    mov byte ptr [rdi], dl
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp next_hex
pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
next_hex:
    add rdi, 3
    cmp ecx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc ecx
    cmp ecx, 16
    jb hex_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
chr_loop:
    cmp rcx, r15
    jae chr_done
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 3f
    mov eax, 0x2e
3:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp chr_loop
chr_done:
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add rsi, r15
    add r13, r15
    jmp row_loop
final:
    mov eax, r12d
    call emit_hex32
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 231
    xor edi, edi
    syscall

emit_hex32:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r14 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

.section .rodata
hexdigits: .ascii "0123456789abcdef"

.bss
.balign 16
inbuf: .space 69632
.balign 16
outbuf: .space 323600
