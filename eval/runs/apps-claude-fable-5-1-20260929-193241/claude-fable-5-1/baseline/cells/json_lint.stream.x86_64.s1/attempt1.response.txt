.text
.globl _start

_start:
        lea     rbx, [rip + inbuf]
        xor     r13, r13
read_loop:
        mov     edx, 70000
        sub     rdx, r13
        jz      read_done
        lea     rsi, [rbx + r13]
        xor     eax, eax
        xor     edi, edi
        syscall
        test    rax, rax
        jle     read_done
        add     r13, rax
        jmp     read_loop
read_done:
        xor     r12, r12
        call    skip_ws
        call    parse_value
        call    skip_ws
        cmp     r12, r13
        jne     error
        mov     eax, 1
        mov     edi, 1
        lea     rsi, [rip + okmsg]
        mov     edx, 3
        syscall
        mov     eax, 231
        xor     edi, edi
        syscall

# ---------------------------------------------------------------
peek:
        cmp     r12, r13
        jae     1f
        movzx   eax, byte ptr [rbx + r12]
        ret
1:      mov     eax, 256
        ret

skip_ws:
        call    peek
        cmp     eax, 0x20
        je      1f
        cmp     eax, 0x09
        je      1f
        cmp     eax, 0x0A
        je      1f
        cmp     eax, 0x0D
        jne     2f
1:      inc     r12
        jmp     skip_ws
2:      ret

# ---------------------------------------------------------------
parse_value:
        call    peek
        cmp     eax, 0x7B
        je      parse_object
        cmp     eax, 0x5B
        je      parse_array
        cmp     eax, 0x22
        je      parse_string
        cmp     eax, 0x2D
        je      parse_number
        cmp     eax, 0x30
        jb      1f
        cmp     eax, 0x39
        jbe     parse_number
1:      cmp     eax, 0x74
        jne     2f
        lea     rdi, [rip + lit_true]
        mov     ecx, 4
        jmp     parse_literal
2:      cmp     eax, 0x66
        jne     3f
        lea     rdi, [rip + lit_false]
        mov     ecx, 5
        jmp     parse_literal
3:      cmp     eax, 0x6E
        jne     error
        lea     rdi, [rip + lit_null]
        mov     ecx, 4
parse_literal:
        call    peek
        movzx   edx, byte ptr [rdi]
        cmp     eax, edx
        jne     error
        inc     r12
        inc     rdi
        dec     ecx
        jnz     parse_literal
        ret

# ---------------------------------------------------------------
parse_string:
        inc     r12
1:      call    peek
        cmp     eax, 0x22
        je      9f
        cmp     eax, 0x5C
        je      2f
        cmp     eax, 0x20
        jb      error
        cmp     eax, 256
        je      error
        inc     r12
        jmp     1b
2:      inc     r12
        call    peek
        cmp     eax, 0x75
        je      3f
        cmp     eax, 0x22
        je      4f
        cmp     eax, 0x5C
        je      4f
        cmp     eax, 0x2F
        je      4f
        cmp     eax, 0x62
        je      4f
        cmp     eax, 0x66
        je      4f
        cmp     eax, 0x6E
        je      4f
        cmp     eax, 0x72
        je      4f
        cmp     eax, 0x74
        je      4f
        jmp     error
4:      inc     r12
        jmp     1b
3:      inc     r12
        mov     ecx, 4
5:      call    peek
        cmp     eax, 0x30
        jb      error
        cmp     eax, 0x39
        jbe     6f
        cmp     eax, 0x41
        jb      error
        cmp     eax, 0x46
        jbe     6f
        cmp     eax, 0x61
        jb      error
        cmp     eax, 0x66
        ja      error
6:      inc     r12
        dec     ecx
        jnz     5b
        jmp     1b
9:      inc     r12
        ret

# ---------------------------------------------------------------
parse_number:
        call    peek
        cmp     eax, 0x2D
        jne     1f
        inc     r12
        call    peek
1:      cmp     eax, 0x30
        jne     2f
        inc     r12
        jmp     4f
2:      cmp     eax, 0x31
        jb      error
        cmp     eax, 0x39
        ja      error
        inc     r12
        call    skip_digits
4:      call    peek
        cmp     eax, 0x2E
        jne     5f
        inc     r12
        call    need_digits
5:      call    peek
        cmp     eax, 0x65
        je      6f
        cmp     eax, 0x45
        jne     7f
6:      inc     r12
        call    peek
        cmp     eax, 0x2B
        je      8f
        cmp     eax, 0x2D
        jne     9f
8:      inc     r12
9:      call    need_digits
7:      ret

need_digits:
        call    peek
        cmp     eax, 0x30
        jb      error
        cmp     eax, 0x39
        ja      error
        inc     r12
skip_digits:
        call    peek
        cmp     eax, 0x30
        jb      1f
        cmp     eax, 0x39
        ja      1f
        inc     r12
        jmp     skip_digits
1:      ret

# ---------------------------------------------------------------
parse_array:
        inc     r12
        call    skip_ws
        call    peek
        cmp     eax, 0x5D
        jne     1f
        inc     r12
        ret
1:      call    parse_value
        call    skip_ws
        call    peek
        cmp     eax, 0x2C
        jne     2f
        inc     r12
        call    skip_ws
        jmp     1b
2:      cmp     eax, 0x5D
        jne     error
        inc     r12
        ret

# ---------------------------------------------------------------
parse_object:
        inc     r12
        call    skip_ws
        call    peek
        cmp     eax, 0x7D
        jne     1f
        inc     r12
        ret
1:      cmp     eax, 0x22
        jne     error
        call    parse_string
        call    skip_ws
        call    peek
        cmp     eax, 0x3A
        jne     error
        inc     r12
        call    skip_ws
        call    parse_value
        call    skip_ws
        call    peek
        cmp     eax, 0x2C
        jne     2f
        inc     r12
        call    skip_ws
        call    peek
        jmp     1b
2:      cmp     eax, 0x7D
        jne     error
        inc     r12
        ret

# ---------------------------------------------------------------
error:
        lea     rdi, [rip + outbuf + 31]
        mov     byte ptr [rdi], 10
        mov     rax, r12
        mov     ecx, 10
1:      xor     edx, edx
        div     rcx
        add     dl, 0x30
        dec     rdi
        mov     byte ptr [rdi], dl
        test    rax, rax
        jnz     1b
        lea     r8, [rip + errmsg]
        mov     ecx, 9
2:      dec     rdi
        mov     al, byte ptr [r8 + rcx - 1]
        mov     byte ptr [rdi], al
        dec     ecx
        jnz     2b
        mov     rsi, rdi
        lea     rdx, [rip + outbuf + 32]
        sub     rdx, rsi
        mov     eax, 1
        mov     edi, 1
        syscall
        mov     eax, 231
        mov     edi, 1
        syscall
        jmp     error

# ---------------------------------------------------------------
.section .rodata
okmsg:      .ascii "ok\n"
errmsg:     .ascii "error at "
lit_true:   .ascii "true"
lit_false:  .ascii "false"
lit_null:   .ascii "null"

.bss
.balign 16
outbuf:     .space 32
inbuf:      .space 70016
