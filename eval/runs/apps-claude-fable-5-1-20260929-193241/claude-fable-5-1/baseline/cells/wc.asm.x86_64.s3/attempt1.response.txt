.text
.globl _start
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in_word flag

read_loop:
    xor eax, eax            # sys_read
    xor edi, edi            # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea rdx, [rsi + rax]

scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je ws
    lea ecx, [rax - 9]
    cmp ecx, 4
    jbe ws
    # word byte
    test r15d, r15d
    jnz 2f
    inc r13
    mov r15d, 1
2:
    cmp rsi, rdx
    jb scan
    jmp read_loop

ws:
    xor r15d, r15d
    cmp rsi, rdx
    jb scan
    jmp read_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call fmt
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call fmt
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call fmt
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1              # sys_write
    syscall
    mov eax, 231            # exit_group
    xor edi, edi
    syscall

# fmt: rax = value, rdi = output pointer; writes decimal, advances rdi
fmt:
    sub rsp, 32
    lea rsi, [rsp + 31]
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rcx, [rsp + 31]
    sub rcx, rsi
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz 2b
    add rsp, 32
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
