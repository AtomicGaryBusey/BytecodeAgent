// sort_lines: static AArch64 Linux executable, no libc
    .bss
    .balign 16
inbuf:  .space 65536
lines:  .space 131072
tmp:    .space 131072
outbuf: .space 131072

    .text
    .globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x2, #65536
    subs    x2, x2, x20
    b.le    read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x21, lines
    add     x21, x21, :lo12:lines
    mov     x22, #0                 // number of lines
    mov     x23, #0                 // start of current line
    mov     x24, #0                 // scan index
split_loop:
    cmp     x24, x20
    b.ge    split_end
    ldrb    w0, [x19, x24]
    cmp     w0, #10
    b.ne    split_next
    add     x1, x19, x23
    sub     x2, x24, x23
    bl      add_line
    add     x23, x24, #1
split_next:
    add     x24, x24, #1
    b       split_loop
split_end:
    cmp     x23, x20
    b.ge    sort_begin
    add     x1, x19, x23
    sub     x2, x20, x23
    bl      add_line

sort_begin:
    adrp    x25, tmp
    add     x25, x25, :lo12:tmp     // x21 = src, x25 = dst
    mov     x26, #1                 // run width
sort_pass:
    cmp     x26, x22
    b.ge    sort_done
    mov     x27, #0                 // i
run_loop:
    cmp     x27, x22
    b.ge    pass_end
    add     x9, x27, x26
    cmp     x9, x22
    csel    x9, x9, x22, lt         // mid
    add     x10, x9, x26
    cmp     x10, x22
    csel    x10, x10, x22, lt       // right
    mov     x11, x27                // a
    mov     x12, x9                 // b
    mov     x13, x27                // k
merge_loop:
    cmp     x11, x9
    b.ge    merge_rest_b
    cmp     x12, x10
    b.ge    merge_rest_a
    lsl     x14, x11, #4
    add     x14, x21, x14
    ldp     x0, x1, [x14]
    lsl     x15, x12, #4
    add     x15, x21, x15
    ldp     x2, x3, [x15]
    bl      cmp_lines
    cmp     x0, #0
    b.gt    take_b
    ldp     x0, x1, [x14]
    add     x11, x11, #1
    b       store_k
take_b:
    ldp     x0, x1, [x15]
    add     x12, x12, #1
store_k:
    lsl     x16, x13, #4
    add     x16, x25, x16
    stp     x0, x1, [x16]
    add     x13, x13, #1
    b       merge_loop
merge_rest_a:
    cmp     x11, x9
    b.ge    run_next
    lsl     x14, x11, #4
    add     x14, x21, x14
    ldp     x0, x1, [x14]
    lsl     x16, x13, #4
    add     x16, x25, x16
    stp     x0, x1, [x16]
    add     x11, x11, #1
    add     x13, x13, #1
    b       merge_rest_a
merge_rest_b:
    cmp     x12, x10
    b.ge    run_next
    lsl     x15, x12, #4
    add     x15, x21, x15
    ldp     x0, x1, [x15]
    lsl     x16, x13, #4
    add     x16, x25, x16
    stp     x0, x1, [x16]
    add     x12, x12, #1
    add     x13, x13, #1
    b       merge_rest_b
run_next:
    add     x27, x27, x26, lsl #1
    b       run_loop
pass_end:
    mov     x0, x21
    mov     x21, x25
    mov     x25, x0
    lsl     x26, x26, #1
    b       sort_pass

sort_done:
    adrp    x28, outbuf
    add     x28, x28, :lo12:outbuf
    mov     x27, #0                 // output length
    mov     x26, #0                 // line index
out_loop:
    cmp     x26, x22
    b.ge    out_flush
    lsl     x0, x26, #4
    add     x0, x21, x0
    ldp     x1, x2, [x0]
    add     x3, x28, x27
copy_loop:
    cbz     x2, copy_done
    ldrb    w4, [x1], #1
    strb    w4, [x3], #1
    sub     x2, x2, #1
    add     x27, x27, #1
    b       copy_loop
copy_done:
    mov     w4, #10
    strb    w4, [x3]
    add     x27, x27, #1
    add     x26, x26, #1
    b       out_loop
out_flush:
    mov     x1, x28
    mov     x2, x27
write_loop:
    cbz     x2, do_exit
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    do_exit
    add     x1, x1, x0
    sub     x2, x2, x0
    b       write_loop
do_exit:
    mov     x0, #0
    mov     x8, #94
    svc     #0
    b       do_exit

// add_line: x1 = ptr, x2 = len; appends to table at x21, count in x22
add_line:
    mov     x3, #8192
    cmp     x22, x3
    b.ge    1f
    lsl     x3, x22, #4
    add     x3, x21, x3
    stp     x1, x2, [x3]
    add     x22, x22, #1
1:
    ret

// cmp_lines: (x0,x1) vs (x2,x3) -> x0 <0, 0, >0 (memcmp-like, shorter prefix first)
cmp_lines:
    cmp     x1, x3
    csel    x4, x1, x3, lo
1:
    cbz     x4, 2f
    ldrb    w5, [x0], #1
    ldrb    w6, [x2], #1
    subs    x5, x5, x6
    b.ne    3f
    sub     x4, x4, #1
    b       1b
2:
    sub     x0, x1, x3
    ret
3:
    mov     x0, x5
    ret
