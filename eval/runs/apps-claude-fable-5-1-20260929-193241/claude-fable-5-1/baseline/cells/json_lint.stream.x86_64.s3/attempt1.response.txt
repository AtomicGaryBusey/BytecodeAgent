.text
.globl _start
_start:
    lea rsi, [rip + buf]
    mov r12, rsi
read_loop:
    xor eax, eax
    xor edi, edi
    lea rdx, [rip + buf_end]
    sub rdx, rsi
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    jmp read_loop
read_done:
    mov rdi, rsi
    mov rsi, r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, rdi
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall

error:
    lea rax, [rip + buf]
    sub rsi, rax
    mov rax, rsi
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:  dec rdi
    xor edx, edx
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    push rdi
    lea rsi, [rip + err_prefix]
    mov ecx, 9
    rep movsb
    pop rsi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall

skip_ws:
    cmp rsi, rdi
    jae 2f
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    jne 2f
1:  inc rsi
    jmp skip_ws
2:  ret

parse_value:
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe parse_number
    cmp al, 't'
    jne 1f
    lea rdx, [rip + lit_t]
    jmp expect_lit
1:  cmp al, 'f'
    jne 1f
    lea rdx, [rip + lit_f]
    jmp expect_lit
1:  cmp al, 'n'
    jne error
    lea rdx, [rip + lit_n]
    jmp expect_lit

expect_lit:
    inc rsi
1:  movzx ecx, byte ptr [rdx]
    test cl, cl
    jz 2f
    cmp rsi, rdi
    jae error
    cmp byte ptr [rsi], cl
    jne error
    inc rsi
    inc rdx
    jmp 1b
2:  ret

parse_string:
    inc rsi
1:  cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je 9f
    cmp al, 0x5C
    je 3f
    cmp al, 0x20
    jb error
    inc rsi
    jmp 1b
3:  inc rsi
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 'u'
    je 5f
    cmp al, 0x22
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, 0x2F
    je 4f
    cmp al, 'b'
    je 4f
    cmp al, 'f'
    je 4f
    cmp al, 'n'
    je 4f
    cmp al, 'r'
    je 4f
    cmp al, 't'
    jne error
4:  inc rsi
    jmp 1b
5:  inc rsi
    mov ecx, 4
6:  cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    mov edx, eax
    sub dl, '0'
    cmp dl, 9
    jbe 7f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
7:  inc rsi
    dec ecx
    jnz 6b
    jmp 1b
9:  inc rsi
    ret

parse_number:
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne 1f
    inc rsi
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
1:  cmp al, '0'
    je 2f
    sub al, '1'
    cmp al, 8
    ja error
    inc rsi
    call skip_digits
    jmp 3f
2:  inc rsi
3:  cmp rsi, rdi
    jae 8f
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne 4f
    inc rsi
    call need_digits
    cmp rsi, rdi
    jae 8f
    movzx eax, byte ptr [rsi]
4:  or al, 0x20
    cmp al, 'e'
    jne 8f
    inc rsi
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:  inc rsi
6:  call need_digits
8:  ret

need_digits:
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
    inc rsi
skip_digits:
    cmp rsi, rdi
    jae 1f
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja 1f
    inc rsi
    jmp skip_digits
1:  ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, rdi
    jae error
    cmp byte ptr [rsi], ']'
    je 3f
1:  call parse_value
    call skip_ws
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp 1b
2:  cmp al, ']'
    jne error
3:  inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, rdi
    jae error
    cmp byte ptr [rsi], '}'
    je 3f
1:  cmp byte ptr [rsi], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp rsi, rdi
    jae error
    cmp byte ptr [rsi], ':'
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, rdi
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    cmp rsi, rdi
    jae error
    jmp 1b
2:  cmp al, '}'
    jne error
3:  inc rsi
    ret

.section .rodata
ok_msg: .ascii "ok\n"
err_prefix: .ascii "error at "
lit_t: .asciz "rue"
lit_f: .asciz "alse"
lit_n: .asciz "ull"

.bss
.balign 16
buf: .space 65552
buf_end:
outbuf: .space 64
