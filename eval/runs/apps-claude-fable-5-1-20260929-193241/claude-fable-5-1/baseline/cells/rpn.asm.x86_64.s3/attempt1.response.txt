.text
.globl _start

.equ INBUFSZ, 65552
.equ OUTSZ, 524288

_start:
    xor r15d, r15d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov edx, INBUFSZ
    sub rdx, r15
    jbe read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    add r15, rsi
    lea r14, [rip + outbuf]
    mov rdi, r14
    lea rbp, [rip + vstack]

main_loop:
    cmp rsi, r15
    jae finish
    xor r12d, r12d
token_loop:
    cmp rsi, r15
    jae end_of_line
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je end_of_line_nl
    cmp al, 32
    je skip_sep
    cmp al, 9
    jne token
skip_sep:
    inc rsi
    jmp token_loop

token:
    xor ecx, ecx
    lea edx, [rax - 48]
    cmp edx, 9
    jbe parse_num
    cmp al, 45
    jne check_op
    lea rdx, [rsi + 1]
    cmp rdx, r15
    jae op_token
    movzx edx, byte ptr [rdx]
    sub edx, 48
    cmp edx, 9
    ja op_token
    inc rsi
    mov ecx, 1
    jmp parse_num
check_op:
    cmp al, 43
    je op_token
    cmp al, 42
    je op_token
    cmp al, 47
    je op_token
    cmp al, 37
    jne error

op_token:
    mov r13d, eax
    inc rsi
    cmp rsi, r15
    jae 1f
    movzx edx, byte ptr [rsi]
    cmp dl, 32
    je 1f
    cmp dl, 9
    je 1f
    cmp dl, 10
    jne error
1:
    cmp r12, 2
    jb error
    mov rbx, qword ptr [rbp + r12*8 - 8]
    mov rax, qword ptr [rbp + r12*8 - 16]
    dec r12
    cmp r13b, 43
    je op_add
    cmp r13b, 45
    je op_sub
    cmp r13b, 42
    je op_mul
    test rbx, rbx
    jz error
    cmp rbx, -1
    jne 2f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error
2:
    cqo
    idiv rbx
    cmp r13b, 47
    je store
    mov rax, rdx
    jmp store
op_add:
    add rax, rbx
    jmp store
op_sub:
    sub rax, rbx
    jmp store
op_mul:
    imul rax, rbx
store:
    mov qword ptr [rbp + r12*8 - 8], rax
    jmp token_loop

parse_num:
    xor eax, eax
    xor r8d, r8d
digit_loop:
    cmp rsi, r15
    jae num_end
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja num_end
    inc r8
    cmp r8, 18
    ja error
    lea rax, [rax + rax*4]
    lea rax, [rdx + rax*2]
    inc rsi
    jmp digit_loop
num_end:
    cmp rsi, r15
    jae 1f
    movzx edx, byte ptr [rsi]
    cmp dl, 32
    je 1f
    cmp dl, 9
    je 1f
    cmp dl, 10
    jne error
1:
    test ecx, ecx
    jz 2f
    neg rax
2:
    mov qword ptr [rbp + r12*8], rax
    inc r12
    jmp token_loop

end_of_line_nl:
    inc rsi
end_of_line:
    cmp r12, 1
    jne error_out
    mov rax, qword ptr [rbp]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], 45
    inc rdi
    neg rax
1:
    lea r9, [rip + numtmp + 32]
    mov r10, 10
2:
    xor edx, edx
    div r10
    add dl, 48
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz 2b
    lea rcx, [rip + numtmp + 32]
    sub rcx, r9
3:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    dec rcx
    jnz 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_done

error:
    cmp rsi, r15
    jae error_out
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
error_out:
    mov eax, 0x6f727265
    mov dword ptr [rdi], eax
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
line_done:
    mov rax, rdi
    sub rax, r14
    cmp rax, OUTSZ - 64
    jb main_loop
    call flush
    jmp main_loop

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp finish

flush:
    push rsi
    mov rsi, r14
    mov rdx, rdi
    sub rdx, r14
1:
    test rdx, rdx
    jz 2f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jmp 1b
2:
    mov rdi, r14
    pop rsi
    ret

.bss
.balign 16
inbuf:
    .space INBUFSZ
.balign 16
outbuf:
    .space OUTSZ
.balign 16
vstack:
    .space 262144
.balign 16
numtmp:
    .space 32
