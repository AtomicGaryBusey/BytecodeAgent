.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // in_word flag
    adrp x23, buf
    add x23, x23, :lo12:buf
read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #0x20
    b.eq ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls ws
    cbnz x22, next
    mov x22, #1
    add x20, x20, #1
    b next
ws:
    mov x22, #0
next:
    cmp x1, x2
    b.lo byte_loop
    b read_loop
done:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #64
    mov x24, x1
    mov w3, #10
    strb w3, [x1, #-1]!
    mov x0, x21
    bl itoa
    mov w3, #32
    strb w3, [x1, #-1]!
    mov x0, x20
    bl itoa
    mov w3, #32
    strb w3, [x1, #-1]!
    mov x0, x19
    bl itoa
    sub x2, x24, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #94
    svc #0
    b .

// itoa: x0 = value, x1 = end pointer (pre-decremented); writes digits backwards
itoa:
    mov x5, #10
1:
    udiv x4, x0, x5
    msub x6, x4, x5, x0
    add w6, w6, #48
    strb w6, [x1, #-1]!
    mov x0, x4
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 64
