.text
.globl _start
_start:
        mov     x19, #0                 // lines
        mov     x20, #0                 // words
        mov     x21, #0                 // bytes
        mov     x22, #0                 // in_word flag
read_loop:
        mov     x0, #0
        adrp    x1, buf
        add     x1, x1, :lo12:buf
        mov     x2, #65536
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    done
        add     x21, x21, x0
        adrp    x3, buf
        add     x3, x3, :lo12:buf
        add     x4, x3, x0
count_loop:
        ldrb    w5, [x3], #1
        cmp     w5, #10
        cinc    x19, x19, eq
        cmp     w5, #0x20
        b.eq    ws
        sub     w6, w5, #9
        cmp     w6, #4
        b.ls    ws
        cbnz    x22, next
        mov     x22, #1
        add     x20, x20, #1
        b       next
ws:
        mov     x22, #0
next:
        cmp     x3, x4
        b.lo    count_loop
        b       read_loop
done:
        adrp    x1, obuf
        add     x1, x1, :lo12:obuf
        add     x1, x1, #64
        mov     w5, #10
        strb    w5, [x1, #-1]!
        mov     x0, x21
        bl      put_num
        mov     w5, #0x20
        strb    w5, [x1, #-1]!
        mov     x0, x20
        bl      put_num
        mov     w5, #0x20
        strb    w5, [x1, #-1]!
        mov     x0, x19
        bl      put_num
        adrp    x2, obuf
        add     x2, x2, :lo12:obuf
        add     x2, x2, #64
        sub     x2, x2, x1
        mov     x0, #1
        mov     x8, #64
        svc     #0
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       .

// x0 = value, x1 = pointer (pre-decremented); clobbers x0, x6, x7, x9
put_num:
        mov     x9, #10
1:
        udiv    x6, x0, x9
        msub    x7, x6, x9, x0
        add     w7, w7, #48
        strb    w7, [x1, #-1]!
        mov     x0, x6
        cbnz    x0, 1b
        ret

.bss
.balign 16
buf:
        .space  65536
obuf:
        .space  64
