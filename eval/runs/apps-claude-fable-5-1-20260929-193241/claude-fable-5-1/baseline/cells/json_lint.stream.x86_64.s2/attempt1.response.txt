.text
.globl _start

.equ BUFSZ, 65600

_start:
    lea rbx, [rip + buf]
    xor r13d, r13d
read_loop:
    mov edx, BUFSZ
    sub edx, r13d
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    mov rsi, rbx
    lea r12, [rbx + r13]
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp error

# peek: eax = current byte, or -1 at EOF
peek:
    cmp rsi, r12
    jae 1f
    movzx eax, byte ptr [rsi]
    ret
1:
    mov eax, -1
    ret

skip_ws:
    call peek
    cmp eax, 0x20
    je 1f
    cmp eax, 0x09
    je 1f
    cmp eax, 0x0a
    je 1f
    cmp eax, 0x0d
    jne 2f
1:
    inc rsi
    jmp skip_ws
2:
    ret

parse_value:
    call peek
    cmp eax, 0x7b
    je parse_object
    cmp eax, 0x5b
    je parse_array
    cmp eax, 0x22
    je parse_string
    cmp eax, 0x74
    je parse_true
    cmp eax, 0x66
    je parse_false
    cmp eax, 0x6e
    je parse_null
    cmp eax, 0x2d
    je parse_number
    sub eax, 0x30
    cmp eax, 9
    jbe parse_number
    jmp error

parse_true:
    lea rdi, [rip + lit_true]
    jmp match_lit
parse_false:
    lea rdi, [rip + lit_false]
    jmp match_lit
parse_null:
    lea rdi, [rip + lit_null]
match_lit:
    movzx ecx, byte ptr [rdi]
    test ecx, ecx
    jz 1f
    call peek
    cmp eax, ecx
    jne error
    inc rsi
    inc rdi
    jmp match_lit
1:
    ret

parse_string:
    inc rsi
str_loop:
    call peek
    cmp eax, 0x22
    je str_end
    cmp eax, 0x5c
    je str_esc
    cmp eax, 0x20
    jb error
    cmp eax, 0xff
    ja error
    inc rsi
    jmp str_loop
str_end:
    inc rsi
    ret
str_esc:
    inc rsi
    call peek
    cmp eax, 0x75
    je str_u
    cmp eax, 0x22
    je esc_ok
    cmp eax, 0x5c
    je esc_ok
    cmp eax, 0x2f
    je esc_ok
    cmp eax, 0x62
    je esc_ok
    cmp eax, 0x66
    je esc_ok
    cmp eax, 0x6e
    je esc_ok
    cmp eax, 0x72
    je esc_ok
    cmp eax, 0x74
    je esc_ok
    jmp error
esc_ok:
    inc rsi
    jmp str_loop
str_u:
    inc rsi
    mov ecx, 4
hex_loop:
    call peek
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    jbe hex_ok
    mov edx, eax
    or edx, 0x20
    sub edx, 0x61
    cmp edx, 5
    jbe hex_ok
    jmp error
hex_ok:
    inc rsi
    dec ecx
    jnz hex_loop
    jmp str_loop

parse_number:
    call peek
    cmp eax, 0x2d
    jne 1f
    inc rsi
    call peek
1:
    cmp eax, 0x30
    jne 2f
    inc rsi
    jmp num_frac
2:
    sub eax, 0x31
    cmp eax, 8
    ja error
    inc rsi
num_digits1:
    call peek
    sub eax, 0x30
    cmp eax, 9
    ja num_frac
    inc rsi
    jmp num_digits1
num_frac:
    call peek
    cmp eax, 0x2e
    jne num_exp
    inc rsi
    call peek
    sub eax, 0x30
    cmp eax, 9
    ja error
    inc rsi
num_digits2:
    call peek
    sub eax, 0x30
    cmp eax, 9
    ja num_exp
    inc rsi
    jmp num_digits2
num_exp:
    call peek
    or eax, 0x20
    cmp eax, 0x65
    jne num_done
    inc rsi
    call peek
    cmp eax, 0x2b
    je 3f
    cmp eax, 0x2d
    jne 4f
3:
    inc rsi
    call peek
4:
    sub eax, 0x30
    cmp eax, 9
    ja error
    inc rsi
num_digits3:
    call peek
    sub eax, 0x30
    cmp eax, 9
    ja num_done
    inc rsi
    jmp num_digits3
num_done:
    ret

parse_array:
    inc rsi
    call skip_ws
    call peek
    cmp eax, 0x5d
    jne arr_elem
    inc rsi
    ret
arr_elem:
    call parse_value
    call skip_ws
    call peek
    cmp eax, 0x5d
    je arr_end
    cmp eax, 0x2c
    jne error
    inc rsi
    call skip_ws
    jmp arr_elem
arr_end:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    call peek
    cmp eax, 0x7d
    jne obj_member
    inc rsi
    ret
obj_member:
    call peek
    cmp eax, 0x22
    jne error
    call parse_string
    call skip_ws
    call peek
    cmp eax, 0x3a
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, 0x7d
    je obj_end
    cmp eax, 0x2c
    jne error
    inc rsi
    call skip_ws
    jmp obj_member
obj_end:
    inc rsi
    ret

error:
    mov rax, rsi
    sub rax, rbx
    lea rdi, [rip + numbuf + 63]
    mov byte ptr [rdi], 0x0a
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    mov rax, qword ptr [rip + errpre]
    mov qword ptr [rdi - 9], rax
    mov al, byte ptr [rip + errpre + 8]
    mov byte ptr [rdi - 1], al
    sub rdi, 9
    mov rsi, rdi
    lea rdx, [rip + numbuf + 64]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall
    jmp error

.section .rodata
okmsg:      .ascii "ok\n"
errpre:     .ascii "error at "
lit_true:   .asciz "true"
lit_false:  .asciz "false"
lit_null:   .asciz "null"

.bss
.balign 16
buf:        .space BUFSZ
numbuf:     .space 64
