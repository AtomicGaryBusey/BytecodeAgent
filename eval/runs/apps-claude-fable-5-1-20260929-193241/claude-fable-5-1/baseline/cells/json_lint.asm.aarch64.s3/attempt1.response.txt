.text
.globl _start

_start:
    adrp    x20, buf
    add     x20, x20, :lo12:buf
    mov     x21, #0                 // bytes read so far
read_loop:
    mov     x2, #65536
    subs    x2, x2, x21             // remaining capacity
    b.le    read_done
    mov     x0, #0
    add     x1, x20, x21
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x21, x21, x0
    b       read_loop
read_done:
    add     x22, x20, x21           // end pointer (buf[len] == 0 sentinel)
    mov     x19, x20                // cursor
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x19, x22
    b.ne    error
    mov     x0, #1
    adrp    x1, okmsg
    add     x1, x1, :lo12:okmsg
    mov     x2, #3
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
skip_ws:
1:  ldrb    w0, [x19]
    cmp     w0, #0x20
    b.eq    2f
    cmp     w0, #0x09
    b.eq    2f
    cmp     w0, #0x0a
    b.eq    2f
    cmp     w0, #0x0d
    b.ne    3f
2:  add     x19, x19, #1
    b       1b
3:  ret

// ---------------------------------------------------------------
parse_value:
    stp     x29, x30, [sp, #-16]!
    ldrb    w0, [x19]
    cmp     w0, #0x7b               // {
    b.eq    pv_obj
    cmp     w0, #0x5b               // [
    b.eq    pv_arr
    cmp     w0, #0x22               // "
    b.eq    pv_str
    cmp     w0, #0x2d               // -
    b.eq    pv_num
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pv_num
    cmp     w0, #0x74               // t
    b.eq    pv_true
    cmp     w0, #0x66               // f
    b.eq    pv_false
    cmp     w0, #0x6e               // n
    b.eq    pv_null
    b       error
pv_obj:
    bl      parse_object
    b       pv_ret
pv_arr:
    bl      parse_array
    b       pv_ret
pv_str:
    bl      parse_string
    b       pv_ret
pv_num:
    bl      parse_number
    b       pv_ret
pv_true:
    adrp    x1, lit_true
    add     x1, x1, :lo12:lit_true
    bl      match_lit
    b       pv_ret
pv_false:
    adrp    x1, lit_false
    add     x1, x1, :lo12:lit_false
    bl      match_lit
    b       pv_ret
pv_null:
    adrp    x1, lit_null
    add     x1, x1, :lo12:lit_null
    bl      match_lit
pv_ret:
    ldp     x29, x30, [sp], #16
    ret

// x1 = asciz literal
match_lit:
1:  ldrb    w2, [x1], #1
    cbz     w2, 2f
    ldrb    w0, [x19]
    cmp     w0, w2
    b.ne    error
    add     x19, x19, #1
    b       1b
2:  ret

// ---------------------------------------------------------------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x19, x19, #1            // {
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x7d               // }
    b.eq    po_end
po_member:
    ldrb    w0, [x19]
    cmp     w0, #0x22
    b.ne    error
    bl      parse_string
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x3a               // :
    b.ne    error
    add     x19, x19, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x2c               // ,
    b.eq    po_comma
    cmp     w0, #0x7d
    b.ne    error
po_end:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret
po_comma:
    add     x19, x19, #1
    bl      skip_ws
    b       po_member

// ---------------------------------------------------------------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x19, x19, #1            // [
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x5d               // ]
    b.eq    pa_end
pa_elem:
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x2c
    b.eq    pa_comma
    cmp     w0, #0x5d
    b.ne    error
pa_end:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret
pa_comma:
    add     x19, x19, #1
    bl      skip_ws
    b       pa_elem

// ---------------------------------------------------------------
parse_string:
    add     x19, x19, #1            // opening "
ps_loop:
    ldrb    w0, [x19]
    cmp     w0, #0x22
    b.eq    ps_end
    cmp     w0, #0x20
    b.lo    error
    cmp     w0, #0x5c               // backslash
    b.ne    ps_plain
    add     x19, x19, #1
    ldrb    w0, [x19]
    cmp     w0, #0x75               // u
    b.eq    ps_u
    cmp     w0, #0x22
    b.eq    ps_plain
    cmp     w0, #0x5c
    b.eq    ps_plain
    cmp     w0, #0x2f               // /
    b.eq    ps_plain
    cmp     w0, #0x62               // b
    b.eq    ps_plain
    cmp     w0, #0x66               // f
    b.eq    ps_plain
    cmp     w0, #0x6e               // n
    b.eq    ps_plain
    cmp     w0, #0x72               // r
    b.eq    ps_plain
    cmp     w0, #0x74               // t
    b.eq    ps_plain
    b       error
ps_plain:
    add     x19, x19, #1
    b       ps_loop
ps_u:
    add     x19, x19, #1
    mov     w3, #4
ps_hex:
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    ps_hexok
    orr     w1, w0, #0x20
    sub     w1, w1, #0x61
    cmp     w1, #5
    b.hi    error
ps_hexok:
    add     x19, x19, #1
    subs    w3, w3, #1
    b.ne    ps_hex
    b       ps_loop
ps_end:
    add     x19, x19, #1
    ret

// ---------------------------------------------------------------
skip_digits:
1:  ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    2f
    add     x19, x19, #1
    b       1b
2:  ret

parse_number:
    stp     x29, x30, [sp, #-16]!
    ldrb    w0, [x19]
    cmp     w0, #0x2d               // -
    b.ne    1f
    add     x19, x19, #1
    ldrb    w0, [x19]
1:  cmp     w0, #0x30               // 0
    b.eq    2f
    sub     w1, w0, #0x31
    cmp     w1, #8
    b.hi    error
    add     x19, x19, #1
    bl      skip_digits
    b       3f
2:  add     x19, x19, #1
3:  ldrb    w0, [x19]
    cmp     w0, #0x2e               // .
    b.ne    4f
    add     x19, x19, #1
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    error
    bl      skip_digits
4:  ldrb    w0, [x19]
    orr     w1, w0, #0x20
    cmp     w1, #0x65               // e / E
    b.ne    5f
    add     x19, x19, #1
    ldrb    w0, [x19]
    cmp     w0, #0x2b               // +
    b.eq    6f
    cmp     w0, #0x2d               // -
    b.ne    7f
6:  add     x19, x19, #1
7:  ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    error
    bl      skip_digits
5:  ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
error:
    sub     x0, x19, x20            // offset
    adrp    x23, numbuf
    add     x23, x23, :lo12:numbuf
    add     x23, x23, #31
    mov     w2, #0x0a
    strb    w2, [x23]
    mov     x3, #10
1:  udiv    x4, x0, x3
    msub    x5, x4, x3, x0
    add     w5, w5, #0x30
    sub     x23, x23, #1
    strb    w5, [x23]
    mov     x0, x4
    cbnz    x0, 1b
    adrp    x24, numbuf
    add     x24, x24, :lo12:numbuf
    add     x24, x24, #32
    sub     x24, x24, x23           // length
    mov     x0, #1
    adrp    x1, errmsg
    add     x1, x1, :lo12:errmsg
    mov     x2, #9
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x1, x23
    mov     x2, x24
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
.section .rodata
okmsg:      .ascii  "ok\n"
errmsg:     .ascii  "error at "
lit_true:   .asciz  "true"
lit_false:  .asciz  "false"
lit_null:   .asciz  "null"

.bss
.balign 16
buf:        .space  65552
numbuf:     .space  32
