.text
.globl _start
_start:
        xor     ebx, ebx                # total bytes read
.Lread:
        mov     edx, 65536
        sub     edx, ebx
        jbe     .Ldone_read
        xor     eax, eax
        xor     edi, edi
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        syscall
        test    rax, rax
        jle     .Ldone_read
        add     rbx, rax
        jmp     .Lread
.Ldone_read:
        lea     rsi, [rip + inbuf]      # current line start
        lea     r8, [rsi + rbx]         # end of input
        lea     rdi, [rip + outbuf]     # output cursor
.Lline:
        cmp     rsi, r8
        jae     .Lwrite
        mov     rcx, rsi
.Lfind:
        cmp     rcx, r8
        jae     .Lfound
        cmp     byte ptr [rcx], 0x0A
        je      .Lfound
        inc     rcx
        jmp     .Lfind
.Lfound:
        mov     rdx, rcx
.Lrev:
        cmp     rdx, rsi
        jbe     .Lrev_done
        dec     rdx
        mov     al, byte ptr [rdx]
        mov     byte ptr [rdi], al
        inc     rdi
        jmp     .Lrev
.Lrev_done:
        mov     byte ptr [rdi], 0x0A
        inc     rdi
        lea     rsi, [rcx + 1]
        jmp     .Lline
.Lwrite:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi                # total output length
.Lwloop:
        test    rdx, rdx
        jle     .Lexit
        mov     eax, 1
        mov     edi, 1
        syscall
        test    rax, rax
        jle     .Lexit
        add     rsi, rax
        sub     rdx, rax
        jmp     .Lwloop
.Lexit:
        mov     eax, 231
        xor     edi, edi
        syscall
        mov     eax, 60
        xor     edi, edi
        syscall
        jmp     .Lexit

.bss
        .balign 16
inbuf:
        .space  65536 + 16
        .balign 16
outbuf:
        .space  65536 + 32
