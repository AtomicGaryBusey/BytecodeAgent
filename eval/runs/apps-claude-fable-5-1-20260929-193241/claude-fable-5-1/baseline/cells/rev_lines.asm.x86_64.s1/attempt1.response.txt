.text
.globl _start
_start:
        lea rbx, [rip + inbuf]
        xor r12d, r12d
read_loop:
        mov edx, 65552
        sub rdx, r12
        jbe read_done
        xor eax, eax
        xor edi, edi
        lea rsi, [rbx + r12]
        syscall
        test rax, rax
        jle read_done
        add r12, rax
        jmp read_loop
read_done:
        lea rbp, [rip + outbuf]
        xor r13d, r13d
        xor r14d, r14d
        xor r15d, r15d
scan_loop:
        cmp r14, r12
        jae scan_end
        cmp byte ptr [rbx + r14], 10
        jne scan_next
        mov rdi, r13
        mov rsi, r14
        call emit_rev
        lea r13, [r14 + 1]
scan_next:
        inc r14
        jmp scan_loop
scan_end:
        cmp r13, r12
        jae do_write
        mov rdi, r13
        mov rsi, r12
        call emit_rev
do_write:
        xor r13d, r13d
write_loop:
        cmp r13, r15
        jae done
        mov eax, 1
        mov edi, 1
        lea rsi, [rbp + r13]
        mov rdx, r15
        sub rdx, r13
        syscall
        test rax, rax
        jle done
        add r13, rax
        jmp write_loop
done:
        mov eax, 231
        xor edi, edi
        syscall
        jmp done

# emit_rev: rdi = start index, rsi = end index (exclusive); appends reversed bytes + newline to outbuf
emit_rev:
        mov rcx, rsi
emit_loop:
        cmp rcx, rdi
        jbe emit_nl
        dec rcx
        movzx eax, byte ptr [rbx + rcx]
        mov byte ptr [rbp + r15], al
        inc r15
        jmp emit_loop
emit_nl:
        mov byte ptr [rbp + r15], 10
        inc r15
        ret

.bss
.balign 16
inbuf:
        .space 65568
.balign 16
outbuf:
        .space 65600
