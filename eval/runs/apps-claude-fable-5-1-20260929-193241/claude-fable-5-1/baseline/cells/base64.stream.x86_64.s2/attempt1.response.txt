.text
.globl _start
_start:
        xor     r12d, r12d              # bytes read so far
read_loop:
        mov     edx, 65536
        sub     rdx, r12
        jbe     read_done
        lea     rsi, [rip + inbuf]
        add     rsi, r12
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        syscall
        test    rax, rax
        jle     read_done
        add     r12, rax
        jmp     read_loop

read_done:
        lea     rsi, [rip + inbuf]
        lea     rdi, [rip + outbuf]
        lea     r9, [rip + alphabet]
        mov     rcx, r12
        xor     r8d, r8d                # groups on current line

full_loop:
        cmp     rcx, 3
        jb      tail
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        movzx   edx, byte ptr [rsi + 1]
        shl     edx, 8
        or      eax, edx
        movzx   edx, byte ptr [rsi + 2]
        or      eax, edx
        call    emit4
        call    linecheck
        add     rsi, 3
        sub     rcx, 3
        jmp     full_loop

tail:
        test    rcx, rcx
        jz      finish
        cmp     rcx, 1
        je      one_byte
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        movzx   edx, byte ptr [rsi + 1]
        shl     edx, 8
        or      eax, edx
        call    emit4
        mov     byte ptr [rdi - 1], '='
        call    linecheck
        jmp     finish
one_byte:
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        call    emit4
        mov     byte ptr [rdi - 1], '='
        mov     byte ptr [rdi - 2], '='
        call    linecheck

finish:
        test    r8d, r8d
        jz      do_write
        mov     byte ptr [rdi], 10
        inc     rdi
do_write:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi                # length
write_loop:
        test    rdx, rdx
        jz      done
        mov     eax, 1
        mov     edi, 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
done:
        mov     eax, 231
        xor     edi, edi
        syscall
        jmp     done

# emit4: eax = 24-bit value, writes 4 chars at rdi, rdi += 4
emit4:
        mov     edx, eax
        shr     edx, 18
        movzx   edx, byte ptr [r9 + rdx]
        mov     byte ptr [rdi], dl
        mov     edx, eax
        shr     edx, 12
        and     edx, 63
        movzx   edx, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 1], dl
        mov     edx, eax
        shr     edx, 6
        and     edx, 63
        movzx   edx, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 2], dl
        mov     edx, eax
        and     edx, 63
        movzx   edx, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 3], dl
        add     rdi, 4
        ret

# linecheck: count group, emit newline after 19 groups (76 chars)
linecheck:
        inc     r8d
        cmp     r8d, 19
        jne     1f
        mov     byte ptr [rdi], 10
        inc     rdi
        xor     r8d, r8d
1:
        ret

.section .rodata
alphabet:
        .ascii  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
        .space  65536
outbuf:
        .space  98304
