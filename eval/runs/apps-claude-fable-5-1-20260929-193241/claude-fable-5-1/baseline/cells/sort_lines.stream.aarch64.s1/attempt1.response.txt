_start:
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #32784
    sub     x2, x2, x20
    cmp     x2, #0
    b.le    read_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    // parse lines into tabA: (ptr, len) pairs
    adrp    x21, tabA
    add     x21, x21, :lo12:tabA
    mov     x22, #0                 // n lines
    mov     x4, x19                 // start of current line
    mov     x5, x19                 // p
    add     x6, x19, x20            // end
parse_loop:
    cmp     x5, x6
    b.hs    parse_end
    ldrb    w7, [x5], #1
    cmp     w7, #10
    b.ne    parse_loop
    sub     x8, x5, #1
    sub     x8, x8, x4              // len
    add     x9, x21, x22, lsl #4
    stp     x4, x8, [x9]
    add     x22, x22, #1
    mov     x4, x5
    b       parse_loop
parse_end:
    cmp     x4, x6
    b.hs    sort
    sub     x8, x6, x4
    add     x9, x21, x22, lsl #4
    stp     x4, x8, [x9]
    add     x22, x22, #1
sort:
    // bottom-up merge sort between tabA and tabB
    mov     x23, x21                // src
    adrp    x24, tabB
    add     x24, x24, :lo12:tabB    // dst
    mov     x25, #1                 // width
sort_outer:
    cmp     x25, x22
    b.hs    sort_done
    mov     x26, #0                 // lo
sort_inner:
    cmp     x26, x22
    b.hs    sort_pass_done
    add     x27, x26, x25           // mid
    cmp     x27, x22
    csel    x27, x27, x22, lo
    add     x28, x27, x25           // hi
    cmp     x28, x22
    csel    x28, x28, x22, lo
    mov     x10, x26                // i
    mov     x11, x27                // j
    mov     x12, x26                // k
merge_loop:
    cmp     x10, x27
    b.hs    merge_right_rest
    cmp     x11, x28
    b.hs    merge_take_left
    add     x13, x23, x10, lsl #4
    ldp     x0, x1, [x13]
    add     x14, x23, x11, lsl #4
    ldp     x2, x3, [x14]
    bl      compare
    cmp     w0, #0
    b.gt    merge_take_right
merge_take_left:
    add     x13, x23, x10, lsl #4
    ldp     x0, x1, [x13]
    add     x14, x24, x12, lsl #4
    stp     x0, x1, [x14]
    add     x10, x10, #1
    add     x12, x12, #1
    b       merge_loop
merge_take_right:
    add     x13, x23, x11, lsl #4
    ldp     x0, x1, [x13]
    add     x14, x24, x12, lsl #4
    stp     x0, x1, [x14]
    add     x11, x11, #1
    add     x12, x12, #1
    b       merge_loop
merge_right_rest:
    cmp     x11, x28
    b.hs    merge_done
    b       merge_take_right
merge_done:
    add     x26, x26, x25, lsl #1
    b       sort_inner
sort_pass_done:
    mov     x0, x23
    mov     x23, x24
    mov     x24, x0
    lsl     x25, x25, #1
    b       sort_outer
sort_done:
    // build output buffer
    adrp    x24, outbuf
    add     x24, x24, :lo12:outbuf
    mov     x10, x24                // out ptr
    mov     x11, #0                 // index
out_loop:
    cmp     x11, x22
    b.hs    out_done
    add     x13, x23, x11, lsl #4
    ldp     x0, x1, [x13]
    add     x11, x11, #1
copy_loop:
    cbz     x1, copy_end
    ldrb    w2, [x0], #1
    strb    w2, [x10], #1
    sub     x1, x1, #1
    b       copy_loop
copy_end:
    mov     w2, #10
    strb    w2, [x10], #1
    b       out_loop
out_done:
    sub     x20, x10, x24           // total output length
    mov     x21, x24
write_loop:
    cbz     x20, do_exit
    mov     x0, #1
    mov     x1, x21
    mov     x2, x20
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    do_exit
    add     x21, x21, x0
    sub     x20, x20, x0
    b       write_loop
do_exit:
    mov     x0, #0
    mov     x8, #94
    svc     #0
    b       do_exit

// compare(x0=ptrA, x1=lenA, x2=ptrB, x3=lenB) -> w0 <0, 0, >0
compare:
    cmp     x1, x3
    csel    x4, x1, x3, lo          // minlen
    mov     x5, #0
cmp_loop:
    cmp     x5, x4
    b.hs    cmp_len
    ldrb    w6, [x0, x5]
    ldrb    w7, [x2, x5]
    add     x5, x5, #1
    cmp     w6, w7
    b.eq    cmp_loop
    sub     w0, w6, w7
    ret
cmp_len:
    sub     x0, x1, x3
    ret
