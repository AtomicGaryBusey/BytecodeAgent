.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r9, [rip + inbuf]
    lea r8, [r9 + rbx]
    lea rdi, [rip + outbuf]
line_loop:
    cmp r9, r8
    jae output
    mov r10, r9
scan:
    cmp r10, r8
    jae scan_done
    cmp byte ptr [r10], 0x0A
    je scan_done
    inc r10
    jmp scan
scan_done:
    mov r11, r10
rev:
    cmp r11, r9
    jbe rev_done
    dec r11
    mov al, byte ptr [r11]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev
rev_done:
    mov byte ptr [rdi], 0x0A
    inc rdi
    cmp r10, r8
    jae output
    lea r9, [r10 + 1]
    jmp line_loop
output:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp done

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 65544
