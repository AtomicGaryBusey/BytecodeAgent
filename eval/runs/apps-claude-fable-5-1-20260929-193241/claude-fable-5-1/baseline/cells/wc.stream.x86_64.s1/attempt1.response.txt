.text
.globl _start
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in_word flag

read_loop:
    xor eax, eax            # read
    xor edi, edi            # fd 0
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    lea rsi, [rip + inbuf]
    mov rcx, rax
    add r14, rax

scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    # word byte
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    dec rcx
    jnz scan
    jmp read_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call fmt
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call fmt
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call fmt
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp done_reading

# fmt: rax = value, rdi = destination; writes decimal digits, advances rdi
fmt:
    sub rsp, 32
    lea rsi, [rsp + 32]
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rcx, [rsp + 32]
    sub rcx, rsi
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz 2b
    add rsp, 32
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 128
