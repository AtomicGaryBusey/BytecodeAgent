.text
.globl _start
_start:
        xor     r13d, r13d              # total bytes read
read_loop:
        lea     rsi, [rip + inbuf]
        add     rsi, r13
        mov     edx, 49152 + 64
        sub     edx, r13d
        jle     read_done
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        syscall
        test    rax, rax
        jle     read_done
        add     r13, rax
        jmp     read_loop

read_done:
        lea     r12, [rip + inbuf]      # input pointer
        lea     r14, [rip + outbuf]     # output pointer
        xor     r15d, r15d              # column count
        lea     rbx, [rip + alphabet]

enc_loop:
        cmp     r13, 3
        jb      enc_tail
        movzx   edx, byte ptr [r12]
        shl     edx, 8
        mov     dl, byte ptr [r12 + 1]
        shl     edx, 8
        mov     dl, byte ptr [r12 + 2]
        add     r12, 3
        sub     r13, 3

        mov     ecx, edx
        shr     ecx, 18
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     ecx, edx
        shr     ecx, 12
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     ecx, edx
        shr     ecx, 6
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     ecx, edx
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        jmp     enc_loop

enc_tail:
        test    r13, r13
        jz      enc_done
        cmp     r13, 1
        je      tail1
        # two bytes remain
        movzx   edx, byte ptr [r12]
        shl     edx, 8
        mov     dl, byte ptr [r12 + 1]
        shl     edx, 8
        mov     ecx, edx
        shr     ecx, 18
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     ecx, edx
        shr     ecx, 12
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     ecx, edx
        shr     ecx, 6
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     eax, '='
        call    emit
        jmp     enc_done
tail1:
        movzx   edx, byte ptr [r12]
        shl     edx, 16
        mov     ecx, edx
        shr     ecx, 18
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     ecx, edx
        shr     ecx, 12
        and     ecx, 63
        movzx   eax, byte ptr [rbx + rcx]
        call    emit
        mov     eax, '='
        call    emit
        mov     eax, '='
        call    emit

enc_done:
        test    r15, r15
        jz      write_out
        mov     byte ptr [r14], 10
        inc     r14

write_out:
        lea     rsi, [rip + outbuf]     # start
        mov     rdx, r14
        sub     rdx, rsi                # length
write_loop:
        test    rdx, rdx
        jle     done
        mov     eax, 1                  # write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop

done:
        mov     eax, 231                # exit_group
        xor     edi, edi
        syscall
        jmp     done

# emit: store al to output, handle 76-char line wrapping
emit:
        mov     byte ptr [r14], al
        inc     r14
        inc     r15
        cmp     r15, 76
        jne     1f
        mov     byte ptr [r14], 10
        inc     r14
        xor     r15d, r15d
1:
        ret

        .section .rodata
alphabet:
        .ascii  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

        .bss
        .balign 16
inbuf:
        .space  49152 + 64
        .balign 16
outbuf:
        .space  70000
