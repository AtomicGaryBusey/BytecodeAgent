.bss
.balign 16
inbuf: .space 65600
outbuf: .space 65600

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x20, #0                 // total bytes read
    mov x23, #0                 // output length
read_loop:
    movz x2, #0x0040
    movk x2, #0x0001, lsl #16   // 65600
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x24, #0                 // line start
    mov x25, #0                 // index
scan_loop:
    cmp x25, x20
    beq scan_end
    ldrb w0, [x19, x25]
    cmp w0, #10
    bne scan_next
    mov x0, x24
    mov x1, x25
    bl emit_line
    add x24, x25, #1
scan_next:
    add x25, x25, #1
    b scan_loop
scan_end:
    cmp x24, x20
    bge write_out
    mov x0, x24
    mov x1, x20
    bl emit_line
write_out:
    mov x26, #0                 // bytes written so far
write_loop:
    sub x2, x23, x26
    cbz x2, finish
    mov x0, #1
    add x1, x22, x26
    mov x8, #64
    svc #0
    cmp x0, #0
    ble finish
    add x26, x26, x0
    b write_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0
    b finish

// emit_line: x0 = start index, x1 = end index (exclusive) in inbuf
// appends reversed bytes + newline to outbuf at x23
emit_line:
    mov x2, x1
1:
    cmp x2, x0
    ble 2f
    sub x2, x2, #1
    ldrb w3, [x19, x2]
    strb w3, [x22, x23]
    add x23, x23, #1
    b 1b
2:
    mov w3, #10
    strb w3, [x22, x23]
    add x23, x23, #1
    ret
