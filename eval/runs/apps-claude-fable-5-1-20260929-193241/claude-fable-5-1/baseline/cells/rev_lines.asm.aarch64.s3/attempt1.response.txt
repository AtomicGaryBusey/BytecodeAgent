.text
.globl _start
_start:
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        adrp    x21, outbuf
        add     x21, x21, :lo12:outbuf
        mov     x20, #0                 // bytes read
read_loop:
        mov     x2, #65536
        sub     x2, x2, x20
        cbz     x2, read_done
        mov     x0, #0
        add     x1, x19, x20
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        ble     read_done
        add     x20, x20, x0
        b       read_loop
read_done:
        mov     x22, #0                 // output length
        mov     x23, #0                 // line start
        mov     x24, #0                 // scan index
scan:
        cmp     x24, x20
        bge     end_scan
        ldrb    w0, [x19, x24]
        cmp     w0, #10
        bne     scan_next
        bl      emit_line
        add     x23, x24, #1
scan_next:
        add     x24, x24, #1
        b       scan
end_scan:
        cmp     x23, x20
        bge     write_out
        bl      emit_line
write_out:
        mov     x25, #0
write_loop:
        cmp     x25, x22
        bge     done
        mov     x0, #1
        add     x1, x21, x25
        sub     x2, x22, x25
        mov     x8, #64
        svc     #0
        cmp     x0, #0
        ble     done
        add     x25, x25, x0
        b       write_loop
done:
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       done

// emit reversed bytes [x23, x24) from inbuf into outbuf at x22, then newline
emit_line:
        mov     x25, x24
1:
        cmp     x25, x23
        ble     2f
        sub     x25, x25, #1
        ldrb    w0, [x19, x25]
        strb    w0, [x21, x22]
        add     x22, x22, #1
        b       1b
2:
        mov     w0, #10
        strb    w0, [x21, x22]
        add     x22, x22, #1
        ret
