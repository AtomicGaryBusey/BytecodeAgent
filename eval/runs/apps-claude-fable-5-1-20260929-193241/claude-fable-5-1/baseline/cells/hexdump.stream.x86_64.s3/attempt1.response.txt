.text
.globl _start
_start:
    xor r12d, r12d                  # total input length
    lea r15, [rip + inbuf]
1:
    xor eax, eax                    # read
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 4096
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    cmp r12, 65536
    jbe 1b
2:
    lea rbx, [rip + hexdigits]
    lea rdi, [rip + outbuf]
    lea rsi, [rip + inbuf]
    xor r13d, r13d                  # current offset
row_loop:
    cmp r13, r12
    jae done
    mov rax, r12
    sub rax, r13
    mov r14d, 16
    cmp rax, 16
    cmovb r14, rax                  # n = min(16, remaining)
    mov eax, r13d
    call emit_hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
hex_loop:
    cmp rcx, r14
    jae pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp next_slot
pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
next_slot:
    cmp ecx, 7
    jne 3f
    mov byte ptr [rdi], 0x20
    inc rdi
3:
    inc ecx
    cmp ecx, 16
    jb hex_loop
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx
chr_loop:
    cmp rcx, r14
    jae chr_done
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    sub edx, 0x20
    cmp edx, 0x5f
    jb 4f
    mov eax, 0x2e
4:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    jmp chr_loop
chr_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    add r13, r14
    add rsi, r14
    jmp row_loop
done:
    mov eax, r12d
    call emit_hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea r15, [rip + outbuf]
    mov r14, rdi
    sub r14, r15                    # output length
wr_loop:
    test r14, r14
    jz exit
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r14
    syscall
    test rax, rax
    jle exit
    add r15, rax
    sub r14, rax
    jmp wr_loop
exit:
    mov eax, 231
    xor edi, edi
    syscall
    jmp exit

# eax = value, rdi = out pointer (advanced by 8), rbx = hexdigits
emit_hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536 + 8192
outbuf:
    .space 4096 * 79 + 64
