.text
.globl _start

.equ INBUF_SIZE, 65600

_start:
    lea rbx, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, INBUF_SIZE
    sub edx, r13d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp read_done

# ---------------------------------------------------------------
# error: print "error at N\n" with N = r12, exit 1
error:
    lea rdi, [rip + outbuf + 32]
    dec rdi
    mov byte ptr [rdi], 0x0A
    mov rax, r12
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    lea rsi, [rip + err_msg]
    xor edx, edx
2:
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [rdi + rdx], al
    inc edx
    cmp edx, 9
    jb 2b
    mov rsi, rdi
    lea rdx, [rip + outbuf + 32]
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
    jmp error

# ---------------------------------------------------------------
# peek: eax = byte at r12, or 0x100 at end of input
peek:
    mov eax, 0x100
    cmp r12, r13
    jae 1f
    movzx eax, byte ptr [rbx + r12]
1:
    ret

skip_ws:
    call peek
    cmp eax, 0x20
    je 1f
    cmp eax, 0x09
    je 1f
    cmp eax, 0x0A
    je 1f
    cmp eax, 0x0D
    jne 2f
1:
    inc r12
    jmp skip_ws
2:
    ret

# ---------------------------------------------------------------
parse_value:
    call peek
    cmp eax, 0x7B
    je parse_object
    cmp eax, 0x5B
    je parse_array
    cmp eax, 0x22
    je parse_string
    cmp eax, 0x2D
    je parse_number
    cmp eax, 0x30
    jb error
    cmp eax, 0x39
    jbe parse_number
    cmp eax, 0x74
    je pv_true
    cmp eax, 0x66
    je pv_false
    cmp eax, 0x6E
    je pv_null
    jmp error
pv_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp parse_lit
pv_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp parse_lit
pv_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4
parse_lit:
    call peek
    movzx edx, byte ptr [rsi]
    cmp eax, edx
    jne error
    inc r12
    inc rsi
    dec ecx
    jnz parse_lit
    ret

# ---------------------------------------------------------------
parse_object:
    inc r12
    call skip_ws
    call peek
    cmp eax, 0x7D
    jne obj_member
    inc r12
    ret
obj_member:
    cmp eax, 0x22
    jne error
    call parse_string
    call skip_ws
    call peek
    cmp eax, 0x3A
    jne error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, 0x2C
    je obj_comma
    cmp eax, 0x7D
    jne error
    inc r12
    ret
obj_comma:
    inc r12
    call skip_ws
    call peek
    jmp obj_member

# ---------------------------------------------------------------
parse_array:
    inc r12
    call skip_ws
    call peek
    cmp eax, 0x5D
    jne arr_elem
    inc r12
    ret
arr_elem:
    call parse_value
    call skip_ws
    call peek
    cmp eax, 0x2C
    je arr_comma
    cmp eax, 0x5D
    jne error
    inc r12
    ret
arr_comma:
    inc r12
    call skip_ws
    jmp arr_elem

# ---------------------------------------------------------------
parse_string:
    inc r12
str_loop:
    call peek
    cmp eax, 0x22
    je str_end
    cmp eax, 0x5C
    je str_esc
    cmp eax, 0x20
    jb error
    cmp eax, 0x100
    jae error
    inc r12
    jmp str_loop
str_end:
    inc r12
    ret
str_esc:
    inc r12
    call peek
    cmp eax, 0x75
    je str_u
    cmp eax, 0x22
    je str_esc_ok
    cmp eax, 0x5C
    je str_esc_ok
    cmp eax, 0x2F
    je str_esc_ok
    cmp eax, 0x62
    je str_esc_ok
    cmp eax, 0x66
    je str_esc_ok
    cmp eax, 0x6E
    je str_esc_ok
    cmp eax, 0x72
    je str_esc_ok
    cmp eax, 0x74
    je str_esc_ok
    jmp error
str_esc_ok:
    inc r12
    jmp str_loop
str_u:
    inc r12
    mov ecx, 4
str_hex:
    call peek
    cmp eax, 0x30
    jb error
    cmp eax, 0x39
    jbe str_hex_ok
    or eax, 0x20
    cmp eax, 0x61
    jb error
    cmp eax, 0x66
    ja error
str_hex_ok:
    inc r12
    dec ecx
    jnz str_hex
    jmp str_loop

# ---------------------------------------------------------------
parse_number:
    call peek
    cmp eax, 0x2D
    jne 1f
    inc r12
    call peek
1:
    cmp eax, 0x30
    jb error
    ja 2f
    inc r12
    jmp num_frac
2:
    cmp eax, 0x39
    ja error
    inc r12
    call skip_digits
num_frac:
    call peek
    cmp eax, 0x2E
    jne num_exp
    inc r12
    call one_or_more_digits
num_exp:
    call peek
    or eax, 0x20
    cmp eax, 0x65
    jne num_done
    inc r12
    call peek
    cmp eax, 0x2B
    je 3f
    cmp eax, 0x2D
    jne 4f
3:
    inc r12
4:
    call one_or_more_digits
num_done:
    ret

one_or_more_digits:
    call peek
    cmp eax, 0x30
    jb error
    cmp eax, 0x39
    ja error
    inc r12
skip_digits:
    call peek
    cmp eax, 0x30
    jb 1f
    cmp eax, 0x39
    ja 1f
    inc r12
    jmp skip_digits
1:
    ret

# ---------------------------------------------------------------
.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space 32
