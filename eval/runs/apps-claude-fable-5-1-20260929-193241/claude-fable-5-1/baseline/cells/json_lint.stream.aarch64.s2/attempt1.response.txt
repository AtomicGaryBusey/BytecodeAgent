.text
.globl _start
_start:
    adrp x20, buf
    add x20, x20, :lo12:buf
    mov x21, #0
    mov x22, #0x10000
read_loop:
    sub x2, x22, x21
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop
read_done:
    mov x19, x20
    bl skip_ws
    bl parse_value
    bl skip_ws
    add x0, x20, x21
    cmp x19, x0
    b.ne error
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b .

error:
    sub x0, x19, x20
    sub sp, sp, #48
    add x1, sp, #47
    mov w2, #10
    strb w2, [x1]
    mov x3, #10
1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #0x30
    sub x1, x1, #1
    strb w5, [x1]
    mov x0, x4
    cbnz x0, 1b
    adrp x6, err_msg
    add x6, x6, :lo12:err_msg
    mov x7, #9
2:
    sub x7, x7, #1
    ldrb w8, [x6, x7]
    sub x1, x1, #1
    strb w8, [x1]
    cbnz x7, 2b
    add x2, sp, #48
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b .

skip_ws:
    ldrb w0, [x19]
    cmp w0, #0x20
    b.eq 1f
    cmp w0, #0x09
    b.eq 1f
    cmp w0, #0x0a
    b.eq 1f
    cmp w0, #0x0d
    b.ne 2f
1:
    add x19, x19, #1
    b skip_ws
2:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    ldrb w0, [x19]
    cmp w0, #0x7b
    b.eq pv_obj
    cmp w0, #0x5b
    b.eq pv_arr
    cmp w0, #0x22
    b.eq pv_str
    cmp w0, #0x2d
    b.eq pv_num
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pv_num
    cmp w0, #0x74
    b.eq pv_true
    cmp w0, #0x66
    b.eq pv_false
    cmp w0, #0x6e
    b.eq pv_null
    b error
pv_obj:
    bl parse_object
    b pv_ret
pv_arr:
    bl parse_array
    b pv_ret
pv_str:
    bl parse_string
    b pv_ret
pv_num:
    bl parse_number
    b pv_ret
pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    bl expect_lit
    b pv_ret
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    bl expect_lit
    b pv_ret
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    bl expect_lit
pv_ret:
    ldp x29, x30, [sp], #16
    ret

expect_lit:
    ldrb w2, [x1], #1
    cbz w2, 1f
    ldrb w0, [x19]
    cmp w0, w2
    b.ne error
    add x19, x19, #1
    b expect_lit
1:
    ret

parse_string:
    add x19, x19, #1
ps_loop:
    ldrb w0, [x19]
    cmp w0, #0x22
    b.eq ps_end
    cmp w0, #0x20
    b.lo error
    cmp w0, #0x5c
    b.ne ps_plain
    add x19, x19, #1
    ldrb w0, [x19]
    cmp w0, #0x75
    b.eq ps_u
    cmp w0, #0x22
    b.eq ps_plain
    cmp w0, #0x5c
    b.eq ps_plain
    cmp w0, #0x2f
    b.eq ps_plain
    cmp w0, #0x62
    b.eq ps_plain
    cmp w0, #0x66
    b.eq ps_plain
    cmp w0, #0x6e
    b.eq ps_plain
    cmp w0, #0x72
    b.eq ps_plain
    cmp w0, #0x74
    b.eq ps_plain
    b error
ps_u:
    add x19, x19, #1
    mov w3, #4
1:
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 2f
    orr w1, w0, #0x20
    sub w1, w1, #0x61
    cmp w1, #5
    b.hi error
2:
    add x19, x19, #1
    subs w3, w3, #1
    b.ne 1b
    b ps_loop
ps_plain:
    add x19, x19, #1
    b ps_loop
ps_end:
    add x19, x19, #1
    ret

parse_number:
    stp x29, x30, [sp, #-16]!
    ldrb w0, [x19]
    cmp w0, #0x2d
    b.ne 1f
    add x19, x19, #1
    ldrb w0, [x19]
1:
    cmp w0, #0x30
    b.ne 2f
    add x19, x19, #1
    b pn_frac
2:
    sub w1, w0, #0x31
    cmp w1, #8
    b.hi error
    add x19, x19, #1
    bl skip_digits
pn_frac:
    ldrb w0, [x19]
    cmp w0, #0x2e
    b.ne pn_exp
    add x19, x19, #1
    bl require_digits
pn_exp:
    ldrb w0, [x19]
    orr w0, w0, #0x20
    cmp w0, #0x65
    b.ne pn_done
    add x19, x19, #1
    ldrb w0, [x19]
    cmp w0, #0x2b
    b.eq 3f
    cmp w0, #0x2d
    b.ne 4f
3:
    add x19, x19, #1
4:
    bl require_digits
pn_done:
    ldp x29, x30, [sp], #16
    ret

require_digits:
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi error
    add x19, x19, #1
skip_digits:
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi 1f
    add x19, x19, #1
    b skip_digits
1:
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #0x7d
    b.eq po_end
po_member:
    ldrb w0, [x19]
    cmp w0, #0x22
    b.ne error
    bl parse_string
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #0x3a
    b.ne error
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #0x2c
    b.eq po_comma
    cmp w0, #0x7d
    b.ne error
po_end:
    add x19, x19, #1
    ldp x29, x30, [sp], #16
    ret
po_comma:
    add x19, x19, #1
    bl skip_ws
    b po_member

parse_array:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #0x5d
    b.eq pa_end
pa_elem:
    bl parse_value
    bl skip_ws
    ldrb w0, [x19]
    cmp w0, #0x2c
    b.eq pa_comma
    cmp w0, #0x5d
    b.ne error
pa_end:
    add x19, x19, #1
    ldp x29, x30, [sp], #16
    ret
pa_comma:
    add x19, x19, #1
    bl skip_ws
    b pa_elem
