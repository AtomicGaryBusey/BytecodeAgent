.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r9, [rip + alphabet]
    mov rcx, rbx
    xor r8d, r8d

enc_loop:
    cmp rcx, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3
    sub rcx, 3
    call enc4
    call advance
    jmp enc_loop

enc_tail:
    test rcx, rcx
    jz enc_end
    cmp rcx, 1
    je tail1
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    call enc4
    mov byte ptr [rdi + 3], 61
    call advance
    jmp enc_end
tail1:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    call enc4
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    call advance

enc_end:
    test r8d, r8d
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi

do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop

done:
    mov eax, 231
    xor edi, edi
    syscall
    jmp done

# enc4: eax = 24-bit group, writes 4 chars at [rdi], does not advance
enc4:
    mov edx, eax
    shr edx, 18
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], dl
    mov edx, eax
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 3], dl
    ret

# advance: rdi += 4, line counter += 4, emit newline at 76
advance:
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
1:  ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 98304
