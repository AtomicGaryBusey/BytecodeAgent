.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
    xor r15d, r15d
read_loop:
    mov edx, 65536
    sub edx, r15d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r15]
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    add r15, rbx
    mov rsi, rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r15
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp read_done

fail:
    mov rax, rsi
    sub rax, rbx
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:
    dec rdi
    xor edx, edx
    div rcx
    add dl, 0x30
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    mov rax, qword ptr [rip + err_msg]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_msg + 8]
    mov byte ptr [rdi + 8], al
    lea rdx, [rip + numbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall
    jmp fail

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    jne 2f
1:
    inc rsi
    jmp skip_ws
2:
    ret

parse_value:
    movzx eax, byte ptr [rsi]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb 1f
    cmp al, 0x39
    jbe parse_number
1:
    cmp al, 0x74
    je lit_true
    cmp al, 0x66
    je lit_false
    cmp al, 0x6E
    je lit_null
    jmp fail

lit_true:
    lea rdi, [rip + s_true]
    mov ecx, 4
    jmp lit_check
lit_false:
    lea rdi, [rip + s_false]
    mov ecx, 5
    jmp lit_check
lit_null:
    lea rdi, [rip + s_null]
    mov ecx, 4
lit_check:
    mov al, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jne fail
    inc rsi
    inc rdi
    dec ecx
    jnz lit_check
    ret

parse_string:
    inc rsi
str_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb fail
    inc rsi
    jmp str_loop
str_end:
    inc rsi
    ret
str_esc:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 0x75
    je str_u
    cmp al, 0x22
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, 0x2F
    je str_esc_ok
    cmp al, 0x62
    je str_esc_ok
    cmp al, 0x66
    je str_esc_ok
    cmp al, 0x6E
    je str_esc_ok
    cmp al, 0x72
    je str_esc_ok
    cmp al, 0x74
    je str_esc_ok
    jmp fail
str_esc_ok:
    inc rsi
    jmp str_loop
str_u:
    inc rsi
    mov ecx, 4
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe 2f
    or al, 0x20
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
2:
    inc rsi
    dec ecx
    jnz 1b
    jmp str_loop

parse_number:
    movzx eax, byte ptr [rsi]
    cmp al, 0x2D
    jne 1f
    inc rsi
    movzx eax, byte ptr [rsi]
1:
    cmp al, 0x30
    jne 2f
    inc rsi
    jmp num_frac
2:
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
    call skip_digits
num_frac:
    movzx eax, byte ptr [rsi]
    cmp al, 0x2E
    jne num_exp
    inc rsi
    call need_digits
num_exp:
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 0x65
    jne num_done
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 0x2B
    je 1f
    cmp al, 0x2D
    jne 2f
1:
    inc rsi
2:
    call need_digits
num_done:
    ret

need_digits:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc rsi
skip_digits:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    jb 1f
    cmp al, 0x39
    ja 1f
    inc rsi
    jmp skip_digits
1:
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], 0x7D
    jne obj_member
    inc rsi
    ret
obj_member:
    cmp byte ptr [rsi], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], 0x3A
    jne fail
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, 0x2C
    je obj_comma
    cmp al, 0x7D
    jne fail
    inc rsi
    ret
obj_comma:
    inc rsi
    call skip_ws
    jmp obj_member

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], 0x5D
    jne arr_elem
    inc rsi
    ret
arr_elem:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, 0x2C
    je arr_comma
    cmp al, 0x5D
    jne fail
    inc rsi
    ret
arr_comma:
    inc rsi
    call skip_ws
    jmp arr_elem

.section .rodata
ok_msg: .ascii "ok\n"
err_msg: .ascii "error at "
s_true: .ascii "true"
s_false: .ascii "false"
s_null: .ascii "null"

.bss
.balign 16
inbuf: .space 65552
numbuf: .space 32
