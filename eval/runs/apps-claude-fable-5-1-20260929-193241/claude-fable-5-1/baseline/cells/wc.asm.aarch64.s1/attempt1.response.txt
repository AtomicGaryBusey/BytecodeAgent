    mov x1, x23
    mov x2, #65536
    mov x8, #63

    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
proc:

    cmp w3, #0x0A
    b.ne 1f
    add x19, x19, #1
1:

    cmp w4, #4
    b.ls ws
    cmp w3, #0x20
    b.eq ws
    cbnz x22, 2f
    add x20, x20, #1
    mov x22, #1
    b 2f
ws:

    cmp x1, x2
    b.lo proc
    b read_loop

    add x5, sp, #80
    mov w4, #0x0A
    strb w4, [x5, #-1]!
    mov x0, x21
    bl fmt

    strb w4, [x5, #-1]!
    mov x0, x20
    bl fmt
    mov w4, #0x20
    strb w4, [x5, #-1]!
    mov x0, x19

    mov x0, #1
    mov x8, #64

    mov x0, #0
