        lea     rsi, [rip + inbuf]
        xor     ebx, ebx
read_loop:

        sub     edx, ebx

        xor     eax, eax

        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     rsi, [rip + inbuf]
        xor     ecx, ecx
find_loop:
        cmp     rcx, rbx
        jae     no_bang
        cmp     byte ptr [rsi + rcx], 0x21
        je      found
        inc     rcx
        jmp     find_loop
found:
        lea     r13, [rsi + rcx + 1]
        lea     r14, [rsi + rbx]
        jmp     have_prog
no_bang:
        lea     r13, [rsi + rbx]
        mov     r14, r13
have_prog:
        mov     rbp, rcx
        xor     ecx, ecx
        xor     edx, edx
        lea     rdi, [rip + jt]
        lea     r8, [rip + bstack]
bpass:
        cmp     rcx, rbp
        jae     bpass_done
        movzx   eax, byte ptr [rsi + rcx]
        cmp     al, 0x5B
        je      bopen
        cmp     al, 0x5D
        je      bclose
        inc     rcx
        jmp     bpass
bopen:
        mov     dword ptr [r8 + rdx*4], ecx
        inc     edx
        inc     rcx
        jmp     bpass
bclose:
        test    edx, edx
        jz      bad
        dec     edx
        mov     eax, dword ptr [r8 + rdx*4]
        mov     dword ptr [rdi + rax*4], ecx
        mov     dword ptr [rdi + rcx*4], eax
        inc     rcx
        jmp     bpass
bpass_done:
        test    edx, edx
        jnz     bad
        lea     r12, [rip + tape]
        lea     r15, [rip + outbuf]
        xor     ebx, ebx
run:
        cmp     rbx, rbp
        jae     done
        movzx   eax, byte ptr [rsi + rbx]
        inc     rbx
        cmp     al, 0x3E
        je      c_right
        cmp     al, 0x3C
        je      c_left
        cmp     al, 0x2B
        je      c_plus
        cmp     al, 0x2D
        je      c_minus
        cmp     al, 0x2E
        je      c_out
        cmp     al, 0x2C
        je      c_in
        cmp     al, 0x5B
        je      c_open
        cmp     al, 0x5D
        je      c_close
        jmp     run
c_right:
        inc     r12
        jmp     run
c_left:
        dec     r12
        jmp     run
c_plus:
        inc     byte ptr [r12]
        jmp     run
c_minus:
        dec     byte ptr [r12]
        jmp     run
c_out:

        mov     byte ptr [r15], al
        inc     r15
        lea     rax, [rip + outbuf + 65536]
        cmp     r15, rax
        jb      run
        call    flush
        jmp     run
c_in:
        xor     eax, eax
        cmp     r13, r14
        jae     1f
        mov     al, byte ptr [r13]
        inc     r13
1:
        mov     byte ptr [r12], al
        jmp     run
c_open:
        cmp     byte ptr [r12], 0
        jne     run
        mov     ebx, dword ptr [rdi + rbx*4 - 4]
        inc     rbx
        jmp     run
c_close:
        cmp     byte ptr [r12], 0
        je      run
        mov     ebx, dword ptr [rdi + rbx*4 - 4]
        inc     rbx
        jmp     run
done:
        call    flush
        mov     eax, 231
        xor     edi, edi
        syscall
bad:
        mov     eax, 231
        mov     edi, 1
        syscall
flush:
        push    rsi
        push    rdi
        push    rdx
        lea     rsi, [rip + outbuf]
flush_loop:
        mov     rdx, r15
        sub     rdx, rsi
        jle     flush_done
        mov     eax, 1
        mov     edi, 1
        syscall
        test    rax, rax
        jle     flush_done
        add     rsi, rax
        jmp     flush_loop
flush_done:
        lea     r15, [rip + outbuf]
        pop     rdx
        pop     rdi
        pop     rsi
        ret
