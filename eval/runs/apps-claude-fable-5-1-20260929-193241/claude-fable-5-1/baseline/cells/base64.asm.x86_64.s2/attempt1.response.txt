.text
.globl _start
_start:
        xor r12d, r12d                  # total bytes read
read_loop:
        mov edx, 65536
        sub edx, r12d
        jle read_done
        lea rsi, [rip + inbuf]
        add rsi, r12
        xor edi, edi
        xor eax, eax
        syscall
        test rax, rax
        jle read_done
        add r12, rax
        jmp read_loop
read_done:
        mov rax, r12
        xor edx, edx
        mov ecx, 3
        div rcx
        mov r14, rdx                    # remainder 0..2
        mov rcx, rax                    # full groups
        lea rsi, [rip + inbuf]
        lea rdi, [rip + outbuf]
        lea rbx, [rip + alphabet]
        xor r13d, r13d                  # line count
        xor r15d, r15d                  # padding count
enc_loop:
        test rcx, rcx
        jz enc_rem
        movzx eax, byte ptr [rsi]
        shl eax, 16
        movzx edx, byte ptr [rsi + 1]
        shl edx, 8
        or eax, edx
        movzx edx, byte ptr [rsi + 2]
        or eax, edx
        add rsi, 3
        call emit4
        dec rcx
        jmp enc_loop
enc_rem:
        test r14, r14
        jz enc_fin
        cmp r14, 1
        jne rem2
        movzx eax, byte ptr [rsi]
        shl eax, 16
        mov r15d, 1
        call emit4
        jmp enc_fin
rem2:
        movzx eax, byte ptr [rsi]
        shl eax, 16
        movzx edx, byte ptr [rsi + 1]
        shl edx, 8
        or eax, edx
        mov r15d, 2
        call emit4
enc_fin:
        test r13d, r13d
        jz do_write
        mov byte ptr [rdi], 10
        inc rdi
do_write:
        lea rsi, [rip + outbuf]
        mov rdx, rdi
        sub rdx, rsi                    # total length
write_loop:
        test rdx, rdx
        jle finish
        mov eax, 1
        mov edi, 1
        syscall
        test rax, rax
        jle finish
        add rsi, rax
        sub rdx, rax
        jmp write_loop
finish:
        mov eax, 231
        xor edi, edi
        syscall
        mov eax, 60
        xor edi, edi
        syscall
        jmp finish

# emit4: eax = 24-bit group, writes 4 chars at rdi, handles padding (r15) and lines
emit4:
        mov edx, eax
        shr edx, 18
        and edx, 63
        mov dl, byte ptr [rbx + rdx]
        mov byte ptr [rdi], dl
        mov edx, eax
        shr edx, 12
        and edx, 63
        mov dl, byte ptr [rbx + rdx]
        mov byte ptr [rdi + 1], dl
        mov edx, eax
        shr edx, 6
        and edx, 63
        mov dl, byte ptr [rbx + rdx]
        mov byte ptr [rdi + 2], dl
        mov edx, eax
        and edx, 63
        mov dl, byte ptr [rbx + rdx]
        mov byte ptr [rdi + 3], dl
        test r15d, r15d
        jz 2f
        mov byte ptr [rdi + 3], 61
        cmp r15d, 2
        jne 2f
        mov byte ptr [rdi + 2], 61
2:
        add rdi, 4
        add r13d, 4
        cmp r13d, 76
        jne 1f
        mov byte ptr [rdi], 10
        inc rdi
        xor r13d, r13d
1:
        ret

.section .rodata
alphabet:
        .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
        .space 65536
.balign 16
outbuf:
        .space 98304
