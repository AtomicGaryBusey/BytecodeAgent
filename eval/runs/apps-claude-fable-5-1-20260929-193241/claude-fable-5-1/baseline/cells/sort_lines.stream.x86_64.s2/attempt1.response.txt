.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 32800
    sub edx, ebx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor ecx, ecx
    xor r8d, r8d
    lea r9, [rip + lines]
    lea rsi, [rip + buf]
parse_loop:
    cmp rcx, rbx
    jae parse_done
    mov edx, ecx
find_nl:
    cmp rcx, rbx
    jae store
    cmp byte ptr [rsi + rcx], 10
    je store
    inc rcx
    jmp find_nl
store:
    mov dword ptr [r9 + r8*8], edx
    mov eax, ecx
    sub eax, edx
    mov dword ptr [r9 + r8*8 + 4], eax
    inc r8
    cmp rcx, rbx
    jae parse_done
    inc rcx
    jmp parse_loop
parse_done:
    mov r10d, 1
    lea r12, [rip + lines]
    lea r13, [rip + tmp]
sort_pass:
    cmp r10, r8
    jae sort_done
    xor r11d, r11d
merge_runs:
    cmp r11, r8
    jae pass_done
    mov r14, r11
    add r14, r10
    cmp r14, r8
    cmova r14, r8
    mov r15, r14
    add r15, r10
    cmp r15, r8
    cmova r15, r8
    mov rdi, r11
    mov rsi, r14
    mov rdx, r11
merge_loop:
    cmp rdi, r14
    jae take_b_rest
    cmp rsi, r15
    jae take_a
    call compare
    test eax, eax
    jz take_a
    mov rax, qword ptr [r12 + rsi*8]
    mov qword ptr [r13 + rdx*8], rax
    inc rsi
    inc rdx
    jmp merge_loop
take_a:
    mov rax, qword ptr [r12 + rdi*8]
    mov qword ptr [r13 + rdx*8], rax
    inc rdi
    inc rdx
    jmp merge_loop
take_b_rest:
    cmp rsi, r15
    jae run_done
    mov rax, qword ptr [r12 + rsi*8]
    mov qword ptr [r13 + rdx*8], rax
    inc rsi
    inc rdx
    jmp take_b_rest
run_done:
    mov r11, r15
    jmp merge_runs
pass_done:
    xchg r12, r13
    add r10, r10
    jmp sort_pass
sort_done:
    lea rdi, [rip + outbuf]
    xor ebx, ebx
    lea r9, [rip + buf]
out_loop:
    cmp rbx, r8
    jae out_done
    mov esi, dword ptr [r12 + rbx*8]
    add rsi, r9
    mov ecx, dword ptr [r12 + rbx*8 + 4]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall

# compare: rdi = index a, rsi = index b, r12 = array base
# returns eax = 0 if a <= b, 1 if a > b
compare:
    mov ecx, dword ptr [r12 + rdi*8 + 4]
    mov ebp, dword ptr [r12 + rsi*8 + 4]
    mov eax, dword ptr [r12 + rdi*8]
    mov ebx, dword ptr [r12 + rsi*8]
    lea r9, [rip + buf]
    add rax, r9
    add rbx, r9
    mov r9, rcx
    cmp r9, rbp
    cmova r9, rbp
    sub rcx, rbp
    test r9, r9
    jz cmp_prefix
cmp_loop:
    movzx ebp, byte ptr [rax]
    cmp bpl, byte ptr [rbx]
    jb cmp_le
    ja cmp_gt
    inc rax
    inc rbx
    dec r9
    jnz cmp_loop
cmp_prefix:
    test rcx, rcx
    jg cmp_gt
cmp_le:
    xor eax, eax
    ret
cmp_gt:
    mov eax, 1
    ret

.bss
.balign 16
buf:
    .space 32800
.balign 16
outbuf:
    .space 33000
.balign 16
lines:
    .space 264000
.balign 16
tmp:
    .space 264000
