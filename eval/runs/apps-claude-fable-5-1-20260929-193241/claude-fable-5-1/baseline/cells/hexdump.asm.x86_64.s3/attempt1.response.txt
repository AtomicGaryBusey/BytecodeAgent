.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65552
    sub edx, ebx
    jle done_read
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle done_read
    add rbx, rax
    jmp read_loop
done_read:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + inbuf]
    xor r12d, r12d
row_loop:
    mov r13, rbx
    sub r13, r12
    jz final
    cmp r13, 16
    jbe 1f
    mov r13d, 16
1:
    mov eax, r12d
    call put_hex32
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
hex_loop:
    cmp ecx, r13d
    jae spaces
    movzx eax, byte ptr [rsi + rcx]
    call put_hex8
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp next_slot
spaces:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
next_slot:
    cmp ecx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc ecx
    cmp ecx, 16
    jb hex_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
asc_loop:
    cmp ecx, r13d
    jae asc_done
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 3f
    mov eax, 0x2e
3:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp asc_loop
asc_done:
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add rsi, r13
    add r12, r13
    jmp row_loop
final:
    mov eax, ebx
    call put_hex32
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall

put_hex32:
    mov edx, 8
    lea r8, [rip + hexdigits]
1:
    rol eax, 4
    mov r9d, eax
    and r9d, 15
    mov r9b, byte ptr [r8 + r9]
    mov byte ptr [rdi], r9b
    inc rdi
    dec edx
    jnz 1b
    ret

put_hex8:
    lea r8, [rip + hexdigits]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    add rdi, 2
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65568
.balign 16
outbuf:
    .zero 330000
