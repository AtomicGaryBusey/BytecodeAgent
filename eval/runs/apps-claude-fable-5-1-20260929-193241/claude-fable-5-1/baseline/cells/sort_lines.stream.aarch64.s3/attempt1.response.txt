.text
.globl _start
_start:
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, #0                 // total bytes read
read_loop:
        mov     x8, #63
        mov     x0, #0
        add     x1, x19, x20
        mov     x2, #32768
        sub     x2, x2, x20
        svc     #0
        cmp     x0, #0
        b.le    read_done
        add     x20, x20, x0
        b       read_loop
read_done:
        adrp    x21, tblA
        add     x21, x21, :lo12:tblA
        adrp    x23, tblB
        add     x23, x23, :lo12:tblB
        mov     x22, #0                 // line count
        mov     x1, x19                 // current line start
        mov     x2, x19                 // scan pointer
        add     x3, x19, x20            // end of input
parse_loop:
        cmp     x2, x3
        b.hs    parse_end
        ldrb    w4, [x2], #1
        cmp     w4, #10
        b.ne    parse_loop
        sub     x5, x2, #1
        sub     x5, x5, x1              // length
        add     x6, x21, x22, lsl #4
        stp     x1, x5, [x6]
        add     x22, x22, #1
        mov     x1, x2
        b       parse_loop
parse_end:
        cmp     x1, x3
        b.hs    sort_start
        sub     x5, x3, x1
        add     x6, x21, x22, lsl #4
        stp     x1, x5, [x6]
        add     x22, x22, #1

sort_start:
        // bottom-up merge sort: src = x21, dst = x23, n = x22, width = x24
        mov     x24, #1
width_loop:
        cmp     x24, x22
        b.hs    sort_done
        mov     x25, #0                 // i
chunk_loop:
        cmp     x25, x22
        b.hs    chunk_end
        add     x26, x25, x24           // mid
        cmp     x26, x22
        csel    x26, x26, x22, lo
        add     x27, x26, x24           // hi
        cmp     x27, x22
        csel    x27, x27, x22, lo
        mov     x9, x25                 // l
        mov     x10, x26                // r
        mov     x11, x25                // out
merge_loop:
        cmp     x9, x26
        b.hs    take_r
        cmp     x10, x27
        b.hs    take_l
        add     x0, x21, x9, lsl #4
        add     x1, x21, x10, lsl #4
        bl      cmp_lines
        cbnz    x0, take_l
take_r:
        cmp     x10, x27
        b.hs    merge_end
        add     x0, x21, x10, lsl #4
        add     x10, x10, #1
        b       copy_entry
take_l:
        add     x0, x21, x9, lsl #4
        add     x9, x9, #1
copy_entry:
        ldp     x2, x3, [x0]
        add     x1, x23, x11, lsl #4
        stp     x2, x3, [x1]
        add     x11, x11, #1
        b       merge_loop
merge_end:
        mov     x25, x27
        b       chunk_loop
chunk_end:
        mov     x0, x21
        mov     x21, x23
        mov     x23, x0
        lsl     x24, x24, #1
        b       width_loop

sort_done:
        // build output in outbuf
        adrp    x25, outbuf
        add     x25, x25, :lo12:outbuf
        mov     x26, x25                // output pointer
        mov     x9, #0                  // index
out_loop:
        cmp     x9, x22
        b.hs    out_done
        add     x0, x21, x9, lsl #4
        ldp     x1, x2, [x0]
copy_loop:
        cbz     x2, copy_done
        ldrb    w3, [x1], #1
        strb    w3, [x26], #1
        sub     x2, x2, #1
        b       copy_loop
copy_done:
        mov     w3, #10
        strb    w3, [x26], #1
        add     x9, x9, #1
        b       out_loop
out_done:
        sub     x2, x26, x25
        cbz     x2, do_exit
        mov     x8, #64
        mov     x0, #1
        mov     x1, x25
        svc     #0
do_exit:
        mov     x8, #93
        mov     x0, #0
        svc     #0
        b       do_exit

// cmp_lines: x0 -> entry A, x1 -> entry B; returns x0 = 1 if A <= B else 0
cmp_lines:
        ldp     x2, x3, [x0]
        ldp     x4, x5, [x1]
        cmp     x3, x5
        csel    x6, x3, x5, lo
cmp_loop:
        cbz     x6, cmp_prefix
        ldrb    w7, [x2], #1
        ldrb    w8, [x4], #1
        sub     x6, x6, #1
        cmp     w7, w8
        b.eq    cmp_loop
        cset    x0, lo
        ret
cmp_prefix:
        cmp     x3, x5
        cset    x0, ls
        ret
