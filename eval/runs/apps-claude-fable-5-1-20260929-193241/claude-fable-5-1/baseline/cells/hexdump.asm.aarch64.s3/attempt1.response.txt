        .text
        .globl _start
_start:
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, #0                 // input length
read_loop:
        mov     x2, #0x10000
        sub     x2, x2, x20
        cbz     x2, read_done
        mov     x0, #0
        add     x1, x19, x20
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        ble     read_done
        add     x20, x20, x0
        b       read_loop
read_done:
        adrp    x21, outbuf
        add     x21, x21, :lo12:outbuf
        mov     x22, x21                // output pointer
        adrp    x23, hexdigits
        add     x23, x23, :lo12:hexdigits
        mov     x24, #0                 // current offset
        mov     x26, #16
row_loop:
        cmp     x24, x20
        bge     finish
        sub     x25, x20, x24
        cmp     x25, x26
        csel    x25, x25, x26, lo       // n = min(16, remaining)
        mov     x0, x24
        bl      put_hex8
        mov     w9, #0x20
        strb    w9, [x22], #1
        strb    w9, [x22], #1
        mov     x27, #0
col_loop:
        cmp     x27, #16
        beq     cols_done
        cmp     x27, x25
        bge     blanks
        add     x11, x24, x27
        ldrb    w10, [x19, x11]
        lsr     x11, x10, #4
        ldrb    w11, [x23, x11]
        strb    w11, [x22], #1
        and     x11, x10, #15
        ldrb    w11, [x23, x11]
        strb    w11, [x22], #1
        strb    w9, [x22], #1
        b       after_col
blanks:
        strb    w9, [x22], #1
        strb    w9, [x22], #1
        strb    w9, [x22], #1
after_col:
        cmp     x27, #7
        bne     1f
        strb    w9, [x22], #1
1:
        add     x27, x27, #1
        b       col_loop
cols_done:
        strb    w9, [x22], #1
        mov     w10, #0x7c
        strb    w10, [x22], #1
        mov     x27, #0
chr_loop:
        cmp     x27, x25
        bge     chr_done
        add     x11, x24, x27
        ldrb    w10, [x19, x11]
        sub     w11, w10, #0x20
        cmp     w11, #0x5e
        bls     1f
        mov     w10, #0x2e
1:
        strb    w10, [x22], #1
        add     x27, x27, #1
        b       chr_loop
chr_done:
        mov     w10, #0x7c
        strb    w10, [x22], #1
        mov     w10, #10
        strb    w10, [x22], #1
        add     x24, x24, #16
        b       row_loop
finish:
        mov     x0, x20
        bl      put_hex8
        mov     w10, #10
        strb    w10, [x22], #1
        mov     x1, x21
        sub     x2, x22, x21
write_loop:
        cbz     x2, done
        mov     x0, #1
        mov     x8, #64
        svc     #0
        cmp     x0, #0
        ble     done
        add     x1, x1, x0
        sub     x2, x2, x0
        b       write_loop
done:
        mov     x0, #0
        mov     x8, #94
        svc     #0
        b       done

// x0 = value; writes 8 lowercase hex digits at x22 (advances x22)
put_hex8:
        mov     x9, #28
1:
        lsr     x10, x0, x9
        and     x10, x10, #15
        ldrb    w10, [x23, x10]
        strb    w10, [x22], #1
        subs    x9, x9, #4
        bge     1b
        ret

        .section .rodata
hexdigits:
        .ascii  "0123456789abcdef"

        .bss
        .balign 16
inbuf:
        .space  65536
        .balign 16
outbuf:
        .space  323616
