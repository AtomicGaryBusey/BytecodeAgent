.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65537
    sub edx, ebx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r8, [rsi + rbx]
    lea rdi, [rip + outbuf]
line_loop:
    cmp rsi, r8
    jae process_done
    mov rcx, rsi
scan_loop:
    cmp rcx, r8
    jae scan_done
    cmp byte ptr [rcx], 10
    je scan_done
    inc rcx
    jmp scan_loop
scan_done:
    mov r9, rcx
rev_loop:
    cmp r9, rsi
    jbe rev_done
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev_loop
rev_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rcx + 1]
    jmp line_loop
process_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle exit_prog
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_prog

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 65552
