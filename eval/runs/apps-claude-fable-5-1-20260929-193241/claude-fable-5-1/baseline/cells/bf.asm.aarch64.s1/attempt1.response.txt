.text
.globl _start

_start:
        // read all of standard input into inbuf
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x24, #0
rd_loop:
        mov     x2, #65536
        sub     x2, x2, x24
        cbz     x2, rd_done
        mov     x8, #63
        mov     x0, #0
        add     x1, x19, x24
        svc     #0
        cmp     x0, #0
        ble     rd_done
        add     x24, x24, x0
        b       rd_loop
rd_done:
        // find first '!' : x20 = program length
        mov     x20, #0
find_bang:
        cmp     x20, x24
        beq     no_bang
        ldrb    w0, [x19, x20]
        cmp     w0, #33
        beq     found_bang
        add     x20, x20, #1
        b       find_bang
found_bang:
        add     x23, x19, x20
        add     x23, x23, #1
        b       after_bang
no_bang:
        add     x23, x19, x24
after_bang:
        add     x24, x19, x24           // x24 = end of input data

        // bracket matching, build jump table
        adrp    x25, jumptab
        add     x25, x25, :lo12:jumptab
        adrp    x28, bstack
        add     x28, x28, :lo12:bstack
        mov     x1, #0
        mov     x2, #0
scan:
        cmp     x1, x20
        beq     scan_done
        ldrb    w3, [x19, x1]
        cmp     w3, #91
        bne     not_open
        str     w1, [x28, x2, lsl #2]
        add     x2, x2, #1
        b       scan_next
not_open:
        cmp     w3, #93
        bne     scan_next
        cbz     x2, fail
        sub     x2, x2, #1
        ldr     w4, [x28, x2, lsl #2]
        str     w1, [x25, x4, lsl #2]
        str     w4, [x25, x1, lsl #2]
scan_next:
        add     x1, x1, #1
        b       scan
scan_done:
        cbnz    x2, fail

        // execute
        adrp    x22, tape
        add     x22, x22, :lo12:tape
        adrp    x26, outbuf
        add     x26, x26, :lo12:outbuf
        mov     x27, #0
        mov     x21, #0
run:
        cmp     x21, x20
        bhs     finish
        ldrb    w0, [x19, x21]
        add     x21, x21, #1
        cmp     w0, #62
        beq     c_right
        cmp     w0, #60
        beq     c_left
        cmp     w0, #43
        beq     c_inc
        cmp     w0, #45
        beq     c_dec
        cmp     w0, #46
        beq     c_out
        cmp     w0, #44
        beq     c_in
        cmp     w0, #91
        beq     c_open
        cmp     w0, #93
        beq     c_close
        b       run
c_right:
        add     x22, x22, #1
        b       run
c_left:
        sub     x22, x22, #1
        b       run
c_inc:
        ldrb    w1, [x22]
        add     w1, w1, #1
        strb    w1, [x22]
        b       run
c_dec:
        ldrb    w1, [x22]
        sub     w1, w1, #1
        strb    w1, [x22]
        b       run
c_out:
        ldrb    w1, [x22]
        strb    w1, [x26, x27]
        add     x27, x27, #1
        mov     x2, #65536
        cmp     x27, x2
        bne     run
        bl      flush
        b       run
c_in:
        cmp     x23, x24
        bhs     in_zero
        ldrb    w1, [x23], #1
        strb    w1, [x22]
        b       run
in_zero:
        strb    wzr, [x22]
        b       run
c_open:
        ldrb    w1, [x22]
        cbnz    w1, run
        sub     x1, x21, #1
        ldr     w1, [x25, x1, lsl #2]
        add     x21, x1, #1
        b       run
c_close:
        ldrb    w1, [x22]
        cbz     w1, run
        sub     x1, x21, #1
        ldr     w1, [x25, x1, lsl #2]
        add     x21, x1, #1
        b       run

finish:
        bl      flush
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       finish

fail:
        mov     x0, #1
        mov     x8, #93
        svc     #0
        b       fail

// flush output buffer: writes x27 bytes from x26 to fd 1, resets x27
flush:
        cbz     x27, 1f
        mov     x1, x26
        mov     x2, x27
2:
        mov     x8, #64
        mov     x0, #1
        svc     #0
        cmp     x0, #0
        ble     1f
        add     x1, x1, x0
        subs    x2, x2, x0
        bne     2b
1:
        mov     x27, #0
        ret

.bss
        .balign 16
inbuf:
        .space  65536
        .balign 16
outbuf:
        .space  65536
        .balign 16
tape:
        .space  30016
        .balign 16
jumptab:
        .space  262144
        .balign 16
bstack:
        .space  262144
