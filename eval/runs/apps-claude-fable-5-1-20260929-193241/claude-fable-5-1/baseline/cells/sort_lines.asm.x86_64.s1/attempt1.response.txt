.text
.globl _start

.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 69632
.equ MAXREC, 1100

_start:
    lea rbx, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov edx, INBUF_SIZE
    sub rdx, r12
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # parse lines into recs (ptr, len)
    xor ecx, ecx
    xor edx, edx
    lea rdi, [rip + recs]
    xor r13d, r13d
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rbx + rcx], 10
    jne 1f
    lea rax, [rbx + rdx]
    mov qword ptr [rdi], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r13
    lea rdx, [rcx + 1]
1:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rdx, r12
    jae parse_done
    lea rax, [rbx + rdx]
    mov qword ptr [rdi], rax
    mov rax, r12
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    inc r13
parse_done:
    test r13, r13
    jz do_exit
    mov r14, r13
    lea r12, [rip + recs]
    lea r13, [rip + tmprecs]
    mov r15d, 1
sort_outer:
    cmp r15, r14
    jae sort_done
    mov qword ptr [rip + i_v], 0
pass_loop:
    mov rax, qword ptr [rip + i_v]
    cmp rax, r14
    jae pass_done
    mov rbp, rax
    lea rcx, [rax + r15]
    cmp rcx, r14
    cmova rcx, r14
    mov qword ptr [rip + mid_v], rcx
    mov r10, rcx
    lea rcx, [rax + r15*2]
    cmp rcx, r14
    cmova rcx, r14
    mov qword ptr [rip + right_v], rcx
    mov rbx, rax
    shl rbx, 4
    add rbx, r13
merge_loop:
    cmp rbp, qword ptr [rip + mid_v]
    jae take_right_rest
    cmp r10, qword ptr [rip + right_v]
    jae take_left
    mov rax, rbp
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rdx, qword ptr [r12 + rax + 8]
    mov rax, r10
    shl rax, 4
    mov rdi, qword ptr [r12 + rax]
    mov rcx, qword ptr [r12 + rax + 8]
    call cmp_lines
    test eax, eax
    jg take_right
take_left:
    mov rax, rbp
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc rbp
    add rbx, 16
    jmp merge_loop
take_right:
    mov rax, r10
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc r10
    add rbx, 16
    jmp merge_loop
take_right_rest:
    cmp r10, qword ptr [rip + right_v]
    jae merge_done
    mov rax, r10
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc r10
    add rbx, 16
    jmp take_right_rest
merge_done:
    mov rax, qword ptr [rip + i_v]
    lea rax, [rax + r15*2]
    mov qword ptr [rip + i_v], rax
    jmp pass_loop
pass_done:
    xchg r12, r13
    shl r15, 1
    jmp sort_outer
sort_done:
    # build output
    lea rdi, [rip + outbuf]
    mov rbx, r12
    xor ebp, ebp
out_loop:
    cmp rbp, r14
    jae out_done
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    inc rbp
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

# cmp_lines: rsi=ptr1, rdx=len1, rdi=ptr2, rcx=len2 -> eax <0, 0, >0
cmp_lines:
    mov r8, rdx
    cmp r8, rcx
    cmova r8, rcx
1:
    test r8, r8
    jz 2f
    movzx eax, byte ptr [rsi]
    movzx r9d, byte ptr [rdi]
    sub eax, r9d
    jnz 3f
    inc rsi
    inc rdi
    dec r8
    jmp 1b
2:
    xor eax, eax
    cmp rdx, rcx
    jb 4f
    ja 5f
    ret
4:
    mov eax, -1
    ret
5:
    mov eax, 1
    ret
3:
    ret

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 16
recs:
    .space MAXREC*16
.balign 16
tmprecs:
    .space MAXREC*16
.balign 8
i_v:
    .space 8
mid_v:
    .space 8
right_v:
    .space 8
