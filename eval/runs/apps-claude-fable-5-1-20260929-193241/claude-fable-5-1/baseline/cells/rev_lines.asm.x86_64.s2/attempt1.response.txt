.text
.globl _start
_start:
        xor     ebx, ebx                # total bytes read
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jz      read_done
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     rsi, [rip + inbuf]
        lea     rdi, [rip + outbuf]     # output write pointer
        xor     r8d, r8d                # start of current line
line_loop:
        cmp     r8, rbx
        jae     done_lines
        mov     r9, r8
find_nl:
        cmp     r9, rbx
        jae     found_end
        cmp     byte ptr [rsi + r9], 10
        je      found_end
        inc     r9
        jmp     find_nl
found_end:
        mov     r10, r9
rev_loop:
        cmp     r10, r8
        je      rev_done
        dec     r10
        mov     al, byte ptr [rsi + r10]
        mov     byte ptr [rdi], al
        inc     rdi
        jmp     rev_loop
rev_done:
        mov     byte ptr [rdi], 10
        inc     rdi
        lea     r8, [r9 + 1]
        jmp     line_loop
done_lines:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
write_loop:
        test    rdx, rdx
        jz      do_exit
        mov     eax, 1                  # write
        mov     edi, 1                  # fd 1
        syscall
        test    rax, rax
        jle     do_exit
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
do_exit:
        mov     eax, 231                # exit_group
        xor     edi, edi
        syscall

.bss
.balign 16
inbuf:
        .space  65536 + 64
outbuf:
        .space  65536 + 64
