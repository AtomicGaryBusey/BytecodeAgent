.bss
inbuf:  .space 65600
jmptab: .space 262144
bstack: .space 262144
tape:   .space 30000
outbuf: .space 65536

.text
.globl _start
_start:
    xor ebx, ebx                    # total bytes read
1:  xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65600
    sub edx, ebx
    jle 2f
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    # find first '!'
    lea rsi, [rip + inbuf]
    xor ecx, ecx
3:  cmp rcx, rbx
    jae 4f
    cmp byte ptr [rsi + rcx], 0x21
    je 5f
    inc rcx
    jmp 3b
5:  lea r12, [rsi + rcx + 1]        # input start
    lea r13, [rsi + rbx]            # input end
    jmp 6f
4:  lea r12, [rsi + rbx]
    mov r13, r12
6:
    mov r14, rcx                    # program length
    # bracket check and jump table
    xor ecx, ecx
    xor edx, edx
    lea rdi, [rip + bstack]
    lea r8, [rip + jmptab]
7:  cmp rcx, r14
    jae 9f
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    jne 8f
    mov dword ptr [rdi + rdx*4], ecx
    inc edx
    jmp 10f
8:  cmp al, 0x5d
    jne 10f
    test edx, edx
    jz err
    dec edx
    mov eax, dword ptr [rdi + rdx*4]
    mov dword ptr [r8 + rax*4], ecx
    mov dword ptr [r8 + rcx*4], eax
10: inc rcx
    jmp 7b
9:  test edx, edx
    jnz err

    # run
    xor ecx, ecx
    lea r15, [rip + tape]
    lea r10, [rip + outbuf]
    xor r9d, r9d
run:
    cmp rcx, r14
    jae done
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x2b
    je c_plus
    cmp al, 0x2d
    je c_minus
    cmp al, 0x3e
    je c_right
    cmp al, 0x3c
    je c_left
    cmp al, 0x5b
    je c_open
    cmp al, 0x5d
    je c_close
    cmp al, 0x2e
    je c_out
    cmp al, 0x2c
    je c_in
    jmp run
c_plus:
    inc byte ptr [r15]
    jmp run
c_minus:
    dec byte ptr [r15]
    jmp run
c_right:
    inc r15
    jmp run
c_left:
    dec r15
    jmp run
c_open:
    cmp byte ptr [r15], 0
    jne run
    mov eax, dword ptr [r8 + rcx*4 - 4]
    lea rcx, [rax + 1]
    jmp run
c_close:
    cmp byte ptr [r15], 0
    je run
    mov eax, dword ptr [r8 + rcx*4 - 4]
    lea rcx, [rax + 1]
    jmp run
c_out:
    mov al, byte ptr [r15]
    mov byte ptr [r10 + r9], al
    inc r9
    cmp r9, 65536
    jb run
    call flush
    jmp run
c_in:
    cmp r12, r13
    jae 11f
    mov al, byte ptr [r12]
    inc r12
    mov byte ptr [r15], al
    jmp run
11: mov byte ptr [r15], 0
    jmp run

done:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
err:
    mov eax, 231
    mov edi, 1
    syscall

# flush: write outbuf[0..r9) to stdout, preserves rcx, resets r9
flush:
    push rcx
    push rbx
    xor ebx, ebx                    # bytes written so far
12: cmp rbx, r9
    jae 13f
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r9
    sub rdx, rbx
    syscall
    test rax, rax
    jle 13f
    add rbx, rax
    jmp 12b
13: xor r9d, r9d
    lea rsi, [rip + inbuf]
    pop rbx
    pop rcx
    ret
