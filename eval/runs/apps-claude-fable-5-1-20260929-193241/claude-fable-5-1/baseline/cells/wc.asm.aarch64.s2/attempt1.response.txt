.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // in_word flag
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
scan:
    ldrb w3, [x1], #1
    cmp w3, #0x0a
    cinc x19, x19, eq
    cmp w3, #0x20
    b.eq ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls ws
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
ws:
    mov x22, #0
next:
    cmp x1, x2
    b.lo scan
    b read_loop

done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x1, x24
    mov x0, x19
    bl putnum
    mov w3, #0x20
    strb w3, [x1], #1
    mov x0, x20
    bl putnum
    mov w3, #0x20
    strb w3, [x1], #1
    mov x0, x21
    bl putnum
    mov w3, #0x0a
    strb w3, [x1], #1
    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// putnum: x0 = value, x1 = output pointer (advanced); clobbers x2-x6
putnum:
    sub sp, sp, #32
    add x2, sp, #32
    mov x6, #10
1:
    udiv x3, x0, x6
    msub x4, x3, x6, x0
    add w4, w4, #48
    strb w4, [x2, #-1]!
    mov x0, x3
    cbnz x0, 1b
    add x5, sp, #32
2:
    ldrb w3, [x2], #1
    strb w3, [x1], #1
    cmp x2, x5
    b.lo 2b
    add sp, sp, #32
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 64
