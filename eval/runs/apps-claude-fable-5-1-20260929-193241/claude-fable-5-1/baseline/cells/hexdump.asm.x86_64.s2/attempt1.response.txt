.text
.globl _start
_start:
    xor r12, r12                    # total input length
read_loop:
    mov edx, 65536
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + inbuf]
    xor r13, r13                    # current offset
row_loop:
    mov rcx, r12
    sub rcx, r13
    jz done_rows
    cmp rcx, 16
    jbe 1f
    mov ecx, 16
1:
    mov eax, r13d
    call put_hex32
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor r8, r8
    lea r9, [rsi + r13]
2:
    cmp r8, rcx
    jae 3f
    movzx eax, byte ptr [r9 + r8]
    call put_hex8
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp 4f
3:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
4:
    cmp r8, 7
    jne 5f
    mov byte ptr [rdi], 0x20
    inc rdi
5:
    inc r8
    cmp r8, 16
    jb 2b
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor r8, r8
6:
    movzx eax, byte ptr [r9 + r8]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 7f
    mov al, 0x2e
7:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, rcx
    jb 6b
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r13, rcx
    jmp row_loop
done_rows:
    mov eax, r12d
    call put_hex32
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall

put_hex32:
    mov r10d, eax
    mov r11d, 8
1:
    rol r10d, 4
    mov eax, r10d
    and eax, 15
    mov al, byte ptr [hexdig + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec r11d
    jnz 1b
    ret

put_hex8:
    mov r10d, eax
    shr eax, 4
    mov al, byte ptr [hexdig + rax]
    mov byte ptr [rdi], al
    mov eax, r10d
    and eax, 15
    mov al, byte ptr [hexdig + rax]
    mov byte ptr [rdi + 1], al
    add rdi, 2
    ret

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 331776
