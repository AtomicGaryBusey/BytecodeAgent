.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cbnz x2, read_call
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output
    add x21, x21, :lo12:output
    mov x22, x21
    adrp x23, digits
    add x23, x23, :lo12:digits
    mov x24, #0

row_loop:
    cmp x24, x20
    b.hs final_offset
    mov x0, x24
    bl emit_offset
    mov w3, #32
    strb w3, [x22], #1
    strb w3, [x22], #1
    sub x25, x20, x24
    mov x4, #16
    cmp x25, x4
    csel x25, x25, x4, lo
    add x26, x19, x24
    mov x27, #0

hex_loop:
    cmp x27, x25
    b.hs blank_slot
    ldrb w4, [x26, x27]
    lsr w5, w4, #4
    ldrb w5, [x23, x5]
    strb w5, [x22], #1
    and w4, w4, #15
    ldrb w5, [x23, x4]
    strb w5, [x22], #1
    b slot_end
blank_slot:
    strb w3, [x22], #1
    strb w3, [x22], #1
slot_end:
    strb w3, [x22], #1
    cmp x27, #7
    b.ne next_slot
    strb w3, [x22], #1
next_slot:
    add x27, x27, #1
    cmp x27, #16
    b.lo hex_loop

    strb w3, [x22], #1
    mov w4, #124
    strb w4, [x22], #1
    mov x27, #0
ascii_loop:
    ldrb w4, [x26, x27]
    sub w5, w4, #32
    cmp w5, #94
    mov w6, #46
    csel w4, w4, w6, ls
    strb w4, [x22], #1
    add x27, x27, #1
    cmp x27, x25
    b.lo ascii_loop
    mov w4, #124
    strb w4, [x22], #1
    mov w4, #10
    strb w4, [x22], #1
    add x24, x24, x25
    b row_loop

final_offset:
    mov x0, x20
    bl emit_offset
    mov w4, #10
    strb w4, [x22], #1
    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

emit_offset:
    mov x9, #28
offset_digit:
    lsr x10, x0, x9
    and x10, x10, #15
    ldrb w11, [x23, x10]
    strb w11, [x22], #1
    subs x9, x9, #4
    b.ge offset_digit
    ret

.section .rodata
