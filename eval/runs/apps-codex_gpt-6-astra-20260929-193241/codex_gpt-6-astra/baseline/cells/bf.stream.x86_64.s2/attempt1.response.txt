.globl _start
.bss
input:
.zero 65537
program:
.zero 65536
.p2align 2
jumps:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input]
    lea rdi, [rip + program]
    lea r8, [rip + jumps]
    lea r9, [rip + brackets]
    xor ebx, ebx
    xor r13d, r13d
    xor r10d, r10d
parse_loop:
    cmp rbx, r12
    jae parse_done
    movzx eax, byte ptr [rsi + rbx]
    inc rbx
    cmp al, 33
    je parse_done
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne parse_loop
save_command:
    mov byte ptr [rdi + r13], al
    inc r13
    jmp parse_loop

open_bracket:
    mov dword ptr [r9 + r10*4], r13d
    inc r10d
    jmp save_command

close_bracket:
    test r10d, r10d
    jz invalid
    dec r10d
    mov edx, dword ptr [r9 + r10*4]
    lea ecx, [r13 + 1]
    mov dword ptr [r8 + rdx*4], ecx
    inc edx
    mov dword ptr [r8 + r13*4], edx
    jmp save_command

parse_done:
    test r10d, r10d
    jnz invalid
    lea r14, [rip + tape]
    lea r15, [rip + output]
    xor ebp, ebp
    xor r9d, r9d
execute:
    cmp r9, r13
    jae finish
    movzx eax, byte ptr [rdi + r9]
    inc r9
    cmp al, 43
    je plus
    cmp al, 45
    je minus
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 46
    je emit
    cmp al, 44
    je consume
    cmp al, 91
    je loop_open
    cmp byte ptr [r14], 0
    je execute
    mov r9d, dword ptr [r8 + r9*4 - 4]
    jmp execute
loop_open:
    cmp byte ptr [r14], 0
    jne execute
    mov r9d, dword ptr [r8 + r9*4 - 4]
    jmp execute
plus:
    inc byte ptr [r14]
    jmp execute
minus:
    dec byte ptr [r14]
    jmp execute
right:
    inc r14
    jmp execute
left:
    dec r14
    jmp execute
emit:
    mov al, byte ptr [r14]
    mov byte ptr [r15 + rbp], al
    inc ebp
    jmp execute
consume:
    xor eax, eax
    cmp rbx, r12
    jae store_input
    mov al, byte ptr [rsi + rbx]
    inc rbx
store_input:
    mov byte ptr [r14], al
    jmp execute

finish:
    mov edx, ebp
    mov rsi, r15
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall
