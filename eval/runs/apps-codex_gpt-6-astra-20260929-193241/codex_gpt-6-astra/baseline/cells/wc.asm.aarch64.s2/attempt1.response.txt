.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, input
    add x23, x23, :lo12:input

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, output_counts
    add x21, x21, x0
    mov x3, x23
    mov x4, x0

count_loop:
    ldrb w5, [x3], #1
    cmp w5, #10
    cinc x19, x19, eq
    cmp w5, #32
    b.eq whitespace
    sub w6, w5, #9
    cmp w6, #4
    b.ls whitespace
    cmp w22, #0
    cinc x20, x20, eq
    mov w22, #1
    b next_byte
whitespace:
    mov w22, #0
next_byte:
    subs x4, x4, #1
    b.ne count_loop
    b read_loop

output_counts:
    adrp x24, output
    add x24, x24, :lo12:output
    mov x25, x24
    mov x0, x19
    bl decimal
    mov w1, #32
    strb w1, [x24], #1
    mov x0, x20
    bl decimal
    mov w1, #32
    strb w1, [x24], #1
    mov x0, x21
    bl decimal
    mov w1, #10
    strb w1, [x24], #1

    mov x0, #1
    mov x1, x25
    sub x2, x24, x25
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

decimal:
    sub sp, sp, #32
    mov x9, sp
    mov x10, #10
decimal_digits:
    udiv x11, x0, x10
    msub x12, x11, x10, x0
    add w12, w12, #48
    strb w12, [x9], #1
    mov x0, x11
    cbnz x0, decimal_digits
    mov x13, sp
decimal_copy:
    ldrb w12, [x9, #-1]!
    strb w12, [x24], #1
    cmp x9, x13
    b.ne decimal_copy
    add sp, sp, #32
    ret

.bss
.balign 16
input:
    .zero 4096
output:
    .zero 64
