.globl _start

.bss
.balign 16
input_buffer:
.space 69632
value_stack:
.space 4096
output_buffer:
.space 32

.section .rodata
error_text:
.ascii "error\n"

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, x19

read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, value_stack
    add x21, x21, :lo12:value_stack

line_loop:
    cmp x19, x20
    b.eq finish
    mov x24, x19

find_line_end:
    cmp x24, x20
    b.eq final_line
    ldrb w0, [x24]
    cmp w0, #10
    b.eq found_newline
    add x24, x24, #1
    b find_line_end

found_newline:
    add x27, x24, #1
    b begin_line

final_line:
    mov x27, x24

begin_line:
    mov x22, #0

token_loop:
    cmp x19, x24
    b.eq line_result
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator
    mov x25, x19

find_token_end:
    add x19, x19, #1
    cmp x19, x24
    b.eq token_found
    ldrb w0, [x19]
    cmp w0, #32
    b.eq token_found
    cmp w0, #9
    b.ne find_token_end

token_found:
    mov x26, x19
    sub x0, x26, x25
    cmp x0, #1
    b.ne parse_literal
    ldrb w3, [x25]
    cmp w3, #43
    b.eq operator
    cmp w3, #45
    b.eq operator
    cmp w3, #42
    b.eq operator
    cmp w3, #47
    b.eq operator
    cmp w3, #37
    b.eq operator

parse_literal:
    mov x1, x25
    mov x2, #0
    ldrb w0, [x1]
    cmp w0, #45
    b.ne literal_digits
    mov x2, #1
    add x1, x1, #1

literal_digits:
    sub x0, x26, x1
    cbz x0, line_error
    cmp x0, #18
    b.hi line_error
    mov x4, #0
    mov x5, #10

digit_loop:
    ldrb w0, [x1], #1
    sub w0, w0, #48
    cmp w0, #9
    b.hi line_error
    madd x4, x4, x5, x0
    cmp x1, x26
    b.ne digit_loop
    cmp x2, #0
    cneg x4, x4, ne
    str x4, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop

skip_separator:
    add x19, x19, #1
    b token_loop

operator:
    cmp x22, #2
    b.lo line_error
    sub x22, x22, #1
    ldr x1, [x21, x22, lsl #3]
    sub x22, x22, #1
    ldr x0, [x21, x22, lsl #3]
    cmp w3, #43
    b.eq op_add
    cmp w3, #45
    b.eq op_sub
    cmp w3, #42
    b.eq op_mul
    cbz x1, line_error
    mov x2, #0x8000000000000000
    cmp x0, x2
    b.ne divide
    cmn x1, #1
    b.eq line_error

divide:
    sdiv x4, x0, x1
    cmp w3, #47
    b.eq push_result
    msub x4, x4, x1, x0
    b push_result

op_add:
    add x4, x0, x1
    b push_result

op_sub:
    sub x4, x0, x1
    b push_result

op_mul:
    mul x4, x0, x1

push_result:
    str x4, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop

line_result:
    cmp x22, #1
    b.ne line_error
    ldr x0, [x21]
    adrp x6, output_buffer
    add x6, x6, :lo12:output_buffer
    add x6, x6, #32
    sub x1, x6, #1
    mov w2, #10
    strb w2, [x1]
    cmp x0, #0
    cset x7, lt
    cneg x0, x0, lt
    mov x5, #10

format_digits:
    udiv x3, x0, x5
    msub x4, x3, x5, x0
    add w4, w4, #48
    strb w4, [x1, #-1]!
    mov x0, x3
    cbnz x0, format_digits
    cbz x7, number_ready
    mov w4, #45
    strb w4, [x1, #-1]!

number_ready:
    sub x2, x6, x1
    b write_line

line_error:
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6

write_line:
    mov x0, #1
    mov x8, #64
    svc #0
    mov x19, x27
    b line_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0
