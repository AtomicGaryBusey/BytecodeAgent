.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    call whitespace
    call value
    call whitespace
    cmp rbx, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

peek:
    cmp rbx, r13
    jae error
    movzx eax, byte ptr [r12 + rbx]
    ret

whitespace:
    cmp rbx, r13
    jae ws_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 32
    je ws_next
    cmp al, 9
    je ws_next
    cmp al, 10
    je ws_next
    cmp al, 13
    jne ws_done
ws_next:
    inc rbx
    jmp whitespace
ws_done:
    ret

value:
    call peek
    cmp al, 34
    je string
    cmp al, 123
    je object
    cmp al, 91
    je array
    cmp al, 116
    je true_value
    cmp al, 102
    je false_value
    cmp al, 110
    je null_value
    cmp al, 45
    je number
    cmp al, 48
    jb error
    cmp al, 57
    ja error
    jmp number

true_value:
    lea rsi, [rip + true_text]
    mov ecx, 4
    jmp literal
false_value:
    lea rsi, [rip + false_text]
    mov ecx, 5
    jmp literal
null_value:
    lea rsi, [rip + null_text]
    mov ecx, 4
literal:
    call peek
    cmp al, byte ptr [rsi]
    jne error
    inc rbx
    inc rsi
    dec ecx
    jnz literal
    ret

string:
    inc rbx
string_loop:
    call peek
    cmp al, 34
    je string_done
    cmp al, 32
    jb error
    cmp al, 92
    je escape
    inc rbx
    jmp string_loop
string_done:
    inc rbx
    ret
escape:
    inc rbx
    call peek
    cmp al, 117
    je unicode
    cmp al, 34
    je escape_done
    cmp al, 92
    je escape_done
    cmp al, 47
    je escape_done
    cmp al, 98
    je escape_done
    cmp al, 102
    je escape_done
    cmp al, 110
    je escape_done
    cmp al, 114
    je escape_done
    cmp al, 116
    jne error
escape_done:
    inc rbx
    jmp string_loop
unicode:
    inc rbx
    mov ecx, 4
hex_loop:
    call peek
    cmp al, 48
    jb error
    cmp al, 57
    jbe hex_good
    or al, 32
    cmp al, 97
    jb error
    cmp al, 102
    ja error
hex_good:
    inc rbx
    dec ecx
    jnz hex_loop
    jmp string_loop

number:
    cmp al, 45
    jne integer
    inc rbx
    call peek
integer:
    cmp al, 48
    jb error
    cmp al, 57
    ja error
    inc rbx
    cmp al, 48
    je fraction
integer_loop:
    cmp rbx, r13
    jae number_done
    movzx eax, byte ptr [r12 + rbx]
    sub eax, 48
    cmp eax, 9
    ja fraction
    inc rbx
    jmp integer_loop
fraction:
    cmp rbx, r13
    jae number_done
    cmp byte ptr [r12 + rbx], 46
    jne exponent
    inc rbx
    call required_digit
fraction_loop:
    cmp rbx, r13
    jae number_done
    movzx eax, byte ptr [r12 + rbx]
    sub eax, 48
    cmp eax, 9
    ja exponent
    inc rbx
    jmp fraction_loop
exponent:
    cmp rbx, r13
    jae number_done
    mov al, byte ptr [r12 + rbx]
    or al, 32
    cmp al, 101
    jne number_done
    inc rbx
    call peek
    cmp al, 43
    je exp_sign
    cmp al, 45
    jne exp_digit
exp_sign:
    inc rbx
exp_digit:
    call required_digit
exp_loop:
    cmp rbx, r13
    jae number_done
    movzx eax, byte ptr [r12 + rbx]
    sub eax, 48
    cmp eax, 9
    ja number_done
    inc rbx
    jmp exp_loop
number_done:
    ret
required_digit:
    call peek
    sub eax, 48
    cmp eax, 9
    ja error
    inc rbx
    ret

array:
    inc rbx
    call whitespace
    call peek
    cmp al, 93
    je container_done
array_value:
    call value
    call whitespace
    call peek
    cmp al, 93
    je container_done
    cmp al, 44
    jne error
    inc rbx
    call whitespace
    jmp array_value

object:
    inc rbx
    call whitespace
    call peek
    cmp al, 125
    je container_done
object_member:
    call peek
    cmp al, 34
    jne error
    call string
    call whitespace
    call peek
    cmp al, 58
    jne error
    inc rbx
    call whitespace
    call value
    call whitespace
    call peek
    cmp al, 125
    je container_done
    cmp al, 44
    jne error
    inc rbx
    call whitespace
    jmp object_member
container_done:
    inc rbx
    ret

error:
    lea rsi, [rip + output_end]
    dec rsi
    mov byte ptr [rsi], 10
    mov rax, rbx
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    sub rsi, 9
    mov rax, 0x746120726f727265
    mov qword ptr [rsi], rax
    mov byte ptr [rsi + 8], 32
    lea rdx, [rip + output_end]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
