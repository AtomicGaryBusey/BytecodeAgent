.globl _start

.bss
.balign 16
input:
.zero 32770
records:
.zero 16000
output:
.zero 33768

.text
_start:
    lea r14, [rip + input]
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    mov edx, 32769
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop

read_done:
    test r15, r15
    jz exit_ok
    cmp byte ptr [r14 + r15 - 1], 10
    je parse_start
    mov byte ptr [r14 + r15], 10
    inc r15

parse_start:
    lea rbx, [rip + records]
    xor r13d, r13d
    xor eax, eax
    xor edx, edx

parse_loop:
    cmp byte ptr [r14 + rax], 10
    jne parse_next
    mov rcx, r13
    shl rcx, 4
    lea rsi, [r14 + rdx]
    mov qword ptr [rbx + rcx], rsi
    mov rsi, rax
    sub rsi, rdx
    mov qword ptr [rbx + rcx + 8], rsi
    inc r13
    lea rdx, [rax + 1]
parse_next:
    inc rax
    cmp rax, r15
    jb parse_loop

    mov r12d, 1
sort_outer:
    cmp r12, r13
    jae build_output
    mov r10, r12
    shl r10, 4
    mov r8, qword ptr [rbx + r10]
    mov r9, qword ptr [rbx + r10 + 8]

sort_inner:
    test r10, r10
    jz sort_place
    mov rdi, qword ptr [rbx + r10 - 16]
    mov rdx, qword ptr [rbx + r10 - 8]
    mov rsi, r8
    mov rcx, r9
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    jb sort_shift
    ja sort_place
compare_lengths:
    cmp r9, rdx
    jae sort_place
sort_shift:
    mov rax, qword ptr [rbx + r10 - 16]
    mov qword ptr [rbx + r10], rax
    mov qword ptr [rbx + r10 + 8], rdx
    sub r10, 16
    jmp sort_inner
sort_place:
    mov qword ptr [rbx + r10], r8
    mov qword ptr [rbx + r10 + 8], r9
    inc r12
    jmp sort_outer

build_output:
    lea rdi, [rip + output]
    xor r12d, r12d
copy_loop:
    mov rax, r12
    shl rax, 4
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r12
    cmp r12, r13
    jb copy_loop

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
