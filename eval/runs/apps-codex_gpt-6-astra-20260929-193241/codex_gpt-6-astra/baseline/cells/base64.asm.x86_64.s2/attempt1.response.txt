.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49153
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    lea rdi, [rip + output]
    mov r14, rdi
    lea rbx, [rip + alphabet]
    mov rsi, r12
    xor r8d, r8d
    test r13, r13
    jz finish

encode_loop:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov r9d, 1
    cmp r13, 1
    je encode_group
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    mov r9d, 2
    cmp r13, 2
    je encode_group
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    mov r9d, 3

encode_group:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl
    mov word ptr [rdi + 2], 0x3d3d
    cmp r9d, 1
    je group_done
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl
    cmp r9d, 2
    je group_done
    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], al

group_done:
    add rdi, 4
    add r8d, 4
    add rsi, r9
    sub r13, r9
    cmp r8d, 76
    jne check_remaining
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
check_remaining:
    test r13, r13
    jnz encode_loop
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    mov rdx, rdi
    sub rdx, r14
    mov rsi, r14
    mov edi, 1
    mov eax, 1
    syscall
finish:
    xor edi, edi
    mov eax, 60
    syscall
