.globl _start

.bss
.balign 16
raw:
.zero 65537
code:
.zero 65536
.balign 8
jumps:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    lea r15, [rip + raw]
    xor r9d, r9d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r9]
    mov edx, 65537
    sub rdx, r9
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_more

read_done:
    lea r12, [rip + code]
    lea r13, [rip + jumps]
    lea r14, [rip + tape]
    xor r8d, r8d
    xor ebx, ebx
    xor edx, edx
parse:
    cmp r8, r9
    jae parsed
    movzx eax, byte ptr [r15 + r8]
    inc r8
    cmp al, 33
    je parsed
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne parse
save_command:
    mov byte ptr [r12 + rbx], al
    inc ebx
    jmp parse

open_bracket:
    mov dword ptr [brackets + rdx*4], ebx
    inc edx
    jmp save_command

close_bracket:
    test edx, edx
    jz invalid
    dec edx
    mov ecx, dword ptr [brackets + rdx*4]
    lea esi, [rbx + 1]
    mov dword ptr [r13 + rcx*4], esi
    inc ecx
    mov dword ptr [r13 + rbx*4], ecx
    jmp save_command

parsed:
    test edx, edx
    jnz invalid
    mov edi, ebx
    xor ebx, ebx
    xor ebp, ebp
    xor r10d, r10d

execute:
    cmp ebx, edi
    jae finished
    movzx eax, byte ptr [r12 + rbx]
    inc ebx
    cmp al, 43
    je cell_inc
    cmp al, 45
    je cell_dec
    cmp al, 62
    je pointer_inc
    cmp al, 60
    je pointer_dec
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close
    cmp al, 46
    je emit

    xor eax, eax
    cmp r8, r9
    jae store_input
    mov al, byte ptr [r15 + r8]
    inc r8
store_input:
    mov byte ptr [r14 + rbp], al
    jmp execute

cell_inc:
    inc byte ptr [r14 + rbp]
    jmp execute
cell_dec:
    dec byte ptr [r14 + rbp]
    jmp execute
pointer_inc:
    inc ebp
    jmp execute
pointer_dec:
    dec ebp
    jmp execute
loop_open:
    cmp byte ptr [r14 + rbp], 0
    jne execute
    mov ebx, dword ptr [r13 + rbx*4 - 4]
    jmp execute
loop_close:
    cmp byte ptr [r14 + rbp], 0
    je execute
    mov ebx, dword ptr [r13 + rbx*4 - 4]
    jmp execute
emit:
    mov al, byte ptr [r14 + rbp]
    mov byte ptr [output + r10], al
    inc r10d
    jmp execute

finished:
    test r10d, r10d
    jz success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov edx, r10d
    syscall
success:
    xor edi, edi
    mov eax, 60
    syscall

invalid:
    mov edi, 1
    mov eax, 60
    syscall
