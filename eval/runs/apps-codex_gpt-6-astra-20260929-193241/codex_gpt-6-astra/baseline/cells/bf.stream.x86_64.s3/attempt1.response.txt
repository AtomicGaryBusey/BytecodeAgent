.globl _start
.text
_start:
    xor r14d, r14d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, r14
    mov edx, 65537
    sub rdx, r14
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    jmp read_more

read_done:
    xor esi, esi
    xor r12d, r12d
    xor r8d, r8d
    mov r13, r14
parse:
    cmp rsi, r14
    jae parsed
    movzx eax, byte ptr [source + rsi]
    inc rsi
    cmp al, 33
    je separator
    cmp al, 62
    je compile_right
    cmp al, 60
    je compile_left
    cmp al, 43
    je compile_plus
    cmp al, 45
    je compile_minus
    cmp al, 46
    je compile_output
    cmp al, 44
    je compile_input
    cmp al, 91
    je compile_open
    cmp al, 93
    je compile_close
    jmp parse

compile_right:
    xor eax, eax
    jmp store_command
compile_left:
    mov eax, 1
    jmp store_command
compile_plus:
    mov eax, 2
    jmp store_command
compile_minus:
    mov eax, 3
    jmp store_command
compile_output:
    mov eax, 4
    jmp store_command
compile_input:
    mov eax, 5
    jmp store_command
compile_open:
    mov dword ptr [brackets + r8*4], r12d
    inc r8d
    mov eax, 6
    jmp store_command
compile_close:
    test r8d, r8d
    jz invalid
    dec r8d
    mov edx, dword ptr [brackets + r8*4]
    lea eax, [r12 + 1]
    mov dword ptr [jumps + rdx*4], eax
    inc edx
    mov dword ptr [jumps + r12*4], edx
    mov eax, 7
store_command:
    mov byte ptr [commands + r12], al
    inc r12d
    jmp parse

separator:
    mov r13, rsi
parsed:
    test r8d, r8d
    jnz invalid
    xor ebx, ebx
    xor ebp, ebp
    xor r15d, r15d
execute:
    cmp rbx, r12
    jae finished
    movzx eax, byte ptr [commands + rbx]
    inc ebx
    jmp qword ptr [dispatch + rax*8]

op_right:
    inc ebp
    jmp execute
op_left:
    dec ebp
    jmp execute
op_plus:
    inc byte ptr [tape + rbp]
    jmp execute
op_minus:
    dec byte ptr [tape + rbp]
    jmp execute
op_output:
    mov al, byte ptr [tape + rbp]
    mov byte ptr [output + r15], al
    inc r15d
    jmp execute
op_input:
    xor eax, eax
    cmp r13, r14
    jae input_store
    mov al, byte ptr [source + r13]
    inc r13
input_store:
    mov byte ptr [tape + rbp], al
    jmp execute
op_open:
    cmp byte ptr [tape + rbp], 0
    jne execute
    mov ebx, dword ptr [jumps + rbx*4 - 4]
    jmp execute
op_close:
    cmp byte ptr [tape + rbp], 0
    je execute
    mov ebx, dword ptr [jumps + rbx*4 - 4]
    jmp execute

finished:
    test r15d, r15d
    jz success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, r15
    syscall
success:
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
.balign 8
dispatch:
    .quad op_right, op_left, op_plus, op_minus
    .quad op_output, op_input, op_open, op_close

.bss
source:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
brackets:
    .zero 4000
tape:
    .zero 30000
output:
    .zero 65536
