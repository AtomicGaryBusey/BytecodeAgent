.text
.globl _start
_start:
    adrp x22, input
    add x22, x22, :lo12:input
    adrp x23, output
    add x23, x23, :lo12:output
    adrp x24, digits
    add x24, x24, :lo12:digits
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x22, x19
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0
    mov x21, x23

row_loop:
    cmp x20, x19
    b.hs final_offset
    sub x25, x19, x20
    mov x0, #16
    cmp x25, x0
    csel x25, x25, x0, lo

    mov x0, x20
    mov x1, x21
    bl hex_offset
    mov x21, x1
    mov w5, #32
    strb w5, [x21], #1
    strb w5, [x21], #1
    add x26, x22, x20
    mov x27, #0

hex_slots:
    cmp x27, x25
    b.hs blank_slot
    ldrb w6, [x26, x27]
    lsr w7, w6, #4
    ldrb w7, [x24, x7]
    strb w7, [x21], #1
    and w6, w6, #15
    ldrb w6, [x24, x6]
    strb w6, [x21], #1
    strb w5, [x21], #1
    b slot_done

blank_slot:
    strb w5, [x21], #1
    strb w5, [x21], #1
    strb w5, [x21], #1

slot_done:
    cmp x27, #7
    b.ne no_extra_space
    strb w5, [x21], #1
no_extra_space:
    add x27, x27, #1
    cmp x27, #16
    b.lo hex_slots

    strb w5, [x21], #1
    mov w6, #124
    strb w6, [x21], #1
    mov x27, #0

ascii_loop:
    ldrb w6, [x26, x27]
    sub w7, w6, #32
    cmp w7, #94
    mov w9, #46
    csel w6, w6, w9, ls
    strb w6, [x21], #1
    add x27, x27, #1
    cmp x27, x25
    b.lo ascii_loop

    mov w6, #124
    strb w6, [x21], #1
    mov w6, #10
    strb w6, [x21], #1
    add x20, x20, x25
    b row_loop

final_offset:
    mov x0, x19
    mov x1, x21
    bl hex_offset
    mov w6, #10
    strb w6, [x1], #1
    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

hex_offset:
    mov x2, #28
hex_offset_loop:
    lsr x3, x0, x2
    and x3, x3, #15
    ldrb w4, [x24, x3]
    strb w4, [x1], #1
    subs x2, x2, #4
    b.ge hex_offset_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 327680
