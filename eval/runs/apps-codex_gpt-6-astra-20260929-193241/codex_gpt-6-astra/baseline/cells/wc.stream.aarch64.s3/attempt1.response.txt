.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, input
    add x23, x23, :lo12:input

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, print_counts
    add x21, x21, x0
    mov x1, x23
    mov x2, x0

count_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace
    cbnz w22, next_byte
    add x20, x20, #1
    mov w22, #1
    b next_byte

whitespace:
    mov w22, #0

next_byte:
    subs x2, x2, #1
    b.ne count_loop
    b read_loop

print_counts:
    adrp x24, output
    add x24, x24, :lo12:output
    mov x25, x24
    mov x0, x19
    bl decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x20
    bl decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x21
    bl decimal
    mov w3, #10
    strb w3, [x25], #1

    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

decimal:
    sub sp, sp, #32
    mov x4, sp
    mov x5, #10
    mov x6, #0
decimal_digits:
    udiv x1, x0, x5
    msub x2, x1, x5, x0
    add w2, w2, #48
    strb w2, [x4], #1
    add x6, x6, #1
    mov x0, x1
    cbnz x0, decimal_digits
decimal_copy:
    ldrb w2, [x4, #-1]!
    strb w2, [x25], #1
    subs x6, x6, #1
    b.ne decimal_copy
    add sp, sp, #32
    ret

.bss
.balign 16
input:
    .zero 4096
output:
    .zero 64
