.bss
input:
.zero 65537
output:
.zero 323593

.section .rodata
digits:
.ascii "0123456789abcdef"

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output]
    lea rbx, [rip + digits]
    xor r14d, r14d

row_loop:
    cmp r14, r13
    jae final_offset
    mov eax, r14d
    call put_offset
    mov ax, 0x2020
    stosw

    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe row_size_ready
    mov r15d, 16
row_size_ready:
    xor r8d, r8d
hex_loop:
    cmp r8, r15
    jae blank_slot
    lea rsi, [r12 + r14]
    movzx edx, byte ptr [rsi + r8]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [rbx + rax]
    stosb
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done
blank_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb
slot_done:
    cmp r8d, 7
    jne no_extra_space
    mov al, 32
    stosb
no_extra_space:
    inc r8d
    cmp r8d, 16
    jb hex_loop

    mov ax, 0x7c20
    stosw
    lea rsi, [r12 + r14]
    xor r8d, r8d
ascii_loop:
    mov al, byte ptr [rsi + r8]
    cmp al, 32
    jb nonprintable
    cmp al, 126
    jbe emit_ascii
nonprintable:
    mov al, 46
emit_ascii:
    stosb
    inc r8d
    cmp r8, r15
    jb ascii_loop
    mov ax, 0x0a7c
    stosw
    add r14, r15
    jmp row_loop

final_offset:
    mov eax, r13d
    call put_offset
    mov al, 10
    stosb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

put_offset:
    mov edx, eax
    mov ecx, 8
offset_digit:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [rbx + rax]
    stosb
    dec ecx
    jnz offset_digit
    ret
