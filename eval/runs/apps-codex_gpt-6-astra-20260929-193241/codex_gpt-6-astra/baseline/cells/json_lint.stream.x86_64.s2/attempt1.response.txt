.bss
buffer:
.zero 65537
outbuf:
.zero 32

.section .rodata
okmsg:
.ascii "ok\n"
errmsg:
.ascii "error at "
trueword:
.asciz "true"
falseword:
.asciz "false"
nullword:
.asciz "null"

.text
.globl _start
_start:
    lea r12, [rip + buffer]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14d, r14d
    call value
    call whitespace
    cmp r14, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

peek:
    cmp r14, r13
    jae eof
    movzx eax, byte ptr [r12 + r14]
    ret
eof:
    mov eax, 256
    ret

whitespace:
    call peek
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    je ws_next
    ret
ws_next:
    inc r14
    jmp whitespace

value:
    call whitespace
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je literal_true
    cmp eax, 102
    je literal_false
    cmp eax, 110
    je literal_null
    cmp eax, 45
    je number
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe number
    jmp error

literal_true:
    lea rbx, [rip + trueword]
    jmp literal
literal_false:
    lea rbx, [rip + falseword]
    jmp literal
literal_null:
    lea rbx, [rip + nullword]
literal:
    movzx edx, byte ptr [rbx]
    test edx, edx
    jz literal_done
    call peek
    cmp eax, edx
    jne error
    inc r14
    inc rbx
    jmp literal
literal_done:
    ret

string:
    inc r14
string_loop:
    call peek
    cmp eax, 256
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_done
    cmp eax, 92
    je escape
    inc r14
    jmp string_loop
string_done:
    inc r14
    ret
escape:
    inc r14
    call peek
    cmp eax, 34
    je escaped
    cmp eax, 92
    je escaped
    cmp eax, 47
    je escaped
    cmp eax, 98
    je escaped
    cmp eax, 102
    je escaped
    cmp eax, 110
    je escaped
    cmp eax, 114
    je escaped
    cmp eax, 116
    je escaped
    cmp eax, 117
    jne error
    inc r14
    mov edx, 4
hex_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_ok
    or eax, 32
    cmp eax, 97
    jb error
    cmp eax, 102
    ja error
hex_ok:
    inc r14
    dec edx
    jnz hex_loop
    jmp string_loop
escaped:
    inc r14
    jmp string_loop

number:
    cmp eax, 45
    jne integer
    inc r14
    call peek
integer:
    cmp eax, 48
    je zero_integer
    cmp eax, 49
    jb error
    cmp eax, 57
    ja error
integer_digits:
    inc r14
    call peek
    cmp eax, 48
    jb fraction
    cmp eax, 57
    jbe integer_digits
    jmp fraction
zero_integer:
    inc r14
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r14
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
fraction_digits:
    inc r14
    call peek
    cmp eax, 48
    jb exponent
    cmp eax, 57
    jbe fraction_digits
exponent:
    cmp eax, 101
    je exponent_start
    cmp eax, 69
    jne number_done
exponent_start:
    inc r14
    call peek
    cmp eax, 43
    je exponent_sign
    cmp eax, 45
    jne exponent_first
exponent_sign:
    inc r14
    call peek
exponent_first:
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
exponent_digits:
    inc r14
    call peek
    cmp eax, 48
    jb number_done
    cmp eax, 57
    jbe exponent_digits
number_done:
    ret

array:
    inc r14
    call whitespace
    cmp eax, 93
    je container_done
array_value:
    call value
    call whitespace
    cmp eax, 93
    je container_done
    cmp eax, 44
    jne error
    inc r14
    jmp array_value

object:
    inc r14
    call whitespace
    cmp eax, 125
    je container_done
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r14
    call value
    call whitespace
    cmp eax, 125
    je container_done
    cmp eax, 44
    jne error
    inc r14
    call whitespace
    jmp object_member
container_done:
    inc r14
    ret

error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 9
    syscall
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov rax, r14
    mov ebx, 10
    mov ecx, 1
decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz decimal_loop
    mov edx, ecx
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
