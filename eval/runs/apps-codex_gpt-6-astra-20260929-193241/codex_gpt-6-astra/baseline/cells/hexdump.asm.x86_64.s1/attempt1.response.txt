.globl _start

.section .rodata
digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input:
.zero 65537
output:
.zero 324000

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output]
    lea rbx, [rip + digits]
    lea r15, [rip + input]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16
row_size_ready:
    mov eax, r13d
    call print_offset
    mov ax, 0x2020
    stosw
    xor r8d, r8d

hex_loop:
    cmp r8, r14
    jae empty_slot
    lea rsi, [r15 + r13]
    movzx edx, byte ptr [rsi + r8]
    mov ecx, edx
    shr ecx, 4
    mov al, byte ptr [rbx + rcx]
    stosb
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done

empty_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb

slot_done:
    cmp r8d, 7
    jne no_separator
    mov al, 32
    stosb
no_separator:
    inc r8d
    cmp r8d, 16
    jb hex_loop

    mov ax, 0x7c20
    stosw
    xor r8d, r8d
    lea rsi, [r15 + r13]
ascii_loop:
    mov al, byte ptr [rsi + r8]
    cmp al, 32
    jb replace_char
    cmp al, 126
    jbe emit_char
replace_char:
    mov al, 46
emit_char:
    stosb
    inc r8d
    cmp r8, r14
    jb ascii_loop
    mov ax, 0x0a7c
    stosw
    add r13, r14
    jmp row_loop

final_offset:
    mov eax, r12d
    call print_offset
    mov al, 10
    stosb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_offset:
    mov r9d, eax
    mov r10d, 8
offset_digit:
    rol r9d, 4
    mov edx, r9d
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    dec r10d
    jnz offset_digit
    ret
