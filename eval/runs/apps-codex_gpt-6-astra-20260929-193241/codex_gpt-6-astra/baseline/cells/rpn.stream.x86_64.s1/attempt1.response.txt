.globl _start
.bss
.align 8
input:
.zero 65537
values:
.zero 8000
number:
.zero 32

.section .rodata
error_text:
.ascii "error\n"

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more
read_done:
    add r13, r12
    lea rbp, [rip + values]
next_line:
    cmp r12, r13
    jae finish
    mov rbx, r12
find_line_end:
    cmp rbx, r13
    jae line_ready
    cmp byte ptr [rbx], 10
    je line_ready
    inc rbx
    jmp find_line_end
line_ready:
    xor r14d, r14d
next_token:
    cmp r12, rbx
    jae evaluate_end
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov rsi, r12
find_token_end:
    inc r12
    cmp r12, rbx
    jae token_ready
    movzx eax, byte ptr [r12]
    cmp al, 32
    je token_ready
    cmp al, 9
    jne find_token_end
token_ready:
    mov rcx, r12
    sub rcx, rsi
    movzx edx, byte ptr [rsi]
    cmp rcx, 1
    jne literal
    cmp dl, '+'
    je operator
    cmp dl, '-'
    je operator
    cmp dl, '*'
    je operator
    cmp dl, '/'
    je operator
    cmp dl, '%'
    je operator
literal:
    xor r9d, r9d
    cmp dl, '-'
    jne literal_digits
    inc rsi
    dec rcx
    mov r9d, 1
literal_digits:
    test rcx, rcx
    jz bad_line
    cmp rcx, 18
    ja bad_line
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja bad_line
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digit_loop
    test r9d, r9d
    jz push_literal
    neg rax
push_literal:
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp next_token
skip_separator:
    inc r12
    jmp next_token
operator:
    cmp r14, 2
    jb bad_line
    dec r14
    mov rdi, qword ptr [rbp + r14*8]
    mov rax, qword ptr [rbp + r14*8 - 8]
    cmp dl, '+'
    je add_values
    cmp dl, '-'
    je subtract_values
    cmp dl, '*'
    je multiply_values
    mov r9d, edx
    test rdi, rdi
    jz bad_line
    cmp rdi, -1
    jne divide_values
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    je bad_line
divide_values:
    cqo
    idiv rdi
    cmp r9b, '%'
    jne store_result
    mov rax, rdx
    jmp store_result
add_values:
    add rax, rdi
    jmp store_result
subtract_values:
    sub rax, rdi
    jmp store_result
multiply_values:
    imul rax, rdi
store_result:
    mov qword ptr [rbp + r14*8 - 8], rax
    jmp next_token
evaluate_end:
    cmp r14, 1
    jne bad_line
    mov rax, qword ptr [rbp]
    lea rsi, [rip + number + 31]
    mov byte ptr [rsi], 10
    xor r9d, r9d
    test rax, rax
    jns format_number
    mov r9d, 1
    neg rax
format_number:
    mov edi, 10
format_digit:
    xor edx, edx
    div rdi
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz format_digit
    test r9d, r9d
    jz number_ready
    dec rsi
    mov byte ptr [rsi], '-'
number_ready:
    lea rdx, [rip + number + 32]
    sub rdx, rsi
    jmp emit
bad_line:
    lea rsi, [rip + error_text]
    mov edx, 6
emit:
    mov eax, 1
    mov edi, 1
    syscall
    mov r12, rbx
    cmp r12, r13
    jae finish
    inc r12
    jmp next_line
finish:
    mov eax, 60
    xor edi, edi
    syscall
