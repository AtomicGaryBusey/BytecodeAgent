.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x21, #65536
    add x21, x21, #1

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output
    add x22, x22, :lo12:output
    mov x23, x22
    mov x24, x19
    mov x25, x19
    add x26, x19, x20

scan_loop:
    cmp x25, x26
    b.eq final_remainder
    ldrb w9, [x25]
    cmp w9, #10
    b.eq complete_line
    add x25, x25, #1
    b scan_loop

complete_line:
    mov x27, x25
reverse_line:
    cmp x27, x24
    b.eq append_newline
    ldrb w9, [x27, #-1]!
    strb w9, [x23], #1
    b reverse_line

append_newline:
    mov w9, #10
    strb w9, [x23], #1
    add x25, x25, #1
    mov x24, x25
    b scan_loop

final_remainder:
    cmp x24, x26
    b.eq write_output
    mov x27, x26
reverse_final:
    cmp x27, x24
    b.eq final_newline
    ldrb w9, [x27, #-1]!
    strb w9, [x23], #1
    b reverse_final

final_newline:
    mov w9, #10
    strb w9, [x23], #1

write_output:
    sub x2, x23, x22
    cbz x2, finish
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 65537
