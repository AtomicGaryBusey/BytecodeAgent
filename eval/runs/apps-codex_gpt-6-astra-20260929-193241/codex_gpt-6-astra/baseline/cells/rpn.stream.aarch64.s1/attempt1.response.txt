.bss
.balign 16
input:
.zero 65537
values:
.zero 8000
outbuf:
.zero 32

.section .rodata
errmsg:
.ascii "error\n"

.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
    movz x21, #1
    movk x21, #1, lsl #16
read_more:
    mov x0, #0
    mov x1, x20
    sub x2, x20, x19
    sub x2, x21, x2
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_more
read_done:
    adrp x21, values
    add x21, x21, :lo12:values
next_line:
    cmp x19, x20
    b.eq finish
    mov x24, x19
find_end:
    cmp x24, x20
    b.eq end_found
    ldrb w0, [x24]
    cmp w0, #10
    b.eq end_found
    add x24, x24, #1
    b find_end
end_found:
    mov x25, x24
    cmp x25, x20
    b.eq no_newline
    add x25, x25, #1
no_newline:
    mov x22, #0
token_loop:
    cmp x19, x24
    b.eq line_done
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_space
    cmp w0, #9
    b.eq skip_space
    mov x26, x19
find_token_end:
    add x19, x19, #1
    cmp x19, x24
    b.eq token_found
    ldrb w0, [x19]
    cmp w0, #32
    b.eq token_found
    cmp w0, #9
    b.ne find_token_end
token_found:
    mov x27, x19
    sub x1, x27, x26
    ldrb w0, [x26]
    cmp x1, #1
    b.ne literal
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42
    b.eq operator
    cmp w0, #47
    b.eq operator
    cmp w0, #37
    b.eq operator
literal:
    mov x2, #0
    cmp w0, #45
    b.ne literal_digits
    mov x2, #1
    add x26, x26, #1
literal_digits:
    sub x1, x27, x26
    cbz x1, line_error
    cmp x1, #18
    b.hi line_error
    mov x3, #0
    mov x4, #10
digit_loop:
    ldrb w0, [x26], #1
    sub w0, w0, #48
    cmp w0, #9
    b.hi line_error
    madd x3, x3, x4, x0
    cmp x26, x27
    b.ne digit_loop
    cmp x2, #0
    cneg x3, x3, ne
    str x3, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop
skip_space:
    add x19, x19, #1
    b token_loop
operator:
    cmp x22, #2
    b.lo line_error
    sub x22, x22, #1
    ldr x2, [x21, x22, lsl #3]
    sub x22, x22, #1
    ldr x1, [x21, x22, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cbz x2, line_error
    mov x3, #0x8000000000000000
    cmp x1, x3
    b.ne divide
    cmn x2, #1
    b.eq line_error
divide:
    sdiv x3, x1, x2
    cmp w0, #47
    b.eq push_result
    msub x3, x3, x2, x1
    b push_result
op_add:
    add x3, x1, x2
    b push_result
op_sub:
    sub x3, x1, x2
    b push_result
op_mul:
    mul x3, x1, x2
push_result:
    str x3, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop
line_done:
    cmp x22, #1
    b.ne line_error
    ldr x3, [x21]
    adrp x4, outbuf
    add x4, x4, :lo12:outbuf
    add x5, x4, #32
    mov x1, x5
    mov w0, #10
    strb w0, [x1, #-1]!
    cmp x3, #0
    cset w6, lt
    cneg x3, x3, lt
    mov x7, #10
format_digits:
    udiv x9, x3, x7
    msub x10, x9, x7, x3
    add w10, w10, #48
    strb w10, [x1, #-1]!
    mov x3, x9
    cbnz x3, format_digits
    cbz w6, write_number
    mov w0, #45
    strb w0, [x1, #-1]!
write_number:
    sub x2, x5, x1
    b write_line
line_error:
    adrp x1, errmsg
    add x1, x1, :lo12:errmsg
    mov x2, #6
write_line:
    mov x0, #1
    mov x8, #64
    svc #0
    mov x19, x25
    b next_line
finish:
    mov x0, #0
    mov x8, #93
    svc #0
