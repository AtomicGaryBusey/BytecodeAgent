.globl _start

.bss
.balign 16
input:
.zero 53248
output:
.zero 66560

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 4096
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    mov rsi, r12
    mov r8, r13
    lea rdi, [rip + output]
    lea r14, [rip + alphabet]
    xor r9d, r9d
    test r8, r8
    jz finish

encode_loop:
    movzx ebx, byte ptr [rsi]
    shl ebx, 16
    mov r10d, 2
    cmp r8, 1
    je encode_group
    movzx eax, byte ptr [rsi + 1]
    shl eax, 8
    or ebx, eax
    mov r10d, 1
    cmp r8, 2
    je encode_group
    movzx eax, byte ptr [rsi + 2]
    or ebx, eax
    xor r10d, r10d

encode_group:
    mov eax, ebx
    shr eax, 18
    mov al, byte ptr [r14 + rax]
    mov byte ptr [rdi], al

    mov eax, ebx
    shr eax, 12
    and eax, 63
    mov al, byte ptr [r14 + rax]
    mov byte ptr [rdi + 1], al

    mov al, 61
    cmp r10d, 2
    je third_ready
    mov eax, ebx
    shr eax, 6
    and eax, 63
    mov al, byte ptr [r14 + rax]
third_ready:
    mov byte ptr [rdi + 2], al

    mov al, 61
    test r10d, r10d
    jnz fourth_ready
    and ebx, 63
    mov al, byte ptr [r14 + rbx]
fourth_ready:
    mov byte ptr [rdi + 3], al
    add rdi, 4
    inc r9d
    cmp r9d, 19
    jne line_ready
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
line_ready:
    cmp r8, 3
    jbe encode_done
    sub r8, 3
    add rsi, 3
    jmp encode_loop

encode_done:
    test r9d, r9d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

finish:
    mov eax, 60
    xor edi, edi
    syscall
