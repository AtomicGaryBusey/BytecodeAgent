.globl _start
.text
_start:
    xor ebx, ebx
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, rbx
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_more

read_done:
    lea rsi, [rip + source]
    lea r9, [rsi + rbx]
    mov r8, r9
    lea r12, [rip + commands]
    xor r13d, r13d
    xor edx, edx
parse:
    cmp rsi, r9
    je parsed
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 33
    je separator
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne parse
save_command:
    mov byte ptr [r12 + r13], al
    inc r13d
    jmp parse

open_bracket:
    mov dword ptr [bracket_stack + rdx*4], r13d
    inc edx
    jmp save_command

close_bracket:
    test edx, edx
    jz invalid
    dec edx
    mov ebx, dword ptr [bracket_stack + rdx*4]
    lea ecx, [r13 + 1]
    mov dword ptr [jumps + rbx*4], ecx
    inc ebx
    mov dword ptr [jumps + r13*4], ebx
    jmp save_command

separator:
    mov r8, rsi
parsed:
    test edx, edx
    jnz invalid
    xor r14d, r14d
    lea r15, [rip + tape]
    xor r10d, r10d

execute:
    cmp r14d, r13d
    jae finished
    movzx eax, byte ptr [r12 + r14]
    inc r14d
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 46
    je output
    cmp al, 44
    je input
    cmp al, 91
    je loop_open
    cmp byte ptr [r15], 0
    je execute
    mov r14d, dword ptr [jumps + r14*4 - 4]
    jmp execute

loop_open:
    cmp byte ptr [r15], 0
    jne execute
    mov r14d, dword ptr [jumps + r14*4 - 4]
    jmp execute

increment:
    inc byte ptr [r15]
    jmp execute
decrement:
    dec byte ptr [r15]
    jmp execute
right:
    inc r15
    jmp execute
left:
    dec r15
    jmp execute
output:
    mov al, byte ptr [r15]
    mov byte ptr [result + r10], al
    inc r10d
    jmp execute
input:
    xor eax, eax
    cmp r8, r9
    jae input_store
    mov al, byte ptr [r8]
    inc r8
input_store:
    mov byte ptr [r15], al
    jmp execute

finished:
    mov edx, r10d
    lea rsi, [rip + result]
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

invalid:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.balign 8
source:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
result:
    .zero 65536
