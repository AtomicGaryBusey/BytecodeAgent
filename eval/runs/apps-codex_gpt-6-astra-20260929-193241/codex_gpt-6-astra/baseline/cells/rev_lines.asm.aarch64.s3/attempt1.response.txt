.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x21, #65536
read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    add x2, x2, #1
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output
    add x22, x22, :lo12:output
    mov x23, x22
    mov x24, x19
    mov x25, x19
    add x26, x19, x20

scan:
    cmp x25, x26
    b.eq final_line
    ldrb w9, [x25], #1
    cmp w9, #10
    b.ne scan
    sub x10, x25, #1
reverse_line:
    cmp x10, x24
    b.eq line_newline
    ldrb w9, [x10, #-1]!
    strb w9, [x23], #1
    b reverse_line
line_newline:
    mov w9, #10
    strb w9, [x23], #1
    mov x24, x25
    b scan

final_line:
    cmp x24, x26
    b.eq write_output
    mov x10, x26
reverse_final:
    ldrb w9, [x10, #-1]!
    strb w9, [x23], #1
    cmp x10, x24
    b.ne reverse_final
    mov w9, #10
    strb w9, [x23], #1

write_output:
    sub x2, x23, x22
    cbz x2, finish
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0
finish:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 65537
