.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
    mov x21, #0
read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    add x21, x21, x0
    b read_loop

read_done:
    adrp x23, values
    add x23, x23, :lo12:values
next_line:
    cmp x19, x20
    b.eq finish
    mov x21, x19
find_end:
    cmp x21, x20
    b.eq end_at_eof
    ldrb w0, [x21]
    cmp w0, #10
    b.eq end_at_newline
    add x21, x21, #1
    b find_end
end_at_newline:
    add x22, x21, #1
    b begin_line
end_at_eof:
    mov x22, x21
begin_line:
    mov x24, #0
token_loop:
    cmp x19, x21
    b.eq line_done
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator
    mov x25, x19
token_end:
    add x19, x19, #1
    cmp x19, x21
    b.eq token_ready
    ldrb w0, [x19]
    cmp w0, #32
    b.eq token_ready
    cmp w0, #9
    b.ne token_end
token_ready:
    sub x26, x19, x25
    ldrb w0, [x25]
    cmp x26, #1
    b.ne literal
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42
    b.eq operator
    cmp w0, #47
    b.eq operator
    cmp w0, #37
    b.eq operator
literal:
    mov x1, x25
    mov x2, x26
    mov w3, #0
    cmp w0, #45
    b.ne literal_digits
    mov w3, #1
    add x1, x1, #1
    sub x2, x2, #1
literal_digits:
    cbz x2, line_error
    cmp x2, #18
    b.hi line_error
    mov x4, #0
    mov x5, #10
digit_loop:
    ldrb w6, [x1], #1
    sub w6, w6, #48
    cmp w6, #9
    b.hi line_error
    madd x4, x4, x5, x6
    subs x2, x2, #1
    b.ne digit_loop
    cmp w3, #0
    cneg x4, x4, ne
    str x4, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop
skip_separator:
    add x19, x19, #1
    b token_loop

operator:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x2, [x23, x24, lsl #3]
    sub x24, x24, #1
    ldr x1, [x23, x24, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cbz x2, line_error
    mov x3, #0x8000000000000000
    cmp x1, x3
    b.ne divide
    cmn x2, #1
    b.eq line_error
divide:
    sdiv x4, x1, x2
    cmp w0, #47
    b.eq store_result
    msub x4, x4, x2, x1
    b store_result
op_add:
    add x4, x1, x2
    b store_result
op_sub:
    sub x4, x1, x2
    b store_result
op_mul:
    mul x4, x1, x2
store_result:
    str x4, [x23, x24, lsl #3]
    add x24, x24, #1
    b token_loop

line_done:
    cmp x24, #1
    b.ne line_error
    ldr x3, [x23]
    adrp x1, output
    add x1, x1, :lo12:output
    add x1, x1, #32
    mov w0, #10
    strb w0, [x1, #-1]!
    mov x2, #1
    cmp x3, #0
    cset w7, lt
    cneg x3, x3, lt
    mov x4, #10
format_digit:
    udiv x5, x3, x4
    msub x6, x5, x4, x3
    add w6, w6, #48
    strb w6, [x1, #-1]!
    add x2, x2, #1
    mov x3, x5
    cbnz x3, format_digit
    cbz w7, emit
    mov w0, #45
    strb w0, [x1, #-1]!
    add x2, x2, #1
    b emit
line_error:
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
emit:
    mov x0, #1
    mov x8, #64
    svc #0
    mov x19, x22
    b next_line
finish:
    mov x0, #0
    mov x8, #93
    svc #0
