.globl _start

.bss
.balign 16
input_buffer:
.zero 65537
output_buffer:
.zero 323600

.section .rodata
digits:
.ascii "0123456789abcdef"

.text
_start:
    xor r13d, r13d
    lea r12, [rip + input_buffer]
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    js read_loop
    add r13, rax
    jmp read_loop

read_done:
    lea rbx, [rip + digits]
    lea rdi, [rip + output_buffer]
    xor r14d, r14d

row_loop:
    cmp r14, r13
    jae final_offset
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe row_size_ready
    mov r15d, 16
row_size_ready:
    mov eax, r14d
    call emit_offset
    mov ax, 0x2020
    stosw
    xor r9d, r9d

hex_loop:
    cmp r9, r15
    jae empty_slot
    lea rsi, [r12 + r14]
    movzx r10d, byte ptr [rsi + r9]
    mov eax, r10d
    shr eax, 4
    mov al, byte ptr [rbx + rax]
    stosb
    mov eax, r10d
    and eax, 15
    mov al, byte ptr [rbx + rax]
    stosb
    mov al, 32
    stosb
    jmp slot_done
empty_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb
slot_done:
    cmp r9d, 7
    jne next_slot
    mov al, 32
    stosb
next_slot:
    inc r9d
    cmp r9d, 16
    jb hex_loop

    mov ax, 0x7c20
    stosw
    xor r9d, r9d
    lea rsi, [r12 + r14]
ascii_loop:
    mov al, byte ptr [rsi + r9]
    cmp al, 32
    jb nonprintable
    cmp al, 126
    jbe emit_character
nonprintable:
    mov al, 46
emit_character:
    stosb
    inc r9d
    cmp r9, r15
    jb ascii_loop
    mov ax, 0x0a7c
    stosw
    add r14, r15
    jmp row_loop

final_offset:
    mov eax, r13d
    call emit_offset
    mov al, 10
    stosb
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov edx, eax
    mov ecx, 8
offset_digit:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [rbx + rax]
    stosb
    dec ecx
    jnz offset_digit
    ret
