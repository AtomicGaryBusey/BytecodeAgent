.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65537
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x22, jumps
    add x22, x22, :lo12:jumps
    adrp x23, bracket_stack
    add x23, x23, :lo12:bracket_stack
    mov x24, #0
    mov x25, #0
    mov x26, x20
    mov x9, #0

parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w10, [x19, x9]
    add x9, x9, #1
    cmp w10, #33
    b.eq separator
    cmp w10, #91
    b.eq open_bracket
    cmp w10, #93
    b.eq close_bracket
    cmp w10, #62
    b.eq save_command
    cmp w10, #60
    b.eq save_command
    cmp w10, #43
    b.eq save_command
    cmp w10, #45
    b.eq save_command
    cmp w10, #46
    b.eq save_command
    cmp w10, #44
    b.ne parse_loop

save_command:
    strb w10, [x21, x25]
    add x25, x25, #1
    b parse_loop

open_bracket:
    str w25, [x23, x24, lsl #2]
    add x24, x24, #1
    b save_command

close_bracket:
    cbz x24, invalid
    sub x24, x24, #1
    ldr w11, [x23, x24, lsl #2]
    add w12, w25, #1
    str w12, [x22, x11, lsl #2]
    add w12, w11, #1
    str w12, [x22, x25, lsl #2]
    b save_command

separator:
    mov x26, x9
parse_done:
    cbnz x24, invalid
    mov x27, x25
    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x25, #0
    mov x9, #0
    mov x10, #0

execute:
    cmp x9, x27
    b.hs finished
    ldrb w11, [x21, x9]
    add x9, x9, #1
    cmp w11, #43
    b.eq increment
    cmp w11, #45
    b.eq decrement
    cmp w11, #62
    b.eq right
    cmp w11, #60
    b.eq left
    cmp w11, #46
    b.eq output_byte
    cmp w11, #44
    b.eq input_byte
    ldrb w12, [x23, x10]
    cmp w11, #91
    b.eq loop_open
    cbz w12, execute
    b jump
loop_open:
    cbnz w12, execute
jump:
    sub x12, x9, #1
    ldr w9, [x22, x12, lsl #2]
    b execute

increment:
    ldrb w12, [x23, x10]
    add w12, w12, #1
    strb w12, [x23, x10]
    b execute
decrement:
    ldrb w12, [x23, x10]
    sub w12, w12, #1
    strb w12, [x23, x10]
    b execute
right:
    add x10, x10, #1
    b execute
left:
    sub x10, x10, #1
    b execute
output_byte:
    ldrb w12, [x23, x10]
    strb w12, [x24, x25]
    add x25, x25, #1
    b execute
input_byte:
    mov w12, #0
    cmp x26, x20
    b.hs store_input
    ldrb w12, [x19, x26]
    add x26, x26, #1
store_input:
    strb w12, [x23, x10]
    b execute

finished:
    cbz x25, success
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
success:
    mov x0, #0
    b terminate
invalid:
    mov x0, #1
terminate:
    mov x8, #93
    svc #0
    b terminate

.bss
.balign 16
