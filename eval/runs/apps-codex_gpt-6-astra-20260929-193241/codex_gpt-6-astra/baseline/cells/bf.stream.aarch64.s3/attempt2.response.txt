.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x18, commands
    add x18, x18, :lo12:commands
    adrp x28, jumps
    add x28, x28, :lo12:jumps
    adrp x17, bracket_stack
    add x17, x17, :lo12:bracket_stack
    mov x16, #0
    mov x21, #0
    mov x22, #0

parse_loop:
    cmp x22, x20
    b.hs parse_done
    ldrb w0, [x19, x22]
    add x22, x22, #1
    cmp w0, #33
    b.eq parse_done
    cmp w0, #91
    b.eq parse_open
    cmp w0, #93
    b.eq parse_close
    cmp w0, #62
    b.eq save_command
    cmp w0, #60
    b.eq save_command
    cmp w0, #43
    b.eq save_command
    cmp w0, #45
    b.eq save_command
    cmp w0, #46
    b.eq save_command
    cmp w0, #44
    b.ne parse_loop

save_command:
    strb w0, [x18, x21]
    add x21, x21, #1
    b parse_loop

parse_open:
    str w21, [x17, x16, lsl #2]
    add x16, x16, #1
    b save_command

parse_close:
    cbz x16, invalid
    sub x16, x16, #1
    ldr w1, [x17, x16, lsl #2]
    add w2, w21, #1
    str w2, [x28, x1, lsl #2]
    add w2, w1, #1
    str w2, [x28, x21, lsl #2]
    b save_command

parse_done:
    cbnz x16, invalid
    adrp x26, tape
    add x26, x26, :lo12:tape
    adrp x27, output_buffer
    add x27, x27, :lo12:output_buffer
    mov x23, #0
    mov x24, #0
    mov x25, #0

execute:
    cmp x23, x21
    b.hs finished
    ldrb w0, [x18, x23]
    add x23, x23, #1
    cmp w0, #43
    b.eq increment
    cmp w0, #45
    b.eq decrement
    cmp w0, #62
    b.eq right
    cmp w0, #60
    b.eq left
    cmp w0, #91
    b.eq open_loop
    cmp w0, #93
    b.eq close_loop
    cmp w0, #46
    b.eq emit

consume:
    mov w1, #0
    cmp x22, x20
    b.hs store_input
    ldrb w1, [x19, x22]
    add x22, x22, #1
store_input:
    strb w1, [x26, x24]
    b execute

increment:
    ldrb w1, [x26, x24]
    add w1, w1, #1
    strb w1, [x26, x24]
    b execute

decrement:
    ldrb w1, [x26, x24]
    sub w1, w1, #1
    strb w1, [x26, x24]
    b execute

right:
    add x24, x24, #1
    b execute

left:
    sub x24, x24, #1
    b execute

open_loop:
    ldrb w1, [x26, x24]
    cbnz w1, execute
    b jump_loop

close_loop:
    ldrb w1, [x26, x24]
    cbz w1, execute

jump_loop:
    sub x1, x23, #1
    ldr w23, [x28, x1, lsl #2]
    b execute

emit:
    ldrb w1, [x26, x24]
    strb w1, [x27, x25]
    add x25, x25, #1
    b execute

finished:
    cbz x25, success
    mov x0, #1
    mov x1, x27
    mov x2, x25
    mov x8, #64
    svc #0
success:
    mov x0, #0
    mov x8, #93
    svc #0
halt:
    b halt

invalid:
    mov x0, #1
    mov x8, #93
    svc #0
    b halt

.bss
.balign 16
input_buffer:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
bracket_stack:
    .zero 4000
tape:
    .zero 30000
output_buffer:
    .zero 65536
