.bss
.balign 16
input_buffer:
.zero 65537
commands:
.zero 65536
.balign 4
jumps:
.zero 262144
bracket_stack:
.zero 4000
tape:
.zero 30000
output_buffer:
.zero 65536

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x22, jumps
    add x22, x22, :lo12:jumps
    adrp x15, bracket_stack
    add x15, x15, :lo12:bracket_stack
    mov x14, #0
    mov x26, #0
    mov x18, #0

parse_loop:
    cmp x18, x20
    b.hs parse_done
    ldrb w0, [x19, x18]
    add x18, x18, #1
    cmp w0, #33
    b.eq parse_done
    cmp w0, #91
    b.eq parse_open
    cmp w0, #93
    b.eq parse_close
    cmp w0, #62
    b.eq save_command
    cmp w0, #60
    b.eq save_command
    cmp w0, #43
    b.eq save_command
    cmp w0, #45
    b.eq save_command
    cmp w0, #46
    b.eq save_command
    cmp w0, #44
    b.ne parse_loop
save_command:
    strb w0, [x21, x26]
    add x26, x26, #1
    b parse_loop

parse_open:
    str w26, [x15, x14, lsl #2]
    add x14, x14, #1
    b save_command

parse_close:
    cbz x14, invalid
    sub x14, x14, #1
    ldr w1, [x15, x14, lsl #2]
    add w2, w26, #1
    str w2, [x22, x1, lsl #2]
    add w2, w1, #1
    str w2, [x22, x26, lsl #2]
    b save_command

parse_done:
    cbnz x14, invalid
    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x25, #0
    mov x27, #0
    mov x28, #0

execute:
    cmp x25, x26
    b.hs finished
    ldrb w0, [x21, x25]
    add x25, x25, #1
    cmp w0, #43
    b.eq increment
    cmp w0, #45
    b.eq decrement
    cmp w0, #62
    b.eq move_right
    cmp w0, #60
    b.eq move_left
    cmp w0, #91
    b.eq loop_open
    cmp w0, #93
    b.eq loop_close
    cmp w0, #46
    b.eq emit

consume:
    mov w1, #0
    cmp x18, x20
    b.hs store_cell
    ldrb w1, [x19, x18]
    add x18, x18, #1
store_cell:
    strb w1, [x23, x27]
    b execute

increment:
    ldrb w1, [x23, x27]
    add w1, w1, #1
    strb w1, [x23, x27]
    b execute

decrement:
    ldrb w1, [x23, x27]
    sub w1, w1, #1
    strb w1, [x23, x27]
    b execute

move_right:
    add x27, x27, #1
    b execute

move_left:
    sub x27, x27, #1
    b execute

loop_open:
    ldrb w1, [x23, x27]
    cbnz w1, execute
    b jump

loop_close:
    ldrb w1, [x23, x27]
    cbz w1, execute
jump:
    sub x1, x25, #1
    ldr w25, [x22, x1, lsl #2]
    b execute

emit:
    ldrb w1, [x23, x27]
    strb w1, [x24, x28]
    add x28, x28, #1
    b execute

finished:
    cbz x28, success
    mov x0, #1
    mov x1, x24
    mov x2, x28
    mov x8, #64
    svc #0
success:
    mov x0, #0
    mov x8, #93
    svc #0
halt:
    b halt

invalid:
    mov x0, #1
    mov x8, #93
    svc #0
    b halt
