.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz print_counts
    js read_more
    add r14, rax
    lea rdi, [rsi + rax]

count_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, rdi
    jb count_loop
    jmp read_more

print_counts:
    lea r8, [rip + output]
    mov rax, r12
    call decimal
    mov byte ptr [r8], 32
    inc r8
    mov rax, r13
    call decimal
    mov byte ptr [r8], 32
    inc r8
    mov rax, r14
    call decimal
    mov byte ptr [r8], 10
    inc r8

    lea rsi, [rip + output]
    mov rdx, r8
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

decimal:
    lea r9, [rip + digits + 32]
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz decimal_loop
    lea r10, [rip + digits + 32]
copy_digits:
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    inc r9
    inc r8
    cmp r9, r10
    jb copy_digits
    ret

.bss
.balign 16
buffer:
    .zero 4096
output:
    .zero 64
digits:
    .zero 32
