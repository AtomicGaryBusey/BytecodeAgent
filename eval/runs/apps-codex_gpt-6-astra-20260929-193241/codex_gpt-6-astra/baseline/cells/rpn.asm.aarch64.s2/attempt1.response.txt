.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
    mov x21, #65536
    add x21, x21, #1
read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    sub x21, x21, x0
    b read_loop
read_done:
    adrp x21, values
    add x21, x21, :lo12:values

line_loop:
    cmp x19, x20
    b.hs finished
    mov x24, x19
find_newline:
    cmp x24, x20
    b.hs final_line
    ldrb w0, [x24]
    cmp w0, #10
    b.eq found_newline
    add x24, x24, #1
    b find_newline
found_newline:
    add x25, x24, #1
    b begin_line
final_line:
    mov x25, x24
begin_line:
    mov x22, #0
token_loop:
    cmp x19, x24
    b.hs line_done
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator
    mov x1, x19
find_token_end:
    cmp x19, x24
    b.hs token_ready
    ldrb w2, [x19]
    cmp w2, #32
    b.eq token_ready
    cmp w2, #9
    b.eq token_ready
    add x19, x19, #1
    b find_token_end
skip_separator:
    add x19, x19, #1
    b token_loop
token_ready:
    sub x2, x19, x1
    cmp x2, #1
    b.ne literal
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42
    b.eq operator
    cmp w0, #47
    b.eq operator
    cmp w0, #37
    b.eq operator
literal:
    mov x3, #0
    cmp w0, #45
    b.ne literal_length
    mov x3, #1
    add x1, x1, #1
    sub x2, x2, #1
literal_length:
    cbz x2, line_error
    cmp x2, #18
    b.hi line_error
    mov x4, #0
    mov x5, #10
literal_digits:
    ldrb w6, [x1], #1
    sub w6, w6, #48
    cmp w6, #9
    b.hi line_error
    madd x4, x4, x5, x6
    subs x2, x2, #1
    b.ne literal_digits
    cmp x3, #0
    cneg x4, x4, ne
    str x4, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop

operator:
    cmp x22, #2
    b.lo line_error
    sub x22, x22, #1
    ldr x4, [x21, x22, lsl #3]
    sub x22, x22, #1
    ldr x3, [x21, x22, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cbz x4, line_error
    mov x5, #0x8000000000000000
    cmp x3, x5
    b.ne divide
    cmn x4, #1
    b.eq line_error
divide:
    sdiv x5, x3, x4
    cmp w0, #47
    b.eq quotient
    msub x3, x5, x4, x3
    b store_result
quotient:
    mov x3, x5
    b store_result
op_add:
    add x3, x3, x4
    b store_result
op_sub:
    sub x3, x3, x4
    b store_result
op_mul:
    mul x3, x3, x4
store_result:
    str x3, [x21, x22, lsl #3]
    add x22, x22, #1
    b token_loop

line_done:
    cmp x22, #1
    b.ne line_error
    ldr x3, [x21]
    adrp x1, output
    add x1, x1, :lo12:output
    add x1, x1, #32
    mov x7, x1
    mov w0, #10
    strb w0, [x1, #-1]!
    cmp x3, #0
    cset x6, lt
    cneg x3, x3, lt
    mov x4, #10
format_digits:
    udiv x5, x3, x4
    msub x0, x5, x4, x3
    add w0, w0, #48
    strb w0, [x1, #-1]!
    mov x3, x5
    cbnz x3, format_digits
    cbz x6, format_done
    mov w0, #45
    strb w0, [x1, #-1]!
format_done:
    sub x2, x7, x1
    b emit
line_error:
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
emit:
    mov x0, #1
    mov x8, #64
    svc #0
    mov x19, x25
    b line_loop

finished:
    mov x0, #0
    mov x8, #93
    svc #0
