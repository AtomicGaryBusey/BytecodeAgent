.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output
    add x21, x21, :lo12:output
    mov x22, x21
    mov x23, x19
    mov x24, x19
    add x25, x19, x20

scan_loop:
    cmp x24, x25
    b.eq input_end
    ldrb w9, [x24], #1
    cmp w9, #10
    b.ne scan_loop
    sub x26, x24, #1

reverse_line:
    cmp x26, x23
    b.eq newline
    ldrb w9, [x26, #-1]!
    strb w9, [x22], #1
    b reverse_line

newline:
    mov w9, #10
    strb w9, [x22], #1
    mov x23, x24
    b scan_loop

input_end:
    cmp x23, x25
    b.eq write_output
    mov x26, x25

reverse_final:
    cmp x26, x23
    b.eq final_newline
    ldrb w9, [x26, #-1]!
    strb w9, [x22], #1
    b reverse_final

final_newline:
    mov w9, #10
    strb w9, [x22], #1

write_output:
    sub x2, x22, x21
    cbz x2, done
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0

done:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 69632
output:
    .zero 65537
