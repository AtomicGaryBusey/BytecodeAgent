    b.eq ws_next
    cmp w0, #9

    cmp w0, #10

    bl peek
    cmp w0, #123
    b.eq object
    cmp w0, #91
    b.eq array
    cmp w0, #34

    cmp w0, #116
    b.eq literal_true
    cmp w0, #102
    b.eq literal_false
    cmp w0, #110
