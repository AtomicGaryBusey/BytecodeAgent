.globl _start

.section .rodata
digits:
.ascii "0123456789abcdef"

.bss
.balign 16
input:
.zero 65552
output:
.zero 323600

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65552
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output]
    lea rsi, [rip + input]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    je final_offset
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16
row_size_ready:
    mov eax, r13d
    call print_offset
    mov al, 32
    stosb
    stosb
    xor r8d, r8d

hex_loop:
    cmp r8, r14
    jae empty_slot
    movzx r9d, byte ptr [rsi + r8]
    mov ebx, r9d
    shr ebx, 4
    mov al, byte ptr [digits + rbx]
    stosb
    and r9d, 15
    mov al, byte ptr [digits + r9]
    stosb
    mov al, 32
    stosb
    jmp slot_done
empty_slot:
    mov al, 32
    stosb
    stosb
    stosb
slot_done:
    cmp r8d, 7
    jne no_extra_space
    mov al, 32
    stosb
no_extra_space:
    inc r8d
    cmp r8d, 16
    jb hex_loop

    mov al, 32
    stosb
    mov al, 124
    stosb
    xor r8d, r8d
ascii_loop:
    mov al, byte ptr [rsi + r8]
    cmp al, 32
    jb replace_char
    cmp al, 126
    jbe emit_char
replace_char:
    mov al, 46
emit_char:
    stosb
    inc r8d
    cmp r8, r14
    jb ascii_loop
    mov al, 124
    stosb
    mov al, 10
    stosb
    add rsi, r14
    add r13, r14
    jmp row_loop

final_offset:
    mov eax, r12d
    call print_offset
    mov al, 10
    stosb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_offset:
    mov edx, eax
    mov ecx, 8
offset_digit:
    rol edx, 4
    mov ebx, edx
    and ebx, 15
    mov al, byte ptr [digits + rbx]
    stosb
    dec ecx
    jnz offset_digit
    ret
