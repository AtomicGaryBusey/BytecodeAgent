.globl _start
.bss
.balign 8
input:
.zero 65537
values:
.zero 8000
output:
.zero 393216
.text
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz input_ready
    add r12, rax
    jmp read_more
input_ready:
    lea r13, [rip + input]
    add r13, r12
    lea r12, [rip + input]
    lea rbp, [rip + output]
next_line:
    cmp r12, r13
    jae finish
    mov r14, r12
find_line_end:
    cmp r14, r13
    jae line_ready
    cmp byte ptr [r14], 10
    je line_ready
    inc r14
    jmp find_line_end
line_ready:
    xor r15d, r15d
next_token:
    cmp r12, r14
    jae line_done
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov rbx, r12
find_token_end:
    inc rbx
    cmp rbx, r14
    jae token_ready
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je token_ready
    cmp al, 9
    jne find_token_end
token_ready:
    mov rax, rbx
    sub rax, r12
    cmp rax, 1
    jne literal
    movzx ecx, byte ptr [r12]
    cmp cl, 43
    je operator
    cmp cl, 45
    je operator
    cmp cl, 42
    je operator
    cmp cl, 47
    je operator
    cmp cl, 37
    je operator
literal:
    xor r8d, r8d
    xor r9d, r9d
    cmp byte ptr [r12], 45
    jne literal_digits
    inc r12
    mov r9d, 1
literal_digits:
    mov rax, rbx
    sub rax, r12
    cmp rax, 1
    jb line_error
    cmp rax, 18
    ja line_error
digit_loop:
    movzx eax, byte ptr [r12]
    sub eax, 48
    cmp eax, 9
    ja line_error
    imul r8, r8, 10
    add r8, rax
    inc r12
    cmp r12, rbx
    jb digit_loop
    test r9d, r9d
    jz push_value
    neg r8
push_value:
    lea rdi, [rip + values]
    mov qword ptr [rdi + r15*8], r8
    inc r15
    mov r12, rbx
    jmp next_token
skip_separator:
    inc r12
    jmp next_token
operator:
    cmp r15, 2
    jb line_error
    lea rdi, [rip + values]
    dec r15
    mov rsi, qword ptr [rdi + r15*8]
    dec r15
    mov r8, qword ptr [rdi + r15*8]
    cmp cl, 43
    je add_values
    cmp cl, 45
    je subtract_values
    cmp cl, 42
    je multiply_values
    test rsi, rsi
    jz line_error
    mov rax, 0x8000000000000000
    cmp r8, rax
    jne divide_values
    cmp rsi, -1
    je line_error
divide_values:
    mov rax, r8
    cqo
    idiv rsi
    mov r8, rax
    cmp cl, 37
    jne push_value
    mov r8, rdx
    jmp push_value
add_values:
    add r8, rsi
    jmp push_value
subtract_values:
    sub r8, rsi
    jmp push_value
multiply_values:
    imul r8, rsi
    jmp push_value
line_done:
    cmp r15, 1
    jne line_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns print_magnitude
    mov byte ptr [rbp], 45
    inc rbp
    neg rax
print_magnitude:
    mov rdi, rbp
    mov esi, 10
decimal_loop:
    xor edx, edx
    div rsi
    add dl, 48
    mov byte ptr [rbp], dl
    inc rbp
    test rax, rax
    jnz decimal_loop
    lea rsi, [rbp - 1]
reverse_digits:
    cmp rdi, rsi
    jae number_done
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp reverse_digits
number_done:
    mov byte ptr [rbp], 10
    inc rbp
    jmp advance_line
line_error:
    mov byte ptr [rbp], 101
    mov byte ptr [rbp + 1], 114
    mov byte ptr [rbp + 2], 114
    mov byte ptr [rbp + 3], 111
    mov byte ptr [rbp + 4], 114
    mov byte ptr [rbp + 5], 10
    add rbp, 6
advance_line:
    mov r12, r14
    cmp r12, r13
    jae finish
    inc r12
    jmp next_line
finish:
    lea rsi, [rip + output]
    mov rdx, rbp
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
