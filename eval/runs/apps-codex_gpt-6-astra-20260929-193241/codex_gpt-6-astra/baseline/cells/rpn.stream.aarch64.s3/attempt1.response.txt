.globl _start

.bss
.balign 16
input_buffer:
.zero 65537

.zero 4096

.zero 32

.section .rodata

.ascii "error\n"

.text

    add x19, x19, :lo12:input_buffer
    mov x20, x19
    movz x25, #1

    add x21, x21, :lo12:value_stack

    b.hs finish
    mov x24, x19

    b.hs line_ready
    ldrb w0, [x24]

    b.eq line_ready

    b.hs line_complete
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator

    cmp x19, x24
    b.hs token_ready
    ldrb w0, [x19]
    cmp w0, #32
    b.eq token_ready
    cmp w0, #9

    ldrb w7, [x4]
    cmp x6, #1
    b.ne literal
    cmp w7, #43
    b.eq operator
    cmp w7, #45
    b.eq operator
    cmp w7, #42
    b.eq operator
    cmp w7, #47
    b.eq operator
    cmp w7, #37
    b.eq operator

    cmp w7, #45
    b.ne literal_digits
    mov x9, #1
    add x4, x4, #1
    sub x6, x6, #1

    cmp x6, #18

    mov x11, #10

    cmp w7, #9

    madd x10, x10, x11, x7
    subs x6, x6, #1
    b.ne parse_digit
    cmp x9, #0

    b next_token

    b.lo line_error
    sub x22, x22, #1
    ldr x10, [x21, x22, lsl #3]
    sub x22, x22, #1
    ldr x9, [x21, x22, lsl #3]
    cmp w7, #43
    b.eq op_add
    cmp w7, #45
    b.eq op_sub
    cmp w7, #42

    mov x11, #0x8000000000000000
    cmp x9, x11

    cmp w7, #47
    b.eq quotient
    msub x9, x11, x10, x9

    b push_result

    b push_result

    add x22, x22, #1

    b.ne line_error
    ldr x9, [x21]
    adrp x12, output_buffer
    add x12, x12, :lo12:output_buffer

    mov x8, #64
    svc #0

    mov x8, #93
