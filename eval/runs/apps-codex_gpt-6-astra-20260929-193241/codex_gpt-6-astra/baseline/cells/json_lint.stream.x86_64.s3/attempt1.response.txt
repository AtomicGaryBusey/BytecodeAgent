.globl _start
.text
_start:
    lea r14, [rip + input]
    mov r13, r14
read_more:
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    lea rdx, [rip + input + 65537]
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more
read_done:
    mov r12, r14
    call value
    call whitespace
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

peek:
    mov eax, -1
    cmp r12, r13
    je peek_done
    movzx eax, byte ptr [r12]
peek_done:
    ret

whitespace:
    call peek
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    jne ws_done
ws_next:
    inc r12
    jmp whitespace
ws_done:
    ret

value:
    call whitespace
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je true_value
    cmp eax, 102
    je false_value
    cmp eax, 110
    je null_value
    cmp eax, 45
    je number
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    jmp number

true_value:
    lea rsi, [rip + literal_true]
    jmp literal
false_value:
    lea rsi, [rip + literal_false]
    jmp literal
null_value:
    lea rsi, [rip + literal_null]
literal:
    movzx edx, byte ptr [rsi]
    test edx, edx
    jz literal_done
    call peek
    cmp eax, edx
    jne error
    inc r12
    inc rsi
    jmp literal
literal_done:
    ret

string:
    inc r12
string_loop:
    call peek
    cmp eax, -1
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_end
    cmp eax, 92
    je escape
    inc r12
    jmp string_loop
string_end:
    inc r12
    ret
escape:
    inc r12
    call peek
    cmp eax, 34
    je escape_simple
    cmp eax, 92
    je escape_simple
    cmp eax, 47
    je escape_simple
    cmp eax, 98
    je escape_simple
    cmp eax, 102
    je escape_simple
    cmp eax, 110
    je escape_simple
    cmp eax, 114
    je escape_simple
    cmp eax, 116
    je escape_simple
    cmp eax, 117
    jne error
    inc r12
    mov edx, 4
hex_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_good
    or eax, 32
    cmp eax, 97
    jb error
    cmp eax, 102
    ja error
hex_good:
    inc r12
    dec edx
    jnz hex_loop
    jmp string_loop
escape_simple:
    inc r12
    jmp string_loop

number:
    cmp eax, 45
    jne integer
    inc r12
    call peek
integer:
    cmp eax, 48
    je zero_integer
    cmp eax, 49
    jb error
    cmp eax, 57
    ja error
    call digits
    jmp fraction
zero_integer:
    inc r12
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r12
    call require_digits
exponent:
    cmp eax, 101
    je exponent_start
    cmp eax, 69
    jne number_done
exponent_start:
    inc r12
    call peek
    cmp eax, 43
    je exponent_sign
    cmp eax, 45
    jne exponent_digits
exponent_sign:
    inc r12
exponent_digits:
    call require_digits
number_done:
    ret
require_digits:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
digits:
    inc r12
    call peek
    cmp eax, 48
    jb digits_done
    cmp eax, 57
    jbe digits
digits_done:
    ret

array:
    inc r12
    call whitespace
    cmp eax, 93
    je array_end
array_value:
    call value
    call whitespace
    cmp eax, 93
    je array_end
    cmp eax, 44
    jne error
    inc r12
    jmp array_value
array_end:
    inc r12
    ret

object:
    inc r12
    call whitespace
    cmp eax, 125
    je object_end
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r12
    call value
    call whitespace
    cmp eax, 125
    je object_end
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp object_member
object_end:
    inc r12
    ret

error:
    mov rax, r12
    sub rax, r14
    lea rsi, [rip + output + 31]
    mov byte ptr [rsi], 10
    mov ebx, 10
decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    sub rsi, 9
    mov byte ptr [rsi], 101
    mov byte ptr [rsi + 1], 114
    mov byte ptr [rsi + 2], 114
    mov byte ptr [rsi + 3], 111
    mov byte ptr [rsi + 4], 114
    mov byte ptr [rsi + 5], 32
    mov byte ptr [rsi + 6], 97
    mov byte ptr [rsi + 7], 116
    mov byte ptr [rsi + 8], 32
    lea rdx, [rip + output + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok:
    .ascii "ok\n"
literal_true:
    .asciz "true"
literal_false:
    .asciz "false"
literal_null:
    .asciz "null"
.bss
input:
    .zero 65537
output:
    .zero 32
