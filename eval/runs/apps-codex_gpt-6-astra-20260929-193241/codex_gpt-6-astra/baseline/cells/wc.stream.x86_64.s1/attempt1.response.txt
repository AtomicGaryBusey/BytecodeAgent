.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz finished
    js read_more
    add r14, rax
    lea r8, [rsi + rax]

scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne classify
    inc r12
classify:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, r8
    jb scan
    jmp read_more

finished:
    lea rdi, [rip + output]
    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

decimal:
    lea r9, [rip + digits + 32]
    mov r10, r9
    mov ecx, 10
decimal_digit:
    xor edx, edx
    div rcx
    add dl, 48
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz decimal_digit
decimal_copy:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    cmp r9, r10
    jb decimal_copy
    ret

.bss
.balign 16
buffer:
    .zero 4096
output:
    .zero 64
digits:
    .zero 32
