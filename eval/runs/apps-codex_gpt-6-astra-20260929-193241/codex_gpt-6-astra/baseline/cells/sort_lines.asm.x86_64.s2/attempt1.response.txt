.globl _start

.bss
.balign 16
input:
.zero 32769
records_a:
.zero 8000
records_b:
.zero 8000
output:
.zero 33768

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz parse
    add r13, rax
    jmp read_loop

parse:
    lea r14, [rip + records_a]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
parse_loop:
    cmp r8, r13
    jae parse_tail
    cmp byte ptr [r12 + r8], 10
    jne parse_next
    mov dword ptr [r14 + r10*8], r9d
    mov eax, r8d
    sub eax, r9d
    mov dword ptr [r14 + r10*8 + 4], eax
    inc r10
    lea r9, [r8 + 1]
parse_next:
    inc r8
    jmp parse_loop
parse_tail:
    cmp r9, r13
    je parsed
    mov dword ptr [r14 + r10*8], r9d
    mov eax, r13d
    sub eax, r9d
    mov dword ptr [r14 + r10*8 + 4], eax
    inc r10
parsed:
    mov r13, r10
    lea r15, [rip + records_b]
    mov ebp, 1

sort_pass:
    cmp rbp, r13
    jae emit
    xor ebx, ebx
merge_pair:
    cmp rbx, r13
    jae pass_done
    mov r8, rbx
    lea r9, [rbx + rbp]
    cmp r9, r13
    cmova r9, r13
    mov r10, r9
    lea r12, [r9 + rbp]
    cmp r12, r13
    cmova r12, r13
merge_loop:
    cmp r8, r9
    jae take_right_tail
    cmp r10, r12
    jae take_left
    mov rax, qword ptr [r14 + r8*8]
    mov rdx, qword ptr [r14 + r10*8]
    call compare
    test eax, eax
    jg take_right
take_left:
    mov rax, qword ptr [r14 + r8*8]
    inc r8
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp merge_loop
take_right_tail:
    cmp r10, r12
    jae merge_pair
take_right:
    mov rax, qword ptr [r14 + r10*8]
    inc r10
    mov qword ptr [r15 + rbx*8], rax
    inc rbx
    jmp merge_loop
pass_done:
    xchg r14, r15
    shl rbp, 1
    jmp sort_pass

compare:
    mov esi, eax
    mov edi, edx
    shr rax, 32
    shr rdx, 32
    mov rcx, rax
    cmp rcx, rdx
    cmova rcx, rdx
    lea r11, [rip + input]
    add rsi, r11
    add rdi, r11
    mov r11, rax
    test rcx, rcx
    jz compare_lengths
compare_bytes:
    movzx eax, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jb compare_less
    ja compare_greater
    inc rsi
    inc rdi
    dec rcx
    jnz compare_bytes
compare_lengths:
    cmp r11, rdx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret

emit:
    lea rdi, [rip + output]
    lea r12, [rip + input]
    xor ebx, ebx
emit_loop:
    cmp rbx, r13
    jae write_output
    mov esi, dword ptr [r14 + rbx*8]
    add rsi, r12
    mov ecx, dword ptr [r14 + rbx*8 + 4]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp emit_loop
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall
