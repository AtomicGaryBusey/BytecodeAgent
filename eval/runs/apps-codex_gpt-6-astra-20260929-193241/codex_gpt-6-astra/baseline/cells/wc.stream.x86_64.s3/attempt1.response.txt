.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz finish
    js read_loop
    add r14, rax
    lea rdi, [rsi + rax]

count_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, rdi
    jb count_loop
    jmp read_loop

finish:
    lea rdi, [rip + output]
    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

decimal:
    lea r8, [rip + digits + 32]
    mov r9, r8
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz decimal_loop
copy_digits:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb copy_digits
    ret

.bss
.balign 16
buffer:
    .zero 4096
output:
    .zero 64
digits:
    .zero 32
