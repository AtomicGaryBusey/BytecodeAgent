.globl _start
.text
_start:
    adrp x19, raw
    add x19, x19, :lo12:raw
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, commands
    add x22, x22, :lo12:commands
    adrp x24, jumps
    add x24, x24, :lo12:jumps
    adrp x11, brackets
    add x11, x11, :lo12:brackets
    mov x21, x20
    mov x23, #0
    mov x9, #0
    mov x10, #0

parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w0, [x19, x9]
    add x9, x9, #1
    cmp w0, #33
    b.eq delimiter
    cmp w0, #91
    b.eq open_bracket
    cmp w0, #93
    b.eq close_bracket
    cmp w0, #62
    b.eq save_command
    cmp w0, #60
    b.eq save_command
    cmp w0, #43
    b.eq save_command
    cmp w0, #45
    b.eq save_command
    cmp w0, #46
    b.eq save_command
    cmp w0, #44
    b.ne parse_loop
save_command:
    strb w0, [x22, x23]
    add x23, x23, #1
    b parse_loop

open_bracket:
    str w23, [x11, x10, lsl #2]
    add x10, x10, #1
    b save_command

close_bracket:
    cbz x10, invalid
    sub x10, x10, #1
    ldr w1, [x11, x10, lsl #2]
    add w2, w23, #1
    str w2, [x24, x1, lsl #2]
    add w2, w1, #1
    str w2, [x24, x23, lsl #2]
    b save_command

delimiter:
    mov x21, x9
parse_done:
    cbnz x10, invalid
    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x26, output
    add x26, x26, :lo12:output
    mov x27, #0
    mov x28, #0

execute:
    cmp x28, x23
    b.hs finished
    ldrb w0, [x22, x28]
    add x28, x28, #1
    cmp w0, #43
    b.eq increment
    cmp w0, #45
    b.eq decrement
    cmp w0, #62
    b.eq right
    cmp w0, #60
    b.eq left
    cmp w0, #91
    b.eq loop_open
    cmp w0, #93
    b.eq loop_close
    cmp w0, #46
    b.eq emit

    mov w1, #0
    cmp x21, x20
    b.hs store_input
    ldrb w1, [x19, x21]
    add x21, x21, #1
store_input:
    strb w1, [x25]
    b execute

increment:
    ldrb w1, [x25]
    add w1, w1, #1
    strb w1, [x25]
    b execute

decrement:
    ldrb w1, [x25]
    sub w1, w1, #1
    strb w1, [x25]
    b execute

right:
    add x25, x25, #1
    b execute

left:
    sub x25, x25, #1
    b execute

loop_open:
    ldrb w1, [x25]
    cbnz w1, execute
    b take_jump

loop_close:
    ldrb w1, [x25]
    cbz w1, execute
take_jump:
    sub x1, x28, #1
    ldr w28, [x24, x1, lsl #2]
    b execute

emit:
    ldrb w1, [x25]
    strb w1, [x26, x27]
    add x27, x27, #1
    b execute

finished:
    cbz x27, success
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0
success:
    mov x0, #0
    b terminate
invalid:
    mov x0, #1
terminate:
    mov x8, #93
    svc #0
    b terminate

.bss
.balign 16
raw:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
brackets:
    .zero 4000
tape:
    .zero 30000
output:
    .zero 65536
