.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
read_loop:
    mov x0, #0
    mov x1, x20
    sub x2, x20, x19
    mov x3, #65536
    add x3, x3, #1
    sub x2, x3, x2
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, x19
    bl whitespace
    bl value
    bl whitespace
    cmp x21, x20
    b.ne error
    mov x0, #1
    adrp x1, ok_message
    add x1, x1, :lo12:ok_message
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

peek:
    cmp x21, x20
    b.eq peek_eof
    ldrb w0, [x21]
    ret
peek_eof:
    mov w0, #-1
    ret

whitespace:
    cmp x21, x20
    b.eq ws_done
    ldrb w0, [x21]
    cmp w0, #32
    b.eq ws_next
    cmp w0, #9
    b.eq ws_next
    cmp w0, #10
    b.eq ws_next
    cmp w0, #13
    b.ne ws_done
ws_next:
    add x21, x21, #1
    b whitespace
ws_done:
    ret

value:
    stp x29, x30, [sp, #-16]!
    bl peek
    cmp w0, #34
    b.eq value_string
    cmp w0, #123
    b.eq object
    cmp w0, #91
    b.eq array
    cmp w0, #116
    b.eq literal_true
    cmp w0, #102
    b.eq literal_false
    cmp w0, #110
    b.eq literal_null
    cmp w0, #45
    b.eq value_number
    sub w1, w0, #48
    cmp w1, #9
    b.ls value_number
    b error
value_string:
    bl string
    b value_done
value_number:
    bl number
    b value_done
literal_true:
    adrp x4, true_text
    add x4, x4, :lo12:true_text
    b literal
literal_false:
    adrp x4, false_text
    add x4, x4, :lo12:false_text
    b literal
literal_null:
    adrp x4, null_text
    add x4, x4, :lo12:null_text
literal:
    ldrb w5, [x4], #1
    cbz w5, value_done
    bl peek
    cmp w0, w5
    b.ne error
    add x21, x21, #1
    b literal

object:
    add x21, x21, #1
    bl whitespace
    bl peek
    cmp w0, #125
    b.eq container_end
object_member:
    bl peek
    cmp w0, #34
    b.ne error
    bl string
    bl whitespace
    bl peek
    cmp w0, #58
    b.ne error
    add x21, x21, #1
    bl whitespace
    bl value
    bl whitespace
    bl peek
    cmp w0, #125
    b.eq container_end
    cmp w0, #44
    b.ne error
    add x21, x21, #1
    bl whitespace
    b object_member

array:
    add x21, x21, #1
    bl whitespace
    bl peek
    cmp w0, #93
    b.eq container_end
array_element:
    bl value
    bl whitespace
    bl peek
    cmp w0, #93
    b.eq container_end
    cmp w0, #44
    b.ne error
    add x21, x21, #1
    bl whitespace
    b array_element
container_end:
    add x21, x21, #1
value_done:
    ldp x29, x30, [sp], #16
    ret

string:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
string_loop:
    bl peek
    cmp w0, #-1
    b.eq error
    cmp w0, #32
    b.lo error
    cmp w0, #34
    b.eq string_end
    cmp w0, #92
    b.eq escape
    add x21, x21, #1
    b string_loop
escape:
    add x21, x21, #1
    bl peek
    cmp w0, #117
    b.eq unicode
    cmp w0, #34
    b.eq escape_ok
    cmp w0, #92
    b.eq escape_ok
    cmp w0, #47
    b.eq escape_ok
    cmp w0, #98
    b.eq escape_ok
    cmp w0, #102
    b.eq escape_ok
    cmp w0, #110
    b.eq escape_ok
    cmp w0, #114
    b.eq escape_ok
    cmp w0, #116
    b.ne error
escape_ok:
    add x21, x21, #1
    b string_loop
unicode:
    add x21, x21, #1
    mov w4, #4
hex_loop:
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls hex_ok
    orr w1, w0, #32
    sub w1, w1, #97
    cmp w1, #5
    b.hi error
hex_ok:
    add x21, x21, #1
    subs w4, w4, #1
    b.ne hex_loop
    b string_loop
string_end:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

number:
    stp x29, x30, [sp, #-16]!
    bl peek
    cmp w0, #45
    b.ne integer
    add x21, x21, #1
    bl peek
integer:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
    cmp w0, #48
    b.eq zero_integer
integer_digits:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls integer_digits
    b fraction
zero_integer:
    add x21, x21, #1
    bl peek
fraction:
    cmp w0, #46
    b.ne exponent
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
fraction_digits:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls fraction_digits
exponent:
    cmp w0, #101
    b.eq exponent_start
    cmp w0, #69
    b.ne number_done
exponent_start:
    add x21, x21, #1
    bl peek
    cmp w0, #43
    b.eq exponent_sign
    cmp w0, #45
    b.ne exponent_required
exponent_sign:
    add x21, x21, #1
    bl peek
exponent_required:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
exponent_digits:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls exponent_digits
number_done:
    ldp x29, x30, [sp], #16
    ret

error:
    sub x22, x21, x19
    mov x0, #1
    adrp x1, error_message
    add x1, x1, :lo12:error_message
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x4, output
    add x4, x4, :lo12:output
    add x4, x4, #32
    mov w5, #10
    strb w5, [x4, #-1]!
    mov x6, x4
    mov x7, #10
decimal_loop:
    udiv x3, x22, x7
    msub x5, x3, x7, x22
    add w5, w5, #48
    strb w5, [x4, #-1]!
    mov x22, x3
    cbnz x22, decimal_loop
    mov x0, #1
    mov x1, x4
    sub x2, x6, x4
    add x2, x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
