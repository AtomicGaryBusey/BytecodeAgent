.globl _start

.section .rodata
hex:
.ascii "0123456789abcdef"

.bss
.balign 16
input:
.zero 65537
output:
.zero 323600

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output]
    lea r13, [rip + input]
    lea rbx, [rip + hex]
    xor r14d, r14d

row_loop:
    cmp r14, r12
    jae final_offset
    mov r9, r12
    sub r9, r14
    cmp r9, 16
    jbe row_size_ready
    mov r9d, 16
row_size_ready:
    mov eax, r14d
    call print_offset
    mov ax, 0x2020
    stosw
    xor r8d, r8d

hex_loop:
    cmp r8, r9
    jae blank_slot
    movzx edx, byte ptr [r13 + r8]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [rbx + rax]
    stosb
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done
blank_slot:
    mov ax, 0x2020
    stosw
    stosb
slot_done:
    cmp r8d, 7
    jne no_extra_space
    mov al, 32
    stosb
no_extra_space:
    inc r8d
    cmp r8d, 16
    jne hex_loop

    mov ax, 0x7c20
    stosw
    xor r8d, r8d
ascii_loop:
    mov al, byte ptr [r13 + r8]
    cmp al, 32
    jb replace_char
    cmp al, 126
    jbe emit_char
replace_char:
    mov al, 46
emit_char:
    stosb
    inc r8d
    cmp r8, r9
    jb ascii_loop
    mov ax, 0x0a7c
    stosw
    add r13, r9
    add r14, r9
    jmp row_loop

final_offset:
    mov eax, r12d
    call print_offset
    mov al, 10
    stosb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_offset:
    mov edx, eax
    mov ecx, 8
offset_digit:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [rbx + rax]
    stosb
    dec ecx
    jnz offset_digit
    ret
