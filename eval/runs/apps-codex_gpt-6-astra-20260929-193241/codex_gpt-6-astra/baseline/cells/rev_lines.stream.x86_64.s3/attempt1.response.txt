.globl _start

.bss
input:
.zero 65537
output:
.zero 65537

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    mov r8, r12
    mov r9, r12
    lea r14, [r12 + r13]
    lea r10, [rip + output]

scan:
    cmp r9, r14
    je final_remainder
    cmp byte ptr [r9], 10
    je complete_line
    inc r9
    jmp scan

complete_line:
    mov rbx, r9
reverse_line:
    cmp rbx, r8
    je line_newline
    dec rbx
    mov al, byte ptr [rbx]
    mov byte ptr [r10], al
    inc r10
    jmp reverse_line

line_newline:
    mov byte ptr [r10], 10
    inc r10
    inc r9
    mov r8, r9
    jmp scan

final_remainder:
    cmp r8, r14
    je write_output
    mov rbx, r14
reverse_final:
    dec rbx
    mov al, byte ptr [rbx]
    mov byte ptr [r10], al
    inc r10
    cmp rbx, r8
    jne reverse_final
    mov byte ptr [r10], 10
    inc r10

write_output:
    lea rsi, [rip + output]
    mov rdx, r10
    sub rdx, rsi
    jz done
    mov eax, 1
    mov edi, 1
    syscall

done:
    mov eax, 60
    xor edi, edi
    syscall
