.globl _start
.bss
buffer:
.zero 65537
outbuf:
.zero 32
.text
_start:
    lea r12, [rip + buffer]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    add r13, r12
    mov r14, r12
    call value
    call whitespace
    cmp r14, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

peek:
    cmp r14, r13
    jae error
    movzx eax, byte ptr [r14]
    ret

whitespace:
    cmp r14, r13
    jae ws_done
    movzx eax, byte ptr [r14]
    cmp al, 32
    je ws_next
    cmp al, 9
    je ws_next
    cmp al, 10
    je ws_next
    cmp al, 13
    jne ws_done
ws_next:
    inc r14
    jmp whitespace
ws_done:
    ret

value:
    call whitespace
    call peek
    cmp al, 34
    je string
    cmp al, 123
    je object
    cmp al, 91
    je array
    cmp al, 116
    je true_value
    cmp al, 102
    je false_value
    cmp al, 110
    je null_value
    cmp al, 45
    je number
    cmp al, 48
    jb error
    cmp al, 57
    ja error
    jmp number

true_value:
    lea rsi, [rip + true_text]
    jmp literal
false_value:
    lea rsi, [rip + false_text]
    jmp literal
null_value:
    lea rsi, [rip + null_text]
literal:
    mov dl, byte ptr [rsi]
    test dl, dl
    jz literal_done
    call peek
    cmp al, dl
    jne error
    inc rsi
    inc r14
    jmp literal
literal_done:
    ret

string:
    inc r14
string_loop:
    call peek
    cmp al, 34
    je string_done
    cmp al, 32
    jb error
    cmp al, 92
    je escape
    inc r14
    jmp string_loop
string_done:
    inc r14
    ret
escape:
    inc r14
    call peek
    cmp al, 117
    je unicode
    cmp al, 34
    je escape_done
    cmp al, 92
    je escape_done
    cmp al, 47
    je escape_done
    cmp al, 98
    je escape_done
    cmp al, 102
    je escape_done
    cmp al, 110
    je escape_done
    cmp al, 114
    je escape_done
    cmp al, 116
    jne error
escape_done:
    inc r14
    jmp string_loop
unicode:
    inc r14
    mov ecx, 4
hex_loop:
    call peek
    cmp al, 48
    jb error
    cmp al, 57
    jbe hex_good
    or al, 32
    cmp al, 97
    jb error
    cmp al, 102
    ja error
hex_good:
    inc r14
    dec ecx
    jnz hex_loop
    jmp string_loop

number:
    cmp al, 45
    jne number_int
    inc r14
    call peek
number_int:
    cmp al, 48
    je zero_int
    cmp al, 49
    jb error
    cmp al, 57
    ja error
    call digits
    jmp fraction
zero_int:
    inc r14
fraction:
    cmp r14, r13
    je number_done
    cmp byte ptr [r14], 46
    jne exponent
    inc r14
    call required_digits
exponent:
    cmp r14, r13
    je number_done
    mov al, byte ptr [r14]
    or al, 32
    cmp al, 101
    jne number_done
    inc r14
    call peek
    cmp al, 43
    je exp_sign
    cmp al, 45
    jne exp_digits
exp_sign:
    inc r14
exp_digits:
    call required_digits
number_done:
    ret
required_digits:
    call peek
    cmp al, 48
    jb error
    cmp al, 57
    ja error
digits:
    inc r14
    cmp r14, r13
    je digits_done
    mov al, byte ptr [r14]
    cmp al, 48
    jb digits_done
    cmp al, 57
    jbe digits
digits_done:
    ret

array:
    inc r14
    call whitespace
    call peek
    cmp al, 93
    je container_done
array_value:
    call value
    call whitespace
    call peek
    cmp al, 93
    je container_done
    cmp al, 44
    jne error
    inc r14
    jmp array_value

object:
    inc r14
    call whitespace
    call peek
    cmp al, 125
    je container_done
object_member:
    call whitespace
    call peek
    cmp al, 34
    jne error
    call string
    call whitespace
    call peek
    cmp al, 58
    jne error
    inc r14
    call value
    call whitespace
    call peek
    cmp al, 125
    je container_done
    cmp al, 44
    jne error
    inc r14
    jmp object_member
container_done:
    inc r14
    ret

error:
    sub r14, r12
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov rax, r14
    mov r8d, 10
decimal:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal
    sub rsi, 9
    mov rax, 0x746120726f727265
    mov qword ptr [rsi], rax
    mov byte ptr [rsi + 8], 32
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok_msg:
.ascii "ok\n"
true_text:
.asciz "true"
false_text:
.asciz "false"
null_text:
.asciz "null"
