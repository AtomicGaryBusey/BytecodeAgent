.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x22, jumps
    add x22, x22, :lo12:jumps
    adrp x23, brackets
    add x23, x23, :lo12:brackets
    mov x24, #0
    mov x25, #0
    mov x26, x20
    mov x9, #0

parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w12, [x19, x9]
    add x9, x9, #1
    cmp w12, #33
    b.eq delimiter
    cmp w12, #91
    b.eq open_bracket
    cmp w12, #93
    b.eq close_bracket
    cmp w12, #62
    b.eq save_command
    cmp w12, #60
    b.eq save_command
    cmp w12, #43
    b.eq save_command
    cmp w12, #45
    b.eq save_command
    cmp w12, #46
    b.eq save_command
    cmp w12, #44
    b.ne parse_loop
save_command:
    strb w12, [x21, x24]
    add x24, x24, #1
    b parse_loop

open_bracket:
    str w24, [x23, x25, lsl #2]
    add x25, x25, #1
    b save_command

close_bracket:
    cbz x25, invalid
    sub x25, x25, #1
    ldr w13, [x23, x25, lsl #2]
    add w14, w24, #1
    str w14, [x22, x13, lsl #2]
    add w14, w13, #1
    str w14, [x22, x24, lsl #2]
    b save_command

delimiter:
    mov x26, x9
parse_done:
    cbnz x25, invalid
    adrp x27, tape
    add x27, x27, :lo12:tape
    adrp x28, output_buffer
    add x28, x28, :lo12:output_buffer
    mov x9, #0
    mov x10, #0
    mov x11, #0

execute:
    cmp x9, x24
    b.hs finished
    ldrb w12, [x21, x9]
    add x9, x9, #1
    cmp w12, #43
    b.eq increment
    cmp w12, #45
    b.eq decrement
    cmp w12, #62
    b.eq right
    cmp w12, #60
    b.eq left
    cmp w12, #46
    b.eq output_byte
    cmp w12, #44
    b.eq input_byte
    ldrb w13, [x27, x10]
    cmp w12, #91
    b.eq loop_open
    cbz w13, execute
    b jump
loop_open:
    cbnz w13, execute
jump:
    sub x9, x9, #1
    ldr w9, [x22, x9, lsl #2]
    b execute

increment:
    ldrb w13, [x27, x10]
    add w13, w13, #1
    strb w13, [x27, x10]
    b execute
decrement:
    ldrb w13, [x27, x10]
    sub w13, w13, #1
    strb w13, [x27, x10]
    b execute
right:
    add x10, x10, #1
    b execute
left:
    sub x10, x10, #1
    b execute
output_byte:
    ldrb w13, [x27, x10]
    strb w13, [x28, x11]
    add x11, x11, #1
    b execute
input_byte:
    mov w13, #0
    cmp x26, x20
    b.hs store_input
    ldrb w13, [x19, x26]
    add x26, x26, #1
store_input:
    strb w13, [x27, x10]
    b execute

finished:
    cbz x11, success
    mov x0, #1
    mov x1, x28
    mov x2, x11
    mov x8, #64
    svc #0
success:
    mov x0, #0
    b terminate
invalid:
    mov x0, #1
terminate:
    mov x8, #93
    svc #0
    b terminate

.bss
.balign 16
input_buffer:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
brackets:
    .zero 4000
tape:
    .zero 30000
output_buffer:
    .zero 65536
