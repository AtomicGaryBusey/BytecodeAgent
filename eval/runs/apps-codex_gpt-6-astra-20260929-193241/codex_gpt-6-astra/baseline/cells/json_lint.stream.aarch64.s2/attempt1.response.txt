.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    bl whitespace
    bl value
    bl whitespace
    cmp x21, x20
    b.ne error
    mov x0, #1
    adrp x1, ok_text
    add x1, x1, :lo12:ok_text
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    b terminate

// Return the current byte, or -1 at end of input.
peek:
    mov w0, #-1
    cmp x21, x20
    b.hs peek_done
    ldrb w0, [x19, x21]
peek_done:
    ret

whitespace:
    cmp x21, x20
    b.hs whitespace_done
    ldrb w0, [x19, x21]
    cmp w0, #32
    b.eq whitespace_next
    cmp w0, #9
    b.eq whitespace_next
    cmp w0, #10
    b.eq whitespace_next
    cmp w0, #13
    b.ne whitespace_done
whitespace_next:
    add x21, x21, #1
    b whitespace
whitespace_done:
    ret

value:
    stp x29, x30, [sp, #-16]!
    bl peek
    cmp w0, #34
    b.eq value_string
    cmp w0, #123
    b.eq value_object
    cmp w0, #91
    b.eq value_array
    cmp w0, #116
    b.eq value_true
    cmp w0, #102
    b.eq value_false
    cmp w0, #110
    b.eq value_null
    cmp w0, #45
    b.eq value_number
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
value_number:
    bl number
    b value_done
value_string:
    bl string
    b value_done
value_object:
    bl object
    b value_done
value_array:
    bl array
    b value_done
value_true:
    adrp x3, true_text
    add x3, x3, :lo12:true_text
    b value_literal
value_false:
    adrp x3, false_text
    add x3, x3, :lo12:false_text
    b value_literal
value_null:
    adrp x3, null_text
    add x3, x3, :lo12:null_text
value_literal:
    ldrb w1, [x3], #1
    cbz w1, value_done
    bl peek
    cmp w0, w1
    b.ne error
    add x21, x21, #1
    b value_literal
value_done:
    ldp x29, x30, [sp], #16
    ret

string:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
string_loop:
    bl peek
    cmp w0, #-1
    b.eq error
    cmp w0, #32
    b.lo error
    cmp w0, #34
    b.eq string_done
    cmp w0, #92
    b.eq string_escape
    add x21, x21, #1
    b string_loop
string_escape:
    add x21, x21, #1
    bl peek
    cmp w0, #117
    b.eq string_unicode
    cmp w0, #34
    b.eq string_accept
    cmp w0, #92
    b.eq string_accept
    cmp w0, #47
    b.eq string_accept
    cmp w0, #98
    b.eq string_accept
    cmp w0, #102
    b.eq string_accept
    cmp w0, #110
    b.eq string_accept
    cmp w0, #114
    b.eq string_accept
    cmp w0, #116
    b.ne error
string_accept:
    add x21, x21, #1
    b string_loop
string_unicode:
    mov w4, #4
unicode_loop:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls unicode_accept
    orr w1, w0, #32
    sub w1, w1, #97
    cmp w1, #5
    b.hi error
unicode_accept:
    subs w4, w4, #1
    b.ne unicode_loop
    add x21, x21, #1
    b string_loop
string_done:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

number:
    stp x29, x30, [sp, #-16]!
    bl peek
    cmp w0, #45
    b.ne number_integer
    add x21, x21, #1
    bl peek
number_integer:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
    add x21, x21, #1
    cmp w0, #48
    b.eq number_fraction
number_digits:
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.hi number_fraction
    add x21, x21, #1
    b number_digits
number_fraction:
    bl peek
    cmp w0, #46
    b.ne number_exponent
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
fraction_digits:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls fraction_digits
number_exponent:
    cmp w0, #101
    b.eq exponent_start
    cmp w0, #69
    b.ne number_done
exponent_start:
    add x21, x21, #1
    bl peek
    cmp w0, #43
    b.eq exponent_sign
    cmp w0, #45
    b.ne exponent_required
exponent_sign:
    add x21, x21, #1
    bl peek
exponent_required:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
exponent_digits:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls exponent_digits
number_done:
    ldp x29, x30, [sp], #16
    ret

array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl whitespace
    bl peek
    cmp w0, #93
    b.eq container_done
array_value:
    bl value
    bl whitespace
    bl peek
    cmp w0, #93
    b.eq container_done
    cmp w0, #44
    b.ne error
    add x21, x21, #1
    bl whitespace
    b array_value

object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl whitespace
    bl peek
    cmp w0, #125
    b.eq container_done
object_member:
    bl peek
    cmp w0, #34
    b.ne error
    bl string
    bl whitespace
    bl peek
    cmp w0, #58
    b.ne error
    add x21, x21, #1
    bl whitespace
    bl value
    bl whitespace
    bl peek
    cmp w0, #125
    b.eq container_done
    cmp w0, #44
    b.ne error
    add x21, x21, #1
    bl whitespace
    b object_member
container_done:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

error:
    adrp x1, output
    add x1, x1, :lo12:output
    add x4, x1, #32
    sub x4, x4, #1
    mov w0, #10
    strb w0, [x4]
    mov x5, #10
    mov x6, x21
decimal_loop:
    udiv x7, x6, x5
    msub x0, x7, x5, x6
    add w0, w0, #48
    strb w0, [x4, #-1]!
    mov x6, x7
    cbnz x6, decimal_loop
    mov x22, x4
    add x23, x1, #32
    sub x23, x23, x4
    mov x0, #1
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #1
    mov x1, x22
    mov x2, x23
    mov x8, #64
    svc #0
    mov x0, #1
terminate:
    mov x8, #93
    svc #0

.section .rodata
ok_text:
    .ascii "ok\n"
error_text:
    .ascii "error at "
true_text:
    .asciz "true"
false_text:
    .asciz "false"
null_text:
    .asciz "null"

.bss
.balign 16
input:
    .zero 65537
output:
    .zero 32
