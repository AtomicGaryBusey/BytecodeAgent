.globl _start

.bss
.balign 16
input:
.zero 65537
output:
.zero 323600

.section .rodata
hex:
.ascii "0123456789abcdef"

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output]
    lea r15, [rip + input]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_length_ready
    mov r14d, 16

row_length_ready:
    mov eax, r13d
    call print_hex8
    mov ax, 0x2020
    stosw
    xor ebx, ebx

hex_slots:
    cmp rbx, r14
    jae empty_slot
    movzx edx, byte ptr [r15 + rbx]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [hex + rax]
    stosb
    and edx, 15
    mov al, byte ptr [hex + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done

empty_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb

slot_done:
    cmp ebx, 7
    jne no_extra_space
    mov al, 32
    stosb
no_extra_space:
    inc ebx
    cmp ebx, 16
    jb hex_slots

    mov al, 32
    stosb
    mov al, 124
    stosb
    xor ebx, ebx

ascii_loop:
    mov al, byte ptr [r15 + rbx]
    cmp al, 32
    jb nonprintable
    cmp al, 126
    jbe emit_character
nonprintable:
    mov al, 46
emit_character:
    stosb
    inc ebx
    cmp rbx, r14
    jb ascii_loop

    mov al, 124
    stosb
    mov al, 10
    stosb
    add r15, r14
    add r13, r14
    jmp row_loop

final_offset:
    mov eax, r12d
    call print_hex8
    mov al, 10
    stosb

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

print_hex8:
    mov r8d, eax
    mov ecx, 28
hex8_loop:
    mov edx, r8d
    shr edx, cl
    and edx, 15
    mov al, byte ptr [hex + rdx]
    stosb
    sub ecx, 4
    jns hex8_loop
    ret
