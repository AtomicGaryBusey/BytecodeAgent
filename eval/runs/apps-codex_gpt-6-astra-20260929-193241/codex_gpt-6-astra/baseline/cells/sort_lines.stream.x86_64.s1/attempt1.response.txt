.globl _start

.bss
.balign 16
buffer:
.zero 32769
records:
.zero 16000

.section .rodata
newline:
.byte 10

.text
_start:
    lea r12, [rip + buffer]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r8, [rip + records]
    xor r14d, r14d
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r13
    jae parse_end
    cmp byte ptr [r12 + rcx], 10
    jne parse_next
    mov rax, r14
    shl rax, 4
    lea rsi, [r12 + rdx]
    mov qword ptr [r8 + rax], rsi
    mov rsi, rcx
    sub rsi, rdx
    mov qword ptr [r8 + rax + 8], rsi
    inc r14
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_end:
    cmp rdx, r13
    je sort_begin
    mov rax, r14
    shl rax, 4
    lea rsi, [r12 + rdx]
    mov qword ptr [r8 + rax], rsi
    sub r13, rdx
    mov qword ptr [r8 + rax + 8], r13
    inc r14

sort_begin:
    mov r9d, 1
sort_outer:
    cmp r9, r14
    jae output_begin
    mov r10, r9
    shl r10, 4
    mov r12, qword ptr [r8 + r10]
    mov r13, qword ptr [r8 + r10 + 8]
sort_inner:
    test r10, r10
    jz sort_place
    mov rbx, qword ptr [r8 + r10 - 16]
    mov rbp, qword ptr [r8 + r10 - 8]
    mov rcx, r13
    cmp rcx, rbp
    cmova rcx, rbp
    test rcx, rcx
    jz compare_lengths
    mov rsi, r12
    mov rdi, rbx
    repe cmpsb
    jb sort_shift
    ja sort_place
compare_lengths:
    cmp r13, rbp
    jae sort_place
sort_shift:
    mov qword ptr [r8 + r10], rbx
    mov qword ptr [r8 + r10 + 8], rbp
    sub r10, 16
    jmp sort_inner
sort_place:
    mov qword ptr [r8 + r10], r12
    mov qword ptr [r8 + r10 + 8], r13
    inc r9
    jmp sort_outer

output_begin:
    xor r12d, r12d
output_loop:
    cmp r12, r14
    jae finish
    mov rax, r12
    shl rax, 4
    mov rsi, qword ptr [r8 + rax]
    mov rdx, qword ptr [r8 + rax + 8]
    test rdx, rdx
    jz output_newline
    mov eax, 1
    mov edi, 1
    syscall
output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r12
    jmp output_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall
