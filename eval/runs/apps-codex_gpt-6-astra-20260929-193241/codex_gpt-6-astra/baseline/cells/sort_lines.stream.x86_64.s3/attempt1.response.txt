.globl _start

.bss
.balign 16
buffer:
.zero 32769
.balign 16
lines:
.zero 16000

.section .rodata
newline:
.byte 10

.text
_start:
    lea r15, [rip + buffer]
    xor r12d, r12d
.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz .Lparse
    add r12, rax
    jmp .Lread

.Lparse:
    lea rbp, [rip + lines]
    xor r14d, r14d
    xor ecx, ecx
    xor edx, edx
.Lscan:
    cmp rcx, r12
    jae .Lremainder
    cmp byte ptr [r15 + rcx], 10
    jne .Lnext
    mov rax, r14
    shl rax, 4
    lea rsi, [r15 + rdx]
    mov qword ptr [rbp + rax], rsi
    mov rsi, rcx
    sub rsi, rdx
    mov qword ptr [rbp + rax + 8], rsi
    inc r14
    lea rdx, [rcx + 1]
.Lnext:
    inc rcx
    jmp .Lscan

.Lremainder:
    cmp rdx, r12
    je .Lsort
    mov rax, r14
    shl rax, 4
    lea rsi, [r15 + rdx]
    mov qword ptr [rbp + rax], rsi
    sub r12, rdx
    mov qword ptr [rbp + rax + 8], r12
    inc r14

.Lsort:
    mov r15, r14
    mov r14d, 1
.Lsort_next:
    cmp r14, r15
    jae .Loutput
    mov rax, r14
    shl rax, 4
    mov r12, qword ptr [rbp + rax]
    mov r13, qword ptr [rbp + rax + 8]
    xor r8d, r8d
    mov r9, r14
.Lbinary:
    cmp r8, r9
    jae .Linsert
    lea r10, [r8 + r9]
    shr r10, 1
    mov rax, r10
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rdi, qword ptr [rbp + rax + 8]
    mov rcx, rdi
    cmp rcx, r13
    cmova rcx, r13
    xor edx, edx
.Lcompare:
    cmp rdx, rcx
    jae .Llength
    movzx eax, byte ptr [rsi + rdx]
    movzx ebx, byte ptr [r12 + rdx]
    cmp eax, ebx
    jb .Llower
    ja .Lupper
    inc rdx
    jmp .Lcompare
.Llength:
    cmp rdi, r13
    ja .Lupper
.Llower:
    lea r8, [r10 + 1]
    jmp .Lbinary
.Lupper:
    mov r9, r10
    jmp .Lbinary

.Linsert:
    mov rcx, r14
    shl rcx, 4
    shl r8, 4
.Lshift:
    cmp rcx, r8
    je .Lstore
    mov rax, qword ptr [rbp + rcx - 16]
    mov rdx, qword ptr [rbp + rcx - 8]
    mov qword ptr [rbp + rcx], rax
    mov qword ptr [rbp + rcx + 8], rdx
    sub rcx, 16
    jmp .Lshift
.Lstore:
    mov qword ptr [rbp + r8], r12
    mov qword ptr [rbp + r8 + 8], r13
    inc r14
    jmp .Lsort_next

.Loutput:
    xor r12d, r12d
.Lwrite_line:
    cmp r12, r15
    jae .Lexit
    mov rax, r12
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rdx, qword ptr [rbp + rax + 8]
    test rdx, rdx
    jz .Lwrite_newline
    mov eax, 1
    mov edi, 1
    syscall
.Lwrite_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r12
    jmp .Lwrite_line
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall
