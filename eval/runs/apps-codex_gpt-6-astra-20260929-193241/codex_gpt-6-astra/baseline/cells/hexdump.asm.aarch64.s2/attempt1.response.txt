.globl _start

.text
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    adrp x22, output
    add x22, x22, :lo12:output
    adrp x23, digits
    add x23, x23, :lo12:digits

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_offset
    sub x24, x20, x21
    mov x0, #16
    cmp x24, x0
    csel x24, x24, x0, lo
    add x25, x19, x21

    mov x0, x21
    mov x1, x22
    bl hex8
    mov w5, #32
    strb w5, [x1], #1
    strb w5, [x1], #1
    mov x26, #0

hex_slots:
    cmp x26, x24
    b.hs empty_slot
    ldrb w6, [x25, x26]
    lsr w7, w6, #4
    ldrb w7, [x23, x7]
    strb w7, [x1], #1
    and w6, w6, #15
    ldrb w7, [x23, x6]
    strb w7, [x1], #1
    strb w5, [x1], #1
    b slot_done

empty_slot:
    strb w5, [x1], #1
    strb w5, [x1], #1
    strb w5, [x1], #1

slot_done:
    cmp x26, #7
    b.ne no_extra_space
    strb w5, [x1], #1
no_extra_space:
    add x26, x26, #1
    cmp x26, #16
    b.lo hex_slots

    strb w5, [x1], #1
    mov w5, #124
    strb w5, [x1], #1
    mov x26, #0

ascii_loop:
    ldrb w6, [x25, x26]
    sub w7, w6, #32
    cmp w7, #94
    mov w7, #46
    csel w6, w6, w7, ls
    strb w6, [x1], #1
    add x26, x26, #1
    cmp x26, x24
    b.lo ascii_loop

    strb w5, [x1], #1
    mov w5, #10
    strb w5, [x1], #1
    sub x2, x1, x22
    mov x1, x22
    mov x0, #1
    mov x8, #64
    svc #0

    add x21, x21, x24
    b row_loop

final_offset:
    mov x0, x20
    mov x1, x22
    bl hex8
    mov w5, #10
    strb w5, [x1]
    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

hex8:
    mov w2, #28
hex8_loop:
    lsr w3, w0, w2
    and w3, w3, #15
    ldrb w4, [x23, x3]
    strb w4, [x1], #1
    subs w2, w2, #4
    b.ge hex8_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 69632
output:
    .zero 80
