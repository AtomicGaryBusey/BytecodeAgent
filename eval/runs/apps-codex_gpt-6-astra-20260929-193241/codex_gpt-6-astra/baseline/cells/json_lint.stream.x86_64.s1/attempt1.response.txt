.globl _start
.text
_start:
    xor r12d, r12d
    lea r13, [rip + buffer]
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ebx, ebx
    call whitespace
    call value
    call whitespace
    cmp rbx, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

peek:
    cmp rbx, r12
    jae error
    movzx eax, byte ptr [r13 + rbx]
    ret

whitespace:
    cmp rbx, r12
    jae ws_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 32
    je ws_next
    cmp al, 9
    je ws_next
    cmp al, 10
    je ws_next
    cmp al, 13
    jne ws_done
ws_next:
    inc rbx
    jmp whitespace
ws_done:
    ret

value:
    call peek
    cmp al, '{'
    je object
    cmp al, '['
    je array
    cmp al, '"'
    je string
    cmp al, 't'
    je literal_true
    cmp al, 'f'
    je literal_false
    cmp al, 'n'
    je literal_null
    cmp al, '-'
    je number
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp number

literal_true:
    lea rsi, [rip + true_text]
    mov ecx, 4
    jmp literal
literal_false:
    lea rsi, [rip + false_text]
    mov ecx, 5
    jmp literal
literal_null:
    lea rsi, [rip + null_text]
    mov ecx, 4
literal:
    call peek
    cmp al, byte ptr [rsi]
    jne error
    inc rsi
    inc rbx
    dec ecx
    jnz literal
    ret

object:
    inc rbx
    call whitespace
    call peek
    cmp al, '}'
    je container_end
object_member:
    cmp al, '"'
    jne error
    call string
    call whitespace
    call peek
    cmp al, ':'
    jne error
    inc rbx
    call whitespace
    call value
    call whitespace
    call peek
    cmp al, '}'
    je container_end
    cmp al, ','
    jne error
    inc rbx
    call whitespace
    call peek
    jmp object_member

array:
    inc rbx
    call whitespace
    call peek
    cmp al, ']'
    je container_end
array_value:
    call value
    call whitespace
    call peek
    cmp al, ']'
    je container_end
    cmp al, ','
    jne error
    inc rbx
    call whitespace
    jmp array_value
container_end:
    inc rbx
    ret

string:
    inc rbx
string_loop:
    call peek
    cmp al, 32
    jb error
    cmp al, '"'
    je string_end
    cmp al, 92
    je string_escape
    inc rbx
    jmp string_loop
string_end:
    inc rbx
    ret
string_escape:
    inc rbx
    call peek
    cmp al, 'u'
    je unicode_escape
    cmp al, '"'
    je escape_done
    cmp al, 92
    je escape_done
    cmp al, '/'
    je escape_done
    cmp al, 'b'
    je escape_done
    cmp al, 'f'
    je escape_done
    cmp al, 'n'
    je escape_done
    cmp al, 'r'
    je escape_done
    cmp al, 't'
    jne error
escape_done:
    inc rbx
    jmp string_loop
unicode_escape:
    inc rbx
    mov ecx, 4
hex_loop:
    call peek
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe hex_done
    or al, 32
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
hex_done:
    inc rbx
    dec ecx
    jnz hex_loop
    jmp string_loop

number:
    cmp al, '-'
    jne number_int
    inc rbx
    call peek
number_int:
    cmp al, '0'
    je zero_int
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
integer_digits:
    inc rbx
    cmp rbx, r12
    jae number_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb fraction_check
    cmp al, '9'
    jbe integer_digits
    jmp fraction_check
zero_int:
    inc rbx
    cmp rbx, r12
    jae number_done
    movzx eax, byte ptr [r13 + rbx]
fraction_check:
    cmp al, '.'
    jne exponent_check
    inc rbx
    call peek
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
fraction_digits:
    inc rbx
    cmp rbx, r12
    jae number_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb exponent_check
    cmp al, '9'
    jbe fraction_digits
exponent_check:
    cmp al, 'e'
    je exponent
    cmp al, 'E'
    jne number_done
exponent:
    inc rbx
    call peek
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_first
exponent_sign:
    inc rbx
    call peek
exponent_first:
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
exponent_digits:
    inc rbx
    cmp rbx, r12
    jae number_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe exponent_digits
number_done:
    ret

error:
    lea rsi, [rip + output + 31]
    mov byte ptr [rsi], 10
    mov ecx, 1
    mov rax, rbx
    mov r8d, 10
decimal_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz decimal_loop
    sub rsi, 9
    mov rax, qword ptr [rip + error_msg]
    mov qword ptr [rsi], rax
    mov byte ptr [rsi + 8], 32
    lea edx, [ecx + 9]
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at "
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.bss
buffer:
    .zero 65537
output:
    .zero 32
