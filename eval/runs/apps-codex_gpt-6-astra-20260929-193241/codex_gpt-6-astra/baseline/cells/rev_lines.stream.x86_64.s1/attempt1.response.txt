.bss
input:
.zero 65537
output:
.zero 65537

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + output]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

scan:
    cmp r8, r13
    je final_line
    cmp byte ptr [r12 + r8], 10
    je complete_line
    inc r8
    jmp scan

complete_line:
    mov rbx, r8
reverse_line:
    cmp rbx, r9
    je append_newline
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_line

append_newline:
    mov byte ptr [r14 + r10], 10
    inc r10
    inc r8
    mov r9, r8
    jmp scan

final_line:
    cmp r9, r13
    je emit
    mov rbx, r13
reverse_final:
    cmp rbx, r9
    je final_newline
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_final

final_newline:
    mov byte ptr [r14 + r10], 10
    inc r10

emit:
    test r10, r10
    jz done
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    syscall

done:
    mov eax, 60
    xor edi, edi
    syscall
