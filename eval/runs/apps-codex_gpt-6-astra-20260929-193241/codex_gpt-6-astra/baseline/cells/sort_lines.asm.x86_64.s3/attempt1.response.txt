.globl _start

.bss
.balign 16
buffer:
.zero 32769
descriptors:
.zero 16000
sorted:
.zero 8000
output:
.zero 33768

.text
_start:
    lea r13, [rip + buffer]
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_start
    add r12, rax
    jmp read_loop

parse_start:
    lea r15, [rip + descriptors]
    xor ecx, ecx
    xor edx, edx
    xor ebx, ebx
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [r13 + rcx], 10
    jne parse_next
    mov rax, rbx
    shl rax, 4
    lea rsi, [r13 + rdx]
    mov qword ptr [r15 + rax], rsi
    mov rsi, rcx
    sub rsi, rdx
    mov qword ptr [r15 + rax + 8], rsi
    inc rbx
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_end:
    cmp rdx, r12
    je sort_start
    mov rax, rbx
    shl rax, 4
    lea rsi, [r13 + rdx]
    mov qword ptr [r15 + rax], rsi
    sub r12, rdx
    mov qword ptr [r15 + rax + 8], r12
    inc rbx

sort_start:
    mov r12, rbx
    lea r14, [rip + sorted]
    xor r13d, r13d
insert_loop:
    cmp r13, r12
    jae build_output
    xor ebx, ebx
    mov rbp, r13
binary_loop:
    cmp rbx, rbp
    jae insert_at
    lea r11, [rbx + rbp]
    shr r11, 1
    mov rdi, r13
    shl rdi, 4
    add rdi, r15
    mov rsi, qword ptr [r14 + r11*8]
    call compare
    test eax, eax
    js lower_half
    lea rbx, [r11 + 1]
    jmp binary_loop
lower_half:
    mov rbp, r11
    jmp binary_loop
insert_at:
    mov rcx, r13
shift_loop:
    cmp rcx, rbx
    je store_item
    mov rax, qword ptr [r14 + rcx*8 - 8]
    mov qword ptr [r14 + rcx*8], rax
    dec rcx
    jmp shift_loop
store_item:
    mov rax, r13
    shl rax, 4
    add rax, r15
    mov qword ptr [r14 + rbx*8], rax
    inc r13
    jmp insert_loop

build_output:
    lea rdi, [rip + output]
    xor ebx, ebx
output_loop:
    cmp rbx, r12
    jae write_output
    mov rax, qword ptr [r14 + rbx*8]
    mov rsi, qword ptr [rax]
    mov rcx, qword ptr [rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp output_loop
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall

compare:
    mov r8, qword ptr [rdi + 8]
    mov r9, qword ptr [rsi + 8]
    mov rdi, qword ptr [rdi]
    mov rsi, qword ptr [rsi]
    mov rcx, r8
    cmp rcx, r9
    cmova rcx, r9
    test rcx, rcx
    jz compare_lengths
compare_bytes:
    movzx eax, byte ptr [rdi]
    movzx edx, byte ptr [rsi]
    sub eax, edx
    jne compare_return
    inc rdi
    inc rsi
    dec rcx
    jnz compare_bytes
compare_lengths:
    xor eax, eax
    cmp r8, r9
    je compare_return
    mov eax, 1
    ja compare_return
    mov eax, -1
compare_return:
    ret
