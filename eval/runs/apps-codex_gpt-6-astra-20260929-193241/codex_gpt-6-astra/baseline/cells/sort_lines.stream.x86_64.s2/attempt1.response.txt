.globl _start

.bss
.balign 16
buffer:
.zero 32769
.balign 16
lines:
.zero 16000

.section .rodata
newline:
.byte 10

.text
_start:
    lea r12, [rip + buffer]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz parse
    add r13, rax
    jmp read_loop

parse:
    lea r14, [rip + lines]
    xor r15d, r15d
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r13
    jae parse_tail
    cmp byte ptr [r12 + rcx], 10
    jne parse_next
    mov rax, r15
    shl rax, 4
    lea rsi, [r12 + rdx]
    mov qword ptr [r14 + rax], rsi
    mov rsi, rcx
    sub rsi, rdx
    mov qword ptr [r14 + rax + 8], rsi
    inc r15
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_tail:
    cmp rdx, r13
    je sort_start
    mov rax, r15
    shl rax, 4
    lea rsi, [r12 + rdx]
    mov qword ptr [r14 + rax], rsi
    sub r13, rdx
    mov qword ptr [r14 + rax + 8], r13
    inc r15

sort_start:
    mov r12d, 1
sort_outer:
    cmp r12, r15
    jae output_start
    mov r10, r12
    shl r10, 4
    mov r8, qword ptr [r14 + r10]
    mov r9, qword ptr [r14 + r10 + 8]
sort_inner:
    test r10, r10
    jz insert_key
    mov rsi, qword ptr [r14 + r10 - 16]
    mov rax, qword ptr [r14 + r10 - 8]
    mov rdi, r8
    mov rcx, rax
    cmp rcx, r9
    cmova rcx, r9
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    ja shift_line
    jb insert_key
compare_lengths:
    cmp rax, r9
    jbe insert_key
shift_line:
    mov rdx, qword ptr [r14 + r10 - 16]
    mov qword ptr [r14 + r10], rdx
    mov qword ptr [r14 + r10 + 8], rax
    sub r10, 16
    jmp sort_inner
insert_key:
    mov qword ptr [r14 + r10], r8
    mov qword ptr [r14 + r10 + 8], r9
    inc r12
    jmp sort_outer

output_start:
    xor r12d, r12d
output_loop:
    cmp r12, r15
    jae finish
    mov rax, r12
    shl rax, 4
    mov rsi, qword ptr [r14 + rax]
    mov rdx, qword ptr [r14 + rax + 8]
    test rdx, rdx
    jz output_newline
    mov eax, 1
    mov edi, 1
    syscall
output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r12
    jmp output_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
