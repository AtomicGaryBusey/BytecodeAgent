.text
.globl _start
_start:
    adrp x19, raw
    add x19, x19, :lo12:raw
    mov x20, #0
read_all:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_all

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x22, jumps
    add x22, x22, :lo12:jumps
    adrp x23, brackets
    add x23, x23, :lo12:brackets
    mov x24, #0
    mov x25, #0
    mov x26, #0

parse:
    cmp x25, x20
    b.hs parsed
    ldrb w3, [x19, x25]
    add x25, x25, #1
    cmp w3, #33
    b.eq parsed
    cmp w3, #91
    b.eq open_bracket
    cmp w3, #93
    b.eq close_bracket
    cmp w3, #62
    b.eq save_command
    cmp w3, #60
    b.eq save_command
    cmp w3, #43
    b.eq save_command
    cmp w3, #45
    b.eq save_command
    cmp w3, #46
    b.eq save_command
    cmp w3, #44
    b.ne parse
save_command:
    strb w3, [x21, x24]
    add x24, x24, #1
    b parse

open_bracket:
    str w24, [x23, x26, lsl #2]
    add x26, x26, #1
    b save_command

close_bracket:
    cbz x26, invalid
    sub x26, x26, #1
    ldr w4, [x23, x26, lsl #2]
    add w5, w24, #1
    str w5, [x22, x4, lsl #2]
    add w5, w4, #1
    str w5, [x22, x24, lsl #2]
    b save_command

parsed:
    cbnz x26, invalid
    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x27, output
    add x27, x27, :lo12:output
    mov x28, #0
    mov x26, #0

execute:
    cmp x26, x24
    b.hs finished
    ldrb w3, [x21, x26]
    add x26, x26, #1
    cmp w3, #43
    b.eq increment
    cmp w3, #45
    b.eq decrement
    cmp w3, #62
    b.eq right
    cmp w3, #60
    b.eq left
    cmp w3, #91
    b.eq loop_start
    cmp w3, #93
    b.eq loop_end
    cmp w3, #46
    b.eq emit

    mov w4, #0
    cmp x25, x20
    b.hs input_store
    ldrb w4, [x19, x25]
    add x25, x25, #1
input_store:
    strb w4, [x23]
    b execute

increment:
    ldrb w4, [x23]
    add w4, w4, #1
    strb w4, [x23]
    b execute

decrement:
    ldrb w4, [x23]
    sub w4, w4, #1
    strb w4, [x23]
    b execute

right:
    add x23, x23, #1
    b execute

left:
    sub x23, x23, #1
    b execute

loop_start:
    ldrb w4, [x23]
    cbnz w4, execute
    b take_jump

loop_end:
    ldrb w4, [x23]
    cbz w4, execute
take_jump:
    sub x4, x26, #1
    ldr w26, [x22, x4, lsl #2]
    b execute

emit:
    ldrb w4, [x23]
    strb w4, [x27, x28]
    add x28, x28, #1
    b execute

finished:
    cbz x28, success
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
success:
    mov x0, #0
    b terminate
invalid:
    mov x0, #1
terminate:
    mov x8, #93
    svc #0
    b terminate

.bss

commands:
    .zero 65536
.balign 4
jumps:
