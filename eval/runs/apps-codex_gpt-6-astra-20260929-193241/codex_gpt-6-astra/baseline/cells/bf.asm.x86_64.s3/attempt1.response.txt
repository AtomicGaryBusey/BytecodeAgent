.bss
.balign 8
source:
.zero 65537
commands:
.zero 65536
matches:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.section .rodata
.balign 8
dispatch:
.quad move_right, move_left, increment, decrement
.quad emit, consume, loop_open, loop_close

.text
.globl _start
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea rbp, [rip + source]
    add rbp, r12
    lea rsi, [rip + source]
    mov r15, rbp
    xor edi, edi
    xor edx, edx
    lea r10, [rip + commands]
    lea r11, [rip + matches]
    lea r9, [rip + brackets]
parse:
    cmp rsi, rbp
    je parsed
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 33
    je separator
    xor ecx, ecx
    cmp al, 62
    je save_command
    inc ecx
    cmp al, 60
    je save_command
    inc ecx
    cmp al, 43
    je save_command
    inc ecx
    cmp al, 45
    je save_command
    inc ecx
    cmp al, 46
    je save_command
    inc ecx
    cmp al, 44
    je save_command
    inc ecx
    cmp al, 91
    je open_bracket
    inc ecx
    cmp al, 93
    jne parse
    test edx, edx
    jz invalid
    dec edx
    mov eax, dword ptr [r9 + rdx*4]
    lea r8d, [rdi + 1]
    mov dword ptr [r11 + rax*4], r8d
    inc eax
    mov dword ptr [r11 + rdi*4], eax
    jmp save_command
open_bracket:
    mov dword ptr [r9 + rdx*4], edi
    inc edx
save_command:
    mov byte ptr [r10 + rdi], cl
    inc edi
    jmp parse
separator:
    mov r15, rsi
parsed:
    test edx, edx
    jnz invalid
    mov r9d, edi
    lea r12, [rip + commands]
    lea r13, [rip + matches]
    lea r14, [rip + tape]
    lea r10, [rip + dispatch]
    lea rsi, [rip + output]
    xor ebx, ebx
    xor r8d, r8d
execute:
    cmp ebx, r9d
    jae finished
    movzx eax, byte ptr [r12 + rbx]
    inc ebx
    jmp qword ptr [r10 + rax*8]

move_right:
    inc r14
    jmp execute
move_left:
    dec r14
    jmp execute
increment:
    inc byte ptr [r14]
    jmp execute
decrement:
    dec byte ptr [r14]
    jmp execute
emit:
    mov al, byte ptr [r14]
    mov byte ptr [rsi + r8], al
    inc r8d
    jmp execute
consume:
    xor eax, eax
    cmp r15, rbp
    jae input_empty
    mov al, byte ptr [r15]
    inc r15
input_empty:
    mov byte ptr [r14], al
    jmp execute
loop_open:
    cmp byte ptr [r14], 0
    jne execute
    mov ebx, dword ptr [r13 + rbx*4 - 4]
    jmp execute
loop_close:
    cmp byte ptr [r14], 0
    je execute
    mov ebx, dword ptr [r13 + rbx*4 - 4]
    jmp execute
finished:
    mov edx, r8d
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall
