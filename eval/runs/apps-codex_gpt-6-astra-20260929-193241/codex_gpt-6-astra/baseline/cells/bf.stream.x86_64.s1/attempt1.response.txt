.globl _start
.bss
.balign 16
source:
.zero 65537
commands:
.zero 65536
links:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    xor r9d, r9d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, r9
    mov edx, 65537
    sub rdx, r9
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_loop

read_done:
    lea r12, [rip + commands]
    lea r13, [rip + links]
    lea r14, [rip + tape]
    xor ebp, ebp
    xor r8d, r8d
    xor ebx, ebx
    lea rdi, [rip + source]
    lea rsi, [rip + brackets]

parse_loop:
    cmp r8, r9
    jae parse_done
    movzx eax, byte ptr [rdi + r8]
    inc r8
    cmp al, 33
    je parse_done
    cmp al, 91
    je parse_open
    cmp al, 93
    je parse_close
    cmp al, 62
    je parse_store
    cmp al, 60
    je parse_store
    cmp al, 43
    je parse_store
    cmp al, 45
    je parse_store
    cmp al, 46
    je parse_store
    cmp al, 44
    jne parse_loop
parse_store:
    mov byte ptr [r12 + rbp], al
    inc ebp
    jmp parse_loop

parse_open:
    mov dword ptr [rsi + rbx*4], ebp
    inc ebx
    jmp parse_store

parse_close:
    test ebx, ebx
    jz invalid
    dec ebx
    mov edx, dword ptr [rsi + rbx*4]
    lea ecx, [rbp + 1]
    mov dword ptr [r13 + rdx*4], ecx
    inc edx
    mov dword ptr [r13 + rbp*4], edx
    jmp parse_store

parse_done:
    test ebx, ebx
    jnz invalid
    xor r15d, r15d
    xor r10d, r10d
execute:
    cmp r15d, ebp
    jae finished
    movzx eax, byte ptr [r12 + r15]
    inc r15d
    cmp al, 43
    je plus
    cmp al, 45
    je minus
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 91
    je open
    cmp al, 93
    je close
    cmp al, 44
    je input_byte
    mov al, byte ptr [r14 + rbx]
    lea rdx, [rip + output]
    mov byte ptr [rdx + r10], al
    inc r10d
    jmp execute

plus:
    inc byte ptr [r14 + rbx]
    jmp execute
minus:
    dec byte ptr [r14 + rbx]
    jmp execute
right:
    inc ebx
    jmp execute
left:
    dec ebx
    jmp execute
open:
    cmp byte ptr [r14 + rbx], 0
    jne execute
    mov r15d, dword ptr [r13 + r15*4 - 4]
    jmp execute
close:
    cmp byte ptr [r14 + rbx], 0
    je execute
    mov r15d, dword ptr [r13 + r15*4 - 4]
    jmp execute
input_byte:
    xor eax, eax
    cmp r8, r9
    jae input_store
    lea rdx, [rip + source]
    mov al, byte ptr [rdx + r8]
    inc r8
input_store:
    mov byte ptr [r14 + rbx], al
    jmp execute

finished:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, r10
    syscall
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall
