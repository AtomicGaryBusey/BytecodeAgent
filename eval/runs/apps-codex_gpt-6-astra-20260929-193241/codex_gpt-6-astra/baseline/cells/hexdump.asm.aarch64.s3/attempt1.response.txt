.globl _start

.text
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output
    add x22, x22, :lo12:output
    adrp x23, digits
    add x23, x23, :lo12:digits
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_offset
    sub x24, x20, x21
    mov x0, #16
    cmp x24, x0
    csel x24, x24, x0, lo
    add x25, x19, x21

    mov x3, x21
    bl emit_offset
    mov w6, #32
    strb w6, [x22], #1
    strb w6, [x22], #1

    mov x26, #0
hex_loop:
    cmp x26, x24
    b.hs empty_slot
    ldrb w7, [x25, x26]
    lsr w9, w7, #4
    ldrb w10, [x23, x9]
    strb w10, [x22], #1
    and w9, w7, #15
    ldrb w10, [x23, x9]
    strb w10, [x22], #1
    strb w6, [x22], #1
    b slot_done

empty_slot:
    strb w6, [x22], #1
    strb w6, [x22], #1
    strb w6, [x22], #1

slot_done:
    cmp x26, #7
    b.ne next_slot
    strb w6, [x22], #1
next_slot:
    add x26, x26, #1
    cmp x26, #16
    b.lo hex_loop

    strb w6, [x22], #1
    mov w7, #124
    strb w7, [x22], #1
    mov x26, #0

ascii_loop:
    ldrb w7, [x25, x26]
    sub w9, w7, #32
    cmp w9, #94
    mov w10, #46
    csel w7, w7, w10, ls
    strb w7, [x22], #1
    add x26, x26, #1
    cmp x26, x24
    b.lo ascii_loop

    mov w7, #124
    strb w7, [x22], #1
    mov w7, #10
    strb w7, [x22], #1
    add x21, x21, x24
    b row_loop

final_offset:
    mov x3, x20
    bl emit_offset
    mov w7, #10
    strb w7, [x22], #1

    mov x0, #1
    adrp x1, output
    add x1, x1, :lo12:output
    sub x2, x22, x1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

emit_offset:
    mov w4, #28
offset_digit:
    lsr w5, w3, w4
    and w5, w5, #15
    ldrb w5, [x23, x5]
    strb w5, [x22], #1
    subs w4, w4, #4
    b.ge offset_digit
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 324000
