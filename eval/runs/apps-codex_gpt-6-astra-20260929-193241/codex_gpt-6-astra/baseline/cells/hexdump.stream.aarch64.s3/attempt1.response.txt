.text
.globl _start
_start:
    adrp x21, input
    add x21, x21, :lo12:input
    adrp x22, output
    add x22, x22, :lo12:output
    adrp x23, digits
    add x23, x23, :lo12:digits
    mov x19, #0
    mov x20, #0
    mov x24, #0

read_more:
    mov x0, #0
    add x1, x21, x20
    mov x2, #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, end_input
    add x20, x20, x0
    cmp x20, #16
    b.lo read_more
    b render_row

end_input:
    mov x24, #1
    cbz x20, final_offset

render_row:
    mov x1, x22
    mov x3, x19
    bl emit_offset
    mov w7, #32
    strb w7, [x1], #1
    strb w7, [x1], #1
    mov x9, #0

hex_slot:
    cmp x9, x20
    b.hs empty_slot
    ldrb w10, [x21, x9]
    lsr w11, w10, #4
    ldrb w11, [x23, x11]
    strb w11, [x1], #1
    and w10, w10, #15
    ldrb w11, [x23, x10]
    strb w11, [x1], #1
    b slot_space

empty_slot:
    strb w7, [x1], #1
    strb w7, [x1], #1

slot_space:
    strb w7, [x1], #1
    cmp x9, #7
    b.ne next_slot
    strb w7, [x1], #1

next_slot:
    add x9, x9, #1
    cmp x9, #16
    b.lo hex_slot
    strb w7, [x1], #1
    mov w7, #124
    strb w7, [x1], #1
    mov x9, #0

ascii_byte:
    ldrb w10, [x21, x9]
    cmp w10, #32
    b.lo replace_byte
    cmp w10, #126
    b.ls store_ascii
replace_byte:
    mov w10, #46
store_ascii:
    strb w10, [x1], #1
    add x9, x9, #1
    cmp x9, x20
    b.lo ascii_byte
    strb w7, [x1], #1
    mov w7, #10
    strb w7, [x1], #1

    sub x2, x1, x22
    mov x1, x22
    mov x0, #1
    mov x8, #64
    svc #0
    add x19, x19, x20
    mov x20, #0
    cbz x24, read_more

final_offset:
    mov x1, x22
    mov x3, x19
    bl emit_offset
    mov w7, #10
    strb w7, [x1]
    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

emit_offset:
    mov x4, #28
offset_digit:
    lsr x5, x3, x4
    and x5, x5, #15
    ldrb w6, [x23, x5]
    strb w6, [x1], #1
    subs x4, x4, #4
    b.ge offset_digit
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 16
output:
    .zero 80
