.bss
input:
.zero 65537
values:
.zero 8000
output:
.zero 393216

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    add r13, r12
    lea rbp, [rip + output]
line_loop:
    cmp r12, r13
    jae finish
    mov r14, r12
find_line_end:
    cmp r14, r13
    jae line_ready
    cmp byte ptr [r14], 10
    je line_ready
    inc r14
    jmp find_line_end
line_ready:
    xor r15d, r15d
token_loop:
    cmp r12, r14
    jae line_done
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov r8, r12
    mov rbx, r12
find_token_end:
    cmp rbx, r14
    jae token_ready
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc rbx
    jmp find_token_end
skip_separator:
    inc r12
    jmp token_loop
token_ready:
    mov r12, rbx
    mov r9, rbx
    sub r9, r8
    cmp r9, 1
    jne literal
    movzx edx, byte ptr [r8]
    cmp dl, 43
    je operator
    cmp dl, 45
    je operator
    cmp dl, 42
    je operator
    cmp dl, 47
    je operator
    cmp dl, 37
    je operator
literal:
    xor r10d, r10d
    cmp byte ptr [r8], 45
    jne literal_digits
    inc r8
    dec r9
    mov r10d, 1
literal_digits:
    test r9, r9
    jz line_error
    cmp r9, 18
    ja line_error
    xor eax, eax
digit_loop:
    movzx ecx, byte ptr [r8]
    sub ecx, 48
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc r8
    cmp r8, rbx
    jb digit_loop
    test r10d, r10d
    jz push_value
    neg rax
push_value:
    lea rsi, [rip + values]
    mov qword ptr [rsi + r15*8], rax
    inc r15
    jmp token_loop

operator:
    cmp r15, 2
    jb line_error
    lea rsi, [rip + values]
    dec r15
    mov rdi, qword ptr [rsi + r15*8]
    dec r15
    mov rax, qword ptr [rsi + r15*8]
    cmp dl, 43
    je addition
    cmp dl, 45
    je subtraction
    cmp dl, 42
    je multiplication
    test rdi, rdi
    jz line_error
    mov r8, 0x8000000000000000
    cmp rax, r8
    jne division_safe
    cmp rdi, -1
    je line_error
division_safe:
    mov ecx, edx
    cqo
    idiv rdi
    cmp cl, 37
    jne push_value
    mov rax, rdx
    jmp push_value
addition:
    add rax, rdi
    jmp push_value
subtraction:
    sub rax, rdi
    jmp push_value
multiplication:
    imul rax, rdi
    jmp push_value

line_done:
    cmp r15, 1
    jne line_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns magnitude_ready
    mov byte ptr [rbp], 45
    inc rbp
    neg rax
magnitude_ready:
    mov r8, rbp
    mov edi, 10
decimal_loop:
    xor edx, edx
    div rdi
    add dl, 48
    mov byte ptr [rbp], dl
    inc rbp
    test rax, rax
    jnz decimal_loop
    lea r9, [rbp - 1]
reverse_loop:
    cmp r8, r9
    jae decimal_done
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_loop
decimal_done:
    mov byte ptr [rbp], 10
    inc rbp
    jmp next_line
line_error:
    mov dword ptr [rbp], 0x6f727265
    mov word ptr [rbp + 4], 0x0a72
    add rbp, 6
next_line:
    mov r12, r14
    cmp r12, r13
    jae finish
    inc r12
    jmp line_loop
finish:
    lea rsi, [rip + output]
    mov rdx, rbp
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
