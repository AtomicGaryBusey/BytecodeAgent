.section .bss
input:
.zero 65537
values:
.zero 8000
output:
.zero 524288

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    lea r14, [r12 + r13]
    lea r15, [rip + output]
    lea rbp, [rip + values]
next_line:
    cmp r12, r14
    jae finish
    xor ebx, ebx

next_token:
    cmp r12, r14
    jae line_done
    movzx eax, byte ptr [r12]
    cmp al, 10
    je newline
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov rsi, r12
scan_token:
    inc r12
    cmp r12, r14
    jae token_ready
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_ready
    cmp al, 32
    je token_ready
    cmp al, 9
    jne scan_token
token_ready:
    mov rcx, r12
    sub rcx, rsi
    movzx r8d, byte ptr [rsi]
    cmp rcx, 1
    jne literal
    cmp r8b, '+'
    je operator
    cmp r8b, '-'
    je operator
    cmp r8b, '*'
    je operator
    cmp r8b, '/'
    je operator
    cmp r8b, '%'
    je operator

literal:
    xor r9d, r9d
    cmp r8b, '-'
    jne digits_start
    mov r9d, 1
    inc rsi
    dec rcx
digits_start:
    test rcx, rcx
    jz bad_line
    cmp rcx, 18
    ja bad_line
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja bad_line
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digit_loop
    test r9d, r9d
    jz push_value
    neg rax
push_value:
    mov qword ptr [rbp + rbx*8], rax
    inc ebx
    jmp next_token

operator:
    cmp ebx, 2
    jb bad_line
    dec ebx
    mov rdi, qword ptr [rbp + rbx*8]
    mov rax, qword ptr [rbp + rbx*8 - 8]
    cmp r8b, '+'
    je do_add
    cmp r8b, '-'
    je do_sub
    cmp r8b, '*'
    je do_mul
    test rdi, rdi
    jz bad_line
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne do_div
    cmp rdi, -1
    je bad_line
do_div:
    cqo
    idiv rdi
    cmp r8b, '%'
    jne store_result
    mov rax, rdx
    jmp store_result
do_add:
    add rax, rdi
    jmp store_result
do_sub:
    sub rax, rdi
    jmp store_result
do_mul:
    imul rax, rdi
store_result:
    mov qword ptr [rbp + rbx*8 - 8], rax
    jmp next_token

skip_separator:
    inc r12
    jmp next_token
newline:
    inc r12
line_done:
    cmp ebx, 1
    jne emit_error
    mov rax, qword ptr [rbp]
    test rax, rax
    jns format_unsigned
    mov byte ptr [r15], '-'
    inc r15
    neg rax
format_unsigned:
    mov rsi, r15
    mov ecx, 10
format_digit:
    xor edx, edx
    div rcx
    add dl, '0'
    mov byte ptr [r15], dl
    inc r15
    test rax, rax
    jnz format_digit
    lea rdi, [r15 - 1]
reverse_digits:
    cmp rsi, rdi
    jae end_result
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp reverse_digits
end_result:
    mov byte ptr [r15], 10
    inc r15
    jmp next_line

bad_line:
    cmp r12, r14
    jae emit_error
    mov al, byte ptr [r12]
    inc r12
    cmp al, 10
    jne bad_line
emit_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6
    jmp next_line

finish:
    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    jz exit_zero
    mov eax, 1
    mov edi, 1
    syscall
exit_zero:
    mov eax, 60
    xor edi, edi
    syscall
