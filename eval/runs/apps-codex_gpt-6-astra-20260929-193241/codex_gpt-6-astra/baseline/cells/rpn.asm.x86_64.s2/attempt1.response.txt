.globl _start

.bss
.balign 16
input:
.zero 65537
values:
.zero 8000
output:
.zero 393216

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    add r13, r12
    lea r15, [rip + output]

next_line:
    cmp r12, r13
    jae finished
    mov rbx, r12
find_line_end:
    cmp rbx, r13
    jae line_ready
    cmp byte ptr [rbx], 10
    je line_ready
    inc rbx
    jmp find_line_end

line_ready:
    xor r14d, r14d

next_token:
    cmp r12, rbx
    jae line_done
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov r8, r12
find_token_end:
    cmp r8, rbx
    jae token_ready
    movzx eax, byte ptr [r8]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc r8
    jmp find_token_end

skip_separator:
    inc r12
    jmp next_token

token_ready:
    mov rcx, r8
    sub rcx, r12
    movzx edi, byte ptr [r12]
    cmp rcx, 1
    jne literal
    cmp dil, 43
    je operator
    cmp dil, 45
    je operator
    cmp dil, 42
    je operator
    cmp dil, 47
    je operator
    cmp dil, 37
    je operator

literal:
    xor r9d, r9d
    cmp dil, 45
    jne literal_length
    mov r9d, 1
    inc r12
    dec rcx
literal_length:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
literal_digit:
    movzx edx, byte ptr [r12]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r12
    cmp r12, r8
    jb literal_digit
    test r9d, r9d
    jz push_value
    neg rax
push_value:
    lea rsi, [rip + values]
    mov qword ptr [rsi + r14*8], rax
    inc r14
    mov r12, r8
    jmp next_token

operator:
    cmp r14, 2
    jb line_error
    lea rsi, [rip + values]
    dec r14
    mov r10, qword ptr [rsi + r14*8]
    dec r14
    mov rax, qword ptr [rsi + r14*8]
    cmp dil, 43
    je do_add
    cmp dil, 45
    je do_sub
    cmp dil, 42
    je do_mul
    test r10, r10
    jz line_error
    cmp r10, -1
    jne do_divide
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
do_divide:
    cqo
    idiv r10
    cmp dil, 37
    jne push_value
    mov rax, rdx
    jmp push_value
do_add:
    add rax, r10
    jmp push_value
do_sub:
    sub rax, r10
    jmp push_value
do_mul:
    imul rax, r10
    jmp push_value

line_done:
    cmp r14, 1
    jne line_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns print_magnitude
    mov byte ptr [r15], 45
    inc r15
    neg rax
print_magnitude:
    mov r8, r15
    mov ecx, 10
decimal_digit:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [r15], dl
    inc r15
    test rax, rax
    jnz decimal_digit
    lea r9, [r15 - 1]
reverse_digits:
    cmp r8, r9
    jae number_done
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_digits
number_done:
    mov byte ptr [r15], 10
    inc r15
    jmp advance_line

line_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6

advance_line:
    mov r12, rbx
    cmp r12, r13
    jae finished
    inc r12
    jmp next_line

finished:
    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
