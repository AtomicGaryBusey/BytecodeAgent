.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input:
.zero 65536
output:
.zero 66560

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    lea r14, [rip + output]
    mov r15, r14
    lea rbx, [rip + alphabet]
    xor ebp, ebp
    test r13, r13
    jz finish

encode_loop:
    movzx eax, byte ptr [r12]
    shl eax, 16
    inc r12
    dec r13
    mov r9d, 1
    test r13, r13
    jz emit
    movzx edx, byte ptr [r12]
    shl edx, 8
    or eax, edx
    inc r12
    dec r13
    inc r9d
    test r13, r13
    jz emit
    movzx edx, byte ptr [r12]
    or eax, edx
    inc r12
    dec r13
    inc r9d

emit:
    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [r15], cl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 1], cl

    mov byte ptr [r15 + 2], 61
    mov byte ptr [r15 + 3], 61
    cmp r9d, 1
    je group_done

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [r15 + 2], cl
    cmp r9d, 2
    je group_done

    and eax, 63
    movzx ecx, byte ptr [rbx + rax]
    mov byte ptr [r15 + 3], cl

group_done:
    add r15, 4
    add ebp, 4
    cmp ebp, 76
    jne check_remaining
    mov byte ptr [r15], 10
    inc r15
    xor ebp, ebp

check_remaining:
    test r13, r13
    jnz encode_loop
    test ebp, ebp
    jz write_output
    mov byte ptr [r15], 10
    inc r15

write_output:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    sub rdx, r14
    syscall

finish:
    mov eax, 60
    xor edi, edi
    syscall
