.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input:
.zero 65536
output:
.zero 66560

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz encode_start
    add r12, rax
    jmp read_loop

encode_start:
    lea r13, [rip + input]
    lea r14, [rip + output]
    lea rbx, [rip + alphabet]
    xor r15d, r15d
    test r12, r12
    jz finish

encode_group:
    movzx eax, byte ptr [r13]
    shl eax, 16
    mov r8d, 1
    cmp r12, 1
    je emit_group
    movzx edx, byte ptr [r13 + 1]
    shl edx, 8
    or eax, edx
    mov r8d, 2
    cmp r12, 2
    je emit_group
    movzx edx, byte ptr [r13 + 2]
    or eax, edx
    mov r8d, 3

emit_group:
    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [r14], cl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [r14 + 1], cl

    mov byte ptr [r14 + 2], 61
    mov byte ptr [r14 + 3], 61
    cmp r8d, 1
    je group_done

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [r14 + 2], cl
    cmp r8d, 2
    je group_done

    and eax, 63
    movzx ecx, byte ptr [rbx + rax]
    mov byte ptr [r14 + 3], cl

group_done:
    add r13, r8
    sub r12, r8
    add r14, 4
    add r15d, 4
    cmp r15d, 76
    jne check_remaining
    mov byte ptr [r14], 10
    inc r14
    xor r15d, r15d

check_remaining:
    test r12, r12
    jnz encode_group
    test r15d, r15d
    jz write_output
    mov byte ptr [r14], 10
    inc r14

write_output:
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

finish:
    mov eax, 60
    xor edi, edi
    syscall
