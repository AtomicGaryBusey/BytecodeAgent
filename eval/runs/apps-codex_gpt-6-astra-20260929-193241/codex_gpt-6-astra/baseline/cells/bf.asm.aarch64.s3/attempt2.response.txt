.globl _start

.text
_start:
    adrp x19, raw
    add x19, x19, :lo12:raw
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x22, jumps
    add x22, x22, :lo12:jumps
    adrp x23, brackets
    add x23, x23, :lo12:brackets
    mov x24, #0
    mov x25, x20
    mov x9, #0
    mov x10, #0

parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w11, [x19, x9]
    add x9, x9, #1
    cmp w11, #33
    b.eq separator
    cmp w11, #91
    b.eq open_bracket
    cmp w11, #93
    b.eq close_bracket
    cmp w11, #62
    b.eq save_command
    cmp w11, #60
    b.eq save_command
    cmp w11, #43
    b.eq save_command
    cmp w11, #45
    b.eq save_command
    cmp w11, #46
    b.eq save_command
    cmp w11, #44
    b.ne parse_loop
save_command:
    strb w11, [x21, x24]
    add x24, x24, #1
    b parse_loop

open_bracket:
    str w24, [x23, x10, lsl #2]
    add x10, x10, #1
    b save_command

close_bracket:
    cbz x10, invalid
    sub x10, x10, #1
    ldr w12, [x23, x10, lsl #2]
    add w13, w24, #1
    str w13, [x22, x12, lsl #2]
    add w13, w12, #1
    str w13, [x22, x24, lsl #2]
    b save_command

separator:
    mov x25, x9
parse_done:
    cbnz x10, invalid
    adrp x26, tape
    add x26, x26, :lo12:tape
    adrp x27, output
    add x27, x27, :lo12:output
    mov x28, #0
    mov x9, #0

execute:
    cmp x9, x24
    b.hs finished
    ldrb w11, [x21, x9]
    add x9, x9, #1
    cmp w11, #43
    b.eq increment
    cmp w11, #45
    b.eq decrement
    cmp w11, #62
    b.eq right
    cmp w11, #60
    b.eq left
    cmp w11, #91
    b.eq loop_open
    cmp w11, #93
    b.eq loop_close
    cmp w11, #46
    b.eq emit

    mov w12, #0
    cmp x25, x20
    b.hs store_input
    ldrb w12, [x19, x25]
    add x25, x25, #1
store_input:
    strb w12, [x26]
    b execute

increment:
    ldrb w12, [x26]
    add w12, w12, #1
    strb w12, [x26]
    b execute

decrement:
    ldrb w12, [x26]
    sub w12, w12, #1
    strb w12, [x26]
    b execute

right:
    add x26, x26, #1
    b execute

left:
    sub x26, x26, #1
    b execute

loop_open:
    ldrb w12, [x26]
    cbnz w12, execute
    sub x13, x9, #1
    ldr w9, [x22, x13, lsl #2]
    b execute

loop_close:
    ldrb w12, [x26]
    cbz w12, execute
    sub x13, x9, #1
    ldr w9, [x22, x13, lsl #2]
    b execute

emit:
    ldrb w12, [x26]
    strb w12, [x27, x28]
    add x28, x28, #1
    b execute

finished:
    cbz x28, success
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
success:
    mov x0, #0
    mov x8, #93
    svc #0
halt:
    b halt

invalid:
    mov x0, #1
    mov x8, #93
    svc #0
    b halt
