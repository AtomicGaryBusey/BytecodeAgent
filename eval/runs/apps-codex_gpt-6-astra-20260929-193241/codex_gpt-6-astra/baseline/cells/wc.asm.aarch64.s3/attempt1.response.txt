.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, print_counts
    add x21, x21, x0
    mov x4, x23
    mov x5, x0

count_loop:
    ldrb w6, [x4], #1
    cmp w6, #10
    cinc x19, x19, eq
    cmp w6, #32
    b.eq whitespace
    sub w7, w6, #9
    cmp w7, #4
    b.ls whitespace
    cbnz w22, next_byte
    add x20, x20, #1
    mov w22, #1
    b next_byte

whitespace:
    mov w22, #0

next_byte:
    subs x5, x5, #1
    b.ne count_loop
    b read_loop

print_counts:
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x25, x24
    mov x0, x19
    bl decimal
    mov w1, #32
    strb w1, [x25], #1
    mov x0, x20
    bl decimal
    mov w1, #32
    strb w1, [x25], #1
    mov x0, x21
    bl decimal
    mov w1, #10
    strb w1, [x25], #1

    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

decimal:
    sub sp, sp, #32
    add x3, sp, #32
    mov x4, x3
    mov x5, #10
decimal_digit:
    udiv x1, x0, x5
    msub x2, x1, x5, x0
    add w2, w2, #48
    strb w2, [x3, #-1]!
    mov x0, x1
    cbnz x0, decimal_digit
decimal_copy:
    ldrb w2, [x3], #1
    strb w2, [x25], #1
    cmp x3, x4
    b.ne decimal_copy
    add sp, sp, #32
    ret

.bss
.balign 16
input_buffer:
    .zero 4096
output_buffer:
    .zero 64
