.globl _start
.bss
input:
.zero 65537
values:
.zero 8008
output:
.zero 393216
digits:
.zero 32

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    add r13, r12
    lea r14, [rip + output]
    lea r15, [rip + values]
new_line:
    cmp r12, r13
    jae finish
    xor ebp, ebp
next_token:
    cmp r12, r13
    jae end_line
    movzx eax, byte ptr [r12]
    cmp al, 10
    je newline
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov rsi, r12
find_end:
    inc r12
    cmp r12, r13
    jae token_ready
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_ready
    cmp al, 32
    je token_ready
    cmp al, 9
    jne find_end
token_ready:
    mov rcx, r12
    sub rcx, rsi
    movzx ebx, byte ptr [rsi]
    cmp rcx, 1
    jne literal
    cmp bl, 43
    je operator
    cmp bl, 45
    je operator
    cmp bl, 42
    je operator
    cmp bl, 47
    je operator
    cmp bl, 37
    je operator
literal:
    xor r8d, r8d
    cmp bl, 45
    jne positive_literal
    inc rsi
    dec rcx
    mov r8d, 1
positive_literal:
    test rcx, rcx
    jz bad_line
    cmp rcx, 18
    ja bad_line
    xor eax, eax
parse_digit:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja bad_line
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz parse_digit
    test r8d, r8d
    jz push_value
    neg rax
push_value:
    mov qword ptr [r15 + rbp*8], rax
    inc ebp
    jmp next_token

skip_separator:
    inc r12
    jmp next_token

operator:
    cmp ebp, 2
    jb bad_line
    dec ebp
    mov rdi, qword ptr [r15 + rbp*8]
    dec ebp
    mov rax, qword ptr [r15 + rbp*8]
    cmp bl, 43
    je addition
    cmp bl, 45
    je subtraction
    cmp bl, 42
    je multiplication
    test rdi, rdi
    jz bad_line
    cmp rdi, -1
    jne division
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je bad_line
division:
    cqo
    idiv rdi
    cmp bl, 37
    jne push_value
    mov rax, rdx
    jmp push_value
addition:
    add rax, rdi
    jmp push_value
subtraction:
    sub rax, rdi
    jmp push_value
multiplication:
    imul rax, rdi
    jmp push_value

newline:
    inc r12
end_line:
    cmp ebp, 1
    jne emit_error
    mov rax, qword ptr [r15]
    test rax, rax
    jns format_number
    mov byte ptr [r14], 45
    inc r14
    neg rax
format_number:
    lea rsi, [rip + digits + 32]
    mov edi, 10
    xor ecx, ecx
format_digit:
    xor edx, edx
    div rdi
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz format_digit
copy_digits:
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc rsi
    inc r14
    dec ecx
    jnz copy_digits
    mov byte ptr [r14], 10
    inc r14
    jmp new_line

bad_line:
    cmp r12, r13
    jae emit_error
    mov al, byte ptr [r12]
    inc r12
    cmp al, 10
    jne bad_line
emit_error:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp new_line

finish:
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
