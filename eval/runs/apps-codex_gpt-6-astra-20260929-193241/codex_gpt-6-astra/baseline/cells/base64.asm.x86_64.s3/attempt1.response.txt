.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
.zero 49153
output:
.zero 66400

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49153
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    mov rsi, r12
    lea rdi, [rip + output]
    lea rbx, [rip + alphabet]
    xor r8d, r8d

encode_loop:
    test r13, r13
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r13, 1
    je encode_group
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    cmp r13, 2
    je encode_group
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

encode_group:
    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], cl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], cl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    cmp r13, 1
    je group_done

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], cl
    cmp r13, 2
    je group_done

    and eax, 63
    movzx ecx, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], cl

group_done:
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne advance
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

advance:
    cmp r13, 3
    jbe finish
    sub r13, 3
    add rsi, 3
    jmp encode_loop

finish:
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall
