.globl _start

.bss
input:
.zero 65537
output:
.zero 65537

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + output]
    xor r15d, r15d
    xor ebx, ebx
    xor ebp, ebp

scan:
    cmp rbx, r13
    je final_remainder
    cmp byte ptr [r12 + rbx], 10
    je terminated_line
    inc rbx
    jmp scan

terminated_line:
    mov r8, rbx
reverse_terminated:
    cmp r8, rbp
    je append_newline
    dec r8
    mov al, byte ptr [r12 + r8]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_terminated

append_newline:
    mov byte ptr [r14 + r15], 10
    inc r15
    inc rbx
    mov rbp, rbx
    jmp scan

final_remainder:
    cmp rbp, r13
    je write_output
    mov r8, r13
reverse_final:
    cmp r8, rbp
    je final_newline
    dec r8
    mov al, byte ptr [r12 + r8]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp reverse_final

final_newline:
    mov byte ptr [r14 + r15], 10
    inc r15

write_output:
    test r15, r15
    jz done
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall

done:
    mov eax, 60
    xor edi, edi
    syscall
