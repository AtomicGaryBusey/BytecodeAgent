.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz output
    js read_more
    add r14, rax
    lea rdi, [rsi + rax]

count_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne check_space
    inc r12

check_space:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    cmp rsi, rdi
    jne count_loop
    jmp read_more

output:
    lea rsi, [rip + outbuf]
    mov rax, r12
    call decimal
    mov byte ptr [rsi], 32
    inc rsi
    mov rax, r13
    call decimal
    mov byte ptr [rsi], 32
    inc rsi
    mov rax, r14
    call decimal
    mov byte ptr [rsi], 10
    inc rsi

    lea rdx, [rip + outbuf]
    sub rsi, rdx
    mov rdx, rsi
    lea rsi, [rip + outbuf]
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

decimal:
    lea r8, [rip + digits + 20]
    mov r9, r8
    mov ecx, 10
decimal_divide:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz decimal_divide
decimal_copy:
    mov al, byte ptr [r8]
    mov byte ptr [rsi], al
    inc r8
    inc rsi
    cmp r8, r9
    jne decimal_copy
    ret

.bss
.balign 16
buffer:
    .zero 4096
outbuf:
    .zero 64
digits:
    .zero 20
