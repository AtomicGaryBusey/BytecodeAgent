.globl _start

.bss
.align 8
inbuf:
    .zero 65536
tape:
    .zero 30000
jumps:
    .zero 262144
bstack:
    .zero 8000
outbuf:
    .zero 65536
dtab:
    .zero 2048

.text
_start:
    xor r12, r12
read_loop:
    mov eax, 65536
    sub rax, r12
    jz read_done
    mov rdx, rax
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor ecx, ecx
    lea rdx, [rip + inbuf]
find_loop:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rdx + rcx], 0x21
    je has_bang
    inc rcx
    jmp find_loop
has_bang:
    mov r13, rcx
    lea r14, [rdx + rcx + 1]
    mov r15, r12
    sub r15, rcx
    dec r15
    jmp do_check
no_bang:
    mov r13, r12
    xor r15, r15
    lea r14, [rdx + r12]
do_check:
    xor ecx, ecx
    xor esi, esi
    lea rdi, [rip + inbuf]
    lea r8, [rip + jumps]
    lea r9, [rip + bstack]
check_loop:
    cmp rcx, r13
    jae check_done
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x5b
    je is_open
    cmp al, 0x5d
    je is_close
    inc rcx
    jmp check_loop
is_open:
    mov qword ptr [r9 + rsi*8], rcx
    inc rsi
    inc rcx
    jmp check_loop
is_close:
    test rsi, rsi
    jz bad
    dec rsi
    mov rax, qword ptr [r9 + rsi*8]
    mov dword ptr [r8 + rax*4], ecx
    mov dword ptr [r8 + rcx*4], eax
    inc rcx
    jmp check_loop
check_done:
    test rsi, rsi
    jnz bad
    lea r11, [rip + dtab]
    lea rax, [rip + c_other]
    xor ecx, ecx
dt_fill:
    mov qword ptr [r11 + rcx*8], rax
    inc ecx
    cmp ecx, 256
    jne dt_fill
    lea rax, [rip + c_plus]
    mov ecx, 0x2b
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_comma]
    mov ecx, 0x2c
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_minus]
    mov ecx, 0x2d
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_dot]
    mov ecx, 0x2e
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_lt]
    mov ecx, 0x3c
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_gt]
    mov ecx, 0x3e
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_lbrack]
    mov ecx, 0x5b
    mov qword ptr [r11 + rcx*8], rax
    lea rax, [rip + c_rbrack]
    mov ecx, 0x5d
    mov qword ptr [r11 + rcx*8], rax
    lea rbx, [rip + tape]
    lea rbp, [rip + inbuf]
    lea r9, [rip + outbuf]
    xor r10, r10
    xor r12, r12
exec_loop:
    cmp r12, r13
    jae good
    movzx eax, byte ptr [rbp + r12]
    jmp qword ptr [r11 + rax*8]
c_other:
    inc r12
    jmp exec_loop
c_plus:
    inc byte ptr [rbx]
    inc r12
    jmp exec_loop
c_minus:
    dec byte ptr [rbx]
    inc r12
    jmp exec_loop
c_gt:
    inc rbx
    inc r12
    jmp exec_loop
c_lt:
    dec rbx
    inc r12
    jmp exec_loop
c_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [r9 + r10], al
    inc r10
    inc r12
    jmp exec_loop
c_comma:
    test r15, r15
    jz c_comma0
    mov al, byte ptr [r14]
    mov byte ptr [rbx], al
    inc r14
    dec r15
    inc r12
    jmp exec_loop
c_comma0:
    mov byte ptr [rbx], 0
    inc r12
    jmp exec_loop
c_lbrack:
    cmp byte ptr [rbx], 0
    jne c_other
    jmp c_dojump
c_rbrack:
    cmp byte ptr [rbx], 0
    je c_other
c_dojump:
    mov eax, dword ptr [r8 + r12*4]
    lea r12, [rax + 1]
    jmp exec_loop
good:
    test r10, r10
    jz good_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r10
    syscall
good_exit:
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall
