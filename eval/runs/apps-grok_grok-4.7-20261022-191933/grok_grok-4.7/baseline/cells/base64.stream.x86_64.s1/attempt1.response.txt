.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
.zero 49152
outbuf:
.zero 70000

.text
.globl _start
_start:
xor r12d, r12d
lea rbx, [rip + inbuf]

read_loop:
cmp r12, 49152
jae read_done
xor eax, eax
xor edi, edi
lea rsi, [rbx + r12]
mov edx, 49152
sub rdx, r12
syscall
test rax, rax
jle read_done
add r12, rax
jmp read_loop

read_done:
lea r13, [rip + inbuf]
lea r14, [rip + outbuf]
xor r15d, r15d
mov rbx, r12

encode_loop:
cmp rbx, 3
jb encode_tail
movzx r8d, byte ptr [r13]
movzx r9d, byte ptr [r13 + 1]
movzx r10d, byte ptr [r13 + 2]
mov edi, r8d
shr edi, 2
call emit_idx
mov edi, r8d
and edi, 3
shl edi, 4
mov eax, r9d
shr eax, 4
or edi, eax
call emit_idx
mov edi, r9d
and edi, 15
shl edi, 2
mov eax, r10d
shr eax, 6
or edi, eax
call emit_idx
mov edi, r10d
and edi, 63
call emit_idx
add r13, 3
sub rbx, 3
jmp encode_loop

encode_tail:
cmp rbx, 1
jb after_data
je one_byte
movzx r8d, byte ptr [r13]
movzx r9d, byte ptr [r13 + 1]
mov edi, r8d
shr edi, 2
call emit_idx
mov edi, r8d
and edi, 3
shl edi, 4
mov eax, r9d
shr eax, 4
or edi, eax
call emit_idx
mov edi, r9d
and edi, 15
shl edi, 2
call emit_idx
mov al, 61
call emit_char
jmp after_data

one_byte:
movzx r8d, byte ptr [r13]
mov edi, r8d
shr edi, 2
call emit_idx
mov edi, r8d
and edi, 3
shl edi, 4
call emit_idx
mov al, 61
call emit_char
mov al, 61
call emit_char

after_data:
test r15d, r15d
jz write_out
mov byte ptr [r14], 10
inc r14

write_out:
lea rsi, [rip + outbuf]
mov rdx, r14
sub rdx, rsi
jz do_exit
mov eax, 1
mov edi, 1
syscall

do_exit:
mov eax, 60
xor edi, edi
syscall
jmp do_exit

emit_idx:
lea rcx, [rip + alphabet]
movzx eax, byte ptr [rcx + rdi]
mov byte ptr [r14], al
inc r14
inc r15d
cmp r15d, 76
jne emit_idx_ret
mov byte ptr [r14], 10
inc r14
xor r15d, r15d
emit_idx_ret:
ret

emit_char:
mov byte ptr [r14], al
inc r14
inc r15d
cmp r15d, 76
jne emit_char_ret
mov byte ptr [r14], 10
inc r14
xor r15d, r15d
emit_char_ret:
ret
