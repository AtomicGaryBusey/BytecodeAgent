.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_CAP, 65536
.equ OK_LEN, 3
.equ ERR_LEN, 9

.data
ok_msg:
  .ascii "ok"
  .byte 0x0A
err_prefix:
  .ascii "error at "
lit_true:
  .ascii "true"
lit_false:
  .ascii "false"
lit_null:
  .ascii "null"

.text
.globl _start

_start:
  cld
  lea r12, [rip + inbuf]
  xor r13, r13
  xor r14, r14
80:
  mov eax, SYS_read
  xor edi, edi
  lea rsi, [r12 + r14]
  mov rdx, BUF_CAP
  sub rdx, r14
  syscall
  test rax, rax
  jle 81f
  add r14, rax
  cmp r14, BUF_CAP
  jb 80b
81:
  call skip_ws
  call parse_value
  call skip_ws
  cmp r13, r14
  jne fail
  mov eax, SYS_write
  mov edi, 1
  lea rsi, [rip + ok_msg]
  mov edx, OK_LEN
  syscall
  mov eax, SYS_exit
  xor edi, edi
  syscall
82:
  jmp 82b

skip_ws:
30:
  cmp r13, r14
  jae 31f
  movzx eax, byte ptr [r12 + r13]
  cmp al, 0x20
  je 32f
  cmp al, 0x09
  je 32f
  cmp al, 0x0A
  je 32f
  cmp al, 0x0D
  je 32f
  ret
32:
  inc r13
  jmp 30b
31:
  ret

parse_value:
  cmp r13, r14
  jae fail
  movzx eax, byte ptr [r12 + r13]
  cmp al, 0x7B
  je parse_object
  cmp al, 0x5B
  je parse_array
  cmp al, 0x22
  je parse_string
  cmp al, 0x74
  je parse_true
  cmp al, 0x66
  je parse_false
  cmp al, 0x6E
  je parse_null
  cmp al, 0x2D
  je parse_number
  cmp al, 0x30
  jb fail
  cmp al, 0x39
  jbe parse_number
  jmp fail

parse_true:
  lea rsi, [rip + lit_true]
  mov ecx, 4
  jmp match_lit

parse_false:
  lea rsi, [rip + lit_false]
  mov ecx, 5
  jmp match_lit

parse_null:
  lea rsi, [rip + lit_null]
  mov ecx, 4
  jmp match_lit

match_lit:
10:
  cmp r13, r14
  jae fail
  movzx eax, byte ptr [r12 + r13]
  movzx edx, byte ptr [rsi]
  cmp al, dl
  jne fail
  inc r13
  inc rsi
  dec ecx
  jnz 10b
  ret

  inc r13
20:
  cmp r13, r14
  jae fail
  movzx eax, byte ptr [r12 + r13]
  cmp al, 0x22

  jmp 93b
