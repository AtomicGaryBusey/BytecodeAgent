.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 96

.text
_start:
    lea r14, [rip + inbuf]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65536
    sub edx, r13d
    test edx, edx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r12d, r12d

row_loop:
    cmp r12, r13
    jae final_line
    mov r15, r13
    sub r15, r12
    cmp r15, 16
    jbe have_n
    mov r15, 16
have_n:
    lea rsi, [r14 + r12]
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    xor ebx, ebx

hex_loop:
    cmp rbx, r15
    jae hex_pad
    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [hexdigits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    movzx eax, byte ptr [hexdigits + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_next
hex_pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
hex_next:
    cmp ebx, 7
    jne hex_no_extra
    mov byte ptr [rdi], 0x20
    inc rdi
hex_no_extra:
    inc ebx
    cmp ebx, 16
    jb hex_loop

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi
    xor ebx, ebx

asc_loop:
    cmp rbx, r15
    jae asc_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    jb asc_dot
    cmp al, 0x7e
    ja asc_dot
    mov byte ptr [rdi], al
    jmp asc_put
asc_dot:
    mov byte ptr [rdi], 0x2e
asc_put:
    inc rdi
    inc ebx
    jmp asc_loop
asc_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r12, r15
    jmp row_loop

final_line:
    lea rdi, [rip + outbuf]
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 28
hex8_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [hexdigits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns hex8_loop
    ret
