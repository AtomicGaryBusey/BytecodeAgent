.globl _start

.data
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
line_buf:
    .zero 80

.text
_start:
    lea r10, [rip + hexdigits]
    xor r12d, r12d

read_loop:
    mov edx, 65536
    sub edx, r12d
    jz read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d
    test r12, r12
    jz final_line

row_loop:
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    lea r15, [rip + input_buf]
    add r15, r13

    lea rdi, [rip + line_buf]
    mov eax, r13d
    call store_hex8

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2

    xor ebx, ebx
byte_loop:
    cmp rbx, r14
    jae 2f
    movzx eax, byte ptr [r15 + rbx]
    mov edx, eax
    shr eax, 4
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    and edx, 15
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp 3f
2:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
3:
    inc ebx
    cmp ebx, 8
    jne 4f
    mov byte ptr [rdi], 0x20
    inc rdi
4:
    cmp ebx, 16
    jb byte_loop

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi

    xor ebx, ebx
ascii_loop:
    cmp rbx, r14
    jae ascii_done
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 0x20
    jb 5f
    cmp al, 0x7e
    ja 5f
    mov byte ptr [rdi], al
    jmp 6f
5:
    mov byte ptr [rdi], 0x2e
6:
    inc rdi
    inc ebx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r13, r14
    cmp r13, r12
    jb row_loop

final_line:
    lea rdi, [rip + line_buf]
    mov eax, r12d
    call store_hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

store_hex8:
    mov ecx, 28
7:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns 7b
    ret
