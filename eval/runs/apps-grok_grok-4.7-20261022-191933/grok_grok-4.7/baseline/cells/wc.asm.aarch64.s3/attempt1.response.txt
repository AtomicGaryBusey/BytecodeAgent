.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 64

.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x23, inbuf
    add x23, x23, :lo12:inbuf

read_more:
    mov x0, #0
    mov x1, x23
    movz x2, #1, lsl #16
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le print_counts

    mov x9, #0
    mov x10, x0

next_byte:
    cmp x9, x10
    b.hs read_more
    ldrb w11, [x23, x9]
    add x9, x9, #1
    add x19, x19, #1

    cmp w11, #0x0A
    b.ne 3f
    add x20, x20, #1
3:
    cmp w11, #0x20
    b.eq mark_ws
    sub w12, w11, #9
    cmp w12, #5
    b.lo mark_ws

    cbnz x22, next_byte
    mov x22, #1
    add x21, x21, #1
    b next_byte

mark_ws:
    mov x22, #0
    b next_byte

print_counts:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x25, x24

    mov x0, x20
    bl append_uint
    mov w11, #0x20
    strb w11, [x24], #1

    mov x0, x21
    bl append_uint
    mov w11, #0x20
    strb w11, [x24], #1

    mov x0, x19
    bl append_uint
    mov w11, #0x0A
    strb w11, [x24], #1

    mov x0, #1
    mov x1, x25
    sub x2, x24, x25
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0
hang:
    b hang

// x0 = value, x24 = destination (advanced). Clobbers x0-x5.
append_uint:
    sub sp, sp, #32
    mov x1, #10
    add x2, sp, #32
1:
    udiv x3, x0, x1
    msub x4, x3, x1, x0
    add w4, w4, #0x30
    sub x2, x2, #1
    strb w4, [x2]
    mov x0, x3
    cbnz x0, 1b
    add x5, sp, #32
2:
    ldrb w4, [x2], #1
    strb w4, [x24], #1
    cmp x2, x5
    b.ne 2b
    add sp, sp, #32
    ret
