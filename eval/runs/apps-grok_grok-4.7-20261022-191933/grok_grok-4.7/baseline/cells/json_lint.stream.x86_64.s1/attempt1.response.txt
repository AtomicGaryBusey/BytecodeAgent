.section .rodata
msg_ok:
    .byte 'o', 'k', 0x0A
msg_err:
    .byte 'e', 'r', 'r', 'o', 'r', ' ', 'a', 't', ' '
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
inbuf:
    .zero 65536
inbuf_end:
outbuf:
    .zero 64
numbuf:
    .zero 32
numbuf_end:

.text
.globl _start
_start:
    lea     r15, [rip + inbuf]
    mov     r12, r15
read_loop:
    xor     eax, eax
    xor     edi, edi
    mov     rsi, r12
    lea     rdx, [rip + inbuf_end]
    sub     rdx, r12
    test    rdx, rdx
    jz      read_done
    syscall
    test    rax, rax
    jle     read_done
    add     r12, rax
    jmp     read_loop
read_done:
    mov     r14, r12
    mov     r13, r15
    call    parse_document
    mov     eax, 1
    mov     edi, 1
    lea     rsi, [rip + msg_ok]
    mov     edx, 3
    syscall
    xor     edi, edi
    jmp     do_exit

error:
    mov     rax, r13
    sub     rax, r15
    lea     r8, [rip + numbuf_end]
    mov     r9d, 10
err_div:
    xor     edx, edx
    div     r9
    add     dl, '0'
    dec     r8
    mov     byte ptr [r8], dl
    test    rax, rax
    jnz     err_div
    lea     rdi, [rip + outbuf]
    lea     rsi, [rip + msg_err]
    mov     ecx, 9
    rep     movsb
    mov     rsi, r8
    lea     rcx, [rip + numbuf_end]
    sub     rcx, r8
err_cp:
    mov     al, byte ptr [rsi]
    mov     byte ptr [rdi], al
    inc     rsi
    inc     rdi
    dec     rcx
    jnz     err_cp
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    lea     rsi, [rip + outbuf]
    mov     rdx, rdi
    sub     rdx, rsi
    mov     eax, 1
    mov     edi, 1
    syscall
    mov     edi, 1
    jmp     do_exit

do_exit:
    mov     eax, 60
    syscall
    jmp     do_exit

parse_document:
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     r13, r14
    jne     error
    ret

skip_ws:
sw_loop:
    cmp     r13, r14
    jae     sw_done
    movzx   eax, byte ptr [r13]
    cmp     al, 0x20
    je      sw_adv
    cmp     al, 0x09
    je      sw_adv
    cmp     al, 0x0A
    je      sw_adv
    cmp     al, 0x0D
    je      sw_adv
    jmp     sw_done
sw_adv:
    inc     r13
    jmp     sw_loop
sw_done:
    ret

parse_value:
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, '{'
    je      parse_object
    cmp     al, '['
    je      parse_array
    cmp     al, 0x22
    je      parse_string
    cmp     al, 't'
    je      pv_true
    cmp     al, 'f'
    je      pv_false
    cmp     al, 'n'
    je      pv_null
    cmp     al, '-'
    je      parse_number
    cmp     al, '0'
    jb      error
    cmp     al, '9'
    ja      error
    jmp     parse_number
pv_true:
    lea     rsi, [rip + lit_true]
    mov     edx, 4
    jmp     match_lit
pv_false:
    lea     rsi, [rip + lit_false]
    mov     edx, 5
    jmp     match_lit
pv_null:
    lea     rsi, [rip + lit_null]
    mov     edx, 4
    jmp     match_lit

parse_object:
    inc     r13
    call    skip_ws
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, '}'
    je      po_close
po_member:
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, 0x22
    jne     error
    call    parse_string
    call    skip_ws
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, ':'
    jne     error
    inc     r13
    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, ','
    jne     po_end
    inc     r13
    call    skip_ws
    jmp     po_member
po_end:
    cmp     al, '}'
    jne     error
po_close:
    inc     r13
    ret

parse_array:
    inc     r13
    call    skip_ws
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, ']'
    je      pa_close
pa_elem:
    call    parse_value
    call    skip_ws
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, ','
    jne     pa_end
    inc     r13
    call    skip_ws
    jmp     pa_elem
pa_end:
    cmp     al, ']'
    jne     error
pa_close:
    inc     r13
    ret

parse_string:
    inc     r13
ps_loop:
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, 0x22
    je      ps_end
    cmp     al, 0x5C
    je      ps_esc
    cmp     al, 0x20
    jb      error
    inc     r13
    jmp     ps_loop
ps_end:
    inc     r13
    ret
ps_esc:
    inc     r13
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, 0x22
    je      ps_esc_ok
    cmp     al, 0x5C
    je      ps_esc_ok
    cmp     al, '/'
    je      ps_esc_ok
    cmp     al, 'b'
    je      ps_esc_ok
    cmp     al, 'f'
    je      ps_esc_ok
    cmp     al, 'n'
    je      ps_esc_ok
    cmp     al, 'r'
    je      ps_esc_ok
    cmp     al, 't'
    je      ps_esc_ok
    cmp     al, 'u'
    je      ps_u
    jmp     error
ps_esc_ok:
    inc     r13
    jmp     ps_loop
ps_u:
    inc     r13
    mov     ecx, 4
ps_hex:
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, '0'
    jb      error
    cmp     al, '9'
    jbe     ps_hex_ok
    cmp     al, 'A'
    jb      error
    cmp     al, 'F'
    jbe     ps_hex_ok
    cmp     al, 'a'
    jb      error
    cmp     al, 'f'
    ja      error
ps_hex_ok:
    inc     r13
    dec     ecx
    jnz     ps_hex
    jmp     ps_loop

parse_number:
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, '-'
    jne     pn_int
    inc     r13
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
pn_int:
    cmp     al, '0'
    jb      error
    cmp     al, '9'
    ja      error
    cmp     al, '0'
    jne     pn_digits
    inc     r13
    jmp     pn_frac
pn_digits:
    inc     r13
pn_digits_loop:
    cmp     r13, r14
    jae     pn_ret
    movzx   eax, byte ptr [r13]
    cmp     al, '0'
    jb      pn_frac_al
    cmp     al, '9'
    ja      pn_frac_al
    inc     r13
    jmp     pn_digits_loop
pn_frac:
    cmp     r13, r14
    jae     pn_ret
    movzx   eax, byte ptr [r13]
pn_frac_al:
    cmp     al, '.'
    jne     pn_exp_al
    inc     r13
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, '0'
    jb      error
    cmp     al, '9'
    ja      error
pn_frac_loop:
    inc     r13
    cmp     r13, r14
    jae     pn_ret
    movzx   eax, byte ptr [r13]
    cmp     al, '0'
    jb      pn_exp_al
    cmp     al, '9'
    ja      pn_exp_al
    jmp     pn_frac_loop
pn_exp_al:
    cmp     al, 'e'
    je      pn_exp
    cmp     al, 'E'
    je      pn_exp
    ret
pn_exp:
    inc     r13
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
    cmp     al, '+'
    je      pn_exp_sign
    cmp     al, '-'
    je      pn_exp_sign
    jmp     pn_exp_dig
pn_exp_sign:
    inc     r13
    cmp     r13, r14
    jae     error
    movzx   eax, byte ptr [r13]
pn_exp_dig:
    cmp     al, '0'
    jb      error
    cmp     al, '9'
    ja      error
pn_exp_loop:
    inc     r13
    cmp     r13, r14
    jae     pn_ret
    movzx   eax, byte ptr [r13]
    cmp     al, '0'
    jb      pn_ret
    cmp     al, '9'
    ja      pn_ret
    jmp     pn_exp_loop
pn_ret:
    ret

match_lit:
ml_loop:
    cmp     r13, r14
    jae     error
    mov     al, byte ptr [rsi]
    cmp     al, byte ptr [r13]
    jne     error
    inc     rsi
    inc     r13
    dec     edx
    jnz     ml_loop
    ret
