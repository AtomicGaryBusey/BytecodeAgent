.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_CAP, 49152
.equ OUT_CAP, 66560
.equ LINE, 76
.equ RD_CHUNK, 4096

.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, xzr

read_loop:
    mov x8, #SYS_read
    mov x0, xzr
    add x1, x19, x20
    mov w2, #IN_CAP
    sub w2, w2, w20
    cmp w2, #RD_CHUNK
    b.ls 1f
    mov w2, #RD_CHUNK
1:
    cbz w2, read_done
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x23, x22
    mov x24, xzr
    mov x25, x19
    mov x26, x20
    adrp x27, alphabet
    add x27, x27, :lo12:alphabet

encode_loop:
    cmp x26, #3
    b.lo partial
    ldrb w0, [x25]
    ldrb w1, [x25, #1]
    ldrb w2, [x25, #2]
    add x25, x25, #3
    sub x26, x26, #3
    lsl w0, w0, #16
    lsl w1, w1, #8
    orr w0, w0, w1
    orr w0, w0, w2
    lsr w1, w0, #18
    and w1, w1, #63
    bl emit_idx
    lsr w1, w0, #12
    and w1, w1, #63
    bl emit_idx
    lsr w1, w0, #6
    and w1, w1, #63
    bl emit_idx
    and w1, w0, #63
    bl emit_idx
    b encode_loop

partial:
    cbz x26, finish
    cmp x26, #1
    b.eq partial_one

partial_two:
    ldrb w0, [x25]
    ldrb w3, [x25, #1]
    lsr w1, w0, #2
    bl emit_idx
    and w0, w0, #3
    lsl w0, w0, #4
    lsr w1, w3, #4
    orr w1, w1, w0
    bl emit_idx
    and w1, w3, #15
    lsl w1, w1, #2
    bl emit_idx
    bl emit_pad
    b finish

partial_one:
    ldrb w0, [x25]
    lsr w1, w0, #2
    bl emit_idx
    and w1, w0, #3
    lsl w1, w1, #4
    bl emit_idx
    bl emit_pad
    bl emit_pad

finish:
    cbz x24, do_write
    mov w2, #10
    strb w2, [x23], #1

do_write:
    sub x2, x23, x22
    cbz x2, do_exit

write_loop:
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x22
    svc #0
    add x22, x22, x0
    sub x2, x23, x22
    cbnz x2, write_loop

do_exit:
    mov x8, #SYS_exit
    mov x0, xzr
    svc #0

emit_idx:
    ldrb w2, [x27, x1]
    b emit_byte

emit_pad:
    mov w2, #61

emit_byte:
    strb w2, [x23], #1
    add x24, x24, #1
    cmp x24, #LINE
    b.ne 1f
    mov w2, #10
    strb w2, [x23], #1
    mov x24, xzr
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    .ascii "abcdefghijklmnopqrstuvwxyz"
    .ascii "0123456789"
    .ascii "+/"

.bss
.align 3
inbuf:
    .zero IN_CAP
outbuf:
    .zero OUT_CAP
