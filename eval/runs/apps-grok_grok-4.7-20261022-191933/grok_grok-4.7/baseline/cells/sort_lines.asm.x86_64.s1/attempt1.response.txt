.globl _start
.text
_start:
    lea rbx, [rip + inbuf]
    mov r12, rbx
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    lea rdx, [rbx + 32768]
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    lea rdx, [rbx + 32768]
    cmp r12, rdx
    jb read_loop
read_done:
    mov r14, rbx
    mov r15, rbx
    xor r13, r13
    lea rbp, [rip + lines]
scan_loop:
    cmp r14, r12
    jae scan_done
    movzx eax, byte ptr [r14]
    cmp al, 10
    jne scan_next
    mov qword ptr [rbp + r13*8], r15
    mov rax, r14
    sub rax, r15
    lea rdx, [rbp + 8000]
    mov qword ptr [rdx + r13*8], rax
    inc r13
    lea r15, [r14 + 1]
scan_next:
    inc r14
    jmp scan_loop
scan_done:
    cmp r15, r12
    je sort_init
    mov qword ptr [rbp + r13*8], r15
    mov rax, r12
    sub rax, r15
    lea rdx, [rbp + 8000]
    mov qword ptr [rdx + r13*8], rax
    inc r13
sort_init:
    lea r12, [rbp + 8000]
    mov r14, 1
sort_loop:
    cmp r14, r13
    jae emit_init
    mov rax, qword ptr [rbp + r14*8]
    mov rbx, qword ptr [r12 + r14*8]
    mov r15, r14
    dec r15
insert_loop:
    mov r8, rbx
    mov r9, qword ptr [r12 + r15*8]
    cmp r8, r9
    cmova r8, r9
    mov rsi, rax
    mov rdi, qword ptr [rbp + r15*8]
chunk_loop:
    cmp r8, 8
    jb tail_bytes
    mov rdx, qword ptr [rsi]
    mov rcx, qword ptr [rdi]
    bswap rdx
    bswap rcx
    cmp rdx, rcx
    jne cmp_done
    add rsi, 8
    add rdi, 8
    sub r8, 8
    jmp chunk_loop
tail_bytes:
    test r8, r8
    jz cmp_lens
tail_loop:
    movzx edx, byte ptr [rsi]
    movzx ecx, byte ptr [rdi]
    cmp dl, cl
    jne cmp_done
    inc rsi
    inc rdi
    dec r8
    jnz tail_loop
cmp_lens:
    cmp rbx, r9
cmp_done:
    jae insert_at
    mov rcx, qword ptr [rbp + r15*8]
    mov rdx, qword ptr [r12 + r15*8]
    mov rsi, r15
    inc rsi
    mov qword ptr [rbp + rsi*8], rcx
    mov qword ptr [r12 + rsi*8], rdx
    test r15, r15
    jz insert_front
    dec r15
    jmp insert_loop
insert_front:
    mov qword ptr [rbp], rax
    mov qword ptr [r12], rbx
    jmp sort_next
insert_at:
    lea rcx, [r15 + 1]
    mov qword ptr [rbp + rcx*8], rax
    mov qword ptr [r12 + rcx*8], rbx
sort_next:
    inc r14
    jmp sort_loop
emit_init:
    xor r14, r14
emit_loop:
    cmp r14, r13
    jae quit
    mov rdx, qword ptr [r12 + r14*8]
    test rdx, rdx
    jz emit_nl
    mov eax, 1
    mov edi, 1
    mov rsi, qword ptr [rbp + r14*8]
    syscall
emit_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nlbyte]
    mov edx, 1
    syscall
    inc r14
    jmp emit_loop
quit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp quit

.section .rodata
nlbyte:
    .byte 10

.bss
.align 8
inbuf:
    .zero 32768
.align 8
lines:
    .zero 8000
    .zero 8000
