.bss
.balign 8
inbuf:
    .zero 65536
outbuf:
    .zero 128

.text
.globl _start

_start:
    mov x19, #0
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x20, x19
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x19
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    mov x2, #1
    lsl x2, x2, #16
    cmp x19, x2
    b.lo read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x19
    b.hs rows_done
    sub x22, x19, x21
    cmp x22, #16
    b.ls row_have_n
    mov x22, #16
row_have_n:
    bl print_row
    add x21, x21, x22
    b row_loop

rows_done:
    bl print_final
    mov x8, #93
    mov x0, #0
    svc #0

print_row:
    stp x29, x30, [sp, #-16]!
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, #0

    mov x0, x21
    bl fmt_hex8
    mov w1, #32
    bl putc
    mov w1, #32
    bl putc

    mov x25, #0
pr_slots:
    cmp x25, x22
    b.hs pr_pad
    add x0, x20, x21
    ldrb w0, [x0, x25]
    bl fmt_hex2
    mov w1, #32
    bl putc
    b pr_after_slot
pr_pad:
    mov w1, #32
    bl putc
    mov w1, #32
    bl putc
    mov w1, #32
    bl putc
pr_after_slot:
    cmp x25, #7
    b.ne pr_no_gap
    mov w1, #32
    bl putc
pr_no_gap:
    add x25, x25, #1
    cmp x25, #16
    b.ne pr_slots

    mov w1, #32
    bl putc
    mov w1, #124
    bl putc

    mov x25, #0
pr_ascii:
    cmp x25, x22
    b.hs pr_ascii_done
    add x0, x20, x21
    ldrb w0, [x0, x25]
    cmp w0, #32
    b.lo pr_dot
    cmp w0, #126
    b.hi pr_dot
    mov w1, w0
    b pr_ascii_putc
pr_dot:
    mov w1, #46
pr_ascii_putc:
    bl putc
    add x25, x25, #1
    b pr_ascii

pr_ascii_done:
    mov w1, #124
    bl putc
    mov w1, #10
    bl putc

    mov x8, #64
    mov x0, #1
    mov x1, x23
    mov x2, x24
    svc #0
    ldp x29, x30, [sp], #16
    ret

print_final:
    stp x29, x30, [sp, #-16]!
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, #0
    mov x0, x19
    bl fmt_hex8
    mov w1, #10
    bl putc
    mov x8, #64
    mov x0, #1
    mov x1, x23
    mov x2, x24
    svc #0
    ldp x29, x30, [sp], #16
    ret

fmt_hex8:
    stp x29, x30, [sp, #-32]!
    stp x25, x26, [sp, #16]
    mov x25, x0
    mov x26, #7
fh8_loop:
    lsl x0, x26, #2
    lsr x0, x25, x0
    and x0, x0, #15
    bl hex_digit
    mov w1, w0
    bl putc
    subs x26, x26, #1
    b.pl fh8_loop
    ldp x25, x26, [sp, #16]
    ldp x29, x30, [sp], #32
    ret

fmt_hex2:
    stp x30, x25, [sp, #-16]!
    mov w25, w0
    lsr w0, w25, #4
    bl hex_digit
    mov w1, w0
    bl putc
    and w0, w25, #15
    bl hex_digit
    mov w1, w0
    bl putc
    ldp x30, x25, [sp], #16
    ret

hex_digit:
    cmp w0, #10
    b.hs hex_digit_alpha
    add w0, w0, #48
    ret
hex_digit_alpha:
    add w0, w0, #87
    ret

putc:
    strb w1, [x23, x24]
    add x24, x24, #1
    ret
