.globl _start

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537

.text
_start:
    lea r14, [rip + inbuf]
    lea r15, [rip + outbuf]
    xor rbx, rbx

read_loop:
    mov edx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz process
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + rbx]
    syscall
    test rax, rax
    jle process
    add rbx, rax
    jmp read_loop

process:
    xor r12, r12
    xor r13, r13

line_loop:
    cmp r12, rbx
    jae write_out
    mov r8, r12

scan:
    cmp r12, rbx
    jae reverse
    movzx eax, byte ptr [r14 + r12]
    cmp al, 10
    je reverse
    inc r12
    jmp scan

reverse:
    mov r10, r12

rev_byte:
    cmp r10, r8
    jbe rev_nl
    dec r10
    movzx eax, byte ptr [r14 + r10]
    mov byte ptr [r15 + r13], al
    inc r13
    jmp rev_byte

rev_nl:
    mov byte ptr [r15 + r13], 10
    inc r13
    cmp r12, rbx
    jae write_out
    inc r12
    jmp line_loop

write_out:
    test r13, r13
    jz do_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r13
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
