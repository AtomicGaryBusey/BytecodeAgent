.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ CH_TAB, 0x09
.equ CH_LF, 0x0A
.equ CH_CR, 0x0D
.equ CH_SP, 0x20
.equ CH_QUOT, 0x22
.equ CH_PLUS, 0x2B
.equ CH_COMMA, 0x2C
.equ CH_MINUS, 0x2D
.equ CH_DOT, 0x2E
.equ CH_SLASH, 0x2F
.equ CH_COLON, 0x3A
.equ CH_LBR, 0x5B
.equ CH_BSL, 0x5C
.equ CH_RBR, 0x5D
.equ CH_LBE, 0x7B
.equ CH_RBE, 0x7D
.equ LEN_TRUE, 4
.equ LEN_FALSE, 5
.equ LEN_NULL, 4
.equ LEN_OK, 3
.equ LEN_ERR, 9

.globl _start
.text
.balign 4

_start:
	adrp x19, input
	add x19, x19, :lo12:input
	mov x20, #0
read_loop:
	mov x0, #0
	add x1, x19, x20
	movz x2, #1, lsl #16
	sub x2, x2, x20
	cbz x2, read_done
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	b.le read_done
	add x20, x20, x0
	b read_loop
read_done:
	mov x21, #0
	bl parse_doc
	cbnz w0, do_error
	mov x0, #1
	adrp x1, ok_msg
	add x1, x1, :lo12:ok_msg
	mov x2, #LEN_OK
	mov x8, #SYS_write
	svc #0
exit_ok:
	mov x0, #0
	mov x8, #SYS_exit
	svc #0
	b exit_ok

do_error:
	adrp x22, outbuf
	add x22, x22, :lo12:outbuf
	mov x24, x22
	adrp x23, err_prefix
	add x23, x23, :lo12:err_prefix
	mov w2, #LEN_ERR
de_cp:
	ldrb w3, [x23], #1
	strb w3, [x22], #1
	subs w2, w2, #1
	b.ne de_cp
	mov w0, w21
	mov w1, #10
	adrp x4, digitbuf
	add x4, x4, :lo12:digitbuf
	mov x5, #0
	cbz w0, de_zero
de_div:
	udiv w2, w0, w1
	msub w3, w2, w1, w0
	add w3, w3, #'0'
	strb w3, [x4, x5]
	add x5, x5, #1
	mov w0, w2
	cbnz w0, de_div
de_emit:
	sub x5, x5, #1
	ldrb w3, [x4, x5]
	strb w3, [x22], #1
	cbnz x5, de_emit
	b de_nl
de_zero:
	mov w3, #'0'
	strb w3, [x22], #1
de_nl:
	mov w3, #CH_LF
	strb w3, [x22], #1
	mov x0, #1
	mov x1, x24
	sub x2, x22, x24
	mov x8, #SYS_write
	svc #0
exit_err:
	mov x0, #1
	mov x8, #SYS_exit
	svc #0
	b exit_err

skip_ws:
sw_loop:
	cmp x21, x20
	b.hs sw_done
	ldrb w0, [x19, x21]
	cmp w0, #CH_SP
	b.eq sw_inc
	cmp w0, #CH_TAB
	b.eq sw_inc
	cmp w0, #CH_LF
	b.eq sw_inc
	cmp w0, #CH_CR
	b.eq sw_inc
	b sw_done
sw_inc:
	add x21, x21, #1
	b sw_loop
sw_done:
	ret

// x0 = address of literal, x1 = its length
parse_lit:
	mov x2, #0
pl_loop:
	cmp x2, x1
	b.hs pl_ok
	cmp x21, x20
	b.hs pl_fail
	ldrb w3, [x19, x21]
	ldrb w4, [x0, x2]
	cmp w3, w4
	b.ne pl_fail
	add x21, x21, #1
	add x2, x2, #1
	b pl_loop
pl_ok:
	mov x0, #0
	ret
pl_fail:
	mov x0, #1
	ret

parse_number:
	ldrb w0, [x19, x21]
	cmp w0, #CH_MINUS
	b.ne pn_int
	add x21, x21, #1
pn_int:
	cmp x21, x20
	b.hs pn_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.eq pn_zero
	cmp w0, #'1'
	b.lo pn_fail
	cmp w0, #'9'
	b.hi pn_fail
	add x21, x21, #1
pn_digits:
	cmp x21, x20
	b.hs pn_ok
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo pn_frac
	cmp w0, #'9'
	b.hi pn_frac
	add x21, x21, #1
	b pn_digits
pn_zero:
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_ok
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo pn_frac
	cmp w0, #'9'
	b.hi pn_frac
	b pn_fail
pn_frac:
	cmp w0, #CH_DOT
	b.ne pn_exp
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo pn_fail
	cmp w0, #'9'
	b.hi pn_fail
	add x21, x21, #1
pn_frac_more:
	cmp x21, x20
	b.hs pn_ok
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo pn_exp
	cmp w0, #'9'
	b.hi pn_exp
	add x21, x21, #1
	b pn_frac_more
pn_exp:
	cmp w0, #'e'
	b.eq pn_doexp
	cmp w0, #'E'
	b.eq pn_doexp
	b pn_ok
pn_doexp:
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_PLUS
	b.eq pn_expsign
	cmp w0, #CH_MINUS
	b.eq pn_expsign
	b pn_expdig
pn_expsign:
	add x21, x21, #1
pn_expdig:
	cmp x21, x20
	b.hs pn_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo pn_fail
	cmp w0, #'9'
	b.hi pn_fail
	add x21, x21, #1
pn_expmore:
	cmp x21, x20
	b.hs pn_ok
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo pn_ok
	cmp w0, #'9'
	b.hi pn_ok
	add x21, x21, #1
	b pn_expmore
pn_ok:
	mov x0, #0
	ret
pn_fail:
	mov x0, #1
	ret

parse_string:
	cmp x21, x20
	b.hs ps_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_QUOT
	b.ne ps_fail
	add x21, x21, #1
ps_loop:
	cmp x21, x20
	b.hs ps_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_QUOT
	b.eq ps_end
	cmp w0, #CH_BSL
	b.eq ps_esc
	cmp w0, #CH_SP
	b.lo ps_fail
	add x21, x21, #1
	b ps_loop
ps_end:
	add x21, x21, #1
	mov x0, #0
	ret
ps_esc:
	add x21, x21, #1
	cmp x21, x20
	b.hs ps_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_QUOT
	b.eq ps_esc_ok
	cmp w0, #CH_BSL
	b.eq ps_esc_ok
	cmp w0, #CH_SLASH
	b.eq ps_esc_ok
	cmp w0, #'b'
	b.eq ps_esc_ok
	cmp w0, #'f'
	b.eq ps_esc_ok
	cmp w0, #'n'
	b.eq ps_esc_ok
	cmp w0, #'r'
	b.eq ps_esc_ok
	cmp w0, #'t'
	b.eq ps_esc_ok
	cmp w0, #'u'
	b.eq ps_u
	b ps_fail
ps_esc_ok:
	add x21, x21, #1
	b ps_loop
ps_u:
	add x21, x21, #1
	mov w1, #4
ps_hex:
	cmp x21, x20
	b.hs ps_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo ps_fail
	cmp w0, #'9'
	b.ls ps_hex_ok
	cmp w0, #'A'
	b.lo ps_fail
	cmp w0, #'F'
	b.ls ps_hex_ok
	cmp w0, #'a'
	b.lo ps_fail
	cmp w0, #'f'
	b.hi ps_fail
ps_hex_ok:
	add x21, x21, #1
	subs w1, w1, #1
	b.ne ps_hex
	b ps_loop
ps_fail:
	mov x0, #1
	ret

parse_value:
	stp x29, x30, [sp, #-16]!
	bl skip_ws
	cmp x21, x20
	b.hs pv_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_LBE
	b.eq pv_obj
	cmp w0, #CH_LBR
	b.eq pv_arr
	cmp w0, #CH_QUOT
	b.eq pv_str
	cmp w0, #'t'
	b.eq pv_true
	cmp w0, #'f'
	b.eq pv_false
	cmp w0, #'n'
	b.eq pv_null
	cmp w0, #CH_MINUS
	b.eq pv_num
	cmp w0, #'0'
	b.lo pv_fail
	cmp w0, #'9'
	b.ls pv_num
	b pv_fail
pv_obj:
	bl parse_object
	b pv_done
pv_arr:
	bl parse_array
	b pv_done
pv_str:
	bl parse_string
	b pv_done
pv_num:
	bl parse_number
	b pv_done
pv_true:
	adrp x0, lit_true
	add x0, x0, :lo12:lit_true
	mov x1, #LEN_TRUE
	bl parse_lit
	b pv_done
pv_false:
	adrp x0, lit_false
	add x0, x0, :lo12:lit_false
	mov x1, #LEN_FALSE
	bl parse_lit
	b pv_done
pv_null:
	adrp x0, lit_null
	add x0, x0, :lo12:lit_null
	mov x1, #LEN_NULL
	bl parse_lit
	b pv_done
pv_done:
	ldp x29, x30, [sp], #16
	ret
pv_fail:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret

parse_object:
	stp x29, x30, [sp, #-16]!
	add x21, x21, #1
	bl skip_ws
	cmp x21, x20
	b.hs po_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_RBE
	b.eq po_end
po_member:
	bl parse_string
	cbnz w0, po_fail
	bl skip_ws
	cmp x21, x20
	b.hs po_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_COLON
	b.ne po_fail
	add x21, x21, #1
	bl parse_value
	cbnz w0, po_fail
	bl skip_ws
	cmp x21, x20
	b.hs po_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_COMMA
	b.eq po_comma
	cmp w0, #CH_RBE
	b.eq po_end
	b po_fail
po_comma:
	add x21, x21, #1
	bl skip_ws
	b po_member
po_end:
	add x21, x21, #1
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret
po_fail:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret

parse_array:
	stp x29, x30, [sp, #-16]!
	add x21, x21, #1
	bl skip_ws
	cmp x21, x20
	b.hs pa_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_RBR
	b.eq pa_end
pa_val:
	bl parse_value
	cbnz w0, pa_fail
	bl skip_ws
	cmp x21, x20
	b.hs pa_fail
	ldrb w0, [x19, x21]
	cmp w0, #CH_COMMA
	b.eq pa_comma
	cmp w0, #CH_RBR
	b.eq pa_end
	b pa_fail
pa_comma:
	add x21, x21, #1
	bl skip_ws
	b pa_val
pa_end:
	add x21, x21, #1
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret
pa_fail:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret

parse_doc:
	stp x29, x30, [sp, #-16]!
	bl parse_value
	cbnz w0, pd_fail
	bl skip_ws
	cmp x21, x20
	b.ne pd_fail
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret
pd_fail:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret

.section .rodata
.balign 4
ok_msg:
	.ascii "ok\n"
err_prefix:
	.ascii "error at "
lit_true:
	.ascii "true"
lit_false:
	.ascii "false"
lit_null:
	.ascii "null"

.section .bss
.balign 8
input:
	.zero 65536
outbuf:
	.zero 32
digitbuf:
	.zero 16
