.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.section .bss
.balign 8
inbuf:
    .zero 65536
linebuf:
    .zero 80

.text
.balign 4
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, linebuf
    add x22, x22, :lo12:linebuf
    adrp x23, hexdigits
    add x23, x23, :lo12:hexdigits

    mov x20, xzr
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    mov x0, xzr
    add x1, x19, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, xzr
row_loop:
    cmp x21, x20
    b.hs final_offset
    sub x24, x20, x21
    cmp x24, #16
    b.ls row_n_ok
    mov x24, #16
row_n_ok:
    mov x0, xzr
    mov w1, #32
clear_loop:
    strb w1, [x22, x0]
    add x0, x0, #1
    cmp x0, #60
    b.lo clear_loop
    mov x0, x21
    mov x1, x22
    mov x2, #8
    bl put_hex
    mov x25, xzr
byte_loop:
    cmp x25, x24
    b.hs bytes_done
    add x0, x21, x25
    ldrb w26, [x19, x0]
    add x3, x25, x25, lsl #1
    cmp x25, #8
    b.hs high_idx
    add x1, x3, #10
    b have_idx
high_idx:
    add x1, x3, #11
have_idx:
    add x1, x22, x1
    mov x0, x26
    mov x2, #2
    bl put_hex
    add x1, x22, #61
    add x1, x1, x25
    cmp w26, #32
    b.lo put_dot
    cmp w26, #126
    b.hi put_dot
    strb w26, [x1]
    b ascii_done
put_dot:
    mov w5, #46
    strb w5, [x1]
ascii_done:
    add x25, x25, #1
    b byte_loop
bytes_done:
    mov w5, #124
    strb w5, [x22, #60]
    add x1, x22, #61
    add x1, x1, x24
    strb w5, [x1]
    mov w5, #10
    strb w5, [x1, #1]
    add x2, x24, #63
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x22
    svc #0
    add x21, x21, x24
    b row_loop
final_offset:
    mov x0, x20
    mov x1, x22
    mov x2, #8
    bl put_hex
    mov w5, #10
    strb w5, [x22, #8]
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x22
    mov x2, #9
    svc #0
    mov x8, #SYS_exit
    mov x0, xzr
    svc #0

// put_hex: write the low x2 hex digits of x0 to [x1]; x23 = hexdigits
put_hex:
    sub x3, x2, #1
    lsl x3, x3, #2
put_hex_loop:
    lsr x4, x0, x3
    and x4, x4, #15
    ldrb w5, [x23, x4]
    strb w5, [x1], #1
    subs x3, x3, #4
    b.pl put_hex_loop
    ret
