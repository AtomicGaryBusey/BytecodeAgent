.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 4096

.bss
.balign 16
buf:
    .zero BUF_SIZE

.balign 16
outbuf:
    .zero 64

.text

// x0 = value, x1 = dest; advances x1. Clobbers x2-x6.
u64_to_dec:
    cbnz x0, 1f
    mov w2, #0x30
    strb w2, [x1], #1
    ret
1:
    sub sp, sp, #32
    mov x2, #0
    mov x4, #10
2:
    udiv x5, x0, x4
    mul x6, x5, x4
    sub x6, x0, x6
    add x6, x6, #0x30
    strb w6, [sp, x2]
    add x2, x2, #1
    mov x0, x5
    cbnz x0, 2b
3:
    sub x2, x2, #1
    ldrb w6, [sp, x2]
    strb w6, [x1], #1
    cbnz x2, 3b
    add sp, sp, #32
    ret

_start:
    // x19 = lines, x20 = words, x21 = bytes, x22 = in_word
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x8, #SYS_read
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #BUF_SIZE
    svc #0
    cmp x0, #0
    b.le done_read
    mov x23, x0
    adrp x24, buf
    add x24, x24, :lo12:buf

byte_loop:
    cbz x23, read_loop
    ldrb w25, [x24], #1
    sub x23, x23, #1
    add x21, x21, #1
    cmp w25, #0x0A
    b.ne 1f
    add x19, x19, #1
1:
    // whitespace: 0x09-0x0D and 0x20
    sub w3, w25, #9
    cmp w3, #5
    b.lo is_ws
    cmp w25, #0x20
    b.eq is_ws
    cbnz x22, byte_loop
    mov x22, #1
    add x20, x20, #1
    b byte_loop

is_ws:
    mov x22, #0
    b byte_loop

done_read:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x28, x1

    mov x0, x19
    bl u64_to_dec
    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x20
    bl u64_to_dec
    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x21
    bl u64_to_dec
    mov w2, #0x0A
    strb w2, [x1], #1

    sub x2, x1, x28
    mov x0, #1
    mov x1, x28
    mov x8, #SYS_write
    svc #0

1:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b 1b
