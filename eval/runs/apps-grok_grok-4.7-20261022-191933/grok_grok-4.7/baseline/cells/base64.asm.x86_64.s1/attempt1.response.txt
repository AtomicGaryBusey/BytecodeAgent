.globl _start

.text
_start:
  xor rbx, rbx
read_loop:
  mov edx, 49152
  sub rdx, rbx
  test rdx, rdx
  jz read_done
  xor eax, eax
  xor edi, edi
  lea rsi, [rip + inbuf]
  add rsi, rbx
  syscall
  test rax, rax
  jle read_done
  add rbx, rax
  jmp read_loop

read_done:
  test rbx, rbx
  jz exit0
  mov rcx, rbx
  lea rsi, [rip + inbuf]
  lea rdi, [rip + outbuf]
  mov r14, rdi
  lea rbx, [rip + b64tab]
  xor r8d, r8d

encode_loop:
  cmp rcx, 3
  jb encode_tail
  movzx eax, byte ptr [rsi]
  movzx edx, byte ptr [rsi + 1]
  movzx r9d, byte ptr [rsi + 2]
  mov r10d, eax
  shr r10d, 2
  mov r11d, eax
  and r11d, 3
  shl r11d, 4
  mov r12d, edx
  shr r12d, 4
  or r11d, r12d
  mov r12d, edx
  and r12d, 15
  shl r12d, 2
  mov r13d, r9d
  shr r13d, 6
  or r12d, r13d
  and r9d, 63
  mov eax, r10d
  call emit_idx
  mov eax, r11d
  call emit_idx
  mov eax, r12d
  call emit_idx
  mov eax, r9d
  call emit_idx
  add rsi, 3
  sub rcx, 3
  jmp encode_loop

encode_tail:
  cmp rcx, 1
  je one_byte
  cmp rcx, 2
  je two_byte
  jmp finish

one_byte:
  movzx eax, byte ptr [rsi]
  mov r10d, eax
  shr r10d, 2
  and eax, 3
  shl eax, 4
  mov r11d, eax
  mov eax, r10d
  call emit_idx
  mov eax, r11d
  call emit_idx
  mov al, 0x3D
  call emit_raw
  mov al, 0x3D
  call emit_raw
  jmp finish

two_byte:
  movzx eax, byte ptr [rsi]
  movzx edx, byte ptr [rsi + 1]
  mov r10d, eax
  shr r10d, 2
  mov r11d, eax
  and r11d, 3
  shl r11d, 4
  mov r12d, edx
  shr r12d, 4
  or r11d, r12d
  and edx, 15
  shl edx, 2
  mov eax, r10d
  call emit_idx
  mov eax, r11d
  call emit_idx
  mov eax, edx
  call emit_idx
  mov al, 0x3D
  call emit_raw

finish:
  test r8d, r8d
  jz do_write
  mov byte ptr [rdi], 10
  inc rdi

do_write:
  mov r15, rdi
write_loop:
  mov rdx, r15
  sub rdx, r14
  test rdx, rdx
  jz exit0
  mov eax, 1
  mov edi, 1
  mov rsi, r14
  syscall
  test rax, rax
  jle exit0
  add r14, rax
  jmp write_loop

exit0:
  mov eax, 60
  xor edi, edi
  syscall

emit_idx:
  movzx eax, byte ptr [rbx + rax]
emit_raw:
  mov byte ptr [rdi], al
  inc rdi
  inc r8d
  cmp r8d, 76
  jne 1f
  mov byte ptr [rdi], 10
  inc rdi
  xor r8d, r8d
1:
  ret
