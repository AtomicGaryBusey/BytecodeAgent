.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_MAX, 65536
.equ SPACE, 32
.equ PIPE, 124
.equ NEWLINE, 10
.equ DOT, 46

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.balign 8
input_buf:
    .zero BUF_MAX
line_buf:
    .zero 128

.text
.balign 4
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #BUF_MAX

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, x21
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    cmp x0, x21
    b.hi read_done
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, read_loop

read_done:
    mov x21, #0
    mov x22, x19

row_loop:
    cmp x21, x20
    b.hs rows_done
    sub x23, x20, x21
    cmp x23, #16
    b.ls row_count_ok
    mov x23, #16
row_count_ok:
    mov x0, x22
    mov x1, x21
    mov x2, x23
    bl format_and_write
    add x22, x22, x23
    add x21, x21, x23
    b row_loop

rows_done:
    bl print_end
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// w11 value, x13 dest (advanced by 8). Clobbers w11, w14, w15, w17, x16.
hex8:
    adrp x16, hextab
    add x16, x16, :lo12:hextab
    mov w14, #8
hex8_loop:
    lsr w15, w11, #28
    and w15, w15, #0xf
    ldrb w17, [x16, x15]
    strb w17, [x13], #1
    lsl w11, w11, #4
    subs w14, w14, #1
    b.ne hex8_loop
    ret

// x0 data, x1 offset, x2 length n (1..16). Preserves x19-x28.
format_and_write:
    stp x29, x30, [sp, #-16]!
    mov x9, x0
    mov x10, x2
    mov w11, w1
    adrp x12, line_buf
    add x12, x12, :lo12:line_buf
    mov x13, x12
    bl hex8
    mov w15, #SPACE
    strb w15, [x13], #1
    strb w15, [x13], #1
    adrp x16, hextab
    add x16, x16, :lo12:hextab
    mov w14, #0
slot_loop:
    cmp w14, #8
    b.ne slot_no_gap
    strb w15, [x13], #1
slot_no_gap:
    cmp w14, w10
    b.hs slot_pad
    ldrb w17, [x9, x14]
    lsr w0, w17, #4
    and w0, w0, #0xf
    ldrb w0, [x16, x0]
    strb w0, [x13], #1
    and w1, w17, #0xf
    ldrb w1, [x16, x1]
    strb w1, [x13], #1
    strb w15, [x13], #1
    b slot_next
slot_pad:
    strb w15, [x13], #1
    strb w15, [x13], #1
    strb w15, [x13], #1
slot_next:
    add w14, w14, #1
    cmp w14, #16
    b.lo slot_loop
    strb w15, [x13], #1
    mov w0, #PIPE
    strb w0, [x13], #1
    mov w14, #0
ascii_loop:
    cmp w14, w10
    b.hs ascii_done
    ldrb w17, [x9, x14]
    cmp w17, #0x20
    b.lo ascii_dot
    cmp w17, #0x7e
    b.hi ascii_dot
    strb w17, [x13], #1
    b ascii_next
ascii_dot:
    mov w17, #DOT
    strb w17, [x13], #1
ascii_next:
    add w14, w14, #1
    b ascii_loop
ascii_done:
    mov w0, #PIPE
    strb w0, [x13], #1
    mov w0, #NEWLINE
    strb w0, [x13], #1
    sub x2, x13, x12
    mov x1, x12
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    ldp x29, x30, [sp], #16
    ret

// Total length is x20. Prints 8 hex digits and a newline.
print_end:
    stp x29, x30, [sp, #-16]!
    adrp x12, line_buf
    add x12, x12, :lo12:line_buf
    mov x13, x12
    mov w11, w20
    bl hex8
    mov w0, #NEWLINE
    strb w0, [x13], #1
    sub x2, x13, x12
    mov x1, x12
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    ldp x29, x30, [sp], #16
    ret
