.globl _start

.text
_start:
    lea rbx, [rip + input_buf]
    xor r12, r12

read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor rcx, rcx

find_bang:
    cmp rcx, r12
    je no_bang
    cmp byte ptr [rbx + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov r13, rcx
    lea r14, [rbx + rcx + 1]
    mov r15, r12
    sub r15, rcx
    dec r15
    jmp do_match

no_bang:
    mov r13, r12
    xor r15, r15
    xor r14, r14

do_match:
    lea rbp, [rip + jump_table]
    lea r11, [rip + bracket_stack]
    xor rdx, rdx
    xor rcx, rcx

match_loop:
    cmp rcx, r13
    je match_done
    mov al, byte ptr [rbx + rcx]
    cmp al, 0x5b
    je match_open
    cmp al, 0x5d
    je match_close
    inc rcx
    jmp match_loop

match_open:
    cmp rdx, 1000
    jae fail
    mov qword ptr [r11 + rdx*8], rcx
    inc rdx
    inc rcx
    jmp match_loop

match_close:
    test rdx, rdx
    jz fail
    dec rdx
    mov rax, qword ptr [r11 + rdx*8]
    lea rsi, [rcx + 1]
    mov dword ptr [rbp + rax*4], esi
    lea rsi, [rax + 1]
    mov dword ptr [rbp + rcx*4], esi
    inc rcx
    jmp match_loop

match_done:
    test rdx, rdx
    jnz fail
    xor r8, r8
    lea r9, [rip + tape]
    xor r10, r10

exec_loop:
    cmp r8, r13
    je ok_exit
    movzx eax, byte ptr [rbx + r8]
    cmp al, 0x3e
    je c_right
    cmp al, 0x3c
    je c_left
    cmp al, 0x2b
    je c_inc
    cmp al, 0x2d
    je c_dec
    cmp al, 0x2e
    je c_out
    cmp al, 0x2c
    je c_in
    cmp al, 0x5b
    je c_lbr
    cmp al, 0x5d
    je c_rbr
    inc r8
    jmp exec_loop

c_right:
    inc r9
    inc r8
    jmp exec_loop

c_left:
    dec r9
    inc r8
    jmp exec_loop

c_inc:
    inc byte ptr [r9]
    inc r8
    jmp exec_loop

c_dec:
    dec byte ptr [r9]
    inc r8
    jmp exec_loop

c_out:
    mov rax, 1
    mov rdi, 1
    mov rsi, r9
    mov rdx, 1
    syscall
    inc r8
    jmp exec_loop

c_in:
    cmp r10, r15
    jae c_in_eof
    mov al, byte ptr [r14 + r10]
    mov byte ptr [r9], al
    inc r10
    inc r8
    jmp exec_loop

c_in_eof:
    mov byte ptr [r9], 0
    inc r8
    jmp exec_loop

c_lbr:
    cmp byte ptr [r9], 0
    jne c_next
    mov r8d, dword ptr [rbp + r8*4]
    jmp exec_loop

c_rbr:
    cmp byte ptr [r9], 0
    je c_next
    mov r8d, dword ptr [rbp + r8*4]
    jmp exec_loop

c_next:
    inc r8
    jmp exec_loop

ok_exit:
    mov rax, 60
    xor rdi, rdi
    syscall

fail:
    mov rax, 60
    mov rdi, 1
    syscall

.bss
.align 8
input_buf:
    .zero 65536
tape:
    .zero 30000
.align 8
jump_table:
    .zero 262144
bracket_stack:
    .zero 8000
