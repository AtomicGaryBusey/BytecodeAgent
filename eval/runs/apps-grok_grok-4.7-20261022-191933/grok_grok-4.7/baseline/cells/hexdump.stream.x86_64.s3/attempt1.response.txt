.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
input:
    .zero 65536
outbuf:
    .zero 128

.text
put_hex32:
    mov ecx, 8
put_hex_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz put_hex_loop
    ret

.globl _start
_start:
    lea r10, [rip + hexdigits]
    lea r14, [rip + input]
    xor r12d, r12d

read_loop:
    cmp r12d, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r12]
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_line
    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe row_n_ok
    mov r15d, 16
row_n_ok:
    lea r8, [r14 + r13]
    lea rdi, [rip + outbuf]
    mov eax, r13d
    call put_hex32
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2
    xor ebx, ebx

hex_loop:
    cmp ebx, r15d
    jae hex_pad
    movzx eax, byte ptr [r8 + rbx]
    mov edx, eax
    and edx, 15
    shr eax, 4
    movzx eax, byte ptr [r10 + rax]
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], al
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp hex_next
hex_pad:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
hex_next:
    inc ebx
    cmp ebx, 8
    jne hex_nomid
    mov byte ptr [rdi], 32
    inc rdi
hex_nomid:
    cmp ebx, 16
    jb hex_loop

    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi
    xor ebx, ebx

char_loop:
    cmp ebx, r15d
    jae char_done
    movzx eax, byte ptr [r8 + rbx]
    cmp al, 32
    jb char_dot
    cmp al, 126
    ja char_dot
    mov byte ptr [rdi], al
    jmp char_stored
char_dot:
    mov byte ptr [rdi], 46
char_stored:
    inc rdi
    inc ebx
    jmp char_loop

char_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    add r13, r15
    jmp row_loop

final_line:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call put_hex32
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
