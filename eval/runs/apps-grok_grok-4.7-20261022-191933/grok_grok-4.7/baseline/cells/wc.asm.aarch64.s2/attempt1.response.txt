.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.text
.globl _start

_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr

read_loop:
    mov x8, #SYS_read
    mov x0, xzr
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #BUF_SIZE
    svc #0
    cmp x0, #0
    b.le print_counts
    add x21, x21, x0
    mov x3, x0
    adrp x4, inbuf
    add x4, x4, :lo12:inbuf

scan_loop:
    cbz x3, read_loop
    ldrb w5, [x4], #1
    sub x3, x3, #1
    cmp w5, #10
    b.ne not_nl
    add x19, x19, #1
not_nl:
    cmp w5, #32
    b.eq is_ws
    sub w6, w5, #9
    cmp w6, #5
    b.lo is_ws
    cbnz x22, scan_loop
    mov x22, #1
    add x20, x20, #1
    b scan_loop
is_ws:
    mov x22, xzr
    b scan_loop

print_counts:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    mov x0, x19
    bl utoa
    mov w1, #32
    strb w1, [x24], #1
    mov x0, x20
    bl utoa
    mov w1, #32
    strb w1, [x24], #1
    mov x0, x21
    bl utoa
    mov w1, #10
    strb w1, [x24], #1
    sub x2, x24, x23
    mov x1, x23
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0

utoa:
    cbnz x0, utoa_nz
    mov w1, #48
    strb w1, [x24], #1
    ret
utoa_nz:
    sub sp, sp, #32
    add x1, sp, #32
    mov x2, x1
    mov x3, #10
utoa_div:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x4
    cbnz x0, utoa_div
utoa_copy:
    ldrb w5, [x2], #1
    strb w5, [x24], #1
    cmp x2, x1
    b.ne utoa_copy
    add sp, sp, #32
    ret

.bss
.align 3
inbuf:
    .zero 65536
outbuf:
    .zero 64
