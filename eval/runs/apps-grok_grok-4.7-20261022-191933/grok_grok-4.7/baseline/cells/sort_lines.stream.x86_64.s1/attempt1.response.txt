.globl _start

.section .rodata
nl_byte:
    .byte 10

.bss
.balign 8
input_buf:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000
indices:
    .zero 4000

.text
_start:
    xor rbx, rbx
read_loop:
    mov rdx, 32768
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + line_off]
    lea r13, [rip + line_len]
    lea r9, [rip + input_buf]
    xor ecx, ecx
    xor edx, edx
    xor r8d, r8d
parse_loop:
    cmp rcx, rbx
    jae parse_done
    movzx eax, byte ptr [r9 + rcx]
    cmp al, 10
    je found_nl
    inc rcx
    jmp parse_loop
found_nl:
    mov dword ptr [r12 + rdx*4], r8d
    mov eax, ecx
    sub eax, r8d
    mov dword ptr [r13 + rdx*4], eax
    inc edx
    inc rcx
    mov r8d, ecx
    jmp parse_loop
parse_done:
    cmp r8, rbx
    jae no_tail
    mov dword ptr [r12 + rdx*4], r8d
    mov eax, ebx
    sub eax, r8d
    mov dword ptr [r13 + rdx*4], eax
    inc edx
no_tail:
    test edx, edx
    jz do_exit
    mov r15d, edx
    lea r14, [rip + indices]
    xor ecx, ecx
init_loop:
    cmp ecx, r15d
    jae init_done
    mov dword ptr [r14 + rcx*4], ecx
    inc ecx
    jmp init_loop
init_done:
    mov eax, r15d
    shr eax, 1
    dec eax
    js heap_built
build_loop:
    mov edi, eax
    mov esi, r15d
    call sift_down
    dec eax
    jns build_loop
heap_built:
    mov ebx, r15d
    dec ebx
extract_loop:
    test ebx, ebx
    jle extract_done
    lea r14, [rip + indices]
    mov eax, dword ptr [r14]
    mov ecx, dword ptr [r14 + rbx*4]
    mov dword ptr [r14], ecx
    mov dword ptr [r14 + rbx*4], eax
    xor edi, edi
    mov esi, ebx
    call sift_down
    dec ebx
    jmp extract_loop
extract_done:
    xor ebx, ebx
out_loop:
    cmp ebx, r15d
    jae do_exit
    lea r14, [rip + indices]
    mov eax, dword ptr [r14 + rbx*4]
    lea r13, [rip + line_len]
    mov ecx, dword ptr [r13 + rax*4]
    lea r12, [rip + line_off]
    mov r8d, dword ptr [r12 + rax*4]
    test ecx, ecx
    jz write_nl
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + input_buf]
    mov edx, r8d
    add rsi, rdx
    mov edx, ecx
    syscall
write_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl_byte]
    mov edx, 1
    syscall
    inc ebx
    jmp out_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

sift_down:
    push rax
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12d, edi
    mov r13d, esi
sift_loop:
    mov r14d, r12d
    shl r14d, 1
    inc r14d
    cmp r14d, r13d
    jae sift_done
    mov r15d, r14d
    inc r15d
    cmp r15d, r13d
    jae sift_have_child
    lea rbx, [rip + indices]
    mov edi, dword ptr [rbx + r14*4]
    mov esi, dword ptr [rbx + r15*4]
    call compare_lines
    test eax, eax
    jns sift_have_child
    mov r14d, r15d
sift_have_child:
    lea rbx, [rip + indices]
    mov edi, dword ptr [rbx + r12*4]
    mov esi, dword ptr [rbx + r14*4]
    call compare_lines
    test eax, eax
    jns sift_done
    lea rbx, [rip + indices]
    mov eax, dword ptr [rbx + r12*4]
    mov ecx, dword ptr [rbx + r14*4]
    mov dword ptr [rbx + r12*4], ecx
    mov dword ptr [rbx + r14*4], eax
    mov r12d, r14d
    jmp sift_loop
sift_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rax
    ret

compare_lines:
    lea r8, [rip + line_off]
    lea r9, [rip + line_len]
    mov eax, dword ptr [r8 + rdi*4]
    mov ecx, dword ptr [r9 + rdi*4]
    mov edx, dword ptr [r8 + rsi*4]
    mov r11d, dword ptr [r9 + rsi*4]
    lea r8, [rip + input_buf]
    lea r9, [r8 + rax]
    lea r10, [r8 + rdx]
    mov r8d, ecx
    cmp r8d, r11d
    cmova r8d, r11d
    xor eax, eax
cmp_loop:
    cmp eax, r8d
    jae cmp_by_len
    movzx edx, byte ptr [r9 + rax]
    movzx esi, byte ptr [r10 + rax]
    cmp edx, esi
    jb cmp_less
    ja cmp_greater
    inc eax
    jmp cmp_loop
cmp_by_len:
    cmp ecx, r11d
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret
