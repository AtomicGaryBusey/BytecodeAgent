.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536

.text
.balign 4
.globl _start
_start:
    mov x19, #0
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf

read_loop:
    mov x8, #SYS_read
    mov x0, #0
    add x1, x20, x19
    mov x2, #IN_MAX
    sub x2, x2, x19
    svc #0
    cmp x0, #0
    b.lt read_loop
    b.eq read_done
    add x19, x19, x0
    cmp x19, #IN_MAX
    b.lo read_loop

read_done:
    mov x21, #0
    adrp x22, linebuf
    add x22, x22, :lo12:linebuf
    adrp x23, hextab
    add x23, x23, :lo12:hextab

row_loop:
    cmp x21, x19
    b.ge print_total
    sub x24, x19, x21
    cmp x24, #16
    b.ls have_n
    mov x24, #16
have_n:
    mov x0, #0
    mov w1, #32
clear_line:
    strb w1, [x22, x0]
    add x0, x0, #1
    cmp x0, #60
    b.lt clear_line

    mov x0, x21
    mov x1, x22
    bl format_hex8

    add x26, x20, x21
    mov x25, #0
hex_bytes:
    cmp x25, x24
    b.ge hex_bytes_done
    ldrb w0, [x26, x25]
    add x3, x25, x25, lsl #1
    add x3, x3, #10
    cmp x25, #8
    b.lt hex_pos_ok
    add x3, x3, #1
hex_pos_ok:
    lsr w4, w0, #4
    and w4, w4, #0xf
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x22, x3]
    and w4, w0, #0xf
    ldrb w4, [x23, w4, uxtw]
    add x3, x3, #1
    strb w4, [x22, x3]
    add x25, x25, #1
    b hex_bytes

hex_bytes_done:
    mov w0, #124
    strb w0, [x22, #60]
    mov x25, #0
ascii_loop:
    cmp x25, x24
    b.ge ascii_done
    ldrb w0, [x26, x25]
    cmp w0, #32
    b.lo ascii_dot
    cmp w0, #126
    b.ls ascii_store
ascii_dot:
    mov w0, #46
ascii_store:
    add x3, x22, #61
    strb w0, [x3, x25]
    add x25, x25, #1
    b ascii_loop

ascii_done:
    add x3, x22, x24
    add x3, x3, #61
    mov w0, #124
    strb w0, [x3]
    mov w0, #10
    strb w0, [x3, #1]

    mov x8, #SYS_write
    mov x0, #1
    mov x1, x22
    add x2, x24, #63
    svc #0

    add x21, x21, x24
    b row_loop

print_total:
    mov x0, x19
    mov x1, x22
    bl format_hex8
    mov w0, #10
    strb w0, [x22, #8]
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x22
    mov x2, #9
    svc #0

exit_now:
    mov x8, #SYS_exit
    mov x0, #0
    svc #0
    b exit_now

// x0 = value, x1 = destination. Writes 8 lowercase hex digits. x23 = hextab.
format_hex8:
    mov w2, #28
format_hex8_loop:
    lsr w3, w0, w2
    and w3, w3, #0xf
    ldrb w3, [x23, w3, uxtw]
    strb w3, [x1], #1
    subs w2, w2, #4
    b.pl format_hex8_loop
    ret

.section .rodata
.balign 16
hextab:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
