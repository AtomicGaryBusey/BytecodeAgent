.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536

.section .rodata
nl:
    .byte 0x0A

.bss
.align 3
inbuf:
    .zero 65536

.text
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, xzr

read_more:
    mov x2, #BUF_SIZE
    sub x2, x2, x19
    cbz x2, parse
    mov x0, xzr
    add x1, x20, x19
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le parse
    add x19, x19, x0
    b read_more

parse:
    mov x21, xzr

next_line:
    cmp x21, x19
    b.hs exit_ok
    mov x22, x21

scan:
    cmp x22, x19
    b.hs do_line
    ldrb w0, [x20, x22]
    cmp w0, #0x0A
    b.eq do_line
    add x22, x22, #1
    b scan

do_line:
    cmp x21, x22
    b.hs after_rev
    mov x23, x21
    sub x24, x22, #1

rev_loop:
    cmp x23, x24
    b.hs after_rev
    ldrb w0, [x20, x23]
    ldrb w1, [x20, x24]
    strb w1, [x20, x23]
    strb w0, [x20, x24]
    add x23, x23, #1
    sub x24, x24, #1
    b rev_loop

after_rev:
    sub x2, x22, x21
    cbz x2, write_nl
    mov x0, #1
    add x1, x20, x21
    mov x8, #SYS_WRITE
    svc #0

write_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x22, x19
    b.hs exit_ok
    add x21, x22, #1
    b next_line

exit_ok:
    mov x0, xzr
    mov x8, #SYS_EXIT
    svc #0
