.globl _start

SYS_read = 0
SYS_write = 1
SYS_exit = 60
BUF_CAP = 65536

.section .bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    lea r15, [rip + inbuf]
    mov r13, r15
read_loop:
    mov rax, r15
    add rax, BUF_CAP
    sub rax, r13
    test rax, rax
    jz count_begin
    mov rdx, rax
    mov eax, SYS_read
    xor edi, edi
    mov rsi, r13
    syscall
    test rax, rax
    jle count_begin
    add r13, rax
    jmp read_loop

count_begin:
    mov r11, r13
    sub r11, r15
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    mov rsi, r15
    mov rcx, r11
    test rcx, rcx
    jz do_format

count_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne check_ws
    inc r8
check_ws:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0D
    ja not_ws
is_ws:
    xor r10d, r10d
    jmp next_byte
not_ws:
    test r10d, r10d
    jnz next_byte
    inc r9
    mov r10d, 1
next_byte:
    dec rcx
    jnz count_loop

do_format:
    lea rdi, [rip + outbuf]
    mov rax, r8
    call utoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r9
    call utoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r11
    call utoa
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall
    mov eax, SYS_exit
    xor edi, edi
    syscall

utoa:
    test rax, rax
    jnz utoa_digits
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
utoa_digits:
    xor ecx, ecx
utoa_div:
    xor edx, edx
    mov r12, 10
    div r12
    add dl, 0x30
    push rdx
    inc ecx
    test rax, rax
    jnz utoa_div
utoa_pop:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz utoa_pop
    ret
