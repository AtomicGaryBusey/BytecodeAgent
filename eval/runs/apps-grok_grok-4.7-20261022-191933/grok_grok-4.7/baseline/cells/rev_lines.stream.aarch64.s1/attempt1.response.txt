.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_CAP, 65536

.globl _start

.section .bss
.balign 16
buf:
	.zero BUF_CAP

.section .rodata
nl:
	.byte 0x0A

.text
.align 2
_start:
	mov	x19, xzr
	adrp	x20, buf
	add	x20, x20, :lo12:buf

read_more:
	mov	x8, #SYS_read
	mov	x0, xzr
	add	x1, x20, x19
	mov	x2, #BUF_CAP
	sub	x2, x2, x19
	svc	#0
	cmp	x0, #0
	b.le	got_input
	add	x19, x19, x0
	cmp	x19, #BUF_CAP
	b.lo	read_more

got_input:
	mov	x21, xzr

next_line:
	cmp	x21, x19
	b.hs	quit
	mov	x22, x21

scan:
	cmp	x22, x19
	b.hs	have_line
	ldrb	w0, [x20, x22]
	cmp	w0, #0x0A
	b.eq	have_line
	add	x22, x22, #1
	b	scan

have_line:
	cmp	x21, x22
	b.hs	emit
	mov	x23, x21
	sub	x24, x22, #1

rev:
	cmp	x23, x24
	b.hs	emit
	ldrb	w0, [x20, x23]
	ldrb	w1, [x20, x24]
	strb	w1, [x20, x23]
	strb	w0, [x20, x24]
	add	x23, x23, #1
	sub	x24, x24, #1
	b	rev

emit:
	sub	x2, x22, x21
	cbz	x2, emit_nl
	mov	x8, #SYS_write
	mov	x0, #1
	add	x1, x20, x21
	svc	#0

emit_nl:
	mov	x8, #SYS_write
	mov	x0, #1
	adrp	x1, nl
	add	x1, x1, :lo12:nl
	mov	x2, #1
	svc	#0
	cmp	x22, x19
	b.hs	quit
	add	x21, x22, #1
	b	next_line

quit:
	mov	x8, #SYS_exit
	mov	x0, xzr
	svc	#0
	b	quit
