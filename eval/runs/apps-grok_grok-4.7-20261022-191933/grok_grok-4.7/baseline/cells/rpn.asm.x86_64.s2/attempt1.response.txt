.globl _start
.text
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
main_loop:
    cmp r13, r12
    jae do_exit
    xor r14, r14
    xor r15, r15
    lea r11, [rip + inbuf]
line_loop:
    cmp r13, r12
    jae emit
    movzx eax, byte ptr [r11 + r13]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    cmp al, 0x0a
    je got_nl
    mov r8, r13
scan_tok:
    inc r13
    cmp r13, r12
    jae have_tok
    movzx eax, byte ptr [r11 + r13]
    cmp al, 0x20
    je have_tok
    cmp al, 0x09
    je have_tok
    cmp al, 0x0a
    je have_tok
    jmp scan_tok
have_tok:
    mov r9, r13
    sub r9, r8
    lea rsi, [r11 + r8]
    cmp r9, 1
    jne parse_num
    movzx eax, byte ptr [rsi]
    cmp al, 0x2b
    je op_add
    cmp al, 0x2d
    je op_sub
    cmp al, 0x2a
    je op_mul
    cmp al, 0x2f
    je op_div
    cmp al, 0x25
    je op_mod
parse_num:
    xor r10d, r10d
    mov rcx, r9
    movzx eax, byte ptr [rsi]
    cmp al, 0x2d
    jne check_digits
    cmp r9, 1
    je bad_line
    inc rsi
    dec rcx
    mov r10d, 1
check_digits:
    cmp rcx, 1
    jb bad_line
    cmp rcx, 18
    ja bad_line
    xor rax, rax
    mov rdi, rsi
    add rdi, rcx
digit_loop:
    movzx edx, byte ptr [rsi]
    cmp dl, 0x30
    jb bad_line
    cmp dl, 0x39
    ja bad_line
    sub edx, 0x30
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    cmp rsi, rdi
    jb digit_loop
    test r10d, r10d
    jz push_lit
    neg rax
push_lit:
    cmp r14, 1024
    jae bad_line
    lea rdi, [rip + rpnstack]
    mov qword ptr [rdi + r14*8], rax
    inc r14
    jmp line_loop
skip_sep:
    inc r13
    jmp line_loop
got_nl:
    inc r13
    jmp emit
op_add:
    call pop_two
    test ecx, ecx
    jnz bad_line
    add rax, rbx
    jmp store_op
op_sub:
    call pop_two
    test ecx, ecx
    jnz bad_line
    sub rax, rbx
    jmp store_op
op_mul:
    call pop_two
    test ecx, ecx
    jnz bad_line
    imul rax, rbx
    jmp store_op
op_div:
    call pop_two
    test ecx, ecx
    jnz bad_line
    test rbx, rbx
    jz bad_line
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne div_ok
    cmp rbx, -1
    je bad_line
div_ok:
    cqo
    idiv rbx
    jmp store_op
op_mod:
    call pop_two
    test ecx, ecx
    jnz bad_line
    test rbx, rbx
    jz bad_line
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne mod_ok
    cmp rbx, -1
    je bad_line
mod_ok:
    cqo
    idiv rbx
    mov rax, rdx
    jmp store_op
store_op:
    lea rdi, [rip + rpnstack]
    mov qword ptr [rdi + r14*8], rax
    inc r14
    jmp line_loop
bad_line:
    mov r15, 1
skip_line:
    cmp r13, r12
    jae emit
    movzx eax, byte ptr [r11 + r13]
    inc r13
    cmp al, 0x0a
    jne skip_line
    jmp emit
emit:
    test r15, r15
    jnz emit_bad
    cmp r14, 1
    jne emit_bad
    lea rdi, [rip + rpnstack]
    mov rax, qword ptr [rdi]
    call print_int
    jmp main_loop
emit_bad:
    call print_err
    jmp main_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit
pop_two:
    xor ecx, ecx
    cmp r14, 2
    jb pop_two_fail
    lea rdi, [rip + rpnstack]
    dec r14
    mov rbx, qword ptr [rdi + r14*8]
    dec r14
    mov rax, qword ptr [rdi + r14*8]
    ret
pop_two_fail:
    mov ecx, 1
    ret
print_int:
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne pint_norm
    lea rsi, [rip + minstr]
    mov edx, 21
    jmp write_out
pint_norm:
    lea r8, [rip + outbuf]
    add r8, 39
    mov byte ptr [r8], 0x0a
    mov rdi, r8
    xor r9d, r9d
    test rax, rax
    jns pint_abs
    neg rax
    mov r9d, 1
pint_abs:
    mov ecx, 10
    test rax, rax
    jnz pint_loop
    dec rdi
    mov byte ptr [rdi], 0x30
    jmp pint_sign
pint_loop:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pint_loop
pint_sign:
    test r9d, r9d
    jz pint_ready
    dec rdi
    mov byte ptr [rdi], 0x2d
pint_ready:
    mov rsi, rdi
    mov rdx, r8
    sub rdx, rdi
    inc rdx
    jmp write_out
print_err:
    lea rsi, [rip + errstr]
    mov edx, 6
    jmp write_out
write_out:
    mov eax, 1
    mov edi, 1
    syscall
    ret
.section .rodata
minstr:
    .ascii "-9223372036854775808"
    .byte 0x0a
errstr:
    .ascii "error"
    .byte 0x0a
.bss
.align 8
inbuf:
    .zero 65536
.align 8
rpnstack:
    .zero 8192
.align 8
outbuf:
    .zero 40
