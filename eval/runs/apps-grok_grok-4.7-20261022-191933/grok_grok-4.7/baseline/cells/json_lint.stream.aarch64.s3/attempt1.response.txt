.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    mov x0, #0
    add x1, x19, x20
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    bl parse_ws
    bl parse_value
    bl parse_ws
    cmp x21, x20
    b.ne fail
    mov x8, #SYS_write
    mov x0, #1
    adrp x1, ok_str
    add x1, x1, :lo12:ok_str
    mov x2, #3
    svc #0
    mov x8, #SYS_exit
    mov x0, #0
    svc #0
    b hang

// x19 = buf, x20 = len, x21 = pos. On mismatch or early EOF, x21 is the error offset.
parse_ws:
pw_loop:
    cmp x21, x20
    b.hs pw_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq pw_skip
    cmp w0, #0x09
    b.eq pw_skip
    cmp w0, #0x0A
    b.eq pw_skip
    cmp w0, #0x0D
    b.eq pw_skip
    ret
pw_skip:
    add x21, x21, #1
    b pw_loop
pw_done:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x7B
    b.eq pv_obj
    cmp w0, #0x5B
    b.eq pv_arr
    cmp w0, #0x22
    b.eq pv_str
    cmp w0, #0x74
    b.eq pv_true
    cmp w0, #0x66
    b.eq pv_false
    cmp w0, #0x6E
    b.eq pv_null
    cmp w0, #0x2D
    b.eq pv_num
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.ls pv_num
    b fail
pv_obj:
    bl parse_object
    b pv_done
pv_arr:
    bl parse_array
    b pv_done
pv_str:
    bl parse_string
    b pv_done
pv_true:
    adrp x0, lit_true
    add x0, x0, :lo12:lit_true
    mov x1, #4
    bl parse_lit
    b pv_done
pv_false:
    adrp x0, lit_false
    add x0, x0, :lo12:lit_false
    mov x1, #5
    bl parse_lit
    b pv_done
pv_null:
    adrp x0, lit_null
    add x0, x0, :lo12:lit_null
    mov x1, #4
    bl parse_lit
    b pv_done
pv_num:
    bl parse_number
pv_done:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl parse_ws
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x7D
    b.eq po_close
po_loop:
    bl parse_string
    bl parse_ws
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x3A
    b.ne fail
    add x21, x21, #1
    bl parse_ws
    bl parse_value
    bl parse_ws
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.ne po_need_end
    add x21, x21, #1
    bl parse_ws
    b po_loop
po_need_end:
    cmp w0, #0x7D
    b.ne fail
po_close:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl parse_ws
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x5D
    b.eq pa_close
pa_loop:
    bl parse_value
    bl parse_ws
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.ne pa_need_end
    add x21, x21, #1
    bl parse_ws
    b pa_loop
pa_need_end:
    cmp w0, #0x5D
    b.ne fail
pa_close:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_string:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne fail
    add x21, x21, #1
ps_loop:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq ps_close
    cmp w0, #0x5C
    b.eq ps_esc
    cmp w0, #0x20
    b.lo fail
    add x21, x21, #1
    b ps_loop
ps_close:
    add x21, x21, #1
    ret
ps_esc:
    add x21, x21, #1
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq ps_esc_ok
    cmp w0, #0x5C
    b.eq ps_esc_ok
    cmp w0, #0x2F
    b.eq ps_esc_ok
    cmp w0, #0x62
    b.eq ps_esc_ok
    cmp w0, #0x66
    b.eq ps_esc_ok
    cmp w0, #0x6E
    b.eq ps_esc_ok
    cmp w0, #0x72
    b.eq ps_esc_ok
    cmp w0, #0x74
    b.eq ps_esc_ok
    cmp w0, #0x75
    b.eq ps_u
    b fail
ps_esc_ok:
    add x21, x21, #1
    b ps_loop
ps_u:
    add x21, x21, #1
    mov w1, #4
ps_hex:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.ls ps_hex_ok
    cmp w0, #0x41
    b.lo fail
    cmp w0, #0x46
    b.ls ps_hex_ok
    cmp w0, #0x61
    b.lo fail
    cmp w0, #0x66
    b.hi fail
ps_hex_ok:
    add x21, x21, #1
    subs w1, w1, #1
    b.ne ps_hex
    b ps_loop

parse_number:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x2D
    b.ne pn_int
    add x21, x21, #1
pn_int:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.eq pn_zero
    cmp w0, #0x31
    b.lo fail
    cmp w0, #0x39
    b.hi fail
    add x21, x21, #1
pn_digs:
    cmp x21, x20
    b.hs pn_frac
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo pn_frac
    cmp w0, #0x39
    b.hi pn_frac
    add x21, x21, #1
    b pn_digs
pn_zero:
    add x21, x21, #1
pn_frac:
    cmp x21, x20
    b.hs pn_ok
    ldrb w0, [x19, x21]
    cmp w0, #0x2E
    b.ne pn_exp
    add x21, x21, #1
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.hi fail
    add x21, x21, #1
pn_frac_digs:
    cmp x21, x20
    b.hs pn_ok
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo pn_exp
    cmp w0, #0x39
    b.hi pn_exp
    add x21, x21, #1
    b pn_frac_digs
pn_exp:
    cmp x21, x20
    b.hs pn_ok
    ldrb w0, [x19, x21]
    cmp w0, #0x65
    b.eq pn_exp_body
    cmp w0, #0x45
    b.ne pn_ok
pn_exp_body:
    add x21, x21, #1
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x2B
    b.eq pn_exp_sign
    cmp w0, #0x2D
    b.ne pn_exp_digit
pn_exp_sign:
    add x21, x21, #1
pn_exp_digit:
    cmp x21, x20
    b.hs fail
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo fail
    cmp w0, #0x39
    b.hi fail
    add x21, x21, #1
pn_exp_digs:
    cmp x21, x20
    b.hs pn_ok
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.lo pn_ok
    cmp w0, #0x39
    b.hi pn_ok
    add x21, x21, #1
    b pn_exp_digs
pn_ok:
    ret

// x0 = literal address, x1 = length
parse_lit:
    mov x2, #0
pl_loop:
    cmp x2, x1
    b.hs pl_ok
    cmp x21, x20
    b.hs fail
    ldrb w3, [x19, x21]
    ldrb w4, [x0, x2]
    cmp w3, w4
    b.ne fail
    add x21, x21, #1
    add x2, x2, #1
    b pl_loop
pl_ok:
    ret

fail:
    adrp x10, msgbuf
    add x10, x10, :lo12:msgbuf
    adrp x11, err_prefix
    add x11, x11, :lo12:err_prefix
    mov x12, #9
fp_copy:
    ldrb w13, [x11], #1
    strb w13, [x10], #1
    subs x12, x12, #1
    b.ne fp_copy
    adrp x14, digbuf
    add x14, x14, :lo12:digbuf
    mov x15, x14
    mov x0, x21
    mov x3, #10
fp_div:
    udiv x1, x0, x3
    msub x2, x1, x3, x0
    strb w2, [x15], #1
    mov x0, x1
    cbnz x0, fp_div
fp_rev:
    sub x15, x15, #1
    ldrb w2, [x15]
    add w2, w2, #0x30
    strb w2, [x10], #1
    cmp x15, x14
    b.ne fp_rev
    mov w2, #0x0A
    strb w2, [x10], #1
    adrp x1, msgbuf
    add x1, x1, :lo12:msgbuf
    sub x2, x10, x1
    mov x8, #SYS_write
    mov x0, #1
    svc #0
    mov x8, #SYS_exit
    mov x0, #1
    svc #0
hang:
    b hang

.section .rodata
.balign 4
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
ok_str:
    .ascii "ok"
    .byte 0x0A
err_prefix:
    .ascii "error at "

.bss
.balign 16
inbuf:
    .zero 65536
msgbuf:
    .zero 64
digbuf:
    .zero 32
