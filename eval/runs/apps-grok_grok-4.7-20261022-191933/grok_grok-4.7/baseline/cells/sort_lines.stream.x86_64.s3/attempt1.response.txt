.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_MAX, 32768

.section .rodata
nl_byte:
    .byte 10

.bss
.align 8
buffer:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov edx, BUF_MAX
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, BUF_MAX
    jb read_loop
read_done:
    xor r12d, r12d
    xor r8d, r8d
    xor esi, esi
    test rbx, rbx
    jz parsed
    lea r14, [rip + buffer]
    lea r15, [rip + line_off]
    lea r13, [rip + line_len]
parse_loop:
    cmp rsi, rbx
    jae parse_final
    movzx eax, byte ptr [r14 + rsi]
    cmp al, 10
    jne parse_next
    mov dword ptr [r15 + r12*4], r8d
    mov eax, esi
    sub eax, r8d
    mov dword ptr [r13 + r12*4], eax
    inc r12d
    mov r8d, esi
    inc r8d
parse_next:
    inc rsi
    jmp parse_loop
parse_final:
    cmp r8, rbx
    jae parsed
    mov dword ptr [r15 + r12*4], r8d
    mov eax, ebx
    sub eax, r8d
    mov dword ptr [r13 + r12*4], eax
    inc r12d
parsed:
    lea r14, [rip + line_off]
    lea r15, [rip + line_len]
    mov ebx, r12d
    cmp ebx, 2
    jb output
    mov edi, ebx
    shr edi, 1
    dec edi
build_loop:
    mov esi, ebx
    push rdi
    call sift_down
    pop rdi
    dec edi
    jns build_loop
    mov r13d, ebx
    dec r13d
extract_loop:
    test r13d, r13d
    jz output
    mov eax, dword ptr [r14]
    mov ecx, dword ptr [r14 + r13*4]
    mov dword ptr [r14], ecx
    mov dword ptr [r14 + r13*4], eax
    mov eax, dword ptr [r15]
    mov ecx, dword ptr [r15 + r13*4]
    mov dword ptr [r15], ecx
    mov dword ptr [r15 + r13*4], eax
    xor edi, edi
    mov esi, r13d
    call sift_down
    dec r13d
    jmp extract_loop
output:
    xor r12d, r12d
out_loop:
    cmp r12d, ebx
    jae do_exit
    mov eax, dword ptr [r14 + r12*4]
    mov edx, dword ptr [r15 + r12*4]
    lea rsi, [rip + buffer]
    add rsi, rax
    mov edi, 1
    test rdx, rdx
    jz out_nl
out_wr:
    mov eax, SYS_write
    syscall
    test rax, rax
    jle out_nl
    add rsi, rax
    sub rdx, rax
    jnz out_wr
out_nl:
    mov eax, SYS_write
    mov edi, 1
    lea rsi, [rip + nl_byte]
    mov edx, 1
    syscall
    inc r12d
    jmp out_loop
do_exit:
    mov eax, SYS_exit
    xor edi, edi
    syscall

sift_down:
    push rbx
    push rbp
    push r12
    push r13
    mov ebx, edi
    mov ebp, esi
sift_loop:
    mov r12d, ebx
    shl r12d, 1
    inc r12d
    cmp r12d, ebp
    jae sift_done
    mov r13d, ebx
    mov edi, dword ptr [r14 + r12*4]
    mov esi, dword ptr [r15 + r12*4]
    mov edx, dword ptr [r14 + r13*4]
    mov ecx, dword ptr [r15 + r13*4]
    call cmp_lines
    test eax, eax
    jle sift_skip_left
    mov r13d, r12d
sift_skip_left:
    mov eax, r12d
    inc eax
    cmp eax, ebp
    jae sift_check_swap
    mov r12d, eax
    mov edi, dword ptr [r14 + r12*4]
    mov esi, dword ptr [r15 + r12*4]
    mov edx, dword ptr [r14 + r13*4]
    mov ecx, dword ptr [r15 + r13*4]
    call cmp_lines
    test eax, eax
    jle sift_check_swap
    mov r13d, r12d
sift_check_swap:
    cmp r13d, ebx
    je sift_done
    mov eax, dword ptr [r14 + rbx*4]
    mov ecx, dword ptr [r14 + r13*4]
    mov dword ptr [r14 + rbx*4], ecx
    mov dword ptr [r14 + r13*4], eax
    mov eax, dword ptr [r15 + rbx*4]
    mov ecx, dword ptr [r15 + r13*4]
    mov dword ptr [r15 + rbx*4], ecx
    mov dword ptr [r15 + r13*4], eax
    mov ebx, r13d
    jmp sift_loop
sift_done:
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

cmp_lines:
    lea r8, [rip + buffer]
    lea r9, [rip + buffer]
    add r8, rdi
    add r9, rdx
    mov r10d, esi
    cmp r10d, ecx
    cmova r10d, ecx
    xor r11d, r11d
cmp_loop:
    cmp r11, r10
    jae cmp_prefix
    movzx eax, byte ptr [r8 + r11]
    movzx edx, byte ptr [r9 + r11]
    cmp eax, edx
    jb cmp_less
    ja cmp_greater
    inc r11
    jmp cmp_loop
cmp_prefix:
    cmp esi, ecx
    jb cmp_less
    ja cmp_greater
    xor eax, eax
    ret
cmp_less:
    mov eax, -1
    ret
cmp_greater:
    mov eax, 1
    ret
