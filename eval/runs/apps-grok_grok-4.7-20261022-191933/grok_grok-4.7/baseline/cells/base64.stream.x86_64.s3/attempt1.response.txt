.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_CAP, 49152
.equ COL_MAX, 76

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
inbuf:
    .zero 49152
outbuf:
    .zero 80000

.text
.globl _start
_start:
    lea r10, [rip + alpha]
    lea r15, [rip + inbuf]
    lea r9, [rip + outbuf]
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + rbx]
    mov edx, IN_CAP
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    test rbx, rbx
    jz do_exit
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
encode_loop:
    mov rax, rbx
    sub rax, r12
    cmp rax, 3
    jb tail
    lea rsi, [r15 + r12]
    movzx ecx, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx r8d, byte ptr [rsi + 2]
    mov eax, ecx
    shr eax, 2
    call emit_idx
    mov eax, ecx
    and eax, 3
    shl eax, 4
    mov esi, edx
    shr esi, 4
    or eax, esi
    call emit_idx
    mov eax, edx
    and eax, 15
    shl eax, 2
    mov esi, r8d
    shr esi, 6
    or eax, esi
    call emit_idx
    mov eax, r8d
    and eax, 63
    call emit_idx
    add r12, 3
    jmp encode_loop
tail:
    cmp rax, 1
    jb finish
    je tail1
    lea rsi, [r15 + r12]
    movzx ecx, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    mov eax, ecx
    shr eax, 2
    call emit_idx
    mov eax, ecx
    and eax, 3
    shl eax, 4
    mov esi, edx
    shr esi, 4
    or eax, esi
    call emit_idx
    mov eax, edx
    and eax, 15
    shl eax, 2
    call emit_idx
    mov al, 61
    call emit_char
    jmp finish
tail1:
    movzx ecx, byte ptr [r15 + r12]
    mov eax, ecx
    shr eax, 2
    call emit_idx
    mov eax, ecx
    and eax, 3
    shl eax, 4
    call emit_idx
    mov al, 61
    call emit_char
    mov al, 61
    call emit_char
finish:
    test r14, r14
    jz do_write
    mov byte ptr [r9 + r13], 10
    inc r13
do_write:
    xor r12d, r12d
write_loop:
    mov eax, SYS_write
    mov edi, 1
    lea rsi, [r9 + r12]
    mov rdx, r13
    sub rdx, r12
    syscall
    test rax, rax
    jle do_exit
    add r12, rax
    cmp r12, r13
    jb write_loop
do_exit:
    mov eax, SYS_exit
    xor edi, edi
    syscall
    jmp do_exit
emit_idx:
    movzx eax, byte ptr [r10 + rax]
emit_char:
    mov byte ptr [r9 + r13], al
    inc r13
    inc r14
    cmp r14, COL_MAX
    jne emit_done
    mov byte ptr [r9 + r13], 10
    inc r13
    xor r14d, r14d
emit_done:
    ret
