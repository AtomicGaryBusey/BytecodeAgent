.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUFSZ, 65536

.globl _start
.text

_start:
    xor r12, r12

read_loop:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, BUFSZ
    sub rdx, r12
    test rdx, rdx
    jz count_init
    syscall
    test rax, rax
    jle count_init
    add r12, rax
    jmp read_loop

count_init:
    xor ecx, ecx
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea r10, [rip + buf]

count_loop:
    cmp rcx, r12
    jae format_out
    movzx eax, byte ptr [r10 + rcx]
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0d
    ja is_word
is_ws:
    cmp al, 0x0a
    jne not_nl
    inc r13
not_nl:
    xor r15, r15
    jmp next_byte
is_word:
    test r15, r15
    jnz next_byte
    inc r14
    mov r15, 1
next_byte:
    inc rcx
    jmp count_loop

format_out:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call format_u64
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call format_u64
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r12
    call format_u64
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    mov eax, SYS_write
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jnz write_loop

do_exit:
    mov eax, SYS_exit
    xor edi, edi
    syscall

format_u64:
    mov rbx, rdi
    mov r8, 10
    xor ecx, ecx
digit_loop:
    xor edx, edx
    div r8
    add dl, 0x30
    push rdx
    inc ecx
    test rax, rax
    jnz digit_loop
pop_loop:
    pop rax
    mov byte ptr [rbx], al
    inc rbx
    dec ecx
    jnz pop_loop
    mov rdi, rbx
    ret

.bss
.balign 16
buf:
    .zero BUFSZ
outbuf:
    .zero 64
