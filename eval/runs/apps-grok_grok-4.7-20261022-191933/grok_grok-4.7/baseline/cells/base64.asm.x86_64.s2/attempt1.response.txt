.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_MAX, 49152
.equ LINE_W, 76

.section .rodata
b64tab:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
inbuf:
.zero 49152
outbuf:
.zero 131072

.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rdx, IN_MAX
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, IN_MAX
    jb read_loop
read_done:
    mov r14, rbx
    lea r13, [rip + inbuf]
    lea r12, [rip + outbuf]
    xor r15, r15
    lea rbx, [rip + b64tab]
encode_loop:
    cmp r14, 3
    jb remainder
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    movzx edx, byte ptr [r13 + 2]
    add r13, 3
    sub r14, 3
    mov esi, eax
    shr esi, 2
    movzx r8d, byte ptr [rbx + rsi]
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ecx
    shr edi, 4
    or esi, edi
    movzx r9d, byte ptr [rbx + rsi]
    mov esi, ecx
    and esi, 15
    shl esi, 2
    mov edi, edx
    shr edi, 6
    or esi, edi
    movzx r10d, byte ptr [rbx + rsi]
    and edx, 63
    movzx r11d, byte ptr [rbx + rdx]
    mov eax, r8d
    call do_emit
    mov eax, r9d
    call do_emit
    mov eax, r10d
    call do_emit
    mov eax, r11d
    call do_emit
    jmp encode_loop
remainder:
    cmp r14, 1
    je one_byte
    cmp r14, 2
    je two_bytes
    jmp finish
two_bytes:
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    mov esi, eax
    shr esi, 2
    movzx r8d, byte ptr [rbx + rsi]
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edx, ecx
    shr edx, 4
    or esi, edx
    movzx r9d, byte ptr [rbx + rsi]
    and ecx, 15
    shl ecx, 2
    movzx r10d, byte ptr [rbx + rcx]
    mov eax, r8d
    call do_emit
    mov eax, r9d
    call do_emit
    mov eax, r10d
    call do_emit
    mov eax, 0x3d
    call do_emit
    jmp finish
one_byte:
    movzx eax, byte ptr [r13]
    mov esi, eax
    shr esi, 2
    movzx r8d, byte ptr [rbx + rsi]
    and eax, 3
    shl eax, 4
    movzx r9d, byte ptr [rbx + rax]
    mov eax, r8d
    call do_emit
    mov eax, r9d
    call do_emit
    mov eax, 0x3d
    call do_emit
    mov eax, 0x3d
    call do_emit
finish:
    test r15, r15
    jz do_write
    mov byte ptr [r12], 10
    inc r12
do_write:
    lea rsi, [rip + outbuf]
    sub r12, rsi
    test r12, r12
    jz do_exit
    mov eax, SYS_write
    mov edi, 1
    mov rdx, r12
    syscall
do_exit:
    mov eax, SYS_exit
    xor edi, edi
    syscall
do_emit:
    mov byte ptr [r12], al
    inc r12
    inc r15
    cmp r15, LINE_W
    jne emit_ret
    mov byte ptr [r12], 10
    inc r12
    xor r15, r15
emit_ret:
    ret
