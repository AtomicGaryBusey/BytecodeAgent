.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_SIZE, 65536
.equ BUF_CAP, 65537

.globl _start

.bss
.balign 8
inbuf:
    .zero BUF_CAP

.text
_start:
    lea r14, [rip + inbuf]
    xor ebx, ebx

read_loop:
    mov eax, SYS_READ
    xor edi, edi
    lea rsi, [r14 + rbx]
    mov edx, BUF_SIZE
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    xor r12d, r12d

line_loop:
    cmp r12, rbx
    jae do_exit
    mov r13, r12

scan_loop:
    cmp r13, rbx
    jae have_line
    cmp byte ptr [r14 + r13], 0x0A
    je have_line
    inc r13
    jmp scan_loop

have_line:
    mov rdx, r13
    sub rdx, r12
    cmp rdx, 2
    jb write_line
    mov r8, r12
    lea r9, [r13 - 1]

rev_loop:
    mov al, byte ptr [r14 + r8]
    mov cl, byte ptr [r14 + r9]
    mov byte ptr [r14 + r8], cl
    mov byte ptr [r14 + r9], al
    inc r8
    dec r9
    cmp r8, r9
    jb rev_loop

write_line:
    mov byte ptr [r14 + r13], 0x0A
    lea rdx, [r13 + 1]
    sub rdx, r12
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [r14 + r12]
    syscall
    lea r12, [r13 + 1]
    jmp line_loop

do_exit:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
