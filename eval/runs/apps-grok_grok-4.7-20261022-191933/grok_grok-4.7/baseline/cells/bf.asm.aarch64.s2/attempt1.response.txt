.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536
.equ MAX_DEPTH, 1000

.text
.globl _start
.align 2

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, xzr

read_loop:
    mov x0, xzr
    add x1, x19, x20
    mov x2, #IN_MAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, xzr
    mov w2, #0x21

find_loop:
    cmp x21, x20
    b.eq no_bang
    ldrb w0, [x19, x21]
    cmp w0, w2
    b.eq found_bang
    add x21, x21, #1
    b find_loop

found_bang:
    add x24, x19, x20
    add x23, x19, x21
    add x23, x23, #1
    b brackets

no_bang:
    mov x21, x20
    add x23, x19, x20
    mov x24, x23

brackets:
    adrp x25, jumptab
    add x25, x25, :lo12:jumptab
    adrp x26, bstack
    add x26, x26, :lo12:bstack
    mov x27, xzr
    mov x28, xzr

match_loop:
    cmp x28, x21
    b.eq match_done
    ldrb w3, [x19, x28]
    cmp w3, #91
    b.eq do_push
    cmp w3, #93
    b.eq do_pop
    add x28, x28, #1
    b match_loop

do_push:
    cmp x27, #MAX_DEPTH
    b.hs bad
    str w28, [x26, x27, lsl #2]
    add x27, x27, #1
    add x28, x28, #1
    b match_loop

do_pop:
    cbz x27, bad
    sub x27, x27, #1
    ldr w3, [x26, x27, lsl #2]
    str w28, [x25, x3, lsl #2]
    str w3, [x25, x28, lsl #2]
    add x28, x28, #1
    b match_loop

match_done:
    cbnz x27, bad

    adrp x10, tape
    add x10, x10, :lo12:tape
    mov x11, xzr
    mov x12, xzr
    adrp x9, class_tab
    add x9, x9, :lo12:class_tab

    sub sp, sp, #80
    adr x0, h_other
    str x0, [sp, #0]
    adr x0, h_gt
    str x0, [sp, #8]
    adr x0, h_lt
    str x0, [sp, #16]
    adr x0, h_plus
    str x0, [sp, #24]
    adr x0, h_minus
    str x0, [sp, #32]
    adr x0, h_dot
    str x0, [sp, #40]
    adr x0, h_comma
    str x0, [sp, #48]
    adr x0, h_lb
    str x0, [sp, #56]
    adr x0, h_rb
    str x0, [sp, #64]
    mov x14, sp

exec:
    cmp x12, x21
    b.eq done_ok
    ldrb w0, [x19, x12]
    ldrb w1, [x9, x0]
    ldr x2, [x14, x1, lsl #3]
    br x2

h_other:
    add x12, x12, #1
    b exec

h_gt:
    add x11, x11, #1
    add x12, x12, #1
    b exec

h_lt:
    sub x11, x11, #1
    add x12, x12, #1
    b exec

h_plus:
    ldrb w0, [x10, x11]
    add w0, w0, #1
    strb w0, [x10, x11]
    add x12, x12, #1
    b exec

h_minus:
    ldrb w0, [x10, x11]
    sub w0, w0, #1
    strb w0, [x10, x11]
    add x12, x12, #1
    b exec

h_dot:
    mov x0, #1
    add x1, x10, x11
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x12, x12, #1
    b exec

h_comma:
    cmp x23, x24
    b.hs comma_eof
    ldrb w0, [x23], #1
    strb w0, [x10, x11]
    add x12, x12, #1
    b exec

comma_eof:
    strb wzr, [x10, x11]
    add x12, x12, #1
    b exec

h_lb:
    ldrb w0, [x10, x11]
    cbz w0, lb_jump
    add x12, x12, #1
    b exec

lb_jump:
    ldr w0, [x25, x12, lsl #2]
    add x12, x0, #1
    b exec

h_rb:
    ldrb w0, [x10, x11]
    cbnz w0, rb_jump
    add x12, x12, #1
    b exec

rb_jump:
    ldr w0, [x25, x12, lsl #2]
    add x12, x0, #1
    b exec

done_ok:
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0

bad:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

.section .rodata
.align 3
class_tab:
    .zero 43
    .byte 3, 6, 4, 5
    .zero 13
    .byte 2, 0, 1
    .zero 28
    .byte 7, 0, 8
    .zero 162

.bss
.align 3
inbuf:
    .zero 65536
tape:
    .zero 30000
jumptab:
    .zero 262144
bstack:
    .zero 4000
