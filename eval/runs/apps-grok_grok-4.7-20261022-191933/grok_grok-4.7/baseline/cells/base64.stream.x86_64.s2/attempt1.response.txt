.globl _start

.equ INMAX, 49152
.equ LINELEN, 76

.text

_start:
    xor r15, r15

read_loop:
    cmp r15, INMAX
    jae encode_start
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov rdx, INMAX
    sub rdx, r15
    syscall
    test rax, rax
    jle encode_start
    add r15, rax
    jmp read_loop

encode_start:
    test r15, r15
    jz do_exit
    xor rbx, rbx
    xor r14, r14
    lea r12, [rip + inbuf]
    lea rdi, [rip + outbuf]

main_loop:
    mov rax, r15
    sub rax, rbx
    cmp rax, 3
    jb do_tail

    movzx eax, byte ptr [r12 + rbx]
    movzx ecx, byte ptr [r12 + rbx + 1]
    movzx edx, byte ptr [r12 + rbx + 2]
    add rbx, 3
    mov r8d, eax
    mov r9d, ecx
    mov r13d, edx

    mov eax, r8d
    shr eax, 2
    call emit_idx

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    call emit_idx

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov ecx, r13d
    shr ecx, 6
    or eax, ecx
    call emit_idx

    mov eax, r13d
    and eax, 63
    call emit_idx
    jmp main_loop

do_tail:
    mov rax, r15
    sub rax, rbx
    cmp rax, 1
    je tail1
    cmp rax, 2
    je tail2
    jmp finish_nl

tail1:
    movzx r8d, byte ptr [r12 + rbx]
    mov eax, r8d
    shr eax, 2
    call emit_idx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    call emit_idx
    mov al, 0x3d
    call emit_raw
    mov al, 0x3d
    call emit_raw
    jmp finish_nl

tail2:
    movzx r8d, byte ptr [r12 + rbx]
    movzx r9d, byte ptr [r12 + rbx + 1]
    mov eax, r8d
    shr eax, 2
    call emit_idx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    call emit_idx
    mov eax, r9d
    and eax, 15
    shl eax, 2
    call emit_idx
    mov al, 0x3d
    call emit_raw

finish_nl:
    test r14, r14
    jz do_write
    mov byte ptr [rdi], 0x0A
    inc rdi

do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit_idx:
    lea r10, [rip + alphabet]
    movzx eax, byte ptr [r10 + rax]
emit_raw:
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    cmp r14, LINELEN
    jne emit_ret
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
emit_ret:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 8
inbuf:
    .zero 49152
outbuf:
    .zero 131072
