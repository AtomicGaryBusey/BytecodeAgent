.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.section .rodata
nl:
    .byte 0x0A

.bss
inbuf:
    .zero BUFSIZE

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    movz x21, #1, lsl #16
    adrp x24, nl
    add x24, x24, :lo12:nl

read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, x21
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, read_loop

read_done:
    mov x22, x19

line_loop:
    cmp x22, x20
    b.hs exit_ok
    mov x23, x22

scan_loop:
    cmp x23, x20
    b.hs final_line
    ldrb w0, [x23]
    cmp w0, #0x0A
    b.eq emit_line
    add x23, x23, #1
    b scan_loop

emit_line:
    mov x1, x22
    mov x2, x23
    bl reverse_range
    mov x0, #1
    mov x1, x22
    sub x2, x23, x22
    add x2, x2, #1
    mov x8, #SYS_write
    svc #0
    add x22, x23, #1
    b line_loop

final_line:
    mov x1, x22
    mov x2, x20
    bl reverse_range
    mov x0, #1
    mov x1, x22
    sub x2, x20, x22
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x1, x24
    mov x2, #1
    mov x8, #SYS_write
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// reverse the half-open byte range [x1, x2)
reverse_range:
    cmp x1, x2
    b.hs rev_done
    sub x2, x2, #1
rev_loop:
    cmp x1, x2
    b.hs rev_done
    ldrb w3, [x1]
    ldrb w4, [x2]
    strb w4, [x1]
    strb w3, [x2]
    add x1, x1, #1
    sub x2, x2, #1
    b rev_loop
rev_done:
    ret
