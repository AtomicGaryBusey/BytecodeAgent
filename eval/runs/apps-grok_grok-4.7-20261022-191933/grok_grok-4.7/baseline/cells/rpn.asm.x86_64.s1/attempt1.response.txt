.section .rodata
err_str:
    .ascii "error"
    .byte 10
err_len = . - err_str

.bss
.balign 8
input_buf:
    .zero 65536
.balign 8
valstack:
    .zero 8192
numbuf:
    .zero 32

.text
.globl _start
_start:
    lea r12, [rip + input_buf]
    lea rbp, [rip + valstack]
    xor r13, r13
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    cmp r13, 65536
    jb read_loop
read_done:
    xor r14, r14
line_loop:
    cmp r14, r13
    jae do_exit
    mov r15, r14
scan_line:
    cmp r15, r13
    jae eval_eof
    movzx eax, byte ptr [r12 + r15]
    cmp al, 10
    je eval_nl
    inc r15
    jmp scan_line
eval_nl:
    call eval_line
    lea r14, [r15 + 1]
    jmp line_loop
eval_eof:
    call eval_line
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

eval_line:
    xor ebx, ebx
tok_loop:
    cmp r14, r15
    jae eval_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 32
    je tok_skip
    cmp al, 9
    je tok_skip
    mov rcx, r14
tok_scan:
    cmp r14, r15
    jae got_token
    movzx eax, byte ptr [r12 + r14]
    cmp al, 32
    je got_token
    cmp al, 9
    je got_token
    inc r14
    jmp tok_scan
tok_skip:
    inc r14
    jmp tok_loop
got_token:
    mov rdx, r14
    sub rdx, rcx
    cmp rdx, 1
    jne try_number
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp try_number

try_number:
    xor r8d, r8d
    movzx eax, byte ptr [r12 + rcx]
    cmp al, '-'
    jne check_len
    mov r8d, 1
    inc rcx
    dec rdx
check_len:
    cmp rdx, 1
    jb eval_error
    cmp rdx, 18
    ja eval_error
    xor eax, eax
parse_digits:
    movzx r9d, byte ptr [r12 + rcx]
    cmp r9d, '0'
    jb eval_error
    cmp r9d, '9'
    ja eval_error
    sub r9d, '0'
    imul rax, rax, 10
    add rax, r9
    inc rcx
    dec rdx
    jnz parse_digits
    test r8d, r8d
    jz push_num
    neg rax
push_num:
    cmp rbx, 1024
    jae eval_error
    mov qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp tok_loop

op_add:
    cmp rbx, 2
    jb eval_error
    dec rbx
    mov rax, qword ptr [rbp + rbx*8]
    dec rbx
    add qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp tok_loop

op_sub:
    cmp rbx, 2
    jb eval_error
    dec rbx
    mov rax, qword ptr [rbp + rbx*8]
    dec rbx
    sub qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp tok_loop

op_mul:
    cmp rbx, 2
    jb eval_error
    dec rbx
    mov rax, qword ptr [rbp + rbx*8]
    dec rbx
    mov r10, qword ptr [rbp + rbx*8]
    imul rax, r10
    mov qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp tok_loop

op_div:
    cmp rbx, 2
    jb eval_error
    dec rbx
    mov rcx, qword ptr [rbp + rbx*8]
    dec rbx
    mov rax, qword ptr [rbp + rbx*8]
    test rcx, rcx
    jz eval_error
    mov r10, 1
    shl r10, 63
    cmp rax, r10
    jne op_div_do
    cmp rcx, -1
    je eval_error
op_div_do:
    cqo
    idiv rcx
    mov qword ptr [rbp + rbx*8], rax
    inc rbx
    jmp tok_loop

op_mod:
    cmp rbx, 2
    jb eval_error
    dec rbx
    mov rcx, qword ptr [rbp + rbx*8]
    dec rbx
    mov rax, qword ptr [rbp + rbx*8]
    test rcx, rcx
    jz eval_error
    mov r10, 1
    shl r10, 63
    cmp rax, r10
    jne op_mod_do
    cmp rcx, -1
    je eval_error
op_mod_do:
    cqo
    idiv rcx
    mov qword ptr [rbp + rbx*8], rdx
    inc rbx
    jmp tok_loop

eval_done:
    cmp rbx, 1
    jne eval_error
    mov rax, qword ptr [rbp]
    call print_i64
    ret

eval_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_str]
    mov edx, err_len
    syscall
    ret

print_i64:
    lea rsi, [rip + numbuf]
    mov r11, rsi
    add r11, 32
    add rsi, 31
    mov byte ptr [rsi], 10
    mov r8, rax
    test rax, rax
    jns print_pos
    neg rax
print_pos:
    mov ecx, 10
print_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz print_loop
    test r8, r8
    jns print_write
    dec rsi
    mov byte ptr [rsi], '-'
print_write:
    mov rdx, r11
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret
