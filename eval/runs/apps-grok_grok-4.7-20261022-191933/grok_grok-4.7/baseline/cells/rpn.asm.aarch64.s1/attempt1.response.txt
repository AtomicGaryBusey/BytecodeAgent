.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_MAX, 65536
.equ STACK_CAP, 1024

.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

line_loop:
    cmp x21, x20
    b.hs do_exit
    mov x22, x21

scan_line:
    cmp x21, x20
    b.hs line_eof
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq line_nl
    add x21, x21, #1
    b scan_line

line_nl:
    add x0, x19, x22
    add x1, x19, x21
    add x21, x21, #1
    bl eval_line
    b line_loop

line_eof:
    cmp x22, x21
    b.eq do_exit
    add x0, x19, x22
    add x1, x19, x21
    bl eval_line

do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b do_exit

// x0 = line start, x1 = line end (exclusive)
eval_line:
    stp x29, x30, [sp, #-96]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    stp x27, x28, [sp, #80]

    mov x19, x0
    mov x20, x1
    mov x21, #0
    adrp x22, rpn_stack
    add x22, x22, :lo12:rpn_stack

skip_sep:
    cmp x19, x20
    b.hs end_tokens
    ldrb w2, [x19]
    cmp w2, #32
    b.eq skip_one
    cmp w2, #9
    b.eq skip_one
    b tok_begin

skip_one:
    add x19, x19, #1
    b skip_sep

tok_begin:
    mov x23, x19

tok_scan:
    cmp x19, x20
    b.hs tok_got
    ldrb w2, [x19]
    cmp w2, #32
    b.eq tok_got
    cmp w2, #9
    b.eq tok_got
    add x19, x19, #1
    b tok_scan

tok_got:
    sub x24, x19, x23
    cmp x24, #1
    b.ne try_num
    ldrb w2, [x23]
    cmp w2, #'+'
    b.eq op_add
    cmp w2, #'-'
    b.eq op_sub
    cmp w2, #'*'
    b.eq op_mul
    cmp w2, #'/'
    b.eq op_div
    cmp w2, #'%'
    b.eq op_mod

try_num:
    mov x25, x23
    mov x26, #0
    ldrb w2, [x25]
    cmp w2, #'-'
    b.ne num_digits
    mov x26, #1
    add x25, x25, #1

num_digits:
    sub x27, x19, x25
    cmp x27, #1
    b.lo bad_line
    cmp x27, #18
    b.hi bad_line
    mov x0, #0
    mov x3, #10

num_loop:
    cmp x25, x19
    b.hs num_done
    ldrb w2, [x25]
    sub w2, w2, #'0'
    cmp w2, #9
    b.hi bad_line
    mul x0, x0, x3
    add x0, x0, x2
    add x25, x25, #1
    b num_loop

num_done:
    cbz x26, push_num
    neg x0, x0

push_num:
    cmp x21, #STACK_CAP
    b.hs bad_line
    str x0, [x22, x21, lsl #3]
    add x21, x21, #1
    b skip_sep

op_add:
    cmp x21, #2
    b.lo bad_line
    sub x21, x21, #1
    ldr x1, [x22, x21, lsl #3]
    sub x4, x21, #1
    ldr x0, [x22, x4, lsl #3]
    add x0, x0, x1
    str x0, [x22, x4, lsl #3]
    b skip_sep

op_sub:
    cmp x21, #2
    b.lo bad_line
    sub x21, x21, #1
    ldr x1, [x22, x21, lsl #3]
    sub x4, x21, #1
    ldr x0, [x22, x4, lsl #3]
    sub x0, x0, x1
    str x0, [x22, x4, lsl #3]
    b skip_sep

op_mul:
    cmp x21, #2
    b.lo bad_line
    sub x21, x21, #1
    ldr x1, [x22, x21, lsl #3]
    sub x4, x21, #1
    ldr x0, [x22, x4, lsl #3]
    mul x0, x0, x1
    str x0, [x22, x4, lsl #3]
    b skip_sep

op_div:
    cmp x21, #2
    b.lo bad_line
    sub x21, x21, #1
    ldr x1, [x22, x21, lsl #3]
    sub x4, x21, #1
    ldr x0, [x22, x4, lsl #3]
    cbz x1, bad_line
    movz x5, #0x8000, lsl #48
    cmp x0, x5
    b.ne div_do
    cmn x1, #1
    b.eq bad_line
div_do:
    sdiv x0, x0, x1
    str x0, [x22, x4, lsl #3]
    b skip_sep

op_mod:
    cmp x21, #2
    b.lo bad_line
    sub x21, x21, #1
    ldr x1, [x22, x21, lsl #3]
    sub x4, x21, #1
    ldr x0, [x22, x4, lsl #3]
    cbz x1, bad_line
    movz x5, #0x8000, lsl #48
    cmp x0, x5
    b.ne mod_do
    cmn x1, #1
    b.eq bad_line
mod_do:
    sdiv x5, x0, x1
    mul x5, x5, x1
    sub x0, x0, x5
    str x0, [x22, x4, lsl #3]
    b skip_sep

end_tokens:
    cmp x21, #1
    b.ne bad_line
    ldr x0, [x22]
    bl print_i64
    b eval_ret

bad_line:
    bl print_err

eval_ret:
    ldp x27, x28, [sp, #80]
    ldp x25, x26, [sp, #64]
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #96
    ret

print_err:
    adrp x1, err_str
    add x1, x1, :lo12:err_str
    mov x0, #1
    mov x2, #6
    mov x8, #SYS_write
    svc #0
    ret

print_i64:
    adrp x3, numbuf
    add x3, x3, :lo12:numbuf
    add x4, x3, #31
    mov w5, #10
    strb w5, [x4]
    mov x6, #10
    mov x2, #0
    cmp x0, #0
    b.ge pi_loop
    mov x2, #1
    movz x1, #0x8000, lsl #48
    cmp x0, x1
    b.eq pi_loop
    neg x0, x0

pi_loop:
    udiv x5, x0, x6
    msub x7, x5, x6, x0
    add w7, w7, #'0'
    sub x4, x4, #1
    strb w7, [x4]
    mov x0, x5
    cbnz x0, pi_loop
    cbz x2, pi_write
    sub x4, x4, #1
    mov w5, #'-'
    strb w5, [x4]

pi_write:
    adrp x5, numbuf
    add x5, x5, :lo12:numbuf
    add x5, x5, #32
    sub x2, x5, x4
    mov x1, x4
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    ret

.section .rodata
err_str:
    .ascii "error\n"

.bss
.balign 8
input_buf:
    .zero 65536
.balign 8
rpn_stack:
    .zero 8192
numbuf:
    .zero 32
