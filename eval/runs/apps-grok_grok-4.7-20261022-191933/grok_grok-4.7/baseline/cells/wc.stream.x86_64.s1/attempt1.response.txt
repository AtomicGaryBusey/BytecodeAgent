.globl _start

.bss
.balign 16
buffer:
    .zero 65536
outbuf:
    .zero 64
numtmp:
    .zero 32

.text
_start:
    xor rbx, rbx
read_loop:
    cmp rbx, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12, r12
    xor r13, r13
    xor r14d, r14d
    mov rcx, rbx
    test rcx, rcx
    jz emit
    lea rsi, [rip + buffer]
scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne classify
    inc r12
classify:
    cmp al, 0x20
    je saw_ws
    cmp al, 0x09
    jb saw_word
    cmp al, 0x0D
    jbe saw_ws
saw_word:
    test r14d, r14d
    jnz scan_next
    inc r13
    mov r14d, 1
    jmp scan_next
saw_ws:
    xor r14d, r14d
scan_next:
    dec rcx
    jnz scan_loop
emit:
    lea r15, [rip + outbuf]
    mov rdi, r15
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, rbx
    call utoa
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov rdx, rdi
    sub rdx, r15
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    syscall
    mov eax, 60
    xor edi, edi
    syscall

utoa:
    lea r8, [rip + numtmp + 24]
    mov r9, r8
    mov r10, 10
utoa_div:
    xor edx, edx
    div r10
    add dl, 0x30
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz utoa_div
utoa_copy:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    cmp r9, r8
    jne utoa_copy
    ret
