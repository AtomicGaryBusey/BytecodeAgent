.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14, r14
line_loop:
    cmp r14, r13
    jae exit_ok
    mov r15, r14
find_nl:
    cmp r15, r13
    jae no_nl
    cmp byte ptr [r12 + r15], 10
    je has_nl
    inc r15
    jmp find_nl
has_nl:
    lea rax, [r15 + 1]
    mov qword ptr [rip + nextpos], rax
    jmp eval_line
no_nl:
    mov qword ptr [rip + nextpos], r13
eval_line:
    xor ebx, ebx
token_loop:
    cmp r14, r15
    jae end_tokens
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    jmp parse_tok
skip_sep:
    inc r14
    jmp token_loop
end_tokens:
    cmp ebx, 1
    jne line_error
    jmp print_num
parse_tok:
    cmp al, 0x2b
    je op_add
    cmp al, 0x2a
    je op_mul
    cmp al, 0x2f
    je op_div
    cmp al, 0x25
    je op_mod
    cmp al, 0x2d
    je op_minus
    cmp al, 0x30
    jb line_error
    cmp al, 0x39
    ja line_error
    xor eax, eax
    xor ecx, ecx
    xor r8d, r8d
    jmp digit_loop
op_add:
    xor r8d, r8d
    jmp check_single_op
op_mul:
    mov r8d, 2
    jmp check_single_op
op_div:
    mov r8d, 3
    jmp check_single_op
op_mod:
    mov r8d, 4
    jmp check_single_op
op_minus:
    lea rcx, [r14 + 1]
    cmp rcx, r15
    jae minus_op
    movzx edx, byte ptr [r12 + rcx]
    cmp dl, 0x20
    je minus_op
    cmp dl, 0x09
    je minus_op
    cmp dl, 0x30
    jb line_error
    cmp dl, 0x39
    ja line_error
    inc r14
    xor eax, eax
    xor ecx, ecx
    mov r8d, 1
    jmp digit_loop
minus_op:
    mov r8d, 1
    jmp check_single_op
check_single_op:
    lea rcx, [r14 + 1]
    cmp rcx, r15
    jae exec_op
    movzx edx, byte ptr [r12 + rcx]
    cmp dl, 0x20
    je exec_op
    cmp dl, 0x09
    je exec_op
    jmp line_error
exec_op:
    mov r14, rcx
    cmp ebx, 2
    jb line_error
    lea rsi, [rip + stack]
    dec ebx
    mov r10, qword ptr [rsi + rbx*8]
    dec ebx
    mov rax, qword ptr [rsi + rbx*8]
    cmp r8d, 0
    je do_add
    cmp r8d, 1
    je do_sub
    cmp r8d, 2
    je do_mul
    cmp r8d, 3
    je do_div
do_mod:
    test r10, r10
    jz line_error
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    jne do_mod_ok
    cmp r10, -1
    je line_error
do_mod_ok:
    cqo
    idiv r10
    mov rax, rdx
    jmp push_val
do_div:
    test r10, r10
    jz line_error
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    jne do_div_ok
    cmp r10, -1
    je line_error
do_div_ok:
    cqo
    idiv r10
    jmp push_val
do_add:
    add rax, r10
    jmp push_val
do_sub:
    sub rax, r10
    jmp push_val
do_mul:
    imul rax, r10
    jmp push_val
digit_loop:
    cmp r14, r15
    jae finish_num
    movzx edx, byte ptr [r12 + r14]
    cmp dl, 0x20
    je finish_num
    cmp dl, 0x09
    je finish_num
    cmp dl, 0x30
    jb line_error
    cmp dl, 0x39
    ja line_error
    inc ecx
    cmp ecx, 18
    ja line_error
    imul rax, rax, 10
    sub edx, 0x30
    add rax, rdx
    inc r14
    jmp digit_loop
finish_num:
    test ecx, ecx
    jz line_error
    test r8d, r8d
    jz store_num
    neg rax
store_num:
    lea rsi, [rip + stack]
    jmp push_val
push_val:
    cmp ebx, 512
    jae line_error
    mov qword ptr [rsi + rbx*8], rax
    inc ebx
    jmp token_loop
print_num:
    mov rax, qword ptr [rip + stack]
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    je print_min
    lea r9, [rip + numbuf]
    add r9, 31
    mov byte ptr [r9], 10
    xor r8d, r8d
    test rax, rax
    jns conv_start
    neg rax
    mov r8d, 1
conv_start:
    mov ecx, 10
conv_loop:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz conv_loop
    test r8d, r8d
    jz write_num
    dec r9
    mov byte ptr [r9], 0x2d
write_num:
    lea rdx, [rip + numbuf]
    add rdx, 32
    sub rdx, r9
    mov rsi, r9
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line
print_min:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + minmsg]
    mov edx, 21
    syscall
    jmp next_line
line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 6
    syscall
next_line:
    mov r14, qword ptr [rip + nextpos]
    jmp line_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_ok
.section .rodata
errmsg:
    .ascii "error"
    .byte 10
minmsg:
    .ascii "-9223372036854775808"
    .byte 10
.bss
.balign 8
nextpos:
    .zero 8
inbuf:
    .zero 65536
.balign 8
stack:
    .zero 4096
numbuf:
    .zero 32
