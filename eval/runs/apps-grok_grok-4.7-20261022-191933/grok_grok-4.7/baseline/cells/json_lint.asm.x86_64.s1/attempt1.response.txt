.globl _start

.text

_start:
    xor r14d, r14d
    lea r15, [rip + buffer]
read_loop:
    mov edx, 65536
    sub rdx, r14
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r14]
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jne do_error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
exit0:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit0

do_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov rax, r13
    lea r12, [rip + numbuf + 32]
    mov rsi, r12
    mov ebx, 10
    test rax, rax
    jnz de_digits
    dec rsi
    mov byte ptr [rsi], 48
    jmp de_write
de_digits:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz de_digits
de_write:
    mov byte ptr [r12], 10
    mov rdx, r12
    sub rdx, rsi
    inc rdx
    mov eax, 1
    mov edi, 1
    syscall
exit1:
    mov eax, 60
    mov edi, 1
    syscall
    jmp exit1

skip_ws:
sw_loop:
    cmp r13, r14
    jae sw_ret
    movzx eax, byte ptr [r15 + r13]
    cmp al, 32
    je sw_inc
    cmp al, 9
    je sw_inc
    cmp al, 10
    je sw_inc
    cmp al, 13
    je sw_inc
sw_ret:
    ret
sw_inc:
    inc r13
    jmp sw_loop

parse_value:
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 116
    je parse_true
    cmp al, 102
    je parse_false
    cmp al, 110
    je parse_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb do_error
    cmp al, 57
    jbe parse_number
    jmp do_error

parse_true:
    lea rsi, [rip + s_true]
    mov ecx, 4
    jmp expect_lit
parse_false:
    lea rsi, [rip + s_false]
    mov ecx, 5
    jmp expect_lit
parse_null:
    lea rsi, [rip + s_null]
    mov ecx, 4
    jmp expect_lit

expect_lit:
el_loop:
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, byte ptr [rsi]
    jne do_error
    inc r13
    inc rsi
    dec ecx
    jnz el_loop
    ret

parse_string:
    cmp r13, r14
    jae do_error
    cmp byte ptr [r15 + r13], 34
    jne do_error
    inc r13
ps_loop:
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 34
    je ps_end
    cmp al, 92
    je ps_esc
    cmp al, 32
    jb do_error
    inc r13
    jmp ps_loop
ps_end:
    inc r13
    ret
ps_esc:
    inc r13
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 34
    je ps_esc_ok
    cmp al, 92
    je ps_esc_ok
    cmp al, 47
    je ps_esc_ok
    cmp al, 98
    je ps_esc_ok
    cmp al, 102
    je ps_esc_ok
    cmp al, 110
    je ps_esc_ok
    cmp al, 114
    je ps_esc_ok
    cmp al, 116
    je ps_esc_ok
    cmp al, 117
    je ps_u
    jmp do_error
ps_esc_ok:
    inc r13
    jmp ps_loop
ps_u:
    inc r13
    mov ecx, 4
ps_hex:
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 48
    jb do_error
    cmp al, 57
    jbe ps_hex_ok
    cmp al, 65
    jb do_error
    cmp al, 70
    jbe ps_hex_ok
    cmp al, 97
    jb do_error
    cmp al, 102
    ja do_error
ps_hex_ok:
    inc r13
    dec ecx
    jnz ps_hex
    jmp ps_loop

parse_number:
    cmp r13, r14
    jae do_error
    cmp byte ptr [r15 + r13], 45
    jne pn_int
    inc r13
pn_int:
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 48
    je pn_zero
    cmp al, 49
    jb do_error
    cmp al, 57
    ja do_error
    inc r13
pn_digs:
    cmp r13, r14
    jae pn_ret
    movzx eax, byte ptr [r15 + r13]
    cmp al, 48
    jb pn_after
    cmp al, 57
    ja pn_after
    inc r13
    jmp pn_digs
pn_zero:
    inc r13
    cmp r13, r14
    jae pn_ret
    movzx eax, byte ptr [r15 + r13]
pn_after:
    cmp al, 46
    je pn_frac
    cmp al, 101
    je pn_exp
    cmp al, 69
    je pn_exp
pn_ret:
    ret
pn_frac:
    inc r13
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 48
    jb do_error
    cmp al, 57
    ja do_error
    inc r13
pn_frac_digs:
    cmp r13, r14
    jae pn_ret
    movzx eax, byte ptr [r15 + r13]
    cmp al, 48
    jb pn_after_frac
    cmp al, 57
    ja pn_after_frac
    inc r13
    jmp pn_frac_digs
pn_after_frac:
    cmp al, 101
    je pn_exp
    cmp al, 69
    je pn_exp
    ret
pn_exp:
    inc r13
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 43
    je pn_sign
    cmp al, 45
    jne pn_expdig
pn_sign:
    inc r13
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
pn_expdig:
    cmp al, 48
    jb do_error
    cmp al, 57
    ja do_error
    inc r13
pn_exp_digs:
    cmp r13, r14
    jae pn_ret
    movzx eax, byte ptr [r15 + r13]
    cmp al, 48
    jb pn_ret
    cmp al, 57
    ja pn_ret
    inc r13
    jmp pn_exp_digs

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae do_error
    cmp byte ptr [r15 + r13], 93
    jne pa_val
    inc r13
    ret
pa_val:
    call parse_value
    call skip_ws
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 44
    je pa_comma
    cmp al, 93
    jne do_error
    inc r13
    ret
pa_comma:
    inc r13
    call skip_ws
    jmp pa_val

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae do_error
    cmp byte ptr [r15 + r13], 125
    jne po_mem
    inc r13
    ret
po_mem:
    call parse_string
    call skip_ws
    cmp r13, r14
    jae do_error
    cmp byte ptr [r15 + r13], 58
    jne do_error
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jae do_error
    movzx eax, byte ptr [r15 + r13]
    cmp al, 44
    je po_comma
    cmp al, 125
    jne do_error
    inc r13
    ret
po_comma:
    inc r13
    call skip_ws
    jmp po_mem

.section .rodata
msg_ok:
    .ascii "ok"
    .byte 10
msg_err:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"

.bss
.balign 16
buffer:
    .zero 65536
numbuf:
    .zero 48
