.text
.globl _start

_start:
  lea r12, [rip + inbuf]
  xor ebx, ebx

read_loop:
  xor eax, eax
  xor edi, edi
  lea rsi, [r12 + rbx]
  mov edx, 65536
  sub rdx, rbx
  test rdx, rdx
  jz read_done
  syscall
  cmp rax, 0
  jle read_done
  add rbx, rax
  jmp read_loop

read_done:
  xor ecx, ecx

find_loop:
  cmp rcx, rbx
  jae no_bang
  cmp byte ptr [r12 + rcx], 0x21
  je has_bang
  inc rcx
  jmp find_loop

has_bang:
  mov r13, rcx
  lea r14, [r12 + rcx + 1]
  mov r15, rbx
  sub r15, rcx
  dec r15
  jmp compile

no_bang:
  mov r13, rbx
  xor r14d, r14d
  xor r15d, r15d

compile:
  lea r8, [rip + dtable]
  mov byte ptr [r8 + 0x2b], 1
  mov byte ptr [r8 + 0x2c], 2
  mov byte ptr [r8 + 0x2d], 3
  mov byte ptr [r8 + 0x2e], 4
  mov byte ptr [r8 + 0x3c], 5
  mov byte ptr [r8 + 0x3e], 6
  mov byte ptr [r8 + 0x5b], 7
  mov byte ptr [r8 + 0x5d], 8
  xor esi, esi
  xor edi, edi
  xor edx, edx
  lea r10, [rip + brstack]
  lea r9, [rip + jumps]

comp_loop:
  cmp rsi, r13
  jae comp_done
  movzx eax, byte ptr [r12 + rsi]
  inc rsi
  movzx ecx, byte ptr [r8 + rax]
  test ecx, ecx
  jz comp_loop
  cmp cl, 7
  je comp_open
  cmp cl, 8
  je comp_close
  mov byte ptr [r12 + rdi], al
  inc rdi
  jmp comp_loop

comp_open:
  cmp rdx, 1000
  jae exit1
  mov byte ptr [r12 + rdi], al
  mov qword ptr [r10 + rdx*8], rdi
  inc rdx
  inc rdi
  jmp comp_loop

comp_close:
  test rdx, rdx
  jz exit1
  dec rdx
  mov byte ptr [r12 + rdi], al
  mov rcx, qword ptr [r10 + rdx*8]
  mov dword ptr [r9 + rcx*4], edi
  mov dword ptr [r9 + rdi*4], ecx
  inc rdi
  jmp comp_loop

comp_done:
  test rdx, rdx
  jnz exit1
  mov r13, rdi
  lea r10, [rip + jtable]
  lea rax, [rip + do_next]
  mov qword ptr [r10], rax
  lea rax, [rip + do_plus]
  mov qword ptr [r10 + 8], rax
  lea rax, [rip + do_comma]
  mov qword ptr [r10 + 16], rax
  lea rax, [rip + do_minus]
  mov qword ptr [r10 + 24], rax
  lea rax, [rip + do_dot]
  mov qword ptr [r10 + 32], rax
  lea rax, [rip + do_left]
  mov qword ptr [r10 + 40], rax
  lea rax, [rip + do_right]
  mov qword ptr [r10 + 48], rax
  lea rax, [rip + do_lbr]
  mov qword ptr [r10 + 56], rax
  lea rax, [rip + do_rbr]
  mov qword ptr [r10 + 64], rax
  xor ebp, ebp
  lea rbx, [rip + tape]

exec_loop:
  cmp rbp, r13
  jae exit0
  movzx eax, byte ptr [r12 + rbp]
  movzx eax, byte ptr [r8 + rax]
  jmp qword ptr [r10 + rax*8]

do_right:
  inc rbx
  inc rbp
  jmp exec_loop

do_left:
  dec rbx
  inc rbp
  jmp exec_loop

do_plus:
  inc byte ptr [rbx]
  inc rbp
  jmp exec_loop

do_minus:
  dec byte ptr [rbx]
  inc rbp
  jmp exec_loop

do_dot:
  mov eax, 1
  mov edi, 1
  mov rsi, rbx
  mov edx, 1
  syscall
  inc rbp
  jmp exec_loop

do_comma:
  test r15, r15
  jz comma_eof
  mov al, byte ptr [r14]
  mov byte ptr [rbx], al
  inc r14
  dec r15
  inc rbp
  jmp exec_loop

comma_eof:
  mov byte ptr [rbx], 0
  inc rbp
  jmp exec_loop

do_lbr:
  cmp byte ptr [rbx], 0
  jne do_next
  mov eax, dword ptr [r9 + rbp*4]
  lea rbp, [rax + 1]
  jmp exec_loop

do_rbr:
  cmp byte ptr [rbx], 0
  je do_next
  mov eax, dword ptr [r9 + rbp*4]
  lea rbp, [rax + 1]
  jmp exec_loop

do_next:
  inc rbp
  jmp exec_loop

exit0:
  mov eax, 60
  xor edi, edi
  syscall

exit1:
  mov eax, 60
  mov edi, 1
  syscall

.bss
.align 8
inbuf:
  .zero 65536
tape:
  .zero 30000
jumps:
  .zero 262144
brstack:
  .zero 8000
dtable:
  .zero 256
jtable:
  .zero 72
