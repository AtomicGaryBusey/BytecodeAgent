.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_CAP, 65536
.equ MAX_NEST, 1000

.globl _start

.text

_start:
    xor ebx, ebx
    lea r12, [rip + inbuf]

read_loop:
    mov eax, BUF_CAP
    sub rax, rbx
    jz read_done
    mov rdx, rax
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    xor ecx, ecx

find_loop:
    cmp rcx, rbx
    je no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_loop

found_bang:
    mov r13, rcx
    lea r8, [r12 + rcx + 1]
    mov r14, rbx
    sub r14, rcx
    dec r14
    jmp scan_brackets

no_bang:
    mov r13, rbx
    xor r14d, r14d
    xor r8d, r8d

scan_brackets:
    xor r9d, r9d
    xor ecx, ecx
    lea r10, [rip + brstack]
    lea r11, [rip + jumps]

scan_loop:
    cmp rcx, r13
    je scan_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5b
    je scan_open
    cmp al, 0x5d
    je scan_close
    inc rcx
    jmp scan_loop

scan_open:
    cmp r9, MAX_NEST
    jae exit_fail
    mov dword ptr [r10 + r9*4], ecx
    inc r9
    inc rcx
    jmp scan_loop

scan_close:
    test r9, r9
    jz exit_fail
    dec r9
    mov eax, dword ptr [r10 + r9*4]
    mov dword ptr [r11 + rcx*4], eax
    mov dword ptr [r11 + rax*4], ecx
    inc rcx
    jmp scan_loop

scan_done:
    test r9, r9
    jnz exit_fail

    lea r9, [rip + disp]
    lea rax, [rip + op_nop]
    xor ecx, ecx

disp_init:
    mov qword ptr [r9 + rcx*8], rax
    inc ecx
    cmp ecx, 256
    jb disp_init

    lea rax, [rip + op_plus]
    mov ecx, 0x2b
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_comma]
    mov ecx, 0x2c
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_minus]
    mov ecx, 0x2d
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_dot]
    mov ecx, 0x2e
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_left]
    mov ecx, 0x3c
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_right]
    mov ecx, 0x3e
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_lbrack]
    mov ecx, 0x5b
    mov qword ptr [r9 + rcx*8], rax

    lea rax, [rip + op_rbrack]
    mov ecx, 0x5d
    mov qword ptr [r9 + rcx*8], rax

    lea r10, [rip + jumps]
    lea rbx, [rip + tape]
    xor r15d, r15d

exec_loop:
    cmp r15, r13
    jae exit_ok
    movzx eax, byte ptr [r12 + r15]
    jmp qword ptr [r9 + rax*8]

op_nop:
    inc r15
    jmp exec_loop

op_plus:
    inc byte ptr [rbx]
    inc r15
    jmp exec_loop

op_minus:
    dec byte ptr [rbx]
    inc r15
    jmp exec_loop

op_right:
    inc rbx
    inc r15
    jmp exec_loop

op_left:
    dec rbx
    inc r15
    jmp exec_loop

op_lbrack:
    cmp byte ptr [rbx], 0
    jne op_nop
    mov eax, dword ptr [r10 + r15*4]
    lea r15, [rax + 1]
    jmp exec_loop

op_rbrack:
    cmp byte ptr [rbx], 0
    je op_nop
    mov eax, dword ptr [r10 + r15*4]
    lea r15, [rax + 1]
    jmp exec_loop

op_dot:
    mov eax, SYS_write
    mov edi, 1
    mov rsi, rbx
    mov edx, 1
    syscall
    inc r15
    jmp exec_loop

op_comma:
    test r14, r14
    jz op_comma_zero
    mov al, byte ptr [r8]
    mov byte ptr [rbx], al
    inc r8
    dec r14
    inc r15
    jmp exec_loop

op_comma_zero:
    mov byte ptr [rbx], 0
    inc r15
    jmp exec_loop

exit_fail:
    mov eax, SYS_exit
    mov edi, 1
    syscall

exit_ok:
    mov eax, SYS_exit
    xor edi, edi
    syscall

.bss
.align 8
inbuf:
    .zero 65536
tape:
    .zero 30000
.align 8
brstack:
    .zero 4000
jumps:
    .zero 262144
disp:
    .zero 2048
