.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ IN_CAP, 65536

.section .bss
.balign 8
inbuf:
    .zero 65536
outbuf:
    .zero 128

.text
.globl _start

_start:
    mov x19, xzr
    mov x7, #IN_CAP
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf

read_loop:
    mov x8, #SYS_READ
    mov x0, xzr
    add x1, x22, x19
    sub x2, x7, x19
    svc #0
    cbz x0, read_done
    cmp x0, #0
    b.lt read_done
    add x19, x19, x0
    cmp x19, x7
    b.lo read_loop

read_done:
    mov x20, xzr

row_loop:
    cmp x20, x19
    b.hs final_line
    sub x21, x19, x20
    cmp x21, #16
    b.ls row_have_n
    mov x21, #16
row_have_n:
    bl emit_row
    add x20, x20, x21
    b row_loop

final_line:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x1, x23
    mov x0, x19
    bl hex8
    mov w4, #10
    strb w4, [x1], #1
    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    mov x0, xzr
    mov x8, #SYS_EXIT
    svc #0
die:
    b die

// x20 = row offset, x21 = row length (1..16), x22 = input base
emit_row:
    stp x29, x30, [sp, #-16]!
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x1, x23
    mov x0, x20
    bl hex8
    mov w4, #32
    strb w4, [x1], #1
    strb w4, [x1], #1
    mov x25, xzr
    add x26, x22, x20

emit_hex_loop:
    cmp x25, x21
    b.hs emit_hex_pad
    ldrb w0, [x26, x25]
    bl hex2
    mov w4, #32
    strb w4, [x1], #1
    b emit_hex_next
emit_hex_pad:
    mov w4, #32
    strb w4, [x1], #1
    strb w4, [x1], #1
    strb w4, [x1], #1
emit_hex_next:
    add x25, x25, #1
    cmp x25, #8
    b.ne emit_hex_no_mid
    mov w4, #32
    strb w4, [x1], #1
emit_hex_no_mid:
    cmp x25, #16
    b.lo emit_hex_loop

    mov w4, #32
    strb w4, [x1], #1
    mov w4, #124
    strb w4, [x1], #1

    mov x25, xzr
emit_asc_loop:
    cmp x25, x21
    b.hs emit_asc_done
    ldrb w0, [x26, x25]
    cmp w0, #32
    b.lo emit_asc_dot
    cmp w0, #126
    b.ls emit_asc_store
emit_asc_dot:
    mov w0, #46
emit_asc_store:
    strb w0, [x1], #1
    add x25, x25, #1
    b emit_asc_loop
emit_asc_done:
    mov w4, #124
    strb w4, [x1], #1
    mov w4, #10
    strb w4, [x1], #1

    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

    ldp x29, x30, [sp], #16
    ret

// write 8 lowercase hex digits of x0 at [x1], advance x1
hex8:
    mov x3, #28
hex8_loop:
    lsr x4, x0, x3
    and x4, x4, #15
    cmp x4, #10
    add x6, x4, #48
    add x5, x4, #87
    csel x4, x6, x5, lo
    strb w4, [x1], #1
    subs x3, x3, #4
    b.pl hex8_loop
    ret

// write 2 lowercase hex digits of the low byte of x0 at [x1], advance x1
hex2:
    lsr x4, x0, #4
    and x4, x4, #15
    cmp x4, #10
    add x6, x4, #48
    add x5, x4, #87
    csel x4, x6, x5, lo
    strb w4, [x1], #1
    and x4, x0, #15
    cmp x4, #10
    add x6, x4, #48
    add x5, x4, #87
    csel x4, x6, x5, lo
    strb w4, [x1], #1
    ret
