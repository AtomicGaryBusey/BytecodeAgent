.globl _start

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    xor r12, r12

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz count
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle count
    add r12, rax
    jmp read_loop

count:
    xor r13, r13
    xor r14, r14
    xor ebx, ebx
    lea rsi, [rip + inbuf]
    mov rcx, r12
    test rcx, rcx
    jz format

scan:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne not_nl
    inc r13
not_nl:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb is_word
    cmp al, 13
    jbe is_ws
is_word:
    test ebx, ebx
    jnz next_byte
    inc r14
    mov ebx, 1
    jmp next_byte
is_ws:
    xor ebx, ebx
next_byte:
    inc rsi
    dec rcx
    jnz scan

format:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

utoa:
    test rax, rax
    jnz utoa_nz
    mov byte ptr [rdi], 48
    inc rdi
    ret
utoa_nz:
    xor ecx, ecx
    mov r9, 10
utoa_div:
    xor edx, edx
    div r9
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz utoa_div
utoa_pop:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz utoa_pop
    ret
