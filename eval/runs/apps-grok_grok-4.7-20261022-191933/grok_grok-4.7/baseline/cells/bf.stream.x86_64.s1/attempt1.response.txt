.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INPUT_MAX, 65536
.equ MAX_DEPTH, 1000

.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d

read_loop:
    mov edx, INPUT_MAX
    sub rdx, r12
    jz read_done
    mov eax, SYS_READ
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor ecx, ecx

find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rbx + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov r13, rcx
    lea r14, [rbx + rcx + 1]
    mov r15, r12
    sub r15, rcx
    dec r15
    jmp validate

no_bang:
    mov r13, r12
    xor r14d, r14d
    xor r15d, r15d

validate:
    lea r12, [rip + prog]
    lea r8, [rip + jump_table]
    lea r11, [rip + bracket_stack]
    xor ecx, ecx
    xor edi, edi
    xor r9d, r9d

filter_loop:
    cmp rcx, r13
    jae filter_done
    movzx eax, byte ptr [rbx + rcx]
    inc rcx
    cmp al, 0x3e
    je store_cmd
    cmp al, 0x3c
    je store_cmd
    cmp al, 0x2b
    je store_cmd
    cmp al, 0x2d
    je store_cmd
    cmp al, 0x2e
    je store_cmd
    cmp al, 0x2c
    je store_cmd
    cmp al, 0x5b
    je store_open
    cmp al, 0x5d
    je store_close
    jmp filter_loop

store_cmd:
    mov byte ptr [r12 + rdi], al
    inc rdi
    jmp filter_loop

store_open:
    cmp r9, MAX_DEPTH
    jae unbalanced
    mov byte ptr [r12 + rdi], al
    mov qword ptr [r11 + r9*8], rdi
    inc r9
    inc rdi
    jmp filter_loop

store_close:
    test r9, r9
    jz unbalanced
    dec r9
    mov byte ptr [r12 + rdi], al
    mov rdx, qword ptr [r11 + r9*8]
    mov dword ptr [r8 + rdx*4], edi
    mov dword ptr [r8 + rdi*4], edx
    inc rdi
    jmp filter_loop

filter_done:
    test r9, r9
    jnz unbalanced
    mov r13, rdi
    mov rbx, r12

    lea r12, [rip + dispatch]
    lea rax, [rip + cmd_skip]
    xor ecx, ecx

fill_disp:
    mov qword ptr [r12 + rcx*8], rax
    inc ecx
    cmp ecx, 256
    jb fill_disp

    lea rax, [rip + cmd_gt]
    mov ecx, 0x3e
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_lt]
    mov ecx, 0x3c
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_plus]
    mov ecx, 0x2b
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_minus]
    mov ecx, 0x2d
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_dot]
    mov ecx, 0x2e
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_comma]
    mov ecx, 0x2c
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_lbrack]
    mov ecx, 0x5b
    mov qword ptr [r12 + rcx*8], rax
    lea rax, [rip + cmd_rbrack]
    mov ecx, 0x5d
    mov qword ptr [r12 + rcx*8], rax

    lea r10, [rip + tape]
    xor r9d, r9d

exec_loop:
    cmp r9, r13
    jae done_ok
    movzx eax, byte ptr [rbx + r9]
    jmp qword ptr [r12 + rax*8]

cmd_skip:
    inc r9
    jmp exec_loop

cmd_gt:
    inc r10
    inc r9
    jmp exec_loop

cmd_lt:
    dec r10
    inc r9
    jmp exec_loop

cmd_plus:
    inc byte ptr [r10]
    inc r9
    jmp exec_loop

cmd_minus:
    dec byte ptr [r10]
    inc r9
    jmp exec_loop

cmd_dot:
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r10
    mov edx, 1
    syscall
    inc r9
    jmp exec_loop

cmd_comma:
    test r15, r15
    jz comma_eof
    mov al, byte ptr [r14]
    mov byte ptr [r10], al
    inc r14
    dec r15
    inc r9
    jmp exec_loop

comma_eof:
    mov byte ptr [r10], 0
    inc r9
    jmp exec_loop

cmd_lbrack:
    cmp byte ptr [r10], 0
    jne lbrack_nz
    mov eax, dword ptr [r8 + r9*4]
    lea r9, [rax + 1]
    jmp exec_loop

lbrack_nz:
    inc r9
    jmp exec_loop

cmd_rbrack:
    cmp byte ptr [r10], 0
    je rbrack_z
    mov eax, dword ptr [r8 + r9*4]
    lea r9, [rax + 1]
    jmp exec_loop

rbrack_z:
    inc r9
    jmp exec_loop

unbalanced:
    mov edi, 1
    jmp do_exit

done_ok:
    xor edi, edi

do_exit:
    mov eax, SYS_EXIT
    syscall
    jmp do_exit

.bss
.balign 8
input_buf:
    .zero 65536
prog:
    .zero 65536
tape:
    .zero 30000
.balign 8
jump_table:
    .zero 262144
bracket_stack:
    .zero 8000
dispatch:
    .zero 2048
