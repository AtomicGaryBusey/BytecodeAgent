.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 96

.text
.globl _start

_start:
    xor rbx, rbx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_loop
read_done:
    lea r13, [rip + inbuf]
    lea r15, [rip + hexdigits]
    xor r12, r12
row_loop:
    cmp r12, rbx
    jae final_offset
    mov r14, rbx
    sub r14, r12
    cmp r14, 16
    jbe have_n
    mov r14, 16
have_n:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2
    lea rsi, [r13 + r12]
    xor ecx, ecx
slots:
    cmp rcx, r14
    jae slot_pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp slot_next
slot_pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
slot_next:
    cmp ecx, 7
    jne slot_no_gap
    mov byte ptr [rdi], 0x20
    inc rdi
slot_no_gap:
    inc ecx
    cmp ecx, 16
    jl slots
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi
    xor ecx, ecx
ascii_loop:
    cmp rcx, r14
    jae ascii_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_store
ascii_dot:
    mov byte ptr [rdi], 0x2e
ascii_store:
    inc rdi
    inc ecx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    add r12, r14
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp row_loop
final_offset:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

put_hex8:
    mov ecx, 8
hex_digit:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz hex_digit
    ret
