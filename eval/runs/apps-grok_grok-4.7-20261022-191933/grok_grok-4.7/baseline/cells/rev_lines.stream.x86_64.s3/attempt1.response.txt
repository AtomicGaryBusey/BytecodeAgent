.globl _start

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537

.text
_start:
    xor r12, r12

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop

read_done:
    lea rbx, [rip + inbuf]
    xor r13, r13
    xor r14, r14

scan_loop:
    cmp r14, r12
    jae end_input
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x0A
    je found_nl
    inc r14
    jmp scan_loop

found_nl:
    call reverse_write
    inc r14
    mov r13, r14
    jmp scan_loop

end_input:
    cmp r13, r12
    jae do_exit
    mov r14, r12
    call reverse_write

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

reverse_write:
    mov rcx, r14
    sub rcx, r13
    lea rdi, [rip + outbuf]
    lea rsi, [rbx + r14]
    dec rsi
    test rcx, rcx
    jz store_nl

rev_copy:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    dec rcx
    jnz rev_copy

store_nl:
    mov byte ptr [rdi], 0x0A
    mov rdx, r14
    sub rdx, r13
    inc rdx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    syscall
    ret
