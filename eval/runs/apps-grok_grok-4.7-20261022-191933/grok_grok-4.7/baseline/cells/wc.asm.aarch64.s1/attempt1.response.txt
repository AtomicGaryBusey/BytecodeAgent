.globl _start
.text
_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr

read_loop:
    mov x8, #63
    mov x0, xzr
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #4096
    svc #0
    cmp x0, #0
    b.le print_counts
    mov x9, x0
    add x21, x21, x9
    adrp x10, inbuf
    add x10, x10, :lo12:inbuf
    mov x11, xzr

scan_loop:
    cmp x11, x9
    b.eq read_loop
    ldrb w12, [x10, x11]
    add x11, x11, #1
    cmp w12, #10
    b.ne check_ws
    add x19, x19, #1
check_ws:
    cmp w12, #32
    b.eq got_ws
    sub w13, w12, #9
    cmp w13, #4
    b.ls got_ws
    cbnz x22, scan_loop
    mov x22, #1
    add x20, x20, #1
    b scan_loop
got_ws:
    mov x22, xzr
    b scan_loop

print_counts:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x0, x19
    bl u64_to_dec
    mov w2, #' '
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x20
    bl u64_to_dec
    mov w2, #' '
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x21
    bl u64_to_dec
    mov w2, #10
    strb w2, [x0], #1
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x0, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, xzr
    mov x8, #93
    svc #0

// x0 = value, x1 = destination; returns x0 = pointer after the digits
u64_to_dec:
    cbnz x0, u64_nz
    mov w2, #'0'
    strb w2, [x1], #1
    mov x0, x1
    ret
u64_nz:
    sub sp, sp, #32
    mov x3, xzr
    mov x4, #10
u64_div:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #'0'
    strb w6, [sp, x3]
    add x3, x3, #1
    mov x0, x5
    cbnz x0, u64_div
u64_rev:
    sub x3, x3, #1
    ldrb w6, [sp, x3]
    strb w6, [x1], #1
    cbnz x3, u64_rev
    add sp, sp, #32
    mov x0, x1
    ret

.bss
.balign 16
inbuf:
    .zero 4096
outbuf:
    .zero 64
