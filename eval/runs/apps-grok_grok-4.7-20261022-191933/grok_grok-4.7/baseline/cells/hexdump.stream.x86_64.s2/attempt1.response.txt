.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_MAX, 65536

.section .rodata
hexdigits:
	.ascii "0123456789abcdef"

.bss
inbuf:
	.zero 65536
linebuf:
	.zero 128

.text
	.globl _start
_start:
	xor r12, r12
read_more:
	mov eax, SYS_read
	xor edi, edi
	lea rsi, [rip + inbuf]
	add rsi, r12
	mov edx, IN_MAX
	sub rdx, r12
	syscall
	test rax, rax
	jle read_done
	add r12, rax
	cmp r12, IN_MAX
	jb read_more
read_done:
	xor r13, r13
dump_loop:
	cmp r13, r12
	jae dump_final
	mov r15, r12
	sub r15, r13
	cmp r15, 16
	jbe row_n_ok
	mov r15, 16
row_n_ok:
	lea r14, [rip + inbuf]
	add r14, r13
	call format_row
	add r13, r15
	jmp dump_loop
dump_final:
	lea r10, [rip + hexdigits]
	lea rdi, [rip + linebuf]
	mov rax, r12
	mov ecx, 8
final_hex:
	mov rbx, rax
	shr rbx, 28
	and ebx, 15
	movzx ebx, byte ptr [r10 + rbx]
	mov byte ptr [rdi], bl
	inc rdi
	shl rax, 4
	dec ecx
	jnz final_hex
	mov byte ptr [rdi], 0x0a
	lea rsi, [rip + linebuf]
	mov edx, 9
	mov eax, SYS_write
	mov edi, 1
	syscall
	mov eax, SYS_exit
	xor edi, edi
	syscall

format_row:
	lea r10, [rip + hexdigits]
	lea rdi, [rip + linebuf]
	mov rax, r13
	mov ecx, 8
hex_off:
	mov rbx, rax
	shr rbx, 28
	and ebx, 15
	movzx ebx, byte ptr [r10 + rbx]
	mov byte ptr [rdi], bl
	inc rdi
	shl rax, 4
	dec ecx
	jnz hex_off
	mov byte ptr [rdi], 0x20
	inc rdi
	mov byte ptr [rdi], 0x20
	inc rdi
	xor ecx, ecx
slot_loop:
	cmp ecx, r15d
	jae slot_pad
	movzx eax, byte ptr [r14 + rcx]
	mov ebx, eax
	shr eax, 4
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [rdi], al
	inc rdi
	and ebx, 15
	movzx eax, byte ptr [r10 + rbx]
	mov byte ptr [rdi], al
	inc rdi
	mov byte ptr [rdi], 0x20
	inc rdi
	jmp slot_after
slot_pad:
	mov byte ptr [rdi], 0x20
	inc rdi
	mov byte ptr [rdi], 0x20
	inc rdi
	mov byte ptr [rdi], 0x20
	inc rdi
slot_after:
	cmp ecx, 7
	jne no_mid_space
	mov byte ptr [rdi], 0x20
	inc rdi
no_mid_space:
	inc ecx
	cmp ecx, 16
	jb slot_loop
	mov byte ptr [rdi], 0x20
	inc rdi
	mov byte ptr [rdi], 0x7c
	inc rdi
	xor ecx, ecx
asc_loop:
	cmp ecx, r15d
	jae asc_done
	movzx eax, byte ptr [r14 + rcx]
	cmp al, 0x20
	jb asc_dot
	cmp al, 0x7e
	ja asc_dot
	jmp asc_store
asc_dot:
	mov al, 0x2e
asc_store:
	mov byte ptr [rdi], al
	inc rdi
	inc ecx
	jmp asc_loop
asc_done:
	mov byte ptr [rdi], 0x7c
	inc rdi
	mov byte ptr [rdi], 0x0a
	inc rdi
	lea rsi, [rip + linebuf]
	mov rdx, rdi
	sub rdx, rsi
	mov eax, SYS_write
	mov edi, 1
	syscall
	ret
