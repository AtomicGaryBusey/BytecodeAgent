.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.globl _start
.text

    mov x4, #0

    b.eq no_bang
    ldrb w0, [x19, x4]
    cmp w0, #33
    b.eq has_bang
    add x4, x4, #1
    b find_bang
has_bang:

    add x21, x19, x4
    add x21, x21, #1
    add x22, x19, x3

    add x28, x28, :lo12:code
    adrp x26, jump
    add x26, x26, :lo12:jump
    adrp x27, brstack
    add x27, x27, :lo12:brstack

    mov x24, #0
    mov x29, #0

    b.eq compile_done
    ldrb w0, [x19, x23]
    add x23, x23, #1
    cmp w0, #62
    b.eq c_gt
    cmp w0, #60
    b.eq c_lt
    cmp w0, #43
    b.eq c_plus
    cmp w0, #45
    b.eq c_minus
    cmp w0, #46
    b.eq c_dot
    cmp w0, #44
    b.eq c_comma
    cmp w0, #91
    b.eq c_lbr
    cmp w0, #93
    b.eq c_rbr
    b compile
c_gt:
    mov w0, #0
    b c_emit
c_lt:
    mov w0, #1
    b c_emit
c_plus:
    mov w0, #2
    b c_emit
c_minus:
    mov w0, #3
    b c_emit
c_dot:
    mov w0, #4
    b c_emit

    b c_emit

    add x24, x24, #1
    mov w0, #6
    b c_emit
c_rbr:

    sub x24, x24, #1
    ldr x1, [x27, x24, lsl #3]
    mov w0, #7
    strb w0, [x28, x29]
    str w29, [x26, x1, lsl #2]
    str w1, [x26, x29, lsl #2]
    add x29, x29, #1
    b compile
c_emit:

    add x29, x29, #1
    b compile
compile_done:

    mov x20, x29
    adrp x25, tape
    add x25, x25, :lo12:tape

    b.hs exit_ok
    ldrb w0, [x28, x23]
    adr x1, dtbl
    add x1, x1, x0, lsl #2

    b do_lt
    b do_plus
    b do_minus
    b do_dot
    b do_comma
    b do_lbr
    b do_rbr
do_gt:
    add x25, x25, #1
    add x23, x23, #1
    b exec
do_lt:
    sub x25, x25, #1
    add x23, x23, #1
    b exec
do_plus:
    ldrb w0, [x25]
    add w0, w0, #1
    strb w0, [x25]
    add x23, x23, #1
    b exec
do_minus:
    ldrb w0, [x25]
    sub w0, w0, #1
    strb w0, [x25]
    add x23, x23, #1
    b exec
do_dot:

    mov x1, x25
    mov x2, #1
    mov x8, #SYS_write
    svc #0

    b exec

    cbz w0, rbr_next
