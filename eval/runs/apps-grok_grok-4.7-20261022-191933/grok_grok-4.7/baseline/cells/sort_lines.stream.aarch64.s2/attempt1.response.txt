.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 32768
.equ NEWLINE, 10

.globl _start

.text

_start:
	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	adrp x23, line_ptr
	add x23, x23, :lo12:line_ptr
	adrp x24, line_len
	add x24, x24, :lo12:line_len
	mov x20, #0
	mov x21, #0

read_loop:
	mov x2, #BUF_SIZE
	sub x2, x2, x20
	cbz x2, parse_start
	mov x0, #0
	add x1, x19, x20
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	b.le parse_start
	add x20, x20, x0
	b read_loop

parse_start:
	mov x22, #0

parse_loop:
	cmp x22, x20
	b.hs sort_start
	add x27, x19, x22
	mov x28, #0

scan_loop:
	cmp x22, x20
	b.hs store_line_final
	ldrb w2, [x19, x22]
	add x22, x22, #1
	cmp w2, #NEWLINE
	b.eq store_line_nl
	add x28, x28, #1
	b scan_loop

store_line_nl:
	str x27, [x23, x21, lsl #3]
	str x28, [x24, x21, lsl #3]
	add x21, x21, #1
	b parse_loop

store_line_final:
	str x27, [x23, x21, lsl #3]
	str x28, [x24, x21, lsl #3]
	add x21, x21, #1

sort_start:
	cmp x21, #1
	b.ls print_lines
	adrp x25, temp_ptr
	add x25, x25, :lo12:temp_ptr
	adrp x26, temp_len
	add x26, x26, :lo12:temp_len
	mov x10, #1

width_loop:
	cmp x10, x21
	b.hs print_lines
	mov x11, #0

i_loop:
	cmp x11, x21
	b.hs width_done
	add x12, x11, x10
	cmp x12, x21
	csel x12, x21, x12, hi
	cmp x12, x21
	b.hs i_advance
	add x13, x11, x10, lsl #1
	cmp x13, x21
	csel x13, x21, x13, hi
	mov x14, x11
	mov x15, x12
	mov x27, x11

merge_loop:
	cmp x14, x12
	b.hs take_rest_q
	cmp x15, x13
	b.hs take_rest_p
	ldr x0, [x23, x14, lsl #3]
	ldr x1, [x24, x14, lsl #3]
	ldr x2, [x23, x15, lsl #3]
	ldr x3, [x24, x15, lsl #3]
	bl compare_gt
	cbnz x0, take_q

take_p:
	ldr x0, [x23, x14, lsl #3]
	ldr x1, [x24, x14, lsl #3]
	str x0, [x25, x27, lsl #3]
	str x1, [x26, x27, lsl #3]
	add x14, x14, #1
	add x27, x27, #1
	b merge_loop

take_q:
	ldr x0, [x23, x15, lsl #3]
	ldr x1, [x24, x15, lsl #3]
	str x0, [x25, x27, lsl #3]
	str x1, [x26, x27, lsl #3]
	add x15, x15, #1
	add x27, x27, #1
	b merge_loop

take_rest_p:
	cmp x14, x12
	b.hs copy_back
	ldr x0, [x23, x14, lsl #3]
	ldr x1, [x24, x14, lsl #3]
	str x0, [x25, x27, lsl #3]
	str x1, [x26, x27, lsl #3]
	add x14, x14, #1
	add x27, x27, #1
	b take_rest_p

take_rest_q:
	cmp x15, x13
	b.hs copy_back
	ldr x0, [x23, x15, lsl #3]
	ldr x1, [x24, x15, lsl #3]
	str x0, [x25, x27, lsl #3]
	str x1, [x26, x27, lsl #3]
	add x15, x15, #1
	add x27, x27, #1
	b take_rest_q

copy_back:
	mov x14, x11

copy_loop:
	cmp x14, x13
	b.hs i_advance
	ldr x0, [x25, x14, lsl #3]
	ldr x1, [x26, x14, lsl #3]
	str x0, [x23, x14, lsl #3]
	str x1, [x24, x14, lsl #3]
	add x14, x14, #1
	b copy_loop

i_advance:
	add x11, x11, x10, lsl #1
	b i_loop

width_done:
	lsl x10, x10, #1
	b width_loop

print_lines:
	mov x25, #0

print_loop:
	cmp x25, x21
	b.hs do_exit
	ldr x0, [x23, x25, lsl #3]
	ldr x1, [x24, x25, lsl #3]
	cbz x1, print_nl
	bl do_write

print_nl:
	adrp x0, nl_byte
	add x0, x0, :lo12:nl_byte
	mov x1, #1
	bl do_write
	add x25, x25, #1
	b print_loop

do_exit:
	mov x0, #0
	mov x8, #SYS_exit
	svc #0
	b do_exit

do_write:
	mov x2, x1
	mov x1, x0
	mov x0, #1
	mov x8, #SYS_write
	svc #0
	ret

compare_gt:
	cmp x1, x3
	csel x4, x3, x1, hi
	mov x5, #0
cg_loop:
	cmp x5, x4
	b.eq cg_prefix
	ldrb w6, [x0, x5]
	ldrb w7, [x2, x5]
	cmp w6, w7
	b.lo cg_le
	b.hi cg_gt
	add x5, x5, #1
	b cg_loop
cg_prefix:
	cmp x1, x3
	b.hi cg_gt
cg_le:
	mov x0, #0
	ret
cg_gt:
	mov x0, #1
	ret

.section .rodata
nl_byte:
	.byte 0x0A

.bss
.align 3
inbuf:
	.zero 32768
.align 3
line_ptr:
	.zero 8000
.align 3
line_len:
	.zero 8000
.align 3
temp_ptr:
	.zero 8000
.align 3
temp_len:
	.zero 8000
