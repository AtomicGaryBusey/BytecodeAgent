.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

.section .rodata
nl:
    .byte 0x0A

.bss
.align 3
inbuf:
    .zero BUF_SIZE
line_off:
    .zero MAX_LINES * 4
line_len:
    .zero MAX_LINES * 4

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, line_off
    add x22, x22, :lo12:line_off
    adrp x23, line_len
    add x23, x23, :lo12:line_len

    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    mov x4, #0
    mov x5, #0
parse_loop:
    cmp x4, x20
    b.hs parse_tail
    ldrb w6, [x19, x4]
    add x4, x4, #1
    cmp w6, #0x0A
    b.ne parse_loop
    sub x6, x4, x5
    sub x6, x6, #1
    str w5, [x22, x21, lsl #2]
    str w6, [x23, x21, lsl #2]
    add x21, x21, #1
    mov x5, x4
    b parse_loop
parse_tail:
    cmp x5, x20
    b.hs do_sort
    sub x6, x20, x5
    str w5, [x22, x21, lsl #2]
    str w6, [x23, x21, lsl #2]
    add x21, x21, #1

do_sort:
    cmp x21, #2
    b.lo do_output
    lsr x25, x21, #1
build_loop:
    sub x25, x25, #1
    mov x0, x25
    mov x1, x21
    bl sift_down
    cbnz x25, build_loop
    mov x24, x21
extract_loop:
    cmp x24, #1
    b.ls do_output
    sub x24, x24, #1
    mov x0, #0
    mov x1, x24
    bl swap_lines
    mov x0, #0
    mov x1, x24
    bl sift_down
    b extract_loop

do_output:
    adrp x26, nl
    add x26, x26, :lo12:nl
    mov x24, #0
out_loop:
    cmp x24, x21
    b.hs do_exit
    ldr w2, [x23, x24, lsl #2]
    ldr w3, [x22, x24, lsl #2]
    cbz w2, out_nl
    mov x0, #1
    add x1, x19, x3
    mov x8, #SYS_write
    svc #0
out_nl:
    mov x0, #1
    mov x1, x26
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x24, x24, #1
    b out_loop

do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b do_exit

// w0 < 0 if line[x0] < line[x1], > 0 if greater, 0 if equal
cmp_line_idx:
    ldr w2, [x22, x0, lsl #2]
    ldr w3, [x23, x0, lsl #2]
    ldr w4, [x22, x1, lsl #2]
    ldr w5, [x23, x1, lsl #2]
    cmp w3, w5
    csel w6, w3, w5, lo
    mov w7, #0
cmp_loop:
    cmp w7, w6
    b.hs cmp_lens
    add x8, x19, x2
    ldrb w9, [x8, x7]
    add x8, x19, x4
    ldrb w10, [x8, x7]
    cmp w9, w10
    b.ne cmp_diff
    add w7, w7, #1
    b cmp_loop
cmp_diff:
    sub w0, w9, w10
    ret
cmp_lens:
    sub w0, w3, w5
    ret

swap_lines:
    ldr w2, [x22, x0, lsl #2]
    ldr w3, [x22, x1, lsl #2]
    str w3, [x22, x0, lsl #2]
    str w2, [x22, x1, lsl #2]
    ldr w2, [x23, x0, lsl #2]
    ldr w3, [x23, x1, lsl #2]
    str w3, [x23, x0, lsl #2]
    str w2, [x23, x1, lsl #2]
    ret

// max-heap sift: x0 = root index, x1 = exclusive end
sift_down:
    stp x29, x30, [sp, #-48]!
    str x0, [sp, #16]
    str x1, [sp, #24]
sift_loop:
    ldr x0, [sp, #16]
    lsl x2, x0, #1
    add x2, x2, #1
    ldr x1, [sp, #24]
    cmp x2, x1
    b.hs sift_done
    add x3, x2, #1
    cmp x3, x1
    b.hs sift_have_child
    stp x2, x3, [sp, #-16]!
    mov x0, x2
    mov x1, x3
    bl cmp_line_idx
    ldp x2, x3, [sp], #16
    cmp w0, #0
    csel x2, x3, x2, lt
sift_have_child:
    str x2, [sp, #32]
    ldr x0, [sp, #16]
    mov x1, x2
    bl cmp_line_idx
    cmp w0, #0
    b.ge sift_done
    ldr x0, [sp, #16]
    ldr x1, [sp, #32]
    bl swap_lines
    ldr x1, [sp, #32]
    str x1, [sp, #16]
    b sift_loop
sift_done:
    ldp x29, x30, [sp], #48
    ret
