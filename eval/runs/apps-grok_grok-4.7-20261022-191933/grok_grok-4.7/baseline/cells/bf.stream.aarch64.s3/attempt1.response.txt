.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536
.equ MAX_NEST, 1000

.globl _start
.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x26, jump_tab
    add x26, x26, :lo12:jump_tab
    adrp x27, tape
    add x27, x27, :lo12:tape
    adrp x28, brack_stack
    add x28, x28, :lo12:brack_stack

    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, #BUF_SIZE
    b.lo read_loop
read_done:
    mov x10, #0
find_loop:
    cmp x10, x20
    b.eq no_bang
    ldrb w0, [x19, x10]
    cmp w0, #0x21
    b.eq found_bang
    add x10, x10, #1
    b find_loop
found_bang:
    mov x23, x10
    add x24, x10, #1
    mov x25, x20
    b match_brackets
no_bang:
    mov x23, x20
    mov x24, x20
    mov x25, x20

match_brackets:
    mov x9, #0
    mov x10, #0
match_loop:
    cmp x10, x23
    b.eq match_done
    ldrb w0, [x19, x10]
    cmp w0, #0x5b
    b.eq do_open
    cmp w0, #0x5d
    b.eq do_close
    add x10, x10, #1
    b match_loop
do_open:
    cmp x9, #MAX_NEST
    b.hs bad
    str x10, [x28, x9, lsl #3]
    add x9, x9, #1
    add x10, x10, #1
    b match_loop
do_close:
    cbz x9, bad
    sub x9, x9, #1
    ldr x1, [x28, x9, lsl #3]
    add x2, x10, #1
    str w2, [x26, x1, lsl #2]
    add x2, x1, #1
    str w2, [x26, x10, lsl #2]
    add x10, x10, #1
    b match_loop
match_done:
    cbnz x9, bad

    mov x21, #0
    mov x22, #0
exec_loop:
    cmp x21, x23
    b.hs done_ok
    ldrb w0, [x19, x21]
    cmp w0, #0x3e
    b.eq c_right
    cmp w0, #0x3c
    b.eq c_left
    cmp w0, #0x2b
    b.eq c_inc
    cmp w0, #0x2d
    b.eq c_dec
    cmp w0, #0x2e
    b.eq c_out
    cmp w0, #0x2c
    b.eq c_in
    cmp w0, #0x5b
    b.eq c_lbr
    cmp w0, #0x5d
    b.eq c_rbr
    add x21, x21, #1
    b exec_loop

c_right:
    add x22, x22, #1
    add x21, x21, #1
    b exec_loop
c_left:
    sub x22, x22, #1
    add x21, x21, #1
    b exec_loop
c_inc:
    ldrb w0, [x27, x22]
    add w0, w0, #1
    strb w0, [x27, x22]
    add x21, x21, #1
    b exec_loop
c_dec:
    ldrb w0, [x27, x22]
    sub w0, w0, #1
    strb w0, [x27, x22]
    add x21, x21, #1
    b exec_loop
c_out:
    mov x0, #1
    add x1, x27, x22
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x21, x21, #1
    b exec_loop
c_in:
    cmp x24, x25
    b.hs c_in_eof
    ldrb w0, [x19, x24]
    add x24, x24, #1
    strb w0, [x27, x22]
    add x21, x21, #1
    b exec_loop
c_in_eof:
    strb wzr, [x27, x22]
    add x21, x21, #1
    b exec_loop
c_lbr:
    ldrb w0, [x27, x22]
    cbz w0, c_lbr_jump
    add x21, x21, #1
    b exec_loop
c_lbr_jump:
    ldr w0, [x26, x21, lsl #2]
    mov x21, x0
    b exec_loop
c_rbr:
    ldrb w0, [x27, x22]
    cbz w0, c_rbr_skip
    ldr w0, [x26, x21, lsl #2]
    mov x21, x0
    b exec_loop
c_rbr_skip:
    add x21, x21, #1
    b exec_loop

done_ok:
    mov x0, #0
    b exit_now
bad:
    mov x0, #1
exit_now:
    mov x8, #SYS_exit
    svc #0
    b exit_now

.bss
.align 3
input_buf:
    .zero 65536
tape:
    .zero 30000
.align 3
jump_tab:
    .zero 262144
brack_stack:
    .zero 8000
