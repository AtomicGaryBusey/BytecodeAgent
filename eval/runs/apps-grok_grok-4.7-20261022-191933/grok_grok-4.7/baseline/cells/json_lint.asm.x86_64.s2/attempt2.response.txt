.text
.globl _start

_start:
  lea r12, [rip + inbuf]
  mov r13, r12
  mov r14, r12
  add r14, 65536
read_loop:
  cmp r13, r14
  jae read_done
  xor eax, eax
  xor edi, edi
  mov rsi, r13
  mov rdx, r14
  sub rdx, r13
  syscall
  test rax, rax
  jle read_done
  add r13, rax
  jmp read_loop
read_done:
  lea rsi, [rip + inbuf]
  mov rdi, r13
  call skip_ws
  cmp rsi, rdi
  jae fail
  call parse_value
  call skip_ws
  cmp rsi, rdi
  jne fail
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + msg_ok]
  mov edx, 3
  syscall
ok_halt:
  mov eax, 60
  xor edi, edi
  syscall
  jmp ok_halt

fail:
  lea rax, [rip + inbuf]
  sub rsi, rax
  mov rax, rsi
  lea rbx, [rip + numtmp]
  add rbx, 20
  mov byte ptr [rbx], 10
  lea r9, [rbx + 1]
  test rax, rax
  jnz fail_digits
  dec rbx
  mov byte ptr [rbx], '0'
  jmp fail_print
fail_digits:
  mov ecx, 10
fail_div_loop:
  xor edx, edx
  div rcx
  add dl, '0'
  dec rbx
  mov byte ptr [rbx], dl
  test rax, rax
  jnz fail_div_loop
fail_print:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + msg_err]
  mov edx, 9
  syscall
  mov eax, 1
  mov edi, 1
  mov rsi, rbx
  mov rdx, r9
  sub rdx, rbx
  syscall
fail_halt:
  mov eax, 60
  mov edi, 1
  syscall
  jmp fail_halt

skip_ws:
sw_loop:
  cmp rsi, rdi
  jae sw_done
  movzx eax, byte ptr [rsi]
  cmp al, 0x20
  je sw_adv
  cmp al, 0x09
  je sw_adv
  cmp al, 0x0A
  je sw_adv
  cmp al, 0x0D
  je sw_adv
  ret
sw_adv:
  inc rsi
  jmp sw_loop
sw_done:
  ret

parse_value:
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, 0x22
  je parse_string
  cmp al, 't'
  je pv_true
  cmp al, 'f'
  je pv_false
  cmp al, 'n'
  je pv_null
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jb fail
  cmp al, '9'
  jbe parse_number
  jmp fail
pv_true:
  lea rax, [rip + s_true]
  mov ecx, 4
  jmp match_lit
pv_false:
  lea rax, [rip + s_false]
  mov ecx, 5
  jmp match_lit
pv_null:
  lea rax, [rip + s_null]
  mov ecx, 4
  jmp match_lit

match_lit:
ml_loop:
  cmp rsi, rdi
  jae fail
  movzx edx, byte ptr [rsi]
  cmp dl, byte ptr [rax]
  jne fail
  inc rsi
  inc rax
  dec ecx
  jnz ml_loop
  ret

parse_number:
  movzx eax, byte ptr [rsi]
  cmp al, '-'
  jne pn_int
  inc rsi
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
pn_int:
  cmp al, '0'
  je pn_zero
  cmp al, '1'
  jb fail
  cmp al, '9'
  ja fail
  inc rsi
pn_digits:
  cmp rsi, rdi
  jae pn_ok
  movzx eax, byte ptr [rsi]
  cmp al, '0'
  jb pn_after
  cmp al, '9'
  ja pn_after
  inc rsi
  jmp pn_digits
pn_zero:
  inc rsi
  cmp rsi, rdi
  jae pn_ok
  movzx eax, byte ptr [rsi]
  cmp al, '0'
  jb pn_after
  cmp al, '9'
  ja pn_after
  jmp fail
pn_after:
  cmp al, '.'
  je pn_frac
  cmp al, 'e'
  je pn_exp
  cmp al, 'E'
  je pn_exp
  ret
pn_frac:
  inc rsi
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, '0'
  jb fail
  cmp al, '9'
  ja fail
pn_frac_digits:
  inc rsi
  cmp rsi, rdi
  jae pn_ok
  movzx eax, byte ptr [rsi]
  cmp al, '0'
  jb pn_after_frac
  cmp al, '9'
  ja pn_after_frac
  jmp pn_frac_digits
pn_after_frac:
  cmp al, 'e'
  je pn_exp
  cmp al, 'E'
  je pn_exp
  ret
pn_exp:
  inc rsi
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, '+'
  je pn_exp_sign
  cmp al, '-'
  je pn_exp_sign
  jmp pn_exp_first
pn_exp_sign:
  inc rsi
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
pn_exp_first:
  cmp al, '0'
  jb fail
  cmp al, '9'
  ja fail
pn_exp_digits:
  inc rsi
  cmp rsi, rdi
  jae pn_ok
  movzx eax, byte ptr [rsi]
  cmp al, '0'
  jb pn_ok
  cmp al, '9'
  ja pn_ok
  jmp pn_exp_digits
pn_ok:
  ret

parse_string:
  inc rsi
ps_loop:
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, 0x22
  je ps_end
  cmp al, 0x5C
  je ps_esc
  cmp al, 0x20
  jb fail
  inc rsi
  jmp ps_loop
ps_end:
  inc rsi
  ret
ps_esc:
  inc rsi
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, 0x22
  je ps_esc_ok
  cmp al, 0x5C
  je ps_esc_ok
  cmp al, '/'
  je ps_esc_ok
  cmp al, 'b'
  je ps_esc_ok
  cmp al, 'f'
  je ps_esc_ok
  cmp al, 'n'
  je ps_esc_ok
  cmp al, 'r'
  je ps_esc_ok
  cmp al, 't'
  je ps_esc_ok
  cmp al, 'u'
  je ps_u
  jmp fail
ps_esc_ok:
  inc rsi
  jmp ps_loop
ps_u:
  inc rsi
  mov ecx, 4
ps_u_loop:
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, '0'
  jb fail
  cmp al, '9'
  jbe ps_u_ok
  or al, 0x20
  cmp al, 'a'
  jb fail
  cmp al, 'f'
  ja fail
ps_u_ok:
  inc rsi
  dec ecx
  jnz ps_u_loop
  jmp ps_loop

parse_array:
  inc rsi
  call skip_ws
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, ']'
  je pa_close
  call parse_value
pa_loop:
  call skip_ws
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, ']'
  je pa_close
  cmp al, ','
  jne fail
  inc rsi
  call skip_ws
  call parse_value
  jmp pa_loop
pa_close:
  inc rsi
  ret

parse_object:
  inc rsi
  call skip_ws
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, '}'
  je po_close
  call parse_member
po_loop:
  call skip_ws
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, '}'
  je po_close
  cmp al, ','
  jne fail
  inc rsi
  call skip_ws
  call parse_member
  jmp po_loop
po_close:
  inc rsi
  ret

parse_member:
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, 0x22
  jne fail
  call parse_string
  call skip_ws
  cmp rsi, rdi
  jae fail
  movzx eax, byte ptr [rsi]
  cmp al, ':'
  jne fail
  inc rsi
  call skip_ws
  jmp parse_value

.section .rodata
msg_ok:
  .ascii "ok\n"
msg_err:
  .ascii "error at "
s_true:
  .ascii "true"
s_false:
  .ascii "false"
s_null:
  .ascii "null"

.bss
.balign 8
inbuf:
  .zero 65536
numtmp:
  .zero 32
