.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ INPUT_CAP, 32768
.equ LINE_CAP, 1000

.globl _start
.text

compare:
  lea rax, [rip + line_off]
  mov ecx, dword ptr [rax + rsi*4]
  mov edx, dword ptr [rax + rdi*4]
  lea rax, [rip + line_len]
  mov r8d, dword ptr [rax + rsi*4]
  mov r9d, dword ptr [rax + rdi*4]
  lea rax, [rip + input_buf]
  lea r10, [rax + rcx]
  lea r11, [rax + rdx]
  mov ecx, r8d
  cmp ecx, r9d
  cmova ecx, r9d
  xor eax, eax
cmp_loop:
  cmp eax, ecx
  jae cmp_pref
  movzx edx, byte ptr [r10 + rax]
  cmp dl, byte ptr [r11 + rax]
  jb cmp_less
  ja cmp_gt
  inc eax
  jmp cmp_loop
cmp_pref:
  cmp r8d, r9d
  jb cmp_less
  ja cmp_gt
  xor eax, eax
  ret
cmp_less:
  mov eax, -1
  ret
cmp_gt:
  mov eax, 1
  ret

sift_down:
  mov r14, rdi
  mov rbp, rsi
sd_loop:
  mov r15, r14
  shl r15, 1
  inc r15
  cmp r15, rbp
  jae sd_done
  mov r13, r14
  mov esi, dword ptr [rbx + r15*4]
  mov edi, dword ptr [rbx + r13*4]
  call compare
  test eax, eax
  jle sd_left_done
  mov r13, r15
sd_left_done:
  lea r8, [r15 + 1]
  cmp r8, rbp
  jae sd_right_done
  mov esi, dword ptr [rbx + r8*4]
  mov edi, dword ptr [rbx + r13*4]
  call compare
  test eax, eax
  jle sd_right_done
  lea r13, [r15 + 1]
sd_right_done:
  cmp r13, r14
  je sd_done
  mov eax, dword ptr [rbx + r14*4]
  mov ecx, dword ptr [rbx + r13*4]
  mov dword ptr [rbx + r14*4], ecx
  mov dword ptr [rbx + r13*4], eax
  mov r14, r13
  jmp sd_loop
sd_done:
  ret

_start:
  xor rbx, rbx
read_loop:
  mov rdx, INPUT_CAP
  sub rdx, rbx
  jz read_done
  xor eax, eax
  xor edi, edi
  lea rsi, [rip + input_buf]
  add rsi, rbx
  syscall
  test rax, rax
  jle read_done
  add rbx, rax
  jmp read_loop
read_done:
  lea rsi, [rip + input_buf]
  lea rdi, [rsi + rbx]
  xor r12, r12
parse_loop:
  cmp rsi, rdi
  jae parse_done
  mov r13, rsi
scan:
  cmp rsi, rdi
  je eof_line
  movzx eax, byte ptr [rsi]
  inc rsi
  cmp al, 10
  jne scan
  mov rdx, rsi
  dec rdx
  sub rdx, r13
  jmp store_line
eof_line:
  mov rdx, rsi
  sub rdx, r13
store_line:
  lea rax, [rip + input_buf]
  mov rcx, r13
  sub rcx, rax
  lea rax, [rip + line_off]
  mov dword ptr [rax + r12*4], ecx
  lea rax, [rip + line_len]
  mov dword ptr [rax + r12*4], edx
  lea rax, [rip + order]
  mov dword ptr [rax + r12*4], r12d
  inc r12
  jmp parse_loop
parse_done:
  cmp r12, 2
  jb sort_done
  lea rbx, [rip + order]
  mov r13, r12
  shr r13, 1
build_loop:
  test r13, r13
  jz extract_init
  dec r13
  push r13
  mov rdi, r13
  mov rsi, r12
  call sift_down
  pop r13
  jmp build_loop
extract_init:
  mov r13, r12
extract_loop:
  cmp r13, 1
  jbe sort_done
  dec r13
  mov eax, dword ptr [rbx]
  mov ecx, dword ptr [rbx + r13*4]
  mov dword ptr [rbx], ecx
  mov dword ptr [rbx + r13*4], eax
  push r13
  xor edi, edi
  mov rsi, r13
  call sift_down
  pop r13
  jmp extract_loop
sort_done:
  xor r13, r13
out_loop:
  cmp r13, r12
  jae exit_ok
  lea rax, [rip + order]
  mov ebx, dword ptr [rax + r13*4]
  lea rax, [rip + line_len]
  mov edx, dword ptr [rax + rbx*4]
  lea rax, [rip + line_off]
  mov eax, dword ptr [rax + rbx*4]
  lea rsi, [rip + input_buf]
  add rsi, rax
  test rdx, rdx
  jz write_nl
  mov eax, SYS_write
  mov edi, 1
  syscall
write_nl:
  mov eax, SYS_write
  mov edi, 1
  lea rsi, [rip + nl]
  mov edx, 1
  syscall
  inc r13
  jmp out_loop
exit_ok:
  mov eax, SYS_exit
  xor edi, edi
  syscall
  jmp exit_ok

.section .rodata
nl:
  .byte 10

.bss
.balign 8
input_buf:
  .zero INPUT_CAP
line_off:
  .zero LINE_CAP * 4
line_len:
  .zero LINE_CAP * 4
order:
  .zero LINE_CAP * 4
