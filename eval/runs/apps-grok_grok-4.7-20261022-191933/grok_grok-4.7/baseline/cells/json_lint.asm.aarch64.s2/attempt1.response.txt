.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
.globl _start

_start:
	adrp x19, buf
	add x19, x19, :lo12:buf
	mov x20, #0
read_loop:
	mov x0, #0
	add x1, x19, x20
	movz x2, #1, lsl #16
	sub x2, x2, x20
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	b.le read_done
	add x20, x20, x0
	movz x3, #1, lsl #16
	cmp x20, x3
	b.lo read_loop
read_done:
	mov x21, #0
	bl skip_ws
	bl parse_value
	bl skip_ws
	cmp x21, x20
	b.ne fail
	mov x0, #1
	adrp x1, msg_ok
	add x1, x1, :lo12:msg_ok
	mov x2, #3
	mov x8, #SYS_write
	svc #0
	mov x0, #0
	mov x8, #SYS_exit
	svc #0
	b .

fail:
	mov x0, x21
	adrp x1, numtmp
	add x1, x1, :lo12:numtmp
	add x1, x1, #32
	mov x5, x1
	mov x3, #10
	cbnz x0, fail_conv
	sub x1, x1, #1
	mov w4, #0x30
	strb w4, [x1]
	b fail_conv_done
fail_conv:
	udiv x4, x0, x3
	msub x6, x4, x3, x0
	add w6, w6, #0x30
	sub x1, x1, #1
	strb w6, [x1]
	mov x0, x4
	cbnz x0, fail_conv
fail_conv_done:
	mov x14, x1
	sub x15, x5, x1
	mov x0, #1
	adrp x1, prefix
	add x1, x1, :lo12:prefix
	mov x2, #9
	mov x8, #SYS_write
	svc #0
	mov x0, #1
	mov x1, x14
	mov x2, x15
	mov x8, #SYS_write
	svc #0
	mov x0, #1
	adrp x1, nlbyte
	add x1, x1, :lo12:nlbyte
	mov x2, #1
	mov x8, #SYS_write
	svc #0
	mov x0, #1
	mov x8, #SYS_exit
	svc #0
	b .

skip_ws:
sw_loop:
	cmp x21, x20
	b.hs sw_done
	ldrb w0, [x19, x21]
	cmp w0, #0x20
	b.eq sw_yes
	cmp w0, #0x09
	b.eq sw_yes
	cmp w0, #0x0A
	b.eq sw_yes
	cmp w0, #0x0D
	b.eq sw_yes
	b sw_done
sw_yes:
	add x21, x21, #1
	b sw_loop
sw_done:
	ret

parse_value:
	str x30, [sp, #-16]!
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x7B
	b.eq pv_obj
	cmp w0, #0x5B
	b.eq pv_arr
	cmp w0, #0x22
	b.eq pv_str
	cmp w0, #0x74
	b.eq pv_true
	cmp w0, #0x66
	b.eq pv_false
	cmp w0, #0x6E
	b.eq pv_null
	cmp w0, #0x2D
	b.eq pv_num
	cmp w0, #0x30
	b.lo fail
	cmp w0, #0x39
	b.hi fail
pv_num:
	bl parse_number
	b pv_done
pv_obj:
	bl parse_object
	b pv_done
pv_arr:
	bl parse_array
	b pv_done
pv_str:
	bl parse_string
	b pv_done
pv_true:
	adrp x2, lit_true
	add x2, x2, :lo12:lit_true
	mov x4, #4
	bl parse_lit
	b pv_done
pv_false:
	adrp x2, lit_false
	add x2, x2, :lo12:lit_false
	mov x4, #5
	bl parse_lit
	b pv_done
pv_null:
	adrp x2, lit_null
	add x2, x2, :lo12:lit_null
	mov x4, #4
	bl parse_lit
	b pv_done
pv_done:
	ldr x30, [sp], #16
	ret

parse_lit:
pl_loop:
	cbz x4, pl_done
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	ldrb w3, [x2]
	cmp w0, w3
	b.ne fail
	add x21, x21, #1
	add x2, x2, #1
	sub x4, x4, #1
	b pl_loop
pl_done:
	ret

parse_string:
	add x21, x21, #1
ps_loop:
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x22
	b.eq ps_end
	cmp w0, #0x5C
	b.eq ps_esc
	cmp w0, #0x20
	b.lo fail
	add x21, x21, #1
	b ps_loop
ps_end:
	add x21, x21, #1
	ret
ps_esc:
	add x21, x21, #1
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x22
	b.eq ps_esc_ok
	cmp w0, #0x5C
	b.eq ps_esc_ok
	cmp w0, #0x2F
	b.eq ps_esc_ok
	cmp w0, #0x62
	b.eq ps_esc_ok
	cmp w0, #0x66
	b.eq ps_esc_ok
	cmp w0, #0x6E
	b.eq ps_esc_ok
	cmp w0, #0x72
	b.eq ps_esc_ok
	cmp w0, #0x74
	b.eq ps_esc_ok
	cmp w0, #0x75
	b.eq ps_u
	b fail
ps_esc_ok:
	add x21, x21, #1
	b ps_loop
ps_u:
	add x21, x21, #1
	mov x4, #4
ps_u_loop:
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.lo fail
	cmp w0, #0x39
	b.ls ps_hex_ok
	cmp w0, #0x41
	b.lo fail
	cmp w0, #0x46
	b.ls ps_hex_ok
	cmp w0, #0x61
	b.lo fail
	cmp w0, #0x66
	b.ls ps_hex_ok
	b fail
ps_hex_ok:
	add x21, x21, #1
	subs x4, x4, #1
	b.ne ps_u_loop
	b ps_loop

parse_number:
	ldrb w0, [x19, x21]
	cmp w0, #0x2D
	b.ne pn_int
	add x21, x21, #1
	cmp x21, x20
	b.hs fail
pn_int:
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.eq pn_zero
	cmp w0, #0x31
	b.lo fail
	cmp w0, #0x39
	b.hi fail
pn_digs:
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_ret
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.lo pn_frac
	cmp w0, #0x39
	b.ls pn_digs
	b pn_frac
pn_zero:
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_ret
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.lo pn_frac
	cmp w0, #0x39
	b.ls fail
	b pn_frac
pn_frac:
	cmp w0, #0x2E
	b.ne pn_exp
	add x21, x21, #1
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.lo fail
	cmp w0, #0x39
	b.hi fail
pn_frac_digs:
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_ret
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.lo pn_exp
	cmp w0, #0x39
	b.ls pn_frac_digs
pn_exp:
	cmp w0, #0x65
	b.eq pn_exp_body
	cmp w0, #0x45
	b.eq pn_exp_body
	ret
pn_exp_body:
	add x21, x21, #1
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x2B
	b.eq pn_exp_sign
	cmp w0, #0x2D
	b.eq pn_exp_sign
	b pn_exp_first
pn_exp_sign:
	add x21, x21, #1
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
pn_exp_first:
	cmp w0, #0x30
	b.lo fail
	cmp w0, #0x39
	b.hi fail
pn_exp_digs:
	add x21, x21, #1
	cmp x21, x20
	b.hs pn_ret
	ldrb w0, [x19, x21]
	cmp w0, #0x30
	b.lo pn_ret
	cmp w0, #0x39
	b.hi pn_ret
	b pn_exp_digs
pn_ret:
	ret

parse_array:
	str x30, [sp, #-16]!
	add x21, x21, #1
	bl skip_ws
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x5D
	b.eq pa_close
	bl parse_value
pa_loop:
	bl skip_ws
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x5D
	b.eq pa_close
	cmp w0, #0x2C
	b.ne fail
	add x21, x21, #1
	bl skip_ws
	bl parse_value
	b pa_loop
pa_close:
	add x21, x21, #1
	ldr x30, [sp], #16
	ret

parse_object:
	str x30, [sp, #-16]!
	add x21, x21, #1
	bl skip_ws
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x7D
	b.eq po_close
po_member:
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x22
	b.ne fail
	bl parse_string
	bl skip_ws
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x3A
	b.ne fail
	add x21, x21, #1
	bl skip_ws
	bl parse_value
	bl skip_ws
	cmp x21, x20
	b.hs fail
	ldrb w0, [x19, x21]
	cmp w0, #0x7D
	b.eq po_close
	cmp w0, #0x2C
	b.ne fail
	add x21, x21, #1
	bl skip_ws
	b po_member
po_close:
	add x21, x21, #1
	ldr x30, [sp], #16
	ret

.section .rodata
msg_ok:
	.ascii "ok"
	.byte 0x0A
prefix:
	.ascii "error at "
nlbyte:
	.byte 0x0A
lit_true:
	.ascii "true"
lit_false:
	.ascii "false"
lit_null:
	.ascii "null"

.bss
.align 3
buf:
	.zero 65536
numtmp:
	.zero 32
