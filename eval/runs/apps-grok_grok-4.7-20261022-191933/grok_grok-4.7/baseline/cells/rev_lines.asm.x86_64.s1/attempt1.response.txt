.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_CAP, 65536

.section .rodata
nlbyte:
    .byte 0x0A

.bss
.balign 8
inbuf:
    .zero 65536

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, SYS_READ
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, BUF_CAP
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    cmp rbx, BUF_CAP
    jb read_loop
read_done:
    lea r14, [rip + inbuf]
    xor r12d, r12d
line_loop:
    cmp r12, rbx
    jae do_exit
    mov r13, r12
scan_loop:
    cmp r12, rbx
    jae final_line
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x0A
    je got_nl
    inc r12
    jmp scan_loop
got_nl:
    mov r15, r12
    call emit_rev
    inc r12
    jmp line_loop
final_line:
    mov r15, rbx
    call emit_rev
do_exit:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

emit_rev:
    mov rsi, r13
    cmp rsi, r15
    jae emit_nl
    mov rdi, r15
    dec rdi
rev_loop:
    cmp rsi, rdi
    jae emit_body
    mov al, byte ptr [r14 + rsi]
    mov dl, byte ptr [r14 + rdi]
    mov byte ptr [r14 + rsi], dl
    mov byte ptr [r14 + rdi], al
    inc rsi
    dec rdi
    jmp rev_loop
emit_body:
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [r14 + r13]
    mov rdx, r15
    sub rdx, r13
    syscall
emit_nl:
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + nlbyte]
    mov edx, 1
    syscall
    ret
