.globl _start
.equ BUFSIZE, 65536

.text
_start:
    xor ebx, ebx
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, BUFSIZE
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, BUFSIZE
    jb read_more
read_done:
    lea r15, [rip + inbuf]
    xor r12d, r12d
    test rbx, rbx
    jz exit_ok
line_loop:
    cmp r12, rbx
    jae exit_ok
    mov r13, r12
scan:
    cmp r12, rbx
    jae no_nl
    mov al, byte ptr [r15 + r12]
    cmp al, 0x0A
    je found_nl
    inc r12
    jmp scan
found_nl:
    mov r14, r12
    call reverse
    lea rsi, [r15 + r13]
    mov rdx, r12
    sub rdx, r13
    inc rdx
    call write_all
    inc r12
    jmp line_loop
no_nl:
    mov r14, r12
    call reverse
    lea rsi, [r15 + r13]
    mov rdx, r12
    sub rdx, r13
    call write_all
    lea rsi, [rip + nl]
    mov edx, 1
    call write_all
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

# reverse bytes [r13, r14) of the buffer based at r15
reverse:
    mov r8, r14
    sub r8, r13
    cmp r8, 2
    jb rev_done
    mov r8, r13
    mov r9, r14
    dec r9
rev_loop:
    mov al, byte ptr [r15 + r8]
    mov dl, byte ptr [r15 + r9]
    mov byte ptr [r15 + r8], dl
    mov byte ptr [r15 + r9], al
    inc r8
    dec r9
    cmp r8, r9
    jb rev_loop
rev_done:
    ret

# write rdx bytes from rsi to stdout; short writes are retried
write_all:
    test rdx, rdx
    jz wr_done
wr_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wr_done
    add rsi, rax
    sub rdx, rax
    jnz wr_loop
wr_done:
    ret

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 16
inbuf:
    .zero 65536
