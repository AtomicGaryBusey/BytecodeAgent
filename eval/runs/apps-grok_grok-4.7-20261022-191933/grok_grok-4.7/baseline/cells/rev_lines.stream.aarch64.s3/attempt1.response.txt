.globl _start

.text
_start:
	mov	x19, xzr
	adrp	x20, inbuf
	add	x20, x20, :lo12:inbuf

read_loop:
	mov	x0, xzr
	add	x1, x20, x19
	movz	x2, #1, lsl #16
	sub	x2, x2, x19
	cbz	x2, process
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	ble	process
	add	x19, x19, x0
	b	read_loop

process:
	mov	x21, xzr

line_loop:
	cmp	x21, x19
	bhs	do_exit
	mov	x22, x21

scan:
	cmp	x22, x19
	bhs	final_line
	ldrb	w3, [x20, x22]
	cmp	w3, #0x0A
	beq	got_nl
	add	x22, x22, #1
	b	scan

got_nl:
	mov	x25, xzr
	b	emit

final_line:
	mov	x25, #1

emit:
	sub	x2, x22, x21
	cbz	x2, place_nl
	mov	x23, x21
	sub	x24, x22, #1

rev_loop:
	cmp	x23, x24
	bhs	place_nl
	ldrb	w3, [x20, x23]
	ldrb	w4, [x20, x24]
	strb	w4, [x20, x23]
	strb	w3, [x20, x24]
	add	x23, x23, #1
	sub	x24, x24, #1
	b	rev_loop

place_nl:
	cbz	x25, do_write
	mov	w3, #0x0A
	strb	w3, [x20, x22]

do_write:
	mov	x0, #1
	add	x1, x20, x21
	sub	x2, x22, x21
	add	x2, x2, #1
	mov	x8, #64
	svc	#0
	cbnz	x25, do_exit
	add	x21, x22, #1
	b	line_loop

do_exit:
	mov	x0, xzr
	mov	x8, #93
	svc	#0
	b	do_exit
