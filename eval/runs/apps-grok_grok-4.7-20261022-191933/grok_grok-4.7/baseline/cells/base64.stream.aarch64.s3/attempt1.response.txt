.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_CAP, 49152
.equ WRAP, 76

.globl _start

.text
.balign 4
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19

read_loop:
    sub x2, x20, x19
    mov x3, #IN_CAP
    sub x2, x3, x2
    cbz x2, read_done
    mov x8, #SYS_read
    mov x0, #0
    mov x1, x20
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, x21
    mov x23, #0
    adrp x24, alpha
    add x24, x24, :lo12:alpha

enc_loop:
    sub x0, x20, x19
    cmp x0, #3
    b.lo enc_tail
    ldrb w1, [x19]
    ldrb w2, [x19, #1]
    ldrb w3, [x19, #2]
    add x19, x19, #3

    lsr w4, w1, #2
    and w5, w1, #3
    lsl w5, w5, #4
    lsr w6, w2, #4
    orr w5, w5, w6
    and w6, w2, #15
    lsl w6, w6, #2
    lsr w7, w3, #6
    orr w6, w6, w7
    and w7, w3, #63

    ldrb w0, [x24, x4]
    bl emit
    ldrb w0, [x24, x5]
    bl emit
    ldrb w0, [x24, x6]
    bl emit
    ldrb w0, [x24, x7]
    bl emit
    b enc_loop

enc_tail:
    cbz x0, enc_finish
    cmp x0, #1
    b.eq enc_one

    ldrb w1, [x19]
    ldrb w2, [x19, #1]
    lsr w4, w1, #2
    and w5, w1, #3
    lsl w5, w5, #4
    lsr w6, w2, #4
    orr w5, w5, w6
    and w6, w2, #15
    lsl w6, w6, #2

    ldrb w0, [x24, x4]
    bl emit
    ldrb w0, [x24, x5]
    bl emit
    ldrb w0, [x24, x6]
    bl emit
    mov w0, #61
    bl emit
    b enc_finish

enc_one:
    ldrb w1, [x19]
    lsr w4, w1, #2
    and w5, w1, #3
    lsl w5, w5, #4
    ldrb w0, [x24, x4]
    bl emit
    ldrb w0, [x24, x5]
    bl emit
    mov w0, #61
    bl emit
    mov w0, #61
    bl emit

enc_finish:
    cbz x23, no_nl
    mov w0, #10
    strb w0, [x21], #1

no_nl:
    sub x2, x21, x22
    cbz x2, do_exit
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x22
    svc #0

do_exit:
    mov x8, #SYS_exit
    mov x0, #0
    svc #0

emit:
    strb w0, [x21], #1
    add x23, x23, #1
    cmp x23, #WRAP
    b.ne emit_ret
    mov w0, #10
    strb w0, [x21], #1
    mov x23, #0
emit_ret:
    ret

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 8
inbuf:
    .zero 49152
outbuf:
    .zero 131072
