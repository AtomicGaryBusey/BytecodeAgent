.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
.globl _start

// x0 = value, x1 = dest; advances x1 by 8. x24 = hex table.
put_hex8:
    mov x2, #28
ph_loop:
    lsr x3, x0, x2
    and x3, x3, #0xf
    ldrb w3, [x24, x3]
    strb w3, [x1], #1
    subs x2, x2, #4
    b.pl ph_loop
    ret

// x0 = buf, x1 = length. Writes to stdout.
write_stdout:
    mov x2, x1
    mov x1, x0
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    ret

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x24, hexdigits
    add x24, x24, :lo12:hexdigits
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x20, #0
    movz x26, #1, lsl #16

read_loop:
    mov x8, #SYS_read
    mov x0, #0
    add x1, x19, x20
    sub x2, x26, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, x26
    b.lo read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_line
    sub x22, x20, x21
    cmp x22, #16
    b.ls row_n
    mov x22, #16
row_n:
    mov x0, x21
    mov x1, x25
    bl put_hex8
    mov x23, x1
    mov w3, #32
    strb w3, [x23], #1
    strb w3, [x23], #1
    mov x5, #0

byte_loop:
    cmp x5, x22
    b.hs byte_pad
    add x7, x19, x21
    ldrb w6, [x7, x5]
    lsr x4, x6, #4
    and x4, x4, #0xf
    ldrb w3, [x24, x4]
    strb w3, [x23], #1
    and x4, x6, #0xf
    ldrb w3, [x24, x4]
    strb w3, [x23], #1
    mov w3, #32
    strb w3, [x23], #1
    b byte_after
byte_pad:
    mov w3, #32
    strb w3, [x23], #1
    strb w3, [x23], #1
    strb w3, [x23], #1
byte_after:
    cmp x5, #7
    b.ne byte_next
    mov w3, #32
    strb w3, [x23], #1
byte_next:
    add x5, x5, #1
    cmp x5, #16
    b.lo byte_loop

    mov w3, #32
    strb w3, [x23], #1
    mov w3, #124
    strb w3, [x23], #1
    mov x5, #0

ascii_loop:
    cmp x5, x22
    b.hs ascii_done
    add x7, x19, x21
    ldrb w6, [x7, x5]
    cmp w6, #32
    b.lo ascii_dot
    cmp w6, #126
    b.hi ascii_dot
    b ascii_put
ascii_dot:
    mov w6, #46
ascii_put:
    strb w6, [x23], #1
    add x5, x5, #1
    b ascii_loop

ascii_done:
    mov w3, #124
    strb w3, [x23], #1
    mov w3, #10
    strb w3, [x23], #1
    mov x0, x25
    sub x1, x23, x25
    bl write_stdout
    add x21, x21, x22
    b row_loop

final_line:
    mov x0, x20
    mov x1, x25
    bl put_hex8
    mov w3, #10
    strb w3, [x1]
    mov x0, x25
    mov x1, #9
    bl write_stdout

exit_now:
    mov x8, #SYS_exit
    mov x0, #0
    svc #0
    b exit_now

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 3
inbuf:
    .zero 65536
outbuf:
    .zero 128
