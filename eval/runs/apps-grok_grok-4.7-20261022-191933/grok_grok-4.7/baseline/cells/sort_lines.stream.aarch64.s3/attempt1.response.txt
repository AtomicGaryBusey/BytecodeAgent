.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 32768

.text
.globl _start

_start:
    adrp x24, buf
    add x24, x24, :lo12:buf
    adrp x25, line_off
    add x25, x25, :lo12:line_off
    adrp x26, line_len
    add x26, x26, :lo12:line_len

    mov x19, #0
read_loop:
    mov x0, #0
    add x1, x24, x19
    mov x2, #BUF_SIZE
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    tbnz x0, #63, read_done
    add x19, x19, x0
    b read_loop
read_done:
    mov x20, #0
    mov x21, #0
parse_loop:
    cmp x21, x19
    b.hs parse_done
    mov x22, x21
parse_scan:
    cmp x21, x19
    b.hs parse_store
    ldrb w0, [x24, x21]
    cmp w0, #10
    b.eq parse_store
    add x21, x21, #1
    b parse_scan
parse_store:
    sub x23, x21, x22
    lsl x0, x20, #2
    str w22, [x25, x0]
    str w23, [x26, x0]
    add x20, x20, #1
    cmp x21, x19
    b.hs parse_done
    add x21, x21, #1
    b parse_loop
parse_done:
    mov x19, x20

    mov x20, #1
ins_i:
    cmp x20, x19
    b.hs sort_done
    lsl x4, x20, #2
    ldr w21, [x25, x4]
    ldr w22, [x26, x4]
    mov x23, x20
ins_j:
    cbz x23, ins_place
    sub x6, x23, #1
    lsl x4, x6, #2
    ldr w0, [x25, x4]
    ldr w1, [x26, x4]
    mov w2, w21
    mov w3, w22
    bl cmp_lines
    cmp w0, #0
    b.le ins_place
    sub x6, x23, #1
    lsl x4, x6, #2
    lsl x7, x23, #2
    ldr w0, [x25, x4]
    ldr w1, [x26, x4]
    str w0, [x25, x7]
    str w1, [x26, x7]
    sub x23, x23, #1
    b ins_j
ins_place:
    lsl x4, x23, #2
    str w21, [x25, x4]
    str w22, [x26, x4]
    add x20, x20, #1
    b ins_i

sort_done:
    mov x20, #0
write_loop:
    cmp x20, x19
    b.hs exit_ok
    lsl x4, x20, #2
    ldr w21, [x25, x4]
    ldr w22, [x26, x4]
    cbz w22, write_nl
    mov x0, #1
    add x1, x24, x21
    mov x2, x22
    mov x8, #SYS_write
    svc #0
write_nl:
    mov x0, #1
    adrp x1, nlbyte
    add x1, x1, :lo12:nlbyte
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x20, x20, #1
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// w0=off_a w1=len_a w2=off_b w3=len_b; w0 > 0 if a > b
cmp_lines:
    cmp w1, w3
    csel w4, w1, w3, lo
    add x9, x24, x0
    add x10, x24, x2
    mov w5, #0
cmp_byte:
    cmp w5, w4
    b.hs cmp_len
    ldrb w7, [x9, x5]
    ldrb w8, [x10, x5]
    cmp w7, w8
    b.ne cmp_diff
    add w5, w5, #1
    b cmp_byte
cmp_diff:
    sub w0, w7, w8
    ret
cmp_len:
    sub w0, w1, w3
    ret

.section .rodata
nlbyte:
    .byte 10

.bss
.balign 8
buf:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000
