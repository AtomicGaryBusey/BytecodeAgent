.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.data
nl:
    .byte 10

.bss
.balign 8
buf:
    .zero 65536

.text
.globl _start
_start:
    adrp x20, buf
    add x20, x20, :lo12:buf
    mov x19, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0

line_loop:
    cmp x21, x19
    b.hs exit_ok
    mov x22, x21

scan_loop:
    cmp x22, x19
    b.hs got_line
    ldrb w0, [x20, x22]
    cmp w0, #10
    b.eq got_line
    add x22, x22, #1
    b scan_loop

got_line:
    cmp x21, x22
    b.hs write_line
    mov x23, x21
    sub x24, x22, #1

rev_loop:
    cmp x23, x24
    b.hs write_line
    ldrb w0, [x20, x23]
    ldrb w1, [x20, x24]
    strb w1, [x20, x23]
    strb w0, [x20, x24]
    add x23, x23, #1
    sub x24, x24, #1
    b rev_loop

write_line:
    sub x2, x22, x21
    cbz x2, write_nl
    mov x0, #1
    add x1, x20, x21
    mov x8, #SYS_write
    svc #0

write_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    cmp x22, x19
    b.hs exit_ok
    add x21, x22, #1
    b line_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b exit_ok
