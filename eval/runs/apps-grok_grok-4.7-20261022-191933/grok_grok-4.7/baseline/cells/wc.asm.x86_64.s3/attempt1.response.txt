.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUF_SIZE, 65536

.globl _start
.text

_start:
    cld
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_more:
    mov eax, SYS_read
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jle print_counts
    mov rcx, rax
    lea rbx, [rip + inbuf]

scan_loop:
    movzx eax, byte ptr [rbx]
    inc rbx
    inc r14
    cmp al, 0x0A
    jne classify
    inc r12
classify:
    cmp al, 0x20
    je got_ws
    cmp al, 0x09
    jb got_word
    cmp al, 0x0D
    jbe got_ws
got_word:
    test r15, r15
    jnz scan_next
    mov r15, 1
    inc r13
    jmp scan_next
got_ws:
    xor r15, r15
scan_next:
    dec rcx
    jnz scan_loop
    jmp read_more

print_counts:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call format_u64
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call format_u64
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call format_u64
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall
    mov eax, SYS_exit
    xor edi, edi
    syscall

format_u64:
    lea rsi, [rip + numtmp]
    add rsi, 32
    mov r8, rsi
    mov r10, 10
    test rax, rax
    jnz fmt_loop
    dec rsi
    mov byte ptr [rsi], 0x30
    jmp fmt_copy
fmt_loop:
    xor rdx, rdx
    div r10
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz fmt_loop
fmt_copy:
    mov rcx, r8
    sub rcx, rsi
    rep movsb
    ret

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 96
numtmp:
    .zero 32
