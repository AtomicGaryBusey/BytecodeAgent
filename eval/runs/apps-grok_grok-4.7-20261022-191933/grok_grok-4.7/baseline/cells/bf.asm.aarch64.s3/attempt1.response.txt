.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536
.equ MAX_NEST, 1000
.equ CH_GT, 62
.equ CH_LT, 60
.equ CH_PLUS, 43
.equ CH_MINUS, 45
.equ CH_DOT, 46
.equ CH_COMMA, 44
.equ CH_LB, 91
.equ CH_RB, 93
.equ CH_BANG, 33

.globl _start
.text
.balign 4
_start:
	adrp x27, dtab
	add x27, x27, :lo12:dtab
	adr x1, cmd_skip
	mov x2, #0
dt_fill:
	str x1, [x27, x2, lsl #3]
	add x2, x2, #1
	cmp x2, #256
	b.ne dt_fill

	adr x1, cmd_gt
	mov x2, #CH_GT
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_lt
	mov x2, #CH_LT
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_plus
	mov x2, #CH_PLUS
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_minus
	mov x2, #CH_MINUS
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_dot
	mov x2, #CH_DOT
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_comma
	mov x2, #CH_COMMA
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_lbrack
	mov x2, #CH_LB
	str x1, [x27, x2, lsl #3]
	adr x1, cmd_rbrack
	mov x2, #CH_RB
	str x1, [x27, x2, lsl #3]

	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	mov x20, #0
read_loop:
	cmp x20, #IN_MAX
	b.hs read_done
	mov x0, #0
	add x1, x19, x20
	mov x2, #IN_MAX
	sub x2, x2, x20
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	b.le read_done
	add x20, x20, x0
	b read_loop
read_done:
	mov x4, #0
find_loop:
	cmp x4, x20
	b.hs no_bang
	ldrb w0, [x19, x4]
	cmp w0, #CH_BANG
	b.eq yes_bang
	add x4, x4, #1
	b find_loop
no_bang:
	mov x23, x20
	mov x24, x20
	mov x25, x20
	b brackets
yes_bang:
	mov x23, x4
	add x24, x4, #1
	mov x25, x20

brackets:
	adrp x26, jumptab
	add x26, x26, :lo12:jumptab
	adrp x7, brstack
	add x7, x7, :lo12:brstack
	mov x8, #0
	mov x9, #0
br_loop:
	cmp x9, x23
	b.hs br_done
	ldrb w0, [x19, x9]
	cmp w0, #CH_LB
	b.eq br_push
	cmp w0, #CH_RB
	b.eq br_pop
	add x9, x9, #1
	b br_loop
br_push:
	cmp x8, #MAX_NEST
	b.hs exit_bad
	str x9, [x7, x8, lsl #3]
	add x8, x8, #1
	add x9, x9, #1
	b br_loop
br_pop:
	cbz x8, exit_bad
	sub x8, x8, #1
	ldr x1, [x7, x8, lsl #3]
	str x9, [x26, x1, lsl #3]
	str x1, [x26, x9, lsl #3]
	add x9, x9, #1
	b br_loop
br_done:
	cbnz x8, exit_bad

	adrp x21, tape
	add x21, x21, :lo12:tape
	adrp x29, outbyte
	add x29, x29, :lo12:outbyte
	mov x22, #0
	mov x28, #0

exec_loop:
	cmp x28, x23
	b.hs exit_ok
	ldrb w0, [x19, x28]
	ldr x1, [x27, x0, lsl #3]
	br x1

cmd_skip:
	add x28, x28, #1
	b exec_loop

cmd_gt:
	add x22, x22, #1
	add x28, x28, #1
	b exec_loop

cmd_lt:
	sub x22, x22, #1
	add x28, x28, #1
	b exec_loop

cmd_plus:
	ldrb w0, [x21, x22]
	add w0, w0, #1
	strb w0, [x21, x22]
	add x28, x28, #1
	b exec_loop

cmd_minus:
	ldrb w0, [x21, x22]
	sub w0, w0, #1
	strb w0, [x21, x22]
	add x28, x28, #1
	b exec_loop

cmd_dot:
	ldrb w0, [x21, x22]
	strb w0, [x29]
	mov x0, #1
	mov x1, x29
	mov x2, #1
	mov x8, #SYS_write
	svc #0
	add x28, x28, #1
	b exec_loop

cmd_comma:
	cmp x24, x25
	b.hs cmd_comma_eof
	ldrb w0, [x19, x24]
	add x24, x24, #1
	b cmd_comma_store
cmd_comma_eof:
	mov w0, #0
cmd_comma_store:
	strb w0, [x21, x22]
	add x28, x28, #1
	b exec_loop

cmd_lbrack:
	ldrb w0, [x21, x22]
	cbz w0, cmd_lbrack_jump
	add x28, x28, #1
	b exec_loop
cmd_lbrack_jump:
	ldr x0, [x26, x28, lsl #3]
	add x28, x0, #1
	b exec_loop

cmd_rbrack:
	ldrb w0, [x21, x22]
	cbz w0, cmd_rbrack_adv
	ldr x0, [x26, x28, lsl #3]
	add x28, x0, #1
	b exec_loop
cmd_rbrack_adv:
	add x28, x28, #1
	b exec_loop

exit_ok:
	mov x0, #0
	b do_exit
exit_bad:
	mov x0, #1
do_exit:
	mov x8, #SYS_exit
	svc #0
	b do_exit

.bss
.balign 8
inbuf:
	.zero 65536
tape:
	.zero 30000
.balign 8
jumptab:
	.zero 524288
brstack:
	.zero 8000
dtab:
	.zero 2048
outbyte:
	.zero 1
