.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_CAP, 32768

.text

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, xzr

read_loop:
    mov x2, #BUF_CAP
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, xzr
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, line_off
    add x21, x21, :lo12:line_off
    adrp x22, line_len
    add x22, x22, :lo12:line_len
    mov x23, xzr
    mov x24, xzr
    mov x25, xzr

parse_loop:
    cmp x24, x20
    b.eq parse_done
    ldrb w0, [x19, x24]
    add x24, x24, #1
    cmp w0, #10
    b.ne parse_loop
    sub x0, x24, x25
    sub x0, x0, #1
    str w25, [x21, x23, lsl #2]
    str w0, [x22, x23, lsl #2]
    add x23, x23, #1
    mov x25, x24
    b parse_loop

parse_done:
    cmp x25, x20
    b.eq no_tail
    sub x0, x20, x25
    str w25, [x21, x23, lsl #2]
    str w0, [x22, x23, lsl #2]
    add x23, x23, #1

no_tail:
    lsr x24, x23, #1
    cbz x24, heap_ready

build_loop:
    sub x24, x24, #1
    mov x0, x24
    mov x1, x23
    bl sift_down
    cbnz x24, build_loop

heap_ready:
    mov x25, x23

extract_loop:
    cmp x25, #1
    b.ls sort_done
    sub x25, x25, #1
    ldr w0, [x21]
    ldr w1, [x21, x25, lsl #2]
    str w1, [x21]
    str w0, [x21, x25, lsl #2]
    ldr w0, [x22]
    ldr w1, [x22, x25, lsl #2]
    str w1, [x22]
    str w0, [x22, x25, lsl #2]
    mov x0, xzr
    mov x1, x25
    bl sift_down
    b extract_loop

sort_done:
    mov x26, xzr

print_loop:
    cmp x26, x23
    b.hs do_exit
    ldr w2, [x22, x26, lsl #2]
    cbz w2, write_nl
    ldr w1, [x21, x26, lsl #2]
    add x1, x19, x1
    mov x0, #1
    mov x8, #SYS_write
    svc #0

write_nl:
    adrp x1, nlbyte
    add x1, x1, :lo12:nlbyte
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x26, x26, #1
    b print_loop

do_exit:
    mov x0, xzr
    mov x8, #SYS_exit
    svc #0

// w0 = 1 if line[x0] > line[x1], -1 if less, 0 if equal
cmp_idx:
    ldr w2, [x21, x0, lsl #2]
    ldr w3, [x22, x0, lsl #2]
    ldr w4, [x21, x1, lsl #2]
    ldr w5, [x22, x1, lsl #2]
    add x6, x19, x2
    add x7, x19, x4
    cmp w3, w5
    csel w8, w3, w5, lo
    mov w9, wzr

cmp_bytes:
    cmp w9, w8
    b.eq cmp_lens
    ldrb w10, [x6, x9]
    ldrb w11, [x7, x9]
    cmp w10, w11
    b.ne cmp_diff
    add w9, w9, #1
    b cmp_bytes

cmp_diff:
    cset w0, hi
    cset w12, lo
    sub w0, w0, w12
    ret

cmp_lens:
    cmp w3, w5
    cset w0, hi
    cset w12, lo
    sub w0, w0, w12
    ret

// sift_down: x0 = index, x1 = heap length. Max-heap on line order.
sift_down:
    stp x29, x30, [sp, #-48]!
    str x1, [sp, #16]
    mov x29, x0

sift_loop:
    lsl x0, x29, #1
    add x0, x0, #1
    ldr x1, [sp, #16]
    cmp x0, x1
    b.hs sift_done
    add x2, x0, #1
    mov x3, x0
    cmp x2, x1
    b.hs sift_cmp_parent
    stp x2, x3, [sp, #24]
    mov x0, x2
    mov x1, x3
    bl cmp_idx
    ldp x2, x3, [sp, #24]
    cmp w0, #0
    b.le sift_cmp_parent
    mov x3, x2

sift_cmp_parent:
    stp x3, x29, [sp, #24]
    mov x0, x3
    mov x1, x29
    bl cmp_idx
    ldp x3, x29, [sp, #24]
    cmp w0, #0
    b.le sift_done
    ldr w0, [x21, x3, lsl #2]
    ldr w1, [x21, x29, lsl #2]
    str w0, [x21, x29, lsl #2]
    str w1, [x21, x3, lsl #2]
    ldr w0, [x22, x3, lsl #2]
    ldr w1, [x22, x29, lsl #2]
    str w0, [x22, x29, lsl #2]
    str w1, [x22, x3, lsl #2]
    mov x29, x3
    b sift_loop

sift_done:
    ldp x29, x30, [sp], #48
    ret

.section .rodata
nlbyte:
    .byte 10

.bss
.align 3
inbuf:
    .zero 32768
.align 3
line_off:
    .zero 4000
.align 3
line_len:
    .zero 4000
