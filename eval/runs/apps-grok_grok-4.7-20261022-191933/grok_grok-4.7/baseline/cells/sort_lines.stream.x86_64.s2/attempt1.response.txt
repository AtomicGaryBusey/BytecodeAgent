.globl _start
.text

cmp_lines:
  lea r8, [rip + line_off]
  mov r10d, dword ptr [r8 + rdi*4]
  mov r11d, dword ptr [r8 + rsi*4]
  lea r8, [rip + line_len]
  mov ecx, dword ptr [r8 + rdi*4]
  mov edx, dword ptr [r8 + rsi*4]
  lea r8, [rip + buffer]
  add r10, r8
  add r11, r8
  mov eax, ecx
  cmp eax, edx
  cmova eax, edx
  xor r8d, r8d
cmp_byte:
  cmp r8d, eax
  jae cmp_len
  movzx r9d, byte ptr [r10 + r8]
  movzx esi, byte ptr [r11 + r8]
  cmp r9d, esi
  jne cmp_diff
  inc r8
  jmp cmp_byte
cmp_diff:
  mov eax, r9d
  sub eax, esi
  ret
cmp_len:
  mov eax, ecx
  sub eax, edx
  ret

_start:
  xor r12, r12
read_loop:
  xor eax, eax
  xor edi, edi
  lea rsi, [rip + buffer]
  add rsi, r12
  mov edx, 32768
  sub rdx, r12
  test rdx, rdx
  jz read_done
  syscall
  test rax, rax
  jle read_done
  add r12, rax
  jmp read_loop
read_done:
  lea r15, [rip + buffer]
  xor r13, r13
  xor r14, r14
  xor rcx, rcx
parse_loop:
  cmp rcx, r12
  jae parse_done
  movzx eax, byte ptr [r15 + rcx]
  cmp al, 10
  jne parse_next
  lea r8, [rip + line_off]
  mov dword ptr [r8 + r13*4], r14d
  mov eax, ecx
  sub eax, r14d
  lea r8, [rip + line_len]
  mov dword ptr [r8 + r13*4], eax
  lea r8, [rip + order]
  mov dword ptr [r8 + r13*4], r13d
  inc r13
  mov r14, rcx
  inc r14
parse_next:
  inc rcx
  jmp parse_loop
parse_done:
  cmp r14, r12
  jae sort_begin
  lea r8, [rip + line_off]
  mov dword ptr [r8 + r13*4], r14d
  mov eax, r12d
  sub eax, r14d
  lea r8, [rip + line_len]
  mov dword ptr [r8 + r13*4], eax
  lea r8, [rip + order]
  mov dword ptr [r8 + r13*4], r13d
  inc r13
sort_begin:
  cmp r13, 1
  jbe print_begin
  mov r12, 1
sort_outer:
  cmp r12, r13
  jae print_begin
  lea rbx, [rip + order]
  mov r14d, dword ptr [rbx + r12*4]
  mov r15, r12
  dec r15
sort_inner:
  lea rbx, [rip + order]
  mov edi, dword ptr [rbx + r15*4]
  mov esi, r14d
  call cmp_lines
  test eax, eax
  jle sort_place
  lea rbx, [rip + order]
  mov eax, dword ptr [rbx + r15*4]
  lea rcx, [r15 + 1]
  mov dword ptr [rbx + rcx*4], eax
  test r15, r15
  jz sort_place0
  dec r15
  jmp sort_inner
sort_place0:
  lea rbx, [rip + order]
  mov dword ptr [rbx], r14d
  jmp sort_next
sort_place:
  lea rbx, [rip + order]
  lea rcx, [r15 + 1]
  mov dword ptr [rbx + rcx*4], r14d
sort_next:
  inc r12
  jmp sort_outer
print_begin:
  xor r12, r12
print_loop:
  cmp r12, r13
  jae do_exit
  lea r8, [rip + order]
  mov eax, dword ptr [r8 + r12*4]
  lea r8, [rip + line_off]
  mov r14d, dword ptr [r8 + rax*4]
  lea r8, [rip + line_len]
  mov r15d, dword ptr [r8 + rax*4]
  test r15d, r15d
  jz print_nl
  lea rsi, [rip + buffer]
  add rsi, r14
  mov eax, 1
  mov edi, 1
  mov rdx, r15
  syscall
print_nl:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + nl]
  mov edx, 1
  syscall
  inc r12
  jmp print_loop
do_exit:
  mov eax, 60
  xor edi, edi
  syscall
  jmp do_exit

.data
nl:
  .byte 10

.bss
.balign 16
buffer:
  .zero 32768
line_off:
  .zero 4000
line_len:
  .zero 4000
order:
  .zero 4000
