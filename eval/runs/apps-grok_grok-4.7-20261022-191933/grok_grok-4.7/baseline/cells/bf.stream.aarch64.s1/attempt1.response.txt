.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536
.equ CH_BANG, 33
.equ CH_PLUS, 43
.equ CH_COMMA, 44
.equ CH_MINUS, 45
.equ CH_DOT, 46
.equ CH_LT, 60
.equ CH_GT, 62
.equ CH_LBR, 91
.equ CH_RBR, 93
.equ OP_PLUS, 1
.equ OP_MINUS, 2
.equ OP_GT, 3
.equ OP_LT, 4
.equ OP_DOT, 5
.equ OP_COMMA, 6
.equ OP_LBR, 7
.equ OP_RBR, 8

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #IN_MAX
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    cmp x20, #IN_MAX
    b.lo read_loop

read_done:
    mov x21, #0
    b find_cond

find_loop:
    ldrb w0, [x19, x21]
    cmp w0, #CH_BANG
    b.eq found_bang
    add x21, x21, #1

find_cond:
    cmp x21, x20
    b.lo find_loop
    mov x23, #0
    b split_done

found_bang:
    add x22, x19, x21
    add x22, x22, #1
    sub x23, x20, x21
    sub x23, x23, #1

split_done:
    adrp x27, code
    add x27, x27, :lo12:code
    adrp x29, targets
    add x29, x29, :lo12:targets
    adrp x10, brstack
    add x10, x10, :lo12:brstack
    mov x12, #0
    mov x13, #0
    mov x11, #0
    b c_cond

c_loop:
    ldrb w0, [x19, x12]
    cmp w0, #CH_PLUS
    b.lo c_next
    cmp w0, #CH_RBR
    b.hi c_next
    cmp w0, #CH_PLUS
    b.eq c_plus
    cmp w0, #CH_COMMA
    b.eq c_comma
    cmp w0, #CH_MINUS
    b.eq c_minus
    cmp w0, #CH_DOT
    b.eq c_dot
    cmp w0, #CH_LT
    b.eq c_lt
    cmp w0, #CH_GT
    b.eq c_gt
    cmp w0, #CH_LBR
    b.eq c_lbr
    cmp w0, #CH_RBR
    b.eq c_rbr

c_next:
    add x12, x12, #1

c_cond:
    cmp x12, x21
    b.lo c_loop
    cbnz x11, bad

    adrp x9, dtab
    add x9, x9, :lo12:dtab
    adr x1, do_nop
    str x1, [x9, #0]
    adr x1, do_plus
    str x1, [x9, #8]
    adr x1, do_minus
    str x1, [x9, #16]
    adr x1, do_gt
    str x1, [x9, #24]
    adr x1, do_lt
    str x1, [x9, #32]
    adr x1, do_dot
    str x1, [x9, #40]
    adr x1, do_comma
    str x1, [x9, #48]
    adr x1, do_lbr
    str x1, [x9, #56]
    adr x1, do_rbr
    str x1, [x9, #64]

    adrp x24, tape
    add x24, x24, :lo12:tape
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
    mov x25, #0
    mov x26, #0

run_loop:
    cmp x26, x13
    b.hs write_out
    ldrb w0, [x27, x26]
    ldr x1, [x9, x0, lsl #3]
    br x1

write_out:
    cbz x25, exit_ok

write_loop:
    mov x0, #1
    mov x1, x28
    mov x2, x25
    mov x8, #SYS_write
    svc #0
    add x28, x28, x0
    sub x25, x25, x0
    cbnz x25, write_loop

exit_ok:
    mov x0, #0
    b do_exit

bad:
    mov x0, #1

do_exit:
    mov x8, #SYS_exit
    svc #0
    b do_exit

c_plus:
    mov w0, #OP_PLUS
    b c_emit

c_comma:
    mov w0, #OP_COMMA
    b c_emit

c_minus:
    mov w0, #OP_MINUS
    b c_emit

c_dot:
    mov w0, #OP_DOT
    b c_emit

c_lt:
    mov w0, #OP_LT
    b c_emit

c_gt:
    mov w0, #OP_GT
    b c_emit

c_emit:
    strb w0, [x27, x13]
    add x13, x13, #1
    b c_next

c_lbr:
    mov w0, #OP_LBR
    strb w0, [x27, x13]
    str w13, [x10, x11, lsl #2]
    add x11, x11, #1
    add x13, x13, #1
    b c_next

c_rbr:
    cbz x11, bad
    sub x11, x11, #1
    ldr w0, [x10, x11, lsl #2]
    add w1, w13, #1
    str w1, [x29, x0, lsl #2]
    add w1, w0, #1
    str w1, [x29, x13, lsl #2]
    mov w0, #OP_RBR
    strb w0, [x27, x13]
    add x13, x13, #1
    b c_next

do_nop:
    add x26, x26, #1
    b run_loop

do_plus:
    ldrb w1, [x24]
    add w1, w1, #1
    strb w1, [x24]
    add x26, x26, #1
    b run_loop

do_minus:
    ldrb w1, [x24]
    sub w1, w1, #1
    strb w1, [x24]
    add x26, x26, #1
    b run_loop

do_gt:
    add x24, x24, #1
    add x26, x26, #1
    b run_loop

do_lt:
    sub x24, x24, #1
    add x26, x26, #1
    b run_loop

do_dot:
    ldrb w1, [x24]
    strb w1, [x28, x25]
    add x25, x25, #1
    add x26, x26, #1
    b run_loop

do_comma:
    cbz x23, comma_eof
    ldrb w1, [x22], #1
    sub x23, x23, #1
    strb w1, [x24]
    add x26, x26, #1
    b run_loop

comma_eof:
    strb wzr, [x24]
    add x26, x26, #1
    b run_loop

do_lbr:
    ldrb w1, [x24]
    cbz w1, lbr_jump
    add x26, x26, #1
    b run_loop

lbr_jump:
    ldr w26, [x29, x26, lsl #2]
    b run_loop

do_rbr:
    ldrb w1, [x24]
    cbz w1, rbr_next
    ldr w26, [x29, x26, lsl #2]
    b run_loop

rbr_next:
    add x26, x26, #1
    b run_loop

.bss
.align 3
dtab:
    .zero 72
.align 3
targets:
    .zero 262144
.align 3
brstack:
    .zero 262144
code:
    .zero 65536
inbuf:
    .zero 65536
outbuf:
    .zero 65536
tape:
    .zero 30000
