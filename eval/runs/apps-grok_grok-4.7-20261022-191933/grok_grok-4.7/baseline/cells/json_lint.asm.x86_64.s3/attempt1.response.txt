.globl _start

.text

_start:
  cld
  lea r15, [rip + input_buf]
  xor r14, r14
read_more:
  xor eax, eax
  xor edi, edi
  lea rsi, [r15 + r14]
  mov rdx, 65536
  sub rdx, r14
  syscall
  test rax, rax
  jle read_done
  add r14, rax
  cmp r14, 65536
  jb read_more
read_done:
  mov r13, r15
  mov rbx, r15
  lea r12, [r15 + r14]
  call skip_ws
  call parse_value
  call skip_ws
  cmp rbx, r12
  jne error_here
print_ok:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip + msg_ok]
  mov edx, 3
  syscall
  mov eax, 60
  xor edi, edi
  syscall
  jmp print_ok

error_here:
  mov rax, rbx
  sub rax, r13
  mov r10, rax
  lea rdi, [rip + outbuf]
  lea rsi, [rip + err_prefix]
  mov ecx, 9
copy_prefix:
  mov al, byte ptr [rsi]
  mov byte ptr [rdi], al
  inc rsi
  inc rdi
  dec ecx
  jnz copy_prefix
  mov rax, r10
  lea r8, [rip + outbuf]
  add r8, 48
  mov r9, r8
  test rax, rax
  jnz div_loop
  dec r8
  mov byte ptr [r8], 0x30
  jmp copy_digits
div_loop:
  xor edx, edx
  mov ecx, 10
  div rcx
  add dl, 0x30
  dec r8
  mov byte ptr [r8], dl
  test rax, rax
  jnz div_loop
copy_digits:
  mov rsi, r8
copy_digits_loop:
  cmp rsi, r9
  jae write_nl
  mov al, byte ptr [rsi]
  mov byte ptr [rdi], al
  inc rsi
  inc rdi
  jmp copy_digits_loop
write_nl:
  mov byte ptr [rdi], 0x0A
  inc rdi
  lea rsi, [rip + outbuf]
  mov rdx, rdi
  sub rdx, rsi
  mov eax, 1
  mov edi, 1
  syscall
  mov eax, 60
  mov edi, 1
  syscall
  jmp error_here

skip_ws:
  cmp rbx, r12
  jae skip_ws_done
  movzx eax, byte ptr [rbx]
  cmp al, 0x20
  je skip_ws_yes
  cmp al, 0x09
  je skip_ws_yes
  cmp al, 0x0A
  je skip_ws_yes
  cmp al, 0x0D
  je skip_ws_yes
  ret
skip_ws_yes:
  inc rbx
  jmp skip_ws
skip_ws_done:
  ret

parse_value:
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x7B
  je parse_object
  cmp al, 0x5B
  je parse_array
  cmp al, 0x22
  je parse_string
  cmp al, 0x74
  je parse_true
  cmp al, 0x66
  je parse_false
  cmp al, 0x6E
  je parse_null
  cmp al, 0x2D
  je parse_number
  cmp al, 0x30
  jb error_here
  cmp al, 0x39
  jbe parse_number
  jmp error_here

parse_true:
  lea rsi, [rip + lit_true]
  mov edx, 4
  jmp expect_lit

parse_false:
  lea rsi, [rip + lit_false]
  mov edx, 5
  jmp expect_lit

parse_null:
  lea rsi, [rip + lit_null]
  mov edx, 4
  jmp expect_lit

expect_lit:
  cmp rbx, r12
  jae error_here
  mov al, byte ptr [rsi]
  cmp al, byte ptr [rbx]
  jne error_here
  inc rsi
  inc rbx
  dec edx
  jnz expect_lit
  ret

parse_string:
  inc rbx
string_loop:
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x22
  je string_end
  cmp al, 0x5C
  je string_esc
  cmp al, 0x20
  jb error_here
  inc rbx
  jmp string_loop
string_end:
  inc rbx
  ret
string_esc:
  inc rbx
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x22
  je string_esc_ok
  cmp al, 0x5C
  je string_esc_ok
  cmp al, 0x2F
  je string_esc_ok
  cmp al, 0x62
  je string_esc_ok
  cmp al, 0x66
  je string_esc_ok
  cmp al, 0x6E
  je string_esc_ok
  cmp al, 0x72
  je string_esc_ok
  cmp al, 0x74
  je string_esc_ok
  cmp al, 0x75
  je string_u
  jmp error_here
string_esc_ok:
  inc rbx
  jmp string_loop
string_u:
  inc rbx
  mov edx, 4
string_u_loop:
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x30
  jb error_here
  cmp al, 0x39
  jbe string_hex_ok
  cmp al, 0x41
  jb error_here
  cmp al, 0x46
  jbe string_hex_ok
  cmp al, 0x61
  jb error_here
  cmp al, 0x66
  ja error_here
string_hex_ok:
  inc rbx
  dec edx
  jnz string_u_loop
  jmp string_loop

parse_number:
  movzx eax, byte ptr [rbx]
  cmp al, 0x2D
  jne number_int
  inc rbx
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
number_int:
  cmp al, 0x30
  je number_zero
  cmp al, 0x31
  jb error_here
  cmp al, 0x39
  ja error_here
number_digs:
  inc rbx
  cmp rbx, r12
  jae number_done
  movzx eax, byte ptr [rbx]
  cmp al, 0x30
  jb number_frac
  cmp al, 0x39
  jbe number_digs
  jmp number_frac
number_zero:
  inc rbx
  cmp rbx, r12
  jae number_done
  movzx eax, byte ptr [rbx]
number_frac:
  cmp al, 0x2E
  jne number_exp
  inc rbx
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x30
  jb error_here
  cmp al, 0x39
  ja error_here
number_frac_digs:
  inc rbx
  cmp rbx, r12
  jae number_done
  movzx eax, byte ptr [rbx]
  cmp al, 0x30
  jb number_exp
  cmp al, 0x39
  ja number_exp
  jmp number_frac_digs
number_exp:
  cmp al, 0x65
  je number_exp_go
  cmp al, 0x45
  jne number_done
number_exp_go:
  inc rbx
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x2B
  je number_exp_sign
  cmp al, 0x2D
  jne number_exp_first
number_exp_sign:
  inc rbx
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
number_exp_first:
  cmp al, 0x30
  jb error_here
  cmp al, 0x39
  ja error_here
number_exp_digs:
  inc rbx
  cmp rbx, r12
  jae number_done
  movzx eax, byte ptr [rbx]
  cmp al, 0x30
  jb number_done
  cmp al, 0x39
  ja number_done
  jmp number_exp_digs
number_done:
  ret

parse_object:
  inc rbx
  call skip_ws
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x7D
  je object_close
object_member:
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x22
  jne error_here
  call parse_string
  call skip_ws
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x3A
  jne error_here
  inc rbx
  call skip_ws
  call parse_value
  call skip_ws
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x7D
  je object_close
  cmp al, 0x2C
  jne error_here
  inc rbx
  call skip_ws
  jmp object_member
object_close:
  inc rbx
  ret

parse_array:
  inc rbx
  call skip_ws
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x5D
  je array_close
array_value:
  call parse_value
  call skip_ws
  cmp rbx, r12
  jae error_here
  movzx eax, byte ptr [rbx]
  cmp al, 0x5D
  je array_close
  cmp al, 0x2C
  jne error_here
  inc rbx
  call skip_ws
  jmp array_value
array_close:
  inc rbx
  ret

.section .rodata
msg_ok:
  .ascii "ok\n"
err_prefix:
  .ascii "error at "
lit_true:
  .ascii "true"
lit_false:
  .ascii "false"
lit_null:
  .ascii "null"

.bss
input_buf:
  .zero 65536
outbuf:
  .zero 64
