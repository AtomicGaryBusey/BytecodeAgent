.globl _start

.section .rodata
newline:
    .byte 10

.bss
.align 8
input_buf:
    .zero 32768
line_off:
    .zero 8000
line_len:
    .zero 8000

.text
_start:
    xor rbx, rbx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, 32768
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, 32768
    jb read_loop
read_done:
    xor r12, r12
    xor rcx, rcx
    xor r8, r8
    lea r9, [rip + input_buf]
    lea r10, [rip + line_off]
    lea r11, [rip + line_len]
parse_loop:
    cmp rcx, rbx
    jae parse_done
    movzx eax, byte ptr [r9 + rcx]
    cmp al, 10
    jne parse_next
    mov qword ptr [r10 + r12*8], r8
    mov rax, rcx
    sub rax, r8
    mov qword ptr [r11 + r12*8], rax
    inc r12
    lea r8, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_done:
    cmp r8, rbx
    jae sort_begin
    mov qword ptr [r10 + r12*8], r8
    mov rax, rbx
    sub rax, r8
    mov qword ptr [r11 + r12*8], rax
    inc r12
sort_begin:
    cmp r12, 2
    jb print_begin
    mov r13, r12
    shr r13, 1
heap_build:
    test r13, r13
    jz heap_extract
    dec r13
    mov rdi, r13
    mov rsi, r12
    call sift_down
    jmp heap_build
heap_extract:
    mov r13, r12
heap_ex_loop:
    cmp r13, 2
    jb print_begin
    dec r13
    xor edi, edi
    mov rsi, r13
    call swap_idx
    xor edi, edi
    mov rsi, r13
    call sift_down
    jmp heap_ex_loop
print_begin:
    xor r13, r13
print_loop:
    cmp r13, r12
    jae exit_ok
    lea rax, [rip + line_off]
    mov rsi, qword ptr [rax + r13*8]
    lea rax, [rip + input_buf]
    add rsi, rax
    lea rax, [rip + line_len]
    mov rdx, qword ptr [rax + r13*8]
    test rdx, rdx
    jz print_nl
    mov eax, 1
    mov edi, 1
    syscall
print_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r13
    jmp print_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

sift_down:
    mov r14, rdi
    mov rbx, rsi
sift_loop:
    mov r15, r14
    shl r15, 1
    inc r15
    cmp r15, rbx
    jae sift_ret
    lea rcx, [r15 + 1]
    cmp rcx, rbx
    jae sift_pick
    push rcx
    mov rdi, r15
    mov rsi, rcx
    call compare_idx
    pop rcx
    cmp eax, 0
    jge sift_pick
    mov r15, rcx
sift_pick:
    mov rdi, r14
    mov rsi, r15
    call compare_idx
    cmp eax, 0
    jge sift_ret
    mov rdi, r14
    mov rsi, r15
    call swap_idx
    mov r14, r15
    jmp sift_loop
sift_ret:
    ret

compare_idx:
    lea rax, [rip + line_off]
    mov r8, qword ptr [rax + rdi*8]
    mov r9, qword ptr [rax + rsi*8]
    lea rax, [rip + line_len]
    mov r10, qword ptr [rax + rdi*8]
    mov rcx, qword ptr [rax + rsi*8]
    lea rax, [rip + input_buf]
    add r8, rax
    add r9, rax
    mov rax, r10
    cmp rax, rcx
    cmova rax, rcx
    xor edx, edx
cmp_loop:
    cmp rdx, rax
    jae cmp_len
    movzx r11d, byte ptr [r8 + rdx]
    cmp r11b, byte ptr [r9 + rdx]
    ja cmp_gt
    jb cmp_lt
    inc rdx
    jmp cmp_loop
cmp_len:
    cmp r10, rcx
    ja cmp_gt
    jb cmp_lt
    xor eax, eax
    ret
cmp_gt:
    mov eax, 1
    ret
cmp_lt:
    mov eax, -1
    ret

swap_idx:
    lea rax, [rip + line_off]
    mov r8, qword ptr [rax + rdi*8]
    mov r9, qword ptr [rax + rsi*8]
    mov qword ptr [rax + rdi*8], r9
    mov qword ptr [rax + rsi*8], r8
    lea rax, [rip + line_len]
    mov r8, qword ptr [rax + rdi*8]
    mov r9, qword ptr [rax + rsi*8]
    mov qword ptr [rax + rdi*8], r9
    mov qword ptr [rax + rsi*8], r8
    ret
