.section .rodata

msg_ok:
  .ascii "ok"
  .byte 0x0A

msg_err:
  .ascii "error at "

lit_true:
  .ascii "true"

lit_false:
  .ascii "false"

lit_null:
  .ascii "null"

.bss
.balign 16

inbuf:
  .zero 65536

numbuf:
  .zero 32

.text
.balign 4
.globl _start

_start:
  adrp x19, inbuf
  add x19, x19, :lo12:inbuf
  mov x20, #0
read_loop:
  movz x2, #1, lsl #16
  sub x2, x2, x20
  cbz x2, read_done
  mov x0, #0
  add x1, x19, x20
  mov x8, #63
  svc #0
  cbz x0, read_done
  add x20, x20, x0
  b read_loop
read_done:
  add x20, x19, x20
  mov x21, x19
  bl skip_ws
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.ne fail
  mov x0, #1
  adrp x1, msg_ok
  add x1, x1, :lo12:msg_ok
  mov x2, #3
  mov x8, #64
  svc #0
  mov x0, #0
  mov x8, #93
  svc #0

skip_ws:
sw_loop:
  cmp x21, x20
  b.hs sw_ret
  ldrb w0, [x21]
  cmp w0, #0x20
  b.eq sw_adv
  cmp w0, #0x09
  b.eq sw_adv
  cmp w0, #0x0A
  b.eq sw_adv
  cmp w0, #0x0D
  b.eq sw_adv
sw_ret:
  ret
sw_adv:
  add x21, x21, #1
  b sw_loop

parse_value:
  stp x29, x30, [sp, #-16]!
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #'{'
  b.eq pv_obj
  cmp w0, #'['
  b.eq pv_arr
  cmp w0, #0x22
  b.eq pv_str
  cmp w0, #'t'
  b.eq pv_true
  cmp w0, #'f'
  b.eq pv_false
  cmp w0, #'n'
  b.eq pv_null
  cmp w0, #'-'
  b.eq pv_num
  cmp w0, #'0'
  b.lo fail
  cmp w0, #'9'
  b.hi fail
pv_num:
  bl parse_number
  b pv_done
pv_obj:
  bl parse_object
  b pv_done
pv_arr:
  bl parse_array
  b pv_done
pv_str:
  bl parse_string
  b pv_done
pv_true:
  bl parse_true
  b pv_done
pv_false:
  bl parse_false
  b pv_done
pv_null:
  bl parse_null
pv_done:
  ldp x29, x30, [sp], #16
  ret

parse_object:
  stp x29, x30, [sp, #-16]!
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #'}'
  b.eq po_close
po_mem:
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #0x22
  b.ne fail
  bl parse_string
  bl skip_ws
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #':'
  b.ne fail
  add x21, x21, #1
  bl skip_ws
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #','
  b.eq po_comma
  cmp w0, #'}'
  b.ne fail
po_close:
  add x21, x21, #1
  ldp x29, x30, [sp], #16
  ret
po_comma:
  add x21, x21, #1
  bl skip_ws
  b po_mem

parse_array:
  stp x29, x30, [sp, #-16]!
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #']'
  b.eq pa_close
pa_val:
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #','
  b.eq pa_comma
  cmp w0, #']'
  b.ne fail
pa_close:
  add x21, x21, #1
  ldp x29, x30, [sp], #16
  ret
pa_comma:
  add x21, x21, #1
  bl skip_ws
  b pa_val

parse_string:
  add x21, x21, #1
ps_loop:
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #0x22
  b.eq ps_end
  cmp w0, #0x5C
  b.eq ps_esc
  cmp w0, #0x20
  b.lo fail
  add x21, x21, #1
  b ps_loop
ps_end:
  add x21, x21, #1
  ret
ps_esc:
  add x21, x21, #1
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #'u'
  b.eq ps_u
  cmp w0, #0x22
  b.eq ps_ok
  cmp w0, #0x5C
  b.eq ps_ok
  cmp w0, #0x2F
  b.eq ps_ok
  cmp w0, #'b'
  b.eq ps_ok
  cmp w0, #'f'
  b.eq ps_ok
  cmp w0, #'n'
  b.eq ps_ok
  cmp w0, #'r'
  b.eq ps_ok
  cmp w0, #'t'
  b.eq ps_ok
  b fail
ps_ok:
  add x21, x21, #1
  b ps_loop
ps_u:
  add x21, x21, #1
  mov w1, #4
ps_hex:
  cbz w1, ps_loop
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #'0'
  b.lo fail
  cmp w0, #'9'
  b.ls ps_hexok
  cmp w0, #'A'
  b.lo fail
  cmp w0, #'F'
  b.ls ps_hexok
  cmp w0, #'a'
  b.lo fail
  cmp w0, #'f'
  b.hi fail
ps_hexok:
  add x21, x21, #1
  sub w1, w1, #1
  b ps_hex

parse_number:
  ldrb w0, [x21]
  cmp w0, #'-'
  b.ne pn_int
  add x21, x21, #1
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
pn_int:
  cmp w0, #'0'
  b.eq pn_zero
  cmp w0, #'1'
  b.lo fail
  cmp w0, #'9'
  b.hi fail
pn_digits:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_ret
  ldrb w0, [x21]
  cmp w0, #'0'
  b.lo pn_tail
  cmp w0, #'9'
  b.hi pn_tail
  b pn_digits
pn_zero:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_ret
  ldrb w0, [x21]
pn_tail:
  cmp w0, #'.'
  b.eq pn_frac
  cmp w0, #'e'
  b.eq pn_exp
  cmp w0, #'E'
  b.eq pn_exp
pn_ret:
  ret
pn_frac:
  add x21, x21, #1
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #'0'
  b.lo fail
  cmp w0, #'9'
  b.hi fail
pn_fracdig:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_ret
  ldrb w0, [x21]
  cmp w0, #'0'
  b.lo pn_afterfrac
  cmp w0, #'9'
  b.hi pn_afterfrac
  b pn_fracdig
pn_afterfrac:
  cmp w0, #'e'
  b.eq pn_exp
  cmp w0, #'E'
  b.eq pn_exp
  ret
pn_exp:
  add x21, x21, #1
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  cmp w0, #'+'
  b.eq pn_sign
  cmp w0, #'-'
  b.eq pn_sign
  b pn_exp1
pn_sign:
  add x21, x21, #1
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
pn_exp1:
  cmp w0, #'0'
  b.lo fail
  cmp w0, #'9'
  b.hi fail
pn_expdigs:
  add x21, x21, #1
  cmp x21, x20
  b.hs pn_ret
  ldrb w0, [x21]
  cmp w0, #'0'
  b.lo pn_ret
  cmp w0, #'9'
  b.hi pn_ret
  b pn_expdigs

parse_true:
  adrp x1, lit_true
  add x1, x1, :lo12:lit_true
  mov w2, #4
  b match_lit

parse_false:
  adrp x1, lit_false
  add x1, x1, :lo12:lit_false
  mov w2, #5
  b match_lit

parse_null:
  adrp x1, lit_null
  add x1, x1, :lo12:lit_null
  mov w2, #4
  b match_lit

match_lit:
ml_loop:
  cbz w2, ml_done
  cmp x21, x20
  b.hs fail
  ldrb w0, [x21]
  ldrb w3, [x1]
  cmp w0, w3
  b.ne fail
  add x21, x21, #1
  add x1, x1, #1
  sub w2, w2, #1
  b ml_loop
ml_done:
  ret

fail:
  sub x0, x21, x19
  adrp x10, numbuf
  add x10, x10, :lo12:numbuf
  mov x11, x10
  adrp x1, msg_err
  add x1, x1, :lo12:msg_err
  mov w2, #9
fail_cpy:
  ldrb w3, [x1], #1
  strb w3, [x11], #1
  subs w2, w2, #1
  b.ne fail_cpy
  add x12, x10, #31
  mov x13, x12
  mov x14, #10
fail_dig:
  udiv x15, x0, x14
  msub x9, x15, x14, x0
  add w9, w9, #'0'
  sub x13, x13, #1
  strb w9, [x13]
  mov x0, x15
  cbnz x0, fail_dig
fail_out:
  cmp x13, x12
  b.hs fail_nl
  ldrb w3, [x13], #1
  strb w3, [x11], #1
  b fail_out
fail_nl:
  mov w3, #0x0A
  strb w3, [x11], #1
  sub x2, x11, x10
  mov x0, #1
  mov x1, x10
  mov x8, #64
  svc #0
  mov x0, #1
  mov x8, #93
  svc #0
