.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_MAX, 65536

.globl _start
.text
_start:
    sub rsp, 8
    lea r9, [rip + hexdigits]
    lea r12, [rip + inbuf]
    xor ebx, ebx

read_more:
    mov edx, IN_MAX
    sub edx, ebx
    jz all_read
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle all_read
    add rbx, rax
    jmp read_more

all_read:
    xor r13d, r13d

row_loop:
    cmp r13, rbx
    jae final_line
    lea r10, [r12 + r13]
    mov r14, rbx
    sub r14, r13
    cmp r14, 16
    jbe n_ready
    mov r14, 16
n_ready:
    lea rdi, [rip + linebuf]
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2
    xor r15d, r15d

slot_loop:
    cmp r15, r14
    jae pad_slot
    movzx eax, byte ptr [r10 + r15]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    movzx eax, byte ptr [r9 + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp slot_gap
pad_slot:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
slot_gap:
    cmp r15d, 7
    jne slot_next
    mov byte ptr [rdi], 0x20
    inc rdi
slot_next:
    inc r15d
    cmp r15d, 16
    jb slot_loop

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi
    xor r15d, r15d

char_loop:
    cmp r15, r14
    jae char_end
    movzx eax, byte ptr [r10 + r15]
    cmp al, 0x20
    jb char_dot
    cmp al, 0x7e
    ja char_dot
    mov byte ptr [rdi], al
    jmp char_store
char_dot:
    mov byte ptr [rdi], 0x2e
char_store:
    inc rdi
    inc r15d
    jmp char_loop

char_end:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    call write_stdout
    add r13, r14
    jmp row_loop

final_line:
    lea rdi, [rip + linebuf]
    mov eax, ebx
    call hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    call write_stdout
    mov eax, SYS_exit
    xor edi, edi
    syscall

hex8:
    mov r8d, 7
hex8_digit:
    mov edx, eax
    mov ecx, r8d
    shl ecx, 2
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec r8d
    jns hex8_digit
    ret

write_stdout:
    mov eax, SYS_write
    mov edi, 1
    syscall
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 65536
linebuf:
    .zero 128
