.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 49152

.globl _start

.section .rodata
.align 3
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 3
inbuf:
    .zero 49152
outbuf:
    .zero 66560

.text
.align 2
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #IN_MAX
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_ok
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, x21
    mov x23, #0
    adrp x24, alpha
    add x24, x24, :lo12:alpha
    mov x25, x19
    mov x26, x20

enc_loop:
    cmp x26, #3
    b.lo enc_tail
    ldrb w0, [x25]
    ldrb w1, [x25, #1]
    ldrb w2, [x25, #2]
    add x25, x25, #3
    sub x26, x26, #3

    lsr w3, w0, #2
    bl emit_index

    and w3, w0, #3
    lsl w3, w3, #4
    lsr w4, w1, #4
    orr w3, w3, w4
    bl emit_index

    and w3, w1, #15
    lsl w3, w3, #2
    lsr w4, w2, #6
    orr w3, w3, w4
    bl emit_index

    and w3, w2, #63
    bl emit_index
    b enc_loop

enc_tail:
    cbz x26, enc_finish
    cmp x26, #1
    b.eq enc_one

    ldrb w0, [x25]
    ldrb w1, [x25, #1]

    lsr w3, w0, #2
    bl emit_index

    and w3, w0, #3
    lsl w3, w3, #4
    lsr w4, w1, #4
    orr w3, w3, w4
    bl emit_index

    and w3, w1, #15
    lsl w3, w3, #2
    bl emit_index

    mov w3, #61
    bl emit_byte
    b enc_finish

enc_one:
    ldrb w0, [x25]

    lsr w3, w0, #2
    bl emit_index

    and w3, w0, #3
    lsl w3, w3, #4
    bl emit_index

    mov w3, #61
    bl emit_byte
    mov w3, #61
    bl emit_byte

enc_finish:
    cbz x23, do_write
    mov w3, #10
    strb w3, [x22], #1

do_write:
    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #SYS_write
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

emit_index:
    ldrb w3, [x24, x3]

emit_byte:
    strb w3, [x22], #1
    add x23, x23, #1
    cmp x23, #76
    b.ne 1f
    mov w4, #10
    strb w4, [x22], #1
    mov x23, #0
1:
    ret
