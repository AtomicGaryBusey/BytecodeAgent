.globl _start

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537

.text
_start:
    xor rbx, rbx

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, 65536
    jb read_more

read_done:
    test rbx, rbx
    jz do_exit
    lea rsi, [rip + inbuf]
    lea rdi, [rsi + rbx]
    lea r8, [rip + outbuf]

next_line:
    cmp rsi, rdi
    jae write_out
    mov rcx, rsi

find_nl:
    cmp rcx, rdi
    jae have_end
    cmp byte ptr [rcx], 10
    je have_end
    inc rcx
    jmp find_nl

have_end:
    mov rdx, rcx

rev_loop:
    cmp rdx, rsi
    je rev_done
    dec rdx
    mov al, byte ptr [rdx]
    mov byte ptr [r8], al
    inc r8
    jmp rev_loop

rev_done:
    mov byte ptr [r8], 10
    inc r8
    cmp rcx, rdi
    jae write_out
    inc rcx
    mov rsi, rcx
    jmp next_line

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, r8
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
