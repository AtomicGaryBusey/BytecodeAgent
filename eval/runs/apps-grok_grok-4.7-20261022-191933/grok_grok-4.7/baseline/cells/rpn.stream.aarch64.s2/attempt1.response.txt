.globl _start

.section .rodata

    .ascii "error"

    .ascii "-9223372036854775808"
    .byte 0x0a

    .zero 65536

    .zero 8192

.text
.balign 4

    sub x2, x2, x20

    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0

    add x20, x20, x0
    b read_more

parse_lines:
    mov x21, #0

    cmp x21, x20
    b.hs do_exit
    mov x22, x21

    b.hs last_line
    ldrb w0, [x19, x22]

    add x22, x22, #1

    b next_line

    mov x0, #0

    b.eq skip_ws
    cmp w0, #0x09

    mov x25, x24

    cmp x24, x22
    b.hs tok_end
    ldrb w0, [x19, x24]
    cmp w0, #0x20

    cmp w0, #0x09

    cmp x6, #1
    b.ne parse_num
    ldrb w0, [x19, x25]
    cmp w0, #0x2b
    b.eq op_add
    cmp w0, #0x2d
    b.eq op_sub
    cmp w0, #0x2a
    b.eq op_mul
    cmp w0, #0x2f
    b.eq op_div
    cmp w0, #0x25
    b.eq op_mod

    mov x5, x25
    ldrb w0, [x19, x25]
    cmp w0, #0x2d
    b.ne check_digits

    add x5, x5, #1

    cbz x6, line_error
    cmp x6, #18

    mov x1, #0
    mov x2, #10

    b.hs digits_done
    ldrb w3, [x19, x5]
    sub w4, w3, #0x30
    cmp w4, #9
    b.hi line_error

    add x1, x1, x4

    b.hs line_error
    str x1, [x28, x23, lsl #3]

    sub x23, x23, #1
    ldr x1, [x28, x23, lsl #3]

    sub x23, x23, #1
    ldr x1, [x28, x23, lsl #3]
    sub x2, x23, #1
