.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.globl _start

.text
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x19, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x21, x19
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0
    mov x23, #0

line_loop:
    cmp x20, x19
    b.hs write_out
    mov x22, x20

scan_loop:
    cmp x22, x19
    b.hs have_line
    ldrb w24, [x21, x22]
    cmp w24, #10
    b.eq have_line
    add x22, x22, #1
    b scan_loop

have_line:
    mov x24, x22

copy_rev:
    cmp x24, x20
    b.eq store_nl
    sub x24, x24, #1
    ldrb w25, [x21, x24]
    strb w25, [x26, x23]
    add x23, x23, #1
    b copy_rev

store_nl:
    mov w25, #10
    strb w25, [x26, x23]
    add x23, x23, #1
    cmp x22, x19
    b.hs write_out
    add x20, x22, #1
    b line_loop

write_out:
    cbz x23, exit_ok
    mov x1, x26
    mov x2, x23

write_loop:
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le exit_ok
    cmp x0, x2
    b.hs exit_ok
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.bss
.balign 8
input_buf:
    .zero 65536
output_buf:
    .zero 65537
