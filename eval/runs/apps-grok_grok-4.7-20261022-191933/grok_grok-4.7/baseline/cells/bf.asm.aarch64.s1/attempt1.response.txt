.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_MAX, 65536
.equ TAPE_SIZE, 30000
.equ MAX_DEPTH, 1000
.equ JTAB_SIZE, 262144
.equ BSTACK_SIZE, 4000
.equ DISP_SIZE, 2048
.equ C_GT, 62
.equ C_LT, 60
.equ C_PLUS, 43
.equ C_MINUS, 45
.equ C_DOT, 46
.equ C_COMMA, 44
.equ C_LBR, 91
.equ C_RBR, 93
.equ C_BANG, 33

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #IN_MAX
read_loop:
    cmp x20, x21
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x4, #0
find_loop:
    cmp x4, x20
    b.eq find_nobang
    ldrb w0, [x19, x4]
    cmp w0, #C_BANG
    b.eq find_bang
    add x4, x4, #1
    b find_loop
find_nobang:
    mov x21, x20
    mov x22, x20
    b find_done
find_bang:
    mov x21, x4
    add x22, x4, #1
find_done:
    adrp x24, bstack
    add x24, x24, :lo12:bstack
    adrp x26, jumptab
    add x26, x26, :lo12:jumptab
    mov x4, #0
    mov x5, #0
    mov x6, #0
pass_loop:
    cmp x4, x21
    b.eq pass_done
    ldrb w0, [x19, x4]
    add x4, x4, #1
    cmp w0, #C_GT
    b.eq pass_keep
    cmp w0, #C_LT
    b.eq pass_keep
    cmp w0, #C_PLUS
    b.eq pass_keep
    cmp w0, #C_MINUS
    b.eq pass_keep
    cmp w0, #C_DOT
    b.eq pass_keep
    cmp w0, #C_COMMA
    b.eq pass_keep
    cmp w0, #C_LBR
    b.eq pass_lbr
    cmp w0, #C_RBR
    b.eq pass_rbr
    b pass_loop
pass_keep:
    strb w0, [x19, x5]
    add x5, x5, #1
    b pass_loop
pass_lbr:
    cmp x6, #MAX_DEPTH
    b.hs bad_brackets
    strb w0, [x19, x5]
    str w5, [x24, x6, lsl #2]
    add x6, x6, #1
    add x5, x5, #1
    b pass_loop
pass_rbr:
    cbz x6, bad_brackets
    sub x6, x6, #1
    ldr w1, [x24, x6, lsl #2]
    strb w0, [x19, x5]
    add w2, w5, #1
    str w2, [x26, x1, lsl #2]
    add w2, w1, #1
    str w2, [x26, x5, lsl #2]
    add x5, x5, #1
    b pass_loop
pass_done:
    cbnz x6, bad_brackets
    mov x21, x5
    adrp x9, disp
    add x9, x9, :lo12:disp
    adr x10, cmd_advance
    mov x11, #0
disp_init_loop:
    str x10, [x9, x11, lsl #3]
    add x11, x11, #1
    cmp x11, #256
    b.ne disp_init_loop
    adr x10, cmd_gt
    mov x11, #C_GT
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_lt
    mov x11, #C_LT
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_plus
    mov x11, #C_PLUS
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_minus
    mov x11, #C_MINUS
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_dot
    mov x11, #C_DOT
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_comma
    mov x11, #C_COMMA
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_lbr
    mov x11, #C_LBR
    str x10, [x9, x11, lsl #3]
    adr x10, cmd_rbr
    mov x11, #C_RBR
    str x10, [x9, x11, lsl #3]
    adrp x27, tape
    add x27, x27, :lo12:tape
    mov x28, #0
    mov x25, #0
exec_loop:
    cmp x25, x21
    b.eq exec_done
    ldrb w0, [x19, x25]
    ldr x1, [x9, x0, lsl #3]
    br x1
exec_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b exec_done
bad_brackets:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
    b bad_brackets
cmd_gt:
    add x28, x28, #1
    b cmd_advance
cmd_lt:
    sub x28, x28, #1
    b cmd_advance
cmd_plus:
    ldrb w0, [x27, x28]
    add w0, w0, #1
    strb w0, [x27, x28]
    b cmd_advance
cmd_minus:
    ldrb w0, [x27, x28]
    sub w0, w0, #1
    strb w0, [x27, x28]
    b cmd_advance
cmd_dot:
    mov x0, #1
    add x1, x27, x28
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    b cmd_advance
cmd_comma:
    cmp x22, x20
    b.hs comma_eof
    ldrb w0, [x19, x22]
    add x22, x22, #1
    strb w0, [x27, x28]
    b cmd_advance
comma_eof:
    strb wzr, [x27, x28]
    b cmd_advance
cmd_lbr:
    ldrb w0, [x27, x28]
    cbnz w0, cmd_advance
    ldr w0, [x26, x25, lsl #2]
    mov x25, x0
    b exec_loop
cmd_rbr:
    ldrb w0, [x27, x28]
    cbz w0, cmd_advance
    ldr w0, [x26, x25, lsl #2]
    mov x25, x0
    b exec_loop
cmd_advance:
    add x25, x25, #1
    b exec_loop

.bss
.balign 8
inbuf:
    .zero IN_MAX
.balign 8
tape:
    .zero TAPE_SIZE
.balign 8
jumptab:
    .zero JTAB_SIZE
.balign 8
bstack:
    .zero BSTACK_SIZE
.balign 8
disp:
    .zero DISP_SIZE
