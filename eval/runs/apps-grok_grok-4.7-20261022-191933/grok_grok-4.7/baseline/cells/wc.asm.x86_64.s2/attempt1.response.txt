.globl _start

.bss
.align 16
buf:
    .zero 8192
numtmp:
    .zero 32
outbuf:
    .zero 64

.text

utoa:
    lea r8, [rip + numtmp]
    mov r9, r8
    mov rcx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [r9], dl
    inc r9
    test rax, rax
    jnz 1b
2:
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    cmp r9, r8
    jne 2b
    ret

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 8192
    syscall
    test rax, rax
    jle finish
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 3f
    inc r12
3:
    cmp al, 32
    je 4f
    cmp al, 13
    ja 5f
    cmp al, 9
    jb 5f
4:
    xor r15d, r15d
    jmp 6f
5:
    test r15, r15
    jnz 6f
    mov r15, 1
    inc r13
6:
    dec rcx
    jnz scan
    jmp read_loop
finish:
    lea rdi, [rip + outbuf]
    mov r10, rdi
    mov rax, r12
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call utoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call utoa
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, r10
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    syscall
    mov eax, 60
    xor edi, edi
    syscall
