.section .rodata
err_msg:
    .ascii "error"
    .byte 10
err_len = . - err_msg
min_msg:
    .ascii "-9223372036854775808"
    .byte 10
min_len = . - min_msg

.bss
.balign 8
input_buf:
    .zero 65536
rpn_stack:
    .zero 16384
numbuf:
    .zero 32

.text
.globl _start
_start:
    xor rbx, rbx
read_more:
    cmp rbx, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_more
read_done:
    lea r12, [rip + input_buf]
    lea r13, [r12 + rbx]
line_loop:
    cmp r12, r13
    jae do_exit
    xor r14, r14
    xor r15, r15
token_loop:
    cmp r12, r13
    jae finish_line
    movzx eax, byte ptr [r12]
    cmp al, 10
    je finish_nl
    cmp al, 32
    je skip_ws
    cmp al, 9
    je skip_ws
    mov rbx, r12
scan_token:
    inc r12
    cmp r12, r13
    jae got_token
    movzx eax, byte ptr [r12]
    cmp al, 32
    je got_token
    cmp al, 9
    je got_token
    cmp al, 10
    je got_token
    jmp scan_token
skip_ws:
    inc r12
    jmp token_loop
got_token:
    test r14, r14
    jnz token_loop
    movzx eax, byte ptr [rbx]
    mov rdx, r12
    sub rdx, rbx
    cmp rdx, 1
    jne parse_number
    cmp al, 43
    je do_op
    cmp al, 45
    je do_op
    cmp al, 42
    je do_op
    cmp al, 47
    je do_op
    cmp al, 37
    je do_op
    jmp parse_number
parse_number:
    mov rsi, rbx
    mov rdx, r12
    sub rdx, rbx
    xor r8d, r8d
    cmp byte ptr [rsi], 45
    jne no_sign
    mov r8d, 1
    inc rsi
    dec rdx
no_sign:
    cmp rdx, 18
    ja set_error
    test rdx, rdx
    jz set_error
    xor rax, rax
digit_loop:
    movzx ecx, byte ptr [rsi]
    sub ecx, 48
    cmp ecx, 9
    ja set_error
    imul rax, rax, 10
    add rax, rcx
    inc rsi
    dec rdx
    jnz digit_loop
    test r8d, r8d
    jz push_num
    neg rax
push_num:
    cmp r15, 2048
    jae set_error
    lea rdi, [rip + rpn_stack]
    mov qword ptr [rdi + r15*8], rax
    inc r15
    jmp token_loop
do_op:
    mov r10b, al
    cmp r15, 2
    jb set_error
    lea rdi, [rip + rpn_stack]
    dec r15
    mov rbx, qword ptr [rdi + r15*8]
    dec r15
    mov rax, qword ptr [rdi + r15*8]
    cmp r10b, 43
    je op_add
    cmp r10b, 45
    je op_sub
    cmp r10b, 42
    je op_mul
    cmp r10b, 47
    je op_div
    jmp op_mod
op_add:
    add rax, rbx
    jmp push_res
op_sub:
    sub rax, rbx
    jmp push_res
op_mul:
    imul rax, rbx
    jmp push_res
op_div:
    test rbx, rbx
    jz set_error
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne div_ok
    cmp rbx, -1
    je set_error
div_ok:
    cqo
    idiv rbx
    jmp push_res
op_mod:
    test rbx, rbx
    jz set_error
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne mod_ok
    cmp rbx, -1
    je set_error
mod_ok:
    cqo
    idiv rbx
    mov rax, rdx
    jmp push_res
push_res:
    lea rdi, [rip + rpn_stack]
    mov qword ptr [rdi + r15*8], rax
    inc r15
    jmp token_loop
set_error:
    mov r14, 1
    jmp token_loop
finish_nl:
    inc r12
finish_line:
    test r14, r14
    jnz emit_err
    cmp r15, 1
    jne emit_err
    lea rdi, [rip + rpn_stack]
    mov rax, qword ptr [rdi]
    call print_i64
    jmp line_loop
emit_err:
    lea rsi, [rip + err_msg]
    mov rdx, err_len
    call write_all
    jmp line_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit
print_i64:
    mov r8, 1
    shl r8, 63
    cmp rax, r8
    jne print_regular
    lea rsi, [rip + min_msg]
    mov rdx, min_len
    call write_all
    ret
print_regular:
    lea rsi, [rip + numbuf]
    add rsi, 31
    mov byte ptr [rsi], 10
    xor r9, r9
    test rax, rax
    jns print_digits
    neg rax
    mov r9, 1
print_digits:
    mov rcx, 10
print_digit_loop:
    xor rdx, rdx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz print_digit_loop
    test r9, r9
    jz print_do_write
    dec rsi
    mov byte ptr [rsi], 45
print_do_write:
    lea rdx, [rip + numbuf]
    add rdx, 32
    sub rdx, rsi
    call write_all
    ret
write_all:
    test rdx, rdx
    jz write_done
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_done:
    ret
