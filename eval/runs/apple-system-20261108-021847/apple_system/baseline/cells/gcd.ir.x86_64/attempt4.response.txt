export function l $gcd(...) {
    .section .text
    .globl  _gcd
_gcd:
    @label
    movl    %rdi, %rax
    cmplq   %rsi, %rax
    jne     .L_gcd_next
    movl    %rax, %rdx
    jmp     .L_gcd_next
.L_gcd_next:
    cmplq   %rcx, %rdx
    jne     .L_gcd_swap
    addl    %rdx, %rax
    jmp     .L_gcd_next
.L_gcd_swap:
    movl    %rdx, %rcx
    movl    %rax, %rdx
    subl    %rcx, %rdx
    cmplq   %rax, %rdx
    jne     .L_gcd_next
    subl    %rdx, %rax
    jmp     .L_gcd_loop
.L_gcd_loop:
    cmplq   %rdx, %rax
    jne     .L_gcd_done
    jmp     .L_gcd_loop
.L_gcd_done:
    popl    %rdx
    popl    %rcx
    popl    %rdx
    popl    %rax
    movl    %rax, %rsp
    ret
