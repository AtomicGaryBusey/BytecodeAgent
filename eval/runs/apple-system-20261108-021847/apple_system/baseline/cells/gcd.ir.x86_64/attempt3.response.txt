export function l $gcd(...) {
    .section .data
    .align 16
    .globl  _gcd, _gcd_end
_gcd:
    .L_gcd_loop:
    @label
    movq    %rdi, %rax
    cmplq   %rsi, %rax
    jne     .L_gcd_next
    movq    %rax, %rdx
    jmp     .L_gcd_next
.L_gcd_next:
    cmplq   %rcx, %rdx
    jne     .L_gcd_swap
    addq    %rdx, %rax
    jmp     .L_gcd_next
.L_gcd_swap:
    movq    %rdx, %rcx
    movq    %rax, %rdx
    subq    %rcx, %rdx
    cmplq   %rax, %rdx
    jne     .L_gcd_next
    subq    %rdx, %rax
    jmp     .L_gcd_loop
.L_gcd_done:
    popl    %rdx
    popl    %rcx
    popl    %rdx
    popl    %rax
    movq    %rax, %rsp
    ret
_gcd_end:
