export function l $fib(...)
@fib
l:
    // Base case: if n == 0, return 0
    csll %n, %n, #0
    je  @fib
    // Initialize fib0 and fib1
    mov %r0, #0
    mov %r1, #1
    // Iteration loop: n from 1 to 90
    @loop
    csll %k, %n, #0
    cmovz %fib0, %fib1, %fib0
    csll %fib1, %fib0, #1
    csll %fib0, %fib0, #-1
    cmovz %w_result, %fib0, %fib1
    // Check if k == 0 (should not happen due to initial conditions)
    csll %k, %n, #0
    je  @done
    // Update n for next iteration
    csll %n, %n, #-1
    // Loop back
    jnz %loop, %k
@done
    // Return fib0 (which holds fib(n))
    ret
