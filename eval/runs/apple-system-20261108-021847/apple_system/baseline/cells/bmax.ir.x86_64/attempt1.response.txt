export function l $bmax(...) {
    @label bmax
    movq %rdi, %rcx
    movq %rsi, %rdx
    movq %rcx, %rax
    loop:
        cmpl %rdx, %rax
        js .fail
        cmpl %rcx, %rax
        jne .loop
        movl %rax, %rdx
        jmp .end
.fail:
        movl %rdx, %rax
.loop:
        jmp .loop
.end:
