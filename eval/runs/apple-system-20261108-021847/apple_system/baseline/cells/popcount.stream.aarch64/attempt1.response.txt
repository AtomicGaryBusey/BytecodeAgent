popcount:
    mov x2, #0
    // Initialize result to 0
    mov x1, x0, lsl #0
    // x0 is the input; shift left by 0 to align for bit testing
    mov x3, #63
    // x3 holds the loop counter (64 bits, one bit at a time)
    // Loop over all 64 bits
.loop:
    // Test bit 0
    mov x4, x1, lsl #0
    // Test bit 1
    mov x5, x1, lsl #1
    // Test bit 2
    mov x6, x1, lsl #2
    // Test bit 3
    mov x7, x1, lsl #3
    // ... continue for bits 4 to 63
    // Use loops with decreasing count
    // Decrement x3 and repeat
    cbz x3, .endloop
    // Increment x2 if bit is set
    cmp x4, #1
    beq .set_one
    // Otherwise, continue to next bit
    // Repeat for remaining bits
.endloop:
    // Check last bit
    mov x8, x1, lsl #63
    // Final check
.set_one:
    // Return result in x0
    mov x0, x2
