popcount:
    mov x2, #0
    mov x1, x0, lsl #0
    mov x3, #63
.loop:
    mov x4, x1, lsl #0
    cmp x4, #1
    beq .set_one
    mov x5, x1, lsl #1
    cmp x5, #1
    beq .set_one
    mov x6, x1, lsl #2
    cmp x6, #1
    beq .set_one
    mov x7, x1, lsl #3
    cmp x7, #1
    beq .set_one
    // Continue for remaining bits
    // Decrement x3 and repeat
    cbz x3, .endloop
    // Increment x2 if bit is set
    mov x0, x2
.endloop:
