gcd:
    mov rax, rdi
    xor rdx, rsi
    test rdx, rdx
    je .Lcheck_zero
.Lcheck_zero:
    mov rcx, 1
.Lloop:
    mov rdx, rax
    cmp rdx, rcx
    jl .Lnext
    mov rax, rdx
    mov rsi, rcx
    shr rsi, 36  # Clear lower 32 bits for division
    div rsi
    mov rdx, rax
    jmp .Lloop
.Lnext:
    mov rdx, rax
    shr rdx, 1
    jne .Lloop
    mov rax, rdx
