```asm
export function l $gcd(...)
    @gcd:
    ldr l0, [l0]
    ldr l1, [l1]
    cmpl l0, l1
    csltl w0, l0, l1
    jnz @gcd_loop
    movl l0, w0
    ret

@gcd_loop
    cmpl w0, l1
    ceql w0, l1, l0
    jnz @gcd_next
    subl l0, l0, l1
    jmp @gcd
    @gcd_next
    addl l1, l0, 1
    jmp @gcd
```
