fib:
    push rbx
.Lcheck:
    mov rax, rdi
    mov rdx, 0
    cmp rdi, rcx
    jl .Lnext
    mov rcx, 1
.Lcalc:
    mov rbx, rax
    sub rbx, rdx
    add rax, rbx
    mov rdx, rax
    sub rcx, 1
    jmp .Lcheck
.Lnext:
    mov rdx, 0
    pop rbx
    cmp rcx, 0
    jl .Lend
    mov rax, rdx
.Lend:
    ret
