fib:
    push rbx
.Lloop:
    mov rax, rdi
    mov rdx, 0
    cmp rdi, rcx
    jl .Lnext
    mov rcx, 1
.Lcalc:
    mov rbx, rax
    sub rbx, rdx
    mov rax, rbx
    add rax, rdx
    mov rdx, rax
    sub rcx, 1
    jmp .Lcalc
.Lnext:
    mov rax, rdx
    pop rbx
    cmp rcx, 0
    jl .Lend
    mov rcx, rax
.Lend:
    mov rax, rcx
    ret
