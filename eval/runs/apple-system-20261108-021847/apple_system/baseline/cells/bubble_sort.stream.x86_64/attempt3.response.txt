bubble_sort:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rax, rdi
    mov rsi, rdi
    mov rcx, rdi
    mov rdx, rdi
    xor rax, rax
.Lloop:
    cmp rcx, 1
    jne .Lend
    mov rax, qword ptr [rsi + rcx*8]
    cmp rax, qword ptr [rsi + rcx*8 + 8]
    jl .Lswap
    mov rax, qword ptr [rsi + rcx*8 + 8]
    jmp .Lnext
.Lswap:
    mov rcx, rax
    mov rsi, rdx
    jmp .Lnext
.Lnext:
    add rcx, 8
    jmp .Lloop
.Lend:
    pop r15
    pop r14
    pop r13
    pop r12
    pop r11
    pop r10
    pop r9
    pop r8
    ret
