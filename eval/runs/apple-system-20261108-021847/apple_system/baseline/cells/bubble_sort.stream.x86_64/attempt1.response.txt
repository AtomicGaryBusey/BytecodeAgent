bubble_sort:
    push r12
    push r13
    push r14
    push r15
    mov rax, rdi
    mov rcx, rsi
    mov rdx, rdx
    mov rsi, rdi
    xor rax, rax
.Lloop:
    cmp rsi, rsi
    jne .Lnext
    mov rax, qword ptr [rsi + rcx*8]
    mov rdx, qword ptr [rsi + rcx*8 + 8]
    sub rax, rdx
    jg .Lswap
    mov rax, qword ptr [rsi + rcx*8]
    jmp .Lnext
.Lswap:
    mov rdi, rax
    mov rcx, rsi
    call __builtin_expect
    jmp .Lloop
.Lend:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
