count_nonzero_prefix:
    // Initialize counter and pointer
    mov x21, #0
    ldr x2, [x0, #0]
    // Check first element
    cbz x2, .Lcheck_prefix
    // Move to next element and decrement counter
    add x0, x0, #8
    sub x21, x21, #1
.Lcheck_prefix:
    // If counter not zero, continue
    cmp x21, #0
    beq .Lend_prefix
    // Move to next element and decrement counter
    add x0, x0, #8
    sub x21, x21, #1
    // Repeat
    b .Lcheck_prefix
.Lfound_nonzero:
    // Return index of first zero
    // Restore callee-saved registers
    ldr x29, [sp, #16]
    ldr x19, [sp, #0]
    ldr x20, [sp, #8]
    // Restore sp to original value
    ldr x29, [sp, #16]
    ldr x19, [sp, #0]
    ldr x20, [sp, #8]
    ret
