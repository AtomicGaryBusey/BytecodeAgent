count_nonzero_prefix:
    // Initialize x19-x28, x29, d8-d15 as callee-saved registers
    // Save sp to preserve alignment
    ldr x19, [sp, #0]
    ldr x20, [sp, #8]
    // Set up loop counter and pointer
    mov x21, #0
    ldr x2, [x0, #0]
    // Check first element
    cbz x2, .Lfound_nonzero
    // Move to next element
    add x0, x0, #8
    // Decrement counter
    sub x21, x21, #1
    // Repeat if not all elements processed
    b .Lcheck_prefix
.Lcheck_prefix:
    // Continue loop if counter not zero
    cmp x21, #0
    beq .Lend_prefix
    // Move to next element
    add x0, x0, #8
    // Decrement counter
    sub x21, x21, #1
    // Repeat
    b .Lcheck_prefix
.Lfound_nonzero:
    // Return index of first zero
    // Restore callee-saved registers
    ldr x29, [sp, #16]
    ldr x19, [sp, #0]
    ldr x20, [sp, #8]
    // Restore sp to original value
    ldr sp, [x19, #0]
