export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %zero =l copy 0
    %j_init =l copy %zero
    jmp @init_loop_cond

@init_loop_cond
    %cond_init =w csltl %j_init, %n
    jnz %cond_init, @init_loop_body
    jmp @init_done

@init_loop_body
    %row_ptr =l add %row, %j_init
    storel %row_ptr, %zero
    %j_next =l add %j_init, 1
    %j_init =l copy %j_next
    jmp @init_loop_cond

@init_done
    %i =l copy %zero
    jmp @outer_cond

@outer_cond
    %cond_outer =w csltl %i, %m
    jnz %cond_outer, @outer_body
    jmp @outer_done

@outer_body
    %diag =l copy %zero
    %j =l copy 1
    %x_i_ptr =l add %x, %i
    %x_i =w loadub %x_i_ptr
    jmp @inner_cond

@inner_cond
    %cond_inner =w cslel %j, %n
    jnz %cond_inner, @inner_body
    jmp @inner_done

@inner_body
    %up_ptr =l add %row, %j
    %up =l loadl %up_ptr
    %y_jm1_ptr =l add %y, %j
    %y_jm1_ptr_dec =l sub %y_jm1_ptr, 1
    %y_jm1 =w loadub %y_jm1_ptr_dec
    %cmp =w ceqw %x_i, %y_jm1
    jnz %cmp, @match
    jmp @nomatch

@match
    %new_val =l add %diag, 1
    storel %up_ptr, %new_val
    jmp @inner_update_diag

@nomatch
    %row_jm1_ptr =l sub %up_ptr, 8
    %row_jm1 =l loadl %row_jm1_ptr
    %cmp_max =w csgtl %up, %row_jm1
    jnz %cmp_max, @keep_up
    %up =l copy %row_jm1
@keep_up
    storel %up_ptr, %up
    jmp @inner_update_diag

@inner_update_diag
    %diag =l copy %up
    %j_next =l add %j, 1
    %j =l copy %j_next
    jmp @inner_cond

@inner_done
    %i_next =l add %i, 1
    %i =l copy %i_next
    jmp @outer_cond

@outer_done
    %result_ptr =l add %row, %n
    %result =l loadl %result_ptr
    ret %result
}
