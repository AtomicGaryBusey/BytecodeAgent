export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %zero =l copy 0
    %one =l copy 1
    %eight =l copy 8
    %j =l copy %zero
    jmp @init_cond

@init_cond
    %cond =w cslel %j, %n
    jnz %cond, @init_body
    jmp @init_done

@init_body
    %j8 =l mul %j, %eight
    %ptr =l add %row, %j8
    storel %ptr, %zero
    %j =l add %j, %one
    jmp @init_cond

@init_done
    %i =l copy %zero
    jmp @outer_cond

@outer_cond
    %cond =w csltl %i, %m
    jnz %cond, @outer_body
    jmp @outer_done

@outer_body
    %diag =l copy %zero
    %j =l copy %one
    %x_ptr =l add %x, %i
    %x_val =w loadub %x_ptr
    jmp @inner_cond

@inner_cond
    %cond =w cslel %j, %n
    jnz %cond, @inner_body
    jmp @inner_done

@inner_body
    %j8 =l mul %j, %eight
    %up_ptr =l add %row, %j8
    %up =l loadl %up_ptr
    %y_ptr =l add %y, %j
    %y_ptr =l sub %y_ptr, %one
    %y_val =w loadub %y_ptr
    %cmp =w ceqw %x_val, %y_val
    jnz %cmp, @match

@nomatch
    %left_ptr =l sub %up_ptr, %eight
    %left =l loadl %left_ptr
    %cmp_max =w csgtl %up, %left
    jnz %cmp_max, @use_up

@use_left
    %new_val =l copy %left
    jmp @store_new

@use_up
    %new_val =l copy %up
    jmp @store_new

@store_new
    storel %up_ptr, %new_val
    jmp @update_diag

@match
    %new_val =l add %diag, %one
    storel %up_ptr, %new_val
    jmp @update_diag

@update_diag
    %diag =l copy %up
    %j =l add %j, %one
    jmp @inner_cond

@inner_done
    %i =l add %i, %one
    jmp @outer_cond

@outer_done
    %n8 =l mul %n, %eight
    %res_ptr =l add %row, %n8
    %res =l loadl %res_ptr
    retl %res
}
