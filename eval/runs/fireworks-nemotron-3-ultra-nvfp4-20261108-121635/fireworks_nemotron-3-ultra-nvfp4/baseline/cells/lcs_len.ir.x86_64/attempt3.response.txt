export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
    %zero =l copy 0
    %one =l copy 1
    %eight =l copy 8
    %j_init =l copy %zero
    jmp @init_cond

@init_cond
    %cond_init =w cslel %j_init, %n
    jnz %cond_init, @init_body

@init_done
    %i =l copy %zero
    jmp @outer_cond

@outer_cond
    %cond_outer =w csltl %i, %m
    jnz %cond_outer, @outer_body

@outer_done
    %n8 =l mul %n, %eight
    %result_ptr =l add %row, %n8
    %result =l loadl %result_ptr
    ret %result

@outer_body
    %diag =l copy %zero
    %j =l copy %one
    %x_i_ptr =l add %x, %i
    %x_i =w loadub %x_i_ptr
    jmp @inner_cond

@inner_cond
    %cond_inner =w cslel %j, %n
    jnz %cond_inner, @inner_body

@inner_done
    %i_next =l add %i, %one
    %i =l copy %i_next
    jmp @outer_cond

@inner_body
    %j8 =l mul %j, %eight
    %up_ptr =l add %row, %j8
    %up =l loadl %up_ptr
    %y_jm1_ptr =l add %y, %j
    %y_jm1_ptr =l sub %y_jm1_ptr, %one
    %y_jm1 =w loadub %y_jm1_ptr
    %cmp =w ceqw %x_i, %y_jm1
    jnz %cmp, @match

@nomatch
    %row_jm1_ptr =l sub %up_ptr, %eight
    %row_jm1 =l loadl %row_jm1_ptr
    %cmp_max =w csgtl %up, %row_jm1
    jnz %cmp_max, @keep_up

@use_row_jm1
    %up =l copy %row_jm1
    jmp @keep_up

@keep_up
    storel %up_ptr, %up
    jmp @inner_update_diag

@match
    %new_val =l add %diag, %one
    storel %up_ptr, %new_val
    jmp @inner_update_diag

@inner_update_diag
    %diag =l copy %up
    %j_next =l add %j, %one
    %j =l copy %j_next
    jmp @inner_cond

@init_body
    %j8_init =l mul %j_init, %eight
    %row_ptr =l add %row, %j8_init
    storel %row_ptr, %zero
    %j_next_init =l add %j_init, %one
    %j_init =l copy %j_next_init
    jmp @init_cond
}
