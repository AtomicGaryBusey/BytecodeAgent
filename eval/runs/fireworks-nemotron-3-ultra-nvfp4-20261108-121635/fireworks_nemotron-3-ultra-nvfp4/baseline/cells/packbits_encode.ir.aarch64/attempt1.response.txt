export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i_w0 =w copy 0
    %out_pos_w0 =w copy 0
    jmp @main_loop

@main_loop
    %i_w =w phi @start %i_w0, @write_repeat %i_w1, @copy_done %i_w2
    %out_pos_w =w phi @start %out_pos_w0, @write_repeat %out_pos_w1, @copy_done %out_pos_w2
    %i_l =l extuw %i_w
    %cond =w csltl %i_l, %n
    jnz %cond, @check_run, @return_success

@check_run
    %i_l =l extuw %i_w
    %src_i_ptr =l add %src, %i_l
    %b0 =w loadub %src_i_ptr
    %run_w0 =w copy 1
    %j_w0 =w add %i_w, 1
    jmp @run_loop

@run_loop
    %run_w =w phi @check_run %run_w0, @run_increment %run_w1
    %j_w =w phi @check_run %j_w0, @run_increment %j_w1
    %b =w phi @check_run %b0, @run_increment %b1
    %j_l =l extuw %j_w
    %cond_j_n =w csltl %j_l, %n
    %limit_w =w add %i_w, 128
    %cond_j_limit =w csltw %j_w, %limit_w
    %cond_continue =w and %cond_j_n, %cond_j_limit
    jnz %cond_continue, @run_compare, @run_done_pass

@run_compare
    %j_l =l extuw %j_w
    %src_j_ptr =l add %src, %j_l
    %b_j =w loadub %src_j_ptr
    %eq =w ceqw %b_j, %b
    jnz %eq, @run_increment, @run_done_pass

@run_increment
    %run_w1 =w add %run_w, 1
    %j_w1 =w add %j_w, 1
    %b1 =w copy %b
    jmp @run_loop

@run_done_pass
    %run_w_pass =w copy %run_w
    %b_pass =w copy %b
    jmp @run_done

@run_done
    %run_w2 =w phi @run_loop %run_w, @run_done_pass %run_w_pass
    %b2 =w phi @run_loop %b, @run_done_pass %b_pass
    %cond_repeat =w csgew %run_w2, 3
    jnz %cond_repeat, @emit_repeat, @emit_literal

@emit_repeat
    %header_w =w sub 257, %run_w2
    %packet_size_w =w copy 2
    %out_pos_l =l extuw %out_pos_w
    %packet_size_l =l extuw %packet_size_w
    %need_l =l add %out_pos_l, %packet_size_l
    %cond_cap =w cslel %need_l, %cap
    jnz %cond_cap, @write_repeat, @return_fail

@write_repeat
    %out_pos_l =l extuw %out_pos_w
    %out_ptr =l add %out, %out_pos_l
    storeb %header_w, %out_ptr
    %out_ptr2 =l add %out_ptr, 1
    storeb %b2, %out_ptr2
    %out_pos_w1 =w add %out_pos_w, 2
    %i_w1 =w add %i_w, %run_w2
    jmp @main_loop

@emit_literal
    %j_w0 =w copy %i_w
    jmp @literal_loop_cond

@literal_loop_cond
    %j_w =w phi @emit_literal %j_w0, @literal_advance_j %j_w1
    %j_l =l extuw %j_w
    %cond_j_n =w csltl %j_l, %n
    %diff_w =w sub %j_w, %i_w
    %cond_diff =w csltw %diff_w, 128
    %cond_continue =w and %cond_j_n, %cond_diff
    jnz %cond_continue, @literal_compute_run, @literal_done

@literal_compute_run
    %j_l =l extuw %j_w
    %src_j_ptr =l add %src, %j_l
    %b_j0 =w loadub %src_j_ptr
    %run_j_w0 =w copy 1
    %k_w0 =w add %j_w, 1
    jmp @literal_run_loop

@literal_run_loop
    %run_j_w =w phi @literal_compute_run %run_j_w0, @literal_run_increment %run_j_w1
    %k_w =w phi @literal_compute_run %k_w0, @literal_run_increment %k_w1
    %b_j =w phi @literal_compute_run %b_j0, @literal_run_increment %b_j1
    %k_l =l extuw %k_w
    %cond_k_n =w csltl %k_l, %n
    %limit_j_w =w add %j_w, 128
    %cond_k_limit =w csltw %k_w, %limit_j_w
    %cond_k =w and %cond_k_n, %cond_k_limit
    jnz %cond_k, @literal_run_compare, @literal_run_done_pass

@literal_run_compare
    %k_l =l extuw %k_w
    %src_k_ptr =l add %src, %k_l
    %b_k =w loadub %src_k_ptr
    %eq =w ceqw %b_k, %b_j
    jnz %eq, @literal_run_increment, @literal_run_done_pass

@literal_run_increment
    %run_j_w1 =w add %run_j_w, 1
    %k_w1 =w add %k_w, 1
    %b_j1 =w copy %b_j
    jmp @literal_run_loop

@literal_run_done_pass
    %run_j_w_pass =w copy %run_j_w
    %b_j_pass =w copy %b_j
    jmp @literal_run_done

@literal_run_done
    %run_j_w2 =w phi @literal_run_loop %run_j_w, @literal_run_done_pass %run_j_w_pass
    %b_j2 =w phi @literal_run_loop %b_j, @literal_run_done_pass %b_j_pass
    %cond_run_ge3 =w csgew %run_j_w2, 3
    jnz %cond_run_ge3, @literal_done, @literal_advance_j

@literal_advance_j
    %j_w1 =w add %j_w, 1
    jmp @literal_loop_cond

@literal_done
    %L_w =w sub %j_w, %i_w
    jmp @write_literal

@write_literal
    %header_w =w sub %L_w, 1
    %packet_size_w =w add 1, %L_w
    %out_pos_l =l extuw %out_pos_w
    %packet_size_l =l extuw %packet_size_w
    %need_l =l add %out_pos_l, %packet_size_l
    %cond_cap =w cslel %need_l, %cap
    jnz %cond_cap, @write_literal_body, @return_fail

@write_literal_body
    %out_pos_l =l extuw %out_pos_w
    %out_ptr =l add %out, %out_pos_l
    storeb %header_w, %out_ptr
    %dst_start =l add %out_ptr, 1
    %i_l =l extuw %i_w
    %src_start_l =l add %src, %i_l
    %copy_idx_w0 =w copy 0
    jmp @copy_loop_cond

@copy_loop_cond
    %copy_idx_w =w phi @write_literal_body %copy_idx_w0, @copy_body %copy_idx_w1
    %cond_copy =w csltw %copy_idx_w, %L_w
    jnz %cond_copy, @copy_body, @copy_done

@copy_body
    %copy_idx_l =l extuw %copy_idx_w
    %src_byte_ptr =l add %src_start_l, %copy_idx_l
    %byte =w loadub %src_byte_ptr
    %dst_byte_ptr =l add %dst_start, %copy_idx_l
    storeb %byte, %dst_byte_ptr
    %copy_idx_w1 =w add %copy_idx_w, 1
    jmp @copy_loop_cond

@copy_done
    %out_pos_w2 =w add %out_pos_w, 1
    %out_pos_w2 =w add %out_pos_w2, %L_w
    %i_w2 =w copy %j_w
    jmp @main_loop

@return_success
    %out_pos_l =l extuw %out_pos_w
    ret %out_pos_l

@return_fail
    ret -1
}
