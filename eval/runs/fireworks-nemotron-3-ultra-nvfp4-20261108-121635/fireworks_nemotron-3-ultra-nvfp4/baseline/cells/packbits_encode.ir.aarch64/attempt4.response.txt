export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %s_i_w0 =w copy 0
    %s_out_pos_w0 =w copy 0
    jmp @main_loop

@main_loop
    %ml_i_w =w phi @start %s_i_w0, @write_repeat %wr_i_w1, @copy_done %cd_i_w2
    %ml_out_pos_w =w phi @start %s_out_pos_w0, @write_repeat %wr_out_pos_w1, @copy_done %cd_out_pos_w2
    %ml_i_l =l extuw %ml_i_w
    %ml_cond =w csltl %ml_i_l, %n
    jnz %ml_cond, @check_run, @return_success

@check_run
    %cr_i_l =l extuw %ml_i_w
    %cr_src_i_ptr =l add %src, %cr_i_l
    %cr_b0 =w loadub %cr_src_i_ptr
    %cr_run_w0 =w copy 1
    %cr_j_w0 =w add %ml_i_w, 1
    jmp @run_loop

@run_loop
    %rl_run_w =w phi @check_run %cr_run_w0, @run_increment %ri_run_w1
    %rl_j_w =w phi @check_run %cr_j_w0, @run_increment %ri_j_w1
    %rl_b =w phi @check_run %cr_b0, @run_increment %ri_b1
    %rl_j_l =l extuw %rl_j_w
    %rl_cond_j_n =w csltl %rl_j_l, %n
    %rl_limit_w =w add %ml_i_w, 128
    %rl_cond_j_limit =w csltw %rl_j_w, %rl_limit_w
    %rl_cond_continue =w and %rl_cond_j_n, %rl_cond_j_limit
    jnz %rl_cond_continue, @run_compare, @run_done

@run_compare
    %rc_j_l =l extuw %rl_j_w
    %rc_src_j_ptr =l add %src, %rc_j_l
    %rc_b_j =w loadub %rc_src_j_ptr
    %rc_eq =w ceqw %rc_b_j, %rl_b
    jnz %rc_eq, @run_increment, @run_done

@run_increment
    %ri_run_w1 =w add %rl_run_w, 1
    %ri_j_w1 =w add %rl_j_w, 1
    %ri_b1 =w copy %rl_b
    jmp @run_loop

@run_done
    %rd_run_w2 =w phi @run_loop %rl_run_w, @run_compare %rl_run_w
    %rd_b2 =w phi @run_loop %rl_b, @run_compare %rl_b
    %rd_cond_repeat =w csgew %rd_run_w2, 3
    jnz %rd_cond_repeat, @emit_repeat, @emit_literal

@emit_repeat
    %er_header_w =w sub 257, %rd_run_w2
    %er_packet_size_w =w copy 2
    %er_out_pos_l =l extuw %ml_out_pos_w
    %er_packet_size_l =l extuw %er_packet_size_w
    %er_need_l =l add %er_out_pos_l, %er_packet_size_l
    %er_cond_cap =w cslel %er_need_l, %cap
    jnz %er_cond_cap, @write_repeat, @return_fail

@write_repeat
    %wr_out_pos_l =l extuw %ml_out_pos_w
    %wr_out_ptr =l add %out, %wr_out_pos_l
    storeb %er_header_w, %wr_out_ptr
    %wr_out_ptr2 =l add %wr_out_ptr, 1
    storeb %rd_b2, %wr_out_ptr2
    %wr_out_pos_w1 =w add %ml_out_pos_w, 2
    %wr_i_w1 =w add %ml_i_w, %rd_run_w2
    jmp @main_loop

@emit_literal
    %el_j_w0 =w copy %ml_i_w
    jmp @literal_loop_cond

@literal_loop_cond
    %llc_j_w =w phi @emit_literal %el_j_w0, @literal_advance_j %laj_j_w1
    %llc_j_l =l extuw %llc_j_w
    %llc_cond_j_n =w csltl %llc_j_l, %n
    %llc_diff_w =w sub %llc_j_w, %ml_i_w
    %llc_cond_diff =w csltw %llc_diff_w, 128
    %llc_cond_continue =w and %llc_cond_j_n, %llc_cond_diff
    jnz %llc_cond_continue, @literal_compute_run, @literal_done

@literal_compute_run
    %lcr_j_l =l extuw %llc_j_w
    %lcr_src_j_ptr =l add %src, %lcr_j_l
    %lcr_b_j0 =w loadub %lcr_src_j_ptr
    %lcr_run_j_w0 =w copy 1
    %lcr_k_w0 =w add %llc_j_w, 1
    jmp @literal_run_loop

@literal_run_loop
    %lrl_run_j_w =w phi @literal_compute_run %lcr_run_j_w0, @literal_run_increment %lri_run_j_w1
    %lrl_k_w =w phi @literal_compute_run %lcr_k_w0, @literal_run_increment %lri_k_w1
    %lrl_b_j =w phi @literal_compute_run %lcr_b_j0, @literal_run_increment %lri_b_j1
    %lrl_k_l =l extuw %lrl_k_w
    %lrl_cond_k_n =w csltl %lrl_k_l, %n
    %lrl_limit_j_w =w add %llc_j_w, 128
    %lrl_cond_k_limit =w csltw %lrl_k_w, %lrl_limit_j_w
    %lrl_cond_k =w and %lrl_cond_k_n, %lrl_cond_k_limit
    jnz %lrl_cond_k, @literal_run_compare, @literal_run_done

@literal_run_compare
    %lrc_k_l =l extuw %lrl_k_w
    %lrc_src_k_ptr =l add %src, %lrc_k_l
    %lrc_b_k =w loadub %lrc_src_k_ptr
    %lrc_eq =w ceqw %lrc_b_k, %lrl_b_j
    jnz %lrc_eq, @literal_run_increment, @literal_run_done

@literal_run_increment
    %lri_run_j_w1 =w add %lrl_run_j_w, 1
    %lri_k_w1 =w add %lrl_k_w, 1
    %lri_b_j1 =w copy %lrl_b_j
    jmp @literal_run_loop

@literal_run_done
    %lrd_run_j_w2 =w phi @literal_run_loop %lrl_run_j_w, @literal_run_compare %lrl_run_j_w
    %lrd_b_j2 =w phi @literal_run_loop %lrl_b_j, @literal_run_compare %lrl_b_j
    %lrd_cond_run_ge3 =w csgew %lrd_run_j_w2, 3
    jnz %lrd_cond_run_ge3, @literal_done, @literal_advance_j

@literal_advance_j
    %laj_j_w1 =w add %llc_j_w, 1
    jmp @literal_loop_cond

@literal_done
    %ld_L_w =w sub %llc_j_w, %ml_i_w
    jmp @write_literal

@write_literal
    %wl_header_w =w sub %ld_L_w, 1
    %wl_packet_size_w =w add 1, %ld_L_w
    %wl_out_pos_l =l extuw %ml_out_pos_w
    %wl_packet_size_l =l extuw %wl_packet_size_w
    %wl_need_l =l add %wl_out_pos_l, %wl_packet_size_l
    %wl_cond_cap =w cslel %wl_need_l, %cap
    jnz %wl_cond_cap, @write_literal_body, @return_fail

@write_literal_body
    %wlb_out_pos_l =l extuw %ml_out_pos_w
    %wlb_out_ptr =l add %out, %wlb_out_pos_l
    storeb %wl_header_w, %wlb_out_ptr
    %wlb_dst_start =l add %wlb_out_ptr, 1
    %wlb_i_l =l extuw %ml_i_w
    %wlb_src_start_l =l add %src, %wlb_i_l
    %wlb_copy_idx_w0 =w copy 0
    jmp @copy_loop_cond

@copy_loop_cond
    %clc_copy_idx_w =w phi @write_literal_body %wlb_copy_idx_w0, @copy_body %cb_copy_idx_w1
    %clc_cond_copy =w csltw %clc_copy_idx_w, %ld_L_w
    jnz %clc_cond_copy, @copy_body, @copy_done

@copy_body
    %cb_copy_idx_l =l extuw %clc_copy_idx_w
    %cb_src_byte_ptr =l add %wlb_src_start_l, %cb_copy_idx_l
    %cb_byte =w loadub %cb_src_byte_ptr
    %cb_dst_byte_ptr =l add %wlb_dst_start, %cb_copy_idx_l
    storeb %cb_byte, %cb_dst_byte_ptr
    %cb_copy_idx_w1 =w add %clc_copy_idx_w, 1
    jmp @copy_loop_cond

@copy_done
    %cd_out_pos_w_tmp =w add %ml_out_pos_w, 1
    %cd_out_pos_w2 =w add %cd_out_pos_w_tmp, %ld_L_w
    %cd_i_w2 =w copy %llc_j_w
    jmp @main_loop

@return_success
    %rs_out_pos_l =l extuw %ml_out_pos_w
    ret %rs_out_pos_l

@return_fail
    ret -1
}
