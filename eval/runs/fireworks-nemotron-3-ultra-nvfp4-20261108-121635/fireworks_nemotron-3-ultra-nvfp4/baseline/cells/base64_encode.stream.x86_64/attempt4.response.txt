base64_encode:
	push rbx
	push r12
	push r13
	push r14
	mov r9, rdx
	mov rax, rsi
	mov rcx, 3
	xor rdx, rdx
	div rcx
	mov r13, rax
	mov r14, rdx
	mov rax, r13
	test r14, r14
	jz .Lno_remainder
	inc rax
.Lno_remainder:
	shl rax, 2
	mov r12, rax
	test r13, r13
	jz .Lremainder
.Lfull_loop:
	movzx r8d, byte ptr [rdi]
	movzx esi, byte ptr [rdi+1]
	movzx edx, byte ptr [rdi+2]
	mov ebx, r8d
	shr ebx, 2
	movzx eax, bl
	mov cl, 0
	lea r10, [rip + .Lafter_map0]
	jmp .Lmap_char
.Lafter_map0:
	mov ebx, r8d
	and ebx, 3
	shl ebx, 4
	mov eax, esi
	shr eax, 4
	or ebx, eax
	movzx eax, bl
	mov cl, 1
	lea r10, [rip + .Lafter_map1]
	jmp .Lmap_char
.Lafter_map1:
	mov ebx, esi
	and ebx, 0xF
	shl ebx, 2
	mov eax, edx
	shr eax, 6
	or ebx, eax
	movzx eax, bl
	mov cl, 2
	lea r10, [rip + .Lafter_map2]
	jmp .Lmap_char
.Lafter_map2:
	mov ebx, edx
	and ebx, 0x3F
	movzx eax, bl
	mov cl, 3
	lea r10, [rip + .Lafter_map3]
	jmp .Lmap_char
.Lafter_map3:
	add rdi, 3
	add r9, 4
	dec r13
	jnz .Lfull_loop
.Lremainder:
	test r14, r14
	jz .Ldone
	cmp r14, 1
	je .Lrem1
	movzx r8d, byte ptr [rdi]
	movzx esi, byte ptr [rdi+1]
	mov ebx, r8d
	shr ebx, 2
	movzx eax, bl
	mov cl, 0
	lea r10, [rip + .Lafter_rem2_0]
	jmp .Lmap_char
.Lafter_rem2_0:
	mov ebx, r8d
	and ebx, 3
	shl ebx, 4
	mov eax, esi
	shr eax, 4
	or ebx, eax
	movzx eax, bl
	mov cl, 1
	lea r10, [rip + .Lafter_rem2_1]
	jmp .Lmap_char
.Lafter_rem2_1:
	mov ebx, esi
	and ebx, 0xF
	shl ebx, 2
	movzx eax, bl
	mov cl, 2
	lea r10, [rip + .Lafter_rem2_2]
	jmp .Lmap_char
.Lafter_rem2_2:
	mov byte ptr [r9+3], 61
	jmp .Ldone
.Lrem1:
	movzx r8d, byte ptr [rdi]
	mov ebx, r8d
	shr ebx, 2
	movzx eax, bl
	mov cl, 0
	lea r10, [rip + .Lafter_rem1_0]
	jmp .Lmap_char
.Lafter_rem1_0:
	mov ebx, r8d
	and ebx, 3
	shl ebx, 4
	movzx eax, bl
	mov cl, 1
	lea r10, [rip + .Lafter_rem1_1]
	jmp .Lmap_char
.Lafter_rem1_1:
	mov byte ptr [r9+2], 61
	mov byte ptr [r9+3], 61
	jmp .Ldone
.Ldone:
	mov rax, r12
	pop r14
	pop r13
	pop r12
	pop rbx
	ret
.Lmap_char:
	cmp al, 26
	jl .Lmap_upper
	cmp al, 52
	jl .Lmap_lower
	cmp al, 62
	jl .Lmap_digit
	je .Lmap_plus
	mov byte ptr [r9 + rcx], 47
	jmp r10
.Lmap_upper:
	add al, 65
	mov byte ptr [r9 + rcx], al
	jmp r10
.Lmap_lower:
	add al, 71
	mov byte ptr [r9 + rcx], al
	jmp r10
.Lmap_digit:
	sub al, 4
	mov byte ptr [r9 + rcx], al
	jmp r10
.Lmap_plus:
	mov byte ptr [r9 + rcx], 43
	jmp r10
