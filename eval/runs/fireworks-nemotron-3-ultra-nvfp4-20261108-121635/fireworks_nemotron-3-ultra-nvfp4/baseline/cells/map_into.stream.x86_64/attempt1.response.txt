map_into:
	push rbx
	push r12
	push r13
	push r14
	push r15
	mov rbx, rcx
	mov r12, rdx
	mov r13, rdi
	mov r14, rsi
	xor r15, r15
.Lloop:
	test r12, r12
	jz .Ldone
	mov rdi, [r14]
	call rbx
	mov [r13], rax
	mov rsi, [r14]
	cmp rax, rsi
	jge .Lnoinc
	inc r15
.Lnoinc:
	add r13, 8
	add r14, 8
	dec r12
	jnz .Lloop
.Ldone:
	mov rax, r15
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbx
	ret
