packbits_decode:
    mov r8, rdx
    mov r9, rcx
    test rsi, rsi
    jz .Ldone
.Lloop:
    test rsi, rsi
    jz .Ldone
    mov r11b, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp r11b, 128
    je .Lcontinue
    jb .Lliteral
    # repeat packet
    movzx eax, r11b
    mov ecx, 257
    sub ecx, eax
    test rsi, rsi
    jz .Lerror_src
    mov r10b, byte ptr [rdi]
    inc rdi
    dec rsi
    cmp ecx, r9d
    jg .Lrepeat_exceeds
    mov eax, ecx
    sub r9d, ecx
    test eax, eax
    jz .Lcontinue
.Lrepeat_loop:
    mov byte ptr [r8], r10b
    inc r8
    dec eax
    jnz .Lrepeat_loop
    jmp .Lcontinue
.Lliteral:
    movzx eax, r11b
    inc eax
    cmp rsi, rax
    jl .Lerror_src
    cmp eax, r9d
    jg .Lliteral_exceeds
    sub r9d, eax
    sub rsi, rax
    test eax, eax
    jz .Lcontinue
.Lliteral_loop:
    mov cl, byte ptr [rdi]
    mov byte ptr [r8], cl
    inc rdi
    inc r8
    dec eax
    jnz .Lliteral_loop
    jmp .Lcontinue
.Lcontinue:
    jmp .Lloop
.Lliteral_exceeds:
    test r9, r9
    jz .Lreturn_minus1
    mov ecx, r9d
.Lliteral_exceeds_loop:
    mov al, byte ptr [rdi]
    mov byte ptr [r8], al
    inc rdi
    inc r8
    dec ecx
    jnz .Lliteral_exceeds_loop
.Lreturn_minus1:
    mov rax, -1
    ret
.Lrepeat_exceeds:
    test r9, r9
    jz .Lreturn_minus1
    mov ecx, r9d
.Lrepeat_exceeds_loop:
    mov byte ptr [r8], r10b
    inc r8
    dec ecx
    jnz .Lrepeat_exceeds_loop
    jmp .Lreturn_minus1
.Lerror_src:
    mov rax, -2
    ret
.Ldone:
    mov rax, r8
    sub rax, rdx
    ret
