qsort_pairs:
	stp	x29, x30, [sp, #-16]!
	stp	x19, x20, [sp, #-16]!
	stp	x21, x22, [sp, #-16]!
	stp	x23, x24, [sp, #-16]!

	mov	x19, x0
	mov	x20, x1
	cmp	x20, #1
	ble	.Lreturn_zero

	// pivot index = n/2
	lsr	x21, x20, #1
	add	x9, x19, x21, lsl #4
	ldr	x23, [x9]          // pivot_key
	ldr	x24, [x9, #8]      // pivot_val

	mov	x21, #0            // i
	sub	x22, x20, #1       // j

.Louter:
	// increment i while record < pivot
.Linc_i_loop:
	add	x9, x19, x21, lsl #4
	ldr	x10, [x9]
	ldr	x11, [x9, #8]
	cmp	x10, x23
	b.lt	.Linc_i_next
	b.gt	.Linc_i_break
	cmp	x11, x24
	b.lt	.Linc_i_next
.Linc_i_break:
	// decrement j while record > pivot
.Ldec_j_loop:
	add	x9, x19, x22, lsl #4
	ldr	x10, [x9]
	ldr	x11, [x9, #8]
	cmp	x10, x23
	b.gt	.Ldec_j_next
	b.lt	.Ldec_j_break
	cmp	x11, x24
	b.gt	.Ldec_j_next
.Ldec_j_break:
	// check if i >= j
	cmp	x21, x22
	b.ge	.Lpartition_done

	// swap records i and j
	add	x9, x19, x21, lsl #4
	ldr	x10, [x9]
	ldr	x11, [x9, #8]
	add	x12, x19, x22, lsl #4
	ldr	x13, [x12]
	ldr	x14, [x12, #8]
	str	x13, [x9]
	str	x14, [x9, #8]
	str	x10, [x12]
	str	x11, [x12, #8]

	add	x21, x21, #1
	sub	x22, x22, #1
	b	.Louter

.Linc_i_next:
	add	x21, x21, #1
	b	.Linc_i_loop

.Ldec_j_next:
	sub	x22, x22, #1
	b	.Ldec_j_loop

.Lpartition_done:
	// left_size = j+1
	mov	x21, x22
	add	x21, x21, #1
	// right_size = n - left_size
	sub	x24, x20, x21
	// right_base = base + left_size * 16
	add	x23, x19, x21, lsl #4

	// sort left part
	cmp	x21, #1
	ble	.Lskip_left
	mov	x0, x19
	mov	x1, x21
	bl	qsort_pairs
.Lskip_left:

	// sort right part
	cmp	x24, #1
	ble	.Lskip_right
	mov	x0, x23
	mov	x1, x24
	bl	qsort_pairs
.Lskip_right:

.Lreturn_zero:
	mov	x0, #0
	ldp	x23, x24, [sp], #16
	ldp	x21, x22, [sp], #16
	ldp	x19, x20, [sp], #16
	ldp	x29, x30, [sp], #16
	ret
