qsort_pairs:
	cmp	x1, #1
	ble	.Lbase_case

	sub	sp, sp, #64
	stp	x29, x30, [sp, #48]
	stp	x19, x20, [sp, #32]
	stp	x21, x22, [sp, #16]
	stp	x23, x24, [sp, #0]

	mov	x19, x0
	mov	x20, x1

	// pivot = last record
	add	x9, x19, x20, lsl #4
	sub	x9, x9, #16
	ldr	x21, [x9]          // pivot_key
	ldr	x22, [x9, #8]      // pivot_val

	mov	x23, #0            // i = 0
	mov	x24, #0            // j = 0
	sub	x10, x20, #1       // n-1 (loop limit)

.Lloop_j:
	cmp	x24, x10
	b.ge	.Lpartition_done

	// load record j
	add	x9, x19, x24, lsl #4
	ldr	x11, [x9]
	ldr	x12, [x9, #8]

	// compare with pivot
	cmp	x11, x21
	b.lt	.Lless
	b.gt	.Lnot_less
	cmp	x12, x22
	b.lt	.Lless
.Lnot_less:
	add	x24, x24, #1
	b	.Lloop_j

.Lless:
	// swap record i and j if i != j
	cmp	x23, x24
	beq	.Lno_swap
	add	x9, x19, x23, lsl #4
	ldr	x13, [x9]
	ldr	x14, [x9, #8]
	str	x11, [x9]
	str	x12, [x9, #8]
	add	x15, x19, x24, lsl #4
	str	x13, [x15]
	str	x14, [x15, #8]
.Lno_swap:
	add	x23, x23, #1
	add	x24, x24, #1
	b	.Lloop_j

.Lpartition_done:
	// swap pivot (last record) with record i
	sub	x10, x20, #1       // n-1
	cmp	x23, x10
	beq	.Lpivot_in_place
	add	x9, x19, x20, lsl #4
	sub	x9, x9, #16        // address of pivot
	add	x10, x19, x23, lsl #4  // address of record i
	ldr	x11, [x9]
	ldr	x12, [x9, #8]
	ldr	x13, [x10]
	ldr	x14, [x10, #8]
	str	x11, [x10]
	str	x12, [x10, #8]
	str	x13, [x9]
	str	x14, [x9, #8]
.Lpivot_in_place:
	// left_size = i
	mov	x24, x23
	// right_size = n - i - 1
	sub	x21, x20, x23
	sub	x21, x21, #1
	// right_base = base + (i+1)*16
	add	x22, x19, x23, lsl #4
	add	x22, x22, #16

	// sort left part
	cmp	x24, #1
	ble	.Lskip_left
	mov	x0, x19
	mov	x1, x24
	bl	qsort_pairs
.Lskip_left:

	// sort right part
	cmp	x21, #1
	ble	.Lskip_right
	mov	x0, x22
	mov	x1, x21
	bl	qsort_pairs
.Lskip_right:

	ldp	x23, x24, [sp, #0]
	ldp	x21, x22, [sp, #16]
	ldp	x19, x20, [sp, #32]
	ldp	x29, x30, [sp, #48]
	add	sp, sp, #64
.Lbase_case:
	mov	x0, #0
	ret
