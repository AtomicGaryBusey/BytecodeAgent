export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    %i =l copy 0
    %count =l copy 0
    jmp @loop

@loop
    %cond =w csltl %i, %n
    jnz %cond, @body, @exit

@body
    %offset =l shl %i, 3
    %addr_src =l add %src, %offset
    %x =l loadl %addr_src
    %y =l call %f(l %x)
    %addr_dst =l add %dst, %offset
    storel %addr_dst, %y
    %cond2 =w csltl %y, %x
    %cond2_l =l zextw %cond2
    %count =l add %count, %cond2_l
    %i =l add %i, 1
    jmp @loop

@exit
    ret %count
}
