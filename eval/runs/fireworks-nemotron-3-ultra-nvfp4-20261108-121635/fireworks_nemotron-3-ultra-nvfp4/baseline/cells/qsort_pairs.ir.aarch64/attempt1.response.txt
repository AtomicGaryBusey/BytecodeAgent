export function l $qsort_pairs(l %a, l %n) {
@start
    %n_le_1 =w clel %n, 1
    jnz %n_le_1, @return_zero

    %slot_a =l alloc 8
    %slot_n =l alloc 8
    %slot_pi =l alloc 8
    %slot_pk =l alloc 8
    %slot_pv =l alloc 8
    %slot_pf =l alloc 8

    store %slot_a, %a
    store %slot_n, %n

    %pivot_idx =l sub %n, 1
    store %slot_pi, %pivot_idx

    %pivot_idx_shl =l shl %pivot_idx, 1
    %pivot_key_ptr =l add %a, %pivot_idx_shl
    %pivot_key =l load %pivot_key_ptr
    store %slot_pk, %pivot_key

    %pivot_val_ptr =l add %pivot_key_ptr, 8
    %pivot_val =l load %pivot_val_ptr
    store %slot_pv, %pivot_val

    jmp @loop_head(-1, 0)

@loop_head %i, %j
    %pivot_idx =l load %slot_pi
    %j_lt_pi =w csll %j, %pivot_idx
    jnz %j_lt_pi, @loop_body(%i, %j), @loop_end(%i)

@loop_body %i, %j
    %a =l load %slot_a
    %pivot_key =l load %slot_pk
    %pivot_val =l load %slot_pv

    %j_shl =l shl %j, 1
    %key_j_ptr =l add %a, %j_shl
    %key_j =l load %key_j_ptr
    %val_j_ptr =l add %key_j_ptr, 8
    %val_j =l load %val_j_ptr

    %key_cmp =w csll %key_j, %pivot_key
    %key_eq =w ceql %key_j, %pivot_key
    %val_cmp =w csll %val_j, %pivot_val
    %val_eq =w ceql %val_j, %pivot_val
    %val_le =w or %val_cmp, %val_eq
    %key_eq_val_le =w and %key_eq, %val_le
    %le =w or %key_cmp, %key_eq_val_le

    jnz %le, @do_swap(%i, %j), @loop_inc(%i, %j)

@do_swap %i, %j
    %i_inc =l add %i, 1
    %i_ne_j =w cnel %i_inc, %j
    jnz %i_ne_j, @swap_pairs(%i_inc, %j), @loop_inc(%i_inc, %j)

@swap_pairs %i_inc, %j
    %a =l load %slot_a
    %i_inc_shl =l shl %i_inc, 1
    %key_i_ptr =l add %a, %i_inc_shl
    %key_i =l load %key_i_ptr
    %val_i_ptr =l add %key_i_ptr, 8
    %val_i =l load %val_i_ptr

    %j_shl =l shl %j, 1
    %key_j_ptr =l add %a, %j_shl
    %key_j =l load %key_j_ptr
    %val_j_ptr =l add %key_j_ptr, 8
    %val_j =l load %val_j_ptr

    store %key_i_ptr, %key_j
    store %val_i_ptr, %val_j
    store %key_j_ptr, %key_i
    store %val_j_ptr, %val_i

    jmp @loop_inc(%i_inc, %j)

@loop_inc %i, %j
    %j_inc =l add %j, 1
    jmp @loop_head(%i, %j_inc)

@loop_end %i
    %pivot_final =l add %i, 1
    store %slot_pf, %pivot_final

    %pivot_idx =l load %slot_pi
    %pf_ne_pi =w cnel %pivot_final, %pivot_idx
    jnz %pf_ne_pi, @swap_pivot, @after_swap_pivot

@swap_pivot
    %a =l load %slot_a
    %pivot_final =l load %slot_pf
    %pivot_idx =l load %slot_pi

    %pf_shl =l shl %pivot_final, 1
    %key_pf_ptr =l add %a, %pf_shl
    %key_pf =l load %key_pf_ptr
    %val_pf_ptr =l add %key_pf_ptr, 8
    %val_pf =l load %val_pf_ptr

    %pi_shl =l shl %pivot_idx, 1
    %key_pi_ptr =l add %a, %pi_shl
    %key_pi =l load %key_pi_ptr
    %val_pi_ptr =l add %key_pi_ptr, 8
    %val_pi =l load %val_pi_ptr

    store %key_pf_ptr, %key_pi
    store %val_pf_ptr, %val_pi
    store %key_pi_ptr, %key_pf
    store %val_pi_ptr, %val_pf

    jmp @after_swap_pivot

@after_swap_pivot
    %a =l load %slot_a
    %n =l load %slot_n
    %pivot_final =l load %slot_pf

    %left_n_le_1 =w clel %pivot_final, 1
    jnz %left_n_le_1, @right_part, @do_left_recurse

@do_left_recurse
    %_ =l call $qsort_pairs(l %a, l %pivot_final)
    jmp @right_part

@right_part
    %pf_plus_1 =l add %pivot_final, 1
    %right_n =l sub %n, %pf_plus_1
    %pf_plus_1_shl =l shl %pf_plus_1, 1
    %right_a =l add %a, %pf_plus_1_shl
    %right_n_le_1 =w clel %right_n, 1
    jnz %right_n_le_1, @return_zero, @do_right_recurse

@do_right_recurse
    %_ =l call $qsort_pairs(l %right_a, l %right_n)
    jmp @return_zero

@return_zero
    ret 0
}
