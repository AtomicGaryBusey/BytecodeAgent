export function l $max_byte(l %p, l %n) {
@start
    %n_zero =l ceql %n, 0
    jnz %n_zero, @ret_zero, @loop_init

@ret_zero
    ret 0

@loop_init
    %max =l copy 0
    %i =l copy 0
    jmp @loop_head

@loop_head
    %cond =l csltl %i, %n
    jnz %cond, @loop_body, @ret_max

@loop_body
    %addr =l add %p, %i
    %byte =l loadub %addr
    %cmp =l csgtl %byte, %max
    jnz %cmp, @update_max, @next_iter

@update_max
    %max =l copy %byte
    jmp @next_iter

@next_iter
    %i =l add %i, 1
    jmp @loop_head

@ret_max
    ret %max
}
