export function l $collatz_max(l %n) {
@start
    %max_steps = copy 0
    %k = copy 1
    jmp @loop_k_cond(%max_steps, %k)

@loop_k_cond(l %max_steps, l %k)
    %cond = slel %k, %n
    jnz %cond, @loop_k_body(%max_steps, %k), @exit(%max_steps)

@loop_k_body(l %max_steps, l %k)
    %steps0 = copy 0
    jmp @loop_x_cond(%k, %steps0, %max_steps, %k)

@loop_x_cond(l %x, l %steps, l %max_steps, l %k)
    %eq1 = ceql %x, 1
    jnz %eq1, @after_x_loop(%steps, %max_steps, %k), @loop_x_body(%x, %steps, %max_steps, %k)

@loop_x_body(l %x, l %steps, l %max_steps, l %k)
    %lsb = andl %x, 1
    %is_even = ceql %lsb, 0
    jnz %is_even, @even(%x, %steps, %max_steps, %k), @odd(%x, %steps, %max_steps, %k)

@even(l %x, l %steps, l %max_steps, l %k)
    %x_new = divl %x, 2
    %steps_new = addl %steps, 1
    jmp @loop_x_cond(%x_new, %steps_new, %max_steps, %k)

@odd(l %x, l %steps, l %max_steps, l %k)
    %x_new = mull %x, 3
    %x_new = addl %x_new, 1
    %steps_new = addl %steps, 1
    jmp @loop_x_cond(%x_new, %steps_new, %max_steps, %k)

@after_x_loop(l %steps, l %max_steps, l %k)
    %gt = sgtl %steps, %max_steps
    jnz %gt, @update_max(%steps, %k), @next_k(%max_steps, %k)

@update_max(l %steps, l %k)
    %max_steps_new = copy %steps
    jmp @next_k(%max_steps_new, %k)

@next_k(l %max_steps, l %k)
    %k_new = addl %k, 1
    jmp @loop_k_cond(%max_steps, %k_new)

@exit(l %max_steps)
    ret %max_steps
}
