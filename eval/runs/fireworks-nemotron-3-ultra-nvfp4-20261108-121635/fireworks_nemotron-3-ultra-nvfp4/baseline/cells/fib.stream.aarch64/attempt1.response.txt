fib:
	cmp x0, #0
	b.eq .Lreturn0
	cmp x0, #1
	b.eq .Lreturn1
	mov x9, #0          // prev = 0
	mov x10, #1         // curr = 1
	mov x11, #2         // i = 2
.Lloop:
	cmp x11, x0
	b.gt .Ldone
	add x12, x9, x10    // next = prev + curr (wraps modulo 2^64)
	mov x9, x10         // prev = curr
	mov x10, x12        // curr = next
	add x11, x11, #1    // i++
	b .Lloop
.Ldone:
	mov x0, x10         // return curr
	ret
.Lreturn0:
	mov x0, #0
	ret
.Lreturn1:
	mov x0, #1
	ret
