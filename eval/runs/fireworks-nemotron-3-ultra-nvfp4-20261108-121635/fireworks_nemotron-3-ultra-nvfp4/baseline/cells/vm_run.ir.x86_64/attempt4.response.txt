export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %c0 =l copy 0
    %c1 =l copy 1
    %c2 =l copy 2
    %c3 =l copy 3
    %c4 =l copy 4
    %c5 =l copy 5
    %c6 =l copy 6
    %c7 =l copy 7
    %c8 =l copy 8
    %c9 =l copy 9
    %c10 =l copy 10
    %c11 =l copy 11
    %pc =l copy %c0
    %sp =l copy %c0
    jmp @loop

@loop
    %cond =w cult %pc, %len
    jnz %cond, @fetch
    jmp @end

@fetch
    %pc8 =l shl %pc, %c3
    %opcode_addr =l add %code, %pc8
    %opcode =l loadl %opcode_addr
    %is_halt =w ceq %opcode, %c0
    jnz %is_halt, @halt
    %is_push =w ceq %opcode, %c1
    jnz %is_push, @push
    %is_add =w ceq %opcode, %c2
    jnz %is_add, @add
    %is_sub =w ceq %opcode, %c3
    jnz %is_sub, @sub
    %is_mul =w ceq %opcode, %c4
    jnz %is_mul, @mul
    %is_dup =w ceq %opcode, %c5
    jnz %is_dup, @dup
    %is_swap =w ceq %opcode, %c6
    jnz %is_swap, @swap
    %is_drop =w ceq %opcode, %c7
    jnz %is_drop, @drop
    %is_over =w ceq %opcode, %c8
    jnz %is_over, @over
    %is_lt =w ceq %opcode, %c9
    jnz %is_lt, @lt
    %is_jz =w ceq %opcode, %c10
    jnz %is_jz, @jz
    %is_jmp =w ceq %opcode, %c11
    jnz %is_jmp, @jmp
    jmp @default

@halt
    jmp @end

@push
    %pc1 =l add %pc, %c1
    %pc1_8 =l shl %pc1, %c3
    %operand_addr =l add %code, %pc1_8
    %operand =l loadl %operand_addr
    %sp8 =l shl %sp, %c3
    %sp_addr =l add %stack, %sp8
    storel %operand, %sp_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c2
    jmp @loop

@add
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %b_addr =l add %stack, %sp8
    %b =l loadl %b_addr
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %a_addr =l add %stack, %sp8
    %a =l loadl %a_addr
    %res =l add %a, %b
    %sp8 =l shl %sp, %c3
    %res_addr =l add %stack, %sp8
    storel %res, %res_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@sub
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %b_addr =l add %stack, %sp8
    %b =l loadl %b_addr
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %a_addr =l add %stack, %sp8
    %a =l loadl %a_addr
    %res =l sub %a, %b
    %sp8 =l shl %sp, %c3
    %res_addr =l add %stack, %sp8
    storel %res, %res_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@mul
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %b_addr =l add %stack, %sp8
    %b =l loadl %b_addr
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %a_addr =l add %stack, %sp8
    %a =l loadl %a_addr
    %res =l mul %a, %b
    %sp8 =l shl %sp, %c3
    %res_addr =l add %stack, %sp8
    storel %res, %res_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@dup
    %sp_minus1 =l sub %sp, %c1
    %sp_minus1_8 =l shl %sp_minus1, %c3
    %val_addr =l add %stack, %sp_minus1_8
    %val =l loadl %val_addr
    %sp8 =l shl %sp, %c3
    %sp_addr =l add %stack, %sp8
    storel %val, %sp_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@swap
    %sp_minus1 =l sub %sp, %c1
    %sp_minus2 =l sub %sp, %c2
    %sp_minus1_8 =l shl %sp_minus1, %c3
    %sp_minus2_8 =l shl %sp_minus2, %c3
    %top_addr =l add %stack, %sp_minus1_8
    %next_addr =l add %stack, %sp_minus2_8
    %top =l loadl %top_addr
    %next =l loadl %next_addr
    storel %top, %next_addr
    storel %next, %top_addr
    %pc =l add %pc, %c1
    jmp @loop

@drop
    %sp =l sub %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@over
    %sp_minus2 =l sub %sp, %c2
    %sp_minus2_8 =l shl %sp_minus2, %c3
    %val_addr =l add %stack, %sp_minus2_8
    %val =l loadl %val_addr
    %sp8 =l shl %sp, %c3
    %sp_addr =l add %stack, %sp8
    storel %val, %sp_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@lt
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %b_addr =l add %stack, %sp8
    %b =l loadl %b_addr
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %a_addr =l add %stack, %sp8
    %a =l loadl %a_addr
    %cmp =w cslt %a, %b
    %res =l zextw %cmp
    %sp8 =l shl %sp, %c3
    %res_addr =l add %stack, %sp8
    storel %res, %res_addr
    %sp =l add %sp, %c1
    %pc =l add %pc, %c1
    jmp @loop

@jz
    %sp =l sub %sp, %c1
    %sp8 =l shl %sp, %c3
    %v_addr =l add %stack, %sp8
    %v =l loadl %v_addr
    %is_zero =w ceq %v, %c0
    jnz %is_zero, @jz_taken
    %pc =l add %pc, %c2
    jmp @loop

@jz_taken
    %pc1 =l add %pc, %c1
    %pc1_8 =l shl %pc1, %c3
    %operand_addr =l add %code, %pc1_8
    %operand =l loadl %operand_addr
    %pc =l copy %operand
    jmp @loop

@jmp
    %pc1 =l add %pc, %c1
    %pc1_8 =l shl %pc1, %c3
    %operand_addr =l add %code, %pc1_8
    %operand =l loadl %operand_addr
    %pc =l copy %operand
    jmp @loop

@default
    %pc =l add %pc, %c1
    jmp @loop

@end
    %cond =w cult %c0, %sp
    jnz %cond, @load_top
    %ret =l copy %c0
    jmp @zero_stack

@load_top
    %sp_minus1 =l sub %sp, %c1
    %sp_minus1_8 =l shl %sp_minus1, %c3
    %top_addr =l add %stack, %sp_minus1_8
    %ret =l loadl %top_addr
    jmp @zero_stack

@zero_stack
    %i =l copy %sp
    jmp @zero_loop_head

@zero_loop_head
    %cond =w cult %i, %cap
    jnz %cond, @zero_body
    jmp @return

@zero_body
    %i8 =l shl %i, %c3
    %addr =l add %stack, %i8
    storel %c0, %addr
    %i =l add %i, %c1
    jmp @zero_loop_head

@return
    ret %ret
}
