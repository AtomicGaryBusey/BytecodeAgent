export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %pc = l 0
    %sp = l 0
    jmp @loop

@loop
    %cond = cultl %pc, %len
    jnz %cond, @fetch
    jmp @end

@fetch
    %opcode_addr = add %code, shl %pc, 3
    %opcode = loadl %opcode_addr
    %is_halt = ceql %opcode, $0
    jnz %is_halt, @halt
    %is_push = ceql %opcode, $1
    jnz %is_push, @push
    %is_add = ceql %opcode, $2
    jnz %is_add, @add
    %is_sub = ceql %opcode, $3
    jnz %is_sub, @sub
    %is_mul = ceql %opcode, $4
    jnz %is_mul, @mul
    %is_dup = ceql %opcode, $5
    jnz %is_dup, @dup
    %is_swap = ceql %opcode, $6
    jnz %is_swap, @swap
    %is_drop = ceql %opcode, $7
    jnz %is_drop, @drop
    %is_over = ceql %opcode, $8
    jnz %is_over, @over
    %is_lt = ceql %opcode, $9
    jnz %is_lt, @lt
    %is_jz = ceql %opcode, $10
    jnz %is_jz, @jz
    %is_jmp = ceql %opcode, $11
    jnz %is_jmp, @jmp
    jmp @default

@halt
    jmp @end

@push
    %pc_plus1 = add %pc, 1
    %operand_addr = add %code, shl %pc_plus1, 3
    %operand = loadl %operand_addr
    %sp_addr = add %stack, shl %sp, 3
    storel %operand, %sp_addr
    %sp = add %sp, 1
    %pc = add %pc, 2
    jmp @loop

@add
    %sp = sub %sp, 1
    %b_addr = add %stack, shl %sp, 3
    %b = loadl %b_addr
    %sp = sub %sp, 1
    %a_addr = add %stack, shl %sp, 3
    %a = loadl %a_addr
    %res = add %a, %b
    %res_addr = add %stack, shl %sp, 3
    storel %res, %res_addr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@sub
    %sp = sub %sp, 1
    %b_addr = add %stack, shl %sp, 3
    %b = loadl %b_addr
    %sp = sub %sp, 1
    %a_addr = add %stack, shl %sp, 3
    %a = loadl %a_addr
    %res = sub %a, %b
    %res_addr = add %stack, shl %sp, 3
    storel %res, %res_addr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@mul
    %sp = sub %sp, 1
    %b_addr = add %stack, shl %sp, 3
    %b = loadl %b_addr
    %sp = sub %sp, 1
    %a_addr = add %stack, shl %sp, 3
    %a = loadl %a_addr
    %res = mul %a, %b
    %res_addr = add %stack, shl %sp, 3
    storel %res, %res_addr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@dup
    %sp_minus1 = sub %sp, 1
    %val_addr = add %stack, shl %sp_minus1, 3
    %val = loadl %val_addr
    %sp_addr = add %stack, shl %sp, 3
    storel %val, %sp_addr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@swap
    %sp_minus1 = sub %sp, 1
    %sp_minus2 = sub %sp, 2
    %top_addr = add %stack, shl %sp_minus1, 3
    %next_addr = add %stack, shl %sp_minus2, 3
    %top = loadl %top_addr
    %next = loadl %next_addr
    storel %top, %next_addr
    storel %next, %top_addr
    %pc = add %pc, 1
    jmp @loop

@drop
    %sp = sub %sp, 1
    %pc = add %pc, 1
    jmp @loop

@over
    %sp_minus2 = sub %sp, 2
    %val_addr = add %stack, shl %sp_minus2, 3
    %val = loadl %val_addr
    %sp_addr = add %stack, shl %sp, 3
    storel %val, %sp_addr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@lt
    %sp = sub %sp, 1
    %b_addr = add %stack, shl %sp, 3
    %b = loadl %b_addr
    %sp = sub %sp, 1
    %a_addr = add %stack, shl %sp, 3
    %a = loadl %a_addr
    %cmp = csltl %a, %b
    %res = zextw %cmp
    %res_addr = add %stack, shl %sp, 3
    storel %res, %res_addr
    %sp = add %sp, 1
    %pc = add %pc, 1
    jmp @loop

@jz
    %sp = sub %sp, 1
    %v_addr = add %stack, shl %sp, 3
    %v = loadl %v_addr
    %is_zero = ceql %v, $0
    jnz %is_zero, @jz_taken
    %pc = add %pc, 2
    jmp @loop

@jz_taken
    %pc_plus1 = add %pc, 1
    %operand_addr = add %code, shl %pc_plus1, 3
    %operand = loadl %operand_addr
    %pc = %operand
    jmp @loop

@jmp
    %pc_plus1 = add %pc, 1
    %operand_addr = add %code, shl %pc_plus1, 3
    %operand = loadl %operand_addr
    %pc = %operand
    jmp @loop

@default
    %pc = add %pc, 1
    jmp @loop

@end
    %cond = csgel %sp, $1
    jnz %cond, @load_top
    %ret = l 0
    jmp @zero_stack

@load_top
    %sp_minus1 = sub %sp, 1
    %top_addr = add %stack, shl %sp_minus1, 3
    %ret = loadl %top_addr
    jmp @zero_stack

@zero_stack
    %i = %sp
    jmp @zero_loop_head

@zero_loop_head
    %cond = cultl %i, %cap
    jnz %cond, @zero_body
    jmp @return

@zero_body
    %addr = add %stack, shl %i, 3
    storel $0, %addr
    %i = add %i, 1
    jmp @zero_loop_head

@return
    ret %ret
}
