export function l $qsort_pairs(l %a, l %n) {
@start
    %one =l copy 1
    %zero =l copy 0
    %two =l copy 2
    %sixteen =l copy 16
    %eight =l copy 8
    %n_le_1 =w cle %n, %one
    jnz %n_le_1, @return_zero
    %pivot_idx =l udiv %n, %two
    %pivot_offset =l mul %pivot_idx, %sixteen
    %pivot_key_addr =l add %a, %pivot_offset
    %pivot_key =l loadl %pivot_key_addr, 0
    %pivot_val_addr =l add %pivot_key_addr, %eight
    %pivot_val =l loadl %pivot_val_addr, 0
    %j_init =l sub %n, %one
    jmp @partition_loop

@partition_loop
    %i =l phi @start %zero, @advance_i %i_adv, @retreat_j %i_ret, @do_swap %i_swp
    %j =l phi @start %j_init, @advance_i %j_adv, @retreat_j %j_ret, @do_swap %j_swp
    %i_le_j =w cle %i, %j
    jnz %i_le_j, @check_i
    jmp @partition_done

@check_i
    %i_offset =l mul %i, %sixteen
    %key_i_addr =l add %a, %i_offset
    %key_i =l loadl %key_i_addr, 0
    %val_i_addr =l add %key_i_addr, %eight
    %val_i =l loadl %val_i_addr, 0
    %key_lt =w csl %key_i, %pivot_key
    %key_eq =w ceq %key_i, %pivot_key
    %val_lt =w csl %val_i, %pivot_val
    %key_eq_and_val_lt =w and %key_eq, %val_lt
    %record_lt =w or %key_lt, %key_eq_and_val_lt
    jnz %record_lt, @advance_i
    jmp @check_j

@advance_i
    %i_adv =l add %i, %one
    %j_adv =l copy %j
    jmp @partition_loop

@check_j
    %j_offset =l mul %j, %sixteen
    %key_j_addr =l add %a, %j_offset
    %key_j =l loadl %key_j_addr, 0
    %val_j_addr =l add %key_j_addr, %eight
    %val_j =l loadl %val_j_addr, 0
    %key_gt =w csg %key_j, %pivot_key
    %key_eq_j =w ceq %key_j, %pivot_key
    %val_gt =w csg %val_j, %pivot_val
    %key_eq_and_val_gt =w and %key_eq_j, %val_gt
    %record_gt =w or %key_gt, %key_eq_and_val_gt
    jnz %record_gt, @retreat_j
    jmp @do_swap

@retreat_j
    %i_ret =l copy %i
    %j_ret =l sub %j, %one
    jmp @partition_loop

@do_swap
    %i_offset2 =l mul %i, %sixteen
    %key_i_addr2 =l add %a, %i_offset2
    %val_i_addr2 =l add %key_i_addr2, %eight
    %key_i2 =l loadl %key_i_addr2, 0
    %val_i2 =l loadl %val_i_addr2, 0
    %j_offset2 =l mul %j, %sixteen
    %key_j_addr2 =l add %a, %j_offset2
    %val_j_addr2 =l add %key_j_addr2, %eight
    %key_j2 =l loadl %key_j_addr2, 0
    %val_j2 =l loadl %val_j_addr2, 0
    storel %key_i_addr2, %key_j2
    storel %val_i_addr2, %val_j2
    storel %key_j_addr2, %key_i2
    storel %val_j_addr2, %val_i2
    %i_swp =l add %i, %one
    %j_swp =l sub %j, %one
    jmp @partition_loop

@partition_done
    %left_n =l add %j, %one
    %right_n =l sub %n, %left_n
    %left_gt_1 =w csg %left_n, %one
    jnz %left_gt_1, @sort_left
    jmp @check_right

@sort_left
    %r_left =l call $qsort_pairs(l %a, l %left_n)
    jmp @check_right

@check_right
    %right_gt_1 =w csg %right_n, %one
    jnz %right_gt_1, @sort_right
    jmp @return_zero

@sort_right
    %right_offset =l mul %left_n, %sixteen
    %right_ptr =l add %a, %right_offset
    %r_right =l call $qsort_pairs(l %right_ptr, l %right_n)
    jmp @return_zero

@return_zero
    ret %zero
}
