stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24
    mov r12, rdi
    mov r13, rdx
    mov r14, rcx
    mov r15, rsi
    mov rdi, [r12]
    call r13
    mov rbx, rax
    mov rbp, rax
    mov [rsp], rax
    and rax, 1
    mov [rsp+8], rax
    add r12, 8
    dec r15
    jz .Lstore_results
.Lloop:
    mov rdi, [r12]
    call r13
    cmp rax, rbx
    cmovl rbx, rax
    cmp rax, rbp
    cmovg rbp, rax
    add [rsp], rax
    test rax, 1
    jz .Lno_odd
    inc qword ptr [rsp+8]
.Lno_odd:
    add r12, 8
    dec r15
    jnz .Lloop
.Lstore_results:
    mov rax, rbx
    mov [r14], rax
    mov rax, rbp
    mov [r14+8], rax
    mov rax, [rsp]
    mov [r14+16], rax
    mov rax, [rsp+8]
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
