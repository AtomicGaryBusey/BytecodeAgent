packbits_encode:
	mov x4, x0              // src base
	mov x5, x1              // n
	mov x6, xzr             // i = 0
	mov x7, x2              // out current
	mov x8, x3              // cap remaining
	mov x9, x2              // out base (for written calculation)
	cbz x5, .Lreturn_zero   // if n == 0 return 0

.Lmain_loop:
	cmp x6, x5
	b.ge .Lreturn_done      // i >= n -> done

	// compute run length at i
	mov x10, #1             // r = 1
	ldrb w11, [x4, x6]      // w11 = src[i]
.Lrun_loop:
	add x12, x6, x10        // i + r
	cmp x12, x5
	b.ge .Lrun_done         // i+r >= n
	cmp x10, #128
	b.ge .Lrun_done         // r >= 128
	ldrb w13, [x4, x12]     // src[i+r]
	cmp w13, w11
	b.ne .Lrun_done
	add x10, x10, #1        // r++
	b .Lrun_loop

.Lrun_done:
	cmp x10, #3
	b.ge .Lrepeat_packet    // if r >= 3 -> repeat packet

	// literal packet
	mov x11, x6             // j = i
.Lliteral_loop:
	cmp x11, x5
	b.ge .Lliteral_done     // j >= n
	sub x12, x11, x6        // j - i
	cmp x12, #128
	b.ge .Lliteral_done     // j-i >= 128

	// compute run at j
	mov x13, #1             // r2 = 1
	ldrb w14, [x4, x11]     // src[j]
.Lrun2_loop:
	add x15, x11, x13       // j + r2
	cmp x15, x5
	b.ge .Lrun2_done
	cmp x13, #128
	b.ge .Lrun2_done
	ldrb w16, [x4, x15]
	cmp w16, w14
	b.ne .Lrun2_done
	add x13, x13, #1
	b .Lrun2_loop
.Lrun2_done:
	cmp x13, #3
	b.ge .Lliteral_done     // run(j) >= 3 -> stop literal
	add x11, x11, #1        // j++
	b .Lliteral_loop

.Lliteral_done:
	sub x12, x11, x6        // L = j - i
	// check capacity: need 1 + L bytes
	add x13, x12, #1        // 1 + L
	cmp x8, x13
	b.lt .Lerror            // cap < 1+L -> error
	// write header L-1
	sub x14, x12, #1        // L-1
	strb w14, [x7], #1      // write header, advance out
	sub x8, x8, #1          // cap--
	// copy L bytes from src[i] to out
	mov x15, x6             // src index
.Lcopy_loop:
	cmp x15, x11
	b.ge .Lcopy_done
	ldrb w16, [x4, x15]
	strb w16, [x7], #1
	add x15, x15, #1
	b .Lcopy_loop
.Lcopy_done:
	sub x8, x8, x12         // cap -= L
	mov x6, x11             // i = j
	b .Lmain_loop

.Lrepeat_packet:
	// need 2 bytes
	cmp x8, #2
	b.lt .Lerror
	// header = 257 - r
	mov x12, #257
	sub x12, x12, x10       // 257 - r
	strb w12, [x7], #1      // write header
	strb w11, [x7], #1      // write byte value
	sub x8, x8, #2          // cap -= 2
	add x6, x6, x10         // i += r
	b .Lmain_loop

.Lerror:
	mov x0, #-1
	ret

.Lreturn_zero:
	mov x0, #0
	ret

.Lreturn_done:
	sub x0, x7, x9          // written = out_current - out_base
	ret
