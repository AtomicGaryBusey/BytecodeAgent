export function l $gcd(l %a, l %b) {
@start
    jmp @loop(%a, %b)

@loop
    %a_phi = phi l %a, %a_next
    %b_phi = phi l %b, %b_next
    %b_zero = ceql %b_phi, 0
    jnz %b_zero, @done, @compute

@compute
    %rem = modl %a_phi, %b_phi
    %a_next = copy %b_phi
    %b_next = copy %rem
    jmp @loop(%a_next, %b_next)

@done
    ret %a_phi
}
