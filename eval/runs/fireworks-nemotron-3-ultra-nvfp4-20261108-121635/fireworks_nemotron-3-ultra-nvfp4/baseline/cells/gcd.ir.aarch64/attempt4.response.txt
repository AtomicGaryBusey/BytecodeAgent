export function l $gcd(l %a, l %b) {
@start
    jmp @loop(%a, %b)

@loop(l %a_phi, l %b_phi)
    %b_zero = ceql %b_phi, 0
    jnz %b_zero, @done(%a_phi), @compute(%a_phi, %b_phi)

@compute(l %a_phi, l %b_phi)
    %rem = modl %a_phi, %b_phi
    jmp @loop(%b_phi, %rem)

@done(l %result)
    ret %result
}
