export function l @fill_down(l %a, l %n, l %start) {
@start
    %nz = ceql %n, 0
    jnz %nz, @exit
    %i = copy 0
    jmp @loop
@loop
    %cond = csltl %i, %n
    jnz %cond, @body
    jmp @exit
@body
    %val = subl %start, %i
    %ptr = addl %a, %i
    storel %val, %ptr
    %i = addl %i, 1
    jmp @loop
@exit
    ret %n
}
