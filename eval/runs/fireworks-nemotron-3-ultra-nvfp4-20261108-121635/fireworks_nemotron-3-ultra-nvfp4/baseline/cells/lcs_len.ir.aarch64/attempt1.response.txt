export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
    %n8 =l mul %n, 8
    %n_plus_1 =l add %n, 1
    %row_ptr =l copy %row
    %j =l copy 0
    jmp @init_loop

@init_loop
    %cond =l csltl %j, %n_plus_1
    jnz %cond, @init_body, @init_done

@init_body
    storel 0, %row_ptr
    %row_ptr =l add %row_ptr, 8
    %j =l add %j, 1
    jmp @init_loop

@init_done
    %i =l copy 0
    jmp @outer_loop

@outer_loop
    %cond =l csltl %i, %m
    jnz %cond, @outer_body, @outer_done

@outer_body
    %diag =l copy 0
    %x_ptr =l add %x, %i
    %xi =w loadub %x_ptr
    %j =l copy 1
    %row_j_ptr =l add %row, 8
    %row_j_minus_1_ptr =l copy %row
    %y_ptr =l copy %y
    jmp @inner_loop

@inner_loop
    %cond =l csltl %j, %n_plus_1
    jnz %cond, @inner_body, @inner_done

@inner_body
    %up =l loadl %row_j_ptr
    %yj =w loadub %y_ptr
    %cmp =w ceqw %xi, %yj
    %row_j_minus_1 =l loadl %row_j_minus_1_ptr
    %cond_max =w csgel %up, %row_j_minus_1
    %max_val =l sel %cond_max, %up, %row_j_minus_1
    %diag_plus_1 =l add %diag, 1
    %new_val =l sel %cmp, %diag_plus_1, %max_val
    storel %new_val, %row_j_ptr
    %diag =l copy %up
    %row_j_ptr =l add %row_j_ptr, 8
    %row_j_minus_1_ptr =l add %row_j_minus_1_ptr, 8
    %y_ptr =l add %y_ptr, 1
    %j =l add %j, 1
    jmp @inner_loop

@inner_done
    %i =l add %i, 1
    jmp @outer_loop

@outer_done
    %result_ptr =l add %row, %n8
    %result =l loadl %result_ptr
    ret %result
}
