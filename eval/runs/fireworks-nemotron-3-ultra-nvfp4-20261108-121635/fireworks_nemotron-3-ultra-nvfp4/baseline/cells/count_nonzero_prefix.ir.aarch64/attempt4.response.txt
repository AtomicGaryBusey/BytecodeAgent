export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %zero = copy 0
    jmp @loop_cond

@loop_cond
    %i = phi l @start %zero @inc %i_next
    %cond = cltu %i %n
    jnz %cond @loop_body @end

@loop_body
    %addr = addl %a %i
    %val = loadl %addr
    %is_zero = ceql %val 0
    jnz %is_zero @end @inc

@inc
    %i_next = addl %i 1
    jmp @loop_cond

@end
    %res = phi l @loop_cond %i @loop_body %i
    ret %res
}
