collatz_max:
    mov rax, 0
    mov rcx, 1
    cmp rdi, 1
    jl .Lreturn
.Louter:
    mov r8, rcx
    xor r9, r9
.Linner:
    cmp r8, 1
    je .Linner_done
    test r8, 1
    jz .Leven
    lea r8, [r8 + r8*2 + 1]
    inc r9
    jmp .Linner
.Leven:
    shr r8, 1
    inc r9
    jmp .Linner
.Linner_done:
    cmp r9, rax
    cmovg rax, r9
    inc rcx
    cmp rcx, rdi
    jle .Louter
.Lreturn:
    ret
