export function l $isqrt(l %n) {
	@entry
	%low =l copy 0
	%high =l copy 4294967296
	jmp @loop

	@loop
	%cmp =w cultl %low %high
	jnz %cmp @body
	jmp @exit

	@body
	%mid =l addl %low %high
	%mid =l shrl %mid 1
	%sq =l mulu %mid %mid
	%cmp2 =w culel %sq %n
	jnz %cmp2 @le
	%high =l copy %mid
	jmp @loop

	@le
	%low =l addl %mid 1
	jmp @loop

	@exit
	%res =l subl %low 1
	ret %res
}
