export function l $isqrt(l %n) {
	@entry
	%low =l copy 0
	%high =l copy 4294967296
	jmp @loop

	@loop
	%cmp =w cltu %low %high
	jnz %cmp @body
	@exit
	%res =l subl %low 1
	ret %res

	@body
	%mid =l addl %low %high
	%mid =l shrl %mid 1
	%sq =l mulu %mid %mid
	%cmp2 =w cleu %sq %n
	jnz %cmp2 @le
	@gt
	%high =l copy %mid
	jmp @loop

	@le
	%low =l addl %mid 1
	jmp @loop
}
