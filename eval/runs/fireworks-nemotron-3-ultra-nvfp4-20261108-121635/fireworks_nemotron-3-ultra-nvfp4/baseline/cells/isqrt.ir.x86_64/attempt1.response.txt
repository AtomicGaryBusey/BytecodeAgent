export function l $isqrt(l %n) {
	@entry
	%low = copy 0
	%high = copy 4294967296
	jmp @loop

	@loop
	%cmp = cult %low %high
	jnz %cmp @body
	jmp @exit

	@body
	%mid = addl %low %high
	%mid = shr %mid 1
	%sq = umul %mid %mid
	%cmp2 = cule %sq %n
	jnz %cmp2 @le
	%high = copy %mid
	jmp @loop

	@le
	%low = addl %mid 1
	jmp @loop

	@exit
	%res = subl %low 1
	ret %res
}
