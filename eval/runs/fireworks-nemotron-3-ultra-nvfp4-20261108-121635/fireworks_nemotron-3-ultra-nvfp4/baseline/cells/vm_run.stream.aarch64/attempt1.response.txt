vm_run:
    mov x10, x0           // code_base
    mov x11, x1           // len
    mov x12, x2           // stack_base
    mov x13, x3           // cap
    mov x14, 0            // pc = 0
    mov x15, x12          // stk_ptr = stack_base (next free slot)

.Lloop:
    cmp x14, x11
    b.eq  .Lexit          // pc == len -> exit

    ldr x9, [x10, x14, lsl #3]   // load opcode

    cbz x9, .Lhalt        // 0 HALT
    cmp x9, 1
    b.eq  .Lpush
    cmp x9, 2
    b.eq  .Ladd
    cmp x9, 3
    b.eq  .Lsub
    cmp x9, 4
    b.eq  .Lmul
    cmp x9, 5
    b.eq  .Ldup
    cmp x9, 6
    b.eq  .Lswap
    cmp x9, 7
    b.eq  .Ldrop
    cmp x9, 8
    b.eq  .Lover
    cmp x9, 9
    b.eq  .Llt
    cmp x9, 10
    b.eq  .Ljz
    cmp x9, 11
    b.eq  .Ljmp

    add x14, x14, 1       // other (should not happen) pc += 1
    b .Lloop

.Lhalt:
    b .Lexit

.Lpush:
    add x16, x14, 1
    ldr x8, [x10, x16, lsl #3]   // operand v
    str x8, [x15], #8            // push v
    add x14, x14, 2              // pc += 2
    b .Lloop

.Ladd:
    sub x15, x15, 8
    ldr x8, [x15]                // pop b
    sub x15, x15, 8
    ldr x9, [x15]                // pop a
    add x8, x9, x8               // a + b
    str x8, [x15], #8            // push result
    add x14, x14, 1
    b .Lloop

.Lsub:
    sub x15, x15, 8
    ldr x8, [x15]                // pop b
    sub x15, x15, 8
    ldr x9, [x15]                // pop a
    sub x8, x9, x8               // a - b
    str x8, [x15], #8
    add x14, x14, 1
    b .Lloop

.Lmul:
    sub x15, x15, 8
    ldr x8, [x15]                // pop b
    sub x15, x15, 8
    ldr x9, [x15]                // pop a
    mul x8, x9, x8               // a * b
    str x8, [x15], #8
    add x14, x14, 1
    b .Lloop

.Ldup:
    ldr x8, [x15, #-8]           // top
    str x8, [x15], #8            // push copy
    add x14, x14, 1
    b .Lloop

.Lswap:
    ldr x8, [x15, #-8]           // top
    ldr x9, [x15, #-16]          // second
    str x8, [x15, #-16]
    str x9, [x15, #-8]
    add x14, x14, 1
    b .Lloop

.Ldrop:
    sub x15, x15, 8              // pop and discard
    add x14, x14, 1
    b .Lloop

.Lover:
    ldr x8, [x15, #-16]          // second from top
    str x8, [x15], #8            // push copy
    add x14, x14, 1
    b .Lloop

.Llt:
    sub x15, x15, 8
    ldr x8, [x15]                // pop b
    sub x15, x15, 8
    ldr x9, [x15]                // pop a
    cmp x9, x8                   // a < b (signed)
    cset x8, lt                  // 1 if a<b else 0
    str x8, [x15], #8
    add x14, x14, 1
    b .Lloop

.Ljz:
    sub x15, x15, 8
    ldr x8, [x15]                // pop v
    cbz x8, .Ljz_taken
    add x14, x14, 2              // pc += 2
    b .Lloop
.Ljz_taken:
    add x16, x14, 1
    ldr x14, [x10, x16, lsl #3]  // pc = t
    b .Lloop

.Ljmp:
    add x16, x14, 1
    ldr x14, [x10, x16, lsl #3]  // pc = t
    b .Lloop

.Lexit:
    mov x0, 0                    // default return 0
    cmp x15, x12
    b.eq  .Lzero_loop            // empty stack
    ldr x0, [x15, #-8]           // top value

.Lzero_loop:
    add x16, x12, x13, lsl #3    // end = stack_base + cap*8
    cmp x15, x16
    b.ge  .Ldone
.Lzero_body:
    str xzr, [x15], #8           // zero remaining slots
    cmp x15, x16
    b.lt  .Lzero_body

.Ldone:
    ret
