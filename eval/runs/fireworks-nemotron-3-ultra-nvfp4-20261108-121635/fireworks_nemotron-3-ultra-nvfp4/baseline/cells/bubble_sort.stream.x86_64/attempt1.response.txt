bubble_sort:
	cmp rsi, 1
	jle .Ldone
	mov r10, rsi
	dec r10
	xor r8, r8
.Louter:
	mov r9, r10
	sub r9, r8
	test r9, r9
	jle .Louter_next
	xor rcx, rcx
.Linner:
	mov rax, [rdi + rcx*8]
	mov rdx, [rdi + rcx*8 + 8]
	cmp rax, rdx
	jle .Lno_swap
	mov [rdi + rcx*8], rdx
	mov [rdi + rcx*8 + 8], rax
.Lno_swap:
	inc rcx
	cmp rcx, r9
	jl .Linner
.Louter_next:
	inc r8
	cmp r8, r10
	jl .Louter
.Ldone:
	xor rax, rax
	ret
