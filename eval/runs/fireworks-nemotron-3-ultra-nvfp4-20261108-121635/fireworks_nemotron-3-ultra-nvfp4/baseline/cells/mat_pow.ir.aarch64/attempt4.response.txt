export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %zero =l 0
    %one =l 1
    %eight =l 8
    %k_eq_zero =w ceql %k, %zero
    jnz %k_eq_zero, @k_zero, @non_zero

@k_zero
    jmp @do_identity_and_zero

@non_zero
    %n_sq =l mull %n, %n
    %n_sq_bytes =l mull %n_sq, %eight
    %tmp_base =l copy %tmp
    %tmp_result =l add %tmp, %n_sq_bytes
    jmp @copy_a_loop_init

@copy_a_loop_init
    %idx_copy =l 0
    jmp @copy_a_loop_head

@copy_a_loop_head
    %idx_phi_ca =l phi @copy_a_loop_init %idx_copy, @copy_a_loop_body %idx_next_ca
    %cond_ca =w cultl %idx_phi_ca, %n_sq
    jnz %cond_ca, @copy_a_loop_body, @copy_a_loop_end

@copy_a_loop_body
    %idx_mul_ca =l mull %idx_phi_ca, %eight
    %src_ca =l add %a, %idx_mul_ca
    %val_ca =l loadl %src_ca
    %dst_ca =l add %tmp_base, %idx_mul_ca
    storel %val_ca, %dst_ca
    %idx_next_ca =l add %idx_phi_ca, %one
    jmp @copy_a_loop_head

@copy_a_loop_end
    jmp @init_identity_loop_init

@init_identity_loop_init
    %i_init =l 0
    jmp @init_i_loop_head

@init_i_loop_head
    %i_phi =l phi @init_identity_loop_init %i_init, @init_i_loop_end %i_next
    %cond_i =w cultl %i_phi, %n
    jnz %cond_i, @init_j_loop_init, @init_i_loop_end

@init_j_loop_init
    %j_init =l 0
    jmp @init_j_loop_head

@init_j_loop_head
    %j_phi =l phi @init_j_loop_init %j_init, @init_j_loop_end %j_next
    %cond_j =w cultl %j_phi, %n
    jnz %cond_j, @init_j_loop_body, @init_j_loop_end

@init_j_loop_body
    %tmp_mul_i =l mull %i_phi, %n
    %index =l add %tmp_mul_i, %j_phi
    %idx_mul =l mull %index, %eight
    %dst_ij =l add %tmp_result, %idx_mul
    %eq =w ceql %i_phi, %j_phi
    %val_ij =l zext %eq
    storel %val_ij, %dst_ij
    %j_next =l add %j_phi, %one
    jmp @init_j_loop_head

@init_j_loop_end
    %i_next =l add %i_phi, %one
    jmp @init_i_loop_head

@init_i_loop_end
    %k_current =l copy %k
    jmp @exp_loop_head

@exp_loop_head
    %k_phi =l phi @init_i_loop_end %k_current, @exp_loop_update_k %k_next
    %cond_k =w cugtl %k_phi, %zero
    jnz %cond_k, @exp_loop_body, @exp_loop_end

@exp_loop_body
    %k_lsb =l and %k_phi, %one
    %lsb_nz =w cugtl %k_lsb, %zero
    jnz %lsb_nz, @multiply_result_base, @skip_multiply_result

@multiply_result_base
    %src1_1 =l copy %tmp_result
    %src2_1 =l copy %tmp_base
    %dst_1 =l copy %out
    jmp @mul1_i_init

@mul1_i_init
    %i_1 =l 0
    jmp @mul1_i_head

@mul1_i_head
    %i_phi1 =l phi @mul1_i_init %i_1, @mul1_i_end %i_next1
    %cond_i1 =w cultl %i_phi1, %n
    jnz %cond_i1, @mul1_j_init, @mul1_done

@mul1_j_init
    %j_1 =l 0
    jmp @mul1_j_head

@mul1_j_head
    %j_phi1 =l phi @mul1_j_init %j_1, @mul1_j_end %j_next1
    %cond_j1 =w cultl %j_phi1, %n
    jnz %cond_j1, @mul1_t_init, @mul1_j_end

@mul1_t_init
    %sum_1 =l 0
    %t_1 =l 0
    jmp @mul1_t_head

@mul1_t_head
    %t_phi1 =l phi @mul1_t_init %t_1, @mul1_t_body %t_next1
    %sum_phi1 =l phi @mul1_t_init %sum_1, @mul1_t_body %sum_next1
    %cond_t1 =w cultl %t_phi1, %n
    jnz %cond_t1, @mul1_t_body, @mul1_t_end

@mul1_t_body
    %tmp_mul_i1 =l mull %i_phi1, %n
    %idx1_1 =l add %tmp_mul_i1, %t_phi1
    %idx1_mul =l mull %idx1_1, %eight
    %ptr1_1 =l add %src1_1, %idx1_mul
    %val1_1 =l loadl %ptr1_1
    %tmp_mul_t1 =l mull %t_phi1, %n
    %idx2_1 =l add %tmp_mul_t1, %j_phi1
    %idx2_mul =l mull %idx2_1, %eight
    %ptr2_1 =l add %src2_1, %idx2_mul
    %val2_1 =l loadl %ptr2_1
    %prod_1 =l mull %val1_1, %val2_1
    %sum_next1 =l add %sum_phi1, %prod_1
    %t_next1 =l add %t_phi1, %one
    jmp @mul1_t_head

@mul1_t_end
    %tmp_mul_i1_end =l mull %i_phi1, %n
    %idx_dst1 =l add %tmp_mul_i1_end, %j_phi1
    %idx_dst_mul =l mull %idx_dst1, %eight
    %ptr_dst1 =l add %dst_1, %idx_dst_mul
    storel %sum_phi1, %ptr_dst1
    jmp @mul1_j_end

@mul1_j_end
    %j_next1 =l add %j_phi1, %one
    jmp @mul1_j_head

@mul1_i_end
    %i_next1 =l add %i_phi1, %one
    jmp @mul1_i_head

@mul1_done
    jmp @copy_result_to_tmp_result

@copy_result_to_tmp_result
    %idx_cr =l 0
    jmp @copy_result_loop_head

@copy_result_loop_head
    %idx_phi_cr =l phi @copy_result_to_tmp_result %idx_cr, @copy_result_loop_body %idx_next_cr
    %cond_cr =w cultl %idx_phi_cr, %n_sq
    jnz %cond_cr, @copy_result_loop_body, @skip_multiply_result

@copy_result_loop_body
    %idx_mul_cr =l mull %idx_phi_cr, %eight
    %src_cr =l add %out, %idx_mul_cr
    %val_cr =l loadl %src_cr
    %dst_cr =l add %tmp_result, %idx_mul_cr
    storel %val_cr, %dst_cr
    %idx_next_cr =l add %idx_phi_cr, %one
    jmp @copy_result_loop_head

@skip_multiply_result
    jmp @multiply_base_base

@multiply_base_base
    %src1_2 =l copy %tmp_base
    %src2_2 =l copy %tmp_base
    %dst_2 =l copy %out
    jmp @mul2_i_init

@mul2_i_init
    %i_2 =l 0
    jmp @mul2_i_head

@mul2_i_head
    %i_phi2 =l phi @mul2_i_init %i_2, @mul2_i_end %i_next2
    %cond_i2 =w cultl %i_phi2, %n
    jnz %cond_i2, @mul2_j_init, @mul2_done

@mul2_j_init
    %j_2 =l 0
    jmp @mul2_j_head

@mul2_j_head
    %j_phi2 =l phi @mul2_j_init %j_2, @mul2_j_end %j_next2
    %cond_j2 =w cultl %j_phi2, %n
    jnz %cond_j2, @mul2_t_init, @mul2_j_end

@mul2_t_init
    %sum_2 =l 0
    %t_2 =l 0
    jmp @mul2_t_head

@mul2_t_head
    %t_phi2 =l phi @mul2_t_init %t_2, @mul2_t_body %t_next2
    %sum_phi2 =l phi @mul2_t_init %sum_2, @mul2_t_body %sum_next2
    %cond_t2 =w cultl %t_phi2, %n
    jnz %cond_t2, @mul2_t_body, @mul2_t_end

@mul2_t_body
    %tmp_mul_i2 =l mull %i_phi2, %n
    %idx1_2 =l add %tmp_mul_i2, %t_phi2
    %idx1_mul2 =l mull %idx1_2, %eight
    %ptr1_2 =l add %src1_2, %idx1_mul2
    %val1_2 =l loadl %ptr1_2
    %tmp_mul_t2 =l mull %t_phi2, %n
    %idx2_2 =l add %tmp_mul_t2, %j_phi2
    %idx2_mul2 =l mull %idx2_2, %eight
    %ptr2_2 =l add %src2_2, %idx2_mul2
    %val2_2 =l loadl %ptr2_2
    %prod_2 =l mull %val1_2, %val2_2
    %sum_next2 =l add %sum_phi2, %prod_2
    %t_next2 =l add %t_phi2, %one
    jmp @mul2_t_head

@mul2_t_end
    %tmp_mul_i2_end =l mull %i_phi2, %n
    %idx_dst2 =l add %tmp_mul_i2_end, %j_phi2
    %idx_dst_mul2 =l mull %idx_dst2, %eight
    %ptr_dst2 =l add %dst_2, %idx_dst_mul2
    storel %sum_phi2, %ptr_dst2
    jmp @mul2_j_end

@mul2_j_end
    %j_next2 =l add %j_phi2, %one
    jmp @mul2_j_head

@mul2_i_end
    %i_next2 =l add %i_phi2, %one
    jmp @mul2_i_head

@mul2_done
    jmp @copy_base_to_tmp_base

@copy_base_to_tmp_base
    %idx_cb =l 0
    jmp @copy_base_loop_head

@copy_base_loop_head
    %idx_phi_cb =l phi @copy_base_to_tmp_base %idx_cb, @copy_base_loop_body %idx_next_cb
    %cond_cb =w cultl %idx_phi_cb, %n_sq
    jnz %cond_cb, @copy_base_loop_body, @exp_loop_update_k

@copy_base_loop_body
    %idx_mul_cb =l mull %idx_phi_cb, %eight
    %src_cb =l add %out, %idx_mul_cb
    %val_cb =l loadl %src_cb
    %dst_cb =l add %tmp_base, %idx_mul_cb
    storel %val_cb, %dst_cb
    %idx_next_cb =l add %idx_phi_cb, %one
    jmp @copy_base_loop_head

@exp_loop_update_k
    %k_next =l shr %k_phi, %one
    jmp @exp_loop_head

@exp_loop_end
    %idx_cf =l 0
    jmp @copy_final_loop_head

@copy_final_loop_head
    %idx_phi_cf =l phi @exp_loop_end %idx_cf, @copy_final_loop_body %idx_next_cf
    %cond_cf =w cultl %idx_phi_cf, %n_sq
    jnz %cond_cf, @copy_final_loop_body, @zero_tmp

@copy_final_loop_body
    %idx_mul_cf =l mull %idx_phi_cf, %eight
    %src_cf =l add %tmp_result, %idx_mul_cf
    %val_cf =l loadl %src_cf
    %dst_cf =l add %out, %idx_mul_cf
    storel %val_cf, %dst_cf
    %idx_next_cf =l add %idx_phi_cf, %one
    jmp @copy_final_loop_head

@zero_tmp
    %total_tmp =l mull %n_sq, 2
    %idx_z =l 0
    jmp @zero_loop_head

@zero_loop_head
    %idx_phi_z =l phi @zero_tmp %idx_z, @zero_loop_body %idx_next_z
    %cond_z =w cultl %idx_phi_z, %total_tmp
    jnz %cond_z, @zero_loop_body, @return_zero

@zero_loop_body
    %idx_mul_z =l mull %idx_phi_z, %eight
    %ptr_z =l add %tmp, %idx_mul_z
    storel %zero, %ptr_z
    %idx_next_z =l add %idx_phi_z, %one
    jmp @zero_loop_head

@return_zero
    ret %zero

@do_identity_and_zero
    %n_sq_id =l mull %n, %n
    %i_id =l 0
    jmp @id_i_loop_head

@id_i_loop_head
    %i_phi_id =l phi @do_identity_and_zero %i_id, @id_i_loop_end %i_next_id
    %cond_i_id =w cultl %i_phi_id, %n
    jnz %cond_i_id, @id_j_loop_init, @id_i_loop_end

@id_j_loop_init
    %j_id =l 0
    jmp @id_j_loop_head

@id_j_loop_head
    %j_phi_id =l phi @id_j_loop_init %j_id, @id_j_loop_end %j_next_id
    %cond_j_id =w cultl %j_phi_id, %n
    jnz %cond_j_id, @id_j_loop_body, @id_j_loop_end

@id_j_loop_body
    %tmp_mul_i_id =l mull %i_phi_id, %n
    %index_id =l add %tmp_mul_i_id, %j_phi_id
    %idx_mul_id =l mull %index_id, %eight
    %ptr_id =l add %out, %idx_mul_id
    %eq_id =w ceql %i_phi_id, %j_phi_id
    %val_id =l zext %eq_id
    storel %val_id, %ptr_id
    %j_next_id =l add %j_phi_id, %one
    jmp @id_j_loop_head

@id_j_loop_end
    %i_next_id =l add %i_phi_id, %one
    jmp @id_i_loop_head

@id_i_loop_end
    %total_tmp2 =l mull %n_sq_id, 2
    %idx_z2 =l 0
    jmp @zero_loop_head2

@zero_loop_head2
    %idx_phi_z2 =l phi @id_i_loop_end %idx_z2, @zero_loop_body2 %idx_next_z2
    %cond_z2 =w cultl %idx_phi_z2, %total_tmp2
    jnz %cond_z2, @zero_loop_body2, @return_zero2

@zero_loop_body2
    %idx_mul_z2 =l mull %idx_phi_z2, %eight
    %ptr_z2 =l add %tmp, %idx_mul_z2
    storel %zero, %ptr_z2
    %idx_next_z2 =l add %idx_phi_z2, %one
    jmp @zero_loop_head2

@return_zero2
    ret %zero
}
