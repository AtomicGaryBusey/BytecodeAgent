export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %k_zero = ceq %k, 0
    jnz %k_zero, @k_zero, @non_zero

@k_zero
    jmp @do_identity_and_zero

@non_zero
    %eight = 8
    %n_sq = mull %n, %n
    %n_sq_bytes = mull %n_sq, %eight
    %tmp_base = %tmp
    %tmp_result = addl %tmp, %n_sq_bytes
    jmp @copy_a_loop_init

@copy_a_loop_init
    %idx_copy = 0
    jmp @copy_a_loop_head

@copy_a_loop_head
    %idx_phi_ca = phi @copy_a_loop_init %idx_copy, @copy_a_loop_body %idx_next_ca
    %cond_ca = cult %idx_phi_ca, %n_sq
    jnz %cond_ca, @copy_a_loop_body, @copy_a_loop_end

@copy_a_loop_body
    %idx_mul_ca = mull %idx_phi_ca, %eight
    %src_ca = addl %a, %idx_mul_ca
    %val_ca = loadl %src_ca
    %dst_ca = addl %tmp_base, %idx_mul_ca
    storel %val_ca, %dst_ca
    %idx_next_ca = addl %idx_phi_ca, 1
    jmp @copy_a_loop_head

@copy_a_loop_end
    jmp @init_i_loop_init

@init_i_loop_init
    %i_init = 0
    jmp @init_i_loop_head

@init_i_loop_head
    %i_phi = phi @init_i_loop_init %i_init, @init_i_loop_end %i_next
    %cond_i = cult %i_phi, %n
    jnz %cond_i, @init_j_loop_init, @init_i_loop_end

@init_j_loop_init
    %j_init = 0
    jmp @init_j_loop_head

@init_j_loop_head
    %j_phi = phi @init_j_loop_init %j_init, @init_j_loop_end %j_next
    %cond_j = cult %j_phi, %n
    jnz %cond_j, @init_j_loop_body, @init_j_loop_end

@init_j_loop_body
    %tmp_mul_i = mull %i_phi, %n
    %index = addl %tmp_mul_i, %j_phi
    %idx_mul = mull %index, %eight
    %dst_ij = addl %tmp_result, %idx_mul
    %eq = ceq %i_phi, %j_phi
    %val_ij = zextl %eq
    storel %val_ij, %dst_ij
    %j_next = addl %j_phi, 1
    jmp @init_j_loop_head

@init_j_loop_end
    %i_next = addl %i_phi, 1
    jmp @init_i_loop_head

@init_i_loop_end
    %k_current = %k
    jmp @exp_loop_head

@exp_loop_head
    %k_phi = phi @init_i_loop_end %k_current, @exp_loop_update_k %k_next
    %cond_k = cugt %k_phi, 0
    jnz %cond_k, @exp_loop_body, @exp_loop_end

@exp_loop_body
    %k_lsb = andl %k_phi, 1
    %lsb_nz = cugt %k_lsb, 0
    jnz %lsb_nz, @multiply_result_base, @skip_multiply_result

@multiply_result_base
    %src1_1 = %tmp_result
    %src2_1 = %tmp_base
    %dst_1 = %out
    jmp @mul1_i_init

@mul1_i_init
    %i_1 = 0
    jmp @mul1_i_head

@mul1_i_head
    %i_phi1 = phi @mul1_i_init %i_1, @mul1_i_end %i_next1
    %cond_i1 = cult %i_phi1, %n
    jnz %cond_i1, @mul1_j_init, @mul1_done

@mul1_j_init
    %j_1 = 0
    jmp @mul1_j_head

@mul1_j_head
    %j_phi1 = phi @mul1_j_init %j_1, @mul1_j_end %j_next1
    %cond_j1 = cult %j_phi1, %n
    jnz %cond_j1, @mul1_t_init, @mul1_j_end

@mul1_t_init
    %sum_1 = 0
    %t_1 = 0
    jmp @mul1_t_head

@mul1_t_head
    %t_phi1 = phi @mul1_t_init %t_1, @mul1_t_body %t_next1
    %sum_phi1 = phi @mul1_t_init %sum_1, @mul1_t_body %sum_next1
    %cond_t1 = cult %t_phi1, %n
    jnz %cond_t1, @mul1_t_body, @mul1_t_end

@mul1_t_body
    %tmp_mul_i1 = mull %i_phi1, %n
    %idx1_1 = addl %tmp_mul_i1, %t_phi1
    %idx1_mul = mull %idx1_1, %eight
    %ptr1_1 = addl %src1_1, %idx1_mul
    %val1_1 = loadl %ptr1_1
    %tmp_mul_t1 = mull %t_phi1, %n
    %idx2_1 = addl %tmp_mul_t1, %j_phi1
    %idx2_mul = mull %idx2_1, %eight
    %ptr2_1 = addl %src2_1, %idx2_mul
    %val2_1 = loadl %ptr2_1
    %prod_1 = mull %val1_1, %val2_1
    %sum_next1 = addl %sum_phi1, %prod_1
    %t_next1 = addl %t_phi1, 1
    jmp @mul1_t_head

@mul1_t_end
    %tmp_mul_i1_end = mull %i_phi1, %n
    %idx_dst1 = addl %tmp_mul_i1_end, %j_phi1
    %idx_dst_mul = mull %idx_dst1, %eight
    %ptr_dst1 = addl %dst_1, %idx_dst_mul
    storel %sum_phi1, %ptr_dst1
    jmp @mul1_j_end

@mul1_j_end
    %j_next1 = addl %j_phi1, 1
    jmp @mul1_j_head

@mul1_i_end
    %i_next1 = addl %i_phi1, 1
    jmp @mul1_i_head

@mul1_done
    jmp @copy_result_to_tmp_result

@copy_result_to_tmp_result
    %idx_cr = 0
    jmp @copy_result_loop_head

@copy_result_loop_head
    %idx_phi_cr = phi @copy_result_to_tmp_result %idx_cr, @copy_result_loop_body %idx_next_cr
    %cond_cr = cult %idx_phi_cr, %n_sq
    jnz %cond_cr, @copy_result_loop_body, @skip_multiply_result

@copy_result_loop_body
    %idx_mul_cr = mull %idx_phi_cr, %eight
    %src_cr = addl %out, %idx_mul_cr
    %val_cr = loadl %src_cr
    %dst_cr = addl %tmp_result, %idx_mul_cr
    storel %val_cr, %dst_cr
    %idx_next_cr = addl %idx_phi_cr, 1
    jmp @copy_result_loop_head

@skip_multiply_result
    jmp @multiply_base_base

@multiply_base_base
    %src1_2 = %tmp_base
    %src2_2 = %tmp_base
    %dst_2 = %out
    jmp @mul2_i_init

@mul2_i_init
    %i_2 = 0
    jmp @mul2_i_head

@mul2_i_head
    %i_phi2 = phi @mul2_i_init %i_2, @mul2_i_end %i_next2
    %cond_i2 = cult %i_phi2, %n
    jnz %cond_i2, @mul2_j_init, @mul2_done

@mul2_j_init
    %j_2 = 0
    jmp @mul2_j_head

@mul2_j_head
    %j_phi2 = phi @mul2_j_init %j_2, @mul2_j_end %j_next2
    %cond_j2 = cult %j_phi2, %n
    jnz %cond_j2, @mul2_t_init, @mul2_j_end

@mul2_t_init
    %sum_2 = 0
    %t_2 = 0
    jmp @mul2_t_head

@mul2_t_head
    %t_phi2 = phi @mul2_t_init %t_2, @mul2_t_body %t_next2
    %sum_phi2 = phi @mul2_t_init %sum_2, @mul2_t_body %sum_next2
    %cond_t2 = cult %t_phi2, %n
    jnz %cond_t2, @mul2_t_body, @mul2_t_end

@mul2_t_body
    %tmp_mul_i2 = mull %i_phi2, %n
    %idx1_2 = addl %tmp_mul_i2, %t_phi2
    %idx1_mul2 = mull %idx1_2, %eight
    %ptr1_2 = addl %src1_2, %idx1_mul2
    %val1_2 = loadl %ptr1_2
    %tmp_mul_t2 = mull %t_phi2, %n
    %idx2_2 = addl %tmp_mul_t2, %j_phi2
    %idx2_mul2 = mull %idx2_2, %eight
    %ptr2_2 = addl %src2_2, %idx2_mul2
    %val2_2 = loadl %ptr2_2
    %prod_2 = mull %val1_2, %val2_2
    %sum_next2 = addl %sum_phi2, %prod_2
    %t_next2 = addl %t_phi2, 1
    jmp @mul2_t_head

@mul2_t_end
    %tmp_mul_i2_end = mull %i_phi2, %n
    %idx_dst2 = addl %tmp_mul_i2_end, %j_phi2
    %idx_dst_mul2 = mull %idx_dst2, %eight
    %ptr_dst2 = addl %dst_2, %idx_dst_mul2
    storel %sum_phi2, %ptr_dst2
    jmp @mul2_j_end

@mul2_j_end
    %j_next2 = addl %j_phi2, 1
    jmp @mul2_j_head

@mul2_i_end
    %i_next2 = addl %i_phi2, 1
    jmp @mul2_i_head

@mul2_done
    jmp @copy_base_to_tmp_base

@copy_base_to_tmp_base
    %idx_cb = 0
    jmp @copy_base_loop_head

@copy_base_loop_head
    %idx_phi_cb = phi @copy_base_to_tmp_base %idx_cb, @copy_base_loop_body %idx_next_cb
    %cond_cb = cult %idx_phi_cb, %n_sq
    jnz %cond_cb, @copy_base_loop_body, @exp_loop_update_k

@copy_base_loop_body
    %idx_mul_cb = mull %idx_phi_cb, %eight
    %src_cb = addl %out, %idx_mul_cb
    %val_cb = loadl %src_cb
    %dst_cb = addl %tmp_base, %idx_mul_cb
    storel %val_cb, %dst_cb
    %idx_next_cb = addl %idx_phi_cb, 1
    jmp @copy_base_loop_head

@exp_loop_update_k
    %k_next = shrl %k_phi, 1
    jmp @exp_loop_head

@exp_loop_end
    %idx_cf = 0
    jmp @copy_final_loop_head

@copy_final_loop_head
    %idx_phi_cf = phi @exp_loop_end %idx_cf, @copy_final_loop_body %idx_next_cf
    %cond_cf = cult %idx_phi_cf, %n_sq
    jnz %cond_cf, @copy_final_loop_body, @zero_tmp

@copy_final_loop_body
    %idx_mul_cf = mull %idx_phi_cf, %eight
    %src_cf = addl %tmp_result, %idx_mul_cf
    %val_cf = loadl %src_cf
    %dst_cf = addl %out, %idx_mul_cf
    storel %val_cf, %dst_cf
    %idx_next_cf = addl %idx_phi_cf, 1
    jmp @copy_final_loop_head

@zero_tmp
    %total_tmp = mull %n_sq, 2
    %idx_z = 0
    jmp @zero_loop_head

@zero_loop_head
    %idx_phi_z = phi @zero_tmp %idx_z, @zero_loop_body %idx_next_z
    %cond_z = cult %idx_phi_z, %total_tmp
    jnz %cond_z, @zero_loop_body, @return_zero

@zero_loop_body
    %idx_mul_z = mull %idx_phi_z, %eight
    %ptr_z = addl %tmp, %idx_mul_z
    storel 0, %ptr_z
    %idx_next_z = addl %idx_phi_z, 1
    jmp @zero_loop_head

@return_zero
    ret 0

@do_identity_and_zero
    %n_sq_id = mull %n, %n
    %i_id = 0
    jmp @id_i_loop_head

@id_i_loop_head
    %i_phi_id = phi @do_identity_and_zero %i_id, @id_i_loop_end %i_next_id
    %cond_i_id = cult %i_phi_id, %n
    jnz %cond_i_id, @id_j_loop_init, @id_i_loop_end

@id_j_loop_init
    %j_id = 0
    jmp @id_j_loop_head

@id_j_loop_head
    %j_phi_id = phi @id_j_loop_init %j_id, @id_j_loop_end %j_next_id
    %cond_j_id = cult %j_phi_id, %n
    jnz %cond_j_id, @id_j_loop_body, @id_j_loop_end

@id_j_loop_body
    %tmp_mul_i_id = mull %i_phi_id, %n
    %index_id = addl %tmp_mul_i_id, %j_phi_id
    %idx_mul_id = mull %index_id, %eight
    %ptr_id = addl %out, %idx_mul_id
    %eq_id = ceq %i_phi_id, %j_phi_id
    %val_id = zextl %eq_id
    storel %val_id, %ptr_id
    %j_next_id = addl %j_phi_id, 1
    jmp @id_j_loop_head

@id_j_loop_end
    %i_next_id = addl %i_phi_id, 1
    jmp @id_i_loop_head

@id_i_loop_end
    %total_tmp2 = mull %n_sq_id, 2
    %idx_z2 = 0
    jmp @zero_loop_head2

@zero_loop_head2
    %idx_phi_z2 = phi @id_i_loop_end %idx_z2, @zero_loop_body2 %idx_next_z2
    %cond_z2 = cult %idx_phi_z2, %total_tmp2
    jnz %cond_z2, @zero_loop_body2, @return_zero2

@zero_loop_body2
    %idx_mul_z2 = mull %idx_phi_z2, %eight
    %ptr_z2 = addl %tmp, %idx_mul_z2
    storel 0, %ptr_z2
    %idx_next_z2 = addl %idx_phi_z2, 1
    jmp @zero_loop_head2

@return_zero2
    ret 0
}
