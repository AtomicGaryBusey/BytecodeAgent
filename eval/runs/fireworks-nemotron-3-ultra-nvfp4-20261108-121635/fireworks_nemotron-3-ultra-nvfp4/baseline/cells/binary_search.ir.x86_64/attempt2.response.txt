export function l $binary_search(l %a, l %n, l %key) {
@start
    %zero = sub %n, %n
    %low0 = copy %zero
    %high0 = copy %n
    %neg1 = sub %zero, 1
    jmp @loop

@loop
    %low = phi @start %low0 @less %low1 @greater %low2
    %high = phi @start %high0 @less %high1 @greater %high2
    %cond = csltl %low, %high
    jnz %cond @loop_body @notfound

@loop_body
    %sum = add %low, %high
    %mid = shr %sum, 1
    %mid8 = mul %mid, 8
    %addr = add %a, %mid8
    %val = loadl %addr, 0
    %eq = ceql %val, %key
    jnz %eq @found @not_equal

@not_equal
    %lt = csltl %val, %key
    jnz %lt @less @greater

@less
    %low1 = add %mid, 1
    %high1 = copy %high
    jmp @loop

@greater
    %low2 = copy %low
    %high2 = copy %mid
    jmp @loop

@found
    ret %mid

@notfound
    ret %neg1
}
