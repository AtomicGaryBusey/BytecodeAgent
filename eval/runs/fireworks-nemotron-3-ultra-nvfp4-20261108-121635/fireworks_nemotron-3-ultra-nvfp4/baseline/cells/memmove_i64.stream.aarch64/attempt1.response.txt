memmove_i64:
	cbz x2, .Lreturn
	cmp x0, x1
	b.lt .Lforward
	b.gt .Lbackward
	b .Lreturn
.Lforward:
	ldr x3, [x1], #8
	str x3, [x0], #8
	subs x2, x2, #1
	b.ne .Lforward
	b .Lreturn
.Lbackward:
	add x3, x1, x2, lsl #3
	add x4, x0, x2, lsl #3
.Lbackward_loop:
	ldr x5, [x3, #-8]!
	str x5, [x4, #-8]!
	subs x2, x2, #1
	b.ne .Lbackward_loop
.Lreturn:
	mov x0, #0
	ret
