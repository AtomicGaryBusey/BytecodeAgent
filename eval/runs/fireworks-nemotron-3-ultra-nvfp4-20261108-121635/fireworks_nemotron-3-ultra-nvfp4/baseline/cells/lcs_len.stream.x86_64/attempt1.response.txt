lcs_len:
    push r12
    push r13
    mov r9, rcx
    inc r9
    xor r10, r10
.Lzero:
    mov qword ptr [r8 + r10*8], 0
    inc r10
    cmp r10, r9
    jne .Lzero
    xor r9, r9
.Louter:
    cmp r9, rsi
    jge .Lend_outer
    movzx eax, byte ptr [rdi + r9]
    xor r10, r10
    mov r11, 1
.Linner:
    cmp r11, rcx
    jg .Lend_inner
    mov r12, qword ptr [r8 + r11*8]
    cmp al, byte ptr [rdx + r11 - 1]
    je .Lequal
    mov r13, qword ptr [r8 + r11*8 - 8]
    cmp r12, r13
    cmovl r13, r12
    mov qword ptr [r8 + r11*8], r13
    jmp .Lnext
.Lequal:
    lea r13, [r10 + 1]
    mov qword ptr [r8 + r11*8], r13
.Lnext:
    mov r10, r12
    inc r11
    jmp .Linner
.Lend_inner:
    inc r9
    jmp .Louter
.Lend_outer:
    mov rax, qword ptr [r8 + rcx*8]
    pop r13
    pop r12
    ret
