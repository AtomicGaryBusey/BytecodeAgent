lcs_len:
    push r12
    push r13
    mov r9, rcx          # r9 = n
    inc r9               # r9 = n+1
    xor r10, r10         # r10 = 0
.Lzero:
    mov qword ptr [r8 + r10*8], 0
    inc r10
    cmp r10, r9
    jne .Lzero
    xor r9, r9           # r9 = i = 0
.Louter:
    cmp r9, rsi          # compare i with m
    jge .Lend_outer
    movzx eax, byte ptr [rdi + r9]  # al = x[i]
    xor r10, r10         # r10 = diag = 0
    mov r11, 1           # r11 = j = 1
.Linner:
    cmp r11, rcx         # compare j with n
    jg .Lend_inner
    mov r12, qword ptr [r8 + r11*8]      # r12 = up = row[j]
    cmp al, byte ptr [rdx + r11 - 1]     # compare x[i] with y[j-1]
    je .Lequal
    mov r13, qword ptr [r8 + r11*8 - 8]  # r13 = row[j-1]
    cmp r12, r13
    cmovg r13, r12       # r13 = max(r12, r13)
    mov qword ptr [r8 + r11*8], r13
    jmp .Lnext
.Lequal:
    lea r13, [r10 + 1]   # r13 = diag + 1
    mov qword ptr [r8 + r11*8], r13
.Lnext:
    mov r10, r12         # diag = up
    inc r11
    jmp .Linner
.Lend_inner:
    inc r9
    jmp .Louter
.Lend_outer:
    mov rax, qword ptr [r8 + rcx*8]  # return row[n]
    pop r13
    pop r12
    ret
