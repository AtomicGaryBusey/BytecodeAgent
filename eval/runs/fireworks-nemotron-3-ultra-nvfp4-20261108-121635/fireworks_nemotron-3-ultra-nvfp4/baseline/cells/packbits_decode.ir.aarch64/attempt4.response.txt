export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%src_ptr0 = copyl %src
	%n_left0 = copyl %n
	%out_ptr0 = copyl %out
	%cap_left0 = copyl %cap
	%total_written0 =l copy 0
	jmp @loop

@loop
	%src_ptr = phi @start %src_ptr0 @noop %src_ptr_next @copy_literal_loop_done %src_ptr_next @fill_repeat_loop_done %src_ptr_next
	%n_left = phi @start %n_left0 @noop %n_left_next @copy_literal_loop_done %n_left_next @fill_repeat_loop_done %n_left_next
	%out_ptr = phi @start %out_ptr0 @noop %out_ptr_next @copy_literal_loop_done %out_ptr_next @fill_repeat_loop_done %out_ptr_next
	%cap_left = phi @start %cap_left0 @noop %cap_left_next @copy_literal_loop_done %cap_left_next @fill_repeat_loop_done %cap_left_next
	%total_written = phi @start %total_written0 @noop %total_written_next @copy_literal_loop_done %total_written_next @fill_repeat_loop_done %total_written_next

	%n_left_zero = ceql %n_left, 0
	jnz %n_left_zero, @done

	%h =w loadsb %src_ptr
	%src_ptr1 = addl %src_ptr, 1
	%n_left1 = subl %n_left, 1

	%h_neg128 = ceqw %h, -128
	jnz %h_neg128, @noop

	%h_ge_zero = csgew %h, 0
	jnz %h_ge_zero, @literal
	jmp @repeat

@noop
	%src_ptr_next = copyl %src_ptr1
	%n_left_next = copyl %n_left1
	%out_ptr_next = copyl %out_ptr
	%cap_left_next = copyl %cap_left
	%total_written_next = copyl %total_written
	jmp @loop

@literal
	%run_len_w = addw %h, 1
	%run_len = zextl %run_len_w

	%n_left_ge = csge %n_left1, %run_len
	jnz %n_left_ge, @literal_check_cap
	jmp @src_error

@literal_check_cap
	%cap_left_ge = csge %cap_left, %run_len
	jnz %cap_left_ge, @literal_copy_full
	jmp @literal_copy_partial

@literal_copy_full
	%copy_count = copyl %run_len
	%is_partial =w copy 0
	jmp @copy_literal_setup

@literal_copy_partial
	%copy_count = copyl %cap_left
	%is_partial =w copy 1
	jmp @copy_literal_setup

@copy_literal_setup
	%src_ptr_in = copyl %src_ptr1
	%out_ptr_in = copyl %out_ptr
	%n_left_in = copyl %n_left1
	%cap_left_in = copyl %cap_left
	%total_written_in = copyl %total_written
	%copy_count_in = copyl %copy_count
	%is_partial_in = copyw %is_partial
	jmp @copy_literal_loop_head

@copy_literal_loop_head
	%src_ptr_l = phi @copy_literal_setup %src_ptr_in @copy_literal_loop_body %src_ptr_next
	%out_ptr_l = phi @copy_literal_setup %out_ptr_in @copy_literal_loop_body %out_ptr_next
	%i = phi @copy_literal_setup 0 @copy_literal_loop_body %i_next
	%cond = csltl %i, %copy_count_in
	jnz %cond, @copy_literal_loop_body
	jmp @copy_literal_loop_done

@copy_literal_loop_body
	%byte =w loadub %src_ptr_l
	storeb %byte, %out_ptr_l
	%src_ptr_next = addl %src_ptr_l, 1
	%out_ptr_next = addl %out_ptr_l, 1
	%i_next = addl %i, 1
	jmp @copy_literal_loop_head

@copy_literal_loop_done
	%n_left_next = subl %n_left_in, %copy_count_in
	%cap_left_next = subl %cap_left_in, %copy_count_in
	%total_written_next = addl %total_written_in, %copy_count_in
	%src_ptr_next = copyl %src_ptr_l
	%out_ptr_next = copyl %out_ptr_l
	%partial = copyw %is_partial_in
	jnz %partial, @return_neg1
	jmp @loop

@repeat
	%repeat_count_w = subw 1, %h
	%repeat_count = zextl %repeat_count_w

	%n_left_ge_1 = csge %n_left1, 1
	jnz %n_left_ge_1, @repeat_load_val
	jmp @src_error

@repeat_load_val
	%val =w loadub %src_ptr1
	%src_ptr2 = addl %src_ptr1, 1
	%n_left2 = subl %n_left1, 1

	%cap_left_ge = csge %cap_left, %repeat_count
	jnz %cap_left_ge, @repeat_fill_full
	jmp @repeat_fill_partial

@repeat_fill_full
	%fill_count = copyl %repeat_count
	%is_partial =w copy 0
	jmp @fill_repeat_setup

@repeat_fill_partial
	%fill_count = copyl %cap_left
	%is_partial =w copy 1
	jmp @fill_repeat_setup

@fill_repeat_setup
	%out_ptr_in = copyl %out_ptr
	%cap_left_in = copyl %cap_left
	%total_written_in = copyl %total_written
	%src_ptr_in = copyl %src_ptr2
	%n_left_in = copyl %n_left2
	%fill_count_in = copyl %fill_count
	%is_partial_in = copyw %is_partial
	jmp @fill_repeat_loop_head

@fill_repeat_loop_head
	%out_ptr_l = phi @fill_repeat_setup %out_ptr_in @fill_repeat_loop_body %out_ptr_next
	%i = phi @fill_repeat_setup 0 @fill_repeat_loop_body %i_next
	%cond = csltl %i, %fill_count_in
	jnz %cond, @fill_repeat_loop_body
	jmp @fill_repeat_loop_done

@fill_repeat_loop_body
	storeb %val, %out_ptr_l
	%out_ptr_next = addl %out_ptr_l, 1
	%i_next = addl %i, 1
	jmp @fill_repeat_loop_head

@fill_repeat_loop_done
	%cap_left_next = subl %cap_left_in, %fill_count_in
	%total_written_next = addl %total_written_in, %fill_count_in
	%out_ptr_next = copyl %out_ptr_l
	%src_ptr_next = copyl %src_ptr_in
	%n_left_next = copyl %n_left_in
	%partial = copyw %is_partial_in
	jnz %partial, @return_neg1
	jmp @loop

@src_error
	ret -2

@return_neg1
	ret -1

@done
	ret %total_written
}
