export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop

@loop
    %cur_a = phi @start %a, @body %next_a
    %cur_n = phi @start %n, @body %next_n
    %cnt = phi @start l 0, @body %next_cnt
    %cond = cugt %cur_n, l 0
    jnz %cond, @body, @done

@body
    %elem = loadl %cur_a
    %gt = cugt %elem, %t
    %add = zext %gt
    %next_cnt = add %cnt, %add
    %next_a = add %cur_a, l 8
    %next_n = sub %cur_n, l 1
    jmp @loop

@done
    ret %cnt
}
