export function l $count_ugt(l %a, l %n, l %t) {
@start
    %cur = copy %a
    %cnt = l 0
    %nrem = copy %n
    jmp @loop

@loop
    %cond = csgte %nrem, l 1
    jnz %cond, @body, @done

@body
    %elem = loadl %cur
    %gt = cugtl %elem, %t
    %add = zextw %gt
    %cnt = add %cnt, %add
    %cur = add %cur, l 8
    %nrem = sub %nrem, l 1
    jmp @loop

@done
    ret %cnt
}
