map_sum:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, 0
    mov r14, rdi
    mov r13, rsi
    mov r12, rdx
    test r13, r13
    jz .Lend
.Lloop:
    mov rdi, [r14]
    call r12
    add rbx, rax
    add r14, 8
    dec r13
    jnz .Lloop
.Lend:
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
