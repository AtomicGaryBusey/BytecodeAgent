export function $popcount(l %x) {
@start
    %t1 = shr %x, 1
    %t2 = and %t1, 0x5555555555555555
    %x = sub %x, %t2
    %t3 = and %x, 0x3333333333333333
    %t4 = shr %x, 2
    %t5 = and %t4, 0x3333333333333333
    %x = add %t3, %t5
    %t6 = shr %x, 4
    %x = add %x, %t6
    %x = and %x, 0x0f0f0f0f0f0f0f0f
    %x = mull %x, 0x0101010101010101
    %x = shr %x, 56
    ret %x
}
