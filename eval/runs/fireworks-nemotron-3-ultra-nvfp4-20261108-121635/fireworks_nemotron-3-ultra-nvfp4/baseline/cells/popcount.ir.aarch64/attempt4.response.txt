@start
    %t1 = shrl %x, 1
    %t2 = andl %t1, 0x5555555555555555
    %x = subl %x, %t2
    %t3 = andl %x, 0x3333333333333333
    %t4 = shrl %x, 2
    %t5 = andl %t4, 0x3333333333333333
    %x = addl %t3, %t5
    %t6 = shrl %x, 4
    %x = addl %x, %t6
    %x = andl %x, 0x0f0f0f0f0f0f0f0f

    %x = shrl %x, 56
    ret %x
}
