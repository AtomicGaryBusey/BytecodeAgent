dot_i32_simd:
    xor rax, rax
    mov r8, rdx
    mov r9, rdi
    mov r10, rsi
    mov rcx, r8
    shr rcx, 2
    test rcx, rcx
    jz .Lscalar_start

    xorps xmm0, xmm0

.Lvector_loop:
    movdqu xmm1, [r9]
    movdqu xmm2, [r10]
    add r9, 16
    add r10, 16

    movdqa xmm3, xmm1
    pmuldq xmm3, xmm2

    pshufd xmm4, xmm1, 0xB1
    pshufd xmm5, xmm2, 0xB1
    pmuldq xmm4, xmm5

    paddq xmm0, xmm3
    paddq xmm0, xmm4

    dec rcx
    jnz .Lvector_loop

    movq rdx, xmm0
    pextrq r11, xmm0, 1
    add rax, rdx
    add rax, r11

.Lscalar_start:
    mov rcx, r8
    and rcx, 3
    test rcx, rcx
    jz .Ldone

.Lscalar_loop:
    movsxd rdx, dword ptr [r9]
    movsxd r11, dword ptr [r10]
    imul rdx, r11
    add rax, rdx
    add r9, 4
    add r10, 4
    dec rcx
    jnz .Lscalar_loop

.Ldone:
    ret
