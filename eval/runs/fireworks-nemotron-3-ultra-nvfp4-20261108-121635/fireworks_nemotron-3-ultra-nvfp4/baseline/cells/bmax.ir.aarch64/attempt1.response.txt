export function l $bmax(l %a, l %n) {
@start
	%max0 = loadl %a
	%i0 = copy 1
	jmp @loop(%i0, %max0)

@loop(l %i, l %max)
	%cond = cltu %i %n
	jnz %cond, @body(%i, %max), @exit(%max)

@body(l %i, l %max)
	%addr = add %a, shl %i, 3
	%val = loadl %addr
	%cmp = csgtl %val %max
	%max_new = sel %cmp %val %max
	%i_new = add %i, 1
	jmp @loop(%i_new, %max_new)

@exit(l %max)
	ret %max
}
