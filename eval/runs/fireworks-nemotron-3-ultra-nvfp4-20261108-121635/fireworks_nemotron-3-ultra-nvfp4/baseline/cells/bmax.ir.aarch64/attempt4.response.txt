export function l @bmax(l %a, l %n) {
@start
	%max0 = loadl %a
	%i0 = addl 0, 1
	jmp @loop(%i0, %max0)

@loop(l %i, l %max)
	%cond = cltul %i %n
	jnz %cond, @body(%i, %max), @exit(%max)

@body(l %i, l %max)
	%sh = shll %i, 3
	%addr = addl %a, %sh
	%val = loadl %addr
	%cmp = cgtl %val %max
	%max_new = sel %cmp %val %max
	%i_new = addl %i, 1
	jmp @loop(%i_new, %max_new)

@exit(l %max)
	ret %max
}
