max_byte:
	cbz x1, .Lzero
	movi v0.16b, #0
.Lvec_loop:
	cmp x1, #16
	blt .Lscalar
	ld1 {v1.16b}, [x0]
	umax v0.16b, v0.16b, v1.16b
	add x0, x0, #16
	sub x1, x1, #16
	b .Lvec_loop
.Lscalar:
	cbz x1, .Lreduce
	ld1 {v1.16b}, [x0]
	umax v0.16b, v0.16b, v1.16b
.Lreduce:
	umaxv b0, v0.16b
	umov x0, v0.b[0]
