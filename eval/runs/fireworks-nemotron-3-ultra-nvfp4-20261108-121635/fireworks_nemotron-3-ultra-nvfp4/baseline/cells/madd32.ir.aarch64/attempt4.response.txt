export function l $madd32(l %a, l %b, l %c) {
@start
	%zero = sub %a, %a
	%one = add %zero, 1
	%mask64 = shl %one, 32
	%mask32 = sub %mask64, 1
	%prod = mul %a, %b
	%prod32 = and %prod, %mask32
	%c32 = and %c, %mask32
	%sum = add %prod32, %c32
	%sum32 = and %sum, %mask32
	%tmp = shl %sum32, 32
	%res = sar %tmp, 32
	ret %res
}
