export function l $madd32(l %a, l %b, l %c) {
@start
	%zero = sub %a, %a
	%one = add %zero, 1
	%mask = shl %one, 32
	%mask = sub %mask, 1
	%a32 = and %a, %mask
	%b32 = and %b, %mask
	%c32 = and %c, %mask
	%prod = mul %a32, %b32
	%prod32 = and %prod, %mask
	%sum = add %prod32, %c32
	%sum32 = and %sum, %mask
	%tmp = shl %sum32, 32
	%res = sar %tmp, 32
	ret %res
}
