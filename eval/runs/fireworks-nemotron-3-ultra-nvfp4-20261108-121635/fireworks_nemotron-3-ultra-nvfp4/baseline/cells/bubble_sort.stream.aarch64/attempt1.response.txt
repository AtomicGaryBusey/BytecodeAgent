bubble_sort:
	cmp x1, #1
	b.le .Lreturn_zero
	mov x9, x1          // n
	mov x10, #0         // i = 0
.Louter:
	sub x12, x9, x10    // n - i
	sub x12, x12, #1    // inner_limit = n - i - 1
	cmp x12, #0
	b.le .Lreturn_zero
	mov x15, #0         // swapped = 0
	mov x11, #0         // j = 0
.Linner:
	cmp x11, x12
	b.ge .Linner_done
	lsl x16, x11, #3    // j * 8
	add x16, x0, x16    // &a[j]
	ldr x13, [x16]      // a[j]
	ldr x14, [x16, #8]  // a[j+1]
	cmp x13, x14
	b.le .Lno_swap
	str x14, [x16]      // swap
	str x13, [x16, #8]
	mov x15, #1         // swapped = 1
.Lno_swap:
	add x11, x11, #1    // j++
	b .Linner
.Linner_done:
	cbz x15, .Lreturn_zero
	add x10, x10, #1    // i++
	b .Louter
.Lreturn_zero:
	mov x0, #0
	ret
