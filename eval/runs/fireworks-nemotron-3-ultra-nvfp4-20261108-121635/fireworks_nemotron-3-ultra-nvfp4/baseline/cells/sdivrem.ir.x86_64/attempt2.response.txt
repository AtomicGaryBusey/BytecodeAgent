export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %b_zero = ceql %b, 0
    jnz %b_zero, @b_zero_case

    %one = copy 1
    %min = shiftl %one, 63
    %a_min = ceql %a, %min
    %b_neg1 = ceql %b, -1
    %special = and %a_min, %b_neg1
    jnz %special, @special_case

    %q = div %a, %b
    %r = rem %a, %b
    jmp @store_and_return

@b_zero_case
    %q = copy 0
    %r = copy %a
    jmp @store_and_return

@special_case
    %q = copy %min
    %r = copy 0
    jmp @store_and_return

@store_and_return
    storel %r, %rem
    ret %q
}
