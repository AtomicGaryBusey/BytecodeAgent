export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %zero = copy 0
    %b_zero = ceql %b, %zero
    jnz %b_zero, @b_zero_case

    %one = copy 1
    %neg1 = copy -1
    %63 = copy 63
    %min = shll %one, %63
    %a_min = ceql %a, %min
    %b_neg1 = ceql %b, %neg1
    %special = andw %a_min, %b_neg1
    jnz %special, @special_case

    %q = divl %a, %b
    %r = reml %a, %b
    jmp @store_and_return

@b_zero_case
    %q = copy 0
    %r = copy %a
    jmp @store_and_return

@special_case
    %q = copy %min
    %r = copy 0
    jmp @store_and_return

@store_and_return
    storel %r, %rem
    ret %q
}
