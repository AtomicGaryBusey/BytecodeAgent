export function l $map_sum(l %a, l %n, l %f) {
    @start
    %sum =l add 0, 0
    %p =l copy %a
    %cnt =l copy %n
    %cond =w csgt %cnt, 0
    jnz %cond, @loop, @end

    @loop
    %val =l loadl %p
    %res =l call %f(l %val)
    %sum =l add %sum, %res
    %p =l add %p, 8
    %cnt =l sub %cnt, 1
    %cond =w csgt %cnt, 0
    jnz %cond, @loop, @end

    @end
    ret %sum
}
