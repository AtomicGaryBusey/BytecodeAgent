export function l $map_sum(l %a, l %n, l %f) {
    @start
    %sum =l addl 0, 0
    %p =l copy %a
    %cnt =l copy %n
    %cond =w ceql %cnt, 0
    jnz %cond, @end, @loop

    @loop
    %val =l loadl %p
    %res =l call %f(l %val)
    %sum =l addl %sum, %res
    %p =l addl %p, 8
    %cnt =l subl %cnt, 1
    %cond =w csgt %cnt, 0
    jnz %cond, @loop, @end

    @end
    ret %sum
}
