sdivrem:
    mov r8, rdx
    mov rax, rdi
    mov rcx, rsi
    test rcx, rcx
    jz .div_zero
    mov r9, 0x8000000000000000
    cmp rax, r9
    jne .do_div
    cmp rcx, -1
    jne .do_div
    mov qword ptr [r8], 0
    mov rax, r9
    ret
.do_div:
    cqo
    idiv rcx
    mov qword ptr [r8], rdx
    ret
.div_zero:
    mov qword ptr [r8], rax
    xor rax, rax
    ret
