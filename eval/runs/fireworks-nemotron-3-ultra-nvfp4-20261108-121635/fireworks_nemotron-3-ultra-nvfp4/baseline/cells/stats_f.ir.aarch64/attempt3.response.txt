export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %a0 =l loadl %a
    %y0 =l call %f(l %a0)
    %min =l copy %y0
    %max =l copy %y0
    %sum =l copy %y0
    %t0 =l and %y0, 1
    %c0 =w cne %t0, 0
    %odd =l sel %c0, 1, 0
    %i =l copy 1
    jmp @loop_head

@loop_head
    %cond =w cslt %i, %n
    jnz %cond, @loop_body, @loop_end

@loop_body
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %ai =l loadl %addr
    %yi =l call %f(l %ai)
    %cmp_min =w cslt %yi, %min
    %min =l sel %cmp_min, %yi, %min
    %cmp_max =w csgt %yi, %max
    %max =l sel %cmp_max, %yi, %max
    %sum =l add %sum, %yi
    %t1 =l and %yi, 1
    %c1 =w cne %t1, 0
    %add_odd =l sel %c1, 1, 0
    %odd =l add %odd, %add_odd
    %i =l add %i, 1
    jmp @loop_head

@loop_end
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
