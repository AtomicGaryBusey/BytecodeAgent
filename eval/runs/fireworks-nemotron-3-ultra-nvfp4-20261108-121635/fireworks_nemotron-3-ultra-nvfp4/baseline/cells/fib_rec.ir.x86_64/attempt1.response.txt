export function l $fib_rec(l %n) {
@entry
    %cmp0 =w ceql %n, 0
    jnz %cmp0, @base0, @check1
@base0
    ret 0
@check1
    %cmp1 =w ceql %n, 1
    jnz %cmp1, @base1, @recurse
@base1
    ret 1
@recurse
    %n1 =l sub %n, 1
    %n2 =l sub %n, 2
    %r1 =l call $fib_rec(l %n1)
    %r2 =l call $fib_rec(l %n2)
    %sum =l add %r1, %r2
    ret %sum
}
