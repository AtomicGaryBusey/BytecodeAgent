export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %zero = const 0
    %one = const 1
    %eight = const 8
    %size = mul %n, %n
    %temp_base = add %tmp, mul %size, %eight

    %k_zero = ceql %k, %zero
    jnz %k_zero, @k_zero

    %idx = %zero
    jmp @copy_loop_cond

@copy_loop_cond
    %cond = csltl %idx, %size
    jnz %cond, @copy_loop_body
    jmp @init_ident

@copy_loop_body
    %off = mul %idx, %eight
    %a_ptr = add %a, %off
    %val = loadl %a_ptr
    %base_ptr = add %tmp, %off
    storel %val, %base_ptr
    %idx = add %idx, %one
    jmp @copy_loop_cond

@init_ident
    %i = %zero
    jmp @ident_i_cond

@ident_i_cond
    %cond = csltl %i, %n
    jnz %cond, @ident_i_body
    jmp @exp_loop_start

@ident_i_body
    %j = %zero
    jmp @ident_j_cond

@ident_j_cond
    %cond = csltl %j, %n
    jnz %cond, @ident_j_body
    %i = add %i, %one
    jmp @ident_i_cond

@ident_j_body
    %i_n = mul %i, %n
    %index = add %i_n, %j
    %off = mul %index, %eight
    %out_ptr = add %out, %off
    %eq = ceql %i, %j
    %val = sel %eq, %one, %zero
    storel %val, %out_ptr
    %j = add %j, %one
    jmp @ident_j_cond

@exp_loop_start
    %exp = %k
    jmp @exp_loop_cond

@exp_loop_cond
    %cond = csgtl %exp, %zero
    jnz %cond, @exp_loop_body
    jmp @zero_tmp

@exp_loop_body
    %bit = and %exp, %one
    %bit_nz = cne %bit, %zero
    jnz %bit_nz, @mult_res_base
    jmp @square_base

@mult_res_base
    %i = %zero
    jmp @mult_res_i_cond

@mult_res_i_cond
    %cond = csltl %i, %n
    jnz %cond, @mult_res_i_body
    jmp @copy_temp_to_out

@mult_res_i_body
    %j = %zero
    jmp @mult_res_j_cond

@mult_res_j_cond
    %cond = csltl %j, %n
    jnz %cond, @mult_res_j_body
    %i = add %i, %one
    jmp @mult_res_i_cond

@mult_res_j_body
    %sum = %zero
    %t = %zero
    jmp @mult_res_t_cond

@mult_res_t_cond
    %cond = csltl %t, %n
    jnz %cond, @mult_res_t_body
    %i_n = mul %i, %n
    %index = add %i_n, %j
    %off = mul %index, %eight
    %dest_ptr = add %temp_base, %off
    storel %sum, %dest_ptr
    %j = add %j, %one
    jmp @mult_res_j_cond

@mult_res_t_body
    %left_idx = add mul %i, %n, %t
    %left_off = mul %left_idx, %eight
    %left_ptr = add %out, %left_off
    %left_val = loadl %left_ptr
    %right_idx = add mul %t, %n, %j
    %right_off = mul %right_idx, %eight
    %right_ptr = add %tmp, %right_off
    %right_val = loadl %right_ptr
    %prod = mul %left_val, %right_val
    %sum = add %sum, %prod
    %t = add %t, %one
    jmp @mult_res_t_cond

@copy_temp_to_out
    %idx = %zero
    jmp @copy_to_out_cond

@copy_to_out_cond
    %cond = csltl %idx, %size
    jnz %cond, @copy_to_out_body
    jmp @square_base

@copy_to_out_body
    %off = mul %idx, %eight
    %src_ptr = add %temp_base, %off
    %val = loadl %src_ptr
    %dst_ptr = add %out, %off
    storel %val, %dst_ptr
    %idx = add %idx, %one
    jmp @copy_to_out_cond

@square_base
    %i = %zero
    jmp @mult_sq_i_cond

@mult_sq_i_cond
    %cond = csltl %i, %n
    jnz %cond, @mult_sq_i_body
    jmp @copy_temp_to_base

@mult_sq_i_body
    %j = %zero
    jmp @mult_sq_j_cond

@mult_sq_j_cond
    %cond = csltl %j, %n
    jnz %cond, @mult_sq_j_body
    %i = add %i, %one
    jmp @mult_sq_i_cond

@mult_sq_j_body
    %sum = %zero
    %t = %zero
    jmp @mult_sq_t_cond

@mult_sq_t_cond
    %cond = csltl %t, %n
    jnz %cond, @mult_sq_t_body
    %i_n = mul %i, %n
    %index = add %i_n, %j
    %off = mul %index, %eight
    %dest_ptr = add %temp_base, %off
    storel %sum, %dest_ptr
    %j = add %j, %one
    jmp @mult_sq_j_cond

@mult_sq_t_body
    %left_idx = add mul %i, %n, %t
    %left_off = mul %left_idx, %eight
    %left_ptr = add %tmp, %left_off
    %left_val = loadl %left_ptr
    %right_idx = add mul %t, %n, %j
    %right_off = mul %right_idx, %eight
    %right_ptr = add %tmp, %right_off
    %right_val = loadl %right_ptr
    %prod = mul %left_val, %right_val
    %sum = add %sum, %prod
    %t = add %t, %one
    jmp @mult_sq_t_cond

@copy_temp_to_base
    %idx = %zero
    jmp @copy_to_base_cond

@copy_to_base_cond
    %cond = csltl %idx, %size
    jnz %cond, @copy_to_base_body
    jmp @shift_exp

@copy_to_base_body
    %off = mul %idx, %eight
    %src_ptr = add %temp_base, %off
    %val = loadl %src_ptr
    %dst_ptr = add %tmp, %off
    storel %val, %dst_ptr
    %idx = add %idx, %one
    jmp @copy_to_base_cond

@shift_exp
    %exp = shr %exp, %one
    jmp @exp_loop_cond

@zero_tmp
    %total = mul %size, const 2
    %idx = %zero
    jmp @zero_cond

@zero_cond
    %cond = csltl %idx, %total
    jnz %cond, @zero_body
    jmp @ret_zero

@zero_body
    %off = mul %idx, %eight
    %ptr = add %tmp, %off
    storel %zero, %ptr
    %idx = add %idx, %one
    jmp @zero_cond

@ret_zero
    ret %zero

@k_zero
    %i = %zero
    jmp @k_zero_i_cond

@k_zero_i_cond
    %cond = csltl %i, %n
    jnz %cond, @k_zero_i_body
    jmp @k_zero_zero_tmp

@k_zero_i_body
    %j = %zero
    jmp @k_zero_j_cond

@k_zero_j_cond
    %cond = csltl %j, %n
    jnz %cond, @k_zero_j_body
    %i = add %i, %one
    jmp @k_zero_i_cond

@k_zero_j_body
    %i_n = mul %i, %n
    %index = add %i_n, %j
    %off = mul %index, %eight
    %out_ptr = add %out, %off
    %eq = ceql %i, %j
    %val = sel %eq, %one, %zero
    storel %val, %out_ptr
    %j = add %j, %one
    jmp @k_zero_j_cond

@k_zero_zero_tmp
    %total = mul %size, const 2
    %idx = %zero
    jmp @k_zero_cond

@k_zero_cond
    %cond = csltl %idx, %total
    jnz %cond, @k_zero_body
    jmp @k_zero_ret

@k_zero_body
    %off = mul %idx, %eight
    %ptr = add %tmp, %off
    storel %zero, %ptr
    %idx = add %idx, %one
    jmp @k_zero_cond

@k_zero_ret
    ret %zero
}
