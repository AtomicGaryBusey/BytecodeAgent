export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %zero =l copy 0
    %one =l copy 1
    %two =l copy 2
    %eight =l copy 8
    %size =l mull %n, %n
    %size8 =l mull %size, %eight
    %temp_base =l addl %tmp, %size8

    %k_zero =w ceql %k, %zero
    jnz %k_zero, @k_zero

    %idx =l copy %zero
    jmp @copy_loop_cond

@copy_loop_cond
    %cond =w csltl %idx, %size
    jnz %cond, @copy_loop_body
    jmp @init_ident

@copy_loop_body
    %off =l mull %idx, %eight
    %a_ptr =l addl %a, %off
    %val =l loadl %a_ptr
    %base_ptr =l addl %tmp, %off
    storel %val, %base_ptr
    %idx =l addl %idx, %one
    jmp @copy_loop_cond

@init_ident
    %i =l copy %zero
    jmp @ident_i_cond

@ident_i_cond
    %cond =w csltl %i, %n
    jnz %cond, @ident_i_body
    jmp @exp_loop_start

@ident_i_body
    %j =l copy %zero
    jmp @ident_j_cond

@ident_j_cond
    %cond =w csltl %j, %n
    jnz %cond, @ident_j_body
    %i =l addl %i, %one
    jmp @ident_i_cond

@ident_j_body
    %i_n =l mull %i, %n
    %index =l addl %i_n, %j
    %off =l mull %index, %eight
    %out_ptr =l addl %out, %off
    %eq =w ceql %i, %j
    %val =l sel %eq, %one, %zero
    storel %val, %out_ptr
    %j =l addl %j, %one
    jmp @ident_j_cond

@exp_loop_start
    %exp =l copy %k
    jmp @exp_loop_cond

@exp_loop_cond
    %cond =w csgtl %exp, %zero
    jnz %cond, @exp_loop_body
    jmp @zero_tmp

@exp_loop_body
    %bit =l andl %exp, %one
    %bit_nz =w cnel %bit, %zero
    jnz %bit_nz, @mult_res_base
    jmp @square_base

@mult_res_base
    %i =l copy %zero
    jmp @mult_res_i_cond

@mult_res_i_cond
    %cond =w csltl %i, %n
    jnz %cond, @mult_res_i_body
    jmp @copy_temp_to_out

@mult_res_i_body
    %j =l copy %zero
    jmp @mult_res_j_cond

@mult_res_j_cond
    %cond =w csltl %j, %n
    jnz %cond, @mult_res_j_body
    %i =l addl %i, %one
    jmp @mult_res_i_cond

@mult_res_j_body
    %sum =l copy %zero
    %t =l copy %zero
    jmp @mult_res_t_cond

@mult_res_t_cond
    %cond =w csltl %t, %n
    jnz %cond, @mult_res_t_body
    %i_n =l mull %i, %n
    %index =l addl %i_n, %j
    %off =l mull %index, %eight
    %dest_ptr =l addl %temp_base, %off
    storel %sum, %dest_ptr
    %j =l addl %j, %one
    jmp @mult_res_j_cond

@mult_res_t_body
    %i_n =l mull %i, %n
    %left_idx =l addl %i_n, %t
    %left_off =l mull %left_idx, %eight
    %left_ptr =l addl %out, %left_off
    %left_val =l loadl %left_ptr
    %t_n =l mull %t, %n
    %right_idx =l addl %t_n, %j
    %right_off =l mull %right_idx, %eight
    %right_ptr =l addl %tmp, %right_off
    %right_val =l loadl %right_ptr
    %prod =l mull %left_val, %right_val
    %sum =l addl %sum, %prod
    %t =l addl %t, %one
    jmp @mult_res_t_cond

@copy_temp_to_out
    %idx =l copy %zero
    jmp @copy_to_out_cond

@copy_to_out_cond
    %cond =w csltl %idx, %size
    jnz %cond, @copy_to_out_body
    jmp @square_base

@copy_to_out_body
    %off =l mull %idx, %eight
    %src_ptr =l addl %temp_base, %off
    %val =l loadl %src_ptr
    %dst_ptr =l addl %out, %off
    storel %val, %dst_ptr
    %idx =l addl %idx, %one
    jmp @copy_to_out_cond

@square_base
    %i =l copy %zero
    jmp @mult_sq_i_cond

@mult_sq_i_cond
    %cond =w csltl %i, %n
    jnz %cond, @mult_sq_i_body
    jmp @copy_temp_to_base

@mult_sq_i_body
    %j =l copy %zero
    jmp @mult_sq_j_cond

@mult_sq_j_cond
    %cond =w csltl %j, %n
    jnz %cond, @mult_sq_j_body
    %i =l addl %i, %one
    jmp @mult_sq_i_cond

@mult_sq_j_body
    %sum =l copy %zero
    %t =l copy %zero
    jmp @mult_sq_t_cond

@mult_sq_t_cond
    %cond =w csltl %t, %n
    jnz %cond, @mult_sq_t_body
    %i_n =l mull %i, %n
    %index =l addl %i_n, %j
    %off =l mull %index, %eight
    %dest_ptr =l addl %temp_base, %off
    storel %sum, %dest_ptr
    %j =l addl %j, %one
    jmp @mult_sq_j_cond

@mult_sq_t_body
    %i_n =l mull %i, %n
    %left_idx =l addl %i_n, %t
    %left_off =l mull %left_idx, %eight
    %left_ptr =l addl %tmp, %left_off
    %left_val =l loadl %left_ptr
    %t_n =l mull %t, %n
    %right_idx =l addl %t_n, %j
    %right_off =l mull %right_idx, %eight
    %right_ptr =l addl %tmp, %right_off
    %right_val =l loadl %right_ptr
    %prod =l mull %left_val, %right_val
    %sum =l addl %sum, %prod
    %t =l addl %t, %one
    jmp @mult_sq_t_cond

@copy_temp_to_base
    %idx =l copy %zero
    jmp @copy_to_base_cond

@copy_to_base_cond
    %cond =w csltl %idx, %size
    jnz %cond, @copy_to_base_body
    jmp @shift_exp

@copy_to_base_body
    %off =l mull %idx, %eight
    %src_ptr =l addl %temp_base, %off
    %val =l loadl %src_ptr
    %dst_ptr =l addl %tmp, %off
    storel %val, %dst_ptr
    %idx =l addl %idx, %one
    jmp @copy_to_base_cond

@shift_exp
    %exp =l shrl %exp, %one
    jmp @exp_loop_cond

@zero_tmp
    %total =l addl %size, %size
    %idx =l copy %zero
    jmp @zero_cond

@zero_cond
    %cond =w csltl %idx, %total
    jnz %cond, @zero_body
    jmp @ret_zero

@zero_body
    %off =l mull %idx, %eight
    %ptr =l addl %tmp, %off
    storel %zero, %ptr
    %idx =l addl %idx, %one
    jmp @zero_cond

@ret_zero
    ret %zero

@k_zero
    %i =l copy %zero
    jmp @k_zero_i_cond

@k_zero_i_cond
    %cond =w csltl %i, %n
    jnz %cond, @k_zero_i_body
    jmp @k_zero_zero_tmp

@k_zero_i_body
    %j =l copy %zero
    jmp @k_zero_j_cond

@k_zero_j_cond
    %cond =w csltl %j, %n
    jnz %cond, @k_zero_j_body
    %i =l addl %i, %one
    jmp @k_zero_i_cond

@k_zero_j_body
    %i_n =l mull %i, %n
    %index =l addl %i_n, %j
    %off =l mull %index, %eight
    %out_ptr =l addl %out, %off
    %eq =w ceql %i, %j
    %val =l sel %eq, %one, %zero
    storel %val, %out_ptr
    %j =l addl %j, %one
    jmp @k_zero_j_cond

@k_zero_zero_tmp
    %total =l addl %size, %size
    %idx =l copy %zero
    jmp @k_zero_cond

@k_zero_cond
    %cond =w csltl %idx, %total
    jnz %cond, @k_zero_body
    jmp @k_zero_ret

@k_zero_body
    %off =l mull %idx, %eight
    %ptr =l addl %tmp, %off
    storel %zero, %ptr
    %idx =l addl %idx, %one
    jmp @k_zero_cond

@k_zero_ret
    ret %zero
}
