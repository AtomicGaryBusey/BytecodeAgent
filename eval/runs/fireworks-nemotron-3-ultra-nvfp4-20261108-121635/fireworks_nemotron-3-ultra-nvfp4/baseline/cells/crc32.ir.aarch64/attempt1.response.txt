export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 0xFFFFFFFF
	%i0 =l copy 0
	jmp @loop(%crc0, %i0)

@loop(w %crc, l %i)
	%cond =w ult %i, %n
	jnz %cond, @body(%crc, %i), @exit(%crc)

@body(w %crc, l %i)
	%byte =w loadub %p, %i
	%crc1 =w xor %crc, %byte
	jmp @inner(%crc1, l 0)

@inner(w %crc_inner, l %bit)
	%bit_cond =w ult %bit, 8
	jnz %bit_cond, @inner_body(%crc_inner, %bit), @inner_done(%crc_inner, %i)

@inner_body(w %crc_inner, l %bit)
	%lsb =w and %crc_inner, 1
	%shr =w shr %crc_inner, 1
	jnz %lsb, @lsb_one(%shr), @lsb_zero(%shr)

@lsb_one(w %shr)
	%crc_next =w xor %shr, 0xEDB88320
	jmp @inner_next(%crc_next, %bit)

@lsb_zero(w %shr)
	%crc_next =w copy %shr
	jmp @inner_next(%crc_next, %bit)

@inner_next(w %crc_next, l %bit)
	%bit_next =l add %bit, 1
	jmp @inner(%crc_next, %bit_next)

@inner_done(w %crc_inner, l %i)
	%i_next =l add %i, 1
	jmp @loop(%crc_inner, %i_next)

@exit(w %crc)
	%result =w xor %crc, 0xFFFFFFFF
	%result_l =l extuw %result
	ret %result_l
}
