export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
	jmp @loop(%crc0, %i0)

@loop(w %crc, l %i)
	%cond =w ult %i, %n
	jnz %cond, @body, @exit

@body(w %crc, l %i)
	%byte =w loadub %p, %i
	%crc1 =w xor %crc, %byte
	jmp @inner(%crc1, l 0, %i)

@inner(w %crc_inner, l %bit, l %i)
	%bit_cond =w ult %bit, 8
	jnz %bit_cond, @inner_body, @inner_done

@inner_body(w %crc_inner, l %bit, l %i)
	%lsb =w and %crc_inner, 1
	%shr =w shr %crc_inner, 1
	%mask =w neg %lsb
	%mask =w and %mask, 3988292384
	%crc_next =w xor %shr, %mask
	%bit_next =l add %bit, 1
	jmp @inner(%crc_next, %bit_next, %i)

@inner_done(w %crc_inner, l %bit, l %i)
	%i_next =l add %i, 1
	jmp @loop(%crc_inner, %i_next)

@exit(w %crc, l %i)
	%result =w xor %crc, 4294967295
	%result_l =l extuw %result
	ret %result_l
}
