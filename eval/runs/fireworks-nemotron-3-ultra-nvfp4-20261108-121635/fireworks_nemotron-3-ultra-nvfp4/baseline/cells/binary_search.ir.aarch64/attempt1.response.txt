export function l $binary_search(l %a, l %n, l %key) {
@start
	%low.0 = copy 0
	%high.0 = subl %n, 1
	jmp @loop(%low.0, %high.0)

@loop(l %low, l %high)
	%cond = csgtl %low, %high
	jnz %cond, @not_found

	%diff = subl %high, %low
	%half = shrl %diff, 1
	%mid = addl %low, %half

	%mid8 = mul %mid, 8
	%addr = addl %a, %mid8
	%val = loadl %addr, 0

	%eq = ceql %val, %key
	jnz %eq, @found(%mid)

	%cmp = csltl %val, %key
	jnz %cmp, @less(%mid, %high)

	%high2 = subl %mid, 1
	jmp @loop(%low, %high2)

@less(l %mid, l %high)
	%low2 = addl %mid, 1
	jmp @loop(%low2, %high)

@found(l %mid)
	ret %mid

@not_found
	ret -1
}
