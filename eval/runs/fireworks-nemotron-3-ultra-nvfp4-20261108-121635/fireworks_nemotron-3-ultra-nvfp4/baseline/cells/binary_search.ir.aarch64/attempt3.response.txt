export function l $binary_search(l %a, l %n, l %key) {
@start
	%low.0 = add 0, 0
	%high.0 = sub %n, 1
	jmp @loop(%low.0, %high.0)

@loop(l %low, l %high)
	%cond = csgt %low, %high
	jnz %cond, @not_found

	%diff = sub %high, %low
	%half = shr %diff, 1
	%mid = add %low, %half

	%mid8 = mul %mid, 8
	%addr = add %a, %mid8
	%val = load %addr, 0

	%eq = ceq %val, %key
	jnz %eq, @found(%mid)

	%cmp = cslt %val, %key
	jnz %cmp, @less(%mid, %high)

	%high2 = sub %mid, 1
	jmp @loop(%low, %high2)

@less(l %mid, l %high)
	%low2 = add %mid, 1
	jmp @loop(%low2, %high)

@found(l %mid)
	ret %mid

@not_found
	ret -1
}
