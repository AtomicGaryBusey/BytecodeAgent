export function l $base64_encode(l %src, l %n, l %out) {
@start
    %full_groups =l div %n, 3
    %rem =l mod %n, 3
    %src_ptr =l copy %src
    %out_ptr =l copy %out
    %i =l copy 0
    jmp @loop_cond

@loop_cond
    %cond =w cltu %i, %full_groups
    jnz %cond, @loop_body, @loop_end

@loop_body
    %b0 =l loadub %src_ptr
    %src_ptr =l add %src_ptr, 1
    %b1 =l loadub %src_ptr
    %src_ptr =l add %src_ptr, 1
    %b2 =l loadub %src_ptr
    %src_ptr =l add %src_ptr, 1

    %tmp0 =l shl %b0, 16
    %tmp1 =l shl %b1, 8
    %tmp2 =l or %tmp0, %tmp1
    %val =l or %tmp2, %b2

    %idx0 =l and (shr %val, 18), 63
    %idx1 =l and (shr %val, 12), 63
    %idx2 =l and (shr %val, 6), 63
    %idx3 =l and %val, 63

    %is_upper0 =w cltu %idx0, 26
    %is_lower0 =w cltu %idx0, 52
    %is_digit0 =w cltu %idx0, 62
    %is_plus0 =w ceq %idx0, 62
    %c_upper0 =l add %idx0, 65
    %c_lower0 =l add %idx0, 71
    %c_digit0 =l add %idx0, -4
    %c_plus0 =l copy 43
    %c_slash0 =l copy 47
    %tmp3_0 =l csel %is_upper0, %c_upper0, %c_lower0
    %tmp4_0 =l csel %is_lower0, %tmp3_0, %c_digit0
    %tmp5_0 =l csel %is_digit0, %tmp4_0, %c_plus0
    %c0 =l csel %is_plus0, %tmp5_0, %c_slash0
    storeb %c0, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %is_upper1 =w cltu %idx1, 26
    %is_lower1 =w cltu %idx1, 52
    %is_digit1 =w cltu %idx1, 62
    %is_plus1 =w ceq %idx1, 62
    %c_upper1 =l add %idx1, 65
    %c_lower1 =l add %idx1, 71
    %c_digit1 =l add %idx1, -4
    %c_plus1 =l copy 43
    %c_slash1 =l copy 47
    %tmp3_1 =l csel %is_upper1, %c_upper1, %c_lower1
    %tmp4_1 =l csel %is_lower1, %tmp3_1, %c_digit1
    %tmp5_1 =l csel %is_digit1, %tmp4_1, %c_plus1
    %c1 =l csel %is_plus1, %tmp5_1, %c_slash1
    storeb %c1, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %is_upper2 =w cltu %idx2, 26
    %is_lower2 =w cltu %idx2, 52
    %is_digit2 =w cltu %idx2, 62
    %is_plus2 =w ceq %idx2, 62
    %c_upper2 =l add %idx2, 65
    %c_lower2 =l add %idx2, 71
    %c_digit2 =l add %idx2, -4
    %c_plus2 =l copy 43
    %c_slash2 =l copy 47
    %tmp3_2 =l csel %is_upper2, %c_upper2, %c_lower2
    %tmp4_2 =l csel %is_lower2, %tmp3_2, %c_digit2
    %tmp5_2 =l csel %is_digit2, %tmp4_2, %c_plus2
    %c2 =l csel %is_plus2, %tmp5_2, %c_slash2
    storeb %c2, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %is_upper3 =w cltu %idx3, 26
    %is_lower3 =w cltu %idx3, 52
    %is_digit3 =w cltu %idx3, 62
    %is_plus3 =w ceq %idx3, 62
    %c_upper3 =l add %idx3, 65
    %c_lower3 =l add %idx3, 71
    %c_digit3 =l add %idx3, -4
    %c_plus3 =l copy 43
    %c_slash3 =l copy 47
    %tmp3_3 =l csel %is_upper3, %c_upper3, %c_lower3
    %tmp4_3 =l csel %is_lower3, %tmp3_3, %c_digit3
    %tmp5_3 =l csel %is_digit3, %tmp4_3, %c_plus3
    %c3 =l csel %is_plus3, %tmp5_3, %c_slash3
    storeb %c3, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %i =l add %i, 1
    jmp @loop_cond

@loop_end
    %rem_eq_1 =w ceq %rem, 1
    jnz %rem_eq_1, @rem1, @rem_not1

@rem_not1
    %rem_eq_2 =w ceq %rem, 2
    jnz %rem_eq_2, @rem2, @end

@rem1
    %b0 =l loadub %src_ptr
    %val =l shl %b0, 16
    %idx0 =l and (shr %val, 18), 63
    %idx1 =l and (shr %val, 12), 63

    %is_upper0r =w cltu %idx0, 26
    %is_lower0r =w cltu %idx0, 52
    %is_digit0r =w cltu %idx0, 62
    %is_plus0r =w ceq %idx0, 62
    %c_upper0r =l add %idx0, 65
    %c_lower0r =l add %idx0, 71
    %c_digit0r =l add %idx0, -4
    %c_plus0r =l copy 43
    %c_slash0r =l copy 47
    %tmp3_0r =l csel %is_upper0r, %c_upper0r, %c_lower0r
    %tmp4_0r =l csel %is_lower0r, %tmp3_0r, %c_digit0r
    %tmp5_0r =l csel %is_digit0r, %tmp4_0r, %c_plus0r
    %c0r =l csel %is_plus0r, %tmp5_0r, %c_slash0r
    storeb %c0r, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %is_upper1r =w cltu %idx1, 26
    %is_lower1r =w cltu %idx1, 52
    %is_digit1r =w cltu %idx1, 62
    %is_plus1r =w ceq %idx1, 62
    %c_upper1r =l add %idx1, 65
    %c_lower1r =l add %idx1, 71
    %c_digit1r =l add %idx1, -4
    %c_plus1r =l copy 43
    %c_slash1r =l copy 47
    %tmp3_1r =l csel %is_upper1r, %c_upper1r, %c_lower1r
    %tmp4_1r =l csel %is_lower1r, %tmp3_1r, %c_digit1r
    %tmp5_1r =l csel %is_digit1r, %tmp4_1r, %c_plus1r
    %c1r =l csel %is_plus1r, %tmp5_1r, %c_slash1r
    storeb %c1r, %out_ptr
    %out_ptr =l add %out_ptr, 1

    storeb 61, %out_ptr
    %out_ptr =l add %out_ptr, 1
    storeb 61, %out_ptr
    %out_ptr =l add %out_ptr, 1
    jmp @end

@rem2
    %b0 =l loadub %src_ptr
    %src_ptr =l add %src_ptr, 1
    %b1 =l loadub %src_ptr
    %val =l or (shl %b0, 16), (shl %b1, 8)
    %idx0 =l and (shr %val, 18), 63
    %idx1 =l and (shr %val, 12), 63
    %idx2 =l and (shr %val, 6), 63

    %is_upper0r2 =w cltu %idx0, 26
    %is_lower0r2 =w cltu %idx0, 52
    %is_digit0r2 =w cltu %idx0, 62
    %is_plus0r2 =w ceq %idx0, 62
    %c_upper0r2 =l add %idx0, 65
    %c_lower0r2 =l add %idx0, 71
    %c_digit0r2 =l add %idx0, -4
    %c_plus0r2 =l copy 43
    %c_slash0r2 =l copy 47
    %tmp3_0r2 =l csel %is_upper0r2, %c_upper0r2, %c_lower0r2
    %tmp4_0r2 =l csel %is_lower0r2, %tmp3_0r2, %c_digit0r2
    %tmp5_0r2 =l csel %is_digit0r2, %tmp4_0r2, %c_plus0r2
    %c0r2 =l csel %is_plus0r2, %tmp5_0r2, %c_slash0r2
    storeb %c0r2, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %is_upper1r2 =w cltu %idx1, 26
    %is_lower1r2 =w cltu %idx1, 52
    %is_digit1r2 =w cltu %idx1, 62
    %is_plus1r2 =w ceq %idx1, 62
    %c_upper1r2 =l add %idx1, 65
    %c_lower1r2 =l add %idx1, 71
    %c_digit1r2 =l add %idx1, -4
    %c_plus1r2 =l copy 43
    %c_slash1r2 =l copy 47
    %tmp3_1r2 =l csel %is_upper1r2, %c_upper1r2, %c_lower1r2
    %tmp4_1r2 =l csel %is_lower1r2, %tmp3_1r2, %c_digit1r2
    %tmp5_1r2 =l csel %is_digit1r2, %tmp4_1r2, %c_plus1r2
    %c1r2 =l csel %is_plus1r2, %tmp5_1r2, %c_slash1r2
    storeb %c1r2, %out_ptr
    %out_ptr =l add %out_ptr, 1

    %is_upper2r2 =w cltu %idx2, 26
    %is_lower2r2 =w cltu %idx2, 52
    %is_digit2r2 =w cltu %idx2, 62
    %is_plus2r2 =w ceq %idx2, 62
    %c_upper2r2 =l add %idx2, 65
    %c_lower2r2 =l add %idx2, 71
    %c_digit2r2 =l add %idx2, -4
    %c_plus2r2 =l copy 43
    %c_slash2r2 =l copy 47
    %tmp3_2r2 =l csel %is_upper2r2, %c_upper2r2, %c_lower2r2
    %tmp4_2r2 =l csel %is_lower2r2, %tmp3_2r2, %c_digit2r2
    %tmp5_2r2 =l csel %is_digit2r2, %tmp4_2r2, %c_plus2r2
    %c2r2 =l csel %is_plus2r2, %tmp5_2r2, %c_slash2r2
    storeb %c2r2, %out_ptr
    %out_ptr =l add %out_ptr, 1

    storeb 61, %out_ptr
    %out_ptr =l add %out_ptr, 1
    jmp @end

@end
    %rem_neq =w cne %rem, 0
    %rem_neq_l =l zext %rem_neq
    %total_groups =l add %full_groups, %rem_neq_l
    %out_len =l mul 4, %total_groups
    ret %out_len
}
