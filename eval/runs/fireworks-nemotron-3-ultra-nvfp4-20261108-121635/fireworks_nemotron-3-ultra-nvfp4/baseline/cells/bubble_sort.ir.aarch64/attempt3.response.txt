export function l $bubble_sort(l %a, l %n) {
    @start
    %n_le_1 = csle %n, l 1
    jnz %n_le_1, @return
    %n_minus_1 = subl %n, l 1
    jmp @outer l 0

    @outer l %i
    %i_lt = cslt %i, %n_minus_1
    jnz %i_lt, @inner %i l 0
    jmp @return

    @inner l %i l %j
    %limit = subl %n_minus_1, %i
    %j_lt = cslt %j, %limit
    jnz %j_lt, @body %i %j
    %i_next = addl %i, l 1
    jmp @outer %i_next

    @body l %i l %j
    %addr_j = addl %a, mull %j, l 8
    %val_j = loadl %addr_j
    %addr_j1 = addl %addr_j, l 8
    %val_j1 = loadl %addr_j1
    %cmp = csgt %val_j, %val_j1
    jnz %cmp, @swap %i %j %addr_j %addr_j1 %val_j %val_j1
    jmp @next %i %j

    @swap l %i l %j l %addr_j l %addr_j1 l %val_j l %val_j1
    storel %addr_j, %val_j1
    storel %addr_j1, %val_j
    jmp @next %i %j

    @next l %i l %j
    %j_next = addl %j, l 1
    jmp @inner %i %j_next

    @return
    ret l 0
}
