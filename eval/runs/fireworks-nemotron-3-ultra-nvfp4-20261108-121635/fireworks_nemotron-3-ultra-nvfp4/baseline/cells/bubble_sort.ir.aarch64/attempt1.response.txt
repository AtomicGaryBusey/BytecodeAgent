export function l $bubble_sort(l %a, l %n) {
    @start
    %n_le_1 = cslel %n, l 1
    jnz %n_le_1, @return
    %n_minus_1 = subl %n, l 1
    jmp @outer_loop l 0

    @outer_loop l %i
    %i_lt = csltl %i, %n_minus_1
    jnz %i_lt, @inner_loop %i l 0
    jmp @return

    @inner_loop l %i l %j
    %inner_limit = subl %n_minus_1, %i
    %j_lt = csltl %j, %inner_limit
    jnz %j_lt, @inner_body %i %j
    %i_next = addl %i, l 1
    jmp @outer_loop %i_next

    @inner_body l %i l %j
    %addr_j = addl %a, mull %j, l 8
    %val_j = loadl %addr_j
    %addr_j1 = addl %addr_j, l 8
    %val_j1 = loadl %addr_j1
    %cmp = csgtrl %val_j, %val_j1
    jnz %cmp, @swap %i %j %addr_j %addr_j1 %val_j %val_j1
    jmp @no_swap %i %j

    @swap l %i l %j l %addr_j l %addr_j1 l %val_j l %val_j1
    storel %addr_j, %val_j1
    storel %addr_j1, %val_j
    jmp @no_swap %i %j

    @no_swap l %i l %j
    %j_next = addl %j, l 1
    jmp @inner_loop %i %j_next

    @return
    ret l 0
}
