export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
    %src_cur =l copy %src
    %src_end =l add %src, %n
    %out_cur =l copy %out
    %out_end =l add %out, %cap
    %written =l copy 0
    jmp @loop

@loop
    %cond1 =w csl_ul %src_cur, %src_end
    %cond2 =w csl_ul %written, %cap
    %cond =w and %cond1, %cond2
    jnz %cond, @read_header, @exit_normal

@read_header
    %h =w loadsb %src_cur
    %src_cur =l add %src_cur, 1
    %h_ge_0 =w csge_w %h, 0
    jnz %h_ge_0, @literal_packet, @check_noop

@check_noop
    %h_eq_128 =w ceq_w %h, -128
    jnz %h_eq_128, @noop_packet, @repeat_packet

@noop_packet
    jmp @loop

@literal_packet
    %len =w add %h, 1
    %len_l =l sexw %len
    %src_needed =l add %src_cur, %len_l
    %enough_src =w csle_ul %src_needed, %src_end
    jnz %enough_src, @literal_check_out, @error_src

@literal_check_out
    %rem_out =l sub %cap, %written
    %rem_out_w =w trunc %rem_out
    %cmp =w csle_w %len, %rem_out_w
    jnz %cmp, @literal_copy_all, @literal_copy_partial

@literal_copy_all
    %copy_cnt =w copy %len
    %copy_src =l copy %src_cur
    %copy_dst =l copy %out_cur
    jmp @literal_copy_loop

@literal_copy_loop
    %copy_cnt_zero =w ceq_w %copy_cnt, 0
    jnz %copy_cnt_zero, @literal_copy_done, @literal_copy_do

@literal_copy_do
    %byte =w loadub %copy_src
    storeb %byte, %copy_dst
    %copy_src =l add %copy_src, 1
    %copy_dst =l add %copy_dst, 1
    %copy_cnt =w sub %copy_cnt, 1
    jmp @literal_copy_loop

@literal_copy_done
    %src_cur =l copy %copy_src
    %out_cur =l copy %copy_dst
    %written =l add %written, %len_l
    jmp @loop

@literal_copy_partial
    %copy_cnt =w copy %rem_out_w
    %copy_src =l copy %src_cur
    %copy_dst =l copy %out_cur
    jmp @literal_copy_partial_loop

@literal_copy_partial_loop
    %copy_cnt_zero =w ceq_w %copy_cnt, 0
    jnz %copy_cnt_zero, @literal_copy_partial_done, @literal_copy_partial_do

@literal_copy_partial_do
    %byte =w loadub %copy_src
    storeb %byte, %copy_dst
    %copy_src =l add %copy_src, 1
    %copy_dst =l add %copy_dst, 1
    %copy_cnt =w sub %copy_cnt, 1
    jmp @literal_copy_partial_loop

@literal_copy_partial_done
    %written =l copy %cap
    ret_l -1

@repeat_packet
    %repeat_count =w sub 1, %h
    %repeat_count_l =l sexw %repeat_count
    %enough_src =w csl_ul %src_cur, %src_end
    jnz %enough_src, @repeat_read_val, @error_src

@repeat_read_val
    %val =w loadub %src_cur
    %src_cur =l add %src_cur, 1
    %rem_out =l sub %cap, %written
    %rem_out_w =w trunc %rem_out
    %cmp =w csle_w %repeat_count, %rem_out_w
    jnz %cmp, @repeat_write_all, @repeat_write_partial

@repeat_write_all
    %write_cnt =w copy %repeat_count
    %write_dst =l copy %out_cur
    jmp @repeat_write_loop

@repeat_write_loop
    %write_cnt_zero =w ceq_w %write_cnt, 0
    jnz %write_cnt_zero, @repeat_write_done, @repeat_write_do

@repeat_write_do
    storeb %val, %write_dst
    %write_dst =l add %write_dst, 1
    %write_cnt =w sub %write_cnt, 1
    jmp @repeat_write_loop

@repeat_write_done
    %out_cur =l copy %write_dst
    %written =l add %written, %repeat_count_l
    jmp @loop

@repeat_write_partial
    %write_cnt =w copy %rem_out_w
    %write_dst =l copy %out_cur
    jmp @repeat_write_partial_loop

@repeat_write_partial_loop
    %write_cnt_zero =w ceq_w %write_cnt, 0
    jnz %write_cnt_zero, @repeat_write_partial_done, @repeat_write_partial_do

@repeat_write_partial_do
    storeb %val, %write_dst
    %write_dst =l add %write_dst, 1
    %write_cnt =w sub %write_cnt, 1
    jmp @repeat_write_partial_loop

@repeat_write_partial_done
    %written =l copy %cap
    ret_l -1

@error_src
    ret_l -2

@exit_normal
    ret_l %written
}
