export function l $base64_encode(l %src, l %n, l %out) {
@start
  %zero =l add 0, 0
  %full_groups =l div %n, 3
  %rem =l mod %n, 3
  %out_len_tmp =l add %n, 2
  %out_len_div =l div %out_len_tmp, 3
  %out_len =l mul %out_len_div, 4
  jmp @loop_head

@loop_head
  %i =l phi @start %zero, @loop_body %i_next
  %src_ptr =l phi @start %src, @loop_body %src_ptr_next
  %out_ptr =l phi @start %out, @loop_body %out_ptr_next
  %cond =l ult %i, %full_groups
  jnz %cond, @loop_body, @loop_end

@loop_body
  %b0_l =l loadub %src_ptr
  %addr1_l =l add %src_ptr, 1
  %b1_l =l loadub %addr1_l
  %addr2_l =l add %src_ptr, 2
  %b2_l =l loadub %addr2_l

  %val0_l =l shl %b0_l, 16
  %val1_l =l shl %b1_l, 8
  %tmp_val_l =l or %val0_l, %val1_l
  %val_l =l or %tmp_val_l, %b2_l

  %tmp_shr18_l =l shr %val_l, 18
  %idx0_l =l and %tmp_shr18_l, 63
  %tmp_shr12_l =l shr %val_l, 12
  %idx1_l =l and %tmp_shr12_l, 63
  %tmp_shr6_l =l shr %val_l, 6
  %idx2_l =l and %tmp_shr6_l, 63
  %idx3_l =l and %val_l, 63

  %lt26_l0 =l ult %idx0_l, 26
  %lt52_l0 =l ult %idx0_l, 52
  %lt62_l0 =l ult %idx0_l, 62
  %c1_l0 =l add 65, %idx0_l
  %sub26_l0 =l sub %idx0_l, 26
  %c2_l0 =l add 97, %sub26_l0
  %sub52_l0 =l sub %idx0_l, 52
  %c3_l0 =l add 48, %sub52_l0
  %c4_l0 =l csel %lt62_l0, 43, 47
  %c34_l0 =l csel %lt62_l0, %c3_l0, %c4_l0
  %c234_l0 =l csel %lt52_l0, %c2_l0, %c34_l0
  %char0_l =l csel %lt26_l0, %c1_l0, %c234_l0

  %lt26_l1 =l ult %idx1_l, 26
  %lt52_l1 =l ult %idx1_l, 52
  %lt62_l1 =l ult %idx1_l, 62
  %c1_l1 =l add 65, %idx1_l
  %sub26_l1 =l sub %idx1_l, 26
  %c2_l1 =l add 97, %sub26_l1
  %sub52_l1 =l sub %idx1_l, 52
  %c3_l1 =l add 48, %sub52_l1
  %c4_l1 =l csel %lt62_l1, 43, 47
  %c34_l1 =l csel %lt62_l1, %c3_l1, %c4_l1
  %c234_l1 =l csel %lt52_l1, %c2_l1, %c34_l1
  %char1_l =l csel %lt26_l1, %c1_l1, %c234_l1

  %lt26_l2 =l ult %idx2_l, 26
  %lt52_l2 =l ult %idx2_l, 52
  %lt62_l2 =l ult %idx2_l, 62
  %c1_l2 =l add 65, %idx2_l
  %sub26_l2 =l sub %idx2_l, 26
  %c2_l2 =l add 97, %sub26_l2
  %sub52_l2 =l sub %idx2_l, 52
  %c3_l2 =l add 48, %sub52_l2
  %c4_l2 =l csel %lt62_l2, 43, 47
  %c34_l2 =l csel %lt62_l2, %c3_l2, %c4_l2
  %c234_l2 =l csel %lt52_l2, %c2_l2, %c34_l2
  %char2_l =l csel %lt26_l2, %c1_l2, %c234_l2

  %lt26_l3 =l ult %idx3_l, 26
  %lt52_l3 =l ult %idx3_l, 52
  %lt62_l3 =l ult %idx3_l, 62
  %c1_l3 =l add 65, %idx3_l
  %sub26_l3 =l sub %idx3_l, 26
  %c2_l3 =l add 97, %sub26_l3
  %sub52_l3 =l sub %idx3_l, 52
  %c3_l3 =l add 48, %sub52_l3
  %c4_l3 =l csel %lt62_l3, 43, 47
  %c34_l3 =l csel %lt62_l3, %c3_l3, %c4_l3
  %c234_l3 =l csel %lt52_l3, %c2_l3, %c34_l3
  %char3_l =l csel %lt26_l3, %c1_l3, %c234_l3

  storeb %char0_l, %out_ptr
  %out_ptr1_l =l add %out_ptr, 1
  storeb %char1_l, %out_ptr1_l
  %out_ptr2_l =l add %out_ptr, 2
  storeb %char2_l, %out_ptr2_l
  %out_ptr3_l =l add %out_ptr, 3
  storeb %char3_l, %out_ptr3_l

  %src_ptr_next =l add %src_ptr, 3
  %out_ptr_next =l add %out_ptr, 4
  %i_next =l add %i, 1
  jmp @loop_head

@loop_end
  %src_ptr_final =l add %src, (mul %full_groups, 3)
  %out_ptr_final =l add %out, (mul %full_groups, 4)
  %rem_eq1 =l ceq %rem, 1
  %rem_eq2 =l ceq %rem, 2
  jnz %rem_eq1, @rem1, @check_rem2

@check_rem2
  jnz %rem_eq2, @rem2, @done

@rem1
  %b0_r1 =l loadub %src_ptr_final
  %val_r1 =l shl %b0_r1, 16
  %tmp_shr18_r1 =l shr %val_r1, 18
  %idx0_r1 =l and %tmp_shr18_r1, 63
  %tmp_shr12_r1 =l shr %val_r1, 12
  %idx1_r1 =l and %tmp_shr12_r1, 63

  %lt26_r10 =l ult %idx0_r1, 26
  %lt52_r10 =l ult %idx0_r1, 52
  %lt62_r10 =l ult %idx0_r1, 62
  %c1_r10 =l add 65, %idx0_r1
  %sub26_r10 =l sub %idx0_r1, 26
  %c2_r10 =l add 97, %sub26_r10
  %sub52_r10 =l sub %idx0_r1, 52
  %c3_r10 =l add 48, %sub52_r10
  %c4_r10 =l csel %lt62_r10, 43, 47
  %c34_r10 =l csel %lt62_r10, %c3_r10, %c4_r10
  %c234_r10 =l csel %lt52_r10, %c2_r10, %c34_r10
  %char0_r1 =l csel %lt26_r10, %c1_r10, %c234_r10

  %lt26_r11 =l ult %idx1_r1, 26
  %lt52_r11 =l ult %idx1_r1, 52
  %lt62_r11 =l ult %idx1_r1, 62
  %c1_r11 =l add 65, %idx1_r1
  %sub26_r11 =l sub %idx1_r1, 26
  %c2_r11 =l add 97, %sub26_r11
  %sub52_r11 =l sub %idx1_r1, 52
  %c3_r11 =l add 48, %sub52_r11
  %c4_r11 =l csel %lt62_r11, 43, 47
  %c34_r11 =l csel %lt62_r11, %c3_r11, %c4_r11
  %c234_r11 =l csel %lt52_r11, %c2_r11, %c34_r11
  %char1_r1 =l csel %lt26_r11, %c1_r11, %c234_r11

  storeb %char0_r1, %out_ptr_final
  %out_ptr_final1_r1 =l add %out_ptr_final, 1
  storeb %char1_r1, %out_ptr_final1_r1
  %out_ptr_final2_r1 =l add %out_ptr_final, 2
  storeb 61, %out_ptr_final2_r1
  %out_ptr_final3_r1 =l add %out_ptr_final, 3
  storeb 61, %out_ptr_final3_r1
  jmp @done

@rem2
  %b0_r2 =l loadub %src_ptr_final
  %addr1_r2 =l add %src_ptr_final, 1
  %b1_r2 =l loadub %addr1_r2
  %val0_r2 =l shl %b0_r2, 16
  %val1_r2 =l shl %b1_r2, 8
  %val_r2 =l or %val0_r2, %val1_r2

  %tmp_shr18_r2 =l shr %val_r2, 18
  %idx0_r2 =l and %tmp_shr18_r2, 63
  %tmp_shr12_r2 =l shr %val_r2, 12
  %idx1_r2 =l and %tmp_shr12_r2, 63
  %tmp_shr6_r2 =l shr %val_r2, 6
  %idx2_r2 =l and %tmp_shr6_r2, 63

  %lt26_r20 =l ult %idx0_r2, 26
  %lt52_r20 =l ult %idx0_r2, 52
  %lt62_r20 =l ult %idx0_r2, 62
  %c1_r20 =l add 65, %idx0_r2
  %sub26_r20 =l sub %idx0_r2, 26
  %c2_r20 =l add 97, %sub26_r20
  %sub52_r20 =l sub %idx0_r2, 52
  %c3_r20 =l add 48, %sub52_r20
  %c4_r20 =l csel %lt62_r20, 43, 47
  %c34_r20 =l csel %lt62_r20, %c3_r20, %c4_r20
  %c234_r20 =l csel %lt52_r20, %c2_r20, %c34_r20
  %char0_r2 =l csel %lt26_r20, %c1_r20, %c234_r20

  %lt26_r21 =l ult %idx1_r2, 26
  %lt52_r21 =l ult %idx1_r2, 52
  %lt62_r21 =l ult %idx1_r2, 62
  %c1_r21 =l add 65, %idx1_r2
  %sub26_r21 =l sub %idx1_r2, 26
  %c2_r21 =l add 97, %sub26_r21
  %sub52_r21 =l sub %idx1_r2, 52
  %c3_r21 =l add 48, %sub52_r21
  %c4_r21 =l csel %lt62_r21, 43, 47
  %c34_r21 =l csel %lt62_r21, %c3_r21, %c4_r21
  %c234_r21 =l csel %lt52_r21, %c2_r21, %c34_r21
  %char1_r2 =l csel %lt26_r21, %c1_r21, %c234_r21

  %lt26_r22 =l ult %idx2_r2, 26
  %lt52_r22 =l ult %idx2_r2, 52
  %lt62_r22 =l ult %idx2_r2, 62
  %c1_r22 =l add 65, %idx2_r2
  %sub26_r22 =l sub %idx2_r2, 26
  %c2_r22 =l add 97, %sub26_r22
  %sub52_r22 =l sub %idx2_r2, 52
  %c3_r22 =l add 48, %sub52_r22
  %c4_r22 =l csel %lt62_r22, 43, 47
  %c34_r22 =l csel %lt62_r22, %c3_r22, %c4_r22
  %c234_r22 =l csel %lt52_r22, %c2_r22, %c34_r22
  %char2_r2 =l csel %lt26_r22, %c1_r22, %c234_r22

  storeb %char0_r2, %out_ptr_final
  %out_ptr_final1_r2 =l add %out_ptr_final, 1
  storeb %char1_r2, %out_ptr_final1_r2
  %out_ptr_final2_r2 =l add %out_ptr_final, 2
  storeb %char2_r2, %out_ptr_final2_r2
  %out_ptr_final3_r2 =l add %out_ptr_final, 3
  storeb 61, %out_ptr_final3_r2
  jmp @done

@done
  ret %out_len
}
