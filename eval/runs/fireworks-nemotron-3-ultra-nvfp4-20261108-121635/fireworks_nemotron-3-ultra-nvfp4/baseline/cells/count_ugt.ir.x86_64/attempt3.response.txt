export function l $count_ugt(l %a, l %n, l %t) {
@start
    jmp @loop

@loop
    %i = phi @start 0l, @body %i_next
    %count = phi @start 0l, @body %count_next
    %p = phi @start %a, @body %p_next
    %cond = cltl %i, %n
    jnz %cond, @body, @end

@body
    %elem = loadl %p
    %cmp = cugtl %elem, %t
    %inc = zextwl %cmp
    %count_next = addl %count, %inc
    %p_next = addl %p, 8
    %i_next = addl %i, 1
    jmp @loop

@end
    %final_count = phi @loop %count
    ret %final_count
}
