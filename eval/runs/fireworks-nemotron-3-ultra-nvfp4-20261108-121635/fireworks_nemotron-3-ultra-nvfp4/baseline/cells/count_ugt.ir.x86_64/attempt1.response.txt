export function l $count_ugt(l %a, l %n, l %t) {
@start
	%p = copy %a
	%i = copy 0
	%count = copy 0
	jmp @loop

@loop
	%cond = cltl %i, %n
	jnz %cond, @body, @end

@body
	%elem = loadl %p
	%cmp = cugtl %elem, %t
	%inc = zextwl %cmp
	%count = addl %count, %inc
	%p = addl %p, 8
	%i = addl %i, 1
	jmp @loop

@end
	ret %count
}
