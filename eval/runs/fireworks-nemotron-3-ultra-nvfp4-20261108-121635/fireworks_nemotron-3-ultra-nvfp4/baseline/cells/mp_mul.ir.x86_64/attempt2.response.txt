export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%zero =l copy 0
	%k =l copy 0
	jmp @zero_loop

@zero_loop
	%cond =w cultl %k, %total
	jnz %cond, @zero_body, @zero_done

@zero_body
	%offset =l shl %k, 3
	%addr =l add %r, %offset
	storel %addr, %zero
	%k =l add %k, 1
	jmp @zero_loop

@zero_done
	%i =l copy 0
	jmp @outer_loop

@outer_loop
	%cond =w cultl %i, %na
	jnz %cond, @outer_body, @outer_done

@outer_body
	%offset_i =l shl %i, 3
	%a_i_addr =l add %a, %offset_i
	%a_i =l loadl %a_i_addr
	%carry =l copy 0
	%j =l copy 0
	jmp @inner_loop

@inner_loop
	%cond =w cultl %j, %nb
	jnz %cond, @inner_body, @inner_done

@inner_body
	%offset_j =l shl %j, 3
	%b_j_addr =l add %b, %offset_j
	%b_j =l loadl %b_j_addr

	%a_lo =l and %a_i, 4294967295
	%a_hi =l shr %a_i, 32
	%b_lo =l and %b_j, 4294967295
	%b_hi =l shr %b_j, 32

	%p0 =l mul %a_lo, %b_lo
	%p1 =l mul %a_lo, %b_hi
	%p2 =l mul %a_hi, %b_lo
	%p3 =l mul %a_hi, %b_hi

	%p1_lo =l and %p1, 4294967295
	%p2_lo =l and %p2, 4294967295
	%sum_lo =l add %p1_lo, %p2_lo
	%carry_lo =w cultl %sum_lo, %p1_lo
	%p1_hi =l shr %p1, 32
	%p2_hi =l shr %p2, 32
	%mid_hi =l add %p1_hi, %p2_hi
	%carry_lo_l =l zext %carry_lo
	%mid_hi =l add %mid_hi, %carry_lo_l

	%sum_lo_shifted =l shl %sum_lo, 32
	%low =l add %p0, %sum_lo_shifted
	%carry_low =w cultl %low, %p0
	%carry_low_l =l zext %carry_low

	%high =l add %p3, %mid_hi
	%high =l add %high, %carry_low_l

	%ij =l add %i, %j
	%offset_ij =l shl %ij, 3
	%r_ij_addr =l add %r, %offset_ij
	%r_ij =l loadl %r_ij_addr

	%t1 =l add %r_ij, %low
	%c1 =w cultl %t1, %r_ij
	%t2 =l add %t1, %carry
	%c2 =w cultl %t2, %t1
	%new_r =l copy %t2
	%c1_l =l zext %c1
	%c2_l =l zext %c2
	%new_carry =l add %high, %c1_l
	%new_carry =l add %new_carry, %c2_l

	storel %r_ij_addr, %new_r
	%carry =l copy %new_carry
	%j =l add %j, 1
	jmp @inner_loop

@inner_done
	%i_nb =l add %i, %nb
	%offset_i_nb =l shl %i_nb, 3
	%r_i_nb_addr =l add %r, %offset_i_nb
	storel %r_i_nb_addr, %carry
	%i =l add %i, 1
	jmp @outer_loop

@outer_done
	%cond =w ceql %total, 0
	jnz %cond, @ret_zero, @ret_loop_start

@ret_zero
	%ret =l copy 0
	jmp @exit

@ret_loop_start
	%k =l sub %total, 1
	jmp @ret_loop

@ret_loop
	%offset_k =l shl %k, 3
	%r_k_addr =l add %r, %offset_k
	%val =l loadl %r_k_addr
	%cond =w cnel %val, 0
	jnz %cond, @ret_found, @ret_continue

@ret_found
	%ret =l add %k, 1
	jmp @exit

@ret_continue
	%cond =w ceql %k, 0
	jnz %cond, @ret_all_zero, @ret_next

@ret_next
	%k =l sub %k, 1
	jmp @ret_loop

@ret_all_zero
	%ret =l copy 0
	jmp @exit

@exit
	ret %ret
}
