lcs_len:
    // zero row[0..n]
    mov x5, #0
.Lzero_loop:
    str xzr, [x4, x5, lsl #3]
    add x5, x5, #1
    cmp x5, x3
    b.le .Lzero_loop

    // outer loop over i = 0..m-1
    mov x5, #0
    cbz x1, .Louter_done
.Louter_loop:
    ldrb w9, [x0, x5]       // x[i]
    mov x7, #0              // diag = 0
    mov x6, #1              // j = 1
    cmp x6, x3
    b.gt .Linner_done
.Linner_loop:
    ldr x8, [x4, x6, lsl #3]      // up = row[j]
    sub x11, x6, #1
    ldrb w10, [x2, x11]           // y[j-1]
    cmp w9, w10
    b.ne .Lnot_equal
    // equal: row[j] = diag + 1
    add x11, x7, #1
    str x11, [x4, x6, lsl #3]
    b .Lupdate_diag
.Lnot_equal:
    // row[j] = max(up, row[j-1])
    ldr x11, [x4, x11, lsl #3]    // row[j-1]
    cmp x8, x11
    csel x11, x8, x11, ge
    str x11, [x4, x6, lsl #3]
.Lupdate_diag:
    mov x7, x8              // diag = up
    add x6, x6, #1
    cmp x6, x3
    b.le .Linner_loop
.Linner_done:
    add x5, x5, #1
    cmp x5, x1
    b.lt .Louter_loop
.Louter_done:
    ldr x0, [x4, x3, lsl #3]      // return row[n]
    ret
