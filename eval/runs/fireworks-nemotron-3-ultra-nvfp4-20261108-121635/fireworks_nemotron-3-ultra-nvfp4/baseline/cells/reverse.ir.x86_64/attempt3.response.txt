export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l subl %n, 1
	jmp @loop

@loop
	%cond =w csltl %i, %j
	jnz %cond, @body, @end

@body
	%idx_i =l shll %i, 3
	%idx_j =l shll %j, 3
	%addr_i =l addl %a, %idx_i
	%addr_j =l addl %a, %idx_j
	%val_i =l loadl %addr_i
	%val_j =l loadl %addr_j
	storel %val_j, %addr_i
	storel %val_i, %addr_j
	%i =l addl %i, 1
	%j =l subl %j, 1
	jmp @loop

@end
	%zero =l copy 0
	ret %zero
}
