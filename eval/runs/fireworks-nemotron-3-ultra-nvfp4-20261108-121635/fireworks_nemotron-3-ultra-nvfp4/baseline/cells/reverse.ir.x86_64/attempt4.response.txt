export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	jmp @loop

@loop
	%cond =w cslt %i, %j
	jnz %cond, @body, @end

@body
	%idx_i =l shl %i, 3
	%idx_j =l shl %j, 3
	%addr_i =l add %a, %idx_i
	%addr_j =l add %a, %idx_j
	%val_i =l load %addr_i
	%val_j =l load %addr_j
	store %val_j, %addr_i
	store %val_i, %addr_j
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop

@end
	%zero =l copy 0
	ret %zero
}
