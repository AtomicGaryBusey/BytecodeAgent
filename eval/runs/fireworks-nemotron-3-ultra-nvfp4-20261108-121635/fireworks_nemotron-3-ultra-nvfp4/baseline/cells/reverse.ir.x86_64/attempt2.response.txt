export function l $reverse(l %a, l %n) {
@start
	%i = copy 0l
	%j = subl %n, 1l
	jmp @loop

@loop
	%cond = csltl %i, %j
	jnz %cond, @body, @end

@body
	%idx_i = shll %i, 3
	%idx_j = shll %j, 3
	%addr_i = addl %a, %idx_i
	%addr_j = addl %a, %idx_j
	%val_i = loadl %addr_i
	%val_j = loadl %addr_j
	storel %val_j, %addr_i
	storel %val_i, %addr_j
	%i = addl %i, 1
	%j = subl %j, 1
	jmp @loop

@end
	ret 0l
}
