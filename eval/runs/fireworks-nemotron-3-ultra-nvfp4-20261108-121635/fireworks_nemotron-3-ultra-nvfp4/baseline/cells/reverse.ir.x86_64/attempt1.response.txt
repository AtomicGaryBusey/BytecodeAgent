export function l $reverse(l %a, l %n) {
@start
	%i = copy 0
	%j = sub %n, 1
	jmp @loop

@loop
	%cond = csltl %i, %j
	jnz %cond, @body, @end

@body
	%idx_i = shl %i, 3
	%idx_j = shl %j, 3
	%addr_i = add %a, %idx_i
	%addr_j = add %a, %idx_j
	%val_i = loadl %addr_i
	%val_j = loadl %addr_j
	storel %val_j, %addr_i
	storel %val_i, %addr_j
	%i = add %i, 1
	%j = sub %j, 1
	jmp @loop

@end
	ret 0
}
