export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%eq_n = ceql %n, 0
	jnz %eq_n, @exit, @check_overlap

@check_overlap
	%cmp = cule %dst, %src
	jnz %cmp, @forward_init, @backward_init

@forward_init
	%src_cur = %src
	%dst_cur = %dst
	%cnt = %n
	jmp @forward_loop

@forward_loop
	%cond = cne %cnt, 0
	jnz %cond, @forward_body, @exit

@forward_body
	%val = loadl %src_cur
	storel %val, %dst_cur
	%src_cur = add %src_cur, 8
	%dst_cur = add %dst_cur, 8
	%cnt = sub %cnt, 1
	jmp @forward_loop

@backward_init
	%byte_n = mul %n, 8
	%src_cur = add %src, %byte_n
	%dst_cur = add %dst, %byte_n
	%cnt = %n
	jmp @backward_loop

@backward_loop
	%cond = cne %cnt, 0
	jnz %cond, @backward_body, @exit

@backward_body
	%src_cur = sub %src_cur, 8
	%dst_cur = sub %dst_cur, 8
	%val = loadl %src_cur
	storel %val, %dst_cur
	%cnt = sub %cnt, 1
	jmp @backward_loop

@exit
	ret 0
}
