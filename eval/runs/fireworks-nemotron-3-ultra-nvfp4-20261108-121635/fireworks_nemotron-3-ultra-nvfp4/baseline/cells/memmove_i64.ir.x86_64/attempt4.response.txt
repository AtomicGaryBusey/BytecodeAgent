export function l memmove_i64(l %dst, l %src, l %n) {
@start
	%eq_n = ceql %n, 0
	jnz %eq_n, @exit, @check_overlap

@check_overlap
	%cmp = cult %dst, %src
	jnz %cmp, @forward_init, @backward_init

@forward_init
	%src_cur_init = %src
	%dst_cur_init = %dst
	%cnt_init = %n
	jmp @forward_loop

@forward_loop
	%src_cur = phi @forward_init %src_cur_init, @forward_body %src_cur_next
	%dst_cur = phi @forward_init %dst_cur_init, @forward_body %dst_cur_next
	%cnt = phi @forward_init %cnt_init, @forward_body %cnt_next
	%cond = cnel %cnt, 0
	jnz %cond, @forward_body, @exit

@forward_body
	%val = loadl %src_cur
	storel %val, %dst_cur
	%src_cur_next = addl %src_cur, 8
	%dst_cur_next = addl %dst_cur, 8
	%cnt_next = subl %cnt, 1
	jmp @forward_loop

@backward_init
	%byte_n = mull %n, 8
	%src_end = addl %src, %byte_n
	%dst_end = addl %dst, %byte_n
	%cnt_init = %n
	jmp @backward_loop

@backward_loop
	%src_cur = phi @backward_init %src_end, @backward_body %src_cur_next
	%dst_cur = phi @backward_init %dst_end, @backward_body %dst_cur_next
	%cnt = phi @backward_init %cnt_init, @backward_body %cnt_next
	%cond = cnel %cnt, 0
	jnz %cond, @backward_body, @exit

@backward_body
	%src_cur_next = subl %src_cur, 8
	%dst_cur_next = subl %dst_cur, 8
	%val = loadl %src_cur_next
	storel %val, %dst_cur_next
	%cnt_next = subl %cnt, 1
	jmp @backward_loop

@exit
	ret 0
}
