export function l $mulhi_u64(l %a, l %b) {
@start
	%a_lo = andl %a, 0xFFFFFFFF
	%a_hi = shrl %a, 32
	%b_lo = andl %b, 0xFFFFFFFF
	%b_hi = shrl %b, 32

	%p0 = mull %a_lo, %b_lo
	%p1 = mull %a_lo, %b_hi
	%p2 = mull %a_hi, %b_lo
	%p3 = mull %a_hi, %b_hi

	%mid = addl %p1, %p2
	%carry1_w = cult %mid, %p1
	%carry1 = zextw %carry1_w

	%t = shll %mid, 32
	%low = addl %p0, %t
	%carry2_w = cult %low, %p0
	%carry2 = zextw %carry2_w

	%mid_hi = shrl %mid, 32
	%carry1_shift = shll %carry1, 32
	%high = addl %p3, %mid_hi
	%high = addl %high, %carry2
	%high = addl %high, %carry1_shift

	ret %high
}
