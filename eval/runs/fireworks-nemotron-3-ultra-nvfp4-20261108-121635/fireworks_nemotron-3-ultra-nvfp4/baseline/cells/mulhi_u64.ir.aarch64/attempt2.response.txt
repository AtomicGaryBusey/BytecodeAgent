export function l $mulhi_u64(l %a, l %b) {
@start
	%mask = copy l 0xFFFFFFFF
	%a_lo = and l %a, %mask
	%a_hi = shr l %a, 32
	%b_lo = and l %b, %mask
	%b_hi = shr l %b, 32

	%p0 = mul l %a_lo, %b_lo
	%p1 = mul l %a_lo, %b_hi
	%p2 = mul l %a_hi, %b_lo
	%p3 = mul l %a_hi, %b_hi

	%mid = add l %p1, %p2
	%carry1_w = cult l %mid, %p1
	%carry1 = ext l %carry1_w

	%t = shl l %mid, 32
	%low = add l %p0, %t
	%carry2_w = cult l %low, %p0
	%carry2 = ext l %carry2_w

	%mid_hi = shr l %mid, 32
	%carry1_shift = shl l %carry1, 32
	%high = add l %p3, %mid_hi
	%high = add l %high, %carry2
	%high = add l %high, %carry1_shift

	ret l %high
}
