binary_search:
    test rsi, rsi
    jz .Lnotfound
    xor rcx, rcx            # low = 0
    mov r8, rsi             # high = n
.Lloop:
    cmp rcx, r8
    jge .Lnotfound          # low >= high -> not found
    mov r9, r8
    sub r9, rcx
    shr r9, 1
    add r9, rcx             # mid = low + (high - low) / 2
    mov r10, [rdi + r9*8]   # load a[mid]
    cmp r10, rdx
    je .Lfound
    jl .Lless
    mov r8, r9              # high = mid
    jmp .Lloop
.Lless:
    lea rcx, [r9 + 1]       # low = mid + 1
    jmp .Lloop
.Lfound:
    mov rax, r9
    ret
.Lnotfound:
    mov rax, -1
    ret
