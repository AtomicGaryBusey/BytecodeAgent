export function l $map_into(l %dst, l %src, l %n, l %f) {
    @start
    %count =l copy 0
    %i =l copy 0
    %offset =l copy 0
    jmp @loop_head

    @loop_head
    %cond =w csltl %i, %n
    jnz %cond, @loop_body, @loop_end

    @loop_body
    %src_ptr =l add %src, %offset
    %val =l loadl %src_ptr
    %res =l call %f(l %val)
    %dst_ptr =l add %dst, %offset
    storel %res, %dst_ptr
    %lt =w csltl %res, %val
    %lt_l =l extwl %lt
    %count =l add %count, %lt_l
    %offset =l add %offset, 8
    %i =l add %i, 1
    jmp @loop_head

    @loop_end
    ret %count
}
