vm_run:
    xor r8, r8
    xor r9, r9
.loop:
    cmp r8, rsi
    je .done
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, 0
    je .op_halt
    cmp rax, 1
    je .op_push
    cmp rax, 2
    je .op_add
    cmp rax, 3
    je .op_sub
    cmp rax, 4
    je .op_mul
    cmp rax, 5
    je .op_dup
    cmp rax, 6
    je .op_swap
    cmp rax, 7
    je .op_drop
    cmp rax, 8
    je .op_over
    cmp rax, 9
    je .op_lt
    cmp rax, 10
    je .op_jz
    cmp rax, 11
    je .op_jmp
    jmp .done

.op_halt:
    jmp .done

.op_push:
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 2
    jmp .loop

.op_add:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    dec r9
    mov r11, qword ptr [rdx + r9*8]
    add r11, r10
    mov qword ptr [rdx + r9*8], r11
    inc r9
    add r8, 1
    jmp .loop

.op_sub:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    dec r9
    mov r11, qword ptr [rdx + r9*8]
    sub r11, r10
    mov qword ptr [rdx + r9*8], r11
    inc r9
    add r8, 1
    jmp .loop

.op_mul:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    dec r9
    mov r11, qword ptr [rdx + r9*8]
    imul r11, r10
    mov qword ptr [rdx + r9*8], r11
    inc r9
    add r8, 1
    jmp .loop

.op_dup:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 1
    jmp .loop

.op_swap:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 8], r11
    mov qword ptr [rdx + r9*8 - 16], r10
    add r8, 1
    jmp .loop

.op_drop:
    dec r9
    add r8, 1
    jmp .loop

.op_over:
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 1
    jmp .loop

.op_lt:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    dec r9
    mov r11, qword ptr [rdx + r9*8]
    cmp r11, r10
    setl r10b
    movzx r10, r10b
    mov qword ptr [rdx + r9*8], r10
    inc r9
    add r8, 1
    jmp .loop

.op_jz:
    dec r9
    mov r10, qword ptr [rdx + r9*8]
    test r10, r10
    jz .jz_taken
    add r8, 2
    jmp .loop
.jz_taken:
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov r8, r10
    jmp .loop

.op_jmp:
    mov r10, qword ptr [rdi + r8*8 + 8]
    mov r8, r10
    jmp .loop

.done:
    test r9, r9
    jz .return_zero
    mov rax, qword ptr [rdx + r9*8 - 8]
    jmp .zero_rest
.return_zero:
    xor rax, rax
.zero_rest:
    mov r10, r9
.zero_loop:
    cmp r10, rcx
    jge .exit
    mov qword ptr [rdx + r10*8], 0
    inc r10
    jmp .zero_loop
.exit:
    ret
