export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%c8_32 = copy 8
	%c8 = extswl %c8_32
	%c3_32 = copy 3
	%c3 = extswl %c3_32
	%c1_32 = copy 1
	%c1 = extswl %c1_32
	%c0_32 = copy 0
	%c0 = extswl %c0_32
	%n8 = shll %n, %c3
	%src_end = addl %src, %n8
	%cond1 = csgtl %dst, %src
	%cond2 = csltl %dst, %src_end
	%cond = andw %cond1, %cond2
	jnz %cond, @backward_init, @forward_init

@forward_init
	%src_ptr = copy %src
	%dst_ptr = copy %dst
	%i = copy %c0
	jmp @forward_loop

@forward_loop
	%cond_f = csltl %i, %n
	jnz %cond_f, @forward_body, @exit

@forward_body
	%val = loadl %src_ptr
	storel %val, %dst_ptr
	%src_ptr = addl %src_ptr, %c8
	%dst_ptr = addl %dst_ptr, %c8
	%i = addl %i, %c1
	jmp @forward_loop

@backward_init
	%src_ptr = addl %src, %n8
	%dst_ptr = addl %dst, %n8
	%i = copy %n
	jmp @backward_loop

@backward_loop
	%cond_b = csgtl %i, %c0
	jnz %cond_b, @backward_body, @exit

@backward_body
	%src_ptr = subl %src_ptr, %c8
	%dst_ptr = subl %dst_ptr, %c8
	%val = loadl %src_ptr
	storel %val, %dst_ptr
	%i = subl %i, %c1
	jmp @backward_loop

@exit
	ret %c0
}
