export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%c8 = copy 8l
	%c1 = copy 1l
	%c0 = copy 0l
	%cm1 = copy -1l
	%n8 = mull %n, %c8
	%src_end = addl %src, %n8
	%cond1 = csgtl %dst, %src
	%cond2 = csltl %dst, %src_end
	%cond = andw %cond1, %cond2
	jnz %cond, @backward_init, @forward_init

@forward_init
	jmp @forward_loop

@forward_loop
	%i = phi @forward_init %c0, @forward_body %i_next
	%cond_f = csltl %i, %n
	jnz %cond_f, @forward_body, @exit

@forward_body
	%tmp = mull %i, %c8
	%src_ptr = addl %src, %tmp
	%dst_ptr = addl %dst, %tmp
	%val = loadl %src_ptr
	storel %val, %dst_ptr
	%i_next = addl %i, %c1
	jmp @forward_loop

@backward_init
	%n_minus_1 = subl %n, %c1
	jmp @backward_loop

@backward_loop
	%i = phi @backward_init %n_minus_1, @backward_body %i_next
	%cond_b = csgtl %i, %cm1
	jnz %cond_b, @backward_body, @exit

@backward_body
	%tmp = mull %i, %c8
	%src_ptr = addl %src, %tmp
	%dst_ptr = addl %dst, %tmp
	%val = loadl %src_ptr
	storel %val, %dst_ptr
	%i_next = subl %i, %c1
	jmp @backward_loop

@exit
	ret %c0
}
