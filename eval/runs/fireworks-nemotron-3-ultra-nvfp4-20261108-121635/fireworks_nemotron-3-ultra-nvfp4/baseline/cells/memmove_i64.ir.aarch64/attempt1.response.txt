export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%n8 = mul %n, 8
	%src_end = add %src, %n8
	%cond1 = csgtl %dst, %src
	%cond2 = csltl %dst, %src_end
	%cond = and %cond1, %cond2
	jnz %cond, @backward_init, @forward_init

@forward_init
	jmp @forward_loop

@forward_loop
	%i = phi @forward_init 0, @forward_body %i_next
	%cond_f = csltl %i, %n
	jnz %cond_f, @forward_body, @exit

@forward_body
	%src_ptr = add %src, mul %i, 8
	%dst_ptr = add %dst, mul %i, 8
	%val = loadl %src_ptr
	storel %val, %dst_ptr
	%i_next = add %i, 1
	jmp @forward_loop

@backward_init
	%n_minus_1 = sub %n, 1
	jmp @backward_loop

@backward_loop
	%i = phi @backward_init %n_minus_1, @backward_body %i_next
	%cond_b = csgtl %i, -1
	jnz %cond_b, @backward_body, @exit

@backward_body
	%src_ptr = add %src, mul %i, 8
	%dst_ptr = add %dst, mul %i, 8
	%val = loadl %src_ptr
	storel %val, %dst_ptr
	%i_next = sub %i, 1
	jmp @backward_loop

@exit
	ret 0
}
