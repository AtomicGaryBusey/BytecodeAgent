export function l $bmax(l %a, l %n) {
@start
    %max0 = loadl %a
    %cnt = subl %n, 1
    %ptr = addl %a, 8
    jnz %cnt, @loop, @end

@loop
    %val = loadl %ptr
    %cmp = csgtnl %val, %max0
    %max1 = %max0
    %cnt_next = %cnt
    %ptr_next = %ptr
    jnz %cmp, @update, @latch

@update
    %max1 = %val
    %cnt_next = %cnt
    %ptr_next = %ptr
    jmp @latch

@latch
    %max0 = %max1
    %cnt = subl %cnt_next, 1
    %ptr = addl %ptr_next, 8
    jnz %cnt, @loop, @end

@end
    ret %max0
}
