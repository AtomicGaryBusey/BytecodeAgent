export function l $bmax(l %a, l %n) {
@start
    %max = loadl %a
    %cnt = subl %n, 1
    %ptr = addl %a, 8
    jnz %cnt, @loop, @end

@loop
    %elem = loadl %ptr
    %cmp = csgtnl %elem, %max
    jnz %cmp, @update, @continue

@update
    %max = %elem
    jmp @continue

@continue
    %cnt = subl %cnt, 1
    %ptr = addl %ptr, 8
    jnz %cnt, @loop, @end

@end
    ret %max
}
