export function l $bmax(l %a, l %n) {
@start
    %max = loadl %a
    %cnt = subl %n, 1
    %ptr = addl %a, 8
    %cond = csgtnl %cnt, 0
    jnz %cond, @loop, @end

@loop
    %val = loadl %ptr
    %cmp = csgtnl %val, %max
    %max1 = copy %max
    %cnt1 = copy %cnt
    %ptr1 = copy %ptr
    jnz %cmp, @update, @latch

@update
    %max1 = copy %val
    %cnt1 = copy %cnt
    %ptr1 = copy %ptr
    jmp @latch

@latch
    %max = copy %max1
    %cnt = subl %cnt1, 1
    %ptr = addl %ptr1, 8
    %cond = csgtnl %cnt, 0
    jnz %cond, @loop, @end

@end
    ret %max
}
