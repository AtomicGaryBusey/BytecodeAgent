export function l $bmax(l %a, l %n) {
@start
    %max = loadl %a
    %cnt = subl %n, 1
    %ptr = addl %a, 8
    %cond = csgtnl %cnt, 0
    jnz %cond, @loop %ptr, %max, %cnt, @end %max

@loop %ptr, %max, %cnt
    %val = loadl %ptr
    %cmp = csgtnl %val, %max
    %nextptr = addl %ptr, 8
    %nextcnt = subl %cnt, 1
    %ncond = csgtnl %nextcnt, 0
    jnz %cmp, @cont %nextptr, %val, %nextcnt, %ncond, @cont %nextptr, %max, %nextcnt, %ncond

@cont %ptr, %max, %cnt, %cond
    jnz %cond, @loop %ptr, %max, %cnt, @end %max

@end %max
    ret %max
}
