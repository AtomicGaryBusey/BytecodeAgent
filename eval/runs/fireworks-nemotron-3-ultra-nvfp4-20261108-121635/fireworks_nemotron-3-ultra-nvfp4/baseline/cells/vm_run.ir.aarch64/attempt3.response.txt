export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	%eight =l copy 8
	jmp @fetch

@fetch
	%pc_lt_len =w csltl %pc, %len
	jnz %pc_lt_len, @do_instr
	jmp @halt_loop

@do_instr
	%pc_off =l mull %pc, %eight
	%opcode =l loadl %code, %pc_off
	%opcode_eq_0 =w ceql %opcode, 0
	jnz %opcode_eq_0, @op_halt
	%opcode_eq_1 =w ceql %opcode, 1
	jnz %opcode_eq_1, @op_push
	%opcode_eq_2 =w ceql %opcode, 2
	jnz %opcode_eq_2, @op_add
	%opcode_eq_3 =w ceql %opcode, 3
	jnz %opcode_eq_3, @op_sub
	%opcode_eq_4 =w ceql %opcode, 4
	jnz %opcode_eq_4, @op_mul
	%opcode_eq_5 =w ceql %opcode, 5
	jnz %opcode_eq_5, @op_dup
	%opcode_eq_6 =w ceql %opcode, 6
	jnz %opcode_eq_6, @op_swap
	%opcode_eq_7 =w ceql %opcode, 7
	jnz %opcode_eq_7, @op_drop
	%opcode_eq_8 =w ceql %opcode, 8
	jnz %opcode_eq_8, @op_over
	%opcode_eq_9 =w ceql %opcode, 9
	jnz %opcode_eq_9, @op_lt
	%opcode_eq_10 =w ceql %opcode, 10
	jnz %opcode_eq_10, @op_jz
	%opcode_eq_11 =w ceql %opcode, 11
	jnz %opcode_eq_11, @op_jmp
	%pc =l addl %pc, 1
	jmp @fetch

@op_halt
	jmp @halt_loop

@op_push
	%pc_plus1 =l addl %pc, 1
	%op_off =l mull %pc_plus1, %eight
	%operand =l loadl %code, %op_off
	%sp_off =l mull %sp, %eight
	storel %operand, %stack, %sp_off
	%sp =l addl %sp, 1
	%pc =l addl %pc, 2
	jmp @fetch

@op_add
	%sp1 =l subl %sp, 1
	%off_b =l mull %sp1, %eight
	%b =l loadl %stack, %off_b
	%sp2 =l subl %sp, 2
	%off_a =l mull %sp2, %eight
	%a =l loadl %stack, %off_a
	%sum =l addl %a, %b
	storel %sum, %stack, %off_a
	%sp =l subl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_sub
	%sp1 =l subl %sp, 1
	%off_b =l mull %sp1, %eight
	%b =l loadl %stack, %off_b
	%sp2 =l subl %sp, 2
	%off_a =l mull %sp2, %eight
	%a =l loadl %stack, %off_a
	%diff =l subl %a, %b
	storel %diff, %stack, %off_a
	%sp =l subl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_mul
	%sp1 =l subl %sp, 1
	%off_b =l mull %sp1, %eight
	%b =l loadl %stack, %off_b
	%sp2 =l subl %sp, 2
	%off_a =l mull %sp2, %eight
	%a =l loadl %stack, %off_a
	%prod =l mull %a, %b
	storel %prod, %stack, %off_a
	%sp =l subl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_dup
	%sp1 =l subl %sp, 1
	%off_top =l mull %sp1, %eight
	%top =l loadl %stack, %off_top
	%sp_off =l mull %sp, %eight
	storel %top, %stack, %sp_off
	%sp =l addl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_swap
	%sp1 =l subl %sp, 1
	%off_top =l mull %sp1, %eight
	%top =l loadl %stack, %off_top
	%sp2 =l subl %sp, 2
	%off_second =l mull %sp2, %eight
	%second =l loadl %stack, %off_second
	storel %top, %stack, %off_second
	storel %second, %stack, %off_top
	%pc =l addl %pc, 1
	jmp @fetch

@op_drop
	%sp =l subl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_over
	%sp2 =l subl %sp, 2
	%off_second =l mull %sp2, %eight
	%second =l loadl %stack, %off_second
	%sp_off =l mull %sp, %eight
	storel %second, %stack, %sp_off
	%sp =l addl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_lt
	%sp1 =l subl %sp, 1
	%off_b =l mull %sp1, %eight
	%b =l loadl %stack, %off_b
	%sp2 =l subl %sp, 2
	%off_a =l mull %sp2, %eight
	%a =l loadl %stack, %off_a
	%cmp =w csltl %a, %b
	%result =l zextw %cmp
	storel %result, %stack, %off_a
	%sp =l subl %sp, 1
	%pc =l addl %pc, 1
	jmp @fetch

@op_jz
	%pc_plus1 =l addl %pc, 1
	%op_off =l mull %pc_plus1, %eight
	%target =l loadl %code, %op_off
	%sp1 =l subl %sp, 1
	%off_v =l mull %sp1, %eight
	%v =l loadl %stack, %off_v
	%sp =l subl %sp, 1
	%v_eq_0 =w ceql %v, 0
	jnz %v_eq_0, @jz_taken
	%pc =l addl %pc, 2
	jmp @fetch

@jz_taken
	%pc =l copy %target
	jmp @fetch

@op_jmp
	%pc_plus1 =l addl %pc, 1
	%op_off =l mull %pc_plus1, %eight
	%target =l loadl %code, %op_off
	%pc =l copy %target
	jmp @fetch

@halt_loop
	%sp_gt_0 =w csltl 0, %sp
	jnz %sp_gt_0, @has_result
	%result =l copy 0
	jmp @zero_stack

@has_result
	%sp1 =l subl %sp, 1
	%off_top =l mull %sp1, %eight
	%result =l loadl %stack, %off_top
	jmp @zero_stack

@zero_stack
	jmp @zero_loop_head

@zero_loop_head
	%cond =w csltl %sp, %cap
	jnz %cond, @zero_body
	jmp @return

@zero_body
	%off =l mull %sp, %eight
	storel 0, %stack, %off
	%sp =l addl %sp, 1
	jmp @zero_loop_head

@return
	ret %result
}
