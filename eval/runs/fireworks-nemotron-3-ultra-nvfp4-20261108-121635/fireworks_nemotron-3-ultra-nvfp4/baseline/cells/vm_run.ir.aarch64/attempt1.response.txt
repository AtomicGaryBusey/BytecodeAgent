export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc = copy 0
	%sp = copy 0
	%eight = copy 8
	jmp @fetch

@fetch
	%pc_ge_len = cgeltl %pc, %len
	jnz %pc_ge_len, @do_instr
	jmp @halt_loop

@do_instr
	%pc_off = mull %pc, %eight
	%opcode = loadl %code, %pc_off
	%opcode_eq_0 = ceql %opcode, 0
	jnz %opcode_eq_0, @op_halt
	%opcode_eq_1 = ceql %opcode, 1
	jnz %opcode_eq_1, @op_push
	%opcode_eq_2 = ceql %opcode, 2
	jnz %opcode_eq_2, @op_add
	%opcode_eq_3 = ceql %opcode, 3
	jnz %opcode_eq_3, @op_sub
	%opcode_eq_4 = ceql %opcode, 4
	jnz %opcode_eq_4, @op_mul
	%opcode_eq_5 = ceql %opcode, 5
	jnz %opcode_eq_5, @op_dup
	%opcode_eq_6 = ceql %opcode, 6
	jnz %opcode_eq_6, @op_swap
	%opcode_eq_7 = ceql %opcode, 7
	jnz %opcode_eq_7, @op_drop
	%opcode_eq_8 = ceql %opcode, 8
	jnz %opcode_eq_8, @op_over
	%opcode_eq_9 = ceql %opcode, 9
	jnz %opcode_eq_9, @op_lt
	%opcode_eq_10 = ceql %opcode, 10
	jnz %opcode_eq_10, @op_jz
	%opcode_eq_11 = ceql %opcode, 11
	jnz %opcode_eq_11, @op_jmp
	%pc = addl %pc, 1
	jmp @fetch

@op_halt
	jmp @halt_loop

@op_push
	%pc_plus1 = addl %pc, 1
	%op_off = mull %pc_plus1, %eight
	%operand = loadl %code, %op_off
	%sp_off = mull %sp, %eight
	storel %operand, %stack, %sp_off
	%sp = addl %sp, 1
	%pc = addl %pc, 2
	jmp @fetch

@op_add
	%sp1 = subl %sp, 1
	%off_b = mull %sp1, %eight
	%b = loadl %stack, %off_b
	%sp2 = subl %sp, 2
	%off_a = mull %sp2, %eight
	%a = loadl %stack, %off_a
	%sum = addl %a, %b
	storel %sum, %stack, %off_a
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_sub
	%sp1 = subl %sp, 1
	%off_b = mull %sp1, %eight
	%b = loadl %stack, %off_b
	%sp2 = subl %sp, 2
	%off_a = mull %sp2, %eight
	%a = loadl %stack, %off_a
	%diff = subl %a, %b
	storel %diff, %stack, %off_a
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_mul
	%sp1 = subl %sp, 1
	%off_b = mull %sp1, %eight
	%b = loadl %stack, %off_b
	%sp2 = subl %sp, 2
	%off_a = mull %sp2, %eight
	%a = loadl %stack, %off_a
	%prod = mull %a, %b
	storel %prod, %stack, %off_a
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_dup
	%sp1 = subl %sp, 1
	%off_top = mull %sp1, %eight
	%top = loadl %stack, %off_top
	%sp_off = mull %sp, %eight
	storel %top, %stack, %sp_off
	%sp = addl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_swap
	%sp1 = subl %sp, 1
	%off_top = mull %sp1, %eight
	%top = loadl %stack, %off_top
	%sp2 = subl %sp, 2
	%off_second = mull %sp2, %eight
	%second = loadl %stack, %off_second
	storel %top, %stack, %off_second
	storel %second, %stack, %off_top
	%pc = addl %pc, 1
	jmp @fetch

@op_drop
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_over
	%sp2 = subl %sp, 2
	%off_second = mull %sp2, %eight
	%second = loadl %stack, %off_second
	%sp_off = mull %sp, %eight
	storel %second, %stack, %sp_off
	%sp = addl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_lt
	%sp1 = subl %sp, 1
	%off_b = mull %sp1, %eight
	%b = loadl %stack, %off_b
	%sp2 = subl %sp, 2
	%off_a = mull %sp2, %eight
	%a = loadl %stack, %off_a
	%cmp = csltl %a, %b
	%result = zextw %cmp
	storel %result, %stack, %off_a
	%sp = subl %sp, 1
	%pc = addl %pc, 1
	jmp @fetch

@op_jz
	%pc_plus1 = addl %pc, 1
	%op_off = mull %pc_plus1, %eight
	%target = loadl %code, %op_off
	%sp1 = subl %sp, 1
	%off_v = mull %sp1, %eight
	%v = loadl %stack, %off_v
	%sp = subl %sp, 1
	%v_eq_0 = ceql %v, 0
	jnz %v_eq_0, @jz_taken
	%pc = addl %pc, 2
	jmp @fetch

@jz_taken
	%pc = copy %target
	jmp @fetch

@op_jmp
	%pc_plus1 = addl %pc, 1
	%op_off = mull %pc_plus1, %eight
	%target = loadl %code, %op_off
	%pc = copy %target
	jmp @fetch

@halt_loop
	%sp_gt_0 = cgltl 0, %sp
	jnz %sp_gt_0, @has_result
	%result = copy 0
	jmp @zero_stack

@has_result
	%sp1 = subl %sp, 1
	%off_top = mull %sp1, %eight
	%result = loadl %stack, %off_top
	jmp @zero_stack

@zero_stack
	jmp @zero_loop_head

@zero_loop_head
	%cond = cgeltl %sp, %cap
	jnz %cond, @zero_body
	jmp @return

@zero_body
	%off = mull %sp, %eight
	storel 0, %stack, %off
	%sp = addl %sp, 1
	jmp @zero_loop_head

@return
	ret %result
}
