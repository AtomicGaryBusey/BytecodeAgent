mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    mov rbx, rsi        # n
    mov rbp, rdx        # k
    mov r12, rdi        # a
    mov r13, rcx        # out
    mov r14, r8         # tmp

    mov rax, rbx
    imul rax, rbx
    mov r15, rax        # n_sq = n*n

    mov rax, rbx
    shl rax, 3
    mov [rsp], rax      # store n*8

    cld

    # copy a -> tmp_base
    mov rcx, r15
    mov rsi, r12
    mov rdi, r14
    rep movsq

    # zero out
    mov rcx, r15
    mov rdi, r13
    xor rax, rax
    rep stosq

    # identity diagonal
    xor r8, r8
.diag_loop:
    mov rax, r8
    imul rax, rbx
    add rax, r8
    mov qword ptr [r13 + rax*8], 1
    inc r8
    cmp r8, rbx
    jne .diag_loop

.pow_loop:
    test rbp, rbp
    jz .pow_end
    test rbp, 1
    jz .skip_mul1

    # multiply out * tmp_base -> tmp_prod
    mov rsi, r13
    mov rdi, r14
    lea r8, [r14 + r15*8]
    xor rcx, rcx
.mul1_i_loop:
    mov r11, rcx
    imul r11, rbx
    shl r11, 3
    add r11, rsi
    xor rdx, rdx
.mul1_j_loop:
    xor r10, r10
    xor r9, r9
.mul1_t_loop:
    mov rax, [r11 + r9*8]
    mov r12, r9
    imul r12, rbx
    shl r12, 3
    add r12, rdi
    imul rax, [r12 + rdx*8]
    add r10, rax
    inc r9
    cmp r9, rbx
    jne .mul1_t_loop
    mov rax, rcx
    imul rax, rbx
    add rax, rdx
    shl rax, 3
    mov [r8 + rax], r10
    inc rdx
    cmp rdx, rbx
    jne .mul1_j_loop
    inc rcx
    cmp rcx, rbx
    jne .mul1_i_loop

    # copy prod to out
    mov rcx, r15
    mov rsi, r8
    mov rdi, r13
    rep movsq

.skip_mul1:
    # square base: tmp_base * tmp_base -> tmp_prod
    mov rsi, r14
    mov rdi, r14
    lea r8, [r14 + r15*8]
    xor rcx, rcx
.mul2_i_loop:
    mov r11, rcx
    imul r11, rbx
    shl r11, 3
    add r11, rsi
    xor rdx, rdx
.mul2_j_loop:
    xor r10, r10
    xor r9, r9
.mul2_t_loop:
    mov rax, [r11 + r9*8]
    mov r12, r9
    imul r12, rbx
    shl r12, 3
    add r12, rdi
    imul rax, [r12 + rdx*8]
    add r10, rax
    inc r9
    cmp r9, rbx
    jne .mul2_t_loop
    mov rax, rcx
    imul rax, rbx
    add rax, rdx
    shl rax, 3
    mov [r8 + rax], r10
    inc rdx
    cmp rdx, rbx
    jne .mul2_j_loop
    inc rcx
    cmp rcx, rbx
    jne .mul2_i_loop

    # copy prod to base
    mov rcx, r15
    mov rsi, r8
    mov rdi, r14
    rep movsq

    shr rbp, 1
    jmp .pow_loop

.pow_end:
    # zero out tmp (2*n*n elements)
    mov rcx, r15
    shl rcx, 1
    mov rdi, r14
    xor rax, rax
    rep stosq

    xor rax, rax
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
