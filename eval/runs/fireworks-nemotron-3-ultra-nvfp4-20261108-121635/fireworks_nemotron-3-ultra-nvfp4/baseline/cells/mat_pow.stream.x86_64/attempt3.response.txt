mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8

    mov rbx, rsi        # n
    mov rbp, rdx        # k
    mov r12, rdi        # a
    mov r13, rcx        # out
    mov r14, r8         # tmp

    mov rax, rbx
    imul rax, rbx
    mov r15, rax        # n_sq = n*n

    mov rax, rbx
    shl rax, 3
    mov [rsp], rax      # store n*8

    cld

    # copy a -> tmp_base
    mov rcx, r15
    mov rsi, r12
    mov rdi, r14
    rep movsq

    # zero out
    mov rcx, r15
    mov rdi, r13
    xor rax, rax
    rep stosq

    # identity diagonal
    xor r8, r8
.diag_loop:
    mov rax, r8
    imul rax, rbx
    add rax, r8
    mov qword ptr [r13 + rax*8], 1
    inc r8
    cmp r8, rbx
    jne .diag_loop

.pow_loop:
    test rbp, rbp
    jz .pow_end
    test rbp, 1
    jz .skip_mul1

    # multiply out * tmp_base -> tmp_prod
    mov rsi, r13
    mov rdi, r14
    lea r8, [r14 + r15*8]
    xor rcx, rcx
.mul1_i_loop:
    mov r10, rcx
    imul r10, rbx
    shl r10, 3
    xor rdx, rdx
.mul1_j_loop:
    lea r11, [rdi + rdx*8]
    xor rax, rax
    xor r9, r9
.mul1_t_loop:
    mov r12, [rsi + r10 + r9*8]
