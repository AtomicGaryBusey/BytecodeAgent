.intel_syntax noprefix
mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15

    mov rbx, rsi        # n
    mov rbp, rdx        # k
    mov r12, rdi        # a
    mov r13, rcx        # out
    mov r14, r8         # tmp

    mov rax, rbx
    imul rax, rbx
    mov r15, rax        # n_sq = n*n

    cld

    # copy a -> tmp_base
    mov rcx, r15
    mov rsi, r12
    mov rdi, r14
    rep movsq

    # zero out
    mov rcx, r15
    mov rdi, r13
    xor rax, rax
    rep stosq

    # identity diagonal
    xor r8, r8
.Ldiag_loop:
    mov rax, r8
    imul rax, rbx
    add rax, r8
    mov qword ptr [r13 + rax*8], 1
    inc r8
    cmp r8, rbx
    jne .Ldiag_loop

.Lpow_loop:
    test rbp, rbp
    jz .Lpow_end
    test rbp, 1
    jz .Lskip_mul

    # multiply out * tmp_base -> tmp_prod
    mov rdi, r13
    mov rsi, r14
    lea rdx, [r14 + r15*8]
    lea rax, [.Lafter_mul1]
