export function l $fib(l %n) {
    @start
    %zero = l 0
    %one = l 1
    %two = l 2
    %n0 =w ceql %n, %zero
    jnz %n0, @return0
    %n1 =w ceql %n, %one
    jnz %n1, @return1

    %a = %zero
    %b = %one
    %i = %two

    @loop
    %cond =w cslel %i, %n
    jnz %cond, @loop_body
    jmp @return_b

    @loop_body
    %c = addl %a, %b
    %a = %b
    %b = %c
    %i = addl %i, %one
    jmp @loop

    @return0
    ret %zero

    @return1
    ret %one

    @return_b
    ret %b
}
