export function l $fib(l %n) {
    @start
    %n0 = ceql %n, l 0
    jnz %n0, @return0
    %n1 = ceql %n, l 1
    jnz %n1, @return1

    %a = copy l 0
    %b = copy l 1
    %i = copy l 2

    @loop
    %cond = cslel %i, %n
    jnz %cond, @loop_body
    jmp @return_b

    @loop_body
    %c = addl %a, %b
    %a = copy %b
    %b = copy %c
    %i = addl %i, l 1
    jmp @loop

    @return0
    ret l 0

    @return1
    ret l 1

    @return_b
    ret %b
}
