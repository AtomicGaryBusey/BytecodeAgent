export function l $fib(l %n) {
    @start
    %n0 =w ceql %n, l 0
    jnz %n0, @return0
    %n1 =w ceql %n, l 1
    jnz %n1, @return1

    %a =l copy l 0
    %b =l copy l 1
    %i =l copy l 2

    @loop
    %cond =w cslel %i, %n
    jnz %cond, @loop_body
    jmp @return_b

    @loop_body
    %c =l addl %a, %b
    %a =l copy %b
    %b =l copy %c
    %i =l addl %i, l 1
    jmp @loop

    @return0
    ret l 0

    @return1
    ret l 1

    @return_b
    ret %b
}
