mp_mul:
    push r12
    push r13
    push r14
    mov r9, rsi            # r9 = na
    mov r10, rcx           # r10 = nb
    mov r11, rdx           # r11 = b
    # rdi = a, r8 = r

    # Zero result array r[0..na+nb-1]
    mov rsi, r9
    add rsi, r10           # rsi = total limbs
    xor rax, rax
    test rsi, rsi
    jz .mul_start
    xor rdx, rdx
.zero_loop:
    mov qword ptr [r8 + rdx*8], rax
    inc rdx
    cmp rdx, rsi
    jne .zero_loop

.mul_start:
    xor rsi, rsi           # i = 0
.outer_loop:
    cmp rsi, r9
    jge .outer_done
    mov rdx, qword ptr [rdi + rsi*8]   # a[i]
    xor r12, r12           # carry = 0
    xor rcx, rcx           # j = 0
    lea r14, [r8 + rsi*8]  # base address for this row
.inner_loop:
    cmp rcx, r10
    jge .inner_done
    mulx r13, rax, qword ptr [r11 + rcx*8]   # rdx * b[j] -> r13:rax
    add rax, qword ptr [r14 + rcx*8]
    adc r13, 0
    add rax, r12
    adc r13, 0
    mov qword ptr [r14 + rcx*8], rax
    mov r12, r13
    inc rcx
    jmp .inner_loop
.inner_done:
    mov qword ptr [r14 + r10*8], r12
    inc rsi
    jmp .outer_loop
.outer_done:

    # Find number of significant limbs
    mov rsi, r9
    add rsi, r10           # total
    test rsi, rsi
    jz .return_zero
    lea rcx, [r8 + rsi*8 - 8]
.scan_loop:
    mov rax, qword ptr [rcx]
    test rax, rax
    jne .found_nonzero
    sub rcx, 8
    dec rsi
    jnz .scan_loop
.return_zero:
    xor rax, rax
    jmp .return
.found_nonzero:
    mov rax, rsi
.return:
    pop r14
    pop r13
    pop r12
    ret
