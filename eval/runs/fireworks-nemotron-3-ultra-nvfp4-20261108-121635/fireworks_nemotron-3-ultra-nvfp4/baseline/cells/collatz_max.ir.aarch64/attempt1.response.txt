export function l $collatz_max(l %n) {
@start
    %max_steps = copy 0
    %k = copy 1
    jmp @outer_head

@outer_head
    %k_phi = phi l [@start: %k, @outer_end: %k_next]
    %max_phi = phi l [@start: %max_steps, @outer_end: %max_next]
    %cond = sult %k_phi %n
    jnz %cond, @outer_body, @return

@outer_body
    %x = copy %k_phi
    %steps = copy 0
    jmp @inner_head

@inner_head
    %x_phi = phi l [@outer_body: %x, @inner_body: %x_next]
    %steps_phi = phi l [@outer_body: %steps, @inner_body: %steps_next]
    %eq1 = ceq %x_phi 1
    jnz %eq1, @inner_end, @inner_body

@inner_body
    %is_even = and %x_phi 1
    %is_even = ceq %is_even 0
    %x_even = shr %x_phi 1
    %x_odd = add %x_phi %x_phi
    %x_odd = add %x_odd %x_phi
    %x_odd = add %x_odd 1
    %x_next = csel %is_even %x_even %x_odd
    %steps_next = add %steps_phi 1
    jmp @inner_head

@inner_end
    %cond_max = sgt %steps_phi %max_phi
    %max_next = csel %cond_max %steps_phi %max_phi
    %k_next = add %k_phi 1
    jmp @outer_end

@outer_end
    jmp @outer_head

@return
    ret %max_phi
}
