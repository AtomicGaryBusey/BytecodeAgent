export function l $collatz_max(l %n) {
@start
    %max_steps =l copy 0
    %k =l copy 1
    jmp @outer_head

@outer_head
    %k_phi =l phi @start %k, @outer_end %k_next
    %max_phi =l phi @start %max_steps, @outer_end %max_next
    %cond =w cule %k_phi %n
    jnz %cond, @outer_body, @return

@outer_body
    %x =l copy %k_phi
    %steps =l copy 0
    jmp @inner_head

@inner_head
    %x_phi =l phi @outer_body %x, @inner_body %x_next
    %steps_phi =l phi @outer_body %steps, @inner_body %steps_next
    %eq1 =w ceq %x_phi 1
    jnz %eq1, @inner_end, @inner_body

@inner_body
    %is_even1 =l andl %x_phi 1
    %is_even =w ceq %is_even1 0
    %x_even =l shrl %x_phi 1
    %x_odd1 =l addl %x_phi %x_phi
    %x_odd2 =l addl %x_odd1 %x_phi
    %x_odd3 =l addl %x_odd2 1
    %x_next =l csel %is_even %x_even %x_odd3
    %steps_next =l addl %steps_phi 1
    jmp @inner_head

@inner_end
    %cond_max =w cugt %steps_phi %max_phi
    %max_next =l csel %cond_max %steps_phi %max_phi
    %k_next =l addl %k_phi 1
    jmp @outer_end

@outer_end
    jmp @outer_head

@return
    ret %max_phi
}
