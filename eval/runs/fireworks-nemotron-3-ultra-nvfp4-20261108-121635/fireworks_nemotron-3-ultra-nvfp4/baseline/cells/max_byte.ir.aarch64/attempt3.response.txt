export function l $max_byte(l %p, l %n) {
@start
    %nz =l ceql %n, 0
    jnz %nz, @end_zero, @loop

@loop
    %max =l copy 0
    %i =l copy 0
    jmp @loop_head

@loop_head
    %cond =l csltl %i, %n
    jnz %cond, @loop_body, @loop_end

@loop_body
    %byte =l loadub %p
    %p =l add %p, 1
    %cmp =w cugtl %byte, %max
    %max =l select %cmp, %byte, %max
    %i =l add %i, 1
    jmp @loop_head

@loop_end
    ret %max

@end_zero
    ret 0
}
