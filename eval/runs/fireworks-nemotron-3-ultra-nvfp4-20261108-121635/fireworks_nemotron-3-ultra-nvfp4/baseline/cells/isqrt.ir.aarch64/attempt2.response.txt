export function l $isqrt(l %n) {
@start
    %low = addl 0, 0
    %high = addl 4294967296, 0
    jmp @loop

@loop
    %cond = csltul %low, %high
    jnz %cond, @body, @exit

@body
    %mid = addl %low, %high
    %mid = shru %mid, 1
    %mid2 = mull %mid, %mid
    %cmp = csleul %mid2, %n
    jnz %cmp, @if_le, @if_gt

@if_le
    %low = addl %mid, 1
    jmp @loop

@if_gt
    %high = copy %mid
    jmp @loop

@exit
    %res = subl %low, 1
    ret %res
}
