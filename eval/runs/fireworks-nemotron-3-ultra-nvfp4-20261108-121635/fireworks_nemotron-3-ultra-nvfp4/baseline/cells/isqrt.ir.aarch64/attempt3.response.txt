export function l $isqrt(l %n) {
@start
    %low = add 0, 0
    %high = add 4294967296, 0
    jmp @loop

@loop
    %cond = cslt %low, %high
    jnz %cond, @body, @exit

@body
    %mid = add %low, %high
    %mid = shr %mid, 1
    %mid2 = mul %mid, %mid
    %cmp = csle %mid2, %n
    jnz %cmp, @if_le, @if_gt

@if_le
    %low = add %mid, 1
    jmp @loop

@if_gt
    %high = mov %mid
    jmp @loop

@exit
    %res = sub %low, 1
    ret %res
}
