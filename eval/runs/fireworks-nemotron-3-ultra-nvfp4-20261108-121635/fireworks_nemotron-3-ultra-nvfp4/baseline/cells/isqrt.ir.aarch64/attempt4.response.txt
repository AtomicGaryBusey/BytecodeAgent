export function l $isqrt(l %n) {
@start
    %low = copy 0
    %high = copy 4294967296
    jmp @loop

@loop
    %cmp = cultl %low, %high
    jnz %cmp, @body, @exit

@body
    %mid = addl %low, %high
    %mid = shrl %mid, 1
    %mid2 = mull %mid, %mid
    %cmp2 = culel %mid2, %n
    jnz %cmp2, @if_le, @if_gt

@if_le
    %low = addl %mid, 1
    jmp @loop

@if_gt
    %high = copy %mid
    jmp @loop

@exit
    %res = subl %low, 1
    ret %res
}
