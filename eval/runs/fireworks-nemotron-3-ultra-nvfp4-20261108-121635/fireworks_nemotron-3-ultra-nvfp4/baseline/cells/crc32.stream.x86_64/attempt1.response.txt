crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jz .Ldone
    mov rcx, rsi
.Lbyte_loop:
    movzx edx, byte ptr [rdi]
    inc rdi
    xor eax, edx
    mov r8d, 8
.Lbit_loop:
    test eax, 1
    jz .Lshift_only
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lnext_bit
.Lshift_only:
    shr eax, 1
.Lnext_bit:
    dec r8d
    jnz .Lbit_loop
    dec rcx
    jnz .Lbyte_loop
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
