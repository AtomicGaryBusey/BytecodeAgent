export function l $fill_down(l %a, l %n, l %start) {
@start
	%i0 = copy 0
	jmp @head

@head
	%i = phi l [%i0, @start], [%i1, @body]
	%cond = csl %i, %n
	jnz %cond, @body, @end

@body
	%val = sub %start, %i
	%off = shl %i, 3
	%addr = add %a, %off
	store %val, %addr
	%i1 = add %i, 1
	jmp @head

@end
	ret %n
}
