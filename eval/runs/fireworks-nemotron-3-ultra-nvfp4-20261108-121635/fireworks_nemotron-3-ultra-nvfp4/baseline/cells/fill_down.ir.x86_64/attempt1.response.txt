export function l $fill_down(l %a, l %n, l %start) {
@start
	%nz = ceql %n, 0
	jnz %nz, @end, @loop_init

@loop_init
	%i0 = copy 0
	jmp @loop_head

@loop_head
	%i = phi l [%i0, @loop_init], [%i1, @loop_body]
	%cond = csltl %i, %n
	jnz %cond, @loop_body, @end

@loop_body
	%val = subl %start, %i
	%off = shll %i, 3
	%addr = addl %a, %off
	storel %val, %addr
	%i1 = addl %i, 1
	jmp @loop_head

@end
	ret %n
}
