export function l $fill_down(l %a, l %n, l %start) {
@start
	%i0 = subl %n, %n
	jmp @head

@head
	%i = phi l [%i0, @start], [%i1, @body]
	%cond = csltl %i, %n
	jnz %cond, @body, @end

@body
	%val = subl %start, %i
	%off = shll %i, 3
	%addr = addl %a, %off
	storel %val, %addr
	%i1 = addl %i, 1
	jmp @head

@end
	ret %n
}
