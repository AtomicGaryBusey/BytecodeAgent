mp_mul:
    add x5, x1, x3                // x5 = na + nb (total limbs)
    cbz x5, .Lret_zero            // if total == 0, return 0
    mov x6, x4                    // x6 = r pointer for zeroing
    mov x7, xzr                   // zero value
.Lzero_loop:
    str x7, [x6], #8              // zero r[i]
    subs x5, x5, #1
    b.ne .Lzero_loop
    cbz x1, .Lret_zero            // if na == 0, return 0
    cbz x3, .Lret_zero            // if nb == 0, return 0
    mov x8, xzr                   // i = 0
.Louter:
    ldr x9, [x0, x8, lsl #3]      // x9 = a[i]
    mov x14, xzr                  // carry = 0
    mov x10, xzr                  // j = 0
    add x15, x4, x8, lsl #3       // x15 = r + i*8
.Linner:
    ldr x11, [x2, x10, lsl #3]    // x11 = b[j]
    ldr x17, [x15, x10, lsl #3]   // x17 = r[i+j]
    mul x12, x9, x11              // x12 = low 64 bits of a[i]*b[j]
    umulh x13, x9, x11            // x13 = high 64 bits of a[i]*b[j]
    adds x12, x12, x17            // x12 += r[i+j], sets carry
    adc x13, x13, xzr             // x13 += carry
    adds x12, x12, x14            // x12 += carry, sets carry
    adc x13, x13, xzr             // x13 += carry
    str x12, [x15, x10, lsl #3]   // store result
    mov x14, x13                  // carry = x13
    add x10, x10, #1              // j++
    cmp x10, x3
    b.ne .Linner
    str x14, [x15, x3, lsl #3]    // store final carry at r[i+nb]
    add x8, x8, #1                // i++
    cmp x8, x1
    b.ne .Louter
    // find most significant non-zero limb
    add x5, x1, x3                // x5 = total
    sub x5, x5, #1                // index = total-1
    cbz x5, .Lcheck_last          // if total==1, check r[0]
.Ltrim_loop:
    ldr x6, [x4, x5, lsl #3]      // load r[index]
    cbnz x6, .Lfound
    subs x5, x5, #1
    b.ge .Ltrim_loop
    mov x0, xzr                   // all zero
    ret
.Lfound:
    add x0, x5, #1                // return index+1
    ret
.Lcheck_last:
    ldr x6, [x4]                  // load r[0]
    cbnz x6, .Lret_one
    mov x0, xzr
    ret
.Lret_one:
    mov x0, #1
    ret
.Lret_zero:
    mov x0, xzr
    ret
