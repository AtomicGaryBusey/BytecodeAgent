mp_mul:
    add x5, x1, x3                // total = na + nb
    cbz x5, .Lret_zero            // if total == 0, return 0
    mov x6, x4                    // x6 = r (pointer for zeroing)
    mov x7, xzr                   // x7 = 0
.Lzero_loop:
    str x7, [x6], #8              // zero r[i]
    subs x5, x5, #1
    b.ne .Lzero_loop
    cbz x1, .Lret_zero            // if na == 0, return 0
    cbz x3, .Lret_zero            // if nb == 0, return 0
    mov x5, #0                    // i = 0
.Louter:
    add x10, x4, x5, lsl #3       // base_i = r + i*8
    ldr x8, [x0, x5, lsl #3]      // a_i = a[i]
    mov x7, #0                    // carry = 0
    mov x6, #0                    // j = 0
.Linner:
    ldr x9, [x2, x6, lsl #3]      // b_j = b[j]
    ldr x11, [x10, x6, lsl #3]    // r_ij = r[i+j]
    umaddl x12, x13, x8, x9, x11  // (x12, x13) = a_i * b_j + r_ij
