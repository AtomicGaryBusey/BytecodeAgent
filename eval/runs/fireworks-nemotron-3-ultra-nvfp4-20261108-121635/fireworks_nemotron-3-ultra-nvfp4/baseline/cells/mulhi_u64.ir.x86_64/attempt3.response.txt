export function l $mulhi_u64(l %a, l %b) {
@start
    %a_lo = andl %a, 0xFFFFFFFF
    %a_hi = shrl %a, 32
    %b_lo = andl %b, 0xFFFFFFFF
    %b_hi = shrl %b, 32

    %t0 = mull %a_lo, %b_lo
    %t1 = mull %a_lo, %b_hi
    %t2 = mull %a_hi, %b_lo
    %t3 = mull %a_hi, %b_hi

    %sum_mid = addl %t1, %t2
    %carry_mid_w = cult %sum_mid, %t1
    %carry_mid = extuw %carry_mid_w

    %sum_mid_lo32 = andl %sum_mid, 0xFFFFFFFF
    %shifted = shll %sum_mid_lo32, 32

    %low_sum = addl %t0, %shifted
    %carry_low_w = cult %low_sum, %t0
    %carry_low = extuw %carry_low_w

    %sum_mid_hi32 = shrl %sum_mid, 32

    %high1 = addl %t3, %carry_low
    %high2 = addl %high1, %sum_mid_hi32
    %carry_mid_shifted = shll %carry_mid, 32
    %result = addl %high2, %carry_mid_shifted

    ret %result
}
