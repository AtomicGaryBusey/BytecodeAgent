export function l $mulhi_u64(l %a, l %b) {
@start
    %a_lo = and %a, 0xFFFFFFFF
    %a_hi = shr %a, 32
    %b_lo = and %b, 0xFFFFFFFF
    %b_hi = shr %b, 32

    %t0 = mul %a_lo, %b_lo
    %t1 = mul %a_lo, %b_hi
    %t2 = mul %a_hi, %b_lo
    %t3 = mul %a_hi, %b_hi

    %sum_mid = add %t1, %t2
    %carry_mid_w = cult %sum_mid, %t1
    %carry_mid = extuw %carry_mid_w

    %sum_mid_lo32 = and %sum_mid, 0xFFFFFFFF
    %shifted = shl %sum_mid_lo32, 32

    %low_sum = add %t0, %shifted
    %carry_low_w = cult %low_sum, %t0
    %carry_low = extuw %carry_low_w

    %sum_mid_hi32 = shr %sum_mid, 32

    %high1 = add %t3, %carry_low
    %high2 = add %high1, %sum_mid_hi32
    %carry_mid_shifted = shl %carry_mid, 32
    %result = add %high2, %carry_mid_shifted

    ret %result
}
