packbits_encode:
    xor r8, r8          # input index i
    xor r9, r9          # output index written
    test rsi, rsi
    jz .Ldone

.Lmain_loop:
    cmp r8, rsi
    jge .Ldone

    # compute run length at r8
    movzx eax, byte ptr [rdi + r8]
    mov r10, 1
.Lrun_loop:
    cmp r10, 128
    jge .Lrun_done
    lea r11, [r8 + r10]
    cmp r11, rsi
    jge .Lrun_done
    cmp byte ptr [rdi + r11], al
    jne .Lrun_done
    inc r10
    jmp .Lrun_loop
.Lrun_done:
    cmp r10, 3
    jl .Lliteral_packet

    # repeat packet: need 2 bytes in output
    lea rax, [r9 + 2]
    cmp rax, rcx
    jg .Lerror
    mov al, 1
    sub al, r10b
    mov byte ptr [rdx + r9], al
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lmain_loop

.Lliteral_packet:
    mov r10, 0          # literal length L
    mov r11, r8         # scan index j
.Lliteral_extend:
    cmp r11, rsi
    jge .Lliteral_done
    mov rax, r11
    sub rax, r8
    cmp rax, 128
    jge .Lliteral_done
    movzx eax, byte ptr [rdi + r11]
    mov r12, 1
.Lrun_check_loop:
    cmp r12, 3
    jge .Lrun_ge3
    lea rax, [r11 + r12]
    cmp rax, rsi
    jge .Lrun_lt3
    cmp byte ptr [rdi + rax], al
    jne .Lrun_lt3
    inc r12
    jmp .Lrun_check_loop
.Lrun_ge3:
    jmp .Lliteral_done
.Lrun_lt3:
    inc r10
    inc r11
    jmp .Lliteral_extend
.Lliteral_done:
    # r10 = L, r11 = j (end of literal)
    lea rax, [r9 + 1 + r10]
    cmp rax, rcx
    jg .Lerror
    mov al, r10b
    dec al
    mov byte ptr [rdx + r9], al
    # copy L bytes from src[i] to out[written+1]
    mov r12, r10        # copy counter
    mov r13, r8         # source index
    lea r14, [rdx + r9 + 1]  # destination pointer
.Lcopy_loop:
    movzx eax, byte ptr [rdi + r13]
    mov byte ptr [r14], al
    inc r13
    inc r14
    dec r12
    jnz .Lcopy_loop
    add r9, 1
    add r9, r10
    mov r8, r11
    jmp .Lmain_loop

.Lerror:
    mov rax, -1
    ret

.Ldone:
    mov rax, r9
    ret
