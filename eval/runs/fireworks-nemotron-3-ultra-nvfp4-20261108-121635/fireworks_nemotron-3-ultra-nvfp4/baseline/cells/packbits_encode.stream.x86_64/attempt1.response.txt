packbits_encode:
    push r12
    push r13
    xor r8, r8
    xor r9, r9
    test rsi, rsi
    jz .Ldone

.Lmain_loop:
    cmp r8, rsi
    jge .Ldone

    movzx eax, byte ptr [rdi + r8]
    mov r10, 1
.Lrun_loop:
    cmp r10, 128
    jge .Lrun_done
    lea r11, [r8 + r10]
    cmp r11, rsi
    jge .Lrun_done
    cmp byte ptr [rdi + r11], al
    jne .Lrun_done
    inc r10
    jmp .Lrun_loop
.Lrun_done:
    cmp r10, 3
    jl .Lliteral_packet

    lea rax, [r9 + 2]
    cmp rax, rcx
    jg .Lerror
    mov al, 1
    sub al, r10b
    mov byte ptr [rdx + r9], al
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Lmain_loop

.Lliteral_packet:
    mov r10, 0
    mov r11, r8
.Lliteral_extend:
    cmp r11, rsi
    jge .Lliteral_done
    mov r12, r11
    sub r12, r8
    cmp r12, 128
    jge .Lliteral_done
    movzx eax, byte ptr [rdi + r11]
    mov r12, 1
.Lrun_check_loop:
    cmp r12, 3
    jge .Lrun_ge3
    lea r13, [r11 + r12]
    cmp r13, rsi
    jge .Lrun_lt3
    cmp byte ptr [rdi + r13], al
    jne .Lrun_lt3
    inc r12
    jmp .Lrun_check_loop
.Lrun_ge3:
    jmp .Lliteral_done
.Lrun_lt3:
    inc r10
    inc r11
    jmp .Lliteral_extend
.Lliteral_done:
    lea rax, [r9 + 1 + r10]
    cmp rax, rcx
    jg .Lerror
    mov al, r10b
    dec al
    mov byte ptr [rdx + r9], al
    mov r8, r11
    mov r11, r10
    lea r12, [rdx + r9 + 1]
    lea r13, [rdi + r8 - r10]
