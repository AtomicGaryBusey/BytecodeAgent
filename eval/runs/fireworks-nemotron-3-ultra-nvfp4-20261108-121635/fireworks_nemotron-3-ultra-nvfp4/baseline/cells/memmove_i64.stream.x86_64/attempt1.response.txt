memmove_i64:
    test rdx, rdx
    jz .done
    cmp rdi, rsi
    jb .forward
    ja .backward
.done:
    xor rax, rax
    ret
.forward:
    xor rcx, rcx
.loop_fwd:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jne .loop_fwd
    jmp .done
.backward:
    mov rcx, rdx
    dec rcx
.loop_bwd:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    test rcx, rcx
    jz .done
    dec rcx
    jmp .loop_bwd
