isqrt:
    mov     x1, #1
    lsl     x1, x1, #62          // m = 1 << 62
    mov     x2, #0               // y = 0
.Lloop:
    orr     x3, x2, x1           // b = y | m
    lsr     x2, x2, #1           // y >>= 1
    cmp     x0, x3               // compare n (x0) with b
    b.lo    .Lskip               // if n < b (unsigned), skip
    sub     x0, x0, x3           // n -= b
    orr     x2, x2, x1           // y |= m
.Lskip:
    lsr     x1, x1, #2           // m >>= 2
    cbnz    x1, .Lloop
    mov     x0, x2               // return y
    ret
