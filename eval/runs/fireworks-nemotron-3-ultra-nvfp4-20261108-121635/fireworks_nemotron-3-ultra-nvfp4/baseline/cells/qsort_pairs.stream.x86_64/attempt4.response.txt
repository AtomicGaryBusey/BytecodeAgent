qsort_pairs:
    cmp rsi, 1
    jle .Lbase_return
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
.Louter_loop:
    cmp r12, 1
    jle .Lpop_return
    mov rax, r12
    shr rax, 1
    shl rax, 4
    mov r13, rax
    mov r14, qword ptr [rbx + r13]
    mov r15, qword ptr [rbx + r13 + 8]
    xor r13, r13
    mov r8, r12
    dec r8
    shl r8, 4
.Lpartition_loop:
    mov r9, qword ptr [rbx + r13]
    mov r10, qword ptr [rbx + r13 + 8]
    cmp r9, r14
    jl .Linc_i
    jg .Lcheck_j
    cmp r10, r15
    jle .Linc_i
.Lcheck_j:
    mov r9, qword ptr [rbx + r8]
    mov r10, qword ptr [rbx + r8 + 8]
    cmp r9, r14
    jg .Ldec_j
    jl .Lswap
    cmp r10, r15
    jge .Ldec_j
.Lswap:
    cmp r13, r8
    jg .Lpartition_done
    mov r9, qword ptr [rbx + r13]
    mov r10, qword ptr [rbx + r13 + 8]
    mov r11, qword ptr [rbx + r8]
    mov rdx, qword ptr [rbx + r8 + 8]
    mov qword ptr [rbx + r13], r11
    mov qword ptr [rbx + r13 + 8], rdx
    mov qword ptr [rbx + r8], r9
    mov qword ptr [rbx + r8 + 8], r10
.Linc_i:
    add r13, 16
    cmp r13, r8
    jle .Lpartition_loop
.Ldec_j:
    sub r8, 16
    cmp r13, r8
    jle .Lpartition_loop
.Lpartition_done:
    sar r13, 4
    mov rdi, rbx
    mov rsi, r13
    sub rsp, 8
    call qsort_pairs
    add rsp, 8
    lea rdi, [rbx + r13*16]
