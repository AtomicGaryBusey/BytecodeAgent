qsort_pairs:
    cmp rsi, 1
    jle .Lbase_return

    push rbx
    push r12
    push r13
    push r14
    push r15

    mov rbx, rdi                # base pointer
    mov r12, 0                  # lo = 0
    mov r13, rsi
    dec r13                     # hi = n-1

    # choose pivot index = (lo + hi) / 2
    mov rax, r12
    add rax, r13
    shr rax, 1
    mov r14, qword ptr [rbx + rax*16]      # pivot key
