export function l $crc32(l %p, l %n) {
@start
    %crc =w copy 4294967295
    %i =l copy 0
    jmp @outer_cond

@outer_cond
    %cond =w csltl %i, %n
    jnz %cond, @outer_body, @outer_end

@outer_body
    %addr =l add %p, %i
    %byte =w loadub %addr
    %crc =w xor %crc, %byte
    %j =l copy 0
    jmp @inner_cond

@inner_cond
    %cond2 =w csltl %j, 8
    jnz %cond2, @inner_body, @inner_end

@inner_body
    %bit =w and %crc, 1
    %crc =w shr %crc, 1
    %mask =w neg %bit
    %mask =w and %mask, 3988292384
    %crc =w xor %crc, %mask
    %j =l add %j, 1
    jmp @inner_cond

@inner_end
    %i =l add %i, 1
    jmp @outer_cond

@outer_end
    %crc =w xor %crc, 4294967295
    %ret =l extuw %crc
    ret %ret
}
