export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %na_zero =w ceql %na, 0
    %nb_zero =w ceql %nb, 0
    %any_zero =w or %na_zero, %nb_zero
    jnz %any_zero, @zero_ret

    %prod =l alloc 128
    %total =l add %na, %nb

    %zero =l copy 0
    %zero_idx =l copy 0
@zero_loop
    %idx_ge =w csgel %zero_idx, %total
    jnz %idx_ge, @zero_done
    %offset =l shl %zero_idx, 3
    %ptr =l add %prod, %offset
    storel %zero, %ptr
    %zero_idx =l add %zero_idx, 1
    jmp @zero_loop
@zero_done

    %i =l copy 0
@outer_loop
    %i_ge_na =w csgel %i, %na
    jnz %i_ge_na, @outer_done

    %a_offset =l shl %i, 3
    %a_ptr =l add %a, %a_offset
    %a_i =l loadl %a_ptr

    %j =l copy 0
    %carry =l copy 0
@inner_loop
    %j_ge_nb =w csgel %j, %nb
    jnz %j_ge_nb, @inner_done

    %b_offset =l shl %j, 3
    %b_ptr =l add %b, %b_offset
    %b_j =l loadl %b_ptr

    %a_lo =l and %a_i, 4294967295
    %a_hi =l shr %a_i, 32
    %b_lo =l and %b_j, 4294967295
    %b_hi =l shr %b_j, 32

    %p0 =l mul %a_lo, %b_lo
    %p1 =l mul %a_lo, %b_hi
    %p2 =l mul %a_hi, %b_lo
    %p3 =l mul %a_hi, %b_hi

    %mid =l add %p1, %p2
    %c_mid =w cultl %mid, %p1

    %mid_lo =l and %mid, 4294967295
    %mid_hi =l shr %mid, 32

    %mid_lo_shift =l shl %mid_lo, 32

    %prod_lo =l add %p0, %mid_lo_shift
    %c_lo =w cultl %prod_lo, %p0

    %c_mid_l =l zextw %c_mid
    %c_mid_shift =l shl %c_mid_l, 32
    %c_lo_l =l zextw %c_lo

    %prod_hi =l add %p3, %mid_hi
    %prod_hi =l add %prod_hi, %c_mid_shift
    %prod_hi =l add %prod_hi, %c_lo_l

    %ij =l add %i, %j
    %ij_offset =l shl %ij, 3
    %prod_ptr =l add %prod, %ij_offset
    %r_ij =l loadl %prod_ptr

    %sum1 =l add %prod_lo, %r_ij
    %c1 =w cultl %sum1, %prod_lo
    %sum2 =l add %sum1, %carry
    %c2 =w cultl %sum2, %sum1

    %c1_l =l zextw %c1
    %c2_l =l zextw %c2
    %carry1 =l add %c1_l, %c2_l

    %new_lo =l copy %sum2
    %new_carry =l add %prod_hi, %carry1

    storel %new_lo, %prod_ptr
    %carry =l copy %new_carry

    %j =l add %j, 1
    jmp @inner_loop
@inner_done

    %i_nb =l add %i, %nb
    %i_nb_offset =l shl %i_nb, 3
    %prod_ptr =l add %prod, %i_nb_offset
    storel %carry, %prod_ptr

    %i =l add %i, 1
    jmp @outer_loop
@outer_done

    %idx =l copy 0
@write_loop
    %idx_ge =w csgel %idx, %total
    jnz %idx_ge, @write_done
    %offset =l shl %idx, 3
    %prod_ptr =l add %prod, %offset
    %val =l loadl %prod_ptr
    %r_ptr =l add %r, %offset
    storel %val, %r_ptr
    %idx =l add %idx, 1
    jmp @write_loop
@write_done

    %sig =l copy 0
    %idx =l sub %total, 1
@scan_loop
    %cond =w sgel %idx, 0
    jnz %cond, @scan_body
    jmp @scan_done
@scan_body
    %offset =l shl %idx, 3
    %prod_ptr =l add %prod, %offset
    %val =l loadl %prod_ptr
    %val_nz =w cne %val, 0
    jnz %val_nz, @found_nonzero
    %idx =l sub %idx, 1
    jmp @scan_loop
@found_nonzero
    %sig =l add %idx, 1
    jmp @scan_done
@scan_done
    ret %sig

@zero_ret
    ret 0
}
