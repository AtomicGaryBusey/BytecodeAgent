export function $fib(l %n) {
@start
    %zero = copy 0
    %one = copy 1
    %cond0 = ceql %n, %zero
    jnz %cond0, @ret0
    %cond1 = ceql %n, %one
    jnz %cond1, @ret1
    %a = copy %zero
    %b = copy %one
    %i = copy 2
    jmp @loop

@loop
    %cond = cslel %i, %n
    jnz %cond, @body
    jmp @done

@body
    %c = addl %a, %b
    %a = copy %b
    %b = copy %c
    %i = addl %i, %one
    jmp @loop

@done
    %ret = copy %b
    ret %ret

@ret0
    ret %zero

@ret1
    ret %one
}
