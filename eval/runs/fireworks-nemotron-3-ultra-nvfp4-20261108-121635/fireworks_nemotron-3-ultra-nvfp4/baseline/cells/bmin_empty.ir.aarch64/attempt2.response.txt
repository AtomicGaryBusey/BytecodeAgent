export function l $bmin_empty(l %a, l %n) {
    @start
    %cmp = ceql %n, 0
    jnz %cmp, @zero, @loop_init

    @zero
    ret 0x7fffffffffffffff

    @loop_init
    %first = loadl %a
    jmp @loop 1, %first

    @loop
    %i = phi l @loop_init 1, @inc %i_next
    %min = phi l @loop_init %first, @inc %min_val
    %cond = csltl %i, %n
    jnz %cond, @body, @done

    @body
    %offset = mul %i, 8
    %ptr = add %a, %offset
    %val = loadl %ptr
    %cmp2 = csltl %val, %min
    jnz %cmp2, @take_val, @take_min

    @take_val
    jmp @inc %val

    @take_min
    jmp @inc %min

    @inc
    %min_val = phi l @take_val %val, @take_min %min
    %i_next = add %i, 1
    jmp @loop %i_next, %min_val

    @done
    ret %min
}
