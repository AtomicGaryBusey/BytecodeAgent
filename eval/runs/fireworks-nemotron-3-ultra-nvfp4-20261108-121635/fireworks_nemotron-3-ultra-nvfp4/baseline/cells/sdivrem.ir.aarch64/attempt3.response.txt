export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%b_zero = ceql %b, 0
	jnz %b_zero, @b_zero, @check_a_min

@b_zero
	%q_zero = copy 0
	storel %a, %rem
	ret %q_zero

@check_a_min
	%a_min = ceql %a, 0x8000000000000000
	jnz %a_min, @check_b_neg1, @normal

@check_b_neg1
	%b_neg1 = ceql %b, -1
	jnz %b_neg1, @overflow, @normal

@overflow
	%q_min = copy 0x8000000000000000
	%r_zero = copy 0
	storel %r_zero, %rem
	ret %q_min

@normal
	%q = divl %a, %b
	%r = reml %a, %b
	storel %r, %rem
	ret %q
}
