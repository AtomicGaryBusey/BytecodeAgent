export function l $reverse(l %a, l %n) {
@start
	%zero = sub %n, %n
	%one = copy 1
	%eight = copy 8
	%cond = csle %n, %one
	jnz %cond, @end, @setup

@setup
	%n1 = sub %n, %one
	%offset = mul %n1, %eight
	%end = add %a, %offset
	%start = copy %a
	jmp @loop

@loop
	%cond2 = cult %start, %end
	jnz %cond2, @body, @end

@body
	%val1 = loadl %start
	%val2 = loadl %end
	storel %val2, %start
	storel %val1, %end
	%start = add %start, %eight
	%end = sub %end, %eight
	jmp @loop

@end
	ret %zero
}
