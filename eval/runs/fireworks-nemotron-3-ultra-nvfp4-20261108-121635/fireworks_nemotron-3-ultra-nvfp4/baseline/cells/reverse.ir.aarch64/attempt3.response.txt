export function l $reverse(l %a, l %n) {
@start
	%zero = subl %n, %n
	%one_w = ceql %n, %n
	%one = extsw %one_w
	%two = addl %one, %one
	%three = addl %two, %one
	%eight = shll %one, %three
	%cond = cslel %n, %one
	jnz %cond, @end, @setup

@setup
	%n1 = subl %n, %one
	%offset = shll %n1, %three
	%end = addl %a, %offset
	%start = copy %a
	jmp @loop

@loop
	%cond2 = cultl %start, %end
	jnz %cond2, @body, @end

@body
	%val1 = loadl %start
	%val2 = loadl %end
	storel %start, %val2
	storel %end, %val1
	%start = addl %start, %eight
	%end = subl %end, %eight
	jmp @loop

@end
	ret %zero
}
