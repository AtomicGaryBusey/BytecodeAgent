export function l bmin_empty(l %a, l %n) {
    @start
    %zero = copy 0
    %n_zero = ceql %n, %zero
    jnz %n_zero, @return_max, @init

    @init
    %min = loadl %a
    %one = copy 1
    %i = copy %one
    %rem = subl %n, %one
    jmp @loop_cond

    @loop_cond
    %cond = csgtl %rem, %zero
    jnz %cond, @loop_body, @return_min

    @loop_body
    %three = copy 3
    %offset = shll %i, %three
    %addr = addl %a, %offset
    %val = loadl %addr
    %cmp = csltl %val, %min
    jnz %cmp, @update_min, @next

    @update_min
    %min = copy %val
    jmp @next

    @next
    %i = addl %i, %one
    %rem = subl %rem, %one
    jmp @loop_cond

    @return_min
    ret %min

    @return_max
    %max = copy 9223372036854775807
    ret %max
}
