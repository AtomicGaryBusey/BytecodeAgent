export function l $bubble_sort(l %a, l %n) {
@entry
    %zero = copy 0l
    %one = copy 1l
    %eight = copy 8l
    %n_gt_1 = csgtl %n, %one
    jnz %n_gt_1, @outer_init
    jump @return_zero

@outer_init
    %i = copy 0l
    jump @outer_cond

@outer_cond
    %n_minus_1 = subl %n, %one
    %i_lt_n_minus_1 = csltl %i, %n_minus_1
    jnz %i_lt_n_minus_1, @inner_init
    jump @return_zero

@inner_init
    %j = copy 0l
    jump @inner_cond

@inner_cond
    %n_minus_i = subl %n, %i
    %limit = subl %n_minus_i, %one
    %j_lt_limit = csltl %j, %limit
    jnz %j_lt_limit, @inner_body
    jump @outer_inc

@inner_body
    %offset_j = mul %j, %eight
    %addr_j = addl %a, %offset_j
    %val_j = loadl %addr_j
    %j1 = addl %j, %one
    %offset_j1 = mul %j1, %eight
    %addr_j1 = addl %a, %offset_j1
    %val_j1 = loadl %addr_j1
    %val_j_gt_val_j1 = csgtl %val_j, %val_j1
    jnz %val_j_gt_val_j1, @swap
    jump @inner_inc

@swap
    storel %addr_j, %val_j1
    storel %addr_j1, %val_j
    jump @inner_inc

@inner_inc
    %j = addl %j, %one
    jump @inner_cond

@outer_inc
    %i = addl %i, %one
    jump @outer_cond

@return_zero
    ret %zero
}
