binary_search:
	cbz x1, .Lnot_found
	mov x3, #0
	sub x4, x1, #1
.Lloop:
	cmp x3, x4
	b.gt .Lnot_found
	add x5, x3, x4
	lsr x5, x5, #1
	ldr x6, [x0, x5, lsl #3]
	cmp x6, x2
	b.eq .Lfound
	b.lt .Llow_adjust
	mov x4, x5
	sub x4, x4, #1
	b .Lloop
.Llow_adjust:
	mov x3, x5
	add x3, x3, #1
	b .Lloop
.Lfound:
	mov x0, x5
	ret
.Lnot_found:
	mov x0, #-1
	ret
