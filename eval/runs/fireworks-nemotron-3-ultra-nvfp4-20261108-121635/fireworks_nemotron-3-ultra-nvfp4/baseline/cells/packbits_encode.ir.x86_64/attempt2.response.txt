export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
    %zero =l copy 0
    jmp @main_loop

@main_loop
    %i =l phi @entry %zero, @repeat_packet %i_repeat, @literal_packet %i_literal
    %out_pos =l phi @entry %zero, @repeat_packet %out_pos_repeat, @literal_packet %out_pos_literal
    %cont =w csltl %i, %n
    jnz %cont, @run_loop_entry, @exit

@run_loop_entry
    %one =l copy 1
    jmp @run_loop

@run_loop
    %i_rl =l phi @run_loop_entry %i, @run_loop %i_rl
    %out_pos_rl =l phi @run_loop_entry %out_pos, @run_loop %out_pos_rl
    %run =l phi @run_loop_entry %one, @run_loop %run_inc
    %cond1 =w csltl %run, 128
    %i_plus_run =l addl %i_rl, %run
    %cond2 =w csltl %i_plus_run, %n
    %cond =w andw %cond1, %cond2
    jnz %cond, @run_check_byte, @run_done

@run_check_byte
    %src_i =l addl %src, %i_rl
    %curr_byte =w loadub %src_i
    %src_i_plus_run =l addl %src, %i_plus_run
    %next_byte =w loadub %src_i_plus_run
    %eq =w ceqw %curr_byte, %next_byte
    jnz %eq, @run_loop_body, @run_done

@run_loop_body
    %run_inc =l addl %run, 1
    jmp @run_loop

@run_done
    %cond_repeat =w csgew %run, 3
    jnz %cond_repeat, @repeat_packet, @literal_packet

@repeat_packet
    %need =l addl %out_pos_rl, 2
    %cap_ok =w cslel %need, %cap
    jnz %cap_ok, @repeat_write, @fail

@repeat_write
    %r_w =w copy %run
    %header =w subw 257, %r_w
    %out_ptr =l addl %out, %out_pos_rl
    storeb %header, %out_ptr
    %out_ptr1 =l addl %out_ptr, 1
    %src_i =l addl %src, %i_rl
    %byte =w loadub %src_i
    storeb %byte, %out_ptr1
    %out_pos_repeat =l addl %out_pos_rl, 2
    %i_repeat =l addl %i_rl, %run
    jmp @main_loop

@literal_packet
    %j =l phi @run_done %i_rl, @literal_scan %j_inc
    %i_lp =l phi @run_done %i_rl, @literal_scan %i_lp
    %out_pos_lp =l phi @run_done %out_pos_rl, @literal_scan %out_pos_lp
    %cond_j1 =w csltl %j, %n
    %j_minus_i =l subl %j, %i_lp
    %cond_j2 =w csltl %j_minus_i, 128
    %cond_j =w andw %cond_j1, %cond_j2
    jnz %cond_j, @check_run_j, @literal_done

@check_run_j
    %j_plus2 =l addl %j, 2
    %cond_j3 =w csltl %j_plus2, %n
    jnz %cond_j3, @check_three_equal, @literal_scan_continue

@check_three_equal
    %src_j =l addl %src, %j
    %b0 =w loadub %src_j
    %src_j1 =l addl %src_j, 1
    %b1 =w loadub %src_j1
    %src_j2 =l addl %src_j1, 1
    %b2 =w loadub %src_j2
    %eq01 =w ceqw %b0, %b1
    %eq12 =w ceqw %b1, %b2
    %all_eq =w andw %eq01, %eq12
    jnz %all_eq, @literal_done, @literal_scan_continue

@literal_scan_continue
    %j_inc =l addl %j, 1
    %i_lp_next =l copy %i_lp
    %out_pos_lp_next =l copy %out_pos_lp
    jmp @literal_packet

@literal_done
    %L =l subl %j, %i_lp
    %need =l addl %out_pos_lp, 1
    %need =l addl %need, %L
    %cap_ok =w cslel %need, %cap
    jnz %cap_ok, @literal_write_header, @fail

@literal_write_header
    %header =w subw %L, 1
    %out_ptr =l addl %out, %out_pos_lp
    storeb %header, %out_ptr
    jmp @literal_copy_head

@literal_copy_head
    %copy_idx =l phi @literal_write_header %zero, @literal_copy_body %copy_idx_inc
    %cond_copy =w csltl %copy_idx, %L
    jnz %cond_copy, @literal_copy_body, @literal_copy_done

@literal_copy_body
    %src_ptr =l addl %src, %i_lp
    %src_ptr =l addl %src_ptr, %copy_idx
    %byte =w loadub %src_ptr
    %out_ptr =l addl %out, %out_pos_lp
    %out_ptr =l addl %out_ptr, 1
    %out_ptr =l addl %out_ptr, %copy_idx
    storeb %byte, %out_ptr
    %copy_idx_inc =l addl %copy_idx, 1
    jmp @literal_copy_head

@literal_copy_done
    %out_pos_literal =l addl %out_pos_lp, 1
    %out_pos_literal =l addl %out_pos_literal, %L
    %i_literal =l copy %j
    jmp @main_loop

@fail
    %neg1 =l copy -1
    ret %neg1

@exit
    ret %out_pos
}
