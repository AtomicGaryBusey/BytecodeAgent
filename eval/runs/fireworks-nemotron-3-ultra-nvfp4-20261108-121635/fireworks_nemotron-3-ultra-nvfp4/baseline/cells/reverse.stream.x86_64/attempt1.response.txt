reverse:
    xor eax, eax                # return 0
    test rsi, rsi
    jle .Lend                   # if n <= 0, done
    mov rcx, rsi                # rcx = n
    dec rcx                     # rcx = n-1 (last index)
    shr rsi, 1                  # rsi = n/2 (half count)
    xor r8, r8                  # r8 = i = 0
    test rsi, rsi
    jz .Lend
.Lloop:
    mov r9, qword ptr [rdi + r8*8]       # a[i]
    mov r10, qword ptr [rdi + rcx*8]     # a[n-1-i]
    mov qword ptr [rdi + r8*8], r10
    mov qword ptr [rdi + rcx*8], r9
    inc r8
    dec rcx
    cmp r8, rsi
    jl .Lloop
.Lend:
    ret
