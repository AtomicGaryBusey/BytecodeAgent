export function l $count_desc(l %a, l %n) {
@entry
	%zero = constw 0
	%one = constw 1
	%two = constw 2
	%eight = constw 8
	%n_minus_one = subl %n, %one
	%cond = csltl %n, %two
	jnz %cond, @return_zero
	jmp @loop_head(%zero, %zero)

@return_zero
	jmp @exit(%zero)

@loop_head
	%count = phi [%zero, @entry], [%c_body, @loop_body]
	%i = phi [%zero, @entry], [%i_body, @loop_body]
	%cond = csltl %i, %n_minus_one
	jnz %cond, @loop_body
	jmp @exit(%count)

@loop_body
	%offset_i = mull %i, %eight
	%val_i = loadl %a, %offset_i
	%i_plus_one = addl %i, %one
	%offset_ip1 = mull %i_plus_one, %eight
	%val_ip1 = loadl %a, %offset_ip1
	%cmp = csgtl %val_i, %val_ip1
	%cmp_l = zextl %cmp
	%c_body = addl %count, %cmp_l
	%i_body = addl %i, %one
	jmp @loop_head(%c_body, %i_body)

@exit
	%count_final = phi [%zero, @return_zero], [%count, @loop_head]
	ret %count_final
}
