export function l $count_desc(l %a, l %n) {
@entry
	%zero = constl 0
	%one = constl 1
	%two = constl 2
	%eight = constl 8
	%n_minus_one = subl %n, %one
	%cond = csltl %n, %two
	jnz %cond, @exit(%zero)
	jmp @loop_head(%zero, %zero)

@loop_head
	%count = phi l [%zero, @entry], [%c_body, @loop_body]
	%i = phi l [%zero, @entry], [%i_body, @loop_body]
	%cond = csltl %i, %n_minus_one
	jnz %cond, @loop_body(%count, %i)
	jmp @exit(%count)

@loop_body
	%offset_i = mull %i, %eight
	%val_i = loadl %a, %offset_i
	%i_plus_one = addl %i, %one
	%offset_ip1 = mull %i_plus_one, %eight
	%val_ip1 = loadl %a, %offset_ip1
	%cmp = csgtl %val_i, %val_ip1
	%c_body = addl %count, zextl %cmp
	%i_body = addl %i, %one
	jmp @loop_head(%c_body, %i_body)

@exit
	%count_final = phi l [%zero, @entry], [%count, @loop_head]
	ret %count_final
}
