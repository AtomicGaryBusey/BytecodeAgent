export function l $count_desc(l %a, l %n) {
@entry
	%zero = const 0
	%one = const 1
	%two = const 2
	%eight = const 8
	%n_minus_one = sub %n, %one
	%cond = clt %n, %two
	jnz %cond, @return_zero
	jmp @loop_head(%zero, %zero)

@return_zero
	jmp @exit(%zero)

@loop_head %count %i
	%cond = clt %i, %n_minus_one
	jnz %cond, @loop_body
	jmp @exit(%count)

@loop_body
	%offset_i = mul %i, %eight
	%val_i = load %a, %offset_i
	%i_plus_one = add %i, %one
	%offset_ip1 = mul %i_plus_one, %eight
	%val_ip1 = load %a, %offset_ip1
	%cmp = cgt %val_i, %val_ip1
	%cmp_l = zext %cmp
	%c_body = add %count, %cmp_l
	%i_body = add %i, %one
	jmp @loop_head(%c_body, %i_body)

@exit %count_final
	ret %count_final
}
