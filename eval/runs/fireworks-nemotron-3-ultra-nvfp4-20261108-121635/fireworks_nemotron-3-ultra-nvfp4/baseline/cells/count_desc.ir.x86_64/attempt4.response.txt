export function l $count_desc(l %a, l %n) {
@start
    %two = copy 2
    %cond = csltl %n, %two
    jnz %cond, @return_zero, @init

@init
    %one = copy 1
    %n_minus_1 = subl %n, %one
    %eight = copy 8
    %tmp = mull %n_minus_1, %eight
    %end = addl %a, %tmp
    jump @loop

@loop
    %p = phi @init %a, @body %p_next
    %count = phi @init 0, @body %count_next
    %cond = csltl %p, %end
    jnz %cond, @body, @done

@body
    %val1 = loadl %p
    %p_next = addl %p, 8
    %val2 = loadl %p_next
    %cmp = csgtl %val1, %val2
    %inc = zextw %cmp
    %count_next = addl %count, %inc
    jump @loop

@done
    ret %count

@return_zero
    %zero = copy 0
    ret %zero
}
