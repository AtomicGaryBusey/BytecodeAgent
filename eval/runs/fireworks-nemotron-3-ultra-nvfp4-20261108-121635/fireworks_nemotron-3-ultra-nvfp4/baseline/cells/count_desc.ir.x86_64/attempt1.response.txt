export function l $count_desc(l %a, l %n) {
@start
    %cond = csltl %n, 2
    jnz %cond, @return_zero, @init

@init
    %n_minus_1 = subl %n, 1
    %end = addl %a, mul %n_minus_1, 8
    %p = %a
    %count = 0
    jump @loop

@loop
    %cond = csltl %p, %end
    jnz %cond, @body, @done

@body
    %val1 = loadl %p
    %val2 = loadl addl %p, 8
    %cmp = csgtl %val1, %val2
    %inc = zextw %cmp
    %count = addl %count, %inc
    %p = addl %p, 8
    jump @loop

@done
    ret %count

@return_zero
    ret 0
}
