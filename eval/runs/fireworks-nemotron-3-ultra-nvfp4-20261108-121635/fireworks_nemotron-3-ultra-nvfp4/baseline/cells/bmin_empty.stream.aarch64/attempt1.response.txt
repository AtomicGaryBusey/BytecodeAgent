bmin_empty:
	cbz x1, .Lreturn_max
	ldr x2, [x0]
	mov x3, x2
	subs x1, x1, #1
	cbz x1, .Ldone
	add x4, x0, #8
.Lloop:
	ldr x5, [x4], #8
	cmp x5, x3
	csel x3, x5, x3, lt
	subs x1, x1, #1
	b.ne .Lloop
.Ldone:
	mov x0, x3
	ret
.Lreturn_max:
	movz x0, #0xFFFF, lsl #48
	movk x0, #0xFFFF, lsl #32
	movk x0, #0xFFFF, lsl #16
	movk x0, #0x7FFF
	ret
