dot_i32_simd:
	cbz x2, .Lreturn_zero
	lsr x3, x2, #2
	and x4, x2, #3
	movi v0.2d, #0
	mov x5, #0
	cbz x3, .Lremainder_start
.Lvector_loop:
	ld1 {v1.4s}, [x0], #16
	ld1 {v2.4s}, [x1], #16
	smlal v0.2d, v1.2s, v2.2s
	smlal2 v0.2d, v1.4s, v2.4s
	subs x3, x3, #1
	b.ne .Lvector_loop
.Lremainder_start:
	cbz x4, .Lreduce_vector
.Lremainder_loop:
	ldrsw x6, [x0], #4
	ldrsw x7, [x1], #4
	mul x6, x6, x7
	add x5, x5, x6
	subs x4, x4, #1
	b.ne .Lremainder_loop
.Lreduce_vector:
	addp v0.2d, v0.2d, v0.2d
	umov x6, v0.d[0]
	add x0, x5, x6
	ret
.Lreturn_zero:
	mov x0, #0
	ret
