export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %i =l copy 0
    %addr0 =l addl %a, 0
    %x0 =l loadl %addr0
    %y0 =l call %f(l %x0)
    %min =l copy %y0
    %max =l copy %y0
    %sum =l copy %y0
    %count =l andl %y0, 1
    %i =l addl %i, 1
    %cond =w cltl %i, %n
    jnz %cond, @loop, @store

@loop
    %offset =l mull %i, 8
    %addr =l addl %a, %offset
    %x =l loadl %addr
    %y =l call %f(l %x)
    %cmp_min =w cltl %y, %min
    %min =l sel %cmp_min, %y, %min
    %cmp_max =w cgtl %y, %max
    %max =l sel %cmp_max, %y, %max
    %sum =l addl %sum, %y
    %odd =l andl %y, 1
    %count =l addl %count, %odd
    %i =l addl %i, 1
    %cond =w cltl %i, %n
    jnz %cond, @loop, @store

@store
    storel %min, %out
    %out1 =l addl %out, 8
    storel %max, %out1
    %out2 =l addl %out, 16
    storel %sum, %out2
    ret %count
}
