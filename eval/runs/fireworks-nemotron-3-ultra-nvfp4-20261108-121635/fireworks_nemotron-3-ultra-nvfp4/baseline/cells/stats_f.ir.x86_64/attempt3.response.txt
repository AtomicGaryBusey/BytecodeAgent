export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %i =l copy 0
    %addr0 =l add %a, 0
    %x0 =l loadl %addr0
    %y0 =l call %f(l %x0)
    %min =l copy %y0
    %max =l copy %y0
    %sum =l copy %y0
    %count =l and %y0, 1
    %i =l add %i, 1
    %cond =w sltl %i, %n
    jnz %cond, @loop, @store

@loop
    %offset =l mul %i, 8
    %addr =l add %a, %offset
    %x =l loadl %addr
    %y =l call %f(l %x)
    %cmp_min =w sltl %y, %min
    %min =l select %cmp_min, %y, %min
    %cmp_max =w sgtl %y, %max
    %max =l select %cmp_max, %y, %max
    %sum =l add %sum, %y
    %odd =l and %y, 1
    %count =l add %count, %odd
    %i =l add %i, 1
    %cond =w sltl %i, %n
    jnz %cond, @loop, @store

@store
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %count
}
