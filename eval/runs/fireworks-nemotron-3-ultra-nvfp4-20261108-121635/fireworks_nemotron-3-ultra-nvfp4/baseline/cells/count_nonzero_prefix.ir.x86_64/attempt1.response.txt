export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %i =l copy 0
    %cnt =l copy 0
    jmp @loop

@loop
    %cond =w cnultl %i, %n
    jnz %cond, @check, @done

@check
    %val =l loadl %a, %i, 3
    %zero =w ceql %val, $0
    jnz %zero, @done, @inc

@inc
    %cnt =l addl %cnt, 1
    %i =l addl %i, 1
    jmp @loop

@done
    ret %cnt
}
