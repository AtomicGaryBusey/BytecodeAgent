export function l $count_nonzero_prefix(l %a, l %n) {
@start
    %p =l copy %a
    %rem =l copy %n
    %cnt =l copy 0
    jmp @loop

@loop
    %cond =w cltl $0, %rem
    jnz %cond, @check, @done

@check
    %val =l loadl %p, 0, 3
    %zero =w ceql %val, $0
    jnz %zero, @done, @inc

@inc
    %p =l addl %p, 8
    %rem =l subl %rem, 1
    %cnt =l addl %cnt, 1
    jmp @loop

@done
    ret %cnt
}
