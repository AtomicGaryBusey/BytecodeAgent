.globl _start
.text
_start:
    lea r12, [rip + input_buf]
    xor rbx, rbx
read_loop:
    cmp rbx, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + rbx]
    mov rdx, 65536
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov rbp, rbx
    mov r13, rbx
    xor rcx, rcx
find_loop:
    cmp rcx, rbx
    jae bal_setup
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 33
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov rbp, rcx
    mov r13, rcx
    inc r13
bal_setup:
    lea r14, [rip + jump_table]
    xor r8, r8
    xor rcx, rcx
bal_loop:
    cmp rcx, rbp
    jae bal_done
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 91
    je bal_open
    cmp al, 93
    je bal_close
    inc rcx
    jmp bal_loop
bal_open:
    lea r9, [rip + br_stack]
    mov dword ptr [r9 + r8*4], ecx
    inc r8
    inc rcx
    jmp bal_loop
bal_close:
    cmp r8, 0
    je unbalanced
    dec r8
    lea r9, [rip + br_stack]
    mov eax, dword ptr [r9 + r8*4]
    mov dword ptr [r14 + rcx*4], eax
    mov dword ptr [r14 + rax*4], ecx
    inc rcx
    jmp bal_loop
bal_done:
    cmp r8, 0
    jne unbalanced
exec_start:
    lea r15, [rip + tape]
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor rdx, rdx
exec_loop:
    cmp r9, rbp
    jae exec_done
    movzx eax, byte ptr [r12 + r9]
    cmp al, 62
    je cmd_gt
    cmp al, 60
    je cmd_lt
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_dot
    cmp al, 44
    je cmd_comma
    cmp al, 91
    je cmd_lbrac
    cmp al, 93
    je cmd_rbrac
    inc r9
    jmp exec_loop
cmd_gt:
    inc r8
    inc r9
    jmp exec_loop
cmd_lt:
    dec r8
    inc r9
    jmp exec_loop
cmd_plus:
    inc byte ptr [r15 + r8]
    inc r9
    jmp exec_loop
cmd_minus:
    dec byte ptr [r15 + r8]
    inc r9
    jmp exec_loop
cmd_dot:
    movzx eax, byte ptr [r15 + r8]
    lea rcx, [rip + out_buf]
    mov byte ptr [rcx + rdx], al
    inc rdx
    inc r9
    jmp exec_loop
cmd_comma:
    mov rax, rbx
    sub rax, r13
    cmp r10, rax
    jae comma_zero
    lea rax, [r13 + r10]
    movzx eax, byte ptr [r12 + rax]
    mov byte ptr [r15 + r8], al
    inc r10
    jmp comma_next
comma_zero:
    mov byte ptr [r15 + r8], 0
comma_next:
    inc r9
    jmp exec_loop
cmd_lbrac:
    cmp byte ptr [r15 + r8], 0
    jne lbrac_next
    mov eax, dword ptr [r14 + r9*4]
    mov r9, rax
    inc r9
    jmp exec_loop
lbrac_next:
    inc r9
    jmp exec_loop
cmd_rbrac:
    cmp byte ptr [r15 + r8], 0
    je rbrac_next
    mov eax, dword ptr [r14 + r9*4]
    mov r9, rax
    inc r9
    jmp exec_loop
rbrac_next:
    inc r9
    jmp exec_loop
exec_done:
    mov rbx, rdx
    xor r8, r8
write_loop:
    cmp r8, rbx
    jae exit_ok
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    add rsi, r8
    mov rdx, rbx
    sub rdx, r8
    syscall
    cmp rax, 0
    jle exit_ok
    add r8, rax
    jmp write_loop
exit_ok:
    mov rax, 60
    mov rdi, 0
    syscall
    jmp hang
unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
hang:
    jmp hang
.bss
.balign 8
input_buf:
    .zero 65536
.balign 8
out_buf:
    .zero 65536
.balign 8
tape:
    .zero 30000
.balign 4
jump_table:
    .zero 262144
.balign 4
br_stack:
    .zero 262144
