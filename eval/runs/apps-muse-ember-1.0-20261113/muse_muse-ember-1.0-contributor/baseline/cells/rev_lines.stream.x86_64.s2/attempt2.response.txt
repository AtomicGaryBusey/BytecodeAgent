.globl _start
.text
_start:
    lea r12, [rip + buf]
    xor r13, r13
read_loop:
    cmp r13, 65536
    jae read_done
    mov rdi, 0
    lea rsi, [r12 + r13]
    mov rdx, 65536
    sub rdx, r13
    mov rax, 0
    syscall
    cmp rax, 0
    je read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14, r14
    xor r15, r15
line_loop:
    cmp r15, r13
    jae final_check
    movzx eax, byte ptr [r12 + r15]
    cmp al, 10
    je handle_newline
    inc r15
    jmp line_loop
handle_newline:
    cmp r15, r14
    je rev1_done
    mov r8, r14
    mov r9, r15
    dec r9
rev1_loop:
    cmp r8, r9
    jae rev1_done
    mov al, byte ptr [r12 + r8]
    mov dl, byte ptr [r12 + r9]
    mov byte ptr [r12 + r8], dl
    mov byte ptr [r12 + r9], al
    inc r8
    dec r9
    jmp rev1_loop
rev1_done:
    mov rdi, 1
    lea rsi, [r12 + r14]
    mov rdx, r15
    sub rdx, r14
    inc rdx
    mov rax, 1
    syscall
    inc r15
    mov r14, r15
    jmp line_loop
final_check:
    cmp r14, r13
    jae done
    mov r8, r14
    mov r9, r13
    dec r9
rev2_loop:
    cmp r8, r9
    jae rev2_done
    mov al, byte ptr [r12 + r8]
    mov dl, byte ptr [r12 + r9]
    mov byte ptr [r12 + r8], dl
    mov byte ptr [r12 + r9], al
    inc r8
    dec r9
    jmp rev2_loop
rev2_done:
    mov rdi, 1
    lea rsi, [r12 + r14]
    mov rdx, r13
    sub rdx, r14
    mov rax, 1
    syscall
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rax, 1
    syscall
done:
    mov rax, 60
    xor rdi, rdi
    syscall
infloop:
    jmp infloop
.section .rodata
nl:
    .byte 10
.bss
buf:
    .zero 65536
