.text
.globl _start
_start:
 xor r12, r12
 xor r13, r13
 xor r14, r14
 xor r15, r15
read_loop:
 mov rax, 0
 mov rdi, 0
 lea rsi, [rip + buffer]
 mov rdx, 4096
 syscall
 cmp rax, 0
 jle done_read
 mov rbp, rax
 xor rbx, rbx
byte_loop:
 cmp rbx, rbp
 jae read_loop
 lea rsi, [rip + buffer]
 movzx eax, byte ptr [rsi + rbx]
 inc r14
 cmp al, 10
 jne not_nl
 inc r12
not_nl:
 cmp al, 32
 je is_ws
 cmp al, 9
 jb is_word
 cmp al, 13
 ja is_word
is_ws:
 test r15, r15
 jz next_byte
 xor r15, r15
 jmp next_byte
is_word:
 test r15, r15
 jnz next_byte
 mov r15, 1
 inc r13
next_byte:
 inc rbx
 jmp byte_loop
done_read:
 lea rsi, [rip + outbuf]
 add rsi, 64
 dec rsi
 mov byte ptr [rsi], 10
 mov rax, r14
 call convert
 dec rsi
 mov byte ptr [rsi], 32
 mov rax, r13
 call convert
 dec rsi
 mov byte ptr [rsi], 32
 mov rax, r12
 call convert
 lea rdx, [rip + outbuf]
 add rdx, 64
 sub rdx, rsi
 mov rax, 1
 mov rdi, 1
 syscall
 mov rax, 60
 xor rdi, rdi
 syscall
convert:
 cmp rax, 0
 jne conv_loop
 dec rsi
 mov byte ptr [rsi], 48
 ret
conv_loop:
 xor rdx, rdx
 mov rcx, 10
 div rcx
 add dl, 48
 dec rsi
 mov byte ptr [rsi], dl
 cmp rax, 0
 jne conv_loop
 ret
.bss
buffer:
 .zero 4096
outbuf:
 .zero 64
