.globl _start
_start:
    lea r12, [rip + inbuf]
    lea r15, [rip + offs]
    lea rbx, [rip + lens]
    xor r13, r13
read_loop:
    cmp r13, 32768
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    add rsi, r13
    mov rdx, 32768
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    test r13, r13
    jz exit_prog
    xor r14, r14
    xor r8, r8
    xor r9, r9
parse_loop:
    cmp r8, r13
    jae parse_end
    cmp byte ptr [r12 + r8], 10
    jne parse_next
    mov rax, r14
    shl rax, 3
    mov qword ptr [r15 + rax], r9
    mov rcx, r8
    sub rcx, r9
    mov qword ptr [rbx + rax], rcx
    inc r14
    mov r9, r8
    inc r9
parse_next:
    inc r8
    jmp parse_loop
parse_end:
    cmp r9, r13
    jae parsed
    mov rax, r14
    shl rax, 3
    mov qword ptr [r15 + rax], r9
    mov rcx, r13
    sub rcx, r9
    mov qword ptr [rbx + rax], rcx
    inc r14
parsed:
    cmp r14, 2
    jb sort_done
    mov r8, 1
sort_outer:
    cmp r8, r14
    jae sort_done
    mov rax, r8
    shl rax, 3
    mov r10, qword ptr [r15 + rax]
    mov r11, qword ptr [rbx + rax]
    mov r9, r8
    dec r9
sort_inner:
    cmp r9, 0
    jl sort_place
    mov rdi, r9
    mov rsi, r8
    call compare
    test eax, eax
    jle sort_place
    mov rax, r9
    shl rax, 3
    mov rcx, qword ptr [r15 + rax]
    mov qword ptr [r15 + rax + 8], rcx
    mov rcx, qword ptr [rbx + rax]
    mov qword ptr [rbx + rax + 8], rcx
    dec r9
    jmp sort_inner
sort_place:
    mov rax, r9
    inc rax
    shl rax, 3
    mov qword ptr [r15 + rax], r10
    mov qword ptr [rbx + rax], r11
    inc r8
    jmp sort_outer
sort_done:
    xor r8, r8
out_loop:
    cmp r8, r14
    jae exit_prog
    mov rax, r8
    shl rax, 3
    mov rsi, qword ptr [r15 + rax]
    add rsi, r12
    mov rdx, qword ptr [rbx + rax]
    mov rdi, 1
    call write_all
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    call write_all
    inc r8
    jmp out_loop
compare:
    push r8
    push r9
    push r10
    push r11
    mov rax, rdi
    shl rax, 3
    mov r8, qword ptr [r15 + rax]
    add r8, r12
    mov r9, qword ptr [rbx + rax]
    mov rax, rsi
    shl rax, 3
    mov r10, qword ptr [r15 + rax]
    add r10, r12
    mov r11, qword ptr [rbx + rax]
    mov rdx, r11
    cmp r9, r11
    jae cmp_start
    mov rdx, r9
cmp_start:
    xor rcx, rcx
cmp_loop:
    cmp rcx, rdx
    jae cmp_len
    movzx edi, byte ptr [r8 + rcx]
    movzx esi, byte ptr [r10 + rcx]
    cmp edi, esi
    jne cmp_diff
    inc rcx
    jmp cmp_loop
cmp_diff:
    cmp edi, esi
    jb cmp_less
    mov eax, 1
    jmp cmp_ret
cmp_less:
    mov eax, -1
    jmp cmp_ret
cmp_len:
    cmp r9, r11
    je cmp_eq
    jb cmp_less_len
    mov eax, 1
    jmp cmp_ret
cmp_less_len:
    mov eax, -1
    jmp cmp_ret
cmp_eq:
    xor eax, eax
cmp_ret:
    pop r11
    pop r10
    pop r9
    pop r8
    ret
write_all:
    test rdx, rdx
    jz wa_ret
wa_loop:
    test rdx, rdx
    jz wa_ret
    mov rax, 1
    syscall
    test rax, rax
    jle wa_ret
    add rsi, rax
    sub rdx, rax
    jmp wa_loop
wa_ret:
    ret
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
