.text
.globl _start
.equ IN_SIZE, 32768
_start:
    lea r14, [rip + inbuf]
    xor r12, r12
read_loop:
    cmp r12, IN_SIZE
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r14
    add rsi, r12
    mov rdx, IN_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit_prog
    lea r15, [rip + offs]
    lea rbx, [rip + lens]
    xor r13, r13
    xor r8, r8
    xor r9, r9
parse_loop:
    cmp r8, r12
    jae parse_end
    cmp byte ptr [r14 + r8], 10
    jne parse_next
    mov rax, r13
    shl rax, 3
    mov qword ptr [r15 + rax], r9
    mov rcx, r8
    sub rcx, r9
    mov qword ptr [rbx + rax], rcx
    inc r13
    mov r9, r8
    inc r9
parse_next:
    inc r8
    jmp parse_loop
parse_end:
    cmp r9, r12
    jae parsedone
    mov rax, r13
    shl rax, 3
    mov qword ptr [r15 + rax], r9
    mov rcx, r12
    sub rcx, r9
    mov qword ptr [rbx + rax], rcx
    inc r13
parsedone:
    cmp r13, 2
    jb sort_done
    mov r8, 1
sort_outer:
    cmp r8, r13
    jae sort_done
    mov rax, r8
    shl rax, 3
    mov r10, qword ptr [r15 + rax]
    mov r11, qword ptr [rbx + rax]
    mov r9, r8
    dec r9
sort_inner:
    cmp r9, 0
    jl sort_place
    mov rdi, r9
    mov rsi, r8
    call compare
    test eax, eax
    jle sort_place
    mov rax, r9
    shl rax, 3
    mov rcx, qword ptr [r15 + rax]
    mov qword ptr [r15 + rax + 8], rcx
    mov rcx, qword ptr [rbx + rax]
    mov qword ptr [rbx + rax + 8], rcx
    dec r9
    jmp sort_inner
sort_place:
    mov rax, r9
    inc rax
    shl rax, 3
    mov qword ptr [r15 + rax], r10
    mov qword ptr [rbx + rax], r11
    inc r8
    jmp sort_outer
sort_done:
    xor r12, r12
out_loop:
    cmp r12, r13
    jae out_done
    mov rax, r12
    shl rax, 3
    mov rsi, qword ptr [r15 + rax]
    add rsi, r14
    mov rdx, qword ptr [rbx + rax]
    mov rdi, 1
    call write_all
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    call write_all
    inc r12
    jmp out_loop
out_done:
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
compare:
    push r8
    push r9
    push r10
    push r11
    mov rax, rdi
    shl rax, 3
    mov r8, qword ptr [r15 + rax]
    add r8, r14
    mov r9, qword ptr [rbx + rax]
    mov rax, rsi
    shl rax, 3
    mov r10, qword ptr [r15 + rax]
    add r10, r14
    mov r11, qword ptr [rbx + rax]
    mov rdx, r9
    cmp r9, r11
    jb cmp_min_a
    mov rdx, r11
    jmp cmp_start
cmp_min_a:
    mov rdx, r9
cmp_start:
    xor rcx, rcx
cmp_loop:
    cmp rcx, rdx
    jae cmp_len
    movzx edi, byte ptr [r8 + rcx]
    movzx esi, byte ptr [r10 + rcx]
    cmp edi, esi
    jne cmp_diff
    inc rcx
    jmp cmp_loop
cmp_diff:
    cmp edi, esi
    jb cmp_less_diff
    mov eax, 1
    jmp cmp_ret
cmp_less_diff:
    mov eax, -1
    jmp cmp_ret
cmp_len:
    cmp r9, r11
    je cmp_equal
    jb cmp_less_len
    mov eax, 1
    jmp cmp_ret
cmp_less_len:
    mov eax, -1
    jmp cmp_ret
cmp_equal:
    xor eax, eax
cmp_ret:
    pop r11
    pop r10
    pop r9
    pop r8
    ret
write_all:
    test rdx, rdx
    jz wa_ret
wa_loop:
    test rdx, rdx
    jz wa_ret
    mov rax, 1
    syscall
    test rax, rax
    jle wa_ret
    add rsi, rax
    sub rdx, rax
    jmp wa_loop
wa_ret:
    ret
.section .rodata
.balign 8

.bss
.balign 8
