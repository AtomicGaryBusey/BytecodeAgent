.text
.globl _start
_start:
	xor r12, r12
read_loop:
	cmp r12, 65536
	jae read_done
	mov rax, 0
	xor rdi, rdi
	lea rsi, [rip + input_buf]
	add rsi, r12
	mov rdx, 65536
	sub rdx, r12
	syscall
	test rax, rax
	jle read_done
	add r12, rax
	jmp read_loop
read_done:
	xor rcx, rcx
find_loop:
	cmp rcx, r12
	jae no_bang
	lea rsi, [rip + input_buf]
	cmp byte ptr [rsi + rcx], 0x21
	je found_bang
	inc rcx
	jmp find_loop
found_bang:
	mov qword ptr [rip + prog_len_var], rcx
	lea rax, [rcx + 1]
	mov qword ptr [rip + input_start_var], rax
	mov rax, r12
	sub rax, qword ptr [rip + input_start_var]
	mov qword ptr [rip + input_len_var], rax
	jmp validate_init
no_bang:
	mov qword ptr [rip + prog_len_var], r12
	mov qword ptr [rip + input_start_var], r12
	mov qword ptr [rip + input_len_var], 0
validate_init:
	xor rbx, rbx
	xor rcx, rcx
check_loop:
	cmp rcx, qword ptr [rip + prog_len_var]
	jae check_done
	lea rsi, [rip + input_buf]
	mov al, byte ptr [rsi + rcx]
	cmp al, 0x5b
	je push_bracket
	cmp al, 0x5d
	je pop_bracket
	inc rcx
	jmp check_loop
push_bracket:
	lea rsi, [rip + match_stack]
	mov dword ptr [rsi + rbx*4], ecx
	inc rbx
	inc rcx
	jmp check_loop
pop_bracket:
	test rbx, rbx
	jz unbalanced
	dec rbx
	lea rsi, [rip + match_stack]
	mov edx, dword ptr [rsi + rbx*4]
	lea rsi, [rip + jump_table]
	mov dword ptr [rsi + rdx*4], ecx
	mov dword ptr [rsi + rcx*4], edx
	inc rcx
	jmp check_loop
check_done:
	test rbx, rbx
	jnz unbalanced
	xor rbx, rbx
	xor rbp, rbp
	xor r12, r12
	mov r13, qword ptr [rip + input_len_var]
	mov r14, qword ptr [rip + input_start_var]
	xor r15, r15
	lea r8, [rip + input_buf]
	lea r9, [rip + tape]
	lea r10, [rip + output_buf]
	lea r11, [rip + jump_table]
exec_loop:
	cmp rbx, qword ptr [rip + prog_len_var]
	jae exec_done
	mov al, byte ptr [r8 + rbx]
	cmp al, 0x2b
	je do_inc
	cmp al, 0x2d
	je do_dec
	cmp al, 0x3e
	je do_right
	cmp al, 0x3c
	je do_left
	cmp al, 0x2e
	je do_output
	cmp al, 0x2c
	je do_input
	cmp al, 0x5b
	je do_open
	cmp al, 0x5d
	je do_close
	inc rbx
	jmp exec_loop
do_inc:
	inc byte ptr [r9 + rbp]
	inc rbx
	jmp exec_loop
do_dec:
	dec byte ptr [r9 + rbp]
	inc rbx
	jmp exec_loop
do_right:
	inc rbp
	inc rbx
	jmp exec_loop
do_left:
	dec rbp
	inc rbx
	jmp exec_loop
do_output:
	mov dl, byte ptr [r9 + rbp]
	mov byte ptr [r10 + r15], dl
	inc r15
	inc rbx
	jmp exec_loop
do_input:
	cmp r12, r13
	jae input_eof
	mov rax, r14
	add rax, r12
	mov dl, byte ptr [r8 + rax]
	mov byte ptr [r9 + rbp], dl
	inc r12
	inc rbx
	jmp exec_loop
input_eof:
	mov byte ptr [r9 + rbp], 0
	inc rbx
	jmp exec_loop
do_open:
	cmp byte ptr [r9 + rbp], 0
	jne open_next
	mov ebx, dword ptr [r11 + rbx*4]
open_next:
	inc rbx
	jmp exec_loop
do_close:
	cmp byte ptr [r9 + rbp], 0
	je close_next
	mov ebx, dword ptr [r11 + rbx*4]
close_next:
	inc rbx
	jmp exec_loop
exec_done:
	xor rbx, rbx
write_loop:
	cmp rbx, r15
	jae exit_zero
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + output_buf]
	add rsi, rbx
	mov rdx, r15
	sub rdx, rbx
	syscall
	test rax, rax
	jle exit_zero
	add rbx, rax
	jmp write_loop
exit_zero:
	mov rax, 60
	xor rdi, rdi
	syscall
unbalanced:
	mov rax, 60
	mov rdi, 1
	syscall
.bss
.balign 8
prog_len_var: .zero 8
input_start_var: .zero 8
input_len_var: .zero 8
input_buf: .zero 65536
tape: .zero 30000
output_buf: .zero 65536
jump_table: .zero 262144
match_stack: .zero 262144
