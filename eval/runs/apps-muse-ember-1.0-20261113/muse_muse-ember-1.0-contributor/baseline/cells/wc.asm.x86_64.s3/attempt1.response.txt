.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 4096
    syscall
    test rax, rax
    jle print
    mov r9, rax
    lea rsi, [rip + buffer]
    xor r8, r8
proc_loop:
    cmp r8, r9
    jae read_loop
    movzx edx, byte ptr [rsi + r8]
    inc r14
    cmp dl, 10
    jne 1f
    inc r12
1:
    cmp dl, 32
    je 2f
    cmp dl, 9
    jb 3f
    cmp dl, 13
    ja 3f
    jmp 2f
3:
    test r15, r15
    jnz 4f
    inc r13
    mov r15, 1
    jmp 4f
2:
    xor r15, r15
4:
    inc r8
    jmp proc_loop
print:
    lea rbx, [rip + outbuf]
    mov rdi, r12
    mov rsi, rbx
    call append_num
    mov rbx, rsi
    mov byte ptr [rbx], 32
    inc rbx
    mov rdi, r13
    mov rsi, rbx
    call append_num
    mov rbx, rsi
    mov byte ptr [rbx], 32
    inc rbx
    mov rdi, r14
    mov rsi, rbx
    call append_num
    mov rbx, rsi
    mov byte ptr [rbx], 10
    inc rbx
    mov rdx, rbx
    lea rsi, [rip + outbuf]
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz exit_prog
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
append_num:
    lea r8, [rip + numtmp]
    xor r9, r9
    test rdi, rdi
    jnz ap_loop
    mov byte ptr [r8], 48
    mov r9, 1
    jmp ap_copy
ap_loop:
    mov rax, rdi
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, 48
    mov byte ptr [r8 + r9], dl
    inc r9
    mov rdi, rax
    test rdi, rdi
    jnz ap_loop
ap_copy:
    xor rcx, rcx
ap_copy_loop:
    cmp rcx, r9
    jae ap_copy_done
    mov rdx, r9
    dec rdx
    sub rdx, rcx
    mov al, byte ptr [r8 + rdx]
    mov byte ptr [rsi + rcx], al
    inc rcx
    jmp ap_copy_loop
ap_copy_done:
    add rsi, r9
    ret
.bss
.align 8
buffer:
    .zero 4096
outbuf:
    .zero 64
numtmp:
    .zero 32
