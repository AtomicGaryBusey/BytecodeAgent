.text
.globl _start
_start:
    xor eax, eax
    mov qword ptr [rip + inlen], rax
read_loop:
    mov rax, qword ptr [rip + inlen]
    lea rsi, [rip + inbuf]
    add rsi, rax
    mov rdx, 65536
    sub rdx, rax
    cmp rdx, 0
    je read_discard
    mov rdi, 0
    mov rax, 0
    syscall
    cmp rax, 0
    jle read_done
    add qword ptr [rip + inlen], rax
    jmp read_loop
read_discard:
    lea rsi, [rip + scratch]
    mov rdx, 1024
    mov rdi, 0
    mov rax, 0
    syscall
    cmp rax, 0
    jle read_done
    jmp read_discard
read_done:
    mov rbx, 0
row_loop:
    mov rax, qword ptr [rip + inlen]
    cmp rbx, rax
    jae print_total
    mov rcx, rax
    sub rcx, rbx
    cmp rcx, 16
    jae n_is_16
    mov r12, rcx
    jmp n_done
n_is_16:
    mov r12, 16
n_done:
    lea rdi, [rip + linebuf]
    mov r13, 7
off_loop:
    mov rax, rbx
    mov rcx, r13
    shl rcx, 2
    shr rax, cl
    and eax, 15
    lea rsi, [rip + hexdig]
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec r13
    cmp r13, 0
    jge off_loop
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    mov r13, 0
slot_loop:
    cmp r13, r12
    jae slot_empty
    lea rsi, [rip + inbuf]
    mov rax, rbx
    add rax, r13
    movzx eax, byte ptr [rsi + rax]
    mov r14, rax
    mov eax, r14d
    shr eax, 4
    lea rsi, [rip + hexdig]
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov eax, r14d
    and eax, 15
    lea rsi, [rip + hexdig]
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    jmp slot_next
slot_empty:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
slot_next:
    cmp r13, 7
    jne slot_noextra
    mov byte ptr [rdi], 32
    inc rdi
slot_noextra:
    inc r13
    cmp r13, 16
    jl slot_loop
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi
    mov r13, 0
char_loop:
    cmp r13, r12
    jae char_done
    lea rsi, [rip + inbuf]
    mov rax, rbx
    add rax, r13
    movzx eax, byte ptr [rsi + rax]
    cmp al, 32
    jb char_dot
    cmp al, 127
    jae char_dot
    mov byte ptr [rdi], al
    inc rdi
    jmp char_next
char_dot:
    mov byte ptr [rdi], 46
    inc rdi
char_next:
    inc r13
    jmp char_loop
char_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
write_row:
    cmp rdx, 0
    je write_row_done
    mov rdi, 1
    mov rax, 1
    syscall
    cmp rax, 0
    jle write_row_done
    add rsi, rax
    sub rdx, rax
    jmp write_row
write_row_done:
    add rbx, 16
    jmp row_loop
print_total:
    lea rdi, [rip + linebuf]
    mov r13, 7
tot_loop:
    mov rax, qword ptr [rip + inlen]
    mov rcx, r13
    shl rcx, 2
    shr rax, cl
    and eax, 15
    lea rsi, [rip + hexdig]
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [rdi], al
    inc rdi
    dec r13
    cmp r13, 0
    jge tot_loop
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
write_tot:
    cmp rdx, 0
    je do_exit
    mov rdi, 1
    mov rax, 1
    syscall
    cmp rax, 0
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_tot
do_exit:
    mov rdi, 0
    mov rax, 60
    syscall
.section .rodata
hexdig: .ascii "0123456789abcdef"
.bss
.balign 8
inlen: .zero 8
inbuf: .zero 65536
linebuf: .zero 256
scratch: .zero 1024
