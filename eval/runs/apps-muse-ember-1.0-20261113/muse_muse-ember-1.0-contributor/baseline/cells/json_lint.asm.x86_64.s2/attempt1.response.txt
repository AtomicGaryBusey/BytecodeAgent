.text
.globl _start
_start:
    xor r13d, r13d
    lea r14, [rip + inbuf]
.Lread_loop:
    cmp r13, 65536
    jae .Lread_done
    mov rax, 0
    xor edi, edi
    lea rsi, [r14 + r13]
    mov rdx, 65536
    sub rdx, r13
    syscall
    cmp rax, 0
    je .Lread_done
    jl .Lread_done
    add r13, rax
    jmp .Lread_loop
.Lread_done:
    xor r12d, r12d
    call parse_value
    call skip_ws
    cmp r12, r13
    jne .Lerror
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    mov rax, 60
    xor edi, edi
    syscall
.Lerror:
    lea rsi, [rip + err_prefix]
    lea rdi, [rip + outbuf]
    mov rcx, 9
    rep movsb
    sub rsp, 32
    lea r10, [rsp + 32]
    mov rax, r12
    test rax, rax
    jnz .Lconv_loop
    dec r10
    mov byte ptr [r10], 48
    jmp .Lconv_done
.Lconv_loop:
    test rax, rax
    jz .Lconv_done
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, 48
    dec r10
    mov byte ptr [r10], dl
    jmp .Lconv_loop
.Lconv_done:
    lea rdi, [rip + outbuf]
    add rdi, 9
    mov rsi, r10
    lea rax, [rsp + 32]
    sub rax, r10
    mov rcx, rax
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
skip_ws:
    cmp r12, r13
    jae .Lskip_ret
    movzx eax, byte ptr [r14 + r12]
    cmp al, 32
    je .Lskip_one
    cmp al, 9
    je .Lskip_one
    cmp al, 10
    je .Lskip_one
    cmp al, 13
    je .Lskip_one
    ret
.Lskip_one:
    inc r12
    jmp skip_ws
.Lskip_ret:
    ret
parse_value:
    call skip_ws
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je .Lpv_string
    cmp al, 116
    je .Lpv_true
    cmp al, 102
    je .Lpv_false
    cmp al, 110
    je .Lpv_null
    cmp al, 45
    je .Lpv_number
    cmp al, 48
    jb .Lerror
    cmp al, 57
    ja .Lerror
.Lpv_number:
    call parse_number
    ret
.Lpv_string:
    call parse_string
    ret
.Lpv_true:
    lea rsi, [rip + str_true]
    mov rcx, 4
    call parse_literal
    ret
.Lpv_false:
    lea rsi, [rip + str_false]
    mov rcx, 5
    call parse_literal
    ret
.Lpv_null:
    lea rsi, [rip + str_null]
    mov rcx, 4
    call parse_literal
    ret
parse_literal:
    xor edx, edx
.Llit_loop:
    cmp rdx, rcx
    jae .Llit_ret
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, byte ptr [rsi + rdx]
    jne .Lerror
    inc r12
    inc rdx
    jmp .Llit_loop
.Llit_ret:
    ret
parse_string:
    inc r12
.Lstr_loop:
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 34
    je .Lstr_end
    cmp al, 92
    je .Lstr_escape
    cmp al, 32
    jb .Lerror
    inc r12
    jmp .Lstr_loop
.Lstr_end:
    inc r12
    ret
.Lstr_escape:
    inc r12
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 34
    je .Lstr_esc_simple
    cmp al, 92
    je .Lstr_esc_simple
    cmp al, 47
    je .Lstr_esc_simple
    cmp al, 98
    je .Lstr_esc_simple
    cmp al, 102
    je .Lstr_esc_simple
    cmp al, 110
    je .Lstr_esc_simple
    cmp al, 114
    je .Lstr_esc_simple
    cmp al, 116
    je .Lstr_esc_simple
    cmp al, 117
    je .Lstr_unicode
    jmp .Lerror
.Lstr_esc_simple:
    inc r12
    jmp .Lstr_loop
.Lstr_unicode:
    inc r12
    mov rcx, 4
.Luni_loop:
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    jb .Lerror
    cmp al, 57
    jbe .Luni_ok
    cmp al, 97
    jb .Luni_upper
    cmp al, 102
    jbe .Luni_ok
    jmp .Lerror
.Luni_upper:
    cmp al, 65
    jb .Lerror
    cmp al, 70
    jbe .Luni_ok
    jmp .Lerror
.Luni_ok:
    inc r12
    dec rcx
    jnz .Luni_loop
    jmp .Lstr_loop
parse_number:
    movzx eax, byte ptr [r14 + r12]
    cmp al, 45
    jne .Lnum_int
    inc r12
.Lnum_int:
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    je .Lnum_zero
    cmp al, 49
    jb .Lerror
    cmp al, 57
    ja .Lerror
.Lnum_int_digits:
    inc r12
    cmp r12, r13
    jae .Lnum_after_int
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    jb .Lnum_after_int
    cmp al, 57
    ja .Lnum_after_int
    jmp .Lnum_int_digits
.Lnum_zero:
    inc r12
.Lnum_after_int:
    cmp r12, r13
    jae .Lnum_ret
    movzx eax, byte ptr [r14 + r12]
    cmp al, 46
    je .Lnum_frac
    cmp al, 101
    je .Lnum_exp
    cmp al, 69
    je .Lnum_exp
    ret
.Lnum_frac:
    inc r12
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    jb .Lerror
    cmp al, 57
    ja .Lerror
.Lnum_frac_digits:
    inc r12
    cmp r12, r13
    jae .Lnum_after_frac
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    jb .Lnum_after_frac
    cmp al, 57
    ja .Lnum_after_frac
    jmp .Lnum_frac_digits
.Lnum_after_frac:
    cmp r12, r13

    movzx eax, byte ptr [r14 + r12]

    je .Lnum_exp
    cmp al, 69
    je .Lnum_exp
    ret
.Lnum_exp:
    inc r12
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 43
    je .Lnum_exp_sign
    cmp al, 45
    je .Lnum_exp_sign
    jmp .Lnum_exp_digits_first
.Lnum_exp_sign:
    inc r12
.Lnum_exp_digits_first:
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    jb .Lerror
    cmp al, 57
    ja .Lerror
.Lnum_exp_digits:
    inc r12
    cmp r12, r13
    jae .Lnum_ret
    movzx eax, byte ptr [r14 + r12]
    cmp al, 48
    jb .Lnum_ret
    cmp al, 57

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 125
    je .Lobj_close
.Lobj_member:
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 34
    jne .Lerror
    call parse_string
    call skip_ws
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 58
    jne .Lerror
    inc r12
    call parse_value
    call skip_ws
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 44
    je .Lobj_comma
    cmp al, 125
    je .Lobj_close
    jmp .Lerror
.Lobj_comma:
    inc r12
    call skip_ws
    jmp .Lobj_member
.Lobj_close:
    inc r12
    ret
parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 93

.Larr_value:
    call parse_value
    call skip_ws
    cmp r12, r13
    jae .Lerror
    movzx eax, byte ptr [r14 + r12]
    cmp al, 44

    cmp al, 93
    je .Larr_close
    jmp .Lerror
.Larr_comma:

.Larr_close:
