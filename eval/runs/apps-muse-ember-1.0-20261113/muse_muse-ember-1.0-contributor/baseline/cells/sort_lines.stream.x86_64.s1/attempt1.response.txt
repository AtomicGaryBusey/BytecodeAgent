.equ IN_CAP, 32768
.equ MAX_LINES, 1000
.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, IN_CAP
    jae parse_start
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rdx, IN_CAP
    sub rdx, rbx
    syscall
    test rax, rax
    jle parse_start
    add rbx, rax
    jmp read_loop
parse_start:
    mov qword ptr [rip + in_len], rbx
    mov r15, rbx
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
parse_loop:
    cmp r13, r15
    jae parse_final
    lea rax, [rip + inbuf]
    movzx edx, byte ptr [rax + r13]
    cmp edx, 10
    je parse_newline
    inc r13
    jmp parse_loop
parse_newline:
    cmp r14, MAX_LINES
    jae parse_update
    lea rcx, [rip + line_start]
    mov qword ptr [rcx + r14*8], r12
    lea rcx, [rip + line_len]
    mov rax, r13
    sub rax, r12
    mov qword ptr [rcx + r14*8], rax
    inc r14
parse_update:
    mov r12, r13
    inc r12
    inc r13
    jmp parse_loop
parse_final:
    cmp r12, r15
    jae parse_done
    cmp r14, MAX_LINES
    jae parse_done
    lea rcx, [rip + line_start]
    mov qword ptr [rcx + r14*8], r12
    lea rcx, [rip + line_len]
    mov rax, r15
    sub rax, r12
    mov qword ptr [rcx + r14*8], rax
    inc r14
parse_done:
    mov qword ptr [rip + line_count], r14
    cmp r14, 2
    jb sort_done
    mov rbx, r14
    mov r12, 1
sort_i_loop:
    cmp r12, rbx
    jae sort_done
    lea rcx, [rip + line_start]
    mov r14, qword ptr [rcx + r12*8]
    lea rcx, [rip + line_len]
    mov r15, qword ptr [rcx + r12*8]
    mov r13, r12
    dec r13
sort_j_loop:
    cmp r13, 0
    jl sort_j_done
    lea rdi, [rip + inbuf]
    lea rcx, [rip + line_start]
    mov rax, qword ptr [rcx + r13*8]
    add rdi, rax
    lea rcx, [rip + line_len]
    mov rsi, qword ptr [rcx + r13*8]
    lea rdx, [rip + inbuf]
    add rdx, r14
    mov rcx, r15
    call compare_lines
    test rax, rax
    jle sort_j_done
    lea r8, [rip + line_start]
    mov r9, qword ptr [r8 + r13*8]
    mov r10, r13
    inc r10
    mov qword ptr [r8 + r10*8], r9
    lea r8, [rip + line_len]
    mov r9, qword ptr [r8 + r13*8]
    mov qword ptr [r8 + r10*8], r9
    dec r13
    jmp sort_j_loop
sort_j_done:
    mov r10, r13
    inc r10
    lea r8, [rip + line_start]
    mov qword ptr [r8 + r10*8], r14
    lea r8, [rip + line_len]
    mov qword ptr [r8 + r10*8], r15
    inc r12
    jmp sort_i_loop
sort_done:
    mov rbx, qword ptr [rip + line_count]
    xor r12d, r12d
out_loop:
    cmp r12, rbx
    jae out_done
    lea r8, [rip + line_len]
    mov rdx, qword ptr [r8 + r12*8]
    test rdx, rdx
    jz out_newline
    lea r8, [rip + line_start]
    mov r9, qword ptr [r8 + r12*8]
    lea rsi, [rip + inbuf]
    add rsi, r9
    mov edi, 1
    call write_all
out_newline:
    mov edi, 1
    lea rsi, [rip + nl_byte]
    mov edx, 1
    call write_all
    inc r12
    jmp out_loop
out_done:
    mov rax, 60
    xor edi, edi
    syscall
exit_loop:
    jmp exit_loop
compare_lines:
    mov r8, rsi
    cmp r8, rcx
    jbe cmp_min_done
    mov r8, rcx
cmp_min_done:
    xor r9d, r9d
cmp_qloop:
    mov rax, r9
    add rax, 8
    cmp rax, r8
    ja cmp_byte_loop
    mov rax, qword ptr [rdi + r9]
    mov r10, qword ptr [rdx + r9]
    mov r11, rax
    xor r11, r10
    test r11, r11
    jz cmp_qnext
    tzcnt r11, r11
    shr r11, 3
    add r11, r9
    movzx eax, byte ptr [rdi + r11]
    movzx r10d, byte ptr [rdx + r11]
    sub eax, r10d
    cdqe
    ret
cmp_qnext:
    add r9, 8
    jmp cmp_qloop
cmp_byte_loop:
    cmp r9, r8
    jae cmp_len
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    cmp eax, r10d
    jne cmp_diff
    inc r9
    jmp cmp_byte_loop
cmp_diff:
    sub eax, r10d
    cdqe
    ret
cmp_len:
    cmp rsi, rcx
    je cmp_equal
    jb cmp_less
    mov eax, 1
    ret
cmp_less:
    mov eax, -1
    ret
cmp_equal:
    xor eax, eax
    ret
write_all:
    mov r8, rsi
    mov r9, rdx
write_loop:
    test r9, r9
    jz write_done
    mov rax, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jle write_done
    add r8, rax
    sub r9, rax
    jmp write_loop
write_done:
    ret
.section .rodata
nl_byte:
    .byte 10
.bss
.balign 8
in_len:
    .space 8
line_count:
    .space 8
inbuf:
    .space 32768
line_start:
    .space 8000
line_len:
    .space 8000
