.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
.Lread:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + buffer]
    mov rdx, 8192
    syscall
    test rax, rax
    jle .Ldone
    add r14, rax
    mov rbx, rax
    lea rsi, [rip + buffer]
.Lbyte_loop:
    test rbx, rbx
    je .Lread
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne .Lnot_newline
    inc r12
.Lnot_newline:
    cmp al, 32
    je .Lis_ws
    cmp al, 9
    je .Lis_ws
    cmp al, 10
    je .Lis_ws
    cmp al, 11
    je .Lis_ws
    cmp al, 12
    je .Lis_ws
    cmp al, 13
    je .Lis_ws
    cmp r15, 0
    jne .Lnext_byte
    mov r15, 1
    inc r13
    jmp .Lnext_byte
.Lis_ws:
    xor r15, r15
.Lnext_byte:
    inc rsi
    dec rbx
    jmp .Lbyte_loop
.Ldone:
    lea rbp, [rip + outbuf]
    mov rsi, rbp
    mov rdi, r12
    call convert
    mov rbp, rsi
    mov byte ptr [rbp], 32
    inc rbp
    mov rsi, rbp
    mov rdi, r13
    call convert
    mov rbp, rsi
    mov byte ptr [rbp], 32
    inc rbp
    mov rsi, rbp
    mov rdi, r14
    call convert
    mov rbp, rsi
    mov byte ptr [rbp], 10
    inc rbp
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, rbp
    sub rdx, rsi
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
convert:
    cmp rdi, 0
    jne .Lconvert_nonzero
    mov byte ptr [rsi], 48
    inc rsi
    ret
.Lconvert_nonzero:
    sub rsp, 32
    mov r8, rsp
    xor r9, r9
    mov r10, 10
.Lconvert_loop:
    xor rdx, rdx
    mov rax, rdi
    div r10
    add dl, 48
    mov byte ptr [r8 + r9], dl
    inc r9
    mov rdi, rax
    cmp rax, 0
    jne .Lconvert_loop
.Lconvert_copy:
    dec r9
    mov dl, byte ptr [r8 + r9]
    mov byte ptr [rsi], dl
    inc rsi
    cmp r9, 0
    jne .Lconvert_copy
    add rsp, 32
    ret
.bss
    .balign 16
buffer:
    .zero 8192
outbuf:
    .zero 64
