.globl _start
.text
_start:
	xor rbx, rbx
	xor r12, r12
	xor r13, r13
	xor r14, r14
	lea r8, [rip + buf]
read_loop:
	mov rax, 0
	xor rdi, rdi
	mov rsi, r8
	mov rdx, 65536
	syscall
	test rax, rax
	jz read_done
	mov r15, rax
	xor rcx, rcx
byte_loop:
	cmp rcx, r15
	jae read_loop
	movzx r9d, byte ptr [r8 + rcx]
	inc r13
	cmp r9, 10
	jne not_newline
	inc rbx
not_newline:
	cmp r9, 32
	je is_ws
	cmp r9, 9
	jb is_word
	cmp r9, 13
	ja is_word
is_ws:
	xor r14, r14
	jmp next_byte
is_word:
	test r14, r14
	jnz next_byte
	inc r12
	mov r14, 1
next_byte:
	inc rcx
	jmp byte_loop
read_done:
	lea rdi, [rip + outbuf]
	mov rsi, rbx
	call print_num
	mov byte ptr [rdi], 32
	inc rdi
	mov rsi, r12
	call print_num
	mov byte ptr [rdi], 32
	inc rdi
	mov rsi, r13
	call print_num
	mov byte ptr [rdi], 10
	inc rdi
	lea rsi, [rip + outbuf]
	mov rdx, rdi
	sub rdx, rsi
	mov rdi, 1
	mov rax, 1
	syscall
	mov rax, 60
	xor rdi, rdi
	syscall
print_num:
	test rsi, rsi
	jnz pn_nonzero
	mov byte ptr [rdi], 48
	inc rdi
	ret
pn_nonzero:
	sub rsp, 32
	xor r8, r8
	mov r10, 10
pn_div:
	xor edx, edx
	mov rax, rsi
	div r10
	add dl, 48
	mov byte ptr [rsp + r8], dl
	inc r8
	mov rsi, rax
	test rsi, rsi
	jnz pn_div
pn_rev:
	test r8, r8
	jz pn_done
	dec r8
	mov al, byte ptr [rsp + r8]
	mov byte ptr [rdi], al
	inc rdi
	jmp pn_rev
pn_done:
	add rsp, 32
	ret
.bss
.align 8
buf:
	.zero 65536
outbuf:
	.zero 64
