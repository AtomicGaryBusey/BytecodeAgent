.globl _start
.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.bss
inbuf: .zero 49152
outbuf: .zero 70000
.text
_start:
	xor r12d, r12d
read_loop:
	mov rax, 0
	xor rdi, rdi
	lea rsi, [rip + inbuf]
	add rsi, r12
	mov rdx, 49152
	sub rdx, r12
	test rdx, rdx
	jz read_done
	syscall
	test rax, rax
	jz read_done
	add r12, rax
	jmp read_loop
read_done:
	lea r14, [rip + outbuf]
	xor r15d, r15d
	xor r13d, r13d
	xor ebx, ebx
encode_loop:
	mov rcx, r12
	sub rcx, rbx
	test rcx, rcx
	jz encode_done
	cmp rcx, 3
	jae do_full
	cmp rcx, 2
	je do_two
do_one:
	movzx r9d, byte ptr [inbuf + rbx]
	shl r9d, 16
	mov eax, r9d
	shr eax, 18
	and eax, 63
	call emit
	mov eax, r9d
	shr eax, 12
	and eax, 63
	call emit
	mov eax, 61
	call emit_ascii
	mov eax, 61
	call emit_ascii
	jmp encode_done
do_two:
	movzx r9d, byte ptr [inbuf + rbx]
	shl r9d, 16
	movzx r10d, byte ptr [inbuf + rbx + 1]
	shl r10d, 8
	or r9d, r10d
	mov eax, r9d
	shr eax, 18
	and eax, 63
	call emit
	mov eax, r9d
	shr eax, 12
	and eax, 63
	call emit
	mov eax, r9d
	shr eax, 6
	and eax, 63
	call emit
	mov eax, 61
	call emit_ascii
	jmp encode_done
do_full:
	movzx r9d, byte ptr [inbuf + rbx]
	shl r9d, 16
	movzx r10d, byte ptr [inbuf + rbx + 1]
	shl r10d, 8
	or r9d, r10d
	movzx r8d, byte ptr [inbuf + rbx + 2]
	or r9d, r8d
	mov eax, r9d
	shr eax, 18
	and eax, 63
	call emit
	mov eax, r9d
	shr eax, 12
	and eax, 63
	call emit
	mov eax, r9d
	shr eax, 6
	and eax, 63
	call emit
	mov eax, r9d
	and eax, 63
	call emit
	add rbx, 3
	jmp encode_loop
encode_done:
	test r15, r15
	jz do_exit
	test r13d, r13d
	jz do_write
	mov byte ptr [r14 + r15], 10
	inc r15
do_write:
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + outbuf]
	mov rdx, r15
	syscall
do_exit:
	mov rax, 60
	xor rdi, rdi
	syscall
emit:
	cmp r13d, 76
	jb emit_no_nl
	mov byte ptr [r14 + r15], 10
	inc r15
	xor r13d, r13d
emit_no_nl:
	movzx ecx, byte ptr [alphabet + rax]
	mov byte ptr [r14 + r15], cl
	inc r15
	inc r13
	ret
emit_ascii:
	cmp r13d, 76
	jb emit_ascii_no_nl
	mov byte ptr [r14 + r15], 10
	inc r15
	xor r13d, r13d
emit_ascii_no_nl:
	mov byte ptr [r14 + r15], al
	inc r15
	inc r13
	ret
