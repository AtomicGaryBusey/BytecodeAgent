.globl _start
.text
_start:
 xor r12, r12
 xor r13, r13
 xor r14, r14
 xor r15, r15
 lea rbx, [rip + buffer]
read_loop:
 mov rax, 0
 mov rdi, 0
 mov rsi, rbx
 mov rdx, 4096
 syscall
 cmp rax, 0
 jle read_done
 mov r10, rax
 xor r8, r8
byte_loop:
 cmp r8, r10
 jae read_loop
 movzx eax, byte ptr [rbx + r8]
 cmp al, 10
 jne not_newline
 inc r12
not_newline:
 cmp al, 32
 je is_ws
 cmp al, 9
 je is_ws
 cmp al, 10
 je is_ws
 cmp al, 11
 je is_ws
 cmp al, 12
 je is_ws
 cmp al, 13
 je is_ws
 cmp r15, 0
 jne byte_next
 inc r13
 mov r15, 1
 jmp byte_next
is_ws:
 xor r15, r15
byte_next:
 inc r8
 jmp byte_loop
read_done:
 lea rsi, [rip + outbuf]
 mov rax, r12
 call convert
 mov byte ptr [rsi], 32
 inc rsi
 mov rax, r13
 call convert
 mov byte ptr [rsi], 32
 inc rsi
 mov rax, r14
 call convert
 mov byte ptr [rsi], 10
 inc rsi
 lea rdi, [rip + outbuf]
 mov rdx, rsi
 sub rdx, rdi
 mov rsi, rdi
write_loop:
 cmp rdx, 0
 je write_done
 mov rax, 1
 mov rdi, 1
 syscall
 cmp rax, 0
 jle write_done
 add rsi, rax
 sub rdx, rax
 jmp write_loop
write_done:
 mov rax, 60
 xor rdi, rdi
 syscall
exit_loop:
 jmp exit_loop
convert:
 sub rsp, 32
 mov r11, rsp
 xor r10, r10
 cmp rax, 0
 jne conv_loop
 mov byte ptr [r11], 48
 mov r10, 1
 jmp conv_copy
conv_loop:
 cmp rax, 0
 je conv_copy
 xor rdx, rdx
 mov r9, 10
 div r9
 add dl, 48
 mov byte ptr [r11 + r10], dl
 inc r10
 jmp conv_loop
conv_copy:
 cmp r10, 0
 je conv_ret
 dec r10
 mov al, byte ptr [r11 + r10]
 mov byte ptr [rsi], al
 inc rsi
 jmp conv_copy
conv_ret:
 add rsp, 32
 ret
.bss
buffer:
 .zero 4096
outbuf:
 .zero 128
