.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r12, rbx
    mov r13, rbx
    xor r14, r14
    xor ecx, ecx
find_loop:
    cmp rcx, rbx
    jae find_done
    lea rsi, [rip + input_buf]
    add rsi, rcx
    movzx eax, byte ptr [rsi]
    cmp eax, 33
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r12, rcx
    mov r13, rcx
    inc r13
    mov r14, rbx
    sub r14, rcx
    dec r14
find_done:
    xor r15, r15
    xor ecx, ecx
check_loop:
    cmp rcx, r12
    jae check_done
    lea rsi, [rip + input_buf]
    add rsi, rcx
    movzx eax, byte ptr [rsi]
    cmp eax, 91
    je push_bracket
    cmp eax, 93
    je pop_bracket
    inc rcx
    jmp check_loop
push_bracket:
    lea rsi, [rip + bstack]
    mov rdx, r15
    shl rdx, 2
    add rsi, rdx
    mov dword ptr [rsi], ecx
    inc r15
    inc rcx
    jmp check_loop
pop_bracket:
    cmp r15, 0
    je exit_fail
    dec r15
    lea rsi, [rip + bstack]
    mov rdx, r15
    shl rdx, 2
    add rsi, rdx
    mov eax, dword ptr [rsi]
    lea rsi, [rip + jump_tab]
    mov rdx, rax
    shl rdx, 2
    add rsi, rdx
    mov dword ptr [rsi], ecx
    lea rsi, [rip + jump_tab]
    mov rdx, rcx
    shl rdx, 2
    add rsi, rdx
    mov dword ptr [rsi], eax
    inc rcx
    jmp check_loop
check_done:
    cmp r15, 0
    jne exit_fail
    xor eax, eax
    mov qword ptr [rip + pc_v], rax
    mov qword ptr [rip + ptr_v], rax
    mov qword ptr [rip + inpos_v], rax
exec_loop:
    mov rax, qword ptr [rip + pc_v]
    cmp rax, r12
    jae exec_done
    lea rsi, [rip + input_buf]
    add rsi, rax
    movzx eax, byte ptr [rsi]
    cmp eax, 62
    je cmd_gt
    cmp eax, 60
    je cmd_lt
    cmp eax, 43
    je cmd_plus
    cmp eax, 45
    je cmd_minus
    cmp eax, 46
    je cmd_dot
    cmp eax, 44
    je cmd_comma
    cmp eax, 91
    je cmd_open
    cmp eax, 93
    je cmd_close
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_gt:
    mov rax, qword ptr [rip + ptr_v]
    inc rax
    mov qword ptr [rip + ptr_v], rax
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_lt:
    mov rax, qword ptr [rip + ptr_v]
    dec rax
    mov qword ptr [rip + ptr_v], rax
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_plus:
    mov rax, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, rax
    movzx eax, byte ptr [rsi]
    inc eax
    mov byte ptr [rsi], al
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_minus:
    mov rax, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, rax
    movzx eax, byte ptr [rsi]
    dec eax
    mov byte ptr [rsi], al
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_dot:
    mov rax, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, rax
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_comma:
    mov rax, qword ptr [rip + inpos_v]
    cmp rax, r14
    jae comma_eof
    mov r9, rax
    lea rsi, [rip + input_buf]
    add rsi, r13
    add rsi, r9
    movzx eax, byte ptr [rsi]
    mov r10, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, r10
    mov byte ptr [rsi], al
    mov rax, qword ptr [rip + inpos_v]
    inc rax
    mov qword ptr [rip + inpos_v], rax
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
comma_eof:
    mov rax, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, rax
    xor eax, eax
    mov byte ptr [rsi], al
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_open:
    mov rax, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, rax
    movzx eax, byte ptr [rsi]
    cmp eax, 0
    je open_jump
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
open_jump:
    mov rax, qword ptr [rip + pc_v]
    lea rsi, [rip + jump_tab]
    mov rdx, rax
    shl rdx, 2
    add rsi, rdx
    mov eax, dword ptr [rsi]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
cmd_close:
    mov rax, qword ptr [rip + ptr_v]
    lea rsi, [rip + tape]
    add rsi, rax
    movzx eax, byte ptr [rsi]
    cmp eax, 0
    jne close_jump
    mov rax, qword ptr [rip + pc_v]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
close_jump:
    mov rax, qword ptr [rip + pc_v]
    lea rsi, [rip + jump_tab]
    mov rdx, rax
    shl rdx, 2
    add rsi, rdx
    mov eax, dword ptr [rsi]
    inc rax
    mov qword ptr [rip + pc_v], rax
    jmp exec_loop
exec_done:
    mov rax, 60
    mov rdi, 0
    syscall
    jmp hang
exit_fail:
    mov rax, 60
    mov rdi, 1
    syscall
hang:
    jmp hang
