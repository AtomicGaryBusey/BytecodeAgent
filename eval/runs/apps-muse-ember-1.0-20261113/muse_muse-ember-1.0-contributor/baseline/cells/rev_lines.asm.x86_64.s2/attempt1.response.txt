.globl _start
.text
_start:
    lea r13, [rip + inbuf]
    lea r14, [rip + outbuf]
    xor r12, r12
read_loop:
    cmp r12, 65536
    jae read_done
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    mov rax, 0
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r15, r15
    xor rbx, rbx
    xor r10, r10
    cmp r12, 0
    je write_out
scan_loop:
    cmp r10, r12
    jae final_line
    movzx eax, byte ptr [r13 + r10]
    cmp al, 10
    jne next_char
    mov r9, r10
    sub r9, rbx
    xor r8, r8
rev_loop1:
    cmp r8, r9
    jae rev_done1
    mov rcx, r10
    sub rcx, r8
    dec rcx
    movzx eax, byte ptr [r13 + rcx]
    mov byte ptr [r14 + r15], al
    inc r15
    inc r8
    jmp rev_loop1
rev_done1:
    mov al, 10
    mov byte ptr [r14 + r15], al
    inc r15
    inc r10
    mov rbx, r10
    jmp scan_loop
next_char:
    inc r10
    jmp scan_loop
final_line:
    cmp rbx, r12
    jae write_out
    mov r9, r12
    sub r9, rbx
    xor r8, r8
rev_loop2:
    cmp r8, r9
    jae rev_done2
    mov rcx, r12
    sub rcx, r8
    dec rcx
    movzx eax, byte ptr [r13 + rcx]
    mov byte ptr [r14 + r15], al
    inc r15
    inc r8
    jmp rev_loop2
rev_done2:
    mov al, 10
    mov byte ptr [r14 + r15], al
    inc r15
write_out:
    cmp r15, 0
    je do_exit
write_loop:
    mov rdi, 1
    mov rsi, r14
    mov rdx, r15
    mov rax, 1
    syscall
    cmp rax, 0
    jle do_exit
    add r14, rax
    sub r15, rax
    cmp r15, 0
    jne write_loop
do_exit:
    mov rdi, 0
    mov rax, 60
    syscall
.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
