.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.globl _start
.text
_start:
    xor r12d, r12d
    lea r13, [rip + inbuf]
    lea rbx, [rip + linebuf]
read_loop:
    mov rax, SYS_read
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, 65536
    jae read_done
    jmp read_loop
read_done:
    xor r14d, r14d
row_loop:
    cmp r14, r12
    jae final
    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe n_ready
    mov r15, 16
n_ready:
    xor ebp, ebp
    lea rsi, [rip + hextab]
    mov rax, r14
    shr rax, 28
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    shr rax, 24
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    shr rax, 20
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    shr rax, 16
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    shr rax, 12
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    shr rax, 8
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    shr rax, 4
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r14
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov byte ptr [rbx + rbp], 32
    inc rbp
    mov byte ptr [rbx + rbp], 32
    inc rbp
    xor r10d, r10d
byte_loop:
    cmp r10, 16
    jae byte_done
    cmp r10, r15
    jae empty_slot
    mov rax, r14
    add rax, r10
    movzx rax, byte ptr [r13 + rax]
    mov rdx, rax
    shr rdx, 4
    and rdx, 15
    movzx rdx, byte ptr [rsi + rdx]
    mov byte ptr [rbx + rbp], dl
    inc rbp
    mov rdx, rax
    and rdx, 15
    movzx rdx, byte ptr [rsi + rdx]
    mov byte ptr [rbx + rbp], dl
    inc rbp
    mov byte ptr [rbx + rbp], 32
    inc rbp
    jmp after_slot
empty_slot:
    mov byte ptr [rbx + rbp], 32
    inc rbp
    mov byte ptr [rbx + rbp], 32
    inc rbp
    mov byte ptr [rbx + rbp], 32
    inc rbp
after_slot:
    cmp r10, 7
    jne no_extra
    mov byte ptr [rbx + rbp], 32
    inc rbp
no_extra:
    inc r10
    jmp byte_loop
byte_done:
    mov byte ptr [rbx + rbp], 32
    inc rbp
    mov byte ptr [rbx + rbp], 124
    inc rbp
    xor r10d, r10d
char_loop:
    cmp r10, r15
    jae char_done
    mov rax, r14
    add rax, r10
    movzx rax, byte ptr [r13 + rax]
    cmp rax, 32
    jb dot
    cmp rax, 126
    ja dot
    mov byte ptr [rbx + rbp], al
    inc rbp
    inc r10
    jmp char_loop
dot:
    mov byte ptr [rbx + rbp], 46
    inc rbp
    inc r10
    jmp char_loop
char_done:
    mov byte ptr [rbx + rbp], 124
    inc rbp
    mov byte ptr [rbx + rbp], 10
    inc rbp
    mov rsi, rbx
    mov rdx, rbp
    call write_all
    add r14, r15
    jmp row_loop
final:
    xor ebp, ebp
    lea rsi, [rip + hextab]
    mov rax, r12
    shr rax, 28
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    shr rax, 24
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    shr rax, 20
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    shr rax, 16
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    shr rax, 12
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    shr rax, 8
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    shr rax, 4
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov rax, r12
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [rbx + rbp], al
    inc rbp
    mov byte ptr [rbx + rbp], 10
    inc rbp
    mov rsi, rbx
    mov rdx, rbp
    call write_all
    mov rax, SYS_exit
    mov rdi, 0
    syscall
write_all:
    mov r8, rdx
    mov r9, rsi
write_loop:
    cmp r8, 0
    je write_ret
    mov rax, SYS_write
    mov rdi, 1
    mov rsi, r9
    mov rdx, r8
    syscall
    cmp rax, 0
    jle write_ret
    add r9, rax
    sub r8, rax
    jmp write_loop
write_ret:
    ret
.section .rodata
hextab:
    .ascii "0123456789abcdef"
.bss
inbuf:
    .zero 65536
linebuf:
    .zero 128
