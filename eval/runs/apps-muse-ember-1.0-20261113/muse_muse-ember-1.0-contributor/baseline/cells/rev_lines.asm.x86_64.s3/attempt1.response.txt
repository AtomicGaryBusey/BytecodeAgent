.text
.globl _start
_start:
    xor rbx, rbx
    lea r15, [rip + inbuf]
    lea r8, [rip + outbuf]
read_loop:
    cmp rbx, 65536
    jge process
    mov rax, 0
    xor rdi, rdi
    mov rsi, r15
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle process
    add rbx, rax
    jmp read_loop
process:
    xor r12, r12
    xor r13, r13
line_loop:
    cmp r12, rbx
    jge write_out
    mov r14, r12
scan_loop:
    cmp r14, rbx
    jge copy_start
    cmp byte ptr [r15 + r14], 10
    je copy_start
    inc r14
    jmp scan_loop
copy_start:
    mov r10, r14
    dec r10
    mov r11, r13
copy_loop:
    cmp r10, r12
    jl copy_done
    mov al, byte ptr [r15 + r10]
    mov byte ptr [r8 + r11], al
    dec r10
    inc r11
    jmp copy_loop
copy_done:
    mov byte ptr [r8 + r11], 10
    inc r11
    mov r13, r11
    cmp r14, rbx
    jge write_out
    mov r12, r14
    inc r12
    jmp line_loop
write_out:
    xor r10, r10
write_loop:
    cmp r10, r13
    jge exit_prog
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    add rsi, r10
    mov rdx, r13
    sub rdx, r10
    syscall
    test rax, rax
    jle exit_prog
    add r10, rax
    jmp write_loop
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
hang:
    jmp hang
.bss
.balign 8
inbuf:
    .space 65536
outbuf:
    .space 65537
