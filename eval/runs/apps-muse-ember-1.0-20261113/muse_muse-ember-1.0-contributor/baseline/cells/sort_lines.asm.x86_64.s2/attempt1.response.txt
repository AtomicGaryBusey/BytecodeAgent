.globl _start
.section .rodata
nl: .byte 10
.bss
.balign 8
inbuf: .zero 32768
offs: .zero 8000
lens: .zero 8000
.text
_start:
  cld
  xor r12d, r12d
  xor r13d, r13d
read_loop:
  cmp r12, 32768
  jae parse
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + inbuf]
  add rsi, r12
  mov rdx, 32768
  sub rdx, r12
  syscall
  test rax, rax
  jle parse
  add r12, rax
  jmp read_loop
parse:
  xor r13d, r13d
  test r12, r12
  jz sort_start
  xor r14d, r14d
  xor r15d, r15d
parse_loop:
  cmp r14, r12
  jae parse_final
  lea rax, [rip + inbuf]
  cmp byte ptr [rax + r14], 10
  jne parse_inc
  lea rax, [rip + offs]
  mov qword ptr [rax + r13*8], r15
  mov rdx, r14
  sub rdx, r15
  lea rax, [rip + lens]
  mov qword ptr [rax + r13*8], rdx
  inc r13
  inc r14
  mov r15, r14
  jmp parse_loop
parse_inc:
  inc r14
  jmp parse_loop
parse_final:
  cmp r15, r12
  jae sort_start
  lea rax, [rip + offs]
  mov qword ptr [rax + r13*8], r15
  mov rdx, r12
  sub rdx, r15
  lea rax, [rip + lens]
  mov qword ptr [rax + r13*8], rdx
  inc r13
sort_start:
  mov r12, r13
  cmp r12, 2
  jb output
  mov rbx, 1
i_loop:
  cmp rbx, r12
  jae output
  lea rax, [rip + offs]
  mov r15, qword ptr [rax + rbx*8]
  lea rax, [rip + lens]
  mov r13, qword ptr [rax + rbx*8]
  mov r14, rbx
  dec r14
j_loop:
  cmp r14, 0
  jl j_done
  lea rax, [rip + offs]
  mov rdi, qword ptr [rax + r14*8]
  mov rdx, r15
  lea rax, [rip + lens]
  mov rsi, qword ptr [rax + r14*8]
  mov rcx, r13
  call compare
  cmp eax, 0
  jle j_done
  lea rax, [rip + offs]
  mov r10, qword ptr [rax + r14*8]
  mov qword ptr [rax + r14*8 + 8], r10
  lea rax, [rip + lens]
  mov r10, qword ptr [rax + r14*8]
  mov qword ptr [rax + r14*8 + 8], r10
  dec r14
  jmp j_loop
j_done:
  lea rax, [rip + offs]
  mov qword ptr [rax + r14*8 + 8], r15
  lea rax, [rip + lens]
  mov qword ptr [rax + r14*8 + 8], r13
  inc rbx
  jmp i_loop
output:
  xor ebx, ebx
out_loop:
  cmp rbx, r12
  jae exit_prog
  lea rax, [rip + offs]
  mov r14, qword ptr [rax + rbx*8]
  lea rax, [rip + lens]
  mov r15, qword ptr [rax + rbx*8]
  test r15, r15
  jz out_nl
  lea rsi, [rip + inbuf]
  add rsi, r14
  mov rdx, r15
out_write_loop:
  test rdx, rdx
  jz out_nl
  mov rax, 1
  mov rdi, 1
  syscall
  test rax, rax
  jle out_nl
  add rsi, rax
  sub rdx, rax
  jmp out_write_loop
out_nl:
  lea rsi, [rip + nl]
  mov rdx, 1
out_nl_loop:
  test rdx, rdx
  jz out_next
  mov rax, 1
  mov rdi, 1
  syscall
  test rax, rax
  jle out_next
  add rsi, rax
  sub rdx, rax
  jmp out_nl_loop
out_next:
  inc rbx
  jmp out_loop
exit_prog:
  mov rax, 60
  xor rdi, rdi
  syscall
  jmp exit_prog
compare:
  mov r8, rsi
  cmp rcx, rsi
  jae cmp_min_done
  mov r8, rcx
cmp_min_done:
  xor r9d, r9d
  lea r10, [rip + inbuf]
  add r10, rdi
  lea r11, [rip + inbuf]
  add r11, rdx
cmp_loop:
  cmp r9, r8
  jae cmp_len_done
  movzx eax, byte ptr [r10 + r9]
  movzx edx, byte ptr [r11 + r9]
  cmp eax, edx
  jb cmp_less
  ja cmp_greater
  inc r9
  jmp cmp_loop
cmp_less:
  mov eax, -1
  ret
cmp_greater:
  mov eax, 1
  ret
cmp_len_done:
  cmp rsi, rcx
  jb cmp_less
  ja cmp_greater
  xor eax, eax
  ret
