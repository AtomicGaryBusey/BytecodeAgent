.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ IN_SIZE, 65536
.globl _start
.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x20, #0
    mov x22, #0
read_loop:
    mov x9, #IN_SIZE
    cmp x20, x9
    b.ge read_done
    mov x8, #SYS_READ
    mov x0, #0
    add x1, x19, x20
    mov x9, #IN_SIZE
    sub x2, x9, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x23, #0
    mov x24, #0
scan_loop:
    cmp x23, x20
    b.ge final_check
    ldrb w9, [x19, x23]
    cmp w9, #10
    b.ne next_byte
    sub x10, x23, #1
rev_loop:
    cmp x10, x24
    b.lt rev_done
    ldrb w11, [x19, x10]
    strb w11, [x21, x22]
    add x22, x22, #1
    sub x10, x10, #1
    b rev_loop
rev_done:
    mov w11, #10
    strb w11, [x21, x22]
    add x22, x22, #1
    add x24, x23, #1
    add x23, x23, #1
    b scan_loop
next_byte:
    add x23, x23, #1
    b scan_loop
final_check:
    cmp x24, x20
    b.ge write_out
    sub x10, x20, #1
final_rev_loop:
    cmp x10, x24
    b.lt final_rev_done
    ldrb w11, [x19, x10]
    strb w11, [x21, x22]
    add x22, x22, #1
    sub x10, x10, #1
    b final_rev_loop
final_rev_done:
    mov w11, #10
    strb w11, [x21, x22]
    add x22, x22, #1
write_out:
    mov x25, #0
write_loop:
    cmp x25, x22
    b.ge exit_prog
    mov x8, #SYS_WRITE
    mov x0, #1
    add x1, x21, x25
    sub x2, x22, x25
    svc #0
    cmp x0, #0
    b.le exit_prog
    add x25, x25, x0
    b write_loop
exit_prog:
    mov x8, #SYS_EXIT
    mov x0, #0
    svc #0
exit_loop:
    b exit_loop
.bss
.balign 8
inbuf:
    .space 65536
outbuf:
    .space 65537
