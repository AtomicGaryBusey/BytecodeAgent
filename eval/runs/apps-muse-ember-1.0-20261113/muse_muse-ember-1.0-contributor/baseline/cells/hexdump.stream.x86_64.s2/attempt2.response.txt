.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
inbuf:
    .space 65536
outbuf:
    .space 128
.text
.globl _start
_start:
    xor r12, r12
    lea r13, [rip + inbuf]
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r14, r14
    cmp r14, r12
    jae final
row_loop:
    lea r15, [rip + outbuf]
    lea rbx, [rip + hexdigits]
    lea r10, [r13 + r14]
    mov r8, 0
    mov r9, 28
offset_loop:
    cmp r8, 8
    jae offset_done
    mov rax, r14
    mov rcx, r9
    shr rax, cl
    and rax, 15
    mov rcx, rax
    movzx edx, byte ptr [rbx + rcx]
    mov byte ptr [r15 + r8], dl
    inc r8
    sub r9, 4
    jmp offset_loop
offset_done:
    mov byte ptr [r15 + 8], 32
    mov byte ptr [r15 + 9], 32
    mov r9, r12
    sub r9, r14
    cmp r9, 16
    jbe n_done
    mov r9, 16
n_done:
    xor r8, r8
    mov rbp, 10
bytes_loop:
    cmp r8, 16
    jae bytes_done
    cmp r8, r9
    jae space_slot
    movzx eax, byte ptr [r10 + r8]
    mov ecx, eax
    shr ecx, 4
    movzx edx, byte ptr [rbx + rcx]
    mov byte ptr [r15 + rbp], dl
    and eax, 15
    mov ecx, eax
    movzx edx, byte ptr [rbx + rcx]
    mov byte ptr [r15 + rbp + 1], dl
    mov byte ptr [r15 + rbp + 2], 32
    jmp slot_adv
space_slot:
    mov byte ptr [r15 + rbp], 32
    mov byte ptr [r15 + rbp + 1], 32
    mov byte ptr [r15 + rbp + 2], 32
slot_adv:
    inc r8
    add rbp, 3
    cmp r8, 8
    jne bytes_loop
    mov byte ptr [r15 + rbp], 32
    inc rbp
    jmp bytes_loop
bytes_done:
    mov byte ptr [r15 + rbp], 32
    inc rbp
    mov byte ptr [r15 + rbp], 124
    inc rbp
    xor r8, r8
ascii_loop:
    cmp r8, r9
    jae ascii_done
    movzx eax, byte ptr [r10 + r8]
    cmp eax, 32
    jb ascii_dot
    cmp eax, 126
    ja ascii_dot
    mov byte ptr [r15 + rbp], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [r15 + rbp], 46
ascii_next:
    inc r8
    inc rbp
    jmp ascii_loop
ascii_done:
    mov byte ptr [r15 + rbp], 124
    inc rbp
    mov byte ptr [r15 + rbp], 10
    inc rbp
    mov rsi, r15
    mov rdx, rbp
write_row_loop:
    cmp rdx, 0
    je write_row_done
    mov rax, 1
    mov rdi, 1
    syscall
    cmp rax, 0
    jle write_row_done
    add rsi, rax
    sub rdx, rax
    jmp write_row_loop
write_row_done:
    add r14, r9
    cmp r14, r12
    jb row_loop
final:
    lea r15, [rip + outbuf]
    lea rbx, [rip + hexdigits]
    mov r8, 0
    mov r9, 28
final_loop:
    cmp r8, 8
    jae final_done
    mov rax, r12
    mov rcx, r9
    shr rax, cl
    and rax, 15
    mov rcx, rax
    movzx edx, byte ptr [rbx + rcx]
    mov byte ptr [r15 + r8], dl
    inc r8
    sub r9, 4
    jmp final_loop
final_done:
    mov byte ptr [r15 + 8], 10
    mov rsi, r15
    mov rdx, 9
final_write_loop:
    cmp rdx, 0
    je final_write_done
    mov rax, 1
    mov rdi, 1
    syscall
    cmp rax, 0
    jle final_write_done
    add rsi, rax
    sub rdx, rax
    jmp final_write_loop
final_write_done:
    mov rax, 60
    xor rdi, rdi
    syscall
