.globl _start
.text
_start:
    lea r13, [rip + inbuf]
    xor r12, r12
read_loop:
    cmp r12, 65536
    je read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r15, [rip + linebuf]
    xor r14, r14
row_loop:
    cmp r14, r12
    jae final_line
    mov rbx, r12
    sub rbx, r14
    cmp rbx, 16
    jbe n_ready
    mov rbx, 16
n_ready:
    mov rdi, r15
    xor r8, r8
off_loop:
    cmp r8, 8
    jae off_done
    mov rcx, 7
    sub rcx, r8
    shl rcx, 2
    mov rax, r14
    shr rax, cl
    and rax, 15
    mov r10, rax
    lea rsi, [rip + hexdigits]
    mov al, byte ptr [rsi + r10]
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp off_loop
off_done:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    xor r8, r8
hex_loop:
    cmp r8, 16
    jae hex_done
    cmp r8, rbx
    jae hex_blank
    mov rcx, r14
    add rcx, r8
    movzx eax, byte ptr [r13 + rcx]
    mov r10, rax
    mov r9, r10
    shr r9, 4
    and r9, 15
    lea rsi, [rip + hexdigits]
    mov r11, r9
    mov al, byte ptr [rsi + r11]
    mov byte ptr [rdi], al
    inc rdi
    mov r11, r10
    and r11, 15
    mov al, byte ptr [rsi + r11]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    jmp hex_next
hex_blank:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
hex_next:
    cmp r8, 7
    jne hex_inc
    mov byte ptr [rdi], 32
    inc rdi
hex_inc:
    inc r8
    jmp hex_loop
hex_done:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi
    xor r8, r8
ascii_loop:
    cmp r8, rbx
    jae ascii_done
    mov rcx, r14
    add rcx, r8
    movzx eax, byte ptr [r13 + rcx]
    cmp rax, 32
    jb ascii_dot
    cmp rax, 126
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_next
ascii_dot:
    mov byte ptr [rdi], 46
ascii_next:
    inc rdi
    inc r8
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, r15
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    syscall
    add r14, rbx
    jmp row_loop
final_line:
    mov rdi, r15
    xor r8, r8
final_loop:
    cmp r8, 8
    jae final_done
    mov rcx, 7
    sub rcx, r8
    shl rcx, 2
    mov rax, r12
    shr rax, cl
    and rax, 15
    mov r10, rax
    lea rsi, [rip + hexdigits]
    mov al, byte ptr [rsi + r10]
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp final_loop
final_done:
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, r15
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
    .balign 16
inbuf:
    .zero 65536
linebuf:
    .zero 256
