.globl _start
.text
_start:
    mov r12, 0
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov r14, r12
    xor r13, r13
    lea rbx, [rip + linebuf]
row_loop:
    cmp r13, r14
    jae final_line
    mov r15, r14
    sub r15, r13
    cmp r15, 16
    jbe n_ready
    mov r15, 16
n_ready:
    lea rsi, [rip + hexdigits]
    lea r10, [rip + inbuf]
    xor r8, r8
    xor r9, r9
offset_loop:
    mov rcx, r9
    shl rcx, 2
    mov rax, 28
    sub rax, rcx
    mov rcx, rax
    mov rax, r13
    shr rax, cl
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx + r8], al
    inc r8
    inc r9
    cmp r9, 8
    jl offset_loop
    mov byte ptr [rbx + r8], 32
    inc r8
    mov byte ptr [rbx + r8], 32
    inc r8
    lea r11, [r10 + r13]
    xor rcx, rcx
hex_loop:
    cmp rcx, r15
    jae pad_slot
    movzx eax, byte ptr [r11 + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rbx + r8], dl
    inc r8
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx + r8], al
    inc r8
    mov byte ptr [rbx + r8], 32
    inc r8
    jmp hex_next
pad_slot:
    mov byte ptr [rbx + r8], 32
    inc r8
    mov byte ptr [rbx + r8], 32
    inc r8
    mov byte ptr [rbx + r8], 32
    inc r8
hex_next:
    cmp rcx, 7
    jne hex_no_extra
    mov byte ptr [rbx + r8], 32
    inc r8
hex_no_extra:
    inc rcx
    cmp rcx, 16
    jl hex_loop
    mov byte ptr [rbx + r8], 32
    inc r8
    mov byte ptr [rbx + r8], 124
    inc r8
    xor rcx, rcx
ascii_loop:
    cmp rcx, r15
    jae ascii_done
    movzx eax, byte ptr [r11 + rcx]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov al, 46
ascii_store:
    mov byte ptr [rbx + r8], al
    inc r8
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rbx + r8], 124
    inc r8
    mov byte ptr [rbx + r8], 10
    inc r8
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, r8
    syscall
    add r13, 16
    jmp row_loop
final_line:
    lea rsi, [rip + hexdigits]
    xor r8, r8
    xor r9, r9
final_loop:
    mov rcx, r9
    shl rcx, 2
    mov rax, 28
    sub rax, rcx
    mov rcx, rax
    mov rax, r14
    shr rax, cl
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx + r8], al
    inc r8
    inc r9
    cmp r9, 8
    jl final_loop
    mov byte ptr [rbx + r8], 10
    inc r8
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, r8
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
    .align 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
