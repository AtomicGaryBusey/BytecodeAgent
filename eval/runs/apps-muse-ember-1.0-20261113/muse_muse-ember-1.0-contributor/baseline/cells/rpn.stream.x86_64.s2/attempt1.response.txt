.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    cmp r13, 65536
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    add rsi, r13
    mov rdx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14, r14
line_loop:
    cmp r14, r13
    jae exit_prog
    mov r8, r14
    mov r9, r8
find_end:
    cmp r9, r13
    jae line_found
    movzx eax, byte ptr [r12 + r9]
    cmp al, 10
    je line_found
    inc r9
    jmp find_end
line_found:
    mov r14, r9
    cmp r9, r13
    jae next_ready
    inc r14
next_ready:
    xor rbx, rbx
    mov r10, r8
token_loop:
skip_sep:
    cmp r10, r9
    jae tokens_done
    movzx eax, byte ptr [r12 + r10]
    cmp al, 32
    je skip_inc
    cmp al, 9
    je skip_inc
    jmp token_start_found
skip_inc:
    inc r10
    jmp skip_sep
token_start_found:
    mov r11, r10
scan_token:
    cmp r10, r9
    jae token_end_found
    movzx eax, byte ptr [r12 + r10]
    cmp al, 32
    je token_end_found
    cmp al, 9
    je token_end_found
    inc r10
    jmp scan_token
token_end_found:
    mov rcx, r10
    sub rcx, r11
    cmp rcx, 1
    jne check_literal
    movzx eax, byte ptr [r12 + r11]
    cmp al, 43
    je do_operator
    cmp al, 45
    je do_operator
    cmp al, 42
    je do_operator
    cmp al, 47
    je do_operator
    cmp al, 37
    je do_operator
check_literal:
    xor r15, r15
    xor eax, eax
    mov rsi, r11
    movzx edx, byte ptr [r12 + r11]
    cmp dl, 45
    jne not_neg
    mov r15, 1
    inc rsi
not_neg:
    mov rcx, r10
    sub rcx, rsi
    cmp rcx, 1
    jb eval_error
    cmp rcx, 18
    ja eval_error
digit_loop:
    cmp rsi, r10
    jae digits_done
    movzx edx, byte ptr [r12 + rsi]
    cmp dl, 48
    jb eval_error
    cmp dl, 57
    ja eval_error
    sub dl, 48
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    jmp digit_loop
digits_done:
    test r15, r15
    je push_value
    neg rax
push_value:
    lea rdi, [rip + eval_stack]
    mov qword ptr [rdi + rbx*8], rax
    inc rbx
    jmp token_loop
do_operator:
    movzx r15d, byte ptr [r12 + r11]
    cmp rbx, 2
    jb eval_error
    dec rbx
    lea rdi, [rip + eval_stack]
    mov rcx, qword ptr [rdi + rbx*8]
    dec rbx
    mov rax, qword ptr [rdi + rbx*8]
    cmp r15, 43
    je op_add
    cmp r15, 45
    je op_sub
    cmp r15, 42
    je op_mul
    cmp r15, 47
    je op_div
    jmp op_mod
op_add:
    add rax, rcx
    jmp push_result
op_sub:
    sub rax, rcx
    jmp push_result
op_mul:
    imul rax, rcx
    jmp push_result
op_div:
    test rcx, rcx
    je eval_error
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    jne div_do
    cmp rcx, -1
    je eval_error
div_do:
    cqo
    idiv rcx
    jmp push_result
op_mod:
    test rcx, rcx
    je eval_error
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    jne mod_do
    cmp rcx, -1
    je eval_error
mod_do:
    cqo
    idiv rcx
    mov rax, rdx
    jmp push_result
push_result:
    lea rdi, [rip + eval_stack]
    mov qword ptr [rdi + rbx*8], rax
    inc rbx
    jmp token_loop
tokens_done:
    cmp rbx, 1
    jne eval_error
    lea rdi, [rip + eval_stack]
    mov rax, qword ptr [rdi]
    lea rsi, [rip + outbuf]
    add rsi, 32
    dec rsi
    mov byte ptr [rsi], 10
    xor r15, r15
    mov rcx, rax
    test rax, rax
    jns conv_loop
    mov r15, 1
    neg rcx
conv_loop:
    mov rax, rcx
    xor edx, edx
    mov rdi, 10
    div rdi
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    mov rcx, rax
    test rcx, rcx
    jne conv_loop
    test r15, r15
    je write_result
    dec rsi
    mov byte ptr [rsi], 45
write_result:
    lea rdx, [rip + outbuf]
    add rdx, 32
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    jmp line_loop
eval_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 6
    syscall
    jmp line_loop
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
    jmp exit_prog
.section .rodata
err_msg:
    .ascii "error\n"
.bss
    .align 8
inbuf:
    .zero 65536
    .align 8
eval_stack:
    .zero 16384
    .align 8
outbuf:
    .zero 32
