.section .rodata
msg_ok:
.ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err:
.ascii "error at "
msg_err_len = . - msg_err
msg_nl:
.ascii "\n"
msg_nl_len = . - msg_nl
lit_true:
.ascii "true"
lit_false:
.ascii "false"
lit_null:
.ascii "null"
.text
.globl _start
_start:
	lea r15, [rip + buf]
	xor r13, r13
	xor r12, r12
	xor r14, r14
read_loop:
	cmp r13, 65536
	jae read_done
	mov rax, 0
	xor rdi, rdi
	mov rsi, r15
	add rsi, r13
	mov rdx, 65536
	sub rdx, r13
	syscall
	cmp rax, 0
	je read_done
	add r13, rax
	jmp read_loop
read_done:
	call parse_document
	jmp ok_out
parse_document:
	call skip_ws
	call parse_value
	call skip_ws
	cmp r12, r13
	je doc_ok
	jmp fail
doc_ok:
	ret
parse_value:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x7b
	je parse_object
	cmp al, 0x5b
	je parse_array
	cmp al, 0x22
	je parse_string
	cmp al, 0x2d
	je parse_number
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	jbe parse_number
	cmp al, 0x74
	je pv_true
	cmp al, 0x66
	je pv_false
	cmp al, 0x6e
	je pv_null
	jmp fail
pv_true:
	lea rsi, [rip + lit_true]
	mov rcx, 4
	jmp parse_literal
pv_false:
	lea rsi, [rip + lit_false]
	mov rcx, 5
	jmp parse_literal
pv_null:
	lea rsi, [rip + lit_null]
	mov rcx, 4
	jmp parse_literal
parse_literal:
	xor r8, r8
pl_loop:
	cmp r8, rcx
	jae pl_done
	mov rax, r12
	add rax, r8
	cmp rax, r13
	jae pl_eof
	movzx eax, byte ptr [r15 + rax]
	cmp al, byte ptr [rsi + r8]
	jne pl_bad
	inc r8
	jmp pl_loop
pl_done:
	add r12, rcx
	ret
pl_eof:
	mov r12, r13
	jmp fail
pl_bad:
	add r12, r8
	jmp fail
parse_object:
	inc r14
	cmp r14, 1000
	ja fail
	inc r12
	call skip_ws
	cmp r12, r13
	jae o_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x7d
	je o_done
o_loop:
	cmp r12, r13
	jae o_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x22
	jne fail
	call parse_string
	call skip_ws
	cmp r12, r13
	jae o_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x3a
	jne fail
	inc r12
	call skip_ws
	call parse_value
	call skip_ws
	cmp r12, r13
	jae o_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x2c
	je o_comma
	cmp al, 0x7d
	je o_done
	jmp fail
o_comma:
	inc r12
	call skip_ws
	jmp o_loop
o_done:
	inc r12
	dec r14
	ret
o_fail_eof:
	mov r12, r13
	jmp fail
parse_array:
	inc r14
	cmp r14, 1000
	ja fail
	inc r12
	call skip_ws
	cmp r12, r13
	jae a_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x5d
	je a_done
a_loop:
	call parse_value
	call skip_ws
	cmp r12, r13
	jae a_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x2c
	je a_comma
	cmp al, 0x5d
	je a_done
	jmp fail
a_comma:
	inc r12
	call skip_ws
	jmp a_loop
a_done:
	inc r12
	dec r14
	ret
a_fail_eof:
	mov r12, r13
	jmp fail
parse_string:
	inc r12
s_loop:
	cmp r12, r13
	jae s_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x22
	je s_done
	cmp al, 0x5c
	je s_esc
	cmp al, 0x20
	jb s_fail
	inc r12
	jmp s_loop
s_done:
	inc r12
	ret
s_fail_eof:
	mov r12, r13
	jmp fail
s_fail:
	jmp fail
s_esc:
	inc r12
	cmp r12, r13
	jae s_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x22
	je s_inc
	cmp al, 0x5c
	je s_inc
	cmp al, 0x2f
	je s_inc
	cmp al, 0x62
	je s_inc
	cmp al, 0x66
	je s_inc
	cmp al, 0x6e
	je s_inc
	cmp al, 0x72
	je s_inc
	cmp al, 0x74
	je s_inc
	cmp al, 0x75
	je s_hex
	jmp s_fail
s_inc:
	inc r12
	jmp s_loop
s_hex:
	inc r12
	mov r8, 0
s_hex_loop:
	cmp r8, 4
	jae s_ent
	cmp r12, r13
	jae s_fail_eof
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	jb s_fail
	cmp al, 0x39
	jbe s_hex_inc
	cmp al, 0x41
	jb s_fail
	cmp al, 0x46
	jbe s_hex_inc
	cmp al, 0x61
	jb s_fail
	cmp al, 0x66
	jbe s_hex_inc
	jmp s_fail
s_hex_inc:
	inc r12
	inc r8
	jmp s_hex_loop
s_ent:
	jmp s_loop
parse_number:
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x2d
	je n_minus
	jmp n_int
n_minus:
	inc r12
	cmp r12, r13
	jae fail
n_int:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	je n_zero
	cmp al, 0x31
	jb fail
	cmp al, 0x39
	ja fail
n_digits:
	cmp r12, r13
	jae n_frac
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	jb n_frac
	cmp al, 0x39
	ja n_frac
	inc r12
	jmp n_digits
n_zero:
	inc r12
n_frac:
	cmp r12, r13
	jae n_done
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x2e
	jne n_exp
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	ja fail
	inc r12
n_frac_digits:
	cmp r12, r13
	jae n_exp
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	jb n_exp
	cmp al, 0x39
	ja n_exp
	inc r12
	jmp n_frac_digits
n_exp:
	cmp r12, r13
	jae n_done
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x65
	je n_expn
	cmp al, 0x45
	je n_expn
	jmp n_done
n_expn:
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x2b
	je n_exps
	cmp al, 0x2d
	je n_exps
	jmp n_exp_digits
n_exps:
	inc r12
	cmp r12, r13
	jae fail
n_exp_digits:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	ja fail
	inc r12
n_exp_loop:
	cmp r12, r13
	jae n_done
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x30
	jb n_done
	cmp al, 0x39
	ja n_done
	inc r12
	jmp n_exp_loop
n_done:
	ret
skip_ws:
sw_loop:
	cmp r12, r13
	jae sw_done
	movzx eax, byte ptr [r15 + r12]
	cmp al, 0x20
	je sw_adv
	cmp al, 0x09
	je sw_adv
	cmp al, 0x0a
	je sw_adv
	cmp al, 0x0d
	je sw_adv
	jmp sw_done
sw_adv:
	inc r12
	jmp sw_loop
sw_done:
	ret
ok_out:
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + msg_ok]
	mov rdx, msg_ok_len
	syscall
	mov rdi, 0
	jmp do_exit
fail:
	mov rax, r12
	lea r8, [rip + numbuf]
	add r8, 32
	mov r9, 0
	cmp rax, 0
	jne conv_loop
	dec r8
	mov byte ptr [r8], 0x30
	mov r9, 1
	jmp conv_done
conv_loop:
	xor rdx, rdx
	mov rcx, 10
	div rcx
	add dl, 0x30
	dec r8
	mov byte ptr [r8], dl
	inc r9
	cmp rax, 0
	jne conv_loop
conv_done:
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + msg_err]
	mov rdx, msg_err_len
	syscall
	mov rax, 1
	mov rdi, 1
	mov rsi, r8
	mov rdx, r9
	syscall
	mov rax, 1
	mov rdi, 1
	lea rsi, [rip + msg_nl]
	mov rdx, msg_nl_len
	syscall
	mov rdi, 1
do_exit:
	mov rax, 60
	syscall
	jmp do_exit
.bss
buf:
	.zero 65536
numbuf:
	.zero 32
