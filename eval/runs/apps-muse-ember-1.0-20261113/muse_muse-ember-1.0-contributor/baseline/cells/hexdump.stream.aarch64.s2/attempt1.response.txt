.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.globl _start
.text
_start:
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    adrp x22, out_buf
    add x22, x22, :lo12:out_buf
    adrp x28, hexdigits
    add x28, x28, :lo12:hexdigits
    mov x20, #0
read_loop:
    mov x9, #1
    lsl x9, x9, #16
    cmp x20, x9
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x9, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
row_loop:
    cmp x21, x20
    b.ge final_line
    sub x24, x20, x21
    cmp x24, #16
    b.ls n_done
    mov x24, #16
n_done:
    mov x23, #0
    mov x14, #7
offset_hex_loop:
    lsl x9, x14, #2
    lsr x10, x21, x9
    and x10, x10, #15
    ldrb w11, [x28, x10]
    add x12, x22, x23
    strb w11, [x12]
    add x23, x23, #1
    cmp x14, #0
    b.eq offset_hex_done
    sub x14, x14, #1
    b offset_hex_loop
offset_hex_done:
    mov w11, #32
    add x12, x22, x23
    strb w11, [x12]
    add x23, x23, #1
    add x12, x22, x23
    strb w11, [x12]
    add x23, x23, #1
    mov x25, #0
slot_loop:
    cmp x25, x24
    b.hs slot_empty
    add x9, x21, x25
    ldrb w10, [x19, x9]
    lsr w11, w10, #4
    and w11, w11, #15
    ldrb w11, [x28, x11]
    add x12, x22, x23
    strb w11, [x12]
    add x23, x23, #1
    and w10, w10, #15
    ldrb w10, [x28, x10]
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    mov w10, #32
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    b slot_after
slot_empty:
    mov w10, #32
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
slot_after:
    cmp x25, #7
    b.ne slot_next
    mov w10, #32
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
slot_next:
    add x25, x25, #1
    cmp x25, #16
    b.ne slot_loop
    mov w10, #32
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    mov w10, #124
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    mov x25, #0
ascii_loop:
    cmp x25, x24
    b.hs ascii_done
    add x9, x21, x25
    ldrb w10, [x19, x9]
    cmp w10, #32
    b.lt ascii_dot
    cmp w10, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w10, #46
ascii_store:
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    add x25, x25, #1
    b ascii_loop
ascii_done:
    mov w10, #124
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    mov w10, #10
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    mov x0, #1
    mov x1, x22
    mov x2, x23
    mov x8, #SYS_WRITE
    svc #0
    add x21, x21, x24
    b row_loop
final_line:
    mov x23, #0
    mov x14, #7
final_hex_loop:
    lsl x9, x14, #2
    lsr x10, x20, x9
    and x10, x10, #15
    ldrb w11, [x28, x10]
    add x12, x22, x23
    strb w11, [x12]
    add x23, x23, #1
    cmp x14, #0
    b.eq final_hex_done
    sub x14, x14, #1
    b final_hex_loop
final_hex_done:
    mov w10, #10
    add x12, x22, x23
    strb w10, [x12]
    add x23, x23, #1
    mov x0, #1
    mov x1, x22
    mov x2, x23
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
exit_loop:
    b exit_loop
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
in_buf:
    .zero 65536
out_buf:
    .zero 128
