.globl _start
.text
_start:
 lea r12, [rip + inbuf]
 lea r14, [rip + inbuf]
 add r14, 65536
 mov r13, r12
read_loop:
 cmp r13, r14
 jae read_done
 mov rax, 0
 mov rdi, 0
 mov rsi, r13
 mov rdx, r14
 sub rdx, r13
 syscall
 test rax, rax
 jle read_done
 add r13, rax
 jmp read_loop
read_done:
 mov r15, r12
line_loop:
 cmp r15, r13
 jae exit_prog
 mov rsi, r15
find_loop:
 cmp rsi, r13
 jae final_line
 cmp byte ptr [rsi], 10
 je newline_line
 inc rsi
 jmp find_loop
newline_line:
 mov r8, r15
 lea r9, [rsi - 1]
rev_nl:
 cmp r8, r9
 jae rev_nl_done
 mov al, byte ptr [r8]
 mov dl, byte ptr [r9]
 mov byte ptr [r8], dl
 mov byte ptr [r9], al
 inc r8
 dec r9
 jmp rev_nl
rev_nl_done:
 lea r10, [rsi + 1]
 mov rdx, r10
 sub rdx, r15
 mov rsi, r15
 call write_all
 mov r15, r10
 jmp line_loop
final_line:
 mov r8, r15
 lea r9, [rsi - 1]
rev_fin:
 cmp r8, r9
 jae rev_fin_done
 mov al, byte ptr [r8]
 mov dl, byte ptr [r9]
 mov byte ptr [r8], dl
 mov byte ptr [r9], al
 inc r8
 dec r9
 jmp rev_fin
rev_fin_done:
 mov rdx, rsi
 sub rdx, r15
 test rdx, rdx
 jz nl_only
 mov rsi, r15
 call write_all
nl_only:
 lea rsi, [rip + nl_byte]
 mov rdx, 1
 call write_all
 jmp exit_prog
write_all:
 test rdx, rdx
 jz wa_ret
 mov r8, rsi
 mov r9, rdx
wa_loop:
 mov rax, 1
 mov rdi, 1
 mov rsi, r8
 mov rdx, r9
 syscall
 test rax, rax
 jle wa_ret
 add r8, rax
 sub r9, rax
 test r9, r9
 jnz wa_loop
wa_ret:
 ret
exit_prog:
 mov rax, 60
 mov rdi, 0
 syscall
.section .rodata
nl_byte:
 .byte 10
.bss
inbuf:
 .zero 65536
