.globl _start
.text
_start:
 lea r12, [rip + inbuf]
 lea r14, [rip + inbuf]
 add r14, 65536
 mov r13, r12
read_loop:
 cmp r13, r14
 jae read_done
 mov rax, 0
 mov rdi, 0
 mov rsi, r13
 mov rdx, r14
 sub rdx, r13
 syscall
 test rax, rax
 jle read_done
 add r13, rax
 jmp read_loop
read_done:
 mov r15, r12
line_loop:
 cmp r15, r13
 jae exit_prog
 mov rsi, r15
find_loop:
 cmp rsi, r13
 jae final_line
 cmp byte ptr [rsi], 10
 je line_with_nl
 inc rsi
 jmp find_loop
line_with_nl:
 mov r8, r15
 lea r9, [rsi - 1]
rev1:
 cmp r8, r9
 jae rev1_done
 mov al, byte ptr [r8]
 mov dl, byte ptr [r9]
 mov byte ptr [r8], dl
 mov byte ptr [r9], al
 inc r8
 dec r9
 jmp rev1
rev1_done:
 mov rdx, rsi
 sub rdx, r15
 inc rdx
 mov rax, 1
 mov rdi, 1
 mov rsi, r15
 syscall
 lea r15, [rsi + 1]
 jmp line_loop
final_line:
 mov r8, r15
 lea r9, [rsi - 1]
rev2:
 cmp r8, r9
 jae rev2_done
 mov al, byte ptr [r8]
 mov dl, byte ptr [r9]
 mov byte ptr [r8], dl
 mov byte ptr [r9], al
 inc r8
 dec r9
 jmp rev2
rev2_done:
 mov rdx, rsi
 sub rdx, r15
 test rdx, rdx
 jz write_nl
 mov rax, 1
 mov rdi, 1
 mov rsi, r15
 syscall
write_nl:
 mov rax, 1
 mov rdi, 1
 lea rsi, [rip + nl_byte]
 mov rdx, 1
 syscall
exit_prog:
 mov rax, 60
 mov rdi, 0
 syscall
.section .rodata
nl_byte:
 .byte 10
.bss
inbuf:
 .zero 65536
