.equ BUFSZ, 65536
.globl _start
.text
_start:
  xor r12, r12
read_loop:
  cmp r12, BUFSZ
  jae read_done
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + inbuf]
  add rsi, r12
  mov rdx, BUFSZ
  sub rdx, r12
  syscall
  cmp rax, 0
  jle read_done
  add r12, rax
  jmp read_loop
read_done:
  lea r15, [rip + inbuf]
  xor r13, r13
line_loop:
  cmp r13, r12
  jae exit_prog
  mov r14, r13
scan_loop:
  cmp r14, r12
  jae scan_done
  movzx eax, byte ptr [r15 + r14]
  cmp al, 10
  je scan_done
  inc r14
  jmp scan_loop
scan_done:
  mov r9, r14
  sub r9, r13
  cmp r9, 2
  jb reverse_done
  mov r10, r13
  mov r8, r14
  dec r8
rev_loop:
  mov al, byte ptr [r15 + r10]
  mov dl, byte ptr [r15 + r8]
  mov byte ptr [r15 + r10], dl
  mov byte ptr [r15 + r8], al
  inc r10
  dec r8
  cmp r10, r8
  jl rev_loop
reverse_done:
  mov rdx, r14
  sub rdx, r13
  jz write_nl
  mov rax, 1
  mov rdi, 1
  lea rsi, [r15 + r13]
  syscall
write_nl:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + newline_byte]
  mov rdx, 1
  syscall
  cmp r14, r12
  jb has_nl
  mov r13, r12
  jmp line_loop
has_nl:
  mov r13, r14
  inc r13
  jmp line_loop
exit_prog:
  mov rax, 60
  xor rdi, rdi
  syscall
hang:
  jmp hang
.section .rodata
newline_byte:
.byte 10
.bss
.align 16
inbuf:
.space 65536
