.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INPUT_MAX, 32768
.globl _start
.bss
.balign 16
inbuf:
.space 32768
.balign 16
outbuf:
.space 34000
.balign 16
line_off:
.space 262144
.balign 16
line_len:
.space 262144
.balign 8
out_len:
.space 8
.text
_start:
    xor r12, r12
read_loop:
    cmp r12, INPUT_MAX
    jae parse_init
    mov rax, SYS_READ
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_init
    js parse_init
    add r12, rax
    jmp read_loop
parse_init:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + inbuf]
parse_loop:
    cmp r15, r12
    jae parse_final
    cmp byte ptr [rbx + r15], 10
    jne parse_next
    mov rdx, r15
    sub rdx, r14
    lea rcx, [rip + line_off]
    mov qword ptr [rcx + r13*8], r14
    lea rcx, [rip + line_len]
    mov qword ptr [rcx + r13*8], rdx
    inc r13
    lea r14, [r15 + 1]
parse_next:
    inc r15
    jmp parse_loop
parse_final:
    cmp r14, r12
    jae sort_start
    mov rdx, r12
    sub rdx, r14
    lea rcx, [rip + line_off]
    mov qword ptr [rcx + r13*8], r14
    lea rcx, [rip + line_len]
    mov qword ptr [rcx + r13*8], rdx
    inc r13
sort_start:
    cmp r13, 2
    jb build_init
    sub rsp, 64
    mov qword ptr [rsp + 48], 1
sort_i_loop:
    mov rax, qword ptr [rsp + 48]
    cmp rax, r13
    jae sort_i_done
    lea rcx, [rip + line_off]
    mov rdx, qword ptr [rcx + rax*8]
    mov qword ptr [rsp], rdx
    lea rcx, [rip + line_len]
    mov rdx, qword ptr [rcx + rax*8]
    mov qword ptr [rsp + 8], rdx
    mov rdx, qword ptr [rsp + 48]
    dec rdx
    mov qword ptr [rsp + 16], rdx
sort_j_loop:
    mov rax, qword ptr [rsp + 16]
    cmp rax, 0
    jl sort_j_done
    lea rcx, [rip + line_off]
    mov rdi, qword ptr [rcx + rax*8]
    lea rcx, [rip + line_len]
    mov rsi, qword ptr [rcx + rax*8]
    mov rdx, qword ptr [rsp]
    mov rcx, qword ptr [rsp + 8]
    call compare_lines
    cmp eax, 0
    jle sort_j_done
    mov rax, qword ptr [rsp + 16]
    lea r8, [rax + 1]
    lea rcx, [rip + line_off]
    mov rdx, qword ptr [rcx + rax*8]
    mov qword ptr [rcx + r8*8], rdx
    lea rcx, [rip + line_len]
    mov rdx, qword ptr [rcx + rax*8]
    mov qword ptr [rcx + r8*8], rdx
    dec qword ptr [rsp + 16]
    jmp sort_j_loop
sort_j_done:
    mov rax, qword ptr [rsp + 16]
    inc rax
    mov rdx, qword ptr [rsp]
    lea rcx, [rip + line_off]
    mov qword ptr [rcx + rax*8], rdx
    mov rdx, qword ptr [rsp + 8]
    lea rcx, [rip + line_len]
    mov qword ptr [rcx + rax*8], rdx
    inc qword ptr [rsp + 48]
    jmp sort_i_loop
sort_i_done:
    add rsp, 64
build_init:
    xor r14, r14
    lea rbx, [rip + outbuf]
    lea rbp, [rip + inbuf]
    xor r15, r15
build_loop:
    cmp r15, r13
    jae build_done
    lea rax, [rip + line_off]
    mov r8, qword ptr [rax + r15*8]
    lea rax, [rip + line_len]
    mov r9, qword ptr [rax + r15*8]
    xor rcx, rcx
copy_loop:
    cmp rcx, r9
    jae copy_done
    mov rax, r8
    add rax, rcx
    mov al, byte ptr [rbp + rax]
    mov byte ptr [rbx + r14], al
    inc rcx
    inc r14
    jmp copy_loop
copy_done:
    mov byte ptr [rbx + r14], 10
    inc r14
    inc r15
    jmp build_loop
build_done:
    mov qword ptr [rip + out_len], r14
    mov r12, r14
    xor r14, r14
write_loop:
    cmp r14, r12
    jae exit_prog
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + outbuf]
    add rsi, r14
    mov rdx, r12
    sub rdx, r14
    syscall
    test rax, rax
    jle exit_prog
    add r14, rax
    jmp write_loop
exit_prog:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
exit_loop:
    jmp exit_loop
compare_lines:
    sub rsp, 32
    mov qword ptr [rsp], rdi
    mov qword ptr [rsp + 8], rsi
    mov qword ptr [rsp + 16], rdx
    mov qword ptr [rsp + 24], rcx
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    lea r10, [rip + inbuf]
    xor r11, r11
cmp_loop:
    cmp r11, r8
    jae cmp_len
    mov rcx, rdi
    add rcx, r11
    movzx eax, byte ptr [r10 + rcx]
    mov rcx, qword ptr [rsp + 16]
    add rcx, r11
    movzx edx, byte ptr [r10 + rcx]
    cmp eax, edx
    jne cmp_diff
    inc r11
    jmp cmp_loop
cmp_diff:
    jb cmp_less
    mov eax, 1
    jmp cmp_ret
cmp_less:
    mov eax, -1
    jmp cmp_ret
cmp_len:
    mov rax, qword ptr [rsp + 8]
    mov rcx, qword ptr [rsp + 24]
    cmp rax, rcx
    je cmp_equal
    jb cmp_less2
    mov eax, 1
    jmp cmp_ret
cmp_less2:
    mov eax, -1
    jmp cmp_ret
cmp_equal:
    xor eax, eax
cmp_ret:
    add rsp, 32
    ret
