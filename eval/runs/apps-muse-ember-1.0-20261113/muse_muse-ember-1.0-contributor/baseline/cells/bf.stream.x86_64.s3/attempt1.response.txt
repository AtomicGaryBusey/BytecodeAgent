.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    cmp rdx, 0
    je read_done
    mov rax, 0
    syscall
    cmp rax, 0
    je read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rax, [rip + input_len]
    mov qword ptr [rax], rbx
    mov r12, rbx
    mov r13, rbx
    xor rcx, rcx
find_loop:
    cmp rcx, rbx
    jae find_done
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x21
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r12, rcx
    mov r13, rcx
    inc r13
find_done:
    xor rbx, rbx
    xor rbp, rbp
bracket_loop:
    cmp rbx, r12
    jae bracket_done
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x5b
    je bracket_push
    cmp al, 0x5d
    je bracket_pop
    inc rbx
    jmp bracket_loop
bracket_push:
    lea rsi, [rip + bstack]
    mov dword ptr [rsi + rbp*4], ebx
    inc rbp
    inc rbx
    jmp bracket_loop
bracket_pop:
    cmp rbp, 0
    je fail_exit
    dec rbp
    lea rsi, [rip + bstack]
    mov r10d, dword ptr [rsi + rbp*4]
    lea rsi, [rip + match]
    mov dword ptr [rsi + rbx*4], r10d
    mov dword ptr [rsi + r10*4], ebx
    inc rbx
    jmp bracket_loop
bracket_done:
    cmp rbp, 0
    jne fail_exit
    lea rax, [rip + input_len]
    mov rbx, qword ptr [rax]
    xor r14, r14
    xor r15, r15
exec_loop:
    cmp r15, r12
    jae exec_done
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + r15]
    inc r15
    cmp al, 62
    je cmd_inc_ptr
    cmp al, 60
    je cmd_dec_ptr
    cmp al, 43
    je cmd_inc_cell
    cmp al, 45
    je cmd_dec_cell
    cmp al, 46
    je cmd_output
    cmp al, 44
    je cmd_input
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    jmp exec_loop
cmd_inc_ptr:
    inc r14
    jmp exec_loop
cmd_dec_ptr:
    dec r14
    jmp exec_loop
cmd_inc_cell:
    lea rsi, [rip + tape]
    inc byte ptr [rsi + r14]
    jmp exec_loop
cmd_dec_cell:
    lea rsi, [rip + tape]
    dec byte ptr [rsi + r14]
    jmp exec_loop
cmd_output:
    mov rdi, 1
    lea rsi, [rip + tape]
    add rsi, r14
    mov rdx, 1
    mov rax, 1
    syscall
    jmp exec_loop
cmd_input:
    cmp r13, rbx
    jae input_eof
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + r13]
    inc r13
    lea rsi, [rip + tape]
    mov byte ptr [rsi + r14], al
    jmp exec_loop
input_eof:
    lea rsi, [rip + tape]
    mov byte ptr [rsi + r14], 0
    jmp exec_loop
cmd_open:
    lea rsi, [rip + tape]
    cmp byte ptr [rsi + r14], 0
    je open_jump
    jmp exec_loop
open_jump:
    lea rsi, [rip + match]
    mov r10, r15
    dec r10
    mov eax, dword ptr [rsi + r10*4]
    mov r15, rax
    inc r15
    jmp exec_loop
cmd_close:
    lea rsi, [rip + tape]
    cmp byte ptr [rsi + r14], 0
    jne close_jump
    jmp exec_loop
close_jump:
    lea rsi, [rip + match]
    mov r10, r15
    dec r10
    mov eax, dword ptr [rsi + r10*4]
    mov r15, rax
    inc r15
    jmp exec_loop
exec_done:
    mov rdi, 0
    mov rax, 60
    syscall
fail_exit:
    mov rdi, 1
    mov rax, 60
    syscall
.bss
.balign 8
input_buf:
    .zero 65536
.balign 8
tape:
    .zero 30000
.balign 8
match:
    .zero 262144
.balign 8
bstack:
    .zero 8192
.balign 8
input_len:
    .zero 8
