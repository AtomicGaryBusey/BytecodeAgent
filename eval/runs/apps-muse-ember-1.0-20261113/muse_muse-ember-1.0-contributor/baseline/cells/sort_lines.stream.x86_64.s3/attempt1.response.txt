.globl _start
_start:
 xor eax, eax
 mov qword ptr [rip + total_bytes], rax
read_loop:
 mov r8, qword ptr [rip + total_bytes]
 cmp r8, 32768
 jae read_done
 lea rsi, [rip + inbuf]
 add rsi, r8
 mov rdx, 32768
 sub rdx, r8
 xor edi, edi
 xor eax, eax
 syscall
 test rax, rax
 je read_done
 mov r8, qword ptr [rip + total_bytes]
 add r8, rax
 mov qword ptr [rip + total_bytes], r8
 jmp read_loop
read_done:
 xor eax, eax
 mov qword ptr [rip + num_lines], rax
 mov qword ptr [rip + cur_start], rax
 mov r12, qword ptr [rip + total_bytes]
 xor r13, r13
 lea r14, [rip + line_starts]
 lea rbx, [rip + line_lens]
 lea r8, [rip + inbuf]
parse_loop:
 cmp r13, r12
 jae parse_done
 movzx eax, byte ptr [r8 + r13]
 cmp al, 10
 jne parse_next
 mov rax, qword ptr [rip + num_lines]
 mov rcx, qword ptr [rip + cur_start]
 mov qword ptr [r14 + rax*8], rcx
 mov rdx, r13
 sub rdx, rcx
 mov qword ptr [rbx + rax*8], rdx
 mov rax, qword ptr [rip + num_lines]
 inc rax
 mov qword ptr [rip + num_lines], rax
 mov rax, r13
 inc rax
 mov qword ptr [rip + cur_start], rax
parse_next:
 inc r13
 jmp parse_loop
parse_done:
 mov rax, qword ptr [rip + cur_start]
 cmp rax, r12
 jae parse_all_done
 mov rcx, qword ptr [rip + num_lines]
 mov qword ptr [r14 + rcx*8], rax
 mov rdx, r12
 sub rdx, rax
 mov qword ptr [rbx + rcx*8], rdx
 inc rcx
 mov qword ptr [rip + num_lines], rcx
parse_all_done:
 mov r12, qword ptr [rip + num_lines]
 cmp r12, 2
 jb sort_done
 mov r13, 1
sort_outer:
 cmp r13, r12
 jae sort_done
 lea r8, [rip + line_starts]
 mov r15, qword ptr [r8 + r13*8]
 lea r8, [rip + line_lens]
 mov rbx, qword ptr [r8 + r13*8]
 mov r14, r13
 dec r14
sort_inner:
 cmp r14, 0
 jl sort_place
 lea r8, [rip + line_starts]
 mov rdi, qword ptr [r8 + r14*8]
 lea r8, [rip + line_lens]
 mov rsi, qword ptr [r8 + r14*8]
 mov rdx, r15
 mov rcx, rbx
 call cmp_lines
 cmp rax, 0
 jle sort_place
 lea r8, [rip + line_starts]
 mov r9, qword ptr [r8 + r14*8]
 lea r8, [rip + line_lens]
 mov r10, qword ptr [r8 + r14*8]
 mov rax, r14
 inc rax
 lea r8, [rip + line_starts]
 mov qword ptr [r8 + rax*8], r9
 lea r8, [rip + line_lens]
 mov qword ptr [r8 + rax*8], r10
 dec r14
 jmp sort_inner
sort_place:
 mov rax, r14
 inc rax
 lea r8, [rip + line_starts]
 mov qword ptr [r8 + rax*8], r15
 lea r8, [rip + line_lens]
 mov qword ptr [r8 + rax*8], rbx
 inc r13
 jmp sort_outer
sort_done:
 mov r12, qword ptr [rip + num_lines]
 xor r13, r13
output_loop:
 cmp r13, r12
 jae output_done
 lea r8, [rip + line_starts]
 mov r14, qword ptr [r8 + r13*8]
 lea r8, [rip + line_lens]
 mov r15, qword ptr [r8 + r13*8]
write_line_loop:
 test r15, r15
 je output_newline
 mov rdi, 1
 lea rsi, [rip + inbuf]
 add rsi, r14
 mov rdx, r15
 mov rax, 1
 syscall
 add r14, rax
 sub r15, rax
 jmp write_line_loop
output_newline:
 mov rdi, 1
 lea rsi, [rip + nl_byte]
 mov rdx, 1
 mov rax, 1
 syscall
 inc r13
 jmp output_loop
output_done:
 xor edi, edi
 mov rax, 60
 syscall
hang:
 jmp hang
cmp_lines:
 mov r8, rsi
 cmp r8, rcx
 cmovae r8, rcx
 test r8, r8
 je cmp_len
 lea r10, [rip + inbuf]
 add r10, rdi
 lea r11, [rip + inbuf]
 add r11, rdx
 xor r9, r9
cmp_loop:
 mov al, byte ptr [r10 + r9]
 cmp al, byte ptr [r11 + r9]
 jne cmp_diff
 inc r9
 cmp r9, r8
 jb cmp_loop
 jmp cmp_len
cmp_diff:
 movzx eax, byte ptr [r10 + r9]
 movzx edx, byte ptr [r11 + r9]
 cmp al, dl
 jb cmp_less
 mov rax, 1
 ret
cmp_len:
 cmp rsi, rcx
 jb cmp_less
 ja cmp_greater
 xor eax, eax
 ret
cmp_less:
 mov rax, -1
 ret
cmp_greater:
 mov rax, 1
 ret
