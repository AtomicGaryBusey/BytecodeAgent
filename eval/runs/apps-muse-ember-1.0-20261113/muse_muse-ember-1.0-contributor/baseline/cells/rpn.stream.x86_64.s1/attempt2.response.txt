.text
.globl _start
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
_start:
    xor r12, r12
    lea rbp, [rip + inbuf]
read_loop:
    cmp r12, 65536
    jae read_done
    mov rax, SYS_READ
    mov rdi, 0
    mov rsi, rbp
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
main_loop:
    cmp r13, r12
    jae done
    mov rcx, r13
find_loop:
    cmp rcx, r12
    jae found_end
    mov al, byte ptr [rbp + rcx]
    cmp al, 10
    je found_end
    inc rcx
    jmp find_loop
found_end:
    mov rbx, rbp
    add rbx, r13
    mov r10, rcx
    sub r10, r13
    cmp rcx, r12
    jae next_no_newline
    mov r13, rcx
    inc r13
    jmp next_done
next_no_newline:
    mov r13, rcx
next_done:
    xor r15, r15
    xor r14, r14
    xor r8, r8
eval_loop:
    test r14, r14
    jnz eval_done
    cmp r8, r10
    jae eval_done
skip_loop:
    cmp r8, r10
    jae eval_done
    mov al, byte ptr [rbx + r8]
    cmp al, 32
    je skip_inc
    cmp al, 9
    je skip_inc
    jmp tok_start
skip_inc:
    inc r8
    jmp skip_loop
tok_start:
    mov r9, r8
tok_scan:
    cmp r8, r10
    jae tok_end
    mov al, byte ptr [rbx + r8]
    cmp al, 32
    je tok_end
    cmp al, 9
    je tok_end
    inc r8
    jmp tok_scan
tok_end:
    mov r11, r8
    sub r11, r9
    cmp r11, 1
    jne literal_check
    mov al, byte ptr [rbx + r9]
    cmp al, 43
    je do_op_plus
    cmp al, 45
    je do_op_minus
    cmp al, 42
    je do_op_mul
    cmp al, 47
    je do_op_div
    cmp al, 37
    je do_op_mod
literal_check:
    mov al, byte ptr [rbx + r9]
    xor rdi, rdi
    cmp al, 45
    jne lit_no_minus
    mov rdi, 1
    mov rcx, r11
    dec rcx
    jmp lit_check_count
lit_no_minus:
    mov rcx, r11
lit_check_count:
    cmp rcx, 0
    je lit_invalid
    cmp rcx, 18
    ja lit_invalid
    mov rsi, rcx
    mov rcx, r9
    add rcx, rdi
    xor rax, rax
lit_digit_loop:
    test rsi, rsi
    je lit_digits_done
    movzx edx, byte ptr [rbx + rcx]
    cmp edx, 48
    jb lit_invalid
    cmp edx, 57
    ja lit_invalid
    sub edx, 48
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    dec rsi
    jmp lit_digit_loop
lit_digits_done:
    test rdi, rdi
    je lit_push
    neg rax
lit_push:
    lea rdx, [rip + eval_stack]
    mov qword ptr [rdx + r15*8], rax
    inc r15
    jmp eval_loop
lit_invalid:
    mov r14, 1
    jmp eval_done
do_op_plus:
    cmp r15, 2
    jb op_error
    lea rsi, [rip + eval_stack]
    dec r15
    mov rcx, qword ptr [rsi + r15*8]
    dec r15
    mov rax, qword ptr [rsi + r15*8]
    add rax, rcx
    mov qword ptr [rsi + r15*8], rax
    inc r15
    jmp eval_loop
do_op_minus:
    cmp r15, 2
    jb op_error
    lea rsi, [rip + eval_stack]
    dec r15
    mov rcx, qword ptr [rsi + r15*8]
    dec r15
    mov rax, qword ptr [rsi + r15*8]
    sub rax, rcx
    mov qword ptr [rsi + r15*8], rax
    inc r15
    jmp eval_loop
do_op_mul:
    cmp r15, 2
    jb op_error
    lea rsi, [rip + eval_stack]
    dec r15
    mov rcx, qword ptr [rsi + r15*8]
    dec r15
    mov rax, qword ptr [rsi + r15*8]
    imul rax, rcx
    mov qword ptr [rsi + r15*8], rax
    inc r15
    jmp eval_loop
do_op_div:
    cmp r15, 2
    jb op_error
    lea rsi, [rip + eval_stack]
    dec r15
    mov rcx, qword ptr [rsi + r15*8]
    dec r15
    mov rax, qword ptr [rsi + r15*8]
    test rcx, rcx
    je op_error
    cmp rcx, -1
    jne do_div
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je op_error
do_div:
    cqo
    idiv rcx
    mov qword ptr [rsi + r15*8], rax
    inc r15
    jmp eval_loop
do_op_mod:
    cmp r15, 2
    jb op_error
    lea rsi, [rip + eval_stack]
    dec r15
    mov rcx, qword ptr [rsi + r15*8]
    dec r15
    mov rax, qword ptr [rsi + r15*8]
    test rcx, rcx
    je op_error
    cmp rcx, -1
    jne do_mod
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je op_error
do_mod:
    cqo
    idiv rcx
    mov rax, rdx
    mov qword ptr [rsi + r15*8], rax
    inc r15
    jmp eval_loop
op_error:
    mov r14, 1
    jmp eval_done
eval_done:
    test r14, r14
    jnz print_error
    cmp r15, 1
    jne print_error
    lea rdx, [rip + eval_stack]
    mov rax, qword ptr [rdx]
    test rax, rax
    jnz conv_nonzero
    lea rsi, [rip + outbuf]
    mov byte ptr [rsi], 48
    mov byte ptr [rsi + 1], 10
    mov rdx, 2
    call do_write
    jmp main_loop
conv_nonzero:
    xor r8, r8
    test rax, rax
    jns conv_pos
    mov r8, 1
    neg rax
conv_pos:
    mov r9, rax
    xor r10, r10
conv_loop:
    test r9, r9
    je conv_build
    mov rax, r9
    xor rdx, rdx
    mov rcx, 10
    div rcx
    mov r9, rax
    add rdx, 48
    lea rsi, [rip + revbuf]
    mov byte ptr [rsi + r10], dl
    inc r10
    jmp conv_loop
conv_build:
    lea rsi, [rip + outbuf]
    xor r11, r11
    test r8, r8
    je conv_digits_init
    mov byte ptr [rsi + r11], 45
    inc r11
conv_digits_init:
    mov rcx, r10
    dec rcx
conv_digit_loop:
    lea rdi, [rip + revbuf]
    mov al, byte ptr [rdi + rcx]
    mov byte ptr [rsi + r11], al
    inc r11
    dec rcx
    cmp rcx, -1
    jne conv_digit_loop
    mov byte ptr [rsi + r11], 10
    inc r11
    mov rdx, r11
    call do_write
    jmp main_loop
print_error:
    lea rsi, [rip + errmsg]
    mov rdx, 6
    call do_write
    jmp main_loop
done:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
do_write:
    mov r8, rdx
    mov r9, rsi
write_loop:
    test r8, r8
    je write_done
    mov rax, SYS_WRITE
    mov rdi, 1
    mov rsi, r9
    mov rdx, r8
    syscall
    test rax, rax
    jle write_done
    add r9, rax
    sub r8, rax
    jmp write_loop
write_done:
    ret
.section .rodata
errmsg: .ascii "error\n"
.bss
.balign 8
inbuf: .zero 65536
eval_stack: .zero 16384
revbuf: .zero 64
outbuf: .zero 64
