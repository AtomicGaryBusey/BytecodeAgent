.globl _start
.bss
buffer:
 .zero 8192
outbuf:
 .zero 64
tmp:
 .zero 16
.text
_start:
 xor r12, r12
 xor r13, r13
 xor r14, r14
 xor r15, r15
L_read:
 mov rax, 0
 xor rdi, rdi
 lea rsi, [rip + buffer]
 mov rdx, 8192
 syscall
 test rax, rax
 jz L_done
 xor rcx, rcx
L_inner:
 movzx edx, byte ptr [rsi + rcx]
 inc r14
 cmp dl, 0x0A
 jne L_notnl
 inc r12
L_notnl:
 cmp dl, 0x20
 je L_ws
 cmp dl, 0x09
 jb L_nonws
 cmp dl, 0x0D
 ja L_nonws
L_ws:
 xor r15, r15
 jmp L_next
L_nonws:
 test r15, r15
 jnz L_next
 inc r13
 mov r15, 1
L_next:
 inc rcx
 cmp rcx, rax
 jb L_inner
 jmp L_read
L_done:
 lea rdi, [rip + outbuf]
 mov rax, r12
 call L_conv
 mov byte ptr [rdi], 0x20
 inc rdi
 mov rax, r13
 call L_conv
 mov byte ptr [rdi], 0x20
 inc rdi
 mov rax, r14
 call L_conv
 mov byte ptr [rdi], 0x0A
 inc rdi
 lea rsi, [rip + outbuf]
 mov rdx, rdi
 sub rdx, rsi
 mov rdi, 1
 mov rax, 1
 syscall
L_exit:
 mov rax, 60
 xor rdi, rdi
 syscall
 jmp L_exit
L_conv:
 cmp rax, 0
 jne L_conv_nz
 mov byte ptr [rdi], 0x30
 inc rdi
 ret
L_conv_nz:
 lea r10, [rip + tmp]
 xor rcx, rcx
 mov rbx, 10
L_conv_loop:
 xor rdx, rdx
 div rbx
 add dl, 0x30
 mov byte ptr [r10 + rcx], dl
 inc rcx
 test rax, rax
 jnz L_conv_loop
L_conv_copy:
 test rcx, rcx
 jz L_conv_done
 dec rcx
 mov r8b, byte ptr [r10 + rcx]
 mov byte ptr [rdi], r8b
 inc rdi
 jmp L_conv_copy
L_conv_done:
 ret
