.text
.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x2, #65536
    subs    x2, x2, x20
    beq     read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    mov     x21, #0                 // scan for first '!'
find_bang:
    cmp     x21, x20
    beq     no_bang
    ldrb    w0, [x19, x21]
    cmp     w0, #33
    beq     found_bang
    add     x21, x21, #1
    b       find_bang
found_bang:
    add     x23, x19, x21
    add     x23, x23, #1            // input start
    add     x24, x19, x20           // input end
    b       compile
no_bang:
    add     x23, x19, x20
    mov     x24, x23

compile:
    adrp    x20, jumps
    add     x20, x20, :lo12:jumps
    adrp    x1, bstack
    add     x1, x1, :lo12:bstack
    adrp    x2, code
    add     x2, x2, :lo12:code
    mov     x3, #0                  // source index
    mov     x4, #0                  // code index
    mov     x5, #0                  // stack depth
cloop:
    cmp     x3, x21
    beq     cdone
    ldrb    w0, [x19, x3]
    add     x3, x3, #1
    cmp     w0, #43
    beq     emit
    cmp     w0, #45
    beq     emit
    cmp     w0, #62
    beq     emit
    cmp     w0, #60
    beq     emit
    cmp     w0, #46
    beq     emit
    cmp     w0, #44
    beq     emit
    cmp     w0, #91
    beq     copen
    cmp     w0, #93
    beq     cclose
    b       cloop
emit:
    strb    w0, [x2, x4]
    add     x4, x4, #1
    b       cloop
copen:
    str     w4, [x1, x5, lsl #2]
    add     x5, x5, #1
    b       emit
cclose:
    cbz     x5, fail
    sub     x5, x5, #1
    ldr     w6, [x1, x5, lsl #2]    // index of matching '['
    add     w7, w4, #1
    str     w7, [x20, x6, lsl #2]   // jumps[open] = close + 1
    add     w7, w6, #1
    str     w7, [x20, x4, lsl #2]   // jumps[close] = open + 1
    b       emit
cdone:
    cbnz    x5, fail
    strb    wzr, [x2, x4]           // terminator
    mov     x19, x2                 // x19 = code base

run:
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    adrp    x25, outbuf
    add     x25, x25, :lo12:outbuf
    mov     x26, #0                 // output count
    mov     x21, #0                 // pc
iloop:
    ldrb    w0, [x19, x21]
    add     x21, x21, #1
    cmp     w0, #43
    beq     op_plus
    cmp     w0, #45
    beq     op_minus
    cmp     w0, #62
    beq     op_right
    cmp     w0, #60
    beq     op_left
    cmp     w0, #91
    beq     op_open
    cmp     w0, #93
    beq     op_close
    cmp     w0, #46
    beq     op_dot
    cmp     w0, #44
    beq     op_comma
    b       finish
op_plus:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       iloop
op_minus:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       iloop
op_right:
    add     x22, x22, #1
    b       iloop
op_left:
    sub     x22, x22, #1
    b       iloop
op_open:
    ldrb    w1, [x22]
    cbnz    w1, iloop
    sub     x2, x21, #1
    ldr     w21, [x20, x2, lsl #2]
    b       iloop
op_close:
    ldrb    w1, [x22]
    cbz     w1, iloop
    sub     x2, x21, #1
    ldr     w21, [x20, x2, lsl #2]
    b       iloop
op_dot:
    ldrb    w1, [x22]
    strb    w1, [x25, x26]
    add     x26, x26, #1
    mov     x2, #65536
    cmp     x26, x2
    blo     iloop
    bl      flush
    b       iloop
op_comma:
    cmp     x23, x24
    bhs     comma_zero
    ldrb    w1, [x23], #1
    strb    w1, [x22]
    b       iloop
comma_zero:
    strb    wzr, [x22]
    b       iloop

finish:
    bl      flush
    mov     x8, #94
    mov     x0, #0
    svc     #0
    b       finish

fail:
    mov     x8, #94
    mov     x0, #1
    svc     #0
    b       fail

// flush: write outbuf[0..x26) to stdout, reset x26
flush:
    mov     x9, x30
    mov     x10, #0                 // written so far
flush_loop:
    subs    x2, x26, x10
    ble     flush_done
    mov     x8, #64
    mov     x0, #1
    add     x1, x25, x10
    svc     #0
    cmp     x0, #0
    ble     flush_done
    add     x10, x10, x0
    b       flush_loop
flush_done:
    mov     x26, #0
    ret     x9
