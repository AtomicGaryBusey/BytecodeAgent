.globl _start
_start:
        xor     ebx, ebx                  # total bytes read
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jle     read_done
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        xor     eax, eax                  # read
        xor     edi, edi                  # fd 0
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     rsi, [rip + inbuf]
        lea     rdi, [rip + outbuf]
        lea     r8, [rip + alphabet]
        xor     ecx, ecx                  # column counter
        mov     rdx, rbx                  # remaining bytes
enc_loop:
        cmp     rdx, 3
        jb      enc_tail
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        movzx   r9d, byte ptr [rsi + 1]
        shl     r9d, 8
        or      eax, r9d
        movzx   r9d, byte ptr [rsi + 2]
        or      eax, r9d
        add     rsi, 3
        sub     rdx, 3
        mov     r9d, eax
        shr     r9d, 18
        mov     r9b, byte ptr [r8 + r9]
        mov     byte ptr [rdi], r9b
        mov     r9d, eax
        shr     r9d, 12
        and     r9d, 63
        mov     r9b, byte ptr [r8 + r9]
        mov     byte ptr [rdi + 1], r9b
        mov     r9d, eax
        shr     r9d, 6
        and     r9d, 63
        mov     r9b, byte ptr [r8 + r9]
        mov     byte ptr [rdi + 2], r9b
        and     eax, 63
        mov     al, byte ptr [r8 + rax]
        mov     byte ptr [rdi + 3], al
        add     rdi, 4
        add     ecx, 4
        cmp     ecx, 76
        jne     enc_loop
        mov     byte ptr [rdi], 10
        inc     rdi
        xor     ecx, ecx
        jmp     enc_loop
enc_tail:
        test    rdx, rdx
        jz      enc_end
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        mov     byte ptr [rdi + 2], 61    # '='
        cmp     rdx, 2
        jne     1f
        movzx   r9d, byte ptr [rsi + 1]
        shl     r9d, 8
        or      eax, r9d
        mov     r9d, eax
        shr     r9d, 6
        and     r9d, 63
        mov     r9b, byte ptr [r8 + r9]
        mov     byte ptr [rdi + 2], r9b
1:
        mov     r9d, eax
        shr     r9d, 18
        mov     r9b, byte ptr [r8 + r9]
        mov     byte ptr [rdi], r9b
        mov     r9d, eax
        shr     r9d, 12
        and     r9d, 63
        mov     r9b, byte ptr [r8 + r9]
        mov     byte ptr [rdi + 1], r9b
        mov     byte ptr [rdi + 3], 61    # '='
        add     rdi, 4
        add     ecx, 4
        cmp     ecx, 76
        jne     enc_end
        mov     byte ptr [rdi], 10
        inc     rdi
        xor     ecx, ecx
enc_end:
        test    ecx, ecx
        jz      do_write
        mov     byte ptr [rdi], 10
        inc     rdi
do_write:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
write_loop:
        test    rdx, rdx
        jz      do_exit
        mov     eax, 1                    # write
        mov     edi, 1                    # fd 1
        syscall
        test    rax, rax
        jle     do_exit
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
do_exit:
        mov     eax, 231                  # exit_group
        xor     edi, edi
        syscall
