.text
.globl _start
_start:
	xor ebx, ebx
read_loop:
	mov edx, 65536
	sub edx, ebx
	jz read_done
	xor eax, eax
	xor edi, edi
	lea rsi, [rip + inbuf]
	add rsi, rbx
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	jmp read_loop
read_done:
	lea rsi, [rip + inbuf]
	xor ecx, ecx
	xor r12d, r12d
	xor r8d, r8d
	lea r13, [rip + arrA]
parse_loop:
	cmp ecx, ebx
	jae parse_end
	movzx eax, byte ptr [rsi + rcx]
	inc ecx
	cmp al, 10
	jne parse_loop
	lea eax, [rcx - 1]
	shl rax, 32
	or rax, r8
	mov qword ptr [r13 + r12*8], rax
	inc r12
	mov r8d, ecx
	jmp parse_loop
parse_end:
	cmp r8d, ebx
	jae no_final
	mov eax, ebx
	shl rax, 32
	or rax, r8
	mov qword ptr [r13 + r12*8], rax
	inc r12
no_final:
	lea r14, [rip + arrB]
	mov r15d, 1
ms_outer:
	cmp r15, r12
	jae ms_done
	xor r9d, r9d
ms_inner:
	cmp r9, r12
	jae ms_pass_done
	lea r10, [r9 + r15]
	cmp r10, r12
	cmova r10, r12
	lea r11, [r10 + r15]
	cmp r11, r12
	cmova r11, r12
	mov rax, r9
	mov rbx, r10
	mov rdx, r9
merge_loop:
	cmp rax, r10
	jae merge_rest_b
	cmp rbx, r11
	jae merge_rest_a
	mov rdi, qword ptr [r13 + rax*8]
	mov rbp, qword ptr [r13 + rbx*8]
	call cmp_le
	test r8d, r8d
	jz merge_take_b
	mov rdi, qword ptr [r13 + rax*8]
	mov qword ptr [r14 + rdx*8], rdi
	inc rax
	inc rdx
	jmp merge_loop
merge_take_b:
	mov rdi, qword ptr [r13 + rbx*8]
	mov qword ptr [r14 + rdx*8], rdi
	inc rbx
	inc rdx
	jmp merge_loop
merge_rest_a:
	cmp rax, r10
	jae merge_block_done
	mov rdi, qword ptr [r13 + rax*8]
	mov qword ptr [r14 + rdx*8], rdi
	inc rax
	inc rdx
	jmp merge_rest_a
merge_rest_b:
	cmp rbx, r11
	jae merge_block_done
	mov rdi, qword ptr [r13 + rbx*8]
	mov qword ptr [r14 + rdx*8], rdi
	inc rbx
	inc rdx
	jmp merge_rest_b
merge_block_done:
	mov r9, r11
	jmp ms_inner
ms_pass_done:
	xchg r13, r14
	add r15, r15
	jmp ms_outer
ms_done:
	lea rdi, [rip + outbuf]
	xor ebx, ebx
	lea rbp, [rip + inbuf]
out_loop:
	cmp rbx, r12
	jae out_done
	mov rax, qword ptr [r13 + rbx*8]
	mov rcx, rax
	shr rcx, 32
	sub ecx, eax
	mov eax, eax
	lea rsi, [rbp + rax]
	rep movsb
	mov byte ptr [rdi], 10
	inc rdi
	inc rbx
	jmp out_loop
out_done:
	lea rsi, [rip + outbuf]
	mov rdx, rdi
	sub rdx, rsi
write_loop:
	test rdx, rdx
	jz do_exit
	mov eax, 1
	mov edi, 1
	syscall
	test rax, rax
	jle do_exit
	add rsi, rax
	sub rdx, rax
	jmp write_loop
do_exit:
	mov eax, 231
	xor edi, edi
	syscall
	jmp do_exit

# rdi = entry A, rbp = entry B; returns r8d = 1 if A <= B else 0
# clobbers rcx, rsi, rdi, rbp, r8
cmp_le:
	mov rcx, rdi
	shr rcx, 32
	sub ecx, edi
	mov r8, rbp
	shr r8, 32
	sub r8d, ebp
	mov edi, edi
	mov ebp, ebp
	lea rsi, [rip + inbuf]
	add rdi, rsi
	add rsi, rbp
	xchg rsi, rdi
	mov rbp, rcx
	sub rbp, r8
	cmp rcx, r8
	cmova rcx, r8
	jrcxz cmp_prefix
	repe cmpsb
	je cmp_prefix
	setb r8b
	movzx r8d, r8b
	ret
cmp_prefix:
	xor r8d, r8d
	test rbp, rbp
	setle r8b
	ret

	.space 65536

	.space 262176
.balign 16
arrB:
	.space 262176
.balign 16
