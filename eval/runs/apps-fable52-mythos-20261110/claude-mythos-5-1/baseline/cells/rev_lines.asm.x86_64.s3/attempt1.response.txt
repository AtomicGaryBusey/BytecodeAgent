.text
.globl _start
_start:
        xor     ebx, ebx                # bytes read so far
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jle     done_read
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        xor     edi, edi
        xor     eax, eax                # read
        syscall
        test    rax, rax
        jle     done_read
        add     rbx, rax
        jmp     read_loop
done_read:
        lea     r8, [rip + inbuf]       # current line start
        lea     r9, [r8 + rbx]          # end of input
        lea     rdi, [rip + outbuf]     # output pointer
line_loop:
        cmp     r8, r9
        jae     finish
        mov     r10, r8
find_nl:
        cmp     r10, r9
        jae     found_nl
        cmp     byte ptr [r10], 10
        je      found_nl
        inc     r10
        jmp     find_nl
found_nl:
        mov     r11, r10
rev_loop:
        cmp     r11, r8
        jbe     rev_done
        dec     r11
        mov     al, byte ptr [r11]
        mov     byte ptr [rdi], al
        inc     rdi
        jmp     rev_loop
rev_done:
        mov     byte ptr [rdi], 10
        inc     rdi
        mov     r8, r10
        cmp     r8, r9
        jae     line_loop
        inc     r8                      # skip the newline
        jmp     line_loop
finish:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
        jz      do_exit
        mov     edi, 1
        mov     eax, 1                  # write
        syscall
do_exit:
        xor     edi, edi
        mov     eax, 231                # exit_group
        syscall
        jmp     do_exit

.bss
.balign 16
inbuf:
        .space  65536
.balign 16
outbuf:
        .space  65600
