.text
.globl _start
_start:
    xor r12d, r12d              # total bytes read
read_loop:
    mov edx, 32768
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # parse lines into (ptr,len) entries
    lea rsi, [rip + inbuf]
    lea rdi, [rip + lines]
    xor ecx, ecx                # line count
    xor r8d, r8d                # pos
    xor r9d, r9d                # line start
parse_loop:
    cmp r8, r12
    jae parse_end
    cmp byte ptr [rsi + r8], 10
    jne parse_next
    lea rax, [rsi + r9]
    mov qword ptr [rdi], rax
    mov rax, r8
    sub rax, r9
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc rcx
    lea r9, [r8 + 1]
parse_next:
    inc r8
    jmp parse_loop
parse_end:
    cmp r9, r12
    jae parse_done
    lea rax, [rsi + r9]
    mov qword ptr [rdi], rax
    mov rax, r12
    sub rax, r9
    mov qword ptr [rdi + 8], rax
    inc rcx
parse_done:
    mov r12, rcx                # n lines
    # bottom-up merge sort
    lea r14, [rip + lines]      # src
    lea r15, [rip + aux]        # dst
    mov ebx, 1                  # width
sort_outer:
    cmp rbx, r12
    jae sort_done
    xor ebp, ebp                # i
sort_inner:
    cmp rbp, r12
    jae sort_next_width
    mov rax, rbp
    add rax, rbx
    cmp rax, r12
    cmova rax, r12              # mid
    mov rdx, rax
    add rdx, rbx
    cmp rdx, r12
    cmova rdx, r12              # right
    mov r8, rbp
    shl r8, 4
    mov r13, r8
    add r13, r15                # dst cursor
    add r8, r14                 # left cursor
    shl rax, 4
    lea r9, [r14 + rax]         # left end
    mov r10, r9                 # right cursor
    shl rdx, 4
    lea r11, [r14 + rdx]        # right end
merge_loop:
    cmp r8, r9
    jae copy_right
    cmp r10, r11
    jae copy_left
    mov rsi, r8
    mov rdi, r10
    call cmp_le
    test eax, eax
    jz take_right
take_left:
    movups xmm0, xmmword ptr [r8]
    movups xmmword ptr [r13], xmm0
    add r8, 16
    add r13, 16
    jmp merge_loop
take_right:
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [r13], xmm0
    add r10, 16
    add r13, 16
    jmp merge_loop
copy_right:
    cmp r10, r11
    jae merge_done
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [r13], xmm0
    add r10, 16
    add r13, 16
    jmp copy_right
copy_left:
    cmp r8, r9
    jae merge_done
    movups xmm0, xmmword ptr [r8]
    movups xmmword ptr [r13], xmm0
    add r8, 16
    add r13, 16
    jmp copy_left
merge_done:
    lea rbp, [rbp + rbx*2]
    jmp sort_inner
sort_next_width:
    xchg r14, r15
    add rbx, rbx
    jmp sort_outer
sort_done:
    # build output
    lea rdi, [rip + outbuf]
    mov r8, r14
    mov r9, r12
out_loop:
    test r9, r9
    jz out_done
    mov rsi, qword ptr [r8]
    mov rcx, qword ptr [r8 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add r8, 16
    dec r9
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall

# cmp_le: rsi = entry A, rdi = entry B; eax = 1 if A <= B else 0
# clobbers rax, rcx, rdx, rsi, rdi
cmp_le:
    mov rax, qword ptr [rsi + 8]    # la
    mov rdx, qword ptr [rdi + 8]    # lb
    mov rcx, rax
    cmp rcx, rdx
    cmova rcx, rdx                  # min length
    mov rsi, qword ptr [rsi]
    mov rdi, qword ptr [rdi]
    jrcxz 1f
    repe cmpsb
    jb 2f
    ja 3f
1:
    cmp rax, rdx
    jbe 2f
3:
    xor eax, eax
    ret
2:
    mov eax, 1
    ret

.bss
.balign 16
inbuf:
    .space 32800
.balign 16
outbuf:
    .space 40000
.balign 16
lines:
    .space 16016
.balign 16
aux:
    .space 16016
