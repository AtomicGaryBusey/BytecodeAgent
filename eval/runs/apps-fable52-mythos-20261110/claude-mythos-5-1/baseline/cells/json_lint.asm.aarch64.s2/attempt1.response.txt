.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x21, #0                 // length
.Lread_loop:
    mov     x2, #65536
    sub     x2, x2, x21
    cmp     x2, #0
    b.le    .Lread_done
    mov     x0, #0
    add     x1, x19, x21
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    .Lread_done
    add     x21, x21, x0
    b       .Lread_loop
.Lread_done:
    mov     x20, #0                 // position
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.ne    fail
    mov     x0, #1
    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #3
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
// fail: print "error at <x20>\n", exit 1 (never returns)
// ---------------------------------------------------------------
fail:
    mov     x22, x20
    sub     sp, sp, #32
    add     x23, sp, #32
    mov     x24, #0
    mov     x4, #10
1:
    udiv    x5, x22, x4
    msub    x6, x5, x4, x22
    add     w6, w6, #0x30
    sub     x23, x23, #1
    strb    w6, [x23]
    add     x24, x24, #1
    mov     x22, x5
    cbnz    x22, 1b
    mov     x0, #1
    adrp    x1, msg_err
    add     x1, x1, :lo12:msg_err
    mov     x2, #9
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x1, x23
    mov     x2, x24
    mov     x8, #64
    svc     #0
    mov     x0, #1
    adrp    x1, msg_nl
    add     x1, x1, :lo12:msg_nl
    mov     x2, #1
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
// skip_ws (leaf)
// ---------------------------------------------------------------
skip_ws:
    ldrb    w0, [x19, x20]
    cmp     w0, #0x20
    b.eq    1f
    cmp     w0, #0x09
    b.eq    1f
    cmp     w0, #0x0A
    b.eq    1f
    cmp     w0, #0x0D
    b.ne    2f
1:
    add     x20, x20, #1
    b       skip_ws
2:
    ret

// ---------------------------------------------------------------
// expect_lit: x1 = asciz literal (leaf)
// ---------------------------------------------------------------
expect_lit:
1:
    ldrb    w2, [x1], #1
    cbz     w2, 2f
    ldrb    w0, [x19, x20]
    cmp     w0, w2
    b.ne    fail
    add     x20, x20, #1
    b       1b
2:
    ret

// ---------------------------------------------------------------
// parse_value
// ---------------------------------------------------------------
parse_value:
    stp     x29, x30, [sp, #-16]!
    ldrb    w0, [x19, x20]
    cmp     w0, #0x7B               // {
    b.eq    .Lv_obj
    cmp     w0, #0x5B               // [
    b.eq    .Lv_arr
    cmp     w0, #0x22               // "
    b.eq    .Lv_str
    cmp     w0, #0x2D               // -
    b.eq    .Lv_num
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    .Lv_num
    cmp     w0, #0x74               // t
    b.eq    .Lv_true
    cmp     w0, #0x66               // f
    b.eq    .Lv_false
    cmp     w0, #0x6E               // n
    b.eq    .Lv_null
    b       fail
.Lv_obj:
    bl      parse_object
    b       .Lv_ret
.Lv_arr:
    bl      parse_array
    b       .Lv_ret
.Lv_str:
    bl      parse_string
    b       .Lv_ret
.Lv_num:
    bl      parse_number
    b       .Lv_ret
.Lv_true:
    adrp    x1, lit_true
    add     x1, x1, :lo12:lit_true
    bl      expect_lit
    b       .Lv_ret
.Lv_false:
    adrp    x1, lit_false
    add     x1, x1, :lo12:lit_false
    bl      expect_lit
    b       .Lv_ret
.Lv_null:
    adrp    x1, lit_null
    add     x1, x1, :lo12:lit_null
    bl      expect_lit
.Lv_ret:
    ldp     x29, x30, [sp], #16
    ret

// ---------------------------------------------------------------
// parse_object
// ---------------------------------------------------------------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    ldrb    w0, [x19, x20]
    cmp     w0, #0x7D               // }
    b.eq    .Lo_end
.Lo_member:
    ldrb    w0, [x19, x20]
    cmp     w0, #0x22
    b.ne    fail
    bl      parse_string
    bl      skip_ws
    ldrb    w0, [x19, x20]
    cmp     w0, #0x3A               // :
    b.ne    fail
    add     x20, x20, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19, x20]
    cmp     w0, #0x2C               // ,
    b.eq    .Lo_comma
    cmp     w0, #0x7D
    b.ne    fail
.Lo_end:
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
.Lo_comma:
    add     x20, x20, #1
    bl      skip_ws
    b       .Lo_member

// ---------------------------------------------------------------
// parse_array
// ---------------------------------------------------------------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    ldrb    w0, [x19, x20]
    cmp     w0, #0x5D               // ]
    b.eq    .La_end
.La_elem:
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19, x20]
    cmp     w0, #0x2C
    b.eq    .La_comma
    cmp     w0, #0x5D
    b.ne    fail
.La_end:
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
.La_comma:
    add     x20, x20, #1
    bl      skip_ws
    b       .La_elem

// ---------------------------------------------------------------
// parse_string (leaf)
// ---------------------------------------------------------------
parse_string:
    add     x20, x20, #1
.Ls_loop:
    ldrb    w0, [x19, x20]
    cmp     w0, #0x22
    b.eq    .Ls_end
    cmp     w0, #0x20
    b.lo    fail
    cmp     w0, #0x5C
    b.eq    .Ls_esc
    add     x20, x20, #1
    b       .Ls_loop
.Ls_end:
    add     x20, x20, #1
    ret
.Ls_esc:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
    cmp     w0, #0x75               // u
    b.eq    .Ls_u
    cmp     w0, #0x22
    b.eq    .Ls_ok1
    cmp     w0, #0x5C
    b.eq    .Ls_ok1
    cmp     w0, #0x2F
    b.eq    .Ls_ok1
    cmp     w0, #0x62               // b
    b.eq    .Ls_ok1
    cmp     w0, #0x66               // f
    b.eq    .Ls_ok1
    cmp     w0, #0x6E               // n
    b.eq    .Ls_ok1
    cmp     w0, #0x72               // r
    b.eq    .Ls_ok1
    cmp     w0, #0x74               // t
    b.ne    fail
.Ls_ok1:
    add     x20, x20, #1
    b       .Ls_loop
.Ls_u:
    add     x20, x20, #1
    mov     w3, #4
.Ls_hex:
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    .Ls_hexok
    sub     w1, w0, #0x41
    cmp     w1, #5
    b.ls    .Ls_hexok
    sub     w1, w0, #0x61
    cmp     w1, #5
    b.hi    fail
.Ls_hexok:
    add     x20, x20, #1
    subs    w3, w3, #1
    b.ne    .Ls_hex
    b       .Ls_loop

// ---------------------------------------------------------------
// parse_number (leaf)
// ---------------------------------------------------------------
parse_number:
    ldrb    w0, [x19, x20]
    cmp     w0, #0x2D
    b.ne    1f
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
1:
    cmp     w0, #0x30
    b.eq    .Ln_zero
    sub     w1, w0, #0x31
    cmp     w1, #8
    b.hi    fail
.Ln_int:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    .Ln_int
    b       .Ln_frac
.Ln_zero:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
.Ln_frac:
    cmp     w0, #0x2E               // .
    b.ne    .Ln_exp
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail
.Ln_fracd:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    .Ln_fracd
.Ln_exp:
    cmp     w0, #0x65               // e
    b.eq    1f
    cmp     w0, #0x45               // E
    b.ne    .Ln_done
1:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
    cmp     w0, #0x2B               // +
    b.eq    2f
    cmp     w0, #0x2D               // -
    b.ne    3f
2:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
3:
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.hi    fail
.Ln_expd:
    add     x20, x20, #1
    ldrb    w0, [x19, x20]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    .Ln_expd
.Ln_done:
    ret
