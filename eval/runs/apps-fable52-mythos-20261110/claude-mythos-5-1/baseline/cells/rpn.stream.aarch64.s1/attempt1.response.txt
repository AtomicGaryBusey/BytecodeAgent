.text
.globl _start

_start:
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, x19
        movz    x26, #0x0800
        movk    x26, #1, lsl #16
        add     x26, x26, x19
read_loop:
        sub     x2, x26, x20
        cbz     x2, read_done
        mov     x0, #0
        mov     x1, x20
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    read_done
        add     x20, x20, x0
        b       read_loop
read_done:
        adrp    x21, outbuf
        add     x21, x21, :lo12:outbuf
        mov     x22, x21
        adrp    x23, vstack
        add     x23, x23, :lo12:vstack

main_loop:
        cmp     x19, x20
        b.hs    finish
        mov     x25, x19
find_eol:
        cmp     x25, x20
        b.hs    eol_found
        ldrb    w0, [x25]
        cmp     w0, #10
        b.eq    eol_found
        add     x25, x25, #1
        b       find_eol
eol_found:
        mov     x24, x23
tok_loop:
        cmp     x19, x25
        b.hs    end_of_line
        ldrb    w0, [x19]
        cmp     w0, #32
        b.eq    skip_sep
        cmp     w0, #9
        b.eq    skip_sep
        mov     x10, x19
scan_tok:
        add     x19, x19, #1
        cmp     x19, x25
        b.hs    tok_end
        ldrb    w0, [x19]
        cmp     w0, #32
        b.eq    tok_end
        cmp     w0, #9
        b.ne    scan_tok
tok_end:
        sub     x11, x19, x10
        ldrb    w0, [x10]
        cmp     x11, #1
        b.ne    literal
        cmp     w0, #43
        b.eq    op_add
        cmp     w0, #45
        b.eq    op_sub
        cmp     w0, #42
        b.eq    op_mul
        cmp     w0, #47
        b.eq    op_div
        cmp     w0, #37
        b.eq    op_mod
literal:
        mov     x12, #0
        cmp     w0, #45
        b.ne    1f
        mov     x12, #1
        add     x10, x10, #1
1:
        sub     x11, x19, x10
        cmp     x11, #1
        b.lt    line_error
        cmp     x11, #18
        b.gt    line_error
        mov     x13, #0
        mov     x14, #10
digit_loop:
        ldrb    w0, [x10], #1
        sub     w0, w0, #48
        cmp     w0, #9
        b.hi    line_error
        madd    x13, x13, x14, x0
        cmp     x10, x19
        b.lo    digit_loop
        cbz     x12, 2f
        neg     x13, x13
2:
        str     x13, [x24], #8
        b       tok_loop
skip_sep:
        add     x19, x19, #1
        b       tok_loop

op_add:
        sub     x9, x24, x23
        cmp     x9, #16
        b.lo    line_error
        ldp     x0, x1, [x24, #-16]!
        add     x0, x0, x1
        str     x0, [x24], #8
        b       tok_loop
op_sub:
        sub     x9, x24, x23
        cmp     x9, #16
        b.lo    line_error
        ldp     x0, x1, [x24, #-16]!
        sub     x0, x0, x1
        str     x0, [x24], #8
        b       tok_loop
op_mul:
        sub     x9, x24, x23
        cmp     x9, #16
        b.lo    line_error
        ldp     x0, x1, [x24, #-16]!
        mul     x0, x0, x1
        str     x0, [x24], #8
        b       tok_loop
op_div:
        sub     x9, x24, x23
        cmp     x9, #16
        b.lo    line_error
        ldp     x0, x1, [x24, #-16]!
        cbz     x1, line_error
        cmn     x1, #1
        b.ne    1f
        movz    x9, #0x8000, lsl #48
        cmp     x0, x9
        b.eq    line_error
1:
        sdiv    x0, x0, x1
        str     x0, [x24], #8
        b       tok_loop
op_mod:
        sub     x9, x24, x23
        cmp     x9, #16
        b.lo    line_error
        ldp     x0, x1, [x24, #-16]!
        cbz     x1, line_error
        cmn     x1, #1
        b.ne    1f
        movz    x9, #0x8000, lsl #48
        cmp     x0, x9
        b.eq    line_error
1:
        sdiv    x9, x0, x1
        msub    x0, x9, x1, x0
        str     x0, [x24], #8
        b       tok_loop

end_of_line:
        sub     x9, x24, x23
        cmp     x9, #8
        b.ne    line_error
        ldr     x0, [x23]
        bl      print_num
        b       next_line
line_error:
        bl      ensure_out
        movz    x9, #0x7265
        movk    x9, #0x6f72, lsl #16
        movk    x9, #0x0a72, lsl #32
        str     x9, [x22]
        add     x22, x22, #6
next_line:
        cmp     x25, x20
        b.hs    1f
        add     x25, x25, #1
1:
        mov     x19, x25
        b       main_loop

finish:
        bl      flush
        mov     x0, #0
        mov     x8, #94
        svc     #0
        b       finish

// flush: write [x21, x22) to stdout, reset x22
flush:
        mov     x9, x21
1:
        subs    x2, x22, x9
        b.le    2f
        mov     x0, #1
        mov     x1, x9
        mov     x8, #64
        svc     #0
        cmp     x0, #0
        b.le    2f
        add     x9, x9, x0
        b       1b
2:
        mov     x22, x21
        ret

// ensure at least 32 bytes free in output buffer
ensure_out:
        sub     x9, x22, x21
        mov     x10, #0xffe0
        cmp     x9, x10
        b.lo    1f
        b       flush
1:
        ret

// print_num: x0 = signed value, append decimal + newline to output
print_num:
        stp     x29, x30, [sp, #-16]!
        mov     x11, x0
        bl      ensure_out
        sub     sp, sp, #32
        add     x12, sp, #32
        mov     x13, x12
        cmp     x11, #0
        b.ge    1f
        neg     x11, x11
        mov     w9, #45
        strb    w9, [x22], #1
1:
        mov     x14, #10
2:
        udiv    x9, x11, x14
        msub    x10, x9, x14, x11
        add     w10, w10, #48
        strb    w10, [x13, #-1]!
        mov     x11, x9
        cbnz    x11, 2b
3:
        ldrb    w9, [x13], #1
        strb    w9, [x22], #1
        cmp     x13, x12
        b.lo    3b
        mov     w9, #10
        strb    w9, [x22], #1
        add     sp, sp, #32
        ldp     x29, x30, [sp], #16
        ret
