.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 98304

.section .rodata
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov edx, 65536
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + table]
    mov rcx, r12
    xor r9d, r9d
enc_loop:
    cmp rcx, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx ebx, byte ptr [rsi + 2]
    shl eax, 16
    shl edx, 8
    or eax, edx
    or eax, ebx
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl
    and eax, 63
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], al
    add rsi, 3
    add rdi, 4
    sub rcx, 3
    add r9d, 4
    cmp r9d, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp enc_loop
enc_tail:
    test rcx, rcx
    jz enc_end
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rcx, 2
    jb tail1
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
tail1:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    cmp rcx, 2
    jb pad2
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], 0x3D
    jmp tailpad_done
pad2:
    mov word ptr [rdi + 2], 0x3D3D
tailpad_done:
    add rdi, 4
    add r9d, 4
enc_end:
    test r9d, r9d
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi
write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
