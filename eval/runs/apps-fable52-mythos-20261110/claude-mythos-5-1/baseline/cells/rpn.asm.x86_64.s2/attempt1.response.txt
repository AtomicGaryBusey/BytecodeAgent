```
.equ INSIZE, 65600
.equ OUTSIZE, 65536

.text
.globl _start
_start:
    xor r13d, r13d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov edx, INSIZE
    sub edx, r13d
    jle read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    add r13, r12
    lea r14, [rip + outbuf]
    lea rbx, [rip + vstack]

line_loop:
    cmp r12, r13
    jae finish
    xor r15d, r15d
tok_loop:
    cmp r12, r13
    jae end_line
    movzx eax, byte ptr [r12]
    cmp al, 10
    je end_line
    cmp al, 32
    je sep
    cmp al, 9
    je sep
    mov rdi, r12
tok_scan:
    inc r12
    cmp r12, r13
    jae tok_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je tok_end
    cmp al, 32
    je tok_end
    cmp al, 9
    je tok_end
    jmp tok_scan
tok_end:
    mov rcx, r12
    sub rcx, rdi
    movzx eax, byte ptr [rdi]
    cmp rcx, 1
    jne parse_num
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
parse_num:
    xor edx, edx
    cmp al, '-'
    jne pn_digits
    inc rdi
    dec rcx
    mov edx, 1
pn_digits:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
pn_loop:
    movzx esi, byte ptr [rdi]
    sub esi, '0'
    cmp esi, 9
    ja line_error
    lea rax, [rax + rax*4]
    lea rax, [rsi + rax*2]
    inc rdi
    dec rcx
    jnz pn_loop
    test edx, edx
    jz pn_push
    neg rax
pn_push:
    mov qword ptr [rbx + r15*8], rax
    inc r15
    jmp tok_loop
sep:
    inc r12
    jmp tok_loop

op_add:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbx + r15*8 - 8]
    add qword ptr [rbx + r15*8 - 16], rax
    dec r15
    jmp tok_loop
op_sub:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbx + r15*8 - 8]
    sub qword ptr [rbx + r15*8 - 16], rax
    dec r15
    jmp tok_loop
op_mul:
    cmp r15, 2
    jb line_error
    mov rax, qword ptr [rbx + r15*8 - 16]
    imul rax, qword ptr [rbx + r15*8 - 8]
    mov qword ptr [rbx + r15*8 - 16], rax
    dec r15
    jmp tok_loop
op_div:
    cmp r15, 2
    jb line_error
    mov rcx, qword ptr [rbx + r15*8 - 8]
    test rcx, rcx
    jz line_error
    mov rax, qword ptr [rbx + r15*8 - 16]
    cmp rcx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
1:
    cqo
    idiv rcx
    mov qword ptr [rbx + r15*8 - 16], rax
    dec r15
    jmp tok_loop
op_mod:
    cmp r15, 2
    jb line_error
    mov rcx, qword ptr [rbx + r15*8 - 8]
    test rcx, rcx
    jz line_error
    mov rax, qword ptr [rbx + r15*8 - 16]
    cmp rcx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
1:
    cqo
    idiv rcx
    mov qword ptr [rbx + r15*8 - 16], rdx
    dec r15
    jmp tok_loop

end_line:
    cmp r12, r13
    jae 1f
    inc r12
1:
    cmp r15, 1
    jne emit_error
    call ensure_space
    mov rax, qword ptr [rbx]
    call print_num
    jmp line_loop

line_error:
    cmp r12, r13
    jae emit_error
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 10
    jne line_error
emit_error:
    call ensure_space
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp line_loop

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp finish

# rax = value; assumes ensure_space already called
print_num:
    test rax, rax
    jns 1f
    mov byte ptr [r14], '-'
    inc r14
    neg rax
1:
    sub rsp, 32
    lea rdi, [rsp + 31]
    mov rsi, rdi
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [rdi]
    mov byte ptr [r14], al
    inc r14
    inc rdi
    cmp rdi, rsi
    jb 3b
    mov byte ptr [r14], 10
    inc r14
    add rsp, 32
    ret

ensure_space:
    lea rax, [rip + outbuf + OUTSIZE - 32]
    cmp r14, rax
    jb 1f
    call flush
1:
    ret

flush:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    jz 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    lea r14, [rip + outbuf]
    ret

.bss
.balign 16
inbuf:
    .space INSIZE
.balign 16
outbuf:
    .space OUTSIZE
.balign 16
vstack:
    .space 8192
```
