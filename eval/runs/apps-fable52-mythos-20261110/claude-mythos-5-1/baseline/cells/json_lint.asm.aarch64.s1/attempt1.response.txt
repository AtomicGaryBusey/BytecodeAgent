.text
.globl _start

_start:
    adrp    x20, buf
    add     x20, x20, :lo12:buf
    mov     x21, x20
    mov     x22, #0x10000
read_loop:
    add     x2, x20, x22
    sub     x2, x2, x21
    cbz     x2, read_done
    mov     x0, #0
    mov     x1, x21
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x21, x21, x0
    b       read_loop
read_done:
    mov     x19, x20
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x19, x21
    bne     error
    adrp    x1, ok_msg
    add     x1, x1, :lo12:ok_msg
    mov     x2, #3
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
error:
    sub     x0, x19, x20
    sub     sp, sp, #48
    add     x1, sp, #47
    mov     w3, #10
    strb    w3, [x1]
    mov     x4, #10
err_digits:
    udiv    x5, x0, x4
    msub    x6, x5, x4, x0
    add     w6, w6, #0x30
    sub     x1, x1, #1
    strb    w6, [x1]
    mov     x0, x5
    cbnz    x0, err_digits
    adrp    x2, err_msg
    add     x2, x2, :lo12:err_msg
    mov     x3, #9
err_copy:
    sub     x3, x3, #1
    ldrb    w4, [x2, x3]
    sub     x1, x1, #1
    strb    w4, [x1]
    cbnz    x3, err_copy
    add     x2, sp, #48
    sub     x2, x2, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .

// ---------------------------------------------------------------
skip_ws:
    ldrb    w0, [x19]
    cmp     w0, #0x20
    beq     skip_ws_adv
    cmp     w0, #0x09
    beq     skip_ws_adv
    cmp     w0, #0x0A
    beq     skip_ws_adv
    cmp     w0, #0x0D
    bne     skip_ws_done
skip_ws_adv:
    add     x19, x19, #1
    b       skip_ws
skip_ws_done:
    ret

// ---------------------------------------------------------------
parse_value:
    ldrb    w0, [x19]
    cmp     w0, #0x7B
    beq     parse_object
    cmp     w0, #0x5B
    beq     parse_array
    cmp     w0, #0x22
    beq     parse_string
    cmp     w0, #0x2D
    beq     parse_number
    sub     w1, w0, #0x30
    cmp     w1, #9
    bls     parse_number
    cmp     w0, #0x74
    beq     pv_true
    cmp     w0, #0x66
    beq     pv_false
    cmp     w0, #0x6E
    beq     pv_null
    b       error
pv_true:
    adrp    x0, lit_true
    add     x0, x0, :lo12:lit_true
    b       match_lit
pv_false:
    adrp    x0, lit_false
    add     x0, x0, :lo12:lit_false
    b       match_lit
pv_null:
    adrp    x0, lit_null
    add     x0, x0, :lo12:lit_null
    b       match_lit

// x0 = pointer to NUL-terminated literal
match_lit:
    ldrb    w1, [x0], #1
    cbz     w1, match_done
    ldrb    w2, [x19]
    cmp     w1, w2
    bne     error
    add     x19, x19, #1
    b       match_lit
match_done:
    ret

// ---------------------------------------------------------------
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x19, x19, #1
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x7D
    beq     obj_end
obj_member:
    ldrb    w0, [x19]
    cmp     w0, #0x22
    bne     error
    bl      parse_string
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x3A
    bne     error
    add     x19, x19, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x2C
    beq     obj_comma
    cmp     w0, #0x7D
    bne     error
obj_end:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret
obj_comma:
    add     x19, x19, #1
    bl      skip_ws
    b       obj_member

// ---------------------------------------------------------------
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x19, x19, #1
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x5D
    beq     arr_end
arr_elem:
    bl      parse_value
    bl      skip_ws
    ldrb    w0, [x19]
    cmp     w0, #0x2C
    beq     arr_comma
    cmp     w0, #0x5D
    bne     error
arr_end:
    add     x19, x19, #1
    ldp     x29, x30, [sp], #16
    ret
arr_comma:
    add     x19, x19, #1
    bl      skip_ws
    b       arr_elem

// ---------------------------------------------------------------
parse_string:
    add     x19, x19, #1
str_loop:
    ldrb    w0, [x19]
    cmp     w0, #0x22
    beq     str_end
    cmp     w0, #0x5C
    beq     str_esc
    cmp     w0, #0x20
    blo     error
    add     x19, x19, #1
    b       str_loop
str_end:
    add     x19, x19, #1
    ret
str_esc:
    add     x19, x19, #1
    ldrb    w0, [x19]
    cmp     w0, #0x75
    beq     str_uesc
    cmp     w0, #0x22
    beq     str_esc_ok
    cmp     w0, #0x5C
    beq     str_esc_ok
    cmp     w0, #0x2F
    beq     str_esc_ok
    cmp     w0, #0x62
    beq     str_esc_ok
    cmp     w0, #0x66
    beq     str_esc_ok
    cmp     w0, #0x6E
    beq     str_esc_ok
    cmp     w0, #0x72
    beq     str_esc_ok
    cmp     w0, #0x74
    beq     str_esc_ok
    b       error
str_esc_ok:
    add     x19, x19, #1
    b       str_loop
str_uesc:
    add     x19, x19, #1
    mov     w3, #4
str_hex:
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    bls     str_hex_ok
    orr     w1, w0, #0x20
    sub     w1, w1, #0x61
    cmp     w1, #5
    bhi     error
str_hex_ok:
    add     x19, x19, #1
    subs    w3, w3, #1
    bne     str_hex
    b       str_loop

// ---------------------------------------------------------------
parse_number:
    ldrb    w0, [x19]
    cmp     w0, #0x2D
    bne     num_int
    add     x19, x19, #1
    ldrb    w0, [x19]
num_int:
    cmp     w0, #0x30
    bne     num_nonzero
    add     x19, x19, #1
    b       num_frac
num_nonzero:
    sub     w1, w0, #0x31
    cmp     w1, #8
    bhi     error
    add     x19, x19, #1
num_int_digits:
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    bhi     num_frac
    add     x19, x19, #1
    b       num_int_digits
num_frac:
    ldrb    w0, [x19]
    cmp     w0, #0x2E
    bne     num_exp
    add     x19, x19, #1
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    bhi     error
num_frac_digits:
    add     x19, x19, #1
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    bls     num_frac_digits
num_exp:
    cmp     w0, #0x65
    beq     num_exp_go
    cmp     w0, #0x45
    bne     num_done
num_exp_go:
    add     x19, x19, #1
    ldrb    w0, [x19]
    cmp     w0, #0x2B
    beq     num_exp_sign
    cmp     w0, #0x2D
    bne     num_exp_first
num_exp_sign:
    add     x19, x19, #1
    ldrb    w0, [x19]
num_exp_first:
    sub     w1, w0, #0x30
    cmp     w1, #9
    bhi     error
num_exp_digits:
    add     x19, x19, #1
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    bls     num_exp_digits
num_done:
    ret

// ---------------------------------------------------------------
.section .rodata
ok_msg:
    .ascii  "ok\n"
err_msg:
    .ascii  "error at "
lit_true:
    .asciz  "true"
lit_false:
    .asciz  "false"
lit_null:
    .asciz  "null"

.bss
.balign 16
buf:
    .space  65552
