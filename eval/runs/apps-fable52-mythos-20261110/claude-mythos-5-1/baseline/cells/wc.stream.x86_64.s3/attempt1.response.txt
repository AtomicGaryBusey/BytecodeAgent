.text
.globl _start
_start:
        lea     rbx, [rip + buf]
        xor     r12d, r12d              # total bytes read
read_loop:
        mov     edx, BUFSIZE
        sub     edx, r12d
        jz      done_read
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        lea     rsi, [rbx + r12]
        syscall
        test    rax, rax
        jle     done_read
        add     r12, rax
        jmp     read_loop
done_read:
        xor     r13d, r13d              # lines
        xor     r14d, r14d              # words
        xor     r15d, r15d              # in_word flag
        xor     ecx, ecx
count_loop:
        cmp     rcx, r12
        jae     count_done
        movzx   eax, byte ptr [rbx + rcx]
        inc     rcx
        cmp     al, 0x20
        je      ws
        lea     edx, [rax - 9]
        cmp     dl, 4
        jbe     ws
        test    r15d, r15d
        jnz     count_loop
        inc     r14
        mov     r15d, 1
        jmp     count_loop
ws:
        xor     r15d, r15d
        cmp     al, 10
        jne     count_loop
        inc     r13
        jmp     count_loop
count_done:
        lea     rdi, [rip + outbuf + 64]
        dec     rdi
        mov     byte ptr [rdi], 10
        mov     rax, r12
        call    putnum
        dec     rdi
        mov     byte ptr [rdi], 0x20
        mov     rax, r14
        call    putnum
        dec     rdi
        mov     byte ptr [rdi], 0x20
        mov     rax, r13
        call    putnum
        mov     rsi, rdi
        lea     rdx, [rip + outbuf + 64]
        sub     rdx, rsi
        mov     edi, 1
        mov     eax, 1
        syscall
        mov     eax, 231
        xor     edi, edi
        syscall
        hlt

# rax = number, rdi = end pointer; writes digits backwards, updates rdi
putnum:
        mov     ecx, 10
1:
        xor     edx, edx
        div     rcx
        add     dl, 0x30
        dec     rdi
        mov     byte ptr [rdi], dl
        test    rax, rax
        jnz     1b
        ret

.bss
.set BUFSIZE, 69632
.balign 16
