.text
.globl _start
_start:
    xor r12d, r12d              # lines
    xor r13d, r13d              # words
    xor r14d, r14d              # bytes
    xor r15d, r15d              # in-word flag
read_loop:
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + inbuf]
    lea rdx, [rsi + rax]
scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    lea ecx, [rax - 9]
    cmp ecx, 4
    jbe ws
    test r15, r15
    jnz 2f
    inc r13
    mov r15d, 1
2:
    cmp rsi, rdx
    jb scan
    jmp read_loop
ws:
    xor r15d, r15d
    cmp rsi, rdx
    jb scan
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call put_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call put_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1                  # write
    mov edi, 1
    syscall
    mov eax, 231                # exit_group
    xor edi, edi
    syscall

# put_num: rax = value, rdi = destination; writes decimal digits, advances rdi
put_num:
    lea rsi, [rip + tmp + 24]
    mov r8, rsi
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    cmp rsi, r8
    jb 2b
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 128
tmp:
    .space 32
