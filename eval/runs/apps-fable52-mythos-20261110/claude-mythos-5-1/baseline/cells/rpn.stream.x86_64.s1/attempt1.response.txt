.text
.globl _start

.equ INBUF_SIZE, 65552
.equ OUTBUF_SIZE, 524288
.equ VSTACK_SIZE, 8192

_start:
    sub rsp, 32
    xor r13d, r13d
read_loop:
    mov edx, INBUF_SIZE
    sub edx, r13d
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, r13
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    add r13, r12
    lea r14, [rip + outbuf]

main_loop:
    cmp r12, r13
    jae finish
    lea rbx, [rip + vstack]
tok_loop:
    cmp r12, r13
    jae end_line
    movzx eax, byte ptr [r12]
    cmp al, 10
    je end_line
    cmp al, 32
    je sep
    cmp al, 9
    je sep
    mov rsi, r12
find_end:
    inc r12
    cmp r12, r13
    jae have_token
    movzx eax, byte ptr [r12]
    cmp al, 10
    je have_token
    cmp al, 32
    je have_token
    cmp al, 9
    je have_token
    jmp find_end
sep:
    inc r12
    jmp tok_loop

have_token:
    mov rcx, r12
    sub rcx, rsi
    movzx eax, byte ptr [rsi]
    cmp rcx, 1
    jne parse_num
    cmp al, '+'
    je do_op
    cmp al, '-'
    je do_op
    cmp al, '*'
    je do_op
    cmp al, '/'
    je do_op
    cmp al, '%'
    je do_op
parse_num:
    xor edx, edx
    cmp al, '-'
    jne pn_digits
    mov edx, 1
    inc rsi
    dec rcx
pn_digits:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
pn_loop:
    movzx edi, byte ptr [rsi]
    sub edi, '0'
    cmp edi, 9
    ja line_error
    lea rax, [rax + rax*4]
    lea rax, [rdi + rax*2]
    inc rsi
    dec rcx
    jnz pn_loop
    test edx, edx
    jz pn_push
    neg rax
pn_push:
    mov qword ptr [rbx], rax
    add rbx, 8
    jmp tok_loop

do_op:
    mov rdx, rbx
    lea rcx, [rip + vstack]
    sub rdx, rcx
    cmp rdx, 16
    jb line_error
    sub rbx, 8
    mov rcx, qword ptr [rbx]
    mov rdx, qword ptr [rbx - 8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne op_divmod
    mov r8, 0x8000000000000000
    cmp rdx, r8
    je line_error
op_divmod:
    mov r8d, eax
    mov rax, rdx
    cqo
    idiv rcx
    cmp r8b, '/'
    je op_store
    mov rax, rdx
    jmp op_store
op_add:
    mov rax, rdx
    add rax, rcx
    jmp op_store
op_sub:
    mov rax, rdx
    sub rax, rcx
    jmp op_store
op_mul:
    mov rax, rdx
    imul rax, rcx
op_store:
    mov qword ptr [rbx - 8], rax
    jmp tok_loop

line_error:
    cmp r12, r13
    jae le_out
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 10
    jne line_error
le_out:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp after_line

end_line:
    cmp r12, r13
    jae 1f
    inc r12
1:
    lea rax, [rip + vstack]
    sub rax, rbx
    cmp rax, -8
    jne le_out
    mov rax, qword ptr [rbx - 8]
    test rax, rax
    jns 2f
    mov byte ptr [r14], '-'
    inc r14
    neg rax
2:
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 10
    mov ecx, 10
3:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 3b
    lea rcx, [rsp + 32]
    sub rcx, rdi
4:
    mov al, byte ptr [rdi]
    mov byte ptr [r14], al
    inc rdi
    inc r14
    dec rcx
    jnz 4b

after_line:
    lea rax, [rip + outbuf]
    mov rdx, r14
    sub rdx, rax
    cmp rdx, OUTBUF_SIZE - 64
    jb main_loop
    call flush
    jmp main_loop

flush:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
fl_loop:
    test rdx, rdx
    jz fl_done
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle fl_done
    add rsi, rax
    sub rdx, rax
    jmp fl_loop
fl_done:
    lea r14, [rip + outbuf]
    ret

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp finish

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 16
vstack:
    .space VSTACK_SIZE
