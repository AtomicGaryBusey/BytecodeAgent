.bss
.balign 16
inbuf:  .space 65536
outbuf: .space 65552

.text
.globl _start
_start:
    xor ebx, ebx                # rbx = bytes read so far
read_loop:
    mov edx, 65536
    sub edx, ebx
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]      # rsi = start of current line
    lea r8, [rsi + rbx]         # r8  = end of input
    lea rdi, [rip + outbuf]     # rdi = output write pointer
line_loop:
    cmp rsi, r8
    jae done_lines
    mov rcx, rsi
find_nl:
    cmp rcx, r8
    jae found_end
    cmp byte ptr [rcx], 10
    je found_end
    inc rcx
    jmp find_nl
found_end:
    mov rdx, rcx                # line is [rsi, rcx)
rev_loop:
    cmp rdx, rsi
    je rev_done
    dec rdx
    mov al, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev_loop
rev_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rcx + 1]          # skip the newline
    jmp line_loop
done_lines:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
