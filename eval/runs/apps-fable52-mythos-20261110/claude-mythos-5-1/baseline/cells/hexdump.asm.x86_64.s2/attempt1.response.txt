.text
.globl _start
_start:
    lea r8, [rip + hexdigits]
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor r13d, r13d
row_loop:
    cmp r13, rbx
    jae finish
    mov eax, r13d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    mov r14, rbx
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    lea rsi, [rip + inbuf]
    add rsi, r13
    xor ecx, ecx
hex_loop:
    cmp ecx, r14d
    jae pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp next
pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
next:
    add rdi, 3
    cmp ecx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc ecx
    cmp ecx, 16
    jb hex_loop
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx
chr_loop:
    cmp ecx, r14d
    jae chr_done
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 3f
    mov eax, 0x2e
3:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp chr_loop
chr_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r14
    jmp row_loop
finish:
    mov eax, ebx
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 231
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 327680
