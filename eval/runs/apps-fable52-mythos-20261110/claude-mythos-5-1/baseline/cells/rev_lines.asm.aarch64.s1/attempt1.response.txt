	.text
	.globl _start
_start:
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0			// input length
	mov	x21, #0x10000
	add	x21, x21, #64		// capacity
read_loop:
	cmp	x20, x21
	b.hs	read_done
	mov	x0, #0
	add	x1, x19, x20
	sub	x2, x21, x20
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	read_done
	add	x20, x20, x0
	b	read_loop
read_done:
	adrp	x22, outbuf
	add	x22, x22, :lo12:outbuf
	mov	x23, #0			// output length
	mov	x9, #0			// line start i
proc_loop:
	cmp	x9, x20
	b.hs	proc_done
	mov	x10, x9			// j
find_nl:
	cmp	x10, x20
	b.hs	found
	ldrb	w11, [x19, x10]
	cmp	w11, #10
	b.eq	found
	add	x10, x10, #1
	b	find_nl
found:
	mov	x12, x10
rev_loop:
	cmp	x12, x9
	b.ls	rev_done
	sub	x12, x12, #1
	ldrb	w11, [x19, x12]
	strb	w11, [x22, x23]
	add	x23, x23, #1
	b	rev_loop
rev_done:
	mov	w11, #10
	strb	w11, [x22, x23]
	add	x23, x23, #1
	add	x9, x10, #1
	b	proc_loop
proc_done:
	mov	x24, #0			// bytes written
write_loop:
	cmp	x24, x23
	b.hs	done
	mov	x0, #1
	add	x1, x22, x24
	sub	x2, x23, x24
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	b.le	done
	add	x24, x24, x0
	b	write_loop
done:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.bss
	.balign 16
inbuf:
	.space	65600
	.balign 16
