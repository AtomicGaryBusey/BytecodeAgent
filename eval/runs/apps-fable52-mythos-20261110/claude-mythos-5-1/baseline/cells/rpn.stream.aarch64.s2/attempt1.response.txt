.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, x19
    movz    x26, #0x0010
    movk    x26, #1, lsl #16
read_loop:
    add     x2, x19, x26
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    mov     x1, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x22, #0
    adrp    x23, stackbuf
    add     x23, x23, :lo12:stackbuf

line_loop:
    cmp     x19, x20
    b.hs    finish
    mov     x25, x19
find_eol:
    cmp     x25, x20
    b.hs    eol_found
    ldrb    w0, [x25]
    cmp     w0, #10
    b.eq    eol_found
    add     x25, x25, #1
    b       find_eol
eol_found:
    bl      eval_line
    cbnz    x1, line_error
    bl      print_number
    b       next_line
line_error:
    bl      print_err
next_line:
    add     x19, x25, #1
    b       line_loop
finish:
    bl      flush
    mov     x0, #0
    mov     x8, #94
    svc     #0
hang:
    b       hang

// eval_line: evaluates [x19, x25). Returns x0 = value, x1 = 0 ok / 1 error.
eval_line:
    mov     x24, #0
    mov     x9, x19
tok_loop:
    cmp     x9, x25
    b.hs    tok_end
    ldrb    w10, [x9]
    cmp     w10, #0x20
    b.eq    tok_skip
    cmp     w10, #0x09
    b.eq    tok_skip
    mov     x11, x9
tok_scan:
    add     x9, x9, #1
    cmp     x9, x25
    b.hs    tok_scanned
    ldrb    w10, [x9]
    cmp     w10, #0x20
    b.eq    tok_scanned
    cmp     w10, #0x09
    b.ne    tok_scan
tok_scanned:
    sub     x12, x9, x11
    ldrb    w10, [x11]
    cmp     x12, #1
    b.ne    tok_number
    cmp     w10, #'+'
    b.eq    op_add
    cmp     w10, #'-'
    b.eq    op_sub
    cmp     w10, #'*'
    b.eq    op_mul
    cmp     w10, #'/'
    b.eq    op_div
    cmp     w10, #'%'
    b.eq    op_mod
tok_number:
    mov     x13, #0
    cmp     w10, #'-'
    b.ne    tok_digits
    mov     x13, #1
    add     x11, x11, #1
    sub     x12, x12, #1
tok_digits:
    cmp     x12, #0
    b.eq    eval_error
    cmp     x12, #18
    b.hi    eval_error
    mov     x14, #0
    mov     x15, #10
tok_dig_loop:
    ldrb    w10, [x11], #1
    sub     w10, w10, #'0'
    cmp     w10, #9
    b.hi    eval_error
    madd    x14, x14, x15, x10
    subs    x12, x12, #1
    b.ne    tok_dig_loop
    cbz     x13, tok_push
    neg     x14, x14
tok_push:
    str     x14, [x23, x24, lsl #3]
    add     x24, x24, #1
    b       tok_loop
tok_skip:
    add     x9, x9, #1
    b       tok_loop

op_add:
    mov     x27, #0
    b       op_common
op_sub:
    mov     x27, #1
    b       op_common
op_mul:
    mov     x27, #2
    b       op_common
op_div:
    mov     x27, #3
    b       op_common
op_mod:
    mov     x27, #4
op_common:
    cmp     x24, #2
    b.lo    eval_error
    sub     x24, x24, #2
    add     x10, x23, x24, lsl #3
    ldp     x16, x17, [x10]
    cbz     x27, do_add
    cmp     x27, #1
    b.eq    do_sub
    cmp     x27, #2
    b.eq    do_mul
    cbz     x17, eval_error
    cmn     x17, #1
    b.ne    do_divok
    mov     x14, #1
    lsl     x14, x14, #63
    cmp     x16, x14
    b.eq    eval_error
do_divok:
    sdiv    x14, x16, x17
    cmp     x27, #3
    b.eq    op_result
    msub    x14, x14, x17, x16
    b       op_result
do_add:
    add     x14, x16, x17
    b       op_result
do_sub:
    sub     x14, x16, x17
    b       op_result
do_mul:
    mul     x14, x16, x17
op_result:
    str     x14, [x10]
    add     x24, x24, #1
    b       tok_loop
tok_end:
    cmp     x24, #1
    b.ne    eval_error
    ldr     x0, [x23]
    mov     x1, #0
    ret
eval_error:
    mov     x1, #1
    ret

// flush: write x22 bytes from x21 to stdout, reset x22
flush:
    mov     x10, #0
flush_loop:
    cmp     x10, x22
    b.hs    flush_done
    mov     x0, #1
    add     x1, x21, x10
    sub     x2, x22, x10
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    flush_done
    add     x10, x10, x0
    b       flush_loop
flush_done:
    mov     x22, #0
    ret

ensure_space:
    stp     x29, x30, [sp, #-16]!
    add     x10, x22, #32
    movz    x11, #1, lsl #16
    cmp     x10, x11
    b.ls    es_done
    bl      flush
es_done:
    ldp     x29, x30, [sp], #16
    ret

print_err:
    stp     x29, x30, [sp, #-16]!
    bl      ensure_space
    add     x10, x21, x22
    adrp    x11, errmsg
    add     x11, x11, :lo12:errmsg
    ldr     w12, [x11]
    str     w12, [x10]
    ldrh    w12, [x11, #4]
    strh    w12, [x10, #4]
    add     x22, x22, #6
    ldp     x29, x30, [sp], #16
    ret

// print_number: x0 = signed value
print_number:
    stp     x29, x30, [sp, #-16]!
    mov     x12, x0
    bl      ensure_space
    add     x10, x21, x22
    sub     sp, sp, #32
    add     x13, sp, #32
    mov     x14, x12
    tbz     x12, #63, pn_pos
    neg     x14, x12
    mov     w15, #'-'
    strb    w15, [x10], #1
pn_pos:
    mov     x15, #10
pn_loop:
    udiv    x16, x14, x15
    msub    x17, x16, x15, x14
    add     w17, w17, #'0'
    strb    w17, [x13, #-1]!
    mov     x14, x16
    cbnz    x14, pn_loop
    add     x16, sp, #32
pn_copy:
    ldrb    w17, [x13], #1
    strb    w17, [x10], #1
    cmp     x13, x16
    b.lo    pn_copy
    mov     w17, #10
    strb    w17, [x10], #1
    sub     x22, x10, x21
    add     sp, sp, #32
    ldp     x29, x30, [sp], #16
    ret
