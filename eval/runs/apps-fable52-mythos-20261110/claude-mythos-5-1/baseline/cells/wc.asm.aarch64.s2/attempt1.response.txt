.text
.globl _start
_start:
	mov	x19, #0			// lines
	mov	x20, #0			// words
	mov	x21, #0			// bytes
	mov	x22, #0			// in-word flag

read_loop:
	mov	x0, #0
	adrp	x1, buf
	add	x1, x1, :lo12:buf
	mov	x2, #4096
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	ble	done
	add	x21, x21, x0
	adrp	x1, buf
	add	x1, x1, :lo12:buf
	add	x4, x1, x0		// end pointer
scan:
	ldrb	w3, [x1], #1
	cmp	w3, #0x0a
	cinc	x19, x19, eq
	cmp	w3, #0x20
	beq	ws
	sub	w5, w3, #9
	cmp	w5, #4
	bls	ws
	cbnz	x22, next
	add	x20, x20, #1
	mov	x22, #1
	b	next
ws:
	mov	x22, #0
next:
	cmp	x1, x4
	blo	scan
	b	read_loop

done:
	sub	sp, sp, #64
	add	x2, sp, #64
	mov	w6, #10
	strb	w6, [x2, #-1]!
	mov	x0, x21
	bl	putnum
	mov	w6, #0x20
	strb	w6, [x2, #-1]!
	mov	x0, x20
	bl	putnum
	mov	w6, #0x20
	strb	w6, [x2, #-1]!
	mov	x0, x19
	bl	putnum
	mov	x1, x2
	add	x3, sp, #64
	sub	x2, x3, x1
	mov	x0, #1
	mov	x8, #64
	svc	#0
	mov	x0, #0
	mov	x8, #93
	svc	#0
hang:
	b	hang

// putnum: x0 = value, x2 = buffer pointer (prepend digits, x2 updated)
putnum:
	mov	x5, #10
1:
	udiv	x4, x0, x5
	msub	x6, x4, x5, x0
	add	w6, w6, #48
	strb	w6, [x2, #-1]!
	mov	x0, x4
	cbnz	x0, 1b
	ret

.bss
.balign 16
buf:
	.space	4096
