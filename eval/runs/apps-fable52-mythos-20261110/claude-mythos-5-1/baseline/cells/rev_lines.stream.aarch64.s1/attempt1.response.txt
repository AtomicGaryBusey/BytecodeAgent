.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
    mov     x21, #65536
read_loop:
    sub     x2, x21, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x22, outbuf
    add     x22, x22, :lo12:outbuf
    mov     x23, #0                 // output length
    mov     x24, #0                 // current line start
line_loop:
    cmp     x24, x20
    bge     write_out
    mov     x25, x24                // scan index
find_nl:
    cmp     x25, x20
    beq     found
    ldrb    w0, [x19, x25]
    cmp     w0, #10
    beq     found
    add     x25, x25, #1
    b       find_nl
found:
    mov     x26, x25                // end of line (exclusive)
rev_loop:
    cmp     x26, x24
    beq     rev_done
    sub     x26, x26, #1
    ldrb    w0, [x19, x26]
    strb    w0, [x22, x23]
    add     x23, x23, #1
    b       rev_loop
rev_done:
    mov     w0, #10
    strb    w0, [x22, x23]
    add     x23, x23, #1
    add     x24, x25, #1
    b       line_loop
write_out:
    mov     x24, #0
write_loop:
    cmp     x24, x23
    bge     do_exit
    mov     x0, #1
    add     x1, x22, x24
    sub     x2, x23, x24
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    ble     do_exit
    add     x24, x24, x0
    b       write_loop
do_exit:
    mov     x0, #0
    mov     x8, #94
    svc     #0
