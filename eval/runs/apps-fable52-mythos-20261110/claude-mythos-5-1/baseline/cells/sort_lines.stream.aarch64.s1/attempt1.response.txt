	.text
	.globl _start
_start:
	// ---- read all of stdin into inbuf ----
	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	mov x20, #0                  // total bytes read
read_loop:
	mov x2, #65536
	subs x2, x2, x20
	b.le read_done
	mov x0, #0
	add x1, x19, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	b.le read_done
	add x20, x20, x0
	b read_loop
read_done:
	// ---- parse lines into table (ptr, len) ----
	adrp x22, lines
	add x22, x22, :lo12:lines
	mov x21, #0                  // line count
	mov x0, x19                  // p
	add x1, x19, x20             // end
	mov x2, x19                  // start of current line
	mov x5, #4096                // cap
parse_loop:
	cmp x0, x1
	b.hs parse_end
	ldrb w3, [x0], #1
	cmp w3, #10
	b.ne parse_loop
	cmp x21, x5
	b.hs parse_end
	sub x4, x0, #1
	sub x4, x4, x2
	add x6, x22, x21, lsl #4
	stp x2, x4, [x6]
	add x21, x21, #1
	mov x2, x0
	b parse_loop
parse_end:
	cmp x2, x1
	b.hs sort
	cmp x21, x5
	b.hs sort
	sub x4, x1, x2
	add x6, x22, x21, lsl #4
	stp x2, x4, [x6]
	add x21, x21, #1

	// ---- bottom-up merge sort: x22 = src, x23 = dst ----
sort:
	adrp x23, temp
	add x23, x23, :lo12:temp
	mov x24, #1                  // width
sort_outer:
	cmp x24, x21
	b.hs sort_done
	mov x25, #0                  // i
sort_inner:
	cmp x25, x21
	b.hs sort_pass_done
	add x26, x25, x24            // mid
	cmp x26, x21
	csel x26, x26, x21, lo
	add x27, x26, x24            // right
	cmp x27, x21
	csel x27, x27, x21, lo
	mov x9, x25                  // a
	mov x10, x26                 // b
	mov x11, x25                 // out
merge_loop:
	cmp x9, x26
	b.lo 1f
	cmp x10, x27
	b.hs merge_done
	b merge_take_b
1:
	cmp x10, x27
	b.hs merge_take_a
	add x12, x22, x9, lsl #4
	ldp x0, x1, [x12]
	add x13, x22, x10, lsl #4
	ldp x2, x3, [x13]
	bl cmp_lines
	cmp x0, #0
	b.gt merge_take_b
merge_take_a:
	add x12, x22, x9, lsl #4
	ldp x0, x1, [x12]
	add x9, x9, #1
	b merge_store
merge_take_b:
	add x13, x22, x10, lsl #4
	ldp x0, x1, [x13]
	add x10, x10, #1
merge_store:
	add x14, x23, x11, lsl #4
	stp x0, x1, [x14]
	add x11, x11, #1
	b merge_loop
merge_done:
	add x25, x25, x24, lsl #1
	b sort_inner
sort_pass_done:
	mov x0, x22
	mov x22, x23
	mov x23, x0
	lsl x24, x24, #1
	b sort_outer
sort_done:

	// ---- output ----
	adrp x23, outbuf
	add x23, x23, :lo12:outbuf
	mov x24, #0                  // out position
	mov x25, #0                  // line index
out_loop:
	cmp x25, x21
	b.hs out_exit
	add x12, x22, x25, lsl #4
	ldr x1, [x12, #8]
	add x2, x24, x1
	add x2, x2, #1
	movz x3, #2, lsl #16
	cmp x2, x3
	b.ls 1f
	bl flush
1:
	ldp x0, x1, [x12]
	add x4, x23, x24
	cbz x1, 3f
2:
	ldrb w5, [x0], #1
	strb w5, [x4], #1
	subs x1, x1, #1
	b.ne 2b
3:
	mov w5, #10
	strb w5, [x4], #1
	sub x24, x4, x23
	add x25, x25, #1
	b out_loop
out_exit:
	bl flush
	mov x0, #0
	mov x8, #94
	svc #0
	b out_exit

	// flush: write outbuf[0..x24) to stdout, reset x24. Clobbers x0,x1,x2,x8,x9.
flush:
	mov x9, #0
1:
	cmp x9, x24
	b.hs 2f
	mov x0, #1
	add x1, x23, x9
	sub x2, x24, x9
	mov x8, #64
	svc #0
	cmp x0, #0
	b.le 2f
	add x9, x9, x0
	b 1b
2:
	mov x24, #0
	ret

	// cmp_lines(x0=p1, x1=len1, x2=p2, x3=len2) -> x0 signed (<0, 0, >0). Clobbers x0-x7.
cmp_lines:
	cmp x1, x3
	csel x4, x1, x3, lo
	add x5, x0, x4
1:
	cmp x0, x5
	b.hs 2f
	ldrb w6, [x0], #1
	ldrb w7, [x2], #1
	subs w6, w6, w7
	b.eq 1b
	sxtw x0, w6
	ret
2:
	sub x0, x1, x3
	ret

	.bss
	.balign 16
inbuf:
	.space 65536
	.balign 16
outbuf:
	.space 131072
	.balign 16
lines:
	.space 65536
	.balign 16
temp:
	.space 65536
