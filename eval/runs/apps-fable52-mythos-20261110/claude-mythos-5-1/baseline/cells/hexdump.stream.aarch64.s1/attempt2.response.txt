.text
.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total input length
read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #4096
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    done_read
    add     x20, x20, x0
    b       read_loop
done_read:
    adrp    x22, outbuf
    add     x22, x22, :lo12:outbuf
    mov     x21, x22                // output cursor
    adrp    x23, hexdigits
    add     x23, x23, :lo12:hexdigits
    mov     x24, #0                 // current offset
row_loop:
    cmp     x24, x20
    b.hs    finish
    sub     x25, x20, x24
    cmp     x25, #16
    mov     x1, #16
    csel    x25, x25, x1, lo        // n = min(16, remaining)
    mov     x0, x24
    bl      put_hex8
    mov     w1, #32
    strb    w1, [x21], #1
    strb    w1, [x21], #1
    mov     x26, #0
hex_loop:
    cmp     x26, x25
    b.hs    hex_pad
    add     x1, x19, x24
    ldrb    w0, [x1, x26]
    lsr     x1, x0, #4
    ldrb    w1, [x23, x1]
    strb    w1, [x21], #1
    and     x1, x0, #15
    ldrb    w1, [x23, x1]
    strb    w1, [x21], #1
    mov     w1, #32
    strb    w1, [x21], #1
    b       hex_next
hex_pad:
    mov     w1, #32
    strb    w1, [x21], #1
    strb    w1, [x21], #1
    strb    w1, [x21], #1
hex_next:
    cmp     x26, #7
    b.ne    1f
    mov     w1, #32
    strb    w1, [x21], #1
1:
    add     x26, x26, #1
    cmp     x26, #16
    b.lo    hex_loop
    mov     w1, #32
    strb    w1, [x21], #1
    mov     w1, #124
    strb    w1, [x21], #1
    mov     x26, #0
asc_loop:
    cmp     x26, x25
    b.hs    asc_done
    add     x1, x19, x24
    ldrb    w0, [x1, x26]
    sub     w1, w0, #32
    cmp     w1, #95
    b.lo    2f
    mov     w0, #46
2:
    strb    w0, [x21], #1
    add     x26, x26, #1
    b       asc_loop
asc_done:
    mov     w1, #124
    strb    w1, [x21], #1
    mov     w1, #10
    strb    w1, [x21], #1
    add     x24, x24, #16
    b       row_loop
finish:
    mov     x0, x20
    bl      put_hex8
    mov     w1, #10
    strb    w1, [x21], #1
    mov     x0, #1
    mov     x1, x22
    sub     x2, x21, x22
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0

// put_hex8: write x0 (low 32 bits) as 8 hex digits at x21, advance x21
put_hex8:
    mov     x2, #8
3:
    lsr     w1, w0, #28
    ldrb    w1, [x23, x1]
    strb    w1, [x21], #1
    lsl     w0, w0, #4
    subs    x2, x2, #1
    b.ne    3b
    ret

.section .rodata
hexdigits:
    .ascii  "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space  69632
.balign 16
outbuf:
    .space  331776
