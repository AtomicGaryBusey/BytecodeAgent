.bss
.balign 16
buf:    .space 65536
outbuf: .space 80

.text
.globl _start
_start:
    mov     x19, #0                 // lines
    mov     x20, #0                 // words
    mov     x21, #0                 // bytes
    mov     x22, #0                 // in_word flag
    adrp    x23, buf
    add     x23, x23, :lo12:buf

read_loop:
    mov     x0, #0
    mov     x1, x23
    mov     x2, #65536
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    done_read
    add     x21, x21, x0
    mov     x1, x23
    add     x2, x23, x0

proc:
    ldrb    w3, [x1], #1
    cmp     w3, #0x0a
    cinc    x19, x19, eq
    cmp     w3, #0x20
    b.eq    ws
    sub     w4, w3, #0x09
    cmp     w4, #4
    b.ls    ws
    cbnz    x22, next
    mov     x22, #1
    add     x20, x20, #1
    b       next
ws:
    mov     x22, #0
next:
    cmp     x1, x2
    b.lo    proc
    b       read_loop

done_read:
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    add     x24, x1, #80
    mov     x1, x24
    mov     w3, #10
    strb    w3, [x1, #-1]!
    mov     x0, x21
    bl      putnum
    mov     w3, #0x20
    strb    w3, [x1, #-1]!
    mov     x0, x20
    bl      putnum
    mov     w3, #0x20
    strb    w3, [x1, #-1]!
    mov     x0, x19
    bl      putnum
    sub     x2, x24, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #94
    svc     #0
    b       .

// putnum: x0 = value, x1 = write pointer (pre-decremented); clobbers x0,x4,x5,x6
putnum:
    mov     x5, #10
1:
    udiv    x4, x0, x5
    msub    x6, x4, x5, x0
    add     w6, w6, #48
    strb    w6, [x1, #-1]!
    mov     x0, x4
    cbnz    x0, 1b
    ret
