.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
read_loop:
    movz    x2, #1, lsl #16
    add     x2, x2, #64
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x23, x21
    mov     x22, #0
row_loop:
    cmp     x22, x20
    bge     rows_done
    mov     x0, x22
    bl      emit_hex8
    mov     w0, #0x20
    strb    w0, [x21], #1
    strb    w0, [x21], #1
    sub     x24, x20, x22
    mov     x25, #16
    cmp     x24, #16
    csel    x24, x24, x25, lo
    mov     x25, #0
hex_loop:
    cmp     x25, x24
    bge     hex_pad
    add     x1, x22, x25
    ldrb    w0, [x19, x1]
    bl      emit_hex2
    mov     w0, #0x20
    strb    w0, [x21], #1
    b       hex_next
hex_pad:
    mov     w0, #0x20
    strb    w0, [x21], #1
    strb    w0, [x21], #1
    strb    w0, [x21], #1
hex_next:
    cmp     x25, #7
    bne     hex_skip
    mov     w0, #0x20
    strb    w0, [x21], #1
hex_skip:
    add     x25, x25, #1
    cmp     x25, #16
    blt     hex_loop
    mov     w0, #0x20
    strb    w0, [x21], #1
    mov     w0, #0x7c
    strb    w0, [x21], #1
    mov     x25, #0
chr_loop:
    cmp     x25, x24
    bge     chr_done
    add     x1, x22, x25
    ldrb    w0, [x19, x1]
    sub     w1, w0, #0x20
    cmp     w1, #0x5e
    mov     w2, #0x2e
    csel    w0, w0, w2, ls
    strb    w0, [x21], #1
    add     x25, x25, #1
    b       chr_loop
chr_done:
    mov     w0, #0x7c
    strb    w0, [x21], #1
    mov     w0, #0x0a
    strb    w0, [x21], #1
    add     x22, x22, #16
    b       row_loop
rows_done:
    mov     x0, x20
    bl      emit_hex8
    mov     w0, #0x0a
    strb    w0, [x21], #1
    sub     x24, x21, x23
    mov     x25, x23
wr_loop:
    cbz     x24, done
    mov     x0, #1
    mov     x1, x25
    mov     x2, x24
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    ble     done
    add     x25, x25, x0
    sub     x24, x24, x0
    b       wr_loop
done:
    mov     x0, #0
    mov     x8, #94
    svc     #0
    b       done

// emit hex digits of w0 at x21: emit_hex8 -> 8 digits, emit_hex2 -> 2 digits
emit_hex2:
    mov     w4, #4
    b       emit_hex_body
emit_hex8:
    mov     w4, #28
emit_hex_body:
    adrp    x3, hexdig
    add     x3, x3, :lo12:hexdig
1:
    lsr     w1, w0, w4
    and     w1, w1, #15
    ldrb    w1, [x3, w1, uxtw]
    strb    w1, [x21], #1
    subs    w4, w4, #4
    bge     1b
    ret

.section .rodata
hexdig:
    .ascii  "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space  65600
.balign 16
outbuf:
    .space  331776
