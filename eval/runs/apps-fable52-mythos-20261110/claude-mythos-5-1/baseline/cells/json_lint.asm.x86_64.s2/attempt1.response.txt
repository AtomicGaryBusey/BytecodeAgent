.equ BUFCAP, 65552

.text
.globl _start
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
read_loop:
    mov edx, BUFCAP
    sub rdx, r13
    jbe read_done
    lea rsi, [r12 + r13]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor esi, esi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp read_done

error:
    mov rax, rsi
    lea rdi, [rip + errbuf + 63]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    mov rax, qword ptr [rip + err_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_prefix + 8]
    mov byte ptr [rdi + 8], al
    lea rdx, [rip + errbuf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
    jmp error

skip_ws:
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    jne 2f
1:
    inc rsi
    jmp skip_ws
2:
    ret

parse_value:
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6E
    je pv_null
    cmp al, 0x2D
    je parse_number
    sub al, 0x30
    cmp al, 9
    jbe parse_number
    jmp error
pv_true:
    lea rdi, [rip + lit_true]
    jmp parse_literal
pv_false:
    lea rdi, [rip + lit_false]
    jmp parse_literal
pv_null:
    lea rdi, [rip + lit_null]
parse_literal:
    movzx eax, byte ptr [rdi]
    test al, al
    jz 1f
    cmp al, byte ptr [r12 + rsi]
    jne error
    inc rsi
    inc rdi
    jmp parse_literal
1:
    ret

parse_number:
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x2D
    jne 1f
    inc rsi
    movzx eax, byte ptr [r12 + rsi]
1:
    cmp al, 0x30
    jne 2f
    inc rsi
    jmp num_frac
2:
    sub al, 0x31
    cmp al, 8
    ja error
    inc rsi
    call skip_digits
num_frac:
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x2E
    jne num_exp
    inc rsi
    call need_digit
    call skip_digits
num_exp:
    movzx eax, byte ptr [r12 + rsi]
    or al, 0x20
    cmp al, 0x65
    jne num_done
    inc rsi
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x2B
    je 3f
    cmp al, 0x2D
    jne 4f
3:
    inc rsi
4:
    call need_digit
    call skip_digits
num_done:
    ret

need_digit:
    movzx eax, byte ptr [r12 + rsi]
    sub al, 0x30
    cmp al, 9
    ja error
    inc rsi
    ret

skip_digits:
    movzx eax, byte ptr [r12 + rsi]
    sub al, 0x30
    cmp al, 9
    ja 1f
    inc rsi
    jmp skip_digits
1:
    ret

parse_string:
    inc rsi
ps_loop:
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x22
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb error
    inc rsi
    jmp ps_loop
ps_end:
    inc rsi
    ret
ps_esc:
    inc rsi
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x75
    je ps_u
    cmp al, 0x22
    je ps_ok1
    cmp al, 0x5C
    je ps_ok1
    cmp al, 0x2F
    je ps_ok1
    cmp al, 0x62
    je ps_ok1
    cmp al, 0x66
    je ps_ok1
    cmp al, 0x6E
    je ps_ok1
    cmp al, 0x72
    je ps_ok1
    cmp al, 0x74
    jne error
ps_ok1:
    inc rsi
    jmp ps_loop
ps_u:
    inc rsi
    mov ecx, 4
ps_hex:
    movzx eax, byte ptr [r12 + rsi]
    mov edx, eax
    sub dl, 0x30
    cmp dl, 9
    jbe ps_hexok
    or al, 0x20
    sub al, 0x61
    cmp al, 5
    ja error
ps_hexok:
    inc rsi
    dec ecx
    jnz ps_hex
    jmp ps_loop

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [r12 + rsi], 0x5D
    jne pa_loop
    inc rsi
    ret
pa_loop:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    jne error
    inc rsi
    ret
pa_comma:
    inc rsi
    call skip_ws
    jmp pa_loop

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [r12 + rsi], 0x7D
    jne po_loop
    inc rsi
    ret
po_loop:
    cmp byte ptr [r12 + rsi], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [r12 + rsi], 0x3A
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    jne error
    inc rsi
    ret
po_comma:
    inc rsi
    call skip_ws
    jmp po_loop

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
.balign 16
buf:
    .space BUFCAP
.balign 16
errbuf:
    .space 64
