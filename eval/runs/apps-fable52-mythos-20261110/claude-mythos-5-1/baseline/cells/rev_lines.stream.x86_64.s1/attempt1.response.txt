.text
.globl _start
_start:
        xor     ebx, ebx                # total bytes read
read_loop:
        mov     edx, 65536
        sub     edx, ebx
        jle     read_done
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        xor     eax, eax                # read
        xor     edi, edi                # fd 0
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     r8, [rip + inbuf]       # current line start
        lea     r9, [r8 + rbx]          # end of input
        lea     rdi, [rip + outbuf]     # output pointer
line_loop:
        cmp     r8, r9
        jae     output
        mov     r10, r8
find_nl:
        cmp     r10, r9
        jae     found
        cmp     byte ptr [r10], 10
        je      found
        inc     r10
        jmp     find_nl
found:
        mov     r11, r10
rev_loop:
        cmp     r11, r8
        jbe     rev_done
        dec     r11
        mov     al, byte ptr [r11]
        mov     byte ptr [rdi], al
        inc     rdi
        jmp     rev_loop
rev_done:
        mov     byte ptr [rdi], 10
        inc     rdi
        lea     r8, [r10 + 1]
        jmp     line_loop
output:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi
write_loop:
        test    rdx, rdx
        jle     done
        mov     eax, 1                  # write
        mov     edi, 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
done:
        mov     eax, 231                # exit_group
        xor     edi, edi
        syscall
        jmp     done

.bss
        .balign 16
inbuf:
        .space  65536 + 64
        .balign 16
outbuf:
        .space  65536 + 64
