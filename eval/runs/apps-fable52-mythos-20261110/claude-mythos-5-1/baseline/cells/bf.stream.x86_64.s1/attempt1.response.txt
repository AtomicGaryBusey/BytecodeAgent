.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + prog]
    lea r10, [rip + iscmd]
    xor ecx, ecx
    xor edx, edx
scan:
    cmp rcx, rbx
    jae scan_noinput
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x21
    je scan_input
    cmp byte ptr [r10 + rax], 0
    je scan
    mov byte ptr [rdi + rdx], al
    inc rdx
    jmp scan
scan_input:
    lea r13, [rsi + rcx]
    lea r14, [rsi + rbx]
    jmp build
scan_noinput:
    lea r13, [rsi + rbx]
    mov r14, r13
build:
    mov byte ptr [rdi + rdx], 0
    lea r9, [rip + jmptab]
    lea r10, [rip + stk]
    xor ecx, ecx
    xor r8d, r8d
build_loop:
    cmp rcx, rdx
    jae build_done
    movzx eax, byte ptr [rdi + rcx]
    cmp al, 0x5B
    je build_push
    cmp al, 0x5D
    jne build_next
    test r8, r8
    jz error
    dec r8
    mov eax, dword ptr [r10 + r8*4]
    mov dword ptr [r9 + rax*4], ecx
    mov dword ptr [r9 + rcx*4], eax
    jmp build_next
build_push:
    mov dword ptr [r10 + r8*4], ecx
    inc r8
build_next:
    inc rcx
    jmp build_loop
build_done:
    test r8, r8
    jnz error
    mov r8, rdi
    lea r12, [rip + tape]
    lea r10, [rip + outbuf]
    xor r15d, r15d
    xor ebx, ebx
run:
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    cmp al, 0x2B
    je do_plus
    cmp al, 0x2D
    je do_minus
    cmp al, 0x3E
    je do_right
    cmp al, 0x3C
    je do_left
    cmp al, 0x5B
    je do_open
    cmp al, 0x5D
    je do_close
    cmp al, 0x2E
    je do_out
    cmp al, 0x2C
    je do_in
    jmp finish
do_plus:
    inc byte ptr [r12]
    jmp run
do_minus:
    dec byte ptr [r12]
    jmp run
do_right:
    inc r12
    jmp run
do_left:
    dec r12
    jmp run
do_open:
    cmp byte ptr [r12], 0
    jne run
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    inc rbx
    jmp run
do_close:
    cmp byte ptr [r12], 0
    je run
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    inc rbx
    jmp run
do_out:
    mov al, byte ptr [r12]
    mov byte ptr [r10 + r15], al
    inc r15
    cmp r15, 65536
    jb run
    call flush
    jmp run
do_in:
    cmp r13, r14
    jae in_zero
    mov al, byte ptr [r13]
    inc r13
    mov byte ptr [r12], al
    jmp run
in_zero:
    mov byte ptr [r12], 0
    jmp run
finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
error:
    mov eax, 231
    mov edi, 1
    syscall
flush:
    test r15, r15
    jz flush_ret
    lea rsi, [rip + outbuf]
    mov rdx, r15
flush_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle flush_ret
    add rsi, rax
    sub rdx, rax
    jnz flush_loop
flush_ret:
    xor r15d, r15d
    ret

.section .rodata
iscmd:
    .zero 43
    .byte 1,1,1,1
    .zero 13
    .byte 1,0,1
    .zero 28
    .byte 1,0,1
    .zero 162

.bss
.balign 16
inbuf:  .space 65536
prog:   .space 65540
.balign 16
jmptab: .space 262144
stk:    .space 262144
tape:   .space 30016
outbuf: .space 65536
