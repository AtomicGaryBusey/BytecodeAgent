.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x22, alphabet
    add x22, x22, :lo12:alphabet
    mov x23, #0                 // output length
    mov x24, #0                 // chars on current line
    mov x25, x19                // input pointer
    add x26, x19, x20           // input end
encode_loop:
    sub x0, x26, x25
    cmp x0, #3
    blt tail
    ldrb w1, [x25], #1
    ldrb w2, [x25], #1
    ldrb w3, [x25], #1
    lsl w1, w1, #16
    orr w1, w1, w2, lsl #8
    orr w1, w1, w3
    lsr w2, w1, #18
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    lsr w2, w1, #12
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    lsr w2, w1, #6
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    and w2, w1, #63
    ldrb w2, [x22, x2]
    bl emit
    b encode_loop
tail:
    cmp x0, #1
    blt finish
    beq tail1
    // two bytes remaining
    ldrb w1, [x25], #1
    ldrb w2, [x25], #1
    lsl w1, w1, #16
    orr w1, w1, w2, lsl #8
    lsr w2, w1, #18
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    lsr w2, w1, #12
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    lsr w2, w1, #6
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    mov w2, #61
    bl emit
    b finish
tail1:
    ldrb w1, [x25], #1
    lsl w1, w1, #16
    lsr w2, w1, #18
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    lsr w2, w1, #12
    and w2, w2, #63
    ldrb w2, [x22, x2]
    bl emit
    mov w2, #61
    bl emit
    mov w2, #61
    bl emit
finish:
    cbz x24, write_loop
    mov w5, #10
    strb w5, [x21, x23]
    add x23, x23, #1
write_loop:
    cbz x23, done
    mov x8, #64
    mov x0, #1
    mov x1, x21
    mov x2, x23
    svc #0
    cmp x0, #0
    ble done
    add x21, x21, x0
    sub x23, x23, x0
    b write_loop
done:
    mov x8, #94
    mov x0, #0
    svc #0
    b done

// emit: store character in w2 to output, handle line wrapping
emit:
    strb w2, [x21, x23]
    add x23, x23, #1
    add x24, x24, #1
    cmp x24, #76
    bne 1f
    mov w5, #10
    strb w5, [x21, x23]
    add x23, x23, #1
    mov x24, #0
1:
    ret
