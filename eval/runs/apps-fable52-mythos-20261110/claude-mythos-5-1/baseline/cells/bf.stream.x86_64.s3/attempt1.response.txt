.section .rodata
optab:
    .zero 0x2B
    .byte 3, 6, 4, 5
    .zero 13
    .byte 2
    .zero 1
    .byte 1
    .zero 28
    .byte 7, 0, 8
    .zero 162
    .balign 8
handlers:
    .quad op_end, op_gt, op_lt, op_inc, op_dec, op_out, op_in, op_lb, op_rb

.bss
    .balign 16
inbuf:  .space 65537
prog:   .space 65537
    .balign 4
jmptab: .space 65537*4
    .balign 4
stack:  .space 65537*4
tape:   .space 30000
    .balign 16
outbuf: .space 65536

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov edx, 65537
    sub edx, r12d
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r15, [rsi + r12]
    mov r14, r15
    lea rdi, [rip + prog]
    lea r11, [rip + jmptab]
    lea r10, [rip + stack]
    lea rbx, [rip + optab]
    xor ecx, ecx
    xor r8d, r8d
parse_loop:
    cmp rsi, r15
    je parse_done
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x21
    je parse_bang
    movzx eax, byte ptr [rbx + rax]
    test al, al
    jz parse_loop
    mov byte ptr [rdi + r8], al
    cmp al, 7
    je parse_open
    cmp al, 8
    je parse_close
    inc r8
    jmp parse_loop
parse_open:
    mov dword ptr [r10 + rcx*4], r8d
    inc rcx
    inc r8
    jmp parse_loop
parse_close:
    test rcx, rcx
    jz parse_error
    dec rcx
    mov r9d, dword ptr [r10 + rcx*4]
    mov dword ptr [r11 + r9*4], r8d
    mov dword ptr [r11 + r8*4], r9d
    inc r8
    jmp parse_loop
parse_bang:
    mov r14, rsi
parse_done:
    mov byte ptr [rdi + r8], 0
    test rcx, rcx
    jnz parse_error

    lea r12, [rip + prog]
    mov rbx, r12
    lea r13, [rip + tape]
    lea rbp, [rip + outbuf]
    lea r10, [rip + handlers]
    lea r9, [rip + jmptab]
dispatch:
    movzx eax, byte ptr [rbx]
    inc rbx
    jmp qword ptr [r10 + rax*8]
op_gt:
    inc r13
    jmp dispatch
op_lt:
    dec r13
    jmp dispatch
op_inc:
    inc byte ptr [r13]
    jmp dispatch
op_dec:
    dec byte ptr [r13]
    jmp dispatch
op_out:
    mov al, byte ptr [r13]
    mov byte ptr [rbp], al
    inc rbp
    lea rax, [rip + outbuf + 65536]
    cmp rbp, rax
    jne dispatch
    call flush
    jmp dispatch
op_in:
    xor eax, eax
    cmp r14, r15
    jae 1f
    mov al, byte ptr [r14]
    inc r14
1:
    mov byte ptr [r13], al
    jmp dispatch
op_lb:
    cmp byte ptr [r13], 0
    jne dispatch
do_jump:
    mov rax, rbx
    sub rax, r12
    mov eax, dword ptr [r9 + rax*4 - 4]
    lea rbx, [r12 + rax + 1]
    jmp dispatch
op_rb:
    cmp byte ptr [r13], 0
    jne do_jump
    jmp dispatch
op_end:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp op_end

parse_error:
    mov eax, 231
    mov edi, 1
    syscall
    jmp parse_error

flush:
    lea rsi, [rip + outbuf]
    mov rdx, rbp
    sub rdx, rsi
flush_loop:
    test rdx, rdx
    jle flush_done
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle flush_done
    add rsi, rax
    sub rdx, rax
    jmp flush_loop
flush_done:
    lea rbp, [rip + outbuf]
    ret
