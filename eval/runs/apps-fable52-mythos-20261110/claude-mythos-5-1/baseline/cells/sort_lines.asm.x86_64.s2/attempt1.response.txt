.bss
.balign 16
inbuf:  .space 65536
outbuf: .space 131072
lines:  .space 8192*16
tmp:    .space 8192*16

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jle read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r8, [rsi + rbx]
    lea rdi, [rip + lines]
    xor r12d, r12d
parse_loop:
    cmp rsi, r8
    jae parse_done
    cmp r12, 8192
    jae parse_done
    mov r9, rsi
find_nl:
    cmp rsi, r8
    jae store_line
    cmp byte ptr [rsi], 10
    je store_line
    inc rsi
    jmp find_nl
store_line:
    mov qword ptr [rdi], r9
    mov rax, rsi
    sub rax, r9
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r12
    inc rsi
    jmp parse_loop
parse_done:
    lea r13, [rip + lines]
    lea r14, [rip + tmp]
    mov r15d, 1
sort_pass:
    cmp r15, r12
    jae sort_done
    xor ebx, ebx
merge_runs:
    cmp rbx, r12
    jae pass_done
    mov rax, rbx
    add rax, r15
    cmp rax, r12
    cmova rax, r12
    mov rdx, rax
    add rdx, r15
    cmp rdx, r12
    cmova rdx, r12
    mov rcx, rbx
    shl rcx, 4
    lea r10, [r13 + rcx]
    lea rdi, [r14 + rcx]
    mov rcx, rax
    shl rcx, 4
    lea rbp, [r13 + rcx]
    mov r11, rbp
    mov rcx, rdx
    shl rcx, 4
    lea rsi, [r13 + rcx]
    mov rbx, rdx
merge_loop:
    cmp r10, rbp
    jae rest_b
    cmp r11, rsi
    jae rest_a
    call cmp_lines
    test eax, eax
    jnz take_b
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [rdi], xmm0
    add r10, 16
    add rdi, 16
    jmp merge_loop
take_b:
    movups xmm0, xmmword ptr [r11]
    movups xmmword ptr [rdi], xmm0
    add r11, 16
    add rdi, 16
    jmp merge_loop
rest_a:
    cmp r10, rbp
    jae merge_runs
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [rdi], xmm0
    add r10, 16
    add rdi, 16
    jmp rest_a
rest_b:
    cmp r11, rsi
    jae merge_runs
    movups xmm0, xmmword ptr [r11]
    movups xmmword ptr [rdi], xmm0
    add r11, 16
    add rdi, 16
    jmp rest_b
pass_done:
    xchg r13, r14
    add r15, r15
    jmp sort_pass
sort_done:
    lea rdi, [rip + outbuf]
    mov rbx, r13
    mov rbp, r12
out_loop:
    test rbp, rbp
    jz out_done
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    dec rbp
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jz do_exit
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rbx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

# cmp_lines: r10 -> entry A, r11 -> entry B
# returns eax = 0 if A <= B, 1 if A > B; clobbers rax, rcx, rdx, r8, r9
cmp_lines:
    mov r8, qword ptr [r10]
    mov r9, qword ptr [r11]
    mov rcx, qword ptr [r10 + 8]
    cmp rcx, qword ptr [r11 + 8]
    cmova rcx, qword ptr [r11 + 8]
    test rcx, rcx
    jz 3f
1:
    movzx eax, byte ptr [r8]
    movzx edx, byte ptr [r9]
    cmp eax, edx
    jne 2f
    inc r8
    inc r9
    dec rcx
    jnz 1b
3:
    mov rax, qword ptr [r10 + 8]
    cmp rax, qword ptr [r11 + 8]
    ja 4f
    xor eax, eax
    ret
4:
    mov eax, 1
    ret
2:
    ja 4b
    xor eax, eax
    ret
