.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                     // input length
read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x28, outbuf
    add     x28, x28, :lo12:outbuf
    mov     x21, x28                    // output cursor
    adrp    x23, hexdig
    add     x23, x23, :lo12:hexdig
    mov     x22, #0                     // current offset
    mov     x25, #16
row_loop:
    sub     x24, x20, x22
    cbz     x24, finish
    cmp     x24, x25
    csel    x24, x24, x25, lo           // n = min(remaining, 16)
    mov     x0, x22
    bl      put_hex8
    mov     w0, #0x20
    strb    w0, [x21], #1
    strb    w0, [x21], #1
    mov     x26, #0
hex_loop:
    cmp     x26, x24
    b.hs    hex_pad
    add     x27, x22, x26
    ldrb    w0, [x19, x27]
    lsr     x10, x0, #4
    ldrb    w10, [x23, x10]
    strb    w10, [x21], #1
    and     x10, x0, #15
    ldrb    w10, [x23, x10]
    strb    w10, [x21], #1
    mov     w0, #0x20
    strb    w0, [x21], #1
    b       hex_next
hex_pad:
    mov     w0, #0x20
    strb    w0, [x21], #1
    strb    w0, [x21], #1
    strb    w0, [x21], #1
hex_next:
    cmp     x26, #7
    b.ne    1f
    mov     w0, #0x20
    strb    w0, [x21], #1
1:
    add     x26, x26, #1
    cmp     x26, #16
    b.lo    hex_loop
    mov     w0, #0x20
    strb    w0, [x21], #1
    mov     w0, #0x7c
    strb    w0, [x21], #1
    mov     x26, #0
chr_loop:
    add     x27, x22, x26
    ldrb    w0, [x19, x27]
    sub     w1, w0, #0x20
    cmp     w1, #0x5f
    b.lo    2f
    mov     w0, #0x2e
2:
    strb    w0, [x21], #1
    add     x26, x26, #1
    cmp     x26, x24
    b.lo    chr_loop
    mov     w0, #0x7c
    strb    w0, [x21], #1
    mov     w0, #0x0a
    strb    w0, [x21], #1
    add     x22, x22, x24
    b       row_loop
finish:
    mov     x0, x20
    bl      put_hex8
    mov     w0, #0x0a
    strb    w0, [x21], #1
    mov     x1, x28
    sub     x2, x21, x28
write_loop:
    cbz     x2, done
    mov     x8, #64
    mov     x0, #1
    svc     #0
    cmp     x0, #0
    b.le    done
    add     x1, x1, x0
    sub     x2, x2, x0
    b       write_loop
done:
    mov     x8, #93
    mov     x0, #0
    svc     #0
    b       done

// x0 = value, writes 8 hex digits at x21, advances x21
put_hex8:
    mov     x9, #28
1:
    lsr     x10, x0, x9
    and     x10, x10, #15
    ldrb    w10, [x23, x10]
    strb    w10, [x21], #1
    subs    x9, x9, #4
    b.pl    1b
    ret

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 327680
