.text
.globl _start
_start:
        # read all input
        lea     rbx, [rip + inbuf]      # rbx = buffer base
        xor     r12d, r12d              # r12 = bytes read so far
1:
        xor     eax, eax
        xor     edi, edi
        lea     rsi, [rbx + r12]
        mov     edx, 65536
        sub     rdx, r12
        jbe     2f
        syscall
        test    rax, rax
        jle     2f
        add     r12, rax
        jmp     1b
2:
        # process lines: rsi = in pointer, r12 = len, rdi = out pointer
        lea     rdi, [rip + outbuf]
        mov     r13, rdi                # r13 = out start
        xor     ecx, ecx                # rcx = i
3:
        cmp     rcx, r12
        jae     7f
        mov     rdx, rcx                # rdx = j
4:
        cmp     rdx, r12
        jae     5f
        cmp     byte ptr [rbx + rdx], 0x0A
        je      5f
        inc     rdx
        jmp     4b
5:
        # line is [rcx, rdx); copy reversed
        mov     r8, rdx
6:
        cmp     r8, rcx
        jbe     8f
        dec     r8
        mov     al, byte ptr [rbx + r8]
        mov     byte ptr [rdi], al
        inc     rdi
        jmp     6b
8:
        mov     byte ptr [rdi], 0x0A
        inc     rdi
        lea     rcx, [rdx + 1]
        jmp     3b
7:
        # write output: r13 = start, rdi = end
        mov     r14, rdi
        sub     r14, r13                # r14 = remaining
9:
        test    r14, r14
        jle     10f
        mov     eax, 1
        mov     edi, 1
        mov     rsi, r13
        mov     rdx, r14
        syscall
        test    rax, rax
        jle     10f
        add     r13, rax
        sub     r14, rax
        jmp     9b
10:
        mov     eax, 231
        xor     edi, edi
        syscall
        hlt

.bss
