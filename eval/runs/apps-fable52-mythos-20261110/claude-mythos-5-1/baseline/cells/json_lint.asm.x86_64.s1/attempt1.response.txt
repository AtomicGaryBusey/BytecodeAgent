.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r12, [rip + buf]
    lea r13, [r12 + rbx]
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp hang

error:
    lea rax, [rip + buf]
    sub r12, rax
    mov rax, r12
    lea rdi, [rip + numbuf + 47]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:
    dec rdi
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    mov rax, qword ptr [rip + err_msg]
    mov qword ptr [rdi - 9], rax
    mov al, byte ptr [rip + err_msg + 8]
    mov byte ptr [rdi - 1], al
    sub rdi, 9
    lea rdx, [rip + numbuf + 48]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall
hang:
    jmp hang

skip_ws:
1:
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    je 2f
    cmp al, 0x0A
    je 2f
    cmp al, 0x0D
    jne 3f
2:
    inc r12
    jmp 1b
3:
    ret

parse_value:
    movzx eax, byte ptr [r12]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb error
    cmp al, 0x39
    jbe parse_number
    cmp al, 0x74
    je lit_true
    cmp al, 0x66
    je lit_false
    cmp al, 0x6E
    je lit_null
    jmp error

lit_true:
    lea rsi, [rip + s_true]
    mov ecx, 4
    jmp parse_lit
lit_false:
    lea rsi, [rip + s_false]
    mov ecx, 5
    jmp parse_lit
lit_null:
    lea rsi, [rip + s_null]
    mov ecx, 4
parse_lit:
1:
    movzx eax, byte ptr [rsi]
    cmp al, byte ptr [r12]
    jne error
    inc rsi
    inc r12
    dec ecx
    jnz 1b
    ret

parse_string:
    inc r12
1:
    movzx eax, byte ptr [r12]
    cmp al, 0x22
    je 9f
    cmp al, 0x5C
    je 2f
    cmp al, 0x20
    jb error
    inc r12
    jmp 1b
2:
    inc r12
    movzx eax, byte ptr [r12]
    cmp al, 0x75
    je 3f
    cmp al, 0x22
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, 0x2F
    je 4f
    cmp al, 0x62
    je 4f
    cmp al, 0x66
    je 4f
    cmp al, 0x6E
    je 4f
    cmp al, 0x72
    je 4f
    cmp al, 0x74
    je 4f
    jmp error
4:
    inc r12
    jmp 1b
3:
    inc r12
    mov ecx, 4
5:
    movzx eax, byte ptr [r12]
    mov edx, eax
    sub dl, 0x30
    cmp dl, 9
    jbe 6f
    mov edx, eax
    or dl, 0x20
    sub dl, 0x61
    cmp dl, 5
    jbe 6f
    jmp error
6:
    inc r12
    dec ecx
    jnz 5b
    jmp 1b
9:
    inc r12
    ret

parse_number:
    movzx eax, byte ptr [r12]
    cmp al, 0x2D
    jne 1f
    inc r12
    movzx eax, byte ptr [r12]
1:
    cmp al, 0x30
    jne 2f
    inc r12
    jmp 4f
2:
    sub al, 0x31
    cmp al, 8
    ja error
    inc r12
3:
    movzx eax, byte ptr [r12]
    sub al, 0x30
    cmp al, 9
    ja 4f
    inc r12
    jmp 3b
4:
    movzx eax, byte ptr [r12]
    cmp al, 0x2E
    jne 6f
    inc r12
    movzx eax, byte ptr [r12]
    sub al, 0x30
    cmp al, 9
    ja error
    inc r12
5:
    movzx eax, byte ptr [r12]
    sub al, 0x30
    cmp al, 9
    ja 6f
    inc r12
    jmp 5b
6:
    movzx eax, byte ptr [r12]
    or al, 0x20
    cmp al, 0x65
    jne 8f
    inc r12
    movzx eax, byte ptr [r12]
    cmp al, 0x2B
    je 7f
    cmp al, 0x2D
    jne 71f
7:
    inc r12
71:
    movzx eax, byte ptr [r12]
    sub al, 0x30
    cmp al, 9
    ja error
    inc r12
72:
    movzx eax, byte ptr [r12]
    sub al, 0x30
    cmp al, 9
    ja 8f
    inc r12
    jmp 72b
8:
    ret

parse_object:
    inc r12
    call skip_ws
    cmp byte ptr [r12], 0x7D
    jne 1f
    inc r12
    ret
1:
    cmp byte ptr [r12], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [r12], 0x3A
    jne error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [r12]
    cmp al, 0x2C
    jne 2f
    inc r12
    call skip_ws
    jmp 1b
2:
    cmp al, 0x7D
    jne error
    inc r12
    ret

parse_array:
    inc r12
    call skip_ws
    cmp byte ptr [r12], 0x5D
    jne 1f
    inc r12
    ret
1:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [r12]
    cmp al, 0x2C
    jne 2f
    inc r12
    call skip_ws
    jmp 1b
2:
    cmp al, 0x5D
    jne error
    inc r12
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"

.bss
.balign 16
buf:
    .space 65536 + 64
numbuf:
    .space 48
