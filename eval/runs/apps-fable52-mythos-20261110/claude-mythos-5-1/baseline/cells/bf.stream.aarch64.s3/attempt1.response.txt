.bss
.balign 16
inbuf:   .space 65552
tape:    .space 30000
.balign 4
jumptab: .space 262144
bstack:  .space 262144
outbuf:  .space 65536

.text
.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     w1, #65536
    cmp     x20, x1
    b.hs    read_done
    sub     x2, x1, x20
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    // find first '!'
    mov     x1, #0
find_loop:
    cmp     x1, x20
    b.hs    find_none
    ldrb    w0, [x19, x1]
    cmp     w0, #0x21
    b.eq    find_found
    add     x1, x1, #1
    b       find_loop
find_found:
    add     x23, x19, x1
    add     x23, x23, #1            // input ptr
    add     x24, x19, x20           // input end
    mov     x20, x1                 // program length
    b       match
find_none:
    add     x23, x19, x20
    mov     x24, x23
match:
    adrp    x25, jumptab
    add     x25, x25, :lo12:jumptab
    adrp    x5, bstack
    add     x5, x5, :lo12:bstack
    mov     x6, #0                  // stack depth
    mov     x1, #0
match_loop:
    cmp     x1, x20
    b.hs    match_end
    ldrb    w0, [x19, x1]
    cmp     w0, #0x5b
    b.eq    match_open
    cmp     w0, #0x5d
    b.eq    match_close
match_next:
    add     x1, x1, #1
    b       match_loop
match_open:
    str     w1, [x5, x6, lsl #2]
    add     x6, x6, #1
    b       match_next
match_close:
    cbz     x6, fail
    sub     x6, x6, #1
    ldr     w2, [x5, x6, lsl #2]
    str     w1, [x25, x2, lsl #2]
    str     w2, [x25, x1, lsl #2]
    b       match_next
match_end:
    cbnz    x6, fail

    // execute
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    adrp    x26, outbuf
    add     x26, x26, :lo12:outbuf
    mov     x27, #0                 // output count
    mov     x21, #0                 // pc
run_loop:
    cmp     x21, x20
    b.hs    run_done
    ldrb    w0, [x19, x21]
    add     x21, x21, #1
    cmp     w0, #0x2b
    b.eq    op_inc
    cmp     w0, #0x3e
    b.eq    op_right
    cmp     w0, #0x2d
    b.eq    op_dec
    cmp     w0, #0x3c
    b.eq    op_left
    cmp     w0, #0x5b
    b.eq    op_open
    cmp     w0, #0x5d
    b.eq    op_close
    cmp     w0, #0x2e
    b.eq    op_out
    cmp     w0, #0x2c
    b.eq    op_in
    b       run_loop
op_right:
    add     x22, x22, #1
    b       run_loop
op_left:
    sub     x22, x22, #1
    b       run_loop
op_inc:
    ldrb    w0, [x22]
    add     w0, w0, #1
    strb    w0, [x22]
    b       run_loop
op_dec:
    ldrb    w0, [x22]
    sub     w0, w0, #1
    strb    w0, [x22]
    b       run_loop
op_out:
    ldrb    w0, [x22]
    strb    w0, [x26, x27]
    add     x27, x27, #1
    cmp     x27, #0x10000
    b.lo    run_loop
    bl      flush
    b       run_loop
op_in:
    cmp     x23, x24
    b.hs    in_eof
    ldrb    w0, [x23], #1
    strb    w0, [x22]
    b       run_loop
in_eof:
    strb    wzr, [x22]
    b       run_loop
op_open:
    ldrb    w0, [x22]
    cbnz    w0, run_loop
    sub     x1, x21, #1
    ldr     w1, [x25, x1, lsl #2]
    add     x21, x1, #1
    b       run_loop
op_close:
    ldrb    w0, [x22]
    cbz     w0, run_loop
    sub     x1, x21, #1
    ldr     w1, [x25, x1, lsl #2]
    add     x21, x1, #1
    b       run_loop
run_done:
    bl      flush
    mov     x0, #0
    mov     x8, #94
    svc     #0
fail:
    mov     x0, #1
    mov     x8, #94
    svc     #0

flush:
    mov     x1, x26
flush_loop:
    cbz     x27, flush_done
    mov     x0, #1
    mov     x2, x27
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    flush_done
    add     x1, x1, x0
    sub     x27, x27, x0
    b       flush_loop
flush_done:
    mov     x27, #0
    ret
