.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x20, #0                 // total input length
    movz x21, #1, lsl #16       // 65536
    add x21, x21, #16           // buffer capacity 65552
read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x25, x24                // output write pointer
    mov x22, #0                 // current offset
row_loop:
    sub x9, x20, x22            // remaining bytes
    cbz x9, finish
    mov x10, #16
    cmp x9, x10
    csel x9, x9, x10, lo        // n = min(remaining, 16)
    mov x0, x22
    bl print_hex8
    mov w11, #32
    strb w11, [x25], #1
    strb w11, [x25], #1
    add x13, x19, x22           // row pointer
    mov x14, #0
col_loop:
    cmp x14, x9
    b.hs col_pad
    ldrb w11, [x13, x14]
    lsr x12, x11, #4
    ldrb w12, [x23, x12]
    strb w12, [x25], #1
    and x12, x11, #15
    ldrb w12, [x23, x12]
    strb w12, [x25], #1
    mov w11, #32
    strb w11, [x25], #1
    b col_next
col_pad:
    mov w11, #32
    strb w11, [x25], #1
    strb w11, [x25], #1
    strb w11, [x25], #1
col_next:
    cmp x14, #7
    b.ne 1f
    mov w11, #32
    strb w11, [x25], #1
1:
    add x14, x14, #1
    cmp x14, #16
    b.lo col_loop
    mov w11, #32
    strb w11, [x25], #1
    mov w11, #124
    strb w11, [x25], #1
    mov x14, #0
asc_loop:
    cmp x14, x9
    b.hs asc_done
    ldrb w11, [x13, x14]
    sub w12, w11, #32
    cmp w12, #95
    b.lo 2f
    mov w11, #46
2:
    strb w11, [x25], #1
    add x14, x14, #1
    b asc_loop
asc_done:
    mov w11, #124
    strb w11, [x25], #1
    mov w11, #10
    strb w11, [x25], #1
    add x22, x22, x9
    b row_loop
finish:
    mov x0, x20
    bl print_hex8
    mov w11, #10
    strb w11, [x25], #1
    mov x26, x24                // write pointer
write_loop:
    sub x2, x25, x26
    cbz x2, done
    mov x0, #1
    mov x1, x26
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le done
    add x26, x26, x0
    b write_loop
done:
    mov x0, #0
    mov x8, #94
    svc #0
    b done

// x0 = value; writes 8 hex digits at x25; clobbers x11, x12
print_hex8:
    mov x12, #28
3:
    lsr x11, x0, x12
    and x11, x11, #15
    ldrb w11, [x23, x11]
    strb w11, [x25], #1
    subs x12, x12, #4
    b.ge 3b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 331776
