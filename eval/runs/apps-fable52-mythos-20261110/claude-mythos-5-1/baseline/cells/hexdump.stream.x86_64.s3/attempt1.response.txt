.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov edx, 65536
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    lea r8, [rip + hexdigits]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae rows_done
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    mov eax, r13d
    call put_hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [rbx + r13]
    xor ecx, ecx
hex_loop:
    cmp rcx, r14
    jae pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp next
pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
next:
    add rdi, 3
    cmp ecx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc ecx
    cmp ecx, 16
    jb hex_loop
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx
chr_loop:
    cmp rcx, r14
    jae chr_done
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 3f
    mov eax, 0x2e
3:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp chr_loop
chr_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, 16
    jmp row_loop
rows_done:
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 231
    xor edi, edi
    syscall
    jmp done

# eax = value, rdi = dest, r8 = hexdigits; writes 8 digits, advances rdi
put_hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 65552
    .balign 16
outbuf:
    .space 331776
