.text
.globl _start
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in-word flag
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea rdx, [rsi + rax]
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    lea ecx, [rax - 9]
    cmp ecx, 4
    jbe ws
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    cmp rsi, rdx
    jb byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call fmt
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call fmt
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call fmt
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall

# fmt: rax = value, rdi = destination pointer (advanced past digits)
fmt:
    sub rsp, 32
    lea rcx, [rsp + 32]
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, 0x30
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    lea rdx, [rsp + 32]
    cmp rcx, rdx
    jb 2b
    add rsp, 32
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
