.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x21, #0                 // bytes read so far
read_loop:
    mov     x2, #65536
    sub     x2, x2, x21
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x21
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x21, x21, x0
    b       read_loop
read_done:
    // find first '!'
    mov     x1, #0
find_loop:
    cmp     x1, x21
    b.hs    no_bang
    ldrb    w0, [x19, x1]
    cmp     w0, #0x21
    b.eq    found_bang
    add     x1, x1, #1
    b       find_loop
found_bang:
    add     x23, x19, x1
    add     x23, x23, #1            // input start
    add     x24, x19, x21           // input end
    mov     x21, x1                 // program length
    b       bracket_check
no_bang:
    add     x23, x19, x21
    mov     x24, x23

bracket_check:
    adrp    x27, jmptab
    add     x27, x27, :lo12:jmptab
    adrp    x9, bstack
    add     x9, x9, :lo12:bstack
    mov     x10, #0                 // stack depth
    mov     x1, #0
bc_loop:
    cmp     x1, x21
    b.hs    bc_end
    ldrb    w0, [x19, x1]
    cmp     w0, #0x5B
    b.eq    bc_open
    cmp     w0, #0x5D
    b.eq    bc_close
bc_next:
    add     x1, x1, #1
    b       bc_loop
bc_open:
    str     w1, [x9, x10, lsl #2]
    add     x10, x10, #1
    b       bc_next
bc_close:
    cbz     x10, bad
    sub     x10, x10, #1
    ldr     w2, [x9, x10, lsl #2]   // index of matching '['
    add     w3, w1, #1
    str     w3, [x27, x2, lsl #2]   // jmp[open] = close+1
    add     w3, w2, #1
    str     w3, [x27, x1, lsl #2]   // jmp[close] = open+1
    b       bc_next
bc_end:
    cbnz    x10, bad

    // run
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    adrp    x25, outbuf
    add     x25, x25, :lo12:outbuf
    mov     x26, #0                 // output count
    mov     x20, #0                 // pc
run:
    cmp     x20, x21
    b.hs    finish
    ldrb    w0, [x19, x20]
    add     x20, x20, #1
    cmp     w0, #0x2B
    b.eq    op_plus
    cmp     w0, #0x2D
    b.eq    op_minus
    cmp     w0, #0x3E
    b.eq    op_right
    cmp     w0, #0x3C
    b.eq    op_left
    cmp     w0, #0x5B
    b.eq    op_open
    cmp     w0, #0x5D
    b.eq    op_close
    cmp     w0, #0x2E
    b.eq    op_dot
    cmp     w0, #0x2C
    b.eq    op_comma
    b       run
op_plus:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       run
op_minus:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       run
op_right:
    add     x22, x22, #1
    b       run
op_left:
    sub     x22, x22, #1
    b       run
op_open:
    ldrb    w1, [x22]
    cbnz    w1, run
    sub     x2, x20, #1
    ldr     w20, [x27, x2, lsl #2]
    b       run
op_close:
    ldrb    w1, [x22]
    cbz     w1, run
    sub     x2, x20, #1
    ldr     w20, [x27, x2, lsl #2]
    b       run
op_dot:
    ldrb    w1, [x22]
    strb    w1, [x25, x26]
    add     x26, x26, #1
    mov     x3, #65536
    cmp     x26, x3
    b.lo    run
    bl      flush
    b       run
op_comma:
    cmp     x23, x24
    b.hs    comma_zero
    ldrb    w1, [x23], #1
    strb    w1, [x22]
    b       run
comma_zero:
    strb    wzr, [x22]
    b       run

finish:
    bl      flush
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       finish

bad:
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       bad

flush:
    stp     x29, x30, [sp, #-16]!
    mov     x1, x25
    mov     x2, x26
flush_loop:
    cbz     x2, flush_done
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    flush_done
    add     x1, x1, x0
    sub     x2, x2, x0
    b       flush_loop
flush_done:
    mov     x26, #0
    ldp     x29, x30, [sp], #16
    ret

.bss
.balign 16
inbuf:
    .space 65536
.balign 16
jmptab:
    .space 262144
.balign 16
bstack:
    .space 262144
.balign 16
tape:
    .space 30016
.balign 16
outbuf:
    .space 65536
