.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // bytes read so far
read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    // find first '!'
    mov     x1, #0
find_bang:
    cmp     x1, x20
    beq     no_bang
    ldrb    w2, [x19, x1]
    cmp     w2, #0x21
    beq     found_bang
    add     x1, x1, #1
    b       find_bang
found_bang:
    add     x21, x19, x1
    add     x21, x21, #1            // input start
    add     x22, x19, x20           // input end
    mov     x20, x1                 // program length
    b       scan
no_bang:
    add     x21, x19, x20
    mov     x22, x21

scan:
    adrp    x24, jumps
    add     x24, x24, :lo12:jumps
    adrp    x9, bstack
    add     x9, x9, :lo12:bstack
    mov     x10, #0                 // stack depth
    mov     x1, #0                  // index
scan_loop:
    cmp     x1, x20
    beq     scan_done
    ldrb    w2, [x19, x1]
    cmp     w2, #0x5B
    beq     scan_open
    cmp     w2, #0x5D
    beq     scan_close
scan_next:
    add     x1, x1, #1
    b       scan_loop
scan_open:
    str     w1, [x9, x10, lsl #2]
    add     x10, x10, #1
    b       scan_next
scan_close:
    cbz     x10, bad
    sub     x10, x10, #1
    ldr     w3, [x9, x10, lsl #2]
    str     w1, [x24, x3, lsl #2]
    str     w3, [x24, x1, lsl #2]
    b       scan_next
scan_done:
    cbnz    x10, bad

    adrp    x23, tape
    add     x23, x23, :lo12:tape
    adrp    x25, outbuf
    add     x25, x25, :lo12:outbuf
    mov     x26, #0                 // output count
    mov     x27, #0                 // pc
run:
    cmp     x27, x20
    bhs     finish
    ldrb    w2, [x19, x27]
    add     x27, x27, #1
    cmp     w2, #0x3E
    beq     c_right
    cmp     w2, #0x3C
    beq     c_left
    cmp     w2, #0x2B
    beq     c_inc
    cmp     w2, #0x2D
    beq     c_dec
    cmp     w2, #0x2E
    beq     c_out
    cmp     w2, #0x2C
    beq     c_in
    cmp     w2, #0x5B
    beq     c_open
    cmp     w2, #0x5D
    beq     c_close
    b       run
c_right:
    add     x23, x23, #1
    b       run
c_left:
    sub     x23, x23, #1
    b       run
c_inc:
    ldrb    w3, [x23]
    add     w3, w3, #1
    strb    w3, [x23]
    b       run
c_dec:
    ldrb    w3, [x23]
    sub     w3, w3, #1
    strb    w3, [x23]
    b       run
c_out:
    ldrb    w3, [x23]
    strb    w3, [x25, x26]
    add     x26, x26, #1
    mov     x4, #65536
    cmp     x26, x4
    bne     run
    bl      flush
    b       run
c_in:
    cmp     x21, x22
    bhs     c_in_zero
    ldrb    w3, [x21], #1
    strb    w3, [x23]
    b       run
c_in_zero:
    strb    wzr, [x23]
    b       run
c_open:
    ldrb    w3, [x23]
    cbnz    w3, run
    b       do_jump
c_close:
    ldrb    w3, [x23]
    cbz     w3, run
do_jump:
    sub     x4, x27, #1
    ldr     w4, [x24, x4, lsl #2]
    add     x27, x4, #1
    b       run

finish:
    bl      flush
    mov     x0, #0
    mov     x8, #94
    svc     #0
    b       finish

bad:
    mov     x0, #1
    mov     x8, #94
    svc     #0
    b       bad

flush:
    cbz     x26, flush_ret
    mov     x9, #0
flush_loop:
    mov     x0, #1
    add     x1, x25, x9
    sub     x2, x26, x9
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    ble     flush_done
    add     x9, x9, x0
    cmp     x9, x26
    blo     flush_loop
flush_done:
    mov     x26, #0
flush_ret:
    ret
