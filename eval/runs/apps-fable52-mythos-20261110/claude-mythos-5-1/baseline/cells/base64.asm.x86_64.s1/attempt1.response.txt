.text
.globl _start
_start:
    xor ebx, ebx                      # total bytes read
read_loop:
    mov edx, 65536
    sub edx, ebx
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax                      # read
    xor edi, edi                      # fd 0
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + table]
    xor ecx, ecx                      # chars on current line
    mov rdx, rbx                      # remaining input bytes

enc_loop:
    cmp rdx, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx r9d, byte ptr [rsi + 1]
    shl r9d, 8
    or eax, r9d
    movzx r9d, byte ptr [rsi + 2]
    or eax, r9d

    mov r9d, eax
    shr r9d, 18
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi], r10b
    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi + 1], r10b
    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi + 2], r10b
    mov r9d, eax
    and r9d, 63
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi + 3], r10b

    add rsi, 3
    add rdi, 4
    sub rdx, 3
    add ecx, 4
    cmp ecx, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor ecx, ecx
    jmp enc_loop

enc_tail:
    test rdx, rdx
    jz enc_end
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rdx, 2
    jb tail_have
    movzx r9d, byte ptr [rsi + 1]
    shl r9d, 8
    or eax, r9d
tail_have:
    mov r9d, eax
    shr r9d, 18
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi], r10b
    mov r9d, eax
    shr r9d, 12
    and r9d, 63
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi + 1], r10b
    cmp rdx, 2
    jb pad_two
    mov r9d, eax
    shr r9d, 6
    and r9d, 63
    mov r10b, byte ptr [r8 + r9]
    mov byte ptr [rdi + 2], r10b
    mov byte ptr [rdi + 3], 61
    jmp tail_done
pad_two:
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
tail_done:
    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi
    jmp write_out

enc_end:
    test ecx, ecx
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1                        # write
    mov edi, 1                        # fd 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop

do_exit:
    mov eax, 231                      # exit_group
    xor edi, edi
    syscall

.section .rodata
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space 65536
.balign 16
outbuf:
    .space 90112
