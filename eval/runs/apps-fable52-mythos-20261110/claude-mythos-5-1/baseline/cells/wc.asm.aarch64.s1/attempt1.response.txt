.text
.globl _start

_start:
        mov     x19, #0                 // lines
        mov     x20, #0                 // words
        mov     x21, #0                 // bytes
        mov     x22, #0                 // in_word flag
        adrp    x23, buf
        add     x23, x23, :lo12:buf

readloop:
        mov     x0, #0
        mov     x1, x23
        mov     x2, #0x10000
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    done
        add     x21, x21, x0
        mov     x1, x23
        add     x2, x23, x0
byteloop:
        ldrb    w3, [x1], #1
        cmp     w3, #10
        cinc    x19, x19, eq
        cmp     w3, #0x20
        b.eq    ws
        sub     w4, w3, #9
        cmp     w4, #4
        b.ls    ws
        cbnz    x22, next
        add     x20, x20, #1
        mov     x22, #1
        b       next
ws:
        mov     x22, #0
next:
        cmp     x1, x2
        b.lo    byteloop
        b       readloop

done:
        sub     sp, sp, #80
        add     x5, sp, #80
        mov     w3, #10
        strb    w3, [x5, #-1]!
        mov     x0, x21
        bl      fmt
        mov     w3, #0x20
        strb    w3, [x5, #-1]!
        mov     x0, x20
        bl      fmt
        mov     w3, #0x20
        strb    w3, [x5, #-1]!
        mov     x0, x19
        bl      fmt
        mov     x0, #1
        mov     x1, x5
        add     x2, sp, #80
        sub     x2, x2, x5
        mov     x8, #64
        svc     #0
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       .

// fmt: write decimal of x0 backwards at x5 (pre-decremented)
fmt:
        mov     x6, #10
1:
        udiv    x7, x0, x6
        msub    x4, x7, x6, x0
        add     w4, w4, #0x30
        strb    w4, [x5, #-1]!
        mov     x0, x7
        cbnz    x0, 1b
        ret

.bss
.balign 16
buf:
        .space  65536
