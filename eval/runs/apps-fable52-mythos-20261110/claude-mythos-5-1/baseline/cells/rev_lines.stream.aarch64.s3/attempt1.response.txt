.globl _start

_start:
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, #0                 // total bytes read
        mov     x21, #65536             // capacity
1:
        subs    x2, x21, x20
        b.le    2f
        mov     x0, #0
        add     x1, x19, x20
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    2f
        add     x20, x20, x0
        b       1b
2:
        // process lines
        adrp    x22, outbuf
        add     x22, x22, :lo12:outbuf
        mov     x23, x22                // out pointer
        add     x21, x19, x20           // input end
        mov     x24, x19                // current position (line start)
3:
        cmp     x24, x21
        b.hs    6f                      // no more lines
        mov     x25, x24                // scan pointer
4:
        cmp     x25, x21
        b.hs    5f
        ldrb    w0, [x25]
        cmp     w0, #10
        b.eq    5f
        add     x25, x25, #1
        b       4b
5:
        // line is [x24, x25); reverse copy
        mov     x26, x25
7:
        cmp     x26, x24
        b.ls    8f
        sub     x26, x26, #1
        ldrb    w0, [x26]
        strb    w0, [x23], #1
        b       7b
8:
        mov     w0, #10
        strb    w0, [x23], #1
        add     x24, x25, #1            // skip newline (or go past end)
        b       3b
6:
        // write output
        sub     x20, x23, x22           // output length
        mov     x24, #0                 // written so far
9:
        subs    x2, x20, x24
        b.le    10f
        mov     x0, #1
        add     x1, x22, x24
        mov     x8, #64
        svc     #0
        cmp     x0, #0
        b.le    10f
        add     x24, x24, x0
        b       9b
10:
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       10b
