_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x20, #0                 // input length

read_loop:

    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63

find_nl:

    b.ge    found
    ldrb    w11, [x19, x10]
    cmp     w11, #10
    b.eq    found
    add     x10, x10, #1

rev_loop:

    b.le    rev_done
    sub     x12, x12, #1
    ldrb    w11, [x19, x12]
    strb    w11, [x21, x22]

rev_done:

    strb    w11, [x21, x22]
    add     x22, x22, #1
    add     x9, x10, #1
    b       proc_loop
proc_done:
    mov     x9, #0                  // bytes written
write_loop:

    b.le    do_exit
    mov     x0, #1
    add     x1, x21, x9
    mov     x8, #64
    svc     #0
    cmp     x0, #0

    add     x9, x9, x0
