.text
.globl _start
_start:
    // ---- read all of standard input ----
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // bytes read so far
    mov x21, #65536             // buffer capacity
read_loop:
    cmp x20, x21
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    // ---- split into lines: table of (ptr, len) ----
    adrp x22, lines_a
    add x22, x22, :lo12:lines_a
    mov x23, #0                 // line count
    mov x24, x19                // current line start
    mov x25, x19                // cursor
    add x26, x19, x20           // end of input
parse_loop:
    cmp x25, x26
    b.hs parse_end
    ldrb w0, [x25], #1
    cmp w0, #10
    b.ne parse_loop
    sub x1, x25, #1
    sub x1, x1, x24             // length without newline
    add x2, x22, x23, lsl #4
    stp x24, x1, [x2]
    add x23, x23, #1
    mov x24, x25
    mov x3, #4095
    cmp x23, x3
    b.lo parse_loop
    mov x24, x26                // table full: drop remainder
parse_end:
    cmp x24, x26
    b.hs sort
    sub x1, x26, x24
    add x2, x22, x23, lsl #4
    stp x24, x1, [x2]
    add x23, x23, #1

    // ---- bottom-up merge sort, x22 = src, x27 = dst ----
sort:
    adrp x27, lines_b
    add x27, x27, :lo12:lines_b
    mov x28, #1                 // run width
sort_pass:
    cmp x28, x23
    b.hs sort_done
    mov x9, #0                  // i
pass_loop:
    cmp x9, x23
    b.hs pass_end
    add x10, x9, x28            // mid
    cmp x10, x23
    csel x10, x10, x23, lo
    add x11, x10, x28           // hi
    cmp x11, x23
    csel x11, x11, x23, lo
    add x12, x22, x9, lsl #4    // left ptr
    add x13, x22, x10, lsl #4   // left end
    mov x14, x13                // right ptr
    add x15, x22, x11, lsl #4   // right end
    add x16, x27, x9, lsl #4    // out ptr
merge_loop:
    cmp x12, x13
    b.hs merge_rest_right
    cmp x14, x15
    b.hs merge_rest_left
    ldp x0, x1, [x12]
    ldp x2, x3, [x14]
    bl cmp_lines
    cmp w0, #0
    b.gt take_right
    ldp x0, x1, [x12], #16
    stp x0, x1, [x16], #16
    b merge_loop
take_right:
    ldp x0, x1, [x14], #16
    stp x0, x1, [x16], #16
    b merge_loop
merge_rest_left:
    cmp x12, x13
    b.hs merge_done
    ldp x0, x1, [x12], #16
    stp x0, x1, [x16], #16
    b merge_rest_left
merge_rest_right:
    cmp x14, x15
    b.hs merge_done
    ldp x0, x1, [x14], #16
    stp x0, x1, [x16], #16
    b merge_rest_right
merge_done:
    add x9, x9, x28, lsl #1
    b pass_loop
pass_end:
    mov x0, x22
    mov x22, x27
    mov x27, x0
    lsl x28, x28, #1
    b sort_pass
sort_done:

    // ---- build output buffer ----
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    mov x10, x9                 // write cursor
    mov x11, #0                 // line index
out_loop:
    cmp x11, x23
    b.hs out_done
    add x12, x22, x11, lsl #4
    ldp x0, x1, [x12]
cpy_loop:
    cbz x1, cpy_done
    ldrb w2, [x0], #1
    strb w2, [x10], #1
    sub x1, x1, #1
    b cpy_loop
cpy_done:
    mov w2, #10
    strb w2, [x10], #1
    add x11, x11, #1
    b out_loop
out_done:
    sub x20, x10, x9            // total output length
    mov x21, x9
write_loop:
    cbz x20, do_exit
    mov x0, #1
    mov x1, x21
    mov x2, x20
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le do_exit
    add x21, x21, x0
    sub x20, x20, x0
    b write_loop
do_exit:
    mov x0, #0
    mov x8, #94
    svc #0
    b do_exit

// cmp_lines: x0=p1, x1=len1, x2=p2, x3=len2 -> w0 <0 / 0 / >0
cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, lo         // min length
    mov x5, #0
1:
    cmp x5, x4
    b.hs 2f
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    add x5, x5, #1
    cmp w6, w7
    b.eq 1b
    sub w0, w6, w7
    ret
2:
    sub x0, x1, x3
    ret

.bss
.balign 16
inbuf:
    .space 65536
.balign 16
lines_a:
    .space 65536
.balign 16
lines_b:
    .space 65536
.balign 16
outbuf:
    .space 69632
