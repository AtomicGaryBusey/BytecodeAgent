.text
.globl _start
_start:
    lea     rsi, [rip + inbuf]
    xor     ebx, ebx
read_loop:
    mov     edx, 65536
    sub     edx, ebx
    jz      read_done
    lea     rsi, [rip + inbuf]
    add     rsi, rbx
    xor     eax, eax
    xor     edi, edi
    syscall
    test    rax, rax
    jle     read_done
    add     rbx, rax
    jmp     read_loop
read_done:
    lea     rsi, [rip + inbuf]
    xor     ecx, ecx
find_loop:
    cmp     rcx, rbx
    jae     no_bang
    cmp     byte ptr [rsi + rcx], 0x21
    je      found
    inc     rcx
    jmp     find_loop
found:
    mov     r12, rcx
    lea     r13, [rsi + rcx + 1]
    lea     r14, [rsi + rbx]
    jmp     scan
no_bang:
    mov     r12, rbx
    lea     r13, [rsi + rbx]
    mov     r14, r13
scan:
    xor     ecx, ecx
    xor     edx, edx
    lea     rdi, [rip + bstack]
    lea     r8, [rip + jt]
scan_loop:
    cmp     rcx, r12
    jae     scan_done
    movzx   eax, byte ptr [rsi + rcx]
    cmp     al, 0x5b
    jne     1f
    mov     dword ptr [rdi + rdx*4], ecx
    inc     edx
    jmp     2f
1:
    cmp     al, 0x5d
    jne     2f
    test    edx, edx
    jz      bad
    dec     edx
    mov     eax, dword ptr [rdi + rdx*4]
    mov     dword ptr [r8 + rax*4], ecx
    mov     dword ptr [r8 + rcx*4], eax
2:
    inc     rcx
    jmp     scan_loop
scan_done:
    test    edx, edx
    jnz     bad

    lea     rbp, [rip + tape]
    lea     r9, [rip + outbuf]
    xor     r15d, r15d
    xor     ecx, ecx
run:
    cmp     rcx, r12
    jae     run_done
    movzx   eax, byte ptr [rsi + rcx]
    inc     rcx
    cmp     al, 0x3e
    je      op_right
    cmp     al, 0x3c
    je      op_left
    cmp     al, 0x2b
    je      op_inc
    cmp     al, 0x2d
    je      op_dec
    cmp     al, 0x5b
    je      op_open
    cmp     al, 0x5d
    je      op_close
    cmp     al, 0x2e
    je      op_out
    cmp     al, 0x2c
    je      op_in
    jmp     run
op_right:
    inc     rbp
    jmp     run
op_left:
    dec     rbp
    jmp     run
op_inc:
    inc     byte ptr [rbp]
    jmp     run
op_dec:
    dec     byte ptr [rbp]
    jmp     run
op_out:
    mov     al, byte ptr [rbp]
    mov     byte ptr [r9 + r15], al
    inc     r15
    cmp     r15, 65536
    jb      run
    call    flush
    jmp     run
op_in:
    xor     eax, eax
    cmp     r13, r14
    jae     1f
    mov     al, byte ptr [r13]
    inc     r13
1:
    mov     byte ptr [rbp], al
    jmp     run
op_open:
    cmp     byte ptr [rbp], 0
    jne     run
    mov     ecx, dword ptr [r8 + rcx*4 - 4]
    inc     rcx
    jmp     run
op_close:
    cmp     byte ptr [rbp], 0
    je      run
    mov     ecx, dword ptr [r8 + rcx*4 - 4]
    inc     rcx
    jmp     run

run_done:
    call    flush
    mov     eax, 231
    xor     edi, edi
    syscall
    jmp     run_done

bad:
    mov     eax, 231
    mov     edi, 1
    syscall
    jmp     bad

flush:
    test    r15, r15
    jz      1f
    push    rcx
    push    rsi
    mov     eax, 1
    mov     edi, 1
    mov     rsi, r9
    mov     rdx, r15
    syscall
    pop     rsi
    pop     rcx
    xor     r15d, r15d
1:
    ret
