.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                     // total bytes read
read_loop:
    mov x2, #32768
    sub x2, x2, x20
    cmp x2, #0
    b.le read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    // parse lines into lines_a: entry = offset | (len << 32)
    adrp x21, lines_a
    add x21, x21, :lo12:lines_a
    mov x22, #0                     // n lines
    mov x3, #0                      // start
    mov x4, #0                      // i
parse_loop:
    cmp x4, x20
    b.ge parse_end
    ldrb w5, [x19, x4]
    add x4, x4, #1
    cmp w5, #10
    b.ne parse_loop
    sub x6, x4, #1
    sub x6, x6, x3                  // len
    orr x7, x3, x6, lsl #32
    str x7, [x21, x22, lsl #3]
    add x22, x22, #1
    mov x3, x4
    b parse_loop
parse_end:
    cmp x3, x20
    b.ge sort
    sub x6, x20, x3
    orr x7, x3, x6, lsl #32
    str x7, [x21, x22, lsl #3]
    add x22, x22, #1

sort:
    adrp x23, lines_b
    add x23, x23, :lo12:lines_b
    mov x24, #1                     // width
sort_outer:
    cmp x24, x22
    b.ge sort_done
    mov x25, #0                     // i
sort_inner:
    cmp x25, x22
    b.ge sort_pass_done
    add x26, x25, x24
    cmp x26, x22
    csel x26, x26, x22, lt          // mid
    add x27, x25, x24, lsl #1
    cmp x27, x22
    csel x27, x27, x22, lt          // right
    mov x9, x25                     // a
    mov x10, x26                    // b
    mov x11, x25                    // k
merge_loop:
    cmp x9, x26
    b.ge merge_rest_b
    cmp x10, x27
    b.ge merge_rest_a
    ldr x0, [x21, x9, lsl #3]
    ldr x1, [x21, x10, lsl #3]
    bl compare
    cmp w0, #0
    b.gt take_b
    ldr x0, [x21, x9, lsl #3]
    str x0, [x23, x11, lsl #3]
    add x9, x9, #1
    add x11, x11, #1
    b merge_loop
take_b:
    ldr x0, [x21, x10, lsl #3]
    str x0, [x23, x11, lsl #3]
    add x10, x10, #1
    add x11, x11, #1
    b merge_loop
merge_rest_a:
    cmp x9, x26
    b.ge merge_done
    ldr x0, [x21, x9, lsl #3]
    str x0, [x23, x11, lsl #3]
    add x9, x9, #1
    add x11, x11, #1
    b merge_rest_a
merge_rest_b:
    cmp x10, x27
    b.ge merge_done
    ldr x0, [x21, x10, lsl #3]
    str x0, [x23, x11, lsl #3]
    add x10, x10, #1
    add x11, x11, #1
    b merge_rest_b
merge_done:
    add x25, x25, x24, lsl #1
    b sort_inner
sort_pass_done:
    mov x0, x21
    mov x21, x23
    mov x23, x0
    lsl x24, x24, #1
    b sort_outer
sort_done:

    // build output
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
    mov x9, #0                      // out length
    mov x10, #0                     // line index
out_loop:
    cmp x10, x22
    b.ge flush
    ldr x0, [x21, x10, lsl #3]
    and x2, x0, #0xffffffff
    add x2, x19, x2
    lsr x3, x0, #32
copy_loop:
    cbz x3, copy_end
    ldrb w4, [x2], #1
    strb w4, [x28, x9]
    add x9, x9, #1
    sub x3, x3, #1
    b copy_loop
copy_end:
    mov w4, #10
    strb w4, [x28, x9]
    add x9, x9, #1
    add x10, x10, #1
    b out_loop

flush:
    mov x10, #0                     // written
write_loop:
    sub x2, x9, x10
    cmp x2, #0
    b.le exit
    mov x0, #1
    add x1, x28, x10
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit
    add x10, x10, x0
    b write_loop

exit:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit

// compare: x0 = entry A, x1 = entry B, x19 = base; returns w0 <0,0,>0
compare:
    and x2, x0, #0xffffffff
    add x2, x19, x2
    lsr x3, x0, #32
    and x4, x1, #0xffffffff
    add x4, x19, x4
    lsr x5, x1, #32
    cmp x3, x5
    csel x6, x3, x5, lo
cmp_loop:
    cbz x6, cmp_len
    ldrb w7, [x2], #1
    ldrb w0, [x4], #1
    sub x6, x6, #1
    cmp w7, w0
    b.eq cmp_loop
    sub w0, w7, w0
    ret
cmp_len:
    sub w0, w3, w5
    ret

.bss
.balign 16
inbuf:
    .space 32800
.balign 16
outbuf:
    .space 40000
.balign 16
lines_a:
    .space 264000
.balign 16
lines_b:
    .space 264000
