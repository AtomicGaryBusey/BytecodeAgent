.text
.globl _start
_start:
        xor     ebp, ebp
read_loop:
        mov     edx, 65536
        sub     edx, ebp
        jle     read_done
        lea     rsi, [rip + inbuf]
        add     rsi, rbp
        xor     eax, eax
        xor     edi, edi
        syscall
        test    rax, rax
        jle     read_done
        add     rbp, rax
        jmp     read_loop
read_done:
        # find first '!'
        lea     rdi, [rip + inbuf]
        mov     rcx, rbp
        test    rcx, rcx
        jz      no_bang
        mov     al, '!'
        repne scasb
        jne     no_bang
        # found: rdi points after '!'
        lea     rsi, [rip + inbuf]
        mov     rcx, rdi
        sub     rcx, rsi
        dec     rcx                     # program length
        mov     r13, rdi                # input start
        lea     r14, [rip + inbuf]
        add     r14, rbp                # input end
        jmp     compile
no_bang:
        lea     rsi, [rip + inbuf]
        mov     rcx, rbp
        lea     r13, [rip + inbuf]
        add     r13, rbp
        mov     r14, r13
compile:
        xor     edx, edx                # code index
        xor     r8d, r8d                # bracket stack depth
        lea     r9, [rip + optab]
compile_loop:
        test    rcx, rcx
        jz      compile_done
        movzx   eax, byte ptr [rsi]
        inc     rsi
        dec     rcx
        movzx   eax, byte ptr [r9 + rax]
        test    eax, eax
        jz      compile_loop
        mov     byte ptr [code + rdx], al
        cmp     al, 7
        je      c_open
        cmp     al, 8
        je      c_close
        inc     rdx
        jmp     compile_loop
c_open:
        mov     dword ptr [bstack + r8*4], edx
        inc     r8
        inc     rdx
        jmp     compile_loop
c_close:
        test    r8, r8
        jz      bad
        dec     r8
        mov     eax, dword ptr [bstack + r8*4]
        lea     edi, [rdx + 1]
        mov     dword ptr [jt + rax*4], edi
        inc     eax
        mov     dword ptr [jt + rdx*4], eax
        inc     rdx
        jmp     compile_loop
compile_done:
        test    r8, r8
        jnz     bad
        mov     byte ptr [code + rdx], 0

        # execute
        xor     ebx, ebx
        lea     r12, [rip + tape]
        lea     r15, [rip + outbuf]
run:
        movzx   eax, byte ptr [code + rbx]
        inc     rbx
        jmp     qword ptr [dispatch + rax*8]
op_right:
        inc     r12
        jmp     run
op_left:
        dec     r12
        jmp     run
op_inc:
        inc     byte ptr [r12]
        jmp     run
op_dec:
        dec     byte ptr [r12]
        jmp     run
op_out:
        mov     al, byte ptr [r12]
        mov     byte ptr [r15], al
        inc     r15
        lea     rax, [rip + outbuf_end]
        cmp     r15, rax
        jb      run
        call    flush
        jmp     run
op_in:
        xor     eax, eax
        cmp     r13, r14
        jae     1f
        mov     al, byte ptr [r13]
        inc     r13
1:
        mov     byte ptr [r12], al
        jmp     run
op_open:
        cmp     byte ptr [r12], 0
        jne     run
        mov     ebx, dword ptr [jt + rbx*4 - 4]
        jmp     run
op_close:
        cmp     byte ptr [r12], 0
        je      run
        mov     ebx, dword ptr [jt + rbx*4 - 4]
        jmp     run
op_end:
        call    flush
        mov     eax, 231
        xor     edi, edi
        syscall
        jmp     op_end

bad:
        mov     eax, 231
        mov     edi, 1
        syscall
        jmp     bad

# flush: write outbuf[0 .. r15), reset r15
flush:
        lea     rsi, [rip + outbuf]
flush_loop:
        mov     rdx, r15
        sub     rdx, rsi
        jle     flush_done
        mov     eax, 1
        mov     edi, 1
        syscall
        test    rax, rax
        jle     flush_done
        add     rsi, rax
        jmp     flush_loop
flush_done:
        lea     r15, [rip + outbuf]
        ret

.section .rodata
.balign 8
dispatch:
        .quad   op_end
        .quad   op_right
        .quad   op_left
        .quad   op_inc
        .quad   op_dec
        .quad   op_out
        .quad   op_in
        .quad   op_open
        .quad   op_close
optab:
        .zero   43
        .byte   3, 6, 4, 5
        .zero   13
        .byte   2, 0, 1
        .zero   28
        .byte   7, 0, 8
        .zero   162

.bss
.balign 16
inbuf:
        .space  65552
code:
        .space  65552
.balign 16
jt:
        .space  262160
bstack:
        .space  262160
tape:
        .space  30016
outbuf:
        .space  65536
outbuf_end:
        .space  16
