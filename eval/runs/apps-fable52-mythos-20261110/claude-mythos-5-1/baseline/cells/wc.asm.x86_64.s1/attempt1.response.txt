.bss
.balign 16
buf:    .space 65536
outbuf: .space 64

.text
.globl _start
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # in_word flag
read_loop:
    xor eax, eax            # read
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    lea rdx, [rsi + rax]
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    sub al, 9
    cmp al, 4
    jbe ws
    test r15, r15
    jnz 2f
    inc r13
    mov r15d, 1
    jmp 2f
ws:
    xor r15d, r15d
2:
    cmp rsi, rdx
    jb byte_loop
    jmp read_loop

done:
    lea rdi, [rip + outbuf + 63]
    mov byte ptr [rdi], 10
    mov rax, r14
    call putnum
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call putnum
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call putnum
    lea rdx, [rip + outbuf + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 231
    xor edi, edi
    syscall
    jmp done

# rax = value, rdi -> byte after digits; writes digits backwards, rdi -> first digit
putnum:
    mov ebx, 10
1:
    xor edx, edx
    div rbx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret
