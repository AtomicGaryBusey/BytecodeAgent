.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // in-word flag
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf
read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_reading
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
proc_loop:
    ldrb w3, [x1], #1
    cmp w3, #0x0A
    cinc x19, x19, eq
    cmp w3, #0x20
    b.eq is_space
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_space
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_space:
    mov x22, #0
next:
    cmp x1, x2
    b.lo proc_loop
    b read_loop
done_reading:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #64
    mov x2, x1
    mov w3, #10
    strb w3, [x1, #-1]!
    mov x0, x21
    bl fmt
    mov w3, #0x20
    strb w3, [x1, #-1]!
    mov x0, x20
    bl fmt
    strb w3, [x1, #-1]!
    mov x0, x19
    bl fmt
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #94
    svc #0

// x0 = value, x1 = pointer (stores with pre-decrement); clobbers x0, x4, x5, x6
fmt:
    mov x6, #10
1:
    udiv x4, x0, x6
    msub x5, x4, x6, x0
    add w5, w5, #48
    strb w5, [x1, #-1]!
    mov x0, x4
    cbnz x0, 1b
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 64
