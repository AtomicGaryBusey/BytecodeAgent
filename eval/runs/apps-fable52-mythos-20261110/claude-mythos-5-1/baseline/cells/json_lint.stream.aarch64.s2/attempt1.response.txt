.text
.globl _start

_start:
    adrp    x21, buf
    add     x21, x21, :lo12:buf
    mov     x20, #0                 // bytes read so far
read_loop:
    mov     x2, #65536
    add     x2, x2, #16
    sub     x2, x2, x20
    cmp     x2, #0
    b.le    read_done
    mov     x0, #0
    add     x1, x21, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    add     x20, x21, x20           // x20 = end pointer
    mov     x19, x21                // x19 = current position
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x19, x20
    b.ne    error
    adrp    x1, ok_msg
    add     x1, x1, :lo12:ok_msg
    mov     x2, #3
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       .

// peek: w0 = byte at x19, or 0x100 at end of input
peek:
    cmp     x19, x20
    b.hs    1f
    ldrb    w0, [x19]
    ret
1:  mov     w0, #0x100
    ret

skip_ws:
1:  cmp     x19, x20
    b.hs    2f
    ldrb    w0, [x19]
    cmp     w0, #0x20
    b.eq    3f
    cmp     w0, #0x09
    b.eq    3f
    cmp     w0, #0x0a
    b.eq    3f
    cmp     w0, #0x0d
    b.ne    2f
3:  add     x19, x19, #1
    b       1b
2:  ret

skip_digits:
1:  cmp     x19, x20
    b.hs    2f
    ldrb    w0, [x19]
    sub     w0, w0, #0x30
    cmp     w0, #9
    b.hi    2f
    add     x19, x19, #1
    b       1b
2:  ret

need_digits:
    cmp     x19, x20
    b.hs    error
    ldrb    w0, [x19]
    sub     w0, w0, #0x30
    cmp     w0, #9
    b.hi    error
    b       skip_digits

parse_string:
    add     x19, x19, #1            // opening quote
1:  cmp     x19, x20
    b.hs    error
    ldrb    w0, [x19]
    cmp     w0, #0x22
    b.eq    9f
    cmp     w0, #0x20
    b.lo    error
    cmp     w0, #0x5c
    b.ne    2f
    add     x19, x19, #1
    cmp     x19, x20
    b.hs    error
    ldrb    w0, [x19]
    cmp     w0, #0x75               // u
    b.eq    3f
    cmp     w0, #0x22
    b.eq    2f
    cmp     w0, #0x5c
    b.eq    2f
    cmp     w0, #0x2f
    b.eq    2f
    cmp     w0, #0x62               // b
    b.eq    2f
    cmp     w0, #0x66               // f
    b.eq    2f
    cmp     w0, #0x6e               // n
    b.eq    2f
    cmp     w0, #0x72               // r
    b.eq    2f
    cmp     w0, #0x74               // t
    b.eq    2f
    b       error
3:  add     x19, x19, #1
    mov     w3, #4
4:  cmp     x19, x20
    b.hs    error
    ldrb    w0, [x19]
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    5f
    orr     w1, w0, #0x20
    sub     w1, w1, #0x61
    cmp     w1, #5
    b.hi    error
5:  add     x19, x19, #1
    subs    w3, w3, #1
    b.ne    4b
    b       1b
2:  add     x19, x19, #1
    b       1b
9:  add     x19, x19, #1
    ret

parse_value:
    stp     x29, x30, [sp, #-16]!
    bl      peek
    cmp     w0, #0x7b               // {
    b.eq    pv_object
    cmp     w0, #0x5b               // [
    b.eq    pv_array
    cmp     w0, #0x22               // "
    b.eq    pv_string
    cmp     w0, #0x74               // t
    b.eq    pv_true
    cmp     w0, #0x66               // f
    b.eq    pv_false
    cmp     w0, #0x6e               // n
    b.eq    pv_null
    cmp     w0, #0x2d               // -
    b.eq    pv_number
    sub     w1, w0, #0x30
    cmp     w1, #9
    b.ls    pv_number
    b       error

pv_object:
    add     x19, x19, #1
    bl      skip_ws
    bl      peek
    cmp     w0, #0x7d               // }
    b.eq    pv_obj_close
pv_obj_member:
    bl      peek
    cmp     w0, #0x22
    b.ne    error
    bl      parse_string
    bl      skip_ws
    bl      peek
    cmp     w0, #0x3a               // :
    b.ne    error
    add     x19, x19, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    bl      peek
    cmp     w0, #0x2c               // ,
    b.eq    1f
    cmp     w0, #0x7d
    b.ne    error
pv_obj_close:
    add     x19, x19, #1
    b       pv_ret
1:  add     x19, x19, #1
    bl      skip_ws
    b       pv_obj_member

pv_array:
    add     x19, x19, #1
    bl      skip_ws
    bl      peek
    cmp     w0, #0x5d               // ]
    b.eq    pv_arr_close
pv_arr_elem:
    bl      parse_value
    bl      skip_ws
    bl      peek
    cmp     w0, #0x2c
    b.eq    1f
    cmp     w0, #0x5d
    b.ne    error
pv_arr_close:
    add     x19, x19, #1
    b       pv_ret
1:  add     x19, x19, #1
    bl      skip_ws
    b       pv_arr_elem

pv_string:
    bl      parse_string
    b       pv_ret

pv_true:
    adrp    x1, lit_true
    add     x1, x1, :lo12:lit_true
    b       pv_literal
pv_false:
    adrp    x1, lit_false
    add     x1, x1, :lo12:lit_false
    b       pv_literal
pv_null:
    adrp    x1, lit_null
    add     x1, x1, :lo12:lit_null
pv_literal:
1:  ldrb    w2, [x1], #1
    cbz     w2, pv_ret
    bl      peek
    cmp     w0, w2
    b.ne    error
    add     x19, x19, #1
    b       1b

pv_number:
    cmp     w0, #0x2d
    b.ne    1f
    add     x19, x19, #1
    bl      peek
1:  cmp     w0, #0x30
    b.eq    2f
    sub     w1, w0, #0x31
    cmp     w1, #8
    b.hi    error
    bl      skip_digits
    b       3f
2:  add     x19, x19, #1
3:  bl      peek
    cmp     w0, #0x2e               // .
    b.ne    4f
    add     x19, x19, #1
    bl      need_digits
    bl      peek
4:  cmp     w0, #0x65               // e
    b.eq    5f
    cmp     w0, #0x45               // E
    b.ne    pv_ret
5:  add     x19, x19, #1
    bl      peek
    cmp     w0, #0x2b               // +
    b.eq    6f
    cmp     w0, #0x2d               // -
    b.ne    7f
6:  add     x19, x19, #1
7:  bl      need_digits
pv_ret:
    ldp     x29, x30, [sp], #16
    ret

error:
    sub     x0, x19, x21            // offset
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    add     x2, x1, #31
    mov     w3, #10
    strb    w3, [x2]
    mov     x4, #10
1:  udiv    x5, x0, x4
    msub    x6, x5, x4, x0
    add     w6, w6, #0x30
    sub     x2, x2, #1
    strb    w6, [x2]
    mov     x0, x5
    cbnz    x0, 1b
    sub     x2, x2, #9
    adrp    x3, err_msg
    add     x3, x3, :lo12:err_msg
    mov     x5, #0
2:  ldrb    w6, [x3, x5]
    strb    w6, [x2, x5]
    add     x5, x5, #1
    cmp     x5, #9
    b.lo    2b
    add     x3, x1, #32
    sub     x3, x3, x2
    mov     x1, x2
    mov     x2, x3
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .

.section .rodata
ok_msg:
    .ascii  "ok\n"
err_msg:
    .ascii  "error at "
lit_true:
    .asciz  "true"
lit_false:
    .asciz  "false"
lit_null:
    .asciz  "null"

.bss
.balign 16
buf:
    .space  65552
.balign 16
outbuf:
    .space  48
