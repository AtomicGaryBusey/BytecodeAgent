.text
.globl _start
_start:
        xor     ebx, ebx                # bytes read so far
read_loop:
        mov     eax, 0
        xor     edi, edi
        lea     rsi, [rip + inbuf]
        add     rsi, rbx
        mov     edx, inbuf_cap
        sub     edx, ebx
        jbe     read_done
        syscall
        test    rax, rax
        jle     read_done
        add     rbx, rax
        jmp     read_loop
read_done:
        lea     rsi, [rip + inbuf]
        lea     rdi, [rip + outbuf]
        mov     rcx, rbx                # remaining input bytes
        xor     r8d, r8d                # column
        lea     r9, [rip + table]
enc_loop:
        cmp     rcx, 3
        jb      enc_tail
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        movzx   edx, byte ptr [rsi + 1]
        shl     edx, 8
        or      eax, edx
        movzx   edx, byte ptr [rsi + 2]
        or      eax, edx
        mov     edx, eax
        shr     edx, 18
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi], dl
        mov     edx, eax
        shr     edx, 12
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 1], dl
        mov     edx, eax
        shr     edx, 6
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 2], dl
        mov     edx, eax
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 3], dl
        add     rsi, 3
        sub     rcx, 3
        add     rdi, 4
        add     r8d, 4
        cmp     r8d, 76
        jne     enc_loop
        mov     byte ptr [rdi], 10
        inc     rdi
        xor     r8d, r8d
        jmp     enc_loop
enc_tail:
        test    rcx, rcx
        jz      enc_end
        movzx   eax, byte ptr [rsi]
        shl     eax, 16
        cmp     rcx, 2
        jb      tail_one
        movzx   edx, byte ptr [rsi + 1]
        shl     edx, 8
        or      eax, edx
        mov     edx, eax
        shr     edx, 6
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 2], dl
        mov     byte ptr [rdi + 3], '='
        jmp     tail_common
tail_one:
        mov     byte ptr [rdi + 2], '='
        mov     byte ptr [rdi + 3], '='
tail_common:
        mov     edx, eax
        shr     edx, 18
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi], dl
        mov     edx, eax
        shr     edx, 12
        and     edx, 63
        mov     dl, byte ptr [r9 + rdx]
        mov     byte ptr [rdi + 1], dl
        add     rdi, 4
        add     r8d, 4
enc_end:
        test    r8d, r8d
        jz      do_write
        mov     byte ptr [rdi], 10
        inc     rdi
do_write:
        lea     rsi, [rip + outbuf]
        mov     rdx, rdi
        sub     rdx, rsi                # total output length
write_loop:
        test    rdx, rdx
        jle     done
        mov     eax, 1
        mov     edi, 1
        syscall
        test    rax, rax
        jle     done
        add     rsi, rax
        sub     rdx, rax
        jmp     write_loop
done:
        mov     eax, 60
        xor     edi, edi
        syscall
        jmp     done
