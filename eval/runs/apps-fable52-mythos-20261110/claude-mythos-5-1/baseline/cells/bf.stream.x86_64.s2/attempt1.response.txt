.text
.globl _start

.equ INBUF_SIZE, 65552
.equ OUTBUF_SIZE, 65536

_start:
    xor r12d, r12d                  # total bytes read
read_loop:
    mov edx, INBUF_SIZE
    sub edx, r12d
    jle read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    # find first '!'
    lea rsi, [rip + inbuf]
    xor ecx, ecx
find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r13, rcx                    # program length
    lea r14, [rcx + 1]              # input start index
    jmp bang_done
no_bang:
    mov r13, r12
    mov r14, r12
bang_done:

    # bracket matching
    xor ecx, ecx                    # pc
    xor edx, edx                    # stack depth
    lea r8, [rip + jmptab]
    lea r9, [rip + bstack]
match_loop:
    cmp rcx, r13
    jae match_end
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b                    # '['
    jne 1f
    mov dword ptr [r9 + rdx*4], ecx
    inc edx
    jmp 2f
1:
    cmp al, 0x5d                    # ']'
    jne 2f
    test edx, edx
    jz fail
    dec edx
    mov eax, dword ptr [r9 + rdx*4]
    mov dword ptr [r8 + rax*4], ecx
    mov dword ptr [r8 + rcx*4], eax
2:
    inc rcx
    jmp match_loop
match_end:
    test edx, edx
    jnz fail

    # interpreter
    lea rbp, [rip + tape]           # data pointer
    lea r10, [rip + outbuf]         # output write pointer
    lea r15, [rip + outbuf]         # output base
    xor ebx, ebx                    # pc
run_loop:
    cmp rbx, r13
    jae run_end
    movzx eax, byte ptr [rsi + rbx]
    inc rbx
    cmp al, 0x2b                    # '+'
    je c_inc
    cmp al, 0x2d                    # '-'
    je c_dec
    cmp al, 0x3e                    # '>'
    je c_right
    cmp al, 0x3c                    # '<'
    je c_left
    cmp al, 0x5b                    # '['
    je c_open
    cmp al, 0x5d                    # ']'
    je c_close
    cmp al, 0x2e                    # '.'
    je c_out
    cmp al, 0x2c                    # ','
    je c_in
    jmp run_loop
c_inc:
    inc byte ptr [rbp]
    jmp run_loop
c_dec:
    dec byte ptr [rbp]
    jmp run_loop
c_right:
    inc rbp
    jmp run_loop
c_left:
    dec rbp
    jmp run_loop
c_open:
    cmp byte ptr [rbp], 0
    jne run_loop
    mov ebx, dword ptr [r8 + rbx*4 - 4]
    inc rbx
    jmp run_loop
c_close:
    cmp byte ptr [rbp], 0
    je run_loop
    mov ebx, dword ptr [r8 + rbx*4 - 4]
    inc rbx
    jmp run_loop
c_out:
    mov al, byte ptr [rbp]
    mov byte ptr [r10], al
    inc r10
    mov rax, r10
    sub rax, r15
    cmp rax, OUTBUF_SIZE
    jb run_loop
    call flush
    jmp run_loop
c_in:
    cmp r14, r12
    jae 1f
    mov al, byte ptr [rsi + r14]
    inc r14
    mov byte ptr [rbp], al
    jmp run_loop
1:
    mov byte ptr [rbp], 0
    jmp run_loop

run_end:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp run_end

fail:
    mov eax, 231
    mov edi, 1
    syscall
    jmp fail

# flush: write outbuf[0 .. r10) to stdout, reset r10
flush:
    push rsi
    push rdx
    mov rsi, r15
flush_loop:
    mov rdx, r10
    sub rdx, rsi
    jle flush_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle flush_done
    add rsi, rax
    jmp flush_loop
flush_done:
    mov r10, r15
    pop rdx
    pop rsi
    ret

.bss
.balign 16
inbuf:
    .space INBUF_SIZE + 16
.balign 16
outbuf:
    .space OUTBUF_SIZE + 16
.balign 16
jmptab:
    .space 65552 * 4
.balign 16
bstack:
    .space 65552 * 4
.balign 16
tape:
    .space 30000 + 64
