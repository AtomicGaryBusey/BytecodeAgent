.text
.globl _start
_start:
    xor r12d, r12d              # lines
    xor r13d, r13d              # words
    xor r14d, r14d              # bytes
    xor r15d, r15d              # in-word flag

read_loop:
    xor eax, eax                # read
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    add r14, rax
    lea rsi, [rip + inbuf]
    lea rdx, [rsi + rax]
byte_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je is_ws
    sub al, 9
    cmp al, 4
    jbe is_ws
    test r15, r15
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
is_ws:
    xor r15d, r15d
next_byte:
    inc rsi
    cmp rsi, rdx
    jb byte_loop
    jmp read_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp done_reading

# rax = value, rdi = destination; returns rdi advanced past digits
print_num:
    lea rsi, [rip + tmp + 32]
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rcx, [rip + tmp + 32]
    sub rcx, rsi
    rep movsb
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 128
tmp:
    .space 32
