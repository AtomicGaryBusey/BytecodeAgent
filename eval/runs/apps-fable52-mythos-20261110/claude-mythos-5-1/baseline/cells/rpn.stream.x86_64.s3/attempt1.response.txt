.text
.globl _start

INBUF_SIZE = 66560
OUTBUF_SIZE = 65536

_start:
    lea rbp, [rip + inbuf]
read_loop:
    lea rdx, [rip + inbuf]
    add rdx, INBUF_SIZE
    sub rdx, rbp
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbp
    syscall
    test rax, rax
    jle read_done
    add rbp, rax
    jmp read_loop
read_done:
    lea rbx, [rip + inbuf]
    xor r15d, r15d

line_loop:
    cmp rbx, rbp
    jae finish
    mov r12, rbx
find_nl:
    cmp r12, rbp
    jae found_nl
    cmp byte ptr [r12], 10
    je found_nl
    inc r12
    jmp find_nl
found_nl:
    xor r13d, r13d
    mov rsi, rbx
token_loop:
    cmp rsi, r12
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je sep
    cmp al, 9
    jne tok_start
sep:
    inc rsi
    jmp token_loop
tok_start:
    mov rdi, rsi
find_tok_end:
    inc rsi
    cmp rsi, r12
    jae tok_end
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je tok_end
    cmp al, 9
    jne find_tok_end
tok_end:
    mov rcx, rsi
    sub rcx, rdi
    movzx eax, byte ptr [rdi]
    cmp rcx, 1
    jne parse_num
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
parse_num:
    xor edx, edx
    cmp al, '-'
    jne no_neg
    inc rdi
    dec rcx
    mov edx, 1
no_neg:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor r8d, r8d
num_loop:
    movzx eax, byte ptr [rdi]
    sub eax, '0'
    cmp eax, 9
    ja line_error
    imul r8, r8, 10
    add r8, rax
    inc rdi
    dec rcx
    jnz num_loop
    test edx, edx
    jz push_val
    neg r8
push_val:
    lea rax, [rip + vstack]
    mov qword ptr [rax + r13*8], r8
    inc r13
    jmp token_loop

op_add:
    cmp r13, 2
    jb line_error
    lea rdx, [rip + vstack]
    mov rcx, qword ptr [rdx + r13*8 - 8]
    dec r13
    mov rax, qword ptr [rdx + r13*8 - 8]
    add rax, rcx
store_result:
    mov qword ptr [rdx + r13*8 - 8], rax
    jmp token_loop
op_sub:
    cmp r13, 2
    jb line_error
    lea rdx, [rip + vstack]
    mov rcx, qword ptr [rdx + r13*8 - 8]
    dec r13
    mov rax, qword ptr [rdx + r13*8 - 8]
    sub rax, rcx
    jmp store_result
op_mul:
    cmp r13, 2
    jb line_error
    lea rdx, [rip + vstack]
    mov rcx, qword ptr [rdx + r13*8 - 8]
    dec r13
    mov rax, qword ptr [rdx + r13*8 - 8]
    imul rax, rcx
    jmp store_result
op_div:
    cmp r13, 2
    jb line_error
    lea rdx, [rip + vstack]
    mov rcx, qword ptr [rdx + r13*8 - 8]
    dec r13
    mov rax, qword ptr [rdx + r13*8 - 8]
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
1:
    cqo
    idiv rcx
    lea rdx, [rip + vstack]
    jmp store_result
op_mod:
    cmp r13, 2
    jb line_error
    lea rdx, [rip + vstack]
    mov rcx, qword ptr [rdx + r13*8 - 8]
    dec r13
    mov rax, qword ptr [rdx + r13*8 - 8]
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
1:
    cqo
    idiv rcx
    mov rax, rdx
    lea rdx, [rip + vstack]
    jmp store_result

line_end:
    cmp r13, 1
    jne line_error
    lea rax, [rip + vstack]
    mov rax, qword ptr [rax]
    call print_num
    jmp next_line
line_error:
    lea rsi, [rip + errmsg]
    mov edx, 6
    call out_write
next_line:
    lea rbx, [r12 + 1]
    jmp line_loop

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp finish

# rax = value
print_num:
    sub rsp, 32
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 10
    mov r8, rax
    test rax, rax
    jns 1f
    neg rax
1:
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 2b
    test r8, r8
    jns 3f
    dec rdi
    mov byte ptr [rdi], '-'
3:
    mov rsi, rdi
    lea rdx, [rsp + 32]
    sub rdx, rdi
    call out_write
    add rsp, 32
    ret

# rsi = ptr, rdx = len
out_write:
    lea rax, [r15 + rdx]
    cmp rax, OUTBUF_SIZE
    jbe 1f
    push rsi
    push rdx
    call flush
    pop rdx
    pop rsi
1:
    lea rdi, [rip + outbuf]
    add rdi, r15
    mov rcx, rdx
    add r15, rdx
    rep movsb
    ret

flush:
    lea rsi, [rip + outbuf]
    mov rdx, r15
1:
    test rdx, rdx
    jz 2f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jmp 1b
2:
    xor r15d, r15d
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 16
inbuf:
    .space 66560
.balign 16
outbuf:
    .space 65536
.balign 16
vstack:
    .space 272000
