.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    mov r13, r12
read_loop:
    lea rdx, [rip + inbuf_end]
    sub rdx, r13
    jz read_done
    xor edi, edi
    mov rsi, r13
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r15, [rip + outbuf]

line_loop:
    cmp r12, r13
    jae finish
    lea r14, [rip + stack]
tok_loop:
    cmp r12, r13
    jae line_end
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    cmp al, 0x0a
    je line_end_nl
    mov rsi, r12
tok_scan:
    inc r12
    cmp r12, r13
    jae tok_scanned
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je tok_scanned
    cmp al, 0x09
    je tok_scanned
    cmp al, 0x0a
    jne tok_scan
tok_scanned:
    mov rcx, r12
    sub rcx, rsi
    movzx eax, byte ptr [rsi]
    cmp rcx, 1
    jne tok_number
    cmp al, 0x2b
    je op_add
    cmp al, 0x2d
    je op_sub
    cmp al, 0x2a
    je op_mul
    cmp al, 0x2f
    je op_div
    cmp al, 0x25
    je op_mod
tok_number:
    xor r8d, r8d
    cmp al, 0x2d
    jne 1f
    inc r8d
    inc rsi
    dec rcx
1:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor edx, edx
2:
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 9
    ja line_error
    imul rdx, rdx, 10
    add rdx, rax
    inc rsi
    dec rcx
    jnz 2b
    test r8d, r8d
    jz 3f
    neg rdx
3:
    mov qword ptr [r14], rdx
    add r14, 8
    jmp tok_loop

skip_sep:
    inc r12
    jmp tok_loop

op_add:
    lea rax, [rip + stack + 16]
    cmp r14, rax
    jb line_error
    mov rbx, qword ptr [r14 - 8]
    mov rax, qword ptr [r14 - 16]
    add rax, rbx
    mov qword ptr [r14 - 16], rax
    sub r14, 8
    jmp tok_loop
op_sub:
    lea rax, [rip + stack + 16]
    cmp r14, rax
    jb line_error
    mov rbx, qword ptr [r14 - 8]
    mov rax, qword ptr [r14 - 16]
    sub rax, rbx
    mov qword ptr [r14 - 16], rax
    sub r14, 8
    jmp tok_loop
op_mul:
    lea rax, [rip + stack + 16]
    cmp r14, rax
    jb line_error
    mov rbx, qword ptr [r14 - 8]
    mov rax, qword ptr [r14 - 16]
    imul rax, rbx
    mov qword ptr [r14 - 16], rax
    sub r14, 8
    jmp tok_loop
op_div:
    lea rax, [rip + stack + 16]
    cmp r14, rax
    jb line_error
    mov rbx, qword ptr [r14 - 8]
    mov rax, qword ptr [r14 - 16]
    test rbx, rbx
    jz line_error
    cmp rbx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
1:
    cqo
    idiv rbx
    mov qword ptr [r14 - 16], rax
    sub r14, 8
    jmp tok_loop
op_mod:
    lea rax, [rip + stack + 16]
    cmp r14, rax
    jb line_error
    mov rbx, qword ptr [r14 - 8]
    mov rax, qword ptr [r14 - 16]
    test rbx, rbx
    jz line_error
    cmp rbx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
1:
    cqo
    idiv rbx
    mov qword ptr [r14 - 16], rdx
    sub r14, 8
    jmp tok_loop

line_error:
    cmp r12, r13
    jae 2f
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 0x0a
    jne line_error
2:
    call ensure_space
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6
    jmp line_loop

line_end_nl:
    inc r12
line_end:
    lea rax, [rip + stack + 8]
    cmp r14, rax
    jne 2b
    call ensure_space
    mov rax, qword ptr [rip + stack]
    test rax, rax
    jns 1f
    mov byte ptr [r15], 0x2d
    inc r15
    neg rax
1:
    lea rdi, [rip + numtmp + 32]
    mov ecx, 10
3:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 3b
    mov rsi, rdi
    lea rcx, [rip + numtmp + 32]
    sub rcx, rsi
    mov rdi, r15
    rep movsb
    mov r15, rdi
    mov byte ptr [r15], 0x0a
    inc r15
    jmp line_loop

ensure_space:
    lea rax, [rip + outbuf + 65536 - 32]
    cmp r15, rax
    jb 1f
    call flush
1:
    ret

flush:
    lea rsi, [rip + outbuf]
2:
    mov rdx, r15
    sub rdx, rsi
    jle 3f
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle 3f
    add rsi, rax
    jmp 2b
3:
    lea r15, [rip + outbuf]
    ret

finish:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp finish

.bss
.balign 16
inbuf:
    .space 65536 + 64
inbuf_end:
outbuf:
    .space 65536 + 64
numtmp:
    .space 64
.balign 16
stack:
    .space 32768 * 8 + 64
