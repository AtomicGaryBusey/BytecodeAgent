    .text
    .globl _start
_start:
    // ---- read all of stdin into buf ----
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0                 // total bytes read
read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #40000
    sub     x2, x2, x20
    cmp     x2, #0
    b.le    read_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    // ---- parse lines into (ptr,len) entries ----
    adrp    x21, lines
    add     x21, x21, :lo12:lines
    mov     x22, #0                 // line count
    mov     x9, x19                 // line start
    mov     x10, x19                // cursor
    add     x11, x19, x20           // end of data
parse_loop:
    cmp     x10, x11
    b.eq    parse_end
    ldrb    w12, [x10], #1
    cmp     w12, #10
    b.ne    parse_loop
    sub     x13, x10, x9
    sub     x13, x13, #1
    add     x14, x21, x22, lsl #4
    stp     x9, x13, [x14]
    add     x22, x22, #1
    mov     x9, x10
    b       parse_loop
parse_end:
    cmp     x9, x11
    b.eq    sort_start
    sub     x13, x11, x9
    add     x14, x21, x22, lsl #4
    stp     x9, x13, [x14]
    add     x22, x22, #1

    // ---- bottom-up merge sort ----
sort_start:
    adrp    x24, tmp
    add     x24, x24, :lo12:tmp
    mov     x25, #1                 // width
sort_outer:
    cmp     x25, x22
    b.ge    sort_done
    mov     x26, #0                 // i
sort_inner:
    cmp     x26, x22
    b.ge    sort_pass_done
    add     x27, x26, x25           // mid
    cmp     x27, x22
    csel    x27, x27, x22, lo
    add     x28, x27, x25           // right
    cmp     x28, x22
    csel    x28, x28, x22, lo
    mov     x9, x26                 // a
    mov     x10, x27                // b
    mov     x11, x26                // k
merge_loop:
    cmp     x9, x27
    b.ge    merge_take_b_rest
    cmp     x10, x28
    b.ge    merge_take_a
    add     x0, x21, x9, lsl #4
    add     x1, x21, x10, lsl #4
    bl      cmp_lines
    cmp     x0, #0
    b.gt    merge_take_b
merge_take_a:
    add     x12, x21, x9, lsl #4
    ldp     x13, x14, [x12]
    add     x12, x24, x11, lsl #4
    stp     x13, x14, [x12]
    add     x9, x9, #1
    add     x11, x11, #1
    b       merge_loop
merge_take_b:
    add     x12, x21, x10, lsl #4
    ldp     x13, x14, [x12]
    add     x12, x24, x11, lsl #4
    stp     x13, x14, [x12]
    add     x10, x10, #1
    add     x11, x11, #1
    b       merge_loop
merge_take_b_rest:
    cmp     x10, x28
    b.ge    merge_done
    add     x12, x21, x10, lsl #4
    ldp     x13, x14, [x12]
    add     x12, x24, x11, lsl #4
    stp     x13, x14, [x12]
    add     x10, x10, #1
    add     x11, x11, #1
    b       merge_take_b_rest
merge_done:
    add     x26, x26, x25, lsl #1
    b       sort_inner
sort_pass_done:
    mov     x12, x21
    mov     x21, x24
    mov     x24, x12
    lsl     x25, x25, #1
    b       sort_outer
sort_done:

    // ---- build output buffer ----
    adrp    x19, outbuf
    add     x19, x19, :lo12:outbuf
    mov     x20, #0                 // output length
    mov     x9, #0                  // line index
out_loop:
    cmp     x9, x22
    b.ge    out_done
    add     x12, x21, x9, lsl #4
    ldp     x13, x14, [x12]
    add     x15, x19, x20
1:
    cbz     x14, 2f
    ldrb    w16, [x13], #1
    strb    w16, [x15], #1
    sub     x14, x14, #1
    b       1b
2:
    mov     w16, #10
    strb    w16, [x15], #1
    sub     x20, x15, x19
    add     x9, x9, #1
    b       out_loop
out_done:

    // ---- write output ----
    mov     x9, #0
write_loop:
    cmp     x9, x20
    b.ge    do_exit
    mov     x0, #1
    add     x1, x19, x9
    sub     x2, x20, x9
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    do_exit
    add     x9, x9, x0
    b       write_loop
do_exit:
    mov     x0, #0
    mov     x8, #93
    svc     #0
    b       do_exit

    // cmp_lines: x0 -> entry A, x1 -> entry B; returns x0 <0, 0, >0
cmp_lines:
    ldp     x2, x3, [x0]
    ldp     x4, x5, [x1]
    cmp     x3, x5
    csel    x6, x3, x5, lo          // min length
    mov     x7, #0
1:
    cmp     x7, x6
    b.eq    2f
    ldrb    w0, [x2, x7]
    ldrb    w1, [x4, x7]
    add     x7, x7, #1
    cmp     w0, w1
    b.eq    1b
    sub     x0, x0, x1
    ret
2:
    sub     x0, x3, x5
    ret

    .bss
    .balign 16
buf:
    .space  40000
    .balign 16
lines:
    .space  16384
    .balign 16
tmp:
    .space  16384
    .balign 16
outbuf:
    .space  48000
