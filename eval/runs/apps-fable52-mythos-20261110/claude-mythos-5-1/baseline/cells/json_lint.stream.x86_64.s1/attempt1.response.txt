BUFSZ = 69632

.section .rodata
ok_msg: .ascii "ok\n"
err_msg: .ascii "error at "
err_len = . - err_msg
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
buf: .zero BUFSZ
outbuf: .zero 64
numbuf: .zero 32

.text
.globl _start
_start:
    lea rbx, [rip + buf]
    xor r13d, r13d
1:
    mov edx, BUFSZ - 1
    sub edx, r13d
    jbe 2f
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    xor r12d, r12d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall

error:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + err_msg]
    mov ecx, err_len
    rep movsb
    mov rax, r12
    lea rsi, [rip + numbuf + 20]
    mov r8, rsi
    mov ecx, 10
3:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 3b
    mov rcx, r8
    sub rcx, rsi
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0a
    je 1f
    cmp al, 0x0d
    jne 2f
1:
    inc r12
    jmp skip_ws
2:
    ret

parse_value:
    movzx eax, byte ptr [rbx + r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, '-'
    je parse_number
    lea ecx, [rax - '0']
    cmp ecx, 9
    jbe parse_number
    cmp al, 't'
    je 1f
    cmp al, 'f'
    je 2f
    cmp al, 'n'
    je 3f
    jmp error
1:
    lea rsi, [rip + lit_true]
    jmp parse_literal
2:
    lea rsi, [rip + lit_false]
    jmp parse_literal
3:
    lea rsi, [rip + lit_null]
parse_literal:
4:
    movzx eax, byte ptr [rsi]
    test al, al
    jz 5f
    cmp al, byte ptr [rbx + r12]
    jne error
    inc r12
    inc rsi
    jmp 4b
5:
    ret

parse_string:
    inc r12
1:
    movzx eax, byte ptr [rbx + r12]
    cmp al, 0x22
    je 9f
    cmp al, 0x5c
    je 2f
    cmp al, 0x20
    jb error
    inc r12
    jmp 1b
2:
    inc r12
    movzx eax, byte ptr [rbx + r12]
    cmp al, 'u'
    je 3f
    cmp al, 0x22
    je 4f
    cmp al, 0x5c
    je 4f
    cmp al, '/'
    je 4f
    cmp al, 'b'
    je 4f
    cmp al, 'f'
    je 4f
    cmp al, 'n'
    je 4f
    cmp al, 'r'
    je 4f
    cmp al, 't'
    je 4f
    jmp error
4:
    inc r12
    jmp 1b
3:
    inc r12
    mov ecx, 4
5:
    movzx eax, byte ptr [rbx + r12]
    lea edx, [rax - '0']
    cmp edx, 9
    jbe 6f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
6:
    inc r12
    dec ecx
    jnz 5b
    jmp 1b
9:
    inc r12
    ret

parse_number:
    movzx eax, byte ptr [rbx + r12]
    cmp al, '-'
    jne 1f
    inc r12
    movzx eax, byte ptr [rbx + r12]
1:
    cmp al, '0'
    jne 2f
    inc r12
    jmp num_frac
2:
    lea ecx, [rax - '1']
    cmp ecx, 8
    ja error
    inc r12
    call skip_digits
num_frac:
    movzx eax, byte ptr [rbx + r12]
    cmp al, '.'
    jne num_exp
    inc r12
    call need_digits
num_exp:
    movzx eax, byte ptr [rbx + r12]
    or al, 0x20
    cmp al, 'e'
    jne 3f
    inc r12
    movzx eax, byte ptr [rbx + r12]
    cmp al, '+'
    je 4f
    cmp al, '-'
    jne 5f
4:
    inc r12
5:
    call need_digits
3:
    ret

skip_digits:
    movzx eax, byte ptr [rbx + r12]
    sub eax, '0'
    cmp eax, 9
    ja 1f
    inc r12
    jmp skip_digits
1:
    ret

need_digits:
    movzx eax, byte ptr [rbx + r12]
    sub eax, '0'
    cmp eax, 9
    ja error
    inc r12
    jmp skip_digits

parse_array:
    inc r12
    call skip_ws
    cmp byte ptr [rbx + r12], ']'
    jne 1f
    inc r12
    ret
1:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rbx + r12]
    cmp al, ','
    jne 2f
    inc r12
    call skip_ws
    jmp 1b
2:
    cmp al, ']'
    jne error
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp byte ptr [rbx + r12], '}'
    jne 1f
    inc r12
    ret
1:
    cmp byte ptr [rbx + r12], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [rbx + r12], ':'
    jne error
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rbx + r12]
    cmp al, ','
    jne 2f
    inc r12
    call skip_ws
    jmp 1b
2:
    cmp al, '}'
    jne error
    inc r12
    ret
