.text
.globl _start
_start:
    xor r12d, r12d              # lines
    xor r13d, r13d              # words
    xor r14d, r14d              # bytes
    xor r15d, r15d              # in-word flag
read_loop:
    xor eax, eax                # read
    xor edi, edi                # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
proc:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    cmp al, 0x09
    jb nonws
    cmp al, 0x0d
    ja nonws
ws:
    xor r15d, r15d
    jmp next
nonws:
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
next:
    dec rcx
    jnz proc
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call putnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call putnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call putnum
    mov byte ptr [rdi], 0x0a
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1                  # write
    mov edi, 1
    syscall
    mov eax, 231                # exit_group
    xor edi, edi
    syscall

# putnum: rax = value, rdi = output pointer (advanced on return)
putnum:
    push rbx
    sub rsp, 32
    lea rbx, [rsp + 32]
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz 1b
2:
    mov al, byte ptr [rbx]
    mov byte ptr [rdi], al
    inc rdi
    inc rbx
    lea rax, [rsp + 32]
    cmp rbx, rax
    jb 2b
    add rsp, 32
    pop rbx
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
