.section .rodata
hextab: .ascii "0123456789abcdef"

.bss
.balign 16
inbuf: .space 65536+16
outbuf: .space 400000

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    lea rdi, [rip + outbuf]
    lea r8, [rip + hextab]
    lea r13, [rip + inbuf]
row_loop:
    cmp r12, rbx
    jae finish
    mov r14, rbx
    sub r14, r12
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    lea rsi, [r13 + r12]
    mov eax, r12d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor r15d, r15d
col_loop:
    cmp r15, r14
    jae col_pad
    movzx eax, byte ptr [rsi + r15]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp col_next
col_pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
col_next:
    add rdi, 3
    cmp r15, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc r15
    cmp r15, 16
    jb col_loop
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7C
    add rdi, 2
    xor r15d, r15d
chr_loop:
    movzx eax, byte ptr [rsi + r15]
    lea edx, [rax - 0x20]
    cmp edx, 0x5F
    jb 3f
    mov al, 0x2E
3:
    mov byte ptr [rdi], al
    inc rdi
    inc r15
    cmp r15, r14
    jb chr_loop
    mov byte ptr [rdi], 0x7C
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r12, 16
    jmp row_loop
finish:
    mov eax, ebx
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
wr_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp wr_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
    jmp do_exit

# hex8: eax -> 8 lowercase hex digits at [rdi], advances rdi. r8 = hextab.
hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret
