.text
.globl _start
_start:
    xor ebx, ebx                    # total bytes read
read_loop:
    mov eax, 65536
    sub eax, ebx
    jz read_done
    mov edx, eax
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    # parse lines into tabA: entries of (ptr, len)
    lea rsi, [rip + inbuf]
    lea rdi, [rip + tabA]
    xor ecx, ecx                    # pos
    xor edx, edx                    # start of current line
    xor r12d, r12d                  # line count
parse_loop:
    cmp rcx, rbx
    jae parse_end
    cmp byte ptr [rsi + rcx], 10
    jne parse_next
    lea rax, [rsi + rdx]
    mov qword ptr [rdi], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r12
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rdx, rbx
    jae parse_fin
    lea rax, [rsi + rdx]
    mov qword ptr [rdi], rax
    mov rax, rbx
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    inc r12
parse_fin:
    # bottom-up merge sort: r13 = src table, r14 = dst table, r15 = width, r12 = n
    lea r13, [rip + tabA]
    lea r14, [rip + tabB]
    mov r15d, 1
pass_loop:
    cmp r15, r12
    jae sort_done
    xor ebx, ebx                    # i (left index)
chunk_loop:
    cmp rbx, r12
    jae chunk_end
    lea r8, [rbx + r15]
    cmp r8, r12
    cmova r8, r12                   # mid
    lea r9, [r8 + r15]
    cmp r9, r12
    cmova r9, r12                   # right
    mov rax, rbx
    shl rax, 4
    lea r10, [r13 + rax]            # a cur
    mov rdi, rax
    mov rax, r8
    shl rax, 4
    lea rbp, [r13 + rax]            # a end
    lea r11, [r13 + rax]            # b cur
    mov rax, r9
    shl rax, 4
    lea rdx, [r13 + rax]            # b end
    lea r8, [r14 + rdi]             # out ptr
merge_loop:
    cmp r10, rbp
    jae copy_b
    cmp r11, rdx
    jae copy_a
    mov rsi, qword ptr [r10]
    mov rcx, qword ptr [r10 + 8]
    mov rdi, qword ptr [r11]
    mov rax, qword ptr [r11 + 8]
    cmp rcx, rax
    cmova rcx, rax                  # min length
    jrcxz merge_lens
    repe cmpsb
    je merge_lens
    ja take_b
    jmp take_a
merge_lens:
    mov rax, qword ptr [r10 + 8]
    cmp rax, qword ptr [r11 + 8]
    ja take_b
take_a:
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [r8], xmm0
    add r10, 16
    add r8, 16
    jmp merge_loop
take_b:
    movups xmm0, xmmword ptr [r11]
    movups xmmword ptr [r8], xmm0
    add r11, 16
    add r8, 16
    jmp merge_loop
copy_a:
    cmp r10, rbp
    jae chunk_next
    movups xmm0, xmmword ptr [r10]
    movups xmmword ptr [r8], xmm0
    add r10, 16
    add r8, 16
    jmp copy_a
copy_b:
    cmp r11, rdx
    jae chunk_next
    movups xmm0, xmmword ptr [r11]
    movups xmmword ptr [r8], xmm0
    add r11, 16
    add r8, 16
    jmp copy_b
chunk_next:
    mov rbx, r9
    jmp chunk_loop
chunk_end:
    xchg r13, r14
    shl r15, 1
    jmp pass_loop
sort_done:
    # build output in outbuf from table r13
    lea rdi, [rip + outbuf]
    mov rbx, r12
    mov r10, r13
out_loop:
    test rbx, rbx
    jz out_done
    mov rsi, qword ptr [r10]
    mov rcx, qword ptr [r10 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add r10, 16
    dec rbx
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit

.bss
.balign 16
inbuf:
    .space 65536
.balign 16
outbuf:
    .space 98304
.balign 16
tabA:
    .space 65536
.balign 16
tabB:
    .space 65536
