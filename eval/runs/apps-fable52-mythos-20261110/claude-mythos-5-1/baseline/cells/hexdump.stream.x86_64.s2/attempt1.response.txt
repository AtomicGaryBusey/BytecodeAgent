.bss
.balign 16
inbuf:  .space 65536+16
outbuf: .space 400000

.section .rodata
hexdig: .ascii "0123456789abcdef"

.text
.globl _start
_start:
    lea r15, [rip + hexdig]
    xor r12d, r12d              # total input length
1:
    mov edx, 65536
    sub edx, r12d
    jle 2f
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    lea rdi, [rip + outbuf]     # output pointer
    lea rsi, [rip + inbuf]
    xor r13d, r13d              # current offset
row_loop:
    mov rax, r12
    sub rax, r13
    jz finish
    cmp rax, 16
    jbe 3f
    mov eax, 16
3:
    mov r14, rax                # n = bytes in this row
    lea rbx, [rsi + r13]        # row base
    mov eax, r13d
    call put_hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
hex_loop:
    cmp rcx, r14
    jae hex_pad
    movzx eax, byte ptr [rbx + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp hex_next
hex_pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
hex_next:
    cmp ecx, 7
    jne 4f
    mov byte ptr [rdi], 0x20
    inc rdi
4:
    inc rcx
    cmp ecx, 16
    jb hex_loop
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    xor ecx, ecx
asc_loop:
    movzx eax, byte ptr [rbx + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5f
    jb 5f
    mov eax, 0x2e
5:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    cmp rcx, r14
    jb asc_loop
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    add r13, r14
    jmp row_loop
finish:
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
wr_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp wr_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
    jmp do_exit

# eax = value, rdi = out ptr; writes 8 lowercase hex digits, advances rdi
put_hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret
