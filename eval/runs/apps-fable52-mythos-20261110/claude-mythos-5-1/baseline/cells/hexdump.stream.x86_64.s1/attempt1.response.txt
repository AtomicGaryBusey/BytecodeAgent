.text
.globl _start
_start:
    xor ebx, ebx                      # total input length
read_loop:
    mov edx, 65536
    sub edx, ebx
    jbe read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor eax, eax                      # read
    xor edi, edi                      # fd 0
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]            # input pointer
    lea rdi, [rip + outbuf]           # output pointer
    lea r13, [rip + hexdigits]
    xor r12d, r12d                    # current offset

row_loop:
    mov rcx, rbx
    sub rcx, r12                      # remaining bytes
    jz rows_done
    cmp rcx, 16
    jbe 1f
    mov ecx, 16
1:
    mov eax, r12d
    call write_hex8
    mov word ptr [rdi], 0x2020        # two spaces
    add rdi, 2
    xor edx, edx                      # i
hex_loop:
    cmp rdx, rcx
    jae hex_pad
    movzx eax, byte ptr [rsi + rdx]
    mov r8d, eax
    shr r8d, 4
    and eax, 15
    mov r8b, byte ptr [r13 + r8]
    mov al, byte ptr [r13 + rax]
    mov byte ptr [rdi], r8b
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp hex_next
hex_pad:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
hex_next:
    cmp edx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc edx
    cmp edx, 16
    jb hex_loop

    mov word ptr [rdi], 0x7C20        # " |"
    add rdi, 2
    xor edx, edx
asc_loop:
    movzx eax, byte ptr [rsi + rdx]
    lea r8d, [rax - 0x20]
    cmp r8d, 0x5F
    jb 3f
    mov eax, 0x2E                     # '.'
3:
    mov byte ptr [rdi], al
    inc rdi
    inc edx
    cmp rdx, rcx
    jb asc_loop
    mov word ptr [rdi], 0x0A7C        # "|\n"
    add rdi, 2
    add rsi, rcx
    add r12, rcx
    jmp row_loop

rows_done:
    mov eax, ebx
    call write_hex8
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                      # output length
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1                        # write
    mov edi, 1                        # fd 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop

do_exit:
    mov eax, 231                      # exit_group
    xor edi, edi
    syscall
    jmp do_exit

# write_hex8: writes eax as 8 lowercase hex digits at [rdi], advances rdi by 8.
# Clobbers r8, r9; preserves rax (rotated back), rcx, rdx, rsi, rbx.
write_hex8:
    mov r9d, 8
1:
    rol eax, 4
    mov r8d, eax
    and r8d, 15
    mov r8b, byte ptr [r13 + r8]
    mov byte ptr [rdi], r8b
    inc rdi
    dec r9d
    jnz 1b
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 331776
