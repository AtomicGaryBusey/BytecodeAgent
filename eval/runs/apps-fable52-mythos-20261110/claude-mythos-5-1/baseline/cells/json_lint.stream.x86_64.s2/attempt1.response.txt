.text
.globl _start

BUFSIZE = 69632

_start:
    lea rbx, [rip + buf]
    xor r13d, r13d
read_loop:
    mov edx, BUFSIZE - 1
    sub rdx, r13
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rbx + r13]
    mov rsi, rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall

fail:
    mov rax, rsi
    sub rax, rbx
    lea rdi, [rip + msgbuf + 32]
    mov byte ptr [rdi], 10
    mov rcx, rdi
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    mov rax, qword ptr [rip + err_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_prefix + 8]
    mov byte ptr [rdi + 8], al
    lea rdx, [rcx + 1]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    jne 2f
1:
    inc rsi
    jmp skip_ws
2:
    ret

parse_value:
    movzx eax, byte ptr [rsi]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2D
    je parse_number
    lea edx, [rax - 0x30]
    cmp dl, 9
    jbe parse_number
    cmp al, 0x74
    je lit_true
    cmp al, 0x66
    je lit_false
    cmp al, 0x6E
    je lit_null
    jmp fail

lit_true:
    lea rdi, [rip + s_true]
    jmp match_lit
lit_false:
    lea rdi, [rip + s_false]
    jmp match_lit
lit_null:
    lea rdi, [rip + s_null]
match_lit:
    movzx eax, byte ptr [rdi]
    test al, al
    jz 2f
    cmp al, byte ptr [rsi]
    jne fail
    inc rsi
    inc rdi
    jmp match_lit
2:
    ret

parse_string:
    inc rsi
ps_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb fail
    inc rsi
    jmp ps_loop
ps_end:
    inc rsi
    ret
ps_esc:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 0x75
    je ps_u
    cmp al, 0x22
    je ps_simple
    cmp al, 0x5C
    je ps_simple
    cmp al, 0x2F
    je ps_simple
    cmp al, 0x62
    je ps_simple
    cmp al, 0x66
    je ps_simple
    cmp al, 0x6E
    je ps_simple
    cmp al, 0x72
    je ps_simple
    cmp al, 0x74
    je ps_simple
    jmp fail
ps_simple:
    inc rsi
    jmp ps_loop
ps_u:
    inc rsi
    mov ecx, 4
ps_hex:
    movzx eax, byte ptr [rsi]
    lea edx, [rax - 0x30]
    cmp dl, 9
    jbe 1f
    or al, 0x20
    sub al, 0x61
    cmp al, 5
    ja fail
1:
    inc rsi
    dec ecx
    jnz ps_hex
    jmp ps_loop

parse_number:
    cmp byte ptr [rsi], 0x2D
    jne 1f
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    je pn_zero
    sub al, 0x31
    cmp al, 8
    ja fail
    inc rsi
    call skip_digits
    jmp pn_frac
pn_zero:
    inc rsi
pn_frac:
    cmp byte ptr [rsi], 0x2E
    jne pn_exp
    inc rsi
    call need_digit
pn_exp:
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 0x65
    jne pn_done
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 0x2B
    je 2f
    cmp al, 0x2D
    jne 3f
2:
    inc rsi
3:
    call need_digit
pn_done:
    ret

need_digit:
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja fail
    inc rsi
skip_digits:
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja 1f
    inc rsi
    jmp skip_digits
1:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], 0x5D
    je pa_end
pa_loop:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    jne fail
pa_end:
    inc rsi
    ret
pa_comma:
    inc rsi
    call skip_ws
    jmp pa_loop

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], 0x7D
    je po_end
po_loop:
    cmp byte ptr [rsi], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], 0x3A
    jne fail
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    jne fail
po_end:
    inc rsi
    ret
po_comma:
    inc rsi
    call skip_ws
    jmp po_loop

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
s_true:
    .asciz "true"
s_false:
    .asciz "false"
s_null:
    .asciz "null"

.bss
.balign 16
buf:
    .space BUFSIZE
msgbuf:
    .space 64
