.text
.globl _start
_start:
    xor r12d, r12d                 # total bytes read
    lea rbx, [rip + inbuf]
read_loop:
    mov eax, 65552
    sub rax, r12
    jz read_done
    mov rdx, rax
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]       # output write pointer
    xor r13d, r13d                # line start index
    xor r14d, r14d                # current index
scan_loop:
    cmp r14, r12
    jae scan_end
    cmp byte ptr [rbx + r14], 0x0A
    jne scan_next
    call emit_line
    lea r13, [r14 + 1]
scan_next:
    inc r14
    jmp scan_loop
scan_end:
    cmp r13, r12
    jae do_write
    call emit_line
do_write:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done:
    mov eax, 231
    xor edi, edi
    syscall
    mov eax, 60
    syscall
    jmp done

# emit_line: reverse inbuf[r13, r14) into [rdi], append newline; advances rdi
emit_line:
    mov rcx, r14
emit_loop:
    cmp rcx, r13
    jbe emit_nl
    dec rcx
    mov al, byte ptr [rbx + rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp emit_loop
emit_nl:
    mov byte ptr [rdi], 0x0A
    inc rdi
    ret

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
outbuf:
    .space 65600
