.text
.globl _start

_start:
        mov     x19, #0                 // lines
        mov     x20, #0                 // words
        mov     x21, #0                 // bytes
        mov     x22, #0                 // in_word flag

read_loop:
        mov     x8, #63                 // read
        mov     x0, #0
        adrp    x1, buf
        add     x1, x1, :lo12:buf
        mov     x2, #65536
        svc     #0
        cmp     x0, #0
        b.le    done
        add     x21, x21, x0
        mov     x3, x1
        add     x4, x1, x0
scan:
        ldrb    w5, [x3], #1
        cmp     w5, #10
        cinc    x19, x19, eq
        cmp     w5, #0x20
        b.eq    is_ws
        sub     w6, w5, #9
        cmp     w6, #4
        b.ls    is_ws
        cbnz    x22, next
        add     x20, x20, #1
        mov     x22, #1
        b       next
is_ws:
        mov     x22, #0
next:
        cmp     x3, x4
        b.lo    scan
        b       read_loop

done:
        adrp    x10, outbuf
        add     x10, x10, :lo12:outbuf
        add     x10, x10, #64           // end of buffer
        mov     x12, #10
        mov     w13, #10
        strb    w13, [x10, #-1]!        // '\n'
        mov     x9, x21
        bl      put_dec
        mov     w13, #32
        strb    w13, [x10, #-1]!
        mov     x9, x20
        bl      put_dec
        mov     w13, #32
        strb    w13, [x10, #-1]!
        mov     x9, x19
        bl      put_dec

        adrp    x2, outbuf
        add     x2, x2, :lo12:outbuf
        add     x2, x2, #64
        sub     x2, x2, x10             // length
        mov     x1, x10
        mov     x0, #1
        mov     x8, #64                 // write
        svc     #0

        mov     x0, #0
        mov     x8, #94                 // exit_group
        svc     #0
        b       .

// put_dec: x9 = value, x10 = pointer (pre-decrement), x12 = 10
put_dec:
        udiv    x11, x9, x12
        msub    x13, x11, x12, x9
        add     w13, w13, #48
        strb    w13, [x10, #-1]!
        mov     x9, x11
        cbnz    x9, put_dec
        ret

.bss
.balign 16
buf:
        .space  65536
outbuf:
        .space  64
