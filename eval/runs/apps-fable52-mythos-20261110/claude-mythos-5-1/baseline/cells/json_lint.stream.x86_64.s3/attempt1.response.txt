.text
.globl _start
_start:
    lea rbx, [rip + buf]
    xor r12d, r12d
read_loop:
    mov edx, 65536
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov byte ptr [rbx + r12], 0
    add r12, rbx
    mov rsi, rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne fail
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall

fail:
    mov rax, rsi
    sub rax, rbx
    lea rdi, [rip + outbuf + 40]
    mov byte ptr [rdi], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    mov rax, qword ptr [rip + err_msg]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_msg + 8]
    mov byte ptr [rdi + 8], al
    lea rdx, [rip + outbuf + 41]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 0x09
    je 1f
    cmp al, 0x0A
    je 1f
    cmp al, 0x0D
    jne 2f
1:
    inc rsi
    jmp skip_ws
2:
    ret

parse_value:
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe parse_number
    cmp al, 't'
    je 1f
    cmp al, 'f'
    je 2f
    cmp al, 'n'
    je 3f
    jmp fail
1:
    lea rdi, [rip + lit_true]
    jmp parse_literal
2:
    lea rdi, [rip + lit_false]
    jmp parse_literal
3:
    lea rdi, [rip + lit_null]
parse_literal:
    movzx eax, byte ptr [rdi]
    test al, al
    jz 1f
    cmp al, byte ptr [rsi]
    jne fail
    inc rsi
    inc rdi
    jmp parse_literal
1:
    ret

parse_string:
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je 9f
    cmp al, 0x5C
    je 2f
    cmp al, 0x20
    jb fail
    inc rsi
    jmp 1b
2:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, 'u'
    je 3f
    cmp al, 0x22
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, '/'
    je 4f
    cmp al, 'b'
    je 4f
    cmp al, 'f'
    je 4f
    cmp al, 'n'
    je 4f
    cmp al, 'r'
    je 4f
    cmp al, 't'
    je 4f
    jmp fail
4:
    inc rsi
    jmp 1b
3:
    inc rsi
    mov ecx, 4
5:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe 6f
    or al, 0x20
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
6:
    inc rsi
    dec ecx
    jnz 5b
    jmp 1b
9:
    inc rsi
    ret

parse_number:
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne 1f
    inc rsi
    movzx eax, byte ptr [rsi]
1:
    cmp al, '0'
    jb fail
    ja 2f
    inc rsi
    jmp 3f
2:
    cmp al, '9'
    ja fail
    inc rsi
    call skip_digits
3:
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne 4f
    inc rsi
    call need_digits
    movzx eax, byte ptr [rsi]
4:
    or al, 0x20
    cmp al, 'e'
    jne 5f
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 6f
    cmp al, '-'
    jne 7f
6:
    inc rsi
7:
    call need_digits
5:
    ret

skip_digits:
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja 1f
    inc rsi
    jmp skip_digits
1:
    ret

need_digits:
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja fail
    inc rsi
    jmp skip_digits

parse_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], '}'
    je 9f
1:
    cmp byte ptr [rsi], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], ':'
    jne fail
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp 1b
2:
    cmp al, '}'
    jne fail
9:
    inc rsi
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], ']'
    je 9f
1:
    call parse_value
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp 1b
2:
    cmp al, ']'
    jne fail
9:
    inc rsi
    ret

.section .rodata
ok_msg: .ascii "ok\n"
err_msg: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
.balign 16
outbuf: .space 64
buf: .space 65552
