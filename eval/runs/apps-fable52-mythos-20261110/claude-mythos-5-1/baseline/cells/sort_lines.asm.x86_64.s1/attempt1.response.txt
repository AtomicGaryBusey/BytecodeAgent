.bss
    .balign 16
inbuf:  .space 32784
outbuf: .space 34832
lines:  .space 16384
tmp:    .space 16384

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov edx, 32768
    sub edx, r12d
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # parse lines
    lea rsi, [rip + inbuf]
    lea rdi, [rip + lines]
    xor ecx, ecx
    xor r13d, r13d
    mov rbx, rsi
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rsi + rcx], 10
    jne no_nl
    lea rax, [rsi + rcx]
    sub rax, rbx
    mov qword ptr [rdi], rbx
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r13
    lea rbx, [rsi + rcx + 1]
no_nl:
    inc rcx
    jmp parse_loop
parse_end:
    lea rax, [rsi + r12]
    cmp rax, rbx
    je no_final
    sub rax, rbx
    mov qword ptr [rdi], rbx
    mov qword ptr [rdi + 8], rax
    inc r13
no_final:
    # bottom-up merge sort
    mov r14d, 1
sort_outer:
    cmp r14, r13
    jae sort_done
    xor r15d, r15d
sort_inner:
    cmp r15, r13
    jae pass_done
    mov rbx, r15
    lea rbp, [r15 + r14]
    cmp rbp, r13
    cmova rbp, r13
    mov r8, rbp
    lea r9, [rbp + r14]
    cmp r9, r13
    cmova r9, r13
    mov r10, r15
merge_loop:
    cmp rbx, rbp
    jae take_right_rest
    cmp r8, r9
    jae take_left
    mov rax, rbx
    shl rax, 4
    lea rsi, [rip + lines]
    add rsi, rax
    mov rax, r8
    shl rax, 4
    lea rdi, [rip + lines]
    add rdi, rax
    call cmp_lines
    test eax, eax
    jg take_right
take_left:
    mov rax, rbx
    shl rax, 4
    lea rsi, [rip + lines]
    movdqu xmm0, xmmword ptr [rsi + rax]
    mov rax, r10
    shl rax, 4
    lea rdi, [rip + tmp]
    movdqu xmmword ptr [rdi + rax], xmm0
    inc rbx
    inc r10
    jmp merge_loop
take_right:
    mov rax, r8
    shl rax, 4
    lea rsi, [rip + lines]
    movdqu xmm0, xmmword ptr [rsi + rax]
    mov rax, r10
    shl rax, 4
    lea rdi, [rip + tmp]
    movdqu xmmword ptr [rdi + rax], xmm0
    inc r8
    inc r10
    jmp merge_loop
take_right_rest:
    cmp r8, r9
    jae merge_done
    mov rax, r8
    shl rax, 4
    lea rsi, [rip + lines]
    movdqu xmm0, xmmword ptr [rsi + rax]
    mov rax, r10
    shl rax, 4
    lea rdi, [rip + tmp]
    movdqu xmmword ptr [rdi + rax], xmm0
    inc r8
    inc r10
    jmp take_right_rest
merge_done:
    lea r15, [r15 + r14*2]
    jmp sort_inner
pass_done:
    lea rsi, [rip + tmp]
    lea rdi, [rip + lines]
    mov rcx, r13
    shl rcx, 4
    rep movsb
    add r14, r14
    jmp sort_outer
sort_done:
    # build output
    lea rdi, [rip + outbuf]
    lea rbx, [rip + lines]
    xor r15d, r15d
out_loop:
    cmp r15, r13
    jae out_done
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    inc r15
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jle do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 231
    xor edi, edi
    syscall
    jmp do_exit

# cmp_lines: rsi -> entry A, rdi -> entry B
# returns eax < 0 if A < B, 0 if equal, > 0 if A > B
# clobbers rax, rcx, rdx, rsi, rdi, r11
cmp_lines:
    mov rcx, qword ptr [rsi + 8]
    mov rdx, qword ptr [rdi + 8]
    mov r11, rcx
    cmp r11, rdx
    cmova r11, rdx
    sub rcx, rdx
    mov rdx, rcx
    mov rsi, qword ptr [rsi]
    mov rdi, qword ptr [rdi]
    xor eax, eax
1:
    cmp rax, r11
    jae 3f
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, byte ptr [rdi + rax]
    jne 2f
    inc rax
    jmp 1b
2:
    movzx eax, byte ptr [rdi + rax]
    sub ecx, eax
    mov eax, ecx
    ret
3:
    mov rax, rdx
    ret
