.equ BUFSIZE, 69632

.section .rodata
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"
s_err:   .ascii "error at "
s_ok:    .ascii "ok\n"

.bss
.balign 16
buf:    .space BUFSIZE
stk:    .space 2048
numbuf: .space 64

.text
.globl _start
_start:
    lea r13, [rip + buf]
    xor r12d, r12d
.Lread:
    mov edx, BUFSIZE
    sub rdx, r12
    jbe .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread
.Lread_done:
    xor esi, esi
    xor ebx, ebx
    lea r14, [rip + stk]
    call skip_ws

value:
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x7B
    je obj_start
    cmp al, 0x5B
    je arr_start
    cmp al, 0x22
    je str_value
    cmp al, 0x74
    je lit_true
    cmp al, 0x66
    je lit_false
    cmp al, 0x6E
    je lit_null
    cmp al, 0x2D
    je number
    sub al, 0x30
    cmp al, 9
    jbe number
    jmp error

obj_start:
    mov byte ptr [r14 + rbx], 1
    inc rbx
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae error
    cmp byte ptr [r13 + rsi], 0x7D
    jne member_key
    inc rsi
    dec rbx
    jmp after_value

member_key:
    cmp rsi, r12
    jae error
    cmp byte ptr [r13 + rsi], 0x22
    jne error
    inc rsi
    call parse_string
    call skip_ws
    cmp rsi, r12
    jae error
    cmp byte ptr [r13 + rsi], 0x3A
    jne error
    inc rsi
    call skip_ws
    jmp value

arr_start:
    mov byte ptr [r14 + rbx], 0
    inc rbx
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae error
    cmp byte ptr [r13 + rsi], 0x5D
    jne value
    inc rsi
    dec rbx
    jmp after_value

str_value:
    inc rsi
    call parse_string
    jmp after_value

lit_true:
    lea rdi, [rip + s_true]
    jmp literal
lit_false:
    lea rdi, [rip + s_false]
    jmp literal
lit_null:
    lea rdi, [rip + s_null]
literal:
    movzx eax, byte ptr [rdi]
    test al, al
    jz after_value
    cmp rsi, r12
    jae error
    cmp al, byte ptr [r13 + rsi]
    jne error
    inc rsi
    inc rdi
    jmp literal

number:
    cmp byte ptr [r13 + rsi], 0x2D
    jne 1f
    inc rsi
1:
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x30
    jne 2f
    inc rsi
    jmp num_frac
2:
    sub al, 0x31
    cmp al, 8
    ja error
    inc rsi
    call skip_digits
num_frac:
    cmp rsi, r12
    jae after_value
    cmp byte ptr [r13 + rsi], 0x2E
    jne num_exp
    inc rsi
    call need_digits
num_exp:
    cmp rsi, r12
    jae after_value
    movzx eax, byte ptr [r13 + rsi]
    or al, 0x20
    cmp al, 0x65
    jne after_value
    inc rsi
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x2B
    je 3f
    cmp al, 0x2D
    jne 4f
3:
    inc rsi
4:
    call need_digits
    jmp after_value

need_digits:
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    sub al, 0x30
    cmp al, 9
    ja error
    inc rsi
skip_digits:
    cmp rsi, r12
    jae 1f
    movzx eax, byte ptr [r13 + rsi]
    sub al, 0x30
    cmp al, 9
    ja 1f
    inc rsi
    jmp skip_digits
1:
    ret

after_value:
    call skip_ws
    test rbx, rbx
    jnz 1f
    cmp rsi, r12
    jne error
    jmp ok
1:
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x2C
    je comma
    cmp byte ptr [r14 + rbx - 1], 0
    jne 2f
    cmp al, 0x5D
    jne error
    jmp close
2:
    cmp al, 0x7D
    jne error
close:
    inc rsi
    dec rbx
    jmp after_value
comma:
    inc rsi
    call skip_ws
    cmp byte ptr [r14 + rbx - 1], 0
    jne member_key
    jmp value

parse_string:
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x22
    je 9f
    cmp al, 0x5C
    je 2f
    cmp al, 0x20
    jb error
    inc rsi
    jmp parse_string
9:
    inc rsi
    ret
2:
    inc rsi
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x75
    je 3f
    cmp al, 0x22
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, 0x2F
    je 4f
    cmp al, 0x62
    je 4f
    cmp al, 0x66
    je 4f
    cmp al, 0x6E
    je 4f
    cmp al, 0x72
    je 4f
    cmp al, 0x74
    je 4f
    jmp error
4:
    inc rsi
    jmp parse_string
3:
    inc rsi
    mov ecx, 4
5:
    cmp rsi, r12
    jae error
    movzx eax, byte ptr [r13 + rsi]
    mov edx, eax
    sub dl, 0x30
    cmp dl, 9
    jbe 6f
    or al, 0x20
    sub al, 0x61
    cmp al, 5
    ja error
6:
    inc rsi
    dec ecx
    jnz 5b
    jmp parse_string

skip_ws:
    cmp rsi, r12
    jae 1f
    movzx eax, byte ptr [r13 + rsi]
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    je 2f
    cmp al, 0x0A
    je 2f
    cmp al, 0x0D
    jne 1f
2:
    inc rsi
    jmp skip_ws
1:
    ret

ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_ok]
    mov edx, 3
    syscall
    mov eax, 231
    xor edi, edi
    syscall

error:
    mov rax, rsi
    lea rdi, [rip + numbuf + 47]
    mov byte ptr [rdi], 0x0A
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    mov rax, qword ptr [rip + s_err]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + s_err + 8]
    mov byte ptr [rdi + 8], al
    lea rdx, [rip + numbuf + 48]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 231
    mov edi, 1
    syscall
