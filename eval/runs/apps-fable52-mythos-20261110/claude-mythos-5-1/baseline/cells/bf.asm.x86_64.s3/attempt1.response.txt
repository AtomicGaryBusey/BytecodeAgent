.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub edx, ebx
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
find_loop:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r15, rcx
    lea r12, [rsi + rcx + 1]
    jmp have_split
no_bang:
    mov r15, rbx
    lea r12, [rsi + rbx]
have_split:
    lea r13, [rsi + rbx]

    # bracket check and jump table
    lea rdi, [rip + jump]
    lea rdx, [rip + bstack]
    xor ecx, ecx
    xor r8d, r8d
bc_loop:
    cmp rcx, r15
    jae bc_end
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je bc_open
    cmp al, 0x5d
    je bc_close
bc_next:
    inc rcx
    jmp bc_loop
bc_open:
    mov dword ptr [rdx + r8*4], ecx
    inc r8
    jmp bc_next
bc_close:
    test r8, r8
    jz bad
    dec r8
    mov eax, dword ptr [rdx + r8*4]
    mov dword ptr [rdi + rax*4], ecx
    mov dword ptr [rdi + rcx*4], eax
    jmp bc_next
bc_end:
    test r8, r8
    jnz bad

    # execute
    lea rdi, [rip + tape]
    lea r8, [rip + jump]
    lea r9, [rip + outbuf]
    xor r14d, r14d
    xor ecx, ecx
run_loop:
    cmp rcx, r15
    jae run_end
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x3e
    je op_right
    cmp al, 0x3c
    je op_left
    cmp al, 0x2b
    je op_inc
    cmp al, 0x2d
    je op_dec
    cmp al, 0x5b
    je op_open
    cmp al, 0x5d
    je op_close
    cmp al, 0x2e
    je op_out
    cmp al, 0x2c
    je op_in
run_next:
    inc rcx
    jmp run_loop
op_right:
    inc rdi
    jmp run_next
op_left:
    dec rdi
    jmp run_next
op_inc:
    inc byte ptr [rdi]
    jmp run_next
op_dec:
    dec byte ptr [rdi]
    jmp run_next
op_open:
    cmp byte ptr [rdi], 0
    jne run_next
    mov ecx, dword ptr [r8 + rcx*4]
    jmp run_next
op_close:
    cmp byte ptr [rdi], 0
    je run_next
    mov ecx, dword ptr [r8 + rcx*4]
    jmp run_next
op_out:
    mov al, byte ptr [rdi]
    mov byte ptr [r9 + r14], al
    inc r14
    cmp r14, 65536
    jb run_next
    call flush
    jmp run_next
op_in:
    xor eax, eax
    cmp r12, r13
    jae 1f
    mov al, byte ptr [r12]
    inc r12
1:
    mov byte ptr [rdi], al
    jmp run_next
run_end:
    call flush
    mov eax, 231
    xor edi, edi
    syscall
    jmp run_end

bad:
    mov eax, 231
    mov edi, 1
    syscall
    jmp bad

flush:
    push rcx
    push rdi
    push rsi
    push rdx
    push r11
    push r10
    xor r10d, r10d
1:
    cmp r10, r14
    jae 2f
    mov eax, 1
    mov edi, 1
    lea rsi, [r9 + r10]
    mov rdx, r14
    sub rdx, r10
    syscall
    test rax, rax
    jle 2f
    add r10, rax
    jmp 1b
2:
    xor r14d, r14d
    pop r10
    pop r11
    pop rdx
    pop rsi
    pop rdi
    pop rcx
    ret

.bss
.balign 16
inbuf:
    .space 65552
.balign 16
tape:
    .space 30016
.balign 16
jump:
    .space 262144
.balign 16
bstack:
    .space 262144
.balign 16
outbuf:
    .space 65552
