.text
.globl _start

_start:
        // ---- read all of stdin into inbuf ----
        adrp    x19, inbuf
        add     x19, x19, :lo12:inbuf
        mov     x20, x19                // fill pointer
        mov     x26, #0x10000
        add     x26, x19, x26           // buffer end
rd_loop:
        sub     x2, x26, x20
        cbz     x2, rd_done
        mov     x0, #0
        mov     x1, x20
        mov     x8, #63
        svc     #0
        cmp     x0, #0
        b.le    rd_done
        add     x20, x20, x0
        b       rd_loop
rd_done:
        // x19 = cursor, x20 = end of input
        adrp    x24, outbuf
        add     x24, x24, :lo12:outbuf
        mov     x21, x24                // output write pointer
        adrp    x22, vstack
        add     x22, x22, :lo12:vstack

line_loop:
        cmp     x19, x20
        b.hs    finish
        // make sure there is room for one more result line
        sub     x0, x21, x24
        movz    x1, #0xFFC0
        cmp     x0, x1
        b.lo    1f
        bl      flush
1:
        mov     x23, #0                 // stack count

token_loop:
        // skip separators
2:
        cmp     x19, x20
        b.hs    end_of_line
        ldrb    w0, [x19]
        cmp     w0, #0x20
        b.eq    3f
        cmp     w0, #0x09
        b.ne    4f
3:
        add     x19, x19, #1
        b       2b
4:
        cmp     w0, #0x0a
        b.eq    end_of_line
        // token start at x5
        mov     x5, x19
5:
        add     x19, x19, #1
        cmp     x19, x20
        b.hs    6f
        ldrb    w0, [x19]
        cmp     w0, #0x20
        b.eq    6f
        cmp     w0, #0x09
        b.eq    6f
        cmp     w0, #0x0a
        b.ne    5b
6:
        sub     x6, x19, x5             // token length
        ldrb    w0, [x5]                // first byte
        cmp     x6, #1
        b.ne    literal
        cmp     w0, #'+'
        b.eq    operator
        cmp     w0, #'-'
        b.eq    operator
        cmp     w0, #'*'
        b.eq    operator
        cmp     w0, #'/'
        b.eq    operator
        cmp     w0, #'%'
        b.eq    operator

literal:
        mov     x7, #0                  // negative flag
        cmp     w0, #'-'
        b.ne    7f
        mov     x7, #1
        add     x5, x5, #1
        sub     x6, x6, #1
7:
        cbz     x6, line_error
        cmp     x6, #18
        b.hi    line_error
        mov     x9, #0
        mov     x1, #10
8:
        ldrb    w0, [x5], #1
        sub     w0, w0, #'0'
        cmp     w0, #9
        b.hi    line_error
        madd    x9, x9, x1, x0
        subs    x6, x6, #1
        b.ne    8b
        cbz     x7, push_result
        neg     x9, x9
        b       push_result

operator:
        cmp     x23, #2
        b.lo    line_error
        sub     x23, x23, #2
        ldr     x1, [x22, x23, lsl #3]  // a
        add     x3, x23, #1
        ldr     x2, [x22, x3, lsl #3]   // b
        cmp     w0, #'+'
        b.ne    1f
        add     x9, x1, x2
        b       push_result
1:
        cmp     w0, #'-'
        b.ne    1f
        sub     x9, x1, x2
        b       push_result
1:
        cmp     w0, #'*'
        b.ne    1f
        mul     x9, x1, x2
        b       push_result
1:
        // '/' or '%'
        cbz     x2, line_error
        cmn     x2, #1
        b.ne    2f
        movz    x3, #0x8000, lsl #48
        cmp     x1, x3
        b.eq    line_error
2:
        sdiv    x9, x1, x2
        cmp     w0, #'/'
        b.eq    push_result
        msub    x9, x9, x2, x1          // a - (a/b)*b

push_result:
        str     x9, [x22, x23, lsl #3]
        add     x23, x23, #1
        b       token_loop

end_of_line:
        cmp     x23, #1
        b.ne    emit_error
        ldr     x0, [x22]
        bl      write_num
        b       consume_newline

line_error:
        // skip rest of line
1:
        cmp     x19, x20
        b.hs    emit_error
        ldrb    w0, [x19]
        cmp     w0, #0x0a
        b.eq    emit_error
        add     x19, x19, #1
        b       1b

emit_error:
        movz    w0, #0x7265             // "er"
        movk    w0, #0x6f72, lsl #16    // "ro"
        str     w0, [x21]
        movz    w0, #0x0a72             // "r\n"
        strh    w0, [x21, #4]
        add     x21, x21, #6

consume_newline:
        cmp     x19, x20
        b.hs    line_loop
        ldrb    w0, [x19]
        cmp     w0, #0x0a
        b.ne    line_loop
        add     x19, x19, #1
        b       line_loop

finish:
        bl      flush
        mov     x0, #0
        mov     x8, #93
        svc     #0
        b       finish

// write_num: x0 = signed value; appends decimal + '\n' at x21
write_num:
        sub     sp, sp, #32
        add     x4, sp, #32             // end of temp area
        mov     x5, x4
        mov     x3, x0
        cmp     x0, #0
        b.ge    1f
        neg     x3, x0
1:
        mov     x6, #10
2:
        udiv    x7, x3, x6
        msub    x8, x7, x6, x3
        add     w8, w8, #'0'
        strb    w8, [x5, #-1]!
        mov     x3, x7
        cbnz    x3, 2b
        tbz     x0, #63, 3f
        mov     w8, #'-'
        strb    w8, [x5, #-1]!
3:
        ldrb    w8, [x5], #1
        strb    w8, [x21], #1
        cmp     x5, x4
        b.lo    3b
        mov     w8, #0x0a
        strb    w8, [x21], #1
        add     sp, sp, #32
        ret

// flush: write outbuf [x24, x21) to stdout, reset x21
flush:
        stp     x29, x30, [sp, #-16]!
        sub     x2, x21, x24
        cbz     x2, 2f
        mov     x25, x24
1:
        mov     x0, #1
        mov     x1, x25
        mov     x8, #64
        svc     #0
        cmp     x0, #0
        b.le    2f
        add     x25, x25, x0
        subs    x2, x2, x0
        b.gt    1b
2:
        mov     x21, x24
        ldp     x29, x30, [sp], #16
        ret

.bss
.balign 16
inbuf:
        .space  65536
.balign 16
outbuf:
        .space  65536
.balign 16
vstack:
        .space  8192
