.globl _start
.text
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in-word flag
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je is_ws
    cmp al, 0x0A
    je is_nl
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    jbe is_ws
is_word:
    test r15d, r15d
    jnz next
    mov r15d, 1
    inc r13
    jmp next
is_nl:
    inc r12
is_ws:
    xor r15d, r15d
next:
    dec rcx
    jnz byte_loop
    jmp read_loop

done:
    # build output backwards from end of outbuf
    lea rdi, [rip + outbuf + 96]
    mov r8, rdi             # end pointer
    dec rdi
    mov byte ptr [rdi], 0x0A
    mov rax, r14
    call conv
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call conv
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call conv

    mov r12, rdi            # current pointer
    mov r13, r8
    sub r13, rdi            # remaining
write_loop:
    test r13, r13
    jle finish
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle finish
    add r12, rax
    sub r13, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# conv: write decimal of rax backwards ending just before rdi; updates rdi
conv:
    mov ebx, 10
1:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 96
