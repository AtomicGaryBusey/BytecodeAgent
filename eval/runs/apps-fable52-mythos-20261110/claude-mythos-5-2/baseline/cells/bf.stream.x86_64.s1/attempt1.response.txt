.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65552
    sub rdx, r13
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor ecx, ecx
find_bang:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov r14, r13
    mov r15, r13
    jmp check
found_bang:
    mov r14, rcx
    lea r15, [rcx + 1]
check:
    lea r10, [rip + jumps]
    mov rbp, rsp
    xor ecx, ecx
chk_loop:
    cmp rcx, r14
    jae chk_end
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp chk_next
1:
    cmp al, 0x5D
    jne chk_next
    cmp rsp, rbp
    je fail
    pop rdx
    mov dword ptr [r10 + rdx*4], ecx
    mov dword ptr [r10 + rcx*4], edx
chk_next:
    inc rcx
    jmp chk_loop
chk_end:
    cmp rsp, rbp
    jne fail

    lea rbx, [rip + tape]
    lea r9, [rip + outbuf]
    xor r8d, r8d
    xor ebp, ebp
run:
    cmp rbp, r14
    jae finish
    movzx eax, byte ptr [r12 + rbp]
    inc rbp
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp run
op_right:
    inc rbx
    jmp run
op_left:
    dec rbx
    jmp run
op_inc:
    inc byte ptr [rbx]
    jmp run
op_dec:
    dec byte ptr [rbx]
    jmp run
op_open:
    cmp byte ptr [rbx], 0
    jne run
    mov ebp, dword ptr [r10 + rbp*4 - 4]
    inc rbp
    jmp run
op_close:
    cmp byte ptr [rbx], 0
    je run
    mov ebp, dword ptr [r10 + rbp*4 - 4]
    inc rbp
    jmp run
op_out:
    mov al, byte ptr [rbx]
    mov byte ptr [r9 + r8], al
    inc r8
    cmp r8, 65536
    jne run
    call flush
    jmp run
op_in:
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
    mov byte ptr [rbx], al
    jmp run
1:
    mov byte ptr [rbx], 0
    jmp run

finish:
    call flush
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov rsp, rbp
    mov edi, 1
    mov eax, 60
    syscall

flush:
    push rbx
    xor ebx, ebx
1:
    cmp rbx, r8
    jae 2f
    mov eax, 1
    mov edi, 1
    lea rsi, [r9 + rbx]
    mov rdx, r8
    sub rdx, rbx
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    pop rbx
    xor r8d, r8d
    ret

.bss
.align 16
inbuf:
    .zero 65568
.align 16
jumps:
    .zero 262208
.align 16
tape:
    .zero 30016
.align 16
outbuf:
    .zero 65552
