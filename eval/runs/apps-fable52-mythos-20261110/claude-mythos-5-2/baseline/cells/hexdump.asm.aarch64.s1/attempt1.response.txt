.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 131072
outbuf:
    .space 4096

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x23, x22
    adrp x24, hextab
    add x24, x24, :lo12:hextab
    mov x20, #0
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

row_loop:
    cmp x21, x20
    b.hs finish
    sub x25, x20, x21
    cmp x25, #16
    mov x9, #16
    csel x25, x25, x9, lo
    mov w0, w21
    bl puthex8
    mov w10, #' '
    strb w10, [x23], #1
    strb w10, [x23], #1
    add x11, x19, x21        // row pointer
    mov x12, #0
hex_loop:
    cmp x12, x25
    b.hs 1f
    ldrb w13, [x11, x12]
    lsr w14, w13, #4
    ldrb w14, [x24, x14]
    strb w14, [x23], #1
    and w14, w13, #15
    ldrb w14, [x24, x14]
    strb w14, [x23], #1
    strb w10, [x23], #1
    b 2f
1:
    strb w10, [x23], #1
    strb w10, [x23], #1
    strb w10, [x23], #1
2:
    cmp x12, #7
    b.ne 3f
    strb w10, [x23], #1
3:
    add x12, x12, #1
    cmp x12, #16
    b.lo hex_loop

    strb w10, [x23], #1
    mov w15, #'|'
    strb w15, [x23], #1
    mov x12, #0
chr_loop:
    ldrb w13, [x11, x12]
    sub w14, w13, #0x20
    cmp w14, #0x5e
    mov w16, #'.'
    csel w13, w13, w16, ls
    strb w13, [x23], #1
    add x12, x12, #1
    cmp x12, x25
    b.lo chr_loop
    strb w15, [x23], #1
    mov w13, #10
    strb w13, [x23], #1

    add x21, x21, x25
    sub x9, x23, x22
    cmp x9, #3900
    b.lo row_loop
    bl flush
    b row_loop

finish:
    mov w0, w20
    bl puthex8
    mov w13, #10
    strb w13, [x23], #1
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0

puthex8:
    mov w1, #28
1:
    lsr w2, w0, w1
    and x2, x2, #15
    ldrb w2, [x24, x2]
    strb w2, [x23], #1
    subs w1, w1, #4
    b.ge 1b
    ret

flush:
    subs x2, x23, x22
    b.eq 1f
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0
    mov x23, x22
1:
    ret
