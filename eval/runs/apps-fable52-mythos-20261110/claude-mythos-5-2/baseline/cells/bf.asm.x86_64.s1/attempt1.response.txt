.globl _start
.equ INSIZE, 65600
.equ OUTSIZE, 65536

.bss
.align 16
inbuf: .space 65600
.align 16
outbuf: .space 65536
.align 16
jmptab: .space 262400
.align 16
tape: .space 30016

.text
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
rd_loop:
    mov edx, INSIZE
    sub rdx, rbx
    jle rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd_loop
rd_done:
    # find first '!'
    xor ecx, ecx
find_loop:
    cmp rcx, rbx
    jae not_found
    cmp byte ptr [r12 + rcx], 0x21
    je found
    inc rcx
    jmp find_loop
found:
    mov r13, rcx
    lea r14, [rcx + 1]
    jmp chk
not_found:
    mov r13, rbx
    mov r14, rbx
chk:
    lea r10, [rip + jmptab]
    mov rbp, rsp
    xor ecx, ecx
chk_loop:
    cmp rcx, r13
    jae chk_end
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp chk_next
1:
    cmp al, 0x5D
    jne chk_next
    cmp rsp, rbp
    je bad
    pop rdx
    mov dword ptr [r10 + rdx*4], ecx
    mov dword ptr [r10 + rcx*4], edx
chk_next:
    inc rcx
    jmp chk_loop
chk_end:
    cmp rsp, rbp
    jne bad

    lea r15, [rip + tape]
    lea r9, [rip + outbuf]
    xor r8d, r8d
    xor ebp, ebp
run:
    cmp rbp, r13
    jae finish
    movzx eax, byte ptr [r12 + rbp]
    inc rbp
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp run
op_right:
    inc r15
    jmp run
op_left:
    dec r15
    jmp run
op_inc:
    inc byte ptr [r15]
    jmp run
op_dec:
    dec byte ptr [r15]
    jmp run
op_open:
    cmp byte ptr [r15], 0
    jne run
    mov ebp, dword ptr [r10 + rbp*4 - 4]
    inc rbp
    jmp run
op_close:
    cmp byte ptr [r15], 0
    je run
    mov ebp, dword ptr [r10 + rbp*4 - 4]
    inc rbp
    jmp run
op_out:
    mov al, byte ptr [r15]
    mov byte ptr [r9 + r8], al
    inc r8
    cmp r8, OUTSIZE
    jb run
    call flush
    jmp run
op_in:
    cmp r14, rbx
    jae 1f
    mov al, byte ptr [r12 + r14]
    inc r14
    mov byte ptr [r15], al
    jmp run
1:
    mov byte ptr [r15], 0
    jmp run

flush:
    test r8, r8
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    syscall
    xor r8d, r8d
1:
    ret

finish:
    call flush
    xor edi, edi
    mov eax, 60
    syscall
bad:
    mov edi, 1
    mov eax, 60
    syscall
