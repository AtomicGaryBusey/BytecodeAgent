.globl _start

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 96
outend:

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws
    cbnz w22, next
    mov w22, #1
    add x20, x20, #1
    b next
is_ws:
    mov w22, #0
next:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

done:
    adrp x1, outend
    add x1, x1, :lo12:outend
    mov x9, x1               // end pointer
    mov w7, #10
    strb w7, [x1, #-1]!
    mov x0, x21
    bl conv
    mov w7, #32
    strb w7, [x1, #-1]!
    mov x0, x20
    bl conv
    mov w7, #32
    strb w7, [x1, #-1]!
    mov x0, x19
    bl conv
    mov x10, x1              // current pointer
write_loop:
    sub x2, x9, x10
    cbz x2, finish
    mov x0, #1
    mov x1, x10
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x10, x10, x0
    b write_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0
hang:
    b hang

// convert x0 to decimal, writing digits backward before x1
conv:
    mov x5, #10
1:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    strb w7, [x1, #-1]!
    mov x0, x6
    cbnz x0, 1b
    ret
