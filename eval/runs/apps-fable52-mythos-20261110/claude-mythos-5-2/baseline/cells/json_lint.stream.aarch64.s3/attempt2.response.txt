.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
.balign 16
buf:
    .space 65552
nest:
    .space 4096
outbuf:
    .space 32

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    adrp x22, nest
    add x22, x22, :lo12:nest
    mov x20, #0
    mov x21, #0
    mov x23, #0
    mov x26, #0x10000
read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x26, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

parse_value:
    bl skipws
    cmp w0, #123
    b.eq v_object
    cmp w0, #91
    b.eq v_array
    cmp w0, #34
    b.eq v_string
    cmp w0, #45
    b.eq v_number
    sub w1, w0, #48
    cmp w1, #9
    b.ls v_number
    cmp w0, #116
    b.eq v_true
    cmp w0, #102
    b.eq v_false
    cmp w0, #110
    b.eq v_null
    b error

v_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    b literal
v_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    b literal
v_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
literal:
    ldrb w2, [x1], #1
    cbz w2, after_value
    bl peek
    cmp w0, w2
    b.ne error
    add x21, x21, #1
    b literal

v_object:
    add x21, x21, #1
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skipws
    cmp w0, #125
    b.ne obj_member
    add x21, x21, #1
    sub x23, x23, #1
    b after_value

v_array:
    add x21, x21, #1
    mov w1, #0
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skipws
    cmp w0, #93
    b.ne parse_value
    add x21, x21, #1
    sub x23, x23, #1
    b after_value

obj_member:
    bl skipws
    cmp w0, #34
    b.ne error
    mov x24, #1
    b string

key_done:
    bl skipws
    cmp w0, #58
    b.ne error
    add x21, x21, #1
    b parse_value

v_string:
    mov x24, #0
string:
    add x21, x21, #1
str_loop:
    bl peek
    cmp w0, #32
    b.lt error
    cmp w0, #34
    b.eq str_end
    cmp w0, #92
    b.eq str_esc
    add x21, x21, #1
    b str_loop
str_end:
    add x21, x21, #1
    cbnz x24, key_done
    b after_value
str_esc:
    add x21, x21, #1
    bl peek
    cmp w0, #34
    b.eq esc_ok
    cmp w0, #92
    b.eq esc_ok
    cmp w0, #47
    b.eq esc_ok
    cmp w0, #98
    b.eq esc_ok
    cmp w0, #102
    b.eq esc_ok
    cmp w0, #110
    b.eq esc_ok
    cmp w0, #114
    b.eq esc_ok
    cmp w0, #116
    b.eq esc_ok
    cmp w0, #117
    b.ne error
    add x21, x21, #1
    mov x25, #4
hex_loop:
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls hex_ok
    orr w1, w0, #0x20
    sub w1, w1, #97
    cmp w1, #5
    b.hi error
hex_ok:
    add x21, x21, #1
    subs x25, x25, #1
    b.ne hex_loop
    b str_loop
esc_ok:
    add x21, x21, #1
    b str_loop

v_number:
    cmp w0, #45
    b.ne num_int
    add x21, x21, #1
num_int:
    bl peek
    cmp w0, #48
    b.eq num_zero
    sub w1, w0, #49
    cmp w1, #8
    b.hi error
    add x21, x21, #1
num_int_loop:
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.hi num_frac
    add x21, x21, #1
    b num_int_loop
num_zero:
    add x21, x21, #1
    bl peek
num_frac:
    cmp w0, #46
    b.ne num_exp
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
num_frac_loop:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls num_frac_loop
num_exp:
    cmp w0, #101
    b.eq num_exp2
    cmp w0, #69
    b.ne after_value
num_exp2:
    add x21, x21, #1
    bl peek
    cmp w0, #43
    b.eq num_sign
    cmp w0, #45
    b.ne num_exp_d
num_sign:
    add x21, x21, #1
    bl peek
num_exp_d:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
num_exp_loop:
    add x21, x21, #1
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls num_exp_loop
    b after_value

after_value:
    bl skipws
    cbnz x23, av_nested
    cmp x21, x20
    b.ne error
    b success
av_nested:
    sub x1, x23, #1
    ldrb w2, [x22, x1]
    cmp w0, #44
    b.ne av_close
    add x21, x21, #1
    cbnz w2, obj_member
    b parse_value
av_close:
    cbnz w2, av_obj
    cmp w0, #93
    b.ne error
    b av_pop
av_obj:
    cmp w0, #125
    b.ne error
av_pop:
    add x21, x21, #1
    sub x23, x23, #1
    b after_value

success:
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x5, outbuf
    add x5, x5, :lo12:outbuf
    add x5, x5, #32
    mov x1, x5
    mov w4, #10
    strb w4, [x1, #-1]!
    mov x10, #10
err_digits:
    udiv x3, x21, x10
    msub x4, x3, x10, x21
    add w4, w4, #48
    strb w4, [x1, #-1]!
    mov x21, x3
    cbnz x21, err_digits
    sub x2, x5, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// peek: w0 = byte at pos, or -1 at end of input
peek:
    mov w0, #-1
    cmp x21, x20
    b.hs 1f
    ldrb w0, [x19, x21]
1:
    ret

// skipws: skip whitespace, then w0 = byte at pos, or -1 at end of input
skipws:
1:
    mov w0, #-1
    cmp x21, x20
    b.hs 2f
    ldrb w0, [x19, x21]
    cmp w0, #32
    b.eq 3f
    cmp w0, #9
    b.eq 3f
    cmp w0, #10
    b.eq 3f
    cmp w0, #13
    b.ne 2f
3:
    add x21, x21, #1
    b 1b
2:
    ret
