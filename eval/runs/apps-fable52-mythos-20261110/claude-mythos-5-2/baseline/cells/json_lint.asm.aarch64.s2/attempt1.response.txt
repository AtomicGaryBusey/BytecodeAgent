parse_value:
    bl skipws
    bl peek
    cmp w0, #'{'
    b.eq v_obj
    cmp w0, #'['
    b.eq v_arr
    cmp w0, #'"'
    b.eq v_str
    cmp w0, #'-'

    sub w1, w0, #'0'
    cmp w1, #9
    b.ls number
    adrp x1, s_true
    add x1, x1, :lo12:s_true
    cmp w0, #'t'
    b.eq literal
    adrp x1, s_false
    add x1, x1, :lo12:s_false
    cmp w0, #'f'
    b.eq literal
    adrp x1, s_null
    add x1, x1, :lo12:s_null
    cmp w0, #'n'

    b err

    mov w1, #1
    strb w1, [x23, x21]
    add x21, x21, #1

    bl peek

    bl peek

    bl peek

    bl peek
    cmp w0, #':'

    cbnz x21, 1f

1:

    sub x2, x21, #1

    add x19, x19, #1

    mov w4, #'}'

    add x19, x19, #1
    sub x21, x21, #1
    b after_value

// string: x19 at opening quote

    cmp w0, #255

    cmp w0, #0x20
    b.lo err
    cmp w0, #'"'

    cmp w0, #'\\'

    b str_loop

    cbnz x24, after_key
    b after_value

    bl peek
    cmp w0, #'u'
    b.eq str_u
    cmp w0, #'"'
    b.eq 1f
    cmp w0, #'\\'

    cmp w0, #'/'
    b.eq 1f
    cmp w0, #'b'
    b.eq 1f
    cmp w0, #'f'
    b.eq 1f
    cmp w0, #'n'
    b.eq 1f
    cmp w0, #'r'
