.globl _start
.equ INSZ, 65536

.bss
.align 16
inbuf: .space 65536
outbuf: .space 65544
table: .space 528000

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, INSZ
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r15, [rip + table]
    xor r14d, r14d
    xor ebx, ebx
parse:
    cmp rbx, r13
    jae parsed
    mov rcx, rbx
1:
    cmp rcx, r13
    jae 2f
    cmp byte ptr [r12 + rcx], 10
    je 2f
    inc rcx
    jmp 1b
2:
    mov rax, r14
    shl rax, 4
    lea rdx, [r12 + rbx]
    mov qword ptr [r15 + rax], rdx
    mov rdx, rcx
    sub rdx, rbx
    mov qword ptr [r15 + rax + 8], rdx
    inc r14
    lea rbx, [rcx + 1]
    jmp parse
parsed:
    mov r8d, 1
outer:
    cmp r8, r14
    jae sorted
    mov rax, r8
    shl rax, 4
    mov r9, qword ptr [r15 + rax]
    mov r10, qword ptr [r15 + rax + 8]
    mov r11, r8
inner:
    test r11, r11
    jz place
    mov rax, r11
    shl rax, 4
    mov rsi, qword ptr [r15 + rax - 16]
    mov rdx, qword ptr [r15 + rax - 8]
    mov rbx, rsi
    mov rcx, rdx
    cmp rcx, r10
    cmova rcx, r10
    test rcx, rcx
    jz 1f
    mov rdi, r9
    repe cmpsb
    jne 2f
1:
    cmp rdx, r10
2:
    jbe place
    mov qword ptr [r15 + rax], rbx
    mov qword ptr [r15 + rax + 8], rdx
    dec r11
    jmp inner
place:
    mov rax, r11
    shl rax, 4
    mov qword ptr [r15 + rax], r9
    mov qword ptr [r15 + rax + 8], r10
    inc r8
    jmp outer
sorted:
    lea rdi, [rip + outbuf]
    xor r8d, r8d
emit:
    cmp r8, r14
    jae emitted
    mov rax, r8
    shl rax, 4
    mov rsi, qword ptr [r15 + rax]
    mov rcx, qword ptr [r15 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r8
    jmp emit
emitted:
    lea rbx, [rip + outbuf]
    mov r13, rdi
    sub r13, rbx
write_loop:
    test r13, r13
    jz finish
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r13
    syscall
    test rax, rax
    jle finish
    add rbx, rax
    sub r13, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
