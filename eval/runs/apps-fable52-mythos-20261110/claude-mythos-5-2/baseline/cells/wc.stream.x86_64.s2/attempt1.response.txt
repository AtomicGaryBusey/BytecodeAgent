.globl _start
.bss
buf: .zero 65536
outbuf: .zero 128
.text
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in-word flag
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je ws
    cmp al, 0x0A
    je nl
    cmp al, 0x09
    jb word
    cmp al,
