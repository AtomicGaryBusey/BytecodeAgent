.globl _start
.bss
buf: .zero 65536
outbuf: .zero 128
.text
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in-word flag
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je is_ws
    cmp al, 0x0A
    je is_nl
    cmp al, 0x09
    jb is_wordbyte
    cmp al, 0x0D
    jbe is_ws
is_wordbyte:
    test r15d, r15d
    jnz next_byte
    mov r15d, 1
    inc r13
    jmp next_byte
is_nl:
    inc r12
is_ws:
    xor r15d, r15d
next_byte:
    dec rcx
    jnz byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call put_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call put_dec
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, rdi = dest; advances rdi
put_dec:
    mov rbx, 10
    xor r8d, r8d
1:
    xor edx, edx
    div rbx
    add dl, 0x30
    push rdx
    inc r8
    test rax, rax
    jnz 1b
2:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec r8
    jnz 2b
    ret
