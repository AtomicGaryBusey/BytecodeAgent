.globl _start
.equ BUFSZ, 65544

.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
.align 16
buf: .zero 65552
stk: .zero 4096
outbuf: .zero 32

.text
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
rd_loop:
    mov edx, BUFSZ
    sub rdx, r13
    jz rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd_loop
rd_done:
    mov rsi, r12
    lea rdi, [r12 + r13]
    xor ebx, ebx
    lea r14, [rip + stk]

parse_value:
    call skipws
    cmp al, '{'
    je v_obj
    cmp al, '['
    je v_arr
    cmp al, '"'
    je v_str
    cmp al, '-'
    je v_num
    cmp al, 't'
    je v_true
    cmp al, 'f'
    je v_false
    cmp al, 'n'
    je v_null
    mov dl, al
    sub dl, '0'
    cmp dl, 9
    jbe v_num
    jmp error

v_obj:
    inc rsi
    mov byte ptr [r14 + rbx], 1
    inc rbx
    call skipws
    cmp al, '}'
    jne expect_key
    inc rsi
    dec rbx
    jmp value_done

v_arr:
    inc rsi
    mov byte ptr [r14 + rbx], 0
    inc rbx
    call skipws
    cmp al, ']'
    jne parse_value
    inc rsi
    dec rbx
    jmp value_done

v_str:
    call string
    jmp value_done

v_true:
    lea rdx, [rip + lit_true]
    jmp literal
v_false:
    lea rdx, [rip + lit_false]
    jmp literal
v_null:
    lea rdx, [rip + lit_null]
literal:
    mov cl, byte ptr [rdx]
    test cl, cl
    jz value_done
    cmp cl, byte ptr [rsi]
    jne error
    inc rsi
    inc rdx
    jmp literal

v_num:
    cmp al, '-'
    jne 1f
    inc rsi
1:
    mov al, byte ptr [rsi]
    cmp al, '0'
    jne 2f
    inc rsi
    jmp num_frac
2:
    sub al, '1'
    cmp al, 8
    ja error
    inc rsi
3:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja num_frac
    inc rsi
    jmp 3b
num_frac:
    mov al, byte ptr [rsi]
    cmp al, '.'
    jne num_exp
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
4:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    jbe 4b
num_exp:
    mov al, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne value_done
    inc rsi
    mov al, byte ptr [rsi]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:
    inc rsi
6:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
7:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    jbe 7b
    jmp value_done

expect_key:
    call skipws
    cmp al, '"'
    jne error
    call string
    call skipws
    cmp al, ':'
    jne error
    inc rsi
    jmp parse_value

value_done:
    call skipws
    test rbx, rbx
    jnz 1f
    cmp rsi, rdi
    je success
    jmp error
1:
    mov cl, byte ptr [r14 + rbx - 1]
    cmp al, ','
    jne 2f
    inc rsi
    test cl, cl
    jnz expect_key
    jmp parse_value
2:
    test cl, cl
    jz 3f
    cmp al, '}'
    jne error
    jmp 4f
3:
    cmp al, ']'
    jne error
4:
    inc rsi
    dec rbx
    jmp value_done

# rsi at opening quote
string:
    inc rsi
s_loop:
    mov al, byte ptr [rsi]
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je s_done
    cmp al, 0x5C
    jne s_loop
    mov al, byte ptr [rsi]
    cmp al, '"'
    je s_esc_ok
    cmp al, 0x5C
    je s_esc_ok
    cmp al, '/'
    je s_esc_ok
    cmp al, 'b'
    je s_esc_ok
    cmp al, 'f'
    je s_esc_ok
    cmp al, 'n'
    je s_esc_ok
    cmp al, 'r'
    je s_esc_ok
    cmp al, 't'
    je s_esc_ok
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
s_hex:
    mov al, byte ptr [rsi]
    mov dl, al
    sub dl, '0'
    cmp dl, 9
    jbe 1f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
1:
    inc rsi
    dec ecx
    jnz s_hex
    jmp s_loop
s_esc_ok:
    inc rsi
    jmp s_loop
s_done:
    ret

skipws:
1:
    movzx eax, byte ptr [rsi]
    cmp al, ' '
    je 2f
    cmp al, 9
    je 2f
    cmp al, 10
    je 2f
    cmp al, 13
    je 2f
    ret
2:
    inc rsi
    jmp 1b

success:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov rax, rsi
    sub rax, r12
    lea r15, [rip + outbuf + 31]
    mov byte ptr [r15], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r15
    mov byte ptr [r15], dl
    test rax, rax
    jnz 1b
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    lea rdx, [rip + outbuf + 32]
    sub rdx, r15
    syscall
    mov eax, 60
    mov edi, 1
    syscall
