.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    dec rcx
    jnz byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call putnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call putnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call putnum
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
wloop:
    test rbx, rbx
    jle finish
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp wloop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, rdi = destination; advances rdi
putnum:
    mov r8d, 10
    xor ecx, ecx
1:
    xor edx, edx
    div r8
    add dl, 0x30
    push rdx
    inc ecx
    test rax, rax
    jnz 1b
2:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 2b
    ret

.bss
    .balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
