.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 65600
outbuf:
    .space 330000

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0              // total length
    mov x25, #65600
read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x25, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x26, x22             // start of output
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x21, #0              // offset
    mov w27, #' '
    mov w28, #'|'
row_loop:
    cmp x21, x20
    b.hs final
    sub x24, x20, x21
    mov x0, #16
    cmp x24, x0
    csel x24, x24, x0, lo    // n
    mov w0, w21
    bl hex8
    strb w27, [x22], #1
    strb w27, [x22], #1
    add x9, x19, x21         // row pointer
    mov x10, #0              // i
hex_loop:
    cmp x10, x24
    b.hs 1f
    ldrb w11, [x9, x10]
    lsr w12, w11, #4
    ldrb w12, [x23, x12]
    strb w12, [x22], #1
    and w12, w11, #15
    ldrb w12, [x23, x12]
    strb w12, [x22], #1
    strb w27, [x22], #1
    b 2f
1:
    strb w27, [x22], #1
    strb w27, [x22], #1
    strb w27, [x22], #1
2:
    cmp x10, #7
    b.ne 3f
    strb w27, [x22], #1
3:
    add x10, x10, #1
    cmp x10, #16
    b.lo hex_loop
    strb w27, [x22], #1
    strb w28, [x22], #1
    mov x10, #0
chr_loop:
    ldrb w11, [x9, x10]
    sub w12, w11, #0x20
    cmp w12, #0x5e
    b.ls 4f
    mov w11, #'.'
4:
    strb w11, [x22], #1
    add x10, x10, #1
    cmp x10, x24
    b.lo chr_loop
    strb w28, [x22], #1
    mov w11, #10
    strb w11, [x22], #1
    add x21, x21, x24
    b row_loop

final:
    mov w0, w20
    bl hex8
    mov w11, #10
    strb w11, [x22], #1
write_loop:
    subs x2, x22, x26
    b.le done
    mov x0, #1
    mov x1, x26
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le done
    add x26, x26, x0
    b write_loop
done:
    mov x0, #0
    mov x8, #93
    svc #0

hex8:
    mov w1, #28
1:
    lsr w2, w0, w1
    and w2, w2, #15
    ldrb w2, [x23, x2]
    strb w2, [x22], #1
    subs w1, w1, #4
    b.ge 1b
    ret
