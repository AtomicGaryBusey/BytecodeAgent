rdloop:

    xor edi, edi

    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle rddone
    add r13, rax
    jmp rdloop
rddone:

    xor r15d, r15d
    mov rsi, r12
    lea r10, [r12 + r13]
parse:
    cmp rsi, r10
    jae parsed
    mov rdi, rsi
scan:
    cmp rsi, r10
    jae endline
    cmp byte ptr [rsi], 10
    je endline
    inc rsi
    jmp scan
endline:

    sub rax, rdi

    mov qword ptr [r14 + rcx], rdi
    mov qword ptr [r14 + rcx + 8], rax

parsed:

outer:

    jae sorted
    mov rax, r8
    shl rax, 4
    lea r9, [r14 + rax]
    mov rbx, qword ptr [r9]
    mov rbp, qword ptr [r9 + 8]

    jbe place

    mov rdx, qword ptr [r9 - 8]

    mov rcx, rdx
    cmp rcx, rbp
    cmova rcx, rbp

    repe cmpsb
    jne 1f

1:

    movups xmmword ptr [r9], xmm0

    jmp inner

    mov rcx, qword ptr [r14 + rax + 8]

wrloop:

    mov edi, 1
    mov rsi, r12

    syscall
