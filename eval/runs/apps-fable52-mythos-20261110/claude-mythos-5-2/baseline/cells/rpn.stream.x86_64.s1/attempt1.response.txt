.globl _start
.equ INSIZE, 65552

.bss
.align 16
inbuf: .zero 65600
outbuf: .zero 524288
vstack: .zero 8192
tmp: .zero 32

.text
_start:
    lea r15, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov edx, INSIZE
    sub rdx, r12
    jz read_done
    lea rsi, [r15 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov rsi, r15
    add r12, r15
    lea rdi, [rip + outbuf]
    lea r13, [rip + vstack]
    mov r10d, 10

line_loop:
    cmp rsi, r12
    jae finish
    xor ebx, ebx
token_loop:
    cmp rsi, r12
    jae line_end
    mov al, byte ptr [rsi]
    cmp al, 10
    je line_end
    cmp al, 32
    je 1f
    cmp al, 9
    jne tok_start
1:
    inc rsi
    jmp token_loop

tok_start:
    mov rdx, rsi
2:
    inc rdx
    cmp rdx, r12
    jae 3f
    mov cl, byte ptr [rdx]
    cmp cl, 32
    je 3f
    cmp cl, 9
    je 3f
    cmp cl, 10
    jne 2b
3:
    mov rcx, rdx
    sub rcx, rsi
    cmp rcx, 1
    jne literal
    cmp al, '+'
    je op_common
    cmp al, '-'
    je op_common
    cmp al, '*'
    je op_common
    cmp al, '/'
    je op_common
    cmp al, '%'
    je op_common

literal:
    mov r8, rsi
    xor r9d, r9d
    cmp al, '-'
    jne 4f
    inc r8
    mov r9d, 1
4:
    mov rcx, rdx
    sub rcx, r8
    jz error_line
    cmp rcx, 18
    ja error_line
    xor eax, eax
5:
    movzx ecx, byte ptr [r8]
    sub ecx, '0'
    cmp ecx, 9
    ja error_line
    imul rax, rax, 10
    add rax, rcx
    inc r8
    cmp r8, rdx
    jb 5b
    test r9d, r9d
    jz 6f
    neg rax
6:
    mov qword ptr [r13 + rbx*8], rax
    inc rbx
    mov rsi, rdx
    jmp token_loop

op_common:
    cmp rbx, 2
    jb error_line
    mov rsi, rdx
    mov r8, qword ptr [r13 + rbx*8 - 8]
    dec rbx
    mov r9, qword ptr [r13 + rbx*8 - 8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    # division or remainder
    test r8, r8
    jz error_line
    cmp r8, -1
    jne 7f
    mov rcx, 0x8000000000000000
    cmp r9, rcx
    je error_line
7:
    mov cl, al
    mov rax, r9
    cqo
    idiv r8
    cmp cl, '/'
    je 8f
    mov rax, rdx
8:
    mov r9, rax
    jmp op_store
op_add:
    add r9, r8
    jmp op_store
op_sub:
    sub r9, r8
    jmp op_store
op_mul:
    imul r9, r8
op_store:
    mov qword ptr [r13 + rbx*8 - 8], r9
    jmp token_loop

error_line:
    cmp rsi, r12
    jae emit_error
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error_line
emit_error:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_loop

line_end:
    cmp rsi, r12
    jae 1f
    inc rsi
1:
    cmp rbx, 1
    jne emit_error
    mov rax, qword ptr [r13]
    test rax, rax
    jns 2f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
2:
    lea r8, [rip + tmp + 24]
    mov r9, r8
3:
    xor edx, edx
    div r10
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 3b
4:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb 4b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_loop

finish:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle do_exit
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rbx, rax
    jmp write_loop
do_exit:
    xor edi, edi
    mov eax, 60
    syscall
