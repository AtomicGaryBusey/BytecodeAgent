.globl _start

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    ble done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
scan:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    beq is_ws
    sub w4, w3, #9
    cmp w4, #4
    bls is_ws
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
next:
    cmp x1, x2
    blo scan
    b read_loop

done:
    adrp x24, outend
    add x24, x24, :lo12:outend
    mov x9, x24
    mov w3, #10
    strb w3, [x9, #-1]!
    mov x0, x21
    bl conv
    mov w3, #32
    strb w3, [x9, #-1]!
    mov x0, x20
    bl conv
    mov w3, #32
    strb w3, [x9, #-1]!
    mov x0, x19
    bl conv

write_loop:
    cmp x9, x24
    bhs finish
    mov x0, #1
    mov x1, x9
    sub x2, x24, x9
    mov x8, #64
    svc #0
    cmp x0, #0
    ble finish
    add x9, x9, x0
    b write_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0

conv:
    mov x10, #10
1:
    udiv x11, x0, x10
    msub x12, x11, x10, x0
    add w12, w12, #48
    strb w12, [x9, #-1]!
    mov x0, x11
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 96
outend:
    .space 16
