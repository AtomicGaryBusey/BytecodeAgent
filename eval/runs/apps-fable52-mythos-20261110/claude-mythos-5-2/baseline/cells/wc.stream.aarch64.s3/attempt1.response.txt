.globl _start

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 96
outend:
    .space 16

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x9, x23
    mov x10, x0
byte_loop:
    ldrb w11, [x9], #1
    cmp w11, #10
    cinc x19, x19, eq
    cmp w11, #32
    b.eq is_ws
    sub w12, w11, #9
    cmp w12, #4
    b.ls is_ws
    cbnz x22, next
    mov x22, #1
    add x20, x20, #1
    b next
is_ws:
    mov x22, #0
next:
    subs x10, x10, #1
    b.ne byte_loop
    b read_loop

done:
    adrp x9, outend
    add x9, x9, :lo12:outend
    mov x13, x9              // end pointer
    mov x14, #10
    mov w2, #10
    strb w2, [x9, #-1]!
    mov x0, x21
    bl conv
    mov w2, #32
    strb w2, [x9, #-1]!
    mov x0, x20
    bl conv
    mov w2, #32
    strb w2, [x9, #-1]!
    mov x0, x19
    bl conv

    mov x0, #1
    mov x1, x9
    sub x2, x13, x9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// x0 = value, x9 = pointer (digits stored downward), x14 = 10
conv:
    udiv x1, x0, x14
    msub x2, x1, x14, x0
    add w2, w2, #48
    strb w2, [x9, #-1]!
    mov x0, x1
    cbnz x1, conv
    ret
