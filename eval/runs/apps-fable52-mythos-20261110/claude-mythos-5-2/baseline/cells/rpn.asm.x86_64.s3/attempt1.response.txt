.globl _start
.equ INCAP, 65600

.bss
.align 16
inbuf: .space 65664
outbuf: .space 400000
stk: .space 264000
tmp: .space 32

.section .rodata
errmsg: .ascii "error\n"

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INCAP
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    add r12, rsi
    lea rdi, [rip + outbuf]
    lea r14, [rip + stk]

line_start:
    cmp rsi, r12
    jae finish
    xor r13d, r13d
tok_loop:
    cmp rsi, r12
    je line_end
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne 1f
    inc rsi
    jmp line_end
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jne token
2:
    inc rsi
    jmp tok_loop

token:
    mov rcx, rsi
    xor r8d, r8d
    cmp al, '-'
    jne 1f
    mov r8d, 1
    inc rcx
1:
    xor eax, eax
    xor r9d, r9d
dig_loop:
    cmp rcx, r12
    je dig_done
    movzx edx, byte ptr [rcx]
    sub edx, '0'
    cmp edx, 9
    ja dig_done
    imul rax, rax, 10
    add rax, rdx
    inc r9
    inc rcx
    jmp dig_loop
dig_done:
    test r9, r9
    jz operator
    cmp r9, 18
    ja error
    mov rdx, rcx
    call is_delim
    jne error
    test r8d, r8d
    jz 1f
    neg rax
1:
    mov qword ptr [r14 + r13*8], rax
    inc r13
    mov rsi, rcx
    jmp tok_loop

operator:
    lea rdx, [rsi + 1]
    call is_delim
    jne error
    cmp r13, 2
    jb error
    mov rcx, qword ptr [r14 + r13*8 - 8]
    mov rax, qword ptr [r14 + r13*8 - 16]
    movzx edx, byte ptr [rsi]
    cmp dl, '+'
    je op_add
    cmp dl, '-'
    je op_sub
    cmp dl, '*'
    je op_mul
    cmp dl, '/'
    je op_div
    cmp dl, '%'
    je op_div
    jmp error
op_add:
    add rax, rcx
    jmp store
op_sub:
    sub rax, rcx
    jmp store
op_mul:
    imul rax, rcx
    jmp store
op_div:
    test rcx, rcx
    jz error
    cmp rcx, -1
    jne 1f
    mov r8, 0x8000000000000000
    cmp rax, r8
    je error
1:
    mov r9d, edx
    cqo
    idiv rcx
    cmp r9b, '/'
    je store
    mov rax, rdx
store:
    mov qword ptr [r14 + r13*8 - 16], rax
    dec r13
    inc rsi
    jmp tok_loop

error:
    cmp rsi, r12
    je emit_err
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
emit_err:
    mov eax, dword ptr [rip + errmsg]
    mov dword ptr [rdi], eax
    mov ax, word ptr [rip + errmsg + 4]
    mov word ptr [rdi + 4], ax
    add rdi, 6
    jmp line_start

line_end:
    cmp r13, 1
    jne emit_err
    mov rax, qword ptr [r14]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    lea r8, [rip + tmp + 24]
    mov r9, r8
    mov r10d, 10
2:
    xor edx, edx
    div r10
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jne 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_start

is_delim:
    cmp rdx, r12
    je 1f
    cmp byte ptr [rdx], 10
    je 1f
    cmp byte ptr [rdx], 32
    je 1f
    cmp byte ptr [rdx], 9
1:
    ret

finish:
    mov rbx, rdi
    lea rsi, [rip + outbuf]
write_loop:
    mov rdx, rbx
    sub rdx, rsi
    jle exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit
    add rsi, rax
    jmp write_loop
exit:
    mov eax, 60
    xor edi, edi
    syscall
