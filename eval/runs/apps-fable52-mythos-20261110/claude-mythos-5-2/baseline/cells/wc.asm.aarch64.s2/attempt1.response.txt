.globl _start
.text
_start:
    mov x19, #0            // lines
    mov x20, #0            // words
    mov x21, #0            // bytes
    mov x22, #0            // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf
read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    sub w4, w3, #9
    cmp w4, #4
    ccmp w3, #32, #4, hi
    b.eq is_ws
    cbnz x22, 1f
    add x20, x20, #1
    mov x22, #1
1:
    cmp x1, x2
    b.lo byte_loop
    b read_loop
is_ws:
    mov x22, #0
    cmp x1, x2
    b.lo byte_loop
    b read_loop

done:
    adrp x5, outend
    add x5, x5, :lo12:outend
    mov x6, x5             // end pointer
    mov w7, #10
    strb w7, [x5, #-1]!
    mov x0, x21
    bl conv
    mov w7, #32
    strb w7, [x5, #-1]!
    mov x0, x20
    bl conv
    mov w7, #32
    strb w7, [x5, #-1]!
    mov x0, x19
    bl conv
write_loop:
    sub x2, x6, x5
    cbz x2, finish
    mov x0, #1
    mov x1, x5
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x5, x5, x0
    b write_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0

// convert x0 to decimal, storing backwards at x5
conv:
    mov x9, #10
1:
    udiv x10, x0, x9
    msub x11, x10, x9, x0
    add w11, w11, #48
    strb w11, [x5, #-1]!
    mov x0, x10
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 96
outend:
    .space 16
