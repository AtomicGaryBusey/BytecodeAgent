    ldrb w0, [x19]
    cmp w0, #10

    cmp w1, #9

    cmp w0, #'-'

    cmp x3, x26

    cbz x4, 8f
    neg x6, x6

    mov x19, x26

    b.lo error_skip
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]  // b
    sub x9, x23, #1
    ldr x2, [x22, x9, lsl #3]   // a
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    // div or mod
    cbz x1, error_skip
    mov x3, #0x8000000000000000
    cmp x2, x3
    b.ne 1f
    cmn x1, #1
    b.eq error_skip
1:

    cmp w0, #'/'
    b.eq op_store
    msub x3, x3, x1, x2

    b op_store

    sub x3, x2, x1

    b.ne emit_err
    ldr x0, [x22]
    cmp x0, #0
    b.ge 1f
    mov w1, #'-'
    strb w1, [x21], #1
    neg x0, x0
