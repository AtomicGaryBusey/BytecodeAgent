.globl _start

.section .rodata
s_ok: .ascii "ok\n"
s_err: .ascii "error at "
s_true: .asciz "rue"
s_false: .asciz "alse"
s_null: .asciz "ull"

.bss
buf: .zero 65600
stk: .zero 2048
outbuf: .zero 48

.text
_start:
    lea rbx, [rip + buf]
    xor r14d, r14d              # length
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r14]
    mov edx, 65536
    sub rdx, r14
    jle 2f
    syscall
    test rax, rax
    jle 2f
    add r14, rax
    jmp 1b
2:
    lea r13, [rbx + r14]        # end pointer
    mov byte ptr [r13], 0       # sentinel
    mov rsi, rbx
    xor r12d, r12d              # depth
    lea r15, [rip + stk]

parse_value:
    call skip_ws
    cmp al, '{'
    je p_obj
    cmp al, '['
    je p_arr
    cmp al, '"'
    je p_str
    cmp al, '-'
    je p_num
    cmp al, 't'
    je p_true
    cmp al, 'f'
    je p_false
    cmp al, 'n'
    je p_null
    mov dl, al
    sub dl, '0'
    cmp dl, 10
    jb p_num
    jmp error

p_true:
    lea rdi, [rip + s_true]
    jmp p_lit
p_false:
    lea rdi, [rip + s_false]
    jmp p_lit
p_null:
    lea rdi, [rip + s_null]
p_lit:
    inc rsi
1:
    mov dl, byte ptr [rdi]
    test dl, dl
    jz value_done
    cmp dl, byte ptr [rsi]
    jne error
    inc rsi
    inc rdi
    jmp 1b

p_obj:
    inc rsi
    call skip_ws
    cmp al, '}'
    jne 1f
    inc rsi
    jmp value_done
1:
    mov byte ptr [r15 + r12], 1
    inc r12
    jmp member

p_arr:
    inc rsi
    call skip_ws
    cmp al, ']'
    jne 1f
    inc rsi
    jmp value_done
1:
    mov byte ptr [r15 + r12], 0
    inc r12
    jmp parse_value

p_str:
    call string
    jmp value_done

member:
    cmp al, '"'
    jne error
    call string
    call skip_ws
    cmp al, ':'
    jne error
    inc rsi
    jmp parse_value

p_num:
    cmp al, '-'
    jne 1f
    inc rsi
1:
    mov al, byte ptr [rsi]
    cmp al, '0'
    jne 2f
    inc rsi
    jmp 4f
2:
    sub al, '1'
    cmp al, 9
    jae error
    inc rsi
3:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae 4f
    inc rsi
    jmp 3b
4:
    cmp byte ptr [rsi], '.'
    jne 6f
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae error
5:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jb 5b
6:
    mov al, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne value_done
    inc rsi
    mov al, byte ptr [rsi]
    cmp al, '+'
    je 7f
    cmp al, '-'
    jne 8f
7:
    inc rsi
8:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jae error
9:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 10
    jb 9b
    jmp value_done

value_done:
    call skip_ws
    test r12, r12
    jnz 1f
    cmp rsi, r13
    jne error
    jmp ok
1:
    mov dl, byte ptr [r15 + r12 - 1]
    cmp al, ','
    jne 2f
    inc rsi
    test dl, dl
    jz parse_value
    call skip_ws
    jmp member
2:
    test dl, dl
    jz 3f
    cmp al, '}'
    jne error
    jmp 4f
3:
    cmp al, ']'
    jne error
4:
    inc rsi
    dec r12
    jmp value_done

# skip whitespace; returns current byte in al
skip_ws:
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
    ret
1:
    inc rsi
    jmp skip_ws

# rsi at opening quote
string:
    inc rsi
1:
    mov al, byte ptr [rsi]
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je 9f
    cmp al, 0x5C
    jne 1b
    mov al, byte ptr [rsi]
    cmp al, '"'
    je 2f
    cmp al, 0x5C
    je 2f
    cmp al, '/'
    je 2f
    cmp al, 'b'
    je 2f
    cmp al, 'f'
    je 2f
    cmp al, 'n'
    je 2f
    cmp al, 'r'
    je 2f
    cmp al, 't'
    je 2f
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
3:
    mov al, byte ptr [rsi]
    mov dl, al
    sub dl, '0'
    cmp dl, 10
    jb 4f
    or al, 0x20
    sub al, 'a'
    cmp al, 6
    jae error
4:
    inc rsi
    dec ecx
    jnz 3b
    jmp 1b
2:
    inc rsi
    jmp 1b
9:
    ret

ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov rax, rsi
    sub rax, rbx
    lea r8, [rip + outbuf + 40]
    mov r9, r8                  # end
    dec r8
    mov byte ptr [r8], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
    sub r8, 9
    mov rax, qword ptr [rip + s_err]
    mov qword ptr [r8], rax
    mov al, byte ptr [rip + s_err + 8]
    mov byte ptr [r8 + 8], al
    mov rsi, r8
    mov rdx, r9
    sub rdx, r8
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
