.globl _start
.equ INSZ, 32768

.bss
.align 16
inbuf: .zero 32768
.align 16
entries: .zero 524304
.align 16
outbuf: .zero 32800

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INSZ
    sub rdx, r13
    jz 2f
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    # parse lines
    mov rsi, r12
    lea r14, [r12 + r13]
    xor r15d, r15d
    lea rbx, [rip + entries]
3:
    cmp rsi, r14
    jae 6f
    mov rax, rsi
4:
    cmp rsi, r14
    jae 5f
    cmp byte ptr [rsi], 10
    je 5f
    inc rsi
    jmp 4b
5:
    mov rcx, r15
    shl rcx, 4
    mov qword ptr [rbx + rcx], rax
    mov rdx, rsi
    sub rdx, rax
    mov qword ptr [rbx + rcx + 8], rdx
    inc r15
    cmp rsi, r14
    jae 3b
    inc rsi
    jmp 3b
6:
    # insertion sort
    shl r15, 4
    mov edx, 16
7:
    cmp rdx, r15
    jae 12f
    mov r8, qword ptr [rbx + rdx]
    mov r9, qword ptr [rbx + rdx + 8]
    mov rax, rdx
8:
    test rax, rax
    jz 11f
    mov rsi, qword ptr [rbx + rax - 16]
    mov r10, qword ptr [rbx + rax - 8]
    mov rdi, r8
    mov rcx, r10
    cmp rcx, r9
    jbe 9f
    mov rcx, r9
9:
    test rcx, rcx
    jz 10f
    repe cmpsb
    je 10f
    ja 13f
    jmp 11f
10:
    cmp r10, r9
    jbe 11f
13:
    mov rsi, qword ptr [rbx + rax - 16]
    mov qword ptr [rbx + rax], rsi
    mov qword ptr [rbx + rax + 8], r10
    sub rax, 16
    jmp 8b
11:
    mov qword ptr [rbx + rax], r8
    mov qword ptr [rbx + rax + 8], r9
    add rdx, 16
    jmp 7b
12:
    # build output
    lea rdi, [rip + outbuf]
    xor eax, eax
14:
    cmp rax, r15
    jae 15f
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rax, 16
    jmp 14b
15:
    lea r12, [rip + outbuf]
    mov r13, rdi
    sub r13, r12
16:
    test r13, r13
    jle 17f
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle 17f
    add r12, rax
    sub r13, rax
    jmp 16b
17:
    mov eax, 60
    xor edi, edi
    syscall
