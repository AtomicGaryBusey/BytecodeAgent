.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rdloop:
    mov edx, 65600
    sub rdx, r13
    jle rddone
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle rddone
    add r13, rax
    jmp rdloop
rddone:
    xor ecx, ecx
    mov r14, r13
    mov r15, r13
findbang:
    cmp rcx, r13
    jae scan
    cmp byte ptr [r12 + rcx], 0x21
    je foundbang
    inc rcx
    jmp findbang
foundbang:
    mov r14, rcx
    lea r15, [rcx + 1]
scan:
    xor ecx, ecx
    xor ebx, ebx
    lea r8, [rip + jumptab]
    lea r9, [rip + stk]
scanloop:
    cmp rcx, r14
    jae scanend
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    mov dword ptr [r9 + rbx*4], ecx
    inc rbx
    jmp scannext
1:
    cmp al, 0x5D
    jne scannext
    test rbx, rbx
    jz bad
    dec rbx
    mov edx, dword ptr [r9 + rbx*4]
    mov dword ptr [r8 + rcx*4], edx
    mov dword ptr [r8 + rdx*4], ecx
scannext:
    inc rcx
    jmp scanloop
scanend:
    test rbx, rbx
    jnz bad

    xor ebx, ebx
    lea rbp, [rip + tape]
    lea r9, [rip + outbuf]
    xor r10d, r10d
run:
    cmp rbx, r14
    jae fin
    movzx eax, byte ptr [r12 + rbx]
    inc rbx
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp run
op_right:
    inc rbp
    jmp run
op_left:
    dec rbp
    jmp run
op_inc:
    inc byte ptr [rbp]
    jmp run
op_dec:
    dec byte ptr [rbp]
    jmp run
op_open:
    cmp byte ptr [rbp], 0
    jne run
    mov ebx, dword ptr [r8 + rbx*4 - 4]
    inc rbx
    jmp run
op_close:
    cmp byte ptr [rbp], 0
    je run
    mov ebx, dword ptr [r8 + rbx*4 - 4]
    inc rbx
    jmp run
op_out:
    mov al, byte ptr [rbp]
    mov byte ptr [r9 + r10], al
    inc r10
    cmp r10, 65536
    jb run
    call flush
    jmp run
op_in:
    xor eax, eax
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
1:
    mov byte ptr [rbp], al
    jmp run

flush:
    test r10, r10
    jz 2f
    mov r11, r9
1:
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r10
    syscall
    lea r9, [rip + outbuf]
    xor r10d, r10d
2:
    ret

fin:
    call flush
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
inbuf:
    .zero 65664
jumptab:
    .zero 262656
stk:
    .zero 262656
tape:
    .zero 30016
outbuf:
    .zero 65600
