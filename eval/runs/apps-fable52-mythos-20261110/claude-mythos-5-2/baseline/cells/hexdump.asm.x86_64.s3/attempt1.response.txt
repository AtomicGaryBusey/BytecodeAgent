.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .space 65600
outbuf:
    .space 340000

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, 65552
    sub rdx, r13
    jle done_read
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle done_read
    add r13, rax
    jmp read_loop
done_read:
    lea rdi, [rip + outbuf]
    xor r14d, r14d
row_loop:
    cmp r14, r13
    jae finish
    mov eax, r14d
    call puthex8
    mov ax, 0x2020
    stosw
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    xor ebx, ebx
hexloop:
    cmp rbx, r15
    jae pad
    lea rsi, [r12 + r14]
    movzx eax, byte ptr [rsi + rbx]
    call putbyte
    mov al, 0x20
    stosb
    jmp next
pad:
    mov al, 0x20
    stosb
    stosb
    stosb
next:
    cmp ebx, 7
    jne 2f
    mov al, 0x20
    stosb
2:
    inc ebx
    cmp ebx, 16
    jb hexloop
    mov ax, 0x7c20
    stosw
    xor ebx, ebx
charloop:
    cmp rbx, r15
    jae 3f
    lea rsi, [r12 + r14]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    jbe ok
dot:
    mov al, 0x2e
ok:
    stosb
    inc ebx
    jmp charloop
3:
    mov ax, 0x0a7c
    stosw
    add r14, r15
    jmp row_loop

finish:
    mov eax, r13d
    call puthex8
    mov al, 10
    stosb
    lea r14, [rip + outbuf]
    mov r15, rdi
    sub r15, r14
write_loop:
    test r15, r15
    jle exit_ok
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall
    test rax, rax
    jle exit_ok
    add r14, rax
    sub r15, rax
    jmp write_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

# eax = value, writes 8 hex digits at rdi, advances rdi
puthex8:
    mov ecx, 8
    lea rsi, [rip + hexdigits]
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

# eax = byte (zero-extended), writes 2 hex digits at rdi, advances rdi
putbyte:
    lea rsi, [rip + hexdigits]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rdi + 1], al
    add rdi, 2
    ret
