.globl _start
.equ INSIZE, 70000

.bss
inbuf:  .space 70016
outbuf: .space 524288
vstack: .space 8192
tmp:    .space 32

.text
_start:
    xor r12d, r12d
read_loop:
    mov edx, INSIZE
    sub rdx, r12
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r13, [rsi + r12]
    lea rdi, [rip + outbuf]
    lea r14, [rip + vstack]

line_loop:
    cmp rsi, r13
    jae done
    xor r15d, r15d
token_loop:
    cmp rsi, r13
    jae line_end_eof
    mov al, byte ptr [rsi]
    cmp al, 10
    je line_end
    cmp al, 32
    je 1f
    cmp al, 9
    jne tok_start
1:
    inc rsi
    jmp token_loop

tok_start:
    mov rbx, rsi
    mov rcx, rsi
2:
    inc rcx
    cmp rcx, r13
    jae 3f
    mov al, byte ptr [rcx]
    cmp al, 10
    je 3f
    cmp al, 32
    je 3f
    cmp al, 9
    jne 2b
3:
    mov rdx, rbx
    xor r8d, r8d
    mov rax, rcx
    sub rax, rbx
    movzx r9d, byte ptr [rbx]
    cmp r9b, '-'
    jne nodash
    cmp rax, 1
    je operator
    mov r8d, 1
    inc rdx
nodash:
    mov rax, rcx
    sub rax, rdx
    cmp rax, 18
    ja error
    xor eax, eax
dl:
    movzx r10d, byte ptr [rdx]
    sub r10d, '0'
    cmp r10d, 9
    ja notnum
    imul rax, rax, 10
    add rax, r10
    inc rdx
    cmp rdx, rcx
    jb dl
    test r8d, r8d
    jz 4f
    neg rax
4:
    mov qword ptr [r14 + r15*8], rax
    inc r15
    mov rsi, rcx
    jmp token_loop

notnum:
    mov rax, rcx
    sub rax, rbx
    cmp rax, 1
    jne error
operator:
    cmp r15, 2
    jb error
    mov r10, qword ptr [r14 + r15*8 - 8]
    mov rax, qword ptr [r14 + r15*8 - 16]
    cmp r9b, '+'
    je op_add
    cmp r9b, '-'
    je op_sub
    cmp r9b, '*'
    je op_mul
    cmp r9b, '/'
    je op_div
    cmp r9b, '%'
    je op_div
    jmp error
op_add:
    add rax, r10
    jmp op_store
op_sub:
    sub rax, r10
    jmp op_store
op_mul:
    imul rax, r10
    jmp op_store
op_div:
    test r10, r10
    jz error
    cmp r10, -1
    jne 5f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error
5:
    cqo
    idiv r10
    cmp r9b, '/'
    je op_store
    mov rax, rdx
op_store:
    dec r15
    mov qword ptr [r14 + r15*8 - 8], rax
    mov rsi, rcx
    jmp token_loop

error:
    cmp rsi, r13
    jae emit_error
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
emit_error:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_loop

line_end:
    inc rsi
line_end_eof:
    cmp r15, 1
    jne emit_error
    mov rax, qword ptr [r14]
    test rax, rax
    jns 6f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
6:
    lea r8, [rip + tmp + 32]
    mov r9, r8
    mov r10d, 10
7:
    xor edx, edx
    div r10
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 7b
8:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, r9
    jb 8b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_loop

done:
    lea rbx, [rip + outbuf]
    mov r12, rdi
    sub r12, rbx
wloop:
    test r12, r12
    jle exit
    mov rsi, rbx
    mov rdx, r12
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle exit
    add rbx, rax
    sub r12, rax
    jmp wloop
exit:
    xor edi, edi
    mov eax, 60
    syscall
