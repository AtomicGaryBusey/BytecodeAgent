.globl _start
.text
_start:
    xor r12d, r12d
rd_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 70000
    sub rdx, r12
    jle rd_done
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    lea rsi, [rip + inbuf]
    add r12, rsi
    lea rdi, [rip + outbuf]
    lea r14, [rip + vstack]

line_loop:
    cmp rsi, r12
    jae done
    xor r13d, r13d
tok_loop:
    cmp rsi, r12
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je line_end_nl
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    # find token end
    mov rbx, rsi
1:
    inc rbx
    cmp rbx, r12
    jae 2f
    mov dl, byte ptr [rbx]
    cmp dl, 10
    je 2f
    cmp dl, 32
    je 2f
    cmp dl, 9
    jne 1b
2:
    mov rcx, rbx
    sub rcx, rsi
    cmp rcx, 1
    jne number
    cmp al, '+'
    je operator
    cmp al, '-'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator
number:
    mov r8, rsi
    xor r9d, r9d
    cmp al, '-'
    jne 1f
    mov r9d, 1
    inc r8
1:
    mov rcx, rbx
    sub rcx, r8
    cmp rcx, 1
    jl error
    cmp rcx, 18
    jg error
    xor eax, eax
2:
    movzx edx, byte ptr [r8]
    sub edx, '0'
    cmp edx, 9
    ja error
    imul rax, rax, 10
    add rax, rdx
    inc r8
    cmp r8, rbx
    jb 2b
    test r9d, r9d
    jz 3f
    neg rax
3:
    mov qword ptr [r14 + r13*8], rax
    inc r13
    mov rsi, rbx
    jmp tok_loop

skip_sep:
    inc rsi
    jmp tok_loop

operator:
    cmp r13, 2
    jb error
    mov rcx, qword ptr [r14 + r13*8 - 8]
    dec r13
    mov r8, qword ptr [r14 + r13*8 - 8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    # division or modulo
    test rcx, rcx
    jz error
    cmp rcx, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp r8, rdx
    je error
1:
    mov r9d, eax
    mov rax, r8
    cqo
    idiv rcx
    cmp r9b, '/'
    je 2f
    mov rax, rdx
2:
    mov r8, rax
    jmp op_store
op_add:
    add r8, rcx
    jmp op_store
op_sub:
    sub r8, rcx
    jmp op_store
op_mul:
    imul r8, rcx
op_store:
    mov qword ptr [r14 + r13*8 - 8], r8
    mov rsi, rbx
    jmp tok_loop

error:
    cmp rsi, r12
    jae err_emit
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
err_emit:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_loop

line_end_nl:
    inc rsi
line_end:
    cmp r13, 1
    jne err_emit
    mov rax, qword ptr [r14]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    lea r8, [rip + tmp + 32]
    mov r9, r8
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, r9
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_loop

done:
    mov rbx, rdi
    lea rsi, [rip + outbuf]
    sub rbx, rsi
wr_loop:
    test rbx, rbx
    jle wr_done
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle wr_done
    add rsi, rax
    sub rbx, rax
    jmp wr_loop
wr_done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
    .balign 16
tmp:
    .zero 64
vstack:
    .zero 8192
inbuf:
    .zero 70016
outbuf:
    .zero 400000
