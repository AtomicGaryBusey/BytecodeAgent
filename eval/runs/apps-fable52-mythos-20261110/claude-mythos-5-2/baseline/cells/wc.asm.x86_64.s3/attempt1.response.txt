.globl _start

.bss
buffer:
    .zero 65536
outbuf:
    .zero 128

.text
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in-word flag

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buffer]
    mov r8, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 0x20
    je is_ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe is_ws
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
is_ws:
    xor r15d, r15d
next_byte:
    dec r8
    jnz byte_loop
    jmp read_loop

done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call put_dec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call put_dec
    mov byte ptr [rdi], 10
    inc rdi

    lea rbx, [rip + outbuf]
    mov r12, rdi
    sub r12, rbx            # remaining length
write_loop:
    test r12, r12
    jle finish
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall
    test rax, rax
    jle finish
    add rbx, rax
    sub r12, rax
    jmp write_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

# put_dec: rax = value, rdi = destination; advances rdi
put_dec:
    mov ecx, 10
    xor r8d, r8d
1:
    xor edx, edx
    div rcx
    add edx, 0x30
    push rdx
    inc r8d
    test rax, rax
    jnz 1b
2:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec r8d
    jnz 2b
    ret
