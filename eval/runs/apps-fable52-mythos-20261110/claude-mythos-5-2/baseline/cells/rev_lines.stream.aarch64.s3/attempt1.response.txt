.globl _start

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
read_loop:
    mov x2, #0x10000
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, x19                // current position
    add x22, x19, x20           // end of input
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x23, x24                // output pointer
line_loop:
    cmp x21, x22
    b.hs finish
    mov x9, x21
1:
    cmp x9, x22
    b.hs 2f
    ldrb w10, [x9]
    cmp w10, #10
    b.eq 2f
    add x9, x9, #1
    b 1b
2:
    mov x11, x9
3:
    cmp x11, x21
    b.ls 4f
    ldrb w10, [x11, #-1]!
    strb w10, [x23], #1
    b 3b
4:
    mov w10, #10
    strb w10, [x23], #1
    add x21, x9, #1
    b line_loop
finish:
    sub x25, x23, x24           // remaining bytes to write
write_loop:
    cbz x25, exit_ok
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x24, x24, x0
    sub x25, x25, x0
    b write_loop
exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_ok

.bss
.balign 16
inbuf:
    .space 0x11000
outbuf:
    .space 0x11000
