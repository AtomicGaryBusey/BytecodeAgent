.globl _start
.section .rodata
hexdig: .ascii "0123456789abcdef"
.bss
.align 16
inbuf: .space 65600
outbuf: .space 400000
.text
_start:
    lea r12, [rip + inbuf]
    lea r8, [rip + hexdig]
    xor r13d, r13d
1:
    mov edx, 65552
    sub rdx, r13
    jz 2f
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    lea rdi, [rip + outbuf]
    xor r14d, r14d
.Lrow:
    mov rcx, r13
    sub rcx, r14
    jz .Lfinal
    cmp rcx, 16
    jbe 1f
    mov ecx, 16
1:
    mov r15, rcx
    mov eax, r14d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ebx, ebx
.Lbyte:
    cmp ebx, r15d
    jae .Lpad
    lea rsi, [r12 + r14]
    movzx eax, byte ptr [rsi + rbx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp .Lnext
.Lpad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
.Lnext:
    add rdi, 3
    cmp ebx, 7
    jne 1f
    mov byte ptr [rdi], 0x20
    inc rdi
1:
    inc ebx
    cmp ebx, 16
    jb .Lbyte
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ebx, ebx
    lea rsi, [r12 + r14]
.Lchr:
    movzx eax, byte ptr [rsi + rbx]
    lea edx, [rax - 0x20]
    cmp dl, 0x5e
    jbe 1f
    mov al, 0x2e
1:
    mov byte ptr [rdi], al
    inc rdi
    inc ebx
    cmp ebx, r15d
    jb .Lchr
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r14, r15
    jmp .Lrow
.Lfinal:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
.Lwr:
    test rbx, rbx
    jle .Lexit
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle .Lexit
    add rsi, rax
    sub rbx, rax
    jmp .Lwr
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret
