.globl _start
.equ BUFSZ, 65600
.equ OUTSZ, 65536

.bss
.align 16
buf:    .zero 65600
outbuf: .zero 65536
tape:   .zero 30016
.align 16
jt:     .zero 262400

.text
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
rd_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, BUFSZ
    sub rdx, r13
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd_loop
rd_done:
    mov rbp, rsp
    lea r11, [rip + jt]
    xor esi, esi            # read index
    xor edi, edi            # write index
pp_loop:
    cmp rsi, r13
    jae pp_done
    movzx eax, byte ptr [r12 + rsi]
    inc rsi
    cmp al, '!'
    je pp_done
    cmp al, '>'
    je pp_store
    cmp al, '<'
    je pp_store
    cmp al, '+'
    je pp_store
    cmp al, '-'
    je pp_store
    cmp al, '.'
    je pp_store
    cmp al, ','
    je pp_store
    cmp al, '['
    je pp_open
    cmp al, ']'
    jne pp_loop
    cmp rsp, rbp
    je bad
    pop rcx
    lea edx, [rdi + 1]
    mov dword ptr [r11 + rcx*4], edx
    lea edx, [rcx + 1]
    mov dword ptr [r11 + rdi*4], edx
    jmp pp_store
pp_open:
    push rdi
pp_store:
    mov byte ptr [r12 + rdi], al
    inc rdi
    jmp pp_loop
pp_done:
    cmp rsp, rbp
    jne bad
    mov byte ptr [r12 + rdi], 0
    lea r8, [r12 + rsi]     # input pointer
    lea r9, [r12 + r13]     # input end
    mov rbp, r11            # jump table
    lea rbx, [rip + tape]
    lea r10, [rip + outbuf]
    xor r15d, r15d
    xor esi, esi
run:
    movzx eax, byte ptr [r12 + rsi]
    inc rsi
    cmp al, '>'
    je c_right
    cmp al, '<'
    je c_left
    cmp al, '+'
    je c_inc
    cmp al, '-'
    je c_dec
    cmp al, '['
    je c_open
    cmp al, ']'
    je c_close
    cmp al, '.'
    je c_out
    cmp al, ','
    je c_in
    # terminator
    test r15, r15
    jz 1f
    call flush
1:
    xor edi, edi
    mov eax, 60
    syscall
c_right:
    inc rbx
    jmp run
c_left:
    dec rbx
    jmp run
c_inc:
    inc byte ptr [rbx]
    jmp run
c_dec:
    dec byte ptr [rbx]
    jmp run
c_open:
    cmp byte ptr [rbx], 0
    jne run
    mov esi, dword ptr [rbp + rsi*4 - 4]
    jmp run
c_close:
    cmp byte ptr [rbx], 0
    je run
    mov esi, dword ptr [rbp + rsi*4 - 4]
    jmp run
c_out:
    mov al, byte ptr [rbx]
    mov byte ptr [r10 + r15], al
    inc r15
    cmp r15, OUTSZ
    jne run
    call flush
    jmp run
c_in:
    xor eax, eax
    cmp r8, r9
    jae 1f
    mov al, byte ptr [r8]
    inc r8
1:
    mov byte ptr [rbx], al
    jmp run

flush:
    push rsi
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r15
    syscall
    pop rsi
    xor r15d, r15d
    ret

bad:
    mov rsp, rbp
    mov edi, 1
    mov eax, 60
    syscall
