.globl _start

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.hi not_ws
is_ws:
    cmp w3, #10
    cinc x19, x19, eq
    mov x22, #0
    b next
not_ws:
    cbnz x22, next
    mov x22, #1
    add x20, x20, #1
next:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

done:
    adrp x24, outend
    add x24, x24, :lo12:outend
    mov x5, x24
    mov w9, #10
    strb w9, [x5, #-1]!
    mov x0, x21
    bl conv
    mov w9, #32
    strb w9, [x5, #-1]!
    mov x0, x20
    bl conv
    mov w9, #32
    strb w9, [x5, #-1]!
    mov x0, x19
    bl conv

write_loop:
    cmp x5, x24
    b.hs finish
    mov x0, #1
    mov x1, x5
    sub x2, x24, x5
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x5, x5, x0
    b write_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0
hang:
    b hang

// x0 = value, x5 = pointer (digits stored before it, x5 updated)
conv:
    mov x6, #10
1:
    udiv x7, x0, x6
    msub x9, x7, x6, x0
    add w9, w9, #48
    strb w9, [x5, #-1]!
    mov x0, x7
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 96
outend:
    .space 16
