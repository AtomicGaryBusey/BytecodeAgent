.globl _start
.text
_start:
    sub rsp, 32
    lea r15, [rip + inbuf]
    mov r12, r15
rd:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    lea rdx, [rip + inbuf_end]
    sub rdx, r12
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd
rd_done:
    mov rsi, r15
    lea rdi, [rip + outbuf]
    lea r13, [rip + vstack]

line_start:
    cmp rsi, r12
    jae finish
    xor r14d, r14d
tl:
    cmp rsi, r12
    jae end_line
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne 1f
    inc rsi
    jmp end_line
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jne tok
2:
    inc rsi
    jmp tl
tok:
    mov rdx, rsi
3:
    inc rdx
    cmp rdx, r12
    jae 4f
    mov cl, byte ptr [rdx]
    cmp cl, 32
    je 4f
    cmp cl, 9
    je 4f
    cmp cl, 10
    jne 3b
4:
    mov rcx, rdx
    sub rcx, rsi
    cmp rcx, 1
    jne literal
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
literal:
    mov rbx, rsi
    xor r8d, r8d
    cmp al, '-'
    jne 5f
    inc r8d
    inc rbx
5:
    mov rsi, rdx
    mov rcx, rdx
    sub rcx, rbx
    jz err
    cmp rcx, 18
    ja err
    xor eax, eax
6:
    movzx ecx, byte ptr [rbx]
    sub ecx, '0'
    cmp ecx, 9
    ja err
    lea rax, [rax + rax*4]
    lea rax, [rcx + rax*2]
    inc rbx
    cmp rbx, rdx
    jb 6b
    test r8d, r8d
    jz 7f
    neg rax
7:
    mov qword ptr [r13 + r14*8], rax
    inc r14
    jmp tl

getab:
    mov rsi, rdx
    cmp r14, 2
    jb 1f
    mov rax, qword ptr [r13 + r14*8 - 16]
    mov rcx, qword ptr [r13 + r14*8 - 8]
    clc
1:
    ret

op_add:
    call getab
    jc err
    add rax, rcx
    jmp store
op_sub:
    call getab
    jc err
    sub rax, rcx
    jmp store
op_mul:
    call getab
    jc err
    imul rax, rcx
    jmp store
op_div:
    call getab
    jc err
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne 1f
    mov rbx, 0x8000000000000000
    cmp rax, rbx
    je err
1:
    cqo
    idiv rcx
    jmp store
op_mod:
    call getab
    jc err
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne 1f
    mov rbx, 0x8000000000000000
    cmp rax, rbx
    je err
1:
    cqo
    idiv rcx
    mov rax, rdx
store:
    dec r14
    mov qword ptr [r13 + r14*8 - 8], rax
    jmp tl

err:
    cmp rsi, r12
    jae err_out
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne err
err_out:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_start

end_line:
    cmp r14, 1
    jne err_out
    mov rax, qword ptr [r13]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    lea r9, [rsp + 32]
    mov r8, r9
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, r9
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_start

finish:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
wloop:
    mov rdx, rbx
    sub rdx, rsi
    jle done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    jmp wloop
done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
    .align 16
inbuf:
    .zero 66000
inbuf_end:
    .align 16
outbuf:
    .zero 400000
    .align 16
vstack:
    .zero 8192
