.equ INSZ, 50000

    .zero 50016
outbuf:
    .zero 70000

    lea r12, [rip + inbuf]
    xor r13d, r13d
readloop:
    mov edx, INSZ
    sub rdx, r13
    jz done_read
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle done_read
    add r13, rax
    jmp readloop
done_read:
    mov rsi, r12
    lea rdi, [rip + outbuf]
    lea r8, [rip + table]
    xor r9d, r9d
encloop:
    cmp r13, 3
    jb tail
    movzx eax, byte ptr [rsi]

    movzx ecx, byte ptr [rsi + 1]

    or eax, ecx

    shr ecx, 12

    mov byte ptr [rdi + 2], cl

    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne encloop
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encloop
tail:
    test r13, r13
    jz fin
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r13, 1
    je 1f
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx

    mov ecx, eax
    shr ecx, 18
    movzx ecx, byte ptr [r8 + rcx]
    mov byte ptr [rdi], cl
    mov ecx, eax
    shr ecx, 12

    test r9d, r9d
    jz 3f
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov r14, rdi
    sub r14, rsi
wloop:
    test r14, r14
    jle done
    mov eax, 1
    mov edi, 1
    mov rdx, r14
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r14, rax
    jmp wloop
done:
    mov eax, 60
    xor edi, edi
    syscall
