.globl _start
.section .rodata
hexdigits: .ascii "0123456789abcdef"

.bss
.align 16
inbuf: .zero 131072
outbuf: .zero 400000

.text
_start:
    lea r12, [rip + inbuf]
    lea r15, [rip + hexdigits]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    cmp r13, 65536
    jb read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor ebx, ebx
row_loop:
    mov rcx, r13
    sub rcx, rbx
    jz final
    mov eax, 16
    cmp rcx, rax
    cmova rcx, rax
    mov r14, rcx
    mov eax, ebx
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [r12 + rbx]
    xor ecx, ecx
slot_loop:
    cmp rcx, r14
    jae pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp next
pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
next:
    add rdi, 3
    cmp ecx, 7
    jne 1f
    mov byte ptr [rdi], 0x20
    inc rdi
1:
    inc ecx
    cmp ecx, 16
    jb slot_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
char_loop:
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 2f
    mov al, 0x2e
2:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    cmp rcx, r14
    jb char_loop
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add rbx, r14
    jmp row_loop
final:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle exit
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle exit
    add rsi, rax
    sub rbx, rax
    jmp write_loop
exit:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret
