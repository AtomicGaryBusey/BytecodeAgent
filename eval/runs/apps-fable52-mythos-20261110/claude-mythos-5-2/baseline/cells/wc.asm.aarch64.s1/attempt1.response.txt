.text
.globl _start
_start:
    mov x19, #0          // lines
    mov x20, #0          // words
    mov x21, #0          // bytes
    mov x22, #0          // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf
read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x9, x23
    add x10, x23, x0
byte_loop:
    ldrb w11, [x9], #1
    cmp w11, #0x20
    b.eq ws
    sub w12, w11, #9
    cmp w12, #4
    b.hi nonws
    cmp w11, #0x0A
    cinc x19, x19, eq
ws:
    mov x22, #0
    b next
nonws:
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
next:
    cmp x9, x10
    b.lo byte_loop
    b read_loop

done:
    // build output backwards from end of outbuf
    adrp x13, outend
    add x13, x13, :lo12:outend
    mov x14, x13         // end pointer
    mov w11, #0x0A
    strb w11, [x13, #-1]!
    mov x15, #10
    mov x0, x21
    bl conv
    mov w11, #0x20
    strb w11, [x13, #-1]!
    mov x0, x20
    bl conv
    mov w11, #0x20
    strb w11, [x13, #-1]!
    mov x0, x19
    bl conv
    sub x24, x14, x13    // remaining length
write_loop:
    cbz x24, exit
    mov x0, #1
    mov x1, x13
    mov x2, x24
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit
    add x13, x13, x0
    sub x24, x24, x0
    b write_loop
exit:
    mov x0, #0
    mov x8, #93
    svc #0

// conv: write decimal of x0 backwards at x13 (pre-decrement); x15 = 10
conv:
1:
    udiv x16, x0, x15
    msub x17, x16, x15, x0
    add w17, w17, #'0'
    strb w17, [x13, #-1]!
    mov x0, x16
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 96
outend:
    .space 16
