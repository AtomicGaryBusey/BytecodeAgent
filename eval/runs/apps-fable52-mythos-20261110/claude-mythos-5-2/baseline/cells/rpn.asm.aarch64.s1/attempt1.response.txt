.globl _start

.text
_start:
    sub sp, sp, #32
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    movz x26, #1, lsl #16
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x27, x21                // output start
    adrp x22, vstack
    add x22, x22, :lo12:vstack

read_loop:
    cbz x26, read_done
    mov x0, #0
    mov x1, x20
    mov x2, x26
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    sub x26, x26, x0
    b read_loop
read_done:

line_start:
    cmp x19, x20
    b.hs finish
    mov x23, #0

tok_loop:
    cmp x19, x20
    b.hs end_line
    ldrb w0, [x19]
    cmp w0, #10
    b.ne 1f
    add x19, x19, #1
    b end_line
1:
    cmp w0, #32
    b.eq 2f
    cmp w0, #9
    b.ne token
2:
    add x19, x19, #1
    b tok_loop

token:
    mov x24, x19                // token start
    mov x25, x19
3:
    cmp x25, x20
    b.hs 4f
    ldrb w1, [x25]
    cmp w1, #32
    b.eq 4f
    cmp w1, #9
    b.eq 4f
    cmp w1, #10
    b.eq 4f
    add x25, x25, #1
    b 3b
4:
    mov x19, x25
    sub x2, x25, x24            // length
    ldrb w0, [x24]
    cmp x2, #1
    b.ne number
    cmp w0, #'+'
    b.eq operator
    cmp w0, #'-'
    b.eq operator
    cmp w0, #'*'
    b.eq operator
    cmp w0, #'/'
    b.eq operator
    cmp w0, #'%'
    b.eq operator

number:
    mov x3, #0                  // neg flag
    mov x4, x24                 // p
    cmp w0, #'-'
    b.ne 5f
    mov x3, #1
    add x4, x4, #1
5:
    sub x5, x25, x4             // digit count
    cmp x5, #1
    b.lt err_skip
    cmp x5, #18
    b.gt err_skip
    mov x6, #0
    mov x7, #10
6:
    ldrb w1, [x4], #1
    sub w1, w1, #'0'
    cmp w1, #9
    b.hi err_skip
    madd x6, x6, x7, x1
    cmp x4, x25
    b.lo 6b
    cbz x3, 7f
    neg x6, x6
7:
    str x6, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

operator:
    cmp x23, #2
    b.lt err_skip
    sub x23, x23, #1
    ldr x2, [x22, x23, lsl #3]  // b
    sub x9, x23, #1
    ldr x1, [x22, x9, lsl #3]   // a
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    // division or remainder
    cbz x2, err_skip
    cmn x2, #1
    b.ne 8f
    movz x10, #0x8000, lsl #48
    cmp x1, x10
    b.eq err_skip
8:
    sdiv x3, x1, x2
    cmp w0, #'/'
    b.eq 9f
    msub x3, x3, x2, x1
9:
    b op_store
op_add:
    add x3, x1, x2
    b op_store
op_sub:
    sub x3, x1, x2
    b op_store
op_mul:
    mul x3, x1, x2
op_store:
    str x3, [x22, x9, lsl #3]
    b tok_loop

err_skip:
    cmp x19, x20
    b.hs err_emit
    ldrb w0, [x19], #1
    cmp w0, #10
    b.ne err_skip
err_emit:
    movz w0, #0x7265
    movk w0, #0x6f72, lsl #16
    str w0, [x21], #4
    movz w0, #0x0a72
    strh w0, [x21], #2
    b line_start

end_line:
    cmp x23, #1
    b.ne err_emit
    ldr x0, [x22]
    cmp x0, #0
    b.ge 1f
    mov w1, #'-'
    strb w1, [x21], #1
    neg x0, x0
1:
    add x13, sp, #32
    mov x9, x13
    mov x10, #10
2:
    udiv x11, x0, x10
    msub x12, x11, x10, x0
    add w12, w12, #'0'
    strb w12, [x9, #-1]!
    mov x0, x11
    cbnz x0, 2b
3:
    ldrb w1, [x9], #1
    strb w1, [x21], #1
    cmp x9, x13
    b.lo 3b
    mov w1, #10
    strb w1, [x21], #1
    b line_start

finish:
    cmp x27, x21
    b.hs do_exit
    mov x0, #1
    mov x1, x27
    sub x2, x21, x27
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le do_exit
    add x27, x27, x0
    b finish

    mov x0, #0
    mov x8, #93
    svc #0
    b do_exit
