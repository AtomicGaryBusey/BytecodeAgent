.globl _start
.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65552
    sub rdx, r12
    jle done_read
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae finish
    mov eax, r13d
    call hex8
    mov ax, 0x2020
    stosw
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    lea rbx, [rip + inbuf]
    add rbx, r13
    xor ecx, ecx
byte_loop:
    cmp rcx, r14
    jae pad
    movzx eax, byte ptr [rbx + rcx]
    call hex2
    mov al, 0x20
    stosb
    jmp next
pad:
    mov al, 0x20
    stosb
    stosb
    stosb
next:
    cmp ecx, 7
    jne 2f
    mov al, 0x20
    stosb
2:
    inc ecx
    cmp ecx, 16
    jb byte_loop
    mov ax, 0x7c20
    stosw
    xor ecx, ecx
char_loop:
    cmp rcx, r14
    jae cdone
    movzx eax, byte ptr [rbx + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 3f
    mov al, 0x2e
3:
    stosb
    inc ecx
    jmp char_loop
cdone:
    mov ax, 0x0a7c
    stosw
    add r13, r14
    jmp row_loop

finish:
    mov eax, r12d
    call hex8
    mov al, 10
    stosb
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle exit_ok
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rbx, rax
    jmp write_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

# eax -> 8 hex digits at rdi; clobbers eax, edx, esi
hex8:
    mov edx, eax
    mov esi, 8
1:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [hexdigits + rax]
    stosb
    dec esi
    jnz 1b
    ret

# al (zero-extended in eax) -> 2 hex digits at rdi; clobbers eax, edx
hex2:
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [hexdigits + rax]
    stosb
    and edx, 15
    mov al, byte ptr [hexdigits + rdx]
    stosb
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .space 65600
outbuf:
    .space 340000
