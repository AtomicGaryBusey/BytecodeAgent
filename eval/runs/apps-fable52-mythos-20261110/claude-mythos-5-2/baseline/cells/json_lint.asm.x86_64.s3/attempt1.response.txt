.globl _start

.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
buf: .zero 131072
stk: .zero 4096
numbuf: .zero 32

.text
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 131072
    sub rdx, r13
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    mov rsi, r12
    lea r15, [r12 + r13]
    xor ebx, ebx
    lea r14, [rip + stk]

parse_value:
    call getc
    cmp al, '{'
    je pv_obj
    cmp al, '['
    je pv_arr
    cmp al, '"'
    je pv_str
    lea rdi, [rip + lit_true]
    cmp al, 't'
    je pv_lit
    lea rdi, [rip + lit_false]
    cmp al, 'f'
    je pv_lit
    lea rdi, [rip + lit_null]
    cmp al, 'n'
    je pv_lit
    cmp al, '-'
    jne num_int
    inc rsi
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
num_int:
    cmp al, '0'
    jne 1f
    inc rsi
    jmp num_frac
1:
    sub al, '1'
    cmp al, 8
    ja error
    inc rsi
    call digits
num_frac:
    cmp rsi, r15
    jae value_done
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne num_exp
    inc rsi
    call digit1
    call digits
num_exp:
    cmp rsi, r15
    jae value_done
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne value_done
    inc rsi
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 1f
    cmp al, '-'
    jne 2f
1:
    inc rsi
2:
    call digit1
    call digits
    jmp value_done

pv_lit:
    mov dl, byte ptr [rdi]
    test dl, dl
    jz value_done
    cmp rsi, r15
    jae error
    cmp dl, byte ptr [rsi]
    jne error
    inc rsi
    inc rdi
    jmp pv_lit

pv_str:
    call parse_string
    jmp value_done

pv_obj:
    inc rsi
    call getc
    cmp al, '}'
    jne 1f
    inc rsi
    jmp value_done
1:
    mov byte ptr [r14 + rbx], 1
    inc rbx
    jmp parse_member

pv_arr:
    inc rsi
    call getc
    cmp al, ']'
    jne 1f
    inc rsi
    jmp value_done
1:
    mov byte ptr [r14 + rbx], 0
    inc rbx
    jmp parse_value

parse_member:
    call getc
    cmp al, '"'
    jne error
    call parse_string
    call getc
    cmp al, ':'
    jne error
    inc rsi
    jmp parse_value

value_done:
    test rbx, rbx
    jnz 1f
    call skipws
    cmp rsi, r15
    jne error
    # ok
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall
1:
    call getc
    movzx edx, byte ptr [r14 + rbx - 1]
    cmp al, ','
    jne 2f
    inc rsi
    test edx, edx
    jnz parse_member
    jmp parse_value
2:
    test edx, edx
    jnz 3f
    cmp al, ']'
    jne error
    jmp 4f
3:
    cmp al, '}'
    jne error
4:
    inc rsi
    dec rbx
    jmp value_done

# require one digit at rsi
digit1:
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
    inc rsi
    ret

# skip any digits
digits:
    cmp rsi, r15
    jae 1f
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja 1f
    inc rsi
    jmp digits
1:
    ret

skipws:
    cmp rsi, r15
    jae 2f
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
2:
    ret
1:
    inc rsi
    jmp skipws

getc:
    call skipws
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
    ret

parse_string:
    inc rsi
ps_loop:
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb error
    cmp al, '"'
    je ps_end
    cmp al, 0x5C
    je ps_esc
    inc rsi
    jmp ps_loop
ps_end:
    inc rsi
    ret
ps_esc:
    inc rsi
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je ps_ok
    cmp al, 0x5C
    je ps_ok
    cmp al, '/'
    je ps_ok
    cmp al, 'b'
    je ps_ok
    cmp al, 'f'
    je ps_ok
    cmp al, 'n'
    je ps_ok
    cmp al, 'r'
    je ps_ok
    cmp al, 't'
    je ps_ok
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
ps_hex:
    cmp rsi, r15
    jae error
    movzx eax, byte ptr [rsi]
    mov edx, eax
    sub dl, '0'
    cmp dl, 9
    jbe 1f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
1:
    inc rsi
    dec ecx
    jnz ps_hex
    jmp ps_loop
ps_ok:
    inc rsi
    jmp ps_loop

error:
    mov rax, rsi
    sub rax, r12
    lea r8, [rip + numbuf + 31]
    mov byte ptr [r8], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    lea rdx, [rip + numbuf + 32]
    sub rdx, r8
    syscall
    mov eax, 60
    mov edi, 1
    syscall
