.globl _start

.equ BUFSZ, 65600

.bss
.balign 16
inbuf:
    .space 65600
outbuf:
    .space 65664

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, BUFSZ
    sub rdx, r13
    jz done_read
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle done_read
    add r13, rax
    jmp read_loop
done_read:
    lea r14, [rip + outbuf]
    mov rdi, r14
    xor ebx, ebx
process:
    cmp rbx, r13
    jae flush
    mov rcx, rbx
find:
    cmp rcx, r13
    jae found
    cmp byte ptr [r12 + rcx], 10
    je found
    inc rcx
    jmp find
found:
    mov rdx, rcx
rev:
    cmp rdx, rbx
    jbe revdone
    dec rdx
    mov al, byte ptr [r12 + rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev
revdone:
    mov byte ptr [rdi], 10
    inc rdi
    lea rbx, [rcx + 1]
    jmp process
flush:
    mov r15, rdi
wloop:
    mov rdx, r15
    sub rdx, r14
    jz finish
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall
    test rax, rax
    jle finish
    add r14, rax
    jmp wloop
finish:
    mov eax, 60
    xor edi, edi
    syscall
