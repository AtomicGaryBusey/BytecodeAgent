.globl _start

.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
.align 16
buf: .zero 69632
nstack: .zero 2048
outbuf: .zero 32

.text
_start:
    lea rbx, [rip + buf]
    mov rsi, rbx
read_loop:
    lea rdx, [rip + buf + 69631]
    sub rdx, rsi
    jle read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    jmp read_loop
read_done:
    mov rdi, rsi
    mov rsi, rbx
    xor r12d, r12d
    lea r13, [rip + nstack]

value:
    call skipws
    cmp al, '{'
    je obj_open
    cmp al, '['
    je arr_open
    cmp al, '"'
    je val_string
    cmp al, '-'
    je val_minus
    cmp al, 't'
    je val_true
    cmp al, 'f'
    je val_false
    cmp al, 'n'
    je val_null
    jmp number

val_string:
    inc rsi
    call string
    jmp after_value

val_true:
    lea rdx, [rip + lit_true]
    jmp lit
val_false:
    lea rdx, [rip + lit_false]
    jmp lit
val_null:
    lea rdx, [rip + lit_null]
lit:
    mov cl, byte ptr [rdx]
    test cl, cl
    jz after_value
    cmp cl, byte ptr [rsi]
    jne error
    inc rsi
    inc rdx
    jmp lit

val_minus:
    inc rsi
    movzx eax, byte ptr [rsi]
number:
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
1:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    jbe 1b
    jmp frac
num_zero:
    inc rsi
frac:
    cmp byte ptr [rsi], '.'
    jne exp
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
1:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    jbe 1b
exp:
    mov al, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne after_value
    inc rsi
    mov al, byte ptr [rsi]
    cmp al, '+'
    je 2f
    cmp al, '-'
    jne 3f
2:
    inc rsi
3:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
1:
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    jbe 1b
    jmp after_value

obj_open:
    inc rsi
    mov byte ptr [r13 + r12], 1
    inc r12
    call skipws
    cmp al, '}'
    jne member
    inc rsi
    dec r12
    jmp after_value

member:
    call skipws
    cmp al, '"'
    jne error
    inc rsi
    call string
    call skipws
    cmp al, ':'
    jne error
    inc rsi
    jmp value

arr_open:
    inc rsi
    mov byte ptr [r13 + r12], 0
    inc r12
    call skipws
    cmp al, ']'
    jne value
    inc rsi
    dec r12
    jmp after_value

after_value:
    call skipws
    test r12, r12
    jz top
    cmp byte ptr [r13 + r12 - 1], 0
    jne in_obj
    cmp al, ','
    jne 1f
    inc rsi
    jmp value
1:
    cmp al, ']'
    jne error
    inc rsi
    dec r12
    jmp after_value
in_obj:
    cmp al, ','
    jne 1f
    inc rsi
    jmp member
1:
    cmp al, '}'
    jne error
    inc rsi
    dec r12
    jmp after_value

top:
    cmp rsi, rdi
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov r14, rsi
    sub r14, rbx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov rax, r14
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

# skip whitespace; returns current byte in eax
skipws:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    jne 2f
1:
    inc rsi
    jmp skipws
2:
    ret

# parse string body; rsi just after opening quote
string:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je str_done
    cmp al, 0x5C
    jne string
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
1:
    movzx eax, byte ptr [rsi]
    lea edx, [rax - '0']
    cmp dl, 9
    jbe 2f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
2:
    inc rsi
    dec ecx
    jnz 1b
    jmp string
str_esc_ok:
    inc rsi
    jmp string
str_done:
    ret
