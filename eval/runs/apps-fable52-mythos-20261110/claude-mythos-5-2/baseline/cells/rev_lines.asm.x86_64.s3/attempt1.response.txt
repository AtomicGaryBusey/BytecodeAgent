.globl _start

.equ BUFSZ, 131072

.bss
.balign 16
buf:
    .zero BUFSZ + 16

.text
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, BUFSZ
    sub rdx, r13
    jz done_read
    syscall
    test rax, rax
    jle done_read
    add r13, rax
    jmp read_loop
done_read:
    test r13, r13
    jz do_exit
    cmp byte ptr [r12 + r13 - 1], 10
    je 1f
    mov byte ptr [r12 + r13], 10
    inc r13
1:
    mov rsi, r12
    lea r14, [r12 + r13]
line_loop:
    cmp rsi, r14
    jae do_write
    mov rdi, rsi
2:
    cmp byte ptr [rdi], 10
    je 3f
    inc rdi
    jmp 2b
3:
    lea rbx, [rdi + 1]
    dec rdi
4:
    cmp rsi, rdi
    jae 5f
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp 4b
5:
    mov rsi, rbx
    jmp line_loop
do_write:
    mov rsi, r12
    mov rdx, r13
write_loop:
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
