.globl _start

.section .rodata
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"
s_ok:    .ascii "ok\n"
s_err:   .ascii "error at "

.bss
.balign 16
buf:    .space 0x11000
stk:    .space 2048
numbuf: .space 32

.text
_start:
    adrp x21, buf
    add x21, x21, :lo12:buf
    mov x19, x21
    movz x24, #0x1000
    movk x24, #0x1, lsl #16
    sub x24, x24, #16
    mov x20, #0
rd_loop:
    mov x0, #0
    add x1, x21, x20
    sub x2, x24, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x20, x20, x0
    b rd_loop
rd_done:
    add x20, x21, x20
    adrp x23, stk
    add x23, x23, :lo12:stk
    mov x22, x23

parse_value:
    bl skipws
    ldrb w0, [x19]
    cmp w0, #123
    b.eq v_obj
    cmp w0, #91
    b.eq v_arr
    cmp w0, #34
    b.eq v_str
    cmp w0, #45
    b.eq v_num
    sub w1, w0, #48
    cmp w1, #9
    b.ls v_num
    cmp w0, #116
    b.eq v_true
    cmp w0, #102
    b.eq v_false
    cmp w0, #110
    b.eq v_null
    b error

v_obj:
    add x19, x19, #1
    bl skipws
    ldrb w0, [x19]
    cmp w0, #125
    b.ne 1f
    add x19, x19, #1
    b value_done
1:
    mov w1, #1
    strb w1, [x22], #1
    b member_key

v_arr:
    add x19, x19, #1
    bl skipws
    ldrb w0, [x19]
    cmp w0, #93
    b.ne 1f
    add x19, x19, #1
    b value_done
1:
    mov w1, #0
    strb w1, [x22], #1
    b parse_value

v_str:
    bl string
    b value_done

v_num:
    bl number
    b value_done

v_true:
    adrp x1, s_true
    add x1, x1, :lo12:s_true
    b literal
v_false:
    adrp x1, s_false
    add x1, x1, :lo12:s_false
    b literal
v_null:
    adrp x1, s_null
    add x1, x1, :lo12:s_null
literal:
    ldrb w2, [x1], #1
    cbz w2, value_done
    ldrb w0, [x19]
    cmp w0, w2
    b.ne error
    add x19, x19, #1
    b literal

member_key:
    bl skipws
    ldrb w0, [x19]
    cmp w0, #34
    b.ne error
    bl string
    bl skipws
    ldrb w0, [x19]
    cmp w0, #58
    b.ne error
    add x19, x19, #1
    b parse_value

value_done:
    bl skipws
    cmp x22, x23
    b.eq top_done
    ldrb w0, [x19]
    ldurb w1, [x22, #-1]
    cmp w0, #44
    b.ne 1f
    add x19, x19, #1
    cbnz w1, member_key
    b parse_value
1:
    mov w2, #93
    mov w3, #125
    cmp w1, #0
    csel w2, w3, w2, ne
    cmp w0, w2
    b.ne error
    add x19, x19, #1
    sub x22, x22, #1
    b value_done

top_done:
    cmp x19, x20
    b.ne error
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, s_err
    add x1, x1, :lo12:s_err
    mov x2, #9
    mov x8, #64
    svc #0
    sub x0, x19, x21
    adrp x5, numbuf
    add x5, x5, :lo12:numbuf
    add x1, x5, #31
    mov w4, #10
    strb w4, [x1]
    mov x2, #10
2:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add w4, w4, #48
    strb w4, [x1, #-1]!
    mov x0, x3
    cbnz x0, 2b
    add x2, x5, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip whitespace at x19
skipws:
    ldrb w0, [x19]
    cmp w0, #32
    b.eq 1f
    cmp w0, #9
    b.eq 1f
    cmp w0, #10
    b.eq 1f
    cmp w0, #13
    b.eq 1f
    ret
1:
    add x19, x19, #1
    b skipws

// string: x19 at opening quote
string:
    add x19, x19, #1
str_loop:
    cmp x19, x20
    b.hs error
    ldrb w0, [x19]
    cmp w0, #34
    b.eq str_end
    cmp w0, #92
    b.eq str_esc
    cmp w0, #32
    b.lo error
    add x19, x19, #1
    b str_loop
str_end:
    add x19, x19, #1
    ret
str_esc:
    add x19, x19, #1
    ldrb w0, [x19]
    cmp w0, #34
    b.eq str_adv
    cmp w0, #92
    b.eq str_adv
    cmp w0, #47
    b.eq str_adv
    cmp w0, #98
    b.eq str_adv
    cmp w0, #102
    b.eq str_adv
    cmp w0, #110
    b.eq str_adv
    cmp w0, #114
    b.eq str_adv
    cmp w0, #116
    b.eq str_adv
    cmp w0, #117
    b.ne error
    add x19, x19, #1
    mov w5, #4
str_hex:
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls 1f
    orr w1, w0, #0x20
    sub w1, w1, #97
    cmp w1, #5
    b.hi error
1:
    add x19, x19, #1
    subs w5, w5, #1
    b.ne str_hex
    b str_loop
str_adv:
    add x19, x19, #1
    b str_loop

// number: x19 at '-' or digit
number:
    ldrb w0, [x19]
    cmp w0, #45
    b.ne 1f
    add x19, x19, #1
    ldrb w0, [x19]
1:
    cmp w0, #48
    b.ne 2f
    add x19, x19, #1
    b num_frac
2:
    sub w1, w0, #49
    cmp w1, #8
    b.hi error
3:
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls 3b
num_frac:
    ldrb w0, [x19]
    cmp w0, #46
    b.ne num_exp
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
4:
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls 4b
num_exp:
    ldrb w0, [x19]
    orr w1, w0, #0x20
    cmp w1, #101
    b.ne num_done
    add x19, x19, #1
    ldrb w0, [x19]
    cmp w0, #43
    b.eq 5f
    cmp w0, #45
    b.ne 6f
5:
    add x19, x19, #1
    ldrb w0, [x19]
6:
    sub w1, w0, #48
    cmp w1, #9
    b.hi error
7:
    add x19, x19, #1
    ldrb w0, [x19]
    sub w1, w0, #48
    cmp w1, #9
    b.ls 7b
num_done:
    ret
