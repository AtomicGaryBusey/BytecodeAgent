.globl _start

.equ BUFSZ, 0x11000

.section .rodata
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"
s_ok:    .ascii "ok\n"
s_err:   .ascii "error at "

.bss
.balign 16
buf:    .space BUFSZ
stk:    .space 2048
numbuf: .space 32

.text
_start:
    adrp x21, buf
    add x21, x21, :lo12:buf
    mov x19, x21
    mov x24, #BUFSZ
    add x24, x21
