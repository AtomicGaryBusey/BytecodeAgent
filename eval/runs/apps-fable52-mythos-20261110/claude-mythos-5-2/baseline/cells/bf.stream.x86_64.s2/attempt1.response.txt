.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
rd:
    mov edx, 65536
    sub edx, ebx
    jz rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd
rd_done:
    xor ecx, ecx
find:
    cmp rcx, rbx
    jae notfound
    cmp byte ptr [r12 + rcx], 0x21
    je found
    inc rcx
    jmp find
found:
    mov r13, rcx
    lea r14, [r12 + rcx + 1]
    lea r15, [r12 + rbx]
    jmp brk
notfound:
    mov r13, rbx
    lea r14, [r12 + rbx]
    mov r15, r14
brk:
    xor ecx, ecx
    xor edx, edx
    lea r8, [rip + jmptab]
    lea r9, [rip + stk]
bloop:
    cmp rcx, r13
    jae bend
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    mov dword ptr [r9 + rdx*4], ecx
    inc edx
    jmp bnext
1:
    cmp al, 0x5D
    jne bnext
    test edx, edx
    jz bad
    dec edx
    mov eax, dword ptr [r9 + rdx*4]
    mov dword ptr [r8 + rcx*4], eax
    mov dword ptr [r8 + rax*4], ecx
bnext:
    inc rcx
    jmp bloop
bend:
    test edx, edx
    jnz bad

    xor ecx, ecx
    xor ebx, ebx
    lea rbp, [rip + tape]
    lea r10, [rip + outbuf]
run:
    cmp rcx, r13
    jae fin
    movzx eax, byte ptr [r12 + rcx]
    inc rcx
    cmp al, 0x3E
    je c_right
    cmp al, 0x3C
    je c_left
    cmp al, 0x2B
    je c_inc
    cmp al, 0x2D
    je c_dec
    cmp al, 0x5B
    je c_open
    cmp al, 0x5D
    je c_close
    cmp al, 0x2E
    je c_out
    cmp al, 0x2C
    je c_in
    jmp run
c_right:
    inc rbp
    jmp run
c_left:
    dec rbp
    jmp run
c_inc:
    inc byte ptr [rbp]
    jmp run
c_dec:
    dec byte ptr [rbp]
    jmp run
c_open:
    cmp byte ptr [rbp], 0
    jne run
    mov ecx, dword ptr [r8 + rcx*4 - 4]
    inc rcx
    jmp run
c_close:
    cmp byte ptr [rbp], 0
    je run
    mov ecx, dword ptr [r8 + rcx*4 - 4]
    inc rcx
    jmp run
c_out:
    mov al, byte ptr [rbp]
    mov byte ptr [r10 + rbx], al
    inc rbx
    cmp rbx, 65536
    jb run
    call flush
    jmp run
c_in:
    xor eax, eax
    cmp r14, r15
    jae 1f
    mov al, byte ptr [r14]
    inc r14
1:
    mov byte ptr [rbp], al
    jmp run
fin:
    call flush
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall

flush:
    test rbx, rbx
    jz 1f
    push rcx
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, rbx
    syscall
    xor ebx, ebx
    pop rcx
1:
    ret

.bss
.align 16
inbuf:
    .zero 65552
jmptab:
    .zero 262144
stk:
    .zero 262144
tape:
    .zero 30016
outbuf:
    .zero 65536
