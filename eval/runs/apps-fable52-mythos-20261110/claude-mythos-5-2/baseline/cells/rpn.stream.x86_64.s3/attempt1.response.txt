.globl _start
.text
_start:
    xor r12d, r12d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 66000
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r13, [rsi + r12]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + vstack]
    mov r10d, 10
main_loop:
    cmp rsi, r13
    jae finish
    xor r14d, r14d
token_loop:
    cmp rsi, r13
    jae line_end
    mov al, byte ptr [rsi]
    cmp al, 10
    je line_end_nl
    cmp al, 32
    je 1f
    cmp al, 9
    jne token_start
1:
    inc rsi
    jmp token_loop
token_start:
    mov rcx, rsi
2:
    inc rcx
    cmp rcx, r13
    jae 3f
    mov al, byte ptr [rcx]
    cmp al, 10
    je 3f
    cmp al, 32
    je 3f
    cmp al, 9
    jne 2b
3:
    mov rdx, rcx
    sub rdx, rsi
    cmp rdx, 1
    jne number
    mov al, byte ptr [rsi]
    cmp al, '+'
    je op_common
    cmp al, '-'
    je op_common
    cmp al, '*'
    je op_common
    cmp al, '/'
    je op_common
    cmp al, '%'
    je op_common
number:
    mov r8, rsi
    xor r9d, r9d
    cmp byte ptr [r8], '-'
    jne 4f
    inc r9d
    inc r8
4:
    mov rdx, rcx
    sub rdx, r8
    jz err_skip
    cmp rdx, 18
    ja err_skip
    xor eax, eax
5:
    movzx edx, byte ptr [r8]
    sub edx, '0'
    cmp edx, 9
    ja err_skip
    imul rax, rax, 10
    add rax, rdx
    inc r8
    cmp r8, rcx
    jb 5b
    test r9d, r9d
    jz 6f
    neg rax
6:
    mov qword ptr [rbx + r14*8], rax
    inc r14
    mov rsi, rcx
    jmp token_loop

op_common:
    cmp r14, 2
    jb err_skip
    mov rsi, rcx
    dec r14
    mov r8, qword ptr [rbx + r14*8]
    mov r9, qword ptr [rbx + r14*8 - 8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    # division or remainder
    test r8, r8
    jz err_skip
    cmp r8, -1
    jne 7f
    mov r11, 0x8000000000000000
    cmp r9, r11
    je err_skip
7:
    mov r11b, al
    mov rax, r9
    cqo
    idiv r8
    cmp r11b, '/'
    je 8f
    mov rax, rdx
8:
    mov r9, rax
    jmp op_store
op_add:
    add r9, r8
    jmp op_store
op_sub:
    sub r9, r8
    jmp op_store
op_mul:
    imul r9, r8
op_store:
    mov qword ptr [rbx + r14*8 - 8], r9
    jmp token_loop

line_end_nl:
    inc rsi
line_end:
    cmp r14, 1
    jne err_out
    mov rax, qword ptr [rbx]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    lea r8, [rip + tmpbuf + 24]
    mov r9, r8
2:
    xor edx, edx
    div r10
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp main_loop

err_skip:
    cmp rsi, r13
    jae err_out
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne err_skip
err_out:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp main_loop

finish:
    lea rsi, [rip + outbuf]
    mov r12, rdi
    sub r12, rsi
wloop:
    test r12, r12
    jle done
    mov eax, 1
    mov edi, 1
    mov rdx, r12
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r12, rax
    jmp wloop
done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
    .balign 16
tmpbuf:
    .zero 32
vstack:
    .zero 264000
inbuf:
    .zero 66016
outbuf:
    .zero 400000
