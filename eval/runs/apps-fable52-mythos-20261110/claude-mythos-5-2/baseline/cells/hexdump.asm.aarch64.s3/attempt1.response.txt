.globl _start

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x23, x22
    adrp x24, hextab
    add x24, x24, :lo12:hextab
    mov x20, #0
readloop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #2, lsl #16
    sub x2, x2, x20
    cbz x2, readdone
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le readdone
    add x20, x20, x0
    b readloop
readdone:
    mov x21, #0
rowloop:
    cmp x21, x20
    b.hs fin
    sub x25, x20, x21
    mov x9, #16
    cmp x25, x9
    csel x25, x25, x9, lo
    mov w0, w21
    bl puthex8
    mov w9, #32
    strb w9, [x23], #1
    strb w9, [x23], #1
    add x26, x19, x21
    mov x10, #0
byteloop:
    cmp x10, x25
    b.hs pad
    ldrb w11, [x26, x10]
    lsr w12, w11, #4
    ldrb w12, [x24, x12]
    strb w12, [x23], #1
    and w12, w11, #15
    ldrb w12, [x24, x12]
    strb w12, [x23], #1
    strb w9, [x23], #1
    b next
pad:
    strb w9, [x23], #1
    strb w9, [x23], #1
    strb w9, [x23], #1
next:
    cmp x10, #7
    b.ne 1f
    strb w9, [x23], #1
1:
    add x10, x10, #1
    cmp x10, #16
    b.lo byteloop
    strb w9, [x23], #1
    mov w12, #124
    strb w12, [x23], #1
    mov x10, #0
    mov w14, #46
charloop:
    ldrb w11, [x26, x10]
    sub w13, w11, #0x20
    cmp w13, #0x5e
    csel w11, w11, w14, ls
    strb w11, [x23], #1
    add x10, x10, #1
    cmp x10, x25
    b.lo charloop
    strb w12, [x23], #1
    mov w12, #10
    strb w12, [x23], #1
    add x21, x21, x25
    sub x0, x23, x22
    cmp x0, #3968
    b.lo rowloop
    bl flush
    b rowloop
fin:
    mov w0, w20
    bl puthex8
    mov w12, #10
    strb w12, [x23], #1
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0
2:
    b 2b

// w0 = value; appends 8 hex digits at x23
puthex8:
    mov w1, #28
1:
    lsr w2, w0, w1
    and w2, w2, #15
    ldrb w2, [x24, x2]
    strb w2, [x23], #1
    subs w1, w1, #4
    b.ge 1b
    ret

flush:
    mov x3, x22
    sub x4, x23, x22
1:
    cmp x4, #0
    b.le 2f
    mov x0, #1
    mov x1, x3
    mov x2, x4
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x3, x3, x0
    sub x4, x4, x0
    b 1b
2:
    mov x23, x22
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss

inbuf:

outbuf:
