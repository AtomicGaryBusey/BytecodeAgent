.globl _start
.equ INSZ, 65552
.equ OUTSZ, 400000

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero INSZ
outbuf:
    .zero OUTSZ

.text
# al = byte -> two hex digits at rdi; clobbers rax, rdx
hex2:
    movzx eax, al
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi + 1], al
    add rdi, 2
    ret

# eax = value -> 8 hex digits at rdi; clobbers rax, rdx, rsi, r8
hex8:
    mov esi, eax
    mov r8d, 4
1:
    rol esi, 8
    mov eax, esi
    call hex2
    dec r8d
    jnz 1b
    ret

_start:
    lea r12, [rip + inbuf]
    lea r9, [rip + hextab]
    xor r13d, r13d
read_loop:
    mov edx, INSZ
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor r14d, r14d
row:
    cmp r14, r13
    jae final
    mov eax, r14d
    call hex8
    mov ax, 0x2020
    stosw
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    lea rbx, [r12 + r14]
    xor ecx, ecx
byte_loop:
    cmp rcx, r15
    jae pad
    mov al, byte ptr [rbx + rcx]
    call hex2
    mov al, ' '
    stosb
    jmp nxt
pad:
    mov al, ' '
    stosb
    stosb
    stosb
nxt:
    cmp ecx, 7
    jne 1f
    mov al, ' '
    stosb
1:
    inc ecx
    cmp ecx, 16
    jb byte_loop
    mov ax, 0x7c20
    stosw
    xor ecx, ecx
char_loop:
    cmp rcx, r15
    jae char_end
    mov al, byte ptr [rbx + rcx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    jbe put
dot:
    mov al, '.'
put:
    stosb
    inc rcx
    jmp char_loop
char_end:
    mov ax, 0x0a7c
    stosw
    add r14, r15
    jmp row
final:
    mov eax, r13d
    call hex8
    mov al, 10
    stosb
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle finish
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
