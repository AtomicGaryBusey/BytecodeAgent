.globl _start
.equ INCAP, 66000
.equ OUTCAP, 65536

.text
_start:
    xor ebx, ebx
rd:
    mov edx, INCAP
    sub edx, ebx
    jz rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [inbuf + rbx]
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd
rd_done:
    mov r14, rbx
    xor ecx, ecx
findb:
    cmp rcx, rbx
    jae notfound
    cmp byte ptr [inbuf + rcx], 0x21
    je found
    inc rcx
    jmp findb
notfound:
    mov r12, rbx
    mov r13, rbx
    jmp chk
found:
    mov r12, rcx
    lea r13, [rcx + 1]
chk:
    xor ecx, ecx
    xor r8d, r8d
chk_loop:
    cmp rcx, r12
    jae chk_done
    mov al, byte ptr [inbuf + rcx]
    cmp al, 0x5b
    jne 1f
    mov dword ptr [stk + r8*4], ecx
    inc r8
    jmp chk_next
1:
    cmp al, 0x5d
    jne chk_next
    test r8, r8
    jz bad
    dec r8
    mov edx, dword ptr [stk + r8*4]
    mov dword ptr [jtab + rcx*4], edx
    mov dword ptr [jtab + rdx*4], ecx
chk_next:
    inc rcx
    jmp chk_loop
chk_done:
    test r8, r8
    jnz bad

    xor esi, esi
    lea rbp, [rip + tape]
    xor r15d, r15d
run:
    cmp rsi, r12
    jae fin
    movzx eax, byte ptr [inbuf + rsi]
    cmp al, 0x3e
    je op_right
    cmp al, 0x3c
    je op_left
    cmp al, 0x2b
    je op_inc
    cmp al, 0x2d
    je op_dec
    cmp al, 0x5b
    je op_open
    cmp al, 0x5d
    je op_close
    cmp al, 0x2e
    je op_out
    cmp al, 0x2c
    je op_in
next:
    inc rsi
    jmp run
op_right:
    inc rbp
    jmp next
op_left:
    dec rbp
    jmp next
op_inc:
    inc byte ptr [rbp]
    jmp next
op_dec:
    dec byte ptr [rbp]
    jmp next
op_open:
    cmp byte ptr [rbp], 0
    jne next
    mov esi, dword ptr [jtab + rsi*4]
    jmp next
op_close:
    cmp byte ptr [rbp], 0
    je next
    mov esi, dword ptr [jtab + rsi*4]
    jmp next
op_out:
    mov al, byte ptr [rbp]
    mov byte ptr [outbuf + r15], al
    inc r15
    cmp r15, OUTCAP
    jb next
    call flush
    jmp next
op_in:
    xor eax, eax
    cmp r13, r14
    jae 1f
    mov al, byte ptr [inbuf + r13]
    inc r13
1:
    mov byte ptr [rbp], al
    jmp next

fin:
    call flush
    xor edi, edi
    mov eax, 60
    syscall

bad:
    mov edi, 1
    mov eax, 60
    syscall

flush:
    push rsi
    push rbx
    xor ebx, ebx
1:
    cmp rbx, r15
    jae 2f
    mov eax, 1
    mov edi, 1
    lea rsi, [outbuf + rbx]
    mov rdx, r15
    sub rdx, rbx
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    xor r15d, r15d
    pop rbx
    pop rsi
    ret

.bss
.balign 16
inbuf:
    .zero 66016
.balign 16
jtab:
    .zero 264064
.balign 16
stk:
    .zero 264064
.balign 16
tape:
    .zero 30016
.balign 16
outbuf:
    .zero 65552
