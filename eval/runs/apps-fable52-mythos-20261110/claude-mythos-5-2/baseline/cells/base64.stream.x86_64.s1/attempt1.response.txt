.globl _start

.section .rodata
tbl:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
inbuf:
    .space 65536
outbuf:
    .space 70000

.text
_start:
    lea r14, [rip + inbuf]
    xor r12d, r12d              # total length
read_loop:
    mov edx, 65536
    sub rdx, r12
    jle read_done
    lea rsi, [r14 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov rsi, r14
    lea rdi, [rip + outbuf]
    lea rbx, [rip + tbl]
    xor r8d, r8d
enc_loop:
    cmp r12, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3
    sub r12, 3
    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl
    and eax, 63
    movzx edx, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], dl
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp enc_loop
tail:
    test r12, r12
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r12, 2
    jne 1f
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
1:
    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 61
    cmp r12, 2
    jne 2f
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl
2:
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r8d, 4
finish:
    test r8d, r8d
    jz 3f
    mov byte ptr [rdi], 10
    inc rdi
3:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
write_loop:
    test r13, r13
    jle done
    mov rdx, r13
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r13, rax
    jmp write_loop
done:
    xor edi, edi
    mov eax, 60
    syscall
