.globl _start
.equ INSZ, 40000

.bss
.align 16
inbuf: .zero 40000
.align 16
outbuf: .zero 40000
.align 16
lines: .zero 32768

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INSZ
    sub rdx, r13
    jbe 2f
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    # parse lines
    mov rsi, r12
    lea r14, [r12 + r13]
    lea rbx, [rip + lines]
    xor r15d, r15d
3:
    cmp rsi, r14
    jae 6f
    mov rdx, rsi
4:
    cmp rsi, r14
    jae 5f
    cmp byte ptr [rsi], 10
    je 5f
    inc rsi
    jmp 4b
5:
    mov rax, r15
    shl rax, 4
    mov qword ptr [rbx + rax], rdx
    mov rcx, rsi
    sub rcx, rdx
    mov qword ptr [rbx + rax + 8], rcx
    inc r15
    inc rsi
    jmp 3b
6:
    # insertion sort
    mov r8d, 1
7:
    cmp r8, r15
    jae 12f
    mov rax, r8
    shl rax, 4
    mov r10, qword ptr [rbx + rax]
    mov r11, qword ptr [rbx + rax + 8]
    mov r9, r8
8:
    test r9, r9
    jz 10f
    mov rax, r9
    shl rax, 4
    mov rsi, qword ptr [rbx + rax - 16]
    mov rdx, qword ptr [rbx + rax - 8]
    mov rcx, rdx
    cmp rcx, r11
    cmova rcx, r11
    mov rdi, r10
    cmp eax, eax
    repe cmpsb
    ja 9f
    jb 10f
    cmp rdx, r11
    jbe 10f
9:
    movups xmm0, xmmword ptr [rbx + rax - 16]
    movups xmmword ptr [rbx + rax], xmm0
    dec r9
    jmp 8b
10:
    mov rax, r9
    shl rax, 4
    mov qword ptr [rbx + rax], r10
    mov qword ptr [rbx + rax + 8], r11
    inc r8
    jmp 7b
12:
    # build output
    lea rdi, [rip + outbuf]
    xor r8d, r8d
13:
    cmp r8, r15
    jae 14f
    mov rax, r8
    shl rax, 4
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r8
    jmp 13b
14:
    lea r12, [rip + outbuf]
    mov r13, rdi
    sub r13, r12
15:
    test r13, r13
    jle 16f
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle 16f
    add r12, rax
    sub r13, rax
    jmp 15b
16:
    mov eax, 60
    xor edi, edi
    syscall
