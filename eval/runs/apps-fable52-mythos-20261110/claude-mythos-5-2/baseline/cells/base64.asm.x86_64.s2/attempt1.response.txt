.section .rodata

    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .align 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072

.text

    lea r12, [rip + inbuf]
    xor ebx, ebx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov rsi, r12
    lea rdi, [rip + outbuf]
    lea r8, [rip + table]
    xor r9d, r9d
enc_loop:
    cmp rbx, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3
    sub rbx, 3
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl
    and eax, 63
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], al
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp enc_loop
tail:
    test rbx, rbx
    jz fin
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp rbx, 2
    jb 1f
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
1:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 61
    cmp rbx, 2
    jb 2f
    shr eax, 6
    and eax, 63
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], al
2:
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r9d, 4
fin:
    test r9d, r9d
    jz 3f
    mov byte ptr [rdi], 10
    inc rdi
3:
    lea r13, [rip + outbuf]
    mov r14, rdi
    sub r14, r13
write_loop:
    test r14, r14
    jle done
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, r14
    syscall
    test rax, rax
    jle done
    add r13, rax
    sub r14, rax
    jmp write_loop
done:
    mov eax, 60
    xor edi, edi
    syscall
