.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd
rd_done:
    mov rsi, r12
    lea rbp, [r12 + r13]
    lea rbx, [rip + tab]
    xor r14d, r14d
pl:
    cmp rsi, rbp
    jae sort
    mov rdi, rsi
scan:
    cmp rsi, rbp
    jae eol
    cmp byte ptr [rsi], 10
    je eol
    inc rsi
    jmp scan
eol:
    mov rax, r14
    shl rax, 4
    mov qword ptr [rbx + rax], rdi
    mov rcx, rsi
    sub rcx, rdi
    mov qword ptr [rbx + rax + 8], rcx
    inc r14
    inc rsi
    jmp pl

sort:
    mov r15d, 1
outer:
    cmp r15, r14
    jae output
    mov rax, r15
    shl rax, 4
    mov r8, qword ptr [rbx + rax]
    mov r9, qword ptr [rbx + rax + 8]
    lea r10, [rbx + rax - 16]
inner:
    cmp r10, rbx
    jb place
    mov rsi, qword ptr [r10]
    mov rdx, qword ptr [r10 + 8]
    mov rdi, r8
    mov rcx, rdx
    cmp rcx, r9
    cmova rcx, r9
    cmp eax, eax
    repe cmpsb
    ja shift
    jb place
    cmp rdx, r9
    jbe place
shift:
    mov rax, qword ptr [r10]
    mov qword ptr [r10 + 16], rax
    mov qword ptr [r10 + 24], rdx
    sub r10, 16
    jmp inner
place:
    mov qword ptr [r10 + 16], r8
    mov qword ptr [r10 + 24], r9
    inc r15
    jmp outer

output:
    lea rdi, [rip + outbuf]
    xor edx, edx
ol:
    cmp rdx, r14
    jae wr
    mov rax, rdx
    shl rax, 4
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rdx
    jmp ol
wr:
    lea r12, [rip + outbuf]
    mov r13, rdi
wl:
    mov rdx, r13
    sub rdx, r12
    jle fin
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    syscall
    test rax, rax
    jle fin
    add r12, rax
    jmp wl
fin:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inbuf:
    .zero 65552
.align 16
tab:
    .zero 65536
.align 16
outbuf:
    .zero 65600
