.globl _start

.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
.align 16
buf: .zero 65600
outbuf: .zero 32

.text
_start:
    lea r12, [rip + buf]
1:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    lea rdx, [rip + buf + 65600]
    sub rdx, r12
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    lea rsi, [rip + buf]
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    lea rax, [rip + buf]
    sub rsi, rax
    mov rbx, rsi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov rax, rbx
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 2f
    cmp al, 9
    je 2f
    cmp al, 10
    je 2f
    cmp al, 13
    je 2f
    ret
2:
    inc rsi
    jmp 1b

parse_value:
    call skip_ws
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    lea rdi, [rip + lit_true]
    cmp al, 't'
    je parse_lit
    lea rdi, [rip + lit_false]
    cmp al, 'f'
    je parse_lit
    lea rdi, [rip + lit_null]
    cmp al, 'n'
    je parse_lit
    cmp al, '-'
    je parse_number
    lea ecx, [rax - '0']
    cmp ecx, 9
    jbe parse_number
    jmp error

parse_lit:
1:
    mov cl, byte ptr [rdi]
    test cl, cl
    jz 2f
    cmp cl, byte ptr [rsi]
    jne error
    inc rsi
    inc rdi
    jmp 1b
2:
    ret

parse_string:
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je 9f
    cmp al, 0x5C
    jne 1b
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je 2f
    cmp al, 0x5C
    je 2f
    cmp al, '/'
    je 2f
    cmp al, 'b'
    je 2f
    cmp al, 'f'
    je 2f
    cmp al, 'n'
    je 2f
    cmp al, 'r'
    je 2f
    cmp al, 't'
    je 2f
    cmp al, 'u'
    jne error
    inc rsi
    mov edx, 4
3:
    movzx eax, byte ptr [rsi]
    lea ecx, [rax - '0']
    cmp ecx, 9
    jbe 4f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
4:
    inc rsi
    dec edx
    jnz 3b
    jmp 1b
2:
    inc rsi
    jmp 1b
9:
    ret

need_digits:
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
skip_digits:
1:
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja 2f
    inc rsi
    jmp 1b
2:
    ret

parse_number:
    cmp al, '-'
    jne 1f
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jne 2f
    inc rsi
    jmp 3f
2:
    sub al, '1'
    cmp al, 8
    ja error
    inc rsi
    call skip_digits
3:
    cmp byte ptr [rsi], '.'
    jne 4f
    inc rsi
    call need_digits
4:
    mov al, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne 9f
    inc rsi
    mov al, byte ptr [rsi]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:
    inc rsi
6:
    call need_digits
9:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp al, ']'
    je 9f
1:
    call parse_value
    call skip_ws
    cmp al, ','
    jne 2f
    inc rsi
    jmp 1b
2:
    cmp al, ']'
    jne error
9:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp al, '}'
    je 9f
1:
    cmp al, '"'
    jne error
    call parse_string
    call skip_ws
    cmp al, ':'
    jne error
    inc rsi
    call parse_value
    call skip_ws
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp 1b
2:
    cmp al, '}'
    jne error
9:
    inc rsi
    ret
