.globl _start

.equ BUFSZ, 0x11000

.text
_start:
    adrp x21, buf
    add x21, x21, :lo12:buf
    mov x19, x21
rd_loop:
    mov x0, #0
    mov x1, x19
    add x2, x21, #0x11, lsl #12
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x19, x19, x0
    b rd_loop
rd_done:
    mov x20, x19
    mov x19, x21
    adrp x22, stk
    add x22, x22, :lo12:stk
    mov x23, #0

parse_value:
    bl skipws
    cmp w0, #'{'
    b.eq pv_obj
    cmp w0, #'['
    b.eq pv_arr
    cmp w0, #'"'
    b.eq pv_str
    cmp w0, #'-'
    b.eq number
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls number
    cmp w0, #'t'
    b.eq pv_true
    cmp w0, #'f'
    b.eq pv_false
    cmp w0, #'n'
    b.eq pv_null
    b error

pv_obj:
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    add x19, x19, #1
    bl skipws
    cmp w0, #'}'
    b.ne member
    add x19, x19, #1
    sub x23, x23, #1
    b value_done

pv_arr:
    mov w1, #0
    strb w1, [x22, x23]
    add x23, x23, #1
    add x19, x19, #1
    bl skipws
    cmp w0, #']'
    b.ne parse_value
    add x19, x19, #1
    sub x23, x23, #1
    b value_done

pv_str:
    bl string
    b value_done

pv_true:
    adrp x1, s_true
    add x1, x1, :lo12:s_true
    b literal
pv_false:
    adrp x1, s_false
    add x1, x1, :lo12:s_false
    b literal
pv_null:
    adrp x1, s_null
    add x1, x1, :lo12:s_null
literal:
    ldrb w2, [x1], #1
    cbz w2, value_done
    bl peek
    cmp w0, w2
    b.ne error
    add x19, x19, #1
    b literal

member:
    bl skipws
    cmp w0, #'"'
    b.ne error
    bl string
    bl skipws
    cmp w0, #':'
    b.ne error
    add x19, x19, #1
    b parse_value

value_done:
    bl skipws
    cbnz x23, 1f
    cmn w0, #1
    b.eq ok
    b error
1:
    sub x1, x23, #1
    ldrb w2, [x22, x1]
    cmp w0, #','
    b.ne 2f
    add x19, x19, #1
    cbnz w2, member
    b parse_value
2:
    cbnz w2, 3f
    cmp w0, #']'
    b.ne error
    b 4f
3:
    cmp w0, #'}'
    b.ne error
4:
    add x19, x19, #1
    sub x23, x23, #1
    b value_done

number:
    cmp w0, #'-'
    b.ne 1f
    add x19, x19, #1
    bl peek
1:
    cmp w0, #'0'
    b.ne 2f
    add x19, x19, #1
    b num_frac
2:
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi error
3:
    add x19, x19, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 3b
num_frac:
    bl peek
    cmp w0, #'.'
    b.ne num_exp
    add x19, x19, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
1:
    add x19, x19, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 1b
num_exp:
    cmp w0, #'e'
    b.eq 1f
    cmp w0, #'E'
    b.ne value_done
1:
    add x19, x19, #1
    bl peek
    cmp w0, #'+'
    b.eq 2f
    cmp w0, #'-'
    b.ne 3f
2:
    add x19, x19, #1
    bl peek
3:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
4:
    add x19, x19, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 4b
    b value_done

// string: x19 at opening quote
string:
    mov x24, x30
    add x19, x19, #1
str_loop:
    bl peek
    cmp w0, #0x20
    b.lt error
    cmp w0, #'"'
    b.eq str_end
    cmp w0, #0x5c
    b.eq str_esc
    add x19, x19, #1
    b str_loop
str_end:
    add x19, x19, #1
    br x24
str_esc:
    add x19, x19, #1
    bl peek
    cmp w0, #'"'
    b.eq 1f
    cmp w0, #0x5c
    b.eq 1f
    cmp w0, #'/'
    b.eq 1f
    cmp w0, #'b'
    b.eq 1f
    cmp w0, #'f'
    b.eq 1f
    cmp w0, #'n'
    b.eq 1f
    cmp w0, #'r'
    b.eq 1f
    cmp w0, #'t'
    b.eq 1f
    cmp w0, #'u'
    b.ne error
    add x19, x19, #1
    mov w3, #4
2:
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 3f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi error
3:
    add x19, x19, #1
    subs w3, w3, #1
    b.ne 2b
    b str_loop
1:
    add x19, x19, #1
    b str_loop

peek:
    cmp x19, x20
    b.hs 1f
    ldrb w0, [x19]
    ret
1:
    mov w0, #-1
    ret

skipws:
    cmp x19, x20
    b.hs 2f
    ldrb w0, [x19]
    cmp w0, #' '
    b.eq 1f
    cmp w0, #9
    b.eq 1f
    cmp w0, #10
    b.eq 1f
    cmp w0, #13
    b.ne 3f
1:
    add x19, x19, #1
    b skipws
2:
    mov w0, #-1
3:
    ret

ok:
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, s_err
    add x1, x1, :lo12:s_err
    mov x2, #9
    mov x8, #64
    svc #0
    sub x3, x19, x21
    adrp x4, outbuf
    add x4, x4, :lo12:outbuf
    add x4, x4, #31
    mov w5, #10
    strb w5, [x4]
    mov x1, x4
    mov x6, #10
1:
    udiv x7, x3, x6
    msub x9, x7, x6, x3
    add w9, w9, #'0'
    strb w9, [x1, #-1]!
    mov x3, x7
    cbnz x3, 1b
    add x2, x4, #1
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"
s_ok:    .ascii "ok\n"
s_err:   .ascii "error at "

.bss
.balign 16
outbuf: .space 32
stk:    .space 2048
buf:    .space BUFSZ
