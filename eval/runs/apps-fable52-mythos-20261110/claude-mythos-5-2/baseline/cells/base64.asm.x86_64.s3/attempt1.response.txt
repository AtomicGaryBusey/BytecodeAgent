.globl _start

.section .rodata
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
inbuf:
    .space 65536
outbuf:
    .space 131072

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + table]
    mov r12, rbx
    xor r13d, r13d
enc_loop:
    cmp r12, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3
    sub r12, 3
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl
    and eax, 63
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], al
    add rdi, 4
    add r13d, 4
    cmp r13d, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r13d, r13d
    jmp enc_loop
tail:
    test r12, r12
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r12, 2
    jne 1f
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
1:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], 61
    cmp r12, 2
    je 2f
    mov byte ptr [rdi + 2], 61
2:
    add rdi, 4
    add r13d, 4
finish:
    test r13d, r13d
    jz 3f
    mov byte ptr [rdi], 10
    inc rdi
3:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle done
    mov rdx, rbx
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rbx, rax
    jmp write_loop
done:
    xor edi, edi
    mov eax, 60
    syscall
