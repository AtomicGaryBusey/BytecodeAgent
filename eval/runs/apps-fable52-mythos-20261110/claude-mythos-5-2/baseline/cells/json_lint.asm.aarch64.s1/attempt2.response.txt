.globl _start
.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
rd:
    mov x0, #0
    add x1, x19, x20
    mov x2, #0x11
    lsl x2, x2, #12
    sub x2, x2, x20
    cmp x2, #0
    b.le rd_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x20, x20, x0
    b rd
rd_done:
    mov x21, #0
    adrp x22, stk
    add x22, x22, :lo12:stk
    mov x23, #0

parse_value:
    bl skipws
    bl peek
    cmp w0, #0x7b
    b.eq v_obj
    cmp w0, #0x5b
    b.eq v_arr
    cmp w0, #0x22
    b.eq v_str
    cmp w0, #0x2d
    b.eq v_num
    cmp w0, #0x74
    b.eq v_true
    cmp w0, #0x66
    b.eq v_false
    cmp w0, #0x6e
    b.eq v_null
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls v_num
    b error

v_obj:
    add x21, x21, #1
    bl skipws
    bl peek
    cmp w0, #0x7d
    b.ne 1f
    add x21, x21, #1
    b value_done
1:
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    b member_key

v_arr:
    add x21, x21, #1
    bl skipws
    bl peek
    cmp w0, #0x5d
    b.ne 1f
    add x21, x21, #1
    b value_done
1:
    strb wzr, [x22, x23]
    add x23, x23, #1
    b parse_value

v_str:
    bl string
    b value_done

member_key:
    bl peek
    cmp w0, #0x22
    b.ne error
    bl string
    bl skipws
    bl peek
    cmp w0, #0x3a
    b.ne error
    add x21, x21, #1
    b parse_value

v_true:
    adrp x1, s_true
    add x1, x1, :lo12:s_true
    b lit
v_false:
    adrp x1, s_false
    add x1, x1, :lo12:s_false
    b lit
v_null:
    adrp x1, s_null
    add x1, x1, :lo12:s_null
lit:
    ldrb w2, [x1], #1
    cbz w2, value_done
    bl peek
    cmp w0, w2
    b.ne error
    add x21, x21, #1
    b lit

v_num:
    cmp w0, #0x2d
    b.ne 1f
    add x21, x21, #1
1:
    bl peek
    cmp w0, #0x30
    b.ne 2f
    add x21, x21, #1
    b n_frac
2:
    sub w1, w0, #0x31
    cmp w1, #8
    b.hi error
3:
    add x21, x21, #1
    bl peek
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 3b
n_frac:
    bl peek
    cmp w0, #0x2e
    b.ne n_exp
    add x21, x21, #1
    bl peek
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi error
4:
    add x21, x21, #1
    bl peek
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 4b
n_exp:
    orr w1, w0, #0x20
    cmp w1, #0x65
    b.ne value_done
    add x21, x21, #1
    bl peek
    cmp w0, #0x2b
    b.eq 5f
    cmp w0, #0x2d
    b.ne 6f
5:
    add x21, x21, #1
    bl peek
6:
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi error
7:
    add x21, x21, #1
    bl peek
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 7b
    b value_done

value_done:
    bl skipws
    cbnz x23, 1f
    cmp x21, x20
    b.ne error
    b ok
1:
    sub x1, x23, #1
    ldrb w2, [x22, x1]
    bl peek
    cmp w0, #0x2c
    b.ne 2f
    add x21, x21, #1
    cbz w2, parse_value
    bl skipws
    b member_key
2:
    mov w3, #0x5d
    mov w4, #0x7d
    cmp w2, #0
    csel w3, w3, w4, eq
    cmp w0, w3
    b.ne error
    add x21, x21, #1
    mov x23, x1
    b value_done

// string: pos at opening quote; returns after closing quote
string:
    mov x25, x30
    add x21, x21, #1
s_loop:
    bl peek
    cmp w0, #0x20
    b.lt error
    add x21, x21, #1
    cmp w0, #0x22
    b.eq s_done
    cmp w0, #0x5c
    b.ne s_loop
    bl peek
    cmp w0, #0x22
    b.eq s_esc
    cmp w0, #0x5c
    b.eq s_esc
    cmp w0, #0x2f
    b.eq s_esc
    cmp w0, #0x62
    b.eq s_esc
    cmp w0, #0x66
    b.eq s_esc
    cmp w0, #0x6e
    b.eq s_esc
    cmp w0, #0x72
    b.eq s_esc
    cmp w0, #0x74
    b.eq s_esc
    cmp w0, #0x75
    b.ne error
    add x21, x21, #1
    mov w5, #4
s_hex:
    bl peek
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls 1f
    orr w1, w0, #0x20
    sub w1, w1, #0x61
    cmp w1, #5
    b.hi error
1:
    add x21, x21, #1
    subs w5, w5, #1
    b.ne s_hex
    b s_loop
s_esc:
    add x21, x21, #1
    b s_loop
s_done:
    br x25

peek:
    mov w0, #-1
    cmp x21, x20
    b.hs 1f
    ldrb w0, [x19, x21]
1:
    ret

skipws:
    cmp x21, x20
    b.hs 2f
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq 1f
    cmp w0, #9
    b.eq 1f
    cmp w0, #10
    b.eq 1f
    cmp w0, #13
    b.ne 2f
1:
    add x21, x21, #1
    b skipws
2:
    ret

ok:
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x2, x1, #9
    sub sp, sp, #32
    mov x3, sp
    mov x4, #0
    mov x0, x21
    mov x10, #10
1:
    udiv x5, x0, x10
    msub x6, x5, x10, x0
    add w6, w6, #0x30
    strb w6, [x3, x4]
    add x4, x4, #1
    mov x0, x5
    cbnz x0, 1b
2:
    sub x4, x4, #1
    ldrb w6, [x3, x4]
    strb w6, [x2], #1
    cbnz x4, 2b
    mov w6, #10
    strb w6, [x2], #1
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
