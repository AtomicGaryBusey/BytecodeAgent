.globl _start
.equ BUFSZ, 65536

.text
_start:
    xor r13d, r13d
L_rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r13
    mov edx, BUFSZ
    sub rdx, r13
    syscall
    test rax, rax
    jle L_rd_done
    add r13, rax
    jmp L_rd
L_rd_done:
    lea rsi, [rip + buf]
    lea r15, [rsi + r13]
    lea r12, [rip + arr]
    xor r14d, r14d
L_parse:
    cmp rsi, r15
    jae L_output
    mov r8, rsi
1:
    cmp rsi, r15
    jae 2f
    cmp byte ptr [rsi], 10
    je 2f
    inc rsi
    jmp 1b
2:
    mov r9, rsi
    sub r9, r8
    cmp rsi, r15
    jae 3f
    inc rsi
3:
    xor ebx, ebx
    mov rbp, r14
L_bsearch:
    cmp rbx, rbp
    jae L_bsdone
    lea rax, [rbx + rbp]
    shr rax, 1
    mov rdx, qword ptr [r12 + rax*8]
    mov r10d, edx
    shr rdx, 32
    mov rcx, r9
    cmp rcx, rdx
    cmova rcx, rdx
    xor edi, edi
L_cmploop:
    cmp rdi, rcx
    jae L_cmpeq
    movzx r11d, byte ptr [r8 + rdi]
    cmp r11b, byte ptr [r10 + rdi]
    jb L_less
    ja L_greater
    inc rdi
    jmp L_cmploop
L_cmpeq:
    cmp r9, rdx
    jb L_less
L_greater:
    lea rbx, [rax + 1]
    jmp L_bsearch
L_less:
    mov rbp, rax
    jmp L_bsearch
L_bsdone:
    mov rcx, r14
L_shift:
    cmp rcx, rbx
    jbe L_shiftdone
    mov rax, qword ptr [r12 + rcx*8 - 8]
    mov qword ptr [r12 + rcx*8], rax
    dec rcx
    jmp L_shift
L_shiftdone:
    mov rax, r9
    shl rax, 32
    or rax, r8
    mov qword ptr [r12 + rbx*8], rax
    inc r14
    jmp L_parse
L_output:
    lea rdi, [rip + outbuf]
    xor ebx, ebx
L_o1:
    cmp rbx, r14
    jae L_o2
    mov rax, qword ptr [r12 + rbx*8]
    mov esi, eax
    shr rax, 32
    mov rcx, rax
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp L_o1
L_o2:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
L_wr:
    test r13, r13
    jle L_exit
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall
    test rax, rax
    jle L_exit
    add rsi, rax
    sub r13, rax
    jmp L_wr
L_exit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
buf:
    .zero 65536
arr:
    .zero 16384
outbuf:
    .zero 65536
