.globl _start
.equ BUFSZ, 65536

.text
_start:
    xor r13d, r13d
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r13
    mov edx, BUFSZ
    sub rdx, r13
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd
rd_done:
    lea rsi, [rip + buf]
    lea r15, [rsi + r13]
    lea r12, [rip + arr]
    xor r14d, r14d
parse:
    cmp rsi, r15
    jae out
    mov r8, rsi
1:
    cmp rsi, r15
    jae 2f
    cmp byte ptr [rsi], 10
    je 2f
    inc rsi
    jmp 1b
2:
    mov r9, rsi
    sub r9, r8
    cmp rsi, r15
    jae 3f
    inc rsi
3:
    xor ebx, ebx
    mov rbp, r14
bs:
    cmp rbx, rbp
    jae bsdone
    lea rax, [rbx + rbp]
    shr rax, 1
    mov rdx, qword ptr [r12 + rax*8]
    mov r10d, edx
    shr rdx, 32
    mov rcx, r9
    cmp rcx, rdx
    cmova rcx, rdx
    xor edi, edi
cl:
    cmp rdi, rcx
    jae ceq
    movzx r11d, byte ptr [r8 + rdi]
    cmp r11b, byte ptr [r10 + rdi]
    jb less
    ja greater
    inc rdi
    jmp cl
ceq:
    cmp r9, rdx
    jb less
greater:
    lea rbx, [rax + 1]
    jmp bs
less:
    mov rbp, rax
    jmp bs
bsdone:
    mov rcx, r14
sh:
    cmp rcx, rbx
    jbe shd
    mov rax, qword ptr [r12 + rcx*8 - 8]
    mov qword ptr [r12 + rcx*8], rax
    dec rcx
    jmp sh
shd:
    mov rax, r9
    shl rax, 32
    or rax, r8
    mov qword ptr [r12 + rbx*8], rax
    inc r14
    jmp parse
out:
    lea rdi, [rip + outbuf]
    xor ebx, ebx
o1:
    cmp rbx, r14
    jae o2
    mov rax, qword ptr [r12 + rbx*8]
    mov esi, eax
    shr rax, 32
    mov rcx, rax
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp o1
o2:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
wr:
    test r13, r13
    jle ex
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall
    test rax, rax
    jle ex
    add rsi, rax
    sub r13, rax
    jmp wr
ex:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
buf:
    .zero 65536
arr:
    .zero 16384
outbuf:
    .zero 65536
