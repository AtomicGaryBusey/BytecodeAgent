.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    ja nonws
    cmp al, 10
    jne ws
    inc r12
ws:
    xor r15d, r15d
    jmp next
nonws:
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
next:
    dec rcx
    jnz byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call putdec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call putdec
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call putdec
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov r12, rdi
    sub r12, rsi
write_loop:
    mov rdx, r12
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub r12, rax
    jnz write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

putdec:
    mov ebx, 10
    xor ecx, ecx
1:
    xor edx, edx
    div rbx
    add dl, 0x30
    push rdx
    inc ecx
    test rax, rax
    jnz 1b
2:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz 2b
    ret

.bss
    .align 16
buf:
    .zero 65536
outbuf:
    .zero 128
