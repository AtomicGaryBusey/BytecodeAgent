.globl _start
.equ CAP, 65536

.bss
buf:
    .space 65552

.text
_start:
    lea r12, [rip + buf]
    xor ebx, ebx
rd:
    mov edx, CAP
    sub rdx, rbx
    jz done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle done
    add rbx, rax
    jmp rd
done:
    test rbx, rbx
    jz finish
    cmp byte ptr [r12 + rbx - 1], 10
    je 1f
    mov byte ptr [r12 + rbx], 10
    inc rbx
1:
    lea r13, [r12 + rbx]
    mov rsi, r12
line:
    cmp rsi, r13
    jae wr
    mov rdi, rsi
find:
    cmp byte ptr [rdi], 10
    je found
    inc rdi
    jmp find
found:
    lea r8, [rdi + 1]
    dec rdi
rev:
    cmp rsi, rdi
    jae next
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp rev
next:
    mov rsi, r8
    jmp line
wr:
    mov rsi, r12
wloop:
    test rbx, rbx
    jz finish
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp wloop
finish:
    mov eax, 60
    xor edi, edi
    syscall
