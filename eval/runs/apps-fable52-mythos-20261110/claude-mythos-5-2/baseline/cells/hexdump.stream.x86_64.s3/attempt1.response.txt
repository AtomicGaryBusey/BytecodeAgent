.globl _start

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .space 65552
outbuf:
    .space 340000

.text
hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

_start:
    lea r12, [rip + inbuf]
    lea r8, [rip + hexd]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65552
    sub rdx, r13
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor r14d, r14d
row_loop:
    cmp r14, r13
    jae final
    mov eax, r14d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    lea rsi, [r12 + r14]
    xor ecx, ecx
slot:
    cmp rcx, r15
    jae pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp adv
pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
adv:
    add rdi, 3
    cmp ecx, 7
    jne 1f
    mov byte ptr [rdi], 0x20
    inc rdi
1:
    inc ecx
    cmp ecx, 16
    jb slot
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
chloop:
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 1f
    mov al, 0x2e
1:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    cmp rcx, r15
    jb chloop
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r14, r15
    jmp row_loop
final:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle finish
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
