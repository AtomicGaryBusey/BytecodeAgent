.globl _start
.equ BUFSZ, 65536 + 64

.bss
.balign 16
inbuf:
    .zero BUFSZ
outbuf:
    .zero BUFSZ

.text
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, BUFSZ - 16
    sub rdx, rbx
    jle read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea r9, [rsi + rbx]
    lea rdi, [rip + outbuf]
line_loop:
    cmp rsi, r9
    jae done
    mov rcx, rsi
1:
    cmp rcx, r9
    je 2f
    cmp byte ptr [rcx], 10
    je 2f
    inc rcx
    jmp 1b
2:
    mov rdx, rcx
3:
    cmp rdx, rsi
    je 4f
    dec rdx
    mov al, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp 3b
4:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rcx + 1]
    jmp line_loop
done:
    lea r12, [rip + outbuf]
    mov r13, rdi
    sub r13, r12
write_loop:
    test r13, r13
    jle finish
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle finish
    add r12, rax
    sub r13, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
