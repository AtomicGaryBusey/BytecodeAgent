.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
readloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
1:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 2f
    inc r12
2:
    cmp al, 32
    je 3f
    sub al, 9
    cmp al, 4
    jbe 3f
    test r15d, r15d
    jnz 4f
    inc r13
    mov r15d, 1
    jmp 4f
3:
    xor r15d, r15d
4:
    dec rcx
    jnz 1b
    jmp readloop

done:
    lea rdi, [rip + outbuf + 64]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call conv
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call conv
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call conv
    mov rsi, rdi
    lea rbx, [rip + outbuf + 64]
wloop:
    mov rdx, rbx
    sub rdx, rsi
    jle finish
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    jmp wloop
finish:
    mov eax, 60
    xor edi, edi
    syscall

conv:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
    .balign 16
buf:
    .zero 65536
outbuf:
    .zero 64
