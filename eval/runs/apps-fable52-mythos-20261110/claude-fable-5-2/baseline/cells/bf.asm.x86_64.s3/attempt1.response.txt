.globl _start
.equ INCAP, 69632
.equ OUTCAP, 65536

.bss
.align 16
inbuf: .zero 69632
.align 16
jmptab: .zero 278528
.align 16
outbuf: .zero 65536
.align 16
tape: .zero 30016

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rd:
    mov edx, INCAP
    sub rdx, r13
    jz rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd
rd_done:
    xor ecx, ecx
fl:
    cmp rcx, r13
    jae fdone
    cmp byte ptr [r12 + rcx], 0x21
    je fdone
    inc rcx
    jmp fl
fdone:
    mov r14, rcx
    lea r15, [rcx + 1]
    cmp r15, r13
    jbe 1f
    mov r15, r13
1:
    # bracket pass
    xor ecx, ecx
    xor r8d, r8d
    lea r9, [rip + jmptab]
bl:
    cmp rcx, r14
    jae bdone
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    inc r8
    jmp bnext
1:
    cmp al, 0x5D
    jne bnext
    test r8, r8
    jz bad
    pop rdx
    dec r8
    mov dword ptr [r9 + rcx*4], edx
    mov dword ptr [r9 + rdx*4], ecx
bnext:
    inc rcx
    jmp bl
bdone:
    test r8, r8
    jnz bad

    lea r8, [rip + outbuf]
    xor r10d, r10d
    lea rbx, [rip + tape]
    xor ebp, ebp
run:
    cmp rbp, r14
    jae fin
    movzx eax, byte ptr [r12 + rbp]
    inc rbp
    cmp al, 0x3E
    je c_right
    cmp al, 0x3C
    je c_left
    cmp al, 0x2B
    je c_plus
    cmp al, 0x2D
    je c_minus
    cmp al, 0x5B
    je c_open
    cmp al, 0x5D
    je c_close
    cmp al, 0x2E
    je c_out
    cmp al, 0x2C
    je c_in
    jmp run
c_right:
    inc rbx
    jmp run
c_left:
    dec rbx
    jmp run
c_plus:
    inc byte ptr [rbx]
    jmp run
c_minus:
    dec byte ptr [rbx]
    jmp run
c_open:
    cmp byte ptr [rbx], 0
    jne run
    mov ebp, dword ptr [r9 + rbp*4 - 4]
    inc rbp
    jmp run
c_close:
    cmp byte ptr [rbx], 0
    je run
    mov ebp, dword ptr [r9 + rbp*4 - 4]
    inc rbp
    jmp run
c_in:
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
    mov byte ptr [rbx], al
    jmp run
1:
    mov byte ptr [rbx], 0
    jmp run
c_out:
    mov al, byte ptr [rbx]
    mov byte ptr [r8 + r10], al
    inc r10
    cmp r10, OUTCAP
    jne run
    call flush
    jmp run

flush:
    test r10, r10
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r10
    syscall
    xor r10d, r10d
1:
    ret

fin:
    call flush
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall
