.globl _start
.equ INCAP, 69632
.equ OUTCAP, 65536

.bss
.align 16
inbuf: .zero 69632
.align 16
jmptab: .zero 278528
.align 16
outbuf: .zero 65536
.align 16
tape: .zero 30016

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, INCAP
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor ecx, ecx
find_loop:
    cmp rcx, r13
    jae find_done
    cmp byte ptr [r12 + rcx], 0x21
    je find_done
    inc rcx
    jmp find_loop
find_done:
    mov r14, rcx
    lea r15, [rcx + 1]
    cmp r15, r13
    jbe 1f
    mov r15, r13
1:
    # bracket pass
    xor ecx, ecx
    xor r8d, r8d
    lea r9, [rip + jmptab]
brk_loop:
    cmp rcx, r14
    jae brk_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    inc r8
    jmp brk_next
1:
    cmp al, 0x5D
    jne brk_next
    test r8, r8
    jz bad_exit
    pop rdx
    dec r8
    mov dword ptr [r9 + rcx*4], edx
    mov dword ptr [r9 + rdx*4], ecx
brk_next:
    inc rcx
    jmp brk_loop
brk_done:
    test r8, r8
    jnz bad_exit

    lea r8, [rip + outbuf]
    xor r10d, r10d
    lea rbx, [rip + tape]
    xor ebp, ebp
run_loop:
    cmp rbp, r14
    jae finish
    movzx eax, byte ptr [r12 + rbp]
    inc rbp
    cmp al, 0x3E
    je c_right
    cmp al, 0x3C
    je c_left
    cmp al, 0x2B
    je c_plus
    cmp al, 0x2D
    je c_minus
    cmp al, 0x5B
    je c_open
    cmp al, 0x5D
    je c_close
    cmp al, 0x2E
    je c_out
    cmp al, 0x2C
    je c_in
    jmp run_loop
c_right:
    inc rbx
    jmp run_loop
c_left:
    dec rbx
    jmp run_loop
c_plus:
    inc byte ptr [rbx]
    jmp run_loop
c_minus:
    dec byte ptr [rbx]
    jmp run_loop
c_open:
    cmp byte ptr [rbx], 0
    jne run_loop
    mov ebp, dword ptr [r9 + rbp*4 - 4]
    inc rbp
    jmp run_loop
c_close:
    cmp byte ptr [rbx], 0
    je run_loop
    mov ebp, dword ptr [r9 + rbp*4 - 4]
    inc rbp
    jmp run_loop
c_in:
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
    mov byte ptr [rbx], al
    jmp run_loop
1:
    mov byte ptr [rbx], 0
    jmp run_loop
c_out:
    mov al, byte ptr [rbx]
    mov byte ptr [r8 + r10], al
    inc r10
    cmp r10, OUTCAP
    jne run_loop
    call flush_out
    jmp run_loop

flush_out:
    test r10, r10
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r10
    syscall
    xor r10d, r10d
1:
    ret

finish:
    call flush_out
    mov eax, 60
    xor edi, edi
    syscall
bad_exit:
    mov eax, 60
    mov edi, 1
    syscall
