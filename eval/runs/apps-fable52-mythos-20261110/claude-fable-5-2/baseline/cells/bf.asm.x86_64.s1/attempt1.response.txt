.globl _start
.equ BUFSZ, 65600
.equ OUTSZ, 65536

.text
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
read_loop:
    mov edx, BUFSZ
    sub edx, ebx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    xor ecx, ecx
find_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov r13, rbx
    mov r14, rbx
    jmp check
found_bang:
    mov r13, rcx
    lea r14, [rcx + 1]
check:
    mov rbp, rsp
    lea rdi, [rip + jt]
    xor ecx, ecx
chk_loop:
    cmp rcx, r13
    jae chk_end
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp chk_next
1:
    cmp al, 0x5D
    jne chk_next
    cmp rsp, rbp
    je bad
    pop rax
    mov dword ptr [rdi + rcx*4], eax
    mov dword ptr [rdi + rax*4], ecx
chk_next:
    inc rcx
    jmp chk_loop
chk_end:
    cmp rsp, rbp
    jne bad

    mov rbp, rdi
    xor r8d, r8d
    xor r9d, r9d
    lea r10, [rip + outbuf]
    lea r15, [rip + tape]
run_loop:
    cmp r8, r13
    jae finish
    movzx eax, byte ptr [r12 + r8]
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
run_next:
    inc r8
    jmp run_loop
op_right:
    inc r15
    jmp run_next
op_left:
    dec r15
    jmp run_next
op_inc:
    inc byte ptr [r15]
    jmp run_next
op_dec:
    dec byte ptr [r15]
    jmp run_next
op_open:
    cmp byte ptr [r15], 0
    jne run_next
    mov r8d, dword ptr [rbp + r8*4]
    jmp run_next
op_close:
    cmp byte ptr [r15], 0
    je run_next
    mov r8d, dword ptr [rbp + r8*4]
    jmp run_next
op_out:
    mov al, byte ptr [r15]
    mov byte ptr [r10 + r9], al
    inc r9
    cmp r9, OUTSZ
    jne run_next
    call flush
    jmp run_next
op_in:
    xor eax, eax
    cmp r14, rbx
    jae 1f
    mov al, byte ptr [r12 + r14]
    inc r14
1:
    mov byte ptr [r15], al
    jmp run_next

finish:
    call flush
    xor edi, edi
    mov eax, 60
    syscall

bad:
    mov edi, 1
    mov eax, 60
    syscall

flush:
    test r9, r9
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r9
    syscall
    xor r9d, r9d
1:
    ret

.bss
.balign 16
inbuf:
    .zero BUFSZ
.balign 16
outbuf:
    .zero OUTSZ
.balign 16
tape:
    .zero 30016
.balign 16
jt:
    .zero 262400
