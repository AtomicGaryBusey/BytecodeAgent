.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
rloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]
bloop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    ja wordb
    cmp al, 10
    jne ws
    inc r12
ws:
    xor r15d, r15d
    jmp next
wordb:
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
next:
    dec rcx
    jnz bloop
    jmp rloop
done:
    lea rdi, [rip + outbuf + 64]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call conv
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call conv
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call conv
    mov rsi, rdi
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi
wloop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle fin
    add rsi, rax
    sub rdx, rax
    jnz wloop
fin:
    mov eax, 60
    xor edi, edi
    syscall

conv:
    mov ebx, 10
1:
    xor edx, edx
    div rbx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
    .balign 16
buf:
    .zero 65536
outbuf:
    .zero 64
