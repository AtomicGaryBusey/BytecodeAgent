.globl _start
.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, x19
    mov x24, #0x20000
    add x24, x19, x24
rd:
    sub x2, x24, x21
    cbz x2, rd_done
    mov x0, #0
    mov x1, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x21, x21, x0
    b rd
rd_done:
    mov x20, x19
    adrp x22, stk
    add x22, x22, :lo12:stk
    mov x23, #0

parse_value:
    bl skipws
    cmp w0, #'{'
    b.eq v_obj
    cmp w0, #'['
    b.eq v_arr
    cmp w0, #'"'
    b.eq v_str
    cmp w0, #'-'
    b.eq v_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls v_num
    cmp w0, #'t'
    b.eq v_true
    cmp w0, #'f'
    b.eq v_false
    cmp w0, #'n'
    b.eq v_null
    b error

v_obj:
    add x20, x20, #1
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skipws
    cmp w0, #'}'
    b.ne parse_member
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

v_arr:
    add x20, x20, #1
    mov w1, #0
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skipws
    cmp w0, #']'
    b.ne parse_value
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

v_str:
    bl parse_string
    b value_done

v_true:
    adr x26, s_true
    b lit
v_false:
    adr x26, s_false
    b lit
v_null:
    adr x26, s_null
lit:
    ldrb w27, [x26], #1
    cbz w27, value_done
    bl peek
    cmp w0, w27
    b.ne error
    add x20, x20, #1
    b lit

v_num:
    cmp w0, #'-'
    b.ne 1f
    add x20, x20, #1
    bl peek
1:
    cmp w0, #'0'
    b.ne 2f
    add x20, x20, #1
    b 3f
2:
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi error
    add x20, x20, #1
    bl digits
3:
    bl peek
    cmp w0, #'.'
    b.ne 4f
    add x20, x20, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
    bl digits
4:
    bl peek
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne value_done
    add x20, x20, #1
    bl peek
    cmp w0, #'+'
    b.eq 5f
    cmp w0, #'-'
    b.ne 6f
5:
    add x20, x20, #1
    bl peek
6:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
    bl digits
    b value_done

value_done:
    bl skipws
    cbnz x23, 1f
    cmn w0, #1
    b.eq ok
    b error
1:
    sub x1, x23, #1
    ldrb w2, [x22, x1]
    cbz w2, 3f
    // object
    cmp w0, #','
    b.ne 2f
    add x20, x20, #1
    bl skipws
    b parse_member
2:
    cmp w0, #'}'
    b.ne error
    add x20, x20, #1
    sub x23, x23, #1
    b value_done
3:
    cmp w0, #','
    b.ne 4f
    add x20, x20, #1
    b parse_value
4:
    cmp w0, #']'
    b.ne error
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

parse_member:
    cmp w0, #'"'
    b.ne error
    bl parse_string
    bl skipws
    cmp w0, #':'
    b.ne error
    add x20, x20, #1
    b parse_value

// returns w0 = current byte or -1 at end
peek:
    cmp x20, x21
    b.hs 1f
    ldrb w0, [x20]
    ret
1:
    mov w0, #-1
    ret

skipws:
    cmp x20, x21
    b.hs 2f
    ldrb w0, [x20]
    cmp w0, #' '
    b.eq 1f
    cmp w0, #9
    b.eq 1f
    cmp w0, #10
    b.eq 1f
    cmp w0, #13
    b.eq 1f
    ret
1:
    add x20, x20, #1
    b skipws
2:
    mov w0, #-1
    ret

digits:
    cmp x20, x21
    b.hs 1f
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi 1f
    add x20, x20, #1
    b digits
1:
    ret

parse_string:
    mov x25, x30
    add x20, x20, #1
ps_loop:
    bl peek
    cmp w0, #0x20
    b.lt error
    add x20, x20, #1
    cmp w0, #'"'
    b.eq ps_done
    cmp w0, #'\\'
    b.ne ps_loop
    bl peek
    cmp w0, #'"'
    b.eq ps_esc
    cmp w0, #'\\'
    b.eq ps_esc
    cmp w0, #'/'
    b.eq ps_esc
    cmp w0, #'b'
    b.eq ps_esc
    cmp w0, #'f'
    b.eq ps_esc
    cmp w0, #'n'
    b.eq ps_esc
    cmp w0, #'r'
    b.eq ps_esc
    cmp w0, #'t'
    b.eq ps_esc
    cmp w0, #'u'
    b.ne error
    add x20, x20, #1
    mov w28, #4
ps_hex:
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 1f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi error
1:
    add x20, x20, #1
    subs w28, w28, #1
    b.ne ps_hex
    b ps_loop
ps_esc:
    add x20, x20, #1
    b ps_loop
ps_done:
    ret x25

ok:
    mov x0, #1
    adr x1, s_ok
    mov x2, #3
    mov x8, #64

    mov x0, #0
    mov x8, #93

    sub sp, sp, #32
    add x1, sp, #32

    mov x3, #10

    udiv x4, x0, x3

    mov x0, x4

    add x2, sp, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
