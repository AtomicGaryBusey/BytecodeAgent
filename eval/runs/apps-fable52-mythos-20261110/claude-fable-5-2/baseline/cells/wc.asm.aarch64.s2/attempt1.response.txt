.globl _start

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
next:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x9, x24
    mov x0, x19
    bl putnum
    mov w3, #32
    strb w3, [x9], #1
    mov x0, x20
    bl putnum
    mov w3, #32
    strb w3, [x9], #1
    mov x0, x21
    bl putnum
    mov w3, #10
    strb w3, [x9], #1

    sub x25, x9, x24         // remaining length
write_loop:
    cbz x25, finish
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x24, x24, x0
    sub x25, x25, x0
    b write_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0
hang:
    b hang

// putnum: x0 = value, x9 = destination pointer (advanced)
putnum:
    sub sp, sp, #32
    mov x10, sp
    mov x11, #10
    mov x12, #0
1:
    udiv x13, x0, x11
    msub x14, x13, x11, x0
    add w14, w14, #48
    strb w14, [x10, x12]
    add x12, x12, #1
    mov x0, x13
    cbnz x0, 1b
2:
    sub x12, x12, #1
    ldrb w14, [x10, x12]
    strb w14, [x9], #1
    cbnz x12, 2b
    add sp, sp, #32
    ret

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 128
