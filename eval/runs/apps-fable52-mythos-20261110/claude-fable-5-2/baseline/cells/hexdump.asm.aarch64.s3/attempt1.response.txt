.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 131072
outbuf:
    .space 400000

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x28, x21
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x20, #0
    mov w27, #0x20

read_loop:
    mov x2, #0x20000
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0

row_loop:
    cmp x22, x20
    b.hs final
    sub x24, x20, x22
    cmp x24, #16
    b.ls 1f
    mov x24, #16
1:
    mov w0, w22
    bl hex8
    strb w27, [x21], #1
    strb w27, [x21], #1
    add x26, x19, x22
    mov x25, #0
2:
    cmp x25, x24
    b.hs 3f
    ldrb w0, [x26, x25]
    lsr w1, w0, #4
    ldrb w1, [x23, x1]
    strb w1, [x21], #1
    and w0, w0, #15
    ldrb w0, [x23, x0]
    strb w0, [x21], #1
    b 4f
3:
    strb w27, [x21], #1
    strb w27, [x21], #1
4:
    strb w27, [x21], #1
    cmp x25, #7
    b.ne 5f
    strb w27, [x21], #1
5:
    add x25, x25, #1
    cmp x25, #16
    b.lo 2b

    strb w27, [x21], #1
    mov w0, #0x7c
    strb w0, [x21], #1
    mov x25, #0
6:
    ldrb w0, [x26, x25]
    sub w1, w0, #0x20
    cmp w1, #0x5e
    b.ls 7f
    mov w0, #0x2e

    strb w0, [x21], #1
    add x25, x25, #1
    cmp x25, x24
    b.lo 6b
    mov w0, #0x7c
    strb w0, [x21], #1
    mov w0, #0x0a
    strb w0, [x21], #1
    add x22, x22, x24
    b row_loop

final:
    mov w0, w20
    bl hex8
    mov w0, #0x0a
    strb w0, [x21], #1

write_loop:
    subs x2, x21, x28
    b.le done
    mov x0, #1
    mov x1, x28
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le done
    add x28, x28, x0
    b write_loop

done:
    mov x0, #0
    mov x8, #93
    svc #0
    b done

hex8:
    mov w1, #28
1:
    lsr w2, w0, w1
    and w2, w2, #15
    ldrb w2, [x23, x2]
    strb w2, [x21], #1
    subs w1, w1, #4
    b.pl 1b
    ret
