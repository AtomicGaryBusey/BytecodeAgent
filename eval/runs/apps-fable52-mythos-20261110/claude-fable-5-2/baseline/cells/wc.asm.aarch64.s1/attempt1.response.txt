.globl _start

.bss
.balign 16
buf:
    .zero 4096
outbuf:
    .zero 64

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    mov x3, x0
byte_loop:
    ldrb w4, [x1], #1
    cmp w4, #0x20
    b.eq is_ws
    sub w5, w4, #9
    cmp w5, #4
    b.hi non_ws
is_ws:
    cmp w4, #10
    cinc x19, x19, eq
    mov x22, #0
    b next
non_ws:
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
next:
    subs x3, x3, #1
    b.ne byte_loop
    b read_loop

done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    add x24, x24, #64        // end of outbuf
    mov x1, x24
    mov w6, #10
    strb w6, [x1, #-1]!
    mov x0, x21
    bl conv
    mov w6, #0x20
    strb w6, [x1, #-1]!
    mov x0, x20
    bl conv
    mov w6, #0x20
    strb w6, [x1, #-1]!
    mov x0, x19
    bl conv
    sub x2, x24, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// conv: write decimal of x0 backwards before x1; updates x1
conv:
    mov x7, #10
1:
    udiv x5, x0, x7
    msub x6, x5, x7, x0
    add w6, w6, #0x30
    strb w6, [x1, #-1]!
    mov x0, x5
    cbnz x0, 1b
    ret
