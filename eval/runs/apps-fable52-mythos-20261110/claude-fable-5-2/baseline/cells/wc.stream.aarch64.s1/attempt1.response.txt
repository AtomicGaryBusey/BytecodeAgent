.globl _start

.text
_start:
    adrp x23, buf
    add x23, x23, :lo12:buf
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
next:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    add x24, x24, #96        // end of buffer
    mov x1, x24
    mov w7, #10
    strb w7, [x1, #-1]!
    mov x0, x21
    bl conv
    mov w7, #32
    strb w7, [x1, #-1]!
    mov x0, x20
    bl conv
    mov w7, #32
    strb w7, [x1, #-1]!
    mov x0, x19
    bl conv
    // x1 = start, x24 = end
    mov x25, x1
write_loop:
    cmp x25, x24
    b.hs finish
    mov x0, #1
    mov x1, x25
    sub x2, x24, x25
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x25, x25, x0
    b write_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0
hang:
    b hang

// conv: x0 = value, x1 = pointer (digits stored going downward)
conv:
    mov x5, #10
1:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    strb w7, [x1, #-1]!
    mov x0, x6
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 96
