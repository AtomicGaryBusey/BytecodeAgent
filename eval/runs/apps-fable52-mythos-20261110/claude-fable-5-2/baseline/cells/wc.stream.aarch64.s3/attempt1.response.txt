.globl _start

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 64

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x1, x23
    add x5, x23, x0          // end pointer
byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws
    // word byte
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
    cmp w3, #10
    b.ne next
    add x19, x19, #1
next:
    cmp x1, x5
    b.lo byte_loop
    b read_loop

done:
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    add x9, x9, #64
    mov x10, x9              // end of output
    mov x11, #10
    mov w12, #10
    strb w12, [x9, #-1]!
    mov x0, x21
    bl conv
    mov w12, #32
    strb w12, [x9, #-1]!
    mov x0, x20
    bl conv
    mov w12, #32
    strb w12, [x9, #-1]!
    mov x0, x19
    bl conv

    sub x13, x10, x9         // remaining length
write_loop:
    mov x0, #1
    mov x1, x9
    mov x2, x13
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x9, x9, x0
    sub x13, x13, x0
    cbnz x13, write_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0

// conv: write decimal of x0 backwards at x9 (pre-decrement); x11 = 10
conv:
1:
    udiv x14, x0, x11
    msub x15, x14, x11, x0
    add w15, w15, #48
    strb w15, [x9, #-1]!
    mov x0, x14
    cbnz x0, 1b
    ret
