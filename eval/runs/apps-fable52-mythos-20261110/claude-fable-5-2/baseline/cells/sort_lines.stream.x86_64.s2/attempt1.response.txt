.globl _start
.equ INSZ, 32768

.bss
.align 16
inbuf: .zero 32784
outbuf: .zero 32784
table: .zero 16032

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, INSZ
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    mov rsi, r12
    lea r14, [r12 + r13]
    lea rbx, [rip + table]
    xor r15d, r15d
parse:
    cmp rsi, r14
    jae parsed
    mov rdi, rsi
scan:
    cmp rsi, r14
    jae eol
    cmp byte ptr [rsi], 10
    je eol
    inc rsi
    jmp scan
eol:
    mov rax, rsi
    sub rax, rdi
    mov rcx, r15
    shl rcx, 4
    mov qword ptr [rbx + rcx], rdi
    mov qword ptr [rbx + rcx + 8], rax
    inc r15
    inc rsi
    jmp parse
parsed:
    mov r14, r15
    shl r14, 4
    add r14, rbx
    lea rbp, [rbx + 16]
outer:
    cmp rbp, r14
    jae sorted
    mov r8, qword ptr [rbp]
    mov r9, qword ptr [rbp + 8]
    mov rdx, rbp
inner:
    cmp rdx, rbx
    jbe place
    mov rsi, qword ptr [rdx - 16]
    mov r10, qword ptr [rdx - 8]
    mov rdi, r8
    mov rcx, r10
    cmp rcx, r9
    cmova rcx, r9
    cmp rcx, rcx
    repe cmpsb
    ja shift
    jb place
    cmp r10, r9
    jbe place
shift:
    mov rax, qword ptr [rdx - 16]
    mov qword ptr [rdx], rax
    mov qword ptr [rdx + 8], r10
    sub rdx, 16
    jmp inner
place:
    mov qword ptr [rdx], r8
    mov qword ptr [rdx + 8], r9
    add rbp, 16
    jmp outer
sorted:
    lea rdi, [rip + outbuf]
    mov r12, rdi
out_loop:
    cmp rbx, r14
    jae flush
    mov rsi, qword ptr [rbx]
    mov rcx, qword ptr [rbx + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add rbx, 16
    jmp out_loop
flush:
    mov r13, rdi
    sub r13, r12
write_loop:
    test r13, r13
    jz finish
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle finish
    add r12, rax
    sub r13, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall
