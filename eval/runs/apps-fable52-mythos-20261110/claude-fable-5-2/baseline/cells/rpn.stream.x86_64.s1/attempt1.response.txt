.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rd_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65552
    sub rdx, r13
    jle rd_done
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd_loop
rd_done:
    mov rsi, r12
    add r13, r12
    lea rdi, [rip + outbuf]
    lea r14, [rip + vstack]
    mov r10d, 10
line_loop:
    cmp rsi, r13
    jae finish
    xor r15d, r15d
tok_loop:
    cmp rsi, r13
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je line_end_nl
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    mov rbx, rsi
1:
    inc rbx
    cmp rbx, r13
    jae 2f
    mov cl, byte ptr [rbx]
    cmp cl, 10
    je 2f
    cmp cl, 32
    je 2f
    cmp cl, 9
    je 2f
    jmp 1b
2:
    mov rcx, rbx
    sub rcx, rsi
    cmp rcx, 1
    jne number
    cmp al, '+'
    je oper
    cmp al, '-'
    je oper
    cmp al, '*'
    je oper
    cmp al, '/'
    je oper
    cmp al, '%'
    je oper
number:
    xor r8d, r8d
    mov rdx, rsi
    cmp al, '-'
    jne 3f
    inc rdx
    mov r8d, 1
3:
    mov rcx, rbx
    sub rcx, rdx
    jz error
    cmp rcx, 18
    ja error
    xor eax, eax
4:
    movzx ecx, byte ptr [rdx]
    sub ecx, '0'
    cmp ecx, 9
    ja error
    imul rax, rax, 10
    add rax, rcx
    inc rdx
    cmp rdx, rbx
    jb 4b
    test r8d, r8d
    jz 5f
    neg rax
5:
    mov qword ptr [r14 + r15*8], rax
    inc r15
    mov rsi, rbx
    jmp tok_loop
skip_sep:
    inc rsi
    jmp tok_loop
oper:
    movzx r9d, al
    cmp r15, 2
    jb error
    mov rcx, qword ptr [r14 + r15*8 - 8]
    mov rax, qword ptr [r14 + r15*8 - 16]
    dec r15
    cmp r9b, '+'
    je op_add
    cmp r9b, '-'
    je op_sub
    cmp r9b, '*'
    je op_mul
    test rcx, rcx
    jz error
    cmp rcx, -1
    jne 6f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error
6:
    cqo
    idiv rcx
    cmp r9b, '/'
    je store
    mov rax, rdx
    jmp store
op_add:
    add rax, rcx
    jmp store
op_sub:
    sub rax, rcx
    jmp store
op_mul:
    imul rax, rcx
store:
    mov qword ptr [r14 + r15*8 - 8], rax
    mov rsi, rbx
    jmp tok_loop

error:
    cmp rsi, r13
    jae emit_err
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
emit_err:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_loop

line_end_nl:
    inc rsi
line_end:
    cmp r15, 1
    jne emit_err
    mov rax, qword ptr [r14]
    test rax, rax
    jns 7f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
7:
    lea r8, [rip + tmpend]
    mov r9, r8
8:
    xor edx, edx
    div r10
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 8b
9:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb 9b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_loop

finish:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
wr_loop:
    test rbx, rbx
    jle wr_done
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle wr_done
    add rsi, rax
    sub rbx, rax
    jmp wr_loop
wr_done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inbuf:
    .zero 65600
tmpbuf:
    .zero 32
tmpend:
    .zero 16
vstack:
    .zero 264000
outbuf:
    .zero 524288
