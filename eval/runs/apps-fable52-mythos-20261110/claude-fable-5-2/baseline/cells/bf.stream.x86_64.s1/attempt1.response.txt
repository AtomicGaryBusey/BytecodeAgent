.globl _start
.equ INSZ, 65552

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INSZ
    sub rdx, r13
    jz rd_done
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd
rd_done:
    xor ecx, ecx
fb:
    cmp rcx, r13
    jae nobang
    cmp byte ptr [r12 + rcx], 0x21
    je found
    inc rcx
    jmp fb
found:
    mov r14, rcx
    lea r15, [rcx + 1]
    jmp chk
nobang:
    mov r14, r13
    mov r15, r13
chk:
    mov rbp, rsp
    lea rdi, [rip + jtab]
    xor ecx, ecx
chk_loop:
    cmp rcx, r14
    jae chk_end
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp chk_next
1:
    cmp al, 0x5D
    jne chk_next
    cmp rsp, rbp
    je bad
    pop rdx
    mov dword ptr [rdi + rcx*4], edx
    mov dword ptr [rdi + rdx*4], ecx
chk_next:
    inc rcx
    jmp chk_loop
chk_end:
    cmp rsp, rbp
    jne bad

    lea rbx, [rip + tape]
    lea r10, [rip + outbuf]
    xor r9d, r9d
    xor r8d, r8d
run:
    cmp r8, r14
    jae done
    movzx eax, byte ptr [r12 + r8]
    inc r8
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp run
op_right:
    inc rbx
    jmp run
op_left:
    dec rbx
    jmp run
op_inc:
    inc byte ptr [rbx]
    jmp run
op_dec:
    dec byte ptr [rbx]
    jmp run
op_open:
    cmp byte ptr [rbx], 0
    jne run
    mov r8d, dword ptr [rdi + r8*4 - 4]
    inc r8
    jmp run
op_close:
    cmp byte ptr [rbx], 0
    je run
    mov r8d, dword ptr [rdi + r8*4 - 4]
    inc r8
    jmp run
op_out:
    cmp r9, 65536
    jae run
    mov al, byte ptr [rbx]
    mov byte ptr [r10 + r9], al
    inc r9
    jmp run
op_in:
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
    mov byte ptr [rbx], al
    jmp run
1:
    mov byte ptr [rbx], 0
    jmp run

done:
    test r9, r9
    jz fin
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r9
    syscall
    test rax, rax
    jle fin
    add r10, rax
    sub r9, rax
    jmp done
fin:
    xor edi, edi
    mov eax, 60
    syscall
bad:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
inbuf:
    .zero 65568
tape:
    .zero 30016
outbuf:
    .zero 65536
.align 16
jtab:
    .zero 262208
