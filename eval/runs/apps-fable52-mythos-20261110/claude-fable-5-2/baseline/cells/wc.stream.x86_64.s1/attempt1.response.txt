.globl _start
.text
_start:
    xor r12d, r12d          # lines
    xor r13d, r13d          # words
    xor r14d, r14d          # bytes
    xor r15d, r15d          # in-word flag
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je is_ws
    cmp al, 0x0A
    je is_nl
    lea ebx, [rax - 9]
    cmp ebx, 4
    jbe is_ws
    # word byte
    test r15d, r15d
    jnz next
    mov r15d, 1
    inc r13
    jmp next
is_nl:
    inc r12
is_ws:
    xor r15d, r15d
next:
    dec rcx
    jnz byte_loop
    jmp read_loop

done:
    # build output backwards
    lea rdi, [rip + outend]
    dec rdi
    mov byte ptr [rdi], 0x0A
    mov rax, r14
    call conv
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call conv
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call conv
    mov rsi, rdi
    lea rbx, [rip + outend]
    sub rbx, rsi            # remaining length
write_loop:
    test rbx, rbx
    jle finish
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rbx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# conv: write decimal of rax backwards ending before rdi; rdi updated
conv:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 96
outend:
    .zero 8
