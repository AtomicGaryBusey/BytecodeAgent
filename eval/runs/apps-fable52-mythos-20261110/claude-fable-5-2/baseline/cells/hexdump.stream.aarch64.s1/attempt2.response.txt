.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 65600
    .balign 16
outbuf:
    .space 327680

.text

// w0 = value, x21 = out pointer, x23 = hex table
hex8:
    mov w1, #28
1:
    lsr w2, w0, w1
    and w2, w2, #15
    ldrb w2, [x23, x2]
    strb w2, [x21], #1
    subs w1, w1, #4
    b.ge 1b
    ret

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x27, x21
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x20, #0
    movz x28, #0x40
    movk x28, #1, lsl #16

readloop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x28, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le readdone
    add x20, x20, x0
    b readloop

readdone:
    mov x22, #0

rowloop:
    cmp x22, x20
    b.hs final
    sub x24, x20, x22
    mov x9, #16
    cmp x24, x9
    csel x24, x24, x9, lo
    mov w0, w22
    bl hex8
    mov w3, #32
    strb w3, [x21], #1
    strb w3, [x21], #1
    add x25, x19, x22
    mov x4, #0
byteloop:
    cmp x4, x24
    b.hs pad
    ldrb w5, [x25, x4]
    lsr w6, w5, #4
    ldrb w6, [x23, x6]
    strb w6, [x21], #1
    and w6, w5, #15
    ldrb w6, [x23, x6]
    strb w6, [x21], #1
    strb w3, [x21], #1
    b next
pad:
    strb w3, [x21], #1
    strb w3, [x21], #1
    strb w3, [x21], #1
next:
    cmp x4, #7
    b.ne 1f
    strb w3, [x21], #1
1:
    add x4, x4, #1
    cmp x4, #16
    b.lo byteloop

    strb w3, [x21], #1
    mov w7, #124
    strb w7, [x21], #1
    mov x4, #0
    mov w7, #46
charloop:
    ldrb w5, [x25, x4]
    sub w6, w5, #0x20
    cmp w6, #0x5e
    csel w5, w5, w7, ls
    strb w5, [x21], #1
    add x4, x4, #1
    cmp x4, x24
    b.lo charloop
    mov w7, #124
    strb w7, [x21], #1
    mov w7, #10
    strb w7, [x21], #1
    add x22, x22, x24
    b rowloop

final:
    mov w0, w20
    bl hex8
    mov w7, #10
    strb w7, [x21], #1

writeloop:
    cmp x27, x21
    b.hs finish
    mov x0, #1
    mov x1, x27
    sub x2, x21, x27
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x27, x27, x0
    b writeloop

finish:
    mov x0, #0
    mov x8, #93
    svc #0
    b finish
