.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 65600
    .balign 16
outbuf:
    .space 327680

.text

// w0 = value, x21 = out pointer, x23 = hex table
hex8:
    mov w1, #28
1:
    lsr w2, w0, w1
    and w2, w2, #15
    ldrb w2, [x23, x2]
    strb w2, [x21], #1
    subs w1, w1, #4
    b.ge 1b
    ret

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x27, x21
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x20, #0
    mov x28, #65552

read
