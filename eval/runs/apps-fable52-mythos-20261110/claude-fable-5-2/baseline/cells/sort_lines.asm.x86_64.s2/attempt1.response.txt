.globl _start
.text
_start:
    lea r12, [rip + buf]
    xor r13d, r13d
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    jz rd_done
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd
rd_done:
    mov rbx, r12
    lea rbp, [r12 + r13]
    xor r14d, r14d
    lea r15, [rip + lines]
parse:
    cmp rbx, rbp
    jae sort
    mov rax, rbx
1:
    cmp rbx, rbp
    jae 2f
    cmp byte ptr [rbx], 10
    je 2f
    inc rbx
    jmp 1b
2:
    mov rcx, rbx
    sub rcx, rax
    mov rdx, r14
    shl rdx, 4
    mov qword ptr [r15 + rdx], rax
    mov qword ptr [r15 + rdx + 8], rcx
    inc r14
    inc rbx
    jmp parse

sort:
    mov r10d, 1
outer:
    cmp r10, r14
    jae output
    mov rax, r10
    shl rax, 4
    mov r8, qword ptr [r15 + rax]
    mov r9, qword ptr [r15 + rax + 8]
    mov r11, r10
inner:
    test r11, r11
    jz place
    mov rax, r11
    shl rax, 4
    mov rsi, qword ptr [r15 + rax - 16]
    mov rdx, qword ptr [r15 + rax - 8]
    mov rbx, rsi
    mov rdi, r8
    mov rcx, rdx
    cmp rcx, r9
    cmova rcx, r9
    cmp eax, eax
    repe cmpsb
    ja shift
    jb place
    cmp rdx, r9
    jbe place
shift:
    mov qword ptr [r15 + rax], rbx
    mov qword ptr [r15 + rax + 8], rdx
    dec r11
    jmp inner
place:
    mov rax, r11
    shl rax, 4
    mov qword ptr [r15 + rax], r8
    mov qword ptr [r15 + rax + 8], r9
    inc r10
    jmp outer

output:
    lea rdi, [rip + obuf]
    xor r10d, r10d
1:
    cmp r10, r14
    jae 3f
    mov rax, r10
    shl rax, 4
    mov rsi, qword ptr [r15 + rax]
    mov rcx, qword ptr [r15 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc r10
    jmp 1b
3:
    lea rsi, [rip + obuf]
    mov rbx, rdi
    sub rbx, rsi
2:
    test rbx, rbx
    jz done
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rbx, rax
    jmp 2b
done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
buf:
    .zero 65536
lines:
    .zero 65536
obuf:
    .zero 70000
