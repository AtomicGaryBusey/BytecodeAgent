.globl _start

.section .rodata
s_true:  .asciz "true"
s_false: .asciz "false"
s_null:  .asciz "null"
s_ok:    .ascii "ok\n"
s_err:   .ascii "error at "

.bss
.balign 16
buf: .space 65536+64
stk: .space 4096
obuf: .space 32

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, #0
rd:
    mov x2, #65536
    sub x2, x2, x21
    cbz x2, rd_done
    mov x0, #0
    add x1, x19, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x21, x21, x0
    b rd
rd_done:
    mov x20, #0
    adrp x22, stk
    add x22, x22, :lo12:stk
    mov x23, #0

value:
    bl skipws
    cmp w0, #'{'
    b.eq obj
    cmp w0, #'['
    b.eq arr
    cmp w0, #'"'
    b.ne 1f
    bl string
    b value_done
1:
    cmp w0, #'-'
    b.eq num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls num
    adrp x1, s_true
    add x1, x1, :lo12:s_true
    cmp w0, #'t'
    b.eq lit
    adrp x1, s_false
    add x1, x1, :lo12:s_false
    cmp w0, #'f'
    b.eq lit
    adrp x1, s_null
    add x1, x1, :lo12:s_null
    cmp w0, #'n'
    b.eq lit
    b error

lit:
    ldrb w2, [x1], #1
    cbz w2, value_done
    ldrb w0, [x19, x20]
    cmp w0, w2
    b.ne error
    add x20, x20, #1
    b lit

obj:
    add x20, x20, #1
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skipws
    cmp w0, #'}'
    b.ne member
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

member:
    bl skipws
    cmp w0, #'"'
    b.ne error
    bl string
    bl skipws
    cmp w0, #':'
    b.ne error
    add x20, x20, #1
    b value

arr:
    add x20, x20, #1
    strb wzr, [x22, x23]
    add x23, x23, #1
    bl skipws
    cmp w0, #']'
    b.ne value
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

value_done:
    cbz x23, top
    bl skipws
    sub x1, x23, #1
    ldrb w1, [x22, x1]
    cmp w0, #','
    b.ne 1f
    add x20, x20, #1
    cbnz w1, member
    b value
1:
    mov w2, #']'
    mov w3, #'}'
    cmp w1, #0
    csel w2, w3, w2, ne
    cmp w0, w2
    b.ne error
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

top:
    bl skipws
    cmp x20, x21
    b.ne error
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

// number
num:
    cmp w0, #'-'
    b.ne 1f
    add x20, x20, #1
    ldrb w0, [x19, x20]
1:
    cmp w0, #'0'
    b.ne 2f
    add x20, x20, #1
    b num_frac
2:
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi error
3:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 3b
num_frac:
    ldrb w0, [x19, x20]
    cmp w0, #'.'
    b.ne num_exp
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
4:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 4b
num_exp:
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne value_done
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'+'
    b.eq 5f
    cmp w0, #'-'
    b.ne 6f
5:
    add x20, x20, #1
    ldrb w0, [x19, x20]
6:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
7:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 7b
    b value_done

// skip whitespace; returns current byte in w0
skipws:
    ldrb w0, [x19, x20]
    cmp w0, #' '
    b.eq 1f
    cmp w0, #9
    b.eq 1f
    cmp w0, #10
    b.eq 1f
    cmp w0, #13
    b.eq 1f
    ret
1:
    add x20, x20, #1
    b skipws

// string: pos at opening quote
string:
    add x20, x20, #1
str_loop:
    ldrb w0, [x19, x20]
    cmp w0, #0x20
    b.lo error
    cmp w0, #'"'
    b.eq str_end
    cmp w0, #'\\'
    b.eq str_esc
    add x20, x20, #1
    b str_loop
str_end:
    add x20, x20, #1
    ret
str_esc:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'"'
    b.eq str_adv
    cmp w0, #'\\'
    b.eq str_adv
    cmp w0, #'/'
    b.eq str_adv
    cmp w0, #'b'
    b.eq str_adv
    cmp w0, #'f'
    b.eq str_adv
    cmp w0, #'n'
    b.eq str_adv
    cmp w0, #'r'
    b.eq str_adv
    cmp w0, #'t'
    b.eq str_adv
    cmp w0, #'u'
    b.ne error
    add x20, x20, #1
    mov w3, #4
1:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 2f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi error
2:
    add x20, x20, #1
    subs w3, w3, #1
    b.ne 1b
    b str_loop
str_adv:
    add x20, x20, #1
    b str_loop

error:
    mov x0, #1
    adrp x1, s_err
    add x1, x1, :lo12:s_err
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x1, obuf
    add x1, x1, :lo12:obuf
    add x1, x1, #31
    mov w2, #10
    strb w2, [x1]
    mov x2, #1
    mov x4, #10
1:
    udiv x5, x20, x4
    msub x6, x5, x4, x20
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    add x2, x2, #1
    mov x20, x5
    cbnz x20, 1b
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
