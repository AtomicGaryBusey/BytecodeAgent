.globl _start
.text

    add x19, x19, :lo12:inbuf
    mov x20, x19
    movz x25, #0x1170
    movk x25, #1, lsl #16

    mov x0, #0
    mov x1, x20
    mov x2, x25
    mov x8, #63
    svc #0
    cmp x0, #0

    add x24, x24, :lo12:outbuf
    mov x21, x24
    adrp x22, vstack
    add x22, x22, :lo12:vstack

    sub x0, x21, x24
    mov x1, #60000
    cmp x0, x1
    b.lo 1f
    bl flush
1:
    cmp x19, x20
    b.hs done
    mov x23, #0

    b.hs line_end
    ldrb w0, [x19]
    cmp w0, #10
    b.eq line_end
    cmp w0, #32
    b.eq 1f
    cmp w0, #9
    b.ne tok_start

    b tok_loop

    mov x9, x19
    mov x10, x19
2:
    cmp x10, x20
    b.hs 3f
    ldrb w0, [x10]
    cmp w0, #10
    b.eq 3f
    cmp w0, #32
    b.eq 3f
    cmp w0, #9
    b.eq 3f
    add x10, x10, #1
    b 2b
3:
    mov x19, x10
    sub x11, x10, x9
    ldrb w0, [x9]
    cmp x11, #1
    b.ne literal
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    cmp w0, #'/'
    b.eq op_div
    cmp w0, #'%'
    b.eq op_mod

    cmp w0, #'-'
    b.ne 4f
    mov x12, #1
    add x9, x9, #1

    sub x11, x10, x9
    cmp x11, #1
    b.lt error
    cmp x11, #18
    b.gt error
    mov x13, #0
    mov x14, #10

    sub w0, w0, #'0'
    cmp w0, #9
    b.hi error
    madd x13, x13, x14, x0
    cmp x9, x10
    b.lo 5b
    cbz x12, 6f
    neg x13, x13

    add x23, x23, #1
    b tok_loop

// pop two: x1 = a, x2 = b ; x23 decremented; result goes to [x23-1]

    cmp x23, #2
    b.lo error
    sub x23, x23, #1
    ldr x2, [x22, x23, lsl #3]
    sub x3, x23, #1
    ldr x1, [x22, x3, lsl #3]
    add x1, x1, x2
    str x1, [x22, x3, lsl #3]
    b tok_loop
op_sub:
    cmp x23, #2
    b.lo error
    sub x23, x23, #1
    ldr x2, [x22, x23, lsl #3]
    sub x3, x23, #1
    ldr x1, [x22, x3, lsl #3]
    sub x1, x1, x2
    str x1, [x22, x3, lsl #3]
    b tok_loop
op_mul:
    cmp x23, #2
    b.lo error
    sub x23, x23, #1
    ldr x2, [x22, x23, lsl #3]
    sub x3, x23, #1
    ldr x1, [x22, x3, lsl #3]

    str x1, [x22, x3, lsl #3]

op_div:

    b divmod

    b.lo error
    sub x23, x23, #1
    ldr x2, [x22, x23, lsl #3]
    sub x3, x23, #1
    ldr x1, [x22, x3, lsl #3]

    mov x4, #0x8000000000000000
    cmp x1, x4
    b.ne 7f
    cmn x2, #1
    b.eq error

    sdiv x5, x1, x2
    cbz x15, 8f
    msub x5, x5, x2, x1
8:
    str x5, [x22, x3, lsl #3]
    b tok_loop

    b.hs 9f
    ldrb w0, [x19], #1
    cmp w0, #10
    b.ne error

    b emit_error

line_end:
    cmp x19, x20
    b.hs 1f
    add x19, x19, #1
1:
    cmp x23, #1
    b.ne emit_error
    ldr x0, [x22]
    tbz x0, #63, 2f
    mov w1, #'-'
    strb w1, [x21], #1
    neg x0, x0

    sub sp, sp, #32
    add x1, sp, #32
    mov x3, #10

    msub x5, x4, x3, x0

    cbnz x0, 3b

    ldrb w5, [x1], #1
    strb w5, [x21], #1
    cmp x1, x2
    b.lo 4b
    add sp, sp, #32
    mov w1, #10
    strb w1, [x21], #1

    strb w0, [x21], #1
    mov w0, #'r'
    strb w0, [x21], #1
    strb w0, [x21], #1
    mov w0, #'o'
    strb w0, [x21], #1
    mov w0, #'r'

    mov x0, #0
    mov x8, #93

    mov x1, x6
    sub x2, x21, x6
    mov x8, #64

    cmp x0, #0
    b.le 2f
