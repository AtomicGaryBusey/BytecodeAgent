.globl _start
.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
.align 16
buf: .zero 131088
nest: .zero 70000
outbuf: .zero 32

.text
_start:
    lea r14, [rip + buf]
    lea r13, [rip + nest]
    mov r12, r14
read_loop:
    lea rdx, [r14 + 131072]
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov byte ptr [r12], 0
    mov rsi, r14
    xor ebx, ebx

value:
    call skipws
    cmp al, '{'
    je obj_start
    cmp al, '['
    je arr_start
    cmp al, '"'
    jne 1f
    call string
    jmp after_value
1:
    cmp al, '-'
    je number
    lea ecx, [rax - '0']
    cmp cl, 9
    jbe number
    lea rdi, [rip + lit_true]
    cmp al, 't'
    je literal
    lea rdi, [rip + lit_false]
    cmp al, 'f'
    je literal
    lea rdi, [rip + lit_null]
    cmp al, 'n'
    je literal
    jmp error

literal:
    mov cl, byte ptr [rdi]
    test cl, cl
    jz after_value
    cmp cl, byte ptr [rsi]
    jne error
    inc rsi
    inc rdi
    jmp literal

obj_start:
    inc rsi
    mov byte ptr [r13 + rbx], 1
    inc rbx
    call skipws
    cmp al, '}'
    je close
member:
    call skipws
    cmp al, '"'
    jne error
    call string
    call skipws
    cmp al, ':'
    jne error
    inc rsi
    jmp value

arr_start:
    inc rsi
    mov byte ptr [r13 + rbx], 0
    inc rbx
    call skipws
    cmp al, ']'
    je close
    jmp value

close:
    inc rsi
    dec rbx
after_value:
    call skipws
    test rbx, rbx
    jz toplevel
    cmp byte ptr [r13 + rbx - 1], 1
    je in_obj
    cmp al, ']'
    je close
    cmp al, ','
    jne error
    inc rsi
    jmp value
in_obj:
    cmp al, '}'
    je close
    cmp al, ','
    jne error
    inc rsi
    jmp member

toplevel:
    cmp rsi, r12
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

number:
    cmp al, '-'
    jne 1f
    inc rsi
    movzx eax, byte ptr [rsi]
1:
    cmp al, '0'
    jne 2f
    inc rsi
    jmp 3f
2:
    call digits
3:
    cmp byte ptr [rsi], '.'
    jne 4f
    inc rsi
    call digits
4:
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne after_value
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:
    inc rsi
6:
    call digits
    jmp after_value

# require one or more digits at rsi
digits:
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja error
1:
    inc rsi
    movzx eax, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    jbe 1b
    ret

# skip whitespace, return current byte in eax
skipws:
1:
    movzx eax, byte ptr [rsi]
    cmp al, ' '
    je 2f
    cmp al, 9
    je 2f
    cmp al, 10
    je 2f
    cmp al, 13
    jne 3f
2:
    inc rsi
    jmp 1b
3:
    ret

# rsi at opening quote
string:
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je 9f
    cmp al, 0x5C
    jne 1b
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je 2f
    cmp al, 0x5C
    je 2f
    cmp al, '/'
    je 2f
    cmp al, 'b'
    je 2f
    cmp al, 'f'
    je 2f
    cmp al, 'n'
    je 2f
    cmp al, 'r'
    je 2f
    cmp al, 't'
    je 2f
    cmp al, 'u'
    jne error
    inc rsi
    mov edx, 4
3:
    movzx eax, byte ptr [rsi]
    lea ecx, [rax - '0']
    cmp cl, 9
    jbe 4f
    or al, 0x20
    lea ecx, [rax - 'a']
    cmp cl, 5
    ja error
4:
    inc rsi
    dec edx
    jnz 3b
    jmp 1b
2:
    inc rsi
    jmp 1b
9:
    ret

error:
    mov rbx, rsi
    sub rbx, r14
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov rax, rbx
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
