.globl _start

.equ INCAP, 65552
.equ OUTCAP, 327680

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space INCAP
outbuf:
    .space OUTCAP

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x27, x21
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x20, #0

read_loop:
    mov x2, #INCAP
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0
    mov w28, #' '

row:
    cmp x22, x20
    bhs final
    sub x24, x20, x22
    cmp x24, #16
    mov x0, #16
    csel x24, x24, x0, lo
    mov w0, w22
    bl hex8
    strb w28, [x21], #1
    strb w28, [x21], #1
    mov x25, #0
    add x26, x19, x22
hexloop:
    cmp x25, x24
    bhs pad
    ldrb w0, [x26, x25]
    lsr w1, w0, #4
    ldrb w1, [x23, x1]
    strb w1, [x21], #1
    and w0, w0, #15
    ldrb w1, [x23, x0]
    strb w1, [x21], #1
    strb w28, [x21], #1
    b next
pad:
    strb w28, [x21], #1
    strb w28, [x21], #1
    strb w28, [x21], #1
next:
    cmp x25, #7
    bne 1f
    strb w28, [x21], #1
1:
    add x25, x25, #1
    cmp x25, #16
    blo hexloop

    strb w28, [x21], #1
    mov w1, #'|'
    strb w1, [x21], #1
    mov x25, #0
charloop:
    cmp x25, x24
    bhs chardone
    ldrb w0, [x26, x25]
    sub w1, w0, #0x20
    cmp w1, #0x5e
    bls 2f
    mov w0, #'.'
2:
    strb w0, [x21], #1
    add x25, x25, #1
    b charloop
chardone:
    mov w1, #'|'
    strb w1, [x21], #1
    mov w1, #10
    strb w1, [x21], #1
    add x22, x22, x24
    b row

final:
    mov w0, w20
    bl hex8
    mov w1, #10
    strb w1, [x21], #1

    // write everything
    mov x24, x27
write_loop:
    sub x2, x21, x24
    cmp x2, #0
    ble done
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0
    cmp x0, #0
    ble done
    add x24, x24, x0
    b write_loop

done:
    mov x0, #0
    mov x8, #93
    svc #0
3:
    b 3b

// w0 = value; writes 8 hex digits at x21, advances x21
hex8:
    mov w2, #28
4:
    lsr w1, w0, w2
    and w1, w1, #15
    ldrb w1, [x23, x1]
    strb w1, [x21], #1
    subs w2, w2, #4
    bpl 4b
    ret
