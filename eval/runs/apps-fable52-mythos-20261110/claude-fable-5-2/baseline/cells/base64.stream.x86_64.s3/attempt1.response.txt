.globl _start

.section .rodata
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072

.text
# emit 4 chars from 24-bit value in eax to [rdi], advance rdi, r8d
emit4:
    mov edx, eax
    shr edx, 18
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl
    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], al
    add rdi, 4
    add r8d, 4
    ret

_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + table]
    xor r8d, r8d
enc_loop:
    cmp r12, 3
    jb tail
    movzx eax, byte ptr [rsi]
    shl eax, 8
    mov al, byte ptr [rsi + 1]
    shl eax, 8
    mov al, byte ptr [rsi + 2]
    add rsi, 3
    sub r12, 3
    call emit4
    cmp r8d, 76
    jne enc_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp enc_loop
tail:
    test r12, r12
    jz finish
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r12, 2
    jne 1f
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
1:
    call emit4
    mov byte ptr [rdi - 1], 61
    cmp r12, 1
    jne finish
    mov byte ptr [rdi - 2], 61
finish:
    test r8d, r8d
    jz 2f
    mov byte ptr [rdi], 10
    inc rdi
2:
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
write_loop:
    test r13, r13
    jle done
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub r13, rax
    jmp write_loop
done:
    mov eax, 60
    xor edi, edi
    syscall
