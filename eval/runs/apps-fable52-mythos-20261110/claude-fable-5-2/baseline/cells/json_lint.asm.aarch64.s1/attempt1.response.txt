.globl _start

.equ BUFSZ, 70000

.text
_start:
    adrp x21, inbuf
    add x21, x21, :lo12:inbuf
    mov x19, x21            // cur
    mov x20, x21            // end
    movz x24, #(BUFSZ & 0xffff)
    movk x24, #(BUFSZ >> 16), lsl #16
    add x24, x21, x24       // buffer limit
read_loop:
    sub x2, x24, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x22, nstack
    add x22, x22, :lo12:nstack
    mov x23, #0             // depth

parse_value:
    bl skipws
    cmp w0, #'{'
    b.eq v_obj
    cmp w0, #'['
    b.eq v_arr
    cmp w0, #'"'
    b.eq v_str
    cmp w0, #'-'
    b.eq v_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls v_num
    cmp w0, #'t'
    b.eq v_true
    cmp w0, #'f'
    b.eq v_false
    cmp w0, #'n'
    b.eq v_null
    b error

v_obj:
    add x19, x19, #1
    bl skipws
    cmp w0, #'}'
    b.ne 1f
    add x19, x19, #1
    b value_done
1:
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    b parse_key

v_arr:
    add x19, x19, #1
    bl skipws
    cmp w0, #']'
    b.ne 1f
    add x19, x19, #1
    b value_done
1:
    strb wzr, [x22, x23]
    add x23, x23, #1
    b parse_value

v_str:
    add x19, x19, #1
    bl parse_string
    b value_done

v_true:
    adrp x26, s_true
    add x26, x26, :lo12:s_true
    b lit
v_false:
    adrp x26, s_false
    add x26, x26, :lo12:s_false
    b lit
v_null:
    adrp x26, s_null
    add x26, x26, :lo12:s_null
lit:
    ldrb w2, [x26], #1
    cbz w2, value_done
    bl peek
    cmp w0, w2
    b.ne error
    add x19, x19, #1
    b lit

v_num:
    cmp w0, #'-'
    b.ne 1f
    add x19, x19, #1
    bl peek
1:
    cmp w0, #'0'
    b.ne 2f
    add x19, x19, #1
    bl peek
    b 3f
2:
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi error
    add x19, x19, #1
    bl skip_digits
3:
    cmp w0, #'.'
    b.ne 4f
    add x19, x19, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
    bl skip_digits
4:
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne value_done
    cmn w0, #1
    b.eq value_done
    add x19, x19, #1
    bl peek
    cmp w0, #'+'
    b.eq 5f
    cmp w0, #'-'
    b.ne 6f
5:
    add x19, x19, #1
    bl peek
6:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error
    bl skip_digits
    b value_done

parse_key:
    // w0 = current char (ws already skipped)
    cmp w0, #'"'
    b.ne error
    add x19, x19, #1
    bl parse_string
    bl skipws
    cmp w0, #':'
    b.ne error
    add x19, x19, #1
    b parse_value

value_done:
    bl skipws
    cbnz x23, 1f
    cmp x19, x20
    b.hs success
    b error
1:
    sub x1, x23, #1
    ldrb w2, [x22, x1]
    cmp w0, #','
    b.ne 2f
    add x19, x19, #1
    cbz w2, parse_value
    bl skipws
    b parse_key
2:
    mov w3, #']'
    mov w4, #'}'
    cmp w2, #0
    csel w3, w3, w4, eq
    cmp w0, w3
    b.ne error
    add x19, x19, #1
    sub x23, x23, #1
    b value_done

// ---- subroutines ----
skipws:
1:
    cmp x19, x20
    b.hs 2f
    ldrb w0, [x19]
    cmp w0, #' '
    b.eq 3f
    cmp w0, #9
    b.eq 3f
    cmp w0, #10
    b.eq 3f
    cmp w0, #13
    b.eq 3f
    ret
3:
    add x19, x19, #1
    b 1b
2:
    mov w0, #-1
    ret

peek:
    cmp x19, x20
    b.hs 1f
    ldrb w0, [x19]
    ret
1:
    mov w0, #-1
    ret

skip_digits:
1:
    cmp x19, x20
    b.hs 2f
    ldrb w0, [x19]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi 3f
    add x19, x19, #1
    b 1b
2:
    mov w0, #-1
3:
    ret

parse_string:
    mov x25, x30
1:
    bl peek
    cmp w0, #0x20
    b.lt error
    add x19, x19, #1
    cmp w0, #'"'
    b.eq 9f
    cmp w0, #'\\'
    b.ne 1b
    bl peek
    cmp w0, #'"'
    b.eq 2f
    cmp w0, #'\\'
    b.eq 2f
    cmp w0, #'/'
    b.eq 2f
    cmp w0, #'b'
    b.eq 2f
    cmp w0, #'f'
    b.eq 2f
    cmp w0, #'n'
    b.eq 2f
    cmp w0, #'r'
    b.eq 2f
    cmp w0, #'t'
    b.eq 2f
    cmp w0, #'u'
    b.ne error
    add x19, x19, #1
    mov w27, #4
3:
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 4f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi error
4:
    add x19, x19, #1
    subs w27, w27, #1
    b.ne 3b
    b 1b
2:
    add x19, x19, #1
    b 1b
9:
    ret x25

// ---- results ----
success:
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, s_err
    add x1, x1, :lo12:s_err
    mov x2, #9
    mov x8, #64
    svc #0
    sub x3, x19, x21        // offset
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #23
    mov w4, #10
    strb w4, [x1]
    mov x2, #1
    mov x5, #10
1:
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #'0'
    strb w7, [x1, #-1]!
    add x2, x2, #1
    mov x3, x6
    cbnz x3, 1b
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
s_ok:
    .ascii "ok\n"
s_err:
    .ascii "error at "
s_true:
    .asciz "true"
s_false:
    .asciz "false"
s_null:
    .asciz "null"

.bss
.balign 16
inbuf:
    .space BUFSZ
.balign 16
nstack:
    .space BUFSZ
.balign 16
numbuf:
    .space 32
