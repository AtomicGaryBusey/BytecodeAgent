.globl _start
.section .rodata
hexdig: .ascii "0123456789abcdef"
.bss
inbuf: .zero 65536+64
outbuf: .zero 340000
.text
_start:
    lea r12, [rip + inbuf]
    lea r8, [rip + hexdig]
    xor r13d, r13d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536+32
    sub rdx, r13
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    lea rdi, [rip + outbuf]
    xor r14d, r14d
row_loop:
    cmp r14, r13
    jae final_part
    mov eax, r14d
    call hex8
    mov ax, 0x2020
    stosw
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe 3f
    mov r15d, 16
3:
    lea rbx, [r12 + r14]
    xor ecx, ecx
byte_loop:
    cmp rcx, r15
    jae pad_part
    movzx eax, byte ptr [rbx + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp next_part
pad_part:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
next_part:
    cmp ecx, 7
    jne 4f
    mov byte ptr [rdi], 0x20
    inc rdi
4:
    inc ecx
    cmp ecx, 16
    jb byte_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
char_loop:
    cmp rcx, r15
    jae 6f
    movzx eax, byte ptr [rbx + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 5f
    mov al, 0x2e
5:
    stosb
    inc rcx
    jmp char_loop
6:
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r14, r15
    jmp row_loop
final_part:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
write_loop:
    test rbx, rbx
    jle done_part
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle done_part
    add rsi, rax
    sub rbx, rax
    jmp write_loop
done_part:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret
