.globl _start
.section .rodata
digits: .ascii "0123456789abcdef"

.bss
.equ INSIZE, 65600
inbuf: .zero 65600
outbuf: .zero 330000

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, INSIZE
    sub rdx, r13
    jz gen
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle gen
    add r13, rax
    jmp read_loop

gen:
    lea rdi, [rip + outbuf]
    lea r8, [rip + digits]
    xor r14d, r14d
row_loop:
    cmp r14, r13
    jae final
    mov eax, r14d
    call hex8
    mov al, ' '
    stosb
    stosb
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    lea rsi, [r12 + r14]
    xor ebx, ebx
slot:
    cmp rbx, r15
    jae pad
    movzx eax, byte ptr [rsi + rbx]
    call hex2
    mov al, ' '
    stosb
    jmp next
pad:
    mov al, ' '
    stosb
    stosb
    stosb
next:
    cmp ebx, 7
    jne 2f
    mov al, ' '
    stosb
2:
    inc ebx
    cmp ebx, 16
    jb slot
    mov al, ' '
    stosb
    mov al, '|'
    stosb
    xor ebx, ebx
chars:
    cmp rbx, r15
    jae cdone
    movzx eax, byte ptr [rsi + rbx]
    lea ecx, [rax - 0x20]
    cmp ecx, 0x5e
    jbe 3f
    mov al, '.'
3:
    stosb
    inc ebx
    jmp chars
cdone:
    mov al, '|'
    stosb
    mov al, 10
    stosb
    add r14, r15
    jmp row_loop

final:
    mov eax, r13d
    call hex8
    mov al, 10
    stosb
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
wloop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle quit
    add rsi, rax
    sub rdx, rax
    jnz wloop
quit:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov edx, eax
    mov ecx, 8
1:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [r8 + rax]
    stosb
    dec ecx
    jnz 1b
    ret

hex2:
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [r8 + rax]
    stosb
    and edx, 15
    mov al, byte ptr [r8 + rdx]
    stosb
    ret
