.globl _start
.equ INSZ, 70000

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .zero 70016
outbuf:
    .zero 400000

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rdloop:
    mov edx, INSZ
    sub rdx, r13
    jz rddone
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle rddone
    add r13, rax
    jmp rdloop
rddone:
    lea rdi, [rip + outbuf]
    xor r14d, r14d
rowloop:
    mov rbx, r13
    sub rbx, r14
    jz final
    cmp rbx, 16
    jbe 1f
    mov ebx, 16
1:
    mov eax, r14d
    call hex8
    mov al, ' '
    stosb
    stosb
    lea rdx, [r12 + r14]
    xor ecx, ecx
slot:
    cmp rcx, rbx
    jae pad
    movzx eax, byte ptr [rdx + rcx]
    call hex2
    mov al, ' '
    stosb
    jmp next
pad:
    mov al, ' '
    stosb
    stosb
    stosb
next:
    cmp ecx, 7
    jne 2f
    mov al, ' '
    stosb
2:
    inc ecx
    cmp ecx, 16
    jb slot
    mov al, ' '
    stosb
    mov al, '|'
    stosb
    xor ecx, ecx
chars:
    movzx eax, byte ptr [rdx + rcx]
    cmp al, 0x20
    jb 3f
    cmp al, 0x7e
    jbe 4f
3:
    mov al, '.'
4:
    stosb
    inc ecx
    cmp rcx, rbx
    jb chars
    mov al, '|'
    stosb
    mov al, 10
    stosb
    add r14, rbx
    jmp rowloop
final:
    mov eax, r13d
    call hex8
    mov al, 10
    stosb
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
wloop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rdx, rax
    jnz wloop
done:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov r8d, eax
    mov r9d, 8
    lea r10, [rip + hexd]
1:
    rol r8d, 4
    mov eax, r8d
    and eax, 15
    mov al, byte ptr [r10 + rax]
    stosb
    dec r9d
    jnz 1b
    ret

hex2:
    mov r8d, eax
    shr eax, 4
    lea r9, [rip + hexd]
    mov al, byte ptr [r9 + rax]
    stosb
    and r8d, 15
    mov al, byte ptr [r9 + r8]
    stosb
    ret
