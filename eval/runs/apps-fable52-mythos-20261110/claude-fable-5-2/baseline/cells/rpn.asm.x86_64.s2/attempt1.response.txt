.globl _start
.text
_start:
    sub rsp, 64
    lea r14, [rip + inbuf]
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov rsi, r14
    lea r15, [r14 + rbx]
    lea rdi, [rip + outbuf]
    lea r12, [rip + vstack]

line_loop:
    cmp rsi, r15
    jae finish
    xor r13d, r13d
tok_loop:
    cmp rsi, r15
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je line_end_nl
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    xor r8d, r8d
    xor ecx, ecx
    xor edx, edx
    cmp al, 45
    jne digits
    lea r9, [rsi + 1]
    cmp r9, r15
    jae operator
    movzx r10d, byte ptr [r9]
    sub r10d, 48
    cmp r10d, 9
    ja operator
    mov r8d, 1
    inc rsi
digits:
    cmp rsi, r15
    jae numend
    movzx eax, byte ptr [rsi]
    lea r10d, [rax - 48]
    cmp r10d, 9
    ja numend
    imul rdx, rdx, 10
    add rdx, r10
    inc ecx
    inc rsi
    jmp digits
numend:
    test ecx, ecx
    jz operator
    cmp ecx, 18
    ja err
    cmp rsi, r15
    jae num_ok
    cmp al, 10
    je num_ok
    cmp al, 32
    je num_ok
    cmp al, 9
    jne err
num_ok:
    test r8d, r8d
    jz 1f
    neg rdx
1:
    mov qword ptr [r12 + r13*8], rdx
    inc r13
    jmp tok_loop

skip_sep:
    inc rsi
    jmp tok_loop

operator:
    lea r9, [rsi + 1]
    cmp r9, r15
    jae op_ok
    movzx ecx, byte ptr [r9]
    cmp cl, 10
    je op_ok
    cmp cl, 32
    je op_ok
    cmp cl, 9
    jne err
op_ok:
    inc rsi
    cmp al, 43
    je 1f
    cmp al, 45
    je 1f
    cmp al, 42
    je 1f
    cmp al, 47
    je 1f
    cmp al, 37
    jne err
1:
    cmp r13, 2
    jb err
    mov rcx, qword ptr [r12 + r13*8 - 8]
    mov r9, qword ptr [r12 + r13*8 - 16]
    dec r13
    cmp al, 43
    jne 1f
    add r9, rcx
    jmp store
1:
    cmp al, 45
    jne 1f
    sub r9, rcx
    jmp store
1:
    cmp al, 42
    jne 1f
    imul r9, rcx
    jmp store
1:
    test rcx, rcx
    jz err
    cmp rcx, -1
    jne 2f
    mov r10, 0x8000000000000000
    cmp r9, r10
    je err
2:
    mov r10d, eax
    mov rax, r9
    cqo
    idiv rcx
    cmp r10b, 47
    cmovne rax, rdx
    mov r9, rax
store:
    mov qword ptr [r12 + r13*8 - 8], r9
    jmp tok_loop

err:
    cmp rsi, r15
    jae err_out
    cmp byte ptr [rsi], 10
    lea rsi, [rsi + 1]
    jne err
err_out:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_loop

line_end_nl:
    inc rsi
line_end:
    cmp r13, 1
    jne err_out
    mov rax, qword ptr [r12]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], 45
    inc rdi
    neg rax
1:
    lea rbx, [rsp + 48]
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, 48
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz 2b
    lea r9, [rsp + 48]
3:
    mov al, byte ptr [rbx]
    mov byte ptr [rdi], al
    inc rbx
    inc rdi
    cmp rbx, r9
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_loop

finish:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
wloop:
    test rdx, rdx
    jle wdone
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wdone
    add rsi, rax
    sub rdx, rax
    jmp wloop
wdone:
    mov eax, 60
    xor edi, edi
    syscall
