.globl _start
.equ INSZ, 70000
.equ OUTSZ, 400000

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero INSZ
outbuf:
    .zero OUTSZ

.text
_start:
    lea r15, [rip + hexdigits]
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INSZ
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor r13d, r13d
row_loop:
    cmp r13, r12
    jae final
    mov eax, r13d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    lea rbx, [rip + inbuf]
    add rbx, r13
    xor ecx, ecx
byte_loop:
    cmp rcx, r14
    jae pad
    movzx eax, byte ptr [rbx + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp next
pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
next:
    cmp ecx, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc ecx
    cmp ecx, 16
    jb byte_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
char_loop:
    movzx eax, byte ptr [rbx + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 3f
    mov al, 0x2e
3:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    cmp rcx, r14
    jb char_loop
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r13, r14
    jmp row_loop
final:
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov r13, rdi
    sub r13, rsi
write_loop:
    test r13, r13
    jle exit
    mov eax, 1
    mov edi, 1
    mov rdx, r13
    syscall
    test rax, rax
    jle exit
    add rsi, rax
    sub r13, rax
    jmp write_loop
exit:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret
