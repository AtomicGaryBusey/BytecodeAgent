.globl _start
.equ INSZ, 65544

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
1:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INSZ
    sub edx, r13d
    jz 2f
    syscall
    test rax, rax
    jle 2f
    add r13, rax
    jmp 1b
2:
    # find first '!'
    xor ecx, ecx
3:
    cmp rcx, r13
    jae 4f
    cmp byte ptr [r12 + rcx], 0x21
    je 5f
    inc rcx
    jmp 3b
4:
    mov r14, r13
    mov r15, r13
    jmp 6f
5:
    mov r14, rcx
    lea r15, [rcx + 1]
6:
    # bracket check
    lea rdi, [rip + jt]
    mov rbp, rsp
    xor ecx, ecx
7:
    cmp rcx, r14
    jae 10f
    movzx eax, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 8f
    push rcx
    jmp 9f
8:
    cmp al, 0x5D
    jne 9f
    cmp rsp, rbp
    je fail
    pop rax
    mov dword ptr [rdi + rax*4], ecx
    mov dword ptr [rdi + rcx*4], eax
9:
    inc rcx
    jmp 7b
10:
    cmp rsp, rbp
    jne fail

    # run
    xor ebx, ebx
    lea r9, [rip + tape]
    lea r10, [rip + outbuf]
run:
    cmp rbx, r14
    jae done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 0x3E
    je c_right
    cmp al, 0x3C
    je c_left
    cmp al, 0x2B
    je c_inc
    cmp al, 0x2D
    je c_dec
    cmp al, 0x5B
    je c_open
    cmp al, 0x5D
    je c_close
    cmp al, 0x2E
    je c_out
    cmp al, 0x2C
    je c_in
next:
    inc rbx
    jmp run
c_right:
    inc r9
    jmp next
c_left:
    dec r9
    jmp next
c_inc:
    inc byte ptr [r9]
    jmp next
c_dec:
    dec byte ptr [r9]
    jmp next
c_open:
    cmp byte ptr [r9], 0
    jne next
    mov ebx, dword ptr [rdi + rbx*4]
    jmp next
c_close:
    cmp byte ptr [r9], 0
    je next
    mov ebx, dword ptr [rdi + rbx*4]
    jmp next
c_out:
    mov al, byte ptr [r9]
    mov byte ptr [r10], al
    inc r10
    jmp next
c_in:
    xor eax, eax
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
1:
    mov byte ptr [r9], al
    jmp next

done:
    lea rbx, [rip + outbuf]
1:
    mov rdx, r10
    sub rdx, rbx
    jle 2f
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    jmp 1b
2:
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
inbuf:
    .zero 65552
tape:
    .zero 30016
outbuf:
    .zero 65552
.align 16
jt:
    .zero 262208
