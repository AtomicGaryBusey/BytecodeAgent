.globl _start
.equ BUFSZ, 65536

.bss
.balign 16
inbuf: .zero 65536
outbuf: .zero 65536
tab: .zero 65536

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rdloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, BUFSZ
    sub rdx, r13
    jz rddone
    syscall
    test rax, rax
    jle rddone
    add r13, rax
    jmp rdloop
rddone:
    mov rsi, r12
    lea rbp, [r12 + r13]
    lea rbx, [rip + tab]
    xor r14d, r14d
parse:
    cmp rsi, rbp
    jae emit
    mov r9, rsi
1:
    cmp rsi, rbp
    jae 2f
    cmp byte ptr [rsi], 10
    je 2f
    inc rsi
    jmp 1b
2:
    mov r10, rsi
    sub r10, r9
    mov r8, rsi
    mov r15, r14
3:
    test r15, r15
    jz 5f
    mov rdx, r15
    shl rdx, 4
    mov rsi, qword ptr [rbx + rdx - 16]
    mov r11, qword ptr [rbx + rdx - 8]
    mov rdi, r9
    mov rcx, r11
    cmp rcx, r10
    cmova rcx, r10
    cmp rcx, rcx
    repe cmpsb
    jne 4f
    cmp r11, r10
4:
    jbe 5f
    movups xmm0, xmmword ptr [rbx + rdx - 16]
    movups xmmword ptr [rbx + rdx], xmm0
    dec r15
    jmp 3b
5:
    mov rdx, r15
    shl rdx, 4
    mov qword ptr [rbx + rdx], r9
    mov qword ptr [rbx + rdx + 8], r10
    inc r14
    lea rsi, [r8 + 1]
    jmp parse

emit:
    lea rdi, [rip + outbuf]
    mov r12, rdi
    xor edx, edx
6:
    cmp rdx, r14
    jae 7f
    mov rax, rdx
    shl rax, 4
    mov rsi, qword ptr [rbx + rax]
    mov rcx, qword ptr [rbx + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rdx
    jmp 6b
7:
    mov r13, rdi
    sub r13, r12
wrloop:
    test r13, r13
    jle fin
    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle fin
    add r12, rax
    sub r13, rax
    jmp wrloop
fin:
    mov eax, 60
    xor edi, edi
    syscall
