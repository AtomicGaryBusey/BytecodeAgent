.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]
byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je ws
    lea edx, [rax - 9]
    cmp edx, 4
    jbe ws
    test r15d, r15d
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
    cmp al, 10
    jne next
    inc r12
next:
    dec rcx
    jnz byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outend]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call putnum
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call putnum
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call putnum
    mov rsi, rdi
    lea rdx, [rip + outend]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

putnum:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
    .balign 16
buf:
    .zero 65536
outbuf:
    .zero 96
outend:
    .zero 16
