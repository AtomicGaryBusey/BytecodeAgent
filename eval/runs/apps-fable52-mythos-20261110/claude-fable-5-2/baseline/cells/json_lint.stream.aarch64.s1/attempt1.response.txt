.globl _start
.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, #0
rd:
    mov x0, #0
    add x1, x19, x21
    mov x2, #0x1170
    movk x2, #1, lsl #16
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    ble rd_done
    add x21, x21, x0
    b rd
rd_done:
    strb wzr, [x19, x21]
    mov x20, #0
    adrp x22, stk
    add x22, x22, :lo12:stk
    mov x23, #0

parse_value:
    bl ws
    cmp w0, #'{'
    beq v_obj
    cmp w0, #'['
    beq v_arr
    cmp w0, #'"'
    beq v_str
    cmp w0, #'-'
    beq v_num
    sub w1, w0, #'0'
    cmp w1, #9
    bls v_num
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    cmp w0, #'t'
    beq v_lit
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    cmp w0, #'f'
    beq v_lit
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    cmp w0, #'n'
    beq v_lit
    b error

v_lit:
    ldrb w2, [x1], #1
    cbz w2, value_done
    ldrb w0, [x19, x20]
    cmp w0, w2
    bne error
    add x20, x20, #1
    b v_lit

v_obj:
    add x20, x20, #1
    bl ws
    cmp w0, #'}'
    bne 1f
    add x20, x20, #1
    b value_done
1:
    mov w1, #1
    strb w1, [x22, x23]
    add x23, x23, #1
    b member

v_arr:
    add x20, x20, #1
    bl ws
    cmp w0, #']'
    bne 1f
    add x20, x20, #1
    b value_done
1:
    strb wzr, [x22, x23]
    add x23, x23, #1
    b parse_value

v_str:
    bl string
    b value_done

member:
    // ws already skipped, w0 = current char
    cmp w0, #'"'
    bne error
    bl string
    bl ws
    cmp w0, #':'
    bne error
    add x20, x20, #1
    b parse_value

value_done:
    bl ws
    cbnz x23, 1f
    cmp x20, x21
    bne error
    b ok
1:
    sub x1, x23, #1
    ldrb w2, [x22, x1]
    cmp w0, #','
    bne 2f
    add x20, x20, #1
    cbz w2, parse_value
    bl ws
    b member
2:
    mov w3, #']'
    mov w4, #'}'
    cmp w2, #0
    csel w3, w3, w4, eq
    cmp w0, w3
    bne error
    add x20, x20, #1
    sub x23, x23, #1
    b value_done

v_num:
    cmp w0, #'-'
    bne 1f
    add x20, x20, #1
    ldrb w0, [x19, x20]
1:
    cmp w0, #'0'
    bne 2f
    add x20, x20, #1
    b n_frac
2:
    sub w1, w0, #'1'
    cmp w1, #8
    bhi error
3:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    bls 3b
n_frac:
    ldrb w0, [x19, x20]
    cmp w0, #'.'
    bne n_exp
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    bhi error
4:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    bls 4b
n_exp:
    orr w1, w0, #0x20
    cmp w1, #'e'
    bne value_done
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'+'
    beq 5f
    cmp w0, #'-'
    bne 6f
5:
    add x20, x20, #1
    ldrb w0, [x19, x20]
6:
    sub w1, w0, #'0'
    cmp w1, #9
    bhi error
7:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    bls 7b
    b value_done

// skip whitespace; returns current byte in w0
ws:
1:
    ldrb w0, [x19, x20]
    cmp w0, #' '
    beq 2f
    cmp w0, #9
    beq 2f
    cmp w0, #10
    beq 2f
    cmp w0, #13
    beq 2f
    ret
2:
    add x20, x20, #1
    b 1b

// parse string; pos at opening quote
string:
    add x20, x20, #1
s_loop:
    ldrb w0, [x19, x20]
    cmp w0, #0x20
    blo error
    cmp w0, #'"'
    beq s_end
    cmp w0, #'\\'
    beq s_esc
    add x20, x20, #1
    b s_loop
s_end:
    add x20, x20, #1
    ret
s_esc:
    add x20, x20, #1
    ldrb w0, [x19, x20]
    cmp w0, #'u'
    beq s_u
    cmp w0, #'"'
    beq s_ok
    cmp w0, #'\\'
    beq s_ok
    cmp w0, #'/'
    beq s_ok
    cmp w0, #'b'
    beq s_ok
    cmp w0, #'f'
    beq s_ok
    cmp w0, #'n'
    beq s_ok
    cmp w0, #'r'
    beq s_ok
    cmp w0, #'t'
    beq s_ok
    b error
s_ok:
    add x20, x20, #1
    b s_loop
s_u:
    add x20, x20, #1
    mov w2, #4
1:
    ldrb w0, [x19, x20]
    sub w1, w0, #'0'
    cmp w1, #9
    bls 2f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    bhi error
2:
    add x20, x20, #1
    subs w2, w2, #1
    bne 1b
    b s_loop

ok:
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #31
    mov w2, #10
    strb w2, [x1]
    mov x3, #1
    mov x4, #10
1:
    udiv x5, x20, x4
    msub x6, x5, x4, x20
    add w6, w6, #'0'
    strb w6, [x1, #-1]!
    add x3, x3, #1
    mov x20, x5
    cbnz x20, 1b
    mov x0, #1
    mov x2, x3
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_true:
    .asciz "true"
lit_false:
    .asciz "false"
lit_null:
    .asciz "null"

.bss
    .balign 16
numbuf:
    .space 32
stk:
    .space 2048
buf:
    .space 70016
