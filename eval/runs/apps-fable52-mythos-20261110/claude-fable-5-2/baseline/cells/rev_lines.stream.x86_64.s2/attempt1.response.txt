.globl _start
.equ CAP, 65544

.bss
buf:
    .zero 65600

.text
_start:
    lea r13, [rip + buf]
    xor r12d, r12d
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, CAP
    sub rdx, r12
    syscall
    test rax, rax
    jle done_rd
    add r12, rax
    jmp rd
done_rd:
    test r12, r12
    jz exit
    cmp byte ptr [r13 + r12 - 1], 10
    je 1f
    mov byte ptr [r13 + r12], 10
    inc r12
1:
    mov rsi, r13
    lea r8, [r13 + r12]
line:
    mov rdi, rsi
2:
    cmp byte ptr [rdi], 10
    je 3f
    inc rdi
    jmp 2b
3:
    lea r9, [rdi + 1]
    dec rdi
4:
    cmp rsi, rdi
    jae 5f
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp 4b
5:
    mov rsi, r9
    cmp rsi, r8
    jb line
wr:
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov rdx, r12
    syscall
    test rax, rax
    jle exit
    add r13, rax
    sub r12, rax
    jnz wr
exit:
    mov eax, 60
    xor edi, edi
    syscall
