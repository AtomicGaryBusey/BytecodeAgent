.globl _start

.bss
.balign 16
buf:
    .space 4096
outbuf:
    .space 64

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0              // bytes
    mov x22, #0              // in-word flag
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done
    add x21, x21, x0
    mov x3, x23
    add x4, x23, x0
byte_loop:
    ldrb w5, [x3], #1
    cmp w5, #10
    cinc x19, x19, eq
    cmp w5, #32
    b.eq is_ws
    sub w6, w5, #9
    cmp w6, #4
    b.ls is_ws
    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
next:
    cmp x3, x4
    b.lo byte_loop
    b read_loop

done:
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    add x9, x9, #64          // end of buffer
    mov x1, x9
    mov x10, #10
    mov w11, #10
    strb w11, [x1, #-1]!     // newline
    mov x0, x21
    bl conv
    mov w11, #32
    strb w11, [x1, #-1]!
    mov x0, x20
    bl conv
    mov w11, #32
    strb w11, [x1, #-1]!
    mov x0, x19
    bl conv

    sub x2, x9, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// conv: write decimal of x0 backwards at x1 (pre-decrement); x10 = 10
conv:
1:
    udiv x12, x0, x10
    msub x13, x12, x10, x0
    add w13, w13, #48
    strb w13, [x1, #-1]!
    mov x0, x12
    cbnz x0, 1b
    ret
