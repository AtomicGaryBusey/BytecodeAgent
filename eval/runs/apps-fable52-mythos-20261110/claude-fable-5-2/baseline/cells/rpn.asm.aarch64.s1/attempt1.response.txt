.globl _start

.bss
.balign 16
inbuf:  .space 70016
outbuf: .space 450000
.balign 16
vstack: .space 8192
tmp:    .space 32

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19            // end pointer
    mov x9, #70000
    add x10, x19, x9        // buffer limit
read_loop:
    mov x0, #0
    mov x1, x20
    sub x2, x10, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x22, vstack
    add x22, x22, :lo12:vstack

line_loop:
    cmp x19, x20
    b.hs finish
    mov x23, #0
tok_loop:
    cmp x19, x20
    b.hs line_end
    ldrb w0, [x19]
    cmp w0, #10
    b.eq line_end_nl
    cmp w0, #32
    b.eq 1f
    cmp w0, #9
    b.ne tok_start
1:
    add x19, x19, #1
    b tok_loop

tok_start:
    mov x1, x19
scan:
    add x19, x19, #1
    cmp x19, x20
    b.hs scan_done
    ldrb w3, [x19]
    cmp w3, #10
    b.eq scan_done
    cmp w3, #32
    b.eq scan_done
    cmp w3, #9
    b.ne scan
scan_done:
    sub x2, x19, x1         // length
    cmp x2, #1
    b.ne literal
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42
    b.eq operator
    cmp w0, #47
    b.eq operator
    cmp w0, #37
    b.eq operator
literal:
    mov x3, #0
    cmp w0, #45
    b.ne 1f
    mov x3, #1
    add x1, x1, #1
    sub x2, x2, #1
1:
    cbz x2, error
    cmp x2, #18
    b.hi error
    mov x4, #0
    mov x7, #10
2:
    ldrb w5, [x1], #1
    sub w5, w5, #48
    cmp w5, #9
    b.hi error
    madd x4, x4, x7, x5
    subs x2, x2, #1
    b.ne 2b
    cbz x3, 3f
    neg x4, x4
3:
    str x4, [x22, x23, lsl #3]
    add x23, x23, #1
    b tok_loop

operator:
    cmp x23, #2
    b.lo error
    sub x23, x23, #1
    ldr x6, [x22, x23, lsl #3]      // b
    sub x9, x23, #1
    ldr x5, [x22, x9, lsl #3]       // a
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    // division or remainder
    cbz x6, error
    cmn x6, #1
    b.ne 1f
    mov x7, #0x8000000000000000
    cmp x5, x7
    b.eq error
1:
    sdiv x7, x5, x6
    cmp w0, #47
    b.eq 2f
    msub x7, x7, x6, x5
2:
    str x7, [x22, x9, lsl #3]
    b tok_loop
op_add:
    add x7, x5, x6
    str x7, [x22, x9, lsl #3]
    b tok_loop
op_sub:
    sub x7, x5, x6
    str x7, [x22, x9, lsl #3]
    b tok_loop
op_mul:
    mul x7, x5, x6
    str x7, [x22, x9, lsl #3]
    b tok_loop

error:
    cmp x19, x20
    b.hs emit_error
    ldrb w0, [x19], #1
    cmp w0, #10
    b.ne error
emit_error:
    mov w0, #101
    strb w0, [x21], #1
    mov w0, #114
    strb w0, [x21], #1
    strb w0, [x21], #1
    mov w0, #111
    strb w0, [x21], #1
    mov w0, #114
    strb w0, [x21], #1
    mov w0, #10
    strb w0, [x21], #1
    b line_loop

line_end_nl:
    add x19, x19, #1
line_end:
    cmp x23, #1
    b.ne emit_error
    ldr x0, [x22]
    cmp x0, #0
    b.ge 1f
    mov w1, #45
    strb w1, [x21], #1
    neg x0, x0
1:
    adrp x1, tmp
    add x1, x1, :lo12:tmp
    add x1, x1, #32
    mov x2, x1              // end
    mov x7, #10
2:
    udiv x3, x0, x7
    msub x4, x3, x7, x0
    add w4, w4, #48
    strb w4, [x1, #-1]!
    mov x0, x3
    cbnz x0, 2b
3:
    ldrb w4, [x1], #1
    strb w4, [x21], #1
    cmp x1, x2
    b.lo 3b
    mov w4, #10
    strb w4, [x21], #1
    b line_loop

finish:
    adrp x19, outbuf
    add x19, x19, :lo12:outbuf
1:
    cmp x19, x21
    b.hs 2f
    mov x0, #1
    mov x1, x19
    sub x2, x21, x19
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x19, x19, x0
    b 1b
2:
    mov x0, #0
    mov x8, #93
    svc #0
