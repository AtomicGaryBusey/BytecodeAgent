.globl _start
.equ BUFSZ, 65536

.bss
.balign 16
inbuf:
    .zero 65552
outbuf:
    .zero 65568

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, BUFSZ
    sub rdx, r13
    jle done_read
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle done_read
    add r13, rax
    jmp read_loop
done_read:
    lea r14, [rip + outbuf]
    xor ebx, ebx
    xor r15d, r15d
    xor ecx, ecx
scan:
    cmp rcx, r13
    jae tail
    cmp byte ptr [r12 + rcx], 10
    je emit
    inc rcx
    jmp scan
tail:
    cmp rbx, r13
    jae do_write
emit:
    mov rsi, rcx
1:
    cmp rsi, rbx
    jbe 2f
    dec rsi
    mov al, byte ptr [r12 + rsi]
    mov byte ptr [r14 + r15], al
    inc r15
    jmp 1b
2:
    mov byte ptr [r14 + r15], 10
    inc r15
    inc rcx
    mov rbx, rcx
    jmp scan
do_write:
    xor ebx, ebx
wloop:
    cmp rbx, r15
    jae finish
    mov eax, 1
    mov edi, 1
    lea rsi, [r14 + rbx]
    mov rdx, r15
    sub rdx, rbx
    syscall
    test rax, rax
    jle finish
    add rbx, rax
    jmp wloop
finish:
    mov eax, 60
    xor edi, edi
    syscall
