.globl _start
.equ BUFSZ, 65552
.equ OUTSZ, 65536

.bss
.align 16
inbuf: .zero 65552
jtab: .zero 262208
tape: .zero 30016
outbuf: .zero 65536

.text
_start:
    xor ebx, ebx
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, BUFSZ
    sub edx, ebx
    jz rd_done
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd
rd_done:
    lea rsi, [rip + inbuf]
    xor ecx, ecx
find:
    cmp rcx, rbx
    jae notfound
    cmp byte ptr [rsi + rcx], 0x21
    je found
    inc rcx
    jmp find
found:
    mov r12, rcx
    lea r13, [rcx + 1]
    jmp chk
notfound:
    mov r12, rbx
    mov r13, rbx
chk:
    mov r14, rbx
    lea r10, [rip + jtab]
    mov rbp, rsp
    xor ecx, ecx
chk_loop:
    cmp rcx, r12
    jae chk_end
    mov al, byte ptr [rsi + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp chk_next
1:
    cmp al, 0x5D
    jne chk_next
    cmp rsp, rbp
    je fail
    pop rax
    mov dword ptr [r10 + rax*4], ecx
    mov dword ptr [r10 + rcx*4], eax
chk_next:
    inc rcx
    jmp chk_loop
chk_end:
    cmp rsp, rbp
    jne fail

    lea r8, [rip + tape]
    lea r15, [rip + outbuf]
    xor r9d, r9d
    xor ecx, ecx
run:
    cmp rcx, r12
    jae done
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp run
op_right:
    inc r8
    jmp run
op_left:
    dec r8
    jmp run
op_inc:
    inc byte ptr [r8]
    jmp run
op_dec:
    dec byte ptr [r8]
    jmp run
op_open:
    cmp byte ptr [r8], 0
    jne run
    mov ecx, dword ptr [r10 + rcx*4 - 4]
    inc rcx
    jmp run
op_close:
    cmp byte ptr [r8], 0
    je run
    mov ecx, dword ptr [r10 + rcx*4 - 4]
    inc rcx
    jmp run
op_out:
    mov al, byte ptr [r8]
    mov byte ptr [r15 + r9], al
    inc r9
    cmp r9, OUTSZ
    jne run
    call flush
    jmp run
op_in:
    cmp r13, r14
    jae in_zero
    mov al, byte ptr [rsi + r13]
    inc r13
    mov byte ptr [r8], al
    jmp run
in_zero:
    mov byte ptr [r8], 0
    jmp run

done:
    test r9, r9
    jz 1f
    call flush
1:
    mov eax, 60
    xor edi, edi
    syscall

fail:
    mov eax, 60
    mov edi, 1
    syscall

flush:
    push rcx
    push rsi
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, r9
    syscall
    xor r9d, r9d
    pop rsi
    pop rcx
    ret
