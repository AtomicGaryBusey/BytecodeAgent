.globl _start
.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x24, jt
    add x24, x24, :lo12:jt
    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x26, outbuf
    add x26, x26, :lo12:outbuf
    mov x27, #0
    mov x20, #0
    movz x9, #0x1000
    movk x9, #0x1, lsl #16
// read all input
1:
    mov x0, #0
    add x1, x19, x20
    sub x2, x9, x20
    cbz x2, 2f
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:
// find first '!'
    mov x21, #0
3:
    cmp x21, x20
    b.hs 4f
    ldrb w10, [x19, x21]
    cmp w10, #'!'
    b.eq 5f
    add x21, x21, #1
    b 3b
4:
    // no bang: input empty
    add x22, x19, x20
    b 6f
5:
    add x22, x19, x21
    add x22, x22, #1
6:
    add x23, x19, x20
// bracket pass
    adrp x11, stk
    add x11, x11, :lo12:stk
    mov x12, #0          // stack depth
    mov x13, #0          // i
7:
    cmp x13, x21
    b.hs 10f
    ldrb w10, [x19, x13]
    cmp w10, #'['
    b.eq 8f
    cmp w10, #']'
    b.eq 9f
    add x13, x13, #1
    b 7b
8:
    str w13, [x11, x12, lsl #2]
    add x12, x12, #1
    add x13, x13, #1
    b 7b
9:
    cbz x12, fail
    sub x12, x12, #1
    ldr w14, [x11, x12, lsl #2]
    str w14, [x24, x13, lsl #2]
    str w13, [x24, x14, lsl #2]
    add x13, x13, #1
    b 7b
10:
    cbnz x12, fail
// run
    mov x28, #0
run:
    cmp x28, x21
    b.hs done
    ldrb w10, [x19, x28]
    cmp w10, #'>'
    b.eq c_right
    cmp w10, #'<'
    b.eq c_left
    cmp w10, #'+'
    b.eq c_inc
    cmp w10, #'-'
    b.eq c_dec
    cmp w10, #'['
    b.eq c_open
    cmp w10, #']'
    b.eq c_close
    cmp w10, #'.'
    b.eq c_out
    cmp w10, #','
    b.eq c_in
next:
    add x28, x28, #1
    b run
c_right:
    add x25, x25, #1
    b next
c_left:
    sub x25, x25, #1
    b next
c_inc:
    ldrb w11, [x25]
    add w11, w11, #1
    strb w11, [x25]
    b next
c_dec:
    ldrb w11, [x25]
    sub w11, w11, #1
    strb w11, [x25]
    b next
c_open:
    ldrb w11, [x25]
    cbnz w11, next
    ldr w28, [x24, x28, lsl #2]
    b next
c_close:
    ldrb w11, [x25]
    cbz w11, next
    ldr w28, [x24, x28, lsl #2]
    b next
c_out:
    ldrb w11, [x25]
    strb w11, [x26, x27]
    add x27, x27, #1
    cmp x27, #65536
    b.lo next
    bl flush
    b next
c_in:
    mov w11, #0
    cmp x22, x23
    b.hs 1f
    ldrb w11, [x22], #1
1:
    strb w11, [x25]
    b next
done:
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0
fail:
    mov x0, #1
    mov x8, #93
    svc #0

flush:
    cbz x27, 1f
    mov x0, #1
    mov x1, x26
    mov x2, x27
    mov x8, #64
    svc #0
    mov x27, #0
1:
    ret
