.globl _start

.bss
.balign 16

.text

    mov x2, x1              // start

    b.hs 2f
    ldrb w3, [x1]
    cmp w3, #10
    b.eq 2f
    add x1, x1, #1
    b 1b

    add x0, x21, x22, lsl #4

    cmp x1, x20

    add x1, x1, #1

    add x0, x21, x14, lsl #4

    cmp x5, x7

    b.eq cmp_len
    ldrb w10, [x4, x15]
    ldrb w11, [x6, x15]
    add x15, x15, #1
    cmp w10, w11
    b.eq cmp_loop

    b ge

    b.lo less

    add x12, x14, #1
    b bs

    b bs

    mov x14, x23

    b.ls shift_done
    add x0, x21, x14, lsl #4
    ldp x6, x7, [x0, #-16]

    mov x2, #0              // index

    ldp x4, x5, [x0]

    strb w3, [x1], #1
    sub x5, x5, #1

    mov x1, x24
    mov x8, #64

    mov x8, #93
