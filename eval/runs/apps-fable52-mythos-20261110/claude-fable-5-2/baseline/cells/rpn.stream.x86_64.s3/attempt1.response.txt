.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov edx, 65600
    sub rdx, r13
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    mov rsi, r12
    add r13, r12
    lea rdi, [rip + outbuf]
    lea rbx, [rip + vstack]

line_loop:
    cmp rsi, r13
    jae finish
    xor r14d, r14d
token_loop:
    cmp rsi, r13
    jae line_end
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je line_end_nl
    cmp al, 32
    je 1f
    cmp al, 9
    jne tok_start
1:
    inc rsi
    jmp token_loop

tok_start:
    mov r8, rsi
2:
    inc r8
    cmp r8, r13
    jae 3f
    movzx edx, byte ptr [r8]
    cmp dl, 10
    je 3f
    cmp dl, 32
    je 3f
    cmp dl, 9
    jne 2b
3:
    mov rdx, r8
    sub rdx, rsi
    mov r9, rsi
    xor r10d, r10d
    cmp al, '-'
    jne 4f
    cmp rdx, 1
    je operator
    mov r10d, 1
    inc r9
    jmp number
4:
    cmp rdx, 1
    jne number
    cmp al, '+'
    je operator
    cmp al, '*'
    je operator
    cmp al, '/'
    je operator
    cmp al, '%'
    je operator
number:
    mov rdx, r8
    sub rdx, r9
    cmp rdx, 18
    ja line_error
    xor eax, eax
5:
    movzx ecx, byte ptr [r9]
    sub ecx, '0'
    cmp ecx, 9
    ja line_error
    lea rax, [rax + rax*4]
    lea rax, [rcx + rax*2]
    inc r9
    cmp r9, r8
    jb 5b
    test r10d, r10d
    jz 6f
    neg rax
6:
    mov qword ptr [rbx + r14*8], rax
    inc r14
    mov rsi, r8
    jmp token_loop

operator:
    cmp r14, 2
    jb line_error
    mov r9d, eax
    mov rcx, qword ptr [rbx + r14*8 - 8]
    mov rax, qword ptr [rbx + r14*8 - 16]
    dec r14
    cmp r9b, '+'
    jne 1f
    add rax, rcx
    jmp op_store
1:
    cmp r9b, '-'
    jne 1f
    sub rax, rcx
    jmp op_store
1:
    cmp r9b, '*'
    jne 1f
    imul rax, rcx
    jmp op_store
1:
    test rcx, rcx
    jz line_error
    cmp rcx, -1
    jne 2f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
2:
    cqo
    idiv rcx
    cmp r9b, '/'
    je op_store
    mov rax, rdx
op_store:
    mov qword ptr [rbx + r14*8 - 8], rax
    mov rsi, r8
    jmp token_loop

line_error:
    cmp rsi, r13
    jae emit_error
    cmp byte ptr [rsi], 10
    lea rsi, [rsi + 1]
    jne line_error
emit_error:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_loop

line_end_nl:
    inc rsi
line_end:
    cmp r14, 1
    jne emit_error
    mov rax, qword ptr [rbx]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    mov ecx, 10
    xor r8d, r8d
2:
    xor edx, edx
    div rcx
    add edx, '0'
    push rdx
    inc r8d
    test rax, rax
    jnz 2b
3:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec r8d
    jnz 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_loop

finish:
    mov r12, rdi
    lea rsi, [rip + outbuf]
1:
    mov rdx, r12
    sub rdx, rsi
    jle 2f
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    jmp 1b

    syscall
