.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space 0x20000
outbuf:
    .space 0x58000

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total length
1:
    mov x0, #0
    add x1, x19, x20
    mov x2, #0x10000
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b
2:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, x21                // output pointer
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    mov x24, #0                 // offset
    mov w9, #' '
row:
    cmp x24, x20
    b.hs done
    sub x25, x20, x24
    mov x10, #16
    cmp x25, x10
    csel x25, x25, x10, lo      // n
    mov x0, x24
    bl puthex8
    strb w9, [x22], #1
    strb w9, [x22], #1
    add x26, x19, x24           // row pointer
    mov x10, #0
hexloop:
    cmp x10, x25
    b.hs pad
    ldrb w11, [x26, x10]
    lsr w12, w11, #4
    ldrb w12, [x23, w12, uxtw]
    strb w12, [x22], #1
    and w12, w11, #15
    ldrb w12, [x23, w12, uxtw]
    strb w12, [x22], #1
    strb w9, [x22], #1
    b next
pad:
    strb w9, [x22], #1
    strb w9, [x22], #1
    strb w9, [x22], #1
next:
    cmp x10, #7
    b.ne 3f
    strb w9, [x22], #1
3:
    add x10, x10, #1
    cmp x10, #16
    b.lo hexloop
    strb w9, [x22], #1
    mov w12, #'|'
    strb w12, [x22], #1
    mov x10, #0
chrloop:
    ldrb w11, [x26, x10]
    sub w13, w11, #0x20
    cmp w13, #0x5e
    b.ls 4f
    mov w11, #'.'
4:
    strb w11, [x22], #1
    add x10, x10, #1
    cmp x10, x25
    b.lo chrloop
    strb w12, [x22], #1
    mov w12, #10
    strb w12, [x22], #1
    add x24, x24, x25
    b row
done:
    mov x0, x20
    bl puthex8
    mov w12, #10
    strb w12, [x22], #1
wloop:
    cmp x21, x22
    b.hs fin
    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le fin
    add x21, x21, x0
    b wloop
fin:
    mov x0, #0
    mov x8, #93
    svc #0

// write w0 as 8 hex digits at x22
puthex8:
    mov w13, #28
5:
    lsr w14, w0, w13
    and w14, w14, #15
    ldrb w14, [x23, w14, uxtw]
    strb w14, [x22], #1
    subs w13, w13, #4
    b.pl 5b
    ret
