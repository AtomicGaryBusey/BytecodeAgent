.globl _start
.equ CAP, 65552

.bss
.align 16
inbuf: .zero 65600
outbuf: .zero 65600

.text
_start:
    lea r12, [rip + inbuf]
    xor ebx, ebx
rd:
    mov edx, CAP
    sub rdx, rbx
    jz done_read
    lea rsi, [r12 + rbx]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle done_read
    add rbx, rax
    jmp rd
done_read:
    mov rsi, r12
    lea r13, [r12 + rbx]
    lea rdi, [rip + outbuf]
ln:
    cmp rsi, r13
    jae out
    mov rdx, rsi
sc:
    cmp rdx, r13
    je found
    cmp byte ptr [rdx], 10
    je found
    inc rdx
    jmp sc
found:
    mov rcx, rdx
cp:
    cmp rcx, rsi
    je cpd
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp cp
cpd:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rdx + 1]
    jmp ln
out:
    lea rsi, [rip + outbuf]
    mov r14, rdi
    sub r14, rsi
wr:
    test r14, r14
    jle fin
    mov rdx, r14
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle fin
    add rsi, rax
    sub r14, rax
    jmp wr
fin:
    xor edi, edi
    mov eax, 60
    syscall
