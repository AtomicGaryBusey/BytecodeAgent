.globl _start
.equ INCAP, 70000
.equ OUTCAP, 460000

.bss
.align 16
inbuf: .zero INCAP
.align 16
outbuf: .zero OUTCAP
.align 16
stk: .zero 8192
tmp: .zero 32
tmpend: .zero 16

.text
_start:
    lea r14, [rip + inbuf]
    xor ebx, ebx
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + rbx]
    mov edx, INCAP
    sub rdx, rbx
    jz rd_done
    syscall
    test rax, rax
    jle rd_done
    add rbx, rax
    jmp rd
rd_done:
    mov rsi, r14
    lea r15, [r14 + rbx]
    lea rdi, [rip + outbuf]
    lea r13, [rip + stk]

line_start:
    cmp rsi, r15
    jae finish
    xor r12d, r12d
tok_loop:
    cmp rsi, r15
    jae end_line
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je nl
    cmp al, 32
    je skipsep
    cmp al, 9
    je skipsep
    mov rcx, rsi
scan:
    inc rcx
    cmp rcx, r15
    jae scan_done
    mov dl, byte ptr [rcx]
    cmp dl, 32
    je scan_done
    cmp dl, 9
    je scan_done
    cmp dl, 10
    je scan_done
    jmp scan
scan_done:
    mov rdx, rcx
    sub rdx, rsi
    cmp rdx, 1
    jne number
    cmp al, '+'
    je op
    cmp al, '-'
    je op
    cmp al, '*'
    je op
    cmp al, '/'
    je op
    cmp al, '%'
    je op
number:
    xor r8d, r8d
    mov rbx, rsi
    cmp al, '-'
    jne 1f
    inc r8d
    inc rbx
1:
    mov rdx, rcx
    sub rdx, rbx
    jz error
    cmp rdx, 18
    ja error
    xor eax, eax
2:
    movzx edx, byte ptr [rbx]
    sub edx, '0'
    cmp edx, 9
    ja error
    lea rax, [rax + rax*4]
    lea rax, [rdx + rax*2]
    inc rbx
    cmp rbx, rcx
    jb 2b
    test r8d, r8d
    jz 3f
    neg rax
3:
    mov qword ptr [r13 + r12*8], rax
    inc r12
    mov rsi, rcx
    jmp tok_loop

skipsep:
    inc rsi
    jmp tok_loop

op:
    cmp r12, 2
    jb error
    mov r10d, eax
    mov r9, qword ptr [r13 + r12*8 - 8]
    mov rax, qword ptr [r13 + r12*8 - 16]
    dec r12
    mov rsi, rcx
    cmp r10b, '+'
    je op_add
    cmp r10b, '-'
    je op_sub
    cmp r10b, '*'
    je op_mul
    # division or remainder
    test r9, r9
    jz error
    cmp r9, -1
    jne 1f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error
1:
    cqo
    idiv r9
    cmp r10b, '/'
    je op_store
    mov rax, rdx
    jmp op_store
op_add:
    add rax, r9
    jmp op_store
op_sub:
    sub rax, r9
    jmp op_store
op_mul:
    imul rax, r9
op_store:
    mov qword ptr [r13 + r12*8 - 8], rax
    jmp tok_loop

nl:
    inc rsi
end_line:
    cmp r12, 1
    jne emit_error
    mov rax, qword ptr [r13]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    lea r9, [rip + tmpend]
    mov rbx, r9
    mov r8d, 10
2:
    xor edx, edx
    div r8
    add dl, '0'
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [rbx]
    mov byte ptr [rdi], al
    inc rbx
    inc rdi
    cmp rbx, r9
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_start

error:
    cmp rsi, r15
    jae emit_error
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error
emit_error:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_start

finish:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
wr:
    test rbx, rbx
    jle done
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    test rax, rax
    jle done
    add rsi, rax
    sub rbx, rax
    jmp wr
done:
    mov eax, 60
    xor edi, edi
    syscall
