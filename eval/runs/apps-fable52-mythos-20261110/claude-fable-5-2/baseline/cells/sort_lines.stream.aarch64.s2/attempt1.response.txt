.globl _start

.equ INCAP, 32784

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
read_loop:
    mov x2, #INCAP
    subs x2, x2, x20
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x21, entries
    add x21, x21, :lo12:entries
    mov x22, #0                 // line count
    mov x1, #0                  // position
parse:
    cmp x1, x20
    b.hs parsed
    mov x2, x1
scan:
    cmp x1, x20
    b.hs endline
    ldrb w3, [x19, x1]
    cmp w3, #10
    b.eq endline
    add x1, x1, #1
    b scan
endline:
    sub x4, x1, x2              // length
    add x5, x19, x2             // pointer
    add x6, x21, x22, lsl #4
    stp x5, x4, [x6]
    add x22, x22, #1
    add x1, x1, #1
    b parse
parsed:
    // binary insertion sort
    mov x24, #1                 // i
sort_outer:
    cmp x24, x22
    b.hs sorted
    add x12, x21, x24, lsl #4
    ldp x4, x5, [x12]           // key ptr, len
    mov x10, #0                 // lo
    mov x11, x24                // hi
bs_loop:
    cmp x10, x11
    b.hs bs_done
    add x15, x10, x11
    lsr x15, x15, #1            // mid
    add x12, x21, x15, lsl #4
    ldp x6, x7, [x12]
    cmp x5, x7
    csel x9, x5, x7, lo         // min length
    mov x13, #0
cmp_loop:
    cmp x13, x9
    b.hs cmp_tie
    ldrb w14, [x4, x13]
    ldrb w16, [x6, x13]
    cmp w14, w16
    b.lo is_less
    b.hi not_less
    add x13, x13, #1
    b cmp_loop
cmp_tie:
    cmp x5, x7
    b.lo is_less
not_less:
    add x10, x15, #1
    b bs_loop
is_less:
    mov x11, x15
    b bs_loop
bs_done:
    mov x13, x24                // j
shift_loop:
    cmp x13, x10
    b.ls shift_done
    add x12, x21, x13, lsl #4
    ldp x14, x16, [x12, #-16]
    stp x14, x16, [x12]
    sub x13, x13, #1
    b shift_loop
shift_done:
    add x12, x21, x10, lsl #4
    stp x4, x5, [x12]
    add x24, x24, #1
    b sort_outer
sorted:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x25, x23                // write pointer
    mov x24, #0
out_loop:
    cmp x24, x22
    b.hs out_done
    add x12, x21, x24, lsl #4
    ldp x4, x5, [x12]
copy_loop:
    cbz x5, copy_done
    ldrb w6, [x4], #1
    strb w6, [x25], #1
    sub x5, x5, #1
    b copy_loop
copy_done:
    mov w6, #10
    strb w6, [x25], #1
    add x24, x24, #1
    b out_loop
out_done:
    sub x26, x25, x23           // remaining bytes
write_loop:
    cbz x26, finish
    mov x0, #1
    mov x1, x23
    mov x2, x26
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le finish
    add x23, x23, x0
    sub x26, x26, x0
    b write_loop
finish:
    mov x0, #0
    mov x8, #93
    svc #0
    b finish

.bss
.balign 16
inbuf:
    .space 32800
.balign 16
outbuf:
    .space 32800
.balign 16
