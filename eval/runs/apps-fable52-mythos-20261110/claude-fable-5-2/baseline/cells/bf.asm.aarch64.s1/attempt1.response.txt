    mov x20, #0

    mov x8, #63

    b.hs find_done
    ldrb w9, [x19, x21]
    cmp w9, #'!'
    b.eq find_done
    add x21, x21, #1
    b find_loop

    add x22, x22, #1
    add x23, x19, x20

    mov w9, #-1          // top
    mov x10, #0          // i

    b.hs chk_done
    ldrb w11, [x19, x10]
    cmp w11, #'['
    b.eq chk_open
    cmp w11, #']'

    mov w12, w9                  // j
    ldr w9, [x24, x12, lsl #2]   // top = jump[j]
    str w10, [x24, x12, lsl #2]  // jump[j] = i
    str w12, [x24, x10, lsl #2]  // jump[i] = j
    b chk_next
chk_done:
    tbz w9, #31, fail

    mov x28, #0
run_loop:
    cmp x28, x21
    b.hs run_done
    ldrb w9, [x19, x28]
    cmp w9, #'>'
    b.eq op_right
    cmp w9, #'<'
    b.eq op_left
    cmp w9, #'+'
    b.eq op_inc
    cmp w9, #'-'
    b.eq op_dec
    cmp w9, #'['
    b.eq op_open
    cmp w9, #']'
    b.eq op_close
    cmp w9, #'.'
    b.eq op_out
    cmp w9, #','

run_next:
    add x28, x28, #1
    b run_loop
op_right:
    add x25, x25, #1
    b run_next
op_left:

    add w10, w10, #1
    strb w10, [x25]

    sub w10, w10, #1

    b run_next

    ldr w28, [x24, x28, lsl #2]
    b run_next

    ldr w28, [x24, x28, lsl #2]

    cmp x27, x9

1:

    b run_next

    mov x8, #93
    svc #0

    mov x8, #93

    mov x8, #64

    sub x11, x11, x0
