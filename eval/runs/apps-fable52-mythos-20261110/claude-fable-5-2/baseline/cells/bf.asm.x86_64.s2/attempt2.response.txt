.globl _start
.equ BUFSZ, 65552

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
L_readloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, BUFSZ
    sub rdx, r13
    jz L_readdone
    syscall
    test rax, rax
    jle L_readdone
    add r13, rax
    jmp L_readloop
L_readdone:
    xor ecx, ecx
L_findloop:
    cmp rcx, r13
    jae L_notfound
    cmp byte ptr [r12 + rcx], 0x21
    je L_found
    inc rcx
    jmp L_findloop
L_found:
    mov r14, rcx
    lea r15, [rcx + 1]
    jmp L_check
L_notfound:
    mov r14, r13
    mov r15, r13
L_check:
    lea r9, [rip + jtab]
    mov rbp, rsp
    xor ecx, ecx
L_chkloop:
    cmp rcx, r14
    jae L_chkend
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp L_chknext
1:
    cmp al, 0x5D
    jne L_chknext
    cmp rsp, rbp
    je L_bad
    pop rdx
    mov dword ptr [r9 + rdx*4], ecx
    mov dword ptr [r9 + rcx*4], edx
L_chknext:
    inc rcx
    jmp L_chkloop
L_chkend:
    cmp rsp, rbp
    jne L_bad

    lea rbx, [rip + tape]
    lea r10, [rip + outbuf]
    xor r8d, r8d
    xor ecx, ecx
L_run:
    cmp rcx, r14
    jae L_fin
    movzx eax, byte ptr [r12 + rcx]
    inc rcx
    cmp al, 0x3E
    je L_op_right
    cmp al, 0x3C
    je L_op_left
    cmp al, 0x2B
    je L_op_inc
    cmp al, 0x2D
    je L_op_dec
    cmp al, 0x5B
    je L_op_open
    cmp al, 0x5D
    je L_op_close
    cmp al, 0x2E
    je L_op_out
    cmp al, 0x2C
    je L_op_in
    jmp L_run
L_op_right:
    inc rbx
    jmp L_run
L_op_left:
    dec rbx
    jmp L_run
L_op_inc:
    inc byte ptr [rbx]
    jmp L_run
L_op_dec:
    dec byte ptr [rbx]
    jmp L_run
L_op_open:
    cmp byte ptr [rbx], 0
    jne L_run
    mov ecx, dword ptr [r9 + rcx*4 - 4]
    inc rcx
    jmp L_run
L_op_close:
    cmp byte ptr [rbx], 0
    je L_run
    mov ecx, dword ptr [r9 + rcx*4 - 4]
    inc rcx
    jmp L_run
L_op_out:
    cmp r8, 65536
    jb 1f
    push rcx
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r8
    syscall
    pop rcx
    xor r8d, r8d
1:
    mov al, byte ptr [rbx]
    mov byte ptr [r10 + r8], al
    inc r8
    jmp L_run
L_op_in:
    xor eax, eax
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
1:
    mov byte ptr [rbx], al
    jmp L_run
L_fin:
    test r8, r8
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r8
    syscall
1:
    mov eax, 60
    xor edi, edi
    syscall
L_bad:
    mov eax, 60
    mov edi, 1
    syscall

.bss
    .align 16
inbuf:
    .zero 65552
    .align 16
jtab:
    .zero 262208
    .align 16
tape:
    .zero 30016
    .align 16
outbuf:
    .zero 65536
