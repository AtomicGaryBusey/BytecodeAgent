.globl _start
.equ BUFSZ, 65552

.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
rl:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, BUFSZ
    sub rdx, r13
    jz rdone
    syscall
    test rax, rax
    jle rdone
    add r13, rax
    jmp rl
rdone:
    xor ecx, ecx
fl:
    cmp rcx, r13
    jae nf
    cmp byte ptr [r12 + rcx], 0x21
    je found
    inc rcx
    jmp fl
found:
    mov r14, rcx
    lea r15, [rcx + 1]
    jmp chk
nf:
    mov r14, r13
    mov r15, r13
chk:
    lea r9, [rip + jt]
    mov rbp, rsp
    xor ecx, ecx
cl:
    cmp rcx, r14
    jae cend
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    jne 1f
    push rcx
    jmp cn
1:
    cmp al, 0x5D
    jne cn
    cmp rsp, rbp
    je bad
    pop rdx
    mov dword ptr [r9 + rdx*4], ecx
    mov dword ptr [r9 + rcx*4], edx
cn:
    inc rcx
    jmp cl
cend:
    cmp rsp, rbp
    jne bad

    lea rbx, [rip + tape]
    lea r10, [rip + outbuf]
    xor r8d, r8d
    xor ecx, ecx
run:
    cmp rcx, r14
    jae fin
    movzx eax, byte ptr [r12 + rcx]
    inc rcx
    cmp al, 0x3E
    je op_right
    cmp al, 0x3C
    je op_left
    cmp al, 0x2B
    je op_inc
    cmp al, 0x2D
    je op_dec
    cmp al, 0x5B
    je op_open
    cmp al, 0x5D
    je op_close
    cmp al, 0x2E
    je op_out
    cmp al, 0x2C
    je op_in
    jmp run
op_right:
    inc rbx
    jmp run
op_left:
    dec rbx
    jmp run
op_inc:
    inc byte ptr [rbx]
    jmp run
op_dec:
    dec byte ptr [rbx]
    jmp run
op_open:
    cmp byte ptr [rbx], 0
    jne run
    mov ecx, dword ptr [r9 + rcx*4 - 4]
    inc rcx
    jmp run
op_close:
    cmp byte ptr [rbx], 0
    je run
    mov ecx, dword ptr [r9 + rcx*4 - 4]
    inc rcx
    jmp run
op_out:
    cmp r8, 65536
    jb 1f
    # flush full buffer
    mov r11, rcx
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r8
    push r11
    syscall
    pop rcx
    xor r8d, r8d
1:
    mov al, byte ptr [rbx]
    mov byte ptr [r10 + r8], al
    inc r8
    jmp run
op_in:
    xor eax, eax
    cmp r15, r13
    jae 1f
    mov al, byte ptr [r12 + r15]
    inc r15
1:
    mov byte ptr [rbx], al
    jmp run
fin:
    test r8, r8
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r8
    syscall
1:
    mov eax, 60
    xor edi, edi
    syscall
bad:
    mov eax, 60
    mov edi, 1
    syscall

.bss
    .align 16
inbuf:
    .zero 65552
    .align 16
jt:
    .zero 262208
    .align 16
tape:
    .zero 30016
    .align 16
outbuf:
    .zero 65536
