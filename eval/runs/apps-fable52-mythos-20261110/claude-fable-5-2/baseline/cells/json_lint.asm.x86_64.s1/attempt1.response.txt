.globl _start

.section .rodata
msg_ok: .ascii "ok\n"
msg_err: .ascii "error at "
lit_true: .asciz "true"
lit_false: .asciz "false"
lit_null: .asciz "null"

.bss
buf: .zero 65664
nstack: .zero 2048
numbuf: .zero 32

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, 65600
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + buf]
    lea r15, [rsi + r12]
    mov byte ptr [r15], 0
    lea r14, [rip + nstack]
    xor ebx, ebx

parse_value:
    call skip_ws
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je open_obj
    cmp al, '['
    je open_arr
    cmp al, '"'
    je val_string
    cmp al, '-'
    je number
    cmp al, 't'
    je val_true
    cmp al, 'f'
    je val_false
    cmp al, 'n'
    je val_null
    mov ecx, eax
    sub ecx, '0'
    cmp ecx, 10
    jb number
    jmp error

open_obj:
    mov byte ptr [r14 + rbx], 1
    inc rbx
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], '}'
    jne obj_key
    inc rsi
    dec rbx
    jmp value_done

open_arr:
    mov byte ptr [r14 + rbx], 0
    inc rbx
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], ']'
    jne parse_value
    inc rsi
    dec rbx
    jmp value_done

obj_key:
    cmp byte ptr [rsi], '"'
    jne error
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], ':'
    jne error
    inc rsi
    jmp parse_value

val_string:
    call parse_string
    jmp value_done

val_true:
    lea rdi, [rip + lit_true]
    jmp literal
val_false:
    lea rdi, [rip + lit_false]
    jmp literal
val_null:
    lea rdi, [rip + lit_null]
literal:
    mov dl, byte ptr [rdi]
    test dl, dl
    jz value_done
    cmp dl, byte ptr [rsi]
    jne error
    inc rdi
    inc rsi
    jmp literal

number:
    cmp byte ptr [rsi], '-'
    jne 1f
    inc rsi
1:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jne 2f
    inc rsi
    jmp num_frac
2:
    sub eax, '1'
    cmp eax, 9
    jae error
    inc rsi
3:
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 10
    jae num_frac
    inc rsi
    jmp 3b
num_frac:
    cmp byte ptr [rsi], '.'
    jne num_exp
    inc rsi
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 10
    jae error
4:
    inc rsi
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 10
    jb 4b
num_exp:
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne value_done
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je 5f
    cmp al, '-'
    jne 6f
5:
    inc rsi
6:
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 10
    jae error
7:
    inc rsi
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 10
    jb 7b
    jmp value_done

value_done:
    call skip_ws
    test rbx, rbx
    jnz 1f
    cmp rsi, r15
    jne error
    jmp success
1:
    movzx eax, byte ptr [rsi]
    cmp byte ptr [r14 + rbx - 1], 1
    jne in_arr
    cmp al, ','
    jne 2f
    inc rsi
    call skip_ws
    jmp obj_key
2:
    cmp al, '}'
    jne error
    inc rsi
    dec rbx
    jmp value_done
in_arr:
    cmp al, ','
    jne 3f
    inc rsi
    jmp parse_value
3:
    cmp al, ']'
    jne error
    inc rsi
    dec rbx
    jmp value_done

skip_ws:
    movzx eax, byte ptr [rsi]
    cmp al, ' '
    je 1f
    cmp al, 9
    je 1f
    cmp al, 10
    je 1f
    cmp al, 13
    je 1f
    ret
1:
    inc rsi
    jmp skip_ws

parse_string:
    inc rsi
ps_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je ps_ret
    cmp al, 0x5C
    jne ps_loop
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, '/'
    je ps_esc_ok
    cmp al, 'b'
    je ps_esc_ok
    cmp al, 'f'
    je ps_esc_ok
    cmp al, 'n'
    je ps_esc_ok
    cmp al, 'r'
    je ps_esc_ok
    cmp al, 't'
    je ps_esc_ok
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
ps_hex:
    movzx eax, byte ptr [rsi]
    mov edx, eax
    sub edx, '0'
    cmp edx, 10
    jb 1f
    or eax, 0x20
    sub eax, 'a'
    cmp eax, 6
    jae error
1:
    inc rsi
    dec ecx
    jnz ps_hex
    jmp ps_loop
ps_esc_ok:
    inc rsi
    jmp ps_loop
ps_ret:
    ret

success:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    lea rax, [rip + buf]
    sub rsi, rax
    mov r13, rsi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea r8, [rip + numbuf + 31]
    mov byte ptr [r8], 10
    mov rax, r13
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
    lea rdx, [rip + numbuf + 32]
    sub rdx, r8
    mov rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
