.globl _start
.equ INSZ, 65600
.equ OUTSZ, 524288
.equ STKSZ, 320000

.bss
.align 16
inbuf: .zero INSZ
outbuf: .zero OUTSZ
vstack: .zero STKSZ
tmp: .zero 32

.text
_start:
    lea r14, [rip + inbuf]
    xor r15d, r15d
read_loop:
    mov edx, INSZ
    sub rdx, r15
    jle read_done
    lea rsi, [r14 + r15]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    add r15, r14
    mov rsi, r14
    lea rdi, [rip + outbuf]
    lea r12, [rip + vstack]

line_start:
    cmp rsi, r15
    jae finish
    xor r13d, r13d
tok_loop:
    cmp rsi, r15
    jae end_line
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne 1f
    inc rsi
    jmp end_line
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jne token
2:
    inc rsi
    jmp tok_loop

token:
    mov rbx, rsi
    mov rcx, rsi
3:
    inc rcx
    cmp rcx, r15
    jae 4f
    movzx eax, byte ptr [rcx]
    cmp al, 10
    je 4f
    cmp al, 32
    je 4f
    cmp al, 9
    jne 3b
4:
    mov rdx, rcx
    sub rdx, rbx
    movzx r10d, byte ptr [rbx]
    cmp rdx, 1
    jne number
    cmp r10b, '+'
    je op
    cmp r10b, '-'
    je op
    cmp r10b, '*'
    je op
    cmp r10b, '/'
    je op
    cmp r10b, '%'
    je op
number:
    mov r8, rbx
    xor r9d, r9d
    cmp r10b, '-'
    jne 5f
    mov r9d, 1
    inc r8
5:
    mov rdx, rcx
    sub rdx, r8
    cmp rdx, 1
    jl error_skip
    cmp rdx, 18
    jg error_skip
    xor eax, eax
6:
    movzx edx, byte ptr [r8]
    sub edx, '0'
    cmp edx, 9
    ja error_skip
    lea rax, [rax + rax*4]
    lea rax, [rdx + rax*2]
    inc r8
    cmp r8, rcx
    jb 6b
    test r9d, r9d
    jz 7f
    neg rax
7:
    mov qword ptr [r12 + r13*8], rax
    inc r13
    mov rsi, rcx
    jmp tok_loop

op:
    mov rsi, rcx
    cmp r13, 2
    jb error_skip
    mov r9, qword ptr [r12 + r13*8 - 8]
    mov rax, qword ptr [r12 + r13*8 - 16]
    dec r13
    cmp r10b, '+'
    jne 1f
    add rax, r9
    jmp op_store
1:
    cmp r10b, '-'
    jne 1f
    sub rax, r9
    jmp op_store
1:
    cmp r10b, '*'
    jne 1f
    imul rax, r9
    jmp op_store
1:
    test r9, r9
    jz error_skip
    cmp r9, -1
    jne 2f
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error_skip
2:
    cqo
    idiv r9
    cmp r10b, '/'
    je op_store
    mov rax, rdx
op_store:
    mov qword ptr [r12 + r13*8 - 8], rax
    jmp tok_loop

error_skip:
    cmp rsi, r15
    jae error_emit
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne error_skip
error_emit:
    mov dword ptr [rdi], 0x6f727265
    mov word ptr [rdi + 4], 0x0a72
    add rdi, 6
    jmp line_start

end_line:
    cmp r13, 1
    jne error_emit
    mov rax, qword ptr [r12]
    test rax, rax
    jns 1f
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
1:
    lea r8, [rip + tmp + 32]
    mov r9, r8
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
3:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb 3b
    mov byte ptr [rdi], 10
    inc rdi
    jmp line_start

finish:
    lea rsi, [rip + outbuf]
    mov rbx, rdi
    sub rbx, rsi
wloop:
    test rbx, rbx
    jle wdone
    mov rdx, rbx
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle wdone
    add rsi, rax
    sub rbx, rax
    jmp wloop
wdone:
    xor edi, edi
    mov eax, 60
    syscall
