.globl _start
.text
_start:
    xor r12d, r12d
    lea r15, [rip + hexd]
rd:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65600
    sub rdx, r12
    jle done
    syscall
    test rax, rax
    jle done
    add r12, rax
    jmp rd
done:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + inbuf]
    xor r13d, r13d
rowloop:
    cmp r13, r12
    jae final
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14d, 16
1:
    mov eax, r13d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor ecx, ecx
2:
    cmp rcx, r14
    jae 3f
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp 4f
3:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
4:
    add rdi, 3
    cmp ecx, 7
    jne 5f
    mov byte ptr [rdi], 0x20
    inc rdi
5:
    inc ecx
    cmp ecx, 16
    jb 2b
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
6:
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp edx, 0x5e
    jbe 7f
    mov al, 0x2e
7:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    cmp rcx, r14
    jb 6b
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add rsi, r14
    add r13, r14
    jmp rowloop
final:
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rbx, [rip + outbuf]
    mov r13, rdi
    sub r13, rbx
wr:
    test r13, r13
    jle fin
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r13
    syscall
    test rax, rax
    jle fin
    add rbx, rax
    sub r13, rax
    jmp wr
fin:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .zero 65664
outbuf:
    .zero 330000
