    jle 2f

2:

    xor r15d, r15d
    lea rbx, [rip + lines]
3:

4:

    jae 6f

    jae 6f

6:

7:

    jae 12f

    mov r11, qword ptr [rbx + rax + 8]

8:

    jz 11f
    mov rax, r9
    shl rax, 4
    add rax, rbx
    mov rsi, qword ptr [rax - 16]
    mov rdx, qword ptr [rax - 8]

    mov qword ptr [rbx + rax + 8], r11

    mov rcx, qword ptr [rbx + rax + 8]

    jle 16f
