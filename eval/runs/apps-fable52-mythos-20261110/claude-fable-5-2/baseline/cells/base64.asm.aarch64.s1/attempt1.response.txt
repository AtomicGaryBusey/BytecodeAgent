.section .rodata

    .space 65536

    .space 131072

.text

    add x1, x19, x20

    cbz x2, read_done
    mov x8, #63

    mov x26, x23                // output base
    mov x24, #0                 // column
    adrp x25, table
    add x25, x25, :lo12:table

    cmp x22, #3
    b.lo tail
    ldrb w0, [x21]
    ldrb w1, [x21, #1]
    ldrb w2, [x21, #2]

    ldrb w1, [x25, w1, uxtw]

    strb w1, [x23], #1
    sub x22, x22, #3
    add x24, x24, #4
    cmp x24, #76
    b.ne enc_loop

    lsl w0, w0, #16
    cmp x22, #2
    b.ne 1f
    ldrb w1, [x21, #1]

    strb w1, [x23], #1
    b 3f

    strb w2, [x23], #1
3:
    strb w2, [x23], #1
    add x24, x24, #4

enc_done:
    cbz x24, 4f
    strb w27, [x23], #1

    sub x22, x23, x26           // bytes to write
    mov x21, x26

    mov x0, #1
    mov x1, x21
    mov x2, x22
