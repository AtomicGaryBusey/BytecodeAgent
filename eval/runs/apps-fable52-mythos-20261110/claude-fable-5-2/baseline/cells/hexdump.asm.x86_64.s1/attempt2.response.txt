.globl _start
.text
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 70000
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rdi, [rip + outbuf]
    xor ebx, ebx
    lea r14, [rip + hexd]
rowloop:
    cmp rbx, r13
    jae final_line
    mov eax, ebx
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    mov r15, r13
    sub r15, rbx
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    lea rsi, [r12 + rbx]
    xor ecx, ecx
byte_loop:
    cmp rcx, r15
    jae pad_slot
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r14 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r14 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp next_slot
pad_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
next_slot:
    add rdi, 3
    cmp ecx, 7
    jne 1f
    mov byte ptr [rdi], 0x20
    inc rdi
1:
    inc ecx
    cmp ecx, 16
    jb byte_loop
    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor ecx, ecx
char_loop:
    movzx eax, byte ptr [rsi + rcx]
    lea edx, [rax - 0x20]
    cmp dl, 0x5e
    jbe 1f
    mov al, 0x2e
1:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    cmp rcx, r15
    jb char_loop
    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add rbx, r15
    jmp rowloop
final_line:
    mov eax, r13d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rbx, [rip + outbuf]
    mov r15, rdi
write_loop:
    mov rdx, r15
    sub rdx, rbx
    jle finish
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall
    test rax, rax
    jle finish
    add rbx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r14 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .space 70016
outbuf:
    .space 400000
