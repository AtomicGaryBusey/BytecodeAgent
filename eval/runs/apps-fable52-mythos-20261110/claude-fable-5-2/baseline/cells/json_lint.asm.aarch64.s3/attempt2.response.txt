.globl _start

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, x19
    mov x21, #0x20000
    add x21, x19, x21
rd_loop:
    sub x2, x21, x20
    cbz x2, rd_done
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le rd_done
    add x20, x20, x0
    b rd_loop
rd_done:
    mov x21, x20
    mov x20, x19
    mov x22, #0
    adrp x23, stk
    add x23, x23, :lo12:stk

parse_value:
    bl skipws
    bl peek
    cmp w0, #'{'
    b.eq v_obj
    cmp w0, #'['
    b.eq v_arr
    cmp w0, #'"'
    b.eq v_str
    cmp w0, #'-'
    b.eq v_num
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls v_num
    cmp w0, #'t'
    b.eq v_true
    cmp w0, #'f'
    b.eq v_false
    cmp w0, #'n'
    b.eq v_null
    b err

v_obj:
    add x20, x20, #1
    bl skipws
    bl peek
    cmp w0, #'}'
    b.ne 1f
    add x20, x20, #1
    b value_done
1:
    mov w1, #2
    strb w1, [x23, x22]
    add x22, x22, #1
    b member

v_arr:
    add x20, x20, #1
    bl skipws
    bl peek
    cmp w0, #']'
    b.ne 1f
    add x20, x20, #1
    b value_done
1:
    mov w1, #1
    strb w1, [x23, x22]
    add x22, x22, #1
    b parse_value

v_str:
    bl string
    b value_done

v_true:
    adrp x3, s_true
    add x3, x3, :lo12:s_true
    b literal
v_false:
    adrp x3, s_false
    add x3, x3, :lo12:s_false
    b literal
v_null:
    adrp x3, s_null
    add x3, x3, :lo12:s_null
literal:
    ldrb w2, [x3], #1
    cbz w2, value_done
    bl peek
    cmp w0, w2
    b.ne err
    add x20, x20, #1
    b literal

v_num:
    cmp w0, #'-'
    b.ne 1f
    add x20, x20, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi err
1:
    add x20, x20, #1
    cmp w0, #'0'
    b.eq 2f
    bl skipdigits
2:
    cmp x20, x21
    b.hs value_done
    ldrb w0, [x20]
    cmp w0, #'.'
    b.ne 3f
    add x20, x20, #1
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi err
    bl skipdigits
    cmp x20, x21
    b.hs value_done
    ldrb w0, [x20]
3:
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne value_done
    add x20, x20, #1
    bl peek
    cmp w0, #'+'
    b.eq 4f
    cmp w0, #'-'
    b.ne 5f
4:
    add x20, x20, #1
    bl peek
5:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi err
    bl skipdigits
    b value_done

member:
    bl skipws
    bl peek
    cmp w0, #'"'
    b.ne err
    bl string
    bl skipws
    bl peek
    cmp w0, #':'
    b.ne err
    add x20, x20, #1
    b parse_value

value_done:
    bl skipws
    cbnz x22, 1f
    cmp x20, x21
    b.ne err
    // ok
    mov x0, #1
    adrp x1, s_ok
    add x1, x1, :lo12:s_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
1:
    bl peek
    sub x2, x22, #1
    ldrb w1, [x23, x2]
    cmp w0, #','
    b.ne 2f
    add x20, x20, #1
    cmp w1, #1
    b.eq parse_value
    b member
2:
    cmp w1, #1
    b.ne 3f
    cmp w0, #']'
    b.ne err
    b 4f
3:
    cmp w0, #'}'
    b.ne err
4:
    add x20, x20, #1
    sub x22, x22, #1
    b value_done

// string: x20 at opening quote
string:
    mov x24, x30
    add x20, x20, #1
str_loop:
    bl peek
    cmp w0, #'"'
    b.eq str_end
    cmp w0, #0x20
    b.lo err
    add x20, x20, #1
    cmp w0, #'\\'
    b.ne str_loop
    bl peek
    cmp w0, #'u'
    b.eq str_u
    cmp w0, #'"'
    b.eq str_esc_ok
    cmp w0, #'\\'
    b.eq str_esc_ok
    cmp w0, #'/'
    b.eq str_esc_ok
    cmp w0, #'b'
    b.eq str_esc_ok
    cmp w0, #'f'
    b.eq str_esc_ok
    cmp w0, #'n'
    b.eq str_esc_ok
    cmp w0, #'r'
    b.eq str_esc_ok
    cmp w0, #'t'
    b.ne err
str_esc_ok:
    add x20, x20, #1
    b str_loop
str_u:
    add x20, x20, #1
    mov w3, #4
1:
    bl peek
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls 2f
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi err
2:
    add x20, x20, #1
    subs w3, w3, #1
    b.ne 1b
    b str_loop
str_end:
    add x20, x20, #1
    mov x30, x24
    ret

skipws:
    cmp x20, x21
    b.hs 2f
    ldrb w0, [x20]
    cmp w0, #' '
    b.eq 1f
    cmp w0, #9
    b.eq 1f
    cmp w0, #10
    b.eq 1f
    cmp w0, #13
    b.ne 2f
1:
    add x20, x20, #1
    b skipws
2:
    ret

skipdigits:
    cmp x20, x21
    b.hs 1f
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi 1f
    add x20, x20, #1
    b skipdigits
1:
    ret

peek:
    cmp x20, x21
    b.hs err
    ldrb w0, [x20]
    ret

err:
    sub x4, x20, x19
    adrp x5, numbuf
    add x5, x5, :lo12:numbuf
    add x1, x5, #23
    mov w2, #10
    strb w2, [x1]
    mov x3, #10
1:
    udiv x6, x4, x3
    msub x7, x6, x3, x4
    add w7, w7, #'0'
    strb w7, [x1, #-1]!
    mov x4, x6
    cbnz x4, 1b
    mov x9, x1
    add x10, x5, #24
    sub x10, x10, x9
    mov x0, #1
    adrp x1, s_err
    add x1, x1, :lo12:s_err
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
