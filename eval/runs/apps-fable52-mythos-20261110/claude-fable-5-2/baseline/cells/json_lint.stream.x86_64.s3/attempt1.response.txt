.globl _start
.text
_start:
    lea rbx, [rip + buf]
    mov r12, rbx
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov edx, 65536
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov rsi, rbx
    mov rdi, r12
    xor r12d, r12d
    lea r13, [rip + stk]

value:
    call skip_ws
    cmp al, '{'
    je obj_start
    cmp al, '['
    je arr_start
    cmp al, '"'
    jne 1f
    call string
    jmp value_done
1:
    lea rdx, [rip + s_true]
    cmp al, 't'
    je literal
    lea rdx, [rip + s_false]
    cmp al, 'f'
    je literal
    lea rdx, [rip + s_null]
    cmp al, 'n'
    je literal
    cmp al, '-'
    jne num_int
    inc rsi
    call peek
num_int:
    cmp al, '0'
    jne 1f
    inc rsi
    jmp num_frac
1:
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
2:
    inc rsi
    call peek
    sub al, '0'
    cmp al, 9
    jbe 2b
num_frac:
    call peek
    cmp al, '.'
    jne num_exp
    inc rsi
    call peek
    sub al, '0'
    cmp al, 9
    ja error
3:
    inc rsi
    call peek
    sub al, '0'
    cmp al, 9
    jbe 3b
num_exp:
    call peek
    or al, 0x20
    cmp al, 'e'
    jne value_done
    inc rsi
    call peek
    cmp al, '+'
    je 4f
    cmp al, '-'
    jne 5f
4:
    inc rsi
    call peek
5:
    sub al, '0'
    cmp al, 9
    ja error
6:
    inc rsi
    call peek
    sub al, '0'
    cmp al, 9
    jbe 6b
    jmp value_done

literal:
    inc rsi
1:
    mov cl, byte ptr [rdx]
    test cl, cl
    jz value_done
    call peek
    cmp al, cl
    jne error
    inc rsi
    inc rdx
    jmp 1b

obj_start:
    inc rsi
    mov byte ptr [r13 + r12], 1
    inc r12
    call skip_ws
    cmp al, '}'
    je close
member:
    cmp al, '"'
    jne error
    call string
    call skip_ws
    cmp al, ':'
    jne error
    inc rsi
    jmp value

arr_start:
    inc rsi
    mov byte ptr [r13 + r12], 0
    inc r12
    call skip_ws
    cmp al, ']'
    je close
    jmp value

close:
    inc rsi
    dec r12
value_done:
    test r12, r12
    jz top_done
    call skip_ws
    cmp byte ptr [r13 + r12 - 1], 0
    jne in_obj
    cmp al, ']'
    je close
    cmp al, ','
    jne error
    inc rsi
    jmp value
in_obj:
    cmp al, '}'
    je close
    cmp al, ','
    jne error
    inc rsi
    call skip_ws
    jmp member

top_done:
    call skip_ws
    cmp rsi, rdi
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + s_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

error:
    mov rax, rsi
    sub rax, rbx
    lea r8, [rip + tmp + 39]
    mov byte ptr [r8], 10
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 1b
    sub r8, 9
    mov rax, qword ptr [rip + s_err]
    mov qword ptr [r8], rax
    mov al, byte ptr [rip + s_err + 8]
    mov byte ptr [r8 + 8], al
    lea rdx, [rip + tmp + 40]
    sub rdx, r8
    mov rsi, r8
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

# skip whitespace; al = current byte or 0 at EOF
skip_ws:
1:
    xor eax, eax
    cmp rsi, rdi
    jae 2f
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je 3f
    cmp al, 9
    je 3f
    cmp al, 10
    je 3f
    cmp al, 13
    jne 2f
3:
    inc rsi
    jmp 1b
2:
    ret

peek:
    xor eax, eax
    cmp rsi, rdi
    jae 1f
    mov al, byte ptr [rsi]
1:
    ret

# rsi at opening quote
string:
    inc rsi
str_loop:
    call peek
    cmp al, 0x20
    jb error
    inc rsi
    cmp al, '"'
    je str_end
    cmp al, 0x5C
    jne str_loop
    call peek
    cmp al, '"'
    je 1f
    cmp al, 0x5C
    je 1f
    cmp al, '/'
    je 1f
    cmp al, 'b'
    je 1f
    cmp al, 'f'
    je 1f
    cmp al, 'n'
    je 1f
    cmp al, 'r'
    je 1f
    cmp al, 't'
    je 1f
    cmp al, 'u'
    jne error
    inc rsi
    mov ecx, 4
2:
    call peek
    mov dl, al
    sub dl, '0'
    cmp dl, 9
    jbe 3f
    or al, 0x20
    sub al, 'a'
    cmp al, 5
    ja error
3:
    inc rsi
    dec ecx
    jnz 2b
    jmp str_loop
1:
    inc rsi
    jmp str_loop
str_end:
    ret

.section .rodata
s_true: .asciz "rue"
s_false: .asciz "alse"
s_null: .asciz "ull"
s_ok: .ascii "ok\n"
s_err: .ascii "error at "

.bss
.align 16
tmp: .zero 64
stk: .zero 4096
buf: .zero 196608
