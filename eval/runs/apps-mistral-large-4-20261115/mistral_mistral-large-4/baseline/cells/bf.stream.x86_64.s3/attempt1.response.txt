.text
.globl _start
_start:
    # read all stdin into input_buf
    xor r12, r12
    lea r13, [rip + input_buf]
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = total length
    # scan for first '!'
    xor rcx, rcx
    lea r14, [rip + input_buf]
scan_loop:
    cmp rcx, r12
    jae no_bang
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 0x21
    je found_bang
    inc rcx
    jmp scan_loop
no_bang:
    mov rbx, r12
    mov r14, r12
    mov r15, r12
    jmp after_bang
found_bang:
    mov rbx, rcx
    lea r14, [rcx + 1]
    mov r15, r12
after_bang:
    # check brackets and build match table
    lea r8, [rip + match]
    lea r9, [rip + bracket_stack]
    xor rcx, rcx
    xor rdx, rdx
    lea r10, [rip + input_buf]
bracket_loop:
    cmp rcx, rbx
    jae bracket_done
    movzx eax, byte ptr [r10 + rcx]
    cmp al, 0x5B
    je is_open
    cmp al, 0x5D
    je is_close
    jmp bracket_next
is_open:
    mov [r9 + rdx*4], ecx
    inc rdx
    jmp bracket_next
is_close:
    test rdx, rdx
    jz unbalanced
    dec rdx
    mov eax, dword ptr [r9 + rdx*4]
    mov [r8 + rcx*4], eax
    mov [r8 + rax*4], ecx
    jmp bracket_next
bracket_next:
    inc rcx
    jmp bracket_loop
bracket_done:
    test rdx, rdx
    jnz unbalanced
    # balanced, run program
    lea r10, [rip + input_buf]
    lea r8, [rip + tape]
    lea r9, [rip + out_buf]
    lea r11, [rip + match]
    xor r12, r12
    xor r13, r13
    mov rbp, r15
    xor r15, r15
run_loop:
    cmp r12, rbx
    jae run_done
    movzx eax, byte ptr [r10 + r12]
    inc r12
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_dot
    cmp al, 0x2C
    je cmd_comma
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    jmp run_loop
cmd_gt:
    inc r13
    jmp run_loop
cmd_lt:
    dec r13
    jmp run_loop
cmd_plus:
    mov al, byte ptr [r8 + r13]
    inc al
    mov byte ptr [r8 + r13], al
    jmp run_loop
cmd_minus:
    mov al, byte ptr [r8 + r13]
    dec al
    mov byte ptr [r8 + r13], al
    jmp run_loop
cmd_dot:
    mov al, byte ptr [r8 + r13]
    mov [r9 + r15], al
    inc r15
    jmp run_loop
cmd_comma:
    cmp r14, rbp
    jae comma_eof
    movzx eax, byte ptr [r10 + r14]
    inc r14
    mov byte ptr [r8 + r13], al
    jmp run_loop
comma_eof:
    mov byte ptr [r8 + r13], 0
    jmp run_loop
cmd_open:
    mov al, byte ptr [r8 + r13]
    test al, al
    jnz open_nonzero
    lea rax, [r12 - 1]
    mov eax, dword ptr [r11 + rax*4]
    inc rax
    mov r12, rax
    jmp run_loop
open_nonzero:
    jmp run_loop
cmd_close:
    mov al, byte ptr [r8 + r13]
    test al, al
    jz close_zero
    lea rax, [r12 - 1]
    mov eax, dword ptr [r11 + rax*4]
    inc rax
    mov r12, rax
    jmp run_loop
close_zero:
    jmp run_loop
run_done:
    test r15, r15
    jz exit_zero
    xor rbx, rbx
write_loop:
    cmp rbx, r15
    jae write_done
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    add rsi, rbx
    mov rdx, r15
    sub rdx, rbx
    syscall
    add rbx, rax
    jmp write_loop
write_done:
exit_zero:
    mov rax, 60
    xor rdi, rdi
    syscall
unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall

.section .bss
.balign 8
input_buf:
    .zero 65536
tape:
    .zero 30000
out_buf:
    .zero 65536
match:
    .zero 262144
bracket_stack:
    .zero 4000
