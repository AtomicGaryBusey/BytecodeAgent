.globl _start

.equ INPUT_SIZE, 65536
.equ VSTACK_BYTES, 262144

.text
_start:
    lea r12, [rip + input_buf]
    xor r14, r14
read_loop:
    cmp r14, INPUT_SIZE
    jae read_done
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    add rsi, r14
    mov rdx, INPUT_SIZE
    sub rdx, r14
    syscall
    cmp rax, 0
    je read_done
    add r14, rax
    jmp read_loop
read_done:
    lea r12, [rip + input_buf]
    lea r13, [rip + input_buf]
    add r13, r14
    lea rbp, [rip + vstack]
    xor r15, r15
    xor r14, r14
main_loop:
    cmp r12, r13
    jae eof
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 10
    je newline
    cmp al, 32
    je sep
    cmp al, 9
    je sep
    mov r14, 1
    mov rdi, r12
    dec rdi
    mov rsi, r12
scan_tok:
    cmp rsi, r13
    jae tok_eof
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je tok_delim
    cmp al, 32
    je tok_delim
    cmp al, 9
    je tok_delim
    inc rsi
    jmp scan_tok
tok_delim:
    mov r12, rsi
    call process_token
    jmp main_loop
tok_eof:
    mov rsi, r13
    mov r12, r13
    call process_token
    jmp main_loop
sep:
    mov r14, 1
    jmp main_loop
newline:
    call finalize_line
    lea rbp, [rip + vstack]
    xor r15, r15
    xor r14, r14
    jmp main_loop
eof:
    cmp r14, 0
    je exit_prog
    call finalize_line
exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall

process_token:
    cmp r15, 0
    jne pt_ret
    mov rax, rsi
    sub rax, rdi
    cmp rax, 1
    jne pt_literal
    movzx r8d, byte ptr [rdi]
    cmp r8b, 43
    je pt_op
    cmp r8b, 45
    je pt_op
    cmp r8b, 42
    je pt_op
    cmp r8b, 47
    je pt_op
    cmp r8b, 37
    je pt_op
    jmp pt_literal
pt_op:
    lea rax, [rip + vstack]
    mov rdx, rbp
    sub rdx, rax
    cmp rdx, 16
    jb pt_error
    sub rbp, 8
    mov rdx, qword ptr [rbp]
    sub rbp, 8
    mov rax, qword ptr [rbp]
    cmp r8b, 43
    je pt_add
    cmp r8b, 45
    je pt_sub
    cmp r8b, 42
    je pt_mul
    cmp rdx, 0
    je pt_error
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne pt_div_ok
    cmp rdx, -1
    je pt_error
pt_div_ok:
    mov rcx, rdx
    cqo
    idiv rcx
    cmp r8b, 47
    je pt_push_rax
    mov rax, rdx
    jmp pt_push_rax
pt_add:
    add rax, rdx
    jmp pt_push_rax
pt_sub:
    sub rax, rdx
    jmp pt_push_rax
pt_mul:
    imul rax, rdx
    jmp pt_push_rax
pt_push_rax:
    mov qword ptr [rbp], rax
    add rbp, 8
pt_ret:
    ret
pt_error:
    mov r15, 1
    ret
pt_literal:
    movzx ecx, byte ptr [rdi]
    xor r8, r8
    cmp cl, 45
    jne pt_lit_nosign
    mov r8, 1
    inc rdi
    dec rax
    test rax, rax
    jz pt_error
pt_lit_nosign:
    cmp rax, 18
    ja pt_error
    xor r9, r9
pt_lit_loop:
    cmp rdi, rsi
    jae pt_lit_done
    movzx ecx, byte ptr [rdi]
    sub ecx, 48
    cmp ecx, 9
    ja pt_error
    imul r9, r9, 10
    add r9, rcx
    inc rdi
    jmp pt_lit_loop
pt_lit_done:
    cmp r8, 0
    je pt_lit_push
    neg r9
pt_lit_push:
    mov qword ptr [rbp], r9
    add rbp, 8
    ret

finalize_line:
    cmp r15, 0
    jne fl_error
    lea rax, [rip + vstack]
    mov rdx, rbp
    sub rdx, rax
    cmp rdx, 8
    jne fl_error
    mov rax, qword ptr [rbp - 8]
    xor r9, r9
    test rax, rax
    jns fl_positive
    mov r9, 1
    neg rax
fl_positive:
    lea r8, [rip + num_buf]
    add r8, 31
    mov byte ptr [r8], 10
    mov rcx, 10
fl_digit:
    xor edx, edx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz fl_digit
    cmp r9, 0
    je fl_len
    dec r8
    mov byte ptr [r8], 45
fl_len:
    lea rdx, [rip + num_buf]
    add rdx, 32
    sub rdx, r8
    mov rsi, r8
    jmp fl_write
fl_error:
    lea rsi, [rip + err_msg]
    mov rdx, 6
fl_write:
    mov rax, 1
    mov rdi, 1
    syscall
fl_done:
    ret

.section .rodata
err_msg:
    .ascii "error\n"

.bss
.align 16
input_buf:
    .zero INPUT_SIZE
vstack:
    .zero VSTACK_BYTES
num_buf:
    .zero 32
