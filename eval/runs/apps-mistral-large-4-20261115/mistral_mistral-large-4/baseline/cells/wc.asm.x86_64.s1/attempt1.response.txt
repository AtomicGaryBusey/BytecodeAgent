.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_SIZE, 65536
.equ OUT_SIZE, 128

.text
.globl _start
_start:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor ebx, ebx
    lea r14, [rip + ws_table]
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jz read_done
    js read_done
    mov r12, rsi
    mov r13, rax
byte_loop:
    test r13, r13
    jz read_loop
    movzx eax, byte ptr [r12]
    inc r12
    dec r13
    inc r10
    cmp al, 10
    jne check_ws
    inc r8
check_ws:
    movzx edx, byte ptr [r14 + rax*1]
    test dl, dl
    jnz set_not_word
    test ebx, ebx
    jnz byte_loop
    inc r9
    mov ebx, 1
    jmp byte_loop
set_not_word:
    xor ebx, ebx
    jmp byte_loop
read_done:
    lea rsi, [rip + outbuf]
    add rsi, OUT_SIZE
    mov rdi, rsi
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r10
    call store_dec
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r9
    call store_dec
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r8
    call store_dec
    mov rdx, rsi
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, SYS_WRITE
    syscall
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

store_dec:
    mov rcx, 10
store_dec_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz store_dec_loop
    ret

.section .rodata
.balign 16
ws_table:
    .zero 9
    .byte 1
    .byte 1
    .byte 1
    .byte 1
    .byte 1
    .zero 18
    .byte 1
    .zero 223

.bss
.balign 16
buf:
    .zero BUF_SIZE
.balign 16
outbuf:
    .zero OUT_SIZE
