.equ BUFSZ, 65536
.equ TAPESZ, 30000
.equ OUTSZ, 65536
.equ MATCHSZ, 262144
.equ BSTACKSZ, 4096
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT_GROUP, 94
.equ CH_EXCL, 0x21
.equ CH_PLUS, 0x2b
.equ CH_COMMA, 0x2c
.equ CH_MINUS, 0x2d
.equ CH_DOT, 0x2e
.equ CH_LT, 0x3c
.equ CH_GT, 0x3e
.equ CH_LB, 0x5b
.equ CH_RB, 0x5d

.text
.globl _start
_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0
read_loop:
  cmp x20, #BUFSZ
  b.hs read_done
  mov x0, #0
  mov x1, x19
  add x1, x1, x20
  mov x2, #BUFSZ
  sub x2, x2, x20
  mov x8, #SYS_READ
  svc #0
  cmp x0, #0
  b.eq read_done
  b.mi read_fail
  add x20, x20, x0
  b read_loop
read_done:
  mov x9, #0
find_loop:
  cmp x9, x20
  b.hs find_none
  ldrb w6, [x19, x9]
  cmp w6, #CH_EXCL
  b.eq find_found
  add x9, x9, #1
  b find_loop
find_found:
  mov x21, x9
  add x22, x19, x9
  add x22, x22, #1
  add x24, x9, #1
  sub x23, x20, x24
  b validate
find_none:
  mov x21, x20
  add x22, x19, x20
  mov x23, #0
  b validate
read_fail:
  mov x0, #1
  mov x8, #SYS_EXIT_GROUP
  svc #0

validate:
  adrp x25, bstack
  add x25, x25, :lo12:bstack
  adrp x26, match
  add x26, x26, :lo12:match
  mov x9, #0
  mov x5, #0
val_loop:
  cmp x9, x21
  b.hs val_end
  ldrb w6, [x19, x9]
  cmp w6, #CH_LB
  b.eq val_lb
  cmp w6, #CH_RB
  b.eq val_rb
  b val_next
val_lb:
  str w9, [x25, x5, lsl #2]
  add x5, x5, #1
  b val_next
val_rb:
  cbz x5, val_fail
  sub x5, x5, #1
  ldr w7, [x25, x5, lsl #2]
  str w9, [x26, x7, lsl #2]
  str w7, [x26, x9, lsl #2]
  b val_next
val_next:
  add x9, x9, #1
  b val_loop
val_end:
  cbnz x5, val_fail
  b run
val_fail:
  mov x0, #1
  mov x8, #SYS_EXIT_GROUP
  svc #0

run:
  adrp x10, tape
  add x10, x10, :lo12:tape
  adrp x27, out_buf
  add x27, x27, :lo12:out_buf
  mov x9, #0
  mov x24, #0
  mov x28, #0
exec_loop:
  cmp x9, x21
  b.hs done
  ldrb w12, [x19, x9]
  cmp w12, #CH_GT
  b.eq cmd_gt
  cmp w12, #CH_LT
  b.eq cmd_lt
  cmp w12, #CH_PLUS
  b.eq cmd_plus
  cmp w12, #CH_MINUS
  b.eq cmd_minus
  cmp w12, #CH_DOT
  b.eq cmd_dot
  cmp w12, #CH_COMMA
  b.eq cmd_comma
  cmp w12, #CH_LB
  b.eq cmd_lb
  cmp w12, #CH_RB
  b.eq cmd_rb
  b cmd_next
cmd_gt:
  add x10, x10, #1
  b cmd_next
cmd_lt:
  sub x10, x10, #1
  b cmd_next
cmd_plus:
  ldrb w13, [x10]
  add w13, w13, #1
  strb w13, [x10]
  b cmd_next
cmd_minus:
  ldrb w13, [x10]
  sub w13, w13, #1
  strb w13, [x10]
  b cmd_next
cmd_dot:
  ldrb w13, [x10]
  strb w13, [x27, x28]
  add x28, x28, #1
  cmp x28, #OUTSZ
  b.lo cmd_next
  mov x0, #1
  mov x1, x27
  mov x2, x28
  mov x8, #SYS_WRITE
  svc #0
  mov x28, #0
  b cmd_next
cmd_comma:
  cmp x24, x23
  b.hs cmd_comma_zero
  ldrb w13, [x22, x24]
  add x24, x24, #1
  strb w13, [x10]
  b cmd_next
cmd_comma_zero:
  strb wzr, [x10]
  b cmd_next
cmd_lb:
  ldrb w13, [x10]
  cbnz w13, cmd_next
  ldr w14, [x26, x9, lsl #2]
  add x9, x14, #1
  b exec_loop
cmd_rb:
  ldrb w13, [x10]
  cbz w13, cmd_next
  ldr w14, [x26, x9, lsl #2]
  add x9, x14, #1
  b exec_loop
cmd_next:
  add x9, x9, #1
  b exec_loop
done:
  cbz x28, exit0
  mov x0, #1
  mov x1, x27
  mov x2, x28
  mov x8, #SYS_WRITE
  svc #0
exit0:
  mov x0, #0
  mov x8, #SYS_EXIT_GROUP
  svc #0

.bss
.p2align 3
input_buf:
  .zero BUFSZ
.p2align 3
tape:
  .zero TAPESZ
.p2align 3
out_buf:
  .zero OUTSZ
.p2align 2
match:
  .zero MATCHSZ
.p2align 2
bstack:
  .zero BSTACKSZ
