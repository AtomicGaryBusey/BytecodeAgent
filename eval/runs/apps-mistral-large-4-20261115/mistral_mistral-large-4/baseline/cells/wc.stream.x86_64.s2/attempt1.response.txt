.text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor r11, r11
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jz end_read
    mov rcx, rax
    lea rsi, [rip + buf]
process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r10
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    test r11, r11
    jnz skip_word
    inc r9
    mov r11, 1
skip_word:
    jmp next_byte
is_ws:
    mov r11, 0
    cmp al, 0x0A
    jne next_byte
    inc r8
next_byte:
    jmp process_loop
end_read:
    lea r12, [rip + out_buf]
    mov rax, r8
    call store_dec
    mov byte ptr [r12], 0x20
    inc r12
    mov rax, r9
    call store_dec
    mov byte ptr [r12], 0x20
    inc r12
    mov rax, r10
    call store_dec
    mov byte ptr [r12], 0x0A
    inc r12
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, r12
    sub rdx, rsi
    syscall
    mov rax, 60
    xor edi, edi
    syscall

store_dec:
    push rbp
    mov rbp, rsp
    xor ecx, ecx
    mov ebx, 10
div_loop:
    xor edx, edx
    div rbx
    add dl, '0'
    push rdx
    inc rcx
    test rax, rax
    jnz div_loop
    test rcx, rcx
    jnz store_loop
    mov dl, '0'
    push rdx
    mov ecx, 1
store_loop:
    pop rax
    mov [r12], al
    inc r12
    dec rcx
    jnz store_loop
    mov rsp, rbp
    pop rbp
    ret

.bss
buf: .zero 65536
out_buf: .zero 64
