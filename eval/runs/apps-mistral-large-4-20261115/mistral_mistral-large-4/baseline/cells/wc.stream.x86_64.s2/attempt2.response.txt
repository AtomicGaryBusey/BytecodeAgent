.text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor r13, r13
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jz end_read
    mov rcx, rax
    lea rsi, [rip + buf]
process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r10
    cmp al, 0x0A
    je newline_byte
    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    test r13, r13
    jnz next_byte
    inc r9
    mov r13, 1
    jmp next_byte
newline_byte:
    inc r8
whitespace:
    xor r13, r13
next_byte:
    jmp process_loop
end_read:
    lea r12, [rip + out_buf]
    mov rax, r8
    call store_dec
    mov byte ptr [r12], 0x20
    inc r12
    mov rax, r9
    call store_dec
    mov byte ptr [r12], 0x20
    inc r12
    mov rax, r10
    call store_dec
    mov byte ptr [r12], 0x0A
    inc r12
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, r12
    sub rdx, rsi
    syscall
    mov rax, 60
    xor edi, edi
    syscall

store_dec:
    push rbp
    mov rbp, rsp
    xor ecx, ecx
    mov ebx, 10
sd_div:
    xor edx, edx
    div rbx
    add dl, '0'
    push rdx
    inc rcx
    test rax, rax
    jnz sd_div
sd_store:
    pop rax
    mov byte ptr [r12], al
    inc r12
    dec rcx
    jnz sd_store
    mov rsp, rbp
    pop rbp
    ret

.bss
buf: .zero 65536
out_buf: .zero 64
