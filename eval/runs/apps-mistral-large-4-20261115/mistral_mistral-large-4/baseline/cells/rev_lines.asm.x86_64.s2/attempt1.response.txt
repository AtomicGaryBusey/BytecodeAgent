.globl _start
.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ IN_CAP, 65537
.equ OUT_CAP, 65538

.text
_start:
    xor r12, r12
read_loop:
    mov rax, SYS_read
    xor rdi, rdi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov rdx, IN_CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, IN_CAP
    jae read_done
    jmp read_loop
read_done:
    lea r13, [rip + in_buf]
    mov r14, r12
    lea r15, [rip + out_buf]
line_loop:
    test r14, r14
    jz write_out
    mov rsi, r13
    mov rcx, r14
    xor rdx, rdx
scan_loop:
    test rcx, rcx
    jz no_newline
    mov al, byte ptr [rsi + rdx]
    cmp al, 10
    je found_newline
    inc rdx
    dec rcx
    jmp scan_loop
found_newline:
    mov r8, rdx
    lea r9, [r13 + r8]
    dec r9
    mov r10, r15
    test r8, r8
    jz after_rev
rev_loop:
    mov al, byte ptr [r9]
    mov byte ptr [r10], al
    inc r10
    dec r9
    dec r8
    jnz rev_loop
after_rev:
    mov r15, r10
    mov byte ptr [r15], 10
    inc r15
    lea rax, [rdx + 1]
    add r13, rax
    sub r14, rax
    jmp line_loop
no_newline:
    mov r8, r14
    lea r9, [r13 + r8]
    dec r9
    mov r10, r15
    test r8, r8
    jz after_rev_final
rev_loop_final:
    mov al, byte ptr [r9]
    mov byte ptr [r10], al
    inc r10
    dec r9
    dec r8
    jnz rev_loop_final
after_rev_final:
    mov r15, r10
    mov byte ptr [r15], 10
    inc r15
    xor r14, r14
    jmp line_loop
write_out:
    lea rdx, [rip + out_buf]
    mov rax, r15
    sub rax, rdx
    test rax, rax
    jz exit_prog
    mov r12, rax
    mov r13, rdx
write_loop:
    test r12, r12
    jz exit_prog
    mov rax, SYS_write
    mov rdi, 1
    mov rsi, r13
    mov rdx, r12
    syscall
    test rax, rax
    jle exit_prog
    add r13, rax
    sub r12, rax
    jmp write_loop
exit_prog:
    mov rax, SYS_exit
    xor rdi, rdi
    syscall

.bss
.balign 16
in_buf:
    .zero IN_CAP
.balign 16
out_buf:
    .zero OUT_CAP
