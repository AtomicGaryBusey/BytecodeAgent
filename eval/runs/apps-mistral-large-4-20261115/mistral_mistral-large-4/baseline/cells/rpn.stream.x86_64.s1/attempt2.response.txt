.equ INPUT_MAX, 65536

.text
.globl _start
_start:
    xor r12d, r12d
.Lread_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    syscall
    test rax, rax
    jz .Lread_done
    add r12, rax
    cmp r12, INPUT_MAX
    jb .Lread_loop
.Lread_done:
    lea r13, [rip + input_buf]
    lea r14, [rip + input_buf]
    add r14, r12
.Lline_loop:
    cmp r13, r14
    jae .Lall_done
    mov r15, r13
.Lfind_nl:
    cmp r15, r14
    jae .Lline_found
    movzx eax, byte ptr [r15]
    cmp al, 0x0A
    je .Lline_found
    inc r15
    jmp .Lfind_nl
.Lline_found:
    mov rsi, r13
    mov rdx, r15
    call process_line
    cmp r15, r14
    jae .Lset_p_end
    lea r13, [r15 + 1]
    jmp .Lline_loop
.Lset_p_end:
    mov r13, r14
    jmp .Lline_loop
.Lall_done:
    mov rax, 60
    xor rdi, rdi
    syscall

process_line:
    push rbx
    mov r8, rsi
    mov r9, rdx
    xor r10d, r10d
    xor r11d, r11d
.Ltoken_loop:
    cmp r8, r9
    jae .Lline_done
    movzx eax, byte ptr [r8]
    cmp al, 0x20
    je .Lskip_sep
    cmp al, 0x09
    je .Lskip_sep
    jmp .Lhave_token
.Lskip_sep:
    inc r8
    jmp .Ltoken_loop
.Lhave_token:
    mov rsi, r8
.Lfind_tok_end:
    cmp r8, r9
    jae .Ltok_end
    movzx eax, byte ptr [r8]
    cmp al, 0x20
    je .Ltok_end
    cmp al, 0x09
    je .Ltok_end
    inc r8
    jmp .Lfind_tok_end
.Ltok_end:
    test r11d, r11d
    jnz .Lskip_rest
    mov rcx, r8
    sub rcx, rsi
    cmp rcx, 1
    jne .Ltry_int
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod
    jmp .Linvalid
.Ltry_int:
    xor ebx, ebx
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne .Ldigits_start
    mov ebx, 1
    inc rsi
    dec rcx
    jz .Linvalid
.Ldigits_start:
    test rcx, rcx
    jz .Linvalid
    cmp rcx, 18
    ja .Linvalid
    xor edx, edx
.Ldigit_loop:
    test rcx, rcx
    jz .Ldigits_done
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 9
    ja .Linvalid
    imul rdx, rdx, 10
    add rdx, rax
    inc rsi
    dec rcx
    jmp .Ldigit_loop
.Ldigits_done:
    test ebx, ebx
    jz .Lpush_val
    neg rdx
.Lpush_val:
    push rdx
    inc r10
    jmp .Ltoken_done
.Lop_add:
    cmp r10, 2
    jb .Lunderflow
    pop rdx
    dec r10
    pop rax
    dec r10
    add rax, rdx
    push rax
    inc r10
    jmp .Ltoken_done
.Lop_sub:
    cmp r10, 2
    jb .Lunderflow
    pop rdx
    dec r10
    pop rax
    dec r10
    sub rax, rdx
    push rax
    inc r10
    jmp .Ltoken_done
.Lop_mul:
    cmp r10, 2
    jb .Lunderflow
    pop rdx
    dec r10
    pop rax
    dec r10
    imul rax, rdx
    push rax
    inc r10
    jmp .Ltoken_done
.Lop_div:
    cmp r10, 2
    jb .Lunderflow
    pop rcx
    dec r10
    test rcx, rcx
    jz .Ldivzero
    pop rax
    dec r10
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    jne .Ldo_div
    cmp rcx, -1
    je .Loverflow
.Ldo_div:
    cqo
    idiv rcx
    push rax
    inc r10
    jmp .Ltoken_done
.Lop_mod:
    cmp r10, 2
    jb .Lunderflow
    pop rcx
    dec r10
    test rcx, rcx
    jz .Ldivzero
    pop rax
    dec r10
    mov rsi, 0x8000000000000000
    cmp rax, rsi
    jne .Ldo_mod
    cmp rcx, -1
    je .Loverflow
.Ldo_mod:
    cqo
    idiv rcx
    push rdx
    inc r10
    jmp .Ltoken_done
.Lunderflow:
.Ldivzero:
.Loverflow:
.Linvalid:
    mov r11d, 1
    jmp .Ltoken_done
.Lskip_rest:
    mov r8, r9
    jmp .Ltoken_loop
.Ltoken_done:
    jmp .Ltoken_loop
.Lline_done:
    test r11d, r11d
    jnz .Lerror
    cmp r10, 1
    jne .Lerror
    pop rax
    call print_number
    jmp .Lret
.Lerror:
    lea rsp, [rsp + r10*8]
    lea rsi, [rip + error_msg]
    mov rdx, 6
    mov rax, 1
    mov rdi, 1
    syscall
.Lret:
    pop rbx
    ret

print_number:
    sub rsp, 32
    mov rdi, rsp
    add rdi, 32
    xor r11d, r11d
    test rax, rax
    jns .Lpos
    mov r11d, 1
    neg rax
.Lpos:
    mov rcx, 10
.Lconv_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz .Lconv_loop
    test r11d, r11d
    jz .Lno_sign
    dec rdi
    mov byte ptr [rdi], '-'
.Lno_sign:
    mov rdx, rsp
    add rdx, 32
    sub rdx, rdi
    mov byte ptr [rdi + rdx], 0x0A
    inc rdx
    mov rsi, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    add rsp, 32
    ret

.section .rodata
error_msg: .ascii "error\n"

.bss
.align 8
input_buf: .zero INPUT_MAX
