.equ INPUT_MAX, 65536
.equ VSTACK_SIZE, 262144

.text
.globl _start
_start:
    xor r12d, r12d
.Lread_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    syscall
    test rax, rax
    jz .Lread_done
    add r12, rax
    cmp r12, INPUT_MAX
    jb .Lread_loop
.Lread_done:
    lea r13, [rip + input_buf]
    lea r14, [rip + input_buf]
    add r14, r12
.Lline_loop:
    cmp r13, r14
    jae .Lall_done
    mov r15, r13
.Lfind_nl:
    cmp r15, r14
    jae .Lline_found
    movzx eax, byte ptr [r15]
    cmp al, 0x0A
    je .Lline_found
    inc r15
    jmp .Lfind_nl
.Lline_found:
    mov rsi, r13
    mov rdx, r15
    call process_line
    cmp r15, r14
    jae .Lline_end
    lea r13, [r15 + 1]
    jmp .Lline_loop
.Lline_end:
    mov r13, r14
    jmp .Lline_loop
.Lall_done:
    mov eax, 60
    xor edi, edi
    syscall

process_line:
    push rbx
    lea rbp, [rip + vstack]
    mov r8, rsi
    mov r9, rdx
    xor r10d, r10d
    xor r11d, r11d
.Ltoken_loop:
    cmp r8, r9
    jae .Lline_done
    movzx eax, byte ptr [r8]
    cmp al, 0x20
    je .Lskip_sep
    cmp al, 0x09
    je .Lskip_sep
    jmp .Lhave_token
.Lskip_sep:
    inc r8
    jmp .Ltoken_loop
.Lhave_token:
    mov rdi, r8
.Lfind_tok_end:
    cmp r8, r9
    jae .Ltok_end
    movzx eax, byte ptr [r8]
    cmp al, 0x20
    je .Ltok_end
    cmp al, 0x09
    je .Ltok_end
    inc r8
    jmp .Lfind_tok_end
.Ltok_end:
    mov rcx, r8
    sub rcx, rdi
    cmp rcx, 1
    jne .Ltry_int
    movzx eax, byte ptr [rdi]
    cmp al, '+'
    je .Lop_add
    cmp al, '-'
    je .Lop_sub
    cmp al, '*'
    je .Lop_mul
    cmp al, '/'
    je .Lop_div
    cmp al, '%'
    je .Lop_mod
.Ltry_int:
    xor ebx, ebx
    movzx eax, byte ptr [rdi]
    cmp al, '-'
    jne .Ldigits_start
    mov ebx, 1
    inc rdi
    dec rcx
    jz .Linvalid
.Ldigits_start:
    test rcx, rcx
    jz .Linvalid
    cmp rcx, 18
    ja .Linvalid
    xor edx, edx
.Ldigit_loop:
    test rcx, rcx
    jz .Ldigits_done
    movzx eax, byte ptr [rdi]
    sub eax, '0'
    cmp eax, 9
    ja .Linvalid
    imul rdx, rdx, 10
    add rdx, rax
    inc rdi
    dec rcx
    jmp .Ldigit_loop
.Ldigits_done:
    test ebx, ebx
    jz .Lpush_val
    neg rdx
.Lpush_val:
    mov [rbp + r10*8], rdx
    inc r10
    jmp .Ltoken_loop
.Lop_add:
    cmp r10, 2
    jb .Lerror_set
    dec r10
    mov rdx, [rbp + r10*8]
    dec r10
    mov rax, [rbp + r10*8]
    add rax, rdx
    mov [rbp + r10*8], rax
    inc r10
    jmp .Ltoken_loop
.Lop_sub:
    cmp r10, 2
    jb .Lerror_set
    dec r10
    mov rdx, [rbp + r10*8]
    dec r10
    mov rax, [rbp + r10*8]
    sub rax, rdx
    mov [rbp + r10*8], rax
    inc r10
    jmp .Ltoken_loop
.Lop_mul:
    cmp r10, 2
    jb .Lerror_set
    dec r10
    mov rdx, [rbp + r10*8]
    dec r10
    mov rax, [rbp + r10*8]
    imul rax, rdx
    mov [rbp + r10*8], rax
    inc r10
    jmp .Ltoken_loop
.Lop_div:
    cmp r10, 2
    jb .Lerror_set
    dec r10
    mov rcx, [rbp + r10*8]
    dec r10
    mov rax, [rbp + r10*8]
    test rcx, rcx
    jz .Lerror_set
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne .Ldo_div
    cmp rcx, -1
    je .Lerror_set
.Ldo_div:
    mov rdx, rax
    sar rdx, 63
    idiv rcx
    mov [rbp + r10*8], rax
    inc r10
    jmp .Ltoken_loop
.Lop_mod:
    cmp r10, 2
    jb .Lerror_set
    dec r10
    mov rcx, [rbp + r10*8]
    dec r10
    mov rax, [rbp + r10*8]
    test rcx, rcx
    jz .Lerror_set
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne .Ldo_mod
    cmp rcx, -1
    je .Lerror_set
.Ldo_mod:
    mov rdx, rax
    sar rdx, 63
    idiv rcx
    mov [rbp + r10*8], rdx
    inc r10
    jmp .Ltoken_loop
.Linvalid:
.Lerror_set:
    mov r11d, 1
    jmp .Lline_done
.Lline_done:
    test r11d, r11d
    jnz .Lerror_out
    cmp r10, 1
    jne .Lerror_out
    mov rax, [rbp]
    call print_number
    jmp .Lret
.Lerror_out:
    lea rsi, [rip + error_msg]
    mov edx, 6
    mov eax, 1
    mov edi, 1
    syscall
.Lret:
    pop rbx
    ret

print_number:
    sub rsp, 32
    mov byte ptr [rsp + 31], 0x0A
    lea r9, [rsp + 31]
    xor r10d, r10d
    test rax, rax
    jns .Lpos
    mov r10d, 1
    neg rax
.Lpos:
    mov ecx, 10
.Lconv_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec r9
    mov [r9], dl
    test rax, rax
    jnz .Lconv_loop
    test r10d, r10d
    jz .Lno_sign
    dec r9
    mov byte ptr [r9], '-'
.Lno_sign:
    mov rsi, r9
    lea rdx, [rsp + 32]
    sub rdx, r9
    mov eax, 1
    mov edi, 1
    syscall
    add rsp, 32
    ret

.section .rodata
error_msg: .ascii "error\n"

.bss
.align 8
input_buf: .zero INPUT_MAX
vstack: .zero VSTACK_SIZE
