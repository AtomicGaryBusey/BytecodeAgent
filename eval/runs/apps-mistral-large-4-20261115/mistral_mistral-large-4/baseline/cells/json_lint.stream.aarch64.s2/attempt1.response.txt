.equ BUFSIZE, 65536
.equ OK_LEN, 3
.equ ERR_PREFIX_LEN, 9
.equ NL_LEN, 1
.equ TRUE_LEN, 4
.equ FALSE_LEN, 5
.equ NULL_LEN, 4
.equ NUMBUF_LEN, 20

.text
.globl _start
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x22, #BUFSIZE
    mov x21, #0
read_loop:
    cmp x21, x22
    b.hs read_done
    mov x8, #63
    mov x0, #0
    add x1, x19, x21
    sub x2, x22, x21
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    b read_loop
read_done:
    mov x20, x21
    mov x0, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x0, x20
    b.lo report_error
    mov x8, #64
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #OK_LEN
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0

skip_ws:
    cmp x0, x20
    b.hs skip_ws_ret
    ldrb w1, [x19, x0]
    cmp w1, #0x20
    b.eq skip_ws_next
    cmp w1, #0x09
    b.eq skip_ws_next
    cmp w1, #0x0A
    b.eq skip_ws_next
    cmp w1, #0x0D
    b.eq skip_ws_next
    b skip_ws_ret
skip_ws_next:
    add x0, x0, #1
    b skip_ws
skip_ws_ret:
    ret

parse_value:
    cmp x0, x20
    b.lo parse_value_load
    mov x0, x20
    b report_error
parse_value_load:
    ldrb w1, [x19, x0]
    cmp w1, #0x7B
    b.eq parse_value_object
    cmp w1, #0x5B
    b.eq parse_value_array
    cmp w1, #0x22
    b.eq parse_string
    cmp w1, #0x74
    b.eq parse_value_true
    cmp w1, #0x66
    b.eq parse_value_false
    cmp w1, #0x6E
    b.eq parse_value_null
    cmp w1, #0x2D
    b.eq parse_number
    sub w2, w1, #0x30
    cmp w2, #9
    b.ls parse_number
    b report_error
parse_value_object:
    add x0, x0, #1
    b parse_object
parse_value_array:
    add x0, x0, #1
    b parse_array
parse_value_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    mov x2, #TRUE_LEN
    b parse_lit
parse_value_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    mov x2, #FALSE_LEN
    b parse_lit
parse_value_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    mov x2, #NULL_LEN
    b parse_lit

parse_lit:
    mov x3, #0
parse_lit_loop:
    cmp x3, x2
    b.hs parse_lit_done
    add x4, x0, x3
    cmp x4, x20
    b.hs parse_lit_eof
    ldrb w5, [x19, x4]
    ldrb w6, [x1, x3]
    cmp w5, w6
    b.ne parse_lit_bad
    add x3, x3, #1
    b parse_lit_loop
parse_lit_done:
    add x0, x0, x2
    ret
parse_lit_eof:
    mov x0, x20
    b report_error
parse_lit_bad:
    mov x0, x4
    b report_error

parse_number:
    ldrb w1, [x19, x0]
    cmp w1, #0x2D
    b.ne num_int
    add x0, x0, #1
    cmp x0, x20
    b.hs num_err_eof
num_int:
    ldrb w1, [x19, x0]
    cmp w1, #0x30
    b.eq num_zero
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi num_err_x0
    add x0, x0, #1
num_int_digits:
    cmp x0, x20
    b.hs num_frac
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi num_frac
    add x0, x0, #1
    b num_int_digits
num_zero:
    add x0, x0, #1
    b num_frac
num_frac:
    cmp x0, x20
    b.hs num_ret
    ldrb w1, [x19, x0]
    cmp w1, #0x2E
    b.ne num_exp
    add x0, x0, #1
    cmp x0, x20
    b.hs num_err_eof
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi num_err_x0
    add x0, x0, #1
num_frac_digits:
    cmp x0, x20
    b.hs num_ret
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi num_exp
    add x0, x0, #1
    b num_frac_digits
num_exp:
    cmp x0, x20
    b.hs num_ret
    ldrb w1, [x19, x0]
    cmp w1, #0x65
    b.eq num_exp_go
    cmp w1, #0x45
    b.ne num_ret
num_exp_go:
    add x0, x0, #1
    cmp x0, x20
    b.hs num_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x2B
    b.eq num_exp_sign
    cmp w1, #0x2D
    b.eq num_exp_sign
    b num_exp_digit
num_exp_sign:
    add x0, x0, #1
    cmp x0, x20
    b.hs num_err_eof
num_exp_digit:
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi num_err_x0
    add x0, x0, #1
num_exp_digits:
    cmp x0, x20
    b.hs num_ret
    ldrb w1, [x19, x0]
    sub w2, w1, #0x30
    cmp w2, #9
    b.hi num_ret
    add x0, x0, #1
    b num_exp_digits
num_ret:
    ret
num_err_eof:
    mov x0, x20
    b report_error
num_err_x0:
    b report_error

parse_string:
    add x0, x0, #1
str_loop:
    cmp x0, x20
    b.hs str_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x22
    b.eq str_end
    cmp w1, #0x5C
    b.eq str_escape
    cmp w1, #0x20
    b.lo str_err_x0
    add x0, x0, #1
    b str_loop
str_end:
    add x0, x0, #1
    ret
str_escape:
    add x1, x0, #1
    cmp x1, x20
    b.hs str_err_eof
    ldrb w2, [x19, x1]
    cmp w2, #0x22
    b.eq str_escape_simple
    cmp w2, #0x5C
    b.eq str_escape_simple
    cmp w2, #0x2F
    b.eq str_escape_simple
    cmp w2, #0x62
    b.eq str_escape_simple
    cmp w2, #0x66
    b.eq str_escape_simple
    cmp w2, #0x6E
    b.eq str_escape_simple
    cmp w2, #0x72
    b.eq str_escape_simple
    cmp w2, #0x74
    b.eq str_escape_simple
    cmp w2, #0x75
    b.eq str_u
    mov x0, x1
    b report_error
str_escape_simple:
    add x0, x0, #2
    b str_loop
str_u:
    add x3, x0, #2
    mov x4, #0
str_u_loop:
    cmp x3, x20
    b.hs str_err_eof
    ldrb w5, [x19, x3]
    sub w6, w5, #0x30
    cmp w6, #9
    b.ls str_u_next
    sub w6, w5, #0x61
    cmp w6, #5
    b.ls str_u_next
    sub w6, w5, #0x41
    cmp w6, #5
    b.ls str_u_next
    mov x0, x3
    b report_error
str_u_next:
    add x3, x3, #1
    add x4, x4, #1
    cmp x4, #4
    b.lo str_u_loop
    mov x0, x3
    b str_loop
str_err_eof:
    mov x0, x20
    b report_error
str_err_x0:
    b report_error

parse_object:
    str x30, [sp, #-16]!
    bl skip_ws
    cmp x0, x20
    b.hs obj_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x7D
    b.eq obj_end
obj_member:
    ldrb w1, [x19, x0]
    cmp w1, #0x22
    b.ne obj_err_x0
    bl parse_string
    bl skip_ws
    cmp x0, x20
    b.hs obj_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x3A
    b.ne obj_err_x0
    add x0, x0, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x0, x20
    b.hs obj_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x2C
    b.eq obj_comma
    cmp w1, #0x7D
    b.eq obj_end
    b obj_err_x0
obj_comma:
    add x0, x0, #1
    bl skip_ws
    cmp x0, x20
    b.hs obj_err_eof
    b obj_member
obj_end:
    add x0, x0, #1
    ldr x30, [sp], #16
    ret
obj_err_eof:
    mov x0, x20
    b report_error
obj_err_x0:
    b report_error

parse_array:
    str x30, [sp, #-16]!
    bl skip_ws
    cmp x0, x20
    b.hs arr_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x5D
    b.eq arr_end
arr_loop:
    bl parse_value
    bl skip_ws
    cmp x0, x20
    b.hs arr_err_eof
    ldrb w1, [x19, x0]
    cmp w1, #0x2C
    b.eq arr_comma
    cmp w1, #0x5D
    b.eq arr_end
    b arr_err_x0
arr_comma:
    add x0, x0, #1
    bl skip_ws
    b arr_loop
arr_end:
    add x0, x0, #1
    ldr x30, [sp], #16
    ret
arr_err_eof:
    mov x0, x20
    b report_error
arr_err_x0:
    b report_error

report_error:
    adrp x5, numbuf
    add x5, x5, :lo12:numbuf
    add x5, x5, #NUMBUF_LEN
    cbz x0, rep_zero
rep_div:
    mov x7, #10
    udiv x1, x0, x7
    msub x2, x1, x7, x0
    add w2, w2, #0x30
    sub x5, x5, #1
    strb w2, [x5]
    mov x0, x1
    cbnz x0, rep_div
    b rep_write
rep_zero:
    sub x5, x5, #1
    mov w2, #0x30
    strb w2, [x5]
rep_write:
    mov x8, #64
    mov x0, #1
    adrp x1, err_prefix
    add x1, x1, :lo12:err_prefix
    mov x2, #ERR_PREFIX_LEN
    svc #0
    mov x8, #64
    mov x0, #1
    mov x1, x5
    adrp x6, numbuf
    add x6, x6, :lo12:numbuf
    add x6, x6, #NUMBUF_LEN
    sub x2, x6, x5
    svc #0
    mov x8, #64
    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #NL_LEN
    svc #0
    mov x8, #93
    mov x0, #1
    svc #0

.section .rodata
ok_msg: .ascii "ok\n"
err_prefix: .ascii "error at "
nl_msg: .ascii "\n"
lit_true: .ascii "true"
lit_false: .ascii "false"
lit_null: .ascii "null"

.bss
.balign 8
buf: .zero BUFSIZE
numbuf: .zero NUMBUF_LEN
