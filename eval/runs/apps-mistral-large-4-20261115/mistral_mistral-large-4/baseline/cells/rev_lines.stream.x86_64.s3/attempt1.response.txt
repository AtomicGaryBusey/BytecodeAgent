.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    lea r12, [rip + input_buf]
    add r12, 65536
read_loop:
    mov eax, 0
    xor edi, edi
    mov rsi, rbx
    mov rdx, r12
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, r12
    jae read_done
    jmp read_loop
read_done:
    mov r13, rbx
    lea r14, [rip + input_buf]
line_loop:
    cmp r14, r13
    jae exit_prog
    mov r15, r14
scan_loop:
    cmp r15, r13
    jae final_line
    movzx eax, byte ptr [r15]
    cmp al, 0x0A
    je have_newline
    inc r15
    jmp scan_loop
have_newline:
    cmp r14, r15
    jae rev_nl_done
    mov rsi, r14
    lea rdi, [r15 - 1]
rev_nl:
    cmp rsi, rdi
    jae rev_nl_done
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp rev_nl
rev_nl_done:
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    sub rdx, r14
    inc rdx
    call write_all
    lea r14, [r15 + 1]
    jmp line_loop
final_line:
    cmp r14, r13
    jae exit_prog
    mov rsi, r14
    lea rdi, [r13 - 1]
rev_final:
    cmp rsi, rdi
    jae rev_final_done
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp rev_final
rev_final_done:
    mov edi, 1
    mov rsi, r14
    mov rdx, r13
    sub rdx, r14
    call write_all
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    jmp exit_prog
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall
write_all:
wa_loop:
    test rdx, rdx
    jz wa_done
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp wa_loop
wa_done:
    ret

.section .rodata
newline:
    .byte 0x0A

.section .bss
.balign 16
input_buf:
    .zero 65536
