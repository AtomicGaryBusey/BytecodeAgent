.equ INPUT_MAX, 49152
.equ OUT_MAX, 66560

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12, INPUT_MAX
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, INPUT_MAX
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + b64table]
    mov r13, r12
    xor r15d, r15d
enc_loop:
    cmp r13, 3
    jb enc_tail
    movzx eax, byte ptr [rsi]
    shl eax, 8
    movzx ebx, byte ptr [rsi+1]
    or eax, ebx
    shl eax, 8
    movzx ebx, byte ptr [rsi+2]
    or eax, ebx
    mov ebx, eax
    mov ecx, ebx
    shr ecx, 18
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov ecx, ebx
    shr ecx, 12
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov ecx, ebx
    shr ecx, 6
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov ecx, ebx
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    add rsi, 3
    sub r13, 3
    jmp enc_loop
enc_tail:
    test r13, r13
    jz finalize
    cmp r13, 1
    jne enc_two
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov ebx, eax
    mov ecx, ebx
    shr ecx, 18
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov ecx, ebx
    shr ecx, 12
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov al, 61
    call put
    mov al, 61
    call put
    jmp finalize
enc_two:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    movzx ebx, byte ptr [rsi+1]
    or eax, ebx
    shl eax, 8
    mov ebx, eax
    mov ecx, ebx
    shr ecx, 18
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov ecx, ebx
    shr ecx, 12
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov ecx, ebx
    shr ecx, 6
    and ecx, 63
    movzx eax, byte ptr [r8 + rcx]
    call put
    mov al, 61
    call put
finalize:
    test r15, r15
    jz write_setup
    mov byte ptr [rdi], 10
    inc rdi
write_setup:
    mov rdx, rdi
    lea rax, [rip + outbuf]
    sub rdx, rax
    lea rsi, [rip + outbuf]
write_loop:
    test rdx, rdx
    jz exit_prog
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

put:
    cmp r15, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r15d, r15d
1:
    mov byte ptr [rdi], al
    inc rdi
    inc r15
    ret

.section .rodata
b64table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero INPUT_MAX
outbuf:
    .zero OUT_MAX
