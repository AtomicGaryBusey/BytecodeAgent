.text
.globl _start

_start:
    adrp x21, buf
    add x21, x21, :lo12:buf
    mov x19, #0
    mov x20, #0
read_loop:
    mov x0, #0
    mov x1, x21
    add x1, x1, x20
    mov x2, #65536
    sub x2, x2, x20
    cmp x2, #0
    b.eq read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop
read_done:
    bl parse_ws
    bl parse_value
    bl parse_ws
    cmp x19, x20
    b.ne error_at_pos
    adrp x0, ok_msg
    add x0, x0, :lo12:ok_msg
    mov x1, x0
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

parse_ws:
parse_ws_loop:
    cmp x19, x20
    b.ge parse_ws_done
    ldrb w0, [x21, x19]
    cmp w0, #0x20
    b.eq parse_ws_skip
    cmp w0, #0x09
    b.eq parse_ws_skip
    cmp w0, #0x0A
    b.eq parse_ws_skip
    cmp w0, #0x0D
    b.eq parse_ws_skip
    b parse_ws_done
parse_ws_skip:
    add x19, x19, #1
    b parse_ws_loop
parse_ws_done:
    ret

parse_string:
    add x19, x19, #1
parse_string_loop:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'"'
    b.eq parse_string_close
    cmp w0, #'\\'
    b.eq parse_string_escape
    cmp w0, #0x20
    b.lo error_at_pos
    add x19, x19, #1
    b parse_string_loop
parse_string_close:
    add x19, x19, #1
    ret
parse_string_escape:
    add x19, x19, #1
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'"'
    b.eq parse_string_escape_simple
    cmp w0, #'\\'
    b.eq parse_string_escape_simple
    cmp w0, #'/'
    b.eq parse_string_escape_simple
    cmp w0, #'b'
    b.eq parse_string_escape_simple
    cmp w0, #'f'
    b.eq parse_string_escape_simple
    cmp w0, #'n'
    b.eq parse_string_escape_simple
    cmp w0, #'r'
    b.eq parse_string_escape_simple
    cmp w0, #'t'
    b.eq parse_string_escape_simple
    cmp w0, #'u'
    b.eq parse_string_escape_u
    b error_at_pos
parse_string_escape_simple:
    add x19, x19, #1
    b parse_string_loop
parse_string_escape_u:
    add x19, x19, #1
    mov x1, #4
parse_string_hex_loop:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'0'
    b.lo parse_string_hex_check_af
    cmp w0, #'9'
    b.le parse_string_hex_ok
parse_string_hex_check_af:
    cmp w0, #'a'
    b.lo parse_string_hex_check_AF
    cmp w0, #'f'
    b.le parse_string_hex_ok
parse_string_hex_check_AF:
    cmp w0, #'A'
    b.lo error_at_pos
    cmp w0, #'F'
    b.le parse_string_hex_ok
    b error_at_pos
parse_string_hex_ok:
    add x19, x19, #1
    subs x1, x1, #1
    b.ne parse_string_hex_loop
    b parse_string_loop

parse_number:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'-'
    b.ne parse_number_int
    add x19, x19, #1
parse_number_int:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'0'
    b.eq parse_number_zero
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error_at_pos
    add x19, x19, #1
parse_number_digits:
    cmp x19, x20
    b.ge parse_number_frac
    ldrb w0, [x21, x19]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi parse_number_frac
    add x19, x19, #1
    b parse_number_digits
parse_number_zero:
    add x19, x19, #1
    b parse_number_frac
parse_number_frac:
    cmp x19, x20
    b.ge parse_number_exp
    ldrb w0, [x21, x19]
    cmp w0, #'.'
    b.ne parse_number_exp
    add x19, x19, #1
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error_at_pos
    add x19, x19, #1
parse_number_frac_digits:
    cmp x19, x20
    b.ge parse_number_exp
    ldrb w0, [x21, x19]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi parse_number_exp
    add x19, x19, #1
    b parse_number_frac_digits
parse_number_exp:
    cmp x19, x20
    b.ge parse_number_done
    ldrb w0, [x21, x19]
    cmp w0, #'e'
    b.eq parse_number_exp_e
    cmp w0, #'E'
    b.eq parse_number_exp_e
    b parse_number_done
parse_number_exp_e:
    add x19, x19, #1
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'+'
    b.eq parse_number_exp_sign
    cmp w0, #'-'
    b.eq parse_number_exp_sign
    b parse_number_exp_digits
parse_number_exp_sign:
    add x19, x19, #1
parse_number_exp_digits:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi error_at_pos
    add x19, x19, #1
parse_number_exp_digits_loop:
    cmp x19, x20
    b.ge parse_number_done
    ldrb w0, [x21, x19]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi parse_number_done
    add x19, x19, #1
    b parse_number_exp_digits_loop
parse_number_done:
    ret

expect_literal:
expect_literal_loop:
    cmp x1, #0
    b.eq expect_literal_done
    cmp x19, x20
    b.ge error_at_pos
    ldrb w2, [x21, x19]
    ldrb w3, [x0], #1
    cmp w2, w3
    b.ne error_at_pos
    add x19, x19, #1
    sub x1, x1, #1
    b expect_literal_loop
expect_literal_done:
    ret

parse_value:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'{'
    b.eq parse_object
    cmp w0, #'['
    b.eq parse_array
    cmp w0, #'"'
    b.eq parse_string
    cmp w0, #'-'
    b.eq parse_number
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls parse_number
    cmp w0, #'t'
    b.eq parse_value_true
    cmp w0, #'f'
    b.eq parse_value_false
    cmp w0, #'n'
    b.eq parse_value_null
    b error_at_pos
parse_value_true:
    stp x30, xzr, [sp, #-16]!
    adrp x0, lit_true
    add x0, x0, :lo12:lit_true
    mov x1, #4
    bl expect_literal
    ldp x30, xzr, [sp], #16
    ret
parse_value_false:
    stp x30, xzr, [sp, #-16]!
    adrp x0, lit_false
    add x0, x0, :lo12:lit_false
    mov x1, #5
    bl expect_literal
    ldp x30, xzr, [sp], #16
    ret
parse_value_null:
    stp x30, xzr, [sp, #-16]!
    adrp x0, lit_null
    add x0, x0, :lo12:lit_null
    mov x1, #4
    bl expect_literal
    ldp x30, xzr, [sp], #16
    ret

parse_object:
    stp x30, xzr, [sp, #-16]!
    add x19, x19, #1
    bl parse_ws
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'}'
    b.eq parse_object_close
    b parse_object_member
parse_object_member:
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #'"'
    b.ne error_at_pos
    bl parse_string
    bl parse_ws
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #':'
    b.ne error_at_pos
    add x19, x19, #1
    bl parse_ws
    bl parse_value
    bl parse_ws
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #','
    b.eq parse_object_comma
    cmp w0, #'}'
    b.eq parse_object_close
    b error_at_pos
parse_object_comma:
    add x19, x19, #1
    bl parse_ws
    b parse_object_member
parse_object_close:
    add x19, x19, #1
    ldp x30, xzr, [sp], #16
    ret

parse_array:
    stp x30, xzr, [sp, #-16]!
    add x19, x19, #1
    bl parse_ws
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #']'
    b.eq parse_array_close
parse_array_value:
    bl parse_value
    bl parse_ws
    cmp x19, x20
    b.ge error_at_pos
    ldrb w0, [x21, x19]
    cmp w0, #','
    b.eq parse_array_comma
    cmp w0, #']'
    b.eq parse_array_close
    b error_at_pos
parse_array_comma:
    add x19, x19, #1
    bl parse_ws
    b parse_array_value
parse_array_close:
    add x19, x19, #1
    ldp x30, xzr, [sp], #16
    ret

error_at_pos:
    adrp x0, err_prefix
    add x0, x0, :lo12:err_prefix
    mov x1, x0
    mov x2, #9
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, x19
    bl print_decimal
    adrp x0, nl
    add x0, x0, :lo12:nl
    mov x1, x0
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

print_decimal:
    adrp x1, decbuf
    add x1, x1, :lo12:decbuf
    add x1, x1, #16
    mov x2, #10
print_decimal_loop:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add x4, x4, #'0'
    sub x1, x1, #1
    strb w4, [x1]
    mov x0, x3
    cmp x0, #0
    b.ne print_decimal_loop
    adrp x5, decbuf
    add x5, x5, :lo12:decbuf
    add x5, x5, #16
    sub x2, x5, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .rodata

lit_false: .ascii "false"

.bss
