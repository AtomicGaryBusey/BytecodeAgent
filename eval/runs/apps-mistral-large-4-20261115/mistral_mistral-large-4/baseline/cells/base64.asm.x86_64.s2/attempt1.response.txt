.equ INPUT_MAX, 49152
.equ OUTBUF_SIZE, 70000
.equ LINE_LEN, 76

.section .rodata
.balign 8
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero INPUT_MAX
outbuf:
    .zero OUTBUF_SIZE

.text
.globl _start
_start:
    xor r13, r13
read_loop:
    mov rdx, INPUT_MAX
    sub rdx, r13
    test rdx, rdx
    jz encode
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    syscall
    test rax, rax
    jz encode
    add r13, rax
    jmp read_loop

encode:
    lea r12, [rip + inbuf]
    lea r14, [rip + outbuf]
    xor r15, r15
    mov rcx, r13
full_loop:
    cmp rcx, 3
    jb handle_rem
    movzx eax, byte ptr [r12]
    movzx ebx, byte ptr [r12 + 1]
    movzx edx, byte ptr [r12 + 2]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ebx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ebx
    and r10d, 15
    shl r10d, 2
    mov r11d, edx
    shr r11d, 6
    or r10d, r11d
    mov r11d, edx
    and r11d, 63
    mov eax, r8d
    call emit_sextet
    mov eax, r9d
    call emit_sextet
    mov eax, r10d
    call emit_sextet
    mov eax, r11d
    call emit_sextet
    add r12, 3
    sub rcx, 3
    jmp full_loop

handle_rem:
    test rcx, rcx
    jz finish
    movzx eax, byte ptr [r12]
    xor ebx, ebx
    xor edx, edx
    cmp rcx, 2
    jb rem_one
    movzx ebx, byte ptr [r12 + 1]
rem_one:
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ebx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ebx
    and r10d, 15
    shl r10d, 2
    cmp rcx, 1
    je rem_one_emit
    mov eax, r8d
    call emit_sextet
    mov eax, r9d
    call emit_sextet
    mov eax, r10d
    call emit_sextet
    mov al, 61
    call emit_byte
    jmp finish
rem_one_emit:
    mov eax, r8d
    call emit_sextet
    mov eax, r9d
    call emit_sextet
    mov al, 61
    call emit_byte
    mov al, 61
    call emit_byte
finish:
    test r15, r15
    jz write_out
    mov byte ptr [r14], 10
    inc r14
write_out:
    lea rax, [rip + outbuf]
    sub r14, rax
    test r14, r14
    jz exit0
    lea rsi, [rip + outbuf]
    mov rdx, r14
write_loop:
    test rdx, rdx
    jz exit0
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit0
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall

emit_sextet:
    lea rsi, [rip + alphabet]
    mov al, byte ptr [rsi + rax]
    jmp emit_byte
emit_byte:
    mov byte ptr [r14], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne emit_byte_done
    mov byte ptr [r14], 10
    inc r14
    xor r15, r15
emit_byte_done:
    ret
