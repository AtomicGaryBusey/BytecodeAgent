.text
.globl _start
.equ BUFSZ, 65536
.equ OUTSZ, 64

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, BUFSZ
    syscall
    test rax, rax
    jz input_done
    mov rcx, rax
    lea rsi, [rip + inbuf]

process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r14
    cmp al, 0x0A
    jne check_ws
    inc r12
    jmp is_ws
check_ws:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    test r15, r15
    jnz process_loop
    inc r13
    mov r15, 1
    jmp process_loop
is_ws:
    xor r15, r15
    jmp process_loop

input_done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call append_num
    mov al, 0x20
    mov byte ptr [rdi], al
    inc rdi
    mov rax, r13
    call append_num
    mov al, 0x20
    mov byte ptr [rdi], al
    inc rdi
    mov rax, r14
    call append_num
    mov al, 0x0A
    mov byte ptr [rdi], al
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_num:
    lea rsi, [rip + outbuf]
    add rsi, OUTSZ
    test rax, rax
    jnz append_div
    dec rsi
    mov al, 0x30
    mov byte ptr [rsi], al
    jmp append_copy
append_div:
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz append_div
append_copy:
    lea r8, [rip + outbuf]
    add r8, OUTSZ
    sub r8, rsi
append_copy_loop:
    test r8, r8
    jz append_done
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec r8
    jmp append_copy_loop
append_done:
    ret

.bss
.balign 16
inbuf:
    .zero BUFSZ
outbuf:
    .zero OUTSZ
