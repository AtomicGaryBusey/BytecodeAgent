.equ BANG, 0x21
.equ CMD_PLUS, 0x2B
.equ CMD_COMMA, 0x2C
.equ CMD_MINUS, 0x2D
.equ CMD_DOT, 0x2E
.equ CMD_LT, 0x3C
.equ CMD_GT, 0x3E
.equ CMD_OPEN, 0x5B
.equ CMD_CLOSE, 0x5D

.text
.balign 4
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0
    movz x9, #1, lsl #16

    mov x8, #63
    mov x0, #0
    add x1, x20, x19
    mov x2, x9
    sub x2, x2, x19

    cmp x0, #0

    b.mi read_error
    add x19, x19, x0
    b read_loop

    mov x22, #0

    cmp x22, x19
    b.hs no_bang
    ldrb w14, [x20, x22]
    cmp w14, #BANG
    b.eq found_bang
    add x22, x22, #1

    add x22, x22, #1

    mov x22, x19

    add x11, x11, :lo12:open_stack

    mov x13, #0

    b.hs pre_done
    ldrb w14, [x20, x12]
    cmp w14, #CMD_OPEN
    b.eq pre_open
    cmp w14, #CMD_CLOSE

    add x12, x12, #1
    b pre_loop
pre_open:
    str w12, [x11, x13, lsl #2]
    add x13, x13, #1
    add x12, x12, #1
    b pre_loop
pre_close:
    cmp x13, #0
    b.eq invalid
    sub x13, x13, #1
    ldr w15, [x11, x13, lsl #2]
    str w15, [x10, x12, lsl #2]
    str w12, [x10, x15, lsl #2]
    add x12, x12, #1
    b pre_loop
pre_done:
    cmp x13, #0
    b.ne invalid
    adrp x24, tape
    add x24, x24, :lo12:tape
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
    mov x25, #0
    mov x26, #0
    mov x27, #0
run_loop:
    cmp x26, x21
    b.hs run_end
    ldrb w14, [x20, x26]
    cmp w14, #CMD_GT
    b.eq cmd_gt
    cmp w14, #CMD_LT
    b.eq cmd_lt
    cmp w14, #CMD_PLUS
    b.eq cmd_plus
    cmp w14, #CMD_MINUS
    b.eq cmd_minus
    cmp w14, #CMD_DOT
    b.eq cmd_dot
    cmp w14, #CMD_COMMA
    b.eq cmd_comma
    cmp w14, #CMD_OPEN
    b.eq cmd_open
    cmp w14, #CMD_CLOSE
    b.eq cmd_close
    add x26, x26, #1
    b run_loop
cmd_gt:
    add x25, x25, #1
    add x26, x26, #1
    b run_loop
cmd_lt:
    sub x25, x25, #1
    add x26, x26, #1
    b run_loop
cmd_plus:
    ldrb w15, [x24, x25]
    add w15, w15, #1
    strb w15, [x24, x25]
    add x26, x26, #1
    b run_loop

    sub w15, w15, #1
    strb w15, [x24, x25]
    add x26, x26, #1
    b run_loop
cmd_dot:
    ldrb w15, [x24, x25]
    strb w15, [x28, x27]
    add x27, x27, #1
    add x26, x26, #1
    b run_loop

    b.hs cmd_comma_zero
    ldrb w15, [x20, x22]
    add x22, x22, #1
    b cmd_comma_store
cmd_comma_zero:
    mov w15, #0

    strb w15, [x24, x25]
    add x26, x26, #1
    b run_loop
cmd_open:
    ldrb w15, [x24, x25]
    cmp w15, #0
    b.eq cmd_open_zero
    add x26, x26, #1
    b run_loop
cmd_open_zero:
    ldr w15, [x10, x26, lsl #2]
    add x26, x15, #1
    b run_loop
cmd_close:
    ldrb w15, [x24, x25]
    cmp w15, #0
    b.eq cmd_close_zero
    ldr w15, [x10, x26, lsl #2]
    add x26, x15, #1
    b run_loop
cmd_close_zero:
    add x26, x26, #1
    b run_loop
run_end:
    mov x9, #0
write_loop:
    cmp x9, x27
    b.hs exit_ok
    mov x8, #64
    mov x0, #1
    add x1, x28, x9
    sub x2, x27, x9
    svc #0
    cmp x0, #0
    b.le exit_error
    add x9, x9, x0
    b write_loop
exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

    mov x0, #1
    mov x8, #93
    svc #0
read_error:
    mov x0, #1
    mov x8, #93
    svc #0
exit_error:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
