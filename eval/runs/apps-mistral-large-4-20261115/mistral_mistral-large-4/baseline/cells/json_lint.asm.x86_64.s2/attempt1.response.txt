.equ BUF_SIZE, 65536
.equ OK_LEN, 3
.equ ERR_PREFIX_LEN, 9

.section .rodata
ok_msg: .ascii "ok\n"
err_prefix: .ascii "error at "
true_str: .ascii "true"
false_str: .ascii "false"
null_str: .ascii "null"

.bss
.balign 8
buf: .zero BUF_SIZE
outbuf: .zero 32
digits: .zero 16

.text
.globl _start
_start:
    lea r13, [rip + buf]
    xor r14, r14
read_loop:
    mov rdx, BUF_SIZE
    sub rdx, r14
    test rdx, rdx
    jz read_done
    xor edi, edi
    lea rsi, [r13 + r14]
    xor eax, eax
    syscall
    cmp rax, 0
    je read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    call parse_document
    test rax, rax
    jnz error_out
    lea rsi, [rip + ok_msg]
    mov edx, OK_LEN
    mov edi, 1
    call write_all
    mov edi, 0
    mov eax, 60
    syscall

write_all:
    test rdx, rdx
    jz wa_done
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_all
wa_done:
    ret

parse_document:
    call skip_ws
    call parse_value
    test rax, rax
    jnz pd_ret
    call skip_ws
    cmp r12, r14
    jne pd_err
    xor eax, eax
    ret
pd_err:
    mov rax, r12
    inc rax
    ret
pd_ret:
    ret

skip_ws:
    cmp r12, r14
    jae sw_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x20
    je sw_next
    cmp al, 0x09
    je sw_next
    cmp al, 0x0A
    je sw_next
    cmp al, 0x0D
    je sw_next
    jmp sw_done
sw_next:
    inc r12
    jmp skip_ws
sw_done:
    ret

parse_value:
    cmp r12, r14
    jae pv_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6E
    je parse_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb pv_err
    cmp al, 0x39
    ja pv_err
    jmp parse_number
pv_eof:
    mov rax, r14
    inc rax
    ret
pv_err:
    mov rax, r12
    inc rax
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r14
    jae po_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x7D
    je po_empty
po_loop:
    cmp r12, r14
    jae po_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x22
    jne po_err
    call parse_string
    test rax, rax
    jnz po_ret
    call skip_ws
    cmp r12, r14
    jae po_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x3A
    jne po_err
    inc r12
    call skip_ws
    call parse_value
    test rax, rax
    jnz po_ret
    call skip_ws
    cmp r12, r14
    jae po_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    je po_done
    jmp po_err
po_comma:
    inc r12
    call skip_ws
    jmp po_loop
po_empty:
    inc r12
    xor eax, eax
    ret
po_done:
    inc r12
    xor eax, eax
    ret
po_eof:
    mov rax, r14
    inc rax
    ret
po_err:
    mov rax, r12
    inc rax
    ret
po_ret:
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r14
    jae pa_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x5D
    je pa_empty
pa_loop:
    call parse_value
    test rax, rax
    jnz pa_ret
    call skip_ws
    cmp r12, r14
    jae pa_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    je pa_done
    jmp pa_err
pa_comma:
    inc r12
    call skip_ws
    jmp pa_loop
pa_empty:
    inc r12
    xor eax, eax
    ret
pa_done:
    inc r12
    xor eax, eax
    ret
pa_eof:
    mov rax, r14
    inc rax
    ret
pa_err:
    mov rax, r12
    inc rax
    ret
pa_ret:
    ret

parse_string:
    inc r12
ps_loop:
    cmp r12, r14
    jae ps_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x22
    je ps_done
    cmp al, 0x5C
    je ps_escape
    cmp al, 0x20
    jb ps_err
    inc r12
    jmp ps_loop
ps_escape:
    inc r12
    cmp r12, r14
    jae ps_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x22
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, 0x2F
    je ps_esc_ok
    cmp al, 0x62
    je ps_esc_ok
    cmp al, 0x66
    je ps_esc_ok
    cmp al, 0x6E
    je ps_esc_ok
    cmp al, 0x72
    je ps_esc_ok
    cmp al, 0x74
    je ps_esc_ok
    cmp al, 0x75
    je ps_unicode
    jmp ps_err
ps_esc_ok:
    inc r12
    jmp ps_loop
ps_unicode:
    inc r12
    mov ecx, 4
ps_hex_loop:
    cmp r12, r14
    jae ps_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb ps_err
    cmp al, 0x39
    jbe ps_hex_ok
    cmp al, 0x61
    jb ps_hex_upper
    cmp al, 0x66
    jbe ps_hex_ok
    jmp ps_err
ps_hex_upper:
    cmp al, 0x41
    jb ps_err
    cmp al, 0x46
    ja ps_err
ps_hex_ok:
    inc r12
    dec ecx
    jnz ps_hex_loop
    jmp ps_loop
ps_done:
    inc r12
    xor eax, eax
    ret
ps_eof:
    mov rax, r14
    inc rax
    ret
ps_err:
    mov rax, r12
    inc rax
    ret

parse_number:
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2D
    jne pn_int
    inc r12
    cmp r12, r14
    jae pn_eof
pn_int:
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    je pn_zero
    cmp al, 0x31
    jb pn_err
    cmp al, 0x39
    ja pn_err
    inc r12
pn_digits:
    cmp r12, r14
    jae pn_after_int
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_after_int
    cmp al, 0x39
    ja pn_after_int
    inc r12
    jmp pn_digits
pn_zero:
    inc r12
    cmp r12, r14
    jae pn_after_int
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_after_int
    cmp al, 0x39
    ja pn_after_int
    jmp pn_err
pn_after_int:
    cmp r12, r14
    jae pn_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2E
    jne pn_exp
    inc r12
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_err
    cmp al, 0x39
    ja pn_err
    inc r12
pn_frac_digits:
    cmp r12, r14
    jae pn_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    ja pn_exp
    inc r12
    jmp pn_frac_digits
pn_exp:
    cmp r12, r14
    jae pn_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x65
    je pn_exp_e
    cmp al, 0x45
    jne pn_done
pn_exp_e:
    inc r12
    cmp r12, r14
    jae pn_eof
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x2B
    je pn_exp_sign
    cmp al, 0x2D
    je pn_exp_sign
    jmp pn_exp_digits
pn_exp_sign:
    inc r12
    cmp r12, r14
    jae pn_eof
pn_exp_digits:
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_err
    cmp al, 0x39
    ja pn_err
    inc r12
pn_exp_digit_loop:
    cmp r12, r14
    jae pn_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x30
    jb pn_done
    cmp al, 0x39
    ja pn_done
    inc r12
    jmp pn_exp_digit_loop
pn_eof:
    mov rax, r14
    inc rax
    ret
pn_err:
    mov rax, r12
    inc rax
    ret
pn_done:
    xor eax, eax
    ret

parse_true:
    lea rsi, [rip + true_str]
    mov ecx, 4
    jmp parse_lit
parse_false:
    lea rsi, [rip + false_str]
    mov ecx, 5
    jmp parse_lit
parse_null:
    lea rsi, [rip + null_str]
    mov ecx, 4
    jmp parse_lit

parse_lit:
    test ecx, ecx
    jz pl_done
    cmp r12, r14
    jae pl_eof
    movzx eax, byte ptr [r13 + r12]
    movzx edx, byte ptr [rsi]
    cmp al, dl
    jne pl_err
    inc r12
    inc rsi
    dec ecx
    jmp parse_lit
pl_done:
    xor eax, eax
    ret
pl_eof:
    mov rax, r14
    inc rax
    ret
pl_err:
    mov rax, r12
    inc rax
    ret

error_out:
    dec rax
    lea r8, [rip + outbuf]
    lea rsi, [rip + err_prefix]
    mov ecx, ERR_PREFIX_LEN
err_copy:
    test ecx, ecx
    jz err_conv
    mov al, byte ptr [rsi]
    mov byte ptr [r8], al
    inc rsi
    inc r8
    dec ecx
    jmp err_copy
err_conv:
    lea rdi, [rip + digits]
    xor ecx, ecx
    mov r9, 10
err_div_loop:
    xor edx, edx
    div r9
    add dl, 0x30
    mov byte ptr [rdi], dl
    inc rdi
    inc ecx
    test rax, rax
    jnz err_div_loop
    lea rsi, [rip + digits]
    add rsi, rcx
    dec rsi
err_copy_digits:
    test ecx, ecx
    jz err_newline
    mov al, byte ptr [rsi]
    mov byte ptr [r8], al
    inc r8
    dec rsi
    dec ecx
    jmp err_copy_digits
err_newline:
    mov byte ptr [r8], 0x0A
    inc r8
    lea rsi, [rip + outbuf]
    mov rdx, r8
    sub rdx, rsi
    mov edi, 1
    call write_all
    mov edi, 1
    mov eax, 60
    syscall
