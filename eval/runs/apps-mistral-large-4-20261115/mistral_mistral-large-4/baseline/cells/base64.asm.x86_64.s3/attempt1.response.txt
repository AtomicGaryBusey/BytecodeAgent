.equ IN_MAX, 49152
.equ OUT_MAX, 70000

.section .rodata
.balign 16
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 32
inbuf:
    .zero IN_MAX
outbuf:
    .zero OUT_MAX

.text
.globl _start
_start:
    xor r12, r12
    lea r13, [rip + inbuf]
read_loop:
    mov rdx, IN_MAX
    sub rdx, r12
    jz read_done
    xor rdi, rdi
    mov rsi, r13
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    add r13, rax
    jmp read_loop
read_done:
    lea r13, [rip + inbuf]
    lea r14, [rip + outbuf]
    lea r9, [rip + b64_table]
    xor r15, r15
    mov rbx, r12
encode_loop:
    cmp rbx, 3
    jb final_group
    cmp r15, 76
    jne 1f
    mov byte ptr [r14], 10
    inc r14
    xor r15, r15
1:
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    movzx edx, byte ptr [r13 + 2]
    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14], r8b
    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, ecx
    shr r10d, 4
    or r8d, r10d
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14 + 1], r8b
    mov r8d, ecx
    and r8d, 15
    shl r8d, 2
    mov r10d, edx
    shr r10d, 6
    or r8d, r10d

    mov byte ptr [r14 + 2], r8b
    mov r8d, edx
    and r8d, 63
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14 + 3], r8b
    add r14, 4
    add r15, 4
    add r13, 3
    sub rbx, 3
    jmp encode_loop
final_group:
    test rbx, rbx
    jz encode_done
    cmp rbx, 1
    je final_one
    cmp r15, 76
    jne 2f
    mov byte ptr [r14], 10
    inc r14
    xor r15, r15
2:
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14], r8b
    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, ecx
    shr r10d, 4
    or r8d, r10d
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14 + 1], r8b
    mov r8d, ecx
    and r8d, 15
    shl r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14 + 2], r8b
    mov byte ptr [r14 + 3], 61
    add r14, 4
    add r15, 4
    jmp encode_done
final_one:
    cmp r15, 76
    jne 3f
    mov byte ptr [r14], 10
    inc r14
    xor r15, r15
3:
    movzx eax, byte ptr [r13]
    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14], r8b
    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [r14 + 1], r8b
    mov byte ptr [r14 + 2], 61
    mov byte ptr [r14 + 3], 61
    add r14, 4
    add r15, 4
encode_done:
    test r15, r15
    jz no_final_newline
    mov byte ptr [r14], 10
    inc r14
no_final_newline:
    lea rdx, [rip + outbuf]
    sub r14, rdx
    mov rdx, r14
    lea rsi, [rip + outbuf]
    mov rdi, 1
    test rdx, rdx
    jz write_done
write_loop:
    mov eax, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_done:
    mov eax, 60
    xor rdi, rdi
    syscall
