    add x24, x24, :lo12:line_ptr

// cmp_lines: x0=ptrA, x1=lenA, x2=ptrB, x3=lenB -> x0=0 (A<B), 1 (equal), 2 (A>B)

    cmp x1, x3
    csel x4, x1, x3, ls
cmp_loop:
    cbz x4, cmp_len
    ldrb w5, [x0], #1
    ldrb w6, [x2], #1
    cmp w5, w6
    b.lo cmp_less
    b.hi cmp_greater
    sub x4, x4, #1
    b cmp_loop

    cmp x1, x3
    b.lo cmp_less
    b.hi cmp_greater
    mov x0, #1
    ret

    mov x0, #0
    ret

    ret

// sort_lines: x0 = count -> x0 = count

    mov x20, x0

    ldr x7, [x22, x23, lsl #3]

    str x7, [x22, x24, lsl #3]
    str x4, [x21, x23, lsl #3]
