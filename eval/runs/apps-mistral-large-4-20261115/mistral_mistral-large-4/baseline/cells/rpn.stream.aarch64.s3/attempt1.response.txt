.equ INPUT_BUF_SIZE, 65536
.equ OUT_SIZE, 400000
.equ VALSTACK_SIZE, 524288
.equ NUM_SIZE, 32
.equ ERR_LEN, 6

.section .rodata
err_str:
    .ascii "error\n"

.section .bss
.balign 8
input_buf:
    .zero INPUT_BUF_SIZE
out_buf:
    .zero OUT_SIZE
val_stack:
    .zero VALSTACK_SIZE
num_buf:
    .zero NUM_SIZE

.text
.globl _start
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x19, x1
    mov x20, #0
read_loop:
    mov x2, #INPUT_BUF_SIZE
    sub x2, x2, x20
    cmp x2, #0
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x23, out_buf
    add x23, x23, :lo12:out_buf
    mov x24, x23
    mov x21, #0
line_loop:
    cmp x21, x20
    b.eq all_done
    mov x22, x21
scan_nl:
    cmp x22, x20
    b.eq final_line
    ldrb w6, [x19, x22]
    cmp w6, #0x0A
    b.eq found_nl
    add x22, x22, #1
    b scan_nl
found_nl:
    add x0, x19, x21
    add x1, x19, x22
    bl process_line
    add x21, x22, #1
    b line_loop
final_line:
    add x0, x19, x21
    add x1, x19, x20
    bl process_line
all_done:
    cmp x24, x23
    b.eq write_done
write_loop:
    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64
    svc #0
    add x23, x23, x0
    cmp x23, x24
    b.ne write_loop
write_done:
    mov x0, #0
    mov x8, #93
    svc #0

process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov x25, x0
    mov x26, x1
    mov x27, #0
token_loop:
    cmp x25, x26
    b.hs line_finish
skip_sep:
    cmp x25, x26
    b.hs line_finish
    ldrb w8, [x25]
    cmp w8, #0x20
    b.eq skip_next
    cmp w8, #0x09
    b.eq skip_next
    b token_found
skip_next:
    add x25, x25, #1
    b skip_sep
token_found:
    mov x28, x25
token_scan:
    cmp x25, x26
    b.hs token_end
    ldrb w8, [x25]
    cmp w8, #0x20
    b.eq token_end
    cmp w8, #0x09
    b.eq token_end
    add x25, x25, #1
    b token_scan
token_end:
    mov x0, x28
    mov x1, x25
    bl process_token
    cmp x0, #0
    b.ne line_error
    b token_loop
line_finish:
    cmp x27, #1
    b.ne line_error
    adrp x0, val_stack
    add x0, x0, :lo12:val_stack
    ldr x0, [x0]
    bl output_value
    b line_done
line_error:
    bl output_error
line_done:
    ldp x29, x30, [sp], #16
    ret

process_token:
    sub x2, x1, x0
    cmp x2, #1
    b.eq pt_len1
    ldrb w3, [x0]
    cmp w3, #0x2D
    b.eq pt_neg
    cmp w3, #0x30
    b.lo pt_invalid
    cmp w3, #0x39
    b.hi pt_invalid
    mov x4, x0
    mov x5, x1
    mov x8, #0
    b pt_parse
pt_neg:
    add x4, x0, #1
    cmp x4, x1
    b.hs pt_invalid
    mov x5, x1
    mov x8, #1
    b pt_parse
pt_len1:
    ldrb w3, [x0]
    cmp w3, #0x2B
    b.eq pt_is_op
    cmp w3, #0x2D
    b.eq pt_is_op
    cmp w3, #0x2A
    b.eq pt_is_op
    cmp w3, #0x2F
    b.eq pt_is_op
    cmp w3, #0x25
    b.eq pt_is_op
    cmp w3, #0x30
    b.lo pt_invalid
    cmp w3, #0x39
    b.hi pt_invalid
    sub w3, w3, #0x30
    mov x6, w3
