.text
.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0
read_loop:
    mov x0, #0
    mov x1, x20
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    cmp x19, #65536
    b.eq read_done
    b read_loop
read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x26, hexdigits
    add x26, x26, :lo12:hexdigits
    mov x22, #0
row_loop:
    cmp x22, x19
    b.hs rows_done
    mov x23, #16
    sub x24, x19, x22
    cmp x24, x23
    csel x23, x24, x23, lo
    mov x25, x21
    mov x3, #7
row_hex_loop:
    lsl x4, x3, #2
    lsr x5, x22, x4
    and x5, x5, #0xf
    ldrb w6, [x26, x5]
    strb w6, [x25], #1
    subs x3, x3, #1
    b.pl row_hex_loop
    mov w6, #0x20
    strb w6, [x25], #1
    strb w6, [x25], #1
    add x27, x20, x22
    mov x28, #0
hex_loop:
    cmp x28, #16
    b.hs hex_done
    cmp x28, x23
    b.hs blank_slot
    ldrb w4, [x27, x28]
    lsr w5, w4, #4
    uxtw x5, w5
    ldrb w6, [x26, x5]
    strb w6, [x25], #1
    and w5, w4, #0xf
    uxtw x5, w5
    ldrb w6, [x26, x5]
    strb w6, [x25], #1
    b after_slot
blank_slot:
    mov w6, #0x20
    strb w6, [x25], #1
    strb w6, [x25], #1
after_slot:
    mov w6, #0x20
    strb w6, [x25], #1
    cmp x28, #7
    b.ne no_extra
    strb w6, [x25], #1
no_extra:
    add x28, x28, #1
    b hex_loop
hex_done:
    mov w6, #0x20
    strb w6, [x25], #1
    mov w6, #0x7c
    strb w6, [x25], #1
    mov x28, #0
ascii_loop:
    cmp x28, x23
    b.hs ascii_done
    ldrb w4, [x27, x28]
    cmp w4, #0x20
    b.lo dot
    cmp w4, #0x7e
    b.hi dot
    b print_char
dot:
    mov w4, #0x2e
print_char:
    strb w4, [x25], #1
    add x28, x28, #1
    b ascii_loop
ascii_done:
    mov w6, #0x7c
    strb w6, [x25], #1
    mov w6, #0x0a
    strb w6, [x25], #1
    sub x2, x25, x21
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0
    add x22, x22, #16
    b row_loop
rows_done:
    mov x25, x21
    mov x3, #7
final_hex_loop:
    lsl x4, x3, #2
    lsr x5, x19, x4
    and x5, x5, #0xf
    ldrb w6, [x26, x5]
    strb w6, [x25], #1
    subs x3, x3, #1
    b.pl final_hex_loop
    mov w6, #0x0a
    strb w6, [x25], #1
    sub x2, x25, x21
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 8
inbuf:
    .zero 65536
.balign 8
outbuf:
    .zero 80
