.globl _start
.equ BUFSIZE, 65536

.text
_start:
    cld
    lea r9, [rip + inbuf]
    xor r8d, r8d
read_loop:
    cmp r8, BUFSIZE
    jae read_done
    mov edi, 0
    mov rsi, r9
    mov rdx, BUFSIZE
    sub rdx, r8
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add r9, rax
    jmp read_loop
read_done:
    lea r9, [rip + inbuf]
    xor r10d, r10d
row_loop:
    cmp r10, r8
    jae final_line
    mov r12, r8
    sub r12, r10
    cmp r12, 16
    jbe n_ok
    mov r12d, 16
n_ok:
    lea rdi, [rip + linebuf]
    mov ecx, 79
    mov al, 0x20
    rep stosb
    lea rdi, [rip + linebuf]
    mov eax, r10d
    mov ecx, 8
    lea rbx, [rip + hexdigits]
offset_loop:
    rol eax, 4
    movzx edx, al
    and edx, 0xF
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi], dl
    inc rdi
    dec ecx
    jnz offset_loop
    xor ecx, ecx
    mov rsi, r9
hex_loop:
    cmp ecx, r12d
    jae hex_done
    mov rax, rcx
    imul rax, rax, 3
    add rax, 10
    cmp ecx, 8
    jb no_extra
    inc rax
no_extra:
    lea rdx, [rip + linebuf]
    add rdx, rax
    movzx eax, byte ptr [rsi]
    mov r15d, eax
    shr eax, 4
    movzx eax, byte ptr [rbx + rax]
    mov [rdx], al
    mov eax, r15d
    and eax, 0xF
    movzx eax, byte ptr [rbx + rax]
    mov [rdx + 1], al
    inc rsi
    inc ecx
    jmp hex_loop
hex_done:
    lea rdi, [rip + linebuf + 60]
    mov byte ptr [rdi], 0x7C
    lea rdi, [rip + linebuf + 61]
    mov rsi, r9
    xor ecx, ecx
ascii_loop:
    cmp ecx, r12d
    jae ascii_done
    movzx eax, byte ptr [rsi]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7E
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov eax, 0x2E
ascii_store:
    mov [rdi], al
    inc rdi
    inc rsi
    inc ecx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 0x7C
    inc rdi
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    add r10, r12
    add r9, r12
    jmp row_loop
final_line:
    lea rdi, [rip + linebuf]
    mov eax, r8d
    mov ecx, 8
    lea rbx, [rip + hexdigits]
final_loop:
    rol eax, 4
    movzx edx, al
    and edx, 0xF
    movzx edx, byte ptr [rbx + rdx]
    mov [rdi], dl
    inc rdi
    dec ecx
    jnz final_loop
    mov byte ptr [rdi], 0x0A
    lea rsi, [rip + linebuf]
    mov edx, 9
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 0
    mov eax, 60
    syscall

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero BUFSIZE
linebuf:
    .zero 80
