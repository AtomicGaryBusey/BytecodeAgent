.equ BUF_SIZE, 4096
.equ OUT_CAP, 32

.text
put_dec:
  cmp x0, #0
  b.ne put_dec_nonzero
  mov w2, #48
  sub x1, x1, #1
  strb w2, [x1]
  mov x0, x1
  ret
put_dec_nonzero:
  mov x3, #10
put_dec_loop:
  udiv x4, x0, x3
  msub x5, x4, x3, x0
  add w5, w5, #48
  sub x1, x1, #1
  strb w5, [x1]
  mov x0, x4
  cmp x0, #0
  b.ne put_dec_loop
  mov x0, x1
  ret

.globl _start
_start:
  mov x19, #0
  mov x20, #0
  mov x21, #0
  mov x22, #0
  adrp x23, buf
  add x23, x23, :lo12:buf
read_loop:
  mov x0, #0
  mov x1, x23
  mov x2, #BUF_SIZE
  mov x8, #63
  svc #0
  cbz x0, eof
  mov x3, x0
  mov x4, x23
  add x5, x4, x3
scan_loop:
  cmp x4, x5
  b.hs scan_done
  ldrb w6, [x4], #1
  add x21, x21, #1
  cmp w6, #10
  b.ne scan_not_nl
  add x19, x19, #1
scan_not_nl:
  cmp w6, #32
  b.eq scan_ws
  cmp w6, #9
  b.lo scan_word
  cmp w6, #13
  b.hi scan_word
scan_ws:
  mov x22, #0
  b scan_next
scan_word:
  cmp x22, #0
  b.ne scan_next
  add x20, x20, #1
  mov x22, #1
scan_next:
  b scan_loop
scan_done:
  b read_loop
eof:
  adrp x7, outbuf
  add x7, x7, :lo12:outbuf
  add x7, x7, #OUT_CAP
  mov x1, x7
  mov w0, #10
  sub x1, x1, #1
  strb w0, [x1]
  mov x0, x21
  bl put_dec
  mov x1, x0
  mov w0, #32
  sub x1, x1, #1
  strb w0, [x1]
  mov x0, x20
  bl put_dec
  mov x1, x0
  mov w0, #32
  sub x1, x1, #1
  strb w0, [x1]
  mov x0, x19
  bl put_dec
  mov x1, x0
  sub x2, x7, x1
  mov x0, #1
  mov x8, #64
  svc #0
  mov x0, #0
  mov x8, #93
  svc #0
  b .

.bss
.balign 16
buf:
  .space BUF_SIZE
.balign 8
outbuf:
  .space OUT_CAP
