.equ IN_MAX, 65536
.equ OUT_MAX, 65537

.text
.globl _start
_start:
    xor r8, r8
read_loop:
    cmp r8, IN_MAX
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r8
    mov rdx, IN_MAX
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:
    lea rbx, [rip + inbuf]
    lea rcx, [rip + outbuf]
    xor r9, r9
    xor r10, r10
line_loop:
    cmp r10, r8
    jae all_done
    mov r11, r10
scan_loop:
    cmp r11, r8
    jae reverse_line
    movzx eax, byte ptr [rbx + r11]
    cmp al, 0x0A
    je reverse_line
    inc r11
    jmp scan_loop
reverse_line:
    mov rsi, r11
    dec rsi
rev_loop:
    cmp rsi, r10
    jl line_rev_done
    movzx eax, byte ptr [rbx + rsi]
    mov byte ptr [rcx + r9], al
    inc r9
    dec rsi
    jmp rev_loop
line_rev_done:
    mov al, 0x0A
    mov byte ptr [rcx + r9], al
    inc r9
    cmp r11, r8
    jae all_done
    inc r11
    mov r10, r11
    jmp line_loop
all_done:
    test r9, r9
    jz exit_prog
    mov eax, 1
    mov edi, 1
    mov rsi, rcx
    mov rdx, r9
    syscall
exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero IN_MAX
outbuf:
    .zero OUT_MAX
