.equ INPUT_MAX, 65536
.equ OUT_MAX, 65536
.equ TAPE_SIZE, 30000
.equ MATCH_TABLE_BYTES, 524288
.equ OPEN_STACK_BYTES, 8000

.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, INPUT_MAX
    sub rdx, rbx
    test rdx, rdx
    jz read_done
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea r10, [rip + input_buf]
    xor rdx, rdx
find_loop:
    cmp rdx, rbx
    jae no_bang
    movzx eax, byte ptr [r10 + rdx]
    cmp al, 0x21
    je found_bang
    inc rdx
    jmp find_loop
found_bang:
    lea r14, [r10 + rdx + 1]
    lea r15, [r10 + rbx]
    mov r8, rdx
    jmp validate
no_bang:
    mov r8, rbx
    lea r14, [r10 + rbx]
    lea r15, [r10 + rbx]
validate:
    lea r11, [r10 + r8]
    lea rsi, [r10]
    xor r9, r9
    lea rcx, [rip + open_stack]
    lea rdx, [rip + match_table]
validate_loop:
    cmp rsi, r11
    jae validate_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x5B
    je v_open
    cmp al, 0x5D
    je v_close
    inc rsi
    jmp validate_loop
v_open:
    mov qword ptr [rcx + r9*8], rsi
    inc r9
    inc rsi
    jmp validate_loop
v_close:
    test r9, r9
    jz unbalanced
    dec r9
    mov rax, qword ptr [rcx + r9*8]
    mov rbx, rsi
    sub rbx, r10
    mov qword ptr [rdx + rbx*8], rax
    mov rbx, rax
    sub rbx, r10
    mov qword ptr [rdx + rbx*8], rsi
    inc rsi
    jmp validate_loop
validate_done:
    test r9, r9
    jnz unbalanced
    lea r12, [r10]
    lea r13, [rip + tape]
    lea rbp, [rip + out_buf]
    mov rbx, rbp
    lea r9, [rbp + OUT_MAX]
    lea r8, [rip + match_table]
    jmp cmd_loop

unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

cmd_loop:
    cmp r12, rsi
    jae done
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_dot
    cmp al, 0x2C
    je cmd_comma
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    jmp cmd_loop

cmd_gt:
    inc r13
    jmp cmd_loop
cmd_lt:
    dec r13
    jmp cmd_loop
cmd_plus:
    inc byte ptr [r13]
    jmp cmd_loop
cmd_minus:
    dec byte ptr [r13]
    jmp cmd_loop
cmd_dot:
    mov al, byte ptr [r13]
    mov byte ptr [rbx], al
    inc rbx
    cmp rbx, r9
    jne cmd_loop
    push rsi
    mov rsi, rbp
    mov rdi, 1
    mov rdx, rbx
    sub rdx, rbp
    mov eax, 1
    syscall
    mov rbx, rbp
    pop rsi
    jmp cmd_loop
cmd_comma:
    cmp r14, r15
    jb comma_read
    mov byte ptr [r13], 0
    jmp cmd_loop
comma_read:
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [r13], al
    jmp cmd_loop
cmd_open:
    cmp byte ptr [r13], 0
    jne cmd_loop
    mov rcx, r12
    dec rcx
    sub rcx, r10
    mov rax, qword ptr [r8 + rcx*8]
    inc rax
    mov r12, rax
    jmp cmd_loop
cmd_close:
    cmp byte ptr [r13], 0
    je cmd_loop
    mov rcx, r12
    dec rcx
    sub rcx, r10
    mov rax, qword ptr [r8 + rcx*8]
    inc rax
    mov r12, rax
    jmp cmd_loop

done:
    cmp rbx, rbp
    je exit0
    mov rsi, rbp
    mov rdi, 1
    mov rdx, rbx
    sub rdx, rbp
    mov eax, 1
    syscall
exit0:
    xor edi, edi
    mov eax, 60
    syscall

.bss
.balign 16
input_buf:
    .zero INPUT_MAX
match_table:
    .zero MATCH_TABLE_BYTES
tape:
    .zero TAPE_SIZE
out_buf:
    .zero OUT_MAX
open_stack:
    .zero OPEN_STACK_BYTES
