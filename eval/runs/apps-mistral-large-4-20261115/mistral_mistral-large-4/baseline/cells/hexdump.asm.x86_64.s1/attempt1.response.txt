.equ INPUT_CAP, 65536
.equ LINE_BUF_SIZE, 79

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rdx, INPUT_CAP
    sub rdx, r12
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rbx, rbx
row_loop:
    cmp rbx, r12
    jae rows_done
    mov r14, r12
    sub r14, rbx
    cmp r14, 16
    jbe got_n
    mov r14d, 16
got_n:
    lea rdi, [rip + line_buf]
    mov al, 0x20
    mov rcx, LINE_BUF_SIZE
    rep stosb
    mov rax, rbx
    lea rdi, [rip + line_buf]
    call hex8
    xor rcx, rcx
slot_loop:
    cmp rcx, r14
    jae slot_done
    lea rax, [rcx + rcx*2]
    add rax, 10
    cmp rcx, 8
    jb no_extra
    inc rax
no_extra:
    lea rdi, [rip + line_buf]
    add rdi, rax
    lea rsi, [rip + input_buf]
    add rsi, rbx
    add rsi, rcx
    movzx eax, byte ptr [rsi]
    call hex2
    inc rcx
    jmp slot_loop
slot_done:
    lea rdi, [rip + line_buf]
    mov byte ptr [rdi + 60], 0x7C
    xor rcx, rcx
ascii_loop:
    cmp rcx, r14
    jae ascii_done
    lea rsi, [rip + input_buf]
    add rsi, rbx
    add rsi, rcx
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7E
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov al, 0x2E
ascii_store:
    lea rdi, [rip + line_buf]
    add rdi, rcx
    mov byte ptr [rdi + 61], al
    inc rcx
    jmp ascii_loop
ascii_done:
    lea rdi, [rip + line_buf]
    add rdi, r14
    mov byte ptr [rdi + 61], 0x7C
    lea rdi, [rip + line_buf]
    add rdi, r14
    mov byte ptr [rdi + 62], 0x0A
    mov rdx, r14
    add rdx, 63
    lea rsi, [rip + line_buf]
    mov edi, 1
    mov eax, 1
    syscall
    add rbx, r14
    jmp row_loop
rows_done:
    lea rdi, [rip + line_buf]
    mov rax, r12
    call hex8
    lea rdi, [rip + line_buf]
    mov byte ptr [rdi + 8], 0x0A
    lea rsi, [rip + line_buf]
    mov edi, 1
    mov eax, 1
    mov edx, 9
    syscall
    xor edi, edi
    mov eax, 60
    syscall

hex8:
    mov r8d, eax
    mov r9, rdi
    mov rcx, 8
    xor r10, r10
hex8_loop:
    rol r8d, 4
    movzx eax, r8b
    and eax, 0x0F
    lea r11, [rip + hex_digits]
    movzx eax, byte ptr [r11 + rax]
    mov byte ptr [r9 + r10], al
    inc r10
    dec rcx
    jnz hex8_loop
    ret

hex2:
    movzx eax, al
    mov r8d, eax
    shr r8d, 4
    lea r9, [rip + hex_digits]
    movzx r10d, byte ptr [r9 + r8]
    mov byte ptr [rdi], r10b
    and eax, 0x0F
    movzx r10d, byte ptr [r9 + rax]
    mov byte ptr [rdi + 1], r10b
    ret

.section .rodata
.align 16
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
line_buf:
    .zero 79
