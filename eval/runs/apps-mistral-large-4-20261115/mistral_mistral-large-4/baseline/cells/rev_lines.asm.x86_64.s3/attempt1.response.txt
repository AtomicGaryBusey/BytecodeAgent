.text
.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ NL, 10
.equ IN_CAP, 65536
.equ OUT_CAP, 65537

_start:
    lea r15, [rip + inbuf]
    xor r12, r12
read_loop:
    mov rax, SYS_read
    xor edi, edi
    mov rsi, r15
    add rsi, r12
    mov rdx, IN_CAP
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, IN_CAP
    jb read_loop
read_done:
    lea r14, [rip + outbuf]
    xor r13, r13
    xor rbx, rbx
scan_loop:
    cmp rbx, r12
    jae scan_done
    mov rcx, rbx
find_loop:
    cmp rcx, r12
    jae find_done
    mov al, byte ptr [r15 + rcx]
    cmp al, NL
    je find_done
    inc rcx
    jmp find_loop
find_done:
    mov rdx, rcx
    sub rdx, rbx
    xor r8, r8
copy_loop:
    cmp r8, rdx
    jae copy_done
    mov r9, rcx
    sub r9, 1
    sub r9, r8
    mov al, byte ptr [r15 + r9]
    lea r10, [r14 + r13]
    add r10, r8
    mov byte ptr [r10], al
    inc r8
    jmp copy_loop
copy_done:
    lea r10, [r14 + r13]
    add r10, rdx
    mov byte ptr [r10], NL
    add r13, rdx
    inc r13
    cmp rcx, r12
    jae set_done
    lea rbx, [rcx + 1]
    jmp scan_loop
set_done:
    mov rbx, r12
    jmp scan_loop
scan_done:
    test r13, r13
    jz exit_prog
    xor r8, r8
write_loop:
    cmp r8, r13
    jae exit_prog
    mov rax, SYS_write
    mov edi, 1
    lea rsi, [r14 + r8]
    mov rdx, r13
    sub rdx, r8
    syscall
    cmp rax, 0
    jle exit_prog
    add r8, rax
    jmp write_loop
exit_prog:
    mov rax, SYS_exit
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero IN_CAP
outbuf:
    .zero OUT_CAP
