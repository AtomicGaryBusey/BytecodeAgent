.globl _start
.equ INPUT_CAP, 65536
.equ STACK_CAP, 1024
.equ NUM_CAP, 32

.section .rodata
err_msg: .ascii "error\n"

.text
_start:
    call read_all
    call process_lines
    mov eax, 60
    xor edi, edi
    syscall

read_all:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
read_loop:
    mov rdx, INPUT_CAP
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    ret

process_lines:
    lea rsi, [rip + input_buf]
    lea r8, [rip + input_buf]
    add r8, r12
line_loop:
    cmp rsi, r8
    jae done_lines
    mov rdi, rsi
scan_nl:
    cmp rdi, r8
    jae line_found
    movzx eax, byte ptr [rdi]
    cmp al, 10
    je line_found
    inc rdi
    jmp scan_nl
line_found:
    mov r9, rdi
    cmp rdi, r8
    jae next_done
    inc r9
next_done:
    push r8
    push r9
    call eval_line
    pop rsi
    pop r8
    jmp line_loop
done_lines:
    ret

eval_line:
    push r13
    push r14
    push r15
    mov r13, rdi
    lea r14, [rip + rpn_stack]
    xor r15d, r15d
token_loop:
    cmp rsi, r13
    jae line_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    jmp have_token
skip_sep:
    inc rsi
    jmp token_loop
have_token:
    mov r8, rsi
tok_scan:
    cmp rsi, r13
    jae tok_end
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je tok_end
    cmp al, 0x09
    je tok_end
    inc rsi
    jmp tok_scan
tok_end:
    mov r9, rsi
    mov rax, r9
    sub rax, r8
    cmp rax, 1
    jne try_literal
    movzx eax, byte ptr [r8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp try_literal
try_literal:
    mov r10, r8
    movzx eax, byte ptr [r10]
    xor r11d, r11d
    cmp al, '-'
    jne lit_start
    inc r10
    mov r11d, 1
lit_start:
    cmp r10, r9
    jae invalid
    xor ebx, ebx
    xor ecx, ecx
lit_loop:
    cmp r10, r9
    jae lit_done
    movzx eax, byte ptr [r10]
    sub eax, '0'
    cmp eax, 9
    ja invalid
    inc r10
    inc rbx
    cmp rbx, 18
    ja invalid
    imul rcx, rcx, 10
    add rcx, rax
    jmp lit_loop
lit_done:
    test r11d, r11d
    jz push_lit
    neg rcx
push_lit:
    cmp r15, STACK_CAP
    jae invalid
    mov qword ptr [r14 + r15*8], rcx
    inc r15
    jmp token_loop
op_add:
    cmp r15, 2
    jb invalid
    dec r15
    mov rbx, qword ptr [r14 + r15*8]
    dec r15
    mov rax, qword ptr [r14 + r15*8]
    add rax, rbx
    mov qword ptr [r14 + r15*8], rax
    inc r15
    jmp token_loop
op_sub:
    cmp r15, 2
    jb invalid
    dec r15
    mov rbx, qword ptr [r14 + r15*8]
    dec r15
    mov rax, qword ptr [r14 + r15*8]
    sub rax, rbx
    mov qword ptr [r14 + r15*8], rax
    inc r15
    jmp token_loop
op_mul:
    cmp r15, 2
    jb invalid
    dec r15
    mov rbx, qword ptr [r14 + r15*8]
    dec r15
    mov rax, qword ptr [r14 + r15*8]
    imul rax, rbx
    mov qword ptr [r14 + r15*8], rax
    inc r15
    jmp token_loop
op_div:
    cmp r15, 2
    jb invalid
    dec r15
    mov rbx, qword ptr [r14 + r15*8]
    dec r15
    mov rax, qword ptr [r14 + r15*8]
    test rbx, rbx
    jz invalid
    mov ecx, 0x80000000
    shl rcx, 32
    cmp rax, rcx
    jne div_ok
    cmp rbx, -1
    je invalid
div_ok:
    cqo
    idiv rbx
    mov qword ptr [r14 + r15*8], rax
    inc r15
    jmp token_loop
op_mod:
    cmp r15, 2
    jb invalid
    dec r15
    mov rbx, qword ptr [r14 + r15*8]
    dec r15
    mov rax, qword ptr [r14 + r15*8]
    test rbx, rbx
    jz invalid
    mov ecx, 0x80000000
    shl rcx, 32
    cmp rax, rcx
    jne mod_ok
    cmp rbx, -1
    je invalid
mod_ok:
    cqo
    idiv rbx
    mov qword ptr [r14 + r15*8], rdx
    inc r15
    jmp token_loop
line_done:
    cmp r15, 1
    jne invalid
    mov rax, qword ptr [r14]
    call out_value
    jmp eval_ret
invalid:
    call out_error
eval_ret:
    pop r15
    pop r14
    pop r13
    ret

out_error:
    lea rsi, [rip + err_msg]
    mov edx, 6
    mov edi, 1
    jmp write_all

out_value:
    xor r9d, r9d
    test rax, rax
    jns pos
    mov r9d, 1
    neg rax
pos:
    lea rdi, [rip + num_buf_end]
    dec rdi
    mov byte ptr [rdi], 10
    mov ebx, 10
digit_loop:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz digit_loop
    test r9d, r9d
    jz send
    dec rdi
    mov byte ptr [rdi], '-'
send:
    lea rsi, [rip + num_buf_end]
    sub rsi, rdi
    mov rdx, rsi
    mov rsi, rdi
    mov edi, 1
    jmp write_all

write_all:
    test rdx, rdx
    jz write_done
    mov eax, 1
    syscall
    cmp rax, 0
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_all
write_done:
    ret

.bss
.balign 8
input_buf: .zero INPUT_CAP
rpn_stack: .zero STACK_CAP*8
num_buf: .zero NUM_CAP
num_buf_end:
