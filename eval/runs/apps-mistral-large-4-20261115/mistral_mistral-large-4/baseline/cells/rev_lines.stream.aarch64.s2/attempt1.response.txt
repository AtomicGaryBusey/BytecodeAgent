.text
.globl _start
.equ INPUT_CAP, 65536
.equ OUT_CAP, 65537

_start:
    adrp x26, input_buf
    add x26, x26, :lo12:input_buf
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x19, #0
read_loop:
    mov x2, #INPUT_CAP
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    mov x1, x26
    add x1, x1, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop
read_done:
    mov x20, #0
    mov x22, #0
process_loop:
    cmp x20, x19
    b.ge process_done
    mov x21, x20
find_nl:
    cmp x21, x19
    b.ge line_end
    ldrb w25, [x26, x21]
    cmp w25, #10
    b.eq line_end
    add x21, x21, #1
    b find_nl
line_end:
    cmp x20, x21
    b.ge append_nl
    sub x27, x21, #1
rev_loop:
    ldrb w25, [x26, x27]
    strb w25, [x24, x22]
    add x22, x22, #1
    cmp x27, x20
    b.eq append_nl
    sub x27, x27, #1
    b rev_loop
append_nl:
    mov w25, #10
    strb w25, [x24, x22]
    add x22, x22, #1
    add x20, x21, #1
    b process_loop
process_done:
    mov x20, #0
write_loop:
    cmp x20, x22
    b.ge write_done
    mov x0, #1
    mov x1, x24
    add x1, x1, x20
    sub x2, x22, x20
    mov x8, #64
    svc #0
    add x20, x20, x0
    b write_loop
write_done:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buf:
    .space INPUT_CAP
out_buf:
    .space OUT_CAP
