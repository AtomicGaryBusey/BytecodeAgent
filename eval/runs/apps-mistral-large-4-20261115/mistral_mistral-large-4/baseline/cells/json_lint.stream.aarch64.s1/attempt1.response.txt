  str x30, [sp]
  bl skip_ws
  cmp x21, x20
  b.ge error_at_len
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.ne error_at_idx
  ldr x30, [sp]
  add sp, sp, #16
  ret

  cmp x21, x20
  b.ge skip_ws_done
  ldrb w0, [x19, x21]
  cmp w0, #0x20
  b.eq skip_ws_inc
  cmp w0, #0x09
  b.eq skip_ws_inc
  cmp w0, #0x0A

  cmp w0, #0x0D
  b.eq skip_ws_inc

  b skip_ws

  mov x2, x0
  mov x0, #1
  mov x8, #64
  svc #0
  mov x0, #1
  adrp x1, nl
  add x1, x1, :lo12:nl
  mov x2, #1
  mov x8, #64
  svc #0
  mov x0, #1
  mov x8, #93
  svc #0

  add x3, x1, #16

  msub x6, x5, x4, x2

parse_value:
  sub sp, sp, #16
  str x30, [sp]
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #'{'
  b.eq pv_obj
  cmp w0, #'['
  b.eq pv_arr
  cmp w0, #'"'
  b.eq pv_str
  cmp w0, #'-'
  b.eq pv_num
  cmp w0, #'0'
  b.lo pv_lit
  cmp w0, #'9'
  b.hi pv_lit
  b pv_num
pv_lit:
  cmp w0, #'t'
  b.eq pv_true
  cmp w0, #'f'
  b.eq pv_false
  cmp w0, #'n'
  b.eq pv_null
  b error_at_idx
pv_obj:
  bl parse_object
  b pv_done
pv_arr:
  bl parse_array
  b pv_done
pv_str:
  bl parse_string
  b pv_done
pv_num:
  bl parse_number
  b pv_done
pv_true:
  bl parse_true
  b pv_done
pv_false:
  bl parse_false
  b pv_done
pv_null:
  bl parse_null
  b pv_done

  ldr x30, [sp]
  add sp, sp, #16
  ret

parse_object:
  sub sp, sp, #16
  str x30, [sp]
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #'}'
  b.eq po_close_empty

  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #'"'
  b.ne error_at_idx
  bl parse_string
  bl skip_ws
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #':'
  b.ne error_at_idx
  add x21, x21, #1
  bl skip_ws
  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #','
  b.eq po_comma
  cmp w0, #'}'
  b.eq po_close
  b error_at_idx

  add x21, x21, #1
  bl skip_ws
  b po_loop
po_close:
  add x21, x21, #1
  b po_done
po_close_empty:
  add x21, x21, #1
po_done:
  ldr x30, [sp]
  add sp, sp, #16
  ret

parse_array:
  sub sp, sp, #16
  str x30, [sp]
  add x21, x21, #1
  bl skip_ws
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #']'
  b.eq pa_close_empty

  bl parse_value
  bl skip_ws
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #','
  b.eq pa_comma
  cmp w0, #']'
  b.eq pa_close
  b error_at_idx

  add x21, x21, #1
  bl skip_ws
  b pa_loop
pa_close:
  add x21, x21, #1
  b pa_done
pa_close_empty:
  add x21, x21, #1
pa_done:
  ldr x30, [sp]
  add sp, sp, #16
  ret

parse_string:
  sub sp, sp, #16
  str x30, [sp]
  add x21, x21, #1

  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #'"'
  b.eq ps_end
  cmp w0, #'\\'
  b.eq ps_escape
  cmp w0, #0x20

  add x21, x21, #1
  b ps_loop

  add x21, x21, #1
  cmp x21, x20
  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #'"'
  b.eq ps_esc_ok
  cmp w0, #'\\'
  b.eq ps_esc_ok
  cmp w0, #'/'
  b.eq ps_esc_ok
  cmp w0, #'b'
  b.eq ps_esc_ok
  cmp w0, #'f'
  b.eq ps_esc_ok
  cmp w0, #'n'
  b.eq ps_esc_ok
  cmp w0, #'r'
  b.eq ps_esc_ok
  cmp w0, #'t'
  b.eq ps_esc_ok
  cmp w0, #'u'
  b.eq ps_unicode
  b error_at_idx

  add x21, x21, #1
  b ps_loop

  mov x1, #4

  b.ge error_at_len
  ldrb w0, [x19, x21]
  cmp w0, #'0'

  cmp w0, #'9'

  cmp w0, #'f'
  b.le ps_hex_ok
  b ps_hex_err

  b.lt ps_hex_err
  cmp w0, #'F'

  sub x1, x1, #1

  ldr x30, [sp]
  add sp, sp, #16

  b.eq pn_zero

  ldrb w0, [x19, x21]
  cmp w0, #'.'

  b.ge error_at_len
  ldrb w0, [x19, x21]
