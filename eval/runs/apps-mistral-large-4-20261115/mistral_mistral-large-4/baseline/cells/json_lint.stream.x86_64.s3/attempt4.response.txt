.equ BUF_SIZE, 65552

.text
.globl _start
_start:
    xor r12, r12
    lea rbx, [rip + input_buf]
read_loop:
    mov rax, 0
    xor edi, edi
    mov rsi, rbx
    mov rdx, BUF_SIZE
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    add rbx, rax
    cmp r12, BUF_SIZE
    jb read_loop
read_done:
    mov qword ptr [rip + len], r12
    mov qword ptr [rip + pos], 0
    call parse_document
    test al, al
    jz valid
    lea rsi, [rip + err_prefix]
    mov edx, 9
    call write_all
    call print_err_num
    mov eax, 60
    mov edi, 1
    syscall
valid:
    lea rsi, [rip + ok_msg]
    mov edx, 3
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_all_done
    mov rax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_all
write_all_done:
    ret

print_err_num:
    mov r10, qword ptr [rip + err_pos]
    mov rbx, r10
    xor r8, r8
count_digits:
    inc r8
    mov rax, rbx
    xor edx, edx
    mov r9, 10
    div r9
    mov rbx, rax
    test rbx, rbx
    jnz count_digits
    lea rdi, [rip + numbuf]
    mov byte ptr [rdi + r8], 0x0A
    mov rcx, r8
fill_digits:
    test rcx, rcx
    jz fill_done
    dec rcx
    mov rax, r10
    xor edx, edx
    mov r9, 10
    div r9
    add dl, 0x30
    mov byte ptr [rdi + rcx], dl
    mov r10, rax
    jmp fill_digits
fill_done:
    lea rsi, [rip + numbuf]
    mov rdx, r8
    inc rdx
    call write_all
    ret

advance:
    mov rax, qword ptr [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    ret

skip_ws:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae skip_ws_done
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x20
    je skip_ws_advance
    cmp ecx, 0x09
    je skip_ws_advance
    cmp ecx, 0x0A
    je skip_ws_advance
    cmp ecx, 0x0D
    je skip_ws_advance
    jmp skip_ws_done
skip_ws_advance:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp skip_ws
skip_ws_done:
    ret

fail:
    mov rax, qword ptr [rip + pos]
    mov qword ptr [rip + err_pos], rax
    mov eax, 1
    ret

parse_document:
    lea rbx, [rip + input_buf]
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    call parse_value
    test al, al
    jnz parse_document_ret
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jne parse_document_error
    xor eax, eax
parse_document_ret:
    ret
parse_document_error:
    jmp fail

parse_value:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x7B
    je parse_object
    cmp ecx, 0x5B
    je parse_array
    cmp ecx, 0x22
    je parse_string
    cmp ecx, 0x74
    je parse_true
    cmp ecx, 0x66
    je parse_false
    cmp ecx, 0x6E
    je parse_null
    cmp ecx, 0x2D
    je parse_number
    cmp ecx, 0x30
    jb parse_value_bad
    cmp ecx, 0x39
    jbe parse_number
parse_value_bad:
    jmp fail

parse_object:
    call advance
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x7D
    je object_empty_close
object_member:
    call parse_string
    test al, al
    jnz object_ret
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x3A
    jne fail
    call advance
    call skip_ws
    call parse_value
    test al, al
    jnz object_ret
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x2C
    je object_comma
    cmp ecx, 0x7D
    je object_close
    jmp fail
object_comma:
    call advance
    call skip_ws
    jmp object_member
object_empty_close:
    call advance
    jmp object_done
object_close:
    call advance
object_done:
    xor eax, eax
object_ret:
    ret

parse_array:
    call advance
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x5D
    je array_empty_close
array_loop:
    call parse_value
    test al, al
    jnz array_ret
    call skip_ws
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x2C
    je array_comma
    cmp ecx, 0x5D
    je array_close
    jmp fail
array_comma:
    call advance
    call skip_ws
    jmp array_loop
array_empty_close:
    call advance
    jmp array_done
array_close:
    call advance
array_done:
    xor eax, eax
array_ret:
    ret

parse_string:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x22
    jne fail
    call advance
string_loop:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x22
    je string_done
    cmp ecx, 0x5C
    je string_escape
    cmp ecx, 0x20
    jb fail
    call advance
    jmp string_loop
string_escape:
    call advance
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x22
    je string_escape_ok
    cmp ecx, 0x5C
    je string_escape_ok
    cmp ecx, 0x2F
    je string_escape_ok
    cmp ecx, 0x62
    je string_escape_ok
    cmp ecx, 0x66
    je string_escape_ok
    cmp ecx, 0x6E
    je string_escape_ok
    cmp ecx, 0x72
    je string_escape_ok
    cmp ecx, 0x74
    je string_escape_ok
    cmp ecx, 0x75
    je string_escape_u
    jmp fail
string_escape_ok:
    call advance
    jmp string_loop
string_escape_u:
    call advance
    mov r8, 4
string_hex_loop:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    jb string_hex_letters_lower_check
    cmp ecx, 0x39
    jbe string_hex_ok
string_hex_letters_lower_check:
    cmp ecx, 0x61
    jb string_hex_upper_check
    cmp ecx, 0x66
    jbe string_hex_ok
string_hex_upper_check:
    cmp ecx, 0x41
    jb fail
    cmp ecx, 0x46
    ja fail
string_hex_ok:
    call advance
    dec r8
    jnz string_hex_loop
    jmp string_loop
string_done:
    call advance
    xor eax, eax
    ret

parse_number:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x2D
    jne number_int
    call advance
number_int:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    je number_zero
    cmp ecx, 0x31
    jb fail
    cmp ecx, 0x39
    ja fail
    call advance
number_int_digits:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae number_frac
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    jb number_frac
    cmp ecx, 0x39
    ja number_frac
    call advance
    jmp number_int_digits
number_zero:
    call advance
number_frac:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae number_exp
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x2E
    jne number_exp
    call advance
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    jb fail
    cmp ecx, 0x39
    ja fail
number_frac_digits:
    call advance
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae number_exp
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    jb number_exp
    cmp ecx, 0x39
    ja number_exp
    jmp number_frac_digits
number_exp:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae number_done
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x65
    je number_exp_start
    cmp ecx, 0x45
    je number_exp_start
    jmp number_done
number_exp_start:
    call advance
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x2B
    je number_exp_sign
    cmp ecx, 0x2D
    je number_exp_sign
    jmp number_exp_digit
number_exp_sign:
    call advance
number_exp_digit:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    jb fail
    cmp ecx, 0x39
    ja fail
number_exp_digits:
    call advance
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae number_done
    movzx ecx, byte ptr [rbx + rax]
    cmp ecx, 0x30
    jb number_done
    cmp ecx, 0x39
    ja number_done
    jmp number_exp_digits
number_done:
    xor eax, eax
    ret

match_lit:
match_lit_loop:
    mov rax, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rax, rdx
    jae fail
    movzx r8d, byte ptr [rbx + rax]
    cmp r8b, byte ptr [rsi]
    jne fail
    inc rsi
    call advance
    dec rcx
    jnz match_lit_loop
    xor eax, eax
    ret

parse_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp match_lit

parse_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp match_lit

parse_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
    jmp match_lit

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
str_true:
    .ascii "true"
str_false:
    .ascii "false"
str_null:
    .ascii "null"

.section .bss
.balign 8
input_buf:
    .zero BUF_SIZE
pos:
    .zero 8
len:
    .zero 8
err_pos:
    .zero 8
numbuf:
    .zero 16
