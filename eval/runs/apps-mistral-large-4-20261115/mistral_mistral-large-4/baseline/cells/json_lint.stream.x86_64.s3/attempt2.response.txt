.globl _start
.equ BUF_SIZE, 65536

.section .rodata
ok_msg:
    .byte 0x6F, 0x6B, 0x0A
.equ OK_LEN, . - ok_msg
err_msg:
    .ascii "error at "
.equ ERR_LEN, . - err_msg
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
buf:
    .zero BUF_SIZE
err_off:
    .zero 8
outbuf:
    .zero 32
digits:
    .zero 16

.text
_start:
    lea rbx, [rip + buf]
    xor r14, r14
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r14]
    mov rdx, BUF_SIZE
    sub rdx, r14
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    cmp r14, BUF_SIZE
    jb read_loop
read_done:
    lea r12, [rbx + r14]
    mov rsi, rbx
    call skip_ws
    call parse_value
    test rax, rax
    js error_path
    call skip_ws
    cmp rsi, r12
    je ok_path
    mov rax, rsi
    sub rax, rbx
    mov [rip + err_off], rax
    jmp error_path
ok_path:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, OK_LEN
    syscall
    mov eax, 60
    xor edi, edi
    syscall
error_path:
    lea rsi, [rip + outbuf]
    lea rdi, [rip + err_msg]
    mov ecx, ERR_LEN
err_copy:
    mov al, [rdi]
    mov [rsi], al
    inc rdi
    inc rsi
    dec ecx
    jnz err_copy
    mov rax, [rip + err_off]
    lea r8, [rip + digits]
    add r8, 16
conv_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 0x30
    dec r8
    mov [r8], dl
    test rax, rax
    jnz conv_loop
    lea r9, [rip + digits]
    add r9, 16
    sub r9, r8
err_copy_digits:
    mov al, [r8]
    mov [rsi], al
    inc r8
    inc rsi
    dec r9
    jnz err_copy_digits
    mov byte ptr [rsi], 0x0A
    inc rsi
    mov rdx, rsi
    lea r
