.equ INPUT_MAX, 65536
.equ TAPE_SIZE, 30000
.equ OUT_MAX, 65536
.equ C_BANG, 0x21
.equ C_GT, 62
.equ C_LT, 60
.equ C_PLUS, 43
.equ C_MINUS, 45
.equ C_DOT, 46
.equ C_COMMA, 44
.equ C_OPEN, 91
.equ C_CLOSE, 93

.text
.globl _start
_start:
    lea r9, [rip + input_buf]
    xor r8d, r8d
read_loop:
    cmp r8, INPUT_MAX
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [r9 + r8]
    mov edx, INPUT_MAX
    sub edx, r8d
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:
    mov rsi, r9
    mov r9, r8
    xor ecx, ecx
find_bang:
    cmp rcx, r9
    jae no_bang
    cmp byte ptr [rsi + rcx], C_BANG
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    lea r14, [rsi + rcx + 1]
    mov r8, rcx
    jmp have_prog
no_bang:
    lea r14, [rsi + r9]
have_prog:
    lea r15, [rsi + r9]
    lea r10, [rip + open_stack]
    lea rdx, [rip + match_tab]
    mov rbx, r10
    xor ecx, ecx
check_loop:
    cmp rcx, r8
    jae check_done
    mov al, byte ptr [rsi + rcx]
    cmp al, C_OPEN
    je push_open
    cmp al, C_CLOSE
    je pop_close
    jmp check_next
push_open:
    mov dword ptr [rbx], ecx
    add rbx, 4
    jmp check_next
pop_close:
    cmp rbx, r10
    je unbalanced
    sub rbx, 4
    mov eax, dword ptr [rbx]
    mov dword ptr [rdx + rax*4], ecx
    mov dword ptr [rdx + rcx*4], eax
    jmp check_next
check_next:
    inc rcx
    jmp check_loop
check_done:
    cmp rbx, r10
    jne unbalanced
    mov r9, rdx
    lea r10, [rip + out_buf]
    lea r11, [rsi + r8]
    mov r12, rsi
    lea r13, [rip + tape]
    xor ebx, ebx
exec:
    cmp r12, r11
    jae done
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, C_GT
    je cmd_gt
    cmp al, C_LT
    je cmd_lt
    cmp al, C_PLUS
    je cmd_plus
    cmp al, C_MINUS
    je cmd_minus
    cmp al, C_DOT
    je cmd_out
    cmp al, C_COMMA
    je cmd_in
    cmp al, C_OPEN
    je cmd_open
    cmp al, C_CLOSE
    je cmd_close
    jmp exec
cmd_gt:
    inc r13
    jmp exec
cmd_lt:
    dec r13
    jmp exec
cmd_plus:
    inc byte ptr [r13]
    jmp exec
cmd_minus:
    dec byte ptr [r13]
    jmp exec
cmd_out:
    mov al, byte ptr [r13]
    mov byte ptr [r10 + rbx], al
    inc rbx
    jmp exec
cmd_in:
    cmp r14, r15
    jae in_zero
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [r13], al
    jmp exec
in_zero:
    mov byte ptr [r13], 0
    jmp exec
cmd_open:
    cmp byte ptr [r13], 0
    jne exec
    lea rax, [r12 - 1]
    sub rax, rsi
    mov eax, dword ptr [r9 + rax*4]
    lea r12, [rsi + rax + 1]
    jmp exec
cmd_close:
    cmp byte ptr [r13], 0
    je exec
    lea rax, [r12 - 1]
    sub rax, rsi
    mov eax, dword ptr [r9 + rax*4]
    lea r12, [rsi + rax + 1]
    jmp exec
done:
    test rbx, rbx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, rbx
    syscall
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 8
input_buf:
    .zero INPUT_MAX
.balign 8
tape:
    .zero TAPE_SIZE
.balign 4
match_tab:
    .zero 262144
.balign 4
open_stack:
    .zero 262144
.balign 8
out_buf:
    .zero OUT_MAX
