.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
read_loop:
    movz x4, #1, lsl #16
    cmp x20, x4
    b.hs read_done
    sub x2, x4, x20
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    bl skip_ws
    cmp x21, x20
    b.hs report_error
    bl parse_value
    cmp x0, #0
    b.ne report_error
    bl skip_ws
    cmp x21, x20
    b.ne report_error
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

report_error:
    mov x0, #1
    adrp x1, err_prefix
    add x1, x1, :lo12:err_prefix
    mov x2, #9
    mov x8, #64
    svc #0
    adrp x1, digits
    add x1, x1, :lo12:digits
    add x1, x1, #16
    mov x0, x21
    cbz x0, dec_zero
dec_loop:
    mov x10, #10
    udiv x2, x0, x10
    msub x3, x2, x10, x0
    add x3, x3, #0x30
    sub x1, x1, #1
    strb w3, [x1]
    mov x0, x2
    cbnz x0, dec_loop
    b dec_write
dec_zero:
    sub x1, x1, #1
    mov w3, #0x30
    strb w3, [x1]
dec_write:
    adrp x2, digits
    add x2, x2, :lo12:digits
    add x2, x2, #16
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
    cmp x21, x20
    b.hs sw_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq sw_next
    cmp w0, #0x09
    b.eq sw_next
    cmp w0, #0x0A
    b.eq sw_next
    cmp w0, #0x0D
    b.eq sw_next
    b sw_done
sw_next:
    add x21, x21, #1
    b skip_ws
sw_done:
    ret

parse_lit:
    cbz x2, pl_ok
pl_loop:
    cmp x21, x20
    b.hs pl_err
    ldrb w3, [x19, x21]
    ldrb w4, [x1], #1
    cmp w3, w4
    b.ne pl_err
    add x21, x21, #1
    sub x2, x2, #1
    cbnz x2, pl_loop
pl_ok:
    mov x0, #0
    ret
pl_err:
    mov x0, #1
    ret

parse_string:
    cmp x21, x20
    b.hs ps_err
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.ne ps_err
    add x21, x21, #1
ps_loop:
    cmp x21, x20
    b.hs ps_err
    ldrb w0, [x19, x21]
    cmp w0, #0x22
    b.eq ps_close
    cmp w0, #0x5C
    b.eq ps_esc
    cmp w0, #0x20
    b.lo ps_err
    add x21, x21, #1
    b ps_loop
ps_close:
    add x21, x21, #1
    mov x0, #0
    ret
ps_esc:
    add x21, x21, #1
    cmp x21, x20
    b.hs ps_err
    ldrb w1, [x19, x21]
    cmp w1, #0x22
    b.eq ps_esc_simple
    cmp w1, #0x5C
    b.eq ps_esc_simple
    cmp w1, #0x2F
    b.eq ps_esc_simple
    cmp w1, #0x62
    b.eq ps_esc_simple
    cmp w1, #0x66
    b.eq ps_esc_simple
    cmp w1, #0x6E
    b.eq ps_esc_simple
    cmp w1, #0x72
    b.eq ps_esc_simple
    cmp w1, #0x74
    b.eq ps_esc_simple
    cmp w1, #0x75
    b.eq ps_esc_u
    b ps_err
ps_esc_simple:
    add x21, x21, #1
    b ps_loop
ps_esc_u:
    add x21, x21, #1
    mov x2, #4
ps_hex_loop:
    cmp x21, x20
    b.hs ps_err
    ldrb w1, [x19, x21]
    sub w3, w1, #0x30
    cmp w3, #9
    b.ls ps_hex_ok
    sub w3, w1, #0x61
    cmp w3, #5
    b.ls ps_hex_ok
    sub w3, w1, #0x41
    cmp w3, #5
    b.ls ps_hex_ok
    b ps_err
ps_hex_ok:
    add x21, x21, #1
    sub x2, x2, #1
    cbnz x2, ps_hex_loop
    b ps_loop
ps_err:
    mov x0, #1
    ret

parse_number:
    cmp x21, x20
    b.hs pn_err
    ldrb w0, [x19, x21]
    cmp w0, #0x2D
    b.ne pn_int
    add x21, x21, #1
pn_int:
    cmp x21, x20
    b.hs pn_err
    ldrb w0, [x19, x21]
    cmp w0, #0x30
    b.eq pn_zero
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi pn_err
    add x21, x21, #1
pn_digits:
    cmp x21, x20
    b.hs pn_frac
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi pn_frac
    add x21, x21, #1
    b pn_digits
pn_zero:
    add x21, x21, #1
    b pn_frac
pn_frac:
    cmp x21, x20
    b.hs pn_exp
    ldrb w0, [x19, x21]
    cmp w0, #0x2E
    b.ne pn_exp
    add x21, x21, #1
    cmp x21, x20
    b.hs pn_err
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi pn_err
pn_frac_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs pn_exp
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pn_frac_digits
    b pn_exp
pn_exp:
    cmp x21, x20
    b.hs pn_ok
    ldrb w0, [x19, x21]
    cmp w0, #0x65
    b.eq pn_exp_body
    cmp w0, #0x45
    b.ne pn_ok
pn_exp_body:
    add x21, x21, #1
    cmp x21, x20
    b.hs pn_err
    ldrb w0, [x19, x21]
    cmp w0, #0x2B
    b.eq pn_exp_sign
    cmp w0, #0x2D
    b.eq pn_exp_sign
    b pn_exp_digits_req
pn_exp_sign:
    add x21, x21, #1
pn_exp_digits_req:
    cmp x21, x20
    b.hs pn_err
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi pn_err
pn_exp_digits:
    add x21, x21, #1
    cmp x21, x20
    b.hs pn_ok
    ldrb w0, [x19, x21]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pn_exp_digits
pn_ok:
    mov x0, #0
    ret
pn_err:
    mov x0, #1
    ret

parse_value:
    str x30, [sp, #-16]!
    cmp x21, x20
    b.hs pv_err
    ldrb w0, [x19, x21]
    cmp w0, #0x7B
    b.eq pv_object
    cmp w0, #0x5B
    b.eq pv_array
    cmp w0, #0x22
    b.eq pv_string
    cmp w0, #0x2D
    b.eq pv_number
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pv_number
    cmp w0, #0x74
    b.eq pv_true
    cmp w0, #0x66
    b.eq pv_false
    cmp w0, #0x6E
    b.eq pv_null
    b pv_err
pv_object:
    bl parse_object
    b pv_out
pv_array:
    bl parse_array
    b pv_out
pv_string:
    bl parse_string
    b pv_out
pv_number:
    bl parse_number
    b pv_out
pv_true:
    adrp x1, lit_true
    add x1, x1, :lo12:lit_true
    mov x2, #4
    bl parse_lit
    b pv_out
pv_false:
    adrp x1, lit_false
    add x1, x1, :lo12:lit_false
    mov x2, #5
    bl parse_lit
    b pv_out
pv_null:
    adrp x1, lit_null
    add x1, x1, :lo12:lit_null
    mov x2, #4
    bl parse_lit
    b pv_out
pv_err:
    mov x0, #1
pv_out:
    ldr x30, [sp], #16
    ret

    str x30, [sp, #-16]!
    add x21, x21, #1

    cmp x21, x20

    cmp x0, #0

    bl skip_ws
    cmp x21, x20
    b.hs po_err
    ldrb w0, [x19, x21]
    cmp w0, #0x3A
    b.ne po_err
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    cmp x0, #0
    b.ne po_out
    bl skip_ws
    cmp x21, x20
    b.hs po_err
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.eq po_comma
    cmp w0, #0x7D
    b.eq po_close
    b po_err

    add x21, x21, #1
    bl skip_ws
    b po_loop
po_close:
    add x21, x21, #1
    mov x0, #0
    b po_out
po_err:
    mov x0, #1
po_out:
    ldr x30, [sp], #16
    ret

parse_array:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs pa_err
    ldrb w0, [x19, x21]
    cmp w0, #0x5D
    b.eq pa_close
pa_loop:
    bl parse_value
    cmp x0, #0
    b.ne pa_out
    bl skip_ws
    cmp x21, x20
    b.hs pa_err
    ldrb w0, [x19, x21]
    cmp w0, #0x2C
    b.eq pa_comma
    cmp w0, #0x5D
    b.eq pa_close
    b pa_err
pa_comma:
    add x21, x21, #1
    bl skip_ws
    b pa_loop
pa_close:
    add x21, x21, #1
    mov x0, #0
    b pa_out
pa_err:
    mov x0, #1
pa_out:
    ldr x30, [sp], #16
    ret
