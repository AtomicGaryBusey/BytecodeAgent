.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r13, [rip + input_buf]
    lea r14, [r13 + r12]
    xor r15, r15
    lea r8, [rip + line_ptr]
    lea r9, [rip + line_len]
parse_loop:
    cmp r13, r14
    jae parse_done
    mov rsi, r13
scan_loop:
    cmp rsi, r14
    jae scan_no_nl
    movzx eax, byte ptr [rsi]
    cmp eax, 0x0A
    je scan_found
    inc rsi
    jmp scan_loop
scan_found:
    mov rax, rsi
    sub rax, r13
    mov qword ptr [r8 + r15*8], r13
    mov qword ptr [r9 + r15*8], rax
    inc r15
    lea r13, [rsi + 1]
    jmp parse_loop
scan_no_nl:
    mov rax, r14
    sub rax, r13
    mov qword ptr [r8 + r15*8], r13
    mov qword ptr [r9 + r15*8], rax
    inc r15
    mov r13, r14
    jmp parse_loop
parse_done:
    cmp r15, 2
    jb sort_done
    mov r12, 1
sort_i:
    cmp r12, r15
    jae sort_done
    mov r13, qword ptr [r8 + r12*8]
    mov r14, qword ptr [r9 + r12*8]
    lea rbp, [r12 - 1]
sort_while:
    cmp rbp, 0
    jl sort_place
    mov rdi, qword ptr [r8 + rbp*8]
    mov rdx, qword ptr [r9 + rbp*8]
    mov rsi, r13
    mov rcx, r14
    call compare
    test eax, eax
    jle sort_place
    mov rax, qword ptr [r8 + rbp*8]
    mov qword ptr [r8 + rbp*8 + 8], rax
    mov rax, qword ptr [r9 + rbp*8]
    mov qword ptr [r9 + rbp*8 + 8], rax
    dec rbp
    jmp sort_while
sort_place:
    lea rax, [rbp + 1]
    mov qword ptr [r8 + rax*8], r13
    mov qword ptr [r9 + rax*8], r14
    inc r12
    jmp sort_i
sort_done:
    lea r12, [rip + out_buf]
    xor r13, r13
out_loop:
    cmp r13, r15
    jae out_done
    mov r14, qword ptr [r8 + r13*8]
    mov r10, qword ptr [r9 + r13*8]
    mov rdi, r12
    mov rsi, r14
    mov rcx, r10
out_copy:
    test rcx, rcx
    jz out_copy_done
    movzx eax, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz out_copy
out_copy_done:
    mov byte ptr [r12], 0x0A
    inc r12
    inc r13
    jmp out_loop
out_done:
    lea rax, [rip + out_buf]
    sub r12, rax
    mov r14, r12
    xor r13, r13
write_loop:
    test r14, r14
    jz write_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, r13
    mov rdx, r14
    syscall
    cmp rax, 0
    jle write_done
    add r13, rax
    sub r14, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall

compare:
    mov r10, rdx
    cmp r10, rcx
    jbe cmp_min_len
    mov r10, rcx
cmp_min_len:
    test r10, r10
    jz cmp_after_loop
cmp_loop:
    movzx eax, byte ptr [rdi]
    movzx r11d, byte ptr [rsi]
    cmp eax, r11d
    jne cmp_diff
    inc rdi
    inc rsi
    dec r10
    jnz cmp_loop
cmp_after_loop:
    cmp rdx, rcx
    je cmp_eq
    jb cmp_lt
    mov eax, 1
    ret
cmp_lt:
    mov eax, -1
    ret
cmp_eq:
    xor eax, eax
    ret
cmp_diff:
    cmp eax, r11d
    ja cmp_gt
    mov eax, -1
    ret
cmp_gt:
    mov eax, 1
    ret

.bss
.balign 8
input_buf:
    .space 32768
.balign 8
out_buf:
    .space 32800
.balign 8
line_ptr:
    .space 8000
.balign 8
line_len:
    .space 8000
