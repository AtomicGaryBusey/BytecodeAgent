.text
.globl _start
_start:
  xor r12, r12
read_loop:
  mov rax, 0
  xor rdi, rdi
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov rdx, 32768
  sub rdx, r12
  syscall
  cmp rax, 0
  jle read_done
  add r12, rax
  cmp r12, 32768
  jb read_loop
read_done:
  lea r13, [rip + input_buf]
  mov r14, r13
  mov r15, r13
  add r15, r12
  xor rbx, rbx
  lea r10, [rip + line_start]
  lea r11, [rip + line_len]
parse_loop:
  cmp r14, r15
  jae parse_done
  mov rsi, r14
scan_loop:
  cmp rsi, r15
  jae scan_no_nl
  movzx eax, byte ptr [rsi]
  cmp eax, 0x0A
  je scan_found
  inc rsi
  jmp scan_loop
scan_found:
  mov rax, rsi
  sub rax, r14
  mov qword ptr [r10 + rbx*8], r14
  mov qword ptr [r11 + rbx*8], rax
  inc rbx
  lea r14, [rsi + 1]
  jmp parse_loop
scan_no_nl:
  mov rax, r15
  sub rax, r14
  mov qword ptr [r10 + rbx*8], r14
  mov qword ptr [r11 + rbx*8], rax
  inc rbx
  mov r14, r15
  jmp parse_loop
parse_done:
  cmp rbx, 2
  jb sort_done
  mov r12, 1
sort_i:
  cmp r12, rbx
  jae sort_done
  mov r13, qword ptr [r10 + r12*8]
  mov r14, qword ptr [r11 + r12*8]
  lea r15, [r12 - 1]
sort_while:
  cmp r15, 0
  jl sort_place
  mov rdi, qword ptr [r10 + r15*8]
  mov rdx, qword ptr [r11 + r15*8]
  mov rsi, r13
  mov rcx, r14
  call compare_lines
  test eax, eax
  jle sort_place
  mov rax, qword ptr [r10 + r15*8]
  mov qword ptr [r10 + r15*8 + 8], rax
  mov rax, qword ptr [r11 + r15*8]
  mov qword ptr [r11 + r15*8 + 8], rax
  dec r15
  jmp sort_while
sort_place:
  lea rax, [r15 + 1]
  mov qword ptr [r10 + rax*8], r13
  mov qword ptr [r11 + rax*8], r14
  inc r12
  jmp sort_i
sort_done:
  lea r12, [rip + out_buf]
  xor r13, r13
out_loop:
  cmp r13, rbx
  jae out_done
  mov r14, qword ptr [r10 + r13*8]
  mov r15, qword ptr [r11 + r13*8]
  mov rdi, r12
  mov rsi, r14
  mov rcx, r15
out_copy:
  test rcx, rcx
  jz out_copy_done
  movzx eax, byte ptr [rsi]
  mov byte ptr [rdi], al
  inc rsi
  inc rdi
  dec rcx
  jmp out_copy
out_copy_done:
  mov byte ptr [r12], 0x0A
  inc r12
  inc r13
  jmp out_loop
out_done:
  lea rax, [rip + out_buf]
  sub r12, rax
  mov rbx, r12
  xor r13, r13
write_loop:
  cmp r13, rbx
  jae write_done
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + out_buf]
  add rsi, r13
  mov rdx, rbx
  sub rdx, r13
  syscall
  cmp rax, 0
  jle write_done
  add r13, rax
  jmp write_loop
write_done:
  mov rax, 60
  xor rdi, rdi
  syscall

compare_lines:
  mov r8, rdx
  cmp r8, rcx
  jbe cl_min_ok
  mov r8, rcx
cl_min_ok:
  test r8, r8
  jz cl_len
cl_loop:
  movzx eax, byte ptr [rdi]
  movzx r9d, byte ptr [rsi]
  cmp eax, r9d
  jne cl_diff
  inc rdi
  inc rsi
  dec r8
  jnz cl_loop
cl_len:
  cmp rdx, rcx
  je cl_ret0
  jb cl_retm1
  mov eax, 1
  ret
cl_retm1:
  mov eax, -1
  ret
cl_ret0:
  xor eax, eax
  ret
cl_diff:
  cmp eax, r9d
  ja cl_ret1
  mov eax, -1
  ret
cl_ret1:
  mov eax, 1
  ret

.section .bss
.balign 8
input_buf:
  .space 32768
out_buf:
  .space 33768
line_start:
  .space 8000
line_len:
  .space 8000
