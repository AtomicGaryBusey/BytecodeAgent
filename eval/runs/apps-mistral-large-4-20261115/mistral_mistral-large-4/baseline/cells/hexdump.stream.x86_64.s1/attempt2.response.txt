.equ INPUT_CAP, 65536
.equ BUF_SIZE, 65552

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, INPUT_CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    lea r10, [rip + hexdigits]
    xor r13, r13
row_loop:
    cmp r13, r12
    jae done_rows
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe have_n
    mov r14d, 16
have_n:
    lea r15, [rip + line_buf]
    mov rbx, r13
    mov ecx, 8
off_loop:
    mov eax, ebx
    shr eax, 28
    movzx r8d, byte ptr [r10 + rax]
    mov byte ptr [r15], r8b
    inc r15
    rol ebx, 4
    dec ecx
    jnz off_loop
    mov word ptr [r15], 0x2020
    add r15, 2
    lea rsi, [rip + input_buf]
    add rsi, r13
    xor ecx, ecx
slot_loop:
    cmp rcx, r14
    jae blank_slot
    movzx eax, byte ptr [rsi + rcx]
    mov rbx, rax
    shr rbx, 4
    movzx r8d, byte ptr [r10 + rbx]
    mov byte ptr [r15], r8b
    inc r15
    mov rbx, rax
    and ebx, 0xF
    movzx r8d, byte ptr [r10 + rbx]
    mov byte ptr [r15], r8b
    inc r15
    mov byte ptr [r15], 0x20
    inc r15
    jmp slot_next
blank_slot:
    mov word ptr [r15], 0x2020
    mov byte ptr [r15 + 2], 0x20
    add r15, 3
slot_next:
    cmp rcx, 7
    jne no_extra
    mov byte ptr [r15], 0x20
    inc r15
no_extra:
    inc rcx
    cmp rcx, 16
    jb slot_loop
    mov byte ptr [r15], 0x20
    inc r15
    mov byte ptr [r15], 0x7C
    inc r15
    xor ecx, ecx
ascii_loop:
    cmp rcx, r14
    jae ascii_done
    movzx eax, byte ptr [rsi + rcx]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7E
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov eax, 0x2E
ascii_store:
    mov byte ptr [r15], al
    inc r15
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r15], 0x7C
    inc r15
    mov byte ptr [r15], 0x0A
    inc r15
    lea rsi, [rip + line_buf]
    mov rdx, r15
    sub rdx, rsi
    mov edi, 1
    call write_all
    add r13, r14
    jmp row_loop
done_rows:
    lea r15, [rip + line_buf]
    mov rbx, r12
    mov ecx, 8
fin_off_loop:
    mov eax, ebx
    shr eax, 28
    movzx r8d, byte ptr [r10 + rax]
    mov byte ptr [r15], r8b
    inc r15
    rol ebx, 4
    dec ecx
    jnz fin_off_loop
    mov byte ptr [r15], 0x0A
    inc r15
    lea rsi, [rip + line_buf]
    mov rdx, r15
    sub rdx, rsi
    mov edi, 1
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

write_all:
    mov eax, 1
wa_loop:
    syscall
    cmp rax, rdx
    je wa_done
    add rsi, rax
    sub rdx, rax
    mov eax, 1
    jmp wa_loop
wa_done:
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero BUF_SIZE
line_buf:
    .zero 96
