.text
.globl _start
_start:
    adrp x16, inbuf
    add x16, x16, :lo12:inbuf
    adrp x17, hexdigits
    add x17, x17, :lo12:hexdigits
    adrp x9, linebuf
    add x9, x9, :lo12:linebuf
    mov x19, #0

read_loop:
    movz x2, #1, lsl #16
    cmp x19, x2
    b.eq read_done
    mov x8, #63
    mov x0, #0
    mov x1, x16
    add x1, x1, x19
    sub x2, x2, x19
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0

row_loop:
    cmp x20, x19
    b.ge rows_done
    mov x21, #16
    sub x22, x19, x20
    cmp x22, x21
    csel x21, x22, x21, le
    mov x10, x9

    mov x11, x20
    mov x12, #28
off_hex_loop:
    lsr x13, x11, x12
    and x13, x13, #0xf
    ldrb w15, [x17, x13]
    strb w15, [x10], #1
    subs x12, x12, #4
    b.pl off_hex_loop

    mov w15, #0x20
    strb w15, [x10], #1
    strb w15, [x10], #1

    mov x11, #0
hex_slot_loop:
    cmp x11, x21
    b.ge slot_missing
    add x13, x16, x20
    add x13, x13, x11
    ldrb w15, [x13]
    lsr w13, w15, #4
    and w13, w13, #0xf
    ldrb w12, [x17, x13]
    strb w12, [x10], #1
    and w13, w15, #0xf
    ldrb w12, [x17, x13]
    strb w12, [x10], #1
    mov w12, #0x20
    strb w12, [x10], #1
    b slot_end
slot_missing:
    mov w12, #0x20
    strb w12, [x10], #1
    strb w12, [x10], #1
    strb w12, [x10], #1
slot_end:
    cmp x11, #7
    b.ne no_extra
    mov w12, #0x20
    strb w12, [x10], #1
no_extra:
    add x11, x11, #1
    cmp x11, #16
    b.lt hex_slot_loop

    mov w12, #0x20
    strb w12, [x10], #1
    mov w12, #0x7c
    strb w12, [x10], #1

    mov x11, #0
char_loop:
    cmp x11, x21
    b.ge char_done
    add x13, x16, x20
    add x13, x13, x11
    ldrb w15, [x13]
    cmp w15, #0x20
    b.lt char_dot
    cmp w15, #0x7e
    b.gt char_dot
    b char_keep
char_dot:
    mov w15, #0x2e
char_keep:
    strb w15, [x10], #1
    add x11, x11, #1
    b char_loop
char_done:
    mov w12, #0x7c
    strb w12, [x10], #1
    mov w12, #0x0a
    strb w12, [x10], #1

    sub x2, x10, x9
    mov x1, x9
    mov x0, #1
    mov x8, #64
    svc #0

    add x20, x20, #16
    b row_loop

rows_done:
    mov x10, x9
    mov x11, x19
    mov x12, #28
final_hex_loop:
    lsr x13, x11, x12
    and x13, x13, #0xf
    ldrb w15, [x17, x13]
    strb w15, [x10], #1
    subs x12, x12, #4
    b.pl final_hex_loop
    mov w15, #0x0a
    strb w15, [x10], #1
    sub x2, x10, x9
    mov x1, x9
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
    b .

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.section .bss
.balign 8
inbuf:
    .zero 65536
linebuf:
    .zero 80
