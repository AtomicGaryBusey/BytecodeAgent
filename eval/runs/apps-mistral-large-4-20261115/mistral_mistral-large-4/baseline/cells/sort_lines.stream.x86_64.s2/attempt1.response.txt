.equ BUF_CAP, 32768
.equ MAX_LINES, 1000

.text
.globl _start
_start:
    xor r12, r12
    lea r13, [rip + buf]
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    add rsi, r12
    mov rdx, BUF_CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, BUF_CAP
    jb read_loop
read_done:
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + lines]
build_loop:
    cmp r14, r12
    jae build_done
    mov rax, r14
scan_loop:
    cmp rax, r12
    jae scan_end
    movzx rdx, byte ptr [r13 + rax]
    cmp dl, 10
    je scan_newline
    inc rax
    jmp scan_loop
scan_newline:
    mov rcx, rax
    sub rcx, r14
    lea rdx, [r13 + r14]
    mov rdi, r15
    shl rdi, 4
    add rdi, rbx
    mov qword ptr [rdi], rdx
    mov qword ptr [rdi + 8], rcx
    inc r15
    lea r14, [rax + 1]
    jmp build_loop
scan_end:
    mov rcx, r12
    sub rcx, r14
    lea rdx, [r13 + r14]
    mov rdi, r15
    shl rdi, 4
    add rdi, rbx
    mov qword ptr [rdi], rdx
    mov qword ptr [rdi + 8], rcx
    inc r15
    mov r14, r12
build_done:
    cmp r15, 2
    jb sort_done
    mov r12, 1
sort_outer:
    cmp r12, r15
    jae sort_done
    mov rax, r12
    shl rax, 4
    add rax, rbx
    mov r13, qword ptr [rax]
    mov r14, qword ptr [rax + 8]
    mov rbp, r12
    dec rbp
sort_inner:
    cmp rbp, 0
    jl sort_insert
    mov rax, rbp
    shl rax, 4
    add rax, rbx
    mov rdi, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    mov rdx, r13
    mov rcx, r14
    call cmp_lines
    test eax, eax
    jg sort_shift
    jmp sort_insert
sort_shift:
    mov rax, rbp
    shl rax, 4
    add rax, rbx
    mov rcx, qword ptr [rax]
    mov qword ptr [rax + 16], rcx
    mov rcx, qword ptr [rax + 8]
    mov qword ptr [rax + 24], rcx
    dec rbp
    jmp sort_inner
sort_insert:
    lea rax, [rbp + 1]
    shl rax, 4
    add rax, rbx
    mov qword ptr [rax], r13
    mov qword ptr [rax + 8], r14
    inc r12
    jmp sort_outer
sort_done:
    xor r12, r12
out_loop:
    cmp r12, r15
    jae out_done
    mov rax, r12
    shl rax, 4
    add rax, rbx
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    test rdx, rdx
    jz out_newline
    mov rax, 1
    mov rdi, 1
    syscall
out_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    inc r12
    jmp out_loop
out_done:
    mov rax, 60
    xor rdi, rdi
    syscall

cmp_lines:
    mov r8, rsi
    cmp r8, rcx
    jbe 1f
    mov r8, rcx
1:
    cmp r8, 8
    jb 2f
    mov rax, qword ptr [rdi]
    mov r9, qword ptr [rdx]
    bswap rax
    bswap r9
    cmp rax, r9
    jne 5f
    add rdi, 8
    add rdx, 8
    sub r8, 8
    jmp 1b
2:
    test r8, r8
    jz 6f
3:
    movzx eax, byte ptr [rdi]
    movzx r9d, byte ptr [rdx]
    cmp eax, r9d
    jne 5f
    inc rdi
    inc rdx
    dec r8
    jnz 3b
6:
    cmp rsi, rcx
    je 9f
    ja 7f
    jmp 8f
5:
    ja 7f
    jb 8f
    xor eax, eax
    ret
7:
    mov eax, 1
    ret
8:
    mov eax, -1
    ret
9:
    xor eax, eax
    ret

.section .rodata
nl:
    .byte 10

.section .bss
.balign 8
buf:
    .zero BUF_CAP
lines:
    .zero MAX_LINES*16
