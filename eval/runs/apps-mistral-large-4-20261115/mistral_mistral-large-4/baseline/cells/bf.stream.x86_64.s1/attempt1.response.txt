.equ INPUT_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MATCH_SIZE, 262144
.equ BSTACK_SIZE, 262144

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    cmp rbx, INPUT_BUF_SIZE
    jae read_done
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, INPUT_BUF_SIZE
    sub rdx, rbx
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    xor ecx, ecx
scan_bang:
    cmp rcx, rbx
    jae no_bang
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp scan_bang
found_bang:
    mov r13, rcx
    lea r14, [rsi + rcx + 1]
    mov r15, rbx
    sub r15, rcx
    dec r15
    jmp split_done
no_bang:
    mov r13, rbx
    lea r14, [rsi + rbx]
    xor r15d, r15d
split_done:
    lea rsi, [rip + input_buf]
    lea rdi, [rip + match]
    lea r8, [rip + bstack]
    xor ecx, ecx
    xor edx, edx
validate_loop:
    cmp rcx, r13
    jae validate_end
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5B
    je bracket_open
    cmp al, 0x5D
    je bracket_close
    jmp validate_next
bracket_open:
    mov dword ptr [r8], ecx
    add r8, 4
    inc rdx
    jmp validate_next
bracket_close:
    test rdx, rdx
    jz unbalanced
    sub r8, 4
    mov r9d, dword ptr [r8]
    mov eax, ecx
    mov dword ptr [rdi + r9*4], eax
    mov eax, r9d
    mov dword ptr [rdi + rcx*4], eax
    dec rdx
validate_next:
    inc rcx
    jmp validate_loop
validate_end:
    test rdx, rdx
    jnz unbalanced
    mov r8, r13
    lea rsi, [rip + input_buf]
    lea rdi, [rip + match]
    lea r12, [rip + tape]
    xor r13d, r13d
    xor ebx, ebx
    xor ebp, ebp
run_loop:
    cmp r13, r8
    jae run_done
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_out
    cmp al, 0x2C
    je cmd_in
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    inc r13
    jmp run_loop
cmd_gt:
    inc r12
    inc r13
    jmp run_loop
cmd_lt:
    dec r12
    inc r13
    jmp run_loop
cmd_plus:
    inc byte ptr [r12]
    inc r13
    jmp run_loop
cmd_minus:
    dec byte ptr [r12]
    inc r13
    jmp run_loop
cmd_out:
    lea rdx, [rip + out_buf]
    mov al, byte ptr [r12]
    mov byte ptr [rdx + rbp], al
    inc rbp
    inc r13
    jmp run_loop
cmd_in:
    cmp rbx, r15
    jae in_eof
    movzx eax, byte ptr [r14 + rbx]
    inc rbx
    jmp in_store
in_eof:
    xor eax, eax
in_store:
    mov byte ptr [r12], al
    inc r13
    jmp run_loop
cmd_open:
    cmp byte ptr [r12], 0
    jne open_nonzero
    mov eax, dword ptr [rdi + r13*4]
    inc rax
    mov r13, rax
    jmp run_loop
open_nonzero:
    inc r13
    jmp run_loop
cmd_close:
    cmp byte ptr [r12], 0
    je close_zero
    mov eax, dword ptr [rdi + r13*4]
    inc rax
    mov r13, rax
    jmp run_loop
close_zero:
    inc r13
    jmp run_loop
run_done:
    test rbp, rbp
    jz exit_ok
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov rdx, rbp
    mov eax, 1
    syscall
exit_ok:
    xor edi, edi
    mov eax, 60
    syscall
unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.balign 16
input_buf:
    .zero INPUT_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE
tape:
    .zero TAPE_SIZE
match:
    .zero MATCH_SIZE
bstack:
    .zero BSTACK_SIZE
