.equ BUF_SIZE, 32768
.equ ARR_SIZE, 4000

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov edx, BUF_SIZE
    sub edx, r12d
    test edx, edx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [buf + r12]
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor eax, eax
    xor ebx, ebx
    xor r13d, r13d
parse_loop:
    cmp eax, r12d
    jae parse_final
    movzx ecx, byte ptr [buf + rax]
    cmp ecx, 10
    je parse_nl
    inc eax
    jmp parse_loop
parse_nl:
    mov edx, eax
    sub edx, ebx
    mov dword ptr [offs + r13*4], ebx
    mov dword ptr [lens + r13*4], edx
    inc r13d
    lea ebx, [rax + 1]
    inc eax
    jmp parse_loop
parse_final:
    cmp ebx, r12d
    jae parse_done
    mov edx, r12d
    sub edx, ebx
    mov dword ptr [offs + r13*4], ebx
    mov dword ptr [lens + r13*4], edx
    inc r13d
parse_done:
    mov r15d, r13d
    xor ecx, ecx
sort_init_loop:
    cmp ecx, r15d
    jae sort_init_done
    mov dword ptr [arr + rcx*4], ecx
    inc ecx
    jmp sort_init_loop
sort_init_done:
    mov ebx, 1
sort_outer:
    cmp ebx, r15d
    jae sort_done
    xor r12d, r12d
sort_seg_loop:
    cmp r12d, r15d
    jae sort_seg_done
    mov r13d, r12d
    add r13d, ebx
    cmp r13d, r15d
    jbe sort_mid_ok
    mov r13d, r15d
sort_mid_ok:
    mov r14d, r12d
    add r14d, ebx
    add r14d, ebx
    cmp r14d, r15d
    jbe sort_right_ok
    mov r14d, r15d
sort_right_ok:
    mov r8d, r12d
    mov r9d, r13d
    mov r10d, r12d
sort_merge_loop:
    cmp r8d, r13d
    jae sort_first_exhausted
    cmp r9d, r14d
    jae sort_second_exhausted
    mov edi, dword ptr [arr + r8*4]
    mov esi, dword ptr [arr + r9*4]
    call compare
    cmp eax, 0
    jle sort_take_first
    jmp sort_take_second
sort_take_first:
    mov eax, dword ptr [arr + r8*4]
    mov dword ptr [tmp + r10*4], eax
    inc r8d
    jmp sort_after_store
sort_take_second:
    mov eax, dword ptr [arr + r9*4]
    mov dword ptr [tmp + r10*4], eax
    inc r9d
sort_after_store:
    inc r10d
    jmp sort_merge_loop
sort_first_exhausted:
    cmp r9d, r14d
    jae sort_merge_end
    mov eax, dword ptr [arr + r9*4]
    mov dword ptr [tmp + r10*4], eax
    inc r9d
    inc r10d
    jmp sort_first_exhausted
sort_second_exhausted:
    cmp r8d, r13d
    jae sort_merge_end
    mov eax, dword ptr [arr + r8*4]
    mov dword ptr [tmp + r10*4], eax
    inc r8d
    inc r10d
    jmp sort_second_exhausted
sort_merge_end:
    mov r12d, r14d
    jmp sort_seg_loop
sort_seg_done:
    xor ecx, ecx
sort_copy_loop:
    cmp ecx, r15d
    jae sort_copy_done
    mov eax, dword ptr [tmp + rcx*4]
    mov dword ptr [arr + rcx*4], eax
    inc ecx
    jmp sort_copy_loop
sort_copy_done:
    shl ebx, 1
    jmp sort_outer
sort_done:
    xor ebx, ebx
out_loop:
    cmp ebx, r15d
    jae out_done
    mov eax, dword ptr [arr + rbx*4]
    mov ecx, dword ptr [offs + rax*4]
    mov edx, dword ptr [lens + rax*4]
    lea rsi, [buf + rcx]
    mov edi, 1
    call write_all
    lea rsi, [rip + nl]
    mov edi, 1
    mov edx, 1
    call write_all
    inc ebx
    jmp out_loop
out_done:
    xor edi, edi
    mov eax, 60
    syscall

compare:
    push rbx
    push rcx
    push rdx
    push rsi
    push rdi
    push r8
    push r9
    push r10
    push r11
    mov eax, dword ptr [lens + rdi*4]
    mov edx, dword ptr [lens + rsi*4]
    mov r8d, dword ptr [offs + rdi*4]
    mov r9d, dword ptr [offs + rsi*4]
    lea r8, [buf + r8]
    lea r9, [buf + r9]
    mov ebx, eax
    cmp edx, ebx
    ja compare_min_ok
    mov ebx, edx
compare_min_ok:
    xor ecx, ecx
compare_byte_loop:
    cmp ecx, ebx
    jae compare_len
    movzx r10d, byte ptr [r8 + rcx]
    movzx r11d, byte ptr [r9 + rcx]
    cmp r10d, r11d
    jb compare_less
    ja compare_greater
    inc ecx
    jmp compare_byte_loop
compare_len:
    cmp eax, edx
    jb compare_less
    ja compare_greater
    xor eax, eax
    jmp compare_ret
compare_less:
    mov eax, 0xFFFFFFFF
    jmp compare_ret
compare_greater:
    mov eax, 1
compare_ret:
    pop r11
    pop r10
    pop r9
    pop r8
    pop rdi
    pop rsi
    pop rdx
    pop rcx
    pop rbx
    ret

write_all:
    test rdx, rdx
    jz write_done
write_loop:
    mov eax, 1
    syscall
    cmp rax, 0
    jle write_error
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_done:
    xor eax, eax
    ret
write_error:
    mov edi, 1
    mov eax, 231
    syscall

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 8
buf:
    .zero BUF_SIZE
offs:
    .zero ARR_SIZE
lens:
    .zero ARR_SIZE
arr:
    .zero ARR_SIZE
tmp:
    .zero ARR_SIZE
