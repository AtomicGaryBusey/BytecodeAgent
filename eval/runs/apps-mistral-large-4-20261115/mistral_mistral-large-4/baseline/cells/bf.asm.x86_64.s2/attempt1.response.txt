.equ INPUT_MAX, 65536

.text
.globl _start
_start:
    lea rsi, [rip + input_buf]
    xor r12, r12
    xor edi, edi
read_loop:
    mov rdx, INPUT_MAX
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add rsi, rax
    jmp read_loop
read_done:
    xor rcx, rcx
scan_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [input_buf + rcx], 0x21
    je found_bang
    inc rcx
    jmp scan_bang
found_bang:
    mov r13, rcx
    lea r14, [rcx + 1]
    jmp have_parts
no_bang:
    mov r13, r12
    mov r14, r12
have_parts:
    xor rbx, rbx
    xor rbp, rbp
    xor r8, r8
pre_loop:
    cmp rbp, r13
    jae pre_done
    movzx eax, byte ptr [input_buf + rbp]
    cmp al, 0x3E
    je pre_add
    cmp al, 0x3C
    je pre_add
    cmp al, 0x2B
    je pre_add
    cmp al, 0x2D
    je pre_add
    cmp al, 0x2E
    je pre_add
    cmp al, 0x2C
    je pre_add
    cmp al, 0x5B
    je pre_open
    cmp al, 0x5D
    je pre_close
    inc rbp
    jmp pre_loop
pre_add:
    mov byte ptr [cmd_code + rbx], al
    inc rbx
    inc rbp
    jmp pre_loop
pre_open:
    mov byte ptr [cmd_code + rbx], al
    mov dword ptr [bracket_stack + r8*4], ebx
    inc r8
    inc rbx
    inc rbp
    jmp pre_loop
pre_close:
    test r8, r8
    jz unbalanced
    dec r8
    mov ecx, dword ptr [bracket_stack + r8*4]
    mov byte ptr [cmd_code + rbx], al
    mov dword ptr [cmd_jump + rbx*4], ecx
    mov dword ptr [cmd_jump + rcx*4], ebx
    inc rbx
    inc rbp
    jmp pre_loop
pre_done:
    test r8, r8
    jnz unbalanced
    mov r13, r12
    mov r12, rbx
    xor rbx, rbx
    xor rbp, rbp
    mov r15, r14
    mov r14, r13
run_loop:
    cmp rbx, r12
    jae run_done
    movzx r8d, byte ptr [cmd_code + rbx]
    cmp r8b, 0x3E
    je do_gt
    cmp r8b, 0x3C
    je do_lt
    cmp r8b, 0x2B
    je do_plus
    cmp r8b, 0x2D
    je do_minus
    cmp r8b, 0x2E
    je do_dot
    cmp r8b, 0x2C
    je do_comma
    cmp r8b, 0x5B
    je do_open
    cmp r8b, 0x5D
    je do_close
    jmp run_done
do_gt:
    inc rbp
    inc rbx
    jmp run_loop
do_lt:
    dec rbp
    inc rbx
    jmp run_loop
do_plus:
    inc byte ptr [tape + rbp]
    inc rbx
    jmp run_loop
do_minus:
    dec byte ptr [tape + rbp]
    inc rbx
    jmp run_loop
do_dot:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + tape]
    add rsi, rbp
    mov edx, 1
    syscall
    inc rbx
    jmp run_loop
do_comma:
    cmp r15, r14
    jae comma_zero
    movzx eax, byte ptr [input_buf + r15]
    inc r15
    mov byte ptr [tape + rbp], al
    inc rbx
    jmp run_loop
comma_zero:
    mov byte ptr [tape + rbp], 0
    inc rbx
    jmp run_loop
do_open:
    cmp byte ptr [tape + rbp], 0
    jne open_nonzero
    mov eax, dword ptr [cmd_jump + rbx*4]
    lea rbx, [rax + 1]
    jmp run_loop
open_nonzero:
    inc rbx
    jmp run_loop
do_close:
    cmp byte ptr [tape + rbp], 0
    je close_zero
    mov eax, dword ptr [cmd_jump + rbx*4]
    lea rbx, [rax + 1]
    jmp run_loop
close_zero:
    inc rbx
    jmp run_loop
run_done:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 8
input_buf:
    .zero 65536
tape:
    .zero 30000
cmd_code:
    .zero 65536
cmd_jump:
    .zero 262144
bracket_stack:
    .zero 262144
