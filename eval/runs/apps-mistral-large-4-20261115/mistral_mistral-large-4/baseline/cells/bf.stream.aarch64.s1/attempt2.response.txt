.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT_GROUP, 94
.equ BANG, 33
.equ OP_GT, 62
.equ OP_LT, 60
.equ OP_PLUS, 43
.equ OP_MINUS, 45
.equ OP_DOT, 46
.equ OP_COMMA, 44
.equ OP_OPEN, 91
.equ OP_CLOSE, 93

.text
.globl _start
_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0
read_loop:
  mov x0, #0
  mov x1, x19
  add x1, x1, x20
  mov x2, #65536
  sub x2, x2, x20
  mov x8, #SYS_READ
  svc #0
  cmp x0, #0
  b.le read_done
  add x20, x20, x0
  cmp x20, #65536
  b.eq read_done
  b read_loop
read_done:
  mov x21, #0
scan_bang:
  cmp x21, x20
  b.eq no_bang
  ldrb w22, [x19, x21]
  cmp w22, #BANG
  b.eq bang_found
  add x21, x21, #1
  b scan_bang
bang_found:
  mov x23, x21
  add x24, x19, x21
  add x24, x24, #1
  sub x25, x20, x21
  sub x25, x25, #1
  b after_parse
no_bang:
  mov x23, x20
  add x24, x19, x20
  mov x25, #0
after_parse:
  adrp x26, match_tab
  add x26, x26, :lo12:match_tab
  adrp x27, open_stack
  add x27, x27, :lo12:open_stack
  mov x28, #0
  mov x5, #0
bracket_loop:
  cmp x5, x23
  b.eq bracket_done
  ldrb w6, [x19, x5]
  cmp w6, #OP_OPEN
  b.eq br_open
  cmp w6, #OP_CLOSE
  b.eq br_close
  b br_next
br_open:
  str w5, [x27, x28, lsl #2]
  add x28, x28, #1
  b br_next
br_close:
  cmp x28, #0
  b.eq unbalanced
  sub x28, x28, #1
  ldr w7, [x27, x28, lsl #2]
  add x8, x5, #1
  str w8, [x26, x5, lsl #2]
  add x8, x7, #1
  str w8, [x26, x7, lsl #2]
  b br_next
br_next:
  add x5, x5, #1
  b bracket_loop
bracket_done:
  cmp x28, #0
  b.ne unbalanced
  b run
unbalanced:
  mov x0, #1
  mov x8, #SYS_EXIT_GROUP
  svc #0
run:
  adrp x14, tape
  add x14, x14, :lo12:tape
  mov x9, x14
  adrp x13, out_buf
  add x13, x13, :lo12:out_buf
  mov x12, #0
  mov x5, #0
exec_loop:
  cmp x5, x23
  b.eq finish
  ldrb w6, [x19, x5]
  cmp w6, #OP_GT
  b.eq cmd_gt
  cmp w6, #OP_LT
  b.eq cmd_lt
  cmp w6, #OP_PLUS
  b.eq cmd_plus
  cmp w6, #OP_MINUS
  b.eq cmd_minus
  cmp w6, #OP_DOT
  b.eq cmd_dot
  cmp w6, #OP_COMMA
  b.eq cmd_comma
  cmp w6, #OP_OPEN
  b.eq cmd_open
  cmp w6, #OP_CLOSE
  b.eq cmd_close
  b next_pc
cmd_gt:
  add x9, x9, #1
  b next_pc
cmd_lt:
  sub x9, x9, #1
  b next_pc
cmd_plus:
  ldrb w7, [x9]
  add w7, w7, #1
  strb w7, [x9]
  b next_pc
cmd_minus:
  ldrb w7, [x9]
  sub w7, w7, #1
  strb w7, [x9]
  b next_pc
cmd_dot:
  ldrb w7, [x9]
  strb w7, [x13, x12]
  add x12, x12, #1
  cmp x12, #65536
  b.lt dot_done
  mov x1, x13
  mov x2, x12
dot_write_loop:
  cmp x2, #0
  b.eq dot_write_done
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  add x1, x1, x0
  sub x2, x2, x0
  b dot_write_loop
dot_write_done:
  mov x12, #0
dot_done:
  b next_pc
cmd_comma:
  cmp x25, #0
  b.eq comma_zero
  ldrb w7, [x24]
  add x24, x24, #1
  sub x25, x25, #1
  strb w7, [x9]
  b next_pc
comma_zero:
  mov w7, #0
  strb w7, [x9]
  b next_pc
cmd_open:
  ldrb w7, [x9]
  cmp w7, #0
  b.eq open_zero
  b next_pc
open_zero:
  ldr w7, [x26, x5, lsl #2]
  sub x5, x7, #1
  b next_pc
cmd_close:
  ldrb w7, [x9]
  cmp w7, #0
  b.eq next_pc
  ldr w7, [x26, x5, lsl #2]
  sub x5, x7, #1
  b next_pc
next_pc:
  add x5, x5, #1
  b exec_loop
finish:
  cmp x12, #0
  b.eq exit_ok
  mov x1, x13
  mov x2, x12
fin_write_loop:
  cmp x2, #0
  b.eq exit_ok
  mov x0, #1
  mov x8, #SYS_WRITE
  svc #0
  add x1, x1, x0
  sub x2, x2, x0
  b fin_write_loop
exit_ok:
  mov x0, #0
  mov x8, #SYS_EXIT_GROUP
  svc #0
