.globl _start

.bss
.balign 16
buf:
    .zero 4096
out:
    .zero 64

.text
_start:
    xor ebx, ebx
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d

read_loop:
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    mov rcx, rax
    lea rsi, [rip + buf]

scan_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r13
    cmp al, 0x0A
    jne 1f
    inc rbx
1:
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    jb 3f
    cmp al, 0x0D
    jbe 2f
    jmp 3f
2:
    xor r14d, r14d
    jmp scan_loop
3:
    test r14d, r14d
    jnz scan_loop
    inc r12
    mov r14d, 1
    jmp scan_loop

input_done:
    lea r8, [rip + out]
    add r8, 64
    dec r8
    mov byte ptr [r8], 0x0A
    mov rax, r13
    call emit_u64
    dec r8
    mov byte ptr [r8], 0x20
    mov rax, r12
    call emit_u64
    dec r8
    mov byte ptr [r8], 0x20
    mov rax, rbx
    call emit_u64
    mov rdi, 1
    mov rsi, r8
    lea rdx, [rip + out]
    add rdx, 64
    sub rdx, r8
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

emit_u64:
    test rax, rax
    jnz 1f
    dec r8
    mov byte ptr [r8], 0x30
    ret
1:
    mov ecx, 10
2:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz 2b
    ret
