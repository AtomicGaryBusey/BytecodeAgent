.globl _start
.equ BUFSIZE, 65536

.bss
.balign 16
buf:
    .zero 65537

.text
_start:
    xor r12d, r12d
read_loop:
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, BUFSIZE
    sub edx, r12d
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit_prog
    lea rbx, [rip + buf]
    mov r13, r12
    mov al, byte ptr [rbx + r12 - 1]
    cmp al, 0x0A
    je no_append
    mov byte ptr [rbx + r12], 0x0A
    inc r13
no_append:
    mov rsi, rbx
    lea r15, [rbx + r12]
line_loop:
    cmp rsi, r15
    jae lines_done
    mov rdi, rsi
scan_loop:
    cmp rdi, r15
    jae no_newline
    mov al, byte ptr [rdi]
    cmp al, 0x0A
    je found_newline
    inc rdi
    jmp scan_loop
found_newline:
    mov r8, rsi
    mov r9, rdi
    jmp rev_loop
no_newline:
    mov r8, rsi
    mov r9, r15
rev_loop:
    cmp r8, r9
    jae rev_done
    lea r10, [r9 - 1]
    mov al, byte ptr [r8]
    mov cl, byte ptr [r10]
    mov byte ptr [r8], cl
    mov byte ptr [r10], al
    inc r8
    dec r9
    jmp rev_loop
rev_done:
    lea rsi, [rdi + 1]
    jmp line_loop
lines_done:
    mov edi, 1
    mov rsi, rbx
    mov rdx, r13
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
exit_prog:
    xor edi, edi
    mov eax, 60
    syscall
