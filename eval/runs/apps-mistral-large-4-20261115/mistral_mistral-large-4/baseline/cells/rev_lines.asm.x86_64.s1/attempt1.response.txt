.text
.equ READ_MAX, 65536
.equ BUF_SIZE, 65537

.globl _start
_start:
    xor r12, r12
read_loop:
    cmp r12, READ_MAX
    jae read_done
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, READ_MAX
    sub rdx, r12
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz no_append
    lea rdx, [rip + inbuf]
    mov cl, byte ptr [rdx + r12 - 1]
    cmp cl, 10
    je no_append
    mov byte ptr [rdx + r12], 10
    inc r12
no_append:
    lea r15, [rip + inbuf]
    lea r8, [rip + outbuf]
    mov rsi, r8
    xor r13, r13
line_loop:
    cmp r13, r12
    jae write_out
    mov rbx, r13
scan_loop:
    mov cl, byte ptr [r15 + rbx]
    cmp cl, 10
    je scan_done
    inc rbx
    jmp scan_loop
scan_done:
    mov r9, rbx
    sub r9, r13
    xor ecx, ecx
copy_loop:
    cmp rcx, r9
    jae copy_done
    mov rdx, rbx
    sub rdx, 1
    sub rdx, rcx
    mov al, byte ptr [r15 + rdx]
    mov byte ptr [rsi + rcx], al
    inc rcx
    jmp copy_loop
copy_done:
    mov byte ptr [rsi + r9], 10
    lea rax, [r9 + 1]
    add rsi, rax
    lea r13, [rbx + 1]
    jmp line_loop
write_out:
    mov rdx, rsi
    sub rdx, r8
    mov rsi, r8
write_loop:
    test rdx, rdx
    jz exit_prog
    mov edi, 1
    mov eax, 1
    syscall
    cmp rax, 0
    jle exit_prog
    cmp rax, rdx
    jae exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop
exit_prog:
    xor edi, edi
    mov eax, 60
    syscall

.bss
.balign 16
inbuf:
    .zero BUF_SIZE
outbuf:
    .zero BUF_SIZE
