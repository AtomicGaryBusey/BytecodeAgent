.globl _start
.equ IN_BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 70000
.equ LINE_LEN, 76

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
in_buf:
    .zero IN_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE

.text
_start:
    xor r12, r12
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + in_buf]
    add rsi, r12
    mov edx, IN_BUF_SIZE
    sub edx, r12d
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, IN_BUF_SIZE
    jb read_loop
read_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + alphabet]
encode_loop:
    cmp r13, r12
    jae encode_done
    mov rax, r12
    sub rax, r13
    cmp rax, 3
    jb final_group
    lea rsi, [rip + in_buf]
    movzx eax, byte ptr [rsi + r13*1]
    movzx ecx, byte ptr [rsi + r13*1 + 1]
    movzx edx, byte ptr [rsi + r13*1 + 2]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ecx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ecx
    and r10d, 15
    shl r10d, 2
    mov r11d, edx
    shr r11d, 6
    or r10d, r11d
    and edx, 63
    lea rsi, [rip + out_buf]
    movzx eax, byte ptr [rbx + r8*1]
    mov byte ptr [rsi + r14*1], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [rsi + r14*1], 10
    inc r14
    xor r15, r15
1:
    movzx eax, byte ptr [rbx + r9*1]
    mov byte ptr [rsi + r14*1], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [rsi + r14*1], 10
    inc r14
    xor r15, r15
1:
    movzx eax, byte ptr [rbx + r10*1]
    mov byte ptr [rsi + r14*1], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [rsi + r14*1], 10
    inc r14
    xor r15, r15

    movzx eax, byte ptr [rbx + rdx*1]
    mov byte ptr [rsi + r14*1], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [rsi + r14*1], 10
    inc r14
    xor r15, r15
1:
    add r13, 3
    jmp encode_loop

final_group:
    cmp rax, 1
    je final_one
    lea rsi, [rip + in_buf]
    movzx eax, byte ptr [rsi + r13*1]
    movzx ecx, byte ptr [rsi + r13*1 + 1]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ecx
    shr r10d, 4
    or r9d, r10d
    mov r10d, ecx
    and r10d, 15
    shl r10d, 2
    lea rsi, [rip + out_buf]
    movzx eax, byte ptr [rbx + r8*1]
    mov byte ptr [rsi + r14*1], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [rsi + r14*1], 10
    inc r14
    xor r15, r15

    movzx eax, byte ptr [rbx + r9*1]
    mov byte ptr [rsi + r14*1], al

    mov byte ptr [rsi + r14*1], 10
    inc r14
    xor r15, r15

    jmp encode_done

final_one:
    lea rsi, [rip + in_buf]
    movzx eax, byte ptr [rsi + r13*1]
    mov r8d, eax
    shr r8d, 2
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    lea rsi, [rip + out_buf]
    movzx eax, byte ptr [rbx + r8*1]
    mov byte ptr [rsi + r14*1], al
    inc r14
    inc r15
    cmp r15, LINE_LEN
    jne 1f
    mov byte ptr [rsi + r14*1], 10

    test r15, r15
    jz write_output
    lea rsi, [rip + out_buf]
    mov byte ptr [rsi + r14*1], 10
    inc r14

    xor r13, r13
    test r14, r14
    jz exit_prog
write_loop:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    add rsi, r13
    mov rdx, r14
    sub rdx, r13
    syscall
    cmp rax, 0
    jle exit_prog
    add r13, rax
    cmp r13, r14
    jb write_loop

    mov eax, 231
    xor edi, edi
    syscall
