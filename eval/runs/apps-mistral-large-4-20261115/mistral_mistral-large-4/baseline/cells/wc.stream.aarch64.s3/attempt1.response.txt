.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536
.equ OUT_SIZE, 64

.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #BUF_SIZE
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.eq read_done
    add x21, x21, x0
    mov x24, x23
    add x25, x23, x0

proc_loop:
    cmp x24, x25
    b.eq proc_done
    ldrb w26, [x24], #1
    cmp w26, #0x0A
    b.eq byte_newline
    cmp w26, #0x20
    b.eq byte_ws
    cmp w26, #0x09
    b.eq byte_ws
    cmp w26, #0x0B
    b.eq byte_ws
    cmp w26, #0x0C
    b.eq byte_ws
    cmp w26, #0x0D
    b.eq byte_ws
    cmp x22, #0
    b.ne proc_loop
    add x20, x20, #1
    mov x22, #1
    b proc_loop

byte_newline:
    add x19, x19, #1
    mov x22, #0
    b proc_loop

byte_ws:
    mov x22, #0
    b proc_loop

proc_done:
    b read_loop

read_done:
    adrp x0, outbuf
    add x0, x0, :lo12:outbuf
    add x1, x0, #OUT_SIZE

    mov w2, #10
    sub x1, x1, #1
    strb w2, [x1]

    mov x0, x21
    bl emit_num

    mov w2, #32
    sub x1, x1, #1
    strb w2, [x1]

    mov x0, x20
    bl emit_num

    mov w2, #32
    sub x1, x1, #1
    strb w2, [x1]

    mov x0, x19
    bl emit_num

    adrp x2, outbuf
    add x2, x2, :lo12:outbuf
    add x2, x2, #OUT_SIZE
    sub x2, x2, x1
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

emit_num:
    cmp x0, #0
    b.ne 1f
    mov w2, #48
    sub x1, x1, #1
    strb w2, [x1]
    ret
1:
    mov x9, #10
2:
    udiv x2, x0, x9
    mul x3, x2, x9
    sub x3, x0, x3
    add x3, x3, #48
    sub x1, x1, #1
    strb w3, [x1]
    mov x0, x2
    cmp x0, #0
    b.ne 2b
    ret

.bss
.balign 8
buf:
    .zero BUF_SIZE
.balign 8
outbuf:
    .zero OUT_SIZE
