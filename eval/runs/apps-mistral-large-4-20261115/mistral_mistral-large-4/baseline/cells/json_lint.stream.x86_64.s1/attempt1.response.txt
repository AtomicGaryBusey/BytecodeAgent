.section .rodata
ok_msg: .ascii "ok\n"
ok_end:
.equ ok_len, ok_end - ok_msg
err_msg: .ascii "error at "
err_end:
.equ err_len, err_end - err_msg
newline: .ascii "\n"
newline_end:
.equ newline_len, newline_end - newline
kw_true: .ascii "true"
kw_false: .ascii "false"
kw_null: .ascii "null"

.bss
.align 8
buf: .zero 65536
len: .zero 8
pos: .zero 8
num_buf: .zero 32

.text
.globl _start
_start:
    xor rbx, rbx
    lea rsi, [rip + buf]
.Lread_loop:
    mov rdi, 0
    mov rdx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz .Lread_done
    xor eax, eax
    syscall
    test rax, rax
    jz .Lread_done
    add rbx, rax
    add rsi, rax
    cmp rbx, 65536
    jae .Lread_done
    jmp .Lread_loop
.Lread_done:
    mov qword ptr [rip + len], rbx
    xor eax, eax
    mov qword ptr [rip + pos], rax
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jae .Lincomplete
    call parse_value
    cmp rax, -1
    jne .Lerror_exit
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jne .Ltrailing
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    call write_all
    mov rax, 60
    xor rdi, rdi
    syscall
.Lincomplete:
    mov rax, qword ptr [rip + len]
    jmp .Lerror_exit
.Ltrailing:
    mov rax, rbx
.Lerror_exit:
    mov r12, rax
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, err_len
    call write_all
    mov rax, r12
    call print_number
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, newline_len
    call write_all
    mov rax, 60
    mov rdi, 1
    syscall

write_all:
.Lwa_loop:
    test rdx, rdx
    jz .Lwa_done
    mov rax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp .Lwa_loop
.Lwa_done:
    ret

print_number:
    mov rbx, 10
    lea rsi, [rip + num_buf + 31]
.Lpn_loop:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .Lpn_loop
    lea rdx, [rip + num_buf + 31]
    sub rdx, rsi
    mov rdi, 1
    call write_all
    ret

skip_ws:
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rsi, [rip + buf]
.Lws_loop:
    cmp rbx, rdx
    jae .Lws_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    je .Lws_adv
    cmp al, 0x09
    je .Lws_adv
    cmp al, 0x0A
    je .Lws_adv
    cmp al, 0x0D
    je .Lws_adv
    jmp .Lws_done
.Lws_adv:
    inc rbx
    jmp .Lws_loop
.Lws_done:
    mov qword ptr [rip + pos], rbx
    ret

parse_value:
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rsi, [rip + buf]
    cmp rbx, rdx
    jae .Lpv_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb .Lpv_bad
    cmp al, '9'
    ja .Lpv_bad
    jmp parse_number
.Lpv_bad:
    mov rax, rbx
    ret
.Lpv_eof:
    mov rax, rdx
    ret

parse_true:
    lea r8, [rip + kw_true]
    mov r9, 4
    jmp parse_keyword
parse_false:
    lea r8, [rip + kw_false]
    mov r9, 5
    jmp parse_keyword
parse_null:
    lea r8, [rip + kw_null]
    mov r9, 4
    jmp parse_keyword

parse_keyword:
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rsi, [rip + buf]
    xor rcx, rcx
.Lkw_loop:
    cmp rcx, r9
    jae .Lkw_done
    cmp rbx, rdx
    jae .Lkw_eof
    movzx eax, byte ptr [rsi + rbx]
    movzx r10d, byte ptr [r8 + rcx]
    cmp al, r10b
    jne .Lkw_bad
    inc rbx
    inc rcx
    jmp .Lkw_loop
.Lkw_done:
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Lkw_bad:
    mov rax, rbx
    ret
.Lkw_eof:
    mov rax, rdx
    ret

parse_number:
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rsi, [rip + buf]
    cmp rbx, rdx
    jae .Lnum_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '-'
    jne .Lnum_int
    inc rbx
.Lnum_int:
    cmp rbx, rdx
    jae .Lnum_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    je .Lnum_zero
    cmp al, '1'
    jb .Lnum_bad
    cmp al, '9'
    ja .Lnum_bad
    inc rbx
.Lnum_digits1:
    cmp rbx, rdx
    jae .Lnum_frac
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb .Lnum_frac
    cmp al, '9'
    ja .Lnum_frac
    inc rbx
    jmp .Lnum_digits1
.Lnum_zero:
    inc rbx
.Lnum_frac:
    cmp rbx, rdx
    jae .Lnum_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '.'
    jne .Lnum_exp
    inc rbx
    cmp rbx, rdx
    jae .Lnum_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb .Lnum_bad
    cmp al, '9'
    ja .Lnum_bad
.Lnum_frac_digits:
    inc rbx
    cmp rbx, rdx
    jae .Lnum_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb .Lnum_exp
    cmp al, '9'
    ja .Lnum_exp
    jmp .Lnum_frac_digits
.Lnum_exp:
    cmp rbx, rdx
    jae .Lnum_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 'e'
    je .Lnum_exp_e
    cmp al, 'E'
    jne .Lnum_done
.Lnum_exp_e:
    inc rbx
    cmp rbx, rdx
    jae .Lnum_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '+'
    je .Lnum_exp_sign
    cmp al, '-'
    je .Lnum_exp_sign
    jmp .Lnum_exp_digits
.Lnum_exp_sign:
    inc rbx
    cmp rbx, rdx
    jae .Lnum_eof
.Lnum_exp_digits:
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb .Lnum_bad
    cmp al, '9'
    ja .Lnum_bad
    inc rbx
.Lnum_exp_digits_loop:
    cmp rbx, rdx
    jae .Lnum_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb .Lnum_done
    cmp al, '9'
    ja .Lnum_done
    inc rbx
    jmp .Lnum_exp_digits_loop
.Lnum_done:
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Lnum_bad:
    mov rax, rbx
    ret
.Lnum_eof:
    mov rax, rdx
    ret

parse_string:
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    lea rsi, [rip + buf]
    cmp rbx, rdx
    jae .Lstr_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '"'
    jne .Lstr_bad
    inc rbx
.Lstr_loop:
    cmp rbx, rdx
    jae .Lstr_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '"'
    je .Lstr_end
    cmp al, 0x5C
    je .Lstr_escape
    cmp al, 0x20
    jb .Lstr_bad
    inc rbx
    jmp .Lstr_loop
.Lstr_end:
    inc rbx
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Lstr_escape:
    inc rbx
    cmp rbx, rdx
    jae .Lstr_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '"'
    je .Lstr_esc_ok
    cmp al, 0x5C
    je .Lstr_esc_ok
    cmp al, '/'
    je .Lstr_esc_ok
    cmp al, 'b'
    je .Lstr_esc_ok
    cmp al, 'f'
    je .Lstr_esc_ok
    cmp al, 'n'
    je .Lstr_esc_ok
    cmp al, 'r'
    je .Lstr_esc_ok
    cmp al, 't'
    je .Lstr_esc_ok
    cmp al, 'u'
    je .Lstr_u
    jmp .Lstr_bad
.Lstr_esc_ok:
    inc rbx
    jmp .Lstr_loop
.Lstr_u:
    inc rbx
    mov rcx, 4
.Lstr_hex_loop:
    cmp rbx, rdx
    jae .Lstr_eof
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb .Lstr_bad
    cmp al, '9'
    jbe .Lstr_hex_next
    cmp al, 'a'
    jb .Lstr_check_upper
    cmp al, 'f'
    jbe .Lstr_hex_next
    jmp .Lstr_bad
.Lstr_check_upper:
    cmp al, 'A'
    jb .Lstr_bad
    cmp al, 'F'
    ja .Lstr_bad
.Lstr_hex_next:
    inc rbx
    dec rcx
    jnz .Lstr_hex_loop
    jmp .Lstr_loop
.Lstr_bad:
    mov rax, rbx
    ret
.Lstr_eof:
    mov rax, rdx
    ret

parse_object:
    mov rbx, qword ptr [rip + pos]
    inc rbx
    mov qword ptr [rip + pos], rbx
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jae .Lobj_eof
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '}'
    je .Lobj_empty
.Lobj_loop:
    call parse_string
    cmp rax, -1
    jne .Lobj_ret
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jae .Lobj_eof
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ':'
    jne .Lobj_bad
    inc rbx
    mov qword ptr [rip + pos], rbx
    call skip_ws
    call parse_value
    cmp rax, -1
    jne .Lobj_ret
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jae .Lobj_eof
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ','
    je .Lobj_comma
    cmp al, '}'
    je .Lobj_end
    jmp .Lobj_bad
.Lobj_comma:
    inc rbx
    mov qword ptr [rip + pos], rbx
    call skip_ws
    jmp .Lobj_loop
.Lobj_end:
    inc rbx
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Lobj_empty:
    inc rbx
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Lobj_bad:
    mov rax, rbx
    ret
.Lobj_eof:
    mov rdx, qword ptr [rip + len]
    mov rax, rdx
    ret
.Lobj_ret:
    ret

parse_array:
    mov rbx, qword ptr [rip + pos]
    inc rbx
    mov qword ptr [rip + pos], rbx
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jae .Larr_eof
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ']'
    je .Larr_empty
.Larr_loop:
    call parse_value
    cmp rax, -1
    jne .Larr_ret
    call skip_ws
    mov rbx, qword ptr [rip + pos]
    mov rdx, qword ptr [rip + len]
    cmp rbx, rdx
    jae .Larr_eof
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ','
    je .Larr_comma
    cmp al, ']'
    je .Larr_end
    jmp .Larr_bad
.Larr_comma:
    inc rbx
    mov qword ptr [rip + pos], rbx
    call skip_ws
    jmp .Larr_loop
.Larr_end:
    inc rbx
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Larr_empty:
    inc rbx
    mov qword ptr [rip + pos], rbx
    mov rax, -1
    ret
.Larr_bad:
    mov rax, rbx
    ret
.Larr_eof:
    mov rdx, qword ptr [rip + len]
    mov rax, rdx
    ret
.Larr_ret:
    ret
