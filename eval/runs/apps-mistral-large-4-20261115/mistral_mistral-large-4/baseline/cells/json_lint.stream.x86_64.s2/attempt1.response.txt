.equ INPUT_CAP, 65536
.equ OUT_CAP, 32
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ TRUE_LEN, 4
.equ FALSE_LEN, 5
.equ NULL_LEN, 4
.equ OK_LEN, 3
.equ ERR_PREFIX_LEN, 9

.section .rodata
str_true: .ascii "true"
str_false: .ascii "false"
str_null: .ascii "null"
ok_msg: .byte 111, 107, 10
err_prefix: .ascii "error at "
newline: .byte 10

.text
.globl _start
_start:
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, INPUT_CAP
    xor r12d, r12d
read_loop:
    test rdx, rdx
    jz read_done
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_CAP
    sub rdx, r12
    jmp read_loop
read_done:
    mov [rip + input_len], r12
    xor eax, eax
    mov [rip + pos], rax
    mov [rip + err_pos], rax
    call parse_document
    test al, al
    jnz report_error
    lea rsi, [rip + ok_msg]
    mov edx, OK_LEN
    mov edi, 1
    call write_all
    mov edi, 0
    mov eax, SYS_EXIT
    syscall

report_error:
    lea rsi, [rip + out_buf]
    add rsi, OUT_CAP
    mov rax, [rip + err_pos]
    mov ebx, 10
conv_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz conv_loop
    mov r8, rsi
    lea rdx, [rip + out_buf]
    add rdx, OUT_CAP
    sub rdx, r8
    lea rsi, [rip + err_prefix]
    mov edx, ERR_PREFIX_LEN
    mov edi, 1
    call write_all
    mov rsi, r8
    lea rdx, [rip + out_buf]
    add rdx, OUT_CAP
    sub rdx, r8
    mov edi, 1
    call write_all
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    call write_all
    mov edi, 1
    mov eax, SYS_EXIT
    syscall

write_all:
    test rdx, rdx
    jz wa_done
    mov eax, SYS_WRITE
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_all
wa_done:
    ret

skip_ws:
    mov rax, [rip + pos]
    mov rdx, [rip + input_len]
sw_loop:
    cmp rax, rdx
    jae sw_done
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x20
    je sw_adv
    cmp cl, 0x09
    je sw_adv
    cmp cl, 0x0A
    je sw_adv
    cmp cl, 0x0D
    je sw_adv
    jmp sw_done
sw_adv:
    inc rax
    jmp sw_loop
sw_done:
    mov [rip + pos], rax
    ret

parse_document:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jb pd_have_value
    mov [rip + err_pos], rax
    mov al, 1
    ret
pd_have_value:
    call parse_value
    test al, al
    jnz pd_ret
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jb pd_trailing
    xor eax, eax
    ret
pd_trailing:
    mov [rip + err_pos], rax
    mov al, 1
pd_ret:
    ret

parse_value:
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae pv_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7B
    je pv_obj
    cmp cl, 0x5B
    je pv_arr
    cmp cl, 0x22
    je pv_str
    cmp cl, 0x2D
    je pv_num
    cmp cl, 0x74
    je pv_true
    cmp cl, 0x66
    je pv_false
    cmp cl, 0x6E
    je pv_null
    cmp cl, 0x30
    jb pv_err
    cmp cl, 0x39
    ja pv_err
    jmp pv_num
pv_obj:
    jmp parse_object
pv_arr:
    jmp parse_array
pv_str:
    jmp parse_string
pv_num:
    jmp parse_number
pv_true:
    lea rsi, [rip + str_true]
    mov edx, TRUE_LEN
    jmp parse_keyword
pv_false:
    lea rsi, [rip + str_false]
    mov edx, FALSE_LEN
    jmp parse_keyword
pv_null:
    lea rsi, [rip + str_null]
    mov edx, NULL_LEN
    jmp parse_keyword
pv_err:
    mov [rip + err_pos], rax
    mov al, 1
    ret
pv_eof:
    mov [rip + err_pos], rax
    mov al, 1
    ret

parse_keyword:
    mov rax, [rip + pos]
    mov r8, [rip + input_len]
    lea r9, [rip + input_buf]
    xor ecx, ecx
pk_loop:
    cmp rcx, rdx
    jae pk_ok
    lea r10, [rax + rcx]
    cmp r10, r8
    jae pk_eof
    movzx r11d, byte ptr [r9 + r10]
    movzx ebx, byte ptr [rsi + rcx]
    cmp r11b, bl
    jne pk_mismatch
    inc rcx
    jmp pk_loop
pk_ok:
    add rax, rdx
    mov [rip + pos], rax
    xor eax, eax
    ret
pk_eof:
    mov [rip + err_pos], r8
    mov al, 1
    ret
pk_mismatch:
    lea r10, [rax + rcx]
    mov [rip + err_pos], r10
    mov al, 1
    ret

parse_string:
    mov rax, [rip + pos]
    inc rax
    mov [rip + pos], rax
ps_loop:
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae ps_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    je ps_close
    cmp cl, 0x5C
    je ps_escape
    cmp cl, 0x20
    jb ps_err_rax
    inc rax
    mov [rip + pos], rax
    jmp ps_loop
ps_close:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret
ps_eof:
    mov rax, [rip + input_len]
    mov [rip + err_pos], rax
    mov al, 1
    ret
ps_err_rax:
    mov [rip + err_pos], rax
    mov al, 1
    ret
ps_escape:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + input_len]
    jae ps_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    je ps_esc_ok
    cmp cl, 0x5C
    je ps_esc_ok
    cmp cl, 0x2F
    je ps_esc_ok
    cmp cl, 0x62
    je ps_esc_ok
    cmp cl, 0x66
    je ps_esc_ok
    cmp cl, 0x6E
    je ps_esc_ok
    cmp cl, 0x72
    je ps_esc_ok
    cmp cl, 0x74
    je ps_esc_ok
    cmp cl, 0x75
    je ps_esc_u
    jmp ps_err_rax
ps_esc_ok:
    inc rax
    mov [rip + pos], rax
    jmp ps_loop
ps_esc_u:
    inc rax
    mov [rip + pos], rax
    xor ecx, ecx
ps_hex_loop:
    cmp rcx, 4
    jae ps_loop
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae ps_eof
    lea rsi, [rip + input_buf]
    movzx edx, byte ptr [rsi + rax]
    cmp dl, 0x30
    jb ps_err_rax
    cmp dl, 0x39
    jbe ps_hex_next
    cmp dl, 0x61
    jb ps_check_upper
    cmp dl, 0x66
    jbe ps_hex_next
ps_check_upper:
    cmp dl, 0x41
    jb ps_err_rax
    cmp dl, 0x46
    ja ps_err_rax
ps_hex_next:
    inc rax
    mov [rip + pos], rax
    inc rcx
    jmp ps_hex_loop

parse_number:
    mov rax, [rip + pos]
    lea rsi, [rip + input_buf]
    mov rdx, [rip + input_len]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2D
    jne pn_no_minus
    inc rax
    cmp rax, rdx
    jae pn_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_err_rax
    cmp cl, 0x39
    ja pn_err_rax
pn_no_minus:
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    je pn_zero
    inc rax
pn_digits_loop:
    cmp rax, rdx
    jae pn_frac_check
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_frac_check
    cmp cl, 0x39
    ja pn_frac_check
    inc rax
    jmp pn_digits_loop
pn_zero:
    inc rax
    cmp rax, rdx
    jae pn_frac_check
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_frac_check
    cmp cl, 0x39
    ja pn_frac_check
    jmp pn_err_rax
pn_frac_check:
    cmp rax, rdx
    jae pn_success
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2E
    jne pn_exp_check
    inc rax
    cmp rax, rdx
    jae pn_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_err_rax
    cmp cl, 0x39
    ja pn_err_rax
    inc rax
pn_frac_digits_loop:
    cmp rax, rdx
    jae pn_exp_check
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_exp_check
    cmp cl, 0x39
    ja pn_exp_check
    inc rax
    jmp pn_frac_digits_loop
pn_exp_check:
    cmp rax, rdx
    jae pn_success
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x65
    je pn_exp
    cmp cl, 0x45
    je pn_exp
    jmp pn_success
pn_exp:
    inc rax
    cmp rax, rdx
    jae pn_eof
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2B
    je pn_exp_sign
    cmp cl, 0x2D
    je pn_exp_sign
    jmp pn_exp_digit
pn_exp_sign:
    inc rax
    cmp rax, rdx
    jae pn_eof
pn_exp_digit:
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_err_rax
    cmp cl, 0x39
    ja pn_err_rax
    inc rax
pn_exp_digits_loop:
    cmp rax, rdx
    jae pn_success
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x30
    jb pn_success
    cmp cl, 0x39
    ja pn_success
    inc rax
    jmp pn_exp_digits_loop
pn_success:
    mov [rip + pos], rax
    xor eax, eax
    ret
pn_eof:
    mov rax, [rip + input_len]
    mov [rip + err_pos], rax
    mov al, 1
    ret
pn_err_rax:
    mov [rip + err_pos], rax
    mov al, 1
    ret

parse_object:
    mov rax, [rip + pos]
    inc rax
    mov [rip + pos], rax
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae po_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x7D
    je po_empty
po_member_loop:
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae po_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x22
    jne po_err_rax
    call parse_string
    test al, al
    jnz po_ret
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae po_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x3A
    jne po_err_rax
    inc rax
    mov [rip + pos], rax
    call skip_ws
    call parse_value
    test al, al
    jnz po_ret
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae po_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2C
    je po_comma
    cmp cl, 0x7D
    je po_close
    jmp po_err_rax
po_comma:
    inc rax
    mov [rip + pos], rax
    call skip_ws
    jmp po_member_loop
po_close:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret
po_empty:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret
po_eof:
    mov rax, [rip + input_len]
    mov [rip + err_pos], rax
    mov al, 1
    ret
po_err_rax:
    mov [rip + err_pos], rax
    mov al, 1
po_ret:
    ret

parse_array:
    mov rax, [rip + pos]
    inc rax
    mov [rip + pos], rax
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae pa_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x5D
    je pa_close
pa_value_loop:
    call parse_value
    test al, al
    jnz pa_ret
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + input_len]
    jae pa_eof
    lea rsi, [rip + input_buf]
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, 0x2C
    je pa_comma
    cmp cl, 0x5D
    je pa_close
    jmp pa_err_rax
pa_comma:
    inc rax
    mov [rip + pos], rax
    call skip_ws
    jmp pa_value_loop
pa_close:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret
pa_eof:
    mov rax, [rip + input_len]
    mov [rip + err_pos], rax
    mov al, 1
    ret
pa_err_rax:
    mov [rip + err_pos], rax
    mov al, 1
pa_ret:
    ret

.bss
.balign 8
input_buf: .zero INPUT_CAP
input_len: .zero 8
pos: .zero 8
err_pos: .zero 8
out_buf: .zero OUT_CAP
