.text
.globl _start
_start:
    xor r12d, r12d
    lea r13, [rip + buf]
read_loop:
    mov eax, 65536
    sub eax, r12d
    test eax, eax
    jz read_done
    mov edx, eax
    xor edi, edi
    mov rsi, r13
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    add r13, rax
    jmp read_loop
read_done:
    xor r14d, r14d
row_loop:
    cmp r14, r12
    jae rows_done
    mov r9d, r12d
    sub r9d, r14d
    cmp r9d, 16
    jbe n_ok
    mov r9d, 16
n_ok:
    lea r15, [rip + outbuf]
    mov r8d, 7
off_loop:
    mov rax, r14
    mov ecx, r8d
    shl ecx, 2
    shr rax, cl
    and eax, 0xF
    lea rdx, [rip + hexdig]
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r15], al
    inc r15
    dec r8d
    jns off_loop
    mov byte ptr [r15], 0x20
    mov byte ptr [r15 + 1], 0x20
    add r15, 2
    xor r8d, r8d
slot_loop:
    cmp r8d, r9d
    jb has_byte
    mov byte ptr [r15], 0x20
    mov byte ptr [r15 + 1], 0x20
    mov byte ptr [r15 + 2], 0x20
    add r15, 3
    jmp slot_after
has_byte:
    lea rsi, [rip + buf]
    lea rsi, [rsi + r14]
    movzx eax, byte ptr [rsi + r8]
    mov edx, eax
    shr edx, 4
    lea rcx, [rip + hexdig]
    movzx edx, byte ptr [rcx + rdx]
    mov byte ptr [r15], dl
    and eax, 0xF
    movzx eax, byte ptr [rcx + rax]
    mov byte ptr [r15 + 1], al
    mov byte ptr [r15 + 2], 0x20
    add r15, 3
slot_after:
    cmp r8d, 7
    jne no_extra
    mov byte ptr [r15], 0x20
    inc r15
no_extra:
    inc r8d
    cmp r8d, 16
    jb slot_loop
    mov byte ptr [r15], 0x20
    inc r15
    mov byte ptr [r15], 0x7C
    inc r15
    xor r8d, r8d
char_loop:
    cmp r8d, r9d
    jae char_done
    lea rsi, [rip + buf]
    lea rsi, [rsi + r14]
    movzx eax, byte ptr [rsi + r8]
    cmp al, 0x20
    jb char_dot
    cmp al, 0x7E
    ja char_dot
    jmp char_store
char_dot:
    mov al, 0x2E
char_store:
    mov byte ptr [r15], al
    inc r15
    inc r8d
    jmp char_loop
char_done:
    mov byte ptr [r15], 0x7C
    inc r15
    mov byte ptr [r15], 10
    inc r15
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    syscall
    add r14, 16
    jmp row_loop
rows_done:
    lea r15, [rip + outbuf]
    mov r14, r12
    mov r8d, 7
final_loop:
    mov rax, r14
    mov ecx, r8d
    shl ecx, 2
    shr rax, cl
    and eax, 0xF
    lea rdx, [rip + hexdig]
    movzx eax, byte ptr [rdx + rax]
    mov byte ptr [r15], al
    inc r15
    dec r8d
    jns final_loop
    mov byte ptr [r15], 10
    inc r15
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.balign 16
buf:
    .zero 65536
outbuf:
    .zero 128
