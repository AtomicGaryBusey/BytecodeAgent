.equ INPUT_MAX, 49152
.equ LINE_LEN, 76

.globl _start
.text
_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdi, 0
    mov rdx, INPUT_MAX
    sub rdx, r12
    xor eax, eax
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    mov rcx, r12
    lea rdi, [rip + outbuf]
    mov r13, rdi
    lea r9, [rip + b64]
    xor r8, r8
encode_loop:
    cmp rcx, 3
    jb encode_tail
    xor r10, r10
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    mov r11d, eax
    mov eax, r11d
    shr eax, 18
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov eax, r11d
    shr eax, 12
    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov eax, r11d
    shr eax, 6
    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov eax, r11d
    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    add rsi, 3
    sub rcx, 3
    jmp group_wrap
encode_tail:
    test rcx, rcx
    jz encode_done
    mov r10, 1
    cmp rcx, 1
    je tail1
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    mov r11d, eax
    mov eax, r11d
    shr eax, 18
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov eax, r11d
    shr eax, 12
    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov eax, r11d
    shr eax, 6
    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov dl, 61
    mov byte ptr [rdi], dl
    inc rdi
    jmp group_wrap
tail1:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov r11d, eax
    mov eax, r11d
    shr eax, 18
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov eax, r11d
    shr eax, 12
    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi
    mov dl, 61
    mov byte ptr [rdi], dl
    inc rdi
    mov dl, 61
    mov byte ptr [rdi], dl
    inc rdi
group_wrap:
    add r8, 4
    cmp r8, LINE_LEN
    jne no_line_nl
    mov dl, 10
    mov byte ptr [rdi], dl
    inc rdi
    xor r8, r8
no_line_nl:
    test r10, r10
    jz encode_loop
    jmp encode_done
encode_done:
    test r8, r8
    jz no_final_nl
    mov dl, 10
    mov byte ptr [rdi], dl
    inc rdi
no_final_nl:
    mov r14, rdi
    sub r14, r13
    test r14, r14
    jz exit_prog
    mov rsi, r13
    mov rdx, r14
write_loop:
    mov rdi, 1
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_loop
exit_prog:
    xor edi, edi
    mov eax, 60
    syscall
