.text
.globl _start
_start:
    lea rsi, [rip + input_buf]
    xor edi, edi
    mov edx, 65537
read_loop:
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    lea rax, [rip + input_buf]
    sub rsi, rax
    mov r8, rsi

    lea rsi, [rip + input_buf]
    mov rcx, r8
find_loop:
    test rcx, rcx
    jz no_bang
    movzx edx, byte ptr [rsi]
    cmp dl, 0x21
    je found_bang
    inc rsi
    dec rcx
    jmp find_loop
no_bang:
    mov rbx, r8
    lea rbp, [rip + input_buf]
    add rbp, r8
    xor r8d, r8d
    jmp setup
found_bang:
    lea rax, [rip + input_buf]
    sub rsi, rax
    mov rbx, rsi
    lea rbp, [rip + input_buf]
    add rbp, rsi
    inc rbp
    mov rax, r8
    sub rax, rsi
    dec rax
    mov r8, rax
setup:
    lea r15, [rip + input_buf]
    xor r14, r14
    xor r9, r9
validate_loop:
    cmp r14, rbx
    jae validate_done
    movzx eax, byte ptr [r15 + r14]
    cmp al, 0x5B
    je val_open
    cmp al, 0x5D
    je val_close
    inc r14
    jmp validate_loop
val_open:
    mov word ptr [bstack + r9*2], r14w
    inc r9
    inc r14
    jmp validate_loop
val_close:
    test r9, r9
    jz invalid
    dec r9
    movzx edx, word ptr [bstack + r9*2]
    mov ecx, r14d
    inc ecx
    mov dword ptr [match_tab + rdx*4], ecx
    mov ecx, edx
    inc ecx
    mov dword ptr [match_tab + r14*4], ecx
    inc r14
    jmp validate_loop
validate_done:
    test r9, r9
    jnz invalid
exec_setup:
    lea r12, [rip + tape]
    lea r13, [rip + out_buf]
    xor r14, r14
exec_loop:
    cmp r14, rbx
    jae exec_done
    movzx eax, byte ptr [r15 + r14]
    cmp al, 0x3E
    je cmd_gt
    cmp al, 0x3C
    je cmd_lt
    cmp al, 0x2B
    je cmd_plus
    cmp al, 0x2D
    je cmd_minus
    cmp al, 0x2E
    je cmd_out
    cmp al, 0x2C
    je cmd_in
    cmp al, 0x5B
    je cmd_open
    cmp al, 0x5D
    je cmd_close
    inc r14
    jmp exec_loop
cmd_gt:
    inc r12
    inc r14
    jmp exec_loop
cmd_lt:
    dec r12
    inc r14
    jmp exec_loop
cmd_plus:
    inc byte ptr [r12]
    inc r14
    jmp exec_loop
cmd_minus:
    dec byte ptr [r12]
    inc r14
    jmp exec_loop
cmd_out:
    mov al, byte ptr [r12]
    mov byte ptr [r13], al
    inc r13
    inc r14
    jmp exec_loop
cmd_in:
    test r8, r8
    jz cmd_in_zero
    mov al, byte ptr [rbp]
    inc rbp
    dec r8
    mov byte ptr [r12], al
    inc r14
    jmp exec_loop
cmd_in_zero:
    mov byte ptr [r12], 0
    inc r14
    jmp exec_loop
cmd_open:
    cmp byte ptr [r12], 0
    jne open_no_jump
    mov edx, dword ptr [match_tab + r14*4]
    mov r14, rdx
    jmp exec_loop
open_no_jump:
    inc r14
    jmp exec_loop
cmd_close:
    cmp byte ptr [r12], 0
    je close_no_jump
    mov edx, dword ptr [match_tab + r14*4]
    mov r14, rdx
    jmp exec_loop
close_no_jump:
    inc r14
    jmp exec_loop
exec_done:
    lea rsi, [rip + out_buf]
    mov rdx, r13
    sub rdx, rsi
    test rdx, rdx
    jz exit0
    mov edi, 1
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
exit0:
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 8
input_buf:
    .zero 65537
.align 8
tape:
    .zero 30000
.align 8
out_buf:
    .zero 65536
.align 8
match_tab:
    .zero 262148
.align 8
bstack:
    .zero 131074
