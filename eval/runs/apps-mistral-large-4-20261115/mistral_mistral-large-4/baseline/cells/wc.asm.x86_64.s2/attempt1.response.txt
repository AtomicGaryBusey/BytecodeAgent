.equ BUF_SIZE, 65536

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + buffer]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    mov rcx, rax

process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je 2f
    cmp al, 9
    jb 3f
    cmp al, 13
    jbe 2f
3:
    test r15d, r15d
    jnz 4f
    inc r13
    mov r15d, 1
    jmp 4f
2:
    xor r15d, r15d
4:
    jmp process_loop

read_done:
    lea rbx, [rip + counts]
    mov qword ptr [rbx], r12
    mov qword ptr [rbx + 8], r13
    mov qword ptr [rbx + 16], r14

    lea rsi, [rip + outbuf]
    mov rdi, rsi
    lea rbx, [rip + counts]
    mov r15d, 3
out_loop:
    mov rax, qword ptr [rbx]
    add rbx, 8
    xor ecx, ecx
    mov r8d, 10
conv_loop:
    xor edx, edx
    div r8
    push rdx
    inc rcx
    test rax, rax
    jnz conv_loop
pop_loop:
    pop rax
    add al, 48
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz pop_loop
    dec r15
    jnz add_space
    jmp out_done
add_space:
    mov byte ptr [rdi], 32
    inc rdi
    jmp out_loop
out_done:
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, rsi
    mov r8, rsi
    mov r9, rdx
write_loop:
    test r9, r9
    jz write_done
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jz write_done
    js write_done
    add r8, rax
    sub r9, rax
    jmp write_loop
write_done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
buffer:
    .space BUF_SIZE
outbuf:
    .space 64
counts:
    .space 24
