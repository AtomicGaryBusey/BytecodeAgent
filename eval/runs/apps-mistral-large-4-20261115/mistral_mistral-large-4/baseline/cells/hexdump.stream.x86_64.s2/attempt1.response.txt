.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rdi, [rip + out_buf]
    xor r8, r8
row_loop:
    cmp r8, rbx
    jae final_line
    mov r9, rbx
    sub r9, r8
    cmp r9, 16
    jbe n_ok
    mov r9, 16
n_ok:
    mov rax, r8
    mov r11, 8
    mov r10, 28
off_loop:
    mov rdx, rax
    mov rcx, r10
    shr rdx, cl
    and edx, 15
    lea rsi, [rip + hexdigits]
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub r10, 4
    dec r11
    jnz off_loop
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2
    xor r10, r10
pos_loop:
    cmp r10, r9
    jae missing_slot
    lea rsi, [rip + input_buf]
    mov rax, r8
    add rax, r10
    movzx eax, byte ptr [rsi + rax]
    mov rdx, rax
    shr rdx, 4
    and edx, 15
    lea rsi, [rip + hexdigits]
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov rdx, rax
    and edx, 15
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    jmp after_slot
missing_slot:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
after_slot:
    cmp r10, 7
    jne no_extra
    mov byte ptr [rdi], 32
    inc rdi
no_extra:
    inc r10
    cmp r10, 16
    jb pos_loop
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi
    xor r10, r10
char_loop:
    cmp r10, r9
    jae char_done
    lea rsi, [rip + input_buf]
    mov rax, r8
    add rax, r10
    movzx eax, byte ptr [rsi + rax]
    cmp eax, 32
    jb dot_char
    cmp eax, 126
    ja dot_char
    mov byte ptr [rdi], al
    jmp char_store
dot_char:
    mov byte ptr [rdi], 46
char_store:
    inc rdi
    inc r10
    jmp char_loop
char_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    add r8, 16
    jmp row_loop
final_line:
    mov rax, rbx
    mov r11, 8
    mov r10, 28
final_off_loop:
    mov rdx, rax
    mov rcx, r10
    shr rdx, cl
    and edx, 15
    lea rsi, [rip + hexdigits]
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub r10, 4
    dec r11
    jnz final_off_loop
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
write_loop:
    mov eax, 1
    syscall
    cmp rax, rdx
    je write_done
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .space 65536
out_buf:
    .space 323593
