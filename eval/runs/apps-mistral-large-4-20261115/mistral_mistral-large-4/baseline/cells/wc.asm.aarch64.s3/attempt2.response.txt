.text
.equ CHUNK, 4096
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
read_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #CHUNK
    mov x8, #63
    svc #0
    cbz x0, finish
    mov x2, x0
    adrp x1, buf
    add x1, x1, :lo12:buf
byte_loop:
    cbz x2, read_loop
    ldrb w3, [x1], #1
    sub x2, x2, #1
    add x19, x19, #1
    cmp w3, #0x0A
    b.ne not_newline
    add x20, x20, #1
not_newline:
    cmp w3, #0x20
    beq whitespace
    sub w4, w3, #0x09
    cmp w4, #4
    bls whitespace
    cbnz x22, byte_loop
    add x21, x21, #1
    mov x22, #1
    b byte_loop
whitespace:
    mov x22, #0
    b byte_loop
finish:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x23, x24
    mov x0, x20
    bl append_num
    mov w3, #0x20
    strb w3, [x23], #1
    mov x0, x21
    bl append_num
    mov w3, #0x20
    strb w3, [x23], #1
    mov x0, x19
    bl append_num
    mov w3, #0x0A
    strb w3, [x23], #1
    sub x2, x23, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

append_num:
    sub sp, sp, #32
    mov x1, sp
    mov x2, #0
    cbz x0, append_num_zero
append_num_loop:
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add x5, x5, #48
    strb w5, [x1], #1
    add x2, x2, #1
    mov x0, x4
    cbnz x0, append_num_loop
    b append_num_copy
append_num_zero:
    mov w5, #48
    strb w5, [x1], #1
    mov x2, #1
append_num_copy:
    cbz x2, append_num_done
    sub x1, x1, #1
    ldrb w7, [x1]
    strb w7, [x23], #1
    sub x2, x2, #1
    b append_num_copy
append_num_done:
    add sp, sp, #32
    ret

.bss
.balign 8
buf: .zero CHUNK
.balign 8
outbuf: .zero 32
