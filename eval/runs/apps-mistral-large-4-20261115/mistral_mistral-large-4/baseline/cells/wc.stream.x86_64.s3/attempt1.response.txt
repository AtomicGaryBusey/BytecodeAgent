.equ BUF_SIZE, 65536

.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jz done
    mov rcx, rax
    lea rsi, [rip + buf]
process_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r12
    cmp al, 0x0A
    jne no_line
    inc r13
no_line:
    cmp al, 0x20
    je ws_byte
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0D
    jbe ws_byte
    jmp not_ws
ws_byte:
    xor r15d, r15d
    jmp process_loop
not_ws:
    test r15d, r15d
    jnz process_loop
    inc r14
    mov r15d, 1
    jmp process_loop
done:
    mov qword ptr [rip + counts], r13
    mov qword ptr [rip + counts + 8], r14
    mov qword ptr [rip + counts + 16], r12
    lea rdi, [rip + outbuf]
    lea r9, [rip + counts]
    xor ebx, ebx
out_loop:
    cmp ebx, 3
    jae out_newline
    mov rax, qword ptr [r9 + rbx*8]
    lea rsi, [rip + digits + 7]
    xor ecx, ecx
conv_loop:
    xor edx, edx
    mov r8, 10
    div r8
    add dl, 0x30
    mov byte ptr [rsi], dl
    dec rsi
    inc rcx
    test rax, rax
    jnz conv_loop
    inc rsi
copy_loop:
    test rcx, rcx
    jz copy_done
    movzx eax, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp copy_loop
copy_done:
    inc ebx
    cmp ebx, 3
    jae out_newline
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp out_loop
out_newline:
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov r8, rdi
    lea rax, [rip + outbuf]
    sub r8, rax
    lea rsi, [rip + outbuf]
    mov edi, 1
write_loop:
    mov eax, 1
    mov rdx, r8
    syscall
    cmp rax, r8
    jae exit_now
    add rsi, rax
    sub r8, rax
    jmp write_loop
exit_now:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
buf:
    .zero BUF_SIZE
outbuf:
    .zero 32
digits:
    .zero 8
counts:
    .zero 24
