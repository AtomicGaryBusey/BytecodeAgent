.equ INPUT_MAX, 32768
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit_group, 94

.text
.globl _start
_start:
    mov x20, #0
read_loop:
    cmp x20, #INPUT_MAX
    b.hs read_done
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x20
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x0, line_start
    add x0, x0, :lo12:line_start
    adrp x1, line_len
    add x1, x1, :lo12:line_len
    adrp x2, input_buf
    add x2, x2, :lo12:input_buf
    mov x3, #0
    mov x4, #0
    mov x5, #0
parse_loop:
    cmp x4, x20
    b.hs parse_end
    ldrb w6, [x2, x4]
    cmp w6, #0x0A
    b.ne parse_next
    sub w7, w4, w5
    str w5, [x0, x3, lsl #2]
    str w7, [x1, x3, lsl #2]
    add x3, x3, #1
    add x5, x4, #1
parse_next:
    add x4, x4, #1
    b parse_loop
parse_end:
    cmp x5, x20
    b.hs parse_done
    str w5, [x0, x3, lsl #2]
    sub w7, w20, w5
    str w7, [x1, x3, lsl #2]
    add x3, x3, #1
parse_done:
    mov x19, x2
    mov x20, x0
    mov x21, x1
    mov x22, x3
    mov x23, #1
sort_outer:
    cmp x23, x22
    b.hs sort_done
    ldr w24, [x20, x23, lsl #2]
    ldr w25, [x21, x23, lsl #2]
    mov x26, #0
    mov x27, x23
bs_loop:
    cmp x26, x27
    b.hs bs_done
    add x28, x26, x27
    lsr x28, x28, #1
    ldr w0, [x20, x28, lsl #2]
    ldr w1, [x21, x28, lsl #2]
    mov x2, x24
    mov x3, x25
    mov x4, x19
    bl compare_lines
    cmp w0, #0
    b.le bs_lower
    mov x27, x28
    b bs_loop
bs_lower:
    add x26, x28, #1
    b bs_loop
bs_done:
    mov x5, x23
shift_loop:
    cmp x5, x26
    b.ls shift_done
    sub x6, x5, #1
    ldr w7, [x20, x6, lsl #2]
    str w7, [x20, x5, lsl #2]
    ldr w7, [x21, x6, lsl #2]
    str w7, [x21, x5, lsl #2]
    sub x5, x5, #1
    b shift_loop
shift_done:
    str w24, [x20, x26, lsl #2]
    str w25, [x21, x26, lsl #2]
    add x23, x23, #1
    b sort_outer
sort_done:
    mov x23, #0
out_loop:
    cmp x23, x22
    b.hs out_done
    ldr w24, [x20, x23, lsl #2]
    ldr w25, [x21, x23, lsl #2]
    cmp x25, #0
    b.eq out_nl
    mov x0, #1
    add x1, x19, x24
    mov x2, x25
    bl write_all
out_nl:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    bl write_all
    add x23, x23, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #SYS_exit_group
    svc #0

compare_lines:
    cmp x1, x3
    csel x5, x1, x3, le
    add x6, x4, x0
    add x7, x4, x2
    mov x8, #0
cmp_loop:
    cmp x8, x5
    b.hs cmp_len
    ldrb w9, [x6, x8]
    ldrb w10, [x7, x8]
    cmp w9, w10
    b.lo cmp_less
    b.hi cmp_greater
    add x8, x8, #1
    b cmp_loop
cmp_len:
    cmp x1, x3
    b.lo cmp_less
    b.hi cmp_greater
    mov w0, #0
    ret
cmp_less:
    mov w0, #0xffffffff
    ret
cmp_greater:
    mov w0, #1
    ret

write_all:
    cbz x2, wa_done
    mov x3, x0
wa_loop:
    mov x0, x3
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le wa_done
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, wa_loop
wa_done:
    ret

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .zero 32768
.balign 8
line_start:
    .zero 4000
.balign 8
line_len:
    .zero 4000
