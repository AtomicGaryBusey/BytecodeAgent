.globl _start
.equ BUF_CAP, 32768
.equ MAX_LINES, 32768

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buf]
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r13 + r12]
    mov rdx, BUF_CAP
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    cmp r12, BUF_CAP
    jae read_done
    jmp read_loop
read_done:
    test r12, r12
    jz exit0
    cmp byte ptr [r13 + r12 - 1], 10
    je no_append
    mov byte ptr [r13 + r12], 10
    inc r12
no_append:
    lea rbx, [rip + line_arr]
    xor r14, r14
    xor r15, r15
parse_loop:
    mov rax, r15
scan_loop:
    cmp byte ptr [r13 + rax], 10
    je found_nl
    inc rax
    jmp scan_loop
found_nl:
    mov rcx, rax
    sub rcx, r15
    mov rdx, r14
    shl rdx, 3
    mov r8, rcx
    shl r8, 32
    or r8, r15
    mov [rbx + rdx], r8
    inc r14
    lea r15, [rax + 1]
    cmp r15, r12
    jb parse_loop
    cmp r14, 2
    jb sort_done
    mov r15, 1
outer_loop:
    cmp r15, r14
    jae sort_done
    mov rdx, r15
    shl rdx, 3
    mov r12, [rbx + rdx]
    mov rbp, r15
    dec rbp
inner_loop:
    cmp rbp, -1
    je insert_zero
    mov rdx, rbp
    shl rdx, 3
    mov rdi, [rbx + rdx]
    mov rsi, r12
    call compare_gt
    test eax, eax
    jz insert_at_j1
    mov rdx, rbp
    shl rdx, 3
    mov [rbx + rdx + 8], rdi
    dec rbp
    jmp inner_loop
insert_zero:
    mov qword ptr [rbx], r12
    jmp next_i
insert_at_j1:
    lea rdx, [rbp + 1]
    shl rdx, 3
    mov [rbx + rdx], r12
next_i:
    inc r15
    jmp outer_loop
sort_done:
    xor r15, r15
output_loop:
    cmp r15, r14
    jae exit0
    mov rdx, r15
    shl rdx, 3
    mov rax, [rbx + rdx]
    mov r8, rax
    mov rdx, rax
    shr rdx, 32
    mov esi, r8d
    lea rdi, [rip + input_buf]
    add rdi, rsi
    mov rsi, rdi
    mov rdi, 1
    call write_all
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rdi, 1
    call write_all
    inc r15
    jmp output_loop
exit0:
    mov rax, 60
    xor rdi, rdi
    syscall

compare_gt:
    mov eax, edi
    mov ecx, esi
    mov rdx, rdi
    shr rdx, 32
    mov r8, rsi
    shr r8, 32
    mov r10, rdx
    mov r9, rdx
    cmp r9, r8
    cmova r9, r8
    lea rdx, [rip + input_buf]
    add rdx, rax
    lea r11, [rip + input_buf]
    add r11, rcx
    xor ecx, ecx
cmp_loop:
    cmp rcx, r9
    jae prefix_done
    movzx eax, byte ptr [rdx + rcx]
    movzx esi, byte ptr [r11 + rcx]
    cmp al, sil
    ja greater
    jb less
    inc rcx
    jmp cmp_loop
prefix_done:
    cmp r10, r8
    ja greater
    xor eax, eax
    ret
greater:
    mov eax, 1
    ret
less:
    xor eax, eax
    ret

write_all:
    test rdx, rdx
    jz write_all_done
write_all_loop:
    mov rax, 1
    syscall
    test rax, rax
    jle write_all_done
    add rsi, rax
    sub rdx, rax
    jnz write_all_loop
write_all_done:
    ret

.section .rodata
newline: .byte 10

.bss
.balign 8
input_buf: .space BUF_CAP + 1
.balign 8
line_arr: .space MAX_LINES * 8
