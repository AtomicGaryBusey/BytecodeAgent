.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUF_SIZE, 65536
.equ MAX_LINES, 1000
.equ NL, 10

.section .rodata
newline:
    .byte 10

.section .bss
.align 8
buf:
    .space BUF_SIZE
.align 8
line_ptr:
    .space MAX_LINES*8
.align 8
line_len:
    .space MAX_LINES*8

.text
.globl _start
_start:
    xor r12, r12
    lea r13, [rip + buf]
read_loop:
    mov rax, SYS_READ
    xor rdi, rdi
    lea rsi, [r13 + r12]
    mov rdx, BUF_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, BUF_SIZE
    jb read_loop
read_done:
    xor r14, r14
    xor r15, r15
    xor rbx, rbx
scan_loop:
    cmp rbx, r12
    jae scan_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, NL
    je found_nl
    inc rbx
    jmp scan_loop
found_nl:
    mov rdi, r15
    mov rsi, rbx
    sub rsi, rdi
    mov rax, r14
    lea rcx, [rip + line_ptr]
    lea rdx, [r13 + rdi]
    mov [rcx + rax*8], rdx
    lea rcx, [rip + line_len]
    mov [rcx + rax*8], rsi
    inc r14
    inc rbx
    mov r15, rbx
    jmp scan_loop
scan_done:
    cmp r15, r12
    jae sort_init
    mov rdi, r15
    mov rsi, r12
    sub rsi, rdi
    mov rax, r14
    lea rcx, [rip + line_ptr]
    lea rdx, [r13 + rdi]
    mov [rcx + rax*8], rdx
    lea rcx, [rip + line_len]
    mov [rcx + rax*8], rsi
    inc r14
sort_init:
    cmp r14, 2
    jb print_init
    mov rbx, 1
sort_outer:
    cmp rbx, r14
    jae print_init
    lea rax, [rip + line_ptr]
    mov r12, [rax + rbx*8]
    lea rax, [rip + line_len]
    mov r13, [rax + rbx*8]
    lea r15, [rbx - 1]
sort_inner:
    test r15, r15
    js sort_place
    lea rax, [rip + line_ptr]
    mov rdi, [rax + r15*8]
    lea rax, [rip + line_len]
    mov rsi, [rax + r15*8]
    mov rdx, r12
    mov r8, r13
    call line_greater
    test al, al
    jz sort_place
    lea rax, [rip + line_ptr]
    mov rcx, [rax + r15*8]
    lea rdx, [r15 + 1]
    mov [rax + rdx*8], rcx
    lea rax, [rip + line_len]
    mov rcx, [rax + r15*8]
    mov [rax + rdx*8], rcx
    dec r15
    jmp sort_inner
sort_place:
    lea rdx, [r15 + 1]
    lea rax, [rip + line_ptr]
    mov [rax + rdx*8], r12
    lea rax, [rip + line_len]
    mov [rax + rdx*8], r13
    inc rbx
    jmp sort_outer
line_greater:
    mov rax, rsi
    cmp r8, rax
    cmovb rax, r8
    xor ecx, ecx
    test rax, rax
    jz 1f
2:
    movzx r9d, byte ptr [rdi + rcx]
    movzx r10d, byte ptr [rdx + rcx]
    cmp r9b, r10b
    jb 3f
    ja 4f
    inc rcx
    cmp rcx, rax
    jb 2b
1:
    cmp rsi, r8
    ja 4f
    jmp 3f
3:
    xor eax, eax
    ret
4:
    mov eax, 1
    ret
write_all:
    test rdx, rdx
    jz 1f
2:
    mov rax, SYS_WRITE
    syscall
    cmp rax, 0
    jle 1f
    add rsi, rax
    sub rdx, rax
    jnz 2b
1:
    ret
print_init:
    xor rbx, rbx
    test r14, r14
    jz exit_prog
print_loop:
    cmp rbx, r14
    jae exit_prog
    lea rax, [rip + line_ptr]
    mov rsi, [rax + rbx*8]
    lea rax, [rip + line_len]
    mov rdx, [rax + rbx*8]
    mov rdi, 1
    call write_all
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rdi, 1
    call write_all
    inc rbx
    jmp print_loop
exit_prog:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall
