.equ BUFSZ, 65536
.equ OUTSZ, 65537

.text
.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    movz x20, #1, lsl #16
read_loop:
    mov x0, #0
    mov x1, x19
    mov x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    sub x20, x20, x0
    cbz x20, read_zero
    b read_loop
read_zero:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x2, #0
    mov x8, #63
    svc #0
read_done:
    adrp x0, inbuf
    add x0, x0, :lo12:inbuf
    adrp x6, outbuf
    add x6, x6, :lo12:outbuf
    mov x5, x6
process_lines:
    cmp x0, x19
    b.hs lines_done
    mov x1, x0
scan:
    cmp x0, x19
    b.hs scan_done
    ldrb w2, [x0], #1
    cmp w2, #10
    b.ne scan
    b scan_done
scan_done:
    ldrb w2, [x0, #-1]
    cmp w2, #10
    b.eq newline_line
    sub x2, x0, x1
    b reverse
newline_line:
    sub x2, x0, x1
    sub x2, x2, #1
reverse:
    cbz x2, put_nl
    add x3, x1, x2
rev_loop:
    ldrb w4, [x3, #-1]!
    strb w4, [x5], #1
    cmp x3, x1
    b.hi rev_loop
put_nl:
    mov w4, #10
    strb w4, [x5], #1
    b process_lines
lines_done:
    sub x2, x5, x6
    cbz x2, exit0
    mov x0, #1
    mov x1, x6
    mov x8, #64
    svc #0
exit0:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 8
inbuf:
    .space BUFSZ
.balign 8
outbuf:
    .space OUTSZ
