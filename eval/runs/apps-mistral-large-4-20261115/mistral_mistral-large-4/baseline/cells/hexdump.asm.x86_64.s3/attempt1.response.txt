.text
.globl _start
_start:
    xor r8, r8
.Lread_loop:
    cmp r8, 65536
    jae .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r8
    mov edx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz .Lread_done
    add r8, rax
    jmp .Lread_loop
.Lread_done:
    mov r12, r8

    lea rdi, [rip + outbuf]
    xor r8, r8
.Lrow_loop:
    cmp r8, r12
    jae .Lrows_done
    mov rax, r12
    sub rax, r8
    cmp rax, 16
    jbe .Ln_ok
    mov rax, 16
.Ln_ok:
    mov r9, rax
    mov rax, r8
    call .Lemit_offset8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [rip + inbuf]
    add rsi, r8
    xor r10, r10
.Lhex_loop:
    cmp r10, r9
    jae .Lpad
    movzx eax, byte ptr [rsi + r10]
    call .Lemit_hex_byte
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp .Lslot_done
.Lpad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
.Lslot_done:
    cmp r10, 7
    jne .Lno_extra
    mov byte ptr [rdi], 0x20
    inc rdi
.Lno_extra:
    inc r10
    cmp r10, 16
    jb .Lhex_loop

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7C
    inc rdi
    xor r10, r10
.Lchar_loop:
    cmp r10, r9
    jae .Lchar_done
    movzx eax, byte ptr [rsi + r10]
    cmp al, 0x20
    jb .Ldot
    cmp al, 0x7E
    ja .Ldot
    mov [rdi], al
    jmp .Lchar_store
.Ldot:
    mov byte ptr [rdi], 0x2E
.Lchar_store:
    inc rdi
    inc r10
    jmp .Lchar_loop
.Lchar_done:
    mov byte ptr [rdi], 0x7C
    inc rdi
    mov byte ptr [rdi], 0x0A
    inc rdi
    add r8, 16
    jmp .Lrow_loop

.Lrows_done:
    mov rax, r12
    call .Lemit_offset8
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rbx, rsi
    mov r12, rdi
    sub r12, rsi
.Lwrite_loop:
    test r12, r12
    jz .Lwrite_done
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall
    add rbx, rax
    sub r12, rax
    jmp .Lwrite_loop
.Lwrite_done:
    mov eax, 60
    xor edi, edi
    syscall

.Lemit_offset8:
    mov r10, rax
    mov r11d, 8
.Loff_loop:
    mov rax, r10
    mov rcx, r11
    dec rcx
    shl rcx, 2
    shr rax, cl
    and eax, 0xF
    cmp al, 10
    jb .Loff_digit
    add al, 87
    jmp .Loff_store
.Loff_digit:
    add al, 48
.Loff_store:
    mov [rdi], al
    inc rdi
    dec r11
    jnz .Loff_loop
    ret

.Lemit_hex_byte:
    movzx eax, al
    mov ecx, eax
    shr ecx, 4
    cmp cl, 10
    jb .Lhex_hi_digit
    add cl, 87
    jmp .Lhex_hi_store
.Lhex_hi_digit:
    add cl, 48
.Lhex_hi_store:
    mov [rdi], cl
    mov ecx, eax
    and cl, 0xF
    cmp cl, 10
    jb .Lhex_lo_digit
    add cl, 87
    jmp .Lhex_lo_store
.Lhex_lo_digit:
    add cl, 48
.Lhex_lo_store:
    mov [rdi + 1], cl
    add rdi, 2
    ret

.section .bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 323600
