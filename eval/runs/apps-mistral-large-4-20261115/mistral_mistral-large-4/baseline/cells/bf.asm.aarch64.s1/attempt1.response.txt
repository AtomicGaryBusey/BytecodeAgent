.equ INPUT_CAP, 65536
.equ OUT_CAP, 65536
.equ BANG, 0x21
.equ GT, 0x3E
.equ LT, 0x3C
.equ PLUS, 0x2B
.equ MINUS, 0x2D
.equ DOT, 0x2E
.equ COMMA, 0x2C
.equ OPEN, 0x5B
.equ CLOSE, 0x5D

.text
.globl _start
_start:
  adrp x4, input_buf
  add x4, x4, :lo12:input_buf
  mov x19, #0
read_loop:
  cmp x19, #INPUT_CAP
  b.hs read_done
  add x1, x4, x19
  mov x2, #INPUT_CAP
  sub x2, x2, x19
  mov x0, #0
  mov x8, #63
  svc #0
  cmp x0, #0
  b.eq read_done
  add x19, x19, x0
  b read_loop
read_done:
  mov x20, #0
find_loop:
  cmp x20, x19
  b.hs not_found
  add x1, x4, x20
  ldrb w2, [x1]
  cmp w2, #BANG
  b.eq found
  add x20, x20, #1
  b find_loop
found:
  mov x21, x20
  add x22, x4, x20
  add x22, x22, #1
  sub x23, x19, x20
  sub x23, x23, #1
  mov x20, x4
  b split_done
not_found:
  mov x21, x19
  mov x20, x4
  add x22, x4, x19
  mov x23, #0
split_done:
  mov x24, #0
  adrp x27, jump_tab
  add x27, x27, :lo12:jump_tab
  adrp x5, open_stack
  add x5, x5, :lo12:open_stack
  mov x6, #0
  mov x7, #0
validate_loop:
  cmp x7, x21
  b.hs validate_done
  add x8, x20, x7
  ldrb w9, [x8]
  cmp w9, #OPEN
  b.eq do_open
  cmp w9, #CLOSE
  b.eq do_close
  b next_char
do_open:
  add x9, x5, x6, lsl #2
  str w7, [x9]
  add x6, x6, #1
  b next_char
do_close:
  cmp x6, #0
  b.eq unbalanced
  sub x6, x6, #1
  add x9, x5, x6, lsl #2
  ldr w10, [x9]
  add x9, x27, x7, lsl #2
  str w10, [x9]
  add x9, x27, x10, lsl #2
  str w7, [x9]
  b next_char
next_char:
  add x7, x7, #1
  b validate_loop
validate_done:
  cmp x6, #0
  b.ne unbalanced
  adrp x28, tape
  add x28, x28, :lo12:tape
  mov x25, #0
  mov x26, #0
  adrp x17, out_buf
  add x17, x17, :lo12:out_buf
  mov x18, #0
run_loop:
  cmp x25, x21
  b.hs run_end
  add x8, x20, x25
  ldrb w9, [x8]
  cmp w9, #GT
  b.eq cmd_gt
  cmp w9, #LT
  b.eq cmd_lt
  cmp w9, #PLUS
  b.eq cmd_plus
  cmp w9, #MINUS
  b.eq cmd_minus
  cmp w9, #DOT
  b.eq cmd_dot
  cmp w9, #COMMA
  b.eq cmd_comma
  cmp w9, #OPEN
  b.eq cmd_open
  cmp w9, #CLOSE
  b.eq cmd_close
  add x25, x25, #1
  b run_loop
cmd_gt:
  add x26, x26, #1
  add x25, x25, #1
  b run_loop
cmd_lt:
  sub x26, x26, #1
  add x25, x25, #1
  b run_loop
cmd_plus:
  add x8, x28, x26
  ldrb w9, [x8]
  add w9, w9, #1
  strb w9, [x8]
  add x25, x25, #1
  b run_loop
cmd_minus:
  add x8, x28, x26
  ldrb w9, [x8]
  sub w9, w9, #1
  strb w9, [x8]
  add x25, x25, #1
  b run_loop
cmd_dot:
  add x8, x28, x26
  ldrb w9, [x8]
  add x10, x17, x18
  strb w9, [x10]
  add x18, x18, #1
  cmp x18, #OUT_CAP
  b.hs flush_out
  add x25, x25, #1
  b run_loop
flush_out:
  mov x0, #1
  mov x1, x17
  mov x2, x18
  mov x8, #64
  svc #0
  mov x18, #0
  add x25, x25, #1
  b run_loop

  add x8, x28, x26
  cmp x24, x23
  b.hs comma_eof
  add x9, x22, x24
  ldrb w10, [x9]
  strb w10, [x8]
  add x24, x24, #1
  add x25, x25, #1
  b run_loop
comma_eof:
  mov w10, #0
  strb w10, [x8]
  add x25, x25, #1
  b run_loop
cmd_open:
  add x8, x28, x26
  ldrb w9, [x8]
  cmp w9, #0
  b.ne open_skip
  add x9, x27, x25, lsl #2
  ldr w10, [x9]
  add x25, x10, #1
  b run_loop
open_skip:
  add x25, x25, #1
  b run_loop
cmd_close:
  add x8, x28, x26
  ldrb w9, [x8]
  cmp w9, #0
  b.eq close_skip
  add x9, x27, x25, lsl #2
  ldr w10, [x9]
  add x25, x10, #1
  b run_loop
close_skip:
  add x25, x25, #1
  b run_loop
run_end:
  cmp x18, #0
  b.eq exit0
  mov x0, #1
  mov x1, x17
  mov x2, x18
  mov x8, #64
  svc #0
exit0:
  mov x0, #0
  mov x8, #93
  svc #0
  b .
unbalanced:
  mov x0, #1
  mov x8, #93
  svc #0
  b .
