.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 70000

.text
.globl _start
_start:
    xor r8d, r8d
    lea r9, [rip + input_buf]
read_loop:
    mov eax, INPUT_MAX
    sub eax, r8d
    test eax, eax
    jz read_done
    xor edi, edi
    lea rsi, [r9 + r8]
    mov rdx, rax
    xor eax, eax
    syscall
    cmp rax, 0
    jle read_done
    add r8, rax
    jmp read_loop
read_done:
    test r8, r8
    jz exit_prog

    lea rsi, [rip + input_buf]
    mov r9, r8
    lea rdi, [rip + output_buf]
    lea r12, [rip + output_buf]
    lea rbx, [rip + b64table]
    xor r11d, r11d

encode_loop:
    cmp r9, 3
    jb final_group
    movzx r13d, byte ptr [rsi]
    movzx r14d, byte ptr [rsi+1]
    movzx r15d, byte ptr [rsi+2]
    shl r13d, 16
    shl r14d, 8
    or r13d, r14d
    or r13d, r15d
    mov eax, r13d
    shr eax, 18
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov eax, r13d
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov eax, r13d
    shr eax, 6
    and eax, 63
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov eax, r13d
    and eax, 63
    movzx eax, byte ptr [rbx + rax]
    call put_char
    add rsi, 3
    sub r9, 3
    jmp encode_loop

final_group:
    test r9, r9
    jz finish
    cmp r9, 1
    je rem1
    movzx r13d, byte ptr [rsi]
    movzx r14d, byte ptr [rsi+1]
    shl r13d, 16
    shl r14d, 8
    or r13d, r14d
    mov eax, r13d
    shr eax, 18
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov eax, r13d
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov eax, r13d
    shr eax, 6
    and eax, 63
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov al, 61
    call put_char
    jmp finish
rem1:
    movzx r13d, byte ptr [rsi]
    shl r13d, 16
    mov eax, r13d
    shr eax, 18
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov eax, r13d
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [rbx + rax]
    call put_char
    mov al, 61
    call put_char
    mov al, 61
    call put_char
finish:
    test r11, r11
    jz write_setup
    mov byte ptr [rdi], 10
    inc rdi
write_setup:
    mov r10, rdi
    sub r10, r12
    mov rsi, r12
write_loop:
    test r10, r10
    jz exit_prog
    mov edi, 1
    mov rdx, r10
    mov eax, 1
    syscall
    test rax, rax
    jle exit_prog
    sub r10, rax
    add rsi, rax
    jmp write_loop

put_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r11
    cmp r11, 76
    jne put_char_ret
    mov byte ptr [rdi], 10
    inc rdi
    xor r11d, r11d
put_char_ret:
    ret

exit_prog:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
b64table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buf:
    .space INPUT_MAX
output_buf:
    .space OUTPUT_MAX
