.bss
.balign 8
buf: .space 65537
.balign 8
numbuf: .space 16

.section .rodata
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg
err_prefix: .ascii "error at "
err_prefix_len = . - err_prefix
str_true: .ascii "true"
str_false: .ascii "false"
str_null: .ascii "null"

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, rbx
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_done
    add rbx, rax
    cmp rbx, 65537
    jae read_done
    jmp read_loop
read_done:
    mov r12, rbx
    xor ebx, ebx
    lea r15, [rip + buf]
    call skip_ws
    cmp rbx, r12
    jae top_eof
    call parse_value
    test rax, rax
    jnz top_error
    call skip_ws
    cmp rbx, r12
    jae top_ok
    mov rax, rbx
    inc rax
    jmp top_error
top_eof:
    mov rax, r12
    inc rax
top_error:
    dec rax
    mov r13, rax
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, err_prefix_len
    syscall
    lea rsi, [rip + numbuf + 16]
    mov byte ptr [rsi - 1], 10
    dec rsi
    mov rax, r13
    mov ecx, 10
conv_loop:
    xor edx, edx
    div ecx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz conv_loop
    lea rdx, [rip + numbuf + 16]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
top_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp rbx, r12
    jae skip_ws_done
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 0x20
    je skip_ws_adv
    cmp al, 9
    je skip_ws_adv
    cmp al, 10
    je skip_ws_adv
    cmp al, 13
    je skip_ws_adv
    jmp skip_ws_done
skip_ws_adv:
    inc rbx
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp rbx, r12
    jae pv_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je pv_true
    cmp al, 'f'
    je pv_false
    cmp al, 'n'
    je pv_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb pv_err
    cmp al, '9'
    ja pv_err
    jmp parse_number
pv_true:
    lea rsi, [rip + str_true]
    mov edx, 4
    jmp parse_word
pv_false:
    lea rsi, [rip + str_false]
    mov edx, 5
    jmp parse_word
pv_null:
    lea rsi, [rip + str_null]
    mov edx, 4
    jmp parse_word
pv_eof:
    mov rax, r12
    inc rax
    ret
pv_err:
    mov rax, rbx
    inc rax
    ret

parse_word:
    xor ecx, ecx
pw_loop:
    cmp rcx, rdx
    jae pw_ok
    lea r10, [rbx + rcx]
    cmp r10, r12
    jae pw_eof
    movzx r8d, byte ptr [r15 + r10]
    movzx r9d, byte ptr [rsi + rcx]
    cmp r8b, r9b
    jne pw_err
    inc rcx
    jmp pw_loop
pw_ok:
    add rbx, rdx
    xor eax, eax
    ret
pw_eof:
    mov rax, r12
    inc rax
    ret
pw_err:
    lea rax, [rbx + rcx]
    inc rax
    ret

parse_number:
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '-'
    jne pn_no_minus
    inc rbx
    cmp rbx, r12
    jae pn_eof
pn_no_minus:
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    je pn_zero
    cmp al, '1'
    jb pn_err
    cmp al, '9'
    ja pn_err
pn_int_loop:
    inc rbx
    cmp rbx, r12
    jae pn_after_int
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    jb pn_after_int
    cmp al, '9'
    ja pn_after_int
    jmp pn_int_loop
pn_zero:
    inc rbx
    jmp pn_after_int
pn_after_int:
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '.'
    jne pn_exp_check
    inc rbx
    cmp rbx, r12
    jae pn_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    jb pn_err
    cmp al, '9'
    ja pn_err
pn_frac_loop:
    inc rbx
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    jb pn_exp_check
    cmp al, '9'
    ja pn_exp_check
    jmp pn_frac_loop
pn_exp_check:
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 'e'
    je pn_exp
    cmp al, 'E'
    je pn_exp
    jmp pn_ok
pn_exp:
    inc rbx
    cmp rbx, r12
    jae pn_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '+'
    je pn_exp_sign
    cmp al, '-'
    je pn_exp_sign
    jmp pn_exp_digits
pn_exp_sign:
    inc rbx
    cmp rbx, r12
    jae pn_eof
pn_exp_digits:
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    jb pn_err
    cmp al, '9'
    ja pn_err
pn_exp_loop:
    inc rbx
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    jb pn_ok
    cmp al, '9'
    ja pn_ok
    jmp pn_exp_loop
pn_eof:
    mov rax, r12
    inc rax
    ret
pn_err:
    mov rax, rbx
    inc rax
    ret
pn_ok:
    xor eax, eax
    ret

parse_string:
    inc rbx
ps_loop:
    cmp rbx, r12
    jae ps_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '"'
    je ps_end
    cmp al, 0x5C
    je ps_escape
    cmp al, 0x20
    jb ps_err
    inc rbx
    jmp ps_loop
ps_end:
    inc rbx
    xor eax, eax
    ret
ps_escape:
    inc rbx
    cmp rbx, r12
    jae ps_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '"'
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, '/'
    je ps_esc_ok
    cmp al, 'b'
    je ps_esc_ok
    cmp al, 'f'
    je ps_esc_ok
    cmp al, 'n'
    je ps_esc_ok
    cmp al, 'r'
    je ps_esc_ok
    cmp al, 't'
    je ps_esc_ok
    cmp al, 'u'
    je ps_unicode
    jmp ps_err
ps_esc_ok:
    inc rbx
    jmp ps_loop
ps_unicode:
    inc rbx
    mov ecx, 4
ps_hex_loop:
    cmp rbx, r12
    jae ps_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '0'
    jb ps_err
    cmp al, '9'
    jbe ps_hex_ok
    cmp al, 'a'
    jb ps_hex_upper
    cmp al, 'f'
    jbe ps_hex_ok
    jmp ps_err
ps_hex_upper:
    cmp al, 'A'
    jb ps_err
    cmp al, 'F'
    jbe ps_hex_ok
    jmp ps_err
ps_hex_ok:
    inc rbx
    dec ecx
    jnz ps_hex_loop
    jmp ps_loop
ps_eof:
    mov rax, r12
    inc rax
    ret
ps_err:
    mov rax, rbx
    inc rax
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae pa_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, ']'
    je pa_close_empty
    call parse_value
    test rax, rax
    jnz pa_ret
pa_loop:
    call skip_ws
    cmp rbx, r12
    jae pa_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, ','
    je pa_comma
    cmp al, ']'
    je pa_close
    jmp pa_err
pa_comma:
    inc rbx
    call skip_ws
    call parse_value
    test rax, rax
    jnz pa_ret
    jmp pa_loop
pa_close:
    inc rbx
    xor eax, eax
    ret
pa_close_empty:
    inc rbx
    xor eax, eax
    ret
pa_eof:
    mov rax, r12
    inc rax
    ret
pa_err:
    mov rax, rbx
    inc rax
pa_ret:
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae po_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '}'
    je po_close_empty
po_member:
    cmp rbx, r12
    jae po_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, '"'
    jne po_err
    call parse_string
    test rax, rax
    jnz po_ret
    call skip_ws
    cmp rbx, r12
    jae po_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, ':'
    jne po_err
    inc rbx
    call skip_ws
    call parse_value
    test rax, rax
    jnz po_ret
    call skip_ws
    cmp rbx, r12
    jae po_eof
    movzx eax, byte ptr [r15 + rbx]
    cmp al, ','
    je po_comma
    cmp al, '}'
    je po_close
    jmp po_err
po_comma:
    inc rbx
    call skip_ws
    jmp po_member
po_close:
    inc rbx
    xor eax, eax
    ret
po_close_empty:
    inc rbx
    xor eax, eax
    ret
po_eof:
    mov rax, r12
    inc rax
    ret
po_err:
    mov rax, rbx
    inc rax
po_ret:
    ret
