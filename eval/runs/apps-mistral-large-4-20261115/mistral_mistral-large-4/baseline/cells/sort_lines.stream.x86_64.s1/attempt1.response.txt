.equ READ_CAP, 32769
.equ BUF_SIZE, 32770
.equ MAX_LINES, 1000

.text
.balign 16
.globl _start
_start:
    lea r15, [rip + input_buf]
    xor r14, r14
read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [r15 + r14]
    mov rdx, READ_CAP
    sub rdx, r14
    syscall
    cmp rax, 0
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r8, r8
    xor r9, r9
    lea r13, [rip + line_entries]
parse_loop:
    cmp r8, r14
    jae parse_done
    mov r10, r8
scan_nl:
    cmp r8, r14
    jae scan_end
    movzx eax, byte ptr [r15 + r8]
    inc r8
    cmp al, 10
    jne scan_nl
    lea r11, [r8 - 1]
    sub r11, r10
    jmp store_entry
scan_end:
    mov r11, r14
    sub r11, r10
store_entry:
    mov rax, r9
    shl rax, 3
    add rax, r13
    mov ecx, r10d
    mov [rax], ecx
    mov ecx, r11d
    mov [rax + 4], ecx
    inc r9
    jmp parse_loop
parse_done:
    lea rax, [rip + line_count]
    mov [rax], r9d
    cmp r9d, 1
    jbe skip_sort
    xor edi, edi
    mov rsi, r9
    dec rsi
    call qs
skip_sort:
    lea rbx, [rip + line_entries]
    lea rax, [rip + line_count]
    mov r13d, [rax]
    xor r12d, r12d
out_loop:
    cmp r12d, r13d
    jae out_done
    mov rax, r12
    shl rax, 3
    add rax, rbx
    mov ecx, [rax]
    mov edx, [rax + 4]
    lea rsi, [rip + input_buf]
    add rsi, rcx
    mov edi, 1
    call write_all
    lea rsi, [rip + newline]
    mov edi, 1
    mov edx, 1
    call write_all
    inc r12d
    jmp out_loop
out_done:
    mov eax, 60
    xor edi, edi
    syscall

compare:
    mov eax, [rdi]
    mov edx, [rsi]
    mov r8d, [rdi + 4]
    mov r9d, [rsi + 4]
    mov ecx, r8d
    cmp r9d, ecx
    cmovb ecx, r9d
    lea rsi, [rip + input_buf]
    add rsi, rax
    lea rdi, [rip + input_buf]
    add rdi, rdx
    test ecx, ecx
    jz cmp_len
    cld
    rep cmpsb
    jc cmp_neg
    jnz cmp_pos
cmp_len:
    cmp r8d, r9d
    jb cmp_neg
    ja cmp_pos
    xor eax, eax
    ret
cmp_neg:
    mov eax, -1
    ret
cmp_pos:
    mov eax, 1
    ret

write_all:
    test rdx, rdx
    jz wa_ret
wa_loop:
    mov eax, 1
    syscall
    test rax, rax
    jle wa_ret
    add rsi, rax
    sub rdx, rax
    jnz wa_loop
wa_ret:
    ret

qs:
    push rbx
    push rbp
    push r12
    push r13
    mov rbx, rdi
    mov rbp, rsi
    cmp rbx, rbp
    jae qs_ret
    mov rax, rbx
    add rax, rbp
    shr rax, 1
    lea rcx, [rip + line_entries]
    mov rdx, rax
    shl rdx, 3
    add rdx, rcx
    mov r8, rbp
    shl r8, 3
    add r8, rcx
    mov r9, [rdx]
    mov r10, [r8]
    mov [rdx], r10
    mov [r8], r9
    lea r12, [rbx - 1]
    mov r13, rbx
qs_loop:
    cmp r13, rbp
    jae qs_part_done
    lea rdi, [rip + line_entries]
    mov rax, r13
    shl rax, 3
    add rdi, rax
    lea rsi, [rip + line_entries]
    mov rax, rbp
    shl rax, 3
    add rsi, rax
    call compare
    test eax, eax
    jns qs_next
    inc r12
    cmp r12, r13
    je qs_no_swap
    lea rdi, [rip + line_entries]
    mov rax, r12
    shl rax, 3
    add rdi, rax
    lea rsi, [rip + line_entries]
    mov rax, r13
    shl rax, 3
    add rsi, rax
    mov rdx, [rdi]
    mov rcx, [rsi]
    mov [rdi], rcx
    mov [rsi], rdx
qs_no_swap:
qs_next:
    inc r13
    jmp qs_loop
qs_part_done:
    lea r12, [r12 + 1]
    cmp r12, rbp
    je qs_no_final_swap
    lea rdi, [rip + line_entries]
    mov rax, r12
    shl rax, 3
    add rdi, rax
    lea rsi, [rip + line_entries]
    mov rax, rbp
    shl rax, 3
    add rsi, rax
    mov rdx, [rdi]
    mov rcx, [rsi]
    mov [rdi], rcx
    mov [rsi], rdx
qs_no_final_swap:
    mov rdi, rbx
    lea rsi, [r12 - 1]
    cmp rdi, rsi
    jge qs_skip_left
    call qs
qs_skip_left:
    lea rdi, [r12 + 1]
    mov rsi, rbp
    cmp rdi, rsi
    jge qs_skip_right
    call qs
qs_skip_right:
qs_ret:
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.section .rodata
newline:
    .byte 10

.bss
.balign 8
input_buf:
    .zero BUF_SIZE
.balign 8
line_entries:
    .zero MAX_LINES * 8
line_count:
    .zero 4
