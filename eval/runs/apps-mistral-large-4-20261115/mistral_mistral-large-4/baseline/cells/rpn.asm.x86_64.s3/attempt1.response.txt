.globl _start
.equ INPUT_SIZE, 65536
.equ STACK_CAP, 32768

.section .rodata
err_msg:
    .ascii "error\n"

.bss
.align 8
input_buf:
    .zero INPUT_SIZE
.align 8
rpn_stack:
    .zero STACK_CAP*8
.align 8
num_buf:
    .zero 32

.text
_start:
    xor r12d, r12d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, INPUT_SIZE
    sub edx, r12d
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, INPUT_SIZE
    jb read_loop
read_done:
    lea rbp, [rip + rpn_stack]
    lea rsi, [rip + input_buf]
    lea rbx, [rip + input_buf]
    add rbx, r12
line_loop:
    cmp rsi, rbx
    je all_done
    mov r14, rsi
scan_nl:
    cmp rsi, rbx
    je final_line
    movzx eax, byte ptr [rsi]
    cmp al, 10
    je found_nl
    inc rsi
    jmp scan_nl
found_nl:
    mov r15, rsi
    inc rsi
    mov r10, rsi
    jmp eval_line
final_line:
    mov r15, rbx
    mov r10, rbx
eval_line:
    mov rsi, r14
    xor r12d, r12d
    xor r13d, r13d
token_loop:
    cmp rsi, r15
    jae end_line
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je sep
    cmp al, 0x09
    je sep
    mov rdi, rsi
tok_scan:
    cmp rsi, r15
    jae tok_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je tok_done
    cmp al, 0x09
    je tok_done
    inc rsi
    jmp tok_scan
tok_done:
    test r12d, r12d
    jnz token_loop
    mov rcx, rsi
    sub rcx, rdi
    cmp rcx, 1
    jne literal_start
    movzx eax, byte ptr [rdi]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
literal_start:
    mov r8, rdi
    xor edx, edx
    movzx eax, byte ptr [r8]
    cmp al, '-'
    jne parse_digits
    inc r8
    mov edx, 1
parse_digits:
    mov rcx, rsi
    sub rcx, r8
    test rcx, rcx
    jz invalid
    cmp rcx, 18
    ja invalid
    xor eax, eax
digit_loop:
    cmp r8, rsi
    jae literal_done
    movzx ecx, byte ptr [r8]
    sub cl, 48
    cmp cl, 9
    ja invalid
    imul rax, rax, 10
    add rax, rcx
    inc r8
    jmp digit_loop
literal_done:
    test edx, edx
    jz push_value
    neg rax
push_value:
    cmp r13, STACK_CAP
    jae invalid
    mov qword ptr [rbp + r13*8], rax
    inc r13
    jmp token_loop
sep:
    inc rsi
    jmp token_loop
invalid:
    mov r12d, 1
    jmp token_loop
op_add:
    cmp r13, 2
    jb invalid
    mov r9, qword ptr [rbp + r13*8 - 8]
    mov r8, qword ptr [rbp + r13*8 - 16]
    lea rax, [r8 + r9]
    sub r13, 2
    jmp push_result
op_sub:
    cmp r13, 2
    jb invalid
    mov r9, qword ptr [rbp + r13*8 - 8]
    mov r8, qword ptr [rbp + r13*8 - 16]
    mov rax, r8
    sub rax, r9
    sub r13, 2
    jmp push_result
op_mul:
    cmp r13, 2
    jb invalid
    mov r9, qword ptr [rbp + r13*8 - 8]
    mov r8, qword ptr [rbp + r13*8 - 16]
    imul rax, r8, r9
    sub r13, 2
    jmp push_result
op_div:
    cmp r13, 2
    jb invalid
    mov r9, qword ptr [rbp + r13*8 - 8]
    mov r8, qword ptr [rbp + r13*8 - 16]
    test r9, r9
    jz invalid
    mov rax, r8
    mov r11, 0x8000000000000000
    cmp rax, r11
    jne do_div
    cmp r9, -1
    je invalid
do_div:
    cqo
    idiv r9
    sub r13, 2
    jmp push_result
op_mod:
    cmp r13, 2
    jb invalid
    mov r9, qword ptr [rbp + r13*8 - 8]
    mov r8, qword ptr [rbp + r13*8 - 16]
    test r9, r9
    jz invalid
    mov rax, r8
    mov r11, 0x8000000000000000
    cmp rax, r11
    jne do_mod
    cmp r9, -1
    je invalid
do_mod:
    cqo
    idiv r9
    mov rax, rdx
    sub r13, 2
    jmp push_result
push_result:
    cmp r13, STACK_CAP
    jae invalid
    mov qword ptr [rbp + r13*8], rax
    inc r13
    jmp token_loop
end_line:
    test r12d, r12d
    jnz error_out
    cmp r13, 1
    jne error_out
    mov rax, qword ptr [rbp]
    call print_number
    jmp line_done
error_out:
    call print_error
line_done:
    mov rsi, r10
    jmp line_loop
all_done:
    mov eax, 60
    xor edi, edi
    syscall

print_number:
    lea rdi, [rip + num_buf + 32]
    mov byte ptr [rdi - 1], 10
    dec rdi
    mov r8, rax
    xor r9d, r9d
    test r8, r8
    jns pn_positive
    mov r9d, 1
    neg rax
pn_positive:
    mov ecx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pn_loop
    test r9d, r9d
    jz pn_no_sign
    dec rdi
    mov byte ptr [rdi], 45
pn_no_sign:
    mov rsi, rdi
    lea rdx, [rip + num_buf + 32]
    sub rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    ret

print_error:
    lea rsi, [rip + err_msg]
    mov edx, 6
    mov eax, 1
    mov edi, 1
    syscall
    ret
