.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFSIZE, 32768
.equ MAXLINES, 1000

.section .rodata
newline:
    .byte 0x0A

.bss
.balign 16
input_buf:
    .zero BUFSIZE
.balign 16
records:
    .zero MAXLINES*16
.balign 16
temp:
    .zero MAXLINES*16

.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0
read_loop:
    cmp x21, #BUFSIZE
    b.eq read_done
    mov x0, #0
    mov x1, x20
    add x1, x1, x21
    mov x2, #BUFSIZE
    sub x2, x2, x21
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop
read_done:
    adrp x29, records
    add x29, x29, :lo12:records
    mov x24, x29
    adrp x27, temp
    add x27, x27, :lo12:temp
    mov x22, x20
    add x23, x20, x21
    mov x25, #0
parse_loop:
    cmp x22, x23
    b.ge parse_done
    mov x26, x22
find_nl:
    cmp x26, x23
    b.ge no_nl
    ldrb w27, [x26]
    cmp w27, #0x0A
    b.eq found_nl
    add x26, x26, #1
    b find_nl
found_nl:
    lsl x28, x25, #4
    add x28, x29, x28
    stp x22, x26, [x28]
    add x25, x25, #1
    add x22, x26, #1
    b parse_loop
no_nl:
    lsl x28, x25, #4
    add x28, x29, x28
    stp x22, x23, [x28]
    add x25, x25, #1
parse_done:
    mov x19, #1
merge_outer:
    cmp x19, x25
    b.ge merge_done
    mov x20, #0
merge_left:
    cmp x20, x25
    b.ge merge_pass_done
    add x21, x20, x19
    cmp x21, x25
    b.le mid_ok
    mov x21, x25
mid_ok:
    add x22, x20, x19
    add x22, x22, x19
    cmp x22, x25
    b.le right_ok
    mov x22, x25
right_ok:
    mov x23, x20
    mov x26, x21
    mov x28, x20
merge_loop:
    cmp x23, x21
    b.ge copy_right_rem
    cmp x26, x22
    b.ge copy_left_rem
    lsl x4, x23, #4
    add x4, x24, x4
    ldp x0, x1, [x4]
    lsl x4, x26, #4
    add x4, x24, x4
    ldp x2, x3, [x4]
    bl compare
    cmp x0, #0
    b.le take_left
    lsl x4, x26, #4
    add x4, x24, x4
    ldp x6, x7, [x4]
    lsl x4, x28, #4
    add x4, x27, x4
    stp x6, x7, [x4]
    add x26, x26, #1
    add x28, x28, #1
    b merge_loop
take_left:
    lsl x4, x23, #4
    add x4, x24, x4
    ldp x6, x7, [x4]
    lsl x4, x28, #4
    add x4, x27, x4
    stp x6, x7, [x4]
    add x23, x23, #1
    add x28, x28, #1
    b merge_loop
copy_left_rem:
    cmp x23, x21
    b.ge merge_next_left
    lsl x4, x23, #4
    add x4, x24, x4
    ldp x6, x7, [x4]
    lsl x4, x28, #4
    add x4, x27, x4
    stp x6, x7, [x4]
    add x23, x23, #1
    add x28, x28, #1
    b copy_left_rem
copy_right_rem:
    cmp x26, x22
    b.ge merge_next_left
    lsl x4, x26, #4
    add x4, x24, x4
    ldp x6, x7, [x4]
    lsl x4, x28, #4
    add x4, x27, x4
    stp x6, x7, [x4]
    add x26, x26, #1
    add x28, x28, #1
    b copy_right_rem
merge_next_left:
    lsl x4, x19, #1
    add x20, x20, x4
    b merge_left
merge_pass_done:
    mov x4, x24
    mov x24, x27
    mov x27, x4
    lsl x19, x19, #1
    b merge_outer
merge_done:
    cmp x24, x29
    b.eq sort_done
    mov x19, #0
copy_records:
    cmp x19, x25
    b.ge sort_done
    lsl x20, x19, #4
    add x20, x24, x20
    ldp x21, x22, [x20]
    lsl x20, x19, #4
    add x20, x29, x20
    stp x21, x22, [x20]
    add x19, x19, #1
    b copy_records
sort_done:
    adrp x26, newline
    add x26, x26, :lo12:newline
    mov x19, #0
out_loop:
    cmp x19, x25
    b.ge out_done
    lsl x20, x19, #4
    add x20, x29, x20
    ldp x1, x2, [x20]
    sub x2, x2, x1
    cmp x2, #0
    b.eq out_newline
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
out_newline:
    mov x0, #1
    mov x1, x26
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    add x19, x19, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

compare:
1:
    cmp x0, x1
    b.eq 2f
    cmp x2, x3
    b.eq 3f
    ldrb w6, [x0], #1
    ldrb w7, [x2], #1
    cmp w6, w7
    b.lo 4f
    b.hi 5f
    b 1b
2:
    cmp x2, x3
    b.eq 6f
    mov x0, #-1
    ret
3:
    mov x0, #1
    ret
4:
    mov x0, #-1
    ret
5:
    mov x0, #1
    ret
6:
    mov x0, #0
    ret
