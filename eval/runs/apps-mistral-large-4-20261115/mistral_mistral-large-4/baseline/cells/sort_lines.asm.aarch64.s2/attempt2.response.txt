.section .rodata
newline: .byte 0x0A

.bss
.balign 8
input_buf: .zero 32768
.balign 16
records: .zero 16000

.text
.globl _start
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #32768
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x1, x1, x0
    subs x2, x2, x0
    b.ne read_loop
read_done:
    mov x3, #32768
    sub x3, x3, x2

    adrp x4, records
    add x4, x4, :lo12:records
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
    add x6, x5, x3
    mov x7, x5
    mov x9, #0
scan_loop:
    cmp x5, x6
    b.hs scan_done
    ldrb w8, [x5], #1
    cmp w8, #10
    b.ne scan_loop
    sub x10, x5, x7
    sub x10, x10, #1
    str x7, [x4]
    str x10, [x4, #8]
    add x4, x4, #16
    add x9, x9, #1
    mov x7, x5
    b scan_loop
scan_done:
    cmp x7, x6
    b.hs no_final
    sub x10, x6, x7
    str x7, [x4]
    str x10, [x4, #8]
    add x9, x9, #1
no_final:
    mov x19, x9
    adrp x20, records
    add x20, x20, :lo12:records

    cmp x19, #2
    b.lo sort_done
    mov x25, #1
i_loop:
    cmp x25, x19
    b.hs sort_done
    add x26, x20, x25, lsl #4
    ldr x21, [x26]
    ldr x22, [x26, #8]
    sub x24, x25, #1
j_loop:
    add x23, x20, x24, lsl #4
    ldr x0, [x23]
    ldr x1, [x23, #8]
    mov x2, x21
    mov x3, x22
    bl cmp_lines
    cmp x0, #0
    b.le j_insert
    ldr x10, [x23]
    ldr x11, [x23, #8]
    add x23, x23, #16
    str x10, [x23]
    str x11, [x23, #8]
    subs x24, x24, #1
    b.mi j_insert
    b j_loop
j_insert:
    add x23, x20, x24, lsl #4
    add x23, x23, #16
    str x21, [x23]
    str x22, [x23, #8]
    add x25, x25, #1
    b i_loop
sort_done:

    mov x25, #0
out_loop:
    cmp x25, x19
    b.hs out_done
    add x26, x20, x25, lsl #4
    ldr x1, [x26]
    ldr x2, [x26, #8]
    mov x0, #1
    bl write_all
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x0, #1
    bl write_all
    add x25, x25, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #93
    svc #0

cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, ls
    mov x5, #0
1:
    cmp x5, x4
    b.hs 2f
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.lo 3f
    b.hi 4f
    add x5, x5, #1
    b 1b
2:
    cmp x1, x3
    b.lo 3f
    b.hi 4f
    mov x0, #0
    ret
3:
    mov x0, #-1
    ret
4:
    mov x0, #1
    ret

write_all:
    cbz x2, 1f
2:
    mov x8, #64
    svc #0
    add x1, x1, x0
    subs x2, x2, x0
    b.ne 2b
1:
    ret
