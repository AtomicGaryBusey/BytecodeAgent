.section .rodata
newline: .byte 0x0A

.bss
.balign 8
input_buf: .zero 32768
.balign 16
records: .zero 16000

.text
.globl _start
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #32768
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x1, x1, x0
    subs x2, x2, x0
    b.ne read_loop
read_done:
    mov x3, #32768
    sub x3, x3, x2

    adrp x4, records
    add x4, x4, :lo12:records
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
    add x6, x5, x3
    mov x7, x5
    mov x9, #0
scan_loop:
    cmp x5, x6
    b.hs scan_done
    ldrb w8, [x5], #1
    cmp w8, #10
    b.ne scan_loop
    sub x10, x5, x7
    sub x10, x10, #1
    str x7, [x4]
    str x10, [x4, #8]
    add x4, x4, #16
    add x9, x9, #1
    mov x7, x5
    b scan_loop
scan_done:
    cmp x7, x6
    b.hs no_final_line
    sub x10, x6, x7
    str x7, [x4]
    str x10, [x4, #8]
    add x9, x9, #1
no_final_line:
    mov x19, x9
    adrp x20, records
    add x20, x20, :lo12:records

    cmp x19, #2
    b.lo sort_done
    mov x21, #1
i_loop:
    cmp x21, x19
    b.hs sort_done
    add x22, x20, x21, lsl #4
    ldp x23, x24, [x22]
    mov x25, x21
    sub x25, x25, #1
j_loop:
    add x26, x20, x25, lsl #4
    mov x0, x26
    mov x1, x23
    mov x2, x24
    bl cmp_lines
    cmp x0, #0
    b.le j_done
    ldp x10, x11, [x26]
    add x26, x26, #16
    stp x10, x11, [x26]
    subs x25, x25, #1
    b.mi j_done
    b j_loop
j_done:
    add x26, x20, x25, lsl #4
    add x26, x26, #16
    stp x23, x24, [x26]
    add x21, x21, #1
    b i_loop
sort_done:

    mov x21, #0
out_loop:
    cmp x21, x19
    b.hs out_done
    add x22, x20, x21, lsl #4
    ldr x1, [x22]
    ldr x2, [x22, #8]
    mov x0, #1
    bl write_all
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x0, #1
    bl write_all
    add x21, x21, #1
    b out_loop
out_done:
    mov x0, #0
    mov x8, #93
    svc #0

cmp_lines:
    ldr x3, [x0]
    ldr x4, [x0, #8]
    cmp x4, x2
    csel x5, x4, x2, ls
    mov x6, #0
cmp_loop:
    cmp x6, x5
    b.hs cmp_min_done
    ldrb w3, [x3, x6]
    ldrb w7, [x1, x6]
    cmp w3, w7
    b.lo cmp_less
    b.hi cmp_greater
    add x6, x6, #1
    b cmp_loop
cmp_min_done:
    cmp x4, x2
    b.lo cmp_less
    b.hi cmp_greater
    mov x0, #0
    ret
cmp_less:
    mov x0, #-1
    ret
cmp_greater:
    mov x0, #1
    ret

write_all:
    cbz x2, wa_done
wa_loop:
    mov x8, #64
    svc #0
    add x1, x1, x0
    subs x2, x2, x0
    b.ne wa_loop
wa_done:
    ret
