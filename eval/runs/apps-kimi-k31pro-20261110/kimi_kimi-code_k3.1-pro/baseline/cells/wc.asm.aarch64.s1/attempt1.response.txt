.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_done

    add x21, x21, x0
    mov x3, x0
    mov x4, x23

byte_loop:
    ldrb w5, [x4], #1

    cmp w5, #10
    b.eq newline
    cmp w5, #32
    b.eq whitespace
    cmp w5, #9
    b.eq whitespace
    cmp w5, #11
    b.eq whitespace
    cmp w5, #12
    b.eq whitespace
    cmp w5, #13
    b.eq whitespace

    cbnz x22, byte_done
    add x20, x20, #1
    mov x22, #1
    b byte_done

newline:
    add x19, x19, #1
whitespace:
    mov x22, #0

byte_done:
    subs x3, x3, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x1, x24

    mov x0, x19
    bl append_uint
    mov w0, #32
    strb w0, [x1], #1

    mov x0, x20
    bl append_uint
    mov w0, #32
    strb w0, [x1], #1

    mov x0, x21
    bl append_uint
    mov w0, #10
    strb w0, [x1], #1

    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

halt:
    b halt

append_uint:
    sub sp, sp, #32
    add x2, sp, #32
    mov x3, #10

    cbnz x0, 1f
    mov w4, #48
    sub x2, x2, #1
    strb w4, [x2]
    b 2f

1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    sub x2, x2, #1
    strb w5, [x2]
    mov x0, x4
    cbnz x0, 1b

2:
    add x6, sp, #32
3:
    cmp x2, x6
    b.hs 4f
    ldrb w7, [x2], #1
    strb w7, [x1], #1
    b 3b

4:
    add sp, sp, #32
    ret

.bss
.balign 16
input_buf:
    .zero 4096
output_buf:
    .zero 64
