.globl _start

.text
_start:
    lea rsi, [rip + input_buffer]
    xor r12d, r12d

read_more:
    cmp r12, 49152
    jae encode_start
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jz encode_start
    add r12, rax
    jmp read_more

encode_start:
    lea rbx, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea r9, [rip + base64_alphabet]
    xor r14d, r14d

encode_group:
    cmp r12, 3
    jb encode_tail

    movzx eax, byte ptr [rbx]
    movzx ecx, byte ptr [rbx + 1]
    movzx edx, byte ptr [rbx + 2]

    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [rdi], r8b

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, ecx
    shr r10d, 4
    or r8d, r10d
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [rdi + 1], r8b

    mov r8d, ecx
    and r8d, 15
    shl r8d, 2
    mov r10d, edx
    shr r10d, 6
    or r8d, r10d
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [rdi + 2], r8b

    and edx, 63
    movzx edx, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 3], dl

    add rbx, 3
    add rdi, 4
    sub r12, 3
    add r14d, 4
    cmp r14d, 76
    jne encode_group

    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
    jmp encode_group

encode_tail:
    test r12, r12
    jz no_partial_group
    cmp r12, 1
    je one_byte_left

    movzx eax, byte ptr [rbx]
    movzx ecx, byte ptr [rbx + 1]

    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [rdi], r8b

    mov r8d, eax
    and r8d, 3
    shl r8d, 4
    mov r10d, ecx
    shr r10d, 4
    or r8d, r10d
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [rdi + 1], r8b

    and ecx, 15
    shl ecx, 2
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi + 2], cl
    mov byte ptr [rdi + 3], 61

    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi
    jmp prepare_output

one_byte_left:
    movzx eax, byte ptr [rbx]

    mov r8d, eax
    shr r8d, 2
    movzx r8d, byte ptr [r9 + r8]
    mov byte ptr [rdi], r8b

    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [r9 + rax]
    mov byte ptr [rdi + 1], al
    mov word ptr [rdi + 2], 0x3d3d

    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi
    jmp prepare_output

no_partial_group:
    test r14d, r14d
    jz prepare_output
    mov byte ptr [rdi], 10
    inc rdi

prepare_output:
    lea rsi, [rip + output_buffer]
    mov r12, rdi
    sub r12, rsi
    jz exit_success

write_output:
    mov eax, 1
    mov edi, 1
    mov rdx, r12
    syscall
    add rsi, rax
    sub r12, rax
    jnz write_output

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
