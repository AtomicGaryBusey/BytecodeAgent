.text
.globl _start

_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    movz x23, #1, lsl #16

read_loop:
    cmp x19, x23
    b.hs read_done
    mov x0, #0
    add x1, x20, x19
    sub x2, x23, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, x20
    add x22, x20, x19

line_loop:
    cmp x21, x22
    b.hs program_exit
    mov x0, x21
    mov x2, x21

find_line_end:
    cmp x2, x22
    b.hs line_found
    ldrb w3, [x2]
    cmp w3, #10
    b.eq line_found
    add x2, x2, #1
    b find_line_end

line_found:
    mov x1, x2
    cmp x2, x22
    b.hs last_line
    add x21, x2, #1
    b evaluate_line
last_line:
    mov x21, x2
evaluate_line:
    bl process_line
    b line_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

process_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!

    mov x19, x0
    mov x20, x1
    mov x21, #0
    adrp x24, value_stack
    add x24, x24, :lo12:value_stack

token_loop:
    cmp x19, x20
    b.hs line_finished
    ldrb w0, [x19]
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator
    b token_begin

skip_separator:
    add x19, x19, #1
    b token_loop

token_begin:
    mov x22, x19

scan_token:
    cmp x19, x20
    b.hs token_scanned
    ldrb w0, [x19]
    cmp w0, #32
    b.eq token_scanned
    cmp w0, #9
    b.eq token_scanned
    add x19, x19, #1
    b scan_token

token_scanned:
    sub x23, x19, x22
    cmp x23, #1
    b.ne parse_literal
    ldrb w0, [x22]
    cmp w0, #43
    b.eq operator_token
    cmp w0, #45
    b.eq operator_token
    cmp w0, #42
    b.eq operator_token
    cmp w0, #47
    b.eq operator_token
    cmp w0, #37
    b.eq operator_token

parse_literal:
    mov x2, x22
    mov x3, x23
    mov x4, #0
    ldrb w0, [x2]
    cmp w0, #45
    b.ne literal_digits
    add x2, x2, #1
    sub x3, x3, #1
    mov x4, #1

literal_digits:
    cbz x3, line_error
    cmp x3, #18
    b.hi line_error
    mov x5, #0
    mov x6, #10

digit_loop:
    ldrb w0, [x2], #1
    sub w1, w0, #48
    cmp w1, #9
    b.hi line_error
    mul x5, x5, x6
    add x5, x5, x1
    subs x3, x3, #1
    b.ne digit_loop
    cbz x4, push_literal
    neg x5, x5

push_literal:
    str x5, [x24, x21, lsl #3]
    add x21, x21, #1
    b token_loop

operator_token:
    cmp x21, #2
    b.lo line_error
    sub x10, x21, #1
    sub x12, x21, #2
    ldr x11, [x24, x10, lsl #3]
    ldr x13, [x24, x12, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cmp w0, #47
    b.eq op_div
    b op_mod

op_add:
    add x14, x13, x11
    b op_store

op_sub:
    sub x14, x13, x11
    b op_store

op_mul:
    mul x14, x13, x11
    b op_store

op_div:
    cbz x11, line_error
    movz x15, #0x8000, lsl #48
    cmp x13, x15
    b.ne div_safe
    cmn x11, #1
    b.eq line_error
div_safe:
    sdiv x14, x13, x11
    b op_store

op_mod:
    cbz x11, line_error
    movz x15, #0x8000, lsl #48
    cmp x13, x15
    b.ne mod_safe
    cmn x11, #1
    b.eq line_error
mod_safe:
    sdiv x16, x13, x11
    msub x14, x16, x11, x13

op_store:
    str x14, [x24, x12, lsl #3]
    sub x21, x21, #1
    b token_loop

line_finished:
    cmp x21, #1
    b.ne line_error
    ldr x0, [x24]
    bl print_value
    b process_return

line_error:
    bl print_error

process_return:
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

print_value:
    adrp x1, number_buf
    add x1, x1, :lo12:number_buf
    add x2, x1, #32
    mov x3, x2
    mov x4, #0
    cmp x0, #0
    b.ge convert_setup
    mov x4, #1
    neg x0, x0

convert_setup:
    mov x5, #10

convert_loop:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    sub x3, x3, #1
    strb w7, [x3]
    mov x0, x6
    cbnz x0, convert_loop
    cbz x4, no_sign
    sub x3, x3, #1
    mov w6, #45
    strb w6, [x3]

no_sign:
    mov w6, #10
    strb w6, [x2]
    sub x2, x2, x3
    add x2, x2, #1
    mov x0, #1
    mov x1, x3
    mov x8, #64
    svc #0
    ret

print_error:
    mov x0, #1
    adrp x1, error_message
    add x1, x1, :lo12:error_message
    mov x2, #6
    mov x8, #64
    svc #0
    ret

.section .rodata
error_message:
    .ascii "error\n"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 8
value_stack:
    .zero 8192
number_buf:
    .zero 40
