.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_CAP, 65536

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #INPUT_CAP
    sub x2, x2, x20
    cbz x2, read_when_full
    mov x1, x19
    add x1, x1, x20
    b read_now
read_when_full:
    adrp x1, scratch_byte
    add x1, x1, :lo12:scratch_byte
    mov x2, #1
read_now:
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cbz x0, input_done
    mov x3, #INPUT_CAP
    cmp x20, x3
    b.hs read_loop
    add x20, x20, x0
    b read_loop

input_done:
    mov x21, #0
    bl skip_ws
    bl parse_value
    cbnz w0, show_error
    bl skip_ws
    cmp x21, x20
    b.ne show_error

    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
exit_hang:
    b exit_hang

show_error:
    adrp x1, out_buf
    add x1, x1, :lo12:out_buf
    mov x5, x1
    adrp x2, error_prefix
    add x2, x2, :lo12:error_prefix
    mov w3, #9
copy_prefix:
    ldrb w0, [x2], #1
    strb w0, [x1], #1
    subs w3, w3, #1
    b.ne copy_prefix

    mov x0, x21
    mov x4, x1
    mov x2, #10
    cbnz x0, decimal_loop
    mov w3, #48
    strb w3, [x1], #1
    b decimal_done
decimal_loop:
    udiv x6, x0, x2
    msub x7, x6, x2, x0
    add w7, w7, #48
    strb w7, [x1], #1
    mov x0, x6
    cbnz x0, decimal_loop
decimal_done:
    sub x6, x1, #1
reverse_digits:
    cmp x4, x6
    b.hs reverse_done
    ldrb w0, [x4]
    ldrb w2, [x6]
    strb w2, [x4]
    strb w0, [x6]
    add x4, x4, #1
    sub x6, x6, #1
    b reverse_digits
reverse_done:
    mov w0, #10
    strb w0, [x1], #1
    sub x2, x1, x5
    mov x0, #1
    mov x1, x5
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
error_exit_hang:
    b error_exit_hang

skip_ws:
    cmp x21, x20
    b.hs skip_ws_return
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq skip_ws_advance
    cmp w0, #0x09
    b.eq skip_ws_advance
    cmp w0, #0x0a
    b.eq skip_ws_advance
    cmp w0, #0x0d
    b.ne skip_ws_return
skip_ws_advance:
    add x21, x21, #1
    b skip_ws
skip_ws_return:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    bl skip_ws
    cmp x21, x20
    b.hs parse_value_error
    ldrb w0, [x19, x21]
    cmp w0, #123
    b.eq parse_value_object
    cmp w0, #91
    b.eq parse_value_array
    cmp w0, #34
    b.eq parse_value_string
    cmp w0, #116
    b.eq parse_value_true
    cmp w0, #102
    b.eq parse_value_false
    cmp w0, #110
    b.eq parse_value_null
    cmp w0, #45
    b.eq parse_value_number
    sub w1, w0, #48
    cmp w1, #9
    b.ls parse_value_number
    b parse_value_error
parse_value_object:
    bl parse_object
    b parse_value_return
parse_value_array:
    bl parse_array
    b parse_value_return
parse_value_string:
    bl parse_string
    b parse_value_return
parse_value_true:
    adrp x0, literal_true
    add x0, x0, :lo12:literal_true
    mov x1, #4
    bl parse_literal
    b parse_value_return
parse_value_false:
    adrp x0, literal_false
    add x0, x0, :lo12:literal_false
    mov x1, #5
    bl parse_literal
    b parse_value_return
parse_value_null:
    adrp x0, literal_null
    add x0, x0, :lo12:literal_null
    mov x1, #4
    bl parse_literal
    b parse_value_return
parse_value_number:
    bl parse_number
    b parse_value_return
parse_value_error:
    mov w0, #1
parse_value_return:
    ldp x29, x30, [sp], #16
    ret

parse_literal:
    cmp x21, x20
    b.hs parse_literal_error
    ldrb w2, [x19, x21]
    ldrb w3, [x0], #1
    cmp w2, w3
    b.ne parse_literal_error
    add x21, x21, #1
    subs x1, x1, #1
    b.ne parse_literal
    mov w0, #0
    ret
parse_literal_error:
    mov w0, #1
    ret

parse_string:
    add x21, x21, #1
parse_string_loop:
    cmp x21, x20
    b.hs parse_string_error
    ldrb w0, [x19, x21]
    cmp w0, #34
    b.eq parse_string_success
    cmp w0, #92
    b.eq parse_string_escape
    cmp w0, #0x20
    b.lo parse_string_error
    add x21, x21, #1
    b parse_string_loop
parse_string_success:
    add x21, x21, #1
    mov w0, #0
    ret
parse_string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_string_error
    ldrb w0, [x19, x21]
    cmp w0, #117
    b.eq parse_string_unicode
    cmp w0, #34
    b.eq parse_string_simple_escape
    cmp w0, #92
    b.eq parse_string_simple_escape
    cmp w0, #47
    b.eq parse_string_simple_escape
    cmp w0, #98
    b.eq parse_string_simple_escape
    cmp w0, #102
    b.eq parse_string_simple_escape
    cmp w0, #110
    b.eq parse_string_simple_escape
    cmp w0, #114
    b.eq parse_string_simple_escape
    cmp w0, #116
    b.ne parse_string_error
parse_string_simple_escape:
    add x21, x21, #1
    b parse_string_loop
parse_string_unicode:
    add x21, x21, #1
    mov w1, #4
parse_string_hex_loop:
    cmp x21, x20
    b.hs parse_string_error
    ldrb w0, [x19, x21]
    sub w2, w0, #48
    cmp w2, #9
    b.ls parse_string_hex_valid
    orr w2, w0, #0x20
    sub w2, w2, #97
    cmp w2, #5
    b.hi parse_string_error
parse_string_hex_valid:
    add x21, x21, #1
    subs w1, w1, #1
    b.ne parse_string_hex_loop
    b parse_string_loop
parse_string_error:
    mov w0, #1
    ret

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #45
    b.ne parse_number_integer
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_error
parse_number_integer:
    ldrb w0, [x19, x21]
    cmp w0, #48
    b.eq parse_number_zero
    sub w1, w0, #49
    cmp w1, #8
    b.hi parse_number_error
parse_number_nonzero:
    add x21, x21, #1
parse_number_integer_loop:
    cmp x21, x20
    b.hs parse_number_fraction_check
    ldrb w0, [x19, x21]
    sub w1, w0, #48
    cmp w1, #9
    b.hi parse_number_fraction_check
    add x21, x21, #1
    b parse_number_integer_loop
parse_number_zero:
    add x21, x21, #1
parse_number_fraction_check:
    cmp x21, x20
    b.hs parse_number_success
    ldrb w0, [x19, x21]
    cmp w0, #46
    b.ne parse_number_exponent_check
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_error
    ldrb w0, [x19, x21]
    sub w1, w0, #48
    cmp w1, #9
    b.hi parse_number_error
parse_number_fraction_loop:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_success
    ldrb w0, [x19, x21]
    sub w1, w0, #48
    cmp w1, #9
    b.ls parse_number_fraction_loop
parse_number_exponent_check:
    cmp x21, x20
    b.hs parse_number_success
    ldrb w0, [x19, x21]
    cmp w0, #101
    b.eq parse_number_exponent
    cmp w0, #69
    b.ne parse_number_success
parse_number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_error
    ldrb w0, [x19, x21]
    cmp w0, #43
    b.eq parse_number_exponent_sign
    cmp w0, #45
    b.ne parse_number_exponent_digit
parse_number_exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_error
parse_number_exponent_digit:
    ldrb w0, [x19, x21]
    sub w1, w0, #48
    cmp w1, #9
    b.hi parse_number_error
parse_number_exponent_loop:
    add x21, x21, #1
    cmp x21, x20
    b.hs parse_number_success
    ldrb w0, [x19, x21]
    sub w1, w0, #48
    cmp w1, #9
    b.ls parse_number_exponent_loop
parse_number_success:
    mov w0, #0
    ret
parse_number_error:
    mov w0, #1
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs parse_array_error
    ldrb w0, [x19, x21]
    cmp w0, #93
    b.eq parse_array_empty
parse_array_value:
    bl parse_value
    cbnz w0, parse_array_return
    bl skip_ws
    cmp x21, x20
    b.hs parse_array_error
    ldrb w0, [x19, x21]
    cmp w0, #93
    b.eq parse_array_close
    cmp w0, #44
    b.ne parse_array_error
    add x21, x21, #1
    b parse_array_value
parse_array_empty:
parse_array_close:
    add x21, x21, #1
    mov w0, #0
    b parse_array_return
parse_array_error:
    mov w0, #1
parse_array_return:
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.hs parse_object_error
    ldrb w0, [x19, x21]
    cmp w0, #125
    b.eq parse_object_empty
parse_object_member:
    cmp x21, x20
    b.hs parse_object_error
    ldrb w0, [x19, x21]
    cmp w0, #34
    b.ne parse_object_error
    bl parse_string
    cbnz w0, parse_object_return
    bl skip_ws
    cmp x21, x20
    b.hs parse_object_error
    ldrb w0, [x19, x21]
    cmp w0, #58
    b.ne parse_object_error
    add x21, x21, #1
    bl parse_value
    cbnz w0, parse_object_return
    bl skip_ws
    cmp x21, x20
    b.hs parse_object_error
    ldrb w0, [x19, x21]
    cmp w0, #125
    b.eq parse_object_close
    cmp w0, #44
    b.ne parse_object_error
    add x21, x21, #1
    bl skip_ws
    b parse_object_member
parse_object_empty:
parse_object_close:
    add x21, x21, #1
    mov w0, #0
    b parse_object_return
parse_object_error:
    mov w0, #1
parse_object_return:
    ldp x29, x30, [sp], #16
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
.balign 16
input_buf:
    .zero INPUT_CAP
out_buf:
    .zero 64
scratch_byte:
    .zero 1
