.globl _start

.text
_start:
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov x23, #0

    adrp x24, input_buffer
    add x24, x24, :lo12:input_buffer

read_loop:
    mov x0, #0
    mov x1, x24
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done
    tbnz x0, #63, input_done

    add x22, x22, x0
    mov x9, x0
    mov x10, x24

byte_loop:
    ldrb w11, [x10], #1

    cmp w11, #10
    b.ne check_space
    add x20, x20, #1

check_space:
    cmp w11, #32
    b.eq whitespace
    sub w12, w11, #9
    cmp w12, #4
    b.ls whitespace

    cbnz x23, next_byte
    add x21, x21, #1
    mov x23, #1
    b next_byte

whitespace:
    mov x23, #0

next_byte:
    subs x9, x9, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x19, output_buffer
    add x19, x19, :lo12:output_buffer
    adrp x26, number_buffer
    add x26, x26, :lo12:number_buffer

    mov x0, x20
    add x1, x26, #32
    bl u64_format
    mov x2, x1
    mov x3, x0
copy_lines:
    ldrb w4, [x3], #1
    strb w4, [x19], #1
    subs x2, x2, #1
    b.ne copy_lines
    mov w4, #32
    strb w4, [x19], #1

    mov x0, x21
    add x1, x26, #32
    bl u64_format
    mov x2, x1
    mov x3, x0
copy_words:
    ldrb w4, [x3], #1
    strb w4, [x19], #1
    subs x2, x2, #1
    b.ne copy_words
    mov w4, #32
    strb w4, [x19], #1

    mov x0, x22
    add x1, x26, #32
    bl u64_format
    mov x2, x1
    mov x3, x0
copy_bytes:
    ldrb w4, [x3], #1
    strb w4, [x19], #1
    subs x2, x2, #1
    b.ne copy_bytes
    mov w4, #10
    strb w4, [x19], #1

    adrp x1, output_buffer
    add x1, x1, :lo12:output_buffer
    sub x2, x19, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

u64_format:
    mov x2, x0
    mov x3, x1
    mov x4, #10
    cbnz x2, 1f

    sub x3, x3, #1
    mov w5, #48
    strb w5, [x3]
    b 2f

1:
    udiv x5, x2, x4
    msub x6, x5, x4, x2
    add w6, w6, #48
    sub x3, x3, #1
    strb w6, [x3]
    mov x2, x5
    cbnz x2, 1b

2:
    sub x1, x1, x3
    mov x0, x3
    ret

.section .bss
.balign 16
input_buffer:
    .zero 4096
number_buffer:
    .zero 32
output_buffer:
    .zero 80
