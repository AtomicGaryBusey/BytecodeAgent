.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, input_ready
    add x20, x20, x0
    b read_loop

input_ready:
    mov x21, #0
    bl skip_ws
    cmp x21, x20
    b.eq error_at_pos
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne error_at_pos
    b valid_document

skip_ws:
    cmp x21, x20
    b.eq skip_ws_done
    ldrb w0, [x19, x21]
    cmp w0, #32
    b.eq skip_ws_consume
    cmp w0, #9
    b.eq skip_ws_consume
    cmp w0, #10
    b.eq skip_ws_consume
    cmp w0, #13
    b.ne skip_ws_done
skip_ws_consume:
    add x21, x21, #1
    b skip_ws
skip_ws_done:
    ret

parse_value:
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #123
    b.eq parse_object
    cmp w0, #91
    b.eq parse_array
    cmp w0, #34
    b.eq parse_string
    cmp w0, #116
    b.eq value_true
    cmp w0, #102
    b.eq value_false
    cmp w0, #110
    b.eq value_null
    cmp w0, #45
    b.eq parse_number
    cmp w0, #48
    b.lo error_at_pos
    cmp w0, #57
    b.ls parse_number
    b error_at_pos

value_true:
    adr x1, literal_true
    mov x2, #4
    b parse_literal

value_false:
    adr x1, literal_false
    mov x2, #5
    b parse_literal

value_null:
    adr x1, literal_null
    mov x2, #4
    b parse_literal

parse_literal:
    mov x3, #0
literal_loop:
    cmp x3, x2
    b.eq literal_done
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    ldrb w4, [x1, x3]
    cmp w0, w4
    b.ne error_at_pos
    add x21, x21, #1
    add x3, x3, #1
    b literal_loop
literal_done:
    ret

parse_array:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #93
    b.eq array_close
    bl parse_value

array_next:
    bl skip_ws
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #93
    b.eq array_close
    cmp w0, #44
    b.ne error_at_pos
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    b array_next

array_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

parse_object:
    str x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #125
    b.eq object_close

object_member:
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #34
    b.ne error_at_pos
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #58
    b.ne error_at_pos
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #125
    b.eq object_close
    cmp w0, #44
    b.ne error_at_pos
    add x21, x21, #1
    bl skip_ws
    b object_member

object_close:
    add x21, x21, #1
    ldr x30, [sp], #16
    ret

parse_string:
    add x21, x21, #1

string_loop:
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #34
    b.eq string_close
    cmp w0, #92
    b.eq string_escape
    cmp w0, #32
    b.lo error_at_pos
    add x21, x21, #1
    b string_loop

string_close:
    add x21, x21, #1
    ret

string_escape:
    add x21, x21, #1
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #117
    b.eq string_unicode
    cmp w0, #34
    b.eq string_simple_escape
    cmp w0, #92
    b.eq string_simple_escape
    cmp w0, #47
    b.eq string_simple_escape
    cmp w0, #98
    b.eq string_simple_escape
    cmp w0, #102
    b.eq string_simple_escape
    cmp w0, #110
    b.eq string_simple_escape
    cmp w0, #114
    b.eq string_simple_escape
    cmp w0, #116
    b.ne error_at_pos

string_simple_escape:
    add x21, x21, #1
    b string_loop

string_unicode:
    add x21, x21, #1
    mov w3, #4

unicode_digit_loop:
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #48
    b.lo error_at_pos
    cmp w0, #57
    b.ls unicode_digit_valid
    cmp w0, #65
    b.lo error_at_pos
    cmp w0, #70
    b.ls unicode_digit_valid
    cmp w0, #97
    b.lo error_at_pos
    cmp w0, #102
    b.hi error_at_pos

unicode_digit_valid:
    add x21, x21, #1
    subs w3, w3, #1
    b.ne unicode_digit_loop
    b string_loop

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #45
    b.ne number_integer
    add x21, x21, #1
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]

number_integer:
    cmp w0, #48
    b.eq number_zero
    cmp w0, #49
    b.lo error_at_pos
    cmp w0, #57
    b.hi error_at_pos

number_integer_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #48
    b.lo number_after_integer
    cmp w0, #57
    b.ls number_integer_digits
    b number_after_integer

number_zero:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]

number_after_integer:
    cmp w0, #46
    b.ne number_exponent_check
    add x21, x21, #1
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #48
    b.lo error_at_pos
    cmp w0, #57
    b.hi error_at_pos

number_fraction_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #48
    b.lo number_exponent_check
    cmp w0, #57
    b.ls number_fraction_digits

number_exponent_check:
    cmp w0, #101
    b.eq number_exponent
    cmp w0, #69
    b.ne number_done

number_exponent:
    add x21, x21, #1
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]
    cmp w0, #43
    b.eq number_exponent_sign
    cmp w0, #45
    b.ne number_exponent_first_digit

number_exponent_sign:
    add x21, x21, #1
    cmp x21, x20
    b.eq error_at_pos
    ldrb w0, [x19, x21]

number_exponent_first_digit:
    cmp w0, #48
    b.lo error_at_pos
    cmp w0, #57
    b.hi error_at_pos

number_exponent_digits:
    add x21, x21, #1
    cmp x21, x20
    b.eq number_done
    ldrb w0, [x19, x21]
    cmp w0, #48
    b.lo number_done
    cmp w0, #57
    b.ls number_exponent_digits

number_done:
    ret

valid_document:
    adr x1, message_ok
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b .

error_at_pos:
    adrp x0, output_buf
    add x0, x0, :lo12:output_buf
    adr x1, error_prefix
    mov x2, #0

error_copy_prefix:
    cmp x2, #9
    b.eq error_convert_start
    ldrb w3, [x1, x2]
    strb w3, [x0, x2]
    add x2, x2, #1
    b error_copy_prefix

error_convert_start:
    mov x3, x21
    add x4, x0, #9
    mov x6, #10

error_digit_loop:
    udiv x7, x3, x6
    msub x8, x7, x6, x3
    add w8, w8, #48
    strb w8, [x4], #1
    mov x3, x7
    cbnz x3, error_digit_loop

    add x6, x0, #9
    sub x7, x4, #1

error_reverse_digits:
    cmp x6, x7
    b.hs error_append_newline
    ldrb w8, [x6]
    ldrb w9, [x7]
    strb w9, [x6], #1
    strb w8, [x7], #-1
    b error_reverse_digits

error_append_newline:
    mov w8, #10
    strb w8, [x4], #1
    sub x2, x4, x0
    mov x1, x0
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b .

.section .rodata

literal_false:
    .ascii "false"
literal_null:
    .ascii "null"
message_ok:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "

.bss

    .space 65537

    .space 32
