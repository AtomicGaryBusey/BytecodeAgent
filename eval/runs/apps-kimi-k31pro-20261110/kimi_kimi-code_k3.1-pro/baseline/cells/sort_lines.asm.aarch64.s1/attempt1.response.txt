.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_loop:
    cmp x20, #32768
    b.eq input_complete
    mov x0, #0
    add x1, x19, x20
    mov x2, #32768
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_complete
    add x20, x20, x0
    b read_loop

input_complete:
    cbz x20, exit_success

    adrp x21, line_table
    add x21, x21, :lo12:line_table
    mov x22, #0
    mov x23, #0
    mov x24, #0

parse_loop:
    cmp x23, x20
    b.hs parse_finish
    ldrb w0, [x19, x23]
    cmp w0, #10
    b.eq parse_newline
    add x23, x23, #1
    b parse_loop

parse_newline:
    add x3, x21, x22, lsl #3
    sub x2, x23, x24
    str w24, [x3]
    str w2, [x3, #4]
    add x22, x22, #1
    add x23, x23, #1
    mov x24, x23
    b parse_loop

parse_finish:
    cmp x24, x20
    b.hs lines_ready
    add x3, x21, x22, lsl #3
    sub x2, x20, x24
    str w24, [x3]
    str w2, [x3, #4]
    add x22, x22, #1

lines_ready:
    mov x0, x21
    adrp x1, merge_table
    add x1, x1, :lo12:merge_table
    mov x2, x22
    bl sort_table
    mov x21, x0

    mov x23, #0
output_loop:
    cmp x23, x22
    b.hs exit_success
    add x3, x21, x23, lsl #3
    ldr w2, [x3, #4]
    cbz w2, output_newline
    ldr w4, [x3]
    mov x0, #1
    add x1, x19, x4
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x2, #1
    mov x8, #64
    svc #0
    add x23, x23, #1
    b output_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

// x0 = first line descriptor, x1 = second line descriptor
// Return a negative, zero, or positive comparison result in w0.
compare_lines:
    ldr w2, [x0]
    ldr w3, [x0, #4]
    ldr w4, [x1]
    ldr w5, [x1, #4]
    adrp x6, input_buffer
    add x6, x6, :lo12:input_buffer
    add x2, x6, x2
    add x4, x6, x4
    cmp w3, w5
    csel w8, w3, w5, lo
    mov w7, #0

compare_bytes:
    cmp w7, w8
    b.hs compare_lengths
    ldrb w0, [x2, x7]
    ldrb w9, [x4, x7]
    subs w0, w0, w9
    b.ne compare_return
    add w7, w7, #1
    b compare_bytes

compare_lengths:
    cmp w3, w5
    b.lo compare_less
    b.hi compare_greater
    mov w0, #0
    ret

compare_less:
    mov w0, #-1
    ret

compare_greater:
    mov w0, #1

compare_return:
    ret

// Bottom-up stable merge sort of eight-byte offset/length descriptors.
// x0 = source table, x1 = scratch table, x2 = line count.
// Return the table containing the sorted descriptors in x0.
sort_table:
    stp x29, x30, [sp, #-96]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    stp x27, x28, [sp, #80]

    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x22, #1

sort_pass:
    cmp x22, x21
    b.hs sort_done
    mov x23, #0

sort_chunk:
    cmp x23, x21
    b.hs sort_pass_done

    add x24, x23, x22
    cmp x24, x21
    b.lo sort_have_middle
    mov x24, x21
sort_have_middle:
    add x25, x24, x22
    cmp x25, x21
    b.lo sort_have_right
    mov x25, x21
sort_have_right:
    mov x26, x23
    mov x27, x24
    mov x28, x23

sort_merge:
    cmp x26, x24
    b.hs sort_copy_right
    cmp x27, x25
    b.hs sort_copy_left
    add x0, x19, x26, lsl #3
    add x1, x19, x27, lsl #3
    bl compare_lines
    cmp w0, #0
    b.gt sort_take_right

sort_take_left:
    ldr x2, [x19, x26, lsl #3]
    str x2, [x20, x28, lsl #3]
    add x26, x26, #1
    add x28, x28, #1
    b sort_merge

sort_take_right:
    ldr x2, [x19, x27, lsl #3]
    str x2, [x20, x28, lsl #3]
    add x27, x27, #1
    add x28, x28, #1
    b sort_merge

sort_copy_left:
    cmp x26, x24
    b.hs sort_chunk_done
    ldr x2, [x19, x26, lsl #3]
    str x2, [x20, x28, lsl #3]
    add x26, x26, #1
    add x28, x28, #1
    b sort_copy_left

sort_copy_right:
    cmp x27, x25
    b.hs sort_chunk_done
    ldr x2, [x19, x27, lsl #3]
    str x2, [x20, x28, lsl #3]
    add x27, x27, #1
    add x28, x28, #1
    b sort_copy_right

sort_chunk_done:
    mov x23, x25
    b sort_chunk

sort_pass_done:
    mov x0, x19
    mov x19, x20
    mov x20, x0
    lsl x22, x22, #1
    b sort_pass

sort_done:
    mov x0, x19
    ldp x27, x28, [sp, #80]
    ldp x25, x26, [sp, #64]
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #96
    ret

.section .rodata
newline_byte:
    .byte 10

.bss
.balign 16
input_buffer:
    .zero 32768
.balign 16
line_table:
    .zero 8000
.balign 16
merge_table:
    .zero 8000
