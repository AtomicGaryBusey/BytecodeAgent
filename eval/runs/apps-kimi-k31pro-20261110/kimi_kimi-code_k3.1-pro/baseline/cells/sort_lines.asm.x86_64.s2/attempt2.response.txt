.globl _start

.text
_start:
    xor r12d, r12d

read_input:
    mov edx, 32769
    sub rdx, r12
    jz parse_input
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall
    test rax, rax
    jle parse_input
    add r12, rax
    jmp read_input

parse_input:
    lea r8, [rip + input_buffer]
    lea r9, [rip + descriptors_a]
    xor ecx, ecx
    xor edx, edx
    xor r10d, r10d
    test r12, r12
    jz sort_setup

scan_input:
    cmp rcx, r12
    jae input_scanned
    cmp byte ptr [r8 + rcx], 10
    jne scan_next

    mov dword ptr [r9 + r10*8], edx
    mov eax, ecx
    sub eax, edx
    mov dword ptr [r9 + r10*8 + 4], eax
    inc r10
    lea rdx, [rcx + 1]

scan_next:
    inc rcx
    jmp scan_input

input_scanned:
    cmp rdx, r12
    je sort_setup
    mov dword ptr [r9 + r10*8], edx
    mov rax, r12
    sub rax, rdx
    mov dword ptr [r9 + r10*8 + 4], eax
    inc r10

sort_setup:
    mov r12, r10
    lea r13, [rip + descriptors_a]
    lea r14, [rip + descriptors_b]
    mov r15d, 1
    cmp r12, 2
    jb output_lines

sort_pass:
    cmp r15, r12
    jae output_lines
    xor ebx, ebx

sort_run:
    cmp rbx, r12
    jae pass_finished

    lea rbp, [rbx + r15]
    cmp rbp, r12
    cmova rbp, r12
    lea r11, [rbp + r15]
    cmp r11, r12
    cmova r11, r12

    mov rsi, rbx
    mov rdi, rbp
    mov rcx, rbx

merge_loop:
    cmp rsi, rbp
    jae copy_right
    cmp rdi, r11
    jae copy_left

    mov rax, qword ptr [r13 + rsi*8]
    mov rdx, qword ptr [r13 + rdi*8]
    mov r8d, eax
    mov r9d, edx
    shr rax, 32
    shr rdx, 32

    lea r10, [rip + input_buffer]
    add r8, r10
    add r9, r10

    mov r10, rax
    cmp r10, rdx
    cmova r10, rdx

compare_bytes:
    test r10, r10
    jz compare_lengths
    mov al, byte ptr [r8]
    cmp al, byte ptr [r9]
    jb take_left
    ja take_right
    inc r8
    inc r9
    dec r10
    jmp compare_bytes

compare_lengths:
    mov rax, qword ptr [r13 + rsi*8]
    shr rax, 32
    cmp rax, rdx
    jbe take_left

take_right:
    mov rax, qword ptr [r13 + rdi*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rdi
    inc rcx
    jmp merge_loop

take_left:
    mov rax, qword ptr [r13 + rsi*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rsi
    inc rcx
    jmp merge_loop

copy_left:
    cmp rsi, rbp
    jae run_finished
    mov rax, qword ptr [r13 + rsi*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rsi
    inc rcx
    jmp copy_left

copy_right:
    cmp rdi, r11
    jae run_finished
    mov rax, qword ptr [r13 + rdi*8]
    mov qword ptr [r14 + rcx*8], rax
    inc rdi
    inc rcx
    jmp copy_right

run_finished:
    mov rbx, r11
    jmp sort_run

pass_finished:
    xchg r13, r14
    shl r15, 1
    jmp sort_pass

output_lines:
    xor ebx, ebx

output_loop:
    cmp rbx, r12
    jae finish
    mov rax, qword ptr [r13 + rbx*8]
    mov rdx, rax
    shr rdx, 32
    mov eax, eax
    lea rsi, [rip + input_buffer]
    add rsi, rax
    test rdx, rdx
    jz output_newline
    mov eax, 1
    mov edi, 1
    syscall

output_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rbx
    jmp output_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 8
input_buffer:
    .zero 32769
descriptors_a:
    .zero 8000
descriptors_b:
    .zero 8000
