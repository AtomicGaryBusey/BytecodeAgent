.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12, 32768
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor ebx, ebx
    xor r13d, r13d
    lea r14, [rip + lines]
    lea r8, [rip + input_buf]

build_loop:
    cmp rbx, r12
    jae build_done
    mov rbp, rbx
scan_line:
    cmp rbx, r12
    jae have_line
    cmp byte ptr [r8 + rbx], 10
    je have_line
    inc rbx
    jmp scan_line

have_line:
    mov rax, r13
    shl rax, 4
    lea r9, [r8 + rbp]
    mov [r14 + rax], r9
    mov r10, rbx
    sub r10, rbp
    mov [r14 + rax + 8], r10
    inc r13
    cmp rbx, r12
    jae build_done
    inc rbx
    jmp build_loop

build_done:
    cmp r13, 1
    jbe output_start
    mov r15, 1

sort_outer:
    mov rax, r15
    shl rax, 4
    mov rbp, [r14 + rax]
    mov r12, [r14 + rax + 8]
    mov rbx, r15

sort_inner:
    test rbx, rbx
    jz place_key
    mov r10, rbx
    shl r10, 4
    lea r11, [r14 + r10]
    mov rdi, rbp
    mov rsi, r12
    mov rdx, [r11 - 16]
    mov rcx, [r11 - 8]
    call compare_lines
    test eax, eax
    jge place_key

    mov r10, rbx
    shl r10, 4
    mov rax, [r14 + r10 - 16]
    mov [r14 + r10], rax
    mov rax, [r14 + r10 - 8]
    mov [r14 + r10 + 8], rax
    dec rbx
    jmp sort_inner

place_key:
    mov rax, rbx
    shl rax, 4
    mov [r14 + rax], rbp
    mov [r14 + rax + 8], r12
    inc r15
    cmp r15, r13
    jb sort_outer

output_start:
    xor r15d, r15d
output_loop:
    cmp r15, r13
    jae program_exit
    mov rax, r15
    shl rax, 4
    mov rsi, [r14 + rax]
    mov rdx, [r14 + rax + 8]
    test rdx, rdx
    jz write_newline
    mov eax, 1
    mov edi, 1
    syscall
write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline_char]
    mov edx, 1
    syscall
    inc r15
    jmp output_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9d, r9d
compare_bytes:
    cmp r9, r8
    jae compare_lengths
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    cmp eax, r10d
    jb line_less
    ja line_greater
    inc r9
    jmp compare_bytes
compare_lengths:
    cmp rsi, rcx
    jb line_less
    ja line_greater
    xor eax, eax
    ret
line_less:
    mov eax, -1
    ret
line_greater:
    mov eax, 1
    ret

.section .rodata
newline_char:
    .byte 10

.bss
.align 16
input_buf:
    .space 32768
.align 16
lines:
    .space 16000
