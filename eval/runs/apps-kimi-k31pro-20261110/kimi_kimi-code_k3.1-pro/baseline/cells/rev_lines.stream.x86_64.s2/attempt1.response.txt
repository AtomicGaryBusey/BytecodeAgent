.globl _start

.text
_start:
    xor r12d, r12d
    lea r15, [rip + input_buffer]

read_more:
    mov rdx, 65536
    sub rdx, r12
    jz process_input
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    syscall
    test rax, rax
    jle process_input
    add r12, rax
    jmp read_more

process_input:
    xor r13d, r13d

next_line:
    cmp r13, r12
    jae exit_program
    mov r14, r13

find_line_end:
    cmp r14, r12
    jae final_line
    cmp byte ptr [r15 + r14], 10
    je terminated_line
    inc r14
    jmp find_line_end

terminated_line:
    lea rbx, [r14 + 1]
    jmp reverse_line

final_line:
    mov rbx, r14

reverse_line:
    lea rsi, [r15 + r13]
    lea rdi, [r15 + r14]
    cmp rsi, rdi
    jae write_line
    dec rdi

reverse_loop:
    cmp rsi, rdi
    jae write_line
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp reverse_loop

write_line:
    mov eax, 1
    mov edi, 1
    lea rsi, [r15 + r13]
    mov rdx, r14
    sub rdx, r13
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    mov r13, rbx
    jmp next_line

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buffer:
    .zero 65536
