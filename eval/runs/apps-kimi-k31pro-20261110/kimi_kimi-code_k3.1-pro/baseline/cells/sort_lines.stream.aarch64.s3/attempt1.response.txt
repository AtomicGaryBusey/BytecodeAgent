.globl _start
.text
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x25, lines
    add x25, x25, :lo12:lines
    mov x19, #0

read_loop:
    mov x2, #32768
    sub x2, x2, x19
    cbz x2, parse_start
    mov x0, #0
    add x1, x21, x19
    mov x8, #63
    svc #0
    cbz x0, parse_start
    add x19, x19, x0
    b read_loop

parse_start:
    mov x20, #0
    mov x26, x21
    add x22, x21, x19
    cmp x26, x22
    b.eq sort_start

parse_line:
    mov x27, x26

find_line_end:
    cmp x26, x22
    b.eq final_line
    ldrb w0, [x26]
    cmp w0, #10
    b.eq newline_found
    add x26, x26, #1
    b find_line_end

newline_found:
    add x1, x25, x20, lsl #4
    sub x2, x26, x27
    stp x27, x2, [x1]
    add x20, x20, #1
    add x26, x26, #1
    cmp x26, x22
    b.eq sort_start
    b parse_line

final_line:
    add x1, x25, x20, lsl #4
    sub x2, x22, x27
    stp x27, x2, [x1]
    add x20, x20, #1

sort_start:
    cmp x20, #1
    b.ls output_start
    mov x26, #1

sort_outer:
    cmp x26, x20
    b.hs output_start
    add x0, x25, x26, lsl #4
    ldp x23, x24, [x0]
    mov x27, x26

sort_inner:
    cbz x27, insert_line
    add x0, x25, x27, lsl #4
    sub x0, x0, #16
    ldr x1, [x0, #8]
    ldr x0, [x0]
    mov x2, x23
    mov x3, x24
    bl line_greater
    cbz w0, insert_line
    add x0, x25, x27, lsl #4
    ldp x2, x3, [x0, #-16]
    stp x2, x3, [x0]
    sub x27, x27, #1
    b sort_inner

insert_line:
    add x0, x25, x27, lsl #4
    stp x23, x24, [x0]
    add x26, x26, #1
    b sort_outer

line_greater:
    cmp x1, x3
    csel x4, x1, x3, lo
    cbz x4, compare_lengths

compare_bytes:
    ldrb w5, [x0], #1
    ldrb w6, [x2], #1
    cmp w5, w6
    b.hi greater
    b.lo not_greater
    subs x4, x4, #1
    b.ne compare_bytes

compare_lengths:
    cmp x1, x3
    cset w0, hi
    ret

greater:
    mov w0, #1
    ret

not_greater:
    mov w0, #0
    ret

output_start:
    mov x26, #0

output_loop:
    cmp x26, x20
    b.hs program_exit
    add x27, x25, x26, lsl #4
    ldr x1, [x27]
    ldr x2, [x27, #8]
    cbz x2, write_newline
    mov x0, #1
    mov x8, #64
    svc #0

write_newline:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    add x26, x26, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buf:
    .space 32768
.balign 16
lines:
    .space 16000
