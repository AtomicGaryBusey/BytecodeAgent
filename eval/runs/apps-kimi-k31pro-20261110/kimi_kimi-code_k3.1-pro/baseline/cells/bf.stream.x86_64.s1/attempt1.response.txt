.globl _start

.text
_start:
    xor ebx, ebx

read_more:
    cmp rbx, 65536
    jae input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle input_ready
    add rbx, rax
    jmp read_more

input_ready:
    lea rsi, [rip + input_buffer]
    xor ecx, ecx

find_separator:
    cmp rcx, rbx
    jae no_separator
    cmp byte ptr [rsi + rcx], 33
    je separator_found
    inc rcx
    jmp find_separator

separator_found:
    mov r12, rcx
    lea r14, [rsi + rcx + 1]
    lea r15, [rsi + rbx]
    jmp validate_start

no_separator:
    mov r12, rbx
    lea r14, [rsi + rbx]
    mov r15, r14

validate_start:
    lea r8, [rip + bracket_map]
    lea r9, [rip + bracket_stack]
    xor ecx, ecx
    xor edx, edx

validate_loop:
    cmp rcx, r12
    jae validate_done
    mov al, byte ptr [rsi + rcx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close

validate_next:
    inc rcx
    jmp validate_loop

validate_open:
    cmp edx, 1000
    jae invalid_program
    mov dword ptr [r9 + rdx*4], ecx
    inc edx
    jmp validate_next

validate_close:
    test edx, edx
    jz invalid_program
    dec edx
    mov eax, dword ptr [r9 + rdx*4]
    mov dword ptr [r8 + rcx*4], eax
    mov dword ptr [r8 + rax*4], ecx
    jmp validate_next

validate_done:
    test edx, edx
    jnz invalid_program

    lea rbx, [rip + tape]
    lea rbp, [rip + output_buffer]
    xor ecx, ecx

execute_loop:
    cmp rcx, r12
    jae execution_done
    movzx eax, byte ptr [rsi + rcx]
    inc rcx

    cmp al, 62
    je cmd_right
    cmp al, 60
    je cmd_left
    cmp al, 43
    je cmd_increment
    cmp al, 45
    je cmd_decrement
    cmp al, 46
    je cmd_output
    cmp al, 44
    je cmd_input
    cmp al, 91
    je cmd_loop_start
    cmp al, 93
    je cmd_loop_end
    jmp execute_loop

cmd_right:
    inc rbx
    jmp execute_loop

cmd_left:
    dec rbx
    jmp execute_loop

cmd_increment:
    inc byte ptr [rbx]
    jmp execute_loop

cmd_decrement:
    dec byte ptr [rbx]
    jmp execute_loop

cmd_output:
    mov al, byte ptr [rbx]
    mov byte ptr [rbp], al
    inc rbp
    jmp execute_loop

cmd_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbx], al
    jmp execute_loop

input_exhausted:
    mov byte ptr [rbx], 0
    jmp execute_loop

cmd_loop_start:
    cmp byte ptr [rbx], 0
    jne execute_loop
    lea rdx, [rcx - 1]
    mov eax, dword ptr [r8 + rdx*4]
    lea rcx, [rax + 1]
    jmp execute_loop

cmd_loop_end:
    cmp byte ptr [rbx], 0
    je execute_loop
    lea rdx, [rcx - 1]
    mov eax, dword ptr [r8 + rdx*4]
    lea rcx, [rax + 1]
    jmp execute_loop

execution_done:
    lea rsi, [rip + output_buffer]
    mov rdx, rbp
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536

.balign 16
bracket_map:
    .zero 262144

.balign 16
bracket_stack:
    .zero 4000

.balign 16
tape:
    .zero 30000

.balign 16
output_buffer:
    .zero 65536
