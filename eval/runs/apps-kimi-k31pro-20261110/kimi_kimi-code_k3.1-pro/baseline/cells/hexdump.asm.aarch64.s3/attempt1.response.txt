.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    beq read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs dump_done

    sub x22, x20, x21
    cmp x22, #16
    b.ls have_row_length
    mov x22, #16
have_row_length:
    mov x1, x23
    mov w0, w21
    bl put_hex8

    mov w4, #32
    strb w4, [x1], #1
    strb w4, [x1], #1
    mov x24, #0

row_slots:
    cmp x24, #16
    b.eq row_slots_done
    cmp x24, x22
    b.hs blank_slot

    add x5, x19, x21
    ldrb w6, [x5, x24]
    adrp x8, hex_digits
    add x8, x8, :lo12:hex_digits
    lsr w7, w6, #4
    ldrb w7, [x8, x7]
    strb w7, [x1], #1
    and w7, w6, #15
    ldrb w7, [x8, x7]
    strb w7, [x1], #1
    mov w7, #32
    strb w7, [x1], #1
    b slot_spacing

blank_slot:
    mov w7, #32
    strb w7, [x1], #1
    strb w7, [x1], #1
    strb w7, [x1], #1

slot_spacing:
    cmp x24, #7
    b.ne no_extra_space
    mov w7, #32
    strb w7, [x1], #1
no_extra_space:
    add x24, x24, #1
    b row_slots

row_slots_done:
    mov w4, #32
    strb w4, [x1], #1
    mov w4, #124
    strb w4, [x1], #1

    mov x24, #0
ascii_loop:
    cmp x24, x22
    b.eq ascii_done
    add x5, x19, x21
    ldrb w6, [x5, x24]
    cmp w6, #32
    b.lo ascii_dot
    cmp w6, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w6, #46
ascii_store:
    strb w6, [x1], #1
    add x24, x24, #1
    b ascii_loop

ascii_done:
    mov w4, #124
    strb w4, [x1], #1
    mov w4, #10
    strb w4, [x1], #1

    sub x2, x1, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0

    add x21, x21, x22
    b row_loop

dump_done:
    mov x1, x23
    mov w0, w20
    bl put_hex8
    mov w4, #10
    strb w4, [x1], #1
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

put_hex8:
    adrp x2, hex_digits
    add x2, x2, :lo12:hex_digits
    mov x3, #28
put_hex8_loop:
    lsr x4, x0, x3
    and x4, x4, #15
    ldrb w5, [x2, x4]
    strb w5, [x1], #1
    subs x3, x3, #4
    b.ge put_hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space 65536
line_buf:
    .space 80
