.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + input_buffer]
    lea r13, [rbx + r12]
    lea r14, [rip + output_buffer]

line_loop:
    cmp rbx, r13
    jae write_output
    mov r15, rbx

scan_line:
    cmp rbx, r13
    jae reverse_line
    cmp byte ptr [rbx], 10
    je reverse_line
    inc rbx
    jmp scan_line

reverse_line:
    mov rcx, rbx
    sub rcx, r15
    lea rsi, [rbx - 1]
    test rcx, rcx
    jz line_reversed

copy_reversed:
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc r14
    dec rsi
    dec rcx
    jnz copy_reversed

line_reversed:
    mov byte ptr [r14], 10
    inc r14
    cmp rbx, r13
    jae write_output
    inc rbx
    jmp line_loop

write_output:
    lea rdx, [rip + output_buffer]
    sub r14, rdx
    mov rdx, r14
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
