.text
.globl _start
_start:
    xor r12d, r12d

read_more:
    mov edx, 49153
    sub rdx, r12
    jz input_done
    lea rsi, [rip + input_buffer]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    lea rbx, [rip + input_buffer]
    lea r13, [rip + output_buffer]
    lea r8, [rip + base64_alphabet]
    xor r14d, r14d

encode_full:
    cmp r12, 3
    jb encode_tail

    movzx eax, byte ptr [rbx]
    shl eax, 16
    movzx edx, byte ptr [rbx + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rbx + 2]
    or eax, edx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13 + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13 + 2], dl

    mov edx, eax
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13 + 3], dl

    add rbx, 3
    sub r12, 3
    add r13, 4
    add r14d, 4
    cmp r14d, 76
    jne encode_full

    mov byte ptr [r13], 10
    inc r13
    xor r14d, r14d
    jmp encode_full

encode_tail:
    test r12, r12
    jnz encode_partial
    test r14d, r14d
    jz write_output
    mov byte ptr [r13], 10
    inc r13
    jmp write_output

encode_partial:
    movzx eax, byte ptr [rbx]
    shl eax, 16
    xor edx, edx
    cmp r12, 1
    je partial_value_ready
    movzx edx, byte ptr [rbx + 1]
    shl edx, 8

partial_value_ready:
    or eax, edx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13 + 1], dl

    cmp r12, 1
    je partial_one_byte

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r13 + 2], dl
    mov byte ptr [r13 + 3], 61
    jmp partial_finish

partial_one_byte:
    mov byte ptr [r13 + 2], 61
    mov byte ptr [r13 + 3], 61

partial_finish:
    add r13, 4
    mov byte ptr [r13], 10
    inc r13

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r13
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_success

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .zero 49153
.balign 16
output_buffer:
    .zero 66560
