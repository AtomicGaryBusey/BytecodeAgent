.globl _start

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf

    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x21, x19, x20
    mov x22, x19

line_loop:
    cmp x22, x21
    b.hs finished
    mov x24, x22

scan_line:
    cmp x22, x21
    b.hs reverse_line
    ldrb w0, [x22]
    cmp w0, #10
    b.eq reverse_line
    add x22, x22, #1
    b scan_line

reverse_line:
    sub x25, x22, x24
    mov x0, x23
    mov x1, x22
    mov x2, x25
    cbz x2, append_newline

1:
    sub x1, x1, #1
    ldrb w3, [x1]
    strb w3, [x0], #1
    subs x2, x2, #1
    b.ne 1b

append_newline:
    mov w3, #10
    strb w3, [x0]

    mov x0, #1
    mov x1, x23
    add x2, x25, #1
    mov x8, #64
    svc #0

    cmp x22, x21
    b.hs line_loop
    add x22, x22, #1
    b line_loop

finished:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 65537
