.text
.globl _start

_start:
    mov x19, #0
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf

read_loop:
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, input_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    b read_loop

input_done:
    mov x21, x19
    mov x22, x19
    mov x0, #0

find_separator:
    cmp x0, x19
    b.hs validate_setup
    ldrb w1, [x20, x0]
    cmp w1, #0x21
    beq separator_found
    add x0, x0, #1
    b find_separator

separator_found:
    mov x21, x0
    add x22, x0, #1

validate_setup:
    adrp x23, jump_table
    add x23, x23, :lo12:jump_table
    adrp x24, bracket_stack
    add x24, x24, :lo12:bracket_stack
    mov x25, #0
    mov x26, #0

validate_loop:
    cmp x26, x21
    b.hs validate_end
    ldrb w0, [x20, x26]
    cmp w0, #0x5b
    beq validate_open
    cmp w0, #0x5d
    beq validate_close

validate_next:
    add x26, x26, #1
    b validate_loop

validate_open:
    cmp x25, #1000
    b.hs invalid_program
    str w26, [x24, x25, lsl #2]
    add x25, x25, #1
    b validate_next

validate_close:
    cbz x25, invalid_program
    sub x25, x25, #1
    ldr w0, [x24, x25, lsl #2]
    add w1, w26, #1
    str w1, [x23, x0, lsl #2]
    add w0, w0, #1
    str w0, [x23, x26, lsl #2]
    b validate_next

validate_end:
    cbnz x25, invalid_program

    adrp x24, tape
    add x24, x24, :lo12:tape
    mov x25, #0
    mov x26, #0

run_loop:
    cmp x26, x21
    b.hs success
    ldrb w0, [x20, x26]
    cmp w0, #0x3e
    beq cmd_right
    cmp w0, #0x3c
    beq cmd_left
    cmp w0, #0x2b
    beq cmd_plus
    cmp w0, #0x2d
    beq cmd_minus
    cmp w0, #0x2e
    beq cmd_output
    cmp w0, #0x2c
    beq cmd_input
    cmp w0, #0x5b
    beq cmd_open
    cmp w0, #0x5d
    beq cmd_close

advance:
    add x26, x26, #1
    b run_loop

cmd_right:
    add x25, x25, #1
    b advance

cmd_left:
    sub x25, x25, #1
    b advance

cmd_plus:
    add x1, x24, x25
    ldrb w0, [x1]
    add w0, w0, #1
    strb w0, [x1]
    b advance

cmd_minus:
    add x1, x24, x25
    ldrb w0, [x1]
    sub w0, w0, #1
    strb w0, [x1]
    b advance

cmd_output:
    add x1, x24, x25
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    b advance

cmd_input:
    cmp x22, x19
    b.hs input_exhausted
    ldrb w0, [x20, x22]
    add x22, x22, #1
    b store_input

input_exhausted:
    mov w0, #0

store_input:
    strb w0, [x24, x25]
    b advance

cmd_open:
    ldrb w0, [x24, x25]
    cbnz w0, advance
    ldr w26, [x23, x26, lsl #2]
    b run_loop

cmd_close:
    ldrb w0, [x24, x25]
    cbz w0, advance
    ldr w26, [x23, x26, lsl #2]
    b run_loop

success:
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 4
input_buf:
    .space 65536

.balign 4
jump_table:
    .space 262144

.balign 4
bracket_stack:
    .space 4000

.balign 16
tape:
    .space 30000
