.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

row_loop:
    cmp x21, x20
    b.eq final_offset
    sub x22, x20, x21
    cmp x22, #16
    b.ls row_size_ready
    mov x22, #16

row_size_ready:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov w0, #32
    mov x1, #61
    mov x2, x23

space_loop:
    strb w0, [x2], #1
    subs x1, x1, #1
    b.ne space_loop

    mov x0, x21
    add x1, x23, #7
    adrp x2, hex_digits
    add x2, x2, :lo12:hex_digits
    mov x3, #8

offset_loop:
    and x4, x0, #0xf
    ldrb w4, [x2, x4]
    strb w4, [x1], #-1
    lsr x0, x0, #4
    subs x3, x3, #1
    b.ne offset_loop

    add x24, x23, #10
    mov x25, #0

hex_loop:
    cmp x25, #16
    b.eq ascii_start
    cmp x25, x22
    b.hs hex_slot_done
    add x0, x19, x21
    ldrb w0, [x0, x25]
    lsr w3, w0, #4
    and w4, w0, #0xf
    ldrb w3, [x2, x3]
    ldrb w4, [x2, x4]
    strb w3, [x24]
    strb w4, [x24, #1]

hex_slot_done:
    add x24, x24, #3
    cmp x25, #7
    b.ne no_extra_space
    add x24, x24, #1

no_extra_space:
    add x25, x25, #1
    b hex_loop

ascii_start:
    mov w0, #124
    strb w0, [x23, #60]
    add x24, x23, #61
    mov x25, #0

ascii_loop:
    cmp x25, x22
    b.eq ascii_done
    add x0, x19, x21
    ldrb w0, [x0, x25]
    cmp w0, #0x20
    b.lo ascii_dot
    cmp w0, #0x7e
    b.hi ascii_dot
    b ascii_store

ascii_dot:
    mov w0, #46

ascii_store:
    strb w0, [x24], #1
    add x25, x25, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1
    sub x2, x24, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0
    add x21, x21, x22
    b row_loop

final_offset:
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    mov x0, x20
    add x1, x23, #7
    adrp x2, hex_digits
    add x2, x2, :lo12:hex_digits
    mov x3, #8

final_hex_loop:
    and x4, x0, #0xf
    ldrb w4, [x2, x4]
    strb w4, [x1], #-1
    lsr x0, x0, #4
    subs x3, x3, #1
    b.ne final_hex_loop

    mov w0, #10
    strb w0, [x23, #8]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

exit_wait:
    b exit_wait

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
line_buf:
    .zero 80
