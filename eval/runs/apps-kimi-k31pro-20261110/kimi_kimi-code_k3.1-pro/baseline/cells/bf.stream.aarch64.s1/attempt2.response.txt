.text
.globl _start

_start:
    adr x20, input_buffer
    mov x19, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    add x21, x20, x19
    mov x9, x20

find_bang:
    cmp x9, x21
    b.eq no_bang
    ldrb w10, [x9]
    cmp w10, #33
    b.eq bang_found
    add x9, x9, #1
    b find_bang

bang_found:
    mov x22, x9
    add x23, x9, #1
    b input_bounds_ready

no_bang:
    mov x22, x21
    mov x23, x21

input_bounds_ready:
    mov x24, x21
    adr x19, program_code
    adr x25, bracket_stack
    adr x26, jump_table
    mov x27, #0
    mov x28, #0
    mov x9, x20

preprocess_loop:
    cmp x9, x22
    b.eq preprocess_done
    ldrb w10, [x9], #1

    cmp w10, #62
    b.eq make_right
    cmp w10, #60
    b.eq make_left
    cmp w10, #43
    b.eq make_plus
    cmp w10, #45
    b.eq make_minus
    cmp w10, #46
    b.eq make_output
    cmp w10, #44
    b.eq make_input
    cmp w10, #91
    b.eq make_open
    cmp w10, #93
    b.eq make_close
    b preprocess_loop

make_right:
    mov w11, #1
    b store_command

make_left:
    mov w11, #2
    b store_command

make_plus:
    mov w11, #3
    b store_command

make_minus:
    mov w11, #4
    b store_command

make_output:
    mov w11, #5
    b store_command

make_input:
    mov w11, #6
    b store_command

make_open:
    cmp x28, #1000
    b.hs bf_invalid
    str w27, [x25, x28, lsl #2]
    add x28, x28, #1
    mov w11, #7
    b store_command

make_close:
    cbz x28, bf_invalid
    sub x28, x28, #1
    ldr w12, [x25, x28, lsl #2]
    str w12, [x26, x27, lsl #2]
    str w27, [x26, x12, lsl #2]
    mov w11, #8

store_command:
    strb w11, [x19, x27]
    add x27, x27, #1
    b preprocess_loop

preprocess_done:
    cbnz x28, bf_invalid
    strb wzr, [x19, x27]

    mov x21, x23
    mov x22, x24
    mov x23, x26
    adr x24, output_buffer
    mov x28, x24
    adr x20, tape
    mov x26, x20
    mov x27, x19
    adr x25, dispatch_table

bf_dispatch:
    ldrb w9, [x27], #1
    cbz w9, bf_done
    ldr x10, [x25, x9, lsl #3]
    br x10

op_right:
    add x26, x26, #1
    b bf_dispatch

op_left:
    sub x26, x26, #1
    b bf_dispatch

op_plus:
    ldrb w9, [x26]
    add w9, w9, #1
    strb w9, [x26]
    b bf_dispatch

op_minus:
    ldrb w9, [x26]
    sub w9, w9, #1
    strb w9, [x26]
    b bf_dispatch

op_output:
    ldrb w9, [x26]
    strb w9, [x28], #1
    b bf_dispatch

op_input:
    cmp x21, x22
    b.eq input_exhausted
    ldrb w9, [x21], #1
    strb w9, [x26]
    b bf_dispatch

input_exhausted:
    strb wzr, [x26]
    b bf_dispatch

op_open:
    ldrb w9, [x26]
    cbnz w9, bf_dispatch
    sub x11, x27, x19
    sub x11, x11, #1
    ldr w11, [x23, x11, lsl #2]
    add x27, x19, x11
    add x27, x27, #1
    b bf_dispatch

op_close:
    ldrb w9, [x26]
    cbz w9, bf_dispatch
    sub x11, x27, x19
    sub x11, x11, #1
    ldr w11, [x23, x11, lsl #2]
    add x27, x19, x11
    add x27, x27, #1
    b bf_dispatch

bf_done:
    sub x2, x28, x24
    cbz x2, exit_success
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

bf_invalid:
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
.balign 8
dispatch_table:
    .quad bf_done
    .quad op_right
    .quad op_left
    .quad op_plus
    .quad op_minus
    .quad op_output
    .quad op_input
    .quad op_open
    .quad op_close

.bss
.balign 16

    .zero 65536

.balign 16
program_code:
    .zero 65537

.balign 16
jump_table:
    .zero 262148

.balign 16
bracket_stack:
    .zero 4000

.balign 16
tape:
    .zero 30000

.balign 16

    .zero 65536
