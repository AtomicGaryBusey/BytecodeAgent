.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + input_buffer]
    lea rbp, [rbx + r12]
    mov r15, rbp
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rbx + rcx], 0x21
    je separator_found
    inc rcx
    jmp find_separator

separator_found:
    lea r14, [rbx + rcx]
    lea r15, [r14 + 1]
    jmp filter_start

no_separator:
    mov r14, rbp

filter_start:
    lea r13, [rip + program]
    xor r12d, r12d

filter_loop:
    cmp rbx, r14
    jae validate_start
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 0x3e
    je filter_keep
    cmp al, 0x3c
    je filter_keep
    cmp al, 0x2b
    je filter_keep
    cmp al, 0x2d
    je filter_keep
    cmp al, 0x2e
    je filter_keep
    cmp al, 0x2c
    je filter_keep
    cmp al, 0x5b
    je filter_keep
    cmp al, 0x5d
    jne filter_loop

filter_keep:
    mov byte ptr [r13 + r12], al
    inc r12
    jmp filter_loop

validate_start:
    xor r14d, r14d
    xor ebx, ebx
    lea r8, [rip + jumps]
    lea r9, [rip + bracket_stack]

validate_loop:
    cmp r14, r12
    jae validate_done
    movzx eax, byte ptr [r13 + r14]
    cmp al, 0x5b
    je validate_open
    cmp al, 0x5d
    je validate_close

validate_next:
    inc r14
    jmp validate_loop

validate_open:
    cmp ebx, 1000
    jae invalid
    mov dword ptr [r9 + rbx*4], r14d
    inc ebx
    jmp validate_next

validate_close:
    test ebx, ebx
    jz invalid
    dec ebx
    mov eax, dword ptr [r9 + rbx*4]
    lea edx, [rax + 1]
    mov dword ptr [r8 + r14*4], edx
    lea edx, [r14 + 1]
    mov dword ptr [r8 + rax*4], edx
    jmp validate_next

validate_done:
    test ebx, ebx
    jnz invalid

    lea rbx, [rip + tape]
    xor r14d, r14d
    xor r10d, r10d
    lea r9, [rip + output_buffer]

execute_loop:
    cmp r14, r12
    jae execute_done
    movzx eax, byte ptr [r13 + r14]
    cmp al, 0x3e
    je command_right
    cmp al, 0x3c
    je command_left
    cmp al, 0x2b
    je command_increment
    cmp al, 0x2d
    je command_decrement
    cmp al, 0x2e
    je command_output
    cmp al, 0x2c
    je command_input
    cmp al, 0x5b
    je command_open
    jmp command_close

command_right:
    inc rbx
    jmp command_next

command_left:
    dec rbx
    jmp command_next

command_increment:
    inc byte ptr [rbx]
    jmp command_next

command_decrement:
    dec byte ptr [rbx]
    jmp command_next

command_output:
    mov al, byte ptr [rbx]
    mov byte ptr [r9 + r10], al
    inc r10
    jmp command_next

command_input:
    cmp r15, rbp
    jae input_exhausted
    mov al, byte ptr [r15]
    inc r15
    mov byte ptr [rbx], al
    jmp command_next

input_exhausted:
    mov byte ptr [rbx], 0
    jmp command_next

command_open:
    cmp byte ptr [rbx], 0
    jne command_next
    mov eax, dword ptr [r8 + r14*4]
    mov r14, rax
    jmp execute_loop

command_close:
    cmp byte ptr [rbx], 0
    je command_next
    mov eax, dword ptr [r8 + r14*4]
    mov r14, rax
    jmp execute_loop

command_next:
    inc r14
    jmp execute_loop

execute_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r10
    syscall
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.balign 16
input_buffer:
    .zero 65536
program:
    .zero 65536
.balign 16
jumps:
    .zero 262144
bracket_stack:
    .zero 4000
.balign 16
tape:
    .zero 30000
output_buffer:
    .zero 65536
