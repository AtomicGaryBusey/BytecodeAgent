.globl _start

.text
_start:
    xor r14d, r14d
read_more:
    cmp r14, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r14
    mov edx, 65536
    sub rdx, r14
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    jmp read_more

read_done:
    lea r12, [rip + input_buffer]
    lea r13, [rip + input_buffer]
    add r13, r14

next_line:
    cmp r12, r13
    jae program_exit
    mov rdi, r12
    mov rsi, rdi
find_line_end:
    cmp rsi, r13
    jae final_line
    cmp byte ptr [rsi], 10
    je newline_line
    inc rsi
    jmp find_line_end
final_line:
    mov r12, rsi
    jmp evaluate
newline_line:
    lea r12, [rsi + 1]
evaluate:
    call eval_line
    mov rsi, rax
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

eval_line:
    lea r8, [rip + value_stack]
    xor r9d, r9d

skip_separators:
    cmp rdi, rsi
    jae line_finished
    mov al, byte ptr [rdi]
    cmp al, 32
    je skip_one
    cmp al, 9
    jne token_begin
skip_one:
    inc rdi
    jmp skip_separators

token_begin:
    mov r10, rdi
scan_token:
    cmp rdi, rsi
    jae token_scanned
    mov al, byte ptr [rdi]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    inc rdi
    jmp scan_token

token_scanned:
    mov rdx, rdi
    sub rdx, r10
    cmp rdx, 1
    jne integer_token
    movzx eax, byte ptr [r10]
    cmp al, 43
    je operator_token
    cmp al, 45
    je operator_token
    cmp al, 42
    je operator_token
    cmp al, 47
    je operator_token
    cmp al, 37
    je operator_token

integer_token:
    mov rcx, r10
    xor r11d, r11d
    cmp byte ptr [rcx], 45
    jne digits_setup
    mov r11d, 1
    inc rcx
digits_setup:
    mov rdx, rdi
    sub rdx, rcx
    test rdx, rdx
    jz eval_error
    cmp rdx, 18
    ja eval_error
    xor eax, eax
digit_loop:
    cmp rcx, rdi
    jae integer_done
    movzx edx, byte ptr [rcx]
    sub edx, 48
    cmp edx, 9
    ja eval_error
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    jmp digit_loop
integer_done:
    test r11d, r11d
    jz push_integer
    neg rax
push_integer:
    mov qword ptr [r8 + r9*8], rax
    inc r9
    jmp skip_separators

operator_token:
    cmp r9, 2
    jb eval_error
    mov rcx, qword ptr [r8 + r9*8 - 16]
    mov rdx, qword ptr [r8 + r9*8 - 8]
    dec r9
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    jmp op_mod

op_add:
    add rcx, rdx
    jmp op_store
op_sub:
    sub rcx, rdx
    jmp op_store
op_mul:
    imul rcx, rdx
    jmp op_store

op_div:
    test rdx, rdx
    jz eval_error
    mov rax, 0x8000000000000000
    cmp rcx, rax
    jne divide_regular
    cmp rdx, -1
    je eval_error
divide_regular:
    mov r10, rdx
    mov rax, rcx
    cqo
    idiv r10
    mov rcx, rax
    jmp op_store

op_mod:
    test rdx, rdx
    jz eval_error
    mov rax, 0x8000000000000000
    cmp rcx, rax
    jne modulo_regular
    cmp rdx, -1
    je eval_error
modulo_regular:
    mov r10, rdx
    mov rax, rcx
    cqo
    idiv r10
    mov rcx, rdx

op_store:
    mov qword ptr [r8 + r9*8 - 8], rcx
    jmp skip_separators

line_finished:
    cmp r9, 1
    jne eval_error
    mov rax, qword ptr [r8]
    lea rcx, [rip + number_buffer + 32]
    lea rdi, [rip + number_buffer + 31]
    mov byte ptr [rdi], 10
    xor r11d, r11d
    test rax, rax
    jns magnitude_ready
    neg rax
    mov r11d, 1
magnitude_ready:
    mov r10, 10
decimal_loop:
    xor edx, edx
    div r10
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz decimal_loop
    test r11d, r11d
    jz decimal_done
    dec rdi
    mov byte ptr [rdi], 45
decimal_done:
    mov rax, rdi
    mov rdx, rcx
    sub rdx, rdi
    ret

eval_error:
    lea rax, [rip + error_message]
    mov edx, 6
    ret

.section .rodata
error_message:
    .ascii "error\n"

.bss
.align 8
input_buffer:
    .zero 65536
value_stack:
    .zero 8192
number_buffer:
    .zero 32
