.globl _start

.text
_start:
    mov x19, #0
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer

read_loop:
    mov x2, #65536
    cmp x19, x2
    b.hs read_done
    mov x0, #0
    add x1, x20, x19
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0

find_separator:
    cmp x21, x19
    b.hs no_separator
    ldrb w0, [x20, x21]
    cmp w0, #33
    b.eq separator_found
    add x21, x21, #1
    b find_separator

separator_found:
    add x22, x21, #1
    b split_done

no_separator:
    mov x21, x19
    mov x22, x19

split_done:
    mov x23, x19
    adrp x27, commands
    add x27, x27, :lo12:commands
    adrp x28, jumps
    add x28, x28, :lo12:jumps
    adrp x29, bracket_stack
    add x29, x29, :lo12:bracket_stack
    mov x24, #0
    mov x25, #0
    mov x26, #0

preprocess_loop:
    cmp x25, x21
    b.hs preprocess_done
    ldrb w0, [x20, x25]
    add x25, x25, #1
    cmp w0, #62
    b.eq store_command
    cmp w0, #60
    b.eq store_command
    cmp w0, #43
    b.eq store_command
    cmp w0, #45
    b.eq store_command
    cmp w0, #46
    b.eq store_command
    cmp w0, #44
    b.eq store_command
    cmp w0, #91
    b.eq open_bracket
    cmp w0, #93
    b.eq close_bracket
    b preprocess_loop

store_command:
    strb w0, [x27, x24]
    add x24, x24, #1
    b preprocess_loop

open_bracket:
    strb w0, [x27, x24]
    cmp x26, #1000
    b.hs invalid_program
    str w24, [x29, x26, lsl #2]
    add x26, x26, #1
    add x24, x24, #1
    b preprocess_loop

close_bracket:
    strb w0, [x27, x24]
    cbz x26, invalid_program
    sub x26, x26, #1
    ldr w9, [x29, x26, lsl #2]
    add x10, x24, #1
    str w10, [x28, x9, lsl #2]
    add x11, x9, #1
    str w11, [x28, x24, lsl #2]
    add x24, x24, #1
    b preprocess_loop

preprocess_done:
    cbnz x26, invalid_program

    adrp x25, tape
    add x25, x25, :lo12:tape
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x19, #0
    mov x21, #0
    mov x9, #0

run_loop:
    cmp x9, x24
    b.hs program_done
    ldrb w10, [x27, x9]
    cmp w10, #62
    b.eq command_right
    cmp w10, #60
    b.eq command_left
    cmp w10, #43
    b.eq command_increment
    cmp w10, #45
    b.eq command_decrement
    cmp w10, #46
    b.eq command_output
    cmp w10, #44
    b.eq command_input
    cmp w10, #91
    b.eq command_open
    b command_close

command_right:
    add x21, x21, #1
    b advance_command

command_left:
    sub x21, x21, #1
    b advance_command

command_increment:
    ldrb w11, [x25, x21]
    add w11, w11, #1
    strb w11, [x25, x21]
    b advance_command

command_decrement:
    ldrb w11, [x25, x21]
    sub w11, w11, #1
    strb w11, [x25, x21]
    b advance_command

command_output:
    ldrb w11, [x25, x21]
    strb w11, [x26, x19]
    add x19, x19, #1
    b advance_command

command_input:
    cmp x22, x23
    b.hs input_exhausted
    ldrb w11, [x20, x22]
    add x22, x22, #1
    b store_input

input_exhausted:
    mov w11, #0

store_input:
    strb w11, [x25, x21]
    b advance_command

command_open:
    ldrb w11, [x25, x21]
    cbnz w11, advance_command
    ldr w9, [x28, x9, lsl #2]
    b run_loop

command_close:
    ldrb w11, [x25, x21]
    cbz w11, advance_command
    ldr w9, [x28, x9, lsl #2]
    b run_loop

advance_command:
    add x9, x9, #1
    b run_loop

program_done:
    mov x0, #1
    mov x1, x26
    mov x2, x19
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.align 4
input_buffer:
    .space 65536
commands:
    .space 65536
jumps:
    .space 262144
bracket_stack:
    .space 4000
tape:
    .space 30000
output_buffer:
    .space 65536
