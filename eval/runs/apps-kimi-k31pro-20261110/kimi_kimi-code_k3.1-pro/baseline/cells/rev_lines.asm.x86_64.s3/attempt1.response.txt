.globl _start

.text
_start:
    xor r12d, r12d
    lea rsi, [rip + input_buf]

read_more:
    mov edx, 65536
    sub rdx, r12
    jz input_ready

    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle input_ready

    add r12, rax
    add rsi, rax
    jmp read_more

input_ready:
    lea rsi, [rip + input_buf]
    lea rdi, [rip + output_buf]
    lea rcx, [rsi + r12]
    mov rbx, rsi

scan_line:
    cmp rsi, rcx
    jae input_scanned

    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne scan_line

    lea rdx, [rsi - 1]

reverse_terminated:
    cmp rbx, rdx
    je terminated_reversed
    dec rdx
    mov al, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_terminated

terminated_reversed:
    mov byte ptr [rdi], 10
    inc rdi
    mov rbx, rsi
    jmp scan_line

input_scanned:
    cmp rbx, rcx
    je write_output

    mov rdx, rcx

reverse_final:
    cmp rbx, rdx
    je final_reversed
    dec rdx
    mov al, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_final

final_reversed:
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_success

    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buf:
    .space 65536
output_buf:
    .space 65537
