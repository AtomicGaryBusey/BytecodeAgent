.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_more

read_done:
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rip + input_buffer + rcx], 0x21
    je separator_found
    inc rcx
    jmp find_separator

no_separator:
    mov rbp, r12
    lea r13, [rip + input_buffer]
    add r13, r12
    xor r14d, r14d
    jmp preprocess

separator_found:
    mov rbp, rcx
    lea r13, [rip + input_buffer]
    add r13, rcx
    inc r13
    mov r14, r12
    sub r14, rcx
    dec r14

preprocess:
    lea r8, [rip + input_buffer]
    lea r9, [rip + jump_map]
    lea r10, [rip + bracket_stack]
    xor ecx, ecx
    xor edx, edx

preprocess_loop:
    cmp rcx, rbp
    jae preprocess_done
    mov al, byte ptr [r8 + rcx]
    cmp al, 0x5b
    je open_bracket
    cmp al, 0x5d
    je close_bracket
    inc rcx
    jmp preprocess_loop

open_bracket:
    cmp edx, 1000
    jae invalid_program
    mov dword ptr [r10 + rdx*4], ecx
    inc edx
    inc rcx
    jmp preprocess_loop

close_bracket:
    test edx, edx
    jz invalid_program
    dec edx
    mov eax, dword ptr [r10 + rdx*4]
    lea esi, [rcx + 1]
    mov dword ptr [r9 + rax*4], esi
    lea esi, [rax + 1]
    mov dword ptr [r9 + rcx*4], esi
    inc rcx
    jmp preprocess_loop

preprocess_done:
    test edx, edx
    jnz invalid_program

    xor ebx, ebx
    lea r12, [rip + tape]
    lea r15, [rip + output_buffer]

execute_loop:
    cmp rbx, rbp
    jae execution_done
    movzx eax, byte ptr [r8 + rbx]
    inc rbx

    cmp al, 0x3e
    je cmd_right
    cmp al, 0x3c
    je cmd_left
    cmp al, 0x2b
    je cmd_increment
    cmp al, 0x2d
    je cmd_decrement
    cmp al, 0x2e
    je cmd_output
    cmp al, 0x2c
    je cmd_input
    cmp al, 0x5b
    je cmd_open
    cmp al, 0x5d
    je cmd_close
    jmp execute_loop

cmd_right:
    inc r12
    jmp execute_loop

cmd_left:
    dec r12
    jmp execute_loop

cmd_increment:
    inc byte ptr [r12]
    jmp execute_loop

cmd_decrement:
    dec byte ptr [r12]
    jmp execute_loop

cmd_output:
    mov al, byte ptr [r12]
    mov byte ptr [r15], al
    inc r15
    jmp execute_loop

cmd_input:
    test r14, r14
    jz input_exhausted
    mov al, byte ptr [r13]
    mov byte ptr [r12], al
    inc r13
    dec r14
    jmp execute_loop

input_exhausted:
    mov byte ptr [r12], 0
    jmp execute_loop

cmd_open:
    cmp byte ptr [r12], 0
    jne execute_loop
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    jmp execute_loop

cmd_close:
    cmp byte ptr [r12], 0
    je execute_loop
    mov ebx, dword ptr [r9 + rbx*4 - 4]
    jmp execute_loop

execution_done:
    lea rsi, [rip + output_buffer]
    mov rdx, r15
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
.align 16
jump_map:
    .zero 262144
.align 16
bracket_stack:
    .zero 4000
.align 16
tape:
    .zero 30000
.align 16
output_buffer:
    .zero 65536
