.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success

    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d
    lea rdx, [rip + lines]

parse_loop:
    cmp rbx, r12
    jae parse_remainder
    cmp byte ptr [rip + input_buf + rbx], 10
    je parse_newline
    inc rbx
    jmp parse_loop

parse_newline:
    mov dword ptr [rdx + r13*8], r14d
    mov rax, rbx
    sub rax, r14
    mov dword ptr [rdx + r13*8 + 4], eax
    inc r13
    inc rbx
    mov r14, rbx
    jmp parse_loop

parse_remainder:
    cmp r14, r12
    jae sort_begin
    mov dword ptr [rdx + r13*8], r14d
    mov rax, r12
    sub rax, r14
    mov dword ptr [rdx + r13*8 + 4], eax
    inc r13

sort_begin:
    mov r15d, 1

sort_outer:
    cmp r15, r13
    jae make_output
    mov r14, qword ptr [rdx + r15*8]
    mov rbx, r15

sort_inner:
    test rbx, rbx
    jz sort_place
    mov r11, qword ptr [rdx + rbx*8 - 8]

    mov eax, r14d
    lea rdi, [rip + input_buf]
    add rdi, rax
    mov eax, r11d
    lea rsi, [rip + input_buf]
    add rsi, rax

    mov r8, r14
    shr r8, 32
    mov r9, r11
    shr r9, 32
    mov rcx, r8
    cmp rcx, r9
    jbe 1f
    mov rcx, r9
1:
    repe cmpsb
    jne 2f
    cmp r8, r9
2:
    jb sort_shift
    jmp sort_place

sort_shift:
    mov qword ptr [rdx + rbx*8], r11
    dec rbx
    jmp sort_inner

sort_place:
    mov qword ptr [rdx + rbx*8], r14
    inc r15
    jmp sort_outer

make_output:
    xor r15d, r15d
    lea rdi, [rip + output_buf]

output_loop:
    cmp r15, r13
    jae output_done
    mov r11, qword ptr [rdx + r15*8]
    mov eax, r11d
    lea rsi, [rip + input_buf]
    add rsi, rax
    mov rcx, r11
    shr rcx, 32
    rep movsb
    mov al, 10
    stosb
    inc r15
    jmp output_loop

output_done:
    mov rdx, rdi
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buf:
    .zero 32768
.balign 16
lines:
    .zero 8000
.balign 16
output_buf:
    .zero 33768
