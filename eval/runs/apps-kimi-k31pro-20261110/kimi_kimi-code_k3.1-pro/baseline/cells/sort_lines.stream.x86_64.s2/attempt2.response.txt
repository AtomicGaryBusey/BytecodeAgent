.globl _start

.text
_start:
    xor r12d, r12d
    lea r15, [rip + input_buf]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r15
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, 32768
    jb read_loop

read_done:
    test r12, r12
    jz exit_success

    lea r8, [rip + lines]
    lea r9, [rip + lengths]
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d

parse_loop:
    cmp rbx, r12
    jae parse_remainder
    mov al, byte ptr [r15 + rbx]
    inc rbx
    cmp al, 10
    jne parse_loop

    lea rax, [r15 + r14]
    mov qword ptr [r8 + r13*8], rax
    mov rax, rbx
    sub rax, r14
    dec rax
    mov qword ptr [r9 + r13*8], rax
    inc r13
    mov r14, rbx
    jmp parse_loop

parse_remainder:
    cmp r14, r12
    jae sort_begin
    lea rax, [r15 + r14]
    mov qword ptr [r8 + r13*8], rax
    mov rax, r12
    sub rax, r14
    mov qword ptr [r9 + r13*8], rax
    inc r13

sort_begin:
    mov ebx, 1

sort_outer:
    cmp rbx, r13
    jae output_begin
    mov r15, qword ptr [r8 + rbx*8]
    mov r12, qword ptr [r9 + rbx*8]
    mov r14, rbx

sort_inner:
    test r14, r14
    jz insert_key
    mov rdi, qword ptr [r8 + r14*8 - 8]
    mov r10, qword ptr [r9 + r14*8 - 8]
    mov rsi, r15
    mov rcx, r10
    cmp rcx, r12
    jbe compare_bytes
    mov rcx, r12

compare_bytes:
    xor edx, edx

compare_loop:
    cmp rdx, rcx
    jae compare_lengths
    mov al, byte ptr [rdi + rdx]
    cmp al, byte ptr [rsi + rdx]
    jne compare_difference
    inc rdx
    jmp compare_loop

compare_difference:
    ja shift_previous
    jmp insert_key

compare_lengths:
    cmp r10, r12
    jbe insert_key

shift_previous:
    mov qword ptr [r8 + r14*8], rdi
    mov qword ptr [r9 + r14*8], r10
    dec r14
    jmp sort_inner

insert_key:
    mov qword ptr [r8 + r14*8], r15
    mov qword ptr [r9 + r14*8], r12
    inc rbx
    jmp sort_outer

output_begin:
    xor ebx, ebx

output_loop:
    cmp rbx, r13
    jae exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, qword ptr [r8 + rbx*8]
    mov rdx, qword ptr [r9 + rbx*8]
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    inc rbx
    jmp output_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buf:
    .zero 32768
.align 8
lines:
    .zero 8000
lengths:
    .zero 8000
