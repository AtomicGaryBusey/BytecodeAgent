.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    movz x21, #1, lsl #16

read_more:
    cmp x20, x21
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_more

read_done:
    mov x21, #0

next_row:
    cmp x21, x20
    b.hs print_total

    mov x22, #16
    sub x23, x20, x21
    cmp x23, x22
    csel x22, x23, x22, lo

    adrp x24, line_buffer
    add x24, x24, :lo12:line_buffer
    mov x27, x24

    mov w0, w21
    mov x1, x24
    bl put_hex8
    mov x24, x1

    mov w6, #32
    strb w6, [x24], #1
    strb w6, [x24], #1

    add x26, x19, x21
    mov x25, #0
    adr x2, hex_table

slot_loop:
    cmp x25, #16
    b.hs slots_done
    cmp x25, x22
    b.hs blank_slot

    ldrb w0, [x26, x25]
    lsr w1, w0, #4
    ldrb w1, [x2, x1]
    strb w1, [x24], #1
    and w0, w0, #15
    ldrb w0, [x2, x0]
    strb w0, [x24], #1
    strb w6, [x24], #1
    b slot_after

blank_slot:
    mov w7, #0x2020
    strh w7, [x24], #2
    strb w6, [x24], #1

slot_after:
    cmp x25, #7
    b.ne 1f
    strb w6, [x24], #1
1:
    add x25, x25, #1
    b slot_loop

slots_done:
    strb w6, [x24], #1
    mov w0, #124
    strb w0, [x24], #1

    mov x25, #0
ascii_loop:
    cmp x25, x22
    b.hs ascii_done
    ldrb w0, [x26, x25]
    cmp w0, #32
    b.lo 1f
    cmp w0, #126
    b.ls 2f
1:
    mov w0, #46
2:
    strb w0, [x24], #1
    add x25, x25, #1
    b ascii_loop

ascii_done:
    mov w0, #124
    strb w0, [x24], #1
    mov w0, #10
    strb w0, [x24], #1

    sub x2, x24, x27
    mov x0, #1
    mov x1, x27
    mov x8, #64
    svc #0

    add x21, x21, x22
    b next_row

print_total:
    adrp x24, line_buffer
    add x24, x24, :lo12:line_buffer
    mov w0, w20
    mov x1, x24
    bl put_hex8
    mov w0, #10
    strb w0, [x1], #1

    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
1:
    b 1b

put_hex8:
    adr x2, hex_table
    mov x3, #28
1:
    lsr w4, w0, w3
    and w4, w4, #15
    ldrb w5, [x2, x4]
    strb w5, [x1], #1
    subs x3, x3, #4
    b.ge 1b
    ret

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
    .align 4
input_buffer:
    .zero 65536
line_buffer:
    .zero 80
