.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    test r12, r12
    jz exit_success
    lea r13, [rip + input_buf]
    lea r14, [r13 + r12]

line_loop:
    cmp r13, r14
    jae exit_success
    mov r15, r13

find_newline:
    cmp r15, r14
    jae have_line
    cmp byte ptr [r15], 10
    je have_line
    inc r15
    jmp find_newline

have_line:
    mov rdi, r13
    mov rsi, r15
    call eval_line
    cmp r15, r14
    jae exit_success
    lea r13, [r15 + 1]
    jmp line_loop

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

eval_line:
    mov r8, rdi
    mov r9, rsi
    xor r10d, r10d
    lea r11, [rip + value_stack]

next_token:
    cmp r8, r9
    jae line_done
    mov al, byte ptr [r8]
    cmp al, 32
    je skip_separator
    cmp al, 9
    jne token_begin

skip_separator:
    inc r8
    jmp next_token

token_begin:
    mov rbx, r8

token_scan:
    cmp r8, r9
    jae token_ready
    mov al, byte ptr [r8]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc r8
    jmp token_scan

token_ready:
    mov rdx, r8
    mov rcx, rdx
    sub rcx, rbx
    cmp rcx, 1
    jne parse_literal
    mov al, byte ptr [rbx]
    cmp al, 43
    je op_add
    cmp al, 45
    je op_sub
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod

parse_literal:
    mov rsi, rbx
    xor edi, edi
    cmp byte ptr [rsi], 45
    jne digit_count
    mov edi, 1
    inc rsi

digit_count:
    mov rcx, rdx
    sub rcx, rsi
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax

digit_loop:
    cmp rsi, rdx
    jae literal_ready
    movzx ecx, byte ptr [rsi]
    sub ecx, 48
    cmp ecx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rcx
    inc rsi
    jmp digit_loop

literal_ready:
    test edi, edi
    jz push_literal
    neg rax

push_literal:
    mov qword ptr [r11 + r10*8], rax
    inc r10
    jmp next_token

op_add:
    cmp r10, 2
    jb line_error
    mov rax, qword ptr [r11 + r10*8 - 16]
    add rax, qword ptr [r11 + r10*8 - 8]
    jmp store_result

op_sub:
    cmp r10, 2
    jb line_error
    mov rax, qword ptr [r11 + r10*8 - 16]
    sub rax, qword ptr [r11 + r10*8 - 8]
    jmp store_result

op_mul:
    cmp r10, 2
    jb line_error
    mov rax, qword ptr [r11 + r10*8 - 16]
    imul rax, qword ptr [r11 + r10*8 - 8]
    jmp store_result

op_div:
    cmp r10, 2
    jb line_error
    mov rcx, qword ptr [r11 + r10*8 - 8]
    test rcx, rcx
    jz line_error
    mov rax, qword ptr [r11 + r10*8 - 16]
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    jne div_ready
    cmp rcx, -1
    je line_error

div_ready:
    cqo
    idiv rcx
    jmp store_result

op_mod:
    cmp r10, 2
    jb line_error
    mov rcx, qword ptr [r11 + r10*8 - 8]
    test rcx, rcx
    jz line_error
    mov rax, qword ptr [r11 + r10*8 - 16]
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    jne mod_ready
    cmp rcx, -1
    je line_error

mod_ready:
    cqo
    idiv rcx
    mov rax, rdx

store_result:
    dec r10
    mov qword ptr [r11 + r10*8 - 8], rax
    jmp next_token

line_done:
    cmp r10, 1
    jne line_error
    mov rax, qword ptr [r11]
    lea rsi, [rip + output_buf + 32]
    dec rsi
    mov byte ptr [rsi], 10
    xor edi, edi
    test rax, rax
    jns convert_setup
    mov edi, 1
    neg rax

convert_setup:
    mov ecx, 10

convert_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert_loop
    test edi, edi
    jz write_result
    dec rsi
    mov byte ptr [rsi], 45

write_result:
    lea rdx, [rip + output_buf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 6
    syscall
    ret

.section .rodata
error_msg:
    .ascii "error\n"

.bss
input_buf:
    .zero 65536
value_stack:
    .zero 8192
output_buf:
    .zero 32
