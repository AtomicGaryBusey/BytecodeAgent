.globl _start
.text
_start:
    xor r12d, r12d

read_input:
    mov edx, 65536
    sub rdx, r12
    jz input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_input

input_done:
    lea r15, [rip + hex_digits]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae print_total

    lea rsi, [rip + line_buffer]
    mov rax, 0x2020202020202020
    mov qword ptr [rsi], rax
    mov qword ptr [rsi + 8], rax
    mov qword ptr [rsi + 16], rax
    mov qword ptr [rsi + 24], rax
    mov qword ptr [rsi + 32], rax
    mov qword ptr [rsi + 40], rax
    mov qword ptr [rsi + 48], rax
    mov qword ptr [rsi + 56], rax

    mov rax, r13
    lea rdi, [rsi + 7]
    mov ecx, 8
offset_digits:
    mov rdx, rax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    shr rax, 4
    dec rdi
    dec ecx
    jnz offset_digits

    mov byte ptr [rsi + 60], '|'

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_length_ready
    mov r14d, 16
row_length_ready:
    lea rbx, [rip + input_buffer]
    add rbx, r13
    lea rdi, [rsi + 10]
    lea r8, [rsi + 61]
    xor ecx, ecx

byte_loop:
    cmp rcx, r14
    jae byte_slot_done

    movzx eax, byte ptr [rbx + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al

    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x20
    jb nonprintable
    cmp al, 0x7e
    jbe printable
nonprintable:
    mov al, '.'
printable:
    mov byte ptr [r8 + rcx], al

byte_slot_done:
    add rdi, 3
    inc rcx
    cmp rcx, 8
    jne no_middle_space
    inc rdi
no_middle_space:
    cmp rcx, 16
    jb byte_loop

    lea rdi, [rsi + 61]
    add rdi, r14
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 10

    mov eax, 1
    mov edi, 1
    lea rdx, [r14 + 63]
    syscall

    add r13, r14
    jmp row_loop

print_total:
    lea rsi, [rip + line_buffer]
    mov rax, r12
    lea rdi, [rsi + 7]
    mov ecx, 8
total_digits:
    mov rdx, rax
    and edx, 15
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    shr rax, 4
    dec rdi
    dec ecx
    jnz total_digits

    mov byte ptr [rsi + 8], 10
    mov eax, 1
    mov edi, 1
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero 65536
line_buffer:
    .zero 80
