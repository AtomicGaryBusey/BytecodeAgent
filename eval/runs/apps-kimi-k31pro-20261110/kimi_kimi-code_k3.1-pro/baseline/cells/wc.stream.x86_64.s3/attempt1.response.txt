.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 4096
    syscall

    test rax, rax
    jle input_done

    add r14, rax
    lea rsi, [rip + input_buffer]
    mov rcx, rax

scan_loop:
    movzx edx, byte ptr [rsi]
    inc rsi

    cmp dl, 10
    jne 1f
    inc r12
1:
    cmp dl, 32
    je 3f
    cmp dl, 9
    jb 2f
    cmp dl, 13
    jbe 3f

2:
    test r15d, r15d
    jnz 4f
    inc r13
    mov r15d, 1
    jmp 4f

3:
    xor r15d, r15d

4:
    dec rcx
    jnz scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_number
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_number:
    mov r8, rdi
    mov r9d, 10

1:
    xor edx, edx
    div r9
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz 1b

    mov rsi, rdi
    dec rsi

2:
    cmp r8, rsi
    jae 3f
    mov al, byte ptr [r8]
    mov dl, byte ptr [rsi]
    mov byte ptr [r8], dl
    mov byte ptr [rsi], al
    inc r8
    dec rsi
    jmp 2b

3:
    ret

.bss
.balign 16
input_buffer:
    .zero 4096
output_buffer:
    .zero 64
