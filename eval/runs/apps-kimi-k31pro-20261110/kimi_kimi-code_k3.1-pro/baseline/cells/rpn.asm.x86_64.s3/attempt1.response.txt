.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    mov edx, 65537
    sub rdx, r12
    jz input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall
    test rax, rax
    jle input_ready
    add r12, rax
    jmp read_more

input_ready:
    lea r13, [rip + input_buffer]
    lea r14, [r13 + r12]

line_loop:
    cmp r13, r14
    jae program_exit

    mov r15, r13
find_line_end:
    cmp r15, r14
    jae line_found
    cmp byte ptr [r15], 10
    je line_found
    inc r15
    jmp find_line_end

line_found:
    mov rbx, r13
    xor ebp, ebp

token_skip:
    cmp rbx, r15
    jae line_done
    mov al, byte ptr [rbx]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp token_begin

skip_one:
    inc rbx
    jmp token_skip

token_begin:
    mov rsi, rbx

token_scan:
    cmp rbx, r15
    jae token_ready
    mov al, byte ptr [rbx]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc rbx
    jmp token_scan

token_ready:
    mov rcx, rbx
    sub rcx, rsi
    cmp rcx, 1
    jne literal_setup

    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    cmp al, '0'
    jb line_error
    cmp al, '9'
    ja line_error

literal_setup:
    xor r8d, r8d
    mov r9, rsi
    mov rdx, rcx
    cmp byte ptr [r9], '-'
    jne literal_length
    mov r8d, 1
    inc r9
    dec rdx

literal_length:
    test rdx, rdx
    jz line_error
    cmp rdx, 18
    ja line_error

    xor r11d, r11d

literal_digits:
    movzx eax, byte ptr [r9]
    sub eax, '0'
    cmp eax, 9
    ja line_error
    imul r11, r11, 10
    add r11, rax
    inc r9
    dec rdx
    jnz literal_digits

    test r8d, r8d
    jz literal_store
    neg r11

literal_store:
    mov qword ptr [value_stack + rbp*8], r11
    inc rbp
    jmp token_skip

op_add:
    cmp rbp, 2
    jb line_error
    mov r10, qword ptr [value_stack + rbp*8 - 8]
    mov r11, qword ptr [value_stack + rbp*8 - 16]
    add r11, r10
    mov qword ptr [value_stack + rbp*8 - 16], r11
    dec rbp
    jmp token_skip

op_sub:
    cmp rbp, 2
    jb line_error
    mov r10, qword ptr [value_stack + rbp*8 - 8]
    mov r11, qword ptr [value_stack + rbp*8 - 16]
    sub r11, r10
    mov qword ptr [value_stack + rbp*8 - 16], r11
    dec rbp
    jmp token_skip

op_mul:
    cmp rbp, 2
    jb line_error
    mov r10, qword ptr [value_stack + rbp*8 - 8]
    mov r11, qword ptr [value_stack + rbp*8 - 16]
    imul r11, r10
    mov qword ptr [value_stack + rbp*8 - 16], r11
    dec rbp
    jmp token_skip

op_div:
    cmp rbp, 2
    jb line_error
    mov r10, qword ptr [value_stack + rbp*8 - 8]
    mov r11, qword ptr [value_stack + rbp*8 - 16]
    test r10, r10
    jz line_error
    mov rax, 0x8000000000000000
    cmp r11, rax
    jne divide_ready
    cmp r10, -1
    je line_error

divide_ready:
    mov rax, r11
    cqo
    idiv r10
    mov qword ptr [value_stack + rbp*8 - 16], rax
    dec rbp
    jmp token_skip

op_mod:
    cmp rbp, 2
    jb line_error
    mov r10, qword ptr [value_stack + rbp*8 - 8]
    mov r11, qword ptr [value_stack + rbp*8 - 16]
    test r10, r10
    jz line_error
    mov rax, 0x8000000000000000
    cmp r11, rax
    jne modulo_ready
    cmp r10, -1
    je line_error

modulo_ready:
    mov rax, r11
    cqo
    idiv r10
    mov qword ptr [value_stack + rbp*8 - 16], rdx
    dec rbp
    jmp token_skip

line_done:
    cmp rbp, 1
    jne line_error
    mov rax, qword ptr [rip + value_stack]

    lea rdi, [rip + output_buffer + 32]
    dec rdi
    mov byte ptr [rdi], 10

    xor r8d, r8d
    test rax, rax
    jns magnitude_ready
    mov r8d, 1
    neg rax

magnitude_ready:
    mov rcx, 10

digit_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz digit_loop

    test r8d, r8d
    jz print_number
    dec rdi
    mov byte ptr [rdi], '-'

print_number:
    lea rdx, [rip + output_buffer + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    jmp next_line

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall

next_line:
    mov r13, r15
    cmp r13, r14
    jae program_exit
    inc r13
    jmp line_loop

program_exit:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
error_message:
    .ascii "error\n"

.bss
.p2align 3
input_buffer:
    .zero 65537
value_stack:
    .zero 4096
output_buffer:
    .zero 32
