.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    jae read_extra

    mov edx, 65536
    sub rdx, r12
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_loop

read_extra:
    lea rsi, [rip + linebuf]
    mov edx, 1
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jg read_extra

input_done:
    xor r13d, r13d
    lea r14, [rip + inbuf]
    lea rbp, [rip + linebuf]

row_loop:
    cmp r13, r12
    jae output_done

    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe 1f
    mov r15d, 16
1:
    mov rax, 0x2020202020202020
    mov qword ptr [rbp], rax
    mov qword ptr [rbp + 8], rax
    mov qword ptr [rbp + 16], rax
    mov qword ptr [rbp + 24], rax
    mov qword ptr [rbp + 32], rax
    mov qword ptr [rbp + 40], rax
    mov qword ptr [rbp + 48], rax
    mov qword ptr [rbp + 56], rax
    mov qword ptr [rbp + 64], rax
    mov qword ptr [rbp + 72], rax

    mov eax, r13d
    mov rdi, rbp
    call put_hex8

    xor ebx, ebx
hex_loop:
    cmp rbx, r15
    jae hex_done

    lea rdi, [rbp + 10]
    lea rax, [rbx + rbx*2]
    add rdi, rax
    cmp ebx, 8
    jb 1f
    inc rdi
1:
    movzx eax, byte ptr [r14 + rbx]
    mov edx, eax
    shr eax, 4
    and edx, 15
    lea r8, [rip + hex_digits]
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl

    inc rbx
    jmp hex_loop

hex_done:
    mov byte ptr [rbp + 59], 32
    mov byte ptr [rbp + 60], 124

    xor ebx, ebx
ascii_loop:
    cmp rbx, r15
    jae ascii_done

    mov al, byte ptr [r14 + rbx]
    cmp al, 0x20
    jb 1f
    cmp al, 0x7e
    jbe 2f
1:
    mov al, 46
2:
    mov byte ptr [rbp + rbx + 61], al
    inc rbx
    jmp ascii_loop

ascii_done:
    lea rax, [rbx + 61]
    mov byte ptr [rbp + rax], 124
    mov byte ptr [rbp + rax + 1], 10
    lea rdx, [rax + 2]

    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    syscall

    add r14, r15
    add r13, r15
    jmp row_loop

output_done:
    mov eax, r13d
    mov rdi, rbp
    call put_hex8
    mov byte ptr [rbp + 8], 10

    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall

put_hex8:
    mov edx, eax
    mov ecx, 8
    lea r8, [rip + hex_digits]
1:
    mov eax, edx
    shr eax, 28
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    shl edx, 4
    dec ecx
    jnz 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
linebuf:
    .zero 80
