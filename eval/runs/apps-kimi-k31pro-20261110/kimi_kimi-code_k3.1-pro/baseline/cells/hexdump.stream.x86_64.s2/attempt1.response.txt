.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .balign 16
input_buffer:
    .zero 65536
line_buffer:
    .zero 80

.text
_start:
    xor r12d, r12d

read_input:
    cmp r12, 65536
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_input

input_done:
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae print_final_offset

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe have_row_length
    mov r14d, 16

have_row_length:
    lea rdi, [rip + line_buffer]
    mov ecx, 79
    mov al, 32
    rep stosb

    mov eax, r13d
    lea rdi, [rip + line_buffer]
    lea r8, [rip + hex_digits]
    mov ecx, 8

offset_digit_loop:
    mov edx, eax
    shr edx, 28
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    shl eax, 4
    inc rdi
    dec ecx
    jnz offset_digit_loop

    lea rsi, [rip + input_buffer]
    add rsi, r13
    lea rdi, [rip + line_buffer + 10]
    xor ecx, ecx

hex_byte_loop:
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    add rdi, 3
    cmp ecx, 7
    jne no_middle_gap
    add rdi, 1

no_middle_gap:
    inc ecx
    cmp ecx, r14d
    jb hex_byte_loop

    mov byte ptr [rip + line_buffer + 60], 124
    lea rdi, [rip + line_buffer + 61]
    xor ecx, ecx

ascii_byte_loop:
    mov al, byte ptr [rsi + rcx]
    cmp al, 32
    jb nonprintable
    cmp al, 126
    jbe store_ascii

nonprintable:
    mov al, 46

store_ascii:
    stosb
    inc ecx
    cmp ecx, r14d
    jb ascii_byte_loop

    mov al, 124
    stosb
    mov al, 10
    stosb

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buffer]
    lea rdx, [r14 + 63]
    syscall

    add r13, r14
    jmp row_loop

print_final_offset:
    mov eax, r12d
    lea rdi, [rip + line_buffer]
    lea r8, [rip + hex_digits]
    mov ecx, 8

final_digit_loop:
    mov edx, eax
    shr edx, 28
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    shl eax, 4
    inc rdi
    dec ecx
    jnz final_digit_loop

    mov byte ptr [rdi], 10
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buffer]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall
