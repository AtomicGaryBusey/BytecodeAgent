.globl _start

.text
hex8:
    mov w5, #28
1:
    lsr w2, w0, w5
    and w2, w2, #15
    ldrb w2, [x26, x2]
    strb w2, [x1], #1
    subs w5, w5, #4
    b.ge 1b
    ret

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x26, hex_digits
    add x26, x26, :lo12:hex_digits
    mov x19, #0

read_loop:
    mov x2, #65536
    cmp x19, x2
    b.eq input_done
    sub x2, x2, x19
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    b read_loop

input_done:
    adrp x22, line_buf
    add x22, x22, :lo12:line_buf
    mov x21, #0

row_loop:
    cmp x21, x19
    b.hs rows_done

    sub x24, x19, x21
    cmp x24, #16
    b.ls 1f
    mov x24, #16
1:
    add x23, x20, x21

    mov x3, #0x2020
    movk x3, #0x2020, lsl #16
    movk x3, #0x2020, lsl #32
    movk x3, #0x2020, lsl #48
    stp x3, x3, [x22, #0]
    stp x3, x3, [x22, #16]
    stp x3, x3, [x22, #32]
    stp x3, x3, [x22, #48]
    stp x3, x3, [x22, #64]

    mov w0, w21
    mov x1, x22
    bl hex8

    mov w14, #0x7c
    strb w14, [x22, #60]
    add x10, x22, #10
    add x11, x22, #61
    mov x25, #0

byte_loop:
    cmp x25, x24
    b.hs bytes_done
    ldrb w12, [x23, x25]

    lsr w13, w12, #4
    ldrb w13, [x26, x13]
    strb w13, [x10], #1
    and w13, w12, #15
    ldrb w13, [x26, x13]
    strb w13, [x10], #1
    mov w13, #0x20
    strb w13, [x10], #1

    mov w14, #0x2e
    cmp w12, #0x20
    b.lo 2f
    cmp w12, #0x7e
    b.hi 2f
    mov w14, w12
2:
    strb w14, [x11], #1

    add x25, x25, #1
    cmp x25, #8
    b.ne 3f
    add x10, x10, #1
3:
    b byte_loop

bytes_done:
    mov w14, #0x7c
    strb w14, [x11], #1
    mov w14, #0x0a
    strb w14, [x11], #1

    sub x2, x11, x22
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0

    add x21, x21, x24
    b row_loop

rows_done:
    mov w0, w19
    mov x1, x22
    bl hex8
    mov w2, #0x0a
    strb w2, [x22, #8]

    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
line_buf:
    .zero 80
