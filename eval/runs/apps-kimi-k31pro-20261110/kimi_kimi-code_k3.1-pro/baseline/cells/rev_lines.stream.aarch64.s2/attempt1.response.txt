.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf

    mov x20, #0
    mov x27, #65536

read_loop:
    cmp x20, x27
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x27, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0

next_line:
    cmp x23, x20
    b.hs output
    mov x24, x23
    mov x25, x23

scan_line:
    cmp x25, x20
    b.eq final_line
    ldrb w0, [x19, x25]
    cmp w0, #10
    b.eq newline_found
    add x25, x25, #1
    b scan_line

newline_found:
    add x26, x25, #1
    b emit_line

final_line:
    mov x26, x25

emit_line:
    cmp x25, x24
    b.eq emit_newline
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b emit_line

emit_newline:
    mov w0, #10
    strb w0, [x21, x22]
    add x22, x22, #1
    mov x23, x26
    b next_line

output:
    cmp x22, #0
    b.eq exit_program
    mov x0, #1
    mov x1, x21
    mov x2, x22
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_program

.bss
.balign 16
input_buf:
    .zero 65536
output_buf:
    .zero 65537
