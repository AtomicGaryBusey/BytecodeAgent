.globl _start

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x9, #65536

read_more:
    cmp x20, x9
    b.hs input_complete
    mov x0, #0
    add x1, x19, x20
    sub x2, x9, x20
    mov x8, #63
    svc #0
    cbz x0, input_complete
    add x20, x20, x0
    b read_more

input_complete:
    mov x21, #0

next_line:
    cmp x21, x20
    b.hs finished
    mov x23, x21

find_line_end:
    cmp x23, x20
    b.hs line_found
    ldrb w0, [x19, x23]
    cmp w0, #10
    b.eq line_found
    add x23, x23, #1
    b find_line_end

line_found:
    add x0, x19, x21
    sub x1, x23, x21
    cmp x23, x20
    b.hs last_line
    add x21, x23, #1
    b evaluate_line
last_line:
    mov x21, x23
evaluate_line:
    bl process_line
    b next_line

finished:
    mov x0, #0
    mov x8, #93
    svc #0

process_line:
    mov x22, x0
    mov x23, x1
    mov x24, #0
    mov x25, #0
    adrp x26, value_stack
    add x26, x26, :lo12:value_stack

skip_separators:
    cmp x24, x23
    b.hs tokens_finished
    ldrb w0, [x22, x24]
    cmp w0, #32
    b.eq consume_separator
    cmp w0, #9
    b.ne token_starts
consume_separator:
    add x24, x24, #1
    b skip_separators

token_starts:
    mov x2, x24

scan_token:
    cmp x24, x23
    b.hs token_ready
    ldrb w0, [x22, x24]
    cmp w0, #32
    b.eq token_ready
    cmp w0, #9
    b.eq token_ready
    add x24, x24, #1
    b scan_token

token_ready:
    sub x3, x24, x2
    ldrb w4, [x22, x2]
    cmp x3, #1
    b.ne literal_begin
    cmp w4, #43
    b.eq operator_begin
    cmp w4, #45
    b.eq operator_begin
    cmp w4, #42
    b.eq operator_begin
    cmp w4, #47
    b.eq operator_begin
    cmp w4, #37
    b.eq operator_begin

literal_begin:
    mov x5, x2
    mov x6, #0
    cmp w4, #45
    b.ne no_minus_prefix
    add x5, x5, #1
    mov x6, #1
no_minus_prefix:
    sub x7, x24, x5
    cmp x7, #1
    b.lo line_error
    cmp x7, #18
    b.hi line_error
    mov x8, #0
    mov x9, #10

parse_digits:
    ldrb w10, [x22, x5]
    sub w10, w10, #48
    cmp w10, #9
    b.hi line_error
    mul x8, x8, x9
    add x8, x8, x10
    add x5, x5, #1
    cmp x5, x24
    b.lo parse_digits
    cbz x6, push_literal
    neg x8, x8
push_literal:
    str x8, [x26, x25, lsl #3]
    add x25, x25, #1
    b skip_separators

operator_begin:
    cmp x25, #2
    b.lo line_error
    sub x9, x25, #1
    ldr x10, [x26, x9, lsl #3]
    sub x11, x25, #2
    ldr x12, [x26, x11, lsl #3]
    cmp w4, #43
    b.eq do_add
    cmp w4, #45
    b.eq do_subtract
    cmp w4, #42
    b.eq do_multiply
    b do_divide_or_remainder

do_add:
    add x13, x12, x10
    b store_result
do_subtract:
    sub x13, x12, x10
    b store_result
do_multiply:
    mul x13, x12, x10
    b store_result

do_divide_or_remainder:
    cbz x10, line_error
    movz x14, #0x8000, lsl #48
    cmp x12, x14
    b.ne division_safe
    cmn x10, #1
    b.eq line_error
division_safe:
    sdiv x13, x12, x10
    cmp w4, #37
    b.ne store_result
    msub x13, x13, x10, x12

store_result:
    str x13, [x26, x11, lsl #3]
    sub x25, x25, #1
    b skip_separators

tokens_finished:
    cmp x25, #1
    b.ne line_error
    ldr x2, [x26]
    adrp x3, number_buf
    add x3, x3, :lo12:number_buf
    add x4, x3, #32
    sub x3, x4, #1
    mov w5, #10
    strb w5, [x3]
    mov x6, #0
    cmp x2, #0
    b.ge magnitude_ready
    mov x6, #1
    neg x2, x2
magnitude_ready:
    mov x9, #10

convert_digits:
    udiv x7, x2, x9
    msub x8, x7, x9, x2
    add w8, w8, #48
    sub x3, x3, #1
    strb w8, [x3]
    mov x2, x7
    cbnz x2, convert_digits
    cbz x6, print_number
    sub x3, x3, #1
    mov w5, #45
    strb w5, [x3]

    sub x2, x4, x3
    mov x0, #1
    mov x1, x3
    mov x8, #64
    svc #0
    ret

    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, #64
    svc #0
    ret
