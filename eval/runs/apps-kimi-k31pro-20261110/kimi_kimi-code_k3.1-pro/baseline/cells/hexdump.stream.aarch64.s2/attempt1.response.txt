.text
.globl _start
.p2align 2

_start:
    mov x19, #0

read_loop:
    mov x0, #0
    adr x1, input_buffer
    add x1, x1, x19
    mov x2, #1
    lsl x2, x2, #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0
    cbz x19, final_offset

row_loop:
    adr x21, line_buffer
    mov x0, x21
    mov x1, #61
    mov w2, #32
1:
    strb w2, [x0], #1
    subs x1, x1, #1
    b.ne 1b

    sub x22, x19, x20
    cmp x22, #16
    b.ls 1f
    mov x22, #16
1:
    mov x0, x20
    mov x1, x21
    bl format_hex8

    mov w0, #124
    strb w0, [x21, #60]

    adr x23, input_buffer
    add x23, x23, x20
    add x24, x21, #10
    add x25, x21, #61
    mov x26, #0
    adr x2, hex_digits

byte_loop:
    cmp x26, x22
    b.hs byte_done

    ldrb w0, [x23, x26]
    lsr w1, w0, #4
    ldrb w1, [x2, x1]
    strb w1, [x24]
    and w1, w0, #15
    ldrb w1, [x2, x1]
    strb w1, [x24, #1]

    cmp w0, #32
    b.lo 1f
    cmp w0, #126
    b.hi 1f
    b 2f
1:
    mov w0, #46
2:
    strb w0, [x25], #1

byte_done:
    add x24, x24, #3
    cmp x26, #7
    b.ne 1f
    add x24, x24, #1
1:
    add x26, x26, #1
    cmp x26, #16
    b.ne byte_loop

    mov w0, #124
    strb w0, [x25], #1
    mov w0, #10
    strb w0, [x25]

    mov x0, #1
    mov x1, x21
    add x2, x22, #63
    mov x8, #64
    svc #0

    add x20, x20, x22
    cmp x20, x19
    b.lo row_loop

final_offset:
    adr x21, line_buffer
    mov x0, x19
    mov x1, x21
    bl format_hex8
    mov w0, #10
    strb w0, [x21, #8]

    mov x0, #1
    mov x1, x21
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

format_hex8:
    mov x2, #8
    mov x3, x0
    adr x4, hex_digits
1:
    lsr x5, x3, #28
    and x5, x5, #15
    ldrb w5, [x4, x5]
    strb w5, [x1], #1
    lsl x3, x3, #4
    subs x2, x2, #1
    b.ne 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buffer:
    .zero 65536
line_buffer:
    .zero 80
