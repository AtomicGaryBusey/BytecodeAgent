.globl _start

.equ BUF_SIZE, 4096

.text
_start:
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov w23, #0

read_loop:
    mov x0, #0
    mov x1, x19
    mov x2, #BUF_SIZE
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_done

    add x20, x20, x0
    mov x3, x19
    mov x4, x0

byte_loop:
    ldrb w5, [x3], #1

    cmp w5, #10
    b.ne 1f
    add x21, x21, #1
1:
    cmp w5, #32
    b.eq whitespace
    cmp w5, #9
    b.lo non_whitespace
    cmp w5, #13
    b.ls whitespace

non_whitespace:
    cmp w23, #0
    b.ne next_byte
    mov w23, #1
    add x22, x22, #1
    b next_byte

whitespace:
    mov w23, #0

next_byte:
    subs x4, x4, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x24, out_buf
    add x24, x24, :lo12:out_buf
    mov x1, x24

    mov x0, x21
    bl append_number
    mov w5, #32
    strb w5, [x1], #1

    mov x0, x22
    bl append_number
    mov w5, #32
    strb w5, [x1], #1

    mov x0, x20
    bl append_number
    mov w5, #10
    strb w5, [x1], #1

    sub x2, x1, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_number:
    sub sp, sp, #32
    mov x2, sp
    add x3, sp, #32
    mov x4, #10

    cbnz x0, 1f
    mov w5, #48
    strb w5, [x1], #1
    b 3f

1:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #48
    strb w6, [x3, #-1]!
    mov x0, x5
    cbnz x0, 1b

2:
    ldrb w5, [x3], #1
    strb w5, [x1], #1
    cmp x3, x2
    b.ne 2b

3:
    add sp, sp, #32
    ret

.bss
.balign 16
in_buf:
    .zero BUF_SIZE
out_buf:
    .zero 64
