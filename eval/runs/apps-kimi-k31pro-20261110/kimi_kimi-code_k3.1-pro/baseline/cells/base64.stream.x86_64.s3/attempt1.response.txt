.globl _start

.text
_start:
    xor r12d, r12d

read_input:
    mov edx, 49152
    sub rdx, r12
    jz encode_start

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    syscall

    test rax, rax
    jle encode_start
    add r12, rax
    jmp read_input

encode_start:
    lea rsi, [rip + input_buffer]
    mov r9, r12
    lea rdi, [rip + output_buffer]
    xor r8d, r8d
    lea r10, [rip + base64_alphabet]

    cmp r9, 3
    jb final_group

encode_full_group:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ebx, byte ptr [rsi + 1]
    shl ebx, 8
    or eax, ebx
    movzx ebx, byte ptr [rsi + 2]
    or eax, ebx

    call encode_group
    call emit_group

    add rsi, 3
    sub r9, 3
    cmp r9, 3
    jae encode_full_group

final_group:
    test r9, r9
    jz finish_lines

    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r9, 1
    je encode_final

    movzx ebx, byte ptr [rsi + 1]
    shl ebx, 8
    or eax, ebx

encode_final:
    call encode_group
    cmp r9, 1
    jne pad_one

    and ebx, 0x0000ffff
    or ebx, 0x3d3d0000
    jmp emit_final

pad_one:
    and ebx, 0x00ffffff
    or ebx, 0x3d000000

emit_final:
    call emit_group

finish_lines:
    test r8d, r8d
    jz prepare_output
    mov byte ptr [rdi], 10
    inc rdi

prepare_output:
    mov r13, rdi
    lea r14, [rip + output_buffer]
    sub r13, r14
    jz program_exit

write_output:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r13
    syscall

    test rax, rax
    jle program_exit
    add r14, rax
    sub r13, rax
    jnz write_output

program_exit:
    xor edi, edi
    mov eax, 60
    syscall

# Convert the low 24 bits of eax into four encoded bytes in ebx.
encode_group:
    mov ebx, eax
    shr ebx, 18
    movzx ebx, byte ptr [r10 + rbx]

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [r10 + rdx]
    shl edx, 8
    or ebx, edx

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [r10 + rdx]
    shl edx, 16
    or ebx, edx

    and eax, 63
    movzx eax, byte ptr [r10 + rax]
    shl eax, 24
    or ebx, eax
    ret

# Emit the four characters in ebx, inserting newlines after each 76 characters.
emit_group:
    mov ecx, 4

emit_character:
    mov byte ptr [rdi], bl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne no_newline

    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

no_newline:
    shr ebx, 8
    dec ecx
    jnz emit_character
    ret

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66560
