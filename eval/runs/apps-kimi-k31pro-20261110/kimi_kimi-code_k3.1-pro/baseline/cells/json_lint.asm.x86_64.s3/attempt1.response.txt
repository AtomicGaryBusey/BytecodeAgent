.globl _start

.text
_start:
    xor ebx, ebx
    lea r12, [rip + input_buf]

read_more:
    cmp rbx, 65536
    jae read_done
    mov edx, 65536
    sub edx, ebx
    lea rsi, [r12 + rbx]
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_more

read_done:
    mov r13, rbx
    xor r14d, r14d
    call skip_ws
    call parse_value
    test eax, eax
    jz report_error
    call skip_ws
    cmp r14, r13
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

report_error:
    lea rdi, [rip + output_buf]
    mov rax, qword ptr [rip + error_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + error_prefix + 8]
    mov byte ptr [rdi + 8], al

    lea rcx, [rdi + 9]
    mov rsi, rcx
    mov rax, r14
    mov r8d, 10

convert_digit:
    xor edx, edx
    div r8
    add dl, 48
    mov byte ptr [rsi], dl
    inc rsi
    test rax, rax
    jnz convert_digit

    lea rdx, [rsi - 1]
reverse_digits:
    cmp rcx, rdx
    jae digits_reversed
    mov r8b, byte ptr [rcx]
    mov r9b, byte ptr [rdx]
    mov byte ptr [rcx], r9b
    mov byte ptr [rdx], r8b
    inc rcx
    dec rdx
    jmp reverse_digits

digits_reversed:
    mov byte ptr [rsi], 10
    inc rsi
    mov rdx, rsi
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

parse_fail:
    xor eax, eax
    ret

parse_ok:
    mov eax, 1
    ret

skip_ws:
    cmp r14, r13
    jae ws_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x20
    je ws_byte
    cmp al, 0x09
    je ws_byte
    cmp al, 0x0a
    je ws_byte
    cmp al, 0x0d
    jne ws_done
ws_byte:
    inc r14
    jmp skip_ws
ws_done:
    ret

parse_value:
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je value_true
    cmp al, 0x66
    je value_false
    cmp al, 0x6e
    je value_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    jbe parse_number
    jmp parse_fail

value_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4
    jmp parse_literal

parse_literal:
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    movzx edx, byte ptr [rsi]
    cmp al, dl
    jne parse_fail
    inc r14
    inc rsi
    dec ecx
    jnz parse_literal
    jmp parse_ok

parse_string:
    inc r14
string_next:
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x22
    je string_close
    cmp al, 0x20
    jb parse_fail
    cmp al, 0x5c
    je string_escape
    inc r14
    jmp string_next

string_close:
    inc r14
    jmp parse_ok

string_escape:
    inc r14
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je simple_escape
    cmp al, 0x5c
    je simple_escape
    cmp al, 0x2f
    je simple_escape
    cmp al, 0x62
    je simple_escape
    cmp al, 0x66
    je simple_escape
    cmp al, 0x6e
    je simple_escape
    cmp al, 0x72
    je simple_escape
    cmp al, 0x74
    jne parse_fail
simple_escape:
    inc r14
    jmp string_next

string_unicode:
    inc r14
    mov ecx, 4
unicode_digit:
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    jbe unicode_hex_ok
    mov edx, eax
    or dl, 0x20
    cmp dl, 0x61
    jb parse_fail
    cmp dl, 0x66
    ja parse_fail
unicode_hex_ok:
    inc r14
    dec ecx
    jnz unicode_digit
    jmp string_next

parse_number:
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2d
    jne number_integer
    inc r14
    cmp r14, r13
    jae parse_fail

number_integer:
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb parse_fail
    cmp al, 0x39
    ja parse_fail

number_integer_digits:
    inc r14
    cmp r14, r13
    jae number_integer_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_integer_done
    cmp al, 0x39
    jbe number_integer_digits
    jmp number_integer_done

number_zero:
    inc r14

number_integer_done:
    cmp r14, r13
    jae parse_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2e
    jne number_exponent_check
    inc r14
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    ja parse_fail

number_fraction_digits:
    inc r14
    cmp r14, r13
    jae parse_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_exponent_check
    cmp al, 0x39
    jbe number_fraction_digits

number_exponent_check:
    cmp r14, r13
    jae parse_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x65
    je number_exponent
    cmp al, 0x45
    jne parse_ok

number_exponent:
    inc r14
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2b
    je number_exponent_sign
    cmp al, 0x2d
    jne number_exponent_first_digit
number_exponent_sign:
    inc r14
    cmp r14, r13
    jae parse_fail

number_exponent_first_digit:
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_fail
    cmp al, 0x39
    ja parse_fail

number_exponent_digits:
    inc r14
    cmp r14, r13
    jae parse_ok
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb parse_ok
    cmp al, 0x39
    jbe number_exponent_digits
    jmp parse_ok

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae parse_fail
    cmp byte ptr [r12 + r14], 0x5d
    je array_close

array_value:
    call parse_value
    test eax, eax
    jz parse_fail
    call skip_ws
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2c
    je array_comma
    cmp al, 0x5d
    je array_close
    jmp parse_fail

array_comma:
    inc r14
    call skip_ws
    jmp array_value

array_close:
    inc r14
    jmp parse_ok

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae parse_fail
    cmp byte ptr [r12 + r14], 0x7d
    je object_close

object_member:
    cmp r14, r13
    jae parse_fail
    cmp byte ptr [r12 + r14], 0x22
    jne parse_fail
    call parse_string
    test eax, eax
    jz parse_fail
    call skip_ws
    cmp r14, r13
    jae parse_fail
    cmp byte ptr [r12 + r14], 0x3a
    jne parse_fail
    inc r14
    call skip_ws
    call parse_value
    test eax, eax
    jz parse_fail
    call skip_ws
    cmp r14, r13
    jae parse_fail
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2c
    je object_comma
    cmp al, 0x7d
    je object_close
    jmp parse_fail

object_comma:
    inc r14
    call skip_ws
    jmp object_member

object_close:
    inc r14
    jmp parse_ok

.section .rodata
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"

.bss
.align 16
input_buf:
    .zero 65536
output_buf:
    .zero 32
