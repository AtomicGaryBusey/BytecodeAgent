.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz input_complete
    add r12, rax
    jmp read_more

input_complete:
    lea r13, [rip + input_buffer]
    lea r14, [rip + input_buffer]
    add r14, r12

next_line:
    cmp r13, r14
    jae finished
    mov r15, r13

find_line_end:
    cmp r13, r14
    jae final_line
    cmp byte ptr [r13], 10
    je newline_line
    inc r13
    jmp find_line_end

newline_line:
    mov rbx, r13
    inc r13
    jmp reverse_line

final_line:
    mov rbx, r13

reverse_line:
    lea rdi, [rip + output_buffer]
    mov rcx, rbx
    sub rcx, r15
    test rcx, rcx
    jz append_newline
    lea rsi, [rbx - 1]

reverse_byte:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rsi
    dec rcx
    jnz reverse_byte

append_newline:
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp next_line

finished:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
input_buffer:
    .zero 65537
output_buffer:
    .zero 65537
