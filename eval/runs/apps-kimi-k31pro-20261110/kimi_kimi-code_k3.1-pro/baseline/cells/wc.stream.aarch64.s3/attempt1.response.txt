.globl _start

.text
_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr
    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_loop:
    mov x0, xzr
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x21, x21, x0
    mov x24, x23
    mov x25, x0

byte_loop:
    ldrb w3, [x24], #1
    cmp w3, #10
    b.eq newline_byte
    cmp w3, #32
    b.eq whitespace_byte
    cmp w3, #9
    b.eq whitespace_byte
    cmp w3, #11
    b.eq whitespace_byte
    cmp w3, #12
    b.eq whitespace_byte
    cmp w3, #13
    b.eq whitespace_byte

    cbnz x22, next_byte
    mov x22, #1
    add x20, x20, #1
    b next_byte

newline_byte:
    add x19, x19, #1
whitespace_byte:
    mov x22, xzr

next_byte:
    subs x25, x25, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer
    mov x26, x23

    mov x0, x19
    mov x1, x26
    bl append_decimal
    mov x26, x1
    mov w4, #32
    strb w4, [x26], #1

    mov x0, x20
    mov x1, x26
    bl append_decimal
    mov x26, x1
    mov w4, #32
    strb w4, [x26], #1

    mov x0, x21
    mov x1, x26
    bl append_decimal
    mov x26, x1
    mov w4, #10
    strb w4, [x26], #1

    mov x0, #1
    mov x1, x23
    sub x2, x26, x23
    mov x8, #64
    svc #0

    mov x0, xzr
    mov x8, #93
    svc #0
1:
    b 1b

append_decimal:
    mov x2, x1
    mov x3, #10
    cbnz x0, 1f
    mov w4, #48
    strb w4, [x1], #1
    ret

1:
    mov x5, x0
2:
    udiv x6, x5, x3
    msub x7, x6, x3, x5
    add w7, w7, #48
    strb w7, [x1], #1
    mov x5, x6
    cbnz x5, 2b

    sub x4, x1, #1
3:
    cmp x2, x4
    b.hs 4f
    ldrb w5, [x2]
    ldrb w6, [x4]
    strb w6, [x2], #1
    strb w5, [x4]
    sub x4, x4, #1
    b 3b
4:
    ret

.bss
.balign 16
input_buffer:
    .space 4096
.balign 16
output_buffer:
    .space 80
