.globl _start

.text
_start:
    lea r12, [rip + input_buffer]
    lea r13, [rip + input_buffer + 65536]

read_loop:
    cmp r12, r13
    jae read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r13
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea r14, [rip + input_buffer]
    mov r13, r12

line_loop:
    cmp r14, r13
    jae program_exit
    mov r12, r14

find_line_end:
    cmp r12, r13
    jae have_line
    cmp byte ptr [r12], 10
    je have_line
    inc r12
    jmp find_line_end

have_line:
    mov rdi, r14
    mov rsi, r12
    call evaluate_line
    lea r14, [r12 + 1]
    jmp line_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

evaluate_line:
    mov r8, rdi
    mov r9, rsi
    xor r10d, r10d
    lea r11, [rip + value_stack]

skip_separators:
    cmp r8, r9
    jae line_done
    mov al, byte ptr [r8]
    cmp al, 32
    je skip_one_separator
    cmp al, 9
    jne token_start

skip_one_separator:
    inc r8
    jmp skip_separators

token_start:
    mov rdx, r8

scan_token:
    cmp r8, r9
    jae token_scanned
    mov al, byte ptr [r8]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    inc r8
    jmp scan_token

token_scanned:
    mov rcx, r8
    sub rcx, rdx
    cmp rcx, 1
    jne parse_literal
    mov al, byte ptr [rdx]
    cmp al, '+'
    je do_add
    cmp al, '-'
    je do_subtract
    cmp al, '*'
    je do_multiply
    cmp al, '/'
    je do_divide
    cmp al, '%'
    je do_remainder

parse_literal:
    mov rsi, rdx
    xor edi, edi
    cmp byte ptr [rsi], '-'
    jne digits_setup
    mov edi, 1
    inc rsi

digits_setup:
    mov rcx, r8
    sub rcx, rsi
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax

digit_loop:
    movzx ebx, byte ptr [rsi]
    sub ebx, '0'
    cmp ebx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rbx
    inc rsi
    dec rcx
    jnz digit_loop
    test edi, edi
    jz push_literal
    neg rax

push_literal:
    mov qword ptr [r11 + r10*8], rax
    inc r10
    jmp skip_separators

do_add:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [r11 + r10*8 - 16]
    mov rax, qword ptr [r11 + r10*8 - 8]
    add rbx, rax
    mov qword ptr [r11 + r10*8 - 16], rbx
    dec r10
    jmp skip_separators

do_subtract:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [r11 + r10*8 - 16]
    mov rax, qword ptr [r11 + r10*8 - 8]
    sub rbx, rax
    mov qword ptr [r11 + r10*8 - 16], rbx
    dec r10
    jmp skip_separators

do_multiply:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [r11 + r10*8 - 16]
    mov rax, qword ptr [r11 + r10*8 - 8]
    imul rbx, rax
    mov qword ptr [r11 + r10*8 - 16], rbx
    dec r10
    jmp skip_separators

do_divide:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [r11 + r10*8 - 16]
    mov rax, qword ptr [r11 + r10*8 - 8]
    test rax, rax
    jz line_error
    cmp rax, -1
    jne divide_ready
    mov rdx, 0x8000000000000000
    cmp rbx, rdx
    je line_error

divide_ready:
    mov rsi, rax
    mov rax, rbx
    cqo
    idiv rsi
    mov qword ptr [r11 + r10*8 - 16], rax
    dec r10
    jmp skip_separators

do_remainder:
    cmp r10, 2
    jb line_error
    mov rbx, qword ptr [r11 + r10*8 - 16]
    mov rax, qword ptr [r11 + r10*8 - 8]
    test rax, rax
    jz line_error
    cmp rax, -1
    jne remainder_ready
    mov rdx, 0x8000000000000000
    cmp rbx, rdx
    je line_error

remainder_ready:
    mov rsi, rax
    mov rax, rbx
    cqo
    idiv rsi
    mov qword ptr [r11 + r10*8 - 16], rdx
    dec r10
    jmp skip_separators

line_done:
    cmp r10, 1
    jne line_error
    mov rax, qword ptr [r11]
    xor edi, edi
    test rax, rax
    jns magnitude_ready
    mov edi, 1
    neg rax

magnitude_ready:
    lea rcx, [rip + output_buffer + 32]
    mov byte ptr [rcx - 1], 10
    lea rsi, [rcx - 1]
    mov rbx, 10

decimal_loop:
    xor edx, edx
    div rbx
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    test edi, edi
    jz write_result
    dec rsi
    mov byte ptr [rsi], '-'

write_result:
    mov rdx, rcx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    ret

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall
    ret

.section .rodata
error_message:
    .ascii "error\n"

.bss
.balign 8
input_buffer:
    .zero 65536
value_stack:
    .zero 8192
output_buffer:
    .zero 32
