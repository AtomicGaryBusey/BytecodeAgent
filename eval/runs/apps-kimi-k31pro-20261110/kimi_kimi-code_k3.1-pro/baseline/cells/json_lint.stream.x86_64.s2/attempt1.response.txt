.globl _start

.text
_start:
    xor r12d, r12d
    lea r13, [rip + input_buf]

read_loop:
    mov rdx, 65537
    sub rdx, r12
    jz input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_loop

input_done:
    mov r14, r13
    lea r15, [r13 + r12]
    call skip_ws
    call parse_value
    test eax, eax
    jz report_error
    call skip_ws
    cmp r14, r15
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

report_error:
    mov rax, r14
    sub rax, r13
    lea rdi, [rip + output_buf]
    lea rsi, [rip + error_prefix]
    mov ecx, 9
    rep movsb

    xor ecx, ecx
    test rax, rax
    jnz convert_loop
    mov byte ptr [rdi], 48
    mov ecx, 1
    jmp converted

convert_loop:
    xor edx, edx
    mov r8d, 10
    div r8
    add dl, 48
    mov byte ptr [rdi + rcx], dl
    inc ecx
    test rax, rax
    jnz convert_loop

converted:
    xor esi, esi
    mov edx, ecx
    dec edx

reverse_loop:
    cmp esi, edx
    jge reverse_done
    mov r8b, byte ptr [rdi + rsi]
    mov r9b, byte ptr [rdi + rdx]
    mov byte ptr [rdi + rsi], r9b
    mov byte ptr [rdi + rdx], r8b
    inc esi
    dec edx
    jmp reverse_loop

reverse_done:
    mov byte ptr [rdi + rcx], 10
    lea rdx, [rcx + 10]
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r14, r15
    jae ws_done
    mov al, byte ptr [r14]
    cmp al, 0x20
    je ws_advance
    cmp al, 0x09
    je ws_advance
    cmp al, 0x0a
    je ws_advance
    cmp al, 0x0d
    jne ws_done
ws_advance:
    inc r14
    jmp skip_ws
ws_done:
    ret

parse_value:
    cmp r14, r15
    jae value_fail
    mov al, byte ptr [r14]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je value_true
    cmp al, 0x66
    je value_false
    cmp al, 0x6e
    je value_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb value_fail
    cmp al, 0x39
    jbe parse_number
value_fail:
    xor eax, eax
    ret
value_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4
    jmp parse_literal

parse_literal:
    cmp r14, r15
    jae literal_fail
    mov al, byte ptr [r14]
    cmp al, byte ptr [rsi]
    jne literal_fail
    inc r14
    inc rsi
    dec ecx
    jnz parse_literal
    mov eax, 1
    ret
literal_fail:
    xor eax, eax
    ret

parse_string:
    inc r14
string_loop:
    cmp r14, r15
    jae string_fail
    mov al, byte ptr [r14]
    cmp al, 0x22
    je string_close
    cmp al, 0x5c
    je string_escape
    cmp al, 0x20
    jb string_fail
    inc r14
    jmp string_loop
string_close:
    inc r14
    mov eax, 1
    ret
string_escape:
    inc r14
    cmp r14, r15
    jae string_fail
    mov al, byte ptr [r14]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je string_simple_escape
    cmp al, 0x5c
    je string_simple_escape
    cmp al, 0x2f
    je string_simple_escape
    cmp al, 0x62
    je string_simple_escape
    cmp al, 0x66
    je string_simple_escape
    cmp al, 0x6e
    je string_simple_escape
    cmp al, 0x72
    je string_simple_escape
    cmp al, 0x74
    jne string_fail
string_simple_escape:
    inc r14
    jmp string_loop
string_unicode:
    inc r14
    mov ecx, 4
unicode_loop:
    cmp r14, r15
    jae string_fail
    mov al, byte ptr [r14]
    cmp al, 0x30
    jb string_fail
    cmp al, 0x39
    jbe unicode_digit
    cmp al, 0x41
    jb string_fail
    cmp al, 0x46
    jbe unicode_digit
    cmp al, 0x61
    jb string_fail
    cmp al, 0x66
    ja string_fail
unicode_digit:
    inc r14
    dec ecx
    jnz unicode_loop
    jmp string_loop
string_fail:
    xor eax, eax
    ret

parse_number:
    cmp byte ptr [r14], 0x2d
    jne number_integer
    inc r14
number_integer:
    cmp r14, r15
    jae number_fail
    mov al, byte ptr [r14]
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb number_fail
    cmp al, 0x39
    ja number_fail
number_integer_digits:
    inc r14
    cmp r14, r15
    jae number_fraction
    mov al, byte ptr [r14]
    cmp al, 0x30
    jb number_fraction
    cmp al, 0x39
    jbe number_integer_digits
    jmp number_fraction
number_zero:
    inc r14
number_fraction:
    cmp r14, r15
    jae number_success
    cmp byte ptr [r14], 0x2e
    jne number_exponent
    inc r14
    cmp r14, r15
    jae number_fail
    mov al, byte ptr [r14]
    cmp al, 0x30
    jb number_fail
    cmp al, 0x39
    ja number_fail
number_fraction_digits:
    inc r14
    cmp r14, r15
    jae number_success
    mov al, byte ptr [r14]
    cmp al, 0x30
    jb number_exponent
    cmp al, 0x39
    jbe number_fraction_digits
number_exponent:
    cmp r14, r15
    jae number_success
    mov al, byte ptr [r14]
    cmp al, 0x65
    je number_exp_start
    cmp al, 0x45
    jne number_success
number_exp_start:
    inc r14
    cmp r14, r15
    jae number_fail
    mov al, byte ptr [r14]
    cmp al, 0x2b
    je number_exp_sign
    cmp al, 0x2d
    jne number_exp_require_digit
number_exp_sign:
    inc r14
number_exp_require_digit:
    cmp r14, r15
    jae number_fail
    mov al, byte ptr [r14]
    cmp al, 0x30
    jb number_fail
    cmp al, 0x39
    ja number_fail
number_exp_digits:
    inc r14
    cmp r14, r15
    jae number_success
    mov al, byte ptr [r14]
    cmp al, 0x30
    jb number_success
    cmp al, 0x39
    jbe number_exp_digits
number_success:
    mov eax, 1
    ret
number_fail:
    xor eax, eax
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r15
    jae array_fail
    cmp byte ptr [r14], 0x5d
    jne array_value
    inc r14
    mov eax, 1
    ret
array_value:
    call parse_value
    test eax, eax
    jz array_fail
    call skip_ws
    cmp r14, r15
    jae array_fail
    mov al, byte ptr [r14]
    cmp al, 0x5d
    je array_close
    cmp al, 0x2c
    jne array_fail
    inc r14
    call skip_ws
    jmp array_value
array_close:
    inc r14
    mov eax, 1
    ret
array_fail:
    xor eax, eax
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r15
    jae object_fail
    cmp byte ptr [r14], 0x7d
    jne object_member
    inc r14
    mov eax, 1
    ret
object_member:
    cmp r14, r15
    jae object_fail
    cmp byte ptr [r14], 0x22
    jne object_fail
    call parse_string
    test eax, eax
    jz object_fail
    call skip_ws
    cmp r14, r15
    jae object_fail
    cmp byte ptr [r14], 0x3a
    jne object_fail
    inc r14
    call skip_ws
    call parse_value
    test eax, eax
    jz object_fail
    call skip_ws
    cmp r14, r15
    jae object_fail
    mov al, byte ptr [r14]
    cmp al, 0x7d
    je object_close
    cmp al, 0x2c
    jne object_fail
    inc r14
    call skip_ws
    jmp object_member
object_close:
    inc r14
    mov eax, 1
    ret
object_fail:
    xor eax, eax
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
input_buf:
    .zero 65537
output_buf:
    .zero 64
