.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    mov x21, #32768

read_loop:
    cmp x20, x21
    b.eq input_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_done
    add x20, x20, x0
    b read_loop

input_done:
    cbz x20, exit_success

    mov x22, #0
    mov x23, #0
    adrp x24, records
    add x24, x24, :lo12:records

parse_outer:
    cmp x22, x20
    b.hs parse_done
    mov x25, x22

scan_line:
    cmp x22, x20
    b.hs line_ready
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq line_ready
    add x22, x22, #1
    b scan_line

line_ready:
    sub x26, x22, x25
    add x0, x19, x25
    add x1, x24, x23, lsl #4
    stp x0, x26, [x1]
    add x23, x23, #1
    cmp x22, x20
    b.hs parse_done
    add x22, x22, #1
    b parse_outer

parse_done:
    adrp x19, records
    add x19, x19, :lo12:records
    adrp x20, temp_records
    add x20, x20, :lo12:temp_records
    mov x21, x23
    mov x22, #1
    cmp x21, #1
    b.ls sort_done

sort_pass:
    mov x23, #0

run_loop:
    cmp x23, x21
    b.hs pass_done

    mov x24, x23
    add x25, x23, x22
    cmp x25, x21
    csel x25, x25, x21, ls
    add x27, x25, x22
    cmp x27, x21
    csel x27, x27, x21, ls
    mov x26, x25
    mov x28, x23

merge_loop:
    cmp x24, x25
    b.eq copy_right_loop
    cmp x26, x27
    b.eq copy_left_loop

    add x5, x19, x24, lsl #4
    ldp x0, x1, [x5]
    add x5, x19, x26, lsl #4
    ldp x2, x3, [x5]
    bl compare_lines
    cbz w0, choose_right

    add x5, x19, x24, lsl #4
    ldp x6, x7, [x5]
    add x5, x20, x28, lsl #4
    stp x6, x7, [x5]
    add x24, x24, #1
    add x28, x28, #1
    b merge_loop

choose_right:
    add x5, x19, x26, lsl #4
    ldp x6, x7, [x5]
    add x5, x20, x28, lsl #4
    stp x6, x7, [x5]
    add x26, x26, #1
    add x28, x28, #1
    b merge_loop

copy_left_loop:
    cmp x24, x25
    b.eq run_done
    add x5, x19, x24, lsl #4
    ldp x6, x7, [x5]
    add x5, x20, x28, lsl #4
    stp x6, x7, [x5]
    add x24, x24, #1
    add x28, x28, #1
    b copy_left_loop

copy_right_loop:
    cmp x26, x27
    b.eq run_done
    add x5, x19, x26, lsl #4
    ldp x6, x7, [x5]
    add x5, x20, x28, lsl #4
    stp x6, x7, [x5]
    add x26, x26, #1
    add x28, x28, #1
    b copy_right_loop

run_done:
    mov x23, x27
    b run_loop

pass_done:
    mov x0, x19
    mov x19, x20
    mov x20, x0
    lsl x22, x22, #1
    cmp x22, x21
    b.lo sort_pass

sort_done:
    mov x22, #0

output_loop:
    cmp x22, x21
    b.hs exit_success
    add x5, x19, x22, lsl #4
    ldp x1, x2, [x5]
    cbz x2, output_newline
    mov x0, #1
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #64
    svc #0
    add x22, x22, #1
    b output_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

compare_lines:
    mov x4, #0
    cmp x1, x3
    csel x5, x1, x3, ls

compare_loop:
    cmp x4, x5
    b.hs compare_prefix
    ldrb w6, [x0, x4]
    ldrb w7, [x2, x4]
    cmp w6, w7
    b.ne compare_different
    add x4, x4, #1
    b compare_loop

compare_different:
    mov w0, #1
    b.lo compare_return
    mov w0, #0
    ret

compare_prefix:
    cmp x1, x3
    mov w0, #1
    b.ls compare_return
    mov w0, #0

compare_return:
    ret

.section .rodata

    .byte 10

.bss
.balign 16

    .zero 32768

    .zero 16000

    .zero 16000
