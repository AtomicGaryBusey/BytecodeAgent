.globl _start

.text
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0
    mov x21, #65536

read_loop:
    cmp x19, x21
    b.hs read_done
    sub x2, x21, x19
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    add x22, x20, x19
    mov x23, x20
    mov x24, x19

line_loop:
    cmp x23, x22
    b.hs write_output
    mov x25, x23
    mov x26, x23

scan_line:
    cmp x26, x22
    b.hs final_line
    ldrb w0, [x26]
    cmp w0, #10
    b.eq terminated_line
    add x26, x26, #1
    b scan_line

terminated_line:
    mov x0, x25
    mov x1, x26
    bl reverse_range
    add x23, x26, #1
    b line_loop

final_line:
    mov x0, x25
    mov x1, x22
    bl reverse_range
    mov w0, #10
    strb w0, [x22]
    add x24, x24, #1
    mov x23, x22
    b line_loop

write_output:
    cbz x24, exit_program
    mov x25, x20
    mov x26, x24

write_loop:
    mov x0, #1
    mov x1, x25
    mov x2, x26
    mov x8, #64
    svc #0
    add x25, x25, x0
    sub x26, x26, x0
    cbnz x26, write_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

reverse_range:
    cmp x0, x1
    b.hs 2f
    sub x1, x1, #1
1:
    cmp x0, x1
    b.hs 2f
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w3, [x0], #1
    strb w2, [x1], #-1
    b 1b

    ret
