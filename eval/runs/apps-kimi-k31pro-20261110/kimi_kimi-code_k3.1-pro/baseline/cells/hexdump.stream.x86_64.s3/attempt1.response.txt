.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 323593

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12d, 65536
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    lea r14, [rip + input_buffer]
    lea r13, [rip + output_buffer]
    xor r15d, r15d
    lea rsi, [rip + hex_digits]

next_row:
    cmp r15, r12
    jae final_offset

    mov rbx, r12
    sub rbx, r15
    cmp rbx, 16
    jbe row_size_ready
    mov ebx, 16
row_size_ready:

    mov eax, r15d
    mov ecx, 28
1:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [r13], dl
    inc r13
    sub ecx, 4
    jns 1b

    mov byte ptr [r13], 32
    mov byte ptr [r13 + 1], 32
    add r13, 2

    xor ecx, ecx
2:
    cmp rcx, rbx
    jae 3f
    movzx eax, byte ptr [r14 + rcx]
    mov edx, eax
    shr eax, 4
    and edx, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [r13], al
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [r13 + 1], al
    mov byte ptr [r13 + 2], 32
    add r13, 3
    jmp 4f
3:
    mov byte ptr [r13], 32
    mov byte ptr [r13 + 1], 32
    mov byte ptr [r13 + 2], 32
    add r13, 3
4:
    inc rcx
    cmp rcx, 8
    jne 5f
    mov byte ptr [r13], 32
    inc r13
5:
    cmp rcx, 16
    jb 2b

    mov byte ptr [r13], 32
    mov byte ptr [r13 + 1], 124
    add r13, 2

    xor ecx, ecx
6:
    cmp rcx, rbx
    jae 7f
    mov al, byte ptr [r14 + rcx]
    cmp al, 32
    jb 8f
    cmp al, 126
    jbe 9f
8:
    mov al, 46
9:
    mov byte ptr [r13], al
    inc r13
    inc rcx
    jmp 6b
7:
    mov byte ptr [r13], 124
    mov byte ptr [r13 + 1], 10
    add r13, 2

    add r14, rbx
    add r15, rbx
    jmp next_row

final_offset:
    mov eax, r12d
    mov ecx, 28
10:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [r13], dl
    inc r13
    sub ecx, 4
    jns 10b
    mov byte ptr [r13], 10
    inc r13

    mov rdx, r13
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
