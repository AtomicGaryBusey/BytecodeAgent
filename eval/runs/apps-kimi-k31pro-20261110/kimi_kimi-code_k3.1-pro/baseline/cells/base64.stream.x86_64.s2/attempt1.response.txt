.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 49152
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    test r12, r12
    jz exit_success

    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea r8, [rip + alphabet]
    xor r13d, r13d
    xor r15d, r15d

next_group:
    mov rax, r12
    sub rax, r13
    cmp rax, 3
    jae full_group
    test rax, rax
    jz encoding_done
    cmp rax, 1
    je one_byte_group
    jmp two_byte_group

full_group:
    movzx eax, byte ptr [rsi + r13]
    shl eax, 16
    movzx ecx, byte ptr [rsi + r13 + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + r13 + 2]
    or eax, ecx

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], dl

    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], al

    add rdi, 4
    add r13, 3
    jmp group_done

one_byte_group:
    movzx eax, byte ptr [rsi + r13]
    shl eax, 16

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl

    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r13, 1
    jmp group_done

two_byte_group:
    movzx eax, byte ptr [rsi + r13]
    shl eax, 16
    movzx ecx, byte ptr [rsi + r13 + 1]
    shl ecx, 8
    or eax, ecx

    mov edx, eax
    shr edx, 18
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], dl

    shr eax, 6
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], al

    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r13, 2

group_done:
    inc r15d
    cmp r15d, 19
    jne next_group
    mov byte ptr [rdi], 10
    inc rdi
    xor r15d, r15d
    jmp next_group

encoding_done:
    test r15d, r15d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero 49152
.align 16
output_buffer:
    .zero 66560
