.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_again:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    lea rdi, [rsi + rax]

scan_byte:
    movzx ecx, byte ptr [rsi]
    inc rsi

    cmp cl, 10
    jne 1f
    inc r12
    jmp 2f

1:
    cmp cl, 32
    je 2f
    cmp cl, 9
    jb 3f
    cmp cl, 13
    jbe 2f

3:
    test r15d, r15d
    jnz 4f
    inc r13
    mov r15d, 1
    jmp 4f

2:
    xor r15d, r15d

4:
    cmp rsi, rdi
    jb scan_byte
    jmp read_again

input_done:
    lea rdi, [rip + output_buffer + 64]
    dec rdi
    mov byte ptr [rdi], 10

    mov rax, r14
    call prepend_number
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call prepend_number
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call prepend_number

    lea rdx, [rip + output_buffer + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

prepend_number:
    mov r8d, 10
    test rax, rax
    jnz 1f
    dec rdi
    mov byte ptr [rdi], 48
    ret

1:
    xor edx, edx
    div r8
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
