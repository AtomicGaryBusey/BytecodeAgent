.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
line_buf:
    .zero 80

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae process_input

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall

    test rax, rax
    jle process_input
    add r12, rax
    jmp read_more

process_input:
    lea rbp, [rip + hex_digits]
    xor r13d, r13d

next_row:
    cmp r13, r12
    jae print_total

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_length_ready
    mov r14d, 16
row_length_ready:

    mov rax, 0x2020202020202020
    mov qword ptr [rip + line_buf], rax
    mov qword ptr [rip + line_buf + 8], rax
    mov qword ptr [rip + line_buf + 16], rax
    mov qword ptr [rip + line_buf + 24], rax
    mov qword ptr [rip + line_buf + 32], rax
    mov qword ptr [rip + line_buf + 40], rax
    mov qword ptr [rip + line_buf + 48], rax
    mov qword ptr [rip + line_buf + 56], rax
    mov byte ptr [rip + line_buf + 60], 124

    mov rax, r13
    lea rdi, [rip + line_buf + 8]
    mov ecx, 8
offset_digit:
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    dec rdi
    mov byte ptr [rdi], dl
    shr rax, 4
    dec ecx
    jnz offset_digit

    lea rsi, [rip + input_buf]
    add rsi, r13
    lea rdi, [rip + line_buf + 10]
    lea r8, [rip + line_buf + 61]
    xor ecx, ecx

byte_loop:
    movzx eax, byte ptr [rsi + rcx]

    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 1], dl

    cmp al, 32
    jb not_printable
    cmp al, 126
    ja not_printable
    mov byte ptr [r8 + rcx], al
    jmp byte_advance

not_printable:
    mov byte ptr [r8 + rcx], 46

byte_advance:
    add rdi, 3
    cmp ecx, 7
    jne no_extra_space
    add rdi, 1
no_extra_space:
    inc ecx
    cmp rcx, r14
    jb byte_loop

    lea rax, [rcx + 61]
    lea rdi, [rip + line_buf]
    mov byte ptr [rdi + rax], 124
    mov byte ptr [rdi + rax + 1], 10
    add rax, 2

    mov rdx, rax
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    syscall

    add r13, r14
    jmp next_row

print_total:
    mov rax, r12
    lea rdi, [rip + line_buf + 8]
    mov ecx, 8
total_digit:
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    dec rdi
    mov byte ptr [rdi], dl
    shr rax, 4
    dec ecx
    jnz total_digit

    mov byte ptr [rip + line_buf + 8], 10
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall
