.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    cmp r12, 32768
    jb read_more

input_done:
    test r12, r12
    jz finish

    lea rbx, [rip + input_buffer]
    mov r14, rbx
    lea r15, [rbx + r12]
    lea r13, [rip + line_array]
    xor ebp, ebp

next_line:
    cmp r14, r15
    jae lines_ready
    mov r8, r14

scan_line:
    cmp r14, r15
    jae line_scanned
    cmp byte ptr [r14], 10
    je line_scanned
    inc r14
    jmp scan_line

line_scanned:
    mov [r13], r8
    mov rax, r14
    sub rax, r8
    mov [r13 + 8], rax
    add r13, 16
    inc ebp
    cmp r14, r15
    je lines_ready
    inc r14
    jmp next_line

lines_ready:
    lea r12, [rip + line_array]
    cmp ebp, 1
    jle make_output

    lea r13, [rip + temp_array]
    mov r14d, 1

width_loop:
    xor r15d, r15d

chunk_loop:
    mov r8, r15
    lea r10, [r15 + r14]
    cmp r10, rbp
    cmova r10, rbp
    mov r9, r10
    lea r11, [r10 + r14]
    cmp r11, rbp
    cmova r11, rbp
    mov rax, r15
    shl rax, 4
    lea rbx, [r13 + rax]

merge_loop:
    cmp r8, r10
    jge drain_right
    cmp r9, r11
    jge drain_left

    mov rax, r8
    shl rax, 4
    lea rdi, [r12 + rax]
    mov rax, r9
    shl rax, 4
    lea rsi, [r12 + rax]
    call compare_lines
    test eax, eax
    jg take_right

    mov rax, r8
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc r8
    add rbx, 16
    jmp merge_loop

take_right:
    mov rax, r9
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc r9
    add rbx, 16
    jmp merge_loop

drain_left:
    cmp r8, r10
    jge chunk_done
    mov rax, r8
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc r8
    add rbx, 16
    jmp drain_left

drain_right:
    cmp r9, r11
    jge chunk_done
    mov rax, r9
    shl rax, 4
    movups xmm0, xmmword ptr [r12 + rax]
    movups xmmword ptr [rbx], xmm0
    inc r9
    add rbx, 16
    jmp drain_right

chunk_done:
    lea r15, [r15 + r14*2]
    cmp r15, rbp
    jl chunk_loop

    xchg r12, r13
    add r14, r14
    cmp r14, rbp
    jl width_loop

make_output:
    mov r13, r12
    lea r14, [rip + output_buffer]
    mov rdi, r14
    mov r15d, ebp

output_loop:
    test r15d, r15d
    jz write_output
    mov rsi, [r13]
    mov rcx, [r13 + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    add r13, 16
    dec r15d
    jmp output_loop

write_output:
    mov rdx, rdi
    sub rdx, r14
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

finish:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    push r8
    push r9
    push r10
    push r11
    mov r8, [rdi]
    mov r9, [rsi]
    mov r10, [rdi + 8]
    mov r11, [rsi + 8]
    mov rcx, r10
    cmp rcx, r11
    cmova rcx, r11
    mov rdi, r8
    mov rsi, r9
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    jb line_less
    ja line_greater

compare_lengths:
    cmp r10, r11
    jb line_less
    ja line_greater
    xor eax, eax
    jmp compare_done

line_less:
    mov eax, -1
    jmp compare_done

line_greater:
    mov eax, 1

compare_done:
    pop r11
    pop r10
    pop r9
    pop r8
    ret

.bss
.balign 16
input_buffer:
    .zero 32768
.balign 16
line_array:
    .zero 16000
.balign 16
temp_array:
    .zero 16000
.balign 16
output_buffer:
    .zero 33768
