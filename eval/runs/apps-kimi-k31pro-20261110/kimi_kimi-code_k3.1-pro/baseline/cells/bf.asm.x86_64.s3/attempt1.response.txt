.globl _start

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_more

read_done:
    lea rsi, [rip + input_buf]
    lea r15, [rsi + r12]
    mov r14, r15
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rsi + rcx], 0x21
    je separator_found
    inc rcx
    jmp find_separator

separator_found:
    mov rdx, rcx
    lea r14, [rsi + rcx + 1]
    jmp begin_parse

no_separator:
    mov rdx, r12

begin_parse:
    xor ebx, ebx
    xor ecx, ecx
    xor r8d, r8d
    lea r10, [rip + code_buf]
    lea r11, [rip + jump_buf]
    lea rdi, [rip + bracket_stack]

parse_loop:
    cmp rcx, rdx
    jae parse_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 62
    je parse_gt
    cmp al, 60
    je parse_lt
    cmp al, 43
    je parse_plus
    cmp al, 45
    je parse_minus
    cmp al, 46
    je parse_dot
    cmp al, 44
    je parse_comma
    cmp al, 91
    je parse_open
    cmp al, 93
    je parse_close
    inc rcx
    jmp parse_loop

parse_gt:
    mov byte ptr [r10 + rbx], 1
    jmp command_parsed

parse_lt:
    mov byte ptr [r10 + rbx], 2
    jmp command_parsed

parse_plus:
    mov byte ptr [r10 + rbx], 3
    jmp command_parsed

parse_minus:
    mov byte ptr [r10 + rbx], 4
    jmp command_parsed

parse_dot:
    mov byte ptr [r10 + rbx], 5
    jmp command_parsed

parse_comma:
    mov byte ptr [r10 + rbx], 6
    jmp command_parsed

parse_open:
    cmp r8d, 1000
    jae invalid_program
    mov dword ptr [rdi + r8*4], ebx
    inc r8d
    mov byte ptr [r10 + rbx], 7
    jmp command_parsed

parse_close:
    test r8d, r8d
    jz invalid_program
    dec r8d
    mov eax, dword ptr [rdi + r8*4]
    mov r9d, ebx
    inc r9d
    mov dword ptr [r11 + rax*4], r9d
    inc eax
    mov dword ptr [r11 + rbx*4], eax
    mov byte ptr [r10 + rbx], 8

command_parsed:
    inc rbx
    inc rcx
    jmp parse_loop

parse_done:
    test r8d, r8d
    jnz invalid_program

    mov r8, rbx
    xor ebx, ebx
    lea r9, [rip + op_table]
    lea r13, [rip + tape]
    lea rbp, [rip + output_buf]

execute:
    cmp rbx, r8
    jae finish_program
    movzx eax, byte ptr [r10 + rbx]
    jmp qword ptr [r9 + rax*8]

op_gt:
    inc r13
    inc rbx
    jmp execute

op_lt:
    dec r13
    inc rbx
    jmp execute

op_plus:
    inc byte ptr [r13]
    inc rbx
    jmp execute

op_minus:
    dec byte ptr [r13]
    inc rbx
    jmp execute

op_dot:
    mov al, byte ptr [r13]
    mov byte ptr [rbp], al
    inc rbp
    inc rbx
    jmp execute

op_comma:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [r13], al
    inc rbx
    jmp execute

input_exhausted:
    mov byte ptr [r13], 0
    inc rbx
    jmp execute

op_open:
    cmp byte ptr [r13], 0
    je open_jump
    inc rbx
    jmp execute

open_jump:
    mov ebx, dword ptr [r11 + rbx*4]
    jmp execute

op_close:
    cmp byte ptr [r13], 0
    jne close_jump
    inc rbx
    jmp execute

close_jump:
    mov ebx, dword ptr [r11 + rbx*4]
    jmp execute

finish_program:
    lea rsi, [rip + output_buf]
    mov rdx, rbp
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
.align 8
op_table:
    .quad finish_program
    .quad op_gt
    .quad op_lt
    .quad op_plus
    .quad op_minus
    .quad op_dot
    .quad op_comma
    .quad op_open
    .quad op_close

.bss
.align 16
input_buf:
    .space 65537
code_buf:
    .space 65536
jump_buf:
    .space 262144
bracket_stack:
    .space 4000
tape:
    .space 30000
output_buf:
    .space 65536
