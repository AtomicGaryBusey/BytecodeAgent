.globl _start

.text
_start:
    xor r12d, r12d
    lea r14, [rip + input_buffer]

read_input:
    cmp r12, 65536
    jae input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle input_ready
    add r12, rax
    jmp read_input

input_ready:
    xor r13d, r13d
    call skip_ws
    call parse_value
    test eax, eax
    jne report_error
    call skip_ws
    cmp r13, r12
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

report_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_prefix]
    mov edx, 9
    syscall

    mov rax, r13
    lea rdi, [rip + number_buffer + 32]
    mov byte ptr [rdi - 1], 10
    lea rsi, [rdi - 1]
    mov ecx, 10

convert_offset:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jne convert_offset

    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    cmp r13, r12
    jae skip_ws_done
    movzx eax, byte ptr [r14 + r13]
    cmp al, 32
    je skip_ws_byte
    cmp al, 9
    je skip_ws_byte
    cmp al, 10
    je skip_ws_byte
    cmp al, 13
    jne skip_ws_done
skip_ws_byte:
    inc r13
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp r13, r12
    jae parse_value_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 116
    je value_true
    cmp al, 102
    je value_false
    cmp al, 110
    je value_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb parse_value_fail
    cmp al, 57
    jbe parse_number
parse_value_fail:
    mov eax, 1
    ret

value_true:
    lea rdx, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal
value_false:
    lea rdx, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal
value_null:
    lea rdx, [rip + literal_null]
    mov ecx, 4

parse_literal:
    xor r8d, r8d
literal_loop:
    cmp r8, rcx
    jae literal_success
    cmp r13, r12
    jae literal_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, byte ptr [rdx + r8]
    jne literal_fail
    inc r13
    inc r8
    jmp literal_loop
literal_success:
    xor eax, eax
    ret
literal_fail:
    mov eax, 1
    ret

parse_string:
    inc r13
string_loop:
    cmp r13, r12
    jae string_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 34
    je string_close
    cmp al, 92
    je string_escape
    cmp al, 32
    jb string_fail
    inc r13
    jmp string_loop

string_close:
    inc r13
    xor eax, eax
    ret

string_escape:
    inc r13
    cmp r13, r12
    jae string_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 34
    je simple_escape
    cmp al, 92
    je simple_escape
    cmp al, 47
    je simple_escape
    cmp al, 98
    je simple_escape
    cmp al, 102
    je simple_escape
    cmp al, 110
    je simple_escape
    cmp al, 114
    je simple_escape
    cmp al, 116
    je simple_escape
    cmp al, 117
    jne string_fail

    inc r13
    mov r8d, 4
unicode_escape_loop:
    cmp r13, r12
    jae string_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    jb string_fail
    cmp al, 57
    jbe unicode_hex_valid
    cmp al, 65
    jb string_fail
    cmp al, 70
    jbe unicode_hex_valid
    cmp al, 97
    jb string_fail
    cmp al, 102
    ja string_fail
unicode_hex_valid:
    inc r13
    dec r8d
    jne unicode_escape_loop
    jmp string_loop

simple_escape:
    inc r13
    jmp string_loop

string_fail:
    mov eax, 1
    ret

parse_number:
    cmp byte ptr [r14 + r13], 45
    jne number_integer
    inc r13

number_integer:
    cmp r13, r12
    jae number_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    je number_zero
    cmp al, 49
    jb number_fail
    cmp al, 57
    ja number_fail

number_integer_digits:
    inc r13
    cmp r13, r12
    jae number_success
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    jb number_after_integer
    cmp al, 57
    jbe number_integer_digits
    jmp number_after_integer

number_zero:
    inc r13
    cmp r13, r12
    jae number_success
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    jb number_after_integer
    cmp al, 57
    jbe number_fail

number_after_integer:
    cmp al, 46
    je number_fraction
    cmp al, 101
    je number_exponent
    cmp al, 69
    je number_exponent
    jmp number_success

number_fraction:
    inc r13
    cmp r13, r12
    jae number_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    jb number_fail
    cmp al, 57
    ja number_fail

number_fraction_digits:
    inc r13
    cmp r13, r12
    jae number_success
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    jb number_after_fraction
    cmp al, 57
    jbe number_fraction_digits
number_after_fraction:
    cmp al, 101
    je number_exponent
    cmp al, 69
    jne number_success

number_exponent:
    inc r13
    cmp r13, r12
    jae number_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 43
    je number_exponent_sign
    cmp al, 45
    jne number_exponent_first_digit
number_exponent_sign:
    inc r13
    cmp r13, r12
    jae number_fail
    movzx eax, byte ptr [r14 + r13]
number_exponent_first_digit:
    cmp al, 48
    jb number_fail
    cmp al, 57
    ja number_fail

number_exponent_digits:
    inc r13
    cmp r13, r12
    jae number_success
    movzx eax, byte ptr [r14 + r13]
    cmp al, 48
    jb number_success
    cmp al, 57
    jbe number_exponent_digits

number_success:
    xor eax, eax
    ret
number_fail:
    mov eax, 1
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r12
    jae array_fail
    cmp byte ptr [r14 + r13], 93
    je array_close

array_value:
    call parse_value
    test eax, eax
    jne array_fail
    call skip_ws
    cmp r13, r12
    jae array_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 93
    je array_close
    cmp al, 44
    jne array_fail
    inc r13
    call skip_ws
    jmp array_value

array_close:
    inc r13
    xor eax, eax
    ret
array_fail:
    mov eax, 1
    ret

parse_object:
    inc r13
    call skip_ws
    cmp r13, r12
    jae object_fail
    cmp byte ptr [r14 + r13], 125
    je object_close

object_member:
    cmp r13, r12
    jae object_fail
    cmp byte ptr [r14 + r13], 34
    jne object_fail
    call parse_string
    test eax, eax
    jne object_fail
    call skip_ws
    cmp r13, r12
    jae object_fail
    cmp byte ptr [r14 + r13], 58
    jne object_fail
    inc r13
    call skip_ws
    call parse_value
    test eax, eax
    jne object_fail
    call skip_ws
    cmp r13, r12
    jae object_fail
    movzx eax, byte ptr [r14 + r13]
    cmp al, 125
    je object_close
    cmp al, 44
    jne object_fail
    inc r13
    call skip_ws
    jmp object_member

object_close:
    inc r13
    xor eax, eax
    ret
object_fail:
    mov eax, 1
    ret

.section .rodata
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"
ok_message:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "

.bss
.align 16
input_buffer:
    .zero 65536
number_buffer:
    .zero 32
