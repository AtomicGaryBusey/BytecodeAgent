.section .bss
.balign 16
input_buf:
    .space 65536
output_buf:
    .space 65537

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x18, #65536
    mov x20, #0

read_loop:
    cmp x20, x18
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x18, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

line_start:
    cmp x21, x20
    b.hs finished
    mov x22, x21

scan_line:
    ldrb w9, [x19, x21]
    cmp w9, #10
    b.eq reverse_line
    add x21, x21, #1
    cmp x21, x20
    b.lo scan_line

reverse_line:
    sub x23, x21, x22
    add x2, x19, x22
    add x2, x2, x23
    mov x3, x24
    mov x4, x23
    cbz x4, append_newline

reverse_loop:
    ldrb w5, [x2, #-1]!
    strb w5, [x3], #1
    subs x4, x4, #1
    b.ne reverse_loop

append_newline:
    mov w5, #10
    strb w5, [x3]
    mov x0, #1
    mov x1, x24
    add x2, x23, #1
    mov x8, #64
    svc #0

    cmp x21, x20
    b.hs finished
    add x21, x21, #1
    b line_start

finished:
    mov x0, #0
    mov x8, #93
    svc #0
