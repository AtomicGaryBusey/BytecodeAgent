.globl _start

.text
_start:
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r15
    mov edx, 65536
    sub rdx, r15
    jnz read_now
    mov edx, 1

read_now:
    syscall
    test rax, rax
    jz input_done
    add r15, rax
    cmp r15, 65536
    jbe read_loop

input_done:
    lea r12, [rip + input]
    mov r14, r15
    xor r13d, r13d

    call skip_ws
    call parse_value
    test eax, eax
    jz report_error
    call skip_ws
    cmp r13, r14
    jne report_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

report_error:
    lea rdi, [rip + outbuf]
    mov dword ptr [rdi], 0x6f727265
    mov dword ptr [rdi + 4], 0x74612072
    mov byte ptr [rdi + 8], 0x20

    lea r8, [rip + outbuf + 9]
    mov rax, r13
    mov ebx, 10
    xor ecx, ecx

decimal_loop:
    xor edx, edx
    div rbx
    add dl, 0x30
    mov byte ptr [r8 + rcx], dl
    inc ecx
    test rax, rax
    jnz decimal_loop

    mov rsi, r8
    lea rdi, [r8 + rcx - 1]

reverse_digits:
    cmp rsi, rdi
    jae digits_reversed
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp reverse_digits

digits_reversed:
    mov byte ptr [r8 + rcx], 0x0a
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    lea rdx, [rcx + 10]
    syscall
    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    cmp r13, r14
    jae skip_ws_done
    mov al, byte ptr [r12 + r13]
    cmp al, 0x20
    je skip_ws_consume
    cmp al, 0x09
    je skip_ws_consume
    cmp al, 0x0a
    je skip_ws_consume
    cmp al, 0x0d
    je skip_ws_consume

skip_ws_done:
    ret

skip_ws_consume:
    inc r13
    jmp skip_ws

parse_value:
    cmp r13, r14
    jae parse_value_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb parse_value_not_number
    cmp al, 0x39
    jbe parse_number

parse_value_not_number:
    cmp al, 0x74
    je value_true
    cmp al, 0x66
    je value_false
    cmp al, 0x6e
    je value_null

parse_value_fail:
    xor eax, eax
    ret

value_true:
    lea rsi, [rip + literal_true]
    mov edx, 4
    jmp parse_literal

value_false:
    lea rsi, [rip + literal_false]
    mov edx, 5
    jmp parse_literal

value_null:
    lea rsi, [rip + literal_null]
    mov edx, 4
    jmp parse_literal

parse_literal:
    xor ecx, ecx

literal_loop:
    cmp ecx, edx
    jae literal_success
    cmp r13, r14
    jae literal_fail
    mov al, byte ptr [r12 + r13]
    cmp al, byte ptr [rsi + rcx]
    jne literal_fail
    inc r13
    inc ecx
    jmp literal_loop

literal_success:
    mov eax, 1
    ret

literal_fail:
    xor eax, eax
    ret

parse_string:
    inc r13

string_loop:
    cmp r13, r14
    jae string_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x22
    je string_close
    cmp al, 0x5c
    je string_escape
    cmp al, 0x20
    jb string_fail
    inc r13
    jmp string_loop

string_close:
    inc r13
    mov eax, 1
    ret

string_escape:
    inc r13
    cmp r13, r14
    jae string_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je string_simple_escape
    cmp al, 0x5c
    je string_simple_escape
    cmp al, 0x2f
    je string_simple_escape
    cmp al, 0x62
    je string_simple_escape
    cmp al, 0x66
    je string_simple_escape
    cmp al, 0x6e
    je string_simple_escape
    cmp al, 0x72
    je string_simple_escape
    cmp al, 0x74
    je string_simple_escape
    jmp string_fail

string_simple_escape:
    inc r13
    jmp string_loop

string_unicode:
    inc r13
    mov ecx, 4

unicode_digit_loop:
    cmp r13, r14
    jae string_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    jb string_fail
    cmp al, 0x39
    jbe unicode_digit_ok
    cmp al, 0x41
    jb string_fail
    cmp al, 0x46
    jbe unicode_digit_ok
    cmp al, 0x61
    jb string_fail
    cmp al, 0x66
    ja string_fail

unicode_digit_ok:
    inc r13
    dec ecx
    jnz unicode_digit_loop
    jmp string_loop

string_fail:
    xor eax, eax
    ret

parse_number:
    cmp byte ptr [r12 + r13], 0x2d
    jne number_integer
    inc r13

number_integer:
    cmp r13, r14
    jae number_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb number_fail
    cmp al, 0x39
    ja number_fail

number_integer_digits:
    inc r13
    cmp r13, r14
    jae number_success
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    jb number_after_integer
    cmp al, 0x39
    jbe number_integer_digits
    jmp number_after_integer

number_zero:
    inc r13

number_after_integer:
    cmp r13, r14
    jae number_success
    mov al, byte ptr [r12 + r13]
    cmp al, 0x2e
    je number_fraction
    cmp al, 0x65
    je number_exponent
    cmp al, 0x45
    je number_exponent
    jmp number_success

number_fraction:
    inc r13
    cmp r13, r14
    jae number_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    jb number_fail
    cmp al, 0x39
    ja number_fail

number_fraction_digits:
    inc r13
    cmp r13, r14
    jae number_success
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    jb number_after_fraction
    cmp al, 0x39
    jbe number_fraction_digits

number_after_fraction:
    cmp al, 0x65
    je number_exponent
    cmp al, 0x45
    je number_exponent
    jmp number_success

number_exponent:
    inc r13
    cmp r13, r14
    jae number_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x2b
    je number_exponent_sign
    cmp al, 0x2d
    jne number_exponent_first_digit

number_exponent_sign:
    inc r13
    cmp r13, r14
    jae number_fail

number_exponent_first_digit:
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    jb number_fail
    cmp al, 0x39
    ja number_fail

number_exponent_digits:
    inc r13
    cmp r13, r14
    jae number_success
    mov al, byte ptr [r12 + r13]
    cmp al, 0x30
    jb number_success
    cmp al, 0x39
    jbe number_exponent_digits

number_success:
    mov eax, 1
    ret

number_fail:
    xor eax, eax
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae array_fail
    cmp byte ptr [r12 + r13], 0x5d
    je array_close

array_value:
    call parse_value
    test eax, eax
    jz array_fail
    call skip_ws
    cmp r13, r14
    jae array_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x5d
    je array_close
    cmp al, 0x2c
    jne array_fail
    inc r13
    call skip_ws
    jmp array_value

array_close:
    inc r13
    mov eax, 1
    ret

array_fail:
    xor eax, eax
    ret

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae object_fail
    cmp byte ptr [r12 + r13], 0x7d
    je object_close

object_member:
    cmp r13, r14
    jae object_fail
    cmp byte ptr [r12 + r13], 0x22
    jne object_fail
    call parse_string
    test eax, eax
    jz object_fail
    call skip_ws
    cmp r13, r14
    jae object_fail
    cmp byte ptr [r12 + r13], 0x3a
    jne object_fail
    inc r13
    call skip_ws
    call parse_value
    test eax, eax
    jz object_fail
    call skip_ws
    cmp r13, r14
    jae object_fail
    mov al, byte ptr [r12 + r13]
    cmp al, 0x7d
    je object_close
    cmp al, 0x2c
    jne object_fail
    inc r13
    call skip_ws
    jmp object_member

object_close:
    inc r13
    mov eax, 1
    ret

object_fail:
    xor eax, eax
    ret

.section .rodata
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"
ok_message:
    .ascii "ok\n"

.bss
.balign 16
input:
    .zero 65537
outbuf:
    .zero 64
