.globl _start

.text
_start:
    adrp x22, input_buf
    add x22, x22, :lo12:input_buf
    mov x19, #0

read_more:
    cmp x19, #65536
    b.hs input_ready
    mov x0, #0
    add x1, x22, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq input_ready
    add x19, x19, x0
    b read_more

input_ready:
    cbz x19, program_exit
    mov x20, #0
    mov x21, #0

scan_input:
    cmp x20, x19
    b.hs input_end
    ldrb w0, [x22, x20]
    cmp w0, #10
    b.eq complete_line
    add x20, x20, #1
    b scan_input

complete_line:
    add x0, x22, x21
    sub x1, x20, x21
    bl evaluate_line
    mov x2, x1
    mov x1, x0
    mov x0, #1
    mov x8, #64
    svc #0
    add x20, x20, #1
    mov x21, x20
    b scan_input

input_end:
    cmp x21, x19
    b.eq program_exit
    add x0, x22, x21
    sub x1, x19, x21
    bl evaluate_line
    mov x2, x1
    mov x1, x0
    mov x0, #1
    mov x8, #64
    svc #0

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

evaluate_line:
    mov x2, x0
    add x3, x0, x1
    adrp x4, value_stack
    add x4, x4, :lo12:value_stack
    mov x5, #0

skip_separators:
    cmp x2, x3
    b.hs line_finished
    ldrb w9, [x2]
    cmp w9, #32
    b.eq skip_one
    cmp w9, #9
    b.eq skip_one
    b token_start

skip_one:
    add x2, x2, #1
    b skip_separators

token_start:
    mov x6, x2

scan_token:
    cmp x2, x3
    b.hs token_ready
    ldrb w9, [x2]
    cmp w9, #32
    b.eq token_ready
    cmp w9, #9
    b.eq token_ready
    add x2, x2, #1
    b scan_token

token_ready:
    sub x8, x2, x6
    cmp x8, #1
    b.ne parse_literal
    ldrb w9, [x6]
    cmp w9, #43
    b.eq operator_token
    cmp w9, #45
    b.eq operator_token
    cmp w9, #42
    b.eq operator_token
    cmp w9, #47
    b.eq operator_token
    cmp w9, #37
    b.eq operator_token

parse_literal:
    mov x7, x6
    mov x10, #0
    mov x11, x8
    ldrb w15, [x6]
    cmp w15, #45
    b.ne check_digit_count
    mov x10, #1
    add x7, x7, #1
    sub x11, x11, #1

check_digit_count:
    cmp x11, #1
    b.lo line_error
    cmp x11, #18
    b.hi line_error
    mov x9, #0
    mov x16, #10

digit_loop:
    ldrb w15, [x7], #1
    sub w15, w15, #48
    cmp w15, #9
    b.hi line_error
    mul x9, x9, x16
    add x9, x9, x15
    subs x11, x11, #1
    b.ne digit_loop
    cbz x10, push_literal
    sub x9, xzr, x9

push_literal:
    str x9, [x4, x5, lsl #3]
    add x5, x5, #1
    b skip_separators

operator_token:
    cmp x5, #2
    b.lo line_error
    sub x5, x5, #1
    ldr x13, [x4, x5, lsl #3]
    sub x5, x5, #1
    ldr x12, [x4, x5, lsl #3]
    cmp w9, #43
    b.eq do_add
    cmp w9, #45
    b.eq do_subtract
    cmp w9, #42
    b.eq do_multiply
    b do_divide_or_remainder

do_add:
    add x14, x12, x13
    b push_result

do_subtract:
    sub x14, x12, x13
    b push_result

do_multiply:
    mul x14, x12, x13
    b push_result

do_divide_or_remainder:
    cbz x13, line_error
    movz x15, #0x8000, lsl #48
    cmp x12, x15
    b.ne division_safe
    cmn x13, #1
    b.eq line_error

division_safe:
    cmp w9, #47
    b.ne do_remainder
    sdiv x14, x12, x13
    b push_result

do_remainder:
    sdiv x15, x12, x13
    msub x14, x15, x13, x12

push_result:
    str x14, [x4, x5, lsl #3]
    add x5, x5, #1
    b skip_separators

line_finished:
    cmp x5, #1
    b.ne line_error
    ldr x0, [x4]
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x2, x1, #32
    mov x3, x2
    mov w4, #10
    strb w4, [x3, #-1]!
    mov x4, #0
    cmp x0, #0
    b.ge magnitude_ready
    mov x4, #1
    sub x0, xzr, x0

magnitude_ready:
    cbnz x0, conversion_setup
    mov w5, #48
    strb w5, [x3, #-1]!
    b add_sign

conversion_setup:
    mov x5, #10

conversion_loop:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    strb w7, [x3, #-1]!
    mov x0, x6
    cbnz x0, conversion_loop

add_sign:
    cbz x4, conversion_done
    mov w5, #45
    strb w5, [x3, #-1]!

conversion_done:
    mov x0, x3
    sub x1, x2, x3
    ret

line_error:
    adrp x0, error_message
    add x0, x0, :lo12:error_message
    mov x1, #6
    ret

.section .rodata
error_message:
    .ascii "error\n"

.bss
.align 3
input_buf:
    .zero 65536
value_stack:
    .zero 8192
output_buf:
    .zero 32
