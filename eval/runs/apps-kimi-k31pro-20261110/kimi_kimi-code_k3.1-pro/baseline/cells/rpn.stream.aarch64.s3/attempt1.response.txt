.section .rodata
error_msg:
    .ascii "error\n"
    .equ error_msg_len, . - error_msg

.text
.globl _start
_start:
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    mov x19, #0

read_more:
    mov x0, #0
    add x1, x20, x19
    movz x2, #1, lsl #16
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    movz x3, #1, lsl #16
    cmp x19, x3
    b.lo read_more

read_done:
    cbz x19, program_exit
    mov x21, x20
    add x22, x20, x19

next_line:
    cmp x21, x22
    b.hs program_exit
    mov x23, x21

find_line_end:
    cmp x23, x22
    b.hs have_line
    ldrb w0, [x23]
    cmp w0, #10
    b.eq have_line
    add x23, x23, #1
    b find_line_end

have_line:
    mov x0, x21
    mov x1, x23
    bl eval_line
    mov x21, x23
    cmp x21, x22
    b.hs next_line
    add x21, x21, #1
    b next_line

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0
    b program_exit

eval_line:
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, #0
    adrp x22, value_stack
    add x22, x22, :lo12:value_stack

eval_next:
    cmp x19, x20
    b.hs eval_finish
    ldrb w0, [x19]
    cmp w0, #32
    b.eq eval_skip_separator
    cmp w0, #9
    b.eq eval_skip_separator
    mov x23, x19

scan_token:
    cmp x19, x20
    b.hs token_ready
    ldrb w0, [x19]
    cmp w0, #32
    b.eq token_ready
    cmp w0, #9
    b.eq token_ready
    add x19, x19, #1
    b scan_token

eval_skip_separator:
    add x19, x19, #1
    b eval_next

token_ready:
    sub x24, x19, x23
    cmp x24, #1
    b.ne parse_integer
    ldrb w0, [x23]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_subtract
    cmp w0, #42
    b.eq op_multiply
    cmp w0, #47
    b.eq op_divide
    cmp w0, #37
    b.eq op_remainder

parse_integer:
    mov x2, x23
    mov x3, #0
    mov x4, x24
    ldrb w0, [x2]
    cmp w0, #45
    b.ne integer_digits_check
    mov x3, #1
    add x2, x2, #1
    sub x4, x4, #1

integer_digits_check:
    cbz x4, eval_error
    cmp x4, #18
    b.hi eval_error
    mov x5, #0
    mov x7, #10

integer_digit_loop:
    ldrb w0, [x2], #1
    sub w6, w0, #48
    cmp w6, #9
    b.hi eval_error
    mul x5, x5, x7
    add x5, x5, x6
    subs x4, x4, #1
    b.ne integer_digit_loop
    cbz x3, integer_push
    neg x5, x5

integer_push:
    cmp x21, #1024
    b.hs eval_error
    str x5, [x22, x21, lsl #3]
    add x21, x21, #1
    b eval_next

op_add:
    cmp x21, #2
    b.lo eval_error
    sub x21, x21, #1
    ldr x2, [x22, x21, lsl #3]
    sub x3, x21, #1
    ldr x4, [x22, x3, lsl #3]
    add x4, x4, x2
    str x4, [x22, x3, lsl #3]
    b eval_next

op_subtract:
    cmp x21, #2
    b.lo eval_error
    sub x21, x21, #1
    ldr x2, [x22, x21, lsl #3]
    sub x3, x21, #1
    ldr x4, [x22, x3, lsl #3]
    sub x4, x4, x2
    str x4, [x22, x3, lsl #3]
    b eval_next

op_multiply:
    cmp x21, #2
    b.lo eval_error
    sub x21, x21, #1
    ldr x2, [x22, x21, lsl #3]
    sub x3, x21, #1
    ldr x4, [x22, x3, lsl #3]
    mul x4, x4, x2
    str x4, [x22, x3, lsl #3]
    b eval_next

op_divide:
    cmp x21, #2
    b.lo eval_error
    sub x21, x21, #1
    ldr x2, [x22, x21, lsl #3]
    sub x3, x21, #1
    ldr x4, [x22, x3, lsl #3]
    cbz x2, eval_error
    cmn x2, #1
    b.ne divide_safe
    movz x6, #0x8000, lsl #48
    cmp x4, x6
    b.eq eval_error

divide_safe:
    sdiv x5, x4, x2
    str x5, [x22, x3, lsl #3]
    b eval_next

op_remainder:
    cmp x21, #2
    b.lo eval_error
    sub x21, x21, #1
    ldr x2, [x22, x21, lsl #3]
    sub x3, x21, #1
    ldr x4, [x22, x3, lsl #3]
    cbz x2, eval_error
    cmn x2, #1
    b.ne remainder_safe
    movz x6, #0x8000, lsl #48
    cmp x4, x6
    b.eq eval_error

remainder_safe:
    sdiv x5, x4, x2
    msub x5, x5, x2, x4
    str x5, [x22, x3, lsl #3]
    b eval_next

eval_finish:
    cmp x21, #1
    b.ne eval_error
    ldr x0, [x22]
    bl print_value
    b eval_return

eval_error:
    bl print_error

eval_return:
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
    ret

print_value:
    adrp x1, output_buffer
    add x1, x1, :lo12:output_buffer
    add x1, x1, #32
    mov w2, #10
    strb w2, [x1, #-1]!
    mov x5, #0
    cmp x0, #0
    b.ge print_digits
    mov x5, #1
    neg x0, x0

print_digits:
    mov x4, #10

print_digit_loop:
    udiv x2, x0, x4
    msub x3, x2, x4, x0
    add w3, w3, #48
    strb w3, [x1, #-1]!
    mov x0, x2
    cbnz x0, print_digit_loop
    cbz x5, print_write
    mov w2, #45
    strb w2, [x1, #-1]!

print_write:
    adrp x6, output_buffer
    add x6, x6, :lo12:output_buffer
    add x6, x6, #32
    sub x2, x6, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

print_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #error_msg_len
    mov x8, #64
    svc #0
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
.balign 8
value_stack:
    .zero 8192
.balign 16
output_buffer:
    .zero 32
