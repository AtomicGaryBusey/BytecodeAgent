.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_input:
    cmp r13, 65536
    jae input_ready
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz input_ready
    add r13, rax
    jmp read_input

input_ready:
    xor r14d, r14d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r14, r13
    jne error_exit

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_message]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp r14, r13
    jae skip_ws_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x20
    je skip_ws_advance
    cmp al, 0x09
    je skip_ws_advance
    cmp al, 0x0a
    je skip_ws_advance
    cmp al, 0x0d
    je skip_ws_advance
skip_ws_done:
    ret
skip_ws_advance:
    inc r14
    jmp skip_ws

parse_value:
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6e
    je parse_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb error_exit
    cmp al, 0x39
    jbe parse_number
    jmp error_exit

parse_true:
    lea rsi, [rip + literal_true]
    mov ecx, 4
    jmp parse_literal
parse_false:
    lea rsi, [rip + literal_false]
    mov ecx, 5
    jmp parse_literal
parse_null:
    lea rsi, [rip + literal_null]
    mov ecx, 4

parse_literal:
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, byte ptr [rsi]
    jne error_exit
    inc r14
    inc rsi
    dec ecx
    jne parse_literal
    ret

parse_string:
    inc r14
string_loop:
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x22
    je string_close
    cmp al, 0x5c
    je string_escape
    cmp al, 0x20
    jb error_exit
    inc r14
    jmp string_loop

string_close:
    inc r14
    ret

string_escape:
    inc r14
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x75
    je string_unicode
    cmp al, 0x22
    je string_simple_escape
    cmp al, 0x5c
    je string_simple_escape
    cmp al, 0x2f
    je string_simple_escape
    cmp al, 0x62
    je string_simple_escape
    cmp al, 0x66
    je string_simple_escape
    cmp al, 0x6e
    je string_simple_escape
    cmp al, 0x72
    je string_simple_escape
    cmp al, 0x74
    je string_simple_escape
    jmp error_exit

string_simple_escape:
    inc r14
    jmp string_loop

string_unicode:
    inc r14
    mov ecx, 4
unicode_digit_loop:
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb error_exit
    cmp al, 0x39
    jbe unicode_digit_ok
    cmp al, 0x41
    jb error_exit
    cmp al, 0x46
    jbe unicode_digit_ok
    cmp al, 0x61
    jb error_exit
    cmp al, 0x66
    ja error_exit
unicode_digit_ok:
    inc r14
    dec ecx
    jne unicode_digit_loop
    jmp string_loop

parse_number:
    cmp byte ptr [r12 + r14], 0x2d
    jne number_integer
    inc r14

number_integer:
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    je number_zero
    cmp al, 0x31
    jb error_exit
    cmp al, 0x39
    ja error_exit
    inc r14

number_integer_digits:
    cmp r14, r13
    jae number_after_integer
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_after_integer
    cmp al, 0x39
    ja number_after_integer
    inc r14
    jmp number_integer_digits

number_zero:
    inc r14

number_after_integer:
    cmp r14, r13
    jae number_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2e
    je number_fraction
    cmp al, 0x65
    je number_exponent
    cmp al, 0x45
    je number_exponent
    ret

number_fraction:
    inc r14
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb error_exit
    cmp al, 0x39
    ja error_exit
    inc r14

number_fraction_digits:
    cmp r14, r13
    jae number_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_maybe_exponent
    cmp al, 0x39
    ja number_maybe_exponent
    inc r14
    jmp number_fraction_digits

number_maybe_exponent:
    cmp al, 0x65
    je number_exponent
    cmp al, 0x45
    je number_exponent
    ret

number_exponent:
    inc r14
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x2b
    je number_exponent_sign
    cmp al, 0x2d
    jne number_exponent_require_digit
number_exponent_sign:
    inc r14

number_exponent_require_digit:
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb error_exit
    cmp al, 0x39
    ja error_exit
    inc r14

number_exponent_digits:
    cmp r14, r13
    jae number_done
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x30
    jb number_done
    cmp al, 0x39
    ja number_done
    inc r14
    jmp number_exponent_digits

number_done:
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae error_exit
    cmp byte ptr [r12 + r14], 0x5d
    je array_close

array_value:
    call parse_value
    call skip_ws
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x5d
    je array_close
    cmp al, 0x2c
    jne error_exit
    inc r14
    call skip_ws
    jmp array_value

array_close:
    inc r14
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae error_exit
    cmp byte ptr [r12 + r14], 0x7d
    je object_close

object_member:
    cmp r14, r13
    jae error_exit
    cmp byte ptr [r12 + r14], 0x22
    jne error_exit
    call parse_string
    call skip_ws
    cmp r14, r13
    jae error_exit
    cmp byte ptr [r12 + r14], 0x3a
    jne error_exit
    inc r14
    call skip_ws
    call parse_value
    call skip_ws
    cmp r14, r13
    jae error_exit
    movzx eax, byte ptr [r12 + r14]
    cmp al, 0x7d
    je object_close
    cmp al, 0x2c
    jne error_exit
    inc r14
    call skip_ws
    jmp object_member

object_close:
    inc r14
    ret

error_exit:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 9
    syscall

    mov rax, r14
    lea rsi, [rip + number_buffer + 32]
    dec rsi
    mov byte ptr [rsi], 0x0a
    mov ecx, 1
    mov ebx, 10

error_digit_loop:
    xor edx, edx
    div rbx
    dec rsi
    add dl, 0x30
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jne error_digit_loop

    mov eax, 1
    mov edi, 1
    mov edx, ecx
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
literal_true:
    .ascii "true"
literal_false:
    .ascii "false"
literal_null:
    .ascii "null"
ok_message:
    .ascii "ok\n"
error_message:
    .ascii "error at "

.bss
.balign 16
input_buffer:
    .zero 65536
number_buffer:
    .zero 32
