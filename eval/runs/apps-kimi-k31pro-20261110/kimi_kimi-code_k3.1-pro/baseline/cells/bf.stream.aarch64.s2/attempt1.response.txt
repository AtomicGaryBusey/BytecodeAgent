.text

    b.hs delimiter_done
    ldrb w1, [x19, x0]

    adrp x3, bracket_stack
    add x3, x3, :lo12:bracket_stack
    mov x0, #0

    b.hs validate_done
    ldrb w2, [x19, x0]
    cmp w2, #91
    b.eq validate_open
    cmp w2, #93

    ldr w4, [x3, x1, lsl #2]
    add w5, w0, #1
    str w5, [x24, x4, lsl #2]
    add w5, w4, #1
    str w5, [x24, x0, lsl #2]

    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf

    mov x27, #0
    mov x28, #0

    b.hs execution_done
    ldrb w0, [x19, x28]
    cmp w0, #62
    b.eq command_right
    cmp w0, #60
    b.eq command_left
    cmp w0, #43
    b.eq command_plus
    cmp w0, #45
    b.eq command_minus
    cmp w0, #46
    b.eq command_output
    cmp w0, #44
    b.eq command_input
    cmp w0, #91
    b.eq command_open
    cmp w0, #93
    b.eq command_close

execute_next:
    add x28, x28, #1
    b execute_loop

    b execute_next

    add w1, w1, #1

command_minus:

    b execute_next

    cbnz w1, execute_next
    ldr w1, [x24, x28, lsl #2]

    cbz w1, execute_next
    ldr w1, [x24, x28, lsl #2]
    mov x28, x1
