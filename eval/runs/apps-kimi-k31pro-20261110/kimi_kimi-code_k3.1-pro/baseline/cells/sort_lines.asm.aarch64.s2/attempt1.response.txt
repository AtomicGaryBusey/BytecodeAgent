.globl _start
.text
_start:
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    mov x19, #0

read_loop:
    movz x2, #32768
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, exit_program

    adrp x21, line_records
    add x21, x21, :lo12:line_records
    mov x23, #0
    mov x3, #0
    mov x4, #0

scan_loop:
    cmp x3, x19
    b.hs scan_done
    ldrb w5, [x20, x3]
    cmp w5, #10
    b.eq found_newline
    add x3, x3, #1
    b scan_loop

found_newline:
    add x6, x20, x4
    sub x7, x3, x4
    lsl x8, x23, #4
    add x8, x21, x8
    stp x6, x7, [x8]
    add x23, x23, #1
    add x3, x3, #1
    mov x4, x3
    b scan_loop

scan_done:
    cmp x4, x19
    b.eq records_done
    add x6, x20, x4
    sub x7, x19, x4
    lsl x8, x23, #4
    add x8, x21, x8
    stp x6, x7, [x8]
    add x23, x23, #1

records_done:
    cmp x23, #2
    b.lo output_setup

    adrp x22, temporary_records
    add x22, x22, :lo12:temporary_records
    mov x24, #1

sort_pass:
    cmp x24, x23
    b.hs output_setup
    mov x25, #0

run_start:
    add x0, x25, x24
    cmp x0, x23
    csel x28, x0, x23, lo
    add x0, x28, x24
    cmp x0, x23
    csel x29, x0, x23, lo

    mov x26, x25
    mov x27, x28
    mov x10, x25

merge_loop:
    cmp x26, x28
    b.hs copy_right_remaining
    cmp x27, x29
    b.hs copy_left_remaining

    lsl x0, x26, #4
    add x0, x21, x0
    ldp x3, x4, [x0]
    lsl x0, x27, #4
    add x0, x21, x0
    ldp x5, x6, [x0]

    cmp x4, x6
    csel x8, x4, x6, lo
    mov x7, #0

compare_bytes:
    cmp x7, x8
    b.hs compare_lengths
    ldrb w0, [x3, x7]
    ldrb w1, [x5, x7]
    cmp w0, w1
    b.lo take_left
    b.hi take_right
    add x7, x7, #1
    b compare_bytes

compare_lengths:
    cmp x4, x6
    b.ls take_left
    b take_right

take_left:
    lsl x0, x26, #4
    add x0, x21, x0
    ldp x3, x4, [x0]
    lsl x1, x10, #4
    add x1, x22, x1
    stp x3, x4, [x1]
    add x26, x26, #1
    add x10, x10, #1
    b merge_loop

take_right:
    lsl x0, x27, #4
    add x0, x21, x0
    ldp x3, x4, [x0]
    lsl x1, x10, #4
    add x1, x22, x1
    stp x3, x4, [x1]
    add x27, x27, #1
    add x10, x10, #1
    b merge_loop

copy_left_remaining:
    cmp x26, x28
    b.hs copy_right_remaining
    lsl x0, x26, #4
    add x0, x21, x0
    ldp x3, x4, [x0]
    lsl x1, x10, #4
    add x1, x22, x1
    stp x3, x4, [x1]
    add x26, x26, #1
    add x10, x10, #1
    b copy_left_remaining

copy_right_remaining:
    cmp x27, x29
    b.hs run_done
    lsl x0, x27, #4
    add x0, x21, x0
    ldp x3, x4, [x0]
    lsl x1, x10, #4
    add x1, x22, x1
    stp x3, x4, [x1]
    add x27, x27, #1
    add x10, x10, #1
    b copy_right_remaining

run_done:
    mov x25, x29
    cmp x25, x23
    b.lo run_start

    mov x0, x21
    mov x21, x22
    mov x22, x0
    lsl x24, x24, #1
    b sort_pass

output_setup:
    adrp x19, newline_byte
    add x19, x19, :lo12:newline_byte
    mov x24, #0

output_loop:
    cmp x24, x23
    b.hs exit_program
    lsl x0, x24, #4
    add x0, x21, x0
    ldp x1, x2, [x0]
    cbz x2, output_newline
    mov x0, #1
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    mov x1, x19
    mov x2, #1
    mov x8, #64
    svc #0
    add x24, x24, #1
    b output_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_program

.section .rodata
newline_byte:
    .byte 10

.bss
    .p2align 4
input_buffer:
    .zero 32768
    .p2align 4
line_records:
    .zero 16000
    .p2align 4
temporary_records:
    .zero 16000
