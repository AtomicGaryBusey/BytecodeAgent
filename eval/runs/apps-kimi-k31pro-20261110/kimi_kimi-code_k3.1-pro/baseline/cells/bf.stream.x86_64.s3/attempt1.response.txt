.globl _start

.text
_start:
    cld
    lea r10, [rip + input_buf]
    xor ebx, ebx

read_loop:
    cmp rbx, 65537
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r10 + rbx]
    mov rdx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    xor ecx, ecx

find_separator:
    cmp rcx, rbx
    jae no_separator
    cmp byte ptr [r10 + rcx], 33
    je separator_found
    inc rcx
    jmp find_separator

separator_found:
    mov r12, rcx
    lea r13, [r10 + rcx + 1]
    lea r14, [r10 + rbx]
    jmp validate_start

no_separator:
    mov r12, rbx
    lea r13, [r10 + rbx]
    mov r14, r13

validate_start:
    xor ebp, ebp
    xor ecx, ecx
    lea r8, [rip + bracket_stack]
    lea r9, [rip + jump_table]

validate_loop:
    cmp rbp, r12
    jae validate_end
    mov al, byte ptr [r10 + rbp]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close

validate_next:
    inc rbp
    jmp validate_loop

validate_open:
    cmp ecx, 1000
    jae invalid_program
    mov dword ptr [r8 + rcx*4], ebp
    inc ecx
    inc rbp
    jmp validate_loop

validate_close:
    test ecx, ecx
    jz invalid_program
    dec ecx
    mov eax, dword ptr [r8 + rcx*4]
    mov dword ptr [r9 + rbp*4], eax
    mov dword ptr [r9 + rax*4], ebp
    inc rbp
    jmp validate_loop

validate_end:
    test ecx, ecx
    jnz invalid_program

    xor ebp, ebp
    xor r15d, r15d
    lea rbx, [rip + tape]
    lea r8, [rip + output_buf]

execute_loop:
    cmp rbp, r12
    jae program_done
    movzx eax, byte ptr [r10 + rbp]
    inc rbp

    cmp al, 62
    je pointer_right
    cmp al, 60
    je pointer_left
    cmp al, 43
    je cell_increment
    cmp al, 45
    je cell_decrement
    cmp al, 46
    je cell_output
    cmp al, 44
    je cell_input
    cmp al, 91
    je bracket_open
    cmp al, 93
    je bracket_close
    jmp execute_loop

pointer_right:
    inc rbx
    jmp execute_loop

pointer_left:
    dec rbx
    jmp execute_loop

cell_increment:
    inc byte ptr [rbx]
    jmp execute_loop

cell_decrement:
    dec byte ptr [rbx]
    jmp execute_loop

cell_output:
    mov al, byte ptr [rbx]
    mov byte ptr [r8 + r15], al
    inc r15
    cmp r15, 65536
    jne execute_loop

    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r15
    syscall
    xor r15d, r15d
    jmp execute_loop

cell_input:
    xor eax, eax
    cmp r13, r14
    jae store_input
    mov al, byte ptr [r13]
    inc r13

store_input:
    mov byte ptr [rbx], al
    jmp execute_loop

bracket_open:
    cmp byte ptr [rbx], 0
    jne execute_loop
    mov ebp, dword ptr [r9 + rbp*4 - 4]
    inc rbp
    jmp execute_loop

bracket_close:
    cmp byte ptr [rbx], 0
    je execute_loop
    mov ebp, dword ptr [r9 + rbp*4 - 4]
    inc rbp
    jmp execute_loop

program_done:
    test r15, r15
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r15
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
input_buf:
    .zero 65537
.align 16
output_buf:
    .zero 65536
.align 16
tape:
    .zero 30000
.align 16
jump_table:
    .zero 262144
.align 16
bracket_stack:
    .zero 4000
