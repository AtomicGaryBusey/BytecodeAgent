.section .rodata
error_msg:
.ascii "error\n"
.equ error_msg_len, 6

.text
.globl _start
_start:
    mov x19, #0
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    movz x21, #1, lsl #16

read_loop:
    sub x2, x21, x19
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, program_done
    mov x22, #0

line_loop:
    cmp x22, x19
    b.hs program_done
    mov x24, x22

find_line_end:
    cmp x24, x19
    b.hs line_end_found
    ldrb w0, [x20, x24]
    cmp w0, #10
    b.eq line_end_found
    add x24, x24, #1
    b find_line_end

line_end_found:
    add x0, x20, x22
    sub x1, x24, x22
    bl eval_line

    cmp x24, x19
    b.hs no_line_newline
    add x22, x24, #1
    b line_position_ready
no_line_newline:
    mov x22, x24

line_position_ready:
    cbnz x0, print_line_error
    mov x0, x1
    bl print_value
    b line_loop

print_line_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #error_msg_len
    mov x8, #64
    svc #0
    b line_loop

program_done:
    mov x0, #0
    mov x8, #93
    svc #0
    b program_done

eval_line:
    add x3, x0, x1
    mov x2, x0
    adrp x4, value_stack
    add x4, x4, :lo12:value_stack
    mov x5, #0

next_token:
    cmp x2, x3
    b.hs tokens_done
    ldrb w6, [x2]
    cmp w6, #32
    b.eq skip_separator
    cmp w6, #9
    b.eq skip_separator
    b token_begin

skip_separator:
    add x2, x2, #1
    b next_token

token_begin:
    mov x7, x2

scan_token:
    cmp x2, x3
    b.hs token_scanned
    ldrb w6, [x2]
    cmp w6, #32
    b.eq token_scanned
    cmp w6, #9
    b.eq token_scanned
    add x2, x2, #1
    b scan_token

token_scanned:
    sub x8, x2, x7
    cmp x8, #1
    b.eq single_byte_token

    mov x9, x7
    mov x10, #0
    ldrb w6, [x7]
    cmp w6, #45
    b.ne positive_literal_length
    mov x10, #1
    add x9, x9, #1
    sub x11, x8, #1
    b check_literal_length

positive_literal_length:
    mov x11, x8

check_literal_length:
    cmp x11, #18
    b.hi eval_error

accumulate_literal:
    mov x12, #0

literal_digit_loop:
    ldrb w6, [x9], #1
    sub w13, w6, #48
    cmp w13, #9
    b.hi eval_error
    mov x14, #10
    mul x12, x12, x14
    add x12, x12, x13
    sub x11, x11, #1
    cbnz x11, literal_digit_loop

    cbz x10, push_literal
    neg x12, x12

push_literal:
    str x12, [x4, x5, lsl #3]
    add x5, x5, #1
    b next_token

single_byte_token:
    ldrb w6, [x7]
    sub w13, w6, #48
    cmp w13, #9
    b.ls single_digit

    cmp w6, #43
    b.eq operator_add
    cmp w6, #45
    b.eq operator_subtract
    cmp w6, #42
    b.eq operator_multiply
    cmp w6, #47
    b.eq operator_divide
    cmp w6, #37
    b.eq operator_remainder
    b eval_error

single_digit:
    mov x9, x7
    mov x10, #0
    mov x11, #1
    b accumulate_literal

operator_add:
    cmp x5, #2
    b.lo eval_error
    sub x14, x5, #1
    ldr x13, [x4, x14, lsl #3]
    sub x14, x5, #2
    ldr x12, [x4, x14, lsl #3]
    add x12, x12, x13
    b store_operator_result

operator_subtract:
    cmp x5, #2
    b.lo eval_error
    sub x14, x5, #1
    ldr x13, [x4, x14, lsl #3]
    sub x14, x5, #2
    ldr x12, [x4, x14, lsl #3]
    sub x12, x12, x13
    b store_operator_result

operator_multiply:
    cmp x5, #2
    b.lo eval_error
    sub x14, x5, #1
    ldr x13, [x4, x14, lsl #3]
    sub x14, x5, #2
    ldr x12, [x4, x14, lsl #3]
    mul x12, x12, x13
    b store_operator_result

operator_divide:
    cmp x5, #2
    b.lo eval_error
    sub x14, x5, #1
    ldr x13, [x4, x14, lsl #3]
    sub x14, x5, #2
    ldr x12, [x4, x14, lsl #3]
    cbz x13, eval_error
    movz x15, #0x8000, lsl #48
    cmp x12, x15
    b.ne divide_ready
    cmn x13, #1
    b.eq eval_error
divide_ready:
    sdiv x12, x12, x13
    b store_operator_result

operator_remainder:
    cmp x5, #2
    b.lo eval_error
    sub x14, x5, #1
    ldr x13, [x4, x14, lsl #3]
    sub x14, x5, #2
    ldr x12, [x4, x14, lsl #3]
    cbz x13, eval_error
    movz x15, #0x8000, lsl #48
    cmp x12, x15
    b.ne remainder_ready
    cmn x13, #1
    b.eq eval_error
remainder_ready:
    sdiv x15, x12, x13
    msub x12, x15, x13, x12

store_operator_result:
    str x12, [x4, x14, lsl #3]
    sub x5, x5, #1
    b next_token

tokens_done:
    cmp x5, #1
    b.ne eval_error
    ldr x1, [x4]
    mov x0, #0
    ret

eval_error:
    mov x0, #1
    ret

print_value:
    adrp x1, output_buffer
    add x1, x1, :lo12:output_buffer
    add x2, x1, #32
    mov x15, x2
    sub x2, x2, #1
    mov w4, #10
    strb w4, [x2]
    mov x3, #10
    mov x5, #0
    tbnz x0, #63, print_negative
    b print_convert

print_negative:
    mov x5, #1
    neg x0, x0

print_convert:
    udiv x6, x0, x3
    msub x7, x6, x3, x0
    add w7, w7, #48
    sub x2, x2, #1
    strb w7, [x2]
    mov x0, x6
    cbnz x0, print_convert
    cbz x5, print_emit
    sub x2, x2, #1
    mov w4, #45
    strb w4, [x2]

print_emit:
    sub x3, x15, x2
    mov x1, x2
    mov x2, x3
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.bss
.p2align 3
input_buffer:
    .zero 65537
.p2align 3
value_stack:
    .zero 4096
output_buffer:
    .zero 32
