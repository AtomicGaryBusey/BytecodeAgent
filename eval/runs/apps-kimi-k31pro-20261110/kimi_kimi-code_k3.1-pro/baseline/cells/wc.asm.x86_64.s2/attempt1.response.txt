.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_chunk:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall

    test rax, rax
    jle input_done

    add r14, rax
    lea rbx, [rip + input_buffer]
    mov rbp, rax

byte_loop:
    movzx eax, byte ptr [rbx]
    inc rbx

    cmp al, 10
    jne check_whitespace
    inc r12

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15b, r15b
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    dec rbp
    jnz byte_loop
    jmp read_chunk

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    lea r8, [rip + digit_buffer + 32]
    mov r9, r8
    mov r10d, 10

convert_digit:
    xor edx, edx
    div r10
    dec r8
    add dl, 48
    mov byte ptr [r8], dl
    test rax, rax
    jnz convert_digit

    mov rcx, r9
    sub rcx, r8
    mov rsi, r8
    rep movsb
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 64
digit_buffer:
    .zero 32
