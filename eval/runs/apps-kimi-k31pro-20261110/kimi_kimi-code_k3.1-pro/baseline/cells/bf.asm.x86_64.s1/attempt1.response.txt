.globl _start

.text
_start:
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buf]
    lea r15, [rsi + r12]
    xor ecx, ecx

find_separator:
    cmp rcx, r12
    jae no_separator
    cmp byte ptr [rsi + rcx], 0x21
    je separator_found
    inc rcx
    jmp find_separator

no_separator:
    mov r14, r15
    mov r8, r15
    jmp filter_setup

separator_found:
    lea r14, [rsi + rcx + 1]
    lea r8, [rsi + rcx]

filter_setup:
    lea rdi, [rip + code_buf]
    lea r9, [rip + bracket_map]
    lea r10, [rip + bracket_stack]
    xor edx, edx
    xor r11d, r11d

filter_loop:
    cmp rsi, r8
    jae filter_done
    movzx eax, byte ptr [rsi]
    cmp al, 62
    je command_gt
    cmp al, 60
    je command_lt
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close
    inc rsi
    jmp filter_loop

command_gt:
    mov eax, 1
    jmp store_command

command_lt:
    mov eax, 2
    jmp store_command

command_plus:
    mov eax, 3
    jmp store_command

command_minus:
    mov eax, 4
    jmp store_command

command_output:
    mov eax, 5
    jmp store_command

command_input:
    mov eax, 6

store_command:
    mov byte ptr [rdi + rdx], al
    inc rdx
    inc rsi
    jmp filter_loop

command_open:
    mov byte ptr [rdi + rdx], 7
    mov dword ptr [r10 + r11*4], edx
    inc r11
    inc rdx
    inc rsi
    jmp filter_loop

command_close:
    test r11, r11
    jz invalid_program
    mov byte ptr [rdi + rdx], 8
    dec r11
    mov eax, dword ptr [r10 + r11*4]
    mov dword ptr [r9 + rax*4], edx
    mov dword ptr [r9 + rdx*4], eax
    inc rdx
    inc rsi
    jmp filter_loop

filter_done:
    test r11, r11
    jnz invalid_program

    mov rbp, rdx
    lea rbx, [rip + code_buf]
    lea r12, [rip + bracket_map]
    lea r13, [rip + tape]
    lea rdi, [rip + output_buf]
    xor esi, esi

interpret_loop:
    cmp rsi, rbp
    jae program_done
    movzx eax, byte ptr [rbx + rsi]
    lea rdx, [rip + dispatch]
    jmp qword ptr [rdx + rax*8]

op_gt:
    inc r13
    inc rsi
    jmp interpret_loop

op_lt:
    dec r13
    inc rsi
    jmp interpret_loop

op_plus:
    inc byte ptr [r13]
    inc rsi
    jmp interpret_loop

op_minus:
    dec byte ptr [r13]
    inc rsi
    jmp interpret_loop

op_output:
    mov al, byte ptr [r13]
    mov byte ptr [rdi], al
    inc rdi
    inc rsi
    jmp interpret_loop

op_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [r13], al
    inc rsi
    jmp interpret_loop

input_exhausted:
    mov byte ptr [r13], 0
    inc rsi
    jmp interpret_loop

op_open:
    cmp byte ptr [r13], 0
    jne op_open_continue
    mov eax, dword ptr [r12 + rsi*4]
    lea esi, [rax + 1]
    jmp interpret_loop

op_open_continue:
    inc rsi
    jmp interpret_loop

op_close:
    cmp byte ptr [r13], 0
    je op_close_continue
    mov eax, dword ptr [r12 + rsi*4]
    lea esi, [rax + 1]
    jmp interpret_loop

op_close_continue:
    inc rsi
    jmp interpret_loop

program_done:
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_success
    mov edi, 1

write_loop:
    mov eax, 1
    syscall
    test rax, rax
    jle exit_success
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
.p2align 3
dispatch:
    .quad interpret_loop
    .quad op_gt
    .quad op_lt
    .quad op_plus
    .quad op_minus
    .quad op_output
    .quad op_input
    .quad op_open
    .quad op_close

.bss
.p2align 4
input_buf:
    .space 65536
code_buf:
    .space 65536
bracket_map:
    .space 262144
bracket_stack:
    .space 262144
tape:
    .space 30000
output_buf:
    .space 65536
