.globl _start

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66560

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jle input_complete
    add r12, rax
    cmp r12, 49152
    jb read_more

input_complete:
    test r12, r12
    jz exit_success

    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea rbp, [rip + base64_alphabet]
    mov rbx, r12
    xor r9d, r9d

encode_loop:
    cmp rbx, 3
    jb final_group

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 2], dl

    and eax, 63
    mov al, byte ptr [rbp + rax]
    mov byte ptr [rdi + 3], al

    add rsi, 3
    add rdi, 4
    sub rbx, 3
    add r9d, 4
    cmp r9d, 76
    jne encode_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encode_loop

final_group:
    test rbx, rbx
    jnz partial_group
    test r9d, r9d
    jz write_output
    jmp append_newline

partial_group:
    cmp rbx, 1
    je one_byte_left

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi + 1], dl

    shr eax, 6
    and eax, 63
    mov al, byte ptr [rbp + rax]
    mov byte ptr [rdi + 2], al
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    jmp append_newline

one_byte_left:
    movzx eax, byte ptr [rsi]
    shl eax, 16

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl

    shr eax, 12
    and eax, 63
    mov al, byte ptr [rbp + rax]
    mov byte ptr [rdi + 1], al
    mov word ptr [rdi + 2], 0x3d3d
    add rdi, 4

append_newline:
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall
