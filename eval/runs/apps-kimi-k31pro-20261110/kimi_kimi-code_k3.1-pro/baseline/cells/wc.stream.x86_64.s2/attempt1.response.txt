.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_input:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 4096
    syscall

    test rax, rax
    jle input_done

    add r14, rax
    lea rsi, [rip + input_buffer]
    mov rcx, rax

scan_byte:
    movzx edx, byte ptr [rsi]
    inc rsi

    cmp dl, 32
    je whitespace
    cmp dl, 9
    je whitespace
    cmp dl, 10
    je newline
    cmp dl, 11
    je whitespace
    cmp dl, 12
    je whitespace
    cmp dl, 13
    je whitespace

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

newline:
    inc r12

whitespace:
    xor r15d, r15d

next_byte:
    dec rcx
    jnz scan_byte
    jmp read_input

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_number
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_number
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_number:
    lea r8, [rip + number_buffer + 32]
    mov r9, r8
    mov r10d, 10

    test rax, rax
    jnz convert_digit
    dec r8
    mov byte ptr [r8], 48
    jmp copy_digits

convert_digit:
    xor edx, edx
    div r10
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz convert_digit

copy_digits:
    cmp r8, r9
    je number_done
    mov dl, byte ptr [r8]
    mov byte ptr [rdi], dl
    inc r8
    inc rdi
    jmp copy_digits

number_done:
    ret

.bss
.balign 16
input_buffer:
    .zero 4096
output_buffer:
    .zero 64
number_buffer:
    .zero 32
