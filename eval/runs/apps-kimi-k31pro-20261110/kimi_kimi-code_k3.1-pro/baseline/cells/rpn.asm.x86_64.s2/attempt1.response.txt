.globl _start

.text
_start:
    xor r13d, r13d

read_input:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r13
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz input_complete
    add r13, rax
    cmp r13, 65537
    jb read_input

input_complete:
    lea r14, [rip + input_buffer]
    lea r15, [rip + input_buffer]
    add r15, r13
    lea rbp, [rip + output_buffer]
    lea rdi, [rip + value_stack]

line_loop:
    cmp r14, r15
    jae write_output

    mov r12, r14
find_line_end:
    cmp r12, r15
    jae line_at_input_end
    cmp byte ptr [r12], 10
    je line_at_newline
    inc r12
    jmp find_line_end

line_at_input_end:
    mov r13, r12
    jmp start_line

line_at_newline:
    lea r13, [r12 + 1]

start_line:
    xor ebx, ebx

parse_line:
    cmp r14, r12
    jae finish_line
    mov al, byte ptr [r14]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov rsi, r14
scan_token:
    cmp r14, r12
    jae token_ready
    mov al, byte ptr [r14]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc r14
    jmp scan_token

skip_separator:
    inc r14
    jmp parse_line

token_ready:
    mov rdx, r14
    mov rcx, rdx
    sub rcx, rsi
    cmp rcx, 1
    jne parse_literal

    mov al, byte ptr [rsi]
    cmp al, 43
    je parse_operator
    cmp al, 45
    je parse_operator
    cmp al, 42
    je parse_operator
    cmp al, 47
    je parse_operator
    cmp al, 37
    je parse_operator

parse_literal:
    mov r9, rsi
    xor r10d, r10d
    cmp byte ptr [r9], 45
    jne literal_digits_start
    mov r10d, 1
    inc r9
    cmp r9, rdx
    je line_error

literal_digits_start:
    xor r8d, r8d
    xor ecx, ecx

literal_digit_loop:
    cmp r9, rdx
    jae literal_complete
    movzx eax, byte ptr [r9]
    sub eax, 48
    cmp eax, 9
    ja line_error
    inc ecx
    cmp ecx, 18
    ja line_error
    imul r8, r8, 10
    add r8, rax
    inc r9
    jmp literal_digit_loop

literal_complete:
    test ecx, ecx
    jz line_error
    test r10d, r10d
    jz push_literal
    neg r8

push_literal:
    mov qword ptr [rdi + rbx*8], r8
    inc rbx
    jmp parse_line

parse_operator:
    cmp ebx, 2
    jb line_error
    mov r8, qword ptr [rdi + rbx*8 - 16]
    mov r9, qword ptr [rdi + rbx*8 - 8]

    cmp al, 43
    je operator_add
    cmp al, 45
    je operator_subtract
    cmp al, 42
    je operator_multiply
    cmp al, 47
    je operator_divide
    jmp operator_remainder

operator_add:
    add r8, r9
    jmp store_operator_result

operator_subtract:
    sub r8, r9
    jmp store_operator_result

operator_multiply:
    imul r8, r9
    jmp store_operator_result

operator_divide:
    test r9, r9
    jz line_error
    mov r10, 0x8000000000000000
    cmp r8, r10
    jne divide_safe
    cmp r9, -1
    je line_error
divide_safe:
    mov rax, r8
    cqo
    idiv r9
    mov r8, rax
    jmp store_operator_result

operator_remainder:
    test r9, r9
    jz line_error
    mov r10, 0x8000000000000000
    cmp r8, r10
    jne remainder_safe
    cmp r9, -1
    je line_error
remainder_safe:
    mov rax, r8
    cqo
    idiv r9
    mov r8, rdx

store_operator_result:
    mov qword ptr [rdi + rbx*8 - 16], r8
    dec rbx
    jmp parse_line

finish_line:
    cmp ebx, 1
    jne line_error
    mov rax, qword ptr [rdi]
    test rax, rax
    jns convert_number
    mov byte ptr [rbp], 45
    inc rbp
    neg rax

convert_number:
    lea rsi, [rip + number_buffer + 32]
    mov r8d, 10

convert_digit:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz convert_digit

    lea rdx, [rip + number_buffer + 32]
copy_number:
    mov al, byte ptr [rsi]
    mov byte ptr [rbp], al
    inc rsi
    inc rbp
    cmp rsi, rdx
    jne copy_number
    mov byte ptr [rbp], 10
    inc rbp
    jmp next_line

line_error:
    mov dword ptr [rbp], 0x6f727265
    mov word ptr [rbp + 4], 0x0a72
    add rbp, 6

next_line:
    mov r14, r13
    jmp line_loop

write_output:
    mov rdx, rbp
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.p2align 3
input_buffer:
    .zero 65537
output_buffer:
    .zero 393216
value_stack:
    .zero 8192
number_buffer:
    .zero 32
