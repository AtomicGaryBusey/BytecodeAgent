.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0

read_input:
    mov x0, #0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    mov x2, #65536
    mov x8, #63
    svc #0

    cmp x0, #0
    b.le input_done

    mov x6, x0
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer

byte_loop:
    ldrb w3, [x1], #1
    add x21, x21, #1

    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w3, #32
    b.eq whitespace
    cmp w3, #9
    b.lo non_whitespace
    cmp w3, #13
    b.ls whitespace

non_whitespace:
    cbnz w22, next_byte
    add x20, x20, #1
    mov w22, #1
    b next_byte

whitespace:
    mov w22, #0

next_byte:
    subs x6, x6, #1
    b.ne byte_loop
    b read_input

input_done:
    adrp x7, output_buffer
    add x7, x7, :lo12:output_buffer
    mov x1, x7

    mov x0, x19
    bl append_decimal
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl append_decimal
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl append_decimal
    mov w2, #10
    strb w2, [x1], #1

    sub x2, x1, x7
    mov x0, #1
    mov x1, x7
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_decimal:
    mov x2, #10
    mov x5, x1
1:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add w4, w4, #48
    strb w4, [x1], #1
    mov x0, x3
    cbnz x0, 1b

    sub x6, x1, #1
2:
    cmp x5, x6
    b.hs 3f
    ldrb w3, [x5]
    ldrb w4, [x6]
    strb w4, [x5], #1
    strb w3, [x6], #-1
    b 2b
3:
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 128
