.globl _start
.text
_start:
    lea r12, [rip + input_buf]
    xor r13d, r13d

.read_loop:
    cmp r13, 65536
    jae .read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle .read_done
    add r13, rax
    jmp .read_loop

.read_done:
    xor r14d, r14d
    call skip_ws
    call parse_value
    test eax, eax
    jz .error
    call skip_ws
    cmp r14, r13
    jne .error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_msg_len
    syscall
    xor edi, edi
    mov eax, 60
    syscall

.error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, err_prefix_len
    syscall

    mov rax, r14
    lea rsi, [rip + num_buf + 64]
    dec rsi
    mov byte ptr [rsi], 10
    mov ebx, 10

.convert_offset:
    xor edx, edx
    div rbx
    dec rsi
    add dl, 48
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .convert_offset

    mov eax, 1
    mov edi, 1
    lea rdx, [rip + num_buf + 64]
    sub rdx, rsi
    syscall
    mov edi, 1
    mov eax, 60
    syscall

skip_ws:
    cmp r14, r13
    jae .sw_done
    mov al, byte ptr [r12 + r14]
    cmp al, 0x20
    je .sw_advance
    cmp al, 0x09
    je .sw_advance
    cmp al, 0x0a
    je .sw_advance
    cmp al, 0x0d
    jne .sw_done
.sw_advance:
    inc r14
    jmp skip_ws
.sw_done:
    ret

parse_value:
    cmp r14, r13
    jae .pv_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x7b
    je parse_object
    cmp al, 0x5b
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je .pv_true
    cmp al, 0x66
    je .pv_false
    cmp al, 0x6e
    je .pv_null
    cmp al, 0x2d
    je parse_number
    cmp al, 0x30
    jb .pv_fail
    cmp al, 0x39
    ja .pv_fail
    jmp parse_number
.pv_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp parse_literal
.pv_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp parse_literal
.pv_null:
    lea rsi, [rip + lit_null]
    mov edx, 4
    jmp parse_literal
.pv_fail:
    xor eax, eax
    ret

parse_literal:
    xor ecx, ecx
.pl_loop:
    cmp ecx, edx
    jae .pl_success
    cmp r14, r13
    jae .pl_fail
    mov al, byte ptr [r12 + r14]
    cmp al, byte ptr [rsi + rcx]
    jne .pl_fail
    inc r14
    inc ecx
    jmp .pl_loop
.pl_success:
    mov eax, 1
    ret
.pl_fail:
    xor eax, eax
    ret

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae .po_fail
    cmp byte ptr [r12 + r14], 0x7d
    jne .po_member
    inc r14
    mov eax, 1
    ret

.po_member:
    call parse_string
    test eax, eax
    jz .po_fail
    call skip_ws
    cmp r14, r13
    jae .po_fail
    cmp byte ptr [r12 + r14], 0x3a
    jne .po_fail
    inc r14
    call skip_ws
    call parse_value
    test eax, eax
    jz .po_fail
    call skip_ws
    cmp r14, r13
    jae .po_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x7d
    je .po_close
    cmp al, 0x2c
    jne .po_fail
    inc r14
    call skip_ws
    jmp .po_member

.po_close:
    inc r14
    mov eax, 1
    ret
.po_fail:
    xor eax, eax
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae .pa_fail
    cmp byte ptr [r12 + r14], 0x5d
    jne .pa_element
    inc r14
    mov eax, 1
    ret

.pa_element:
    call parse_value
    test eax, eax
    jz .pa_fail
    call skip_ws
    cmp r14, r13
    jae .pa_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x5d
    je .pa_close
    cmp al, 0x2c
    jne .pa_fail
    inc r14
    call skip_ws
    jmp .pa_element

.pa_close:
    inc r14
    mov eax, 1
    ret
.pa_fail:
    xor eax, eax
    ret

parse_string:
    cmp r14, r13
    jae .ps_fail
    cmp byte ptr [r12 + r14], 0x22
    jne .ps_fail
    inc r14

.ps_loop:
    cmp r14, r13
    jae .ps_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x22
    je .ps_close
    cmp al, 0x20
    jb .ps_fail
    cmp al, 0x5c
    je .ps_escape
    inc r14
    jmp .ps_loop

.ps_escape:
    inc r14
    cmp r14, r13
    jae .ps_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x75
    je .ps_unicode
    cmp al, 0x22
    je .ps_simple_escape
    cmp al, 0x5c
    je .ps_simple_escape
    cmp al, 0x2f
    je .ps_simple_escape
    cmp al, 0x62
    je .ps_simple_escape
    cmp al, 0x66
    je .ps_simple_escape
    cmp al, 0x6e
    je .ps_simple_escape
    cmp al, 0x72
    je .ps_simple_escape
    cmp al, 0x74
    jne .ps_fail

.ps_simple_escape:
    inc r14
    jmp .ps_loop

.ps_unicode:
    inc r14
    mov ecx, 4
.ps_hex_loop:
    cmp r14, r13
    jae .ps_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb .ps_fail
    cmp al, 0x39
    jbe .ps_hex_valid
    mov dl, al
    or dl, 0x20
    cmp dl, 0x61
    jb .ps_fail
    cmp dl, 0x66
    ja .ps_fail
.ps_hex_valid:
    inc r14
    dec ecx
    jnz .ps_hex_loop
    jmp .ps_loop

.ps_close:
    inc r14
    mov eax, 1
    ret
.ps_fail:
    xor eax, eax
    ret

parse_number:
    cmp byte ptr [r12 + r14], 0x2d
    jne .pn_int
    inc r14

.pn_int:
    cmp r14, r13
    jae .pn_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    je .pn_zero
    cmp al, 0x31
    jb .pn_fail
    cmp al, 0x39
    ja .pn_fail
    inc r14

.pn_int_digits:
    cmp r14, r13
    jae .pn_after_int
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb .pn_after_int
    cmp al, 0x39
    ja .pn_after_int
    inc r14
    jmp .pn_int_digits

.pn_zero:
    inc r14

.pn_after_int:
    cmp r14, r13
    jae .pn_success
    cmp byte ptr [r12 + r14], 0x2e
    je .pn_fraction
    jmp .pn_exp_check

.pn_fraction:
    inc r14
    cmp r14, r13
    jae .pn_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb .pn_fail
    cmp al, 0x39
    ja .pn_fail
    inc r14

.pn_frac_digits:
    cmp r14, r13
    jae .pn_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb .pn_exp_check
    cmp al, 0x39
    ja .pn_exp_check
    inc r14
    jmp .pn_frac_digits

.pn_exp_check:
    cmp r14, r13
    jae .pn_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x65
    je .pn_exponent
    cmp al, 0x45
    jne .pn_success

.pn_exponent:
    inc r14
    cmp r14, r13
    jae .pn_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x2b
    je .pn_exp_sign
    cmp al, 0x2d
    jne .pn_exp_require_digit
.pn_exp_sign:
    inc r14

.pn_exp_require_digit:
    cmp r14, r13
    jae .pn_fail
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb .pn_fail
    cmp al, 0x39
    ja .pn_fail
    inc r14

.pn_exp_digits:
    cmp r14, r13
    jae .pn_success
    mov al, byte ptr [r12 + r14]
    cmp al, 0x30
    jb .pn_success
    cmp al, 0x39
    ja .pn_success
    inc r14
    jmp .pn_exp_digits

.pn_success:
    mov eax, 1
    ret
.pn_fail:
    xor eax, eax
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
.equ ok_msg_len, . - ok_msg
err_prefix:
    .ascii "error at "
.equ err_prefix_len, . - err_prefix
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.align 16
input_buf:
    .zero 65536
num_buf:
    .zero 64
