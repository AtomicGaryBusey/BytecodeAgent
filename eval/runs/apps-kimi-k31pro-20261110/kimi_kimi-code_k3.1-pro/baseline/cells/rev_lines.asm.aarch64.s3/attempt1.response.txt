.globl _start

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x20, x20, x0
    b read_loop

input_done:
    cbz x20, done
    mov x21, #0
    mov x22, #0

scan_loop:
    cmp x22, x20
    b.eq at_eof
    ldrb w3, [x19, x22]
    cmp w3, #10
    b.eq newline_found
    add x22, x22, #1
    b scan_loop

newline_found:
    sub x23, x22, x21
    mov x27, #0
    b output_line

at_eof:
    cmp x21, x20
    b.eq done
    sub x23, x20, x21
    mov x27, #1

output_line:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    add x0, x19, x21
    add x1, x0, x23
    mov x2, x24
    cbz x23, reverse_done

reverse_loop:
    sub x1, x1, #1
    ldrb w3, [x1]
    strb w3, [x2], #1
    subs x23, x23, #1
    b.ne reverse_loop

reverse_done:
    mov w3, #10
    strb w3, [x2], #1
    sub x2, x2, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

    cbnz x27, done
    add x22, x22, #1
    mov x21, x22
    b scan_loop

done:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
inbuf:
    .space 65536
.balign 16
outbuf:
    .space 65537
