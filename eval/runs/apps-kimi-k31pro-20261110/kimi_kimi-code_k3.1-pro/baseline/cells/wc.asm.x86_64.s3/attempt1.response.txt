.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, 4096
    syscall
    test rax, rax
    jz input_done

    add r12, rax
    mov rcx, rax
    lea rsi, [rip + input_buf]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r13
1:
    lea rdx, [rip + whitespace]
    movzx eax, byte ptr [rdx + rax]
    test al, al
    jz non_whitespace

    xor r15d, r15d
    dec rcx
    jnz scan_loop
    jmp read_loop

non_whitespace:
    test r15d, r15d
    jnz 2f
    inc r14
    mov r15d, 1
2:
    dec rcx
    jnz scan_loop
    jmp read_loop

input_done:
    lea rdi, [rip + output_buf]

    mov rax, r13
    call append_uint
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_uint
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r12
    call append_uint
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

append_uint:
    lea rsi, [rip + digit_buf + 32]
    mov ebx, 10
    test rax, rax
    jnz 2f

    dec rsi
    mov byte ptr [rsi], 48
    jmp 3f

2:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 2b

3:
    lea rdx, [rip + digit_buf + 32]
4:
    cmp rsi, rdx
    je 5f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    jmp 4b
5:
    ret

.section .rodata
whitespace:
    .zero 9
    .byte 1, 1, 1, 1, 1
    .zero 18
    .byte 1
    .zero 223

.bss
.align 16
input_buf:
    .zero 4096
output_buf:
    .zero 80
digit_buf:
    .zero 32
