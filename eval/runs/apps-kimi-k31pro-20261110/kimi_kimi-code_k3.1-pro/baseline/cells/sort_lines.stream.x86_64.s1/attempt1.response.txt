.globl _start

.text
_start:
    xor r12d, r12d

read_input:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz input_complete
    add r12, rax
    jmp read_input

input_complete:
    lea r14, [rip + input_buffer]
    lea r15, [r14 + r12]
    xor r13d, r13d
    cmp r14, r15
    je lines_parsed

parse_line:
    mov rbx, r14
    mov rdx, r14
    xor ebp, ebp

scan_line:
    cmp rdx, r15
    jae store_line
    cmp byte ptr [rdx], 10
    je found_newline
    inc rdx
    jmp scan_line

found_newline:
    mov ebp, 1

store_line:
    mov rax, r13
    shl rax, 4
    lea rdi, [rip + line_ptrs]
    add rdi, rax
    mov qword ptr [rdi], rbx
    mov rcx, rdx
    sub rcx, rbx
    mov qword ptr [rdi + 8], rcx
    inc r13

    test ebp, ebp
    jz lines_parsed
    lea r14, [rdx + 1]
    cmp r14, r15
    jb parse_line

lines_parsed:
    cmp r13, 2
    jb output_begin
    lea rdi, [rip + line_ptrs]
    lea rsi, [rip + temp_lines]
    mov rdx, r13
    call sort_range

output_begin:
    xor ebx, ebx

output_loop:
    cmp rbx, r13
    jae program_exit
    mov rax, rbx
    shl rax, 4
    lea r14, [rip + line_ptrs]
    add r14, rax

    mov eax, 1
    mov edi, 1
    mov rsi, qword ptr [r14]
    mov rdx, qword ptr [r14 + 8]
    syscall

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

    inc rbx
    jmp output_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

# sort_range(rdi = descriptors, rsi = temporary storage, rdx = count)
sort_range:
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15

    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    cmp r14, 2
    jb sort_done

    mov r15, r14
    shr r15, 1
    mov rbx, r14
    sub rbx, r15

    mov rdi, r12
    mov rsi, r13
    mov rdx, r15
    call sort_range

    mov rax, r15
    shl rax, 4
    lea rdi, [r12 + rax]
    lea rsi, [r13 + rax]
    mov rdx, rbx
    call sort_range

    mov r8, r12
    mov rax, r15
    shl rax, 4
    lea r9, [r12 + rax]
    mov r10, r9
    mov rax, r14
    shl rax, 4
    lea r11, [r12 + rax]
    mov rbp, r13

merge_loop:
    cmp r8, r10
    je merge_right_only
    cmp r9, r11
    je merge_left_only

    mov rdi, r8
    mov rsi, r9
    call compare_desc
    test eax, eax
    jg merge_take_right

merge_take_left:
    mov rax, qword ptr [r8]
    mov qword ptr [rbp], rax
    mov rax, qword ptr [r8 + 8]
    mov qword ptr [rbp + 8], rax
    add r8, 16
    add rbp, 16
    jmp merge_loop

merge_take_right:
    mov rax, qword ptr [r9]
    mov qword ptr [rbp], rax
    mov rax, qword ptr [r9 + 8]
    mov qword ptr [rbp + 8], rax
    add r9, 16
    add rbp, 16
    jmp merge_loop

merge_left_only:
    cmp r8, r10
    je merge_copy_back
    mov rax, qword ptr [r8]
    mov qword ptr [rbp], rax
    mov rax, qword ptr [r8 + 8]
    mov qword ptr [rbp + 8], rax
    add r8, 16
    add rbp, 16
    jmp merge_left_only

merge_right_only:
    cmp r9, r11
    je merge_copy_back
    mov rax, qword ptr [r9]
    mov qword ptr [rbp], rax
    mov rax, qword ptr [r9 + 8]
    mov qword ptr [rbp + 8], rax
    add r9, 16
    add rbp, 16
    jmp merge_right_only

merge_copy_back:
    mov rsi, r13
    mov rdi, r12
    mov rcx, r14
    shl rcx, 1
    rep movsq

sort_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret

# Returns -1, 0, or 1. Preserves r8-r15, rbx, and rbp.
compare_desc:
    mov rcx, qword ptr [rdi + 8]
    mov rdx, qword ptr [rsi + 8]
    mov rax, rcx
    cmp rax, rdx
    cmova rax, rdx
    xchg rcx, rax

    mov rsi, qword ptr [rsi]
    mov rdi, qword ptr [rdi]
    xchg rdi, rsi

    test rcx, rcx
    jz compare_prefix
    repe cmpsb
    jne compare_different

compare_prefix:
    cmp rax, rdx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_different:
    jb compare_less

compare_greater:
    mov eax, 1
    ret

compare_less:
    mov eax, -1
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero 32769
.align 16
line_ptrs:
    .zero 16000
.align 16
temp_lines:
    .zero 16000
