.globl _start

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    cmp x20, x21
    b.eq input_done
    sub x2, x21, x20
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_done
    add x20, x20, x0
    b read_loop

input_done:
    mov x22, #0
    mov x23, #0

scan_loop:
    cmp x22, x20
    b.eq end_of_input
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq found_newline
    add x22, x22, #1
    b scan_loop

found_newline:
    sub x24, x22, x23
    bl emit_line
    add x22, x22, #1
    mov x23, x22
    b scan_loop

end_of_input:
    cmp x23, x20
    b.eq program_exit
    sub x24, x20, x23
    bl emit_line

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

emit_line:
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    mov x10, #0
    cmp x24, #0
    b.eq append_newline

    add x11, x19, x23
    add x11, x11, x24
    sub x11, x11, #1

reverse_loop:
    ldrb w12, [x11]
    strb w12, [x9, x10]
    sub x11, x11, #1
    add x10, x10, #1
    cmp x10, x24
    b.ne reverse_loop

append_newline:
    mov w12, #10
    strb w12, [x9, x10]
    add x2, x10, #1
    mov x0, #1
    mov x1, x9
    mov x8, #64
    svc #0
    ret

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 65537
