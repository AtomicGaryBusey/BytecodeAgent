.globl _start

.section .text
_start:
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jae input_ready

    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall

    test rax, rax
    jle input_ready
    add r12, rax
    jmp read_more

input_ready:
    lea rsi, [rip + input_buffer]
    lea r8, [rsi + r12]
    lea rdi, [rip + output_buffer]
    mov rbx, rsi

scan_input:
    cmp rsi, r8
    jae input_end

    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne scan_input

    lea rcx, [rsi - 1]
reverse_terminated:
    cmp rbx, rcx
    jae terminated_reversed
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_terminated

terminated_reversed:
    mov byte ptr [rdi], 10
    inc rdi
    mov rbx, rsi
    jmp scan_input

input_end:
    cmp rbx, r8
    jae write_output

    mov rcx, r8
reverse_final:
    cmp rbx, rcx
    jae final_reversed
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_final

final_reversed:
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_success

    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
