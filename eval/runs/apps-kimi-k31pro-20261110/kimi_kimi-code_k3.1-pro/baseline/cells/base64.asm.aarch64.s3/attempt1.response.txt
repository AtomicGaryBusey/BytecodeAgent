.section .rodata
.balign 16
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    movz x21, #0xc000

read_loop:
    cmp x20, x21
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_success

    adrp x22, output_buffer
    add x22, x22, :lo12:output_buffer
    mov x23, x22
    mov x24, #0
    mov x25, x20
    adrp x26, base64_alphabet
    add x26, x26, :lo12:base64_alphabet

encode_groups:
    cmp x25, #3
    b.lo encode_tail

    ldrb w0, [x19]
    ldrb w1, [x19, #1]
    ldrb w2, [x19, #2]
    add x19, x19, #3

    lsr w3, w0, #2
    ldrb w3, [x26, x3]
    strb w3, [x22], #1

    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    ldrb w4, [x26, x4]
    strb w4, [x22], #1

    and w3, w1, #15
    lsl w3, w3, #2
    lsr w5, w2, #6
    orr w3, w3, w5
    ldrb w3, [x26, x3]
    strb w3, [x22], #1

    and w4, w2, #63
    ldrb w4, [x26, x4]
    strb w4, [x22], #1

    add x24, x24, #4
    cmp x24, #76
    b.ne 1f
    mov w0, #10
    strb w0, [x22], #1
    mov x24, #0
1:
    sub x25, x25, #3
    b encode_groups

encode_tail:
    cbnz x25, tail_present
    cbz x24, write_output
    mov w0, #10
    strb w0, [x22], #1
    b write_output

tail_present:
    ldrb w0, [x19]

    lsr w3, w0, #2
    ldrb w3, [x26, x3]
    strb w3, [x22], #1

    and w4, w0, #3
    lsl w4, w4, #4
    cmp x25, #1
    b.eq one_byte_tail

    ldrb w1, [x19, #1]
    lsr w5, w1, #4
    orr w4, w4, w5
    ldrb w4, [x26, x4]
    strb w4, [x22], #1

    and w3, w1, #15
    lsl w3, w3, #2
    ldrb w3, [x26, x3]
    strb w3, [x22], #1

    mov w4, #61
    strb w4, [x22], #1
    b tail_newline

one_byte_tail:
    ldrb w4, [x26, x4]
    strb w4, [x22], #1
    mov w4, #61
    strb w4, [x22], #1
    strb w4, [x22], #1

tail_newline:
    mov w0, #10
    strb w0, [x22], #1

write_output:
    mov x0, #1
    mov x1, x23
    sub x2, x22, x23
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input_buffer:
    .space 49152
.balign 16
output_buffer:
    .space 66560
