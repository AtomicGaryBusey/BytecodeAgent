.globl _start

.section .rodata
error_message:
    .ascii "error\n"

.text
_start:
    xor r15d, r15d

.read_more:
    mov rdx, 65536
    sub rdx, r15
    lea rsi, [rip + input_buffer]
    add rsi, r15
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle .read_done
    add r15, rax
    jmp .read_more

.read_done:
    test r15, r15
    jz .exit_program
    lea r12, [rip + input_buffer]
    lea r13, [rip + input_buffer]
    add r13, r15

.next_line:
    cmp r12, r13
    jae .exit_program
    mov r14, r12

.find_line_end:
    cmp r14, r13
    jae .line_at_input_end
    cmp byte ptr [r14], 10
    je .line_with_newline
    inc r14
    jmp .find_line_end

.line_with_newline:
    lea r15, [r14 + 1]
    jmp .begin_evaluation

.line_at_input_end:
    mov r15, r13

.begin_evaluation:
    xor ebp, ebp
    mov rbx, r12

.next_token:
    cmp rbx, r14
    jae .line_complete
    mov al, byte ptr [rbx]
    cmp al, 32
    je .skip_separator
    cmp al, 9
    jne .token_begin

.skip_separator:
    inc rbx
    jmp .next_token

.token_begin:
    mov rsi, rbx
    mov rdx, rbx

.find_token_end:
    cmp rdx, r14
    jae .token_found
    mov al, byte ptr [rdx]
    cmp al, 32
    je .token_found
    cmp al, 9
    je .token_found
    inc rdx
    jmp .find_token_end

.token_found:
    mov rbx, rdx
    mov rcx, rdx
    sub rcx, rsi
    cmp rcx, 1
    jne .parse_literal
    mov al, byte ptr [rsi]
    cmp al, 43
    je .operator_add
    cmp al, 45
    je .operator_subtract
    cmp al, 42
    je .operator_multiply
    cmp al, 47
    je .operator_divide
    cmp al, 37
    je .operator_remainder

.parse_literal:
    xor r9d, r9d
    mov rdi, rsi
    cmp byte ptr [rdi], 45
    jne .digits_start
    mov r9d, 1
    inc rdi

.digits_start:
    cmp rdi, rdx
    jae .line_error
    xor r8d, r8d
    xor ecx, ecx

.digit_loop:
    cmp rdi, rdx
    jae .literal_complete
    movzx eax, byte ptr [rdi]
    sub eax, 48
    cmp eax, 9
    ja .line_error
    inc ecx
    cmp ecx, 18
    ja .line_error
    imul r8, r8, 10
    add r8, rax
    inc rdi
    jmp .digit_loop

.literal_complete:
    test r9d, r9d
    jz .push_literal
    neg r8

.push_literal:
    mov qword ptr [value_stack + rbp*8], r8
    inc rbp
    jmp .next_token

.operator_add:
    xor r10d, r10d
    jmp .binary_operator

.operator_subtract:
    mov r10d, 1
    jmp .binary_operator

.operator_multiply:
    mov r10d, 2
    jmp .binary_operator

.operator_divide:
    mov r10d, 3
    jmp .binary_operator

.operator_remainder:
    mov r10d, 4

.binary_operator:
    cmp rbp, 2
    jb .line_error
    dec rbp
    mov r9, qword ptr [value_stack + rbp*8]
    dec rbp
    mov r8, qword ptr [value_stack + rbp*8]
    cmp r10d, 0
    je .do_addition
    cmp r10d, 1
    je .do_subtraction
    cmp r10d, 2
    je .do_multiplication
    jmp .do_division

.do_addition:
    add r8, r9
    jmp .push_result

.do_subtraction:
    sub r8, r9
    jmp .push_result

.do_multiplication:
    imul r8, r9
    jmp .push_result

.do_division:
    test r9, r9
    jz .line_error
    cmp r9, -1
    jne .division_ready
    mov r11, 0x8000000000000000
    cmp r8, r11
    je .line_error

.division_ready:
    mov rax, r8
    cqo
    idiv r9
    cmp r10d, 3
    jne .remainder_result
    mov r8, rax
    jmp .push_result

.remainder_result:
    mov r8, rdx

.push_result:
    mov qword ptr [value_stack + rbp*8], r8
    inc rbp
    jmp .next_token

.line_complete:
    cmp rbp, 1
    jne .line_error
    mov r10, qword ptr [value_stack]
    xor r9d, r9d
    test r10, r10
    jns .magnitude_ready
    mov r9d, 1
    neg r10

.magnitude_ready:
    lea rdi, [rip + number_buffer + 32]
    mov byte ptr [rdi - 1], 10
    dec rdi
    mov rax, r10
    mov r11, 10

.output_digit_loop:
    xor edx, edx
    div r11
    dec rdi
    add dl, 48
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .output_digit_loop
    test r9d, r9d
    jz .write_number
    dec rdi
    mov byte ptr [rdi], 45

.write_number:
    lea rdx, [rip + number_buffer + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    jmp .advance_line

.line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_message]
    mov edx, 6
    syscall

.advance_line:
    mov r12, r15
    jmp .next_line

.exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 8
input_buffer:
    .zero 65537
.align 8
value_stack:
    .zero 8192
number_buffer:
    .zero 32
