.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x2, #49153
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, encode_init
    add x20, x20, x0
    b read_loop

encode_init:
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x21
    mov x23, x19
    mov x25, #0
    adrp x26, base64_alphabet
    add x26, x26, :lo12:base64_alphabet

full_group_test:
    cmp x20, #3
    b.lo tail_test

    ldrb w0, [x23]
    ldrb w1, [x23, #1]
    ldrb w2, [x23, #2]
    lsl w27, w0, #16
    orr w27, w27, w1, lsl #8
    orr w27, w27, w2

    ubfx w1, w27, #18, #6
    ldrb w1, [x26, x1]
    strb w1, [x22], #1
    ubfx w1, w27, #12, #6
    ldrb w1, [x26, x1]
    strb w1, [x22], #1
    ubfx w1, w27, #6, #6
    ldrb w1, [x26, x1]
    strb w1, [x22], #1
    and w1, w27, #63
    ldrb w1, [x26, x1]
    strb w1, [x22], #1

    add x23, x23, #3
    sub x20, x20, #3
    add x25, x25, #4
    cmp x25, #76
    b.ne full_group_test
    mov w1, #10
    strb w1, [x22], #1
    mov x25, #0
    b full_group_test

tail_test:
    cbz x20, no_tail

    ldrb w0, [x23]
    lsl w27, w0, #16
    cmp x20, #1
    b.eq tail_value_ready
    ldrb w0, [x23, #1]
    orr w27, w27, w0, lsl #8

tail_value_ready:
    ubfx w1, w27, #18, #6
    ldrb w1, [x26, x1]
    strb w1, [x22], #1
    ubfx w1, w27, #12, #6
    ldrb w1, [x26, x1]
    strb w1, [x22], #1

    cmp x20, #1
    b.eq tail_one_byte

    ubfx w1, w27, #6, #6
    ldrb w1, [x26, x1]
    strb w1, [x22], #1
    mov w1, #61
    strb w1, [x22], #1
    b tail_newline

tail_one_byte:
    mov w1, #61
    strb w1, [x22], #1
    strb w1, [x22], #1

tail_newline:
    mov w1, #10
    strb w1, [x22], #1
    b write_output

no_tail:
    cbz x25, write_output
    mov w1, #10
    strb w1, [x22], #1

write_output:
    sub x2, x22, x21
    cbz x2, exit_success
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0

    mov x8, #93
    svc #0
