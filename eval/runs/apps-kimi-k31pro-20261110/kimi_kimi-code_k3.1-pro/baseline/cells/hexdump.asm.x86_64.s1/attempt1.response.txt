.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"
blank16:
    .ascii "                "

.bss
.balign 16
input_buffer:
    .zero 65536
.balign 16
line_buffer:
    .zero 80

.text
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle input_done
    add r12, rax
    jmp read_more

input_done:
    xor r13d, r13d
    lea rbp, [rip + hex_digits]

row_test:
    cmp r13, r12
    jae print_final_offset

    mov rax, r12
    sub rax, r13
    mov r14d, 16
    cmp rax, 16
    jge row_size_ready
    mov r14, rax

row_size_ready:
    lea rbx, [rip + line_buffer]
    movdqu xmm0, xmmword ptr [rip + blank16]
    movdqu xmmword ptr [rbx], xmm0
    movdqu xmmword ptr [rbx + 16], xmm0
    movdqu xmmword ptr [rbx + 32], xmm0
    movdqu xmmword ptr [rbx + 48], xmm0
    movdqu xmmword ptr [rbx + 64], xmm0

    mov rax, r13
    mov rdi, rbx
    mov ecx, 8
offset_hex_loop:
    rol rax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz offset_hex_loop

    mov byte ptr [rbx + 60], '|'
    lea r15, [rip + input_buffer]
    add r15, r13
    lea rdi, [rbx + 10]
    lea r8, [rbx + 61]
    xor ecx, ecx

byte_loop:
    movzx eax, byte ptr [r15 + rcx]
    mov r10d, eax
    mov edx, eax
    shr edx, 4
    and eax, 15
    mov r9b, byte ptr [rbp + rdx]
    mov al, byte ptr [rbp + rax]
    mov byte ptr [rdi], r9b
    mov byte ptr [rdi + 1], al

    cmp r10b, 0x20
    jb ascii_dot
    cmp r10b, 0x7e
    ja ascii_dot
    mov byte ptr [r8], r10b
    jmp ascii_done

ascii_dot:
    mov byte ptr [r8], '.'

ascii_done:
    add rdi, 3
    inc r8
    inc rcx
    cmp rcx, 7
    jne no_middle_gap
    add rdi, 1

no_middle_gap:
    cmp rcx, r14
    jl byte_loop

    mov byte ptr [r8], '|'
    mov byte ptr [r8 + 1], 10
    lea rdx, [r8 + 2]
    sub rdx, rbx
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall

    add r13, r14
    jmp row_test

print_final_offset:
    lea rbx, [rip + line_buffer]
    mov rax, r12
    mov rdi, rbx
    mov ecx, 8
final_hex_loop:
    rol rax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz final_hex_loop

    mov byte ptr [rbx + 8], 10
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov edx, 9
    syscall

    mov eax, 60
    xor edi, edi
    syscall
