.globl _start

.text
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0

read_input:
    mov x9, #65536
    cmp x20, x9
    b.hs input_complete
    mov x0, #0
    add x1, x19, x20
    sub x2, x9, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_complete
    add x20, x20, x0
    b read_input

input_complete:
    mov x21, #0
    adrp x25, value_stack
    add x25, x25, :lo12:value_stack

next_line:
    cmp x21, x20
    b.hs program_exit

    mov x22, x21
find_line_end:
    cmp x22, x20
    b.hs line_end_found
    ldrb w9, [x19, x22]
    cmp w9, #10
    b.eq line_end_found
    add x22, x22, #1
    b find_line_end

line_end_found:
    mov x23, x21
    mov x24, #0

skip_separators:
    cmp x23, x22
    b.hs finish_line
    ldrb w9, [x19, x23]
    cmp w9, #32
    b.eq skip_one_separator
    cmp w9, #9
    b.eq skip_one_separator

    mov x26, x23
scan_token:
    cmp x23, x22
    b.hs token_scanned
    ldrb w9, [x19, x23]
    cmp w9, #32
    b.eq token_scanned
    cmp w9, #9
    b.eq token_scanned
    add x23, x23, #1
    b scan_token

token_scanned:
    mov x27, x23
    sub x28, x27, x26
    ldrb w9, [x19, x26]
    cmp x28, #1
    b.ne multi_byte_token

    sub w10, w9, #48
    cmp w10, #9
    b.ls positive_literal
    cmp w9, #43
    b.eq have_operator
    cmp w9, #45
    b.eq have_operator
    cmp w9, #42
    b.eq have_operator
    cmp w9, #47
    b.eq have_operator
    cmp w9, #37
    b.eq have_operator
    b line_error

multi_byte_token:
    cmp w9, #45
    b.ne positive_literal
    mov x10, #1
    sub x11, x28, #1
    cmp x11, #18
    b.hi line_error
    mov x12, #1
    b parse_literal

positive_literal:
    mov x10, #0
    cmp x28, #18
    b.hi line_error
    mov x12, #0

parse_literal:
    mov x13, #0
literal_digit_loop:
    cmp x10, x28
    b.hs literal_complete
    add x14, x26, x10
    ldrb w9, [x19, x14]
    sub w14, w9, #48
    cmp w14, #9
    b.hi line_error
    lsl x15, x13, #3
    add x13, x15, x13, lsl #1
    add x13, x13, x14
    add x10, x10, #1
    b literal_digit_loop

literal_complete:
    cbz x12, push_literal
    neg x13, x13
push_literal:
    str x13, [x25, x24, lsl #3]
    add x24, x24, #1
    b skip_separators

have_operator:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x14, [x25, x24, lsl #3]
    sub x24, x24, #1
    ldr x13, [x25, x24, lsl #3]

    cmp w9, #43
    b.eq operator_add
    cmp w9, #45
    b.eq operator_subtract
    cmp w9, #42
    b.eq operator_multiply
    b operator_divide_or_remainder

operator_add:
    add x13, x13, x14
    b store_operator_result

operator_subtract:
    sub x13, x13, x14
    b store_operator_result

operator_multiply:
    mul x13, x13, x14
    b store_operator_result

operator_divide_or_remainder:
    cbz x14, line_error
    mov x15, #1
    lsl x15, x15, #63
    cmp x13, x15
    b.ne division_safe
    cmn x14, #1
    b.eq line_error
division_safe:
    sdiv x15, x13, x14
    cmp w9, #37
    b.eq calculate_remainder
    mov x13, x15
    b store_operator_result
calculate_remainder:
    msub x13, x15, x14, x13

store_operator_result:
    str x13, [x25, x24, lsl #3]
    add x24, x24, #1
    b skip_separators

finish_line:
    cmp x24, #1
    b.ne line_error
    ldr x13, [x25]

    mov x10, #0
    mov x18, x13
    tbz x13, #63, magnitude_ready
    neg x18, x13
    mov x10, #1

magnitude_ready:
    adrp x16, number_buffer
    add x16, x16, :lo12:number_buffer
    add x17, x16, #31
    mov w9, #10
    strb w9, [x17], #-1
    cbnz x18, convert_digits
    mov w9, #48
    strb w9, [x17], #-1
    b conversion_sign

convert_digits:
    mov x11, #10
    udiv x12, x18, x11
    msub x14, x12, x11, x18
    add w14, w14, #48
    strb w14, [x17], #-1
    mov x18, x12
    cbnz x18, convert_digits

conversion_sign:
    cbz x10, write_number
    mov w9, #45
    strb w9, [x17], #-1

write_number:
    add x1, x17, #1
    add x2, x16, #32
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    b line_consumed

line_error:
    mov x0, #1
    adrp x1, error_message
    add x1, x1, :lo12:error_message
    mov x2, #6
    mov x8, #64
    svc #0

line_consumed:
    cmp x22, x20
    b.hs program_exit
    add x21, x22, #1
    b next_line

skip_one_separator:
    add x23, x23, #1
    b skip_separators

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_message:
    .ascii "error\n"

.bss
.balign 16
input_buffer:
    .zero 65536

    .zero 8192

    .zero 32
