.text
.globl _start
_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer

read_loop:
    mov x0, xzr
    mov x2, #65536
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x19, x19, x0
    mov x3, x1
    mov x4, x0

byte_loop:
    ldrb w5, [x3], #1

    cmp w5, #10
    b.ne 1f
    add x20, x20, #1
1:
    sub w6, w5, #9
    cmp w6, #4
    b.ls whitespace
    cmp w5, #32
    b.eq whitespace

    cbnz x22, byte_done
    add x21, x21, #1
    mov x22, #1
    b byte_done

whitespace:
    mov x22, xzr

byte_done:
    subs x4, x4, #1
    b.ne byte_loop
    b read_loop

input_done:
    adrp x23, output_buffer
    add x23, x23, :lo12:output_buffer
    mov x24, x23

    mov x0, x20
    bl append_number
    mov w5, #32
    strb w5, [x24], #1

    mov x0, x21
    bl append_number
    mov w5, #32
    strb w5, [x24], #1

    mov x0, x19
    bl append_number
    mov w5, #10
    strb w5, [x24], #1

    sub x2, x24, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0

    mov x0, xzr
    mov x8, #93
    svc #0
1:
    b 1b

append_number:
    add x1, x24, #32
    mov x2, x1
    mov x4, #10
1:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #48
    strb w5, [x1, #-1]!
    mov x0, x3
    cbnz x0, 1b
2:
    cmp x1, x2
    b.hs 3f
    ldrb w5, [x1], #1
    strb w5, [x24], #1
    b 2b
3:
    ret

.bss
.balign 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 128
