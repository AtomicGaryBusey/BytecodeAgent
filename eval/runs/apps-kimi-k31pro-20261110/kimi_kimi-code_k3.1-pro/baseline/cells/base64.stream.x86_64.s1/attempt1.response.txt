.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor r12d, r12d

read_input:
    cmp r12, 49152
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 49152
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_input

input_done:
    lea r13, [rip + input_buffer]
    lea r14, [r13 + r12]
    lea r15, [rip + output_buffer]
    lea rbp, [rip + alphabet]
    xor ebx, ebx

encode_loop:
    mov rax, r14
    sub rax, r13
    cmp rax, 3
    jae full_group
    test rax, rax
    jz finish_lines
    cmp rax, 1
    je final_one
    jmp final_two

full_group:
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]
    movzx edx, byte ptr [r13 + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r15], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r15 + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r15 + 2], dl

    and eax, 63
    mov al, byte ptr [rbp + rax]
    mov byte ptr [r15 + 3], al

    add r13, 3
    add r15, 4
    add ebx, 4
    cmp ebx, 76
    jne encode_loop
    mov byte ptr [r15], 10
    inc r15
    xor ebx, ebx
    jmp encode_loop

final_one:
    movzx eax, byte ptr [r13]

    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r15], dl

    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbp + rax]
    mov byte ptr [r15 + 1], al

    mov byte ptr [r15 + 2], 61
    mov byte ptr [r15 + 3], 61
    add r15, 4
    add ebx, 4
    jmp finish_lines

final_two:
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13 + 1]

    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [rbp + rdx]
    mov byte ptr [r15], dl

    and eax, 3
    shl eax, 4
    mov edx, ecx
    shr edx, 4
    or eax, edx
    mov al, byte ptr [rbp + rax]
    mov byte ptr [r15 + 1], al

    and ecx, 15
    shl ecx, 2
    mov cl, byte ptr [rbp + rcx]
    mov byte ptr [r15 + 2], cl

    mov byte ptr [r15 + 3], 61
    add r15, 4
    add ebx, 4

finish_lines:
    test ebx, ebx
    jz write_output
    mov byte ptr [r15], 10
    inc r15

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r15
    sub rdx, rsi
    jz exit_success
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66560
