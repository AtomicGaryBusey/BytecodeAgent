.globl _start

.equ INPUT_CAP, 65536

.text
_start:
    xor r12d, r12d

read_more:
    cmp r12, INPUT_CAP
    jae input_ready

    mov edx, INPUT_CAP
    sub rdx, r12
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall

    test rax, rax
    jle input_ready
    add r12, rax
    jmp read_more

input_ready:
    lea r8, [rip + input_buf]
    lea r10, [r8 + r12]
    lea rdi, [rip + output_buf]

next_line:
    cmp r8, r10
    jae output_ready

    mov r9, r8

find_line_end:
    cmp r9, r10
    jae final_line
    cmp byte ptr [r9], 10
    je terminated_line
    inc r9
    jmp find_line_end

terminated_line:
    mov rcx, r9

reverse_terminated:
    cmp rcx, r8
    jbe terminated_reversed
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_terminated

terminated_reversed:
    mov byte ptr [rdi], 10
    inc rdi
    lea r8, [r9 + 1]
    jmp next_line

final_line:
    mov rcx, r10

reverse_final:
    cmp rcx, r8
    jbe final_reversed
    dec rcx
    mov al, byte ptr [rcx]
    mov byte ptr [rdi], al
    inc rdi
    jmp reverse_final

final_reversed:
    mov byte ptr [rdi], 10
    inc rdi

output_ready:
    lea rsi, [rip + output_buf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_success

    mov edi, 1
    mov eax, 1
    syscall

exit_success:
    xor edi, edi
    mov eax, 60
    syscall

.bss
.balign 16
input_buf:
    .zero INPUT_CAP
output_buf:
    .zero INPUT_CAP + 1
