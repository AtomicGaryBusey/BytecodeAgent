.globl _start

.text
_start:
    lea r15, [rip + input_buf]
    xor r12d, r12d

read_more:
    cmp r12, 32768
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_more

read_done:
    lea rbp, [rip + lines]
    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d

scan_loop:
    cmp rbx, r12
    jae scan_done
    cmp byte ptr [r15 + rbx], 10
    jne scan_next

    mov dword ptr [rbp + r13*8], r14d
    mov rax, rbx
    sub rax, r14
    mov dword ptr [rbp + r13*8 + 4], eax
    inc r13
    lea r14, [rbx + 1]

scan_next:
    inc rbx
    jmp scan_loop

scan_done:
    cmp r14, r12
    je sort_begin
    mov dword ptr [rbp + r13*8], r14d
    mov rax, r12
    sub rax, r14
    mov dword ptr [rbp + r13*8 + 4], eax
    inc r13

sort_begin:
    mov ebx, 1

sort_outer:
    cmp rbx, r13
    jae output_begin
    mov r14, qword ptr [rbp + rbx*8]
    mov r12, rbx

sort_inner:
    test r12, r12
    jz sort_place
    mov rdi, qword ptr [rbp + r12*8 - 8]
    mov rsi, r14
    call compare_lines
    test eax, eax
    jle sort_place
    mov rax, qword ptr [rbp + r12*8 - 8]
    mov qword ptr [rbp + r12*8], rax
    dec r12
    jmp sort_inner

sort_place:
    mov qword ptr [rbp + r12*8], r14
    inc rbx
    jmp sort_outer

compare_lines:
    mov r8, rdi
    shr r8, 32
    mov r9, rsi
    shr r9, 32
    mov r10, r8
    cmp r10, r9
    cmova r10, r9

    mov eax, edi
    lea rdx, [rip + input_buf]
    add rdx, rax
    mov eax, esi
    lea r11, [rip + input_buf]
    add r11, rax
    xor ecx, ecx

compare_bytes:
    cmp rcx, r10
    jae compare_lengths
    movzx eax, byte ptr [rdx + rcx]
    movzx edi, byte ptr [r11 + rcx]
    cmp eax, edi
    jb compare_less
    ja compare_greater
    inc rcx
    jmp compare_bytes

compare_lengths:
    cmp r8, r9
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

output_begin:
    lea r14, [rip + output_buf]
    mov rdi, r14
    xor ebx, ebx

output_loop:
    cmp rbx, r13
    jae output_done
    mov rax, qword ptr [rbp + rbx*8]
    mov rcx, rax
    shr rcx, 32
    mov esi, eax
    add rsi, r15
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp output_loop

output_done:
    cmp rdi, r14
    je exit_success
    mov rdx, rdi
    sub rdx, r14
    mov rsi, r14
    mov eax, 1
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.balign 16
input_buf:
    .zero 32768
.balign 16
lines:
    .zero 8000
.balign 16
output_buf:
    .zero 32769
