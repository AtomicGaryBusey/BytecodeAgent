	.globl _start
	.text
_start:
	// read all input into buffer
	mov	x8, #63			// sys_read
	mov	x0, #0			// fd = stdin
	adrp	x1, input_buf
	add	x1, x1, :lo12:input_buf
	mov	x2, #65536		// max bytes
1:	svc	0
	cmp	x0, #0
	beq	2f			// EOF
	// x0 = bytes read, advance pointer
	add	x1, x1, x0
	b	1b
2:
	// write "ok\n"
	mov	x8, #64			// sys_write
	mov	x0, #1			// fd = stdout
	adrp	x1, ok_msg
	add	x1, x1, :lo12:ok_msg
	mov	x2, #3			// length
	svc	0
	// exit status 0
	mov	x8, #93			// sys_exit
	mov	x0, #0
	svc	0

	// data
	.section .rodata
ok_msg:
	.ascii "ok\n"

	.bss
	.align 4
input_buf:
	.zero 65536
