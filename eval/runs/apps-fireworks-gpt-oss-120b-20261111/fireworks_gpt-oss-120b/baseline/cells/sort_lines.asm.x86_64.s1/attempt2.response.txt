.globl _start

.section .bss
.align 16
input_buffer:
    .space 32768
line_offsets:
    .space 1000*4
line_lengths:
    .space 1000*4

.section .rodata
newline_char:
    .byte 0x0A

.text
_start:
    xor r8, r8                # total_read = 0

read_loop:
    mov rax, 0                # sys_read
    mov rdi, 0                # fd = stdin
    lea rsi, [rip + input_buffer]
    add rsi, r8
    mov rdx, 32768
    sub rdx, r8               # remaining space
    syscall
    test rax, rax
    jz read_done
    cmp rax, 0
    jl exit_err
    add r8, rax               # total_read += rax
    cmp r8, 32768
    jae read_done
    jmp read_loop

read_done:
    test r8, r8
    jz exit_success

    xor ebx, ebx              # line_count = 0
    xor r9, r9                # src_offset = 0

parse_loop:
    cmp r9, r8
    jae parse_done

    mov r10, r9               # line_start = src_offset
    lea rdi, [rip + input_buffer]
    add rdi, r9               # rdi points to current byte
    mov rcx, r8
    sub rcx, r9               # remaining bytes

scan_loop:
    test rcx, rcx
    jz no_newline
    mov al, byte ptr [rdi]
    cmp al, 0x0A
    je found_newline
    inc rdi
    inc r9
    dec rcx
    jmp scan_loop

found_newline:
    mov rax, r9
    sub rax, r10              # length = r9 - line_start
    mov edx, ebx
    lea rsi, [rip + line_offsets]
    mov [rsi + rdx*4], r10d
    lea rsi, [rip + line_lengths]
    mov [rsi + rdx*4], eax
    inc ebx
    inc r9                     # skip newline
    jmp parse_loop

no_newline:
    mov rax, r8
    sub rax, r10               # length = end - line_start
    mov edx, ebx
    lea rsi, [rip + line_offsets]
    mov [rsi + rdx*4], r10d
    lea rsi, [rip + line_lengths]
    mov [rsi + rdx*4], eax
    inc ebx
    mov r9, r8
    jmp parse_loop

parse_done:
    # insertion sort
    xor esi, esi               # i = 0

ins_outer:
    cmp esi, ebx
    jge ins_done

    mov edx, esi
    lea rdi, [rip + line_offsets]
    mov eax, [rdi + rdx*4]
    mov r10d, eax              # key_offset
    lea rdi, [rip + line_lengths]
    mov eax, [rdi + rdx*4]
    mov r11d, eax              # key_len

    mov ecx, esi
    dec ecx                    # j = i-1

ins_inner:
    test ecx, ecx
    jl ins_place_key           # if j < 0, place key

    lea rdi, [rip + line_offsets]
    mov eax, [rdi + rcx*4]
    mov r12d, eax              # offset a
    lea rdi, [rip + line_lengths]
    mov eax, [rdi + rcx*4]
    mov r13d, eax              # len a

    push rcx
    mov rdx, r12                # offset a
    mov r8d, r13d               # len a
    mov rcx, r10                # offset b (key)
    mov r9d, r11d               # len b
    call line_compare
    pop rcx

    cmp eax, 0
    jg ins_shift               # if a > b, shift

    jmp ins_place_key

ins_shift:
    mov edx, ecx
    inc edx                     # dest = j+1
    lea rdi, [rip + line_offsets]
    mov eax, [rdi + rcx*4]
    mov [rdi + rdx*4], eax
    lea rdi, [rip + line_lengths]
    mov eax, [rdi + rcx*4]
    mov [rdi + rdx*4], eax
    dec ecx
    jmp ins_inner

ins_place_key:
    inc ecx                     # place at j+1
    mov edx, ecx
    lea rdi, [rip + line_offsets]
    mov [rdi + rdx*4], r10d
    lea rdi, [rip + line_lengths]
    mov [rdi + rdx*4], r11d
    inc esi
    jmp ins_outer

ins_done:
    xor edi, edi                # output index i = 0

output_loop:
    cmp edi, ebx
    jge output_done

    lea rsi, [rip + line_offsets]
    mov eax, [rsi + rdi*4]
    mov r8d, eax                # offset
    lea rsi, [rip + line_lengths]
    mov eax, [rsi + rdi*4]
    mov r9d, eax                # length

    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    lea rsi, [rip + input_buffer]
    add rsi, r8
    mov rdx, r9
    call write_all

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    call write_all

    inc edi
    jmp output_loop

output_done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

exit_err:
    mov rax, 60
    mov rdi, 1
    syscall

# ------------------------------------------------------------
# write_all: write full buffer (rdi=fd, rsi=ptr, rdx=count)
write_all:
    test rdx, rdx
    jz .w_done
    mov rax, 1
    syscall
    cmp rax, 0
    je .w_error
    sub rdx, rax
    add rsi, rax
    jnz write_all
.w_done:
    ret
.w_error:
    mov rax, 60
    xor rdi, rdi
    syscall

# ------------------------------------------------------------
# line_compare: compare two lines
# rdx = offset_a, r8d = len_a
# rcx = offset_b, r9d = len_b
# return signed int in eax: <0 if a<b, 0 if equal, >0 if a>b
line_compare:
    mov r10d, r8d          # len_a_rem
    mov r11d, r9d          # len_b_rem

    lea rsi, [rip + input_buffer]
    add rsi, rdx           # ptr_a
    lea rdi, [rip + input_buffer]
    add rdi, rcx           # ptr_b

cmp_loop:
    test r10d, r10d
    jz cmp_end
    test r11d, r11d
    jz cmp_end
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rdi]
    cmp al, bl
    jne cmp_diff
    inc rsi
    inc rdi
    dec r10d
    dec r11d
    jmp cmp_loop

cmp_end:
    mov eax, r8d
    sub eax, r9d
    ret

cmp_diff:
    movzx eax, al
    movzx ebx, bl
    sub eax, ebx
    ret
