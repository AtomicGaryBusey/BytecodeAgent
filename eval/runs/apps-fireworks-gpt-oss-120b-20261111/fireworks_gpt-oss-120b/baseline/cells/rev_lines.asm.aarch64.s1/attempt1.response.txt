    .global _start
    .text
    .align 2

    // constants
    .equ BUFFER_SIZE, 65536

_start:
    // x19 = total length read so far
    mov     x19, #0
    // x20 = address of buffer
    adr     x20, buffer

read_loop:
    // x0 = fd 0 (stdin)
    mov     x0, #0
    // x1 = buffer + total_len
    add     x1, x20, x19
    // x2 = remaining space = BUFFER_SIZE - total_len
    movz    x2, #0x1, lsl #16        // x2 = 65536
    sub     x2, x2, x19
    // syscall read
    mov     x8, #63
    svc     #0
    // if returned 0 -> EOF
    cbz     x0, read_done
    // add returned count to total_len
    add     x19, x19, x0
    b       read_loop

read_done:
    // if total length zero, exit
    cbz     x19, exit

    // x21 = current offset (index) into buffer
    mov     x21, #0

line_loop:
    // if offset >= total length, done
    cmp     x21, x19
    bge     exit

    // x22 = scan index, start at current offset
    mov     x22, x21

scan_newline:
    // if scan index >= total length, no newline found
    cmp     x22, x19
    bge     no_newline
    // load byte
    ldrb    w0, [x20, x22]
    // compare with '\n' (0x0A)
    cmp     w0, #10
    beq     found_newline
    // advance scan index
    add     x22, x22, #1
    b       scan_newline

found_newline:
    // line length = x22 - x21
    sub     x23, x22, x21          // x23 = line_len
    // reverse output loop
    mov     x24, x23               // remaining bytes
    sub     x25, x23, #1           // i = line_len-1

rev_byte_loop:
    cbz     x24, rev_done
    // address = buffer + start + i
    add     x26, x21, x25
    ldrb    w0, [x20, x26]
    // write single byte
    adr     x1, out_buf
    strb    w0, [x1]
    mov     x0, #1                 // fd stdout
    mov     x2, #1
    mov     x8, #64
    svc     #0
    // decrement counters
    sub     x25, x25, #1
    sub     x24, x24, #1
    b       rev_byte_loop

rev_done:
    // write newline after line
    adr     x1, out_buf
    mov     w0, #10                // '\n'
    strb    w0, [x1]
    mov     x0, #1
    mov     x2, #1
    mov     x8, #64
    svc     #0
    // advance offset past the newline
    add     x21, x22, #1
    b       line_loop

no_newline:
    // final line without terminating newline
    sub     x23, x19, x21          // line_len = total_len - offset
    // if zero length, nothing to output
    cbz     x23, exit

    // reverse output loop for final line
    mov     x24, x23
    sub     x25, x23, #1

rev_final_loop:
    cbz     x24, rev_final_done
    add     x26, x21, x25
    ldrb    w0, [x20, x26]
    adr     x1, out_buf
    strb    w0, [x1]
    mov     x0, #1
    mov     x2, #1
    mov     x8, #64
    svc     #0
    sub     x25, x25, #1
    sub     x24, x24, #1
    b       rev_final_loop

rev_final_done:
    // write final newline
    adr     x1, out_buf
    mov     w0, #10
    strb    w0, [x1]
    mov     x0, #1
    mov     x2, #1
    mov     x8, #64
    svc     #0
    b       exit

exit:
    // exit status 0
    mov     x0, #0
    mov     x8, #93
    svc     #0

    // -------------------------------------------------------------------------
    // Data sections
    .section .bss
    .align 3
buffer:
    .zero BUFFER_SIZE
out_buf:
    .zero 1
