.globl _start
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 16
buffer:      .space MAX_INPUT
line_ptrs:   .space MAX_LINES*8
line_lens:   .space MAX_LINES*4

.section .rodata
nl: .byte 0x0a

.text
_start:
    xor ebp, ebp                    # line count = 0

    # read loop
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, MAX_INPUT
    xor rax, rax                    # sys_read = 0
    syscall
    test rax, rax
    jz parse_input
    mov r12, rax                    # total bytes read so far

read_more:
    mov rdi, 0
    lea rsi, [rip + buffer + r12]
