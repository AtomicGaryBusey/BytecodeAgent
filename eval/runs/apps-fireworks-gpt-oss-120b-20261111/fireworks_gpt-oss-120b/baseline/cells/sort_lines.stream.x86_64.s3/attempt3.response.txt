.globl _start
.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 16
buffer:      .space MAX_INPUT
line_ptrs:   .space MAX_LINES*8
line_lens:   .space MAX_LINES*4

.section .rodata
nl: .byte 0x0a

.text
_start:
    xor    ebx, ebx            # line count = 0
    xor    r12, r12            # total bytes read = 0

read_loop:
    mov    rax, 0              # sys_read
    mov    rdi, 0              # fd = stdin
    lea    rsi, [rip + buffer]
    add    rsi, r12            # buffer + offset
    mov    rdx, MAX_INPUT
    sub    rdx, r12            # remaining space
    syscall
    test   rax, rax
    jz     parse_input
    add    r12, rax
    jmp    read_loop

parse_input:
    xor    rsi, rsi            # rsi = 0 (offset into buffer)
    mov    rcx, r12            # rcx = remaining bytes
    jrcxz  sort_and_output     # no input

parse_line:
    cmp    rcx, 0
    je     sort_and_output

    xor    r8d, r8d            # r8 = line length counter
scan_newline:
    cmp    r8, rcx
    jae    no_newline
    mov    al, byte ptr [rip + buffer + rsi + r8]
