.section .bss
    .align 16
buffer:     .zero 65536
outbuf:     .zero 65537

.text
.globl _start
_start:
    xor rcx, rcx                     # total length = 0
    lea rbx, [rip + buffer]          # write pointer = buffer

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    mov rsi, rbx                     # destination
    mov rdx, 65536
    sub rdx, rcx                     # remaining space
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                     # total length += bytes read
    add rbx, rax                     # advance write pointer
    jmp read_loop

read_done:
    lea rsi, [rip + buffer]          # rsi = line start
    mov rdi, rbx                     # rdi = buffer + total length (end)

process_loop:
    cmp rsi, rdi
    jge exit_program                 # no more data

    mov rbx, rsi                     # rbx = scan pointer

find_nl:
    cmp rbx, rdi
    je no_nl_found
    mov al, [rbx]
    cmp al, 0x0A
    je have_nl
    inc rbx
    jmp find_nl

have_nl:
    mov rax, rbx                     # rax = newline address
    sub rax, rsi                     # rax = line length
    call reverse_write
    inc rbx                          # skip newline
    mov rsi, rbx
    jmp process_loop

no_nl_found:
    mov rax, rdi                     # rax = end address
    sub rax, rsi                     # rax = final line length
    test rax, rax
    jz exit_program                  # empty remainder => no line
    call reverse_write
    jmp exit_program

# rsi = line start, rax = line length
reverse_write:
    lea r8, [rip + outbuf]           # output buffer pointer
    mov rcx, rax                     # rcx = remaining length
    test rcx, rcx
    jz write_nl
    lea r9, [rsi + rax]              # r9 = line_end (past last byte)

rev_loop:
    dec r9
    mov al, [r9]
    mov [r8], al
    inc r8
    dec rcx
    jnz rev_loop

write_nl:
    mov byte ptr [r8], 0x0A
    inc r8
    mov rdx, rax                     # rdx = original length
    inc rdx                          # + newline
    mov rax, 1                       # sys_write
    mov rdi, 1                       # fd = stdout
    lea rsi, [rip + outbuf]
    syscall
    ret

exit_program:
    mov rax, 60                      # sys_exit
    xor rdi, rdi
    syscall
