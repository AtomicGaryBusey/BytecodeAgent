.section .bss
    .align 16
buffer: .zero 65536
outbuf: .zero 65537        # space for reversed line + newline
input_len: .quad 0

.text
.globl _start
_start:
    # read whole input into buffer
    lea rbx, [rip + buffer]        # current write pointer
    xor rcx, rcx                  # total length so far
read_loop:
    mov rax, 0                    # sys_read
    mov rdi, 0                    # fd stdin
    mov rsi, rbx                  # dst = current pointer
    mov rdx, 65536
    sub rdx, rcx                  # remaining space
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                  # total += bytes read
    add rbx, rax                  # advance pointer
    jmp read_loop
read_done:
    mov [rip + input_len], rcx

    # set up scanning pointers
    lea rsi, [rip + buffer]       # rsi = current scan position
    mov rdx, rcx                  # rdx = total length
    add rdx, rsi                  # rdx = end pointer (buffer + total)

scan_loop:
    cmp rsi, rdx
    jge end_program               # no more data

    # find next newline
    mov rbx, rsi                  # rbx = scan pointer
find_nl:
    cmp rbx, rdx
    jge no_nl                     # reached end without newline
    mov al, byte ptr [rbx]
    cmp al, 0x0A
    je have_nl
    inc rbx
    jmp find_nl

have_nl:
    # line length = rbx - rsi
    mov rcx, rbx
    sub rcx, rsi
    call process_line
    inc rbx                       # skip the newline
    mov rsi, rbx
    jmp scan_loop

no_nl:
    # final line (no terminating newline)
    mov rcx, rdx
    sub rcx, rsi                  # length = end - start
    test rcx, rcx
    jz end_program                # empty remainder => no line
    call process_line
    jmp end_program

process_line:                     # rsi = line start, rcx = line length
    lea r8, [rip + outbuf]        # output buffer pointer
    mov rdx, rcx                  # save original length
    test rcx, rcx
    jz write_nl
    lea r9, [rsi + rcx]           # r9 = line_end (past last byte)
rev_loop:
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [r8], al
    inc r8
    dec rcx
    jnz rev_loop
write_nl:
    mov byte ptr [r8], 0x0A
    inc r8
    mov rax, 1                    # sys_write
    mov rdi, 1                    # stdout
    lea rsi, [rip + outbuf]
    mov rdx, rdx                  # original length
    inc rdx                       # + newline
    syscall
    ret

end_program:
    mov rax, 60                   # sys_exit
    xor rdi, rdi
    syscall
