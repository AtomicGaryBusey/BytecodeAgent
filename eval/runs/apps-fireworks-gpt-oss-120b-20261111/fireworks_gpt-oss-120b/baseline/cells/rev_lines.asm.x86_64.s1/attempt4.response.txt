.section .bss
    .align 16
buffer: .zero 65536
outbuf: .zero 65537

.text
.globl _start
_start:
    xor rcx, rcx                     # total bytes read = 0
    lea rbx, [rip + buffer]          # write pointer = buffer

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    mov rsi, rbx                     # dest = current pointer
    mov rdx, 65536
    sub rdx, rcx                     # space left
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                     # total += bytes read
    add rbx, rax                     # advance pointer
    jmp read_loop

read_done:
    lea rsi, [rip + buffer]          # line start
    mov rdi, rbx                     # end = buffer + total length
    mov rbx, rsi                     # rbx = current line start

process_loop:
    cmp rbx, rdi
    jae exit_program                 # no more data

    mov rcx, rbx                     # rcx = scan pointer

find_nl:
    cmp rcx, rdi
    jae no_nl_found                  # reached end, no newline
    mov al, byte ptr [rcx]
    cmp al, 0x0A
    je have_nl
    inc rcx
    jmp find_nl

have_nl:
    mov rax, rcx                     # rax = newline address
    sub rax, rbx                     # rax = line length
    # reverse copy into outbuf
    mov rdx, rax                     # rdx = remaining
    test rdx, rdx
    jz write_nl_only                 # empty line
    lea rsi, [rbx]                   # src start
    lea rdi, [rip + outbuf]          # dest start
    add rsi, rdx                     # point to src_end
rev_copy:
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rdx
    jnz rev_copy

write_nl_only:
    mov byte ptr [rdi], 0x0A
    inc rdi
    mov rdx, rax                     # rdx = line length
    inc rdx                          # + newline
    mov rax, 1                       # sys_write
    mov rdi, 1                       # fd = stdout
    lea rsi, [rip + outbuf]
    syscall

    inc rcx                          # skip newline
    mov rbx, rcx
    jmp process_loop

no_nl_found:
    mov rax, rdi                     # rax = end address
    sub rax, rbx                     # rax = final line length
    test rax, rax
    jz exit_program                  # empty remainder => no line
    # reverse copy final line
    mov rdx, rax
    lea rsi, [rbx]
    lea rdi, [rip + outbuf]
    add rsi, rdx
rev_copy_last:
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    dec rdx
    jnz rev_copy_last

    mov byte ptr [rdi], 0x0A
    inc rdi
    mov rdx, rax
    inc rdx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    syscall

    jmp exit_program

exit_program:
    mov rax, 60                      # sys_exit
    xor rdi, rdi
    syscall
