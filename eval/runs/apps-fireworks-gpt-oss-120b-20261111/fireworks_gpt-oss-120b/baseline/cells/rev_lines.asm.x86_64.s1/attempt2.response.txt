.section .bss
    .align 16
buffer:     .zero 65536
outbuf:     .zero 65537
input_len:  .zero 8

.text
.globl _start
_start:
    lea rbx, [rip + buffer]          # write pointer
    xor rcx, rcx                     # total length so far

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # stdin
    mov rsi, rbx                     # destination buffer
    mov rdx, 65536
    sub rdx, rcx                     # remaining space
    syscall
    test rax, rax
    jz read_done
    add rcx, rax
    add rbx, rax
    jmp read_loop

read_done:
    mov [rip + input_len], rcx

    lea rsi, [rip + buffer]          # scan position
    mov rdx, rcx                     # total length
    add rdx, rsi                     # end pointer (buffer + length)

scan_loop:
    cmp rsi, rdx
    jge end_program                  # no more data

    mov rbx, rsi                     # search pointer

find_nl:
    cmp rbx, rdx
    jge no_nl                         # reached end without newline
    mov al, [rbx]
    cmp al, 0x0A
    je have_nl
    inc rbx
    jmp find_nl

have_nl:
    mov rcx, rbx
    sub rcx, rsi                     # line length
    call process_line
    inc rbx                           # skip newline
    mov rsi, rbx
    jmp scan_loop

no_nl:
    mov rcx, rdx
    sub rcx, rsi                     # length of final line
    test rcx, rcx
    jz end_program                    # empty remainder => no line
    call process_line
    jmp end_program

# rsi = line start, rcx = line length
process_line:
    lea r8, [rip + outbuf]           # output buffer pointer
    mov rdx, rcx                     # save original length
    test rcx, rcx
    jz write_nl
    lea r9, [rsi + rcx]              # line_end (past last byte)

rev_loop:
    dec r9
    mov al, [r9]
    mov [r8], al
    inc r8
    dec rcx
    jnz rev_loop

write_nl:
    mov byte ptr [r8], 0x0A
    inc r8
    mov rax, 1                       # sys_write
    mov rdi, 1                       # stdout
    lea rsi, [rip + outbuf]
    mov rdx, rdx                     # original length
    inc rdx                          # + newline
    syscall
    ret

end_program:
    mov rax, 60                      # sys_exit
    xor rdi, rdi
    syscall
