.globl _start

.section .bss
.align 16
input_buf:
    .zero 65536               # input buffer
stack:
    .zero 2000*8              # 2000 int64 slots
out_buf:
    .zero 32                  # temporary output buffer

.section .rodata
error_msg:
    .asciz "error\n"
error_len = . - error_msg

.section .text

_start:
    # -------------------------------------------------
    # read all input into input_buf (max 65536 bytes)
    # rbx = total length
    xor rbx, rbx                # total length = 0
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    lea rsi, [rel input_buf + rbx]
