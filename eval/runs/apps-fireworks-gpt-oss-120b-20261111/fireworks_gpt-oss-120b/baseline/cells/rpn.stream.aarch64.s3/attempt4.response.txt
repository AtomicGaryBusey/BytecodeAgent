    .globl _start
    .text
    .equ    MAX_INPUT, 65536
    .equ    MAX_STACK, 2048          // 2048 entries * 8 = 16384 bytes
    .equ    OUTBUF_SIZE, 32

// ---------------------------------------------------------------------
// Data
    .section .rodata
err_msg:
    .ascii  "error\n"
err_len = . - err_msg

newline:
    .byte   10                     // '\n'

    .section .bss
    .align  8
input_buf:
    .zero   MAX_INPUT

    .align  8
stack_buf:
    .zero   MAX_STACK * 8

    .align  8
out_buf:
    .zero   OUTBUF_SIZE

// ---------------------------------------------------------------------
// Entry point
_start:
    // -------------------------------------------------------------
    // Read entire input into input_buf
    mov     x9, #0                  // total_len = 0
