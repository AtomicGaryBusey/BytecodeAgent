    .globl _start
    .text
_start:
    // stack base address
    adr     x4, stack               // x4 = &stack[0]

    // read entire input into input_buf
    mov     x0, #0                  // fd = stdin
    adr     x1, input_buf
    mov     x2, #65536
    mov     x8, #63                 // sys_read
    svc     #0
    cbz     x0, read_done
    mov     x9, x0                  // total_len = bytes read
read_loop:
    mov     x0, #0
    adr     x1, input_buf
    add     x1, x1, x9              // buffer + total_len
    mov     x2, #65536
    sub     x2, x2, x9              // remaining space
    mov     x8, #63
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0              // total_len += bytes read
    b       read_loop
read_done:
    // x9 = total_len (0 .. 65536)
    mov     x10, #0                 // cur_offset = 0

main_loop:
    cmp     x10, x9
    bge     finish

    // line_start = cur_offset
    mov     x11, x10                // x11 = line_start
    // find newline or end
    mov     x12, x10                // x12 = iterator
find_nl:
    cmp     x12, x9
    beq     line_end_no_nl
    adr     x13, input_buf
    add     x13, x13, x12
    ldrb    w14, [x13]
    cmp     w14, #10                // '\n'
    beq     line_end_nl
    add     x12, x12, #1
    b       find_nl
line_end_nl:
    // x12 points at newline
    b       line_process_start
line_end_no_nl:
    // no newline, line ends at total_len
    // x12 already equals total_len
line_process_start:
    // x12 = line_end (position of newline or total_len)
    // initialize line evaluation
    mov     x15, #0                 // stack depth = 0
    mov     x20, #0                 // error flag = 0
    mov     x16, x11                // token_ptr = line_start

token_loop:
    // skip separators (space 0x20, tab 0x09)
    cmp     x16, x12
    bge     tokens_done
    adr     x13, input_buf
    add     x13, x13, x16
    ldrb    w14, [x13]
    cmp     w14, #32                // space
    beq     inc_token_ptr
    cmp     w14, #9                 // tab
    beq     inc_token_ptr
    b       token_start
inc_token_ptr:
    add     x16, x16, #1
    b       token_loop

token_start:
    // token_start_offset = x16
    mov     x17, x16                // start
    mov     x18, x16                // end iterator
tok_find_end:
    cmp     x18, x12
    bge     tok_end
    adr     x13, input_buf
    add     x13, x13, x18
    ldrb    w14, [x13]
    cmp     w14, #32                // space
    beq     tok_end
    cmp     w14, #9                 // tab
    beq     tok_end
    add     x18, x18, #1
    b       tok_find_end
tok_end:
    // token_len = x18 - x17
    sub     x19, x18, x17            // token_len

    // load first byte of token
    adr     x13, input_buf
    add     x13, x13, x17
    ldrb    w20, [x13]               // first char

    // check for single-char operator
    cmp     x19, #1
    bne     not_single_char
    // possible operator
    cmp     w20, #'+'
    beq     op_add
    cmp     w20, #'-'
    beq     op_sub
    cmp     w20, #'*'
    beq     op_mul
    cmp     w20, #'/'
    beq     op_div
    cmp     w20, #'%'
    beq     op_mod
    // not an operator -> fall through to literal handling
    b       literal_start

not_single_char:
literal_start:
    // determine if literal (optional leading '-')
    mov     x21, #0                 // is_negative = 0
    cmp     w20, #'-'
    bne     lit_check_digits
    mov     x21, #1                 // negative sign
    // need at least one more character
    cmp     x19, #2
    blt     token_invalid
    // digit_start = x17 + 1
    add     x22, x17, #1
    b       lit_digit_loop_start
lit_check_digits:
    // first char must be digit
    cmp     w20, #'0'
    blt     token_invalid
    cmp     w20, #'9'
    bgt     token_invalid
    // digit_start = x17
    mov     x22, x17
lit_digit_loop_start:
    // count digits and validate, also count length <=18
    mov     x23, #0                 // digit count
lit_digit_loop:
    cmp     x22, x18
    bge     lit_digits_done
    adr     x13, input_buf
    add     x13, x13, x22
    ldrb    w24, [x13]
    cmp     w24, #'0'
    blt     token_invalid
    cmp     w24, #'9'
    bgt     token_invalid
    add     x23, x23, #1
    cmp     x23, #18
    bgt     token_invalid
    add     x22, x22, #1
    b       lit_digit_loop
lit_digits_done:
    // build integer value
    mov     x25, #0                 // accumulator
    // set digit pointer start again
    cmp     x21, #0
    beq     digit_start_is_x17
    add     x26, x17, #1
    b       digit_parse_loop
digit_start_is_x17:
    mov     x26, x17
    // x23 holds digit count
    mov     x27, #0                 // index i
digit_parse_iter:
    cmp     x27, x23
    beq     digit_parse_done
    add     x28, x26, x27
    adr     x13, input_buf
    add     x13, x13, x28
    ldrb    w29, [x13]
    sub     w29, w29, #'0'          // digit value 0-9
    // x25 = x25 * 10 + digit
    mov     x30, x25
    lsl     x30, x30, #3
    add     x30, x30, x25, lsl #1   // *8 + *2 = *10
    add     x30, x30, x29
    mov     x25, x30
    add     x27, x27, #1
    b       digit_parse_iter
digit_parse_done:
    // apply sign if needed
    cmp     x21, #0
    beq     push_literal
    sub     x25, xzr, x25           // negate
push_literal:
    // push x25 onto stack
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    str     x25, [x6]
    add     x15, x15, #1
    // advance to next token
    mov     x16, x18
    b       token_loop

token_invalid:
    mov     x20, #1                 // set error flag
    mov     x16, x12                // skip rest of line
    b       token_loop

// operator implementations
op_add:
    cmp     x15, #2
    blt     line_error
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x9, [x6]                // b
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x10, [x6]               // a
    add     x11, x10, x9            // a+b (wrap)
    str     x11, [x6]
    add     x15, x15, #1
    mov     x16, x18
    b       token_loop

op_sub:
    cmp     x15, #2
    blt     line_error
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x9, [x6]                // b
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x10, [x6]               // a
    sub     x11, x10, x9            // a-b
    str     x11, [x6]
    add     x15, x15, #1
    mov     x16, x18
    b       token_loop

op_mul:
    cmp     x15, #2
    blt     line_error
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x9, [x6]                // b
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x10, [x6]               // a
    mul     x11, x10, x9
    str     x11, [x6]
    add     x15, x15, #1
    mov     x16, x18
    b       token_loop

op_div:
    cmp     x15, #2
    blt     line_error
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x9, [x6]                // b
    cbz     x9, line_error          // divide by zero
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x10, [x6]               // a
    // overflow check: a == INT64_MIN && b == -1
    movz    x30, #0x8000, lsl #48   // x30 = INT64_MIN
    cmp     x10, x30
    bne     div_normal
    cmp     x9, #-1
    beq     line_error
div_normal:
    sdiv    x11, x10, x9
    str     x11, [x6]
    add     x15, x15, #1
    mov     x16, x18
    b       token_loop

op_mod:
    cmp     x15, #2
    blt     line_error
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x9, [x6]                // b
    cbz     x9, line_error          // modulo by zero
    sub     x15, x15, #1
    mov     x5, x15
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x10, [x6]               // a
    // overflow check same as div
    movz    x30, #0x8000, lsl #48
    cmp     x10, x30
    bne     mod_normal
    cmp     x9, #-1
    beq     line_error
mod_normal:
    sdiv    x11, x10, x9            // q = a/b
    mul     x12, x11, x9            // q*b
    sub     x13, x10, x12           // a - q*b
    str     x13, [x6]
    add     x15, x15, #1
    mov     x16, x18
    b       token_loop

line_error:
    mov     x20, #1                 // set error flag
    mov     x16, x12                // skip rest of line
    b       token_loop

tokens_done:
    // after processing all tokens of the line
    cmp     x20, #0
    bne     line_error_output
    cmp     x15, #1
    bne     line_error_output
    // stack has exactly one value at top
    sub     x5, x15, #1
    lsl     x5, x5, #3
    add     x6, x4, x5
    ldr     x0, [x6]                // result in x0

    // convert result to decimal string in out_buf, then write it
    // sign handling
    asr     x7, x0, #63             // sign mask
    mov     x9, #0                  // sign flag
    cbz     x7, pos_abs
    mov     x9, #1                  // negative
    sub     x8, xzr, x0             // abs = -value
    b       abs_ready
pos_abs:
    mov     x8, x0                  // abs = value
abs_ready:
    adr     x10, out_buf
    add     x11, x10, #31           // ptr = end of buffer
    mov     x12, #0                 // length = 0
    cbz     x8, store_zero
conv_loop:
    mov     x14, #10
    udiv    x13, x8, x14            // quotient
    mul     x15, x13, x14
    sub     x15, x8, x15            // remainder
    add     x15, x15, #'0'
    strb    w15, [x11]
    sub     x11, x11, #1
    add     x12, x12, #1
    mov     x8, x13
    cbz     x8, conv_done
    b       conv_loop
conv_done:
    b       output_digits
store_zero:
    add     x15, xzr, #'0'
