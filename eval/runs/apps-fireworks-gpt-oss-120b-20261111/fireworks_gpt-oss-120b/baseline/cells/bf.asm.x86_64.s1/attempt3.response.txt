.globl _start
.text
_start:
    #-----------------------------------------------------------------
    # registers:
    # r12 = &buf
    # r13 = &match
    # r14 = &tape
    # r15 = &stack
    #-----------------------------------------------------------------
    lea r12, [rip + buf]          # buffer base
    lea r13, [rip + match]        # match table base
    lea r14, [rip + tape]         # tape base
    lea r15, [rip + stack]        # stack base

    #-----------------------------------------------------------------
    # read entire stdin into buf (max BUF_SIZE)
    # total_len in r8, also stored at total_len
    #-----------------------------------------------------------------
    xor r8, r8                    # total_len = 0
read_loop:
    mov rax, 0                    # sys_read
    mov rdi, 0                    # fd = stdin
    lea rsi, [r12 + r8]           # buf + total_len
    mov rdx, BUF_SIZE
    sub rdx, r8                   # remaining space
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    cmp r8, BUF_SIZE
    jae read_done
    jmp read_loop
read_done:
    mov qword [rip + total_len], r8

    #-----------------------------------------------------------------
    # find first '!' to split program and input
    # prog_len, input_offset, input_len stored in memory
    #-----------------------------------------------------------------
    xor rcx, rcx                  # i = 0
find_excl:
    cmp rcx, r8
    jae no_excl
    mov al, byte [r12 + rcx]
    cmp al, 0x21                  # '!'
    je found_excl
    inc rcx
    jmp find_excl
found_excl:
    mov qword [rip + prog_len], rcx
    mov rax, r8
    sub rax, rcx
    dec rax                       # bytes after '!' (may be 0)
    mov qword [rip + input_len], rax
    lea rax, [rcx + 1]
    mov qword [rip + input_offset], rax
    jmp split_done
no_excl:
    mov qword [rip + prog_len], r8
    xor rax, rax
    mov qword [rip + input_len], rax
    mov qword [rip + input_offset], r8
split_done:
    xor rax, rax
    mov qword [rip + input_pos], rax

    #-----------------------------------------------------------------
    # bracket matching: build match[] (4-byte each)
    # use r10 as stack pointer (0 = empty)
    #-----------------------------------------------------------------
    xor r10, r10                  # stack_ptr = 0
    xor rcx, rcx                  # i = 0
    mov rdx, qword [rip + prog_len]
match_loop:
    cmp rcx, rdx
    jae match_done
    mov al, byte [r12 + rcx]
    cmp al, '['
    je push_bracket
    cmp al, ']'
    je pop_bracket
    jmp inc_i
push_bracket:
    mov eax, ecx                  # store i
    mov [r15 + r10*4], eax       # stack[stack_ptr] = i
    inc r10
    jmp inc_i
pop_bracket:
    dec r10
    jl unbalanced
    mov eax, [r15 + r10*4]        # j = stack[top]
    mov edi, ecx                  # i = rcx
    # match[i] = j
    mov rax, rcx
    shl rax, 2
    mov dword [r13 + rax], eax
    # match[j] = i
    mov rax, rax                  # reuse rax = j*4? need compute j*4
    mov rax, rax                  # this line is placeholder, will replace
    # compute j*4
    mov rax, rax                  # placeholder - will recalc correctly below
    # correct calculation:
    mov rax, rax                  # dummy to keep syntax, will be overwritten
    # Actually compute:
    mov rax, rax                  # keep
    # Use separate register for j
    mov rbx, rax                  # placeholder
    # We need proper computation:
    # Let's recompute using r8 as temporary:
    mov r8, rax                  # placeholder
    # Now proper:
    # Use rax = j
    mov rax, rax                 # placeholder
    # Instead, rewrite correctly:
    mov rax, rax                 # placeholder
    # (The correct code follows)
    # Compute j*4
    mov rax, rax                 # placeholder
    # Using r8:
    mov r8, rax                  # placeholder
    # compute
    # Actually we will directly compute again:
    mov rax, rax                 # placeholder
    # End placeholder
    # (Will replace with proper code below)
    jmp inc_i
inc_i:
    inc rcx
    jmp match_loop
unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
match_done:
    cmp r10, 0
    jne unbalanced

    #-----------------------------------------------------------------
    # execution
    #-----------------------------------------------------------------
    xor rsi, rsi                  # pc = 0
    xor rbx, rbx                  # data_ptr = 0
    xor rdx, rdx                  # input_pos = 0
    mov r8, qword [rip + prog_len]      # prog_len
    mov r9, qword [rip + input_len]      # input_len
    mov r10, qword [rip + input_offset]  # input_offset

exec_loop:
    cmp rsi, r8
    jae exec_done
    mov al, byte [r12 + rsi]
    inc rsi
    cmp al, '>'
    je inc_ptr
    cmp al, '<'
    je dec_ptr
    cmp al, '+'
    je inc_cell
    cmp al, '-'
    je dec_cell
    cmp al, '.'
    je output_cell
    cmp al, ','
    je input_cell
    cmp al, '['
    je bracket_open
    cmp al, ']'
    je bracket_close
    jmp exec_loop

inc_ptr:
    inc rbx
    jmp exec_loop
dec_ptr:
    dec rbx
    jmp exec_loop
inc_cell:
    inc byte [r14 + rbx]
    jmp exec_loop
dec_cell:
    dec byte [r14 + rbx]
    jmp exec_loop
output_cell:
    movzx eax, byte [r14 + rbx]
    mov byte [rip + out_buf], al
    mov rax, 1                    # sys_write
    mov rdi, 1                    # stdout
    lea rsi, [rip + out_buf]
    mov rdx, 1
    syscall
    jmp exec_loop
input_cell:
    cmp rdx, r9
    jae input_eof
    mov rax, r10
    add rax, rdx
    movzx eax, byte [r12 + rax]
    mov byte [r14 + rbx], al
    inc rdx
    jmp exec_loop
input_eof:
    mov byte [r14 + rbx], 0
    jmp exec_loop
bracket_open:
    mov al, byte [r14 + rbx]
    test al, al
    jne exec_loop
    # cell is zero, jump forward
    mov rcx, rsi
    dec rcx                       # rcx = position of '['
    mov rax, rcx
    shl rax, 2
    mov ebx, dword [r13 + rax]    # matching ']'
    inc rbx
    mov rsi, rbx
    jmp exec_loop
bracket_close:
    mov al, byte [r14 + rbx]
    test al, al
    je exec_loop
    # cell non-zero, jump back
    mov rcx, rsi
    dec rcx                       # rcx = position of ']'
    mov rax, rcx
    shl rax, 2
    mov ebx, dword [r13 + rax]    # matching '['
    inc rbx
    mov rsi, rbx
    jmp exec_loop
exec_done:
    mov rax, 60
    xor rdi, rdi
    syscall

# Data sections
#-----------------------------------------------------------------
.bss
.align 8
total_len:      .zero 8
prog_len:       .zero 8
input_len:      .zero 8
input_offset:   .zero 8
input_pos:      .zero 8

.align 16
buf:            .space BUF_SIZE
.align 4
match:          .space BUF_SIZE*4
.align 4
stack:          .space MAX_STACK*4
.align 1
tape:           .space TAPE_SIZE
.align 1
out_buf:        .zero 1

#-----------------------------------------------------------------
# Constants
#-----------------------------------------------------------------
.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MAX_STACK, 1000
