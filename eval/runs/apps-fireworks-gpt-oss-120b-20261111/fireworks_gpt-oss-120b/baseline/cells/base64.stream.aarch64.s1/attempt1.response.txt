	.globl	_start
	.text
	.align	2

// constants
	.equ	READ_SYSCALL,	63
	.equ	WRITE_SYSCALL,	64
	.equ	EXITGROUP_SYSCALL, 94
	.equ	STDIN_FD,	0
	.equ	STDOUT_FD,	1
	.equ	INPUT_BUF_SIZE,	49152
	.equ	OUTPUT_BUF_MAX, 70000
	.equ	LINE_MAX,	76
	.equ	NL_CHAR,	0x0A
	.equ	PAD_CHAR,	0x3D   // '='

_start:
	// x19 = input_len (initialized to 0 by .bss)
	// x20 = input buffer pointer
	// x21 = output buffer pointer
	// x22 = line_len counter
	// x23 = output_len counter

	// load address of input buffer into x20
	adrp	x20, input_buf
	add	x20, x20, :lo12:input_buf

	// set input_len to 0 (already zeroed)
	// read loop
read_loop:
	// syscall read(stdin, x20, INPUT_BUF_SIZE)
	mov	x0, #STDIN_FD
	mov	x1, x20
	mov	x2, #INPUT_BUF_SIZE
	mov	x8, #READ_SYSCALL
	svc	#0
	cmp	x0, #0
	b.eq	read_done			// EOF
	// add returned bytes to input_len (x19)
	add	x19, x19, x0
	// advance input buffer pointer
	add	x20, x20, x0
	b	read_loop

read_done:
	// if input_len == 0, exit silently
	cmp	x19, #0
	b.eq	exit_success

	// reset pointers for encoding
	// x20 = start of input buffer
	adrp	x20, input_buf
	add	x20, x20, :lo12:input_buf

	// x21 = start of output buffer
	adrp	x21, output_buf
	add	x21, x21, :lo12:output_buf

	// x22 = line_len = 0
	mov	x22, #0
	// x23 = output_len = 0
	mov	x23, #0

	// x24 = base64 table pointer
	adrp	x24, b64_table
	add	x24, x24, :lo12:b64_table

	// x25 = remaining bytes = input_len
	mov	x25, x19

	// main groups of 3 bytes
encode_loop:
	cmp	x25, #3
	blt	encode_remainder

	// load three bytes
	ldrb	w0, [x20], #1        // b0
	ldrb	w1, [x20], #1        // b1
	ldrb	w2, [x20], #1        // b2

	// compute sextets
	// sextet0 = b0 >> 2
	ubfx	w3, w0, #2, #6       // (b0 & 0xFC) >> 2
	// sextet1 = ((b0 & 0x3) << 4) | (b1 >> 4)
	ubfx	w4, w0, #0, #2       // lower 2 bits of b0
	lsl	w4, w4, #4
	ubfx	w5, w1, #4, #4       // top 4 bits of b1
	orr	w4, w4, w5

	// sextet2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	ubfx	w5, w1, #0, #4       // lower 4 bits of b1
	lsl	w5, w5, #2
	ubfx	w6, w2, #6, #2       // top 2 bits of b2
	orr	w5, w5, w6

	// sextet3 = b2 & 0x3F
	ands	w6, w2, #0x3F

	// output the four characters
	// sextet0
	ldr	b7, [x24, w3]        // load byte from table
