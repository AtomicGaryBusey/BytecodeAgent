	.globl	_start
	.text
	.align	2

	// system call numbers
	.equ	READ_SYSCALL,	63
	.equ	WRITE_SYSCALL,	64
	.equ	EXIT_SYSCALL,	93

	// constants
	.equ	STDIN_FD,	0
	.equ	STDOUT_FD,	1
	.equ	INPUT_BUF_SIZE,	49152
	.equ	OUTPUT_BUF_SIZE,	70000
	.equ	LINE_MAX,	76
	.equ	NL_CHAR,	0x0A
	.equ	PAD_CHAR,	0x3D	// '='

_start:
	// x19 = current input pointer (write position)
	adrp	x19, input_buf
	add	x19, x19, :lo12:input_buf

	// x20 = start of input buffer (for later reset)
	adrp	x20, input_buf
	add	x20, x20, :lo12:input_buf

	// x21 = total input length
	mov	x21, #0

// ---------- read loop ----------
read_loop:
	mov	x0, #STDIN_FD
	mov	x1, x19
	mov	x2, #INPUT_BUF_SIZE
	mov	x8, #READ_SYSCALL
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x21, x21, x0
	add	x19, x19, x0
	b	read_loop

read_done:
	cmp	x21, #0
	beq	exit_success

	// x22 = src pointer for encoding (reset to start)
	adrp	x22, input_buf
	add	x22, x22, :lo12:input_buf

	// x23 = dst pointer for output
	adrp	x23, output_buf
	add	x23, x23, :lo12:output_buf

	// x24 = current line length
	mov	x24, #0

	// x25 = base64 table pointer
	adrp	x25, b64_table
	add	x25, x25, :lo12:b64_table

	// x26 = remaining bytes to encode
	mov	x26, x21

// ---------- main encode loop (full groups of 3) ----------
encode_loop:
	cmp	x26, #3
	blt	encode_remainder

	// load three bytes
	ldrb	w0, [x22], #1		// b0
	ldrb	w1, [x22], #1		// b1
	ldrb	w2, [x22], #1		// b2

	// sextet0 = b0 >> 2
	ubfx	w3, w0, #2, #6

	// sextet1 = ((b0 & 0x3) << 4) | (b1 >> 4)
	and	w4, w0, #0x3
	lsl	w4, w4, #4
	ubfx	w5, w1, #4, #4
	orr	w4, w4, w5

	// sextet2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	and	w5, w1, #0xF
	lsl	w5, w5, #2
	ubfx	w6, w2, #6, #2
	orr	w5, w5, w6

	// sextet3 = b2 & 0x3F
	and	w6, w2, #0x3F

	// output sextet0
	uxtw	x0, w3
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	1f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
1:

	// output sextet1
	uxtw	x0, w4
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	2f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
2:

	// output sextet2
	uxtw	x0, w5
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	3f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
3:

	// output sextet3
	uxtw	x0, w6
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	4f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
4:

	sub	x26, x26, #3
	b	encode_loop

// ---------- remainder handling ----------
encode_remainder:
	cmp	x26, #0
	beq	encoding_done

	// load remaining bytes
	ldrb	w0, [x22], #1		// first byte always present
	mov	w1, #0
	cmp	x26, #2
	blt	rem_one_byte
	ldrb	w1, [x22], #1		// second byte
rem_one_byte:

	// sextet0
	ubfx	w3, w0, #2, #6

	// sextet1 base
	and	w4, w0, #0x3
	lsl	w4, w4, #4

	cmp	x26, #2
	beq	rem_two_bytes
	// one byte case: no addition to sextet1
	b	rem_one_byte_done
rem_two_bytes:
	ubfx	w5, w1, #4, #4
	orr	w4, w4, w5
rem_one_byte_done:

	// output sextet0
	uxtw	x0, w3
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	5f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
5:

	// output sextet1
	uxtw	x0, w4
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	6f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
6:

	cmp	x26, #2
	beq	rem_two_output

	// one byte remainder: output two '=' pads
pad_one_eq:
	mov	w7, #PAD_CHAR
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	7f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
7:
	mov	w7, #PAD_CHAR
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	encoding_done
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
	b	encoding_done

rem_two_output:
	// compute sextet2 = ((b1 & 0xF) << 2)
	and	w5, w1, #0xF
	lsl	w5, w5, #2

	// output sextet2
	uxtw	x0, w5
	ldrb	w7, [x25, x0]
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	8f
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
8:

	// pad with one '='
	mov	w7, #PAD_CHAR
	strb	w7, [x23], #1
	add	x24, x24, #1
	cmp	x24, #LINE_MAX
	bne	encoding_done
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	mov	x24, #0
	b	encoding_done

encoding_done:
	// ensure final newline if needed
	cmp	x24, #0
	beq	final_write
	mov	w7, #NL_CHAR
	strb	w7, [x23], #1
	// fall through

final_write:
	// x0 = output buffer address
	adrp	x0, output_buf
	add	x0, x0, :lo12:output_buf
	// x1 = length = dst - start
	sub	x1, x23, x0
	// write syscall
	mov	x2, x1
	mov	x0, #STDOUT_FD
	mov	x8, #WRITE_SYSCALL
	svc	#0

	// exit
exit_success:
	mov	x0, #0
	mov	x8, #EXIT_SYSCALL
	svc	#0

	// -----------------------------------------------------------------
	.bss
	.align	4
input_buf:
	.space	INPUT_BUF_SIZE
output_buf:
	.space	OUTPUT_BUF_SIZE

	.section	.rodata
	.align	4
b64_table:
	.ascii	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
