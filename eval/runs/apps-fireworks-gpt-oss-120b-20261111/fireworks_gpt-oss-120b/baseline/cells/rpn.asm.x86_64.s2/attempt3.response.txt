.section .data
err_msg:
    .ascii "error\n"
err_len = . - err_msg
nl:
    .byte 0x0A

.section .bss
    .align 8
input_buf:
    .space 65536
stack_vals:
    .space 8192
outbuf:
    .space 32

.section .text
.globl _start
_start:
    # read whole input (max 65536 bytes) into input_buf
    xor r12, r12                 # total bytes read = 0
read_loop:
    mov rdi, 0                   # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, r12                 # buf + offset
    mov rdx, 65536
    sub rdx, r12                 # remaining space
    mov rax, 0                   # sys_read
    syscall
    test rax, rax
    jz read_done
    add r12, rax                 # total += bytes read
    cmp r12, 65536
    jl read_loop
read_done:
    lea r14, [rip + input_buf]   # r14 = input start
    lea r15, [r14 + r12]         # r15 = input end (one past)

    # line processing
    mov rbx, r14                 # rbx = ptr to current line start
line_loop:
    cmp rbx, r15
    jae exit_success

    # find line end (newline or input end)
    mov r8, rbx                  # scanner
scan_newline:
    cmp r8, r15
    jae no_newline
    mov al, [r8]
    cmp al, 0x0A                 # '\n'
    je found_newline
    inc r8
    jmp scan_newline
found_newline:
    mov r9d, 1                   # newline present flag
    jmp after_scan
no_newline:
    mov r9d, 0                   # no newline
after_scan:
    mov r10, r8                  # r10 = line_end (points at newline or end)

    # evaluate line [rbx, r10)
    xor r13d, r13d               # stack depth = 0
    xor r11d, r11d               # error flag = 0

token_loop:
    # skip separators (space or tab)
skip_sep:
    cmp rbx, r10
    jae token_done
    mov al, [rbx]
    cmp al, 0x20                 # space
    je inc_ptr_sep
    cmp al, 0x09                 # tab
    je inc_ptr_sep
    jmp after_skip
inc_ptr_sep:
    inc rbx
    jmp skip_sep
after_skip:
    cmp rbx, r10
    jae token_done

    # token start at rsi = rbx
    mov rsi, rbx
    # find token end in rbx
scan_token:
    cmp rbx, r10
    jae token_end
    mov al, [rbx]
    cmp al, 0x20
    je token_end
    cmp al, 0x09
    je token_end
    inc rbx
    jmp scan_token
token_end:
    mov rdi, rbx                 # rdi = token_end
    mov rcx, rdi
    sub rcx, rsi                 # rcx = token_len

    # check for operator (single-char token)
    cmp rcx, 1
    jne not_operator
    mov al, [rsi]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    # not a known operator -> fall through to literal handling
not_operator:
    # parse integer literal
    xor rax, rax                 # accumulator
    xor r8d, r8d                 # digit count
    mov bl, [rsi]                # first byte
    cmp bl, '-'
    jne lit_positive
    inc rsi                      # skip '-'
    mov r9b, 1                   # sign flag = negative
    jmp lit_digits
lit_positive:
    xor r9b, r9b                 # sign flag = 0
lit_digits:
    cmp rsi, rdi
    jae lit_no_digit
    mov al, [rsi]
    cmp al, '0'
    jb lit_invalid
    cmp al, '9'
    ja lit_invalid
    inc r8d
    cmp r8d, 19
    jae lit_invalid
    sub al, '0'
    movzx rbx, al
    imul rax, rax, 10
    add rax, rbx
    inc rsi
    jmp lit_digits
lit_no_digit:
    cmp r8d, 0
    je lit_invalid
    test r9b, r9b
    je lit_push
    neg rax                      # two's complement neg (works for INT64_MIN)
    jmp lit_push
lit_invalid:
    mov r11d, 1                  # set error flag
    jmp output_error

lit_push:
    # push rax onto stack_vals
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov [rbx], rax
    inc r13d
    jmp token_loop

# operator implementations
op_add:
    cmp r13d, 2
    jb set_error
    dec r13d
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov rbx, [rbx]               # b
    dec r13d
    lea rax, [rip + stack_vals]
    mov rcx, r13
    shl rcx, 3
    add rax, rcx
    mov rax, [rax]               # a
    add rax, rbx
    jmp push_result
op_sub:
    cmp r13d, 2
    jb set_error
    dec r13d
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov rbx, [rbx]               # b
    dec r13d
    lea rax, [rip + stack_vals]
    mov rcx, r13
    shl rcx, 3
    add rax, rcx
    mov rax, [rax]               # a
    sub rax, rbx
    jmp push_result
op_mul:
    cmp r13d, 2
    jb set_error
    dec r13d
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov rbx, [rbx]               # b
    dec r13d
    lea rax, [rip + stack_vals]
    mov rcx, r13
    shl rcx, 3
    add rax, rcx
    mov rax, [rax]               # a
    imul rax, rax, rbx
    jmp push_result
op_div:
    cmp r13d, 2
    jb set_error
    dec r13d
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov rcx, [rbx]               # divisor b
    dec r13d
    lea rax, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rax, rdx
    mov rax, [rax]               # dividend a
    test rcx, rcx
    jz set_error
    mov r8, 0x8000000000000000
    cmp rax, r8
    jne div_ok
    cmp rcx, -1
    je set_error
div_ok:
    cqo
    idiv rcx                     # quotient -> rax
    jmp push_result
op_mod:
    cmp r13d, 2
    jb set_error
    dec r13d
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov rcx, [rbx]               # divisor b
    dec r13d
    lea rax, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rax, rdx
    mov rax, [rax]               # dividend a
    test rcx, rcx
    jz set_error
    mov r8, 0x8000000000000000
    cmp rax, r8
    jne mod_ok
    cmp rcx, -1
    je set_error
mod_ok:
    cqo
    idiv rcx                     # remainder -> rdx
    mov rax, rdx
    jmp push_result

set_error:
    mov r11d, 1
    jmp output_error

push_result:
    lea rbx, [rip + stack_vals]
    mov rdx, r13
    shl rdx, 3
    add rbx, rdx
    mov [rbx], rax
    inc r13d
    jmp token_loop

token_done:
    # line finished
    test r11d, r11d
    jne output_error

    cmp r13d, 1
    jne output_error

    # fetch result
    lea rax, [rip + stack_vals]
    mov rcx, r13
    dec rcx
    shl rcx, 3
    add rax, rcx
    mov rax, [rax]               # result

    # convert to decimal string
    lea rdi, [rip + outbuf + 31] # point to last byte of buffer
    xor r8d, r8d                 # sign flag = 0
    mov rbx, rax
    test rbx, rbx
    jge conv_start
    # negative number
    neg rbx
    mov r8d, 1
conv_start:
    xor rcx, rcx                 # digit count
conv_loop:
    mov rdx, 0
    mov rsi, 10
    div rsi                       # rax = quotient, rdx = remainder
    add dl, '0'
    dec rdi
    mov [rdi], dl
    inc rcx
    mov rbx, rax
    test rax, rax
    jne conv_loop
    cmp r8d, 0
    je write_number
    dec rdi
    mov byte [rdi], '-'
    inc rcx
write_number:
    mov rsi, rdi
    mov rdx, rcx
    mov rdi, 1                   # stdout
    mov rax, 1                   # sys_write
    syscall

    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall

    jmp after_line

output_error:
    # write "error\n"
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, err_len
    syscall
    jmp after_line

after_line:
    # advance to next line
    cmp r9d, 1
    jne no_newline_next
    inc r10                      # skip newline character
no_newline_next:
    mov rbx, r10
    jmp line_loop

exit_success:
    mov rax, 60                  # sys_exit
    xor rdi, rdi
    syscall
