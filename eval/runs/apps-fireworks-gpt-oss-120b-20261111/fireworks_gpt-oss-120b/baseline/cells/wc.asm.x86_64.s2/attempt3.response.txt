.globl _start
.equ BUF_SIZE, 65536

.section .bss
.align 16
buf:    .zero BUF_SIZE
outbuf: .zero 32

.section .text
_start:
    xor     rbx, rbx            # total bytes
    xor     r12, r12            # line count
    xor     r13, r13            # word count
    xor     r14, r14            # in_word flag (0)

read_loop:
    mov     rax, 0              # sys_read
    mov     rdi, 0              # fd 0
    lea     rsi, [rip + buf]
    mov     rdx, BUF_SIZE
    syscall
    test    rax, rax
    jz      finish_read
    mov     r8, rax             # bytes read this chunk
    add     rbx, rax            # total bytes

    lea     rsi, [rip + buf]    # pointer to start of chunk
proc_loop:
    test    r8, r8
    jz      after_chunk
    mov     al, [rsi]

    cmp     al, 0x0A
    jne     not_nl
    inc     r12                 # line count
not_nl:
    # whitespace check
    cmp     al, 0x20
    je      is_ws
    cmp     al, 0x09
    je      is_ws
    cmp     al, 0x0A
    je      is_ws
    cmp     al, 0x0B
    je      is_ws
    cmp     al, 0x0C
    je      is_ws
    cmp     al, 0x0D
    je      is_ws
    jmp     not_ws
is_ws:
    test    r14, r14
    jz      ws_no_word
    inc     r13                 # word count
    xor     r14, r14
ws_no_word:
    jmp     next_char
not_ws:
    mov     r14, 1              # set in_word flag
next_char:
    inc     rsi
    dec     r8
    jmp     proc_loop
after_chunk:
    jmp     read_loop

finish_read:
    test    r14, r14
    jz      no_final_word
    inc     r13                 # last word
no_final_word:
    # build output string
    lea     rdi, [rip + outbuf]

    mov     rax, r12            # lines
    call    u64_to_dec
    mov     byte ptr [rdi], 0x20
    inc     rdi

    mov     rax, r13            # words
    call    u64_to_dec
    mov     byte ptr [rdi], 0x20
    inc     rdi

    mov     rax, rbx            # bytes
    call    u64_to_dec
    mov     byte ptr [rdi], 0x0A
    inc     rdi

    # write output
    mov     rdx, rdi
    lea     rsi, [rip + outbuf]
    sub     rdx, rsi            # length
    mov     rax, 1              # sys_write
    mov     rdi, 1              # fd 1
    syscall

    # exit
    mov     rax, 60             # sys_exit
    xor     rdi, rdi
    syscall

# -------------------------------------------------
# rax = value, rdi = buffer pointer -> writes decimal text, updates rdi, returns
u64_to_dec:
    push    rbx
    push    rcx
    push    rdx
    push    r8
    push    r9

    cmp     rax, 0
    jne     .convert
    mov     byte ptr [rdi], '0'
    inc     rdi
    jmp     .done

.convert:
    sub     rsp, 32                 # temporary space
    lea     r8, [rsp + 31]          # end of temp buffer
    xor     rcx, rcx                # digit count
.dig_loop:
    xor     rdx, rdx
    mov     r9, 10
    div     r9                      # rax = rax/10, rdx = remainder
    add     dl, '0'
    mov     byte ptr [r8], dl
    dec     r8
    inc     rcx
    test    rax, rax
    jne     .dig_loop
    lea     r9, [r8 + 1]            # start of digits
.copy_loop:
    mov     al, [r9]
    mov     [rdi], al
    inc     rdi
    inc     r9
    dec     rcx
    jne     .copy_loop
    add     rsp, 32

.done:
    pop     r9
    pop     r8
    pop     rdx
    pop     rcx
    pop     rbx
    ret
