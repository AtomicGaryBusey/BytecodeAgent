.globl _start

.equ BUF_SIZE, 65536
.equ CHUNK, 4096

.section .bss
.align 16
buf:        .zero BUF_SIZE
outbuf:     .zero 32

.section .text
_start:
    xor     rbx, rbx        # total bytes
    xor     r10, r10       # line count
    xor     r11, r11       # word count
    xor     r15, r15       # offset in buffer
    xor     bl, bl         # in_word flag (use low byte of rbx, but rbx is total bytes; we'll use a separate register)
    mov     bl, 0

read_loop:
    mov     rax, 0          # sys_read
    mov     rdi, 0          # fd 0
    lea     rsi, [rip + buf]
    add     rsi, r15        # buffer + offset
    mov     rdx, CHUNK
    syscall
    test    rax, rax
    jz      finish_read

    mov     r12, rax        # bytes read this chunk
    add     rbx, rax        # total bytes

    lea     rsi, [rip + buf]
    add     rsi, r15        # start pointer for processing
    mov     rcx, r12        # loop counter

proc_loop:
    mov     al, [rsi]

    cmp     al, 0x0a
    jne     not_nl
    inc     r10             # line count
not_nl:

    # whitespace check
    cmp     al, 0x20
    je      is_ws
    cmp     al, 0x09
    je      is_ws
    cmp     al, 0x0a
    je      is_ws
    cmp     al, 0x0b
    je      is_ws
    cmp     al, 0x0c
    je      is_ws
    cmp     al, 0x0d
    je      is_ws
    jmp     not_ws

is_ws:
    cmp     bl, 1
    jne     ws_no_word
    inc     r11             # word count
    mov     bl, 0
ws_no_word:
    jmp     after_ws

not_ws:
    mov     bl, 1
after_ws:
    inc     rsi
    dec     rcx
    jnz     proc_loop

    add     r15, r12        # advance offset
    jmp     read_loop

finish_read:
    cmp     bl, 1
    jne     no_final_word
    inc     r11
no_final_word:

    # build output string
    lea     rdi, [rip + outbuf]   # output pointer

    mov     rax, r10              # lines
    call    u64_to_dec

    mov     byte ptr [rdi], 0x20
    inc     rdi

    mov     rax, r11              # words
    call    u64_to_dec

    mov     byte ptr [rdi], 0x20
    inc     rdi

    mov     rax, rbx              # bytes
    call    u64_to_dec

    mov     byte ptr [rdi], 0x0a
    inc     rdi

    # write output
    mov     rdx, rdi
    lea     rsi, [rip + outbuf]
    sub     rdx, rsi              # length
    mov     rax, 1                # sys_write
    mov     rdi, 1                # fd 1
    syscall

    # exit
    mov     rax, 60               # sys_exit
    xor     rdi, rdi              # status 0
    syscall

# -------------------------------------------------
# rax = value, rdi = buffer ptr -> writes decimal text, updates rdi, returns
u64_to_dec:
    push    rbx
    push    rcx
    push    rdx
    push    r8
    push    r9

    cmp     rax, 0
    jne     .convert
    mov     byte ptr [rdi], '0'
    inc     rdi
    jmp     .done

.convert:
    sub     rsp, 32              # temporary space
    lea     rbx, [rsp + 31]      # ptr to end of temp buffer
    xor     rcx, rcx             # digit count

.dig_loop:
    xor     rdx, rdx
    mov     r8, 10
    div     r8                   # rax = rax/10, rdx = rax%10
    add     dl, '0'
    mov     byte ptr [rbx], dl
    dec     rbx
    inc     rcx
    test    rax, rax
    jne     .dig_loop

    lea     rsi, [rbx + 1]       # start of digits
.copy_loop:
    mov     al, [rsi]
    mov     [rdi], al
    inc     rsi
    inc     rdi
    dec     rcx
    jne     .copy_loop

    add     rsp, 32

.done:
    pop     r9
    pop     r8
    pop     rdx
    pop     rcx
    pop     rbx
    ret
