.globl _start

.section .bss
    .align 8
input_buf:
    .zero 65536
input_len:
    .zero 8
pos:
    .zero 8
error_offset:
    .zero 8
dec_buf:
    .zero 6

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
err_prefix:
    .ascii "error at "
err_prefix_len = . - err_prefix
newline:
    .ascii "\n"
newline_len = . - newline

.text
_start:
    # read whole stdin
    xor rdi, rdi                     # fd = 0
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    mov rax, 0                       # sys_read
read_loop:
    syscall
    test rax, rax
    jz read_done
    mov rbx, rax
    mov rax, [rip + input_len]
    add rax, rbx
    mov [rip + input_len], rax
    jmp read_loop
read_done:
    xor rax, rax
    mov [rip + pos], rax

    # parse document
    call parse_ws
    call parse_value
    call parse_ws
    # check for extra non‑ws
    mov rax, [rip + pos]
    mov rbx, [rip + input_len]
    cmp rax, rbx
    je ok_path
    # extra data -> error at current position
    mov [rip + error_offset], rax
    jmp error

ok_path:
    # write "ok\n"
    mov rax, 1                       # sys_write
    mov rdi, 1                       # stdout
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    syscall
    # exit 0
    mov rax, 60                      # sys_exit
    xor rdi, rdi
    syscall

# ---------- parsing helpers ----------
parse_ws:
    lea rdx, [rip + input_buf]
    mov rcx, [rip + input_len]
    mov r8, [rip + pos]
ws_loop:
    cmp r8, rcx
    jae ws_done
    mov al, byte [rdx + r8]
