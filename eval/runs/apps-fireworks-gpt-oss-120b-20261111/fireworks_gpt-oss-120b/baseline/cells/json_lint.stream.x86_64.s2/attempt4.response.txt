.globl _start

.section .bss
    .align 8
input_buf:
    .zero 65536
input_len:
    .zero 8
err_num_buf:
    .zero 24

.section .rodata
msg_ok:
    .asciz "ok\n"
msg_err_prefix:
    .asciz "error at "
msg_newline:
    .asciz "\n"

.text
_start:
    # read whole stdin into input_buf
    xor r8, r8                     # total length = 0
    mov r9, 65536                  # max size
read_loop:
    mov rax, 0                     # sys_read
    mov rdi, 0                     # fd 0
    mov rdx, r9
    sub rdx, r8                    # remaining space
    je read_done
    lea rsi, [rel input_buf + r8]
