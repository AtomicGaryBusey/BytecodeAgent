	.global _start
	.text
	.align 2
_start:
	// initialize buf_len = 0
	adrp	x0, buf_len
	add	x0, x0, :lo12:buf_len
	mov	x1, #0
	str	x1, [x0]

	// load address of newline byte into x9
	adr	x9, nl_char

read_loop:
	// load current length
	adrp	x2, buf_len
	add	x2, x2, :lo12:buf_len
	ldr	x3, [x2]               // x3 = len

	// compute dest = input_buf + len
	adrp	x4, input_buf
	add	x4, x4, :lo12:input_buf
	add	x5, x4, x3             // x5 = dest

	// remaining = 65536 - len
	mov	x6, #65536
	sub	x6, x6, x3

	// syscall read(fd=0, buf=dest, count=remaining)
	mov	x0, #0
	mov	x1, x5
	mov	x2, x6
	mov	x8, #63
	svc	#0
	cbnz	x0, read_got_data
	// EOF
	b	read_done

read_got_data:
	// len += returned count
	adrp	x7, buf_len
	add	x7, x7, :lo12:buf_len
	ldr	x8, [x7]
	add	x8, x8, x0
	str	x8, [x7]
	b	read_loop

read_done:
	// load total length into x20
	adrp	x20, buf_len
	add	x20, x20, :lo12:buf_len
	ldr	x20, [x20]             // total_len

	// base address of input_buf into x21
	adrp	x21, input_buf
	add	x21, x21, :lo12:input_buf

	// i = 0
	mov	x19, #0                 // current offset

process_loop:
	cmp	x19, x20
	bge	exit_prog

	// line_start = i
	mov	x22, x19                // line_start

	// scan index = i
	mov	x23, x19                // scan index

scan_loop:
	cmp	x23, x20
	bge	line_end_no_nl
	add	x24, x21, x23
	ldrb	w25, [x24]
	cmp	w25, #0x0a
	beq	line_end_nl
	add	x23, x23, #1
	b	scan_loop

line_end_nl:
	// newline at x23, line_end = x23 (exclude newline)
	sub	x26, x23, x22           // len = line_end - line_start
	cbnz	x26, rev_nonempty
	// empty line: just write newline
	b	empty_line

rev_nonempty:
	// idx = line_start + len - 1
	add	x27, x22, x26
	sub	x27, x27, #1            // idx

rev_loop:
	add	x28, x21, x27           // address of byte
	// write(fd=1, buf=addr, count=1)
	mov	x0, #1
	mov	x1, x28
	mov	x2, #1
	mov	x8, #64
	svc	#0
	subs	x27, x27, #1
	cmp	x27, x22
	bge	rev_loop

	// write newline after reversed line
write_newline_common:
	mov	x0, #1
	mov	x1, x9
	mov	x2, #1
	mov	x8, #64
	svc	#0
	// advance i past newline
	add	x19, x23, #1
	b	process_loop

empty_line:
	// write newline for empty line
	b	write_newline_common

line_end_no_nl:
	// end of input without newline
	sub	x26, x20, x22           // len = total_len - line_start
	cbnz	x26, rev_no_nl
	// no data left, nothing to output, exit
	b	exit_prog

rev_no_nl:
	// idx = line_start + len - 1
	add	x27, x22, x26
	sub	x27, x27, #1

rev_loop_no_nl:
	add	x28, x21, x27
	mov	x0, #1
	mov	x1, x28
	mov	x2, #1
	mov	x8, #64
	svc	#0
	subs	x27, x27, #1
	cmp	x27, x22
	bge	rev_loop_no_nl

	// write final newline
	b	write_newline_common

exit_prog:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.section .rodata
	.align 1
nl_char:
	.byte	0x0a

	.bss
	.align 8
input_buf:
	.space	65536
buf_len:
	.space	8
