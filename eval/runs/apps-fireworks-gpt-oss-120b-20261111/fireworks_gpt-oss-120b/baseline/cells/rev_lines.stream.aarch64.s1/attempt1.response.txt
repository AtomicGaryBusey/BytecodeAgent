	.globl _start
	.text
	.align 2
_start:
	// read loop: fill input_buf, keep length in x9 (len)
	// initialize len = 0
	adrp	x9, buf_len
	add	x9, x9, :lo12:buf_len
	str	xzr, [x9]               // buf_len = 0

read_loop:
	// load current length
	adrp	x10, buf_len
	add	x10, x10, :lo12:buf_len
	ldr	x11, [x10]              // x11 = len

	// compute dest = input_buf + len
	adrp	x12, input_buf
	add	x12, x12, :lo12:input_buf
	add	x12, x12, x11           // x12 = dest address

	// remaining space = 65536 - len
	mov	x2, #65536
	sub	x2, x2, x11

	// syscall read(fd=0, buf=dest, count=remaining)
	mov	x0, #0
	mov	x1, x12
	mov	x8, #63
	svc	#0
	cbnz	x0, read_got_data
	// x0 == 0 -> EOF
	b	read_done

read_got_data:
	// add returned count to len
	adrp	x13, buf_len
	add	x13, x13, :lo12:buf_len
	ldr	x14, [x13]
	add	x14, x14, x0
	str	x14, [x13]
	// loop again
	b	read_loop

read_done:
	// load total length into x20
	adrp	x20, buf_len
	add	x20, x20, :lo12:buf_len
	ldr	x20, [x20]              // x20 = total_len

	// i = 0
	mov	x19, #0                 // i

process_loop:
	cmp	x19, x20
	bge	done

	// line_start = i
	mov	x21, x19                // line_start

	// find newline or end
	mov	x22, x19                // scan index

find_nl:
	cmp	x22, x20
	bge	line_end_no_nl
	adrp	x23, input_buf
	add	x23, x23, :lo12:input_buf
	add	x24, x23, x22
	ldrb	w25, [x24]
	cmp	w25, #0x0a
	beq	line_end_nl
	add	x22, x22, #1
	b	find_nl

line_end_nl:
	// newline at x22, line_end = x22 (exclude newline)
	sub	x26, x22, x21           // len = line_end - line_start
	cbnz	x26, rev_nonempty
	// empty line: just write newline
	b	write_newline

rev_nonempty:
	// ptr = line_start + len - 1
	mov	x27, x26
	sub	x27, x27, #1
	add	x27, x21, x27          // x27 = index of last byte

rev_loop:
	// load byte from input_buf[x27]
	adrp	x28, input_buf
	add	x28, x28, :lo12:input_buf
	add	x29, x28, x27
	ldrb	w30, [x29]
	// store to tmp
	adrp	x31, tmp
	add	x31, x31, :lo12:tmp
