.global _start
.section .bss
.align 16
inbuf:   .zero 65536
revbuf:  .zero 65536

.section .rodata
newline: .byte 0x0A

.section .text
_start:
    xor ebx, ebx                # total_len = 0
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + inbuf]
    add rsi, rbx                # buffer + offset
    mov rdx, 65536
    sub rdx, rbx                # remaining space
    syscall
    cmp rax, 0
    je process_input
    jl exit_error               # negative => error
    add rbx, rax                # total_len += bytes read
    cmp rbx, 65536
    jl read_loop

process_input:
    mov rcx, rbx                # rcx = bytes remaining
    lea rsi, [rip + inbuf]      # rsi = current pointer

process_loop:
    test rcx, rcx
    jz exit_success
    xor r8, r8                  # r8 = offset within current buffer
find_nl:
    cmp r8, rcx
    je final_line               # no newline found
    mov al, byte ptr [rsi + r8]
    cmp al, 0x0A
    je have_nl
    inc r8
    jmp find_nl

have_nl:
    mov r9, r8                  # r9 = line_len copy
    test r9, r9
    jz write_nl
    lea rbx, [rip + revbuf]     # rbx = dest base
copy_rev:
    dec r9
    mov al, byte ptr [rsi + r9]
    mov byte ptr [rbx + r9], al
    test r9, r9
    jne copy_rev
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    lea rsi, [rip + revbuf]
    mov rdx, r8                 # length
    syscall

write_nl:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    add rsi, r8                 # skip line
    inc rsi                     # skip newline
    sub rcx, r8
    dec rcx                     # remove newline
    jmp process_loop

final_line:
    test rcx, rcx
    jz exit_success
    mov r8, rcx                 # r8 = line_len
    lea rbx, [rip + revbuf]
copy_rev_last:
    dec r8
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rbx + r8], al
    test r8, r8
    jne copy_rev_last
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + revbuf]
    mov rdx, rcx
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    jmp exit_success

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
