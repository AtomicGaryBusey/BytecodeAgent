.global _start

.section .bss
.align 16
inbuf: .zero 65536

.section .rodata
newline: .byte 0x0A

.section .text
_start:
    xor rbx, rbx                 # rbx = total bytes read

read_loop:
    mov rax, 0                   # sys_read
    mov rdi, 0                   # fd 0
    lea rsi, [rip + inbuf]
    add rsi, rbx                 # buffer + offset
    mov rdx, 65536
    sub rdx, rbx                 # remaining space
    syscall
    cmp rax, 0
    je process_input
    jl exit_error                # negative -> error
    add rbx, rax                 # total += bytes read
    cmp rbx, 65536
    jl read_loop

process_input:
    mov rsi, rip
