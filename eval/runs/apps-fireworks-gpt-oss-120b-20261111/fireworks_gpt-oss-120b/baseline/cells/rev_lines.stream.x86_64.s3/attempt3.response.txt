.global _start

.section .bss
.align 16
inbuf: .zero 65536

.section .rodata
newline: .byte 0x0A

.section .text
_start:
    xor rbx, rbx                # total bytes read = 0

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + inbuf]
    add rsi, rbx                # buffer + offset
    mov rdx, 65536
    sub rdx, rbx                # remaining space
    syscall
    cmp rax, 0
    je after_read
    js exit_error               # negative -> error
    add rbx, rax                # total += bytes read
    cmp rbx, 65536
    jl read_loop

after_read:
    xor r10, r10                # offset = 0

process_loop:
    cmp r10, rbx
    jge done                    # offset >= total -> finished

    mov r8, r10                 # line_start = offset

search_newline:
    cmp r10, rbx
    je no_newline               # reached end without newline
    lea rsi, [rip + inbuf]
    add rsi, r10
    mov al, byte ptr [rsi]
    cmp al, 0x0A
    je found_newline
    inc r10
    jmp search_newline

found_newline:
    # length = r10 - line_start
    mov rcx, r10
    sub rcx, r8                 # rcx = length

    # reverse output
    mov r9, r8                  # r9 = i = line_start + length - 1
    add r9, rcx
    dec r9

rev_loop:
    cmp r9, r8
    jl rev_done
    lea rsi, [rip + inbuf]
    add rsi, r9
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd 1
    mov rdx, 1                  # count 1
    syscall
    dec r9
    jmp rev_loop

rev_done:
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    inc r10                     # skip the newline
    jmp process_loop

no_newline:
    # final line without trailing newline
    mov rcx, rbx                # rcx = total
    sub rcx, r8                 # rcx = length
    test rcx, rcx
    je done                     # zero-length remainder -> nothing to output

    # reverse output
    mov r9, r8
    add r9, rcx
    dec r9

rev_loop2:
    cmp r9, r8
    jl rev_done2
    lea rsi, [rip + inbuf]
    add rsi, r9
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r9
    jmp rev_loop2

rev_done2:
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    jmp done

done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1                  # non‑zero status
    syscall
