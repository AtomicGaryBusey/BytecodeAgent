	.globl	_start
	.text
_start:
	// total input length in x19 = 0
	mov	x19, #0

read_loop:
	// remaining space = INPUT_BUF_SIZE - total_len
	mov	x20, #49152
	sub	x20, x20, x19
	cbnz	x20, 1f
	b	read_done
1:
	// addr = input_buf + total_len
	adrp	x21, input_buf
	add	x21, x21, :lo12:input_buf
	add	x21, x21, x19

	// read(fd=0, buf, count)
	mov	x0, #0
	mov	x2, x20
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	beq	read_done
	bltz	x0, exit_error
	add	x19, x19, x0
	b	read_loop

read_done:
	// if no input, exit success
	cbnz	x19, 1f
	b	exit_success
1:
	// src pointer = input_buf
	adrp	x22, input_buf
	add	x22, x22, :lo12:input_buf
	mov	x23, x19			// src_len
	// dst pointer = output_buf
	adrp	x24, output_buf
	add	x24, x24, :lo12:output_buf
	mov	x25, #0			// line character count

encode_loop:
	cmp	x23, #3
	blt	encode_tail

	// load three bytes
	ldrb	w0, [x22], #1
	ldrb	w1, [x22], #1
	ldrb	w2, [x22], #1
	sub	x23, x23, #3

	// i0 = b0>>2
	lsr	w3, w0, #2
	// i1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	w4, w0, #3
	lsl	w4, w4, #4
	orr	w4, w4, w1, lsr #4
	// i2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	and	w5, w1, #0xF
	lsl	w5, w5, #2
	orr	w5, w5, w2, lsr #6
	// i3 = b2 & 0x3F
	and	w6, w2, #0x3F

	adrp	x5, b64_table
	add	x5, x5, :lo12:b64_table

	ldrb	w7, [x5, w3]
	strb	w7, [x24], #1
	ldrb	w7, [x5, w4]
	strb	w7, [x24], #1
	ldrb	w7, [x5, w5]
	strb	w7, [x24], #1
	ldrb	w7, [x5, w6]
	strb	w7, [x24], #1

	add	x25, x25, #4
	cmp	x25, #76
	bne	encode_loop

	// newline after 76 chars
	mov	w7, #0x0A
	strb	w7, [x24], #1
	mov	x25, #0
	b	encode_loop

encode_tail:
	cbnz	x23, 1f
	b	encode_finish
1:
	// load first byte
	ldrb	w0, [x22]
	adrp	x5, b64_table
	add	x5, x5, :lo12:b64_table

	// i0 = b0 >> 2
	lsr	w3, w0, #2
	ldrb	w7, [x5, w3]
	strb	w7, [x24], #1

	cmp	x23, #2
	bne	tail_one

	// length == 2
	ldrb	w1, [x22, #1]

	// i1 = ((b0 &3) <<4) | (b1>>4)
	and	w4, w0, #3
	lsl	w4, w4, #4
	orr	w4, w4, w1, lsr #4
	// i2 = ((b1 &0xF) <<2)
	and	w5, w1, #0xF
	lsl	w5, w5, #2

	ldrb	w7, [x5, w4]
	strb	w7, [x24], #1
	ldrb	w7, [x5, w5]
	strb	w7, [x24], #1
	// padding '='
	mov	w7, #'='
	strb	w7, [x24], #1
	strb	w7, [x24], #1

	add	x25, x25, #4
	cmp	x25, #76
	bne	encode_finish
	// newline after 76 chars
	mov	w7, #0x0A
	strb	w7, [x24], #1
	mov	x25, #0
	b	encode_finish

tail_one:
	// length == 1
	// i1 = (b0 &3) <<4
	and	w4, w0, #3
	lsl	w4, w4, #4

	ldrb	w7, [x5, w4]
	strb	w7, [x24], #1
	// two padding '='
	mov	w7, #'='
	strb	w7, [x24], #1
	strb	w7, [x24], #1

	add	x25, x25, #4
	cmp	x25, #76
	bne	encode_finish
	// newline after 76 chars
	mov	w7, #0x0A
	strb	w7, [x24], #1
	mov	x25, #0

encode_finish:
	// if line count non‑zero, emit final newline
	cbnz	x25, 1f
	b	write_output
1:
	mov	w7, #0x0A
	strb	w7, [x24], #1

write_output:
	// length = x24 - output_buf
	adrp	x0, output_buf
	add	x0, x0, :lo12:output_buf
	sub	x1, x24, x0		// total_len in x1

	mov	x2, x1			// remaining
	mov	x3, x0			// ptr
write_loop:
	cbnz	x2, 1f
	b	exit_success
1:
	mov	x0, #1			// fd stdout
	mov	x1, x3
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	ble	exit_error
	sub	x2, x2, x0
	add	x3, x3, x0
	b	write_loop

exit_success:
	mov	x0, #0
	mov	x8, #93
	svc	#0

exit_error:
	mov	x0, #1
	mov	x8, #93
	svc	#0

	.section	.rodata
b64_table:
