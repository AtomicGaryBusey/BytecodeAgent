.globl _start

.section .bss
.align 8
buffer:          .zero 65536          # input buffer

total_lines:     .zero 8
total_words:    .zero 8
total_bytes:    .zero 8
in_word:        .zero 1
numbuf:         .zero 32

.section .rodata
space_char:     .byte 0x20
newline_char:   .byte 0x0A

.section .text
_start:
    # rsi = buffer address, rdx = buffer size
    lea rsi, [rip + buffer]
    mov rdx, 65536

read_loop:
    mov rdi, 0                  # fd = stdin
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz output_results           # EOF

    # add bytes read to total_bytes
    add qword ptr [rip + total_bytes], rax

    # rcx = bytes remaining in this chunk
    mov rcx, rax
    # rbx = pointer to current byte
    mov rbx, rsi

    # load in_word flag into r8b
    movzx r8d, byte ptr [rip + in_word]

proc_loop:
    mov al, byte ptr [rbx]
    inc rbx
    dec rcx

    # newline handling
    cmp al, 0x0A
    jne not_newline
    inc qword ptr [rip + total_lines]
not_newline:
    # whitespace test
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    # not whitespace: possible start of word
    test r8b, r8b
    jne proc_continue          # already in word
    inc qword ptr [rip + total_words]
    mov r8b, 1
    jmp proc_continue

is_ws:
    test r8b, r8b
    je proc_continue           # already not in word
    mov r8b, 0

proc_continue:
    cmp rcx, 0
    jne proc_loop

    # store back in_word flag
    mov byte ptr [rip + in_word], r8b
    jmp read_loop

output_results:
    # --- print total_lines ---
    mov rax, qword ptr [rip + total_lines]
    call uint_to_ascii
    mov rdx, rcx                # length
    mov rsi, rax                # buffer pointer
    mov rdi, 1                  # stdout
    mov rax, 1                  # sys_write
    syscall

    # write space
    mov rdx, 1
    lea rsi, [rip + space_char]
    mov rdi, 1
    mov rax, 1
    syscall

    # --- print total_words ---
    mov rax, qword ptr [rip + total_words]
    call uint_to_ascii
    mov rdx, rcx
    mov rsi, rax
    mov rdi, 1
    mov rax, 1
    syscall

    # write space
    mov rdx, 1
    lea rsi, [rip + space_char]
    mov rdi, 1
    mov rax, 1
    syscall

    # --- print total_bytes ---
    mov rax, qword ptr [rip + total_bytes]
    call uint_to_ascii
    mov rdx, rcx
    mov rsi, rax
    mov rdi, 1
    mov rax, 1
    syscall

    # write newline
    mov rdx, 1
    lea rsi, [rip + newline_char]
    mov rdi, 1
    mov rax, 1
    syscall

    # exit
    mov rdi, 0
    mov rax, 60                 # sys_exit
    syscall

# ------------------------------------------------------------
# Convert unsigned integer in RAX to decimal ASCII.
# Returns:
#   RAX = pointer to start of string
#   RCX = length of string
# Uses: RDX, RDI, RBX (preserved by caller if needed)
# ------------------------------------------------------------
uint_to_ascii:
    # rbx = pointer to end of numbuf
    lea rbx, [rip + numbuf + 32]
    xor rcx, rcx                # length = 0
    cmp rax, 0
    jne .convert_loop
    # value is zero
    dec rbx
    mov byte ptr [rbx], '0'
    inc rcx
    jmp .done

.convert_loop:
    xor rdx, rdx
    mov rdi, 10
    div rdi                     # rax = quotient, rdx = remainder
    add dl, '0'
    dec rbx
    mov byte ptr [rbx], dl
    inc rcx
    cmp rax, 0
    jne .convert_loop

.done:
    mov rax, rbx                # return pointer in rax
    ret
