.globl _start
.text
.equ   PROG_MAX, 65536
.equ   TAPE_SIZE, 30000
.equ   STACK_MAX, 1000

_start:
    # -------------------------------------------------
    # read all stdin into prog_buf, length -> rbx
    # -------------------------------------------------
    mov     rdi, 0                      # fd = stdin
    lea     rsi, [rip + prog_buf]       # buffer
    mov     rdx, PROG_MAX               # max count
    xor     rbx, rbx                    # total read so far
read_loop:
    mov     rax, 0                      # sys_read
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax
    cmp     rbx, PROG_MAX
    jae     read_done
    jmp     read_loop
read_done:
    mov     [prog_len], rbx

    # -------------------------------------------------
    # locate first '!' (0x21)
    # -------------------------------------------------
    mov     rcx, rbx                    # rcx = prog_len
    xor     rdx, rdx                    # index i = 0
find_excl:
    cmp     rdx, rcx
    jae     no_excl
    mov     al, byte ptr [rip + prog_buf + rdx]
