	.global _start
	.text
	.align 2

_start:
	// x9 = total_len = 0
	mov	x9, #0

	// x0 = fd = 0
	mov	x0, #0
	// x1 = buffer = input_buf
	adrp	x1, input_buf
	add	x1, x1, #:lo12:input_buf
	// x2 = BUFFER_SIZE
	mov	x2, #65536
	// x8 = sys_read
	mov	x8, #63

read_loop:
	// remaining = BUFFER_SIZE - total_len
	mov	x3, #65536
	sub	x3, x3, x9
	cmp	x3, #0
	b.eq	read_done
	// set x2 = remaining
	mov	x2, x3
	// set x1 = input_buf + total_len
	adrp	x4, input_buf
	add	x4, x4, #:lo12:input_buf
	add	x1, x4, x9
	svc	#0			// read
	cmp	x0, #0
	beq	read_done
	cmp	x0, #0
	b.lt	error_exit		// negative => error
	add	x9, x9, x0
	b	read_loop

read_done:
	// x19 = prog_ptr = input_buf
	adrp	x19, input_buf
	add	x19, x19, #:lo12:input_buf

	// Find first '!' (0x21)
	mov	x20, #0			// prog_len candidate
find_exclam:
	cmp	x20, x9
	b.ge	no_exclam		// reached end
	ldrb	w0, [x19, x20]
	cmp	w0, #0x21
	beq	exclam_found
	add	x20, x20, #1
	b	find_exclam

exclam_found:
	// prog_len = x20
	// input_len = total_len - prog_len - 1
	sub	x22, x9, x20
	sub	x22, x22, #1
	// input_ptr = prog_ptr + prog_len + 1
	add	x21, x19, x20
	add	x21, x21, #1
	b	after_split

no_exclam:
	// whole input is program, no program input
	mov	x22, #0			// input_len = 0
	// prog_len = total_len
	mov	x20, x9
	// input_ptr = prog_ptr (won't be used)
	add	x21, x19, #0

after_split:
	// Set up other bases
	adrp	x28, bracket_map
	add	x28, x28, #:lo12:bracket_map
	adrp	x25, stack
	add	x25, x25, #:lo12:stack
	adrp	x27, tape
	add	x27, x27, #:lo12:tape

	// Initialize bracket checking
	mov	x23, #0			// stack_idx
	mov	x24, #0			// pc (used as index)

bracket_loop:
	cmp	x24, x20
	b.ge	bracket_done
	ldrb	w0, [x19, x24]
	cmp	w0, #'['
	beq	bracket_lpush
	cmp	w0, #']'
	beq	bracket_rpop
	// other char: ignore
	add	x24, x24, #1
	b	bracket_loop

bracket_lpush:
	// push x24 onto stack
	lsl	x5, x23, #2
	add	x5, x25, x5
	str	w24, [x5]
	add	x23, x23, #1
	add	x24, x24, #1
	b	bracket_loop

bracket_rpop:
	// if stack empty => error
	cbnz	x23, bracket_rpop_ok
	// error: unbalanced
	b	error_unbalanced
bracket_rpop_ok:
	sub	x23, x23, #1
	lsl	x5, x23, #2
	add	x5, x25, x5
	ldr	w1, [x5]			// w1 = matching '[' index

	// store match for '['
	uxtw	x6, w1
	lsl	x6, x6, #2
	add	x6, x28, x6
	mov	w2, w24
	str	w2, [x6]

	// store match for ']'
	lsl	x7, x24, #2
	add	x7, x28, x7
	mov	w2, w1
	str	w2, [x7]

	add	x24, x24, #1
	b	bracket_loop

bracket_done:
	// after loop, stack must be empty
	cmp	x23, #0
	b.ne	error_unbalanced

	// Initialize execution registers
	mov	x24, #0			// pc
	mov	x26, #0			// input index
	mov	x29, #0			// data pointer

exec_loop:
	cmp	x24, x20
	b.ge	exec_done
	ldrb	w0, [x19, x24]

	cmp	w0, #'>'
	beq	cmd_inc_ptr
	cmp	w0, #'<'
	beq	cmd_dec_ptr
	cmp	w0, #'+'
	beq	cmd_inc_cell
	cmp	w0, #'-'
	beq	cmd_dec_cell
	cmp	w0, #'.'
	beq	cmd_output
	cmp	w0, #','
	beq	cmd_input
	cmp	w0, #'['
	beq	cmd_bracket_l
	cmp	w0, #']'
	beq	cmd_bracket_r
	// ignore any other character
	add	x24, x24, #1
	b	exec_loop

cmd_inc_ptr:
	add	x29, x29, #1
	add	x24, x24, #1
	b	exec_loop

cmd_dec_ptr:
	sub	x29, x29, #1
	add	x24, x24, #1
	b	exec_loop

cmd_inc_cell:
	add	x5, x27, x29
	ldrb	w1, [x5]
	add	w1, w1, #1
	and	w1, w1, #0xFF
	strb	w1, [x5]
	add	x24, x24, #1
	b	exec_loop

cmd_dec_cell:
	add	x5, x27, x29
	ldrb	w1, [x5]
	sub	w1, w1, #1
	and	w1, w1, #0xFF
	strb	w1, [x5]
	add	x24, x24, #1
	b	exec_loop

cmd_output:
	mov	x0, #1
	add	x1, x27, x29
	mov	x2, #1
	mov	x8, #64
	svc	#0
	add	x24, x24, #1
	b	exec_loop

cmd_input:
	cmp	x26, x22
	b.ge	input_eof
	add	x5, x21, x26
	ldrb	w0, [x5]
	add	x5, x27, x29
	strb	w0, [x5]
	add	x26, x26, #1
	b	input_done
input_eof:
	add	x5, x27, x29
	mov	w0, #0
	strb	w0, [x5]
input_done:
	add	x24, x24, #1
	b	exec_loop

cmd_bracket_l:
	add	x5, x27, x29
	ldrb	w0, [x5]
	cbz	w0, bracket_l_jump
	add	x24, x24, #1
	b	exec_loop
bracket_l_jump:
	lsl	x5, x24, #2
	add	x5, x28, x5
	ldr	w1, [x5]			// matching ']'
	add	w1, w1, #1
	mov	x24, x1
	b	exec_loop

cmd_bracket_r:
	add	x5, x27, x29
	ldrb	w0, [x5]
	cbz	w0, bracket_r_skip
	lsl	x5, x24, #2
	add	x5, x28, x5
	ldr	w1, [x5]			// matching '['
	add	w1, w1, #1
	mov	x24, x1
	b	exec_loop
bracket_r_skip:
	add	x24, x24, #1
	b	exec_loop

exec_done:
	// exit 0
	mov	w0, #0
	mov	x8, #93
	svc	#0

error_unbalanced:
	// exit 1
	mov	w0, #1
	mov	x8, #93
	svc	#0

error_exit:
	// exit 1 (generic error)
	mov	w0, #1
	mov	x8, #93
	svc	#0

	.section .bss
	.align 8
input_buf:
	.space 65536

	.align 8
bracket_map:
	.space 262144		// 65536 * 4

	.align 8
stack:
	.space 4000		// 1000 * 4

	.align 8
tape:
	.space 30000
