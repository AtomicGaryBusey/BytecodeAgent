.globl _start

.section .bss
.align 16
buffer:      .zero 65536
input_len:   .zero 8
error_offset:.zero 8

.section .rodata
msg_ok:      .string "ok\n"
msg_ok_len  = . - msg_ok
msg_err_prefix: .string "error at "
msg_err_prefix_len = . - msg_err_prefix
msg_err_suffix: .string "\n"
msg_err_suffix_len = . - msg_err_suffix

.section .text
_start:
    # read stdin into buffer
    xor    rdi, rdi                 # fd = 0
    lea    rsi, [rip + buffer]
    mov    rdx, 65536
    xor    rax, rax                 # sys_read
    syscall
    test   rax, rax
    jz     .done_read
    mov    rbx, rax                 # total bytes read
.read_loop:
    xor    rdi, rdi
    lea    rsi, [rip + buffer]
    add    rsi, rbx                 # buffer + rbx
    mov    rdx, 65536
    sub    rdx, rbx
    xor    rax, rax
    syscall
    test   rax, rax
    jz     .done_read
    add    rbx, rax
    cmp    rbx, 65536
    jb     .read_loop
.done_read:
    mov    qword ptr [rip + input_len], rbx

    xor    rsi, rsi                 # idx = 0
    call   parse_ws
    mov    rsi, rax
    call   parse_value
    mov    rsi, rax
    call   parse_ws
    mov    rsi, rax
    mov    rax, qword ptr [rip + input_len]
    cmp    rsi, rax
    jne    .error_extra

    # success output
    mov    rdi, 1
    lea    rsi, [rip + msg_ok]
    mov    rdx, msg_ok_len
    mov    rax, 1
    syscall
    xor    edi, edi
    mov    eax, 60
    syscall

.error_extra:
    mov    rax, rsi
    mov    qword ptr [rip + error_offset], rax
    jmp    .print_error

parse_error_eof:
    mov    rax, qword ptr [rip + input_len]
    jmp    .set_error

.set_error:
    mov    qword ptr [rip + error_offset], rax
    jmp    .print_error

.print_error:
    mov    rdi, 1
    lea    rsi, [rip + msg_err_prefix]
    mov    rdx, msg_err_prefix_len
    mov    rax, 1
    syscall
    mov    rax, qword ptr [rip + error_offset]
    call   write_number
    mov    rdi, 1
    lea    rsi, [rip + msg_err_suffix]
    mov    rdx, msg_err_suffix_len
    mov    rax, 1
    syscall
    mov    edi, 1
    mov    eax, 60
    syscall

# write_number: rax = number to write
write_number:
    push   rbp
    mov    rbp, rsp
    sub    rsp, 32
    mov    rbx, rax
    xor    rcx, rcx
    test   rbx, rbx
    jne    .wn_loop
    mov    byte ptr [rsp+31], '0'
    inc    rcx
    jmp    .wn_write
.wn_loop:
    xor    rdx, rdx
    mov    rdi, 10
    div    rdi
    add    dl, '0'
    mov    byte ptr [rsp+31 - rcx], dl
    inc    rcx
    mov    rbx, rax
    test   rbx, rbx
    jne    .wn_loop
.wn_write:
    lea    rsi, [rsp+32 - rcx]
    mov    rdx, rcx
    mov    rdi, 1
    mov    rax, 1
    syscall
    add    rsp, 32
    pop    rbp
    ret

parse_ws:
    mov    rax, rsi
    mov    rdx, qword ptr [rip + input_len]
.ws_loop:
    cmp    rax, rdx
    jae    .ws_done
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rax]
    cmp    bl, 0x20
    je     .ws_inc
    cmp    bl, 0x09
    je     .ws_inc
    cmp    bl, 0x0A
    je     .ws_inc
    cmp    bl, 0x0D
    je     .ws_inc
    jmp    .ws_done
.ws_inc:
    inc    rax
    jmp    .ws_loop
.ws_done:
    ret

parse_value:
    mov    rdx, qword ptr [rip + input_len]
    mov    rax, rsi
    cmp    rax, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rax]
    cmp    bl, '{'
    je     .val_object
    cmp    bl, '['
    je     .val_array
    cmp    bl, '"'
    je     .val_string
    cmp    bl, '-'
    je     .val_number
    cmp    bl, 't'
    je     .val_true
    cmp    bl, 'f'
    je     .val_false
    cmp    bl, 'n'
    je     .val_null
    cmp    bl, '0'
    je     .val_number
    cmp    bl, '1'
    je     .val_number
    cmp    bl, '2'
    je     .val_number
    cmp    bl, '3'
    je     .val_number
    cmp    bl, '4'
    je     .val_number
    cmp    bl, '5'
    je     .val_number
    cmp    bl, '6'
    je     .val_number
    cmp    bl, '7'
    je     .val_number
    cmp    bl, '8'
    je     .val_number
    cmp    bl, '9'
    je     .val_number
    mov    rax, rsi
    jmp    .set_error
.val_object:
    mov    rdi, rsi
    call   parse_object
    ret
.val_array:
    mov    rdi, rsi
    call   parse_array
    ret
.val_string:
    mov    rdi, rsi
    call   parse_string
    ret
.val_number:
    mov    rdi, rsi
    call   parse_number
    ret
.val_true:
    mov    rdi, rsi
    call   parse_true
    ret
.val_false:
    mov    rdi, rsi
    call   parse_false
    ret
.val_null:
    mov    rdi, rsi
    call   parse_null
    ret

parse_object:
    lea    rsi, [rdi + 1]           # after '{'
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '}'
    je     .obj_end
.member_loop:
    mov    rdi, rsi
    call   parse_string
    mov    rsi, rax
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, ':'
    jne    .obj_error
    lea    rsi, [rsi + 1]
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    mov    rdi, rsi
    call   parse_value
    mov    rsi, rax
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, ','
    je     .obj_comma
    cmp    bl, '}'
    je     .obj_end
    jmp    .obj_error
.obj_comma:
    lea    rsi, [rsi + 1]
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    jmp    .member_loop
.obj_end:
    lea    rax, [rsi + 1]
    ret
.obj_error:
    mov    rax, rsi
    jmp    .set_error

parse_array:
    lea    rsi, [rdi + 1]           # after '['
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, ']'
    je     .arr_end
.value_loop:
    mov    rdi, rsi
    call   parse_value
    mov    rsi, rax
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, ','
    je     .arr_comma
    cmp    bl, ']'
    je     .arr_end
    jmp    .arr_error
.arr_comma:
    lea    rsi, [rsi + 1]
    mov    rdi, rsi
    call   parse_ws
    mov    rsi, rax
    jmp    .value_loop
.arr_end:
    lea    rax, [rsi + 1]
    ret
.arr_error:
    mov    rax, rsi
    jmp    .set_error

parse_string:
    mov    rsi, rdi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '"'
    jne    .str_error
    inc    rsi
.str_loop:
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    mov    al, byte ptr [rbx + rsi]
    cmp    al, '"'
    je     .str_done
    cmp    al, '\\'
    je     .str_escape
    cmp    al, 0x20
    jb     .str_error
    inc    rsi
    jmp    .str_loop
.str_escape:
    inc    rsi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    mov    al, byte ptr [rbx + rsi]
    cmp    al, '"'
    je     .esc_ok
    cmp    al, '\\'
    je     .esc_ok
    cmp    al, '/'
    je     .esc_ok
    cmp    al, 'b'
    je     .esc_ok
    cmp    al, 'f'
    je     .esc_ok
    cmp    al, 'n'
    je     .esc_ok
    cmp    al, 'r'
    je     .esc_ok
    cmp    al, 't'
    je     .esc_ok
    cmp    al, 'u'
    je     .unicode_escape
    jmp    .str_error
.esc_ok:
    inc    rsi
    jmp    .str_loop
.unicode_escape:
    inc    rsi
    mov    rcx, 4
.unicode_loop:
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    mov    al, byte ptr [rbx + rsi]
    mov    bl, al
    cmp    bl, '0'
    jb     .unicode_invalid
    cmp    bl, '9'
    jbe    .unicode_next
    cmp    bl, 'a'
    jb     .unicode_check_upper
    cmp    bl, 'f'
    jbe    .unicode_next
.unicode_check_upper:
    cmp    bl, 'A'
    jb     .unicode_invalid
    cmp    bl, 'F'
    ja     .unicode_invalid
.unicode_next:
    inc    rsi
    dec    rcx
    jnz    .unicode_loop
    jmp    .str_loop
.unicode_invalid:
    jmp    .str_error
.str_done:
    inc    rsi
    mov    rax, rsi
    ret
.str_error:
    mov    rax, rsi
    jmp    .set_error

parse_number:
    mov    rsi, rdi
    mov    rdx, qword ptr [rip + input_len]
    lea    rbx, [rip + buffer]
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '-'
    jne    .num_int_start
    inc    rsi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
.num_int_start:
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '0'
    je     .num_int_zero
    cmp    bl, '1'
    jb     .num_error
    cmp    bl, '9'
    ja     .num_error
.num_int_nonzero:
    inc    rsi
.num_int_loop:
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    .num_frac_check
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '0'
    jb     .num_frac_check
    cmp    bl, '9'
    ja     .num_frac_check
    inc    rsi
    jmp    .num_int_loop
.num_int_zero:
    inc    rsi
.num_frac_check:
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    .num_exp_check
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '.'
    jne    .num_exp_check
    inc    rsi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '0'
    jb     .num_error
    cmp    bl, '9'
    ja     .num_error
.num_frac_loop:
    inc    rsi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    .num_exp_check
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '0'
    jb     .num_exp_check
    cmp    bl, '9'
    ja     .num_exp_check
    jmp    .num_frac_loop
.num_exp_check:
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    .num_done
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, 'e'
    je     .num_exp_start
    cmp    bl, 'E'
    jne    .num_done
.num_exp_start:
    inc    rsi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '+'
    je     .num_exp_sign
    cmp    bl, '-'
    je     .num_exp_sign
    jmp    .num_exp_digit_check
.num_exp_sign:
    inc    rsi
.num_exp_digit_check:
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    parse_error_eof
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '0'
    jb     .num_error
    cmp    bl, '9'
    ja     .num_error
.num_exp_loop:
    inc    rsi
    mov    rdx, qword ptr [rip + input_len]
    cmp    rsi, rdx
    jae    .num_done
    mov    bl, byte ptr [rbx + rsi]
    cmp    bl, '0'
    jb     .num_done
    cmp    bl, '9'
    ja     .num_done
    jmp    .num_exp_loop
.num_done:
    mov    rax, rsi
    ret
.num_error:
    mov    rax, rsi
    jmp    .set_error

parse_true:
    mov    rsi, rdi
    mov    rdx, qword ptr [rip + input_len]
    mov    rcx, rsi
    add    rcx, 4
    cmp    rcx, rdx
    ja     parse_error_eof
    lea    rbx, [rip + buffer]
    mov    al, byte ptr [rbx + rsi]
    cmp    al, 't'
    jne    .true_error
    mov    al, byte ptr [rbx + rsi + 1]
    cmp    al, 'r'
    jne    .true_error
    mov    al, byte ptr [rbx + rsi + 2]
    cmp    al, 'u'
    jne    .true_error
    mov    al, byte ptr [rbx + rsi + 3]
    cmp    al, 'e'
    jne    .true_error
    lea    rax, [rsi + 4]
    ret
.true_error:
    mov    rax, rsi
    jmp    .set_error

parse_false:
    mov    rsi, rdi
    mov    rdx, qword ptr [rip + input_len]
    mov    rcx, rsi
    add    rcx, 5
    cmp    rcx, rdx
    ja     parse_error_eof
    lea    rbx, [rip + buffer]
    mov    al, byte ptr [rbx + rsi]
    cmp    al, 'f'
    jne    .false_error
    mov    al, byte ptr [rbx + rsi + 1]
    cmp    al, 'a'
    jne    .false_error
    mov    al, byte ptr [rbx + rsi + 2]
    cmp    al, 'l'
    jne    .false_error
    mov    al, byte ptr [rbx + rsi + 3]
    cmp    al, 's'
    jne    .false_error
    mov    al, byte ptr [rbx + rsi + 4]
    cmp    al, 'e'
    jne    .false_error
    lea    rax, [rsi + 5]
    ret
.false_error:
    mov    rax, rsi
    jmp    .set_error

parse_null:
    mov    rsi, rdi
    mov    rdx, qword ptr [rip + input_len]
    mov    rcx, rsi
    add    rcx, 4
    cmp    rcx, rdx
    ja     parse_error_eof
    lea    rbx, [rip + buffer]
    mov    al, byte ptr [rbx + rsi]
    cmp    al, 'n'
    jne    .null_error
    mov    al, byte ptr [rbx + rsi + 1]
    cmp    al, 'u'
    jne    .null_error
    mov    al, byte ptr [rbx + rsi + 2]
    cmp    al, 'l'
    jne    .null_error
    mov    al, byte ptr [rbx + rsi + 3]
    cmp    al, 'l'
    jne    .null_error
    lea    rax, [rsi + 4]
    ret
.null_error:
    mov    rax, rsi
    jmp    .set_error
