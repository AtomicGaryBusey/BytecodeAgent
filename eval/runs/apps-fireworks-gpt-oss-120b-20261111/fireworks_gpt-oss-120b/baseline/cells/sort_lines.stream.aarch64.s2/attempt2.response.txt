.text
.global _start
.align 4
_start:
    // base addresses
    adrp    x9, buf
    add     x9, x9, :lo12:buf          // x9 = buffer base
    adrp    x10, ptrs
    add     x10, x10, :lo12:ptrs       // x10 = ptr array base
    adrp    x11, lens
    add     x11, x11, :lo12:lens       // x11 = len array base

    // counters
    mov     x12, #0                    // total_len
    mov     x13, #0                    // line_count

read_loop:
    mov     x0, #0                     // fd = stdin
    add     x1, x9, x12                // buffer + total_len
    mov     x2, #32768
    sub     x2, x2, x12                // remaining space
    mov     x8, #63                    // sys_read
    svc     #0
    cbz     x0, read_done
    add     x12, x12, x0               // total_len += bytes read
    b       read_loop
read_done:

    // parse lines
    mov     x14, #0                    // line_start offset
    mov     x15, #0                    // idx offset

parse_loop:
    cmp     x15, x12
    bge     parse_end
    add     x4, x9, x15
    ldrb    w5, [x4]
    cmp     w5, #0x0A
    bne     parse_continue

    // newline found
    sub     x6, x15, x14               // len = idx - start
    add     x7, x9, x14                // pointer to start
    // store pointer
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x10, x8
    str     x7, [x8]
    // store length (8 bytes)
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x11, x8
    str     x6, [x8]
    add     x13, x13, #1               // line_count++
    add     x14, x15, #1               // next start after newline
parse_continue:
    add     x15, x15, #1
    b       parse_loop

parse_end:
    cmp     x14, x12
    bge     no_final
    sub     x6, x12, x14               // final line length
    add     x7, x9, x14                // pointer
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x10, x8
    str     x7, [x8]
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x11, x8
    str     x6, [x8]
    add     x13, x13, #1
no_final:
    cbz     x13, exit

    // insertion sort
    mov     x19, #1                    // i = 1
sort_i_loop:
    cmp     x19, x13
    bge     sort_done

    // key = ptrs[i], len = lens[i]
    mov     x20, x19
    lsl     x20, x20, #3
    add     x21, x10, x20
    ldr     x22, [x21]                 // key_ptr
    add     x23, x11, x20
    ldr     x24, [x23]                 // key_len

    // j = i - 1
    sub     x25, x19, #1               // j
sort_while:
    cmp     x25, #-1
    blt     sort_insert_place

    // load ptrs[j] and lens[j]
    mov     x26, x25
    lsl     x26, x26, #3
    add     x27, x10, x26
    ldr     x28, [x27]                 // ptr_j
    add     x29, x11, x26
    ldr     x30, [x29]                 // len_j

    // compare line j with key (a = j, b = key)
    mov     x0, x28                    // a_ptr
    mov     x1, x30                    // a_len
    mov     x2, x22                    // b_ptr
    mov     x3, x24                    // b_len
    bl      cmp_lines

    // result in w0: -1 if a<b, 0 if equal, 1 if a>b
    cmp     w0, #0
    ble     sort_insert_place          // a <= b -> stop shifting

    // shift ptrs[j] to position j+1
    add     x26, x26, #8               // (j+1)*8
    add     x27, x10, x26
    str     x28, [x27]
    // shift len
    add     x29, x11, x26
    str     x30, [x29]

    sub     x25, x25, #1               // j--
    b       sort_while

sort_insert_place:
    // place key at position j+1
    add     x26, x25, #1
    lsl     x26, x26, #3
    add     x27, x10, x26
    str     x22, [x27]
    add     x29, x11, x26
    str     x24, [x29]

    add     x19, x19, #1
    b       sort_i_loop

sort_done:
    // output sorted lines
    mov     x5, #0                     // line index i = 0
out_loop:
    cmp     x5, x13
    bge     exit

    // load ptr and len for line i
    mov     x0, x5
    lsl     x0, x0, #3
    add     x1, x10, x0
    ldr     x2, [x1]                   // line_ptr
    add     x1, x11, x0
    ldr     x3, [x1]                   // line_len

    // write line
    mov     x0, #1                     // fd = stdout
    mov     x1, x2                     // buffer
    mov     x2, x3                     // count
    mov     x8, #64
    svc     #0

    // write newline
    adr     x1, newline
    mov     x2, #1
    mov     x0, #1
    mov     x8, #64
    svc     #0

    add     x5, x5, #1
    b       out_loop

exit:
    mov     x0, #0
    mov     x8, #93
    svc     #0

// -------------------------------------------------------------
// compare_lines:
//   x0 = a_ptr, x1 = a_len, x2 = b_ptr, x3 = b_len
// Returns:
//   w0 = -1 if a<b, 0 if a==b, 1 if a>b
// Clobbers x4-x9, preserves others (including x30 as return address)
cmp_lines:
    mov     x4, x1                     // a_len
    mov     x5, x3                     // b_len
    mov     x6, #0                     // index

cmp_loop:
    cmp     x6, x4
    b.ge    a_done
    cmp     x6, x5
    b.ge    b_done

    add     x7, x0, x6
    ldrb    w8, [x7]                  // a_byte
    add     x7, x2, x6
    ldrb    w9, [x7]                  // b_byte
    cmp     w8, w9
    b.ne    diff_bytes
    add     x6, x6, #1
    b       cmp_loop

a_done:
    // a exhausted
    cmp     x5, x4
    beq     equal
    // a shorter
    mov     w0, #-1
    ret

b_done:
    // b exhausted
    cmp     x4, x5
    beq     equal
    // a longer
    mov     w0, #1
    ret

diff_bytes:
    cmp     w8, w9
    b.lt    less
    // greater
    mov     w0, #1
    ret
less:
    mov     w0, #-1
    ret

equal:
    mov     w0, #0
    ret

.section .rodata
newline:
    .byte   0x0A

.bss
.align 8
buf:
    .space 32768
ptrs:
    .space 8000      // 1000 * 8
lens:
    .space 8000      // 1000 * 8 (store lengths as 64‑bit)
