.text
.globl _start
.align 4
_start:
    // load addresses of buffers
    adrp    x19, buf
    add     x19, x19, :lo12:buf          // x19 = buffer base
    adrp    x22, line_ptrs
    add     x22, x22, :lo12:line_ptrs    // x22 = line pointers array
    adrp    x23, line_lens
    add     x23, x23, :lo12:line_lens    // x23 = line lengths array

    mov     x20, #0                      // total_len = 0
    mov     x21, #0                      // line_count = 0

    // read loop
    mov     x0, #0                       // fd = 0
    mov     x1, x19                      // buffer address
    mov     x2, #32768
    mov     x8, #63                      // sys_read
read_loop:
    svc     #0
    cbz     x0, read_done
    add     x20, x20, x0                 // total_len += rdx
    // set up next read position
    add     x1, x19, x20                 // buf + total_len
    mov     x2, #32768
    sub     x2, x2, x20                  // remaining space
    b       read_loop
read_done:

    // parse lines
    mov     x24, #0                      // line_start = 0
    mov     x25, #0                      // idx = 0
parse_loop:
    cmp     x25, x20
    bge     parse_end
    add     x5, x19, x25
    ldrb    w6, [x5]
    cmp     w6, #0x0A
    bne     parse_continue
    // newline found
    sub     x7, x25, x24                 // len = idx - start
    // store pointer
    mov     x8, x21
    lsl     x8, x8, #3
    add     x9, x22, x8
    str     x24, [x9]
    // store length
    add     x9, x23, x8
    str     x7, [x9]
    add     x21, x21, #1                 // line_count++
    add     x24, x25, #1                 // start = idx+1
parse_continue:
    add     x25, x25, #1
    b       parse_loop
parse_end:
    cmp     x24, x20
    bge     no_final_line
    sub     x7, x20, x24                 // final line length
    mov     x8, x21
    lsl     x8, x8, #3
    add     x9, x22, x8
    str     x24, [x9]
    add     x9, x23, x8
    str     x7, [x9]
    add     x21, x21, #1
no_final_line:

    // insertion sort
    mov     x10, #1                      // i = 1
sort_i_loop:
    cmp     x10, x21
    bge     sort_done
    // key = ptrs[i]
    mov     x11, x10
    lsl     x11, x11, #3
    add     x12, x22, x11
    ldr     x13, [x12]                   // key_ptr
    add     x14, x23, x11
    ldr     x15, [x14]                   // key_len
    sub     x16, x10, #1                 // j = i-1
sort_while:
    cmp     x16, #-1
    blt     sort_insert_place
    // load ptrs[j] and lens[j]
    mov     x17, x16
    lsl     x17, x17, #3
    add     x18, x22, x17
    ldr     x19a, [x18]                  // ptr_j
