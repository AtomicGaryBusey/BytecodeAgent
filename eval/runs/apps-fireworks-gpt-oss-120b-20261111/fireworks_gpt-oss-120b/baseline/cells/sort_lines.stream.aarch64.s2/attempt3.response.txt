.text
.global _start
.align 4
_start:
    // base addresses
    adrp    x9, buf
    add     x9, x9, :lo12:buf          // x9 = buffer base
    adrp    x10, ptrs
    add     x10, x10, :lo12:ptrs       // x10 = ptr array base
    adrp    x11, lens
    add     x11, x11, :lo12:lens       // x11 = len array base

    // counters
    mov     x12, #0                    // total_len
    mov     x13, #0                    // line_count

read_loop:
    mov     x0, #0                     // fd = stdin
    add     x1, x9, x12                // buffer + total_len
    mov     x2, #32768
    sub     x2, x2, x12                // remaining space
    mov     x8, #63                    // sys_read
    svc     #0
    cbz     x0, read_done
    add     x12, x12, x0               // total_len += bytes read
    b       read_loop
read_done:

    // parse lines
    mov     x14, #0                    // line_start offset
    mov     x15, #0                    // idx offset

parse_loop:
    cmp     x15, x12
    b.ge    parse_end
    add     x4, x9, x15
    ldrb    w5, [x4]
    cmp     w5, #0x0A
    b.ne    parse_continue

    // newline found
    sub     x6, x15, x14               // len = idx - start
    add     x7, x9, x14                // pointer to start
    // store pointer
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x10, x8
    str     x7, [x8]
    // store length
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x11, x8
    str     x6, [x8]
    add     x13, x13, #1               // line_count++
    add     x14, x15, #1               // next start after newline
parse_continue:
    add     x15, x15, #1
    b       parse_loop

parse_end:
    cmp     x14, x12
    b.ge    no_final
    sub     x6, x12, x14               // final line length
    add     x7, x9, x14                // pointer
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x10, x8
    str     x7, [x8]
    mov     x8, x13
    lsl     x8, x8, #3
    add     x8, x11, x8
    str     x6, [x8]
    add     x13, x13, #1
no_final:
    cbz     x13, exit_prog

    // insertion sort
    mov     x19, #1                    // i = 1
sort_i_loop:
    cmp     x19, x13
    b.ge    sort_done

    // key = ptrs[i], len = lens[i]
    mov     x20, x19
    lsl     x20, x20, #3
    add     x21, x10, x20
    ldr     x22, [x21]                 // key_ptr
    add     x23, x11, x20
    ldr     x24, [x23]                 // key_len

    // j = i - 1
    sub     x25, x19, #1               // j
sort_while:
    cmp     x25, #-1
    blt     sort_insert_place

    // load ptrs[j] and lens[j]
    mov     x26, x25
    lsl     x26, x26, #3
    add     x27, x10, x26
    ldr     x28, [x27]                 // ptr_j
    add     x29, x11, x26
    ldr     x30, [x29]                 // len_j (will be overwritten by bl)

    // compare line j with key (a = j, b = key)
    mov     x0, x28                    // a_ptr
    mov     x1, x30                    // a_len
    mov     x2, x22                    // b_ptr
    mov     x3, x24                    // b_len
    bl      cmp_lines

    // result in w0: -1 if a<b, 0 if equal, 1 if a>b
    cmp     w0, #0
    ble     sort_insert_place          // a <= b -> stop shifting

    // shift ptrs[j] to position j+1
    add     x26, x26, #8               // (j+1)*8
    add     x27, x10, x26
    str     x28, [x27]
    // reload len_j (since x30 was clobbered by the call)
    sub     x31, x26, #8               // j*8
