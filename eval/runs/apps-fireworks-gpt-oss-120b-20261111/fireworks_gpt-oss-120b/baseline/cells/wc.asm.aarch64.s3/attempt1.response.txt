.text
.global _start
.align 2

_start:
    // counters: lines(x19), words(x20), bytes(x21), in_word flag(w22)
    mov     x19, #0
    mov     x20, #0
    mov     x21, #0
    mov     w22, #0

read_loop:
    // sys_read(fd=0, buf, 65536)
    mov     x8, #63
    mov     x0, #0
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x2, #65536
    svc     #0
    cbz     x0, after_read          // EOF

    // x0 = nread
    mov     x23, x0                 // remaining bytes
    mov     x24, #0                 // offset

process_byte:
    add     x25, x1, x24
    ldrb    w0, [x25]               // byte in w0

    // bytes++
    add     x21, x21, #1

    // newline?
    cmp     w0, #10
    bne     not_newline
    add     x19, x19, #1
not_newline:
    // whitespace check
    mov     w1, w0
    cmp     w1, #9
    beq     is_ws
    cmp     w1, #10
    beq     is_ws
    cmp     w1, #11
    beq     is_ws
    cmp     w1, #12
    beq     is_ws
    cmp     w1, #13
    beq     is_ws
    cmp     w1, #32
    beq     is_ws
    // not whitespace
    cmp     w22, #0
    bne     not_new_word
    // start new word
    add     x20, x20, #1
    mov     w22, #1
    b       not_new_word
is_ws:
    mov     w22, #0
not_new_word:
    // advance offset
    add     x24, x24, #1
    subs    x23, x23, #1
    bne     process_byte
    b       read_loop

after_read:
    // output buffer pointer
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf

    // write lines
    mov     x2, x19
    bl      write_decimal
    // space
    mov     w0, #' '
    strb    w0, [x1], #1

    // write words
    mov     x2, x20
    bl      write_decimal
    // space
    mov     w0, #' '
    strb    w0, [x1], #1

    // write bytes
    mov     x2, x21
    bl      write_decimal
    // newline
    mov     w0, #10
    strb    w0, [x1], #1

    // compute length and write
    adrp    x4, outbuf
    add     x4, x4, :lo12:outbuf
    sub     x2, x1, x4             // length
    mov     x8, #64                // sys_write
    mov     x0, #1                 // fd stdout
    mov     x1, x4                 // buffer start
    svc     #0

    // exit
    mov     x8, #93                // sys_exit
    mov     x0, #0
    svc     #0

// ------------------------------------------------------------
// write_decimal: x2 = number, x1 = dest pointer
// writes decimal representation, advances x1, returns.
write_decimal:
    // digit buffer
    adrp    x3, decbuf
    add     x3, x3, :lo12:decbuf
    add     x4, x3, #24            // end of buffer
    mov     x5, x4                 // current ptr

    mov     w9, #10                // divisor

    cbz     x2, wd_zero

wd_loop:
    udiv    x6, x2, x9             // quotient
    mul     x7, x6, x9             // quotient*10
    sub     x7, x2, x7             // remainder
    add     w7, w7, #'0'
    sub     x5, x5, #1
    strb    w7, [x5]
    mov     x2, x6
    cbnz    x2, wd_loop
    b       wd_copy

wd_zero:
    sub     x5, x5, #1
    mov     w0, #'0'
    strb    w0, [x5]

wd_copy:
    // copy digits to output
wd_copy_loop:
    cmp     x5, x4
    bge     wd_done
    ldrb    w0, [x5], #1
    strb    w0, [x1], #1
    b       wd_copy_loop
wd_done:
    ret

.section .bss
.align 8
buf:
    .zero   65536
decbuf:
    .zero   24
outbuf:
    .zero   32
