.text
.global _start
.align 2

_start:
    // counters: lines (x19), words (x20), bytes (x21), in_word flag (w22)
    mov     x19, #0
    mov     x20, #0
    mov     x21, #0
    mov     w22, #0

read_loop:
    // sys_read(fd=0, buf, 65536)
    mov     x8, #63                 // sys_read
    mov     x0, #0                  // fd 0
    adr     x1, buf                 // buffer address
    mov     x2, #65536
    svc     #0
    cbz     x0, after_read          // EOF (x0 == 0)

    // x0 = nread bytes
    mov     x23, x0                 // remaining count
    mov     x24, #0                 // offset within buffer

process_byte:
    // address = buf + offset
    adr     x25, buf
    add     x25, x25, x24
    ldrb    w0, [x25]               // load byte

    // bytes++
    add     x21, x21, #1

    // newline?
    cmp     w0, #10                 // '\n'
    bne     not_newline
    add     x19, x19, #1
not_newline:
    // whitespace check
    cmp     w0, #9                  // '\t'
    beq     ws
    cmp     w0, #10                 // '\n'
    beq     ws
    cmp     w0, #11                 // vertical tab
    beq     ws
    cmp     w0, #12                 // form feed
    beq     ws
    cmp     w0, #13                 // carriage return
    beq     ws
    cmp     w0, #32                 // space
    beq     ws
    // not whitespace
    cbz     w22, new_word
    b       after_ws
ws:
    mov     w22, #0                 // leave word
    b       after_ws
new_word:
    add     x20, x20, #1            // new word
    mov     w22, #1
after_ws:
    // advance offset
    add     x24, x24, #1
    subs    x23, x23, #1
    bne     process_byte
    b       read_loop

after_read:
    // output buffer pointer
    adr     x1, outbuf

    // write lines
    mov     x2, x19
    bl      write_decimal
    // space
    mov     w0, #' '
    strb    w0, [x1], #1

    // write words
    mov     x2, x20
    bl      write_decimal
    // space
    mov     w0, #' '
    strb    w0, [x1], #1

    // write bytes
    mov     x2, x21
    bl      write_decimal
    // newline
    mov     w0, #10
    strb    w0, [x1], #1

    // compute length and write
    adr     x4, outbuf
    sub     x2, x1, x4              // length
    mov     x8, #64                 // sys_write
    mov     x0, #1                  // fd stdout
    mov     x1, x4                  // buffer start
    svc     #0

    // exit
    mov     x8, #93                 // sys_exit
    mov     x0, #0
    svc     #0

// ------------------------------------------------------------
// write_decimal: x2 = number, x1 = dest pointer (will be advanced)
write_decimal:
    // temporary buffer for digits (24 bytes, enough for 64-bit max)
    adr     x3, decbuf
    add     x4, x3, #24             // end of buffer
    mov     x5, x4                  // current write pointer

    mov     w9, #10                 // divisor

    cbz     x2, wd_zero

wd_loop:
    udiv    x6, x2, x9              // quotient = x2 / 10
    mul     x7, x6, x9              // quotient * 10
    sub     x7, x2, x7              // remainder
    add     w7, w7, #'0'            // convert to ASCII
    sub     x5, x5, #1
    strb    w7, [x5]
    mov     x2, x6
    cbnz    x2, wd_loop
    b       wd_copy

wd_zero:
    sub     x5, x5, #1
    mov     w0, #'0'
    strb    w0, [x5]

wd_copy:
    // copy digits to output buffer, advancing x1
wd_copy_loop:
    cmp     x5, x4
    bge     wd_done
    ldrb    w0, [x5], #1
    strb    w0, [x1], #1
    b       wd_copy_loop
wd_done:
    ret

.section .bss
.align 8
buf:
    .zero   65536
decbuf:
    .zero   24
outbuf:
    .zero   32
