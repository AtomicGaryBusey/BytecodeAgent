	.global _start
	.text
	.align 2

	// constants
	.equ SYS_READ, 63
	.equ SYS_WRITE, 64
	.equ SYS_EXIT, 93
	.equ BUFSIZE, 65536

_start:
	// total length = 0
	mov	x19, #0                 // total_len
	adr	x20, buf                // buffer base address

read_loop:
	// x0 = fd 0
	mov	x0, #0
	// x1 = buffer + total_len
	add	x1, x20, x19
	// x2 = max count
	mov	x2, #BUFSIZE
	// syscall
	mov	x8, #SYS_READ
	svc	#0
	// result in x0
	cmp	x0, #0
	beq	read_done
	add	x19, x19, x0            // total_len += nread
	b	read_loop

read_done:
	// x19 = total_len
	mov	x21, #0                 // offset = 0

row_loop:
	cmp	x21, x19
	bge	rows_done

	// remaining = total_len - offset
	sub	x22, x19, x21
	// row_len = min(16, remaining)
	cmp	x22, #16
	csel	x23, x22, #16, lt      // x23 = row_len
