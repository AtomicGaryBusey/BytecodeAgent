.globl _start
.text
.p2align 2

.equ BUF_SIZE, 65536
.equ STDIN, 0
.equ STDOUT, 1
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // zero counters
    mov     x19, #0          // lines
    mov     x20, #0          // words
    mov     x21, #0          // bytes
    mov     x6, #0           // in_word flag = false

    // buffer addresses
    adrp    x22, input_buf
    add     x22, x22, :lo12:input_buf   // x22 = input buffer
    adrp    x23, out_buf
    add     x23, x23, :lo12:out_buf     // x23 = output buffer (base)

read_loop:
    // sys_read(stdin, buf, BUF_SIZE)
    mov     x8, #SYS_READ
    mov     x0, #STDIN
    mov     x1, x22
    mov     x2, #BUF_SIZE
    svc     #0
    cbz     x0, after_read          // EOF when read returns 0

    // add to total byte count
    add     x21, x21, x0

    // set up pointers for processing this chunk
    mov     x27, x0                 // remaining byte count
    mov     x24, x22                // pointer to current byte

process_byte:
    cbz     x27, read_loop          // if no more bytes, read next chunk

    ldrb    w25, [x24], #1          // load byte, post-increment
    subs    x27, x27, #1

    // line count (newline = 0x0A)
    cmp     w25, #0x0A
    bne     not_newline
    add     x19, x19, #1
not_newline:

    // whitespace test
    cmp     w25, #0x20          // space
    beq     is_ws
    cmp     w25, #0x09          // tab
    beq     is_ws
    cmp     w25, #0x0A          // newline
    beq     is_ws
    cmp     w25, #0x0B          // vertical tab
    beq     is_ws
    cmp     w25, #0x0C          // form feed
    beq     is_ws
    cmp     w25, #0x0D          // carriage return
    beq     is_ws
    b       not_ws

is_ws:
    // whitespace => not inside a word
    mov     x6, #0
    b       not_ws_done

not_ws:
    // non‑whitespace
    cbz     x6, new_word
    b       not_ws_done

new_word:
    add     x20, x20, #1          // increment word count
    mov     x6, #1               // now inside a word

not_ws_done:
    b       process_byte

after_read:
    // x23 points to start of output buffer
    mov     x24, x23              // x24 = output pointer (will be advanced)

    // -----------------
    // convert lines (x19)
    // -----------------
    mov     x0, x19
    mov     x12, #10              // divisor constant
    sub     sp, sp, #32
    mov     x9, sp                // save base of temp area

conv_loop_lines:
    cbz     x0, conv0_lines
conv_loop_body_lines:
    udiv    x11, x0, x12          // x11 = x0 / 10
    mul     x13, x11, x12         // x13 = (x0/10)*10
    sub     x14, x0, x13          // remainder
    add     w14, w14, #'0'
    strb    w14, [sp], #1
    mov     x0, x11
    cbnz    x0, conv_loop_body_lines
    b       conv_done_lines

conv0_lines:
    mov     w14, #'0'
    strb    w14, [sp], #1

conv_done_lines:
    // copy digits back to output buffer
conv_copy_lines:
    cmp     sp, x9
    beq     conv_copy_done_lines
    sub     sp, sp, #1
    ldrb    w14, [sp]
    strb    w14, [x24], #1
    b       conv_copy_lines
conv_copy_done_lines:
    add     sp, sp, #32

    // space after first number
    mov     w14, #' '
    strb    w14, [x24], #1

    // -----------------
    // convert words (x20)
    // -----------------
    mov     x0, x20
    mov     x12, #10
    sub     sp, sp, #32
    mov     x9, sp

conv_loop_words:
    cbz     x0, conv0_words
conv_loop_body_words:
    udiv    x11, x0, x12
    mul     x13, x11, x12
    sub     x14, x0, x13
    add     w14, w14, #'0'
    strb    w14, [sp], #1
    mov     x0, x11
    cbnz    x0, conv_loop_body_words
    b       conv_done_words

conv0_words:
    mov     w14, #'0'
    strb    w14, [sp], #1

conv_done_words:
conv_copy_words:
    cmp     sp, x9
    beq     conv_copy_done_words
    sub     sp, sp, #1
    ldrb    w14, [sp]
    strb    w14, [x24], #1
    b       conv_copy_words
conv_copy_done_words:
    add     sp, sp, #32

    // space after second number
    mov     w14, #' '
    strb    w14, [x24], #1

    // -----------------
    // convert bytes (x21)
    // -----------------
    mov     x0, x21
    mov     x12, #10
    sub     sp, sp, #32
    mov     x9, sp

conv_loop_bytes:
    cbz     x0, conv0_bytes
conv_loop_body_bytes:
    udiv    x11, x0, x12
    mul     x13, x11, x12
    sub     x14, x0, x13
    add     w14, w14, #'0'
    strb    w14, [sp], #1
    mov     x0, x11
    cbnz    x0, conv_loop_body_bytes
    b       conv_done_bytes

conv0_bytes:
    mov     w14, #'0'
    strb    w14, [sp], #1

conv_done_bytes:
conv_copy_bytes:
    cmp     sp, x9
    beq     conv_copy_done_bytes
    sub     sp, sp, #1
    ldrb    w14, [sp]
    strb    w14, [x24], #1
    b       conv_copy_bytes
conv_copy_done_bytes:
    add     sp, sp, #32

    // newline after third number
    mov     w14, #10
    strb    w14, [x24], #1

    // compute length of output string
    sub     x25, x24, x23          // x25 = length

    // write result to stdout
    mov     x8, #SYS_WRITE
    mov     x0, #STDOUT
    mov     x1, x23
    mov     x2, x25
    svc     #0

    // exit with status 0
    mov     x8, #SYS_EXIT
    mov     x0, #0
    svc     #0

.section .bss
.p2align 3
input_buf:
    .space BUF_SIZE
out_buf:
    .space 32

.end
