.global _start
.text
.p2align 2

// syscall numbers
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // read input into buffer
    adr     x10, buf          // x10 = current write pointer
    mov     x11, #0           // total length = 0

read_loop:
    mov     x0, #0            // fd = stdin
    mov     x2, #65536
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done     // x0 == 0 => EOF
    add     x11, x11, x0      // total += bytes read
    add     x10, x10, x0      // advance write pointer
    b       read_loop

read_done:
    // x9 = total length
    mov     x9, x11
    // x12 = offset = 0
    mov     x12, #0
    // x13 = buffer start address
    adr     x13, buf

dump_loop:
    cmp     x12, x9
    b.ge    dump_done

    // remaining = total - offset
    sub     x14, x9, x12
    // n = min(remaining, 16)
    mov     x15, #16
    cmp     x14, #16
    b.lt    set_n
    b       n_ready
set_n:
    mov     x15, x14
n_ready:

    // line buffer pointer
    adr     x20, linebuf
    mov     x21, x20          // current write ptr = start

    // ----- print offset (8 hex digits) -----
    mov     x22, x12          // value to print
    mov     x23, #28          // shift amount

offset_hex_loop:
    lsr     x24, x22, x23
    and     x24, x24, #0xf
    adr     x25, hex_digits
    add     x25, x25, x24, uxtw
    ldrb    w26, [x25]
    strb    w26, [x21], #1
    subs    x23, x23, #4
    b.ge    offset_hex_loop

    // two spaces after offset
    mov     w26, #' '
    strb    w26, [x21], #1
    strb    w26, [x21], #1

    // ----- print byte slots -----
    mov     x27, #0           // i = 0
byte_loop:
    cmp     x27, #16
    b.eq    after_bytes

    cmp     x27, x15
    b.ge    missing_byte

    // have a byte
    add     x28, x13, x12
    add     x28, x28, x27
    ldrb    w29, [x28]

    // high nibble
    lsr     w30, w29, #4
    and     w30, w30, #0xf
    // low nibble
    and     w31, w29, #0xf

    adr     x5, hex_digits
    add     x5, x5, x30, uxtw
    ldrb    w6, [x5]
    strb    w6, [x21], #1

    adr     x5, hex_digits
    add     x5, x5, x31, uxtw
    ldrb    w6, [x5]
    strb    w6, [x21], #1

    // space after byte
    mov     w6, #' '
    strb    w6, [x21], #1
    b       after_byte

missing_byte:
    // three spaces
    mov     w6, #' '
    strb    w6, [x21], #1
    strb    w6, [x21], #1
    strb    w6, [x21], #1

after_byte:
    // extra space after byte 7
    cmp     x27, #7
    b.ne    no_extra_space
    mov     w6, #' '
    strb    w6, [x21], #1
no_extra_space:
    add     x27, x27, #1
    b       byte_loop

after_bytes:
    // one space before '|'
    mov     w6, #' '
    strb    w6, [x21], #1
    // pipe
    mov     w6, #'|'
    strb    w6, [x21], #1

    // ----- ASCII representation -----
    mov     x28, #0           // i = 0
ascii_loop:
    cmp     x28, x15
    b.ge    after_ascii
    add     x29, x13, x12
    add     x29, x29, x28
    ldrb    w30, [x29]

    // printable check 0x20 <= w30 <= 0x7e
    cmp     w30, #0x20
    b.lt    ascii_dot
    cmp     w30, #0x7e
    b.gt    ascii_dot
    // printable
    strb    w30, [x21], #1
    b       ascii_next
ascii_dot:
    mov     w30, #'.'
    strb    w30, [x21], #1
ascii_next:
    add     x28, x28, #1
    b       ascii_loop

after_ascii:
    // closing pipe and newline
    mov     w6, #'|'
    strb    w6, [x21], #1
    mov     w6, #0x0a
    strb    w6, [x21], #1

    // write the line
    sub     x2, x21, x20      // length = current - start
    mov     x0, #1            // stdout
    mov     x1, x20
    mov     x8, #SYS_WRITE
    svc     #0

    // advance offset
    add     x12, x12, #16
    b       dump_loop

dump_done:
    // ----- print total length line -----
    adr     x20, linebuf
    mov     x21, x20

    // total length hex (8 digits)
    mov     x22, x9           // value
    mov     x23, #28

total_hex_loop:
    lsr     x24, x22, x23
    and     x24, x24, #0xf
    adr     x25, hex_digits
    add     x25, x25, x24, uxtw
    ldrb    w26, [x25]
    strb    w26, [x21], #1
    subs    x23, x23, #4
    b.ge    total_hex_loop

    // newline
    mov     w26, #0x0a
    strb    w26, [x21], #1

    // write total line
    sub     x2, x21, x20
    mov     x0, #1
    mov     x1, x20
    mov     x8, #SYS_WRITE
    svc     #0

    // exit
    mov     x0, #0
    mov     x8, #SYS_EXIT
    svc     #0

.section .rodata
.p2align 2
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.p2align 3
buf:
    .zero   65536
linebuf:
    .zero   96
