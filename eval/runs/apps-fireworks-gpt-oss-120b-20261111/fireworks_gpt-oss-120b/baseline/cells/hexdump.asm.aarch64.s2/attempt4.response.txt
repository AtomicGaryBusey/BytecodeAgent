.global _start
.text
.p2align 2

// syscall numbers
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // read input into buffer
    adr     x10, buf          // current write pointer
    mov     x11, #0           // total length = 0

read_loop:
    mov     x0, #0            // fd = stdin
    mov     x1, x10           // buffer
    mov     x2, #65536
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done     // x0 == 0 => EOF
    add     x11, x11, x0      // total += bytes read
    add     x10, x10, x0      // advance write pointer
    b       read_loop

read_done:
    // x11 = total length
    mov     x12, #0           // offset = 0
    adr     x13, buf          // buffer base address

dump_loop:
    cmp     x12, x11
    b.ge    dump_done

    // remaining = total - offset
    sub     x14, x11, x12
    // n = min(remaining, 16)
    mov     x15, #16
    cmp     x14, #16
    csel    x15, x14, x15, lt // if remaining < 16 then n = remaining

    // line buffer pointer
    adr     x16, linebuf
    mov     x17, x16          // current output pointer

    // ----- print offset (8 hex digits) -----
    mov     x18, x12          // value to print
    mov     x19, #28          // shift amount

offset_hex_loop:
    lsr     x20, x18, x19
    and     x20, x20, #0xf
    adr     x21, hex_digits
    add     x21, x21, w20, uxtw
    ldrb    w22, [x21]
    strb    w22, [x17], #1
    sub     x19, x19, #4
    cmp     x19, #0
    b.ge    offset_hex_loop

    // two spaces after offset
    mov     w22, #' '
    strb    w22, [x17], #1
    strb    w22, [x17], #1

    // ----- print byte slots -----
    mov     x23, #0           // i = 0
byte_loop:
    cmp     x23, #16
    b.eq    after_bytes

    cmp     x23, x15
    b.ge    missing_byte

    // have a byte
    add     x24, x13, x12
    add     x24, x24, x23
    ldrb    w25, [x24]        // byte value

    // high nibble
    lsr     w26, w25, #4
    and     w26, w26, #0xf
    // low nibble
    and     w27, w25, #0xf

    adr     x28, hex_digits
    add     x28, x28, w26, uxtw
    ldrb    w29, [x28]
    strb    w29, [x17], #1

    adr     x28, hex_digits
    add     x28, x28, w27, uxtw
    ldrb    w29, [x28]
    strb    w29, [x17], #1

    // space after byte
    mov     w29, #' '
    strb    w29, [x17], #1
    b       after_byte

missing_byte:
    // three spaces
    mov     w29, #' '
    strb    w29, [x17], #1
    strb    w29, [x17], #1
    strb    w29, [x17], #1

after_byte:
    // extra space after byte 7
    cmp     x23, #7
    b.ne    no_extra_space
    mov     w29, #' '
    strb    w29, [x17], #1
no_extra_space:
    add     x23, x23, #1
    b       byte_loop

after_bytes:
    // one space before '|'
    mov     w29, #' '
    strb    w29, [x17], #1
    // pipe
    mov     w29, #'|'
    strb    w29, [x17], #1

    // ----- ASCII representation -----
    mov     x24, #0           // i = 0
ascii_loop:
    cmp     x24, x15
    b.ge    after_ascii
    add     x25, x13, x12
    add     x25, x25, x24
    ldrb    w26, [x25]

    // printable check 0x20 <= w26 <= 0x7e
    cmp     w26, #0x20
    b.lt    ascii_dot
    cmp     w26, #0x7e
    b.gt    ascii_dot
    // printable
    strb    w26, [x17], #1
    b       ascii_next
ascii_dot:
    mov     w26, #'.'
    strb    w26, [x17], #1
ascii_next:
    add     x24, x24, #1
    b       ascii_loop

after_ascii:
    // closing pipe and newline
    mov     w29, #'|'
    strb    w29, [x17], #1
    mov     w29, #0x0a
    strb    w29, [x17], #1

    // write the line
    sub     x2, x17, x16      // length = current - start
    mov     x0, #1            // stdout
    mov     x1, x16
    mov     x8, #SYS_WRITE
    svc     #0

    // advance offset
    add     x12, x12, #16
    b       dump_loop

dump_done:
    // ----- print total length line -----
    adr     x16, linebuf
    mov     x17, x16

    // total length hex (8 digits)
    mov     x18, x11          // value
    mov     x19, #28

total_hex_loop:
    lsr     x20, x18, x19
    and     x20, x20, #0xf
    adr     x21, hex_digits
    add     x21, x21, w20, uxtw
    ldrb    w22, [x21]
    strb    w22, [x17], #1
    sub     x19, x19, #4
    cmp     x19, #0
    b.ge    total_hex_loop

    // newline
    mov     w22, #0x0a
    strb    w22, [x17], #1

    // write total line
    sub     x2, x17, x16
    mov     x0, #1
    mov     x1, x16
    mov     x8, #SYS_WRITE
    svc     #0

    // exit
    mov     x0, #0
    mov     x8, #SYS_EXIT
    svc     #0

.section .rodata
.p2align 2
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.p2align 3
buf:
    .zero   65536
linebuf:
    .zero   96
