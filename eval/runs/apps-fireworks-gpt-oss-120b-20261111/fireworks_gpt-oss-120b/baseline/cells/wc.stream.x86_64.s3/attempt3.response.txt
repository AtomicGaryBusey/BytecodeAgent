.globl _start
.equ BUF_SIZE, 65536

.section .bss
.align 16
buf:
    .zero BUF_SIZE
outbuf:
    .zero 64
numbuf:
    .zero 64

.section .text
_start:
    xor     rbx, rbx            # lines = 0
    xor     r12, r12            # words = 0
    xor     r13, r13            # bytes = 0
    xor     r11, r11            # in_word = 0

read_loop:
    mov     rax, 0              # sys_read
    mov     rdi, 0              # fd = stdin
    lea     rsi, [rip + buf]    # buffer address
    mov     rdx, BUF_SIZE
    syscall
    test    rax, rax
    je      done_read           # EOF
    add     r13, rax            # total bytes
    mov     r15, rax            # n = bytes read
    xor     r8, r8              # i = 0
    lea     rsi, [rip + buf]    # buffer pointer

proc_loop:
    cmp     r8, r15
    jae     after_proc
    mov     al, byte [rsi + r8] # load byte
