.global _start
.equ BUF_SIZE, 65536

.section .bss
buf:
    .zero BUF_SIZE
outbuf:
    .zero 64

.section .text
_start:
    xor     rbx, rbx            # lines = 0
    xor     rcx, rcx            # words = 0
    xor     rdx, rdx            # bytes = 0
    xor     rsi, rsi            # in_word = 0
    lea     r12, [rip + buf]    # buffer pointer

read_loop:
    mov     rax, 0              # sys_read
    mov     rdi, 0              # fd = stdin
    mov     rsi, r12            # buf
    mov     rdx, BUF_SIZE
    syscall
    test    rax, rax
    je      done_read           # EOF
    mov     r13, rax            # n = bytes read
    xor     r14, r14            # i = 0

process_loop:
    cmp     r14, r13
    jae     after_process
    movzx   eax, byte ptr [r12 + r14]
    inc     rdx                 # total_bytes++
    cmp     al, 0x0A
    jne     not_nl
    inc     rbx                 # lines++
not_nl:
    # whitespace test
    mov     r8b, 0
    cmp     al, 0x20
    je      is_ws
    cmp     al, 0x09
    je      is_ws
    cmp     al, 0x0A
    je      is_ws
    cmp     al, 0x0B
    je      is_ws
    cmp     al, 0x0C
    je      is_ws
    cmp     al, 0x0D
    je      is_ws
    jmp     not_ws
is_ws:
    mov     r8b, 1
not_ws:
    cmp     r8b, 0
    jne     ws_true
    # not whitespace
    cmp     rsi, 0
    jne     already_word
    inc     rcx                 # words++
    mov     rsi, 1              # in_word = 1
    jmp     after_word
already_word:
    jmp     after_word
ws_true:
    xor     rsi, rsi            # in_word = 0
after_word:
    inc     r14
    jmp     process_loop

after_process:
    jmp     read_loop

done_read:
    # output: lines words bytes\n
    lea     rdi, [rip + outbuf]     # output pointer

    mov     rax, rbx                # lines
    call    utoa
    mov     byte ptr [rdi], ' '
    inc     rdi

    mov     rax, rcx                # words
    call    utoa
    mov     byte ptr [rdi], ' '
    inc     rdi

    mov     rax, rdx                # bytes
    call    utoa
    mov     byte ptr [rdi], 0x0A
    inc     rdi

    lea     rsi, [rip + outbuf]     # buffer start
    mov     rdx, rdi
    sub     rdx, rsi                # length
    mov     rax, 1                  # sys_write
    mov     rdi, 1                  # fd = stdout
    syscall

    mov     rax, 60                 # sys_exit
    xor     rdi, rdi                # status 0
    syscall

# ------------------------------------------------------------
# utoa: convert unsigned integer in rax to decimal ASCII.
# Input:  rax = value, rdi = pointer to output buffer (where to write)
# Output: rdi advanced past written characters, registers clobbered.
# ------------------------------------------------------------
utoa:
    push    rbx
    push    rcx
    push    rdx
    push    r8
    push    r9
    push    r10
    push    r11
    sub     rsp, 32                 # temporary space (max 20 digits)

    mov     r10, 10                 # divisor
    xor     rcx, rcx                # digit count

    cmp     rax, 0
    jne     utoa_loop_start
    # value is zero
    mov     byte ptr [rdi], '0'
    inc     rdi
    jmp     utoa_done

utoa_loop_start:
    xor     rdx, rdx
    div     r10                     # rax = rax/10, rdx = rax%10
    add     dl, '0'
    mov     byte ptr [rsp + rcx], dl
    inc     rcx
    test    rax, rax
    jne     utoa_loop_start

    # rcx = number of digits, copy in reverse
    dec     rcx                     # index of last digit
utoa_copy:
    mov     al, byte ptr [rsp + rcx]
    mov     byte ptr [rdi], al
    inc     rdi
    dec     rcx
    jns     utoa_copy               # while rcx >= 0

utoa_done:
    add     rsp, 32
    pop     r11
    pop     r10
    pop     r9
    pop     r8
    pop     rdx
    pop     rcx
    pop     rbx
    ret
