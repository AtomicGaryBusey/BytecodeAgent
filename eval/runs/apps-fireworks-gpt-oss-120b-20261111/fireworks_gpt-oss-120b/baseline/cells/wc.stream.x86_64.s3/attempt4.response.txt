.globl _start
.equ BUF_SIZE, 65536

.section .bss
.align 16
buf:
    .zero BUF_SIZE
outbuf:
    .zero 64
tmpdigits:
    .space 32

.section .text
_start:
    xor     rbx, rbx            # lines = 0
    xor     r12, r12            # words = 0
    xor     r13, r13            # bytes = 0
    xor     r11, r11            # in_word = 0

read_loop:
    mov     rax, 0              # sys_read
    xor     rdi, rdi            # fd = 0 (stdin)
    lea     rsi, [rip + buf]    # buffer address
    mov     rdx, BUF_SIZE
    syscall
    test    rax, rax
    je      eof                 # EOF
    add     r13, rax            # total bytes += nread
    mov     r15, rax            # r15 = nread
    xor     r8, r8              # i = 0
    lea     rsi, [rip + buf]    # buffer pointer

proc_loop:
    cmp     r8, r15
    jae     after_proc
    mov     al, byte ptr [rsi + r8]   # load byte
    cmp     al, 0x0A
    jne     not_newline
    inc     rbx                      # line count
not_newline:
    # check whitespace
    mov     dl, al
    cmp     dl, 0x20
    je      is_ws
    cmp     dl, 0x09
    je      is_ws
    cmp     dl, 0x0A
    je      is_ws
    cmp     dl, 0x0B
    je      is_ws
    cmp     dl, 0x0C
    je      is_ws
    cmp     dl, 0x0D
    je      is_ws
    # not whitespace
    mov     r11, 1
    jmp     next_byte
is_ws:
    test    r11, r11
    je      next_byte
    inc     r12          # word count
    xor     r11, r11
next_byte:
    inc     r8
    jmp     proc_loop
after_proc:
    jmp     read_loop

eof:
    cmp     r11, 0
    je      after_word
    inc     r12            # count last word if any
after_word:
    # format output into outbuf
    lea     rdi, [rip + outbuf]   # output pointer

    mov     rax, rbx
    call    fmt_uint

    mov     byte ptr [rdi], 0x20
    inc     rdi

    mov     rax, r12
    call    fmt_uint

    mov     byte ptr [rdi], 0x20
    inc     rdi

    mov     rax, r13
    call    fmt_uint

    mov     byte ptr [rdi], 0x0A
    inc     rdi

    # compute length = rdi - outbuf
    lea     rsi, [rip + outbuf]
    mov     rcx, rdi
    sub     rcx, rsi          # rcx = length

    # write output
    mov     rax, 1            # sys_write
    mov     rdi, 1            # fd = stdout
    lea     rsi, [rip + outbuf]
    mov     rdx, rcx
    syscall

    # exit with status 0
    mov     rax, 60           # sys_exit
    xor     rdi, rdi
    syscall

# ------------------------------------------------------------
# fmt_uint: rax = unsigned integer, rdi = destination pointer
# writes decimal representation, returns with rdi advanced
fmt_uint:
    push    rbx
    push    rcx
    push    rdx
    push    rsi

    mov     rcx, 0                # digit count
    lea     rsi, [rip + tmpdigits + 31]   # point to end of temp buffer

    cmp     rax, 0
    jne     .conv_loop
    mov     byte ptr [rsi], '0'
    dec     rsi
    inc     rcx
    jmp     .copy_digits
.conv_loop:
    xor     rdx, rdx
    mov     r8, 10
    div     r8                     # rax = quotient, rdx = remainder
    add     dl, '0'
    mov     byte ptr [rsi], dl
    dec     rsi
    inc     rcx
    cmp     rax, 0
    jne     .conv_loop
.copy_digits:
    inc     rsi                    # rsi -> first digit
    mov     rbx, rcx               # copy counter
.copy_loop2:
    test    rbx, rbx
    jz      .done_fmt
    mov     al, byte ptr [rsi]
    mov     byte ptr [rdi], al
    inc     rsi
    inc     rdi
    dec     rbx
    jmp     .copy_loop2
.done_fmt:
    pop     rsi
    pop     rdx
    pop     rcx
    pop     rbx
    ret
