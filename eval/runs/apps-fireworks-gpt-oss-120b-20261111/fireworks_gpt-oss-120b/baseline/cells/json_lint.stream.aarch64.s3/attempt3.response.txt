	.global _start
	.text
	.align 2

	// syscall numbers
	.equ SYS_READ,   63
	.equ SYS_WRITE,  64
	.equ SYS_EXIT,   93

	// file descriptors
	.equ FD_STDIN,   0
	.equ FD_STDOUT,  1

	// limits
	.equ INPUT_MAX,  65536
	.equ DEC_BUF_SIZE, 20
	.equ MAX_NESTING, 1000

	// messages
	.section .rodata
msg_ok:
	.asciz "ok\n"
msg_ok_len = . - msg_ok

msg_error_prefix:
	.asciz "error at "
msg_error_prefix_len = . - msg_error_prefix

msg_newline:
	.asciz "\n"
msg_newline_len = . - msg_newline

	.bss
	.align 8
input_buf:
	.zero INPUT_MAX
input_len:
	.zero 8
error_offset:
	.zero 8
dec_buf:
	.zero DEC_BUF_SIZE

	.text

// entry point
_start:
	// x19 = current offset (ptr) = 0
	mov		x19, #0

	// x20 = &input_buf
	adrp	x20, input_buf
	add		x20, x20, :lo12:input_buf

	// x21 = length = 0
	mov		x21, #0

read_loop:
	// calculate remaining space = INPUT_MAX - x21
	// load constant INPUT_MAX into x3
	movz	x3, #0x0, LSL #0
	movk	x3, #0x1, LSL #16	// x3 = 0x10000

	sub		x4, x3, x21		// x4 = count

	// address = input_buf + x21
	add		x5, x20, x21

	// sys_read(fd=0, buf=x5, count=x4)
	mov		x0, FD_STDIN
	mov		x1, x5
	mov		x2, x4
	mov		x8, SYS_READ
	svc		0

	// result in x0
	cmp		x0, #0
	beq		read_done
	// add to length
	add		x21, x21, x0
	b		read_loop

read_done:
	// start parsing
	bl		skip_ws
	bl		parse_value
	bl		skip_ws

	// after value, check if we consumed all input
	cmp		x19, x21
	bne		parse_extra_error

	// success: output "ok\n"
	adrp	x0, msg_ok
	add		x0, x0, :lo12:msg_ok
	ldr		x1, =msg_ok_len	// will be resolved by assembler as absolute address -> not allowed
