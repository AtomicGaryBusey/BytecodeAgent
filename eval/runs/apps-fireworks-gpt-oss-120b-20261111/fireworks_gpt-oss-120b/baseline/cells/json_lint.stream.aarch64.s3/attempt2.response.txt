	.global _start
	.text
	.align 2

// syscall numbers
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

// file descriptors
.equ FD_STDIN, 0
.equ FD_STDOUT, 1

// buffer sizes
.equ INPUT_MAX, 65536
.equ DEC_BUF_SIZE, 20

// message strings
	.section .rodata
msg_ok:
	.asciz "ok\n"
msg_error_prefix:
	.asciz "error at "

// bss area
	.bss
	.align 8
input_buf:
	.zero INPUT_MAX
input_len:
	.zero 8
error_offset:
	.zero 8
dec_buf:
	.zero DEC_BUF_SIZE

// text start
	.text
_start:
	// x19 = current offset, start 0
	mov x19, #0
	// x20 = input buffer address
	adrp x20, input_buf
	add  x20, x20, :lo12:input_buf
	// x21 = input length (initial 0)
	mov x21, #0

read_loop:
	// sys_read(fd=0, buf = input_buf + input_len, count = INPUT_MAX - input_len)
	mov x0, FD_STDIN
	adrp x2, input_buf
	add  x2, x2, :lo12:input_buf
	add  x2, x2, x21          // buf = base + len
	mov x8, SYS_READ
	// count = INPUT_MAX - input_len
	ldr x3, =INPUT_MAX
