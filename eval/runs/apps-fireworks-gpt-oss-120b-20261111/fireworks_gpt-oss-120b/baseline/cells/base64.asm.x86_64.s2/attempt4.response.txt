.globl _start
.section .bss
    .align 64
input_buf:
    .zero 49152
output_buf:
    .zero 70000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # rbx = input buffer address
    lea rbx, [rip + input_buf]
    xor r12d, r12d                 # total bytes read = 0

read_loop:
    mov rax, 0                     # sys_read
    mov rdi, 0                     # fd = stdin
    lea rsi, [rbx + r12]           # buf = input_buf + total_read
    mov rdx, 49152
    sub rdx, r12                   # remaining space
    syscall
    test rax, rax
    jz read_done
    add r12, rax                   # total_read += bytes_read
    cmp r12, 49152
    jae read_done
    jmp read_loop

read_done:
    test r12d, r12d
    jz exit_success                # empty input: no output

    # rsi = input pointer, rcx = remaining input length
    mov rsi, rbx
    mov rcx, r12
    lea r8, [rip + output_buf]     # output base
    mov rdi, r8                    # output pointer
    xor r9d, r9d                   # line position (0..75)

    # table base address
    lea r10, [rip + b64_table]

full_loop:
    cmp rcx, 3
    jb tail_encode

    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx edx, byte ptr [rsi+2]

    shl eax, 16
    shl ebx, 8
    or  eax, ebx
    or  eax, edx                   # eax = 24-bit chunk

    # idx0
    mov r11, rax
    shr r11, 18
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl0
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl0:

    # idx1
    mov r11, rax
    shr r11, 12
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl1:

    # idx2
    mov r11, rax
    shr r11, 6
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl2:

    # idx3
    mov r11, rax
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl3:

    add rsi, 3
    sub rcx, 3
    jmp full_loop

tail_encode:
    cmp rcx, 0
    je final_newline_check

    cmp rcx, 1
    je one_byte_case

    # two-byte case
    movzx eax, byte ptr [rsi]          # first byte
    movzx ebx, byte ptr [rsi+1]        # second byte
    shl eax, 16
    shl ebx, 8
    or  eax, ebx                       # 24-bit value with low 8 bits zero

    # idx0
    mov r11, rax
    shr r11, 18
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl4:

    # idx1
    mov r11, rax
    shr r11, 12
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl5
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl5:

    # idx2
    mov r11, rax
    shr r11, 6
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl6
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl6:

    # padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl7
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl7:

    jmp final_newline_check

one_byte_case:
    movzx eax, byte ptr [rsi]          # first byte
    shl eax, 16

    # idx0
    mov r11, rax
    shr r11, 18
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl8
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl8:

    # idx1
    mov r11, rax
    shr r11, 12
    and r11, 0x3F
    mov al, byte ptr [r10 + r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl9
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl9:

    # first padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl10
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl10:

    # second padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r9d
    cmp r9d, 76
    jne .no_nl11
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9d, r9d
.no_nl11:

    jmp final_newline_check

final_newline_check:
    test r9d, r9d
    je after_newline
    mov byte ptr [rdi], 0x0A
    inc rdi
after_newline:
    # rdx = output length
    mov rdx, rdi
    sub rdx, r8

    # write output
    mov rax, 1                     # sys_write
    mov rdi, 1                     # stdout
    mov rsi, r8
    syscall

    # exit
    mov rax, 60
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
