.globl _start
.section .bss
    .align 64
input_buf:
    .zero 49152
output_buf:
    .zero 70000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # load input buffer address into rbx
    lea rbx, [rip + input_buf]
    xor r12, r12                 # total bytes read = 0

read_loop:
    mov rax, 0                   # sys_read
    mov rdi, 0                   # stdin
    lea rsi, [rbx + r12]         # buffer pointer
    mov rdx, 49152
    sub rdx, r12                 # remaining space
    syscall
    test rax, rax
    jz read_done
    add r12, rax                 # accumulate bytes read
    cmp r12, 49152
    jae read_done
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success              # empty input: nothing to output

    # rsi = input pointer, rdx = remaining input length
    mov rsi, rbx
    mov rdx, r12
    lea r8, [rip + output_buf]   # output base
    mov rdi, r8                  # output pointer
    xor r9b, r9b                 # line length counter (0..75)

full_loop:
    cmp rdx, 3
    jb tail_encode

    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]

    shl eax, 16
    shl ebx, 8
    or  eax, ebx
    or  eax, ecx                 # eax = 24-bit chunk

    # idx0
    mov ebx, eax
    shr ebx, 18
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # idx1
    mov ebx, eax
    shr ebx, 12
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # idx2
    mov ebx, eax
    shr ebx, 6
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # idx3
    mov ebx, eax
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:

    add rsi, 3
    sub rdx, 3
    jmp full_loop

tail_encode:
    cmp rdx, 0
    je final_newline_check

    cmp rdx, 1
    je one_byte_case

    # two-byte case
    movzx eax, byte ptr [rsi]        # first byte
    movzx ebx, byte ptr [rsi+1]      # second byte
    shl eax, 16
    shl ebx, 8
    or  eax, ebx                     # 16-bit data in high bits

    # idx0
    mov ebx, eax
    shr ebx, 18
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # idx1
    mov ebx, eax
    shr ebx, 12
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # idx2
    mov ebx, eax
    shr ebx, 6
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    jmp final_newline_check

one_byte_case:
    movzx eax, byte ptr [rsi]        # first byte
    shl eax, 16

    # idx0
    mov ebx, eax
    shr ebx, 18
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # idx1
    mov ebx, eax
    shr ebx, 12
    and ebx, 0x3F
    movzx ebx, byte ptr [rip + b64_table + rbx]
    mov [rdi], bl
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # first padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    # second padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r9b
    cmp r9b, 76
    jne 0f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r9b, r9b
0:
    jmp final_newline_check

final_newline_check:
    # if any chars were output, ensure trailing newline unless already at line start
    test r9b, r9b
    je after_newline
    mov byte ptr [rdi], 0x0A
    inc rdi
after_newline:
    # compute output length in rdx
    mov rdx, rdi
    sub rdx, r8

    # write output
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    mov rsi, r8
    syscall

    # exit
    mov rax, 60
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
