.globl _start
.section .bss
    .align 64
input_buf:
    .zero 49152
output_buf:
    .zero 70000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # rbx = input buffer base
    lea rbx, [rip + input_buf]
    xor r9, r9                # r9 = total bytes read

read_loop:
    mov rax, 0                # sys_read
    mov rdi, 0                # stdin
    mov rsi, rbx
    add rsi, r9               # buf = input_buf + r9
    mov rdx, 49152
    sub rdx, r9               # count = remaining space
    syscall
    test rax, rax
    jz read_done              # rax == 0 -> EOF
    add r9, rax               # total += bytes read
    cmp r9, 49152
    jae read_done
    jmp read_loop
read_done:
    test r9, r9
    jz exit_success           # empty input, exit

    # rsi = input pointer
    mov rsi, rbx
    # rcx = remaining input length
    mov rcx, r9
    # r8 = output buffer base
    lea r8, [rip + output_buf]
    mov rdi, r8               # output pointer
    xor r10d, r10d            # line_len = 0

# ----- encode full 3-byte groups -----
full_loop:
    cmp rcx, 3
    jl tail_encode

    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx edx, byte ptr [rsi+2]

    shl eax, 16
    shl ebx, 8
    or  eax, ebx
    or  eax, edx               # eax now holds 24 bits

    mov r11d, eax
    shr r11d, 18
    and r11d, 0x3F

    mov r12d, eax
    shr r12d, 12
    and r12d, 0x3F

    mov r13d, eax
    shr r13d, 6
    and r13d, 0x3F

    mov r14d, eax
    and r14d, 0x3F

    lea rdx, [rip + b64_table]

    # first character
    movzx ebp, byte ptr [rdx + r11]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl1:

    # second character
    movzx ebp, byte ptr [rdx + r12]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl2:

    # third character
    movzx ebp, byte ptr [rdx + r13]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl3:

    # fourth character
    movzx ebp, byte ptr [rdx + r14]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl4:

    add rsi, 3
    sub rcx, 3
    jmp full_loop

# ----- handle remaining 1 or 2 bytes -----
tail_encode:
    cmp rcx, 0
    je final_newline_check

    # there is 1 or 2 bytes left
    movzx eax, byte ptr [rsi]          # first byte
    mov ebx, 0
    mov edx, 0
    mov ecx, 0
    mov r12d, 0                         # padding count (0,1,2)

    cmp rcx, 2
    jne .one_byte

    # two bytes case
    movzx ebx, byte ptr [rsi+1]        # second byte
    shl eax, 16
    shl ebx, 8
    or  eax, ebx                       # bits in eax (16 bits of data)
    mov r12d, 1                        # one '=' padding later

    # indices
    mov r11d, eax
    shr r11d, 18
    and r11d, 0x3F

    mov r13d, eax
    shr r13d, 12
    and r13d, 0x3F

    mov r14d, eax
    shr r14d, 6
    and r14d, 0x3F

    # output first three chars
    lea rdx, [rip + b64_table]

    movzx ebp, byte ptr [rdx + r11]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_t1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_t1:

    movzx ebp, byte ptr [rdx + r13]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_t2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_t2:

    movzx ebp, byte ptr [rdx + r14]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_t3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_t3:

    # padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_t4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_t4:

    jmp final_newline_check

.one_byte:
    # one byte case
    shl eax, 16                         # bits = b1 << 16
    mov r12d, 2                         # two '=' paddings

    mov r11d, eax
    shr r11d, 18
    and r11d, 0x3F

    mov r13d, eax
    shr r13d, 12
    and r13d, 0x3F

    lea rdx, [rip + b64_table]

    movzx ebp, byte ptr [rdx + r11]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_o1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_o1:

    movzx ebp, byte ptr [rdx + r13]
    mov [rdi], bpl
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_o2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_o2:

    # two paddings
    mov byte ptr [rdi], '='
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_o3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_o3:

    mov byte ptr [rdi], '='
    inc rdi
    inc r10d
    cmp r10d, 76
    jne .no_nl_o4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r10d, r10d
.no_nl_o4:

    jmp final_newline_check

# ----- ensure final newline if needed -----
final_newline_check:
    test r10d, r10d
    je maybe_no_output
    mov byte ptr [rdi], 0x0A
    inc rdi
maybe_no_output:
    # rdx = output length
    mov rdx, rdi
    sub rdx, r8

    # write output
    mov rax, 1               # sys_write
    mov rdi, 1               # stdout
    mov rsi, r8
    syscall

    # exit
    mov rax, 60              # sys_exit
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
