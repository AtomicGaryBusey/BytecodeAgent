.globl _start
.text

# system call numbers
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

# constants
.equ BUFFER_SIZE, 65536
.equ MAX_DEPTH, 1000

_start:
    # rsi = buffer base
    lea rsi, [rip + buffer]

    xor rcx, rcx                # total bytes read = 0
read_loop:
    mov rax, SYS_READ
    mov rdi, 0                  # stdin
    lea rdx, [rip + buffer + rcx]   # dst = buffer + rcx
