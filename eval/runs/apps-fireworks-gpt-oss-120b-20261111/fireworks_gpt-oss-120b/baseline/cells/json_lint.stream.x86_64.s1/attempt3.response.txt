.global _start
.text

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFFER_SIZE, 65536

_start:
    # rsi = address of buffer
    lea rsi, [rip + buffer]

    xor rcx, rcx            # total bytes read = 0
.read_loop:
    mov rax, SYS_READ
    mov rdi, 0
    lea rdx, [rsi + rcx]    # dst = buffer + rcx
    mov rbx, BUFFER_SIZE
    sub rbx, rcx
    mov rdx, rbx
    syscall
    test rax, rax
    jz .read_done
    add rcx, rax
    cmp rcx, BUFFER_SIZE
    jl .read_loop
.read_done:
    mov [rip + input_len], rcx
    mov r8, rcx             # r8 = input_len
    xor rcx, rcx            # rcx = position (offset)
    lea rdi, [rip + buffer] # rdi = pointer to current byte

    call parse_value
    test rax, rax
    jnz .error_output

    call skip_ws
    cmp rcx, r8
    jne .error_output

    # success: write "ok\n"
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

.error_output:
    # set error position if not already set (for extra data case)
    mov [rip + error_pos], rcx

    # write "error at "
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + msg_err_prefix]
    mov rdx, msg_err_prefix_len
    syscall

    # convert error_pos to decimal
    mov rax, [rip + error_pos]
    lea rbx, [rip + numbuf + 20]   # rbx = end of buffer
    mov rcx, rbx

    test rax, rax
    jnz .conv_loop
    dec rcx
    mov byte ptr [rcx], '0'
    jmp .conv_done
.conv_loop:
    xor rdx, rdx
    mov rsi, 10
    div rsi                 # rax = quotient, rdx = remainder
    dec rcx
    add dl, '0'
    mov byte ptr [rcx], dl
    test rax, rax
    jnz .conv_loop
.conv_done:
    mov rdx, rbx
    sub rdx, rcx            # length = rbx - rcx
    mov rsi, rcx
    mov rax, SYS_WRITE
    mov rdi, 1
    syscall

    # write newline
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + msg_newline]
    mov rdx, msg_newline_len
    syscall

    mov rax, SYS_EXIT
    mov rdi, 1
    syscall

# -------------------------------------------------
# Parsing utilities
# -------------------------------------------------
skip_ws:
    cmp rcx, r8
    jae .skip_done
.sw_loop:
    mov al, byte [rdi]
