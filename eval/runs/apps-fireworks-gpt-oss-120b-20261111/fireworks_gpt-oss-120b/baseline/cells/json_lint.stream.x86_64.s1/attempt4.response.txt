.global _start
.text

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFFER_SIZE, 65536

# -------------------------------------------------
# Data
# -------------------------------------------------
.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

msg_err_prefix:
    .ascii "error at "
msg_err_prefix_len = . - msg_err_prefix

msg_newline:
    .ascii "\n"
msg_newline_len = . - msg_newline

# -------------------------------------------------
# BSS
# -------------------------------------------------
.section .bss
.align 8
buffer:          .zero BUFFER_SIZE
input_len:       .quad 0
