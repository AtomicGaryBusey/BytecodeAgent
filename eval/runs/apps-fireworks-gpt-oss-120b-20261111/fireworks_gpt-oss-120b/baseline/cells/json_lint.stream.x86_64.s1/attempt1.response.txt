.globl _start
.text

_start:
    # read input into buffer (max 65536 bytes)
    lea rsi, [rip + buffer]
    xor rcx, rcx                # total length = 0
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add rcx, rax
    jmp read_loop
read_done:
    mov r8, rcx                 # r8 = input length

    xor rdx, rdx                # offset = 0
    call parse_ws               # skip leading ws
    mov rdx, rax                # offset after ws
    call parse_value            # parse the value
    mov rdx, rax                # offset after value
    call parse_ws               # trailing ws
    mov rdx, rax                # final offset

    cmp rdx, r8
    jne error_extra_bytes

    # success: output "ok\n" and exit 0
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    call write_stdout
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

error_extra_bytes:
    # extra non‑whitespace after value
    mov rdx, rax                # offending offset
    jmp parse_error

# -------------------------------
# parse_ws: skip whitespace
# Input: rdx = offset
# Output: rax = new offset
parse_ws:
    mov rcx, rdx
.ws_loop:
    cmp rcx, r8
    jae .ws_done
    mov al, byte [rsi + rcx]
