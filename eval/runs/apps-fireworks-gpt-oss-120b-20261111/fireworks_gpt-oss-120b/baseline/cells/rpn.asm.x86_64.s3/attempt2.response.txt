.globl _start
.equ INPUT_BUF_SIZE, 65536
.equ STACK_MAX, 8192

.section .bss
.align 16
input_buf:
    .zero INPUT_BUF_SIZE
input_len:
    .zero 8
eval_stack:
    .zero STACK_MAX*8
outbuf:
    .zero 32

.section .rodata
err_msg:
    .ascii "error\n"
err_len = . - err_msg

.section .text
_start:
    xor rcx, rcx                    # total bytes read = 0
read_loop:
    mov rdi, 0                      # fd stdin
    lea rsi, [rel input_buf + rcx]
    mov rdx, INPUT_BUF_SIZE
    sub rdx, rcx
    mov rax, 0                      # sys_read
    syscall
    test rax, rax
    jz read_done
    add rcx, rax
    cmp rcx, INPUT_BUF_SIZE
    jb read_loop
read_done:
    mov [rel input_len], rcx

    xor rbx, rbx                    # offset into input_buf

outer_loop:
    mov rax, [rel input_len]
    cmp rbx, rax
    jae exit_prog

    lea rsi, [rel input_buf + rbx]  # line start pointer

    mov rcx, rax
    sub rcx, rbx                    # remaining bytes
    mov rdi, rsi                    # scan pointer
    xor r9d, r9d                    # newline flag = 0
scan_loop:
    test rcx, rcx
    jz no_newline
    mov al, [rdi]
    cmp al, 0x0A
    je have_newline
    inc rdi
    dec rcx
    jmp scan_loop
have_newline:
    mov r9d, 1
    jmp after_scan
no_newline:
    lea rdi, [rsi + rcx]            # end pointer (no newline)
after_scan:
    mov rax, rdi
    sub rax, [rel input_buf]        # offset to line end
    test r9d, r9d
    je .no_nl_inc
    inc rax                         # include newline char
.no_nl_inc:
    mov r10, rax                    # saved next offset

    xor ebp, ebp                    # stack count = 0
    xor r11b, r11b                  # error flag = 0

    mov rdx, rdi                    # line_end pointer

token_loop:
    cmp rsi, rdx
    jae token_done
    mov al, [rsi]
    cmp al, ' '
    je inc_ptr
    cmp al, 9
    je inc_ptr
    jmp token_start
inc_ptr:
    inc rsi
    jmp token_loop

token_start:
    mov al, [rsi]
    cmp al, '-'
    je handle_minus
    cmp al, '0'
    jb invalid_tok
    cmp al, '9'
    ja invalid_tok
    xor r8b, r8b                    # sign = 0
    jmp parse_number

handle_minus:
    lea rcx, [rsi+1]
    cmp rcx, rdx
    jae treat_op_minus
    mov bl, [rcx]
    cmp bl, '0'
    jb treat_op_minus
    cmp bl, '9'
    ja treat_op_minus
    mov r8b, 1                     # sign = 1 (negative number)
    inc rsi
    jmp parse_number

treat_op_minus:
    lea rcx, [rsi+1]
    cmp rcx, rdx
    je op_minus_ok
    mov bl, [rcx]
    cmp bl, ' '
    je op_minus_ok
    cmp bl, 9
    je op_minus_ok
    jmp invalid_tok
op_minus_ok:
    inc rsi
    mov bl, '-'
    jmp process_operator

parse_number:
    xor rax, rax
    xor rcx, rcx                    # digit count
num_loop:
    cmp rsi, rdx
    jae after_number
    mov bl, [rsi]
    cmp bl, '0'
    jb after_number
    cmp bl, '9'
    ja after_number
    inc rcx
    movzx rbx, bl
    sub rbx, '0'
    imul rax, rax, 10
    add rax, rbx
    inc rsi
    jmp num_loop
after_number:
    test rcx, rcx
    jz invalid_tok
    cmp rcx, 19
    jae invalid_tok
    cmp r8b, 0
    je push_number
    neg rax
push_number:
    lea rdi, [rel eval_stack + ebp*8]
    mov [rdi], rax
    inc ebp
    jmp token_loop

invalid_tok:
    mov r11b, 1
    mov rsi, rdx                    # skip to end of line
    jmp token_loop

process_operator:
    cmp ebp, 2
    jb invalid_tok
    dec ebp
    lea rdi, [rel eval_stack + ebp*8]
    mov rcx, [rdi]                 # b
    dec ebp
    lea rdi, [rel eval_stack + ebp*8]
    mov rax, [rdi]                 # a
    cmp bl, '+'
    je op_do_add
    cmp bl, '-'
    je op_do_sub
    cmp bl, '*'
    je op_do_mul
    cmp bl, '/'
    je op_do_div
    cmp bl, '%'
    je op_do_mod
    jmp invalid_tok

op_do_add:
    add rax, rcx
    jmp push_result
op_do_sub:
    sub rax, rcx
    jmp push_result
op_do_mul:
    imul rax, rcx
    jmp push_result
op_do_div:
    test rcx, rcx
    jz invalid_tok
    cmp rax, 0x8000000000000000
    jne .div_ok
    cmp rcx, -1
    je invalid_tok
.div_ok:
    cqo
    idiv rcx
    jmp push_result
op_do_mod:
    test rcx, rcx
    jz invalid_tok
    cmp rax, 0x8000000000000000
    jne .mod_ok
    cmp rcx, -1
    je invalid_tok
.mod_ok:
    cqo
    idiv rcx
    mov rax, rdx
    jmp push_result
push_result:
    lea rdi, [rel eval_stack + ebp*8]
    mov [rdi], rax
    inc ebp
    jmp token_loop

token_done:
    cmp r11b, 1
    je output_error
    cmp ebp, 1
    jne output_error
    mov rax, [rel eval_stack]

    # convert rax to decimal in outbuf
    lea rdi, [rel outbuf + 32]     # points just past buffer end
    mov rbx, rdi
    xor rcx, rcx                   # digit count
    mov r8b, 0
    test rax, rax
    jge .conv_nonneg
    mov r8b, 1
    neg rax
.conv_nonneg:
    test rax, rax
    jnz .conv_loop_start
    dec rbx
    mov byte [rbx], '0'
    inc rcx
    jmp .conv_done
.conv_loop_start:
.conv_loop:
    xor rdx, rdx
    mov r9, 10
    div r9
    dec rbx
    add dl, '0'
    mov [rbx], dl
    inc rcx
    test rax, rax
    jnz .conv_loop
.conv_done:
    cmp r8b, 0
    je .no_sign
    dec rbx
    mov byte [rbx], '-'
    inc rcx
.no_sign:
    mov rsi, rbx
    mov rdx, rcx
    mov rdi, 1
    mov rax, 1
    syscall                         # write number
    # write newline
    mov byte [rel outbuf], 10
    lea rsi, [rel outbuf]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall
    jmp line_finish

output_error:
    lea rsi, [rel err_msg]
    mov rdx, err_len
    mov rdi, 1
    mov rax, 1
    syscall
    jmp line_finish

line_finish:
    mov rbx, r10
    jmp outer_loop

exit_prog:
    mov rdi, 0
    mov rax, 60
    syscall
