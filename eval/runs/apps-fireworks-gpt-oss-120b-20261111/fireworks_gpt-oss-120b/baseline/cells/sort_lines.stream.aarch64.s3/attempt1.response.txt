.global _start
.text
_start:
    // read input
    adr     x19, buf               // buffer base
    mov     x20, #0                // total_len = 0
read_loop:
    mov     x0, #0                 // fd = stdin
    add     x1, x19, x20           // buf + total_len
    mov     x2, #32768
    sub     x2, x2, x20            // remaining space
    cmp     x2, #0
    ble     read_done
    mov     x8, #63                // sys_read
    svc     #0
    cmp     x0, #0
    beq     read_done
    add     x20, x20, x0           // total_len += ret
    b       read_loop
read_done:
    // split into lines
    mov     x21, x19               // cur = buf
    add     x22, x19, x20          // end = buf+total_len
    adr     x23, lines_ptr
    adr     x24, lines_len
    mov     x25, #0                // line_cnt = 0
parse_loop:
    cmp     x21, x22
    beq     parse_done
    // store pointer
    mov     x0, x25
    lsl     x0, x0, #3             // *8
    add     x5, x23, x0
    str     x21, [x5]              // lines_ptr[line_cnt] = cur
    // search newline
    mov     x6, x21                // search ptr
search_nl:
    cmp     x6, x22
    beq     no_nl_found
    ldrb    w7, [x6]
    cmp     w7, #10
    beq     nl_found
    add     x6, x6, #1
    b       search_nl
nl_found:
    sub     x8, x6, x21            // length = nl - cur
    // store length
    mov     x0, x25
    lsl     x0, x0, #3
    add     x5, x24, x0
    str     x8, [x5]
    // advance cur past newline
    add     x6, x6, #1
    mov     x21, x6
    add     x25, x25, #1
    b       parse_loop
no_nl_found:
    sub     x8, x22, x21           // length = end - cur
    mov     x0, x25
    lsl     x0, x0, #3
    add     x5, x24, x0
    str     x8, [x5]
    mov     x21, x22
    add     x25, x25, #1
    b       parse_loop
parse_done:
    // insertion sort
    cmp     x25, #1
    ble     after_sort
    mov     x26, #1                // i = 1
i_loop:
    cmp     x26, x25
    bge     after_sort
    // load element i
    mov     x0, x26
    lsl     x0, x0, #3
    add     x5, x23, x0
    ldr     x9, [x5]               // p_i
    add     x5, x24, x0
    ldr     x10, [x5]              // len_i
    mov     x27, x26               // j = i
j_loop:
    cmp     x27, #0
    beq     j_done
    sub     x0, x27, #1
    lsl     x0, x0, #3
    add     x5, x23, x0
    ldr     x11, [x5]              // p_j-1
    add     x5, x24, x0
    ldr     x12, [x5]              // len_j-1
    // compare(j-1, i)
    mov     x0, x11
    mov     x1, x12
    mov     x2, x9
    mov     x3, x10
    bl      compare
    cmp     w0, #0
    ble     j_done
    // shift j-1 forward to j
    mov     x0, x27
    lsl     x0, x0, #3
    add     x5, x23, x0
    str     x11, [x5]
    add     x5, x24, x0
    str     x12, [x5]
    sub     x27, x27, #1
    b       j_loop
j_done:
    // place element i at position j
    mov     x0, x27
    lsl     x0, x0, #3
    add     x5, x23, x0
    str     x9, [x5]
    add     x5, x24, x0
    str     x10, [x5]
    add     x26, x26, #1
    b       i_loop
after_sort:
    // output
    mov     x26, #0                // idx = 0
out_loop:
    cmp     x26, x25
    bge     out_done
    mov     x0, x26
    lsl     x0, x0, #3
    add     x5, x23, x0
    ldr     x9, [x5]               // ptr
    add     x5, x24, x0
    ldr     x10, [x5]              // len
    // write line
    mov     x0, #1
    mov     x1, x9
    mov     x2, x10
    mov     x8, #64
    svc     #0
    // write newline
    adr     x1, newline
    mov     x0, #1
    mov     x2, #1
    mov     x8, #64
    svc     #0
    add     x26, x26, #1
    b       out_loop
out_done:
    // exit
    mov     x0, #0
    mov     x8, #93
    svc     #0

// compare function: returns 1 if ptr1>ptr2, -1 if <, 0 if equal
// x0 = ptr1, x1 = len1, x2 = ptr2, x3 = len2
compare:
    mov     x4, #0                 // offset
cmp_loop:
    cmp     x4, x1
    bge     cmp_len_done
    cmp     x4, x3
    bge     cmp_len_done
    add     x5, x0, x4
    ldrb    w6, [x5]
    add     x5, x2, x4
    ldrb    w7, [x5]
    cmp     w6, w7
    beq     cmp_next
    // differ
    cmp     w6, w7
    bgt     cmp_ret_pos
    mov     w0, #-1
    ret
cmp_ret_pos:
    mov     w0, #1
    ret
cmp_next:
    add     x4, x4, #1
    b       cmp_loop
cmp_len_done:
    cmp     x1, x3
    beq     cmp_eq
    cmp     x1, x3
    bgt     cmp_ret_pos
    mov     w0, #-1
    ret
cmp_eq:
    mov     w0, #0
    ret

.section .rodata
newline:
    .byte 0x0A

.bss
.align 8
buf:
    .space 32768
lines_ptr:
    .space 8000
lines_len:
    .space 8000
