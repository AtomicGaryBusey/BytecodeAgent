	.text
	.globl _start
	.align 2

	.equ SYS_READ, 63
	.equ SYS_WRITE, 64
	.equ SYS_EXIT, 93

	// buffers in .bss
	.bss
	.align 4
in_buf:
	.space 65536
out_buf:
	.space 131072

	.text
_start:
	// x19 = &in_buf
	adrp	x19, in_buf
	add	x19, x19, :lo12:in_buf
	// x20 = &out_buf
	adrp	x20, out_buf
	add	x20, x20, :lo12:out_buf

	// x21 = total_len = 0
	mov	x21, xzr

read_loop:
	// remaining = 65536 - total_len
	mov	x0, #65536
	sub	x0, x0, x21
	cbz	x0, read_done               // no space left

	// x1 = in_buf + total_len
	add	x1, x19, x21
	// x2 = count = remaining
	mov	x2, x0
	// fd = 0 (stdin)
	mov	x0, #0
	mov	x8, #SYS_READ
	svc	#0

	// x0 = bytes read
	cbz	x0, read_done               // EOF
	add	x21, x21, x0                // total_len += bytes
	b	read_loop

read_done:
	// src = in_buf
	mov	x22, x19
	// src_end = in_buf + total_len
	add	x24, x19, x21
	// dst = out_buf
	mov	x23, x20

	// if total_len == 0, skip processing
	cbz	x21, write_output

process_loop:
	cmp	x22, x24
	b.ge	write_output                // src >= src_end

	// line_start = src
	mov	x25, x22

	// find newline or end
find_nl:
	cmp	x22, x24
	b.ge	nl_found
	ldrb	w0, [x22]
	cmp	w0, #0x0a
	beq	nl_found
	add	x22, x22, #1
	b	find_nl

nl_found:
	// line_end = src (x22)
	// rev copy from line_end-1 down to line_start
	mov	x26, x22                      // tmp = line_end

rev_copy_loop:
	cmp	x26, x25
	beq	after_rev
	sub	x26, x26, #1
	ldrb	w0, [x26]
	strb	w0, [x23], #1
	b	rev_copy_loop

after_rev:
	// write newline
	mov	w0, #0x0a
	strb	w0, [x23], #1

	// if we stopped at a newline, consume it
	cmp	x22, x24
	b.ge	write_output
	ldrb	w0, [x22]
	cmp	w0, #0x0a
	bne	write_output
	add	x22, x22, #1          // skip newline

	b	process_loop

write_output:
	// out_len = dst - out_buf
	sub	x0, x23, x20
	cbz	x0, exit_prog

	// write(fd=1, buf=out_buf, count=out_len)
	mov	x1, x20
	mov	x2, x0
	mov	x0, #1
	mov	x8, #SYS_WRITE
	svc	#0

exit_prog:
	mov	x0, #0
	mov	x8, #SYS_EXIT
	svc	#0
