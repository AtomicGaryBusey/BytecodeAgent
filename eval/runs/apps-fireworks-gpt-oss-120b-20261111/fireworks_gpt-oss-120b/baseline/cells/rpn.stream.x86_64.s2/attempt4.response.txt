.section .bss
.align 16
input_buf:      .space 65536
output_buf:     .space 32
stack_buf:      .space 8192          # 1024 * 8
num_buf:        .space 22           # max 20 digits + sign + possible newline

.section .rodata
error_msg:      .ascii "error\n"
error_len = . - error_msg
newline_msg:    .ascii "\n"
newline_len = . - newline_msg

.section .text
.globl _start
_start:
    # ----- read whole input -----
    xor     rbx, rbx                # total length = 0
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]
    mov     rdx, 65536
    xor     rax, rax                # sys_read
read_loop:
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                # accumulate length
    jmp     read_loop
read_done:
    mov     r12, rbx                # r12 = total input length
    xor     r13, r13                # r13 = current offset

main_loop:
    cmp     r13, r12
    jge     finish

    # ----- line initialization -----
    lea     rsi, [rip + stack_buf]  # rsi = stack base
    xor     r14, r14                # r14 = stack depth
    xor     r15, r15                # r15 = error flag (0 = ok)

line_process:
    # skip leading separators (space or tab) within line
skip_sep:
    cmp     r13, r12
    je      line_end
    mov     al, byte [rip + input_buf + r13]
