.section .bss
.align 16
input_buf:      .space 65536
output_buf:     .space 32
stack_buf:      .space 8192          # 1024 entries * 8
num_buf:        .space 20

.section .text
.globl _start
_start:
    # ---------- read whole input ----------
    xor     rbx, rbx                # total length = 0
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]
    mov     rdx, 65536
    xor     rax, rax                # sys_read
read_loop:
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                # accumulate length
    jmp     read_loop
read_done:
    mov     r12, rbx                # r12 = total input length
    xor     r13, r13                # r13 = current offset

    # ---------- main loop over lines ----------
main_loop:
    cmp     r13, r12
    jge     finish

    # set up stack
    lea     rsi, [rip + stack_buf]  # rsi = stack base
    xor     r14, r14                # r14 = stack pointer (count)
    xor     r15, r15                # r15 = error flag (0 = ok)

line_loop:
    # check end of input
    cmp     r13, r12
    je      line_end_no_nl

    mov     al, byte ptr [rip + input_buf + r13]
