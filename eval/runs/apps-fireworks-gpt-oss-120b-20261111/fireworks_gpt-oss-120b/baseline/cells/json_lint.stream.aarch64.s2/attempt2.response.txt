	.globl	_start
	.text
_start:
	// x19 = address of input buffer
	adrp	x19, input_buf
	add	x19, x19, :lo12:input_buf

	// read loop
	mov	x0, #0			// fd = stdin
	mov	x2, #65536		// count
	mov	x3, #0			// total read so far (use x3)
ReadLoop:
	cmp	x3, #65536
	b.ge	ReadDone
	// buf = input_buf + total
	add	x1, x19, x3
	sub	x2, #65536, x3		// remaining
