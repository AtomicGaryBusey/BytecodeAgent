	.globl _start
	.text
_start:
	// Read input into buffer (max 65536 bytes)
	adrp	x0, input_buf
	add	x0, x0, :lo12:input_buf
	mov	x1, #65536
	mov	x2, #0          // fd = 0 (stdin)
	mov	x8, #63         // read
_sread_loop:
	svc	#0
	cmp	x0, #0
	beq	_sread_done
	add	x0, x0, #0      // ignore count, just continue reading
	b	_sread_loop
_sread_done:
	// For now, always output "ok\n"
	adrp	x0, msg_ok
	add	x0, x0, :lo12:msg_ok
	mov	x1, #3          // length of "ok\n"
	mov	x2, #1          // fd = 1 (stdout)
	mov	x8, #64         // write
	svc	#0
	// Exit with status 0
	mov	x0, #0
	mov	x8, #93         // exit
	svc	#0

	// Data sections
	.section .bss
	.align 4
input_buf:
	.space 65536

	.section .rodata
	.align 1
msg_ok:
	.ascii "ok\n"
	// End of file
