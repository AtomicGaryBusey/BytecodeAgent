.globl _start
.text

.equ    MAX_INPUT, 32768
.equ    MAX_LINES, 1000

_start:
    # read all input into buffer
    xor     rbx, rbx                # total_read = 0
read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]
    add     rsi, rbx                # buf + total_read
    mov     rdx, MAX_INPUT
    sub     rdx, rbx                # remaining space
    syscall
    test    rax, rax
    jz      read_done               # EOF
    js      exit_err                # negative = error
    add     rbx, rax                # total_read += nread
    cmp     rbx, MAX_INPUT
    jb      read_loop
read_done:
    test    rbx, rbx
    jz      exit_success            # empty input, nothing to do

    # parse lines, store offsets and lengths
    xor     rcx, rcx                # scan index i = 0
    xor     rdx, rdx                # line_count = 0
    xor     r8,  r8                 # start = 0 (offset of current line)
parse_loop:
    cmp     rcx, rbx
    jae     parse_done
    lea     rsi, [rip + input_buf]
    add     rsi, rcx
    mov     al, byte ptr [rsi]
    cmp     al, 0x0A                # newline ?
    jne     not_newline
    # newline found, store line [start, rcx)
    mov     r9, rcx
    sub     r9, r8                  # length = rcx - start
    mov     rax, rdx                # line index
    lea     rsi, [rip + line_offsets]
    mov     [rsi + rax*8], r8
    lea     rsi, [rip + line_lengths]
    mov     [rsi + rax*8], r9
    inc     rdx                     # line_count++
    lea     r8, [rcx + 1]           # start = rcx+1 (next char)
not_newline:
    inc     rcx
    jmp     parse_loop
parse_done:
    # handle final line if any bytes after last newline
    cmp     r8, rbx
    jae     store_linecnt           # start >= total_read => no remainder
    mov     r9, rbx
    sub     r9, r8                  # length = total_read - start
    mov     rax, rdx
    lea     rsi, [rip + line_offsets]
    mov     [rsi + rax*8], r8
    lea     rsi, [rip + line_lengths]
    mov     [rsi + rax*8], r9
    inc     rdx                     # line_count++
store_linecnt:
    mov     dword ptr [rip + line_cnt], edx

    # -------------------------------------------------
    # insertion sort (ascending)
    xor     r14, r14                # i = 0
    inc     r14                     # start from 1
sort_i_loop:
    mov     eax, dword ptr [rip + line_cnt]
    cmp     r14d, eax
    jge     sort_done
    mov     r15, r14                # j = i
sort_j_loop:
    cmp     r15, 0
    je      sort_i_next
    mov     rax, r15
    dec     rax                     # a_index = j-1
    mov     rbx, r15                # b_index = j

    # load offsets
    lea     rsi, [rip + line_offsets]
    mov     r8,  [rsi + rax*8]      # off_a
    mov     r9,  [rsi + rbx*8]      # off_b
    # load lengths
    lea     rsi, [rip + line_lengths]
    mov     r10, [rsi + rax*8]      # len_a
    mov     r11, [rsi + rbx*8]      # len_b

    # compare a vs b
    mov     rdi, r8                 # off_a
    mov     rsi, r10                # len_a
    mov     rdx, r9                 # off_b
    mov     rcx, r11                # len_b
    call    compare
    test    rax, rax
    jle     sort_i_next             # a <= b, keep order

    # swap entries at indices a_index (rax) and b_index (rbx)
    mov     r12d, eax               # a_index (low dword)
    mov     r13d, ebx               # b_index

    lea     rsi, [rip + line_offsets]
    mov     rax, [rsi + r12*8]
    mov     rdx, [rsi + r13*8]
    mov     [rsi + r12*8], rdx
    mov     [rsi + r13*8], rax

    lea     rsi, [rip + line_lengths]
    mov     rax, [rsi + r12*8]
    mov     rdx, [rsi + r13*8]
    mov     [rsi + r12*8], rdx
    mov     [rsi + r13*8], rax

    dec     r15
    jmp     sort_j_loop

sort_i_next:
    inc     r14
    jmp     sort_i_loop

sort_done:

    # -------------------------------------------------
    # output sorted lines
    xor     rcx, rcx                # idx = 0
output_loop:
    mov     eax, dword ptr [rip + line_cnt]
    cmp     rcx, eax
