.globl _start

#--------------------------------------------------------------------
# Data
#--------------------------------------------------------------------
.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

#--------------------------------------------------------------------
# BSS
#--------------------------------------------------------------------
.section .bss
.align 16
input_buf:
    .skip 65536          # 64 KiB input buffer
line_buf:
    .skip 80             # line buffer (max 79 chars + possible null)

#--------------------------------------------------------------------
# Text
#--------------------------------------------------------------------
.section .text
_start:
    # ------------------------------------------------------------
    # Read entire stdin into input_buf, total length in rbx
    # ------------------------------------------------------------
    xor     rbx, rbx                 # total = 0

read_loop:
    mov     rdi, 0                   # fd = stdin
    lea     rsi, [rip + input_buf + rbx]
