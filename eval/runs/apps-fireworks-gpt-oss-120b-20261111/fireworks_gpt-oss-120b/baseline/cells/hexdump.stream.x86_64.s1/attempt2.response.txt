.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.equ HEX_START, 10
.equ HEX_EXTRA_SPACE_POS, (HEX_START + 8*3)          # 34
.equ AFTER_HEX_SPACE_POS, (HEX_START + 16*3 + 1)    # 59
.equ ASCII_START, (AFTER_HEX_SPACE_POS + 2)        # 61

.section .bss
.align 16
input_buf:
    .zero 65536
line_buf:
    .zero 80

.section .text
_start:
    # ------------------------------------------------------------
    # Read all input into input_buf, total length in rbx
    # ------------------------------------------------------------
    xor     rbx, rbx                 # total = 0
read_loop:
    mov     rdi, 0                   # fd = stdin
    lea     rsi, [rip + input_buf]   # base of buffer
    add     rsi, rbx                 # current pointer = base + total
    mov     rdx, 65536
    sub     rdx, rbx                 # remaining space
    mov     rax, 0                   # sys_read
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                 # total += bytes read
    jmp     read_loop
read_done:
    # ------------------------------------------------------------
    # Prepare pointers and counters for output
    # ------------------------------------------------------------
    mov     rsi, rip
