.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf:
    .zero 65536
line_buf:
    .zero 80

.section .text
_start:
    # ------------------------------------------------------------
    # Read all input into input_buf, total length in rbx
    # ------------------------------------------------------------
    xor     rbx, rbx                 # total = 0
read_loop:
    mov     rdi, 0                   # fd = stdin
    lea     rsi, [rip + input_buf]   # base of buffer
    add     rsi, rbx                 # current pointer = base + total
    mov     rdx, 65536
    sub     rdx, rbx                 # remaining space
    mov     rax, 0                   # sys_read
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                 # total += bytes read
    jmp     read_loop
read_done:

    # ------------------------------------------------------------
    # Output rows
    # ------------------------------------------------------------
    xor     rcx, rcx                 # offset = 0
    lea     r8, [rip + line_buf]     # line buffer pointer
    lea     r9, [rip + hex_digits]   # hex digits table

row_loop:
    cmp     rcx, rbx
    jae     after_rows               # finished all rows

    # n = min(16, total - offset)
    mov     rax, rbx
    sub     rax, rcx
    cmp     rax, 16
    cmova   rdx, rax                 # if remaining < 16, n = remaining
    mov     rdx, 16                  # else n = 16
    cmp     rax, 16
    jb      set_n_small
set_n_full:
    mov     rdx, 16
    jmp     n_ready
set_n_small:
    mov     rdx, rax
n_ready:
    # source pointer for this row
    lea     r14, [rip + input_buf]
    add     r14, rcx                 # r14 = input_buf + offset

    # ------------------------------------------------------------
    # Write offset (8 hex digits) at line_buf[0..7]
    # ------------------------------------------------------------
    xor     r10, r10                 # i = 0
offset_digit_loop:
    mov     rcx, 7
    sub     rcx, r10
    shl     rcx, 2                   # shift = (7-i)*4
    mov     rax, rcx                 # backup shift in rax for later
    mov     rax, rcx                 # need shift count in cl
    mov     rax, rcx
    mov     rcx, rax
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx
    mov     rax, rcx

    # shift amount is in rcx, need it in cl
    mov     cl, cl
    mov     rax, rcx
    mov     rax, rcx

    mov     rax, rcx
    mov     rax, rcx

    mov     rax, rcx
    mov     rax, rcx

    mov     rax, rcx
    mov     rax, rcx

    mov     rax, rcx
    mov     rax, rcx

    mov     rax, rcx
    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    mov     rax, rcx

    ; extract nibble
