.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buf:
    .zero 65536
line_buf:
    .zero 80

.section .text
_start:
    # ------------------------------------------------------------
    # Read all input into input_buf, total length in r13
    # ------------------------------------------------------------
    xor     rbx, rbx                 # total = 0
read_loop:
    mov     rdi, 0                   # fd = stdin
    lea     rsi, [rip + input_buf]   # buffer base
    add     rsi, rbx                 # current pointer = base + total
    mov     rdx, 65536
    sub     rdx, rbx                 # remaining space
    mov     rax, 0                   # sys_read
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                 # total += bytes read
    jmp     read_loop
read_done:
    mov     r13, rbx                 # r13 = total length (keep safe)

    # ------------------------------------------------------------
    # Process rows of 16 bytes
    # ------------------------------------------------------------
    xor     r12, r12                 # offset = 0
row_loop:
    cmp     r12, r13
    jae     after_rows

    # rdx = row_len = min(16, total - offset)
    mov     rax, r13
    sub     rax, r12
    mov     rdx, 16
    cmp     rax, 16
    cmova   rdx, rax

    # r8 = pointer to start of this row
    lea     r8, [rip + input_buf]
    add     r8, r12

    # rdi = line buffer base
    lea     rdi, [rip + line_buf]

    # ------------------------------------------------------------
    # Write offset (8 hex digits)
    # ------------------------------------------------------------
    lea     rsi, [rip + hex_digits]   # hex table
    xor     r9, r9                    # i = 0
offset_loop:
    cmp     r9, 8
    je      offset_done
    mov     r10, 7
    sub     r10, r9
    shl     r10, 2                    # shift = (7-i)*4
    mov     rcx, r10
    mov     rax, r12                  # offset value
    shr     rax, cl
    and     al, 0xF
    mov     bl, [rsi + rax]
    mov     [rdi + r9], bl
    inc     r9
    jmp     offset_loop
offset_done:
    mov     byte ptr [rdi + 8], ' '
    mov     byte ptr [rdi + 9], ' '

    # ------------------------------------------------------------
    # Hex dump of the row (first 8 bytes)
    # ------------------------------------------------------------
    xor     r10, r10                  # i = 0
first8_loop:
    cmp     r10, 8
    je      after_first8
    ; destination pointer = line_buf + 10 + i*3
