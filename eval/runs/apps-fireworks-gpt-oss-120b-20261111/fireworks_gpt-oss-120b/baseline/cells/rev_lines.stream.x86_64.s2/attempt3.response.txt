.globl _start
.text
_start:
    # constants
    .equ BUFSIZE, 65536          # maximum input size
    .equ OUTSIZE, BUFSIZE+1      # output buffer size (extra newline)

    # set up buffer pointers
    lea r13, [rip + inbuf]       # r13 = input buffer base
    lea r14, [rip + outbuf]      # r14 = output buffer base

    xor r12, r12                 # r12 = read offset within input buffer
    xor r15, r15                 # r15 = total bytes read

read_loop:
    mov rsi, r13                 # rsi = inbuf
    add rsi, r12                 # rsi = inbuf + offset
    mov eax, BUFSIZE
    sub eax, r12d                # eax = remaining space
    test eax, eax
    jz read_done                 # should not happen (input limit)
    xor edi, edi                 # edi = 0 (stdin)
    mov edx, eax                 # edx = count
    mov eax, 0                   # syscall: read
    syscall
    test rax, rax
    jz read_done                 # EOF
    add r15, rax                 # total += bytes read
    add r12, rax                 # offset += bytes read
    jmp read_loop

read_done:
    # r15 = total input length
    mov rsi, r13                 # rsi = current pointer (start of line)
    lea rdx, [r13 + r15]         # rdx = end pointer (inbuf + total)
    mov rdi, r14                 # rdi = output pointer

process_loop:
    cmp rsi, rdx
    jae proc_done                # no more input

    mov r8, rsi                  # r8  = line_start
    mov r9, rsi                  # r9  = scan pointer

scan_loop:
    cmp r9, rdx
    jae no_newline               # reached end without newline
    mov al, [r9]
    cmp al, 0x0a                 # newline?
    je newline_found
    inc r9
    jmp scan_loop

newline_found:
    # r9 points at newline byte
    mov rax, r9
    dec rax                      # rax = line_end - 1
rev_loop:
    cmp rax, r8
    jl rev_done
    mov bl, [rax]
    mov [rdi], bl
    inc rdi
    dec rax
    jmp rev_loop
rev_done:
    mov byte ptr [rdi], 0x0a
    inc rdi
    inc r9                       # skip newline
    mov rsi, r9
    jmp process_loop

no_newline:
    # last line without terminating newline
    mov rax, rdx
    dec rax                      # rax = end - 1
rev_last_loop:
    cmp rax, r8
    jl rev_last_done
    mov bl, [rax]
    mov [rdi], bl
    inc rdi
    dec rax
    jmp rev_last_loop
rev_last_done:
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rsi, rdx
    jmp process_loop

proc_done:
    # rdi = pointer after last output byte
    mov r8, rdi
    sub r8, r14                  # r8 = output length
    mov rsi, r14                 # rsi = output buffer start

write_loop:
    test r8, r8
    jz exit_prog
    mov rdx, r8
    mov edi, 1                   # fd = stdout
    mov eax, 1                   # syscall: write
    syscall
    test rax, rax
    jz exit_prog
    sub r8, rax
    add rsi, rax
    jmp write_loop

exit_prog:
    mov eax, 60                  # syscall: exit
    xor edi, edi                 # status = 0
    syscall

.section .bss
    .align 16
inbuf:  .zero BUFSIZE
outbuf: .zero OUTSIZE
