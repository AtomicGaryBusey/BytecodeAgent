.globl _start
.text
_start:
    # constants
    # (defined in .equ below, but we can also embed numeric literals)

    # set up pointers to buffers
    lea r10, [rip + buf]          # r10 = input buffer base
    lea r13, [rip + outbuf]       # r13 = output buffer base

    xor r12, r12                  # r12 = offset into input buffer
    xor rbx, rbx                  # rbx = total bytes read

read_loop:
    mov rsi, r10                  # rsi = buf
    add rsi, r12                  # rsi = buf + offset
    mov eax, BUFSIZE
    sub eax, r12d                 # eax = remaining space
    test eax, eax
    jz read_done                  # should not happen, input limit reached
    xor edi, edi                  # fd = 0 (stdin)
    mov edx, eax                  # count = remaining space
    mov eax, 0                    # syscall: read
    syscall
    test rax, rax
    jz read_done                  # EOF
    add rbx, rax                  # total += bytes read
    add r12, rax                  # offset += bytes read
    jmp read_loop

read_done:
    # rbx = total input length
    mov rsi, r10                  # rsi = cur pointer (start of line)
    lea rdx, [r10 + rbx]          # rdx = end pointer (buf + total)
    mov rdi, r13                  # rdi = outptr (output buffer)

process_loop:
    cmp rsi, rdx
    jae proc_done                 # no more input

    mov r8, rsi                   # r8 = line_start
    mov r9, rsi                   # r9 = scan pointer

scan_loop:
    cmp r9, rdx
    jae no_newline                # reached end without newline
    mov al, [r9]
    cmp al, 0x0a                  # newline?
    je newline_found
    inc r9
    jmp scan_loop

newline_found:
    # r9 points at newline byte
    mov rax, r9
    dec rax                       # rax = line_end - 1
rev_loop:
    cmp rax, r8
    jl rev_done
    mov bl, [rax]
    mov [rdi], bl
    inc rdi
    dec rax
    jmp rev_loop
rev_done:
    mov byte ptr [rdi], 0x0a
    inc rdi
    inc r9                        # skip newline
    mov rsi, r9
    jmp process_loop

no_newline:
    # last line without terminating newline
    mov rax, rdx
    dec rax                       # rax = end - 1
rev_last_loop:
    cmp rax, r8
    jl rev_last_done
    mov bl, [rax]
    mov [rdi], bl
    inc rdi
    dec rax
    jmp rev_last_loop
rev_last_done:
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rsi, rdx
    jmp process_loop

proc_done:
    # rdi = outptr end, r13 = outbuf base
    mov r8, rdi
    sub r8, r13                    # r8 = out_len
    mov rsi, r13                   # rsi = output pointer

write_loop:
    test r8, r8
    jz exit_prog
    mov rdx, r8
    mov edi, 1                     # fd = 1 (stdout)
    mov eax, 1                     # syscall: write
    syscall
    test rax, rax
    jz exit_prog
    sub r8, rax
    add rsi, rax
    jmp write_loop

exit_prog:
    mov eax, 60                    # syscall: exit
    xor edi, edi                   # status = 0
    syscall

.section .bss
    .align 16
buf:        .zero BUFSIZE
