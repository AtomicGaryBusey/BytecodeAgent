.globl _start
.text
_start:
    # buffer base in r10, outbuf base in r13
    lea r10, [rip + buf]
    lea r13, [rip + outbuf]

    xor r12, r12          # offset into input buffer
    xor rbx, rbx          # total bytes read

read_loop:
    mov rsi, r10
    add rsi, r12          # rsi = buf + offset
    mov edx, BUFSIZE
    sub edx, r12d         # remaining space
    xor edi, edi          # fd = 0 (stdin)
    mov eax, 0            # syscall: read
    syscall
    test rax, rax
    jz read_done
    add rbx, rax          # total += bytes read
    add r12, rax          # offset += bytes read
    jmp read_loop

read_done:
    # rbx = total input length
    # rsi = cur pointer, rdx = end pointer
    mov rsi, r10          # cur = buf
    mov rdx, r10
    add rdx, rbx          # end = buf + total
    mov rdi, r13          # out_ptr = outbuf

process_loop:
    cmp rsi, rdx
    jae proc_done

    mov r8, rsi           # line_start
    mov r9, rsi           # scan pointer

scan_loop:
    cmp r9, rdx
    jae no_newline
    mov al, [r9]
    cmp al, 0x0a
    je newline_found
    inc r9
    jmp scan_loop

newline_found:
    # reverse bytes from r9-1 down to r8
    mov rax, r9
    dec rax               # rax = line_end-1
rev_loop:
    cmp rax, r8
    jl rev_done
    mov bl, [rax]
    mov [rdi], bl
    inc rdi
    dec rax
    jmp rev_loop
rev_done:
    mov byte ptr [rdi], 0x0a
    inc rdi
    inc r9                # skip newline
    mov rsi, r9
    jmp process_loop

no_newline:
    # reverse remaining bytes (last line without newline)
    mov rax, rdx
    dec rax               # rax = end-1
rev_last_loop:
    cmp rax, r8
    jl rev_last_done
    mov bl, [rax]
    mov [rdi], bl
    inc rdi
    dec rax
    jmp rev_last_loop
rev_last_done:
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rsi, rdx
    jmp process_loop

proc_done:
    # compute output length = rdi - outbuf base (r13)
    mov r8, rdi
    sub r8, r13           # r8 = out_len
    mov rsi, r13          # rsi = output pointer

write_loop:
    test r8, r8
    jz exit_prog
    mov rdx, r8
    mov edi, 1            # fd = 1 (stdout)
    mov eax, 1            # syscall: write
    syscall
    test rax, rax
    jz exit_prog
    sub r8, rax
    add rsi, rax
    jmp write_loop

exit_prog:
    mov eax, 60           # syscall: exit
    xor edi, edi          # status = 0
    syscall

.bss
    .align 16
buf:        .zero BUFSIZE
