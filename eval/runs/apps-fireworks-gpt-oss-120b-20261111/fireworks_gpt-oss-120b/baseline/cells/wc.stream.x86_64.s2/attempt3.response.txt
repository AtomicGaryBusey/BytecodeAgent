.text
.globl _start
_start:
    xor r8, r8          # r8 = total bytes
    xor r9, r9          # r9 = line count
    xor r10, r10        # r10 = word count
    xor r11, r11        # r11 = in_word flag (0)

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_read                # rax == 0 -> EOF
    mov rcx, rax                # rcx = bytes read
    lea rbx, [rip + buffer]     # rbx = ptr to current byte

proc_loop:
    mov al, byte ptr [rbx]
    inc rbx
    inc r8                      # total bytes++
    cmp al, 10                  # newline?
    jne not_nl
    inc r9                      # line++
not_nl:
    ; whitespace test
