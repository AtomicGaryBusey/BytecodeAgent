.text
.global _start
_start:
    xor r8, r8          # total bytes
    xor r9, r9          # line count
    xor r10, r10        # word count
    xor r11, r11        # in_word flag (0)

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # stdin
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_read                # rax == 0 -> EOF
    mov rcx, rax                # rcx = bytes read
    lea rbx, [rip + buffer]     # rbx = ptr to current byte

proc_loop:
    mov al, [rbx]               # load byte
    inc rbx
    inc r8                      # total bytes++
    cmp al, 0x0A                # newline?
    jne not_newline
    inc r9                      # line++
not_newline:
    # whitespace test
    cmp al, 0x09
    je ws
    cmp al, 0x0A
    je ws
    cmp al, 0x0B
    je ws
    cmp al, 0x0C
    je ws
    cmp al, 0x0D
    je ws
    cmp al, 0x20
    je ws
    # non‑whitespace
    test r11, r11
    jne continue_proc
    inc r10                     # new word
    mov r11, 1
    jmp continue_proc
ws:
    xor r11, r11                # leave word
continue_proc:
    dec rcx
    jnz proc_loop
    jmp read_loop

done_read:
    # build output line "<lines> <words> <bytes>\n"
    lea rdi, [rip + outbuf]     # output pointer
    mov rax, r9                 # lines
    call append_number
    mov rbx, rdi
    mov byte ptr [rbx], ' '
    inc rbx
    mov rdi, rbx
    mov rax, r10                # words
    call append_number
    mov rbx, rdi
    mov byte ptr [rbx], ' '
    inc rbx
    mov rdi, rbx
    mov rax, r8                 # bytes
    call append_number
    mov rbx, rdi
    mov byte ptr [rbx], 0x0A    # newline
    inc rbx

    # write the result
    lea rax, [rip + outbuf]     # start address
    mov rdx, rbx
    sub rdx, rax                # length
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    lea rsi, [rip + outbuf]
    syscall

    # exit
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

# ------------------------------------------------------------
# void append_number(uint64_t value in rax, char *ptr in rdi)
# returns updated rdi (pointer after written digits)
append_number:
    push rbp
    mov rbp, rsp
    mov rcx, 0                  # digit count
    cmp rax, 0
    jne .num_loop
    mov byte ptr [rdi], '0'
    inc rdi
    jmp .num_done
.num_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx                     # rax = quotient, rdx = remainder
    push rdx                    # save remainder
    inc rcx
    test rax, rax
    jne .num_loop
.num_pop:
    pop rax
    add al, '0'
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz .num_pop
.num_done:
    pop rbp
    ret

.bss
.align 16
buffer: .zero 65536
outbuf: .zero 64
