.text
.globl _start
_start:
    xor r8, r8          # byte count
    xor r9, r9          # line count
    xor r10, r10        # word count
    xor r11, r11        # in_word flag

read_loop:
    mov rax, 0          # sys_read
    mov rdi, 0          # fd = stdin
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_read
    mov rcx, rax        # bytes read
    lea rbx, [rip + buffer]

proc_loop:
    mov al, byte ptr [rbx]
    inc rbx
    inc r8               # total bytes
    cmp al, 10
    jne .not_nl
    inc r9               # line count
.not_nl:
    cmp al, 9
    je .ws
    cmp al, 10
    je .ws
    cmp al, 11
    je .ws
    cmp al, 12
    je .ws
    cmp al, 13
    je .ws
    cmp al, 32
    je .ws
    # non‑whitespace
    test r11, r11
    jne .proc_next
    inc r10              # new word
    mov r11, 1
    jmp .proc_next
.ws:
    xor r11, r11         # in_word = 0
.proc_next:
    dec rcx
    jne proc_loop
    jmp read_loop

done_read:
    lea rsi, [rip + outbuf]
    mov rdi, r9
    call uint_to_str
    mov byte ptr [rsi], ' '
    inc rsi
    mov rdi, r10
    call uint_to_str
    mov byte ptr [rsi], ' '
    inc rsi
    mov rdi, r8
    call uint_to_str
    mov byte ptr [rsi], 10
    inc rsi

    lea rdx, [rip + outbuf]
    mov rax, rsi
    sub rax, rdx          # length in rax
    mov rdx, rax
    mov rax, 1            # sys_write
    mov rdi, 1            # stdout
    lea rsi, [rip + outbuf]
    syscall

    mov rax, 60           # sys_exit
    xor rdi, rdi
    syscall

# ------------------------------------------------------------
# uint_to_str: rdi = number, rsi = output pointer, returns with rsi advanced
uint_to_str:
    test rdi, rdi
    jne .nonzero
    mov byte ptr [rsi], '0'
    inc rsi
    ret
.nonzero:
    mov rax, rdi
    mov rbx, 10
    xor rcx, rcx          # digit count
.conv_loop:
    xor rdx, rdx
    div rbx               # rax = rax/10, rdx = remainder
    sub rsp, 1
    add dl, '0'
    mov [rsp], dl
    inc rcx
    test rax, rax
    jne .conv_loop
.copy_loop:
    mov al, [rsp]
    mov [rsi], al
    inc rsi
    inc rsp
    dec rcx
    jne .copy_loop
    ret

.bss
.align 16
buffer: .zero 65536
outbuf: .space 32
