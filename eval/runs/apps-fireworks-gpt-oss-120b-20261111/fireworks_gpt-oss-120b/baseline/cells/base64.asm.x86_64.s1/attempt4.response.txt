.section .bss
.align 16
input_buf:
    .space 49152
out_buf:
    .space 70000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    xor rcx, rcx                # total bytes read

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + input_buf]
    add rsi, rcx
    mov rdx, 4096
    syscall
    test rax, rax
    jz after_read
    add rcx, rax
    cmp rcx, 49152
    jb read_loop
    jmp read_loop               # should not exceed limit

after_read:
    test rcx, rcx
    jz exit_success

    # rsi = input pointer, rdx = remaining input bytes
    lea rsi, [rip + input_buf]
    mov rdx, rcx                # remaining input length
    # rbx = out start, rdi = out ptr, r8b = line counter
    lea rbx, [rip + out_buf]
    mov rdi, rbx
    xor r8b, r8b

    # r9 = base64 table pointer
    lea r9, [rip + b64_table]

main_loop:
    cmp rdx, 3
    jb leftovers

    # ---- encode 3 bytes to 4 chars ----
    # idx0
    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 2
    mov al, byte ptr [r9 + rcx]
    call store_char

    # idx1
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    movzx ecx, byte ptr [rsi+1]
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [r9 + rax]
    call store_char

    # idx2
    movzx eax, byte ptr [rsi+1]
    and eax, 0xF
    shl eax, 2
    movzx ecx, byte ptr [rsi+2]
    shr ecx, 6
    or eax, ecx
    mov al, byte ptr [r9 + rax]
    call store_char

    # idx3
    movzx eax, byte ptr [rsi+2]
    and eax, 0x3F
    mov al, byte ptr [r9 + rax]
    call store_char

    add rsi, 3
    sub rdx, 3
    jmp main_loop

leftovers:
    cmp rdx, 0
    je finish_output
    cmp rdx, 1
    je leftover_one

    # rdx == 2
    # idx0
    movzx eax, byte ptr [rsi]
    shr eax, 2
    mov al, byte ptr [r9 + rax]
    call store_char

    # idx1
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    movzx ecx, byte ptr [rsi+1]
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [r9 + rax]
    call store_char

    # idx2
    movzx eax, byte ptr [rsi+1]
    and eax, 0xF
    shl eax, 2
    mov al, byte ptr [r9 + rax]
    call store_char

    # padding '='
    mov al, '='
    call store_char
    jmp finish_output

leftover_one:
    # idx0
    movzx eax, byte ptr [rsi]
    shr eax, 2
    mov al, byte ptr [r9 + rax]
    call store_char

    # idx1
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    mov al, byte ptr [r9 + rax]
    call store_char

    # padding '=='
    mov al, '='
    call store_char
    mov al, '='
    call store_char
    jmp finish_output

finish_output:
    # if any output generated and current line not terminated, add newline
    cmp rdi, rbx
    je write_and_exit            # no output at all
    cmp r8b, 0
    je write_and_exit            # line already ended with newline
    mov al, 10                   # newline
    call store_char

write_and_exit:
    # rbx = start, rdi = end
    mov rsi, rbx
    mov rdx, rdi
    sub rdx, rsi                 # length
    test rdx, rdx
    je exit_success

    mov rax, 1                   # sys_write
    mov rdi, 1                   # fd 1
    syscall

    mov rax, 60                  # sys_exit
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

# -------------------------------------------------
# store_char: al = character to output
# uses rdi (output pointer) and r8b (line counter)
store_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r8b
    cmp r8b, 76
    jne .sc_ret
    mov byte ptr [rdi], 10
    inc rdi
    xor r8b, r8b
.sc_ret:
    ret
