.section .bss
.align 16
input_buf:
    .space 49152
out_buf:
    .space 66400

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    # read all input
    xor rcx, rcx                # total bytes read
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + input_buf + rcx]
    mov rdx, 4096
    syscall
    test rax, rax
    jz after_read
    add rcx, rax
    cmp rcx, 49152
    jb read_loop
    jmp read_loop               # should not exceed limit
after_read:
    test rcx, rcx
    jz exit_success

    # set up pointers and counters
    lea rsi, [rip + input_buf]  # input pointer
    lea rdi, [rip + out_buf]    # output pointer
    xor r8d, r8d                # line length counter
    mov rdx, rcx                # remaining input bytes
    lea r9, [rip + b64_table]   # base64 table address

main_loop:
    cmp rdx, 3
    jb leftovers

    # load three bytes
    movzx eax, byte ptr [rsi]        # a
    movzx ebx, byte ptr [rsi+1]      # b
    movzx ecx, byte ptr [rsi+2]      # c

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, byte ptr [r9 + rax]
    call output_char

    # index1 = ((a & 3) << 4) | (b >> 4)
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    movzx ebx, byte ptr [rsi+1]
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [r9 + rax]
    call output_char

    # index2 = ((b & 0xF) << 2) | (c >> 6)
    movzx ebx, byte ptr [rsi+1]
    and ebx, 0xF
    shl ebx, 2
    movzx ecx, byte ptr [rsi+2]
    shr ecx, 6
    or ebx, ecx
    mov eax, ebx
    mov al, byte ptr [r9 + rax]
    call output_char

    # index3 = c & 0x3F
    movzx ecx, byte ptr [rsi+2]
    and ecx, 0x3F
    mov eax, ecx
    mov al, byte ptr [r9 + rax]
    call output_char

    add rsi, 3
    sub rdx, 3
    jmp main_loop

leftovers:
    cmp rdx, 0
    je finish_output
    cmp rdx, 1
    je leftover_one
    # rdx == 2
    # idx0
    movzx eax, byte ptr [rsi]
    shr eax, 2
    mov al, byte ptr [r9 + rax]
    call output_char
    # idx1 = ((a & 3) << 4) | (b >> 4)
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    movzx ebx, byte ptr [rsi+1]
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [r9 + rax]
    call output_char
    # idx2 = (b & 0xF) << 2
    movzx ebx, byte ptr [rsi+1]
    and ebx, 0xF
    shl ebx, 2
    mov eax, ebx
    mov al, byte ptr [r9 + rax]
    call output_char
    # padding '='
    mov al, '='
    call output_char
    jmp finish_output

leftover_one:
    # idx0
    movzx eax, byte ptr [rsi]
    shr eax, 2
    mov al, byte ptr [r9 + rax]
    call output_char
    # idx1 = (a & 3) << 4
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    mov al, byte ptr [r9 + rax]
    call output_char
    # padding '=='
    mov al, '='
    call output_char
    mov al, '='
    call output_char
    jmp finish_output

finish_output:
    cmp r8d, 0
    je write_output
    mov al, 10                # newline
    call output_char

write_output:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1                # sys_write
    mov rdi, 1                # fd 1
    syscall

    mov rax, 60               # sys_exit
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

# -------------------------------------------------
# output_char: al = character to output
# uses rdi (output pointer) and r8d (line counter)
output_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r8d
    cmp r8d, 76
    jne .oc_ret
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
.oc_ret:
    ret
