.section .bss
.align 16
input_buf:
    .space 49152
out_buf:
    .space 66400

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    # load base address of input buffer into rbx
    lea rbx, [rip + input_buf]

    # read loop, rcx = total bytes read
    xor rcx, rcx
read_loop:
    mov rax, 0                # sys_read
    mov rdi, 0                # fd 0
    lea rsi, [rbx + rcx]      # buffer pointer
    mov rdx, 4096
    syscall
    test rax, rax
    jz encode_start
    add rcx, rax
    jmp read_loop

encode_start:
    test rcx, rcx
    jz exit_success           # empty input, no output

    # pointers
    lea rsi, [rbx]            # input pointer
    lea rdi, [rip + out_buf]  # output pointer
    xor r8d, r8d              # line length counter
    mov rdx, rcx              # remaining input bytes
    lea r9, [rip + b64_table] # base64 table

main_loop:
    cmp rdx, 3
    jb leftovers

    # load three bytes
    mov al, [rsi]
    mov bl, [rsi+1]
    mov cl, [rsi+2]

    # index0 = al >> 2
    movzx eax, al
    shr eax, 2
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after0
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after0:

    # index1 = ((al & 3) << 4) | (bl >> 4)
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebx, bl
    shr ebx, 4
    or eax, ebx
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after1
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after1:

    # index2 = ((bl & 0xF) << 2) | (cl >> 6)
    movzx eax, bl
    and eax, 0xF
    shl eax, 2
    movzx ebx, cl
    shr ebx, 6
    or eax, ebx
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after2
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after2:

    # index3 = cl & 0x3F
    movzx eax, cl
    and eax, 0x3F
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after3
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after3:

    add rsi, 3
    sub rdx, 3
    jmp main_loop

leftovers:
    cmp rdx, 0
    je final_newline
    cmp rdx, 1
    je handle_one
    # rdx == 2
handle_two:
    mov al, [rsi]
    mov bl, [rsi+1]

    # index0
    movzx eax, al
    shr eax, 2
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after2_0
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after2_0:

    # index1 = ((al & 3) << 4) | (bl >> 4)
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebx, bl
    shr ebx, 4
    or eax, ebx
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after2_1
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after2_1:

    # index2 = (bl & 0xF) << 2
    movzx eax, bl
    and eax, 0xF
    shl eax, 2
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after2_2
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after2_2:

    # padding '='
    mov byte [rdi], '='
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after2_eq
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after2_eq:
    jmp final_newline

handle_one:
    mov al, [rsi]

    # index0
    movzx eax, al
    shr eax, 2
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after1_0
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after1_0:

    # index1 = (al & 3) << 4
    movzx eax, al
    and eax, 3
    shl eax, 4
    movzx ebp, byte [r9 + rax]
    mov [rdi], bpl
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after1_1
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after1_1:

    # padding '='
    mov byte [rdi], '='
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after1_eq1
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after1_eq1:

    mov byte [rdi], '='
    inc rdi
    inc r8d
    cmp r8d, 76
    jne after1_eq2
    mov byte [rdi], 10
    inc rdi
    xor r8d, r8d
after1_eq2:
    jmp final_newline

final_newline:
    cmp r8d, 0
    je write_output
    mov byte [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1                # sys_write
    mov rdi, 1                # fd 1
    syscall

    mov rax, 60               # sys_exit
    xor rdi, rdi
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
