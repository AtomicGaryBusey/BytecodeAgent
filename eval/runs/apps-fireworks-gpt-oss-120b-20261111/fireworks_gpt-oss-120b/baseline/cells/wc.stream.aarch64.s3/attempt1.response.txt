	.global _start
	.text
	.align 2

_start:
	// buffer address in x23
	adrp	x23, input_buf
	add	x23, x23, :lo12:input_buf
	// offset = 0 in x24
	mov	x24, #0

read_loop:
	// fd = 0
	mov	x0, #0
	// buf = base + offset
	add	x1, x23, x24
	// count = 65536 - offset
	mov	x2, #65536
	sub	x2, x2, x24
	// syscall read
	mov	x8, #63
	svc	#0
	// result in x0
	cbnz	x0, read_continue
	// x0 == 0 => EOF
	b	read_done
read_continue:
	add	x24, x24, x0
	b	read_loop
read_done:
	// total bytes = x24 -> x21
	mov	x21, x24
	// lines = 0 -> x19
	mov	x19, #0
	// words = 0 -> x20
	mov	x20, #0
	// prev_is_space = 1 -> x22
	mov	x22, #1
	// index = 0 -> x25
	mov	x25, #0

process_loop:
	cmp	x25, x24
	b.eq	process_done
	// load byte
	add	x0, x23, x25
	ldrb	w1, [x0]
	add	x25, x25, #1
	// newline?
	cmp	w1, #10
	b.ne	not_newline
	add	x19, x19, #1
not_newline:
	// whitespace checks
	cmp	w1, #0x20
	beq	is_ws
	cmp	w1, #0x09
	beq	is_ws
	cmp	w1, #0x0a
	beq	is_ws
	cmp	w1, #0x0b
	beq	is_ws
	cmp	w1, #0x0c
	beq	is_ws
	cmp	w1, #0x0d
	beq	is_ws
	// not whitespace
	cmp	x22, #1
	b.ne	not_word_start
	add	x20, x20, #1
	mov	x22, #0
	b	continue_loop
is_ws:
	mov	x22, #1
	b	continue_loop
not_word_start:
	// already inside a word, nothing to do
continue_loop:
	b	process_loop

process_done:
	// output buffer address in x1
	adrp	x1, outbuf
	add	x1, x1, :lo12:outbuf
	// lines
	mov	x0, x19
	bl	utoa
	// space
	mov	w2, #' '
	strb	w2, [x1], #1
	// words
	mov	x0, x20
	bl	utoa
	// space
	mov	w2, #' '
	strb	w2, [x1], #1
	// bytes
	mov	x0, x21
	bl	utoa
	// newline
	mov	w2, #10
	strb	w2, [x1], #1
	// compute length = x1 - outbuf
	adrp	x2, outbuf
	add	x2, x2, :lo12:outbuf
	sub	x3, x1, x2		// length in x3
	// write syscall
	mov	x0, #1			// fd stdout
	mov	x1, x2			// buf
	mov	x2, x3			// count
	mov	x8, #64
	svc	#0
	// exit
	mov	x0, #0
	mov	x8, #93
	svc	#0

// -------- utoa: convert unsigned x0 to decimal string at [x1], returns with x1 advanced ----------
	.align 2
utoa:
	// if x0 == 0 => store '0' and return
	cbnz	x0, utoa_nonzero
	// zero case
	mov	w2, #'0'
	strb	w2, [x1], #1
	ret
utoa_nonzero:
	// x2 = digitbuf_end
	adrp	x2, digitbuf_end
	add	x2, x2, :lo12:digitbuf_end
	// divisor 10 in x3
	mov	x3, #10
utoa_loop:
	udiv	x4, x0, x3          // quotient
	mul	x5, x4, x3
	sub	x5, x0, x5          // remainder
	sub	x2, x2, #1
	add	w6, w5, #'0'
	strb	w6, [x2]
	mov	x0, x4
	cbnz	x0, utoa_loop
	// copy digits to destination
	adrp	x7, digitbuf_end
	add	x7, x7, :lo12:digitbuf_end
utoa_copy:
	cmp	x2, x7
	b.eq	utoa_done
	ldrb	w8, [x2], #1
	strb	w8, [x1], #1
	b	utoa_copy
utoa_done:
	ret

	.section .bss
	.align 3
input_buf:
	.space	65536
digitbuf:
	.space	32
digitbuf_end:
outbuf:
	.space	64
