	.global	_start
	.text
_start:
	// ---------- read entire stdin into prog_buf ----------
	mov	x0, #0				// fd = stdin
	adrp	x1, prog_buf
	add	x1, x1, :lo12:prog_buf		// buffer address
	mov	x3, #0				// total length = 0

read_loop:
	cmp	x3, #65536
	bge	read_done
	adrp	x2, prog_buf
	add	x2, x2, :lo12:prog_buf
	add	x2, x2, x3			// buffer + offset
	mov	x4, #65536
	sub	x4, x4, x3			// remaining space
	mov	x8, #63			// sys_read
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x3, x3, x0
	b	read_loop

read_done:
	// ---------- split program and input at first '!' ----------
	mov	x5, x3				// total_len
	mov	x6, #0				// i = 0

find_excl:
	cmp	x6, x5
	bge	no_excl
	adrp	x7, prog_buf
	add	x7, x7, :lo12:prog_buf
	add	x7, x7, x6
	ldrb	w8, [x7]
	cmp	w8, #33			// '!'
	bne	inc_find
	// found
	mov	x9, x6				// prog_len = i
	add	x10, x6, #1			// input_start = i+1
	sub	x11, x5, x10			// input_len = total_len - input_start
	b	after_split

inc_find:
	add	x6, x6, #1
	b	find_excl

no_excl:
	// no '!'
	mov	x9, x5				// prog_len = total_len
	mov	x11, #0				// input_len = 0
	add	x10, x5, #0			// input_start = total_len (empty)

after_split:
	// x9 = prog_len, x10 = input_start offset, x11 = input_len

	// ---------- set up pointers ----------
	adrp	x12, tape
	add	x12, x12, :lo12:tape		// tape base
	mov	x13, x12				// data pointer (address)

	adrp	x14, prog_buf
	add	x14, x14, :lo12:prog_buf	// program base

	adrp	x15, prog_buf
	add	x15, x15, :lo12:prog_buf
	add	x15, x15, x10			// input_ptr = prog_buf + input_start
	add	x16, x15, x11			// input_end = input_ptr + input_len

	adrp	x17, br_map
	add	x17, x17, :lo12:br_map		// bracket map base

	// ---------- bracket balance check and map ----------
	mov	x18, #0				// stack depth
	mov	x19, #0				// i = 0

bracket_loop:
	cmp	x19, x9
	bge	brackets_done
	adrp	x20, prog_buf
	add	x20, x20, :lo12:prog_buf
	add	x20, x20, x19
	ldrb	w21, [x20]

	// '[' ?
	cmp	w21, #0x5b			// '['
	bne	not_open
	// push i onto stack
	adrp	x22, br_stack
	add	x22, x22, :lo12:br_stack
	lsl	x23, x18, #2
	add	x23, x23, x22
	str	w19, [x23]
	add	x18, x18, #1
	b	inc_i

not_open:
	// ']' ?
	cmp	w21, #0x5d			// ']'
	bne	not_close
	cmp	x18, #0
	beq	bracket_unbalanced
	sub	x18, x18, #1
	adrp	x22, br_stack
	add	x22, x22, :lo12:br_stack
	lsl	x23, x18, #2
	add	x23, x23, x22
	ldr	w24, [x23]			// open index

	// forward map for '[' -> close+1
	add	w25, w19, #1			// target = i+1
	adrp	x26, br_map
	add	x26, x26, :lo12:br_map
	lsl	x27, w24, #2
	add	x27, x27, x26
	str	w25, [x27]

	// backward map for ']' -> open+1
	add	w25, w24, #1
	lsl	x27, w19, #2
	add	x27, x27, x26
	str	w25, [x27]

	b	inc_i

not_close:
	// other character
	// fall through

inc_i:
	add	x19, x19, #1
	b	bracket_loop

bracket_unbalanced:
	mov	x0, #1
	mov	x8, #93			// sys_exit
	svc	#0

brackets_done:
	cmp	x18, #0
	bne	bracket_unbalanced

	// ---------- execution ----------
	mov	x28, #0				// pc = 0

exec_loop:
	cmp	x28, x9
	bge	exec_done
	adrp	x29, prog_buf
	add	x29, x29, :lo12:prog_buf
	add	x29, x29, x28
	ldrb	w30, [x29]

	// '>'
	cmp	w30, #0x3e
	beq	cmd_inc_ptr
	// '<'
	cmp	w30, #0x3c
	beq	cmd_dec_ptr
	// '+'
	cmp	w30, #0x2b
	beq	cmd_inc_cell
	// '-'
	cmp	w30, #0x2d
	beq	cmd_dec_cell
	// '.'
	cmp	w30, #0x2e
	beq	cmd_output
	// ','
	cmp	w30, #0x2c
	beq	cmd_input
	// '['
	cmp	w30, #0x5b
	beq	cmd_bracket_open
	// ']'
	cmp	w30, #0x5d
	beq	cmd_bracket_close
	// ignore
	add	x28, x28, #1
	b	exec_loop

cmd_inc_ptr:
	add	x13, x13, #1
	add	x28, x28, #1
	b	exec_loop

cmd_dec_ptr:
	sub	x13, x13, #1
	add	x28, x28, #1
	b	exec_loop

cmd_inc_cell:
	ldrb	w0, [x13]
	add	w0, w0, #1
	and	w0, w0, #0xff
	strb	w0, [x13]
	add	x28, x28, #1
	b	exec_loop

cmd_dec_cell:
	ldrb	w0, [x13]
	sub	w0, w0, #1
	and	w0, w0, #0xff
	strb	w0, [x13]
	add	x28, x28, #1
	b	exec_loop

cmd_output:
	mov	x0, #1				// fd = stdout
	mov	x1, x13				// buffer = cell address
	mov	x2, #1
	mov	x8, #64			// sys_write
	svc	#0
	add	x28, x28, #1
	b	exec_loop

cmd_input:
	cmp	x15, x16
	bge	input_eof
	ldrb	w0, [x15]
	strb	w0, [x13]
	add	x15, x15, #1
	b	input_done
input_eof:
	mov	w0, #0
	strb	w0, [x13]
input_done:
	add	x28, x28, #1
	b	exec_loop

cmd_bracket_open:
	ldrb	w0, [x13]
	cbnz	w0, bracket_open_continue
	// jump forward
	adrp	x1, br_map
	add	x1, x1, :lo12:br_map
	mov	x2, x28
	lsl	x2, x2, #2
	add	x2, x2, x1
	ldr	w3, [x2]
	mov	x28, w3
	b	exec_loop
bracket_open_continue:
	add	x28, x28, #1
	b	exec_loop

cmd_bracket_close:
	ldrb	w0, [x13]
	cbz	w0, bracket_close_continue
	// jump back
	adrp	x1, br_map
	add	x1, x1, :lo12:br_map
	mov	x2, x28
	lsl	x2, x2, #2
	add	x2, x2, x1
	ldr	w3, [x2]
	mov	x28, w3
	b	exec_loop
bracket_close_continue:
	add	x28, x28, #1
	b	exec_loop

exec_done:
	mov	x0, #0
	mov	x8, #93			// sys_exit
	svc	#0

	// ---------- data ----------
	.bss
	.align	4
prog_buf:
	.zero	65536
tape:
	.zero	30000
br_map:
	.zero	262144			// 65536 * 4
br_stack:
	.zero	4000			// 1000 * 4
