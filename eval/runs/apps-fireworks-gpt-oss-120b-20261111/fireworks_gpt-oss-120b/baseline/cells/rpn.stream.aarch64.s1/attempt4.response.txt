.global _start
.text

/* system call numbers */
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

/* constants */
.equ INPUT_MAX, 65536
.equ STACK_MAX, 1024          // per line stack entries
.equ NUMBUF_SIZE, 24          // sign + up to 20 digits + extra

/* sections */
.section .bss
.align 8
input_buf:
    .zero INPUT_MAX
stack_arr:
    .zero STACK_MAX*8
num_buf:
    .zero NUMBUF_SIZE

.section .rodata
error_msg:
    .ascii "error\n"
error_len = . - error_msg
nl_char:
    .byte 0x0a

.section .text
_start:
    /* Load addresses */
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf          // x19 = input buffer base
    adrp    x9, stack_arr
    add     x9, x9, :lo12:stack_arr            // x9 = stack base

    /* Read entire input */
    mov     x20, #0                           // total bytes read
read_loop:
    mov     x2, #INPUT_MAX
    sub     x2, x2, x20
    cbz     x2, read_done
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #SYS_READ
    svc     #0
    cmp     x0, #0
    beq     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    cbz     x20, exit_success

    /* Pointers */
    mov     x21, x19                     // cur = input_buf
    add     x22, x19, x20                // end = input_buf + total

process_loop:
    cmp     x21, x22
    b.ge    exit_success

    /* Find line end */
    mov     x24, x21                     // line_start
line_find:
    cmp     x21, x22
    b.ge    line_no_nl
    ldrb    w0, [x21]
    cmp     w0, #10                      // '\n'
    beq     line_has_nl
    add     x21, x21, #1
    b       line_find
line_has_nl:
    mov     x25, x21                     // line_end (exclusive, before newline)
    b       line_process
line_no_nl:
    mov     x25, x22                     // line_end = end (no newline)

line_process:
    /* Evaluate line [x24, x25) */
    mov     x26, #0                      // stack size
    mov     w27, #0                      // error flag

    mov     x23, x24                     // token pointer

token_loop:
    cmp     x23, x25
    b.ge    token_done

    /* Skip separators (space, tab) */
    ldrb    w0, [x23]
    cmp     w0, #32
    beq     skip_sep
    cmp     w0, #9
    beq     skip_sep

    /* Token start */
    mov     x28, x23                     // token_start

scan_token:
    cmp     x23, x25
    b.ge    token_end
    ldrb    w0, [x23]
    cmp     w0, #32
    beq     token_end
    cmp     w0, #9
    beq     token_end
    add     x23, x23, #1
    b       scan_token
token_end:
    mov     x29, x23                     // token_end
    sub     x30, x29, x28                // token_len in x30

    cmp     w27, #0
    b.ne    after_token

    /* Check for operator (single char) */
    cmp     x30, #1
    b.ne    try_literal
    ldrb    w0, [x28]
    cmp     w0, #'+'
    beq     op_add
    cmp     w0, #'-'
    beq     op_sub
    cmp     w0, #'*'
    beq     op_mul
    cmp     w0, #'/'
    beq     op_div
    cmp     w0, #'%'
    beq     op_mod
    b       try_literal

/* ----- operators ----- */
op_add:
    cmp     x26, #2
    blt     set_error
    /* b = stack[sp-1] */
    sub     x5, x26, #1
    lsl     x5, x5, #3
    ldr     x5, [x9, x5]
    /* a = stack[sp-2] */
    sub     x6, x26, #2
    lsl     x6, x6, #3
    ldr     x6, [x9, x6]
    add     x6, x6, x5
    /* store result */
    sub     x7, x26, #2
    lsl     x7, x7, #3
    str     x6, [x9, x7]
    sub     x26, x26, #1
    b       after_token

op_sub:
    cmp     x26, #2
    blt     set_error
    sub     x5, x26, #1
    lsl     x5, x5, #3
    ldr     x5, [x9, x5]          // b
    sub     x6, x26, #2
    lsl     x6, x6, #3
    ldr     x6, [x9, x6]          // a
    sub     x6, x6, x5
    sub     x7, x26, #2
    lsl     x7, x7, #3
    str     x6, [x9, x7]
    sub     x26, x26, #1
    b       after_token

op_mul:
    cmp     x26, #2
    blt     set_error
    sub     x5, x26, #1
    lsl     x5, x5, #3
    ldr     x5, [x9, x5]          // b
    sub     x6, x26, #2
    lsl     x6, x6, #3
    ldr     x6, [x9, x6]          // a
    mul     x6, x6, x5
    sub     x7, x26, #2
    lsl     x7, x7, #3
    str     x6, [x9, x7]
    sub     x26, x26, #1
    b       after_token

op_div:
    cmp     x26, #2
    blt     set_error
    sub     x5, x26, #1
    lsl     x5, x5, #3
    ldr     x5, [x9, x5]          // b
    cbz     x5, set_error         // division by zero
    sub     x6, x26, #2
    lsl     x6, x6, #3
    ldr     x6, [x9, x6]          // a
    /* overflow: a == INT64_MIN && b == -1 */
    movz    x7, #0x8000, lsl #48   // INT64_MIN
    cmp     x6, x7
    b.ne    div_no_overflow
    mov     x8, #-1
    cmp     x5, x8
    b.ne    div_no_overflow
    b       set_error
div_no_overflow:
    sdiv    x6, x6, x5
    sub     x7, x26, #2
    lsl     x7, x7, #3
    str     x6, [x9, x7]
    sub     x26, x26, #1
    b       after_token

op_mod:
    cmp     x26, #2
    blt     set_error
    sub     x5, x26, #1
    lsl     x5, x5, #3
    ldr     x5, [x9, x5]          // b
    cbz     x5, set_error         // division by zero
    sub     x6, x26, #2
    lsl     x6, x6, #3
    ldr     x6, [x9, x6]          // a
    /* overflow check */
    movz    x7, #0x8000, lsl #48   // INT64_MIN
    cmp     x6, x7
    b.ne    mod_no_overflow
    mov     x8, #-1
    cmp     x5, x8
    b.ne    mod_no_overflow
    b       set_error
mod_no_overflow:
    sdiv    x8, x6, x5            // quotient
    mul     x11, x8, x5           // q*b
    sub     x6, x6, x11           // remainder
    sub     x7, x26, #2
    lsl     x7, x7, #3
    str     x6, [x9, x7]
    sub     x26, x26, #1
    b       after_token

set_error:
    mov     w27, #1
    b       after_token

try_literal:
    /* Parse integer literal */
    mov     x31, x28               // ptr = token_start
