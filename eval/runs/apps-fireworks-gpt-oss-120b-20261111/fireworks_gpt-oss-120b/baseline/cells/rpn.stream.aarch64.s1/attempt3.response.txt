.global _start
.text

/* system call numbers */
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

/* constants */
.equ INPUT_MAX, 65536
.equ STACK_MAX, 1024          // max stack entries per line
.equ NUMBUF_SIZE, 24          // enough for sign + 20 digits + newline

/* sections */
.section .bss
.align 8
input_buf:
    .zero INPUT_MAX
stack_arr:
    .zero STACK_MAX*8
num_buf:
    .zero NUMBUF_SIZE

.section .rodata
error_msg:
    .ascii "error\n"
error_len = . - error_msg

.section .text
_start:
    /* ---- read whole input into input_buf ---- */
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf          // x19 = input buffer base
    mov     x20, #0                           // total bytes read

read_loop:
    mov     x2, #INPUT_MAX
    sub     x2, x2, x20
    cbz     x2, read_done
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #SYS_READ
    svc     #0
    cmp     x0, #0
    beq     read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    /* if no input, exit */
    cbz     x20, exit_success

    /* pointers */
    add     x21, x19, x20                     // x22 = end pointer
    mov     x22, x21
    mov     x21, x19                          // x21 = current pointer

process_input:
    cmp     x21, x22
    b.ge    exit_success                      // done

    /* find end of current line */
    mov     x24, x21                          // line_start = x24
line_find_nl:
    cmp     x21, x22
    b.ge    line_no_nl
    ldrb    w0, [x21]
    cmp     w0, #10                           // '\n'
    beq     line_found_nl
    add     x21, x21, #1
    b       line_find_nl
line_found_nl:
    /* line_end = x21 (points to '\n') */
    b       line_process
line_no_nl:
    /* reached end without newline, line_end = x22 */
    mov     x21, x22

line_process:
    /* line_start in x24, line_end = x21 (points to NL or end) */
    mov     x23, x24                          // cur pointer within line
    mov     x25, x22                          // line_end ptr
    /* initialize stack */
    mov     x26, #0                           // stack size
    mov     w27, #0                           // error flag (0=no,1=yes)

token_loop:
    cmp     x23, x25
    b.ge    line_finish

    ldrb    w0, [x23]
    /* skip separators: space (0x20) and tab (0x09) */
    cmp     w0, #32
    beq     token_skip_sep
    cmp     w0, #9
    beq     token_skip_sep
    /* token start */
    mov     x28, x23                          // token_start
    /* find token end */
token_scan:
    cmp     x23, x25
    b.ge    token_end
    ldrb    w0, [x23]
    cmp     w0, #32
    beq     token_end
    cmp     w0, #9
    beq     token_end
    cmp     w0, #10
    beq     token_end
    add     x23, x23, #1
    b       token_scan
token_end:
    mov     x29, x23
    sub     x30, x29, x28                     // token_len in x30

    /* if already error, ignore token */
    cmp     w27, #0
    b.ne    token_done

    /* check if operator (single char) */
    cmp     x30, #1
    b.ne    try_literal
    ldrb    w0, [x28]
    cmp     w0, #'+'
    beq     op_add
    cmp     w0, #'-'
    beq     op_sub
    cmp     w0, #'*'
    beq     op_mul
    cmp     w0, #'/'
    beq     op_div
    cmp     w0, #'%'
    beq     op_mod
    b       try_literal

/* ------------ operators --------------- */
op_add:
    cmp     x26, #2
    blt     token_error
    sub     x26, x26, #1
    ldr     x5, [stack_arr, x26, LSL #3]      // b
