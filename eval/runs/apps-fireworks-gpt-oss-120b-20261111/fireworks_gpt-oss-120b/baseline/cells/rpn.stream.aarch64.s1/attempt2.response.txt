.global _start
.text
_start:
    // addresses of buffers
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf          // x19 = input buffer base
    adrp    x23, stack_arr
    add     x23, x23, :lo12:stack_arr          // x23 = stack base
    adrp    x27, outbuf
    add     x27, x27, :lo12:outbuf             // x27 = output buffer base

    // read all input (max 65536 bytes)
    mov     x20, #0                           // total bytes read
read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    add     x1, x19, x20
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    beq     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    cbz     x20, exit_success

    // parsing pointers
    mov     x21, x19                          // cur pointer
    add     x22, x19, x20                     // end pointer
    mov     x24, #0                           // stack size
    mov     x25, #0                           // error flag for current line

parse_loop:
    cmp     x21, x22
    b.ge    eof_process

    ldrb    w0, [x21]
    cmp     w0, #10
    beq     line_newline
    cmp     w0, #32
    beq     inc_cur
    cmp     w0, #9
    beq     inc_cur

    // token start
    mov     x26, x21                          // token_start

token_scan:
    cmp     x21, x22
    b.ge    token_end
    ldrb    w0, [x21]
    cmp     w0, #10
    beq     token_end
    cmp     w0, #32
    beq     token_end
    cmp     w0, #9
    beq     token_end
    add     x21, x21, #1
    b       token_scan
token_end:
    sub     x27, x21, x26                     // token_len

    // if error already set, ignore token
    cmp     x25, #0
    b.ne    token_done

    // operator check (single-char)
    cmp     x27, #1
    b.ne    try_literal
    ldrb    w0, [x26]
    cmp     w0, #'+'
    beq     op_add
    cmp     w0, #'-'
    beq     op_sub
    cmp     w0, #'*'
    beq     op_mul
    cmp     w0, #'/'
    beq     op_div
    cmp     w0, #'%'
    beq     op_mod
    // not operator => literal
    b       try_literal

op_add:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]           // b
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]           // a
    add     x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_sub:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]
    sub     x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_mul:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]
    mul     x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_div:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]           // b
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]           // a
    cbz     x5, token_error                  // divide by zero
    // INT64_MIN = 0x8000000000000000 in x6
    movz    x6, #0x0000, LSL #0
    movk    x6, #0x8000, LSL #48
    cmp     x4, x6
    b.ne    div_ok
    mov     x7, #-1
    cmp     x5, x7
    b.ne    div_ok
    b       token_error
div_ok:
    sdiv    x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_mod:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]           // b
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]           // a
    cbz     x5, token_error                  // mod by zero
    movz    x6, #0x0000, LSL #0
    movk    x6, #0x8000, LSL #48
    cmp     x4, x6
    b.ne    mod_ok
    mov     x7, #-1
    cmp     x5, x7
    b.ne    mod_ok
    b       token_error
mod_ok:
    sdiv    x7, x4, x5
    mul     x8, x7, x5
    sub     x6, x4, x8
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

try_literal:
    // token at [x26, x27)
    mov     x28, #0          // sign flag
    mov     x29, #0          // digit count
    mov     x14, #0          // accumulator
    mov     x15, #0          // index

    // optional leading '-'
    cmp     x27, #0
    beq     token_error
    ldrb    w0, [x26]
    cmp     w0, #'-'
    b.ne    lit_no_sign
    mov     x28, #1
    add     x15, x15, #1
    cmp     x15, x27
    beq     token_error
lit_no_sign:
lit_loop:
    cmp     x15, x27
    beq     lit_done
    add     x9, x26, x15
    ldrb    w0, [x9]
    cmp     w0, #48          // '0'
    blt     token_error
    cmp     w0, #57          // '9'
    bgt     token_error
    add     x29, x29, #1
    cmp     x29, #18
    bgt     token_error
    // acc = acc*10 + (w0 - '0')
    mov     x0, x14
    mov     x1, #10
    mul     x14, x0, x1
    sub     x2, w0, #48
