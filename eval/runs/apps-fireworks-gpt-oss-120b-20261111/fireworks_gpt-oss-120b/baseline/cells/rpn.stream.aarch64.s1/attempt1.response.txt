    .global _start
    .text
_start:
    // -------------------------------------------------
    // Set up addresses of .bss buffers
    adrp    x19, input_buf            // x19 = &input_buf
    add     x19, x19, :lo12:input_buf

    adrp    x23, stack_arr            // x23 = &stack_arr
    add     x23, x23, :lo12:stack_arr

    // -------------------------------------------------
    // Read entire input (max 65536 bytes) into input_buf
    mov     x20, #0                   // x20 = total bytes read

read_loop:
    // remaining = 65536 - x20
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done

    // address = input_buf + x20
    add     x1, x19, x20

    mov     x0, #0                    // fd = stdin
    mov     x8, #63                   // sys_read
    svc     #0
    cmp     x0, #0
    beq     read_done
    // x0 = bytes read
    add     x20, x20, x0
    b       read_loop

read_done:
    // if no input, exit normally
    cbz     x20, exit_success

    // -------------------------------------------------
    // Initialize parsing pointers
    mov     x21, x19                  // cur = input_buf
    add     x22, x19, x20             // end = input_buf + total

    mov     x24, #0                   // stack size = 0
    mov     x25, #0                   // error flag = 0

parse_loop:
    // if cur >= end, finish
    cmp     x21, x22
    b.ge    exit_success

    ldrb    w0, [x21]                 // current byte
    cmp     w0, #10                   // newline ?
    beq     line_end

    // skip spaces and tabs
    cmp     w0, #32
    beq     skip_sep
    cmp     w0, #9
    beq     skip_sep

    // ---------------- token start --------------------
    mov     x26, x21                  // token_start
    // find token end
token_scan:
    cmp     x21, x22
    beq     token_end
    ldrb    w0, [x21]
    cmp     w0, #10
    beq     token_end
    cmp     w0, #32
    beq     token_end
    cmp     w0, #9
    beq     token_end
    // not separator
    add     x21, x21, #1
    b       token_scan
token_end:
    sub     x27, x21, x26             // token_len = cur - start

    // -------------------------------------------------
    // If already in error, ignore token
    cmp     x25, #0
    b.ne    token_skip

    // Check for single-char operators
    cmp     x27, #1
    b.ne    try_literal
    ldrb    w0, [x26]                 // first char
    cmp     w0, #'+'
    beq     op_add
    cmp     w0, #'-'
    beq     op_sub
    cmp     w0, #'*'
    beq     op_mul
    cmp     w0, #'/'
    beq     op_div
    cmp     w0, #'%'
    beq     op_mod
    // not an operator, treat as literal
    b       try_literal

    // ---------------- operator implementations ----------------
op_add:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]   // b
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]   // a
    add     x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_sub:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]
    sub     x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_mul:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]
    mul     x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_div:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]   // b
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]   // a
    cbz     x5, token_error          // divide by zero

    // generate INT64_MIN constant in x6
    movz    x6, #0x0000, LSL #0
    movk    x6, #0x8000, LSL #48
    cmp     x4, x6
    b.ne    div_no_overflow
    mov     x7, #-1
    cmp     x5, x7
    b.ne    div_no_overflow
    b       token_error
div_no_overflow:
    sdiv    x6, x4, x5
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

op_mod:
    cmp     x24, #2
    blt     token_error
    sub     x24, x24, #1
    ldr     x5, [x23, x24, LSL #3]   // b
    sub     x24, x24, #1
    ldr     x4, [x23, x24, LSL #3]   // a
    cbz     x5, token_error          // mod by zero

    // generate INT64_MIN constant in x6
    movz    x6, #0x0000, LSL #0
    movk    x6, #0x8000, LSL #48
    cmp     x4, x6
    b.ne    mod_no_overflow
    mov     x7, #-1
    cmp     x5, x7
    b.ne    mod_no_overflow
    b       token_error
mod_no_overflow:
    sdiv    x7, x4, x5                // quotient
    mul     x8, x7, x5                // q*b
    sub     x6, x4, x8                // remainder
    str     x6, [x23, x24, LSL #3]
    add     x24, x24, #1
    b       token_done

    // -------------------------------------------------
    // Try to parse as integer literal
try_literal:
    // token in [x26, x27)
    // sign flag in x28 (0 positive, 1 negative)
    // digit count in x29
    // accumulator in x14
    mov     x28, #0
    mov     x29, #0
    mov     x14, #0
    mov     x15, #0               // offset i

    // check for leading '-'
    cmp     x27, #0
    beq     token_error           // empty token shouldn't happen
    ldrb    w0, [x26]
    cmp     w0, #'-'
    b.ne    lit_no_sign
    // sign negative
    mov     x28, #1
    add     x15, x15, #1
    // ensure at least one more byte
    cmp     x15, x27
    beq     token_error

lit_no_sign:
lit_loop:
    cmp     x15, x27
    beq     lit_done
    add     x9, x26, x15
    ldrb    w0, [x9]
    // check digit
    cmp     w0, #'0'
    blt     token_error
    cmp     w0, #'9'
    bgt     token_error
    // digit ok
    add     x29, x29, #1
    // acc = acc*10 + (w0-'0')
    mov     x0, x14
    mov     x1, #10
    mul     x0, x0, x1
    sub     x2, w0, #'0'
