	.global _start
	.text
	.p2align 2

_start:
	// buffer address in x19
	adrp	x19, buffer
	add	x19, x19, :lo12:buffer
	// outbuf address in x27
	adrp	x27, outbuf
	add	x27, x27, :lo12:outbuf
	// length = 0
	mov	x20, #0
	// cur = 0
	mov	x21, #0

read_loop:
	// remaining = BUFFER_SIZE - length
	mov	x2, #65536
	sub	x2, x2, x20
	cbnz	x2, .Lread_continue
	b	.read_done

.Lread_continue:
	// x1 = buffer + length
	add	x1, x19, x20
	// fd = 0
	mov	x0, #0
	// syscall 63 (read)
	mov	x8, #63
	svc	#0
	// result in x0 = bytes read
	cbnz	x0, .Lgot_bytes
	// EOF
	b	.read_done

.Lgot_bytes:
	add	x20, x20, x0
	b	read_loop

.read_done:
	// processing loop
process_loop:
	cmp	x21, x20
	bge	.exit_success

	// line_end = cur
	mov	x22, x21

scan_loop:
	cmp	x22, x20
	bge	.no_newline
	// load byte at buffer+line_end
	add	x0, x19, x22
	ldrb	w3, [x0]
	cmp	w3, #10
	beq	.found_newline
	add	x22, x22, #1
	b	scan_loop

.found_newline:
	// length = line_end - cur
	sub	x23, x22, x21
	cbnz	x23, .rev_write_loop
	// empty line, just write newline
	b	.write_newline

.rev_write_loop:
	// counter = length
	mov	x24, x23
.rev_write_iter:
	cbnz	x24, .rev_write_body
	b	.write_newline
.rev_write_body:
	// offset = cur + (counter-1)
	sub	x25, x24, #1
	add	x26, x21, x25
	add	x0, x19, x26
	ldrb	w3, [x0]
	// store to outbuf
	strb	w3, [x27]
	// write(fd=1, outbuf, 1)
	mov	x0, #1
	mov	x1, x27
	mov	x2, #1
	mov	x8, #64
	svc	#0
	// decrement counter
	sub	x24, x24, #1
	b	.rev_write_iter

.write_newline:
	// store newline to outbuf
	mov	w3, #10
	strb	w3, [x27]
	// write newline
	mov	x0, #1
	mov	x1, x27
	mov	x2, #1
	mov	x8, #64
	svc	#0

	// skip newline if present
	cmp	x22, x20
	bge	.set_cur_done
	add	x0, x19, x22
	ldrb	w3, [x0]
	cmp	w3, #10
	bne	.set_cur_done
	add	x22, x22, #1

.set_cur_done:
	// cur = line_end (or line_end+1 if newline skipped)
	mov	x21, x22
	b	process_loop

.no_newline:
	// length = len - cur
	sub	x23, x20, x21
	cbnz	x23, .rev_write_loop2
	// no more data
	b	.exit_success

.rev_write_loop2:
	// counter = length
	mov	x24, x23
.rev_write_iter2:
	cbnz	x24, .rev_write_body2
	b	.write_newline2
.rev_write_body2:
	sub	x25, x24, #1
	add	x26, x21, x25
	add	x0, x19, x26
	ldrb	w3, [x0]
	strb	w3, [x27]
	mov	x0, #1
	mov	x1, x27
	mov	x2, #1
	mov	x8, #64
	svc	#0
	sub	x24, x24, #1
	b	.rev_write_iter2

.write_newline2:
	mov	w3, #10
	strb	w3, [x27]
	mov	x0, #1
	mov	x1, x27
	mov	x2, #1
	mov	x8, #64
	svc	#0
	// cur = len (end)
	mov	x21, x20
	b	process_loop

.exit_success:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.section .bss
	.p2align 3
buffer:
	.space	65536
outbuf:
	.space	1
