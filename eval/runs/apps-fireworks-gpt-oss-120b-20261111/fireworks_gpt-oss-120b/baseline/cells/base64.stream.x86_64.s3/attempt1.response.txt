.text
.globl _start
_start:
    xor     r12, r12                # total_read = 0

read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0
    lea     rsi, [rip + input_buf]
    add     rsi, r12
    mov     rdx, INPUT_MAX
    sub     rdx, r12
    syscall
    test    rax, rax
    jz      encode_start
    add     r12, rax
    jmp     read_loop

encode_start:
    cmp     r12, 0
    je      exit_program

    xor     r13, r13                # input offset
    xor     r14, r14                # output offset
    xor     r15d, r15d              # line position
    lea     r8, [rip + input_buf]   # r8 = input base
    lea     r9, [rip + output_buf]  # r9 = output base
    lea     rbx, [rip + b64_table]  # rbx = base64 table

encode_loop:
    mov     rax, r12
    sub     rax, r13
    cmp     rax, 3
    jl      tail

    movzx   eax, byte [r8 + r13]          # a
