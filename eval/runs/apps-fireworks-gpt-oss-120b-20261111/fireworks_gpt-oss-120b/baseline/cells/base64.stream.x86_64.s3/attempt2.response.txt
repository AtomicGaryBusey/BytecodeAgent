.globl _start

.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 66000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buf:
    .zero INPUT_MAX
output_buf:
    .zero OUTPUT_MAX

.text
_start:
    xor    r12, r12                # total_read = 0
read_loop:
    mov    rax, 0                  # sys_read
    mov    rdi, 0
    lea    rsi, [rip + input_buf]
    add    rsi, r12
    mov    rdx, INPUT_MAX
    sub    rdx, r12
    syscall
    test   rax, rax
    jz     encode_start
    add    r12, rax
    cmp    r12, INPUT_MAX
    jae    encode_start
    jmp    read_loop

encode_start:
    cmp    r12, 0
    je     exit_program

    xor    r13, r13                # input offset
    xor    r14, r14                # output offset
    xor    r15d, r15d              # line position (0..75)

    lea    rbx, [rip + b64_table]  # base64 table
    lea    r8,  [rip + input_buf]  # input base
    lea    r9,  [rip + output_buf] # output base

enc_loop:
    mov    rax, r12
    sub    rax, r13
    cmp    rax, 3
    jb     enc_tail

    movzx  eax, byte ptr [r8 + r13]        # a
    movzx  ecx, byte ptr [r8 + r13 + 1]    # b
    movzx  edx, byte ptr [r8 + r13 + 2]    # c
    shl    eax, 16
    shl    ecx, 8
    or     eax, ecx
    or     eax, edx                         # eax = 24 bits

    mov    esi, eax
    shr    esi, 18
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    esi, eax
    shr    esi, 12
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    esi, eax
    shr    esi, 6
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    esi, eax
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    add    r13, 3
    jmp    enc_loop

enc_tail:
    mov    rax, r12
    sub    rax, r13
    cmp    rax, 0
    je     enc_done
    cmp    rax, 1
    je     enc_one
    # remaining == 2
    movzx  eax, byte ptr [r8 + r13]
    movzx  ecx, byte ptr [r8 + r13 + 1]
    shl    eax, 16
    shl    ecx, 8
    or     eax, ecx                       # eax = bits

    mov    esi, eax
    shr    esi, 18
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    esi, eax
    shr    esi, 12
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    esi, eax
    shr    esi, 6
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    dil, '='
    call   write_char
    jmp    enc_done

enc_one:
    movzx  eax, byte ptr [r8 + r13]
    shl    eax, 16                        # bits

    mov    esi, eax
    shr    esi, 18
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    esi, eax
    shr    esi, 12
    and    esi, 0x3F
    movzx  edi, byte ptr [rbx + rsi]
    call   write_char

    mov    dil, '='
    call   write_char
    call   write_char
    jmp    enc_done

enc_done:
    cmp    r14, 0
    je     exit_program
    cmp    r15d, 0
    je     after_newline
    mov    dil, 0x0A
    call   write_char
after_newline:
    mov    rdi, 1
    mov    rsi, r9
    mov    rdx, r14
write_loop:
    test   rdx, rdx
    jz     exit_program
    mov    rax, 1          # sys_write
    syscall
    test   rax, rax
    jz     exit_program
    sub    rdx, rax
    add    rsi, rax
    jmp    write_loop

write_char:
    mov    byte ptr [r9 + r14], dil
    inc    r14
    inc    r15d
    cmp    r15d, 76
    jne    .wcn_end
    mov    byte ptr [r9 + r14], 0x0A
    inc    r14
    xor    r15d, r15d
.wcn_end:
    ret

exit_program:
    mov    edi, 0
    mov    eax, 60          # sys_exit
    syscall
