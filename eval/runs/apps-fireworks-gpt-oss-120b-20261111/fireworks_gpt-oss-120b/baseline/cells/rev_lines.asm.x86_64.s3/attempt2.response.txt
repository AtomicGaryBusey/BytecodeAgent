.section .bss
.align 16
buffer:
    .zero 65536
temp:
    .zero 1

.text
.globl _start
_start:
    xor     rbx, rbx                      # total bytes read = 0

read_loop:
    mov     rax, 0                        # sys_read
    mov     rdi, 0                        # fd = stdin
    lea     rsi, [rip + buffer]
    add     rsi, rbx                      # buffer + offset
    mov     rdx, 65536
    sub     rdx, rbx                      # remaining space
    test    rdx, rdx
    jz      read_done
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                      # total += bytes read
    jmp     read_loop

read_done:
    xor     rcx, rcx                      # current offset = 0

line_loop:
    cmp     rcx, rbx
    jae     exit_success                  # all processed

    mov     rdx, rcx                      # search pointer

search_newline:
    cmp     rdx, rbx
    jae     no_newline                    # reached end of input
    mov     al, byte ptr [rip + buffer + rdx]
    cmp     al, 0x0A
    je      newline_found
    inc     rdx
    jmp     search_newline

newline_found:
    mov     rsi, rdx                      # rsi = newline position
    sub     rsi, rcx                      # length = rsi - rcx
    test    rsi, rsi
    je      write_nl_only
    mov     r8, rsi                       # r8 = length

rev_loop:
    dec     r8
    mov     al, byte ptr [rip + buffer + rcx + r8]
    mov     byte ptr [rip + temp], al
    mov     rax, 1                        # sys_write
    mov     rdi, 1                        # fd = stdout
    lea     rsi, [rip + temp]
    mov     rdx, 1
    syscall
    cmp     r8, 0
    jne     rev_loop

write_nl_only:
    mov     al, 0x0A
    mov     byte ptr [rip + temp], al
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + temp]
    mov     rdx, 1
    syscall
    mov     rcx, rdx                      # rdx still holds newline position
    inc     rcx                           # advance past newline
    jmp     line_loop

no_newline:
    mov     rsi, rbx
    sub     rsi, rcx                      # length = total - rcx
    test    rsi, rsi
    je      exit_success                  # no remaining bytes
    mov     r8, rsi

rev_last_loop:
    dec     r8
    mov     al, byte ptr [rip + buffer + rcx + r8]
    mov     byte ptr [rip + temp], al
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + temp]
    mov     rdx, 1
    syscall
    cmp     r8, 0
    jne     rev_last_loop

    mov     al, 0x0A
    mov     byte ptr [rip + temp], al
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + temp]
    mov     rdx, 1
    syscall
    jmp     exit_success

exit_success:
    mov     rax, 60                       # sys_exit
    xor     rdi, rdi                      # status = 0
    syscall
