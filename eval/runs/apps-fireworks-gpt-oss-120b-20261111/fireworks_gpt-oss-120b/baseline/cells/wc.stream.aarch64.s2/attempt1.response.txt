	.text
	.global _start
	.type _start, %function
_start:
	// registers:
	// x9  = buffer base
	// x10 = lines counter
	// x11 = words counter
	// x12 = bytes counter
	// x13 = in_word flag (0/1)

	// initialize counters
	mov	x10, #0
	mov	x11, #0
	mov	x12, #0
	mov	x13, #0

	// buffer address
	adr	x9, buffer

	// buffer size = 65536 (1 << 16)
	movz	x20, #1, lsl #16

read_loop:
	// sys_read(fd=0, buf=x9, count=65536)
	mov	x0, #0                 // fd
	mov	x1, x9                // buf
	mov	x2, x20               // count
	mov	x8, #63               // sys_read
	svc	#0
	cmp	x0, #0
	beq	after_read            // end of input

	// x0 = nread
	mov	x17, x0               // remaining bytes in this chunk
	mov	x16, #0               // offset within buffer

process_loop:
	// load byte
	ldrb	w19, [x9, x16]

	// total bytes++
	add	x12, x12, #1

	// line count on '\n'
	cmp	w19, #0x0A
	bne	_not_newline
	add	x10, x10, #1
_not_newline:

	// whitespace test
	cmp	w19, #0x20
	beq	_is_ws
	cmp	w19, #0x09
	beq	_is_ws
	cmp	w19, #0x0A
	beq	_is_ws
	cmp	w19, #0x0B
	beq	_is_ws
	cmp	w19, #0x0C
	beq	_is_ws
	cmp	w19, #0x0D
	beq	_is_ws
	// not whitespace
	cmp	x13, #0
	bne	_not_word_start
	// start new word
	add	x11, x11, #1
	mov	x13, #1
	b	_continue_proc
_not_word_start:
	b	_continue_proc
_is_ws:
	// whitespace => leave word
	mov	x13, #0
_continue_proc:
	// advance offset
	add	x16, x16, #1
	subs	x17, x17, #1
	bne	process_loop

	// more data?
	b	read_loop

after_read:
	// x10 = lines, x11 = words, x12 = bytes
	// outbuf pointer in x14
	adr	x14, outbuf

	// ---- output lines ----
	mov	x0, x10
	adr	x1, numbuf
	bl	uint_to_str
	mov	x15, x0          // len
	subs	x2, x15, #1
lines_rev:
	ldrb	w3, [x1, x2]
	strb	w3, [x14], #1
	subs	x2, x2, #1
	bge	lines_rev
	// space separator
	mov	w3, #' '
	strb	w3, [x14], #1

	// ---- output words ----
	mov	x0, x11
	adr	x1, numbuf
	bl	uint_to_str
	mov	x15, x0
	subs	x2, x15, #1
words_rev:
	ldrb	w3, [x1, x2]
	strb	w3, [x14], #1
	subs	x2, x2, #1
	bge	words_rev
	// space separator
	mov	w3, #' '
	strb	w3, [x14], #1

	// ---- output bytes ----
	mov	x0, x12
	adr	x1, numbuf
	bl	uint_to_str
	mov	x15, x0
	subs	x2, x15, #1
bytes_rev:
	ldrb	w3, [x1, x2]
	strb	w3, [x14], #1
	subs	x2, x2, #1
	bge	bytes_rev
	// newline
	mov	w3, #0x0A
	strb	w3, [x14], #1

	// compute length = outptr - outbuf
	adr	x5, outbuf
	sub	x6, x14, x5   // x6 = length

	// write(fd=1, buf=outbuf, count=length)
	mov	x0, #1
	mov	x1, x5
	mov	x2, x6
	mov	x8, #64          // sys_write
	svc	#0

	// exit(0)
	mov	x0, #0
	mov	x8, #93          // sys_exit
	svc	#0

	// ----------------------------------------------------
	// uint_to_str: convert unsigned 64-bit in x0 to decimal string
	// Input: x0 = value, x1 = pointer to temporary buffer (at least 24 bytes)
	// Output: x0 = length, buffer filled with digits in reverse order
	// Clobbers: x2-x6, w3-w5
	.type uint_to_str, %function
uint_to_str:
	// x2 = length counter
	mov	x2, #0
	cbnz	x0, .Lconv_loop
	// value == 0
	mov	w3, #'0'
	strb	w3, [x1]
	mov	x2, #1
	b	.Lconv_done

.Lconv_loop:
	// divide by 10
	mov	x3, #10
	udiv	x4, x0, x3        // x4 = quotient
	msub	x5, x4, x3, x0    // x5 = remainder
	add	w5, w5, #'0'       // digit char
	// store at buffer+len
	add	x6, x1, x2
	strb	w5, [x6]
	add	x2, x2, #1
	mov	x0, x4
	cbnz	x0, .Lconv_loop

.Lconv_done:
	mov	x0, x2                // return length
	ret

	.bss
	.align 8
buffer:
	.space 65536
	.align 8
numbuf:
	.space 24
	.align 8
outbuf:
	.space 64
