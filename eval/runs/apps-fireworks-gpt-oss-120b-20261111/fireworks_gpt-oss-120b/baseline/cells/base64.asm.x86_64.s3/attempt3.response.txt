.globl _start
.section .bss
.align 16
buf_in:
    .zero 49152
buf_out:
    .zero 70000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor     r8, r8                  # total bytes read
read_loop:
    mov     rax, 0                  # sys_read
    xor     rdi, rdi                # fd 0
    lea     rsi, [rip + buf_in]
    add     rsi, r8                 # buf_in + offset
    mov     rdx, 49152
    sub     rdx, r8                 # remaining space
    syscall
    test    rax, rax
    jz      read_done
    add     r8, rax
    cmp     r8, 49152
    jb      read_loop
read_done:
    test    r8, r8
    jz      exit_success            # empty input, no output

    # rsi = src pointer, rcx = remaining input length
    lea     rsi, [rip + buf_in]
    mov     rcx, r8
    # rdi = dst pointer
    lea     rdi, [rip + buf_out]
    xor     r9d, r9d                # line length counter

encode_main:
    cmp     rcx, 3
    jb      encode_leftover

    # load three bytes into r8b, r9b, r10b
    mov     r8b,  [rsi]
    mov     r9b,  [rsi+1]
    mov     r10b, [rsi+2]

    # idx0 = b0 >> 2
    movzx   rdx, r8b
    shr     rdx, 2
    call    put_char_idx

    # idx1 = ((b0 & 3) << 4) | (b1 >> 4)
    movzx   rdx, r8b
    and     rdx, 3
    shl     rdx, 4
    movzx   rax, r9b
    shr     rax, 4
    or      rdx, rax
    call    put_char_idx

    # idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    movzx   rdx, r9b
    and     rdx, 0xF
    shl     rdx, 2
    movzx   rax, r10b
    shr     rax, 6
    or      rdx, rax
    call    put_char_idx

    # idx3 = b2 & 0x3F
    movzx   rdx, r10b
    and     rdx, 0x3F
    call    put_char_idx

    add     rsi, 3
    sub     rcx, 3
    jmp     encode_main

encode_leftover:
    cmp     rcx, 0
    je      finish_output
    cmp     rcx, 1
    je      encode_one

    # rcx == 2
    mov     r8b, [rsi]          # b0
    mov     r9b, [rsi+1]        # b1

    # idx0 = b0 >> 2
    movzx   rdx, r8b
    shr     rdx, 2
    call    put_char_idx

    # idx1 = ((b0 & 3) << 4) | (b1 >> 4)
    movzx   rdx, r8b
    and     rdx, 3
    shl     rdx, 4
    movzx   rax, r9b
    shr     rax, 4
    or      rdx, rax
    call    put_char_idx

    # idx2 = ((b1 & 0xF) << 2)
    movzx   rdx, r9b
    and     rdx, 0xF
    shl     rdx, 2
    call    put_char_idx

    # padding '=' twice
    mov     al, '='
    call    put_char
    call    put_char
    jmp     finish_output

encode_one:
    mov     r8b, [rsi]          # b0

    # idx0 = b0 >> 2
    movzx   rdx, r8b
    shr     rdx, 2
    call    put_char_idx

    # idx1 = (b0 & 3) << 4
    movzx   rdx, r8b
    and     rdx, 3
    shl     rdx, 4
    call    put_char_idx

    # padding '=' twice
    mov     al, '='
    call    put_char
    call    put_char
    jmp     finish_output

# -------------------------------------------------
# put_char_idx: rdx holds 0..63, loads character from table into al, then falls through to put_char
put_char_idx:
    lea     rbx, [rip + b64_table]
    mov     al, [rbx + rdx]
    # fall through

# put_char: al holds character to write
put_char:
    mov     [rdi], al
    inc     rdi
    inc     r9d
    cmp     r9d, 76
    jne     .no_nl
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r9d, r9d
.no_nl:
    ret

finish_output:
    cmp     r9d, 0
    je      output_done
    mov     byte ptr [rdi], 0x0A
    inc     rdi
output_done:
    # write out buffer
    lea     rsi, [rip + buf_out]
    mov     rdx, rdi
    sub     rdx, rsi            # length
    mov     rax, 1              # sys_write
    mov     rdi, 1              # stdout
    syscall

exit_success:
    mov     rax, 60
    xor     rdi, rdi
    syscall
