.globl _start
.equ IN_BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 70000

.section .bss
.align 16
buf_in:
    .zero IN_BUF_SIZE
buf_out:
    .zero OUT_BUF_SIZE

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    xor     r8, r8                     # total bytes read
read_loop:
    mov     rax, 0                     # sys_read
    xor     rdi, rdi                   # fd 0
    lea     rsi, [rip + buf_in]
    add     rsi, r8
    mov     rdx, IN_BUF_SIZE
    sub     rdx, r8
    syscall
    test    rax, rax
    jz      read_done
    add     r8, rax
    cmp     r8, IN_BUF_SIZE
    jb      read_loop
read_done:
    test    r8, r8
    jz      exit_success               # empty input, no output

    mov     r12, r8                    # remaining input length
    lea     rsi, [rip + buf_in]        # src pointer
    lea     rdi, [rip + buf_out]       # dst pointer
    xor     r9d, r9d                   # line length counter

encode_loop:
    cmp     r12, 3
    jb      encode_leftover
    movzx   r8d, byte [rsi]            # b0
    movzx   r9d, byte [rsi+1]          # b1
    movzx   r10d, byte [rsi+2]         # b2

    # idx0 = b0 >> 2
    mov     edx, r8d
    shr     edx, 2
    call    put_char_idx

    # idx1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov     edx, r8d
    and     edx, 3
    shl     edx, 4
    mov     eax, r9d
    shr     eax, 4
    or      edx, eax
    call    put_char_idx

    # idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    mov     edx, r9d
    and     edx, 0xF
    shl     edx, 2
    mov     eax, r10d
    shr     eax, 6
    or      edx, eax
    call    put_char_idx

    # idx3 = b2 & 0x3F
    mov     edx, r10d
    and     edx, 0x3F
    call    put_char_idx

    add     rsi, 3
    sub     r12, 3
    jmp     encode_loop

encode_leftover:
    cmp     r12, 0
    je      finish_output
    cmp     r12, 1
    je      encode_one

    # r12 == 2
    movzx   r8d, byte [rsi]            # b0
    movzx   r9d, byte [rsi+1]          # b1

    # idx0 = b0 >> 2
    mov     edx, r8d
    shr     edx, 2
    call    put_char_idx

    # idx1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov     edx, r8d
    and     edx, 3
    shl     edx, 4
    mov     eax, r9d
    shr     eax, 4
    or      edx, eax
    call    put_char_idx

    # idx2 = ((b1 & 0xF) << 2)
    mov     edx, r9d
    and     edx, 0xF
    shl     edx, 2
    call    put_char_idx

    # single padding '='
    mov     al, '='
    call    put_char
    jmp     finish_output

encode_one:
    movzx   r8d, byte [rsi]            # b0

    # idx0 = b0 >> 2
    mov     edx, r8d
    shr     edx, 2
    call    put_char_idx

    # idx1 = (b0 & 3) << 4
    mov     edx, r8d
    and     edx, 3
    shl     edx, 4
    call    put_char_idx

    # two paddings '='
    mov     al, '='
    call    put_char
    call    put_char
    jmp     finish_output

# rdx holds index 0..63, loads character into al and falls through to put_char
put_char_idx:
    lea     rbx, [rip + b64_table]
    mov     al, [rbx + rdx]
    # fall through

put_char:
    mov     [rdi], al
    inc     rdi
    inc     r9d
    cmp     r9d, 76
    jne     .no_nl
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r9d, r9d
.no_nl:
    ret

finish_output:
    cmp     r9d, 0
    je      output_done
    mov     byte ptr [rdi], 0x0A
    inc     rdi
output_done:
    lea     rsi, [rip + buf_out]
    mov     rdx, rdi
    sub     rdx, rsi                 # length
    mov     rax, 1                   # sys_write
    mov     rdi, 1                   # stdout
    syscall

    mov     rax, 60                  # exit
    xor     rdi, rdi
    syscall

exit_success:
    mov     rax, 60
    xor     rdi, rdi
    syscall
