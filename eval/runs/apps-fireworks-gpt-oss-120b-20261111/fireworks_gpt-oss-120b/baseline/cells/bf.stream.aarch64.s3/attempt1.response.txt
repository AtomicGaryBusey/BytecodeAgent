	.global _start
	.text
_start:
	// x19 = base of input buffer
	adrp	x19, input_buf
	add	x19, x19, :lo12:input_buf
	// x20 = total length = 0
	mov	x20, #0

read_loop:
	// x1 = buffer + total_len
	add	x1, x19, x20
	// x2 = max read size
	mov	x2, #65536
	// x0 = fd 0
	mov	x0, #0
	// syscall 63 = read
	mov	x8, #63
	svc	#0
	cbnz	x0, read_continue
	// x0 == 0, end of input
	b	read_done
read_continue:
	add	x20, x20, x0
	b	read_loop
read_done:
	// Find first '!' character
	mov	x24, #0          // found flag
	mov	x0, #0           // index i
find_loop:
	cmp	x0, x20
	b.cs	find_done
	add	x1, x19, x0
	ldrb	w1, [x1]
	cmp	w1, #'!'
	b.ne	find_not
	// found '!'
	mov	x24, #1
	mov	x21, x0          // program_len = i
	add	x23, x0, #1      // input_start offset
	sub	x22, x20, x23    // input_len = total_len - (i+1)
	b	find_done
find_not:
	add	x0, x0, #1
	b	find_loop
find_done:
	cmp	x24, #0
	b.ne	after_find
	// No '!' found
	mov	x21, x20         // program_len = total_len
	mov	x22, #0          // input_len = 0
	mov	x23, #0          // input_start unused
after_find:
	// Bracket validation
	mov	x26, #0           // pc index for validation
	mov	x27, #0           // depth counter
bal_loop:
	cmp	x26, x21
	b.cs	bal_done
	add	x1, x19, x26
	ldrb	w1, [x1]
	cmp	w1, #'['
	b.eq	bal_inc
	cmp	w1, #']'
	b.eq	bal_dec
	add	x26, x26, #1
	b	bal_loop
bal_inc:
	add	x27, x27, #1
	add	x26, x26, #1
	b	bal_loop
bal_dec:
	subs	x27, x27, #1
	b.mi	bracket_error
	add	x26, x26, #1
	b	bal_loop
bal_done:
	cbnz	x27, bracket_error
	// Prepare for execution
	adrp	x29, tape
	add	x29, x29, :lo12:tape
	mov	x28, #0           // data pointer offset
	mov	x25, #0           // input read index
	mov	x26, #0           // program counter (pc)

exec_loop:
	cmp	x26, x21
	b.cs	exec_done
	add	x1, x19, x26
	ldrb	w1, [x1]
	cmp	w1, #'>'
	b.eq	cmd_gt
	cmp	w1, #'<'
	b.eq	cmd_lt
	cmp	w1, #'+'
	b.eq	cmd_inc
	cmp	w1, #'-'
	b.eq	cmd_dec
	cmp	w1, #'.'
	b.eq	cmd_dot
	cmp	w1, #','
	b.eq	cmd_comma
	cmp	w1, #'['
	b.eq	cmd_lbracket
	cmp	w1, #']'
	b.eq	cmd_rbracket
	// ignore any other character
	add	x26, x26, #1
	b	exec_loop

cmd_gt:
	add	x28, x28, #1
	add	x26, x26, #1
	b	exec_loop

cmd_lt:
	sub	x28, x28, #1
	add	x26, x26, #1
	b	exec_loop

cmd_inc:
	add	x2, x29, x28
	ldrb	w2, [x2]
	add	w2, w2, #1
	and	w2, w2, #0xFF
	strb	w2, [x2]
	add	x26, x26, #1
	b	exec_loop

cmd_dec:
	add	x2, x29, x28
	ldrb	w2, [x2]
	sub	w2, w2, #1
	and	w2, w2, #0xFF
	strb	w2, [x2]
	add	x26, x26, #1
	b	exec_loop

cmd_dot:
	add	x2, x29, x28
	ldrb	w2, [x2]
	adrp	x3, out_buf
	add	x3, x3, :lo12:out_buf
	strb	w2, [x3]
	// write(fd=1, buf, 1)
	mov	x0, #1
	mov	x1, x3
	mov	x2, #1
	mov	x8, #64
	svc	#0
	add	x26, x26, #1
	b	exec_loop

cmd_comma:
	cmp	x25, x22
	b.cs	comma_eof
	// have input
	add	x1, x19, x23
	add	x1, x1, x25
	ldrb	w1, [x1]
	add	x2, x29, x28
	strb	w1, [x2]
	add	x25, x25, #1
	b	comma_done
comma_eof:
	add	x2, x29, x28
	mov	w1, #0
	strb	w1, [x2]
comma_done:
	add	x26, x26, #1
	b	exec_loop

cmd_lbracket:
	add	x2, x29, x28
	ldrb	w2, [x2]
	cbnz	w2, bracket_continue
	// cell is zero, skip forward
	mov	x27, #1
	add	x26, x26, #1       // move to next instruction after '['
bracket_skip_fwd:
	cmp	x26, x21
	b.cs	exec_done
	add	x1, x19, x26
	ldrb	w1, [x1]
	cmp	w1, #'['
	b.eq	bracket_inc_nest_fwd
	cmp	w1, #']'
	b.eq	bracket_dec_nest_fwd
	add	x26, x26, #1
	b	bracket_skip_fwd
bracket_inc_nest_fwd:
	add	x27, x27, #1
	add	x26, x26, #1
	b	bracket_skip_fwd
bracket_dec_nest_fwd:
	sub	x27, x27, #1
	add	x26, x26, #1
	cbnz	x27, bracket_skip_fwd
	// nesting zero, continue execution after ']'
	b	exec_loop
bracket_continue:
	add	x26, x26, #1
	b	exec_loop

cmd_rbracket:
	add	x2, x29, x28
	ldrb	w2, [x2]
	cbnz	w2, bracket_back
	// cell zero, just move past ']'
	add	x26, x26, #1
	b	exec_loop
bracket_back:
	mov	x27, #1
	sub	x26, x26, #1       // start at previous instruction
bracket_scan_back:
	cmp	x26, #0
	b.lt	exec_done
	add	x1, x19, x26
	ldrb	w1, [x1]
	cmp	w1, #']'
	b.eq	bracket_inc_nest_back
	cmp	w1, #'['
	b.eq	bracket_dec_nest_back
	sub	x26, x26, #1
	b	bracket_scan_back
bracket_inc_nest_back:
	add	x27, x27, #1
	sub	x26, x26, #1
	b	bracket_scan_back
bracket_dec_nest_back:
	sub	x27, x27, #1
	sub	x26, x26, #1
	cbnz	x27, bracket_scan_back
	// found matching '['
	add	x26, x26, #1       // position after '['
	b	exec_loop

exec_done:
	// exit status 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

bracket_error:
	// exit status 1
	mov	x0, #1
	mov	x8, #93
	svc	#0

	.section .bss
	.align 4
input_buf:
	.space 65536
tape:
	.space 30000
out_buf:
	.byte 0
