.global _start
.text
_start:
    // x19 = address of input buffer
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf

    // x20 = total length read so far = 0
    mov     x20, #0

read_loop:
    // x0 = fd 0 (stdin)
    mov     x0, #0
    // x1 = buffer address = input_buf + total_len
    add     x1, x19, x20
    // x2 = max count = 65536 - total_len
    mov     x3, #65536
    sub     x2, x3, x20
    // syscall read
    mov     x8, #63
    svc     #0
    // x0 = bytes read
    cbz     x0, read_done
    // add to total_len
    add     x20, x20, x0
    b       read_loop
read_done:
    // Find first '!' separator
    // x21 = program_len, x22 = input_len, x23 = input_start offset
    mov     x21, x20          // default: whole buffer is program
    mov     x22, #0
    mov     x23, #0
    mov     x24, #0          // flag found
    mov     x25, #0          // index i

find_sep:
    cmp     x25, x20
    b.cs    after_find
    add     x2, x19, x25
    ldrb    w2, [x2]
    cmp     w2, #'!'
    b.ne    inc_find
    // found separator
    mov     x24, #1
    mov     x21, x25          // program_len = i
    add     x23, x25, #1      // input_start offset = i+1
    sub     x22, x20, x23     // input_len = total_len - (i+1)
    b       after_find
inc_find:
    add     x25, x25, #1
    b       find_sep
after_find:
    // If separator not found, program_len already set, input_len =0
    // Bracket validation
    mov     x26, #0          // pc for validation
    mov     x27, #0          // depth
val_loop:
    cmp     x26, x21
    b.cs    val_done
    add     x2, x19, x26
    ldrb    w2, [x2]
    cmp     w2, #'['
    b.eq    inc_depth
    cmp     w2, #']'
    b.eq    dec_depth
    add     x26, x26, #1
    b       val_loop
inc_depth:
    add     x27, x27, #1
    add     x26, x26, #1
    b       val_loop
dec_depth:
    subs    x27, x27, #1
    b.mi    error_unbalanced
    add     x26, x26, #1
    b       val_loop
val_done:
    cbnz    x27, error_unbalanced

    // Set up pointers
    // x24 = tape base
    adrp    x24, tape
    add     x24, x24, :lo12:tape
    // x28 = data pointer offset = 0
    mov     x28, #0
    // x25 = input position = 0
    mov     x25, #0
    // x26 = program counter = 0
    mov     x26, #0

exec_loop:
    cmp     x26, x21
    b.cs    exec_done
    add     x2, x19, x26
    ldrb    w2, [x2]          // w2 = current command
    // compare against known commands
    cmp     w2, #'>'
    b.eq    cmd_gt
    cmp     w2, #'<'
    b.eq    cmd_lt
    cmp     w2, #'+'
    b.eq    cmd_inc
    cmp     w2, #'-'
    b.eq    cmd_dec
    cmp     w2, #'.'
    b.eq    cmd_dot
    cmp     w2, #','
    b.eq    cmd_comma
    cmp     w2, #'['
    b.eq    cmd_lbracket
    cmp     w2, #']'
    b.eq    cmd_rbracket
    // ignore any other character
    add     x26, x26, #1
    b       exec_loop

cmd_gt:
    add     x28, x28, #1
    add     x26, x26, #1
    b       exec_loop

cmd_lt:
    sub     x28, x28, #1
    add     x26, x26, #1
    b       exec_loop

cmd_inc:
    add     x3, x24, x28
    ldrb    w3, [x3]
    add     w3, w3, #1
    and     w3, w3, #0xFF
    strb    w3, [x3]
    add     x26, x26, #1
    b       exec_loop

cmd_dec:
    add     x3, x24, x28
    ldrb    w3, [x3]
    sub     w3, w3, #1
    and     w3, w3, #0xFF
    strb    w3, [x3]
    add     x26, x26, #1
    b       exec_loop

cmd_dot:
    // load cell
    add     x3, x24, x28
    ldrb    w3, [x3]
    // store to out_buf
    adrp    x4, out_buf
    add     x4, x4, :lo12:out_buf
    strb    w3, [x4]
    // write(fd=1, buf=out_buf, count=1)
    mov     x0, #1
    mov     x1, x4
    mov     x2, #1
    mov     x8, #64
    svc     #0
    add     x26, x26, #1
    b       exec_loop

cmd_comma:
    // if input_pos < input_len
    cmp     x25, x22
    b.cs    comma_eof
    // load input byte
    add     x3, x19, x23
    add     x3, x3, x25
    ldrb    w3, [x3]
    b       comma_store
comma_eof:
    mov     w3, #0
comma_store:
    // store into tape cell
    add     x4, x24, x28
    strb    w3, [x4]
    // advance input_pos if not EOF
    cmp     x25, x22
    b.cs    after_comma_inc
    add     x25, x25, #1
after_comma_inc:
    add     x26, x26, #1
    b       exec_loop

cmd_lbracket:
    // load current cell
    add     x3, x24, x28
    ldrb    w3, [x3]
    cbnz    w3, lbracket_continue
    // cell is zero -> skip forward to matching ]
    mov     x27, #1          // nesting depth
    add     x26, x26, #1     // start after '['
lbracket_skip:
    cmp     x26, x21
    b.cs    exec_done        // should not happen, but protect
    add     x4, x19, x26
    ldrb    w4, [x4]
    cmp     w4, #'['
    b.eq    nest_inc_fwd
    cmp     w4, #']'
    b.eq    nest_dec_fwd
    add     x26, x26, #1
    b       lbracket_skip
nest_inc_fwd:
    add     x27, x27, #1
    add     x26, x26, #1
    b       lbracket_skip
nest_dec_fwd:
    sub     x27, x27, #1
    add     x26, x26, #1
    cbnz    x27, lbracket_skip
    // found matching ]
    b       exec_loop
lbracket_continue:
    // just move past '['
    add     x26, x26, #1
    b       exec_loop

cmd_rbracket:
    // load current cell
    add     x3, x24, x28
    ldrb    w3, [x3]
    cbz     w3, rbracket_continue
    // cell non-zero -> jump back to matching [
    mov     x27, #1          // nesting depth
    sub     x26, x26, #1     // start before ']'
rbracket_back:
    cmp     x26, #0
    b.lt    exec_done        // protect
    add     x4, x19, x26
    ldrb    w4, [x4]
    cmp     w4, #']'
    b.eq    nest_inc_back
    cmp     w4, #'['
    b.eq    nest_dec_back
    sub     x26, x26, #1
    b       rbracket_back
nest_inc_back:
    add     x27, x27, #1
    sub     x26, x26, #1
    b       rbracket_back
nest_dec_back:
    sub     x27, x27, #1
    sub     x26, x26, #1
    cbnz    x27, rbracket_back
    // now x26 points at matching '['
    add     x26, x26, #1     // move to instruction after '['
    b       exec_loop
rbracket_continue:
    // cell zero -> just move past ']'
    add     x26, x26, #1
    b       exec_loop

exec_done:
    // exit status 0
    mov     x0, #0
    mov     x8, #93
    svc     #0

error_unbalanced:
    // exit status 1
    mov     x0, #1
    mov     x8, #93
    svc     #0

.section .bss
.align 8
input_buf:
    .space 65536
tape:
    .space 30000
out_buf:
    .zero 1
