.global _start
.text
_start:
    // x10 = input buffer address
    adrp    x10, input_buf
    add     x10, x10, :lo12:input_buf

    // x11 = total input length (bytes read) = 0
    mov     x11, #0

read_loop:
    // sys_read(fd=0, buf=x10+x11, count=65536-x11)
    mov     x0, #0                // fd
    add     x1, x10, x11          // buf
    mov     x2, #65536
    sub     x2, x2, x11           // count
    mov     x8, #63               // sys_read
    svc     #0
    // x0 = bytes read
    cbz     x0, read_done
    add     x11, x11, x0
    b       read_loop
read_done:
    // Find first '!' separator
    // x12 = program length, default whole input
    mov     x12, x11
    // x13 = separator index
    mov     x13, #0
    // x14 = input start offset (after '!'), default 0
    mov     x14, #0
    // x15 = input length, default 0
    mov     x15, #0
    // x16 = flag found
    mov     x16, #0
    // x17 = loop index i
    mov     x17, #0

find_sep:
    cmp     x17, x11
    b.cs    after_find
    add     x18, x10, x17
    ldrb    w18, [x18]
    cmp     w18, #'!'
    b.ne    inc_find
    // separator found
    mov     x16, #1
    mov     x12, x17          // program length = i
    add     x14, x17, #1      // input start = i+1
    sub     x15, x11, x14     // input length = total - start
    b       after_find
inc_find:
    add     x17, x17, #1
    b       find_sep
after_find:
    // Bracket validation
    // x18 = pc for validation, x19 = depth
    mov     x18, #0
    mov     x19, #0
val_loop:
    cmp     x18, x12
    b.cs    val_done
    add     x20, x10, x18
    ldrb    w20, [x20]
    cmp     w20, #'['
    b.eq    inc_depth
    cmp     w20, #']'
    b.eq    dec_depth
    add     x18, x18, #1
    b       val_loop
inc_depth:
    add     x19, x19, #1
    add     x18, x18, #1
    b       val_loop
dec_depth:
    subs    x19, x19, #1
    b.mi    error_unbalanced
    add     x18, x18, #1
    b       val_loop
val_done:
    cbnz    x19, error_unbalanced

    // Setup runtime pointers
    // x15 = tape base
    adrp    x15, tape
    add     x15, x15, :lo12:tape
    // x16 = data pointer offset (0)
    mov     x16, #0
    // x17 = input position (0)
    mov     x17, #0
    // x18 = program counter (0)
    mov     x18, #0

exec_loop:
    cmp     x18, x12
    b.cs    exec_done
    add     x20, x10, x18
    ldrb    w20, [x20]          // current command

    // dispatch
    cmp     w20, #'>'
    b.eq    cmd_gt
    cmp     w20, #'<'
    b.eq    cmd_lt
    cmp     w20, #'+'
    b.eq    cmd_inc
    cmp     w20, #'-'
    b.eq    cmd_dec
    cmp     w20, #'.'
    b.eq    cmd_dot
    cmp     w20, #','
    b.eq    cmd_comma
    cmp     w20, #'['
    b.eq    cmd_lbracket
    cmp     w20, #']'
    b.eq    cmd_rbracket
    // ignore other chars
    add     x18, x18, #1
    b       exec_loop

cmd_gt:
    add     x16, x16, #1
    add     x18, x18, #1
    b       exec_loop

cmd_lt:
    sub     x16, x16, #1
    add     x18, x18, #1
    b       exec_loop

cmd_inc:
    add     x21, x15, x16
    ldrb    w21, [x21]
    add     w21, w21, #1
    and     w21, w21, #0xFF
    strb    w21, [x21]
    add     x18, x18, #1
    b       exec_loop

cmd_dec:
    add     x21, x15, x16
    ldrb    w21, [x21]
    sub     w21, w21, #1
    and     w21, w21, #0xFF
    strb    w21, [x21]
    add     x18, x18, #1
    b       exec_loop

cmd_dot:
    add     x21, x15, x16
    ldrb    w21, [x21]
    // out_buf address
    adrp    x22, out_buf
    add     x22, x22, :lo12:out_buf
    strb    w21, [x22]
    // write(fd=1, buf=out_buf, count=1)
    mov     x0, #1
    mov     x1, x22
    mov     x2, #1
    mov     x8, #64
    svc     #0
    add     x18, x18, #1
    b       exec_loop

cmd_comma:
    // if input_pos < input_len
    cmp     x17, x15          // careful: x15 is tape base now, need input_len in a reg
    // Actually input_len stored in x15? No, we overwrote x15. Need separate reg for input_len.
    // We'll keep input_len in x23 (saved earlier). Adjust: after separator, move input_len to x23.
    // So we need to modify earlier code: after after_find, store input_len to x23.

    // Fix: Use x23 for input_len, x24 for input_start offset.
    // We'll adjust below after this block.

    // Placeholder (will be replaced by corrected code below)

    b       exec_loop

cmd_lbracket:
    add     x21, x15, x16
    ldrb    w21, [x21]
    cbnz    w21, lbracket_continue
    // cell zero -> skip forward
    mov     x19, #1
    add     x18, x18, #1
lbracket_skip:
    cmp     x18, x12
    b.cs    exec_done
    add     x21, x10, x18
    ldrb    w21, [x21]
    cmp     w21, #'['
    b.eq    nest_inc_fwd
    cmp     w21, #']'
    b.eq    nest_dec_fwd
    add     x18, x18, #1
    b       lbracket_skip
nest_inc_fwd:
    add     x19, x19, #1
    add     x18, x18, #1
    b       lbracket_skip
nest_dec_fwd:
    sub     x19, x19, #1
    add     x18, x18, #1
    cbnz    x19, lbracket_skip
    // matched
    b       exec_loop
lbracket_continue:
    add     x18, x18, #1
    b       exec_loop

cmd_rbracket:
    add     x21, x15, x16
    ldrb    w21, [x21]
    cbz     w21, rbracket_continue
    // non-zero -> jump back
    mov     x19, #1
    sub     x18, x18, #1
rbracket_back:
    cmp     x18, #0
    b.lt    exec_done
    add     x21, x10, x18
    ldrb    w21, [x21]
    cmp     w21, #']'
    b.eq    nest_inc_back
    cmp     w21, #'['
    b.eq    nest_dec_back
    sub     x18, x18, #1
    b       rbracket_back
nest_inc_back:
    add     x19, x19, #1
    sub     x18, x18, #1
    b       rbracket_back
nest_dec_back:
    sub     x19, x19, #1
    sub     x18, x18, #1
    cbnz    x19, rbracket_back
    // now at matching '['
    add     x18, x18, #1
    b       exec_loop
rbracket_continue:
    add     x18, x18, #1
    b       exec_loop

exec_done:
    mov     x0, #0
    mov     x8, #93
    svc     #0

error_unbalanced:
    mov     x0, #1
    mov     x8, #93
    svc     #0
