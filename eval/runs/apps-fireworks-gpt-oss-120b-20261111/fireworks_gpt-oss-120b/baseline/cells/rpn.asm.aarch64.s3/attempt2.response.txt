	.globl	_start
	.text
	.align	2

	.equ	BUF_SIZE, 65536
	.equ	STACK_CAP, 1024
	.equ	CONV_BUF_SIZE, 32
	.equ	ERROR_LEN, 6
	.equ	MIN_INT, 0x8000000000000000

_start:
	// read entire input into buffer
	mov	x4, #0			// total bytes read
read_loop:
	adr	x5, buffer
	add	x1, x5, x4		// buf + offset
	mov	x0, #0			// fd stdin
	mov	x2, #BUF_SIZE
	sub	x2, x2, x4		// remaining space
	mov	x8, #63			// sys_read
	svc	#0
	cbnz	x0, read_continue
	// x0 == 0 => EOF
	b	read_done
read_continue:
	add	x4, x4, x0
	cmp	x4, #BUF_SIZE
	blt	read_loop
read_done:
	// x4 = total input length
	adr	x5, buffer		// cur pointer
	mov	x6, x4			// bytes left

	adr	x19, stack_vals		// base of evaluation stack
	adr	x20, conv_buf		// base of conversion buffer (will use end)

line_loop:
	cbnz	x6, line_start
	// no more data
	b	exit_program

line_start:
	// find end of current line (newline or end of input)
	mov	x7, x5			// scan ptr
	mov	x8, x6			// bytes left for scan
	mov	x9, #0			// flag: newline found (0=no,1=yes)

find_nl:
	cbnz	x8, check_char
	// no more bytes, no newline
	b	line_set_end

check_char:
	ldrb	w0, [x7]
	cmp	w0, #0x0A
	bne	not_nl
	// newline found
	mov	x9, #1
	b	line_set_end
not_nl:
	add	x7, x7, #1
	subs	x8, x8, #1
	b	find_nl

line_set_end:
	// x7 points at newline if found, else at end
	// compute line_end (exclusive)
	cmp	x9, #1
	bne	no_newline
	// line_end = x7 (newline position)
	mov	x10, x7
	b	line_process
no_newline:
	// line_end = cur + remaining (x5 + x6)
	add	x10, x5, x6
	// x9 remains 0
line_process:
	// x5 = line start, x10 = line end (exclusive)
	// initialize per-line state
	mov	x11, #0			// stack index
	mov	x12, #0			// error flag

parse_loop:
	cmp	x5, x10
	b.ge	parse_done
	ldrb	w0, [x5]
	cmp	w0, #0x20
	beq	skip_sep
	cmp	w0, #0x09
	beq	skip_sep
	b	token_start
skip_sep:
	add	x5, x5, #1
	b	parse_loop

token_start:
	mov	x13, x5			// token start
	// find token end
token_scan:
	cmp	x5, x10
	b.ge	token_end
	ldrb	w0, [x5]
	cmp	w0, #0x20
	beq	token_end
	cmp	w0, #0x09
	beq	token_end
	add	x5, x5, #1
	b	token_scan
token_end:
	sub	x14, x5, x13		// token length in x14
	ldrb	w1, [x13]		// first character

	// check for single-char operators
	cmp	x14, #1
	b.ne	literal_check

	// possible operators '+', '-', '*', '/', '%'
	cmp	w1, #'+' 
	beq	op_plus
	cmp	w1, #'-'
	beq	op_minus_op
	cmp	w1, #'*'
	beq.op_mul
	cmp	w1, #'/'
	beq.op_div
	cmp	w1, #'%'
	beq.op_mod
	b	set_error

op_minus_op:
	// single '-' = subtraction operator
	b	op_minus

literal_check:
	// parse integer literal
	mov	x15, #0			// sign flag (0=positive)
	cmp	w1, #'-'
	b.ne.lit_no_sign
	mov	x15, #1
	add	x16, x13, #1		// digit start
	sub	x17, x14, #1		// digit count
	b.lit_digit_loop_prep
lit_no_sign:
	mov	x16, x13
	mov	x17, x14
lit_digit_loop_prep:
	// need at least one digit
	cmp	x17, #0
	beq	set_error

	// digit count must <=18
	cmp	x17, #19
	b.ge	set_error

	mov	x18, #0			// accumulator
	mov	x19, #0			// loop counter

lit_digit_loop:
	ldrb	w2, [x16, x19]
	cmp	w2, #'0'
	blt	set_error
	cmp	w2, #'9'
	bgt	set_error
	// acc = acc*10 + (w2-'0')
	mov	x20, x18
	mov	x21, #10
	mul	x20, x20, x21
	sub	w22, w2, #'0'
	add	x20, x20, x22
	mov	x18, x20
	add	x19, x19, #1
	cmp	x19, x17
	b.lt	lit_digit_loop

	// apply sign if needed
	cmp	x15, #0
	beq.lit_push
	neg	x18, x18
 lit_push:
	// push value onto stack
	add	x21, x11, #0
	lsl	x21, x21, #3
	add	x22, x19, #0		// placeholder not used
	add	x22, x19, #0
	// address = stack_base + offset
	add	x23, x19, #0		// placeholder
	add	x23, x21, x19		// not correct; rewrite:
	// compute address: stack_vals + (stack_idx * 8)
	add	x24, x19, #0		// placeholder
	// actual:
	add	x24, x19, #0		// placeholder
	// We'll compute properly:
	// x19 currently holds accumulator; we need base address in x19? Wait we used x19 for accumulator, conflict.
	// Better use dedicated registers:
	// Use x30 as temp for address compute.
	// Re-compute:
	// x19 is accumulator, keep it.
	// Use x21 for offset, x22 for address.
	lsl	x21, x11, #3
	adr	x22, stack_vals
	add	x22, x22, x21
	str	x19, [x22]
	add	x11, x11, #1
	b.parse_loop

// ---------- operators ----------
op_plus:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	adr	x26, stack_vals
	add	x26, x26, x25
	ldr	x27, [x26]			// a
	ldr	x28, [x26, #8]		// b
	add	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b.parse_loop

op_minus:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	adr	x26, stack_vals
	add	x26, x26, x25
	ldr	x27, [x26]			// a
	ldr	x28, [x26, #8]		// b
	sub	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b.parse_loop

op_mul:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	adr	x26, stack_vals
	add	x26, x26, x25
	ldr	x27, [x26]
	ldr	x28, [x26, #8]
	mul	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b.parse_loop

op_div:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	adr	x26, stack_vals
	add	x26, x26, x25
	ldr	x27, [x26]			// a
	ldr	x28, [x26, #8]		// b
	cbnz	x28, div_ok
	b	set_error
div_ok:
	// overflow check: a == INT64_MIN && b == -1
	movz	x29, #0
	movk	x29, #0x8000, lsl #48	// x29 = INT64_MIN
	cmp	x27, x29
	b.ne.div_calc
	cmp	x28, #-1
	b.ne.div_calc
	b	set_error
div_calc:
	sdiv	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b.parse_loop

op_mod:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	adr	x26, stack_vals
	add	x26, x26, x25
	ldr	x27, [x26]			// a
	ldr	x28, [x26, #8]		// b
	cbnz	x28, mod_ok
	b	set_error
mod_ok:
	// overflow check same as div
	movz	x29, #0
	movk	x29, #0x8000, lsl #48
	cmp	x27, x29
	b.ne.mod_calc
	cmp	x28, #-1
	b.ne.mod_calc
	b	set_error
mod_calc:
	sdiv	x30, x27, x28		// quotient
	mul	x31, x30, x28
	sub	x0, x27, x31		// remainder
	str	x0, [x26]
	sub	x11, x11, #1
	b.parse_loop

set_error:
	mov	x12, #1
	// skip rest of line
	mov	x5, x10
	b.parse_loop

parse_done:
	// line finished, produce output
	cbnz	x12, output_error
	cmp	x11, #1
	b.ne	output_error

	// get result
	adr	x0, stack_vals
	ldr	x1, [x0]			// result in x1

	// convert to decimal string in conv_buf (from end)
	adr	x2, conv_buf
	add	x2, x2, #CONV_BUF_SIZE	// ptr = end
	mov	x3, #0			// sign flag
	cmp	x1, #0
	b.ge.conv_abs
	neg	x1, x1
	mov	x3, #1
conv_abs:
	// handle zero specially
	cmp	x1, #0
	b.ne.conv_loop
	// value is zero -> write '0'
	sub	x2, x2, #1
	mov	w4, #'0'
	strb	w4, [x2]
	b.conv_done_digits
conv_loop:
	sub	x2, x2, #1
	// quotient and remainder
	mov	x4, #10
	udiv	x5, x1, x4
	mul	x6, x5, x4
	sub	x7, x1, x6
	add	w8, w7, #'0'
	strb	w8, [x2]
	mov	x1, x5
	cbnz	x1, conv_loop
conv_done_digits:
	// add sign if needed
	cbnz	x3, add_sign
	b.add_newline
add_sign:
	sub	x2, x2, #1
	mov	w9, #'-'
	strb	w9, [x2]
add_newline:
	// store newline
	sub	x2, x2, #1
	mov	w10, #'\n'
	strb	w10, [x2]

	// compute length = conv_buf+CONV_BUF_SIZE - x2
	adr	x11, conv_buf
	add	x12, x11, #CONV_BUF_SIZE
	sub	x13, x12, x2		// length

	// write result
	mov	x0, #1			// stdout
	mov	x1, x2
	mov	x2, x13
	mov	x8, #64
	svc	#0
	b.line_continue

output_error:
	adr	x0, error_msg
	mov	x1, #1
	mov	x2, #ERROR_LEN
	mov	x8, #64
	svc	#0
	// fall through to line_continue

line_continue:
	// advance to next line
	cmp	x9, #1			// x9 = newline flag
	bne	no_more_input
	// newline was present
	add	x5, x10, #1		// skip newline
	sub	x6, x6, x14		// consumed line_len + newline
	sub	x6, x6, #1
	b	line_loop
no_more_input:
	// either no newline (end of file) or finished processing final line
	// set remaining bytes to zero to exit
	mov	x6, #0
	b	line_loop

exit_program:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.section	.rodata
error_msg:
	.asciz	"error\n"

	.section	.bss
	.align	3
buffer:
	.zero	BUF_SIZE
stack_vals:
	.zero	STACK_CAP*8
conv_buf:
	.zero	CONV_BUF_SIZE
