	.globl	_start
	.text
	.align	2

	// constants
	.equ	BUF_SIZE, 65536
	.equ	STACK_CAP, 1024
	.equ	CONV_BUF_SIZE, 32
	.equ	ERROR_LEN, 6
	.equ	MIN_INT_HI, 0x8000		// high 16 bits for 0x8000000000000000
	.equ	MIN_INT_LO, 0x0000

_start:
	// read whole input into buffer
	mov	x4, #0			// offset / total length
read_loop:
	// x5 = &buffer
	adr	x5, buffer
	add	x1, x5, x4		// buffer + offset
	mov	x0, #0			// fd stdin
	mov	x2, #BUF_SIZE
	sub	x2, x2, x4		// remaining space
	mov	x8, #63			// sys_read
	svc	#0
	cbz	x0, read_done
	add	x4, x4, x0		// increase offset
	b	read_loop
read_done:
	// x4 now holds total_len

	// set up pointers for line processing
	adr	x6, buffer		// cur_ptr
	mov	x7, x4			// bytes remaining

line_loop:
	cbnz	x7, process_line_entry
	b	exit_program

process_line_entry:
	// find end of line (newline or end of input)
	mov	x9, x6			// scan_ptr
	mov	x10, x7			// bytes left for scan
	mov	x11, #0			// line_len (excludes newline)
scan_newline:
	cbnz	x10, scan_check
	// no newline left
	b	line_no_newline
scan_check:
	ldrb	w12, [x9], #1
	subs	x10, x10, #1
	cmp	w12, #0x0A
	bne	scan_newline
	// newline found, x9 points after newline, line_len = consumed-1
	sub	x11, x9, x6		// total consumed including newline
	sub	x11, x11, #1		// line length without newline
	b	line_found

line_no_newline:
	// line is the rest of the input
	add	x11, x6, x7		// line_end = cur + remaining
	b	line_process

line_found:
	// line_end = cur + line_len
	add	x11, x6, x11

line_process:
	// x6 = line_start, x11 = line_end (exclusive)
	mov	x20, #0			// stack index
	mov	x21, #0			// error flag

parse_loop:
	// skip separators (space or tab)
	cmp	x6, x11
	b.ge	parse_done
	ldrb	w13, [x6]
	cmp	w13, #0x20
	beq	skip_sep
	cmp	w13, #0x09
	beq	skip_sep
	b	token_start
skip_sep:
	add	x6, x6, #1
	b	parse_loop

token_start:
	mov	x14, x6			// token_start
	// find token end
token_scan:
	cmp	x6, x11
	b.ge	token_end
	ldrb	w13, [x6]
	cmp	w13, #0x20
	beq	token_end
	cmp	w13, #0x09
	beq	token_end
	add	x6, x6, #1
	b	token_scan
token_end:
	sub	x15, x6, x14		// token_len
	ldrb	w16, [x14]		// first char

	// check for single-char operators
	cmp	x15, #1
	b.ne	check_literal

	// possible operators '+', '-', '*', '/', '%'
	cmp	w16, #'+'
	beq	op_plus
	cmp	w16, #'-'
	beq	op_minus_operator
	cmp	w16, #'*'
	beq	op_mul
	cmp	w16, #'/'
	beq	op_div
	cmp	w16, #'%'
	beq	op_mod
	b	set_error

op_minus_operator:
	// single '-' is subtraction operator
	b	op_minus

check_literal:
	// parse integer literal, optional leading '-'
	mov	x17, #0			// negative flag
	cmp	w16, #'-'
	b.ne	lit_no_sign
	mov	x17, #1
	add	x18, x14, #1		// start of digits
	sub	x19, x15, #1		// digit count
	b	lit_digit_loop_check
lit_no_sign:
	mov	x18, x14
	mov	x19, x15

	// need at least one digit
	cmp	x19, #0
	beq	set_error

	// digit parsing
	mov	x20d, #0		// digit count
	mov	x21v, #0		// accumulator

lit_digit_loop:
	cmp	x20d, #18
	b.ge	set_error
	ldrb	w22, [x18, x20d]
	cmp	w22, #'0'
	blt	set_error
	cmp	w22, #'9'
	bgt	set_error
	// acc = acc*10 + (w22-'0')
	mov	x23, x21v
	mov	x24, #10
	mul	x23, x23, x24
	sub	w24, w22, #'0'
	add	x23, x23, x24
	mov	x21v, x23
	add	x20d, x20d, #1
	cmp	x20d, x19
	b.lt	lit_digit_loop

	// apply sign if needed
	cmp	x17, #0
	beq	lit_push
	neg	x21v, x21v
lit_push:
	// push onto stack
	adr	x0, stack_vals
	madd	x1, x20, #8, x0
	str	x21v, [x1]
	add	x20, x20, #1
	b	parse_loop

// ---------- operators ----------
op_plus:
	cmp	x20, #2
	b.lt	set_error
	sub	x25, x20, #2
	adr	x0, stack_vals
	madd	x1, x25, #8, x0
	ldr	x2, [x1]			// a
	ldr	x3, [x1, #8]			// b
	add	x4, x2, x3
	str	x4, [x1]			// store result at a
	sub	x20, x20, #1
	b	parse_loop

op_minus:
	cmp	x20, #2
	b.lt	set_error
	sub	x25, x20, #2
	adr	x0, stack_vals
	madd	x1, x25, #8, x0
	ldr	x2, [x1]			// a
	ldr	x3, [x1, #8]			// b
	sub	x4, x2, x3
	str	x4, [x1]
	sub	x20, x20, #1
	b	parse_loop

op_mul:
	cmp	x20, #2
	b.lt	set_error
	sub	x25, x20, #2
	adr	x0, stack_vals
	madd	x1, x25, #8, x0
	ldr	x2, [x1]
	ldr	x3, [x1, #8]
	mul	x4, x2, x3
	str	x4, [x1]
	sub	x20, x20, #1
	b	parse_loop

op_div:
	cmp	x20, #2
	b.lt	set_error
	sub	x25, x20, #2
	adr	x0, stack_vals
	madd	x1, x25, #8, x0
	ldr	x2, [x1]			// a
	ldr	x3, [x1, #8]			// b
	cbnz	x3, div_nonzero
	b	set_error
div_nonzero:
	// load MIN_INT constant into x5
	movz	x5, #0
	movk	x5, #MIN_INT_HI, lsl #48	// 0x8000000000000000
	cmp	x2, x5
	b.ne	div_ok
	cmp	x3, #-1
	b.ne	div_ok
	b	set_error
div_ok:
	sdiv	x4, x2, x3
	str	x4, [x1]
	sub	x20, x20, #1
	b	parse_loop

op_mod:
	cmp	x20, #2
	b.lt	set_error
	sub	x25, x20, #2
	adr	x0, stack_vals
	madd	x1, x25, #8, x0
	ldr	x2, [x1]			// a
	ldr	x3, [x1, #8]			// b
	cbnz	x3, mod_nonzero
	b	set_error
mod_nonzero:
	// check overflow same as div
	movz	x5, #0
	movk	x5, #MIN_INT_HI, lsl #48
	cmp	x2, x5
	b.ne	mod_ok
	cmp	x3, #-1
	b.ne	mod_ok
	b	set_error
mod_ok:
	sdiv	x6, x2, x3			// quotient
	mul	x7, x6, x3
	sub	x4, x2, x7			// remainder
	str	x4, [x1]
	sub	x20, x20, #1
	b	parse_loop

set_error:
	mov	x21, #1
	// skip rest of line
	mov	x6, x11
	b	parse_loop

parse_done:
	// line finished, produce output
	cbnz	x21, output_error
	cmp	x20, #1
	b.ne	output_error

	// load result
	adr	x0, stack_vals
	ldr	x1, [x0]			// result in x1

	// convert to decimal string
	adr	x5, conv_buf
	add	x5, x5, #CONV_BUF_SIZE		// pointer to end
	mov	x6, #0				// sign flag
	cmp	x1, #0
	b.ge	conv_abs_start
	neg	x1, x1
	mov	x6, #1
conv_abs_start:
	// loop extracting digits
conv_digit_loop:
	mov	x8, #10
	udiv	x9, x1, x8			// quotient
	mul	x10, x9, x8
	sub	x11, x1, x10			// remainder
	add	w11, w11, #'0'
	sub	x5, x5, #1
	strb	w11, [x5]
	mov	x1, x9
	cbnz	x1, conv_digit_loop
	// add sign if needed
	cbnz	x6, add_sign
	b	conv_number_ready
add_sign:
	sub	x5, x5, #1
	mov	w0, #'-'
	strb	w0, [x5]
conv_number_ready:
	// append newline
	add	x12, x5, #0
	// compute length
	adr	x0, conv_buf
	add	x13, x0, #CONV_BUF_SIZE
	sub	x2, x13, x5			// length without newline
	add	x2, x2, #1			// include newline
	// store newline
	add	x14, x5, x2
	add	x14, x5, x2
	sub	x14, x5, #-1		// x14 = x5 + (length-1)
	add	x14, x5, x2
	add	x14, x5, x2
	sub	x14, x5, #-1
	add	x14, x5, x2
	add	x14, x5, x2
	// simpler: store newline at x5+len-1
	add	x15, x5, x2
	sub	x15, x15, #1
	mov	w0, #'\n'
	strb	w0, [x15]

	// write result
	mov	x0, #1				// stdout
	mov	x1, x5
	// x2 already length
	mov	x8, #64
	svc	#0
	b	line_continue

output_error:
	adr	x0, error_msg
	mov	x1, #1
	mov	x2, #ERROR_LEN
	mov	x8, #64
	svc	#0
	// fall through

line_continue:
	// advance to next line
	cmp	x9, #0				// x9 was set earlier only in newline case; reuse logic
	// determine if we processed a newline-terminated line or final line
	// if we are here from newline case, x9 points after newline; otherwise we need to finish.
	// Use the earlier computed variables:
	// after newline case we had x9 as after newline, after no-newline case we need to exit
	// Check if remaining bytes (x7) were consumed
	cbnz	x7, line_loop
	b	exit_program

	// place holder for unreachable
	b	.

exit_program:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// -------------------------------------------------
	// Data sections
	.section	.rodata
error_msg:
	.asciz	"error\n"

	.section	.bss
	.align	3
buffer:
	.zero	BUF_SIZE
stack_vals:
	.zero	STACK_CAP*8
conv_buf:
	.zero	CONV_BUF_SIZE
