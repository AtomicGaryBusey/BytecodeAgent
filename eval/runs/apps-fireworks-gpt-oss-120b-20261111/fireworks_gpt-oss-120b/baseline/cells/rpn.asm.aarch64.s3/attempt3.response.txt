	.globl	_start
	.text
	.align	2

	.equ	BUF_SIZE, 65536
	.equ	STACK_CAP, 1024
	.equ	CONV_BUF_SIZE, 32
	.equ	ERROR_LEN, 6
	.equ	INT_MIN_CONST, 0x8000000000000000

_start:
	// -------- read entire input ----------
	mov	x4, #0				// total bytes read
read_loop:
	adr	x1, buffer
	add	x1, x1, x4			// buf + offset
	mov	x0, #0				// fd stdin
	mov	x2, #BUF_SIZE
	sub	x2, x2, x4			// remaining space
	mov	x8, #63				// sys_read
	svc	#0
	cbnz	x0, read_continue
	b	read_done
read_continue:
	add	x4, x4, x0
	cmp	x4, #BUF_SIZE
	blt	read_loop
read_done:
	// x4 = total input length
	adr	x5, buffer			// current pointer
	mov	x6, x4				// bytes left

	adr	x24, stack_vals		// stack base address (keep in x24)

line_loop:
	cbnz	x6, line_start
	b	exit_program			// no more data

line_start:
	// x5 = line start, x6 = bytes left
	mov	x20, x5				// save line start
	mov	x21, x6				// save remaining before line

	// find end of line
	mov	x7, x5				// scan pointer
	mov	x8, x6				// bytes left for scan
	mov	x9, #0				// newline flag (0 = no, 1 = yes)

find_nl:
	cbnz	x8, check_char
	b	line_end_no_nl
check_char:
	ldrb	w0, [x7]
	cmp	w0, #0x0A
	beq	found_nl
	add	x7, x7, #1
	sub	x8, x8, #1
	b	find_nl
found_nl:
	mov	x9, #1
	b	line_end_set
line_end_no_nl:
	// x7 already points past last byte
line_end_set:
	cmp	x9, #0
	beq	no_newline
	mov	x10, x7			// line_end = newline position
	b	line_process
no_newline:
	add	x10, x5, x6		// line_end = end of buffer
line_process:
	// per‑line state
	mov	x11, #0			// stack idx
	mov	x12, #0			// error flag

token_loop:
	cmp	x5, x10
	b.ge	parse_done

	ldrb	w0, [x5]
	cmp	w0, #0x20
	beq	skip_sep
	cmp	w0, #0x09
	beq	skip_sep
	b	token_start
skip_sep:
	add	x5, x5, #1
	b	token_loop

token_start:
	mov	x13, x5			// token start
scan_token:
	cmp	x5, x10
	b.ge	token_end
	ldrb	w0, [x5]
	cmp	w0, #0x20
	beq	token_end
	cmp	w0, #0x09
	beq	token_end
	add	x5, x5, #1
	b	scan_token
token_end:
	sub	x14, x5, x13		// token length
	ldrb	w1, [x13]		// first char

	// ---------- operator check ----------
	cmp	x14, #1
	b.ne	literal_path

	cmp	w1, #'+' 
	beq	op_plus
	cmp	w1, #'-'
	beq	op_minus_op
	cmp	w1, #'*'
	beq	op_mul
	cmp	w1, #'/'
	beq	op_div
	cmp	w1, #'%'
	beq	op_mod
	b	set_error

op_minus_op:
	b	op_minus

// ---------- literal ----------
literal_path:
	mov	x15, #0			// sign flag
	cmp	w1, #'-'
	b.ne	lit_no_sign
	// sign present, need digits after it
	cmp	x14, #1
	beq	set_error
	mov	x15, #1
	add	x16, x13, #1		// digit start
	sub	x17, x14, #1		// digit count
	b	lit_digit_prep
lit_no_sign:
	add	x16, x13, #0
	mov	x17, x14
lit_digit_prep:
	cmp	x17, #0
	beq	set_error
	cmp	x17, #19
	b.ge	set_error
	mov	x18, #0			// accumulator
	mov	x19, #0			// loop index
lit_digit_loop:
	add	x20, x16, x19
	ldrb	w2, [x20]
	cmp	w2, #'0'
	blt	set_error
	cmp	w2, #'9'
	bgt	set_error
	mov	x21, x18
	mov	x22, #10
	mul	x21, x21, x22
	sub	w23, w2, #'0'
	add	x21, x21, x23
	mov	x18, x21
	add	x19, x19, #1
	cmp	x19, x17
	b.lt	lit_digit_loop
	cmp	x15, #0
	beq	lit_push
	neg	x18, x18
lit_push:
	// push x18
	lsl	x21, x11, #3
	add	x22, x24, x21
	str	x18, [x22]
	add	x11, x11, #1
	b	token_loop

// ---------- operators ----------
op_plus:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	add	x26, x24, x25
	ldr	x27, [x26]            // a
	ldr	x28, [x26, #8]        // b
	add	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b	token_loop

op_minus:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	add	x26, x24, x25
	ldr	x27, [x26]            // a
	ldr	x28, [x26, #8]        // b
	sub	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b	token_loop

op_mul:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	add	x26, x24, x25
	ldr	x27, [x26]
	ldr	x28, [x26, #8]
	mul	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b	token_loop

op_div:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	add	x26, x24, x25
	ldr	x27, [x26]            // a
	ldr	x28, [x26, #8]        // b
	cbnz	x28, div_ok
	b	set_error
div_ok:
	// overflow check a == INT_MIN && b == -1
	movz	x29, #0x0000
	movk	x29, #0x8000, lsl #48   // INT_MIN
	cmp	x27, x29
	b.ne	div_compute
	cmp	x28, #-1
	b.ne	div_compute
	b	set_error
div_compute:
	sdiv	x0, x27, x28
	str	x0, [x26]
	sub	x11, x11, #1
	b	token_loop

op_mod:
	cmp	x11, #2
	blt	set_error
	sub	x25, x11, #2
	lsl	x25, x25, #3
	add	x26, x24, x25
	ldr	x27, [x26]            // a
	ldr	x28, [x26, #8]        // b
	cbnz	x28, mod_ok
	b	set_error
mod_ok:
	// overflow check a == INT_MIN && b == -1
	movz	x29, #0x0000
	movk	x29, #0x8000, lsl #48
	cmp	x27, x29
	b.ne	mod_compute
	cmp	x28, #-1
	b.ne	mod_compute
	b	set_error
mod_compute:
	sdiv	x20, x27, x28          // quotient
	mul	x21, x20, x28
	sub	x0, x27, x21           // remainder
	str	x0, [x26]
	sub	x11, x11, #1
	b	token_loop

set_error:
	mov	x12, #1
	// skip rest of line
	mov	x5, x10
	b	token_loop

parse_done:
	// ---------- output ----------
	cmp	x12, #0
	b.ne	output_error
	cmp	x11, #1
	b.ne	output_error

	ldr	x1, [x24]               // result

	// check INT_MIN special case
	movz	x29, #0x0000
	movk	x29, #0x8000, lsl #48
	cmp	x1, x29
	beq	output_intmin

	// normal conversion
	adr	x2, conv_buf
	add	x2, x2, #CONV_BUF_SIZE   // ptr = end
	mov	x3, #0                  // sign flag
	cmp	x1, #0
	b.ge	conv_abs
	neg	x1, x1
	mov	x3, #1
conv_abs:
	cmp	x1, #0
	b.ne	conv_loop
	sub	x2, x2, #1
	mov	w4, #'0'
	strb	w4, [x2]
	b	conv_done_digits
conv_loop:
	sub	x2, x2, #1
	mov	x4, #10
	udiv	x5, x1, x4
	mul	x6, x5, x4
	sub	x7, x1, x6
	add	w8, w7, #'0'
	strb	w8, [x2]
	mov	x1, x5
	cbnz	x1, conv_loop
conv_done_digits:
	cmp	x3, #0
	beq	add_newline
	sub	x2, x2, #1
	mov	w9, #'-'
	strb	w9, [x2]
add_newline:
	sub	x2, x2, #1
	mov	w10, #'\n'
	strb	w10, [x2]
	adr	x11, conv_buf
	add	x12, x11, #CONV_BUF_SIZE
	sub	x13, x12, x2            // length
	mov	x0, #1                  // stdout
	mov	x1, x2
	mov	x2, x13
	mov	x8, #64
	svc	#0
	b	line_continue

output_intmin:
	// write pre‑built "-9223372036854775808\n"
	mov	x0, #1
	adr	x1, int_min_str
	mov	x2, #21
	mov	x8, #64
	svc	#0
	b	line_continue

output_error:
	mov	x0, #1
	adr	x1, error_msg
	mov	x2, #ERROR_LEN
	mov	x8, #64
	svc	#0
	b	line_continue

line_continue:
	// compute bytes consumed
	// line_len = line_end - line_start
	sub	x22, x10, x20
	// consumed = line_len + newline_flag
	add	x22, x22, x9
	// new remaining = original_remaining - consumed
	sub	x6, x21, x22
	// set next pointer
	cmp	x9, #1
	b.ne	no_newline_skip
	add	x5, x10, #1
	b	line_loop
no_newline_skip:
	mov	x5, x10
	b	line_loop

exit_program:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.section	.rodata
error_msg:
	.asciz	"error\n"
int_min_str:
	.asciz	"-9223372036854775808\n"

	.bss
	.align	3
buffer:
	.zero	BUF_SIZE
stack_vals:
	.zero	STACK_CAP*8
conv_buf:
	.zero	CONV_BUF_SIZE
