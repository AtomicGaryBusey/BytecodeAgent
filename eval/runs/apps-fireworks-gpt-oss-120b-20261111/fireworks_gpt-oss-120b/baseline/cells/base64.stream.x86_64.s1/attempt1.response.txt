.globl _start

.equ INPUT_MAX, 49152
.equ OUT_MAX, 66400

.section .bss
.align 16
input_buf:
    .zero INPUT_MAX
out_buf:
    .zero OUT_MAX

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
_start:
    # read loop
    xor rbx, rbx                # total read = 0
read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, rbx                # buffer + offset
    mov rdx, INPUT_MAX
    sub rdx, rbx                # remaining space
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done
    add rbx, rax                # total += bytes read
    cmp rbx, INPUT_MAX
    jae read_done
    jmp read_loop
read_done:
    mov rcx, rbx                # rcx = input length
    test rcx, rcx
    jz exit_ok

    # set up pointers
    lea rsi, [rip + input_buf]  # src
    lea rdi, [rip + out_buf]    # dst
    xor r8, r8                  # line_len = 0
    lea rbx, [rip + b64_table]  # table pointer

process_full:
    cmp rcx, 3
    jb process_remain

    # load three bytes and build 24-bit word
    movzx eax, byte [rsi]
