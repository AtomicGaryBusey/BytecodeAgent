.globl _start
.equ INPUT_MAX, 49152
.equ OUT_MAX, 70000

.section .bss
.align 16
input_buf:
    .zero INPUT_MAX
out_buf:
    .zero OUT_MAX

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # ---------------- read entire input ----------------
    xor rbx, rbx                     # total bytes read = 0
read_loop:
    mov rdi, 0                       # stdin
    lea rsi, [rip + input_buf]
    add rsi, rbx                     # buffer + offset
    mov rdx, INPUT_MAX
    sub rdx, rbx                     # remaining space
    mov rax, 0                       # sys_read
    syscall
    test rax, rax
    jz read_done
    add rbx, rax                     # total += bytes read
    cmp rbx, INPUT_MAX
    jae read_done
    jmp read_loop
read_done:
    mov rdx, rbx                     # rdx = remaining input length
    test rdx, rdx
    jz exit_ok                       # empty input

    # ---------------- set up pointers ----------------
    lea rsi, [rip + input_buf]       # source pointer
    lea r10, [rip + out_buf]         # destination pointer
    xor r8, r8                       # line_len = 0
    lea r9, [rip + b64_table]        # base64 table

process_full:
    cmp rdx, 3
    jb process_remain

    # load three bytes
    movzx rax, byte [rsi]            # b0
