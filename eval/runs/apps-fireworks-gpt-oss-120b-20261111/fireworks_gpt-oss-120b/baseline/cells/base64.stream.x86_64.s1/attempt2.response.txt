.globl _start

.equ INPUT_MAX, 49152
.equ OUT_MAX, 66400

.section .bss
.align 16
input_buf:
    .zero INPUT_MAX
out_buf:
    .zero OUT_MAX

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
_start:
    # ----- read entire input -----
    xor rbx, rbx                # total bytes read = 0
read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, rbx                # buffer + offset
    mov rdx, INPUT_MAX
    sub rdx, rbx                # remaining space
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done
    add rbx, rax                # total += bytes read
    cmp rbx, INPUT_MAX
    jae read_done
    jmp read_loop
read_done:
    mov rcx, rbx                # rcx = input length
    test rcx, rcx
    jz exit_ok                  # empty input, exit

    # ----- set up pointers -----
    lea rsi, [rip + input_buf]  # src pointer
    lea r10, [rip + out_buf]    # dst pointer
    xor r8, r8                  # line_len = 0
    lea r9, [rip + b64_table]   # table pointer

    # ----- process full 3‑byte groups -----
process_full:
    cmp rcx, 3
    jb process_remain

    # load three bytes
    movzx eax, byte ptr [rsi]          # b0
    movzx edx, byte ptr [rsi+1]        # b1
    movzx ebp, byte ptr [rsi+2]        # b2

    # indices a,b,c,d
    mov edi, eax
    shr edi, 2                         # a

    mov esi, eax
    and esi, 3
    shl esi, 4
    mov ebx, edx
    shr ebx, 4
    or esi, ebx                        # b

    mov ebx, edx
    and ebx, 0xF
    shl ebx, 2
    mov edx, ebp
    shr edx, 6
    or ebx, edx                        # c

    mov edx, ebp
    and edx, 0x3F                      # d

    # output a
    mov al, byte ptr [r9 + rdi]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl

    # output b
    mov al, byte ptr [r9 + rsi]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl

    # output c
    mov al, byte ptr [r9 + rbx]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl

    # output d
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl

    # advance to next group
    add rsi, 3
    sub rcx, 3
    jmp process_full

write_nl:
    mov byte ptr [r10], 0x0A
    inc r10
    xor r8, r8
    jmp check_continue

check_continue:
    # after newline we may be in middle of processing a group;
    # determine where we came from
    # If we just wrote d, we need to advance src and rcx
    # This label is reached only from the d‑output check.
    add rsi, 3
    sub rcx, 3
    jmp process_full

process_remain:
    # rcx = 0,1,2
    cmp rcx, 0
    je final_newline
    cmp rcx, 1
    je handle_one
    # rcx == 2
handle_two:
    movzx eax, byte ptr [rsi]          # b0
    movzx edx, byte ptr [rsi+1]        # b1

    # a = b0>>2
    mov edi, eax
    shr edi, 2

    # b = ((b0 &3)<<4) | (b1>>4)
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov ebx, edx
    shr ebx, 4
    or esi, ebx

    # c = ((b1 &0xF) <<2)
    mov ebx, edx
    and ebx, 0xF
    shl ebx, 2

    # output a
    mov al, byte ptr [r9 + rdi]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_two

    # output b
    mov al, byte ptr [r9 + rsi]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_two

    # output c
    mov al, byte ptr [r9 + rbx]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_two

    # output '=' padding
    mov byte ptr [r10], '='
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_two

    jmp final_newline

write_nl_two:
    mov byte ptr [r10], 0x0A
    inc r10
    xor r8, r8
    jmp final_newline

handle_one:
    movzx eax, byte ptr [rsi]          # b0

    # a = b0>>2
    mov edi, eax
    shr edi, 2

    # b = (b0 &3) <<4
    mov esi, eax
    and esi, 3
    shl esi, 4

    # output a
    mov al, byte ptr [r9 + rdi]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_one

    # output b
    mov al, byte ptr [r9 + rsi]
    mov byte ptr [r10], al
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_one

    # output '==' padding
    mov byte ptr [r10], '='
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_one

    mov byte ptr [r10], '='
    inc r10
    inc r8
    cmp r8, 76
    je write_nl_one

    jmp final_newline

write_nl_one:
    mov byte ptr [r10], 0x0A
    inc r10
    xor r8, r8
    jmp final_newline

final_newline:
    cmp r8, 0
    je done_output
    # write final newline
    mov byte ptr [r10], 0x0A
    inc r10
    # r8 no longer needed

done_output:
    # ----- write output buffer -----
    mov rdx, r10
    lea rax, [rip + out_buf]
    sub rdx, rax                # rdx = output length
    mov rdi, 1                  # fd = stdout
    mov rsi, rax                # buffer
    mov rax, 1                  # sys_write
    syscall

    # ----- exit -----
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
