.equ SYSCALL_READ,    63
.equ SYSCALL_WRITE,   64
.equ SYSCALL_EXIT,    93

.equ INPUT_BUF_SIZE,  65536
.equ LINE_BUF_SIZE,   80

.globl _start
.text
_start:
    // read entire input into input_buf, store length in x19
    mov     x19, #0                      // total length
read_loop:
    // x0 = fd 0
    mov     x0, #0
    // x1 = input_buf + total_len
    adrp    x1, input_buf
    add     x1, x1, :lo12:input_buf
    add     x1, x1, x19
    // x2 = remaining space
    mov     x2, #INPUT_BUF_SIZE
    sub     x2, x2, x19
    mov     x8, #SYSCALL_READ
    svc     #0
    cbz     x0, read_done                // EOF
    add     x19, x19, x0
    b       read_loop
read_done:
    // x19 now holds total_len

    // line buffer pointer
    adrp    x20, line_buf
    add     x20, x20, :lo12:line_buf     // x20 = start of line buffer

    // offset iterator in x21
    mov     x21, #0

dump_loop:
    cmp     x21, x19
    b.ge    dump_final_offset

    // bytes remaining = total_len - offset
    sub     x22, x19, x21
    // n = min(16, remaining)
    mov     x23, #16
    cmp     x22, #16
    csel    x23, x22, x23, lo            // x23 = n

    // x24 = current write pointer (starts at line start)
    mov     x24, x20

    // ----- write offset (8 hex digits) -----
    adrp    x25, hex_digits
    add     x25, x25, :lo12:hex_digits
    mov     x26, x21                     // value to convert
    mov     x27, #8
off_hex_loop:
    sub     x28, x27, #1
    lsl     x28, x28, #2                 // shift = (pos-1)*4
    lsr     x29, x26, x28
    and     x29, x29, #0xf
    ldrb    w30, [x25, x29]              // get digit
    strb    w30, [x24], #1
    subs    x27, x27, #1
    b.ne    off_hex_loop

    // two spaces after offset
    mov     w30, #' '
    strb    w30, [x24], #1
    strb    w30, [x24], #1

    // ----- write hex bytes area -----
    mov     x28, #0                     // i = 0
hex_byte_loop:
    cmp     x28, #16
    b.eq    hex_byte_done
    cmp     x28, x23
    b.ge    hex_byte_missing

    // load byte from input_buf
    adrp    x29, input_buf
    add     x29, x29, :lo12:input_buf
    add     x30, x29, x21
    add     x30, x30, x28
    ldrb    w31, [x30]

    // high nibble
    lsr     w0, w31, #4
    and     w0, w0, #0xf
    ldrb    w1, [x25, x0]
    strb    w1, [x24], #1
    // low nibble
    and     w0, w31, #0xf
    ldrb    w1, [x25, x0]
    strb    w1, [x24], #1
    // space
    mov     w1, #' '
    strb    w1, [x24], #1
    b       hex_byte_next

hex_byte_missing:
    // three spaces for missing byte
    mov     w1, #' '
    strb    w1, [x24], #1
    strb    w1, [x24], #1
    strb    w1, [x24], #1

hex_byte_next:
    // extra space after eighth byte
    cmp     x28, #7
    b.ne    hex_byte_continue
    mov     w1, #' '
    strb    w1, [x24], #1
hex_byte_continue:
    add     x28, x28, #1
    b       hex_byte_loop
hex_byte_done:

    // space and '|'
    mov     w1, #' '
    strb    w1, [x24], #1
    mov     w1, #'|'
    strb    w1, [x24], #1

    // ----- write ASCII representation -----
    mov     x28, #0                     // i = 0
ascii_loop:
    cmp     x28, x23
    b.ge    ascii_done
    // load byte
    adrp    x29, input_buf
    add     x29, x29, :lo12:input_buf
    add     x30, x29, x21
    add     x30, x30, x28
    ldrb    w31, [x30]

    // printable? 0x20 <= w31 <= 0x7e
    cmp     w31, #0x20
    b.lt    ascii_put_dot
    cmp     w31, #0x7e
    b.gt    ascii_put_dot
    // printable
    strb    w31, [x24], #1
    b       ascii_next
ascii_put_dot:
    mov     w31, #'.'
    strb    w31, [x24], #1
ascii_next:
    add     x28, x28, #1
    b       ascii_loop
ascii_done:
    // closing '|' and newline
    mov     w1, #'|'
    strb    w1, [x24], #1
    mov     w1, #'\n'
    strb    w1, [x24], #1

    // write the line: length = x24 - line_buf start (x20)
    sub     x2, x24, x20                // length
    mov     x0, #1                      // fd 1
    mov     x1, x20
    mov     x8, #SYSCALL_WRITE
    svc     #0

    // advance offset by 16
    add     x21, x21, #16
    b       dump_loop

// ------------------------------------------------------------
dump_final_offset:
    // x20 still points to start of line_buf, reuse it
    mov     x24, x20

    // write total length as 8 hex digits
    adrp    x25, hex_digits
    add     x25, x25, :lo12:hex_digits
    mov     x26, x19                    // total length
    mov     x27, #8
final_off_loop:
    sub     x28, x27, #1
    lsl     x28, x28, #2
    lsr     x29, x26, x28
    and     x29, x29, #0xf
    ldrb    w30, [x25, x29]
    strb    w30, [x24], #1
    subs    x27, x27, #1
    b.ne    final_off_loop

    // newline
    mov     w30, #'\n'
    strb    w30, [x24], #1

    // write final line
    sub     x2, x24, x20                // length
    mov     x0, #1
    mov     x1, x20
    mov     x8, #SYSCALL_WRITE
    svc     #0

    // exit with status 0
    mov     x0, #0
    mov     x8, #SYSCALL_EXIT
    svc     #0

// ------------------------------------------------------------
.section .rodata
hex_digits:
    .ascii  "0123456789abcdef"

.bss
    .p2align 4
input_buf:
    .space  INPUT_BUF_SIZE
line_buf:
    .space  LINE_BUF_SIZE
</final>
