.global _start
.section .bss
.align 16
input_buf:
    .space 49152
output_buf:
    .space 67000

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
_start:
    # ----- read entire input -----
    lea rsi, [rip + input_buf]          # rsi = write pointer
    mov rdx, 49152                      # max read size
    xor rbx, rbx                        # total bytes read
read_loop:
    mov rax, 0                          # sys_read
    xor rdi, rdi                        # fd = 0 (stdin)
    syscall
    test rax, rax
    jz read_done
    add rbx, rax                        # total += bytes read
    add rsi, rax                        # advance write pointer
    jmp read_loop
read_done:
    # rbx = total input size
    # ----- prepare encoding -----
    lea rsi, [rip + input_buf]          # rsi = input pointer
    lea r9,  [rip + output_buf]         # r9 = output start
    mov rdi, r9                         # rdi = output write pointer
    xor r10, r10                        # r10 = column counter (0..75)

encode_loop:
    cmp rbx, 3
    jb encode_remaining
    # load three bytes
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, dl
    call store_char

    # index1 = ((a & 3) << 4) | (b >> 4)
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov ebp, ebx
    shr ebp, 4
    or edx, ebp
    mov al, dl
    call store_char

    # index2 = ((b & 0xF) << 2) | (c >> 6)
    mov edx, ebx
    and edx, 0xF
    shl edx, 2
    mov ebp, ecx
    shr ebp, 6
    or edx, ebp
    mov al, dl
    call store_char

    # index3 = c & 0x3F
    mov edx, ecx
    and edx, 0x3F
    mov al, dl
    call store_char

    add rsi, 3
    sub rbx, 3
    jmp encode_loop

encode_remaining:
    cmp rbx, 0
    je finish_output

    cmp rbx, 1
    je encode_one_byte

    # rbx == 2
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, dl
    call store_char

    # index1 = ((a & 3) << 4) | (b >> 4)
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov ebp, ebx
    shr ebp, 4
    or edx, ebp
    mov al, dl
    call store_char

    # index2 = (b & 0xF) << 2
    mov edx, ebx
    and edx, 0xF
    shl edx, 2
    mov al, dl
    call store_char

    # padding '='
    mov al, '='
    call store_char
    jmp finish_output

encode_one_byte:
    movzx eax, byte ptr [rsi]

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, dl
    call store_char

    # index1 = (a & 3) << 4
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov al, dl
    call store_char

    # padding '=' twice
    mov al, '='
    call store_char
    mov al, '='
    call store_char
    jmp finish_output

# -------------------------------------------------
# store_char: al = index (0..63) or literal character
# uses rdx for index lookup, r11 clobbered is fine
store_char:
    push rdx                # preserve rdx (may hold column later)
    push rax                # preserve al if it's literal; we'll re-use al
    # if index is <= 63 we look up alphabet, otherwise literal (e.g., '=')
    cmp al, 63
    jg .literal_char
    movzx edx, al
    mov bl, byte ptr [rip + alphabet + rdx]
