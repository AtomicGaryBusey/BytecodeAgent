.global _start
.section .bss
.align 16
input_buf:
    .space 49152
output_buf:
    .space 67000

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
_start:
    # ----- read entire input -----
    lea rsi, [rip + input_buf]          # write pointer
    mov rdx, 49152                      # max read size
    xor rbx, rbx                        # total bytes read
read_loop:
    mov rax, 0                          # sys_read
    xor rdi, rdi                        # fd = 0
    syscall
    test rax, rax
    jz read_done
    add rbx, rax                        # total += bytes read
    add rsi, rax                        # advance write pointer
    jmp read_loop
read_done:
    # rbx = total input size
    # ----- prepare encoding -----
    lea rsi, [rip + input_buf]          # input pointer
    lea rdi, [rip + output_buf]         # output pointer
    xor rcx, rcx                        # column counter (0..75)

    # rbx = remaining bytes
encode_loop:
    cmp rbx, 3
    jb encode_remaining

    # load three bytes
    movzx eax, byte ptr [rsi]           # a
    movzx ebp, byte ptr [rsi+1]         # b
    movzx r8d, byte ptr [rsi+2]         # c

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, dl
    call store_idx

    # index1 = ((a & 3) << 4) | (b >> 4)
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov ebx, ebp
    shr ebx, 4
    or edx, ebx
    mov al, dl
    call store_idx

    # index2 = ((b & 0xF) << 2) | (c >> 6)
    mov edx, ebp
    and edx, 0xF
    shl edx, 2
    mov ebx, r8d
    shr ebx, 6
    or edx, ebx
    mov al, dl
    call store_idx

    # index3 = c & 0x3F
    mov edx, r8d
    and edx, 0x3F
    mov al, dl
    call store_idx

    add rsi, 3
    sub rbx, 3
    jmp encode_loop

encode_remaining:
    cmp rbx, 0
    je finish_output

    cmp rbx, 1
    je encode_one_byte

    # rbx == 2
    movzx eax, byte ptr [rsi]           # a
    movzx ebp, byte ptr [rsi+1]         # b

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, dl
    call store_idx

    # index1 = ((a & 3) << 4) | (b >> 4)
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov ebx, ebp
    shr ebx, 4
    or edx, ebx
    mov al, dl
    call store_idx

    # index2 = (b & 0xF) << 2
    mov edx, ebp
    and edx, 0xF
    shl edx, 2
    mov al, dl
    call store_idx

    # padding '='
    mov al, '='
    call store_lit
    jmp finish_output

encode_one_byte:
    movzx eax, byte ptr [rsi]           # a

    # index0 = a >> 2
    mov edx, eax
    shr edx, 2
    mov al, dl
    call store_idx

    # index1 = (a & 3) << 4
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov al, dl
    call store_idx

    # padding '=' twice
    mov al, '='
    call store_lit
    mov al, '='
    call store_lit
    jmp finish_output

# -------------------------------------------------
# store_idx: al = index (0..63)
# translates via alphabet and stores with line handling
store_idx:
    movzx ebx, al           # ebx = index
    lea r8, [rip + alphabet]
    mov al, byte ptr [r8 + rbx]
    jmp store_common

# store_lit: al = literal character (e.g., '=')
store_lit:
    ; al already contains literal
