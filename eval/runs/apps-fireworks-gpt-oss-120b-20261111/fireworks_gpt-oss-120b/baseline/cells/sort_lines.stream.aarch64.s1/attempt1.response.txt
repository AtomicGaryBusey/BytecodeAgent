	.global _start
	.text
	.align 2

	.equ MAX_INPUT, 32768
	.equ MAX_LINES, 1000

_start:
	// base addresses
	adrp	x9, buf
	add	x9, x9, :lo12:buf          // x9 = buf base
	adrp	x10, line_offsets
	add	x10, x10, :lo12:line_offsets // x10 = offsets array base
	adrp	x11, line_lengths
	add	x11, x11, :lo12:line_lengths // x11 = lengths array base

	// read input
	mov	w12, #0                    // total_len = 0
read_loop:
	sub	w2, #MAX_INPUT, w12       // remaining = MAX_INPUT - total_len
