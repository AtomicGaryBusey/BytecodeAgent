	.global _start
	.text
	.align 2

	.equ    MAX_INPUT, 32768
	.equ    MAX_LINES, 1000

	/* registers usage:
	   x9  = buffer base
	   x10 = offsets array base
	   x12 = lengths array base
	   w13 = total input length
	   w14 = line count
	*/

_start:
	/* load addresses */
	adrp	x9, buf
	add	x9, x9, :lo12:buf
	adrp	x10, line_offsets
	add	x10, x10, :lo12:line_offsets
	adrp	x12, line_lengths
	add	x12, x12, :lo12:line_lengths

	/* read input */
	mov	w13, #0			// total_len = 0
read_loop:
	/* w2 = MAX_INPUT - total_len */
	mov	w2, #MAX_INPUT
	sub	w2, w2, w13
	cbnz	w2, do_read
	b	parse_input		// no space left, stop reading

do_read:
	add	x1, x9, w13		// buf + total_len
