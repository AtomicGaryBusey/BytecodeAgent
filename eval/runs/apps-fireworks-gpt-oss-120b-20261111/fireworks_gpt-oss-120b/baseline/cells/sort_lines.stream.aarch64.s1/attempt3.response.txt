.global _start
.text
.align 2

.equ    MAX_INPUT,    32768
.equ    MAX_LINES,    1000
.equ    SYSCALL_READ,     63
.equ    SYSCALL_WRITE,    64
.equ    SYSCALL_EXIT,     93
.equ    FD_STDIN,         0
.equ    FD_STDOUT,        1

_start:
    // load base addresses
    adrp    x9,  buf
    add     x9,  x9, :lo12:buf          // x9 = buffer base
    adrp    x10, line_offsets
    add     x10, x10, :lo12:line_offsets   // x10 = offsets array base
    adrp    x12, line_lengths
    add     x12, x12, :lo12:line_lengths    // x12 = lengths array base

    // read input fully
    mov     w13, #0                     // total_len = 0
read_loop:
    mov     w2, #MAX_INPUT
    sub     w2, w2, w13                 // w2 = remaining space
    cbz     w2, parse_input
    // x1 = buf + total_len
    add     x1, x9, w13, uxtw #0
    mov     x0, #FD_STDIN
    mov     x8, #SYSCALL_READ
    svc     #0
    cbz     w0, parse_input             // EOF
    add     w13, w13, w0                // total_len += bytes_read
    b       read_loop

parse_input:
    // line parsing
    mov     w14, #0                     // line_count = 0
    mov     w15, #0                     // cursor = 0
    mov     w16, #0                     // line_start = 0

parse_loop:
    cmp     w15, w13
    b.ge    after_parse_loop

    add     x2, x9, w15, uxtw #0
    ldrb    w3, [x2]
    cmp     w3, #0x0A                  // newline ?
    b.ne    not_nl

    // newline found: store line [line_start, w15)
    sub     w4, w15, w16               // length = cursor - line_start
    // store offset
    add     x5, x10, x14, LSL #2
    str     w16, [x5]
    // store length
    add     x5, x12, x14, LSL #2
    str     w4, [x5]
    add     x14, x14, #1               // line_count++
    // advance start
    add     w16, w15, #1
    mov     w15, w16
    b       parse_loop

not_nl:
    add     w15, w15, #1
    b       parse_loop

after_parse_loop:
    // handle final line if any
    cmp     w16, w13
    b.ge    parse_done
    sub     w4, w13, w16               // length = total_len - line_start
    add     x5, x10, x14, LSL #2
    str     w16, [x5]
    add     x5, x12, x14, LSL #2
    str     w4, [x5]
    add     x14, x14, #1               // line_count++

parse_done:
    // -------------------------------------------------
    // sort lines using insertion sort
    // -------------------------------------------------
    mov     x15, #1                     // i = 1
sort_outer:
    cmp     x15, x14
    b.ge    sort_done

    // key = line i
    add     x2, x10, x15, LSL #2
    ldr     w0, [x2]                    // key_offset
    add     x2, x12, x15, LSL #2
    ldr     w1, [x2]                    // key_len

    sub     x16, x15, #1               // j = i-1

sort_inner:
    cmp     x16, #0
    blt     insert_key

    // load line j
    add     x2, x10, x16, LSL #2
    ldr     w2, [x2]                    // off_j
    add     x3, x12, x16, LSL #2
    ldr     w3, [x3]                    // len_j

    // compare line j with key
    // pointers
    add     x20, x9, w2, uxtw #0        // ptr1 = buf + off_j
    add     x21, x9, w0, uxtw #0        // ptr2 = buf + key_offset

    // min_len = (len_j < key_len) ? len_j : key_len
    cmp     w3, w1
    csel    w4, w3, w1, lt              // w4 = min_len

    mov     w5, #0                      // result = 0
    mov     w6, #0                      // index = 0

cmp_loop:
    cmp     w6, w4
    b.ge    after_cmp_loop
    ldrb    w7, [x20, w6, uxtw #0]
    ldrb    w8, [x21, w6, uxtw #0]
    cmp     w7, w8
    b.ne    diff_found
    add     w6, w6, #1
    b       cmp_loop

diff_found:
    sub     w5, w7, w8                  // signed difference
    b       after_cmp_loop

after_cmp_loop:
    cmp     w5, #0
    b.ne    compare_done
    // lengths decide
    sub     w5, w3, w1                  // len_j - key_len (may be zero)
compare_done:

    // if line_j > key, shift
    cmp     w5, #0
    ble     insert_key

    // shift offsets[j+1] = offsets[j]
    add     x5, x10, x16, LSL #2
    add     x5, x5, #4
    str     w2, [x5]
    // shift lengths[j+1] = lengths[j]
    add     x5, x12, x16, LSL #2
    add     x5, x5, #4
    str     w3, [x5]

    sub     x16, x16, #1
    b       sort_inner

insert_key:
    // dest = j+1
    add     x5, x10, x16, LSL #2
    add     x5, x5, #4
    str     w0, [x5]
    add     x5, x12, x16, LSL #2
    add     x5, x5, #4
    str     w1, [x5]

    add     x15, x15, #1
    b       sort_outer

sort_done:
    // -------------------------------------------------
    // output sorted lines
    // -------------------------------------------------
    mov     x18, #0                     // index = 0

output_loop:
    cmp     x18, x14
    b.ge    exit_prog

    // load offset and length
    add     x2, x10, x18, LSL #2
    ldr     w0, [x2]                    // offset
    add     x2, x12, x18, LSL #2
    ldr     w1, [x2]                    // length

    // write line
    mov     x0, #FD_STDOUT
    add     x1, x9, w0, uxtw #0
    mov     w2, w1
    mov     x8, #SYSCALL_WRITE
    svc     #0

    // write newline
    adr     x1, nl
    mov     w2, #1
    mov     x0, #FD_STDOUT
    mov     x8, #SYSCALL_WRITE
    svc     #0

    add     x18, x18, #1
    b       output_loop

exit_prog:
    mov     w0, #0
    mov     x8, #SYSCALL_EXIT
    svc     #0

.section .rodata
.align 2
nl:
    .byte 0x0A

.bss
.align 4
buf:
    .zero   MAX_INPUT
line_offsets:
    .zero   MAX_LINES*4
line_lengths:
    .zero   MAX_LINES*4
