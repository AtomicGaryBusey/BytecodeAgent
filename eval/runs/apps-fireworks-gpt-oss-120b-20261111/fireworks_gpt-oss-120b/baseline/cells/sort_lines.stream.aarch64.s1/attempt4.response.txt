.global _start
.text
.align 2

.equ    MAX_INPUT,    32768
.equ    MAX_LINES,    1000
.equ    SYSCALL_READ,     63
.equ    SYSCALL_WRITE,    64
.equ    SYSCALL_EXIT,     93
.equ    FD_STDIN,         0
.equ    FD_STDOUT,        1

_start:
    // base addresses
    adrp    x9, buf
    add     x9, x9, :lo12:buf          // x9 = buffer base
    adrp    x10, line_offsets
    add     x10, x10, :lo12:line_offsets   // x10 = offsets array base
    adrp    x12, line_lengths
    add     x12, x12, :lo12:line_lengths    // x12 = lengths array base

    // read input
    mov     w13, #0                     // total_len = 0
read_loop:
    mov     w2, #MAX_INPUT
    sub     w2, w2, w13                 // remaining space
    cbz     w2, parse_input
    add     x1, x9, w13, uxtw #0        // buf + total_len
    mov     x0, #FD_STDIN
    mov     x8, #SYSCALL_READ
    svc     #0
    cbz     w0, parse_input             // EOF
    add     w13, w13, w0                // total_len += bytes_read
    b       read_loop

parse_input:
    // parse lines
    mov     w14, #0                     // line_count = 0
    mov     w15, #0                     // cursor = 0
    mov     w16, #0                     // line_start = 0

parse_loop:
    cmp     w15, w13
    b.ge    after_parse_loop

    add     x2, x9, w15, uxtw #0
    ldrb    w3, [x2]
    cmp     w3, #0x0A
    b.ne    not_nl

    // newline: store line [line_start, w15)
    sub     w4, w15, w16               // length
    // offset
    add     x5, x10, x14, lsl #2
    str     w16, [x5]
    // length
    add     x5, x12, x14, lsl #2
    str     w4, [x5]
    add     w14, w14, #1               // line_count++

    // advance start after newline
    add     w16, w15, #1
    add     w15, w16, #0
    b       parse_loop

not_nl:
    add     w15, w15, #1
    b       parse_loop

after_parse_loop:
    // final line if any data left
    cmp     w16, w13
    b.ge    parse_done
    sub     w4, w13, w16               // length
    add     x5, x10, w14, lsl #2
