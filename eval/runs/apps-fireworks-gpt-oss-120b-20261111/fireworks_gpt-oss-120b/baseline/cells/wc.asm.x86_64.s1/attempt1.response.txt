.globl _start

.equ BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 64

.section .bss
.align 16
buf:        .skip BUF_SIZE
outbuf:     .skip OUT_BUF_SIZE

.text
_start:
    xor rbx, rbx          # total bytes
    xor r12, r12          # lines
    xor r13, r13          # words
    xor r14, r14          # in_word flag

read_loop:
    mov rax, 0            # sys_read
    mov rdi, 0            # fd 0
    lea rsi, [rip + buf]
    mov rdx, BUF_SIZE
    syscall
    test rax, rax
    jz done_read          # EOF

    mov r8, rax           # bytes read this chunk
    add rbx, rax          # total bytes
    lea r10, [rip + buf]  # base address
    xor r9, r9            # index i = 0

proc_loop:
    cmp r9, r8
    jae read_loop         # finished this chunk

    mov al, [r10 + r9]    # load byte

    cmp al, 0x0A
    jne not_nl
    inc r12               # newline -> line count
not_nl:
    mov r11, 0
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    jmp not_ws
is_ws:
    mov r11, 1
not_ws:
    cmp r11, 1
    je ws_case

    # non‑whitespace
    cmp r14, 0
    jne non_ws_already
    inc r13               # new word
    mov r14, 1
    jmp after_ws
non_ws_already:
    jmp after_ws

ws_case:
    mov r14, 0            # exit word
after_ws:
    inc r9
    jmp proc_loop

done_read:
    # output: "<lines> <words> <bytes>\n"
    lea rdi, [rip + outbuf]

    mov rax, r12
    call write_num
    mov rdi, rax
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r13
    call write_num
    mov rdi, rax
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, rbx
    call write_num
    mov rdi, rax
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi          # length

    mov rax, 1            # sys_write
    mov rdi, 1            # stdout
    syscall

    mov rax, 60           # sys_exit
    xor rdi, rdi
    syscall

# ------------------------------------------------------------
# write_num: rax = number, rdi = dest pointer
# returns rax = pointer after written digits
write_num:
    mov r8, rax           # copy number
    test r8, r8
    jnz .digit_loop_start
    # zero case
    mov byte ptr [rdi], '0'
    inc rdi
    mov rax, rdi
    ret

.digit_loop_start:
    xor r9, r9            # digit count = 0
.digit_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx               # rax = rax/10, rdx = remainder
    add dl, '0'
    push rdx
    inc r9
    test rax, rax
    jne .digit_loop

.pop_loop:
    dec r9
    jl .pop_done
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    jmp .pop_loop
.pop_done:
    mov rax, rdi
    ret
