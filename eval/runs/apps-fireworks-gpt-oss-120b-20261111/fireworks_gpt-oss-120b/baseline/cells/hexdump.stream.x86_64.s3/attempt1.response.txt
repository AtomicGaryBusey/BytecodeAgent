.globl _start
_start:
    # -------- read entire stdin into input_buf ----------
    xor     rbx, rbx                    # rbx = total length read
read_loop:
    mov     rax, 0                      # sys_read
    mov     rdi, 0                      # fd = stdin
    lea     rsi, [rip + input_buf]
    add     rsi, rbx                    # destination = input_buf + total
    mov     rdx, BUF_SIZE
    sub     rdx, rbx                    # remaining space
    test    rdx, rdx
    jz      read_done
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                    # total += bytes_read
    cmp     rbx, BUF_SIZE
    jl      read_loop
read_done:
    mov     qword ptr [rip + input_len], rbx

    # -------- dump the buffer ----------
    xor     rcx, rcx                    # rcx = offset of current line (0)

dump_loop:
    mov     rax, qword ptr [rip + input_len]
    cmp     rcx, rax
    jae     dump_finish

    # row_len = min(16, input_len - rcx)
    mov     rdx, rax
    sub     rdx, rcx                    # remaining bytes
    mov     r8, 16
    cmp     rdx, r8
    cmovb   r8, rdx                     # r8 = row_len (1..16 or 0)

    # pointers
    lea     rdi, [rip + line_buf]       # destination buffer pointer
    lea     rdx, [rip + input_buf]      # base of input buffer
    add     rdx, rcx                    # rdx = pointer to start of row data

    # ---- write offset (8 hex digits) ----

    mov     r10, 28                     # shift count
