.globl _start

.equ    BUF_SIZE, 65536

.section .bss
    .align 16
input_buf:
    .zero   BUF_SIZE
    .align 8
input_len:
    .zero   8
    .align 16
line_buf:
    .zero   80

.section .rodata
hex_digits:
    .ascii  "0123456789abcdef"

.text
_start:
    xor     rsi, rsi                # total bytes read = 0
read_loop:
    cmp     rsi, BUF_SIZE
    jae     read_done
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rdx, [rip + input_buf]
    add     rdx, rsi                # buffer + offset
    mov     rcx, BUF_SIZE
    sub     rcx, rsi                # remaining space
    mov     rdx, rcx
    syscall
    test    rax, rax
    jz      read_done
    add     rsi, rax                # increase total
    jmp     read_loop
read_done:
    mov     [rip + input_len], rsi

    xor     rdi, rdi                # line offset = 0
output_loop:
    mov     rax, [rip + input_len]
    cmp     rdi, rax
    jae     final_len_line

    mov     rbx, rax
    sub     rbx, rdi                # remaining bytes
    mov     rcx, 16
    cmp     rbx, rcx
    jae     have_full
    mov     rcx, rbx                # rcx = n (bytes this line)
have_full:
    # rdi = offset, rcx = n, rbx = n (also keep in rbx for later)
    mov     rbx, rcx

    # ---- build line in line_buf ----
    lea     r8, [rip + line_buf]    # dest pointer

    # offset as 8 hex digits (write from end backwards)
    mov     r9, rdi                 # value
    lea     r10, [r8 + 7]           # pointer to last digit
    mov     r11, 8
off_rev_loop:
    mov     rax, r9
    xor     rdx, rdx
    mov     rcx, 16
    div     rcx                     # rdx = remainder
    movzx   rdx, dl
    mov     al, byte ptr [rip + hex_digits + rdx]
