.globl _start
.equ BUF_SIZE, 65536

.section .bss
.align 16
input_buf:
    .zero BUF_SIZE
.align 8
input_len:
    .zero 8
.align 16
line_buf:
    .zero 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
_start:
    xor     rbx, rbx                # total bytes read = 0
read_loop:
    cmp     rbx, BUF_SIZE
    jae     read_done
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]
    add     rsi, rbx                # buffer + offset
    mov     rdx, BUF_SIZE
    sub     rdx, rbx                # remaining space
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                # increase total
    jmp     read_loop
read_done:
    mov     [rip + input_len], rbx

    xor     rdi, rdi                # offset = 0
output_loop:
    mov     rax, qword [rip + input_len]
