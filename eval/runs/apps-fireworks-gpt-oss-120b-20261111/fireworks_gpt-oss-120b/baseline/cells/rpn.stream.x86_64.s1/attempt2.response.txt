.globl _start
.text
_start:
    # ----- read whole input (max 65536 bytes) into input_buf -----
    xor    rcx, rcx                # rcx = current length
read_loop:
    mov    rax, 0                  # sys_read
    mov    rdi, 0                  # stdin
    lea    rsi, [rip + input_buf]  # base address
    add    rsi, rcx                # rsi = base + len
    mov    rdx, 65536
    sub    rdx, rcx                # remaining space
    syscall
    test   rax, rax
    jz     after_read
    add    rcx, rax
    jmp    read_loop
after_read:
    mov    [rip + input_len], rcx

    # ----- set up pointers to input buffer -----
    lea    rbx, [rip + input_buf]          # rbx = cur ptr
    mov    r12, rbx
    add    r12, [rip + input_len]          # r12 = end ptr

line_loop:
    cmp    rbx, r12
    jae    exit_program

    # start new line
    mov    qword [rip + stack_ptr], 0
