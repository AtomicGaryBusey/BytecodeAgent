.globl _start
.text
_start:
    # read entire input into input_buf (max 65536 bytes)
    mov     rdi, 0                      # fd = stdin
    lea     rsi, [rip + input_buf]
    xor     rdx, rdx                    # offset = 0
read_loop:
    mov     rax, 0                      # sys_read
    mov     rdi, 0
    lea     rsi, [rip + input_buf]
    mov     rdx, 65536
    sub     rdx, [rip + input_len]
    syscall
    test    rax, rax
    jz      after_read
    add     [rip + input_len], rax
    jmp     read_loop
after_read:
    # rdx = current pointer, r10 = end pointer
    lea     rdx, [rip + input_buf]
    lea     r10, [rip + input_buf + input_len]
