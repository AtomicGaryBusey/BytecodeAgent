.globl _start
.text
_start:
    # ---------- read whole input (max 65536 bytes) ----------
    xor     rcx, rcx                    # rcx = current length
read_loop:
    mov     rax, 0                      # sys_read
    mov     rdi, 0                      # stdin
    lea     rsi, [rip + input_buf]      # base address
    add     rsi, rcx                    # rsi = base + len
    mov     rdx, INPUT_MAX
    sub     rdx, rcx                    # remaining space
    syscall
    test    rax, rax
    jz      after_read
    add     rcx, rax
    cmp     rcx, INPUT_MAX
    jae     after_read
    jmp     read_loop
after_read:
    mov     qword ptr [rip + input_len], rcx

    # ---------- pointers ----------
    lea     rbx, [rip + input_buf]      # rbx = cur ptr
    mov     r12, rbx
    add     r12, rcx                    # r12 = end ptr

line_loop:
    cmp     rbx, r12
    jae     exit_program

    # ---------- find end of line ----------
    mov     r13, rbx                    # r13 = line start
    mov     r14, rbx                    # r14 = scan ptr
find_nl:
    cmp     r14, r12
    jae     no_nl
    mov     al, byte ptr [r14]
    cmp     al, 0x0A
    je      has_nl
    inc     r14
    jmp     find_nl
has_nl:
    # r14 points to newline, line bytes are [r13, r14)
    jmp     line_found
no_nl:
    # line ends at end of input
    mov     r14, r12
line_found:
    # ---------- process line ----------
    xor     r10d, r10d                  # error flag = 0
    mov     qword ptr [rip + stack_ptr], 0

    mov     rsi, r13                    # token pointer

token_loop:
    cmp     rsi, r14
    jae     token_done
    # skip separators (space or tab)
    mov     al, byte ptr [rsi]
    cmp     al, 0x20
    je      inc_si
    cmp     al, 0x09
    je      inc_si
    jmp     token_start
inc_si:
    inc     rsi
    jmp     token_loop

token_start:
    mov     rdx, rsi                    # token start ptr
scan_tok:
    cmp     rsi, r14
    jae     token_end
    mov     al, byte ptr [rsi]
    cmp     al, 0x20
    je      token_end
    cmp     al, 0x09
    je      token_end
    inc     rsi
    jmp     scan_tok
token_end:
    mov     r11, rsi                    # token end ptr
    sub     r11, rdx                    # token_len in r11
    cmp     r11, 0
    je      token_loop                  # empty token (should not happen)

    # if error already set, skip rest of line
    cmp     r10d, 0
    jne     line_error_skip

    # check for single-char operator
    cmp     r11, 1
    jne     not_operator
    mov     al, byte ptr [rdx]
    cmp     al, '+'
    je      op_plus
    cmp     al, '-'
    je      op_minus
    cmp     al, '*'
    je      op_mul
    cmp     al, '/'
    je      op_div
    cmp     al, '%'
    je      op_mod
    jmp     not_operator

# ---------- operators ----------
op_plus:
    call    pop_two
    add     rax, rcx
    jmp     push_result
op_minus:
    call    pop_two
    sub     rax, rcx
    jmp     push_result
op_mul:
    call    pop_two
    mul     rcx                     # unsigned mul, low part in rax
    jmp     push_result
op_div:
    call    pop_two
    test    rcx, rcx
    jz      set_error
    mov     r8, 0x8000000000000000
    cmp     rax, r8
    jne     do_idiv
    cmp     rcx, -1
    je      set_error
do_idiv:
    cqo
    idiv    rcx
    jmp     push_result
op_mod:
    call    pop_two
    test    rcx, rcx
    jz      set_error
    mov     r8, 0x8000000000000000
    cmp     rax, r8
    jne     do_idiv_mod
    cmp     rcx, -1
    je      set_error
do_idiv_mod:
    cqo
    idiv    rcx
    mov     rax, rdx
    jmp     push_result

set_error:
    mov     r10d, 1
    jmp     token_continue

push_result:
    # rax holds result
    mov     rbx, qword ptr [rip + stack_ptr]
    lea     rdx, [rip + stack + rbx*8]
