	.globl _start
	.text
_start:
	// read all input into input_buf
	mov	x19, #0				// total bytes read
read_loop:
	mov	x0, #0				// fd = stdin
	adrp	x1, input_buf
	add	x1, x1, :lo12:input_buf
	mov	x2, #32768
	mov	x8, #63			// sys_read
	svc	#0
	cbz	x0, read_done			// EOF
	add	x19, x19, x0
	b	read_loop
read_done:
	// store total length
	adrp	x0, input_len
	add	x0, x0, :lo12:input_len
	str	w19, [x0]

	// parse lines
	mov	w20, #0				// start offset
	mov	w21, #0				// line count index
	adrp	x0, input_len
	add	x0, x0, :lo12:input_len
	ldr	w22, [x0]			// total length
	cmp	w22, #0
	beq	no_lines_to_parse

	mov	w23, #0				// current position
parse_loop:
	cmp	w23, w22
	bge	parse_done

	adrp	x0, input_buf
	add	x0, x0, :lo12:input_buf
	add	x0, x0, w23, uxtw
	ldrb	w24, [x0]

	cmp	w24, #10			// newline?
	bne	not_newline

	// newline found: store line [start, pos)
	sub	w25, w23, w20			// length = pos - start
	adrp	x0, line_off
	add	x0, x0, :lo12:line_off
	add	x0, x0, w21, lsl #2
	str	w20, [x0]
	adrp	x1, line_len
	add	x1, x1, :lo12:line_len
	add	x1, x1, w21, lsl #2
	str	w25, [x1]
	add	w21, w21, #1
	add	w20, w23, #1			// new start = pos+1
not_newline:
	add	w23, w23, #1
	b	parse_loop
parse_done:
	// final line without trailing newline?
	cmp	w20, w22
	bge	no_final_line
	sub	w25, w22, w20			// length = total - start
	adrp	x0, line_off
	add	x0, x0, :lo12:line_off
	add	x0, x0, w21, lsl #2
	str	w20, [x0]
	adrp	x1, line_len
	add	x1, x1, :lo12:line_len
	add	x1, x1, w21, lsl #2
	str	w25, [x1]
	add	w21, w21, #1
no_final_line:
	adrp	x0, line_cnt
	add	x0, x0, :lo12:line_cnt
	str	w21, [x0]

no_lines_to_parse:
	// sort lines with bubble sort
	adrp	x20, input_buf
	add	x20, x20, :lo12:input_buf	// base address of input_buf in x20

	adrp	x0, line_cnt
	add	x0, x0, :lo12:line_cnt
	ldr	w26, [x0]			// n = line count
	cmp	w26, #1
	ble	sort_done

	mov	w27, #0				// i = 0
outer_loop:
	// if i >= n-1, done
	add	w28, w26, #-1
	cmp	w27, w28
	bge	sort_done

	// limit = n - i - 1
	sub	w29, w26, w27
	sub	w29, w29, #1			// number of comparisons this pass
	mov	w30, #0				// j = 0
inner_loop:
	cmp	w30, w29
	bge	inner_done

	// load a index = j
	mov	w31, w30
	// load b index = j+1
	add	w8, w30, #1

	// offsets
	adrp	x0, line_off
	add	x0, x0, :lo12:line_off
	add	x0, x0, w31, lsl #2
	ldr	w5, [x0]			// a_off
	adrp	x1, line_off
	add	x1, x1, :lo12:line_off
	add	x1, x1, w8, lsl #2
	ldr	w9, [x1]			// b_off

	// lengths
	adrp	x2, line_len
	add	x2, x2, :lo12:line_len
	add	x2, x2, w31, lsl #2
	ldr	w6, [x2]			// a_len
	adrp	x3, line_len
	add	x3, x3, :lo12:line_len
	add	x3, x3, w8, lsl #2
	ldr	w10, [x3]			// b_len

	// compare a and b
	// min_len = (a_len > b_len) ? b_len : a_len
	mov	w16, w6
	cmp	w6, w10
	csel	w16, w10, w6, hi		// if a_len > b_len select b_len

	// pointers
	add	x21, x20, w5, uxtw		// a_ptr
	add	x22, x20, w9, uxtw		// b_ptr
	mov	w17, w16			// remaining = min_len
	cmp	w17, #0
	beq	compare_done_min

compare_loop:
	ldrb	w18, [x21], #1
	ldrb	w19, [x22], #1
	cmp	w18, w19
	bne	diff_found
	subs	w17, w17, #1
	bne	compare_loop

compare_done_min:
	// all min bytes equal, decide based on lengths
	cmp	w6, w10
	bgt	need_swap			// a_len > b_len => a > b
	b	inner_next			// a <= b, no swap
diff_found:
	cmp	w18, w19
	ble	inner_next			// a <= b
need_swap:
	// swap offsets
	adrp	x0, line_off
	add	x0, x0, :lo12:line_off
	add	x0, x0, w31, lsl #2		// addr a_off
	ldr	w24, [x0]			// a_off
	add	x1, x0, #4			// not correct, recompute b address
	adrp	x1, line_off
	add	x1, x1, :lo12:line_off
	add	x1, x1, w8, lsl #2		// addr b_off
	ldr	w25, [x1]			// b_off
	str	w25, [x0]			// store b_off at a
	str	w24, [x1]			// store a_off at b

	// swap lengths
	adrp	x2, line_len
	add	x2, x2, :lo12:line_len
	add	x2, x2, w31, lsl #2		// addr a_len
	ldr	w24, [x2]
	adrp	x3, line_len
	add	x3, x3, :lo12:line_len
	add	x3, x3, w8, lsl #2		// addr b_len
	ldr	w25, [x3]
	str	w25, [x2]
	str	w24, [x3]

inner_next:
	add	w30, w30, #1
	b	inner_loop
inner_done:
	add	w27, w27, #1
	b	outer_loop

sort_done:
	// output sorted lines
	adrp	x0, line_cnt
	add	x0, x0, :lo12:line_cnt
	ldr	w26, [x0]			// n
	cbnz	w26, do_output
	b	exit_program

do_output:
	mov	w30, #0				// idx = 0
output_loop:
	cmp	w30, w26
	bge	output_done

	// load offset and length
	adrp	x0, line_off
	add	x0, x0, :lo12:line_off
	add	x0, x0, w30, lsl #2
	ldr	w5, [x0]			// off
	adrp	x1, line_len
	add	x1, x1, :lo12:line_len
	add	x1, x1, w30, lsl #2
	ldr	w6, [x1]			// len

	// write line bytes if length > 0
	cbnz	w6, write_line
	b	write_newline

write_line:
	mov	x0, #1				// fd = stdout
	add	x1, x20, w5, uxtw		// buf = input_buf + off
	mov	w2, w6
	mov	x8, #64			// sys_write
	svc	#0
write_newline:
	mov	x0, #1
	adr	x1, newline
	mov	w2, #1
	mov	x8, #64
	svc	#0

	add	w30, w30, #1
	b	output_loop
output_done:
	// exit
exit_program:
	mov	x0, #0
	mov	x8, #93			// sys_exit
	svc	#0

	.section .rodata
newline:
	.byte	0x0a

	.bss
	.align 8
input_buf:
	.space	32768
	.align 4
line_off:
	.space	4000			// 1000 * 4
	.align 4
line_len:
	.space	4000
	.align 4
input_len:
	.word	0
	.align 4
line_cnt:
	.word	0
