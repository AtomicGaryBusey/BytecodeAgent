	.global	_start
	.text
	.p2align	2

_start:
	// x19 = address of input buffer
	adrp	x19, input_buf
	add	x19, x19, :lo12:input_buf
	// x20 = address of line buffer
	adrp	x20, line_buf
	add	x20, x20, :lo12:line_buf
	// x25 = address of hex table
	adrp	x25, hex_digits
	add	x25, x25, :lo12:hex_digits

	// x21 = total length = 0
	mov	x21, xzr

read_loop:
	// x0 = fd 0
	mov	x0, #0
	// x1 = buffer + total_len
	add	x1, x19, x21
	// x2 = max read size (65536)
	mov	x2, #65536
	// syscall read = 63
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x21, x21, x0			// total_len += bytes_read
	b	read_loop

read_done:
	// if total_len == 0, skip rows
	cmp	x21, #0
	beq	print_final_offset

	// x22 = offset = 0
	mov	x22, xzr

row_loop:
	// if offset >= total_len, done
	cmp	x22, x21
	bge	rows_done

	// remaining = total_len - offset
	sub	x23, x21, x22
	// row_len = min(16, remaining)
	mov	x24, #16
	cmp	x23, #16
	csel	x24, x23, x24, lo		// unsigned lower -> if remaining < 16 use remaining

	// x9 = pointer into line buffer (start)
	mov	x9, x20

	// ---- write offset (8 hex digits) ----
	mov	x10, x22			// value to format
	mov	x11, #28			// shift amount
	mov	x12, #8				// digit count
offset_digit_loop:
	lsr	x13, x10, x11
	and	x13, x13, #0xf
	ldrb	w13, [x25, x13]
	strb	w13, [x9], #1
	sub	x11, x11, #4
	subs	x12, x12, #1
	bne	offset_digit_loop

	// two spaces
	mov	w13, #' '
	strb	w13, [x9], #1
	strb	w13, [x9], #1

	// ---- hex byte area ----
	mov	x28, #0				// i = 0
hex_byte_loop:
	cmp	x28, #16
	bge	hex_byte_done

	cmp	x28, x24
	bcs	hex_missing_byte		// i >= row_len -> missing

	// load byte: buf + offset + i
	add	x14, x19, x22
	add	x14, x14, x28
	ldrb	w15, [x14]

	// high nibble
	ubfx	w16, w15, #4, #4
	ldrb	w16, [x25, w16]
	strb	w16, [x9], #1
	// low nibble
	and	w17, w15, #0xf
	ldrb	w17, [x25, w17]
	strb	w17, [x9], #1
	// space
	mov	w18, #' '
	strb	w18, [x9], #1
	b	hex_byte_next

hex_missing_byte:
	// three spaces for missing byte
	mov	w18, #' '
	strb	w18, [x9], #1
	strb	w18, [x9], #1
	strb	w18, [x9], #1

hex_byte_next:
	// extra space after 8th byte (i == 7)
	cmp	x28, #7
	bne	hex_byte_cont
	mov	w18, #' '
	strb	w18, [x9], #1
hex_byte_cont:
	add	x28, x28, #1
	b	hex_byte_loop

hex_byte_done:
	// space, '|'
	mov	w18, #' '
	strb	w18, [x9], #1
	mov	w18, #'|'
	strb	w18, [x9], #1

	// ---- ascii column ----
	mov	x28, #0				// i = 0
ascii_loop:
	cmp	x28, x24
	bcs	ascii_done			// i >= row_len

	add	x14, x19, x22
	add	x14, x14, x28
	ldrb	w15, [x14]

	// printable check 0x20 <= byte <= 0x7e
	cmp	w15, #0x20
	blt	ascii_dot
	cmp	w15, #0x7e
	bgt	ascii_dot
	// printable
	mov	w18, w15
	b	ascii_store

ascii_dot:
	mov	w18, #'.'

ascii_store:
	strb	w18, [x9], #1

	add	x28, x28, #1
	b	ascii_loop

ascii_done:
	// trailing '|', newline
	mov	w18, #'|'
	strb	w18, [x9], #1
	mov	w18, #'\n'
	strb	w18, [x9], #1

	// write line
	// length = x9 - line_buf
	sub	x2, x9, x20			// count
	mov	x1, x20			// buf
	mov	x0, #1			// stdout
	mov	x8, #64			// write
	svc	#0

	// offset += 16
	add	x22, x22, #16
	b	row_loop

rows_done:
	// fall through to print final offset

print_final_offset:
	// x22 = total_len (x21)
	mov	x22, x21

	// x9 = line buffer start
	mov	x9, x20

	// write offset (8 hex digits)
	mov	x10, x22
	mov	x11, #28
	mov	x12, #8
offset_final_loop:
	lsr	x13, x10, x11
	and	x13, x13, #0xf
	ldrb	w13, [x25, x13]
	strb	w13, [x9], #1
	sub	x11, x11, #4
	subs	x12, x12, #1
	bne	offset_final_loop

	// newline
	mov	w13, #'\n'
	strb	w13, [x9], #1

	// write final line
	sub	x2, x9, x20
	mov	x1, x20
	mov	x0, #1
	mov	x8, #64
	svc	#0

	// exit
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// should never reach here
	b	.

	// --------------------------------------------------------
	// Data sections
	.section	.rodata
	.p2align	2
hex_digits:
	.asciz	"0123456789abcdef"

	.section	.bss
	.p2align	4
	.align	4
input_buf:
	.space	65536
	.align	4
line_buf:
	.space	80
	.text
	.p2align	2
