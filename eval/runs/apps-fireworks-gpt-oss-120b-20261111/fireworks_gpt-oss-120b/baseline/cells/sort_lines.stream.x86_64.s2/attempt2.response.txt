.globl _start

.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.section .bss
.align 16
input_buf:
    .zero MAX_INPUT
.align 8
line_ptrs:
    .zero MAX_LINES*8
.align 4
line_lens:
    .zero MAX_LINES*4

.section .rodata
nl:
    .byte 0x0A

.text
_start:
    # r12 = input buffer base
    lea r12, [rip + input_buf]
    xor r13d, r13d                 # total_len = 0

read_loop:
    mov rax, SYS_READ
    xor rdi, rdi                  # fd 0
    lea rsi, [r12 + r13]          # buffer + total_len
    mov rdx, MAX_INPUT
    sub rdx, r13d                 # remaining space
