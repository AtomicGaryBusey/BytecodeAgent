.globl _start
.section .bss
input_buf:
    .zero 65536
line_buf:
    .zero 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .text
_start:
    xor rbx, rbx                # total_len = 0
read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    sub rdx, rbx                # remaining space
    mov rax, 0                  # sys_read
    syscall
    cmp rax, 0
    je read_done
    test rax, rax
    jle exit_error
    add rbx, rax                # total_len += bytes_read
    jmp read_loop
read_done:
    xor r12, r12                # line_offset = 0
dump_loop:
    cmp r12, rbx
    jae dump_done

    mov rdx, rbx
    sub rdx, r12                # remaining bytes
    mov r13, 16
    cmp rdx, r13
    cmova rdx, r13              # rdx = n (bytes in this line)

    # r14 = pointer to start of line in input_buf
    lea r14, [rip + input_buf]
    add r14, r12                # r14 = input_buf + line_offset

    # r15 = pointer to line_buf
    lea r15, [rip + line_buf]

    # ---- offset (8 hex digits) ----
    mov rax, r12                # offset value
    mov rcx, 8
offset_digit_loop:
    dec rcx
    mov rbx, rax
    shr rbx, 28
    and rbx, 0xF
    mov bl, byte [rip + hex_digits + rbx]
    mov byte [r15], bl
    inc r15
    shl rax, 4
    test rcx, rcx
    jnz offset_digit_loop

    # two spaces after offset
    mov byte [r15], ' '
    inc r15
    mov byte [r15], ' '
    inc r15

    # ---- hex bytes (16 slots) ----
    xor rcx, rcx                # i = 0
hex_byte_loop:
    cmp rcx, 16
    je hex_bytes_done

    cmp rcx, rdx
    jb have_byte
    # write three spaces for missing byte
    mov byte [r15], ' '
    inc r15
    mov byte [r15], ' '
    inc r15
    mov byte [r15], ' '
    inc r15
    jmp after_hex_byte

have_byte:
    movzx rax, byte [r14 + rcx]
    mov rbx, rax
    shr rbx, 4
    and rbx, 0xF
    mov bl, byte [rip + hex_digits + rbx]
    mov byte [r15], bl
    inc r15
    and rax, 0xF
    mov bl, byte [rip + hex_digits + rax]
    mov byte [r15], bl
    inc r15
    mov byte [r15], ' '
    inc r15

after_hex_byte:
    inc rcx
    cmp rcx, 8
    jne hex_byte_loop
    # extra space after 8th byte
    mov byte [r15], ' '
    inc r15
    jmp hex_byte_loop

hex_bytes_done:
    # space and pipe before ascii
    mov byte [r15], ' '
    inc r15
    mov byte [r15], '|'
    inc r15

    # ---- ascii column ----
    xor rcx, rcx                # i = 0
ascii_loop:
    cmp rcx, rdx
    jae ascii_done
    movzx rax, byte [r14 + rcx]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7e
    jg ascii_dot
    # printable
    mov byte [r15], al
    inc r15
    jmp ascii_next
ascii_dot:
    mov byte [r15], '.'
    inc r15
ascii_next:
    inc rcx
    jmp ascii_loop
ascii_done:

    # closing pipe and newline
    mov byte [r15], '|'
    inc r15
    mov byte [r15], 0x0a
    inc r15

    # write line
    mov rdi, 1                  # stdout fd
    lea rsi, [rip + line_buf]
    mov rdx, r15
    sub rdx, rsi                # length = r15 - line_buf
    mov rax, 1                  # sys_write
    syscall

    add r12, 16                 # line_offset += 16
    jmp dump_loop

dump_done:
    # ---- final offset line (total length) ----
    lea r15, [rip + line_buf]
    mov rax, rbx                # total_len
    mov rcx, 8
final_offset_loop:
    dec rcx
    mov rdx, rax
    shr rdx, 28
    and rdx, 0xF
    mov dl, byte [rip + hex_digits + rdx]
    mov byte [r15], dl
    inc r15
    shl rax, 4
    test rcx, rcx
    jnz final_offset_loop

    mov byte [r15], 0x0a
    inc r15

    # write final offset line
    mov rdi, 1
    lea rsi, [rip + line_buf]
    mov rdx, r15
    sub rdx, rsi
    mov rax, 1
    syscall

    # exit
    mov rax, 60
    xor rdi, rdi
    syscall

exit_error:
    mov rax, 60
    mov rdi, 1
    syscall
