.globl _start

.section .bss
input_buf:
    .zero 65536
line_buf:
    .zero 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .text
_start:
    xor     r12, r12                # total_len = 0

read_loop:
    mov     rdi, 0                  # stdin
    lea     rsi, [rip + input_buf]
    mov     rdx, 65536
    sub     rdx, r12                # remaining buffer space
    mov     rax, 0                  # sys_read
    syscall
    cmp     rax, 0
    je      read_done
    test    rax, rax
    jle     exit_error
    add     r12, rax                # total_len += bytes_read
    jmp     read_loop

read_done:
    xor     r13, r13                # line_offset = 0

dump_loop:
    cmp     r13, r12
    jae     dump_done

    mov     rdx, r12
    sub     rdx, r13                # remaining bytes
    cmp     rdx, 16
    jae     set_full
    ; rdx already < 16
    jmp     after_min
set_full:
    mov     rdx, 16
after_min:
    ; source pointer
    lea     r9, [rip + input_buf]
    add     r9, r13
    ; destination pointer
    lea     r10, [rip + line_buf]

    ; ----- offset (8 hex digits) -----
    mov     rax, r13
    mov     rcx, 8
    lea     rsi, [rip + hex_digits]
off_digit:
    dec     rcx
    mov     rbx, rax
    shr     rbx, 28
    and     rbx, 0xF
    movzx   ebx, byte ptr [rsi + rbx]
    mov     byte ptr [r10], bl
    inc     r10
    shl     rax, 4
    test    rcx, rcx
    jnz     off_digit

    ; two spaces after offset
    mov     byte ptr [r10], ' '
    inc     r10
    mov     byte ptr [r10], ' '
    inc     r10

    ; ----- hex bytes (16 slots) -----
    xor     rcx, rcx                # i = 0
hex_loop:
    cmp     rcx, 16
    je      hex_done
    cmp     rcx, rdx
    jb      have_byte
    ; three spaces for missing byte
    mov     byte ptr [r10], ' '
    inc     r10
    mov     byte ptr [r10], ' '
    inc     r10
    mov     byte ptr [r10], ' '
    inc     r10
    jmp     after_hex
have_byte:
    movzx   eax, byte ptr [r9 + rcx]
    mov     rbx, rax
    shr     rbx, 4
    and     rbx, 0xF
    movzx   ebx, byte ptr [rsi + rbx]
    mov     byte ptr [r10], bl
    inc     r10
    and     eax, 0xF
    movzx   ebx, byte ptr [rsi + rax]
    mov     byte ptr [r10], bl
    inc     r10
    mov     byte ptr [r10], ' '
    inc     r10
after_hex:
    inc     rcx
    cmp     rcx, 8
    jne     hex_loop
    ; extra space after the eighth byte
    mov     byte ptr [r10], ' '
    inc     r10
    jmp     hex_loop
hex_done:

    ; space and pipe before ASCII
    mov     byte ptr [r10], ' '
    inc     r10
    mov     byte ptr [r10], '|'
    inc     r10

    ; ----- ASCII column -----
    xor     rcx, rcx
ascii_loop:
    cmp     rcx, rdx
    jae     ascii_done
    movzx   eax, byte ptr [r9 + rcx]
    cmp     al, 0x20
    jl      ascii_dot
    cmp     al, 0x7e
    jg      ascii_dot
    ; printable
    mov     byte ptr [r10], al
    inc     r10
    jmp     ascii_next
ascii_dot:
    mov     byte ptr [r10], '.'
    inc     r10
ascii_next:
    inc     rcx
    jmp     ascii_loop
ascii_done:
    mov     byte ptr [r10], '|'
    inc     r10
    mov     byte ptr [r10], 0x0a
    inc     r10

    ; ----- write line -----
    mov     rdi, 1                  # stdout
    lea     rsi, [rip + line_buf]
    mov     rdx, r10
    lea     rcx, [rip + line_buf]
    sub     rdx, rcx                # length = r10 - line_buf
    mov     rax, 1                  # sys_write
    syscall

    add     r13, 16                 # next line offset
    jmp     dump_loop

dump_done:
    ; ----- final offset line (total length) -----
    lea     r10, [rip + line_buf]
    mov     rax, r12
    mov     rcx, 8
    lea     rsi, [rip + hex_digits]
final_off:
    dec     rcx
    mov     rdx, rax
    shr     rdx, 28
    and     rdx, 0xF
    movzx   edx, byte ptr [rsi + rdx]
    mov     byte ptr [r10], dl
    inc     r10
    shl     rax, 4
    test    rcx, rcx
    jnz     final_off
    mov     byte ptr [r10], 0x0a
    inc     r10

    ; write final offset line
    mov     rdi, 1
    lea     rsi, [rip + line_buf]
    mov     rdx, r10
    lea     rcx, [rip + line_buf]
    sub     rdx, rcx
    mov     rax, 1
    syscall

    ; exit
    mov     rax, 60
    xor     rdi, rdi
    syscall

exit_error:
    mov     rax, 60
    mov     rdi, 1
    syscall
