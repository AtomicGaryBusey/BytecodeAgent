.globl _start

.section .bss
input_buf:
    .zero 65536
line_buf:
    .zero 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .text
_start:
    xor     rbx, rbx                # total_len = 0

read_loop:
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]
    mov     rdx, 65536
    sub     rdx, rbx                # remaining buffer space
    mov     rax, 0                  # sys_read
    syscall
    cmp     rax, 0
    je      read_done
    test    rax, rax
    jle     exit_error
    add     rbx, rax                # total_len += bytes_read
    jmp     read_loop

read_done:
    xor     r12, r12                # line_offset = 0

dump_loop:
    cmp     r12, rbx
    jae     dump_done

    mov     rdx, rbx
    sub     rdx, r12                # remaining bytes
    mov     r13, 16
    cmp     rdx, r13
    cmova   rdx, r13                # rdx = bytes_this_line (n)

    lea     r14, [rip + input_buf]  # src base
    add     r14, r12                # src pointer for this line

    lea     r15, [rip + line_buf]   # dst pointer

    # ---- offset (8 hex digits) ----
    mov     rax, r12                # offset value
    mov     rcx, 8
    lea     rsi, [rip + hex_digits]# hex table

offset_digit_loop:
    dec     rcx
    mov     rbx, rax
    shr     rbx, 28
    and     rbx, 0xF
    movzx   ebx, byte ptr [rsi + rbx]
    mov     byte ptr [r15], bl
    inc     r15
    shl     rax, 4
    test    rcx, rcx
    jnz     offset_digit_loop

    # two spaces after offset
    mov     byte ptr [r15], ' '
    inc     r15
    mov     byte ptr [r15], ' '
    inc     r15

    # ---- hex bytes (16 slots) ----
    xor     rcx, rcx                # i = 0
hex_byte_loop:
    cmp     rcx, 16
    je      hex_bytes_done

    cmp     rcx, rdx
    jb      have_byte
    # three spaces for missing byte
    mov     byte ptr [r15], ' '
    inc     r15
    mov     byte ptr [r15], ' '
    inc     r15
    mov     byte ptr [r15], ' '
    inc     r15
    jmp     after_hex_byte

have_byte:
    movzx   eax, byte ptr [r14 + rcx]   # load input byte
    mov     rbx, rax
    shr     rbx, 4
    and     rbx, 0xF
    movzx   ebx, byte ptr [rsi + rbx]
    mov     byte ptr [r15], bl
    inc     r15
    and     eax, 0xF
    movzx   ebx, byte ptr [rsi + rax]
    mov     byte ptr [r15], bl
    inc     r15
    mov     byte ptr [r15], ' '
    inc     r15

after_hex_byte:
    inc     rcx
    cmp     rcx, 8
    jne     hex_byte_loop
    # extra space after 8th byte
    mov     byte ptr [r15], ' '
    inc     r15
    jmp     hex_byte_loop

hex_bytes_done:
    # space and pipe before ascii
    mov     byte ptr [r15], ' '
    inc     r15
    mov     byte ptr [r15], '|'
    inc     r15

    # ---- ascii column ----
    xor     rcx, rcx
ascii_loop:
    cmp     rcx, rdx
    jae     ascii_done
    movzx   eax, byte ptr [r14 + rcx]
    cmp     al, 0x20
    jl      ascii_dot
    cmp     al, 0x7e
    jg      ascii_dot
    # printable
    mov     byte ptr [r15], al
    inc     r15
    jmp     ascii_next
ascii_dot:
    mov     byte ptr [r15], '.'
    inc     r15
ascii_next:
    inc     rcx
    jmp     ascii_loop

ascii_done:
    # closing pipe and newline
    mov     byte ptr [r15], '|'
    inc     r15
    mov     byte ptr [r15], 0x0a
    inc     r15

    # write line
    mov     rdi, 1                  # stdout
    lea     rsi, [rip + line_buf]
    mov     rdx, r15
    lea     rcx, [rip + line_buf]
    sub     rdx, rcx                # length = r15 - line_buf
    mov     rax, 1                  # sys_write
    syscall

    add     r12, 16                 # next line offset
    jmp     dump_loop

dump_done:
    # ---- final offset line (total length) ----
    lea     r15, [rip + line_buf]
    mov     rax, rbx                # total_len
    mov     rcx, 8
    lea     rsi, [rip + hex_digits]

final_offset_loop:
    dec     rcx
    mov     rdx, rax
    shr     rdx, 28
    and     rdx, 0xF
    movzx   edx, byte ptr [rsi + rdx]
    mov     byte ptr [r15], dl
    inc     r15
    shl     rax, 4
    test    rcx, rcx
    jnz     final_offset_loop

    mov     byte ptr [r15], 0x0a
    inc     r15

    # write final offset line
    mov     rdi, 1
    lea     rsi, [rip + line_buf]
    mov     rdx, r15
    lea     rcx, [rip + line_buf]
    sub     rdx, rcx
    mov     rax, 1
    syscall

    # exit
    mov     rax, 60
    xor     rdi, rdi
    syscall

exit_error:
    mov     rax, 60
    mov     rdi, 1
    syscall
