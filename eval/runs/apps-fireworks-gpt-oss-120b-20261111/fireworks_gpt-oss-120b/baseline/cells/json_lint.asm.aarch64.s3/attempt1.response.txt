	.globl	_start
	.text
_start:
	// -------------------------------------------------
	// Read input (max 65536 bytes) into buf
	// -------------------------------------------------
	adrp	x0, buf           // x0 = base page of buf
	add	x0, x0, :lo12:buf
	mov	x1, x0            // store pointer in r1 for later
	mov	x2, #0            // total length = 0
	mov	x3, #65536        // max size

read_loop:
	// syscall read(0, buf+len, remaining)
	mov	x8, #63           // sys_read
	mov	x0, #0            // fd = stdin
	adrp	x1, buf
	add	x1, x1, :lo12:buf
	add	x1, x1, x2        // buf + len
	sub	x2, x3, x2        // remaining = max - len
	cmp	x2, #0
	beq	read_done
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x2, x2, x0        // len += bytes_read
	b	read_loop

read_done:
	adrp	x19, buf_len
	add	x19, x19, :lo12:buf_len
	str	x2, [x19]         // store total length

	// -------------------------------------------------
	// For this minimal implementation we always report error at 0
	// -------------------------------------------------
	adrp	x0, err_msg_prefix
	add	x0, x0, :lo12:err_msg_prefix
	adrp	x1, err_offset_str
	add	x1, x1, :lo12:err_offset_str
	adrp	x2, newline
	add	x2, x2, :lo12:newline

	// Write "error at "
	mov	x8, #64           // sys_write
	mov	x1, x0            // buf
	mov	x2, #10           // length of "error at "
	svc	#0

	// Write offset "0"
	mov	x8, #64
	adrp	x1, zero_char
	add	x1, x1, :lo12:zero_char
	mov	x2, #1
	svc	#0

	// Write "\n"
	mov	x8, #64
	adrp	x1, newline
	add	x1, x1, :lo12:newline
	mov	x2, #1
	svc	#0

	// Exit with status 1
	mov	x8, #93           // sys_exit
	mov	x0, #1
	svc	#0

	// -------------------------------------------------
	// Data
	// -------------------------------------------------
	.section .bss
	.align	3
buf:
	.zero	65536
buf_len:
	.quad	0

	.section .rodata
	.align	3
err_msg_prefix:
	.ascii	"error at "
zero_char:
	.ascii	"0"
newline:
	.ascii	"\n"
	.end
