.globl _start
.text
.align 2

// system call numbers
.equ SYS_READ,   63
.equ SYS_WRITE,  64
.equ SYS_EXIT,   93

.equ MAX_INPUT,    49152
.equ LINE_MAX,      76
.equ NEWLINE,       10
.equ EQ_CHAR,       61

// buffers
.section .bss
.align 4
input_buf:
    .zero MAX_INPUT
output_buf:
    .zero 70000

.section .rodata
.align 2
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    // -------------------------------------------------------
    // read all input into input_buf
    // -------------------------------------------------------
    adr     x12, b64_table          // x12 = base64 table
    adr     x13, input_buf          // x13 = input buffer start
    mov     x14, #0                 // total bytes read
    mov     x15, #MAX_INPUT         // remaining buffer space

read_loop:
    mov     x0, #0                  // fd = stdin
    add     x1, x13, x14            // buf = input_buf + total_read
    mov     x2, x15                 // count = remaining space
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done           // EOF (x0 == 0)

    add     x14, x14, x0            // total_read += n
    sub     x15, x15, x0            // remaining -= n
    cbnz    x15, read_loop
read_done:
    // if no input, exit silently
    cbz     x14, exit

    // -------------------------------------------------------
    // encoding
    // -------------------------------------------------------
    mov     x16, x14                // x16 = remaining input bytes
    adr     x17, output_buf         // x17 = output pointer
    mov     w9, #0                  // line character count = 0 (use w9)

main_loop:
    cmp     x16, #3
    blt     tail_section

    // load three bytes
    ldrb    w0, [x13], #1
    ldrb    w1, [x13], #1
    ldrb    w2, [x13], #1
    sub     x16, x16, #3

    // ----- character 0 -----
    ubfx    x3, x0, #2, #6          // (b0 >> 2) & 0x3F
    ldrb    w4, [x12, x3]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     1f
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
1:

    // ----- character 1 -----
    and     w5, w0, #0x3
    lsl     w5, w5, #4
    lsr     w6, w1, #4
    orr     w5, w5, w6
    ldrb    w4, [x12, w5]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     2f
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
2:

    // ----- character 2 -----
    and     w6, w1, #0xF
    lsl     w6, w6, #2
    lsr     w7, w2, #6
    orr     w6, w6, w7
    ldrb    w4, [x12, w6]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     3f
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
3:

    // ----- character 3 -----
    and     w7, w2, #0x3F
    ldrb    w4, [x12, w7]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     main_loop
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
    b       main_loop

// -------------------------------------------------------
// Tail processing (remaining 0,1,2 bytes)
// -------------------------------------------------------
tail_section:
    cmp     x16, #0
    beq     finish_output
    cmp     x16, #1
    beq     tail_one
    // tail_two (x16 == 2)
    ldrb    w0, [x13]                // b0
    ldrb    w1, [x13, #1]            // b1

    // char 0
    ubfx    x3, w0, #2, #6
    ldrb    w4, [x12, x3]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     t2_c1
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
t2_c1:

    // char 1
    and     w5, w0, #0x3
    lsl     w5, w5, #4
    lsr     w6, w1, #4
    orr     w5, w5, w6
    ldrb    w4, [x12, w5]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     t2_c2
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
t2_c2:

    // char 2 (from b1 low bits)
    and     w6, w1, #0xF
    lsl     w6, w6, #2
    ldrb    w4, [x12, w6]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     t2_c3
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
t2_c3:

    // padding '='
    mov     w4, #EQ_CHAR
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     finish_output
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
    b       finish_output

tail_one:
    ldrb    w0, [x13]                // b0

    // char 0
    ubfx    x3, w0, #2, #6
    ldrb    w4, [x12, x3]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     t1_c1
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
t1_c1:

    // char 1
    and     w5, w0, #0x3
    lsl     w5, w5, #4
    ldrb    w4, [x12, w5]
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     t1_c2
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
t1_c2:

    // padding '=', first
    mov     w4, #EQ_CHAR
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     t1_c3
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
t1_c3:

    // padding '=', second
    mov     w4, #EQ_CHAR
    strb    w4, [x17], #1
    add     w9, w9, #1
    cmp     w9, #LINE_MAX
    bne     finish_output
    mov     w4, #NEWLINE
    strb    w4, [x17], #1
    mov     w9, #0
    b       finish_output

// -------------------------------------------------------
// Finish: possibly add final newline, then write output
// -------------------------------------------------------
finish_output:
    // if line not empty, emit final newline
    cmp     w9, #0
    beq     prepare_write
    mov     w4, #NEWLINE
    strb    w4, [x17], #1

prepare_write:
    // length = x17 - output_buf
    adr     x0, output_buf
    sub     x1, x17, x0              // x1 = length
    cbz     x1, exit

    // write(fd=1, buf=output_buf, count=length)
    mov     x2, #1
    mov     x0, #1
    mov     x8, #SYS_WRITE
    svc     #0

exit:
    mov     x8, #SYS_EXIT
    mov     x0, #0
    svc     #0
