	.global	_start
	.text
_start:
	// x19 = buffer start
	adrp	x19, buffer
	add	x19, x19, :lo12:buffer
	// x9 = address of temporary byte storage
	adrp	x9, tmp
	add	x9, x9, :lo12:tmp

	// x20 = current write pointer (initially buffer start)
	mov	x20, x19
	// x24 = total bytes read
	mov	x24, #0

	// x5 = BUFFER_SIZE (0x10000)
	movz	x5, #0x1, lsl #16	// x5 = 65536

read_loop:
	// x0 = fd 0
	mov	x0, #0
	// x1 = buffer pointer
	mov	x1, x20
	// x2 = remaining space = BUFFER_SIZE - total
	sub	x2, x5, x24
	// syscall read
	mov	x8, #63
	svc	#0
	// result in x0
	cbnz	x0, read_continue
	// x0 == 0 => EOF
	b	read_done

read_continue:
	// advance pointers
	add	x24, x24, x0
	add	x20, x20, x0
	cmp	x24, x5
	b.lt	read_loop

read_done:
	// x23 = end pointer (buffer + total)
	mov	x23, x20

	// x22 = current processing pointer (start of buffer)
	mov	x22, x19

process_loop:
	cmp	x22, x23
	b.ge	done

	// x25 = scan pointer
	mov	x25, x22

scan_next:
	cmp	x25, x23
	b.ge	no_newline
	ldrb	w26, [x25]
	cmp	w26, #0x0A
	b.eq	found_newline
	add	x25, x25, #1
	b	scan_next

found_newline:
	// line length = x25 - x22
	sub	x27, x25, x22
	// src = x22 + len - 1
	add	x28, x22, x27
	sub	x28, x28, #1

rev_loop:
	cmp	x27, #0
	beq	rev_done
	ldrb	w0, [x28]		// byte to write
	strb	w0, [x9]		// store to tmp
	// write(fd=1, &tmp, 1)
	mov	x0, #1
	mov	x1, x9
	mov	x2, #1
	mov	x8, #64
	svc	#0
	sub	x28, x28, #1
	sub	x27, x27, #1
	b	rev_loop

rev_done:
	// write newline
	mov	w0, #0x0A
	strb	w0, [x9]
	mov	x0, #1
	mov	x1, x9
	mov	x2, #1
	mov	x8, #64
	svc	#0

	// advance past the newline we just processed
	add	x22, x25, #1
	b	process_loop

no_newline:
	// line length = x23 - x22
	sub	x27, x23, x22
	// src = x22 + len - 1
	add	x28, x22, x27
	sub	x28, x28, #1

rev_loop2:
	cmp	x27, #0
	beq	rev_done2
	ldrb	w0, [x28]
	strb	w0, [x9]
	mov	x0, #1
	mov	x1, x9
	mov	x2, #1
	mov	x8, #64
	svc	#0
	sub	x28, x28, #1
	sub	x27, x27, #1
	b	rev_loop2

rev_done2:
	// write newline
	mov	w0, #0x0A
	strb	w0, [x9]
	mov	x0, #1
	mov	x1, x9
	mov	x2, #1
	mov	x8, #64
	svc	#0

	// finish processing
	mov	x22, x23
	b	process_loop

done:
	// exit status 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.bss
	.align	3
buffer:
	.zero	65536
tmp:
	.zero	1
