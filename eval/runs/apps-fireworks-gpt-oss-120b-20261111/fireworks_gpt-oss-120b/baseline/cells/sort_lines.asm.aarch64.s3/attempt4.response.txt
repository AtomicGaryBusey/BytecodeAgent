.global _start
.text

// -------------------------------------------------
// Constants
.equ    BUFFER_SIZE, 32768
.equ    MAX_LINES, 1000

// -------------------------------------------------
// Entry point
_start:
    // x19 = base of input buffer
    adrp    x19, buf
    add     x19, x19, :lo12:buf

    // x20 = lines_off base
    adrp    x20, lines_off
    add     x20, x20, :lo12:lines_off

    // x21 = lines_len base
    adrp    x21, lines_len
    add     x21, x21, :lo12:lines_len

    // w22 = total length read so far = 0
    mov     w22, #0

read_loop:
    // sys_read(fd=0, buf+total_len, remaining)
    mov     x0, #0                     // fd
    add     x1, x19, w22, uxtw         // buf + total_len
    mov     w2, #BUFFER_SIZE
    sub     w2, w2, w22                // remaining bytes
    mov     x8, #63                    // sys_read
    svc     #0
    cbz     x0, read_done              // eof
    add     w22, w22, w0               // total_len += bytes_read
    b       read_loop

read_done:
    // -------------------------------------------------
    // Parse lines into offset/length tables
    // w23 = cursor offset = 0
    // w24 = line count = 0
    mov     w23, #0
    mov     w24, #0

parse_loop:
    cmp     w23, w22
    b.ge    parse_done                 // cursor >= total_len

    // w25 = start_offset = w23
    mov     w25, w23

scan_loop:
    cmp     w23, w22
    b.ge    no_newline                 // reached end without newline
    add     x2, x19, w23, uxtw
    ldrb    w2, [x2]
    cmp     w2, #10                    // newline?
    b.eq    found_newline

    add     w23, w23, #1
    b       scan_loop

found_newline:
    // line length = w23 - w25
    sub     w26, w23, w25
    // store offset
    lsl     w0, w24, #2                // byte offset = line_index*4
    str     w25, [x20, x0]
    // store length
    str     w26, [x21, x0]
    // increment line count
    add     w24, w24, #1

    // skip newline
    add     w23, w23, #1
    b       parse_loop

no_newline:
    // final line (no terminating newline)
    sub     w26, w23, w25              // length = cursor - start_offset
    cmp     w26, #0
    beq     parse_done                 // zero-length remainder is not a line
    // store offset
    lsl     w0, w24, #2
    str     w25, [x20, x0]
    // store length
    str     w26, [x21, x0]
    add     w24, w24, #1

parse_done:
    // w24 = number of lines
    cbz     w24, exit_program          // no lines -> exit

    // -------------------------------------------------
    // Simple bubble sort (ascending)
    // w27 = outer remaining = line_count - 1
    sub     w27, w24, #1

outer_loop:
    cbz     w27, sort_done

    // w28 = inner limit = w27
    mov     w28, w27
    // w29 = j = 0
    mov     w29, #0

inner_loop:
    // compute byte offset = j*4
    lsl     w0, w29, #2
    // addresses for line j
    add     x2, x20, x0                 // addr_off_j
    add     x3, x21, x0                 // addr_len_j

    // addresses for line j+1 (offset +4)
    add     x4, x2, #4                  // addr_off_j1
    add     x5, x3, #4                  // addr_len_j1

    // load offsets and lengths
    ldr     w6, [x2]                    // off_a
    ldr     w7, [x4]                    // off_b
    ldr     w8, [x3]                    // len_a
    ldr     w9, [x5]                    // len_b

    // call compare: result in w0 (1 if a > b)
    mov     x0, x19                     // buffer base
    mov     w1, w6
    mov     w2, w8
    mov     w3, w7
    mov     w4, w9
    bl      _cmp

    // if result == 1, swap entries
    cbz     w0, next_inner

    // recompute addresses (they may have been clobbered)
    lsl     w0, w29, #2
    add     x2, x20, x0
    add     x3, x21, x0
    add     x4, x2, #4
    add     x5, x3, #4

    // swap offsets
    str     w7, [x2]
    str     w6, [x4]
    // swap lengths
    str     w9, [x3]
    str     w8, [x5]

next_inner:
    add     w29, w29, #1
    cmp     w29, w28
    blt     inner_loop

    // decrement outer remaining
    sub     w27, w27, #1
    b       outer_loop

sort_done:
    // -------------------------------------------------
    // Output sorted lines
    // w30 = i = 0
    mov     w30, #0

output_loop:
    cmp     w30, w24
    b.ge    exit_program

    // load offset and length for line i
    lsl     w0, w30, #2
    ldr     w1, [x20, x0]               // offset
    ldr     w2, [x21, x0]               // length

    // write line
    mov     x0, #1                      // stdout
    add     x1, x19, w1, uxtw
    mov     x2, w2                      // length zero-extended
    mov     x8, #64
    svc     #0

    // write newline
    mov     x0, #1
    adrp    x1, newline
    add     x1, x1, :lo12:newline
    mov     x2, #1
    mov     x8, #64
    svc     #0

    add     w30, w30, #1
    b       output_loop

exit_program:
    mov     x0, #0
    mov     x8, #93                     // sys_exit
    svc     #0

// -------------------------------------------------
// Compare subroutine
// Inputs:
//   x0 = buffer base
//   w1 = offset a
//   w2 = len a
//   w3 = offset b
//   w4 = len b
// Output:
//   w0 = 1 if a > b, else 0
_cmp:
    // w5 = min(len_a, len_b)
    mov     w5, w2
    cmp     w2, w4
    csel    w5, w2, w4, lt

    // w6 = i = 0
    mov     w6, #0

cmp_loop:
    cmp     w6, w5
    beq     equal_prefix

    // load byte from a: buf + offset_a + i
    add     x7, x0, w1, uxtw
    add     x7, x7, w6, uxtw
    ldrb    w8, [x7]

    // load byte from b: buf + offset_b + i
    add     x9, x0, w3, uxtw
    add     x9, x9, w6, uxtw
    ldrb    w9, [x9]

    cmp     w8, w9
    blt     a_less
    bgt     a_greater

    // equal, continue
    add     w6, w6, #1
    b       cmp_loop

a_less:
    mov     w0, #0
    ret

a_greater:
    mov     w0, #1
    ret

equal_prefix:
    // lengths decide
    cmp     w2, w4
    blt     a_less
    bgt     a_greater
    mov     w0, #0
    ret

// -------------------------------------------------
.section .rodata
newline:
    .byte   0x0A

.bss
    .align 8
buf:
    .space  BUFFER_SIZE
lines_off:
    .space  MAX_LINES * 4
lines_len:
    .space  MAX_LINES * 4
