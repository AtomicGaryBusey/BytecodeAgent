	.equ INPUT_MAX, 49152
	.equ OUT_MAX, 72000
	.equ LINE_MAX, 76

	.global	_start
	.text

_start:
	// ----- set up buffers -----
	adrp	x20, input_buf
	add	x20, x20, :lo12:input_buf          // x20 = input buffer base
	adrp	x21, input_buf
	add	x21, x21, :lo12:input_buf          // x21 = src pointer (will be advanced)
	adrp	x22, output_buf
	add	x22, x22, :lo12:output_buf          // x22 = output pointer
	// ----- read all input -----
read_loop:
	// compute remaining space = INPUT_MAX - (src - base)
	sub	x0, x21, x20                      // bytes already read
	mov	x1, #INPUT_MAX
	sub	x2, x1, x0                        // remaining space
	cmp	x2, #0
	ble	read_done                         // no space left

	// sys_read(fd=0, buf=src, count=remaining)
	mov	x0, #0
	mov	x1, x21
	mov	x2, x2
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	beq	read_done                         // EOF
	add	x21, x21, x0                      // advance src pointer
	b	read_loop

read_done:
	// x20 = input base, x21 = ptr after last byte read
	sub	x9, x21, x20                      // x9 = total input length

	// ----- set up for encoding -----
	adrp	x23, input_buf
	add	x23, x23, :lo12:input_buf          // x23 = src pointer = input base
	add	x24, x23, x9                       // x24 = src_end = input base + len
	adrp	x25, b64_table
	add	x25, x25, :lo12:b64_table          // x25 = base64 table
	mov	x26, #0                            // x26 = output length
	mov	x27, #0                            // x27 = line count
	mov	w28, #LINE_MAX                     // w28 = 76
	mov	w29, #10                           // w29 = newline '\n'

	// ----- main loop processing 3-byte groups -----
main_loop:
	sub	x0, x24, x23                       // bytes left
	cmp	x0, #3
	blt	remainder

	// load three bytes
	ldrb	w0, [x23], #1                     // b0
	ldrb	w1, [x23], #1                     // b1
	ldrb	w2, [x23], #1                     // b2

	// idx0 = b0 >> 2
	ubfx	w3, w0, #2, #6

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	w4, w0, #3
	lsl	w4, w4, #4
	ubfx	w5, w1, #4, #4
	orr	w4, w4, w5                         // w4 = idx1

	// idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	and	w5, w1, #0xF
	lsl	w5, w5, #2
	ubfx	w6, w2, #6, #2
	orr	w5, w5, w6                         // w5 = idx2

	// idx3 = b2 & 0x3F
	and	w6, w2, #0x3F

	// output idx0
	add	x10, x25, w3, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output idx1
	add	x10, x25, w4, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output idx2
	add	x10, x25, w5, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output idx3
	add	x10, x25, w6, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	b	main_loop

remainder:
	sub	x0, x24, x23                       // remaining bytes (0,1,2)
	cmp	x0, #0
	beq	finish_encoding

	cmp	x0, #1
	beq	rem_one

	// remaining == 2
	ldrb	w0, [x23]                         // b0
	ldrb	w1, [x23, #1]                     // b1

	// idx0 = b0 >> 2
	ubfx	w3, w0, #2, #6

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	w4, w0, #3
	lsl	w4, w4, #4
	ubfx	w5, w1, #4, #4
	orr	w4, w4, w5                         // w4 = idx1

	// idx2 = ((b1 & 0xF) << 2)
	and	w5, w1, #0xF
	lsl	w5, w5, #2                         // w5 = idx2
	// idx3 = padding

	// output idx0
	add	x10, x25, w3, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output idx1
	add	x10, x25, w4, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output idx2
	add	x10, x25, w5, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output '=' padding
	mov	w7, #'='
	strb	w7, [x22], #1
	bl	update_line
	b	finish_encoding

rem_one:
	ldrb	w0, [x23]                         // b0

	// idx0 = b0 >> 2
	ubfx	w3, w0, #2, #6

	// idx1 = ((b0 & 3) << 4)
	and	w4, w0, #3
	lsl	w4, w4, #4                         // w4 = idx1

	// output idx0
	add	x10, x25, w3, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output idx1
	add	x10, x25, w4, uxtw #0
	ldrb	w7, [x10]
	strb	w7, [x22], #1
	bl	update_line

	// output '==' padding
	mov	w7, #'='
	strb	w7, [x22], #1
	bl	update_line
	strb	w7, [x22], #1
	bl	update_line
	b	finish_encoding

// -------------------------------------------------------
update_line:
	// increment output length
	add	x26, x26, #1
	// increment line count
	add	x27, x27, #1
	cmp	x27, #LINE_MAX
	b.lt	1f
	// line full, emit newline
	mov	w7, #10
	strb	w7, [x22], #1
	add	x26, x26, #1
	mov	x27, #0
1:
	ret

finish_encoding:
	// if there was any output and line count != 0, emit final newline
	cmp	x26, #0
	beq	write_output
	cmp	x27, #0
	beq	write_output
	// emit newline
	mov	w7, #10
	strb	w7, [x22], #1
	add	x26, x26, #1
	b	write_output

write_output:
	// write all output in one syscall, handling partial writes
	mov	x0, #1                // fd = stdout
	mov	x1, x22               // x22 currently points after last byte; need start address
	sub	x1, x1, x26           // x1 = output_buf start
	mov	x2, x26               // length
write_loop:
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	beq	exit_success
	subs	x2, x2, x0
	add	x1, x1, x0
	bne	write_loop

exit_success:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// -----------------------------------------------------------------
	.section .rodata
b64_table:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

	.bss
	.align 4
input_buf:
	.space INPUT_MAX
output_buf:
	.space OUT_MAX
