	.global	_start
	.text
_start:
	// ---------- read all input into input_buf ----------
	adrp	x20, input_buf          // x20 = base page of input buffer
	add	x20, x20, :lo12:input_buf
	mov	x21, x20                // x21 = current write pointer (also start)
	adrp	x22, input_buf_end
	add	x22, x22, :lo12:input_buf_end   // x22 = end of buffer (input_buf + INPUT_MAX)

read_loop:
	cmp	x21, x22
	b.ge	read_done               // buffer full, stop reading
	// x0 = fd = 0
	mov	x0, #0
	mov	x1, x21                 // destination
	sub	x2, x22, x21            // remaining space
	mov	x8, #63                 // sys_read
	svc	#0
	cmp	x0, #0
	beq	read_done                // EOF
	add	x21, x21, x0            // advance write pointer
	b	read_loop

read_done:
	// x9 = total input length
	sub	x9, x21, input_buf
