.globl _start
.text
_start:
    xor rbx, rbx                 # total_len = 0
    lea r8, [rip + buffer]      # ptr = buffer

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    mov rsi, r8
    mov rdx, READ_CHUNK
    syscall
    test rax, rax
    je dump
    add rbx, rax                # total_len += bytes read
    add r8, rax                 # ptr += bytes read
    jmp read_loop

dump:
    lea rsi, [rip + buffer]     # buffer base
    xor rdx, rdx                # offset = 0

row_loop:
    cmp rdx, rbx
    jae final_len_line

    mov rcx, rbx
    sub rcx, rdx                # remaining bytes
    mov r8, 16
    cmp rcx, r8
    cmova rcx, r8               # n = min(16, remaining)

    lea rdi, [rip + linebuf]    # line buffer pointer
    mov rax, rdx
    call hex8                   # write offset (8 hex chars)

    mov byte [rdi], ' '
