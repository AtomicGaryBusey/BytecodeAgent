.globl _start
.text
_start:
    # registers:
    # r12 = buffer base
    # rbx = total length
    # rdx = current offset for dumping
    # r9  = temporary (n bytes in row)
    # r8  = pointer to line buffer
    # r10 = remaining bytes for read
    # r13 = pointer for hex conversion (temp)

    # buffer pointer
    lea r12, [rip + buffer]

    xor rbx, rbx                     # total_len = 0

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    lea rsi, [r12 + rbx]             # buf = buffer + total_len
    mov rdx, BUFFER_SIZE
    sub rdx, rbx                     # count = BUFFER_SIZE - total_len
    syscall
    test rax, rax
    je dump
    add rbx, rax                     # total_len += bytes read
    cmp rbx, BUFFER_SIZE
    jae dump
    jmp read_loop

dump:
    xor rdx, rdx                     # offset = 0

row_loop:
    cmp rdx, rbx
    jae final_len_line

    # n = min(16, total_len - offset)
    mov r9, rbx
    sub r9, rdx
    mov rcx, 16
    cmp r9, rcx
    cmova r9, rcx                    # if remaining < 16, n = remaining

    # line buffer pointer
    lea r8, [rip + linebuf]          # r8 = dst

    # ----- write offset (8 hex digits) -----
    mov r13, rdx                     # r13 = offset copy
    mov rcx, 8
hex8_loop:
    mov rax, r13
    shr rax, 28                      # high nibble
    and al, 0xF
    mov bl, [rip + hex_digits + rax]
