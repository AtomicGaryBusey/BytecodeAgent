.globl _start
.text
_start:
    # r12 = buffer base, rbx = total length, rdx = current offset
    lea     r12, [rip + buffer]          # buffer base
    xor     rbx, rbx                     # total_len = 0

read_loop:
    mov     rax, 0                       # sys_read
    xor     rdi, rdi                     # fd = 0 (stdin)
    lea     rsi, [r12 + rbx]             # buf = buffer + total_len
    mov     rdx, BUFFER_SIZE
    sub     rdx, rbx                     # count = remaining space
    syscall
    test    rax, rax
    jz      dump_start
    add     rbx, rax                     # total_len += bytes read
    cmp     rbx, BUFFER_SIZE
    jb      read_loop
    # buffer full, continue to dump

dump_start:
    xor     rdx, rdx                     # offset = 0

row_loop:
    cmp     rdx, rbx
    jae     final_len_line               # if offset >= total_len, finish

    # rcx = bytes in this row = min(16, total_len - offset)
    mov     rcx, rbx
    sub     rcx, rdx
    mov     r8, 16
    cmp     rcx, r8
    cmova   rcx, r8

    # r8 = pointer to line buffer
    lea     r8, [rip + linebuf]

    # ---- write offset (8 hex digits) ----
    mov     r9, rdx                      # copy offset
    mov     r10, 7                       # nibble index
offset_nibble_loop:
    mov     rax, r9
    mov     rcx, r10
    shl     rcx, 2                       # rcx = nibble*4
    shr     rax, cl
    and     al, 0xF
    movzx   rax, byte ptr [rip + hex_digits + rax]
