.globl _start
.text
_start:
    # r12 = buffer base, rbx = total length, rdx = current offset
    lea r12, [rip + buffer]          # buffer base
    xor rbx, rbx                     # total_len = 0

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    lea rsi, [r12 + rbx]             # buf = buffer + total_len
    mov rdx, BUFFER_SIZE
    sub rdx, rbx                     # count = remaining space
    syscall
    test rax, rax
    jz dump_start
    add rbx, rax                     # total_len += bytes read
    cmp rbx, BUFFER_SIZE
    jb read_loop
    # buffer full, continue to dump
dump_start:
    xor rdx, rdx                     # offset = 0

row_loop:
    cmp rdx, rbx
    jae final_len_line

    # r9 = remaining bytes, rcx = n (bytes this row)
    mov r9, rbx
    sub r9, rdx
    mov rcx, 16
    cmp r9, rcx
    cmovb rcx, r9                    # rcx = n (min(16, remaining))

    # r8 = line buffer pointer
    lea r8, [rip + linebuf]

    # ----- offset (8 hex digits) -----
    mov r10, rdx                     # copy of offset
    mov r13, 7                       # nibble index (7..0)
    lea r15, [rip + hex_digits]     # hex table
offset_nibble_loop:
    mov rax, r10
    mov r14, r13
    shl r14, 2                       # r14 = nibble_index*4
    shr rax, cl                      # shift right by nibble bits
    and al, 0xF
    mov bl, [r15 + rax]              # hex char
    mov [r8], bl
    inc r8
    dec r13
    jns offset_nibble_loop

    # two spaces after offset
    mov byte ptr [r8], ' '
    inc r8
    mov byte ptr [r8], ' '
    inc r8

    # ----- hex byte dump (16 slots) -----
    xor r14, r14                     # i = 0
hex_bytes_loop:
    cmp r14, 16
    je after_hex_bytes
    cmp r14, rcx
    jae hex_byte_absent

    # present byte
    mov al, [r12 + rdx + r14]        # load byte
