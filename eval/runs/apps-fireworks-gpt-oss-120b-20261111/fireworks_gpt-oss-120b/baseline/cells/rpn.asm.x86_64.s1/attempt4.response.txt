.globl _start

.section .bss
.align 16
input_buf:
    .zero 65536
stack:
    .zero 8192          # 1024 * 8 bytes
outbuf:
    .zero 32

.section .rodata
err_msg:
    .ascii "error\n"
err_len = . - err_msg
nl:
    .byte 0x0A

.section .text
_start:
    # ---------- read whole input ----------
    lea r8, [rip + input_buf]          # r8 = buffer base
    xor r9, r9                         # r9 = total bytes read
read_loop:
    mov rax, 0                         # sys_read
    mov rdi, 0                         # fd = stdin
    lea rsi, [r8 + r9]                 # dst = base + already read
    mov rdx, 65536
    sub rdx, r9                        # remaining space
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_loop
read_done:
    # rsi = current line start, rdx = end of input
    lea rsi, [r8]                      # rsi points to start of buffer
    lea rdx, [r8 + r9]                 # rdx points just past last byte

line_loop:
    cmp rsi, rdx
    jae exit_program                   # no more data

    # find end of line (newline or end)
    mov rbx, rsi                       # rbx scans for newline
line_find_nl:
    cmp rbx, rdx
    jae line_end_found
    mov al, byte [rbx]
    cmp al, 0x0A
    je line_end_found
    inc rbx
    jmp line_find_nl
line_end_found:
    # rbx now points to newline or end (line end exclusive)

    # ---------- evaluate line ----------
    lea r14, [rip + stack]             # r14 = stack base
    xor r8, r8                         # r8 = stack count
    xor r9b, r9b                       # r9b = error flag

token_loop:
    cmp rsi, rbx
    jae token_done                     # reached line end

    # skip whitespace (space or tab)
skip_ws:
    cmp rsi, rbx
    jae token_done
    mov al, byte [rsi]
    cmp al, 0x20
    je inc_ws
    cmp al, 0x09
    je inc_ws
    jmp token_start
inc_ws:
    inc rsi
    jmp skip_ws

token_start:
    mov r12, rsi                       # r12 = token start ptr
    mov r13, rsi                       # r13 = token scan ptr
scan_token:
    cmp r13, rbx
    jae token_end
    mov al, byte [r13]
    cmp al, 0x20
    je token_end
    cmp al, 0x09
    je token_end
    inc r13
    jmp scan_token
token_end:
    mov rcx, r13
    sub rcx, r12                       # rcx = token length
    # after processing, rsi will be set to r13 (next token)

    # ---------- token classification ----------
    cmp rcx, 1
    jne not_operator
    mov al, byte [r12]                 # single char token
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp not_operator

not_operator:
    # ---------- integer literal ----------
    xor rax, rax                       # accumulator
    xor rbx, rbx                       # sign flag 0=positive,1=negative
    mov rdi, r12                       # pointer into token
    mov rdx, rcx                       # remaining length
    mov al, byte [rdi]
    cmp al, '-'
    jne parse_digits
    mov bl, 1
    inc rdi
    dec rdx
    cmp rdx, 0
    je token_error
parse_digits:
    test rdx, rdx
    je token_error
digit_loop:
    mov al, byte [rdi]
    cmp al, '0'
    jb token_error
    cmp al, '9'
    ja token_error
    sub al, '0'
    movzx rdx, al
    imul rax, rax, 10
    add rax, rdx
    inc rdi
    dec rcx
    jne digit_loop                     # rcx was token length, now zero when done
    test bl, bl
    jz push_number
    neg rax
push_number:
    mov [r14 + r8*8], rax
    inc r8
    jmp token_continue

op_plus:
    cmp r8, 2
    jl token_error
    dec r8
    mov rbx, [r14 + r8*8]              # b
    dec r8
    mov rax, [r14 + r8*8]              # a
    add rax, rbx
    jmp push_res
op_minus:
    cmp r8, 2
    jl token_error
    dec r8
    mov rbx, [r14 + r8*8]
    dec r8
    mov rax, [r14 + r8*8]
    sub rax, rbx
    jmp push_res
op_mul:
    cmp r8, 2
    jl token_error
    dec r8
    mov rbx, [r14 + r8*8]
    dec r8
    mov rax, [r14 + r8*8]
    mul rbx                # unsigned multiply, result low in rax
    jmp push_res
op_div:
    cmp r8, 2
    jl token_error
    dec r8
    mov rbx, [r14 + r8*8]      # b
    dec r8
    mov rax, [r14 + r8*8]      # a
    test rbx, rbx
    je token_error
    mov rcx, rax
    mov rdx, 0x8000000000000000
    cmp rcx, rdx
    jne do_div
    cmp rbx, -1
    je token_error
do_div:
    cqo
    idiv rbx
    jmp push_res
op_mod:
    cmp r8, 2
    jl token_error
    dec r8
    mov rbx, [r14 + r8*8]      # b
    dec r8
    mov rax, [r14 + r8*8]      # a
    test rbx, rbx
    je token_error
    mov rcx, rax
    mov rdx, 0x8000000000000000
    cmp rcx, rdx
    jne do_mod
    cmp rbx, -1
    je token_error
do_mod:
    cqo
    idiv rbx
    mov rax, rdx               # remainder
    jmp push_res
push_res:
    mov [r14 + r8*8], rax
    inc r8
    jmp token_continue

token_error:
    mov r9b, 1
    jmp token_done

token_continue:
    mov rsi, r13                # advance to after token
    jmp token_loop

token_done:
    # ---------- after token processing ----------
    test r9b, r9b
    jnz output_error
    cmp r8, 1
    jne output_error

    # result is the only element on stack (at index 0)
    mov rax, [r14]

    # ---------- convert result to decimal ----------
    lea rdi, [rip + outbuf]
    add rdi, 32                 # point past buffer end
    xor rcx, rcx                # digit count
    xor rdx, rdx
    xor r9b, r9b                # sign flag
    test rax, rax
    jge conv_pos
    mov r9b, 1
    neg rax
conv_pos:
    cmp rax, 0
    jne conv_loop
    dec rdi
    mov byte [rdi], '0'
    inc rcx
    jmp conv_done
conv_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx                     # rax = rax/10, rdx = remainder
    dec rdi
    add dl, '0'
    mov byte [rdi], dl
    inc rcx
    cmp rax, 0
    jne conv_loop
conv_done:

    je no_sign
    dec rdi
    mov byte [rdi], '-'
    inc rcx

    mov rax, rbx               # rbx = line end (newline or end)
    cmp rax, rdx
    je set_cur_end
    inc rax                    # skip newline
set_cur_end:
    mov rsi, rax
    jmp line_loop
