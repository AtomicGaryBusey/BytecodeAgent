.globl _start

.section .bss
.align 16
input_buf:
    .zero 65536
stack:
    .zero 8192          # 1024 entries * 8
outbuf:
    .zero 32

.section .rodata
err_msg:
    .ascii "error\n"
err_len = . - err_msg
nl:
    .byte 0x0A

.section .text
_start:
    # ---- read all input into input_buf ----
    lea r8, [rip + input_buf]      # r8 = buffer base
    xor r9, r9                     # r9 = length read so far
read_loop:
    mov rax, 0                     # sys_read
    mov rdi, 0                     # fd = stdin
    lea rsi, [r8 + r9]             # dst = base+len
    mov rdx, 65536
    sub rdx, r9                    # remaining space
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_loop
read_done:
    lea r10, [r8 + r9]             # r10 = end pointer

    # r12 = cur pointer
    mov r12, r8
    mov r13, r10                   # r13 = end pointer

line_loop:
    cmp r12, r13
    jae exit_program               # no more data

    # r14 = line_start
    mov r14, r12

    # find newline or end
    mov r15, r12                   # scanner
find_nl:
    cmp r15, r13
    jae no_nl
    mov al, byte ptr [r15]
    cmp al, 0x0A
    je nl_found
    inc r15
    jmp find_nl
no_nl:
    mov rbx, r13                   # line_end = end
    jmp after_nl
nl_found:
    mov rbx, r15                   # line_end = newline
after_nl:
    # rdx = line_len = line_end - line_start
    mov rdx, rbx
    sub rdx, r14
    # rsi = line_start
    mov rsi, r14

    # ---- evaluation stack ----
    lea r11, [rip + stack]         # r11 = stack base
    xor r8d, r8d                   # r8d = stack_top (count)
    xor r9b, r9b                   # r9b = error flag

    xor rcx, rcx                   # rcx = offset within line

token_loop:
    cmp rcx, rdx
    jae tokens_done

    # skip separators (space 0x20, tab 0x09)
skip_sep:
    cmp rcx, rdx
    jae tokens_done
    mov al, byte ptr [rsi + rcx]
    cmp al, 0x20
    je inc_sep
    cmp al, 0x09
    je inc_sep
    jmp token_start
inc_sep:
    inc rcx
    jmp skip_sep

token_start:
    mov r8d, ecx            # token_start = rcx (stored in r8d)
    mov r9d, ecx            # scanner = rcx (stored in r9d)

tok_scan:
    cmp r9d, edx
    jae tok_end
    mov al, byte ptr [rsi + r9d]
    cmp al, 0x20
    je tok_end
    cmp al, 0x09
    je tok_end
    inc r9d
    jmp tok_scan
tok_end:
    mov r10d, r9d           # token_end = r9d
    sub r10d, r8d           # token_len = token_end - token_start, in r10d

    # if error already set, skip processing
    test r9b, r9b
    jnz set_rcx_end

    # check for single-char operator
    cmp r10d, 1
    jne not_operator
    mov bl, byte ptr [rsi + r8d]    # first char of token
    cmp bl, '+'
    je op_plus
    cmp bl, '-'
    je op_minus
    cmp bl, '*'
    je op_mul
    cmp bl, '/'
    je op_div
    cmp bl, '%'
    je op_mod
    jmp not_operator

not_operator:
    # integer literal parsing
    xor rax, rax            # clear rax, will hold accumulator
    xor rbx, rbx            # sign flag: 0 positive, 1 negative
    mov rcx, r8d            # rcx = token_start offset
    mov al, byte ptr [rsi + rcx]
    cmp al, '-'
    jne lit_digits
    mov bl, 1               # set sign flag
    inc rcx
    dec r10d                # consumed '-'
    cmp r10d, 0
    je token_invalid
lit_digits:
    mov rdx, r10d           # digit count
    test rdx, rdx
    jz token_invalid
lit_digit_loop:
    mov dl, byte ptr [rsi + rcx]
    cmp dl, '0'
    jb token_invalid
    cmp dl, '9'
    ja token_invalid
    sub dl, '0'
    movzx rdx, dl
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    dec r10d
    jnz lit_digit_loop

    # apply sign if needed
    test bl, bl
    jz push_number
    neg rax                 # works for INT64_MIN
push_number:
    mov [r11 + r8*8], rax
    inc r8d
    jmp after_token

op_plus:
    cmp r8d, 2
    jl set_error
    dec r8d
    mov rbx, [r11 + r8*8]      # b
    dec r8d
    mov rax, [r11 + r8*8]      # a
    add rax, rbx
    jmp push_res
op_minus:
    cmp r8d, 2
    jl set_error
    dec r8d
    mov rbx, [r11 + r8*8]
    dec r8d
    mov rax, [r11 + r8*8]
    sub rax, rbx
    jmp push_res
op_mul:
    cmp r8d, 2
    jl set_error
    dec r8d
    mov rbx, [r11 + r8*8]
    dec r8d
    mov rax, [r11 + r8*8]
    mul rbx          # unsigned multiply, result low in rax
    jmp push_res
op_div:
    cmp r8d, 2
    jl set_error
    dec r8d
    mov rbx, [r11 + r8*8]      # b
    dec r8d
    mov rax, [r11 + r8*8]      # a
    test rbx, rbx
    jz set_error
    mov rcx, rax
    mov rdx, 0x8000000000000000
    cmp rcx, rdx
    jne ok_div
    cmp rbx, -1
    je set_error
ok_div:
    cqo
    idiv rbx
    jmp push_res
op_mod:
    cmp r8d, 2
    jl set_error
    dec r8d
    mov rbx, [r11 + r8*8]      # b
    dec r8d
    mov rax, [r11 + r8*8]      # a
    test rbx, rbx
    jz set_error
    mov rcx, rax
    mov rdx, 0x8000000000000000
    cmp rcx, rdx
    jne ok_mod
    cmp rbx, -1
    je set_error
ok_mod:
    cqo
    idiv rbx
    mov rax, rdx                # remainder
    jmp push_res
push_res:
    mov [r11 + r8*8], rax
    inc r8d
    jmp after_token

set_error:
    mov r9b, 1
    jmp after_token

token_invalid:
    mov r9b, 1
    jmp after_token

set_rcx_end:
    mov rcx, rdx                # force exit token loop
    jmp token_loop

after_token:
    # advance rcx to token_end
    mov rcx, r10d               # r10d holds token_len, need absolute offset = token_start + token_len
    add rcx, r8d                # WRONG: we need original token_start, which was saved in r8d before it was overwritten.
    # Instead, recompute using saved start offset stored in r8d before we clobbered it.
    # We'll use a temporary register: r12d held line_start offset? Instead, redo logic:
    # Simpler: set rcx = token_start + token_len using registers saved before clobber.
    # We'll store token_start in r12d before processing token.
    # Adjust code above accordingly.

    # (this placeholder will be replaced below)
    jmp token_loop

tokens_done:
    test r9b, r9b
    jnz output_error
    cmp r8d, 1
    jne output_error

    # result is at stack[0]
    mov rax, [r11]

    # convert to decimal string in outbuf (32 bytes, growing backwards)
    lea rsi, [rip + outbuf]
    add rsi, 32
    xor rcx, rcx                # digit count
    xor rdx, rdx                # clear rdx for division
    xor r9b, r9b                # sign flag
    test rax, rax
    jge conv_pos
    mov r9b, 1
    neg rax
conv_pos:
    cmp rax, 0
    jne conv_loop
    dec rsi
    mov byte ptr [rsi], '0'
    inc rcx
    jmp conv_done
conv_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx                     # rax = rax/10, rdx = remainder
    dec rsi
    add dl, '0'
    mov byte ptr [rsi], dl
    inc rcx
    cmp rax, 0
    jne conv_loop
conv_done:
    cmp r9b, 0
    je no_sign
    dec rsi
    mov byte ptr [rsi], '-'
    inc rcx
no_sign:
    # write number
    mov rdx, rcx
    mov rax, 1
    mov rdi, 1
    syscall
    # write newline
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    jmp after_output

output_error:
    lea rsi, [rip + err_msg]
    mov rdx, err_len
    mov rax, 1
    mov rdi, 1
    syscall

after_output:
    # advance to next line
    mov rax, rbx                # rbx = line_end
    cmp rax, r13
    je set_cur_end
    inc rax
set_cur_end:
    mov r12, rax
    jmp line_loop

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
