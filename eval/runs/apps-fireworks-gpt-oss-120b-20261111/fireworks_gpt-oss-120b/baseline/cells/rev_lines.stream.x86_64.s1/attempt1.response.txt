.globl _start

.section .bss
.align 8
buffer:
    .zero 65536
outbuf:
    .zero 65536

.section .rodata
newline:
    .byte 0x0A

.section .text
_start:
    # read all input into buffer
    lea r8, [rip + buffer]        # r8 = base address of buffer
    xor rbx, rbx                  # total bytes read = 0

read_loop:
    mov rax, 0                    # sys_read
    mov rdi, 0                    # fd = stdin
    mov rsi, r8
    add rsi, rbx                  # ptr = buffer + total
    mov rdx, 65536
    sub rdx, rbx                  # remaining space
    test rdx, rdx
    jz read_done
    syscall
    cmp rax, 0
    je read_done
    add rbx, rax                  # total += bytes read
    jmp read_loop

read_done:
    # rbx = total length
    mov rsi, r8                   # rsi = current pointer in buffer
    mov rcx, rbx                  # rcx = remaining bytes

process_loop:
    test rcx, rcx
    jz exit_program

    xor rdx, rdx                  # rdx = line_len = 0

scan_loop:
    cmp rdx, rcx
    je final_line                 # no newline found
    mov al, byte ptr [rsi + rdx]
    cmp al, 0x0A
    je have_newline
    inc rdx
    jmp scan_loop

have_newline:
    mov r8, rdx                   # r8 = line_len

    test r8, r8
    jz write_nl_only

    xor r9, r9                    # r9 = i = 0
copy_rev:
    mov al, byte ptr [rsi + r8 - 1 - r9]
