.global _start

.section .bss
.align 8
buffer:
    .zero 65536
outbyte:
    .zero 1

.section .rodata
newline:
    .byte 0x0A

.text
_start:
    # ----- read all input into buffer -----
    xor rbx, rbx                # total bytes read = 0

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buffer]
    add rsi, rbx                # buf = buffer + total
    mov rdx, 65536
    sub rdx, rbx                # remaining space
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax               # rax = bytes read
    jz read_done
    add rbx, rax                # total += bytes read
    jmp read_loop

read_done:
    # rbx = total length
    test rbx, rbx
    jz exit_program

    lea rsi, [rip + buffer]    # rsi = current pointer
    mov rcx, rbx                # rcx = bytes remaining

line_loop:
    test rcx, rcx
    jz exit_program

    xor rdx, rdx                # rdx = line_len = 0

scan_loop:
    cmp rdx, rcx
    je final_line_no_nl         # no newline found
    mov al, [rsi + rdx]
    cmp al, 0x0A
    je have_newline
    inc rdx
    jmp scan_loop

have_newline:
    mov r8, rdx                 # r8 = line_len

    # write reversed line
    call write_rev_line

    # write newline
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = stdout
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    # advance past line and newline
    add rsi, r8
    inc rsi                     # skip '\n'
    sub rcx, r8
    dec rcx                     # remove newline from remaining count
    jmp line_loop

final_line_no_nl:
    # rcx is the length of the final line (no terminating newline)
    mov r8, rcx                 # r8 = line_len
    test r8, r8
    jz exit_program             # no remaining bytes

    call write_rev_line

    # write newline for the final line
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    jmp exit_program

# -------------------------------------------------
# write_rev_line:
#   Input: rsi = start of line, r8 = length
#   Writes the line bytes in reverse order, one byte at a time.
#   Preserves rsi and r8 for caller.
# -------------------------------------------------
write_rev_line:
    push rsi
    push r8
    xor r9, r9                  # i = 0
rev_loop:
    cmp r9, r8
    je rev_done
    mov rax, r8
    dec rax                     # rax = length-1
    sub rax, r9                 # rax = length-1-i
    lea rdx, [rsi + rax]        # address of byte to output
    mov al, [rdx]               # load byte

    lea rdx, [rip + outbyte]
    mov [rdx], al               # store byte to temporary buffer

    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = stdout
    mov rsi, rdx                # buffer = outbyte
    mov rdx, 1                  # count = 1
    syscall

    inc r9
    jmp rev_loop
rev_done:
    pop r8
    pop rsi
    ret

exit_program:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status = 0
    syscall
