.global _start

.section .bss
.align 8
buffer:
    .zero 65536
outbyte:
    .zero 1

.section .rodata
.align 1
newline:
    .byte 0x0A

.text
_start:
    xor     rbx, rbx                # rbx = total bytes read = 0

read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + buffer]
    add     rsi, rbx                # buffer + total
    mov     rdx, 65536
    sub     rdx, rbx                # remaining space
    test    rdx, rdx
    jz      read_done               # buffer full
    syscall
    test    rax, rax                # rax = bytes read
    jz      read_done               # EOF
    add     rbx, rax                # total += bytes read
    jmp     read_loop

read_done:
    lea     rsi, [rip + buffer]     # rsi = current pointer
    mov     rbx, rbx                # rbx already holds remaining length

process_loop:
    test    rbx, rbx
    jz      exit_program

    xor     rdx, rdx                # rdx = line_len = 0

find_nl:
    cmp     rdx, rbx
    jae     final_no_nl             # no newline in remaining data
    mov     al, [rsi + rdx]
    cmp     al, 0x0A
    je      have_nl
    inc     rdx
    jmp     find_nl

have_nl:
    # rdx = length of line before newline
    mov     r9, rdx                 # r9 = line_len (save for later)
    mov     r8, rdx                 # r8 = rev_idx

rev_out_loop:
    test    r8, r8
    jz      rev_done
    dec     r8
    mov     al, [rsi + r8]
    mov     [rip + outbyte], al
    mov     rax, 1                  # sys_write
    mov     rdi, 1                  # stdout
    lea     rsi, [rip + outbyte]
    mov     rdx, 1
    syscall
    jmp     rev_out_loop

rev_done:
    # write newline
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + newline]
    mov     rdx, 1
    syscall

    # advance past the line and its newline
    add     rsi, r9                 # skip line bytes
    inc     rsi                     # skip the '\n'
    sub     rbx, r9
    dec     rbx                     # remove newline from remaining count
    jmp     process_loop

final_no_nl:
    # rbx = length of final line (no terminating newline)
    test    rbx, rbx
    jz      exit_program            # no remaining bytes

    mov     r8, rbx                 # rev_idx = remaining length

final_rev_loop:
    test    r8, r8
    jz      final_rev_done
    dec     r8
    mov     al, [rsi + r8]
    mov     [rip + outbyte], al
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + outbyte]
    mov     rdx, 1
    syscall
    jmp     final_rev_loop

final_rev_done:
    # write newline for the final line
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + newline]
    mov     rdx, 1
    syscall

    jmp     exit_program

exit_program:
    mov     rax, 60                 # sys_exit
    xor     rdi, rdi                # status = 0
    syscall
