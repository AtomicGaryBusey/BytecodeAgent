.global _start

.section .bss
.align 8
buffer:
    .zero 65536
outbyte:
    .zero 1

.section .rodata
newline:
    .byte 0x0A

.text
_start:
    xor rbx, rbx                # total bytes read = 0

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buffer]
    add rsi, rbx                # buffer + total
    mov rdx, 65536
    sub rdx, rbx                # remaining space
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax               # rax = bytes read
    jz read_done
    add rbx, rax                # total += bytes read
    jmp read_loop

read_done:
    lea rsi, [rip + buffer]     # rsi = start of data
    mov rcx, rbx                # rcx = remaining bytes

process_loop:
    test rcx, rcx
    jz exit_program

    xor rdx, rdx                # rdx = line_len = 0

find_nl:
    cmp rdx, rcx
    je final_no_nl               # no newline found
    mov al, [rsi + rdx]
    cmp al, 0x0A
    je have_nl
    inc rdx
    jmp find_nl

have_nl:
    # rdx = length of line (bytes before newline)
    mov r10, rdx                # r10 = rev_idx = line_len
rev_out_loop:
    test r10, r10
    jz rev_done
    dec r10
    mov al, [rsi + r10]
    lea r9, [rip + outbyte]
    mov [r9], al
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    mov rsi, r9
    mov rdx, 1
    syscall
    jmp rev_out_loop
rev_done:
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    # advance past line and newline
    add rsi, rdx                # rdx still holds line_len
    inc rsi                     # skip '\n'
    sub rcx, rdx
    dec rcx                     # remove newline from remaining count
    jmp process_loop

final_no_nl:
    # rcx is the length of the final line (no terminating newline)
    test rcx, rcx
    jz exit_program             # no remaining bytes

    mov rdx, rcx                # line_len = rcx
    mov r10, rdx                # rev_idx
final_rev_loop:
    test r10, r10
    jz final_rev_done
    dec r10
    mov al, [rsi + r10]
    lea r9, [rip + outbyte]
    mov [r9], al
    mov rax, 1
    mov rdi, 1
    mov rsi, r9
    mov rdx, 1
    syscall
    jmp final_rev_loop
final_rev_done:
    # write newline for the final line
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    jmp exit_program

exit_program:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status = 0
    syscall
