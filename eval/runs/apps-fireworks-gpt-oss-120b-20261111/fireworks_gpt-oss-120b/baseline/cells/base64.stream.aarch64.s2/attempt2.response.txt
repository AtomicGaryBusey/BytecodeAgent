.global _start
.text
.align 2

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_MAX, 49152
.equ OUTPUT_MAX, 70000
.equ LINE_LIMIT, 76

_start:
    // read all input into input_buf
    adr x19, input_buf          // x19 = base of input buffer
    mov x20, #0                 // total bytes read

read_loop:
    cmp x20, #INPUT_MAX
    b.ge read_done

    mov x2, #INPUT_MAX
    sub x2, x2, x20             // space left

    mov x0, #0                  // fd = stdin
    add x1, x19, x20            // buffer = input_buf + total
    mov x8, #SYS_READ
    svc #0
    // x0 = bytes read
    cmp x0, #0
    beq read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_success       // no input -> exit 0

    // set up pointers for encoding
    adr x19, input_buf          // x19 = input pointer
    add x20, x19, x20           // x20 = input end pointer
    adr x21, output_buf         // x21 = output pointer
    mov x22, #0                 // line length counter
    adr x23, b64_table          // base64 alphabet

encode_loop:
    cmp x19, x20
    b.eq encode_finish

    sub x24, x20, x19           // remaining bytes
    cmp x24, #3
    b.ge encode_three
    cmp x24, #2
    beq encode_two
    b encode_one

encode_three:
    ldrb w4, [x19], #1          // b0
    ldrb w5, [x19], #1          // b1
    ldrb w6, [x19], #1          // b2

    // s0 = b0 >> 2
    mov w7, w4, lsr #2
    // s1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and w8, w4, #0x3
    lsl w8, w8, #4
    mov w9, w5, lsr #4
    orr w8, w8, w9
    // s2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and w9, w5, #0xF
    lsl w9, w9, #2
    mov w10, w6, lsr #6
    orr w9, w9, w10
    // s3 = b2 & 0x3F
    and w10, w6, #0x3F

    // output chars
    add x25, x23, w7, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    add x25, x23, w8, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    add x25, x23, w9, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    add x25, x23, w10, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    b encode_loop

encode_two:
    ldrb w4, [x19], #1          // b0
    ldrb w5, [x19], #1          // b1

    // s0 = b0 >> 2
    mov w7, w4, lsr #2
    // s1 = ((b0 & 0x3) << 4) | (b1 >> 4)
    and w8, w4, #0x3
    lsl w8, w8, #4
    mov w9, w5, lsr #4
    orr w8, w8, w9
    // s2 = ((b1 & 0xF) << 2)
    and w9, w5, #0xF
    lsl w9, w9, #2

    add x25, x23, w7, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    add x25, x23, w8, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    add x25, x23, w9, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    mov w11, #'='
    strb w11, [x21], #1
    bl add_line_check

    b encode_loop

encode_one:
    ldrb w4, [x19], #1          // b0

    // s0 = b0 >> 2
    mov w7, w4, lsr #2
    // s1 = ((b0 & 0x3) << 4)
    and w8, w4, #0x3
    lsl w8, w8, #4

    add x25, x23, w7, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    add x25, x23, w8, uxtw
    ldrb w11, [x25]
    strb w11, [x21], #1
    bl add_line_check

    mov w11, #'='
    strb w11, [x21], #1
    bl add_line_check
    strb w11, [x21], #1
    bl add_line_check

    b encode_loop

add_line_check:
    add w22, w22, #1
    cmp w22, #LINE_LIMIT
    b.lt .Lno_newline
    mov w11, #0x0A
    strb w11, [x21], #1
    mov w22, #0
.Lno_newline:
    ret

encode_finish:
    adr x24, output_buf
    cmp x21, x24
    beq exit_success               // no output (should not happen)

    cmp w22, #0
    beq write_output               // already at line start
    mov w11, #0x0A
    strb w11, [x21], #1

write_output:
    adr x24, output_buf
    sub x25, x21, x24              // length
    mov x0, #1                     // stdout
    mov x1, x24
    mov x2, x25
    mov x8, #SYS_WRITE
write_loop:
    cmp x2, #0
    beq exit_success
    svc #0
    cmp x0, #0
    beq exit_success
    sub x2, x2, x0
    add x1, x1, x0
    b write_loop

exit_success:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.section .rodata
.align 3
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
input_buf:
    .space INPUT_MAX
output_buf:
    .space OUTPUT_MAX
