	.global _start
	.text
	.align 2

// system call numbers
	.equ SYS_READ, 63
	.equ SYS_WRITE, 64
	.equ SYS_EXIT, 93

// constants
	.equ INPUT_MAX, 49152
	.equ OUTPUT_MAX, 70000
	.equ LINE_LIMIT, 76

_start:
	// ----- read all input into input_buf -----
	adr	x1, input_buf               // x1 = base of input buffer
	mov	x9, #0                       // total bytes read = 0

read_loop:
	cmp	x9, #INPUT_MAX
	b.ge	read_done                  // should not happen (input limited)

	// compute address = input_buf + total_read
	add	x2, x1, x9                  // x2 = dest address
	// remaining space
	mov	x3, #INPUT_MAX
	sub	x3, x3, x9                  // x3 = count
	mov	x0, #0                       // fd = stdin
	mov	x1, x2                       // buffer pointer
	mov	x2, x3                       // count
	mov	x8, #SYS_READ
	svc	#0
	cbnz	x0, have_read
	// x0 == 0 -> EOF
	b	read_done

have_read:
	add	x9, x9, x0                  // total += bytes read
	b	read_loop

read_done:
	// if no input, exit with 0 (no output)
	cbnz	x9, encode_start
	mov	x0, #0
	mov	x8, #SYS_EXIT
	svc	#0

encode_start:
	// set up base pointers
	adr	x1, input_buf               // x1 = input base
	adr	x3, output_buf              // x3 = output base
	adr	x2, b64_table               // x2 = base of base64 alphabet

	// registers:
	// x10 = input index
	// x4  = output index (shared with write_char)
	// x5  = line length counter
	mov	x10, #0                     // input index = 0
	mov	x4,  #0                     // output index = 0
	mov	x5,  #0                     // line length = 0

encode_loop:
	cmp	x10, x9
	b.ge	encode_finish

	// remaining bytes = total - idx
	sub	x6, x9, x10                 // x6 = remaining
	cmp	x6, #3
	b.ge	encode_three

	cmp	x6, #2
	beq	encode_two
	// must be 1
	b	encode_one

// ------------------------------------------------
// Encode three bytes -> four chars
encode_three:
	// load three bytes and build 24-bit value in x14
	ldrb	w7, [x1, x10]               // byte0
	ldrb	w8, [x1, x10, #1]           // byte1
