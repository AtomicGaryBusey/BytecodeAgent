.globl _start
.equ BUF_SIZE, 65536

.section .bss
buf:        .zero BUF_SIZE
buf_len:    .zero 8
error_pos:  .zero 8
err_buf:    .zero 24

.section .rodata
ok_msg:     .ascii "ok\n"
ok_len = . - ok_msg
err_prefix: .ascii "error at "
err_prefix_len = . - err_prefix
newline:    .ascii "\n"
lit_true:   .ascii "true"
lit_false:  .ascii "false"
lit_null:   .ascii "null"

.text
_start:
    # rbx = buffer base
    lea rbx, [rip + buf]

read_loop:
    xor rdi, rdi                # fd 0
    mov rsi, rbx                # buffer address
    mov rdx, BUF_SIZE
    xor rax, rax                # sys_read
    syscall
    cmp rax, 0
    je read_done
    add qword [rip + buf_len], rax
