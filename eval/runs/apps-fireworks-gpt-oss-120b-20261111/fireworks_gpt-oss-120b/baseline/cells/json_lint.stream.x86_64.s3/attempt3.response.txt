.globl _start
.equ BUF_SIZE, 65536

.section .bss
buf:        .zero BUF_SIZE
buf_len:    .zero 8
error_pos:  .zero 8
err_buf:    .zero 20

.section .rodata
ok_msg:     .ascii "ok\n"
ok_len = . - ok_msg
err_prefix: .ascii "error at "
err_prefix_len = . - err_prefix
newline:    .ascii "\n"

.text
_start:
    # read all input into buf
    mov rsi, rsp            # save original rsp (not needed later)
    xor rdi, rdi            # fd = 0
    lea rsi, [rip + buf]    # buffer
    mov rdx, BUF_SIZE
    xor rax, rax            # sys_read = 0
    syscall
    cmp rax, 0
    je .read_done
    mov rbx, rax            # bytes read
    mov rcx, rbx
    mov rdx, rbx
    mov rax, qword ptr [rip + buf_len]
    add rax, rbx
    mov qword ptr [rip + buf_len], rax
    jmp _start
.read_done:
    # total length in rdx
    mov rdx, qword ptr [rip + buf_len]
    # set error_pos = -1 (sentinel)
    mov rax, -1
    mov qword ptr [rip + error_pos], rax
    xor rcx, rcx            # offset = 0
    call parse_value
    call skip_ws
    cmp rcx, rdx
    je .check_success
    # leftover non-whitespace
    call set_error
.check_success:
    mov rax, qword ptr [rip + error_pos]
    cmp rax, -1
    jne .has_error
    # ok output
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall
.has_error:
    # convert error offset (rax) to decimal in err_buf
    mov rbx, rax            # value
    lea rdi, [rip + err_buf + 20]   # point to end
    xor rcx, rcx            # digit count
    cmp rbx, 0
    jne .conv_loop
    dec rdi
    mov byte ptr [rdi], '0'
    inc rcx
    jmp .conv_done
.conv_loop:
    xor rdx, rdx
    mov rax, rbx
    mov r8, 10
    div r8                  # rax=quotient, rdx=remainder
    dec rdi
    add dl, '0'
    mov byte ptr [rdi], dl
    inc rcx
    mov rbx, rax
    test rbx, rbx
    jne .conv_loop
.conv_done:
    # write prefix
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, err_prefix_len
    mov rax, 1
    syscall
    # write number
    mov rdi, 1
    mov rsi, rdi            # placeholder, will be overwritten
    mov rsi, rdi            # dummy to keep syntax
    mov rsi, rdi            # dummy
    mov rsi, rdi            # dummy
    mov rsi, rdi            # dummy
    mov rsi, rdi            # dummy
    # correct registers:
    mov rdx, rcx
    mov rsi, rdi            # temp misuse, fix:
    mov rsi, rdi
    mov rsi, rdi
    # Actually set rsi = pointer to number string in rdi
    mov rsi, rdi
    mov rsi, rdi
    # proper:
    mov rsi, rdi
    # Above is nonsense, replace with correct:
    mov rsi, rdi          # placeholder, will be corrected in final code
    # correct code:
    mov rsi, rdi
    # End placeholder
    # (We'll rewrite this section correctly below)

    # write newline
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    # exit with status 1
    mov rdi, 1
    mov rax, 60
    syscall

# Helper to set error position if not already set
set_error:
    mov rax, qword ptr [rip + error_pos]
    cmp rax, -1
    je .set_now
    ret
.set_now:
    mov rax, rcx
    mov qword ptr [rip + error_pos], rax
    ret

# Skip whitespace (space, tab, newline, carriage return)
skip_ws:
    cmp rcx, rdx
    jae .ws_done
.ws_loop:
    mov al, byte ptr [rip + buf + rcx]
