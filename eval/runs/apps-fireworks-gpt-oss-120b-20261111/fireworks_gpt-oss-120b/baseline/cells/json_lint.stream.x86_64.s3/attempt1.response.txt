.globl _start

#--------------------------------------------------------------------
#  Constants
#--------------------------------------------------------------------
.equ BUF_SIZE, 65536

#--------------------------------------------------------------------
#  Read input
#--------------------------------------------------------------------
read_input:
    xor r8, r8                     # total bytes read = 0
.read_loop:
    mov rax, 0                     # sys_read
    mov rdi, 0                     # fd = stdin
    lea rsi, [rip+buf]
    add rsi, r8                    # buffer + offset
    mov rcx, BUF_SIZE
    sub rcx, r8                    # remaining space
    cmp rcx, 0
    jle .read_done
    mov rdx, rcx
    syscall
    test rax, rax
    jz .read_done                  # EOF
    add r8, rax
    jmp .read_loop
.read_done:
    mov qword ptr [rip+buf_len], r8
    ret

#--------------------------------------------------------------------
#  Skip whitespace
#--------------------------------------------------------------------
skip_ws:
.ws_loop:
    cmp rbx, rdx
    jae .ws_done
    mov al, byte ptr [rsi + rbx]
    cmp al, 0x20
    je .ws_inc
    cmp al, 0x09
    je .ws_inc
    cmp al, 0x0A
    je .ws_inc
    cmp al, 0x0D
    je .ws_inc
    jmp .ws_done
.ws_inc:
    inc rbx
    jmp .ws_loop
.ws_done:
    xor eax, eax
    ret

#--------------------------------------------------------------------
#  Parse value
#--------------------------------------------------------------------
parse_value:
    call skip_ws
    mov rax, rbx
    cmp rax, rdx
    jae .val_error_eof
    mov al, byte ptr [rsi + rbx]
    cmp al, '{'
    je .val_object
    cmp al, '['
    je .val_array
    cmp al, '"'
    je .val_string
    cmp al, '-'
    je .val_number
    cmp al, 't'
    je .val_true
    cmp al, 'f'
    je .val_false
    cmp al, 'n'
    je .val_null
    ; digit?
