	.globl _start
	.equ MAX_INPUT, 65536

	.text
_start:
	// buffer base in x5, length accumulator x4 = 0
	adrp	x5, input_buf
	add	x5, x5, :lo12:input_buf
	mov	x4, #0

read_loop:
	// remaining = MAX_INPUT - x4
	mov	x2, #MAX_INPUT
	sub	x2, x2, x4
	cbnz	x2, 1f
	b	read_done
1:
	// sys_read(fd=0, buf= input_buf+x4, count=remaining)
	mov	x0, #0
	add	x1, x5, x4
	svc	#0
	// result in x0
	cmp	x0, #0
	beq	read_done
	blt	error_read
	add	x4, x4, x0
	b	read_loop

read_done:
	// store length in x19
	mov	x19, x4

	// set cur_ptr = input_buf, end_ptr = input_buf + length
	adrp	x9, input_buf
	add	x9, x9, :lo12:input_buf
	adrp	x10, end_ptr
	add	x10, x10, :lo12:end_ptr
	add	x11, x9, x19
	str	x9, [x9]          // temporary, will be overwritten
	// store cur_ptr
	adrp	x12, cur_ptr
	add	x12, x12, :lo12:cur_ptr
	str	x9, [x12]
	// store end_ptr
	str	x11, [x10]

	// parse document
	bl	skip_ws
	bl	parse_value
	bl	skip_ws

	// ensure at end
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]          // cur
	adrp	x13, end_ptr
	add	x13, x13, :lo12:end_ptr
	ldr	x13, [x13]          // end
	cmp	x12, x13
	b.ne	parse_error

	// success: write "ok\n"
	adrp	x0, ok_msg
	add	x0, x0, :lo12:ok_msg
	mov	x1, #4
	bl	write_stdout
	// exit 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

// ---------- error handling ----------
parse_error:
	// offset = cur - input_buf
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]          // cur
	adrp	x13, input_buf
	add	x13, x13, :lo12:input_buf
	sub	x0, x12, x13        // offset in x0

	// write "error at "
	adrp	x1, err_prefix
	add	x1, x1, :lo12:err_prefix
	mov	x2, #9
	bl	write_stdout

	// convert offset to decimal (store reversed in rev_buf)
	adrp	x5, rev_buf_end
	add	x5, x5, :lo12:rev_buf_end
	mov	x6, x0              // number
	cmp	x6, #0
	b.ne	conv_loop
	// number is zero, store '0'
	mov	x7, #'0'
	sub	x5, x5, #1
	strb	w7, [x5]
	b	conv_done
conv_loop:
	// divide x6 by 10
	mov	x10, #10
	udiv	x11, x6, x10
	mul	x12, x11, x10
	sub	x13, x6, x12        // remainder
	add	x13, x13, #'0'      // digit
	sub	x5, x5, #1
	strb	w13, [x5]
	mov	x6, x11
	cmp	x6, #0
	b.ne	conv_loop
conv_done:
	// x5 points to first digit, length = rev_buf_end - x5
	adrp	x7, rev_buf_end
	add	x7, x7, :lo12:rev_buf_end
	sub	x2, x7, x5          // digit length
	// write digits
	mov	x0, x5
	bl	write_stdout
	// write newline
	adrp	x0, nl_msg
	add	x0, x0, :lo12:nl_msg
	mov	x1, #1
	bl	write_stdout
	// exit 1
	mov	x0, #1
	mov	x8, #93
	svc	#0

// ---------- write helper ----------
write_stdout:
	// x0 = address, x1 = length
	mov	x2, x1
	mov	x1, x0
	mov	x0, #1               // fd = stdout
	mov	x8, #64
	svc	#0
	ret

// ---------- whitespace ----------
skip_ws:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]          // cur
	adrp	x13, end_ptr
	add	x13, x13, :lo12:end_ptr
	ldr	x13, [x13]          // end
1:	cmp	x12, x13
	b.eq	2f
	ldrb	w14, [x12]
	cmp	w14, #' '
	b.eq	ws_skip
	cmp	w14, #9
	b.eq	ws_skip
	cmp	w14, #10
	b.eq	ws_skip
	cmp	w14, #13
	b.eq	ws_skip
	b	2f
ws_skip:
	add	x12, x12, #1
	b	1b
2:	// store updated cur
	str	x12, [x11]
	ret

// ---------- value ----------
parse_value:
	bl	skip_ws
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	ldrb	w13, [x12]
	cmp	w13, #'{'
	beq	parse_object
	cmp	w13, #'['
	beq	parse_array
	cmp	w13, #'"'
	beq	parse_string
	cmp	w13, #'-'
	beq	parse_number
	// digit?
	cmp	w13, #'0'
	blt	literal_check
	cmp	w13, #'9'
	bgt	literal_check
	b	parse_number
literal_check:
	cmp	w13, #'t'
	beq	parse_true
	cmp	w13, #'f'
	beq	parse_false
	cmp	w13, #'n'
	beq	parse_null
	b	parse_error

// ---------- true ----------
parse_true:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	// compare "true"
	ldrb	w14, [x12], #1
	cmp	w14, #'t'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'r'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'u'
	b.ne	parse_error
	ldrb	w14, [x12]
	cmp	w14, #'e'
	b.ne	parse_error
	add	x12, x12, #1
	str	x12, [x11]
	ret

// ---------- false ----------
parse_false:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	ldrb	w14, [x12], #1
	cmp	w14, #'f'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'a'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'l'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'s'
	b.ne	parse_error
	ldrb	w14, [x12]
	cmp	w14, #'e'
	b.ne	parse_error
	add	x12, x12, #1
	str	x12, [x11]
	ret

// ---------- null ----------
parse_null:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	ldrb	w14, [x12], #1
	cmp	w14, #'n'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'u'
	b.ne	parse_error
	ldrb	w14, [x12], #1
	cmp	w14, #'l'
	b.ne	parse_error
	ldrb	w14, [x12]
	cmp	w14, #'l'
	b.ne	parse_error
	add	x12, x12, #1
	str	x12, [x11]
	ret

// ---------- object ----------
parse_object:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	// consume '{'
	add	x12, x12, #1
	str	x12, [x11]
	bl	skip_ws
	adrp	x13, cur_ptr
	add	x13, x13, :lo12:cur_ptr
	ldr	x14, [x13]
	adrp	x15, end_ptr
	add	x15, x15, :lo12:end_ptr
	ldr	x15, [x15]
	ldrb	w16, [x14]
	cmp	w16, #'}'
	b.eq	obj_empty
obj_members:
	bl	parse_string
	bl	skip_ws
	// expect ':'
	adrp	x17, cur_ptr
	add	x17, x17, :lo12:cur_ptr
	ldr	x18, [x17]
	ldrb	w19, [x18]
	cmp	w19, #':'
	b.ne	parse_error
	add	x18, x18, #1
	str	x18, [x17]
	bl	skip_ws
	bl	parse_value
	bl	skip_ws
	// check for ',' or '}'
	adrp	x20, cur_ptr
	add	x20, x20, :lo12:cur_ptr
	ldr	x21, [x20]
	ldrb	w22, [x21]
	cmp	w22, #','
	b.eq	obj_next
	cmp	w22, #'}'
	b.eq	obj_end
	b	parse_error
obj_next:
	add	x21, x21, #1
	str	x21, [x20]
	bl	skip_ws
	b	obj_members
obj_end:
	add	x21, x21, #1
	str	x21, [x20]
	ret
obj_empty:
	add	x14, x14, #1
	str	x14, [x11]
	ret

// ---------- array ----------
parse_array:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	// consume '['
	add	x12, x12, #1
	str	x12, [x11]
	bl	skip_ws
	adrp	x13, cur_ptr
	add	x13, x13, :lo12:cur_ptr
	ldr	x14, [x13]
	adrp	x15, end_ptr
	add	x15, x15, :lo12:end_ptr
	ldr	x15, [x15]
	ldrb	w16, [x14]
	cmp	w16, #']'
	b.eq	arr_empty
arr_elements:
	bl	parse_value
	bl	skip_ws
	adrp	x17, cur_ptr
	add	x17, x17, :lo12:cur_ptr
	ldr	x18, [x17]
	ldrb	w19, [x18]
	cmp	w19, #','
	b.eq	arr_next
	cmp	w19, #']'
	b.eq	arr_end
	b	parse_error
arr_next:
	add	x18, x18, #1
	str	x18, [x17]
	bl	skip_ws
	b	arr_elements
arr_end:
	add	x18, x18, #1
	str	x18, [x17]
	ret
arr_empty:
	add	x14, x14, #1
	str	x14, [x11]
	ret

// ---------- string ----------
parse_string:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	// expect opening '"'
	ldrb	w13, [x12]
	cmp	w13, #'"'
	b.ne	parse_error
	add	x12, x12, #1
	str	x12, [x11]
str_loop:
	adrp	x13, cur_ptr
	add	x13, x13, :lo12:cur_ptr
	ldr	x14, [x13]
	adrp	x15, end_ptr
	add	x15, x15, :lo12:end_ptr
	ldr	x15, [x15]
	cmp	x14, x15
	b.eq	parse_error          // unexpected EOF
	ldrb	w16, [x14]
	cmp	w16, #'"'
	b.eq	str_done
	cmp	w16, #'\\'
	b.eq	str_escape
	// control chars < 0x20 are errors
	cmp	w16, #0x20
	b.lo	parse_error
	// normal char
	add	x14, x14, #1
	str	x14, [x13]
	b	str_loop
str_escape:
	add	x14, x14, #1
	cmp	x14, x15
	b.eq	parse_error
	ldrb	w17, [x14]
	cmp	w17, #'u'
	b.eq	str_unicode
	// simple escapes
	cmp	w17, #'"'
	beq	esc_ok
	cmp	w17, #'\\'
	beq	esc_ok
	cmp	w17, #'/'
	beq	esc_ok
	cmp	w17, #'b'
	beq	esc_ok
	cmp	w17, #'f'
	beq	esc_ok
	cmp	w17, #'n'
	beq	esc_ok
	cmp	w17, #'r'
	beq	esc_ok
	cmp	w17, #'t'
	beq	esc_ok
	b	parse_error
esc_ok:
	add	x14, x14, #1
	str	x14, [x13]
	b	str_loop
str_unicode:
	add	x14, x14, #1
	mov	x19, #4               // remaining hex digits
unicode_loop:
	cmp	x19, #0
	b.eq	unicode_done
	cmp	x14, x15
	b.eq	parse_error
	ldrb	w20, [x14]
	// 0-9
	cmp	w20, #'0'
	blt	parse_error
	cmp	w20, #'9'
	ble	unicode_next
	// A-F
	cmp	w20, #'A'
	blt	parse_error
	cmp	w20, #'F'
	ble	unicode_next
	// a-f
	cmp	w20, #'a'
	blt	parse_error
	cmp	w20, #'f'
	ble	unicode_next
	b	parse_error
unicode_next:
	add	x14, x14, #1
	sub	x19, x19, #1
	b	unicode_loop
unicode_done:
	str	x14, [x13]
	b	str_loop
str_done:
	add	x14, x14, #1
	str	x14, [x11]
	ret

// ---------- number ----------
parse_number:
	adrp	x11, cur_ptr
	add	x11, x11, :lo12:cur_ptr
	ldr	x12, [x11]
	adrp	x13, end_ptr
	add	x13, x13, :lo12:end_ptr
	// optional '-'
	ldrb	w14, [x12]
	cmp	w14, #'-'
	b.ne	num_int
	add	x12, x12, #1
	// need at least one digit after '-'
	ldrb	w14, [x12]
	cmp	w14, #'0'
	blt	parse_error
	cmp	w14, #'9'
	bgt	parse_error
num_int:
	ldrb	w14, [x12]
	cmp	w14, #'0'
	b.eq	num_int_zero
	// first digit 1-9
	cmp	w14, #'1'
	blt	parse_error
	cmp	w14, #'9'
	bgt	parse_error
	// consume subsequent digits
1:	add	x12, x12, #1
	cmp	x12, x13
	b.eq	num_frac_exp_check
	ldrb	w14, [x12]
	cmp	w14, #'0'
	blt	num_frac_exp_check
	cmp	w14, #'9'
	bgt	num_frac_exp_check
	b	1b
num_int_zero:
	// leading zero
	add	x12, x12, #1
	b	num_frac_exp_check
num_frac_exp_check:
	// optional fraction
	cmp	x12, x13
	b.eq	num_exp_check
	ldrb	w14, [x12]
	cmp	w14, #'.'
	b.ne	num_exp_check
	// consume '.'
	add	x12, x12, #1
	// need at least one digit
	cmp	x12, x13
	b.eq	parse_error
	ldrb	w14, [x12]
	cmp	w14, #'0'
	blt	parse_error
	cmp	w14, #'9'
	bgt	parse_error
	// consume fraction digits
2:	add	x12, x12, #1
	cmp	x12, x13
	b.eq	num_exp_check
	ldrb	w14, [x12]
	cmp	w14, #'0'
	blt	num_exp_check
	cmp	w14, #'9'
	bgt	num_exp_check
	b	2b
num_exp_check:
	// optional exponent
	cmp	x12, x13
	b.eq	num_done
	ldrb	w14, [x12]
	cmp	w14, #'e'
	beq	exp_start
	cmp	w14, #'E'
	b.ne	num_done
exp_start:
	// consume e/E
	add	x12, x12, #1
	// optional sign
	cmp	x12, x13
	b.eq	parse_error
	ldrb	w14, [x12]
	cmp	w14, #'+'
	beq	exp_sign
	cmp	w14, #'-'
	beq	exp_sign
	b	exp_digits
exp_sign:
	add	x12, x12, #1
exp_digits:
	// need at least one digit
	cmp	x12, x13
	b.eq	parse_error
	ldrb	w14, [x12]
	cmp	w14, #'0'
	blt	parse_error
	cmp	w14, #'9'
	bgt	parse_error
	// consume exponent digits
3:	add	x12, x12, #1
	cmp	x12, x13
	b.eq	num_done
	ldrb	w14, [x12]
	cmp	w14, #'0'
	blt	num_done
	cmp	w14, #'9'
	bgt	num_done
	b	3b
num_done:
	str	x12, [x11]
	ret

	// ---------- data ----------
	.section .rodata
	.p2align 2
ok_msg:
	.asciz "ok\n"
err_prefix:
	.asciz "error at "
nl_msg:
	.asciz "\n"

	.section .bss
	.p2align 3
input_buf:
	.space MAX_INPUT
cur_ptr:
	.zero 8
end_ptr:
	.zero 8
rev_buf:
	.space 12
rev_buf_end:
	.zero 0

error_read:
	// read failure, exit with status 1
	mov	x0, #1
	mov	x8, #93
	svc	#0
