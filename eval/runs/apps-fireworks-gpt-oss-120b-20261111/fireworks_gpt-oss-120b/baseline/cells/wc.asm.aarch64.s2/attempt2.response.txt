.global _start
.text

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536

_start:
	// counters: x19=bytes, x20=lines, x21=words, x22=in_word
	mov	x19, xzr
	mov	x20, xzr
	mov	x21, xzr
	mov	x22, xzr

read_loop:
	// read(fd=0, buf, BUF_SIZE)
	mov	x0, #0
	adr	x1, buf
	mov	x2, #BUF_SIZE
	mov	x8, #SYS_READ
	svc	#0
	cbnz	x0, process_chunk
	// EOF
	b	done

process_chunk:
	// x0 = nread
	mov	x23, x0			// remaining bytes in chunk
	adr	x24, buf		// pointer to current byte

process_byte:
	ldrb	w25, [x24], #1		// load byte, advance pointer
	add	x19, x19, #1		// total bytes++

	// newline check
	cmp	w25, #10
	bne	not_newline
	add	x20, x20, #1
not_newline:

	// whitespace test
	cmp	w25, #' '
	beq	is_ws
	cmp	w25, #9
	beq	is_ws
	cmp	w25, #10
	beq	is_ws
	cmp	w25, #11
	beq=is_ws
	cmp	w25, #12
	beq=is_ws
	cmp	w25, #13
	beq=is_ws
	b	not_ws
is_ws:
	// whitespace => exit word
	mov	x22, xzr
	b	after_ws
not_ws:
	// non‑whitespace
	cbnz	x22, after_ws		// already in a word?
	add	x21, x21, #1		// start new word
	mov	x22, #1
after_ws:
	subs	x23, x23, #1
	bne	process_byte

	b	read_loop

done:
	// prepare output in outbuf
	adr	x0, outbuf		// destination pointer

	// lines
	mov	x1, x0
	mov	x0, x20
	bl	convert
	// space
	mov	w5, #' '
	strb	w5, [x0], #1

	// words
	mov	x1, x0
	mov	x0, x21
	bl	convert
	// space
	mov	w5, #' '
	strb	w5, [x0], #1

	// bytes
	mov	x1, x0
	mov	x0, x19
	bl	convert
	// newline
	mov	w5, #10
	strb	w5, [x0], #1

	// length = x0 - outbuf
	adr	x1, outbuf
	sub	x2, x0, x1

	// write(fd=1, outbuf, len)
	mov	x0, #1
	mov	x8, #SYS_WRITE
	svc	#0

	// exit(0)
	mov	x0, #0
	mov	x8, #SYS_EXIT
	svc	#0

// ------------------------------------------------------------
// convert: x0 = number, x1 = destination pointer
// returns x0 = pointer after written number
// clobbers: x2‑x10
// ------------------------------------------------------------
convert:
	// handle zero quickly
	cbnz	x0, .Lconvert_loop
	// number == 0
	mov	w5, #'0'
	strb	w5, [x1], #1
	mov	x0, x1
	ret

.Lconvert_loop:
	// divisor constant 10
	mov	w10, #10

	// revbuf base
	adr	x2, revbuf
	mov	x3, xzr			// digit count = 0

.Ldiv_loop:
	udiv	x4, x0, x10
	mul	x5, x4, x10
	sub	x5, x0, x5		// remainder
	add	w5, w5, #'0'
	strb	w5, [x2, x3]
	add	x3, x3, #1
	mov	x0, x4
	cbnz	x0, .Ldiv_loop

	// copy digits in reverse order to destination
.Lcopy_loop:
	subs	x3, x3, #1
	blt	.Lcopy_done
	ldrb	w5, [x2, x3]
	strb	w5, [x1], #1
	b	.Lcopy_loop
.Lcopy_done:
	mov	x0, x1
	ret

.bss
.p2align 6
buf:
	.space BUF_SIZE
.p2align 3
outbuf:
	.space 64
.p2align 3
revbuf:
	.space 32
