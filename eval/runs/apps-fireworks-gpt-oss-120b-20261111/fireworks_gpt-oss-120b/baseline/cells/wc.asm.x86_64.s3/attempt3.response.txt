.section .bss
    .align 8
lines:          .space 8          # 64‑bit counter
words:          .space 8
bytes_cnt:      .space 8
in_word:        .space 1          # 0 = not in word, 1 = in word
    .align 8
inbuf:          .space 65536
outbuf:         .space 80          # enough for three numbers, spaces and newline

.section .text
.globl _start
_start:
    lea rbx, [rip + inbuf]          # buffer base

read_loop:
    mov rdi, 0                      # stdin
    mov rsi, rbx                    # buffer
    mov rdx, 65536                  # size
    mov rax, 0                      # sys_read
    syscall
    test rax, rax
    jz done_read                    # 0 bytes = EOF

    mov rcx, rax                    # bytes this chunk
    add qword [rip + bytes_cnt], rax

    xor r8, r8                      # index = 0

process_loop:
    cmp r8, rcx
    jae after_process

    mov al, byte [rbx + r8]
    inc r8

    cmp al, 0x0A                    # newline?
    jne not_newline
    inc qword [rip + lines]
not_newline:

    # whitespace test
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    jmp not_ws

is_ws:
    mov byte [rip + in_word], 0
    jmp cont_process

not_ws:
    mov al, byte [rip + in_word]
    test al, al
    jne cont_process                # already in word
    inc qword [rip + words]
    mov byte [rip + in_word], 1

cont_process:
    jmp process_loop

after_process:
    jmp read_loop

done_read:
    # build output line backwards in outbuf
    lea rbx, [rip + outbuf + 80]    # rbx = past end of buffer
    mov rdi, rbx

    dec rdi
    mov byte [rdi], 0x0A            # newline

    mov rax, qword [rip + bytes_cnt]
    call u64_to_rev

    dec rdi
    mov byte [rdi], ' '

    mov rax, qword [rip + words]
    call u64_to_rev

    dec rdi
    mov byte [rdi], ' '

    mov rax, qword [rip + lines]
    call u64_to_rev

    # write the result
    mov rsi, rdi                    # start address
    mov rdx, rbx
    sub rdx, rdi                    # length = end - start

    mov rdi, 1                      # stdout
    mov rax, 1                      # sys_write
    syscall

    mov rax, 60                     # sys_exit
    xor rdi, rdi
    syscall

# ------------------------------------------------------------
# u64_to_rev: rax = number, rdi points past the end of output.
# writes decimal digits backwards, updates rdi, returns with rdi
# pointing to first digit.
u64_to_rev:
    push rbx
    mov rbx, rax
    cmp rbx, 0
    jne .nonzero
    dec rdi
    mov byte [rdi], '0'
    jmp .done
.nonzero:
digit_loop:
    xor rdx, rdx
    mov rax, rbx
    mov rcx, 10
    div rcx                 # rax = rbx/10, rdx = rbx%10
    dec rdi
    add dl, '0'
    mov byte [rdi], dl
    mov rbx, rax
    test rbx, rbx
    jne digit_loop
.done:
    pop rbx
    ret
