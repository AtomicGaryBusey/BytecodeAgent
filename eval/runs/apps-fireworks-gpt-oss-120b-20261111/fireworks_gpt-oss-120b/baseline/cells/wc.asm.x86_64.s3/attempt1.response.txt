.section .bss
    .align 8
lines:          .quad 0
words:          .quad 0
bytes_cnt:      .quad 0
in_word:        .byte 0
    .align 8
inbuf:          .space 65536
outbuf:         .space 64

.section .text
.global _start
_start:
    lea rbx, [rip + inbuf]          # base of input buffer

read_loop:
    mov rdi, 0                      # fd = stdin
    mov rsi, rbx                    # buffer address
    mov rdx, 65536                  # max size
    mov rax, 0                      # sys_read
    syscall
    test rax, rax
    jz done_read

    mov rcx, rax                    # bytes read in rcx
    add qword [rip + bytes_cnt], rax
    xor r8, r8                      # index = 0

process_loop:
    cmp r8, rcx
    jae after_process

    mov al, byte [rbx + r8]         # load byte
    inc r8

    cmp al, 0x0A
    jne not_newline
    inc qword [rip + lines]
not_newline:

    mov dl, 0
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    jmp not_ws
is_ws:
    mov dl, 1
not_ws:
    test dl, dl
    jne ws_action

    # non‑whitespace
    mov al, byte [rip + in_word]
    test al, al
    jne already_in_word
    inc qword [rip + words]
    mov byte [rip + in_word], 1
    jmp cont_process
already_in_word:
    jmp cont_process

ws_action:
    mov byte [rip + in_word], 0

cont_process:
    jmp process_loop

after_process:
    jmp read_loop

done_read:
    # Build output line in outbuf (backwards)
    lea rdi, [rip + outbuf + 64]    # end of buffer

    dec rdi
    mov byte [rdi], 0x0A            # newline

    mov rax, [rip + bytes_cnt]
    call u64_to_rev

    dec rdi
    mov byte [rdi], ' '

    mov rax, [rip + words]
    call u64_to_rev

    dec rdi
    mov byte [rdi], ' '

    mov rax, [rip + lines]
    call u64_to_rev

    lea rsi, [rip + outbuf + 64]    # buffer end
    mov rcx, rsi
    sub rcx, rdi                    # length = end - start

    mov rdx, rcx
    mov rsi, rdi                    # pointer to start
    mov rdi, 1                      # stdout
    mov rax, 1                      # sys_write
    syscall

    mov rax, 60                     # sys_exit
    xor rdi, rdi
    syscall

# ------------------------------------------------------------
# u64_to_rev: rax = number, rdi points past the end of output.
# writes the decimal representation backwards, updating rdi.
# Returns with rdi pointing to the first digit.
u64_to_rev:
    push rbx
    mov rbx, rax
    cmp rbx, 0
    jne .nonzero
    dec rdi
    mov byte [rdi], '0'
    jmp .done
.nonzero:
    .digit_loop:
        xor rdx, rdx
        mov rax, rbx
        mov rcx, 10
        div rcx                # rax = rbx/10, rdx = rbx%10
        dec rdi
        add dl, '0'
        mov byte [rdi], dl
        mov rbx, rax
        test rbx, rbx
        jne .digit_loop
.done:
    pop rbx
    ret
