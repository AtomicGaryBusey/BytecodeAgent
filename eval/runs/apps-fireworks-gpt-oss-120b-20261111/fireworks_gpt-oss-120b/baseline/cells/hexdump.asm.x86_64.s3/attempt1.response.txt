.globl _start

.equ INPUT_MAX, 65536

.section .bss
input_buf:
    .skip INPUT_MAX
line_buf:
    .skip 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.text
_start:
    # read all input into input_buf
    lea rsi, [rel input_buf]          # rsi = write pointer
    xor r12, r12                      # total bytes read = 0
read_loop:
    mov rdi, 0                        # fd = stdin
    mov rdx, INPUT_MAX
    sub rdx, r12                      # remaining space
    mov rax, 0                        # sys_read
    syscall
    test rax, rax
    jz after_read
    add r12, rax                      # total += bytes read
    add rsi, rax                      # advance write pointer
    cmp r12, INPUT_MAX
    jb read_loop

after_read:
    # r12 = total length
    xor r8, r8                        # offset = 0

row_loop:
    cmp r8, r12
    jae after_rows

    # n = min(16, total - offset)
    mov rcx, r12
    sub rcx, r8
    cmp rcx, 16
    jle have_n
    mov rcx, 16
have_n:
    mov r13, rcx                      # n bytes in this row

    # line buffer pointer
    lea r9, [rel line_buf]

    # ----- offset (8 hex digits) -----
    mov ebp, 8                        # digit count
    mov edx, 28                       # shift amount
offset_loop:
    mov rcx, r8
    mov cl, dl
    shr rcx, cl
    and rcx, 0xF
    mov al, byte ptr [rel hex_digits + rcx]
    mov byte ptr [r9], al
    inc r9
    sub dl, 4
    dec ebp
    jne offset_loop

    # two spaces after offset
    mov byte ptr [r9], ' '
    inc r9
    mov byte ptr [r9], ' '
    inc r9

    # data pointer for this row
    lea rdx, [rel input_buf]
    add rdx, r8

    # ----- hex bytes and spaces -----
    xor r10d, r10d                    # i = 0
hex_byte_loop:
    cmp r10, 8
    jne after_mid_space
    mov byte ptr [r9], ' '
    inc r9
after_mid_space:
    cmp r10, 16
    jae after_hex_bytes

    cmp r10, r13
    jae missing_byte

    # present byte
    mov al, byte ptr [rdx + r10]
    mov bl, al

    # high nibble
    shr al, 4
    and al, 0xF
    movzx edx, al
    mov al, byte ptr [rel hex_digits + rdx]   # use rdx as temporary index, safe because rdx holds data ptr not needed now
    mov byte ptr [r9], al
    inc r9

    # low nibble
    mov al, bl
    and al, 0xF
    movzx edx, al
    mov al, byte ptr [rel hex_digits + rdx]
    mov byte ptr [r9], al
    inc r9

    # space after byte
    mov byte ptr [r9], ' '
    inc r9
    jmp next_hex_byte

missing_byte:
    # three spaces for missing byte
    mov byte ptr [r9], ' '
    inc r9
    mov byte ptr [r9], ' '
    inc r9
    mov byte ptr [r9], ' '
    inc r9

next_hex_byte:
    inc r10
    jmp hex_byte_loop

after_hex_bytes:
    # space before '|'
    mov byte ptr [r9], ' '
    inc r9
    mov byte ptr [r9], '|'
    inc r9

    # ----- ASCII representation -----
    xor r10d, r10d                    # i = 0
ascii_loop:
    cmp r10, r13
    jae after_ascii
    mov al, byte ptr [rdx + r10]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7e
    jg ascii_dot
    jmp ascii_store
ascii_dot:
    mov al, '.'
ascii_store:
    mov byte ptr [r9], al
    inc r9
    inc r10
    jmp ascii_loop

after_ascii:
    # closing '|', newline
    mov byte ptr [r9], '|'
    inc r9
    mov byte ptr [r9], 0x0a
    inc r9

    # write the line
    lea rdx, [rel line_buf]
    mov rbx, r9
    sub rbx, rdx                     # length
    mov rdi, 1                       # stdout
    mov rsi, rdx
    mov rdx, rbx
    mov rax, 1                       # sys_write
    syscall

    # next row
    add r8, 16
    jmp row_loop

after_rows:
    # ----- final total offset line -----
    lea rbx, [rel line_buf]

    mov ebp, 8
    mov edx, 28
total_offset_loop:
    mov rcx, r12
    mov cl, dl
    shr rcx, cl
    and rcx, 0xF
    mov al, byte ptr [rel hex_digits + rcx]
    mov byte ptr [rbx], al
    inc rbx
    sub dl, 4
    dec ebp
    jne total_offset_loop

    mov byte ptr [rbx], 0x0a
    inc rbx

    lea rdx, [rel line_buf]
    mov rbx, rbx
    sub rbx, rdx                     # length
    mov rdi, 1
    mov rsi, rdx
    mov rdx, rbx
    mov rax, 1
    syscall

    # exit
    mov rax, 60
    xor rdi, rdi
    syscall
