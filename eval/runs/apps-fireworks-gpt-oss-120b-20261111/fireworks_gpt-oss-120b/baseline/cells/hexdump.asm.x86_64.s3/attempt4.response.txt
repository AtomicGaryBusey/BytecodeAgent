.globl _start

.equ INPUT_MAX, 65536

.section .bss
.align 16
input_buf:
    .zero INPUT_MAX
total_len:
    .zero 8
line_buf:
    .zero 80

.section .rodata
.align 16
hex_digits:
    .ascii "0123456789abcdef"

.text
_start:
    # -------------------------------------------------
    # read all input into input_buf, total_len = bytes read
    # -------------------------------------------------
    xor r8, r8                    # r8 = total_len
    lea r9, [rip + input_buf]    # r9 = write pointer

read_loop:
    mov rdi, 0                    # fd = stdin
    mov rdx, INPUT_MAX
    sub rdx, r8                   # remaining space
    mov rax, 0                    # sys_read
    syscall
    test rax, rax
    jz read_done
    add r8, rax                   # total_len += bytes read
    add r9, rax                   # advance pointer
    cmp r8, INPUT_MAX
    jb read_loop

read_done:
    mov [rip + total_len], r8

    # -------------------------------------------------
    # dump rows
    # -------------------------------------------------
    xor r8, r8                    # r8 = offset (row start)

row_loop:
    mov rax, [rip + total_len]   # total_len
    cmp r8, rax
    jae final_offset_line

    # n = min(16, total_len - offset)
    mov r10, rax
    sub r10, r8
    cmp r10, 16
    jle have_n
    mov r10, 16
have_n:
    # r10 = n (bytes in this row)

    # pointers
    lea r11, [rip + input_buf]   # base of input
    add r11, r8                   # r11 = &input_buf[offset]
    lea rsi, [rip + line_buf]    # rsi = write pointer into line_buf
    lea rbx, [rip + hex_digits]  # rbx = hex table base

    # ----- offset (8 hex digits) -----
    mov r12, 8                    # digit counter
    mov rcx, 28                   # shift amount
offset_loop:
    mov rax, r8
    shr rax, cl                   # shift by rcx low 8 bits
    and rax, 0xF
    mov al, [rbx + rax]
    mov [rsi], al
    inc rsi
    sub rcx, 4
    dec r12
    jnz offset_loop

    # two spaces after offset
    mov byte [rsi], ' '
    inc rsi
    mov byte [rsi], ' '
    inc rsi

    # ----- hex bytes and spaces -----
    xor rcx, rcx                  # i = 0
hex_loop:
    cmp rcx, 8
    jne after_mid_space
    mov byte [rsi], ' '
    inc rsi
after_mid_space:
    cmp rcx, 16
    jae after_hex_bytes

    cmp rcx, r10
    jae missing_byte

    # present byte
    mov al, [r11 + rcx]           # byte value
    mov bl, al
    shr bl, 4
    and bl, 0xF
    movzx rdx, bl
    mov dl, [rbx + rdx]           # high nibble char
    mov [rsi], dl
    inc rsi

    mov bl, al
    and bl, 0xF
    movzx rdx, bl
    mov dl, [rbx + rdx]           # low nibble char
    mov [rsi], dl
    inc rsi

    mov byte [rsi], ' '
    inc rsi
    jmp next_hex

missing_byte:
    # three spaces for missing byte
    mov byte [rsi], ' '
    inc rsi
    mov byte [rsi], ' '
    inc rsi
    mov byte [rsi], ' '
    inc rsi

next_hex:
    inc rcx
    jmp hex_loop

after_hex_bytes:
    # space before '|'
    mov byte [rsi], ' '
    inc rsi
    mov byte [rsi], '|'
    inc rsi

    # ----- ASCII representation -----
    xor rcx, rcx                  # i = 0
ascii_loop:
    cmp rcx, r10
    jae after_ascii
    mov al, [r11 + rcx]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7e
    jg ascii_dot
    mov [rsi], al
    inc rsi
    jmp ascii_next
ascii_dot:
    mov byte [rsi], '.'
    inc rsi
ascii_next:
    inc rcx
    jmp ascii_loop

after_ascii:
    mov byte [rsi], '|'
    inc rsi
    mov byte [rsi], 0x0a
    inc rsi

    # write the line (rsi - line_buf) bytes
    lea rdx, [rip + line_buf]
    mov rcx, rsi
    sub rcx, rdx                  # rcx = length
    mov rdi, 1                    # stdout
    mov rsi, rdx
    mov rdx, rcx
    mov rax, 1                    # sys_write
    syscall

    # next row
    add r8, 16
    jmp row_loop

    # -------------------------------------------------
    # final total offset line
    # -------------------------------------------------
final_offset_line:
    lea rsi, [rip + line_buf]    # rsi = write pointer
    lea rbx, [rip + hex_digits]  # hex table

    mov rdx, [rip + total_len]   # total length value
    mov r12, 8                    # digit counter
    mov rcx, 28                   # shift amount
final_offset_loop:
    mov rax, rdx
    shr rax, cl
    and rax, 0xF
    mov al, [rbx + rax]
    mov [rsi], al
    inc rsi
    sub rcx, 4
    dec r12
    jnz final_offset_loop

    mov byte [rsi], 0x0a
    inc rsi

    # write final line
    lea rdx, [rip + line_buf]
    mov rcx, rsi
    sub rcx, rdx
    mov rdi, 1
    mov rsi, rdx
    mov rdx, rcx
    mov rax, 1
    syscall

    # -------------------------------------------------
    # exit
    # -------------------------------------------------
    mov rax, 60                  # sys_exit
    xor rdi, rdi                 # status 0
    syscall
