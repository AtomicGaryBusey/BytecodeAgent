	.text
	.globl _start
_start:
	// x19 = input buffer address
	adrp	x19, input_buf
	add	x19, x19, :lo12:input_buf

	// x20 = input length (starts at 0)
	mov	x20, #0

read_loop:
	// read(fd=0, buf = input_buf + len, count = 65536 - len)
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #65536
	sub	x2, x2, x20
	svc	#0
	cmp	x0, #0
	beq	read_done
	cmp	x0, #0
	blt	read_error
	add	x20, x20, x0
	cmp	x20, #65536
	blt	read_loop
read_done:
	// store length
	adrp	x0, input_len
	add	x0, x0, :lo12:input_len
	str	x20, [x0]

	// cur offset = 0
	mov	x21, #0
	// error flag = 0
	mov	x23, #0

	// parse document
	bl	parse_ws
	bl	parse_value
	bl	parse_ws

	// check for extra bytes
	cmp	x21, x20
	beq	success
	// if not already error, set error at cur
	cmp	x23, #0
	bne	after_error_check
	mov	x22, x21          // error_offset = cur
	mov	x23, #1
after_error_check:
	b	error

read_error:
	// exit with status 1 on read error
	mov	x0, #1
	mov	x8, #93
	svc	#0

	// ------------------------------------------------------------
	// Parsing helpers
	// ------------------------------------------------------------
parse_ws:
	// skip whitespace (space, tab, LF, CR)
1:	cmp	x21, x20
	bhs	2f
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #0x20
	beq	ws_inc
	cmp	w0, #0x09
	beq	ws_inc
	cmp	w0, #0x0A
	beq	ws_inc
	cmp	w0, #0x0D
	beq	ws_inc
	b	2f
ws_inc:
	add	x21, x21, #1
	b	1b
2:	ret

parse_value:
	// check error flag
	cmp	x23, #0
	bne	parse_value_exit
	// ensure we have at least one byte
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	// dispatch based on first byte
	cmp	w0, #'{'		// object
	beq	parse_object
	cmp	w0, #'['		// array
	beq	parse_array
	cmp	w0, #'"'		// string
	beq	parse_string
	cmp	w0, #'-'		// number
	beq	parse_number
	cmp	w0, #'t'		// true
	beq	parse_true
	cmp	w0, #'f'		// false
	beq	parse_false
	cmp	w0, #'n'		// null
	beq	parse_null
	// none matched -> error
	b	parse_error_invalid

parse_value_exit:
	ret

parse_error_incomplete:
	// set error at end of input (incomplete document)
	mov	x22, x20
	mov	x23, #1
	ret

parse_error_invalid:
	// set error at current position
	mov	x22, x21
	mov	x23, #1
	ret

// ------------------------------------------------------------
// Literal parsing (true, false, null)
// ------------------------------------------------------------
parse_true:
	adrp	x1, lit_true
	add	x1, x1, :lo12:lit_true
	mov	x2, #4
	b	literal_match

parse_false:
	adrp	x1, lit_false
	add	x1, x1, :lo12:lit_false
	mov	x2, #5
	b	literal_match

parse_null:
	adrp	x1, lit_null
	add	x1, x1, :lo12:lit_null
	mov	x2, #4
	b	literal_match

literal_match:
	// x1 = literal address, x2 = length
	mov	x3, #0
1:	cmp	x3, x2
	bhs	2f
	cmp	x21, x20
	bhs	parse_error_incomplete
	ldrb	w4, [x19, x21]
	ldrb	w5, [x1, x3]
	cmp	w4, w5
	bne	parse_error_invalid
	add	x21, x21, #1
	add	x3, x3, #1
	b	1b
2:	ret

// ------------------------------------------------------------
// Object parsing
// ------------------------------------------------------------
parse_object:
	// consume '{'
	add	x21, x21, #1
	bl	parse_ws
	// check for empty object
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'}'
	beq	empty_object_done
	// parse first member
	bl	parse_string
	cmp	x23, #0
	bne	parse_object_exit
	bl	parse_ws
	// expect ':'
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #':'
	bne	parse_error_invalid
	add	x21, x21, #1
	bl	parse_ws
	bl	parse_value
	cmp	x23, #0
	bne	parse_object_exit
	bl	parse_ws
	// loop for additional members
1:	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #','
	bne	2f
	add	x21, x21, #1
	bl	parse_ws
	bl	parse_string
	cmp	x23, #0
	bne	parse_object_exit
	bl	parse_ws
	// expect ':'
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #':'
	bne	parse_error_invalid
	add	x21, x21, #1
	bl	parse_ws
	bl	parse_value
	cmp	x23, #0
	bne	parse_object_exit
	bl	parse_ws
	b	1b
2:	// expect '}'
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'}'
	bne	parse_error_invalid
	add	x21, x21, #1
	b	3f
empty_object_done:
	// consume '}'
	add	x21, x21, #1
3:
	ret
parse_object_exit:
	ret

// ------------------------------------------------------------
// Array parsing
// ------------------------------------------------------------
parse_array:
	// consume '['
	add	x21, x21, #1
	bl	parse_ws
	// check for empty array
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #']'
	beq	empty_array_done
	// parse first element
	bl	parse_value
	cmp	x23, #0
	bne	parse_array_exit
	bl	parse_ws
	// loop for additional elements
1:	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #','
	bne	2f
	add	x21, x21, #1
	bl	parse_ws
	bl	parse_value
	cmp	x23, #0
	bne	parse_array_exit
	bl	parse_ws
	b	1b
2:	// expect ']'
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #']'
	bne	parse_error_invalid
	add	x21, x21, #1
	b	3f
empty_array_done:
	// consume ']'
	add	x21, x21, #1
3:
	ret
parse_array_exit:
	ret

// ------------------------------------------------------------
// String parsing
// ------------------------------------------------------------
parse_string:
	// expect opening quote
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'"'
	bne	parse_error_invalid
	add	x21, x21, #1
	// loop over characters
1:	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'"'          // end of string
	beq	end_string
	cmp	w0, #'\'
	beq	escape_sequence
	// check for control characters (0x00-0x1F)
	cmp	w0, #0x20
	bhs	2f
	// invalid control char
	b	parse_error_invalid
2:	// regular character, advance
	add	x21, x21, #1
	b	1b
escape_sequence:
	// consume backslash
	add	x21, x21, #1
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	// check allowed simple escapes
	cmp	w0, #'"'
	beq	simple_escape
	cmp	w0, #'\\'
	beq	simple_escape
	cmp	w0, #'/'
	beq	simple_escape
	cmp	w0, #'b'
	beq	simple_escape
	cmp	w0, #'f'
	beq	simple_escape
	cmp	w0, #'n'
	beq	simple_escape
	cmp	w0, #'r'
	beq	simple_escape
	cmp	w0, #'t'
	beq	simple_escape
	cmp	w0, #'u'          // unicode escape
	beq	unicode_escape
	// invalid escape
	b	parse_error_invalid
simple_escape:
	// consume escaped character
	add	x21, x21, #1
	b	1b
unicode_escape:
	// consume 'u'
	add	x21, x21, #1
	// need exactly 4 hex digits
	mov	x2, #4
1u:	cmp	x2, #0
	beq	2u
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	// check hex digit
	cmp	w0, #'0'
	blt	parse_error_invalid
	cmp	w0, #'9'
	ble	hex_ok
	cmp	w0, #'a'
	blt	parse_error_invalid
	cmp	w0, #'f'
	ble	hex_ok
	cmp	w0, #'A'
	blt	parse_error_invalid
	cmp	w0, #'F'
	ble	hex_ok
	b	parse_error_invalid
hex_ok:
	add	x21, x21, #1
	sub	x2, x2, #1
	b	1u
2u:
	b	1b
end_string:
	// consume closing quote
	add	x21, x21, #1
	ret

// ------------------------------------------------------------
// Number parsing
// ------------------------------------------------------------
parse_number:
	// optional '-'
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'-'
	bne	num_int_part
	add	x21, x21, #1
	// after '-', must have int part
	cmp	x21, x20
	bhs	parse_error_incomplete

num_int_part:
	// check first digit
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'0'
	beq	num_int_zero
	cmp	w0, #'1'
	blt	parse_error_invalid
	cmp	w0, #'9'
	bgt	parse_error_invalid
	// non-zero start, consume digits
	add	x21, x21, #1
1int_loop:
	cmp	x21, x20
	bhs	num_frac_check
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'0'
	blt	num_frac_check
	cmp	w0, #'9'
	bgt	num_frac_check
	add	x21, x21, #1
	b	1int_loop
num_int_zero:
	// integer is exactly '0'
	add	x21, x21, #1
	// after zero, may have fraction or exponent or end
	b	num_frac_check

num_frac_check:
	// optional fraction
	cmp	x21, x20
	bhs	num_exp_check
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'.'
	bne	num_exp_check
	// consume '.'
	add	x21, x21, #1
	// must have at least one digit
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'0'
	blt	parse_error_invalid
	cmp	w0, #'9'
	bgt	parse_error_invalid
	// consume first digit
	add	x21, x21, #1
	// consume subsequent digits
1frac_loop:
	cmp	x21, x20
	bhs	num_exp_check
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'0'
	blt	num_exp_check
	cmp	w0, #'9'
	bgt	num_exp_check
	add	x21, x21, #1
	b	1frac_loop

num_exp_check:
	// optional exponent
	cmp	x21, x20
	bhs	num_done
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'e'
	bne	num_exp_check2
	cmp	w0, #'E'
	bne	num_exp_check2
	// exponent indicator
	add	x21, x21, #1
	// optional sign
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'+'
	beq	exp_sign_consumed
	cmp	w0, #'-'
	beq	exp_sign_consumed
	b	exp_digits_needed
exp_sign_consumed:
	add	x21, x21, #1
exp_digits_needed:
	// must have at least one digit
	cmp	x21, x20
	bhs	parse_error_incomplete
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'0'
	blt	parse_error_invalid
	cmp	w0, #'9'
	bgt	parse_error_invalid
	// consume first exponent digit
	add	x21, x21, #1
	// consume remaining exponent digits
1exp_loop:
	cmp	x21, x20
	bhs	num_done
	add	x0, x19, x21
	ldrb	w0, [x0]
	cmp	w0, #'0'
	blt	num_done
	cmp	w0, #'9'
	bgt	num_done
	add	x21, x21, #1
	b	1exp_loop
num_exp_check2:
	// no exponent
	b	num_done
num_done:
	ret

// ------------------------------------------------------------
// Error handling output
// ------------------------------------------------------------
error:
	// write "error at "
	mov	x0, #1
	adrp	x1, err_prefix
	add	x1, x1, :lo12:err_prefix
	mov	x2, #error_prefix_len
	mov	x8, #64
	svc	#0
	// write error offset (x22)
	// if x22 not set, default to input length
	cmp	x23, #0
	bne	write_number
	// no error flag set (should not happen), use length
	adrp	x0, input_len
	add	x0, x0, :lo12:input_len
	ldr	x22, [x0]
write_number:
	mov	x0, x22
	cmp	x0, #0
	bne	conv_loop
	// write single '0'
	adrp	x1, zero_char
	add	x1, x1, :lo12:zero_char
	mov	x2, #1
	mov	x8, #64
	svc	#0
	b	write_nl
conv_loop:
	// convert x0 to decimal, store reversed in num_buf
	mov	x5, #0          // digit count
1c:	udiv	x6, x0, #10
	mul	x7, x6, #10
	sub	x7, x0, x7     // remainder
	adrp	x8, num_buf
	add	x8, x8, :lo12:num_buf
	add	x8, x8, x5
	add	x7, x7, #'0'
	strb	w7, [x8]
	add	x5, x5, #1
	mov	x0, x6
	cmp	x0, #0
	bne	1b
	// write digits in correct order
	mov	x6, x5
2c:	subs	x6, x6, #1
	blt	write_nl
	adrp	x7, num_buf
	add	x7, x7, :lo12:num_buf
	add	x7, x7, x6
	mov	x0, #1
	mov	x1, x7
	mov	x2, #1
	mov	x8, #64
	svc	#0
	b	2b
write_nl:
	// write newline
	mov	x0, #1
	adrp	x1, nl_char
	add	x1, x1, :lo12:nl_char
	mov	x2, #1
	mov	x8, #64
	svc	#0
	// exit with status 1
	mov	x0, #1
	mov	x8, #93
	svc	#0

// ------------------------------------------------------------
// Success output
// ------------------------------------------------------------
success:
	// write "ok\n"
	mov	x0, #1
	adrp	x1, ok_msg
	add	x1, x1, :lo12:ok_msg
	mov	x2, #ok_len
	mov	x8, #64
	svc	#0
	// exit 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// ------------------------------------------------------------
	// Data
	// ------------------------------------------------------------
	.section .rodata
ok_msg:
	.ascii "ok\n"
ok_len = . - ok_msg

err_prefix:
	.ascii "error at "
error_prefix_len = . - err_prefix

nl_char:
	.byte 0x0A
zero_char:
	.byte '0'

lit_true:
	.ascii "true"
lit_false:
	.ascii "false"
lit_null:
	.ascii "null"

	.bss
	.align 8
input_buf:
	.zero 65536
input_len:
	.quad 0
	// no additional bss needed; num_buf is in .bss as well
num_buf:
	.zero 20
	// end of file
