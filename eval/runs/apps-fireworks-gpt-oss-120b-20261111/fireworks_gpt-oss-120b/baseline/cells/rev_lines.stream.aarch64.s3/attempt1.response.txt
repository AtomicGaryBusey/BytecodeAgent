.globl _start
.equ BUFFER_SIZE, 65536
.equ OUT_BUF_SIZE, 131072

.text
_start:
    // load addresses of buffers
    adrp    x9, input_buf
    add     x9, x9, :lo12:input_buf          // x9 = input buffer base
    adrp    x13, output_buf
    add     x13, x13, :lo12:output_buf        // x13 = output buffer base

    // in_len = 0
    mov     x10, #0

read_loop:
    // x0 = fd (0)
    mov     x0, #0
    // x1 = buffer + in_len
    add     x1, x9, x10
    // x2 = remaining capacity
    mov     x2, #BUFFER_SIZE
    sub     x2, x2, x10
    // syscall read
    mov     x8, #63
    svc     #0
    cbz     x0, read_done                     // x0 == 0 => EOF
    // add bytes read to in_len
    add     x10, x10, x0
    // continue if buffer not full
    cmp     x10, #BUFFER_SIZE
    b.lt    read_loop
read_done:

    // set pointers for processing
    adrp    x11, input_buf
    add     x11, x11, :lo12:input_buf          // src = input_buf
    add     x12, x11, x10                     // src_end = input_buf + in_len
    mov     x14, x13                          // out_ptr = output_buf base

process_loop:
    cmp     x11, x12
    b.ge    write_output                      // src >= end -> done

    // find newline
    mov     x15, x11                          // tmp = src
find_nl:
    cmp     x15, x12
    b.ge    no_nl_found
    ldrb    w0, [x15]
    cmp     w0, #0x0A
    beq     nl_found
    add     x15, x15, #1
    b       find_nl
nl_found:
    sub     x5, x15, x11                      // line_len = tmp - src
    b       copy_reverse
no_nl_found:
    sub     x5, x12, x11                      // line_len = src_end - src

copy_reverse:
    // copy reversed line bytes
    mov     x6, x5                            // i = line_len
rev_loop:
    cbz     x6, rev_done
    sub     x7, x6, #1
    add     x8, x11, x7
    ldrb    w0, [x8]
    strb    w0, [x14], #1                     // store and post-increment out_ptr
    sub     x6, x6, #1
    b       rev_loop
rev_done:
    // store newline
    mov     w0, #0x0A
    strb    w0, [x14], #1

    // advance src past newline if present
    cmp     x15, x12
    b.ge    src_done
    add     x11, x15, #1
    b       process_loop
src_done:
    mov     x11, x12
    b       process_loop

write_output:
    // out_len = out_ptr - output_buf
    sub     x0, x14, x13
    cbz     x0, exit_prog

    mov     x1, x13          // buf = output_buf
    mov     x2, x0           // remaining = out_len
write_loop:
    cmp     x2, #0
    beq     exit_prog
    mov     x0, #1           // fd = stdout
    mov     x8, #64          // sys_write
    svc     #0
    // x0 = bytes_written
    sub     x2, x2, x0
    add     x1, x1, x0
    b       write_loop

exit_prog:
    mov     x0, #0
    mov     x8, #93          // sys_exit
    svc     #0

.bss
.align 4
input_buf:
    .zero BUFFER_SIZE
output_buf:
    .zero OUT_BUF_SIZE
