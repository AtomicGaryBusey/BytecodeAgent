.globl _start
.equ BUFFER_SIZE, 65536

.section .bss
.align 16
buffer: .zero BUFFER_SIZE
outbuf: .zero 1

.section .text
_start:
    # r12 = total length read so far
    xor r12, r12                # total_len = 0
    lea rsi, [rip + buffer]     # rsi = write pointer
    mov rdx, BUFFER_SIZE        # rdx = remaining space

read_loop:
    mov rdi, 0                  # fd = stdin
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done                # rax == 0 => EOF
    add rsi, rax                # advance write pointer
    sub rdx, rax                # decrease remaining space
    add r12, rax                # total_len += bytes read
    jmp read_loop

read_done:
    # r12 = total_len
    lea rdi, [rip + buffer]     # rdi = cur pointer (start of buffer)
    lea rsi, [rip + buffer]     # rsi = start pointer
    add rsi, r12                # rsi = end pointer (start + total_len)

process_loop:
    cmp rdi, rsi
    jae exit_program            # cur >= end -> done

    # Find next newline
    mov rbx, rdi                # rbx = scan pointer
scan_newline:
    cmp rbx, rsi
    jae no_newline              # reached end without newline
    mov al, [rbx]
    cmp al, 0x0A
    je newline_found
    inc rbx
    jmp scan_newline

newline_found:
    # line_len = rbx - rdi (excluding newline)
    mov rcx, rbx
    sub rcx, rdi                # rcx = line_len
    jmp have_line_len

no_newline:
    # line_len = rsi - rdi (rest of data)
    mov rcx, rsi
    sub rcx, rdi                # rcx = line_len

have_line_len:
    # rdx will point to end of line (cur + line_len)
    lea rdx, [rdi + rcx]        # rdx = line_end (first byte after line)

rev_loop:
    cmp rcx, 0
    je rev_done
    dec rdx                     # point to previous byte
    mov al, [rdx]               # load byte to output
    # write single byte
    mov byte [rip + outbuf], al
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    lea rsi, [rip + outbuf]
    mov rdx, 1
    syscall
    dec rcx
    jmp rev_loop

rev_done:
    # write newline
    mov byte [rip + outbuf], 0x0A
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    lea rsi, [rip + outbuf]
    mov rdx, 1
    syscall

    # advance cur pointer past processed line and optional newline
    cmp rbx, rsi
    je advance_done_no_nl       # no newline (rbx == end)
    # newline was found; rbx points to newline byte
    inc rbx                     # skip newline
advance_done_no_nl:
    mov rdi, rbx                # cur = new position
    jmp process_loop

exit_program:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
