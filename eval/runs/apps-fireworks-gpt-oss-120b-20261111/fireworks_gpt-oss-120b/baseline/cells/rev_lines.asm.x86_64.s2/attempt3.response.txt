.global _start

.equ BUFFER_SIZE, 65536

.section .bss
.align 16
buffer: .zero BUFFER_SIZE
outbuf: .zero 1

.section .text
_start:
    # read entire input into buffer
    xor r12, r12                # total_len = 0
    lea rsi, [rip + buffer]     # rsi = write pointer
    mov rdx, BUFFER_SIZE        # remaining space

read_loop:
    mov rdi, 0                  # fd = stdin
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done                # rax == 0 -> EOF
    add rsi, rax                # advance write pointer
    sub rdx, rax                # decrease remaining
    add r12, rax                # total_len += bytes read
    jmp read_loop

read_done:
    # r12 = total_len
    lea rsi, [rip + buffer]     # cur = start of buffer
    lea r9, [rip + buffer]      # r9 = start of buffer
    add r9, r12                 # r9 = start + total_len

process_loop:
    cmp rsi, r9
    jae exit_program            # cur >= end -> done

    # scan for newline
    mov r8, rsi                 # r8 = scan pointer

scan_loop:
    cmp r8, r9
    jae no_newline              # reached end without newline
    mov al, byte [r8]
    cmp al, 0x0A
    je newline_found
    inc r8
    jmp scan_loop

newline_found:
    # line_len = r8 - rsi (excluding newline)
    mov rbx, r8
    sub rbx, rsi                # rbx = line_len
    jmp have_len

no_newline:
    # line_len = r9 - rsi (rest of data)
    mov rbx, r9
    sub rbx, rsi                # rbx = line_len

have_len:
    # reverse and write the line
    mov r10, rsi                # r10 = line start
    mov rcx, rbx                # rcx = remaining bytes to output

rev_loop:
    test rcx, rcx
    jz rev_done
    dec rcx
    mov al, byte [r10 + rcx]
    mov byte [rip + outbuf], al
    mov rdi, 1                  # stdout
    lea rsi, [rip + outbuf]
    mov rdx, 1
    mov rax, 1                  # sys_write
    syscall
    jmp rev_loop

rev_done:
    # write newline
    mov al, 0x0A
    mov byte [rip + outbuf], al
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, 1
    mov rax, 1
    syscall

    # advance cur pointer past processed line (and newline if present)
    mov rsi, r8
    cmp r8, r9
    jae process_loop            # no newline, reached end
    inc rsi                     # skip newline
    jmp process_loop

exit_program:
    mov rdi, 0
    mov rax, 60                 # sys_exit
    syscall
