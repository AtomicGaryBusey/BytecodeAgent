.global _start

.equ BUFFER_SIZE, 65536

.section .bss
.align 16
buffer: .zero BUFFER_SIZE
outbuf: .zero BUFFER_SIZE

.section .rodata
newline: .byte 0x0A

.section .text
_start:
    ; initialize pointers and counters
    xor r13, r13                 # total length = 0
    lea rbx, [rip + buffer]      # write pointer = buffer start
    mov r12, BUFFER_SIZE         # remaining space

read_loop:
    mov rdi, 0                   # fd = stdin
    mov rsi, rbx                 # buffer pointer
    mov rdx, r12                 # max bytes to read
    mov rax, 0                   # sys_read
    syscall
    test rax, rax
    jz read_done                 # rax == 0 -> EOF
    add rbx, rax                 # advance write pointer
    sub r12, rax                 # decrease remaining space
    add r13, rax                 # total length += bytes read
    jmp read_loop

read_done:
    ; r13 = total length, rsi = start of buffer, rdx = end pointer
    lea rsi, [rip + buffer]      # cur = buffer start
    lea rdx, [rip + buffer + r13]# end = buffer start + total length

process_loop:
    cmp rsi, rdx
    jae exit_program             # cur >= end -> done

    ; remaining = end - cur
    mov rax, rdx
    sub rax, rsi                 # rax = remaining bytes

    ; find newline (0x0A) in remaining bytes
    mov al, 0x0A
    mov rdi, rsi                 # scan pointer
    mov rcx, rax                 # rcx = remaining
    repne scasb
    jnz no_newline               # ZF=0 -> not found

    ; newline found: line_len = (rdi-1) - rsi
    lea rbx, [rdi - 1]           # pointer to byte before newline
    sub rbx, rsi                 # rbx = line length
    jmp have_len

no_newline:
    ; no newline: line_len = remaining
    mov rbx, rax                 # rbx = line length (remaining)

have_len:
    ; reverse line of length rbx from [rsi] into outbuf
    test rbx, rbx
    jz skip_copy

    lea r8, [rsi + rbx - 1]      # src = rsi + line_len - 1
    lea r9, [rip + outbuf]       # dst = outbuf
    mov rcx, rbx

rev_copy_loop:
    mov al, byte ptr [r8]
    mov byte ptr [r9], al
    dec r8
    inc r9
    dec rcx
    jnz rev_copy_loop

skip_copy:
    ; write reversed line
    mov rdi, 1                   # fd = stdout
    lea rsi, [rip + outbuf]
    mov rdx, rbx                 # length
    mov rax, 1                   # sys_write
    syscall

    ; write newline
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall

    ; advance cur pointer
    mov rsi, rdi                 # rdi after scasb points after newline or at end
    jmp process_loop

exit_program:
    mov rdi, 0                   # status 0
    mov rax, 60                  # sys_exit
    syscall
