	.global _start
	.text
_start:
	// ---------- read all input ----------
	// total length will be in x3, pointer will be in x9, remaining bytes in x10
	adr	x9, input_buf
	mov	x3, #0                // total length
read_loop:
	// x3 = bytes already read, x9 = current write pointer
	mov	x8, #63              // sys_read
	mov	x0, #0               // fd = stdin
	mov	x1, x9               // buffer
	mov	x2, #65536
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x3, x3, x0
	add	x9, x9, x0
	b	read_loop
read_done:
	// x3 = total input length
	// set up initial pointers
	adr	x9, input_buf        // current pointer
	mov	x10, x3              // remaining bytes
	adr	x14, stack_buf       // stack base
	mov	x12, #0              // stack count
	mov	x13, #0              // line_error flag
	// flag whether any input existed
	cmp	x3, #0
	mov	x31, #0
