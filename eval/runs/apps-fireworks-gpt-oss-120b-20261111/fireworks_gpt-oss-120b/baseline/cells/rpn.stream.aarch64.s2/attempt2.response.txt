.global _start
.text
_start:
    // read all input into input_buf
    adr     x9, input_buf
    mov     x3, #0                // total bytes read
read_loop:
    mov     x8, #63               // sys_read
    mov     x0, #0                // fd = stdin
    mov     x1, x9                // buffer
    mov     x2, #65536
    svc     #0
    cmp     x0, #0
    beq     read_done
    add     x3, x3, x0
    add     x9, x9, x0
    b       read_loop
read_done:
    // x3 = total input length
    adr     x12, input_buf        // current pointer (p)
    mov     x13, x3               // remaining bytes (len)

line_loop:
    cbz     x13, exit_success

    // per‑line initialisation
    adr     x20, stack_buf        // stack base
    mov     x21, #0               // stack count
    mov     x22, #0               // line_error flag (0 = ok)

line_proc:
    // end of line ?
    cbz     x13, line_end
    ldrb    w0, [x12]
    cmp     w0, #0x0A            // newline ?
    beq     line_end

    // skip spaces / tabs
    cmp     w0, #0x20
    beq     skip_sep
    cmp     w0, #0x09
    beq     skip_sep

    // token start
    cmp     w0, #0x2D            // '-'
    beq     token_minus_start
    // digit ?
    cmp     w0, #0x30
    blt     token_invalid
    cmp     w0, #0x39
    bgt     token_invalid

    // --- positive number ---
    mov     x23, #0               // accumulator
    mov     x24, #0               // digit count
parse_number:
    // w0 is digit
    mov     x0, #10
    mul     x23, x23, x0
    sub     w0, w0, #0x30
    add     x23, x23, x0

    // advance pointer
    add     x12, x12, #1
    sub     x13, x13, #1
    add     x24, x24, #1

    // next character ?
    cbz     x13, number_done
    ldrb    w0, [x12]
    cmp     w0, #0x30
    blt     number_done
    cmp     w0, #0x39
    bgt     number_done
    b       parse_number
number_done:
    cmp     x24, #19
    bge     token_invalid
    // push value
    lsl     x25, x21, #3
    add     x26, x20, x25
    str     x23, [x26]
    add     x21, x21, #1
    b       line_proc

token_minus_start:
    // lookahead to decide number vs operator
    cmp     x13, #1
    blt     op_minus
    ldrb    w1, [x12, #1]
    cmp     w1, #0x30
    blt     op_minus
    cmp     w1, #0x39
    bgt     op_minus

    // --- negative number ---
    // consume '-'
    add     x12, x12, #1
    sub     x13, x13, #1
    // parse digits
    mov     x23, #0
    mov     x24, #0
parse_neg_number:
    ldrb    w0, [x12]                // digit
    mov     x0, #10
    mul     x23, x23, x0
    sub     w0, w0, #0x30
    add     x23, x23, x0

    add     x12, x12, #1
    sub     x13, x13, #1
    add     x24, x24, #1

    cbz     x13, neg_number_done
    ldrb    w0, [x12]
    cmp     w0, #0x30
    blt     neg_number_done
    cmp     w0, #0x39
    bgt     neg_number_done
    b       parse_neg_number
neg_number_done:
    cmp     x24, #19
    bge     token_invalid
    // negate
    neg     x23, x23
    // push
    lsl     x25, x21, #3
    add     x26, x20, x25
    str     x23, [x26]
    add     x21, x21, #1
    b       line_proc

op_minus:
    // token is operator '-'
    mov     w0, #0x2D
    b       operator_common

// other operators will be reached directly with their ASCII in w0
operator_common:
    // need at least two stack items
    cmp     x21, #2
    blt     set_error

    // pop b
    sub     x21, x21, #1
    lsl     x25, x21, #3
    add     x26, x20, x25
    ldr     x6, [x26]               // b

    // pop a
    sub     x21, x21, #1
    lsl     x25, x21, #3
    add     x26, x20, x25
    ldr     x7, [x26]               // a

    // dispatch on operator
    cmp     w0, #0x2B               // '+'
    beq     op_add
    cmp     w0, #0x2D               // '-'
    beq     op_sub
    cmp     w0, #0x2A               // '*'
    beq     op_mul
    cmp     w0, #0x2F               // '/'
    beq     op_div
    cmp     w0, #0x25               // '%'
    beq     op_mod
    // unknown -> error
    b       set_error

op_add:
    add     x8, x7, x6
    b       push_result
op_sub:
    sub     x8, x7, x6
    b       push_result
op_mul:
    mul     x8, x7, x6
    b       push_result
op_div:
    // divisor zero ?
    cmp     x6, #0
    beq     set_error
    // overflow a = INT64_MIN, b = -1 ?
    mov     x0, #0x8000000000000000
    cmp     x7, x0
    bne     div_normal
    cmp     x6, #-1
    beq     set_error
div_normal:
    sdiv    x8, x7, x6
    b       push_result
op_mod:
    // divisor zero ?
    cmp     x6, #0
    beq     set_error
    // overflow a = INT64_MIN, b = -1 ?
    mov     x0, #0x8000000000000000
    cmp     x7, x0
    bne     mod_normal
    cmp     x6, #-1
    beq     set_error
mod_normal:
    srem    x8, x7, x6
