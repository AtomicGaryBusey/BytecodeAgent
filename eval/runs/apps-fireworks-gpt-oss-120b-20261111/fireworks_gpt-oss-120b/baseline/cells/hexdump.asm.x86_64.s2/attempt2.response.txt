.globl _start

.bss
	.align 16
input_buf:
	.zero 65536
line_buf:
	.zero 80

.section .rodata
hex_tbl:
	.ascii "0123456789abcdef"

.text
_start:
	# read all input into input_buf
	xor rbx, rbx            # total_len = 0
read_loop:
	mov rax, 0              # sys_read
	mov rdi, 0              # fd 0
	lea rsi, [input_buf]    # buffer base
	add rsi, rbx            # buf = input_buf + total_len
	mov rdx, 65536
	sub rdx, rbx            # count = remaining space
	test rdx, rdx
	je read_done
	syscall
	cmp rax, 0
	je read_done
	add rbx, rax            # total_len += bytes_read
	jmp read_loop
read_done:
	mov r12, rbx            # r12 = total_len

	# if no data, skip to final length output
	xor rcx, rcx            # offset = 0
	cmp r12, 0
	je final_len

process_row:
	# r12 = total_len, rcx = offset
	# compute remaining bytes in this row
	mov r8, r12
	sub r8, rcx             # r8 = total_len - offset
	mov r9, 16              # row_len = 16
	cmp r8, 16
	jae row_len_ok
	mov r9, r8              # row_len = remaining
row_len_ok:

	# build line in line_buf
	lea rsi, [line_buf]

	# offset hex (8 digits)
	mov rdx, rcx
	mov r10, 8
offset_hex_loop:
	mov rax, rdx
	shr rax, 60
	and rax, 0xF
	movzx eax, byte ptr [hex_tbl + rax]
	mov byte ptr [rsi], al
	inc rsi
	shl rdx, 4
	dec r10
	jnz offset_hex_loop

	# two spaces after offset
	mov byte ptr [rsi], ' '
	inc rsi
	mov byte ptr [rsi], ' '
	inc rsi

	# pointer to row data
	lea rbx, [input_buf]
	add rbx, rcx            # rbx = input_buf + offset

	# byte/space columns (16 slots)
	xor r10, r10            # i = 0
byte_loop:
	cmp r10, 16
	je after_bytes
	cmp r10, r9
	jb byte_present
	# missing byte: three spaces
	mov byte ptr [rsi], ' '
	inc rsi
	mov byte ptr [rsi], ' '
	inc rsi
	mov byte ptr [rsi], ' '
	inc rsi
	jmp after_byte_common
byte_present:
	# load byte
	mov al, byte ptr [rbx + r10]
	# high nibble
	mov dl, al
	shr dl, 4
	and dl, 0xF
	movzx edx, byte ptr [hex_tbl + rdx]
	mov byte ptr [rsi], dl
	inc rsi
	# low nibble
	mov dl, al
	and dl, 0xF
	movzx edx, byte ptr [hex_tbl + rdx]
	mov byte ptr [rsi], dl
	inc rsi
	# trailing space
	mov byte ptr [rsi], ' '
	inc rsi
after_byte_common:
	inc r10
	cmp r10, 8
	jne byte_loop
	# extra space after 8th slot
	mov byte ptr [rsi], ' '
	inc rsi
	jmp byte_loop
after_bytes:
	# space and '|' before ASCII
	mov byte ptr [rsi], ' '
	inc rsi
	mov byte ptr [rsi], '|'
	inc rsi

	# ASCII representation
	xor r10, r10            # i = 0
ascii_loop:
	cmp r10, r9
	je after_ascii
	mov al, byte ptr [rbx + r10]
	cmp al, 0x20
	jb not_printable
	cmp al, 0x7e
	jg not_printable
	# printable
	mov byte ptr [rsi], al
	inc rsi
	jmp ascii_next
not_printable:
	mov byte ptr [rsi], '.'
	inc rsi
ascii_next:
	inc r10
	jmp ascii_loop
after_ascii:
	# closing '|' and newline
	mov byte ptr [rsi], '|'
	inc rsi
	mov byte ptr [rsi], 0x0a
	inc rsi

	# write the line
	push rcx                     # preserve offset across syscall
	lea rdx, [line_buf]
	mov rax, rsi
	sub rax, rdx                 # length = rsi - line_buf
	mov rdi, 1                   # fd 1
	mov rsi, rdx
	mov rdx, rax
	mov rax, 1                   # sys_write
	syscall
	pop rcx

	# advance to next row
	add rcx, 16
	cmp rcx, r12
	jb process_row

final_len:
	# output total length as 8 hex digits + newline
	lea rsi, [line_buf]
	mov rdx, r12                 # length value
	mov r10, 8
len_hex_loop:
	mov rax, rdx
	shr rax, 60
	and rax, 0xF
	movzx eax, byte ptr [hex_tbl + rax]
	mov byte ptr [rsi], al
	inc rsi
	shl rdx, 4
	dec r10
	jnz len_hex_loop
	# newline
	mov byte ptr [rsi], 0x0a
	inc rsi

	# write length line
	lea rdx, [line_buf]
	mov rax, rsi
	sub rax, rdx
	mov rdi, 1
	mov rsi, rdx
	mov rdx, rax
	mov rax, 1
	syscall

	# exit
	mov rax, 60
	xor rdi, rdi
	syscall
