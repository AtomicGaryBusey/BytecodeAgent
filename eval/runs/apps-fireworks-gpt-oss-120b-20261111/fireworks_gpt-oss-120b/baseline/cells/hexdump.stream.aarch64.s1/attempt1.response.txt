.section .bss
    .align 8
buf:        .zero   65536
linebuf:    .space  80

.section .rodata
hex_digits: .ascii  "0123456789abcdef"

.text
    .global _start
    .p2align 2
_start:
    // x19 = buffer base
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, xzr               // len = 0

read_loop:
    // remaining = 65536 - len
    mov     x1, #65536
    sub     x2, x1, x20
    cbz     x2, read_done          // no space left

    // read(fd=0, buf+len, remaining)
    mov     x0, #0
    add     x1, x19, x20           // address
    mov     x8, #63
    svc     #0
    // x0 = bytes read
    cbz     x0, read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    // x20 = total length
    mov     x21, xzr               // offset = 0

print_loop:
    cmp     x21, x20
    bcs     print_done

    // n = min(16, len - offset)
    sub     x22, x20, x21           // remaining bytes
    mov     x23, #16
    cmp     x22, x23
    csel    x24, x22, x23, lt       // n in x24

    // line buffer pointer
    adrp    x25, linebuf
    add     x25, x25, :lo12:linebuf

    // ----- offset hex (8 digits) -----
    mov     x26, #8                // count
    mov     x27, x21               // offset value
offset_hex_loop:
    sub     x28, x26, #1
    lsl     x28, x28, #2           // shift = (count-1)*4
    lsr     x29, x27, x28
    and     x29, x29, #0xF
    adrp    x30, hex_digits
    add     x30, x30, :lo12:hex_digits
    add     x30, x30, x29
    ldrb    w0, [x30]
    strb    w0, [x25], #1
    subs    x26, x26, #1
    bne     offset_hex_loop

    // two spaces after offset
    mov     w0, #' '
    strb    w0, [x25], #1
    strb    w0, [x25], #1

    // ----- hex bytes (16 slots) -----
    mov     x26, #0                // i = 0
hex_bytes_loop:
    cmp     x26, #16
    bge     after_hex_bytes

    // decide whether to print byte or spaces
    cmp     x26, x24
    bge     hex_slot_spaces

    // print actual byte
    add     x0, x19, x21
    add     x0, x0, x26
    ldrb    w0, [x0]

    // high nibble
    and     x1, x0, #0xF0
    lsr     x1, x1, #4
    // low nibble
    and     x2, x0, #0xF

    // high char
    adrp    x3, hex_digits
    add     x3, x3, :lo12:hex_digits
    add     x3, x3, x1
    ldrb    w3, [x3]
    strb    w3, [x25], #1

    // low char
    adrp    x4, hex_digits
    add     x4, x4, :lo12:hex_digits
    add     x4, x4, x2
    ldrb    w4, [x4]
    strb    w4, [x25], #1

    // trailing space
    mov     w5, #' '
    strb    w5, [x25], #1

    b       hex_increment

hex_slot_spaces:
    // three spaces for missing byte
    mov     w0, #' '
    strb    w0, [x25], #1
    strb    w0, [x25], #1
    strb    w0, [x25], #1

hex_increment:
    add     x26, x26, #1
    // extra space after the 8th slot (i == 8)
    cmp     x26, #8
    bne     hex_bytes_loop
    mov     w0, #' '
    strb    w0, [x25], #1
    b       hex_bytes_loop

after_hex_bytes:
    // space then '|'
    mov     w0, #' '
    strb    w0, [x25], #1
    mov     w0, #'|'
    strb    w0, [x25], #1

    // ----- ASCII representation -----
    mov     x26, #0                // i = 0
ascii_loop:
    cmp     x26, x24
    bge     after_ascii

    add     x0, x19, x21
    add     x0, x0, x26
    ldrb    w0, [x0]

    // printable?
    cmp     w0, #0x20
    blt     ascii_not_printable
    cmp     w0, #0x7e
    bgt     ascii_not_printable
    // printable
    strb    w0, [x25], #1
    b       ascii_next

ascii_not_printable:
    mov     w0, #'.'
    strb    w0, [x25], #1

ascii_next:
    add     x26, x26, #1
    b       ascii_loop

after_ascii:
    // closing '|' and newline
    mov     w0, #'|'
    strb    w0, [x25], #1
    mov     w0, #10
    strb    w0, [x25], #1

    // write the line
    adr     x0, linebuf
    sub     x1, x25, x0           // length
    mov     x2, x1
    mov     x0, #1                // fd stdout
    mov     x8, #64
    svc     #0

    // next offset
    add     x21, x21, #16
    b       print_loop

print_done:
    // ----- final length line -----
    adrp    x25, linebuf
    add     x25, x25, :lo12:linebuf
    mov     x27, x20               // total length

    // offset hex (8 digits)
    mov     x26, #8
final_offset_hex:
    sub     x28, x26, #1
    lsl     x28, x28, #2
    lsr     x29, x27, x28
    and     x29, x29, #0xF
    adrp    x30, hex_digits
    add     x30, x30, :lo12:hex_digits
    add     x30, x30, x29
    ldrb    w0, [x30]
    strb    w0, [x25], #1
    subs    x26, x26, #1
    bne     final_offset_hex

    // newline
    mov     w0, #10
    strb    w0, [x25], #1

    // write it
    adr     x0, linebuf
    sub     x1, x25, x0
    mov     x2, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0

    // exit 0
    mov     x0, #0
    mov     x8, #93
    svc     #0
