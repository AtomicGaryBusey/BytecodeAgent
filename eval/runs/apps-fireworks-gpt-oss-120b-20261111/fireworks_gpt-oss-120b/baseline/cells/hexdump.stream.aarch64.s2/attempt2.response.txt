	.text
	.globl _start
	.p2align 2

// System call numbers
	.equ SYS_READ, 63
	.equ SYS_WRITE, 64
	.equ SYS_EXIT, 93

// Buffer sizes
	.equ BUF_SIZE, 65536
	.equ CHUNK_MAX, 4096

_start:
	// ---- load addresses into callee‑saved registers ----
	adrp	x19, buf                // x19 = &buf
	add	x19, x19, :lo12:buf

	adrp	x20, input_len          // x20 = &input_len
	add	x20, x20, :lo12:input_len

	adrp	x21, linebuf            // x21 = &linebuf
	add	x21, x21, :lo12:linebuf

	adrp	x22, hexchars           // x22 = &hexchars
	add	x22, x22, :lo12:hexchars

	// ---- read all input into buf ----
	// input_len is zero‑initialised in .bss
read_loop:
	ldr	w0, [x20]                // w0 = current length
	subs	x1, xzr, w0             // x1 = -length (for remaining calc)
