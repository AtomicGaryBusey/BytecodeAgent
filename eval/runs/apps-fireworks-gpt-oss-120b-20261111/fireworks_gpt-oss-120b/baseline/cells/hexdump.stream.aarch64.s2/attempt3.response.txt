	.text
	.globl _start
	.p2align 2

	// system call numbers
	.equ SYS_READ, 63
	.equ SYS_WRITE, 64
	.equ SYS_EXIT, 93

	// constants
	.equ BUF_SIZE, 65536
	.equ LINEBUF_SIZE, 80

_start:
	// x19 = &buf
	adrp	x19, buf
	add	x19, x19, :lo12:buf

	// x20 = &input_len
	adrp	x20, input_len
	add	x20, x20, :lo12:input_len

	// x21 = &linebuf
	adrp	x21, linebuf
	add	x21, x21, :lo12:linebuf

	// x22 = &hexchars
	adrp	x22, hexchars
	add	x22, x22, :lo12:hexchars

	// ---------- read loop ----------
read_loop:
	ldr	w3, [x20]                // current length (low 32)
	uxtw	x3, w3                    // zero‑extend to 64‑bit
	mov	x4, #BUF_SIZE
	sub	x5, x4, x3                // remaining space
	cbz	x5, read_done             // no space left

	add	x1, x19, x3              // buffer + length
	mov	x0, #0                    // fd = stdin
	mov	x2, x5                    // count = remaining
	mov	x8, #SYS_READ
	svc	#0
	cmp	x0, #0
	beq	read_done                 // EOF

	add	x3, x3, x0                // new length
	// store back (low 32 bits are enough)
	ldr	w4, =0xFFFFFFFF           // dummy to force 32‑bit store size
