.globl _start
.section .bss
.p2align 3
prog_buf:        .zero 65536
prog_len:        .zero 8
input_len:       .zero 8
input_offset:    .zero 8
stack_buf:       .zero 8000          # 1000 * 8 bytes for bracket stack
bracket_map_buf: .zero 262144        # 65536 * 4 bytes
tape:            .zero 30000

.section .text
_start:
    # ---- read whole stdin into prog_buf ----
    xor r12, r12                     # total_len = 0
    lea rsi, [rip + prog_buf]        # buffer pointer (temporary)

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r12, rax                     # total_len += rax
    add rsi, rax                     # advance buffer pointer
    jmp read_loop

read_done:
    mov [prog_len], r12

    # ---- split at first '!' ----
    lea rdi, [rip + prog_buf]        # base address of program buffer
    xor rcx, rcx                     # index i = 0
    xor rdx, rdx                     # found flag = 0

find_excl_loop:
    cmp rcx, r12
    jae find_excl_done
    mov al, [rdi + rcx]
    cmp al, '!'
    je  found_excl
    inc rcx
    jmp find_excl_loop

found_excl:
    mov rdx, 1                       # flag = found

find_excl_done:
    test rdx, rdx
    jz no_excl

    # exclamation found at rcx
    mov [prog_len], rcx              # program length = rcx
    # input_len = total_len - (rcx + 1)
    mov rax, r12
    sub rax, rcx
    dec rax
    mov [input_len], rax
    # input_offset = rcx + 1
    lea rax, [rcx + 1]
    mov [input_offset], rax
    jmp after_split

no_excl:
    xor rax, rax
    mov [input_len], rax
    mov [input_offset], r12

after_split:

    # ---- bracket matching (balance check) ----
    xor rcx, rcx          # i = 0
    xor rbx, rbx          # stack depth = 0

bracket_scan_loop:
    mov rax, [prog_len]
    cmp rcx, rax
    jae bracket_scan_done
    mov al, [prog_buf + rcx]
    cmp al, '['
    je  bracket_open
    cmp al, ']'
    je  bracket_close
    inc rcx
    jmp bracket_scan_loop

bracket_open:
    # push rcx onto stack_buf
    lea rdi, [rip + stack_buf]
    mov [rdi + rbx*8], rcx
    inc rbx
    inc rcx
    jmp bracket_scan_loop

bracket_close:
    test rbx, rbx
    jz bracket_error_unbalanced
    dec rbx
    lea rdi, [rip + stack_buf]
    mov rax, [rdi + rbx*8]                # opening index
    # map open -> close
    mov [rip + bracket_map_buf + rcx*4], eax
    # map close -> open
    mov [rip + bracket_map_buf + rax*4], ecx
    inc rcx
    jmp bracket_scan_loop

bracket_scan_done:
    test rbx, rbx
    jnz bracket_error_unbalanced
    jmp exec_start

bracket_error_unbalanced:
    mov rax, 60          # sys_exit
    mov rdi, 1           # status 1
    syscall

    # ---- execution ----
exec_start:
    xor r9, r9               # pc = 0
    lea r8, [rip + tape]     # data pointer
    xor r10, r10             # input_pos = 0

exec_loop:
    mov rax, [prog_len]
    cmp r9, rax
    jae exec_finish
    mov al, [prog_buf + r9]

    cmp al, '>'
    je  cmd_incptr
    cmp al, '<'
    je  cmd_decptr
    cmp al, '+'
    je  cmd_inc
    cmp al, '-'
    je  cmd_dec
    cmp al, '.'
    je  cmd_out
    cmp al, ','
    je  cmd_in
    cmp al, '['
    je  cmd_bracket_open
    cmp al, ']'
    je  cmd_bracket_close

    # ignore any other character
    inc r9
    jmp exec_loop

cmd_incptr:
    inc r8
    inc r9
    jmp exec_loop

cmd_decptr:
    dec r8
    inc r9
    jmp exec_loop

cmd_inc:
    inc byte ptr [r8]
    inc r9
    jmp exec_loop

cmd_dec:
    dec byte ptr [r8]
    inc r9
    jmp exec_loop

cmd_out:
    mov rax, 1               # sys_write
    mov rdi, 1               # stdout
    mov rsi, r8              # buffer = data pointer
    mov rdx, 1
    syscall
    inc r9
    jmp exec_loop

cmd_in:
    mov rax, [input_len]
    cmp r10, rax
    jae input_eof
    mov rcx, [input_offset]
    add rcx, r10
    mov al, [prog_buf + rcx]
    mov [r8], al
    inc r10
    jmp after_input

input_eof:
    mov byte ptr [r8], 0

after_input:
    inc r9
    jmp exec_loop

cmd_bracket_open:
    mov al, [r8]
    test al, al
    jnz open_not_zero
    # cell is zero -> jump after matching ]
    mov eax, [rip + bracket_map_buf + r9*4]
    inc rax
    mov r9, rax
    jmp exec_loop

open_not_zero:
    inc r9
    jmp exec_loop

cmd_bracket_close:
    mov al, [r8]
    test al, al
    jz close_is_zero
    # cell non-zero -> jump after matching [
    mov eax, [rip + bracket_map_buf + r9*4]
    inc rax
    mov r9, rax
    jmp exec_loop

close_is_zero:
    inc r9
    jmp exec_loop

exec_finish:
    mov rax, 60          # sys_exit
    xor rdi, rdi         # status 0
    syscall
