.global _start
.section .bss
prog_buf:               .space 65536
prog_len:               .quad 0
input_len:              .quad 0
input_offset:           .quad 0
input_pos:              .quad 0
stack_ptr:              .quad 0
stack:                  .space 8000          # 1000 * 8
bracket_map:            .space 262144        # 65536 * 4
tape:                   .space 30000
.section .text
_start:
    # ---- read whole stdin into prog_buf ----
    xor r12, r12                     # total_len = 0
    lea rsi, [rel prog_buf]          # buffer pointer
read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    mov rdx, 65536
    syscall
    test rax, rax
    je  read_done
    add r12, rax                     # total_len += rax
    add rsi, rax                     # advance buffer pointer
    jmp read_loop
read_done:
    mov [rel prog_len], r12

    # ---- split at first '!' ----
    lea rdi, [rel prog_buf]          # base address
    xor rcx, rcx                     # index i = 0
    xor rdx, rdx                     # found flag = 0
find_excl_loop:
    cmp rcx, r12
    jae find_excl_done
    mov al, byte ptr [rdi + rcx]
    cmp al, '!'
    je  found_excl
    inc rcx
    jmp find_excl_loop
found_excl:
    mov rdx, 1                       # found = 1
find_excl_done:
    test rdx, rdx
    jz  no_excl
    # exclamation found at rcx
    mov [rel prog_len], rcx          # program length = rcx
    # input_len = total_len - (rcx + 1)
    mov rax, r12
    sub rax, rcx
    dec rax
    mov [rel input_len], rax
    # input_offset = rcx + 1
    lea rax, [rcx + 1]
    mov [rel input_offset], rax
    jmp after_split
no_excl:
    # no '!' => input empty
    mov [rel input_len], rax         # rax is zero
    mov [rel input_offset], r12
after_split:

    # ---- bracket matching ----
    xor rcx, rcx                     # i = 0
    xor rbx, rbx                     # stack depth = 0
    mov [rel stack_ptr], rbx
bracket_scan_loop:
    mov rax, [rel prog_len]
    cmp rcx, rax
    jae bracket_scan_done
    mov al, byte ptr [rel prog_buf + rcx]
    cmp al, '['
    je  bracket_open
    cmp al, ']'
    je  bracket_close
    inc rcx
    jmp bracket_scan_loop
bracket_open:
    # push rcx onto stack
    mov rax, rcx
    mov rbx, [rel stack_ptr]
    lea rdi, [rel stack + rbx*8]
    mov [rdi], rax
    inc rbx
    mov [rel stack_ptr], rbx
    inc rcx
    jmp bracket_scan_loop
bracket_close:
    mov rbx, [rel stack_ptr]
    test rbx, rbx
    jz bracket_error_unbalanced
    dec rbx
    mov [rel stack_ptr], rbx
    lea rdi, [rel stack + rbx*8]
    mov rax, [rdi]                   # opening index
    # map open -> close
    lea rsi, [rel bracket_map + rcx*4]
    mov dword ptr [rsi], eax
    # map close -> open
    lea rsi, [rel bracket_map + rax*4]
    mov dword ptr [rsi], ecx
    inc rcx
    jmp bracket_scan_loop
bracket_scan_done:
    mov rbx, [rel stack_ptr]
    test rbx, rbx
    jnz bracket_error_unbalanced
    jmp exec_start
bracket_error_unbalanced:
    # exit status 1
    mov rax, 60
    mov rdi, 1
    syscall

    # ---- execution ----
exec_start:
    xor r9, r9                       # pc = 0
    lea rdx, [rel tape]              # data pointer
    xor r10, r10                     # input_pos = 0
exec_loop:
    mov rax, [rel prog_len]
    cmp r9, rax
    jae exec_finish
    mov al, byte ptr [rel prog_buf + r9]
    cmp al, '>'
    je  cmd_incptr
    cmp al, '<'
    je  cmd_decptr
    cmp al, '+'
    je  cmd_inc
    cmp al, '-'
    je  cmd_dec
    cmp al, '.'
    je  cmd_out
    cmp al, ','
    je  cmd_in
    cmp al, '['
    je  cmd_bracket_open
    cmp al, ']'
    je  cmd_bracket_close
    # ignore any other character
    inc r9
    jmp exec_loop

cmd_incptr:
    inc rdx
    inc r9
    jmp exec_loop
cmd_decptr:
    dec rdx
    inc r9
    jmp exec_loop
cmd_inc:
    inc byte ptr [rdx]
    inc r9
    jmp exec_loop
cmd_dec:
    dec byte ptr [rdx]
    inc r9
    jmp exec_loop
cmd_out:
    mov rax, 1                       # sys_write
    mov rdi, 1                       # stdout
    mov rsi, rdx                     # buffer = data pointer
    mov rdx, 1
    syscall
    inc r9
    jmp exec_loop
cmd_in:
    mov rax, [rel input_len]
    cmp r10, rax
    jae input_eof
    mov rcx, [rel input_offset]
    add rcx, r10
    mov al, byte ptr [rel prog_buf + rcx]
    mov byte ptr [rdx], al
    inc r10
    jmp after_input
input_eof:
    mov byte ptr [rdx], 0
after_input:
    inc r9
    jmp exec_loop
cmd_bracket_open:
    mov al, byte ptr [rdx]
    test al, al
    jnz open_not_zero
    # cell is zero -> jump after matching ]
    mov eax, dword ptr [rel bracket_map + r9*4]
    inc rax
    mov r9, rax
    jmp exec_loop
open_not_zero:
    inc r9
    jmp exec_loop
cmd_bracket_close:
    mov al, byte ptr [rdx]
    test al, al
    jz close_is_zero
    # cell non-zero -> jump after matching [
    mov eax, dword ptr [rel bracket_map + r9*4]
    inc rax
    mov r9, rax
    jmp exec_loop
close_is_zero:
    inc r9
    jmp exec_loop

exec_finish:
    mov rax, 60
    xor rdi, rdi
    syscall
