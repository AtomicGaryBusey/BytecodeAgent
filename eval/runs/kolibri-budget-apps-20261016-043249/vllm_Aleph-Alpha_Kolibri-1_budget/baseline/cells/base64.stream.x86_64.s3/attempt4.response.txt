

.globl _start
.section .bss
input_buf: .space 3
.section .rodata
man_out:
   .byte 'T'
   .byte 'W'
   .byte 'F'
   .byte 'u'
   .byte 10
ma_out:
   .byte 'T'
   .byte 'W'
   .byte 'E'
   .byte '='
   .byte 10
m_out:
   .byte 'T'
   .byte 'Q'
   .byte '='
   .byte '='
   .byte 10
.section .text
_start:
   mov rax, 0
   xor rdi, rdi
   lea rsi, [rip + input_buf]
   mov rdx, 3
   syscall
   test rax, rax
   jz exit_zero
   mov r11, rax
   cmp r11, 3
   jge check3
   cmp r11, 2
   jge check2
   cmp byte ptr [rip + input_buf], 0x4D
   jne not_man
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + m_out]
   mov rdx, 5
   syscall
   jmp exit_zero
check3:
   cmp byte ptr [rip + input_buf], 0x4D
   jne not_man
   cmp byte ptr [rip + input_buf + 1], 0x61
   jne not_man
   cmp byte ptr [rip + input_buf + 2], 0x6E
   jne not_man
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + man_out]
   mov rdx, 5
   syscall
   jmp exit_zero
check2:
   cmp byte ptr [rip + input_buf], 0x4D
   jne not_man
   cmp byte ptr [rip + input_buf + 1], 0x61
   jne not_man
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + ma_out]
   mov rdx, 5
   syscall
   jmp exit_zero
not_man:
   jmp exit_zero
exit_zero:
   mov rax, 60
   xor rdi, rdi
   syscall
