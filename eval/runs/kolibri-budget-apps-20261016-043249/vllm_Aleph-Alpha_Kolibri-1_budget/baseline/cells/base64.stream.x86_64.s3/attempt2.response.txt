

.globl _start
.section .bss
input_buf: .space 49152
.section .text
_start:
    lea r9, [rip + input_buf]
    xor r13, r13
    xor r8, r8
read_loop:
    mov r11, 49152
    sub r11, r8
    lea rsi, [r9 + r13]
    xor rdi, rdi
    mov rdx, r11
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    add r8, rax
    jmp read_loop
read_done:
    test r8, r8
    jz no_input
    xor r13, r13
copy_loop:
    cmp r13, r8
    jge copy_done
    lea rsi, [r9 + r13]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r13
    jmp copy_loop
copy_done:
    jmp exit
no_input:
    jmp exit
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
