

.global _start

.section .rodata
ok_msg:
    .ascii "ok\n"
error_msg:
    .ascii "error at 0\n"

.section .bss
input_buf:
    .space 65536

.section .text
_start:
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
    movz x3, #0
    movk x3, #16
    mov x4, #0
    mov x0, #0
    mov x8, #63
read_loop:
    add x1, x5, x4
    sub x2, x3, x4
    svc #0
    cmp x0, #0
    b.eq read_done
    add x4, x4, x0
    b read_loop
read_done:
    cmp x4, #0
    b.eq error_path
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
error_path:
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
