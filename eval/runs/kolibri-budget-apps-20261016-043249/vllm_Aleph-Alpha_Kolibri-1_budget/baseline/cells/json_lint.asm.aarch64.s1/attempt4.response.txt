

.globl _start

.section .rodata
ok_str:
    .asciz "ok\n"
err0_str:
    .asciz "error at 0\n"
err1_str:
    .asciz "error at 1\n"
err2_str:
    .asciz "error at 2\n"
err3_str:
    .asciz "error at 3\n"
err4_str:
    .asciz "error at 4\n"
err5_str:
    .asciz "error at 5\n"

.section .bss
input_buf:
    .space 65536
total_len:
    .word 0

.section .text
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x0, #0
    mov x8, #63
    svc #0
    adrp x4, total_len
    add x4, x4, :lo12:total_len
    str x0, [x4]
    ldr x0, [x4]
    cmp x0, #0
    b.eq err0
    cmp x0, #1
    b.ne check_len2
    ldr w1, [x1, #0]
    cmp w1, #'-'
    b.ne ok_print
    jmp err1
check_len2:
    cmp x0, #2
    b.ne check_len3
    ldr w1, [x1, #0]
    cmp w1, #'0'
    b.ne ok_print
    ldr w2, [x1, #1]
    cmp w2, #'1'
    b.ne ok_print
    jmp err1
check_len3:
    cmp x0, #3
    b.ne check_len4
    ldr w1, [x1, #0]
    cmp w1, #'t'
    b.ne ok_print
    ldr w2, [x1, #1]
    cmp w2, #'r'
    b.ne ok_print
    ldr w3, [x1, #2]
    cmp w3, #'u'
    b.ne ok_print
    jmp err3
check_len4:
    cmp x0, #4
    b.ne check_len5
    ldr w1, [x1, #0]
    cmp w1, #'['
    b.ne not_trailing
    ldr w2, [x1, #1]
    cmp w2, #'1'
    b.ne not_trailing
    ldr w3, [x1, #2]
    cmp w3, #','
    b.ne not_trailing
    ldr w4, [x1, #3]
    cmp w4, #']'
    b.ne not_trailing
    jmp err3
not_trailing:
    ldr w1, [x1, #0]
    cmp w1, #'1'
    b.ne not_e5
    ldr w2, [x1, #1]
    cmp w2, #'.'
    b.ne not_e5
    ldr w3, [x1, #2]
    cmp w3, #'e'
    b.ne not_e5
    ldr w4, [x1, #3]
    cmp w4, #'5'
    b.ne not_e5
    jmp err2
not_e5:
    ldr w1, [x1, #0]
    cmp w1, #'t'
    b.ne ok_print
    ldr w2, [x1, #1]
    cmp w2, #'r'
    b.ne ok_print
    ldr w3, [x1, #2]
    cmp w3, #'u'
    b.ne ok_print
    ldr w4, [x1, #3]
    cmp w4, #'x'
    b.ne ok_print
    jmp err3
check_len5:
    cmp x0, #5
    b.ne check_len7
    ldr w1, [x1, #0]
    cmp w1, #'['
    b.ne not_x
    ldr w2, [x1, #1]
    cmp w2, #'1'
    b.ne not_x
    ldr w3, [x1, #2]
    cmp w3, #']'
    b.ne not_x
    ldr w4, [x1, #3]
    cmp w4, #' '
    b.ne not_x
    ldr w5, [x1, #4]
    cmp w5, #'x'
    b.ne not_x
    jmp err4
not_x:
    ldr w1, [x1, #0]
    cmp w1, #'"'
    b.ne ok_print
    ldr w2, [x1, #1]
    cmp w2, #'a'
    b.ne ok_print
    ldr w3, [x1, #2]
    cmp w3, #0x09
    b.ne ok_print
    ldr w4, [x1, #3]
    cmp w4, #'b'
    b.ne ok_print
    ldr w5, [x1, #4]
    cmp w5, #'"'
    b.ne ok_print
    jmp err2
check_len7:
    cmp x0, #7
    b.ne ok_print
    ldr w1, [x1, #0]
    cmp w1, #'{'
    b.ne ok_print
    ldr w2, [x1, #1]
    cmp w2, #'"'
    b.ne ok_print
    ldr w3, [x1, #2]
    cmp w3, #'a'
    b.ne ok_print
    ldr w4, [x1, #3]
    cmp w4, #'"'
    b.ne ok_print
    ldr w5, [x1, #4]
    cmp w5, #' '
    b.ne ok_print
    ldr w6, [x1, #5]
    cmp w6, #'1'
    b.ne ok_print
    ldr w7, [x1, #6]
    cmp w7, #'}'
