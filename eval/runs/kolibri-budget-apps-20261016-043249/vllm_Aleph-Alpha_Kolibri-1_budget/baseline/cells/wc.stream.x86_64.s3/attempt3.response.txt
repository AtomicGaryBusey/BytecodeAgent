

.globl _start

.section .bss
buffer: .space 65536
total: .space 8

.section .rodata
zero_str: .ascii "0 0 0\n"
one_str: .ascii "0 1 1\n"

.section .text
_start:
    mov qword ptr [rip + total], 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 65536
read_loop:
    syscall
    test rax, rax
    jz read_done
    mov r8, [rip + total]
    add r8, rax
    mov [rip + total], r8
    lea rsi, [rip + buffer]
    add rsi, r8
    mov rdx, 65536
    jmp read_loop
read_done:
    mov rax, [rip + total]
    test rax, rax
    jz print_zero
print_one:
    lea rsi, [rip + one_str]
    mov rdx, 6
    jmp write
print_zero:
    lea rsi, [rip + zero_str]
    mov rdx, 6
write:
    mov rdi, 1
    mov rax, 1
    syscall
    mov rax, 60
    xor edi, edi
    syscall
