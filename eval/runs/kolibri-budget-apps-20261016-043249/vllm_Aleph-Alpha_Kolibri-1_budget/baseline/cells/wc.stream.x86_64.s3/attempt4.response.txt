

.globl _start

.section .bss
buffer: .space 65536
out_buf: .space 32

.section .text
print_num:
    test rax, rax
    jz print_num_zero
    mov r14, rsp
push_loop:
    test rax, rax
    jz pop_loop
    mov rbx, 10
    div rbx
    push rdx
    jmp push_loop
pop_loop:
    cmp rsp, r14
    je done
    pop rax
    add al, '0'
    mov [rsi + r15], al
    inc r15
    jmp pop_loop
done:
    ret
print_num_zero:
    mov al, '0'
    mov [rsi + r15], al
    inc r15
    ret

_start:
    xor r8, r8
    mov r13, 0

read_loop:
    mov rdi, 0
    lea rsi, [rip + buffer]
    add rsi, r13
    mov rdx, 65536
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add r13, rax
    jmp read_loop

read_done:
    lea rsi, [rip + buffer]
    mov rcx, r8
    xor r9, r9
    xor r10, r10
    xor r12, r12

count_loop:
    test rcx, rcx
    jz count_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x0A
    je has_newline
    cmp al, 0x20
    je is_whitespace
    cmp al, 0x09
    je is_whitespace
    cmp al, 0x0B
    je is_whitespace
    cmp al, 0x0C
    je is_whitespace
    cmp al, 0x0D
    je is_whitespace
    test r12, r12
    jz start_word
    jmp next_byte
has_newline:
    inc r9
is_whitespace:
    test r12, r12
    jz no_word
    xor r12, r12
    jmp next_byte
no_word:
    jmp next_byte
start_word:
    inc r10
    mov r12, 1
    jmp next_byte
next_byte:
    inc rsi
    dec rcx
    jnz count_loop

count_done:
    lea rsi, [rip + out_buf]
    xor r15, r15

    mov rax, r9
    call print_num
    mov al, ' '
    mov [rsi + r15], al
    inc r15

    mov rax, r10
    call print_num
    mov al, ' '
    mov [rsi + r15], al
    inc r15

    mov rax, r8
    call print_num
    mov al, '\n'
    mov [rsi + r15], al
    inc r15

    mov rdi, 1
    mov rdx, r15
    mov rax, 1
    syscall

    mov rax, 60
    xor edi, edi
    syscall
