

.section .bss
input_buf .space 49152

.section .text
.globl _start
_start:
    lea rsi, [rip + input_buf]
    xor r12, r12
.read_loop:
    mov rdx, 49152
    sub rdx, r12
    xor rax, rax
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    add rsi, rax
    jmp .read_loop
.read_done:
    lea r13, [rip + input_buf]
    test r12, r12
    jz .empty_exit
    mov rsi, r13
    mov rdx, r12
    mov rax, 1
    mov rdi, 1
    syscall
    jmp .exit
.empty_exit:
    jmp .exit
.exit:
    mov rax, 60
    xor rdi, rdi
    syscall
