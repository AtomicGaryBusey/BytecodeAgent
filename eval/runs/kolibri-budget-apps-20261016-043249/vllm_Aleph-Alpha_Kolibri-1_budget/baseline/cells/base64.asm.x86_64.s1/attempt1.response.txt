

.section .rodata
alphabet .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.section .data
newline .byte 10
.section .bss
input_buf .space 49152
out_buf .space 66400
.section .text
.globl _start
_start:
    lea rsi, [rip + input_buf]
    xor r12, r12
    mov rdi, 0
.read_loop:
    mov rdx, 49152
    sub rdx, r12
    xor rax, rax
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    add rsi, rax
    jmp .read_loop
.read_done:
    lea r13, [rip + input_buf]
    lea r15, [rip + out_buf]
    lea r10, [rip + out_buf]
    xor r8, r8
    cmp r12, 3
    jl .handle_remainder
.encoding_loop:
    movzx rax, byte ptr [r13]
    shr rax, 2
    and rax, 63
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    movzx rax, byte ptr [r13]
    and rax, 3
    shl rax, 4
    movzx rbx, byte ptr [r13 + 1]
    shr rbx, 4
    and rbx, 15
    or rax, rbx
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    movzx rax, byte ptr [r13 + 1]
    and rax, 15
    shl rax, 2
    movzx rbx, byte ptr [r13 + 2]
    shr rbx, 6
    and rbx, 3
    or rax, rbx
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    movzx rax, byte ptr [r13 + 2]
    and rax, 63
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    add r13, 3
    sub r12, 3
    jmp .encoding_loop
.handle_remainder:
    cmp r12, 2
    je .handle_two
    cmp r12, 1
    je .handle_one
    jmp .write_output
.handle_one:
    movzx rax, byte ptr [r13]
    shr rax, 2
    and rax, 63
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    and rax, 3
    shl rax, 4
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    mov al, 61
    mov byte ptr [r10], al
    inc r10
    inc r8
    mov al, 61
    mov byte ptr [r10], al
    inc r10
    inc r8
    jmp .write_output
.handle_two:
    movzx rax, byte ptr [r13]
    shr rax, 2
    and rax, 63
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    movzx rax, byte ptr [r13]
    and rax, 3
    shl rax, 4
    movzx rbx, byte ptr [r13 + 1]
    shr rbx, 4
    and rbx, 15
    or rax, rbx
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    movzx rax, byte ptr [r13 + 1]
    and rax, 15
    shl rax, 2
    movzx edx, byte ptr [rip + alphabet + rax]
    mov byte ptr [r10], dl
    inc r10
    inc r8
    mov al, 61
    mov byte ptr [r10], al
    inc r10
    inc r8
    jmp .write_output
.write_output:
    test r8, r8
    jz .exit
    lea r14, [rip + out_buf]
    mov r12, r8
.write_loop:
    cmp r12, 76
    jge .write_full
    mov rdx, r12
    jmp .do_write
.write_full:
    mov rdx, 76
.do_write:
    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    syscall
    add r14, rdx
    sub r12, rdx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    test r12, r12
    jz .exit
    jmp .write_loop
.exit:
    mov rax, 60
    xor rdi, rdi
    syscall
