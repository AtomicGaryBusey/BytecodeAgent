

.global _start

.section .bss
in_buf: .space 65536

.section .rodata
newline: .byte 10

.section .text
_start:
    lea rsi, [rip + in_buf]
    mov rdi, 0
    mov rdx, 65536
    xor rax, rax
    syscall
    mov r12, rsi
    mov r13, rax
    test r13, r13
    jz exit
    xor r8, r8
    lea r15, [rip + newline]

outer_loop:
    mov r9, r8
scan_newline:
    cmp r9, r13
    jge no_newline
    mov al, byte ptr [r12 + r9]
    cmp al, 10
    je found_newline
    inc r9
    jmp scan_newline
found_newline:
    mov r14, r9
    mov r10, r14
    dec r10
print_loop:
    cmp r10, r8
    jl print_newline
    lea rsi, [r12 + r10]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r10
    jmp print_loop
print_newline:
    mov rsi, r15
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    mov r8, r14
    inc r8
    cmp r8, r13
    jge exit
    jmp outer_loop
no_newline:
    cmp r8, r13
    jge exit
    mov r10, r13
    dec r10
print_loop_final:
    cmp r10, r8
    jl after_final_newline
    lea rsi, [r12 + r10]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r10
    jmp print_loop_final
after_final_newline:
    mov rsi, r15
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    jmp exit
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
