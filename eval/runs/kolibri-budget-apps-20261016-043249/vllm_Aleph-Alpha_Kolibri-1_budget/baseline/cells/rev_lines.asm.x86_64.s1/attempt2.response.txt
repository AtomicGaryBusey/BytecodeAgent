

.global _start

.section .bss
in_buf: .space 65536

.section .rodata
nl: .byte 10

.section .text
_start:
   xor rax, rax
   mov rdi, 0
   mov rsi, offset in_buf
   mov rdx, 65536
   syscall
   mov r10, rax
   test r10, r10
   jz exit

   xor r8, r8
   mov r9, r8

scan_loop:
   cmp r8, r10
   jge end_of_input

   mov al, byte ptr [in_buf + r8]
   cmp al, 10
   je found_newline

   inc r8
   jmp scan_loop

found_newline:
   mov r12, r8
   dec r12
   cmp r12, r9
   jge print_bytes
   jmp after_line

print_bytes:
   mov al, byte ptr [in_buf + r12]
   mov rax, 1
   mov rdi, 1
   mov rsi, r12
   mov rdx, 1
   syscall
   dec r12
   cmp r12, r9
   jge print_bytes

after_line:
   lea rsi, [rip + nl]
   mov rax, 1
   mov rdi, 1
   mov rdx, 1
   syscall
   inc r8
   mov r9, r8
   jmp scan_loop

end_of_input:
   cmp r9, r10
   je exit
   mov r12, r10
   dec r12
   jmp print_bytes_last

print_bytes_last:
   mov al, byte ptr [in_buf + r12]
   mov rax, 1
   mov rdi, 1
   mov rsi, r12
   mov rdx, 1
   syscall
   dec r12
   cmp r12, r9
   jge print_bytes_last

after_line_last:
   lea rsi, [rip + nl]
   mov rax, 1
   mov rdi, 1
   mov rdx, 1
   syscall
   jmp exit

exit:
   mov rax, 60
   xor rdi, rdi
   syscall
