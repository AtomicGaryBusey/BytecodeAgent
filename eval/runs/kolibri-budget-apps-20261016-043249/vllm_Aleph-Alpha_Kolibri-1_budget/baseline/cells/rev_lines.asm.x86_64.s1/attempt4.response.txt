

.global _start

.section .rodata
newline: .byte 10

.section .bss
in_buf: .space 65536

.section .text
_start:
    lea r12, [rip + in_buf]
    xor rbx, rbx
read_loop:
    lea rsi, [r12 + rbx]
    mov rdi, 0
    mov rdx, 65536
    sub rdx, rbx
    test rdx, rdx
    jle read_done
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r13, rbx
    xor r8, r8
outer_loop:
    mov r9, r8
scan_newline:
    cmp r9, r13
    jge no_newline
    movzx eax, byte ptr [r12 + r9]
    cmp eax, 10
    je found_newline
    inc r9
    jmp scan_newline
found_newline:
    mov r14, r9
    mov r10, r14
    dec r10
print_loop:
    cmp r10, r8
    jl print_newline
    lea rsi, [r12 + r10]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r10
    jmp print_loop
print_newline:
    lea rsi, [rip + newline]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    inc r14
    mov r8, r14
    cmp r8, r13
    jge exit
    jmp outer_loop
no_newline:
    cmp r8, r13
    jge exit
    mov r10, r13
    dec r10
print_loop_final:
    cmp r10, r8
    jl after_final_newline
    lea rsi, [r12 + r10]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r10
    jmp print_loop_final
after_final_newline:
    lea rsi, [rip + newline]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    jmp exit
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
