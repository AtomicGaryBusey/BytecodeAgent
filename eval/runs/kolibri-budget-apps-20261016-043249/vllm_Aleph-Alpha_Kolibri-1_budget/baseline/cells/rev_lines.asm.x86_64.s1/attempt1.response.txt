

.global _start

.section .bss
in_buf: .space 65536

.section .rodata
nl: .byte 10

.section .text
_start:
   xor r8, r8
   xor rdi, rdi
_read_loop:
   lea rsi, [in_buf + r8]
   mov rdx, 1
   mov rax, 0
   syscall
   test rax, rax
   jz done_reading
   inc r8
   jmp _read_loop
done_reading:
   mov r10, r8
   xor r8, r8
   xor r9, r9
_find_newline:
   cmp r8, r10
   jge no_newline
   mov al, byte ptr [in_buf + r8]
   cmp al, 10
   je found_newline
   inc r8
   jmp _find_newline
found_newline:
   call process_line
   inc r8
   mov r9, r8
   jmp _find_newline
no_newline:
   cmp r9, r10
   jge done
   mov r8, r10
   call process_line
done:
   mov rax, 60
   xor rdi, rdi
   syscall

process_line:
   mov r12, r8
   sub r12, r9
   test r12, r12
   jz write_newline_only
   dec r12
write_reversed:
   lea r13, [in_buf + r9 + r12]
   mov rax, 1
   mov rdi, 1
   mov rsi, r13
   mov rdx, 1
   syscall
   dec r12
   test r12, r12
   jge write_reversed
write_newline_only:
   mov rax, 1
   mov rdi, 1
   mov rsi, nl
   mov rdx, 1
   syscall
   ret
