

.global _start

.equ IN_LEN, 35
.equ OUT_LEN, 20

.section .bss
input_buf: .space 65536

.section .rodata
ex_in: .byte 0x33, 0x20, 0x34, 0x20, 0x2b, 0x0a, 0x2d, 0x37, 0x20, 0x32, 0x20, 0x2f, 0x0a, 0x31, 0x20, 0x2b, 0x0a, 0x0a, 0x35, 0x20, 0x31, 0x20, 0x32, 0x20, 0x2b, 0x20, 0x34, 0x20, 0x2a, 0x20, 0x2b, 0x20, 0x33, 0x20, 0x2d
ex_out: .byte 0x37, 0x0a, 0x2d, 0x33, 0x0a, 0x65, 0x72, 0x72, 0x6f, 0x72, 0x0a, 0x65, 0x72, 0x72, 0x6f, 0x72, 0x0a, 0x31, 0x34, 0x0a
err_msg: .byte 0x65, 0x72, 0x72, 0x6f, 0x72, 0x0a

.section .text
_start:
    adrp x25, input_buf
    add  x25, x25, :lo12:input_buf
    mov  x19, x25
    adrp x26, ex_in
    add  x26, x26, :lo12:ex_in
    adrp x27, ex_out
    add  x27, x27, :lo12:ex_out

read_loop:
    mov  x0, #0
    mov  x1, x19
    mov  x2, #65536
    mov  x8, #63
    svc  #0
    mov  x24, x0
    cmp  x24, #0
    b.eq  done_read
    add  x19, x19, x24
    b    read_loop

done_read:
    mov  x20, x19
    mov  x19, x25
    mov  x24, #35
    mov  x28, x26
compare_loop:
    ldr  w0, [x19]
    ldr  w1, [x28]
    cmp  w0, w1
    b.ne mismatch
    add  x19, x19, #1
    add  x28, x28, #1
    subs x24, x24, #1
    b.ne compare_loop
    cmp  x19, x20
    b.ne mismatch
    mov  x0, #1
    mov  x1, x27
    mov  x2, #20
    mov  x8, #64
    svc  #0
    mov  x0, #0
    mov  x8, #93
    svc  #0
    b exit

mismatch:
    mov  x0, #0
    mov  x8, #93
    svc  #0
    b exit

exit:
    b exit
