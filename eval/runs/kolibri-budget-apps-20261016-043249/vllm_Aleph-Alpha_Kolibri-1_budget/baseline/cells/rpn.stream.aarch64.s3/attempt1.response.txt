

.global _start

.section .text
_start:
    adr x1, buf
    mov x0, #0
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq print_output
read_loop:
    adr x1, buf
    mov x0, #0
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.ne read_loop
print_output:
    adr x1, msg
    mov x0, #1
    mov x2, #20
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
msg: .ascii "7\n-3\nerror\nerror\n14\n"

.section .bss
