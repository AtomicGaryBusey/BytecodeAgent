

.section .data
in_len: .quad 0
.section .bss
inbuf: .space 49152
.section .text
.globl _start
_start:
    mov rbx, 0
    mov r12, offset inbuf
    mov r14, 0
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r14]
    mov rdx, 4096
    syscall
    add rbx, rax
    add r14, rax
    test rax, rax
    jz read_done
    jmp read_loop
read_done:
    mov qword ptr [in_len], rbx
    mov rdi, 1
    mov rsi, r12
    mov rdx, rbx
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
