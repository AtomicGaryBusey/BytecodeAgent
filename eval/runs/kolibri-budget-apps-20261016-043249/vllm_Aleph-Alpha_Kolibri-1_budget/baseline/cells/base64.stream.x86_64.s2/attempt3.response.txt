

.section .bss
inbuf: .space 49152
outbuf: .space 66400
.section .rodata
b64_table: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline: .byte 0x0A
.section .data
in_len: .quad 0
.section .text
.globl _start
_start:
    mov r12, offset inbuf
    xor r10, r10
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r12 + r10]
    mov rdx, 4096
    syscall
    add r10, rax
    test rax, rax
    jz read_done
    jmp read_loop
read_done:
    mov [in_len], r10
    mov r9, r10
    xor r10, r10
    mov r8, offset outbuf
    xor r11, r11
    jmp encoding_loop
encoding_loop:
    cmp r10, r9
    jge encoding_done
    lea rcx, [r9 - r10]
