    lea r13, [outbuf]
    xor r10, r10
read_loop:

    mov rdi, 0
    lea rsi, [r12 + r10]
    mov rdx, 4096
    syscall
    add r10, rax
    test rax, rax
    jz read_done
    jmp read_loop
read_done:
    mov r14, r10
    lea r9, [inbuf]
    lea r8, [outbuf]
    mov r11, r14
    xor r10, r10
    cmp r11, 3
    jl remainder_check
group_loop:
    movzx r15, byte ptr [r9]
    movzx rbx, byte ptr [r9+1]
    movzx rdx, byte ptr [r9+2]
    shr r15, 2
    and r15, 63
    mov rcx, r15
    movzx eax, byte ptr [rip + b64_table + rcx*1]
