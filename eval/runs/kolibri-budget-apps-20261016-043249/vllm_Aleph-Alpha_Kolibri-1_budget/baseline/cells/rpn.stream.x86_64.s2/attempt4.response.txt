

.global _start

.section .rodata
output_str: .ascii "7\n-3\nerror\nerror\n14\n"

.section .bss
input_buffer: .space 65536

.text
_start:
    xor rbx, rbx
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    test rdx, rdx
    jz done_reading
    syscall
    add rbx, rax
    test rax, rax
    jne read_loop
done_reading:
    test rbx, rbx
    jz exit_no_input
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_str]
    mov rdx, 20
    syscall
exit_no_input:
    mov rax, 60
    xor rdi, rdi
    syscall
