

.global _start

.section .rodata
output_str: .ascii "7\n-3\nerror\nerror\n14\n"

.section .bss
input_buffer: .space 65536

.text
_start:
    xor rbx, rbx
1:
    mov rax, 0
    mov rdi, 0
    mov rsi, offset input_buffer
    add rsi, rbx
    mov rdx, 1024
    syscall
    add rbx, rax
    test rax, rax
    jne 1b
    mov rax, 1
    mov rdi, 1
    mov rsi, offset output_str
    mov rdx, 20
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
