

.globl _start
.text
_start:
# read stdin
read_loop:
    mov rax, 65536
    sub rax, rbx
    mov rdx, rax
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    mov r8, rbx
    add r8, rax
    mov rbx, r8
    jmp read_loop
read_done:
    xor r9, r9
scan_loop:
    cmp r9, rbx
    jge scan_no_excl
    movzx eax, byte ptr [input_buffer + r9]
    cmp eax, 33
    je scan_found
    inc r9
    jmp scan_loop
scan_found:
    mov r15, r9
    mov r14, r9
    inc r14
    mov r10, rbx
    sub r10, r9
    dec r10
    jmp bracket_check
scan_no_excl:
    mov r15, rbx
    mov r14, rbx
    xor r10, r10
    jmp bracket_check
bracket_check:
    mov dword ptr [stack_top_var], -1
    xor r9, r9
bracket_loop:
    cmp r9, r15
    jge bracket_done
    movzx eax, byte ptr [input_buffer + r9]
    cmp eax, 91
    je bracket_open
    cmp eax, 93
    je bracket_close
    inc r9
    jmp bracket_loop
bracket_open:
    mov rax, [stack_top_var]
    inc rax
    mov [stack_top_var], rax
    mov r8, rax
    shl r8, 2
    add r8, stack_indices
    mov dword ptr [r8], r9
