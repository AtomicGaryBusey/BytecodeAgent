

.global _start

.section .bss
input_buf: .space 65536

.section .data
total_read: .quad 0
prog_len: .quad 0
input_start: .quad 0
input_len: .quad 0

.section .text
_start:
    lea rsi, [rip + input_buf]
    xor eax, eax
    mov qword ptr [total_read], rax
read_loop:
    xor rax, rax
    xor rdi, rdi
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    mov rcx, qword ptr [total_read]
    add rcx, rax
    mov qword ptr [total_read], rcx
    add rsi, rax
    jmp read_loop
read_done:
    lea rsi, [rip + input_buf]
    mov rcx, qword ptr [total_read]
    xor r8, r8
scan_loop:
    test rcx, rcx
    jz no_excl
    movzx eax, byte ptr [rsi]
    cmp al, 21
    je found
    inc r8
    inc rsi
    dec rcx
    jmp scan_loop
found:
    mov qword ptr [prog_len], r8
    add r8, 1
    mov qword ptr [input_start], r8
    mov rcx, qword ptr [total_read]
    sub rcx, r8
    mov qword ptr [input_len], rcx
    jmp bracket_check
no_excl:
    mov rcx, qword ptr [total_read]
    mov qword ptr [prog_len], rcx
    mov qword ptr [input_start], rcx
    xor eax, eax
    mov qword ptr [input_len], rax
bracket_check:
    lea rsi, [rip + input_buf]
    mov rcx, qword ptr [prog_len]
    xor r8, r8
bracket_loop:
    test rcx, rcx
    jz bracket_end
    movzx eax, byte ptr [rsi]
    cmp al, 91
    je bracket_open
    cmp al, 93
    je bracket_close
    inc rsi
    dec rcx
    jmp bracket_loop
bracket_open:
    inc r8
    inc rsi
    dec rcx
    jmp bracket_loop
bracket_close:
    test r8, r8
    jz bracket_unbalanced
    dec r8
    inc rsi
    dec rcx
    jmp bracket_loop
bracket_end:
    test r8, r8
    jz bracket_balanced
bracket_unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
    jmp bracket_unbalanced
bracket_balanced:
    xor rdi, rdi
    mov rax, 60
    syscall
