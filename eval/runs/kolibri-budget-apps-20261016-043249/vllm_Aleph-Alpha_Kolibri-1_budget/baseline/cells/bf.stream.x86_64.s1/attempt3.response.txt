

.global _start

.section .bss
input_buf: .space 65536
output_buf: .space 65536

.section .data
prog_len: .quad 0
input_start: .quad 0
input_len: .quad 0
total_read: .quad 0

.section .text
_start:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    xor r8, r8
read_loop:
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add rsi, rax
    jmp read_loop
read_done:
    mov [total_read], r8
    mov rax, [total_read]
    lea rdi, [rip + input_buf]
    lea rsi, [rip + input_buf]
    add rsi, rax
    mov rcx, rax
scan:
    test rcx, rcx
    jz no_excl
    movzx eax, byte ptr [rdi]
    cmp al, 21
    je found
    inc rdi
    dec rcx
    jmp scan
found:
    mov rax, [total_read]
    sub rax, rcx
    mov [prog_len], rax
    add rdi, 1
    mov [input_start], rdi
    dec rcx
    mov [input_len], rcx
    jmp output_input
no_excl:
    mov rax, [total_read]
    mov [prog_len], rax
    mov [input_len], 0
