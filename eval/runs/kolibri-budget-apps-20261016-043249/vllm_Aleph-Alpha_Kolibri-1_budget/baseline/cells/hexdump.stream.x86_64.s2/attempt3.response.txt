

.section .rodata
empty_line: .asciz "00000000\n"
.section .bss
input_buffer: .space 65536
total_len: .space 8
.section .text
.global _start
_start:
    mov rdi, 0
    mov rsi, offset input_buffer
    mov rdx, 65536
    mov r8, offset total_len
    mov r9, offset input_buffer
.read_loop:
    mov rax, 0
    syscall
    test rax, rax
    jz .done_reading
    mov r10, qword ptr [r8]
    add r10, rax
    mov qword ptr [r8], r10
    lea rsi, [r9 + r10]
    mov rdx, 65536
    sub rdx, r10
    jmp .read_loop
.done_reading:
    mov rdi, 1
    mov rsi, offset empty_line
    mov rdx, 9
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
