

.global _start

.section .bss
stdin_buf: .space 65536
tape: .space 30000
match: .space 262144
bracket_stack: .space 4000
align 4
prog_len: .space 4
input_start: .space 4

.section .text
_start:
    lea r12, [rip + stdin_buf]
    xor r13, r13
    mov r13, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r14, r12
    xor r11, r11
    lea r10, [rip + match]
    lea r15, [rip + bracket_stack]
scan_loop:
    cmp r14, r13
    jge scan_no_delim
    movzx eax, byte ptr [r14]
    cmp al, 33
    je found_delim
    cmp al, 91
    je bracket_open
    cmp al, 93
    je bracket_close
    add r14, 1
    jmp scan_loop
found_delim:
    mov eax, r14
    sub eax, r12
    mov dword ptr [prog_len], eax
    inc eax
    mov dword ptr [input_start], eax
    jmp scan_done
bracket_open:
    mov eax, r14
    sub eax, r12
    lea ecx, [r15 + r11*4]
    mov dword ptr [ecx], eax
    inc r11
    add r14, 1
    jmp scan_loop
bracket_close:
    test r11, r11
    jz unbalanced
    dec r11
    lea eax, [r15 + r11*4]
    mov eax, dword ptr [eax]
    mov ecx, r14
    sub ecx, r12
    mov ebx, eax
    mov edx, ecx
    inc edx
    lea eax, [r10 + ebx*4]
    mov dword ptr [eax], edx
    inc ebx
    lea eax, [r10 + ecx*4]
    mov dword ptr [eax], ebx
    add r14, 1
    jmp scan_loop
unbalanced:
    mov rax, 60
    mov rdi, 1
    xor rsi, rsi
    xor rdx, rdx
    syscall
scan_no_delim:
    mov eax, r13
    sub eax, r12
    mov dword ptr [prog_len], eax
    mov dword ptr [input_start], eax
scan_done:
    test r11, r11
    jz balanced
    mov rax, 60
    mov rdi, 1
    xor rsi, rsi
    xor rdx, rdx
    syscall
balanced:
    mov r10, r12
    mov rax, r12
    mov ebx, dword ptr [prog_len]
    add rax, ebx
    mov r11, rax
    mov rax, r12
    mov ebx, dword ptr [input_start]
    add rax, ebx
    mov r9, rax
    mov r14, r13
    lea r8, [rip + tape]
    lea r15, [rip + match]
main_loop:
    cmp r10, r11
    jge program_end
    movzx eax, byte ptr [r10]
    cmp al, 62
    je cmd_gt
    cmp al, 60
    je cmd_lt
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_dot
    cmp al, 44
    je cmd_comm
    cmp al, 91
    je cmd_lbracket
    cmp al, 93
    je cmd_rbracket
    add r10, 1
    jmp main_loop
cmd_gt:
    add r8, 1
    add r10, 1
    jmp main_loop
cmd_lt:
    sub r8, 1
    add r10, 1
    jmp main_loop
cmd_plus:
    movzx eax, byte ptr [r8]
    inc eax
    mov byte ptr [r8], al
    add r10, 1
    jmp main_loop
cmd_minus:
    movzx eax, byte ptr [r8]
    dec eax
    mov byte ptr [r8], al
    add r10, 1
    jmp main_loop
cmd_dot:
    movzx eax, byte ptr [r8]
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    syscall
    add r10, 1
    jmp main_loop
cmd_comm:
    cmp r9, r14
    jge input_exhausted
    movzx eax, byte ptr [r9]
    mov byte ptr [r8], al
    add r9, 1
    add r10, 1
    jmp main_loop
input_exhausted:
    xor eax, eax
    mov byte ptr [r8], al
    add r10, 1
    jmp main_loop
cmd_lbracket:
    movzx eax, byte ptr [r8]
    test eax, eax
    jz skip_loop
    mov ecx, r10
    sub ecx, r12
    lea eax, [r15 + ecx*4]
    mov eax, dword ptr [eax]
    mov rax, r12
    add rax, eax
    mov r10, rax
    jmp main_loop
skip_loop:
    add r10, 1
    jmp main_loop
cmd_rbracket:
    movzx eax, byte ptr [r8]
    test eax, eax
    jz continue_rbracket
    mov ecx, r10
    sub ecx, r12
    lea eax, [r15 + ecx*4]
    mov eax, dword ptr [eax]
    mov rax, r12
    add rax, eax
    mov r10, rax
    jmp main_loop
continue_rbracket:
    add r10, 1
    jmp main_loop
program_end:
    mov rax, 60
    xor rdi, rdi
    xor rsi, rsi
    xor rdx, rdx
    syscall
