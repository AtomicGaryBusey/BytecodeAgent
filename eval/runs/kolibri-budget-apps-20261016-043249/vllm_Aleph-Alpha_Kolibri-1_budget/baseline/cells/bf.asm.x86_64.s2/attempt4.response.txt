

.global _start

.section .bss
stdin_buf: .space 65536
tape: .space 30000
match: .space 262144
bracket_stack: .space 4000
prog_len: .space 4
input_start: .space 4
total_len: .space 4

.section .text
_start:
    lea r12, [rip + stdin_buf]
    xor r13, r13
read_loop:
    lea rsi, [r12 + r13]
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r8, r12
    mov r11, r13
scan_delim_loop:
    test r11, r11
    jz scan_no_delim
    movzx eax, byte ptr [r8]
    cmp al, 33
    je found_delim
    add r8, 1
    dec r11
    jmp scan_delim_loop
found_delim:
    mov eax, r8
    sub eax, r12
    mov dword ptr [rip + prog_len], eax
    inc eax
    mov dword ptr [rip + input_start], eax
    mov eax, r13
    sub eax, r12
    mov dword ptr [rip + total_len], eax
    jmp bracket_scan
scan_no_delim:
    mov eax, r13
    sub eax, r12
    mov dword ptr [rip + prog_len], eax
    mov dword ptr [rip + input_start], eax
    mov eax, r13
    sub eax, r12
    mov dword ptr [rip + total_len], eax
    jmp bracket_scan
bracket_scan:
    lea r10, [rip + match]
    movzx eax, dword ptr [rip + prog_len]
    mov r11, eax
    mov r8, r12
    xor r15, r15
unbalanced:
    mov rax, 60
    mov rdi, 1
    xor rsi, rsi
    xor rdx, rdx
    syscall
bracket_scan_loop:
    test r11, r11
    jz bracket_scan_done
    movzx eax, byte ptr [r8]
    cmp al, 91
    je bracket_open
    cmp al, 93
    je bracket_close
    add r8, 1
    dec r11
    jmp bracket_scan_loop
bracket_open:
    mov eax, r8
    sub eax, r12
    mov dword ptr [r15], eax
    add r15, 4
    add r8, 1
    dec r11
    jmp bracket_scan_loop
bracket_close:
    test r15, r15
    jz unbalanced
    sub r15, 4
    mov eax, dword ptr [r15]
    mov ecx, eax
    mov ebx, r8
    sub ebx, r12
    lea eax, [r10 + rcx*4]
    mov dword ptr [eax], ebx
    lea eax, [r10 + rbx*4]
    mov dword ptr [eax], ecx
    add r8, 1
    dec r11
    jmp bracket_scan_loop
bracket_scan_done:
    test r15, r15
    jne unbalanced
    mov r14, r12
    movzx eax, dword ptr [rip + input_start]
    lea r9, [r12 + rax]
    lea r8, [rip + tape]
    mov r13, r8
    movzx eax, dword ptr [rip + prog_len]
    lea r15, [r12 + rax]
main_loop:
    cmp r14, r15
    jge program_end
    movzx eax, byte ptr [r14]
    cmp al, 62
    je cmd_gt
    cmp al, 60
    je cmd_lt
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_dot
    cmp al, 44
    je cmd_comm
    cmp al, 91
    je cmd_lbracket
    cmp al, 93
    je cmd_rbracket
    add r14, 1
    jmp main_loop
cmd_gt:
    add r13, 1
    add r14, 1
    jmp main_loop
cmd_lt:
    sub r13, 1
    add r14, 1
    jmp main_loop
cmd_plus:
    movzx eax, byte ptr [r13]
    inc eax
    mov byte ptr [r13], al
    add r14, 1
    jmp main_loop
cmd_minus:
    movzx eax, byte ptr [r13]
    dec eax
    mov byte ptr [r13], al
    add r14, 1
    jmp main_loop
cmd_dot:
    movzx eax, byte ptr [r13]
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    mov rdx, 1
    syscall
    add r14, 1
    jmp main_loop
cmd_comm:
    movzx eax, dword ptr [rip + total_len]
    lea rcx, [r12 + rax]
    cmp r9, rcx
    jge input_exhausted
    movzx eax, byte ptr [r9]
    mov byte ptr [r13], al
    add r9, 1
    add r14, 1
    jmp main_loop
input_exhausted:
    xor eax, eax
    mov byte ptr [r13], al
    add r14, 1
    jmp main_loop
cmd_lbracket:
    movzx eax, byte ptr [r13]
    test eax, eax
    jnz continue_lbracket
    mov ecx, r14
    sub ecx, r12
    lea eax, [r10 + rcx*4]
    mov eax, dword ptr [eax]
    lea eax, [r12 + rax]
    inc eax
    mov r14, eax
    jmp main_loop
continue_lbracket:
    add r14, 1
    jmp main_loop
cmd_rbracket:
    movzx eax, byte ptr [r13]
    test eax, eax
    jz continue_rbracket
    mov ecx, r14
    sub ecx, r12
    lea eax, [r10 + rcx*4]
    mov eax, dword ptr [eax]
    lea eax, [r12 + rax]
    inc eax
    mov r14, eax
    jmp main_loop
continue_rbracket:
    add r14, 1
    jmp main_loop
program_end:
    mov rax, 60
    xor rdi, rdi
    xor rsi, rsi
    xor rdx, rdx
    syscall
