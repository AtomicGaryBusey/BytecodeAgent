

.global _start
.section .bss
stdin_buf: .space 65536
tape: .space 30000
match: .space 262144
bracket_stack: .space 4000
.align 4
prog_len: .space 4
.align 4
input_start: .space 4
.section .text
_start:
lea r12, [rip + stdin_buf]
lea r14, [rip + stdin_buf]
read_loop:
mov rax, 0
mov rdi, 0
mov rsi, r14
mov rdx, 65536
syscall
test rax, rax
jz read_done
add r14, rax
jmp read_loop
read_done:
mov r13, r14
mov r14, r12
lea r15, [rip + bracket_stack]
lea r10, [rip + match]
xor r11, r11
scan_loop:
cmp r14, r13
jge scan_done_no_delim
movzx eax, byte ptr [r14]
cmp al, 33
je found_delim
cmp al, 91
je bracket_open
cmp al, 93
je bracket_close
add r14, 1
jmp scan_loop
found_delim:
mov rax, r14
sub rax, r12
mov dword ptr [prog_len], eax
inc rax
mov dword ptr [input_start], eax
jmp scan_done
bracket_open:
lea rax, [r15 + r11*4]
mov [rax], r14
inc r11
add r14, 1
jmp scan_loop
bracket_close:
test r11, r11
jz unbalanced
dec r11
lea rax, [r15 + r11*4]
mov eax, [rax]
mov rdx, r14
sub rdx, r12
sub rax, r12
lea rcx, [r10 + rdx*4]
mov dword ptr [rcx], eax
lea rcx, [r10 + rax*4]
mov dword ptr [rcx], edx
add r14, 1
jmp scan_loop
scan_done_no_delim:
mov rax, r13
sub rax, r12
mov dword ptr [prog_len], eax
mov dword ptr [input_start], eax
jmp scan_done
scan_done:
test r11, r11
jz balanced
jmp unbalanced
balanced:
lea r8, [rip + tape]
mov rax, [input_start]
mov r9, r12
add r9, rax
mov r10, r12
mov rax, [prog_len]
mov r11, r12
add r11, rax
mov r14, r13
lea r15, [rip + match]
main_loop:
cmp r10, r11
jge program_end
movzx eax, byte ptr [r10]
cmp al, 62
je cmd_gt
cmp al, 60
je cmd_lt
cmp al, 43
je cmd_plus
cmp al, 45
je cmd_minus
cmp al, 46
je cmd_dot
cmp al, 44
je cmd_comm
cmp al, 91
je cmd_lbracket
cmp al, 93
je cmd_rbracket
add r10, 1
jmp main_loop
cmd_gt:
add r8, 1
add r10, 1
jmp main_loop
cmd_lt:
sub r8, 1
add r10, 1
jmp main_loop
cmd_plus:
movzx eax, byte ptr [r8]
inc eax
mov byte ptr [r8], al
add r10, 1
jmp main_loop
cmd_minus:
movzx eax, byte ptr [r8]
dec eax
mov byte ptr [r8], al
add r10, 1
jmp main_loop
cmd_dot:
mov rax, 1
mov rdi, 1
mov rsi, r8
mov rdx, 1
syscall
add r10, 1
jmp main_loop
cmd_comm:
cmp r9, r14
jge input_exhausted
movzx eax, byte ptr [r9]
mov byte ptr [r8], al
add r9, 1
add r10, 1
jmp main_loop
input_exhausted:
xor eax, eax
mov byte ptr [r8], al
add r10, 1
jmp main_loop
cmd_lbracket:
movzx eax, byte ptr [r8]
test eax, eax
jz skip_loop
add r10, 1
jmp main_loop
skip_loop:
mov rdx, r10
sub rdx, r12
lea rax, [r15 + rdx*4]
mov eax, [rax]
lea r10, [r12 + rax]
add r10, 1
jmp main_loop
cmd_rbracket:
movzx eax, byte ptr [r8]
test eax, eax
jz continue_rbracket
mov rdx, r10
sub rdx, r12
lea rax, [r15 + rdx*4]
mov eax, [rax]
lea r10, [r12 + rax]
add r10, 1
jmp main_loop
continue_rbracket:
add r10, 1
jmp main_loop
unbalanced:
mov rax, 60
mov rdi, 1
mov rsi, 0
mov rdx, 0
syscall
program_end:
mov rax, 60
mov rdi, 0
mov rsi, 0
mov rdx, 0
syscall
