

.global _start
.section .bss
stdin_buf: .space 65536
prog_len: .space 8
bal: .space 8
.section .text
_start:
lea rsi, [rip + stdin_buf]
mov r13, rsi
mov rdx, 65536
read_loop:
mov rax, 0
mov rdi, 0
mov rsi, r13
mov rdx, 65536
syscall
test rax, rax
jz read_done
add r13, rax
jmp read_loop
read_done:
lea r14, [rip + stdin_buf]
scan_delim:
cmp r14, r13
jge no_delim
movzx eax, byte ptr [r14]
cmp eax, 0x21
je found_delim
add r14, 1
jmp scan_delim
found_delim:
lea r15, [rip + stdin_buf]
sub r14, r14, r15
mov [prog_len], r14
jmp check_brackets
no_delim:
lea r15, [rip + stdin_buf]
sub r13, r13, r15
mov [prog_len], r13
jmp check_brackets
check_brackets:
lea r14, [rip + stdin_buf]
mov rax, [prog_len]
lea r15, [rip + stdin_buf]
add r15, rax
mov r12, 0
scan_brackets:
cmp r14, r15
jge done_scan
movzx eax, byte ptr [r14]
cmp eax, 0x5B
je bracket_open
cmp eax, 0x5D
je bracket_close
add r14, 1
jmp scan_brackets
bracket_open:
inc r12
add r14, 1
jmp scan_brackets
bracket_close:
dec r12
jc unbalanced
add r14, 1
jmp scan_brackets
done_scan:
test r12, r12
jz balanced
jmp unbalanced
balanced:
mov rax, 60
mov rdi, 0
mov rsi, 0
mov rdx, 0
syscall
unbalanced:
mov rax, 60
mov rdi, 1
mov rsi, 0
mov rdx, 0
syscall
