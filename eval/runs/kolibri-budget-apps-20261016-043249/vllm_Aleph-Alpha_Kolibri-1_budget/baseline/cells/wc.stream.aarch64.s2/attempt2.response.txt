

.global _start
.section .bss

.section .rodata

newline: .ascii "\n"
.section .text
_start:
    adrp x9, buf
    add x9, x9, :lo12:buf
    mov x10, #0
    mov x11, #0
    mov x12, #0
    mov x13, #0
    mov x14, #0
read_loop:
    add x1, x9, x10
    mov x2, #65536
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_end
    add x10, x10, x0
    add x11, x11, x0
    b read_loop
read_end:
    mov x0, x9
    add x1, x9, x11
    mov x12, #0
    mov x13, #0
    mov x14, #0
analysis_loop:
    cmp x0, x1
    b.ge analysis_end
    ldrb w2, [x0], #1
    cmp w2, #0x0A
    b.eq line_inc
    cmp w2, #0x20
    b.eq ws
    cmp w2, #0x09
    b.eq ws
    cmp w2, #0x0B
    b.eq ws
    cmp w2, #0x0C
    b.eq ws
    cmp w2, #0x0D
    b.eq ws
    tst x14, #1
    b.ne skip_word
    inc x13, x13, #1
