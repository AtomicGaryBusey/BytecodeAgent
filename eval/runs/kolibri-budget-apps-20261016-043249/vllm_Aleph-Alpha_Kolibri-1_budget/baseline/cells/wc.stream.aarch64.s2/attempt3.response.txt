

.global _start
.section .bss
buf: .space 65536
digit_buf: .space 12
.section .rodata
newline: .byte 10
space: .byte 32
.section .text
_start:
    adrp x9, buf
    add x9, x9, :lo12:buf
    mov x12, x9
    mov x11, #0
read_loop:
    mov x0, #0
    mov x1, x9
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_end
    add x11, x11, x0
    add x9, x9, x0
    b read_loop
read_end:
    mov x10, x9
    mov x13, #0
    mov x14, #0
    mov x15, #0
    mov x9, x12
scan_loop:
    cmp x9, x10
    b.ge scan_end
    ldrb w16, [x9], #1
    cmp w16, #0x0A
    b.eq nl
    cmp w16, #0x20
    b.eq ws
    cmp w16, #0x09
    b.eq ws
    cmp w16, #0x0B
    b.eq ws
    cmp w16, #0x0C
    b.eq ws
    cmp w16, #0x0D
    b.eq ws
    tst x15, #1
    b.eq in_word
    inc x14, x14, #1
