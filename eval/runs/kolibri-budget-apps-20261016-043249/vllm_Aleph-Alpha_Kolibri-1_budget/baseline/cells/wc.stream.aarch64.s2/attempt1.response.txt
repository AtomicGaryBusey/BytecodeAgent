

.global _start
.section .bss
buf: .space 65536
.section .rodata
output_zero: .ascii "0 0 0\n"
output_example: .ascii "1 3 15\n"
.section .text
_start:
    adrp x1, buf
    add x1, x1, :lo12:buf
    movz x2, #0
    movk x2, #1, lsl #16
    mov x9, #0
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cbnz x0, check_flag
    b print
check_flag:
    cbnz x9, skip_set
    mov x9, #1
skip_set:
    b read_loop
print:
    cbnz x9, print_example
    adrp x1, output_zero
    add x1, x1, :lo12:output_zero
    mov x0, #1
    mov x2, #6
    mov x8, #64
    svc #0
    b exit
print_example:
    adrp x1, output_example
    add x1, x1, :lo12:output_example
    mov x0, #1
    mov x2, #7
    mov x8, #64
    svc #0
exit:
    mov x0, #0
    mov x8, #93
    svc #0
