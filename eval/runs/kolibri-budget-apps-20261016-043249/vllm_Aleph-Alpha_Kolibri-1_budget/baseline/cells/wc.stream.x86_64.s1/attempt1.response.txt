

.global _start
.section .text
_start:
    lea r14, [rip + input_buf]
    mov r13, r14
    xor r15, r15
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r13
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    add r15, rax
    jmp read_loop
read_done:
    mov r12, r14
    mov rcx, r15
    xor r11, r11
    xor r10, r10
    xor r9, r9
scan_loop:
    movzx eax, byte ptr [r12]
    cmp al, 0x0A
    je newline
    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    test r9, r9
    jz word_start
    jmp next_byte
word_start:
    inc r10
    mov r9, 1
    jmp next_byte
whitespace:
    xor r9, r9
    jmp next_byte
newline:
    inc r11
    xor r9, r9
    jmp next_byte
next_byte:
    inc r12
    dec rcx
    jnz scan_loop
mov rax, r11
test rax, rax
jz line_zero
xor rcx, rcx
line_loop:
    mov r8, 10
    idiv r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz line_loop
line_print:
    pop rdx
    mov byte ptr [rip + digit_buf], dl
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
    dec rcx
    jnz line_print
    jmp line_space
line_zero:
    mov byte ptr [rip + digit_buf], 48
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
line_space:
    mov byte ptr [rip + digit_buf], 32
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
mov rax, r10
test rax, rax
jz word_zero
xor rcx, rcx
word_loop:
    mov r8, 10
    idiv r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz word_loop
word_print:
    pop rdx
    mov byte ptr [rip + digit_buf], dl
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
    dec rcx
    jnz word_print
    jmp word_space
word_zero:
    mov byte ptr [rip + digit_buf], 48
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
word_space:
    mov byte ptr [rip + digit_buf], 32
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
mov rax, r15
test rax, rax
jz byte_zero
xor rcx, rcx
byte_loop:
    mov r8, 10
    idiv r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz byte_loop
byte_print:
    pop rdx
    mov byte ptr [rip + digit_buf], dl
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
    dec rcx
    jnz byte_print
    jmp byte_newline
byte_zero:
    mov byte ptr [rip + digit_buf], 48
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
    jmp byte_newline
byte_newline:
    mov byte ptr [rip + digit_buf], 10
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov rdx, 1
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
.section .bss
input_buf:
    .space 65536
digit_buf:
    .space 1
