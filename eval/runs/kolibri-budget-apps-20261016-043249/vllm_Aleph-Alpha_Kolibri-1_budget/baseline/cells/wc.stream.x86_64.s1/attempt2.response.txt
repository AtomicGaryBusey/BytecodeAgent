

.global _start
.global input_buf
.global output_buf
.section .bss
input_buf:
   .space 65536
output_buf:
   .space 64
.section .text
convert_number:
   xor r9, r9
   test r8, r8
   jz zero_case
division_loop:
   xor rdx, rdx
   mov rax, r8
   mov rcx, 10
   idiv rcx
   add rdx, 48
   push rdx
   inc r9
   mov r8, rax
   test r8, r8
   jnz division_loop
   jmp digit_write
zero_case:
   push 48
   inc r9
digit_write:
pop_loop:
   pop rax
   mov byte ptr [r15], al
   inc r15
   dec r9
   jnz pop_loop
   ret

_start:
   mov r14, offset input_buf
   xor r13, r13

read_loop:
   test r13, r13
   jge read_done
   mov rdx, 65536
   sub rdx, r13
   mov rax, 0
   mov rdi, 0
   lea rsi, [r14 + r13]
   syscall
   test rax, rax
   jz read_done
   add r13, rax
   jmp read_loop

read_done:
   mov r12, r14
   mov rcx, r13
   xor r8, r8
   xor r10, r10
   xor r11, r11

scan_loop:
   test rcx, rcx
   jz scan_done
   movzx eax, byte ptr [r12]
   cmp al, 0x20
   je whitespace
   cmp al, 0x09
   je whitespace
   cmp al, 0x0A
   je newline
   cmp al, 0x0B
   je whitespace
   cmp al, 0x0C
   je whitespace
   cmp al, 0x0D
   je whitespace
   test r8, r8
   jz word_start
   jmp next_byte
word_start:
   inc r11
   mov r8, 1
   jmp next_byte
whitespace:
   xor r8, r8
   jmp next_byte
newline:
   inc r10
   xor r8, r8
   jmp next_byte
next_byte:
   inc r12
   dec rcx
   jnz scan_loop

scan_done:
   mov r12, offset output_buf
   mov r15, r12
   mov r8, r10
   call convert_number
   mov byte ptr [r15], 32
   inc r15
   mov r8, r11
   call convert_number
   mov byte ptr [r15], 32
   inc r15
   mov r8, r13
   call convert_number
   mov byte ptr [r15], 10
   inc r15
   mov rax, 1
   mov rdi, 1
   mov rsi, r12
   mov rdx, r15
   sub rdx, r12
   syscall
   mov rax, 60
   xor rdi, rdi
   syscall
