

.section .rodata
newline:
    .byte 10

.section .bss
inbuf:
    .space 65536

.section .text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x21, #0
    movz x20, #1, lsl #16
read_loop:
    add x1, x19, x21
    sub x2, x20, x21
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    cmp x21, x20
    b.ge read_done
    b read_loop

read_done:
    mov x22, #0
    mov x23, #0
parse_loop:
    cmp x23, x21
    b.ge parse_done
    add x4, x19, x23
    ldrb w5, [x4]
    cmp w5, #10
    b.ne next_byte
    add x6, x19, x23
    sub x6, x6, #1
    add x7, x19, x22
    cmp x6, x7
    b.lt skip_line
output_line:
    ldrb w5, [x6]
    mov x0, #1
    mov x1, x6
    mov x2, #1
    mov x8, #64
    svc #0
    sub x6, x6, #1
    cmp x6, x7
    b.ge output_line
skip_line:
    adr x1, newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    add x22, x23, #1
next_byte:
    add x23, x23, #1
    b parse_loop

parse_done:
    cmp x22, x21
    b.ge exit
    add x6, x19, x21
    sub x6, x6, #1
    add x7, x19, x22
    cmp x6, x7
    b.lt skip_remaining
output_remaining:
    ldrb w5, [x6]
    mov x0, #1
    mov x1, x6
    mov x2, #1
    mov x8, #64
    svc #0
    sub x6, x6, #1
    cmp x6, x7
    b.ge output_remaining
skip_remaining:
    adr x1, newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
exit:
    mov x0, #0
    mov x8, #93
    svc #0
