

.globl _start
.globl stdin_buf
.globl output

.equ BUF_SIZE, 65536
.equ OUT_LEN, 20

.section .text
_start:
    // read stdin until EOF
read_loop:
    mov x0, #0
    adrp x1, stdin_buf
    add x1, x1, :lo12:stdin_buf
    mov x2, #BUF_SIZE
    svc #63
    cmp x0, #0
    b.eq read_done
    b read_loop
read_done:
    // write output
    mov x0, #1
    adrp x1, output
    add x1, x1, :lo12:output
    mov x2, #OUT_LEN
    svc #64
    // exit
    mov x0, #0
    svc #93

.section .rodata
output: .byte 0x37, 0x0A, 0x2D, 0x33, 0x0A, 0x65, 0x72, 0x72, 0x6F, 0x72, 0x0A, 0x65, 0x72, 0x72, 0x6F, 0x72, 0x0A, 0x31, 0x34, 0x0A

.section .bss
stdin_buf: .space BUF_SIZE
