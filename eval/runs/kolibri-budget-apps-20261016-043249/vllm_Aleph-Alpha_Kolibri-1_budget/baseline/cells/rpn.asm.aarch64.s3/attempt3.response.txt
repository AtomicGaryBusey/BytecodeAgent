

.globl _start

.section .text
_start:
    // set stack pointer down by 8000 bytes
    mov x0, #8000
    sub sp, sp, x0

    // read stdin: fd 0, buffer address, count 4096
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #4096
    mov x8, #63
    svc #0

    // exit with status 0
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
input_buf:
    .space 65536
