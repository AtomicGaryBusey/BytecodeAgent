

.globl _start

.section .text
_start:
    // read stdin
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc x8
    // store read count
    adrp x2, input_len
    add x2, x2, :lo12:input_len
    str x0, [x2]
    ldr x4, [x2]
    cmp x4, #0
    b.eq exit
    // load err_str address
    adrp x6, err_str
    add x6, x6, :lo12:err_str
    mov x3, #0
    mov x5, #0
scan_loop:
    cmp x3, x4
    b.ge scan_done
    ldr w7, [x1, x3, lsl #0]
    cmp w7, #0x0A
    b.eq newline
    mov x5, #1
    b scan_next
newline:
    mov x0, #1
    mov x1, x6
    mov x2, #6
    mov x8, #64
    svc x8
    mov x5, #0
scan_next:
    add x3, x3, #1
    b scan_loop
scan_done:
    cmp x5, #0
    b.eq exit
    mov x0, #1
    mov x1, x6
    mov x2, #6
    mov x8, #64
    svc x8
exit:
    mov x0, #0
    mov x8, #93
    svc x8

.section .rodata
err_str:
    .byte 0x65, 0x72, 0x72, 0x6F, 0x72, 0x0A

.section .bss
input_buf:
    .space 65536
input_len:
    .space 8
