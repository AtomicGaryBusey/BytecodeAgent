

.globl _start

.section .bss
buf:
    .space 65536

.section .rodata
buf_size:
    .word 65536

.section .text
_start:
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x5, x1
    adrp x4, buf_size
    add x4, x4, :lo12:buf_size
    ldr x2, [x4]
    mov x3, #0
    b read_loop

read_loop:
    cmp x3, x2
    b.ge done_read
    sub x4, x2, x3
    mov x0, #0
    mov x8, #63
    svc #0
    cbnz w0, continue_read
    b done_read
continue_read:
    add x1, x1, x0
    add x3, x3, x0
    b read_loop

done_read:
    cmp x3, #0
    b.eq do_exit
    mov x0, #1
    mov x1, x5
    mov x2, x3
    mov x8, #64
    svc #0
do_exit:
    mov x0, #0
    mov x8, #93
    svc #0
