

.global _start

.section .rodata
newline_char: .byte 0x0A

.section .bss
input_buffer: .space 32768

.section .text
_start:
    lea r15, [rip + input_buffer]
    mov rsi, r15
    xor r14, r14
    mov rdi, 0
read_loop:
    mov rdx, 32768
    sub rdx, r14
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    add rsi, rax
    jmp read_loop
read_done:
    test r14, r14
    jz exit
    mov rdi, 1
    mov rsi, r15
    mov rdx, r14
    syscall
    mov rax, r14
    sub rax, 1
    add rax, r15
    mov bl, byte ptr [rax]
    cmp bl, 0x0A
    jne need_newline
    jmp exit
need_newline:
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
