

.global _start

.section .rodata
newline_char: .byte 0x0A

.section .bss
input_buffer: .space 32768

.section .text
_start:
    lea r12, [rip + input_buffer]
    xor r13, r13
read_loop:
    mov rdi, 0
    lea rsi, [r12 + r13]
    mov rdx, 32768
    sub rdx, r13
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    test r13, r13
    jz exit
    mov rdi, 1
    mov rsi, r12
    mov rdx, r13
    mov rax, 1
    syscall
    mov rax, r13
    dec rax
    add rax, r12
    movzx ebx, byte ptr [rax]
    cmp bl, 0x0A
    jne add_newline
    jmp exit
add_newline:
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    mov rax, 1
    syscall
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
