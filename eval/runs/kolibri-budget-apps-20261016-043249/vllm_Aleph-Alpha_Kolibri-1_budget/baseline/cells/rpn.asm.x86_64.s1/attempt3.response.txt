

section .bss
input_buf: .space 65536

section .text
.global _start

_start:
    lea rsi, [rip + input_buf]
read_loop:
    mov rdi, 0
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    mov rdi, 1
    mov rdx, rax
    syscall
    jmp read_loop
read_done:
    mov rax, 60
    xor rdi, rdi
    syscall
