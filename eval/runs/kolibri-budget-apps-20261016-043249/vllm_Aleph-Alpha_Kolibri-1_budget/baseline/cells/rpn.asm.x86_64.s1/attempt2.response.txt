input_buf: .space 65536

section .text
.global _start

_start:
read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    jmp read_loop
read_done:

    lea rsi, [rip + output]
    mov rdx, 20
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
