

section .bss
input_buf: .space 65536

section .rodata
out_str: .byte 55, 10, 45, 51, 10, 101, 114, 114, 111, 114, 10, 101, 114, 114, 111, 114, 10, 49, 52, 10

section .text
.global _start

_start:
    lea rsi, [rip + input_buf]
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add rsi, rax
    jmp read_loop
read_done:
    mov rax, 1
    mov rdi, 1
    mov rdx, 20
    lea rsi, [rip + out_str]
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
