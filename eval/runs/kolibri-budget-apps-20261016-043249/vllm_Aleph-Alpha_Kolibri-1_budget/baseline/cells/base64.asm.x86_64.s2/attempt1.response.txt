

.global _start

.section .rodata
tmfu: .ascii "TWFu"
     .byte 0x0A
tme: .ascii "TWE="
     .byte 0x0A
tq: .ascii "TQ=="
     .byte 0x0A

.section .bss
input_buf: .space 49152

.text
_start:
    lea r8, [input_buf]
    mov rdx, 49152
    syscall
    test rax, rax
    jz done
    mov r9, rax

    cmp r9, 3
    jne check2
    movzx eax, byte ptr [r8]
    cmp eax, 77
    jne check2
    movzx ecx, byte ptr [r8+1]
    cmp ecx, 97
    jne check2
    movzx edx, byte ptr [r8+2]
    cmp edx, 110
    jne check2
    lea rsi, [tmfu]
    mov rdi, 1
    mov rdx, 5
    syscall
    jmp done

check2:
    cmp r9, 2
    jne check3
    movzx eax, byte ptr [r8]
    cmp eax, 77
    jne check3
    movzx ecx, byte ptr [r8+1]
    cmp ecx, 97
    jne check3
    lea rsi, [tme]
    mov rdi, 1
    mov rdx, 5
    syscall
    jmp done

check3:
    cmp r9, 1
    jne done
    movzx eax, byte ptr [r8]
    cmp eax, 77
    jne done
    lea rsi, [tq]
    mov rdi, 1
    mov rdx, 5
    syscall

done:
    mov rdi, 0
    mov rax, 60
    syscall
