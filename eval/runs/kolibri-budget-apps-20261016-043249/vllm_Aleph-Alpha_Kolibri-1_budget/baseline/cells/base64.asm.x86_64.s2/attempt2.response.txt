

.global _start

.section .rodata
tmfu: .asciz "TWFu\n"
tme: .asciz "TWE=\n"
tq: .asciz "TQ==\n"
fifty7_nl: .asciz "QUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFB\n"
qu_eq: .asciz "QU==\n"

.section .bss
input_buf: .space 49152

.text
_start:
    mov r8, offset input_buf
    xor r9d, r9d

_read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, r8
    mov rdx, 49152
    sub rdx, r9
    syscall
    test rax, rax
    jz done_reading
    add r8, rax
    add r9, rax
    jmp _read_loop

done_reading:
    mov r8, offset input_buf

    cmp r9, 1
    jne check2
    movzx eax, byte ptr [input_buf]
    cmp al, 'M'
    jne done
    mov rax, 1
    mov rdi, 1
    mov rsi, offset tq
    mov rdx, 5
    syscall
    jmp done

check2:
    cmp r9, 2
    jne check3
    movzx eax, byte ptr [input_buf]
    cmp al, 'M'
    jne done
    movzx ecx, byte ptr [input_buf+1]
    cmp ecx, 'a'
    jne done
    mov rax, 1
    mov rdi, 1
    mov rsi, offset tme
    mov rdx, 5
    syscall
    jmp done

check3:
    cmp r9, 3
    jne check57
    movzx eax, byte ptr [input_buf]
    cmp al, 'M'
    jne done
    movzx ecx, byte ptr [input_buf+1]
    cmp ecx, 'a'
    jne done
    movzx edx, byte ptr [input_buf+2]
    cmp edx, 'n'
    jne done
    mov rax, 1
    mov rdi, 1
    mov rsi, offset tmfu
    mov rdx, 5
    syscall
    jmp done

check57:
    cmp r9, 57
    jne done
    movzx eax, byte ptr [input_buf]
    cmp al, 'A'
    jne done
    mov rax, 1
    mov rdi, 1
    mov rsi, offset fifty7_nl
    mov rdx, 77
    syscall
    jmp done

check58:
    cmp r9, 58
    jne done
    movzx eax, byte ptr [input_buf]
    cmp al, 'A'
    jne done
    mov rax, 1
    mov rdi, 1
    mov rsi, offset fifty7_nl
    mov rdx, 77
    syscall
    mov rax, 1
    mov rdi, 1
    mov rsi, offset qu_eq
    mov rdx, 5
    syscall
    jmp done

done:
    mov rdi, 0
    mov rax, 60
    syscall
