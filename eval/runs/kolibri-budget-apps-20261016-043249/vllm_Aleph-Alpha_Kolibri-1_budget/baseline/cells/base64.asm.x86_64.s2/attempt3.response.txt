

.global _start

.section .rodata
b64_chars: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
input_buf: .space 49152

.text
write_char:
  push rsi
  push r10
  mov rdi, 1
  mov rsi, r9d
  mov rdx, 1
  mov rax, 1
  syscall
  pop r10
  pop rsi
  ret

_start:
  mov rsi, offset input_buf
  mov rdx, 49152
  mov rax, 0
  syscall
  mov r10, rax
  xor r8d, r8d
  lea r12, [rip + b64_chars]

  test r10, r10
  jz finalize

encode_loop:
  cmp r10, 3
  jb handle_remainder

  movzx r13d, byte ptr [rsi]
  movzx r14d, byte ptr [rsi+1]
  movzx r15d, byte ptr [rsi+2]

  mov eax, r13d
  shr eax, 2
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne next1
  mov r9d, 10
  call write_char
  xor r8d, r8d
next1:
  mov eax, r13d
  and eax, 3
  shl eax, 4
  mov edx, r14d
  shr edx, 4
  or eax, edx
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne next2
  mov r9d, 10
  call write_char
  xor r8d, r8d
next2:
  mov eax, r14d
  and eax, 15
  shl eax, 2
  mov edx, r15d
  shr edx, 6
  or eax, edx
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne next3
  mov r9d, 10
  call write_char
  xor r8d, r8d
next3:
  mov eax, r15d
  and eax, 63
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne after_group
  mov r9d, 10
  call write_char
  xor r8d, r8d
after_group:
  add rsi, 3
  sub r10, 3
  jmp encode_loop

handle_remainder:
  test r10, r10
  jz finalize
  cmp r10, 2
  jge handle_2
  movzx eax, byte ptr [rsi]
  shr eax, 2
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne check1
  mov r9d, 10
  call write_char
  xor r8d, r8d
check1:
  movzx eax, byte ptr [rsi]
  and eax, 3
  shl eax, 4
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne check2
  mov r9d, 10
  call write_char
  xor r8d, r8d
check2:
  mov r9d, 61
  call write_char
  inc r8d
  cmp r8d, 76
  jne check3
  mov r9d, 10
  call write_char
  xor r8d, r8d
check3:
  mov r9d, 61
  call write_char
  inc r8d
  jmp finalize

handle_2:
  movzx eax, byte ptr [rsi]
  shr eax, 2
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne check2_1
  mov r9d, 10
  call write_char
  xor r8d, r8d
check2_1:
  movzx eax, byte ptr [rsi]
  and eax, 3
  shl eax, 4
  mov edx, byte ptr [rsi+1]
  shr edx, 4
  or eax, edx
  lea rdi, [r12 + rax]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne check3_1
  mov r9d, 10
  call write_char
  xor r8d, r8d
check3_1:
  movzx edx, byte ptr [rsi+1]
  and edx, 15
  shl edx, 2
  lea rdi, [r12 + rdx]
  movzx r9d, byte ptr [rdi]
  call write_char
  inc r8d
  cmp r8d, 76
  jne check4
  mov r9d, 10
  call write_char
  xor r8d, r8d
check4:
  mov r9d, 61
  call write_char
  inc r8d
  jmp finalize

finalize:
  cmp r8d, 0
  jle done
  mov r9d, 10
  call write_char
done:
  mov rdi, 0
  mov rax, 60
  syscall
