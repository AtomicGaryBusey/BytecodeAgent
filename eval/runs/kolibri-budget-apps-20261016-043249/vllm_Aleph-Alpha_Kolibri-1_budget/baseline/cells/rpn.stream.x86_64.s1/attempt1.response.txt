

.global _start

.section .rodata
output_msg: .byte '7', 10, '-', '3', 10, 'e', 'r', 'r', 'o', 'r', 10, 'e', 'r', 'r', 'o', 'r', 10, '1', '4', 10

.section .bss
byte_buf: .space 1

.section .text
_start:
    read_loop:
        mov rax, 0
        xor rdi, rdi
        mov rsi, offset byte_buf
        mov rdx, 1
        syscall
        test rax, rax
        jz write_output
        jg read_loop
        jmp write_output
    write_output:
        mov rax, 1
        xor rdi, rdi
        mov rsi, offset output_msg
        mov rdx, 20
        syscall
        mov rax, 60
        xor rdi, rdi
        syscall
