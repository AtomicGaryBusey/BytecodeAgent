

.global _start

.section .bss
buffer: .space 65536

.section .text
_start:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    test rax, rax
    jl error
    jz done
    mov rdi, 1
    mov rdx, rax
    mov rax, 1
    syscall
done:
    mov rax, 60
    xor rdi, rdi
    syscall
error:
    mov rax, 60
    mov rdi, 1
    syscall
