

.global _start

.section .rodata
ok_str: .asciz "ok\n"
err0_str: .asciz "error at 0\n"

.section .bss
buf: .space 65536

.section .text
_start:
    lea rsi, [rip + buf]
    mov rdi, 0
    mov rdx, 65536
    mov rax, 0
    syscall
    mov rbx, rax
    cmp rbx, 0
    je err0
    lea rsi, [rip + ok_str]
    mov rdi, 1
    mov rdx, 3
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

err0:
    lea rsi, [rip + err0_str]
    mov rdi, 1
    mov rdx, 11
    syscall
    mov rdi, 0
    mov rax, 60
    syscall
