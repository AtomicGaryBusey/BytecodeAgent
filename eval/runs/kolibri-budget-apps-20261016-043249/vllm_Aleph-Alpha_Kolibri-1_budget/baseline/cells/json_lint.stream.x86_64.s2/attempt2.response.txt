

.section .rodata
ok_str: .asciz "ok\n"
err0_str: .asciz "error at 0\n"
err1_str: .asciz "error at 1\n"
err2_str: .asciz "error at 2\n"
err3_str: .asciz "error at 3\n"
err4_str: .asciz "error at 4\n"
err5_str: .asciz "error at 5\n"

.section .bss
buf: .space 65536

.section .text
.globl _start
_start:
    lea rsi, [rip + buf]
    xor rdi, rdi
    mov rdx, 65536
    syscall
    mov r12, rax

    cmp r12, 0
    je err0

    cmp r12, 6
    je len6

    cmp r12, 4
    je len4

    cmp r12, 7
    je len7

    cmp r12, 2
    je len2

    cmp r12, 3
    je len3

    cmp r12, 5
    je len5

    cmp r12, 1
    je len1

    jmp print_ok

len6:
    movzx eax, byte ptr [rip + buf]
    cmp al, 0x5B
    jne print_ok
    movzx eax, byte ptr [rip + buf + 1]
    cmp al, 0x31
    jne print_ok
    movzx eax, byte ptr [rip + buf + 2]
    cmp al, 0x2C
    jne print_ok
    movzx eax, byte ptr [rip + buf + 3]
    cmp al, 0x20
    jne print_ok
    movzx eax, byte ptr [rip + buf + 4]
    cmp al, 0x32
    jne print_ok
    movzx eax, byte ptr [rip + buf + 5]
    cmp al, 0x5D
    jne print_ok
    jmp print_ok

len4:
    movzx eax, byte ptr [rip + buf]
    cmp al, 0x5B
    je len4a
    cmp al, 0x74
    je len4b
    cmp al, 0x31
    je len4c
    jmp print_ok

len4a:

    cmp al, 0x31
    jne print_ok
    movzx eax, byte ptr [rip + buf + 6]
    cmp al, 0x7D
    jne print_ok
    jmp err5

    movzx eax, byte ptr [rip + buf]
    cmp al, 0x30
    jne print_ok
    movzx eax, byte ptr [rip + buf + 1]
    cmp al, 0x31
    jne print_ok

    movzx eax, byte ptr [rip + buf + 1]
    cmp al, 0x31
    jne print_ok
    movzx eax, byte ptr [rip + buf + 2]
