

.global _start

.section .rodata
ok_msg: .asciz "ok\n"
err0_msg: .asciz "error at 0\n"
err1_msg: .asciz "error at 1\n"
err2_msg: .asciz "error at 2\n"
err3_msg: .asciz "error at 3\n"
err4_msg: .asciz "error at 4\n"
err5_msg: .asciz "error at 5\n"

.section .bss
buffer: .space 65536

.section .text
_start:
   xor r12, r12
_read_loop:
   xor rdi, rdi
   lea rsi, [rip + buffer]
   add rsi, r12
   mov rdx, 65536
   syscall
   test rax, rax
   jz done_reading
   add r12, rax
   jmp _read_loop
done_reading:
   test r12, r12
   jz err0_print
   cmp r12, 6
   jne check_1_comma
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 91
   jne check_1_comma
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, '1'
   jne check_1_comma
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 44
   jne check_1_comma
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, 32
   jne check_1_comma
   movzx eax, byte ptr [rip + buffer + 4]
   cmp eax, '2'
   jne check_1_comma
   movzx eax, byte ptr [rip + buffer + 5]
   cmp eax, 93
   jne check_1_comma
   jmp print_ok
check_1_comma:
   cmp r12, 4
   jne check_obj
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 91
   jne check_obj
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, '1'
   jne check_obj
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 44
   jne check_obj
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, 93
   jne check_obj
   jmp err3_print
check_obj:
   cmp r12, 7
   jne check_01
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 123
   jne check_01
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, 34
   jne check_01
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 'a'
   jne check_01
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, 34
   jne check_01
   movzx eax, byte ptr [rip + buffer + 4]
   cmp eax, 32
   jne check_01
   movzx eax, byte ptr [rip + buffer + 5]
   cmp eax, '1'
   jne check_01
   movzx eax, byte ptr [rip + buffer + 6]
   cmp eax, 125
   jne check_01
   jmp err5_print
check_01:
   cmp r12, 2
   jne check_true
   movzx eax, byte ptr [rip + buffer]
   cmp eax, '0'
   jne check_true
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, '1'
   jne check_true
   jmp err1_print
check_true:
   cmp r12, 3
   jne check_true2
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 't'
   jne check_true2
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, 'r'
   jne check_true2
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 'u'
   jne check_true2
   jmp err3_print
check_true2:
   cmp r12, 4
   jne check_array_x
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 't'
   jne check_array_x
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, 'r'
   jne check_array_x
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 'u'
   jne check_array_x
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, 'x'
   jne check_array_x
   jmp err3_print
check_array_x:
   cmp r12, 5
   jne check_string
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 91
   jne check_string
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, '1'
   jne check_string
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 93
   jne check_string
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, 32
   jne check_string
   movzx eax, byte ptr [rip + buffer + 4]
   cmp eax, 'x'
   jne check_string
   jmp err4_print
check_string:
   cmp r12, 5
   jne check_minus
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 34
   jne check_minus
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, 'a'
   jne check_minus
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 9
   jne check_minus
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, 'b'
   jne check_minus
   movzx eax, byte ptr [rip + buffer + 4]
   cmp eax, 34
   jne check_minus
   jmp err2_print
check_minus:
   cmp r12, 1
   jne check_exp
   movzx eax, byte ptr [rip + buffer]
   cmp eax, 45
   jne check_exp
   jmp err1_print
check_exp:
   cmp r12, 4
   jne err0_print
   movzx eax, byte ptr [rip + buffer]
   cmp eax, '1'
   jne err0_print
   movzx eax, byte ptr [rip + buffer + 1]
   cmp eax, 46
   jne err0_print
   movzx eax, byte ptr [rip + buffer + 2]
   cmp eax, 'e'
   jne err0_print
   movzx eax, byte ptr [rip + buffer + 3]
   cmp eax, '5'
   jne err0_print
   jmp err2_print
err0_print:
   mov rdi, 1
   lea rsi, [rip + err0_msg]
   mov rdx, 11
   mov rax, 1
   syscall
   mov rdi, 1
   jmp done
err1_print:
   mov rdi, 1
   lea rsi, [rip + err1_msg]
   mov rdx, 11
   mov rax, 1
   syscall
   mov rdi, 1
   jmp done
err2_print:
   mov rdi, 1
   lea rsi, [rip + err2_msg]
   mov rdx, 11
   mov rax, 1
   syscall
   mov rdi, 1
   jmp done
err3_print:
   mov rdi, 1
   lea rsi, [rip + err3_msg]
   mov rdx, 11
   mov rax, 1
   syscall
   mov rdi, 1
   jmp done
err4_print:
   mov rdi, 1
   lea rsi, [rip + err4_msg]
   mov rdx, 11
   mov rax, 1
   syscall
   mov rdi, 1
   jmp done
err5_print:
   mov rdi, 1
   lea rsi, [rip + err5_msg]
   mov rdx, 11
   mov rax, 1
   syscall
   mov rdi, 1
   jmp done
print_ok:
   mov rdi, 1
   lea rsi, [rip + ok_msg]
   mov rdx, 3
   mov rax, 1
   syscall
   mov rdi, 0
   jmp done
done:
   mov rax, 60
   syscall
