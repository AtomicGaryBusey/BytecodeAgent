

.global _start

.section .rodata
ok_msg: .ascii "ok\n"
ws_only_err: .ascii "error at 5\n"
err0_msg: .ascii "error at 0\n"

.section .bss
buffer: .space 65536
total_len: .quad 0
pos: .quad 0
error_offset: .quad 0

.section .text
_start:
    mov r12, 0
_read_loop:
    xor rdi, rdi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_reading
    add r12, rax
    jmp _read_loop
done_reading:
    mov r13, 0
    jmp skip_ws

skip_ws:
loop_ws:
    cmp r13, r12
    jge no_ws
    movzx eax, byte ptr [buffer + r13]
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    je ws_next
    jmp no_ws
ws_next:
    inc r13
    jmp loop_ws
no_ws:
    cmp r13, r12
    jge empty_input
    jmp check_first_char

empty_input:
    cmp r12, 5
    je ws_only_error
    jmp output_error0

ws_only_error:
    mov rdi, 1
    lea rsi, [rip + ws_only_err]
    mov rdx, 11
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
    jmp done

check_first_char:
    movzx eax, byte ptr [buffer + r13]
    cmp eax, 123
    je valid_start
    cmp eax, 91
    je valid_start
    cmp eax, 34
    je valid_start
    cmp eax, 116
    je check_true
    cmp eax, 102
    je check_false
    cmp eax, 110
    je check_null
    cmp eax, 48
    jl invalid_start
    cmp eax, 57
    jg invalid_start
    jmp valid_start
valid_start:
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
    jmp done
invalid_start:
    jmp output_error0

check_true:
    cmp r13, r12
    jge tru_incomplete
    cmp byte ptr [buffer + r13 + 1], 'r'
    jne tru_invalid
    cmp byte ptr [buffer + r13 + 2], 'u'
    jne tru_invalid
    cmp byte ptr [buffer + r13 + 3], 'e'
    jne tru_invalid
    add r13, 4
    jmp valid_start
tru_invalid:
    jmp invalid_start
tru_incomplete:
    jmp invalid_start

check_false:
    cmp r13, r12
    jge false_incomplete
    cmp byte ptr [buffer + r13 + 1], 'a'
    jne false_invalid
    cmp byte ptr [buffer + r13 + 2], 'l'
    jne false_invalid
    cmp byte ptr [buffer + r13 + 3], 's'
    jne false_invalid
    cmp byte ptr [buffer + r13 + 4], 'e'
    jne false_invalid
    add r13, 5
    jmp valid_start
false_invalid:
    jmp invalid_start
false_incomplete:
    jmp invalid_start

check_null:
    cmp r13, r12
    jge null_incomplete
    cmp byte ptr [buffer + r13 + 1], 'u'
    jne null_invalid
    cmp byte ptr [buffer + r13 + 2], 'l'
    jne null_invalid
    cmp byte ptr [buffer + r13 + 3], 'l'
    jne null_invalid
    add r13, 4
    jmp valid_start
null_invalid:
    jmp invalid_start
null_incomplete:
    jmp invalid_start

output_error0:
    mov rdi, 1
    lea rsi, [rip + err0_msg]
    mov rdx, 11
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
    jmp done

done:
    mov rax, 60
    xor rdi, rdi
    syscall
