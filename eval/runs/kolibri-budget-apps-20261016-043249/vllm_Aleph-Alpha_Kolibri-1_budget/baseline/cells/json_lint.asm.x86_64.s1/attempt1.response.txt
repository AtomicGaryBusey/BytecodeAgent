

.global _start

.section .bss
buffer: .space 65536

.section .rodata
ok_msg: .ascii "ok\n"
err_msg: .ascii "error at 0\n"

.section .text
_start:
    xor rdi, rdi
    lea rsi, [rip + buffer]
    mov rdx, 65536
    xor rax, rax
    syscall
    test rax, rax
    jz no_input
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
no_input:
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, 11
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
