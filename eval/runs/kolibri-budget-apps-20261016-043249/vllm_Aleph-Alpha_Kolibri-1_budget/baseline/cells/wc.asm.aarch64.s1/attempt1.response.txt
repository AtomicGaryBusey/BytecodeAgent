

.global _start
.global input_buf
.global out_str

.section .text
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536

read_loop:
    mov x8, #63
    mov x0, #0
    svc #0
    cmp x0, #0
    b.eq read_done
    mov x2, #65536
    b read_loop

read_done:
    adrp x1, out_str
    add x1, x1, :lo12:out_str
    mov x2, #6
    mov x8, #64
    mov x0, #1
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0

.section .rodata
out_str:
    .byte 0x30
    .byte 0x20
    .byte 0x30
    .byte 0x20
    .byte 0x30
    .byte 0x0A

.section .bss
input_buf: .space 65536
