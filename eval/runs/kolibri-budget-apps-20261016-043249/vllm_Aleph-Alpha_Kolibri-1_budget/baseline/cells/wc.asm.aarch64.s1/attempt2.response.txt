

.global _start

.section .text
_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read2
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
process_loop:
    cmp x0, #0
    b.eq process_done
    ldrb w3, [x1], #1
    cmp w3, #0x0A
    b.eq nl
    cmp w3, #0x20
    b.eq ws
    cmp w3, #0x09
    b.eq ws
    cmp w3, #0x0B
    b.eq ws
    cmp w3, #0x0C
    b.eq ws
    cmp w3, #0x0D
    b.eq ws
    tst x22, #1
    b.eq not_in_word
    b.next
not_in_word:
    add x20, x20, #1
    mov x22, #1
next:
    add x21, x21, #1
    b.process_loop
nl:
    add x19, x19, #1
    mov x22, #0
    add x21, x21, #1
    b.process_loop
ws:
    mov x22, #0
    add x21, x21, #1
    b.process_loop
process_done:
read2:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq print_counts
    b.print_counts
print_counts:
    mov x0, x19
    bl print_number_less100
    mov w0, #0x20
    bl print_byte
    mov x0, x20
    bl print_number_less100
    mov w0, #0x20
    bl print_byte
    mov x0, x21
    bl print_number_less100
    mov w0, #0x0A
    bl print_byte
    mov x8, #93
    mov x0, #0
    svc #0

print_number_less100:
    cmp x0, #10
    b.lt print_single
    mov x1, #10
    udiv x1, x0, x1
    umod x4, x0, x1
    mov w1, x1
    add w1, w1, #0x30
    mov w0, w1
    adrp x3, temp_digit
    add x3, x3, :lo12:temp_digit
    str w0, [x3]
    mov x0, #1
    mov x2, #1
    mov x1, x3
    mov x8, #64
    svc #0
    mov w2, x4
    add w2, w2, #0x30
    mov w0, w2
    str w0, [x3]
    mov x0, #1
    mov x2, #1
    mov x1, x3
    mov x8, #64
    svc #0
    b.print_done
print_single:
    mov w0, x0
    add w0, w0, #0x30
    adrp x1, temp_digit
    add x1, x1, :lo12:temp_digit
    str w0, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
print_done:
    ret

print_byte:
    adrp x1, temp_digit
    add x1, x1, :lo12:temp_digit
    str w0, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ret

.section .bss
input_buf: .space 65536
temp_digit: .space 1
