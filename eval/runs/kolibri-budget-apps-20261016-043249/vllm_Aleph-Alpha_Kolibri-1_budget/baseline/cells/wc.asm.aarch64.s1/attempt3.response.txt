

.global _start

.section .bss
input_buf: .space 65536
digit_buf: .space 5
temp_char: .byte 0
lines: .word 0
words: .word 0
bytes_cnt: .word 0
in_word: .byte 0

.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov x23, #0
read_loop:
    mov x0, #0
    mov x1, x19
    movk x2, #0x1, lsl #16
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
process_chunk:
    cmp x0, #0
    b.eq process_chunk_done
    ldrb w3, [x1], #1
    add x22, x22, #1
    cmp w3, #0x0A
    b.eq newline
    cmp w3, #0x20
    b.eq ws
    cmp w3, #0x09
    b.eq ws
    cmp w3, #0x0B
    b.eq ws
    cmp w3, #0x0C
    b.eq ws
    cmp w3, #0x0D
    b.eq ws
    tst x23, #1
    b.eq word_start
    b.weiter
word_start:
    add x21, x21, #1
    mov x23, #1
    b.weiter
ws:
    mov x23, #0
    b.weiter
newline:
    add x20, x20, #1
    mov x23, #0
    b.weiter
weiter:
    sub x0, x0, #1
    b.process_chunk
process_chunk_done:
    b.read_loop
read_done:
    adrp x1, digit_buf
    add x1, x1, :lo12:digit_buf
    add x1, x1, #4
    mov x0, x20
    bl print_number
    mov w0, #0x20
    bl print_char
    mov x0, x21
    adrp x1, digit_buf
    add x1, x1, :lo12:digit_buf
    add x1, x1, #4
    bl print_number
    mov w0, #0x20
    bl print_char
    mov x0, x22
    adrp x1, digit_buf
    add x1, x1, :lo12:digit_buf
    add x1, x1, #4
    bl print_number
    mov w0, #0x0A
    bl print_char
    mov x0, #0
    mov x8, #93
    svc #0

print_char:
    adrp x1, temp_char
    add x1, x1, :lo12:temp_char
    strb w0, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ret

print_number_zero:
    mov w0, #0x30
    bl print_char
    ret

print_number:
    cmp x0, #0
    b.eq print_number_zero
    mov x5, x1
    mov x2, #10
division_loop:
    cmp x0, #0
    b.eq division_done
    udiv x3, x0, x2
    mul x4, x3, x2
    sub x0, x0, x4
    add w0, w0, #0x30
    strb w0, [x5]
    sub x5, x5, #1
    mov x0, x3
    b.division_loop
division_done:
    add x5, x5, #1
print_loop:
    cmp x5, x1
    b.gt print_number_done
    ldrb w0, [x5]
    bl print_char
    add x5, x5, #1
    b.print_loop
print_number_done:
    ret
