

.section .rodata
space_char:
  .byte 32
newline_char:
  .byte 10

.section .bss
input_buf:
  .space 65537
digit_buf:
  .space 12

.section .text
.globl _start

_start:
  xor r12, r12
  xor r13, r13
  xor r14, r14
  xor r15, r15
  mov r8, 0

read_loop:
  lea rsi, [rip + input_buf]
  add rsi, r8
  mov rdi, 0
  mov rdx, 65536
  sub rdx, r8
  xor rax, rax
  syscall
  test rax, rax
  jz read_done
  mov r10, rax
  mov rcx, r10
process_chunk:
  test rcx, rcx
  jz process_done
  movzx eax, byte ptr [rsi]
  cmp eax, 10
  je newline
  cmp eax, 9
  je whitespace
  cmp eax, 11
  je whitespace
  cmp eax, 12
  je whitespace
  cmp eax, 13
  je whitespace
  cmp eax, 32
  je whitespace
  test r15, r15
  jz word_start
  jmp next_byte
newline:
  inc r13
  xor r15, r15
  jmp next_byte
whitespace:
  xor r15, r15
  jmp next_byte
word_start:
  inc r14
  mov r15, 1
  jmp next_byte
next_byte:
  inc rsi
  dec rcx
  jnz process_chunk
process_done:
  add r12, r10
  add r8, r10
  jmp read_loop

read_done:
  mov rax, r12
  lea r8, [rip + digit_buf]
  add r8, 11
  xor r9, r9
  test rax, rax
  jz zero_bytes
div_bytes:
  mov rdx, 10
  div rdx
  add dl, 48
  mov byte ptr [r8], dl
  dec r8
  inc r9
  test rax, rax
  jnz div_bytes
  inc r8
  jmp write_bytes
zero_bytes:
  mov byte ptr [r8], 48
  mov r9, 1
write_bytes:
  mov rsi, r8
  mov rdx, r9
  mov rdi, 1
  mov rax, 1
  syscall
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + space_char]
  mov rdx, 1
  syscall
  mov rax, r13
  lea r8, [rip + digit_buf]
  add r8, 11
  xor r9, r9
  test rax, rax
  jz zero_lines
div_lines:
  mov rdx, 10
  div rdx
  add dl, 48
  mov byte ptr [r8], dl
  dec r8
  inc r9
  test rax, rax
  jnz div_lines
  inc r8
  jmp write_lines
zero_lines:
  mov byte ptr [r8], 48
  mov r9, 1
write_lines:
  mov rsi, r8
  mov rdx, r9
  mov rdi, 1
  mov rax, 1
  syscall
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + space_char]
  mov rdx, 1
  syscall
  mov rax, r14
  lea r8, [rip + digit_buf]
  add r8, 11
  xor r9, r9
  test rax, rax
  jz zero_words
div_words:
  mov rdx, 10
  div rdx
  add dl, 48
  mov byte ptr [r8], dl
  dec r8
  inc r9
  test rax, rax
  jnz div_words
  inc r8
  jmp write_words
zero_words:
  mov byte ptr [r8], 48
  mov r9, 1
write_words:
  mov rsi, r8
  mov rdx, r9
  mov rdi, 1
  mov rax, 1
  syscall
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + newline_char]
  mov rdx, 1
  syscall
  mov rax, 60
  xor rdi, rdi
  syscall
