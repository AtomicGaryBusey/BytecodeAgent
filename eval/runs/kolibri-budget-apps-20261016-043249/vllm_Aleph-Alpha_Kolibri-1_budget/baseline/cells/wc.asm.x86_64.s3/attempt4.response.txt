

.section .rodata
space_char:
  .byte 32
newline_char:
  .byte 10

.section .bss
input_buf:
  .space 65536
digit_buf:
  .space 12

.section .text
.globl _start

_start:
  xor r12, r12
  xor r13, r13
  xor r14, r14
  xor r15, r15

read_loop:
  xor rdi, rdi
  lea rsi, [rip + input_buf]
  mov rdx, 65536
  xor rax, rax
  syscall
  test rax, rax
  jz read_done
  mov r10, rax
  mov rcx, rax
process_chunk:
  test rcx, rcx
  jz process_done
  movzx eax, byte ptr [rsi]
  cmp eax, 10
  je line
  cmp eax, 9
  je whitespace
  cmp eax, 11
  je whitespace
  cmp eax, 12
  je whitespace
  cmp eax, 13
  je whitespace
  cmp eax, 32
  je whitespace
  test r15, r15
  jz word_start
  jmp next
line:
  inc r13
  xor r15, r15
  jmp next
whitespace:
  xor r15, r15
  jmp next
word_start:
  inc r14
  mov r15, 1
  jmp next
next:
  inc rsi
  dec rcx
  jnz process_chunk
process_done:
  add r12, r10
  jmp read_loop

read_done:
  mov rax, r13
  lea r8, [rip + digit_buf]
  add r8, 11
  xor r9, r9
  test rax, rax
  jz lines_zero
lines_loop:
  test rax, rax
  jz lines_after
  mov r10, 10
  div r10
  add dl, 48
  mov byte ptr [r8], dl
  dec r8
  inc r9
  jmp lines_loop
lines_after:
  inc r8
  jmp lines_print
lines_zero:
  mov byte ptr [r8], 48
  mov r9, 1
  jmp lines_print
lines_print:
  mov rsi, r8
  mov rdx, r9
  mov rdi, 1
  mov rax, 1
  syscall

  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + space_char]
  mov rdx, 1
  syscall

  mov rax, r14
  lea r8, [rip + digit_buf]
  add r8, 11
  xor r9, r9
  test rax, rax
  jz words_zero
words_loop:
  test rax, rax
  jz words_after
  mov r10, 10
  div r10
  add dl, 48
  mov byte ptr [r8], dl
  dec r8
  inc r9
  jmp words_loop
words_after:
  inc r8
  jmp words_print
words_zero:
  mov byte ptr [r8], 48
  mov r9, 1
  jmp words_print
words_print:
  mov rsi, r8
  mov rdx, r9
  mov rdi, 1
  mov rax, 1
  syscall

  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + space_char]
  mov rdx, 1
  syscall

  mov rax, r12
  lea r8, [rip + digit_buf]
  add r8, 11
  xor r9, r9
  test rax, rax
  jz bytes_zero
bytes_loop:
  test rax, rax
  jz bytes_after
  mov r10, 10
  div r10
  add dl, 48
  mov byte ptr [r8], dl
  dec r8
  inc r9
  jmp bytes_loop
bytes_after:
  inc r8
  jmp bytes_print
bytes_zero:
  mov byte ptr [r8], 48
  mov r9, 1
  jmp bytes_print
bytes_print:
  mov rsi, r8
  mov rdx, r9
  mov rdi, 1
  mov rax, 1
  syscall

  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + newline_char]
  mov rdx, 1
  syscall

  xor rdi, rdi
  mov rax, 60
  syscall
