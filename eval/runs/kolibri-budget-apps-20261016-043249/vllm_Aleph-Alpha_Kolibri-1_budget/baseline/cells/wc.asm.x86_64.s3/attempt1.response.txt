

.section .rodata
space_byte:
  .byte 32
newline_byte:
  .byte 10

.section .bss
input_buf:
  .space 65536
digit_buf:
  .space 12

.section .text
.globl _start

convert_number:
  mov r8, offset digit_buf
  add r8, 11
  xor r9, r9
  test rax, rax
  jz zero_digit
div_loop:
  div 10
  add dl, 48
  mov [r8], dl
  dec r8
  inc r9
  test rax, rax
  jnz div_loop
  add r8, 1
  jmp done
zero_digit:
  mov byte ptr [r8], 48
  inc r9
  jmp done
done:
  mov rsi, r8
  mov rdx, r9
  ret

_start:
  xor r12, r12
  xor r13, r13
  xor r14, r14
  xor r15, r15

read_loop:
  mov rax, 0
  mov rdi, 0
  mov rsi, offset input_buf
  mov rdx, 65536
  syscall
  test rax, rax
  jz read_done
  mov rcx, rax
process_chunk:
  test rcx, rcx
  jz process_done
  movzx eax, byte ptr [rsi]
  cmp eax, 10
  je newline
  cmp eax, 9
  je whitespace
  cmp eax, 11
  je whitespace
  cmp eax, 12
  je whitespace
  cmp eax, 13
  je whitespace
  cmp eax, 32
  je whitespace
  test r15, r15
  jz word_start
  jmp next_byte
newline:
  inc r13
  xor r15, r15
  jmp next_byte
whitespace:
  xor r15, r15
  jmp next_byte
word_start:
  inc r14
  mov r15, 1
  jmp next_byte
next_byte:
  inc r12
  inc rsi
  dec rcx
  jnz process_chunk
process_done:
  jmp read_loop

read_done:
  mov rax, r13
  call convert_number
  mov rdi, 1
  mov rax, 1
  syscall
  mov rsi, offset space_byte
  mov rdx, 1
  mov rax, 1
  syscall
  mov rax, r14
  call convert_number
  mov rdi, 1
  mov rax, 1
  syscall
  mov rsi, offset space_byte
  mov rdx, 1
  mov rax, 1
  syscall
  mov rax, r12
  call convert_number
  mov rdi, 1
  mov rax, 1
  syscall
  mov rsi, offset newline_byte
  mov rdx, 1
  mov rax, 1
  syscall
  mov rax, 60
  mov rdi, 0
  syscall
