

.global _start

.section .rodata
ok_str: .asciz "ok\n"
error_str: .asciz "error at 0\n"

.section .bss
buffer: .space 65536

.section .text
_start:
    mov r8, 0
read_loop:
    mov rdi, 0
    lea rsi, [rip + buffer]
    add rsi, r8
    mov rdx, 65536
    mov rax, 0
    syscall
    add r8, rax
    test rax, rax
    jz done_reading
    jmp read_loop
done_reading:
    test r8, r8
    jz empty
    mov rdi, 1
    lea rsi, [rip + ok_str]
    mov rdx, 3
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall
empty:
    mov rdi, 1
    lea rsi, [rip + error_str]
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall
