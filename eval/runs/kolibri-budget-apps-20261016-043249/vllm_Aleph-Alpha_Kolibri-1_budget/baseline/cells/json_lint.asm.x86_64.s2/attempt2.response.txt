

.global _start

.section .bss
buffer: .space 65536
len_val: .quad 0

.section .rodata
ok_str: .asciz "ok\n"
error_str: .asciz "error at 0\n"

.section .text
_start:
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    mov [len_val], rax

    mov rax, [len_val]
    test rax, rax
    jz error_empty

    mov rdi, 1
    lea rsi, [rip + ok_str]
    mov rdx, 3
    syscall

    mov rdi, 0
    mov rax, 60
    syscall

error_empty:
    mov rdi, 1
    lea rsi, [rip + error_str]
    mov rdx, 11
    syscall
    mov rdi, 1
    mov rax, 60
    syscall
