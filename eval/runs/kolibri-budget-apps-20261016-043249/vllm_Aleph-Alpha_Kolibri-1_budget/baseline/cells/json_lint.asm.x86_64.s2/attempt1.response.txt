

.global _start

.section .bss
buffer: .space 4096

.section .rodata
ok_str: .asciz "ok\n"
error_str: .asciz "error at 0\n"

.section .text
_start:
    mov rdi, 0
    mov rsi, buffer
    mov rdx, 4096
    syscall

    test rax, rax
    jz error

    mov rdi, 1
    mov rsi, ok_str
    mov rdx, 3
    syscall

    mov rdi, 0
    syscall

error:
    mov rdi, 1
    mov rsi, error_str
    mov rdx, 11
    syscall

    mov rdi, 1
    syscall
