

.section .bss
cur_byte:
    .space 1
line_cnt:
    .space 4
word_cnt:
    .space 4
byte_cnt:
    .space 4
digit_buf:
    .space 5
space_val:
    .byte 32
newline_val:
    .byte 10
zero_char:
    .byte 48

.section .text
.globl _start

_start:
    adr x1, cur_byte
    mov w5, #0
    mov w6, #0
    mov w7, #0
    mov w8, #0
read_loop:
    mov x0, #0
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    ldrb w3, [x1]
    cmp w3, #10
    b.eq newline
    cmp w3, #9
    b.eq ws
    cmp w3, #11
    b.eq ws
    cmp w3, #12
    b.eq ws
    cmp w3, #13
    b.eq ws
    cmp w3, #32
    b.eq ws
    cmp w8, #0
    b.ne in_word
    add w6, w6, #1
    mov w8, #1
    b.done
ws:
    mov w8, #0
    b.done
newline:
    add w5, w5, #1
    mov w8, #0
    b.done
done:
    add w7, w7, #1
    b read_loop

read_done:
    adr x0, line_cnt
    str w5, [x0]
    adr x1, word_cnt
    str w6, [x1]
    adr x2, byte_cnt
    str w7, [x2]
    b print_counts

print_counts:
    adr x0, line_cnt
    ldr w0, [x0]
    bl print_int
    adr x1, space_val
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    adr x0, word_cnt
    ldr w0, [x0]
    bl print_int
    adr x1, space_val
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    adr x0, byte_cnt
    ldr w0, [x0]
    bl print_int
    adr x1, newline_val
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

print_int:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cmp w0, #0
    b.eq print_zero

    adr x1, digit_buf
    mov w1, x1
    add w1, w1, #5
    mov w4, #0

div_loop:
    cmp w0, #0
    b.eq div_done
    mov w2, w0
    mov w3, #0
div_sub_loop:
    cmp w2, #10
    b.lt div_sub_done
    sub w2, w2, #10
    add w3, w3, #1
    b div_sub_loop
div_sub_done:
    add w2, w2, #48
    sub w1, w1, #1
    strb w2, [w1]
    add w4, w4, #1
    mov w0, w3
    b div_loop
div_done:
print_digit_loop:
    cmp w4, #0
    b.eq print_digits_done
    ldrb w0, [w1]
    mov x0, #1
    mov x1, w1
    mov x2, #1
    mov x8, #64
    svc #0
    add w1, w1, #1
    sub w4, w4, #1
    b print_digit_loop
print_digits_done:
    ldp x29, x30, [sp], #16
    ret

print_zero:
    adr x1, zero_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ldp x29, x30, [sp], #16
    ret
