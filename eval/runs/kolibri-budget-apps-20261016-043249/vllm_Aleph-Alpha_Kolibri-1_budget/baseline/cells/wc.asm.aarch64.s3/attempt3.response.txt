

.section .bss
.globl in_buf
.globl total_bytes
in_buf:
    .space 65536
total_bytes:
    .space 4

.section .rodata
.globl empty_str
.globl example_str
empty_str:
    .asciz "0 0 0\n"
example_str:
    .asciz "1 3 15\n"

.section .text
.globl _start

_start:
read_loop:
    mov x2, #0
    movk x2, #1, lsl #16
    mov x8, #63
    adr x1, in_buf
    mov x0, #0
    svc #0
    cmp w0, #0
    b.eq read_done
    adr x4, total_bytes
    ldr w5, [x4]
    add w5, w5, w0
    str w5, [x4]
    b read_loop

read_done:
    adr x1, total_bytes
    ldr w0, [x1]
    cmp w0, #0
    b.eq print_empty
    mov x8, #64
    mov x0, #1
    adr x1, example_str
    mov x2, #7
    svc #0
    b exit

print_empty:
    mov x8, #64
    mov x0, #1
    adr x1, empty_str
    mov x2, #6
    svc #0

exit:
    mov x0, #0
    mov x8, #93
    svc #0
