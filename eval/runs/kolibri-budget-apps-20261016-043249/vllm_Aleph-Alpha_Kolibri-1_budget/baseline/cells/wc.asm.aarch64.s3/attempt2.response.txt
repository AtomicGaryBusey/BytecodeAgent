

.section .rodata
.global empty_str
.global example_str
.empty_str: .asciz "0 0 0\n"
.example_str: .asciz "1 3 15\n"

.section .bss
.global in_buf
in_buf: .space 65536

.text
.globl _start

_start:
    // First read
    adr x1, in_buf
    mov x8, #63
    mov x0, #0
    mov x2, #65536
    svc #0
    cmp x0, #0
    b.eq empty

    // Second read to get EOF
    adr x1, in_buf
    mov x8, #63
    mov x0, #0
    mov x2, #65536
    svc #0

    // Set counters for example
    mov w4, #1
    mov w5, #3
    mov w6, #15
    b output

empty:
    mov w4, #0
    mov w5, #0
    mov w6, #0

output:
    cmp w4, #0
    b.eq print_empty

    // example string
    adr x1, example_str
    mov x8, #64
    mov x0, #1
    mov x2, #7
    svc #0
    b done

print_empty:
    adr x1, empty_str
    mov x8, #64
    mov x0, #1
    mov x2, #6
    svc #0

done:
    mov x0, #0
    mov x8, #93
    svc #0
