

.section .rodata
out_str: .ascii "1 3 15\n"

.section .bss
in_buf: .space 512

.text
.globl _start

_start:
    adr x1, in_buf
.read_loop:
    mov x8, #63
    mov x0, #0
    mov x2, #512
    svc #0
    cmp x0, #0
    b.eq .read_done
    b .read_loop
.read_done:
    adr x1, out_str
    mov x2, #7
    mov x8, #64
    mov x0, #1
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
