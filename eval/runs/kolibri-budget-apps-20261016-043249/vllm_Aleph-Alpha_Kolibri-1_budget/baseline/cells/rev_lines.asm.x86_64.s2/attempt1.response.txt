

.section .bss
input_buf:
    .space 65536
output_buf:
    .space 65537
.section .text
    .globl _start
_start:
    mov rax, 0
    xor rdi, rdi
    mov rsi, offset input_buf
    mov rdx, 65536
    syscall
    test rax, rax
    je no_input
    mov r12, rax
    mov r13, offset input_buf
    mov r14, r13
    mov r15, offset output_buf
    mov rdi, r15
loop_start:
    test r12, r12
    je after_loop
    mov al, byte ptr [r13]
    cmp al, 10
    je newline
    inc r13
    dec r12
    jmp loop_start
newline:
    mov r8, r13
    sub r8, r14
    test r8, r8
    jne copy_line
    mov byte ptr [rdi], 10
    inc rdi
    jmp after_newline
copy_line:
    mov r9, r13
    dec r9
copy_loop:
    mov [rdi], byte ptr [r9]
    inc rdi
    dec r9
    dec r8
    jnz copy_loop
    mov byte ptr [rdi], 10
    inc rdi
after_newline:
    add r14, 1
    dec r12
    inc r13
    jmp loop_start
after_loop:
    cmp r14, r13
    je no_final_line
    mov r8, r13
    sub r8, r14
    mov r9, r13
    dec r9
copy_final_loop:
    mov [rdi], byte ptr [r9]
    inc rdi
    dec r9
    dec r8
    jnz copy_final_loop
    mov byte ptr [rdi], 10
    inc rdi
no_final_line:
    mov rax, 1
    mov rdx, rdi
    sub rdx, r15
    mov rsi, r15
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
no_input:
    mov rax, 60
    xor rdi, rdi
    syscall
