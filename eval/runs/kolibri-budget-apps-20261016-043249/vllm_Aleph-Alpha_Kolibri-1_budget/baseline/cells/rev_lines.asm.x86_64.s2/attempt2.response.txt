

.section .bss
input_buf:
    .space 65536
output_buf:
    .space 131072
.section .text
.globl _start
_start:
    lea r8, [rip + input_buf]
    lea r9, [rip + output_buf]
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r8 + r13]
    mov rdx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    je done_reading
    add r13, rax
    jmp read_loop
done_reading:
    mov r12, r13
    xor r13, r13
    xor r14, r14
    xor r15, r15
scan_loop:
    cmp r13, r12
    je after_scan
    mov al, byte ptr [r8 + r13]
    cmp al, 10
    je newline
    inc r13
    jmp scan_loop
newline:
    mov r10, r13
    dec r10
    cmp r10, r14
    jl reverse_done
reverse_loop:
    mov al, byte ptr [r8 + r10]
    mov byte ptr [r9 + r15], al
    inc r15
    dec r10
    cmp r10, r14
    jge reverse_loop
reverse_done:
    mov byte ptr [r9 + r15], 10
    inc r15
    lea r14, [r13 + 1]
    mov r13, r14
    jmp scan_loop
final_reverse_loop:
    mov al, byte ptr [r8 + r10]
    mov byte ptr [r9 + r15], al
    inc r15
    dec r10
    cmp r10, r14
    jge final_reverse_loop
final_reverse_done:
    mov byte ptr [r9 + r15], 10
    inc r15
    jmp no_final_line
after_scan:
    cmp r14, r12
    jge no_final_line
    mov r10, r12
    dec r10
    jmp final_reverse_loop
no_final_line:
    mov rax, 1
    mov rdi, 1
    mov rsi, r9
    mov rdx, r15
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
