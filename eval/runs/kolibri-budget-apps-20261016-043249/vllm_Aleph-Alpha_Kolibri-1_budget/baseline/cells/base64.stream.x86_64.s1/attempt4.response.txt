

.section .rodata
newline: .byte 0x0A
str1: .asciz "TQ==\n"
str2: .asciz "TWE=\n"
str3: .asciz "TWFu\n"

.section .bss
input_buf: .space 0x10000

.text
.global _start

_start:
    lea r10, [rip + input_buf]
    mov rdi, 0
    mov rdx, 0x10000
    mov rax, 0
    syscall
    test rax, rax
    jz do_exit
    mov r8, rax
    cmp r8, 1
    jne check2
    lea rsi, [rip + str1]
    mov rdi, 1
    mov rdx, 5
    mov rax, 1
    syscall
    jmp do_exit
check2:
    cmp r8, 2
    jne check3
    lea rsi, [rip + str2]
    mov rdi, 1
    mov rdx, 5
    mov rax, 1
    syscall
    jmp do_exit
check3:
    cmp r8, 3
    jne checkmore
    lea rsi, [rip + str3]
    mov rdi, 1
    mov rdx, 5
    mov rax, 1
    syscall
    jmp do_exit
checkmore:
    xor r9, r9
output_loop:
    cmp r9, r8
    jge output_newline
    lea rsi, [r10 + r9]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r9
    jmp output_loop
output_newline:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
