

.equ read, 0
.equ write, 1
.equ exit, 60
.equ exit_group, 231
.equ input_size, 49152
.equ output_size, 66400

.globl _start

.section .rodata
base64_alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
input_buf:
    .space input_size
output_buf:
    .space output_size

.section .text
_start:
    xor r8, r8
.read_loop:
    xor rdi, rdi
    lea rsi, [rip + input_buf + r8]
    mov rdx, input_size
    sub rdx, r8
    syscall
    test rax, rax
    jz .read_done
    add r8, rax
    jmp .read_loop
.read_done:
    test r8, r8
    jz .empty_input
    lea rsi, [rip + input_buf]
    lea r11, [rip + output_buf]
    lea r10, [rip + output_buf]
    lea rdi, [rip + base64_alpha]
    xor r9, r9
    xor rbx, rbx
    mov r8, r8
.group_loop:
    cmp r8, 3
    jb .remainder
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]
    shr eax, 2
    and eax, 63
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .char0_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.char0_done:
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .char1_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.char1_done:
    movzx ebx, byte ptr [rsi+1]
    and ebx, 15
    shl ebx, 2
    shr ecx, 6
    or ebx, ecx
    mov eax, ebx
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .char2_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.char2_done:
    movzx ecx, byte ptr [rsi+2]
    and ecx, 63
    mov eax, ecx
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .group_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.group_done:
    add rsi, 3
    sub r8, 3
    jmp .group_loop
.remainder:
    test r8, r8
    jz .finalize
    cmp r8, 1
    je .remainder1
    jmp .remainder2
.remainder1:
    movzx eax, byte ptr [rsi]
    shr eax, 2
    and eax, 63
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .rem1_char0_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.rem1_char0_done:
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .rem1_char1_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.rem1_char1_done:
    mov byte ptr [r10], 61
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .rem1_first_eq_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.rem1_first_eq_done:
    mov byte ptr [r10], 61
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .finalize
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
    jmp .finalize
.remainder2:
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    shr eax, 2
    and eax, 63
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .rem2_char0_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.rem2_char0_done:
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .rem2_char1_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.rem2_char1_done:
    and ebx, 15
    shl ebx, 2
    mov eax, ebx
    mov al, byte ptr [rdi + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .rem2_char2_done
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.rem2_char2_done:
    mov byte ptr [r10], 61
    add r10, 1
    inc rbx
    inc r9
    cmp r9, 76
    jne .finalize
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
    jmp .finalize
.finalize:
    test r9, r9
    jz .do_write
    mov byte ptr [r10], 10
    add r10, 1
    inc rbx
    xor r9, r9
.do_write:
    lea rsi, [rip + output_buf]
    mov rdi, 1
    mov rax, write
    mov rdx, rbx
    syscall
    xor rdi, rdi
    mov rax, exit
    syscall
.empty_input:
    xor rdi, rdi
    mov rax, exit
    syscall
