

.equ read, 0
.equ write, 1
.equ exit, 60
.globl _start

.section .rodata
base64_alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline:
    .byte 10

.section .bss
input_buf:
    .space 49152
output_buf:
    .space 66400

.section .text
_start:
    xor r15, r15
.read_loop:
    lea rsi, [rip + input_buf]
    add rsi, r15
    mov rdx, 49152
    xor rdi, rdi
    mov rax, read
    syscall
    test rax, rax
    jz .read_done
    add r15, rax
    jmp .read_loop
.read_done:
    test r15, r15
    jz .empty_input
    lea r12, [rip + input_buf]
    lea r10, [rip + output_buf]
    lea r13, [rip + base64_alpha]
    mov r8, r15
    xor r14, r14
    jmp .encode_loop
.empty_input:
    xor rdi, rdi
    mov rax, exit
    syscall
.encode_loop:
    cmp r8, 3
    jge .full_group
    jmp .remainder
.full_group:
    movzx rax, byte ptr [r12]
    movzx rbx, byte ptr [r12+1]
    movzx rcx, byte ptr [r12+2]
    shr rax, 2
    and rax, 63
    mov r9, rax
    movzx rax, byte ptr [r12]
    and rax, 3
    shl rax, 4
    shr rbx, 4
    or rax, rbx
    mov r11, rax
    movzx rbx, byte ptr [r12+1]
    and rbx, 15
    shl rbx, 2
    movzx rcx, byte ptr [r12+2]
    shr rcx, 6
    or rbx, rcx
    movzx rcx, byte ptr [r12+2]
    and rcx, 63
    mov al, byte ptr [r13 + r9]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov al, byte ptr [r13 + r11]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov al, byte ptr [r13 + rbx]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov al, byte ptr [r13 + rcx]
    mov byte ptr [r10], al
    inc r10
    inc r14
    add r12, 3
    sub r8, 3
    jmp .encode_loop
.remainder:
    test r8, r8
    jz .done_encoding
    cmp r8, 2
    je .remainder2
    jmp .remainder1
.remainder2:
    movzx rax, byte ptr [r12]
    movzx rbx, byte ptr [r12+1]
    shr rax, 2
    and rax, 63
    mov r9, rax
    movzx rax, byte ptr [r12]
    and rax, 3
    shl rax, 4
    shr rbx, 4
    or rax, rbx
    mov r11, rax
    movzx rbx, byte ptr [r12+1]
    and rbx, 15
    shl rbx, 2
    mov al, byte ptr [r13 + r9]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov al, byte ptr [r13 + r11]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov al, byte ptr [r13 + rbx]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov byte ptr [r10], 61
    inc r10
    inc r14
    xor r8, r8
    jmp .done_encoding
.remainder1:
    movzx rax, byte ptr [r12]
    shr rax, 2
    and rax, 63
    mov r9, rax
    movzx rax, byte ptr [r12]
    and rax, 3
    shl rax, 4
    mov al, byte ptr [r13 + r9]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov al, byte ptr [r13 + rax]
    mov byte ptr [r10], al
    inc r10
    inc r14
    mov byte ptr [r10], 61
    inc r10
    inc r14
    mov byte ptr [r10], 61
    inc r10
    inc r14
    xor r8, r8
    jmp .done_encoding
.done_encoding:
    mov rdi, 1
    lea r12, [rip + output_buf]
.write_loop:
    test r14, r14
    jz .done_write
    cmp r14, 76
    jle .write_last
    mov rsi, r12
    mov rdx, 76
    syscall write
    add r12, 76
    sub r14, 76
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall write
    jmp .write_loop
.write_last:
    mov rsi, r12
    mov rdx, r14
    syscall write
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall write
.done_write:
    xor rdi, rdi
    mov rax, exit
    syscall
