

.equ read, 0
.equ write, 1
.equ exit, 60
.equ exit_group, 231
.equ input_size, 49152
.equ output_size, 66400

.globl _start

.section .rodata
base64_alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
input_buf:
    .space input_size
output_buf:
    .space output_size

.section .text
_start:
    xor r8, r8
    mov r9, 49152
.read_loop:
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, r9
    xor rdi, rdi
    mov rax, read
    syscall
    test rax, rax
    jz .read_done
    add r8, rax
    sub r9, rax
    jmp .read_loop
.read_done:
    test r8, r8
    jz .empty_input
    lea r12, [rip + output_buf]
    mov r10, r12
    xor r14, r14
    xor r9, r9
    lea rsi, [rip + input_buf]
.encode_loop:
    cmp r8, 3
    jge .full_group
    jmp .remainder
.full_group:
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]
.char0:
    shr eax, 2
    and eax, 63
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .char1
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.char1:
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    movzx ebx, byte ptr [rsi+1]
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .char2
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.char2:
    movzx eax, byte ptr [rsi+1]
    and eax, 15
    shl eax, 2
    movzx ebx, byte ptr [rsi+2]
    shr ebx, 6
    or eax, ebx
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .char3
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.char3:
    movzx eax, byte ptr [rsi+2]
    and eax, 63
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .group_done
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.group_done:
    add rsi, 3
    sub r8, 3
    jmp .encode_loop
.remainder:
    test r8, r8
    jz .done_encoding
    cmp r8, 2
    je .remainder2
    jmp .remainder1
.remainder2:
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
.rem2_char0:
    shr eax, 2
    and eax, 63
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .rem2_char1
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.rem2_char1:
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    movzx ebx, byte ptr [rsi+1]
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .rem2_char2
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.rem2_char2:
    movzx eax, byte ptr [rsi+1]
    and eax, 15
    shl eax, 2
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .rem2_eq
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.rem2_eq:
    mov byte ptr [r10], 61
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .done_encoding
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
    jmp .done_encoding
.remainder1:
    movzx eax, byte ptr [rsi]
.rem1_char0:
    shr eax, 2
    and eax, 63
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .rem1_char1
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.rem1_char1:
    movzx eax, byte ptr [rsi]
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rip + base64_alpha + rax]
    mov byte ptr [r10], al
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .rem1_eq1
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.rem1_eq1:
    mov byte ptr [r10], 61
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .rem1_eq2
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.rem1_eq2:
    mov byte ptr [r10], 61
    add r10, 1
    inc r14
    inc r9
    cmp r9, 76
    jne .done_encoding
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
    jmp .done_encoding
.done_encoding:
    test r9, r9
    jz .no_newline
    mov byte ptr [r10], 10
    add r10, 1
    inc r14
    xor r9, r9
.no_newline:
    mov rdi, 1
    mov rsi, r12
    mov rax, write
    mov rdx, r14
    syscall
    xor rdi, rdi
    mov rax, exit
    syscall
.empty_input:
    xor rdi, rdi
    mov rax, exit
    syscall
