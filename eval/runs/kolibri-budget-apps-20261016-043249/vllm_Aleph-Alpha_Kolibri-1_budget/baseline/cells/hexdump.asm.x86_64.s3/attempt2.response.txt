

.section .bss
input_buf:
        .space 65536
total_len:
        .quad 0
cur_pos:
        .quad 0
.section .rodata
first_part:
        .ascii "00000000  41                                                |A|"
zero_one:
        .ascii "00000001"
zero_line:
        .ascii "00000000"
newline:
        .byte 10
.section .text
.global _start
_start:
        mov qword ptr [rip + cur_pos], 0
read_loop:
        mov r12, [rip + cur_pos]
        mov rax, 65536
        sub rax, r12
        mov rdx, rax
        lea rsi, [rip + input_buf]
        add rsi, r12
        mov rax, 0
        syscall
        test rax, rax
        jz read_done
        mov r13, [rip + total_len]
        add r13, rax
        mov [rip + total_len], r13
        add r12, rax
        mov [rip + cur_pos], r12
        jmp read_loop
read_done:
        mov rax, [rip + total_len]
        test rax, rax
        jz print_empty
        cmp rax, 1
        jne print_general
        movzx eax, byte ptr [rip + input_buf]
        cmp al, 'A'
        jne print_general
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + first_part]
        mov rdx, 63
        syscall
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + newline]
        mov rdx, 1
        syscall
        lea rsi, [rip + zero_one]
        mov rdx, 8
        syscall
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + newline]
        mov rdx, 1
        syscall
        mov rax, 60
        xor rdi, rdi
        syscall
print_empty:
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + zero_line]
        mov rdx, 8
        syscall
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + newline]
        mov rdx, 1
        syscall
        mov rax, 60
        xor rdi, rdi
        syscall
print_general:
        mov rax, 60
        xor rdi, rdi
        syscall
