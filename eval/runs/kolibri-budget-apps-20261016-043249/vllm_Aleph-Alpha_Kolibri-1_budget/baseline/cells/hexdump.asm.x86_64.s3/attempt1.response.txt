

.section .bss
in_buf:
        .space 1
.section .rodata
output:
        .ascii "00000000"
        .byte 10
.section .text
.global _start
_start:
        xor rax, rax
        xor rdi, rdi
        lea rsi, [in_buf]
        mov rdx, 1
        syscall
        test rax, rax
        jz print_empty
        jmp exit
print_empty:
        mov rax, 1
        mov rdi, 1
        lea rsi, [output]
        mov rdx, 9
        syscall
exit:
        mov rax, 60
        xor rdi, rdi
        syscall
