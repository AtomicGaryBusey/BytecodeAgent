

.global _start

.section .rodata
zero_str:
   .ascii "00000000\n"

.section .data
total_len:
   .quad 0

.section .bss
buffer:
   .space 65536

.section .text
_start:
  mov rdi, 0
  mov rsi, offset buffer
  mov rdx, 65536
read_loop:
  mov rax, 0
  syscall
  test rax, rax
  jz done_reading
  mov r8, rax
  mov rax, [total_len]
  add rax, r8
  mov [total_len], rax
  mov rsi, offset buffer
  add rsi, rax
  jmp read_loop
done_reading:
  mov rax, [total_len]
  test rax, rax
  jz print_zero
  mov rdx, rax
  mov rax, 1
  mov rdi, 1
  mov rsi, offset buffer
  syscall
  jmp exit
print_zero:
  mov rax, 1
  mov rdi, 1
  mov rsi, offset zero_str
  mov rdx, 9
  syscall
exit:
  mov rax, 60
  xor rdi, rdi
  syscall
