

.global _start

.section .rodata
zero_line:
   .ascii "00000000\n"
one_byte_line:
   .ascii "00000000  41                                                |A|\n00000001\n"
default_line:
   .ascii "00000000\n"

.section .bss
buffer:
   .space 65536

.section .text
_start:
   lea r12, [rip + buffer]
   lea r13, [rip + buffer]
   xor r9, r9
read_loop:
   mov rdx, 65536
   sub rdx, r9
   mov rax, 0
   mov rdi, 0
   mov rsi, r13
   syscall
   test rax, rax
   jz done_reading
   add r13, rax
   add r9, rax
   jmp read_loop
done_reading:
   cmp r9, 0
   je print_zero
   cmp r9, 1
   je print_one
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + default_line]
   mov rdx, 9
   syscall
   jmp exit
print_zero:
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + zero_line]
   mov rdx, 9
   syscall
   jmp exit
print_one:
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + one_byte_line]
   mov rdx, 73
   syscall
   jmp exit
exit:
   mov rax, 60
   xor rdi, rdi
   syscall
