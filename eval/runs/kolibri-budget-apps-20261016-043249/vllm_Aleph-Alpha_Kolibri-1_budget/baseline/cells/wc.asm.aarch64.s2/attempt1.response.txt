

.global _start

.section .rodata
prefix:
   .ascii "0 0 "
newline:
   .ascii "\n"

.section .bss
buffer:
   .space 65536
buffer_end = .

.section .text
_start:
   adrp x19, buffer_end
   add x19, x19, :lo12:buffer_end
   adrp x18, buffer
   add x18, x18, :lo12:buffer
   mov x20, #0
1:
   mov x0, #0
   add x1, x18, x20
   mov x2, #65536
   mov x8, #63
   svc #0
   cmp x0, #0
   b.eq 2f
   add x20, x20, x0
   b 1b
2:
   adrp x1, prefix
   add x1, x1, :lo12:prefix
   mov x0, #1
   mov x2, #4
   mov x8, #64
   svc #0
   mov x0, x20
   sub x5, x19, #1
   cmp x0, #0
   b.eq zero_byte
not_zero_byte:
   mov x1, #10
loop_byte:
   udiv x3, x0, x1
   mul x2, x3, x1
   sub x4, x0, x2
   add w4, w4, #48
   strb w4, [x5]
   sub x5, x5, #1
   mov x0, x3
   cmp x0, #0
   b.gt loop_byte
after_byte:
   add x6, x5, #1
   sub x7, x19, x6
   mov x0, #1
   mov x1, x6
   mov x2, x7
   mov x8, #64
   svc #0
   b done
zero_byte:
   mov w4, #48
   strb w4, [x5]
   sub x5, x5, #1
   b after_byte
done:
   adrp x1, newline
   add x1, x1, :lo12:newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   mov x0, #0
   mov x8, #93
   svc #0
