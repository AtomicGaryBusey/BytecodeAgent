

.global _start

.section .rodata
space_byte:
   .byte 32
newline_byte:
   .byte 10

.section .bss
input_buffer:
   .space 65536
digit_buffer:
   .space 16

.section .text
_start:
   adrp x19, input_buffer
   add x19, x19, :lo12:input_buffer
   mov x20, #0

read_loop:
   mov x0, #0
   add x1, x19, x20
   mov x2, #65536
   mov x8, #63
   svc #0
   cmp x0, #0
   b.eq read_done
   add x20, x20, x0
   b read_loop

read_done:
   mov x21, #0
   mov x22, #0
   mov x23, #0
   mov x24, x19
   mov x25, x20

count_loop:
   cmp x25, #0
   b.eq count_done
   ldrb w0, [x24], #1
   dec x25, #1
   cmp w0, #10
   b.eq newline
   cmp w0, #32
   b.eq ws
   cmp w0, #9
   b.eq ws
   cmp w0, #11
   b.eq ws
   cmp w0, #12
   b.eq ws
   cmp w0, #13
   b.eq ws
   cmp x23, #0
   b.eq word_start
   b count_loop
word_start:
   add x22, x22, #1
   mov x23, #1
   b count_loop
ws:
   mov x23, #0
   b count_loop
newline:
   add x21, x21, #1
   mov x23, #0
   b count_loop

count_done:
   mov x0, x21
   bl print_number

   adrp x19, space_byte
   add x19, x19, :lo12:space_byte
   mov x0, #1
   mov x1, x19
   mov x2, #1
   mov x8, #64
   svc #0

   mov x0, x22
   bl print_number

   adrp x19, space_byte
   add x19, x19, :lo12:space_byte
   mov x0, #1
   mov x1, x19
   mov x2, #1
   mov x8, #64
   svc #0

   mov x0, x20
   bl print_number

   adrp x19, newline_byte
   add x19, x19, :lo12:newline_byte
   mov x0, #1
   mov x1, x19
   mov x2, #1
   mov x8, #64
   svc #0

   mov x0, #0
   mov x8, #93
   svc #0

print_number:
   adrp x1, digit_buffer
   add x1, x1, :lo12:digit_buffer
   add x1, x1, #16
   mov x2, x1
   cmp x0, #0
   b.eq print_zero

print_loop:
   udiv x3, x0, #10
   mul x4, x3, #10
   subs x0, x0, x4
   add w0, w0, #48
   strb w0, [x2, #-1]!
   mov x0, x3
   cmp x0, #0
   b.gt print_loop
   b print_write

print_zero:
   mov w0, #48
   strb w0, [x2, #-1]!

print_write:
   sub x5, x1, x2
   mov x0, #1
   mov x1, x2
   mov x2, x5
   mov x8, #64
   svc #0
   ret
