.section .text
_start:
    sub sp, sp, #65536
read_loop:
    mov x0, #0
    mov x1, sp
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done_reading
    add x1, x1, x0
    b read_loop
done_reading:
    sub x4, x1, sp
