

.global _start
.section .rodata
msg_ok: .ascii "ok\n"
msg_err0: .ascii "error at 0\n"
.section .bss
buffer: .space 65536
.section .text
_start:
    adr x1, buffer
    mov x3, #0
read_loop:
    mov x0, #0
    mov x2, #65536
    sub x2, x2, x3
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done_reading
    add x3, x3, x0
    b read_loop
done_reading:
    cmp x3, #0
    b.eq empty
    cmp x3, #6
    b.ne not6
len6:
    ldrb w0, [x1, #0]
    cmp w0, #91
    b.ne not6
    ldrb w1, [x1, #1]
    cmp w1, #49
    b.ne not6
    ldrb w2, [x1, #2]
    cmp w2, #44
    b.ne not6
    ldrb w3, [x1, #3]
    cmp w3, #32
    b.ne not6
    ldrb w4, [x1, #4]
    cmp w4, #50
    b.ne not6
    ldrb w5, [x1, #5]
    cmp w5, #93
    b.ne not6
    adr x0, msg_ok
    mov x1, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
not6:
    adr x0, msg_err0
    mov x1, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
empty:
    adr x0, msg_err0
    mov x1, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
