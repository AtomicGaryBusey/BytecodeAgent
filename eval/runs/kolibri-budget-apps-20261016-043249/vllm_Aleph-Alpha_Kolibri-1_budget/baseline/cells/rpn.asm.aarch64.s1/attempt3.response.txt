

.global _start

.section .bss
.balign 1
in_buf:
    .space 16

.section .rodata
.balign 1
error_msg:
    .ascii "error\n"
newline_char:
    .byte 0x0A

.text
_start:
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf

    mov x0, #0
    mov x1, x19
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq no_input

    ldr w2, [x19, #0]

    cmp w2, #'0'
    b.lt not_digit
    cmp w2, #'9'
    b.gt not_digit

    mov x0, #0
    mov x1, x19
    add x1, x1, #1
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq digit_only

    ldr w3, [x19, #1]
    cmp w3, #'\n'
    b.ne extra_char

    mov x0, #0
    mov x1, x19
    add x1, x1, #2
    mov x2, #1
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq digit_newline

    b error

digit_only:
    mov x0, #1
    mov x1, x19
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x3, newline_char
    add x3, x3, :lo12:newline_char
    mov x0, #1
    mov x1, x3
    mov x2, #1
    mov x8, #64
    svc #0
    b exit

digit_newline:
    mov x0, #1
    mov x1, x19
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x3, newline_char
    add x3, x3, :lo12:newline_char
    mov x0, #1
    mov x1, x3
    mov x2, #1
    mov x8, #64
    svc #0
    b exit

extra_char:
    b error

not_digit:
    b error

no_input:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit

error:
    adrp x3, error_msg
    add x3, x3, :lo12:error_msg
    mov x0, #1
    mov x1, x3
    mov x2, #6
    mov x8, #64
    svc #0
    b exit

exit:
    mov x0, #0
    mov x8, #93
    svc #0
