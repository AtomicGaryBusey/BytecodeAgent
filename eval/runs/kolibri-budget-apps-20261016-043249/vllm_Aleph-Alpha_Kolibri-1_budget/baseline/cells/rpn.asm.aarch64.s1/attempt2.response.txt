

.global _start

.section .bss
.balign 1
in_buf:
    .space 1

.section .rodata
.balign 1
error_msg:
    .ascii "error\n"

.text
_start:
    adr x21, error_msg
    adr x20, in_buf
    mov x1, x20
    mov x19, #0
    mov x2, #1
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    mov x19, #1
    b read_loop
read_done:
    cmp x19, #0
    b.eq exit_start
    mov x0, #1
    mov x1, x21
    mov x2, #6
    mov x8, #64
    svc #0
exit_start:
    mov x0, #0
    mov x8, #93
    svc #0
