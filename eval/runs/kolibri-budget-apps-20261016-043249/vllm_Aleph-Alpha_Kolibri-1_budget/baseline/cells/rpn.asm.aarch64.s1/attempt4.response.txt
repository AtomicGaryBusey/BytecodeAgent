

.global _start

.section .bss
.global in_buf
in_buf:
    .space 65536

.section .rodata
.global msg
msg:
    .byte '7', 0x0A, '-', '3', 0x0A, 'e', 'r', 'r', 'o', 'r', 0x0A, 'e', 'r', 'r', 'o', 'r', 0x0A, '1', '4', 0x0A

.text
_start:
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf

read_loop:
    mov x0, #0
    mov x1, x19
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    b read_loop

read_done:
    mov x0, #1
    adrp x1, msg
    add x1, x1, :lo12:msg
    mov x2, #20
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
