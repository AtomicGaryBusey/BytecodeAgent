

.globl _start

.section .rodata
newline: .byte 10

.section .bss
input_buf: .space 32768

.section .text
_start:
   lea rsi, [rip + input_buf]
   lea r13, [rip + input_buf]
   mov rdx, 32768
   xor r12, r12
read_loop:
   mov rax, 0
   syscall
   test rax, rax
   jz read_done
   add r12, rax
   add rsi, rax
   sub rdx, rax
   jmp read_loop
read_done:
   mov rsi, r13
   mov rax, 1
   mov rdi, 1
   mov rdx, r12
   syscall
   test r12, r12
   jz exit_now
   lea r14, [r13 + r12]
   dec r14
   mov al, byte ptr [r14]
   cmp al, 10
   jne no_newline
   jmp exit_now
no_newline:
   lea rsi, [rip + newline]
   mov rdx, 1
   mov rax, 1
   mov rdi, 1
   syscall
exit_now:
   mov rax, 60
   mov rdi, 0
   syscall
