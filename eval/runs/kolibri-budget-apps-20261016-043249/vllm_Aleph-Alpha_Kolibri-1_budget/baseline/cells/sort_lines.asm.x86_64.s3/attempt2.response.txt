

.globl _start

.section .rodata
newline: .byte 10

.section .bss
input_buf: .space 32768

.section .text
_start:
   xor r10, r10
   lea rsi, [rip + input_buf]
   mov rdx, 32768
read_loop:
   mov rax, 0
   syscall
   test rax, rax
   jz read_done
   add r10, rax
   add rsi, rax
   sub rdx, rax
   jmp read_loop
read_done:
   lea r11, [rip + input_buf]
   mov rax, 1
   mov rdi, 1
   mov rsi, r11
   mov rdx, r10
   syscall
   cmp r10, 0
   jle exit_now
   mov r12, r11
   add r12, r10
   sub r12, 1
   mov al, byte ptr [r12]
   cmp al, 10
   jne exit_now
   lea rsi, [rip + newline]
   mov rdx, 1
   mov rax, 1
   mov rdi, 1
   syscall
exit_now:
   mov rax, 60
   mov rdi, 0
   syscall
