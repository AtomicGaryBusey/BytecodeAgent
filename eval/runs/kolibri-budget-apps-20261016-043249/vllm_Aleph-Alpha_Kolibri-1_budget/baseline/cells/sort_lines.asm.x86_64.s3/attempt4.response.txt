

.globl _start

.section .bss
input_buf: .space 32768
starts: .space 8000
lens: .space 8000
total_len: .quad 0
line_count: .quad 0
line_start_offset: .quad 0
current_offset: .quad 0

.section .rodata
newline: .byte 10

.section .text
_start:
    lea rsi, [rip + input_buf]
    xor rdx, rdx
    mov rax, 0
    mov rdi, 0
    mov rdx, 32768
    syscall
    mov [rip + total_len], rax
    test rax, rax
    jz exit_now

    mov rax, [rip + total_len]
    mov r14, rax

    lea r13, [rip + input_buf]
    lea r9, [rip + starts]
    lea r8, [rip + lens]
    xor r12, r12
    xor r15, r15
    xor r10, r10

parse_loop:
    cmp r15, r14
    jge parse_done

    movzx eax, byte ptr [r13 + r15]
    cmp al, 10
    jne parse_no_newline

    lea rax, [r13 + r12]
    mov [r9 + r10*8], rax
    mov rcx, r15
    sub rcx, r12
    mov [r8 + r10*8], rcx
    inc r10
    inc r15
    mov r12, r15
    jmp parse_loop

parse_no_newline:
    inc r15
    jmp parse_loop

parse_done:
    cmp r12, r14
    jge no_remaining
    lea rax, [r13 + r12]
    mov [r9 + r10*8], rax
    mov rcx, r14
    sub rcx, r12
    mov [r8 + r10*8], rcx
    inc r10
no_remaining:

    cmp r10, 1
    jle sort_done

    mov r14, r10
    dec r14

    xor r11, r11

sort_outer:
    cmp r11, r14
    jge sort_done

    mov r12, r11
    mov r13, r11
    inc r13

sort_inner_loop:
    cmp r13, r10
    jge sort_inner_done

    mov rcx, [r8 + r12*8]
    mov r15, [r8 + r13*8]
    cmp rcx, 0
    jne min_len_nonzero
    cmp r15, 0
    jle both_zero
    jmp not_less
both_zero:
    jmp not_less
min_len_nonzero:
    cmp r15, 0
    jle j_len_zero
    mov rax, [r9 + r12*8]
    movzx eax, byte ptr [rax]
    mov rdx, [r9 + r13*8]
    movzx ecx, byte ptr [rdx]
    cmp al, cl
    jle not_less
    mov r12, r13
    jmp not_less
j_len_zero:
    mov r12, r13
not_less:
    inc r13
    jmp sort_inner_loop

sort_inner_done:
    cmp r12, r11
    jne sort_swap
    inc r11
    jmp sort_outer

sort_swap:
    mov rax, [r9 + r11*8]
    mov rcx, [r8 + r11*8]
    mov rdx, [r9 + r12*8]
    mov r15, [r8 + r12*8]
    mov [r9 + r11*8], rdx
    mov [r8 + r11*8], r15
    mov [r9 + r12*8], rax
    mov [r8 + r12*8], rcx
    inc r11
    jmp sort_outer

sort_done:
    test r10, r10
    jz output_done

    mov r11, 0
    mov rdi, 1

output_loop:
    cmp r11, r10
    jge output_done

    mov rsi, [r9 + r11*8]
    mov rdx, [r8 + r11*8]
    mov rax, 1
    syscall

    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall

    inc r11
    jmp output_loop

output_done:
    mov rax, 60
    mov rdi, 0
    syscall

exit_now:
    mov rax, 60
    mov rdi, 0
    syscall
