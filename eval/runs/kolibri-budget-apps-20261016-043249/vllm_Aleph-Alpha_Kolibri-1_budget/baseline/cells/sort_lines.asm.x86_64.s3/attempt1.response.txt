

.global _start

.section .bss
buf: .space 32768

.section .text
_start:
    mov rdi, 0
    mov rsi, offset buf
    mov rdx, 32768
    xor r12, r12
read_loop:
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add rsi, rax
    mov rdx, 32768
    jmp read_loop
read_done:
    mov rax, 1
    mov rdi, 1
    mov rsi, offset buf
    mov rdx, r12
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
