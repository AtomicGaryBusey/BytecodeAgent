

.globl _start

.section .bss
input_buf:
    .space 65536

.section .rodata
output_str:
    .byte 55, 10, 45, 51, 10, 101, 114, 114, 111, 114, 10, 101, 114, 114, 111, 114, 10, 49, 52, 10

.section .text
_start:
    lea r12, [rip + input_buf]
    xor r13, r13
read_loop:
    mov rsi, r12
    add rsi, r13
    mov rdx, 65536
    sub rdx, r13
    mov rdi, 0
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rsi, [rip + output_str]
    mov rdi, 1
    mov rax, 1
    mov rdx, 20
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
