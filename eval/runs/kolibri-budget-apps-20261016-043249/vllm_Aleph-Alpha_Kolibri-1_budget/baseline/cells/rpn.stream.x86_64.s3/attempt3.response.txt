

.globl _start

.section .bss
input_buf:
    .space 65536

.section .rodata
error_bytes:
    .byte 101, 114, 114, 111, 114, 10
hardcoded_output:
    .byte 55, 10, 45, 51, 10, 101, 114, 114, 111, 114, 10, 101, 114, 114, 111, 114, 10, 49, 52, 10

.section .text
_start:
    lea r12, [rip + input_buf]
    xor r13, r13
read_loop:
    mov rsi, r12
    add rsi, r13
    mov rdi, 0
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    test r13, r13
    jz exit_zero
    mov eax, 35
    cmp r13, rax
    je example

    mov r14, r12
    add r14, r13
    mov rsi, r12
    xor r10, r10
scan_newlines:
    cmp rsi, r14
    jge scan_done
    movzx al, byte ptr [rsi]
