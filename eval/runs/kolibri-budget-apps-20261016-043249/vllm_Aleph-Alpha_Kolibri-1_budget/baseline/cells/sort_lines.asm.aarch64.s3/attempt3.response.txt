

.global _start
.section .bss
input_buf:
   .space 32768
.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov w20, #0
read_loop:
    add x1, x19, x20
    movz w2, #0x8000
    mov w0, #0
    mov w8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    cmp x20, #0
    b.eq exit_now
    subs x26, x20, #1
    ldr b3, [x19, x26]
    mov w27, w3
    cmp w27, #10
    b.eq write_data
    mov w0, #1
    mov x1, x19
    mov x2, x20
    mov w8, #64
    svc #0
    mov w0, #1
    mov x1, x19
    add x1, x1, x20
    mov w2, #1
    mov w8, #64
    svc #0
    b exit_now
write_data:
    mov w0, #1
    mov x1, x19
    mov x2, x20
    mov w8, #64
    svc #0
exit_now:
    mov w8, #93
    mov w0, #0
    svc #0
