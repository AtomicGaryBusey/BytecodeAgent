

.global _start
.section .bss
  .align 16
buffer:
  .space 32768
.section .text
_start:
  movz x4, #0x800
  lsl x4, x4, #4
  adrp x20, buffer
  add x20, x20, :lo12:buffer
  mov x19, #0
read_loop:
  add x1, x20, x19
  sub x3, x4, x19
  mov x0, #0
  mov x8, #63
  svc #0
  cbz x0, read_done
  add x19, x19, x0
  b read_loop
read_done:
  mov x8, #64
  mov x0, #1
  mov x1, x20
  mov x2, x19
  svc #0
  mov x8, #93
  mov x0, #0
  svc #0
