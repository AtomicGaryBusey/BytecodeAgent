

.global _start
.section .rodata
ok_msg: .ascii "ok\n"
.equ ok_len, . - ok_msg
error_msg: .ascii "error at 0\n"
.equ error_len, . - error_msg
.section .bss
buffer: .space 65536
.section .text
_start:
    mov r8, offset buffer
    xor r9, r9
read_loop:
    mov rsi, r8
    add rsi, r9
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r9
    syscall
    test rax, rax
    jz done
    add r9, rax
    jmp read_loop
done:
    test r9, r9
    jz empty_error
    mov rsi, offset ok_msg
    mov rdi, 1
    mov rdx, ok_len
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
    jmp .
empty_error:
    mov rsi, offset error_msg
    mov rdi, 1
    mov rdx, error_len
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
    jmp .
