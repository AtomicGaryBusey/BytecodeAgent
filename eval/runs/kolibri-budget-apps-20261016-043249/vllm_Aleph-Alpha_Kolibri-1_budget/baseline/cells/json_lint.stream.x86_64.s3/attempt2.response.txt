

.global _start
.section .rodata
ok_msg: .ascii "ok\n"
error_msg: .ascii "error at 0\n"
.section .bss
buffer: .space 65536
.section .text
_start:
    lea rbx, [rip + buffer]
    mov rax, 0
    mov rdi, 0
    mov rdx, 65536
read_loop:
    mov rsi, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    lea rax, [rip + buffer]
    sub rbx, rax
    test rbx, rbx
    jz error_zero
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
error_zero:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, 11
    syscall
    mov rax, 60
    mov rdi, 1
    syscall
