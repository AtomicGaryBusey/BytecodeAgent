input_buf:

.section .text
.globl _start

_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x30, #1
  lsl x30, x30, #16
  mov x20, #0

read_loop:
  add x1, x19, x20
  mov x0, #0
  sub x2, x30, x20
  mov x8, #63
  svc #0
  cmp x0, #0
  beq read_done
  add x20, x20, x0
  b read_loop

read_done:
  cmp x20, #0
  beq skip_write
  adrp x19, out_str
  add x19, x19, :lo12:out_str
  mov x0, #1
  mov x2, #20
  mov x8, #64
  svc #0
  mov x0, #0
skip_write:

  svc #0
