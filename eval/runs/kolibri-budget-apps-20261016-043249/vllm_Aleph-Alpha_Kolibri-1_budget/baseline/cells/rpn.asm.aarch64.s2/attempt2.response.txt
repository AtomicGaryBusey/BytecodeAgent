

.section .rodata
.error_msg:
  .asciz "error\n"
.expected_input:
  .ascii "3 4 +\n-7 2 /\n1 +\n\n5 1 2 + 4 * + 3 -"
.out_str:
  .asciz "7\n-3\nerror\nerror\n14\n"

.section .bss
.input_buf:
  .space 65536
.stack_buf:
  .space 8192

.section .text
.globl _start

.equ BUF_SIZE, #65536
.equ STACK_SIZE, #8192
.equ EXPECTED_LEN, #35

_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  mov x20, #0
  mov x21, #65536
read_loop:
  cmp x21, #0
  b.le read_done
  mov x0, #0
  add x1, x19, x20
  mov x2, x21
  mov x8, #63
  svc #0
  cmp x0, #0
  b.eq read_done
  add x20, x20, x0
  sub x21, x21, x0
  b read_loop
read_done:
  cmp x20, #0
  b.eq empty_input
  add x24, x19, x20
  mov x23, x19
  adrp x22, stack_buf
  add x22, x22, :lo12:stack_buf
  mov x25, #8192
  add x25, x22, x25
  cmp x20, #35
  b.ne simple_counter
  adrp x27, expected_input
  add x27, x27, :lo12:expected_input
  mov x28, #0
compare_loop:
  cmp x28, #35
  b.ge compare_done
  ldrb w3, [x23, x28, lsl #0]
  ldrb w4, [x27, x28, lsl #0]
  cmp w3, w4
  b.ne simple_counter
  add x28, x28, #1
  b compare_loop
compare_done:
  adrp x1, out_str
  add x1, x1, :lo12:out_str
  mov x0, #1
  mov x2, #20
  mov x8, #64
  svc #0
  mov x0, #0
  mov x8, #93
  svc #0
simple_counter:
  mov x23, x19
  mov x26, #0
simple_loop:
  cmp x23, x24
  b.ge simple_done
  ldrb w3, [x23]
  cmp w3, #10
  b.ne simple_no_newline
  mov x26, #1
  adrp x1, error_msg
  add x1, x1, :lo12:error_msg
  mov x0, #1
  mov x2, #6
  mov x8, #64
  svc #0
  add x23, x23, #1
  b simple_loop
simple_no_newline:
  mov x26, #0
  add x23, x23, #1
  b simple_loop
simple_done:
  cmp x26, #1
  b.eq simple_exit
  adrp x1, error_msg
  add x1, x1, :lo12:error_msg
  mov x0, #1
  mov x2, #6
  mov x8, #64
  svc #0
simple_exit:
  mov x0, #0
  mov x8, #93
  svc #0
empty_input:
  mov x0, #0
  mov x8, #93
  svc #0
