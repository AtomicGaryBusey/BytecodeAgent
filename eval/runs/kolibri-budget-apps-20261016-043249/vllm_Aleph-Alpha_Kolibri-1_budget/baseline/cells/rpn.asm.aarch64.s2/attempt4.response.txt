.section .text
.globl _start

_start:
  adr x19, input_buf
  mov x20, #65536
  mov x21, #0
read_loop:

  b.ge read_done
  mov x0, #0
  add x1, x19, x21
  sub x2, x20, x21

  cmp x0, #0
  b.le read_done

  mov x8, #93
