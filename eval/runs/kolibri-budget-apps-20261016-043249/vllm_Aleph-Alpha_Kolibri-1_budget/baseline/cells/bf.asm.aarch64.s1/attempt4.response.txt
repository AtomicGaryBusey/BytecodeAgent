

.global _start

.section .bss
in_buf:
    .space 65536

.section .text
_start:
    adrp x3, in_buf
    add x3, x3, :lo12:in_buf
read_loop:
    mov x0, #0
    mov x1, x3
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    b read_loop
read_done:
    mov x0, #0
    mov x8, #93
    svc #0
