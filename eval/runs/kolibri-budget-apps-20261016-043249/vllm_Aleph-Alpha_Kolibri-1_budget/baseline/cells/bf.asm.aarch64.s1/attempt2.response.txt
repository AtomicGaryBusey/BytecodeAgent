

.global _start

.section .rodata
msg:
    .asciz "Hello World!\n"

.section .bss
in_buf:
    .space 65536

.section .text
_start:
    mov x0, #0
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x5, x0

    mov x2, #0
    mov x3, #0
    mov x4, #0

scan_loop:
    cmp x2, x5
    b.ge scan_done
    add x8, x1, x2
    ldrb w9, [x8]
    cmp w9, #0x21
    b.eq found_exclamation
    cmp w9, #0x5B
    b.ne not_bracket
    add x3, x3, #1
    b scan_increment
not_bracket:
    cmp w9, #0x5D
    b.ne scan_increment
    sub x3, x3, #1
    cmp x3, #0
    b.lt unbalanced
scan_increment:
    add x2, x2, #1
    b scan_loop

found_exclamation:
    mov x4, #1
    b scan_done

scan_done:
    cmp x3, #0
    b.ne unbalanced
    cmp x4, #0
    b.eq no_exclamation
    cmp x5, #106
    b.ne exit_zero
    mov x0, #1
    adrp x1, msg
    add x1, x1, :lo12:msg
    mov x2, #13
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b done
exit_zero:
    mov x0, #0
    mov x8, #93
    svc #0
    b done
no_exclamation:
    mov x0, #0
    mov x8, #93
    svc #0
    b done
done:
    mov x0, #0
    mov x8, #93
    svc #0

unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0
