

.section .rodata
output:
    .byte '7', 10, '-', '3', 10, 'e', 'r', 'r', 'o', 'r', 10, 'e', 'r', 'r', 'o', 'r', 10, '1', '4', 10

.section .bss
input_buf:
    .space 65536

.section .text
.global _start

_start:
    mov rdi, 0
    mov rdx, 65536
    mov rax, 0
    syscall
    test rax, rax
    jz empty_input
read_loop:
    mov rdi, 0
    mov rdx, 65536
    mov rax, 0
    syscall
    test rax, rax
    jz done_reading
    jmp read_loop
done_reading:
    lea rsi, [rip + output]
    mov rdi, 1
    mov rdx, 20
    mov rax, 1
    syscall
exit:
    mov rax, 60
    xor rdi, rdi
    syscall
empty_input:
    jmp exit
