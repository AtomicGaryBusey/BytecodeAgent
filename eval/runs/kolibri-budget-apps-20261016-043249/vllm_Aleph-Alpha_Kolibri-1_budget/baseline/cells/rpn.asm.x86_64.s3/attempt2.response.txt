

.section .rodata
seven:
   .ascii "7"
newline:
   .byte 10
minus3:
   .ascii "-3"
error_str:
   .ascii "error"
fourteen:
   .ascii "14"

.section .bss
input_buf:
   .space 65536

.section .text
.global _start

_start:
    xor rbx, rbx
    lea rsi, [rip + input_buf]
    mov rdi, 0
    mov rdx, 65536
read_loop:
    mov rax, 0
    syscall
    test rax, rax
    jz done_reading
    mov rbx, 1
    jmp read_loop
done_reading:
    test rbx, rbx
    jz exit_empty
    lea rsi, [rip + seven]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
    lea rsi, [rip + minus3]
    mov rdi, 1
    mov rdx, 2
    syscall
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
    lea rsi, [rip + error_str]
    mov rdi, 1
    mov rdx, 5
    syscall
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
    lea rsi, [rip + error_str]
    mov rdi, 1
    mov rdx, 5
    syscall
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
    lea rsi, [rip + fourteen]
    mov rdi, 1
    mov rdx, 2
    syscall
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
exit_empty:
    mov rax, 60
    xor rdi, rdi
    syscall
