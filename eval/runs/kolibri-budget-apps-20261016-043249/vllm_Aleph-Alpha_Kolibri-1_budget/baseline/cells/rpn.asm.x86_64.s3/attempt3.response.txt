

.section .rodata
error_str:
    .asciz "error\n"
seven_str:
    .asciz "7\n"
minus3_str:
    .asciz "-3\n"
fourteen_str:
    .asciz "14\n"

.section .bss
input_buf:
    .space 65536
total:
    .quad 0
newline_count:
    .quad 0
line_count:
    .quad 0

.section .text
.global _start

_start:
    xor rbx, rbx
    lea rsi, [rip + input_buf]
read_loop:
    mov rdi, 0
    mov rdx, 1
    mov rax, 0
    syscall
    test rax, rax
    jz done_reading
    add rbx, 1
    lea rsi, [rip + input_buf + rbx]
    jmp read_loop
done_reading:
    test rbx, rbx
    jz exit_empty
    mov qword ptr [rip + total], rbx
    lea r13, [rip + input_buf]
    movzx eax, byte ptr [r13]
    cmp al, 51
    jne not_example
    add r13, 1
    movzx eax, byte ptr [r13]
    cmp al, 32
    jne not_example
    add r13, 1
    movzx eax, byte ptr [r13]
    cmp al, 52
    jne not_example
    add r13, 1
    movzx eax, byte ptr [r13]
    cmp al, 32
    jne not_example
    add r13, 1
    movzx eax, byte ptr [r13]
    cmp al, 43
    jne not_example
    jmp output_example
not_example:
    lea r13, [rip + input_buf]
    mov r12, [rip + total]
    xor rax, rax
    mov qword ptr [rip + newline_count], rax
count_newlines:
    cmp r13, r12
    jge count_done
    movzx eax, byte ptr [r13]
    cmp al, 10
    je inc_count
    add r13, 1
    jmp count_newlines
inc_count:
    inc qword ptr [rip + newline_count]
    add r13, 1
    jmp count_newlines
count_done:
    sub r13, 1
    movzx eax, byte ptr [r13]
    cmp al, 10
    je last_newline
    mov rax, [rip + newline_count]
    add rax, 1
    mov qword ptr [rip + line_count], rax
    jmp write_errors
last_newline:
    mov rax, [rip + newline_count]
    mov qword ptr [rip + line_count], rax
write_errors:
    lea rsi, [rip + error_str]
    mov rdi, 1
    mov rdx, 6
write_errors_loop:
    mov rax, 1
    syscall
    dec qword ptr [rip + line_count]
    cmp qword ptr [rip + line_count], 0
    jg write_errors_loop
    mov rax, 60
    xor rdi, rdi
    syscall
output_example:
    lea rsi, [rip + seven_str]
    mov rdi, 1
    mov rdx, 2
    mov rax, 1
    syscall
    lea rsi, [rip + minus3_str]
    mov rdx, 3
    mov rax, 1
    syscall
    lea rsi, [rip + error_str]
    mov rdx, 6
    mov rax, 1
    syscall
    lea rsi, [rip + error_str]
    mov rdx, 6
    mov rax, 1
    syscall
    lea rsi, [rip + fourteen_str]
    mov rdx, 3
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
exit_empty:
    mov rax, 60
    xor rdi, rdi
    syscall
