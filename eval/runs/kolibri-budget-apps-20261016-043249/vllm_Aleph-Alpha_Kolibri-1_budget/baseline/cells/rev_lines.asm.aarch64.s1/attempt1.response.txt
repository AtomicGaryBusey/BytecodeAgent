

.global _start

.section .bss
input_buf:
    .space 65536

.section .data
newline:
    .byte 0x0a

.section .text
_start:
    adrp x4, input_buf
    add x4, x4, :lo12:input_buf
    mov x5, #0
    movz x6, #1
    lsl x6, x6, #16
read_loop:
    sub x2, x6, x5
    mov x0, #0
    add x1, x4, x5
    mov x8, #63
    svc #0
    cbz x0, done_reading
    add x5, x5, x0
    b read_loop
done_reading:
    adrp x7, newline
    add x7, x7, :lo12:newline
    mov x9, #0
    mov x10, #0
scan_loop:
    cmp x9, x5
    bge done_scan
    ldrb w11, [x4, x9, lsl #0]
    cmp w11, #10
    bne not_newline
    sub x12, x9, x10
    cmp x12, #0
    ble skip_rev
    sub x11, x9, #1
rev_loop:
    cmp x11, x10
    blt done_rev
    mov x0, #1
    add x1, x4, x11
    mov x2, #1
    mov x8, #64
    svc #0
    sub x11, x11, #1
    b rev_loop
done_rev:
skip_rev:
    mov x0, #1
    mov x1, x7
    mov x2, #1
    mov x8, #64
    svc #0
    add x10, x9, #1
    add x9, x9, #1
    b scan_loop
not_newline:
    add x9, x9, #1
    b scan_loop
done_scan:
    cmp x10, x5
    bge done
    sub x12, x5, x10
    cmp x12, #0
    ble final_skip_rev
    sub x11, x5, #1
final_rev_loop:
    cmp x11, x10
    blt done_final_rev
    mov x0, #1
    add x1, x4, x11
    mov x2, #1
    mov x8, #64
    svc #0
    sub x11, x11, #1
    b final_rev_loop
done_final_rev:
final_skip_rev:
    mov x0, #1
    mov x1, x7
    mov x2, #1
    mov x8, #64
    svc #0
done:
    mov x0, #0
    mov x8, #93
    svc #0
