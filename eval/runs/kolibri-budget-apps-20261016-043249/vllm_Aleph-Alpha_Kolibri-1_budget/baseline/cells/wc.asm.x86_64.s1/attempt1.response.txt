

.global _start
.section .data
.global space
.global newline
.global out_buf
.global digits_buf
space:
   .byte 0x20
newline:
   .byte 0x0A
out_buf:
   .space 64
digits_buf:
   .space 16
.section .bss
.global input_buf
input_buf:
   .space 65536
.section .text
_start:
   xor r12, r12
   xor r13, r13
   xor r14, r14
   xor r15, r15
   mov rbx, offset input_buf
read_loop:
   mov rax, 0
   mov rdi, 0
   mov rsi, rbx
   mov rdx, 65536
   syscall
   mov rcx, rax
   test rcx, rcx
   jz read_done
   mov r10, rbx
inner_loop:
   movzx eax, byte ptr [r10]
   cmp eax, 0x0A
   je newline_inc
   jmp check_whitespace
newline_inc:
   inc r13
check_whitespace:
   cmp eax, 0x20
   je is_whitespace
   cmp eax, 0x09
   je is_whitespace
   cmp eax, 0x0A
   je is_whitespace
   cmp eax, 0x0B
   je is_whitespace
   cmp eax, 0x0C
   je is_whitespace
   cmp eax, 0x0D
   je is_whitespace
   jmp non_whitespace_check
is_whitespace:
   xor r15, r15
   jmp increment
non_whitespace_check:
   test r15, r15
   jz non_word_start
   jmp increment
non_word_start:
   inc r14
   mov r15, 1
increment:
   inc r12
   inc r10
   dec rcx
   jnz inner_loop
   jmp read_loop
read_done:
   mov r10, offset out_buf
   mov rsi, r10
   test r13, r13
   jz lines_zero
lines_nonzero:
   mov r8, offset digits_buf
   add r8, 16
   mov r11, r8
divide_loop1:
   mov rax, r13
   xor rdx, rdx
   mov r9, 10
   div r9
   add rdx, '0'
   dec r8
   mov byte ptr [r8], dl
   mov r13, rax
   test r13, r13
   jnz divide_loop1
   mov rax, r11
   sub rax, r8
   mov rcx, rax
copy_loop1:
   mov al, byte ptr [r8]
   mov byte ptr [rsi], al
   inc r8
   inc rsi
   dec rcx
   jnz copy_loop1
   mov rdi, 1
   mov rsi, r10
   mov rdx, rax
   syscall
   jmp after_lines
lines_zero:
   mov byte ptr [rsi], '0'
   mov rax, 1
   mov rdi, 1
   mov rsi, r10
   mov rdx, rax
   syscall
after_lines:
   mov rdi, 1
   mov rsi, offset space
   mov rdx, 1
   syscall
   mov r10, offset out_buf
   mov rsi, r10
   test r14, r14
   jz words_zero
words_nonzero:
   mov r8, offset digits_buf
   add r8, 16
   mov r11, r8
divide_loop2:
   mov rax, r14
   xor rdx, rdx
   mov r9, 10
   div r9
   add rdx, '0'
   dec r8
   mov byte ptr [r8], dl
   mov r14, rax
   test r14, r14
   jnz divide_loop2
   mov rax, r11
   sub rax, r8
   mov rcx, rax
copy_loop2:
   mov al, byte ptr [r8]
   mov byte ptr [rsi], al
   inc r8
   inc rsi
   dec rcx
   jnz copy_loop2
   mov rdi, 1
   mov rsi, r10
   mov rdx, rax
   syscall
   jmp after_words
words_zero:
   mov byte ptr [rsi], '0'
   mov rax, 1
   mov rdi, 1
   mov rsi, r10
   mov rdx, rax
   syscall
after_words:
   mov rdi, 1
   mov rsi, offset space
   mov rdx, 1
   syscall
   mov r10, offset out_buf
   mov rsi, r10
   test r12, r12
   jz bytes_zero
bytes_nonzero:
   mov r8, offset digits_buf
   add r8, 16
   mov r11, r8
divide_loop3:
   mov rax, r12
   xor rdx, rdx
   mov r9, 10
   div r9
   add rdx, '0'
   dec r8
   mov byte ptr [r8], dl
   mov r12, rax
   test r12, r12
   jnz divide_loop3
   mov rax, r11
   sub rax, r8
   mov rcx, rax
copy_loop3:
   mov al, byte ptr [r8]
   mov byte ptr [rsi], al
   inc r8
   inc rsi
   dec rcx
   jnz copy_loop3
   mov rdi, 1
   mov rsi, r10
   mov rdx, rax
   syscall
   jmp after_bytes
bytes_zero:
   mov byte ptr [rsi], '0'
   mov rax, 1
   mov rdi, 1
   mov rsi, r10
   mov rdx, rax
   syscall
after_bytes:
   mov rdi, 1
   mov rsi, offset newline
   mov rdx, 1
   syscall
   mov rax, 60
   xor rdi, rdi
   syscall
