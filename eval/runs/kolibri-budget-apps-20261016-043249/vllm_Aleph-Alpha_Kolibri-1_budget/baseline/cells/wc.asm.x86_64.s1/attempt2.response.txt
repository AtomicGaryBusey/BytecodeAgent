

.global _start
.section .rodata
space:
   .byte 0x20
newline:
   .byte 0x0A
.section .bss
input_buf:
   .space 65536
out_buf:
   .space 64
digits_buf:
   .space 16
.section .text
_start:
   xor r12, r12
   xor r13, r13
   xor r14, r14
   xor r15, r15
   mov rbx, offset input_buf
   mov r10, rbx
read_loop:
   mov rax, 0
   mov rdi, 0
   mov rsi, r10
   mov rdx, 65536
   syscall
   mov rcx, rax
   test rcx, rcx
   jz read_done
   mov r9, rcx
process_loop:
   movzx eax, byte ptr [r10]
   inc r12
   cmp eax, 0x0A
   je line_inc
   cmp eax, 0x20
   je is_ws
   cmp eax, 0x09
   je is_ws
   cmp eax, 0x0B
   je is_ws
   cmp eax, 0x0C
   je is_ws
   cmp eax, 0x0D
   je is_ws
   test r15, r15
   jz word_start
   jmp process_end
line_inc:
   inc r13
   xor r15, r15
   jmp process_end
is_ws:
   xor r15, r15
   jmp process_end
word_start:
   inc r14
   mov r15, 1
process_end:
   inc r10
   dec r9
   jnz process_loop
   jmp read_loop
read_done:
   mov r10, offset out_buf
   mov r15, r10
test r13, r13
jz lines_zero
lines_nonzero:
   mov r8, offset digits_buf
   add r8, 16
   mov r9, r8
convert_lines:
   xor rdx, rdx
   mov r11, 10
   div r11
   add dl, '0'
   dec r8
   mov byte ptr [r8], dl
   test rax, rax
   jnz convert_lines
   mov rcx, r9
   sub rcx, r8
copy_lines:
   mov al, byte ptr [r8]
   mov byte ptr [r10], al
   inc r8
   inc r10
   dec rcx
   jnz copy_lines
   jmp after_lines
lines_zero:
   mov byte ptr [r10], '0'
   inc r10
   jmp after_lines
after_lines:
   mov byte ptr [r10], ' '
   inc r10
mov r13, r14
test r13, r13
jz words_zero
words_nonzero:
   mov r8, offset digits_buf
   add r8, 16
   mov r9, r8
convert_words:
   xor rdx, rdx
   mov r11, 10
   div r11
   add dl, '0'
   dec r8
   mov byte ptr [r8], dl
   test rax, rax
   jnz convert_words
   mov rcx, r9
   sub rcx, r8
copy_words:
   mov al, byte ptr [r8]
   mov byte ptr [r10], al
   inc r8
   inc r10
   dec rcx
   jnz copy_words
   jmp after_words
words_zero:
   mov byte ptr [r10], '0'
   inc r10
   jmp after_words
after_words:
   mov byte ptr [r10], ' '
   inc r10
mov r13, r12
test r13, r13
jz bytes_zero
bytes_nonzero:
   mov r8, offset digits_buf
   add r8, 16
   mov r9, r8
convert_bytes:
   xor rdx, rdx
   mov r11, 10
   div r11
   add dl, '0'
   dec r8
   mov byte ptr [r8], dl
   test rax, rax
   jnz convert_bytes
   mov rcx, r9
   sub rcx, r8
copy_bytes:
   mov al, byte ptr [r8]
   mov byte ptr [r10], al
   inc r8
   inc r10
   dec rcx
   jnz copy_bytes
   jmp after_bytes
bytes_zero:
   mov byte ptr [r10], '0'
   inc r10
   jmp after_bytes
after_bytes:
   mov byte ptr [r10], 0x0A
   inc r10
mov rdx, r10
sub rdx, r15
mov rax, 1
mov rdi, 1
mov rsi, r15
syscall
mov rax, 60
xor rdi, rdi
syscall
