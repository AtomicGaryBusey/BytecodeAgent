

.global _start

.section .rodata
bufsize:
    .quad 4096

.section .bss
copy_buf:
    .space 4096

.section .text
_start:
    adrp x24, bufsize
    add x24, x24, :lo12:bufsize
    ldr x24, [x24]
    adrp x19, copy_buf
    add x19, x19, :lo12:copy_buf
copy_loop:
    mov x0, #0
    mov x1, x19
    mov x2, x24
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le copy_done
    mov x2, x0
    mov x0, #1
    mov x1, x19
    mov x8, #64
    svc #0
    b copy_loop
copy_done:
    mov x0, #0
    mov x1, #0
    mov x2, #0
    mov x8, #93
    svc #0
