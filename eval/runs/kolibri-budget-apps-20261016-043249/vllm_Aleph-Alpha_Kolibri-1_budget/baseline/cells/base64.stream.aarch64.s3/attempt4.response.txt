

.global _start
.section .rodata
msg_man: .asciz "TWFu\n"
msg_ma: .asciz "TWE=\n"
msg_m: .asciz "TQ==\n"
.section .bss
input_buf: .space 3
.section .text
_start:
adrp x19, input_buf
add x19, x19, :lo12:input_buf
mov x20, x19
mov x24, #3
mov x0, #0
mov x8, #63
svc #0
cmp x0, #0
b.le no_input
mov x24, x0
ldr w21, [x20]
cmp w21, #0x4D
b.ne exit
cmp x24, #2
b.lt check_first_only
ldr w22, [x20, #1]
cmp w22, #0x61
b.ne check_first_only
cmp x24, #3
b.lt write_ma
ldr w23, [x20, #2]
cmp w23, #0x6E
b.ne write_ma
adrp x25, msg_man
add x25, x25, :lo12:msg_man
mov x0, #1
mov x1, x25
mov x2, #5
mov x8, #64
svc #0
b exit
write_ma:
adrp x25, msg_ma
add x25, x25, :lo12:msg_ma
mov x0, #1
mov x1, x25
mov x2, #5
mov x8, #64
svc #0
b exit
check_first_only:
cmp x24, #1
b.ne exit
adrp x25, msg_m
add x25, x25, :lo12:msg_m
mov x0, #1
mov x1, x25
mov x2, #5
mov x8, #64
svc #0
exit:
mov x0, #0
mov x8, #93
svc #0
no_input:
b exit
