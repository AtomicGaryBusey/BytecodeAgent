

.global _start

.section .rodata
base64_alphabet: .asciz "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
input_buf: .space 49152
output_buf: .space 70000

.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf
    mov x23, #49152
    mov x21, #0

read_loop:
    cmp x21, x23
    b.ge done_read
    sub x2, x23, x21
    mov x1, x19
    add x1, x1, x21
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_read
    add x21, x21, x0
    b read_loop

done_read:
    cmp x21, #0
    b.eq no_input
    mov x22, x19
    mov x24, x20
    mov x25, #0
    adrp x29, base64_alphabet
    add x29, x29, :lo12:base64_alphabet

encoding_loop:
    cmp x21, #3
    b.le last_group
    ldr w26, [x22]
    ldr w27, [x22, #1]
    ldr w28, [x22, #2]
    lsl w27, w27, #8
    orr w27, w27, w28
    lsl w26, w26, #16
    orr x30, w26, w27
