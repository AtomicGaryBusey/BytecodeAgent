

.global _start

.section .rodata
newline: .byte 10

.section .bss
input_buf: .space 32768
line_starts: .space 8000
line_lens: .space 8000

.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, line_starts
    add x20, x20, :lo12:line_starts
    adrp x21, line_lens
    add x21, x21, :lo12:line_lens
    mov x4, #0
    mov x5, #0
    mov x6, #0
    mov x7, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x5
    mov x2, #4096
    svc #0
    cbz x0, done_read
    add x5, x5, x0
    b read_loop

done_read:
    mov x4, x5

parse_loop:
    cmp x5, x4
    b.ge parse_end
    ldrb w8, [x19, x5]
    mov w9, #10
    cmp w8, w9
    b.ne parse_not_newline
    lsl x10, x7, #3
    add x10, x20, x10
    add x13, x19, x6
    str x13, [x10]
    lsl x11, x7, #3
    add x11, x21, x11
    sub x12, x5, x6
    str x12, [x11]
    add x7, x7, #1
    add x6, x5, #1
    add x5, x5, #1
    b parse_loop
parse_not_newline:
    add x5, #1
