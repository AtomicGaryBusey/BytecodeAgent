

.global _start

.section .rodata
newline: .byte 10

.section .bss
.balign 16
input_buf: .space 32768

.section .text
_start:
    adrp x4, input_buf
    add x4, x4, :lo12:input_buf
    mov x5, #0
read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x4, x5
    mov x2, #4096
    svc #0
    cbz x0, done_read
    add x5, x5, x0
    b read_loop
done_read:
    mov x8, #64
    mov x0, #1
    mov x1, x4
    mov x2, x5
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0
