

.global _start

.section .rodata
newline: .byte 10

.section .bss
input_buf: .space 32768
line_starts: .space 8000
line_lens: .space 8000

.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, line_starts
    add x20, x20, :lo12:line_starts
    adrp x21, line_lens
    add x21, x21, :lo12:line_lens
    adrp x3, newline
    add x3, x3, :lo12:newline
    mov x4, #0
    mov x5, #0
    mov x6, #0
    mov x7, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x5
    mov x2, #4096
    svc #0
    cbz x0, done_read
    add x5, x5, x0
    b read_loop

done_read:
    mov x4, x5
parse_loop:
    cmp x5, x4
    b.ge parse_end
    ldrb w8, [x19, x5]
    mov w9, #10
    cmp w8, w9
    b.ne parse_not_newline
    lsl x10, x7, #3
    add x10, x20, x10
    add x13, x19, x6
    str x13, [x10]
    lsl x11, x7, #3
    add x11, x21, x11
    sub x12, x5, x6
    str x12, [x11]
    add x7, x7, #1
    add x5, x5, #1
    mov x6, x5
    b parse_loop
parse_not_newline:
    add x5, x5, #1
    b parse_loop
parse_end:
    cmp x6, x4
    b.ge parse_no_more
    lsl x10, x7, #3
    add x10, x20, x10
    add x13, x19, x6
    str x13, [x10]
    lsl x11, x7, #3
    add x11, x21, x11
    sub x12, x4, x6
    str x12, [x11]
    add x7, x7, #1
parse_no_more:
    mov x13, #0
print_loop:
    cmp x13, x7
    b.ge print_done
    lsl x10, x13, #3
    add x10, x20, x10
    ldr x1, [x10]
    lsl x11, x13, #3
    add x11, x21, x11
    ldr x2, [x11]
    mov x8, #64
    mov x0, #1
    svc #0
    mov x2, #1
    mov x1, x3
    svc #0
    add x13, x13, #1
    b print_loop
print_done:
    mov x8, #93
    mov x0, #0
    svc #0
