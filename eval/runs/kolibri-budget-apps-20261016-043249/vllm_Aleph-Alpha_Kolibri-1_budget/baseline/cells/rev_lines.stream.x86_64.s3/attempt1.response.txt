

.globl _start

.section .data
newline: .byte 10

.section .bss
buf: .space 65536

.section .text
.equ BUF_SIZE, 65536
_start:
    mov r8, 0
    lea r12, [rip + buf]
read_loop:
    mov rdi, 0
    lea rsi, [r12 + r8]
    mov rdx, BUF_SIZE
    sub rdx, r8
    syscall
    test rax, rax
    jz done_reading
    add r8, rax
    jmp read_loop
done_reading:
    test r8, r8
    jz exit
    lea r13, [r12 + r8]
    lea r10, [r12]
    mov r9, 0
scan_loop:
    cmp r10, r13
    jge end_of_input
    movzx eax, byte ptr [r10]
    cmp eax, 10
    je newline_found
    inc r10
    jmp scan_loop

newline_found:
    mov r15, r10
    mov r14, r15
    sub r14, r12
    sub r14, r9
    dec r10
    lea r11, [r12 + r9]
rev_loop:
    cmp r11, r10
    jge rev_done
    xchg byte ptr [r11], byte ptr [r10]
