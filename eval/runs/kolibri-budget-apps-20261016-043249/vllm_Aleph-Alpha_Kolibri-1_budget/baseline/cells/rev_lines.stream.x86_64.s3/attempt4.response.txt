

.globl _start

.section .rodata
newline: .byte 10

.section .bss
buf: .space 65536

.section .text
_start:
    lea r12, [rip + buf]
    xor r8, r8
read_loop:
    mov rdi, 0
    mov rsi, r12
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz done_reading
    add r8, rax
    cmp r8, 65536
    jl read_loop
done_reading:
    mov r13, r12
    add r13, r8
    mov r10, r12
    mov r11, r12
scan_loop:
    cmp r10, r13
    jge end_of_input
    movzx eax, byte ptr [r10]
    cmp eax, 10
    je found_newline
    inc r10
    jmp scan_loop
found_newline:
    mov r14, r10
    dec r14
    mov r9, r11
print_rev_loop:
    cmp r14, r9
    jl print_rev_done
    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    mov rdx, 1
    syscall
    dec r14
    jmp print_rev_loop
print_rev_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r10
    mov r11, r10
    jmp scan_loop
end_of_input:
    cmp r11, r13
    jge exit_start
    mov r14, r13
    dec r14
    mov r9, r11
print_final_rev_loop:
    cmp r14, r9
    jl print_final_rev_done
    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    mov rdx, 1
    syscall
    dec r14
    jmp print_final_rev_loop
print_final_rev_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
exit_start:
    mov rax, 60
    xor rdi, rdi
    syscall
