

section .bss
stdin_buf: .space 65536
tape: .space 30000
match_array: .space 262144
bracket_stack: .space 4000
total_read: .space 8
program_len: .space 8
input_start: .space 8
input_len: .space 8

section .text
.global _start

_start:
   lea r14, [rip + stdin_buf]
   mov r15, r14
   mov rdi, 0
   mov rdx, 65536
   mov rax, 0
read_loop:
   syscall
   test rax, rax
   jz read_done
   add r15, rax
   jmp read_loop
read_done:
   mov rdx, r15
   sub rdx, r14
   mov qword ptr [total_read], rdx
   mov r12, 0
   mov r9, qword ptr [total_read]
find_excl_loop:
   cmp r12, r9
   jge no_excl
   movzx eax, byte ptr [r14 + r12]
   cmp al, 0x21
   je found_excl
   inc r12
   jmp find_excl_loop
no_excl:
   mov r8, r9
   mov qword ptr [program_len], r8
   mov qword ptr [input_start], r8
   mov qword ptr [input_len], 0
   jmp setup_validation
found_excl:
   mov r8, r12
   mov qword ptr [program_len], r8
   inc r12
   mov qword ptr [input_start], r12
   mov r9, qword ptr [total_read]
   sub r9, r12
   mov qword ptr [input_len], r9
   jmp setup_validation
setup_validation:
   lea r10, [rip + match_array]
   lea r15, [rip + bracket_stack]
   add r15, 4000
   mov r11, r15
   mov r12, 0
   mov r9, qword ptr [program_len]
validate_loop:
   cmp r12, r9
   jge validate_done
   movzx eax, byte ptr [r14 + r12]
   cmp al, '['
   je validate_push
   cmp al, ']'
   je validate_pop
   inc r12
   jmp validate_loop
validate_push:
   sub r11, 4
   mov eax, r12
   mov dword ptr [r11], eax
   inc r12
   jmp validate_loop
validate_pop:
   cmp r11, r15
   je validate_unbalanced
   mov eax, dword ptr [r11]
   add r11, 4
   lea rdx, [r10 + r12*4]
   mov dword ptr [rdx], eax
   movzx rax, eax
   lea rdx, [r10 + rax*4]
   mov eax, r12
   mov dword ptr [rdx], eax
   inc r12
   jmp validate_loop
validate_done:
   cmp r11, r15
   jne validate_unbalanced
   lea rbx, [rip + tape]
   lea r15, [rip + match_array]
   xor r8, r8
   xor r12, r12
   mov r9, qword ptr [program_len]
   mov r10, qword ptr [input_len]
   mov r11, qword ptr [input_start]
   lea r13, [r14 + r11]
   jmp execution_loop
validate_unbalanced:
   mov rax, 60
   mov rdi, 1
   syscall
   jmp exit_fail
execution_loop:
   cmp r12, r9
   jge program_end
   movzx eax, byte ptr [r14 + r12]
   cmp al, '>'
   je exec_gt
   cmp al, '<'
   je exec_lt
   cmp al, '+'
   je exec_plus
   cmp al, '-'
   je exec_minus
   cmp al, '.'
   je exec_dot
   cmp al, ','
   je exec_comm
   cmp al, '['
   je exec_lbracket
   cmp al, ']'
   je exec_rbracket
   inc r12
   jmp execution_loop
exec_gt:
   inc r8
   jmp after_op
exec_lt:
   dec r8
   jmp after_op
exec_plus:
   movzx eax, byte ptr [rbx + r8]
   inc eax
   mov byte ptr [rbx + r8], al
   jmp after_op
exec_minus:
   movzx eax, byte ptr [rbx + r8]
   dec eax
   mov byte ptr [rbx + r8], al
   jmp after_op
exec_dot:
   lea rsi, [rbx + r8]
   mov rdi, 1
   mov rdx, 1
   mov rax, 1
   syscall
   jmp after_op
exec_comm:
   test r10, r10
   jz comm_exhausted
   movzx eax, byte ptr [r13]
   mov byte ptr [rbx + r8], al
   inc r13
   dec r10
   jmp after_op
comm_exhausted:
   mov byte ptr [rbx + r8], 0
   jmp after_op
exec_lbracket:
   movzx eax, byte ptr [rbx + r8]
   test al, al
   jz lbracket_zero
   jmp after_op
lbracket_zero:
   lea rdx, [r15 + r12*4]
   mov rax, dword ptr [rdx]
   add rax, 1
   mov r12, rax
   jmp execution_loop
exec_rbracket:
   movzx eax, byte ptr [rbx + r8]
   test al, al
   jz rbracket_zero
   lea rdx, [r15 + r12*4]
   mov rax, dword ptr [rdx]
   add rax, 1
   mov r12, rax
   jmp execution_loop
rbracket_zero:
   jmp after_op
after_op:
   inc r12
   jmp execution_loop
program_end:
   mov rax, 60
   xor rdi, rdi
   syscall
   jmp program_end
exit_fail:
   mov rax, 60
   mov rdi, 1
   syscall
   jmp exit_fail
