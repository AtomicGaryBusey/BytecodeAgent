

section .bss
buffer: .space 65536
tape: .space 30000
match: .space 262144
stack: .space 4000
total_read: .space 8
program_len: .space 8
input_start: .space 8
input_len: .space 8

section .text
global _start

_start:
   lea r14, [rip + buffer]
   mov qword ptr [total_read], 0
read_loop:
   mov r8, qword ptr [total_read]
   lea rsi, [r14 + r8]
   mov rdi, 0
   mov rdx, 65536
   mov rax, 0
   syscall
   test rax, rax
   jz done_read
   add qword ptr [total_read], rax
   jmp read_loop
done_read:
   mov r9, qword ptr [total_read]
   xor r13, r13
find_excl:
   cmp r13, r9
   jge no_excl
   movzx eax, byte ptr [r14 + r13]
   cmp al, 0x21
   je found_excl
   inc r13
   jmp find_excl
no_excl:
   mov qword ptr [program_len], r9
   mov qword ptr [input_start], r9
   mov qword ptr [input_len], 0
   jmp setup_validation
found_excl:
   mov qword ptr [input_start], r13 + 1
   mov qword ptr [program_len], r13
   mov qword ptr [input_len], r9 - r13 - 1
   jmp setup_validation
setup_validation:
   lea r10, [rip + match]
   lea r11, [rip + stack]
   mov r8, qword ptr [program_len]
   xor r12, r12
   xor r13, r13
validate_loop:
   cmp r13, r8
   jge validate_done
   movzx eax, byte ptr [r14 + r13]
   cmp al, '['
   je push_bracket
   cmp al, ']'
   je pop_bracket
   inc r13
   jmp validate_loop
push_bracket:
   mov dword ptr [r11 + r12*4], r13
   inc r12
   inc r13
   jmp validate_loop
pop_bracket:
   test r12, r12
   jz unbalanced
   dec r12
   mov eax, dword ptr [r11 + r12*4]
   lea rdx, [r10 + rax*4]
   lea r9, [r13 + 1]
   mov dword ptr [rdx], r9
   lea rdx, [r10 + r13*4]
   lea r9, [rax + 1]
   mov dword ptr [rdx], r9
   inc r13
   jmp validate_loop
validate_done:
   test r12, r12
   jne unbalanced
   mov r9, qword ptr [program_len]
   xor r12, r12
   xor r8, r8
   xor r13, r13
   lea rbx, [rip + tape]
   mov r11, qword ptr [input_start]
   lea r15, [r14 + r11]
   jmp execution_loop
unbalanced:
   mov rax, 60
   mov rdi, 1
   syscall
   jmp fail
fail:
   jmp fail
execution_loop:
   cmp r12, r9
   jge program_end
   movzx eax, byte ptr [r14 + r12]
   cmp al, '['
   je lbracket
   cmp al, ']'
   je rbracket
   cmp al, '>'
   je gt
   cmp al, '<'
   je lt
   cmp al, '+'
   je inc_cmd
   cmp al, '-'
   je dec_cmd
   cmp al, '.'
   je out_cmd
   cmp al, ','
   je inn_cmd
   inc r12
   jmp execution_loop
gt:
   inc r8
   jmp after_op
lt:
   dec r8
   jmp after_op
inc_cmd:
   movzx eax, byte ptr [rbx + r8]
   inc eax
   mov byte ptr [rbx + r8], al
   jmp after_op
dec_cmd:
   movzx eax, byte ptr [rbx + r8]
   dec eax
   jz dec_wrap
   mov byte ptr [rbx + r8], al
   jmp after_op
dec_wrap:
   mov byte ptr [rbx + r8], 0xff
   jmp after_op
out_cmd:
   mov rdi, 1
   lea rsi, [rbx + r8]
   mov rdx, 1
   mov rax, 1
   syscall
   jmp after_op
inn_cmd:
   mov r11, qword ptr [input_len]
   cmp r13, r11
   jge input_exhausted
   movzx eax, byte ptr [r15 + r13]
   mov byte ptr [rbx + r8], al
   inc r13
   jmp after_op
input_exhausted:
   mov byte ptr [rbx + r8], 0
   jmp after_op
after_op:
   inc r12
   jmp execution_loop
lbracket:
   movzx eax, byte ptr [rbx + r8]
   test al, al
   jz lbracket_zero
   jmp after_op
lbracket_zero:
   lea rax, [r10 + r12*4]
   mov rax, [rax]
   mov r12, rax
   jmp execution_loop
rbracket:
   movzx eax, byte ptr [rbx + r8]
   test al, al
   jz rbracket_zero
   lea rax, [r10 + r12*4]
   mov rax, [rax]
   mov r12, rax
   jmp execution_loop
rbracket_zero:
   jmp after_op
program_end:
   mov rax, 60
   xor rdi, rdi
   syscall
