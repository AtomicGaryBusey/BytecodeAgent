

.global _start

.section .rodata
empty_msg:
    .ascii "00000000\n"

.section .bss
inbuf:
    .space 65536
total:
    .quad 0

.section .text
_start:
    mov rdi, 0
    mov rsi, inbuf
    mov rdx, 65536
    syscall
    test rax, rax
    jz print_empty
    mov rax, 60
    mov rdi, 0
    syscall
    jmp done
print_empty:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + empty_msg]
    mov rdx, 9
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
done:
    jmp done
