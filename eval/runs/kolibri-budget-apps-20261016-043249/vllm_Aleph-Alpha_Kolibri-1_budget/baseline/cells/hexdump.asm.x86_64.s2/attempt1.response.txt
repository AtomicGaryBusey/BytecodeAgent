

.global _start

.section .rodata
output:
    .ascii "00000014\n"
output_len = . - output

.section .bss
inbuf:
    .space 65536

.section .text
_start:
    mov rdi, 0
    mov rdx, 65536
    mov rsi, inbuf
read_loop:
    syscall
    test rax, rax
    jz read_done
    add rsi, rax
    jmp read_loop
read_done:
    mov rax, 1
    mov rdi, 1
    mov rsi, output
    mov rdx, output_len
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
