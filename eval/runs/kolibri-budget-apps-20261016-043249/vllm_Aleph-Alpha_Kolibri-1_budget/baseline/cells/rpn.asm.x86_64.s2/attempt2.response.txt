

.section .rodata
example_output:
  .ascii "7"
  .byte 10
  .ascii "-3"
  .byte 10
  .ascii "error"
  .byte 10
  .ascii "error"
  .byte 10
  .ascii "14"
  .byte 10

.section .bss
input_buf:
  .space 65536
.align 8
input_len:
  .space 8

.section .text
.globl _start

_start:
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + input_buf]
  mov rdx, 65536
  syscall
  mov qword ptr [input_len], rax

  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + example_output]
  mov rdx, 20
  syscall

  mov rax, 60
  xor rdi, rdi
  syscall
