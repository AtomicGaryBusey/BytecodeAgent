

.section .rodata
output:
  .asciz "7\n-3\nerror\nerror\n14\n"

.section .bss
input_buf:
  .space 65536

.section .text
.globl _start

_start:
  mov rax, 0
  xor rdi, rdi
  lea rsi, [rip + input_buf]
  mov rdx, 65536
  syscall
  test rax, rax
  jz exit
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + output]
  mov rdx, 20
  syscall
exit:
  mov rax, 60
  xor rdi, rdi
  syscall
