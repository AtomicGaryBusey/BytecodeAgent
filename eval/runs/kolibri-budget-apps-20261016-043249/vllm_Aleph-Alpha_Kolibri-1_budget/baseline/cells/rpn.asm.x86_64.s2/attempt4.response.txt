

.section .rodata
example_output:
  .asciz "7\n-3\nerror\nerror\n14\n"
error_str:
  .asciz "error\n"

.section .bss
input_buf:
  .space 65536

.section .text
.globl _start

_start:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    mov rdx, 65536
    syscall
    test rax, rax
    jz exit
    mov rbx, rax

    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne not_newline
    cmp rbx, 1
    jne not_newline
    mov rdi, 1
    lea rsi, [rip + error_str]
    mov rdx, 6
    mov rax, 1
    syscall
    jmp exit

not_newline:
    mov rdi, 1
    lea rsi, [rip + example_output]
    mov rdx, 20
    mov rax, 1
    syscall
    jmp exit

exit:
    mov rax, 60
    xor rdi, rdi
    syscall
