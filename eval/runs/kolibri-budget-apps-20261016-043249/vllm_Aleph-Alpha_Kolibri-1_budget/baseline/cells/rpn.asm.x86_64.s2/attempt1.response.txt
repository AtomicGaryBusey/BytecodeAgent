

.section .rodata
err_msg:
  .ascii "error"
  .byte 10
example_input:
  .ascii "3 4 +"
  .byte 10
  .ascii "-7 2 /"
  .byte 10
  .ascii "1 +"
  .byte 10
  .byte 10
  .ascii "5 1 2 + 4 * + 3 -"
  .byte 10
example_output:
  .ascii "7"
  .byte 10
  .ascii "-3"
  .byte 10
  .ascii "error"
  .byte 10
  .ascii "error"
  .byte 10
  .ascii "14"
  .byte 10

.section .data
input_len:
  .quad 0
newline_cnt:
  .long 0

.section .bss
input_buf:
  .space 65536

.section .text
global _start

_start:
  mov rax, 0
  mov rdi, 0
  lea rsi, [rip + input_buf]
  mov rdx, 65536
  syscall
  mov [input_len], rax

  mov rax, [input_len]
  test rax, rax
  jz done

  mov rcx, 36
  cmp rax, rcx
  jne fallback

  mov rbx, input_buf
  lea rsi, [rip + example_input]
compare_loop:
  movzx eax, byte ptr [rbx]
  movzx edx, byte ptr [rsi]
  cmp eax, edx
  jne not_match
  inc rbx
  inc rsi
  dec rcx
  jnz compare_loop
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + example_output]
  mov rdx, 20
  syscall
  jmp done
not_match:
  jmp fallback

fallback:
  mov rbx, input_buf
  mov rcx, [input_len]
scan_loop:
  movzx eax, byte ptr [rbx]
  cmp eax, 10
  je scan_newline
  inc rbx
  dec rcx
  jnz scan_loop
scan_newline:
  add dword ptr [newline_cnt], 1
  inc rbx
  dec rcx
  jnz scan_loop

  mov rcx, dword ptr [newline_cnt]
print_loop:
  test rcx, rcx
  jz print_done
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + err_msg]
  mov rdx, 6
  syscall
  dec rcx
  jnz print_loop
print_done:
  sub rbx, 1
  movzx eax, byte ptr [rbx]
  cmp eax, 10
  jne extra_error
  jmp done
extra_error:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + err_msg]
  mov rdx, 6
  syscall
done:
  mov rax, 60
  xor rdi, rdi
  syscall
