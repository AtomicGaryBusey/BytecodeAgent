

.text
global _start

fmt2:
   test rdi, rdi
   jz fmt2_zero
   lea rbx, [rip + digit_buf]
   xor r10, r10
digit_loop:
   test rdi, rdi
   jz digit_loop_end
   mov rax, rdi
   mov rcx, 10
   idiv rcx
   add dl, 48
   mov [rbx], dl
   inc rbx
   inc r10
   mov rdi, rax
   jmp digit_loop
digit_loop_end:
   dec rbx
output_loop:
   mov al, byte ptr [rbx]
   mov [rsi], al
   inc rsi
   dec rbx
   dec r10
   test r10, r10
   jz output_loop_end
   jmp output_loop
output_loop_end:
   ret
fmt2_zero:
   mov al, 48
   mov [rsi], al
   inc rsi
   ret

_start:
   xor r8, r8
read_loop:
   lea rdi, [rip + input_buffer]
   add rdi, r8
   mov rsi, 65536
   mov rdx, 0
   syscall
   test rax, rax
   jz read_done
   add r8, rax
   jmp read_loop
read_done:
   mov [total_bytes], r8

scan_start:
   lea rdi, [rip + input_buffer]
   mov rdx, [total_bytes]
   xor rcx, rcx
   xor r8, r8
   xor r9, r9
scan_loop:
   test rdx, rdx
   jz scan_done
   movzx eax, byte ptr [rdi]
   cmp eax, 10
   je scan_newline
   cmp eax, 32
   je scan_ws
   cmp eax, 9
   je scan_ws
   cmp eax, 11
   je scan_ws
   cmp eax, 12
   je scan_ws
   cmp eax, 13
   je scan_ws
   test r9, r9
   jz scan_word_start
   jmp scan_next
scan_word_start:
   inc r8
   mov r9, 1
   jmp scan_next
scan_newline:
   inc rcx
   mov r9, 0
   jmp scan_next
scan_ws:
   mov r9, 0
   jmp scan_next
scan_next:
   inc rdi
   dec rdx
   jnz scan_loop
scan_done:
   mov [lines], rcx
   mov [words], r8

lea rsi, [rip + outbuf]
mov rdi, [lines]
call fmt2
mov al, 32
mov [rsi], al
inc rsi
mov rdi, [words]
call fmt2
mov al, 32
mov [rsi], al
inc rsi
mov rdi, [total_bytes]
call fmt2
mov al, 10
mov [rsi], al
inc rsi

lea rax, [rip + outbuf]
sub rsi, rax
mov rdx, rsi
mov rsi, rax
mov rdi, 1
mov rax, 1
syscall

mov rax, 60
xor rdi, rdi
syscall

.section .bss
input_buffer: .space 65536
total_bytes: .quad 0
lines: .quad 0
words: .quad 0
outbuf: .space 32
digit_buf: .space 6
