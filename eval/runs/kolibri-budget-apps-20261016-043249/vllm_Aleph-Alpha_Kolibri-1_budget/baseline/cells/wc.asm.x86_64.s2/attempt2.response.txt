

.section .rodata
space_char: .byte 32
newline_char: .byte 10
char_zero: .byte 48
.section .bss
read_buf: .space 65536
digit_buf: .space 6
.section .text
global _start

_start:
   xor r10, r10
   xor r12, r12
   xor r13, r13
   xor r14, r14
read_loop:
   xor rdi, rdi
   lea rsi, [rip + read_buf]
   mov rdx, 65536
   syscall
   test rax, rax
   jz read_done
   mov r8, rax
   lea r9, [rip + read_buf]
process_chunk:
   test r8, r8
   jz process_next
   movzx eax, byte ptr [r9]
   cmp eax, 10
   je newline
   cmp eax, 32
   je whitespace
   cmp eax, 9
   je whitespace
   cmp eax, 11
   je whitespace
   cmp eax, 12
   je whitespace
   cmp eax, 13
   je whitespace
   test r14, r14
   jz word_start
   jmp next_byte
word_start:
   inc r13
   mov r14, 1
   jmp next_byte
newline:
   inc r12
   mov r14, 0
   jmp next_byte
whitespace:
   mov r14, 0
   jmp next_byte
next_byte:
   inc r10
   inc r9
   dec r8
   jnz process_chunk
process_next:
   jmp read_loop
read_done:
   mov rdi, r12
   call print_num
   lea rsi, [rip + space_char]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   mov rdi, r13
   call print_num
   lea rsi, [rip + space_char]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   mov rdi, r10
   call print_num
   lea rsi, [rip + newline_char]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   mov rax, 60
   xor rdi, rdi
   syscall

print_num:
   test rdi, rdi
   jz print_zero
   lea rsi, [rip + digit_buf + 5]
   xor r8, r8
digit_loop:
   test rdi, rdi
   jz digit_done
   mov rax, rdi
   mov rcx, 10
   idiv rcx
   add dl, 48
   mov [rsi], dl
   dec rsi
   inc r8
   mov rdi, rax
   jmp digit_loop
digit_done:
   inc rsi
   mov rdx, r8
   mov rdi, 1
   mov rax, 1
   syscall
   ret
print_zero:
   lea rsi, [rip + char_zero]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   ret
