

.section .rodata
space_char: .byte 32
newline_char: .byte 10
zero_char: .byte 48

.section .bss
read_buf: .space 65536
digit_buf: .space 6

.section .text
.globl _start

_start:
   xor r12, r12
   xor r13, r13
   xor r14, r14
   xor r15, r15
   xor rdi, rdi
   lea rsi, [rip + read_buf]
   mov rdx, 65536
read_loop:
   mov rax, 0
   syscall
   test rax, rax
   jz read_done
   mov r8, rax
   lea r9, [rip + read_buf]
process_chunk:
   test r8, r8
   jz process_next
   movzx eax, byte ptr [r9]
   cmp eax, 10
   je newline
   cmp eax, 32
   je whitespace
   cmp eax, 9
   je whitespace
   cmp eax, 11
   je whitespace
   cmp eax, 12
   je whitespace
   cmp eax, 13
   je whitespace
   test r15, r15
   jz word_start
   jmp next_byte
word_start:
   inc r13
   mov r15, 1
   jmp next_byte
newline:
   inc r12
   mov r15, 0
   jmp next_byte
whitespace:
   mov r15, 0
   jmp next_byte
next_byte:
   inc r14
   inc r9
   dec r8
   jnz process_chunk
process_next:
   jmp read_loop
read_done:
   push r13
   push r14
   mov rdi, r12
   call print_num
   pop r13
   lea rsi, [rip + space_char]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   mov rdi, r13
   call print_num
   pop r14
   lea rsi, [rip + space_char]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   mov rdi, r14
   call print_num
   lea rsi, [rip + newline_char]
   mov rdx, 1
   mov rdi, 1
   mov rax, 1
   syscall
   mov rax, 60
   xor rdi, rdi
   syscall

print_num:
   test rdi, rdi
   jz print_zero
   mov rcx, 10
   lea rsi, [rip + digit_buf + 5]
   xor r8, r8
division_loop:
   mov rax, rdi
   xor rdx, rdx
   idiv rcx
   add dl, 48
   mov [rsi], dl
   dec rsi
   inc r8
   mov rdi, rax
   test rax, rax
   jnz division_loop
   inc rsi
print_digits:
   mov rax, 1
   mov rdi, 1
   mov rdx, 1
   syscall
   inc rsi
   dec r8
   test r8, r8
   jnz print_digits
   ret
print_zero:
   mov rax, 1
   mov rdi, 1
   lea rsi, [rip + zero_char]
   mov rdx, 1
   syscall
   ret
