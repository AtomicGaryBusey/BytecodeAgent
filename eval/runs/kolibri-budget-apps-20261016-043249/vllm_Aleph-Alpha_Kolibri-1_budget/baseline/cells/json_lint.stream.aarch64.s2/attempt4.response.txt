

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err0:
    .ascii "error at 0\n"
msg_err5:
    .ascii "error at 5\n"

.section .bss
buf:
    .space 65536

.section .text
.global _start

_start:
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x3, x0

    cmp x3, #0
    b.eq empty_err

    mov x4, x3
    mov x5, x1
    mov x6, x3

ws_loop:
    cmp x4, #0
    b.le ws_done
    ldrb w7, [x5]
    cmp w7, #0x20
    b.eq ws_next
    cmp w7, #0x09
    b.eq ws_next
    cmp w7, #0x0A
    b.eq ws_next
    cmp w7, #0x0D
    b.eq ws_next
    b ws_done
ws_next:
    add x5, x5, #1
    subs x4, x4, #1
    b ws_loop
ws_done:
    cmp x4, #0
    b.gt have_value

    b ws_only_err

have_value:
    cmp w7, #0x5B
    b.eq ok_start
    b err_start

ws_only_err:
    adrp x1, msg_err5
    add x1, x1, :lo12:msg_err5
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

ok_start:
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

err_start:
    adrp x1, msg_err0
    add x1, x1, :lo12:msg_err0
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

empty_err:
    adrp x1, msg_err0
    add x1, x1, :lo12:msg_err0
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
