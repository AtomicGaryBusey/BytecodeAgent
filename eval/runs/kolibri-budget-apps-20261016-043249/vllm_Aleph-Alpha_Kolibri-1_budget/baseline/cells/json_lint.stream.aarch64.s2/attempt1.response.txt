

.section .rodata
msg_ok:
    .asciz "ok\n"
msg_err0:
    .asciz "error at 0\n"

.section .bss
buf:
    .space 65536

.section .text
.global _start

_start:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x3, x0

    cmp x3, #0
    b.ne non_zero

empty:
    adrp x1, msg_err0
    add x1, x1, :lo12:msg_err0
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

non_zero:
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
