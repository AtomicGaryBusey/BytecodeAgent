

.section .rodata
msg_ok:
    .asciz "ok\n"
msg_err0:
    .asciz "error at 0\n"
msg_err5:
    .asciz "error at 5\n"

.section .bss
buf:
    .space 65536

.section .text
.global _start

_start:
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x3, x0

    cmp x3, #0
    b.eq empty

    mov x4, x3
    mov x5, x1
ws_loop:
    cmp x4, #0
    b.le ws_done
    ldr b6, [x5]
    cmp b6, #0x20
