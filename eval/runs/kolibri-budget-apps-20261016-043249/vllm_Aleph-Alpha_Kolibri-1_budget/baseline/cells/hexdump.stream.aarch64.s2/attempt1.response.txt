

.section .bss
.global buf
buf: .space 65536

.section .rodata
.global zero_line
zero_line: .byte 48,48,48,48,48,48,48,48,10

.section .text
.global _start

_start:
    adrp x9, buf
    add x9, x9, :lo12:buf
    mov x10, #0
read_loop:
    add x1, x9, x10
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    cbz x0, done_reading
    add x10, x10, x0
    b read_loop
done_reading:
    cmp x10, #0
    b.ne non_zero
    adrp x1, zero_line
    add x1, x1, :lo12:zero_line
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0
    b exit
non_zero:
    b exit
exit:
    mov x0, #93
    mov x1, #0
    svc #0
