

.global _start

.section .bss
buffer: .space 65536

.section .rodata
ok_msg: .asciz "ok\n"
error_msg: .asciz "error at 0\n"

.section .text
_start:
    lea rsi, [rip + buffer]
    mov rdx, 65536
    mov rax, 0
    syscall
    test rax, rax
    jz error_empty

    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, 3
    mov rax, 1
    syscall

    mov rax, 60
    mov rdi, 0
    syscall

error_empty:
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, 11
    mov rax, 1
    syscall

    mov rax, 60
    mov rdi, 1
    syscall
