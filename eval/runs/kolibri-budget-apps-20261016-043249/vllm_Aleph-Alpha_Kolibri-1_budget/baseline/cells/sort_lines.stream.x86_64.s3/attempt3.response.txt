

.global _start

.equ BUF_SIZE, 32768

.text
_start:
    lea rsi, [rip + buffer]
    mov rdx, BUF_SIZE
    xor rax, rax
    syscall
    mov [rip + total_bytes], rax
    mov rax, [rip + total_bytes]
    mov rdi, 1
    mov rdx, rax
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
buffer:
    .space BUF_SIZE
total_bytes:
    .space 8
