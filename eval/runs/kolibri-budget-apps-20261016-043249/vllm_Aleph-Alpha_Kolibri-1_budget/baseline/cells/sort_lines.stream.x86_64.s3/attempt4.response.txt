

.global _start

.section .bss
buffer:
    .space 32768
total:
    .space 8

.section .rodata
newline:
    .byte 10

.text
_start:
    xor rax, rax
    lea rdi, [buffer]
    mov rdx, 32768
    syscall
    mov [total], rax
    test rax, rax
    jz .exit
    mov rax, 1
    mov rdi, 1
    lea rsi, [buffer]
    mov rdx, [total]
    syscall
    mov r8, [total]
    lea rdi, [buffer]
    add rdi, r8
    sub rdi, 1
    mov al, [rdi]
    cmp al, 10
    je .exit
    mov rax, 1
    mov rdi, 1
    lea rsi, [newline]
    mov rdx, 1
    syscall
.exit:
    mov rax, 60
    xor rdi, rdi
    syscall
