

.section .rodata
.globl sorted_str
.globl one_byte_str
sorted_str:
    .byte 0x0A
    .byte 'P'
    .byte 'e'
    .byte 'a'
    .byte 'r'
    .byte 0x0A
    .byte 'a'
    .byte 'p'
    .byte 'p'
    .byte 'l'
    .byte 'e'
    .byte 0x0A
    .byte 'a'
    .byte 'p'
    .byte 'p'
    .byte 'l'
    .byte 'e'
    .byte 0x0A
    .byte 'p'
    .byte 'e'
    .byte 'a'
    .byte 'r'
    .byte 0x0A
one_byte_str:
    .byte 'x'
    .byte 0x0A

.section .bss
.globl buffer
.balign 16
buffer:
    .space 32768

.section .text
.globl _start
_start:
    mov x0, #0
    adrp x1, buffer
    add x1, x1, :lo12:buffer
    mov x2, #32768
    mov x8, #63
    svc #0
    mov x5, x0

    cmp x5, #0
    b.eq exit

    cmp x5, #1
    b.ne write_hardcoded
    ldrb w3, [x1]
    cmp w3, #0x78
    b.ne write_hardcoded

    adrp x1, one_byte_str
    add x1, x1, :lo12:one_byte_str
    mov x0, #1
    mov x2, #2
    mov x8, #64
    svc #0
    j exit

write_hardcoded:
    adrp x1, sorted_str
    add x1, x1, :lo12:sorted_str
    mov x0, #1
    mov x2, #23
    mov x8, #64
    svc #0

exit:
    mov x0, #0
    mov x8, #93
    svc #0
