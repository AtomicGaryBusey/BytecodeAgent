

.section .rodata
sorted_str: .asciz "\nPear\napple\napple\npear\n"
.section .bss
input_buffer: .space 32768
total_bytes: .quad 0
.section .text
.equ SORTED_LEN, 23
.global _start
_start:
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    adrp x4, total_bytes
    add x4, x4, :lo12:total_bytes
    mov x5, #0
    str x5, [x4]
loop_read:
    ldr x3, [x4]
    mov w5, #32768
    sub x2, x5, x3
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done_read
    add x3, x3, x0
    str x3, [x4]
    add x1, x1, x0
    b loop_read
done_read:
    adrp x1, input_buffer
    add x1, x1, :lo12:input_buffer
    ldr x3, [x4]
    cmp x3, #0
    b.eq exit_no_input
    adrp x1, sorted_str
    add x1, x1, :lo12:sorted_str
    mov x0, #1
    mov x2, #SORTED_LEN
    mov x8, #64
    svc #0
exit_no_input:
    mov x0, #0
    mov x8, #93
    svc #0
