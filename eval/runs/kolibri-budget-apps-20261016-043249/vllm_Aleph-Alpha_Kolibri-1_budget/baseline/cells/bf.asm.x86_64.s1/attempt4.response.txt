

.section .bss
align 8
stdin_buf:
    .space 65536
tape:
    .space 30000
match_arr:
    .space 65536*4
bracket_stack:
    .space 4000*4

.section .data
total_len:
    .quad 0
prog_len:
    .quad 0
input_start:
    .quad 0
input_len:
    .quad 0
status:
    .long 0

.section .text
.globl _start
_start:
    lea r12, [rip + stdin_buf]
    lea r13, [rip + tape]
    lea r14, [rip + match_arr]
    lea r15, [rip + bracket_stack]
    xor r10, r10
read_loop:
    mov eax, 0
    mov rdi, 0
    lea rsi, [r12 + r10]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_reading
    add r10, rax
    jmp read_loop
done_reading:
    mov qword ptr [rip + total_len], r10
    xor r11, r11
find_excl:
    mov rax, qword ptr [rip + total_len]
    mov r11, rax
    xor r8, r8
scan:
    cmp r8, r11
    jge no_excl
    movzx al, byte ptr [r12 + r8]
    cmp al, 0x21
    je found_excl
    inc r8
    jmp scan
found_excl:
    mov qword ptr [rip + prog_len], r8
    lea rax, [r8 + 1]
    mov qword ptr [rip + input_start], rax
    sub r11, r8
    sub r11, 1
    mov qword ptr [rip + input_len], r11
    jmp bracket_scanning
no_excl:
    mov qword ptr [rip + prog_len], r11
    mov qword ptr [rip + input_start], r11
    xor r10, r10
    mov qword ptr [rip + input_len], r10
    jmp bracket_scanning
bracket_scanning:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    mov rax, qword ptr [rip + prog_len]
    mov r11, rax
scan_loop:
    cmp r10, r11
    jge scan_done
    movzx al, byte ptr [r12 + r10]
    cmp al, '['
    je scan_lbracket
    cmp al, ']'
    je scan_rbracket
    inc r10
    jmp scan_loop
scan_lbracket:
    lea rax, [r15 + r8*4]
    mov dword ptr [rax], r10
    inc r8
    inc r10
    jmp scan_loop
scan_rbracket:
    test r8, r8
    jz unbalanced
    lea rax, [r15 + r8*4 - 4]
    mov eax, dword ptr [rax]
    lea rax, [r14 + r10*4]
    mov dword ptr [rax], eax
    lea rax, [r14 + eax*4]
    mov dword ptr [rax], r10
    dec r8
    inc r10
    jmp scan_loop
scan_done:
    test r8, r8
    jz balanced
unbalanced:
    mov dword ptr [rip + status], 1
    jmp exit_start
balanced:
    mov dword ptr [rip + status], 0
    mov rax, qword ptr [rip + prog_len]
    mov r15, rax
    mov rax, qword ptr [rip + input_start]
    lea r8, [r12 + rax]
    xor r9, r9
    xor r10, r10
    jmp execution_loop
execution_loop:
    cmp r10, r15
    jge program_end
    movzx al, byte ptr [r12 + r10]
    cmp al, '>'
    je do_gt
    cmp al, '<'
    je do_lt
    cmp al, '+'
    je do_plus
    cmp al, '-'
    je do_minus
    cmp al, '.'
    je do_dot
    cmp al, ','
    je do_comm
    cmp al, '['
    je do_lbracket
    cmp al, ']'
    je do_rbracket
    inc r10
    jmp execution_loop
do_gt:
    inc r9
    inc r10
    jmp execution_loop
do_lt:
    dec r9
    inc r10
    jmp execution_loop
do_plus:
    movzx al, byte ptr [r13 + r9]
    inc al
    mov byte ptr [r13 + r9], al
    inc r10
    jmp execution_loop
do_minus:
    movzx al, byte ptr [r13 + r9]
    dec al
    mov byte ptr [r13 + r9], al
    inc r10
    jmp execution_loop
do_dot:
    mov eax, 1
    mov rdi, 1
    lea rsi, [r13 + r9]
    mov rdx, 1
    syscall
    inc r10
    jmp execution_loop
do_comm:
    mov rax, qword ptr [rip + input_len]
    lea rdx, [r8 + rax]
    cmp r8, rdx
    jge input_exhausted
    movzx al, byte ptr [r8]
    inc r8
    mov byte ptr [r13 + r9], al
    inc r10
    jmp execution_loop
input_exhausted:
    xor al, al
    mov byte ptr [r13 + r9], al
    inc r10
    jmp execution_loop
do_lbracket:
    movzx al, byte ptr [r13 + r9]
    test al, al
    jnz continue_lbracket
    mov eax, dword ptr [r14 + r10*4]
    inc eax
    mov r10, rax
    jmp execution_loop
continue_lbracket:
    inc r10
    jmp execution_loop
do_rbracket:
    movzx al, byte ptr [r13 + r9]
    test al, al
    jz continue_rbracket
    mov eax, dword ptr [r14 + r10*4]
    inc eax
    mov r10, rax
    jmp execution_loop
continue_rbracket:
    inc r10
    jmp execution_loop
program_end:
    mov dword ptr [rip + status], 0
    jmp exit_start
exit_start:
    mov eax, dword ptr [rip + status]
    mov rdi, rax
    mov eax, 60
    syscall
