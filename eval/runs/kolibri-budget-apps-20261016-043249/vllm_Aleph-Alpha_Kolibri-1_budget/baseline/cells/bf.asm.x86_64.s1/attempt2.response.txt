

.section .bss
align 8
stdin_buf:
    .space 65536
align 8
tape:
    .space 30000
align 8
match:
    .space 262144
align 8
bracket_stack:
    .space 4096

.section .data
total_len:
    .quad 0
prog_len:
    .quad 0
input_start:
    .quad 0
input_len:
    .quad 0

.section .text
.globl _start
_start:
    lea r12, [rip + stdin_buf]
    lea r13, [rip + tape]
    lea r14, [rip + match]
    lea r15, [rip + bracket_stack]

    xor r10, r10
read_loop:
    xor eax, eax
    mov rdi, 0
    lea rsi, [r12 + r10]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_reading
    add r10, rax
    jmp read_loop
done_reading:
    mov [rip + total_len], r10

    xor r11, r11
find_excl:
    cmp r11, r10
    jge no_excl
    movzx al, byte ptr [r12 + r11]
    cmp al, 0x21
    je found_excl
    inc r11
    jmp find_excl
found_excl:
    mov [rip + prog_len], r11
    lea rax, [r11 + 1]
    mov [rip + input_start], rax
    sub r10, r11
    sub r10, 1
    mov [rip + input_len], r10
    jmp do_bracket_scanning
no_excl:
    mov r11, r10
    mov [rip + prog_len], r11
    mov [rip + input_start], r11
    xor r10, r10
    mov [rip + input_len], r10
    jmp do_bracket_scanning

do_bracket_scanning:
    mov r10, [rip + prog_len]
    xor r8, r8
    xor r11, r11
scan_loop:
    cmp r11, r10
    jge scan_done
    movzx al, byte ptr [r12 + r11]
    cmp al, '['
    je scan_lbracket
    cmp al, ']'
    je scan_rbracket
    inc r11
    jmp scan_loop
scan_lbracket:
    lea rax, [r15 + r8*4]
    mov [rax], r11
    inc r8
    inc r11
    jmp scan_loop
scan_rbracket:
    test r8, r8
    jz unbalanced
    dec r8
    lea rax, [r15 + r8*4]
    mov rax, [rax]
    lea rdx, [r14 + r11*4]
    mov dword ptr [rdx], eax
    lea rdx, [r14 + rax*4]
    mov dword ptr [rdx], r11d
    inc r11
    jmp scan_loop
scan_done:
    test r8, r8
    jz balanced
unbalanced:
    mov eax, 60
    mov rdi, 1
    syscall
    jmp done
balanced:
    mov r8, 0
    xor r9, r9
    xor r10, r10
    jmp execution_loop

done:
    jmp done

execution_loop:
    mov rax, [rip + prog_len]
    cmp r8, rax
    jge program_end

    movzx al, byte ptr [r12 + r8]

    cmp al, '>'
    je do_gt
    cmp al, '<'
    je do_lt
    cmp al, '+'
    je do_plus
    cmp al, '-'
    je do_minus
    cmp al, '.'
    je do_dot
    cmp al, ','
    je do_comm
    cmp al, '['
    je do_lbracket
    cmp al, ']'
    je do_rbracket

    inc r8
    jmp execution_loop

do_gt:
    inc r9
    jmp execution_loop

do_lt:
    dec r9
    jmp execution_loop

do_plus:
    movzx al, byte ptr [r13 + r9]
    inc al
    mov byte ptr [r13 + r9], al
    jmp execution_loop

do_minus:
    movzx al, byte ptr [r13 + r9]
    dec al
    mov byte ptr [r13 + r9], al
    jmp execution_loop

do_dot:
    lea rsi, [r13 + r9]
    mov eax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    jmp execution_loop

do_comm:
    mov rax, [rip + input_len]
    cmp r10, rax
    jge input_exhausted
    mov rax, [rip + input_start]
    add rax, r10
    lea rsi, [r12 + rax]
    movzx al, byte ptr [rsi]
    mov byte ptr [r13 + r9], al
    inc r10
    jmp execution_loop
input_exhausted:
    xor al, al
    mov byte ptr [r13 + r9], al
    jmp execution_loop

do_lbracket:
    movzx al, byte ptr [r13 + r9]
    test al, al
    jz jump_to_matching
    inc r8
    jmp execution_loop
jump_to_matching:
    lea rdx, [r14 + r8*4]
    mov eax, dword ptr [rdx]
    inc eax
    mov r8, eax
    jmp execution_loop

do_rbracket:
    movzx al, byte ptr [r13 + r9]
    test al, al
    jz continue_after
    lea rdx, [r14 + r8*4]
    mov eax, dword ptr [rdx]
    inc eax
    mov r8, eax
    jmp execution_loop
continue_after:
    inc r8
    jmp execution_loop

program_end:
    mov eax, 60
    xor rdi, rdi
    syscall
