

.section .bss
stdin_buf: .space 65536
tape: .space 30000
match: .space 262144
bracket_stack: .space 4096
total_len: .quad 0
stack_top: .quad 0
prog_len: .quad 0
input_start: .quad 0
input_len: .quad 0
temp_byte: .byte 0
.section .text
.globl _start
_start:
    lea r13, [rip + stdin_buf]
    xor r10d, r10d
read_loop:
    xor eax, eax
    mov rdi, 0
    lea rsi, [rip + stdin_buf]
    add rsi, r10
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_reading
    add r10, rax
    jmp read_loop
done_reading:
    mov qword ptr [rip + total_len], r10
    xor r14d, r14d
find_excl:
    movzx al, byte ptr [r13 + r14]
    cmp al, 0x21
    je found_excl
    inc r14d
    cmp r14d, r10d
    jl find_excl
    jmp no_excl
found_excl:
    mov r15d, r14
    mov qword ptr [rip + prog_len], r15
    lea rax, [r15 + 1]
    mov qword ptr [rip + input_start], rax
    sub r10, r15
    sub r10, 1
    mov qword ptr [rip + input_len], r10
    jmp do_bracket_scanning
no_excl:
    mov r15d, r10
    mov qword ptr [rip + prog_len], r15
    mov qword ptr [rip + input_start], r15
    mov qword ptr [rip + input_len], 0
do_bracket_scanning:
    lea r8, [rip + match]
    lea r12, [rip + bracket_stack]
    lea rax, [rip + prog_len]
    mov r11, [rax]
    xor r15d, r15d
    xor r14d, r14d
scan_loop:
    cmp r14d, r11d
    jge scan_done
    movzx al, byte ptr [r13 + r14]
    cmp al, '['
    je scan_lbracket
    cmp al, ']'
    je scan_rbracket
    inc r14d
    jmp scan_loop
scan_lbracket:
    lea rax, [r12 + r15*4]
    mov [rax], r14
    inc r15d
    inc r14d
    jmp scan_loop
scan_rbracket:
    test r15, r15
    jz unbalanced
    dec r15d
    lea rax, [r12 + r15*4]
    mov r10, [rax]
    lea rax, [r8 + r10*4]
    mov [rax], r14
    lea rax, [r8 + r14*4]
    mov [rax], r10
    inc r14d
    jmp scan_loop
scan_done:
    test r15, r15
    jz balanced
unbalanced:
    mov eax, 60
    mov rdi, 1
    syscall
balanced:
    mov eax, 60
    mov rdi, 0
    syscall
