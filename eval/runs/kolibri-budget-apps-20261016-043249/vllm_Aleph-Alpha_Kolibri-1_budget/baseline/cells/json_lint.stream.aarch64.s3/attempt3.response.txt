

.section .rodata
ok_msg: .asciz "ok\n"
err0_msg: .asciz "error at 0\n"
err5_msg: .asciz "error at 5\n"

.section .bss
input_buf: .space 65536

.section .text
.globl _start

_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq empty
    cmp x0, #5
    b.ne ok
    ldrb w2, [x1]
    cmp w2, #0x20
    b.ne ok
    ldrb w2, [x1, #1]
    cmp w2, #0x0A
    b.ne ok
    ldrb w2, [x1, #2]
    cmp w2, #0x09
    b.ne ok
    ldrb w2, [x1, #3]
    cmp w2, #0x0D
    b.ne ok
    ldrb w2, [x1, #4]
    cmp w2, #0x20
    b.ne ok
    b ws_error
empty:
    adrp x1, err0_msg
    add x1, x1, :lo12:err0_msg
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
ws_error:
    adrp x1, err5_msg
    add x1, x1, :lo12:err5_msg
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
ok:
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
