

.section .rodata
ok_msg: .asciz "ok\n"
err_msg: .asciz "error at 0\n"

.section .bss
input_buf: .space 65536

.section .text
.globl _start

error_path:
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

ok_path:
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

_start:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq error_path
    b ok_path
