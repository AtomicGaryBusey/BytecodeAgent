

.global _start
.section .rodata
char_table: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
newline_byte: .byte 0x0A
eq_char: .byte 61
.section .bss
in_buf: .space 4
.section .text
_start:
    adrp x1, in_buf
    add x1, x1, :lo12:in_buf
    mov x2, #3
    mov x0, #0
    mov x8, #63
    svc #0
    cmp w0, #0
    b.eq exit_empty
    cmp w0, #3
    b.eq full_group
    cmp w0, #2
    b.eq two_bytes
    cmp w0, #1
    b.eq one_byte
    b exit

full_group:
    ldrb w0, [x1]
    add x1, x1, #1
    ldrb w1, [x1]
    add x1, x1, #1
    ldrb w2, [x1]
    adrp x7, char_table
    add x7, x7, :lo12:char_table
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    or w4, w4, w5
    and w5, w1, #15
    lsl w5, w5, #2
    lsr w6, w2, #6
    or w5, w5, w6
    and w6, w2, #63
    add x9, x7, x3
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x7, x4
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x7, x5
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x7, x6
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    b exit

two_bytes:
    ldrb w0, [x1]
    add x1, x1, #1
    ldrb w1, [x1]
    adrp x7, char_table
    add x7, x7, :lo12:char_table
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    or w4, w4, w5
    and w5, w1, #15
    lsl w5, w5, #2
    add x9, x7, x3
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x7, x4
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x7, x5
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x1, eq_char
    add x1, x1, :lo12:eq_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    b exit

one_byte:
    ldrb w0, [x1]
    adrp x7, char_table
    add x7, x7, :lo12:char_table
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    add x9, x7, x3
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x7, x4
    ldrb w0, [x9]
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x1, eq_char
    add x1, x1, :lo12:eq_char
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    b exit

exit_empty:
    mov x0, #0
    mov x8, #93
    svc #0

exit:
    mov x0, #0
    mov x8, #93
    svc #0
