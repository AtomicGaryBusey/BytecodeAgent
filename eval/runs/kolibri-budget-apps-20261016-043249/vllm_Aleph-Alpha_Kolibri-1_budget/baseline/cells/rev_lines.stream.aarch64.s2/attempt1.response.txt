

.global _start

.section .rodata
max_count:
    .word 65536
newline:
    .byte 10

.section .bss
buf:
    .space 65536

.section .text
_start:
    adrp x0, buf
    add x0, x0, :lo12:buf
    mov x19, x0
    mov x21, x19
    adrp x0, max_count
    add x0, x0, :lo12:max_count
    ldr w18, [x0]
read_loop:
    mov x1, x21
    mov w2, w18
    mov x0, #0
    mov x8, #63
    svc #0
    mov x5, x0
    cmp x5, #0
    b.eq read_done
    add x21, x21, x5
    b read_loop
read_done:
    mov x20, x21
    mov x21, x19
    adrp x0, newline
    add x0, x0, :lo12:newline
    mov x22, x0
scan_outer:
    cmp x21, x20
    b.ge scan_done
    mov x23, x21
search_loop:
    cmp x23, x20
    b.ge search_no_newline
    ldrb w24, [x23]
    cmp w24, #10
    b.ne search_next
    b search_found
search_next:
    add x23, x23, #1
    b search_loop
search_found:
    b process_line
search_no_newline:
    mov x23, x20
    b process_line
process_line:
    sub x24, x23, x21
    mov x25, x21
    sub x26, x23, #1
rev_loop:
    cmp x21, x26
    b.ge rev_done
    ldrb w0, [x21]
    ldrb w1, [x26]
    strb w1, [x21]
    strb w0, [x26]
    add x21, x21, #1
    sub x26, x26, #1
    b rev_loop
rev_done:
    mov x0, #1
    mov x1, x25
    mov x2, x24
    mov x8, #64
    svc #0
    mov x1, x22
    mov x2, #1
    mov x0, #1
    svc #0
    cmp x23, x20
    b.eq scan_next_final
    add x21, x23, #1
    b scan_outer
scan_next_final:
    mov x21, x20
    b scan_outer
scan_done:
    mov x0, #0
    mov x8, #93
    svc #0
