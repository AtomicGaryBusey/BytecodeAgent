

.section .text
.global _start

_start:

    xor rdi, rdi
    lea rsi, [rip + input_buf]
    mov rdx, 40960
    syscall

    xor rdi, rdi
    lea rsi, [rip + input_buf]

    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

input_buf:
