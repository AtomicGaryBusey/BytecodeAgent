

.section .text
.globl _start

_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + read_buf]
    xor rdi, rdi
    mov rdx, 4096
    syscall
    test rax, rax
    jz read_done
    lea rdi, [rip + input_buf]
    add rdi, r12
    mov rcx, rax
    rep movsb
    add r12, rax
    jmp read_loop
read_done:
    cmp r12, 0
    jle exit_prog
    lea rax, [rip + input_buf + r12 - 1]
