

.section .bss
input_buf: .space 32768
line_start: .space 8000
line_len: .space 8000

.section .text
.globl _start

_start:
    lea rsi, [rip + input_buf]
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    jmp read_loop
done_read:
    lea r8, [rip + input_buf]
    lea r15, [rip + input_buf]
    add r15, r12
    xor r13, r13
    xor r14, r14
    lea r10, [rip + line_start]
    lea r9, [rip + line_len]
parse_loop:
    cmp r8, r15
    jge parse_done
    movzx eax, byte ptr [r8]
    cmp al, 10
    je found_newline
    inc r8
    jmp parse_loop
found_newline:
    mov rdx, r8
    sub rdx, r13
    lea rax, [r10 + r14*8]
    mov qword ptr [rax], r13
    lea rax, [r9 + r14*8]
    mov qword ptr [rax], rdx
    inc r14
    lea rax, [r8 + 1]
    mov r13, rax
    inc r8
    jmp parse_loop
parse_done:
    cmp r13, r15
    jge parse_final
    mov rdx, r15
    sub rdx, r13
    lea rax, [r10 + r14*8]
    mov qword ptr [rax], r13
    lea rax, [r9 + r14*8]
    mov qword ptr [rax], rdx
    inc r14
parse_final:
    mov r12, 0
sorting_outer:
    cmp r12, r14
    jge sorting_done
    mov r13, r12
    inc r13
sorting_inner:
    cmp r13, r14
    jge inner_done
    lea rax, [r10 + r12*8]
    mov r15, qword ptr [rax]
    lea rax, [r9 + r12*8]
    mov rdx, qword ptr [rax]
    lea rax, [r10 + r13*8]
    mov r8, qword ptr [rax]
    lea rax, [r9 + r13*8]
    mov r11, qword ptr [rax]
    mov rsi, r15
    mov rdi, r8
    cmp rdx, r11
    jle set_min
    mov rcx, r11
    jmp rep_cmpsb
set_min:
    mov rcx, rdx
rep_cmpsb:
    jz prefix_case
    jc no_swap_diff
    jmp swap
prefix_case:
    cmp rdx, r11
    jle no_swap_prefix
    jmp swap
no_swap_diff:
    jmp next_j
no_swap_prefix:
    jmp next_j
swap:
    mov rax, r15
    mov r15, r8
    mov r8, rax
    mov rax, rdx
    mov rdx, r11
    mov r11, rax
    lea rax, [r10 + r12*8]
    mov qword ptr [rax], r15
    lea rax, [r10 + r13*8]
    mov qword ptr [rax], r8
    lea rax, [r9 + r12*8]
    mov qword ptr [rax], rdx
    lea rax, [r9 + r13*8]
    mov qword ptr [rax], r11
next_j:
    inc r13
    jmp sorting_inner
inner_done:
    inc r12
    jmp sorting_outer
sorting_done:
    mov r12, 0
output_loop:
    cmp r12, r14
    jge output_done
    lea rax, [r10 + r12*8]
    mov r15, qword ptr [rax]
    lea rax, [r9 + r12*8]
    mov rdx, qword ptr [rax]
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + input_buf]
    add rsi, r15
    inc rdx
    syscall
    inc r12
    jmp output_loop
output_done:
    mov rax, 60
    xor rdi, rdi
    syscall
