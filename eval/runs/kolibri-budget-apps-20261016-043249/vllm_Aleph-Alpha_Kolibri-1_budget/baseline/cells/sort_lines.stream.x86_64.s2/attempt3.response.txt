

.section .bss
input_buf: .space 32768

.section .text
.globl _start

_start:
    xor r12, r12
read_loop:
    xor rax, rax
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    jmp read_loop
done_read:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, r12
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
