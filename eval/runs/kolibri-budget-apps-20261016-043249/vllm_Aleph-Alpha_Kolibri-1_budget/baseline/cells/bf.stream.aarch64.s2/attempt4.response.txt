

.global _start

.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BANG, 0x21
.equ LBRACKET, 0x5B
.equ RBRACKET, 0x5D
.equ GREATER, 0x3E
.equ LESS, 0x3C
.equ PLUS, 0x2B
.equ MINUS, 0x2D
.equ DOT, 0x2E
.equ COMMA, 0x2C
_start:
    adrp x19, stdin_buf
    add x19, x19, :lo12:stdin_buf
    mov x20, #0
read_loop:
    mov x8, #SYS_READ
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
find_excl:
    cmp x21, x20
    b.ge no_excl
    add x1, x19, x21
    ldrb w22, [x1]
    cmp w22, #BANG
    b.eq found_excl
    add x21, x21, #1
    b find_excl
no_excl:
    add x22, x19, x20
    mov x23, x22
    mov x24, x22
    b after_parse
found_excl:
    add x22, x19, x21
    add x23, x22, #1
    add x24, x19, x20
after_parse:
    mov x25, x19
    mov x26, x23
bracket_loop:
    cmp x25, x22
    b.ge bracket_done
    ldrb w27, [x25]
    cmp w27, #LBRACKET
    b.eq bracket_inc
    cmp w27, #RBRACKET
    b.eq bracket_dec
    add x25, x25, #1
    b bracket_loop
bracket_inc:
    add x26, x26, #1
    add x25, x25, #1
    b bracket_loop
bracket_dec:
    sub x26, x26, #1
    add x25, x25, #1
    b bracket_loop
bracket_done:
    cmp x26, #0
    b.ne unbalanced
    adrp x28, tape
    add x28, x28, :lo12:tape
    mov x30, x28
    adrp x28, out_buf
    add x28, x28, :lo12:out_buf
    mov x25, x19
    mov x26, x23
interpreter_loop:
    cmp x25, x22
    b.ge interpreter_done
    ldrb w27, [x25]
    cmp w27, #GREATER
    b.eq op_right
    cmp w27, #LESS
    b.eq op_left
    cmp w27, #PLUS
    b.eq op_plus
    cmp w27, #MINUS
    b.eq op_minus
    cmp w27, #DOT
    b.eq op_dot
    cmp w27, #COMMA
    b.eq op_comma
    cmp w27, #LBRACKET
    b.eq op_lbracket
    cmp w27, #RBRACKET
    b.eq op_rbracket
    add x25, x25, #1
    b interpreter_loop
op_right:
    add x30, x30, #1
    add x25, x25, #1
    b interpreter_loop
op_left:
    sub x30, x30, #1
    add x25, x25, #1
    b interpreter_loop
op_plus:
    ldrb w29, [x30]
    adds w29, w29, #1
    and w29, w29, #255
    strb w29, [x30]
    add x25, x25, #1
    b interpreter_loop
op_minus:
    ldrb w29, [x30]
    subs w29, w29, #1
    and w29, w29, #255
    strb w29, [x30]
    add x25, x25, #1
    b interpreter_loop
op_dot:
    ldrb w29, [x30]
    strb w29, [x28]
    mov x0, #1
    mov x1, x28
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    add x25, x25, #1
    b interpreter_loop
op_comma:
    cmp x26, x24
    b.ge no_input
    ldrb w29, [x26]
    add x26, x26, #1
    strb w29, [x30]
    add x25, x25, #1
    b interpreter_loop
no_input:
    mov w29, #0
    strb w29, [x30]
    add x25, x25, #1
    b interpreter_loop
op_lbracket:
    ldrb w29, [x30]
    cmp w29, #0
    b.ne op_lbracket_cont
    add x1, x25, #1
    mov x2, #1
find_close:
    cmp x1, x22
    b.ge no_close
    ldrb w30, [x1]
    cmp w30, #LBRACKET
    b.eq close_inc
    cmp w30, #RBRACKET
    b.eq close_dec
    add x1, x1, #1
    b find_close
close_inc:
    add x2, x2, #1
    add x1, x1, #1
    b find_close
close_dec:
    sub x2, x2, #1
    cmp x2, #0
    b.eq found_close
    add x1, x1, #1
    b find_close
found_close:
    add x25, x1, #1
    b interpreter_loop
no_close:
    mov x25, x22
    b interpreter_loop
op_lbracket_cont:
    add x25, x25, #1
    b interpreter_loop
op_rbracket:
    ldrb w29, [x30]
    cmp w29, #0
    b.eq op_rbracket_next
    sub x1, x25, #1
    mov x2, #1
find_open:
    cmp x1, x19
    b.lt no_open
    ldrb w30, [x1]
    cmp w30, #LBRACKET
    b.eq open_dec
    cmp w30, #RBRACKET
    b.eq open_inc
    sub x1, x1, #1
    b find_open
open_dec:
    sub x2, x2, #1
    cmp x2, #0
    b.eq found_open
    sub x1, x1, #1
    b find_open
found_open:
    add x25, x1, #1
    b interpreter_loop
no_open:
    mov x25, x19
    b interpreter_loop
op_rbracket_next:
    add x25, x25, #1
    b interpreter_loop
interpreter_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
unbalanced:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.section .bss
stdin_buf:
    .space 65536
tape:
    .space 30000
out_buf:
    .space 1
