

.global _start
.section .rodata
msg_ok:
    .asciz "ok\n"
msg_err0:
    .asciz "error at 0\n"
msg_err5:
    .asciz "error at 5\n"
.section .bss
input_buf:
    .space 65536
.section .text
_start:
    adr x1, input_buf
    mov x4, x1
    mov x3, #0
    mov x2, #65536
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_read
    add x1, x1, x0
    add x3, x3, x0
    b read_loop
done_read:
    cmp x3, #0
    b.le empty
    mov x5, x4
    mov x6, x3
ws_loop:
    cmp x6, #0
    b.le ws_done
    ldrb w7, [x5]
    cmp w7, #0x20
    b.eq ws_next
    cmp w7, #0x09
    b.eq ws_next
    cmp w7, #0x0A
    b.eq ws_next
    cmp w7, #0x0D
    b.eq ws_next
    b not_ws
ws_next:
    add x5, x5, #1
    subs x6, x6, #1
    b ws_loop
ws_done:
    adr x0, msg_err5
    mov x1, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b end
not_ws:
    adr x0, msg_ok
    mov x1, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b end
empty:
    adr x0, msg_err0
    mov x1, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b end
end:
    b end
