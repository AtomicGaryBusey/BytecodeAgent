

.global _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err0:
    .ascii "error at 0\n"

.section .text
_start:
    movz x3, #0x0001, lsl #16
    sub sp, sp, x3
    mov x1, sp
    mov x19, #0
    mov x20, #0
    b read_loop
read_loop:
    sub x2, x3, x19
    cmp x2, #0
    b.le read_done
    mov x0, #0
    add x4, x1, x19
    mov x1, x4
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    add x19, x19, x0
    cmp x0, x2
    b.ne read_done
    b read_loop
read_done:
    sub x1, x1, x20
    cmp x20, #0
    b.ne not_empty
    adrp x21, msg_err0
    add x21, x21, :lo12:msg_err0
    mov x0, #1
    mov x1, x21
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
not_empty:
    adrp x21, msg_ok
    add x21, x21, :lo12:msg_ok
    mov x0, #1
    mov x1, x21
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
