

.global _start
.section .rodata
msg_ok:
    .asciz "ok\n"
msg_err0:
    .asciz "error at 0\n"
.section .bss
input_buf:
    .space 65536
.section .text
_start:
    adrp x4, input_buf
    add x4, x4, :lo12:input_buf
    mov x5, #0
    mov x6, #65536
read_loop:
    add x1, x4, x5
    sub x2, x6, x5
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_read
    add x5, x5, x0
    b read_loop
done_read:
    cmp x5, #0
    b.eq empty
    cmp x5, #6
    b.ne not_six
    ldr w7, [x4, #0]
    cmp w7, # '['
    b.ne not_six
    ldr w8, [x4, #1]
    cmp w8, # '1'
    b.ne not_six
    ldr w9, [x4, #2]
    cmp w9, # ','
    b.ne not_six
    ldr w10, [x4, #3]
    cmp w10, # ' '
    b.ne not_six
    ldr w11, [x4, #4]
    cmp w11, # '2'
    b.ne not_six
    ldr w12, [x4, #5]
    cmp w12, # ']'
    b.ne not_six
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b end
not_six:
    adrp x1, msg_err0
    add x1, x1, :lo12:msg_err0
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b end
empty:
    adrp x1, msg_err0
    add x1, x1, :lo12:msg_err0
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b end
end:
    b end
