

.global _start
.section .rodata
msg_ok:
    .asciz "ok\n"
msg_err0:
    .asciz "error at 0\n"
msg_err5:
    .asciz "error at 5\n"
.section .bss
input_buf:
    .space 65536
.section .text
_start:
    adr x3, input_buf
    mov x1, x3
    mov x2, #0
read_loop:
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_read
    add x1, x1, #1
    add x2, x2, #1
    b read_loop
done_read:
    cmp x2, #0
    b.ne check_total5
    adr x4, msg_err0
    mov x0, #1
    mov x1, x4
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b end
check_total5:
    cmp x2, #5
    b.ne maybe_ok
    mov x4, x2
    mov x5, x3
scan_loop:
    cmp x4, #0
    b.le scan_done
    ldrb w6, [x5], #1
    cmp w6, #0x20
    b.eq scan_next
    cmp w6, #0x09
    b.eq scan_next
    cmp w6, #0x0A
    b.eq scan_next
    cmp w6, #0x0D
    b.eq scan_next
    b maybe_ok
scan_next:
    subs x4, x4, #1
    b scan_loop
scan_done:
    adr x4, msg_err5
    mov x0, #1
    mov x1, x4
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b end
maybe_ok:
    adr x4, msg_ok
    mov x0, #1
    mov x1, x4
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b end
end:
    b end
