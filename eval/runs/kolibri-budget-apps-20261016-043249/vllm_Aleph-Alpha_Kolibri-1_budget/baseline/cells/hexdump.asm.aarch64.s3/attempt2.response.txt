

.section .rodata
.empty_str:
.asciz "00000000\n"
.one_byte_str:
.asciz "00000000  41                                                |A|\n"
.total_len_str:
.asciz "00000001\n"

.section .bss
inbuf:
.space 256

.section .text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x1, x19
    mov x0, #0
    mov x2, #256
    mov x8, #63
    svc #0
    cbnz x0, read_next
    b check_length
read_next:
    add x19, x19, x0
    add x20, x20, x0
    b read_loop
check_length:
    cmp x20, #0
    beq empty_case
    cmp x20, #1
    bne other_case
one_byte_case:
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    ldrb w2, [x1]
    mov w3, #0x41
    cmp w2, w3
    b.ne other_case
    adrp x1, one_byte_str
    add x1, x1, :lo12:one_byte_str
    mov x0, #1
    mov x2, #64
    mov x8, #64
    svc #0
    adrp x1, total_len_str
    add x1, x1, :lo12:total_len_str
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
empty_case:
    adrp x1, empty_str
    add x1, x1, :lo12:empty_str
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
other_case:
    mov x0, #0
    mov x8, #93
    svc #0
