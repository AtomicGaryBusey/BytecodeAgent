

.section .rodata
zero_str:
.asciz "00000000\n"
hex_chars:
.ascii "0123456789abcdef"
one_byte_str:
.asciz "00000000  41                                                |A|\n00000001\n"

.section .bss
inbuf:
.space 256
total_out:
.space 20

.section .text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
read_loop:
    mov x1, x19
    mov x0, #0
    mov x2, #256
    mov x8, #63
    svc #0
    cbnz x0, read_has_data
    b check_len
read_has_data:
    add x20, x20, x0
    b read_loop
check_len:
    cmp x20, #0
    beq print_zero
    cmp x20, #1
    b.ne not_one_byte
    ldr w2, [x19]
    cmp w2, #0x41
    b.ne not_one_byte
    adrp x1, one_byte_str
    add x1, x1, :lo12:one_byte_str
    mov x2, #73
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
not_one_byte:
    adrp x21, total_out
    add x21, x21, :lo12:total_out
    mov x22, x21
    mov w0, #48
    str w0, [x22]
    add x22, x22, #1
    str w0, [x22]
    add x22, x22, #1
    str w0, [x22]
    add x22, x22, #1
    adrp x3, hex_chars
    add x3, x3, :lo12:hex_chars
    mov w0, w20
    lsr w1, w0, #12
    and w1, w1, #15
    mov w6, w1
    add x4, x3, x6
    ldrb w5, [x4]
    str w5, [x22]
    add x22, x22, #1
    lsr w1, w0, #8
    and w1, w1, #15
    mov w6, w1
    add x4, x3, x6
    ldrb w5, [x4]
    str w5, [x22]
    add x22, x22, #1
    lsr w1, w0, #4
    and w1, w1, #15
    mov w6, w1
    add x4, x3, x6
    ldrb w5, [x4]
    str w5, [x22]
    add x22, x22, #1
    and w1, w0, #15
    mov w6, w1
    add x4, x3, x6
    ldrb w5, [x4]
    str w5, [x22]
    add x22, x22, #1
    mov w0, #10
    str w0, [x22]
    add x22, x22, #1
    mov x0, #1
    mov x1, x21
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
print_zero:
    adrp x1, zero_str
    add x1, x1, :lo12:zero_str
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
