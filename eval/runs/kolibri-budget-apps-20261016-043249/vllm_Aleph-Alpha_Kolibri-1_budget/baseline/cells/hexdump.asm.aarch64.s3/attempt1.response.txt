

.section .bss
.global inbuf
inbuf:
.space 1

.section .rodata
.global output
output:
.ascii "00000000\n"

.section .text
.globl _start

_start:
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x8, #63
read_loop:
    mov x0, #0
    mov x2, #1
    svc #0
    cbnz x0, read_loop
    adrp x1, output
    add x1, x1, :lo12:output
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
