

section .data
newline:
  .byte 10

section .bss
input_buf:
  .space 65536

section .text
.global _start

.equ BUF_SIZE, 65536

_start:
  mov r13, offset input_buf
  mov r14, 0
read_loop:
  cmp r14, BUF_SIZE
  jae done_reading
  mov rax, 0
  mov rdi, 0
  lea rsi, [r13 + r14]
  mov rdx, BUF_SIZE
  sub rdx, r14
  syscall
  add r14, rax
  test rax, rax
  jne read_loop
done_reading:
  mov r15, 0
  mov r17, offset newline
process_loop:
  cmp r15, r14
  jae done_processing
  mov r16, r15
scan:
  cmp r16, r14
  jae last_line
  mov al, byte ptr [r13 + r16]
  test al, al
  jne scan_next
  mov rcx, r16
  sub rcx, r15
  mov rdi, r15
  mov rsi, r16
  dec rsi
reverse_loop:
  cmp rdi, rsi
  jae reverse_done
  mov al, byte ptr [r13 + rdi]
  mov bl, byte ptr [r13 + rsi]
  mov byte ptr [r13 + rdi], bl
  mov byte ptr [r13 + rsi], al
  inc rdi
  dec rsi
  jmp reverse_loop
reverse_done:
  mov rax, 1
  mov rdi, 1
  lea rsi, [r13 + r15]
  mov rdx, rcx
  syscall
  mov rsi, r17
  mov rdx, 1
  syscall
  inc r16
  mov r15, r16
  jmp process_loop
scan_next:
  inc r16
  jmp scan
last_line:
  mov rcx, r14
  sub rcx, r15
  mov rdi, r15
  mov rsi, r14
  dec rsi
reverse_loop2:
  cmp rdi, rsi
  jae reverse_done2
  mov al, byte ptr [r13 + rdi]
  mov bl, byte ptr [r13 + rsi]
  mov byte ptr [r13 + rdi], bl
  mov byte ptr [r13 + rsi], al
  inc rdi
  dec rsi
  jmp reverse_loop2
reverse_done2:
  mov rax, 1
  mov rdi, 1
  lea rsi, [r13 + r15]
  mov rdx, rcx
  syscall
  mov rsi, r17
  mov rdx, 1
  syscall
done_processing:
  mov rax, 60
  xor rdi, rdi
  syscall
