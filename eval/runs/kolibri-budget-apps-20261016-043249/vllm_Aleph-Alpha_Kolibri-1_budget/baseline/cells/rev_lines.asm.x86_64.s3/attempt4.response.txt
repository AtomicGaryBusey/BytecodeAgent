

.section .rodata
newline:
  .byte 10

.section .bss
input_buf:
  .space 65536

.section .text
.global _start

_start:
  lea r8, [rip + input_buf]
  xor r10, r10
read_loop:
  cmp r10, 65536
  jae read_done
  mov rdx, 65536
  sub rdx, r10
  lea rsi, [r8 + r10]
  mov rax, 0
  mov rdi, 0
  syscall
  add r10, rax
  test rax, rax
  jne read_loop
read_done:
  mov r9, r10
  xor r10, r10
  xor r12, r12
  xor r13, r13
  jmp outer_loop

outer_loop:
  cmp r12, r9
  jae exit
  mov r13, r12
scan_loop:
  cmp r13, r9
  jae no_newline
  mov al, byte ptr [r8 + r13]
  cmp al, 10
  je found_newline
  inc r13
  jmp scan_loop
found_newline:
  mov r15, r13
  cmp r12, r15
  je empty_line_found
  mov r14, r15
  dec r14
  mov r10, 1
  jmp print_loop
empty_line_found:
  mov r14, r12
  dec r14
  mov r10, 1
  jmp print_loop
no_newline:
  mov r14, r9
  dec r14
  mov r10, 0
  jmp print_loop

print_loop:
  cmp r14, r12
  jl print_newline
  mov rax, 1
  mov rdi, 1
  lea rsi, [r8 + r14]
  mov rdx, 1
  syscall
  dec r14
  jmp print_loop
print_newline:
  lea rsi, [rip + newline]
  mov rax, 1
  mov rdi, 1
  mov rdx, 1
  syscall
  test r10, r10
  jz exit
  lea r12, [r15 + 1]
  jmp outer_loop

exit:
  mov rax, 60
  xor rdi, rdi
  syscall
