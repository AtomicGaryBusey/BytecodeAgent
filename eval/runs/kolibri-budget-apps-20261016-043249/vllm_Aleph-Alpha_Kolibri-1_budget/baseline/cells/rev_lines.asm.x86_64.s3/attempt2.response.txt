

section .rodata
newline:
  .byte 10

section .bss
input_buf:
  .space 65536

section .text
.global _start

_start:
  lea r8, [rip + input_buf]
  xor r9, r9
read_loop:
  cmp r9, 65536
  jae done_reading
  mov rdx, 65536
  sub rdx, r9
  mov rax, 0
  mov rdi, 0
  mov rsi, r8
  syscall
  add r9, rax
  test rax, rax
  jne read_loop
done_reading:
  xor r10, r10
outer_loop:
  cmp r10, r9
  jae done_processing
  mov r11, r10
scan_loop:
  cmp r11, r9
  jae scan_no_newline
  mov al, byte ptr [r8 + r11]
  cmp al, 10
  je scan_found_newline
  inc r11
  jmp scan_loop
scan_found_newline:
  mov r12, r11
  jmp after_scan
scan_no_newline:
  mov r12, r9
  jmp after_scan
after_scan:
  cmp r12, r9
  jl scan_has_newline
  mov r14, 0
  mov r13, r9
  sub r13, 1
  jmp write_loop
scan_has_newline:
  mov r14, 1
  mov r13, r12
  dec r13
  jmp write_loop
write_loop:
  cmp r13, r10
  jl write_done
  mov rax, 1
  mov rdi, 1
  lea rsi, [r8 + r13]
  mov rdx, 1
  syscall
  dec r13
  jmp write_loop
write_done:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + newline]
  mov rdx, 1
  syscall
  test r14, r14
  jz update_no_newline
  add r12, 1
  mov r10, r12
  jmp outer_loop
update_no_newline:
  mov r10, r9
  jmp outer_loop
done_processing:
  mov rax, 60
  xor rdi, rdi
  syscall
