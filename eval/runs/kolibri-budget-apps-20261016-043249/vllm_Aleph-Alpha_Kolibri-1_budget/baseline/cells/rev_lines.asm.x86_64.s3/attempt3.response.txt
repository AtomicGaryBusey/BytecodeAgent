

section .rodata
newline:
  .byte 10

section .bss
input_buf:
  .space 65536

section .text
.global _start

_start:
  lea r8, [rip + input_buf]
  xor r10, r10
read_loop:
  cmp r10, 65536
  jae read_done
  mov rdx, 65536
  sub rdx, r10
  lea rsi, [r8 + r10]
  mov rax, 0
  mov rdi, 0
  syscall
  add r10, rax
  test rax, rax
  jne read_loop
read_done:
  mov r9, r10
  xor r10, r10
  jmp outer_loop
outer_loop:
  cmp r10, r9
  jae done_processing
  mov r11, r10
scan_loop:
  cmp r11, r9
  jae no_newline
  mov al, byte ptr [r8 + r11]
  cmp al, 10
  je found_newline
  inc r11
  jmp scan_loop
found_newline:
  mov r14, r11
  dec r14
  mov r12, 1
  jmp print_loop
no_newline:
  mov r14, r9
  dec r14
  mov r12, 0
  jmp print_loop
print_loop:
  cmp r14, r10
  jl print_newline
  mov rax, 1
  mov rdi, 1
  lea rsi, [r8 + r14]
  mov rdx, 1
  syscall
  dec r14
  jmp print_loop
print_newline:
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + newline]
  mov rdx, 1
  syscall
  test r12, r12
  jz update_no_newline
  lea r10, [r11 + 1]
  jmp outer_loop
update_no_newline:
  mov r10, r9
  jmp outer_loop
done_processing:
  mov rax, 60
  xor rdi, rdi
  syscall
