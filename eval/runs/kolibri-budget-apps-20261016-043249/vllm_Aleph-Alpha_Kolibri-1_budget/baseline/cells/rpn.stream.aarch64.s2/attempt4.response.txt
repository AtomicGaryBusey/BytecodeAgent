.section .text
_start:
    adr x1, buf
read_loop:
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done_read
    b read_loop
done_read:
    adr x1, output
    mov x2, #20
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
buf:
    .space 65536
