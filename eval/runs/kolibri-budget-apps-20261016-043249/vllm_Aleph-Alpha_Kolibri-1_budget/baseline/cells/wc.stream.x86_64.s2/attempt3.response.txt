

.globl _start

.section .bss
read_buf: .space 65536
out_buf: .space 32
temp_buf: .space 8

.section .text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
    mov r10, read_buf
    mov r8, read_buf

read_loop:
    mov rdi, 0
    mov rsi, r8
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add r8, rax
    jmp read_loop

read_done:
    mov rsi, r10
    mov rdx, r12
    mov r15, 0
    jmp process_loop

print_zero:
    mov al, '0'
    mov [r11], al
    inc r11
    ret

digit_done:
    lea rax, [rip + temp_buf]
    add rax, 8
    sub rax, rsi
    mov rcx, rax

copy_loop:
    mov al, byte ptr [rsi]
    mov [r11], al
    inc r11
    inc rsi
    dec rcx
    jnz copy_loop
    ret

digit_loop:
    dec rsi
    xor rdx, rdx
    mov r8, 10
    div r8
    mov al, '0'
    add al, rdx
