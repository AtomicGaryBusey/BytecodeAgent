

.globl _start

.section .rodata
msg: .ascii "0 0 0\n"

.section .bss
buf: .space 65536

.section .text
.equ BUF_SIZE, 65536
_start:
read_loop:
    mov rdi, 0
    lea rsi, [rip + buf]
    mov rdx, BUF_SIZE
    syscall
    test rax, rax
    jz read_done
    jmp read_loop
read_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg]
    mov rdx, 6
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
