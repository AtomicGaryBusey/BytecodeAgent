

.globl _start

.section .rodata

.section .bss
read_buf: .space 65536
out_buf: .space 32

.section .text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    mov r15, 1
    lea r10, [rip + read_buf]

read_loop:
    mov rdi, 0
    mov rsi, r10
    mov rdx, 65536
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    mov r8, r10
    mov r9, rax
process_loop:
    movzx eax, byte ptr [r8]
    cmp al, 0x0A
    je is_newline
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    test r15, r15
    jz non_ws_no_word
    inc r14
non_ws_no_word:
    mov r15, 0
    jmp next_byte
is_newline:
    inc r13
    mov r15, 1
    jmp next_byte
is_ws:
    mov r15, 1
next_byte:
    inc r8
    dec r9
    jnz process_loop
    jmp read_loop

read_done:
    xor r11, r11
    mov al, '0'
    add al, r13b
    mov [out_buf + r11], al
    incr r11
