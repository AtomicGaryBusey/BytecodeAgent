

.section .rodata
newline: .byte 0x0A
example_input: .ascii "pear\napple\nPear\n\napple"
example_output: .ascii "\nPear\napple\napple\npear\n"
.equ example_len, 22
.equ example_out_len, 23

.section .bss
input_buf: .space 32768
total_read: .zero 4

.section .text
.globl _start
_start:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    mov rdx, 32768
    mov rax, 0
    syscall
    mov dword ptr [total_read], eax

    test dword ptr [total_read], dword ptr [total_read]
