

.equ INPUT_SIZE, 32768

.section .rodata
newline: .byte 0x0A

.section .bss
input_buf: .space INPUT_SIZE

.section .text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
read_loop:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, INPUT_SIZE
    sub rdx, r8
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    add r8, rax
    jmp read_loop
read_done:
    test r9, r9
    jz exit
    lea r10, [rip + input_buf]
    add r10, r9
    xor r8, r8
    lea rsi, [rip + input_buf]
output_loop:
    cmp rsi, r10
    jge output_done
    mov al, byte ptr [rsi]
    cmp al, 0x0A
    jne output_next
    mov rdx, rsi
    sub rdx, r8
    mov rdi, 1
    lea r12, [rip + input_buf]
    add r12, r8
    mov rax, 1
    syscall
    mov rdi, 1
    lea r13, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    lea r8, [rsi + 1]
    inc rsi
    jmp output_loop
output_next:
    inc rsi
    jmp output_loop
output_done:
    cmp r8, r10
    jne output_final
    jmp exit
output_final:
    mov rdx, r10
    sub rdx, r8
    mov rdi, 1
    lea r12, [rip + input_buf]
    add r12, r8
    mov rax, 1
    syscall
    mov rdi, 1
    lea r13, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
exit:
    mov rdi, 0
    mov rax, 60
    syscall
