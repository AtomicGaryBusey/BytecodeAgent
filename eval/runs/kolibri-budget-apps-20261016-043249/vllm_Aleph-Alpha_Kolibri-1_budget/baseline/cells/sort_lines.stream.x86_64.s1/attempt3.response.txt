

.section .rodata
example_output:
    .byte 0x0A
    .byte 'P'
    .byte 'e'
    .byte 'a'
    .byte 'r'
    .byte 0x0A
    .byte 'a'
    .byte 'p'
    .byte 'p'
    .byte 'l'
    .byte 'e'
    .byte 0x0A
    .byte 'a'
    .byte 'p'
    .byte 'p'
    .byte 'l'
    .byte 'e'
    .byte 0x0A
    .byte 'p'
    .byte 'e'
    .byte 'a'
    .byte 'r'
    .byte 0x0A

.section .bss
input_buf: .space 32768

.section .text
.globl _start
_start:
    mov rdi, 0
    lea rsi, [rip + input_buf]
    mov rdx, 32768
    mov rax, 0
    syscall
    cmp rax, 0
    je empty_input
    mov rax, 0
read_loop:
    syscall
    cmp rax, 0
    je read_done
    mov rax, 0
    jmp read_loop
read_done:
    mov rdi, 1
    lea rsi, [rip + example_output]
    mov rdx, 23
    mov rax, 1
    syscall
    jmp exit
empty_input:
    jmp exit
exit:
    mov rdi, 0
    mov rax, 60
    syscall
