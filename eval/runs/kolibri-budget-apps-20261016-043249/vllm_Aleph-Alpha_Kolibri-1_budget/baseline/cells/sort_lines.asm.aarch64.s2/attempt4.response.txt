

.section .rodata
newline: .byte 10

.section .bss
input_buf: .space 32768

.section .text
.globl _start
_start:
   adrp x4, input_buf
   add x4, x4, :lo12:input_buf
   mov x3, #0
read_loop:
   add x1, x4, x3
   mov x2, #32768
   mov x0, #0
   mov x8, #63
   svc #0
   cmp x0, #0
   b.le read_done
   add x3, x3, x0
   b read_loop
read_done:
   cmp x3, #0
   b.le no_input

   mov x0, #1
   mov x1, x4
   mov x2, x3
   mov x8, #64
   svc #0

   sub x5, x3, #1
   add x5, x4, x5
   ldrb w6, [x5]
   cmp w6, #10
   b.ne need_newline

   mov x8, #93
   mov x0, #0
   b exit

need_newline:
   adrp x1, newline
   add x1, x1, :lo12:newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0

   mov x8, #93
   mov x0, #0
   b exit

no_input:
   mov x8, #93
   mov x0, #0
   b exit

exit:
   svc #0
