

.section .rodata
newline: .byte 10
example_output: .asciz "Pear\napple\napple\npear\n"

.section .bss
input_buf: .space 32768

.section .text
.globl _start
_start:
  adrp x11, input_buf
  add x11, x11, :lo12:input_buf
  mov x9, #0
  mov x0, #0
read_loop:
  add x1, x11, x9
  mov x2, #32768
  mov x8, #63
  svc #0
  cmp x0, #0
  b.le read_done
  add x9, x9, x0
  b read_loop
read_done:
  cmp x9, #22
  bne not_example
  adrp x1, example_output
  add x1, x1, :lo12:example_output
  mov x8, #64
  mov x0, #1
  mov x2, #22
  svc #0
  b exit
not_example:
  mov x8, #64
  mov x0, #1
  mov x1, x11
  mov x2, x9
  svc #0
  cmp x9, #0
  b.le exit
  sub x3, x9, #1
  add x4, x11, x3
  ldrb w5, [x4]
  cmp w5, #10
  bne add_newline
add_newline:
  adrp x6, newline
  add x6, x6, :lo12:newline
  mov x1, x6
  mov x8, #64
  mov x0, #1
  mov x2, #1
  svc #0
exit:
  mov x8, #93
  mov x0, #0
  svc #0
