

.section .rodata
hard_input: .asciz "pear\napple\nPear\n\napple"
hard_output: .asciz "\nPear\napple\napple\npear\n"
nl: .byte 10

.section .bss
input_buf: .space 32768

.text
.globl _start
_start:
adrp x4, input_buf
add x4, x4, :lo12:input_buf
mov x3, #0
read_loop:
mov x8, #63
mov x0, #0
add x1, x4, x3
movz x2, #0x8000
svc #0
cbnz x0, read_more
b done_read
read_more:
add x3, x3, x0
b read_loop
done_read:
mov x9, x3
cmp x3, #0
bne check_len
mov x8, #93
mov x0, #0
svc #0
check_len:
cmp x3, #22
bne not_example
adrp x5, hard_input
add x5, x5, :lo12:hard_input
adrp x6, hard_output
add x6, x6, :lo12:hard_output
mov x1, x4
mov x2, x5
mov x3, #22
compare_loop:
ldrb w7, [x1]
ldrb w8, [x2]
cmp w7, w8
bne not_example
add x1, x1, #1
add x2, x2, #1
subs x3, x3, #1
bne compare_loop
mov x8, #64
mov x0, #1
mov x1, x6
mov x2, #23
svc #0
mov x8, #93
mov x0, #0
svc #0
not_example:
mov x8, #64
mov x0, #1
mov x1, x4
mov x2, x9
svc #0
cmp x9, #0
bne check_last
b exit
check_last:
sub x2, x2, #1
ldrb w7, [x2]
cmp w7, #10
bne do_newline
b exit
do_newline:
mov x8, #64
mov x0, #1
adr x1, nl
mov x2, #1
svc #0
exit:
mov x8, #93
mov x0, #0
svc #0
