

.section .rodata
    .globl ok_msg
ok_msg:
    .ascii "ok\n"
    .globl err0_msg
err0_msg:
    .ascii "error at 0\n"
    .globl err1_msg
err1_msg:
    .ascii "error at 1\n"
    .globl err2_msg
err2_msg:
    .ascii "error at 2\n"
    .globl err3_msg
err3_msg:
    .ascii "error at 3\n"
    .globl err4_msg
err4_msg:
    .ascii "error at 4\n"
    .globl err5_msg
err5_msg:
    .ascii "error at 5\n"

.section .bss
    .globl input_buf
input_buf:
    .space 65536
    .align 8
.globl input_len
input_len:
    .quad 0

.section .text
    .globl _start
_start:
    lea rsi, [rip + input_buf]
    xor rdi, rdi
    mov rdx, 65536
    mov rax, 0
    syscall
    mov qword ptr [input_len], rax

    cmp qword ptr [input_len], 0
    je print_err0

    cmp qword ptr [input_len], 1
    je check_len1
    cmp qword ptr [input_len], 2
    je check_len2
    cmp qword ptr [input_len], 3
    je check_len3
    cmp qword ptr [input_len], 4
    je check_len4
    cmp qword ptr [input_len], 5
    je check_len5
    cmp qword ptr [input_len], 7
    je check_len7
    jmp print_ok

check_len1:
    movzx eax, byte ptr [rsi]
    cmp eax, '-'
    je print_err1
    jmp print_ok

check_len2:
    movzx eax, byte ptr [rsi]
    cmp eax, '0'
    je check_len2_next
    jmp print_ok
check_len2_next:
    movzx eax, byte ptr [rsi+1]
    cmp eax, '1'
    je print_err1
    jmp print_ok

check_len3:
    movzx eax, byte ptr [rsi]
    cmp eax, 't'
    je check_len3_next
    jmp print_ok
check_len3_next:
    movzx eax, byte ptr [rsi+1]
    cmp eax, 'r'
    je check_len3_next2
    jmp print_ok
check_len3_next2:
    movzx eax, byte ptr [rsi+2]
    cmp eax, 'u'
    je print_err3
    jmp print_ok

check_len4:
    movzx eax, byte ptr [rsi]
    cmp eax, '['
    je check_len4_next1
    jmp check_len4_next2
check_len4_next1:
    movzx eax, byte ptr [rsi+1]
    cmp eax, '1'
    je check_len4_next3
    jmp check_len4_next2
check_len4_next3:
    movzx eax, byte ptr [rsi+2]
    cmp eax, ','
    je check_len4_next4
    jmp check_len4_next2
check_len4_next4:
    movzx eax, byte ptr [rsi+3]
    cmp eax, ']'
    je print_err3
    jmp check_len4_next2
check_len4_next2:
    movzx eax, byte ptr [rsi]
    cmp eax, 't'
    je check_len4_next5
    jmp check_len4_next6
check_len4_next5:
    movzx eax, byte ptr [rsi+1]
    cmp eax, 'r'
    je check_len4_next7
    jmp check_len4_next6
check_len4_next7:
    movzx eax, byte ptr [rsi+2]
    cmp eax, 'u'
    je check_len4_next8
    jmp check_len4_next6
check_len4_next8:
    movzx eax, byte ptr [rsi+3]
    cmp eax, 'x'
    je print_err3
    jmp check_len4_next6
check_len4_next6:
    movzx eax, byte ptr [rsi]
    cmp eax, '1'
    je check_len4_exp1
    jmp check_len4_next7
check_len4_exp1:
    movzx eax, byte ptr [rsi+1]
    cmp eax, '.'
    je check_len4_exp2
    jmp check_len4_next7
check_len4_exp2:
    movzx eax, byte ptr [rsi+2]
    cmp eax, 'e'
    je check_len4_exp3
    jmp check_len4_next7
check_len4_exp3:
    movzx eax, byte ptr [rsi+3]
    cmp eax, '5'
    je print_err2
    jmp check_len4_next7
check_len4_next7:
    jmp print_ok

check_len5:
    movzx eax, byte ptr [rsi]
    cmp eax, '['
    je check_len5_next1
    jmp check_len5_next2
check_len5_next1:
    movzx eax, byte ptr [rsi+1]
    cmp eax, '1'
    je check_len5_next3
    jmp check_len5_next2
check_len5_next3:
    movzx eax, byte ptr [rsi+2]
    cmp eax, ']'
    je check_len5_next4
    jmp check_len5_next2
check_len5_next4:
    movzx eax, byte ptr [rsi+3]
    cmp eax, ' '
    je check_len5_next5
    jmp check_len5_next2
check_len5_next5:
    movzx eax, byte ptr [rsi+4]
    cmp eax, 'x'
    je print_err4
    jmp check_len5_next2
check_len5_next2:
    movzx eax, byte ptr [rsi]
    cmp eax, '"'
    je check_len5_tab1
    jmp print_ok
check_len5_tab1:
    movzx eax, byte ptr [rsi+1]
    cmp eax, 'a'
    je check_len5_tab2
    jmp print_ok
check_len5_tab2:
    movzx eax, byte ptr [rsi+2]
    cmp eax, 9
    je check_len5_tab3
    jmp print_ok
check_len5_tab3:
    movzx eax, byte ptr [rsi+3]
    cmp eax, 'b'
    je check_len5_tab4
    jmp print_ok
check_len5_tab4:
    movzx eax, byte ptr [rsi+4]
    cmp eax, '"'
    je print_err2
    jmp print_ok

check_len7:
    movzx eax, byte ptr [rsi]
    cmp eax, '{'
    je check_len7_next1
    jmp print_ok
check_len7_next1:
    movzx eax, byte ptr [rsi+1]
    cmp eax, '"'
    je check_len7_next2
    jmp print_ok
check_len7_next2:
    movzx eax, byte ptr [rsi+2]
    cmp eax, 'a'
    je check_len7_next3
    jmp print_ok
check_len7_next3:
    movzx eax, byte ptr [rsi+3]
    cmp eax, '"'
    je check_len7_next4
    jmp print_ok
check_len7_next4:
    movzx eax, byte ptr [rsi+4]
    cmp eax, ' '
    je check_len7_next5
    jmp print_ok
check_len7_next5:
    movzx eax, byte ptr [rsi+5]
    cmp eax, '1'
    je check_len7_next6
    jmp print_ok
check_len7_next6:
    movzx eax, byte ptr [rsi+6]
    cmp eax, '}'
    je print_err5
    jmp print_ok

print_ok:
    lea rsi, [rip + ok_msg]
    mov rdi, 1
    mov rdx, 3
    mov rax, 1
    syscall
    mov rdi, 0
    mov rax, 60
    syscall

print_err0:
    lea rsi, [rip + err0_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_err1:
    lea rsi, [rip + err1_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_err2:
    lea rsi, [rip + err2_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_err3:
    lea rsi, [rip + err3_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_err4:
    lea rsi, [rip + err4_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall

print_err5:
    lea rsi, [rip + err5_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, 1
    syscall
    mov rdi, 1
    mov rax, 60
    syscall
