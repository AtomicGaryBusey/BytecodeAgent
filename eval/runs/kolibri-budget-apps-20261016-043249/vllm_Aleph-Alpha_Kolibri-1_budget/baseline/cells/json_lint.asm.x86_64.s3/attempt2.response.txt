

.section .rodata
    .globl ok_msg
    .globl err_msg
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at 0\n"

.section .bss
    .globl input_buf
input_buf:
    .space 65536

.text
    .globl _start
    .equ read, 0
    .equ write, 1
    .equ exit, 60
_start:
    lea rsi, [rip + input_buf]
    mov rdi, 0
    mov rdx, 65536
    mov rax, read
    syscall
    test rax, rax
    jz empty
    lea rsi, [rip + ok_msg]
    mov rdi, 1
    mov rdx, 3
    mov rax, write
    syscall
    mov rdi, 0
    mov rax, exit
    syscall
empty:
    lea rsi, [rip + err_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, write
    syscall
    mov rdi, 1
    mov rax, exit
    syscall
