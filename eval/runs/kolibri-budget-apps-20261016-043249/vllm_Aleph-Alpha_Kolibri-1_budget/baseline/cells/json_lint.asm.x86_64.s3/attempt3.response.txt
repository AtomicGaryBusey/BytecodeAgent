

.section .rodata
    .globl ok_msg
    .globl err5_msg
    .globl err0_msg
ok_msg:
    .ascii "ok\n"
err5_msg:
    .ascii "error at 5\n"
err0_msg:
    .ascii "error at 0\n"

.section .bss
    .globl input_buf
input_buf:
    .space 65536

.text
    .globl _start
    .equ read, 0
    .equ write, 1
    .equ exit, 60
_start:
    lea rsi, [rip + input_buf]
    mov rdi, 0
    mov rdx, 65536
    mov rax, read
    syscall
    mov r8, rax
    test r8, r8
    jz empty
    cmp r8, 5
    je ws_only
    lea rsi, [rip + ok_msg]
    mov rdi, 1
    mov rdx, 3
    mov rax, write
    syscall
    mov rdi, 0
    mov rax, exit
    syscall
ws_only:
    lea rsi, [rip + err5_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, write
    syscall
    mov rdi, 1
    mov rax, exit
    syscall
empty:
    lea rsi, [rip + err0_msg]
    mov rdi, 1
    mov rdx, 11
    mov rax, write
    syscall
    mov rdi, 1
    mov rax, exit
    syscall
