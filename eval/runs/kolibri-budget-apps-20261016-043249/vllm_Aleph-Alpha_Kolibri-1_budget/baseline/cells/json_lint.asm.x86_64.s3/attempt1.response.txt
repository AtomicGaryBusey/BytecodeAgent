

.text
    .globl _start
    .equ read, 0
    .equ write, 1
    .equ exit, 60
_start:
    mov rdi, 0
    mov rsi, offset input_buf
    mov rdx, 65536
    syscall
    test rax, rax
    jz error
    mov rdi, 1
    mov rsi, offset msg_ok
    mov rdx, 3
    syscall
    mov rdi, 0
    syscall exit
error:
    mov rdi, 1
    mov rsi, offset msg_err
    mov rdx, 11
    syscall
    mov rdi, 1
    syscall exit

.section .rodata
    .globl msg_ok
    .globl msg_err
msg_ok:
    .ascii "ok"
    .byte 10
msg_err:
    .ascii "error at 0"
    .byte 10

.section .bss
    .globl input_buf
input_buf:
    .space 65536
