

.global _start

.section .bss
inbuf:
   .space 65536

.section .text
_start:
   xor rbx, rbx
   mov rdi, 0
   lea rsi, [inbuf]
   mov rdx, 65536
.read_loop:
   syscall
   test rax, rax
   jle .read_end
   add rbx, rax
   add rsi, rax
   mov rdx, 65536
   sub rdx, rbx
   jmp .read_loop
.read_end:
   lea rsi, [inbuf]
   mov rcx, rbx
   xor rbx, rbx
   xor r8, r8
.scan_loop:
   test rcx, rcx
   jz .scan_end
   mov al, byte ptr [rsi]
   cmp al, 0x21
   je .found
   cmp al, 0x5B
   je .bracket_open
   cmp al, 0x5D
   je .bracket_close
   jmp .next
.bracket_open:
   inc rbx
   jmp .next
.bracket_close:
   dec rbx
   test rbx, rbx
   jl .unbalanced
   jmp .next
.found:
   mov r8, 1
   jmp .scan_end
.next:
   add rsi, 1
   dec rcx
   jmp .scan_loop
.scan_end:
   test rbx, rbx
   jz .exit_zero
   jmp .unbalanced
.unbalanced:
   mov rax, 60
   mov rdi, 1
   syscall
   jmp .unbalanced
.exit_zero:
   mov rax, 60
   xor rdi, rdi
   syscall
