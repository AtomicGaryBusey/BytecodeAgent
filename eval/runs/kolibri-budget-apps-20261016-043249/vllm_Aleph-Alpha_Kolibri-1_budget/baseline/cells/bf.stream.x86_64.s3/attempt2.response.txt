

.global _start

.section .bss
inbuf:
   .space 65536

.section .text
_start:
   mov r12, offset inbuf
   xor r14, r14
.read_loop:
   mov rdi, 0
   lea rsi, [r12 + r14]
   mov rdx, 65536
   sub rdx, r14
   mov rax, 0
   syscall
   test rax, rax
   jz .read_done
   add r14, rax
   jmp .read_loop
.read_done:
   mov rcx, r14
   lea rsi, [r12]
   xor r8, r8
.scan_loop:
   test rcx, rcx
   jz .scan_end
   mov al, byte ptr [rsi]
   cmp al, 0x21
   je .found
   cmp al, 0x5B
   je .bracket_open
   cmp al, 0x5D
   je .bracket_close
   jmp .scan_next
.bracket_open:
   inc r8
   jmp .scan_next
.bracket_close:
   dec r8
   test r8, r8
   jl .unbalanced
   jmp .scan_next
.found:
   jmp .scan_end
.scan_next:
   inc rsi
   dec rcx
   jmp .scan_loop
.scan_end:
   test r8, r8
   jne .unbalanced
   mov rax, 60
   xor rdi, rdi
   syscall
   jmp .done
.unbalanced:
   mov rax, 60
   mov rdi, 1
   syscall
   jmp .done
.done:
   jmp .done
