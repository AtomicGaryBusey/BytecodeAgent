

.globl _start
.section .bss
inbuf:
   .space 65536
tape:
   .space 30000
.section .text
_start:
   mov rdi, 0
   lea rsi, [rip + inbuf]
   xor r10, r10
read_loop:
   mov rax, 0
   mov rdx, 65536
   syscall
   test rax, rax
   jz read_done
   add r10, rax
   jmp read_loop
read_done:
   lea rbx, [rip + inbuf]
   lea r12, [rbx + r10]
   lea rsi, [rbx]
find_exclamation:
   movzx eax, byte ptr [rsi]
   cmp eax, 33
   je found_exclamation
   cmp rsi, r12
   jge not_found
   inc rsi
   jmp find_exclamation
found_exclamation:
   sub rsi, rbx
   mov r11, rsi
   add rsi, 1
   mov r12, rsi
   sub r10, rsi
   jmp check_brackets
not_found:
   mov r11, r10
   mov r12, r10
   xor r10, r10
check_brackets:
   lea r13, [rbx + r11]
   lea r14, [rbx]
   xor ecx, ecx
scan_loop:
   cmp r14, r13
   jge scan_done
   movzx eax, byte ptr [r14]
   cmp eax, 91
   je scan_open
   cmp eax, 93
   je scan_close
   jmp scan_next
scan_open:
   inc ecx
   jmp scan_next
scan_close:
   dec ecx
   test ecx, ecx
   js unbalanced
   jmp scan_next
scan_next:
   inc r14
   jmp scan_loop
scan_done:
   test ecx, ecx
   jz balanced
   jmp unbalanced
balanced:
   xor eax, eax
   jmp exit_syscall
unbalanced:
   mov eax, 1
exit_syscall:
   mov rdi, eax
