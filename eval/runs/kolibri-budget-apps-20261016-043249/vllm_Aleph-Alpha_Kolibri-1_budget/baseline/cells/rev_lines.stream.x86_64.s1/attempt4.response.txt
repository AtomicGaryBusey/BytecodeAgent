

.section .bss
buffer: .space 65537
.section .rodata
newline: .byte 10
.section .text
.globl _start
_start:
    mov r12, offset buffer
    xor r13, r13
read_loop:
    mov rsi, r12
    add rsi, r13
    mov rdx, 65536
    sub rdx, r13
    xor rax, rax
    mov rdi, 0
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r14, r12
    mov r15, r12
    add r15, r13
line_loop:
    mov r16, r14
find_nl:
    cmp r14, r15
    jge no_nl
    movzx eax, byte ptr [r14]
    cmp al, 10
    je found_nl
    inc r14
    jmp find_nl
found_nl:
    mov r8, r14
    dec r8
    jmp output_rev
output_rev:
    cmp r8, r16
    jl after_output_rev
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    jmp output_rev
after_output_rev:
    mov rsi, offset newline
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r14
    jmp line_loop
no_nl:
    cmp r16, r15
    jge done
    mov r8, r15
    dec r8
    jmp output_final_rev
output_final_rev:
    cmp r8, r16
    jl after_final_newline
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    jmp output_final_rev
after_final_newline:
    mov rsi, offset newline
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    mov r14, r15
    jmp done
done:
    mov rdi, 0
    mov rax, 60
    syscall
