

.section .rodata
newline: .byte 10

.section .bss
buffer: .space 65536

.section .text
.global _start
_start:
    lea r12, [rip + buffer]
    xor r13, r13
read_loop:
    xor rax, rax
    mov rdi, 0
    lea rsi, [r12 + r13]
    mov rdx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r9, [r12 + r13]
    lea r14, [r12]
    lea r15, [r12]
scan_loop:
    cmp r14, r9
    jge scan_done
    movzx eax, byte ptr [r14]
    cmp eax, 10
    je found_newline
    inc r14
    jmp scan_loop
found_newline:
    cmp r15, r14
    je write_newline
    mov r8, r14
    sub r8, 1
output_loop:
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    cmp r8, r15
    jge output_loop
write_newline:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    lea r15, [r14 + 1]
    inc r14
    jmp scan_loop
scan_done:
    cmp r15, r14
    jge after_scan
    mov r8, r14
    sub r8, 1
last_output_loop:
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    cmp r8, r15
    jge last_output_loop
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
after_scan:
    mov rdi, 0
    xor rax, rax
    syscall
