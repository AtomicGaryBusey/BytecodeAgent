

.section .bss
buffer: .space 65536
.section .rodata
newline: .byte 10
.section .text
.global _start
_start:
    lea r12, [rip + buffer]
    xor r13, r13
read_loop:
    lea rsi, [r12 + r13]
    mov rdx, 65536
    sub rdx, r13
    xor rax, rax
    mov rdi, 0
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r15, r12
    add r15, r13
    mov r14, r12
line_loop:
    mov r10, r14
find_newline:
    cmp r10, r15
    jge no_newline
    movzx eax, byte ptr [r10]
    cmp eax, 10
    je newline_found
    inc r10
    jmp find_newline
newline_found:
    mov r8, r10
    dec r8
output_rev:
    cmp r8, r14
    jl after_output
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    jmp output_rev
after_output:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    mov r14, r10
    inc r14
    jmp line_loop
no_newline:
    cmp r14, r15
    jge exit_empty
    mov r8, r15
    dec r8
output_final_rev:
    cmp r8, r14
    jl after_final_newline
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    jmp output_final_rev
after_final_newline:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    jmp exit_empty
exit_empty:
    mov rdi, 0
    xor rax, rax
    syscall
