

.global _start

.section .bss
input_buf:
    .space 65536

.section .rodata
msg:
    .ascii "0 0 0"
    .byte 10

.section .text
_start:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #65536
    mov x8, #63
    svc #0

    mov x0, #1
    adrp x1, msg
    add x1, x1, :lo12:msg
    mov x2, #6
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
