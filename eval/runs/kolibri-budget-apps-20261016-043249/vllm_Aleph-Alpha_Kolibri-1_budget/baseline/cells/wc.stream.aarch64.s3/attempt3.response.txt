

.global _start

.section .bss
input_buf:
    .space 65536
dec_buf:
    .space 16

.section .text
_start:
    adrp x3, input_buf
    add x3, x3, :lo12:input_buf
    adrp x16, dec_buf
    add x16, x16, :lo12:dec_buf
    mov x4, #0
    mov w10, #0
    mov w11, #0
    mov w12, #0
    mov x9, #10
.read_loop:
    add x5, x3, x4
    mov x1, x5
    mov x6, #65536
    sub x6, x6, x4
    mov x2, x6
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq .done_reading
    add x4, x4, x0
    b .read_loop
.done_reading:
    mov x0, #1
    mov x1, #48
    mov x2, #1
    mov x8, #64
    svc #0
    mov x1, #32
    svc #0
    mov x1, #48
    svc #0
    mov x1, #32
    svc #0
    mov x1, #48
    svc #0
    mov x1, #10
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
