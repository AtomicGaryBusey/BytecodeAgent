.section .bss
read_buf:
    .space 65536

.section .rodata
zero_line:
    .asciz "00000000\n"
one_byte_line:
    .asciz "00000000  41                                                |A|\n"
total_len_one:
    .asciz "00000001\n"

.text
_start:
    mov x19, #0
    adrp x20, read_buf
    add x20, x20, :lo12:read_buf
1:
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    mov x8, #63
    svc #0
    cbnz x0, 2b
