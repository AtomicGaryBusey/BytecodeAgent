

.section .bss
input_buffer:
    .space 65536

.text
.global _start

_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x21, #0

read_loop:
    cmp x21, #65536
    b.ge read_done
    mov x2, #65536
    sub x2, x2, x21
    mov x0, #0
    mov x1, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    beq read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, #0

line_loop:
    cmp x22, x21
    b.ge line_done

    mov x23, x22
find_newline:
    cmp x23, x21
    b.ge no_newline
    ldrb w1, [x19, x23]
    cmp w1, #10
    beq found_newline
    add x23, x23, #1
    b find_newline

no_newline:
    mov x24, x22
    sub x25, x21, #1
reverse_loop:
    cmp x24, x25
    b.ge reverse_done
    ldrb w1, [x19, x24]
    ldrb w2, [x19, x25]
    strb w1, [x19, x25]
    strb w2, [x19, x24]
    add x24, x24, #1
    sub x25, x25, #1
    b reverse_loop
reverse_done:
    sub x26, x21, x22
    add x1, x19, x22
    mov x0, #1
    mov x8, #64
    mov x2, x26
    svc #0
    mov x0, #1
    mov x1, #10
    mov x2, #1
    mov x8, #64
    svc #0
    mov x22, x21
    b line_done

found_newline:
    sub x26, x23, x22
    mov x24, x22
    sub x25, x23, #1
reverse_loop2:
    cmp x24, x25
    b.ge reverse_done2
    ldrb w1, [x19, x24]
    ldrb w2, [x19, x25]
    strb w1, [x19, x25]
    strb w2, [x19, x24]
    add x24, x24, #1
    sub x25, x25, #1
    b reverse_loop2
reverse_done2:
    add x1, x19, x22
    mov x0, #1
    mov x8, #64
    mov x2, x26
    svc #0
    mov x0, #1
    mov x1, #10
    mov x2, #1
    mov x8, #64
    svc #0
    add x22, x23, #1
    b line_loop

line_done:
    mov x0, #0
    mov x8, #93
    svc #0
