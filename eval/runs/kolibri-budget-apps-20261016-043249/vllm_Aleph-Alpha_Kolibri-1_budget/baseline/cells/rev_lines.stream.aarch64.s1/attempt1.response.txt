

.text
.global _start

_start:
    adr x19, input_buffer
    adr x28, newline_byte
    mov x20, #0
    mov x21, #0
    mov x24, #0
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    movk x2, #1, lsl #16
    sub x2, x2, x20
    svc #0
    cmp x0, #0
    beq read_done
    add x21, x21, x0
    add x20, x20, x0
    b read_loop

read_done:
    cmp x21, #0
    beq exit
    b scan_loop

scan_loop:
    cmp x22, x21
    b.ge no_newline
    ldrb w1, [x19, x22]
    cmp w1, #10
    beq found_newline
    add x22, x22, #1
    b scan_loop

found_newline:
    sub x25, x22, x24
    mov x27, x24
    add x26, x24, x25
    sub x26, x26, #1
    b reverse_loop

reverse_loop:
    cmp x24, x26
    b.ge reverse_done
    ldrb w1, [x19, x24]
    ldrb w2, [x19, x26]
    strb w1, [x19, x26]
    strb w2, [x19, x24]
    add x24, x24, #1
    sub x26, x26, #1
    b reverse_loop

reverse_done:
    add x1, x19, x27
    mov x0, #1
    mov x2, x25
    svc #0
    mov x0, #1
    mov x1, x28
    mov x2, #1
    svc #0
    cmp x22, x21
    beq exit
    add x24, x22, #1
    mov x22, x24
    b scan_loop

no_newline:
    cmp x24, x21
    b.ge exit
    sub x25, x21, x24
    mov x27, x24
    sub x26, x21, #1
    b reverse_loop

exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline_byte:
    .byte 10

.section .bss
input_buffer:
    .space 65536
