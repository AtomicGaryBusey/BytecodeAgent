

.text
.global _start

_start:
    adr x19, input_buffer
    adr x28, newline_byte
    mov x20, #0
    mov x21, #0
    mov x22, #0
    movk x3, #1, lsl #16
    b read_loop

read_loop:
    cmp x20, x3
    b.ge read_done
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    sub x2, x3, x20
    svc #0
    cmp x0, #0
    beq read_done
    add x20, x20, x0
    b read_loop

read_done:
    cmp x20, #0
    beq exit_helper
    mov x21, #0
    mov x22, #0
    b scan_loop

scan_loop:
    cmp x21, x20
    b.ge final_line_check
    ldrb w1, [x19, x21]
    cmp w1, #10
    beq found_newline
    add x21, x21, #1
    b scan_loop

found_newline:
    sub x25, x21, x22
    add x27, x19, x22
    sub x26, x21, #1
    mov x29, #0
    b reverse_loop

final_line_check:
    cmp x22, x20
    b.ge exit_helper
    sub x25, x20, x22
    add x27, x19, x22
    sub x26, x20, #1
    mov x29, #1
    b reverse_loop

reverse_loop:
    cmp x22, x26
    b.ge reverse_done
    ldrb w1, [x19, x22]
    ldrb w2, [x19, x26]
    strb w1, [x19, x26]
    strb w2, [x19, x22]
    add x22, x22, #1
    sub x26, x26, #1
    b reverse_loop

reverse_done:
    mov x0, #1
    mov x1, x27
    mov x2, x25
    mov x8, #64
    svc #0
    mov x0, #1
    mov x1, x28
    mov x2, #1
    mov x8, #64
    svc #0
    cmp x29, #1
    b.eq exit_helper
    add x22, x21, #1
    mov x21, x22
    b scan_loop

exit_helper:
    mov x0, #0
    mov x8, #93
    svc #0

.rodata
