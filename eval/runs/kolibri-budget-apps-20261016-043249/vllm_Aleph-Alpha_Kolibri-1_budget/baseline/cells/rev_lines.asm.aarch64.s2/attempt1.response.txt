

.global _start

.section .rodata
newline:
    .byte 0x0A

.section .bss
inbuf:
    .space 65536

.section .text
_start:
    adrp x3, inbuf
    add x3, x3, :lo12:inbuf
    adrp x12, newline
    add x12, x12, :lo12:newline
    mov x4, #0
read_loop:
    add x1, x3, x4
    mov x0, #0
    mov x2, #1024
    mov x8, #63
    svc #0
    cbnz x0, read_continue
    mov x5, x3
    add x6, x3, x4
    cmp x4, #0
    b.ne process_loop
    mov x0, #0
    mov x8, #93
    svc #0
    b done_after_newline
read_continue:
    add x4, x4, x0
    b read_loop
process_loop:
    mov x7, x5
scan_loop:
    cmp x5, x6
    b.ge no_newline
    ldrb w0, [x5]
    cmp w0, #10
    b.ne scan_next
    sub x9, x5, x7
    mov x11, x7
    add x10, x7, x9
    sub x10, x10, #1
reverse_loop:
    cmp x11, x10
    b.ge reverse_done
    ldrb w0, [x11]
    ldrb w1, [x10]
    strb w1, [x11]
    strb w0, [x10]
    add x11, x11, #1
    sub x10, x10, #1
    b reverse_loop
reverse_done:
    mov x0, #1
    mov x1, x7
    mov x2, x9
    mov x8, #64
    svc #0
    mov x1, x12
    mov x2, #1
    mov x8, #64
    svc #0
    add x5, x5, #1
    cmp x5, x6
    b.ge done_after_newline
    mov x7, x5
    b scan_loop
scan_next:
    add x5, x5, #1
    b scan_loop
no_newline:
    sub x9, x6, x7
    mov x11, x7
    add x10, x7, x9
    sub x10, x10, #1
reverse_loop2:
    cmp x11, x10
    b.ge reverse_done2
    ldrb w0, [x11]
    ldrb w1, [x10]
    strb w1, [x11]
    strb w0, [x10]
    add x11, x11, #1
    sub x10, x10, #1
    b reverse_loop2
reverse_done2:
    mov x0, #1
    mov x1, x7
    mov x2, x9
    mov x8, #64
    svc #0
    mov x1, x12
    mov x2, #1
    mov x8, #64
    svc #0
done_after_newline:
    mov x0, #0
    mov x8, #93
    svc #0
