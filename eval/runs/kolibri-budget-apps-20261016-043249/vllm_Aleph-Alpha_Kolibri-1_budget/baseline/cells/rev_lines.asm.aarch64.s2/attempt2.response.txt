

.global _start

.section .rodata
newline:
    .byte 10
max_len:
    .quad 65536

.section .bss
inbuf:
    .space 65536

.section .text
_start:
    adrp x3, inbuf
    add x3, x3, :lo12:inbuf
    adrp x12, newline
    add x12, x12, :lo12:newline
    adrp x13, max_len
    add x13, x13, :lo12:max_len
    ldr x14, [x13]
    mov x4, #0
read_loop:
    cmp x4, x14
    b.ge process
    add x1, x3, x4
    mov x0, #0
    sub x2, x14, x4
    mov x8, #63
    svc #0
    cbnz x0, read_continue
    b process
read_continue:
    add x4, x4, x0
    b read_loop
process:
    cmp x4, #0
    b.le exit
    add x7, x3, x4
    mov x5, x3
    b scan_loop
scan_loop:
    mov x6, x5
    cmp x6, x7
    b.ge no_newline
    ldrb w0, [x6]
    cmp w0, #10
    b.ne scan_next
    sub x9, x6, x5
    cmp x9, #0
    b.le print_newline_only
    mov x10, x6
    sub x10, x10, #1
    mov x11, x9
print_bytes:
    cmp x11, #0
    b.le done_bytes
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0
    sub x10, x10, #1
    sub x11, x11, #1
    b print_bytes
done_bytes:
print_newline_only:
    mov x0, #1
    mov x1, x12
    mov x2, #1
    mov x8, #64
    svc #0
    add x5, x6, #1
    cmp x5, x7
    b.ge exit
    b scan_loop
scan_next:
    add x6, x6, #1
    b scan_loop
no_newline:
    sub x9, x7, x5
    cmp x9, #0
    b.le print_newline_only2
    mov x10, x7
    sub x10, x10, #1
    mov x11, x9
print_bytes2:
    cmp x11, #0
    b.le done_bytes2
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0
    sub x10, x10, #1
    sub x11, x11, #1
    b print_bytes2
done_bytes2:
print_newline_only2:
    mov x0, #1
    mov x1, x12
    mov x2, #1
    mov x8, #64
    svc #0
    b exit
exit:
    mov x0, #0
    mov x8, #93
    svc #0
