

.global _start

.section .rodata
newline:
    .byte 10

.section .bss
inbuf:
    .space 65536

.section .text
_start:
    adrp x3, inbuf
    add x3, x3, :lo12:inbuf
    mov x14, #65536
    mov x4, #0
read_loop:
    cmp x4, x14
    b.ge process
    add x1, x3, x4
    mov x0, #0
    sub x2, x14, x4
    mov x8, #63
    svc #0
    cbnz x0, read_continue
    b process
read_continue:
    add x4, x4, x0
    b read_loop
process:
    mov x5, #0
    mov x6, #0
    mov x7, #0
    adrp x10, newline
    add x10, x10, :lo12:newline
scan_loop:
    cmp x5, x4
    b.ge after_scan
    add x1, x3, x5
    ldrb w7, [x1]
    cmp w7, #10
    b.ne scan_next
    cmp x6, x5
    b.ge newline_only
    mov x7, x5
    sub x7, x7, #1
print_reversed:
    cmp x7, x6
    b.lt done_reversed
    add x1, x3, x7
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    sub x7, x7, #1
    b print_reversed
done_reversed:
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0
    add x6, x5, #1
    add x5, x5, #1
    b scan_loop
newline_only:
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0
    add x6, x5, #1
    add x5, x5, #1
    b scan_loop
scan_next:
    add x5, x5, #1
    b scan_loop
after_scan:
    cmp x6, x4
    b.ge exit_after_scan
    mov x7, x4
    sub x7, x7, #1
print_final_reversed:
    cmp x7, x6
    b.lt done_final_reversed
    add x1, x3, x7
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    sub x7, x7, #1
    b print_final_reversed
done_final_reversed:
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc #0
exit_after_scan:
    mov x0, #0
    mov x8, #93
    svc #0
