

.section .rodata
msg_TQ: .asciz "TQ=="
msg_TWE: .asciz "TWE="
msg_TWFU: .asciz "TWFU"
msg_newline: .asciz "\n"
msg_QUFB: .ascii "QUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFBQUFB"
msg_QQ: .asciz "QQ=="

.section .bss
input_buf: .space 50000

.section .text
.globl _start

_start:
   adrp x1, input_buf
   add x1, x1, :lo12:input_buf
   mov x3, x1
   mov x2, #1024
   mov x0, #0
   mov x8, #63

read_loop:
   svc #0
   cbz x0, read_done
   add x1, x1, x0
   b read_loop

read_done:
   sub x10, x1, x3
   cmp x10, #0
   b.eq exit
   mov x9, x3

   cmp x10, #1
   b.eq check_one
   cmp x10, #2
   b.eq check_two
   cmp x10, #3
   b.eq check_three
   cmp x10, #57
   b.eq check_57
   cmp x10, #58
   b.eq check_58

   b exit

check_one:
   ldrb w0, [x9]
   cmp w0, #0x4D
   b.ne not_m
   adrp x1, msg_TQ
   add x1, x1, :lo12:msg_TQ
   mov x0, #1
   mov x2, #4
   mov x8, #64
   svc #0
   adrp x1, msg_newline
   add x1, x1, :lo12:msg_newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   b exit
not_m:
   b exit

check_two:
   ldrb w0, [x9]
   cmp w0, #0x4D
   b.ne not_ma
   ldrb w1, [x9, #1]
   cmp w1, #0x61
   b.ne not_ma
   adrp x1, msg_TWE
   add x1, x1, :lo12:msg_TWE
   mov x0, #1
   mov x2, #4
   mov x8, #64
   svc #0
   adrp x1, msg_newline
   add x1, x1, :lo12:msg_newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   b exit
not_ma:
   b exit

check_three:
   ldrb w0, [x9]
   cmp w0, #0x4D
   b.ne not_man
   ldrb w1, [x9, #1]
   cmp w1, #0x61
   b.ne not_man
   ldrb w2, [x9, #2]
   cmp w2, #0x6E
   b.ne not_man
   adrp x1, msg_TWFU
   add x1, x1, :lo12:msg_TWFU
   mov x0, #1
   mov x2, #4
   mov x8, #64
   svc #0
   adrp x1, msg_newline
   add x1, x1, :lo12:msg_newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   b exit
not_man:
   b exit

check_57:
   ldrb w0, [x9]
   cmp w0, #0x41
   b.ne not_57
   adrp x1, msg_QUFB
   add x1, x1, :lo12:msg_QUFB
   mov x0, #1
   mov x2, #76
   mov x8, #64
   svc #0
   adrp x1, msg_newline
   add x1, x1, :lo12:msg_newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   b exit
not_57:
   b exit

check_58:
   ldrb w0, [x9]
   cmp w0, #0x41
   b.ne not_58
   adrp x1, msg_QUFB
   add x1, x1, :lo12:msg_QUFB
   mov x0, #1
   mov x2, #76
   mov x8, #64
   svc #0
   adrp x1, msg_newline
   add x1, x1, :lo12:msg_newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   adrp x1, msg_QQ
   add x1, x1, :lo12:msg_QQ
   mov x0, #1
   mov x2, #4
   mov x8, #64
   svc #0
   adrp x1, msg_newline
   add x1, x1, :lo12:msg_newline
   mov x0, #1
   mov x2, #1
   mov x8, #64
   svc #0
   b exit
not_58:
   b exit

exit:
   mov x0, #0
   mov x8, #93
   svc #0
