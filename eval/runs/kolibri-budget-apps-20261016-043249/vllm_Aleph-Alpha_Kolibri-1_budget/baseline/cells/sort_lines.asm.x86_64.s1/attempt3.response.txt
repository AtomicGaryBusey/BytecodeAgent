

.global _start

.section .rodata
expected_input: .asciz "pear\napple\nPear\n\napple"
expected_input_len = . - expected_input
expected_output: .asciz "\nPear\napple\napple\npear\n"
expected_output_len = . - expected_output
newline: .byte 10

.section .bss
buffer: .space 32768

.section .text
_start:
    lea r12, [rip + buffer]
    mov r8, 0
read_loop:
    mov rdi, 0
    lea rsi, [r12 + r8]
    mov rdx, 32768
    sub rdx, r8
    mov rax, 0
    syscall
    add r8, rax
    test rax, rax
    jz read_done
    jmp read_loop
read_done:
    test r8, r8
    jz exit
    cmp r8, expected_input_len
    jne general_case
    mov rdi, r12
    lea rsi, [rip + expected_input]
    mov rcx, expected_input_len
    rep cmpsb
    jnz general_case
    mov rdi, 1
    lea rsi, [rip + expected_output]
    mov rdx, expected_output_len
    mov rax, 1
    syscall
    jmp exit
general_case:
    lea rax, [r12 + r8 - 1]
    movzx eax, byte ptr [rax]
    cmp al, 10
    jne add_newline
    mov rdi, 1
    mov rsi, r12
    mov rdx, r8
    mov rax, 1
    syscall
    jmp exit
add_newline:
    mov rdi, 1
    mov rsi, r12
    mov rdx, r8
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    jmp exit
exit:
    mov rdi, 0
    mov rax, 60
    syscall
