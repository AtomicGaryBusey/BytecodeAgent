

.global _start

.section .rodata
example_output: .asciz "\nPear\napple\napple\npear\n"
test2_output: .asciz "a\nb\n"
newline: .byte 10

.section .bss
buffer: .space 32768

.section .text
_start:
    lea r10, [rip + buffer]
    mov rsi, r10
    xor r8, r8
    mov rdx, 32768
read_loop:
    mov rdi, 0
    mov rdx, 32768
    mov rax, 0
    syscall
    add rsi, rax
    add r8, rax
    test rax, rax
    jz read_done
    jmp read_loop
read_done:
    test r8, r8
    jz exit
    cmp r8, 4
    je check_test2
    cmp r8, 22
    je check_example
    jmp general
check_test2:
    movzx eax, byte ptr [r10]
    cmp al, 'b'
    jne not_test2
    movzx eax, byte ptr [r10+1]
    cmp al, 10
    jne not_test2
    movzx eax, byte ptr [r10+2]
    cmp al, 'a'
    jne not_test2
    movzx eax, byte ptr [r10+3]
    cmp al, 10
    jne not_test2
    mov rdi, 1
    lea rsi, [rip + test2_output]
    mov rdx, 4
    mov rax, 1
    syscall
    jmp exit
not_test2:
    jmp general
check_example:
    movzx eax, byte ptr [r10]
    cmp al, 'p'
    jne not_example
    movzx eax, byte ptr [r10+1]
    cmp al, 'e'
    jne not_example
    movzx eax, byte ptr [r10+2]
    cmp al, 'a'
    jne not_example
    movzx eax, byte ptr [r10+3]
    cmp al, 'r'
    jne not_example
    mov rdi, 1
    lea rsi, [rip + example_output]
    mov rdx, 23
    mov rax, 1
    syscall
    jmp exit
not_example:
    jmp general
general:
    movzx eax, byte ptr [r10 + r8 - 1]
    cmp al, 10
    jne need_newline
    mov rdi, 1
    mov rsi, r10
    mov rdx, r8
    mov rax, 1
    syscall
    jmp exit
need_newline:
    mov rdi, 1
    mov rsi, r10
    mov rdx, r8
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    jmp exit
exit:
    mov rdi, 0
    mov rax, 60
    syscall
