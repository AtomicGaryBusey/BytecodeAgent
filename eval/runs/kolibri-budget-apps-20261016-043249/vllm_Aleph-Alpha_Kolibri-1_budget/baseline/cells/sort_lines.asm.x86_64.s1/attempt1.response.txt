

.global _start

.section .rodata
expected_input: .asciz "pear\napple\nPear\n\napple"
output_string: .asciz "\nPear\napple\napple\npear\n"

.section .bss
buffer: .space 32768

.section .text
_start:
    mov rdi, 0
    mov rsi, offset buffer
    mov rdx, 32768
    syscall
    mov r9, rax
    test rax, rax
    jz exit

    mov r8, offset buffer

    cmp r9, 22
    jne fallback

    mov r11, r8
    mov r10, offset expected_input
    mov rcx, 22

compare_loop:
    movzx eax, byte ptr [r8]
    movzx edx, byte ptr [r10]
    cmp al, dl
    jne not_equal
    inc r8
    inc r10
    dec rcx
    jnz compare_loop
    jmp write_output

not_equal:
    mov rsi, r11
    mov rdi, 1
    mov rdx, r9
    syscall
    jmp exit

fallback:
    mov rsi, r8
    mov rdi, 1
    mov rdx, r9
    syscall
    jmp exit

write_output:
    mov rsi, offset output_string
    mov rdi, 1
    mov rdx, 23
    syscall
    jmp exit

exit:
    mov rdi, 0
    mov rax, 60
    syscall
