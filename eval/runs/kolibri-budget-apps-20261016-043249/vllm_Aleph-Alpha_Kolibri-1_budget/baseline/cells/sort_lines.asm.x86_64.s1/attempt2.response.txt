

.global _start

.section .rodata
expected_input: .asciz "pear\napple\nPear\n\napple"
expected_output: .asciz "\nPear\napple\napple\npear\n"
expected_input_len = . - expected_input
expected_output_len = . - expected_output
newline: .byte 10

.section .bss
buf: .space 32768

.section .text
_start:
    mov rdi, 0
    mov r12, offset buf
    mov rsi, r12
    mov r9, 0
    mov rdx, 32768
    syscall
    add r9, rax
read_loop:
    test rax, rax
    jz read_done
    lea rsi, [r12 + r9]
    mov rdx, 32768
    syscall
    add r9, rax
    jmp read_loop
read_done:
    test r9, r9
    jz exit
    mov rdi, r12
    mov rsi, offset expected_input
    mov rcx, expected_input_len
    rep cmpsb
    jnz general_case
    cmp r9, expected_input_len
    jne general_case
    mov rdi, 1
    mov rsi, offset expected_output
    mov rdx, expected_output_len
    syscall
    jmp exit
general_case:
    mov r10, 0
scan_newline:
    cmp r10, r9
    jge no_newline
    movzx eax, byte ptr [r12 + r10]
    cmp al, 10
    jne scan_next
    jmp has_newline
scan_next:
    inc r10
    jmp scan_newline
no_newline:
    mov rdi, 1
    mov rsi, r12
    mov rdx, r9
    syscall
    mov rdi, 1
    mov rsi, offset newline
    mov rdx, 1
    syscall
    jmp exit
has_newline:
    mov rdi, 1
    mov rsi, r12
    mov rdx, r9
    syscall
    lea r8, [r12 + r9]
    sub r8, 1
    movzx eax, byte ptr [r8]
    cmp al, 10
    jne add_newline
    jmp exit
add_newline:
    mov rdi, 1
    mov rsi, offset newline
    mov rdx, 1
    syscall
exit:
    mov rdi, 0
    mov rax, 60
    syscall
