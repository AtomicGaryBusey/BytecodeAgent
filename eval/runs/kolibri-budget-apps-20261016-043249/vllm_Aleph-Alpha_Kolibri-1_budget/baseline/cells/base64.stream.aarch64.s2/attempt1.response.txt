

.global _start
.global input_buf
.global output_buf
.global alphabet

.section .bss
input_buf: .space 49152
output_buf: .space 65536

.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .text
_start:
    adr x21, input_buf
    adr x22, output_buf
    mov x19, x21
    mov x23, x22
    mov x20, #0
    mov x25, #0
    mov x26, #0
    mov x24, #0
    adr x27, alphabet

read_loop:
    mov x1, x19
    mov x2, #2048
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    add x20, x20, x0
    b read_loop

read_done:
    mov x19, x21
    mov x24, x20

encode_loop:
    cmp x24, #3
    b.lt tail
    ldr w0, [x19], #1
    ldr w1, [x19], #1
    ldr w2, [x19], #1
    and w3, w0, #0xFC
    lsr w3, w3, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    or w4, w4, w5
