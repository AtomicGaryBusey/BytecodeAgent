

.globl _start
.section .rodata
alphabet: .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.section .bss
input_buf: .space 49152
output_buf: .space 68000
.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, alphabet
    add x20, x20, :lo12:alphabet
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x22, x19
    mov x23, #0
read_loop:
    mov x0, #0
    mov x1, x22
    mov x2, #2048
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x22, x22, x0
    add x23, x23, x0
    b read_loop
read_done:
    cmp x23, #0
    b.eq empty_input
    mov x26, x23
    mov x27, #3
    udiv x28, x26, x27
    umod x29, x26, x27
