.section .text

    add x19, x19, :lo12:input_buf
    adrp x20, alphabet
    add x20, x20, :lo12:alphabet

read_loop:

    mov x1, x22
    mov x2, #2048
    mov x8, #63

    cmp x0, #0
    b.eq read_done

    b.eq empty_input
    add x24, x19, x23
    mov x25, x19
    mov x26, x21
    mov x27, #0
    mov x28, #0

encode_loop:

    cmp x22, #3
    b.lt tail_loop
    ldr w0, [x25], #1
    ldr w1, [x25], #1
    ldr w2, [x25], #1
    and w3, w0, #0xFC
    lsr w3, w3, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5

    lsr w6, w2, #6
    orr w5, w5, w6

    ldr w0, [x20, w3, lsl #0]
