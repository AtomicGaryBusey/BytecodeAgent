

.global _start

.section .bss
tape: .space 30000
stdin_buf: .space 65536
bracket_stack: .space 8000
match: .space 262144
stdin_len: .space 8
bang_pos: .space 8
prog_len: .space 8
input_len: .space 8

.section .text
_start:
    lea rbx, [rip + stdin_buf]
    xor r14, r14
_read_loop:
    mov edi, 0
    lea rsi, [rbx + r14]
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jz _after_read
    add r14, rax
    jmp _read_loop
_after_read:
    mov qword ptr [rip + stdin_len], r14
    mov r12, qword ptr [rip + stdin_len]
    xor r13, r13
    jmp _find_bang
_found_bang:
    mov qword ptr [rip + bang_pos], r13
    mov qword ptr [rip + prog_len], r13
    sub r12, r13
    sub r12, 1
    mov qword ptr [rip + input_len], r12
    jmp _bracket_check
_no_bang:
    mov qword ptr [rip + prog_len], r12
    mov qword ptr [rip + input_len], 0
    mov qword ptr [rip + bang_pos], -1
    jmp _bracket_check
_bracket_check:
    lea r10, [rip + bracket_stack]
    lea r11, [rip + match]
    xor r12, r12
    mov r15, qword ptr [rip + prog_len]
    xor r13, r13
_bracket_loop:
    cmp r13, r15
    jge _bracket_done_check
    movzx eax, byte ptr [rbx + r13]
    cmp al, 91
    je _bracket_open_check
    cmp al, 93
    je _bracket_close_check
    inc r13
    jmp _bracket_loop
_bracket_open_check:
    mov qword ptr [r10 + r12], r13
    add r12, 8
    inc r13
    jmp _bracket_loop
_bracket_close_check:
    cmp r12, 0
    je _bracket_unbalanced_check
    sub r12, 8
    mov rax, qword ptr [r10 + r12]
    mov dword ptr [r11 + rax*4], r13
