

.global _start

.section .bss
input_buf:
    .space 65536

.section .rodata
newline:
    .byte 0x0A

.section .text
_start:
    lea r13, [rip + input_buf]
    xor r12, r12
    xor r10, r10
    xor rbx, rbx

read_loop:
    mov rdi, 0
    mov rdx, 65536
    sub rdx, rbx
    test rdx, rdx
    jle process_loop
    mov rsi, r13
    add rsi, rbx
    syscall
    test rax, rax
    jz process_loop
    add rbx, rax
    jmp read_loop

process_loop:
    cmp r10, rbx
    jge process_done
    movzx eax, byte ptr [r13 + r10]
    cmp al, 10
    je found_newline
    inc r10
    jmp process_loop

found_newline:
    mov r14, r10
    dec r14
write_line:
    cmp r14, r12
    jb write_line_done
    mov rsi, r13
    add rsi, r14
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r14
    jmp write_line
write_line_done:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
    inc r10
    mov r12, r10
    jmp process_loop

process_done:
    cmp r12, rbx
    jge process_exit
    mov r14, rbx
    dec r14
write_final_line:
    cmp r14, r12
    jb write_final_line_done
    mov rsi, r13
    add rsi, r14
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r14
    jmp write_final_line
write_final_line_done:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    syscall
process_exit:
    xor rdi, rdi
    mov rax, 60
    syscall
