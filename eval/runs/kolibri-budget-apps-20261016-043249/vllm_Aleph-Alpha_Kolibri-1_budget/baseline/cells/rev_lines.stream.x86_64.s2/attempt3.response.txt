

.globl _start

.section .bss
input_buf:
    .space 65536

.section .rodata
newline:
    .byte 0x0A

.section .text
.equ BUF_SIZE, 65536
.equ NEWLINE, 10
_start:
    lea r13, [rip + input_buf]
    xor r10, r10
read_loop:
    mov rdi, 0
    mov rsi, r13
    add rsi, r10
    mov rdx, BUF_SIZE
    sub rdx, r10
    test rdx, rdx
    jz read_done
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r10, rax
    jmp read_loop
read_done:
    lea r15, [r13 + r10]
    mov r12, r13
    mov r14, r13
process_loop:
    cmp r14, r15
    jge process_done
    movzx eax, byte ptr [r14]
    cmp al, NEWLINE
    je newline_found
    inc r14
    jmp process_loop
newline_found:
    mov r8, r14
    dec r8
print_line_rev:
    cmp r8, r12
    jb print_newline
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    jmp print_line_rev
print_newline:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r14
    mov r12, r14
    jmp process_loop
process_done:
    cmp r12, r15
    jge process_exit
    mov r8, r15
    dec r8
print_final_line_rev:
    cmp r8, r12
    jb print_final_newline
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    mov rax, 1
    syscall
    dec r8
    jmp print_final_line_rev
print_final_newline:
    lea rsi, [rip + newline]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
process_exit:
    xor rdi, rdi
    mov rax, 60
    syscall
