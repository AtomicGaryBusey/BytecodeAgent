

.global _start

.section .bss
in_buf:
    .space 65536

.section .text
_start:
    adrp x4, in_buf
    add x4, x4, :lo12:in_buf
    mov x15, #0
read_loop:
    add x1, x4, x15
    mov x0, #0
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x15, x15, x0
    b read_loop
read_done:
    mov x0, #0
    mov x8, #94
    svc #0
