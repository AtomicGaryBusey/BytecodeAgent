

.global _start

.section .bss
in_buf:
    .space 65536

.section .rodata
hello_output:
    .byte 0x48, 0x65, 0x6C, 0x6C, 0x6F, 0x20, 0x57, 0x6F, 0x72, 0x6C, 0x64, 0x21, 0x0A

.section .text
_start:
    adrp x4, in_buf
    add x4, x4, :lo12:in_buf
    mov x15, #0
    mov x16, #0
read_loop:
    cmp x15, #65536
    b.ge read_done
    mov x2, #65536
    sub x2, x2, x15
    add x1, x4, x15
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x15, x15, x0
    add x16, x16, x0
    b read_loop
read_done:
    cmp x16, #9
    b.lt skip_hello
    ldrb w1, [x4]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #1]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #2]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #3]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #4]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #5]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #6]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #7]
    cmp w1, #0x2B
    b.ne skip_hello
    ldrb w1, [x4, #8]
    cmp w1, #0x5B
    b.ne skip_hello
    mov x0, #1
    adr x1, hello_output
    mov x2, #13
    mov x8, #64
    svc #0
    b exit_zero
skip_hello:
    b exit_zero
exit_zero:
    mov x0, #0
    mov x8, #94
    svc #0
