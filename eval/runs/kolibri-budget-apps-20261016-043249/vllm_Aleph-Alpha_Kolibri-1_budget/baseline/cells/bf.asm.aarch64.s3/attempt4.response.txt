

.global _start

.section .bss
.balign 4
in_buf:
    .space 65536
.balign 4
cells:
    .space 30000
.balign 4
output_buf:
    .space 65536
.balign 4
match_table:
    .space 262144
.balign 4
bracket_stack:
    .space 4000

.section .text
_start:
    adrp x9, in_buf
    add x9, x9, :lo12:in_buf
    adrp x10, cells
    add x10, x10, :lo12:cells
    adrp x11, output_buf
    add x11, x11, :lo12:output_buf
    adrp x12, match_table
    add x12, x12, :lo12:match_table
    adrp x13, bracket_stack
    add x13, x13, :lo12:bracket_stack
    mov x14, #0
    mov x15, #0
read_loop:
    mov x0, #0
    add x1, x9, x15
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x15, x15, x0
    add x14, x14, x0
    b read_loop
read_done:
    mov x16, #0
find_exclamation:
    cmp x16, x14
    b.ge no_exclamation
    ldrb w17, [x9, x16]
    cmp w17, #0x21
    b.eq found_exclamation
    add x16, x16, #1
    b find_exclamation
no_exclamation:
    mov x15, x14
    mov x18, x14
    b after_find
found_exclamation:
    mov x15, x16
    add x18, x16, #1
after_find:
    mov x16, #0
    mov x20, #0
    mov x21, #0
pre_pass_loop:
    cmp x16, x15
    b.ge pre_pass_done
    ldrb w17, [x9, x16]
    cmp w17, #0x5B
    b.eq pre_pass_open
    cmp w17, #0x5D
    b.eq pre_pass_close
    add x16, x16, #1
    b pre_pass_loop
pre_pass_open:
    str w16, [x13, x20, lsl #2]
    add x20, x20, #1
    add x16, x16, #1
    b pre_pass_loop
pre_pass_close:
    cmp x20, #0
    b.eq pre_pass_close_unbalanced
    sub x20, x20, #1
    ldr w22, [x13, x20, lsl #2]
    str w16, [x12, x22, lsl #2]
    str w22, [x12, x16, lsl #2]
    add x16, x16, #1
    b pre_pass_loop
pre_pass_close_unbalanced:
    mov x21, #1
    mov x16, x15
    b pre_pass_done
pre_pass_done:
    cmp x20, #0
    b.ne exit_unbalanced
    cmp x21, #0
    b.ne exit_unbalanced
    mov x16, #0
    mov x17, #0
    mov x19, #0
    mov x20, #0
interp_loop:
    cmp x16, x15
    b.ge interp_done
    ldrb w17, [x9, x16]
    cmp w17, #0x3E
    b.eq interp_gt
    cmp w17, #0x3C
    b.eq interp_lt
    cmp w17, #0x2B
    b.eq interp_plus
    cmp w17, #0x2D
    b.eq interp_minus
    cmp w17, #0x2E
    b.eq interp_dot
    cmp w17, #0x2C
    b.eq interp_comm
    cmp w17, #0x5B
    b.eq interp_open
    cmp w17, #0x5D
    b.eq interp_close
    add x16, x16, #1
    b interp_loop
interp_gt:
    add x17, x17, #1
    add x16, x16, #1
    b interp_loop
interp_lt:
    sub x17, x17, #1
    add x16, x16, #1
    b interp_loop
interp_plus:
    ldr w18, [x10, x17]
    add w18, w18, #1
    and w18, w18, #0xFF
    str w18, [x10, x17]
    add x16, x16, #1
    b interp_loop
interp_minus:
    ldr w18, [x10, x17]
    sub w18, w18, #1
    and w18, w18, #0xFF
    str w18, [x10, x17]
    add x16, x16, #1
    b interp_loop
interp_dot:
    ldr w18, [x10, x17]
    str w18, [x11, x19]
    add x19, x19, #1
    add x16, x16, #1
    b interp_loop
interp_comm:
    cmp x18, x14
    b.ge interp_comm_zero
    ldr w18, [x9, x18]
    str w18, [x10, x17]
    add x18, x18, #1
    b interp_comm_next
interp_comm_zero:
    mov w18, #0
    str w18, [x10, x17]
interp_comm_next:
    add x16, x16, #1
    b interp_loop
interp_open:
    ldr w18, [x12, x16, lsl #2]
    ldr w19, [x10, x17]
    cmp w19, #0
    b.eq interp_open_zero
    add x16, x16, #1
    b interp_loop
interp_open_zero:
    add x16, w18, #1
    b interp_loop
interp_close:
    ldr w18, [x12, x16, lsl #2]
    ldr w19, [x10, x17]
    cmp w19, #0
    b.eq interp_close_zero
    add x16, w18, #1
    b interp_loop
interp_close_zero:
    add x16, x16, #1
    b interp_loop
interp_done:
    mov x0, #1
    mov x1, x11
    mov x2, x19
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #94
    svc #0
exit_unbalanced:
    mov x0, #1
    mov x8, #94
    svc #0
