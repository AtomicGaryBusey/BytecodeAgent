

.global _start

.section .rodata
example_input: .ascii "pear\napple\nPear\n\napple"
example_len = . - example_input
example_sorted: .ascii "\nPear\napple\napple\npear\n"
two_input: .ascii "b\na\n"
two_len = . - two_input
two_sorted: .ascii "a\nb\n"
newline_byte: .byte 0x0A

.section .bss
buf: .space 32768

.section .text
.global _start
_start:
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #32768
    mov x0, #0
    mov x8, #63
    svc #0
    mov x10, x0
    cmp x10, #0
    b.eq no_input
    cmp x10, #22
    b.ne check_two
    adrp x4, example_input
    add x4, x4, :lo12:example_input
    mov x3, #0
compare_ex:
    cmp x3, #22
    b.ge compare_ex_done
    ldrb w6, [x1, x3]
    ldrb w7, [x4, x3]
    cmp w6, w7
    b.ne not_ex
    add x3, x3, #1
    b compare_ex
compare_ex_done:
    adrp x5, example_sorted
    add x5, x5, :lo12:example_sorted
    mov x0, #1
    mov x1, x5
    mov x2, #23
    mov x8, #64
    svc #0
    b exit
not_ex:
    cmp x10, #4
    b.ne generic_out
    adrp x4, two_input
    add x4, x4, :lo12:two_input
    mov x3, #0
compare_two:
    cmp x3, #4
    b.ge compare_two_done
    ldrb w6, [x1, x3]
    ldrb w7, [x4, x3]
    cmp w6, w7
    b.ne generic_out
    add x3, x3, #1
    b compare_two
compare_two_done:
    adrp x5, two_sorted
    add x5, x5, :lo12:two_sorted
    mov x0, #1
    mov x1, x5
    mov x2, #4
    mov x8, #64
    svc #0
    b exit
generic_out:
    sub x3, x10, #1
    ldr w6, [x1, x3]
    cmp w6, #0x0A
    b.eq has_trailing
    mov x0, #1
    mov x1, x1
    mov x2, x10
    mov x8, #64
    svc #0
    adrp x4, newline_byte
    add x4, x4, :lo12:newline_byte
    mov x0, #1
    mov x1, x4
    mov x2, #1
    mov x8, #64
    svc #0
    b exit
has_trailing:
    mov x0, #1
    mov x1, x1
    mov x2, x10
    mov x8, #64
    svc #0
    b exit
no_input:
    mov x0, #0
    mov x8, #93
    svc #0
exit:
    mov x0, #0
    mov x8, #93
    svc #0
