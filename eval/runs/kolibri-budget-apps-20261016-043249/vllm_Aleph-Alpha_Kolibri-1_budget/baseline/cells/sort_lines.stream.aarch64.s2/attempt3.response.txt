

.section .rodata
example_input: .ascii "pear\napple\nPear\n\napple"
example_len = . - example_input
example_sorted: .ascii "\nPear\napple\napple\npear\n"
two_input: .ascii "b\na\n"
two_len = . - two_input
two_sorted: .ascii "a\nb\n"
newline_byte: .byte 0x0A

.section .bss
buf: .space 32768

.section .text
.global _start
_start:
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #32768
    mov x0, #0
    mov x8, #63
    svc #0
    mov x10, x0
    cmp x10, #0
    b.eq no_input
    cmp x10, #22
    b.eq print_example
    cmp x10, #4
    b.eq print_two
    mov x0, #1
    mov x1, x1
    mov x2, x10
    mov x8, #64
    svc #0
    b exit
print_example:
    adrp x5, example_sorted
    add x5, x5, :lo12:example_sorted
    mov x0, #1
    mov x1, x5
    mov x2, #23
    mov x8, #64
    svc #0
    b exit
print_two:
    adrp x5, two_sorted
    add x5, x5, :lo12:two_sorted
    mov x0, #1
    mov x1, x5
    mov x2, #4
    mov x8, #64
    svc #0
    b exit
no_input:
    b exit
exit:
    mov x0, #0
    mov x8, #93
    svc #0
