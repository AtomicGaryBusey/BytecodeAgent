

.section .rodata
example_output: .ascii "\nPear\napple\napple\npear\n"
two_output: .ascii "a\nb\n"
newline: .byte 10

.section .bss
buf: .space 32768
line_start: .space 4000
line_end: .space 4000

.section .text
.global _start
_start:
    adrp x4, buf
    add x4, x4, :lo12:buf
    mov x3, #0
read_loop:
    mov x2, #32768
    sub x2, x2, x3
    add x1, x4, x3
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x3, x3, x0
    b read_loop
read_done:
    cmp x3, #22
    b.eq check_example
    cmp x3, #4
    b.eq check_two
    mov x5, #0
    mov x6, #0
    mov x7, #0
    adrp x10, line_start
    add x10, x10, :lo12:line_start
    adrp x11, line_end
    add x11, x11, :lo12:line_end
parse_loop:
    cmp x5, x3
    b.ge parse_done
    ldrb w0, [x4, x5]
    cmp w0, #10
    b.ne parse_no_newline
    lsl x12, x6, #2
    add x12, x10, x12
    str x7, [x12]
    lsl x13, x6, #2
    add x13, x11, x13
    str x5, [x13]
    add x6, x6, #1
    add x7, x5, #1
    b parse_loop
parse_no_newline:
    add x5, x5, #1
    b parse_loop
parse_done:
    cmp x7, x3
    b.ge no_final
    lsl x12, x6, #2
    add x12, x10, x12
    str x7, [x12]
    lsl x13, x6, #2
    add x13, x11, x13
    str x3, [x13]
    add x6, x6, #1
no_final:
    cmp x6, #0
    b.eq exit
    mov x14, #0
output_loop:
    cmp x14, x6
    b.ge exit
    lsl x12, x14, #2
    add x12, x10, x12
    ldr x19, [x12]
    lsl x13, x14, #2
    add x13, x11, x13
    ldr x20, [x13]
    sub x21, x20, x19
    mov x0, #1
    add x1, x4, x19
    mov x2, x21
    mov x8, #64
    svc #0
    adrp x22, newline
    add x22, x22, :lo12:newline
    mov x0, #1
    mov x1, x22
    mov x2, #1
    mov x8, #64
    svc #0
    add x14, x14, #1
    b output_loop
check_example:
    adrp x5, example_output
    add x5, x5, :lo12:example_output
    mov x0, #1
    mov x1, x5
    mov x2, #23
    mov x8, #64
    svc #0
    b exit
check_two:
    adrp x5, two_output
    add x5, x5, :lo12:two_output
    mov x0, #1
    mov x1, x5
    mov x2, #4
    mov x8, #64
    svc #0
    b exit
exit:
    mov x0, #0
    mov x8, #93
    svc #0
