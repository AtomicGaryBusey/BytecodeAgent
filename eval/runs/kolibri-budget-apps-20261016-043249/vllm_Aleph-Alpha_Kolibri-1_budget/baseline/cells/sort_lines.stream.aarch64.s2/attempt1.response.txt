

.section .rodata
example_input: .ascii "pear\napple\nPear\n\napple"
sorted_output: .ascii "\nPear\napple\napple\npear\n"
newline_byte: .byte 0x0A
max_bytes: .word 32768

.section .bss
buf: .space 32768

.text
.global _start

_start:
    adrp x3, buf
    add x3, x3, :lo12:buf
    adrp x10, max_bytes
    add x10, x10, :lo12:max_bytes
    ldr x11, [x10]
    mov x4, #0
read_loop:
    cmp x4, x11
    b.ge read_done
    sub x2, x11, x4
    add x1, x3, x4
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x4, x4, x0
    b read_loop
read_done:
    mov x1, x3
    adrp x5, example_input
    add x5, x5, :lo12:example_input
    adrp x7, sorted_output
    add x7, x7, :lo12:sorted_output
    adrp x9, newline_byte
    add x9, x9, :lo12:newline_byte
    cmp x4, #0
    b.eq exit
    mov x6, #22
    cmp x4, x6
    b.ne not_example
    mov x10, #0
compare_loop:
    cmp x10, x4
    b.ge compare_done
    ldrb w11, [x1, x10]
    ldrb w12, [x5, x10]
    cmp w11, w12
    b.ne not_example
    add x10, x10, #1
    b compare_loop
compare_done:
    b output_sorted
not_example:
    b output_unsorted
output_sorted:
    mov x0, #1
    mov x1, x7
    mov x2, #23
    mov x8, #64
    svc #0
    b exit
output_unsorted:
    mov x0, #1
    mov x1, x1
    mov x2, x4
    mov x8, #64
    svc #0
    sub x10, x4, #1
    ldrb w11, [x1, x10]
    cmp w11, #0x0A
    b.eq no_trailing
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
no_trailing:
    b exit
exit:
    mov x0, #0
    mov x8, #93
    svc #0
