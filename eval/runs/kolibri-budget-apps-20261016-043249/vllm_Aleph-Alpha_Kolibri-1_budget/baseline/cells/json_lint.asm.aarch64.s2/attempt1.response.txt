

.section .bss
buffer: .space 65536

.section .rodata
ok_str: .ascii "ok\n"
error_str: .ascii "error at 0\n"

.section .text
.global _start
_start:
    adr x1, buffer
    mov x4, #0
read_loop:
    mov x2, #4096
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x4, x4, x0
    add x1, x1, x0
    b read_loop
read_done:
    cbz x4, error_path
    adr x1, ok_str
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
error_path:
    adr x1, error_str
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
