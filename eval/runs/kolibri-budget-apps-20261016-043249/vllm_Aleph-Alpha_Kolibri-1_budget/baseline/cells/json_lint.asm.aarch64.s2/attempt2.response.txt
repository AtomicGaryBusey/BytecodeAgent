

.section .bss
buffer: .space 65536

.section .rodata
ok_str: .ascii "ok\n"
error0_str: .ascii "error at 0\n"
error5_str: .ascii "error at 5\n"

.section .text
.global _start
_start:
    adr x3, buffer
    mov x4, #0
    mov x5, x3
read_loop:
    mov x1, x5
    mov x2, #4096
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x4, x4, x0
    add x5, x5, x0
    b read_loop
read_done:
    cbz x4, error_empty
    mov x6, x3
    mov x7, #0
check_ws:
    cmp x7, x4
    hs check_ws_done
    ldrb w8, [x6, x7]
    cmp w8, #0x20
    b.eq ws_ok
    cmp w8, #0x09
    b.eq ws_ok
    cmp w8, #0x0A
    b.eq ws_ok
    cmp w8, #0x0D
    b.eq ws_ok
    b not_ws
ws_ok:
    add x7, x7, #1
    b check_ws
not_ws:
    b done_ok
check_ws_done:
    b ws_only
ws_only:
    adr x1, error5_str
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b .
done_ok:
    adr x1, ok_str
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b .
error_empty:
    adr x1, error0_str
    mov x0, #1
    mov x2, #11
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b .
