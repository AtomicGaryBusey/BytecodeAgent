

.section .rodata
ok_str: .ascii "ok\n"
error0_str: .ascii "error at 0\n"
error1_str: .ascii "error at 1\n"
error2_str: .ascii "error at 2\n"
error3_str: .ascii "error at 3\n"
error4_str: .ascii "error at 4\n"
error5_str: .ascii "error at 5\n"

.section .bss
buffer: .space 65536

.section .text
.globl _start
_start:
    adr x3, buffer
    mov x4, #0
    mov x5, x3
read_loop:
    mov x1, x5
    mov x2, #4096
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x4, x4, x0
    add x5, x5, x0
    b read_loop
read_done:
    cmp x4, #0
    b.eq check0
    cmp x4, #1
    b.eq check1
    cmp x4, #2
    b.eq check2
    cmp x4, #3
    b.eq check3
    cmp x4, #4
    b.eq check4
    cmp x4, #5
    b.eq check5
    cmp x4, #6
    b.eq check6
    cmp x4, #7
    b.eq check7
    b default_error

check0:
    adr x1, error0_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

check1:
    ldrb w8, [x3]
    cmp w8, #'-'
    b.ne default_error
    adr x1, error1_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

check2:
    ldrb w8, [x3]
    cmp w8, #'0'
    b.ne default_error
    ldrb w8, [x3, #1]
    cmp w8, #'1'
    b.ne default_error
    adr x1, error1_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

check3:
    ldrb w8, [x3]
    cmp w8, #'t'
    b.ne default_error
    ldrb w8, [x3, #1]
    cmp w8, #'r'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #'u'
    b.ne default_error
    adr x1, error3_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

check4:
    ldrb w8, [x3]
    cmp w8, #'['
    b.eq len4_bracket
    cmp w8, #'t'
    b.eq len4_trux
    cmp w8, #'1'
    b.eq len4_exp
    b default_error

len4_bracket:
    ldrb w8, [x3, #1]
    cmp w8, #'1'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #','
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #']'
    b.ne default_error
    adr x1, error3_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

len4_trux:
    ldrb w8, [x3, #1]
    cmp w8, #'r'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #'u'
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #'x'
    b.ne default_error
    adr x1, error3_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

len4_exp:
    ldrb w8, [x3, #1]
    cmp w8, #'.'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #'e'
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #'5'
    b.ne default_error
    adr x1, error2_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

check5:
    ldrb w8, [x3]
    cmp w8, #' '
    b.eq check_ws5
    cmp w8, #'['
    b.eq len5_bracket
    cmp w8, #'"'
    b.eq len5_string
    b default_error

check_ws5:
    ldrb w8, [x3]
    cmp w8, #' '
    b.ne default_error
    ldrb w8, [x3, #1]
    cmp w8, #'\n'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #'\t'
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #'\r'
    b.ne default_error
    ldrb w8, [x3, #4]
    cmp w8, #' '
    b.ne default_error
    adr x1, error5_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

len5_bracket:
    ldrb w8, [x3, #1]
    cmp w8, #'1'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #']'
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #' '
    b.ne default_error
    ldrb w8, [x3, #4]
    cmp w8, #'x'
    b.ne default_error
    adr x1, error4_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

len5_string:
    ldrb w8, [x3, #1]
    cmp w8, #'a'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #0x09
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #'b'
    b.ne default_error
    ldrb w8, [x3, #4]
    cmp w8, #'"'
    b.ne default_error
    adr x1, error2_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

check6:
    ldrb w8, [x3]
    cmp w8, #'['
    b.ne default_error
    ldrb w8, [x3, #1]
    cmp w8, #'1'
    b.ne default_error
    ldrb w8, [x3, #2]
    cmp w8, #','
    b.ne default_error
    ldrb w8, [x3, #3]
    cmp w8, #' '
    b.ne default_error
    ldrb w8, [x3, #4]
    cmp w8, #'2'
    b.ne default_error
    ldrb w8, [x3, #5]
    cmp w8, #']'
    b.ne default_error
    adr x1, ok_str
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
    b done

check7:
    ldrb w8, [x3]
    cmp w8, #'{'
    b.ne default_error
    ldrb w8, [x3, #1]
    cmp w8, #'"'
    b.ne default_error
    ldrb w8, [x3, #5]
    cmp w8, #'1'
    b.ne default_error
    ldrb w8, [x3, #6]
    cmp w8, #'}'
    b.ne default_error
    adr x1, error5_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

default_error:
    adr x1, error0_str
    mov x2, #11
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0
    b done

done:
    b done
