

section .text
.globl _start

_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    adrp x9, alphabet
    add x9, x9, :lo12:alphabet

    mov x23, x22
    mov x25, #0
    mov x20, x21
    mov x19, #0

read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, #4096
    movz x8, #63
    svc #0
    cmp x0, #0
    ble done_reading
    add x19, x19, x0
    add x20, x20, x0
    b read_loop

done_reading:
    cmp x19, #0
    beq empty_input

    mov x20, x21
    mov x24, x19

process_loop:
    cmp x24, #3
    blt process_remainder

    ldr w0, [x20]
    ldr w1, [x20, #1]
    ldr w2, [x20, #2]

    lsr w3, w0, #2
    and w3, w3, #0x3F
    add x7, x9, x3
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_char0
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0
after_char0:
    and w3, w0, #0x03
    lsl w3, w3, #4
    lsr w4, w1, #4
    and w4, w4, #0x0F
    orr w3, w3, w4
    add x7, x9, x3
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_char1
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0
after_char1:
    and w4, w1, #0x0F
    lsl w4, w4, #2
    lsr w5, w2, #6
    and w5, w5, #0x03
    orr w4, w4, w5
    add x7, x9, x4
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_char2
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0
after_char2:
    and w5, w2, #0x3F
    add x7, x9, x5
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_char3
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0
after_char3:
    sub x24, x24, #3
    b process_loop

process_remainder:
    cmp x24, #2
    beq process_two_bytes
    cmp x24, #1
    beq process_one_byte
    b done_processing

process_two_bytes:
    ldr w0, [x20]
    ldr w1, [x20, #1]

    lsr w3, w0, #2
    and w3, w3, #0x3F
    add x7, x9, x3
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_two_char0
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0
after_two_char0:
    and w3, w0, #0x03
    lsl w3, w3, #4
    lsr w4, w1, #4
    and w4, w4, #0x0F
    orr w3, w3, w4
    add x7, x9, x3
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_two_char1
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0

    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_two_eq

    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0
after_two_eq:
    mov x24, #0
    b done_processing

process_one_byte:
    ldr w0, [x20]

    lsr w3, w0, #2
    and w3, w3, #0x3F
    add x7, x9, x3
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1
    cmp x25, #76
    bne after_one_char0
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
    mov x25, #0

    and w3, w0, #0x03
    lsl w3, w3, #4
    add x7, x9, x3
    ldrb w8, [x7]
    strb w8, [x23]
    add x23, x23, #1
    add x25, x25, #1

    bne after_one_char1

    mov x0, #0
    movz x8, #93
    svc #0

    cmp x25, #0
    beq write_output
    mov w8, #10
    strb w8, [x23]
    add x23, x23, #1
