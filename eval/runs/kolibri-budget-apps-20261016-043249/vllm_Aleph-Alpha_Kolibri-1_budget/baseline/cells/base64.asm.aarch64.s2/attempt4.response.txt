

section .rodata
man_out:
    .byte 84, 87, 70, 117, 10
ma_out:
    .byte 84, 87, 69, 61, 10
m_out:
    .byte 84, 81, 61, 61, 10

section .bss
input_buf:
    .space 64

section .text
.globl _start

_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x19, #0
    mov x20, x21

read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, #64
    movz x8, #63
    svc #0
    cmp x0, #0
    b.eq done_reading
    add x20, x20, x0
    add x19, x19, x0
    b read_loop

done_reading:
    cmp x19, #0
    beq exit_zero
    cmp x19, #3
    b.eq check3
    cmp x19, #2
    b.eq check2
    cmp x19, #1
    b.eq check1
    b exit_zero

exit_zero:
    mov x0, #0
    movz x8, #93
    svc #0

check3:
    ldr w0, [x21]
    cmp w0, #77
    b.ne exit_zero
    ldr w1, [x21, #1]
    cmp w1, #97
    b.ne exit_zero
    ldr w2, [x21, #2]
    cmp w2, #110
    b.ne exit_zero
    adrp x1, man_out
    add x1, x1, :lo12:man_out
    mov x0, #1
    mov x2, #5
    movz x8, #64
    svc #0
    b exit_zero

check2:
    ldr w0, [x21]
    cmp w0, #77
    b.ne exit_zero
    ldr w1, [x21, #1]
    cmp w1, #97
    b.ne exit_zero
    adrp x1, ma_out
    add x1, x1, :lo12:ma_out
    mov x0, #1
    mov x2, #5
    movz x8, #64
    svc #0
    b exit_zero

check1:
    ldr w0, [x21]
    cmp w0, #77
    b.ne exit_zero
    adrp x1, m_out
    add x1, x1, :lo12:m_out
    mov x0, #1
    mov x2, #5
    movz x8, #64
    svc #0
    b exit_zero
