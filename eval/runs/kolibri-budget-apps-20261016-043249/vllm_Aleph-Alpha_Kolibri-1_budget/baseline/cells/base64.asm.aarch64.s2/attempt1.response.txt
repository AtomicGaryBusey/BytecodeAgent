

section .text
global _start

_start:
adrp x9, buf
add x9, x9, :lo12:buf
b read_loop

read_loop:
mov x0, #0
mov x1, x9
mov x2, #1
movz x8, #63
svc #0
cbnz x0, write_byte
mov x0, #0
movz x8, #93
svc #0

write_byte:
mov x0, #1
mov x1, x9
mov x2, #1
movz x8, #64
svc #0
b read_loop

section .data
buf:
.byte 0
