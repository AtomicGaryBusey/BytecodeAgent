

.section .bss
input_buf:
.space 65536
output_buf:
.space 65537

.section .text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x21, output_buf
    add x21, x21, :lo12:output_buf
    mov x20, #0
    mov x22, #0
    mov x23, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1, lsl #16
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop
read_done:
scan_loop:
    cmp x23, x20
    b.ge scan_done
    mov x24, x23
find_newline:
    cmp x24, x20
    b.ge no_newline
    ldrb w26, [x19, x24]
    cmp w26, #10
    b.ne next_char
    sub x25, x24, #1
    add x28, x24, #1
    b print_line
next_char:
    add x24, x24, #1
    b find_newline
no_newline:
    sub x25, x20, #1
    mov x28, x20
print_line:
    mov x27, x25
print_rev:
    cmp x27, x23
    b.lt print_newline
    ldrb w26, [x19, x27]
    strb w26, [x21, x22]
    add x22, x22, #1
    subs x27, x27, #1
    b print_rev
print_newline:
    mov w26, #10
    strb w26, [x21, x22]
    add x22, x22, #1
    mov x23, x28
    b scan_loop
scan_done:
    mov x0, #1
    mov x1, x21
    mov x2, x22
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
