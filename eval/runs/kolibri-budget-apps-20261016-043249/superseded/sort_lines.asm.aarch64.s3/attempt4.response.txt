

.global _start
.section .rodata
newline:
    .byte 10
.section .bss
input_buf:
    .space 32768
.section .text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, #32768
read_loop:
    add x1, x19, x20
    mov x0, #0
    mov x2, x21
    mov x8, #63
    svc #0
    cbnz x0, read_loop2
    b read_done
read_loop2:
    add x20, x20, x0
    subs x21, x21, x0
    cmp x21, #0
    b.le read_done
    b read_loop
read_done:
    cmp x20, #0
    b.eq exit_now
    subs x22, x20, #1
    ldr w23, [x19, x22]
    cmp w23, #10
    b.eq write_all
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov w8, #64
    svc #0
    adrp x3, newline
    add x3, x3, :lo12:newline
    mov x0, #1
    mov x1, x3
    mov x2, #1
    mov w8, #64
    svc #0
    b exit_now
write_all:
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov w8, #64
    svc #0
exit_now:
    mov w8, #93
    mov w0, #0
    svc #0
