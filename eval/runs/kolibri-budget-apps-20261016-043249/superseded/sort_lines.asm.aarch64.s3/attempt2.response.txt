

.global _start
.section .bss
buffer:
  .space 32768
.section .text
_start:
  adr x19, buffer
  movz x3, #0x800
  lsl x3, x3, #4
  mov x20, #0
read_loop:
  add x1, x19, x20
  mov x0, #0
  mov x2, x3
  mov x8, #63
  svc #0
  cbz x0, read_done
  add x20, x20, x0
  b read_loop
read_done:
  mov x8, #64
  mov x0, #1
  mov x1, x19
  mov x2, x20
  svc #0
  mov x8, #93
  mov x0, #0
  svc #0
