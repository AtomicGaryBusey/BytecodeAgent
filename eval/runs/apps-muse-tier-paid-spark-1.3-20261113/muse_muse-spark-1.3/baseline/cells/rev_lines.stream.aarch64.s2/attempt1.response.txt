.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ NL, 10
.globl _start
_start:
  adrp x19, in_buf
  add x19, x19, :lo12:in_buf
  adrp x21, out_buf
  add x21, x21, :lo12:out_buf
  mov x20, #0
read_loop:
  movz x2, #1, lsl #16
  sub x2, x2, x20
  cbz x2, read_done
  add x1, x19, x20
  mov x0, #FD_STDIN
  mov x8, #SYS_READ
  svc #0
  cbz x0, read_done
  add x20, x20, x0
  b read_loop
read_done:
  mov x22, #0
  mov x23, #0
outer_loop:
  cmp x23, x20
  b.hs write_start
  mov x24, x23
scan_loop:
  cmp x24, x20
  b.hs scan_done
  add x0, x19, x24
  ldrb w0, [x0]
  cmp w0, #NL
  b.eq scan_done
  add x24, x24, #1
  b scan_loop
scan_done:
  sub x25, x24, x23
  cbz x25, copy_done
  mov x26, #0
copy_loop:
  cmp x26, x25
  b.hs copy_done
  sub x0, x25, #1
  sub x0, x0, x26
  add x0, x0, x23
  add x0, x19, x0
  ldrb w0, [x0]
  add x1, x21, x22
  add x1, x1, x26
  strb w0, [x1]
  add x26, x26, #1
  b copy_loop
copy_done:
  add x22, x22, x25
  add x0, x21, x22
  mov w1, #NL
  strb w1, [x0]
  add x22, x22, #1
  cmp x24, x20
  b.hs write_start
  add x23, x24, #1
  b outer_loop
write_start:
  mov x23, #0
write_loop:
  cmp x23, x22
  b.hs exit_ok
  sub x2, x22, x23
  add x1, x21, x23
  mov x0, #FD_STDOUT
  mov x8, #SYS_WRITE
  svc #0
  add x23, x23, x0
  b write_loop
exit_ok:
  mov x0, #0
  mov x8, #SYS_EXIT
  svc #0
  b exit_ok
.bss
.balign 16
in_buf:
.zero 65536
.balign 16
out_buf:
.zero 65537
