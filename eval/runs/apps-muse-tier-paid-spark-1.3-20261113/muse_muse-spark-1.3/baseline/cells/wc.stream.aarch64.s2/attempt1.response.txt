.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ BUF_SIZE, 4096
.text
.globl _start
.balign 4
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x23, buf
    add x23, x23, :lo12:buf
read_loop:
    mov x0, #FD_STDIN
    mov x1, x23
    mov x2, #BUF_SIZE
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    tbnz x0, #63, read_done
    add x21, x21, x0
    mov x24, x23
    add x25, x23, x0
byte_loop:
    cmp x24, x25
    b.hs read_loop
    ldrb w0, [x24], #1
    cmp w0, #10
    b.ne check_ws
    add x19, x19, #1
check_ws:
    cmp w0, #32
    b.eq is_space
    cmp w0, #9
    b.lo is_word
    cmp w0, #13
    b.ls is_space
    b is_word
is_space:
    mov x22, #0
    b byte_loop
is_word:
    cbnz x22, byte_loop
    mov x22, #1
    add x20, x20, #1
    b byte_loop
read_done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    adrp x25, digits
    add x25, x25, :lo12:digits
    mov x0, x19
    cbnz x0, conv1_nz
    mov w1, #48
    strb w1, [x24], #1
    b conv1_done
conv1_nz:
    mov x1, #0
    mov x2, #10
conv1_loop:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add x4, x4, #48
    strb w4, [x25, x1]
    add x1, x1, #1
    mov x0, x3
    cbnz x0, conv1_loop
conv1_rev:
    sub x1, x1, #1
    ldrb w4, [x25, x1]
    strb w4, [x24], #1
    cbnz x1, conv1_rev
conv1_done:
    mov w1, #32
    strb w1, [x24], #1
    mov x0, x20
    cbnz x0, conv2_nz
    mov w1, #48
    strb w1, [x24], #1
    b conv2_done
conv2_nz:
    mov x1, #0
    mov x2, #10
conv2_loop:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add x4, x4, #48
    strb w4, [x25, x1]
    add x1, x1, #1
    mov x0, x3
    cbnz x0, conv2_loop
conv2_rev:
    sub x1, x1, #1
    ldrb w4, [x25, x1]
    strb w4, [x24], #1
    cbnz x1, conv2_rev
conv2_done:
    mov w1, #32
    strb w1, [x24], #1
    mov x0, x21
    cbnz x0, conv3_nz
    mov w1, #48
    strb w1, [x24], #1
    b conv3_done
conv3_nz:
    mov x1, #0
    mov x2, #10
conv3_loop:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add x4, x4, #48
    strb w4, [x25, x1]
    add x1, x1, #1
    mov x0, x3
    cbnz x0, conv3_loop
conv3_rev:
    sub x1, x1, #1
    ldrb w4, [x25, x1]
    strb w4, [x24], #1
    cbnz x1, conv3_rev
conv3_done:
    mov w1, #10
    strb w1, [x24], #1
    mov x1, x23
    sub x2, x24, x23
write_loop:
    mov x0, #FD_STDOUT
    mov x8, #SYS_WRITE
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_loop
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
hang:
    b hang
.bss
.balign 16
buf:
    .zero 4096
outbuf:
    .zero 64
digits:
    .zero 32
