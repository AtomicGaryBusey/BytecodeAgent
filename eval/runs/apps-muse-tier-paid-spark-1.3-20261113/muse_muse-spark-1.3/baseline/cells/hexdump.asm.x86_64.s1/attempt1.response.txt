.text
.globl _start
_start:
lea r14, [rip + buffer]
lea r10, [rip + hexdigits]
xor r12d, r12d
read_loop:
cmp r12, 65536
jae read_done
mov eax, 0
mov edi, 0
mov rsi, r14
add rsi, r12
mov rdx, 65536
sub rdx, r12
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_done:
xor r15d, r15d
row_loop:
cmp r15, r12
jae final_line
mov rbx, r12
sub rbx, r15
cmp rbx, 16
jbe have_n
mov ebx, 16
have_n:
mov rsi, r14
add rsi, r15
lea rdi, [rip + outbuf]
mov eax, r15d
mov ecx, 28
off_loop:
mov r8d, eax
shr r8d, cl
and r8d, 15
movzx edx, byte ptr [r10 + r8]
mov byte ptr [rdi], dl
inc rdi
sub ecx, 4
jns off_loop
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
xor ecx, ecx
hex_loop:
cmp ecx, ebx
jae blank_slot
movzx eax, byte ptr [rsi + rcx]
mov r8d, eax
shr r8d, 4
movzx edx, byte ptr [r10 + r8]
mov byte ptr [rdi], dl
inc rdi
and eax, 15
movzx edx, byte ptr [r10 + rax]
mov byte ptr [rdi], dl
inc rdi
mov byte ptr [rdi], 32
inc rdi
jmp hex_after
blank_slot:
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 32
inc rdi
hex_after:
cmp ecx, 7
jne hex_next
mov byte ptr [rdi], 32
inc rdi
hex_next:
inc ecx
cmp ecx, 16
jb hex_loop
mov byte ptr [rdi], 32
inc rdi
mov byte ptr [rdi], 124
inc rdi
xor ecx, ecx
ascii_loop:
cmp ecx, ebx
jae ascii_done
movzx eax, byte ptr [rsi + rcx]
cmp al, 32
jb ascii_dot
cmp al, 126
ja ascii_dot
jmp ascii_store
ascii_dot:
mov al, 46
ascii_store:
mov byte ptr [rdi], al
inc rdi
inc ecx
jmp ascii_loop
ascii_done:
mov byte ptr [rdi], 124
inc rdi
mov byte ptr [rdi], 10
inc rdi
lea r8, [rip + outbuf]
mov rdx, rdi
sub rdx, r8
mov rsi, r8
mov eax, 1
mov edi, 1
syscall
add r15, rbx
jmp row_loop
final_line:
lea rdi, [rip + outbuf]
mov eax, r12d
mov ecx, 28
final_loop:
mov r8d, eax
shr r8d, cl
and r8d, 15
movzx edx, byte ptr [r10 + r8]
mov byte ptr [rdi], dl
inc rdi
sub ecx, 4
jns final_loop
mov byte ptr [rdi], 10
inc rdi
lea rsi, [rip + outbuf]
mov edx, 9
mov eax, 1
mov edi, 1
syscall
mov eax, 60
xor edi, edi
syscall
