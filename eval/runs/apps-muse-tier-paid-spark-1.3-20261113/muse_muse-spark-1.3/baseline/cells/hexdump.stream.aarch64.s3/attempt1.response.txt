.text
.globl _start
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x23, line_buf
    add x23, x23, :lo12:line_buf
    adrp x24, hexdigits
    add x24, x24, :lo12:hexdigits
    mov x19, #0
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop
read_done:
    mov x21, #0
row_loop:
    cmp x21, x19
    b.hs final_line
    sub x22, x19, x21
    cmp x22, #16
    b.ls have_n
    mov x22, #16
have_n:
    lsr x9, x21, #28
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23]
    lsr x9, x21, #24
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #1]
    lsr x9, x21, #20
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #2]
    lsr x9, x21, #16
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #3]
    lsr x9, x21, #12
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #4]
    lsr x9, x21, #8
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #5]
    lsr x9, x21, #4
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #6]
    and x9, x21, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #7]
    mov w10, #32
    strb w10, [x23, #8]
    strb w10, [x23, #9]
    mov x26, #0
hex_loop:
    cmp x26, #16
    b.hs hex_done
    lsl x9, x26, #1
    add x9, x9, x26
    add x9, x9, #10
    cmp x26, #8
    b.lo hex_no_extra
    add x9, x9, #1
hex_no_extra:
    add x9, x9, x23
    cmp x26, x22
    b.hs hex_blank
    add x11, x20, x21
    add x11, x11, x26
    ldrb w10, [x11]
    lsr w11, w10, #4
    ldrb w12, [x24, x11]
    strb w12, [x9]
    and x11, x10, #15
    ldrb w12, [x24, x11]
    strb w12, [x9, #1]
    mov w12, #32
    strb w12, [x9, #2]
    b hex_next
hex_blank:
    mov w12, #32
    strb w12, [x9]
    strb w12, [x9, #1]
    strb w12, [x9, #2]
hex_next:
    add x26, x26, #1
    b hex_loop
hex_done:
    mov w10, #32
    strb w10, [x23, #34]
    strb w10, [x23, #59]
    mov w10, #124
    strb w10, [x23, #60]
    mov x26, #0
ascii_loop:
    cmp x26, x22
    b.hs ascii_done
    add x9, x20, x21
    add x9, x9, x26
    ldrb w10, [x9]
    cmp w10, #32
    b.lo is_dot
    cmp w10, #126
    b.hi is_dot
    b store_ascii
is_dot:
    mov w10, #46
store_ascii:
    add x9, x23, #61
    add x9, x9, x26
    strb w10, [x9]
    add x26, x26, #1
    b ascii_loop
ascii_done:
    add x9, x23, #61
    add x9, x9, x22
    mov w10, #124
    strb w10, [x9]
    mov w10, #10
    strb w10, [x9, #1]
    add x2, x22, #63
    mov x0, #1
    mov x1, x23
    mov x8, #SYS_write
    svc #0
    add x21, x21, x22
    b row_loop
final_line:
    lsr x9, x19, #28
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23]
    lsr x9, x19, #24
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #1]
    lsr x9, x19, #20
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #2]
    lsr x9, x19, #16
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #3]
    lsr x9, x19, #12
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #4]
    lsr x9, x19, #8
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #5]
    lsr x9, x19, #4
    and x9, x9, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #6]
    and x9, x19, #15
    ldrb w10, [x24, x9]
    strb w10, [x23, #7]
    mov w10, #10
    strb w10, [x23, #8]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
.section .rodata
hexdigits:
    .ascii "0123456789abcdef"
.bss
    .balign 16
input_buf:
    .zero 65536
line_buf:
    .zero 80
