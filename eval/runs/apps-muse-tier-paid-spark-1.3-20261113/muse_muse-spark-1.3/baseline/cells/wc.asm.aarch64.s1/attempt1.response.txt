.text
.globl _start
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
_start:
	mov x19, #0
	mov x20, #0
	mov x21, #0
	mov x22, #0
read_loop:
	mov x0, #0
	adrp x1, buf
	add x1, x1, :lo12:buf
	mov x2, #8192
	mov x8, #SYS_read
	svc #0
	cbz x0, print_counts
	adrp x1, buf
	add x1, x1, :lo12:buf
	add x4, x1, x0
	mov x3, x1
byte_loop:
	ldrb w5, [x3], #1
	add x21, x21, #1
	cmp w5, #10
	b.ne check_ws
	add x19, x19, #1
	mov x22, #0
	b byte_next
check_ws:
	cmp w5, #32
	b.eq is_ws
	cmp w5, #9
	b.lo is_word
	cmp w5, #13
	b.ls is_ws
	b is_word
is_ws:
	mov x22, #0
	b byte_next
is_word:
	cbnz x22, byte_next
	add x20, x20, #1
	mov x22, #1
byte_next:
	cmp x3, x4
	b.ne byte_loop
	b read_loop
print_counts:
	adrp x1, outbuf
	add x1, x1, :lo12:outbuf
	mov x3, x19
	bl write_num
	mov w2, #32
	strb w2, [x1], #1
	mov x3, x20
	bl write_num
	mov w2, #32
	strb w2, [x1], #1
	mov x3, x21
	bl write_num
	mov w2, #10
	strb w2, [x1], #1
	adrp x3, outbuf
	add x3, x3, :lo12:outbuf
	sub x2, x1, x3
	mov x0, #1
	mov x1, x3
	mov x8, #SYS_write
	svc #0
	mov x0, #0
	mov x8, #SYS_exit
	svc #0
write_num:
	mov x2, #1
	mov x5, #10
div_loop:
	udiv x4, x3, x2
	cmp x4, #10
	b.lt dig_loop
	mul x2, x2, x5
	b div_loop
dig_loop:
	udiv x4, x3, x2
	msub x3, x4, x2, x3
	add x4, x4, #48
	strb w4, [x1], #1
	udiv x2, x2, x5
	cbnz x2, dig_loop
	ret
.bss
.balign 16
buf:
	.space 8192
outbuf:
	.space 64
