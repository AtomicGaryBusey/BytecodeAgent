.text
.globl _start
_start:
xor r12, r12
lea r13, [rip + input_buf]
read_loop:
cmp r12, 32768
jae parse_start
mov rdi, 0
lea rsi, [r13 + r12]
mov rdx, 32768
sub rdx, r12
mov rax, 0
syscall
test rax, rax
jz parse_start
js parse_start
add r12, rax
jmp read_loop
parse_start:
xor r14, r14
xor r15, r15
xor rbx, rbx
lea r10, [rip + line_off]
lea r9, [rip + line_len]
parse_loop:
cmp rbx, r12
jae parse_tail
movzx eax, byte ptr [r13 + rbx]
cmp eax, 10
jne parse_next
mov qword ptr [r10 + r14*8], r15
mov rax, rbx
sub rax, r15
mov qword ptr [r9 + r14*8], rax
inc r14
lea r15, [rbx + 1]
parse_next:
inc rbx
jmp parse_loop
parse_tail:
cmp r15, r12
jae sort_check
mov qword ptr [r10 + r14*8], r15
mov rax, r12
sub rax, r15
mov qword ptr [r9 + r14*8], rax
inc r14
sort_check:
cmp r14, 1
jbe write_start
mov rbx, 1
outer_loop:
cmp rbx, r14
jae write_start
mov rsi, rbx
inner_loop:
cmp rsi, 0
je outer_next
mov rax, qword ptr [r10 + rsi*8]
mov rdx, qword ptr [r9 + rsi*8]
mov rcx, rsi
dec rcx
mov rdi, qword ptr [r10 + rcx*8]
mov r8, qword ptr [r9 + rcx*8]
add rax, r13
add rdi, r13
mov r15, rdx
cmp r15, r8
jbe min_done
mov r15, r8
min_done:
xor rcx, rcx
cmp_loop:
cmp rcx, r15
jae cmp_len
movzx r11d, byte ptr [rax + rcx]
movzx ebp, byte ptr [rdi + rcx]
cmp r11d, ebp
jne cmp_diff
inc rcx
jmp cmp_loop
cmp_diff:
jb do_swap
jmp outer_next
cmp_len:
cmp rdx, r8
jb do_swap
jmp outer_next
do_swap:
mov rcx, rsi
dec rcx
mov rax, qword ptr [r10 + rsi*8]
mov rdx, qword ptr [r10 + rcx*8]
mov qword ptr [r10 + rsi*8], rdx
mov qword ptr [r10 + rcx*8], rax
mov rax, qword ptr [r9 + rsi*8]
mov rdx, qword ptr [r9 + rcx*8]
mov qword ptr [r9 + rsi*8], rdx
mov qword ptr [r9 + rcx*8], rax
dec rsi
jmp inner_loop
outer_next:
inc rbx
jmp outer_loop
write_start:
test r14, r14
jz exit_ok
xor rbx, rbx
write_loop:
cmp rbx, r14
jae exit_ok
mov rax, qword ptr [r10 + rbx*8]
mov rdx, qword ptr [r9 + rbx*8]
test rdx, rdx
jz write_nl
lea rsi, [r13 + rax]
mov rdi, 1
mov rax, 1
syscall
write_nl:
lea rsi, [rip + nl]
mov rdx, 1
mov rdi, 1
mov rax, 1
syscall
inc rbx
jmp write_loop
exit_ok:
mov rdi, 0
mov rax, 60
syscall
jmp exit_ok
