.text
.globl _start
_start:
  xor r12d, r12d
  xor r13d, r13d
  xor r14d, r14d
  xor r15d, r15d
read_loop:
  mov eax, 0
  mov edi, 0
  lea rsi, [rip + buffer]
  mov edx, 4096
  syscall
  test rax, rax
  jz done_input
  add r14, rax
  mov rbx, rax
  lea rsi, [rip + buffer]
byte_loop:
  movzx eax, byte ptr [rsi]
  inc rsi
  cmp al, 10
  jne not_nl
  inc r12
not_nl:
  cmp al, 32
  je is_ws
  cmp al, 9
  je is_ws
  cmp al, 10
  je is_ws
  cmp al, 11
  je is_ws
  cmp al, 12
  je is_ws
  cmp al, 13
  je is_ws
  test r15d, r15d
  jnz next_byte
  inc r13
  mov r15d, 1
  jmp next_byte
is_ws:
  xor r15d, r15d
next_byte:
  dec rbx
  jnz byte_loop
  jmp read_loop
done_input:
  lea rdi, [rip + outbuf]
  mov rax, r12
  call print_num
  mov byte ptr [rdi], 32
  inc rdi
  mov rax, r13
  call print_num
  mov byte ptr [rdi], 32
  inc rdi
  mov rax, r14
  call print_num
  mov byte ptr [rdi], 10
  inc rdi
  mov rax, rdi
  lea rdx, [rip + outbuf]
  sub rax, rdx
  mov rdx, rax
  lea rsi, [rip + outbuf]
  mov eax, 1
  mov edi, 1
  syscall
  mov eax, 60
  xor edi, edi
  syscall
print_num:
  push rbx
  mov ebx, 10
  test rax, rax
  jnz pn_divloop
  mov byte ptr [rdi], 48
  inc rdi
  pop rbx
  ret
pn_divloop:
  xor ecx, ecx
pn_div:
  xor edx, edx
  div rbx
  push rdx
  inc ecx
  test rax, rax
  jnz pn_div
pn_pop:
  pop rax
  add al, 48
  mov byte ptr [rdi], al
  inc rdi
  dec ecx
  jnz pn_pop
  pop rbx
  ret
.bss
.align 16
buffer:
.zero 4096
outbuf:
.zero 64
