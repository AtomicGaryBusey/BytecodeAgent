parse_skip:

parse_next:
  inc r14

parse_end:

  jae parse_done
  cmp r13, 1000
  jae parse_done
  mov qword ptr [r10 + r13*8], rbx
  mov rax, r12
  sub rax, rbx

  jae sort_done

  lea rsi, [r15 + r8]

  movzx edx, byte ptr [rsi]

sort_cmp_diff:
