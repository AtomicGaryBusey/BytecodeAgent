.text
.globl _start
_start:
  xor r12d, r12d
read_loop:
  cmp r12d, 65536
  jae read_done
  mov eax, 0
  mov edi, 0
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov edx, 65536
  sub edx, r12d
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_done:
  xor r13d, r13d
  cmp r13d, r12d
  jae final_line
row_loop:
  mov r14d, r12d
  sub r14d, r13d
  cmp r14d, 16
  jbe n_ok
  mov r14d, 16
n_ok:
  lea r15, [rip + input_buf]
  add r15, r13
  lea r8, [rip + out_buf]
  mov rdi, r8
  lea r9, [rip + hexdigits]
  mov eax, r13d
  mov ecx, 8
off_loop:
  rol eax, 4
  mov edx, eax
  and edx, 15
  mov dl, byte ptr [r9 + rdx]
  mov byte ptr [rdi], dl
  inc rdi
  dec ecx
  jnz off_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  add rdi, 2
  xor r10d, r10d
hex_loop:
  cmp r10d, r14d
  jae hex_blank
  movzx eax, byte ptr [r15 + r10]
  mov edx, eax
  shr edx, 4
  mov dl, byte ptr [r9 + rdx]
  mov byte ptr [rdi], dl
  and eax, 15
  mov al, byte ptr [r9 + rax]
  mov byte ptr [rdi + 1], al
  mov byte ptr [rdi + 2], 32
  add rdi, 3
  jmp hex_after
hex_blank:
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  mov byte ptr [rdi + 2], 32
  add rdi, 3
hex_after:
  cmp r10d, 7
  jne hex_no_extra
  mov byte ptr [rdi], 32
  inc rdi
hex_no_extra:
  inc r10d
  cmp r10d, 16
  jb hex_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 124
  add rdi, 2
  xor r10d, r10d
ascii_loop:
  cmp r10d, r14d
  jae ascii_done
  movzx eax, byte ptr [r15 + r10]
  cmp al, 32
  jb ascii_dot
  cmp al, 126
  ja ascii_dot
  mov byte ptr [rdi], al
  jmp ascii_next
ascii_dot:
  mov byte ptr [rdi], 46
ascii_next:
  inc rdi
  inc r10d
  jmp ascii_loop
ascii_done:
  mov byte ptr [rdi], 124
  mov byte ptr [rdi + 1], 10
  add rdi, 2
  mov rsi, r8
  sub rdi, r8
  mov rdx, rdi
  mov eax, 1
  mov edi, 1
  syscall
  add r13d, r14d
  cmp r13d, r12d
  jb row_loop
final_line:
  lea r8, [rip + out_buf]
  mov rdi, r8
  lea r9, [rip + hexdigits]
  mov eax, r12d
  mov ecx, 8
final_off_loop:
  rol eax, 4
  mov edx, eax
  and edx, 15
  mov dl, byte ptr [r9 + rdx]
  mov byte ptr [rdi], dl
  inc rdi
  dec ecx
  jnz final_off_loop
  mov byte ptr [rdi], 10
  inc rdi
  mov rsi, r8
  sub rdi, r8
  mov rdx, rdi
  mov eax, 1
  mov edi, 1
  syscall
  mov eax, 60
  xor edi, edi
  syscall
.section .rodata
hexdigits:
  .ascii "0123456789abcdef"
.bss
  .align 16
input_buf:
  .zero 65536
out_buf:
  .zero 80
