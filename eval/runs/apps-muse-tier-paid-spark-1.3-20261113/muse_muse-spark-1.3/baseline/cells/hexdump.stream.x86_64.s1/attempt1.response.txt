.text
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN_FD, 0
.equ STDOUT_FD, 1
.equ INPUT_MAX, 65536
.globl _start
_start:
xor r12d, r12d
read_loop:
cmp r12, INPUT_MAX
jae read_done
mov eax, SYS_READ
mov edi, STDIN_FD
lea rsi, [rip + input_buf]
add rsi, r12
mov edx, INPUT_MAX
sub rdx, r12
syscall
test rax, rax
jz read_done
add r12, rax
jmp read_loop
read_done:
xor r13d, r13d
row_loop:
cmp r13, r12
jae print_final
mov rax, r12
sub rax, r13
cmp rax, 16
jbe row_n_ok
mov eax, 16
row_n_ok:
mov r14, rax
lea r15, [rip + outbuf]
lea r8, [rip + hex_digits]
lea rsi, [rip + input_buf]
add rsi, r13
mov ecx, 28
off_loop:
mov eax, r13d
shr eax, cl
and eax, 15
mov al, byte ptr [r8 + rax]
mov byte ptr [r15], al
add r15, 1
sub ecx, 4
jns off_loop
mov byte ptr [r15], 32
mov byte ptr [r15 + 1], 32
add r15, 2
xor r10d, r10d
hex_loop:
cmp r10, r14
jae hex_pad
movzx eax, byte ptr [rsi + r10]
mov ebx, eax
shr ebx, 4
and ebx, 15
mov bl, byte ptr [r8 + rbx]
mov byte ptr [r15], bl
and eax, 15
mov al, byte ptr [r8 + rax]
mov byte ptr [r15 + 1], al
mov byte ptr [r15 + 2], 32
add r15, 3
jmp hex_after
hex_pad:
mov byte ptr [r15], 32
mov byte ptr [r15 + 1], 32
mov byte ptr [r15 + 2], 32
add r15, 3
hex_after:
cmp r10d, 7
jne hex_next
mov byte ptr [r15], 32
add r15, 1
hex_next:
add r10d, 1
cmp r10d, 16
jb hex_loop
mov byte ptr [r15], 32
mov byte ptr [r15 + 1], 124
add r15, 2
xor r10d, r10d
ascii_loop:
cmp r10, r14
jae ascii_done
movzx eax, byte ptr [rsi + r10]
cmp al, 32
jb ascii_dot
cmp al, 126
ja ascii_dot
jmp ascii_store
ascii_dot:
mov al, 46
ascii_store:
mov byte ptr [r15], al
add r15, 1
add r10d, 1
jmp ascii_loop
ascii_done:
mov byte ptr [r15], 124
mov byte ptr [r15 + 1], 10
add r15, 2
mov eax, SYS_WRITE
mov edi, STDOUT_FD
lea rsi, [rip + outbuf]
mov rdx, r15
sub rdx, rsi
syscall
add r13, r14
jmp row_loop
print_final:
lea r15, [rip + outbuf]
lea r8, [rip + hex_digits]
mov ecx, 28
final_loop:
mov eax, r12d
shr eax, cl
and eax, 15
mov al, byte ptr [r8 + rax]
mov byte ptr [r15], al
add r15, 1
sub ecx, 4
jns final_loop
mov byte ptr [r15], 10
add r15, 1
mov eax, SYS_WRITE
mov edi, STDOUT_FD
lea rsi, [rip + outbuf]
mov edx, 9
syscall
mov eax, SYS_EXIT
xor edi, edi
syscall
.section .rodata
hex_digits:
.ascii "0123456789abcdef"
.bss
input_buf:
.zero 65536
outbuf:
.zero 80
