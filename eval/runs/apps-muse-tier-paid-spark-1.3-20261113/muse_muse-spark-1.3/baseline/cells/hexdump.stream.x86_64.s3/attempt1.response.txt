.text
.globl _start
_start:
  xor r12, r12
  xor r13, r13
read_loop:
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov rdx, 65536
  sub rdx, r12
  test rdx, rdx
  jz read_done
  mov rax, 0
  mov rdi, 0
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_done:
  xor r13, r13
  lea r10, [rip + hexdigits]
row_loop:
  cmp r13, r12
  jae rows_done
  mov rax, r12
  sub rax, r13
  mov r14, 16
  cmp rax, 16
  jae have_n
  mov r14, rax
have_n:
  lea rbx, [rip + line_buf]
  lea rsi, [rip + input_buf]
  add rsi, r13
  mov r8d, 28
off_loop:
  mov eax, r13d
  mov cl, r8b
  shr eax, cl
  and eax, 15
  mov ecx, eax
  mov al, byte ptr [r10 + rcx]
  mov byte ptr [rbx], al
  inc rbx
  sub r8d, 4
  cmp r8d, 0
  jge off_loop
  mov byte ptr [rbx], 32
  mov byte ptr [rbx + 1], 32
  add rbx, 2
  xor r15, r15
hex_loop:
  cmp r15, 8
  jne no_extra
  mov byte ptr [rbx], 32
  inc rbx
no_extra:
  cmp r15, r14
  jae blank_slot
  movzx eax, byte ptr [rsi + r15]
  mov ecx, eax
  shr ecx, 4
  and ecx, 15
  mov dl, byte ptr [r10 + rcx]
  mov byte ptr [rbx], dl
  and eax, 15
  mov dl, byte ptr [r10 + rax]
  mov byte ptr [rbx + 1], dl
  mov byte ptr [rbx + 2], 32
  add rbx, 3
  jmp hex_next
blank_slot:
  mov byte ptr [rbx], 32
  mov byte ptr [rbx + 1], 32
  mov byte ptr [rbx + 2], 32
  add rbx, 3
hex_next:
  inc r15
  cmp r15, 16
  jb hex_loop
  mov byte ptr [rbx], 32
  mov byte ptr [rbx + 1], 124
  add rbx, 2
  xor r15, r15
ascii_loop:
  cmp r15, r14
  jae ascii_done
  movzx eax, byte ptr [rsi + r15]
  cmp al, 32
  jb ascii_dot
  cmp al, 126
  ja ascii_dot
  jmp ascii_store
ascii_dot:
  mov al, 46
ascii_store:
  mov byte ptr [rbx], al
  inc rbx
  inc r15
  jmp ascii_loop
ascii_done:
  mov byte ptr [rbx], 124
  mov byte ptr [rbx + 1], 10
  add rbx, 2
  lea rcx, [rip + line_buf]
  mov rdx, rbx
  sub rdx, rcx
  mov rsi, rcx
  mov rax, 1
  mov rdi, 1
  syscall
  add r13, r14
  jmp row_loop
rows_done:
  lea rbx, [rip + line_buf]
  mov r8d, 28
final_loop:
  mov eax, r12d
  mov cl, r8b
  shr eax, cl
  and eax, 15
  mov ecx, eax
  mov al, byte ptr [r10 + rcx]
  mov byte ptr [rbx], al
  inc rbx
  sub r8d, 4
  cmp r8d, 0
  jge final_loop
  mov byte ptr [rbx], 10
  inc rbx
  lea rcx, [rip + line_buf]
  mov rdx, rbx
  sub rdx, rcx
  mov rsi, rcx
  mov rax, 1
  mov rdi, 1
  syscall
  mov rax, 60
  xor rdi, rdi
  syscall
.section .rodata
hexdigits:
  .ascii "0123456789abcdef"
.bss
input_buf:
  .zero 65536
line_buf:
  .zero 80
