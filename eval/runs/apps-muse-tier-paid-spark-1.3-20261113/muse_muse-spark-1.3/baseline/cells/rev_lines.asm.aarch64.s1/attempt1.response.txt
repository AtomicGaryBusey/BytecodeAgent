.text
.globl _start
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ NEWLINE, 10
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0
read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #FD_STDIN
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    cbz x20, do_exit
    mov x21, #0
    mov x22, #0
scan_loop:
    cmp x22, x20
    b.hs scan_done
    add x0, x19, x22
    ldrb w0, [x0]
    cmp w0, #NEWLINE
    b.ne scan_next
    add x0, x19, x21
    sub x1, x22, x21
    bl reverse_range
    add x21, x22, #1
scan_next:
    add x22, x22, #1
    b scan_loop
scan_done:
    cmp x21, x20
    b.hs write_setup
    add x0, x19, x21
    sub x1, x20, x21
    bl reverse_range
    add x0, x19, x20
    mov w1, #NEWLINE
    strb w1, [x0]
    add x20, x20, #1
write_setup:
    mov x21, #0
write_loop:
    cmp x21, x20
    b.hs do_exit
    mov x0, #FD_STDOUT
    add x1, x19, x21
    sub x2, x20, x21
    mov x8, #SYS_WRITE
    svc #0
    add x21, x21, x0
    b write_loop
do_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    b do_exit
reverse_range:
    cmp x1, #1
    b.ls rev_done
    add x2, x0, x1
    sub x2, x2, #1
rev_loop:
    cmp x0, x2
    b.hs rev_done
    ldrb w3, [x0]
    ldrb w4, [x2]
    strb w4, [x0]
    strb w3, [x2]
    add x0, x0, #1
    sub x2, x2, #1
    b rev_loop
rev_done:
    ret
.bss
.balign 16
buf:
    .zero 65568
