.text
.globl _start
_start:
    xor r12d, r12d
    lea r10, [rip + input_buf]
read_loop:
    cmp r12, 65536
    jae drain_loop
    mov rdi, 0
    lea rsi, [r10 + r12*1]
    mov rdx, 65536
    sub rdx, r12
    mov eax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
drain_loop:
    mov rdi, 0
    lea rsi, [rip + dummy]
    mov edx, 1
    mov eax, 0
    syscall
    test rax, rax
    jz read_done
    jmp drain_loop
read_done:
    xor ecx, ecx
scan_loop:
    cmp rcx, r12
    jae no_bang
    movzx eax, byte ptr [r10 + rcx*1]
    cmp al, 33
    je found_bang
    inc rcx
    jmp scan_loop
found_bang:
    mov r13, rcx
    lea r14, [r10 + rcx*1 + 1]
    lea r15, [r10 + r12*1]
    jmp scan_done
no_bang:
    mov r13, r12
    lea r14, [r10 + r12*1]
    mov r15, r14
scan_done:
    lea r8, [rip + match_tab]
    xor ebx, ebx
    dec rbx
    xor ecx, ecx
    xor r12d, r12d
filter_loop:
    cmp rcx, r13
    jae filter_done
    movzx eax, byte ptr [r10 + rcx*1]
    cmp al, 62
    je filt_copy
    cmp al, 60
    je filt_copy
    cmp al, 43
    je filt_copy
    cmp al, 45
    je filt_copy
    cmp al, 46
    je filt_copy
    cmp al, 44
    je filt_copy
    cmp al, 91
    je filt_open
    cmp al, 93
    je filt_close
    inc rcx
    jmp filter_loop
filt_copy:
    mov byte ptr [r10 + r12*1], al
    inc r12
    inc rcx
    jmp filter_loop
filt_open:
    mov byte ptr [r10 + r12*1], al
    mov dword ptr [r8 + r12*4], ebx
    mov rbx, r12
    inc r12
    inc rcx
    jmp filter_loop
filt_close:
    cmp rbx, -1
    je unbalanced
    mov byte ptr [r10 + r12*1], al
    mov eax, dword ptr [r8 + rbx*4]
    mov dword ptr [r8 + rbx*4], r12d
    mov dword ptr [r8 + r12*4], ebx
    movsxd rbx, eax
    inc r12
    inc rcx
    jmp filter_loop
filter_done:
    cmp rbx, -1
    je run_init
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall
    jmp unbalanced
run_init:
    mov r13, r12
    lea rbx, [rip + tape]
    xor ebp, ebp
    xor r12d, r12d
    lea r9, [rip + out_buf]
interp_loop:
    cmp rbp, r13
    jae interp_done
    movzx eax, byte ptr [r10 + rbp*1]
    cmp al, 62
    je do_right
    cmp al, 60
    je do_left
    cmp al, 43
    je do_inc
    cmp al, 45
    je do_dec
    cmp al, 46
    je do_out
    cmp al, 44
    je do_in
    cmp al, 91
    je do_open
    cmp al, 93
    je do_close
    inc rbp
    jmp interp_loop
do_right:
    inc rbx
    inc rbp
    jmp interp_loop
do_left:
    dec rbx
    inc rbp
    jmp interp_loop
do_inc:
    inc byte ptr [rbx]
    inc rbp
    jmp interp_loop
do_dec:
    dec byte ptr [rbx]
    inc rbp
    jmp interp_loop
do_out:
    mov al, byte ptr [rbx]
    mov byte ptr [r9 + r12*1], al
    inc r12
    inc rbp
    jmp interp_loop
do_in:
    cmp r14, r15
    jae in_eof
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbx], al
    inc rbp
    jmp interp_loop
in_eof:
    mov byte ptr [rbx], 0
    inc rbp
    jmp interp_loop
do_open:
    mov al, byte ptr [rbx]
    test al, al
    jnz open_nz
    mov eax, dword ptr [r8 + rbp*4]
    inc rax
    mov rbp, rax
    jmp interp_loop
open_nz:
    inc rbp
    jmp interp_loop
do_close:
    mov al, byte ptr [rbx]
    test al, al
    jz close_z
    mov eax, dword ptr [r8 + rbp*4]
    inc rax
    mov rbp, rax
    jmp interp_loop
close_z:
    inc rbp
    jmp interp_loop
interp_done:
    test r12, r12
    jz exit0
    xor ebx, ebx
write_loop:
    mov rdi, 1
    lea rsi, [r9 + rbx*1]
    mov rdx, r12
    sub rdx, rbx
    mov eax, 1
    syscall
    add rbx, rax
    cmp rbx, r12
    jb write_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit0
.bss
input_buf:
    .zero 65536
tape:
    .zero 30000
out_buf:
    .zero 65536
    .balign 4
match_tab:
    .zero 262144
dummy:
    .zero 16
