.text
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ MAX_INPUT, 32768
.globl _start
_start:
# r12 = total bytes
xor r12, r12
read_loop:
cmp r12, MAX_INPUT
jae read_done
mov rax, SYS_READ
mov rdi, FD_STDIN
lea rsi, [rip + input_buf]
add rsi, r12
mov rdx, MAX_INPUT
sub rdx, r12
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_done:
xor r13, r13
test r12, r12
jz exit_ok
xor rbx, rbx
xor rcx, rcx
lea rbp, [rip + input_buf]
parse_loop:
cmp rcx, r12
jae parse_tail
movzx eax, byte ptr [rbp + rcx]
cmp al, 10
jne parse_next
mov rdx, rcx
sub rdx, rbx
lea rsi, [rip + line_off]
mov qword ptr [rsi + r13*8], rbx
lea rsi, [rip + line_len]
mov qword ptr [rsi + r13*8], rdx
inc r13
lea rbx, [rcx + 1]
parse_next:
inc rcx
jmp parse_loop
parse_tail:
cmp rbx, r12
jae parse_done
mov rdx, r12
sub rdx, rbx
lea rsi, [rip + line_off]
mov qword ptr [rsi + r13*8], rbx
lea rsi, [rip + line_len]
mov qword ptr [rsi + r13*8], rdx
inc r13
parse_done:
cmp r13, 1
jle write_init
mov r14, 1
outer_loop:
cmp r14, r13
jge write_init
lea rsi, [rip + line_off]
mov r8, qword ptr [rsi + r14*8]
lea rsi, [rip + line_len]
mov r9, qword ptr [rsi + r14*8]
mov r15, r14
dec r15
inner_loop:
cmp r15, 0
jl inner_done
lea rsi, [rip + line_off]
mov r10, qword ptr [rsi + r15*8]
lea rsi, [rip + line_len]
mov rdi, qword ptr [rsi + r15*8]
mov rsi, rdi
cmp rsi, r9
cmova rsi, r9
lea rax, [rip + input_buf]
mov rdx, rax
add rdx, r10
mov rcx, rax
add rcx, r8
xor rbx, rbx
cmp_loop:
cmp rbx, rsi
jae cmp_len
movzx eax, byte ptr [rdx + rbx]
movzx ebp, byte ptr [rcx + rbx]
cmp al, bpl
je cmp_next
ja do_shift
jmp inner_done
cmp_next:
inc rbx
jmp cmp_loop
cmp_len:
cmp rdi, r9
ja do_shift
jmp inner_done
do_shift:
lea rdx, [r15 + 1]
lea rax, [rip + line_off]
mov qword ptr [rax + rdx*8], r10
lea rax, [rip + line_len]
mov qword ptr [rax + rdx*8], rdi
dec r15
jmp inner_loop
inner_done:
lea rdx, [r15 + 1]
lea rax, [rip + line_off]
mov qword ptr [rax + rdx*8], r8
lea rax, [rip + line_len]
mov qword ptr [rax + rdx*8], r9
inc r14
jmp outer_loop
write_init:
xor r14, r14
write_loop:
cmp r14, r13
jge exit_ok
lea rax, [rip + line_off]
mov r15, qword ptr [rax + r14*8]
lea rax, [rip + line_len]
mov r12, qword ptr [rax + r14*8]
test r12, r12
jz write_nl
mov rax, SYS_WRITE
mov rdi, FD_STDOUT
lea rsi, [rip + input_buf]
add rsi, r15
mov rdx, r12
syscall
write_nl:
mov rax, SYS_WRITE
mov rdi, FD_STDOUT
lea rsi, [rip + newline]
mov rdx, 1
syscall
inc r14
jmp write_loop
exit_ok:
mov rax, SYS_EXIT
xor rdi, rdi
syscall
.section .rodata
newline:
.byte 10
.byte 0
.byte 0
.byte 0
.byte 0
.byte 0
.byte 0
.byte 0
.bss
.balign 8
input_buf:
.space 32768
.balign 8
line_off:
.space 8000
.balign 8
line_len:
.space 8000
