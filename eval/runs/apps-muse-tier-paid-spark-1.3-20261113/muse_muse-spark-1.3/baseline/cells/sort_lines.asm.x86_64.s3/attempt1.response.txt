.text
.globl _start
_start:
  xor r12d, r12d
read_loop:
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov edx, 32784
  sub edx, r12d
  jz parse_start
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz parse_start
  js parse_start
  add r12, rax
  jmp read_loop
parse_start:
  lea rbx, [rip + input_buf]
  xor r13d, r13d
  xor r14d, r14d
  xor r15d, r15d
parse_loop:
  cmp r14, r12
  jae parse_end
  mov al, byte ptr [rbx + r14]
  cmp al, 10
  jne parse_next
  mov rax, rbx
  add rax, r15
  mov rcx, r14
  sub rcx, r15
  lea rdx, [rip + line_ptr]
  mov qword ptr [rdx + r13*8], rax
  lea rdx, [rip + line_len]
  mov qword ptr [rdx + r13*8], rcx
  inc r13
  mov r15, r14
  inc r15
parse_next:
  inc r14
  jmp parse_loop
parse_end:
  cmp r15, r12
  jae sort_check
  mov rax, rbx
  add rax, r15
  mov rcx, r12
  sub rcx, r15
  lea rdx, [rip + line_ptr]
  mov qword ptr [rdx + r13*8], rax
  lea rdx, [rip + line_len]
  mov qword ptr [rdx + r13*8], rcx
  inc r13
sort_check:
  cmp r13, 1
  jle output_start
  mov r14d, 1
outer_loop:
  cmp r14, r13
  jge output_start
  lea rax, [rip + line_ptr]
  mov rbx, qword ptr [rax + r14*8]
  lea rax, [rip + line_len]
  mov rbp, qword ptr [rax + r14*8]
  mov r15, r14
  dec r15
inner_loop:
  cmp r15, 0
  jl insert_key
  lea rax, [rip + line_ptr]
  mov r8, qword ptr [rax + r15*8]
  lea rax, [rip + line_len]
  mov r9, qword ptr [rax + r15*8]
  mov rcx, r9
  cmp rcx, rbp
  jbe min_done
  mov rcx, rbp
min_done:
  xor edx, edx
cmp_loop:
  cmp rdx, rcx
  jae cmp_len
  mov al, byte ptr [r8 + rdx]
  cmp al, byte ptr [rbx + rdx]
  jne cmp_diff
  inc rdx
  jmp cmp_loop
cmp_diff:
  ja do_shift
  jmp insert_key
cmp_len:
  cmp r9, rbp
  ja do_shift
  jmp insert_key
do_shift:
  mov rax, r15
  inc rax
  lea r10, [rip + line_ptr]
  mov qword ptr [r10 + rax*8], r8
  lea r10, [rip + line_len]
  mov qword ptr [r10 + rax*8], r9
  dec r15
  jmp inner_loop
insert_key:
  mov rax, r15
  inc rax
  lea r10, [rip + line_ptr]
  mov qword ptr [r10 + rax*8], rbx
  lea r10, [rip + line_len]
  mov qword ptr [r10 + rax*8], rbp
  inc r14
  jmp outer_loop
output_start:
  xor r14d, r14d
out_loop:
  cmp r14, r13
  jge exit_ok
  lea rax, [rip + line_ptr]
  mov rsi, qword ptr [rax + r14*8]
  lea rax, [rip + line_len]
  mov rdx, qword ptr [rax + r14*8]
  test rdx, rdx
  jz write_nl
  mov edi, 1
  mov eax, 1
  syscall
write_nl:
  lea rsi, [rip + nl]
  mov edx, 1
  mov edi, 1
  mov eax, 1
  syscall
  inc r14
  jmp out_loop
exit_ok:
  xor edi, edi
  mov eax, 60
  syscall
exit_hang:
  jmp exit_hang
.section .rodata
nl:
.byte 10
.bss
.align 8
input_buf:
.zero 32784
.align 8
line_ptr:
.zero 8000
.align 8
line_len:
.zero 8000
