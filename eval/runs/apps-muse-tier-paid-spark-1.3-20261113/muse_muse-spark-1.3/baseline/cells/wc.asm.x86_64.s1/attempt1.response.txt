.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 4096
    syscall
    test rax, rax
    jz done
    js done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buffer]
inner_loop:
    test rcx, rcx
    jz read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb is_word
    cmp al, 13
    jbe is_ws
    jmp is_word
is_ws:
    xor r15, r15
    jmp inner_loop
is_word:
    test r15, r15
    jnz inner_loop
    inc r13
    mov r15, 1
    jmp inner_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
print_num:
    cmp rax, 0
    jne pn_nz
    mov byte ptr [rdi], 48
    inc rdi
    ret
pn_nz:
    xor ecx, ecx
    mov rsi, 10
pn_div:
    xor edx, edx
    div rsi
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz pn_div
pn_pop:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz pn_pop
    ret
.bss
buffer:
    .zero 4096
outbuf:
    .zero 64
