.globl _start
.text
_start:
 xor r12, r12
 xor r13, r13
 xor r14, r14
 xor r15, r15
read_loop:
 mov rax, 0
 mov rdi, 0
 lea rsi, [rip + buffer]
 mov rdx, 4096
 syscall
 test rax, rax
 jle done
 mov rcx, rax
 lea rsi, [rip + buffer]
process_byte:
 movzx eax, byte ptr [rsi]
 inc rsi
 inc r14
 cmp al, 10
 jne check_ws
 inc r12
check_ws:
 cmp al, 32
 je is_space
 cmp al, 9
 jb is_word
 cmp al, 13
 jbe is_space
is_word:
 test r15, r15
 jnz next_byte
 inc r13
 mov r15, 1
 jmp next_byte
is_space:
 xor r15, r15
next_byte:
 dec rcx
 jnz process_byte
 jmp read_loop
done:
 lea rdi, [rip + outbuf]
 mov rax, r12
 call write_num
 mov byte ptr [rdi], 32
 inc rdi
 mov rax, r13
 call write_num
 mov byte ptr [rdi], 32
 inc rdi
 mov rax, r14
 call write_num
 mov byte ptr [rdi], 10
 inc rdi
 lea rsi, [rip + outbuf]
 mov rdx, rdi
 sub rdx, rsi
write_loop:
 mov rax, 1
 mov rdi, 1
 syscall
 test rax, rax
 jle write_done
 sub rdx, rax
 jz write_done
 add rsi, rax
 jmp write_loop
write_done:
 mov rax, 60
 xor edi, edi
 syscall
write_num:
 test rax, rax
 jnz write_num_nz
 mov byte ptr [rdi], 48
 inc rdi
 ret
write_num_nz:
 mov rbx, 10
 xor ecx, ecx
div_loop:
 xor edx, edx
 div rbx
 push rdx
 inc rcx
 test rax, rax
 jnz div_loop
pop_loop:
 pop rax
 add al, 48
 mov byte ptr [rdi], al
 inc rdi
 dec rcx
 jnz pop_loop
 ret
.bss
buffer:
 .zero 4096
outbuf:
 .zero 64
