.text
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ IN_CAP, 65536
.equ SCRATCH_LEN, 64
.globl _start
_start:
lea r12, [rip + inbuf]
xor ebx, ebx
read_loop:
mov edx, IN_CAP
sub edx, ebx
jz drain_start
mov eax, SYS_READ
mov edi, FD_STDIN
lea rsi, [r12 + rbx]
syscall
cmp rax, 0
jle read_done
add rbx, rax
cmp rbx, IN_CAP
jb read_loop
drain_start:
drain_loop:
mov eax, SYS_READ
mov edi, FD_STDIN
lea rsi, [rip + scratch]
mov edx, SCRATCH_LEN
syscall
cmp rax, 0
jg drain_loop
read_done:
mov r13, rbx
mov r14, rbx
mov r15, rbx
test rbx, rbx
jz scan_done
xor ecx, ecx
scan_loop:
cmp rcx, rbx
jae scan_done
cmp byte ptr [r12 + rcx], 0x21
je found_bang
inc rcx
jmp scan_loop
found_bang:
mov r13, rcx
mov r14, rcx
inc r14
scan_done:
lea r10, [rip + match]
xor ebx, ebx
xor ebp, ebp
bracket_loop:
cmp rbx, r13
jae bracket_done
movzx eax, byte ptr [r12 + rbx]
cmp al, 0x5b
je bracket_open
cmp al, 0x5d
je bracket_close
inc rbx
jmp bracket_loop
bracket_open:
push rbx
inc rbp
inc rbx
jmp bracket_loop
bracket_close:
test rbp, rbp
jz unbalanced
pop rdx
dec rbp
mov dword ptr [r10 + rdx*4], ebx
mov dword ptr [r10 + rbx*4], edx
inc rbx
jmp bracket_loop
bracket_done:
test rbp, rbp
jnz unbalanced
exec_setup:
lea r8, [rip + tape]
lea r9, [rip + outbuf]
xor ebx, ebx
xor ebp, ebp
xor r11d, r11d
exec_loop:
cmp rbp, r13
jae exec_done
movzx eax, byte ptr [r12 + rbp]
cmp al, 0x3e
je do_gt
cmp al, 0x3c
je do_lt
cmp al, 0x2b
je do_plus
cmp al, 0x2d
je do_minus
cmp al, 0x2e
je do_dot
cmp al, 0x2c
je do_comma
cmp al, 0x5b
je do_open
cmp al, 0x5d
je do_close
inc rbp
jmp exec_loop
do_gt:
inc rbx
inc rbp
jmp exec_loop
do_lt:
dec rbx
inc rbp
jmp exec_loop
do_plus:
inc byte ptr [r8 + rbx]
inc rbp
jmp exec_loop
do_minus:
dec byte ptr [r8 + rbx]
inc rbp
jmp exec_loop
do_dot:
mov al, byte ptr [r8 + rbx]
mov byte ptr [r9 + r11], al
inc r11
inc rbp
jmp exec_loop
do_comma:
cmp r14, r15
jae comma_empty
mov al, byte ptr [r12 + r14]
inc r14
mov byte ptr [r8 + rbx], al
inc rbp
jmp exec_loop
comma_empty:
mov byte ptr [r8 + rbx], 0
inc rbp
jmp exec_loop
do_open:
cmp byte ptr [r8 + rbx], 0
jne open_nz
mov eax, dword ptr [r10 + rbp*4]
mov rbp, rax
inc rbp
jmp exec_loop
open_nz:
inc rbp
jmp exec_loop
do_close:
cmp byte ptr [r8 + rbx], 0
je close_z
mov eax, dword ptr [r10 + rbp*4]
mov rbp, rax
inc rbp
jmp exec_loop
close_z:
inc rbp
jmp exec_loop
exec_done:
test r11, r11
jz exit_ok
mov rbx, r11
xor ebp, ebp
write_loop:
mov eax, SYS_WRITE
mov edi, FD_STDOUT
lea rsi, [r9 + rbp]
mov rdx, rbx
syscall
cmp rax, 0
jle exit_ok
add rbp, rax
sub rbx, rax
jnz write_loop
exit_ok:
mov eax, SYS_EXIT
xor edi, edi
syscall
jmp exit_ok
unbalanced:
mov eax, SYS_EXIT
mov edi, 1
syscall
jmp unbalanced
.bss
inbuf:
.zero 65536
tape:
.zero 30000
outbuf:
.zero 65536
match:
.zero 262144
scratch:
.zero 64
