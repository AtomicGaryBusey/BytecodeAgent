.text
.globl _start
_start:
    xor r12, r12 # lines
    xor r13, r13 # words
    xor r14, r14 # bytes
    xor r15, r15 # in_word flag
read_loop:
    xor eax, eax # read = 0
    xor edi, edi # fd = 0
    lea rsi, [rip + buffer]
    mov edx, 8192
    syscall
    test rax, rax
    jle done
    mov rcx, rax
    lea rbx, [rip + buffer]
byte_loop:
    movzx eax, byte ptr [rbx]
    inc r14
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    cmp al, 32
    je is_ws
    cmp al, 9
    je is_ws
    cmp al, 10
    je is_ws
    cmp al, 11
    je is_ws
    cmp al, 12
    je is_ws
    cmp al, 13
    je is_ws
    test r15, r15
    jnz byte_next
    inc r13
    mov r15, 1
    jmp byte_next
is_ws:
    xor r15, r15
byte_next:
    inc rbx
    dec rcx
    jnz byte_loop
    jmp read_loop
done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_uint
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_uint
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_uint
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
hang:
    jmp hang
print_uint:
    mov rbx, 10
    xor ecx, ecx
div_loop:
    xor edx, edx
    div rbx
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz div_loop
copy_loop:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz copy_loop
    ret
.bss
.balign 16
buffer:
    .zero 8192
.balign 16
outbuf:
    .zero 64
