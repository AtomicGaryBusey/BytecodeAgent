.text
.globl _start
_start:
    xor r12d, r12d
    lea rbx, [rip + inbuf]
    lea rbp, [rip + outbuf]
read_loop:
    cmp r12, 65536
    jae read_done
    mov eax, 0
    mov edi, 0
    mov rsi, rbx
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    jz read_done
    js read_done
    add r12, rax
    jmp read_loop
read_done:
    test r12, r12
    jz do_exit
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
proc_loop:
    cmp r13, r12
    jae proc_tail
    mov al, byte ptr [rbx + r13]
    cmp al, 10
    jne proc_next
    mov rsi, r13
    dec rsi
    mov rdi, r14
rev1_loop:
    cmp rsi, rdi
    jl rev1_done
    mov al, byte ptr [rbx + rsi]
    mov byte ptr [rbp + r15], al
    inc r15
    dec rsi
    jmp rev1_loop
rev1_done:
    mov byte ptr [rbp + r15], 10
    inc r15
    mov r14, r13
    inc r14
proc_next:
    inc r13
    jmp proc_loop
proc_tail:
    cmp r14, r12
    jae write_start
    mov rsi, r12
    dec rsi
    mov rdi, r14
rev2_loop:
    cmp rsi, rdi
    jl rev2_done
    mov al, byte ptr [rbx + rsi]
    mov byte ptr [rbp + r15], al
    inc r15
    dec rsi
    jmp rev2_loop
rev2_done:
    mov byte ptr [rbp + r15], 10
    inc r15
write_start:
    xor r12d, r12d
write_loop:
    cmp r12, r15
    jae do_exit
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    add rsi, r12
    mov rdx, r15
    sub rdx, r12
    syscall
    test rax, rax
    jz do_exit
    js do_exit
    add r12, rax
    jmp write_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
    jmp do_exit
