.text
.globl _start
_start:
    // load base addresses for buffers and table
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x22, alphabet
    add x22, x22, :lo12:alphabet
    // total input length starts at zero
    mov x19, #0
read_loop:
    // remaining space in input buffer
    movz x2, #49152
    sub x2, x2, x19
    cbz x2, read_done
    // read from standard input
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop
read_done:
    // empty input prints nothing
    cbz x19, exit_ok
    // init encode offsets and column
    mov x23, #0
    mov x24, #0
    mov x25, #0
encode_full_loop:
    // check for at least three bytes left
    sub x9, x19, x23
    cmp x9, #3
    b.lo encode_remainder
    // load three bytes
    add x9, x20, x23
    ldrb w10, [x9]
    ldrb w11, [x9, #1]
    ldrb w12, [x9, #2]
    // first six bits
    lsr w13, w10, #2
    // second six bits
    and w14, w10, #3
    lsl w14, w14, #4
    lsr w9, w11, #4
    orr w14, w14, w9
    // third six bits
    and w15, w11, #15
    lsl w15, w15, #2
    lsr w9, w12, #6
    orr w15, w15, w9
    // fourth six bits
    and w16, w12, #63
    // emit four characters
    add x9, x22, x13
    ldrb w9, [x9]
    bl emit_char
    add x9, x22, x14
    ldrb w9, [x9]
    bl emit_char
    add x9, x22, x15
    ldrb w9, [x9]
    bl emit_char
    add x9, x22, x16
    ldrb w9, [x9]
    bl emit_char
    add x23, x23, #3
    b encode_full_loop
encode_remainder:
    // remaining bytes after full groups
    sub x9, x19, x23
    cbz x9, encode_done
    cmp x9, #1
    b.eq remainder_one
    // two bytes left
    add x9, x20, x23
    ldrb w10, [x9]
    ldrb w11, [x9, #1]
    lsr w13, w10, #2
    and w14, w10, #3
    lsl w14, w14, #4
    lsr w9, w11, #4
    orr w14, w14, w9
    and w15, w11, #15
    lsl w15, w15, #2
    add x9, x22, x13
    ldrb w9, [x9]
    bl emit_char
    add x9, x22, x14
    ldrb w9, [x9]
    bl emit_char
    add x9, x22, x15
    ldrb w9, [x9]
    bl emit_char
    mov w9, #61
    bl emit_char
    b encode_done
remainder_one:
    // one byte left
    add x9, x20, x23
    ldrb w10, [x9]
    lsr w13, w10, #2
    and w14, w10, #3
    lsl w14, w14, #4
    add x9, x22, x13
    ldrb w9, [x9]
    bl emit_char
    add x9, x22, x14
    ldrb w9, [x9]
    bl emit_char
    mov w9, #61
    bl emit_char
    mov w9, #61
    bl emit_char
encode_done:
    // terminate last line if needed
    cbz x25, write_out
    mov w9, #10
    add x10, x21, x24
    strb w9, [x10]
    add x24, x24, #1
    mov x25, #0
write_out:
    mov x23, #0
write_loop:
    sub x9, x24, x23
    cbz x9, exit_ok
    add x1, x21, x23
    mov x2, x9
    mov x0, #1
    mov x8, #64
    svc #0
    add x23, x23, x0
    b write_loop
exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0
    b exit_ok
emit_char:
    // store char and wrap lines at seventy six
    add x10, x21, x24
    strb w9, [x10]
    add x24, x24, #1
    add x25, x25, #1
    cmp x25, #76
    b.ne emit_done
    mov w9, #10
    add x10, x21, x24
    strb w9, [x10]
    add x24, x24, #1
    mov x25, #0
emit_done:
    ret
.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.bss
inbuf:
    .zero 49152
outbuf:
    .zero 66560
