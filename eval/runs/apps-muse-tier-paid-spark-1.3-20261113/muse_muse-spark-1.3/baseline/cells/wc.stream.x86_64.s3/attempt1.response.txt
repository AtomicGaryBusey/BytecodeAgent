.text
.globl _start
_start:
 xor r12, r12
 xor r13, r13
 xor r14, r14
 xor r15, r15
read_loop:
 mov rax, 0
 mov rdi, 0
 lea rsi, [rip + buffer]
 mov rdx, 4096
 syscall
 test rax, rax
 jz print_out
 mov rcx, rax
 lea rsi, [rip + buffer]
byte_loop:
 test rcx, rcx
 jz read_loop
 movzx eax, byte ptr [rsi]
 inc rsi
 dec rcx
 cmp al, 10
 jne check_ws
 inc r12
check_ws:
 cmp al, 32
 je is_ws
 cmp al, 9
 jb is_word
 cmp al, 13
 jbe is_ws
is_word:
 test r15, r15
 jnz byte_loop
 inc r13
 mov r15, 1
 jmp byte_loop
is_ws:
 xor r15d, r15d
 jmp byte_loop
print_out:
 lea rsi, [rip + outbuf]
 add rsi, 64
 mov r8, rsi
 dec rsi
 mov byte ptr [rsi], 10
 mov rax, r14
 mov rbx, 10
 test rax, rax
 jnz conv_bytes_loop
 dec rsi
 mov byte ptr [rsi], 48
 jmp conv_bytes_done
conv_bytes_loop:
 xor edx, edx
 div rbx
 add dl, 48
 dec rsi
 mov byte ptr [rsi], dl
 test rax, rax
 jnz conv_bytes_loop
conv_bytes_done:
 dec rsi
 mov byte ptr [rsi], 32
 mov rax, r13
 test rax, rax
 jnz conv_words_loop
 dec rsi
 mov byte ptr [rsi], 48
 jmp conv_words_done
conv_words_loop:
 xor edx, edx
 div rbx
 add dl, 48
 dec rsi
 mov byte ptr [rsi], dl
 test rax, rax
 jnz conv_words_loop
conv_words_done:
 dec rsi
 mov byte ptr [rsi], 32
 mov rax, r12
 test rax, rax
 jnz conv_lines_loop
 dec rsi
 mov byte ptr [rsi], 48
 jmp conv_lines_done
conv_lines_loop:
 xor edx, edx
 div rbx
 add dl, 48
 dec rsi
 mov byte ptr [rsi], dl
 test rax, rax
 jnz conv_lines_loop
conv_lines_done:
 mov rdx, r8
 sub rdx, rsi
 mov rax, 1
 mov rdi, 1
 syscall
 mov rax, 60
 xor edi, edi
 syscall
.bss
buffer:
 .zero 4096
outbuf:
 .zero 64
