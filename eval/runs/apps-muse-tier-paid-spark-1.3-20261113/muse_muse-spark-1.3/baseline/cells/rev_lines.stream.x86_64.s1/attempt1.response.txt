.text
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INPUT_CAP, 65536
.globl _start
_start:
    xor r12d, r12d
    lea r15, [rip + input_buf]
read_loop:
    mov edx, INPUT_CAP
    sub rdx, r12
    jz drain
    mov rsi, r15
    add rsi, r12
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
drain:
    lea rsi, [rip + dummy]
    mov edx, 1
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    js read_done
    jmp drain
read_done:
    lea r15, [rip + input_buf]
    lea r14, [rip + output_buf]
    xor r13d, r13d
    xor ebx, ebx
outer:
    cmp rbx, r12
    jae write_out
    mov rcx, rbx
scan:
    cmp rcx, r12
    jae scan_done
    cmp byte ptr [r15 + rcx], 0x0A
    je scan_done
    inc rcx
    jmp scan
scan_done:
    mov r8, rcx
    sub r8, rbx
    jz emit_nl
    lea rsi, [r15 + rbx]
    lea rdi, [r14 + r13]
    add rsi, r8
    dec rsi
    mov rdx, r8
rev_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    dec rsi
    inc rdi
    dec rdx
    jnz rev_loop
    add r13, r8
emit_nl:
    mov byte ptr [r14 + r13], 0x0A
    inc r13
    mov rbx, rcx
    cmp rbx, r12
    jae write_out
    inc rbx
    jmp outer
write_out:
    test r13, r13
    jz exit0
    mov r12, r13
    lea r15, [rip + output_buf]
write_loop:
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r15
    mov rdx, r12
    syscall
    add r15, rax
    sub r12, rax
    jnz write_loop
exit0:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
hang:
    jmp hang
.bss
.align 16
input_buf:
    .zero 65536
output_buf:
    .zero 65537
dummy:
    .zero 1
