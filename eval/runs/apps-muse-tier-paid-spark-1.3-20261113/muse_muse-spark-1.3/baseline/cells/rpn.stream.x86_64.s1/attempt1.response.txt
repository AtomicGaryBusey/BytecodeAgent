.globl _start
.section .rodata
err_msg:
.byte 0x65
.byte 0x72
.byte 0x72
.byte 0x6f
.byte 0x72
.byte 0x0A
.bss
.align 16
input_buf:
.zero 65536
.align 8
eval_stack:
.zero 8192
.align 16
out_buf:
.zero 32
.text
_start:
lea r15, [rip + input_buf]
lea rbp, [rip + eval_stack]
xor r12, r12
read_loop:
mov rax, 65536
sub rax, r12
jz read_full
mov rdx, rax
mov rsi, r15
add rsi, r12
mov rdi, 0
mov rax, 0
syscall
test rax, rax
jz read_done
js read_done
add r12, rax
jmp read_loop
read_full:
lea rsi, [rip + out_buf]
mov rdx, 32
mov rdi, 0
mov rax, 0
syscall
test rax, rax
jz read_done
js read_done
jmp read_full
read_done:
test r12, r12
jz exit0
xor r14, r14
line_loop:
cmp r14, r12
jge exit0
mov r8, r15
add r8, r14
mov r10, r14
find_nl_loop:
cmp r10, r12
jge find_nl_eof
mov rax, r15
add rax, r10
cmp byte ptr [rax], 0x0A
je find_nl_found
inc r10
jmp find_nl_loop
find_nl_found:
mov r9, r10
sub r9, r14
mov r14, r10
inc r14
jmp eval_start
find_nl_eof:
mov r9, r12
sub r9, r14
mov r14, r12
eval_start:
xor rbx, rbx
xor r10, r10
token_skip:
cmp r10, r9
jge tokens_done
mov rax, r8
add rax, r10
movzx eax, byte ptr [rax]
cmp al, 0x20
je token_skip_adv
cmp al, 0x09
je token_skip_adv
jmp token_begin
token_skip_adv:
inc r10
jmp token_skip
token_begin:
mov r13, r10
token_scan:
cmp r10, r9
jge token_scan_done
mov rax, r8
add rax, r10
movzx eax, byte ptr [rax]
cmp al, 0x20
je token_scan_done
cmp al, 0x09
je token_scan_done
inc r10
jmp token_scan
token_scan_done:
mov rsi, r8
add rsi, r13
mov rdx, r10
sub rdx, r13
cmp rdx, 1
jne try_literal
movzx eax, byte ptr [rsi]
cmp al, 0x2B
je do_operator
cmp al, 0x2D
je do_operator
cmp al, 0x2A
je do_operator
cmp al, 0x2F
je do_operator
cmp al, 0x25
je do_operator
jmp try_literal
try_literal:
movzx eax, byte ptr [rsi]
cmp al, 0x2D
je lit_negative
cmp rdx, 18
ja line_error
xor rax, rax
xor rcx, rcx
lit_pos_loop:
cmp rcx, rdx
jge lit_pos_done
movzx r11d, byte ptr [rsi + rcx]
cmp r11b, 0x30
jb line_error
cmp r11b, 0x39
ja line_error
sub r11b, 0x30
imul rax, rax, 10
add rax, r11
inc rcx
jmp lit_pos_loop
lit_pos_done:
cmp rbx, 1024
jge line_error
mov qword ptr [rbp + rbx*8], rax
inc rbx
jmp token_skip
lit_negative:
cmp rdx, 2
jb line_error
cmp rdx, 19
ja line_error
xor rax, rax
mov rcx, 1
lit_neg_loop:
cmp rcx, rdx
jge lit_neg_done
movzx r11d, byte ptr [rsi + rcx]
cmp r11b, 0x30
jb line_error
cmp r11b, 0x39
ja line_error
sub r11b, 0x30
imul rax, rax, 10
add rax, r11
inc rcx
jmp lit_neg_loop
lit_neg_done:
neg rax
cmp rbx, 1024
jge line_error
mov qword ptr [rbp + rbx*8], rax
inc rbx
jmp token_skip
do_operator:
mov r11b, al
cmp rbx, 2
jb line_error
dec rbx
mov rcx, qword ptr [rbp + rbx*8]
dec rbx
mov rax, qword ptr [rbp + rbx*8]
cmp r11b, 0x2B
je op_add
cmp r11b, 0x2D
je op_sub
cmp r11b, 0x2A
je op_mul
cmp r11b, 0x2F
je op_div
jmp op_mod
op_add:
add rax, rcx
jmp op_push
op_sub:
sub rax, rcx
jmp op_push
op_mul:
imul rax, rcx
jmp op_push
op_div:
test rcx, rcx
jz line_error
cmp rcx, -1
jne div_ok
mov rdx, 1
shl rdx, 63
cmp rax, rdx
je line_error
div_ok:
cqo
idiv rcx
jmp op_push
op_mod:
test rcx, rcx
jz line_error
cmp rcx, -1
jne mod_ok
mov rdx, 1
shl rdx, 63
cmp rax, rdx
je line_error
mod_ok:
cqo
idiv rcx
mov rax, rdx
jmp op_push
op_push:
mov qword ptr [rbp + rbx*8], rax
inc rbx
jmp token_skip
tokens_done:
cmp rbx, 1
jne line_error
mov rax, qword ptr [rbp]
lea rdi, [rip + out_buf]
mov rsi, rdi
add rsi, 32
dec rsi
mov byte ptr [rsi], 0x0A
test rax, rax
jns conv_pos
mov r8b, 1
neg rax
jmp conv_digits
conv_pos:
mov r8b, 0
conv_digits:
mov rcx, 10
conv_loop:
xor rdx, rdx
div rcx
add dl, 0x30
dec rsi
mov byte ptr [rsi], dl
test rax, rax
jnz conv_loop
cmp r8b, 0
je conv_done
dec rsi
mov byte ptr [rsi], 0x2D
conv_done:
mov rdx, rdi
add rdx, 32
sub rdx, rsi
mov rdi, 1
mov rax, 1
syscall
jmp line_loop
line_error:
lea rsi, [rip + err_msg]
mov rdx, 6
mov rdi, 1
mov rax, 1
syscall
jmp line_loop
exit0:
mov rdi, 0
mov rax, 60
syscall
mov rax, 231
syscall
jmp exit0
