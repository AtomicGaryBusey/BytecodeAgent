_start:
  xor r12, r12
read_loop:
  lea rsi, [rip + buffer]
  add rsi, r12
  mov rax, 65536
  sub rax, r12
  test rax, rax
  jz read_done
  mov rdx, rax
  mov rdi, 0
  mov rax, 0
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_done:
  mov qword ptr [rip + input_len], r12
  xor eax, eax
  mov qword ptr [rip + cur_pos], rax
  call skip_ws
  mov rax, qword ptr [rip + cur_pos]
  mov rdx, qword ptr [rip + input_len]
  cmp rax, rdx
  jae handle_error
  call parse_value
  test rax, rax
  jnz handle_error
  call skip_ws
  mov rax, qword ptr [rip + cur_pos]
  mov rdx, qword ptr [rip + input_len]
  cmp rax, rdx
  jb handle_error
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + msg_ok]
  mov rdx, 3
  syscall
  mov rax, 60
  mov rdi, 0
  syscall
handle_error:
  mov rax, qword ptr [rip + cur_pos]
  lea rbx, [rip + numbuf + 32]
  mov rcx, 10
  xor r12, r12
conv_loop:
  xor rdx, rdx
  div rcx
  add dl, 0x30
  dec rbx
  mov byte ptr [rbx], dl
  inc r12
  test rax, rax
  jnz conv_loop
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + msg_err]
  mov rdx, 9
  syscall
  mov rax, 1
  mov rdi, 1
  mov rsi, rbx
  mov rdx, r12
  syscall
  mov rax, 1
  mov rdi, 1
  lea rsi, [rip + msg_nl]
  mov rdx, 1
  syscall
  mov rax, 60
  mov rdi, 1
  syscall
skip_ws:
  mov rax, qword ptr [rip + cur_pos]
  mov rdx, qword ptr [rip + input_len]
  lea rsi, [rip + buffer]
skip_ws_loop:
  cmp rax, rdx
  jae skip_ws_done
  movzx ecx, byte ptr [rsi + rax]
  cmp ecx, 0x20
  je skip_ws_next
  cmp ecx, 0x09
  je skip_ws_next
  cmp ecx, 0x0A
  je skip_ws_next
  cmp ecx, 0x0D
  je skip_ws_next
  jmp skip_ws_done
skip_ws_next:
  inc rax
  jmp skip_ws_loop
skip_ws_done:
  mov qword ptr [rip + cur_pos], rax
  ret
parse_string:
  mov rax, qword ptr [rip + cur_pos]
  mov rdx, qword ptr [rip + input_len]
  lea rsi, [rip + buffer]
  cmp rax, rdx
  jae parse_string_fail
  movzx ecx, byte ptr [rsi + rax]
  cmp ecx, 0x22
  jne parse_string_fail
  inc rax

  cmp rax, rdx
  jae parse_string_fail
  movzx ecx, byte ptr [rsi + rax]
  cmp ecx, 0x22
  je parse_string_close
  cmp ecx, 0x5C
  je parse_string_escape
  cmp ecx, 0x20
  jb parse_string_fail
  inc rax
  jmp parse_string_loop
parse_string_close:
  inc rax

  xor eax, eax
  ret

  je parse_string_esc_simple
  cmp ecx, 0x2F
  je parse_string_esc_simple
  cmp ecx, 0x62
  je parse_string_esc_simple
  cmp ecx, 0x66
  je parse_string_esc_simple
  cmp ecx, 0x6E
  je parse_string_esc_simple
  cmp ecx, 0x72
  je parse_string_esc_simple
  cmp ecx, 0x74
  je parse_string_esc_simple
  cmp ecx, 0x75
  je parse_string_esc_u
  jmp parse_string_fail
parse_string_esc_simple:
  inc rax
  jmp parse_string_loop
parse_string_esc_u:

parse_string_hex_loop:

  jae parse_string_fail
  movzx ecx, byte ptr [rsi + rax]
