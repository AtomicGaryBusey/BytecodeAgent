.text
.globl _start
_start:
  xor r12d, r12d
  lea r13, [rip + buffer]
  lea rbx, [rip + outbuf]
read_loop:
  mov edx, 65536
  sub edx, r12d
  jz read_done
  mov rsi, r13
  add rsi, r12
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  cmp r12, 65536
  jb read_loop
read_done:
  test r12, r12
  jz do_exit
  xor r14d, r14d
  xor r15d, r15d
outer:
  cmp r14, r12
  jae write_phase
  mov rsi, r14
scan:
  cmp rsi, r12
  jae scan_done
  mov al, byte ptr [r13 + rsi]
  cmp al, 10
  je scan_done
  inc rsi
  jmp scan
scan_done:
  mov rcx, rsi
  sub rcx, r14
  test rcx, rcx
  jz append_nl
  mov rdi, rbx
  add rdi, r15
  mov rdx, r13
  add rdx, rsi
  dec rdx
rev_loop:
  mov al, byte ptr [rdx]
  mov byte ptr [rdi], al
  inc rdi
  dec rdx
  dec rcx
  jnz rev_loop
append_nl:
  mov rcx, rsi
  sub rcx, r14
  mov rax, rbx
  add rax, r15
  add rax, rcx
  mov byte ptr [rax], 10
  inc rcx
  add r15, rcx
  cmp rsi, r12
  jae write_phase
  mov r14, rsi
  inc r14
  jmp outer
write_phase:
  xor r14d, r14d
  test r15, r15
  jz do_exit
write_loop:
  mov rsi, rbx
  add rsi, r14
  mov rdx, r15
  sub rdx, r14
  mov edi, 1
  mov eax, 1
  syscall
  add r14, rax
  cmp r14, r15
  jb write_loop
do_exit:
  xor edi, edi
  mov eax, 60
  syscall
hang:
  jmp hang
.bss
.align 16
buffer:
  .zero 65536
outbuf:
  .zero 65537
