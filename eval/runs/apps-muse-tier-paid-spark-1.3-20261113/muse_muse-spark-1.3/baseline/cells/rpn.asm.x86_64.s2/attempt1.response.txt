.globl _start
_start:

jz read_discard

no_nl:

eval_start:

token_loop:
cmp rdi, r10
jae tokens_done
movzx eax, byte ptr [rdi]
cmp al, 32
je is_sep
cmp al, 9
je is_sep
jmp token_found
is_sep:
inc rdi
jmp token_loop
token_found:
mov rsi, rdi
find_tok_end:
cmp rdi, r10
jae tok_end_found
movzx eax, byte ptr [rdi]
cmp al, 32
je tok_end_found
cmp al, 9
je tok_end_found
inc rdi
jmp find_tok_end
tok_end_found:
mov rcx, rdi
sub rcx, rsi
cmp rcx, 1
jne try_literal
movzx eax, byte ptr [rsi]
cmp al, 43
je do_operator
cmp al, 45
je do_operator
cmp al, 42
je do_operator
cmp al, 47
je do_operator
cmp al, 37
je do_operator

mov r8b, al

jb line_error
sub r9, 2
mov rax, qword ptr [r12 + r9*8]
mov rcx, qword ptr [r12 + r9*8 + 8]
cmp r8b, 43
je op_add
cmp r8b, 45
je op_sub
cmp r8b, 42
je op_mul
cmp r8b, 47
je op_div

add rax, rcx

inc r9
jmp token_loop
op_sub:
sub rax, rcx
mov qword ptr [r12 + r9*8], rax
inc r9
jmp token_loop
op_mul:

mov qword ptr [r12 + r9*8], rax
inc r9
jmp token_loop
op_div:
test rcx, rcx
jz line_error
cmp rcx, -1
jne div_no_overflow
mov r11d, 1
shl r11, 63

je line_error
div_no_overflow:
cqo
idiv rcx
mov qword ptr [r12 + r9*8], rax
inc r9
jmp token_loop
op_mod:
test rcx, rcx
jz line_error
cmp rcx, -1
jne mod_no_overflow
mov r11d, 1
shl r11, 63
cmp rax, r11
je line_error
mod_no_overflow:
cqo
idiv rcx
mov qword ptr [r12 + r9*8], rdx
inc r9
jmp token_loop
try_literal:
xor r8d, r8d
movzx eax, byte ptr [rsi]
cmp al, 45
jne lit_check_len
mov r8d, 1
inc rsi
dec rcx
jz line_error
lit_check_len:
test rcx, rcx
jz line_error
cmp rcx, 18
ja line_error
xor r11d, r11d
lit_digit_loop:
test rcx, rcx
jz lit_done
movzx eax, byte ptr [rsi]
cmp eax, 48
jb line_error
cmp eax, 57
ja line_error
sub eax, 48
imul r11, r11, 10
add r11, rax
inc rsi
dec rcx
jmp lit_digit_loop
lit_done:

jz lit_push

lit_push:
cmp r9, 32768
jae line_error
mov qword ptr [r12 + r9*8], r11
inc r9
jmp token_loop
tokens_done:
