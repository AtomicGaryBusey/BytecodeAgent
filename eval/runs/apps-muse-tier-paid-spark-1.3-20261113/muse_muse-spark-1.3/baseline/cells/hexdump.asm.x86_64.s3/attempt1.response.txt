.text
.globl _start
_start:
  xor r12d, r12d
read_loop:
  lea rsi, [rip + input_buf]
  add rsi, r12
  mov edx, 65536
  sub rdx, r12
  jz drain_loop
  mov eax, 0
  mov edi, 0
  syscall
  test rax, rax
  jz dump_begin
  js dump_begin
  add r12, rax
  jmp read_loop
drain_loop:
  lea rsi, [rip + dummy_byte]
  mov edx, 1
  mov eax, 0
  mov edi, 0
  syscall
  test rax, rax
  jz dump_begin
  js dump_begin
  jmp drain_loop
dump_begin:
  xor r13d, r13d
row_loop:
  cmp r13, r12
  jae final_line
  mov r14, r12
  sub r14, r13
  cmp r14, 16
  jbe n_ok
  mov r14d, 16
n_ok:
  lea rbx, [rip + line_buf]
  lea rbp, [rip + hexdigits]
  mov rdi, rbx
  mov r10d, 28
off_loop:
  mov rax, r13
  mov cl, r10b
  shr rax, cl
  and eax, 15
  mov al, byte ptr [rbp + rax]
  mov byte ptr [rdi], al
  inc rdi
  sub r10d, 4
  jns off_loop
  mov byte ptr [rdi], 0x20
  mov byte ptr [rdi + 1], 0x20
  add rdi, 2
  lea rsi, [rip + input_buf]
  add rsi, r13
  xor r8d, r8d
hex_loop:
  cmp r8d, r14d
  jae hex_space
  movzx r10d, byte ptr [rsi + r8]
  mov eax, r10d
  shr eax, 4
  and eax, 15
  mov al, byte ptr [rbp + rax]
  mov byte ptr [rdi], al
  mov eax, r10d
  and eax, 15
  mov al, byte ptr [rbp + rax]
  mov byte ptr [rdi + 1], al
  mov byte ptr [rdi + 2], 0x20
  add rdi, 3
  jmp hex_after
hex_space:
  mov byte ptr [rdi], 0x20
  mov byte ptr [rdi + 1], 0x20
  mov byte ptr [rdi + 2], 0x20
  add rdi, 3
hex_after:
  cmp r8d, 7
  jne hex_no_extra
  mov byte ptr [rdi], 0x20
  inc rdi
hex_no_extra:
  inc r8d
  cmp r8d, 16
  jb hex_loop
  mov byte ptr [rdi], 0x20
  mov byte ptr [rdi + 1], 0x7C
  add rdi, 2
  xor r8d, r8d
ascii_loop:
  cmp r8d, r14d
  jae ascii_done
  movzx eax, byte ptr [rsi + r8]
  cmp al, 0x20
  jb ascii_dot
  cmp al, 0x7E
  ja ascii_dot
  jmp ascii_store
ascii_dot:
  mov al, 0x2E
ascii_store:
  mov byte ptr [rdi], al
  inc rdi
  inc r8d
  jmp ascii_loop
ascii_done:
  mov byte ptr [rdi], 0x7C
  mov byte ptr [rdi + 1], 0x0A
  add rdi, 2
  mov rdx, rdi
  sub rdx, rbx
  mov rsi, rbx
  mov edi, 1
  mov eax, 1
  syscall
  add r13, r14
  jmp row_loop
final_line:
  lea rbx, [rip + line_buf]
  lea rbp, [rip + hexdigits]
  mov rdi, rbx
  mov r10d, 28
final_loop:
  mov rax, r12
  mov cl, r10b
  shr rax, cl
  and eax, 15
  mov al, byte ptr [rbp + rax]
  mov byte ptr [rdi], al
  inc rdi
  sub r10d, 4
  jns final_loop
  mov byte ptr [rdi], 0x0A
  inc rdi
  mov rdx, rdi
  sub rdx, rbx
  mov rsi, rbx
  mov edi, 1
  mov eax, 1
  syscall
  mov eax, 60
  xor edi, edi
  syscall
hang:
  jmp hang
.section .rodata
hexdigits:
.ascii "0123456789abcdef"
.bss
.align 16
input_buf:
.zero 65536
line_buf:
.zero 80
dummy_byte:
.zero 1
