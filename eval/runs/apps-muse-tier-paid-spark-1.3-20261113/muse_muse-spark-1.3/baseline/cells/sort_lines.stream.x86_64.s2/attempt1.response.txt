.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ INPUT_CAP, 32768
.equ MAX_LINES, 1000
.equ DISCARD_CAP, 1024
.text
.globl _start
_start:
  lea r14, [rip + input_buf]
  lea r15, [rip + offs]
  lea rbx, [rip + lens]
  xor r12d, r12d
  xor r13d, r13d
read_loop:
  mov eax, INPUT_CAP
  sub rax, r12
  test rax, rax
  jz discard_read
  mov rsi, r14
  add rsi, r12
  mov rdx, rax
  mov edi, STDIN
  mov eax, SYS_READ
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
discard_read:
  lea rsi, [rip + discard_buf]
  mov edx, DISCARD_CAP
  mov edi, STDIN
  mov eax, SYS_READ
  syscall
  test rax, rax
  jz read_done
  js read_done
  jmp discard_read
read_done:
  test r12, r12
  jz no_lines
  xor r8d, r8d
  xor r9d, r9d
  xor r13d, r13d
parse_loop:
  cmp r9, r12
  jae parse_tail
  movzx eax, byte ptr [r14 + r9]
  cmp al, 10
  jne parse_next
  cmp r13, MAX_LINES
  jae parse_skip_store
  mov r10, r9
  sub r10, r8
  mov qword ptr [r15 + r13*8], r8
  mov qword ptr [rbx + r13*8], r10
  inc r13
parse_skip_store:
  mov r8, r9
  inc r8
parse_next:
  inc r9
  jmp parse_loop
parse_tail:
  cmp r8, r12
  jae parse_done
  cmp r13, MAX_LINES
  jae parse_done
  mov r10, r12
  sub r10, r8
  mov qword ptr [r15 + r13*8], r8
  mov qword ptr [rbx + r13*8], r10
  inc r13
parse_done:
  cmp r13, 1
  jbe output_phase
  mov r12, r13
  shr r12, 1
gap_loop:
  test r12, r12
  jz output_phase
  mov r10, r12
i_loop:
  cmp r10, r13
  jae gap_next
  mov r11, r10
j_loop:
  cmp r11, r12
  jb i_next
  mov r8, r11
  sub r8, r12
  mov rax, qword ptr [r15 + r11*8]
  mov rcx, qword ptr [rbx + r11*8]
  mov rdi, qword ptr [r15 + r8*8]
  mov rsi, qword ptr [rbx + r8*8]
  add rax, r14
  add rdi, r14
  xor r9d, r9d
cmp_loop:
  cmp r9, rcx
  jae compare_lengths
  cmp r9, rsi
  jae compare_lengths
  movzx edx, byte ptr [rax + r9]
  movzx ebp, byte ptr [rdi + r9]
  cmp edx, ebp
  jb need_swap
  ja no_swap
  inc r9
  jmp cmp_loop
compare_lengths:
  cmp rcx, rsi
  jb need_swap
  jmp no_swap
need_swap:
  mov rax, qword ptr [r15 + r11*8]
  mov rcx, qword ptr [r15 + r8*8]
  mov qword ptr [r15 + r11*8], rcx
  mov qword ptr [r15 + r8*8], rax
  mov rax, qword ptr [rbx + r11*8]
  mov rcx, qword ptr [rbx + r8*8]
  mov qword ptr [rbx + r11*8], rcx
  mov qword ptr [rbx + r8*8], rax
  mov r11, r8
  jmp j_loop
no_swap:
i_next:
  inc r10
  jmp i_loop
gap_next:
  shr r12, 1
  jmp gap_loop
output_phase:
  test r13, r13
  jz exit0
  xor r12d, r12d
out_loop:
  cmp r12, r13
  jae exit0
  mov rax, qword ptr [r15 + r12*8]
  mov r8, qword ptr [rbx + r12*8]
  test r8, r8
  jz write_nl
  mov r9, r14
  add r9, rax
write_content_loop:
  test r8, r8
  jz write_nl
  mov eax, SYS_WRITE
  mov edi, STDOUT
  mov rsi, r9
  mov rdx, r8
  syscall
  test rax, rax
  jle write_nl
  sub r8, rax
  add r9, rax
  jmp write_content_loop
write_nl:
  mov eax, SYS_WRITE
  mov edi, STDOUT
  lea rsi, [rip + newline]
  mov edx, 1
  syscall
  inc r12
  jmp out_loop
no_lines:
  xor r13d, r13d
exit0:
  mov eax, SYS_EXIT
  mov edi, 0
  syscall
hang:
  jmp hang
.section .rodata
newline:
  .byte 10
.bss
.align 8
input_buf:
  .zero 32768
offs:
  .zero 8000
lens:
  .zero 8000
discard_buf:
  .zero 1024
