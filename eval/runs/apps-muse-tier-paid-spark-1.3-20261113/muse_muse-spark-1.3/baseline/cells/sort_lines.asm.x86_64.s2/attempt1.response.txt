.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ SYS_EXIT_GROUP, 231
.equ STDOUT, 1
.equ INPUT_MAX, 32768
.equ MAX_LINES, 1000
.text
.globl _start
_start:
  lea r15, [rip + input_buf]
  lea r14, [rip + line_entries]
  xor r12d, r12d
read_loop:
  mov eax, INPUT_MAX
  sub eax, r12d
  jnz read_have_space
  lea rsi, [rip + out_buf]
  mov edx, 1024
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz read_done
  js read_done
  jmp read_loop
read_have_space:
  mov edx, eax
  lea rsi, [r15 + r12]
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jz read_done
  js read_done
  add r12, rax
  jmp read_loop
read_done:
  xor ebx, ebx
  xor ecx, ecx
  xor r13d, r13d
  test r12, r12
  jz lines_done
parse_loop:
  cmp rbx, r12
  jae parse_tail
  movzx eax, byte ptr [r15 + rbx]
  cmp al, 10
  jne parse_next
  cmp r13, MAX_LINES
  jae parse_newline_skip
  mov eax, ebx
  sub eax, ecx
  mov dword ptr [r14 + r13*8], ecx
  mov dword ptr [r14 + r13*8 + 4], eax
  inc r13
parse_newline_skip:
  inc rbx
  mov rcx, rbx
  jmp parse_loop
parse_next:
  inc rbx
  jmp parse_loop
parse_tail:
  cmp rcx, r12
  jae lines_done
  cmp r13, MAX_LINES
  jae lines_done
  mov eax, r12d
  sub eax, ecx
  mov dword ptr [r14 + r13*8], ecx
  mov dword ptr [r14 + r13*8 + 4], eax
  inc r13
lines_done:
  cmp r13, 2
  jb sort_done
  mov r12, r13
  shr r12, 1
gap_loop:
  test r12, r12
  jz sort_done
  mov rbx, r12
i_loop:
  cmp rbx, r13
  jae next_gap
  mov r10, qword ptr [r14 + rbx*8]
  mov rbp, rbx
j_loop:
  cmp rbp, r12
  jb store_temp
  mov rax, rbp
  sub rax, r12
  mov r8, qword ptr [r14 + rax*8]
  mov eax, r8d
  shr r8, 32
  mov ecx, r10d
  mov r9, r10
  shr r9, 32
  lea rsi, [r15 + rax]
  lea rdi, [r15 + rcx]
  mov rdx, r8
  cmp rdx, r9
  cmova rdx, r9
  xor eax, eax
cmp_loop:
  cmp rax, rdx
  jae cmp_prefix_equal
  movzx r11d, byte ptr [rsi + rax]
  movzx ecx, byte ptr [rdi + rax]
  cmp r11d, ecx
  jne cmp_bytes_differ
  inc rax
  jmp cmp_loop
cmp_bytes_differ:
  ja do_shift
  jmp store_temp
cmp_prefix_equal:
  cmp r8, r9
  je store_temp
  ja do_shift
  jmp store_temp
do_shift:
  mov rax, rbp
  sub rax, r12
  mov r8, qword ptr [r14 + rax*8]
  mov qword ptr [r14 + rbp*8], r8
  sub rbp, r12
  jmp j_loop
store_temp:
  mov qword ptr [r14 + rbp*8], r10
  inc rbx
  jmp i_loop
next_gap:
  shr r12, 1
  jmp gap_loop
sort_done:
  lea r12, [rip + out_buf]
  xor ebx, ebx
  xor ebp, ebp
  test r13, r13
  jz write_setup
copy_loop:
  cmp rbp, r13
  jae write_setup
  mov rax, qword ptr [r14 + rbp*8]
  mov r8, rax
  shr r8, 32
  mov r9d, eax
  lea rsi, [r15 + r9]
  lea rdi, [r12 + rbx]
  xor ecx, ecx
copy_byte_loop:
  cmp rcx, r8
  jae copy_line_done
  movzx eax, byte ptr [rsi + rcx]
  mov byte ptr [rdi + rcx], al
  inc rcx
  jmp copy_byte_loop
copy_line_done:
  add rbx, r8
  mov byte ptr [r12 + rbx], 10
  inc rbx
  inc rbp
  jmp copy_loop
write_setup:
  test rbx, rbx
  jz do_exit
  xor r13d, r13d
write_loop:
  cmp r13, rbx
  jae do_exit
  mov eax, SYS_WRITE
  mov edi, STDOUT
  lea rsi, [r12 + r13]
  mov rdx, rbx
  sub rdx, r13
  syscall
  test rax, rax
  jle do_exit
  add r13, rax
  jmp write_loop
do_exit:
  mov eax, SYS_EXIT
  xor edi, edi
  syscall
  mov eax, SYS_EXIT_GROUP
  xor edi, edi
  syscall
exit_hang:
  jmp exit_hang
.bss
.balign 8
input_buf:
.zero 32768
.balign 8
line_entries:
.zero 8000
.balign 8
out_buf:
.zero 33768
