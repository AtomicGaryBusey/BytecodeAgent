.globl _start
.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz done_read
    js read_loop
    mov rcx, rax
    lea rsi, [rip + buffer]
proc_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    inc r14
    cmp al, 10
    jne not_nl
    inc r12
not_nl:
    cmp al, 32
    je is_ws
    sub al, 9
    cmp al, 4
    jbe is_ws
    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
    jmp next_byte
is_ws:
    xor r15, r15
next_byte:
    dec rcx
    jnz proc_loop
    jmp read_loop
done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call print_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call print_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
exit_loop:
    mov eax, 60
    xor edi, edi
    syscall
    jmp exit_loop
print_num:
    test rax, rax
    jnz pn_start
    mov byte ptr [rdi], 48
    inc rdi
    ret
pn_start:
    xor ecx, ecx
    mov ebx, 10
pn_div:
    xor edx, edx
    div rbx
    push rdx
    inc rcx
    test rax, rax
    jnz pn_div
pn_pop:
    pop rax
    add al, 48
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz pn_pop
    ret
.bss
.align 16
buffer:
    .zero 4096
outbuf:
    .zero 128
