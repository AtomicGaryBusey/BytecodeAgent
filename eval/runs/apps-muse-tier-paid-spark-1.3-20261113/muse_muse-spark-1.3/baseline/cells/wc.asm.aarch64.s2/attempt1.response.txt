.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ FD_STDIN, 0
.equ FD_STDOUT, 1
.equ BUFSIZE, 4096
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0
    adrp x25, inbuf
    add x25, x25, :lo12:inbuf
read_loop:
    mov x0, #FD_STDIN
    mov x1, x25
    mov x2, #BUFSIZE
    mov x8, #SYS_READ
    svc #0
    cbz x0, done_read
    mov x23, x25
    add x24, x25, x0
byte_loop:
    ldrb w3, [x23], #1
    add x21, x21, #1
    cmp w3, #10
    b.ne check_ws
    add x19, x19, #1
check_ws:
    cmp w3, #32
    b.eq got_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls got_ws
    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte
got_ws:
    mov x22, #0
next_byte:
    cmp x23, x24
    b.lo byte_loop
    b read_loop
done_read:
    adrp x26, outbuf
    add x26, x26, :lo12:outbuf
    mov x1, x26
    mov x0, x19
    bl print_uint
    mov w2, #32
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x20
    bl print_uint
    mov w2, #32
    strb w2, [x0], #1
    mov x1, x0
    mov x0, x21
    bl print_uint
    mov w2, #10
    strb w2, [x0], #1
    sub x2, x0, x26
    mov x1, x26
    mov x0, #FD_STDOUT
    mov x8, #SYS_WRITE
    svc #0
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
print_uint:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    sub sp, sp, #32
    cbnz x0, print_nonzero
    mov w2, #48
    strb w2, [x1], #1
    mov x0, x1
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret
print_nonzero:
    mov x2, x0
    mov x3, #0
    mov x4, sp
    mov x5, #10
div_loop:
    udiv x6, x2, x5
    msub x7, x6, x5, x2
    add w7, w7, #48
    add x8, x4, x3
    strb w7, [x8]
    add x3, x3, #1
    mov x2, x6
    cbnz x2, div_loop
copy_loop:
    sub x3, x3, #1
    add x8, x4, x3
    ldrb w7, [x8]
    strb w7, [x1], #1
    cbnz x3, copy_loop
    mov x0, x1
    add sp, sp, #32
    ldp x29, x30, [sp], #16
    ret
.bss
.balign 16
inbuf:
    .zero 4096
outbuf:
    .zero 64
