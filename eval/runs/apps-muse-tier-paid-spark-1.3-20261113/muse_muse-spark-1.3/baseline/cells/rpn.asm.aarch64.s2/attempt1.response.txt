.text
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.globl _start
_start:
  adrp x19, input_buf
  add x19, x19, :lo12:input_buf
  adrp x22, eval_stack
  add x22, x22, :lo12:eval_stack
  adrp x23, out_buf
  add x23, x23, :lo12:out_buf
  adrp x24, rev_buf
  add x24, x24, :lo12:rev_buf
  mov x9, #0
read_loop:
  movz x10, #1, lsl #16
  sub x10, x10, x9
  cbz x10, read_done
  add x1, x19, x9
  mov x0, #0
  mov x2, x10
  mov x8, #SYS_READ
  svc #0
  cbz x0, read_done
  add x9, x9, x0
  b read_loop
read_done:
  add x20, x19, x9
  mov x21, x19
line_loop:
  cmp x21, x20
  b.eq exit_ok
  mov x10, x21
find_nl:
  cmp x10, x20
  b.eq no_nl
  ldrb w11, [x10]
  cmp w11, #0x0A
  b.eq found_nl
  add x10, x10, #1
  b find_nl
found_nl:
  mov x25, x10
  add x26, x10, #1
  b eval_start
no_nl:
  mov x25, x20
  mov x26, x20
eval_start:
  mov x27, #0
  mov x28, x21
token_skip:
  cmp x28, x25
  b.eq end_tokens
  ldrb w10, [x28]
  cmp w10, #0x20
  b.eq token_skip_adv
  cmp w10, #0x09
  b.ne token_begin
token_skip_adv:
  add x28, x28, #1
  b token_skip
token_begin:
  mov x9, x28
tok_end_loop:
  cmp x9, x25
  b.eq tok_end_found
  ldrb w10, [x9]
  cmp w10, #0x20
  b.eq tok_end_found
  cmp w10, #0x09
  b.eq tok_end_found
  add x9, x9, #1
  b tok_end_loop
tok_end_found:
  sub x10, x9, x28
  cmp x10, #1
  b.ne parse_literal
  ldrb w11, [x28]
  cmp w11, #0x2B
  b.eq do_op
  cmp w11, #0x2D
  b.eq do_op
  cmp w11, #0x2A
  b.eq do_op
  cmp w11, #0x2F
  b.eq do_op
  cmp w11, #0x25
  b.eq do_op
  b parse_literal
parse_literal:
  ldrb w11, [x28]
  cmp w11, #0x2D
  b.eq lit_negative
  cmp x10, #18
  b.hi error_line
  mov x11, x28
  mov x12, x10
  mov x13, #0
  mov x14, #0
  b lit_setup
lit_negative:
  sub x12, x10, #1
  cbz x12, error_line
  cmp x12, #18
  b.hi error_line
  add x11, x28, #1
  mov x13, #0
  mov x14, #1
lit_setup:
  add x16, x11, x12
  mov x15, x11
lit_loop:
  cmp x15, x16
  b.eq lit_done
  ldrb w17, [x15]
  add x15, x15, #1
  sub w17, w17, #0x30
  cmp w17, #10
  b.hs error_line
  mov x18, #10
  mul x13, x13, x18
  add x13, x13, x17
  b lit_loop
lit_done:
  cmp x14, #1
  b.ne lit_push
  neg x13, x13
lit_push:
  str x13, [x22, x27, lsl #3]
  add x27, x27, #1
  mov x28, x9
  b token_skip
do_op:
  mov w12, w11
  cmp x27, #2
  b.lo error_line
  sub x27, x27, #1
  ldr x14, [x22, x27, lsl #3]
  sub x27, x27, #1
  ldr x13, [x22, x27, lsl #3]
  cmp w12, #0x2B
  b.eq op_add
  cmp w12, #0x2D
  b.eq op_sub
  cmp w12, #0x2A
  b.eq op_mul
  cmp w12, #0x2F
  b.eq op_div
  b op_mod
op_add:
  add x15, x13, x14
  b op_push
op_sub:
  sub x15, x13, x14
  b op_push
op_mul:
  mul x15, x13, x14
  b op_push
op_div:
  cbz x14, error_line
  movz x16, #0x8000, lsl #48
  cmn x14, #1
  b.ne op_div_ok
  cmp x13, x16
  b.eq error_line
op_div_ok:
  sdiv x15, x13, x14
  b op_push
op_mod:
  cbz x14, error_line
  movz x16, #0x8000, lsl #48
  cmn x14, #1
  b.ne op_mod_ok
  cmp x13, x16
  b.eq error_line
op_mod_ok:
  sdiv x16, x13, x14
  msub x15, x16, x14, x13
  b op_push
op_push:
  str x15, [x22, x27, lsl #3]
  add x27, x27, #1
  mov x28, x9
  b token_skip
end_tokens:
  cmp x27, #1
  b.ne error_line
  ldr x10, [x22]
  cmp x10, #0
  b.ge print_pos
  mov w11, #0x2D
  strb w11, [x23]
  mov x11, #1
  neg x12, x10
  b print_convert
print_pos:
  mov x11, #0
  mov x12, x10
print_convert:
  mov x13, #0
  mov x14, #10
div_loop:
  udiv x15, x12, x14
  msub x16, x15, x14, x12
  add x16, x16, #0x30
  add x17, x24, x13
  strb w16, [x17]
  add x13, x13, #1
  mov x12, x15
  cbnz x12, div_loop
  mov x15, #0
copy_loop:
  cmp x15, x13
  b.eq copy_done
  sub x16, x13, x15
  sub x16, x16, #1
  add x18, x24, x16
  ldrb w16, [x18]
  add x18, x11, x15
  add x18, x23, x18
  strb w16, [x18]
  add x15, x15, #1
  b copy_loop
copy_done:
  add x16, x11, x13
  add x17, x23, x16
  mov w18, #0x0A
  strb w18, [x17]
  add x16, x16, #1
  mov x0, #1
  mov x1, x23
  mov x2, x16
  mov x8, #SYS_WRITE
  svc #0
  b next_line
error_line:
  adrp x1, err_msg
  add x1, x1, :lo12:err_msg
  mov x0, #1
  mov x2, #6
  mov x8, #SYS_WRITE
  svc #0
  b next_line
next_line:
  mov x21, x26
  b line_loop
exit_ok:
  mov x0, #0
  mov x8, #SYS_EXIT
  svc #0
.section .rodata
err_msg:
  .ascii "error\n"
.bss
.balign 8
input_buf:
  .zero 65536
eval_stack:
  .zero 8192
out_buf:
  .zero 32
rev_buf:
  .zero 32
