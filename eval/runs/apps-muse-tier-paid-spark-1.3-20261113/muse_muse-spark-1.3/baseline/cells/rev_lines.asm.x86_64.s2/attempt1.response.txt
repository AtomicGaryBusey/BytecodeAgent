.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ STDIN, 0
.equ STDOUT, 1
.equ IN_CAP, 65536
.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov eax, IN_CAP
    sub eax, r12d
    jz done_read
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, eax
    mov edi, STDIN
    mov eax, SYS_READ
    syscall
    test rax, rax
    jz done_read
    add r12, rax
    jmp read_loop
done_read:
    test r12, r12
    jz exit_ok
    xor r13d, r13d
    xor r14d, r14d
    lea r15, [rip + inbuf]
    lea rbx, [rip + outbuf]
line_loop:
    cmp r13, r12
    jae done_lines
    mov rsi, r13
    mov rdi, r13
find_nl:
    cmp rdi, r12
    jae have_line
    mov al, byte ptr [r15 + rdi]
    cmp al, 10
    je have_line
    inc rdi
    jmp find_nl
have_line:
    mov rcx, rdi
    sub rcx, rsi
    test rcx, rcx
    jz emit_nl
    xor edx, edx
copy_loop:
    mov r8, rdi
    dec r8
    sub r8, rdx
    mov al, byte ptr [r15 + r8]
    mov r9, r14
    add r9, rdx
    mov byte ptr [rbx + r9], al
    inc rdx
    cmp rdx, rcx
    jb copy_loop
    add r14, rcx
emit_nl:
    mov byte ptr [rbx + r14], 10
    inc r14
    cmp rdi, r12
    jae done_lines
    lea r13, [rdi + 1]
    jmp line_loop
done_lines:
    xor r13d, r13d
write_loop:
    cmp r13, r14
    jae exit_ok
    mov rax, r14
    sub rax, r13
    mov rsi, rbx
    add rsi, r13
    mov rdx, rax
    mov edi, STDOUT
    mov eax, SYS_WRITE
    syscall
    add r13, rax
    jmp write_loop
exit_ok:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
    jmp exit_ok
.bss
inbuf:
    .zero 65536
outbuf:
    .zero 65537
