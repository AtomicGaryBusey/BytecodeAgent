    .equ IN_BUF_SIZE, 65600
    .equ OUT_BUF_SIZE, 8192
    .equ RPN_STACK_CAP, 1024

    .text
    .globl _start

_start:
    adrp x19, in_buf
    add  x19, x19, :lo12:in_buf
    mov  x20, x19
    movz x24, #0x0040
    movk x24, #0x0001, lsl #16
    mov  x25, #0

read_loop:
    sub  x2, x24, x25
    cbz  x2, read_done
    mov  x0, #0
    mov  x1, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    add  x25, x25, x0
    b    read_loop

read_done:
    adrp x21, out_buf
    add  x21, x21, :lo12:out_buf
    mov  x22, #0
    adrp x23, rpn_stack
    add  x23, x23, :lo12:rpn_stack

line_loop:
    cmp  x19, x20
    b.hs all_done
    mov  x0, x19

find_nl:
    cmp  x19, x20
    b.eq line_no_nl
    ldrb w2, [x19]
    cmp  w2, #0x0A
    b.eq line_found_nl
    add  x19, x19, #1
    b    find_nl

line_found_nl:
    mov  x1, x19
    add  x19, x19, #1
    b    do_eval

line_no_nl:
    mov  x1, x19

do_eval:
    bl   eval_line
    b    line_loop

all_done:
    bl   flush_out
    mov  x0, #0
    mov  x8, #93
    svc  #0

// eval_line(x0 = line_start, x1 = line_end)
eval_line:
    mov  x9, x0
    mov  x10, x1
    mov  x11, #0

tok_skip_ws:
    cmp  x9, x10
    b.hs line_finished
    ldrb w2, [x9]
    cmp  w2, #0x20
    b.eq 1f
    cmp  w2, #0x09
    b.ne tok_found
1:
    add  x9, x9, #1
    b    tok_skip_ws

tok_found:
    mov  x12, x9

tok_scan_end:
    add  x9, x9, #1
    cmp  x9, x10
    b.hs tok_got_end
    ldrb w2, [x9]
    cmp  w2, #0x20
    b.eq tok_got_end
    cmp  w2, #0x09
    b.ne tok_scan_end

tok_got_end:
    mov  x13, x9
    sub  x14, x13, x12
    cmp  x14, #1
    b.ne parse_number

    ldrb w2, [x12]
    cmp  w2, #'+'
    b.eq op_add
    cmp  w2, #'-'
    b.eq op_sub
    cmp  w2, #'*'
    b.eq op_mul
    cmp  w2, #'/'
    b.eq op_div
    cmp  w2, #'%'
    b.eq op_mod

    sub  w3, w2, #'0'
    cmp  w3, #9
    b.hi line_error
    str  x3, [x23, x11, lsl #3]
    add  x11, x11, #1
    b    tok_skip_ws

parse_number:
    mov  x15, x12
    mov  x16, #0
    ldrb w2, [x15]
    cmp  w2, #'-'
    b.ne 1f
    mov  x16, #1
    add  x15, x15, #1
1:
    sub  x14, x13, x15
    cmp  x14, #1
    b.lt line_error
    cmp  x14, #18
    b.gt line_error
    mov  x0, #0
    mov  x17, #10
2:
    ldrb w2, [x15], #1
    sub  w3, w2, #'0'
    cmp  w3, #9
    b.hi line_error
    mul  x0, x0, x17
    add  x0, x0, x3
    cmp  x15, x13
    b.lo 2b

    cbz  x16, 3f
    neg  x0, x0
3:
    str  x0, [x23, x11, lsl #3]
    add  x11, x11, #1
    b    tok_skip_ws

op_add:
    cmp  x11, #2
    b.lt line_error
    sub  x11, x11, #1
    sub  x14, x11, #1
    ldr  x0, [x23, x14, lsl #3]
    ldr  x1, [x23, x11, lsl #3]
    add  x0, x0, x1
    str  x0, [x23, x14, lsl #3]
    b    tok_skip_ws

op_sub:
    cmp  x11, #2
    b.lt line_error
    sub  x11, x11, #1
    sub  x14, x11, #1
    ldr  x0, [x23, x14, lsl #3]
    ldr  x1, [x23, x11, lsl #3]
    sub  x0, x0, x1
    str  x0, [x23, x14, lsl #3]
    b    tok_skip_ws

op_mul:
    cmp  x11, #2
    b.lt line_error
    sub  x11, x11, #1
    sub  x14, x11, #1
    ldr  x0, [x23, x14, lsl #3]
    ldr  x1, [x23, x11, lsl #3]
    mul  x0, x0, x1
    str  x0, [x23, x14, lsl #3]
    b    tok_skip_ws

op_div:
    cmp  x11, #2
    b.lt line_error
    sub  x11, x11, #1
    sub  x14, x11, #1
    ldr  x0, [x23, x14, lsl #3]
    ldr  x1, [x23, x11, lsl #3]
    cbz  x1, line_error
    movz x2, #0x8000, lsl #48
    cmp  x0, x2
    b.ne 1f
    cmn  x1, #1
    b.eq line_error
1:
    sdiv x0, x0, x1
    str  x0, [x23, x14, lsl #3]
    b    tok_skip_ws

op_mod:
    cmp  x11, #2
    b.lt line_error
    sub  x11, x11, #1
    sub  x14, x11, #1
    ldr  x0, [x23, x14, lsl #3]
    ldr  x1, [x23, x11, lsl #3]
    cbz  x1, line_error
    movz x2, #0x8000, lsl #48
    cmp  x0, x2
    b.ne 1f
    cmn  x1, #1
    b.eq line_error
1:
    sdiv x2, x0, x1
    msub x0, x2, x1, x0
    str  x0, [x23, x14, lsl #3]
    b    tok_skip_ws

line_finished:
    cmp  x11, #1
    b.ne line_error
    ldr  x0, [x23]
    b    emit_i64

line_error:
    b    emit_error

emit_error:
    stp  x29, x30, [sp, #-16]!
    add  x0, x22, #6
    mov  x1, #OUT_BUF_SIZE
    cmp  x0, x1
    b.ls 1f
    bl   flush_out
1:
    add  x3, x21, x22
    mov  w0, #'e'
    strb w0, [x3]
    mov  w0, #'r'
    strb w0, [x3, #1]
    strb w0, [x3, #2]
    strb w0, [x3, #4]
    mov  w0, #'o'
    strb w0, [x3, #3]
    mov  w0, #'\n'
    strb w0, [x3, #5]
    add  x22, x22, #6
    ldp  x29, x30, [sp], #16
    ret

emit_i64:
    stp  x29, x30, [sp, #-48]!
    mov  x9, x0
    add  x0, x22, #24
    mov  x1, #OUT_BUF_SIZE
    cmp  x0, x1
    b.ls 1f
    bl   flush_out
1:
    add  x3, x21, x22
    cmp  x9, #0
    b.ge 2f
    mov  w0, #'-'
    strb w0, [x3], #1
    neg  x9, x9
2:
    add  x4, sp, #16
    mov  x5, x4
    mov  x10, #10
3:
    udiv x1, x9, x10
    msub x2, x1, x10, x9
    add  w2, w2, #'0'
    strb w2, [x5], #1
    mov  x9, x1
    cbnz x9, 3b
4:
    ldrb w0, [x5, #-1]!
    strb w0, [x3], #1
    cmp  x5, x4
    b.ne 4b
    mov  w0, #'\n'
    strb w0, [x3], #1
    sub  x22, x3, x21
    ldp  x29, x30, [sp], #48
    ret

flush_out:
    cbz  x22, flush_ret
    mov  x3, x21
    mov  x4, x22
flush_loop:
    mov  x0, #1
    mov  x1, x3
    mov  x2, x4
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le flush_done
    add  x3, x3, x0
    sub  x4, x4, x0
    cbnz x4, flush_loop
flush_done:
    mov  x22, #0
flush_ret:
    ret

    .bss
    .p2align 4
in_buf:
    .space IN_BUF_SIZE
    .p2align 4
out_buf:
    .space OUT_BUF_SIZE
    .p2align 4
rpn_stack:
    .space RPN_STACK_CAP * 8
