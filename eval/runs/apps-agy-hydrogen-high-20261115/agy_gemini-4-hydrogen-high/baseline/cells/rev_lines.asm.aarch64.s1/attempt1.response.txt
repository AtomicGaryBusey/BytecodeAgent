    .text
    .globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    movz x21, #0x0100
    movk x21, #1, lsl #16

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x20, x19, x20
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, x21
    mov x9, x19

process_loop:
    cmp x9, x20
    b.hs process_done
    mov x10, x9

find_nl:
    cmp x10, x20
    b.hs no_nl
    ldrb w11, [x10]
    cmp w11, #10
    b.eq found_nl
    add x10, x10, #1
    b find_nl

found_nl:
    add x12, x10, #1
    b reverse_copy

no_nl:
    mov x12, x20

reverse_copy:
    mov x13, x10

rev_loop:
    cmp x13, x9
    b.ls rev_done
    ldrb w11, [x13, #-1]!
    strb w11, [x22], #1
    b rev_loop

rev_done:
    mov w11, #10
    strb w11, [x22], #1
    mov x9, x12
    b process_loop

process_done:
write_loop:
    sub x2, x22, x21
    cbz x2, write_done
    mov x0, #1
    mov x1, x21
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_done
    add x21, x21, x0
    b write_loop

write_done:
    mov x0, #0
    mov x8, #93
    svc #0

    .bss
    .balign 16

outbuf:
