.globl _start

.text
_start:
    xor r12d, r12d
    lea r13, [rip + inbuf]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, 65552
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov rsi, r13
    add r13, r12
    lea r8, [rip + prog]
    lea r9, [rip + cmd_map]
    lea r10, [rip + stack_br]
    lea r11, [rip + jump_tab]
    xor edi, edi
    xor ebx, ebx

parse_loop:
    cmp rsi, r13
    jae parse_no_bang
    movzx eax, byte ptr [rsi]
    inc rsi
    movzx eax, byte ptr [r9 + rax]
    cmp al, 8
    jb parse_cmd
    je parse_bang
    jmp parse_loop

parse_cmd:
    mov byte ptr [r8 + rdi], al
    cmp al, 6
    je parse_lbr
    cmp al, 7
    je parse_rbr
    inc rdi
    jmp parse_loop

parse_lbr:
    mov dword ptr [r10 + rbx*4], edi
    inc rbx
    inc rdi
    jmp parse_loop

parse_rbr:
    test rbx, rbx
    jz exit_err
    dec rbx
    mov edx, dword ptr [r10 + rbx*4]
    lea eax, [rdi + 1]
    mov dword ptr [r11 + rdx*4], eax
    inc edx
    mov dword ptr [r11 + rdi*4], edx
    inc rdi
    jmp parse_loop

parse_bang:
    mov r14, rsi
    mov r15, r13
    jmp parse_done

parse_no_bang:
    mov r14, r13
    mov r15, r13

parse_done:
    test rbx, rbx
    jnz exit_err
    mov byte ptr [r8 + rdi], 8

    lea r9, [rip + dispatch_table]
    lea r10, [rip + outbuf]
    lea rbp, [rip + tape]
    xor ecx, ecx
    xor ebx, ebx
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_plus:
    inc byte ptr [rbp]
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_minus:
    dec byte ptr [rbp]
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_right:
    inc rbp
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_left:
    dec rbp
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_dot:
    mov al, byte ptr [rbp]
    mov byte ptr [r10 + rbx], al
    inc rbx
    cmp rbx, 65536
    je flush_out
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_comma:
    xor eax, eax
    cmp r14, r15
    jae 1f
    movzx eax, byte ptr [r14]
    inc r14
1:
    mov byte ptr [rbp], al
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_lbr:
    cmp byte ptr [rbp], 0
    jne 1f
    mov ecx, dword ptr [r11 + rcx*4 - 4]
1:
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_rbr:
    cmp byte ptr [rbp], 0
    je 1f
    mov ecx, dword ptr [r11 + rcx*4 - 4]
1:
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

flush_out:
    push rcx
    xor r12d, r12d
flush_loop:
    cmp r12, rbx
    jae flush_done
    mov eax, 1
    mov edi, 1
    lea rsi, [r10 + r12]
    mov rdx, rbx
    sub rdx, r12
    syscall
    test rax, rax
    jle flush_done
    add r12, rax
    jmp flush_loop
flush_done:
    xor ebx, ebx
    pop rcx
    lea r11, [rip + jump_tab]
    movzx eax, byte ptr [r8 + rcx]
    inc rcx
    jmp qword ptr [r9 + rax*8]

op_halt:
    xor r12d, r12d
halt_flush_loop:
    cmp r12, rbx
    jae exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [r10 + r12]
    mov rdx, rbx
    sub rdx, r12
    syscall
    test rax, rax
    jle exit_ok
    add r12, rax
    jmp halt_flush_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

exit_err:
    mov eax, 60
    mov edi, 1
    syscall
