.text
.globl _start

_start:
    // Read all of stdin into in_buf
    adrp x20, in_buf
    add x20, x20, :lo12:in_buf
    mov x19, #0
    movz x21, #1, lsl #16

read_loop:
    sub x2, x21, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.gt read_advance
    cmn x0, #4
    b.eq read_loop
    b read_done

read_advance:
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, exit_zero

    // Parse lines into lines[] as (ptr, len) pairs
    mov x21, x20
    add x22, x20, x19
    mov x23, x20
    adrp x24, lines
    add x24, x24, :lo12:lines
    mov x25, #0

parse_loop:
    cmp x21, x22
    b.hs parse_done
    ldrb w0, [x21]
    cmp w0, #10
    b.ne parse_next

    sub x1, x21, x23
    add x2, x24, x25, lsl #4
    stp x23, x1, [x2]
    add x25, x25, #1
    add x21, x21, #1
    mov x23, x21
    b parse_loop

parse_next:
    add x21, x21, #1
    b parse_loop

parse_done:
    cmp x23, x22
    b.hs parse_end
    sub x1, x22, x23
    add x2, x24, x25, lsl #4
    stp x23, x1, [x2]
    add x25, x25, #1

parse_end:
    // Heapsort lines[0 .. num_lines - 1]
    mov x19, x24
    mov x20, x25
    cmp x20, #1
    b.le sort_done

    lsr x21, x20, #1
    sub x21, x21, #1

build_heap_loop:
    mov x9, x21
    mov x10, x20
    bl sift_down
    subs x21, x21, #1
    b.ge build_heap_loop

    sub x21, x20, #1

sort_extract_loop:
    cmp x21, #0
    b.le sort_done

    add x13, x19, x21, lsl #4
    ldp x0, x1, [x19]
    ldp x2, x3, [x13]
    stp x2, x3, [x19]
    stp x0, x1, [x13]

    mov x9, #0
    mov x10, x21
    bl sift_down
    sub x21, x21, #1
    b sort_extract_loop

sort_done:
    // Buffered output to stdout
    mov x21, #0
    adrp x22, out_buf
    add x22, x22, :lo12:out_buf
    mov x23, #0
    movz x24, #1, lsl #16

out_loop:
    cmp x21, x20
    b.hs out_flush

    add x0, x19, x21, lsl #4
    ldp x25, x26, [x0]

    add x0, x23, x26
    add x0, x0, #1
    cmp x0, x24
    b.ls out_copy

    cbz x23, out_check_huge
    mov x0, x22
    mov x1, x23
    bl write_all
    mov x23, #0

out_check_huge:
    add x0, x26, #1
    cmp x0, x24
    b.ls out_copy

    mov x0, x25
    mov x1, x26
    bl write_all
    mov w0, #10
    strb w0, [x22]
    mov x0, x22
    mov x1, #1
    bl write_all
    add x21, x21, #1
    b out_loop

out_copy:
    add x0, x22, x23
    mov x1, x25
    mov x2, x26
    cbz x2, out_newline

out_copy_loop:
    ldrb w3, [x1], #1
    strb w3, [x0], #1
    subs x2, x2, #1
    b.ne out_copy_loop

out_newline:
    mov w3, #10
    strb w3, [x0]
    add x23, x23, x26
    add x23, x23, #1
    add x21, x21, #1
    b out_loop

out_flush:
    cbz x23, exit_zero
    mov x0, x22
    mov x1, x23
    bl write_all

exit_zero:
    mov x0, #0
    mov x8, #93
    svc #0

// cmp_lines(x0=ptrA, x1=lenA, x2=ptrB, x3=lenB) -> x0 (<0, 0, >0)
// Clobbers only x0-x7
cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, lo
    cbz x4, 2f
1:
    ldrb w5, [x0], #1
    ldrb w6, [x2], #1
    subs w7, w5, w6
    b.ne 3f
    subs x4, x4, #1
    b.ne 1b
2:
    sub x0, x1, x3
    ret
3:
    sxtw x0, w7
    ret

// sift_down(x19=lines_base, x9=root, x10=count)
// Clobbers x0-x7, x9, x11-x15
sift_down:
    mov x15, x30
1:
    lsl x11, x9, #1
    add x11, x11, #1
    cmp x11, x10
    b.hs 3f

    add x12, x11, #1
    cmp x12, x10
    b.hs 2f

    add x13, x19, x11, lsl #4
    ldp x0, x1, [x13]
    add x14, x19, x12, lsl #4
    ldp x2, x3, [x14]
    bl cmp_lines
    cmp x0, #0
    b.ge 2f
    mov x11, x12
2:
    add x13, x19, x9, lsl #4
    ldp x0, x1, [x13]
    add x14, x19, x11, lsl #4
    ldp x2, x3, [x14]
    bl cmp_lines
    cmp x0, #0
    b.ge 3f

    ldp x0, x1, [x13]
    ldp x2, x3, [x14]
    stp x2, x3, [x13]
    stp x0, x1, [x14]

    mov x9, x11
    b 1b
3:
    mov x30, x15
    ret

// write_all(x0=buf, x1=count)
write_all:
    mov x2, x1
    mov x1, x0
1:
    cbz x2, 2f
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.gt 3f
    cmn x0, #4
    b.eq 1b
    b 2f
3:
    add x1, x1, x0
    sub x2, x2, x0
    b 1b
2:
    ret

.bss
.balign 16
lines:
    .space 524288
in_buf:
    .space 65536
out_buf:
    .space 65536
