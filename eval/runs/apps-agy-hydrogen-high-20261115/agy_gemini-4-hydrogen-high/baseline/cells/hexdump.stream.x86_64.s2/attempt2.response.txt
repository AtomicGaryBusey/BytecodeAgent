.globl _start

.text
_start:
    xor ebx, ebx                # offset = 0
    xor r12d, r12d              # leftover = 0
    xor r13d, r13d              # out_pos = 0
    lea r14, [rip + inbuf]
    lea r15, [rip + outbuf]

read_loop:
    xor eax, eax                # SYS_read = 0
    xor edi, edi                # fd = 0 (stdin)
    lea rsi, [r14 + r12]        # buf = inbuf + leftover
    mov edx, 4096               # count = 4096
    syscall
    test rax, rax
    jle eof

    lea rbp, [r12 + rax]        # avail = leftover + bytes_read
    mov rsi, r14                # cur_ptr = inbuf

row_loop:
    cmp rbp, 16
    jb row_loop_done

    mov eax, ebx
    lea rdi, [r15 + r13]
    mov edx, 16
    call format_row
    add r13, rax
    add rbx, 16
    add rsi, 16
    sub rbp, 16

    cmp r13, 65536 - 128
    jb row_loop
    call flush_out
    jmp row_loop

row_loop_done:
    mov r12, rbp                # leftover = avail
    test rbp, rbp
    jz read_loop
    cmp rsi, r14
    je read_loop
    xor ecx, ecx
copy_rem:
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [r14 + rcx], al
    inc rcx
    cmp rcx, rbp
    jb copy_rem
    jmp read_loop

eof:
    test r12, r12
    jz final_offset

    mov eax, ebx
    lea rdi, [r15 + r13]
    mov rsi, r14
    mov rdx, r12
    call format_row
    add r13, rax
    add rbx, r12

final_offset:
    cmp r13, 65536 - 16
    jb 1f
    call flush_out
1:
    mov eax, ebx
    lea rdi, [r15 + r13]
    call write_hex32
    mov byte ptr [r15 + r13 + 8], 0x0a
    add r13, 9

    call flush_out

    mov eax, 60                 # SYS_exit = 60
    xor edi, edi                # status = 0
    syscall

# format_row(eax=offset, rdi=dst, rsi=src, rdx=n [1..16]) -> rax=bytes_written
# Preserves rbx, rbp, rsi, r12, r13, r14, r15
format_row:
    mov r10, rdx
    call write_hex32

    mov rax, 0x2020202020202020
    mov qword ptr [rdi + 8], rax
    mov qword ptr [rdi + 16], rax
    mov qword ptr [rdi + 24], rax
    mov qword ptr [rdi + 32], rax
    mov qword ptr [rdi + 40], rax
    mov qword ptr [rdi + 48], rax
    mov qword ptr [rdi + 56], rax

    mov byte ptr [rdi + 60], 0x7c

    lea r9, [rip + hex_digits]
    xor ecx, ecx
    lea r8, [rdi + 10]
1:
    cmp rcx, 8
    jne 2f
    inc r8
2:
    movzx eax, byte ptr [rsi + rcx]

    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r8], dl

    mov edx, eax
    and edx, 0x0f
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [r8 + 1], dl

    add r8, 3

    mov edx, 0x2e               # '.'
    lea r11d, [rax - 0x20]
    cmp r11b, 0x5e
    cmova eax, edx
    mov byte ptr [rdi + rcx + 61], al

    inc rcx
    cmp rcx, r10
    jb 1b

    mov byte ptr [rdi + r10 + 61], 0x7c
    mov byte ptr [rdi + r10 + 62], 0x0a

    lea rax, [r10 + 63]
    ret

# write_hex32(eax=val, rdi=dst)
# Preserves rbx, rcx, rdi, rsi, rbp, r10, r12..r15
write_hex32:
    lea r9, [rip + hex_digits]
    mov r8d, 7
1:
    mov edx, eax
    and edx, 0x0f
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + r8], dl
    shr eax, 4
    dec r8
    jns 1b
    ret

# flush_out: writes r13 bytes from r15, resets r13 = 0
# Preserves rbx, rbp, rsi, r12, r14, r15
flush_out:
    test r13, r13
    jz 2f
    push rsi
    xor r8d, r8d
1:
    mov eax, 1                  # SYS_write = 1
    mov edi, 1                  # fd = 1 (stdout)
    lea rsi, [r15 + r8]
    mov rdx, r13
    sub rdx, r8
    syscall
    test rax, rax
    jle 3f
    add r8, rax
    cmp r8, r13
    jb 1b
3:
    xor r13d, r13d
    pop rsi
2:
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .space 4112
    .align 16
outbuf:
    .space 65536
