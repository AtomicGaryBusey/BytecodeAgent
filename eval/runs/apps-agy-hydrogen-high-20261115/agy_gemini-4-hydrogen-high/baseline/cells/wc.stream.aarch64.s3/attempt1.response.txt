.text
.globl _start

_start:
    mov x19, #0                 // lines = 0
    mov x20, #0                 // words = 0
    mov x21, #0                 // bytes = 0
    mov w22, #0                 // in_word = 0

    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0                  // fd = 0 (stdin)
    mov x1, x23                 // buf
    movz x2, #1, lsl #16        // count = 65536
    mov x8, #63                 // __NR_read
    svc #0

    cmp x0, #0
    b.le read_done              // EOF or error

    add x21, x21, x0            // bytes += nread
    mov x1, x23                 // ptr = buf
    add x2, x23, x0             // end = buf + nread

byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #0x0A               // '\n'
    b.ne 1f
    add x19, x19, #1            // lines++
    mov w22, #0                 // in_word = 0
    b 3f
1:
    cmp w3, #0x20               // ' '
    b.eq 2f
    sub w4, w3, #0x09           // '\t'..'\r' (0x09..0x0D)
    cmp w4, #4
    b.ls 2f

    cbnz w22, 3f
    add x20, x20, #1            // words++
    mov w22, #1                 // in_word = 1
    b 3f
2:
    mov w22, #0                 // in_word = 0
3:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

read_done:
    sub sp, sp, #128            // output buffer on stack
    add x24, sp, #128           // end of output buffer
    mov x1, x24                 // write cursor (grows downward)

    mov w0, #0x0A               // '\n'
    strb w0, [x1, #-1]!

    mov x0, x21                 // bytes
    bl format_u64

    mov w0, #0x20               // ' '
    strb w0, [x1, #-1]!

    mov x0, x20                 // words
    bl format_u64

    mov w0, #0x20               // ' '
    strb w0, [x1, #-1]!

    mov x0, x19                 // lines
    bl format_u64

    mov x25, x1                 // start of formatted output
    sub x26, x24, x1            // remaining length

write_loop:
    cbz x26, exit_ok
    mov x0, #1                  // fd = 1 (stdout)
    mov x1, x25
    mov x2, x26
    mov x8, #64                 // __NR_write
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x25, x25, x0
    sub x26, x26, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #93                 // __NR_exit
    svc #0

// Formats unsigned 64-bit integer in x0 in decimal backwards from x1.
// Returns updated pointer in x1. Clobbers x0, x2, x3, x4.
format_u64:
    mov x4, #10
1:
    udiv x2, x0, x4
    msub x3, x2, x4, x0
    add w3, w3, #0x30           // '0'
    strb w3, [x1, #-1]!
    mov x0, x2
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space 65536
