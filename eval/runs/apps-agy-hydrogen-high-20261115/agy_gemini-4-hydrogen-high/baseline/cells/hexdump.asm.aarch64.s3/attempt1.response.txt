    .text
    .globl _start

_start:
    mov x19, #0
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf
    mov x21, x20
    add x22, x20, #4096
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits
    mov x24, #0
    adrp x25, rowbuf
    add x25, x25, :lo12:rowbuf
    adrp x26, inbuf
    add x26, x26, :lo12:inbuf

read_loop:
    mov x0, #0
    mov x1, x26
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le eof

    mov x27, x26
    mov x28, x0

consume_loop:
    cbz x28, read_loop
    cbnz x24, slow_byte
    cmp x28, #16
    b.lo slow_byte

    mov w0, w19
    mov x1, x27
    mov w2, #16
    bl format_row
    add x19, x19, #16
    add x27, x27, #16
    sub x28, x28, #16
    b consume_loop

slow_byte:
    ldrb w3, [x27], #1
    strb w3, [x25, x24]
    add x24, x24, #1
    sub x28, x28, #1
    cmp x24, #16
    b.lo consume_loop

    mov w0, w19
    mov x1, x25
    mov w2, #16
    bl format_row
    add x19, x19, #16
    mov x24, #0
    b consume_loop

eof:
    cbz x24, 1f
    mov w0, w19
    mov x1, x25
    mov w2, w24
    bl format_row
    add x19, x19, x24
1:
    cmp x21, x22
    b.lo 2f
    bl flush_out
2:
    mov w0, w19
    bl format_hex8
    mov w3, #0x0a
    strb w3, [x21], #1
    bl flush_out

    mov x0, #0
    mov x8, #93
    svc #0

// flush_out: write [x20, x21) to stdout and reset x21 = x20
// Clobbers: x0-x4, x8, x21
flush_out:
    sub x4, x21, x20
    cbz x4, 2f
    mov x3, x20
1:
    mov x0, #1
    mov x1, x3
    mov x2, x4
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x3, x3, x0
    sub x4, x4, x0
    cbnz x4, 1b
2:
    mov x21, x20
    ret

// format_hex8: format w0 as 8 lowercase hex digits at x21, advance x21 by 8
// Clobbers: w0, x2, x3, x4
format_hex8:
    add x2, x21, #8
    mov x4, x2
1:
    sub x2, x2, #1
    and w3, w0, #0xf
    ldrb w3, [x23, x3]
    strb w3, [x2]
    lsr w0, w0, #4
    cmp x2, x21
    b.hi 1b
    mov x21, x4
    ret

// format_row: format one row (1..16 bytes) into outbuf at x21
// Inputs: w0 = off, x1 = data ptr, w2 = n
// Clobbers: x0-x5, x8-x15, updates x21
format_row:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    mov w9, w0
    mov x10, x1
    mov w11, w2

    cmp x21, x22
    b.lo 1f
    bl flush_out
1:
    mov w0, w9
    bl format_hex8

    mov w3, #0x20
    strb w3, [x21], #1
    strb w3, [x21], #1

    mov w12, #0
2:
    cmp w12, w11
    b.hs 3f
    ldrb w13, [x10, x12]
    lsr w14, w13, #4
    and w15, w13, #0xf
    ldrb w14, [x23, x14]
    ldrb w15, [x23, x15]
    strb w14, [x21], #1
    strb w15, [x21], #1
    mov w3, #0x20
    strb w3, [x21], #1
    b 4f
3:
    mov w3, #0x20
    strb w3, [x21], #1
    strb w3, [x21], #1
    strb w3, [x21], #1
4:
    cmp w12, #7
    b.ne 5f
    mov w3, #0x20
    strb w3, [x21], #1
5:
    add w12, w12, #1
    cmp w12, #16
    b.lo 2b

    mov w3, #0x20
    strb w3, [x21], #1
    mov w3, #0x7c
    strb w3, [x21], #1

    mov w12, #0
6:
    ldrb w3, [x10, x12]
    sub w4, w3, #0x20
    cmp w4, #0x5e
    mov w5, #0x2e
    csel w3, w3, w5, ls
    strb w3, [x21], #1
    add w12, w12, #1
    cmp w12, w11
    b.lo 6b

    mov w3, #0x7c
    strb w3, [x21], #1
    mov w3, #0x0a
    strb w3, [x21], #1

    ldp x29, x30, [sp], #16
    ret

    .section .rodata
hex_digits:
    .ascii "0123456789abcdef"

    .bss
    .balign 16
inbuf:
    .space 4096
    .balign 16
outbuf:
    .space 8192
    .balign 16
rowbuf:
    .space 16
