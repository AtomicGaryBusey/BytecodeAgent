.equ IN_CAP, 65536
.equ OUT_CAP, 65536
.equ OUT_LIMIT_OFF, OUT_CAP - 128

.text
.globl _start

_start:
    lea rbp, [rip + hex_tbl]
    xor r12d, r12d
    xor r13d, r13d
    lea r14, [rip + out_buf]
    lea r15, [r14 + OUT_LIMIT_OFF]

read_more:
    lea rsi, [rip + in_buf]
    add rsi, r13
    mov edx, IN_CAP
    sub rdx, r13
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle eof
    add r13, rax
    cmp r13, 16
    jb read_more

    lea rbx, [rip + in_buf]
row_loop:
    cmp r14, r15
    jb 1f
    call flush_out
1:
    mov rsi, rbx
    mov edx, 16
    call format_row
    add rbx, 16
    add r12d, 16
    sub r13, 16
    cmp r13, 16
    jae row_loop

    test r13, r13
    jz read_more
    lea rdi, [rip + in_buf]
    mov rsi, rbx
    mov rcx, r13
    rep movsb
    jmp read_more

eof:
    test r13, r13
    jz 2f
    cmp r14, r15
    jb 1f
    call flush_out
1:
    lea rsi, [rip + in_buf]
    mov rdx, r13
    call format_row
    add r12d, r13d
2:
    cmp r14, r15
    jb 3f
    call flush_out
3:
    mov rdi, r14
    mov eax, r12d
    call format_hex32
    mov byte ptr [r14 + 8], 0x0a
    add r14, 9
    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

# Write 8 lowercase hex digits of eax to [rdi..rdi+7]
# Clobbers rax, rcx, rdx
format_hex32:
    bswap eax
    movzx edx, al
    mov cx, word ptr [rbp + rdx*2]
    mov word ptr [rdi], cx
    shr eax, 8
    movzx edx, al
    mov cx, word ptr [rbp + rdx*2]
    mov word ptr [rdi + 2], cx
    shr eax, 8
    movzx edx, al
    mov cx, word ptr [rbp + rdx*2]
    mov word ptr [rdi + 4], cx
    shr eax, 8
    mov cx, word ptr [rbp + rax*2]
    mov word ptr [rdi + 6], cx
    ret

# Format a row of rdx (1..16) bytes from [rsi] at offset r12d into [r14]
# Updates r14
format_row:
    mov r10, rdx
    mov r11, rsi
    mov rdi, r14
    mov eax, r12d
    call format_hex32

    mov rax, 0x2020202020202020
    mov qword ptr [rdi + 8], rax
    mov qword ptr [rdi + 16], rax
    mov qword ptr [rdi + 24], rax
    mov qword ptr [rdi + 32], rax
    mov qword ptr [rdi + 40], rax
    mov qword ptr [rdi + 48], rax
    mov dword ptr [rdi + 56], eax
    mov byte ptr [rdi + 60], 0x7c

    xor ecx, ecx
1:
    movzx eax, byte ptr [r11 + rcx]
    cmp rcx, 8
    sbb r8, r8
    lea r9, [rcx + rcx*2 + 11]
    add r9, r8
    mov r8w, word ptr [rbp + rax*2]
    mov word ptr [rdi + r9], r8w

    lea r8d, [rax - 0x20]
    cmp r8b, 0x5e
    mov r9d, 0x2e
    cmova eax, r9d
    mov byte ptr [rdi + rcx + 61], al

    inc rcx
    cmp rcx, r10
    jb 1b

    mov word ptr [rdi + r10 + 61], 0x0a7c
    lea r14, [rdi + r10 + 63]
    ret

# Flush out_buf up to r14 and reset r14 to out_buf
flush_out:
    lea rsi, [rip + out_buf]
    mov rdx, r14
    sub rdx, rsi
    jz 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    lea r14, [rip + out_buf]
    ret

.section .rodata
.p2align 4
hex_tbl:

    .ascii "101112131415161718191a1b1c1d1e1f"

    .ascii "303132333435363738393a3b3c3d3e3f"
    .ascii "404142434445464748494a4b4c4d4e4f"

    .ascii "707172737475767778797a7b7c7d7e7f"
    .ascii "808182838485868788898a8b8c8d8e8f"
    .ascii "909192939495969798999a9b9c9d9e9f"
    .ascii "a0a1a2a3a4a5a6a7a8a9aaabacadaeaf"
    .ascii "b0b1b2b3b4b5b6b7b8b9babbbcbdbebf"

    .ascii "f0f1f2f3f4f5f6f7f8f9fafbfcfdfeff"

.bss
.p2align 4
in_buf:
    .zero IN_CAP
.p2align 4
out_buf:
    .zero OUT_CAP
