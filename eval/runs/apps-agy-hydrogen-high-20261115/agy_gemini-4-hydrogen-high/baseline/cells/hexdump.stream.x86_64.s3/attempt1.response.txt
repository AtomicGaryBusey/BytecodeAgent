.text
.globl _start

_start:
    lea rbx, [rip + hex_digits]
    lea r12, [rip + inbuf]
    lea r13, [rip + outbuf]
    mov rbp, r13
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r14]
    mov edx, 65536
    sub rdx, r14
    syscall

    test rax, rax
    jz eof
    jns 1f
    cmp rax, -4
    je read_loop
    jmp eof
1:
    lea r10, [r14 + rax]
    xor r14d, r14d

row_loop:
    lea rax, [r14 + 16]
    cmp rax, r10
    ja row_loop_done

    lea rsi, [r12 + r14]
    mov ecx, 16
    mov eax, r15d
    mov rdi, rbp
    call format_row
    mov rbp, rdi

    add r14, 16
    add r15d, 16

    mov rax, rbp
    sub rax, r13
    cmp rax, 65280
    jb row_loop
    call flush_outbuf
    jmp row_loop

row_loop_done:
    sub r10, r14
    jz copy_done
    test r14, r14
    jz copy_done

    xor ecx, ecx
1:
    mov al, byte ptr [r12 + r14]
    mov byte ptr [r12 + rcx], al
    inc r14
    inc rcx
    cmp rcx, r10
    jb 1b

copy_done:
    mov r14, r10
    jmp read_loop

eof:
    test r14, r14
    jz write_final_offset

    mov rsi, r12
    mov rcx, r14
    mov eax, r15d
    mov rdi, rbp
    call format_row
    mov rbp, rdi
    add r15d, r14d

write_final_offset:
    mov eax, r15d
    mov rdi, rbp
    call format_hex32
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rbp, rdi

    call flush_outbuf

    mov eax, 60
    xor edi, edi
    syscall

format_hex32:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 0x0f
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

format_row:
    mov r9, rcx
    call format_hex32
    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r8d, r8d
1:
    cmp r8, 8
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    cmp r8, r9
    jae 3f
    movzx eax, byte ptr [rsi + r8]
    mov edx, eax
    shr edx, 4
    and eax, 0x0f
    mov dl, byte ptr [rbx + rdx]
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], dl
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp 4f
3:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
4:
    add rdi, 3
    inc r8
    cmp r8, 16
    jb 1b

    mov word ptr [rdi], 0x7c20
    add rdi, 2

    xor r8d, r8d
5:
    movzx eax, byte ptr [rsi + r8]
    mov edx, eax
    sub edx, 0x20
    cmp dl, 0x5f
    mov edx, 0x2e
    cmovae eax, edx
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, r9
    jb 5b

    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    ret

flush_outbuf:
    mov rdx, rbp
    sub rdx, r13
    jz 2f
    mov rsi, r13
1:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    ja 1b
2:
    mov rbp, r13
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 65536
