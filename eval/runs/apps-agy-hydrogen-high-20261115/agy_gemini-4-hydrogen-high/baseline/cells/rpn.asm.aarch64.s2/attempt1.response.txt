.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.equ INBUF_SIZE, 131072
.equ OUTBUF_SIZE, 65536
.equ OUTBUF_FLUSH, 65000
.equ RPN_MAX_DEPTH, 1024
.equ RPN_STACK_BYTES, 8192

.equ CH_TAB, 0x09
.equ CH_LF, 0x0A
.equ CH_SPACE, 0x20
.equ CH_PCT, 0x25
.equ CH_MUL, 0x2A
.equ CH_ADD, 0x2B
.equ CH_SUB, 0x2D
.equ CH_DIV, 0x2F
.equ CH_ZERO, 0x30

.text
.globl _start
.balign 4
_start:
    sub sp, sp, #32

    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    movz x0, #2, lsl #16
    add x21, x19, x0

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x21, x23
    movz x0, #OUTBUF_FLUSH
    add x22, x23, x0

    adrp x24, rpn_stack
    add x24, x24, :lo12:rpn_stack

line_loop:
    cmp x19, x20
    b.hs all_done

    mov x25, x19
    mov x26, x19
1:
    cmp x26, x20
    b.hs 2f
    ldrb w0, [x26]
    cmp w0, #CH_LF
    b.eq 2f
    add x26, x26, #1
    b 1b
2:
    cmp x26, x20
    b.hs 3f
    add x19, x26, #1
    b 4f
3:
    mov x19, x26
4:
    mov x27, #0

tok_loop:
1:
    cmp x25, x26
    b.hs end_of_line
    ldrb w0, [x25]
    cmp w0, #CH_SPACE
    b.eq 2f
    cmp w0, #CH_TAB
    b.eq 2f
    b tok_start
2:
    add x25, x25, #1
    b 1b

tok_start:
    add x28, x25, #1
3:
    cmp x28, x26
    b.hs 4f
    ldrb w0, [x28]
    cmp w0, #CH_SPACE
    b.eq 4f
    cmp w0, #CH_TAB
    b.eq 4f
    add x28, x28, #1
    b 3b
4:
    sub x1, x28, x25
    ldrb w0, [x25]
    cmp x1, #1
    b.ne parse_literal

    cmp w0, #CH_ADD
    b.eq op_add
    cmp w0, #CH_SUB
    b.eq op_sub
    cmp w0, #CH_MUL
    b.eq op_mul
    cmp w0, #CH_DIV
    b.eq op_div
    cmp w0, #CH_PCT
    b.eq op_mod

parse_literal:
    mov x2, #0
    mov x3, x25
    cmp w0, #CH_SUB
    b.ne check_digit_len
    mov x2, #1
    add x3, x3, #1
    sub x1, x1, #1

check_digit_len:
    cmp x1, #18
    b.hi line_error
    mov x4, #0
    mov x5, #10

digit_loop:
    ldrb w6, [x3], #1
    sub w6, w6, #CH_ZERO
    cmp w6, #9
    b.hi line_error
    mul x4, x4, x5
    add x4, x4, x6
    cmp x3, x28
    b.lo digit_loop

    cbz x2, 1f
    neg x4, x4
1:
    cmp x27, #RPN_MAX_DEPTH
    b.hs line_error
    str x4, [x24, x27, lsl #3]
    add x27, x27, #1
    mov x25, x28
    b tok_loop

op_add:
    cmp x27, #2
    b.lo line_error
    sub x27, x27, #1
    ldr x2, [x24, x27, lsl #3]
    sub x3, x27, #1
    ldr x1, [x24, x3, lsl #3]
    add x0, x1, x2
    str x0, [x24, x3, lsl #3]
    mov x25, x28
    b tok_loop

op_sub:
    cmp x27, #2
    b.lo line_error
    sub x27, x27, #1
    ldr x2, [x24, x27, lsl #3]
    sub x3, x27, #1
    ldr x1, [x24, x3, lsl #3]
    sub x0, x1, x2
    str x0, [x24, x3, lsl #3]
    mov x25, x28
    b tok_loop

op_mul:
    cmp x27, #2
    b.lo line_error
    sub x27, x27, #1
    ldr x2, [x24, x27, lsl #3]
    sub x3, x27, #1
    ldr x1, [x24, x3, lsl #3]
    mul x0, x1, x2
    str x0, [x24, x3, lsl #3]
    mov x25, x28
    b tok_loop

op_div:
    cmp x27, #2
    b.lo line_error
    sub x27, x27, #1
    ldr x2, [x24, x27, lsl #3]
    cbz x2, line_error
    sub x3, x27, #1
    ldr x1, [x24, x3, lsl #3]
    cmn x2, #1
    b.ne 1f
    movz x4, #0x8000, lsl #48
    cmp x1, x4
    b.eq line_error
1:
    sdiv x0, x1, x2
    str x0, [x24, x3, lsl #3]
    mov x25, x28
    b tok_loop

op_mod:
    cmp x27, #2
    b.lo line_error
    sub x27, x27, #1
    ldr x2, [x24, x27, lsl #3]
    cbz x2, line_error
    sub x3, x27, #1
    ldr x1, [x24, x3, lsl #3]
    cmn x2, #1
    b.ne 1f
    movz x4, #0x8000, lsl #48
    cmp x1, x4
    b.eq line_error
1:
    sdiv x4, x1, x2
    msub x0, x4, x2, x1
    str x0, [x24, x3, lsl #3]
    mov x25, x28
    b tok_loop

end_of_line:
    cmp x27, #1
    b.ne line_error
    ldr x0, [x24]

    cmp x0, #0
    b.ge 1f
    mov w1, #CH_SUB
    strb w1, [x21], #1
    neg x0, x0
1:
    add x2, sp, #32
    mov x3, x2
    mov x4, #10
2:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #CH_ZERO
    strb w6, [x3, #-1]!
    mov x0, x5
    cbnz x0, 2b
3:
    ldrb w6, [x3], #1
    strb w6, [x21], #1
    cmp x3, x2
    b.lo 3b

    mov w6, #CH_LF
    strb w6, [x21], #1

    cmp x21, x22
    b.lo line_loop
    bl flush_out
    b line_loop

line_error:
    adrp x0, err_str
    add x0, x0, :lo12:err_str
    ldrb w1, [x0, #0]
    strb w1, [x21, #0]
    ldrb w1, [x0, #1]
    strb w1, [x21, #1]
    ldrb w1, [x0, #2]
    strb w1, [x21, #2]
    ldrb w1, [x0, #3]
    strb w1, [x21, #3]
    ldrb w1, [x0, #4]
    strb w1, [x21, #4]
    ldrb w1, [x0, #5]
    strb w1, [x21, #5]
    add x21, x21, #6

    cmp x21, x22
    b.lo line_loop
    bl flush_out
    b line_loop

all_done:
    bl flush_out
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

flush_out:
    mov x9, x23
1:
    sub x2, x21, x9
    cbz x2, 2f
    mov x0, #1
    mov x1, x9
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le 2f
    add x9, x9, x0
    b 1b
2:
    mov x21, x23
    ret

.section .rodata
err_str:
    .ascii "error\n"

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 8
rpn_stack:
    .space RPN_STACK_BYTES
