.bss
.align 16
buf:
    .space 65552

.text
.globl _start
_start:
    lea rbx, [rip + buf]
    xor r12d, r12d          # total bytes read

1:
    xor eax, eax            # SYS_read = 0
    xor edi, edi            # fd = 0 (stdin)
    lea rsi, [rbx + r12]
    mov edx, 65552
    sub edx, r12d
    jle 2f
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b

2:
    test r12, r12
    jz 9f                   # empty input: nothing to print

    cmp byte ptr [rbx + r12 - 1], 10
    je 3f
    mov byte ptr [rbx + r12], 10
    inc r12

3:
    mov rdi, rbx            # current scan pointer
    lea r13, [rbx + r12]    # end of buffer
    mov eax, 10             # newline byte 0x0A

4:
    cmp rdi, r13
    jae 7f

    mov rsi, rdi            # line start
    mov rcx, r13
    sub rcx, rdi            # remaining bytes
    repne scasb             # rdi advances past '\n'
    lea rdx, [rdi - 2]      # line end (last byte before '\n')

5:
    cmp rsi, rdx
    jae 4b
    mov cl, byte ptr [rsi]
    mov r8b, byte ptr [rdx]
    mov byte ptr [rsi], r8b
    mov byte ptr [rdx], cl
    inc rsi
    dec rdx
    jmp 5b

7:
    xor r14d, r14d          # total bytes written

8:
    cmp r14, r12
    jae 9f
    mov eax, 1              # SYS_write = 1
    mov edi, 1              # fd = 1 (stdout)
    lea rsi, [rbx + r14]
    mov rdx, r12
    sub rdx, r14
    syscall
    test rax, rax
    jle 9f
    add r14, rax
    jmp 8b

9:
    mov eax, 60             # SYS_exit = 60
    xor edi, edi            # exit status 0
    syscall
