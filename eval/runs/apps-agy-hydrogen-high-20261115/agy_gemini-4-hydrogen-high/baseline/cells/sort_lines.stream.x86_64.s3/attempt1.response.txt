.globl _start

.text
_start:
    lea r15, [rip + in_buf]
    xor ebx, ebx

read_loop:
    mov rdx, 65536
    sub rdx, rbx
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_ok

    lea r14, [rip + lines]
    xor r13d, r13d
    mov rsi, r15
    mov rcx, r15
    lea rdx, [r15 + rbx]

parse_loop:
    cmp rcx, rdx
    jae parse_after
    cmp byte ptr [rcx], 10
    je parse_eol
    inc rcx
    jmp parse_loop

parse_eol:
    mov rax, rsi
    sub rax, r15
    mov r8, rcx
    sub r8, rsi
    shl r8, 32
    or rax, r8
    mov qword ptr [r14 + r13*8], rax
    inc r13
    inc rcx
    mov rsi, rcx
    jmp parse_loop

parse_after:
    cmp rsi, rdx
    jae parse_done
    mov rax, rsi
    sub rax, r15
    mov r8, rdx
    sub r8, rsi
    shl r8, 32
    or rax, r8
    mov qword ptr [r14 + r13*8], rax
    inc r13

parse_done:
    test r13, r13
    jz exit_ok

    xor edi, edi
    lea rsi, [r13 - 1]
    call quicksort

    lea r12, [rip + out_buf]
    mov rdi, r12
    xor ebx, ebx

build_loop:
    cmp rbx, r13
    jae build_done
    mov rax, qword ptr [r14 + rbx*8]
    mov edx, eax
    shr rax, 32
    lea rsi, [r15 + rdx]
    mov rcx, rax
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp build_loop

build_done:
    sub rdi, r12
    mov r13, rdi
    xor ebx, ebx

write_loop:
    mov rdx, r13
    sub rdx, rbx
    jbe exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle exit_ok
    add rbx, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

cmp_lines:
    mov eax, edi
    mov edx, esi
    shr rdi, 32
    shr rsi, 32
    mov r8, rdi
    mov r9, rsi
    lea rsi, [r15 + rax]
    lea rdi, [r15 + rdx]
    mov rcx, r8
    cmp r8, r9
    cmova rcx, r9
    test rcx, rcx
    jz 1f
    repe cmpsb
    jb 2f
    ja 3f
1:
    cmp r8, r9
    jb 2f
    ja 3f
    xor eax, eax
    ret
2:
    mov eax, -1
    ret
3:
    mov eax, 1
    ret

quicksort:
    push rbx
    push rbp
    push r12
    push r13
    mov rbx, rdi
    mov r12, rsi

qs_loop:
    cmp rbx, r12
    jge qs_done

    lea r10, [rbx + r12]
    shr r10, 1

    mov rdi, qword ptr [r14 + rbx*8]
    mov rsi, qword ptr [r14 + r10*8]
    call cmp_lines
    test eax, eax
    jle 1f
    mov rax, qword ptr [r14 + rbx*8]
    mov rcx, qword ptr [r14 + r10*8]
    mov qword ptr [r14 + rbx*8], rcx
    mov qword ptr [r14 + r10*8], rax
1:
    mov rdi, qword ptr [r14 + rbx*8]
    mov rsi, qword ptr [r14 + r12*8]
    call cmp_lines
    test eax, eax
    jle 2f
    mov rax, qword ptr [r14 + rbx*8]
    mov rcx, qword ptr [r14 + r12*8]
    mov qword ptr [r14 + rbx*8], rcx
    mov qword ptr [r14 + r12*8], rax
2:
    mov rdi, qword ptr [r14 + r10*8]
    mov rsi, qword ptr [r14 + r12*8]
    call cmp_lines
    test eax, eax
    jle 3f
    mov rax, qword ptr [r14 + r10*8]
    mov rcx, qword ptr [r14 + r12*8]
    mov qword ptr [r14 + r10*8], rcx
    mov qword ptr [r14 + r12*8], rax
3:
    mov r13, qword ptr [r14 + r10*8]
    lea rbp, [rbx - 1]
    lea r11, [r12 + 1]

part_loop:
4:
    inc rbp
    mov rdi, qword ptr [r14 + rbp*8]
    mov rsi, r13
    call cmp_lines
    test eax, eax
    jl 4b

5:
    dec r11
    mov rdi, qword ptr [r14 + r11*8]
    mov rsi, r13
    call cmp_lines
    test eax, eax
    jg 5b

    cmp rbp, r11
    jge part_done

    mov rax, qword ptr [r14 + rbp*8]
    mov rcx, qword ptr [r14 + r11*8]
    mov qword ptr [r14 + rbp*8], rcx
    mov qword ptr [r14 + r11*8], rax
    jmp part_loop

part_done:
    mov rax, r11
    sub rax, rbx
    mov rcx, r12
    sub rcx, r11
    dec rcx
    cmp rax, rcx
    jge right_smaller

    mov rdi, rbx
    mov rsi, r11
    lea rbx, [r11 + 1]
    call quicksort
    jmp qs_loop

right_smaller:
    lea rdi, [r11 + 1]
    mov rsi, r12
    mov r12, r11
    call quicksort
    jmp qs_loop

qs_done:
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.bss
.p2align 4
in_buf:
    .space 65536
out_buf:
    .space 131072
lines:
    .space 524288
