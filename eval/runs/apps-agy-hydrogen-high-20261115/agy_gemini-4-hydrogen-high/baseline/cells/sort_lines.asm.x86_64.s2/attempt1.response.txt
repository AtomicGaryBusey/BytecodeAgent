.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 65536
.equ LINES_BYTES, 32768

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
outbuf:
    .space OUTBUF_SIZE
lines_a:
    .space LINES_BYTES
lines_b:
    .space LINES_BYTES

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    mov eax, SYS_READ
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, INBUF_SIZE
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_zero

    xor ebx, ebx
    xor r13d, r13d
    xor r14d, r14d
    lea r15, [rip + inbuf]
    lea rbp, [rip + lines_a]

parse_loop:
    cmp rbx, r12
    jae parse_after
    cmp byte ptr [r15 + rbx], 0x0A
    jne parse_next
    lea rax, [r15 + r13]
    mov rcx, rbx
    sub rcx, r13
    mov rdx, r14
    shl rdx, 4
    mov qword ptr [rbp + rdx], rax
    mov qword ptr [rbp + rdx + 8], rcx
    inc r14
    lea r13, [rbx + 1]

parse_next:
    inc rbx
    jmp parse_loop

parse_after:
    cmp r13, r12
    jae parse_done
    lea rax, [r15 + r13]
    mov rcx, r12
    sub rcx, r13
    mov rdx, r14
    shl rdx, 4
    mov qword ptr [rbp + rdx], rax
    mov qword ptr [rbp + rdx + 8], rcx
    inc r14

parse_done:
    lea rbp, [rip + lines_a]
    lea r12, [rip + lines_b]
    mov r13d, 1

sort_pass:
    cmp r13, r14
    jae sort_done
    xor ebx, ebx

sort_chunk:
    cmp rbx, r14
    jae sort_pass_end

    lea r8, [rbx + r13]
    cmp r8, r14
    cmova r8, r14

    lea r9, [rbx + r13 * 2]
    cmp r9, r14
    cmova r9, r14

    mov r10, rbx
    mov r11, r8
    mov r15, rbx

merge_loop:
    cmp r10, r8
    jae merge_copy_right
    cmp r11, r9
    jae merge_copy_left

    mov rax, r10
    shl rax, 4
    mov rdi, qword ptr [rbp + rax]
    mov rsi, qword ptr [rbp + rax + 8]

    mov rax, r11
    shl rax, 4
    mov rdx, qword ptr [rbp + rax]
    mov rcx, qword ptr [rbp + rax + 8]

    mov rax, rsi
    cmp rax, rcx
    cmova rax, rcx

cmp_qword:
    cmp rax, 8
    jb cmp_byte
    mov rsi, qword ptr [rdi]
    mov rcx, qword ptr [rdx]
    cmp rsi, rcx
    jne cmp_qword_diff
    add rdi, 8
    add rdx, 8
    sub rax, 8
    jmp cmp_qword

cmp_qword_diff:
    bswap rsi
    bswap rcx
    cmp rsi, rcx
    jb take_left
    jmp take_right

cmp_byte:
    test rax, rax
    jz cmp_len
    mov sil, byte ptr [rdi]
    cmp sil, byte ptr [rdx]
    jb take_left
    ja take_right
    inc rdi
    inc rdx
    dec rax
    jmp cmp_byte

cmp_len:
    mov rax, r10
    shl rax, 4
    mov rsi, qword ptr [rbp + rax + 8]
    mov rax, r11
    shl rax, 4
    cmp rsi, qword ptr [rbp + rax + 8]
    jbe take_left
    jmp take_right

take_left:
    mov rax, r10
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rcx, qword ptr [rbp + rax + 8]
    mov rax, r15
    shl rax, 4
    mov qword ptr [r12 + rax], rsi
    mov qword ptr [r12 + rax + 8], rcx
    inc r10
    inc r15
    jmp merge_loop

take_right:
    mov rax, r11
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rcx, qword ptr [rbp + rax + 8]
    mov rax, r15
    shl rax, 4
    mov qword ptr [r12 + rax], rsi
    mov qword ptr [r12 + rax + 8], rcx
    inc r11
    inc r15
    jmp merge_loop

merge_copy_left:
    cmp r10, r8
    jae merge_done
    mov rax, r10
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rcx, qword ptr [rbp + rax + 8]
    mov rax, r15
    shl rax, 4
    mov qword ptr [r12 + rax], rsi
    mov qword ptr [r12 + rax + 8], rcx
    inc r10
    inc r15
    jmp merge_copy_left

merge_copy_right:
    cmp r11, r9
    jae merge_done
    mov rax, r11
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rcx, qword ptr [rbp + rax + 8]
    mov rax, r15
    shl rax, 4
    mov qword ptr [r12 + rax], rsi
    mov qword ptr [r12 + rax + 8], rcx
    inc r11
    inc r15
    jmp merge_copy_right

merge_done:
    lea rbx, [rbx + r13 * 2]
    jmp sort_chunk

sort_pass_end:
    xchg rbp, r12
    shl r13, 1
    jmp sort_pass

sort_done:
    lea rdi, [rip + outbuf]
    xor ebx, ebx

out_loop:
    cmp rbx, r14
    jae out_done
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [rbp + rax]
    mov rcx, qword ptr [rbp + rax + 8]
    rep movsb
    mov byte ptr [rdi], 0x0A
    inc rdi
    inc rbx
    jmp out_loop

out_done:
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    mov r12, rdi
    xor ebx, ebx

write_loop:
    cmp rbx, r12
    jae exit_zero
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, rbx
    mov rdx, r12
    sub rdx, rbx
    syscall
    test rax, rax
    jle exit_zero
    add rbx, rax
    jmp write_loop

exit_zero:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
