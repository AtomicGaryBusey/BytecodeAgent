.section .rodata

ok_msg:
    .ascii "ok\n"
ok_msg_len = . - ok_msg

err_prefix:
    .ascii "error at "
err_prefix_len = . - err_prefix

.bss
    .balign 16
in_buf:
    .space 131072

.text
    .globl _start

_start:
    adrp x21, in_buf
    add x21, x21, :lo12:in_buf
    mov x19, x21
    movz x22, #2, lsl #16
    add x22, x21, x22

read_loop:
    sub x2, x22, x19
    cbz x2, read_done
    mov x0, #0
    mov x1, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, x19
    mov x19, x21

    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.ne fail

    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #ok_msg_len
    bl write_all

    mov x0, #0
    mov x8, #93
    svc #0

fail:
    sub sp, sp, #64
    add x4, sp, #64
    mov w1, #0x0A
    strb w1, [x4, #-1]!

    sub x0, x19, x21
    mov x3, #10
fail_dec_loop:
    udiv x1, x0, x3
    msub x2, x1, x3, x0
    add w2, w2, #0x30
    strb w2, [x4, #-1]!
    mov x0, x1
    cbnz x0, fail_dec_loop

    sub x4, x4, #err_prefix_len
    adrp x6, err_prefix
    add x6, x6, :lo12:err_prefix
    mov x7, x4
    mov w2, #err_prefix_len
fail_copy_loop:
    ldrb w3, [x6], #1
    strb w3, [x7], #1
    subs w2, w2, #1
    b.ne fail_copy_loop

    add x2, sp, #64
    sub x2, x2, x4
    mov x1, x4
    bl write_all

    mov x0, #1
    mov x8, #93
    svc #0

// Write x2 bytes from x1 to stdout (fd 1).
write_all:
    mov x3, x2
write_all_loop:
    cbz x3, write_all_done
    mov x0, #1
    mov x2, x3
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_all_done
    add x1, x1, x0
    sub x3, x3, x0
    b write_all_loop
write_all_done:
    ret

// Skip JSON whitespace (0x20, 0x09, 0x0A, 0x0D).
skip_ws:
skip_ws_loop:
    cmp x19, x20
    b.hs skip_ws_done
    ldrb w0, [x19]
    cmp w0, #0x20
    b.eq skip_ws_next
    cmp w0, #0x09
    b.eq skip_ws_next
    cmp w0, #0x0A
    b.eq skip_ws_next
    cmp w0, #0x0D
    b.ne skip_ws_done
skip_ws_next:
    add x19, x19, #1
    b skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x7B           // '{'
    b.eq parse_object
    cmp w0, #0x5B           // '['
    b.eq parse_array
    cmp w0, #0x22           // '"'
    b.eq parse_string
    cmp w0, #0x74           // 't'
    b.eq parse_true
    cmp w0, #0x66           // 'f'
    b.eq parse_false
    cmp w0, #0x6E           // 'n'
    b.eq parse_null
    cmp w0, #0x2D           // '-'
    b.eq parse_number
    sub w1, w0, #0x30       // '0'
    cmp w1, #9
    b.ls parse_number
    b fail

parse_true:
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x72           // 'r'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x75           // 'u'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x65           // 'e'
    b.ne fail
    add x19, x19, #1
    ret

parse_false:
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x61           // 'a'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x6C           // 'l'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x73           // 's'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x65           // 'e'
    b.ne fail
    add x19, x19, #1
    ret

parse_null:
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x75           // 'u'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x6C           // 'l'
    b.ne fail
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x6C           // 'l'
    b.ne fail
    add x19, x19, #1
    ret

parse_number:
    ldrb w0, [x19]
    cmp w0, #0x2D           // '-'
    b.ne pnum_int
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi fail
pnum_int:
    cmp w0, #0x30           // '0'
    b.ne pnum_digits
    add x19, x19, #1
    b pnum_after_int
pnum_digits:
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_done
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pnum_digits
pnum_after_int:
    cmp x19, x20
    b.hs pnum_done
    ldrb w0, [x19]
    cmp w0, #0x2E           // '.'
    b.ne pnum_exp
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi fail
pnum_frac_digits:
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_done
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pnum_frac_digits
pnum_exp:
    cmp w0, #0x65           // 'e'
    b.eq pnum_exp_body
    cmp w0, #0x45           // 'E'
    b.ne pnum_done
pnum_exp_body:
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x2B           // '+'
    b.eq pnum_exp_sign
    cmp w0, #0x2D           // '-'
    b.ne pnum_exp_req_digit
pnum_exp_sign:
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
pnum_exp_req_digit:
    sub w1, w0, #0x30
    cmp w1, #9
    b.hi fail
pnum_exp_digits:
    add x19, x19, #1
    cmp x19, x20
    b.hs pnum_done
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pnum_exp_digits
pnum_done:
    ret

parse_string:
    add x19, x19, #1        // skip opening '"'
pstr_loop:
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x22           // '"'
    b.eq pstr_end
    cmp w0, #0x20
    b.lo fail
    cmp w0, #0x5C           // '\'
    b.eq pstr_escape
    add x19, x19, #1
    b pstr_loop
pstr_escape:
    add x19, x19, #1
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x22           // '"'
    b.eq pstr_esc_ok
    cmp w0, #0x5C           // '\'
    b.eq pstr_esc_ok
    cmp w0, #0x2F           // '/'
    b.eq pstr_esc_ok
    cmp w0, #0x62           // 'b'
    b.eq pstr_esc_ok
    cmp w0, #0x66           // 'f'
    b.eq pstr_esc_ok
    cmp w0, #0x6E           // 'n'
    b.eq pstr_esc_ok
    cmp w0, #0x72           // 'r'
    b.eq pstr_esc_ok
    cmp w0, #0x74           // 't'
    b.eq pstr_esc_ok
    cmp w0, #0x75           // 'u'
    b.ne fail
    add x19, x19, #1
    mov w2, #4
pstr_hex_loop:
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    sub w1, w0, #0x30
    cmp w1, #9
    b.ls pstr_hex_ok
    orr w1, w0, #0x20
    sub w1, w1, #0x61       // 'a'
    cmp w1, #5
    b.hi fail
pstr_hex_ok:
    add x19, x19, #1
    subs w2, w2, #1
    b.ne pstr_hex_loop
    b pstr_loop
pstr_esc_ok:
    add x19, x19, #1
    b pstr_loop
pstr_end:
    add x19, x19, #1        // skip closing '"'
    ret

parse_array:
    str x30, [sp, #-16]!
    add x19, x19, #1        // skip '['
    bl skip_ws
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x5D           // ']'
    b.eq parr_end
parr_elem:
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x5D           // ']'
    b.eq parr_end
    cmp w0, #0x2C           // ','
    b.ne fail
    add x19, x19, #1
    bl skip_ws
    b parr_elem
parr_end:
    add x19, x19, #1        // skip ']'
    ldr x30, [sp], #16
    ret

parse_object:
    str x30, [sp, #-16]!
    add x19, x19, #1        // skip '{'
    bl skip_ws
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x7D           // '}'
    b.eq pobj_end
pobj_member:
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x22           // '"'
    b.ne fail
    bl parse_string
    bl skip_ws
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x3A           // ':'
    b.ne fail
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x19, x20
    b.hs fail
    ldrb w0, [x19]
    cmp w0, #0x7D           // '}'
    b.eq pobj_end
    cmp w0, #0x2C           // ','
    b.ne fail
    add x19, x19, #1
    bl skip_ws
    b pobj_member
pobj_end:
    add x19, x19, #1        // skip '}'
    ldr x30, [sp], #16
    ret
