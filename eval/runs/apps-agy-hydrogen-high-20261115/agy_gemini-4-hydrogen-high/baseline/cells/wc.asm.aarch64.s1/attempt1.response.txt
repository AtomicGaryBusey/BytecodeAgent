.equ buf_size, 65536

.text
.balign 4
.globl _start
_start:
    mov x19, #0             // lines
    mov x20, #0             // words
    mov x21, #0             // bytes
    mov w22, #0             // in_word flag

    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, #0              // fd = 0 (stdin)
    mov x1, x23             // buf
    mov x2, #buf_size       // count
    mov x8, #63             // SYS_read
    svc #0
    cmp x0, #0
    b.le eof

    add x21, x21, x0        // bytes += nread
    mov x1, x23             // ptr = buf
    add x2, x23, x0         // end = buf + nread

byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #0x0A           // '\n'
    b.ne 1f
    add x19, x19, #1        // lines++
    mov w22, #0             // in_word = 0
    b 4f
1:
    cmp w3, #0x20           // ' '
    b.eq 2f
    sub w4, w3, #0x09       // '\t'..'\r' -> 0..4
    cmp w4, #4
    b.hi 3f
2:
    mov w22, #0             // in_word = 0
    b 4f
3:
    cbnz w22, 4f
    add x20, x20, #1        // words++
    mov w22, #1             // in_word = 1
4:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

eof:
    sub sp, sp, #128
    add x24, sp, #128
    mov x1, x24

    mov w5, #0x0A           // '\n'
    strb w5, [x1, #-1]!

    mov x0, x21             // bytes
    bl format_u64

    mov w5, #0x20           // ' '
    strb w5, [x1, #-1]!

    mov x0, x20             // words
    bl format_u64

    mov w5, #0x20           // ' '
    strb w5, [x1, #-1]!

    mov x0, x19             // lines
    bl format_u64

    sub x2, x24, x1         // output length
write_loop:
    mov x0, #1              // fd = 1 (stdout)
    mov x8, #64             // SYS_write
    svc #0
    cmp x0, #0
    b.le done
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_loop

done:
    add sp, sp, #128
    mov x0, #0              // status = 0
    mov x8, #93             // SYS_exit
    svc #0

// Format unsigned 64-bit integer x0 in decimal backwards before x1.
// Returns updated start pointer in x1.
format_u64:
    mov x3, #10
1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #0x30       // '0'
    strb w5, [x1, #-1]!
    mov x0, x4
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .space buf_size
