.equ INBUF_CAP, 69632
.equ OUTBUF_CAP, 65536
.equ TAPE_SIZE, 32768
.equ PROG_CAP, 65536

.text
.globl _start

_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INBUF_CAP
    sub rdx, r12
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea r15, [rsi + r12]
    mov rcx, rsi

find_bang:
    cmp rcx, r15
    je no_bang
    cmp byte ptr [rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang

no_bang:
    mov r14, r15
    jmp parse_start

found_bang:
    lea r14, [rcx + 1]

parse_start:
    lea r8, [rip + prog]
    lea r9, [rip + bstack]
    xor edi, edi
    xor ebp, ebp

parse_loop:
    cmp rsi, rcx
    jae parse_done
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, '>'
    je p_right
    cmp al, '<'
    je p_left
    cmp al, '+'
    je p_plus
    cmp al, '-'
    je p_minus
    cmp al, '.'
    je p_dot
    cmp al, ','
    je p_comma
    cmp al, '['
    je p_lbrack
    cmp al, ']'
    je p_rbrack
    jmp parse_loop

p_right:
    xor eax, eax
    jmp p_store

p_left:
    mov eax, 1
    jmp p_store

p_plus:
    mov eax, 2
    jmp p_store

p_minus:
    mov eax, 3
    jmp p_store

p_dot:
    mov eax, 4
    jmp p_store

p_comma:
    mov eax, 5

p_store:
    mov dword ptr [r8 + rdi*4], eax
    inc rdi
    jmp parse_loop

p_lbrack:
    mov dword ptr [r9 + rbp*4], edi
    inc rbp
    mov dword ptr [r8 + rdi*4], 6
    inc rdi
    jmp parse_loop

p_rbrack:
    test rbp, rbp
    jz exit_error
    dec rbp
    mov edx, dword ptr [r9 + rbp*4]
    mov eax, edi
    shl eax, 8
    or eax, 6
    mov dword ptr [r8 + rdx*4], eax
    mov eax, edx
    shl eax, 8
    or eax, 7
    mov dword ptr [r8 + rdi*4], eax
    inc rdi
    jmp parse_loop

parse_done:
    test rbp, rbp
    jnz exit_error
    mov dword ptr [r8 + rdi*4], 8

    xor ebx, ebx
    mov r12, r8
    lea r13, [rip + tape]
    lea r9, [rip + outbuf]
    xor ebp, ebp
    lea r10, [rip + jump_table]

    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_right:
    inc r13
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_left:
    dec r13
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_plus:
    inc byte ptr [r13]
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_minus:
    dec byte ptr [r13]
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_dot:
    mov al, byte ptr [r13]
    mov byte ptr [r9 + rbp], al
    inc rbp
    cmp rbp, OUTBUF_CAP
    jb 1f
    call flush_out
1:
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_comma:
    xor eax, eax
    cmp r14, r15
    jae 1f
    mov al, byte ptr [r14]
    inc r14
1:
    mov byte ptr [r13], al
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_lbrack:
    cmp byte ptr [r13], 0
    jne 1f
    shr eax, 8
    mov ebx, eax
1:
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_rbrack:
    cmp byte ptr [r13], 0
    je 1f
    shr eax, 8
    mov ebx, eax
1:
    inc rbx
    mov eax, dword ptr [r12 + rbx*4]
    movzx ecx, al
    jmp qword ptr [r10 + rcx*8]

op_end:
    call flush_out
    mov eax, 60
    xor edi, edi
    syscall

exit_error:
    mov eax, 60
    mov edi, 1
    syscall

flush_out:
    test rbp, rbp
    jz 2f
    xor r8d, r8d
1:
    mov eax, 1
    mov edi, 1
    lea rsi, [r9 + r8]
    mov rdx, rbp
    sub rdx, r8
    syscall
    test rax, rax
    jle 2f
    add r8, rax
    cmp r8, rbp
    jb 1b
    xor ebp, ebp
2:
    ret

.section .rodata
.balign 8
jump_table:
    .quad op_right
    .quad op_left
    .quad op_plus
    .quad op_minus
    .quad op_dot
    .quad op_comma
    .quad op_lbrack
    .quad op_rbrack
    .quad op_end

.bss
.balign 16
inbuf:
    .zero INBUF_CAP
.balign 16
outbuf:
    .zero OUTBUF_CAP
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
prog:
    .zero (PROG_CAP + 1) * 4
.balign 16
bstack:
    .zero PROG_CAP * 4
