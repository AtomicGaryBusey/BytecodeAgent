.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    movz x21, #1, lsl #16
    add x21, x21, #16

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_ok
    sub x1, x20, #1
    ldrb w0, [x19, x1]
    cmp w0, #10
    b.eq process_lines
    mov w0, #10
    strb w0, [x19, x20]
    add x20, x20, #1

process_lines:
    mov x3, x19
    add x4, x19, x20

line_loop:
    cmp x3, x4
    b.hs write_loop
    mov x5, x3

find_nl:
    ldrb w0, [x5]
    cmp w0, #10
    b.eq found_nl
    add x5, x5, #1
    b find_nl

found_nl:
    mov x7, x3
    sub x6, x5, #1

rev_loop:
    cmp x7, x6
    b.hs rev_done
    ldrb w0, [x7]
    ldrb w1, [x6]
    strb w1, [x7], #1
    strb w0, [x6], #-1
    b rev_loop

rev_done:
    add x3, x5, #1
    b line_loop

write_loop:
    cbz x20, exit_ok
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x19, x19, x0
    sub x20, x20, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
inbuf:
    .space 65568
