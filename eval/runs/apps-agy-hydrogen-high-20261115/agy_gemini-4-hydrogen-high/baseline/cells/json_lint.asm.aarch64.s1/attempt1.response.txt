.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.equ BUF_SIZE, 0x20000
.equ STACK_SIZE, 4096

.equ TYPE_ARRAY, 0
.equ TYPE_OBJECT, 1

.equ CH_TAB, 0x09
.equ CH_LF, 0x0a
.equ CH_CR, 0x0d
.equ CH_SPACE, 0x20
.equ CH_QUOTE, 0x22
.equ CH_PLUS, 0x2b
.equ CH_COMMA, 0x2c
.equ CH_MINUS, 0x2d
.equ CH_DOT, 0x2e
.equ CH_SLASH, 0x2f
.equ CH_0, 0x30
.equ CH_1, 0x31
.equ CH_COLON, 0x3a
.equ CH_E, 0x45
.equ CH_LBRACK, 0x5b
.equ CH_BSLASH, 0x5c
.equ CH_RBRACK, 0x5d
.equ CH_a, 0x61
.equ CH_b, 0x62
.equ CH_e, 0x65
.equ CH_f, 0x66
.equ CH_n, 0x6e
.equ CH_r, 0x72
.equ CH_t, 0x74
.equ CH_u, 0x75
.equ CH_LBRACE, 0x7b
.equ CH_RBRACE, 0x7d

.text

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x21, #2, lsl #16

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x21, x19, x20
    mov x20, x19
    adrp x22, container_stack
    add x22, x22, :lo12:container_stack
    mov x23, #0

    bl skip_ws

parse_value:
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]

    cmp w0, #CH_QUOTE
    b.eq do_string

    cmp w0, #CH_LBRACE
    b.eq do_object

    cmp w0, #CH_LBRACK
    b.eq do_array

    cmp w0, #CH_t
    b.eq do_true

    cmp w0, #CH_f
    b.eq do_false

    cmp w0, #CH_n
    b.eq do_null

    cmp w0, #CH_MINUS
    b.eq do_number
    sub w1, w0, #CH_0
    cmp w1, #9
    b.ls do_number

    b report_error

do_object:
    add x20, x20, #1
    mov w1, #TYPE_OBJECT
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skip_ws
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_RBRACE
    b.ne parse_key
    add x20, x20, #1
    sub x23, x23, #1
    b after_value

parse_key:
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_QUOTE
    b.ne report_error
    bl parse_string_sub
    bl skip_ws
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_COLON
    b.ne report_error
    add x20, x20, #1
    bl skip_ws
    b parse_value

do_array:
    add x20, x20, #1
    mov w1, #TYPE_ARRAY
    strb w1, [x22, x23]
    add x23, x23, #1
    bl skip_ws
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_RBRACK
    b.ne parse_value
    add x20, x20, #1
    sub x23, x23, #1
    b after_value

do_string:
    bl parse_string_sub
    b after_value

do_true:
    adrp x1, lit_rue
    add x1, x1, :lo12:lit_rue
    mov w2, #3
    b match_lit

do_false:
    adrp x1, lit_alse
    add x1, x1, :lo12:lit_alse
    mov w2, #4
    b match_lit

do_null:
    adrp x1, lit_ull
    add x1, x1, :lo12:lit_ull
    mov w2, #3

match_lit:
    add x20, x20, #1
1:
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    ldrb w3, [x1], #1
    cmp w0, w3
    b.ne report_error
    add x20, x20, #1
    sub w2, w2, #1
    cbnz w2, 1b
    b after_value

do_number:
    ldrb w0, [x20]
    cmp w0, #CH_MINUS
    b.ne 1f
    add x20, x20, #1
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
1:
    cmp w0, #CH_0
    b.ne 2f
    add x20, x20, #1
    b num_frac
2:
    sub w1, w0, #CH_1
    cmp w1, #8
    b.hi report_error
    add x20, x20, #1
3:
    cmp x20, x21
    b.eq after_value
    ldrb w0, [x20]
    sub w1, w0, #CH_0
    cmp w1, #9
    b.hi num_frac
    add x20, x20, #1
    b 3b

num_frac:
    cmp x20, x21
    b.eq after_value
    ldrb w0, [x20]
    cmp w0, #CH_DOT
    b.ne num_exp
    add x20, x20, #1
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    sub w1, w0, #CH_0
    cmp w1, #9
    b.hi report_error
    add x20, x20, #1
4:
    cmp x20, x21
    b.eq after_value
    ldrb w0, [x20]
    sub w1, w0, #CH_0
    cmp w1, #9
    b.hi num_exp
    add x20, x20, #1
    b 4b

num_exp:
    cmp x20, x21
    b.eq after_value
    ldrb w0, [x20]
    cmp w0, #CH_e
    b.eq 5f
    cmp w0, #CH_E
    b.ne after_value
5:
    add x20, x20, #1
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_PLUS
    b.eq 6f
    cmp w0, #CH_MINUS
    b.ne 7f
6:
    add x20, x20, #1
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
7:
    sub w1, w0, #CH_0
    cmp w1, #9
    b.hi report_error
    add x20, x20, #1
8:
    cmp x20, x21
    b.eq after_value
    ldrb w0, [x20]
    sub w1, w0, #CH_0
    cmp w1, #9
    b.hi after_value
    add x20, x20, #1
    b 8b

after_value:
    bl skip_ws
    cbz x23, check_eof
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    sub x1, x23, #1
    ldrb w1, [x22, x1]
    cmp w1, #TYPE_ARRAY
    b.eq 1f

    cmp w0, #CH_RBRACE
    b.eq 2f
    cmp w0, #CH_COMMA
    b.ne report_error
    add x20, x20, #1
    bl skip_ws
    b parse_key
1:
    cmp w0, #CH_RBRACK
    b.eq 2f
    cmp w0, #CH_COMMA
    b.ne report_error
    add x20, x20, #1
    bl skip_ws
    b parse_value
2:
    add x20, x20, #1
    sub x23, x23, #1
    b after_value

check_eof:
    cmp x20, x21
    b.ne report_error
    b report_ok

parse_string_sub:
    add x20, x20, #1
1:
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_SPACE
    b.lo report_error
    cmp w0, #CH_QUOTE
    b.eq 2f
    cmp w0, #CH_BSLASH
    b.eq 3f
    add x20, x20, #1
    b 1b
2:
    add x20, x20, #1
    ret
3:
    add x20, x20, #1
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    cmp w0, #CH_QUOTE
    b.eq 4f
    cmp w0, #CH_BSLASH
    b.eq 4f
    cmp w0, #CH_SLASH
    b.eq 4f
    cmp w0, #CH_b
    b.eq 4f
    cmp w0, #CH_f
    b.eq 4f
    cmp w0, #CH_n
    b.eq 4f
    cmp w0, #CH_r
    b.eq 4f
    cmp w0, #CH_t
    b.eq 4f
    cmp w0, #CH_u
    b.ne report_error

    add x20, x20, #1
    mov w2, #4
5:
    cmp x20, x21
    b.eq report_error
    ldrb w0, [x20]
    sub w1, w0, #CH_0
    cmp w1, #9
    b.ls 6f
    orr w1, w0, #0x20
    sub w1, w1, #CH_a
    cmp w1, #5
    b.hi report_error
6:
    add x20, x20, #1
    sub w2, w2, #1
    cbnz w2, 5b
    b 1b
4:
    add x20, x20, #1
    b 1b

skip_ws:
1:
    cmp x20, x21
    b.eq 2f
    ldrb w0, [x20]
    cmp w0, #CH_SPACE
    b.hi 2f
    b.eq 3f
    cmp w0, #CH_LF
    b.eq 3f
    cmp w0, #CH_CR
    b.eq 3f
    cmp w0, #CH_TAB
    b.ne 2f
3:
    add x20, x20, #1
    b 1b
2:
    ret

report_ok:
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    bl write_all
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

report_error:
    sub x0, x20, x19
    sub sp, sp, #64
    add x15, sp, #64
    mov x10, x15
    mov w11, #CH_LF
    strb w11, [x10, #-1]!
    mov x12, #10
1:
    udiv x13, x0, x12
    msub x14, x13, x12, x0
    add w14, w14, #CH_0
    strb w14, [x10, #-1]!
    mov x0, x13
    cbnz x0, 1b

    adrp x11, err_prefix
    add x11, x11, :lo12:err_prefix
    sub x10, x10, #9
    mov x12, #0
2:
    ldrb w13, [x11, x12]
    strb w13, [x10, x12]
    add x12, x12, #1
    cmp x12, #9
    b.lo 2b

    mov x1, x10
    sub x2, x15, x10
    bl write_all
    add sp, sp, #64
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

write_all:
1:
    cbz x2, 2f
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 2f
    add x1, x1, x0
    sub x2, x2, x0
    b 1b
2:
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_rue:
    .ascii "rue"
lit_alse:
    .ascii "alse"
lit_ull:
    .ascii "ull"

.bss
.balign 16
input_buf:
    .space BUF_SIZE
.balign 16
container_stack:
    .space STACK_SIZE
