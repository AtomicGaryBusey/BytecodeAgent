.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .balign 16
inbuf:
    .space 65536
outbuf:
    .space 131072

.text
.globl _start
_start:
    xor r12d, r12d
    lea r13, [rip + inbuf]

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_zero

    mov rsi, r13
    lea rdi, [rip + outbuf]
    mov r14, rdi
    lea r8, [rip + b64_table]
    xor r9d, r9d

encode_loop:
    cmp r12, 3
    jb encode_tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx

    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], cl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    movzx ecx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], cl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    movzx ecx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 2], cl

    and eax, 0x3F
    movzx ecx, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], cl

    add rsi, 3
    sub r12, 3
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne encode_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encode_loop

encode_tail:
    test r12, r12
    jz finish_line

    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r12, 1
    je tail_one

    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx

    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], cl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    movzx ecx, byte ptr [r8 + rdx]
    mov byte ptr [rdi + 1], cl

    shr eax, 6
    and eax, 0x3F
    movzx ecx, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], cl

    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4
    add r9d, 4
    jmp finish_line

tail_one:
    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], cl

    shr eax, 12
    and eax, 0x3F
    movzx ecx, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], cl

    mov byte ptr [rdi + 2], 0x3D
    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4
    add r9d, 4

finish_line:
    test r9d, r9d
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    mov rbx, rdi
    sub rbx, r14

write_loop:
    test rbx, rbx
    jz exit_zero
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, rbx
    syscall
    test rax, rax
    jle exit_zero
    add r14, rax
    sub rbx, rax
    jmp write_loop

exit_zero:
    mov eax, 60
    xor edi, edi
    syscall
