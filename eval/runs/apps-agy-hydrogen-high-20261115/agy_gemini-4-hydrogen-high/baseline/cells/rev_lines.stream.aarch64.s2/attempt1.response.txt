.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_READ_CAP, 131072
.equ BUF_SIZE, 131088

.bss
.balign 16
buf:
    .zero BUF_SIZE

.text
.balign 4
.globl _start
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, x19
    movz x21, #2, lsl #16
    add x21, x19, x21

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cmp x20, x19
    b.eq exit_prog

    sub x1, x20, #1
    ldrb w0, [x1]
    cmp w0, #10
    b.eq start_reverse
    mov w0, #10
    strb w0, [x20], #1

start_reverse:
    mov x22, x19

line_loop:
    cmp x22, x20
    b.hs write_all
    mov x23, x22

find_nl:
    ldrb w0, [x22], #1
    cmp w0, #10
    b.ne find_nl
    sub x24, x22, #2
    b swap_cond

swap_loop:
    ldrb w0, [x23]
    ldrb w1, [x24]
    strb w1, [x23], #1
    strb w0, [x24], #-1

swap_cond:
    cmp x23, x24
    b.lo swap_loop
    b line_loop

write_all:
    mov x22, x19

write_loop:
    sub x2, x20, x22
    cbz x2, exit_prog
    mov x0, #1
    mov x1, x22
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le exit_prog
    add x22, x22, x0
    b write_loop

exit_prog:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
