.equ BUF_CAP, 131072
.equ READ_MAX, 131071

.bss
.p2align 4
in_buf:
    .zero BUF_CAP

.text
.globl _start
_start:
    lea rbx, [rip + in_buf]
    xor r12d, r12d

read_loop:
    mov rdx, READ_MAX
    sub rdx, r12
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz done_exit

    mov rsi, rbx
    lea r13, [rbx + r12]

line_loop:
    cmp rsi, r13
    jae write_start

    mov rdi, rsi
find_nl:
    cmp rdi, r13
    jae rev_start
    cmp byte ptr [rdi], 10
    je rev_start
    inc rdi
    jmp find_nl

rev_start:
    mov rax, rsi
    lea rcx, [rdi - 1]
rev_loop:
    cmp rax, rcx
    jae rev_done
    mov dl, byte ptr [rax]
    mov dh, byte ptr [rcx]
    mov byte ptr [rax], dh
    mov byte ptr [rcx], dl
    inc rax
    dec rcx
    jmp rev_loop

rev_done:
    cmp rdi, r13
    jae append_nl
    lea rsi, [rdi + 1]
    jmp line_loop

append_nl:
    mov byte ptr [r13], 10
    inc r12

write_start:
    xor r13d, r13d

write_loop:
    cmp r13, r12
    jae done_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r13]
    mov rdx, r12
    sub rdx, r13
    syscall
    test rax, rax
    jle done_exit
    add r13, rax
    jmp write_loop

done_exit:
    mov eax, 60
    xor edi, edi
    syscall
