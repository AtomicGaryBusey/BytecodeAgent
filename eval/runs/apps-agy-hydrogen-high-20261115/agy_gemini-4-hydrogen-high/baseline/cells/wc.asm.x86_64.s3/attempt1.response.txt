.globl _start

.bss
    .balign 16
in_buf:
    .space 65536
out_buf:
    .space 128

.text
_start:
    # r12 = lines, r13 = words, r14 = bytes, r15 = in_word (all 0 at entry)
read_loop:
    xor eax, eax                # sys_read = 0
    xor edi, edi                # fd = 0 (stdin)
    lea rsi, [rip + in_buf]
    mov edx, 65536
    syscall

    test rax, rax
    jle done_reading

    add r14, rax                # bytes += count
    mov rcx, rax                # loop counter = count
    lea rsi, [rip + in_buf]

byte_loop:
    movzx eax, byte ptr [rsi]
    cmp eax, 10
    jne 1f
    inc r12                     # lines++
    xor r15d, r15d              # in_word = 0
    jmp 2f
1:
    cmp eax, 32
    je 3f
    lea edx, [rax - 9]
    cmp edx, 4
    jbe 3f
    mov eax, 1
    sub eax, r15d
    add r13, rax                # words += (1 - in_word)
    mov r15d, 1                 # in_word = 1
    jmp 2f
3:
    xor r15d, r15d              # in_word = 0
2:
    inc rsi
    dec rcx
    jnz byte_loop

    jmp read_loop

done_reading:
    lea rdi, [rip + out_buf]

    mov rax, r12
    call fmt_u64
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call fmt_u64
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call fmt_u64
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    mov eax, 1                  # sys_write = 1
    mov edi, 1                  # fd = 1 (stdout)
    syscall
    cmp rax, rdx
    jge done_write
    test rax, rax
    jle done_write
    add rsi, rax
    sub rdx, rax
    jmp write_loop

done_write:
    mov eax, 60                 # sys_exit = 60
    xor edi, edi                # status = 0
    syscall

fmt_u64:
    xor ecx, ecx
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, 48
    push rdx
    inc rcx
    test rax, rax
    jnz 1b
2:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    dec rcx
    jnz 2b
    ret
