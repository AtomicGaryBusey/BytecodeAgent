.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ IN_CAP, 65536
.equ OUT_CAP, 131072
.equ MAX_LINES, 32768

.bss
.balign 16
in_buf:
    .space IN_CAP
.balign 16
out_buf:
    .space OUT_CAP
.balign 8
lines:
    .space MAX_LINES * 8

.text
.globl _start

_start:
    xor ebx, ebx
    lea r12, [rip + in_buf]

read_loop:
    mov rdx, IN_CAP
    sub rdx, rbx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_ok

    lea r13, [rip + lines]
    xor r14d, r14d
    xor ecx, ecx
    xor edx, edx

parse_loop:
    cmp rdx, rbx
    jae parse_tail
    cmp byte ptr [r12 + rdx], 10
    jne parse_next
    mov rax, rdx
    sub rax, rcx
    shl rax, 32
    or rax, rcx
    mov qword ptr [r13 + r14 * 8], rax
    inc r14
    lea rcx, [rdx + 1]

parse_next:
    inc rdx
    jmp parse_loop

parse_tail:
    cmp rcx, rbx
    jae parse_done
    mov rax, rbx
    sub rax, rcx
    shl rax, 32
    or rax, rcx
    mov qword ptr [r13 + r14 * 8], rax
    inc r14

parse_done:
    test r14, r14
    jz exit_ok
    cmp r14, 1
    jbe sort_done

    mov r15, r14
    shr r15, 1

heapify_loop:
    test r15, r15
    jz sort_loop_init
    dec r15
    mov rdi, r15
    mov rsi, r14
    call sift_down
    jmp heapify_loop

sort_loop_init:
    mov r15, r14

sort_loop:
    cmp r15, 1
    jbe sort_done
    dec r15
    mov rax, qword ptr [r13]
    mov rdx, qword ptr [r13 + r15 * 8]
    mov qword ptr [r13], rdx
    mov qword ptr [r13 + r15 * 8], rax
    xor edi, edi
    mov rsi, r15
    call sift_down
    jmp sort_loop

sort_done:
    xor ebx, ebx
    lea rbp, [rip + out_buf]
    xor r15d, r15d

out_loop:
    cmp r15, r14
    jae out_flush
    mov rax, qword ptr [r13 + r15 * 8]
    mov edx, eax
    lea rsi, [r12 + rdx]
    mov rcx, rax
    shr rcx, 32
    lea rax, [rbx + rcx + 1]
    cmp rax, OUT_CAP
    jbe 1f
    push rsi
    push rcx
    mov rdi, rbp
    mov rsi, rbx
    call write_all
    xor ebx, ebx
    pop rcx
    pop rsi
1:
    lea rdi, [rbp + rbx]
    lea rbx, [rbx + rcx + 1]
    rep movsb
    mov byte ptr [rdi], 10
    inc r15
    jmp out_loop

out_flush:
    test rbx, rbx
    jz exit_ok
    mov rdi, rbp
    mov rsi, rbx
    call write_all

exit_ok:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

sift_down:
    push rbx
    push rbp
    push r14
    mov rbx, rdi
    mov rbp, rsi
1:
    lea r14, [rbx + rbx + 1]
    cmp r14, rbp
    jae 3f
    lea rax, [r14 + 1]
    cmp rax, rbp
    jae 2f
    mov rdi, qword ptr [r13 + r14 * 8]
    mov rsi, qword ptr [r13 + rax * 8]
    call cmp_lines
    test eax, eax
    jge 2f
    inc r14
2:
    mov rdi, qword ptr [r13 + rbx * 8]
    mov rsi, qword ptr [r13 + r14 * 8]
    call cmp_lines
    test eax, eax
    jge 3f
    mov rax, qword ptr [r13 + rbx * 8]
    mov rdx, qword ptr [r13 + r14 * 8]
    mov qword ptr [r13 + rbx * 8], rdx
    mov qword ptr [r13 + r14 * 8], rax
    mov rbx, r14
    jmp 1b
3:
    pop r14
    pop rbp
    pop rbx
    ret

cmp_lines:
    mov eax, edi
    lea r8, [r12 + rax]
    mov r9, rdi
    shr r9, 32
    mov eax, esi
    lea r10, [r12 + rax]
    mov r11, rsi
    shr r11, 32

    mov rcx, r9
    cmp rcx, r11
    jbe 1f
    mov rcx, r11
1:
    test rcx, rcx
    jz 3f
2:
    movzx eax, byte ptr [r8]
    movzx edx, byte ptr [r10]
    sub eax, edx
    jnz 4f
    inc r8
    inc r10
    dec rcx
    jnz 2b
3:
    cmp r9, r11
    jb 5f
    ja 6f
    xor eax, eax
4:
    ret
5:
    mov eax, -1
    ret
6:
    mov eax, 1
    ret

write_all:
    mov r8, rdi
    mov r9, rsi
1:
    test r9, r9
    jz 2f
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jle 2f
    add r8, rax
    sub r9, rax
    jmp 1b
2:
    ret
