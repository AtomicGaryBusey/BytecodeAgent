.globl _start

.text
_start:
    lea rbx, [rip + inbuf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65600
    sub edx, r12d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_zero

    cmp byte ptr [rbx + r12 - 1], 10
    je lines_start
    mov byte ptr [rbx + r12], 10
    inc r12

lines_start:
    mov rsi, rbx
    lea r13, [rbx + r12]

line_loop:
    cmp rsi, r13
    jae write_all
    mov rdi, rsi

find_nl:
    cmp byte ptr [rdi], 10
    je found_nl
    inc rdi
    jmp find_nl

found_nl:
    mov rdx, rsi
    lea rcx, [rdi - 1]

rev_loop:
    cmp rdx, rcx
    jae rev_done
    mov al, byte ptr [rdx]
    mov r8b, byte ptr [rcx]
    mov byte ptr [rdx], r8b
    mov byte ptr [rcx], al
    inc rdx
    dec rcx
    jmp rev_loop

rev_done:
    lea rsi, [rdi + 1]
    jmp line_loop

write_all:
    xor r13d, r13d

write_loop:
    cmp r13, r12
    jae exit_zero
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r13]
    mov rdx, r12
    sub rdx, r13
    syscall
    test rax, rax
    jle exit_zero
    add r13, rax
    jmp write_loop

exit_zero:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .space 65600
