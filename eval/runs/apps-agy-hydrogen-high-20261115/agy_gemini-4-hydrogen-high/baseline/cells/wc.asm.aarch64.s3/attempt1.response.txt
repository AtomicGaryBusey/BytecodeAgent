.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 128

.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #INBUF_SIZE
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done

    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0

byte_loop:
    cmp x1, x2
    b.hs read_loop
    ldrb w3, [x1], #1
    cmp w3, #0x0a
    cinc x19, x19, eq
    cmp w3, #0x20
    b.eq is_space
    sub w4, w3, #0x09
    cmp w4, #4
    b.ls is_space

    cmp w22, #0
    cinc x20, x20, eq
    mov w22, #1
    b byte_loop

is_space:
    mov w22, #0
    b byte_loop

read_done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    add x25, x24, #OUTBUF_SIZE
    mov x1, x25

    mov w0, #0x0a
    strb w0, [x1, #-1]!

    mov x0, x21
    bl fmt_u64_rev

    mov w0, #0x20
    strb w0, [x1, #-1]!

    mov x0, x20
    bl fmt_u64_rev

    mov w0, #0x20
    strb w0, [x1, #-1]!

    mov x0, x19
    bl fmt_u64_rev

    sub x2, x25, x1
write_loop:
    cbz x2, exit_ok
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

fmt_u64_rev:
    mov x2, #10
1:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add w4, w4, #0x30
    strb w4, [x1, #-1]!
    mov x0, x3
    cbnz x0, 1b
    ret

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
