.globl _start

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 128

.text
_start:
    xor r12d, r12d          # lines = 0
    xor r13d, r13d          # words = 0
    xor r14d, r14d          # bytes = 0
    xor r15d, r15d          # in_word = 0

read_loop:
    xor eax, eax            # sys_read = 0
    xor edi, edi            # fd = 0 (stdin)
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall

    test rax, rax
    jle read_done

    add r14, rax
    lea rsi, [rip + inbuf]
    lea rcx, [rsi + rax]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 0x0A
    je is_newline

    cmp al, 0x20
    je is_space

    sub al, 0x09
    cmp al, 0x04
    jbe is_space

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

is_newline:
    inc r12
is_space:
    xor r15d, r15d

next_byte:
    cmp rsi, rcx
    jb byte_loop
    jmp read_loop

read_done:
    lea rdi, [rip + outbuf + 128]
    mov r8, rdi

    dec rdi
    mov byte ptr [rdi], 0x0A

    mov rax, r14
    call format_u64

    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r13
    call format_u64

    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r12
    call format_u64

    mov rsi, rdi
    mov rdx, r8
    sub rdx, rdi

write_loop:
    mov eax, 1              # sys_write = 1
    mov edi, 1              # fd = 1 (stdout)
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    ja write_loop

exit_ok:
    mov eax, 60             # sys_exit = 60
    xor edi, edi            # status = 0
    syscall

format_u64:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret
