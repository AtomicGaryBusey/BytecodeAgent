.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ STDIN, 0
.equ STDOUT, 1
.equ TOK_ARRAY, 0
.equ TOK_OBJECT, 1

.section .rodata
msg_ok:
    .ascii "ok\n"
.equ msg_ok_len, . - msg_ok

msg_err:
    .ascii "error at "
.equ msg_err_len, . - msg_err

lit_rue:
    .asciz "rue"
lit_alse:
    .asciz "alse"
lit_ull:
    .asciz "ull"

.bss
.balign 16
input_buf:
    .space 131072
stack_buf:
    .space 65536
out_buf:
    .space 64
out_buf_end:

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    mov x21, x19
    movz x25, #2, lsl #16

read_loop:
    sub x2, x25, x21
    add x2, x2, x19
    cbz x2, read_done
    mov x0, #STDIN
    mov x1, x21
    mov x8, #SYS_READ
    svc #0
    cbz x0, read_done
    cmp x0, #0
    b.lt read_err
    add x21, x21, x0
    b read_loop

read_err:
    cmn x0, #4
    b.eq read_loop

read_done:
    movz x22, #0x2600
    movk x22, #1, lsl #32
    adrp x23, stack_buf
    add x23, x23, :lo12:stack_buf
    mov x24, x23

expect_value:
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'{'
    b.eq do_open_obj
    cmp w0, #'['
    b.eq do_open_arr
    cmp w0, #0x22
    b.eq do_string
    cmp w0, #'-'
    b.eq parse_number
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls parse_number
    cmp w0, #'t'
    b.eq do_true
    cmp w0, #'f'
    b.eq do_false
    cmp w0, #'n'
    b.eq do_null
    b fail

do_open_obj:
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'}'
    b.eq 1f
    mov w1, #TOK_OBJECT
    strb w1, [x24], #1
    b obj_key
1:
    add x20, x20, #1
    b after_value

do_open_arr:
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #']'
    b.eq 1f
    mov w1, #TOK_ARRAY
    strb w1, [x24], #1
    b expect_value
1:
    add x20, x20, #1
    b after_value

do_string:
    bl parse_string
    b after_value

do_true:
    adrp x2, lit_rue
    add x2, x2, :lo12:lit_rue
    b match_lit

do_false:
    adrp x2, lit_alse
    add x2, x2, :lo12:lit_alse
    b match_lit

do_null:
    adrp x2, lit_ull
    add x2, x2, :lo12:lit_ull

match_lit:
    add x20, x20, #1
1:
    ldrb w1, [x2], #1
    cbz w1, after_value
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, w1
    b.ne fail
    add x20, x20, #1
    b 1b

obj_key:
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #0x22
    b.ne fail
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #':'
    b.ne fail
    add x20, x20, #1
    b expect_value

after_value:
    bl skip_ws
    cmp x24, x23
    b.eq top_done
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    ldrb w1, [x24, #-1]
    cbnz w1, after_obj_val
    cmp w0, #','
    b.eq 1f
    cmp w0, #']'
    b.ne fail
    add x20, x20, #1
    sub x24, x24, #1
    b after_value
1:
    add x20, x20, #1
    b expect_value

after_obj_val:
    cmp w0, #','
    b.eq 1f
    cmp w0, #'}'
    b.ne fail
    add x20, x20, #1
    sub x24, x24, #1
    b after_value
1:
    add x20, x20, #1
    b obj_key

top_done:
    cmp x20, x21
    b.ne fail
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #msg_ok_len
    mov x26, #0
    b write_and_exit

// Leaf helper: advance x20 past JSON whitespace (0x09, 0x0A, 0x0D, 0x20)
skip_ws:
    cmp x20, x21
    b.hs 2f
    ldrb w0, [x20]
    cmp w0, #0x20
    b.hi 2f
    lsr x1, x22, x0
    tbz x1, #0, 2f
    add x20, x20, #1
    b skip_ws
2:
    ret

// Leaf helper: parse JSON string starting at opening '"'
parse_string:
    add x20, x20, #1
str_loop:
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #0x20
    b.lo fail
    cmp w0, #0x22
    b.eq str_done
    cmp w0, #0x5C
    b.eq str_esc
    add x20, x20, #1
    b str_loop
str_done:
    add x20, x20, #1
    ret

str_esc:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq str_esc_one
    cmp w0, #0x5C
    b.eq str_esc_one
    cmp w0, #0x2F
    b.eq str_esc_one
    cmp w0, #'b'
    b.eq str_esc_one
    cmp w0, #'f'
    b.eq str_esc_one
    cmp w0, #'n'
    b.eq str_esc_one
    cmp w0, #'r'
    b.eq str_esc_one
    cmp w0, #'t'
    b.eq str_esc_one
    cmp w0, #'u'
    b.eq str_esc_u
    b fail

str_esc_one:
    add x20, x20, #1
    b str_loop

str_esc_u:
    add x20, x20, #1
    mov w2, #4
str_hex_loop:
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls str_hex_ok
    orr w1, w0, #0x20
    sub w1, w1, #'a'
    cmp w1, #5
    b.hi fail
str_hex_ok:
    add x20, x20, #1
    subs w2, w2, #1
    b.ne str_hex_loop
    b str_loop

// Parse JSON number and jump to after_value
parse_number:
    cmp w0, #'-'
    b.ne num_int
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]

num_int:
    cmp w0, #'0'
    b.ne num_1_9
    add x20, x20, #1
    cmp x20, x21
    b.hs after_value
    ldrb w0, [x20]
    b num_check_frac

num_1_9:
    sub w1, w0, #'1'
    cmp w1, #8
    b.hi fail
num_int_loop:
    add x20, x20, #1
    cmp x20, x21
    b.hs after_value
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls num_int_loop

num_check_frac:
    cmp w0, #'.'
    b.ne num_check_exp
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
num_frac_loop:
    add x20, x20, #1
    cmp x20, x21
    b.hs after_value
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls num_frac_loop

num_check_exp:
    orr w1, w0, #0x20
    cmp w1, #'e'
    b.ne after_value
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
    cmp w0, #'+'
    b.eq num_exp_sign
    cmp w0, #'-'
    b.ne num_exp_first_digit
num_exp_sign:
    add x20, x20, #1
    cmp x20, x21
    b.hs fail
    ldrb w0, [x20]
num_exp_first_digit:
    sub w1, w0, #'0'
    cmp w1, #9
    b.hi fail
num_exp_loop:
    add x20, x20, #1
    cmp x20, x21
    b.hs after_value
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #9
    b.ls num_exp_loop
    b after_value

fail:
    sub x0, x20, x19
    adrp x2, out_buf_end
    add x2, x2, :lo12:out_buf_end
    mov x6, x2
    mov w3, #0x0A
    strb w3, [x2, #-1]!
    mov x3, #10
1:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #'0'
    strb w5, [x2, #-1]!
    mov x0, x4
    cbnz x0, 1b

    adrp x3, msg_err
    add x3, x3, :lo12:msg_err
    mov w4, #msg_err_len
    sub x2, x2, x4
    mov x1, x2
2:
    ldrb w5, [x3], #1
    strb w5, [x1], #1
    subs w4, w4, #1
    b.ne 2b

    mov x1, x2
    sub x2, x6, x2
    mov x26, #1

write_and_exit:
    mov x27, x1
    mov x28, x2
1:
    cbz x28, 2f
    mov x0, #STDOUT
    mov x1, x27
    mov x2, x28
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.lt 3f
    cbz x0, 2f
    add x27, x27, x0
    sub x28, x28, x0
    b 1b
3:
    cmn x0, #4
    b.eq 1b
2:
    mov x0, x26
    mov x8, #SYS_EXIT
    svc #0
