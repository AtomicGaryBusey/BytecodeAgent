    .globl _start
    .text
_start:
    // Read all of stdin into in_buf
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0
    movz x21, #1, lsl #16
    add x21, x21, #64

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    // Parse program, find '!', check brackets, build jump table
    add x27, x19, x20
    mov x25, x27
    mov x0, #0
    adrp x1, prog_buf
    add x1, x1, :lo12:prog_buf
    mov x2, #0
    adrp x3, bstack
    add x3, x3, :lo12:bstack
    mov x4, #0
    adrp x5, jumps
    add x5, x5, :lo12:jumps

parse_loop:
    cmp x0, x20
    b.eq parse_end
    ldrb w6, [x19, x0]
    add x0, x0, #1
    cmp w6, #0x21
    b.ne 1f
    add x25, x19, x0
    b parse_end
1:
    cmp w6, #0x3e
    b.eq is_gt
    cmp w6, #0x3c
    b.eq is_lt
    cmp w6, #0x2b
    b.eq is_plus
    cmp w6, #0x2d
    b.eq is_minus
    cmp w6, #0x2e
    b.eq is_dot
    cmp w6, #0x2c
    b.eq is_comma
    cmp w6, #0x5b
    b.eq is_lbrack
    cmp w6, #0x5d
    b.eq is_rbrack
    b parse_loop

is_gt:
    mov w7, #0
    b emit_op
is_lt:
    mov w7, #1
    b emit_op
is_plus:
    mov w7, #2
    b emit_op
is_minus:
    mov w7, #3
    b emit_op
is_dot:
    mov w7, #4
    b emit_op
is_comma:
    mov w7, #5
    b emit_op
is_lbrack:
    str w2, [x3, x4, lsl #2]
    add x4, x4, #1
    mov w7, #6
    b emit_op
is_rbrack:
    cbz x4, bracket_err
    sub x4, x4, #1
    ldr w8, [x3, x4, lsl #2]
    str w2, [x5, x8, lsl #2]
    str w8, [x5, x2, lsl #2]
    mov w7, #7

emit_op:
    strb w7, [x1, x2]
    add x2, x2, #1
    b parse_loop

parse_end:
    cbnz x4, bracket_err
    mov w7, #8
    strb w7, [x1, x2]

    // Initialize VM registers
    mov x19, x1
    mov x20, #0
    adrp x21, out_buf
    add x21, x21, :lo12:out_buf
    adrp x22, tape
    add x22, x22, :lo12:tape
    mov x23, x21
    mov x24, x5
    adr x26, dispatch_table
    movz x28, #1, lsl #16
    add x28, x21, x28

loop_top:
    ldrb w8, [x19, x20]
    add x9, x26, x8, lsl #2
    br x9

dispatch_table:
    b op_gt
    b op_lt
    b op_plus
    b op_minus
    b op_dot
    b op_comma
    b op_lbrack
    b op_rbrack
    b done

op_gt:
    add x22, x22, #1
    add x20, x20, #1
    b loop_top

op_lt:
    sub x22, x22, #1
    add x20, x20, #1
    b loop_top

op_plus:
    ldrb w8, [x22]
    add w8, w8, #1
    strb w8, [x22]
    add x20, x20, #1
    b loop_top

op_minus:
    ldrb w8, [x22]
    sub w8, w8, #1
    strb w8, [x22]
    add x20, x20, #1
    b loop_top

op_dot:
    cmp x23, x28
    b.lo 1f
    bl flush_out
1:
    ldrb w8, [x22]
    strb w8, [x23], #1
    add x20, x20, #1
    b loop_top

op_comma:
    cmp x25, x27
    b.hs 1f
    ldrb w8, [x25], #1
    b 2f
1:
    mov w8, #0
2:
    strb w8, [x22]
    add x20, x20, #1
    b loop_top

op_lbrack:
    ldrb w8, [x22]
    cbnz w8, 1f
    ldr w20, [x24, x20, lsl #2]
1:
    add x20, x20, #1
    b loop_top

op_rbrack:
    ldrb w8, [x22]
    cbz w8, 1f
    ldr w20, [x24, x20, lsl #2]
1:
    add x20, x20, #1
    b loop_top

done:
    bl flush_out
    mov x0, #0
    mov x8, #93
    svc #0

bracket_err:
    mov x0, #1
    mov x8, #93
    svc #0

flush_out:
    sub x9, x23, x21
    cbz x9, 2f
    mov x10, x21
1:
    mov x0, #1
    mov x1, x10
    mov x2, x9
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x10, x10, x0
    sub x9, x9, x0
    cbnz x9, 1b
2:
    mov x23, x21
    ret
