.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
inbuf:
    .space 57008
outbuf:
    .space 77056

.text
_start:
main_loop:
    xor ebp, ebp

read_loop:
    mov edx, 57000
    sub rdx, rbp
    jz process_full_chunk
    lea rsi, [rip + inbuf]
    add rsi, rbp
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle eof_reached
    add rbp, rax
    jmp read_loop

process_full_chunk:
    call encode_and_write
    jmp main_loop

eof_reached:
    test rbp, rbp
    jz done
    call encode_and_write

done:
    xor edi, edi
    mov eax, 60
    syscall

encode_and_write:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r8, [rip + b64_table]
    mov r9, rbp

line_loop:
    mov r10, r9
    cmp r10, 57
    jbe 1f
    mov r10, 57
1:
    sub r9, r10

group_loop:
    cmp r10, 3
    jb line_tail
    mov eax, dword ptr [rsi]
    bswap eax
    shr eax, 8

    mov ecx, eax
    and ecx, 0x3F
    movzx edx, byte ptr [r8 + rcx]
    shl edx, 8

    shr eax, 6
    mov ecx, eax
    and ecx, 0x3F
    mov dl, byte ptr [r8 + rcx]
    shl edx, 8

    shr eax, 6
    mov ecx, eax
    and ecx, 0x3F
    mov dl, byte ptr [r8 + rcx]
    shl edx, 8

    shr eax, 6
    mov dl, byte ptr [r8 + rax]
    mov dword ptr [rdi], edx

    add rsi, 3
    add rdi, 4
    sub r10, 3
    jmp group_loop

line_tail:
    test r10, r10
    jz emit_nl
    cmp r10, 1
    jne tail_two

    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 2
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], dl
    and eax, 3
    shl eax, 4
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], dl
    mov word ptr [rdi + 2], 0x3D3D
    add rdi, 4
    jmp emit_nl

tail_two:
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    shl eax, 8
    or eax, ecx
    mov ecx, eax
    shr ecx, 10
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], dl
    mov ecx, eax
    shr ecx, 4
    and ecx, 0x3F
    mov dl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 1], dl
    and eax, 0xF
    shl eax, 2
    mov dl, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4

emit_nl:
    mov byte ptr [rdi], 0x0A
    inc rdi
    test r9, r9
    jnz line_loop

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz write_done
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop

write_done:
    ret
