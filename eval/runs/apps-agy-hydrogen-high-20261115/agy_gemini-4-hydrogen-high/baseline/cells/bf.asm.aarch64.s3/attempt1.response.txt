.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAX_BUF, 65536

.equ CH_BANG, 33
.equ CH_PLUS, 43
.equ CH_COMMA, 44
.equ CH_MINUS, 45
.equ CH_DOT, 46
.equ CH_LT, 60
.equ CH_GT, 62
.equ CH_LBRACK, 91
.equ CH_RBRACK, 93

.text
.globl _start
.p2align 2

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #MAX_BUF

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    mov x22, #0
    adrp x23, jumps
    add x23, x23, :lo12:jumps
    adrp x24, bstack
    add x24, x24, :lo12:bstack
    mov x25, #0

scan_loop:
    cmp x21, x20
    b.hs scan_no_bang
    ldrb w0, [x19, x21]
    cmp w0, #CH_BANG
    b.eq scan_found_bang
    cmp w0, #CH_GT
    b.eq is_gt
    cmp w0, #CH_LT
    b.eq is_lt
    cmp w0, #CH_PLUS
    b.eq is_plus
    cmp w0, #CH_MINUS
    b.eq is_minus
    cmp w0, #CH_LBRACK
    b.eq is_open
    cmp w0, #CH_RBRACK
    b.eq is_close
    cmp w0, #CH_DOT
    b.eq is_dot
    cmp w0, #CH_COMMA
    b.eq is_comma
    add x21, x21, #1
    b scan_loop

is_gt:
    mov w3, #0
    b store_cmd
is_lt:
    mov w3, #1
    b store_cmd
is_plus:
    mov w3, #2
    b store_cmd
is_minus:
    mov w3, #3
    b store_cmd
is_dot:
    mov w3, #6
    b store_cmd
is_comma:
    mov w3, #7
store_cmd:
    strb w3, [x19, x22]
    add x22, x22, #1
    add x21, x21, #1
    b scan_loop

is_open:
    str w22, [x24, x25, lsl #2]
    add x25, x25, #1
    mov w3, #4
    b store_cmd

is_close:
    cbz x25, bad_brackets
    sub x25, x25, #1
    ldr w1, [x24, x25, lsl #2]
    add w2, w22, #1
    str w2, [x23, x1, lsl #2]
    add w3, w1, #1
    str w3, [x23, x22, lsl #2]
    mov w3, #5
    b store_cmd

scan_found_bang:
    add x26, x21, #1
    b scan_done

scan_no_bang:
    mov x26, x20

scan_done:
    cbnz x25, bad_brackets

    add x20, x19, x20
    add x26, x19, x26
    mov x21, #0
    adrp x24, tape
    add x24, x24, :lo12:tape
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x27, #0
    adr x28, handlers_base

exec_loop:
    cmp x21, x22
    b.hs exec_done
    ldrb w0, [x19, x21]
    add x1, x28, x0, lsl #5
    br x1

.p2align 5
handlers_base:
    add x24, x24, #1
    add x21, x21, #1
    b exec_loop

.p2align 5
    sub x24, x24, #1
    add x21, x21, #1
    b exec_loop

.p2align 5
    ldrb w1, [x24]
    add w1, w1, #1
    strb w1, [x24]
    add x21, x21, #1
    b exec_loop

.p2align 5
    ldrb w1, [x24]
    sub w1, w1, #1
    strb w1, [x24]
    add x21, x21, #1
    b exec_loop

.p2align 5
    ldrb w1, [x24]
    cbz w1, 1f
    add x21, x21, #1
    b exec_loop
1:
    ldr w21, [x23, x21, lsl #2]
    b exec_loop

.p2align 5
    ldrb w1, [x24]
    cbnz w1, 1f
    add x21, x21, #1
    b exec_loop
1:
    ldr w21, [x23, x21, lsl #2]
    b exec_loop

.p2align 5
    ldrb w1, [x24]
    strb w1, [x25, x27]
    add x27, x27, #1
    add x21, x21, #1
    cmp x27, #MAX_BUF
    b.eq flush_and_loop
    b exec_loop

.p2align 5
    cmp x26, x20
    mov w1, #0
    b.hs 1f
    ldrb w1, [x26], #1
1:
    strb w1, [x24]
    add x21, x21, #1
    b exec_loop

flush_and_loop:
    bl flush_out
    b exec_loop

exec_done:
    bl flush_out
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

bad_brackets:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

flush_out:
    mov x9, #0
1:
    cmp x9, x27
    b.hs 2f
    mov x0, #1
    add x1, x25, x9
    sub x2, x27, x9
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le 2f
    add x9, x9, x0
    b 1b
2:
    mov x27, #0
    ret

.bss
.p2align 4
inbuf:
    .zero 65536
tape:
    .zero 30000
.p2align 2
outbuf:
    .zero 65536
.p2align 2
jumps:
    .zero 262144
bstack:
    .zero 262144
