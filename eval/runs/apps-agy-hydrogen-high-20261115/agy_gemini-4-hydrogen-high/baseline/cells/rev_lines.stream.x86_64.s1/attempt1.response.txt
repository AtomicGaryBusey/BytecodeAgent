.text
.globl _start

_start:
    lea rbx, [rip + in_buf]
    xor r12d, r12d
    mov r14d, 65600

read_loop:
    mov rdx, r14
    sub rdx, r12
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov r8, rbx
    add r12, rbx
    lea r13, [rip + out_buf]

1:
    cmp r8, r12
    jae done_lines
    mov r9, r8

2:
    cmp r9, r12
    jae 3f
    cmp byte ptr [r9], 10
    je 3f
    inc r9
    jmp 2b

3:
    mov r10, r9

4:
    cmp r10, r8
    je 5f
    dec r10
    mov al, byte ptr [r10]
    mov byte ptr [r13], al
    inc r13
    jmp 4b

5:
    mov byte ptr [r13], 10
    inc r13
    lea r8, [r9 + 1]
    jmp 1b

done_lines:
    lea rsi, [rip + out_buf]
    mov rdx, r13
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jle exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.bss
    .align 16
in_buf:
    .zero 65600
    .align 16
out_buf:
    .zero 65600
