.text
.globl _start

_start:
    lea r12, [rip + in_buf]
    xor ebx, ebx

read_loop:
    mov edx, 65540
    sub rdx, rbx
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_ok

    cmp byte ptr [r12 + rbx - 1], 10
    je 1f
    mov byte ptr [r12 + rbx], 10
    inc rbx

1:
    mov rdi, r12
    lea r13, [r12 + rbx]

line_loop:
    cmp rdi, r13
    jae write_all

    mov rsi, rdi
2:
    cmp byte ptr [rsi], 10
    je 3f
    inc rsi
    jmp 2b

3:
    lea r14, [rsi + 1]
    jmp 5f

4:
    mov al, byte ptr [rdi]
    mov cl, byte ptr [rsi]
    mov byte ptr [rdi], cl
    mov byte ptr [rsi], al
    inc rdi

5:
    dec rsi
    cmp rdi, rsi
    jb 4b

    mov rdi, r14
    jmp line_loop

write_all:
    xor r13d, r13d

write_loop:
    cmp r13, rbx
    jae exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + r13]
    mov rdx, rbx
    sub rdx, r13
    syscall
    test rax, rax
    jle exit_ok
    add r13, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
in_buf:
    .space 65544
