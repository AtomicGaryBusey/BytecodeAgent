.text
.globl _start

_start:
    xor r12d, r12d          # lines = 0
    xor r13d, r13d          # words = 0
    xor r14d, r14d          # bytes = 0
    xor r15d, r15d          # in_word = 0

read_loop:
    xor eax, eax            # SYS_read = 0
    xor edi, edi            # fd = 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax            # bytes += n
    mov rcx, rax

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a            # '\n'
    je is_newline
    cmp al, 0x20            # ' '
    je is_space
    sub al, 0x09            # 0x09..0x0d -> 0..4
    cmp al, 4
    jbe is_space

    test r15d, r15d
    jnz next_byte
    inc r13                 # words++
    mov r15d, 1             # in_word = 1
    jmp next_byte

is_newline:
    inc r12                 # lines++
is_space:
    xor r15d, r15d          # in_word = 0

next_byte:
    dec rcx
    jnz byte_loop
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf_end]
    dec rdi
    mov byte ptr [rdi], 0x0a

    mov rax, r14            # bytes
    call fmt_uint

    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r13            # words
    call fmt_uint

    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r12            # lines
    call fmt_uint

    lea rdx, [rip + outbuf_end]
    sub rdx, rdi
    mov rsi, rdi

write_loop:
    mov eax, 1              # SYS_write = 1
    mov edi, 1              # fd = 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    ja write_loop

exit_prog:
    mov eax, 60             # SYS_exit = 60
    xor edi, edi            # status = 0
    syscall

# Format unsigned 64-bit integer in rax as decimal backwards from rdi.
# Returns updated start pointer in rdi. Clobbers rax, rcx, rdx.
fmt_uint:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.p2align 4
buf:
    .space 65536
outbuf:
    .space 64
outbuf_end:
