.equ BUF_SIZE, 131072
.equ STACK_SIZE, 131072

.text
.globl _start

_start:
    lea r12, [rip + input_buf]
    mov r13, r12
    lea rbx, [r12 + BUF_SIZE]

read_loop:
    mov rdx, rbx
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    mov rsi, r12
    lea r14, [rip + stack_buf]
    mov rbp, r14
    lea r15, [rip + char_table]

    call skip_ws

parse_value:
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je parse_string_val
    cmp al, '{'
    je start_object
    cmp al, '['
    je start_array
    cmp al, '-'
    je parse_number_minus
    cmp al, '0'
    je parse_number_zero
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    lea edx, [rax - '1']
    cmp edx, 9
    jb parse_number_digits
    jmp fail

parse_string_val:
    call parse_string
    jmp after_value

start_array:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], ']'
    je empty_container
    mov byte ptr [rbp], 0
    inc rbp
    jmp parse_value

start_object:
    inc rsi
    call skip_ws
    cmp byte ptr [rsi], '}'
    je empty_container
    mov byte ptr [rbp], 1
    inc rbp

object_member:
    cmp byte ptr [rsi], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], ':'
    jne fail
    inc rsi
    call skip_ws
    jmp parse_value

empty_container:
    inc rsi

after_value:
    call skip_ws
    cmp rbp, r14
    je after_doc_value
    cmp byte ptr [rbp - 1], 0
    jne after_object_value

after_array_value:
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je array_comma
    cmp al, ']'
    jne fail
    inc rsi
    dec rbp
    jmp after_value

array_comma:
    inc rsi
    call skip_ws
    jmp parse_value

after_object_value:
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je object_comma
    cmp al, '}'
    jne fail
    inc rsi
    dec rbp
    jmp after_value

object_comma:
    inc rsi
    call skip_ws
    jmp object_member

after_doc_value:
    cmp rsi, r13
    jne fail

success:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

parse_true:
    inc rsi
    cmp byte ptr [rsi], 'r'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'u'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'e'
    jne fail
    inc rsi
    jmp after_value

parse_false:
    inc rsi
    cmp byte ptr [rsi], 'a'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'l'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 's'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'e'
    jne fail
    inc rsi
    jmp after_value

parse_null:
    inc rsi
    cmp byte ptr [rsi], 'u'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'l'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'l'
    jne fail
    inc rsi
    jmp after_value

parse_number_minus:
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je parse_number_zero
    lea edx, [rax - '1']
    cmp edx, 9
    jb parse_number_digits
    jmp fail

parse_number_zero:
    inc rsi
    movzx eax, byte ptr [rsi]
    jmp num_check_frac

parse_number_digits:
    inc rsi
num_int_loop:
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 2
    jz num_check_frac
    inc rsi
    jmp num_int_loop

num_check_frac:
    cmp al, '.'
    jne num_check_exp
    inc rsi
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 2
    jz fail
    inc rsi
num_frac_loop:
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 2
    jz num_check_exp
    inc rsi
    jmp num_frac_loop

num_check_exp:
    or al, 0x20
    cmp al, 'e'
    jne after_value
    inc rsi
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je num_exp_sign
    cmp al, '-'
    jne num_exp_first_digit
num_exp_sign:
    inc rsi
    movzx eax, byte ptr [rsi]
num_exp_first_digit:
    test byte ptr [r15 + rax], 2
    jz fail
    inc rsi
num_exp_loop:
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 2
    jz after_value
    inc rsi
    jmp num_exp_loop

skip_ws:
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 1
    jz 1f
    inc rsi
    jmp skip_ws
1:
    ret

parse_string:
    inc rsi
str_loop:
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 8
    jz str_special
    inc rsi
    jmp str_loop
str_special:
    cmp al, 0x22
    je str_done
    cmp al, 0x5C
    jne fail
    inc rsi
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 16
    jz str_check_u
    inc rsi
    jmp str_loop
str_check_u:
    cmp al, 'u'
    jne fail
    inc rsi
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 4
    jz fail
    inc rsi
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 4
    jz fail
    inc rsi
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 4
    jz fail
    inc rsi
    movzx eax, byte ptr [rsi]
    test byte ptr [r15 + rax], 4
    jz fail
    inc rsi
    jmp str_loop
str_done:
    inc rsi
    ret

fail:
    mov rax, rsi
    sub rax, r12
    lea rcx, [rip + out_buf + 32]
    dec rcx
    mov byte ptr [rcx], 10
    mov r8d, 10
1:
    xor edx, edx
    div r8
    add dl, '0'
    dec rcx
    mov byte ptr [rcx], dl
    test rax, rax
    jnz 1b
    dec rcx
    mov byte ptr [rcx], 0x20
    sub rcx, 8
    mov r9, 0x746120726f727265
    mov qword ptr [rcx], r9
    lea rdx, [rip + out_buf + 32]
    sub rdx, rcx
    mov eax, 1
    mov edi, 1
    mov rsi, rcx
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata

    .byte 9, 8, 16, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 24
    .byte 14, 14, 14, 14, 14, 14, 14, 14, 14, 14, 8, 8, 8, 8, 8, 8
    .byte 8, 12, 12, 12, 12, 12, 12, 8, 8, 8, 8, 8, 8, 8, 8, 8
    .byte 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 16, 8, 8, 8
    .byte 8, 12, 28, 12, 12, 12, 28, 8, 8, 8, 8, 8, 8, 8, 24, 8
    .byte 8, 8, 24, 8, 24, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8

    .byte 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8
    .byte 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8
    .byte 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8
    .byte 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8, 8

.bss
    .align 16

    .zero BUF_SIZE + 64

    .zero STACK_SIZE

    .zero 32
