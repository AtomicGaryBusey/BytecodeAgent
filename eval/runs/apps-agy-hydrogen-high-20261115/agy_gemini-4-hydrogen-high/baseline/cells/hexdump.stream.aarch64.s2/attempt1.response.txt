.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INBUF_SIZE, 4096
.equ OUTBUF_SIZE, 4096
.equ OUTBUF_FLUSH_LIMIT, 3968

.globl _start

.text
_start:
    mov x19, #0                 // offset = 0
    mov x20, #0                 // line_len = 0

    adrp x21, linebuf
    add x21, x21, :lo12:linebuf
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23                // out_ptr = outbuf
    add x25, x23, #OUTBUF_FLUSH_LIMIT

read_loop:
    mov x0, #0                  // fd = stdin
    mov x1, x22                 // buf = inbuf
    mov x2, #INBUF_SIZE
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le eof

    mov x26, x0                 // remaining bytes in inbuf
    mov x27, x22                // in_ptr

process_loop:
    ldrb w0, [x27], #1
    strb w0, [x21, x20]
    add x20, x20, #1
    cmp x20, #16
    b.ne next_byte

    cmp x24, x25
    b.lo 1f
    mov x0, x23
    sub x1, x24, x23
    bl write_all
    mov x24, x23
1:
    mov x0, x21
    mov w1, #16
    mov x2, x19
    mov x3, x24
    bl format_row
    mov x24, x0
    add x19, x19, #16
    mov x20, #0

next_byte:
    subs x26, x26, #1
    b.ne process_loop
    b read_loop

eof:
    cbz x20, emit_final_offset

    cmp x24, x25
    b.lo 1f
    mov x0, x23
    sub x1, x24, x23
    bl write_all
    mov x24, x23
1:
    mov x0, x21
    mov w1, w20
    mov x2, x19
    mov x3, x24
    bl format_row
    mov x24, x0
    add x19, x19, x20

emit_final_offset:
    cmp x24, x25
    b.lo 1f
    mov x0, x23
    sub x1, x24, x23
    bl write_all
    mov x24, x23
1:
    adrp x4, hex_digits
    add x4, x4, :lo12:hex_digits
    mov w5, #28
2:
    lsr w6, w19, w5
    and w6, w6, #0xf
    ldrb w7, [x4, x6]
    strb w7, [x24], #1
    subs w5, w5, #4
    b.ge 2b

    mov w6, #0x0a               // '\n'
    strb w6, [x24], #1

    mov x0, x23
    sub x1, x24, x23
    bl write_all

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// format_row(x0 = linebuf, w1 = len (1..16), x2 = offset, x3 = out_ptr) -> x0 = new out_ptr
format_row:
    adrp x4, hex_digits
    add x4, x4, :lo12:hex_digits

    mov w5, #28
1:
    lsr w6, w2, w5
    and w6, w6, #0xf
    ldrb w7, [x4, x6]
    strb w7, [x3], #1
    subs w5, w5, #4
    b.ge 1b

    mov w8, #0x20               // ' '
    strb w8, [x3], #1
    strb w8, [x3], #1

    mov w5, #0
2:
    cmp w5, w1
    b.hs 3f
    ldrb w6, [x0, x5]
    lsr w7, w6, #4
    ldrb w7, [x4, x7]
    strb w7, [x3], #1
    and w7, w6, #0xf
    ldrb w7, [x4, x7]
    strb w7, [x3], #1
    strb w8, [x3], #1
    b 4f
3:
    strb w8, [x3], #1
    strb w8, [x3], #1
    strb w8, [x3], #1
4:
    cmp w5, #7
    b.ne 5f
    strb w8, [x3], #1
5:
    add w5, w5, #1
    cmp w5, #16
    b.lo 2b

    strb w8, [x3], #1
    mov w6, #0x7c               // '|'
    strb w6, [x3], #1

    mov w5, #0
    mov w9, #0x2e               // '.'
6:
    ldrb w6, [x0, x5]
    sub w7, w6, #0x20
    cmp w7, #0x5e
    csel w6, w6, w9, ls
    strb w6, [x3], #1
    add w5, w5, #1
    cmp w5, w1
    b.lo 6b

    mov w6, #0x7c               // '|'
    strb w6, [x3], #1
    mov w6, #0x0a               // '\n'
    strb w6, [x3], #1

    mov x0, x3
    ret

// write_all(x0 = buf, x1 = count)
write_all:
    mov x3, x0
    mov x4, x1
1:
    cbz x4, 2f
    mov x0, #1                  // fd = stdout
    mov x1, x3
    mov x2, x4
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le 2f
    add x3, x3, x0
    sub x4, x4, x0
    b 1b
2:
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
linebuf:
    .space 16
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
