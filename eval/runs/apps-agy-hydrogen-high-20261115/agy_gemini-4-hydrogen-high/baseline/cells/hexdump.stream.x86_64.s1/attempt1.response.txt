.equ INBUF_SIZE, 4096
.equ OUTBUF_SIZE, 8192
.equ OUTBUF_THRESH, 7680

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .space INBUF_SIZE
    .balign 16
outbuf:
    .space OUTBUF_SIZE

.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    lea r14, [rip + outbuf]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov rdx, INBUF_SIZE
    sub rdx, r13
    syscall
    test rax, rax
    jle eof_reached
    add r13, rax

    lea r15, [rip + inbuf]
row_loop:
    cmp r13, 16
    jb row_loop_done

    mov rsi, r15
    mov rdi, r14
    mov eax, r12d
    mov ecx, 16
    call format_row
    mov r14, rdi

    add r15, 16
    sub r13, 16
    add r12, 16

    lea rax, [rip + outbuf]
    mov rdx, r14
    sub rdx, rax
    cmp rdx, OUTBUF_THRESH
    jb row_loop
    call flush_out
    jmp row_loop

row_loop_done:
    test r13, r13
    jz read_loop
    lea rdi, [rip + inbuf]
    cmp r15, rdi
    je read_loop
    xor ecx, ecx
copy_rem:
    mov al, byte ptr [r15 + rcx]
    mov byte ptr [rdi + rcx], al
    inc rcx
    cmp rcx, r13
    jb copy_rem
    jmp read_loop

eof_reached:
    test r13, r13
    jz write_final_offset
    lea rsi, [rip + inbuf]
    mov rdi, r14
    mov eax, r12d
    mov rcx, r13
    call format_row
    mov r14, rdi
    add r12, r13

write_final_offset:
    mov rdi, r14
    mov eax, r12d
    call write_hex32
    mov byte ptr [rdi], 10
    inc rdi
    mov r14, rdi
    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

# Write 32-bit value in eax as 8 lowercase hex digits to [rdi], advance rdi by 8.
# Sets r9 = hex_chars. Clobbers eax, edx, r8, r9.
write_hex32:
    mov r8d, 8
    lea r9, [rip + hex_chars]
1:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec r8d
    jnz 1b
    ret

# Format a row of rcx bytes (1 <= rcx <= 16) from rsi at offset eax into rdi.
# Advances rdi past the trailing newline.
format_row:
    call write_hex32
    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r8d, r8d
1:
    cmp r8, rcx
    jae 2f
    movzx eax, byte ptr [rsi + r8]
    mov edx, eax
    shr eax, 4
    and edx, 0xf
    mov al, byte ptr [r9 + rax]
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], al
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 0x20
    jmp 3f
2:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
3:
    add rdi, 3
    cmp r8d, 7
    jne 4f
    mov byte ptr [rdi], 0x20
    inc rdi
4:
    inc r8d
    cmp r8d, 16
    jb 1b

    mov word ptr [rdi], 0x7c20
    add rdi, 2

    xor r8d, r8d
    mov r10d, 0x2e
5:
    movzx eax, byte ptr [rsi + r8]
    lea edx, [rax - 0x20]
    cmp dl, 0x5e
    cmova eax, r10d
    mov byte ptr [rdi], al
    inc rdi
    inc r8d
    cmp r8, rcx
    jb 5b

    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    ret

# Write outbuf..r14 to stdout and reset r14 to outbuf.
flush_out:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    jz 2f
    mov edi, 1
1:
    mov eax, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    lea r14, [rip + outbuf]
    ret
