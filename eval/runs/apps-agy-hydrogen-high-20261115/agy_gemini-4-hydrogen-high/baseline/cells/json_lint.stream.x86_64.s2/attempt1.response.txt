.globl _start

.text
_start:
    lea r12, [rip + buf]
    xor ebx, ebx

read_loop:
    mov edx, 131072
    sub edx, ebx
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov rsi, r12
    lea r13, [r12 + rbx]
    lea r14, [rip + stack]
    mov rbp, r14

expect_value:
    call skip_ws
    cmp rsi, r13
    je error
    movzx eax, byte ptr [rsi]
    cmp al, 0x7B
    je do_object
    cmp al, 0x5B
    je do_array
    cmp al, 0x22
    je do_string
    cmp al, 0x74
    je do_true
    cmp al, 0x66
    je do_false
    cmp al, 0x6E
    je do_null
    cmp al, 0x2D
    je parse_number
    sub al, 0x30
    cmp al, 9
    jbe parse_number
    jmp error

do_object:
    inc rsi
    call skip_ws
    cmp rsi, r13
    je error
    cmp byte ptr [rsi], 0x7D
    je close_empty
    mov byte ptr [rbp], 1
    inc rbp
    jmp parse_member_prefix

do_array:
    inc rsi
    call skip_ws
    cmp rsi, r13
    je error
    cmp byte ptr [rsi], 0x5D
    je close_empty
    mov byte ptr [rbp], 0
    inc rbp
    jmp expect_value

close_empty:
    inc rsi
    jmp after_value

do_string:
    call parse_string
    jmp after_value

do_true:
    inc rsi
    lea rdi, [rip + str_rue]
    mov ecx, 3
    call match_bytes
    jmp after_value

do_false:
    inc rsi
    lea rdi, [rip + str_alse]
    mov ecx, 4
    call match_bytes
    jmp after_value

do_null:
    inc rsi
    lea rdi, [rip + str_ull]
    mov ecx, 3
    call match_bytes
    jmp after_value

parse_member_prefix:
    call parse_string
    call skip_ws
    cmp rsi, r13
    je error
    cmp byte ptr [rsi], 0x3A
    jne error
    inc rsi
    jmp expect_value

after_value:
    call skip_ws
    cmp rbp, r14
    je check_eof
    cmp rsi, r13
    je error
    cmp byte ptr [rbp - 1], 1
    je after_obj_val
    cmp byte ptr [rsi], 0x5D
    je close_container
    cmp byte ptr [rsi], 0x2C
    jne error
    inc rsi
    jmp expect_value

after_obj_val:
    cmp byte ptr [rsi], 0x7D
    je close_container
    cmp byte ptr [rsi], 0x2C
    jne error
    inc rsi
    call skip_ws
    jmp parse_member_prefix

close_container:
    inc rsi
    dec rbp
    jmp after_value

check_eof:
    cmp rsi, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + str_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    mov rdx, 0x100002600
skip_ws_loop:
    cmp rsi, r13
    jae skip_ws_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    ja skip_ws_done
    bt rdx, rax
    jnc skip_ws_done
    inc rsi
    jmp skip_ws_loop
skip_ws_done:
    ret

match_bytes:
    cmp rsi, r13
    je error
    mov al, byte ptr [rsi]
    cmp al, byte ptr [rdi]
    jne error
    inc rsi
    inc rdi
    dec ecx
    jnz match_bytes
    ret

parse_string:
    cmp rsi, r13
    je error
    cmp byte ptr [rsi], 0x22
    jne error
    inc rsi
str_loop:
    cmp rsi, r13
    je error
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    jb error
    cmp al, 0x22
    je str_done
    cmp al, 0x5C
    je str_escape
    inc rsi
    jmp str_loop
str_done:
    inc rsi
    ret
str_escape:
    inc rsi
    cmp rsi, r13
    je error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, 0x2F
    je str_esc_ok
    cmp al, 0x62
    je str_esc_ok
    cmp al, 0x66
    je str_esc_ok
    cmp al, 0x6E
    je str_esc_ok
    cmp al, 0x72
    je str_esc_ok
    cmp al, 0x74
    je str_esc_ok
    cmp al, 0x75
    je str_esc_u
    jmp error
str_esc_ok:
    inc rsi
    jmp str_loop
str_esc_u:
    inc rsi
    mov ecx, 4
str_hex_loop:
    cmp rsi, r13
    je error
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    jbe str_hex_ok
    movzx eax, byte ptr [rsi]
    or al, 0x20
    sub al, 0x61
    cmp al, 5
    ja error
str_hex_ok:
    inc rsi
    dec ecx
    jnz str_hex_loop
    jmp str_loop

parse_number:
    cmp byte ptr [rsi], 0x2D
    jne num_int
    inc rsi
    cmp rsi, r13
    je error
num_int:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    je num_zero
    sub al, 0x31
    cmp al, 8
    ja error
    inc rsi
num_digits_loop:
    cmp rsi, r13
    je after_value
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja num_check_frac
    inc rsi
    jmp num_digits_loop
num_zero:
    inc rsi
num_check_frac:
    cmp rsi, r13
    je after_value
    mov al, byte ptr [rsi]
    cmp al, 0x2E
    jne num_check_exp
    inc rsi
    cmp rsi, r13
    je error
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja error
    inc rsi
num_frac_loop:
    cmp rsi, r13
    je after_value
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja num_check_exp
    inc rsi
    jmp num_frac_loop
num_check_exp:
    cmp rsi, r13
    je after_value
    mov al, byte ptr [rsi]
    or al, 0x20
    cmp al, 0x65
    jne after_value
    inc rsi
    cmp rsi, r13
    je error
    mov al, byte ptr [rsi]
    cmp al, 0x2B
    je num_exp_sign
    cmp al, 0x2D
    jne num_exp_req_digit
num_exp_sign:
    inc rsi
    cmp rsi, r13
    je error
num_exp_req_digit:
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja error
    inc rsi
num_exp_loop:
    cmp rsi, r13
    je after_value
    movzx eax, byte ptr [rsi]
    sub al, 0x30
    cmp al, 9
    ja after_value
    inc rsi
    jmp num_exp_loop

error:
    mov rax, rsi
    sub rax, r12
    and rsp, -16
    sub rsp, 64
    lea rdi, [rsp + 63]
    mov byte ptr [rdi], 0x0A
    mov ecx, 10
err_div_loop:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz err_div_loop
    sub rdi, 9
    mov rax, 0x746120726f727265
    mov qword ptr [rdi], rax
    mov byte ptr [rdi + 8], 0x20
    lea rdx, [rsp + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
str_ok:
    .ascii "ok\n"
str_rue:
    .ascii "rue"
str_alse:
    .ascii "alse"
str_ull:
    .ascii "ull"

.bss
    .p2align 4
buf:
    .space 131072
stack:
    .space 65536
