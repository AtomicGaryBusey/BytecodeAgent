.text
.globl _start

_start:
    mov x19, #0                 // lines = 0
    mov x20, #0                 // words = 0
    mov x21, #0                 // bytes = 0
    mov w22, #0                 // in_word = 0
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf
    mov x24, #8192              // read chunk size

read_loop:
    mov x0, #0                  // fd = 0 (stdin)
    mov x1, x23                 // buf
    mov x2, x24                 // count
    mov x8, #63                 // sys_read
    svc #0
    cmp x0, #0
    b.le read_done              // EOF or error

    add x21, x21, x0            // bytes += nread
    mov x1, x23                 // ptr = buf
    add x2, x23, x0             // end = buf + nread

byte_loop:
    ldrb w3, [x1], #1           // w3 = *ptr++
    cmp w3, #0x0a               // '\n'
    cinc x19, x19, eq           // if '\n', lines++

    cmp w3, #0x20               // ' '
    b.eq 1f
    sub w4, w3, #0x09           // check 0x09..0x0d (\t, \n, \v, \f, \r)
    cmp w4, #4
    b.ls 1f

    cbnz w22, 2f                // non-whitespace: if already in word, continue
    add x20, x20, #1            // words++
    mov w22, #1                 // in_word = 1
    b 2f

1:
    mov w22, #0                 // whitespace: in_word = 0

2:
    cmp x1, x2
    b.lo byte_loop
    b read_loop

read_done:
    adrp x25, outbuf_end
    add x25, x25, :lo12:outbuf_end
    mov x26, x25

    mov w2, #0x0a               // trailing '\n'
    strb w2, [x26, #-1]!

    mov x0, x21                 // bytes
    bl u64_to_dec

    mov w2, #0x20               // ' '
    strb w2, [x26, #-1]!

    mov x0, x20                 // words
    bl u64_to_dec

    mov w2, #0x20               // ' '
    strb w2, [x26, #-1]!

    mov x0, x19                 // lines
    bl u64_to_dec

write_loop:
    mov x0, #1                  // fd = 1 (stdout)
    mov x1, x26                 // buf
    sub x2, x25, x26            // remaining count
    cbz x2, exit_ok
    mov x8, #64                 // sys_write
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x26, x26, x0
    b write_loop

exit_ok:
    mov x0, #0                  // status = 0
    mov x8, #93                 // sys_exit
    svc #0

// Convert x0 (unsigned 64-bit) to decimal ASCII backwards from x26
u64_to_dec:
    mov x3, #10
1:
    udiv x1, x0, x3
    msub x2, x1, x3, x0         // x2 = x0 % 10
    add w2, w2, #0x30           // '0' + digit
    strb w2, [x26, #-1]!
    mov x0, x1
    cbnz x0, 1b
    ret

.bss
    .balign 16
inbuf:
    .zero 8192
outbuf:
    .zero 128
outbuf_end:
