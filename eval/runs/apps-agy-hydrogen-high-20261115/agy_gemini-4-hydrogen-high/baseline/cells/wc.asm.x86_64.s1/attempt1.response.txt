    xor edi, edi
    lea rsi, [rip + in_buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle read_done

    add r14, rax
    lea rbx, [rip + in_buf]
    lea rbp, [rbx + rax]

1:

    xor ecx, ecx
    cmp al, 10
    sete cl
    add r12, rcx
    movzx edx, byte ptr [r8 + rax]
    mov ecx, r15d
    xor ecx, 1
    and ecx, edx
    add r13, rcx
    mov r15d, edx
    inc rbx
    cmp rbx, rbp

    jmp read_loop

read_done:

    add r8, 128
    mov rdi, r8
    mov r9d, 10

    call write_u64

    dec rdi

    mov rax, r12
    call write_u64

    mov rbx, rdi
    mov rbp, r8
    sub rbp, rdi

write_loop:

    jle write_done
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, rbp
    syscall
    test rax, rax
    jle write_done
    add rbx, rax
    sub rbp, rax
    jmp write_loop

write_done:
    mov eax, 60
    xor edi, edi
    syscall
