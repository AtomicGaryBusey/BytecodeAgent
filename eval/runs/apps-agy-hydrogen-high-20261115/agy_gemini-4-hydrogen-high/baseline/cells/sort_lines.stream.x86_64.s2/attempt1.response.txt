.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 131072
.equ LINES_SIZE, 262144

.bss
.p2align 4
in_buf:
    .zero IN_BUF_SIZE
.p2align 4
out_buf:
    .zero OUT_BUF_SIZE
.p2align 3
lines:
    .zero LINES_SIZE

.text
.globl _start
_start:
    lea r15, [rip + in_buf]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    mov edx, IN_BUF_SIZE
    sub rdx, r12
    jbe read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_ok

    lea rbp, [rip + lines]
    xor r14d, r14d
    xor ebx, ebx
    xor ecx, ecx

parse_loop:
    cmp byte ptr [r15 + rcx], 10
    jne 1f
    mov rax, rcx
    sub rax, rbx
    shl rax, 32
    or rax, rbx
    mov qword ptr [rbp + r14*8], rax
    inc r14
    lea rbx, [rcx + 1]
1:
    inc rcx
    cmp rcx, r12
    jb parse_loop

    cmp rbx, r12
    jae parse_done
    mov rax, r12
    sub rax, rbx
    shl rax, 32
    or rax, rbx
    mov qword ptr [rbp + r14*8], rax
    inc r14

parse_done:
    test r14, r14
    jz exit_ok
    cmp r14, 1
    jbe sort_done

    mov rbx, r14
    shr rbx, 1
    dec rbx

heapify_loop:
    mov rdi, rbx
    mov rsi, r14
    call sift_down
    dec rbx
    jns heapify_loop

    lea rbx, [r14 - 1]

extract_loop:
    mov rax, qword ptr [rbp]
    mov rdx, qword ptr [rbp + rbx*8]
    mov qword ptr [rbp], rdx
    mov qword ptr [rbp + rbx*8], rax
    xor edi, edi
    mov rsi, rbx
    call sift_down
    dec rbx
    jnz extract_loop

sort_done:
    lea rdi, [rip + out_buf]
    xor ebx, ebx

build_out_loop:
    mov rax, qword ptr [rbp + rbx*8]
    mov esi, eax
    add rsi, r15
    mov rcx, rax
    shr rcx, 32
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    cmp rbx, r14
    jb build_out_loop

    lea rbx, [rip + out_buf]
    sub rdi, rbx
    mov r12, rdi

write_loop:
    test r12, r12
    jz exit_ok
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall
    test rax, rax
    jle exit_ok
    add rbx, rax
    sub r12, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

sift_down:
    push rbx
    push r12
    push r13
    push r14
    mov rbx, rdi
    mov r12, rsi
    mov r13, qword ptr [rbp + rbx*8]
1:
    lea r14, [rbx + rbx + 1]
    cmp r14, r12
    jae 3f
    lea rcx, [r14 + 1]
    cmp rcx, r12
    jae 2f
    mov rax, qword ptr [rbp + r14*8]
    mov rdx, qword ptr [rbp + rcx*8]
    call cmp_lines
    test eax, eax
    jge 2f
    inc r14
2:
    mov rax, r13
    mov rdx, qword ptr [rbp + r14*8]
    call cmp_lines
    test eax, eax
    jge 3f
    mov rax, qword ptr [rbp + r14*8]
    mov qword ptr [rbp + rbx*8], rax
    mov rbx, r14
    jmp 1b
3:
    mov qword ptr [rbp + rbx*8], r13
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

cmp_lines:
    mov r8d, eax
    mov r9, rax
    shr r9, 32
    mov r10d, edx
    mov r11, rdx
    shr r11, 32
    mov rcx, r9
    cmp rcx, r11
    cmova rcx, r11
    test rcx, rcx
    jz 1f
    lea rsi, [r15 + r8]
    lea rdi, [r15 + r10]
    repe cmpsb
    je 1f
    jb 2f
    mov eax, 1
    ret
2:
    mov eax, -1
    ret
1:
    cmp r9, r11
    jb 2b
    ja 3f
    xor eax, eax
    ret
3:
    mov eax, 1
    ret
