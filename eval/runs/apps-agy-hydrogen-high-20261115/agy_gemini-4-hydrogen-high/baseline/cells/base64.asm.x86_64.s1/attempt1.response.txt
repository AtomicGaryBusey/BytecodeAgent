.text
.globl _start

_start:
    lea r12, [rip + in_buf]
    xor r13d, r13d

read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    mov rsi, r12
    mov r12, r13
    lea rdi, [rip + out_buf]
    lea rbx, [rip + b64_table]

line_loop:
    cmp r12, 57
    jb rem_loop
    mov r8d, 19

group57_loop:
    call encode3
    add rsi, 3
    add rdi, 4
    dec r8d
    jnz group57_loop

    mov byte ptr [rdi], 10
    inc rdi
    sub r12, 57
    jmp line_loop

rem_loop:
    test r12, r12
    jz done_encode

rem3_loop:
    cmp r12, 3
    jb rem_tail
    call encode3
    add rsi, 3
    add rdi, 4
    sub r12, 3
    jmp rem3_loop

rem_tail:
    test r12, r12
    jz finish_line
    cmp r12, 1
    je tail_1

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx

    mov ecx, eax
    shr ecx, 18
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    shr ecx, 12
    and ecx, 0x3F
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 1], cl

    shr eax, 6
    and eax, 0x3F
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 2], al

    mov byte ptr [rdi + 3], '='
    add rdi, 4
    jmp finish_line

tail_1:
    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl

    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al

    mov byte ptr [rdi + 2], '='
    mov byte ptr [rdi + 3], '='
    add rdi, 4

finish_line:
    mov byte ptr [rdi], 10
    inc rdi

done_encode:
    lea r12, [rip + out_buf]
    mov r13, rdi
    sub r13, r12
    xor r14d, r14d

write_loop:
    cmp r14, r13
    jae exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + r14]
    mov rdx, r13
    sub rdx, r14
    syscall
    test rax, rax
    jle exit_ok
    add r14, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

encode3:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    movzx ecx, byte ptr [rsi + 1]
    or eax, ecx
    shl eax, 8
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx

    mov ecx, eax
    shr ecx, 18
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    shr ecx, 12
    and ecx, 0x3F
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 1], cl

    mov ecx, eax
    shr ecx, 6
    and ecx, 0x3F
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 2], cl

    and eax, 0x3F
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], al
    ret

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
in_buf:
    .space 65536
.balign 16
out_buf:
    .space 131072
