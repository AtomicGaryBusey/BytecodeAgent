.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_BUF_SIZE, 4096
.equ OUT_BUF_SIZE, 4096
.equ OUT_FLUSH_THRESHOLD, 4000

.text
.globl _start

_start:
    sub sp, sp, #16

    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, x19
    mov x21, x19

    adrp x22, out_buf
    add x22, x22, :lo12:out_buf
    mov x23, x22
    add x24, x22, #OUT_FLUSH_THRESHOLD

    mov x25, #0

    adrp x26, hex_digits
    add x26, x26, :lo12:hex_digits

    movz x27, #0x2020
    movk x27, #0x2020, lsl #16
    movk x27, #0x2020, lsl #32
    movk x27, #0x2020, lsl #48

main_loop:
    mov x28, #0

read_row_loop:
    cmp x20, x21
    b.lo 1f
    mov x0, #0
    mov x1, x19
    mov x2, #IN_BUF_SIZE
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_row_done
    mov x20, x19
    add x21, x19, x0
1:
    ldrb w0, [x20], #1
    strb w0, [sp, x28]
    add x28, x28, #1
    cmp x28, #16
    b.lo read_row_loop

read_row_done:
    cbz x28, done_all

    cmp x23, x24
    b.lo 2f
    bl flush_out
2:
    mov w0, w25
    mov x1, x23
    bl format_hex32

    str x27, [x23, #8]
    str x27, [x23, #16]
    str x27, [x23, #24]
    str x27, [x23, #32]
    str x27, [x23, #40]
    str x27, [x23, #48]
    str x27, [x23, #56]

    mov w0, #0x7c
    strb w0, [x23, #60]

    mov x2, #0
    add x3, x23, #10
    add x4, x23, #61

row_byte_loop:
    ldrb w5, [sp, x2]
    lsr w6, w5, #4
    and w7, w5, #0xf
    ldrb w6, [x26, x6]
    ldrb w7, [x26, x7]
    strb w6, [x3]
    strb w7, [x3, #1]
    add x3, x3, #3
    cmp x2, #7
    cinc x3, x3, eq

    cmp w5, #0x20
    b.lo 3f
    cmp w5, #0x7e
    b.ls 4f
3:
    mov w5, #0x2e
4:
    strb w5, [x4], #1

    add x2, x2, #1
    cmp x2, x28
    b.lo row_byte_loop

    mov w0, #0x7c
    strb w0, [x4], #1
    mov w0, #0x0a
    strb w0, [x4], #1
    mov x23, x4

    add x25, x25, x28
    cmp x28, #16
    b.eq main_loop

done_all:
    cmp x23, x24
    b.lo 5f
    bl flush_out
5:
    mov w0, w25
    mov x1, x23
    bl format_hex32
    mov w0, #0x0a
    strb w0, [x23, #8]
    add x23, x23, #9
    bl flush_out

    add sp, sp, #16
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

// Write buffered output [x22..x23) to stdout and reset x23 = x22
flush_out:
    sub x2, x23, x22
    cbz x2, 1f
    mov x0, #1
    mov x1, x22
    mov x8, #SYS_write
    svc #0
    mov x23, x22
1:
    ret

// Format 32-bit w0 as 8 lowercase hex digits at [x1..x1+7]
format_hex32:
    mov x2, #7
1:
    and x3, x0, #0xf
    ldrb w4, [x26, x3]
    strb w4, [x1, x2]
    lsr w0, w0, #4
    subs x2, x2, #1
    b.pl 1b
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .balign 16
in_buf:
    .space IN_BUF_SIZE
    .balign 16
out_buf:
    .space OUT_BUF_SIZE
