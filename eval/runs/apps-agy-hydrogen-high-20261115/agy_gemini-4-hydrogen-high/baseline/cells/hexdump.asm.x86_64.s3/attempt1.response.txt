.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
in_buf:
    .space 4096
.balign 16
out_buf:
    .space 65536

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    lea r14, [rip + out_buf]
    lea r15, [rip + in_buf]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r13]
    mov edx, 4096
    sub edx, r13d
    syscall
    test rax, rax
    jle eof
    add r13, rax

    xor ebx, ebx
row_loop:
    lea rax, [rbx + 16]
    cmp rax, r13
    ja row_done

    lea rsi, [r15 + rbx]
    mov edx, 16
    call emit_row

    lea rax, [rip + out_buf]
    add rax, 65000
    cmp r14, rax
    jb 1f
    call flush_out
1:
    add rbx, 16
    jmp row_loop

row_done:
    sub r13, rbx
    test r13, r13
    jz read_loop
    test rbx, rbx
    jz read_loop

    xor ecx, ecx
2:
    mov al, byte ptr [r15 + rbx]
    mov byte ptr [r15 + rcx], al
    inc rbx
    inc rcx
    cmp rcx, r13
    jb 2b
    jmp read_loop

eof:
    test r13, r13
    jz final_offset
    mov rsi, r15
    mov rdx, r13
    call emit_row

final_offset:
    mov rdi, r14
    mov eax, r12d
    call format_hex32
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov r14, rdi
    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

# format_hex32: write eax as 8 lowercase hex digits at [rdi], advance rdi by 8
# Clobbers: rax, rcx, rdx, r8
format_hex32:
    lea r8, [rip + hex_digits]
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

# emit_row: format rdx (1..16) bytes at rsi using offset r12d into [r14]
# Updates r14 and advances r12 by rdx
# Clobbers: rax, rcx, rdx, rdi, r8, r9, r10, r11
emit_row:
    mov r9, rdx
    mov rdi, r14
    mov eax, r12d
    call format_hex32

    mov rax, 0x2020202020202020
    mov qword ptr [rdi], rax
    mov qword ptr [rdi + 8], rax
    mov qword ptr [rdi + 16], rax
    mov qword ptr [rdi + 24], rax
    mov qword ptr [rdi + 32], rax
    mov qword ptr [rdi + 40], rax
    mov qword ptr [rdi + 48], rax
    mov byte ptr [r14 + 60], 0x7c

    xor ecx, ecx
    lea r10, [r14 + 10]
    lea r11, [r14 + 61]
1:
    cmp ecx, 8
    jne 2f
    inc r10
2:
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r10], dl

    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [r10 + 1], dl

    add r10, 3

    lea edx, [eax - 0x20]
    cmp dl, 0x5f
    mov edx, 0x2e
    cmovb edx, eax
    mov byte ptr [r11 + rcx], dl

    inc rcx
    cmp rcx, r9
    jb 1b

    mov word ptr [r11 + rcx], 0x0a7c
    lea r14, [r11 + rcx + 2]
    add r12, r9
    ret

# flush_out: write [out_buf .. r14) to stdout and reset r14 to out_buf
# Clobbers: rax, rcx, rdx, rdi, rsi, r11
flush_out:
    lea rsi, [rip + out_buf]
    mov rdx, r14
    sub rdx, rsi
    jz 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    ja 1b
2:
    lea r14, [rip + out_buf]
    ret
