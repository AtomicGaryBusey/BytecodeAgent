.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 64

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + inbuf]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov edx, 65536
    syscall
    test rax, rax
    jle read_done

    add r14, rax
    mov rsi, rbx
    lea rdx, [rbx + rax]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0a
    jne 1f
    inc r12
    jmp 2f
1:
    cmp al, 0x20
    je 2f
    sub eax, 9
    cmp al, 4
    jbe 2f
    test r15d, r15d
    jnz 3f
    inc r13
    mov r15d, 1
    jmp 3f
2:
    xor r15d, r15d
3:
    cmp rsi, rdx
    jb byte_loop
    jmp read_loop

read_done:
    lea rdi, [rip + outbuf]
    add rdi, 64
    mov r8, rdi

    dec rdi
    mov byte ptr [rdi], 0x0a

    mov rax, r14
    call write_u64

    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r13
    call write_u64

    dec rdi
    mov byte ptr [rdi], 0x20

    mov rax, r12
    call write_u64

    mov r12, r8
    sub r12, rdi
    mov rsi, rdi

write_loop:
    mov eax, 1
    mov edi, 1
    mov rdx, r12
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub r12, rax
    ja write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

write_u64:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret
