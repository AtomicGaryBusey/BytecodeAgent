.equ INBUF_SIZE, 58368
.equ INBUF_ALLOC, 58372
.equ OUTBUF_SIZE, 78848

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
inbuf:
    .zero INBUF_ALLOC
    .align 16
outbuf:
    .zero OUTBUF_SIZE

.text
.globl _start

_start:
    xor r14d, r14d

read_chunk:
    xor r12d, r12d

read_more:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, INBUF_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle got_eof
    add r12, rax
    cmp r12, INBUF_SIZE
    jb read_more
    jmp process_chunk

got_eof:
    mov r14d, 1
    test r12, r12
    jz exit_ok

process_chunk:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + b64_table]
    mov r13d, 19

    cmp r12, 3
    jb tail_check

loop3:
    mov eax, dword ptr [rsi]
    bswap eax
    shr eax, 8

    mov edx, eax
    shr edx, 18
    mov cl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], cl

    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov cl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], cl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov cl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], cl

    and eax, 0x3F
    mov cl, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], cl

    add rsi, 3
    add rdi, 4
    dec r13d
    jnz 1f
    mov byte ptr [rdi], 10
    inc rdi
    mov r13d, 19
1:
    sub r12, 3
    cmp r12, 3
    jae loop3

tail_check:
    test r12, r12
    jz finish_line

    cmp r12, 1
    jne tail_two

tail_one:
    movzx eax, byte ptr [rsi]
    shl eax, 4

    mov edx, eax
    shr edx, 6
    mov cl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], cl

    and eax, 0x3F
    mov cl, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], cl

    mov byte ptr [rdi + 2], 0x3D
    mov byte ptr [rdi + 3], 0x3D
    jmp tail_done

tail_two:
    movzx eax, byte ptr [rsi]
    shl eax, 10
    movzx edx, byte ptr [rsi + 1]
    shl edx, 2
    or eax, edx

    mov edx, eax
    shr edx, 12
    mov cl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], cl

    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov cl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], cl

    and eax, 0x3F
    mov cl, byte ptr [rbx + rax]
    mov byte ptr [rdi + 2], cl

    mov byte ptr [rdi + 3], 0x3D

tail_done:
    add rdi, 4
    dec r13d
    jnz finish_line
    mov byte ptr [rdi], 10
    inc rdi
    mov r13d, 19

finish_line:
    cmp r13d, 19
    je write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    ja write_loop

    test r14d, r14d
    jz read_chunk

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
