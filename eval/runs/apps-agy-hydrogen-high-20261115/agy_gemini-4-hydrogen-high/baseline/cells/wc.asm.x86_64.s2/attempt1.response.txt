.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading

    add r14, rax
    lea r9, [rsi + rax]

byte_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    xor ecx, ecx
    cmp al, 0x0A
    sete cl
    add r12, rcx

    lea ecx, [rax - 9]
    cmp ecx, 4
    setbe cl
    cmp al, 0x20
    sete r8b
    or cl, r8b
    xor cl, 1

    mov r8d, r15d
    xor r8d, 1
    and r8b, cl
    movzx r8d, r8b
    add r13, r8
    movzx r15d, cl

    cmp rsi, r9
    jb byte_loop
    jmp read_loop

done_reading:
    lea r10, [rip + outbuf_end]
    mov rdi, r10

    dec rdi
    mov byte ptr [rdi], 10

    mov rax, r14
    call write_dec

    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call write_dec

    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call write_dec

    mov rsi, rdi
    mov rdx, r10
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jle exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

write_dec:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
.balign 16
buf:
    .space 65536
outbuf:
    .space 128
outbuf_end:
