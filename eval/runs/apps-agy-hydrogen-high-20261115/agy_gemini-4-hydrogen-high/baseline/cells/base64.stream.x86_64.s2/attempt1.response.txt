.globl _start

.text
_start:
    lea rsi, [rip + inbuf]
    xor r12d, r12d

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    cmp rax, -4
    je read_loop
    test rax, rax
    jle read_done
    add rsi, rax
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_prog

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + b64_table]

line_loop:
    cmp r12, 57
    jb last_line
    mov r13d, 19

full_group_loop:
    call encode3
    dec r13d
    jnz full_group_loop

    mov byte ptr [rdi], 10
    inc rdi
    sub r12, 57
    jnz line_loop
    jmp encode_done

last_line:
    cmp r12, 3
    jb rem_bytes

last_group_loop:
    call encode3
    sub r12, 3
    cmp r12, 3
    jae last_group_loop

rem_bytes:
    test r12, r12
    jz append_nl
    cmp r12, 1
    jne rem_two

    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 2
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 1], al
    mov word ptr [rdi + 2], 0x3D3D
    add rdi, 4
    jmp append_nl

rem_two:
    movzx eax, byte ptr [rsi]
    shl eax, 8
    movzx ecx, byte ptr [rsi + 1]
    or eax, ecx
    mov ecx, eax
    shr ecx, 10
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl
    mov ecx, eax
    shr ecx, 4
    and ecx, 63
    mov cl, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 1], cl
    and eax, 15
    shl eax, 2
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 2], al
    mov byte ptr [rdi + 3], 0x3D
    add rdi, 4

append_nl:
    mov byte ptr [rdi], 10
    inc rdi

encode_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_prog

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, -4
    je write_loop
    test rax, rax
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

encode3:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx

    mov ecx, eax
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    shl edx, 8
    shr eax, 6
    mov ecx, eax
    and ecx, 63
    movzx ecx, byte ptr [rbx + rcx]
    or edx, ecx
    shl edx, 8
    shr eax, 6
    mov ecx, eax
    and ecx, 63
    movzx ecx, byte ptr [rbx + rcx]
    or edx, ecx
    shl edx, 8
    shr eax, 6
    movzx ecx, byte ptr [rbx + rax]
    or edx, ecx
    mov dword ptr [rdi], edx

    add rsi, 3
    add rdi, 4
    ret

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 98304
