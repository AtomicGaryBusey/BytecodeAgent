.equ INBUF_SIZE, 69632
.equ OUTBUF_SIZE, 65536
.equ TAPE_SIZE, 32768
.equ PROG_SIZE, 69632
.equ TABLE_SIZE, 278528

.bss
.balign 16
inbuf:
    .zero INBUF_SIZE
.balign 16
outbuf:
    .zero OUTBUF_SIZE
.balign 16
tape:
    .zero TAPE_SIZE
.balign 16
prog:
    .zero PROG_SIZE
.balign 16
jumps:
    .zero TABLE_SIZE
.balign 16
bstack:
    .zero TABLE_SIZE

.section .rodata
.balign 8
dispatch_table:
    .quad op_halt
    .quad op_gt
    .quad op_lt
    .quad op_plus
    .quad op_minus
    .quad op_dot
    .quad op_comma
    .quad op_lbrack
    .quad op_rbrack

.text
.globl _start
_start:
    # Read all of stdin into inbuf
    lea rsi, [rip + inbuf]
    mov r15, rsi
    add r15, INBUF_SIZE

read_loop:
    mov rdx, r15
    sub rdx, rsi
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    jmp read_loop

read_done:
    mov r15, rsi

    # Parse program, check bracket balance, build jump table
    lea r14, [rip + inbuf]
    lea r12, [rip + prog]
    lea r13, [rip + jumps]
    lea r8, [rip + bstack]
    mov r9, r8
    xor ebx, ebx

parse_loop:
    cmp r14, r15
    je parse_done
    mov al, byte ptr [r14]
    inc r14
    cmp al, 0x21
    je parse_done
    cmp al, 0x3e
    je p_gt
    cmp al, 0x3c
    je p_lt
    cmp al, 0x2b
    je p_plus
    cmp al, 0x2d
    je p_minus
    cmp al, 0x2e
    je p_dot
    cmp al, 0x2c
    je p_comma
    cmp al, 0x5b
    je p_lbrack
    cmp al, 0x5d
    je p_rbrack
    jmp parse_loop

p_gt:
    mov byte ptr [r12 + rbx], 1
    inc rbx
    jmp parse_loop

p_lt:
    mov byte ptr [r12 + rbx], 2
    inc rbx
    jmp parse_loop

p_plus:
    mov byte ptr [r12 + rbx], 3
    inc rbx
    jmp parse_loop

p_minus:
    mov byte ptr [r12 + rbx], 4
    inc rbx
    jmp parse_loop

p_dot:
    mov byte ptr [r12 + rbx], 5
    inc rbx
    jmp parse_loop

p_comma:
    mov byte ptr [r12 + rbx], 6
    inc rbx
    jmp parse_loop

p_lbrack:
    mov byte ptr [r12 + rbx], 7
    mov dword ptr [r9], ebx
    add r9, 4
    inc rbx
    jmp parse_loop

p_rbrack:
    cmp r9, r8
    je exit_error
    sub r9, 4
    mov eax, dword ptr [r9]
    mov byte ptr [r12 + rbx], 8
    mov dword ptr [r13 + rax*4], ebx
    mov dword ptr [r13 + rbx*4], eax
    inc rbx
    jmp parse_loop

parse_done:
    cmp r9, r8
    jne exit_error
    mov byte ptr [r12 + rbx], 0

    # Initialize interpreter state
    xor ebx, ebx
    lea rbp, [rip + tape]
    lea r8, [rip + outbuf]
    mov r9, r8
    add r9, OUTBUF_SIZE
    lea r10, [rip + dispatch_table]

loop_next:
    movzx eax, byte ptr [r12 + rbx]
    jmp qword ptr [r10 + rax*8]

op_halt:
    call flush_out
    xor edi, edi
    mov eax, 60
    syscall

op_gt:
    inc rbp
    inc rbx
    jmp loop_next

op_lt:
    dec rbp
    inc rbx
    jmp loop_next

op_plus:
    inc byte ptr [rbp]
    inc rbx
    jmp loop_next

op_minus:
    dec byte ptr [rbp]
    inc rbx
    jmp loop_next

op_dot:
    mov al, byte ptr [rbp]
    mov byte ptr [r8], al
    inc r8
    cmp r8, r9
    je 1f
    inc rbx
    jmp loop_next
1:
    call flush_out
    inc rbx
    jmp loop_next

op_comma:
    xor eax, eax
    cmp r14, r15
    jae 1f
    mov al, byte ptr [r14]
    inc r14
1:
    mov byte ptr [rbp], al
    inc rbx
    jmp loop_next

op_lbrack:
    cmp byte ptr [rbp], 0
    jne 1f
    mov ebx, dword ptr [r13 + rbx*4]
1:
    inc rbx
    jmp loop_next

op_rbrack:
    cmp byte ptr [rbp], 0
    je 1f
    mov ebx, dword ptr [r13 + rbx*4]
1:
    inc rbx
    jmp loop_next

flush_out:
    lea rsi, [rip + outbuf]
    mov rdx, r8
    sub rdx, rsi
    jz flush_done
flush_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle flush_done
    add rsi, rax
    sub rdx, rax
    jnz flush_loop
flush_done:
    lea r8, [rip + outbuf]
    ret

exit_error:
    mov edi, 1
    mov eax, 60
    syscall
