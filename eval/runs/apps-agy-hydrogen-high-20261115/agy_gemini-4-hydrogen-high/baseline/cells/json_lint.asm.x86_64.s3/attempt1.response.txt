.globl _start

.text

_start:
    lea rbx, [rip + input_buf]
    xor r12d, r12d
    mov r13d, 131072

read_loop:
    mov rdx, r13
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov rsi, rbx
    add r12, rbx

    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne parse_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall

    mov eax, 60
    xor edi, edi
    syscall

parse_error:
    mov rax, rsi
    sub rax, rbx
    lea rdi, [rip + err_buf + 31]
    mov byte ptr [rdi], 10
    mov ecx, 10

itoa_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz itoa_loop

    sub rdi, 9
    mov rax, qword ptr [rip + err_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_prefix + 8]
    mov byte ptr [rdi + 8], al

    lea rdx, [rip + err_buf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp rsi, r12
    jae ws_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je ws_next
    cmp al, 0x09
    je ws_next
    cmp al, 0x0A
    je ws_next
    cmp al, 0x0D
    jne ws_done
ws_next:
    inc rsi
    jmp skip_ws
ws_done:
    ret

parse_value:
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22                # '"'
    je parse_string
    cmp al, 0x7B                # '{'
    je parse_object
    cmp al, 0x5B                # '['
    je parse_array
    cmp al, 0x74                # 't'
    je parse_true
    cmp al, 0x66                # 'f'
    je parse_false
    cmp al, 0x6E                # 'n'
    je parse_null
    cmp al, 0x2D                # '-'
    je parse_number
    sub al, 0x30                # '0'
    cmp al, 9
    jbe parse_number
    jmp parse_error

parse_true:
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x72    # 'r'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x75    # 'u'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x65    # 'e'
    jne parse_error
    inc rsi
    ret

parse_false:
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x61    # 'a'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x6C    # 'l'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x73    # 's'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x65    # 'e'
    jne parse_error
    inc rsi
    ret

parse_null:
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x75    # 'u'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x6C    # 'l'
    jne parse_error
    inc rsi
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x6C    # 'l'
    jne parse_error
    inc rsi
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x5D    # ']'
    jne arr_loop
    inc rsi
    ret

arr_loop:
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x5D                # ']'
    je arr_close
    cmp al, 0x2C                # ','
    jne parse_error
    inc rsi
    call skip_ws
    jmp arr_loop

arr_close:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x7D    # '}'
    jne obj_loop
    inc rsi
    ret

obj_loop:
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x22    # '"'
    jne parse_error
    call parse_string
    call skip_ws
    cmp rsi, r12
    jae parse_error
    cmp byte ptr [rsi], 0x3A    # ':'
    jne parse_error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x7D                # '}'
    je obj_close
    cmp al, 0x2C                # ','
    jne parse_error
    inc rsi
    call skip_ws
    jmp obj_loop

obj_close:
    inc rsi
    ret

parse_string:
    inc rsi
str_loop:
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22                # '"'
    je str_done
    cmp al, 0x20
    jb parse_error
    cmp al, 0x5C                # '\'
    je str_escape
    inc rsi
    jmp str_loop

str_escape:
    inc rsi
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22                # '"'
    je str_esc_ok
    cmp al, 0x5C                # '\'
    je str_esc_ok
    cmp al, 0x2F                # '/'
    je str_esc_ok
    cmp al, 0x62                # 'b'
    je str_esc_ok
    cmp al, 0x66                # 'f'
    je str_esc_ok
    cmp al, 0x6E                # 'n'
    je str_esc_ok
    cmp al, 0x72                # 'r'
    je str_esc_ok
    cmp al, 0x74                # 't'
    je str_esc_ok
    cmp al, 0x75                # 'u'
    jne parse_error
    inc rsi
    mov ecx, 4
str_hex_loop:
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x30                # '0'
    jb parse_error
    cmp al, 0x39                # '9'
    jbe str_hex_ok
    or al, 0x20
    sub al, 0x61                # 'a'
    cmp al, 5
    ja parse_error
str_hex_ok:
    inc rsi
    dec ecx
    jnz str_hex_loop
    jmp str_loop

str_esc_ok:
    inc rsi
    jmp str_loop

str_done:
    inc rsi
    ret

parse_number:
    cmp byte ptr [rsi], 0x2D    # '-'
    jne num_int
    inc rsi
num_int:
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x30                # '0'
    je num_zero
    sub al, 0x31                # '1'
    cmp al, 8
    ja parse_error
    inc rsi
num_int_loop:
    cmp rsi, r12
    jae num_done
    movzx eax, byte ptr [rsi]
    sub al, 0x30                # '0'
    cmp al, 9
    ja num_after_int
    inc rsi
    jmp num_int_loop

num_zero:
    inc rsi

num_after_int:
    cmp rsi, r12
    jae num_done
    cmp byte ptr [rsi], 0x2E    # '.'
    jne num_check_exp
    inc rsi
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    sub al, 0x30                # '0'
    cmp al, 9
    ja parse_error
    inc rsi
num_frac_loop:
    cmp rsi, r12
    jae num_done
    movzx eax, byte ptr [rsi]
    sub al, 0x30                # '0'
    cmp al, 9
    ja num_check_exp
    inc rsi
    jmp num_frac_loop

num_check_exp:
    cmp rsi, r12
    jae num_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x65                # 'e'
    je num_exp
    cmp al, 0x45                # 'E'
    jne num_done
num_exp:
    inc rsi
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
    cmp al, 0x2B                # '+'
    je num_exp_sign
    cmp al, 0x2D                # '-'
    jne num_exp_digit
num_exp_sign:
    inc rsi
    cmp rsi, r12
    jae parse_error
    movzx eax, byte ptr [rsi]
num_exp_digit:
    sub al, 0x30                # '0'
    cmp al, 9
    ja parse_error
    inc rsi
num_exp_loop:
    cmp rsi, r12
    jae num_done
    movzx eax, byte ptr [rsi]
    sub al, 0x30                # '0'
    cmp al, 9
    ja num_done
    inc rsi
    jmp num_exp_loop

num_done:
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "

.bss
.balign 16
input_buf:
    .zero 131072
err_buf:
    .zero 32
