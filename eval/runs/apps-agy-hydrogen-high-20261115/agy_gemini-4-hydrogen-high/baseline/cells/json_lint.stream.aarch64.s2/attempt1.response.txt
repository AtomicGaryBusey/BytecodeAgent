.text
.globl _start

_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    mov  x20, #0
    movz x21, #2, lsl #16

read_loop:
    mov  x0, #0
    add  x1, x19, x20
    sub  x2, x21, x20
    cbz  x2, read_done
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    add  x20, x19, x20
    mov  x21, x19
    adrp x22, nest_stack
    add  x22, x22, :lo12:nest_stack
    mov  x23, x22

expect_value:
    bl   skip_ws
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]

    cmp  w0, #'{'
    b.eq handle_lbrace
    cmp  w0, #'['
    b.eq handle_lbracket
    cmp  w0, #0x22
    b.eq handle_string
    cmp  w0, #'t'
    b.eq handle_true
    cmp  w0, #'f'
    b.eq handle_false
    cmp  w0, #'n'
    b.eq handle_null
    cmp  w0, #'-'
    b.eq handle_number
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.ls handle_number
    b    fail

handle_lbrace:
    add  x21, x21, #1
    bl   skip_ws
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #'}'
    b.ne 1f
    add  x21, x21, #1
    b    after_value
1:
    mov  w1, #1
    strb w1, [x23], #1
    b    expect_key_no_ws

handle_lbracket:
    add  x21, x21, #1
    bl   skip_ws
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #']'
    b.ne 1f
    add  x21, x21, #1
    b    after_value
1:
    mov  w1, #0
    strb w1, [x23], #1
    b    expect_value

expect_key:
    bl   skip_ws
expect_key_no_ws:
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #0x22
    b.ne fail
    bl   parse_string
    bl   skip_ws
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #':'
    b.ne fail
    add  x21, x21, #1
    b    expect_value

handle_string:
    bl   parse_string
    b    after_value

handle_true:
    adrp x1, lit_rue
    add  x1, x1, :lo12:lit_rue
    b    match_tail

handle_false:
    adrp x1, lit_alse
    add  x1, x1, :lo12:lit_alse
    b    match_tail

handle_null:
    adrp x1, lit_ull
    add  x1, x1, :lo12:lit_ull

match_tail:
    add  x21, x21, #1
    ldrb w2, [x1], #1
    cbz  w2, after_value
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, w2
    b.eq match_tail
    b    fail

handle_number:
    ldrb w0, [x21]
    cmp  w0, #'-'
    b.ne 1f
    add  x21, x21, #1
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
1:
    cmp  w0, #'0'
    b.ne 2f
    add  x21, x21, #1
    b    num_frac
2:
    sub  w1, w0, #'1'
    cmp  w1, #8
    b.hi fail
    add  x21, x21, #1
3:
    cmp  x21, x20
    b.hs after_value
    ldrb w0, [x21]
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.hi num_frac
    add  x21, x21, #1
    b    3b

num_frac:
    cmp  x21, x20
    b.hs after_value
    ldrb w0, [x21]
    cmp  w0, #'.'
    b.ne num_exp
    add  x21, x21, #1
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.hi fail
    add  x21, x21, #1
4:
    cmp  x21, x20
    b.hs after_value
    ldrb w0, [x21]
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.hi num_exp
    add  x21, x21, #1
    b    4b

num_exp:
    cmp  w0, #'e'
    b.eq 5f
    cmp  w0, #'E'
    b.ne after_value
5:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #'+'
    b.eq 6f
    cmp  w0, #'-'
    b.ne 7f
6:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
7:
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.hi fail
    add  x21, x21, #1
8:
    cmp  x21, x20
    b.hs after_value
    ldrb w0, [x21]
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.hi after_value
    add  x21, x21, #1
    b    8b

after_value:
    bl   skip_ws
    cmp  x23, x22
    b.eq 1f
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    ldurb w1, [x23, #-1]
    cmp  w0, #','
    b.ne 2f
    add  x21, x21, #1
    cbnz w1, expect_key
    b    expect_value
2:
    cbnz w1, 3f
    cmp  w0, #']'
    b.ne fail
    add  x21, x21, #1
    sub  x23, x23, #1
    b    after_value
3:
    cmp  w0, #'}'
    b.ne fail
    add  x21, x21, #1
    sub  x23, x23, #1
    b    after_value

1:
    cmp  x21, x20
    b.lo fail

    mov  x0, #1
    adrp x1, ok_msg
    add  x1, x1, :lo12:ok_msg
    mov  x2, #3
    mov  x8, #64
    svc  #0

    mov  x0, #0
    mov  x8, #93
    svc  #0

fail:
    sub  x0, x21, x19
    sub  sp, sp, #32
    add  x4, sp, #31
    mov  w5, #0x0A
    strb w5, [x4]
    mov  x6, #10
1:
    udiv x7, x0, x6
    msub x8, x7, x6, x0
    add  w8, w8, #'0'
    strb w8, [x4, #-1]!
    mov  x0, x7
    cbnz x0, 1b

    sub  x1, x4, #9
    adrp x2, err_prefix
    add  x2, x2, :lo12:err_prefix
    mov  w3, #9
2:
    ldrb w5, [x2], #1
    strb w5, [x1], #1
    subs w3, w3, #1
    b.ne 2b

    sub  x1, x4, #9
    add  x2, sp, #32
    sub  x2, x2, x1
    mov  x0, #1
    mov  x8, #64
    svc  #0

    mov  x0, #1
    mov  x8, #93
    svc  #0

skip_ws:
    cmp  x21, x20
    b.hs 2f
1:
    ldrb w0, [x21]
    cmp  w0, #0x20
    b.hi 2f
    b.eq 3f
    cmp  w0, #0x0A
    b.eq 3f
    cmp  w0, #0x09
    b.eq 3f
    cmp  w0, #0x0D
    b.ne 2f
3:
    add  x21, x21, #1
    cmp  x21, x20
    b.lo 1b
2:
    ret

parse_string:
    add  x21, x21, #1
1:
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #0x22
    b.eq 2f
    cmp  w0, #0x5C
    b.eq 3f
    cmp  w0, #0x20
    b.lo fail
    add  x21, x21, #1
    b    1b

2:
    add  x21, x21, #1
    ret

3:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    cmp  w0, #0x22
    b.eq 4f
    cmp  w0, #0x5C
    b.eq 4f
    cmp  w0, #'/'
    b.eq 4f
    cmp  w0, #'b'
    b.eq 4f
    cmp  w0, #'f'
    b.eq 4f
    cmp  w0, #'n'
    b.eq 4f
    cmp  w0, #'r'
    b.eq 4f
    cmp  w0, #'t'
    b.eq 4f
    cmp  w0, #'u'
    b.ne fail

    add  x21, x21, #1
    mov  w2, #4
5:
    cmp  x21, x20
    b.hs fail
    ldrb w0, [x21]
    sub  w1, w0, #'0'
    cmp  w1, #9
    b.ls 6f
    orr  w0, w0, #0x20
    sub  w1, w0, #'a'
    cmp  w1, #5
    b.hi fail
6:
    add  x21, x21, #1
    subs w2, w2, #1
    b.ne 5b
    b    1b

4:
    add  x21, x21, #1
    b    1b

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_rue:
    .asciz "rue"
lit_alse:
    .asciz "alse"
lit_ull:
    .asciz "ull"

.bss
.p2align 4

    .space 131072
.p2align 4

    .space 65536
