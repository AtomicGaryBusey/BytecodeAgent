.globl _start
.text

_start:
    // Read all of stdin into in_buf
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0
    movz x21, #1, lsl #16
    add x21, x21, #16

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    // Split at first '!', compact valid commands into prog_buf, check brackets
    mov x0, #0
    adrp x1, prog_buf
    add x1, x1, :lo12:prog_buf
    mov x2, #0
    adrp x3, bstack
    add x3, x3, :lo12:bstack
    mov x4, #0
    adrp x5, jumps
    add x5, x5, :lo12:jumps

scan_loop:
    cmp x0, x20
    b.hs scan_no_bang
    ldrb w6, [x19, x0]
    add x0, x0, #1

    cmp w6, #33
    b.eq scan_bang

    cmp w6, #43
    b.eq cmd_plus
    cmp w6, #45
    b.eq cmd_minus
    cmp w6, #62
    b.eq cmd_gt
    cmp w6, #60
    b.eq cmd_lt
    cmp w6, #46
    b.eq cmd_dot
    cmp w6, #44
    b.eq cmd_comma
    cmp w6, #91
    b.eq cmd_lbrack
    cmp w6, #93
    b.eq cmd_rbrack
    b scan_loop

cmd_plus:
    mov w7, #0
    b store_cmd
cmd_minus:
    mov w7, #1
    b store_cmd
cmd_gt:
    mov w7, #2
    b store_cmd
cmd_lt:
    mov w7, #3
    b store_cmd
cmd_dot:
    mov w7, #4
    b store_cmd
cmd_comma:
    mov w7, #5
    b store_cmd

store_cmd:
    strb w7, [x1, x2]
    add x2, x2, #1
    b scan_loop

cmd_lbrack:
    mov w7, #6
    strb w7, [x1, x2]
    add x2, x2, #1
    str w2, [x3, x4, lsl #2]
    add x4, x4, #1
    b scan_loop

cmd_rbrack:
    cbz x4, unbalanced_exit
    sub x4, x4, #1
    ldr w7, [x3, x4, lsl #2]
    mov w8, #7
    strb w8, [x1, x2]
    add x2, x2, #1
    str w2, [x5, x7, lsl #2]
    str w7, [x5, x2, lsl #2]
    b scan_loop

scan_no_bang:
    add x24, x19, x20
    mov x25, x24
    b check_brackets

scan_bang:
    add x24, x19, x0
    add x25, x19, x20

check_brackets:
    cbnz x4, unbalanced_exit
    mov w7, #8
    strb w7, [x1, x2]

    // Initialize interpreter registers
    mov x19, x1
    mov x20, #0
    adr x21, jump_table
    adrp x22, tape
    add x22, x22, :lo12:tape
    mov x23, x5
    adrp x26, out_buf
    add x26, x26, :lo12:out_buf
    mov x27, #0

dispatch:
    ldrb w0, [x19, x20]
    add x20, x20, #1
    add x1, x21, x0, lsl #2
    br x1

jump_table:
    b op_plus
    b op_minus
    b op_gt
    b op_lt
    b op_dot
    b op_comma
    b op_lbrack
    b op_rbrack
    b op_halt

op_plus:
    ldrb w2, [x22]
    add w2, w2, #1
    strb w2, [x22]
    b dispatch

op_minus:
    ldrb w2, [x22]
    sub w2, w2, #1
    strb w2, [x22]
    b dispatch

op_gt:
    add x22, x22, #1
    b dispatch

op_lt:
    sub x22, x22, #1
    b dispatch

op_dot:
    ldrb w2, [x22]
    strb w2, [x26, x27]
    add x27, x27, #1
    cmp x27, #65536
    b.lo dispatch
    bl flush_out
    mov x27, #0
    b dispatch

op_comma:
    cmp x24, x25
    b.hs 1f
    ldrb w2, [x24], #1
    strb w2, [x22]
    b dispatch
1:
    strb wzr, [x22]
    b dispatch

op_lbrack:
    ldrb w2, [x22]
    cbnz w2, dispatch
    ldr w20, [x23, x20, lsl #2]
    b dispatch

op_rbrack:
    ldrb w2, [x22]
    cbz w2, dispatch
    ldr w20, [x23, x20, lsl #2]
    b dispatch

op_halt:
    bl flush_out
    mov x0, #0
    mov x8, #93
    svc #0

unbalanced_exit:
    mov x0, #1
    mov x8, #93
    svc #0

flush_out:
    mov x28, #0
1:
    cmp x28, x27
    b.hs 2f
    mov x0, #1
    add x1, x26, x28
    sub x2, x27, x28
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x28, x28, x0
    b 1b
2:
    ret

.bss
.balign 16
in_buf:
    .zero 65552
out_buf:
    .zero 65552
tape:
    .zero 30016
prog_buf:
    .zero 65552
jumps:
    .zero 262160
bstack:
    .zero 262160
