.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 65536
.equ LINES_BYTES, 524288

.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #INBUF_SIZE

read_loop:
    subs x2, x21, x20
    b.eq read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_ok

    adrp x21, lines
    add x21, x21, :lo12:lines
    mov x22, #0
    mov x2, x19
    add x3, x19, x20
    mov x4, x19
    mov x5, x21

parse_loop:
    cmp x2, x3
    b.hs parse_tail
    ldrb w6, [x2], #1
    cmp w6, #10
    b.ne parse_loop
    sub x7, x2, #1
    sub x7, x7, x4
    stp x4, x7, [x5], #16
    add x22, x22, #1
    mov x4, x2
    b parse_loop

parse_tail:
    cmp x4, x3
    b.hs parse_done
    sub x7, x3, x4
    stp x4, x7, [x5], #16
    add x22, x22, #1

parse_done:
    cmp x22, #1
    b.ls sort_done

    lsr x23, x22, #1

heapify_loop:
    cbz x23, heapify_done
    sub x23, x23, #1
    mov x0, x21
    mov x1, x23
    mov x2, x22
    bl sift_down
    b heapify_loop

heapify_done:
    mov x23, x22

sort_loop:
    cmp x23, #1
    b.ls sort_done
    sub x23, x23, #1
    add x9, x21, x23, lsl #4
    ldp x10, x11, [x21]
    ldp x12, x13, [x9]
    stp x12, x13, [x21]
    stp x10, x11, [x9]
    mov x0, x21
    mov x1, #0
    mov x2, x23
    bl sift_down
    b sort_loop

sort_done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x25, #0
    mov x26, #OUTBUF_SIZE

out_line_loop:
    cbz x22, out_flush
    ldp x19, x20, [x21], #16
    sub x22, x22, #1

out_copy_loop:
    cbz x20, out_newline
    cmp x25, x26
    b.lo 1f
    mov x0, x24
    mov x1, x25
    bl write_all
    mov x25, #0
1:
    ldrb w11, [x19], #1
    strb w11, [x24, x25]
    add x25, x25, #1
    sub x20, x20, #1
    b out_copy_loop

out_newline:
    cmp x25, x26
    b.lo 2f
    mov x0, x24
    mov x1, x25
    bl write_all
    mov x25, #0
2:
    mov w11, #10
    strb w11, [x24, x25]
    add x25, x25, #1
    b out_line_loop

out_flush:
    cbz x25, exit_ok
    mov x0, x24
    mov x1, x25
    bl write_all

exit_ok:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// sift_down(x0 = base, x1 = root, x2 = limit)
sift_down:
    stp x19, x20, [sp, #-48]!
    stp x21, x22, [sp, #16]
    stp x23, x30, [sp, #32]
    mov x19, x0
    mov x20, x1
    mov x21, x2

1:
    lsl x22, x20, #1
    add x22, x22, #1
    cmp x22, x21
    b.hs 3f
    add x23, x22, #1
    cmp x23, x21
    b.hs 2f
    add x0, x19, x22, lsl #4
    add x1, x19, x23, lsl #4
    bl cmp_lines
    cmp w0, #0
    csel x22, x23, x22, lt

2:
    add x0, x19, x20, lsl #4
    add x1, x19, x22, lsl #4
    bl cmp_lines
    cmp w0, #0
    b.ge 3f
    add x9, x19, x20, lsl #4
    add x10, x19, x22, lsl #4
    ldp x11, x12, [x9]
    ldp x13, x14, [x10]
    stp x13, x14, [x9]
    stp x11, x12, [x10]
    mov x20, x22
    b 1b

3:
    ldp x23, x30, [sp, #32]
    ldp x21, x22, [sp, #16]
    ldp x19, x20, [sp], #48
    ret

// cmp_lines(x0 = &lineA, x1 = &lineB) -> w0 (-1, 0, 1)
cmp_lines:
    ldp x2, x3, [x0]
    ldp x4, x5, [x1]
    cmp x3, x5
    csel x6, x3, x5, ls
    cbz x6, 2f

1:
    ldrb w7, [x2], #1
    ldrb w8, [x4], #1
    cmp w7, w8
    b.ne 3f
    subs x6, x6, #1
    b.ne 1b

2:
    cmp x3, x5
    b.lo 4f
    b.hi 5f
    mov w0, #0
    ret

3:
    b.lo 4f

5:
    mov w0, #1
    ret

4:
    mov w0, #-1
    ret

// write_all(x0 = buf, x1 = count)
write_all:
    mov x3, x0
    mov x4, x1
1:
    cbz x4, 2f
    mov x0, #1
    mov x1, x3
    mov x2, x4
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 2f
    add x3, x3, x0
    sub x4, x4, x0
    b 1b
2:
    ret

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 16
lines:
    .space LINES_BYTES
