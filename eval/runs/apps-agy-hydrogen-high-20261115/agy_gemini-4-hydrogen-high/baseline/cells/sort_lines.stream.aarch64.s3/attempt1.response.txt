.globl _start

.text
_start:
    // Read all of stdin into in_buf
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_ok

    // Parse lines into lines array: (ptr, len)
    adrp x22, lines
    add x22, x22, :lo12:lines
    mov x23, #0
    mov x0, x19
    add x1, x19, x20
    mov x2, x19
    mov w3, #10

parse_loop:
    cmp x0, x1
    b.hs parse_after
    ldrb w4, [x0]
    cmp w4, w3
    b.ne parse_next
    sub x5, x0, x2
    add x6, x22, x23, lsl #4
    stp x2, x5, [x6]
    add x23, x23, #1
    add x0, x0, #1
    mov x2, x0
    b parse_loop

parse_next:
    add x0, x0, #1
    b parse_loop

parse_after:
    cmp x2, x1
    b.hs parse_done
    sub x5, x1, x2
    add x6, x22, x23, lsl #4
    stp x2, x5, [x6]
    add x23, x23, #1

parse_done:
    cbz x23, exit_ok

    // Sort lines[0 .. n_lines) using mergesort
    adrp x24, tmp_lines
    add x24, x24, :lo12:tmp_lines
    mov x0, #0
    mov x1, x23
    bl mergesort

    // Build output in out_buf
    adrp x19, out_buf
    add x19, x19, :lo12:out_buf
    mov x20, x19
    mov x0, #0
    mov w5, #10

out_line_loop:
    cmp x0, x23
    b.hs out_write
    add x1, x22, x0, lsl #4
    ldp x2, x3, [x1]

copy_bytes:
    cbz x3, copy_nl
    ldrb w4, [x2], #1
    strb w4, [x20], #1
    sub x3, x3, #1
    b copy_bytes

copy_nl:
    strb w5, [x20], #1
    add x0, x0, #1
    b out_line_loop

out_write:
    sub x21, x20, x19
    mov x20, #0

write_loop:
    cmp x20, x21
    b.hs exit_ok
    mov x0, #1
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x20, x20, x0
    b write_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

// mergesort(x0 = lo, x1 = hi)
// Expects x22 = lines, x24 = tmp_lines
mergesort:
    sub x2, x1, x0
    cmp x2, #1
    b.le ms_ret

    stp x29, x30, [sp, #-48]!
    stp x19, x20, [sp, #16]
    str x21, [sp, #32]

    mov x19, x0
    mov x21, x1
    add x20, x0, x1
    lsr x20, x20, #1

    mov x0, x19
    mov x1, x20
    bl mergesort

    mov x0, x20
    mov x1, x21
    bl mergesort

    mov x9, x19
    mov x10, x20
    mov x11, x19

merge_loop:
    cmp x9, x20
    b.hs merge_rem_j
    cmp x10, x21
    b.hs merge_rem_i

    add x12, x22, x9, lsl #4
    ldp x0, x2, [x12]
    add x13, x22, x10, lsl #4
    ldp x1, x3, [x13]

    mov x12, x0
    mov x13, x2
    mov x14, x1
    mov x15, x3
    bl cmp_le
    add x4, x24, x11, lsl #4
    cbz w0, take_j

take_i:
    stp x12, x13, [x4]
    add x9, x9, #1
    add x11, x11, #1
    b merge_loop

take_j:
    stp x14, x15, [x4]
    add x10, x10, #1
    add x11, x11, #1
    b merge_loop

merge_rem_i:
    cmp x9, x20
    b.hs copy_back
    add x12, x22, x9, lsl #4
    ldp x0, x1, [x12]
    add x4, x24, x11, lsl #4
    stp x0, x1, [x4]
    add x9, x9, #1
    add x11, x11, #1
    b merge_rem_i

merge_rem_j:
    cmp x10, x21
    b.hs copy_back
    add x13, x22, x10, lsl #4
    ldp x0, x1, [x13]
    add x4, x24, x11, lsl #4
    stp x0, x1, [x4]
    add x10, x10, #1
    add x11, x11, #1
    b merge_rem_j

copy_back:
    mov x9, x19

copy_loop:
    cmp x9, x21
    b.hs ms_done
    add x4, x24, x9, lsl #4
    ldp x0, x1, [x4]
    add x5, x22, x9, lsl #4
    stp x0, x1, [x5]
    add x9, x9, #1
    b copy_loop

ms_done:
    ldr x21, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #48

ms_ret:
    ret

// cmp_le(x0 = ptr1, x1 = ptr2, x2 = len1, x3 = len2) -> w0 = 1 if line1 <= line2 else 0
// Uses only x0..x6
cmp_le:
    cmp x2, x3
    csel x6, x2, x3, ls

cmp_loop:
    cbz x6, cmp_len
    ldrb w4, [x0], #1
    ldrb w5, [x1], #1
    cmp w4, w5
    b.lo cmp_ret_true
    b.hi cmp_ret_false
    sub x6, x6, #1
    b cmp_loop

cmp_len:
    cmp x2, x3
    cset w0, ls
    ret

cmp_ret_true:
    mov w0, #1
    ret

cmp_ret_false:
    mov w0, #0
    ret

.bss
.align 4
in_buf:
    .space 65536
out_buf:
    .space 65536
lines:
    .space 16384
tmp_lines:
    .space 16384
