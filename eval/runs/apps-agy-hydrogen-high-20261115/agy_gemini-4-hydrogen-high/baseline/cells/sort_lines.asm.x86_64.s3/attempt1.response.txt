.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 65536
.equ MAX_LINES, 65536

.bss
.balign 16
in_buf:
    .zero IN_BUF_SIZE
.balign 16
out_buf:
    .zero OUT_BUF_SIZE
.balign 16
lines:
    .zero MAX_LINES * 8

.text
.globl _start

_start:
    lea rbx, [rip + in_buf]
    xor r13d, r13d

.read_loop:
    mov edx, IN_BUF_SIZE
    sub rdx, r13
    jz .read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    syscall
    test rax, rax
    jle .read_done
    add r13, rax
    jmp .read_loop

.read_done:
    test r13, r13
    jz .exit_ok

    lea rbp, [rip + lines]
    xor r14d, r14d
    xor r12d, r12d

.parse_loop:
    cmp r12, r13
    jae .parse_done
    mov rcx, r12
.scan_nl:
    cmp rcx, r13
    jae .found_end
    cmp byte ptr [rbx + rcx], 0x0A
    je .found_nl
    inc rcx
    jmp .scan_nl

.found_nl:
    mov rax, rcx
    sub rax, r12
    shl rax, 32
    or rax, r12
    mov qword ptr [rbp + r14 * 8], rax
    inc r14
    lea r12, [rcx + 1]
    jmp .parse_loop

.found_end:
    mov rax, rcx
    sub rax, r12
    shl rax, 32
    or rax, r12
    mov qword ptr [rbp + r14 * 8], rax
    inc r14
    mov r12, rcx

.parse_done:
    cmp r14, 1
    jbe .sort_done

    mov r15, r14
    shr r15, 1
.heapify_loop:
    test r15, r15
    jz .heapify_done
    dec r15
    mov r9, r15
    mov r10, r14
    call sift_down
    jmp .heapify_loop

.heapify_done:
    mov r15, r14
.extract_loop:
    cmp r15, 1
    jbe .sort_done
    dec r15
    mov rax, qword ptr [rbp]
    mov rdx, qword ptr [rbp + r15 * 8]
    mov qword ptr [rbp], rdx
    mov qword ptr [rbp + r15 * 8], rax
    xor r9d, r9d
    mov r10, r15
    call sift_down
    jmp .extract_loop

.sort_done:
    lea r13, [rip + out_buf]
    lea r12, [r13 + OUT_BUF_SIZE]
    mov rdi, r13
    xor r15d, r15d

.out_loop:
    cmp r15, r14
    jae .out_flush
    mov rax, qword ptr [rbp + r15 * 8]
    mov esi, eax
    add rsi, rbx
    mov rcx, rax
    shr rcx, 32

    mov rax, r12
    sub rax, rdi
    lea rdx, [rcx + 1]
    cmp rdx, rax
    jbe .out_copy

    push rsi
    push rcx
    mov rsi, r13
    mov rdx, rdi
    sub rdx, r13
    call write_all
    mov rdi, r13
    pop rcx
    pop rsi

    lea rdx, [rcx + 1]
    cmp rdx, OUT_BUF_SIZE
    jbe .out_copy
    mov rdx, rcx
    call write_all
    mov rdi, r13
    mov byte ptr [rdi], 0x0A
    inc rdi
    jmp .out_next

.out_copy:
    rep movsb
    mov byte ptr [rdi], 0x0A
    inc rdi

.out_next:
    inc r15
    jmp .out_loop

.out_flush:
    mov rsi, r13
    mov rdx, rdi
    sub rdx, r13
    call write_all

.exit_ok:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

# sift_down: r9 = root, r10 = end, rbp = lines, rbx = in_buf
sift_down:
    mov r11, qword ptr [rbp + r9 * 8]
.sift_loop:
    lea r12, [r9 + r9 + 1]
    cmp r12, r10
    jae .sift_done
    lea r13, [r12 + 1]
    cmp r13, r10
    jae .sift_no_right
    mov rax, qword ptr [rbp + r12 * 8]
    mov rdx, qword ptr [rbp + r13 * 8]
    call cmp_lines
    test eax, eax
    jge .sift_no_right
    mov r12, r13
.sift_no_right:
    mov rax, r11
    mov rdx, qword ptr [rbp + r12 * 8]
    call cmp_lines
    test eax, eax
    jge .sift_done
    mov rdx, qword ptr [rbp + r12 * 8]
    mov qword ptr [rbp + r9 * 8], rdx
    mov r9, r12
    jmp .sift_loop
.sift_done:
    mov qword ptr [rbp + r9 * 8], r11
    ret

# cmp_lines: rax = entry_a, rdx = entry_b, rbx = in_buf -> eax = -1, 0, 1
cmp_lines:
    mov esi, eax
    add rsi, rbx
    mov edi, edx
    add rdi, rbx
    shr rax, 32
    shr rdx, 32
    mov rcx, rax
    cmp rcx, rdx
    jbe 1f
    mov rcx, rdx
1:
    test rcx, rcx
    jz 2f
    repe cmpsb
    je 2f
    sbb eax, eax
    or eax, 1
    ret
2:
    cmp rax, rdx
    jb 3f
    ja 4f
    xor eax, eax
    ret
3:
    mov eax, -1
    ret
4:
    mov eax, 1
    ret

# write_all: rsi = buf, rdx = count
write_all:
    test rdx, rdx
    jz 2f
1:
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    ret
