.bss
.balign 16
buf:
    .space 65600

.text
.globl _start
.balign 4
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0                 // total bytes read
    movz x21, #1, lsl #16       // 65536
    add x21, x21, #64           // 65600 byte capacity

1:
    sub x2, x21, x20            // remaining capacity
    cbz x2, 2f
    mov x0, #0                  // fd 0 (stdin)
    add x1, x19, x20            // buf + total_read
    mov x8, #63                 // sys_read
    svc #0
    cmp x0, #0
    b.le 2f
    add x20, x20, x0
    b 1b

2:
    cbz x20, 9f                 // empty input -> exit 0

    // Ensure input ends with '\n'
    sub x0, x20, #1
    ldrb w1, [x19, x0]
    cmp w1, #10
    b.eq 3f
    mov w1, #10
    strb w1, [x19, x20]
    add x20, x20, #1

3:
    mov x9, x19                 // line_start = buf
    add x10, x19, x20           // buf_end = buf + total_len

4:
    cmp x9, x10
    b.hs 8f                     // done reversing all lines
    mov x11, x9                 // scan = line_start

5:
    ldrb w12, [x11]
    cmp w12, #10
    b.eq 6f
    add x11, x11, #1
    b 5b

6:
    mov x12, x9                 // left = line_start
    sub x13, x11, #1            // right = newline_ptr - 1
    add x9, x11, #1             // next line_start = newline_ptr + 1

7:
    cmp x12, x13
    b.hs 4b
    ldrb w14, [x12]
    ldrb w15, [x13]
    strb w15, [x12], #1
    strb w14, [x13], #-1
    b 7b

8:
    mov x21, #0                 // total bytes written

0:
    cmp x21, x20
    b.hs 9f
    mov x0, #1                  // fd 1 (stdout)
    add x1, x19, x21            // buf + written
    sub x2, x20, x21            // remaining bytes to write
    mov x8, #64                 // sys_write
    svc #0
    cmp x0, #0
    b.le 9f
    add x21, x21, x0
    b 0b

9:
    mov x0, #0                  // exit status 0
    mov x8, #93                 // sys_exit
    svc #0
