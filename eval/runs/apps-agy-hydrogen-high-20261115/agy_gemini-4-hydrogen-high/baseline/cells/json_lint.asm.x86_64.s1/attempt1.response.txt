.globl _start

.equ BUF_SIZE, 65600

.text

_start:
    lea r12, [rip + buf]
    xor r14d, r14d

read_loop:
    mov edx, BUF_SIZE
    sub rdx, r14
    jbe read_done
    lea rsi, [r12 + r14]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop

read_done:
    lea r13, [r12 + r14]
    mov rsi, r12

    call skip_ws
    cmp rsi, r13
    jae error
    call parse_value
    call skip_ws
    cmp rsi, r13
    jne error

    lea rsi, [rip + ok_msg]
    mov edx, 3
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
    cmp rsi, r13
    jae 2f
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    jne 2f
3:
    inc rsi
    cmp rsi, r13
    jb 1b
2:
    ret

parse_value:
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6E
    je parse_null
    cmp al, 0x2D
    je parse_number
    sub eax, 0x30
    cmp eax, 10
    jb parse_number
    jmp error

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x7D
    jne 1f
    inc rsi
    ret
1:
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x3A
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x7D
    je 2f
    cmp al, 0x2C
    jne error
    inc rsi
    call skip_ws
    jmp 1b
2:
    inc rsi
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x5D
    jne 1f
    inc rsi
    ret
1:
    call parse_value
    call skip_ws
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x5D
    je 2f
    cmp al, 0x2C
    jne error
    inc rsi
    call skip_ws
    jmp 1b
2:
    inc rsi
    ret

parse_string:
    inc rsi
str_loop:
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je str_done
    cmp al, 0x5C
    je str_escape
    cmp al, 0x20
    jb error
    inc rsi
    jmp str_loop
str_done:
    inc rsi
    ret

str_escape:
    inc rsi
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x22
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, 0x2F
    je str_esc_ok
    cmp al, 0x62
    je str_esc_ok
    cmp al, 0x66
    je str_esc_ok
    cmp al, 0x6E
    je str_esc_ok
    cmp al, 0x72
    je str_esc_ok
    cmp al, 0x74
    je str_esc_ok
    cmp al, 0x75
    jne error
    inc rsi
    mov ecx, 4
str_hex_loop:
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 10
    jb str_hex_next
    movzx eax, byte ptr [rsi]
    or al, 0x20
    sub eax, 0x61
    cmp eax, 6
    jae error
str_hex_next:
    inc rsi
    dec ecx
    jnz str_hex_loop
    jmp str_loop

str_esc_ok:
    inc rsi
    jmp str_loop

parse_true:
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x72
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x75
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x65
    jne error
    inc rsi
    ret

parse_false:
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x61
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x6C
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x73
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x65
    jne error
    inc rsi
    ret

parse_null:
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x75
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x6C
    jne error
    inc rsi
    cmp rsi, r13
    jae error
    cmp byte ptr [rsi], 0x6C
    jne error
    inc rsi
    ret

parse_number:
    cmp byte ptr [rsi], 0x2D
    jne 1f
    inc rsi
    cmp rsi, r13
    jae error
1:
    movzx eax, byte ptr [rsi]
    cmp al, 0x30
    je 2f
    sub eax, 0x31
    cmp eax, 9
    jae error
    inc rsi
3:
    cmp rsi, r13
    jae 4f
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 10
    jae 4f
    inc rsi
    jmp 3b
2:
    inc rsi
4:
    cmp rsi, r13
    jae 9f
    cmp byte ptr [rsi], 0x2E
    jne 5f
    inc rsi
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 10
    jae error
    inc rsi
6:
    cmp rsi, r13
    jae 9f
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 10
    jae 5f
    inc rsi
    jmp 6b
5:
    cmp rsi, r13
    jae 9f
    movzx eax, byte ptr [rsi]
    or al, 0x20
    cmp al, 0x65
    jne 9f
    inc rsi
    cmp rsi, r13
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, 0x2B
    je 7f
    cmp al, 0x2D
    jne 8f
7:
    inc rsi
    cmp rsi, r13
    jae error
8:
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 10
    jae error
    inc rsi
10:
    cmp rsi, r13
    jae 9f
    movzx eax, byte ptr [rsi]
    sub eax, 0x30
    cmp eax, 10
    jae 9f
    inc rsi
    jmp 10b
9:
    ret

error:
    mov rax, rsi
    sub rax, r12
    lea r8, [rip + outbuf + 64]
    mov rdi, r8
    dec rdi
    mov byte ptr [rdi], 0x0A
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    sub rdi, 9
    mov rax, qword ptr [rip + err_prefix]
    mov qword ptr [rdi], rax
    mov al, byte ptr [rip + err_prefix + 8]
    mov byte ptr [rdi + 8], al
    mov rdx, r8
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "

.bss
    .align 16
buf:
    .zero BUF_SIZE
outbuf:
    .zero 64
