.globl _start

.bss
.p2align 4
inbuf:
    .space 65600

.text
_start:
    lea rbx, [rip + inbuf]
    xor r12d, r12d

read_loop:
    mov edx, 65540
    sub rdx, r12
    jbe read_done
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_program

    cmp byte ptr [rbx + r12 - 1], 10
    je process_lines
    mov byte ptr [rbx + r12], 10
    inc r12

process_lines:
    xor r8d, r8d

line_loop:
    cmp r8, r12
    jae write_output
    mov r9, r8

find_nl:
    cmp byte ptr [rbx + r9], 10
    je found_nl
    inc r9
    jmp find_nl

found_nl:
    lea r10, [rbx + r8]
    lea r11, [rbx + r9]

rev_loop:
    dec r11
    cmp r10, r11
    jae next_line
    mov al, byte ptr [r10]
    mov cl, byte ptr [r11]
    mov byte ptr [r10], cl
    mov byte ptr [r11], al
    inc r10
    jmp rev_loop

next_line:
    lea r8, [r9 + 1]
    jmp line_loop

write_output:
    mov rsi, rbx
    mov r13, r12

write_loop:
    test r13, r13
    jz exit_program
    mov edi, 1
    mov rdx, r13
    mov eax, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub r13, rax
    jmp write_loop

exit_program:
    xor edi, edi
    mov eax, 60
    syscall
