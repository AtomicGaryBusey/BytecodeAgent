.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0                 // total bytes read
    movz x25, #2, lsl #16       // BUF_SIZE = 0x20000

1:
    sub x2, x25, x20            // remaining capacity
    cbz x2, 2f
    mov x0, #0                  // fd = 0 (stdin)
    add x1, x19, x20            // inbuf + total
    mov x8, #63                 // read
    svc #0
    cmp x0, #0
    b.le 2f                     // EOF or error
    add x20, x20, x0
    b 1b

2:
    cbz x20, 9f                 // empty input -> exit(0)

    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, x21                // out_ptr = outbuf
    mov x1, x19                 // cur = inbuf
    add x2, x19, x20            // end = inbuf + total

3:                              // line loop (cur < end)
    mov x3, x1                  // line_start = cur

4:                              // scan for 0x0a or end
    ldrb w4, [x1]
    cmp w4, #0x0a
    b.eq 5f
    add x1, x1, #1
    cmp x1, x2
    b.lo 4b

    // reached end without newline
    mov x5, x1                  // p = end
    b 6f

5:                              // found newline at x1
    mov x5, x1                  // p = line_end
    add x1, x1, #1              // next_cur = line_end + 1

6:                              // reverse [line_start, p) into outbuf
    cmp x5, x3
    b.ls 7f
    ldrb w4, [x5, #-1]!
    strb w4, [x22], #1
    b 6b

7:                              // append newline
    mov w4, #0x0a
    strb w4, [x22], #1
    cmp x1, x2
    b.lo 3b

    sub x23, x22, x21           // total bytes to write
    cbz x23, 9f

8:                              // write loop
    mov x0, #1                  // fd = 1 (stdout)
    mov x1, x21                 // buf
    mov x2, x23                 // count
    mov x8, #64                 // write
    svc #0
    cmp x0, #0
    b.le 9f
    add x21, x21, x0
    sub x23, x23, x0
    cbnz x23, 8b

9:
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

.bss
.balign 16
inbuf:
    .zero 0x20000
outbuf:
    .zero 0x20000
