.equ sys_read, 63
.equ sys_write, 64
.equ sys_exit, 93
.equ buf_size, 8192

.bss
.balign 16
buf:
    .space buf_size

.text
.balign 4
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov w22, #0                 // in_word (0 or 1)
    adrp x23, buf
    add x23, x23, :lo12:buf
    mov x24, #buf_size

read_loop:
    mov x0, #0                  // fd = 0 (stdin)
    mov x1, x23
    mov x2, x24
    mov x8, #sys_read
    svc #0
    cmp x0, #0
    b.le done_read

    add x21, x21, x0            // bytes += nread
    mov x1, x23
    add x2, x23, x0

byte_loop:
    ldrb w3, [x1], #1
    cmp w3, #0x0A
    cinc x19, x19, eq           // lines += (b == '\n')
    cmp w3, #0x20
    b.eq is_ws
    sub w4, w3, #0x09
    cmp w4, #4
    b.ls is_ws

    // Non-whitespace byte
    eor w4, w22, #1
    add x20, x20, x4            // words += (in_word == 0)
    mov w22, #1
    cmp x1, x2
    b.lo byte_loop
    b read_loop

is_ws:
    mov w22, #0
    cmp x1, x2
    b.lo byte_loop
    b read_loop

done_read:
    sub sp, sp, #128
    add x25, sp, #128
    mov x0, x25

    mov w4, #0x0A               // '\n'
    strb w4, [x0, #-1]!
    mov x1, x21                 // bytes
    bl fmt_u64

    mov w4, #0x20               // ' '
    strb w4, [x0, #-1]!
    mov x1, x20                 // words
    bl fmt_u64

    mov w4, #0x20               // ' '
    strb w4, [x0, #-1]!
    mov x1, x19                 // lines
    bl fmt_u64

    mov x26, x0                 // output start pointer
    sub x27, x25, x0            // output length

write_loop:
    mov x0, #1                  // fd = 1 (stdout)
    mov x1, x26
    mov x2, x27
    mov x8, #sys_write
    svc #0
    cmp x0, #0
    b.le exit_ok
    add x26, x26, x0
    sub x27, x27, x0
    cbnz x27, write_loop

exit_ok:
    mov x0, #0
    mov x8, #sys_exit
    svc #0

// Format unsigned 64-bit integer in x1 backwards from x0.
// Returns updated pointer in x0. Clobbers x1, x2, x3, x4.
fmt_u64:
    mov x2, #10
1:
    udiv x3, x1, x2
    msub x4, x3, x2, x1
    add w4, w4, #0x30           // '0'
    strb w4, [x0, #-1]!
    mov x1, x3
    cbnz x1, 1b
    ret
