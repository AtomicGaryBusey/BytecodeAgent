.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 8192
.equ OUT_SIZE, 128

.bss
.balign 16
inbuf:
    .zero BUF_SIZE
outbuf:
    .zero OUT_SIZE

.text
.globl _start
_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov w22, #0                 // in_word (0 = whitespace, 1 = word)
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf

read_loop:
    mov x0, #0                  // fd = 0 (stdin)
    mov x1, x23
    mov x2, #BUF_SIZE
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done

    add x21, x21, x0            // bytes += n
    mov x1, x23                 // buffer cursor
    mov x2, x0                  // remaining bytes in chunk

1:
    ldrb w3, [x1], #1
    cmp w3, #0x20               // space
    b.eq 3f
    sub w4, w3, #9              // 0x09..0x0D -> 0..4
    cmp w4, #4
    b.ls 2f

    // Non-whitespace byte
    cmp w22, #0
    cinc x20, x20, eq
    mov w22, #1
    b 4f

2:
    // 0x09..0x0D whitespace
    cmp w3, #0x0A               // newline
    cinc x19, x19, eq

3:
    mov w22, #0

4:
    subs x2, x2, #1
    b.ne 1b
    b read_loop

read_done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    add x24, x24, #OUT_SIZE
    mov x1, x24

    mov w2, #0x0A               // '\n'
    strb w2, [x1, #-1]!

    mov x0, x21                 // bytes
    bl fmt_u64

    mov w2, #0x20               // ' '
    strb w2, [x1, #-1]!

    mov x0, x20                 // words
    bl fmt_u64

    mov w2, #0x20               // ' '
    strb w2, [x1, #-1]!

    mov x0, x19                 // lines
    bl fmt_u64

    sub x2, x24, x1             // output byte count

write_loop:
    mov x0, #1                  // fd = 1 (stdout)
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le exit_prog
    add x1, x1, x0
    subs x2, x2, x0
    b.gt write_loop

exit_prog:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// Format unsigned 64-bit integer in x0 backwards starting before x1.
// Returns updated start pointer in x1. Clobbers x0, x2, x3, x4.
fmt_u64:
    mov x3, #10
1:
    udiv x2, x0, x3
    msub x4, x2, x3, x0
    add w4, w4, #0x30           // '0'
    strb w4, [x1, #-1]!
    mov x0, x2
    cbnz x0, 1b
    ret
