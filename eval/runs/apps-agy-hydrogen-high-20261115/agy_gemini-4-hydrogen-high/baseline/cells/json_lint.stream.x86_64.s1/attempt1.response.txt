.globl _start

.section .rodata
ok_msg:
    .ascii "ok\n"
    .align 8
err_prefix:
    .ascii "error at "

.bss
    .align 16
buf:
    .space 65544
cstack:
    .space 65544

.text
_start:
    lea r12, [rip + buf]
    xor ebx, ebx

read_loop:
    mov edx, 65540
    sub rdx, rbx
    jbe read_done
    lea rsi, [r12 + rbx]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov rsi, r12
    lea r13, [r12 + rbx]
    lea r14, [rip + cstack]
    mov r15, r14

expect_value:
    call skip_ws
expect_value_ready:
    mov al, byte ptr [rsi]
    cmp al, '{'
    je val_object
    cmp al, '['
    je val_array
    cmp al, 0x22
    je val_string
    cmp al, '-'
    je val_neg_num
    mov cl, al
    sub cl, '0'
    cmp cl, 9
    jbe val_num_digits
    cmp al, 't'
    je val_true
    cmp al, 'f'
    je val_false
    cmp al, 'n'
    je val_null
    jmp fail

val_object:
    inc rsi
    mov byte ptr [r15], 1
    inc r15
    call skip_ws
    cmp byte ptr [rsi], '}'
    jne expect_key_ready
    inc rsi
    dec r15
    jmp after_value

val_array:
    inc rsi
    mov byte ptr [r15], 0
    inc r15
    call skip_ws
    cmp byte ptr [rsi], ']'
    jne expect_value_ready
    inc rsi
    dec r15
    jmp after_value

val_string:
    call parse_string
    jmp after_value

val_neg_num:
    inc rsi
    mov al, byte ptr [rsi]
val_num_digits:
    cmp al, '0'
    je num_zero
    sub al, '1'
    cmp al, 8
    ja fail
    inc rsi
num_int_loop:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja num_frac
    inc rsi
    jmp num_int_loop

num_zero:
    inc rsi

num_frac:
    mov al, byte ptr [rsi]
    cmp al, '.'
    jne num_exp
    inc rsi
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja fail
    inc rsi
num_frac_loop:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja num_exp
    inc rsi
    jmp num_frac_loop

num_exp:
    mov al, byte ptr [rsi]
    or al, 0x20
    cmp al, 'e'
    jne after_value
    inc rsi
    mov al, byte ptr [rsi]
    cmp al, '+'
    je num_exp_sign
    cmp al, '-'
    jne num_exp_first_digit
num_exp_sign:
    inc rsi
    mov al, byte ptr [rsi]
num_exp_first_digit:
    sub al, '0'
    cmp al, 9
    ja fail
    inc rsi
num_exp_loop:
    mov al, byte ptr [rsi]
    sub al, '0'
    cmp al, 9
    ja after_value
    inc rsi
    jmp num_exp_loop

val_true:
    inc rsi
    cmp byte ptr [rsi], 'r'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'u'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'e'
    jne fail
    inc rsi
    jmp after_value

val_false:
    inc rsi
    cmp byte ptr [rsi], 'a'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'l'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 's'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'e'
    jne fail
    inc rsi
    jmp after_value

val_null:
    inc rsi
    cmp byte ptr [rsi], 'u'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'l'
    jne fail
    inc rsi
    cmp byte ptr [rsi], 'l'
    jne fail
    inc rsi
    jmp after_value

expect_key:
    call skip_ws
expect_key_ready:
    cmp byte ptr [rsi], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp byte ptr [rsi], ':'
    jne fail
    inc rsi
    jmp expect_value

after_value:
    call skip_ws
    cmp r15, r14
    je check_eof
    cmp byte ptr [r15 - 1], 0
    je after_array
    mov al, byte ptr [rsi]
    cmp al, ','
    je obj_next
    cmp al, '}'
    jne fail
    inc rsi
    dec r15
    jmp after_value

obj_next:
    inc rsi
    jmp expect_key

after_array:
    mov al, byte ptr [rsi]
    cmp al, ','
    je arr_next
    cmp al, ']'
    jne fail
    inc rsi
    dec r15
    jmp after_value

arr_next:
    inc rsi
    jmp expect_value

check_eof:
    cmp rsi, r13
    jne fail
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je skip_ws_adv
    cmp al, 0x0A
    je skip_ws_adv
    cmp al, 0x09
    je skip_ws_adv
    cmp al, 0x0D
    je skip_ws_adv
    ret
skip_ws_adv:
    inc rsi
    jmp skip_ws

parse_string:
    inc rsi
str_loop:
    mov al, byte ptr [rsi]
    cmp al, 0x22
    je str_done
    cmp al, 0x5C
    je str_escape
    cmp al, 0x20
    jb fail
    inc rsi
    jmp str_loop

str_done:
    inc rsi
    ret

str_escape:
    inc rsi
    mov al, byte ptr [rsi]
    cmp al, 'u'
    je str_unicode
    cmp al, 0x22
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, 0x2F
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    jne fail
str_esc_ok:
    inc rsi
    jmp str_loop

str_unicode:
    inc rsi
    mov ecx, 4
str_hex_loop:
    mov al, byte ptr [rsi]
    cmp al, '0'
    jb fail
    cmp al, '9'
    jbe str_hex_ok
    or al, 0x20
    cmp al, 'a'
    jb fail
    cmp al, 'f'
    ja fail
str_hex_ok:
    inc rsi
    dec ecx
    jnz str_hex_loop
    jmp str_loop

fail:
    mov rax, rsi
    sub rax, r12
    sub rsp, 32
    lea rdi, [rsp + 31]
    mov byte ptr [rdi], 0x0A
    mov ecx, 10
fail_div_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz fail_div_loop
    sub rdi, 9
    mov rax, qword ptr [rip + err_prefix]
    mov qword ptr [rdi], rax
    mov byte ptr [rdi + 8], ' '
    lea rdx, [rsp + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
