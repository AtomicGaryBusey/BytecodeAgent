.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 65536
.equ LINES_SIZE, 524288

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 16
lines:
    .space LINES_SIZE

.text
.globl _start

// cmp_lines(x0 = ptrA, x1 = lenA, x2 = ptrB, x3 = lenB) -> x0 (<0, 0, >0)
cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, #0
1:
    cmp x5, x4
    b.hs 2f
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.ne 3f
    add x5, x5, #1
    b 1b
2:
    sub x0, x1, x3
    ret
3:
    sub x0, x6, x7
    ret

// sift_down(x0 = lines, x1 = start, x2 = heap_size)
sift_down:
    stp x29, x30, [sp, #-16]!
    mov x8, x0
    mov x9, x1
    mov x10, x2
    lsl x14, x9, #4
    add x14, x8, x14
    ldp x11, x12, [x14]
1:
    lsl x13, x9, #1
    add x13, x13, #1
    cmp x13, x10
    b.hs 4f

    add x14, x13, #1
    cmp x14, x10
    b.hs 2f

    lsl x15, x13, #4
    add x15, x8, x15
    ldp x0, x1, [x15]
    lsl x15, x14, #4
    add x15, x8, x15
    ldp x2, x3, [x15]
    bl cmp_lines
    cmp x0, #0
    b.ge 2f
    add x13, x13, #1
2:
    lsl x15, x13, #4
    add x15, x8, x15
    ldp x2, x3, [x15]
    mov x0, x11
    mov x1, x12
    bl cmp_lines
    cmp x0, #0
    b.ge 4f

    lsl x15, x13, #4
    add x15, x8, x15
    ldp x16, x17, [x15]
    lsl x14, x9, #4
    add x14, x8, x14
    stp x16, x17, [x14]

    mov x9, x13
    b 1b
4:
    lsl x14, x9, #4
    add x14, x8, x14
    stp x11, x12, [x14]
    ldp x29, x30, [sp], #16
    ret

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #INBUF_SIZE

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, exit_zero

    adrp x21, lines
    add x21, x21, :lo12:lines
    mov x22, #0
    mov x23, x19
    add x24, x19, x20
    mov x25, x19

parse_loop:
    cmp x23, x24
    b.hs parse_end
    ldrb w0, [x23]
    cmp w0, #10
    b.ne parse_next
    sub x1, x23, x25
    lsl x2, x22, #4
    add x2, x21, x2
    stp x25, x1, [x2]
    add x22, x22, #1
    add x25, x23, #1
parse_next:
    add x23, x23, #1
    b parse_loop

parse_end:
    cmp x25, x24
    b.hs parse_done
    sub x1, x24, x25
    lsl x2, x22, #4
    add x2, x21, x2
    stp x25, x1, [x2]
    add x22, x22, #1

parse_done:
    cmp x22, #1
    b.ls sort_done

    lsr x23, x22, #1
heapify_loop:
    cbz x23, heapify_done
    sub x23, x23, #1
    mov x0, x21
    mov x1, x23
    mov x2, x22
    bl sift_down
    b heapify_loop

heapify_done:
    mov x23, x22
sort_loop:
    cmp x23, #1
    b.ls sort_done
    sub x23, x23, #1
    ldp x0, x1, [x21]
    lsl x2, x23, #4
    add x2, x21, x2
    ldp x3, x4, [x2]
    stp x3, x4, [x21]
    stp x0, x1, [x2]
    mov x0, x21
    mov x1, #0
    mov x2, x23
    bl sift_down
    b sort_loop

sort_done:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    mov x25, #0

build_out_loop:
    cmp x25, x22
    b.hs build_out_done
    lsl x0, x25, #4
    add x0, x21, x0
    ldp x1, x2, [x0]
1:
    cbz x2, 2f
    ldrb w3, [x1], #1
    strb w3, [x24], #1
    sub x2, x2, #1
    b 1b
2:
    mov w3, #10
    strb w3, [x24], #1
    add x25, x25, #1
    b build_out_loop

build_out_done:
    sub x24, x24, x23
    mov x25, #0

write_loop:
    cmp x25, x24
    b.hs exit_zero
    mov x0, #1
    add x1, x23, x25
    sub x2, x24, x25
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le exit_zero
    add x25, x25, x0
    b write_loop

exit_zero:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
