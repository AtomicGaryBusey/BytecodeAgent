.equ INBUF_SIZE, 4096
.equ OUTBUF_FLUSH, 4096
.equ OUTBUF_SIZE, 8192

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero INBUF_SIZE
outbuf:
    .zero OUTBUF_SIZE

.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
    lea rbp, [rip + outbuf]
    xor r12d, r12d
    xor r13d, r13d
    mov r14, rbp
    lea r15, [rip + hex_digits]

main_loop:
    cmp r13, 16
    jae process_rows

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    mov edx, INBUF_SIZE
    sub rdx, r13
    syscall
    cmp rax, -4
    je read_more
    test rax, rax
    jle eof_reached
    add r13, rax
    cmp r13, 16
    jb read_more

process_rows:
    xor r8d, r8d
row_loop:
    mov rax, r13
    sub rax, r8
    cmp rax, 16
    jb shift_rem

    lea rsi, [rbx + r8]
    mov r10d, 16
    call format_row
    add r12d, 16
    add r8, 16

    mov rax, r14
    sub rax, rbp
    cmp rax, OUTBUF_FLUSH
    jb row_loop
    call flush_out
    jmp row_loop

shift_rem:
    lea rsi, [rbx + r8]
    sub r13, r8
    jz main_loop
    xor ecx, ecx
shift_loop:
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [rbx + rcx], al
    inc rcx
    cmp rcx, r13
    jb shift_loop
    jmp main_loop

eof_reached:
    test r13, r13
    jz print_final_offset
    mov rsi, rbx
    mov r10, r13
    call format_row
    add r12d, r13d

print_final_offset:
    mov rdi, r14
    mov eax, r12d
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov r14, rdi

    call flush_out

    mov eax, 60
    xor edi, edi
    syscall

format_row:
    mov rdi, r14
    mov eax, r12d
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b

    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor ecx, ecx
2:
    cmp rcx, r10
    jae 3f
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 0xf
    mov al, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    jmp 4f
3:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
4:
    add rdi, 3
    cmp rcx, 7
    jne 5f
    mov byte ptr [rdi], 0x20
    inc rdi
5:
    inc rcx
    cmp rcx, 16
    jb 2b

    mov word ptr [rdi], 0x7c20
    add rdi, 2

    xor ecx, ecx
6:
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    sub edx, 0x20
    cmp edx, 0x5e
    jbe 7f
    mov al, 0x2e
7:
    mov byte ptr [rdi], al
    inc rdi
    inc rcx
    cmp rcx, r10
    jb 6b

    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    mov r14, rdi
    ret

flush_out:
    mov rdx, r14
    sub rdx, rbp
    jz 3f
    mov rsi, rbp
1:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, -4
    je 1b
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    mov r14, rbp
3:
    ret
