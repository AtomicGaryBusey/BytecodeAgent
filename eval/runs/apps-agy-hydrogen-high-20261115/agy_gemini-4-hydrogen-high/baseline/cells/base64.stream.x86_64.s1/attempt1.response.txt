.equ INBUF_SIZE, 57000
.equ OUTBUF_SIZE, 77000

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space INBUF_SIZE + 4
.balign 16
outbuf:
    .space OUTBUF_SIZE

.text
.globl _start
_start:
    lea     r12, [rip + b64_table]
    lea     r13, [rip + inbuf]
    lea     r14, [rip + outbuf]
    xor     ebx, ebx
    xor     r15d, r15d

.Lread_loop:
    cmp     rbx, INBUF_SIZE
    jae     .Lread_done
    xor     eax, eax
    xor     edi, edi
    lea     rsi, [r13 + rbx]
    mov     edx, INBUF_SIZE
    sub     rdx, rbx
    syscall
    test    rax, rax
    jle     .Lgot_eof
    add     rbx, rax
    jmp     .Lread_loop

.Lgot_eof:
    mov     r15d, 1

.Lread_done:
    test    rbx, rbx
    jz      .Lexit

    mov     rsi, r13
    mov     rdi, r14
    mov     rcx, rbx

.Lchunk57_loop:
    cmp     rcx, 57
    jb      .Lchunk57_done
    mov     ebp, 19

.Lgroup3_in_57:
    mov     eax, dword ptr [rsi]
    bswap   eax

    mov     edx, eax
    shr     edx, 26
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi], dl

    mov     edx, eax
    shr     edx, 20
    and     edx, 63
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi + 1], dl

    mov     edx, eax
    shr     edx, 14
    and     edx, 63
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi + 2], dl

    shr     eax, 8
    and     eax, 63
    movzx   eax, byte ptr [r12 + rax]
    mov     byte ptr [rdi + 3], al

    add     rsi, 3
    add     rdi, 4
    dec     rbp
    jnz     .Lgroup3_in_57

    mov     byte ptr [rdi], 10
    inc     rdi
    sub     rcx, 57
    jmp     .Lchunk57_loop

.Lchunk57_done:
    test    rcx, rcx
    jz      .Lflush_out

.Lrem_3_loop:
    cmp     rcx, 3
    jb      .Lrem_tail
    mov     eax, dword ptr [rsi]
    bswap   eax

    mov     edx, eax
    shr     edx, 26
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi], dl

    mov     edx, eax
    shr     edx, 20
    and     edx, 63
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi + 1], dl

    mov     edx, eax
    shr     edx, 14
    and     edx, 63
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi + 2], dl

    shr     eax, 8
    and     eax, 63
    movzx   eax, byte ptr [r12 + rax]
    mov     byte ptr [rdi + 3], al

    add     rsi, 3
    add     rdi, 4
    sub     rcx, 3
    jmp     .Lrem_3_loop

.Lrem_tail:
    test    rcx, rcx
    jz      .Lappend_nl
    cmp     rcx, 1
    je      .Ltail_1

    movzx   eax, byte ptr [rsi]
    shl     eax, 16
    movzx   edx, byte ptr [rsi + 1]
    shl     edx, 8
    or      eax, edx

    mov     edx, eax
    shr     edx, 18
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi], dl

    mov     edx, eax
    shr     edx, 12
    and     edx, 63
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi + 1], dl

    shr     eax, 6
    and     eax, 63
    movzx   eax, byte ptr [r12 + rax]
    mov     byte ptr [rdi + 2], al

    mov     byte ptr [rdi + 3], 61
    add     rdi, 4
    jmp     .Lappend_nl

.Ltail_1:
    movzx   eax, byte ptr [rsi]
    mov     edx, eax
    shr     edx, 2
    movzx   edx, byte ptr [r12 + rdx]
    mov     byte ptr [rdi], dl

    and     eax, 3
    shl     eax, 4
    movzx   eax, byte ptr [r12 + rax]
    mov     byte ptr [rdi + 1], al

    mov     byte ptr [rdi + 2], 61
    mov     byte ptr [rdi + 3], 61
    add     rdi, 4

.Lappend_nl:
    mov     byte ptr [rdi], 10
    inc     rdi

.Lflush_out:
    mov     rsi, r14
    mov     rdx, rdi
    sub     rdx, r14

.Lwrite_loop:
    test    rdx, rdx
    jz      .Lwrite_done
    mov     eax, 1
    mov     edi, 1
    syscall
    test    rax, rax
    jle     .Lexit
    add     rsi, rax
    sub     rdx, rax
    jmp     .Lwrite_loop

.Lwrite_done:
    test    r15, r15
    jnz     .Lexit
    xor     ebx, ebx
    jmp     .Lread_loop

.Lexit:
    mov     eax, 60
    xor     edi, edi
    syscall
