.globl _start

_start:
    lea rsi, [rip + inbuf]
    lea rbx, [rip + inbuf + 65792]
1:
    mov rdx, rbx
    sub rdx, rsi
    jz 2f
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    jmp 1b
2:
    mov r14, rsi

    lea rsi, [rip + inbuf]
    mov r13, r14
    mov r12, r14
3:
    cmp rsi, r14
    jae 4f
    cmp byte ptr [rsi], 0x21
    je 5f
    inc rsi
    jmp 3b
5:
    mov r12, rsi
    lea r13, [rsi + 1]
4:
    lea rsi, [rip + inbuf]
    lea r8, [rip + ops]
    lea r9, [rip + jumps]
    lea r10, [rip + char_to_op]
    lea r11, [rip + bstack]
    xor ebx, ebx
    xor ecx, ecx

compile_loop:
    cmp rsi, r12
    jae compile_done
    movzx eax, byte ptr [rsi]
    inc rsi
    movzx eax, byte ptr [r10 + rax]
    cmp al, 7
    ja compile_loop
    cmp al, 6
    jb compile_simple
    je compile_lbrack

compile_rbrack:
    test rcx, rcx
    jz unbalanced
    dec rcx
    mov edx, dword ptr [r11 + rcx*4]
    lea edi, [rbx + 1]
    mov dword ptr [r9 + rdx*4], edi
    lea edi, [rdx + 1]
    mov dword ptr [r9 + rbx*4], edi
    mov byte ptr [r8 + rbx], al
    inc rbx
    jmp compile_loop

compile_lbrack:
    mov byte ptr [r8 + rbx], al
    mov dword ptr [r11 + rcx*4], ebx
    inc rcx
    inc rbx
    jmp compile_loop

compile_simple:
    mov byte ptr [r8 + rbx], al
    inc rbx
    jmp compile_loop

compile_done:
    test rcx, rcx
    jnz unbalanced
    mov byte ptr [r8 + rbx], 8

    lea r12, [rip + tape]
    lea r15, [rip + outbuf]
    lea rbp, [rip + jump_table]
    xor ebx, ebx
    movzx eax, byte ptr [r8]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_gt:
    inc r12
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_lt:
    dec r12
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_plus:
    inc byte ptr [r12]
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_minus:
    dec byte ptr [r12]
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_dot:
    mov al, byte ptr [r12]
    mov byte ptr [r15], al
    inc r15
    lea rdx, [rip + outbuf + 65536]
    cmp r15, rdx
    je flush_outbuf
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

flush_outbuf:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
1:
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    lea r15, [rip + outbuf]
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_comma:
    xor eax, eax
    cmp r13, r14
    jae 1f
    mov al, byte ptr [r13]
    inc r13
1:
    mov byte ptr [r12], al
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_lbrack:
    cmp byte ptr [r12], 0
    jne 1f
    mov ebx, dword ptr [r9 + rbx*4 - 4]
1:
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_rbrack:
    cmp byte ptr [r12], 0
    je 1f
    mov ebx, dword ptr [r9 + rbx*4 - 4]
1:
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    jmp qword ptr [rbp + rax*8]

op_halt:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
    jz exit_ok
1:
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jnz 1b
exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

unbalanced:
    mov edi, 1
    mov eax, 60
    syscall
