.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.equ STDIN_FD, 0
.equ STDOUT_FD, 1

.equ INBUF_SIZE, 4096
.equ OUTBUF_SIZE, 4096

.equ CH_LF, 0x0a
.equ CH_SPACE, 0x20
.equ CH_DOT, 0x2e
.equ CH_BAR, 0x7c

.globl _start

.text
.balign 4

_start:
    mov x19, #0
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    mov x20, x22
    mov x21, x22
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    add x25, x23, #OUTBUF_SIZE
    adrp x26, hex_digits
    add x26, x26, :lo12:hex_digits
    adrp x27, rowbuf
    add x27, x27, :lo12:rowbuf
    mov x28, #0

main_loop:
    bl read_row
    cbz x0, done
    mov x9, x0

    // Ensure at least 80 bytes free in outbuf
    sub x1, x25, x24
    cmp x1, #80
    b.ge 1f
    bl flush_out
1:
    // Offset as 8 lowercase hex digits followed by two spaces
    mov w0, w19
    bl write_hex32
    mov w3, #CH_SPACE
    strb w3, [x24], #1
    strb w3, [x24], #1

    // 16 hex byte slots
    mov x5, #0
hex_loop:
    cmp x5, x9
    b.ge hex_pad
    ldrb w3, [x27, x5]
    lsr w4, w3, #4
    ldrb w4, [x26, x4]
    strb w4, [x24], #1
    and w4, w3, #0xf
    ldrb w4, [x26, x4]
    strb w4, [x24], #1
    mov w4, #CH_SPACE
    strb w4, [x24], #1
    b hex_after_slot
hex_pad:
    mov w4, #CH_SPACE
    strb w4, [x24], #1
    strb w4, [x24], #1
    strb w4, [x24], #1
hex_after_slot:
    cmp x5, #7
    b.ne 2f
    mov w4, #CH_SPACE
    strb w4, [x24], #1
2:
    add x5, x5, #1
    cmp x5, #16
    b.lt hex_loop

    // Space, pipe, ASCII column, pipe, newline
    mov w4, #CH_SPACE
    strb w4, [x24], #1
    mov w4, #CH_BAR
    strb w4, [x24], #1

    mov x5, #0
ascii_loop:
    ldrb w3, [x27, x5]
    cmp w3, #0x20
    b.lo 3f
    cmp w3, #0x7e
    b.ls 4f
3:
    mov w3, #CH_DOT
4:
    strb w3, [x24], #1
    add x5, x5, #1
    cmp x5, x9
    b.lt ascii_loop

    mov w4, #CH_BAR
    strb w4, [x24], #1
    mov w4, #CH_LF
    strb w4, [x24], #1

    add x19, x19, x9
    b main_loop

done:
    // Ensure at least 16 bytes free in outbuf
    sub x1, x25, x24
    cmp x1, #16
    b.ge 5f
    bl flush_out
5:
    mov w0, w19
    bl write_hex32
    mov w4, #CH_LF
    strb w4, [x24], #1

    bl flush_out

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// Read up to 16 bytes into rowbuf (x27), returns count in x0
read_row:
    mov x4, #0
1:
    cmp x4, #16
    b.ge 3f
    cmp x20, x21
    b.lo 2f
    cbnz x28, 3f
    mov x0, #STDIN_FD
    mov x1, x22
    mov x2, #INBUF_SIZE
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.gt 4f
    mov x28, #1
    b 3f
4:
    mov x20, x22
    add x21, x22, x0
2:
    ldrb w3, [x20], #1
    strb w3, [x27, x4]
    add x4, x4, #1
    b 1b
3:
    mov x0, x4
    ret

// Format w0 as 8 lowercase hex digits at [x24], advancing x24
write_hex32:
    mov w1, #28
1:
    lsr w2, w0, w1
    and w2, w2, #0xf
    ldrb w3, [x26, x2]
    strb w3, [x24], #1
    subs w1, w1, #4
    b.ge 1b
    ret

// Write buffered bytes in outbuf to stdout and reset x24
flush_out:
    subs x2, x24, x23
    b.le 2f
    mov x1, x23
1:
    mov x0, #STDOUT_FD
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 2f
    add x1, x1, x0
    subs x2, x2, x0
    b.gt 1b
2:
    mov x24, x23
    ret

.section .rodata
.balign 16
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space INBUF_SIZE
.balign 16
outbuf:
    .space OUTBUF_SIZE
.balign 16
rowbuf:
    .space 16
