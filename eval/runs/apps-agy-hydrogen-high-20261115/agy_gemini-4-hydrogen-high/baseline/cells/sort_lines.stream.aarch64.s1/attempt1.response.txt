.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ IN_BUF_SIZE, 65536
.equ OUT_BUF_SIZE, 65536
.equ MAX_LINES, 4096
.equ LINES_BYTES, 65536

.bss
.balign 16
in_buf:
    .zero IN_BUF_SIZE
.balign 16
out_buf:
    .zero OUT_BUF_SIZE
.balign 16
lines:
    .zero LINES_BYTES
.balign 16
tmp_lines:
    .zero LINES_BYTES

.text
.globl _start
.balign 4

_start:
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0
    mov x21, #IN_BUF_SIZE

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, do_exit

    adrp x21, lines
    add x21, x21, :lo12:lines
    mov x22, #0
    mov x23, x19
    add x24, x19, x20
    mov x25, x19
    mov x26, #MAX_LINES

parse_loop:
    cmp x23, x24
    b.hs parse_end
    ldrb w0, [x23]
    cmp w0, #10
    b.ne parse_next
    cmp x22, x26
    b.hs parse_done
    sub x1, x23, x25
    lsl x2, x22, #4
    add x2, x21, x2
    stp x25, x1, [x2]
    add x22, x22, #1
    add x23, x23, #1
    mov x25, x23
    b parse_loop

parse_next:
    add x23, x23, #1
    b parse_loop

parse_end:
    cmp x25, x24
    b.hs parse_done
    cmp x22, x26
    b.hs parse_done
    sub x1, x24, x25
    lsl x2, x22, #4
    add x2, x21, x2
    stp x25, x1, [x2]
    add x22, x22, #1

parse_done:
    cmp x22, #1
    b.ls skip_sort
    mov x0, x21
    adrp x1, tmp_lines
    add x1, x1, :lo12:tmp_lines
    mov x2, x22
    bl mergesort

skip_sort:
    adrp x23, out_buf
    add x23, x23, :lo12:out_buf
    mov x24, #0
    mov x25, #OUT_BUF_SIZE
    mov x26, #0

out_loop:
    cmp x26, x22
    b.hs out_flush_end
    lsl x0, x26, #4
    add x0, x21, x0
    ldp x27, x28, [x0]
    add x0, x28, #1
    add x1, x24, x0
    cmp x1, x25
    b.ls out_copy_line
    mov x0, x23
    mov x1, x24
    bl write_all
    mov x24, #0
    add x0, x28, #1
    cmp x0, x25
    b.ls out_copy_line
    mov x0, x27
    mov x1, x28
    bl write_all
    mov w0, #10
    strb w0, [x23]
    mov x0, x23
    mov x1, #1
    bl write_all
    b out_next_line

out_copy_line:
    add x0, x23, x24
    mov x1, x27
    mov x2, x28
    cbz x2, out_append_nl

out_byte_loop:
    ldrb w3, [x1], #1
    strb w3, [x0], #1
    sub x2, x2, #1
    cbnz x2, out_byte_loop

out_append_nl:
    mov w3, #10
    strb w3, [x0]
    add x24, x24, x28
    add x24, x24, #1

out_next_line:
    add x26, x26, #1
    b out_loop

out_flush_end:
    mov x0, x23
    mov x1, x24
    bl write_all

do_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

// cmp_le(x0=a_ptr, x1=a_len, x2=b_ptr, x3=b_len) -> x0 = 1 if a <= b else 0
cmp_le:
    cmp x1, x3
    csel x4, x1, x3, lo
    cbz x4, 2f
    mov x5, #0
1:
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.ne 3f
    add x5, x5, #1
    cmp x5, x4
    b.lo 1b
2:
    cmp x1, x3
    cset x0, ls
    ret
3:
    cset x0, lo
    ret

// mergesort(x0=arr, x1=tmp, x2=n)
mergesort:
    cmp x2, #1
    b.ls mergesort_done
    stp x29, x30, [sp, #-96]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    stp x27, x28, [sp, #80]

    mov x19, x0
    mov x20, x1
    mov x21, x2
    lsr x22, x21, #1

    mov x0, x19
    mov x1, x20
    mov x2, x22
    bl mergesort

    lsl x3, x22, #4
    add x0, x19, x3
    add x1, x20, x3
    sub x2, x21, x22
    bl mergesort

    mov x23, x19
    lsl x3, x22, #4
    add x24, x19, x3
    mov x25, x24
    lsl x3, x21, #4
    add x26, x19, x3
    mov x27, x20

merge_loop:
    cmp x23, x24
    b.hs merge_right_rem
    cmp x25, x26
    b.hs merge_left_rem
    ldp x0, x1, [x23]
    ldp x2, x3, [x25]
    bl cmp_le
    cbz x0, merge_take_right
    ldp x0, x1, [x23], #16
    stp x0, x1, [x27], #16
    b merge_loop

merge_take_right:
    ldp x0, x1, [x25], #16
    stp x0, x1, [x27], #16
    b merge_loop

merge_left_rem:
    cmp x23, x24
    b.hs merge_copy_back
    ldp x0, x1, [x23], #16
    stp x0, x1, [x27], #16
    b merge_left_rem

merge_right_rem:
    cmp x25, x26
    b.hs merge_copy_back
    ldp x0, x1, [x25], #16
    stp x0, x1, [x27], #16
    b merge_right_rem

merge_copy_back:
    mov x0, x20
    mov x1, x19

merge_copy_loop:
    cmp x0, x27
    b.hs merge_ret
    ldp x2, x3, [x0], #16
    stp x2, x3, [x1], #16
    b merge_copy_loop

merge_ret:
    ldp x27, x28, [sp, #80]
    ldp x25, x26, [sp, #64]
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #96

mergesort_done:
    ret

// write_all(x0=buf, x1=len)
write_all:
    cbz x1, 2f
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    mov x19, x0
    mov x20, x1
1:
    mov x0, #1
    mov x1, x19
    mov x2, x20
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 3f
    add x19, x19, x0
    sub x20, x20, x0
    cbnz x20, 1b
3:
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
2:
    ret
