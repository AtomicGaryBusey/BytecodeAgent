.text
.globl _start

_start:
    lea r12, [rip + stdin_buf]
    mov r13d, 65536

read_loop:
    test r13d, r13d
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov edx, r13d
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    sub r13d, eax
    jmp read_loop

read_done:
    lea rsi, [rip + stdin_buf]
    lea rdi, [rip + code_buf]
    mov r10, rdi
    lea r8, [rip + bstack]
    mov r9, r8

compile_loop:
    cmp rsi, r12
    jae compile_done
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x21
    je compile_done
    cmp al, 0x3E
    je emit_right
    cmp al, 0x3C
    je emit_left
    cmp al, 0x2B
    je emit_inc
    cmp al, 0x2D
    je emit_dec
    cmp al, 0x2E
    je emit_out
    cmp al, 0x2C
    je emit_in
    cmp al, 0x5B
    je emit_open
    cmp al, 0x5D
    je emit_close
    jmp compile_loop

emit_right:
    mov dword ptr [rdi], 1
    add rdi, 4
    jmp compile_loop

emit_left:
    mov dword ptr [rdi], 2
    add rdi, 4
    jmp compile_loop

emit_inc:
    mov dword ptr [rdi], 3
    add rdi, 4
    jmp compile_loop

emit_dec:
    mov dword ptr [rdi], 4
    add rdi, 4
    jmp compile_loop

emit_out:
    mov dword ptr [rdi], 5
    add rdi, 4
    jmp compile_loop

emit_in:
    mov dword ptr [rdi], 6
    add rdi, 4
    jmp compile_loop

emit_open:
    mov rax, rdi
    sub rax, r10
    mov dword ptr [r9], eax
    add r9, 4
    add rdi, 4
    jmp compile_loop

emit_close:
    cmp r9, r8
    je err_exit
    sub r9, 4
    mov eax, dword ptr [r9]
    mov rcx, rdi
    sub rcx, r10
    lea edx, [rcx + 4]
    shl edx, 8
    or edx, 7
    mov dword ptr [r10 + rax], edx
    lea edx, [rax + 4]
    shl edx, 8
    or edx, 8
    mov dword ptr [rdi], edx
    add rdi, 4
    jmp compile_loop

compile_done:
    cmp r9, r8
    jne err_exit
    mov dword ptr [rdi], 0

    mov r13, rsi
    mov r14, r12
    mov r12, r10
    lea rbp, [rip + tape]
    lea r15, [rip + out_buf]
    lea r8, [r15 + 65536]
    lea rbx, [rip + jump_table]

dispatch:
    mov eax, dword ptr [r12]
    movzx ecx, al
    jmp qword ptr [rbx + rcx*8]

op_right:
    inc rbp
    add r12, 4
    jmp dispatch

op_left:
    dec rbp
    add r12, 4
    jmp dispatch

op_inc:
    inc byte ptr [rbp]
    add r12, 4
    jmp dispatch

op_dec:
    dec byte ptr [rbp]
    add r12, 4
    jmp dispatch

op_out:
    mov al, byte ptr [rbp]
    mov byte ptr [r15], al
    inc r15
    cmp r15, r8
    je flush_out
    add r12, 4
    jmp dispatch

flush_out:
    lea rsi, [rip + out_buf]
    mov rdx, 65536
1:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    lea r15, [rip + out_buf]
    add r12, 4
    jmp dispatch

op_in:
    cmp r13, r14
    jae 1f
    mov al, byte ptr [r13]
    inc r13
    mov byte ptr [rbp], al
    add r12, 4
    jmp dispatch
1:
    mov byte ptr [rbp], 0
    add r12, 4
    jmp dispatch

op_open:
    cmp byte ptr [rbp], 0
    je 1f
    add r12, 4
    jmp dispatch
1:
    shr eax, 8
    lea r12, [r10 + rax]
    jmp dispatch

op_close:
    cmp byte ptr [rbp], 0
    jne 1f
    add r12, 4
    jmp dispatch
1:
    shr eax, 8
    lea r12, [r10 + rax]
    jmp dispatch

op_halt:
    lea rsi, [rip + out_buf]
    mov rdx, r15
    sub rdx, rsi
    jz 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle 2f
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    mov eax, 60
    xor edi, edi
    syscall

err_exit:
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
.p2align 3
jump_table:
    .quad op_halt
    .quad op_right
    .quad op_left
    .quad op_inc
    .quad op_dec
    .quad op_out
    .quad op_in
    .quad op_open
    .quad op_close

.bss
.p2align 4
stdin_buf:
    .zero 65536
out_buf:
    .zero 65536
tape:
    .zero 30000
.p2align 2
code_buf:
    .zero 262148
bstack:
    .zero 262144
