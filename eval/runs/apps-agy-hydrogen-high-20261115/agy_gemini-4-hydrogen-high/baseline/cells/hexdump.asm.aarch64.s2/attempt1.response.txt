.text
.globl _start
.balign 4

_start:
    mov x19, #0
    mov x20, #0
    adrp x21, rowbuf
    add x21, x21, :lo12:rowbuf
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, #0
    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

read_loop:
    mov x0, #0
    mov x1, x22
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le eof_reached

    mov x26, #0
    mov x27, x0
byte_loop:
    ldrb w9, [x22, x26]
    strb w9, [x21, x20]
    add x20, x20, #1
    cmp x20, #16
    b.lo next_byte

    mov w0, w19
    mov x1, x21
    mov x2, #16
    bl emit_row
    add x19, x19, #16
    mov x20, #0

next_byte:
    add x26, x26, #1
    cmp x26, x27
    b.lo byte_loop
    b read_loop

eof_reached:
    cbz x20, emit_total
    mov w0, w19
    mov x1, x21
    mov x2, x20
    bl emit_row
    add x19, x19, x20

emit_total:
    mov w0, w19
    bl write_hex32
    add x9, x23, x24
    mov w10, #0x0a
    strb w10, [x9]
    add x24, x24, #1
    bl flush_outbuf

    mov x0, #0
    mov x8, #93
    svc #0

// Write 8 lowercase hex digits of w0 into outbuf[outpos..outpos+7]
write_hex32:
    add x9, x23, x24
    mov w10, #8
1:
    lsr w11, w0, #28
    ldrb w12, [x25, x11]
    strb w12, [x9], #1
    lsl w0, w0, #4
    subs w10, w10, #1
    b.ne 1b
    add x24, x24, #8
    ret

// Emit one row: w0 = offset, x1 = rowbuf, x2 = n (1..16)
emit_row:
    stp x19, x30, [sp, #-16]!
    stp x20, x21, [sp, #-16]!
    mov x19, x1
    mov x20, x2

    bl write_hex32

    add x9, x23, x24
    mov w10, #0x20
    strb w10, [x9], #1
    strb w10, [x9], #1

    mov x11, #0
1:
    cmp x11, x20
    b.hs 2f
    ldrb w12, [x19, x11]
    lsr w13, w12, #4
    and w12, w12, #0xf
    ldrb w14, [x25, x13]
    strb w14, [x9], #1
    ldrb w14, [x25, x12]
    strb w14, [x9], #1
    strb w10, [x9], #1
    b 3f
2:
    strb w10, [x9], #1
    strb w10, [x9], #1
    strb w10, [x9], #1
3:
    cmp x11, #7
    b.ne 4f
    strb w10, [x9], #1
4:
    add x11, x11, #1
    cmp x11, #16
    b.lo 1b

    strb w10, [x9], #1
    mov w12, #0x7c
    strb w12, [x9], #1

    mov x11, #0
    mov w13, #0x2e
5:
    ldrb w14, [x19, x11]
    sub w15, w14, #0x20
    cmp w15, #0x5e
    csel w14, w14, w13, ls
    strb w14, [x9], #1
    add x11, x11, #1
    cmp x11, x20
    b.lo 5b

    strb w12, [x9], #1
    mov w14, #0x0a
    strb w14, [x9], #1

    sub x24, x9, x23
    cmp x24, #4096
    b.lo 6f
    bl flush_outbuf
6:
    ldp x20, x21, [sp], #16
    ldp x19, x30, [sp], #16
    ret

// Flush outbuf[0..outpos-1] to stdout and reset outpos to 0
flush_outbuf:
    mov x9, x23
    mov x10, x24
1:
    cbz x10, 2f
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le 2f
    add x9, x9, x0
    sub x10, x10, x0
    b 1b
2:
    mov x24, #0
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
rowbuf:
    .space 16
inbuf:
    .space 4096
outbuf:
    .space 8192
