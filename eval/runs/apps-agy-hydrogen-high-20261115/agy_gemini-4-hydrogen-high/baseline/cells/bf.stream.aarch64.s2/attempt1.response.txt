.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.text
.globl _start
_start:
    adrp x19, in_buf
    add x19, x19, :lo12:in_buf
    mov x20, #0
    movz x21, #0x0400
    movk x21, #1, lsl #16

read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x0, #0
    add x24, x19, x20

find_bang:
    cmp x0, x20
    b.eq no_bang
    ldrb w1, [x19, x0]
    cmp w1, #0x21
    b.eq found_bang
    add x0, x0, #1
    b find_bang

found_bang:
    mov x2, x0
    add x23, x19, x0
    add x23, x23, #1
    b parse_start

no_bang:
    mov x2, x20
    mov x23, x24

parse_start:
    adrp x4, code_buf
    add x4, x4, :lo12:code_buf
    adrp x6, bracket_stack
    add x6, x6, :lo12:bracket_stack
    adrp x21, jump_table
    add x21, x21, :lo12:jump_table
    mov x3, #0
    mov x5, #0
    mov x7, #0

parse_loop:
    cmp x3, x2
    b.eq parse_done
    ldrb w0, [x19, x3]
    add x3, x3, #1

    cmp w0, #0x3E
    b.eq emit_gt
    cmp w0, #0x3C
    b.eq emit_lt
    cmp w0, #0x2B
    b.eq emit_plus
    cmp w0, #0x2D
    b.eq emit_minus
    cmp w0, #0x2E
    b.eq emit_dot
    cmp w0, #0x2C
    b.eq emit_comma
    cmp w0, #0x5B
    b.eq emit_lbrack
    cmp w0, #0x5D
    b.eq emit_rbrack
    b parse_loop

emit_gt:
    mov w1, #1
    b emit_store

emit_lt:
    mov w1, #2
    b emit_store

emit_plus:
    mov w1, #3
    b emit_store

emit_minus:
    mov w1, #4
    b emit_store

emit_dot:
    mov w1, #5
    b emit_store

emit_comma:
    mov w1, #6

emit_store:
    strb w1, [x4, x5]
    add x5, x5, #1
    b parse_loop

emit_lbrack:
    str w5, [x6, x7, lsl #2]
    add x7, x7, #1
    mov w1, #7
    strb w1, [x4, x5]
    add x5, x5, #1
    b parse_loop

emit_rbrack:
    cbz x7, exit_err
    sub x7, x7, #1
    ldr w8, [x6, x7, lsl #2]
    add w9, w5, #1
    str w9, [x21, x8, lsl #2]
    add w9, w8, #1
    str w9, [x21, x5, lsl #2]
    mov w1, #8
    strb w1, [x4, x5]
    add x5, x5, #1
    b parse_loop

parse_done:
    cbnz x7, exit_err
    strb wzr, [x4, x5]

    mov x19, x4
    mov x20, #0
    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x25, out_buf
    add x25, x25, :lo12:out_buf
    mov x26, #0
    adr x27, dispatch_table

exec_loop:
    ldrb w0, [x19, x20]
    add x20, x20, #1
    add x1, x27, x0, lsl #2
    br x1

dispatch_table:
    b op_halt
    b op_gt
    b op_lt
    b op_plus
    b op_minus
    b op_dot
    b op_comma
    b op_lbrack
    b op_rbrack

op_gt:
    add x22, x22, #1
    b exec_loop

op_lt:
    sub x22, x22, #1
    b exec_loop

op_plus:
    ldrb w0, [x22]
    add w0, w0, #1
    strb w0, [x22]
    b exec_loop

op_minus:
    ldrb w0, [x22]
    sub w0, w0, #1
    strb w0, [x22]
    b exec_loop

op_dot:
    ldrb w0, [x22]
    strb w0, [x25, x26]
    add x26, x26, #1
    cmp x26, #65536
    b.lo exec_loop
    bl flush_output
    b exec_loop

op_comma:
    cmp x23, x24
    b.hs 1f
    ldrb w0, [x23], #1
    strb w0, [x22]
    b exec_loop
1:
    strb wzr, [x22]
    b exec_loop

op_lbrack:
    ldrb w0, [x22]
    cbnz w0, exec_loop
    sub x2, x20, #1
    ldr w20, [x21, x2, lsl #2]
    b exec_loop

op_rbrack:
    ldrb w0, [x22]
    cbz w0, exec_loop
    sub x2, x20, #1
    ldr w20, [x21, x2, lsl #2]
    b exec_loop

op_halt:
    bl flush_output
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

exit_err:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

flush_output:
    cbz x26, 2f
    mov x3, #0
1:
    mov x0, #1
    add x1, x25, x3
    sub x2, x26, x3
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 2f
    add x3, x3, x0
    cmp x3, x26
    b.lo 1b
2:
    mov x26, #0
    ret

.bss
.balign 16
in_buf:
    .zero 66560

code_buf:
    .zero 66560
.balign 16
out_buf:
    .zero 65536
.balign 16
tape:
    .zero 30016
.balign 16

    .zero 266240
.balign 16
bracket_stack:
    .zero 266240
