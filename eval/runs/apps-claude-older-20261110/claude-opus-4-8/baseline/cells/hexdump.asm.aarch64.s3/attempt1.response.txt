    mov x24, #16

    cmp x26, x23

    strb w0, [x22], #1
    mov w0, #0x7c
    strb w0, [x22], #1
    mov x26, #0

    bge chloop_done
    add x0, x21, x20
    ldrb w0, [x0, x26]
    cmp w0, #0x20
    blt dot
    cmp w0, #0x7e

    add x26, x26, #1

    add x22, x22, :lo12:line
    mov x0, x19
    mov x1, x22
    bl emit8hex
    mov w0, #0x0a
    strb w0, [x1], #1
    mov x2, #9
    mov x1, x22
    mov x0, #1
    mov x8, #64

    mov x0, #0

    strb w12, [x1], #1
