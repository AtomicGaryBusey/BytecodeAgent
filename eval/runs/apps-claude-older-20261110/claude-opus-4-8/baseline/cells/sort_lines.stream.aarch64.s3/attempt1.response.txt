.globl _start

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 16
inbuf:
    .zero 32800
.balign 16
descs:
    .zero 16000

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
read_loop:
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    movz    x2, #32800
    sub     x2, x2, x20
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    adrp    x21, descs
    add     x21, x21, :lo12:descs
    mov     x22, #0          // line count
    mov     x23, #0          // start index
    mov     x24, #0          // i
parse_loop:
    cmp     x24, x20
    b.ge    parse_tail
    ldrb    w0, [x19, x24]
    cmp     w0, #0x0A
    b.ne    1f
    add     x25, x19, x23
    sub     x26, x24, x23
    lsl     x0, x22, #4
    add     x0, x21, x0
    stp     x25, x26, [x0]
    add     x22, x22, #1
    add     x23, x24, #1
1:
    add     x24, x24, #1
    b       parse_loop
parse_tail:
    cmp     x23, x20
    b.ge    parse_done
    add     x25, x19, x23
    sub     x26, x20, x23
    lsl     x0, x22, #4
    add     x0, x21, x0
    stp     x25, x26, [x0]
    add     x22, x22, #1
parse_done:

    mov     x9, #1
sort_outer:
    cmp     x9, x22
    b.ge    sort_done
    lsl     x10, x9, #4
    add     x10, x21, x10
    ldp     x11, x12, [x10]
    sub     x13, x9, #1
sort_inner:
    tbnz    x13, #63, sort_insert
    lsl     x14, x13, #4
    add     x14, x21, x14
    ldp     x0, x1, [x14]
    mov     x2, x11
    mov     x3, x12
    bl      cmp_lines
    cmp     w0, #0
    b.le    sort_insert
    ldp     x0, x1, [x14]
    stp     x0, x1, [x14, #16]
    sub     x13, x13, #1
    b       sort_inner
sort_insert:
    add     x14, x13, #1
    lsl     x14, x14, #4
    add     x14, x21, x14
    stp     x11, x12, [x14]
    add     x9, x9, #1
    b       sort_outer
sort_done:

    mov     x9, #0
out_loop:
    cmp     x9, x22
    b.ge    out_done
    lsl     x10, x9, #4
    add     x10, x21, x10
    ldp     x1, x2, [x10]
    mov     x8, #64
    mov     x0, #1
    svc     #0
    mov     x8, #64
    mov     x0, #1
    adrp    x1, nl
    add     x1, x1, :lo12:nl
    mov     x2, #1
    svc     #0
    add     x9, x9, #1
    b       out_loop
out_done:
    mov     x8, #93
    mov     x0, #0
    svc     #0

// cmp_lines(x0=p1,x1=l1,x2=p2,x3=l2) -> w0 sign
cmp_lines:
    cmp     x1, x3
    csel    x4, x1, x3, ls
    mov     x5, #0
1:
    cmp     x5, x4
    b.ge    2f
    ldrb    w6, [x0, x5]
    ldrb    w7, [x2, x5]
    cmp     w6, w7
    b.ne    3f
    add     x5, x5, #1
    b       1b
3:
    sub     w0, w6, w7
    ret
2:
    sub     w0, w1, w3
    ret
