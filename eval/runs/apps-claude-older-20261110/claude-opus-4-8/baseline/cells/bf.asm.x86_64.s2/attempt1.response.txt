.text
.globl _start

_start:
    lea rbx, [rip+buf]
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = total bytes
    xor rcx, rcx
find_bang:
    cmp rcx, r12
    jae no_bang
    cmp byte ptr [rbx+rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov rbp, r12
    mov r14, r12
    mov r15, r12
    jmp validate
found_bang:
    mov rbp, rcx
    lea r14, [rcx+1]
    mov r15, r12

validate:
    lea rdi, [rip+jmp]
    lea rsi, [rip+bstack]
    xor rcx, rcx
    xor rdx, rdx
vloop:
    cmp rcx, rbp
    jae vend
    movzx eax, byte ptr [rbx+rcx]
    cmp al, 0x5b
    je v_open
    cmp al, 0x5d
    je v_close
    inc rcx
    jmp vloop
v_open:
    mov dword ptr [rsi+rdx*4], ecx
    inc rdx
    inc rcx
    jmp vloop
v_close:
    test rdx, rdx
    jz verror
    dec rdx
    mov eax, dword ptr [rsi+rdx*4]
    mov dword ptr [rdi+rax*4], ecx
    mov dword ptr [rdi+rcx*4], eax
    inc rcx
    jmp vloop
vend:
    test rdx, rdx
    jnz verror
    jmp run

verror:
    mov eax, 60
    mov edi, 1
    syscall

run:
    xor r12, r12
    lea r13, [rip+tape]
    xor r8, r8
    lea r9, [rip+jmp]
    lea r10, [rip+outbuf]

run_loop:
    cmp r12, rbp
    jae run_done
    movzx eax, byte ptr [rbx+r12]
    cmp al, 0x3e
    je c_right
    cmp al, 0x3c
    je c_left
    cmp al, 0x2b
    je c_plus
    cmp al, 0x2d
    je c_minus
    cmp al, 0x2e
    je c_dot
    cmp al, 0x2c
    je c_comma
    cmp al, 0x5b
    je c_lb
    cmp al, 0x5d
    je c_rb
next:
    inc r12
    jmp run_loop

c_right:
    inc r13
    jmp next
c_left:
    dec r13
    jmp next
c_plus:
    inc byte ptr [r13]
    jmp next
c_minus:
    dec byte ptr [r13]
    jmp next
c_dot:
    mov al, byte ptr [r13]
    mov byte ptr [r10+r8], al
    inc r8
    cmp r8, 65536
    jb next
    call flush
    jmp next
c_comma:
    cmp r14, r15
    jae c_comma_zero
    mov al, byte ptr [rbx+r14]
    inc r14
    mov byte ptr [r13], al
    jmp next
c_comma_zero:
    mov byte ptr [r13], 0
    jmp next
c_lb:
    cmp byte ptr [r13], 0
    jne next
    mov eax, dword ptr [r9+r12*4]
    mov r12d, eax
    jmp next
c_rb:
    cmp byte ptr [r13], 0
    je next
    mov eax, dword ptr [r9+r12*4]
    mov r12d, eax
    jmp next

run_done:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    test r8, r8
    jz 1f
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r8
    syscall
    xor r8, r8
1:
    ret

.bss
buf:
    .zero 65536
tape:
    .zero 30000
jmp:
    .zero 262144
bstack:
    .zero 4096
outbuf:
    .zero 65536
