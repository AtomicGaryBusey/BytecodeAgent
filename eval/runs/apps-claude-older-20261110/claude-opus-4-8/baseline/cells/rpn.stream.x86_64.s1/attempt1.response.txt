.globl _start
.text

_start:
    lea rbp, [rip + inbuf]
    xor r14, r14
rdl:
    mov rax, 0
    xor rdi, rdi
    mov rsi, rbp
    add rsi, r14
    mov rdx, 70000
    sub rdx, r14
    jle rdone
    syscall
    test rax, rax
    jle rdone
    add r14, rax
    jmp rdl
rdone:
    xor rbx, rbx

line_loop:
    cmp rbx, r14
    jae all_done
    mov r15, rbx
fnl:
    cmp r15, r14
    jae have_line
    movzx eax, byte ptr [rbp + r15]
    cmp al, 10
    je have_line
    inc r15
    jmp fnl

have_line:
    xor r13, r13
    mov r12, rbx
tok_loop:
    cmp r12, r15
    jae tok_done
    movzx eax, byte ptr [rbp + r12]
    cmp al, 0x20
    je sepc
    cmp al, 0x09
    jne token_start
sepc:
    inc r12
    jmp tok_loop

token_start:
    mov r8, r12
tfind:
    cmp r12, r15
    jae tfound
    movzx eax, byte ptr [rbp + r12]
    cmp al, 0x20
    je tfound
    cmp al, 0x09
    je tfound
    inc r12
    jmp tfind
tfound:
    mov r9, r12
    mov rcx, r9
    sub rcx, r8
    cmp rcx, 1
    jne try_int
    movzx eax, byte ptr [rbp + r8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

try_int:
    mov rsi, r8
    xor r10d, r10d
    movzx eax, byte ptr [rbp + rsi]
    cmp al, '-'
    jne ti_digits
    mov r10d, 1
    inc rsi
ti_digits:
    mov rcx, r9
    sub rcx, rsi
    cmp rcx, 1
    jl do_error
    cmp rcx, 18
    jg do_error
    xor rax, rax
ti_loop:
    cmp rsi, r9
    jae ti_done
    movzx edx, byte ptr [rbp + rsi]
    sub edx, '0'
    cmp edx, 9
    ja do_error
    lea rax, [rax + rax*4]
    add rax, rax
    add rax, rdx
    inc rsi
    jmp ti_loop
ti_done:
    test r10d, r10d
    jz ti_push
    neg rax
ti_push:
    lea rdi, [rip + valstack]
    mov qword ptr [rdi + r13*8], rax
    inc r13
    jmp tok_loop

op_add:
    cmp r13, 2
    jl do_error
    lea rdi, [rip + valstack]
    mov rdx, qword ptr [rdi + r13*8 - 8]
    mov rax, qword ptr [rdi + r13*8 - 16]
    sub r13, 2
    add rax, rdx
    mov qword ptr [rdi + r13*8], rax
    inc r13
    jmp tok_loop

op_sub:
    cmp r13, 2
    jl do_error
    lea rdi, [rip + valstack]
    mov rdx, qword ptr [rdi + r13*8 - 8]
    mov rax, qword ptr [rdi + r13*8 - 16]
    sub r13, 2
    sub rax, rdx
    mov qword ptr [rdi + r13*8], rax
    inc r13
    jmp tok_loop

op_mul:
    cmp r13, 2
    jl do_error
    lea rdi, [rip + valstack]
    mov rdx, qword ptr [rdi + r13*8 - 8]
    mov rax, qword ptr [rdi + r13*8 - 16]
    sub r13, 2
    imul rax, rdx
    mov qword ptr [rdi + r13*8], rax
    inc r13
    jmp tok_loop

op_div:
    cmp r13, 2
    jl do_error
    lea rdi, [rip + valstack]
    mov rcx, qword ptr [rdi + r13*8 - 8]
    mov rax, qword ptr [rdi + r13*8 - 16]
    sub r13, 2
    test rcx, rcx
    je do_error
    cmp rcx, -1
    jne ddiv
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je do_error
ddiv:
    cqo
    idiv rcx
    mov qword ptr [rdi + r13*8], rax
    inc r13
    jmp tok_loop

op_mod:
    cmp r13, 2
    jl do_error
    lea rdi, [rip + valstack]
    mov rcx, qword ptr [rdi + r13*8 - 8]
    mov rax, qword ptr [rdi + r13*8 - 16]
    sub r13, 2
    test rcx, rcx
    je do_error
    cmp rcx, -1
    jne dmod
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je do_error
dmod:
    cqo
    idiv rcx
    mov qword ptr [rdi + r13*8], rdx
    inc r13
    jmp tok_loop

tok_done:
    cmp r13, 1
    jne do_error
    lea rdi, [rip + valstack]
    mov rax, qword ptr [rdi]
    call emit_num
    jmp advance

do_error:
    call emit_error
    jmp advance

advance:
    lea rbx, [r15 + 1]
    jmp line_loop

all_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, qword ptr [rip + out_idx]
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

emit_num:
    lea rdi, [rip + outbuf]
    add rdi, qword ptr [rip + out_idx]
    test rax, rax
    jns pos2
    mov byte ptr [rdi], '-'
    inc rdi
    neg rax
pos2:
    lea rsi, [rip + numtmp]
    mov r8, rsi
    mov rcx, 10
convloop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [rsi], dl
    inc rsi
    test rax, rax
    jnz convloop
copyrev:
    dec rsi
    movzx eax, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rdi
    cmp rsi, r8
    jne copyrev
    mov byte ptr [rdi], 10
    inc rdi
    lea rax, [rip + outbuf]
    sub rdi, rax
    mov qword ptr [rip + out_idx], rdi
    ret

emit_error:
    lea rdi, [rip + outbuf]
    add rdi, qword ptr [rip + out_idx]
    lea rsi, [rip + errstr]
    mov rcx, 6
cpyerr:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz cpyerr
    lea rax, [rip + outbuf]
    sub rdi, rax
    mov qword ptr [rip + out_idx], rdi
    ret

.section .rodata
errstr:
    .ascii "error\n"

.bss
.align 16
inbuf:
    .zero 70000
valstack:
    .zero 16384
numtmp:
    .zero 32
out_idx:
    .zero 8
outbuf:
    .zero 500000
