.globl _start

.text

_start:
    adrp x21, buf
    add x21, x21, :lo12:buf
    mov x19, #0
.read_loop:
    mov x0, #0
    add x1, x21, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le .read_done
    add x19, x19, x0
    b .read_loop
.read_done:
    mov x20, x19
    mov x19, #0

    bl skip_ws
    bl parse_value
    cbnz w0, .error
    bl skip_ws
    cmp x19, x20
    b.lt .error

    // print "ok\n"
    adrp x1, ok_str
    add x1, x1, :lo12:ok_str
    mov x2, #3
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

.error:
    // print "error at "
    adrp x1, err_str
    add x1, x1, :lo12:err_str
    mov x2, #9
    mov x0, #1
    mov x8, #64
    svc #0
    // convert x19 to decimal into numbuf
    adrp x4, numbuf
    add x4, x4, :lo12:numbuf
    add x5, x4, #20
    mov x3, x19
    mov x6, #10
    cbnz x3, .conv_loop
    sub x5, x5, #1
    mov w9, #0x30
    strb w9, [x5]
    b .conv_done
.conv_loop:
    cbz x3, .conv_done
    udiv x7, x3, x6
    msub x9, x7, x6, x3
    add w9, w9, #0x30
    sub x5, x5, #1
    strb w9, [x5]
    mov x3, x7
    b .conv_loop
.conv_done:
    add x10, x4, #20
    sub x2, x10, x5
    mov x1, x5
    mov x0, #1
    mov x8, #64
    svc #0
    // newline
    adrp x1, nl_str
    add x1, x1, :lo12:nl_str
    mov x2, #1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

// skip_ws: advance x19 past whitespace
skip_ws:
.sw_loop:
    cmp x19, x20
    b.ge .sw_done
    ldrb w0, [x21, x19]
    cmp w0, #0x20
    b.eq .sw_adv
    cmp w0, #0x09
    b.eq .sw_adv
    cmp w0, #0x0A
    b.eq .sw_adv
    cmp w0, #0x0D
    b.eq .sw_adv
    b .sw_done
.sw_adv:
    add x19, x19, #1
    b .sw_loop
.sw_done:
    ret

// match_literal: x0 = ptr, x1 = len. returns w0 = 0 ok, 1 fail
match_literal:
    mov x2, #0
.ml_loop:
    cmp x2, x1
    b.ge .ml_ok
    cmp x19, x20
    b.ge .ml_fail
    ldrb w3, [x21, x19]
    ldrb w4, [x0, x2]
    cmp w3, w4
    b.ne .ml_fail
    add x19, x19, #1
    add x2, x2, #1
    b .ml_loop
.ml_ok:
    mov w0, #0
    ret
.ml_fail:
    mov w0, #1
    ret

// parse_value
parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x19, x20
    b.ge .pv_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x7B
    b.eq .pv_obj
    cmp w0, #0x5B
    b.eq .pv_arr
    cmp w0, #0x22
    b.eq .pv_str
    cmp w0, #0x74
    b.eq .pv_true
    cmp w0, #0x66
    b.eq .pv_false
    cmp w0, #0x6E
    b.eq .pv_null
    cmp w0, #0x2D
    b.eq .pv_num
    cmp w0, #0x30
    b.lt .pv_fail
    cmp w0, #0x39
    b.gt .pv_fail
.pv_num:
    bl parse_number
    b .pv_ret
.pv_obj:
    bl parse_object
    b .pv_ret
.pv_arr:
    bl parse_array
    b .pv_ret
.pv_str:
    bl parse_string
    b .pv_ret
.pv_true:
    adrp x0, lit_true
    add x0, x0, :lo12:lit_true
    mov x1, #4
    bl match_literal
    b .pv_ret
.pv_false:
    adrp x0, lit_false
    add x0, x0, :lo12:lit_false
    mov x1, #5
    bl match_literal
    b .pv_ret
.pv_null:
    adrp x0, lit_null
    add x0, x0, :lo12:lit_null
    mov x1, #4
    bl match_literal
    b .pv_ret
.pv_fail:
    mov w0, #1
.pv_ret:
    ldp x29, x30, [sp], #16
    ret

// parse_string: current byte is '"'
parse_string:
    add x19, x19, #1
.ps_loop:
    cmp x19, x20
    b.ge .ps_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x22
    b.eq .ps_close
    cmp w0, #0x5C
    b.eq .ps_esc
    cmp w0, #0x20
    b.lt .ps_fail
    add x19, x19, #1
    b .ps_loop
.ps_close:
    add x19, x19, #1
    mov w0, #0
    ret
.ps_esc:
    add x19, x19, #1
    cmp x19, x20
    b.ge .ps_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x22
    b.eq .ps_esc1
    cmp w0, #0x5C
    b.eq .ps_esc1
    cmp w0, #0x2F
    b.eq .ps_esc1
    cmp w0, #0x62
    b.eq .ps_esc1
    cmp w0, #0x66
    b.eq .ps_esc1
    cmp w0, #0x6E
    b.eq .ps_esc1
    cmp w0, #0x72
    b.eq .ps_esc1
    cmp w0, #0x74
    b.eq .ps_esc1
    cmp w0, #0x75
    b.eq .ps_u
    b .ps_fail
.ps_esc1:
    add x19, x19, #1
    b .ps_loop
.ps_u:
    add x19, x19, #1
    mov x5, #4
.ps_u_loop:
    cmp x5, #0
    b.eq .ps_loop
    cmp x19, x20
    b.ge .ps_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .ps_fail
    cmp w0, #0x39
    b.le .ps_u_ok
    cmp w0, #0x41
    b.lt .ps_fail
    cmp w0, #0x46
    b.le .ps_u_ok
    cmp w0, #0x61
    b.lt .ps_fail
    cmp w0, #0x66
    b.gt .ps_fail
.ps_u_ok:
    add x19, x19, #1
    sub x5, x5, #1
    b .ps_u_loop
.ps_fail:
    mov w0, #1
    ret

// parse_number
parse_number:
    cmp x19, x20
    b.ge .pn_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x2D
    b.ne .pn_int
    add x19, x19, #1
.pn_int:
    cmp x19, x20
    b.ge .pn_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .pn_fail
    cmp w0, #0x39
    b.gt .pn_fail
    cmp w0, #0x30
    b.eq .pn_zero
    add x19, x19, #1
.pn_digits:
    cmp x19, x20
    b.ge .pn_frac
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .pn_frac
    cmp w0, #0x39
    b.gt .pn_frac
    add x19, x19, #1
    b .pn_digits
.pn_zero:
    add x19, x19, #1
.pn_frac:
    cmp x19, x20
    b.ge .pn_ok
    ldrb w0, [x21, x19]
    cmp w0, #0x2E
    b.ne .pn_exp
    add x19, x19, #1
    cmp x19, x20
    b.ge .pn_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .pn_fail
    cmp w0, #0x39
    b.gt .pn_fail
    add x19, x19, #1
.pn_frac_d:
    cmp x19, x20
    b.ge .pn_ok
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .pn_exp
    cmp w0, #0x39
    b.gt .pn_exp
    add x19, x19, #1
    b .pn_frac_d
.pn_exp:
    cmp x19, x20
    b.ge .pn_ok
    ldrb w0, [x21, x19]
    cmp w0, #0x65
    b.eq .pn_exp_go
    cmp w0, #0x45
    b.eq .pn_exp_go
    b .pn_ok
.pn_exp_go:
    add x19, x19, #1
    cmp x19, x20
    b.ge .pn_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x2B
    b.eq .pn_exp_sign
    cmp w0, #0x2D
    b.eq .pn_exp_sign
    b .pn_exp_digit
.pn_exp_sign:
    add x19, x19, #1
.pn_exp_digit:
    cmp x19, x20
    b.ge .pn_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .pn_fail
    cmp w0, #0x39
    b.gt .pn_fail
    add x19, x19, #1
.pn_exp_d2:
    cmp x19, x20
    b.ge .pn_ok
    ldrb w0, [x21, x19]
    cmp w0, #0x30
    b.lt .pn_ok
    cmp w0, #0x39
    b.gt .pn_ok
    add x19, x19, #1
    b .pn_exp_d2
.pn_ok:
    mov w0, #0
    ret
.pn_fail:
    mov w0, #1
    ret

// parse_object: current byte is '{'
parse_object:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.ge .po_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x7D
    b.eq .po_close
.po_member:
    cmp x19, x20
    b.ge .po_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x22
    b.ne .po_fail
    bl parse_string
    cbnz w0, .po_ret
    bl skip_ws
    cmp x19, x20
    b.ge .po_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x3A
    b.ne .po_fail
    add x19, x19, #1
    bl skip_ws
    bl parse_value
    cbnz w0, .po_ret
    bl skip_ws
    cmp x19, x20
    b.ge .po_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x2C
    b.eq .po_comma
    cmp w0, #0x7D
    b.eq .po_close
    b .po_fail
.po_comma:
    add x19, x19, #1
    bl skip_ws
    b .po_member
.po_close:
    add x19, x19, #1
    mov w0, #0
    b .po_ret
.po_fail:
    mov w0, #1
.po_ret:
    ldp x29, x30, [sp], #16
    ret

// parse_array: current byte is '['
parse_array:
    stp x29, x30, [sp, #-16]!
    add x19, x19, #1
    bl skip_ws
    cmp x19, x20
    b.ge .pa_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x5D
    b.eq .pa_close
.pa_elem:
    bl parse_value
    cbnz w0, .pa_ret
    bl skip_ws
    cmp x19, x20
    b.ge .pa_fail
    ldrb w0, [x21, x19]
    cmp w0, #0x2C
    b.eq .pa_comma
    cmp w0, #0x5D
    b.eq .pa_close
    b .pa_fail
.pa_comma:
    add x19, x19, #1
    bl skip_ws
    b .pa_elem
.pa_close:
    add x19, x19, #1
    mov w0, #0
    b .pa_ret
.pa_fail:
    mov w0, #1
.pa_ret:
    ldp x29, x30, [sp], #16
    ret

.section .rodata
ok_str:    .ascii "ok\n"
err_str:   .ascii "error at "
nl_str:    .ascii "\n"
lit_true:  .ascii "true"
lit_false: .ascii "false"
lit_null:  .ascii "null"

.bss
.balign 16
buf:    .zero 65536
numbuf: .zero 32
