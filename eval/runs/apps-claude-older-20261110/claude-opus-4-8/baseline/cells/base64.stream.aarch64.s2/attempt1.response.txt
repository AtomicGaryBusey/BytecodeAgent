.section .bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072

.section .rodata
.balign 16
b64table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    adrp x21, inbuf
    add  x21, x21, :lo12:inbuf
    mov  x23, x21               // read pos
read_loop:
    mov  x0, #0
    mov  x1, x23
    mov  x2, #65536
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x23, x23, x0
    b    read_loop
read_done:
    sub  x22, x23, x21          // in_len

    adrp x19, outbuf
    add  x19, x19, :lo12:outbuf
    mov  x28, x19               // out base
    mov  w20, #0                // col
    mov  x24, #0                // i
    adrp x23, b64table
    add  x23, x23, :lo12:b64table

main_loop:
    add  x0, x24, #3
    cmp  x0, x22
    b.gt tail
    ldrb w25, [x21, x24]
    add  x0, x24, #1
    ldrb w26, [x21, x0]
    add  x0, x24, #2
    ldrb w27, [x21, x0]
    add  x24, x24, #3

    lsr  w0, w25, #2
    bl   emit_idx
    and  w0, w25, #3
    lsl  w0, w0, #4
    lsr  w1, w26, #4
    orr  w0, w0, w1
    bl   emit_idx
    and  w0, w26, #0xf
    lsl  w0, w0, #2
    lsr  w1, w27, #6
    orr  w0, w0, w1
    bl   emit_idx
    and  w0, w27, #0x3f
    bl   emit_idx
    b    main_loop

tail:
    sub  w0, w22, w24
    cbz  w0, finish
    cmp  w0, #1
    b.eq rem1

    ldrb w25, [x21, x24]
    add  x0, x24, #1
    ldrb w26, [x21, x0]
    lsr  w0, w25, #2
    bl   emit_idx
    and  w0, w25, #3
    lsl  w0, w0, #4
    lsr  w1, w26, #4
    orr  w0, w0, w1
    bl   emit_idx
    and  w0, w26, #0xf
    lsl  w0, w0, #2
    bl   emit_idx
    mov  w0, #61
    bl   emit_char
    b    finish

rem1:
    ldrb w25, [x21, x24]
    lsr  w0, w25, #2
    bl   emit_idx
    and  w0, w25, #3
    lsl  w0, w0, #4
    bl   emit_idx
    mov  w0, #61
    bl   emit_char
    mov  w0, #61
    bl   emit_char

finish:
    cbz  w20, write_out
    mov  w0, #10
    strb w0, [x19], #1
write_out:
    mov  x1, x28
    sub  x2, x19, x28
    mov  x0, #1
    mov  x8, #64
    svc  #0
    mov  x0, #0
    mov  x8, #93
    svc  #0

emit_idx:
    ldrb w0, [x23, w0, uxtw]
emit_char:
    strb w0, [x19], #1
    add  w20, w20, #1
    cmp  w20, #76
    b.ne 1f
    mov  w0, #10
    strb w0, [x19], #1
    mov  w20, #0
1:
    ret
