.globl _start

.equ BUFCAP, 65536

.text

skip_ws:
    cmp r12, r13
    jae sw_done
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0A
    je sw_adv
    cmp al, 0x0D
    je sw_adv
    ret
sw_adv:
    inc r12
    jmp skip_ws
sw_done:
    ret

pv:
    cmp r12, r13
    jae pv_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x7B
    je pv_obj
    cmp al, 0x5B
    je pv_arr
    cmp al, 0x22
    je pstr
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6E
    je pv_null
    cmp al, 0x2D
    je pnum
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    jbe pnum
pv_fail:
    mov eax, 1
    ret
pv_obj:
    jmp pobj
pv_arr:
    jmp parr
pv_true:
    lea rsi, [rip+lit_true]
    mov edx, 4
    jmp match_lit
pv_false:
    lea rsi, [rip+lit_false]
    mov edx, 5
    jmp match_lit
pv_null:
    lea rsi, [rip+lit_null]
    mov edx, 4
    jmp match_lit

match_lit:
    xor ecx, ecx
ml_loop:
    cmp ecx, edx
    jae ml_ok
    cmp r12, r13
    jae ml_fail
    movzx eax, byte ptr [r14+r12]
    movzx r8d, byte ptr [rsi+rcx]
    cmp al, r8b
    jne ml_fail
    inc r12
    inc ecx
    jmp ml_loop
ml_ok:
    xor eax, eax
    ret
ml_fail:
    mov eax, 1
    ret

pstr:
    inc r12
ps_loop:
    cmp r12, r13
    jae ps_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x22
    je ps_close
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb ps_fail
    inc r12
    jmp ps_loop
ps_close:
    inc r12
    xor eax, eax
    ret
ps_fail:
    mov eax, 1
    ret
ps_esc:
    inc r12
    cmp r12, r13
    jae ps_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x22
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, 0x2F
    je ps_esc_ok
    cmp al, 0x62
    je ps_esc_ok
    cmp al, 0x66
    je ps_esc_ok
    cmp al, 0x6E
    je ps_esc_ok
    cmp al, 0x72
    je ps_esc_ok
    cmp al, 0x74
    je ps_esc_ok
    cmp al, 0x75
    je ps_esc_u
    jmp ps_fail
ps_esc_ok:
    inc r12
    jmp ps_loop
ps_esc_u:
    inc r12
    mov ecx, 4
ps_u_loop:
    cmp r12, r13
    jae ps_fail
    movzx eax, byte ptr [r14+r12]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    jbe ps_u_ok
    mov edx, eax
    or edx, 0x20
    sub edx, 0x61
    cmp edx, 5
    jbe ps_u_ok
    jmp ps_fail
ps_u_ok:
    inc r12
    dec ecx
    jnz ps_u_loop
    jmp ps_loop

pnum:
    cmp r12, r13
    jae pn_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x2D
    jne pn_int
    inc r12
pn_int:
    cmp r12, r13
    jae pn_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x30
    je pn_zero
    mov edx, eax
    sub edx, 0x31
    cmp edx, 8
    ja pn_fail
    inc r12
pn_digs:
    cmp r12, r13
    jae pn_frac
    movzx eax, byte ptr [r14+r12]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    ja pn_frac
    inc r12
    jmp pn_digs
pn_zero:
    inc r12
pn_frac:
    cmp r12, r13
    jae pn_ok
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x2E
    jne pn_exp
    inc r12
    cmp r12, r13
    jae pn_fail
    movzx eax, byte ptr [r14+r12]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    ja pn_fail
    inc r12
pn_frac_digs:
    cmp r12, r13
    jae pn_ok
    movzx eax, byte ptr [r14+r12]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    ja pn_exp
    inc r12
    jmp pn_frac_digs
pn_exp:
    cmp r12, r13
    jae pn_ok
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x65
    je pn_exp_go
    cmp al, 0x45
    je pn_exp_go
    jmp pn_ok
pn_exp_go:
    inc r12
    cmp r12, r13
    jae pn_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x2B
    je pn_exp_sign
    cmp al, 0x2D
    je pn_exp_sign
    jmp pn_exp_dig1
pn_exp_sign:
    inc r12
pn_exp_dig1:
    cmp r12, r13
    jae pn_fail
    movzx eax, byte ptr [r14+r12]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    ja pn_fail
    inc r12
pn_exp_digs:
    cmp r12, r13
    jae pn_ok
    movzx eax, byte ptr [r14+r12]
    mov edx, eax
    sub edx, 0x30
    cmp edx, 9
    ja pn_ok
    inc r12
    jmp pn_exp_digs
pn_ok:
    xor eax, eax
    ret
pn_fail:
    mov eax, 1
    ret

pobj:
    inc r12
    call skip_ws
    cmp r12, r13
    jae po_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x7D
    je po_close
po_member:
    cmp r12, r13
    jae po_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x22
    jne po_fail
    call pstr
    test eax, eax
    jnz po_prop
    call skip_ws
    cmp r12, r13
    jae po_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x3A
    jne po_fail
    inc r12
    call skip_ws
    call pv
    test eax, eax
    jnz po_prop
    call skip_ws
    cmp r12, r13
    jae po_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x7D
    je po_close
    cmp al, 0x2C
    jne po_fail
    inc r12
    call skip_ws
    jmp po_member
po_close:
    inc r12
    xor eax, eax
    ret
po_fail:
    mov eax, 1
po_prop:
    ret

parr:
    inc r12
    call skip_ws
    cmp r12, r13
    jae pa_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x5D
    je pa_close
pa_elem:
    call pv
    test eax, eax
    jnz pa_prop
    call skip_ws
    cmp r12, r13
    jae pa_fail
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x5D
    je pa_close
    cmp al, 0x2C
    jne pa_fail
    inc r12
    call skip_ws
    jmp pa_elem
pa_close:
    inc r12
    xor eax, eax
    ret
pa_fail:
    mov eax, 1
pa_prop:
    ret

_start:
    lea r14, [rip+buf]
    xor r13, r13
read_loop:
    mov rdx, BUFCAP
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r14+r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r12, r12
    call skip_ws
    call pv
    test eax, eax
    jnz do_error
    call skip_ws
    cmp r12, r13
    jne do_error
    lea rsi, [rip+msg_ok]
    mov edx, 3
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

do_error:
    mov rax, r12
    lea rsi, [rip+numbuf+31]
    mov rcx, 10
conv_loop:
    xor edx, edx
    div rcx
    add dl, 0x30
    mov byte ptr [rsi], dl
    dec rsi
    test rax, rax
    jnz conv_loop
    inc rsi
    lea rdx, [rip+numbuf+32]
    sub rdx, rsi
    push rsi
    push rdx
    lea rsi, [rip+msg_err]
    mov edx, 9
    mov eax, 1
    mov edi, 1
    syscall
    pop rdx
    pop rsi
    mov eax, 1
    mov edi, 1
    syscall
    lea rsi, [rip+msg_nl]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
lit_true:  .ascii "true"
lit_false: .ascii "false"
lit_null:  .ascii "null"
msg_ok:    .ascii "ok\n"
msg_err:   .ascii "error at "
msg_nl:    .ascii "\n"

.bss
.align 16
buf:    .zero 65536
numbuf: .zero 32
