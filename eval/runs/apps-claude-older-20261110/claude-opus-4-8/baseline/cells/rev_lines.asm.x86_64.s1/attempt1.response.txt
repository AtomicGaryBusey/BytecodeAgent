.globl _start

.bss
.align 16
inbuf:
.zero 65536
outbuf:
.zero 65540

.text
_start:
    lea rsi, [rip + inbuf]
    xor r12, r12            # total length

read_loop:
    xor eax, eax            # read
    xor edi, edi            # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    # process lines
    xor rbx, rbx            # i
    xor r13, r13            # output index o
    lea r14, [rip + inbuf]
    lea r15, [rip + outbuf]

line_loop:
    cmp rbx, r12
    jae finish
    # find newline
    mov rcx, rbx            # j
scan:
    cmp rcx, r12
    jae scan_done
    mov al, byte ptr [r14 + rcx]
    cmp al, 0x0A
    je scan_done
    inc rcx
    jmp scan
scan_done:
    # line is [rbx, rcx)
    mov rdx, rcx            # k = j
rev:
    cmp rdx, rbx
    jbe rev_done
    dec rdx
    mov al, byte ptr [r14 + rdx]
    mov byte ptr [r15 + r13], al
    inc r13
    jmp rev
rev_done:
    mov byte ptr [r15 + r13], 0x0A
    inc r13
    # advance i
    cmp rcx, r12
    jae set_end
    lea rbx, [rcx + 1]
    jmp line_loop
set_end:
    mov rbx, r12
    jmp line_loop

finish:
    # write output
    test r13, r13
    jz do_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall
