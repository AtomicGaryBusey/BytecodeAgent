.globl _start

.bss
.align 16
buf:
    .zero 65536

.text
_start:
    xor r12, r12          # lines
    xor r13, r13          # words
    xor r14, r14          # bytes
    xor r15, r15          # in_word flag (0 = not in word)

read_loop:
    xor eax, eax          # read
    xor edi, edi          # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax          # bytes += n
    lea rsi, [rip + buf]
    lea rcx, [rsi + rax]  # end pointer

proc_loop:
    cmp rsi, rcx
    je read_loop
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne not_nl
    inc r12
not_nl:
    # check whitespace
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    cmp al, 0x0A
    je is_ws
    # word byte
    test r15, r15
    jnz proc_loop
    mov r15, 1
    inc r13
    jmp proc_loop
is_ws:
    xor r15, r15
    jmp proc_loop

done_read:
    # print r12, r13, r14 separated by spaces, newline
    mov rax, r12
    call print_num
    call print_space
    mov rax, r13
    call print_num
    call print_space
    mov rax, r14
    call print_num
    call print_newline

    mov eax, 60
    xor edi, edi
    syscall

# print number in rax (decimal) to stdout
print_num:
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 0
    mov rbx, 10
    test rax, rax
    jnz pn_loop
    dec rdi
    mov byte ptr [rdi], '0'
    jmp pn_write
pn_loop:
    test rax, rax
    jz pn_write
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    jmp pn_loop
pn_write:
    # rdi points to start of string, compute length
    lea rsi, [rip + numbuf + 31]
    mov rdx, rsi
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    push rcx
    syscall
    pop rcx
    ret

print_space:
    lea rsi, [rip + spacech]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    push rcx
    syscall
    pop rcx
    ret

print_newline:
    lea rsi, [rip + nlch]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    push rcx
    syscall
    pop rcx
    ret

.section .rodata
spacech:
    .byte 0x20
nlch:
    .byte 0x0A

.bss
.align 16
numbuf:
    .zero 32
