.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x20, #0

read_loop:
    mov w0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cmp x2, #0
    b.le read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    mov x23, #0
    mov x24, #0
    mov x22, #0
loop_i:
    cmp x24, x20
    b.ge after
    ldrb w0, [x19, x24]
    cmp w0, #0x0a
    b.ne next_i
    mov x25, x24
revloop:
    cmp x25, x23
    b.le revdone
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b revloop
revdone:
    mov w0, #0x0a
    strb w0, [x21, x22]
    add x22, x22, #1
    add x23, x24, #1
next_i:
    add x24, x24, #1
    b loop_i
after:
    cmp x23, x20
    b.ge write_out
    mov x25, x20
revloop2:
    cmp x25, x23
    b.le revdone2
    sub x25, x25, #1
    ldrb w0, [x19, x25]
    strb w0, [x21, x22]
    add x22, x22, #1
    b revloop2
revdone2:
    mov w0, #0x0a
    strb w0, [x21, x22]
    add x22, x22, #1

write_out:
    cmp x22, #0
    b.eq do_exit
    mov x0, #1
    mov x1, x21
    mov x2, x22
    mov x8, #64
    svc #0
do_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 70000
