.intel_syntax noprefix
.globl _start

.text
_start:
    # read all of stdin into inbuf
    lea     rsi, [rip + inbuf]
    xor     rbx, rbx
read_loop:
    cmp     ebx, 65536
    jae     read_done
    xor     eax, eax            # read = 0
    xor     edi, edi            # fd 0
    lea     rsi, [rip + inbuf]
    add     rsi, rbx
    mov     edx, 65536
    sub     edx, ebx
    syscall
    test    rax, rax
    jle     read_done
    add     rbx, rax
    jmp     read_loop
read_done:
    mov     [rip + inlen], rbx

    # find first '!'
    lea     rsi, [rip + inbuf]
    xor     rcx, rcx
find_loop:
    cmp     rcx, rbx
    je      no_bang
    cmp     byte ptr [rsi + rcx], 0x21
    je      found_bang
    inc     rcx
    jmp     find_loop
no_bang:
    mov     [rip + proglen], rbx
    mov     [rip + inputpos], rbx
    jmp     after_split
found_bang:
    mov     [rip + proglen], rcx
    inc     rcx
    mov     [rip + inputpos], rcx
after_split:

    # validate brackets and build jump table
    lea     rsi, [rip + inbuf]
    lea     r8, [rip + jumptable]
    lea     r9, [rip + bstack]
    xor     rcx, rcx
    xor     rdx, rdx
vloop:
    cmp     rcx, [rip + proglen]
    jae     vend
    movzx   eax, byte ptr [rsi + rcx]
    cmp     al, 0x5b
    je      push_b
    cmp     al, 0x5d
    je      pop_b
    inc     rcx
    jmp     vloop
push_b:
    mov     [r9 + rdx*4], ecx
    inc     rdx
    inc     rcx
    jmp     vloop
pop_b:
    test    rdx, rdx
    jz      unbalanced
    dec     rdx
    mov     eax, [r9 + rdx*4]
    mov     [r8 + rax*4], ecx
    mov     [r8 + rcx*4], eax
    inc     rcx
    jmp     vloop
vend:
    test    rdx, rdx
    jnz     unbalanced

    # run
    lea     rsi, [rip + inbuf]
    lea     r8, [rip + jumptable]
    lea     r14, [rip + tape]
    xor     r13, r13
    mov     r12, [rip + proglen]
runloop:
    cmp     r13, r12
    jae     run_done
    movzx   eax, byte ptr [rsi + r13]
    cmp     al, 0x3e
    je      c_gt
    cmp     al, 0x3c
    je      c_lt
    cmp     al, 0x2b
    je      c_plus
    cmp     al, 0x2d
    je      c_minus
    cmp     al, 0x2e
    je      c_dot
    cmp     al, 0x2c
    je      c_comma
    cmp     al, 0x5b
    je      c_lb
    cmp     al, 0x5d
    je      c_rb
    inc     r13
    jmp     runloop
c_gt:
    inc     r14
    inc     r13
    jmp     runloop
c_lt:
    dec     r14
    inc     r13
    jmp     runloop
c_plus:
    inc     byte ptr [r14]
    inc     r13
    jmp     runloop
c_minus:
    dec     byte ptr [r14]
    inc     r13
    jmp     runloop
c_dot:
    mov     ecx, [rip + outpos]
    movzx   eax, byte ptr [r14]
    lea     rdi, [rip + outbuf]
    mov     [rdi + rcx], al
    inc     ecx
    mov     [rip + outpos], ecx
    inc     r13
    jmp     runloop
c_comma:
    mov     rcx, [rip + inputpos]
    cmp     rcx, [rip + inlen]
    jae     comma_zero
    mov     al, [rsi + rcx]
    mov     [r14], al
    inc     rcx
    mov     [rip + inputpos], rcx
    inc     r13
    jmp     runloop
comma_zero:
    mov     byte ptr [r14], 0
    inc     r13
    jmp     runloop
c_lb:
    cmp     byte ptr [r14], 0
    jne     lb_no
    mov     r13d, [r8 + r13*4]
lb_no:
    inc     r13
    jmp     runloop
c_rb:
    cmp     byte ptr [r14], 0
    je      rb_no
    mov     r13d, [r8 + r13*4]
rb_no:
    inc     r13
    jmp     runloop

run_done:
    mov     ecx, [rip + outpos]
    test    ecx, ecx
    jz      exit0
    mov     eax, 1
    mov     edi, 1
    lea     rsi, [rip + outbuf]
    mov     edx, ecx
    syscall
exit0:
    mov     eax, 60
    xor     edi, edi
    syscall

unbalanced:
    mov     eax, 60
    mov     edi, 1
    syscall

.bss
.align 8
inlen:      .zero 8
proglen:    .zero 8
inputpos:   .zero 8
outpos:     .zero 8
inbuf:      .zero 65536
outbuf:     .zero 65536
tape:       .zero 30000
jumptable:  .zero 262144
bstack:     .zero 4096
