.globl _start
.text
_start:
    lea r14, [rip + inbuf]
    xor r15, r15
read_loop:
    mov edx, 65536
    sub edx, r15d
    test edx, edx
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    lea rbx, [rip + inbuf]
    lea r13, [rbx + r15]
    lea rax, [rip + outbuf]
    mov [rip + outpos], rax

line_loop:
    cmp rbx, r13
    jae all_done
    mov r12, rbx
scan_nl:
    cmp r12, r13
    jae eval_start
    cmp byte ptr [r12], 10
    je eval_start
    inc r12
    jmp scan_nl
eval_start:
    mov r14, rbx
    xor ebp, ebp
    lea r15, [rip + vstack]

tok_loop:
    cmp r14, r12
    jae tok_end
    movzx eax, byte ptr [r14]
    cmp al, 32
    je sep
    cmp al, 9
    je sep
    jmp tok_got
sep:
    inc r14
    jmp tok_loop
tok_got:
    mov r10, r14
tscan:
    cmp r14, r12
    jae tdone
    movzx eax, byte ptr [r14]
    cmp al, 32
    je tdone
    cmp al, 9
    je tdone
    inc r14
    jmp tscan
tdone:
    mov rcx, r14
    sub rcx, r10
    cmp rcx, 1
    jne try_number
    movzx eax, byte ptr [r10]
    cmp al, 43
    je do_add
    cmp al, 45
    je do_sub
    cmp al, 42
    je do_mul
    cmp al, 47
    je do_div
    cmp al, 37
    je do_mod

try_number:
    mov rsi, r10
    xor r9d, r9d
    movzx eax, byte ptr [rsi]
    cmp al, 45
    jne no_neg
    mov r9d, 1
    inc rsi
no_neg:
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jle print_error
    cmp rdx, 18
    jg print_error
    xor r8, r8
digloop:
    movzx ecx, byte ptr [rsi]
    sub ecx, 48
    cmp ecx, 9
    ja print_error
    lea r8, [r8 + r8*4]
    lea r8, [r8*2 + rcx]
    inc rsi
    cmp rsi, r14
    jb digloop
    test r9d, r9d
    jz num_push
    neg r8
num_push:
    mov [r15 + rbp*8], r8
    inc rbp
    jmp tok_loop

do_add:
    cmp rbp, 2
    jl print_error
    dec rbp
    mov rax, [r15 + rbp*8]
    dec rbp
    mov rdx, [r15 + rbp*8]
    add rdx, rax
    mov [r15 + rbp*8], rdx
    inc rbp
    jmp tok_loop

do_sub:
    cmp rbp, 2
    jl print_error
    dec rbp
    mov rax, [r15 + rbp*8]
    dec rbp
    mov rdx, [r15 + rbp*8]
    sub rdx, rax
    mov [r15 + rbp*8], rdx
    inc rbp
    jmp tok_loop

do_mul:
    cmp rbp, 2
    jl print_error
    dec rbp
    mov rax, [r15 + rbp*8]
    dec rbp
    mov rcx, [r15 + rbp*8]
    imul rax, rcx
    mov [r15 + rbp*8], rax
    inc rbp
    jmp tok_loop

do_div:
    cmp rbp, 2
    jl print_error
    dec rbp
    mov rcx, [r15 + rbp*8]
    dec rbp
    mov rax, [r15 + rbp*8]
    test rcx, rcx
    je print_error
    cmp rcx, -1
    jne div_ok
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je print_error
div_ok:
    cqo
    idiv rcx
    mov [r15 + rbp*8], rax
    inc rbp
    jmp tok_loop

do_mod:
    cmp rbp, 2
    jl print_error
    dec rbp
    mov rcx, [r15 + rbp*8]
    dec rbp
    mov rax, [r15 + rbp*8]
    test rcx, rcx
    je print_error
    cmp rcx, -1
    jne mod_ok
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je print_error
mod_ok:
    cqo
    idiv rcx
    mov [r15 + rbp*8], rdx
    inc rbp
    jmp tok_loop

tok_end:
    cmp rbp, 1
    jne print_error
    mov rax, [r15]
    mov rdi, [rip + outpos]
    test rax, rax
    jns pn_pos
    mov byte ptr [rdi], 45
    inc rdi
    neg rax
pn_pos:
    lea rsi, [rip + numbuf + 32]
    mov rcx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz pn_loop
    lea r8, [rip + numbuf + 32]
pn_copy:
    cmp rsi, r8
    jae pn_copydone
    mov al, [rsi]
    mov [rdi], al
    inc rdi
    inc rsi
    jmp pn_copy
pn_copydone:
    mov byte ptr [rdi], 10
    inc rdi
    mov [rip + outpos], rdi
    jmp advance

print_error:
    mov rdi, [rip + outpos]
    mov byte ptr [rdi], 101
    mov byte ptr [rdi + 1], 114
    mov byte ptr [rdi + 2], 114
    mov byte ptr [rdi + 3], 111
    mov byte ptr [rdi + 4], 114
    mov byte ptr [rdi + 5], 10
    add rdi, 6
    mov [rip + outpos], rdi
    jmp advance

advance:
    cmp r12, r13
    jae set_p_end
    lea rbx, [r12 + 1]
    jmp line_loop
set_p_end:
    mov rbx, r13
    jmp line_loop

all_done:
    lea rsi, [rip + outbuf]
    mov rbx, [rip + outpos]
wloop:
    cmp rsi, rbx
    jae wdone
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    sub rdx, rsi
    syscall
    test rax, rax
    jle wdone
    add rsi, rax
    jmp wloop
wdone:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 8
outpos:
    .zero 8
inbuf:
    .zero 65536
outbuf:
    .zero 400000
numbuf:
    .zero 32
vstack:
    .zero 8192
