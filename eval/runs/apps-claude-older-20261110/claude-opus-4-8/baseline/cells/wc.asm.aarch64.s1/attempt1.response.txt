.globl _start

.text
_start:
    mov     x19, #0             // lines
    mov     x20, #0             // words
    mov     x21, #0             // bytes
    mov     x22, #0             // in_word flag

read_loop:
    mov     x0, #0
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x2, #65536
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     done_read
    add     x21, x21, x0        // bytes += count
    adrp    x3, buf
    add     x3, x3, :lo12:buf
    mov     x4, #0

proc_loop:
    cmp     x4, x0
    bge     read_loop
    ldrb    w5, [x3, x4]
    add     x4, x4, #1
    cmp     w5, #0x0A
    bne     1f
    add     x19, x19, #1
1:
    cmp     w5, #0x20
    beq     is_ws
    sub     w6, w5, #0x09
    cmp     w6, #4
    bls     is_ws
    cbnz    x22, proc_loop
    mov     x22, #1
    add     x20, x20, #1
    b       proc_loop
is_ws:
    mov     x22, #0
    b       proc_loop

done_read:
    adrp    x23, outbuf
    add     x23, x23, :lo12:outbuf
    mov     x1, x23
    mov     x0, x19
    bl      num2str
    mov     w6, #0x20
    strb    w6, [x1], #1
    mov     x0, x20
    bl      num2str
    mov     w6, #0x20
    strb    w6, [x1], #1
    mov     x0, x21
    bl      num2str
    mov     w6, #0x0A
    strb    w6, [x1], #1

    sub     x2, x1, x23
    mov     x0, #1
    mov     x1, x23
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0

// num2str: x0 = value, x1 = dest pointer; returns advanced pointer in x1
num2str:
    mov     x9, x1
    mov     x10, #10
    cbnz    x0, 1f
    mov     w11, #'0'
    strb    w11, [x9], #1
    mov     x1, x9
    ret
1:
    sub     sp, sp, #32
    mov     x12, sp
    mov     x13, x12
2:
    udiv    x14, x0, x10
    msub    x15, x14, x10, x0
    add     w15, w15, #'0'
    strb    w15, [x13], #1
    mov     x0, x14
    cbnz    x0, 2b
3:
    sub     x13, x13, #1
    ldrb    w15, [x13]
    strb    w15, [x9], #1
    cmp     x13, x12
    bhi     3b
    add     sp, sp, #32
    mov     x1, x9
    ret

.bss
.align 4
buf:
    .zero 65536
outbuf:
    .zero 64
