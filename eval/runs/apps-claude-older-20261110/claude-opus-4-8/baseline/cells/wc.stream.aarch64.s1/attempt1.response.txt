.globl _start

.text
_start:
    // read all input into buffer
    adrp    x19, buf
    add     x19, x19, :lo12:buf
    mov     x20, #0              // total bytes read
read_loop:
    mov     x8, #63              // read
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    // x20 = byte count
    // count lines and words
    mov     x21, #0              // lines
    mov     x22, #0              // words
    mov     x23, #0              // in_word flag
    mov     x24, #0              // index
count_loop:
    cmp     x24, x20
    beq     count_done
    ldrb    w0, [x19, x24]
    add     x24, x24, #1
    // newline?
    cmp     w0, #0x0A
    bne     1f
    add     x21, x21, #1
1:
    // whitespace check
    cmp     w0, #0x20
    beq     is_ws
    cmp     w0, #0x09
    beq     is_ws
    cmp     w0, #0x0A
    beq     is_ws
    cmp     w0, #0x0B
    beq     is_ws
    cmp     w0, #0x0C
    beq     is_ws
    cmp     w0, #0x0D
    beq     is_ws
    // word byte
    cbnz    x23, count_loop
    mov     x23, #1
    add     x22, x22, #1
    b       count_loop
is_ws:
    mov     x23, #0
    b       count_loop
count_done:
    // print lines, words, bytes separated by spaces, newline
    mov     x0, x21
    bl      print_num
    bl      print_space
    mov     x0, x22
    bl      print_num
    bl      print_space
    mov     x0, x20
    bl      print_num
    bl      print_newline
    // exit
    mov     x8, #93
    mov     x0, #0
    svc     #0

// print_num: print x0 in decimal
print_num:
    adrp    x1, numbuf
    add     x1, x1, :lo12:numbuf
    add     x1, x1, #32          // end pointer
    mov     x2, #0               // length
    mov     x3, #10
    // handle zero
    cbnz    x0, 2f
    mov     w4, #'0'
    sub     x1, x1, #1
    strb    w4, [x1]
    mov     x2, #1
    b       3f
2:
    udiv    x4, x0, x3
    msub    x5, x4, x3, x0       // remainder
    add     w5, w5, #'0'
    sub     x1, x1, #1
    strb    w5, [x1]
    add     x2, x2, #1
    mov     x0, x4
    cbnz    x0, 2b
3:
    // write
    mov     x8, #64
    mov     x0, #1
    svc     #0
    ret

print_space:
    adrp    x1, spacebuf
    add     x1, x1, :lo12:spacebuf
    mov     x8, #64
    mov     x0, #1
    mov     x2, #1
    svc     #0
    ret

print_newline:
    adrp    x1, nlbuf
    add     x1, x1, :lo12:nlbuf
    mov     x8, #64
    mov     x0, #1
    mov     x2, #1
    svc     #0
    ret

.section .rodata
spacebuf:
    .ascii " "
nlbuf:
    .ascii "\n"

.bss
.align 4
buf:
    .zero 65536
numbuf:
    .zero 32
