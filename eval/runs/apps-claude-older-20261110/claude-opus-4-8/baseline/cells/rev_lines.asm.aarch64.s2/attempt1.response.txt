.globl _start

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0                 // total read

read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x22, #0                 // outlen
    mov     x23, #0                 // line_start
    mov     x24, #0                 // si

scan_loop:
    cmp     x24, x20
    bge     scan_end
    ldrb    w0, [x19, x24]
    cmp     w0, #10
    bne     scan_next
    // process line [x23, x24)
    bl      emit_line
    add     x23, x24, #1
scan_next:
    add     x24, x24, #1
    b       scan_loop

scan_end:
    cmp     x23, x20
    bge     write_out
    mov     x24, x20
    bl      emit_line

write_out:
    mov     x0, #1
    mov     x1, x21
    mov     x2, x22
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0

// emit_line: reverse bytes [x23, x24) into output, then newline
// clobbers x0, x1, x2
emit_line:
    mov     x1, x24                 // j = end
el_loop:
    cmp     x1, x23
    ble     el_nl
    sub     x1, x1, #1
    ldrb    w0, [x19, x1]
    strb    w0, [x21, x22]
    add     x22, x22, #1
    b       el_loop
el_nl:
    mov     w0, #10
    strb    w0, [x21, x22]
    add     x22, x22, #1
    ret

.bss
.align 4
inbuf:
    .zero   65536
outbuf:
    .zero   65537
