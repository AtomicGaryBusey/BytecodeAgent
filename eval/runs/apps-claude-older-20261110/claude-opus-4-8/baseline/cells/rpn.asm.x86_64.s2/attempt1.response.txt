.text
.globl _start

_start:
    xor r13, r13
.readloop:
    mov rdx, 65536
    sub rdx, r13
    test rdx, rdx
    jz .readdone
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    syscall
    test rax, rax
    jle .readdone
    add r13, rax
    jmp .readloop
.readdone:
    lea rbx, [rip + inbuf]
    lea r12, [rbx + r13]
    lea r15, [rip + outbuf]

.lineloop:
    cmp rbx, r12
    jae .done
    xor r14, r14

.tokloop:
    cmp rbx, r12
    jae .endline
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je .sep
    cmp al, 0x09
    je .sep
    cmp al, 0x0A
    je .endline
    mov rsi, rbx
.scanend:
    cmp rbx, r12
    jae .haveend
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je .haveend
    cmp al, 0x09
    je .haveend
    cmp al, 0x0A
    je .haveend
    inc rbx
    jmp .scanend
.haveend:
    mov rcx, rbx
    sub rcx, rsi
    cmp rcx, 1
    jne .tryint
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je .op
    cmp al, '-'
    je .op
    cmp al, '*'
    je .op
    cmp al, '/'
    je .op
    cmp al, '%'
    je .op
.tryint:
    mov rdi, rsi
    xor r8d, r8d
    movzx eax, byte ptr [rdi]
    cmp al, '-'
    jne .nodash
    mov r8d, 1
    inc rdi
.nodash:
    mov rdx, rbx
    sub rdx, rdi
    test rdx, rdx
    jle .error
    cmp rdx, 18
    jg .error
    xor r9, r9
.digloop:
    cmp rdi, rbx
    jae .digdone
    movzx eax, byte ptr [rdi]
    sub eax, '0'
    cmp eax, 9
    ja .error
    lea r9, [r9 + r9*4]
    add r9, r9
    add r9, rax
    inc rdi
    jmp .digloop
.digdone:
    test r8d, r8d
    jz .pushv
    neg r9
.pushv:
    mov qword ptr [valstack + r14*8], r9
    inc r14
    jmp .tokloop

.sep:
    inc rbx
    jmp .tokloop

.op:
    cmp r14, 2
    jb .error
    mov rcx, r14
    mov r10, [valstack + rcx*8 - 8]
    mov r9, [valstack + rcx*8 - 16]
    cmp al, '+'
    je .doadd
    cmp al, '-'
    je .dosub
    cmp al, '*'
    je .domul
    cmp al, '/'
    je .dodiv
    jmp .domod
.doadd:
    add r9, r10
    jmp .opstore
.dosub:
    sub r9, r10
    jmp .opstore
.domul:
    imul r9, r10
    jmp .opstore
.dodiv:
    test r10, r10
    jz .error
    mov r11, 0x8000000000000000
    cmp r9, r11
    jne .divok
    cmp r10, -1
    je .error
.divok:
    mov rax, r9
    cqo
    idiv r10
    mov r9, rax
    jmp .opstore
.domod:
    test r10, r10
    jz .error
    mov r11, 0x8000000000000000
    cmp r9, r11
    jne .modok
    cmp r10, -1
    je .error
.modok:
    mov rax, r9
    cqo
    idiv r10
    mov r9, rdx
    jmp .opstore
.opstore:
    dec r14
    mov [valstack + r14*8 - 8], r9
    jmp .tokloop

.endline:
    cmp rbx, r12
    jae .checkout
    inc rbx
.checkout:
    cmp r14, 1
    jne .doerrout
    mov r9, qword ptr [valstack]
    jmp .outnum

.error:
.skiploop:
    cmp rbx, r12
    jae .doerrout
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 0x0A
    jne .skiploop
.doerrout:
    mov byte ptr [r15 + 0], 'e'
    mov byte ptr [r15 + 1], 'r'
    mov byte ptr [r15 + 2], 'r'
    mov byte ptr [r15 + 3], 'o'
    mov byte ptr [r15 + 4], 'r'
    mov byte ptr [r15 + 5], 0x0A
    add r15, 6
    jmp .lineloop

.outnum:
    test r9, r9
    jns .nonneg
    mov byte ptr [r15], '-'
    inc r15
    neg r9
.nonneg:
    mov rax, r9
    lea rcx, [numbuf + 24]
    test rax, rax
    jnz .convloop
    dec rcx
    mov byte ptr [rcx], '0'
    jmp .convwrite
.convloop:
    test rax, rax
    jz .convwrite
    xor edx, edx
    mov r8, 10
    div r8
    add dl, '0'
    dec rcx
    mov [rcx], dl
    jmp .convloop
.convwrite:
    lea r8, [numbuf + 24]
.cw:
    cmp rcx, r8
    jae .cwdone
    mov al, [rcx]
    mov [r15], al
    inc r15
    inc rcx
    jmp .cw
.cwdone:
    mov byte ptr [r15], 0x0A
    inc r15
    jmp .lineloop

.done:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
.wloop:
    test rdx, rdx
    jz .exit
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .exit
    add rsi, rax
    sub rdx, rax
    jmp .wloop
.exit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 65536
valstack:
    .zero 16384
numbuf:
    .zero 32
outbuf:
    .zero 800000
