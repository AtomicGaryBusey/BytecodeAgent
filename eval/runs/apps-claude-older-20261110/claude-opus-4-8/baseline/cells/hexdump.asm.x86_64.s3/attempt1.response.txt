.text
.globl _start
_start:
    lea rsi, [rip + inbuf]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov edx, 65536
    sub edx, r13d
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r12d, r12d

row_loop:
    cmp r12, r13
    jae rows_done
    mov r14, r13
    sub r14, r12
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    lea rbx, [rsi + r12]
    mov eax, r12d
    call emit8hex
    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r15d, r15d
hex_pos_loop:
    cmp r15, r14
    jae short_fill
    movzx eax, byte ptr [rbx + r15]
    call emit2hex
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp afterpos
short_fill:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
afterpos:
    cmp r15, 7
    jne nospace
    mov byte ptr [rdi], 0x20
    inc rdi
nospace:
    inc r15
    cmp r15, 16
    jb hex_pos_loop

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2

    xor r15d, r15d
ascii_loop:
    cmp r15, r14
    jae ascii_done
    movzx eax, byte ptr [rbx + r15]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    ja dot
    jmp putc
dot:
    mov al, 0x2e
putc:
    mov [rdi], al
    inc rdi
    inc r15
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    add r12, 16
    jmp row_loop

rows_done:
    mov eax, r13d
    call emit8hex
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit8hex:
    mov ecx, 8
e8:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [hextable + rdx]
    mov [rdi], dl
    inc rdi
    dec ecx
    jnz e8
    ret

emit2hex:
    movzx eax, al
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [hextable + rdx]
    mov [rdi], dl
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [hextable + rdx]
    mov [rdi + 1], dl
    add rdi, 2
    ret

.section .rodata
hextable:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 400000
