.intel_syntax noprefix
.globl _start

.text
_start:
    lea rbp, [rip + hextab]
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov eax, 0
    xor edi, edi
    mov edx, 65536
    sub edx, r12d
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    cmp r12, 65536
    jb read_loop
done_read:
    xor r13, r13
row_loop:
    cmp r13, r12
    jae rows_done
    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe 3f
    mov r15d, 16
3:
    lea rbx, [rip + linebuf]
    mov eax, r13d
    mov ecx, 8
hexoff:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    mov [rbx], dl
    inc rbx
    dec ecx
    jnz hexoff
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx+1], 0x20
    add rbx, 2
    xor r14, r14
byteloop:
    cmp r14, 16
    jae bytedone
    cmp r14, r15
    jb emitbyte
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx+1], 0x20
    mov byte ptr [rbx+2], 0x20
    add rbx, 3
    jmp afterslot
emitbyte:
    movzx edx, byte ptr [inbuf + r13 + r14]
    mov eax, edx
    shr eax, 4
    and eax, 15
    mov al, byte ptr [rbp + rax]
    mov [rbx], al
    inc rbx
    mov eax, edx
    and eax, 15
    mov al, byte ptr [rbp + rax]
    mov [rbx], al
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
afterslot:
    cmp r14, 7
    jne 2f
    mov byte ptr [rbx], 0x20
    inc rbx
2:
    inc r14
    jmp byteloop
bytedone:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx+1], 0x7c
    add rbx, 2
    xor r14, r14
charloop:
    cmp r14, r15
    jae chardone
    movzx edx, byte ptr [inbuf + r13 + r14]
    cmp edx, 0x20
    jb dot
    cmp edx, 0x7e
    ja dot
    jmp storechar
dot:
    mov edx, 0x2e
storechar:
    mov [rbx], dl
    inc rbx
    inc r14
    jmp charloop
chardone:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx+1], 0x0a
    add rbx, 2
    lea rsi, [rip + linebuf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add r13, 16
    jmp row_loop
rows_done:
    lea rbx, [rip + linebuf]
    mov eax, r12d
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbp + rdx]
    mov [rbx], dl
    inc rbx
    dec ecx
    jnz 1b
    mov byte ptr [rbx], 0x0a
    inc rbx
    lea rsi, [rip + linebuf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.align 16
linebuf:
    .zero 128
inbuf:
    .zero 65536
