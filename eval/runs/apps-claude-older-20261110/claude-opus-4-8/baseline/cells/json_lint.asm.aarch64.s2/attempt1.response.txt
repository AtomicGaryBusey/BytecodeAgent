.text
.globl _start

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
rd_loop:
    mov     x0, #0
    add     x1, x19, x20
    movz    x2, #1, lsl #16
    subs    x2, x2, x20
    beq     rd_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     rd_done
    add     x20, x20, x0
    b       rd_loop
rd_done:
    mov     x0, #0
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x0, x20
    bcc     main_err
    b       ok_out
main_err:
    b       do_error

// skip_ws: x0 = index, x19 = base, x20 = len. returns x0 after ws.
skip_ws:
ws_loop:
    cmp     x0, x20
    bcs     ws_done
    ldrb    w1, [x19, x0]
    cmp     w1, #0x20
    beq     ws_adv
    cmp     w1, #0x09
    beq     ws_adv
    cmp     w1, #0x0A
    beq     ws_adv
    cmp     w1, #0x0D
    beq     ws_adv
    b       ws_done
ws_adv:
    add     x0, x0, #1
    b       ws_loop
ws_done:
    ret

// parse_value: x0 = index. returns x0 after value, or jumps to do_error.
parse_value:
    cmp     x0, x20
    bcs     pv_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x7B
    beq     parse_object
    cmp     w1, #0x5B
    beq     parse_array
    cmp     w1, #0x22
    beq     parse_string
    cmp     w1, #0x74
    beq     parse_true
    cmp     w1, #0x66
    beq     parse_false
    cmp     w1, #0x6E
    beq     parse_null
    cmp     w1, #0x2D
    beq     parse_number
    cmp     w1, #0x30
    bcc     pv_err
    cmp     w1, #0x39
    bls     parse_number
pv_err:
    b       do_error

parse_true:
    adrp    x3, lit_true
    add     x3, x3, :lo12:lit_true
    mov     x4, #4
    b       match_lit
parse_false:
    adrp    x3, lit_false
    add     x3, x3, :lo12:lit_false
    mov     x4, #5
    b       match_lit
parse_null:
    adrp    x3, lit_null
    add     x3, x3, :lo12:lit_null
    mov     x4, #4
    b       match_lit

// match_lit: x0=index, x3=literal addr, x4=len
match_lit:
    mov     x5, #0
ml_loop:
    cmp     x5, x4
    bcs     ml_done
    cmp     x0, x20
    bcs     ml_err
    ldrb    w6, [x19, x0]
    ldrb    w7, [x3, x5]
    cmp     w6, w7
    bne     ml_err
    add     x0, x0, #1
    add     x5, x5, #1
    b       ml_loop
ml_err:
    b       do_error
ml_done:
    ret

// parse_string: x0 points at opening quote
parse_string:
    add     x0, x0, #1
str_loop:
    cmp     x0, x20
    bcs     str_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x22
    beq     str_close
    cmp     w1, #0x5C
    beq     str_esc
    cmp     w1, #0x20
    bcc     str_err
    add     x0, x0, #1
    b       str_loop
str_close:
    add     x0, x0, #1
    ret
str_err:
    b       do_error
str_esc:
    add     x0, x0, #1
    cmp     x0, x20
    bcs     str_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x22
    beq     str_esc_s
    cmp     w1, #0x5C
    beq     str_esc_s
    cmp     w1, #0x2F
    beq     str_esc_s
    cmp     w1, #0x62
    beq     str_esc_s
    cmp     w1, #0x66
    beq     str_esc_s
    cmp     w1, #0x6E
    beq     str_esc_s
    cmp     w1, #0x72
    beq     str_esc_s
    cmp     w1, #0x74
    beq     str_esc_s
    cmp     w1, #0x75
    beq     str_esc_u
    b       str_err
str_esc_s:
    add     x0, x0, #1
    b       str_loop
str_esc_u:
    add     x0, x0, #1
    mov     x5, #4
str_hex:
    cbz     x5, str_loop
    cmp     x0, x20
    bcs     str_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x30
    bcc     str_err
    cmp     w1, #0x39
    bls     str_hex_ok
    cmp     w1, #0x61
    bcc     str_hex_A
    cmp     w1, #0x66
    bls     str_hex_ok
    b       str_err
str_hex_A:
    cmp     w1, #0x41
    bcc     str_err
    cmp     w1, #0x46
    bhi     str_err
str_hex_ok:
    add     x0, x0, #1
    sub     x5, x5, #1
    b       str_hex

// parse_number: x0 points at '-' or digit
parse_number:
    ldrb    w1, [x19, x0]
    cmp     w1, #0x2D
    bne     num_int
    add     x0, x0, #1
num_int:
    cmp     x0, x20
    bcs     num_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x30
    beq     num_zero
    cmp     w1, #0x31
    bcc     num_err
    cmp     w1, #0x39
    bhi     num_err
    add     x0, x0, #1
num_digs:
    cmp     x0, x20
    bcs     num_frac
    ldrb    w1, [x19, x0]
    cmp     w1, #0x30
    bcc     num_frac
    cmp     w1, #0x39
    bhi     num_frac
    add     x0, x0, #1
    b       num_digs
num_zero:
    add     x0, x0, #1
num_frac:
    cmp     x0, x20
    bcs     num_done
    ldrb    w1, [x19, x0]
    cmp     w1, #0x2E
    bne     num_exp
    add     x0, x0, #1
    cmp     x0, x20
    bcs     num_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x30
    bcc     num_err
    cmp     w1, #0x39
    bhi     num_err
    add     x0, x0, #1
num_fdigs:
    cmp     x0, x20
    bcs     num_exp
    ldrb    w1, [x19, x0]
    cmp     w1, #0x30
    bcc     num_exp
    cmp     w1, #0x39
    bhi     num_exp
    add     x0, x0, #1
    b       num_fdigs
num_exp:
    cmp     x0, x20
    bcs     num_done
    ldrb    w1, [x19, x0]
    cmp     w1, #0x65
    beq     num_edo
    cmp     w1, #0x45
    beq     num_edo
    b       num_done
num_edo:
    add     x0, x0, #1
    cmp     x0, x20
    bcs     num_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x2B
    beq     num_esign
    cmp     w1, #0x2D
    beq     num_esign
    b       num_echk
num_esign:
    add     x0, x0, #1
    cmp     x0, x20
    bcs     num_err
    ldrb    w1, [x19, x0]
num_echk:
    cmp     w1, #0x30
    bcc     num_err
    cmp     w1, #0x39
    bhi     num_err
    add     x0, x0, #1
num_edigs:
    cmp     x0, x20
    bcs     num_done
    ldrb    w1, [x19, x0]
    cmp     w1, #0x30
    bcc     num_done
    cmp     w1, #0x39
    bhi     num_done
    add     x0, x0, #1
    b       num_edigs
num_done:
    ret
num_err:
    b       do_error

// parse_array: x0 points at '['
parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x0, x0, #1
    bl      skip_ws
    cmp     x0, x20
    bcs     arr_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x5D
    beq     arr_end
arr_loop:
    bl      parse_value
    bl      skip_ws
    cmp     x0, x20
    bcs     arr_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x5D
    beq     arr_end
    cmp     w1, #0x2C
    bne     arr_err
    add     x0, x0, #1
    bl      skip_ws
    b       arr_loop
arr_end:
    add     x0, x0, #1
    ldp     x29, x30, [sp], #16
    ret
arr_err:
    b       do_error

// parse_object: x0 points at '{'
parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x0, x0, #1
    bl      skip_ws
    cmp     x0, x20
    bcs     obj_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x7D
    beq     obj_end
obj_loop:
    cmp     x0, x20
    bcs     obj_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x22
    bne     obj_err
    bl      parse_string
    bl      skip_ws
    cmp     x0, x20
    bcs     obj_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x3A
    bne     obj_err
    add     x0, x0, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x0, x20
    bcs     obj_err
    ldrb    w1, [x19, x0]
    cmp     w1, #0x7D
    beq     obj_end
    cmp     w1, #0x2C
    bne     obj_err
    add     x0, x0, #1
    bl      skip_ws
    b       obj_loop
obj_end:
    add     x0, x0, #1
    ldp     x29, x30, [sp], #16
    ret
obj_err:
    b       do_error

do_error:
    mov     x19, x0
    mov     x0, #1
    adrp    x1, msg_err
    add     x1, x1, :lo12:msg_err
    mov     x2, #9
    mov     x8, #64
    svc     #0
    adrp    x3, numbuf
    add     x3, x3, :lo12:numbuf
    add     x4, x3, #24
    mov     x5, x4
    mov     x0, x19
    mov     x6, #10
de_conv:
    udiv    x7, x0, x6
    msub    x9, x7, x6, x0
    add     x9, x9, #0x30
    sub     x5, x5, #1
    strb    w9, [x5]
    mov     x0, x7
    cbnz    x0, de_conv
    mov     x0, #1
    mov     x1, x5
    sub     x2, x4, x5
    mov     x8, #64
    svc     #0
    mov     x0, #1
    adrp    x1, msg_nl
    add     x1, x1, :lo12:msg_nl
    mov     x2, #1
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0

ok_out:
    mov     x0, #1
    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #3
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
msg_err:  .ascii "error at "
msg_ok:   .ascii "ok\n"
msg_nl:   .ascii "\n"
lit_true: .ascii "true"
lit_false:.ascii "false"
lit_null: .ascii "null"

.bss
.balign 16
inbuf:  .zero 65536
numbuf: .zero 32
