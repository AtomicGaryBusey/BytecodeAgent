.globl _start

.text
_start:
    xor rbx, rbx
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rdx, 40000
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:

    lea r12, [rip + starts]
    lea r13, [rip + lens]
    lea r8, [rip + inbuf]
    xor r14, r14
    xor rcx, rcx
    xor r9, r9
parse_loop:
    cmp rcx, rbx
    jge parse_tail
    movzx eax, byte ptr [r8 + rcx]
    cmp al, 0x0A
    jne parse_next
    mov rdi, r8
    add rdi, r9
    mov [r12 + r14*8], rdi
    mov rsi, rcx
    sub rsi, r9
    mov [r13 + r14*8], rsi
    inc r14
    lea r9, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_tail:
    cmp r9, rbx
    jge parse_done
    mov rdi, r8
    add rdi, r9
    mov [r12 + r14*8], rdi
    mov rsi, rbx
    sub rsi, r9
    mov [r13 + r14*8], rsi
    inc r14
parse_done:

    mov rbx, 1
outer:
    cmp rbx, r14
    jge sort_done
    mov r15, [r12 + rbx*8]
    mov r11, [r13 + rbx*8]
    mov rbp, rbx
    dec rbp
inner:
    cmp rbp, 0
    jl inner_done
    mov rdi, [r12 + rbp*8]
    mov rsi, [r13 + rbp*8]
    mov rdx, r15
    mov rcx, r11
    call compare
    cmp rax, 0
    jle inner_done
    mov rax, [r12 + rbp*8]
    mov [r12 + rbp*8 + 8], rax
    mov rax, [r13 + rbp*8]
    mov [r13 + rbp*8 + 8], rax
    dec rbp
    jmp inner
inner_done:
    mov [r12 + rbp*8 + 8], r15
    mov [r13 + rbp*8 + 8], r11
    inc rbx
    jmp outer
sort_done:

    xor rbx, rbx
outloop:
    cmp rbx, r14
    jge end
    mov rsi, [r12 + rbx*8]
    mov rdx, [r13 + rbx*8]
    mov rax, 1
    mov rdi, 1
    syscall
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rax, 1
    mov rdi, 1
    syscall
    inc rbx
    jmp outloop
end:
    mov rax, 60
    xor rdi, rdi
    syscall

compare:
    mov rax, rsi
    cmp rax, rcx
    jbe 1f
    mov rax, rcx
1:
    xor r8, r8
2:
    cmp r8, rax
    jae 3f
    movzx r9d, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp r9d, r10d
    jne 4f
    inc r8
    jmp 2b
3:
    mov rax, rsi
    sub rax, rcx
    ret
4:
    sub r9d, r10d
    movsxd rax, r9d
    ret

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 8
inbuf:
    .zero 40000
starts:
    .zero 8000
lens:
    .zero 8000
