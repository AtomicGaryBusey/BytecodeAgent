.text
.globl _start
_start:
    lea rbp, [rip + hextab]
    lea r10, [rip + inbuf]
    xor r15d, r15d

read_loop:
    mov rdx, 65536
    sub rdx, r15
    jz read_done
    xor edi, edi
    mov rsi, r10
    add rsi, r15
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:

    lea r14, [rip + outbuf]
    xor ebx, ebx

row_loop:
    cmp rbx, r15
    jae rows_done

    mov rax, r15
    sub rax, rbx
    cmp rax, 16
    jbe 1f
    mov eax, 16
1:
    mov r13, rax

    mov r11, r10
    add r11, rbx

    mov r8d, ebx
    mov ecx, 8
2:
    rol r8d, 4
    mov edx, r8d
    and edx, 15
    movzx edx, byte ptr [rbp + rdx]
    mov [r14], dl
    inc r14
    dec ecx
    jnz 2b

    mov word ptr [r14], 0x2020
    add r14, 2

    xor r12d, r12d
3:
    cmp r12, 16
    jae 4f
    cmp r12, r13
    jb 5f
    mov word ptr [r14], 0x2020
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
    jmp 6f
5:
    movzx eax, byte ptr [r11 + r12]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [rbp + rdx]
    mov [r14], dl
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [rbp + rdx]
    mov [r14 + 1], dl
    mov byte ptr [r14 + 2], 0x20
    add r14, 3
6:
    cmp r12, 7
    jne 7f
    mov byte ptr [r14], 0x20
    inc r14
7:
    inc r12
    jmp 3b
4:
    mov byte ptr [r14], 0x20
    mov byte ptr [r14 + 1], 0x7c
    add r14, 2

    xor r12d, r12d
8:
    cmp r12, r13
    jae 9f
    movzx eax, byte ptr [r11 + r12]
    cmp al, 0x20
    jb putdot
    cmp al, 0x7e
    ja putdot
    jmp putc
putdot:
    mov al, 0x2e
putc:
    mov [r14], al
    inc r14
    inc r12
    jmp 8b
9:
    mov byte ptr [r14], 0x7c
    mov byte ptr [r14 + 1], 0x0a
    add r14, 2

    add rbx, 16
    jmp row_loop

rows_done:
    mov r8d, r15d
    mov ecx, 8
10:
    rol r8d, 4
    mov edx, r8d
    and edx, 15
    movzx edx, byte ptr [rbp + rdx]
    mov [r14], dl
    inc r14
    dec ecx
    jnz 10b
    mov byte ptr [r14], 0x0a
    inc r14

    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 400000
