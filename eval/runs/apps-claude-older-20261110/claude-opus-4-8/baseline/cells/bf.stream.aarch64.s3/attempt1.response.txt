.globl _start

.text
_start:
    // read all stdin into inbuf
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0              // total bytes read
read_loop:
    mov     x2, #65536
    sub     x2, x2, x20          // remaining
    cbz     x2, read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    mov     x26, x20             // total

    // find first '!' (0x21)
    mov     x1, #0
1:
    cmp     x1, x26
    beq     2f
    ldrb    w2, [x19, x1]
    cmp     w2, #0x21
    beq     3f
    add     x1, x1, #1
    b       1b
2:  // no bang
    mov     x20, x26             // proglen
    add     x23, x19, x26        // input ptr = end
    add     x24, x19, x26        // input end
    b       4f
3:  // bang at x1
    mov     x20, x1              // proglen
    add     x23, x19, x1
    add     x23, x23, #1         // input start
    add     x24, x19, x26        // input end
4:

    // build jump table and validate brackets
    adrp    x25, jumptab
    add     x25, x25, :lo12:jumptab
    adrp    x27, bstack
    add     x27, x27, :lo12:bstack
    mov     x28, #0              // stack depth
    mov     x1, #0
5:
    cmp     x1, x20
    beq     6f
    ldrb    w2, [x19, x1]
    cmp     w2, #0x5b            // '['
    beq     push_open
    cmp     w2, #0x5d            // ']'
    beq     pop_close
    b       cont
push_open:
    str     w1, [x27, x28, lsl #2]
    add     x28, x28, #1
    b       cont
pop_close:
    cbz     x28, unbalanced
    sub     x28, x28, #1
    ldr     w4, [x27, x28, lsl #2]        // open index
    str     w1, [x25, x4, lsl #2]         // jumptab[open] = close
    str     w4, [x25, x1, lsl #2]         // jumptab[close] = open
cont:
    add     x1, x1, #1
    b       5b
6:
    cbnz    x28, unbalanced

    // run interpreter
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    mov     x21, #0              // pc
run_loop:
    cmp     x21, x20
    beq     done
    ldrb    w2, [x19, x21]
    cmp     w2, #0x3e            // '>'
    beq     c_gt
    cmp     w2, #0x3c            // '<'
    beq     c_lt
    cmp     w2, #0x2b            // '+'
    beq     c_plus
    cmp     w2, #0x2d            // '-'
    beq     c_minus
    cmp     w2, #0x2e            // '.'
    beq     c_dot
    cmp     w2, #0x2c            // ','
    beq     c_comma
    cmp     w2, #0x5b            // '['
    beq     c_lb
    cmp     w2, #0x5d            // ']'
    beq     c_rb
    b       next
c_gt:
    add     x22, x22, #1
    b       next
c_lt:
    sub     x22, x22, #1
    b       next
c_plus:
    ldrb    w3, [x22]
    add     w3, w3, #1
    strb    w3, [x22]
    b       next
c_minus:
    ldrb    w3, [x22]
    sub     w3, w3, #1
    strb    w3, [x22]
    b       next
c_dot:
    mov     x8, #64
    mov     x0, #1
    mov     x1, x22
    mov     x2, #1
    svc     #0
    b       next
c_comma:
    cmp     x23, x24
    bhs     store_zero
    ldrb    w3, [x23]
    add     x23, x23, #1
    strb    w3, [x22]
    b       next
store_zero:
    strb    wzr, [x22]
    b       next
c_lb:
    ldrb    w3, [x22]
    cbnz    w3, next
    ldr     w3, [x25, x21, lsl #2]
    mov     x21, x3
    b       next
c_rb:
    ldrb    w3, [x22]
    cbz     w3, next
    ldr     w3, [x25, x21, lsl #2]
    mov     x21, x3
    b       next
next:
    add     x21, x21, #1
    b       run_loop
done:
    mov     x8, #93
    mov     x0, #0
    svc     #0
unbalanced:
    mov     x8, #93
    mov     x0, #1
    svc     #0

.bss
.balign 16
inbuf:
    .zero 65536
tape:
    .zero 30000
.balign 4
jumptab:
    .zero 262144
bstack:
    .zero 4096
