.text
.globl _start

_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, #0
.read_loop:
    mov x9, #0x10000
    sub x2, x9, x21
    cbz x2, .read_done
    mov x8, #63
    mov x0, #0
    add x1, x19, x21
    svc #0
    cmp x0, #0
    b.le .read_done
    add x21, x21, x0
    b .read_loop
.read_done:
    mov x20, #0
    bl skip_ws
    bl parse_value
    cbnz w0, .fail
    bl skip_ws
    cmp x20, x21
    b.ne .fail

    // ok
    mov x8, #64
    mov x0, #1
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x2, #3
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0

.fail:
    // write "error at "
    mov x8, #64
    mov x0, #1
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x2, #9
    svc #0
    // convert x20 to decimal in numbuf
    adrp x3, numbuf
    add x3, x3, :lo12:numbuf
    add x3, x3, #20
    mov x9, x20
    mov x10, #10
    cbnz x9, .conv_loop
    mov w11, #0x30
    sub x3, x3, #1
    strb w11, [x3]
    b .conv_done
.conv_loop:
    cbz x9, .conv_done
    udiv x12, x9, x10
    msub x13, x12, x10, x9
    add w13, w13, #0x30
    sub x3, x3, #1
    strb w13, [x3]
    mov x9, x12
    b .conv_loop
.conv_done:
    adrp x5, numbuf
    add x5, x5, :lo12:numbuf
    add x5, x5, #20
    sub x2, x5, x3
    mov x8, #64
    mov x0, #1
    mov x1, x3
    svc #0
    // newline
    mov x8, #64
    mov x0, #1
    adrp x1, msg_nl
    add x1, x1, :lo12:msg_nl
    mov x2, #1
    svc #0
    mov x8, #93
    mov x0, #1
    svc #0

// -------- skip_ws --------
skip_ws:
1:
    cmp x20, x21
    b.hs 2f
    ldrb w9, [x19, x20]
    cmp w9, #0x20
    b.eq 3f
    cmp w9, #0x09
    b.eq 3f
    cmp w9, #0x0A
    b.eq 3f
    cmp w9, #0x0D
    b.eq 3f
    b 2f
3:
    add x20, x20, #1
    b 1b
2:
    ret

// -------- parse_value --------
parse_value:
    cmp x20, x21
    b.hs .val_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x7B
    b.eq .val_obj
    cmp w9, #0x5B
    b.eq .val_arr
    cmp w9, #0x22
    b.eq .val_str
    cmp w9, #0x74
    b.eq .val_true
    cmp w9, #0x66
    b.eq .val_false
    cmp w9, #0x6E
    b.eq .val_null
    cmp w9, #0x2D
    b.eq .val_num
    sub w10, w9, #0x30
    cmp w10, #9
    b.ls .val_num
.val_fail:
    mov w0, #1
    ret
.val_obj:
    b parse_object
.val_arr:
    b parse_array
.val_str:
    b parse_string
.val_num:
    b parse_number
.val_true:
    adrp x0, lit_true
    add x0, x0, :lo12:lit_true
    b check_literal
.val_false:
    adrp x0, lit_false
    add x0, x0, :lo12:lit_false
    b check_literal
.val_null:
    adrp x0, lit_null
    add x0, x0, :lo12:lit_null
    b check_literal

// -------- check_literal (x0 = asciz) --------
check_literal:
1:
    ldrb w10, [x0], #1
    cbz w10, 2f
    cmp x20, x21
    b.hs 3f
    ldrb w9, [x19, x20]
    cmp w9, w10
    b.ne 3f
    add x20, x20, #1
    b 1b
2:
    mov w0, #0
    ret
3:
    mov w0, #1
    ret

// -------- parse_object --------
parse_object:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs .obj_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x7D
    b.eq .obj_close
.obj_member:
    bl parse_string
    cbnz w0, .obj_ret_fail
    bl skip_ws
    cmp x20, x21
    b.hs .obj_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x3A
    b.ne .obj_fail
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    cbnz w0, .obj_ret_fail
    bl skip_ws
    cmp x20, x21
    b.hs .obj_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2C
    b.eq .obj_comma
    cmp w9, #0x7D
    b.eq .obj_close
    b .obj_fail
.obj_comma:
    add x20, x20, #1
    bl skip_ws
    b .obj_member
.obj_close:
    add x20, x20, #1
    mov w0, #0
    ldp x29, x30, [sp], #16
    ret
.obj_fail:
    mov w0, #1
.obj_ret_fail:
    ldp x29, x30, [sp], #16
    ret

// -------- parse_array --------
parse_array:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs .arr_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x5D
    b.eq .arr_close
.arr_elem:
    bl parse_value
    cbnz w0, .arr_ret_fail
    bl skip_ws
    cmp x20, x21
    b.hs .arr_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2C
    b.eq .arr_comma
    cmp w9, #0x5D
    b.eq .arr_close
    b .arr_fail
.arr_comma:
    add x20, x20, #1
    bl skip_ws
    b .arr_elem
.arr_close:
    add x20, x20, #1
    mov w0, #0
    ldp x29, x30, [sp], #16
    ret
.arr_fail:
    mov w0, #1
.arr_ret_fail:
    ldp x29, x30, [sp], #16
    ret

// -------- parse_string --------
parse_string:
    cmp x20, x21
    b.hs .str_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x22
    b.ne .str_fail
    add x20, x20, #1
.str_loop:
    cmp x20, x21
    b.hs .str_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x22
    b.eq .str_end
    cmp w9, #0x5C
    b.eq .str_esc
    cmp w9, #0x20
    b.lo .str_fail
    add x20, x20, #1
    b .str_loop
.str_esc:
    add x20, x20, #1
    cmp x20, x21
    b.hs .str_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x22
    b.eq .str_simple
    cmp w9, #0x5C
    b.eq .str_simple
    cmp w9, #0x2F
    b.eq .str_simple
    cmp w9, #0x62
    b.eq .str_simple
    cmp w9, #0x66
    b.eq .str_simple
    cmp w9, #0x6E
    b.eq .str_simple
    cmp w9, #0x72
    b.eq .str_simple
    cmp w9, #0x74
    b.eq .str_simple
    cmp w9, #0x75
    b.eq .str_u
    b .str_fail
.str_simple:
    add x20, x20, #1
    b .str_loop
.str_u:
    add x20, x20, #1
    mov w10, #4
.str_u_loop:
    cmp x20, x21
    b.hs .str_fail
    ldrb w9, [x19, x20]
    sub w11, w9, #0x30
    cmp w11, #9
    b.ls .str_u_ok
    orr w11, w9, #0x20
    sub w11, w11, #0x61
    cmp w11, #5
    b.hi .str_fail
.str_u_ok:
    add x20, x20, #1
    subs w10, w10, #1
    b.ne .str_u_loop
    b .str_loop
.str_end:
    add x20, x20, #1
    mov w0, #0
    ret
.str_fail:
    mov w0, #1
    ret

// -------- parse_number --------
parse_number:
    cmp x20, x21
    b.hs .num_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2D
    b.ne .num_int
    add x20, x20, #1
.num_int:
    cmp x20, x21
    b.hs .num_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x30
    b.eq .num_zero
    sub w10, w9, #0x31
    cmp w10, #8
    b.hi .num_fail
    add x20, x20, #1
.num_digits1:
    cmp x20, x21
    b.hs .num_frac
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi .num_frac
    add x20, x20, #1
    b .num_digits1
.num_zero:
    add x20, x20, #1
.num_frac:
    cmp x20, x21
    b.hs .num_ok
    ldrb w9, [x19, x20]
    cmp w9, #0x2E
    b.ne .num_exp
    add x20, x20, #1
    cmp x20, x21
    b.hs .num_fail
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi .num_fail
    add x20, x20, #1
.num_frac_digits:
    cmp x20, x21
    b.hs .num_ok
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi .num_exp
    add x20, x20, #1
    b .num_frac_digits
.num_exp:
    cmp x20, x21
    b.hs .num_ok
    ldrb w9, [x19, x20]
    cmp w9, #0x65
    b.eq .num_exp2
    cmp w9, #0x45
    b.eq .num_exp2
    b .num_ok
.num_exp2:
    add x20, x20, #1
    cmp x20, x21
    b.hs .num_fail
    ldrb w9, [x19, x20]
    cmp w9, #0x2B
    b.eq .num_exp_sign
    cmp w9, #0x2D
    b.eq .num_exp_sign
    b .num_exp_digit0
.num_exp_sign:
    add x20, x20, #1
.num_exp_digit0:
    cmp x20, x21
    b.hs .num_fail
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi .num_fail
    add x20, x20, #1
.num_exp_digits:
    cmp x20, x21
    b.hs .num_ok
    ldrb w9, [x19, x20]
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi .num_ok
    add x20, x20, #1
    b .num_exp_digits
.num_ok:
    mov w0, #0
    ret
.num_fail:
    mov w0, #1
    ret

.section .rodata
msg_ok:   .ascii "ok\n"
msg_err:  .ascii "error at "
msg_nl:   .ascii "\n"
lit_true:  .asciz "true"
lit_false: .asciz "false"
lit_null:  .asciz "null"

.bss
.balign 16
buf:    .space 65536
numbuf: .space 24
