.text
.globl _start

_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz done_read
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    xor r13, r13
    xor r14, r14
    xor r8d, r8d
    xor rcx, rcx
    lea rbx, [rip + buf]
proc_loop:
    cmp rcx, r12
    jae proc_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x0A
    jne 5f
    inc r13
5:
    cmp al, 0x20
    je 6f
    cmp al, 0x09
    jb 7f
    cmp al, 0x0D
    jbe 6f
7:
    test r8d, r8d
    jnz 8f
    inc r14
    mov r8d, 1
8:
    jmp next
6:
    xor r8d, r8d
next:
    inc rcx
    jmp proc_loop
proc_done:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call itoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call itoa
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r12
    call itoa
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

itoa:
    mov rcx, 10
    xor r10, r10
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
1:
    xor edx, edx
    div rcx
    add dl, 0x30
    push rdx
    inc r10
    test rax, rax
    jnz 1b
2:
    pop rdx
    mov [rdi], dl
    inc rdi
    dec r10
    jnz 2b
    ret

outbuf:
