.section .rodata
nlbyte:
	.byte 0x0A

.bss
	.align 4
buf:
	.zero 32768
arr_ptr:
	.zero 8192
arr_len:
	.zero 8192

.text
.globl _start

// cmpf(x0=i, x1=j) -> w0 sign, compares lines
cmpf:
	ldr x2, [x19, x0, lsl #3]
	ldr x3, [x20, x0, lsl #3]
	ldr x4, [x19, x1, lsl #3]
	ldr x5, [x20, x1, lsl #3]
	cmp x3, x5
	csel x6, x3, x5, ls
	mov x7, #0
1:
	cmp x7, x6
	b.hs 2f
	ldrb w8, [x2, x7]
	ldrb w9, [x4, x7]
	subs w10, w8, w9
	b.ne 3f
	add x7, x7, #1
	b 1b
2:
	cmp x3, x5
	b.lo 4f
	b.hi 5f
	mov w0, #0
	ret
3:
	mov w0, w10
	ret
4:
	mov w0, #-1
	ret
5:
	mov w0, #1
	ret

// siftdown(x0=start, x1=end)
siftdown:
	stp x29, x30, [sp, #-64]!
	stp x21, x22, [sp, #16]
	stp x23, x24, [sp, #32]
	mov x21, x0
	mov x22, x1
sd_loop:
	lsl x23, x21, #1
	add x23, x23, #1
	cmp x23, x22
	b.hs sd_done
	add x24, x23, #1
	cmp x24, x22
	b.hs sd_noright
	mov x0, x23
	mov x1, x24
	bl cmpf
	cmp w0, #0
	b.ge sd_noright
	mov x23, x24
sd_noright:
	mov x0, x21
	mov x1, x23
	bl cmpf
	cmp w0, #0
	b.ge sd_done
	ldr x5, [x19, x21, lsl #3]
	ldr x6, [x19, x23, lsl #3]
	str x6, [x19, x21, lsl #3]
	str x5, [x19, x23, lsl #3]
	ldr x5, [x20, x21, lsl #3]
	ldr x6, [x20, x23, lsl #3]
	str x6, [x20, x21, lsl #3]
	str x5, [x20, x23, lsl #3]
	mov x21, x23
	b sd_loop
sd_done:
	ldp x23, x24, [sp, #32]
	ldp x21, x22, [sp, #16]
	ldp x29, x30, [sp], #64
	ret

_start:
	adrp x19, arr_ptr
	add x19, x19, :lo12:arr_ptr
	adrp x20, arr_len
	add x20, x20, :lo12:arr_len

	adrp x9, buf
	add x9, x9, :lo12:buf
	mov x10, x9
	add x12, x9, #8, lsl #12
readloop:
	sub x2, x12, x10
	cbz x2, readdone
	mov x0, #0
	mov x1, x10
	mov x8, #63
	svc #0
	cmp x0, #0
	b.le readdone
	add x10, x10, x0
	b readloop
readdone:
	mov x22, x9
	mov x23, x10
	mov x24, x9
	mov x28, #0
parseloop:
	cmp x22, x23
	b.hs parseend
	ldrb w7, [x22]
	cmp w7, #0x0A
	b.ne notnl
	str x24, [x19, x28, lsl #3]
	sub x7, x22, x24
	str x7, [x20, x28, lsl #3]
	add x28, x28, #1
	add x22, x22, #1
	mov x24, x22
	b parseloop
notnl:
	add x22, x22, #1
	b parseloop
parseend:
	cmp x24, x23
	b.hs dosort
	str x24, [x19, x28, lsl #3]
	sub x7, x23, x24
	str x7, [x20, x28, lsl #3]
	add x28, x28, #1

dosort:
	cmp x28, #1
	b.le output
	lsr x27, x28, #1
	sub x27, x27, #1
build:
	cmp x27, #0
	b.lt afterbuild
	mov x0, x27
	mov x1, x28
	bl siftdown
	sub x27, x27, #1
	b build
afterbuild:
	sub x27, x28, #1
sortloop:
	cmp x27, #0
	b.le output
	ldr x5, [x19]
	ldr x6, [x19, x27, lsl #3]
	str x6, [x19]
	str x5, [x19, x27, lsl #3]
	ldr x5, [x20]
	ldr x6, [x20, x27, lsl #3]
	str x6, [x20]
	str x5, [x20, x27, lsl #3]
	mov x0, #0
	mov x1, x27
	bl siftdown
	sub x27, x27, #1
	b sortloop

output:
	mov x25, #0
outloop:
	cmp x25, x28
	b.hs doexit
	ldr x1, [x19, x25, lsl #3]
	ldr x2, [x20, x25, lsl #3]
	mov x0, #1
	mov x8, #64
	svc #0
	mov x0, #1
	adr x1, nlbyte
	mov x2, #1
	mov x8, #64
	svc #0
	add x25, x25, #1
	b outloop
doexit:
	mov x0, #0
	mov x8, #93
	svc #0
