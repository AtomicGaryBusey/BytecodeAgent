.text
.globl _start

_start:
    xor r12, r12
rloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle rdone
    add r12, rax
    jmp rloop
rdone:
    lea rsi, [rip + inbuf]
    xor ebx, ebx
    call sw
    call pv
    test eax, eax
    jnz do_error
    call sw
    cmp rbx, r12
    jne do_error
    lea rsi, [rip + ok_msg]
    mov edx, 3
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

do_error:
    lea rsi, [rip + err_msg]
    mov edx, 9
    mov eax, 1
    mov edi, 1
    syscall
    mov rax, rbx
    lea rdi, [rip + numbuf + 32]
    dec rdi
    mov byte ptr [rdi], 0x0A
    mov ecx, 10
de_conv:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz de_conv
    mov rsi, rdi
    lea rdx, [rip + numbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

# skip whitespace, advances rbx, no return value
sw:
    cmp rbx, r12
    jae sw_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0A
    je sw_adv
    cmp al, 0x0D
    je sw_adv
    ret
sw_adv:
    inc rbx
    jmp sw
sw_done:
    ret

# parse value, rax=0 ok / 1 fail, rbx = position
pv:
    cmp rbx, r12
    jae pv_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '{'
    je po
    cmp al, '['
    je pa
    cmp al, '"'
    je ps
    cmp al, 't'
    je pv_true
    cmp al, 'f'
    je pv_false
    cmp al, 'n'
    je pv_null
    cmp al, '-'
    je pn
    cmp al, '0'
    jb pv_fail
    cmp al, '9'
    jbe pn
pv_fail:
    mov eax, 1
    ret

pv_true:
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'r'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'u'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'e'
    jne kw_fail
    inc rbx
    xor eax, eax
    ret

pv_false:
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'a'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'l'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 's'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'e'
    jne kw_fail
    inc rbx
    xor eax, eax
    ret

pv_null:
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'u'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'l'
    jne kw_fail
    inc rbx
    cmp rbx, r12
    jae kw_fail
    cmp byte ptr [rsi + rbx], 'l'
    jne kw_fail
    inc rbx
    xor eax, eax
    ret
kw_fail:
    mov eax, 1
    ret

# parse string, entry buf[rbx]=='"'
ps:
    inc rbx
ps_loop:
    cmp rbx, r12
    jae ps_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '"'
    je ps_end
    cmp al, 0x5C
    je ps_esc
    cmp al, 0x20
    jb ps_fail
    inc rbx
    jmp ps_loop
ps_end:
    inc rbx
    xor eax, eax
    ret
ps_fail:
    mov eax, 1
    ret
ps_esc:
    inc rbx
    cmp rbx, r12
    jae ps_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '"'
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, '/'
    je ps_esc_ok
    cmp al, 'b'
    je ps_esc_ok
    cmp al, 'f'
    je ps_esc_ok
    cmp al, 'n'
    je ps_esc_ok
    cmp al, 'r'
    je ps_esc_ok
    cmp al, 't'
    je ps_esc_ok
    cmp al, 'u'
    je ps_u
    jmp ps_fail
ps_esc_ok:
    inc rbx
    jmp ps_loop
ps_u:
    inc rbx
    mov ecx, 4
ps_u_loop:
    cmp rbx, r12
    jae ps_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb ps_fail
    cmp al, '9'
    jbe ps_u_good
    or al, 0x20
    cmp al, 'a'
    jb ps_fail
    cmp al, 'f'
    ja ps_fail
ps_u_good:
    inc rbx
    dec ecx
    jnz ps_u_loop
    jmp ps_loop

# parse number, entry buf[rbx] is '-' or digit
pn:
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '-'
    jne pn_int
    inc rbx
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [rsi + rbx]
pn_int:
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
    cmp al, '0'
    je pn_zero
    inc rbx
pn_digs:
    cmp rbx, r12
    jae pn_frac
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb pn_frac
    cmp al, '9'
    ja pn_frac
    inc rbx
    jmp pn_digs
pn_zero:
    inc rbx
pn_frac:
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '.'
    jne pn_exp
    inc rbx
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
    inc rbx
pn_fdigs:
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb pn_exp
    cmp al, '9'
    ja pn_exp
    inc rbx
    jmp pn_fdigs
pn_exp:
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 'e'
    je pn_exp2
    cmp al, 'E'
    je pn_exp2
    jmp pn_ok
pn_exp2:
    inc rbx
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '+'
    je pn_exps
    cmp al, '-'
    je pn_exps
    jmp pn_expd_first
pn_exps:
    inc rbx
pn_expd_first:
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
    inc rbx
pn_expd:
    cmp rbx, r12
    jae pn_ok
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '0'
    jb pn_ok
    cmp al, '9'
    ja pn_ok
    inc rbx
    jmp pn_expd
pn_ok:
    xor eax, eax
    ret
pn_fail:
    mov eax, 1
    ret

# parse array, entry buf[rbx]=='['
pa:
    inc rbx
    call sw
    cmp rbx, r12
    jae pa_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ']'
    jne pa_loop
    inc rbx
    xor eax, eax
    ret
pa_loop:
    call pv
    test eax, eax
    jnz pa_ret
    call sw
    cmp rbx, r12
    jae pa_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ']'
    je pa_close
    cmp al, ','
    jne pa_fail
    inc rbx
    call sw
    jmp pa_loop
pa_close:
    inc rbx
    xor eax, eax
pa_ret:
    ret
pa_fail:
    mov eax, 1
    ret

# parse object, entry buf[rbx]=='{'
po:
    inc rbx
    call sw
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '}'
    jne po_loop
    inc rbx
    xor eax, eax
    ret
po_loop:
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '"'
    jne po_fail
    call ps
    test eax, eax
    jnz po_ret
    call sw
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, ':'
    jne po_fail
    inc rbx
    call sw
    call pv
    test eax, eax
    jnz po_ret
    call sw
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [rsi + rbx]
    cmp al, '}'
    je po_close
    cmp al, ','
    jne po_fail
    inc rbx
    call sw
    jmp po_loop
po_close:
    inc rbx
    xor eax, eax
po_ret:
    ret
po_fail:
    mov eax, 1
    ret

.section .rodata
err_msg:
    .ascii "error at "
ok_msg:
    .ascii "ok\n"

.bss
.balign 16
inbuf:
    .zero 65536
numbuf:
    .zero 32
