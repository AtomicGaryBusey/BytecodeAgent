.globl _start

.bss
.balign 16
buf:
.zero 65536

.text
_start:
    // x19 = lines, x20 = words, x21 = bytes
    mov x19, #0
    mov x20, #0
    mov x21, #0
    // x22 = inword flag (0 = not in word)
    mov x22, #0

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    svc #0
    cmp x0, #0
    ble done_read
    // x0 = bytes read
    add x21, x21, x0
    mov x23, x0           // count remaining
    adrp x24, buf
    add x24, x24, :lo12:buf

byte_loop:
    cbz x23, read_loop
    ldrb w9, [x24], #1
    sub x23, x23, #1
    // check newline
    cmp w9, #0x0A
    b.ne not_nl
    add x19, x19, #1
not_nl:
    // check whitespace
    cmp w9, #0x20
    b.eq is_ws
    cmp w9, #0x09
    b.eq is_ws
    cmp w9, #0x0A
    b.eq is_ws
    cmp w9, #0x0B
    b.eq is_ws
    cmp w9, #0x0C
    b.eq is_ws
    cmp w9, #0x0D
    b.eq is_ws
    // non-whitespace
    cbnz x22, byte_loop
    mov x22, #1
    add x20, x20, #1
    b byte_loop
is_ws:
    mov x22, #0
    b byte_loop

done_read:
    // print lines, space, words, space, bytes, newline
    // use outbuf in bss
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x26, x25          // write pointer

    mov x0, x19
    bl print_num
    mov w9, #0x20
    strb w9, [x26], #1

    mov x0, x20
    bl print_num
    mov w9, #0x20
    strb w9, [x26], #1

    mov x0, x21
    bl print_num
    mov w9, #0x0A
    strb w9, [x26], #1

    // write
    mov x8, #64
    mov x0, #1
    mov x1, x25
    sub x2, x26, x25
    svc #0

    mov x8, #93
    mov x0, #0
    svc #0

// print_num: x0 = number, append decimal to [x26]
print_num:
    // handle zero
    cbnz x0, pn_nonzero
    mov w9, #0x30
    strb w9, [x26], #1
    ret
pn_nonzero:
    // use stack to reverse digits
    mov x10, x26         // save not needed
    sub sp, sp, #32
    mov x11, sp          // temp buffer
    mov x12, #0          // count
    mov x13, #10
pn_div:
    cbz x0, pn_emit
    udiv x14, x0, x13
    msub x15, x14, x13, x0   // remainder
    add w15, w15, #0x30
    strb w15, [x11, x12]
    add x12, x12, #1
    mov x0, x14
    b pn_div
pn_emit:
    // digits in x11 reversed, count x12
    sub x12, x12, #1
pn_emit_loop:
    ldrb w15, [x11, x12]
    strb w15, [x26], #1
    cbz x12, pn_done
    sub x12, x12, #1
    b pn_emit_loop
pn_done:
    add sp, sp, #32
    ret

.bss
.balign 16
outbuf:
.zero 64
