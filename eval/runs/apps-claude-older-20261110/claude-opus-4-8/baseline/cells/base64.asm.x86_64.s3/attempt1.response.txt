.text
.globl _start

_start:
    xor r12d, r12d

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 49152
    sub edx, r12d
    jz read_done
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbp, [rip + alpha]
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    mov rcx, r12
    xor r14d, r14d

g_loop:
    cmp rcx, 3
    jb g_tail
    movzx eax, byte ptr [rsi]
    movzx r8d, byte ptr [rsi+1]
    movzx r9d, byte ptr [rsi+2]
    shl eax, 16
    shl r8d, 8
    or eax, r8d
    or eax, r9d
    mov r10d, eax
    mov edx, r10d
    shr edx, 18
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov edx, r10d
    shr edx, 12
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov edx, r10d
    shr edx, 6
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov edx, r10d
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    add rsi, 3
    sub rcx, 3
    jmp g_loop

g_tail:
    test rcx, rcx
    jz finish
    cmp rcx, 1
    je tail1
    movzx eax, byte ptr [rsi]
    movzx r8d, byte ptr [rsi+1]
    shl eax, 16
    shl r8d, 8
    or eax, r8d
    mov r10d, eax
    mov edx, r10d
    shr edx, 18
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov edx, r10d
    shr edx, 12
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov edx, r10d
    shr edx, 6
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov al, 0x3D
    call emit_char
    jmp finish

tail1:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    mov r10d, eax
    mov edx, r10d
    shr edx, 18
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov edx, r10d
    shr edx, 12
    and edx, 63
    mov al, byte ptr [rbp + rdx]
    call emit_char
    mov al, 0x3D
    call emit_char
    mov al, 0x3D
    call emit_char

finish:
    test r14d, r14d
    jz no_final
    mov byte ptr [rdi], 10
    inc rdi
no_final:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

w_loop:
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp w_loop

done:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r14d
    cmp r14d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
1:
    ret

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000
