.text
.globl _start
_start:
  xor r12, r12
rdloop:
  xor eax, eax
  xor edi, edi
  lea rsi, [rip+inbuf]
  add rsi, r12
  mov rdx, 70000
  sub rdx, r12
  syscall
  test rax, rax
  jle rddone
  add r12, rax
  jmp rdloop
rddone:
  xor r13, r13
  xor rbx, rbx
mloop:
  cmp rbx, r12
  jae finish
  mov rsi, rbx
fnl:
  cmp rsi, r12
  jae nlnf
  movzx eax, byte ptr [inbuf + rsi]
  cmp al, 0x0A
  je nlf
  inc rsi
  jmp fnl
nlf:
  mov r14, rsi
  mov rdi, rbx
  mov rsi, r14
  call eval
  lea rbx, [r14+1]
  jmp mloop
nlnf:
  mov rdi, rbx
  mov rsi, r12
  call eval
  mov rbx, r12
  jmp mloop
finish:
  mov eax, 1
  mov edi, 1
  lea rsi, [rip+outbuf]
  mov rdx, r13
  syscall
  mov eax, 60
  xor edi, edi
  syscall

eval:
  mov r10, rsi
  mov r8, rdi
  xor r9d, r9d
skipsep:
  cmp r8, r10
  jae toksdone
  movzx eax, byte ptr [inbuf + r8]
  cmp al, 0x20
  je issep
  cmp al, 0x09
  je issep
  jmp tstart
issep:
  inc r8
  jmp skipsep
tstart:
  mov r11, r8
scantok:
  cmp r8, r10
  jae scandone
  movzx eax, byte ptr [inbuf + r8]
  cmp al, 0x20
  je scandone
  cmp al, 0x09
  je scandone
  inc r8
  jmp scantok
scandone:
  mov rcx, r11
  xor edx, edx
  movzx eax, byte ptr [inbuf + rcx]
  cmp al, 0x2D
  jne nonneg
  mov edx, 1
  inc rcx
nonneg:
  mov rax, r8
  sub rax, rcx
  cmp rax, 1
  jl notlit
  cmp rax, 18
  jg notlit
  xor esi, esi
pdig:
  cmp rcx, r8
  jae parsed
  movzx eax, byte ptr [inbuf + rcx]
  sub eax, 0x30
  cmp eax, 9
  ja notlit
  lea rsi, [rsi + rsi*4]
  add rsi, rsi
  add rsi, rax
  inc rcx
  jmp pdig
parsed:
  test edx, edx
  jz pushv
  neg rsi
pushv:
  mov qword ptr [valstack + r9*8], rsi
  inc r9
  jmp skipsep
notlit:
  mov rax, r8
  sub rax, r11
  cmp rax, 1
  jne everr
  movzx eax, byte ptr [inbuf + r11]
  cmp r9, 2
  jl everr
  mov rcx, qword ptr [valstack + r9*8 - 8]
  mov rsi, qword ptr [valstack + r9*8 - 16]
  sub r9, 2
  cmp al, 0x2B
  je opadd
  cmp al, 0x2D
  je opsub
  cmp al, 0x2A
  je opmul
  cmp al, 0x2F
  je opdiv
  cmp al, 0x25
  je opmod
  jmp everr
opadd:
  add rsi, rcx
  jmp pushr
opsub:
  sub rsi, rcx
  jmp pushr
opmul:
  imul rsi, rcx
  jmp pushr
opdiv:
  test rcx, rcx
  jz everr
  mov r11, 0x8000000000000000
  cmp rsi, r11
  jne 1f
  cmp rcx, -1
  je everr
1:
  mov rax, rsi
  cqo
  idiv rcx
  mov rsi, rax
  jmp pushr
opmod:
  test rcx, rcx
  jz everr
  mov r11, 0x8000000000000000
  cmp rsi, r11
  jne 1f
  cmp rcx, -1
  je everr
1:
  mov rax, rsi
  cqo
  idiv rcx
  mov rsi, rdx
  jmp pushr
pushr:
  mov qword ptr [valstack + r9*8], rsi
  inc r9
  jmp skipsep
toksdone:
  cmp r9, 1
  jne everr
  mov rax, qword ptr [valstack]
  call outnum
  ret
everr:
  call outerr
  ret

outerr:
  lea rsi, [rip+errmsg]
  mov ecx, 6
oe1:
  movzx eax, byte ptr [rsi]
  mov byte ptr [outbuf + r13], al
  inc r13
  inc rsi
  dec ecx
  jnz oe1
  ret

outnum:
  test rax, rax
  jns npos
  mov byte ptr [outbuf + r13], 0x2D
  inc r13
  neg rax
npos:
  xor ecx, ecx
  test rax, rax
  jnz cvl
  mov byte ptr [numtmp], 0x30
  mov ecx, 1
  jmp emitn
cvl:
  mov r8, 10
cvl2:
  xor edx, edx
  div r8
  add dl, 0x30
  mov byte ptr [numtmp + rcx], dl
  inc rcx
  test rax, rax
  jnz cvl2
emitn:
  dec rcx
  movzx eax, byte ptr [numtmp + rcx]
  mov byte ptr [outbuf + r13], al
  inc r13
  test rcx, rcx
  jnz emitn
  mov byte ptr [outbuf + r13], 0x0A
  inc r13
  ret

.section .rodata
errmsg: .ascii "error\n"

.bss
.balign 8
inbuf: .zero 70000
outbuf: .zero 600000
valstack: .zero 8192
numtmp: .zero 32
