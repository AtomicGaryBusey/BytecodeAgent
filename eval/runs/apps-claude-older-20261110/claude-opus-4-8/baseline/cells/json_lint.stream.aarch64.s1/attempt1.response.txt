_start:
    adrp x19, buf
    add  x19, x19, :lo12:buf
    mov  x22, #0
rd:
    mov  x0, #0
    add  x1, x19, x22
    mov  x2, #0x20000
    sub  x2, x2, x22
    mov  x8, #63
    svc  #0

    add  x22, x22, x0
    b    rd

    mov  x20, x22
    mov  x21, #0
    bl   skip_ws
    bl   parse_value
    cbnz w0, do_error
    bl   skip_ws
    cmp  x21, x20
    b.ne do_error
    b    do_ok

// ---- skip_ws: advance pos over ws ----
skip_ws:
sw1:
    cmp  x21, x20
    b.ge sw2
    ldrb w0, [x19, x21]
    cmp  w0, #0x20
    b.eq sw3
    cmp  w0, #0x09
    b.eq sw3
    cmp  w0, #0x0A
    b.eq sw3
    cmp  w0, #0x0D
    b.eq sw3
    b    sw2
sw3:
    add  x21, x21, #1
    b    sw1
sw2:
    ret

// ---- expect_lit: x2=ptr, x3=len ----
expect_lit:
    mov  x4, #0
el1:
    cmp  x4, x3
    b.ge el2
    add  x5, x21, x4
    cmp  x5, x20
    b.ge el3
    ldrb w6, [x19, x5]
    ldrb w7, [x2, x4]
    cmp  w6, w7
    b.ne el3
    add  x4, x4, #1
    b    el1
el3:
    add  x21, x21, x4
    mov  w0, #1
    ret
el2:
    add  x21, x21, x3
    mov  w0, #0
    ret

// ---- parse_value ----
parse_value:
    stp  x29, x30, [sp, #-16]!
    cmp  x21, x20
    b.ge perr
    ldrb w0, [x19, x21]
    cmp  w0, #0x7B
    b.eq pv_obj
    cmp  w0, #0x5B
    b.eq pv_arr
    cmp  w0, #0x22
    b.eq pv_str
    cmp  w0, #0x2D
    b.eq pv_num
    sub  w1, w0, #0x30
    cmp  w1, #9
    b.ls pv_num
    cmp  w0, #0x74
    b.eq pv_true
    cmp  w0, #0x66
    b.eq pv_false
    cmp  w0, #0x6E
    b.eq pv_null
perr:
    mov  w0, #1
    b    pv_ret
pv_obj:
    bl   parse_object
    b    pv_ret
pv_arr:
    bl   parse_array
    b    pv_ret
pv_str:
    bl   parse_string
    b    pv_ret
pv_num:
    bl   parse_number
    b    pv_ret
pv_true:
    adrp x2, lit_true
    add  x2, x2, :lo12:lit_true
    mov  x3, #4
    bl   expect_lit
    b    pv_ret
pv_false:
    adrp x2, lit_false
    add  x2, x2, :lo12:lit_false
    mov  x3, #5
    bl   expect_lit
    b    pv_ret
pv_null:
    adrp x2, lit_null
    add  x2, x2, :lo12:lit_null
    mov  x3, #4
    bl   expect_lit
    b    pv_ret
pv_ret:
    ldp  x29, x30, [sp], #16
    ret

// ---- parse_object ----
parse_object:
    stp  x29, x30, [sp, #-16]!
    add  x21, x21, #1
    bl   skip_ws
    cmp  x21, x20
    b.ge po_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x7D
    b.eq po_close
po_member:
    cmp  x21, x20
    b.ge po_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x22
    b.ne po_err
    bl   parse_string
    cmp  w0, #0
    b.ne po_fail
    bl   skip_ws
    cmp  x21, x20
    b.ge po_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x3A
    b.ne po_err
    add  x21, x21, #1
    bl   skip_ws
    bl   parse_value
    cmp  w0, #0
    b.ne po_fail
    bl   skip_ws
    cmp  x21, x20
    b.ge po_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x2C
    b.eq po_comma
    cmp  w0, #0x7D
    b.eq po_close
    b    po_err
po_comma:
    add  x21, x21, #1
    bl   skip_ws
    b    po_member
po_close:
    add  x21, x21, #1
    mov  w0, #0
    ldp  x29, x30, [sp], #16
    ret
po_err:
    mov  w0, #1
po_fail:
    ldp  x29, x30, [sp], #16
    ret

// ---- parse_array ----
parse_array:
    stp  x29, x30, [sp, #-16]!
    add  x21, x21, #1
    bl   skip_ws
    cmp  x21, x20
    b.ge pa_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x5D
    b.eq pa_close
pa_elem:
    bl   parse_value
    cmp  w0, #0
    b.ne pa_fail
    bl   skip_ws
    cmp  x21, x20
    b.ge pa_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x2C
    b.eq pa_comma
    cmp  w0, #0x5D
    b.eq pa_close
    b    pa_err
pa_comma:
    add  x21, x21, #1
    bl   skip_ws
    b    pa_elem
pa_close:
    add  x21, x21, #1
    mov  w0, #0
    ldp  x29, x30, [sp], #16
    ret
pa_err:
    mov  w0, #1
pa_fail:
    ldp  x29, x30, [sp], #16
    ret

// ---- parse_string (leaf) ----
parse_string:
    add  x21, x21, #1
ps_loop:
    cmp  x21, x20
    b.ge ps_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x22
    b.eq ps_close
    cmp  w0, #0x5C
    b.eq ps_esc
    cmp  w0, #0x20
    b.lo ps_err
    add  x21, x21, #1
    b    ps_loop
ps_esc:
    add  x21, x21, #1
    cmp  x21, x20
    b.ge ps_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x22
    b.eq ps_esc_ok
    cmp  w0, #0x5C
    b.eq ps_esc_ok
    cmp  w0, #0x2F
    b.eq ps_esc_ok
    cmp  w0, #0x62
    b.eq ps_esc_ok
    cmp  w0, #0x66
    b.eq ps_esc_ok
    cmp  w0, #0x6E
    b.eq ps_esc_ok
    cmp  w0, #0x72
    b.eq ps_esc_ok
    cmp  w0, #0x74
    b.eq ps_esc_ok
    cmp  w0, #0x75
    b.eq ps_esc_u
    b    ps_err
ps_esc_ok:
    add  x21, x21, #1
    b    ps_loop
ps_esc_u:
    add  x21, x21, #1
    mov  x3, #4
ps_u_loop:
    cmp  x3, #0
    b.eq ps_loop
    cmp  x21, x20
    b.ge ps_err
    ldrb w0, [x19, x21]
    sub  w1, w0, #0x30
    cmp  w1, #9
    b.ls ps_u_ok
    orr  w1, w0, #0x20
    sub  w2, w1, #0x61
    cmp  w2, #5
    b.ls ps_u_ok
    b    ps_err
ps_u_ok:
    add  x21, x21, #1
    sub  x3, x3, #1
    b    ps_u_loop
ps_close:
    add  x21, x21, #1
    mov  w0, #0
    ret
ps_err:
    mov  w0, #1
    ret

// ---- parse_number (leaf) ----
parse_number:
    cmp  x21, x20
    b.ge pn_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x2D
    b.ne pn_int
    add  x21, x21, #1
pn_int:
    cmp  x21, x20
    b.ge pn_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x30
    b.eq pn_zero
    sub  w1, w0, #0x31
    cmp  w1, #8
    b.hi pn_err
    add  x21, x21, #1

    cmp  x21, x20
    b.ge pn_frac
    ldrb w0, [x19, x21]
    sub  w1, w0, #0x30
    cmp  w1, #9
    b.hi pn_frac
    add  x21, x21, #1
    b    pn_digits
pn_zero:

    b.ge pn_ok
    ldrb w0, [x19, x21]
    cmp  w0, #0x2E
    b.ne pn_exp
    add  x21, x21, #1
    cmp  x21, x20
    b.ge pn_err
    ldrb w0, [x19, x21]
    sub  w1, w0, #0x30
    cmp  w1, #9
    b.hi pn_err
    add  x21, x21, #1
pn_frac_digits:
    cmp  x21, x20
    b.ge pn_ok
    ldrb w0, [x19, x21]
    sub  w1, w0, #0x30
    cmp  w1, #9
    b.hi pn_exp
    add  x21, x21, #1
    b    pn_frac_digits
pn_exp:

    b.ge pn_ok
    ldrb w0, [x19, x21]
    cmp  w0, #0x65
    b.eq pn_exp2
    cmp  w0, #0x45
    b.eq pn_exp2

pn_exp2:
    add  x21, x21, #1
    cmp  x21, x20
    b.ge pn_err
    ldrb w0, [x19, x21]
    cmp  w0, #0x2B
    b.eq pn_exp_sign
    cmp  w0, #0x2D
    b.eq pn_exp_sign
    b    pn_exp_digit0

pn_exp_digit0:

    b.ge pn_err
    ldrb w0, [x19, x21]
    sub  w1, w0, #0x30
    cmp  w1, #9
    b.hi pn_err
    add  x21, x21, #1

    b.ge pn_ok
    ldrb w0, [x19, x21]
    sub  w1, w0, #0x30
    cmp  w1, #9

pn_err:

// ---- output ----
