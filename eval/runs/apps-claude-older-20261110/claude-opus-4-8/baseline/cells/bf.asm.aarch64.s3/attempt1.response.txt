.globl _start

.text
_start:
    // read all stdin into input_buf
    adrp x25, input_buf
    add x25, x25, :lo12:input_buf
    mov x19, #0
read_loop:
    movz x2, #1, lsl #16        // 65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    add x1, x25, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop
read_done:

    // find first '!'
    mov x20, #0
scan:
    cmp x20, x19
    b.ge no_bang
    ldrb w0, [x25, x20]
    cmp w0, #0x21
    b.eq found_bang
    add x20, x20, #1
    b scan
found_bang:
    add x21, x20, #1           // input ptr
    mov x22, x19               // input end
    b after_find
no_bang:
    mov x20, x19               // prog_len
    mov x21, x19
    mov x22, x19
after_find:

    // validate brackets + build jump table
    adrp x26, jumptab
    add x26, x26, :lo12:jumptab
    adrp x9, bstack
    add x9, x9, :lo12:bstack
    mov x10, #0                // stack count
    mov x11, #0                // scan pc
bscan:
    cmp x11, x20
    b.ge bscan_done
    ldrb w0, [x25, x11]
    cmp w0, #0x5b
    b.eq is_open
    cmp w0, #0x5d
    b.eq is_close
    add x11, x11, #1
    b bscan
is_open:
    str w11, [x9, x10, lsl #2]
    add x10, x10, #1
    add x11, x11, #1
    b bscan
is_close:
    cbz x10, unbalanced
    sub x10, x10, #1
    ldr w12, [x9, x10, lsl #2]
    add w13, w11, #1
    str w13, [x26, x12, lsl #2]
    add w14, w12, #1
    str w14, [x26, x11, lsl #2]
    add x11, x11, #1
    b bscan
bscan_done:
    cbz x10, run_start
unbalanced:
    mov x0, #1
    mov x8, #93
    svc #0

run_start:
    adrp x24, tape
    add x24, x24, :lo12:tape
    mov x23, #0                // pc
    mov x27, #0                // out_len
    adrp x28, outbuf
    add x28, x28, :lo12:outbuf
run_loop:
    cmp x23, x20
    b.ge run_done
    ldrb w0, [x25, x23]
    cmp w0, #0x3e
    b.eq c_gt
    cmp w0, #0x3c
    b.eq c_lt
    cmp w0, #0x2b
    b.eq c_plus
    cmp w0, #0x2d
    b.eq c_minus
    cmp w0, #0x2e
    b.eq c_dot
    cmp w0, #0x2c
    b.eq c_comma
    cmp w0, #0x5b
    b.eq c_open
    cmp w0, #0x5d
    b.eq c_close
    add x23, x23, #1
    b run_loop
c_gt:
    add x24, x24, #1
    add x23, x23, #1
    b run_loop
c_lt:
    sub x24, x24, #1
    add x23, x23, #1
    b run_loop
c_plus:
    ldrb w1, [x24]
    add w1, w1, #1
    strb w1, [x24]
    add x23, x23, #1
    b run_loop
c_minus:
    ldrb w1, [x24]
    sub w1, w1, #1
    strb w1, [x24]
    add x23, x23, #1
    b run_loop
c_dot:
    ldrb w1, [x24]
    strb w1, [x28, x27]
    add x27, x27, #1
    movz x3, #1, lsl #16
    cmp x27, x3
    b.lt 1f
    bl flush
1:
    add x23, x23, #1
    b run_loop
c_comma:
    cmp x21, x22
    b.ge 1f
    ldrb w1, [x25, x21]
    add x21, x21, #1
    strb w1, [x24]
    b 2f
1:
    strb wzr, [x24]
2:
    add x23, x23, #1
    b run_loop
c_open:
    ldrb w1, [x24]
    cbnz w1, 1f
    ldr w23, [x26, x23, lsl #2]
    b run_loop
1:
    add x23, x23, #1
    b run_loop
c_close:
    ldrb w1, [x24]
    cbz w1, 1f
    ldr w23, [x26, x23, lsl #2]
    b run_loop
1:
    add x23, x23, #1
    b run_loop
run_done:
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0

flush:
    cbz x27, 1f
    mov x0, #1
    mov x1, x28
    mov x2, x27
    mov x8, #64
    svc #0
    mov x27, #0
1:
    ret

.bss
.align 4
input_buf:
    .zero 65536
tape:
    .zero 30000
jumptab:
    .zero 262144
bstack:
    .zero 4000
outbuf:
    .zero 65536
