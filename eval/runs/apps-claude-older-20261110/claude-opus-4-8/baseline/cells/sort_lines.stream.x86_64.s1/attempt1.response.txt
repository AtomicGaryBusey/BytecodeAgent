.text
.globl _start

_start:
    # ---- read all input ----
    xor r12, r12            # total bytes read
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [inbuf + r12]
    mov rdx, 40000
    sub rdx, r12
    syscall
    test rax, rax
    jle .read_done
    add r12, rax
    jmp .read_loop
.read_done:

    # ---- parse lines ----
    mov r14, r12            # total length
    xor rsi, rsi            # pos
    xor r15, r15            # nlines
.parse_loop:
    cmp rsi, r14
    jae .parse_done
    mov rbx, rsi            # line_start
.parse_scan:
    cmp rsi, r14
    jae .parse_store
    cmp byte ptr [inbuf + rsi], 0x0A
    je .parse_store
    inc rsi
    jmp .parse_scan
.parse_store:
    lea rax, [inbuf + rbx]
    mov [line_ptr + r15*8], rax
    mov rcx, rsi
    sub rcx, rbx
    mov [line_len + r15*8], rcx
    inc r15
    cmp rsi, r14
    jae .parse_loop
    inc rsi                 # skip newline
    jmp .parse_loop
.parse_done:

    # ---- selection sort ----
    xor r12, r12            # i
.sort_outer:
    mov rax, r15
    dec rax
    cmp r12, rax
    jge .sort_done
    mov r14, r12            # min = i
    lea r13, [r12 + 1]      # j
.sort_inner:
    cmp r13, r15
    jge .sort_inner_done
    mov rdi, r13
    mov rsi, r14
    call cmp_lines
    test eax, eax
    jns .sort_noupd
    mov r14, r13
.sort_noupd:
    inc r13
    jmp .sort_inner
.sort_inner_done:
    mov rax, [line_ptr + r12*8]
    mov rcx, [line_ptr + r14*8]
    mov [line_ptr + r12*8], rcx
    mov [line_ptr + r14*8], rax
    mov rax, [line_len + r12*8]
    mov rcx, [line_len + r14*8]
    mov [line_len + r12*8], rcx
    mov [line_len + r14*8], rax
    inc r12
    jmp .sort_outer
.sort_done:

    # ---- build output ----
    xor r12, r12            # line index
    xor r13, r13            # out offset
.build_loop:
    cmp r12, r15
    jge .build_done
    mov rsi, [line_ptr + r12*8]
    mov rcx, [line_len + r12*8]
    lea rdi, [outbuf + r13]
    add r13, rcx
    rep movsb
    mov byte ptr [outbuf + r13], 0x0A
    inc r13
    inc r12
    jmp .build_loop
.build_done:

    # ---- write output ----
    lea r14, [outbuf]
.write_loop:
    test r13, r13
    jz .write_done
    mov rax, 1
    mov rdi, 1
    mov rsi, r14
    mov rdx, r13
    syscall
    test rax, rax
    jle .write_done
    add r14, rax
    sub r13, rax
    jmp .write_loop
.write_done:

    mov rax, 60
    xor rdi, rdi
    syscall

# cmp_lines(rdi=index a, rsi=index b) -> eax (<0 if a<b)
cmp_lines:
    mov r8,  [line_ptr + rdi*8]
    mov r9,  [line_len + rdi*8]
    mov r10, [line_ptr + rsi*8]
    mov r11, [line_len + rsi*8]
    mov rcx, r9
    cmp r11, rcx
    cmovb rcx, r11          # min length
    xor rdx, rdx
.cmp_loop:
    cmp rdx, rcx
    jae .cmp_eq
    movzx eax, byte ptr [r8 + rdx]
    movzx edi, byte ptr [r10 + rdx]
    cmp eax, edi
    jne .cmp_diff
    inc rdx
    jmp .cmp_loop
.cmp_diff:
    sub eax, edi
    ret
.cmp_eq:
    mov eax, r9d
    sub eax, r11d
    ret

.bss
.align 16
inbuf:
    .zero 40000
line_ptr:
    .zero 8192
line_len:
    .zero 8192
outbuf:
    .zero 40000
