.text
.globl _start

_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf      // input buffer base
    mov  x20, #0                    // total bytes read

read_loop:
    mov  x0, #0                     // fd 0
    add  x1, x19, x20
    mov  x2, #49152
    sub  x2, x2, x20
    cbz  x2, read_done
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop
read_done:

    mov  x21, #0                    // i index
    adrp x25, outbuf
    add  x25, x25, :lo12:outbuf     // output base
    mov  x22, x25                   // output pointer
    mov  w23, #0                    // column count
    adrp x24, alphabet
    add  x24, x24, :lo12:alphabet

encode_loop:
    cmp  x21, x20
    b.ge encode_done
    sub  x9, x20, x21               // remaining
    add  x10, x19, x21
    ldrb w11, [x10]
    cmp  x9, #3
    b.lt partial

    ldrb w12, [x10, #1]
    ldrb w13, [x10, #2]
    lsr  x0, x11, #2
    bl   emit_idx
    and  w0, w11, #3
    lsl  w0, w0, #4
    lsr  w14, w12, #4
    orr  w0, w0, w14
    bl   emit_idx
    and  w0, w12, #0xF
    lsl  w0, w0, #2
    lsr  w14, w13, #6
    orr  w0, w0, w14
    bl   emit_idx
    and  w0, w13, #0x3F
    bl   emit_idx
    add  x21, x21, #3
    b    encode_loop

partial:
    cmp  x9, #2
    b.eq two

    // one byte
    lsr  x0, x11, #2
    bl   emit_idx
    and  w0, w11, #3
    lsl  w0, w0, #4
    bl   emit_idx
    mov  w0, #0x3D
    bl   emit_raw
    mov  w0, #0x3D
    bl   emit_raw
    add  x21, x21, #1
    b    encode_loop

two:
    ldrb w12, [x10, #1]
    lsr  x0, x11, #2
    bl   emit_idx
    and  w0, w11, #3
    lsl  w0, w0, #4
    lsr  w14, w12, #4
    orr  w0, w0, w14
    bl   emit_idx
    and  w0, w12, #0xF
    lsl  w0, w0, #2
    bl   emit_idx
    mov  w0, #0x3D
    bl   emit_raw
    add  x21, x21, #2
    b    encode_loop

encode_done:
    cbz  x20, do_write
    cbz  w23, do_write
    mov  w0, #0x0A
    strb w0, [x22], #1

do_write:
    sub  x2, x22, x25               // length
    mov  x1, x25
write_loop:
    cbz  x2, exit0
    mov  x0, #1
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit0
    add  x1, x1, x0
    sub  x2, x2, x0
    b    write_loop

exit0:
    mov  x0, #0
    mov  x8, #93
    svc  #0

emit_idx:
    ldrb w0, [x24, x0]
emit_raw:
    strb w0, [x22], #1
    add  w23, w23, #1
    cmp  w23, #76
    b.ne 1f
    mov  w0, #0x0A
    strb w0, [x22], #1
    mov  w23, #0
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000
