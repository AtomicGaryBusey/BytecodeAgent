    add     x19, x19, :lo12:inbuf

    mov     x26, #0

    mov     x8, #63

    b.ge    nobang
    ldrb    w0, [x19, x27]

    mov     x24, x23

    add     x25, x25, :lo12:jump

    b.ge    ve
    ldrb    w13, [x19, x12]

    cmp     w13, #0x5D

    add     x11, x11, #1
    b       vn

    ldr     w14, [x10, x11, lsl #2]

    str     w14, [x25, x12, lsl #2]

    b       vl

    mov     x8, #93

    b.ge    runend
    ldrb    w13, [x19, x21]
    cmp     w13, #0x3E
    b.eq    cgt
    cmp     w13, #0x3C
    b.eq    clt
    cmp     w13, #0x2B
    b.eq    cplus
    cmp     w13, #0x2D
    b.eq    cminus
    cmp     w13, #0x2E
    b.eq    cdot
    cmp     w13, #0x2C
    b.eq    ccomma
    cmp     w13, #0x5B
    b.eq    copen
    cmp     w13, #0x5D
    b.eq    cclose

cgt:
    add     x22, x22, #1
    b       runnext

    b       runnext

    add     w14, w14, #1
    strb    w14, [x22]
    b       runnext

    cbnz    w14, runnext
    ldr     w14, [x25, x21, lsl #2]

    b       runnext

    cbz     w14, runnext
    ldr     w14, [x25, x21, lsl #2]
    mov     x21, x14

    mov     x0, #0
    mov     x8, #93
    svc     #0
