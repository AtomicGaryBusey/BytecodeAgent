.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rdx, 49152
    sub rdx, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    cmp rbx, 49152
    jb read_loop
read_done:
    lea rsi, [rip + inbuf]
    mov rcx, rbx
    lea rdi, [rip + outbuf]
    lea r13, [rip + outbuf]
    lea r12, [rip + alpha]
    xor r8d, r8d

loop_start:
    test rcx, rcx
    jz encode_done
    cmp rcx, 3
    jb tail
    movzx r10d, byte ptr [rsi]
    movzx r11d, byte ptr [rsi+1]
    movzx edx, byte ptr [rsi+2]
    mov eax, r10d
    shr eax, 2
    call emit_idx
    mov eax, r10d
    and eax, 3
    shl eax, 4
    mov ebx, r11d
    shr ebx, 4
    or eax, ebx
    call emit_idx
    mov eax, r11d
    and eax, 15
    shl eax, 2
    mov ebx, edx
    shr ebx, 6
    or eax, ebx
    call emit_idx
    mov eax, edx
    and eax, 63
    call emit_idx
    add rsi, 3
    sub rcx, 3
    jmp loop_start

tail:
    cmp rcx, 2
    je two
    movzx r10d, byte ptr [rsi]
    mov eax, r10d
    shr eax, 2
    call emit_idx
    mov eax, r10d
    and eax, 3
    shl eax, 4
    call emit_idx
    mov al, '='
    call emit_raw
    mov al, '='
    call emit_raw
    jmp encode_done
two:
    movzx r10d, byte ptr [rsi]
    movzx r11d, byte ptr [rsi+1]
    mov eax, r10d
    shr eax, 2
    call emit_idx
    mov eax, r10d
    and eax, 3
    shl eax, 4
    mov ebx, r11d
    shr ebx, 4
    or eax, ebx
    call emit_idx
    mov eax, r11d
    and eax, 15
    shl eax, 2
    call emit_idx
    mov al, '='
    call emit_raw
    jmp encode_done

encode_done:
    test r8, r8
    jz write_out
    mov byte ptr [rdi], 0x0A
    inc rdi
write_out:
    mov rsi, r13
    mov rdx, rdi
    sub rdx, r13
write_loop:
    test rdx, rdx
    jz do_exit
    mov rax, 1
    mov rdi, 1
    syscall
    test rax, rax
    jle do_exit
    add rsi, rax
    sub rdx, rax
    jmp write_loop
do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall

emit_idx:
    movzx eax, byte ptr [r12 + rax]
emit_raw:
    mov [rdi], al
    inc rdi
    inc r8
    cmp r8, 76
    jne 1f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r8d, r8d
1:
    ret
