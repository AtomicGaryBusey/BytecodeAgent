.section .rodata
msg_err: .ascii "error at "
msg_ok:  .ascii "ok\n"
msg_nl:  .ascii "\n"
str_true:  .ascii "true"
str_false: .ascii "false"
str_null:  .ascii "null"

.text
.globl _start

_start:
    adrp x19, buf
    add  x19, x19, :lo12:buf
    mov  x20, #0
.rloop:
    mov  x8, #63
    mov  x0, #0
    add  x1, x19, x20
    movz x2, #0x2, lsl #16
    sub  x2, x2, x20
    svc  #0
    cmp  x0, #0
    b.le .rdone
    add  x20, x20, x0
    b    .rloop
.rdone:
    mov  x21, #0
    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x21, x20
    b.ne report_error
    mov  x0, #1
    adrp x1, msg_ok
    add  x1, x1, :lo12:msg_ok
    mov  x2, #3
    mov  x8, #64
    svc  #0
    mov  x0, #0
    mov  x8, #93
    svc  #0

// leaf: advances x21 over whitespace, uses x0
skip_ws:
.ws_loop:
    cmp  x21, x20
    b.hs .ws_done
    ldrb w0, [x19, x21]
    cmp  w0, #0x20
    b.eq .ws_adv
    cmp  w0, #0x09
    b.eq .ws_adv
    cmp  w0, #0x0A
    b.eq .ws_adv
    cmp  w0, #0x0D
    b.eq .ws_adv
    b    .ws_done
.ws_adv:
    add  x21, x21, #1
    b    .ws_loop
.ws_done:
    ret

parse_value:
    stp  x29, x30, [sp, #-16]!
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'{'
    b.eq .pv_obj
    cmp  w0, #'['
    b.eq .pv_arr
    cmp  w0, #'"'
    b.eq .pv_str
    cmp  w0, #'t'
    b.eq .pv_true
    cmp  w0, #'f'
    b.eq .pv_false
    cmp  w0, #'n'
    b.eq .pv_null
    cmp  w0, #'-'
    b.eq .pv_num
    cmp  w0, #'0'
    b.lo report_error
    cmp  w0, #'9'
    b.hi report_error
.pv_num:
    bl   parse_number
    b    .pv_ret
.pv_str:
    bl   parse_string
    b    .pv_ret
.pv_obj:
    bl   parse_object
    b    .pv_ret
.pv_arr:
    bl   parse_array
    b    .pv_ret
.pv_true:
    adrp x0, str_true
    add  x0, x0, :lo12:str_true
    mov  x1, #4
    bl   match_literal
    b    .pv_ret
.pv_false:
    adrp x0, str_false
    add  x0, x0, :lo12:str_false
    mov  x1, #5
    bl   match_literal
    b    .pv_ret
.pv_null:
    adrp x0, str_null
    add  x0, x0, :lo12:str_null
    mov  x1, #4
    bl   match_literal
    b    .pv_ret
.pv_ret:
    ldp  x29, x30, [sp], #16
    ret

// leaf: x0=ptr x1=len; matches literal, advances x21
match_literal:
    mov  x2, #0
.ml_loop:
    cmp  x2, x1
    b.hs .ml_done
    cmp  x21, x20
    b.hs report_error
    ldrb w3, [x19, x21]
    ldrb w4, [x0, x2]
    cmp  w3, w4
    b.ne report_error
    add  x21, x21, #1
    add  x2, x2, #1
    b    .ml_loop
.ml_done:
    ret

// leaf: parses string assuming buf[x21]=='"'
parse_string:
    add  x21, x21, #1
.ps_loop:
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'"'
    b.eq .ps_close
    cmp  w0, #0x5c
    b.eq .ps_esc
    cmp  w0, #0x20
    b.lo report_error
    add  x21, x21, #1
    b    .ps_loop
.ps_esc:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'"'
    b.eq .ps_simple
    cmp  w0, #0x5c
    b.eq .ps_simple
    cmp  w0, #'/'
    b.eq .ps_simple
    cmp  w0, #'b'
    b.eq .ps_simple
    cmp  w0, #'f'
    b.eq .ps_simple
    cmp  w0, #'n'
    b.eq .ps_simple
    cmp  w0, #'r'
    b.eq .ps_simple
    cmp  w0, #'t'
    b.eq .ps_simple
    cmp  w0, #'u'
    b.eq .ps_u
    b    report_error
.ps_simple:
    add  x21, x21, #1
    b    .ps_loop
.ps_u:
    add  x21, x21, #1
    mov  w1, #4
.ps_uloop:
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo report_error
    cmp  w0, #'9'
    b.ls .ps_uok
    cmp  w0, #'A'
    b.lo report_error
    cmp  w0, #'F'
    b.ls .ps_uok
    cmp  w0, #'a'
    b.lo report_error
    cmp  w0, #'f'
    b.ls .ps_uok
    b    report_error
.ps_uok:
    add  x21, x21, #1
    subs w1, w1, #1
    b.ne .ps_uloop
    b    .ps_loop
.ps_close:
    add  x21, x21, #1
    ret

// leaf: parses number assuming buf[x21] is '-' or digit
parse_number:
    ldrb w0, [x19, x21]
    cmp  w0, #'-'
    b.ne .pn_int
    add  x21, x21, #1
.pn_int:
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo report_error
    cmp  w0, #'9'
    b.hi report_error
    cmp  w0, #'0'
    b.eq .pn_zero
    add  x21, x21, #1
.pn_digits:
    cmp  x21, x20
    b.hs .pn_frac
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo .pn_frac
    cmp  w0, #'9'
    b.hi .pn_frac
    add  x21, x21, #1
    b    .pn_digits
.pn_zero:
    add  x21, x21, #1
.pn_frac:
    cmp  x21, x20
    b.hs .pn_done
    ldrb w0, [x19, x21]
    cmp  w0, #'.'
    b.ne .pn_exp
    add  x21, x21, #1
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo report_error
    cmp  w0, #'9'
    b.hi report_error
.pn_fracd:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs .pn_done
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo .pn_exp
    cmp  w0, #'9'
    b.hi .pn_exp
    b    .pn_fracd
.pn_exp:
    cmp  x21, x20
    b.hs .pn_done
    ldrb w0, [x19, x21]
    cmp  w0, #'e'
    b.eq .pn_exp2
    cmp  w0, #'E'
    b.eq .pn_exp2
    b    .pn_done
.pn_exp2:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'+'
    b.eq .pn_exps
    cmp  w0, #'-'
    b.eq .pn_exps
    b    .pn_expd1
.pn_exps:
    add  x21, x21, #1
.pn_expd1:
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo report_error
    cmp  w0, #'9'
    b.hi report_error
.pn_expd:
    add  x21, x21, #1
    cmp  x21, x20
    b.hs .pn_done
    ldrb w0, [x19, x21]
    cmp  w0, #'0'
    b.lo .pn_done
    cmp  w0, #'9'
    b.hi .pn_done
    b    .pn_expd
.pn_done:
    ret

parse_object:
    stp  x29, x30, [sp, #-16]!
    add  x21, x21, #1
    bl   skip_ws
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'}'
    b.eq .po_close
.po_member:
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #'"'
    b.ne report_error
    bl   parse_string
    bl   skip_ws
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #':'
    b.ne report_error
    add  x21, x21, #1
    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #','
    b.eq .po_comma
    cmp  w0, #'}'
    b.eq .po_close
    b    report_error
.po_comma:
    add  x21, x21, #1
    bl   skip_ws
    b    .po_member
.po_close:
    add  x21, x21, #1
    ldp  x29, x30, [sp], #16
    ret

parse_array:
    stp  x29, x30, [sp, #-16]!
    add  x21, x21, #1
    bl   skip_ws
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #']'
    b.eq .pa_close
.pa_elem:
    bl   parse_value
    bl   skip_ws
    cmp  x21, x20
    b.hs report_error
    ldrb w0, [x19, x21]
    cmp  w0, #','
    b.eq .pa_comma
    cmp  w0, #']'
    b.eq .pa_close
    b    report_error
.pa_comma:
    add  x21, x21, #1
    bl   skip_ws
    b    .pa_elem
.pa_close:
    add  x21, x21, #1
    ldp  x29, x30, [sp], #16
    ret

report_error:
    mov  x0, #1
    adrp x1, msg_err
    add  x1, x1, :lo12:msg_err
    mov  x2, #9
    mov  x8, #64
    svc  #0
    adrp x3, dbuf
    add  x3, x3, :lo12:dbuf
    add  x5, x3, #32
    mov  x4, x5
    mov  x6, x21
    mov  x7, #10
    cbnz x6, .re_loop
    mov  w11, #'0'
    sub  x5, x5, #1
    strb w11, [x5]
    b    .re_done
.re_loop:
    cbz  x6, .re_done
    udiv x9, x6, x7
    msub x10, x9, x7, x6
    add  w10, w10, #'0'
    sub  x5, x5, #1
    strb w10, [x5]
    mov  x6, x9
    b    .re_loop
.re_done:
    mov  x0, #1
    mov  x1, x5
    sub  x2, x4, x5
    mov  x8, #64
    svc  #0
    mov  x0, #1
    adrp x1, msg_nl
    add  x1, x1, :lo12:msg_nl
    mov  x2, #1
    mov  x8, #64
    svc  #0
    mov  x0, #1
    mov  x8, #93
    svc  #0

.bss
