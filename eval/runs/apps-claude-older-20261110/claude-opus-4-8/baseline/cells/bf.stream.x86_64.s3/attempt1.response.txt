.text
.globl _start

_start:
    xor r12, r12
read_loop:
    xor rax, rax
    xor rdi, rdi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jl read_loop
read_done:

    xor rbx, rbx
find_bang:
    cmp rbx, r12
    jge no_bang
    movzx eax, byte ptr [inbuf + rbx]
    cmp al, 33
    je found_bang
    inc rbx
    jmp find_bang
no_bang:
    mov r13, r12
    mov r14, r12
    jmp after_bang
found_bang:
    mov r13, rbx
    lea r14, [rbx + 1]
after_bang:

    xor rcx, rcx
    xor rbp, rbp
check_loop:
    cmp rcx, r13
    jge check_end
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, 91
    je push_bracket
    cmp al, 93
    je pop_bracket
    inc rcx
    jmp check_loop
push_bracket:
    mov dword ptr [brkstack + rbp*4], ecx
    inc rbp
    inc rcx
    jmp check_loop
pop_bracket:
    test rbp, rbp
    jz imbalance
    dec rbp
    mov edx, dword ptr [brkstack + rbp*4]
    mov dword ptr [jumptable + rdx*4], ecx
    mov dword ptr [jumptable + rcx*4], edx
    inc rcx
    jmp check_loop
check_end:
    test rbp, rbp
    jnz imbalance

    lea rbx, [rip + tape]
    xor rcx, rcx
    xor r15, r15
exec_loop:
    cmp rcx, r13
    jge exec_end
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, 62
    je do_gt
    cmp al, 60
    je do_lt
    cmp al, 43
    je do_plus
    cmp al, 45
    je do_minus
    cmp al, 46
    je do_dot
    cmp al, 44
    je do_comma
    cmp al, 91
    je do_open
    cmp al, 93
    je do_close
    inc rcx
    jmp exec_loop

do_gt:
    inc rbx
    inc rcx
    jmp exec_loop
do_lt:
    dec rbx
    inc rcx
    jmp exec_loop
do_plus:
    inc byte ptr [rbx]
    inc rcx
    jmp exec_loop
do_minus:
    dec byte ptr [rbx]
    inc rcx
    jmp exec_loop
do_dot:
    movzx eax, byte ptr [rbx]
    mov byte ptr [outbuf + r15], al
    inc r15
    inc rcx
    jmp exec_loop
do_comma:
    cmp r14, r12
    jge comma_zero
    movzx eax, byte ptr [inbuf + r14]
    inc r14
    mov byte ptr [rbx], al
    inc rcx
    jmp exec_loop
comma_zero:
    mov byte ptr [rbx], 0
    inc rcx
    jmp exec_loop
do_open:
    cmp byte ptr [rbx], 0
    jne 1f
    mov ecx, dword ptr [jumptable + rcx*4]
    inc rcx
    jmp exec_loop
1:
    inc rcx
    jmp exec_loop
do_close:
    cmp byte ptr [rbx], 0
    je 1f
    mov ecx, dword ptr [jumptable + rcx*4]
    inc rcx
    jmp exec_loop
1:
    inc rcx
    jmp exec_loop

exec_end:
    xor r14, r14
wloop:
    cmp r14, r15
    jge done
    mov rax, 1
    mov rdi, 1
    lea rsi, [outbuf + r14]
    mov rdx, r15
    sub rdx, r14
    syscall
    test rax, rax
    jle done
    add r14, rax
    jmp wloop
done:
    mov rax, 60
    xor rdi, rdi
    syscall

imbalance:
    mov rax, 60
    mov rdi, 1
    syscall

.bss
.align 16
inbuf:
    .zero 65536
tape:
    .zero 30000
outbuf:
    .zero 65536
brkstack:
    .zero 4096
jumptable:
    .zero 262144
