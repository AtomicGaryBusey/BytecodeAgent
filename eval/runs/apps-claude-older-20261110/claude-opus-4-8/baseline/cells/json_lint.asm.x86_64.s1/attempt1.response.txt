.text
.globl _start

_start:
    lea r14, [rip + inbuf]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r12]
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    mov r13, r12
    xor r12, r12

    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jb fail
    jmp ok

skip_ws:
    cmp r12, r13
    jae skip_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x20
    je skip_adv
    cmp al, 0x09
    je skip_adv
    cmp al, 0x0A
    je skip_adv
    cmp al, 0x0D
    je skip_adv
    ret
skip_adv:
    inc r12
    jmp skip_ws
skip_done:
    ret

match_literal:
    xor edx, edx
ml_loop:
    cmp r12, r13
    jae fail
    mov al, byte ptr [r14 + r12]
    cmp al, byte ptr [rsi + rdx]
    jne fail
    inc r12
    inc rdx
    cmp rdx, rcx
    jb ml_loop
    ret

parse_value:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6E
    je pv_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe parse_number
    jmp fail
pv_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp match_literal
pv_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp match_literal
pv_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
    jmp match_literal

parse_string:
    inc r12
ps_loop:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x22
    je ps_close
    cmp al, 0x5C
    je ps_escape
    cmp al, 0x20
    jb fail
    inc r12
    jmp ps_loop
ps_close:
    inc r12
    ret
ps_escape:
    inc r12
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x22
    je ps_simple
    cmp al, 0x5C
    je ps_simple
    cmp al, 0x2F
    je ps_simple
    cmp al, 0x62
    je ps_simple
    cmp al, 0x66
    je ps_simple
    cmp al, 0x6E
    je ps_simple
    cmp al, 0x72
    je ps_simple
    cmp al, 0x74
    je ps_simple
    cmp al, 0x75
    je ps_uesc
    jmp fail
ps_simple:
    inc r12
    jmp ps_loop
ps_uesc:
    inc r12
    mov ecx, 4
ps_hexloop:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe ps_hexok
    cmp al, 0x41
    jb fail
    cmp al, 0x46
    jbe ps_hexok
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    jbe ps_hexok
    jmp fail
ps_hexok:
    inc r12
    dec ecx
    jnz ps_hexloop
    jmp ps_loop

parse_number:
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x2D
    jne num_int
    inc r12
num_int:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    je num_zero
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
    inc r12
num_digits:
    cmp r12, r13
    jae num_frac
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    jb num_frac
    cmp al, 0x39
    ja num_frac
    inc r12
    jmp num_digits
num_zero:
    inc r12
num_frac:
    cmp r12, r13
    jae num_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x2E
    jne num_exp
    inc r12
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc r12
num_fracdigits:
    cmp r12, r13
    jae num_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    jb num_exp
    cmp al, 0x39
    ja num_exp
    inc r12
    jmp num_fracdigits
num_exp:
    cmp r12, r13
    jae num_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x65
    je num_exp2
    cmp al, 0x45
    je num_exp2
    jmp num_done
num_exp2:
    inc r12
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x2B
    je num_expsign
    cmp al, 0x2D
    je num_expsign
    jmp num_expdig
num_expsign:
    inc r12
num_expdig:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc r12
num_expdigits:
    cmp r12, r13
    jae num_done
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x30
    jb num_done
    cmp al, 0x39
    ja num_done
    inc r12
    jmp num_expdigits
num_done:
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x5D
    je arr_close
arr_elem:
    call parse_value
    call skip_ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x5D
    je arr_close
    cmp al, 0x2C
    jne fail
    inc r12
    call skip_ws
    jmp arr_elem
arr_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x7D
    je obj_close
obj_member:
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x3A
    jne fail
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    jae fail
    movzx eax, byte ptr [r14 + r12]
    cmp al, 0x7D
    je obj_close
    cmp al, 0x2C
    jne fail
    inc r12
    call skip_ws
    jmp obj_member
obj_close:
    inc r12
    ret

ok:
    lea rsi, [rip + str_ok]
    mov edx, 3
    call write1
    mov eax, 60
    xor edi, edi
    syscall

fail:
    lea rsi, [rip + str_error]
    mov edx, 9
    call write1
    mov rax, r12
    lea rdi, [rip + numbuf + 32]
    mov r9, rdi
    mov ebx, 10
conv_loop:
    xor edx, edx
    div rbx
    add dl, 0x30
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz conv_loop
    mov rsi, rdi
    mov rdx, r9
    sub rdx, rdi
    call write1
    lea rsi, [rip + str_nl]
    mov edx, 1
    call write1
    mov eax, 60
    mov edi, 1
    syscall

write1:
    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
str_true:  .ascii "true"
str_false: .ascii "false"
str_null:  .ascii "null"
str_error: .ascii "error at "
str_ok:    .ascii "ok\n"
str_nl:    .byte 10

.bss
.align 16
inbuf:  .zero 65536
numbuf: .zero 32
