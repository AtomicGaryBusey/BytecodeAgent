.globl _start

.equ msg_ok_len, 3
.equ msg_error_len, 9

.text

_start:
    lea r13, [rip + buffer]
    xor r14, r14
read_loop:
    mov edx, 70000
    sub rdx, r14
    jle read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r14]
    syscall
    cmp rax, 0
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    mov r12, r14

    xor ebx, ebx
    call skip_ws
    call parse_value
    test eax, eax
    jnz do_error
    call skip_ws
    cmp rbx, r12
    jne do_error

do_ok:
    lea rsi, [rip + msg_ok]
    mov edx, msg_ok_len
    call write_stdout
    xor edi, edi
    jmp do_exit

do_error:
    lea rsi, [rip + msg_error]
    mov edx, msg_error_len
    call write_stdout
    mov rax, rbx
    lea rdi, [rip + numbuf]
    add rdi, 31
    mov byte ptr [rdi], 0x0A
    dec rdi
    mov ecx, 10
conv_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    mov [rdi], dl
    dec rdi
    test rax, rax
    jnz conv_loop
    inc rdi
    lea rdx, [rip + numbuf]
    add rdx, 32
    sub rdx, rdi
    mov rsi, rdi
    call write_stdout
    mov edi, 1
    jmp do_exit

do_exit:
    mov eax, 60
    syscall

write_stdout:
    mov eax, 1
    mov edi, 1
    syscall
    ret

skip_ws:
    cmp rbx, r12
    jae sw_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0A
    je sw_adv
    cmp al, 0x0D
    je sw_adv
    ret
sw_adv:
    inc rbx
    jmp skip_ws
sw_done:
    ret

parse_value:
    cmp rbx, r12
    jae pv_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb pv_fail
    cmp al, '9'
    jbe parse_number
pv_fail:
    mov eax, 1
    ret

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '}'
    je po_end
po_member:
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    jne po_fail
    call parse_string
    test eax, eax
    jnz po_ret_fail
    call skip_ws
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ':'
    jne po_fail
    inc rbx
    call skip_ws
    call parse_value
    test eax, eax
    jnz po_ret_fail
    call skip_ws
    cmp rbx, r12
    jae po_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '}'
    je po_end
    cmp al, ','
    jne po_fail
    inc rbx
    call skip_ws
    jmp po_member
po_end:
    inc rbx
    xor eax, eax
    ret
po_fail:
    mov eax, 1
po_ret_fail:
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r12
    jae pa_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ']'
    je pa_end
pa_elem:
    call parse_value
    test eax, eax
    jnz pa_ret_fail
    call skip_ws
    cmp rbx, r12
    jae pa_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, ']'
    je pa_end
    cmp al, ','
    jne pa_fail
    inc rbx
    call skip_ws
    jmp pa_elem
pa_end:
    inc rbx
    xor eax, eax
    ret
pa_fail:
    mov eax, 1
pa_ret_fail:
    ret

parse_string:
    inc rbx
ps_loop:
    cmp rbx, r12
    jae ps_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je ps_end
    cmp al, 0x5C
    je ps_escape
    cmp al, 0x20
    jb ps_fail
    inc rbx
    jmp ps_loop
ps_end:
    inc rbx
    xor eax, eax
    ret
ps_fail:
    mov eax, 1
    ret
ps_escape:
    inc rbx
    cmp rbx, r12
    jae ps_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '"'
    je ps_esc_simple
    cmp al, 0x5C
    je ps_esc_simple
    cmp al, '/'
    je ps_esc_simple
    cmp al, 'b'
    je ps_esc_simple
    cmp al, 'f'
    je ps_esc_simple
    cmp al, 'n'
    je ps_esc_simple
    cmp al, 'r'
    je ps_esc_simple
    cmp al, 't'
    je ps_esc_simple
    cmp al, 'u'
    je ps_esc_u
    jmp ps_fail
ps_esc_simple:
    inc rbx
    jmp ps_loop
ps_esc_u:
    inc rbx
    mov ecx, 4
ps_hex_loop:
    cmp rbx, r12
    jae ps_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb ps_fail
    cmp al, '9'
    jbe ps_hex_ok
    cmp al, 'A'
    jb ps_fail
    cmp al, 'F'
    jbe ps_hex_ok
    cmp al, 'a'
    jb ps_fail
    cmp al, 'f'
    jbe ps_hex_ok
    jmp ps_fail
ps_hex_ok:
    inc rbx
    dec ecx
    jnz ps_hex_loop
    jmp ps_loop

parse_number:
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '-'
    jne pn_int
    inc rbx
pn_int:
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb pn_fail
    je pn_zero
    cmp al, '9'
    ja pn_fail
    inc rbx
pn_digits:
    cmp rbx, r12
    jae pn_frac_check
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb pn_frac_check
    cmp al, '9'
    ja pn_frac_check
    inc rbx
    jmp pn_digits
pn_zero:
    inc rbx
pn_frac_check:
    cmp rbx, r12
    jae pn_exp_check
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '.'
    jne pn_exp_check
    inc rbx
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
    inc rbx
pn_frac_digits:
    cmp rbx, r12
    jae pn_exp_check
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb pn_exp_check
    cmp al, '9'
    ja pn_exp_check
    inc rbx
    jmp pn_frac_digits
pn_exp_check:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, 'e'
    je pn_exp
    cmp al, 'E'
    je pn_exp
    jmp pn_done
pn_exp:
    inc rbx
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '+'
    je pn_exp_sign
    cmp al, '-'
    je pn_exp_sign
    jmp pn_exp_digit0
pn_exp_sign:
    inc rbx
pn_exp_digit0:
    cmp rbx, r12
    jae pn_fail
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb pn_fail
    cmp al, '9'
    ja pn_fail
    inc rbx
pn_exp_digits:
    cmp rbx, r12
    jae pn_done
    movzx eax, byte ptr [r13 + rbx]
    cmp al, '0'
    jb pn_done
    cmp al, '9'
    ja pn_done
    inc rbx
    jmp pn_exp_digits
pn_done:
    xor eax, eax
    ret
pn_fail:
    mov eax, 1
    ret

parse_true:
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'r'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'u'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'e'
    jne lit_fail
    inc rbx
    xor eax, eax
    ret

parse_false:
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'a'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'l'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 's'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'e'
    jne lit_fail
    inc rbx
    xor eax, eax
    ret

parse_null:
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'u'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'l'
    jne lit_fail
    inc rbx
    cmp rbx, r12
    jae lit_fail
    cmp byte ptr [r13 + rbx], 'l'
    jne lit_fail
    inc rbx
    xor eax, eax
    ret

lit_fail:
    mov eax, 1
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_error:
    .ascii "error at "

.bss
.align 16
numbuf:
    .zero 32
buffer:
    .zero 70016
