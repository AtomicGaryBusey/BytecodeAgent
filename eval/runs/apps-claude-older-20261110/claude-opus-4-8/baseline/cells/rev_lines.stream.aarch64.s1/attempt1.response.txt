.globl _start

.text
_start:
    // x19 = inbuf, x20 = n (bytes read)
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0

read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    // x21 = outbuf, x22 = o
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x22, #0

    mov     x9, #0          // i
outer:
    cmp     x9, x20
    bge     done
    // find newline starting at i
    mov     x10, x9
inner:
    cmp     x10, x20
    bge     found
    ldrb    w11, [x19, x10]
    cmp     w11, #10
    beq     found
    add     x10, x10, #1
    b       inner
found:
    // line bytes [x9 .. x10-1], reverse-copy
    sub     x12, x10, #1
rev_loop:
    cmp     x12, x9
    blt     rev_done
    ldrb    w11, [x19, x12]
    strb    w11, [x21, x22]
    add     x22, x22, #1
    sub     x12, x12, #1
    b       rev_loop
rev_done:
    mov     w11, #10
    strb    w11, [x21, x22]
    add     x22, x22, #1
    add     x9, x10, #1
    b       outer
done:
    // write outbuf
    mov     x0, #1
    mov     x1, x21
    mov     x2, x22
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0

.bss
.align 4
inbuf:
    .space 65536
outbuf:
    .space 65600
