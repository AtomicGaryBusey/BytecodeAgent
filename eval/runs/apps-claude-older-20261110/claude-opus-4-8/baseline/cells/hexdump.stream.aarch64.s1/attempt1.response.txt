.section .bss
inbuf:
    .zero 65536
outbuf:
    .zero 128

.section .rodata
hextable:
    .ascii "0123456789abcdef"

.text
.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
read_loop:
    mov     x0, #0
    add     x1, x19, x20
    movz    w2, #1, lsl #16
    sub     w2, w2, w20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x22, hextable
    add     x22, x22, :lo12:hextable

    mov     x23, #0
row_loop:
    cmp     x23, x20
    b.ge    rows_done
    sub     x24, x20, x23
    mov     x25, #16
    cmp     x24, #16
    csel    x24, x24, x25, lt

    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x26, x21

    mov     w0, w23
    mov     w5, #28
hex_off_loop:
    lsr     w6, w0, w5
    and     w6, w6, #0xf
    ldrb    w7, [x22, x6]
    strb    w7, [x21], #1
    subs    w5, w5, #4
    b.ge    hex_off_loop

    mov     w7, #0x20
    strb    w7, [x21], #1
    strb    w7, [x21], #1

    mov     x27, #0
byte_loop:
    cmp     x27, #16
    b.ge    byte_done
    cmp     x27, x24
    b.ge    pad3
    add     x9, x19, x23
    ldrb    w4, [x9, x27]
    lsr     w6, w4, #4
    and     w6, w6, #0xf
    ldrb    w7, [x22, x6]
    strb    w7, [x21], #1
    and     w6, w4, #0xf
    ldrb    w7, [x22, x6]
    strb    w7, [x21], #1
    mov     w7, #0x20
    strb    w7, [x21], #1
    b       after_slot
pad3:
    mov     w7, #0x20
    strb    w7, [x21], #1
    strb    w7, [x21], #1
    strb    w7, [x21], #1
after_slot:
    cmp     x27, #7
    b.ne    noextra
    mov     w7, #0x20
    strb    w7, [x21], #1
noextra:
    add     x27, x27, #1
    b       byte_loop
byte_done:
    mov     w7, #0x20
    strb    w7, [x21], #1
    mov     w7, #0x7c
    strb    w7, [x21], #1

    mov     x27, #0
char_loop:
    cmp     x27, x24
    b.ge    char_done
    add     x9, x19, x23
    ldrb    w4, [x9, x27]
    cmp     w4, #0x20
    b.lt    dot
    cmp     w4, #0x7e
    b.gt    dot
    b       putc
dot:
    mov     w4, #0x2e
putc:
    strb    w4, [x21], #1
    add     x27, x27, #1
    b       char_loop
char_done:
    mov     w7, #0x7c
    strb    w7, [x21], #1
    mov     w7, #0x0a
    strb    w7, [x21], #1

    mov     x0, #1
    mov     x1, x26
    sub     x2, x21, x26
    mov     x8, #64
    svc     #0

    add     x23, x23, #16
    b       row_loop
rows_done:
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x26, x21
    mov     w0, w20
    mov     w5, #28
tot_loop:
    lsr     w6, w0, w5
    and     w6, w6, #0xf
    ldrb    w7, [x22, x6]
    strb    w7, [x21], #1
    subs    w5, w5, #4
    b.ge    tot_loop
    mov     w7, #0x0a
    strb    w7, [x21], #1
    mov     x0, #1
    mov     x1, x26
    sub     x2, x21, x26
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0
