.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
in_buf:
.zero 49160
out_buf:
.zero 70000

.text
.globl _start

_start:
    adrp x19, in_buf
    add  x19, x19, :lo12:in_buf
    adrp x25, out_buf
    add  x25, x25, :lo12:out_buf
    adrp x23, alphabet
    add  x23, x23, :lo12:alphabet
    mov  x21, x25

    // read all input
    mov  x24, #0
read_loop:
    mov  x0, #0
    add  x1, x19, x24
    mov  x2, #49152
    sub  x2, x2, x24
    cbz  x2, read_done
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    ble  read_done
    add  x24, x24, x0
    b    read_loop
read_done:
    mov  x20, x24

    // encode
    mov  x24, #0
    mov  x22, #0
enc_loop:
    sub  x9, x20, x24
    cmp  x9, #3
    b.lt rem
    ldrb w10, [x19, x24]
    add  x11, x24, #1
    ldrb w11, [x19, x11]
    add  x12, x24, #2
    ldrb w12, [x19, x12]
    add  x24, x24, #3
    lsr  w0, w10, #2
    bl   emit6
    and  w0, w10, #3
    lsl  w0, w0, #4
    lsr  w1, w11, #4
    orr  w0, w0, w1
    bl   emit6
    and  w0, w11, #15
    lsl  w0, w0, #2
    lsr  w1, w12, #6
    orr  w0, w0, w1
    bl   emit6
    and  w0, w12, #63
    bl   emit6
    b    enc_loop

rem:
    cbz  x9, finish
    cmp  x9, #1
    b.eq one
    // two bytes
    ldrb w10, [x19, x24]
    add  x11, x24, #1
    ldrb w11, [x19, x11]
    lsr  w0, w10, #2
    bl   emit6
    and  w0, w10, #3
    lsl  w0, w0, #4
    lsr  w1, w11, #4
    orr  w0, w0, w1
    bl   emit6
    and  w0, w11, #15
    lsl  w0, w0, #2
    bl   emit6
    mov  w0, #0x3D
    bl   emitraw
    b    finish
one:
    ldrb w10, [x19, x24]
    lsr  w0, w10, #2
    bl   emit6
    and  w0, w10, #3
    lsl  w0, w0, #4
    bl   emit6
    mov  w0, #0x3D
    bl   emitraw
    mov  w0, #0x3D
    bl   emitraw

finish:
    cbz  x22, after_nl
    mov  w0, #0x0A
    strb w0, [x21], #1
after_nl:

    // write output
    mov  x1, x25
    sub  x2, x21, x25
wloop:
    cbz  x2, wdone
    mov  x0, #1
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    ble  wdone
    add  x1, x1, x0
    sub  x2, x2, x0
    b    wloop
wdone:
    mov  x0, #0
    mov  x8, #93
    svc  #0

emit6:
    ldrb w0, [x23, x0]
emitraw:
    strb w0, [x21], #1
    add  x22, x22, #1
    cmp  x22, #76
    b.ne 1f
    mov  w0, #0x0A
    strb w0, [x21], #1
    mov  x22, #0
1:
    ret
