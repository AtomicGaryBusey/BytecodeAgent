.text
.globl _start

_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [inbuf]
    add rsi, r12
    mov edx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    xor rbx, rbx
    xor r13, r13
    xor r14, r14
parse_loop:
    cmp r14, r12
    jae parse_end
    movzx eax, byte ptr [inbuf + r14]
    cmp al, 0x0A
    jne 1f
    mov ecx, r13d
    mov [starts + rbx*4], ecx
    mov eax, r14d
    sub eax, r13d
    mov [lens + rbx*4], eax
    inc rbx
    lea r13, [r14 + 1]
1:
    inc r14
    jmp parse_loop
parse_end:
    cmp r13, r12
    jae no_final
    mov ecx, r13d
    mov [starts + rbx*4], ecx
    mov eax, r12d
    sub eax, r13d
    mov [lens + rbx*4], eax
    inc rbx
no_final:

    mov r15, 1
outer:
    cmp r15, rbx
    jae sort_done
    mov r12d, [starts + r15*4]
    mov r13d, [lens + r15*4]
    mov r14, r15
    dec r14
inner:
    test r14, r14
    js place
    mov edi, [starts + r14*4]
    mov esi, [lens + r14*4]
    mov edx, r12d
    mov ecx, r13d
    call cmp_lines
    test eax, eax
    jle place
    mov eax, [starts + r14*4]
    mov [starts + r14*4 + 4], eax
    mov eax, [lens + r14*4]
    mov [lens + r14*4 + 4], eax
    dec r14
    jmp inner
place:
    mov [starts + r14*4 + 4], r12d
    mov [lens + r14*4 + 4], r13d
    inc r15
    jmp outer
sort_done:

    xor r15, r15
    xor r14, r14
build:
    cmp r15, rbx
    jae build_done
    mov esi, [starts + r15*4]
    mov ecx, [lens + r15*4]
    lea rax, [inbuf]
    add rsi, rax
    lea rdi, [outbuf]
    add rdi, r14
copyloop:
    test rcx, rcx
    jz copydone
    mov al, [rsi]
    mov [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jmp copyloop
copydone:
    mov byte ptr [rdi], 0x0A
    mov ecx, [lens + r15*4]
    add r14, rcx
    inc r14
    inc r15
    jmp build
build_done:

    mov eax, 1
    mov edi, 1
    lea rsi, [outbuf]
    mov rdx, r14
    syscall

    mov eax, 60
    xor edi, edi
    syscall

cmp_lines:
    mov r8, rsi
    mov r9, rcx
    mov r10, r8
    cmp r9, r10
    cmovb r10, r9
    lea r11, [inbuf]
    add rdi, r11
    add rdx, r11
    xor r11, r11
cmp_byte:
    cmp r11, r10
    jae cmp_eqlen
    movzx eax, byte ptr [rdi + r11]
    movzx ecx, byte ptr [rdx + r11]
    sub eax, ecx
    jnz cmp_ret
    inc r11
    jmp cmp_byte
cmp_eqlen:
    mov eax, r8d
    sub eax, r9d
cmp_ret:
    ret

.bss
.balign 16
inbuf:
    .zero 32768
starts:
    .zero 4000
lens:
    .zero 4000
outbuf:
    .zero 40000
