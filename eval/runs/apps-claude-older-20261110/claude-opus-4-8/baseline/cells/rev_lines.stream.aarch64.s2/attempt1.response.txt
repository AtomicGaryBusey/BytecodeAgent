.globl _start

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    mov     x20, #0            // total read

read_loop:
    mov     x8, #63
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    cmp     x2, #0
    b.eq    read_done
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    // x24 = out length
    mov     x24, #0
    mov     x23, #0            // line_start
    mov     x25, #0            // i

scan_loop:
    cmp     x25, x20
    b.ge    scan_end
    ldrb    w0, [x19, x25]
    cmp     w0, #0x0A
    b.ne    scan_next
    // emit line [x23, x25)
    bl      emit_line
    add     x23, x25, #1
scan_next:
    add     x25, x25, #1
    b       scan_loop

scan_end:
    // final line if line_start < total
    cmp     x23, x20
    b.ge    do_write
    bl      emit_line

do_write:
    mov     x8, #64
    mov     x0, #1
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    mov     x2, x24
    svc     #0

    mov     x8, #93
    mov     x0, #0
    svc     #0

// emit reversed bytes from [x23, x25) into outbuf, then newline
// uses x0,x1,x2
emit_line:
    mov     x1, x25            // e
emit_copy:
    cmp     x1, x23
    b.le    emit_nl
    sub     x1, x1, #1
    ldrb    w0, [x19, x1]
    strb    w0, [x21, x24]
    add     x24, x24, #1
    b       emit_copy
emit_nl:
    mov     w0, #0x0A
    strb    w0, [x21, x24]
    add     x24, x24, #1
    ret

.bss
.align 4
inbuf:
    .zero   65536
outbuf:
    .zero   70000
