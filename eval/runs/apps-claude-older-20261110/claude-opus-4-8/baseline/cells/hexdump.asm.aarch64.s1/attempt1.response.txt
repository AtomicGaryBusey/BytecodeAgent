.section .rodata
hextab:
	.ascii "0123456789abcdef"

.bss
	.balign 16
inbuf:
	.zero 65536
outbuf:
	.zero 524288

.text
	.globl _start
_start:
	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x20, #0
read_loop:
	mov	x0, #0
	add	x1, x19, x20
	mov	x2, #65536
	sub	x2, x2, x20
	cbz	x2, read_done
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	ble	read_done
	add	x20, x20, x0
	b	read_loop
read_done:

	adrp	x21, outbuf
	add	x21, x21, :lo12:outbuf
	adrp	x23, hextab
	add	x23, x23, :lo12:hextab
	mov	x22, #0

row_loop:
	cmp	x22, x20
	bge	rows_done
	sub	x24, x20, x22
	cmp	x24, #16
	mov	x25, #16
	csel	x25, x25, x24, gt

	mov	w0, w22
	bl	append_hex8
	mov	w0, #0x20
	strb	w0, [x21], #1
	strb	w0, [x21], #1

	mov	x26, #0
byte_loop:
	cmp	x26, #16
	bge	byte_done
	cmp	x26, x25
	bge	pad3
	add	x27, x22, x26
	ldrb	w0, [x19, x27]
	bl	append_hex2
	mov	w0, #0x20
	strb	w0, [x21], #1
	b	after_slot
pad3:
	mov	w0, #0x20
	strb	w0, [x21], #1
	strb	w0, [x21], #1
	strb	w0, [x21], #1
after_slot:
	cmp	x26, #7
	bne	noextra
	mov	w0, #0x20
	strb	w0, [x21], #1
noextra:
	add	x26, x26, #1
	b	byte_loop
byte_done:
	mov	w0, #0x20
	strb	w0, [x21], #1
	mov	w0, #0x7c
	strb	w0, [x21], #1
	mov	x26, #0
ascii_loop:
	cmp	x26, x25
	bge	ascii_done
	add	x27, x22, x26
	ldrb	w0, [x19, x27]
	cmp	w0, #0x20
	blt	dot
	cmp	w0, #0x7e
	bgt	dot
	b	putc
dot:
	mov	w0, #0x2e
putc:
	strb	w0, [x21], #1
	add	x26, x26, #1
	b	ascii_loop
ascii_done:
	mov	w0, #0x7c
	strb	w0, [x21], #1
	mov	w0, #0x0a
	strb	w0, [x21], #1
	add	x22, x22, #16
	b	row_loop

rows_done:
	mov	w0, w20
	bl	append_hex8
	mov	w0, #0x0a
	strb	w0, [x21], #1

	mov	x0, #1
	adrp	x1, outbuf
	add	x1, x1, :lo12:outbuf
	sub	x2, x21, x1
	mov	x8, #64
	svc	#0

	mov	x0, #0
	mov	x8, #93
	svc	#0

append_hex8:
	mov	w1, w0
	mov	w2, #28
1:
	lsr	w3, w1, w2
	and	w3, w3, #0xf
	ldrb	w3, [x23, x3]
	strb	w3, [x21], #1
	subs	w2, w2, #4
	bge	1b
	ret

append_hex2:
	and	w1, w0, #0xf0
	lsr	w1, w1, #4
	ldrb	w1, [x23, x1]
	strb	w1, [x21], #1
	and	w1, w0, #0xf
	ldrb	w1, [x23, x1]
	strb	w1, [x21], #1
	ret
