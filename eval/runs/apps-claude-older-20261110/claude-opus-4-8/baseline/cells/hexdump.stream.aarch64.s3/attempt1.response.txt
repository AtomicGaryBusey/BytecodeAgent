.globl _start

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x21, outbuf
    add     x21, x21, :lo12:outbuf
    adrp    x23, hextab
    add     x23, x23, :lo12:hextab
    mov     w22, #0                 // count

read_loop:
    mov     w3, #65536
    subs    w3, w3, w22
    b.eq    read_done
    mov     x0, #0
    add     x1, x19, w22, uxtw
    mov     w2, w3
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     w22, w22, w0
    b       read_loop
read_done:

    mov     w24, #0                 // off
row_loop:
    cmp     w24, w22
    b.hs    rows_done
    sub     w4, w22, w24
    cmp     w4, #16
    b.lo    1f
    mov     w4, #16
1:
    // offset 8 hex digits
    mov     w5, #28
2:
    lsr     w6, w24, w5
    and     w6, w6, #0xf
    ldrb    w7, [x23, w6, uxtw]
    strb    w7, [x21], #1
    subs    w5, w5, #4
    b.ge    2b
    mov     w7, #0x20
    strb    w7, [x21], #1
    strb    w7, [x21], #1
    // byte columns
    mov     w5, #0
3:
    cmp     w5, w4
    b.hs    4f
    add     w9, w24, w5
    ldrb    w10, [x19, w9, uxtw]
    lsr     w11, w10, #4
    and     w11, w11, #0xf
    ldrb    w12, [x23, w11, uxtw]
    strb    w12, [x21], #1
    and     w11, w10, #0xf
    ldrb    w12, [x23, w11, uxtw]
    strb    w12, [x21], #1
    mov     w12, #0x20
    strb    w12, [x21], #1
    b       5f
4:
    mov     w12, #0x20
    strb    w12, [x21], #1
    strb    w12, [x21], #1
    strb    w12, [x21], #1
5:
    cmp     w5, #7
    b.ne    6f
    mov     w12, #0x20
    strb    w12, [x21], #1
6:
    add     w5, w5, #1
    cmp     w5, #16
    b.lo    3b
    // one space then |
    mov     w12, #0x20
    strb    w12, [x21], #1
    mov     w12, #0x7c
    strb    w12, [x21], #1
    // char column
    mov     w5, #0
7:
    cmp     w5, w4
    b.hs    8f
    add     w9, w24, w5
    ldrb    w10, [x19, w9, uxtw]
    cmp     w10, #0x20
    b.lo    9f
    cmp     w10, #0x7e
    b.hi    9f
    b       10f
9:
    mov     w10, #0x2e
10:
    strb    w10, [x21], #1
    add     w5, w5, #1
    b       7b
8:
    mov     w12, #0x7c
    strb    w12, [x21], #1
    mov     w12, #0x0a
    strb    w12, [x21], #1
    add     w24, w24, #16
    b       row_loop
rows_done:

    // final total length line
    mov     w5, #28
11:
    lsr     w6, w22, w5
    and     w6, w6, #0xf
    ldrb    w7, [x23, w6, uxtw]
    strb    w7, [x21], #1
    subs    w5, w5, #4
    b.ge    11b
    mov     w7, #0x0a
    strb    w7, [x21], #1

    // write output
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x21, x1
write_loop:
    cbz     x2, write_done
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    write_done
    add     x1, x1, x0
    sub     x2, x2, x0
    b       write_loop
write_done:

    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
hextab:
    .ascii  "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero   65536
outbuf:
    .zero   524288
