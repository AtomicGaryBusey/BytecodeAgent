.text
.globl _start
_start:
    xor rbx, rbx
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    jz done_read
    syscall
    test rax, rax
    jz done_read
    add rbx, rax
    jmp read_loop
done_read:
    xor r12, r12
    xor r8, r8
    xor rcx, rcx
scan:
    cmp rcx, rbx
    jge after
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, 0x0A
    je found
    inc rcx
    jmp scan
found:
    call do_reverse
    inc rcx
    mov r8, rcx
    jmp scan
after:
    cmp r8, rbx
    jge write_out
    mov rcx, rbx
    call do_reverse
write_out:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r12
    syscall
    mov eax, 60
    xor edi, edi
    syscall

do_reverse:
    mov r9, rcx
.rl:
    cmp r9, r8
    jle .rend
    dec r9
    movzx eax, byte ptr [inbuf + r9]
    mov byte ptr [outbuf + r12], al
    inc r12
    jmp .rl
.rend:
    mov byte ptr [outbuf + r12], 0x0A
    inc r12
    ret

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
