.text
.globl _start

_start:
    xor rbx, rbx
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov rax, 0
    xor rdi, rdi
    mov rdx, 49152
    sub rdx, rbx
    jz done_read
    syscall
    test rax, rax
    jz done_read
    add rbx, rax
    jmp read_loop

done_read:
    lea rsi, [rip + inbuf]
    mov rcx, rbx
    lea rdi, [rip + outbuf]
    lea r14, [rip + table]
    xor r13, r13

g3:
    cmp rcx, 3
    jb tail
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]
    add rsi, 3
    sub rcx, 3

    mov eax, r8d
    shr eax, 2
    call emitidx

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    call emitidx

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    call emitidx

    mov eax, r10d
    and eax, 63
    call emitidx
    jmp g3

tail:
    test rcx, rcx
    jz finish
    cmp rcx, 1
    je one

    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    mov eax, r8d
    shr eax, 2
    call emitidx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    call emitidx
    mov eax, r9d
    and eax, 15
    shl eax, 2
    call emitidx
    mov al, 0x3d
    call emitc
    jmp finish

one:
    movzx r8d, byte ptr [rsi]
    mov eax, r8d
    shr eax, 2
    call emitidx
    mov eax, r8d
    and eax, 3
    shl eax, 4
    call emitidx
    mov al, 0x3d
    call emitc
    mov al, 0x3d
    call emitc

finish:
    test r13, r13
    jz writeout
    mov byte ptr [rdi], 10
    inc rdi

writeout:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

emitidx:
    movzx eax, byte ptr [r14 + rax]
emitc:
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r13, r13
1:
    ret

.section .rodata
table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072
