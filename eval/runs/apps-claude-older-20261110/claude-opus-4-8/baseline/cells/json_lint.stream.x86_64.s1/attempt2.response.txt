.globl _start

.equ BUFSIZE, 70000

.text

skip_ws:
sw_loop:
    cmp r13, r14
    jae sw_done
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x20
    je sw_adv
    cmp al, 0x09
    je sw_adv
    cmp al, 0x0A
    je sw_adv
    cmp al, 0x0D
    je sw_adv
    jmp sw_done
sw_adv:
    inc r13
    jmp sw_loop
sw_done:
    ret

parse_literal:
pl_loop:
    test rcx, rcx
    jz pl_ok
    cmp r13, r14
    jae pl_fail
    movzx eax, byte ptr [r12 + r13]
    movzx edx, byte ptr [rsi]
    cmp al, dl
    jne pl_fail
    inc r13
    inc rsi
    dec rcx
    jmp pl_loop
pl_ok:
    mov eax, 1
    ret
pl_fail:
    xor eax, eax
    ret

parse_value:
    cmp r13, r14
    jae pv_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je pv_true
    cmp al, 0x66
    je pv_false
    cmp al, 0x6E
    je pv_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb pv_fail
    cmp al, 0x39
    jbe parse_number
pv_fail:
    xor eax, eax
    ret
pv_true:
    lea rsi, [rip + s_true]
    mov ecx, 4
    jmp parse_literal
pv_false:
    lea rsi, [rip + s_false]
    mov ecx, 5
    jmp parse_literal
pv_null:
    lea rsi, [rip + s_null]
    mov ecx, 4
    jmp parse_literal

parse_string:
    inc r13
pstr_loop:
    cmp r13, r14
    jae pstr_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x22
    je pstr_close
    cmp al, 0x5C
    je pstr_esc
    cmp al, 0x20
    jb pstr_fail
    inc r13
    jmp pstr_loop
pstr_close:
    inc r13
    mov eax, 1
    ret
pstr_esc:
    inc r13
    cmp r13, r14
    jae pstr_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x22
    je pstr_esc1
    cmp al, 0x5C
    je pstr_esc1
    cmp al, 0x2F
    je pstr_esc1
    cmp al, 0x62
    je pstr_esc1
    cmp al, 0x66
    je pstr_esc1
    cmp al, 0x6E
    je pstr_esc1
    cmp al, 0x72
    je pstr_esc1
    cmp al, 0x74
    je pstr_esc1
    cmp al, 0x75
    je pstr_escu
    jmp pstr_fail
pstr_esc1:
    inc r13
    jmp pstr_loop
pstr_escu:
    inc r13
    mov ecx, 4
pstr_hex:
    cmp r13, r14
    jae pstr_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb pstr_fail
    cmp al, 0x39
    jbe pstr_hexok
    cmp al, 0x41
    jb pstr_fail
    cmp al, 0x46
    jbe pstr_hexok
    cmp al, 0x61
    jb pstr_fail
    cmp al, 0x66
    jbe pstr_hexok
    jmp pstr_fail
pstr_hexok:
    inc r13
    dec ecx
    jnz pstr_hex
    jmp pstr_loop
pstr_fail:
    xor eax, eax
    ret

parse_number:
    cmp r13, r14
    jae pn_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2D
    jne pn_intpart
    inc r13
pn_intpart:
    cmp r13, r14
    jae pn_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    je pn_zero
    cmp al, 0x31
    jb pn_fail
    cmp al, 0x39
    ja pn_fail
    inc r13
pn_digits:
    cmp r13, r14
    jae pn_frac
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb pn_frac
    cmp al, 0x39
    ja pn_frac
    inc r13
    jmp pn_digits
pn_zero:
    inc r13
pn_frac:
    cmp r13, r14
    jae pn_done
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2E
    jne pn_exp
    inc r13
    cmp r13, r14
    jae pn_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb pn_fail
    cmp al, 0x39
    ja pn_fail
    inc r13
pn_fracdig:
    cmp r13, r14
    jae pn_done
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb pn_exp
    cmp al, 0x39
    ja pn_exp
    inc r13
    jmp pn_fracdig
pn_exp:
    cmp r13, r14
    jae pn_done
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x65
    je pn_exps
    cmp al, 0x45
    je pn_exps
    jmp pn_done
pn_exps:
    inc r13
    cmp r13, r14
    jae pn_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2B
    je pn_expsign
    cmp al, 0x2D
    je pn_expsign
    jmp pn_expd1
pn_expsign:
    inc r13
pn_expd1:
    cmp r13, r14
    jae pn_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb pn_fail
    cmp al, 0x39
    ja pn_fail
    inc r13
pn_expdig:
    cmp r13, r14
    jae pn_done
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb pn_done
    cmp al, 0x39
    ja pn_done
    inc r13
    jmp pn_expdig
pn_done:
    mov eax, 1
    ret
pn_fail:
    xor eax, eax
    ret

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae po_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x7D
    je po_close
po_member:
    cmp r13, r14
    jae po_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x22
    jne po_fail
    call parse_string
    test al, al
    jz po_retfail
    call skip_ws
    cmp r13, r14
    jae po_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x3A
    jne po_fail
    inc r13
    call skip_ws
    call parse_value
    test al, al
    jz po_retfail
    call skip_ws
    cmp r13, r14
    jae po_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2C
    je po_comma
    cmp al, 0x7D
    je po_close
    jmp po_fail
po_comma:
    inc r13
    call skip_ws
    jmp po_member
po_close:
    inc r13
    mov eax, 1
    ret
po_fail:
    xor eax, eax
po_retfail:
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae pa_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x5D
    je pa_close
pa_elem:
    call parse_value
    test al, al
    jz pa_retfail
    call skip_ws
    cmp r13, r14
    jae pa_fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2C
    je pa_comma
    cmp al, 0x5D
    je pa_close
    jmp pa_fail
pa_comma:
    inc r13
    call skip_ws
    jmp pa_elem
pa_close:
    inc r13
    mov eax, 1
    ret
pa_fail:
    xor eax, eax
pa_retfail:
    ret

write_all:
wa_loop:
    test rdx, rdx
    jz wa_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wa_done
    add rsi, rax
    sub rdx, rax
    jmp wa_loop
wa_done:
    ret

_start:
    lea r12, [rip + buffer]
    xor r14, r14
read_loop:
    mov rdx, BUFSIZE
    sub rdx, r14
    jbe read_done
    lea rsi, [r12 + r14]
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r13, r13
    call skip_ws
    call parse_value
    test al, al
    jz do_err
    call skip_ws
    cmp r13, r14
    jne do_err

    lea rsi, [rip + msg_ok]
    mov edx, 3
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

do_err:
    mov rax, r13
    lea r15, [rip + numbuf + 32]
    mov rdi, r15
    mov ebx, 10
    test rax, rax
    jnz itoa_loop
    dec rdi
    mov byte ptr [rdi], 0x30
    jmp itoa_done
itoa_loop:
    xor edx, edx
    div rbx
    add dl, 0x30
    dec rdi
    mov [rdi], dl
    test rax, rax
    jnz itoa_loop
itoa_done:
    mov rbx, rdi
    lea rsi, [rip + msg_err]
    mov edx, 9
    call write_all
    mov rsi, rbx
    mov rdx, r15
    sub rdx, rbx
    call write_all
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
s_true:
    .ascii "true"
s_false:
    .ascii "false"
s_null:
    .ascii "null"
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
newline:
    .ascii "\n"

.bss
.align 16
buffer:
    .zero BUFSIZE
numbuf:
    .zero 32
