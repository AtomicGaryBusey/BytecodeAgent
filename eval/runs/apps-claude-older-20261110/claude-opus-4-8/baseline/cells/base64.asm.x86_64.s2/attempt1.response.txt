.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
.zero 49152
outbuf:
.zero 70000

.text
emitchar:
    mov byte ptr [rdi], dl
    inc rdi
    inc r14b
    cmp r14b, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14b, r14b
1:
    ret

_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 49152
    sub edx, r12d
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea rbx, [rip + alphabet]
    xor r13, r13
    xor r14d, r14d

full_loop:
    lea rax, [r13 + 3]
    cmp rax, r12
    jg tail
    movzx eax, byte ptr [rsi + r13]
    movzx r8d, byte ptr [rsi + r13 + 1]
    movzx r9d, byte ptr [rsi + r13 + 2]
    shl eax, 16
    shl r8d, 8
    or eax, r8d
    or eax, r9d
    mov r8d, eax
    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov ecx, r8d
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    add r13, 3
    jmp full_loop

tail:
    mov rax, r12
    sub rax, r13
    jz done
    cmp rax, 1
    je tail1

    movzx eax, byte ptr [rsi + r13]
    movzx r8d, byte ptr [rsi + r13 + 1]
    shl eax, 16
    shl r8d, 8
    or eax, r8d
    mov r8d, eax
    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov ecx, r8d
    shr ecx, 6
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov edx, 61
    call emitchar
    jmp done

tail1:
    movzx eax, byte ptr [rsi + r13]
    shl eax, 16
    mov r8d, eax
    mov ecx, r8d
    shr ecx, 18
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov ecx, r8d
    shr ecx, 12
    and ecx, 63
    movzx edx, byte ptr [rbx + rcx]
    call emitchar
    mov edx, 61
    call emitchar
    mov edx, 61
    call emitchar

done:
    test r14b, r14b
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    mov edi, 0
    mov eax, 60
    syscall
