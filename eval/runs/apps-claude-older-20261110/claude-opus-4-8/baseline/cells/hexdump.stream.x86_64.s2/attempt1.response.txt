.intel_syntax noprefix
.globl _start

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .zero 65536
outbuf:
    .zero 400000

.text

# emit_offset: eax = 32-bit value, writes 8 hex digits at [r13]
emit_offset:
    mov r9d, 8
    mov ecx, 28
eo_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0x0f
    mov r10b, byte ptr [hextab + rdx]
    mov byte ptr [r13], r10b
    inc r13
    sub ecx, 4
    dec r9d
    jnz eo_loop
    ret

# emit_hex_byte: al = byte, writes 2 hex digits at [r13]
emit_hex_byte:
    mov cl, al
    movzx edx, al
    shr edx, 4
    mov r10b, byte ptr [hextab + rdx]
    mov byte ptr [r13], r10b
    inc r13
    movzx edx, cl
    and edx, 0x0f
    mov r10b, byte ptr [hextab + rdx]
    mov byte ptr [r13], r10b
    inc r13
    ret

_start:
    # read all input into inbuf
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:

    lea r13, [rip + outbuf]
    xor ebx, ebx

row_loop:
    cmp rbx, r12
    jae rows_done

    mov rax, r12
    sub rax, rbx
    cmp rax, 16
    jbe have_n
    mov rax, 16
have_n:
    mov r14, rax

    # offset
    mov eax, ebx
    call emit_offset
    mov byte ptr [r13], ' '
    inc r13
    mov byte ptr [r13], ' '
    inc r13

    xor r15, r15
byte_loop:
    cmp r15, r14
    jae pad_slot
    lea rsi, [rbx + r15]
    movzx eax, byte ptr [inbuf + rsi]
    call emit_hex_byte
    mov byte ptr [r13], ' '
    inc r13
    jmp after_slot
pad_slot:
    mov byte ptr [r13], ' '
    inc r13
    mov byte ptr [r13], ' '
    inc r13
    mov byte ptr [r13], ' '
    inc r13
after_slot:
    cmp r15, 7
    jne no_extra
    mov byte ptr [r13], ' '
    inc r13
no_extra:
    inc r15
    cmp r15, 16
    jb byte_loop

    mov byte ptr [r13], ' '
    inc r13
    mov byte ptr [r13], '|'
    inc r13

    xor r15, r15
char_loop:
    cmp r15, r14
    jae chars_done
    lea rsi, [rbx + r15]
    movzx eax, byte ptr [inbuf + rsi]
    cmp al, 0x20
    jb put_dot
    cmp al, 0x7e
    ja put_dot
    jmp put_char
put_dot:
    mov al, '.'
put_char:
    mov byte ptr [r13], al
    inc r13
    inc r15
    jmp char_loop
chars_done:
    mov byte ptr [r13], '|'
    inc r13
    mov byte ptr [r13], 10
    inc r13

    add rbx, 16
    jmp row_loop

rows_done:
    mov eax, r12d
    call emit_offset
    mov byte ptr [r13], 10
    inc r13

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    lea rcx, [rip + outbuf]
    sub rdx, rcx
    syscall

    mov rax, 60
    xor edi, edi
    syscall
