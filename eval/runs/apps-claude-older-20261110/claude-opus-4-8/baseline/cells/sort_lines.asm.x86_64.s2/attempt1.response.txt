.intel_syntax noprefix

.text
_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea r15, [rip + lines]
    lea r13, [rip + inbuf]
    xor r8, r8
    xor r9, r9
    xor r10, r10
parse_loop:
    cmp r10, r12
    jae parse_tail
    movzx eax, byte ptr [r13 + r10]
    cmp al, 0x0A
    jne parse_next
    mov rax, r8
    shl rax, 4
    lea rcx, [r13 + r9]
    mov [r15 + rax], rcx
    mov rcx, r10
    sub rcx, r9
    mov [r15 + rax + 8], rcx
    inc r8
    lea r9, [r10 + 1]
parse_next:
    inc r10
    jmp parse_loop
parse_tail:
    cmp r9, r12
    jae parse_done
    mov rax, r8
    shl rax, 4
    lea rcx, [r13 + r9]
    mov [r15 + rax], rcx
    mov rcx, r12
    sub rcx, r9
    mov [r15 + rax + 8], rcx
    inc r8
parse_done:
    mov rbx, r8

    mov rbp, 1
sort_outer:
    cmp rbp, rbx
    jae sort_done
    mov rax, rbp
    shl rax, 4
    mov r12, [r15 + rax]
    mov r13, [r15 + rax + 8]
    mov r14, rbp
    dec r14
sort_inner:
    test r14, r14
    js sort_insert
    mov rax, r14
    shl rax, 4
    mov rdi, [r15 + rax]
    mov rsi, [r15 + rax + 8]
    mov rdx, r12
    mov rcx, r13
    call line_cmp
    test eax, eax
    jle sort_insert
    mov rax, r14
    shl rax, 4
    mov rcx, [r15 + rax]
    mov rdx, [r15 + rax + 8]
    mov [r15 + rax + 16], rcx
    mov [r15 + rax + 24], rdx
    dec r14
    jmp sort_inner
sort_insert:
    mov rax, r14
    inc rax
    shl rax, 4
    mov [r15 + rax], r12
    mov [r15 + rax + 8], r13
    inc rbp
    jmp sort_outer
sort_done:

    xor rbp, rbp
    lea rdi, [rip + outbuf]
out_loop:
    cmp rbp, rbx
    jae out_done
    mov rax, rbp
    shl rax, 4
    mov rsi, [r15 + rax]
    mov rcx, [r15 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 0x0A
    inc rdi
    inc rbp
    jmp out_loop
out_done:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    xor edi, edi
    syscall

line_cmp:
    mov r8, rsi
    cmp r8, rcx
    jbe 1f
    mov r8, rcx
1:
    xor r9, r9
2:
    cmp r9, r8
    jae 3f
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    cmp eax, r10d
    jb lc_lt
    ja lc_gt
    inc r9
    jmp 2b
3:
    cmp rsi, rcx
    jb lc_lt
    ja lc_gt
    xor eax, eax
    ret
lc_lt:
    mov eax, -1
    ret
lc_gt:
    mov eax, 1
    ret
