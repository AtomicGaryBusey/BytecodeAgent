.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000

.text
.globl _start

_start:
    adrp x20, inbuf
    add  x20, x20, :lo12:inbuf
    adrp x25, outbuf
    add  x25, x25, :lo12:outbuf
    mov  x21, x25
    adrp x23, alphabet
    add  x23, x23, :lo12:alphabet
    mov  x22, #0
    mov  x19, #0

read_loop:
    mov  x8, #63
    mov  x0, #0
    add  x1, x20, x19
    mov  x2, #49152
    sub  x2, x2, x19
    cbz  x2, encode_start
    svc  #0
    cmp  x0, #0
    b.le encode_start
    add  x19, x19, x0
    b    read_loop

encode_start:
    mov  x9, #0
enc_loop:
    sub  x10, x19, x9
    cmp  x10, #3
    b.lt remainder
    add  x12, x20, x9
    ldrb w11, [x12]
    ldrb w13, [x12, #1]
    ldrb w14, [x12, #2]
    lsr  w15, w11, #2
    and  w0, w11, #3
    lsl  w0, w0, #4
    lsr  w1, w13, #4
    orr  w16, w0, w1
    and  w0, w13, #15
    lsl  w0, w0, #2
    lsr  w1, w14, #6
    orr  w17, w0, w1
    and  w7, w14, #63
    ldrb w0, [x23, x15]
    bl   emit_char
    ldrb w0, [x23, x16]
    bl   emit_char
    ldrb w0, [x23, x17]
    bl   emit_char
    ldrb w0, [x23, x7]
    bl   emit_char
    add  x9, x9, #3
    b    enc_loop

remainder:
    cbz  x10, done
    cmp  x10, #1
    b.eq one_byte
    add  x12, x20, x9
    ldrb w11, [x12]
    ldrb w13, [x12, #1]
    lsr  w15, w11, #2
    and  w0, w11, #3
    lsl  w0, w0, #4
    lsr  w1, w13, #4
    orr  w16, w0, w1
    and  w0, w13, #15
    lsl  w17, w0, #2
    ldrb w0, [x23, x15]
    bl   emit_char
    ldrb w0, [x23, x16]
    bl   emit_char
    ldrb w0, [x23, x17]
    bl   emit_char
    mov  w0, #0x3d
    bl   emit_char
    b    done

one_byte:
    add  x12, x20, x9
    ldrb w11, [x12]
    lsr  w15, w11, #2
    and  w0, w11, #3
    lsl  w16, w0, #4
    ldrb w0, [x23, x15]
    bl   emit_char
    ldrb w0, [x23, x16]
    bl   emit_char
    mov  w0, #0x3d
    bl   emit_char
    mov  w0, #0x3d
    bl   emit_char

done:
    cmp  x22, #0
    b.eq write_out
    mov  w0, #10
    strb w0, [x21], #1

write_out:
    mov  x1, x25
    sub  x2, x21, x25
write_loop:
    cbz  x2, do_exit
    mov  x8, #64
    mov  x0, #1
    svc  #0
    cmp  x0, #0
    b.le do_exit
    add  x1, x1, x0
    sub  x2, x2, x0
    b    write_loop

do_exit:
    mov  x8, #93
    mov  x0, #0
    svc  #0

emit_char:
    strb w0, [x21], #1
    add  x22, x22, #1
    cmp  x22, #76
    b.ne 1f
    mov  w0, #10
    strb w0, [x21], #1
    mov  x22, #0
1:
    ret
