.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    test rdx, rdx
    jle done_read
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    lea rbp, [rip + outbuf]
    xor r13, r13
    xor r14, r14
scan:
    cmp r14, r12
    jae after_scan
    movzx eax, byte ptr [rbx + r14]
    cmp al, 0x0A
    jne 1f
    mov rsi, r14
rev1:
    cmp rsi, r13
    jbe 2f
    dec rsi
    mov al, byte ptr [rbx + rsi]
    mov byte ptr [rbp], al
    inc rbp
    jmp rev1
2:
    mov byte ptr [rbp], 0x0A
    inc rbp
    inc r14
    mov r13, r14
    jmp scan
1:
    inc r14
    jmp scan
after_scan:
    cmp r13, r12
    jae write_out
    mov rsi, r12
rev2:
    cmp rsi, r13
    jbe 3f
    dec rsi
    mov al, byte ptr [rbx + rsi]
    mov byte ptr [rbp], al
    inc rbp
    jmp rev2
3:
    mov byte ptr [rbp], 0x0A
    inc rbp
write_out:
    lea rax, [rip + outbuf]
    mov rsi, rax
    mov rdx, rbp
    sub rdx, rax
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
