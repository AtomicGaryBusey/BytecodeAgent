.text
.globl _start

_start:
    xor r15, r15                      # total bytes read
read_loop:
    xor eax, eax                      # read = 0
    xor edi, edi                      # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov rdx, 32768
    sub rdx, r15
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:

    # parse lines
    lea rbp, [rip + inbuf]            # base
    xor rbx, rbx                      # i
    xor r12, r12                      # start
    xor r13, r13                      # line count
parse_loop:
    cmp rbx, r15
    jae parse_end
    movzx eax, byte ptr [rbp + rbx]
    cmp al, 0x0A
    jne parse_skip
    mov rax, rbp
    add rax, r12
    mov qword ptr [rec_ptr + r13*8], rax
    mov rcx, rbx
    sub rcx, r12
    mov qword ptr [rec_len + r13*8], rcx
    inc r13
    lea r12, [rbx + 1]
parse_skip:
    inc rbx
    jmp parse_loop
parse_end:
    cmp r12, r15
    jae no_final
    mov rax, rbp
    add rax, r12
    mov qword ptr [rec_ptr + r13*8], rax
    mov rcx, r15
    sub rcx, r12
    mov qword ptr [rec_len + r13*8], rcx
    inc r13
no_final:
    mov qword ptr [rip + nlines], r13

    # init index array idxA[i]=i
    lea rdi, [rip + idxA]
    xor rcx, rcx
init_loop:
    cmp rcx, r13
    jae init_done
    mov dword ptr [rdi + rcx*4], ecx
    inc rcx
    jmp init_loop
init_done:
    lea rax, [rip + idxA]
    mov qword ptr [rip + src_p], rax
    lea rax, [rip + idxB]
    mov qword ptr [rip + dst_p], rax

    # bottom-up merge sort
    mov qword ptr [rip + cur_width], 1
outer:
    mov rax, qword ptr [rip + cur_width]
    mov rcx, qword ptr [rip + nlines]
    cmp rax, rcx
    jae sort_done

    xor r15, r15                      # i
inner_top:
    mov rax, qword ptr [rip + nlines]
    cmp r15, rax
    jae inner_done
    mov rbx, r15                      # l
    mov rax, qword ptr [rip + cur_width]
    lea r13, [r15 + rax]              # lend
    mov rcx, qword ptr [rip + nlines]
    cmp r13, rcx
    jbe 10f
    mov r13, rcx
10:
    mov rbp, r13                      # r
    mov rax, qword ptr [rip + cur_width]
    lea r14, [r15 + rax*2]            # rend
    cmp r14, rcx
    jbe 11f
    mov r14, rcx
11:
    mov r12, r15                      # k

merge_top:
    cmp rbx, r13
    jae copy_right
    cmp rbp, r14
    jae copy_left
    mov rax, qword ptr [rip + src_p]
    mov edi, dword ptr [rax + rbx*4]
    mov esi, dword ptr [rax + rbp*4]
    call cmpline
    cmp eax, 0
    jle take_left
    mov rax, qword ptr [rip + src_p]
    mov ecx, dword ptr [rax + rbp*4]
    mov rdx, qword ptr [rip + dst_p]
    mov dword ptr [rdx + r12*4], ecx
    inc r12
    inc rbp
    jmp merge_top
take_left:
    mov rax, qword ptr [rip + src_p]
    mov ecx, dword ptr [rax + rbx*4]
    mov rdx, qword ptr [rip + dst_p]
    mov dword ptr [rdx + r12*4], ecx
    inc r12
    inc rbx
    jmp merge_top

copy_left:
    cmp rbx, r13
    jae seg_done
    mov rax, qword ptr [rip + src_p]
    mov ecx, dword ptr [rax + rbx*4]
    mov rdx, qword ptr [rip + dst_p]
    mov dword ptr [rdx + r12*4], ecx
    inc r12
    inc rbx
    jmp copy_left

copy_right:
    cmp rbp, r14
    jae seg_done
    mov rax, qword ptr [rip + src_p]
    mov ecx, dword ptr [rax + rbp*4]
    mov rdx, qword ptr [rip + dst_p]
    mov dword ptr [rdx + r12*4], ecx
    inc r12
    inc rbp
    jmp copy_right

seg_done:
    mov rax, qword ptr [rip + cur_width]
    lea r15, [r15 + rax*2]
    jmp inner_top

inner_done:
    mov rax, qword ptr [rip + src_p]
    mov rcx, qword ptr [rip + dst_p]
    mov qword ptr [rip + src_p], rcx
    mov qword ptr [rip + dst_p], rax
    mov rax, qword ptr [rip + cur_width]
    add rax, rax
    mov qword ptr [rip + cur_width], rax
    jmp outer

sort_done:
    # build output
    lea r13, [rip + outbuf]
    xor r12, r12                      # out offset
    xor rbx, rbx                      # k
    mov r14, qword ptr [rip + src_p]
out_loop:
    cmp rbx, qword ptr [rip + nlines]
    jae out_write
    mov eax, dword ptr [r14 + rbx*4]
    mov rsi, qword ptr [rec_ptr + rax*8]
    mov rdx, qword ptr [rec_len + rax*8]
    xor rcx, rcx
copy_bytes:
    cmp rcx, rdx
    jae copy_bytes_done
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [r13 + r12], al
    inc r12
    inc rcx
    jmp copy_bytes
copy_bytes_done:
    mov byte ptr [r13 + r12], 0x0A
    inc r12
    inc rbx
    jmp out_loop
out_write:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r12
    syscall
    mov eax, 60
    xor edi, edi
    syscall

# cmpline(rdi=index a, rsi=index b) -> eax sign
cmpline:
    mov r8,  qword ptr [rec_ptr + rdi*8]
    mov r9,  qword ptr [rec_len + rdi*8]
    mov r10, qword ptr [rec_ptr + rsi*8]
    mov r11, qword ptr [rec_len + rsi*8]
    mov rcx, r9
    cmp r11, r9
    cmovb rcx, r11
    xor edx, edx
cmp_loop:
    cmp rdx, rcx
    jae cmp_equal
    movzx eax, byte ptr [r8 + rdx]
    movzx esi, byte ptr [r10 + rdx]
    cmp eax, esi
    jne cmp_diff
    inc rdx
    jmp cmp_loop
cmp_equal:
    mov eax, r9d
    sub eax, r11d
    ret
cmp_diff:
    sub eax, esi
    ret

.bss
.align 16
inbuf:
    .zero 32768
.align 16
outbuf:
    .zero 34816
.align 16
rec_ptr:
    .zero 8000
rec_len:
    .zero 8000
idxA:
    .zero 4000
idxB:
    .zero 4000
nlines:
    .zero 8
cur_width:
    .zero 8
src_p:
    .zero 8
dst_p:
    .zero 8
