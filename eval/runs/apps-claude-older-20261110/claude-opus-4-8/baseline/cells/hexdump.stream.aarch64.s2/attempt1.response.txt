.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
    .balign 16
inbuf:
    .zero 65536
    .balign 16
outbuf:
    .zero 400000

.text
.globl _start

// emit 8 hex digits of w0 to [x21], advancing x21
emit_hex8:
    adrp x11, hextab
    add x11, x11, :lo12:hextab
    mov w9, #28
1:
    lsr w10, w0, w9
    and w10, w10, #0xf
    ldrb w10, [x11, w10, uxtw]
    strb w10, [x21], #1
    subs w9, w9, #4
    b.ge 1b
    ret

_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x19, #0
read_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x19, x19, x0
    b read_loop
read_done:

    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    mov x22, #0

row_loop:
    cmp x22, x19
    bge rows_done

    sub x23, x19, x22
    cmp x23, #16
    mov x24, #16
    csel x25, x24, x23, ge

    // offset
    mov w0, w22
    bl emit_hex8
    mov w14, #0x20
    strb w14, [x21], #1
    strb w14, [x21], #1

    mov x26, #0
byte_loop:
    cmp x26, #16
    bge byte_done
    cmp x26, x25
    bge empty_slot
    add x11, x20, x22
    ldrb w0, [x11, x26]
    lsr w12, w0, #4
    and w13, w0, #0xf
    adrp x11, hextab
    add x11, x11, :lo12:hextab
    ldrb w12, [x11, w12, uxtw]
    ldrb w13, [x11, w13, uxtw]
    strb w12, [x21], #1
    strb w13, [x21], #1
    mov w14, #0x20
    strb w14, [x21], #1
    b slot_done
empty_slot:
    mov w14, #0x20
    strb w14, [x21], #1
    strb w14, [x21], #1
    strb w14, [x21], #1
slot_done:
    cmp x26, #7
    bne 1f
    mov w14, #0x20
    strb w14, [x21], #1
1:
    add x26, x26, #1
    b byte_loop
byte_done:

    mov w14, #0x20
    strb w14, [x21], #1
    mov w14, #0x7c
    strb w14, [x21], #1

    mov x26, #0
char_loop:
    cmp x26, x25
    bge char_done
    add x11, x20, x22
    ldrb w0, [x11, x26]
    cmp w0, #0x20
    blt notp
    cmp w0, #0x7e
    bgt notp
    b putc
notp:
    mov w0, #0x2e
putc:
    strb w0, [x21], #1
    add x26, x26, #1
    b char_loop
char_done:
    mov w14, #0x7c
    strb w14, [x21], #1
    mov w14, #0x0a
    strb w14, [x21], #1

    add x22, x22, #16
    b row_loop

rows_done:
    mov w0, w19
    bl emit_hex8
    mov w14, #0x0a
    strb w14, [x21], #1

    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x21, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
