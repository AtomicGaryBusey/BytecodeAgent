.text
.globl _start

_start:
    lea r13, [rip + inbuf]
    xor r12, r12

read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbx, [rip + outbuf]
    xor r14, r14

row_loop:
    cmp r14, r12
    jae rows_done

    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15, 16
1:
    mov rbp, r13
    add rbp, r14

    mov edx, r14d
    call emit_hex8
    mov word ptr [rbx], 0x2020
    add rbx, 2

    xor ecx, ecx
2:
    cmp ecx, 16
    jae 3f
    cmp rcx, r15
    jb have_byte
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx+1], 0x20
    mov byte ptr [rbx+2], 0x20
    add rbx, 3
    jmp after_slot
have_byte:
    movzx eax, byte ptr [rbp + rcx]
    mov edi, eax
    shr edi, 4
    and eax, 15
    mov dl, byte ptr [hextab + rdi]
    mov byte ptr [rbx], dl
    mov dl, byte ptr [hextab + rax]
    mov byte ptr [rbx+1], dl
    mov byte ptr [rbx+2], 0x20
    add rbx, 3
after_slot:
    cmp ecx, 7
    jne 6f
    mov byte ptr [rbx], 0x20
    inc rbx
6:
    inc ecx
    jmp 2b
3:
    mov byte ptr [rbx], 0x20
    mov byte ptr [rbx+1], 0x7c
    add rbx, 2

    xor ecx, ecx
7:
    cmp rcx, r15
    jae 8f
    movzx eax, byte ptr [rbp + rcx]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    ja dot
    jmp putc
dot:
    mov al, 0x2e
putc:
    mov byte ptr [rbx], al
    inc rbx
    inc ecx
    jmp 7b
8:
    mov byte ptr [rbx], 0x7c
    mov byte ptr [rbx+1], 0x0a
    add rbx, 2
    add r14, 16
    jmp row_loop

rows_done:
    mov edx, r12d
    call emit_hex8
    mov byte ptr [rbx], 0x0a
    inc rbx

    lea rsi, [rip + outbuf]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    mov ecx, 8
9:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [hextab + rax]
    mov byte ptr [rbx], al
    inc rbx
    dec ecx
    jnz 9b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 400000
