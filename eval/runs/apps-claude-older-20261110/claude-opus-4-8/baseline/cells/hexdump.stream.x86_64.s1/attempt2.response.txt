.text
.globl _start
_start:
    lea r13, [rip+hextab]
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    test rdx, rdx
    jz read_done
    lea rsi, [rip+inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    xor rbx, rbx
    lea rdi, [rip+outbuf]
row_loop:
    cmp rbx, r12
    jae rows_done

    mov esi, ebx
    call emit_offset
    mov word ptr [rdi], 0x2020
    add rdi, 2

    mov r9, r12
    sub r9, rbx
    cmp r9, 16
    jbe 1f
    mov r9, 16
1:
    lea rsi, [rip+inbuf]
    add rsi, rbx
    xor r8, r8
byte_loop:
    cmp r8, 16
    jae byte_done
    cmp r8, r9
    jb have_byte
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi+1], 0x20
    mov byte ptr [rdi+2], 0x20
    add rdi, 3
    jmp after_byte
have_byte:
    movzx eax, byte ptr [rsi+r8]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r13+rdx]
    mov byte ptr [rdi], dl
    and eax, 0xf
    movzx eax, byte ptr [r13+rax]
    mov byte ptr [rdi+1], al
    mov byte ptr [rdi+2], 0x20
    add rdi, 3
after_byte:
    cmp r8, 7
    jne 2f
    mov byte ptr [rdi], 0x20
    inc rdi
2:
    inc r8
    jmp byte_loop
byte_done:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi+1], 0x7c
    add rdi, 2

    xor r8, r8
char_loop:
    cmp r8, r9
    jae char_done
    movzx eax, byte ptr [rsi+r8]
    cmp al, 0x20
    jb dot
    cmp al, 0x7e
    ja dot
    jmp putc
dot:
    mov al, 0x2e
putc:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp char_loop
char_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi+1], 0x0a
    add rdi, 2

    add rbx, 16
    jmp row_loop
rows_done:
    mov esi, r12d
    call emit_offset
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rax, [rip+outbuf]
    mov rsi, rax
    mov rdx, rdi
    sub rdx, rax
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_offset:
    mov r8d, 32
3:
    sub r8d, 4
    mov edx, esi
    mov ecx, r8d
    shr edx, cl
    and edx, 0xf
    movzx edx, byte ptr [r13+rdx]
    mov byte ptr [rdi], dl
    inc rdi
    test r8d, r8d
    jnz 3b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 400000
