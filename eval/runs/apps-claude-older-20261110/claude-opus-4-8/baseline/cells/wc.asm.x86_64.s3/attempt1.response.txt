.globl _start

.section .bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 64

.text
_start:
    xor r12, r12        # lines
    xor r13, r13        # words
    xor r14, r14        # bytes
    xor r15, r15        # in_word flag

read_loop:
    xor eax, eax        # read
    xor edi, edi        # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax        # bytes += n
    mov rcx, rax        # count
    lea rsi, [rip + buf]
proc_byte:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne not_nl
    inc r12
not_nl:
    # check whitespace
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0D
    jbe is_ws
not_ws:
    # word byte
    test r15, r15
    jnz skip_word
    inc r13
    mov r15, 1
skip_word:
    jmp next_byte
is_ws:
    xor r15, r15
next_byte:
    dec rcx
    jnz proc_byte
    jmp read_loop

done_read:
    # print r12, r13, r14 separated by spaces and newline
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call put_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call put_num
    mov byte ptr [rdi], 0x0A
    inc rdi
    # write
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

# put decimal number in rax to [rdi], advance rdi
put_num:
    mov r8, rdi
    mov rcx, rax
    test rcx, rcx
    jnz pn_nonzero
    mov byte ptr [rdi], 0x30
    inc rdi
    ret
pn_nonzero:
    # generate digits onto stack reversed
    mov r9, rsp
    mov rbx, 10
pn_div:
    xor edx, edx
    mov rax, rcx
    div rbx
    mov rcx, rax
    add dl, 0x30
    dec r9
    mov byte ptr [r9], dl
    test rcx, rcx
    jnz pn_div
    # copy from r9..rsp to rdi
pn_copy:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    inc r9
    cmp r9, rsp
    jne pn_copy
    ret
