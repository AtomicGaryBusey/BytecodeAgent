.text
.globl _start

_start:
    # read all input into inbuf
    xor ebx, ebx                 # total bytes read
read_loop:
    xor eax, eax                 # read
    xor edi, edi                 # fd 0
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 49152
    sub edx, ebx
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    cmp rbx, 49152
    jb read_loop
read_done:

    lea rsi, [rip + inbuf]       # input pointer
    lea rdi, [rip + outbuf]      # output pointer
    xor r8d, r8d                 # column count
    mov rcx, rbx                 # remaining bytes

enc_loop:
    cmp rcx, 3
    jb enc_tail
    movzx r12d, byte ptr [rsi]
    movzx r13d, byte ptr [rsi+1]
    movzx r14d, byte ptr [rsi+2]
    add rsi, 3
    sub rcx, 3
    mov eax, r12d
    shr eax, 2
    call emitv
    mov eax, r12d
    and eax, 3
    shl eax, 4
    mov edx, r13d
    shr edx, 4
    or eax, edx
    call emitv
    mov eax, r13d
    and eax, 15
    shl eax, 2
    mov edx, r14d
    shr edx, 6
    or eax, edx
    call emitv
    mov eax, r14d
    and eax, 63
    call emitv
    jmp enc_loop

enc_tail:
    cmp rcx, 0
    je enc_done
    cmp rcx, 1
    je tail1
    # two bytes remain
    movzx r12d, byte ptr [rsi]
    movzx r13d, byte ptr [rsi+1]
    mov eax, r12d
    shr eax, 2
    call emitv
    mov eax, r12d
    and eax, 3
    shl eax, 4
    mov edx, r13d
    shr edx, 4
    or eax, edx
    call emitv
    mov eax, r13d
    and eax, 15
    shl eax, 2
    call emitv
    mov al, 61
    call emit
    jmp enc_done
tail1:
    movzx r12d, byte ptr [rsi]
    mov eax, r12d
    shr eax, 2
    call emitv
    mov eax, r12d
    and eax, 3
    shl eax, 4
    call emitv
    mov al, 61
    call emit
    mov al, 61
    call emit
enc_done:
    cmp r8d, 0
    je write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    cmp rdx, 0
    jle done_write
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle done_write
    add rsi, rax
    sub rdx, rax
    jmp write_loop
done_write:
    mov eax, 60
    xor edi, edi
    syscall

# emitv: al = 6-bit value, translate via alphabet, then emit
emitv:
    movzx eax, al
    lea r11, [rip + alpha]
    mov al, byte ptr [r11 + rax]
# emit: al = raw char
emit:
    mov byte ptr [rdi], al
    inc rdi
    inc r8d
    cmp r8d, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
1:
    ret

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 49152
outbuf:
    .zero 131072
