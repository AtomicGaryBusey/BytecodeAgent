.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    mov edx, 65536
    sub edx, r12d
    test edx, edx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = inlen ; find first '!'
    xor rcx, rcx
find_loop:
    cmp rcx, r12
    jae no_bang
    lea rsi, [rip + inbuf]
    cmp byte ptr [rsi + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_loop
no_bang:
    mov r14, r12
    mov qword ptr [rip + inputpos], r12
    mov qword ptr [rip + inputend], r12
    jmp after_split
found_bang:
    mov r14, rcx
    lea rax, [rcx + 1]
    mov qword ptr [rip + inputpos], rax
    mov qword ptr [rip + inputend], r12
after_split:
    # bracket check + build jump table
    lea rsi, [rip + inbuf]
    lea rdi, [rip + jump]
    lea r8, [rip + bstack]
    xor r9d, r9d
    xor rcx, rcx
bchk_loop:
    cmp rcx, r14
    jae bchk_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, '['
    je is_open
    cmp al, ']'
    je is_close
    inc rcx
    jmp bchk_loop
is_open:
    mov dword ptr [r8 + r9*4], ecx
    inc r9
    inc rcx
    jmp bchk_loop
is_close:
    test r9, r9
    jz berror
    dec r9
    mov edx, dword ptr [r8 + r9*4]
    mov dword ptr [rdi + rdx*4], ecx
    mov dword ptr [rdi + rcx*4], edx
    inc rcx
    jmp bchk_loop
bchk_done:
    test r9, r9
    jnz berror

    # run
    lea r12, [rip + inbuf]
    lea r15, [rip + tape]
    mov r13, r15
    lea rbp, [rip + jump]
    xor ebx, ebx
run_loop:
    cmp rbx, r14
    jae run_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, '>'
    je c_gt
    cmp al, '<'
    je c_lt
    cmp al, '+'
    je c_plus
    cmp al, '-'
    je c_minus
    cmp al, '.'
    je c_dot
    cmp al, ','
    je c_comma
    cmp al, '['
    je c_lb
    cmp al, ']'
    je c_rb
next:
    inc rbx
    jmp run_loop
c_gt:
    inc r13
    jmp next
c_lt:
    dec r13
    jmp next
c_plus:
    inc byte ptr [r13]
    jmp next
c_minus:
    dec byte ptr [r13]
    jmp next
c_dot:
    mov rcx, qword ptr [rip + outpos]
    mov al, byte ptr [r13]
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + rcx], al
    inc rcx
    mov qword ptr [rip + outpos], rcx
    cmp rcx, 65536
    jb next
    call flush
    jmp next
c_comma:
    mov rax, qword ptr [rip + inputpos]
    cmp rax, qword ptr [rip + inputend]
    jae comma_zero
    movzx edx, byte ptr [r12 + rax]
    mov byte ptr [r13], dl
    inc rax
    mov qword ptr [rip + inputpos], rax
    jmp next
comma_zero:
    mov byte ptr [r13], 0
    jmp next
c_lb:
    cmp byte ptr [r13], 0
    jne next
    mov ebx, dword ptr [rbp + rbx*4]
    jmp next
c_rb:
    cmp byte ptr [r13], 0
    je next
    mov ebx, dword ptr [rbp + rbx*4]
    jmp next
run_done:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

berror:
    mov eax, 60
    mov edi, 1
    syscall

flush:
    mov rdx, qword ptr [rip + outpos]
    test rdx, rdx
    jz 1f
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    syscall
    mov qword ptr [rip + outpos], 0
1:
    ret

.bss
.align 8
inputpos: .zero 8
inputend: .zero 8
outpos:   .zero 8
inbuf:    .zero 65536
tape:     .zero 30000
outbuf:   .zero 65536
jump:     .zero 262144
bstack:   .zero 4096
