.globl _start

.text
_start:
    mov x19, #0              // lines
    mov x20, #0              // words
    mov x21, #0             // bytes
    mov x22, #0             // in_word

read_loop:
    mov x0, #0
    adrp x1, rdbuf
    add x1, x1, :lo12:rdbuf
    mov x2, #4096
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le done_read
    mov x3, x0              // count
    adrp x1, rdbuf
    add x1, x1, :lo12:rdbuf
    mov x4, #0             // index

byte_loop:
    cmp x4, x3
    b.ge read_loop
    ldrb w5, [x1, x4]
    add x4, x4, #1
    add x21, x21, #1       // bytes++
    cmp w5, #0x0A
    b.ne 1f
    add x19, x19, #1       // lines++
1:
    cmp w5, #0x20
    b.eq is_ws
    sub w6, w5, #9
    cmp w6, #4
    b.ls is_ws
    cmp x22, #0
    b.ne byte_loop
    mov x22, #1
    add x20, x20, #1
    b byte_loop
is_ws:
    mov x22, #0
    b byte_loop

done_read:
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    mov x0, x19
    bl print_num
    mov w12, #0x20
    strb w12, [x9], #1
    mov x0, x20
    bl print_num
    mov w12, #0x20
    strb w12, [x9], #1
    mov x0, x21
    bl print_num
    mov w12, #0x0A
    strb w12, [x9], #1

    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x9, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_num:
    mov x10, #10
    sub sp, sp, #32
    mov x11, #0
    cbnz x0, 1f
    mov w12, #0x30
    strb w12, [sp]
    mov x11, #1
    b 2f
1:
3:
    cbz x0, 2f
    udiv x13, x0, x10
    msub x14, x13, x10, x0
    add w14, w14, #0x30
    strb w14, [sp, x11]
    add x11, x11, #1
    mov x0, x13
    b 3b
2:
    mov x15, x11
4:
    cbz x15, 5f
    sub x15, x15, #1
    ldrb w12, [sp, x15]
    strb w12, [x9], #1
    b 4b
5:
    add sp, sp, #32
    ret

.bss
.align 4
rdbuf:
    .space 4096
outbuf:
    .space 32
