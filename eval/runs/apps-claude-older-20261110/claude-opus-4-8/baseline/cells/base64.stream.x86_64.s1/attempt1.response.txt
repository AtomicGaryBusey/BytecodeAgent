.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000

.text
.globl _start

emit:
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne 1f
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14d, r14d
1:
    ret

_start:
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:

    lea r13, [rip + inbuf]
    mov rbx, r12
    lea rdi, [rip + outbuf]
    xor r14d, r14d
    lea rbp, [rip + alphabet]

enc_loop:
    cmp rbx, 3
    jb enc_tail
    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13+1]
    movzx edx, byte ptr [r13+2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx
    mov r15d, eax

    mov eax, r15d
    shr eax, 18
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov eax, r15d
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov eax, r15d
    shr eax, 6
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov eax, r15d
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    add r13, 3
    sub rbx, 3
    jmp enc_loop

enc_tail:
    test rbx, rbx
    jz enc_done
    cmp rbx, 1
    je tail1

    movzx eax, byte ptr [r13]
    movzx ecx, byte ptr [r13+1]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    mov r15d, eax

    mov eax, r15d
    shr eax, 18
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov eax, r15d
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov eax, r15d
    shr eax, 6
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov al, 0x3D
    call emit
    jmp enc_done

tail1:
    movzx eax, byte ptr [r13]
    shl eax, 16
    mov r15d, eax

    mov eax, r15d
    shr eax, 18
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov eax, r15d
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [rbp + rax]
    call emit

    mov al, 0x3D
    call emit
    mov al, 0x3D
    call emit

enc_done:
    test r14d, r14d
    jz write_out
    mov byte ptr [rdi], 0x0A
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall
