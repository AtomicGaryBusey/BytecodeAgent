.globl _start

.text
_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, #0
read_loop:
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    mov     x24, x20                // total length

    // find first '!'
    mov     x21, #0
find_loop:
    cmp     x21, x24
    b.ge    no_bang
    ldrb    w0, [x19, x21]
    cmp     w0, #0x21
    b.eq    found_bang
    add     x21, x21, #1
    b       find_loop
found_bang:
    mov     x20, x21                // proglen
    add     x25, x21, #1            // input index
    b       after_bang
no_bang:
    mov     x20, x24
    mov     x25, x24
after_bang:

    // validate brackets and build jump table
    adrp    x26, jump_tbl
    add     x26, x26, :lo12:jump_tbl
    adrp    x27, bstack
    add     x27, x27, :lo12:bstack
    mov     x28, #0                 // stack depth
    mov     x21, #0                 // i
val_loop:
    cmp     x21, x20
    b.ge    val_end
    ldrb    w0, [x19, x21]
    cmp     w0, #0x5b
    b.eq    is_open
    cmp     w0, #0x5d
    b.eq    is_close
    add     x21, x21, #1
    b       val_loop
is_open:
    str     w21, [x27, x28, lsl #2]
    add     x28, x28, #1
    add     x21, x21, #1
    b       val_loop
is_close:
    cbz     x28, bracket_err
    sub     x28, x28, #1
    ldr     w1, [x27, x28, lsl #2]
    str     w21, [x26, w1, uxtw #2]
    str     w1, [x26, x21, lsl #2]
    add     x21, x21, #1
    b       val_loop
val_end:
    cbz     x28, run_start
bracket_err:
    mov     x0, #1
    mov     x8, #93
    svc     #0

run_start:
    adrp    x22, tape
    add     x22, x22, :lo12:tape
    mov     x21, #0                 // ip
exec_loop:
    cmp     x21, x20
    b.ge    exec_done
    ldrb    w0, [x19, x21]
    cmp     w0, #0x3e
    b.eq    op_gt
    cmp     w0, #0x3c
    b.eq    op_lt
    cmp     w0, #0x2b
    b.eq    op_plus
    cmp     w0, #0x2d
    b.eq    op_minus
    cmp     w0, #0x2e
    b.eq    op_dot
    cmp     w0, #0x2c
    b.eq    op_comma
    cmp     w0, #0x5b
    b.eq    op_open
    cmp     w0, #0x5d
    b.eq    op_close
next_ip:
    add     x21, x21, #1
    b       exec_loop

op_gt:
    add     x22, x22, #1
    b       next_ip
op_lt:
    sub     x22, x22, #1
    b       next_ip
op_plus:
    ldrb    w1, [x22]
    add     w1, w1, #1
    strb    w1, [x22]
    b       next_ip
op_minus:
    ldrb    w1, [x22]
    sub     w1, w1, #1
    strb    w1, [x22]
    b       next_ip
op_dot:
    mov     x0, #1
    mov     x1, x22
    mov     x2, #1
    mov     x8, #64
    svc     #0
    b       next_ip
op_comma:
    cmp     x25, x24
    b.ge    comma_zero
    ldrb    w1, [x19, x25]
    add     x25, x25, #1
    strb    w1, [x22]
    b       next_ip
comma_zero:
    strb    wzr, [x22]
    b       next_ip
op_open:
    ldrb    w1, [x22]
    cbnz    w1, next_ip
    ldr     w1, [x26, x21, lsl #2]
    mov     x21, x1
    b       next_ip
op_close:
    ldrb    w1, [x22]
    cbz     w1, next_ip
    ldr     w1, [x26, x21, lsl #2]
    mov     x21, x1
    b       next_ip
exec_done:
    mov     x0, #0
    mov     x8, #93
    svc     #0

.bss
.align 4
input_buf:
    .zero   65536
jump_tbl:
    .zero   262144
bstack:
    .zero   4096
tape:
    .zero   30000
