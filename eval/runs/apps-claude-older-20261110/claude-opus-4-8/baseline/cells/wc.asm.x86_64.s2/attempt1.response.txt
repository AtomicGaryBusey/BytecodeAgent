.text
.globl _start
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # in_word flag

read_loop:
    xor eax, eax            # read
    xor edi, edi            # fd 0
    lea rsi, [rip + rdbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jle done_read
    add r14, rax            # bytes += n
    lea rsi, [rip + rdbuf]
    mov rcx, rax            # count
proc_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne 1f
    inc r12
1:
    mov dl, al
    cmp dl, 0x20
    je is_ws
    sub dl, 0x09
    cmp dl, 4
    jbe is_ws
    # word byte
    test r15, r15
    jnz 2f
    inc r13
    mov r15d, 1
    jmp 2f
is_ws:
    xor r15d, r15d
2:
    dec rcx
    jnz proc_loop
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call itoa
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, r13
    call itoa
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, r14
    call itoa
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi            # length
    mov eax, 1             # write
    mov edi, 1            # fd 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

# itoa: value in rax, dest ptr in rdi, returns new rdi
itoa:
    mov rbx, 10
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], '0'
    inc rdi
    ret
1:
    xor rcx, rcx
2:
    xor rdx, rdx
    div rbx
    add dl, '0'
    push rdx
    inc rcx
    test rax, rax
    jnz 2b
3:
    pop rdx
    mov [rdi], dl
    inc rdi
    dec rcx
    jnz 3b
    ret

.bss
.align 16
rdbuf:
    .zero 4096
outbuf:
    .zero 64
