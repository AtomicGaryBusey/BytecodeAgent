.text
.globl _start

_start:
    xor r12, r12
rdloop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [inbuf + r12]
    mov rdx, 32768
    sub rdx, r12
    syscall
    test rax, rax
    jle rddone
    add r12, rax
    jmp rdloop
rddone:
    mov r13, r12
    xor r12, r12
    xor rbx, rbx
    xor rbp, rbp
ploop:
    cmp rbx, r13
    jae ptail
    cmp byte ptr [inbuf + rbx], 0x0A
    jne pnext
    mov qword ptr [line_off + r12*8], rbp
    mov rax, rbx
    sub rax, rbp
    mov qword ptr [line_len + r12*8], rax
    inc r12
    lea rbp, [rbx + 1]
pnext:
    inc rbx
    jmp ploop
ptail:
    cmp rbp, r13
    jae pdone
    mov qword ptr [line_off + r12*8], rbp
    mov rax, r13
    sub rax, rbp
    mov qword ptr [line_len + r12*8], rax
    inc r12
pdone:

    mov rbx, 1
outer:
    cmp rbx, r12
    jae sort_done
    mov r14, qword ptr [line_off + rbx*8]
    mov r15, qword ptr [line_len + rbx*8]
    mov rbp, rbx
    dec rbp
inner:
    test rbp, rbp
    js place
    mov rdi, qword ptr [line_off + rbp*8]
    mov rsi, qword ptr [line_len + rbp*8]
    mov rdx, r14
    mov rcx, r15
    call cmp_lines
    test eax, eax
    jle place
    mov rax, qword ptr [line_off + rbp*8]
    mov qword ptr [line_off + rbp*8 + 8], rax
    mov rax, qword ptr [line_len + rbp*8]
    mov qword ptr [line_len + rbp*8 + 8], rax
    dec rbp
    jmp inner
place:
    mov qword ptr [line_off + rbp*8 + 8], r14
    mov qword ptr [line_len + rbp*8 + 8], r15
    inc rbx
    jmp outer
sort_done:

    xor rbx, rbx
    xor r13, r13
oloop:
    cmp rbx, r12
    jae owrite
    mov r8, qword ptr [line_off + rbx*8]
    mov rcx, qword ptr [line_len + rbx*8]
    lea rdi, [outbuf + r13]
    lea r8, [inbuf + r8]
    xor r9, r9
cpl:
    cmp r9, rcx
    jae cpdone
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi + r9], al
    inc r9
    jmp cpl
cpdone:
    add r13, rcx
    mov byte ptr [outbuf + r13], 0x0A
    inc r13
    inc rbx
    jmp oloop
owrite:
    mov rax, 1
    mov rdi, 1
    lea rsi, [outbuf]
    mov rdx, r13
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

cmp_lines:
    mov r8, rsi
    cmp rcx, r8
    cmovb r8, rcx
    lea r9, [inbuf]
    add rdi, r9
    add rdx, r9
    xor r10, r10
.cl:
    cmp r10, r8
    jae .lendiff
    movzx eax, byte ptr [rdi + r10]
    movzx r11d, byte ptr [rdx + r10]
    cmp eax, r11d
    jne .diff
    inc r10
    jmp .cl
.diff:
    sub eax, r11d
    ret
.lendiff:
    mov eax, esi
    sub eax, ecx
    ret

.bss
.align 16
inbuf:
    .zero 32768
line_off:
    .zero 8008
line_len:
    .zero 8008
outbuf:
