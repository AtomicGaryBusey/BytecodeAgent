.text
.globl _start

_start:
    xor r12, r12
rd_loop:
    mov rdx, 65536
    sub rdx, r12
    jle rd_done
    xor eax, eax
    xor edi, edi
    lea rsi, [buffer + r12]
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    mov [len], r12
    mov qword ptr [pos], 0
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, [pos]
    cmp rax, [len]
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

peek:
    mov rcx, [pos]
    cmp rcx, [len]
    jae 1f
    movzx eax, byte ptr [buffer + rcx]
    ret
1:  mov eax, -1
    ret

skip_ws:
    mov rcx, [pos]
1:  cmp rcx, [len]
    jae 2f
    movzx eax, byte ptr [buffer + rcx]
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
    jmp 2f
3:  inc rcx
    jmp 1b
2:  mov [pos], rcx
    ret

parse_value:
    call peek
    cmp eax, '{'
    je parse_object
    cmp eax, '['
    je parse_array
    cmp eax, '"'
    je parse_string
    cmp eax, '-'
    je parse_number
    cmp eax, '0'
    jb pv_err
    cmp eax, '9'
    jbe parse_number
    cmp eax, 't'
    je pv_true
    cmp eax, 'f'
    je pv_false
    cmp eax, 'n'
    je pv_null
pv_err:
    jmp error

pv_true:
    inc qword ptr [pos]
    call peek
    cmp eax, 'r'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 'u'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 'e'
    jne error
    inc qword ptr [pos]
    ret

pv_false:
    inc qword ptr [pos]
    call peek
    cmp eax, 'a'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 'l'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 's'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 'e'
    jne error
    inc qword ptr [pos]
    ret

pv_null:
    inc qword ptr [pos]
    call peek
    cmp eax, 'u'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 'l'
    jne error
    inc qword ptr [pos]
    call peek
    cmp eax, 'l'
    jne error
    inc qword ptr [pos]
    ret

parse_object:
    inc qword ptr [pos]
    call skip_ws
    call peek
    cmp eax, '}'
    je po_close
po_loop:
    call peek
    cmp eax, '"'
    jne error
    call parse_string
    call skip_ws
    call peek
    cmp eax, ':'
    jne error
    inc qword ptr [pos]
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, ','
    je po_comma
    cmp eax, '}'
    je po_close
    jmp error
po_comma:
    inc qword ptr [pos]
    call skip_ws
    jmp po_loop
po_close:
    inc qword ptr [pos]
    ret

parse_array:
    inc qword ptr [pos]
    call skip_ws
    call peek
    cmp eax, ']'
    je pa_close
pa_loop:
    call parse_value
    call skip_ws
    call peek
    cmp eax, ','
    je pa_comma
    cmp eax, ']'
    je pa_close
    jmp error
pa_comma:
    inc qword ptr [pos]
    call skip_ws
    jmp pa_loop
pa_close:
    inc qword ptr [pos]
    ret

parse_string:
    inc qword ptr [pos]
ps_loop:
    call peek
    cmp eax, -1
    je error
    cmp eax, '"'
    je ps_end
    cmp eax, 0x5C
    je ps_esc
    cmp eax, 0x20
    jb error
    inc qword ptr [pos]
    jmp ps_loop
ps_end:
    inc qword ptr [pos]
    ret
ps_esc:
    inc qword ptr [pos]
    call peek
    cmp eax, '"'
    je ps_simple
    cmp eax, 0x5C
    je ps_simple
    cmp eax, '/'
    je ps_simple
    cmp eax, 'b'
    je ps_simple
    cmp eax, 'f'
    je ps_simple
    cmp eax, 'n'
    je ps_simple
    cmp eax, 'r'
    je ps_simple
    cmp eax, 't'
    je ps_simple
    cmp eax, 'u'
    je ps_u
    jmp error
ps_simple:
    inc qword ptr [pos]
    jmp ps_loop
ps_u:
    inc qword ptr [pos]
    mov r13d, 4
ps_u_loop:
    call peek
    cmp eax, '0'
    jb error
    cmp eax, '9'
    jbe ps_u_ok
    cmp eax, 'a'
    jb ps_u_chkA
    cmp eax, 'f'
    jbe ps_u_ok
    jmp error
ps_u_chkA:
    cmp eax, 'A'
    jb error
    cmp eax, 'F'
    ja error
ps_u_ok:
    inc qword ptr [pos]
    dec r13d
    jnz ps_u_loop
    jmp ps_loop

parse_number:
    call peek
    cmp eax, '-'
    jne pn_int
    inc qword ptr [pos]
pn_int:
    call peek
    cmp eax, '0'
    je pn_zero
    cmp eax, '1'
    jb error
    cmp eax, '9'
    ja error
    inc qword ptr [pos]
pn_digits:
    call peek
    cmp eax, '0'
    jb pn_frac
    cmp eax, '9'
    ja pn_frac
    inc qword ptr [pos]
    jmp pn_digits
pn_zero:
    inc qword ptr [pos]
pn_frac:
    call peek
    cmp eax, '.'
    jne pn_exp
    inc qword ptr [pos]
    call peek
    cmp eax, '0'
    jb error
    cmp eax, '9'
    ja error
    inc qword ptr [pos]
pn_frac_digits:
    call peek
    cmp eax, '0'
    jb pn_exp
    cmp eax, '9'
    ja pn_exp
    inc qword ptr [pos]
    jmp pn_frac_digits
pn_exp:
    call peek
    cmp eax, 'e'
    je pn_exp_go
    cmp eax, 'E'
    je pn_exp_go
    ret
pn_exp_go:
    inc qword ptr [pos]
    call peek
    cmp eax, '+'
    je pn_exp_sign
    cmp eax, '-'
    je pn_exp_sign
    jmp pn_exp_d1
pn_exp_sign:
    inc qword ptr [pos]
pn_exp_d1:
    call peek
    cmp eax, '0'
    jb error
    cmp eax, '9'
    ja error
    inc qword ptr [pos]
pn_exp_digits:
    call peek
    cmp eax, '0'
    jb pn_done
    cmp eax, '9'
    ja pn_done
    inc qword ptr [pos]
    jmp pn_exp_digits
pn_done:
    ret

error:
    mov eax, 1
    mov edi, 1
    lea rsi, [err_msg]
    mov edx, 9
    syscall
    mov rax, [pos]
    lea rbp, [numbuf + 20]
    mov byte ptr [rbp], 10
    mov rsi, rbp
    mov rbx, 10
en_conv:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz en_conv
    mov rdx, rbp
    sub rdx, rsi
    inc rdx
    mov rax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok_msg:  .ascii "ok\n"
err_msg: .ascii "error at "

.bss
.balign 8
pos:     .zero 8
len:     .zero 8
numbuf:  .zero 32
buffer:  .zero 65552
