.globl _start

.text
_start:
    # read all input into inbuf
    xor r15, r15                      # r15 = bytes read so far
read_loop:
    cmp r15, 65536
    jae read_done
    xor eax, eax                      # read
    xor edi, edi                      # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov edx, 65536
    sub edx, r15d
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    mov rbp, r15                      # rbp = n (total length)

    # find first '!'
    xor rcx, rcx
find_bang:
    cmp rcx, rbp
    je no_bang
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, '!'
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r13, rcx                      # progLen
    lea r15, [rcx + 1]                # input start index
    jmp split_done
no_bang:
    mov r13, rbp                      # progLen = n
    mov r15, rbp                      # input start = n (empty)
split_done:

    # build jump table / validate brackets
    xor r8, r8                        # stack depth
    xor rcx, rcx                      # pos
build_loop:
    cmp rcx, r13
    je build_done
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, '['
    je push_b
    cmp al, ']'
    je pop_b
build_cont:
    inc rcx
    jmp build_loop
push_b:
    mov dword ptr [stackbuf + r8*4], ecx
    inc r8
    jmp build_cont
pop_b:
    test r8, r8
    jz unbalanced
    dec r8
    mov edx, dword ptr [stackbuf + r8*4]
    mov dword ptr [jumps + rcx*4], edx
    mov dword ptr [jumps + rdx*4], ecx
    jmp build_cont
build_done:
    test r8, r8
    jnz unbalanced

    # run
    xor r12, r12                      # ip
    lea r14, [rip + tape]             # data pointer
    xor ebx, ebx                      # output length
run_loop:
    cmp r12, r13
    jae run_done
    movzx eax, byte ptr [inbuf + r12]
    cmp al, '>'
    je c_gt
    cmp al, '<'
    je c_lt
    cmp al, '+'
    je c_plus
    cmp al, '-'
    je c_minus
    cmp al, '.'
    je c_dot
    cmp al, ','
    je c_comma
    cmp al, '['
    je c_lb
    cmp al, ']'
    je c_rb
next:
    inc r12
    jmp run_loop
c_gt:
    inc r14
    jmp next
c_lt:
    dec r14
    jmp next
c_plus:
    inc byte ptr [r14]
    jmp next
c_minus:
    dec byte ptr [r14]
    jmp next
c_dot:
    mov al, byte ptr [r14]
    mov byte ptr [outbuf + rbx], al
    inc rbx
    jmp next
c_comma:
    cmp r15, rbp
    jae store_zero
    movzx eax, byte ptr [inbuf + r15]
    inc r15
    mov byte ptr [r14], al
    jmp next
store_zero:
    mov byte ptr [r14], 0
    jmp next
c_lb:
    cmp byte ptr [r14], 0
    jne next
    mov r12d, dword ptr [jumps + r12*4]
    jmp next
c_rb:
    cmp byte ptr [r14], 0
    je next
    mov r12d, dword ptr [jumps + r12*4]
    jmp next

run_done:
    # flush output
    xor r12, r12                      # written so far
flush_loop:
    cmp r12, rbx
    jae exit0
    mov eax, 1                        # write
    mov edi, 1                        # fd 1
    lea rsi, [rip + outbuf]
    add rsi, r12
    mov rdx, rbx
    sub rdx, r12
    syscall
    test rax, rax
    jle exit0
    add r12, rax
    jmp flush_loop
exit0:
    mov eax, 60
    xor edi, edi
    syscall

unbalanced:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65536
tape:
    .zero 30000
jumps:
    .zero 262144
stackbuf:
    .zero 262144
