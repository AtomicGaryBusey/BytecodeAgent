.text
.globl _start
_start:
    lea rsi, [rip + buf]
    xor r12d, r12d          # total bytes read
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jl read_loop
read_done:
    # r12 = total length
    # process lines into outbuf
    lea rsi, [rip + buf]     # input base
    lea rdi, [rip + outbuf]  # output ptr
    xor rbx, rbx             # line start index
    xor rcx, rcx             # current index
scan:
    cmp rcx, r12
    jge final
    mov al, byte ptr [rsi + rcx]
    cmp al, 10
    jne scan_next
    # line is [rbx, rcx), reverse it
    mov r8, rcx
rev_loop:
    cmp r8, rbx
    jle rev_done
    dec r8
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev_loop
rev_done:
    mov byte ptr [rdi], 10
    inc rdi
    lea rbx, [rcx + 1]
scan_next:
    inc rcx
    jmp scan
final:
    cmp rbx, r12
    jge write_out
    # final line [rbx, r12)
    mov r8, r12
rev_loop2:
    cmp r8, rbx
    jle rev_done2
    dec r8
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rdi], al
    inc rdi
    jmp rev_loop2
rev_done2:
    mov byte ptr [rdi], 10
    inc rdi
write_out:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
buf:
    .zero 65536
outbuf:
    .zero 131072
