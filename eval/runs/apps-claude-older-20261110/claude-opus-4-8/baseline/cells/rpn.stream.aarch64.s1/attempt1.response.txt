.globl _start

.section .rodata
errstr:
    .ascii "error\n"

.bss
.balign 8
valstack:
    .zero 8192
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 524288

.text
_start:
    adrp    x24, inbuf
    add     x24, x24, :lo12:inbuf
    mov     x19, #0
read_loop:
    mov     x2, #65536
    sub     x2, x2, x19
    cbz     x2, read_done
    mov     x8, #63
    mov     x0, #0
    add     x1, x24, x19
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x19, x19, x0
    b       read_loop
read_done:
    add     x23, x24, x19
    mov     x21, x24
    adrp    x22, outbuf
    add     x22, x22, :lo12:outbuf

line_loop:
    cmp     x21, x23
    bhs     all_done
    mov     x25, x21
find_nl:
    cmp     x25, x23
    bhs     nl_notfound
    ldrb    w0, [x25]
    cmp     w0, #0x0A
    beq     nl_found
    add     x25, x25, #1
    b       find_nl
nl_found:
    mov     x26, x21
    mov     x27, x25
    add     x21, x25, #1
    b       do_process
nl_notfound:
    mov     x26, x21
    mov     x27, x25
    mov     x21, x25
do_process:
    bl      process_line
    b       line_loop

all_done:
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x22, x1
    cbz     x2, exit_prog
    mov     x0, #1
    mov     x8, #64
    svc     #0
exit_prog:
    mov     x8, #93
    mov     x0, #0
    svc     #0

// process_line: x26 = start, x27 = end
process_line:
    adrp    x10, valstack
    add     x10, x10, :lo12:valstack
    mov     x11, #0
    mov     x9, x26

tok_loop:
skip_sep:
    cmp     x9, x27
    bhs     tok_end
    ldrb    w0, [x9]
    cmp     w0, #0x20
    beq     advance_sep
    cmp     w0, #0x09
    beq     advance_sep
    b       tok_start
advance_sep:
    add     x9, x9, #1
    b       skip_sep

tok_start:
    mov     x12, x9
scan_tok:
    cmp     x9, x27
    bhs     tok_scanned
    ldrb    w0, [x9]
    cmp     w0, #0x20
    beq     tok_scanned
    cmp     w0, #0x09
    beq     tok_scanned
    add     x9, x9, #1
    b       scan_tok
tok_scanned:
    sub     x13, x9, x12
    cmp     x13, #1
    bne     try_int
    ldrb    w0, [x12]
    cmp     w0, #'+'
    beq     op_found
    cmp     w0, #'-'
    beq     op_found
    cmp     w0, #'*'
    beq     op_found
    cmp     w0, #'/'
    beq     op_found
    cmp     w0, #'%'
    beq     op_found
    b       try_int

op_found:
    cmp     x11, #2
    blt     line_error
    sub     x11, x11, #1
    ldr     x2, [x10, x11, lsl #3]
    sub     x11, x11, #1
    ldr     x1, [x10, x11, lsl #3]
    cmp     w0, #'+'
    beq     do_add
    cmp     w0, #'-'
    beq     do_sub
    cmp     w0, #'*'
    beq     do_mul
    cmp     w0, #'/'
    beq     do_div
    b       do_mod

do_add:
    add     x3, x1, x2
    b       push_res
do_sub:
    sub     x3, x1, x2
    b       push_res
do_mul:
    mul     x3, x1, x2
    b       push_res
do_div:
    cbz     x2, line_error
    cmn     x2, #1
    bne     div_ok
    mov     x4, #1
    lsl     x4, x4, #63
    cmp     x1, x4
    beq     line_error
div_ok:
    sdiv    x3, x1, x2
    b       push_res
do_mod:
    cbz     x2, line_error
    cmn     x2, #1
    bne     mod_ok
    mov     x4, #1
    lsl     x4, x4, #63
    cmp     x1, x4
    beq     line_error
mod_ok:
    sdiv    x4, x1, x2
    msub    x3, x4, x2, x1
    b       push_res

push_res:
    str     x3, [x10, x11, lsl #3]
    add     x11, x11, #1
    b       tok_loop

try_int:
    mov     x14, x12
    mov     x15, #0
    ldrb    w0, [x14]
    cmp     w0, #'-'
    bne     digits_check
    mov     x15, #1
    add     x14, x14, #1
digits_check:
    sub     x5, x9, x14
    cmp     x5, #1
    blt     line_error
    cmp     x5, #18
    bgt     line_error
    mov     x6, #0
int_conv:
    cmp     x14, x9
    bhs     int_done
    ldrb    w0, [x14]
    sub     w0, w0, #'0'
    cmp     w0, #9
    bhi     line_error
    mov     x7, #10
    mul     x6, x6, x7
    add     x6, x6, x0, uxtw
    add     x14, x14, #1
    b       int_conv
int_done:
    cbz     x15, int_push
    neg     x6, x6
int_push:
    str     x6, [x10, x11, lsl #3]
    add     x11, x11, #1
    b       tok_loop

tok_end:
    cmp     x11, #1
    bne     line_error
    ldr     x0, [x10]
    // print signed integer in x0 to outbuf
    mov     x3, x0
    tbz     x3, #63, print_pos
    mov     w0, #'-'
    strb    w0, [x22], #1
    neg     x3, x3
print_pos:
    sub     sp, sp, #32
    mov     x4, sp
    add     x4, x4, #31
    mov     x5, #0
    cbnz    x3, conv_loop
    mov     w0, #'0'
    strb    w0, [x4]
    sub     x4, x4, #1
    mov     x5, #1
    b       conv_emit
conv_loop:
    cbz     x3, conv_emit
    mov     x6, #10
    udiv    x7, x3, x6
    msub    x8, x7, x6, x3
    add     w8, w8, #'0'
    strb    w8, [x4]
    sub     x4, x4, #1
    mov     x3, x7
    add     x5, x5, #1
    b       conv_loop
conv_emit:
    add     x4, x4, #1
copy_loop:
    cbz     x5, copy_done
    ldrb    w0, [x4], #1
    strb    w0, [x22], #1
    sub     x5, x5, #1
    b       copy_loop
copy_done:
    mov     w0, #0x0A
    strb    w0, [x22], #1
    add     sp, sp, #32
    ret

line_error:
    adrp    x0, errstr
    add     x0, x0, :lo12:errstr
    mov     x5, #6
err_copy:
    ldrb    w1, [x0], #1
    strb    w1, [x22], #1
    sub     x5, x5, #1
    cbnz    x5, err_copy
    ret
