.section .bss
.align 4
inbuf:
    .zero 65536
outbuf:
    .zero 524288
valstack:
    .zero 9600

.text
.globl _start

_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    adrp x21, outbuf
    add  x21, x21, :lo12:outbuf
    mov  x22, #0
    mov  x20, #0
read_loop:
    mov  x0, #0
    add  x1, x19, x20
    mov  x2, #65536
    sub  x2, x2, x20
    cbz  x2, read_done
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop
read_done:
    add  x20, x19, x20        // x20 = end
    mov  x23, x19             // p = start
proc_loop:
    cmp  x23, x20
    b.ge proc_done
    mov  x0, x23              // lstart
find_nl:
    cmp  x23, x20
    b.ge got_lineend
    ldrb w9, [x23]
    cmp  w9, #0x0a
    b.eq got_lineend
    add  x23, x23, #1
    b    find_nl
got_lineend:
    mov  x1, x23
    bl   process_line
    cmp  x23, x20
    b.ge proc_loop
    add  x23, x23, #1
    b    proc_loop
proc_done:
    mov  x0, #1
    adrp x1, outbuf
    add  x1, x1, :lo12:outbuf
    mov  x2, x22
    mov  x8, #64
    svc  #0
    mov  x0, #0
    mov  x8, #93
    svc  #0

// process_line(x0=start, x1=end)
process_line:
    stp  x29, x30, [sp, #-16]!
    mov  x24, x0              // q
    mov  x25, x1              // end
    adrp x26, valstack
    add  x26, x26, :lo12:valstack
    mov  x27, #0              // count
tok_loop:
skipsep:
    cmp  x24, x25
    b.ge end_tokens
    ldrb w9, [x24]
    cmp  w9, #0x20
    b.eq sep_adv
    cmp  w9, #0x09
    b.eq sep_adv
    b    got_token
sep_adv:
    add  x24, x24, #1
    b    skipsep
got_token:
    mov  x10, x24            // tstart
scantok:
    cmp  x24, x25
    b.ge tokend
    ldrb w9, [x24]
    cmp  w9, #0x20
    b.eq tokend
    cmp  w9, #0x09
    b.eq tokend
    add  x24, x24, #1
    b    scantok
tokend:
    mov  x11, x24           // tend
    sub  x12, x11, x10      // len
    cmp  x12, #1
    b.ne not_op
    ldrb w9, [x10]
    cmp  w9, #'+'
    b.eq do_op
    cmp  w9, #'-'
    b.eq do_op
    cmp  w9, #'*'
    b.eq do_op
    cmp  w9, #'/'
    b.eq do_op
    cmp  w9, #'%'
    b.eq do_op
not_op:
    mov  x13, x10           // c
    mov  x14, #0            // neg
    ldrb w9, [x13]
    cmp  w9, #'-'
    b.ne nodash
    mov  x14, #1
    add  x13, x13, #1
nodash:
    mov  x15, x13           // digstart
    mov  x0, #0             // val
    mov  x16, #10
parse_dig:
    cmp  x13, x11
    b.ge parse_done
    ldrb w9, [x13]
    sub  w9, w9, #'0'
    cmp  w9, #9
    b.hi Lerror
    madd x0, x0, x16, x9
    add  x13, x13, #1
    b    parse_dig
