.section .rodata
hexchars:
	.ascii "0123456789abcdef"

.bss
	.balign 16
inbuf:
	.zero 65536
	.balign 16
linebuf:
	.zero 128

.text
.globl _start

emit_hex_byte:
	adrp x9, hexchars
	add x9, x9, :lo12:hexchars
	and w10, w0, #0xf0
	lsr w10, w10, #4
	ldrb w11, [x9, x10]
	strb w11, [x1], #1
	and w10, w0, #0x0f
	ldrb w11, [x9, x10]
	strb w11, [x1], #1
	ret

emit_hex32:
	adrp x9, hexchars
	add x9, x9, :lo12:hexchars
	mov w12, #28
1:
	lsr w10, w0, w12
	and w10, w10, #0xf
	ldrb w11, [x9, x10]
	strb w11, [x1], #1
	subs w12, w12, #4
	bcs 1b
	ret

_start:
	adrp x19, inbuf
	add x19, x19, :lo12:inbuf
	mov x20, #0
read_loop:
	mov x0, #0
	add x1, x19, x20
	mov x2, #65536
	sub x2, x2, x20
	cbz x2, read_done
	mov x8, #63
	svc #0
	cmp x0, #0
	ble read_done
	add x20, x20, x0
	b read_loop
read_done:

	mov x21, #0
row_loop:
	cmp x21, x20
	bge rows_done
	sub x22, x20, x21
	mov x23, #16
	cmp x22, #16
	csel x22, x23, x22, ge

	adrp x24, linebuf
	add x24, x24, :lo12:linebuf
	mov x25, x24

	mov w0, w21
	mov x1, x24
	bl emit_hex32
	mov x24, x1
	mov w0, #0x20
	strb w0, [x24], #1
	strb w0, [x24], #1

	mov x26, #0
slot_loop:
	cmp x26, #16
	bge slot_done
	cmp x26, x22
	bge slot_pad
	add x9, x19, x21
	ldrb w0, [x9, x26]
	mov x1, x24
	bl emit_hex_byte
	mov x24, x1
	b slot_space
slot_pad:
	mov w0, #0x20
	strb w0, [x24], #1
	strb w0, [x24], #1
slot_space:
	mov w0, #0x20
	strb w0, [x24], #1
	cmp x26, #7
	bne 1f
	strb w0, [x24], #1
1:
	add x26, x26, #1
	b slot_loop
slot_done:

	mov w0, #0x20
	strb w0, [x24], #1
	mov w0, #0x7c
	strb w0, [x24], #1

	mov x26, #0
char_loop:
	cmp x26, x22
	bge char_done
	add x9, x19, x21
	ldrb w0, [x9, x26]
	cmp w0, #0x20
	blo 2f
	cmp w0, #0x7e
	bhi 2f
	b 3f
2:
	mov w0, #0x2e
3:
	strb w0, [x24], #1
	add x26, x26, #1
	b char_loop
char_done:
	mov w0, #0x7c
	strb w0, [x24], #1
	mov w0, #0x0a
	strb w0, [x24], #1

	mov x0, #1
	mov x1, x25
	sub x2, x24, x25
	mov x8, #64
	svc #0

	add x21, x21, #16
	b row_loop

rows_done:
	adrp x24, linebuf
	add x24, x24, :lo12:linebuf
	mov w0, w20
	mov x1, x24
	bl emit_hex32
	mov w0, #0x0a
	strb w0, [x1], #1
	mov x0, #1
	adrp x1, linebuf
	add x1, x1, :lo12:linebuf
	mov x2, #9
	mov x8, #64
	svc #0

	mov x0, #0
	mov x8, #93
	svc #0
