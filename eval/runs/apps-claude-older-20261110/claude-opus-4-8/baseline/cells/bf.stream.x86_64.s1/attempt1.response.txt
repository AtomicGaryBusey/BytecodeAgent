.text
.globl _start
_start:
    # read all of stdin into inbuf
    xor r14, r14
rd_loop:
    mov rdx, 65536
    sub rdx, r14
    jz rd_done
    xor rax, rax
    xor rdi, rdi
    lea rsi, [inbuf + r14]
    syscall
    test rax, rax
    jz rd_done
    add r14, rax
    jmp rd_loop
rd_done:
    mov [rip + in_end], r14

    # find first '!' (0x21)
    xor rcx, rcx
fb_loop:
    cmp rcx, r14
    jae fb_none
    cmp byte ptr [inbuf + rcx], 0x21
    je fb_found
    inc rcx
    jmp fb_loop
fb_found:
    mov [rip + prog_len], rcx
    lea rax, [rcx + 1]
    mov [rip + in_ptr], rax
    jmp validate
fb_none:
    mov [rip + prog_len], r14
    mov [rip + in_ptr], r14

validate:
    mov r12, [rip + prog_len]
    xor rcx, rcx
    xor rbp, rbp
val_loop:
    cmp rcx, r12
    jae val_end
    movzx eax, byte ptr [inbuf + rcx]
    cmp al, 0x5B
    je v_push
    cmp al, 0x5D
    je v_pop
    inc rcx
    jmp val_loop
v_push:
    mov dword ptr [brstack + rbp*4], ecx
    inc rbp
    inc rcx
    jmp val_loop
v_pop:
    test rbp, rbp
    jz unbalanced
    dec rbp
    mov edx, dword ptr [brstack + rbp*4]
    mov dword ptr [jumptab + rdx*4], ecx
    mov dword ptr [jumptab + rcx*4], edx
    inc rcx
    jmp val_loop
val_end:
    test rbp, rbp
    jnz unbalanced

    # run
    xor rbx, rbx
    xor r13, r13
run_loop:
    cmp rbx, r12
    jae run_done
    movzx eax, byte ptr [inbuf + rbx]
    cmp al, 0x3E
    je c_gt
    cmp al, 0x3C
    je c_lt
    cmp al, 0x2B
    je c_plus
    cmp al, 0x2D
    je c_minus
    cmp al, 0x2E
    je c_dot
    cmp al, 0x2C
    je c_comma
    cmp al, 0x5B
    je c_open
    cmp al, 0x5D
    je c_close
next:
    inc rbx
    jmp run_loop

c_gt:
    inc r13
    jmp next
c_lt:
    dec r13
    jmp next
c_plus:
    inc byte ptr [tape + r13]
    jmp next
c_minus:
    dec byte ptr [tape + r13]
    jmp next
c_dot:
    mov al, byte ptr [tape + r13]
    mov rdx, [rip + out_len]
    mov byte ptr [outbuf + rdx], al
    inc rdx
    mov [rip + out_len], rdx
    jmp next
c_comma:
    mov rdx, [rip + in_ptr]
    cmp rdx, [rip + in_end]
    jae comma_zero
    mov al, byte ptr [inbuf + rdx]
    inc rdx
    mov [rip + in_ptr], rdx
    mov byte ptr [tape + r13], al
    jmp next
comma_zero:
    mov byte ptr [tape + r13], 0
    jmp next
c_open:
    cmp byte ptr [tape + r13], 0
    jne next
    mov ebx, dword ptr [jumptab + rbx*4]
    jmp next
c_close:
    cmp byte ptr [tape + r13], 0
    je next
    mov ebx, dword ptr [jumptab + rbx*4]
    jmp next

run_done:
    xor r8, r8
fl_loop:
    mov rdx, [rip + out_len]
    sub rdx, r8
    jz fl_done
    mov rax, 1
    mov rdi, 1
    lea rsi, [outbuf + r8]
    syscall
    test rax, rax
    jle fl_done
    add r8, rax
    jmp fl_loop
fl_done:
    mov rax, 60
    xor rdi, rdi
    syscall

unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall

.bss
.align 16
inbuf:    .zero 65536
tape:     .zero 30000
outbuf:   .zero 65536
jumptab:  .zero 262144
brstack:  .zero 8192
prog_len: .zero 8
in_ptr:   .zero 8
in_end:   .zero 8
out_len:  .zero 8
