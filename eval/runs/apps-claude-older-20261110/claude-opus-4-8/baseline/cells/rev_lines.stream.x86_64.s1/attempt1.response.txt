.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rbx, rbx
    xor r13, r13
    xor rcx, rcx
scan:
    cmp rcx, r12
    jae after
    mov al, byte ptr [inbuf + rcx]
    cmp al, 0x0A
    jne 1f
    call emit_line
    inc rcx
    mov rbx, rcx
    jmp scan
1:
    inc rcx
    jmp scan
after:
    cmp rbx, r12
    jae done
    mov rcx, r12
    call emit_line
done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall
    mov eax, 60
    xor edi, edi
    syscall

emit_line:
    mov rdx, rcx
emit_loop:
    cmp rdx, rbx
    jbe emit_nl
    dec rdx
    mov al, byte ptr [inbuf + rdx]
    mov byte ptr [outbuf + r13], al
    inc r13
    jmp emit_loop
emit_nl:
    mov byte ptr [outbuf + r13], 0x0A
    inc r13
    ret

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65552
