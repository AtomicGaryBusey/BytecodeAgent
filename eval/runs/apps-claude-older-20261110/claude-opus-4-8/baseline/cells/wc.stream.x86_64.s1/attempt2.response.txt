.globl _start

.section .rodata
wstable:
.zero 9
.byte 1,1,1,1,1
.zero 18
.byte 1
.zero 223

.bss
.align 16
buf:
.zero 65536
outbuf:
.zero 64

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jz done_read
    add r13, rax
    mov rcx, rax
    lea rsi, [rip + buf]
    lea rbp, [rip + wstable]

proc:
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rbp + rax]
    cmp al, 10
    sete al
    movzx eax, al
    add r12, rax
    test edx, edx
    jnz is_ws
    test r15d, r15d
    jnz skip_word
    inc r14
    mov r15d, 1
    jmp cont
is_ws:
    xor r15d, r15d
    jmp cont
skip_word:
cont:
    inc rsi
    dec rcx
    jnz proc
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf]
    mov rsi, r12
    call putnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rsi, r14
    call putnum
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rsi, r13
    call putnum
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

putnum:
    mov rax, rsi
    mov r8, rdi
    mov r9, rsp
    mov rbx, 10
    test rax, rax
    jnz pn_loop
    mov byte ptr [r8], 0x30
    inc r8
    mov rdi, r8
    ret
pn_loop:
    test rax, rax
    jz pn_emit
    xor edx, edx
    div rbx
    add dl, 0x30
    dec r9
    mov byte ptr [r9], dl
    jmp pn_loop
pn_emit:
    mov rax, rsp
pn_copy:
    cmp r9, rax
    jae pn_done
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    inc r8
    inc r9
    jmp pn_copy
pn_done:
    mov rdi, r8
    ret
