.bss
.align 16
inbuf:
.zero 65536
outbuf:
.zero 65540

.text
.globl _start
_start:
    lea rsi, [rip + inbuf]
    xor r12, r12            # total bytes read
read_loop:
    xor eax, eax            # read
    xor edi, edi            # fd 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = total input bytes
    lea r13, [rip + inbuf]      # input base
    lea r14, [rip + outbuf]     # output write ptr
    xor rbx, rbx                # line_start index
    xor r15, r15                # i
scan:
    cmp r15, r12
    jae tail
    mov al, byte ptr [r13 + r15]
    cmp al, 0x0A
    jne scan_next
    # emit reverse of [rbx, r15)
    mov rsi, r15                # end (exclusive)
    mov rdi, rbx                # start
emit_rev:
    cmp rsi, rdi
    jbe emit_nl
    dec rsi
    mov cl, byte ptr [r13 + rsi]
    mov byte ptr [r14], cl
    inc r14
    jmp emit_rev
emit_nl:
    mov byte ptr [r14], 0x0A
    inc r14
    lea rbx, [r15 + 1]
scan_next:
    inc r15
    jmp scan
tail:
    cmp rbx, r12
    jae do_write
    mov rsi, r12
    mov rdi, rbx
tail_rev:
    cmp rsi, rdi
    jbe tail_nl
    dec rsi
    mov cl, byte ptr [r13 + rsi]
    mov byte ptr [r14], cl
    inc r14
    jmp tail_rev
tail_nl:
    mov byte ptr [r14], 0x0A
    inc r14
do_write:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
do_exit:
    mov eax, 60
    xor edi, edi
    syscall
