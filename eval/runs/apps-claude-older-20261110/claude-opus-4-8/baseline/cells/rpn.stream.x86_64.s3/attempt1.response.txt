.globl _start

.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov rdx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    cmp r13, 65536
    jb read_loop
read_done:
    lea r15, [rip + outbuf]
    mov rbx, r12
    lea r14, [r12 + r13]

line_loop:
    cmp rbx, r14
    jae all_done
    mov rdi, rbx
    mov rsi, rbx
scan_nl:
    cmp rsi, r14
    jae scan_nl_end
    cmp byte ptr [rsi], 10
    je scan_nl_end
    inc rsi
    jmp scan_nl
scan_nl_end:
    mov rbx, rsi
    cmp rsi, r14
    jae no_adv
    inc rbx
no_adv:
    call eval_line
    jmp line_loop

all_done:
    lea rbx, [rip + outbuf]
    mov r14, r15
wloop:
    cmp rbx, r14
    jae wdone
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r14
    sub rdx, rbx
    syscall
    test rax, rax
    jle wdone
    add rbx, rax
    jmp wloop
wdone:
    mov eax, 60
    xor edi, edi
    syscall

# eval_line: rdi=start rsi=end. clobbers r8-r13,rbp,rax-rdx,rsi,rdi
eval_line:
    mov r8, rdi
    mov r9, rsi
    lea r10, [rip + evalstack]
    xor r11d, r11d
tok_loop:
    cmp r8, r9
    jae tok_end
    movzx eax, byte ptr [r8]
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    jmp tok_start
skip_sep:
    inc r8
    jmp tok_loop
tok_start:
    mov rdi, r8
scan_tok:
    cmp r8, r9
    jae tok_got
    movzx eax, byte ptr [r8]
    cmp al, 32
    je tok_got
    cmp al, 9
    je tok_got
    inc r8
    jmp scan_tok
tok_got:
    mov rdx, r8
    sub rdx, rdi
    cmp rdx, 1
    jne try_int
    movzx eax, byte ptr [rdi]
    cmp al, '+'
    je do_op
    cmp al, '-'
    je do_op
    cmp al, '*'
    je do_op
    cmp al, '/'
    je do_op
    cmp al, '%'
    je do_op
try_int:
    mov rsi, rdi
    mov rdx, r8
    xor r12d, r12d
    cmp byte ptr [rsi], '-'
    jne no_neg
    mov r12d, 1
    inc rsi
no_neg:
    xor rcx, rcx
    xor ebp, ebp
digit_loop:
    cmp rsi, rdx
    jae digits_done
    movzx eax, byte ptr [rsi]
    sub eax, '0'
    cmp eax, 9
    ja line_error
    lea rcx, [rcx + rcx*4]
    lea rcx, [rax + rcx*2]
    inc ebp
    inc rsi
    jmp digit_loop
digits_done:
    test ebp, ebp
    jz line_error
    cmp ebp, 18
    ja line_error
    test r12d, r12d
    jz int_pos
    neg rcx
int_pos:
    mov rax, r11
    mov qword ptr [r10 + rax*8], rcx
    inc r11d
    jmp tok_loop

do_op:
    cmp r11d, 2
    jb line_error
    mov rcx, r11
    mov rsi, qword ptr [r10 + rcx*8 - 8]
    mov rdi, qword ptr [r10 + rcx*8 - 16]
    sub r11d, 2
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
op_mod:
    test rsi, rsi
    jz line_error
    mov rcx, 0x8000000000000000
    cmp rdi, rcx
    jne mod_ok
    cmp rsi, -1
    je line_error
mod_ok:
    mov rax, rdi
    cqo
    idiv rsi
    mov rdi, rdx
    jmp push_res
op_add:
    add rdi, rsi
    jmp push_res
op_sub:
    sub rdi, rsi
    jmp push_res
op_mul:
    imul rdi, rsi
    jmp push_res
op_div:
    test rsi, rsi
    jz line_error
    mov rcx, 0x8000000000000000
    cmp rdi, rcx
    jne div_ok
    cmp rsi, -1
    je line_error
div_ok:
    mov rax, rdi
    cqo
    idiv rsi
    mov rdi, rax
    jmp push_res
push_res:
    mov rcx, r11
    mov qword ptr [r10 + rcx*8], rdi
    inc r11d
    jmp tok_loop

tok_end:
    cmp r11d, 1
    jne line_error
    mov rax, qword ptr [r10]
    call emit_num
    ret
line_error:
    call emit_error
    ret

emit_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6
    ret

emit_num:
    test rax, rax
    jns num_pos
    mov byte ptr [r15], '-'
    inc r15
    neg rax
num_pos:
    mov r8, 10
    lea rcx, [rsp - 1]
num_div:
    xor edx, edx
    div r8
    add dl, '0'
    mov byte ptr [rcx], dl
    dec rcx
    test rax, rax
    jnz num_div
    inc rcx
    lea rdx, [rsp - 1]
num_copy:
    mov al, byte ptr [rcx]
    mov byte ptr [r15], al
    inc r15
    cmp rcx, rdx
    je num_done
    inc rcx
    jmp num_copy
num_done:
    mov byte ptr [r15], 10
    inc r15
    ret

.bss
.balign 16
inbuf:
    .zero 65536
evalstack:
    .zero 8192
outbuf:
    .zero 700000
