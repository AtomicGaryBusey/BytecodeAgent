.globl _start

.bss
.balign 16
buf:
.zero 65536
outbuf:
.zero 64

.text
_start:
    xor r12, r12          # lines
    xor r13, r13          # words
    xor r14, r14          # bytes
    xor r15, r15          # inword flag (0=not in word)

read_loop:
    xor eax, eax          # read
    xor edi, edi          # fd 0
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax          # bytes += n

    lea rsi, [rip + buf]
    mov rcx, rax          # count
proc:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A
    jne not_nl
    inc r12
not_nl:
    # check whitespace
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb not_ws
    cmp al, 0x0D
    jbe is_ws
not_ws:
    # word byte
    test r15, r15
    jnz skip_count
    inc r13
    mov r15, 1
skip_count:
    jmp next
is_ws:
    xor r15, r15
next:
    dec rcx
    jnz proc
    jmp read_loop

done_read:
    # output: lines words bytes
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call put_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call put_num
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

# put_num: writes rax as decimal to [rdi], advances rdi
put_num:
    mov r8, rdi
    mov r9, rsi
    lea rsi, [rip + outbuf + 60]   # temp area end
    mov rcx, 10
    test rax, rax
    jnz pn_loop
    dec rsi
    mov byte ptr [rsi], '0'
    jmp pn_copy
pn_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz pn_loop
pn_copy:
    # copy from rsi to [r8], until outbuf+60
    lea rax, [rip + outbuf + 60]
pn_c2:
    cmp rsi, rax
    jae pn_done
    mov dl, byte ptr [rsi]
    mov byte ptr [r8], dl
    inc rsi
    inc r8
    jmp pn_c2
pn_done:
    mov rdi, r8
    mov rsi, r9
    ret
