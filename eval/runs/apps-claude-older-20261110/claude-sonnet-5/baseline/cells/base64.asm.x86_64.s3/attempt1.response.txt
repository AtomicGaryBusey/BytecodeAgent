.equ INBUF_SIZE, 49152
.equ OUTBUF_SIZE, 70000

.globl _start

.text

_start:
    xor r15, r15
.read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov rdi, 0
    mov rdx, INBUF_SIZE
    sub rdx, r15
    xor rax, rax
    syscall
    test rax, rax
    jz .read_done
    add r15, rax
    jmp .read_loop
.read_done:

    lea rbx, [rip + inbuf]
    lea r10, [rip + outbuf]
    xor r13, r13
    xor r14, r14
    xor r12, r12
    mov r9, r15

.group_loop:
    cmp r9, 3
    jl .tail
    movzx ecx, byte ptr [rbx + r12]
    movzx edx, byte ptr [rbx + r12 + 1]
    movzx r8d, byte ptr [rbx + r12 + 2]

    mov eax, ecx
    shr eax, 2
    call emit_idx

    mov eax, ecx
    and eax, 3
    shl eax, 4
    mov esi, edx
    shr esi, 4
    or eax, esi
    call emit_idx

    mov eax, edx
    and eax, 0xf
    shl eax, 2
    mov esi, r8d
    shr esi, 6
    or eax, esi
    call emit_idx

    mov eax, r8d
    and eax, 0x3f
    call emit_idx

    add r12, 3
    sub r9, 3
    jmp .group_loop

.tail:
    cmp r9, 1
    je .tail1
    cmp r9, 2
    je .tail2
    jmp .finish

.tail1:
    movzx ecx, byte ptr [rbx + r12]
    mov eax, ecx
    shr eax, 2
    call emit_idx
    mov eax, ecx
    and eax, 3
    shl eax, 4
    call emit_idx
    mov eax, 61
    call emit_char
    mov eax, 61
    call emit_char
    jmp .finish

.tail2:
    movzx ecx, byte ptr [rbx + r12]
    movzx edx, byte ptr [rbx + r12 + 1]
    mov eax, ecx
    shr eax, 2
    call emit_idx
    mov eax, ecx
    and eax, 3
    shl eax, 4
    mov esi, edx
    shr esi, 4
    or eax, esi
    call emit_idx
    mov eax, edx
    and eax, 0xf
    shl eax, 2
    call emit_idx
    mov eax, 61
    call emit_char
    jmp .finish

.finish:
    test r14, r14
    jz .do_write
    mov byte ptr [r10 + r13], 10
    inc r13
    xor r14, r14

.do_write:
    xor r12, r12
.write_loop:
    cmp r12, r13
    jge .do_exit
    lea rsi, [rip + outbuf]
    add rsi, r12
    mov rdi, 1
    mov rdx, r13
    sub rdx, r12
    mov rax, 1
    syscall
    add r12, rax
    jmp .write_loop

.do_exit:
    mov rax, 231
    xor rdi, rdi
    syscall

emit_idx:
    lea r11, [rip + b64table]
    movzx eax, byte ptr [r11 + rax]
    call emit_char
    ret

emit_char:
    mov byte ptr [r10 + r13], al
    inc r13
    inc r14
    cmp r14, 76
    jne .ec_done
    mov byte ptr [r10 + r13], 10
    inc r13
    xor r14, r14
.ec_done:
    ret

.section .rodata
b64table:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
.zero INBUF_SIZE
outbuf:
.zero OUTBUF_SIZE
