.text
.globl _start

_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0            // total bytes read

read_loop:
    mov x21, #65536
    sub x21, x21, x20
    cbz x21, after_read
    mov x8, #63            // read
    mov x0, #0
    add x1, x19, x20
    mov x2, x21
    svc #0
    cmp x0, #0
    beq after_read
    add x20, x20, x0
    b read_loop

after_read:
    mov x22, #0            // lines
    mov x23, #0            // words
    mov x24, #0            // in_word flag
    mov x25, #0            // index

process_loop:
    cmp x25, x20
    bge process_done
    ldrb w0, [x19, x25]
    add x25, x25, #1

    cmp w0, #0x0A
    bne not_nl
    add x22, x22, #1
not_nl:
    cmp w0, #9
    blt not_ws
    cmp w0, #13
    ble is_ws
    cmp w0, #0x20
    beq is_ws
    b not_ws
is_ws:
    mov x24, #0
    b process_loop
not_ws:
    cbnz x24, process_loop
    add x23, x23, #1
    mov x24, #1
    b process_loop

process_done:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf

    mov x0, x22
    bl write_num
    mov x1, x0
    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x23
    bl write_num
    mov x1, x0
    mov w2, #0x20
    strb w2, [x1], #1

    mov x0, x20
    bl write_num
    mov x1, x0
    mov w2, #0x0A
    strb w2, [x1], #1

    adrp x3, outbuf
    add x3, x3, :lo12:outbuf
    sub x4, x1, x3

    mov x8, #64            // write
    mov x0, #1
    mov x1, x3
    mov x2, x4
    svc #0

    mov x8, #93            // exit
    mov x0, #0
    svc #0

write_num:
    cbnz x0, wn_nonzero
    mov w2, #'0'
    strb w2, [x1], #1
    mov x0, x1
    ret
wn_nonzero:
    sub sp, sp, #32
    mov x2, sp
    mov x3, x0
    mov x4, #0
wn_loop:
    mov x6, #10
    udiv x7, x3, x6
    msub x8, x7, x6, x3
    add x8, x8, #'0'
    strb w8, [x2], #1
    add x4, x4, #1
    mov x3, x7
    cbnz x3, wn_loop
wn_copy:
    sub x2, x2, #1
    ldrb w8, [x2]
    strb w8, [x1], #1
    subs x4, x4, #1
    bne wn_copy
    add sp, sp, #32
    mov x0, x1
    ret

.bss
.align 3
buf:
    .zero 65536
outbuf:
    .zero 32
