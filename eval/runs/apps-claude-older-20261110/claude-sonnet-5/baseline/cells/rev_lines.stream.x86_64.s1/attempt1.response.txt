```
.section .bss
buf: .zero 65536

.section .rodata
nl: .byte 10

.text
.globl _start
_start:
    xor r15, r15        # total bytes read

read_loop:
    lea rsi, [rip + buf]
    add rsi, r15
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r15
    cmp rdx, 0
    je after_read
    mov rax, 0
    syscall
    test rax, rax
    jz after_read
    add r15, rax
    jmp read_loop

after_read:
    xor rbx, rbx        # pos
    xor r13, r13        # start of current line

process_loop:
    cmp rbx, r15
    jge process_done
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp eax, 10
    jne not_newline
    call reverse_line
    inc rbx
    mov r13, rbx
    jmp process_loop
not_newline:
    inc rbx
    jmp process_loop

process_done:
    cmp r13, r15
    jge no_remainder
    mov rbx, r15
    call reverse_line

no_remainder:
    test r15, r15
    jz write_extra_check
    lea rsi, [rip + buf]
    mov rdi, 1
    mov rdx, r15
    mov rax, 1
    syscall

write_extra_check:
    cmp r13, r15
    jge exit_prog
    lea rsi, [rip + nl]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall

exit_prog:
    mov rax, 60
    mov rdi, 0
    syscall

reverse_line:
    lea rsi, [rip + buf]
    mov rax, r13
    mov rcx, rbx
    dec rcx
rev_loop:
    cmp rax, rcx
    jge rev_done
    movzx r8d, byte ptr [rsi + rax]
    movzx r9d, byte ptr [rsi + rcx]
    mov byte ptr [rsi + rax], r9b
    mov byte ptr [rsi + rcx], r8b
    inc rax
    dec rcx
    jmp rev_loop
rev_done:
    ret
```
