```
.globl _start

.equ INPUT_CAP, 65536
.equ TABLE_CAP, 65536
.equ STACK_CAP, 1000

.text

_start:
    # ---- read all of stdin into input_buf, accumulate length in r12 ----
    xor r12, r12
read_loop:
    xor rdi, rdi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, INPUT_CAP
    sub rdx, r12
    xor rax, rax
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop
read_done:
    # r12 = total_len

    lea rbx, [rip + input_buf]   # program_base / full buffer base

    # ---- scan for first '!' ----
    xor rcx, rcx
    mov rdx, -1
scan_loop:
    cmp rcx, r12
    jge scan_end
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, 0x21
    je scan_set
    inc rcx
    jmp scan_loop
scan_set:
    mov rdx, rcx
scan_end:
    cmp rdx, -1
    je no_bang

    mov r10, rbx
    add r10, rdx
    add r10, 1
    mov rax, r12
    sub rax, rdx
    sub rax, 1
    mov r11, rax
    mov r12, rdx
    jmp scan_done

no_bang:
    xor r11, r11
    mov r10, rbx
scan_done:
    # r12 = program_len, r10 = prog_input_base, r11 = prog_input_len

    lea r13, [rip + match_table]
    lea r14, [rip + tape]
    lea r15, [rip + output_buf]

    # ---- build match table ----
    lea rsi, [rip + bracket_stack]
    xor rdx, rdx           # depth
    xor r9, r9             # loop index
build_loop:
    cmp r9, r12
    jge build_done
    movzx eax, byte ptr [rbx + r9]
    cmp eax, 0x5B
    je build_open
    cmp eax, 0x5D
    je build_close
    jmp build_next
build_open:
    mov qword ptr [rsi + rdx*8], r9
    inc rdx
    jmp build_next
build_close:
    cmp rdx, 0
    je error_exit
    dec rdx
    mov rax, qword ptr [rsi + rdx*8]
    mov dword ptr [r13 + r9*4], eax
    mov dword ptr [r13 + rax*4], r9d
    jmp build_next
build_next:
    inc r9
    jmp build_loop
build_done:
    cmp rdx, 0
    jne error_exit

    # ---- interpret ----
    xor r9, r9             # ip
    xor r8, r8             # pointer
    xor rbp, rbp            # output_len
    xor rcx, rcx            # prog_input_pos

interp_loop:
    cmp r9, r12
    jge interp_done
    movzx eax, byte ptr [rbx + r9]
    cmp eax, 0x3E
    je do_gt
    cmp eax, 0x3C
    je do_lt
    cmp eax, 0x2B
    je do_plus
    cmp eax, 0x2D
    je do_minus
    cmp eax, 0x2E
    je do_dot
    cmp eax, 0x2C
    je do_comma
    cmp eax, 0x5B
    je do_open
    cmp eax, 0x5D
    je do_close
    inc r9
    jmp interp_loop

do_gt:
    inc r8
    inc r9
    jmp interp_loop
do_lt:
    dec r8
    inc r9
    jmp interp_loop
do_plus:
    inc byte ptr [r14 + r8]
    inc r9
    jmp interp_loop
do_minus:
    dec byte ptr [r14 + r8]
    inc r9
    jmp interp_loop
do_dot:
    movzx eax, byte ptr [r14 + r8]
    mov byte ptr [r15 + rbp], al
    inc rbp
    inc r9
    jmp interp_loop
do_comma:
    cmp rcx, r11
    jge comma_zero
    movzx eax, byte ptr [r10 + rcx]
    mov byte ptr [r14 + r8], al
    inc rcx
    jmp comma_done
comma_zero:
    mov byte ptr [r14 + r8], 0
comma_done:
    inc r9
    jmp interp_loop
do_open:
    cmp byte ptr [r14 + r8], 0
    jne open_cont
    mov eax, dword ptr [r13 + r9*4]
    mov r9, rax
    inc r9
    jmp interp_loop
open_cont:
    inc r9
    jmp interp_loop
do_close:
    cmp byte ptr [r14 + r8], 0
    je close_cont
    mov eax, dword ptr [r13 + r9*4]
    mov r9, rax
    inc r9
    jmp interp_loop
close_cont:
    inc r9
    jmp interp_loop

interp_done:
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rdx, rbp
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

error_exit:
    mov rax, 60
    mov rdi, 1
    syscall

.bss
.align 8
input_buf:
    .zero INPUT_CAP
match_table:
    .zero TABLE_CAP * 4
bracket_stack:
    .zero STACK_CAP * 8
tape:
    .zero 30000
output_buf:
    .zero 65536
```
