.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 16
input_buf:
    .zero 65536
    .align 16
linebuf:
    .zero 128

.text
.globl _start

write_hex8:
    mov ecx, 28
    xor r8d, r8d
    lea r9, [rip + hex_digits]
whex_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xF
    movzx edx, byte ptr [r9 + rdx]
    mov byte ptr [rdi + r8], dl
    inc r8
    sub cl, 4
    cmp r8, 8
    jl whex_loop
    ret

_start:
    lea r14, [rip + input_buf]
    lea r10, [rip + linebuf]
    xor r15, r15

read_loop:
    lea rsi, [r14 + r15]
    mov rdx, 65536
    sub rdx, r15
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:

    xor r13, r13

row_loop:
    cmp r13, r15
    jge after_rows

    mov eax, r13d
    mov rdi, r10
    call write_hex8

    mov byte ptr [r10 + 8], 0x20
    mov byte ptr [r10 + 9], 0x20

    mov r12, r15
    sub r12, r13
    cmp r12, 16
    jle n_ok
    mov r12, 16
n_ok:

    mov r11, r14
    add r11, r13

    xor rcx, rcx
    mov rbx, 10

row_byte_loop:
    cmp rcx, 16
    jge row_byte_done
    cmp rcx, 8
    jne skip_extra
    mov byte ptr [r10 + rbx], 0x20
    inc rbx
skip_extra:
    cmp rcx, r12
    jl have_byte
    mov byte ptr [r10 + rbx], 0x20
    mov byte ptr [r10 + rbx + 1], 0x20
    mov byte ptr [r10 + rbx + 2], 0x20
    add rbx, 3
    jmp row_byte_cont
have_byte:
    movzx eax, byte ptr [r11 + rcx]
    mov edx, eax
    shr edx, 4
    and edx, 0xF
    lea r9, [rip + hex_digits]
    movzx edx, byte ptr [r9 + rdx]
    mov byte ptr [r10 + rbx], dl
    mov edx, eax
    and edx, 0xF
    movzx edx, byte ptr [r9 + rdx]
    mov byte ptr [r10 + rbx + 1], dl
    mov byte ptr [r10 + rbx + 2], 0x20
    add rbx, 3
row_byte_cont:
    inc rcx
    jmp row_byte_loop
row_byte_done:

    mov byte ptr [r10 + rbx], 0x20
    inc rbx
    mov byte ptr [r10 + rbx], 0x7C
    inc rbx

    xor rcx, rcx
ascii_loop:
    cmp rcx, r12
    jge ascii_done
    movzx eax, byte ptr [r11 + rcx]
    cmp eax, 0x20
    jl dot_char
    cmp eax, 0x7E
    jg dot_char
    jmp store_char
dot_char:
    mov eax, 0x2E
store_char:
    mov byte ptr [r10 + rbx], al
    inc rbx
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r10 + rbx], 0x7C
    inc rbx
    mov byte ptr [r10 + rbx], 0x0A
    inc rbx

    mov rdi, 1
    mov rsi, r10
    mov rdx, rbx
    mov rax, 1
    syscall

    add r13, 16
    jmp row_loop

after_rows:
    mov eax, r15d
    mov rdi, r10
    call write_hex8
    mov byte ptr [r10 + 8], 0x0A

    mov rdi, 1
    mov rsi, r10
    mov rdx, 9
    mov rax, 1
    syscall

    xor rdi, rdi
    mov rax, 60
    syscall
