```asm
.bss
    .align 8
input_buf:
    .zero 65536
tape:
    .zero 30000
    .align 4
match:
    .zero 262144
bstack:
    .zero 4004
outbuf:
    .zero 65536

.text
.globl _start
_start:
    # ---- read all of stdin into input_buf ----
    xor r8, r8                  # r8 = total length read so far
read_loop:
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    cmp rdx, 0
    jle read_done
    xor edi, edi                # fd = 0
    mov rax, 0                  # sys_read
    syscall
    cmp rax, 0
    je read_done
    add r8, rax
    jmp read_loop
read_done:
    mov r10, r8                 # r10 = total_len

    # ---- find first '!' ----
    xor rcx, rcx
    lea r9, [rip + input_buf]
find_bang:
    cmp rcx, r8
    jge no_bang
    movzx eax, byte ptr [r9 + rcx]
    cmp al, 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov rbx, rcx                 # program_len
    lea r11, [rcx + 1]           # input_start
    jmp set_input_start
no_bang:
    mov rbx, r8                  # program_len = total len
    mov r11, r8                  # input_start = len (no input left)
set_input_start:
    mov r9, r11                  # r9 = current input read position
    # r10 = total_len (end bound for comma reads)
    # rbx = program_len

    # ---- validate brackets ----
    xor rcx, rcx                 # i
    xor rdx, rdx                 # sp
    lea r14, [rip + input_buf]
    lea r15, [rip + bstack]
validate_loop:
    cmp rcx, rbx
    jge validate_done
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 0x5B                 # '['
    je v_lbracket
    cmp al, 0x5D                 # ']'
    je v_rbracket
    inc rcx
    jmp validate_loop
v_lbracket:
    cmp rdx, 1000
    jge validate_fail
    mov dword ptr [r15 + rdx*4], ecx
    inc rdx
    inc rcx
    jmp validate_loop
v_rbracket:
    cmp rdx, 0
    je validate_fail
    dec rdx
    mov eax, dword ptr [r15 + rdx*4]
    lea r13, [rip + match]
    mov dword ptr [r13 + rax*4], ecx
    mov dword ptr [r13 + rcx*4], eax
    inc rcx
    jmp validate_loop
validate_fail:
    mov edi, 1
    mov rax, 60
    syscall
validate_done:
    cmp rdx, 0
    jne validate_fail

    # ---- run the program ----
    lea r14, [rip + input_buf]
    lea r15, [rip + match]
    lea r12, [rip + outbuf]
    lea r13, [rip + tape]
    xor r8, r8                   # outlen
    xor rsi, rsi                 # ip

run_loop:
    cmp rsi, rbx
    jge done_exec
    movzx eax, byte ptr [r14 + rsi]
    cmp al, 0x3E                 # '>'
    je op_gt
    cmp al, 0x3C                 # '<'
    je op_lt
    cmp al, 0x2B                 # '+'
    je op_plus
    cmp al, 0x2D                 # '-'
    je op_minus
    cmp al, 0x2E                 # '.'
    je op_dot
    cmp al, 0x2C                 # ','
    je op_comma
    cmp al, 0x5B                 # '['
    je op_lbracket
    cmp al, 0x5D                 # ']'
    je op_rbracket
    inc rsi
    jmp run_loop

op_gt:
    inc r13
    inc rsi
    jmp run_loop
op_lt:
    dec r13
    inc rsi
    jmp run_loop
op_plus:
    inc byte ptr [r13]
    inc rsi
    jmp run_loop
op_minus:
    dec byte ptr [r13]
    inc rsi
    jmp run_loop
op_dot:
    mov al, byte ptr [r13]
    mov byte ptr [r12 + r8], al
    inc r8
    inc rsi
    jmp run_loop
op_comma:
    cmp r9, r10
    jge comma_zero
    movzx eax, byte ptr [r14 + r9]
    inc r9
    mov byte ptr [r13], al
    jmp comma_done
comma_zero:
    mov byte ptr [r13], 0
comma_done:
    inc rsi
    jmp run_loop
op_lbracket:
    cmp byte ptr [r13], 0
    jne lb_cont
    mov eax, dword ptr [r15 + rsi*4]
    mov rsi, rax
    inc rsi
    jmp run_loop
lb_cont:
    inc rsi
    jmp run_loop
op_rbracket:
    cmp byte ptr [r13], 0
    je rb_cont
    mov eax, dword ptr [r15 + rsi*4]
    mov rsi, rax
    inc rsi
    jmp run_loop
rb_cont:
    inc rsi
    jmp run_loop

done_exec:
    cmp r8, 0
    je no_write
    mov rdx, r8
    lea rsi, [rip + outbuf]
    mov edi, 1
    mov rax, 1
    syscall
no_write:
    xor edi, edi
    mov rax, 60
    syscall
```
