```asm
.globl _start

.section .rodata
nl:
    .byte 0x0a

.bss
    .balign 8
buf:
    .zero 32768
off:
    .zero 4000
len:
    .zero 4000

.text

// compare_lines(x0=offA, x1=lenA, x2=offB, x3=lenB) -> x0 = -1/0/1
// uses x20 as base pointer to buf (must be set by caller before call)
compare_lines:
    cmp x1, x3
    b.lo cmp_min_is_a
    mov x6, x3
    b cmp_have_min
cmp_min_is_a:
    mov x6, x1
cmp_have_min:
    mov x7, #0
cmp_byte_loop:
    cmp x7, x6
    b.ge cmp_len_check
    add x8, x0, x7
    add x9, x2, x7
    ldrb w10, [x20, x8]
    ldrb w11, [x20, x9]
    cmp w10, w11
    b.eq cmp_next
    b.lo cmp_less
    b cmp_greater
cmp_next:
    add x7, x7, #1
    b cmp_byte_loop
cmp_len_check:
    cmp x1, x3
    b.eq cmp_equal
    b.lo cmp_less
    b cmp_greater
cmp_less:
    mov x0, #-1
    ret
cmp_greater:
    mov x0, #1
    ret
cmp_equal:
    mov x0, #0
    ret

_start:
    mov x19, #0            // total bytes read
    adrp x20, buf
    add x20, x20, :lo12:buf

read_loop:
    mov x2, #32768
    sub x2, x2, x19
    cmp x2, #0
    b.eq read_done
    mov x0, #0
    add x1, x20, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0            // start of current line
    mov x22, #0            // scan index
    mov x23, #0            // line count
    adrp x24, off
    add x24, x24, :lo12:off
    adrp x25, len
    add x25, x25, :lo12:len

parse_loop:
    cmp x22, x19
    b.ge parse_end
    ldrb w0, [x20, x22]
    cmp w0, #0x0a
    b.ne parse_next
    sub x1, x22, x21
    str w21, [x24, x23, lsl #2]
    str w1, [x25, x23, lsl #2]
    add x23, x23, #1
    add x21, x22, #1
parse_next:
    add x22, x22, #1
    b parse_loop

parse_end:
    cmp x21, x19
    b.ge no_remainder
    sub x1, x19, x21
    str w21, [x24, x23, lsl #2]
    str w1, [x25, x23, lsl #2]
    add x23, x23, #1
no_remainder:

    mov x12, #1
sort_outer:
    cmp x12, x23
    b.ge sort_done
    ldr w13, [x24, x12, lsl #2]
    ldr w14, [x25, x12, lsl #2]
    sub x15, x12, #1
sort_inner:
    tbnz x15, #63, sort_insert
    ldr w16, [x24, x15, lsl #2]
    ldr w17, [x25, x15, lsl #2]
    mov x0, x16
    mov x1, x17
    mov x2, x13
    mov x3, x14
    bl compare_lines
    cmp x0, #0
    b.le sort_insert
    add x4, x15, #1
    str w16, [x24, x4, lsl #2]
    str w17, [x25, x4, lsl #2]
    sub x15, x15, #1
    b sort_inner
sort_insert:
    add x4, x15, #1
    str w13, [x24, x4, lsl #2]
    str w14, [x25, x4, lsl #2]
    add x12, x12, #1
    b sort_outer
sort_done:

    mov x12, #0
out_loop:
    cmp x12, x23
    b.ge out_done
    ldr w5, [x24, x12, lsl #2]
    ldr w6, [x25, x12, lsl #2]
    mov x0, #1
    add x1, x20, x5
    mov x2, x6
    mov x8, #64
    svc #0
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #64
    svc #0
    add x12, x12, #1
    b out_loop

out_done:
    mov x0, #0
    mov x8, #93
    svc #0
```
