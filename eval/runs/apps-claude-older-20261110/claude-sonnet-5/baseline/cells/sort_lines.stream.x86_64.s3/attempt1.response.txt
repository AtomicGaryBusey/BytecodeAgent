.section .bss
.align 16
input_buf: .zero 32768
output_buf: .zero 32800
lines: .zero 16000
tmp: .zero 16000

.text
.globl _start

_start:
    xor r12, r12
read_loop:
    xor rax, rax
    mov rdi, 0
    lea rsi, [rip+input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop
read_done:

    xor r13, r13
    xor r14, r14
    xor r15, r15
parse_loop:
    cmp r14, r12
    jge parse_tail
    lea rax, [rip+input_buf]
    movzx ecx, byte ptr [rax+r14]
    cmp cl, 0x0A
    jne parse_next
    mov rdx, r14
    sub rdx, r15
    lea rax, [rip+input_buf]
    add rax, r15
    mov r8, r13
    shl r8, 1
    lea rdi, [rip+lines]
    lea rdi, [rdi+r8*8]
    mov [rdi], rax
    mov [rdi+8], rdx
    inc r13
    lea r15, [r14+1]
parse_next:
    inc r14
    jmp parse_loop
parse_tail:
    cmp r15, r12
    jge parse_done
    mov rdx, r12
    sub rdx, r15
    lea rax, [rip+input_buf]
    add rax, r15
    mov r8, r13
    shl r8, 1
    lea rdi, [rip+lines]
    lea rdi, [rdi+r8*8]
    mov [rdi], rax
    mov [rdi+8], rdx
    inc r13
parse_done:

    cmp r13, 1
    jle skip_sort
    lea rdi, [rip+lines]
    mov rsi, r13
    call merge_sort
skip_sort:

    xor r14, r14
    xor rbx, rbx
out_loop:
    cmp rbx, r13
    jge out_done
    mov r8, rbx
    shl r8, 1
    lea rax, [rip+lines]
    lea rax, [rax+r8*8]
    mov rsi, [rax]
    mov rdx, [rax+8]
    lea rdi, [rip+output_buf]
    add rdi, r14
    xor rcx, rcx
copy_loop:
    cmp rcx, rdx
    jge copy_done
    mov r9b, byte ptr [rsi+rcx]
    mov byte ptr [rdi+rcx], r9b
    inc rcx
    jmp copy_loop
copy_done:
    add r14, rdx
    lea rdi, [rip+output_buf]
    add rdi, r14
    mov byte ptr [rdi], 0x0A
    inc r14
    inc rbx
    jmp out_loop
out_done:

    xor r15, r15
write_loop:
    cmp r15, r14
    jge write_done
    mov rdi, 1
    lea rsi, [rip+output_buf]
    add rsi, r15
    mov rdx, r14
    sub rdx, r15
    mov rax, 1
    syscall
    add r15, rax
    jmp write_loop
write_done:

    mov rdi, 0
    mov rax, 60
    syscall

merge_sort:
    push rbp
    mov rbp, rsp
    cmp rsi, 1
    jle ms_done
    mov rax, rsi
    shr rax, 1
    sub rsp, 32
    mov [rsp], rdi
    mov [rsp+8], rsi
    mov [rsp+16], rax
    mov rsi, rax
    call merge_sort
    mov rdi, [rsp]
    mov rax, [rsp+16]
    mov r8, rax
    shl r8, 1
    lea rdi, [rdi+r8*8]
    mov rsi, [rsp+8]
    sub rsi, rax
    call merge_sort
    mov rdi, [rsp]
    mov rsi, [rsp+16]
    mov rdx, [rsp+8]
    sub rdx, rsi
    call merge
    add rsp, 32
ms_done:
    mov rsp, rbp
    pop rbp
    ret

merge:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov rax, r13
    shl rax, 1
    lea r15, [r12+rax*8]
    xor rbx, rbx
    xor rbp, rbp
    xor r11, r11
mg_l1:
    cmp rbp, r13
    jge mg_l1done
    cmp r11, r14
    jge mg_l1done
    mov rax, rbp
    shl rax, 1
    lea rax, [r12+rax*8]
    mov rdi, [rax]
    mov rsi, [rax+8]
    mov rax, r11
    shl rax, 1
    lea rax, [r15+rax*8]
    mov rdx, [rax]
    mov rcx, [rax+8]
    call cmp_lines
    cmp rax, 0
    jg mg_take_right
    mov rax, rbp
    shl rax, 1
    lea rax, [r12+rax*8]
    mov r8, [rax]
    mov r9, [rax+8]
    mov rax, rbx
    shl rax, 1
    lea rdi, [rip+tmp]
    lea rdi, [rdi+rax*8]
    mov [rdi], r8
    mov [rdi+8], r9
    inc rbp
    jmp mg_inc
mg_take_right:
    mov rax, r11
    shl rax, 1
    lea rax, [r15+rax*8]
    mov r8, [rax]
    mov r9, [rax+8]
    mov rax, rbx
    shl rax, 1
    lea rdi, [rip+tmp]
    lea rdi, [rdi+rax*8]
    mov [rdi], r8
    mov [rdi+8], r9
    inc r11
mg_inc:
    inc rbx
    jmp mg_l1
mg_l1done:
mg_l2:
    cmp rbp, r13
    jge mg_l2done
    mov rax, rbp
    shl rax, 1
    lea rax, [r12+rax*8]
    mov r8, [rax]
    mov r9, [rax+8]
    mov rax, rbx

mg_l2done:
mg_l3:

    lea rax, [r15+rax*8]
    mov r8, [rax]
    mov r9, [rax+8]
    mov rax, rbx

    lea rdi, [rip+tmp]
    lea rdi, [rdi+rax*8]
    mov [rdi], r8
    mov [rdi+8], r9
    inc r11
    inc rbx
    jmp mg_l3
mg_l3done:
    xor rbp, rbp
mg_l4:
    cmp rbp, rbx

    mov r10, rbp

    lea rax, [rip+tmp]
    lea rax, [rax+r10*8]

    mov r9, [rax+8]
    lea rdi, [r12+r10*8]

    jmp mg_l4
mg_l4done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx

cmp_lines:
    mov r8, rsi
    cmp r8, rcx
    jle cl_have_min
    mov r8, rcx
cl_have_min:
    xor r9, r9
cl_loop:
    cmp r9, r8
    jge cl_tail
    movzx eax, byte ptr [rdi+r9]
    movzx r10d, byte ptr [rdx+r9]
    cmp eax, r10d
    jne cl_diff
    inc r9
    jmp cl_loop
cl_diff:
    sub eax, r10d
    movsxd rax, eax
    ret
cl_tail:
    mov rax, rsi
    sub rax, rcx
    ret
