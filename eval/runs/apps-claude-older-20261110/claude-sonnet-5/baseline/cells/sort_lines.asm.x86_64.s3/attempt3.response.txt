.equ MAX_INPUT, 32768

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, MAX_INPUT
    sub rdx, r12
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:

    xor r13, r13
    xor rbx, rbx
    xor r14, r14

parse_loop:
    cmp rbx, r12
    jge parse_after
    lea rsi, [rip + input_buf]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 0x0A
    jne parse_next
    mov rcx, rbx
    sub rcx, r14
    lea rdi, [rip + line_off]
    mov qword ptr [rdi + r13*8], r14
    lea rdi, [rip + line_len]
    mov qword ptr [rdi + r13*8], rcx
    inc r13
    lea r14, [rbx + 1]
parse_next:
    inc rbx
    jmp parse_loop

parse_after:
    cmp r14, r12
    jge sort_start
    mov rcx, r12
    sub rcx, r14
    lea rdi, [rip + line_off]
    mov qword ptr [rdi + r13*8], r14
    lea rdi, [rip + line_len]
    mov qword ptr [rdi + r13*8], rcx
    inc r13

sort_start:
    cmp r13, 1
    jle sort_done

    mov r15, 1
insertion_outer:
    cmp r15, r13
    jge sort_done
    lea rdi, [rip + line_off]
    mov rax, qword ptr [rdi + r15*8]
    lea rdi, [rip + line_len]
    mov rdx, qword ptr [rdi + r15*8]
    mov rbx, r15
    dec rbx

insertion_inner:
    cmp rbx, 0
    jl insertion_place
    lea rdi, [rip + line_off]
    mov r8, qword ptr [rdi + rbx*8]
    lea rdi, [rip + line_len]
    mov r9, qword ptr [rdi + rbx*8]
    call do_compare
    cmp r11, 0
    jle insertion_place
    lea rdi, [rip + line_off]
    mov qword ptr [rdi + rbx*8 + 8], r8
    lea rdi, [rip + line_len]
    mov qword ptr [rdi + rbx*8 + 8], r9
    dec rbx
    jmp insertion_inner

insertion_place:
    inc rbx
    lea rdi, [rip + line_off]
    mov qword ptr [rdi + rbx*8], rax
    lea rdi, [rip + line_len]
    mov qword ptr [rdi + rbx*8], rdx
    inc r15
    jmp insertion_outer

sort_done:

    xor r15, r15
    xor rbx, rbx
output_loop:
    cmp r15, r13
    jge output_done_building
    lea rdi, [rip + line_off]
    mov rax, qword ptr [rdi + r15*8]
    lea rdi, [rip + line_len]
    mov rdx, qword ptr [rdi + r15*8]
    lea rsi, [rip + input_buf]
    add rsi, rax
    lea rdi, [rip + output_buf]
    add rdi, rbx
    mov rcx, rdx
    test rcx, rcx
    jz skip_copy
copy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz copy_loop
skip_copy:
    add rbx, rdx
    lea rdi, [rip + output_buf]
    mov byte ptr [rdi + rbx], 0x0A
    inc rbx
    inc r15
    jmp output_loop

output_done_building:
    xor r12, r12
write_loop:
    cmp r12, rbx
    jge exit_prog
    lea rsi, [rip + output_buf]
    add rsi, r12
    mov rdx, rbx
    sub rdx, r12
    mov rdi, 1
    mov rax, 1
    syscall
    add r12, rax
    jmp write_loop

exit_prog:
    mov rdi, 0
    mov rax, 60
    syscall

do_compare:
    mov rbp, r9
    cmp rbp, rdx
    jle cmp_minset
    mov rbp, rdx
cmp_minset:
    xor r10, r10
    lea rsi, [rip + input_buf]
    add rsi, r8
    lea rdi, [rip + input_buf]
    add rdi, rax
cmp_loop:
    cmp r10, rbp
    jge cmp_lentail
    movzx r11d, byte ptr [rsi + r10]
    movzx ecx, byte ptr [rdi + r10]
    cmp r11d, ecx
    jne cmp_differ
    inc r10
    jmp cmp_loop
cmp_lentail:
    cmp r9, rdx
    je cmp_equal
    jl cmp_less
    jmp cmp_greater
cmp_differ:
    jb cmp_less
    jmp cmp_greater
cmp_less:
    mov r11, -1
    ret
cmp_equal:
    xor r11, r11
    ret
cmp_greater:
    mov r11, 1
    ret

.bss
.align 8
input_buf:
    .zero 32768
line_off:
    .zero 8000
line_len:
    .zero 8000
output_buf:
    .zero 40000
