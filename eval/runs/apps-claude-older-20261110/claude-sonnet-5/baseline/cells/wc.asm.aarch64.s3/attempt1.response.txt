// int_to_str: x0 = value (unsigned), x1 = dest pointer (in/out)
// on return, x1 points just past the last digit written.
int_to_str:
    sub sp, sp, #32
    mov x10, sp
    mov x11, x10
    cbz x0, .Lzero_case
    mov x13, #10
.Lloop_digits:
    udiv x14, x0, x13
    msub x15, x14, x13, x0
    add x15, x15, #48
    strb w15, [x11], #1
    mov x0, x14
    cbnz x0, .Lloop_digits
    b .Lwrite_digits
.Lzero_case:
    mov w15, #48
    strb w15, [x11], #1
.Lwrite_digits:
    sub x11, x11, #1
.Lrev_loop:
    ldrb w15, [x11]
    strb w15, [x1], #1
    cmp x11, x10
    beq .Lrev_done
    sub x11, x11, #1
    b .Lrev_loop
.Lrev_done:
    add sp, sp, #32
    ret

_start:
    adrp x20, buf
    add x20, x20, :lo12:buf
    mov x19, #0

.Lread_loop:
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    cmp x2, #0
    beq .Lprocess
    mov x8, #63
    svc #0
    cmp x0, #0
    beq .Lprocess
    add x19, x19, x0
    b .Lread_loop

.Lprocess:
    mov x21, #0
    mov x22, #0
    mov x23, #0
    mov x24, #0

.Lscan_loop:
    cmp x24, x19
    bge .Lscan_done
    ldrb w3, [x20, x24]
    cmp w3, #0x20
    beq .Lws
    cmp w3, #0x09
    beq .Lws
    cmp w3, #0x0A
    beq .Lnl
    cmp w3, #0x0B
    beq .Lws
    cmp w3, #0x0C
    beq .Lws
    cmp w3, #0x0D
    beq .Lws
    cbnz x23, .Lcont_scan
    add x22, x22, #1
    mov x23, #1
    b .Lcont_scan
.Lnl:
    add x21, x21, #1
    mov x23, #0
    b .Lcont_scan
.Lws:
    mov x23, #0
.Lcont_scan:
    add x24, x24, #1
    b .Lscan_loop

.Lscan_done:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x1, x25

    mov x0, x21
    bl int_to_str
    mov w3, #0x20
    strb w3, [x1], #1

    mov x0, x22
    bl int_to_str
    mov w3, #0x20
    strb w3, [x1], #1

    mov x0, x19
    bl int_to_str
    mov w3, #0x0A
    strb w3, [x1], #1

    sub x2, x1, x25
    mov x0, #1
    mov x1, x25
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.bss
    .align 4
buf:
    .zero 65536
outbuf:
    .zero 64
