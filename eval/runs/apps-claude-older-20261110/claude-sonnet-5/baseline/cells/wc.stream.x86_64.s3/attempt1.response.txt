```
.globl _start

.section .rodata

.bss
    .align 16
buffer:
    .space 65536
outbuf:
    .space 64

.text

itoa:
    sub rsp, 32
    mov r9, 10
    xor rcx, rcx
conv_loop:
    xor rdx, rdx
    div r9
    mov byte ptr [rsp + rcx], dl
    inc rcx
    test rax, rax
    jnz conv_loop
conv_write:
    dec rcx
    movzx edx, byte ptr [rsp + rcx]
    add dl, '0'
    mov byte ptr [r10], dl
    inc r10
    test rcx, rcx
    jnz conv_write
    add rsp, 32
    ret

_start:
    xor r14, r14

read_loop:
    xor rdi, rdi
    lea rsi, [rip + buffer]
    add rsi, r14
    mov rdx, 65536
    sub rdx, r14
    xor rax, rax
    syscall
    test rax, rax
    jz process
    add r14, rax
    jmp read_loop

process:
    lea rbx, [rip + buffer]
    xor rcx, rcx
    xor r12, r12
    xor r13, r13
    xor r15, r15

count_loop:
    cmp rcx, r14
    jge count_done
    movzx eax, byte ptr [rbx + rcx]

    cmp al, 0x0A
    jne not_newline
    inc r12
not_newline:

    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws

    test r15, r15
    jnz cont_char
    inc r13
    mov r15, 1
    jmp cont_char

is_ws:
    xor r15, r15

cont_char:
    inc rcx
    jmp count_loop

count_done:
    lea r8, [rip + outbuf]
    mov r10, r8

    mov rax, r12
    call itoa
    mov byte ptr [r10], ' '
    inc r10

    mov rax, r13
    call itoa
    mov byte ptr [r10], ' '
    inc r10

    mov rax, r14
    call itoa
    mov byte ptr [r10], 0x0A
    inc r10

    mov rdx, r10
    sub rdx, r8

    mov rdi, 1
    mov rsi, r8
    mov rax, 1
    syscall

    mov rdi, 0
    mov rax, 231
    syscall
```
