.text
.globl _start
_start:
    mov x19, #0            // total_len

read_loop:
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf
    mov x21, x20            // output cursor

    mov x22, #0             // current row offset

row_loop:
    cmp x22, x19
    b.ge after_rows

    sub x23, x19, x22
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, lt  // x23 = n (bytes in this row)

    // print offset (x22) as 8 hex digits
    mov x25, x22
    mov x26, #28
print_off_loop:
    lsr x27, x25, x26
    and x27, x27, #0xf
    adrp x28, hex_digits
    add x28, x28, :lo12:hex_digits
    ldrb w0, [x28, x27]
    strb w0, [x21], #1
    subs x26, x26, #4
    b.ge print_off_loop

    mov w0, #' '
    strb w0, [x21], #1
    strb w0, [x21], #1

    mov x9, #0
byte_loop:
    cmp x9, x23
    b.lt have_byte
    mov w0, #' '
    strb w0, [x21], #1
    strb w0, [x21], #1
    strb w0, [x21], #1
    b after_byte
have_byte:
    adrp x10, input_buf
    add x10, x10, :lo12:input_buf
    add x10, x10, x22
    add x10, x10, x9
    ldrb w11, [x10]
    lsr w12, w11, #4
    and w13, w11, #0xf
    adrp x28, hex_digits
    add x28, x28, :lo12:hex_digits
    ldrb w0, [x28, x12]
    strb w0, [x21], #1
    ldrb w0, [x28, x13]
    strb w0, [x21], #1
    mov w0, #' '
    strb w0, [x21], #1
after_byte:
    cmp x9, #7
    b.ne skip_extra
    mov w0, #' '
    strb w0, [x21], #1
skip_extra:
    add x9, x9, #1
    cmp x9, #16
    b.lt byte_loop

    mov w0, #' '
    strb w0, [x21], #1
    mov w0, #'|'
    strb w0, [x21], #1

    mov x9, #0
ascii_loop:
    cmp x9, x23
    b.ge ascii_done
    adrp x10, input_buf
    add x10, x10, :lo12:input_buf
    add x10, x10, x22
    add x10, x10, x9
    ldrb w11, [x10]
    cmp w11, #0x20
    b.lt dot
    cmp w11, #0x7e
    b.gt dot
    b store_char
dot:
    mov w11, #'.'
store_char:
    strb w11, [x21], #1
    add x9, x9, #1
    b ascii_loop
ascii_done:
    mov w0, #'|'
    strb w0, [x21], #1
    mov w0, #'\n'
    strb w0, [x21], #1

    add x22, x22, x23
    b row_loop

after_rows:
    mov x25, x19
    mov x26, #28
final_off_loop:
    lsr x27, x25, x26
    and x27, x27, #0xf
    adrp x28, hex_digits
    add x28, x28, :lo12:hex_digits
    ldrb w0, [x28, x27]
    strb w0, [x21], #1
    subs x26, x26, #4
    b.ge final_off_loop
    mov w0, #'\n'
    strb w0, [x21], #1

    sub x2, x21, x20
    mov x1, x20
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .space 65536
output_buf:
    .space 400000
