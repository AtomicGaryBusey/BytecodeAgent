.globl _start

_start:
    mov     x20, #0                 // total bytes read so far
read_loop:
    adrp    x9, buf
    add     x9, x9, :lo12:buf
    add     x9, x9, x20
    mov     x10, #65536
    sub     x10, x10, x20
    cmp     x10, #0
    b.le    read_done
    mov     x0, #0
    mov     x1, x9
    mov     x2, x10
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    adrp    x9, g_len
    add     x9, x9, :lo12:g_len
    str     x20, [x9]
    adrp    x9, g_pos
    add     x9, x9, :lo12:g_pos
    str     xzr, [x9]

    bl      skip_ws
    bl      parse_value
    cbz     x0, do_error
    bl      skip_ws
    bl      get_pos
    mov     x21, x0
    bl      get_len
    cmp     x21, x0
    b.ne    do_error

    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #3
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0

do_error:
    bl      get_pos
    mov     x21, x0

    adrp    x1, msg_errprefix
    add     x1, x1, :lo12:msg_errprefix
    mov     x2, #9
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, x21
    bl      itoa
    mov     x2, x0
    adrp    x1, digitbuf
    add     x1, x1, :lo12:digitbuf
    mov     x0, #1
    mov     x8, #64
    svc     #0

    adrp    x1, msg_nl
    add     x1, x1, :lo12:msg_nl
    mov     x2, #1
    mov     x0, #1
    mov     x8, #64
    svc     #0

    mov     x0, #1
    mov     x8, #93
    svc     #0

// ---------------------------------------------------------------
// leaf helper functions
// ---------------------------------------------------------------

get_pos:
    adrp    x9, g_pos
    add     x9, x9, :lo12:g_pos
    ldr     x0, [x9]
    ret

get_len:
    adrp    x9, g_len
    add     x9, x9, :lo12:g_len
    ldr     x0, [x9]
    ret

adv:
    adrp    x9, g_pos
    add     x9, x9, :lo12:g_pos
    ldr     x0, [x9]
    add     x0, x0, #1
    str     x0, [x9]
    ret

peek_byte:
    adrp    x9, g_pos
    add     x9, x9, :lo12:g_pos
    ldr     x1, [x9]
    adrp    x10, g_len
    add     x10, x10, :lo12:g_len
    ldr     x2, [x10]
    cmp     x1, x2
    b.ge    .Lpb_eof
    adrp    x11, buf
    add     x11, x11, :lo12:buf
    ldrb    w0, [x11, x1]
    ret
.Lpb_eof:
    mov     x0, #-1
    ret

is_hex_digit:
    cmp     x0, #0x30
    b.lt    .Lhd_no
    cmp     x0, #0x39
    b.le    .Lhd_yes
    cmp     x0, #0x41
    b.lt    .Lhd_no
    cmp     x0, #0x46
    b.le    .Lhd_yes
    cmp     x0, #0x61
    b.lt    .Lhd_no
    cmp     x0, #0x66
    b.le    .Lhd_yes
.Lhd_no:
    mov     x0, #0
    ret
.Lhd_yes:
    mov     x0, #1
    ret

itoa:
    adrp    x9, digitbuf
    add     x9, x9, :lo12:digitbuf
    mov     x1, x0
    cbnz    x1, .Lit_nonzero
    mov     w2, #0x30
    strb    w2, [x9]
    mov     x0, #1
    ret
.Lit_nonzero:
    mov     x2, #0
    sub     sp, sp, #16
.Lit_loop:
    mov     x3, #10
    udiv    x4, x1, x3
    msub    x5, x4, x3, x1
    add     w5, w5, #0x30
    strb    w5, [sp, x2]
    add     x2, x2, #1
    mov     x1, x4
    cbnz    x1, .Lit_loop
    mov     x6, #0
.Lit_rev:
    sub     x7, x2, x6
    sub     x7, x7, #1
    ldrb    w8, [sp, x7]
    strb    w8, [x9, x6]
    add     x6, x6, #1
    cmp     x6, x2
    b.lt    .Lit_rev
    mov     x0, x2
    add     sp, sp, #16
    ret

// ---------------------------------------------------------------
// recursive-descent parser (non-leaf functions save/restore x30)
// ---------------------------------------------------------------

parse_value:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    mov     x1, x0
    cmp     x1, #0x7B
    b.eq    .Lpv_object
    cmp     x1, #0x5B
    b.eq    .Lpv_array
    cmp     x1, #0x22
    b.eq    .Lpv_string
    cmp     x1, #0x74
    b.eq    .Lpv_true
    cmp     x1, #0x66
    b.eq    .Lpv_false
    cmp     x1, #0x6E
    b.eq    .Lpv_null
    cmp     x1, #0x2D
    b.eq    .Lpv_number
    cmp     x1, #0x30
    b.lt    .Lpv_fail
    cmp     x1, #0x39
    b.gt    .Lpv_fail
    b       .Lpv_number
.Lpv_object:
    bl      parse_object
    b       .Lpv_ret
.Lpv_array:
    bl      parse_array
    b       .Lpv_ret
.Lpv_string:
    bl      parse_string
    b       .Lpv_ret
.Lpv_true:
    bl      match_true
    b       .Lpv_ret
.Lpv_false:
    bl      match_false
    b       .Lpv_ret
.Lpv_null:
    bl      match_null
    b       .Lpv_ret
.Lpv_number:
    bl      parse_number
    b       .Lpv_ret
.Lpv_fail:
    mov     x0, #0
.Lpv_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

skip_ws:
    adrp    x9, buf
    add     x9, x9, :lo12:buf
.Lsw_loop:
    adrp    x10, g_pos
    add     x10, x10, :lo12:g_pos
    ldr     x11, [x10]
    adrp    x12, g_len
    add     x12, x12, :lo12:g_len
    ldr     x13, [x12]
    cmp     x11, x13
    b.ge    .Lsw_done
    ldrb    w14, [x9, x11]
    cmp     w14, #0x20
    b.eq    .Lsw_adv
    cmp     w14, #0x09
    b.eq    .Lsw_adv
    cmp     w14, #0x0A
    b.eq    .Lsw_adv
    cmp     w14, #0x0D
    b.eq    .Lsw_adv
    b       .Lsw_done
.Lsw_adv:
    add     x11, x11, #1
    str     x11, [x10]
    b       .Lsw_loop
.Lsw_done:
    ret

parse_object:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      adv
    bl      skip_ws
    bl      peek_byte
    cmp     x0, #0x7D
    b.ne    .Lpo_members
    bl      adv
    mov     x0, #1
    b       .Lpo_ret
.Lpo_members:
.Lpo_loop:
    bl      parse_member
    cbz     x0, .Lpo_fail
    bl      skip_ws
    bl      peek_byte
    cmp     x0, #0x7D
    b.eq    .Lpo_close
    cmp     x0, #0x2C
    b.eq    .Lpo_comma
    b       .Lpo_fail
.Lpo_comma:
    bl      adv
    bl      skip_ws
    b       .Lpo_loop
.Lpo_close:
    bl      adv
    mov     x0, #1
    b       .Lpo_ret
.Lpo_fail:
    mov     x0, #0
.Lpo_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

parse_member:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    cmp     x0, #0x22
    b.ne    .Lpm_fail
    bl      parse_string
    cbz     x0, .Lpm_fail
    bl      skip_ws
    bl      peek_byte
    cmp     x0, #0x3A
    b.ne    .Lpm_fail
    bl      adv
    bl      skip_ws
    bl      parse_value
    b       .Lpm_ret
.Lpm_fail:
    mov     x0, #0
.Lpm_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

parse_array:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      adv
    bl      skip_ws
    bl      peek_byte
    cmp     x0, #0x5D
    b.ne    .Lpa_elems
    bl      adv
    mov     x0, #1
    b       .Lpa_ret
.Lpa_elems:
.Lpa_loop:
    bl      parse_value
    cbz     x0, .Lpa_fail
    bl      skip_ws
    bl      peek_byte
    cmp     x0, #0x5D
    b.eq    .Lpa_close
    cmp     x0, #0x2C
    b.eq    .Lpa_comma
    b       .Lpa_fail
.Lpa_comma:
    bl      adv
    bl      skip_ws
    b       .Lpa_loop
.Lpa_close:
    bl      adv
    mov     x0, #1
    b       .Lpa_ret
.Lpa_fail:
    mov     x0, #0
.Lpa_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

parse_string:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      adv
.Lps_loop:
    bl      peek_byte
    cmp     x0, #0x22
    b.eq    .Lps_close
    cmp     x0, #0x5C
    b.eq    .Lps_escape
    cmp     x0, #0x20
    b.lt    .Lps_fail
    bl      adv
    b       .Lps_loop
.Lps_escape:
    bl      adv
    bl      peek_byte
    cmp     x0, #0x22
    b.eq    .Lps_esc_simple
    cmp     x0, #0x5C
    b.eq    .Lps_esc_simple
    cmp     x0, #0x2F
    b.eq    .Lps_esc_simple
    cmp     x0, #0x62
    b.eq    .Lps_esc_simple
    cmp     x0, #0x66
    b.eq    .Lps_esc_simple
    cmp     x0, #0x6E
    b.eq    .Lps_esc_simple
    cmp     x0, #0x72
    b.eq    .Lps_esc_simple
    cmp     x0, #0x74
    b.eq    .Lps_esc_simple
    cmp     x0, #0x75
    b.eq    .Lps_esc_u
    b       .Lps_fail
.Lps_esc_simple:
    bl      adv
    b       .Lps_loop
.Lps_esc_u:
    bl      adv
    bl      chk_hex
    cbz     x0, .Lps_fail
    bl      chk_hex
    cbz     x0, .Lps_fail
    bl      chk_hex
    cbz     x0, .Lps_fail
    bl      chk_hex
    cbz     x0, .Lps_fail
    b       .Lps_loop
.Lps_close:
    bl      adv
    mov     x0, #1
    b       .Lps_ret
.Lps_fail:
    mov     x0, #0
.Lps_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

chk_hex:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    bl      is_hex_digit
    cbz     x0, .Lch_fail
    bl      adv
    mov     x0, #1
    b       .Lch_ret
.Lch_fail:
    mov     x0, #0
.Lch_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

parse_number:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    cmp     x0, #0x2D
    b.ne    .Lpn_int
    bl      adv
.Lpn_int:
    bl      peek_byte
    cmp     x0, #0x30
    b.lt    .Lpn_fail
    cmp     x0, #0x39
    b.gt    .Lpn_fail
    cmp     x0, #0x30
    b.eq    .Lpn_int_zero
    bl      adv
.Lpn_int_rest:
    bl      peek_byte
    cmp     x0, #0x30
    b.lt    .Lpn_frac
    cmp     x0, #0x39
    b.gt    .Lpn_frac
    bl      adv
    b       .Lpn_int_rest
.Lpn_int_zero:
    bl      adv
    b       .Lpn_frac
.Lpn_frac:
    bl      peek_byte
    cmp     x0, #0x2E
    b.ne    .Lpn_exp
    bl      adv
    bl      peek_byte
    cmp     x0, #0x30
    b.lt    .Lpn_fail
    cmp     x0, #0x39
    b.gt    .Lpn_fail
    bl      adv
.Lpn_frac_rest:
    bl      peek_byte
    cmp     x0, #0x30
    b.lt    .Lpn_exp
    cmp     x0, #0x39
    b.gt    .Lpn_exp
    bl      adv
    b       .Lpn_frac_rest
.Lpn_exp:
    bl      peek_byte
    cmp     x0, #0x65
    b.eq    .Lpn_exp_go
    cmp     x0, #0x45
    b.eq    .Lpn_exp_go
    b       .Lpn_success
.Lpn_exp_go:
    bl      adv
    bl      peek_byte
    cmp     x0, #0x2B
    b.eq    .Lpn_exp_sign
    cmp     x0, #0x2D
    b.eq    .Lpn_exp_sign
    b       .Lpn_exp_digit1
.Lpn_exp_sign:
    bl      adv
.Lpn_exp_digit1:
    bl      peek_byte
    cmp     x0, #0x30
    b.lt    .Lpn_fail
    cmp     x0, #0x39
    b.gt    .Lpn_fail
    bl      adv
.Lpn_exp_rest:
    bl      peek_byte
    cmp     x0, #0x30
    b.lt    .Lpn_success
    cmp     x0, #0x39
    b.gt    .Lpn_success
    bl      adv
    b       .Lpn_exp_rest
.Lpn_success:
    mov     x0, #1
    b       .Lpn_ret
.Lpn_fail:
    mov     x0, #0
.Lpn_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

match_true:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    cmp     x0, #0x74
    b.ne    .Lmt_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x72
    b.ne    .Lmt_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x75
    b.ne    .Lmt_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x65
    b.ne    .Lmt_fail
    bl      adv
    mov     x0, #1
    b       .Lmt_ret
.Lmt_fail:
    mov     x0, #0
.Lmt_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

match_false:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    cmp     x0, #0x66
    b.ne    .Lmf_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x61
    b.ne    .Lmf_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x6C
    b.ne    .Lmf_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x73
    b.ne    .Lmf_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x65
    b.ne    .Lmf_fail
    bl      adv
    mov     x0, #1
    b       .Lmf_ret
.Lmf_fail:
    mov     x0, #0
.Lmf_ret:
    ldr     x30, [sp]
    add     sp, sp, #16
    ret

match_null:
    sub     sp, sp, #16
    str     x30, [sp]
    bl      peek_byte
    cmp     x0, #0x6E
    b.ne    .Lmn_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x75
    b.ne    .Lmn_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x6C
    b.ne    .Lmn_fail
    bl      adv
    bl      peek_byte
    cmp     x0, #0x6C
    b.ne    .Lmn_fail
