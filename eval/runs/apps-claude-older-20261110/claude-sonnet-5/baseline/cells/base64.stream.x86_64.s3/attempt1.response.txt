```asm
.globl _start

.section .rodata
b64tab:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000

.text
_start:
    xor r12, r12            # total bytes read so far

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 49152
    sub rdx, r12
    cmp rdx, 0
    je done_read
    xor rax, rax            # sys_read
    xor rdi, rdi            # fd 0
    syscall
    cmp rax, 0
    jle done_read
    add r12, rax
    jmp read_loop

done_read:
    xor rbx, rbx            # input offset
    xor r13, r13            # output offset
    xor r14, r14            # current line length

main_loop:
    mov rax, r12
    sub rax, rbx
    cmp rax, 3
    jb tail

    lea rsi, [rip + inbuf]
    add rsi, rbx
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    shl eax, 16
    shl ecx, 8
    or eax, ecx
    or eax, edx

    mov r8d, eax
    shr r8d, 18
    and r8d, 0x3F
    mov r9d, eax
    shr r9d, 12
    and r9d, 0x3F
    mov r10d, eax
    shr r10d, 6
    and r10d, 0x3F
    mov r11d, eax
    and r11d, 0x3F

    lea rdx, [rip + b64tab]
    movzx r8d, byte ptr [rdx + r8]
    movzx r9d, byte ptr [rdx + r9]
    movzx r10d, byte ptr [rdx + r10]
    movzx r11d, byte ptr [rdx + r11]

    lea rdi, [rip + outbuf]
    add rdi, r13
    mov byte ptr [rdi], r8b
    mov byte ptr [rdi + 1], r9b
    mov byte ptr [rdi + 2], r10b
    mov byte ptr [rdi + 3], r11b

    add r13, 4
    add r14, 4
    add rbx, 3

    cmp r14, 76
    jne main_loop

    lea rdi, [rip + outbuf]
    add rdi, r13
    mov byte ptr [rdi], 10
    inc r13
    xor r14, r14
    jmp main_loop

tail:
    cmp rax, 0
    je finish
    cmp rax, 1
    je tail1
    jmp tail2

tail1:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    movzx eax, byte ptr [rsi]
    mov ecx, eax
    shr ecx, 2
    and ecx, 0x3F
    mov edx, eax
    shl edx, 4
    and edx, 0x3F

    lea r8, [rip + b64tab]
    movzx r9d, byte ptr [r8 + rcx]
    movzx r10d, byte ptr [r8 + rdx]

    lea rdi, [rip + outbuf]
    add rdi, r13
    mov byte ptr [rdi], r9b
    mov byte ptr [rdi + 1], r10b
    mov byte ptr [rdi + 2], 0x3D
    mov byte ptr [rdi + 3], 0x3D

    add r13, 4
    add r14, 4
    jmp finish

tail2:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    movzx eax, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    shl eax, 8
    or eax, ecx

    mov ecx, eax
    shr ecx, 10
    and ecx, 0x3F
    mov edx, eax
    shr edx, 4
    and edx, 0x3F
    mov r8d, eax
    and r8d, 0xF
    shl r8d, 2

    lea r9, [rip + b64tab]
    movzx r10d, byte ptr [r9 + rcx]
    movzx r11d, byte ptr [r9 + rdx]
    movzx eax, byte ptr [r9 + r8]

    lea rdi, [rip + outbuf]
    add rdi, r13
    mov byte ptr [rdi], r10b
    mov byte ptr [rdi + 1], r11b
    mov byte ptr [rdi + 2], al
    mov byte ptr [rdi + 3], 0x3D

    add r13, 4
    add r14, 4
    jmp finish

finish:
    cmp r14, 0
    je write_output
    lea rdi, [rip + outbuf]
    add rdi, r13
    mov byte ptr [rdi], 10
    inc r13

write_output:
    cmp r13, 0
    je exit_prog
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + outbuf]
    mov rdx, r13
    syscall

exit_prog:
    mov rax, 60
    xor rdi, rdi
    syscall
```
