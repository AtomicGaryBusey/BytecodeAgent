.section .rodata

.bss
buffer:     .zero 65536
digits_tmp: .zero 24
outbuf:     .zero 64

.text
.globl _start

_start:
    mov rbx, 0          # total bytes read

read_loop:
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    cmp rdx, 0
    je done_read
    mov rdi, 0
    mov rax, 0
    mov rdi, rsi
    mov rsi, rdi
    jmp do_read

do_read:
    # fix up registers properly for syscall
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    mov rdi, 0
    mov rax, 0
    syscall
    cmp rax, 0
    je done_read
    add rbx, rax
    jmp read_loop

done_read:
    xor rcx, rcx        # index
    xor r8, r8          # lines
    xor r9, r9          # words
    xor r10, r10        # in_word flag
    lea rsi, [rip + buffer]

scan_loop:
    cmp rcx, rbx
    jae scan_end
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x0A
    jne not_newline
    inc r8
not_newline:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0A
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    cmp r10, 0
    jne in_word_already
    mov r10, 1
    inc r9
in_word_already:
    jmp scan_next
is_ws:
    mov r10, 0
scan_next:
    inc rcx
    jmp scan_loop
scan_end:

    lea r15, [rip + outbuf]
    mov rdi, r15

    mov rax, r8
    call write_num
    mov byte ptr [rdi], 0x20
    inc rdi

    mov rax, r9
    call write_num
    mov byte ptr [rdi], 0x20
    inc rdi

    mov rax, rbx
    call write_num
    mov byte ptr [rdi], 0x0A
    inc rdi

    mov rdx, rdi
    sub rdx, r15
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    syscall

    mov rax, 60
    mov rdi, 0
    syscall

write_num:
    push rcx
    push rdx
    push rsi
    push r8
    push r9
    test rax, rax
    jnz wn_nonzero
    mov byte ptr [rdi], 0x30
    inc rdi
    jmp wn_done
wn_nonzero:
    lea rsi, [rip + digits_tmp]
    xor rcx, rcx
    mov r9, 10
wn_loop:
    xor rdx, rdx
    div r9
    add dl, 0x30
    mov byte ptr [rsi + rcx], dl
    inc rcx
    test rax, rax
    jnz wn_loop
    mov rax, rcx
wn_copy:
    dec rax
    movzx edx, byte ptr [rsi + rax]
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz wn_copy
wn_done:
    pop r9
    pop r8
    pop rsi
    pop rdx
    pop rcx
    ret
