.globl _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
    .align 3
input_buf:
    .zero 65536
output_buf:
    .zero 330000

.text

_start:
    mov x19, #0
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf

read_loop:
    cmp x19, #65536
    b.ge read_done
    mov x0, #0
    add x1, x21, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x9, x26

    adrp x25, hex_table
    add x25, x25, :lo12:hex_table

    mov x23, #0

row_loop:
    cmp x23, x19
    b.ge after_rows

    sub x24, x19, x23
    cmp x24, #16
    b.lt row_n_small
    mov x24, #16
row_n_small:

    mov w0, w23
    bl write_offset_hex

    mov w1, #' '
    strb w1, [x9], #1
    strb w1, [x9], #1

    mov x10, #0

hexbyte_loop:
    cmp x10, #16
    b.ge after_hex
    cmp x10, x24
    b.lt do_hex
    b do_pad

do_hex:
    add x11, x21, x23
    add x11, x11, x10
    ldrb w12, [x11]
    lsr w13, w12, #4
    and x13, x13, #0xf
    ldrb w14, [x25, x13]
    strb w14, [x9], #1
    and w13, w12, #0xf
    and x13, x13, #0xf
    ldrb w14, [x25, x13]
    strb w14, [x9], #1
    mov w1, #' '
    strb w1, [x9], #1
    b hexbyte_after

do_pad:
    mov w1, #' '
    strb w1, [x9], #1
    strb w1, [x9], #1
    strb w1, [x9], #1

hexbyte_after:
    cmp x10, #7
    b.ne hexbyte_next
    mov w1, #' '
    strb w1, [x9], #1
hexbyte_next:
    add x10, x10, #1
    b hexbyte_loop

after_hex:
    mov w1, #' '
    strb w1, [x9], #1
    mov w1, #'|'
    strb w1, [x9], #1

    mov x10, #0

ascii_loop:
    cmp x10, x24
    b.ge ascii_done
    add x11, x21, x23
    add x11, x11, x10
    ldrb w12, [x11]
    cmp w12, #0x20
    b.lt ascii_dot
    cmp w12, #0x7e
    b.gt ascii_dot
    strb w12, [x9], #1
    b ascii_next
ascii_dot:
    mov w1, #'.'
    strb w1, [x9], #1
ascii_next:
    add x10, x10, #1
    b ascii_loop

ascii_done:
    mov w1, #'|'
    strb w1, [x9], #1
    mov w1, #10
    strb w1, [x9], #1

    add x23, x23, #16
    b row_loop

after_rows:
    mov w0, w19
    bl write_offset_hex
    mov w1, #10
    strb w1, [x9], #1

    sub x2, x9, x26
    mov x1, x26
    mov x0, #1

write_loop:
    mov x8, #64
    svc #0
    cmp x0, x2
    b.ge exit_prog
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

exit_prog:
    mov x8, #93
    mov x0, #0
    svc #0
exit_spin:
    b exit_spin

write_offset_hex:
    mov w2, w0
    mov w3, #28
wo_loop:
    lsr w4, w2, w3
    and x4, x4, #0xf
    ldrb w5, [x25, x4]
    strb w5, [x9], #1
    subs w3, w3, #4
    b.ge wo_loop
    ret
