.globl _start

.bss
.balign 8
inbuf:
    .space 65536
outbuf:
    .space 65537

.text

reverse_segment:
    // x23 = start (inclusive), x22 = end (exclusive)
    // x19 = inbuf base, x20 = outbuf base, x24 = outpos (updated)
    mov x1, x22
    sub x1, x1, #1      // last index = end - 1
    mov x2, x23
rev_loop:
    cmp x1, x2
    blt rev_done
    ldrb w3, [x19, x1]
    strb w3, [x20, x24]
    add x24, x24, #1
    sub x1, x1, #1
    b rev_loop
rev_done:
    ret

_start:
    // read all of stdin into inbuf
    mov x9, #0          // total bytes read so far

read_loop:
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x9
    mov x2, #65536
    sub x2, x2, x9
    cmp x2, #0
    beq done_read
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    beq done_read
    add x9, x9, x0
    b read_loop

done_read:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x20, outbuf
    add x20, x20, :lo12:outbuf

    mov x22, #0         // i
    mov x23, #0         // start
    mov x24, #0         // outpos

scan_loop:
    cmp x22, x9
    bge after_scan
    ldrb w0, [x19, x22]
    cmp w0, #10
    bne scan_next
    // found newline at x22, reverse [start, i) into outbuf
    bl reverse_segment
    mov w3, #10
    strb w3, [x20, x24]
    add x24, x24, #1
    add x23, x22, #1
    add x22, x22, #1
    b scan_loop

scan_next:
    add x22, x22, #1
    b scan_loop

after_scan:
    cmp x23, x9
    bge no_trailing
    mov x22, x9
    bl reverse_segment
    mov w3, #10
    strb w3, [x20, x24]
    add x24, x24, #1

no_trailing:
    // write outbuf[0:x24) to stdout
    mov x5, x20
    mov x6, x24
    mov x7, #0

write_loop:
    cmp x7, x6
    bge write_done
    mov x0, #1
    add x1, x5, x7
    sub x2, x6, x7
    mov x8, #64
    svc #0
    add x7, x7, x0
    b write_loop

write_done:
    mov x0, #0
    mov x8, #93
    svc #0
