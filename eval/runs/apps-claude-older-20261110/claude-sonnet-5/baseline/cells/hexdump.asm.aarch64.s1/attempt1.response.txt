```asm
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 8
inbuf:
    .space 65536
outbuf:
    .space 128

.text
.globl _start
_start:
    mov x19, #0            // total bytes read

read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    beq read_done
    add x19, x19, x0
    b read_loop
read_done:

    mov x20, #0            // current row offset

row_loop:
    cmp x20, x19
    bge after_rows

    sub x21, x19, x20      // remaining
    mov x22, #16
    cmp x21, x22
    csel x21, x21, x22, lt // x21 = n = min(remaining,16)

    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23           // write pointer

    // offset as 8 hex digits
    mov x25, x20
    mov x26, #28
hex_offset_loop:
    lsr x27, x25, x26
    and x27, x27, #0xf
    adrp x28, hexdigits
    add x28, x28, :lo12:hexdigits
    ldrb w29, [x28, x27]
    strb w29, [x24], #1
    subs x26, x26, #4
    bge hex_offset_loop

    // two spaces
    mov w29, #' '
    strb w29, [x24], #1
    strb w29, [x24], #1

    mov x25, #0            // i
hex_bytes_loop:
    cmp x25, #16
    bge hex_bytes_done
    cmp x25, x21
    blt have_byte
    mov w29, #' '
    strb w29, [x24], #1
    strb w29, [x24], #1
    strb w29, [x24], #1
    b after_byte
have_byte:
    adrp x28, inbuf
    add x28, x28, :lo12:inbuf
    add x28, x28, x20
    add x28, x28, x25
    ldrb w27, [x28]
    lsr w9, w27, #4
    and w9, w9, #0xf
    adrp x28, hexdigits
    add x28, x28, :lo12:hexdigits
    ldrb w29, [x28, x9]
    strb w29, [x24], #1
    and w9, w27, #0xf
    ldrb w29, [x28, x9]
    strb w29, [x24], #1
    mov w29, #' '
    strb w29, [x24], #1
after_byte:
    cmp x25, #7
    bne skip_extra_space
    mov w29, #' '
    strb w29, [x24], #1
skip_extra_space:
    add x25, x25, #1
    b hex_bytes_loop
hex_bytes_done:

    mov w29, #' '
    strb w29, [x24], #1
    mov w29, #'|'
    strb w29, [x24], #1

    mov x25, #0
ascii_loop:
    cmp x25, x21
    bge ascii_done
    adrp x28, inbuf
    add x28, x28, :lo12:inbuf
    add x28, x28, x20
    add x28, x28, x25
    ldrb w27, [x28]
    cmp w27, #0x20
    blt dot_char
    cmp w27, #0x7e
    bgt dot_char
    mov w29, w27
    b store_ascii
dot_char:
    mov w29, #'.'
store_ascii:
    strb w29, [x24], #1
    add x25, x25, #1
    b ascii_loop
ascii_done:

    mov w29, #'|'
    strb w29, [x24], #1
    mov w29, #'\n'
    strb w29, [x24], #1

    sub x2, x24, x23
    mov x1, x23
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

    add x20, x20, #16
    b row_loop

after_rows:
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x24, x23
    mov x25, x19
    mov x26, #28
final_hex_loop:
    lsr x27, x25, x26
    and x27, x27, #0xf
    adrp x28, hexdigits
    add x28, x28, :lo12:hexdigits
    ldrb w29, [x28, x27]
    strb w29, [x24], #1
    subs x26, x26, #4
    bge final_hex_loop
    mov w29, #'\n'
    strb w29, [x24], #1

    sub x2, x24, x23
    mov x1, x23
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
```
