.globl _start

// ---------------------------------------------------------------
// _start: read all of stdin, parse it as JSON, report result.
// ---------------------------------------------------------------
_start:
    adrp x9, buf
    add  x9, x9, :lo12:buf
    mov  x19, #0                 // total bytes read

rd_loop:
    movz x2, #1, lsl #16         // x2 = 65536
    sub  x2, x2, x19
    cmp  x2, #0
    b.le rd_done
    mov  x0, #0                  // fd = 0
    add  x1, x9, x19
    mov  x8, #63                 // read
    svc  #0
    cmp  x0, #0
    b.le rd_done
    add  x19, x19, x0
    b    rd_loop

rd_done:
    mov  x11, x19                // x11 = length
    mov  x20, #0                 // x20 = cursor

    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x20, x11
    b.ne fail

    mov  x0, #1
    adrp x1, msg_ok
    add  x1, x1, :lo12:msg_ok
    mov  x2, #3
    mov  x8, #64
    svc  #0
    mov  x0, #0
    mov  x8, #93
    svc  #0

// ---------------------------------------------------------------
// fail: print "error at N\n" and exit(1). N is in x20.
// ---------------------------------------------------------------
fail:
    adrp x7, numbuf_end
    add  x7, x7, :lo12:numbuf_end
    mov  x3, #0
    mov  x2, x20
    cbnz x2, nz_conv
    sub  x7, x7, #1
    mov  w0, #0x30
    strb w0, [x7]
    mov  x3, #1
    b    conv_done
nz_conv:
    mov  x4, #10
cv_loop:
    cbz  x2, conv_done
    udiv x5, x2, x4
    msub x6, x5, x4, x2
    add  w6, w6, #0x30
    sub  x7, x7, #1
    strb w6, [x7]
    add  x3, x3, #1
    mov  x2, x5
    b    cv_loop
conv_done:
    mov  x0, #1
    adrp x1, msg_err_prefix
    add  x1, x1, :lo12:msg_err_prefix
    mov  x2, #9
    mov  x8, #64
    svc  #0

    mov  x0, #1
    mov  x1, x7
    mov  x2, x3
    mov  x8, #64
    svc  #0

    mov  x0, #1
    adrp x1, msg_nl
    add  x1, x1, :lo12:msg_nl
    mov  x2, #1
    mov  x8, #64
    svc  #0

    mov  x0, #1
    mov  x8, #93
    svc  #0

// ---------------------------------------------------------------
// skip_ws: advance x20 past ws bytes (space, tab, LF, CR)
// ---------------------------------------------------------------
skip_ws:
sw_loop:
    cmp  x20, x11
    b.ge sw_done
    ldrb w0, [x9, x20]
    cmp  w0, #0x20
    b.eq sw_adv
    cmp  w0, #0x09
    b.eq sw_adv
    cmp  w0, #0x0A
    b.eq sw_adv
    cmp  w0, #0x0D
    b.eq sw_adv
    b    sw_done
sw_adv:
    add  x20, x20, #1
    b    sw_loop
sw_done:
    ret

// ---------------------------------------------------------------
// parse_value: dispatch on current byte at x20
// ---------------------------------------------------------------
parse_value:
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x7B                // '{'
    b.eq parse_object
    cmp  w0, #0x5B                // '['
    b.eq parse_array
    cmp  w0, #0x22                // '"'
    b.eq parse_string
    cmp  w0, #0x2D                // '-'
    b.eq parse_number
    cmp  w0, #0x30
    b.lt pv_check_letters
    cmp  w0, #0x39
    b.gt pv_check_letters
    b    parse_number
pv_check_letters:
    cmp  w0, #0x74                // 't'
    b.eq pv_true
    cmp  w0, #0x66                // 'f'
    b.eq pv_false
    cmp  w0, #0x6E                // 'n'
    b.eq pv_null
    b    fail
pv_true:
    adrp x1, lit_true
    add  x1, x1, :lo12:lit_true
    mov  x2, #4
    b    match_literal
pv_false:
    adrp x1, lit_false
    add  x1, x1, :lo12:lit_false
    mov  x2, #5
    b    match_literal
pv_null:
    adrp x1, lit_null
    add  x1, x1, :lo12:lit_null
    mov  x2, #4
    b    match_literal

// ---------------------------------------------------------------
// match_literal: compare x2 bytes at x9+x20 against x1
// ---------------------------------------------------------------
match_literal:
    mov  x3, #0
ml_loop:
    cmp  x3, x2
    b.ge ml_done
    add  x4, x20, x3
    cmp  x4, x11
    b.ge ml_fail
    ldrb w5, [x9, x4]
    ldrb w6, [x1, x3]
    cmp  w5, w6
    b.ne ml_fail
    add  x3, x3, #1
    b    ml_loop
ml_done:
    add  x20, x20, x2
    ret
ml_fail:
    mov  x20, x4
    b    fail

// ---------------------------------------------------------------
// parse_number
// ---------------------------------------------------------------
parse_number:
    ldrb w0, [x9, x20]
    cmp  w0, #0x2D
    b.ne pn_after_sign
    add  x20, x20, #1
pn_after_sign:
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.eq pn_zero
    cmp  w0, #0x31
    b.lt fail
    cmp  w0, #0x39
    b.gt fail
    add  x20, x20, #1
pn_int_extra:
    cmp  x20, x11
    b.ge pn_after_int
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.lt pn_after_int
    cmp  w0, #0x39
    b.gt pn_after_int
    add  x20, x20, #1
    b    pn_int_extra
pn_zero:
    add  x20, x20, #1
pn_after_int:
    cmp  x20, x11
    b.ge pn_exp_check
    ldrb w0, [x9, x20]
    cmp  w0, #0x2E
    b.ne pn_exp_check
    add  x20, x20, #1
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.lt fail
    cmp  w0, #0x39
    b.gt fail
    add  x20, x20, #1
pn_frac_extra:
    cmp  x20, x11
    b.ge pn_exp_check
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.lt pn_exp_check
    cmp  w0, #0x39
    b.gt pn_exp_check
    add  x20, x20, #1
    b    pn_frac_extra
pn_exp_check:
    cmp  x20, x11
    b.ge pn_done
    ldrb w0, [x9, x20]
    cmp  w0, #0x65
    b.eq pn_exp_yes
    cmp  w0, #0x45
    b.eq pn_exp_yes
    b    pn_done
pn_exp_yes:
    add  x20, x20, #1
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x2B
    b.eq pn_exp_sign_yes
    cmp  w0, #0x2D
    b.eq pn_exp_sign_yes
    b    pn_exp_digit_req
pn_exp_sign_yes:
    add  x20, x20, #1
pn_exp_digit_req:
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.lt fail
    cmp  w0, #0x39
    b.gt fail
    add  x20, x20, #1
pn_exp_extra:
    cmp  x20, x11
    b.ge pn_done
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.lt pn_done
    cmp  w0, #0x39
    b.gt pn_done
    add  x20, x20, #1
    b    pn_exp_extra
pn_done:
    ret

// ---------------------------------------------------------------
// parse_string
// ---------------------------------------------------------------
parse_string:
    add  x20, x20, #1
ps_loop:
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x22
    b.eq ps_close
    cmp  w0, #0x5C
    b.eq ps_escape
    cmp  w0, #0x20
    b.lt fail
    add  x20, x20, #1
    b    ps_loop
ps_escape:
    add  x20, x20, #1
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x22
    b.eq ps_simple_esc
    cmp  w0, #0x5C
    b.eq ps_simple_esc
    cmp  w0, #0x2F
    b.eq ps_simple_esc
    cmp  w0, #0x62
    b.eq ps_simple_esc
    cmp  w0, #0x66
    b.eq ps_simple_esc
    cmp  w0, #0x6E
    b.eq ps_simple_esc
    cmp  w0, #0x72
    b.eq ps_simple_esc
    cmp  w0, #0x74
    b.eq ps_simple_esc
    cmp  w0, #0x75
    b.eq ps_unicode_esc
    b    fail
ps_simple_esc:
    add  x20, x20, #1
    b    ps_loop
ps_unicode_esc:
    add  x20, x20, #1
    mov  x3, #0
ps_hex_loop:
    cmp  x3, #4
    b.ge ps_loop
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x30
    b.lt fail
    cmp  w0, #0x39
    b.le ps_hex_ok
    cmp  w0, #0x41
    b.lt fail
    cmp  w0, #0x46
    b.le ps_hex_ok
    cmp  w0, #0x61
    b.lt fail
    cmp  w0, #0x66
    b.gt fail
ps_hex_ok:
    add  x20, x20, #1
    add  x3, x3, #1
    b    ps_hex_loop
ps_close:
    add  x20, x20, #1
    ret

// ---------------------------------------------------------------
// parse_object
// ---------------------------------------------------------------
parse_object:
    sub  sp, sp, #16
    str  x30, [sp]
    add  x20, x20, #1
    bl   skip_ws
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x7D
    b.eq po_close
po_member:
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x22
    b.ne fail
    bl   parse_string
    bl   skip_ws
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x3A
    b.ne fail
    add  x20, x20, #1
    bl   skip_ws
    bl   parse_value
    bl   skip_ws
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x2C
    b.eq po_comma
    cmp  w0, #0x7D
    b.eq po_close
    b    fail
po_comma:
    add  x20, x20, #1
    bl   skip_ws
    b    po_member
po_close:
    add  x20, x20, #1
    ldr  x30, [sp]
    add  sp, sp, #16
    ret

// ---------------------------------------------------------------
// parse_array
// ---------------------------------------------------------------
parse_array:
    sub  sp, sp, #16
    str  x30, [sp]
    add  x20, x20, #1
    bl   skip_ws
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x5D
    b.eq pa_close
pa_elem:
    bl   parse_value
    bl   skip_ws
    cmp  x20, x11
    b.ge fail
    ldrb w0, [x9, x20]
    cmp  w0, #0x2C
    b.eq pa_comma
    cmp  w0, #0x5D
    b.eq pa_close
    b    fail
pa_comma:
    add  x20, x20, #1
    bl   skip_ws
    b    pa_elem
pa_close:
    add  x20, x20, #1
    ldr  x30, [sp]
    add  sp, sp, #16
    ret

// ---------------------------------------------------------------
// Data
// ---------------------------------------------------------------
.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err_prefix:
    .ascii "error at "
msg_nl:
    .ascii "\n"
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 8
buf:
    .zero 65536
.balign 8
numbuf:
    .zero 24
numbuf_end:
