```
.section .rodata
.balign 8
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 8
input_buf:
    .zero 65536
output_buf:
    .zero 330000

.text
.globl _start

_start:
    // Read all of stdin into input_buf, tracking total length in x19
    adrp x23, input_buf
    add  x23, x23, :lo12:input_buf
    mov  x19, #0

read_loop:
    mov  x0, #0              // fd = 0
    add  x1, x23, x19        // buf = input_buf + x19
    mov  x2, #65536
    sub  x2, x2, x19         // remaining capacity
    mov  x8, #63             // read
    svc  #0
    cmp  x0, #0
    beq  read_done
    add  x19, x19, x0
    b    read_loop
read_done:

    // Set up output pointer
    adrp x22, output_buf
    add  x22, x22, :lo12:output_buf
    mov  x20, x22            // save base of output buffer

    mov  x21, #0             // row offset

row_loop:
    cmp  x21, x19
    bge  after_rows

    add  x27, x23, x21       // row base pointer in input

    // n = min(16, total - offset)
    sub  x24, x19, x21
    mov  x25, #16
    cmp  x24, x25
    csel x25, x24, x25, lt   // x25 = n

    // write offset (8 hex digits)
    mov  w0, w21
    bl   write_hex8

    // two spaces
    mov  w0, #' '
    strb w0, [x22], #1
    strb w0, [x22], #1

    mov  x26, #0              // byte index i
byte_loop:
    cmp  x26, #16
    bge  byte_loop_done
    cmp  x26, x25
    blt  have_byte

    mov  w0, #' '
    strb w0, [x22], #1
    strb w0, [x22], #1
    strb w0, [x22], #1
    b    after_byte

have_byte:
    ldrb w0, [x27, x26]
    bl   write_hex2
    mov  w0, #' '
    strb w0, [x22], #1

after_byte:
    cmp  x26, #7
    bne  no_extra
    mov  w0, #' '
    strb w0, [x22], #1
no_extra:
    add  x26, x26, #1
    b    byte_loop
byte_loop_done:

    mov  w0, #' '
    strb w0, [x22], #1
    mov  w0, #'|'
    strb w0, [x22], #1

    mov  x26, #0
char_loop:
    cmp  x26, x25
    bge  char_loop_done
    ldrb w0, [x27, x26]
    cmp  w0, #0x20
    blt  notprint
    cmp  w0, #0x7E
    bgt  notprint
    b    store_char
notprint:
    mov  w0, #'.'
store_char:
    strb w0, [x22], #1
    add  x26, x26, #1
    b    char_loop
char_loop_done:

    mov  w0, #'|'
    strb w0, [x22], #1
    mov  w0, #10
    strb w0, [x22], #1

    add  x21, x21, #16
    b    row_loop

after_rows:
    mov  w0, w19
    bl   write_hex8
    mov  w0, #10
    strb w0, [x22], #1

    // write output buffer to stdout
    mov  x5, x20              // current write pointer
    sub  x6, x22, x20         // remaining length

write_loop:
    cbz  x6, write_done
    mov  x0, #1               // fd = 1
    mov  x1, x5
    mov  x2, x6
    mov  x8, #64              // write
    svc  #0
    add  x5, x5, x0
    sub  x6, x6, x0
    b    write_loop
write_done:

    mov  x0, #0
    mov  x8, #93              // exit
    svc  #0

// Writes 8 lowercase hex digits of w0 to [x22], advancing x22 by 8.
write_hex8:
    adrp x10, hexdigits
    add  x10, x10, :lo12:hexdigits
    mov  x4, #28
hex8_loop:
    lsr  w2, w0, w4
    and  w2, w2, #0xF
    ldrb w3, [x10, x2]
    strb w3, [x22], #1
    subs x4, x4, #4
    bge  hex8_loop
    ret

// Writes 2 lowercase hex digits of w0 (low byte) to [x22], advancing x22 by 2.
write_hex2:
    adrp x10, hexdigits
    add  x10, x10, :lo12:hexdigits
    lsr  w2, w0, #4
    and  w2, w2, #0xF
    ldrb w3, [x10, x2]
    strb w3, [x22], #1
    and  w2, w0, #0xF
    ldrb w3, [x10, x2]
    strb w3, [x22], #1
    ret
```
