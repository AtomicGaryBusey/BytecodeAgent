.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err:
    .ascii "error at "
msg_err_len = . - msg_err
msg_nl:
    .ascii "\n"
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.align 8
g_pos:
    .zero 8
g_len:
    .zero 8
buf:
    .zero 65536
digit_buf:
    .zero 24

.text
.globl _start

_start:
    call read_all
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, -1
    je 1f
    jmp fail
1:
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, msg_ok_len
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

read_all:
1:
    mov r8, qword ptr [rip + g_len]
    cmp r8, 65536
    jae 2f
    lea rsi, [rip + buf]
    add rsi, r8
    mov edx, 65536
    sub rdx, r8
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle 2f
    add r8, rax
    mov qword ptr [rip + g_len], r8
    jmp 1b
2:
    ret

peek:
    mov rax, qword ptr [rip + g_pos]
    cmp rax, qword ptr [rip + g_len]
    jae 1f
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rax]
    ret
1:
    mov eax, -1
    ret

advance:
    inc qword ptr [rip + g_pos]
    ret

skip_ws:
1:
    call peek
    cmp eax, 0x20
    je 2f
    cmp eax, 0x09
    je 2f
    cmp eax, 0x0A
    je 2f
    cmp eax, 0x0D
    je 2f
    ret
2:
    call advance
    jmp 1b

check_hex_digit:
    call peek
    cmp eax, '0'
    jl 1f
    cmp eax, '9'
    jle 2f
1:
    cmp eax, 'A'
    jl 3f
    cmp eax, 'F'
    jle 2f
    cmp eax, 'a'
    jl 3f
    cmp eax, 'f'
    jg 3f
2:
    call advance
    ret
3:
    jmp fail

parse_literal:
    mov r9d, esi
    xor ecx, ecx
1:
    cmp ecx, r9d
    jae 2f
    call peek
    movzx edx, byte ptr [rdi + rcx]
    cmp eax, edx
    jne 3f
    call advance
    inc ecx
    jmp 1b
2:
    ret
3:
    jmp fail

parse_value:
    call peek
    cmp eax, 0x22
    je 1f
    cmp eax, '{'
    je 2f
    cmp eax, '['
    je 3f
    cmp eax, 't'
    je 4f
    cmp eax, 'f'
    je 5f
    cmp eax, 'n'
    je 6f
    cmp eax, '-'
    je 7f
    cmp eax, '0'
    jl 9f
    cmp eax, '9'
    jg 9f
    jmp 7f
1:
    call parse_string
    ret
2:
    call parse_object
    ret
3:
    call parse_array
    ret
4:
    lea rdi, [rip + lit_true]
    mov esi, 4
    call parse_literal
    ret
5:
    lea rdi, [rip + lit_false]
    mov esi, 5
    call parse_literal
    ret
6:
    lea rdi, [rip + lit_null]
    mov esi, 4
    call parse_literal
    ret
7:
    call parse_number
    ret
9:
    jmp fail

parse_number:
    call peek
    cmp eax, '-'
    jne 1f
    call advance
1:
    call peek
    cmp eax, '0'
    je 2f
    cmp eax, '1'
    jl 9f
    cmp eax, '9'
    jg 9f
    call advance
3:
    call peek
    cmp eax, '0'
    jl 4f
    cmp eax, '9'
    jg 4f
    call advance
    jmp 3b
2:
    call advance
4:
    call peek
    cmp eax, '.'
    jne 6f
    call advance
    call peek
    cmp eax, '0'
    jl 9f
    cmp eax, '9'
    jg 9f
    call advance
5:
    call peek
    cmp eax, '0'
    jl 6f
    cmp eax, '9'
    jg 6f
    call advance
    jmp 5b
6:
    call peek
    cmp eax, 'e'
    je 7f
    cmp eax, 'E'
    je 7f
    ret
7:
    call advance
    call peek
    cmp eax, '+'
    je 8f
    cmp eax, '-'
    je 8f
    jmp 10f
8:
    call advance
10:
    call peek
    cmp eax, '0'
    jl 9f
    cmp eax, '9'
    jg 9f
    call advance
11:
    call peek
    cmp eax, '0'
    jl 12f
    cmp eax, '9'
    jg 12f
    call advance
    jmp 11b
12:
    ret
9:
    jmp fail

parse_string:
    call advance
1:
    call peek
    cmp eax, -1
    je 9f
    cmp eax, 0x22
    je 2f
    cmp eax, 0x5C
    je 3f
    cmp eax, 0x20
    jl 9f
    call advance
    jmp 1b
2:
    call advance
    ret
3:
    call advance
    call peek
    cmp eax, -1
    je 9f
    cmp eax, 0x22
    je 4f
    cmp eax, 0x5C
    je 4f
    cmp eax, '/'
    je 4f
    cmp eax, 'b'
    je 4f
    cmp eax, 'f'
    je 4f
    cmp eax, 'n'
    je 4f
    cmp eax, 'r'
    je 4f
    cmp eax, 't'
    je 4f
    cmp eax, 'u'
    je 5f
    jmp 9f
4:
    call advance
    jmp 1b
5:
    call advance
    call check_hex_digit
    call check_hex_digit
    call check_hex_digit
    call check_hex_digit
    jmp 1b
9:
    jmp fail

parse_array:
    call advance
    call skip_ws
    call peek
    cmp eax, ']'
    je 5f
1:
    call parse_value
    call skip_ws
    call peek
    cmp eax, ','
    je 2f
    cmp eax, ']'
    je 3f
    jmp 9f
2:
    call advance
    call skip_ws
    jmp 1b
3:
    call advance
    ret
5:
    call advance
    ret
9:
    jmp fail

parse_object:
    call advance
    call skip_ws
    call peek
    cmp eax, '}'
    je 6f
1:
    call peek
    cmp eax, 0x22
    jne 9f
    call parse_string
    call skip_ws
    call peek
    cmp eax, ':'
    jne 9f
    call advance
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, ','
    je 2f
    cmp eax, '}'
    je 3f
    jmp 9f
2:
    call advance
    call skip_ws
    jmp 1b
3:
    call advance
    ret
6:
    call advance
    ret
9:
    jmp fail

fail:
    mov rax, qword ptr [rip + g_pos]
    lea r9, [rip + digit_buf + 23]
    xor r8, r8
    test rax, rax
    jnz 1f
    dec r9
    mov byte ptr [r9], '0'
    inc r8
    jmp 2f
1:
    test rax, rax
    jz 2f
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    inc r8
    jmp 1b
2:
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, msg_err_len
    mov eax, 1
    syscall
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    mov eax, 1
    syscall
    mov edi, 1
    lea rsi, [rip + msg_nl]
    mov edx, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
