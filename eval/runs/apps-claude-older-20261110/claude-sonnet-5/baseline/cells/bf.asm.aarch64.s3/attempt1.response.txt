```asm
.text
.globl _start

_start:
    // ---- read all of stdin into input_buf ----
    mov     x9, #0                  // total bytes read
    adrp    x10, input_buf
    add     x10, x10, :lo12:input_buf

read_loop:
    mov     x11, #65536
    cmp     x9, x11
    b.ge    read_done
    sub     x2, x11, x9             // remaining capacity
    add     x1, x10, x9             // buf + offset
    mov     x0, #0                  // fd = 0
    mov     x8, #63                 // read
    svc     #0
    cmp     x0, #0
    b.eq    read_done
    add     x9, x9, x0
    b       read_loop

read_done:
    // x9 = total length, x10 = base of input buffer
    mov     x11, #0
    mov     x6, #0                  // found flag
    mov     x12, #0                 // position of '!'

find_loop:
    cmp     x11, x9
    b.ge    find_done
    ldrb    w13, [x10, x11]
    cmp     w13, #0x21
    b.eq    found_bang
    add     x11, x11, #1
    b       find_loop
found_bang:
    mov     x12, x11
    mov     x6, #1
find_done:
    cbz     x6, no_bang
    mov     x20, x12                // prog_end
    add     x14, x12, #1            // input_start
    sub     x15, x9, x14            // input_len
    b       split_done
no_bang:
    mov     x20, x9                 // prog_end = total length
    mov     x14, x9                 // input_start
    mov     x15, #0                 // input_len
split_done:
    mov     x19, x10                // program base
    add     x25, x10, x14           // input region base
    mov     x26, x15                // input remaining

    // ---- bracket matching ----
    adrp    x23, bracket_match
    add     x23, x23, :lo12:bracket_match
    adrp    x16, bstack
    add     x16, x16, :lo12:bstack
    mov     x17, #0                 // stack depth
    mov     x24, #0                 // parse index

parse_loop:
    cmp     x24, x20
    b.ge    parse_done
    ldrb    w13, [x19, x24]
    cmp     w13, #0x5B              // '['
    b.eq    is_open
    cmp     w13, #0x5D              // ']'
    b.eq    is_close
    b       parse_next
is_open:
    str     w24, [x16, x17, lsl #2]
    add     x17, x17, #1
    b       parse_next
is_close:
    cmp     x17, #0
    b.eq    unbalanced_exit
    sub     x17, x17, #1
    ldr     w18, [x16, x17, lsl #2] // open_idx (zero extended into x18)
    str     w24, [x23, x18, lsl #2] // bracket_match[open] = close
    str     w18, [x23, x24, lsl #2] // bracket_match[close] = open
    b       parse_next
parse_next:
    add     x24, x24, #1
    b       parse_loop
parse_done:
    cmp     x17, #0
    b.ne    unbalanced_exit
    b       run_program

unbalanced_exit:
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       unbalanced_exit

run_program:
    adrp    x21, tape
    add     x21, x21, :lo12:tape
    mov     x22, #0                 // data pointer offset
    mov     x24, #0                 // ip
    adrp    x27, output_buf
    add     x27, x27, :lo12:output_buf
    mov     x28, #0                 // output count

main_loop:
    cmp     x24, x20
    b.ge    main_done
    ldrb    w13, [x19, x24]
    cmp     w13, #0x3E              // '>'
    b.eq    op_gt
    cmp     w13, #0x3C              // '<'
    b.eq    op_lt
    cmp     w13, #0x2B              // '+'
    b.eq    op_plus
    cmp     w13, #0x2D              // '-'
    b.eq    op_minus
    cmp     w13, #0x2E              // '.'
    b.eq    op_dot
    cmp     w13, #0x2C              // ','
    b.eq    op_comma
    cmp     w13, #0x5B              // '['
    b.eq    op_open
    cmp     w13, #0x5D              // ']'
    b.eq    op_close
    b       main_next

op_gt:
    add     x22, x22, #1
    b       main_next
op_lt:
    sub     x22, x22, #1
    b       main_next
op_plus:
    add     x0, x21, x22
    ldrb    w1, [x0]
    add     w1, w1, #1
    strb    w1, [x0]
    b       main_next
op_minus:
    add     x0, x21, x22
    ldrb    w1, [x0]
    sub     w1, w1, #1
    strb    w1, [x0]
    b       main_next
op_dot:
    add     x0, x21, x22
    ldrb    w1, [x0]
    strb    w1, [x27, x28]
    add     x28, x28, #1
    b       main_next
op_comma:
    cmp     x26, #0
    b.eq    comma_zero
    ldrb    w1, [x25], #1
    sub     x26, x26, #1
    add     x0, x21, x22
    strb    w1, [x0]
    b       main_next
comma_zero:
    add     x0, x21, x22
    strb    wzr, [x0]
    b       main_next
op_open:
    add     x0, x21, x22
    ldrb    w1, [x0]
    cmp     w1, #0
    b.ne    main_next
    ldr     w24, [x23, x24, lsl #2]
    b       main_next
op_close:
    add     x0, x21, x22
    ldrb    w1, [x0]
    cmp     w1, #0
    b.eq    main_next
    ldr     w24, [x23, x24, lsl #2]
    b       main_next
main_next:
    add     x24, x24, #1
    b       main_loop
main_done:
    cmp     x28, #0
    b.eq    exit_ok
    mov     x0, #1
    mov     x1, x27
    mov     x2, x28
    mov     x8, #64
    svc     #0
exit_ok:
    mov     x0, #0
    mov     x8, #93
    svc     #0

.bss
.balign 8
input_buf:      .space 65536
.balign 8
tape:           .space 30000
.balign 8
bracket_match:  .space 262144
.balign 8
bstack:         .space 4000
.balign 8
output_buf:     .space 65536
```
