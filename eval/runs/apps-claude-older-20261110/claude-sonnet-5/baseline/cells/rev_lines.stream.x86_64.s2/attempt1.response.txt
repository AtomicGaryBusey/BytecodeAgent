.section .rodata
nl:
    .byte 0x0A

.bss
    .align 8
inbuf:
    .zero 65600

.text
.globl _start

_start:
    xor r12, r12            # total bytes read so far
    lea r15, [rip + inbuf]  # base address of input buffer

read_loop:
    mov rdi, 0
    lea rsi, [r15 + r12]
    mov rdx, 65600
    sub rdx, r12
    mov rax, 0
    syscall
    test rax, rax
    jle process
    add r12, rax
    jmp read_loop

process:
    mov rbx, r12            # len
    xor r13, r13            # start
    xor r14, r14            # i

scan_loop:
    cmp r14, rbx
    jge scan_done
    movzx eax, byte ptr [r15 + r14]
    cmp al, 0x0A
    jne scan_next
    call reverse_line
    mov rsi_off_dummy, 0
    l
