.section .rodata

.bss
    .align 8
inbuf:
    .zero 65536
outbuf:
    .zero 65537

.text
.globl _start

_start:
    xor r12, r12            # total bytes read so far

read_more:
    mov rdi, 0
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    mov rax, 0
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    cmp r12, 65536
    jae read_done
    jmp read_more

read_done:
    lea r13, [rip + inbuf]
    lea r14, [rip + outbuf]
    xor r15, r15             # output offset
    xor rbx, rbx             # start of current line
    xor rcx, rcx             # scan index

process_loop:
    cmp rcx, r12
    jge process_last
    movzx eax, byte ptr [r13 + rcx]
    cmp al, 0x0A
    jne process_next
    call emit_reversed_line
    mov rbx, rcx
    inc rbx
    inc rcx
    jmp process_loop

process_next:
    inc rcx
    jmp process_loop

process_last:
    cmp rbx, r12
    jge process_end
    mov rcx, r12
    call emit_reversed_line

process_end:
    xor r8, r8               # bytes written so far

write_loop:
    cmp r8, r15
    jge do_exit
    mov rdi, 1
    lea rsi, [rip + outbuf]
    add rsi, r8
    mov rdx, r15
    sub rdx, r8
    mov rax, 1
    syscall
    add r8, rax
    jmp write_loop

do_exit:
    mov rdi, 0
    mov rax, 60
    syscall

# emit_reversed_line: reverses inbuf[rbx..rcx) into outbuf at r15, then appends newline
# inputs: rbx = start index (inclusive), rcx = end index (exclusive)
# uses/updates: r15 (output offset)
emit_reversed_line:
    push rax
    push r9
    push r10
    mov r9, rcx
    dec r9
    mov r10, rbx

copy_loop:
    cmp r10, rcx
    jge copy_done
    movzx eax, byte ptr [r13 + r9]
    mov byte ptr [r14 + r15], al
    inc r15
    dec r9
    inc r10
    jmp copy_loop

copy_done:
    mov byte ptr [r14 + r15], 0x0A
    inc r15
    pop r10
    pop r9
    pop rax
    ret
