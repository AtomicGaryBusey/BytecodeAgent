```asm
.equ CAP, 65536

.bss
.align 4
input_buf:  .zero CAP
output_buf: .zero CAP+1

.text
.globl _start
_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x21, output_buf
    add  x21, x21, :lo12:output_buf

    mov  x16, #CAP
    mov  x9, #0            // read offset

read_loop:
    cmp  x9, x16
    beq  read_full
    mov  x0, #0            // fd = 0
    add  x1, x19, x9
    sub  x2, x16, x9
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    beq  read_full
    add  x9, x9, x0
    b    read_loop

read_full:
    mov  x20, x9           // total bytes read

    mov  x10, #0           // i
    mov  x11, #0           // line_start
    mov  x22, #0           // output length

proc_loop:
    cmp  x10, x20
    beq  proc_end
    ldrb w12, [x19, x10]
    cmp  w12, #10
    bne  proc_next
    bl   emit_line
    add  x11, x10, #1
proc_next:
    add  x10, x10, #1
    b    proc_loop

proc_end:
    cmp  x11, x20
    beq  proc_done
    mov  x10, x20
    bl   emit_line
proc_done:

    mov  x9, #0            // write offset
write_loop:
    cmp  x9, x22
    beq  write_done
    mov  x0, #1            // fd = 1
    add  x1, x21, x9
    sub  x2, x22, x9
    mov  x8, #64
    svc  #0
    add  x9, x9, x0
    b    write_loop

write_done:
    mov  x0, #0
    mov  x8, #93
    svc  #0

emit_line:
    sub  x15, x10, x11     // length = end - start
    cbz  x15, el_nl
    add  x13, x11, x15
    sub  x13, x13, #1      // j = end - 1
el_loop:
    ldrb w14, [x19, x13]
    strb w14, [x21, x22]
    add  x22, x22, #1
    subs x15, x15, #1
    beq  el_nl
    sub  x13, x13, #1
    b    el_loop
el_nl:
    mov  w14, #10
    strb w14, [x21, x22]
    add  x22, x22, #1
    ret
```
