.section .rodata

.text
.globl _start

itoa:
    mov rcx, 10
    lea r8, [rsp - 64]
    mov r9, r8
    xor r10, r10
.div_loop:
    test rax, rax
    jz .after_div
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [r9], dl
    inc r9
    inc r10
    jmp .div_loop
.after_div:
    test r10, r10
    jnz .copy_setup
    mov byte ptr [r9], '0'
    inc r9
.copy_setup:
.copy_loop:
    cmp r9, r8
    je .copy_done
    dec r9
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc rdi
    jmp .copy_loop
.copy_done:
    ret

_start:
    xor r12, r12

read_loop:
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdi, 0
    mov rdx, 65536
    sub rdx, r12
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    xor ebx, ebx

scan_loop:
    cmp r15, r12
    jge scan_done
    lea rsi, [rip + buffer]
    movzx eax, byte ptr [rsi + r15]
    cmp al, 0x20
    je is_space
    cmp al, 0x09
    je is_space
    cmp al, 0x0a
    je is_newline
    cmp al, 0x0b
    je is_space
    cmp al, 0x0c
    je is_space
    cmp al, 0x0d
    je is_space
    cmp ebx, 0
    jne continue_word
    mov ebx, 1
    inc r14
    jmp continue_word

is_newline:
    inc r13
    mov ebx, 0
    jmp continue_word

is_space:
    mov ebx, 0
    jmp continue_word

continue_word:
    inc r15
    jmp scan_loop

scan_done:
    lea rdi, [rip + outbuf]
    mov rax, r13
    call itoa
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, r14
    call itoa
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, r12
    call itoa
    mov byte ptr [rdi], 0x0a
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall

    mov rax, 60
    mov rdi, 0
    syscall

.bss
.align 16
buffer:
    .zero 65536
outbuf:
    .zero 32
