.section .rodata

.bss
.align 16
inbuf: .zero 65536
outbuf: .zero 65537

.text
.globl _start

_start:
    xor r12, r12            # total bytes read

read_loop:
    cmp r12, 65536
    jae read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    xor rdi, rdi
    mov rax, 0
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13            # line start
    xor r14, r14            # current index
    xor r15, r15            # output length

process_loop:
    cmp r14, r12
    jae process_tail
    lea rax, [rip + inbuf]
    movzx ecx, byte ptr [rax + r14]
    cmp cl, 0x0A
    je found_nl
    inc r14
    jmp process_loop

found_nl:
    call reverse_line
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + r15], 0x0A
    inc r15
    inc r14
    mov r13, r14
    jmp process_loop

process_tail:
    cmp r13, r14
    je all_done
    call reverse_line
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + r15], 0x0A
    inc r15

all_done:
    lea rsi, [rip + outbuf]
    mov rdi, 1
    mov rdx, r15

write_loop:
    test rdx, rdx
    jz exit_prog
    mov rax, 1
    syscall
    cmp rax, 0
    jle exit_prog
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_prog:
    mov rax, 60
    mov rdi, 0
    syscall

reverse_line:
    mov rcx, r14
    sub rcx, r13
    test rcx, rcx
    jz rl_ret
    lea rax, [rip + inbuf]
    lea rdx, [rip + outbuf]
    mov r8, r14
    dec r8
    mov r9, r15

rl_loop:
    movzx ebx, byte ptr [rax + r8]
    mov byte ptr [rdx + r9], bl
    dec r8
    inc r9
    dec rcx
    jnz rl_loop
    mov r15, r9

rl_ret:
    ret
