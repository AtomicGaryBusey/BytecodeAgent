.section .rodata
// no read-only data needed

.bss
.align 3
inbuf:
    .zero 65536
outbuf:
    .zero 65537

.text
.globl _start
_start:
    // x19 = inbuf base
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf

    // x20 = total bytes read so far
    mov x20, #0

read_loop:
    // compute remaining capacity = 65536 - total
    mov x9, #65536
    sub x9, x9, x20
    cmp x9, #0
    beq read_done

    mov x0, #0            // fd = 0
    add x1, x19, x20       // buf + total
    mov x2, x9             // count = remaining
    mov x8, #63            // read
    svc #0

    cmp x0, #0
    beq read_done          // EOF

    add x20, x20, x0
    b read_loop

read_done:
    // x21 = outbuf base
    adrp x21, outbuf
    add  x21, x21, :lo12:outbuf

    mov x22, #0            // outlen
    mov x23, #0            // pos

main_loop:
    cmp x23, x20
    bge write_output

    // scan for newline starting at pos
    mov x24, x23            // i = pos

scan_loop:
    cmp x24, x20
    bge scan_done
    ldrb w9, [x19, x24]
    cmp w9, #0x0A
    beq scan_done
    add x24, x24, #1
    b scan_loop

scan_done:
    // lineLen = i - pos
    sub x25, x24, x23

    // reverse copy buf[pos..i-1] into outbuf[outlen..], reversed
    mov x10, #0             // k

rev_loop:
    cmp x10, x25
    bge rev_done
    sub x11, x24, x10
    sub x11, x11, #1        // src index = i-1-k
    ldrb w12, [x19, x11]
    add x13, x22, x10
    strb w12, [x21, x13]
    add x10, x10, #1
    b rev_loop

rev_done:
    add x22, x22, x25

    mov w12, #0x0A
    strb w12, [x21, x22]
    add x22, x22, #1

    // advance pos
    cmp x24, x20
    blt has_newline
    mov x23, x20
    b main_loop

has_newline:
    add x23, x24, #1
    b main_loop

write_output:
    mov x14, #0             // written so far

write_loop:
    cmp x14, x22
    bge do_exit
    mov x0, #1
    add x1, x21, x14
    sub x2, x22, x14
    mov x8, #64
    svc #0
    add x14, x14, x0
    b write_loop

do_exit:
    mov x0, #0
    mov x8, #93
    svc #0
