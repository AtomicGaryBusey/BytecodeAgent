.section .rodata
newline:
    .byte 0x0A

.bss
    .align 8
inbuf:
    .zero 32768
line_off:
    .zero 4000
line_len:
    .zero 4000

.text
.globl _start

_start:
    xor r12, r12            # total bytes read

read_loop:
    mov eax, 32768
    sub eax, r12d
    cmp eax, 0
    je read_done
    mov rdx, rax
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    mov eax, 0
    syscall
    cmp rax, 0
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13             # line count
    xor r14, r14             # current line start
    xor r15, r15             # scan index

parse_loop:
    cmp r15, r12
    jge parse_end
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + r15]
    cmp al, 0x0A
    jne parse_next
    mov rax, r15
    sub rax, r14
    lea rcx, [rip + line_off]
    mov dword ptr [rcx + r13*4], r14d
    lea rcx, [rip + line_len]
    mov dword ptr [rcx + r13*4], eax
    inc r13
    lea r14, [r15 + 1]
parse_next:
    inc r15
    jmp parse_loop

parse_end:
    cmp r14, r12
    jge no_final_line
    mov rax, r12
    sub rax, r14
    lea rcx, [rip + line_off]
    mov dword ptr [rcx + r13*4], r14d
    lea rcx, [rip + line_len]
    mov dword ptr [rcx + r13*4], eax
    inc r13
no_final_line:

    mov rcx, 1               # i
sort_outer:
    cmp rcx, r13
    jge sort_done
    lea rax, [rip + line_off]
    mov edx, dword ptr [rax + rcx*4]   # key_off
    lea rax, [rip + line_len]
    mov esi, dword ptr [rax + rcx*4]   # key_len
    mov rdi, rcx
    dec rdi                            # j

sort_inner:
    cmp rdi, -1
    jle sort_insert
    lea rax, [rip + line_off]
    mov r8d, dword ptr [rax + rdi*4]
    lea rax, [rip + line_len]
    mov r9d, dword ptr [rax + rdi*4]

    push rcx
    push rdi
    push rdx
    push rsi

    mov rdi, r8
    mov rsi, r9
    mov rdx, [rsp + 8]
    mov rcx, [rsp + 0]
    call compare_lines

    mov rsi, [rsp + 0]
    mov rdx, [rsp + 8]
    mov rdi, [rsp + 16]
    mov rcx, [rsp + 24]
    add rsp, 32

    cmp eax, 0
    jle sort_insert

    lea rax, [rip + line_off]
    mov r10d, dword ptr [rax + rdi*4]
    mov dword ptr [rax + rdi*4 + 4], r10d
    lea rax, [rip + line_len]
    mov r10d, dword ptr [rax + rdi*4]
    mov dword ptr [rax + rdi*4 + 4], r10d
    dec rdi
    jmp sort_inner

sort_insert:
    lea rax, [rip + line_off]
    mov dword ptr [rax + rdi*4 + 4], edx
    lea rax, [rip + line_len]
    mov dword ptr [rax + rdi*4 + 4], esi
    inc rcx
    jmp sort_outer

sort_done:
    xor r14, r14
output_loop:
    cmp r14, r13
    jge output_done
    lea rax, [rip + line_off]
    mov r15d, dword ptr [rax + r14*4]
    lea rax, [rip + line_len]
    mov r8d, dword ptr [rax + r14*4]

    cmp r8, 0
    je skip_content
    mov edi, 1
    lea rsi, [rip + inbuf]
    add rsi, r15
    mov rdx, r8
    mov eax, 1
    syscall
skip_content:
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    syscall

    inc r14
    jmp output_loop

output_done:
    mov edi, 0
    mov eax, 60
    syscall

# compare_lines(rdi=off1, rsi=len1, rdx=off2, rcx=len2) -> eax
# returns <0 if line1<line2, 0 if equal, >0 if line1>line2
compare_lines:
    lea r8, [rip + inbuf]
    add rdi, r8
    add rdx, r8
    mov r9, rsi
    cmp r9, rcx
    jle cmp_minok
    mov r9, rcx
cmp_minok:
    xor r10, r10
cmp_loop:
    cmp r10, r9
    jge cmp_tail
    movzx eax, byte ptr [rdi + r10]
    movzx r11d, byte ptr [rdx + r10]
    cmp eax, r11d
    jne cmp_diff
    inc r10
    jmp cmp_loop
cmp_diff:
    sub eax, r11d
    ret
cmp_tail:
    mov eax, esi
    sub eax, ecx
    ret
