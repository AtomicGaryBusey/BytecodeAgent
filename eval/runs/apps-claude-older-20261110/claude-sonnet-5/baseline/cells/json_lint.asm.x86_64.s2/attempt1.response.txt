```asm
.section .rodata
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

msg_error:
    .ascii "error at "
msg_error_len = . - msg_error

msg_nl:
    .ascii "\n"

.bss
    .align 8
buf:
    .zero 65536
len:
    .zero 8
pos:
    .zero 8
err_pos:
    .zero 8
numbuf:
    .zero 32

.text
.globl _start

_start:
    xor r12, r12
.read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    cmp rax, 0
    jle .read_done
    add r12, rax
    cmp r12, 65536
    jae .read_done
    jmp .read_loop
.read_done:
    mov qword ptr [rip + len], r12
    mov qword ptr [rip + pos], 0

    call parse_document
    test al, al
    jnz .ok_path

    lea rdi, [rip + msg_error]
    mov rsi, msg_error_len
    call write_all

    mov rax, [rip + err_pos]
    call print_number

    lea rdi, [rip + msg_nl]
    mov rsi, 1
    call write_all

    mov rdi, 1
    jmp do_exit

.ok_path:
    lea rdi, [rip + msg_ok]
    mov rsi, msg_ok_len
    call write_all
    mov rdi, 0

do_exit:
    mov rax, 60
    syscall
    jmp do_exit

write_all:
    mov r13, rdi
    mov r14, rsi
.wa_loop:
    test r14, r14
    jz .wa_done
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    mov rdx, r14
    syscall
    add r13, rax
    sub r14, rax
    jmp .wa_loop
.wa_done:
    ret

print_number:
    lea rdi, [rip + numbuf + 31]
    mov rbx, rdi
    test rax, rax
    jnz .pn2_loop
    dec rdi
    mov byte ptr [rdi], '0'
    jmp .pn2_done
.pn2_loop:
    test rax, rax
    jz .pn2_done
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    jmp .pn2_loop
.pn2_done:
    mov rsi, rbx
    sub rsi, rdi
    call write_all
    ret

skip_ws:
.Lws_loop:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .Lws_done
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x20
    je .Lws_adv
    cmp dl, 0x09
    je .Lws_adv
    cmp dl, 0x0A
    je .Lws_adv
    cmp dl, 0x0D
    je .Lws_adv
    jmp .Lws_done
.Lws_adv:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .Lws_loop
.Lws_done:
    ret

parse_document:
    call skip_ws
    call parse_value
    test al, al
    jz .pd_fail
    call skip_ws
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jne .pd_trailing
    mov eax, 1
    ret
.pd_trailing:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret
.pd_fail:
    xor eax, eax
    ret

parse_value:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pv_fail_atpos
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '{'
    je parse_object
    cmp dl, '['
    je parse_array
    cmp dl, 0x22
    je parse_string
    cmp dl, 't'
    je .pv_true
    cmp dl, 'f'
    je .pv_false
    cmp dl, 'n'
    je .pv_null
    cmp dl, '-'
    je parse_number
    cmp dl, '0'
    jb .pv_fail_atpos
    cmp dl, '9'
    ja .pv_fail_atpos
    jmp parse_number
.pv_true:
    lea rdi, [rip + lit_true]
    mov rsi, 4
    jmp parse_literal
.pv_false:
    lea rdi, [rip + lit_false]
    mov rsi, 5
    jmp parse_literal
.pv_null:
    lea rdi, [rip + lit_null]
    mov rsi, 4
    jmp parse_literal
.pv_fail_atpos:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret

parse_literal:
    mov r8, [rip + pos]
    mov r9, [rip + len]
    xor rcx, rcx
.pl_loop:
    cmp rcx, rsi
    je .pl_success
    mov r10, r8
    add r10, rcx
    cmp r10, r9
    jae .pl_fail
    lea r11, [rip + buf]
    movzx eax, byte ptr [r11 + r10]
    movzx ebx, byte ptr [rdi + rcx]
    cmp al, bl
    jne .pl_fail
    inc rcx
    jmp .pl_loop
.pl_success:
    mov rax, r8
    add rax, rsi
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.pl_fail:
    mov qword ptr [rip + err_pos], r10
    xor eax, eax
    ret

parse_object:
    mov rax, [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .po_loop_entry
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '}'
    jne .po_loop_entry
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.po_loop_entry:
.po_loop:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .po_fail_key
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x22
    jne .po_fail_key
    call parse_string
    test al, al
    jz .po_ret_fail
    call skip_ws
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .po_fail_colon
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ':'
    jne .po_fail_colon
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    call parse_value
    test al, al
    jz .po_ret_fail
    call skip_ws
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .po_fail_sep
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ','
    je .po_comma
    cmp dl, '}'
    je .po_close
    jmp .po_fail_sep
.po_comma:
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    jmp .po_loop
.po_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.po_fail_key:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret
.po_fail_colon:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret
.po_fail_sep:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret
.po_ret_fail:
    xor eax, eax
    ret

parse_array:
    mov rax, [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pa_loop_entry
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ']'
    jne .pa_loop_entry
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.pa_loop_entry:
.pa_loop:
    call parse_value
    test al, al
    jz .pa_ret_fail
    call skip_ws
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pa_fail_sep
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, ','
    je .pa_comma
    cmp dl, ']'
    je .pa_close
    jmp .pa_fail_sep
.pa_comma:
    inc rax
    mov qword ptr [rip + pos], rax
    call skip_ws
    jmp .pa_loop
.pa_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.pa_fail_sep:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret
.pa_ret_fail:
    xor eax, eax
    ret

parse_string:
    mov rax, [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
.ps_loop:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .ps_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x22
    je .ps_close
    cmp dl, 0x5C
    je .ps_escape
    cmp dl, 0x20
    jb .ps_fail
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .ps_loop
.ps_close:
    inc rax
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.ps_escape:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .ps_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x22
    je .ps_esc_simple
    cmp dl, 0x5C
    je .ps_esc_simple
    cmp dl, '/'
    je .ps_esc_simple
    cmp dl, 'b'
    je .ps_esc_simple
    cmp dl, 'f'
    je .ps_esc_simple
    cmp dl, 'n'
    je .ps_esc_simple
    cmp dl, 'r'
    je .ps_esc_simple
    cmp dl, 't'
    je .ps_esc_simple
    cmp dl, 'u'
    je .ps_esc_u
    jmp .ps_fail
.ps_esc_simple:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .ps_loop
.ps_esc_u:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rcx, 4
.ps_u_loop:
    test rcx, rcx
    jz .ps_u_done
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .ps_fail
    lea rdx, [rip + buf]
    movzx r8d, byte ptr [rdx + rax]
    cmp r8b, '0'
    jb .ps_fail
    cmp r8b, '9'
    jbe .ps_u_ok
    cmp r8b, 'A'
    jb .ps_fail
    cmp r8b, 'F'
    jbe .ps_u_ok
    cmp r8b, 'a'
    jb .ps_fail
    cmp r8b, 'f'
    ja .ps_fail
.ps_u_ok:
    inc rax
    dec rcx
    jmp .ps_u_loop
.ps_u_done:
    mov qword ptr [rip + pos], rax
    jmp .ps_loop
.ps_fail:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret

parse_number:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '-'
    jne .pn_int
    inc rax
    mov qword ptr [rip + pos], rax
.pn_int:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    je .pn_zero
    cmp dl, '1'
    jb .pn_fail
    cmp dl, '9'
    ja .pn_fail
    inc rax
    mov qword ptr [rip + pos], rax
.pn_digits_loop:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_frac
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .pn_frac
    cmp dl, '9'
    ja .pn_frac
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .pn_digits_loop
.pn_zero:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .pn_frac
.pn_frac:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_exp
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '.'
    jne .pn_exp
    inc rax
    mov qword ptr [rip + pos], rax
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .pn_fail
    cmp dl, '9'
    ja .pn_fail
    inc rax
    mov qword ptr [rip + pos], rax
.pn_frac_digits:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_exp
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .pn_exp
    cmp dl, '9'
    ja .pn_exp
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .pn_frac_digits
.pn_exp:
    mov rax, [rip + pos]
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_success
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 'e'
    je .pn_exp_go
    cmp dl, 'E'
    je .pn_exp_go
    jmp .pn_success
.pn_exp_go:
    inc rax
    mov qword ptr [rip + pos], rax
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '+'
    je .pn_exp_sign
    cmp dl, '-'
    je .pn_exp_sign
    jmp .pn_exp_digit1
.pn_exp_sign:
    inc rax
    mov qword ptr [rip + pos], rax
.pn_exp_digit1:
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_fail
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .pn_fail
    cmp dl, '9'
    ja .pn_fail
    inc rax
    mov qword ptr [rip + pos], rax
.pn_exp_digits:
    mov rbx, [rip + len]
    cmp rax, rbx
    jae .pn_success
    lea rcx, [rip + buf]
    movzx edx, byte ptr [rcx + rax]
    cmp dl, '0'
    jb .pn_success
    cmp dl, '9'
    ja .pn_success
    inc rax
    mov qword ptr [rip + pos], rax
    jmp .pn_exp_digits
.pn_success:
    mov qword ptr [rip + pos], rax
    mov eax, 1
    ret
.pn_fail:
    mov qword ptr [rip + err_pos], rax
    xor eax, eax
    ret
```
