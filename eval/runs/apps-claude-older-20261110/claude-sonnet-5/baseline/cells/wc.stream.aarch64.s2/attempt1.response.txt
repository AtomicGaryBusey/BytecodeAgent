.globl _start
.text

_start:
    mov x20, #0                 // total bytes read

    adrp x9, buf
    add x9, x9, :lo12:buf

read_loop:
    mov x0, #0                  // fd = 0
    add x1, x9, x20
    mov x2, #65536
    sub x2, x2, x20
    mov x8, #63                 // read
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x10, #0                 // index
    mov x22, #0                 // lines
    mov x23, #0                 // words
    mov x24, #0                 // in_word flag

process_loop:
    cmp x10, x20
    b.ge process_done
    ldrb w11, [x9, x10]
    add x10, x10, #1

    cmp w11, #32
    b.eq is_ws
    cmp w11, #9
    b.eq is_ws
    cmp w11, #10
    b.eq is_nl
    cmp w11, #11
    b.eq is_ws
    cmp w11, #12
    b.eq is_ws
    cmp w11, #13
    b.eq is_ws

    cbnz x24, process_loop
    add x23, x23, #1
    mov x24, #1
    b process_loop

is_nl:
    add x22, x22, #1
    mov x24, #0
    b process_loop

is_ws:
    mov x24, #0
    b process_loop

process_done:
    adrp x12, out_buf
    add x12, x12, :lo12:out_buf

    adrp x13, digits_tmp
    add x13, x13, :lo12:digits_tmp
    add x14, x13, #24

    // lines
    mov x0, x22
    mov x1, x14
    bl num_to_str
    mov x3, x1
    mov x4, x2
copy1:
    cbz x4, copy1_done
    ldrb w5, [x3], #1
    strb w5, [x12], #1
    sub x4, x4, #1
    b copy1
copy1_done:
    mov w5, #32
    strb w5, [x12], #1

    // words
    mov x0, x23
    mov x1, x14
    bl num_to_str
    mov x3, x1
    mov x4, x2
copy2:
    cbz x4, copy2_done
    ldrb w5, [x3], #1
    strb w5, [x12], #1
    sub x4, x4, #1
    b copy2
copy2_done:
    mov w5, #32
    strb w5, [x12], #1

    // bytes
    mov x0, x20
    mov x1, x14
    bl num_to_str
    mov x3, x1
    mov x4, x2
copy3:
    cbz x4, copy3_done
    ldrb w5, [x3], #1
    strb w5, [x12], #1
    sub x4, x4, #1
    b copy3
copy3_done:
    mov w5, #10
    strb w5, [x12], #1

    adrp x15, out_buf
    add x15, x15, :lo12:out_buf
    sub x2, x12, x15
    mov x1, x15

write_loop:
    mov x0, #1
    mov x8, #64                 // write
    svc #0
    cmp x0, x2
    b.eq write_done
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

write_done:
    mov x0, #0
    mov x8, #93                 // exit
    svc #0

num_to_str:
    // x0 = value, x1 = end pointer (exclusive)
    // returns: x1 = start pointer of digits, x2 = length
    mov x7, x1
nts_loop:
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add x5, x5, #48
    sub x1, x1, #1
    strb w5, [x1]
    mov x0, x4
    cbnz x0, nts_loop
    sub x2, x7, x1
    ret

.bss
.align 3
buf:
    .zero 65536
digits_tmp:
    .zero 24
out_buf:
    .zero 64
