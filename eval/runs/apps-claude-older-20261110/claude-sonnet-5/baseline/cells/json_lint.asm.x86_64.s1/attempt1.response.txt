.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

msg_err:
    .ascii "error at "
msg_err_len = . - msg_err

msg_nl:
    .ascii "\n"

.equ QUOTE, 0x22
.equ BACKSLASH, 0x5C

.bss
    .align 8
buf:
    .zero 65536
    .align 8
pos:
    .zero 8
    .align 8
len:
    .zero 8
    .align 8
numbuf:
    .zero 24

.text
.globl _start

# ---- peek: returns current byte in eax, or -1 if EOF ----
peek:
    mov rcx, qword ptr [rip + pos]
    cmp rcx, qword ptr [rip + len]
    jae peek_eof
    lea rsi, [rip + buf]
    movzx eax, byte ptr [rsi + rcx]
    ret
peek_eof:
    mov eax, -1
    ret

# ---- advance: pos += 1 ----
advance:
    inc qword ptr [rip + pos]
    ret

# ---- match_char: r8d = expected char, compares peek, advances or fails ----
match_char:
    call peek
    cmp eax, r8d
    jne fail
    call advance
    ret

# ---- skip_ws ----
skip_ws:
sw_loop:
    call peek
    cmp eax, 0x20
    je sw_adv
    cmp eax, 0x09
    je sw_adv
    cmp eax, 0x0A
    je sw_adv
    cmp eax, 0x0D
    je sw_adv
    ret
sw_adv:
    call advance
    jmp sw_loop

# ---- check_hex_digit ----
check_hex_digit:
    call peek
    cmp eax, -1
    je fail
    cmp eax, '0'
    jl fail
    cmp eax, '9'
    jle chd_ok
    cmp eax, 'a'
    jl chd_checkupper
    cmp eax, 'f'
    jle chd_ok
    jmp fail
chd_checkupper:
    cmp eax, 'A'
    jl fail
    cmp eax, 'F'
    jle chd_ok
    jmp fail
chd_ok:
    call advance
    ret

# ---- parse_string: expects peek == QUOTE ----
parse_string:
    call advance
ps_loop:
    call peek
    cmp eax, -1
    je fail
    cmp eax, QUOTE
    je ps_close
    cmp eax, BACKSLASH
    je ps_escape
    cmp eax, 0x20
    jl fail
    call advance
    jmp ps_loop
ps_escape:
    call advance
    call peek
    cmp eax, -1
    je fail
    cmp eax, QUOTE
    je ps_simple
    cmp eax, BACKSLASH
    je ps_simple
    cmp eax, '/'
    je ps_simple
    cmp eax, 'b'
    je ps_simple
    cmp eax, 'f'
    je ps_simple
    cmp eax, 'n'
    je ps_simple
    cmp eax, 'r'
    je ps_simple
    cmp eax, 't'
    je ps_simple
    cmp eax, 'u'
    je ps_uesc
    jmp fail
ps_simple:
    call advance
    jmp ps_loop
ps_uesc:
    call advance
    call check_hex_digit
    call check_hex_digit
    call check_hex_digit
    call check_hex_digit
    jmp ps_loop
ps_close:
    call advance
    ret

# ---- parse_number: expects peek == '-' or digit ----
parse_number:
    call peek
    cmp eax, '-'
    jne pn_check_int
    call advance
pn_check_int:
    call peek
    cmp eax, '0'
    jne pn_nonzero
    call advance
    jmp pn_frac
pn_nonzero:
    cmp eax, '1'
    jl fail
    cmp eax, '9'
    jg fail
    call advance
pn_digits:
    call peek
    cmp eax, '0'
    jl pn_frac
    cmp eax, '9'
    jg pn_frac
    call advance
    jmp pn_digits
pn_frac:
    call peek
    cmp eax, '.'
    jne pn_exp
    call advance
    call peek
    cmp eax, '0'
    jl fail
    cmp eax, '9'
    jg fail
    call advance
pn_frac_digits:
    call peek
    cmp eax, '0'
    jl pn_exp
    cmp eax, '9'
    jg pn_exp
    call advance
    jmp pn_frac_digits
pn_exp:
    call peek
    cmp eax, 'e'
    je pn_exp_go
    cmp eax, 'E'
    je pn_exp_go
    ret
pn_exp_go:
    call advance
    call peek
    cmp eax, '+'
    je pn_exp_sign
    cmp eax, '-'
    je pn_exp_sign
    jmp pn_exp_digit1
pn_exp_sign:
    call advance
pn_exp_digit1:
    call peek
    cmp eax, '0'
    jl fail
    cmp eax, '9'
    jg fail
    call advance
pn_exp_digits:
    call peek
    cmp eax, '0'
    jl pn_ret
    cmp eax, '9'
    jg pn_ret
    call advance
    jmp pn_exp_digits
pn_ret:
    ret

# ---- expect_true/false/null: expects leading char already peeked ----
expect_true:
    call advance
    mov r8d, 'r'
    call match_char
    mov r8d, 'u'
    call match_char
    mov r8d, 'e'
    call match_char
    ret

expect_false:
    call advance
    mov r8d, 'a'
    call match_char
    mov r8d, 'l'
    call match_char
    mov r8d, 's'
    call match_char
    mov r8d, 'e'
    call match_char
    ret

expect_null:
    call advance
    mov r8d, 'u'
    call match_char
    mov r8d, 'l'
    call match_char
    mov r8d, 'l'
    call match_char
    ret

# ---- parse_object: expects peek == '{' ----
parse_object:
    call advance
    call skip_ws
    call peek
    cmp eax, '}'
    je po_close
po_member:
    call peek
    cmp eax, QUOTE
    jne fail
    call parse_string
    call skip_ws
    mov r8d, ':'
    call match_char
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, ','
    je po_comma
    cmp eax, '}'
    je po_close
    jmp fail
po_comma:
    call advance
    call skip_ws
    jmp po_member
po_close:
    call advance
    ret

# ---- parse_array: expects peek == '[' ----
parse_array:
    call advance
    call skip_ws
    call peek
    cmp eax, ']'
    je pa_close
pa_elem:
    call parse_value
    call skip_ws
    call peek
    cmp eax, ','
    je pa_comma
    cmp eax, ']'
    je pa_close
    jmp fail
pa_comma:
    call advance
    call skip_ws
    jmp pa_elem
pa_close:
    call advance
    ret

# ---- parse_value ----
parse_value:
    call peek
    cmp eax, '{'
    je pv_obj
    cmp eax, '['
    je pv_arr
    cmp eax, QUOTE
    je pv_str
    cmp eax, 't'
    je pv_true
    cmp eax, 'f'
    je pv_false
    cmp eax, 'n'
    je pv_null
    cmp eax, '-'
    je pv_num
    cmp eax, '0'
    jl fail
    cmp eax, '9'
    jg fail
    jmp pv_num
pv_obj:
    call parse_object
    ret
pv_arr:
    call parse_array
    ret
pv_str:
    call parse_string
    ret
pv_true:
    call expect_true
    ret
pv_false:
    call expect_false
    ret
pv_null:
    call expect_null
    ret
pv_num:
    call parse_number
    ret

# ---- fail: prints "error at N\n" and exits 1 ----
fail:
    mov rax, qword ptr [rip + pos]
    lea rcx, [rip + numbuf + 24]
    mov rbx, rcx
    mov r9, 10
    test rax, rax
    jnz f_nz
    dec rcx
    mov byte ptr [rcx], '0'
    jmp f_done
f_nz:
f_loop:
    test rax, rax
    jz f_done
    dec rcx
    xor rdx, rdx
    div r9
    add dl, '0'
    mov byte ptr [rcx], dl
    jmp f_loop
f_done:
    mov r8, rcx
    mov r10, rbx
    sub r10, rcx

    mov rdi, 1
    lea rsi, [rip + msg_err]
    mov rdx, msg_err_len
    mov rax, 1
    syscall

    mov rdi, 1
    mov rsi, r8
    mov rdx, r10
    mov rax, 1
    syscall

    mov rdi, 1
    lea rsi, [rip + msg_nl]
    mov rdx, 1
    mov rax, 1
    syscall

    mov rdi, 1
    mov rax, 60
    syscall

# ---- _start ----
_start:
    xor rbx, rbx
read_loop:
    mov rdx, 65536
    sub rdx, rbx
    cmp rdx, 0
    je read_done
    lea rsi, [rip + buf]
    add rsi, rbx
    mov rdi, 0
    mov rax, 0
    syscall
    cmp rax, 0
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov qword ptr [rip + len], rbx
    mov qword ptr [rip + pos], 0

    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, -1
    jne fail

    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    mov rax, 1
    syscall

    mov rdi, 0
    mov rax, 60
    syscall
