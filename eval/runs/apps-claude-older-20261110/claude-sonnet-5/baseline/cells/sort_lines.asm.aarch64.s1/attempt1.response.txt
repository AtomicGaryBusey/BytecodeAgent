```asm
.globl _start

.text

// cmp_lines: compare two byte sequences like memcmp, with shorter-is-less
// on common prefix. Input: x0=ptr1, x1=len1, x2=ptr2, x3=len2.
// Output: w0 <0 if seq1<seq2, 0 if equal, >0 if seq1>seq2.
// Scratch registers used: x0,x1,x2,x3,x10,x11,x12,x13.
cmp_lines:
    cmp     x1, x3
    csel    x10, x1, x3, lo   // x10 = min(len1,len2)
    mov     x11, #0           // i = 0
cl_loop:
    cmp     x11, x10
    beq     cl_tie
    ldrb    w12, [x0, x11]
    ldrb    w13, [x2, x11]
    cmp     w12, w13
    bne     cl_diff
    add     x11, x11, #1
    b       cl_loop
cl_diff:
    sub     w0, w12, w13
    ret
cl_tie:
    sub     x0, x1, x3
    ret

_start:
    // ---- read all of stdin into input_buf ----
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, #0            // total bytes read

read_loop:
    mov     x0, #0             // fd = 0
    add     x1, x19, x20       // buf = input_buf + len
    mov     x2, #32768
    sub     x2, x2, x20        // remaining capacity
    cmp     x2, #0
    beq     read_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    beq     read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    // ---- parse lines into line_start[] / line_len[] ----
    adrp    x21, line_start
    add     x21, x21, :lo12:line_start
    adrp    x22, line_len
    add     x22, x22, :lo12:line_len
    mov     x23, #0            // pos
    mov     x24, #0            // n (line count)

parse_loop:
    cmp     x23, x20
    bge     parse_done
    mov     x25, x23           // scan index
find_nl:
    cmp     x25, x20
    bge     no_nl
    ldrb    w0, [x19, x25]
    cmp     w0, #0x0a
    beq     found_nl
    add     x25, x25, #1
    b       find_nl
found_nl:
    sub     x0, x25, x23       // length = idx - pos
    str     w23, [x21, x24, lsl #2]
    str     w0,  [x22, x24, lsl #2]
    add     x24, x24, #1
    add     x23, x25, #1
    b       parse_loop
no_nl:
    sub     x0, x20, x23       // length = len - pos
    str     w23, [x21, x24, lsl #2]
    str     w0,  [x22, x24, lsl #2]
    add     x24, x24, #1
    mov     x23, x20
    b       parse_loop
parse_done:

    // ---- insertion sort line_start[]/line_len[] by content ----
    cmp     x24, #2
    blt     sort_done
    mov     x5, #1             // i
outer_loop:
    cmp     x5, x24
    bge     sort_done
    ldr     w7, [x21, x5, lsl #2]   // key_start
    ldr     w8, [x22, x5, lsl #2]   // key_len
    sub     x6, x5, #1              // j
inner_loop:
    cmp     x6, #0
    blt     inner_done
    ldr     w0, [x21, x6, lsl #2]   // line_start[j]
    ldr     w1, [x22, x6, lsl #2]   // line_len[j]
    add     x0, x19, x0             // ptr_j
    mov     x2, x19
    add     x2, x2, x7              // ptr_key
    mov     x3, x8                  // key_len
    bl      cmp_lines
    cmp     w0, #0
    ble     inner_done
    add     x9, x6, #1
    ldr     w0, [x21, x6, lsl #2]
    str     w0, [x21, x9, lsl #2]
    ldr     w0, [x22, x6, lsl #2]
    str     w0, [x22, x9, lsl #2]
    sub     x6, x6, #1
    b       inner_loop
inner_done:
    add     x9, x6, #1
    str     w7, [x21, x9, lsl #2]
    str     w8, [x22, x9, lsl #2]
    add     x5, x5, #1
    b       outer_loop
sort_done:

    // ---- print sorted lines ----
    mov     x5, #0
out_loop:
    cmp     x5, x24
    bge     out_done
    ldr     w6, [x21, x5, lsl #2]   // start
    ldr     w7, [x22, x5, lsl #2]   // len
    cbz     w7, skip_content
    mov     x0, #1
    add     x1, x19, x6
    mov     x2, x7
    mov     x8, #64
    svc     #0
skip_content:
    mov     x0, #1
    adrp    x1, newline_byte
    add     x1, x1, :lo12:newline_byte
    mov     x2, #1
    mov     x8, #64
    svc     #0
    add     x5, x5, #1
    b       out_loop
out_done:

    mov     x0, #0
    mov     x8, #93
    svc     #0
halt:
    b       halt

.section .rodata
newline_byte:
    .byte   0x0a

.bss
.align 3
input_buf:
    .zero   32768
.align 2
line_start:
    .zero   4000
line_len:
    .zero   4000
```
