```asm
.globl _start

.bss
.balign 4
INBUF: .zero 65536
TAPE: .zero 30000
.balign 4
MATCH: .zero 262144
STACK: .zero 4000
.balign 4
OBUF: .zero 65536

.text
_start:
    // ---- read all of stdin into INBUF, total length -> x9 ----
    mov x9, #0
read_loop:
    adrp x1, INBUF
    add x1, x1, :lo12:INBUF
    add x1, x1, x9
    mov x2, #65536
    sub x2, x2, x9
    cmp x2, #0
    b.le read_done
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x9, x9, x0
    b read_loop
read_done:
    // x9 = INLEN
    adrp x20, INBUF
    add x20, x20, :lo12:INBUF

    mov x10, #0
find_loop:
    cmp x10, x9
    b.ge not_found
    ldrb w11, [x20, x10]
    cmp w11, #0x21
    b.eq found
    add x10, x10, #1
    b find_loop

found:
    mov x19, x10
    add x13, x10, #1
    sub x21, x9, x13
    add x22, x20, x13
    b after_split

not_found:
    mov x19, x9
    mov x21, #0
    mov x22, x20

after_split:
    // x19=programlen, x20=PROG base, x21=inputlen, x22=INPUT base

    // ---- bracket balance check and match table build ----
    adrp x24, MATCH
    add x24, x24, :lo12:MATCH
    adrp x15, STACK
    add x15, x15, :lo12:STACK
    mov x16, #0
    mov x26, #0
check_loop:
    cmp x26, x19
    b.ge check_done
    ldrb w11, [x20, x26]
    cmp w11, #0x5B
    b.eq check_open
    cmp w11, #0x5D
    b.eq check_close
    add x26, x26, #1
    b check_loop
check_open:
    str w26, [x15, x16, lsl #2]
    add x16, x16, #1
    add x26, x26, #1
    b check_loop
check_close:
    cmp x16, #0
    b.eq error_exit
    sub x16, x16, #1
    ldr w17, [x15, x16, lsl #2]
    str w17, [x24, x26, lsl #2]
    str w26, [x24, x17, lsl #2]
    add x26, x26, #1
    b check_loop
check_done:
    cmp x16, #0
    b.ne error_exit

    // ---- set up execution state ----
    adrp x23, TAPE
    add x23, x23, :lo12:TAPE
    adrp x29, OBUF
    add x29, x29, :lo12:OBUF
    mov x25, #0
    mov x26, #0
    mov x27, #0
    mov x28, #0

exec_loop:
    cmp x26, x19
    b.ge exec_done
    ldrb w11, [x20, x26]
    cmp w11, #0x3E
    b.eq op_gt
    cmp w11, #0x3C
    b.eq op_lt
    cmp w11, #0x2B
    b.eq op_plus
    cmp w11, #0x2D
    b.eq op_minus
    cmp w11, #0x2E
    b.eq op_dot
    cmp w11, #0x2C
    b.eq op_comma
    cmp w11, #0x5B
    b.eq op_open
    cmp w11, #0x5D
    b.eq op_close
    add x26, x26, #1
    b exec_loop

op_gt:
    add x25, x25, #1
    add x26, x26, #1
    b exec_loop
op_lt:
    sub x25, x25, #1
    add x26, x26, #1
    b exec_loop
op_plus:
    ldrb w12, [x23, x25]
    add w12, w12, #1
    strb w12, [x23, x25]
    add x26, x26, #1
    b exec_loop
op_minus:
    ldrb w12, [x23, x25]
    sub w12, w12, #1
    strb w12, [x23, x25]
    add x26, x26, #1
    b exec_loop
op_dot:
    ldrb w12, [x23, x25]
    strb w12, [x29, x28]
    add x28, x28, #1
    add x26, x26, #1
    b exec_loop
op_comma:
    cmp x27, x21
    b.ge comma_zero
    ldrb w12, [x22, x27]
    add x27, x27, #1
    b comma_store
comma_zero:
    mov w12, #0
comma_store:
    strb w12, [x23, x25]
    add x26, x26, #1
    b exec_loop
op_open:
    ldrb w12, [x23, x25]
    cmp w12, #0
    b.ne open_cont
    ldr w13, [x24, x26, lsl #2]
    add x26, x13, #1
    b exec_loop
open_cont:
    add x26, x26, #1
    b exec_loop
op_close:
    ldrb w12, [x23, x25]
    cmp w12, #0
    b.eq close_cont
    ldr w13, [x24, x26, lsl #2]
    add x26, x13, #1
    b exec_loop
close_cont:
    add x26, x26, #1
    b exec_loop

exec_done:
    cmp x28, #0
    b.eq exit0
    mov x0, #1
    mov x1, x29
    mov x2, x28
    mov x8, #64
    svc #0
exit0:
    mov x0, #0
    mov x8, #93
    svc #0

error_exit:
    mov x0, #1
    mov x8, #93
    svc #0
```
