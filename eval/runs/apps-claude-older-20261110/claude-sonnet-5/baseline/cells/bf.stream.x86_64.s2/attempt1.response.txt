```asm
.section .bss
input_buf:
    .space 65536
tape:
    .space 30000
output_buf:
    .space 65536
match_table:
    .space 262144
bracket_stack:
    .space 4000
prog_len:
    .zero 8
in_start:
    .zero 8
in_len:
    .zero 8

.text
.globl _start
_start:
    xor r8, r8
read_loop:
    cmp r8, 65536
    jge read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:

    lea rcx, [rip + input_buf]
    xor rax, rax
find_loop:
    cmp rax, r8
    je not_found
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x21
    je found
    inc rax
    jmp find_loop
found:
    mov qword ptr [rip + prog_len], rax
    lea rdx, [rax + 1]
    mov qword ptr [rip + in_start], rdx
    mov rdx, r8
    sub rdx, rax
    dec rdx
    mov qword ptr [rip + in_len], rdx
    jmp split_done
not_found:
    mov qword ptr [rip + prog_len], r8
    mov qword ptr [rip + in_start], r8
    mov qword ptr [rip + in_len], 0
split_done:

    lea rcx, [rip + input_buf]
    lea r8, [rip + match_table]
    lea r9, [rip + bracket_stack]
    xor r10, r10
    mov r11, qword ptr [rip + prog_len]
    xor rax, rax
match_loop:
    cmp rax, r11
    je match_done
    movzx edx, byte ptr [rcx + rax]
    cmp dl, 0x5b
    je is_open
    cmp dl, 0x5d
    je is_close
    jmp match_next
is_open:
    mov dword ptr [r9 + r10*4], eax
    inc r10
    jmp match_next
is_close:
    cmp r10, 0
    je error_exit
    dec r10
    mov edx, dword ptr [r9 + r10*4]
    mov dword ptr [r8 + rdx*4], eax
    mov dword ptr [r8 + rax*4], edx
match_next:
    inc rax
    jmp match_loop
match_done:
    cmp r10, 0
    jne error_exit
    jmp run_program

error_exit:
    mov rax, 60
    mov rdi, 1
    syscall

run_program:
    lea r12, [rip + input_buf]
    lea r13, [rip + tape]
    lea r14, [rip + output_buf]
    lea r15, [rip + match_table]
    xor rbx, rbx
    xor rsi, rsi
    xor rdi, rdi
    xor rdx, rdx
    mov r8, qword ptr [rip + prog_len]
    mov r9, qword ptr [rip + in_len]
    mov r10, qword ptr [rip + in_start]
    lea r11, [r12 + r10]

exec_loop:
    cmp rsi, r8
    jge exec_done
    movzx eax, byte ptr [r12 + rsi]
    cmp al, 0x3e
    je op_gt
    cmp al, 0x3c
    je op_lt
    cmp al, 0x2b
    je op_plus
    cmp al, 0x2d
    je op_minus
    cmp al, 0x2e
    je op_dot
    cmp al, 0x2c
    je op_comma
    cmp al, 0x5b
    je op_open
    cmp al, 0x5d
    je op_close
    jmp op_next

op_gt:
    inc rbx
    jmp op_next
op_lt:
    dec rbx
    jmp op_next
op_plus:
    inc byte ptr [r13 + rbx]
    jmp op_next
op_minus:
    dec byte ptr [r13 + rbx]
    jmp op_next
op_dot:
    mov al, byte ptr [r13 + rbx]
    mov byte ptr [r14 + rdi], al
    inc rdi
    jmp op_next
op_comma:
    cmp rdx, r9
    jge comma_zero
    movzx eax, byte ptr [r11 + rdx]
    mov byte ptr [r13 + rbx], al
    inc rdx
    jmp op_next
comma_zero:
    mov byte ptr [r13 + rbx], 0
    jmp op_next
op_open:
    cmp byte ptr [r13 + rbx], 0
    jne op_next
    mov eax, dword ptr [r15 + rsi*4]
    mov rsi, rax
    jmp op_next
op_close:
    cmp byte ptr [r13 + rbx], 0
    je op_next
    mov eax, dword ptr [r15 + rsi*4]
    mov rsi, rax
    jmp op_next
op_next:
    inc rsi
    jmp exec_loop

exec_done:
    mov rdx, rdi
    mov rdi, 1
    lea rsi, [rip + output_buf]
    mov rax, 1
    syscall
    mov rax, 60
    mov rdi, 0
    syscall
```
