.globl _start
_start:
    // ---- Step 1: read all of stdin into input_buf ----
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf   // x19 = base of input_buf
    mov  x20, #0                     // x20 = total_len

read_loop:
    mov  x9, #65536
    sub  x9, x9, x20                 // x9 = remaining space
    cbz  x9, read_done
    mov  x8, #63                     // read
    mov  x0, #0                      // fd = 0
    add  x1, x19, x20                // buf = input_buf + total_len
    mov  x2, x9                      // count = remaining
    svc  #0
    cbz  x0, read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    // x20 = total_len

    // ---- Step 2: find first '!' ----
    mov  x21, #0                     // index i
find_loop:
    cmp  x21, x20
    b.ge not_found
    ldrb w0, [x19, x21]
    cmp  w0, #0x21
    b.eq found_bang
    add  x21, x21, #1
    b    find_loop

found_bang:
    mov  x22, x21                    // prog_len = index
    add  x23, x21, #1                // input_start = index + 1
    b    split_done

not_found:
    mov  x22, x20                    // prog_len = total_len
    mov  x23, x20                    // input_start = total_len (input_len = 0)

split_done:
    sub  x24, x20, x23               // input_len = total_len - input_start

    // ---- Step 3: parse brackets, build jump_match ----
    adrp x25, jump_match
    add  x25, x25, :lo12:jump_match  // base of jump_match array
    adrp x26, stack_buf
    add  x26, x26, :lo12:stack_buf   // base of bracket stack
    mov  x27, #0                     // stack_ptr
    mov  x28, #0                     // i

parse_loop:
    cmp  x28, x22
    b.ge parse_done
    ldrb w0, [x19, x28]
    cmp  w0, #0x5b                   // '['
    b.eq parse_open
    cmp  w0, #0x5d                   // ']'
    b.eq parse_close
    add  x28, x28, #1
    b    parse_loop

parse_open:
    str  w28, [x26, x27, lsl #2]
    add  x27, x27, #1
    add  x28, x28, #1
    b    parse_loop

parse_close:
    cmp  x27, #0
    b.eq unbalanced
    sub  x27, x27, #1
    ldr  w1, [x26, x27, lsl #2]      // j = matching '[' position
    str  w28, [x25, x1, lsl #2]      // jump_match[j] = i
    str  w1,  [x25, x28, lsl #2]     // jump_match[i] = j
    add  x28, x28, #1
    b    parse_loop

parse_done:
    cmp  x27, #0
    b.ne unbalanced
    b    execute

unbalanced:
    mov  x8, #93
    mov  x0, #1
    svc  #0
unbalanced_halt:
    b    unbalanced_halt

execute:
    adrp x11, tape
    add  x11, x11, :lo12:tape        // x11 = tape pointer (current cell)
    mov  x12, #0                     // pc
    mov  x13, #0                     // input_pos
    adrp x9, output_buf
    add  x9, x9, :lo12:output_buf    // x9 = output_buf base
    mov  x10, #0                     // out_pos

main_loop:
    cmp  x12, x22
    b.ge main_done
    ldrb w0, [x19, x12]
    cmp  w0, #0x3e                   // '>'
    b.eq op_gt
    cmp  w0, #0x3c                   // '<'
    b.eq op_lt
    cmp  w0, #0x2b                   // '+'
    b.eq op_plus
    cmp  w0, #0x2d                   // '-'
    b.eq op_minus
    cmp  w0, #0x2e                   // '.'
    b.eq op_dot
    cmp  w0, #0x2c                   // ','
    b.eq op_comma
    cmp  w0, #0x5b                   // '['
    b.eq op_lbracket
    cmp  w0, #0x5d                   // ']'
    b.eq op_rbracket
    add  x12, x12, #1
    b    main_loop

op_gt:
    add  x11, x11, #1
    add  x12, x12, #1
    b    main_loop

op_lt:
    sub  x11, x11, #1
    add  x12, x12, #1
    b    main_loop

op_plus:
    ldrb w0, [x11]
    add  w0, w0, #1
    strb w0, [x11]
    add  x12, x12, #1
    b    main_loop

op_minus:
    ldrb w0, [x11]
    sub  w0, w0, #1
    strb w0, [x11]
    add  x12, x12, #1
    b    main_loop

op_dot:
    ldrb w0, [x11]
    strb w0, [x9, x10]
    add  x10, x10, #1
    add  x12, x12, #1
    b    main_loop

op_comma:
    cmp  x13, x24
    b.ge comma_zero
    add  x1, x23, x13
    ldrb w0, [x19, x1]
    add  x13, x13, #1
    strb w0, [x11]
    add  x12, x12, #1
    b    main_loop
comma_zero:
    mov  w0, #0
    strb w0, [x11]
    add  x12, x12, #1
    b    main_loop

op_lbracket:
    ldrb w0, [x11]
    cmp  w0, #0
    b.ne lb_continue
    ldr  w1, [x25, x12, lsl #2]
    add  x12, x1, #1
    b    main_loop
lb_continue:
    add  x12, x12, #1
    b    main_loop

op_rbracket:
    ldrb w0, [x11]
    cmp  w0, #0
    b.eq rb_continue
    ldr  w1, [x25, x12, lsl #2]
    add  x12, x1, #1
    b    main_loop
rb_continue:
    add  x12, x12, #1
    b    main_loop

main_done:
    mov  x8, #64
    mov  x0, #1
    mov  x1, x9
    mov  x2, x10
    svc  #0
    mov  x8, #93
    mov  x0, #0
    svc  #0
end_halt:
    b    end_halt
