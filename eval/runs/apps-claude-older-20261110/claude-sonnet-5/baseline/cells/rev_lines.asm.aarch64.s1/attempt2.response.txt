.globl _start

.bss
    .align 3
buf:
    .space 65537

.text

_start:
    adrp x19, buf
    add  x19, x19, :lo12:buf
    mov  x20, #0            // total bytes read so far

read_loop:
    mov  x0, #0             // fd = stdin
    add  x1, x19, x20       // buf + offset
    mov  x2, #65537
    sub  x2, x2, x20        // remaining capacity
    mov  x8, #63            // read
    svc  #0
    cmp  x0, #0
    beq  read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    mov  x21, #0            // p = start of current line
    mov  x22, #0            // i = scan index

process_loop:
    cmp  x22, x20
    bge  process_after
    add  x9, x19, x22
    ldrb w10, [x9]
    cmp  w10, #10
    bne  process_continue
    // reverse segment [p, i-1]
    mov  x0, x21
    sub  x1, x22, #1
    bl   reverse_segment
    add  x21, x22, #1       // p = i + 1
process_continue:
    add  x22, x22, #1
    b    process_loop

process_after:
    cmp  x21, x20
    bge  write_output
    // final line without trailing newline
    mov  x0, x21
    sub  x1, x20, #1
    bl   reverse_segment
    // append newline
    add  x9, x19, x20
    mov  w10, #10
    strb w10, [x9]
    add  x20, x20, #1

write_output:
    mov  x23, #0            // written offset

write_loop:
    cmp  x23, x20
    bge  do_exit
    mov  x0, #1             // fd = stdout
    add  x1, x19, x23
    sub  x2, x20, x23
    mov  x8, #64            // write
    svc  #0
    add  x23, x23, x0
    b    write_loop

do_exit:
    mov  x0, #0
    mov  x8, #93            // exit
    svc  #0

reverse_segment:            // x0 = lo index, x1 = hi index (into buf), x19 = base
    cmp  x0, x1
    bge  reverse_done
    add  x2, x19, x0
    add  x3, x19, x1
    ldrb w4, [x2]
    ldrb w5, [x3]
    strb w5, [x2]
    strb w4, [x3]
    add  x0, x0, #1
    sub  x1, x1, #1
    b    reverse_segment
reverse_done:
    ret
