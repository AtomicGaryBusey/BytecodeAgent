```asm
.globl _start

.section .bss
.align 16
buffer: .space 65536
outbuf: .space 131072

.text

_start:
    lea rdi, [rip + buffer]
    xor r12, r12            # total bytes read

read_loop:
    lea rdi, [rip + buffer]
    add rdi, r12
    mov rax, 65536
    sub rax, r12
    jz process               # buffer full, stop reading (shouldn't happen per limits)
    mov rsi, rdi
    mov rdi, 0
    mov rdx, rax
    mov rax, 0
    syscall
    test rax, rax
    jz process
    add r12, rax
    jmp read_loop

process:
    # r12 = total length in buffer
    lea rsi, [rip + buffer]      # start of input
    xor r13, r13                 # read index
    lea r14, [rip + outbuf]      # write pointer
    xor r15, r15                 # output length

line_loop:
    cmp r13, r12
    jge done_process

    # find end of line (newline) starting at r13, within [0, r12)
    mov rbx, r13      # line start
    mov rcx, r13      # scan index

find_nl:
    cmp rcx, r12
    jge no_nl_found
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 10
    je found_nl
    inc rcx
    jmp find_nl

no_nl_found:
    # line from rbx to r12 (exclusive), no newline present, treat as final line
    mov rdx, r12
    jmp emit_line

found_nl:
    mov rdx, rcx      # line end (exclusive), rcx points to newline

emit_line:
    # write bytes from rdx-1 down to rbx into outbuf at r14, then newline
    mov rax, rdx
    sub rax, rbx     # line length
    test rax, rax
    jz emit_nl_only

    mov rdi, rdx
    dec rdi          # last byte index of line

emit_byte_loop:
    movzx eax, byte ptr [rsi + rdi]
    mov byte ptr [r14], al
    inc r14
    inc r15
    cmp rdi, rbx
    je emit_nl_only
    dec rdi
    jmp emit_byte_loop

emit_nl_only:
    mov byte ptr [r14], 10
    inc r14
    inc r15

    # advance r13 past this line
    cmp rdx, r12
    jl have_newline_boundary
    mov r13, r12
    jmp line_loop

have_newline_boundary:
    mov r13, rdx
    inc r13
    jmp line_loop

done_process:
    # write outbuf of length r15 to stdout
    xor r8, r8
write_loop:
    cmp r8, r15
    jge finish
    lea rdi, [rip + outbuf]
    add rdi, r8
    mov rsi, rdi
    mov rdi, 1
    mov rdx, r15
    sub rdx, r8
    mov rax, 1
    syscall
    add r8, rax
    jmp write_loop

finish:
    mov rax, 60
    xor rdi, rdi
    syscall
```
