.globl _start

.equ BUFSIZE, 65536

.section .bss
.align 3
buf:
    .zero BUFSIZE
.align 3
out_buf:
    .zero BUFSIZE+1

.text
_start:
    adrp x9, buf
    add  x9, x9, :lo12:buf
    adrp x10, out_buf
    add  x10, x10, :lo12:out_buf

    mov x19, #0        // total bytes read

read_loop:
    mov x16, #BUFSIZE
    cmp x19, x16
    b.ge read_done
    add x1, x9, x19
    sub x2, x16, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    b read_loop
read_done:

    mov x20, #0        // scan index
    mov x21, #0        // line start
    mov x22, #0        // out write index

main_loop:
    cmp x20, x19
    b.eq after_loop
    ldrb w3, [x9, x20]
    cmp w3, #0x0A
    b.ne not_newline

    // reverse copy [x21, x20) to out_buf, then newline
    mov x23, x20
    mov x24, x21
rev_loop:
    cmp x24, x23
    b.ge rev_done
    sub x25, x23, #1
    ldrb w3, [x9, x25]
    strb w3, [x10, x22]
    add x22, x22, #1
    mov x23, x25
    b rev_loop
rev_done:
    mov w3, #0x0A
    strb w3, [x10, x22]
    add x22, x22, #1
    add x20, x20, #1
    mov x21, x20
    b main_loop

not_newline:
    add x20, x20, #1
    b main_loop

after_loop:
    cmp x21, x19
    b.ge write_output
    mov x23, x19
    mov x24, x21
fin_rev_loop:
    cmp x24, x23
    b.ge fin_done
    sub x25, x23, #1
    ldrb w3, [x9, x25]
    strb w3, [x10, x22]
    add x22, x22, #1
    mov x23, x25
    b fin_rev_loop
fin_done:
    mov w3, #0x0A
    strb w3, [x10, x22]
    add x22, x22, #1

write_output:
    mov x0, #1
    mov x1, x10
    mov x2, x22
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0
