.bss
input_buf:      .space 65536
tape:           .space 30000
output_buf:     .space 65536
match_table:    .space 524288
bracket_stack:  .space 8000

.text
.globl _start
_start:
    # ---- read all of stdin into input_buf ----
    xor r8, r8                       # r8 = total bytes read so far

read_loop:
    mov rax, 65536
    sub rax, r8
    jle read_done
    mov rdx, rax                     # count
    lea rsi, [rip + input_buf]
    add rsi, r8                      # buf = input_buf + r8
    xor rdi, rdi                      # fd = 0
    mov rax, 0                        # sys_read
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:
    # r8 = total_len

    # ---- find first '!' ----
    lea rsi, [rip + input_buf]
    xor rcx, rcx
find_loop:
    cmp rcx, r8
    jge not_found
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x21
    je found
    inc rcx
    jmp find_loop

found:
    mov r9, rcx                       # prog_len
    lea r10, [rsi + rcx + 1]          # input_start
    mov r11, r8
    sub r11, rcx
    dec r11                           # input_len
    jmp have_prog

not_found:
    mov r9, r8                        # prog_len = total_len
    lea r10, [rsi + r8]                # input_start (unused, len 0)
    xor r11, r11                       # input_len = 0

have_prog:
    # r9 = prog_len, r10 = input_start addr, r11 = input_len, rsi = input_buf base

    # ---- bracket balance check ----
    lea r12, [rip + bracket_stack]    # bracket_stack_base
    mov rdi, r12                       # stack top pointer
    xor rcx, rcx                       # position index

check_loop:
    cmp rcx, r9
    jge check_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, '['
    je is_open
    cmp al, ']'
    je is_close
    jmp check_next

is_open:
    lea rdx, [rsi + rcx]
    mov [rdi], rdx
    add rdi, 8
    jmp check_next

is_close:
    cmp rdi, r12
    je unbalanced_exit
    sub rdi, 8
    mov rdx, [rdi]                     # address of matching '['
    lea r13, [rsi + rcx]                # address of this ']'
    lea r14, [rip + match_table]
    mov rax, rdx
    sub rax, rsi
    lea r15, [r14 + rax*8]
    mov [r15], r13
    mov rax, r13
    sub rax, rsi
    lea r15, [r14 + rax*8]
    mov [r15], rdx
    jmp check_next

check_next:
    inc rcx
    jmp check_loop

check_done:
    cmp rdi, r12
    jne unbalanced_exit
    jmp run_program

unbalanced_exit:
    mov edi, 1
    mov eax, 231
    syscall

run_program:
    mov r12, rsi                       # input_buf_base
    lea rbx, [rsi + r9]                 # prog_end
    mov r8, rsi                         # pc
    lea r9, [rip + tape]                 # tape_ptr
    mov r13, r10                         # input_cursor
    mov r14, r11                         # input_remaining
    lea r15, [rip + output_buf]          # output_cursor

main_loop:
    cmp r8, rbx
    jge exec_done
    movzx eax, byte ptr [r8]
    cmp al, '>'
    je op_gt
    cmp al, '<'
    je op_lt
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '.'
    je op_dot
    cmp al, ','
    je op_comma
    cmp al, '['
    je op_open
    cmp al, ']'
    je op_close
    jmp next_pc

op_gt:
    inc r9
    jmp next_pc
op_lt:
    dec r9
    jmp next_pc
op_plus:
    inc byte ptr [r9]
    jmp next_pc
op_minus:
    dec byte ptr [r9]
    jmp next_pc
op_dot:
    mov al, byte ptr [r9]
    mov byte ptr [r15], al
    inc r15
    jmp next_pc
op_comma:
    cmp r14, 0
    jle comma_zero
    mov al, byte ptr [r13]
    inc r13
    dec r14
    mov byte ptr [r9], al
    jmp next_pc
comma_zero:
    mov byte ptr [r9], 0
    jmp next_pc
op_open:
    cmp byte ptr [r9], 0
    jne next_pc
    mov rax, r8
    sub rax, r12
    lea rdx, [rip + match_table]
    mov r8, [rdx + rax*8]
    jmp next_pc
op_close:
    cmp byte ptr [r9], 0
    je next_pc
    mov rax, r8
    sub rax, r12
    lea rdx, [rip + match_table]
    mov r8, [rdx + rax*8]
    jmp next_pc
next_pc:
    inc r8
    jmp main_loop

exec_done:
    lea rsi, [rip + output_buf]
    mov rbp, r15
    sub rbp, rsi                       # rbp = remaining bytes to write

write_loop:
    cmp rbp, 0
    jle write_done
    mov rdx, rbp
    mov rdi, 1
    mov rax, 1
    syscall
    add rsi, rax
    sub rbp, rax
    jmp write_loop

write_done:
    mov edi, 0
    mov eax, 231
    syscall
