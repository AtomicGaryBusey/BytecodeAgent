```asm
.globl _start

.section .rodata
nl_byte:
    .byte 0x0A

.bss
.align 3
buf:
    .zero 65536

.text
_start:
    mov x19, #0              // total bytes read
    adrp x20, buf
    add x20, x20, :lo12:buf

read_loop:
    cmp x19, #65536
    bge read_done

    mov x0, #0                // fd = 0 (stdin)
    add x1, x20, x19          // buf + total
    mov x2, #65536
    sub x2, x2, x19           // remaining capacity
    mov x8, #63                // read
    svc #0

    cmp x0, #0
    beq read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, #0               // start index

process_loop:
    cmp x21, x19
    bge all_done

    mov x22, x21               // scan index
    mov x26, #0                // found-newline flag

find_nl:
    cmp x22, x19
    bge no_nl_found
    add x0, x20, x22
    ldrb w1, [x0]
    cmp w1, #0x0A
    beq nl_found
    add x22, x22, #1
    b find_nl

nl_found:
    mov x23, x22
    mov x26, #1
    b reverse_line

no_nl_found:
    mov x23, x19
    mov x26, #0

reverse_line:
    mov x24, x21
    sub x25, x23, #1

rev_loop:
    cmp x24, x25
    bge rev_done
    add x0, x20, x24
    add x1, x20, x25
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w3, [x0]
    strb w2, [x1]
    add x24, x24, #1
    sub x25, x25, #1
    b rev_loop

rev_done:
    sub x2, x23, x21
    cbz x2, skip_write
    mov x0, #1
    add x1, x20, x21

write_retry:
    mov x8, #64                // write
    svc #0
    cmp x0, x2
    beq skip_write
    add x1, x1, x0
    sub x2, x2, x0
    b write_retry

skip_write:
    adrp x5, nl_byte
    add x5, x5, :lo12:nl_byte
    mov x0, #1
    mov x1, x5
    mov x2, #1

write_nl_retry:
    mov x8, #64                // write
    svc #0
    cmp x0, #1
    beq nl_written
    b write_nl_retry

nl_written:
    cbnz x26, continue_with_newline
    mov x21, x19
    b process_loop

continue_with_newline:
    mov x21, x23
    add x21, x21, #1
    b process_loop

all_done:
    mov x0, #0
    mov x8, #93                // exit
    svc #0
```
