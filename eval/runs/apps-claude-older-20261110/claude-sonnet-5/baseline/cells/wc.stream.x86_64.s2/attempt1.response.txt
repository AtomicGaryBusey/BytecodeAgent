```
.globl _start

.section .rodata
space_char:
    .ascii " "
newline_char:
    .ascii "\n"

.section .bss
.align 16
buffer:
    .zero 65536
numbuf:
    .zero 24

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor rax, rax
    xor rdi, rdi
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    test rax, rax
    jz done_read
    mov rbx, rax
    add r14, rbx
    lea rsi, [rip + buffer]
    xor rcx, rcx
process_loop:
    cmp rcx, rbx
    jge read_loop
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x0A
    jne not_newline
    inc r12
not_newline:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jl not_ws
    cmp al, 0x0D
    jg not_ws
is_ws:
    xor r15, r15
    jmp next_char
not_ws:
    test r15, r15
    jnz next_char
    inc r13
    mov r15, 1
next_char:
    inc rcx
    jmp process_loop

done_read:
    mov rax, r12
    call to_decimal
    mov rdi, 1
    mov rdx, rcx
    mov rax, 1
    syscall

    lea rsi, [rip + space_char]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall

    mov rax, r13
    call to_decimal
    mov rdi, 1
    mov rdx, rcx
    mov rax, 1
    syscall

    lea rsi, [rip + space_char]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall

    mov rax, r14
    call to_decimal
    mov rdi, 1
    mov rdx, rcx
    mov rax, 1
    syscall

    lea rsi, [rip + newline_char]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

to_decimal:
    lea rdi, [rip + numbuf + 23]
    xor rcx, rcx
td_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc rcx
    test rax, rax
    jnz td_loop
    mov rsi, rdi
    ret
```
