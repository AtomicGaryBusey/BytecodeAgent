```asm
.text
.globl _start

_start:
    lea r14, [rip + buf]
    xor r15, r15
rd_loop:
    mov rdx, 65600
    sub rdx, r15
    cmp rdx, 0
    jle rd_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [r14 + r15]
    syscall
    cmp rax, 0
    jle rd_done
    add r15, rax
    jmp rd_loop
rd_done:
    mov r13, r15
    xor r12, r12
    call ws
    call parse_value
    call ws
    cmp r12, r13
    jne fail
    lea rsi, [rip + msg_ok]
    mov rdx, 3
    call write_buf
    mov edi, 0
    jmp do_exit

fail:
    lea rsi, [rip + msg_error]
    mov rdx, 9
    call write_buf
    call to_decimal
    call write_buf
    lea rsi, [rip + nl]
    mov rdx, 1
    call write_buf
    mov edi, 1
    jmp do_exit

do_exit:
    mov eax, 60
    syscall

ws:
ws_loop:
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x20
    je ws_inc
    cmp al, 0x09
    je ws_inc
    cmp al, 0x0A
    je ws_inc
    cmp al, 0x0D
    je ws_inc
    ret
ws_inc:
    inc r12
    jmp ws_loop

parse_value:
    movzx eax, byte ptr [r14+r12]
    cmp al, '{'
    je pv_call_object
    cmp al, '['
    je pv_call_array
    cmp al, 0x22
    je pv_call_string
    cmp al, 't'
    je pv_true
    cmp al, 'f'
    je pv_false
    cmp al, 'n'
    je pv_null
    cmp al, '-'
    je pv_call_number
    cmp al, '0'
    jb pv_invalid
    cmp al, '9'
    ja pv_invalid
    jmp pv_call_number
pv_invalid:
    jmp fail
pv_call_object:
    jmp parse_object
pv_call_array:
    jmp parse_array
pv_call_string:
    jmp parse_string
pv_call_number:
    jmp parse_number
pv_true:
    lea rsi, [rip + lit_true]
    mov edx, 4
    jmp match_literal
pv_false:
    lea rsi, [rip + lit_false]
    mov edx, 5
    jmp match_literal
pv_null:
    lea rsi, [rip + lit_null]
    mov edx, 4
    jmp match_literal

parse_object:
    inc r12
    call ws
    movzx eax, byte ptr [r14+r12]
    cmp al, '}'
    jne po_member
    inc r12
    ret
po_member:
    call parse_member
po_loop:
    call ws
    movzx eax, byte ptr [r14+r12]
    cmp al, '}'
    je po_close
    cmp al, ','
    jne po_err
    inc r12
    call ws
    call parse_member
    jmp po_loop
po_close:
    inc r12
    ret
po_err:
    jmp fail

parse_member:
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x22
    jne pm_err
    call parse_string
    call ws
    movzx eax, byte ptr [r14+r12]
    cmp al, ':'
    jne pm_err
    inc r12
    call ws
    call parse_value
    ret
pm_err:
    jmp fail

parse_array:
    inc r12
    call ws
    movzx eax, byte ptr [r14+r12]
    cmp al, ']'
    jne pa_value
    inc r12
    ret
pa_value:
    call parse_value
pa_loop:
    call ws
    movzx eax, byte ptr [r14+r12]
    cmp al, ']'
    je pa_close
    cmp al, ','
    jne pa_err
    inc r12
    call ws
    call parse_value
    jmp pa_loop
pa_close:
    inc r12
    ret
pa_err:
    jmp fail

parse_string:
    inc r12
ps_loop:
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x22
    je ps_close
    cmp al, 0x5C
    je ps_escape
    cmp al, 0x20
    jb ps_err
    inc r12
    jmp ps_loop
ps_close:
    inc r12
    ret
ps_escape:
    inc r12
    movzx eax, byte ptr [r14+r12]
    cmp al, 0x22
    je ps_esc_simple
    cmp al, 0x5C
    je ps_esc_simple
    cmp al, '/'
    je ps_esc_simple
    cmp al, 'b'
    je ps_esc_simple
    cmp al, 'f'
    je ps_esc_simple
    cmp al, 'n'
    je ps_esc_simple
    cmp al, 'r'
    je ps_esc_simple
    cmp al, 't'
    je ps_esc_simple
    cmp al, 'u'
    je ps_esc_u
    jmp ps_err
ps_esc_simple:
    inc r12
    jmp ps_loop
ps_esc_u:
    inc r12
    mov ecx, 4
ps_u_loop:
    test ecx, ecx
    jz ps_loop
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    jb ps_err
    cmp al, '9'
    jbe ps_u_ok
    cmp al, 'A'
    jb ps_err
    cmp al, 'F'
    jbe ps_u_ok
    cmp al, 'a'
    jb ps_err
    cmp al, 'f'
    ja ps_err
ps_u_ok:
    inc r12
    dec ecx
    jmp ps_u_loop
ps_err:
    jmp fail

parse_number:
    movzx eax, byte ptr [r14+r12]
    cmp al, '-'
    jne pn_int
    inc r12
pn_int:
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    je pn_zero
    cmp al, '1'
    jb pn_err
    cmp al, '9'
    ja pn_err
    inc r12
pn_int_loop:
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    jb pn_frac
    cmp al, '9'
    ja pn_frac
    inc r12
    jmp pn_int_loop
pn_zero:
    inc r12
    jmp pn_frac
pn_err:
    jmp fail
pn_frac:
    movzx eax, byte ptr [r14+r12]
    cmp al, '.'
    jne pn_exp
    inc r12
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    jb pn_err
    cmp al, '9'
    ja pn_err
    inc r12
pn_frac_loop:
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    jb pn_exp
    cmp al, '9'
    ja pn_exp
    inc r12
    jmp pn_frac_loop
pn_exp:
    movzx eax, byte ptr [r14+r12]
    cmp al, 'e'
    je pn_exp_go
    cmp al, 'E'
    je pn_exp_go
    ret
pn_exp_go:
    inc r12
    movzx eax, byte ptr [r14+r12]
    cmp al, '+'
    je pn_exp_sign
    cmp al, '-'
    je pn_exp_sign
    jmp pn_exp_digit_check
pn_exp_sign:
    inc r12
pn_exp_digit_check:
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    jb pn_err
    cmp al, '9'
    ja pn_err
    inc r12
pn_exp_loop:
    movzx eax, byte ptr [r14+r12]
    cmp al, '0'
    jb pn_done
    cmp al, '9'
    ja pn_done
    inc r12
    jmp pn_exp_loop
pn_done:
    ret

match_literal:
    lea r10, [r14+r12]
    xor ecx, ecx
ml_loop:
    cmp ecx, edx
    je ml_done
    movzx eax, byte ptr [r10+rcx]
    movzx ebx, byte ptr [rsi+rcx]
    cmp al, bl
    jne ml_fail
    inc ecx
    jmp ml_loop
ml_done:
    add r12, rdx
    ret
ml_fail:
    add r12, rcx
    jmp fail

write_buf:
wb_loop:
    test rdx, rdx
    jz wb_done
    mov rax, 1
    mov rdi, 1
    syscall
    cmp rax, 0
    jle wb_done
    add rsi, rax
    sub rdx, rax
    jmp wb_loop
wb_done:
    ret

to_decimal:
    lea rdi, [rip + numbuf + 23]
    mov rax, r12
    xor r9, r9
    test rax, rax
    jnz td_loop
    mov byte ptr [rdi], '0'
    dec rdi
    inc r9
    jmp td_done
td_loop:
    test rax, rax
    jz td_done
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, '0'
    mov byte ptr [rdi], dl
    dec rdi
    inc r9
    jmp td_loop
td_done:
    inc rdi
    mov rsi, rdi
    mov rdx, r9
    ret

.section .rodata
msg_error: .ascii "error at "
msg_ok: .ascii "ok\n"
nl: .ascii "\n"
lit_true: .ascii "true"
lit_false: .ascii "false"
lit_null: .ascii "null"

.bss
buf: .zero 65600
numbuf: .zero 24
```
