.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
    .align 16
inbuf:
    .space 65536
outbuf:
    .space 128

.text
.globl _start

_start:
    xor r12, r12
read_loop:
    cmp r12, 65536
    jge read_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    xor rdi, rdi
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:

    xor rbx, rbx
process_loop:
    cmp rbx, r12
    jge final_line

    mov r13, r12
    sub r13, rbx
    cmp r13, 16
    jle n_done
    mov r13, 16
n_done:

    lea rdi, [rip + outbuf]
    mov eax, ebx
    call write_hex32

    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi

    xor r8, r8
byte_loop:
    cmp r8, 16
    jge byte_loop_done
    cmp r8, r13
    jl have_byte
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3
    jmp byte_loop_tail
have_byte:
    lea rsi, [rip + inbuf]
    add rsi, rbx
    movzx eax, byte ptr [rsi + r8]
    call write_hex8
    mov byte ptr [rdi], ' '
    inc rdi
byte_loop_tail:
    cmp r8, 7
    jne byte_loop_next
    mov byte ptr [rdi], ' '
    inc rdi
byte_loop_next:
    inc r8
    jmp byte_loop
byte_loop_done:

    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor r8, r8
char_loop:
    cmp r8, r13
    jge char_loop_done
    lea rsi, [rip + inbuf]
    add rsi, rbx
    movzx eax, byte ptr [rsi + r8]
    cmp eax, 0x20
    jl char_dot
    cmp eax, 0x7e
    jg char_dot
    jmp char_store
char_dot:
    mov eax, '.'
char_store:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    jmp char_loop
char_loop_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rax
    mov rsi, rax
    mov rdi, 1
    mov rax, 1
    syscall

    add rbx, r13
    jmp process_loop

final_line:
    lea rdi, [rip + outbuf]
    mov eax, r12d
    call write_hex32
    mov byte ptr [rdi], 10
    inc rdi

    lea rax, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rax
    mov rsi, rax
    mov rdi, 1
    mov rax, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

write_hex32:
    mov ecx, 32
whex32_loop:
    sub ecx, 4
    mov edx, eax
    shr edx, cl
    and edx, 0xf
    lea r10, [rip + hexdigits]
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    test ecx, ecx
    jnz whex32_loop
    ret

write_hex8:
    mov edx, eax
    shr edx, 4
    and edx, 0xf
    lea r10, [rip + hexdigits]
    movzx ecx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], cl
    inc rdi
    mov edx, eax
    and edx, 0xf
    movzx ecx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], cl
    inc rdi
    ret
