.section .bss
.align 8
var_proglen:   .space 8
var_instart:   .space 8
var_totallen:  .space 8

.align 8
buf_input: .space 65536

.align 4
arr_match: .space 262144

.align 4
arr_bstack: .space 4000

.align 8
arr_tape: .space 30000

.align 8
arr_outbuf: .space 65536

.text
.globl _start

_start:
    # ---- read all of stdin into buf_input, up to 65536 bytes ----
    xor r12, r12
read_loop:
    mov rdi, 0
    lea rsi, [rip + buf_input]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    mov rax, 0
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jae read_done
    jmp read_loop
read_done:
    mov qword ptr [rip + var_totallen], r12

    # ---- find first '!' (0x21) ----
    xor r13, r13
    lea rbx, [rip + buf_input]
find_bang:
    cmp r13, r12
    jae no_bang
    movzx eax, byte ptr [rbx + r13]
    cmp eax, 0x21
    je found_bang
    inc r13
    jmp find_bang
found_bang:
    mov r15, r13
    inc r13
    jmp have_split
no_bang:
    mov r15, r12
    mov r13, r12
have_split:
    mov qword ptr [rip + var_proglen], r15
    mov qword ptr [rip + var_instart], r13

    # ---- bracket matching / validation ----
    xor rcx, rcx            # i
    xor rdx, rdx            # stack depth
    lea rbx, [rip + buf_input]
    lea rsi, [rip + arr_match]
    lea rdi, [rip + arr_bstack]
match_loop:
    cmp rcx, qword ptr [rip + var_proglen]
    jae match_done
    movzx eax, byte ptr [rbx + rcx]
    cmp eax, 0x5B
    je is_open
    cmp eax, 0x5D
    je is_close
    jmp match_next
is_open:
    mov dword ptr [rdi + rdx*4], ecx
    inc rdx
    jmp match_next
is_close:
    cmp rdx, 0
    je do_invalid
    dec rdx
    mov eax, dword ptr [rdi + rdx*4]
    mov dword ptr [rsi + rcx*4], eax
    mov dword ptr [rsi + rax*4], ecx
match_next:
    inc rcx
    jmp match_loop
match_done:
    cmp rdx, 0
    jne do_invalid
    jmp run_program

do_invalid:
    mov rdi, 1
    mov rax, 60
    syscall

run_program:
    xor r8, r8                      # ip
    xor r9, r9                      # dp
    mov r10, qword ptr [rip + var_instart]   # inpos (absolute index into buf_input)
    xor r11, r11                    # outpos

    lea rbx, [rip + buf_input]      # program bytes base
    lea rsi, [rip + arr_match]
    lea rdi, [rip + arr_tape]
    lea rdx, [rip + arr_outbuf]

exec_loop:
    cmp r8, qword ptr [rip + var_proglen]
    jae exec_done
    movzx eax, byte ptr [rbx + r8]
    cmp eax, 0x3E
    je op_gt
    cmp eax, 0x3C
    je op_lt
    cmp eax, 0x2B
    je op_plus
    cmp eax, 0x2D
    je op_minus
    cmp eax, 0x2E
    je op_dot
    cmp eax, 0x2C
    je op_comma
    cmp eax, 0x5B
    je op_open
    cmp eax, 0x5D
    je op_close
    jmp op_next

op_gt:
    inc r9
    jmp op_next
op_lt:
    dec r9
    jmp op_next
op_plus:
    inc byte ptr [rdi + r9]
    jmp op_next
op_minus:
    dec byte ptr [rdi + r9]
    jmp op_next
op_dot:
    movzx eax, byte ptr [rdi + r9]
    mov byte ptr [rdx + r11], al
    inc r11
    jmp op_next
op_comma:
    cmp r10, qword ptr [rip + var_totallen]
    jae comma_zero
    movzx eax, byte ptr [rbx + r10]
    mov byte ptr [rdi + r9], al
    inc r10
    jmp op_next
comma_zero:
    mov byte ptr [rdi + r9], 0
    jmp op_next
op_open:
    cmp byte ptr [rdi + r9], 0
    jne op_next
    mov eax, dword ptr [rsi + r8*4]
    mov r8, rax
    jmp op_next
op_close:
    cmp byte ptr [rdi + r9], 0
    je op_next
    mov eax, dword ptr [rsi + r8*4]
    mov r8, rax
    jmp op_next
op_next:
    inc r8
    jmp exec_loop

exec_done:
    xor r12, r12
write_loop:
    cmp r12, r11
    jae write_done
    mov rdi, 1
    lea rsi, [rip + arr_outbuf]
    add rsi, r12
    mov rdx, r11
    sub rdx, r12
    mov rax, 1
    syscall
    add r12, rax
    jmp write_loop
write_done:
    mov rdi, 0
    mov rax, 60
    syscall
