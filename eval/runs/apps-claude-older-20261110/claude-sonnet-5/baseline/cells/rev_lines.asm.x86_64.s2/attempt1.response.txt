.globl _start

.section .bss
.align 16
buffer: .space 65536

.text

_start:
    lea rdi, [rip + buffer]
    xor r12, r12

read_loop:
    lea rsi, [rdi + r12]
    mov rax, 65536
    sub rax, r12
    mov rdx, rax
    mov rax, 0
    mov rdi, 0
    syscall
    test rax, rax
    jle process
    add r12, rax
    cmp r12, 65536
    jb read_loop

process:
    # r12 = total bytes in buffer
    lea rbx, [rip + buffer]
    xor r13, r13        # current position (start of line)

line_loop:
    cmp r13, r12
    jae done

    # find newline starting at r13
    mov rsi, r13
find_nl:
    cmp rsi, r12
    jae no_nl_found
    movzx eax, byte ptr [rbx + rsi]
    cmp al, 10
    je found_nl
    inc rsi
    jmp find_nl

no_nl_found:
    # final line without newline, line is [r13, r12)
    mov r14, r13
    mov r15, r12
    call print_reversed
    call print_newline
    mov r13, r12
    jmp line_loop

found_nl:
    # line is [r13, rsi)
    mov r14, r13
    mov r15, rsi
    call print_reversed
    call print_newline
    lea r13, [rsi + 1]
    jmp line_loop

done:
    mov rax, 60
    xor rdi, rdi
    syscall

# print_reversed: print bytes from r14 to r15 (exclusive) in reverse order
print_reversed:
    push rax
    push rcx
    push rdx
    push rsi
    push rdi
    push r8
    push r9

    mov r8, r15
    mov r9, r14

rev_loop:
    cmp r8, r9
    jle rev_done
    dec r8
    lea rsi, [rbx + r8]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall
    jmp rev_loop

rev_done:
    pop r9
    pop r8
    pop rdi
    pop rsi
    pop rdx
    pop rcx
    pop rax
    ret

print_newline:
    push rax
    push rcx
    push rdx
    push rsi
    push rdi

    lea rsi, [rip + nl_byte]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    pop rdi
    pop rsi
    pop rdx
    pop rcx
    pop rax
    ret

.section .rodata
nl_byte: .byte 10
