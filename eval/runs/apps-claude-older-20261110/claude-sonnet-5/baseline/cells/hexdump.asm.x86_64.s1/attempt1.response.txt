.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .space 65536
output_buf:
    .space 400000

.text
.globl _start

write_hex32:
    # in: r9d = value (32-bit), rdi = dest pointer
    # out: rdi advanced by 8 (8 hex digits written)
    mov ecx, 28
write_hex32_loop:
    mov eax, r9d
    shr eax, cl
    and eax, 0xf
    lea rdx, [rip + hexdigits]
    movzx eax, byte ptr [rdx + rax]
    mov [rdi], al
    inc rdi
    sub ecx, 4
    jge write_hex32_loop
    ret

write_hex_byte:
    # in: r9d = byte value (0-255), rdi = dest pointer
    # out: rdi advanced by 2 (2 hex digits written)
    mov ecx, 4
write_hex_byte_loop:
    mov eax, r9d
    shr eax, cl
    and eax, 0xf
    lea rdx, [rip + hexdigits]
    movzx eax, byte ptr [rdx + rax]
    mov [rdi], al
    inc rdi
    sub ecx, 4
    jge write_hex_byte_loop
    ret

_start:
    xor r12, r12            # r12 = total_len read so far

read_loop:
    mov rax, 65536
    sub rax, r12
    jz read_done
    mov rdx, rax
    lea rsi, [rip + input_buf]
    add rsi, r12
    xor eax, eax
    mov edi, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:

    lea r13, [rip + output_buf]   # r13 = output write pointer
    xor r14, r14                   # r14 = processed offset

row_loop:
    cmp r14, r12
    jae rows_done

    mov rax, r12
    sub rax, r14
    cmp rax, 16
    jbe set_n
    mov rax, 16
set_n:
    mov r15, rax    # r15 = n for this row

    mov r9d, r14d
    mov rdi, r13
    call write_hex32
    mov r13, rdi

    mov word ptr [r13], 0x2020
    add r13, 2

    xor rbx, rbx
col_loop:
    cmp rbx, r15
    jae col_pad
    mov rax, r14
    add rax, rbx
    lea rdx, [rip + input_buf]
    movzx r9d, byte ptr [rdx + rax]
    mov rdi, r13
    call write_hex_byte
    mov r13, rdi
    mov byte ptr [r13], 0x20
    inc r13
    jmp col_after
col_pad:
    mov byte ptr [r13], 0x20
    mov byte ptr [r13+1], 0x20
    mov byte ptr [r13+2], 0x20
    add r13, 3
col_after:
    cmp rbx, 7
    jne col_next
    mov byte ptr [r13], 0x20
    inc r13
col_next:
    inc rbx
    cmp rbx, 16
    jl col_loop

    mov byte ptr [r13], 0x20
    mov byte ptr [r13+1], 0x7c
    add r13, 2

    xor rbx, rbx
char_loop:
    cmp rbx, r15
    jae char_done
    mov rax, r14
    add rax, rbx
    lea rdx, [rip + input_buf]
    movzx eax, byte ptr [rdx + rax]
    cmp eax, 0x20
    jb char_dot
    cmp eax, 0x7e
    jbe char_store
char_dot:
    mov eax, 0x2e
char_store:
    mov byte ptr [r13], al
    inc r13
    inc rbx
    jmp char_loop
char_done:
    mov byte ptr [r13], 0x7c
    mov byte ptr [r13+1], 0x0a
    add r13, 2

    add r14, r15
    jmp row_loop

rows_done:
    mov r9d, r12d
    mov rdi, r13
    call write_hex32
    mov r13, rdi
    mov byte ptr [r13], 0x0a
    inc r13

    lea rsi, [rip + output_buf]
    mov rdx, r13
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_prog
    mov rax, 1
    mov rdi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_prog:
    mov eax, 231
    xor edi, edi
    syscall
