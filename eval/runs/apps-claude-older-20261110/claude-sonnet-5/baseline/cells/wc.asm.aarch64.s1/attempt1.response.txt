.text
.globl _start

_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0          // total bytes read so far
    mov  x21, #65536      // capacity

read_loop:
    sub  x2, x21, x20
    cbz  x2, read_done
    add  x1, x19, x20
    mov  x0, #0
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    beq  read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    mov  x22, #0          // lines
    mov  x23, #0          // words
    mov  x24, #0          // in_word flag
    mov  x25, #0          // index

proc_loop:
    cmp  x25, x20
    bge  proc_done
    ldrb w0, [x19, x25]
    cmp  w0, #0x20
    beq  is_ws
    cmp  w0, #0x09
    blt  not_ws
    cmp  w0, #0x0D
    bgt  not_ws
    b    is_ws

not_ws:
    cmp  x24, #0
    bne  skip_inc_word
    add  x23, x23, #1
    mov  x24, #1
skip_inc_word:
    b    proc_next

is_ws:
    cmp  w0, #0x0A
    bne  not_newline
    add  x22, x22, #1
not_newline:
    mov  x24, #0

proc_next:
    add  x25, x25, #1
    b    proc_loop

proc_done:
    adrp x9, outbuf
    add  x9, x9, :lo12:outbuf
    add  x9, x9, #64       // x9 = end pointer of outbuf

    sub  x9, x9, #1
    mov  w0, #10
    strb w0, [x9]

    mov  x0, x20
    mov  x1, x9
    bl   num_to_buf
    mov  x9, x0

    sub  x9, x9, #1
    mov  w0, #0x20
    strb w0, [x9]

    mov  x0, x23
    mov  x1, x9
    bl   num_to_buf
    mov  x9, x0

    sub  x9, x9, #1
    mov  w0, #0x20
    strb w0, [x9]

    mov  x0, x22
    mov  x1, x9
    bl   num_to_buf
    mov  x9, x0

    adrp x10, outbuf
    add  x10, x10, :lo12:outbuf
    add  x10, x10, #64

    sub  x2, x10, x9       // length
    mov  x1, x9
    mov  x0, #1
    mov  x8, #64
    svc  #0

    mov  x0, #0
    mov  x8, #93
    svc  #0

num_to_buf:
    mov  x3, x1
    cbnz x0, nb_loop
    sub  x3, x3, #1
    mov  w4, #48
    strb w4, [x3]
    b    nb_done
nb_loop:
    mov  x5, #10
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add  w7, w7, #48
    sub  x3, x3, #1
    strb w7, [x3]
    mov  x0, x6
    cbnz x0, nb_loop
nb_done:
    mov  x0, x3
    ret

.bss
.align 3
inbuf:
    .zero 65536
outbuf:
    .zero 64
