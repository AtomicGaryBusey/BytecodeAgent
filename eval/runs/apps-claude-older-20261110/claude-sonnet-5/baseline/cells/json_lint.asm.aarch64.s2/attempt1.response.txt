_start:
    mov     x19, #0                 // total bytes read

read_loop:
    mov     x2, #65536
    subs    x2, x2, x19
    b.le    read_done
    adrp    x1, input_buf
    add     x1, x1, :lo12:input_buf
    add     x1, x1, x19
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x19, x19, x0
    b       read_loop

    add     x9, x9, :lo12:input_len
    str     w19, [x9]

    mov     w0, #0
    bl      ws
    bl      value
    cbz     w1, do_fail

    bl      ws
    adrp    x9, input_len
    ldr     w10, [x9, :lo12:input_len]

    b       do_ok

    b       do_fail

    add     x1, x1, :lo12:msg_ok
    mov     x0, #1
    mov     x2, #3
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0

do_fail:

    add     x1, x1, :lo12:numbuf

    add     w6, w6, #0x30

    strb    w6, [x1]
    add     w2, w2, #1
    mov     w0, w5
    b       fmt_loop

    mov     x2, #9

    adrp    x1, msg_nl

    mov     x2, #1
    mov     x8, #64
    svc     #0

    mov     x0, #1
    mov     x8, #93
    svc     #0

// =====================================================
// peek_byte: w0 = pos -> w0 = byte(0-255) or -1
// =====================================================
peek_byte:
    adrp    x9, input_len
    ldr     w10, [x9, :lo12:input_len]
    cmp     w0, w10
    b.lt    1f
    mov     w0, #-1
    ret
1:
    adrp    x9, input_buf
    add     x9, x9, :lo12:input_buf
    add     x9, x9, w0, uxtw
    ldrb    w0, [x9]
    ret

// =====================================================
// expect_byte: w0=pos, w1=expected byte -> w0=newpos, w1=ok
// =====================================================
expect_byte:
    adrp    x9, input_len
    ldr     w10, [x9, :lo12:input_len]
    cmp     w0, w10
    b.ge    expect_byte_fail
    adrp    x9, input_buf
    add     x9, x9, :lo12:input_buf
    add     x9, x9, w0, uxtw
    ldrb    w11, [x9]
    cmp     w11, w1
    b.ne    expect_byte_fail
    add     w0, w0, #1
    mov     w1, #1
    ret
expect_byte_fail:
    adrp    x9, max_fail
    add     x9, x9, :lo12:max_fail
    ldr     w10, [x9]
    cmp     w0, w10
    b.le    1f
    str     w0, [x9]
1:
    mov     w1, #0
    ret

// =====================================================
// record_fail: w0=pos
// =====================================================
record_fail:
    adrp    x9, max_fail
    add     x9, x9, :lo12:max_fail
    ldr     w10, [x9]
    cmp     w0, w10
    b.le    1f
    str     w0, [x9]
1:
    ret

// =====================================================
// ws: w0=pos -> w0=newpos (always succeeds)
// =====================================================
ws:
    stp     x29, x30, [sp, #-32]!
    str     x19, [sp, #16]
    mov     x29, sp
    mov     w19, w0
ws_loop:
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x20
    b.eq    ws_adv
    cmp     w0, #0x09
    b.eq    ws_adv
    cmp     w0, #0x0A
    b.eq    ws_adv
    cmp     w0, #0x0D
    b.eq    ws_adv
    b       ws_end
ws_adv:
    add     w19, w19, #1
    b       ws_loop
ws_end:
    mov     w0, w19
    ldr     x19, [sp, #16]
    ldp     x29, x30, [sp], #32
    ret

// =====================================================
// value: w0=pos -> w0=newpos, w1=ok
// =====================================================
value:
    stp     x29, x30, [sp, #-32]!
    str     x19, [sp, #16]
    mov     x29, sp
    mov     w19, w0

    mov     w0, w19
    bl      object
    cbnz    w1, value_ok

    mov     w0, w19
    bl      array
    cbnz    w1, value_ok

    mov     w0, w19
    bl      string
    cbnz    w1, value_ok

    mov     w0, w19
    bl      number
    cbnz    w1, value_ok

    mov     w0, w19
    adrp    x1, lit_true
    add     x1, x1, :lo12:lit_true
    mov     w2, #4
    bl      match_lit

    mov     w0, w19

    add     x1, x1, :lo12:lit_false
    mov     w2, #5
    bl      match_lit
    cbnz    w1, value_ok

    mov     w0, w19
    adrp    x1, lit_null
    add     x1, x1, :lo12:lit_null
    mov     w2, #4
    bl      match_lit
    cbnz    w1, value_ok

    mov     w1, #0
    b       value_end
value_ok:
value_end:
    ldr     x19, [sp, #16]
    ldp     x29, x30, [sp], #32
    ret

// =====================================================
// object: w0=pos -> w0=newpos, w1=ok
// =====================================================
object:
    stp     x29, x30, [sp, #-32]!
    stp     x19, x20, [sp, #16]

    bl      string
    cbz     w1, object_fail

    mov     w0, w19
    bl      ws
    mov     w19, w0

    mov     w0, w19
    mov     w1, #0x3A
    bl      expect_byte
    cbz     w1, object_fail
    mov     w19, w0

    mov     w0, w19
    bl      ws

    ldp     x19, x20, [sp, #16]

// string: w0=pos -> w0=newpos, w1=ok
// =====================================================
string:

    stp     x19, x20, [sp, #16]
    stp     x21, x22, [sp, #32]
    mov     x29, sp
    mov     w19, w0

    bl      peek_byte
    mov     w20, w0

    cmp     w20, #0x22
    b.eq    string_close
    cmp     w20, #0x5C
    b.eq    string_escape

    cmp     w20, #0x20
    b.lt    string_bad_char
    add     w19, w19, #1
    b       string_loop

string_close:

    b       string_success

string_bad_char:

    b       string_fail

string_escape:

    mov     w0, w21
    bl      peek_byte
    mov     w20, w0

    cmp     w20, #0x22
    b.eq    string_esc_simple
    cmp     w20, #0x5C
    b.eq    string_esc_simple
    cmp     w20, #0x2F
    b.eq    string_esc_simple
    cmp     w20, #0x62
    b.eq    string_esc_simple
    cmp     w20, #0x66
    b.eq    string_esc_simple
    cmp     w20, #0x6E
    b.eq    string_esc_simple
    cmp     w20, #0x72
    b.eq    string_esc_simple
    cmp     w20, #0x74
    b.eq    string_esc_simple
    cmp     w20, #0x75
    b.eq    string_esc_u

    mov     w0, w21
    bl      record_fail
    b       string_fail

string_esc_simple:

string_esc_u:

    mov     w20, #0

    cmp     w0, #0x39

    cmp     w0, #0x61
    b.lt    string_hex_check_upper
    cmp     w0, #0x66
    b.le    string_hex_ok

string_hex_check_upper:
    cmp     w0, #0x41

    cmp     w0, #0x46

    b       string_hex_bad
string_hex_ok:

    add     w20, w20, #1
    b       string_hex_loop
string_hex_bad:

    b       string_fail
string_hex_done:

    b       string_loop

string_fail:

string_success:
    mov     w0, w19
    mov     w1, #1
string_end:
    ldp     x21, x22, [sp, #32]
    ldp     x19, x20, [sp, #16]
    ldp     x29, x30, [sp], #48
    ret

// =====================================================
// number: w0=pos -> w0=newpos, w1=ok
// =====================================================

    stp     x29, x30, [sp, #-32]!

number_int:

    bl      peek_byte
    cmp     w0, #0x30
    b.eq    number_zero
    cmp     w0, #0x31
    b.lt    number_int_fail
    cmp     w0, #0x39
    b.gt    number_int_fail
    add     w19, w19, #1
number_int_digits:
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x30
    b.lt    number_int_done
    cmp     w0, #0x39
    b.gt    number_int_done
    add     w19, w19, #1
    b       number_int_digits

number_zero:
    add     w19, w19, #1
    b       number_int_done

number_int_fail:
    mov     w0, w19
    bl      record_fail
    mov     w1, #0
    b       number_end

number_int_done:
    mov     w20, w19
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x2E
    b.ne    number_exp_check
    add     w19, w19, #1
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x30
    b.lt    number_frac_bad
    cmp     w0, #0x39
    b.gt    number_frac_bad
    add     w19, w19, #1
number_frac_digits:
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x30
    b.lt    number_exp_check
    cmp     w0, #0x39
    b.gt    number_exp_check
    add     w19, w19, #1
    b       number_frac_digits

number_frac_bad:
    mov     w0, w19
    bl      record_fail
    mov     w19, w20
    b       number_exp_check

number_exp_check:
    mov     w20, w19
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x65
    b.eq    number_exp_go
    cmp     w0, #0x45
    b.eq    number_exp_go
    b       number_success

number_exp_go:
    add     w19, w19, #1
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x2B
    b.eq    number_exp_sign
    cmp     w0, #0x2D
    b.eq    number_exp_sign
    b       number_exp_digit1

number_exp_sign:
    add     w19, w19, #1
    b       number_exp_digit1

number_exp_digit1:
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x30
    b.lt    number_exp_bad
    cmp     w0, #0x39
    b.gt    number_exp_bad
    add     w19, w19, #1
number_exp_digits:
    mov     w0, w19
    bl      peek_byte
    cmp     w0, #0x30
    b.lt    number_success
    cmp     w0, #0x39
    b.gt    number_success
    add     w19, w19, #1
    b       number_exp_digits

number_exp_bad:
    mov     w0, w19
    bl      record_fail
    mov     w19, w20
    b       number_success

number_success:
    mov     w0, w19
    mov     w1, #1
number_end:
    ldp     x19, x20, [sp, #16]
    ldp     x29, x30, [sp], #32
    ret

// =====================================================
// match_lit: w0=pos, x1=table addr, w2=len -> w0=newpos, w1=ok

match_lit:
    stp     x29, x30, [sp, #-48]!
    stp     x19, x20, [sp, #16]
    stp     x21, x22, [sp, #32]
    mov     x29, sp
    mov     w19, w0
    mov     x20, x1
    mov     w21, w2
    mov     w22, #0

ml_loop:
    cmp     w22, w21
    b.ge    ml_success
    add     x3, x20, w22, uxtw
    ldrb    w1, [x3]
    mov     w0, w19
    bl      expect_byte
    cbz     w1, ml_fail
    mov     w19, w0
    add     w22, w22, #1
    b       ml_loop

ml_fail:
    mov     w1, #0
    b       ml_end
ml_success:
    mov     w0, w19
    mov     w1, #1
ml_end:
    ldp     x21, x22, [sp, #32]
    ldp     x19, x20, [sp, #16]
    ldp     x29, x30, [sp], #48
    ret
