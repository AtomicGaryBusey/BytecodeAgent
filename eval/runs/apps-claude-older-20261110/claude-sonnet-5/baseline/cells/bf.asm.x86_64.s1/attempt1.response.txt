```asm
.globl _start

.section .bss
.align 8
INBUF:   .zero 65536
OUTBUF:  .zero 65536
TAPE:    .zero 30000
MATCH:   .zero 262144
STACK:   .zero 4004

.text
_start:
    xor r8, r8

read_loop:
    xor rax, rax
    xor rdi, rdi
    lea rsi, [rip + INBUF]
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz after_read
    add r8, rax
    cmp r8, 65536
    jae after_read
    jmp read_loop

after_read:
    mov r9, r8
    xor r10, r10
    lea rbx, [rip + INBUF]

find_bang:
    cmp r10, r9
    jge no_bang
    movzx eax, byte ptr [rbx + r10]
    cmp eax, '!'
    je found_bang
    inc r10
    jmp find_bang

found_bang:
    mov r12, r10
    lea r13, [rbx + r10 + 1]
    mov r14, r9
    sub r14, r10
    sub r14, 1
    jmp have_split

no_bang:
    mov r12, r9
    lea r13, [rbx + r9]
    xor r14, r14

have_split:
    mov r15, rbx

    lea rsi, [rip + MATCH]
    lea rdi, [rip + STACK]
    xor rcx, rcx
    xor rdx, rdx

match_loop:
    cmp rdx, r12
    jge match_done
    movzx eax, byte ptr [r15 + rdx]
    cmp eax, '['
    je is_open
    cmp eax, ']'
    je is_close
    jmp match_next

is_open:
    mov dword ptr [rdi + rcx*4], edx
    inc rcx
    jmp match_next

is_close:
    cmp rcx, 0
    je mismatch_exit
    dec rcx
    mov r8d, dword ptr [rdi + rcx*4]
    mov dword ptr [rsi + r8*4], edx
    mov dword ptr [rsi + rdx*4], r8d

match_next:
    inc rdx
    jmp match_loop

match_done:
    cmp rcx, 0
    jne mismatch_exit
    jmp exec_init

mismatch_exit:
    mov rax, 60
    mov rdi, 1
    syscall

exec_init:
    xor rdx, rdx
    xor rbx, rbx
    xor rcx, rcx
    xor rdi, rdi
    lea r8, [rip + TAPE]
    lea r9, [rip + OUTBUF]

exec_loop:
    cmp rdx, r12
    jge exec_done
    movzx eax, byte ptr [r15 + rdx]
    cmp eax, '>'
    je op_gt
    cmp eax, '<'
    je op_lt
    cmp eax, '+'
    je op_plus
    cmp eax, '-'
    je op_minus
    cmp eax, '.'
    je op_dot
    cmp eax, ','
    je op_comma
    cmp eax, '['
    je op_open
    cmp eax, ']'
    je op_close
    jmp op_next

op_gt:
    inc rbx
    jmp op_next

op_lt:
    dec rbx
    jmp op_next

op_plus:
    inc byte ptr [r8 + rbx]
    jmp op_next

op_minus:
    dec byte ptr [r8 + rbx]
    jmp op_next

op_dot:
    mov al, byte ptr [r8 + rbx]
    mov byte ptr [r9 + rdi], al
    inc rdi
    jmp op_next

op_comma:
    cmp rcx, r14
    jge comma_zero
    movzx eax, byte ptr [r13 + rcx]
    mov byte ptr [r8 + rbx], al
    inc rcx
    jmp op_next

comma_zero:
    mov byte ptr [r8 + rbx], 0
    jmp op_next

op_open:
    cmp byte ptr [r8 + rbx], 0
    jne op_next
    mov eax, dword ptr [rsi + rdx*4]
    mov rdx, rax
    jmp op_next

op_close:
    cmp byte ptr [r8 + rbx], 0
    je op_next
    mov eax, dword ptr [rsi + rdx*4]
    mov rdx, rax
    jmp op_next

op_next:
    inc rdx
    jmp exec_loop

exec_done:
    test rdi, rdi
    jz do_exit
    mov rdx, rdi
    lea rsi, [rip + OUTBUF]
    mov rdi, 1
    mov rax, 1
    syscall

do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
```
