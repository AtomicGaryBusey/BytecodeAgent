```asm
.section .rodata
.align 8
b64tab:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buf:
.zero 49152
.align 16
out_buf:
.zero 66500

.text
.globl _start
_start:
    lea r12, [rip + input_buf]
    xor r13, r13

read_loop:
    mov rdx, 49152
    sub rdx, r13
    jz read_done
    lea rsi, [r12 + r13]
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + out_buf]
    mov r10, r12
    mov r11, r13
    xor r15, r15
    lea rbx, [rip + b64tab]

main_loop:
    cmp r11, 3
    jb remainder_handle
    movzx eax, byte ptr [r10]
    movzx ecx, byte ptr [r10 + 1]
    movzx edx, byte ptr [r10 + 2]
    add r10, 3
    sub r11, 3

    mov r9d, eax
    shr r9d, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r8d, ecx
    shr r8d, 4
    or r9d, r8d
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov r9d, ecx
    and r9d, 0xF
    shl r9d, 2
    mov r8d, edx
    shr r8d, 6
    or r9d, r8d
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov r9d, edx
    and r9d, 0x3F
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    inc r15
    cmp r15, 19
    jne main_loop
    mov byte ptr [r14], 10
    inc r14
    xor r15, r15
    jmp main_loop

remainder_handle:
    cmp r11, 0
    je end_handle
    cmp r11, 1
    je rem1

    # r11 == 2
    movzx eax, byte ptr [r10]
    movzx ecx, byte ptr [r10 + 1]

    mov r9d, eax
    shr r9d, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r8d, ecx
    shr r8d, 4
    or r9d, r8d
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov r9d, ecx
    and r9d, 0xF
    shl r9d, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov byte ptr [r14], 0x3D
    inc r14
    inc r15
    jmp end_handle

rem1:
    movzx eax, byte ptr [r10]

    mov r9d, eax
    shr r9d, 2
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    movzx r9d, byte ptr [rbx + r9]
    mov byte ptr [r14], r9b
    inc r14

    mov byte ptr [r14], 0x3D
    inc r14
    mov byte ptr [r14], 0x3D
    inc r14
    inc r15
    jmp end_handle

end_handle:
    cmp r15, 0
    je write_out
    mov byte ptr [r14], 10
    inc r14

write_out:
    lea rsi, [rip + out_buf]
    mov rdx, r14
    sub rdx, rsi
    test rdx, rdx
    jz do_exit
    mov rdi, 1
    mov rax, 1
    syscall

do_exit:
    mov rax, 60
    xor rdi, rdi
    syscall
```
