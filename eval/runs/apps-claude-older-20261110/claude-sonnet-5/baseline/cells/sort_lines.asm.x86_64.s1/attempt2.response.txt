.globl _start

.text

_start:
    # ---- read all of stdin into input_buf ----
    xor r13, r13                 # r13 = total bytes read
.Lread_loop:
    mov rax, 0                   # sys_read
    mov rdi, 0                   # fd 0
    lea rsi, [rip + input_buf]
    add rsi, r13
    mov rdx, 33000
    sub rdx, r13
    syscall
    test rax, rax
    jz .Lread_done
    add r13, rax
    jmp .Lread_loop
.Lread_done:

    # ---- parse lines ----
    lea rbx, [rip + input_buf]
    lea r12, [rip + lines_arr]
    xor r14, r14                 # line_start
    xor r15, r15                 # line_count
    xor rcx, rcx                 # idx
.Lparse_loop:
    cmp rcx, r13
    jge .Lparse_end
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 0x0A
    jne .Lparse_next
    mov rax, rbx
    add rax, r14
    mov rdx, rcx
    sub rdx, r14
    mov rsi, r15
    shl rsi, 4
    add rsi, r12
    mov qword ptr [rsi], rax
    mov qword ptr [rsi+8], rdx
    inc r15
    mov r14, rcx
    inc r14
.Lparse_next:
    inc rcx
    jmp .Lparse_loop
.Lparse_end:
    cmp r14, r13
    jge .Lno_final
    mov rax, rbx
    add rax, r14
    mov rdx, r13
    sub rdx, r14
    mov rsi, r15
    shl rsi, 4
    add rsi, r12
    mov qword ptr [rsi], rax
    mov qword ptr [rsi+8], rdx
    inc r15
.Lno_final:
    lea rax, [rip + total_lines]
    mov qword ptr [rax], r15

    # ---- sort ----
    mov rdi, 0
    lea rax, [rip + total_lines]
    mov rsi, qword ptr [rax]
    call merge_sort

    # ---- build output ----
    lea rbx, [rip + output_buf]
    xor r12, r12                 # output length
    xor r13, r13                 # line index
.Lout_loop:
    lea rax, [rip + total_lines]
    cmp r13, qword ptr [rax]
    jge .Lout_done
    lea rax, [rip + lines_arr]
    mov rcx, r13
    shl rcx, 4
    add rax, rcx
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax+8]
    xor rcx, rcx
.Lcopy_loop:
    cmp rcx, rdx
    jge .Lcopy_done
    movzx eax, byte ptr [rsi + rcx]
    mov byte ptr [rbx + r12], al
    inc r12
    inc rcx
    jmp .Lcopy_loop
.Lcopy_done:
    mov byte ptr [rbx + r12], 0x0A
    inc r12
    inc r13
    jmp .Lout_loop
.Lout_done:

    # ---- write output ----
    xor r14, r14
.Lwrite_loop:
    cmp r14, r12
    jge .Lwrite_done
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buf]
    add rsi, r14
    mov rdx, r12
    sub rdx, r14
    syscall
    add r14, rax
    jmp .Lwrite_loop
.Lwrite_done:

    mov rax, 60
    mov rdi, 0
    syscall

# ---------------------------------------------------------
# merge_sort(rdi=lo, rsi=hi)  sorts lines_arr[lo..hi)
# ---------------------------------------------------------
merge_sort:
    mov rax, rsi
    sub rax, rdi
    cmp rax, 1
    jle .Lms_ret
    push rdi
    push rsi
    mov rax, rdi
    add rax, rsi
    shr rax, 1
    push rax

    mov rdi, qword ptr [rsp+16]
    mov rsi, qword ptr [rsp]
    call merge_sort

    mov rdi, qword ptr [rsp]
    mov rsi, qword ptr [rsp+8]
    call merge_sort

    mov rdi, qword ptr [rsp+16]
    mov rsi, qword ptr [rsp]
    mov rdx, qword ptr [rsp+8]
    call merge

    add rsp, 24
.Lms_ret:
    ret

# ---------------------------------------------------------
# merge(rdi=lo, rsi=mid, rdx=hi)
# ---------------------------------------------------------
merge:
    lea rax, [rip + g_lo]
    mov qword ptr [rax], rdi
    lea rax, [rip + g_mid]
    mov qword ptr [rax], rsi
    lea rax, [rip + g_hi]
    mov qword ptr [rax], rdx
    lea rax, [rip + g_i]
    mov qword ptr [rax], rdi
    lea rax, [rip + g_j]
    mov qword ptr [rax], rsi
    lea rax, [rip + g_k]
    mov qword ptr [rax], rdi

.Lmg_loop:
    lea rax, [rip + g_i]
    mov rax, qword ptr [rax]
    lea rcx, [rip + g_mid]
    cmp rax, qword ptr [rcx]
    jge .Lmg_right_only
    lea rax, [rip + g_j]
    mov rax, qword ptr [rax]
    lea rcx, [rip + g_hi]
    cmp rax, qword ptr [rcx]
    jge .Lmg_left_only

    lea rcx, [rip + lines_arr]
    lea rax, [rip + g_i]
    mov rax, qword ptr [rax]
    shl rax, 4
    add rax, rcx
    lea rcx, [rip + lines_arr]
    lea rsi, [rip + g_j]
    mov rsi, qword ptr [rsi]
    shl rsi, 4
    add rsi, rcx

    mov rdi, qword ptr [rax]
    mov r8,  qword ptr [rax+8]
    mov rdx, qword ptr [rsi]
    mov r9,  qword ptr [rsi+8]
    mov rsi, r8
    mov rcx, r9
    call compare
    cmp rax, 0
    jg .Lmg_from_j

    lea rcx, [rip + lines_arr]
    lea rax, [rip + g_i]
    mov rax, qword ptr [rax]
    shl rax, 4
    add rax, rcx
    lea rcx, [rip + tmp_arr]
    lea rdx, [rip + g_k]
    mov rdx, qword ptr [rdx]
    shl rdx, 4
    add rdx, rcx
    mov r10, qword ptr [rax]
    mov r11, qword ptr [rax+8]
    mov qword ptr [rdx], r10
    mov qword ptr [rdx+8], r11
    lea rax, [rip + g_i]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    lea rax, [rip + g_k]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    jmp .Lmg_loop

.Lmg_from_j:
    lea rcx, [rip + lines_arr]
    lea rax, [rip + g_j]
    mov rax, qword ptr [rax]
    shl rax, 4
    add rax, rcx
    lea rcx, [rip + tmp_arr]
    lea rdx, [rip + g_k]
    mov rdx, qword ptr [rdx]
    shl rdx, 4
    add rdx, rcx
    mov r10, qword ptr [rax]
    mov r11, qword ptr [rax+8]
    mov qword ptr [rdx], r10
    mov qword ptr [rdx+8], r11
    lea rax, [rip + g_j]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    lea rax, [rip + g_k]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    jmp .Lmg_loop

.Lmg_left_only:
    lea rcx, [rip + g_i]
    mov rax, qword ptr [rcx]
    lea rcx, [rip + g_mid]
    cmp rax, qword ptr [rcx]
    jge .Lmg_copy_back
    lea rcx, [rip + lines_arr]
    shl rax, 4
    add rax, rcx
    lea rcx, [rip + tmp_arr]
    lea rdx, [rip + g_k]
    mov rdx, qword ptr [rdx]
    shl rdx, 4
    add rdx, rcx
    mov r10, qword ptr [rax]
    mov r11, qword ptr [rax+8]
    mov qword ptr [rdx], r10
    mov qword ptr [rdx+8], r11
    lea rax, [rip + g_i]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    lea rax, [rip + g_k]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    jmp .Lmg_left_only

.Lmg_right_only:
    lea rcx, [rip + g_j]
    mov rax, qword ptr [rcx]
    lea rcx, [rip + g_hi]
    cmp rax, qword ptr [rcx]
    jge .Lmg_copy_back
    lea rcx, [rip + lines_arr]
    shl rax, 4
    add rax, rcx
    lea rcx, [rip + tmp_arr]
    lea rdx, [rip + g_k]
    mov rdx, qword ptr [rdx]
    shl rdx, 4
    add rdx, rcx
    mov r10, qword ptr [rax]
    mov r11, qword ptr [rax+8]
    mov qword ptr [rdx], r10
    mov qword ptr [rdx+8], r11
    lea rax, [rip + g_j]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    lea rax, [rip + g_k]
    mov rcx, qword ptr [rax]
    inc rcx
    mov qword ptr [rax], rcx
    jmp .Lmg_right_only

.Lmg_copy_back:
    lea rax, [rip + g_lo]
    mov rax, qword ptr [rax]
    lea rcx, [rip + g_i]
    mov qword ptr [rcx], rax
.Lmg_cb_loop:
    lea rcx, [rip + g_i]
    mov r8, qword ptr [rcx]
    lea rcx, [rip + g_hi]
    cmp r8, qword ptr [rcx]
    jge .Lmg_ret
    mov rax, r8
    shl rax, 4
    lea rcx, [rip + tmp_arr]
    add rax, rcx
    mov rdx, r8
    shl rdx, 4
    lea rcx, [rip + lines_arr]
    add rdx, rcx
    mov r10, qword ptr [rax]
    mov r11, qword ptr [rax+8]
    mov qword ptr [rdx], r10
    mov qword ptr [rdx+8], r11
    inc r8
    lea rcx, [rip + g_i]
    mov qword ptr [rcx], r8
    jmp .Lmg_cb_loop
.Lmg_ret:
    ret

# ---------------------------------------------------------
# compare(rdi=ptrA, rsi=lenA, rdx=ptrB, rcx=lenB) -> rax
# ---------------------------------------------------------
compare:
    mov r8, rsi
    cmp r8, rcx
    jle .Lcmp_have_min
    mov r8, rcx
.Lcmp_have_min:
    xor r9, r9
.Lcmp_loop:
    cmp r9, r8
    jge .Lcmp_tie
    movzx rax, byte ptr [rdi + r9]
    movzx r11, byte ptr [rdx + r9]
    cmp rax, r11
    jne .Lcmp_diff
    inc r9
    jmp .Lcmp_loop
.Lcmp_diff:
    sub rax, r11
    ret
.Lcmp_tie:
    mov rax, rsi
    sub rax, rcx
    ret

.bss
.align 8
input_buf:   .zero 33000
output_buf:  .zero 33000
lines_arr:   .zero 16000
tmp_arr:     .zero 16000
total_lines: .zero 8
g_lo:        .zero 8
g_mid:       .zero 8
g_hi:        .zero 8
g_i:         .zero 8
g_j:         .zero 8
g_k:         .zero 8
