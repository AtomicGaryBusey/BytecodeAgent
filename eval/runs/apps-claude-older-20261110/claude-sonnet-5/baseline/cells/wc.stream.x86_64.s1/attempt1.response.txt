.globl _start

.bss
.align 16
buffer:

outbuf:
    .zero 64

_start:
    xor r12, r12
    lea rbx, [rip + buffer]

read_loop:
    mov rdi, 0
    lea rsi, [rbx + r12]
    mov rdx, 65536
    sub rdx, r12
    xor rax, rax
    syscall
    test rax, rax
    jz process_start
    add r12, rax
    jmp read_loop

process_start:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    xor r10, r10

process_loop:
    cmp r10, r12
    jge process_done
    movzx eax, byte ptr [rbx + r10]
    cmp al, 0x0A
    jne 1f
    inc r13
1:
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    jb 3f
    cmp al, 0x0D
    jbe 2f
    jmp 3f
2:
    mov r15, 0
    jmp 4f
3:
    cmp r15, 0
    jne 4f
    inc r14
    mov r15, 1
4:
    inc r10
    jmp process_loop

process_done:
    lea r9, [rip + outbuf]
    mov rsi, r9

    mov rax, r13
    call write_num
    mov byte ptr [rsi], 0x20
    inc rsi

    mov rax, r14
    call write_num
    mov byte ptr [rsi], 0x20
    inc rsi

    mov rax, r12
    call write_num
    mov byte ptr [rsi], 0x0A
    inc rsi

    mov rdx, rsi
    sub rdx, r9

    mov rdi, 1
    mov rsi, r9
    mov rax, 1
    syscall

    mov rdi, 0
    mov rax, 60
    syscall

write_num:
    sub rsp, 32
    xor rcx, rcx
    test rax, rax
    jnz 10f
    mov byte ptr [rsp], 0x30
    inc rcx
    jmp 11f
10:
12:

    mov r9, 10
    div r9
    add dl, 0x30
    mov byte ptr [rsp + rcx], dl
    inc rcx
    test rax, rax
    jnz 12b
11:

    dec r8
13:
    mov dl, byte ptr [rsp + r8]
    mov byte ptr [rsi], dl
    inc rsi
    test r8, r8
    jz 14f
    dec r8
    jmp 13b
14:
