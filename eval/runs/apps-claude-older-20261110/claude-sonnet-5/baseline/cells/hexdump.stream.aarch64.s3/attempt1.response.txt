```asm
.equ INBUF_SIZE, 65536
.equ OUTBUF_SIZE, 330000

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .space INBUF_SIZE
outbuf:
    .space OUTBUF_SIZE

.text
.globl _start

_start:
    mov x19, #0

read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x21, outbuf
    add x21, x21, :lo12:outbuf
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    mov x23, #0

row_loop:
    cmp x23, x19
    b.hs rows_done

    sub x24, x19, x23
    cmp x24, #16
    b.ls set_n_remaining
    mov x25, #16
    b n_done
set_n_remaining:
    mov x25, x24
n_done:

    mov w0, w23
    mov x1, x21
    bl print_hex8
    mov x21, x1

    mov w0, #32
    strb w0, [x21], #1
    strb w0, [x21], #1

    mov x26, #0
byte_loop:
    cmp x26, #16
    b.ge byte_loop_end
    cmp x26, x25
    b.lt do_hex
    mov w0, #32
    strb w0, [x21], #1
    strb w0, [x21], #1
    strb w0, [x21], #1
    b after_hex
do_hex:
    add x0, x22, x23
    add x0, x0, x26
    ldrb w0, [x0]
    mov x1, x21
    bl print_hex_byte
    mov x21, x1
    mov w0, #32
    strb w0, [x21], #1
after_hex:
    cmp x26, #7
    b.ne skip_extra
    mov w0, #32
    strb w0, [x21], #1
skip_extra:
    add x26, x26, #1
    b byte_loop
byte_loop_end:

    mov w0, #32
    strb w0, [x21], #1
    mov w0, #124
    strb w0, [x21], #1

    mov x26, #0
ascii_loop:
    cmp x26, x25
    b.ge ascii_loop_end
    add x0, x22, x23
    add x0, x0, x26
    ldrb w27, [x0]
    cmp w27, #32
    b.lt not_printable
    cmp w27, #126
    b.gt not_printable
    b store_char
not_printable:
    mov w27, #46
store_char:
    strb w27, [x21], #1
    add x26, x26, #1
    b ascii_loop
ascii_loop_end:

    mov w0, #124
    strb w0, [x21], #1
    mov w0, #10
    strb w0, [x21], #1

    add x23, x23, x25
    b row_loop

rows_done:
    mov w0, w19
    mov x1, x21
    bl print_hex8
    mov x21, x1
    mov w0, #10
    strb w0, [x21], #1

    adrp x3, outbuf
    add x3, x3, :lo12:outbuf
    sub x2, x21, x3
    mov x1, x3
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

print_hex8:
    mov w4, #28
    adrp x3, hex_table
    add x3, x3, :lo12:hex_table
hex8_loop:
    lsr w2, w0, w4
    and w2, w2, #0xF
    ldrb w5, [x3, x2]
    strb w5, [x1], #1
    subs w4, w4, #4
    b.ge hex8_loop
    ret

print_hex_byte:
    adrp x3, hex_table
    add x3, x3, :lo12:hex_table
    lsr w2, w0, #4
    and w2, w2, #0xF
    ldrb w4, [x3, x2]
    strb w4, [x1], #1
    and w2, w0, #0xF
    ldrb w4, [x3, x2]
    strb w4, [x1], #1
    ret
```
