```asm
.globl _start

.section .rodata

.text

_start:
    // x19 = buffer base, x20 = total bytes read
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cmp x2, #0
    beq read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    beq read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x22, #0   // lines
    mov x23, #0   // words
    mov x21, #0   // in_word flag
    mov x24, #0   // index

proc_loop:
    cmp x24, x20
    bge proc_done
    ldrb w0, [x19, x24]
    add x24, x24, #1

    cmp w0, #0x20
    beq is_ws
    cmp w0, #0x09
    blt not_ws
    cmp w0, #0x0D
    bgt not_ws
    b is_ws

is_ws:
    cmp w0, #0x0A
    bne skip_line
    add x22, x22, #1
skip_line:
    mov x21, #0
    b proc_loop

not_ws:
    cmp x21, #0
    bne proc_loop
    mov x21, #1
    add x23, x23, #1
    b proc_loop

proc_done:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x9, x25

    mov x0, x22
    bl write_num
    mov w0, #' '
    strb w0, [x9], #1

    mov x0, x23
    bl write_num
    mov w0, #' '
    strb w0, [x9], #1

    mov x0, x20
    bl write_num
    mov w0, #'\n'
    strb w0, [x9], #1

    mov x0, #1
    mov x1, x25
    sub x2, x9, x25
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

write_num:
    // x0 = value, writes decimal digits to [x9], advances x9
    mov x10, x0
    adrp x11, digits_buf
    add x11, x11, :lo12:digits_buf
    add x11, x11, #20
    mov x12, #0
    cbnz x10, wn_loop
    mov w0, #'0'
    strb w0, [x9], #1
    ret
wn_loop:
    cbz x10, wn_done
    mov x13, #10
    udiv x14, x10, x13
    msub x15, x14, x13, x10
    add w15, w15, #'0'
    sub x11, x11, #1
    strb w15, [x11]
    add x12, x12, #1
    mov x10, x14
    b wn_loop
wn_done:
    mov x16, #0
wn_copy:
    cmp x16, x12
    bge wn_copy_done
    ldrb w17, [x11, x16]
    strb w17, [x9], #1
    add x16, x16, #1
    b wn_copy
wn_copy_done:
    ret

.bss
.balign 8
inbuf:
    .zero 65536
digits_buf:
    .zero 20
outbuf:
    .zero 32
```
