.section .bss
.align 8
input_buf: .zero 32768
line_off: .zero 4000
line_len: .zero 4000
output_buf: .zero 34000

.text
.globl _start

_start:
    xor r12, r12            # total bytes read so far

read_loop:
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    cmp rdx, 0
    je read_done
    xor edi, edi             # fd = 0
    xor eax, eax             # syscall read = 0
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    cmp r12, 32768
    jae read_done
    jmp read_loop
read_done:

    # parse lines
    xor r13, r13             # scan pos
    xor r14, r14             # start of current line
    xor r15, r15             # line count
    lea rbx, [rip + input_buf]
    lea r8, [rip + line_off]
    lea r9, [rip + line_len]

parse_loop:
    cmp r13, r12
    jae parse_end
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x0A
    jne parse_cont
    mov rcx, r13
    sub rcx, r14
    mov dword ptr [r8 + r15*4], r14d
    mov dword ptr [r9 + r15*4], ecx
    inc r15
    inc r13
    mov r14, r13
    jmp parse_loop
parse_cont:
    inc r13
    jmp parse_loop
parse_end:
    cmp r14, r12
    jae no_trailing
    mov rcx, r12
    sub rcx, r14
    mov dword ptr [r8 + r15*4], r14d
    mov dword ptr [r9 + r15*4], ecx
    inc r15
no_trailing:

    mov r12, r15              # r12 = number of lines (n)

    # insertion sort of (line_off,line_len) pairs, using r8,r9 as base pointers
    mov rbx, 1
Lsort_outer:
    cmp rbx, r12
    jae Lsort_done
    mov r13d, dword ptr [r8 + rbx*4]   # key_off
    mov r14d, dword ptr [r9 + rbx*4]   # key_len
    mov r15, rbx
    dec r15
Lsort_inner:
    cmp r15, 0
    jl Lsort_insert
    mov edi, dword ptr [r8 + r15*4]
    mov esi, dword ptr [r9 + r15*4]
    mov edx, r13d
    mov ecx, r14d
    call compare_lines
    cmp eax, 0
    jle Lsort_insert
    mov eax, dword ptr [r8 + r15*4]
    mov dword ptr [r8 + r15*4 + 4], eax
    mov eax, dword ptr [r9 + r15*4]
    mov dword ptr [r9 + r15*4 + 4], eax
    dec r15
    jmp Lsort_inner
Lsort_insert:
    inc r15
    mov dword ptr [r8 + r15*4], r13d
    mov dword ptr [r9 + r15*4], r14d
    inc rbx
    jmp Lsort_outer
Lsort_done:

    # build output buffer
    xor rbx, rbx
    lea r10, [rip + output_buf]
    lea r14, [rip + input_buf]
Lout_loop:
    cmp rbx, r12
    jae Lout_done
    mov eax, dword ptr [r8 + rbx*4]
    mov ecx, dword ptr [r9 + rbx*4]
    lea rsi, [r14 + rax]
    mov rdi, r10
    xor rdx, rdx
Lcopy_loop:
    cmp rdx, rcx
    jae Lcopy_done
    mov al, byte ptr [rsi + rdx]
    mov byte ptr [rdi + rdx], al
    inc rdx
    jmp Lcopy_loop
Lcopy_done:
    add r10, rcx
    mov byte ptr [r10], 0x0A
    inc r10
    inc rbx
    jmp Lout_loop
Lout_done:

    lea rax, [rip + output_buf]
    mov rdx, r10
    sub rdx, rax
    mov rsi, rax
    mov rdi, 1
    mov eax, 1
    syscall

    mov edi, 0
    mov eax, 60
    syscall

compare_lines:
    push rcx
    lea r10, [rip + input_buf]
    add rdi, r10
    add rdx, r10
    mov r10, rsi
    cmp r10, rcx
    jbe Lcmp_minok
    mov r10, rcx
Lcmp_minok:
    xor r11, r11
Lcmp_loop:
    cmp r11, r10
    jae Lcmp_tail
    movzx eax, byte ptr [rdi + r11]
    movzx ecx, byte ptr [rdx + r11]
    cmp eax, ecx
    jne Lcmp_diff
    inc r11
    jmp Lcmp_loop
Lcmp_diff:
    sub eax, ecx
    pop rcx
    ret
Lcmp_tail:
    pop rcx
    mov eax, esi
    sub eax, ecx
    ret
