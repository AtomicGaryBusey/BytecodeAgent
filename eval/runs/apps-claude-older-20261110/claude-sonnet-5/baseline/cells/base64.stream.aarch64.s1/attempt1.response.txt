```asm
.section .rodata
b64table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 4
inbuf:
    .zero 49152
outbuf:
    .zero 70000

.text
.globl _start
_start:
    // ---- read all of stdin into inbuf ----
    adrp x9, inbuf
    add  x9, x9, :lo12:inbuf
    mov  x10, #0          // total bytes read so far

read_loop:
    mov  x0, #0           // fd = 0
    add  x1, x9, x10      // buf = inbuf + total
    mov  x2, #49152
    sub  x2, x2, x10      // remaining space
    mov  x8, #63          // read
    svc  #0
    cmp  x0, #0
    beq  read_done
    add  x10, x10, x0
    b    read_loop

read_done:
    mov  x11, x10          // n
    mov  x12, #3
    udiv x13, x11, x12     // full_groups = n / 3
    mul  x14, x13, x12
    sub  x15, x11, x14     // remainder = n % 3

    adrp x16, inbuf
    add  x16, x16, :lo12:inbuf   // in ptr
    adrp x17, outbuf
    add  x17, x17, :lo12:outbuf  // out ptr
    adrp x18, b64table
    add  x18, x18, :lo12:b64table

    mov  x0, #0            // group counter (resets every 19 groups = 76 chars)

    mov  x1, #0            // loop index

full_loop:
    cmp  x1, x13
    bge  full_done
    ldrb w2, [x16], #1     // b0
    ldrb w3, [x16], #1     // b1
    ldrb w4, [x16], #1     // b2

    lsr  w5, w2, #2        // idx0
    and  w6, w2, #3
    lsl  w6, w6, #4
    lsr  w7, w3, #4
    orr  w6, w6, w7        // idx1

    and  w7, w3, #0xF
    lsl  w7, w7, #2
    lsr  w9, w4, #6
    orr  w7, w7, w9        // idx2

    and  w9, w4, #0x3F     // idx3

    ldrb w5, [x18, x5]
    strb w5, [x17], #1
    ldrb w6, [x18, x6]
    strb w6, [x17], #1
    ldrb w7, [x18, x7]
    strb w7, [x17], #1
    ldrb w9, [x18, x9]
    strb w9, [x17], #1

    add  x0, x0, #1
    cmp  x0, #19
    bne  1f
    mov  w2, #10
    strb w2, [x17], #1
    mov  x0, #0
1:
    add  x1, x1, #1
    b    full_loop

full_done:
    cmp  x15, #0
    beq  no_remainder
    cmp  x15, #1
    beq  rem1

    // remainder == 2
    ldrb w2, [x16], #1    // b0
    ldrb w3, [x16], #1    // b1
    lsr  w5, w2, #2
    and  w6, w2, #3
    lsl  w6, w6, #4
    lsr  w7, w3, #4
    orr  w6, w6, w7
    and  w7, w3, #0xF
    lsl  w7, w7, #2
    ldrb w5, [x18, x5]
    strb w5, [x17], #1
    ldrb w6, [x18, x6]
    strb w6, [x17], #1
    ldrb w7, [x18, x7]
    strb w7, [x17], #1
    mov  w9, #61
    strb w9, [x17], #1
    b    remainder_done

rem1:
    ldrb w2, [x16], #1    // b0
    lsr  w5, w2, #2
    and  w6, w2, #3
    lsl  w6, w6, #4
    ldrb w5, [x18, x5]
    strb w5, [x17], #1
    ldrb w6, [x18, x6]
    strb w6, [x17], #1
    mov  w9, #61
    strb w9, [x17], #1
    strb w9, [x17], #1

remainder_done:
    add  x0, x0, #1
    cmp  x0, #19
    bne  2f
    mov  w2, #10
    strb w2, [x17], #1
    mov  x0, #0
2:

no_remainder:
    cmp  x0, #0
    beq  write_output
    mov  w2, #10
    strb w2, [x17], #1

write_output:
    adrp x20, outbuf
    add  x20, x20, :lo12:outbuf
    sub  x21, x17, x20    // total output length
    mov  x22, x20

write_loop:
    cmp  x21, #0
    beq  exit_prog
    mov  x0, #1
    mov  x1, x22
    mov  x2, x21
    mov  x8, #64
    svc  #0
    add  x22, x22, x0
    sub  x21, x21, x0
    b    write_loop

exit_prog:
    mov  x0, #0
    mov  x8, #93
    svc  #0
```
