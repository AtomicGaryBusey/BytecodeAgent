.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 65536
linebuf:
    .zero 128

.text
.globl _start

_start:
    lea r15, [rip + inbuf]
    xor r12, r12

read_loop:
    cmp r12, 65536
    jae read_done
    lea rsi, [r15 + r12]
    mov rdx, 65536
    sub rdx, r12
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13

row_loop:
    cmp r13, r12
    jae rows_done

    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jbe row_n_ok
    mov rax, 16
row_n_ok:
    mov rbx, rax

    lea r14, [r15 + r13]

    lea rdi, [rip + linebuf]

    mov eax, r13d
    call write_hex32

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    add rdi, 2

    xor r8, r8
hexcol_loop:
    cmp r8, 16
    jae hexcol_done
    cmp r8, rbx
    jl hexcol_present

    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3
    jmp hexcol_check

hexcol_present:
    movzx edx, byte ptr [r14 + r8]
    call write_hex_byte
    mov byte ptr [rdi], ' '
    inc rdi

hexcol_check:
    cmp r8, 7
    jne hexcol_next
    mov byte ptr [rdi], ' '
    inc rdi

hexcol_next:
    inc r8
    jmp hexcol_loop

hexcol_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor r8, r8
char_loop:
    cmp r8, rbx
    jae char_done
    movzx eax, byte ptr [r14 + r8]
    cmp al, 0x20
    jl char_dot
    cmp al, 0x7e
    jg char_dot
    jmp char_store
char_dot:
    mov al, '.'
char_store:
    mov [rdi], al
    inc rdi
    inc r8
    jmp char_loop

char_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + linebuf]
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall

    add r13, 16
    jmp row_loop

rows_done:
    lea rdi, [rip + linebuf]
    mov eax, r12d
    call write_hex32
    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + linebuf]
    sub rdx, rsi
    mov rdi, 1
    mov rax, 1
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

write_hex_byte:
    movzx eax, dl
    mov ecx, eax
    shr ecx, 4
    and ecx, 0xf
    lea rsi, [rip + hexdigits]
    movzx ecx, byte ptr [rsi + rcx]
    mov [rdi], cl

    and eax, 0xf
    movzx eax, byte ptr [rsi + rax]
    mov [rdi+1], al

    add rdi, 2
    ret

write_hex32:
    mov ecx, 28
hex32_loop:
    mov edx, eax
    shr edx, cl
    and edx, 0xf
    lea rsi, [rip + hexdigits]
    movzx edx, byte ptr [rsi + rdx]
    mov [rdi], dl
    inc rdi
    sub ecx, 4
    jns hex32_loop
    ret
