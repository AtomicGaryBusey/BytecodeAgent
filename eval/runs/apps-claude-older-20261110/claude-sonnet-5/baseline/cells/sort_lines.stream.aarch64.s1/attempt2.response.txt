```asm
.globl _start

.section .rodata
newline_buf:
    .byte 0x0a

.bss
    .p2align 3
input_buf:
    .zero 32768
    .p2align 3
lines:
    .zero 8000      // up to 1000 entries of {offset:4, len:4}

.text

// cmp_lines: x0=offset1,x1=len1,x2=offset2,x3=len2
// uses global buffer base in x9
// returns w0 = -1,0,1
cmp_lines:
    cmp x1, x3
    csel x4, x1, x3, lo
    mov  x5, #0
cl_loop:
    cmp  x5, x4
    b.eq cl_prefix
    add  x6, x9, x0
    add  x6, x6, x5
    ldrb w6, [x6]
    add  x7, x9, x2
    add  x7, x7, x5
    ldrb w7, [x7]
    cmp  w6, w7
    b.lo cl_less
    b.hi cl_greater
    add  x5, x5, #1
    b    cl_loop
cl_prefix:
    cmp x1, x3
    b.lo cl_less
    b.hi cl_greater
    mov w0, #0
    ret
cl_less:
    mov w0, #-1
    ret
cl_greater:
    mov w0, #1
    ret

_start:
    adrp x9, input_buf
    add  x9, x9, :lo12:input_buf

    mov  x11, #0          // total bytes read
    mov  x12, #32768      // capacity

read_loop:
    cmp  x11, x12
    b.eq read_done
    add  x1, x9, x11
    sub  x2, x12, x11
    mov  x0, #0
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.eq read_done
    add  x11, x11, x0
    b    read_loop
read_done:

    // parse lines
    adrp x13, lines
    add  x13, x13, :lo12:lines
    mov  x14, #0          // n
    mov  x15, #0          // current line start
    mov  x16, #0          // index

parse_loop:
    cmp  x16, x11
    b.eq parse_end_check
    add  x0, x9, x16
    ldrb w0, [x0]
    cmp  w0, #10
    b.ne parse_next
    sub  x1, x16, x15
    str  w15, [x13]
    str  w1, [x13, #4]
    add  x13, x13, #8
    add  x14, x14, #1
    add  x16, x16, #1
    mov  x15, x16
    b    parse_loop
parse_next:
    add  x16, x16, #1
    b    parse_loop
parse_end_check:
    cmp  x15, x11
    b.eq sort_start
    sub  x1, x11, x15
    str  w15, [x13]
    str  w1, [x13, #4]
    add  x13, x13, #8
    add  x14, x14, #1

sort_start:
    adrp x17, lines
    add  x17, x17, :lo12:lines

    cmp  x14, #2
    b.lt sort_done

    mov  x19, #1          // i
outer_loop:
    cmp  x19, x14
    b.ge sort_done
    lsl  x0, x19, #3
    add  x1, x17, x0
    ldr  w21, [x1]
    ldr  w22, [x1, #4]
    mov  x20, x19         // j = i

inner_loop:
    cbz  x20, insert_here
    sub  x5, x20, #1
    lsl  x6, x5, #3
    add  x7, x17, x6
    ldr  w5, [x7]
    ldr  w6, [x7, #4]
    mov  x0, x21
    mov  x1, x22
    mov  x2, x5
    mov  x3, x6
    bl   cmp_lines
    cmp  w0, #0
    b.ge insert_here
    lsl  x0, x20, #3
    add  x1, x17, x0
    str  w5, [x1]
    str  w6, [x1, #4]
    sub  x20, x20, #1
    b    inner_loop

insert_here:
    lsl  x0, x20, #3
    add  x1, x17, x0
    str  w21, [x1]
    str  w22, [x1, #4]
    add  x19, x19, #1
    b    outer_loop

sort_done:
    adrp x17, lines
    add  x17, x17, :lo12:lines
    mov  x19, #0

out_loop:
    cmp  x19, x14
    b.ge out_done
    lsl  x5, x19, #3
    add  x6, x17, x5
    ldr  w23, [x6]
    ldr  w24, [x6, #4]
    add  x1, x9, x23
    mov  x0, #1
    mov  x2, x24
    mov  x8, #64
    svc  #0
    mov  x0, #1
    adrp x1, newline_buf
    add  x1, x1, :lo12:newline_buf
    mov  x2, #1
    mov  x8, #64
    svc  #0
    add  x19, x19, #1
    b    out_loop

out_done:
    mov  x0, #0
    mov  x8, #93
    svc  #0
```
