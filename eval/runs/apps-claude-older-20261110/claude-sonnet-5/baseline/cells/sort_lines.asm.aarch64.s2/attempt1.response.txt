.equ MAXIN, 32768
.equ MAXLINES, 1000

.section .bss
.align 3
input_buf:
    .zero MAXIN
.align 3
ptr_arr:
    .zero MAXLINES*8
.align 2
len_arr:
    .zero MAXLINES*4
.align 3
out_buf:
    .zero MAXIN+MAXLINES+16

.text
.globl _start
_start:
    // ---------- read entire stdin ----------
    mov x19, #0
read_loop:
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    add x1, x9, x19
    mov x0, #0
    mov x2, #32768
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cmp x0, #0
    beq read_done
    add x19, x19, x0
    b read_loop
read_done:
    // x19 = total_len

    // ---------- set up persistent base pointers ----------
    adrp x25, input_buf
    add x25, x25, :lo12:input_buf
    adrp x26, ptr_arr
    add x26, x26, :lo12:ptr_arr
    adrp x27, len_arr
    add x27, x27, :lo12:len_arr

    // ---------- parse lines ----------
    mov x20, #0      // n
    mov x21, #0      // line_start
    mov x22, #0      // i
parse_loop:
    cmp x22, x19
    bge parse_check_last
    ldrb w0, [x25, x22]
    cmp w0, #10
    bne parse_next
    sub x1, x22, x21
    add x2, x25, x21
    lsl x3, x20, #3
    str x2, [x26, x3]
    lsl x4, x20, #2
    str w1, [x27, x4]
    add x20, x20, #1
    add x21, x22, #1
parse_next:
    add x22, x22, #1
    b parse_loop
parse_check_last:
    cmp x21, x19
    bge parse_done
    sub x1, x19, x21
    add x2, x25, x21
    lsl x3, x20, #3
    str x2, [x26, x3]
    lsl x4, x20, #2
    str w1, [x27, x4]
    add x20, x20, #1
parse_done:

    // ---------- insertion sort ----------
    mov x21, #1
sort_outer:
    cmp x21, x20
    bge sort_done
    lsl x0, x21, #3
    ldr x23, [x26, x0]
    lsl x0, x21, #2
    ldr w24, [x27, x0]
    mov x22, x21
sort_inner:
    cmp x22, #0
    beq sort_insert
    sub x10, x22, #1
    lsl x11, x10, #3
    ldr x0, [x26, x11]
    lsl x12, x10, #2
    ldr w1, [x27, x12]
    mov x2, x23
    mov w3, w24
    bl cmp_bytes
    cmp w0, #0
    ble sort_insert
    ldr x0, [x26, x11]
    ldr w1, [x27, x12]
    lsl x13, x22, #3
    str x0, [x26, x13]
    lsl x14, x22, #2
    str w1, [x27, x14]
    mov x22, x10
    b sort_inner
sort_insert:
    lsl x13, x22, #3
    str x23, [x26, x13]
    lsl x14, x22, #2
    str w24, [x27, x14]
    add x21, x21, #1
    b sort_outer
sort_done:

    // ---------- write output ----------
    adrp x25, out_buf
    add x25, x25, :lo12:out_buf
    mov x9, x25
    mov x21, #0
out_loop:
    cmp x21, x20
    bge out_done
    lsl x0, x21, #3
    ldr x1, [x26, x0]
    lsl x0, x21, #2
    ldr w2, [x27, x0]
    mov x3, #0
copy_loop:
    cmp x3, x2
    bge copy_done
    ldrb w4, [x1, x3]
    strb w4, [x9, x3]
    add x3, x3, #1
    b copy_loop
copy_done:
    add x9, x9, x2
    mov w4, #10
    strb w4, [x9]
    add x9, x9, #1
    add x21, x21, #1
    b out_loop
out_done:
    sub x2, x9, x25
    mov x1, x25
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

cmp_bytes:
    mov x4, #0
    cmp w1, w3
    csel w5, w1, w3, lo
cmp_loop:
    cmp x4, x5
    bge cmp_tail
    ldrb w6, [x0, x4]
    ldrb w7, [x2, x4]
    cmp w6, w7
    bne cmp_diff
    add x4, x4, #1
    b cmp_loop
cmp_diff:
    sub w0, w6, w7
    ret
cmp_tail:
    sub w0, w1, w3
    ret
