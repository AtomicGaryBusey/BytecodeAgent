.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 16
input:
    .zero 65536
output:
    .zero 320000

.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov rdx, 65536
    sub rdx, r12
    cmp rdx, 0
    je read_done
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input]
    add rsi, r12
    syscall
    cmp rax, 0
    je read_done
    add r12, rax
    jmp read_loop
read_done:
    mov r13, r12
    xor r14, r14
    lea r15, [rip + output]

row_loop:
    cmp r14, r13
    jae after_rows
    mov rax, r13
    sub rax, r14
    cmp rax, 16
    jbe set_n
    mov rax, 16
set_n:
    mov rbx, rax

    mov rax, r14
    call write_offset_hex

    mov byte ptr [r15], ' '
    mov byte ptr [r15 + 1], ' '
    add r15, 2

    xor rcx, rcx
hex_loop:
    cmp rcx, 16
    jae hex_loop_done
    cmp rcx, rbx
    jb hex_write_byte
    mov byte ptr [r15], ' '
    mov byte ptr [r15 + 1], ' '
    mov byte ptr [r15 + 2], ' '
    add r15, 3
    jmp hex_loop_next
hex_write_byte:
    lea rdx, [rip + input]
    add rdx, r14
    movzx eax, byte ptr [rdx + rcx]
    call write_hex_pair
    mov byte ptr [r15], ' '
    inc r15
hex_loop_next:
    cmp rcx, 7
    jne hex_loop_cont
    mov byte ptr [r15], ' '
    inc r15
hex_loop_cont:
    inc rcx
    jmp hex_loop
hex_loop_done:

    mov byte ptr [r15], ' '
    mov byte ptr [r15 + 1], '|'
    add r15, 2

    xor rcx, rcx
ascii_loop:
    cmp rcx, rbx
    jae ascii_loop_done
    lea rdx, [rip + input]
    add rdx, r14
    movzx eax, byte ptr [rdx + rcx]
    cmp eax, 0x20
    jb ascii_dot
    cmp eax, 0x7e
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov eax, '.'
ascii_store:
    mov byte ptr [r15], al
    inc r15
    inc rcx
    jmp ascii_loop
ascii_loop_done:
    mov byte ptr [r15], '|'
    mov byte ptr [r15 + 1], 10
    add r15, 2

    add r14, rbx
    jmp row_loop

after_rows:
    mov rax, r13
    call write_offset_hex
    mov byte ptr [r15], 10
    inc r15

    lea rax, [rip + output]
    mov rdx, r15
    sub rdx, rax
    mov rsi, rax
    mov rdi, 1
    mov rax, 1
    syscall

    mov rdi, 0
    mov rax, 60
    syscall

write_hex_pair:
    push rax
    mov r8d, eax
    shr r8d, 4
    and r8d, 0xf
    lea r9, [rip + hex_digits]
    movzx r10d, byte ptr [r9 + r8]
    mov byte ptr [r15], r10b
    pop rax
    and eax, 0xf
    movzx r10d, byte ptr [r9 + rax]
    mov byte ptr [r15 + 1], r10b
    add r15, 2
    ret

write_offset_hex:
    mov r10, rax
    mov rcx, 28
    lea r9, [rip + hex_digits]
wo_loop:
    mov rax, r10
    shr rax, cl
    and eax, 0xf
    movzx r8d, byte ptr [r9 + rax]
    mov byte ptr [r15], r8b
    inc r15
    sub rcx, 4
    jl wo_done
    jmp wo_loop
wo_done:
    ret
