.text
.globl _start

_start:
    // ---- read all of stdin into inbuf ----
    mov     x19, #0                 // total bytes read so far

read_loop:
    adrp    x1, inbuf
    add     x1, x1, :lo12:inbuf
    add     x1, x1, x19             // buf = inbuf + x19
    mov     x2, #49152
    sub     x2, x2, x19             // remaining capacity
    cbz     x2, read_done
    mov     x0, #0                  // fd = 0
    mov     x8, #63                 // read
    svc     #0
    cbz     x0, read_done           // 0 = EOF
    add     x19, x19, x0
    b       read_loop

read_done:
    // x19 = total input length

    adrp    x1, inbuf
    add     x1, x1, :lo12:inbuf     // input pointer
    adrp    x2, outbuf
    add     x2, x2, :lo12:outbuf    // output pointer
    adrp    x20, outbuf
    add     x20, x20, :lo12:outbuf  // output base (saved)
    adrp    x14, alphabet
    add     x14, x14, :lo12:alphabet // alphabet base

    mov     x4, #0                  // column counter
    mov     x5, x19                 // remaining input bytes

loop3:
    cmp     x5, #3
    blt     tail
    ldrb    w6, [x1], #1
    ldrb    w7, [x1], #1
    ldrb    w8, [x1], #1
    sub     x5, x5, #3
    lsl     w9, w6, #16
    orr     w9, w9, w7, lsl #8
    orr     w9, w9, w8

    lsr     w10, w9, #18
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    lsr     w10, w9, #12
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    lsr     w10, w9, #6
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    and     w10, w9, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    b       loop3

tail:
    cmp     x5, #0
    beq     finish
    cmp     x5, #1
    beq     tail1

    // exactly 2 bytes remaining
    ldrb    w6, [x1], #1
    ldrb    w7, [x1], #1
    lsl     w9, w6, #16
    orr     w9, w9, w7, lsl #8

    lsr     w10, w9, #18
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    lsr     w10, w9, #12
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    lsr     w10, w9, #6
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    mov     w0, #0x3D               // '='
    bl      emit_byte
    b       finish

tail1:
    ldrb    w6, [x1], #1
    lsl     w9, w6, #16

    lsr     w10, w9, #18
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    lsr     w10, w9, #12
    and     w10, w10, #0x3F
    ldrb    w0, [x14, x10]
    bl      emit_byte

    mov     w0, #0x3D
    bl      emit_byte
    mov     w0, #0x3D
    bl      emit_byte
    b       finish

finish:
    cbz     x4, write_out
    mov     w16, #10
    strb    w16, [x2], #1

write_out:
    mov     x9, x20                 // write pointer
    sub     x10, x2, x20            // bytes remaining to write

write_loop:
    cbz     x10, do_exit
    mov     x0, #1
    mov     x1, x9
    mov     x2, x10
    mov     x8, #64
    svc     #0
    add     x9, x9, x0
    sub     x10, x10, x0
    b       write_loop

do_exit:
    mov     x0, #0
    mov     x8, #93
    svc     #0

// ---- emit_byte: append byte in w0 to output buffer, handling line wrap ----
// uses/updates x2 (output pointer), x4 (column counter)
emit_byte:
    strb    w0, [x2], #1
    add     x4, x4, #1
    cmp     x4, #76
    bne     emit_ret
    mov     w16, #10
    strb    w16, [x2], #1
    mov     x4, #0
emit_ret:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 4
inbuf:
    .zero 49152
outbuf:
    .zero 70000
