```
.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
linebuf:
    .zero 96

.text
.globl _start
_start:
    xor r12, r12            # r12 = total bytes read so far (length)
    lea rbx, [rip + inbuf]  # rbx = base of input buffer

read_loop:
    cmp r12, 65536
    jae read_done
    mov rdi, 0
    lea rsi, [rbx]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13             # r13 = current row offset

row_loop:
    cmp r13, r12
    jae rows_done

    mov rax, r12
    sub rax, r13
    cmp rax, 16
    jbe n_set
    mov rax, 16
n_set:
    mov r8, rax              # r8 = n (bytes in this row)

    lea r9, [rip + linebuf]  # r9 = line buffer base
    lea r10, [rip + hex_table]

    mov rcx, r13
    mov edx, 7
off_digit_loop:
    mov rax, rcx
    and rax, 0xF
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [r9 + rdx], al
    shr rcx, 4
    dec edx
    jns off_digit_loop

    mov byte ptr [r9 + 8], ' '
    mov byte ptr [r9 + 9], ' '

    mov r11, 10
    xor rcx, rcx
hexfield_loop:
    cmp rcx, 8
    jne skip_extra
    mov byte ptr [r9 + r11], ' '
    inc r11
skip_extra:
    cmp rcx, r8
    jl have_byte
    mov byte ptr [r9 + r11], ' '
    mov byte ptr [r9 + r11 + 1], ' '
    mov byte ptr [r9 + r11 + 2], ' '
    jmp hexfield_cont
have_byte:
    movzx eax, byte ptr [rbx + r13 + rcx]
    mov edx, eax
    shr edx, 4
    and edx, 0xF
    movzx edx, byte ptr [r10 + rdx]
    mov byte ptr [r9 + r11], dl
    and eax, 0xF
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [r9 + r11 + 1], al
    mov byte ptr [r9 + r11 + 2], ' '
hexfield_cont:
    add r11, 3
    inc rcx
    cmp rcx, 16
    jl hexfield_loop

    mov byte ptr [r9 + r11], ' '
    inc r11
    mov byte ptr [r9 + r11], '|'
    inc r11

    xor rcx, rcx
ascii_loop:
    cmp rcx, r8
    jge ascii_done
    movzx eax, byte ptr [rbx + r13 + rcx]
    cmp eax, 0x20
    jl dot_char
    cmp eax, 0x7e
    jg dot_char
    jmp store_char
dot_char:
    mov eax, '.'
store_char:
    mov byte ptr [r9 + r11], al
    inc r11
    inc rcx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r9 + r11], '|'
    inc r11
    mov byte ptr [r9 + r11], 10
    inc r11

    mov rdx, r11
    mov rsi, r9
    mov rdi, 1
    call write_all

    add r13, 16
    jmp row_loop

rows_done:
    lea r9, [rip + linebuf]
    lea r10, [rip + hex_table]
    mov rcx, r12
    mov edx, 7
final_digit_loop:
    mov rax, rcx
    and rax, 0xF
    movzx eax, byte ptr [r10 + rax]
    mov byte ptr [r9 + rdx], al
    shr rcx, 4
    dec edx
    jns final_digit_loop
    mov byte ptr [r9 + 8], 10

    mov rdx, 9
    mov rsi, r9
    mov rdi, 1
    call write_all

    mov rax, 60
    xor rdi, rdi
    syscall

write_all:
    cmp rdx, 0
    je wa_ret
wa_loop:
    mov rax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    cmp rdx, 0
    jne wa_loop
wa_ret:
    ret
```
