```asm
.section .bss
.align 16
inbuf:
.zero 65536
outbuf:
.zero 65537

.text
.globl _start
_start:
    xor r12, r12            # total bytes read so far

read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    mov rdx, 65536
    sub rdx, r12
    xor eax, eax
    syscall
    test rax, rax
    jz process
    add r12, rax
    jmp read_loop

process:
    xor r13, r13             # current line start index
    lea rbx, [rip + inbuf]
    lea r14, [rip + outbuf]

scan_loop:
    cmp r13, r12
    jge done
    mov r15, r13

find_nl:
    cmp r15, r12
    jge no_nl
    movzx eax, byte ptr [rbx + r15]
    cmp al, 10
    je found_nl
    inc r15
    jmp find_nl

found_nl:
    call reverse_and_print
    mov r13, r15
    inc r13
    jmp scan_loop

no_nl:
    cmp r13, r12
    jge done
    mov r15, r12
    call reverse_and_print
    mov r13, r12
    jmp scan_loop

done:
    xor edi, edi
    mov eax, 60
    syscall

reverse_and_print:
    mov r9, r15
    sub r9, r13               # len
    xor r10, r10              # i

copy_loop:
    cmp r10, r9
    jge after_copy
    mov rax, r13
    add rax, r9
    dec rax
    sub rax, r10
    movzx ecx, byte ptr [rbx + rax]
    mov byte ptr [r14 + r10], cl
    inc r10
    jmp copy_loop

after_copy:
    mov byte ptr [r14 + r9], 10
    mov rsi, r14
    mov rdx, r9
    inc rdx
    mov edi, 1
    mov eax, 1
    syscall
    ret
```
