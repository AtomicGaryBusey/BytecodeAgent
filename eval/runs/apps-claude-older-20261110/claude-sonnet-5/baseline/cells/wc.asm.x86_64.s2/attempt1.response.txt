.globl _start

.section .bss
.align 16
buffer: .space 65536

.text
_start:
    xor r12, r12            # lines
    xor r13, r13            # words
    xor r14, r14            # bytes
    xor r15, r15            # in-word flag (0 = not in word, 1 = in word)

    lea rbx, [rip + buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    mov rdx, 65536
    syscall
    test rax, rax
    jle done_reading

    add r14, rax
    mov rcx, rax
    xor rsi, rsi

process_byte:
    cmp rsi, rcx
    jge read_loop

    movzx eax, byte ptr [rbx + rsi]
    inc rsi

    cmp al, 0x0A
    jne not_nl
    inc r12
    jmp check_ws
not_nl:
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    je is_ws
    cmp al, 0x0B
    je is_ws
    cmp al, 0x0C
    je is_ws
    cmp al, 0x0D
    je is_ws
    jmp not_ws

is_ws:
check_ws:
    mov r15, 0
    jmp process_byte

not_ws:
    test r15, 1
    jnz process_byte
    mov r15, 1
    inc r13
    jmp process_byte

done_reading:
    lea rdi, [rip + outbuf]
    lea rsi, [rip + outbuf_end]
    mov rax, r12
    call write_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r13
    call write_num
    mov byte ptr [rdi], 0x20
    inc rdi
    mov rax, r14
    call write_num
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall

    mov rax, 60
    mov rdi, 0
    syscall

# write_num: converts rax to decimal ascii at rdi, advances rdi, uses rsi as scratch stack end
write_num:
    mov r8, rsi            # temp buffer end pointer (scratch area)
    mov r9, rdi            # save start of output area? not needed
    lea r10, [rip + numbuf + 24]
    mov r11, r10

    test rax, rax
    jnz wn_loop
    dec r11
    mov byte ptr [r11], '0'
    jmp wn_copy

wn_loop:
    test rax, rax
    jz wn_copy
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    dec r11
    mov byte ptr [r11], dl
    jmp wn_loop

wn_copy:
    mov rcx, r10
    sub rcx, r11
wn_copy_loop:
    test rcx, rcx
    jz wn_done
    mov al, byte ptr [r11]
    mov byte ptr [rdi], al
    inc rdi
    inc r11
    dec rcx
    jmp wn_copy_loop
wn_done:
    ret

.section .bss
.align 8
numbuf: .space 32

.section .bss
.align 8
outbuf: .space 64
outbuf_end:
