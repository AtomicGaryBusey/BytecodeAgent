```asm
// rpn: evaluate Reverse Polish Notation expressions, one per input line.
// AArch64, Linux, no libc.

.section .rodata
msg_error:
    .ascii "error\n"
msg_error_len = . - msg_error

.bss
.align 3
input_buf:      .space 65536
.align 3
line_stack:     .space 8192
.align 3
digits_tmp:     .space 24
.align 3
outbuf:         .space 32
.align 3
g_input_len:    .space 8
g_pos:          .space 8
g_line_start:   .space 8
g_line_end:     .space 8
g_tok_pos:      .space 8
g_error:        .space 8
g_stack_count:  .space 8

.text
.globl _start

_start:
    mov x19, #0
read_loop:
    mov x9, #65536
    sub x9, x9, x19
    cbz x9, read_done
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, x19
    mov x2, x9
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    beq read_done
    add x19, x19, x0
    b read_loop
read_done:
    adrp x9, g_input_len
    add x9, x9, :lo12:g_input_len
    str x19, [x9]
    adrp x9, g_pos
    add x9, x9, :lo12:g_pos
    str xzr, [x9]

main_loop:
    adrp x9, g_pos
    add x9, x9, :lo12:g_pos
    ldr x0, [x9]
    adrp x10, g_input_len
    add x10, x10, :lo12:g_input_len
    ldr x1, [x10]
    cmp x0, x1
    bge all_done

    adrp x11, g_line_start
    add x11, x11, :lo12:g_line_start
    str x0, [x11]

    mov x2, x0
find_nl:
    cmp x2, x1
    bge nl_not_found
    adrp x3, input_buf
    add x3, x3, :lo12:input_buf
    ldrb w4, [x3, x2]
    cmp w4, #10
    beq nl_found
    add x2, x2, #1
    b find_nl

nl_found:
    adrp x11, g_line_end
    add x11, x11, :lo12:g_line_end
    str x2, [x11]
    bl process_line
    adrp x11, g_line_end
    add x11, x11, :lo12:g_line_end
    ldr x2, [x11]
    add x2, x2, #1
    adrp x9, g_pos
    add x9, x9, :lo12:g_pos
    str x2, [x9]
    b main_loop

nl_not_found:
    adrp x11, g_line_end
    add x11, x11, :lo12:g_line_end
    str x2, [x11]
    bl process_line
    adrp x11, g_line_end
    add x11, x11, :lo12:g_line_end
    ldr x2, [x11]
    adrp x9, g_pos
    add x9, x9, :lo12:g_pos
    str x2, [x9]
    b main_loop

all_done:
    mov x0, #0
    mov x8, #93
    svc #0
halt:
    b halt

// ---------------------------------------------------------------
process_line:
    sub sp, sp, #16
    str x30, [sp]

    adrp x9, g_stack_count
    add x9, x9, :lo12:g_stack_count
    str xzr, [x9]
    adrp x9, g_error
    add x9, x9, :lo12:g_error
    str xzr, [x9]
    adrp x9, g_line_start
    add x9, x9, :lo12:g_line_start
    ldr x0, [x9]
    adrp x9, g_tok_pos
    add x9, x9, :lo12:g_tok_pos
    str x0, [x9]

token_loop:
    adrp x9, g_error
    add x9, x9, :lo12:g_error
    ldr x0, [x9]
    cbnz x0, line_done

    adrp x9, g_tok_pos
    add x9, x9, :lo12:g_tok_pos
    ldr x1, [x9]
    adrp x10, g_line_end
    add x10, x10, :lo12:g_line_end
    ldr x2, [x10]

skip_sep:
    cmp x1, x2
    bge line_done
    adrp x3, input_buf
    add x3, x3, :lo12:input_buf
    ldrb w4, [x3, x1]
    cmp w4, #0x20
    beq sep_adv
    cmp w4, #0x09
    beq sep_adv
    b token_found_start
sep_adv:
    add x1, x1, #1
    b skip_sep

token_found_start:
    mov x5, x1
scan_tok:
    add x1, x1, #1
    cmp x1, x2
    bge tok_end
    adrp x3, input_buf
    add x3, x3, :lo12:input_buf
    ldrb w4, [x3, x1]
    cmp w4, #0x20
    beq tok_end
    cmp w4, #0x09
    beq tok_end
    b scan_tok
tok_end:
    adrp x9, g_tok_pos
    add x9, x9, :lo12:g_tok_pos
    str x1, [x9]
    sub x6, x1, x5
    mov x0, x5
    mov x1, x6
    bl handle_token
    b token_loop

line_done:
    adrp x9, g_error
    add x9, x9, :lo12:g_error
    ldr x0, [x9]
    cbnz x0, PL_error
    adrp x9, g_stack_count
    add x9, x9, :lo12:g_stack_count
    ldr x0, [x9]
    cmp x0, #1
    bne PL_error
    adrp x9, line_stack
    add x9, x9, :lo12:line_stack
    ldr x0, [x9]
    bl print_number
    b PL_exit
PL_error:
    bl print_error
PL_exit:
    ldr x30, [sp]
    add sp, sp, #16
    ret

// ---------------------------------------------------------------
// x0 = token start index, x1 = token length
handle_token:
    sub sp, sp, #16
    str x30, [sp]

    mov x7, x0
    mov x8, x1
    adrp x6, input_buf
    add x6, x6, :lo12:input_buf

    cmp x8, #1
    bne try_literal
    ldrb w9, [x6, x7]
    cmp w9, #'+'
    beq do_add
    cmp w9, #'-'
    beq do_sub
    cmp w9, #'*'
    beq do_mul
    cmp w9, #'/'
    beq do_div
    cmp w9, #'%'
    beq do_mod

try_literal:
    mov x10, #0
    mov x11, x7
    mov x12, x8
    ldrb w9, [x6, x11]
    cmp w9, #'-'
    bne no_sign
    mov x10, #1
    add x11, x11, #1
    sub x12, x12, #1
no_sign:
    cmp x12, #1
    blt invalid_token
    cmp x12, #18
    bgt invalid_token
    mov x13, #0
    mov x14, #0
parse_digits:
    cmp x14, x12
    bge digits_done
    ldrb w9, [x6, x11]
    cmp w9, #'0'
    blt invalid_token
    cmp w9, #'9'
    bgt invalid_token
    sub w9, w9, #'0'
    mov x15, #10
    mul x13, x13, x15
    add x13, x13, w9, uxtw
    add x11, x11, #1
    add x14, x14, #1
    b parse_digits
digits_done:
    cbz x10, do_push_literal
    neg x13, x13
do_push_literal:
    mov x0, x13
    bl stack_push
    ldr x30, [sp]
    add sp, sp, #16
    ret

invalid_token:
    adrp x1, g_error
    add x1, x1, :lo12:g_error
    mov x0, #1
    str x0, [x1]
    ldr x30, [sp]
    add sp, sp, #16
    ret

do_add:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    cmp x2, #2
    blt invalid_token
    bl stack_pop
    mov x20, x0
    bl stack_pop
    mov x19, x0
    add x0, x19, x20
    bl stack_push
    ldr x30, [sp]
    add sp, sp, #16
    ret

do_sub:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    cmp x2, #2
    blt invalid_token
    bl stack_pop
    mov x20, x0
    bl stack_pop
    mov x19, x0
    sub x0, x19, x20
    bl stack_push
    ldr x30, [sp]
    add sp, sp, #16
    ret

do_mul:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    cmp x2, #2
    blt invalid_token
    bl stack_pop
    mov x20, x0
    bl stack_pop
    mov x19, x0
    mul x0, x19, x20
    bl stack_push
    ldr x30, [sp]
    add sp, sp, #16
    ret

do_div:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    cmp x2, #2
    blt invalid_token
    bl stack_pop
    mov x20, x0
    bl stack_pop
    mov x19, x0
    cbz x20, invalid_token
    mov x21, #1
    lsl x21, x21, #63
    cmp x19, x21
    bne div_ok
    cmn x20, #1
    beq invalid_token
div_ok:
    sdiv x0, x19, x20
    bl stack_push
    ldr x30, [sp]
    add sp, sp, #16
    ret

do_mod:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    cmp x2, #2
    blt invalid_token
    bl stack_pop
    mov x20, x0
    bl stack_pop
    mov x19, x0
    cbz x20, invalid_token
    mov x21, #1
    lsl x21, x21, #63
    cmp x19, x21
    bne mod_ok
    cmn x20, #1
    beq invalid_token
mod_ok:
    sdiv x22, x19, x20
    msub x0, x22, x20, x19
    bl stack_push
    ldr x30, [sp]
    add sp, sp, #16
    ret

// ---------------------------------------------------------------
// x0 = value to push
stack_push:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    adrp x3, line_stack
    add x3, x3, :lo12:line_stack
    str x0, [x3, x2, lsl #3]
    add x2, x2, #1
    str x2, [x1]
    ret

// returns x0 = popped value (assumes count >= 1)
stack_pop:
    adrp x1, g_stack_count
    add x1, x1, :lo12:g_stack_count
    ldr x2, [x1]
    sub x2, x2, #1
    str x2, [x1]
    adrp x3, line_stack
    add x3, x3, :lo12:line_stack
    ldr x0, [x3, x2, lsl #3]
    ret

// ---------------------------------------------------------------
// x0 = signed 64-bit value to print, followed by newline
print_number:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    cbnz x0, pn_nonzero
    mov w2, #'0'
    strb w2, [x1]
    mov x3, #1
    b pn_append_nl

pn_nonzero:
    mov x4, #0
    cmp x0, #0
    bge pn_skip_neg
    mov x4, #1
    neg x0, x0
pn_skip_neg:
    adrp x7, digits_tmp
    add x7, x7, :lo12:digits_tmp
    add x8, x7, #23
pn_divloop:
    mov x10, #10
    udiv x11, x0, x10
    msub x12, x11, x10, x0
    add w12, w12, #'0'
    strb w12, [x8]
    sub x8, x8, #1
    mov x0, x11
    cbnz x0, pn_divloop
    add x13, x8, #1
    add x14, x7, #24
    sub x3, x14, x13
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov x15, x1
    cbz x4, pn_copy_digits
    mov w2, #'-'
    strb w2, [x15]
    add x15, x15, #1
pn_copy_digits:
    mov x16, #0
pn_copy_loop:
    cmp x16, x3
    bge pn_copy_done
    ldrb w2, [x13, x16]
    strb w2, [x15, x16]
    add x16, x16, #1
    b pn_copy_loop
pn_copy_done:
    cbz x4, pn_append_nl
    add x3, x3, #1

pn_append_nl:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    mov w2, #10
    strb w2, [x1, x3]
    add x3, x3, #1
    mov x0, #1
    mov x2, x3
    mov x8, #64
    svc #0
    ret

// ---------------------------------------------------------------
print_error:
    adrp x1, msg_error
    add x1, x1, :lo12:msg_error
    mov x0, #1
    mov x2, #msg_error_len
    mov x8, #64
    svc #0
    ret
```
