.globl _start

.section .rodata
b64table:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
	.zero 49152
outbuf:
	.zero 70000

.text
_start:
	lea r8, [rip + inbuf]
	lea r9, [rip + outbuf]
	lea r10, [rip + b64table]
	xor r12, r12            # total bytes read so far

.read_loop:
	mov edi, 0
	lea rsi, [r8 + r12]
	mov rdx, 49152
	sub rdx, r12
	xor eax, eax
	syscall
	test rax, rax
	jz .read_done
	add r12, rax
	jmp .read_loop

.read_done:
	xor r14, r14            # input position
	xor r13, r13            # output position
	xor r15, r15            # chars in current line

.enc_loop:
	mov rax, r12
	sub rax, r14
	cmp rax, 3
	jl .enc_tail

	movzx ecx, byte ptr [r8 + r14]
	movzx edx, byte ptr [r8 + r14 + 1]
	movzx esi, byte ptr [r8 + r14 + 2]

	mov eax, ecx
	shr eax, 2
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13], al

	mov eax, ecx
	and eax, 3
	shl eax, 4
	mov ebx, edx
	shr ebx, 4
	or eax, ebx
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13 + 1], al

	mov eax, edx
	and eax, 0xF
	shl eax, 2
	mov ebx, esi
	shr ebx, 6
	or eax, ebx
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13 + 2], al

	mov eax, esi
	and eax, 0x3F
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13 + 3], al

	add r13, 4
	add r14, 3
	add r15, 4
	cmp r15, 76
	jne .enc_loop
	mov byte ptr [r9 + r13], 0x0A
	inc r13
	xor r15, r15
	jmp .enc_loop

.enc_tail:
	test rax, rax
	jnz .enc_tail_nonzero

	test r15, r15
	jz .enc_finish
	mov byte ptr [r9 + r13], 0x0A
	inc r13
	jmp .enc_finish

.enc_tail_nonzero:
	cmp rax, 1
	je .tail1

	# exactly 2 bytes remain
	movzx ecx, byte ptr [r8 + r14]
	movzx edx, byte ptr [r8 + r14 + 1]

	mov eax, ecx
	shr eax, 2
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13], al

	mov eax, ecx
	and eax, 3
	shl eax, 4
	mov ebx, edx
	shr ebx, 4
	or eax, ebx
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13 + 1], al

	mov eax, edx
	and eax, 0xF
	shl eax, 2
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13 + 2], al

	mov byte ptr [r9 + r13 + 3], 0x3D
	add r13, 4
	jmp .enc_write_final_nl

.tail1:
	movzx ecx, byte ptr [r8 + r14]

	mov eax, ecx
	shr eax, 2
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13], al

	mov eax, ecx
	and eax, 3
	shl eax, 4
	movzx eax, byte ptr [r10 + rax]
	mov byte ptr [r9 + r13 + 1], al

	mov byte ptr [r9 + r13 + 2], 0x3D
	mov byte ptr [r9 + r13 + 3], 0x3D
	add r13, 4

.enc_write_final_nl:
	mov byte ptr [r9 + r13], 0x0A
	inc r13

.enc_finish:
	xor r14, r14            # bytes written so far

.write_loop:
	cmp r14, r13
	jge .do_exit
	mov edi, 1
	lea rsi, [r9 + r14]
	mov rdx, r13
	sub rdx, r14
	mov eax, 1
	syscall
	add r14, rax
	jmp .write_loop

.do_exit:
	mov edi, 0
	mov eax, 60
	syscall
