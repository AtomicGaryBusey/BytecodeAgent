.globl _start

.text
_start:
    # Read all of stdin
    xor r12, r12
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    # Find '!' separator
    lea rdi, [rip + input_buffer]
    mov rcx, r12
    test rcx, rcx
    jz no_sep
    mov al, '!'
    repne scasb
    jne no_sep
    
    lea rsi, [rip + input_buffer]
    mov r13, rdi
    sub r13, rsi
    dec r13
    mov r14, rdi
    mov r15, r12
    sub r15, r13
    dec r15
    jmp check_brackets

no_sep:
    mov r13, r12
    lea r14, [rip + input_buffer]
    add r14, r12
    xor r15, r15

check_brackets:
    lea rdi, [rip + input_buffer]
    lea rsi, [rip + bracket_stack]
    xor r8, r8
    xor r9, r9
    
1:
    cmp r9, r13
    jge 2f
    
    movzx rax, byte ptr [rdi + r9]
    cmp al, '['
    je 3f
    cmp al, ']'
    je 4f
    
    inc r9
    jmp 1b
    
3:
    mov dword ptr [rsi + r8*4], r9d
    inc r8
    inc r9
    jmp 1b
    
4:
    test r8, r8
    jz unbalanced
    dec r8
    mov eax, dword ptr [rsi + r8*4]
    
    lea rcx, [rip + jump_table]
    mov dword ptr [rcx + r9*4], eax
    mov dword ptr [rcx + rax*4], r9d
    
    inc r9
    jmp 1b
    
2:
    test r8, r8
    jnz unbalanced
    jmp execute

unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall

execute:
    lea rbx, [rip + tape]
    xor r12, r12
    
1:
    cmp r12, r13
    jge done
    
    lea rdi, [rip + input_buffer]
    movzx rax, byte ptr [rdi + r12]
    
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_inc
    cmp al, '-'
    je cmd_dec
    cmp al, '.'
    je cmd_out
    cmp al, ','
    je cmd_in
    cmp al, '['
    je cmd_lbr
    cmp al, ']'
    je cmd_rbr
    
    inc r12
    jmp 1b

cmd_right:
    inc rbx
    inc r12
    jmp 1b

cmd_left:
    dec rbx
    inc r12
    jmp 1b

cmd_inc:
    inc byte ptr [rbx]
    inc r12
    jmp 1b

cmd_dec:
    dec byte ptr [rbx]
    inc r12
    jmp 1b

cmd_out:
    push r12
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, 1
    syscall
    pop r12
    inc r12
    jmp 1b

cmd_in:
    test r15, r15
    jz 2f
    movzx rax, byte ptr [r14]
    mov byte ptr [rbx], al
    inc r14
    dec r15
    jmp 3f
2:
    mov byte ptr [rbx], 0
3:
    inc r12
    jmp 1b

cmd_lbr:
    cmp byte ptr [rbx], 0
    jne 2f
    lea rdi, [rip + jump_table]
    mov r12d, dword ptr [rdi + r12*4]
2:
    inc r12
    jmp 1b

cmd_rbr:
    cmp byte ptr [rbx], 0
    je 2f
    lea rdi, [rip + jump_table]
    mov r12d, dword ptr [rdi + r12*4]
2:
    inc r12
    jmp 1b

done:
    mov rax, 60
    xor rdi, rdi
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
tape:
    .zero 30000
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
