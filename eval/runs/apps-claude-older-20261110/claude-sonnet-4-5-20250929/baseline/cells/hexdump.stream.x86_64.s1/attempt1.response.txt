```asm
.globl _start

.section .bss
.align 16
input_buffer:
    .zero 65536

.align 16
output_buffer:
    .zero 4096

.text
_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    
    add r12, rax
    jmp read_loop
    
read_done:
    xor r13, r13
    
process_loop:
    cmp r13, r12
    jge final_offset
    
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jle 1f
    mov r14, 16
1:
    
    lea rdi, [rip + output_buffer]
    
    mov rax, r13
    call print_hex8
    
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    
    xor r15, r15
hex_loop:
    cmp r15, 16
    jge hex_done
    
    cmp r15, r14
    jge hex_pad
    
    lea rsi, [rip + input_buffer]
    add rsi, r13
    add rsi, r15
    movzx rax, byte ptr [rsi]
    call print_hex2
    
    mov byte ptr [rdi], ' '
    inc rdi
    jmp hex_next
    
hex_pad:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], ' '
    inc rdi
    
hex_next:
    cmp r15, 7
    jne 1f
    mov byte ptr [rdi], ' '
    inc rdi
1:
    inc r15
    jmp hex_loop
    
hex_done:
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi
    
    xor r15, r15
ascii_loop:
    cmp r15, r14
    jge ascii_done
    
    lea rsi, [rip + input_buffer]
    add rsi, r13
    add rsi, r15
    movzx rax, byte ptr [rsi]
    
    cmp al, 0x20
    jb 1f
    cmp al, 0x7E
    ja 1f
    mov byte ptr [rdi], al
    jmp 2f
1:
    mov byte ptr [rdi], '.'
2:
    inc rdi
    inc r15
    jmp ascii_loop
    
ascii_done:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi
    
    lea rsi, [rip + output_buffer]
    sub rdi, rsi
    mov rdx, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    
    add r13, 16
    jmp process_loop
    
final_offset:
    lea rdi, [rip + output_buffer]
    mov rax, r12
    call print_hex8
    mov byte ptr [rdi], 10
    inc rdi
    
    lea rsi, [rip + output_buffer]
    sub rdi, rsi
    mov rdx, rdi
    mov rax, 1
    mov rdi, 1
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

print_hex8:
    push rax
    push rbx
    push rcx
    push rdx
    
    lea rbx, [rip + hex_digits]
    mov rcx, 28
    
1:
    mov rdx, rax
    shr rdx, cl
    and rdx, 0xF
    movzx rdx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub rcx, 4
    jge 1b
    
    pop rdx
    pop rcx
    pop rbx
    pop rax
    ret

print_hex2:
    push rax
    push rbx
    push rcx
    
    lea rbx, [rip + hex_digits]
    
    movzx rcx, al
    shr rcx, 4
    movzx rcx, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl
    inc rdi
    
    movzx rcx, al
    and rcx, 0xF
    movzx rcx, byte ptr [rbx + rcx]
    mov byte ptr [rdi], cl
    inc rdi
    
    pop rcx
    pop rbx
    pop rax
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"
```
