```asm
.globl _start

.bss
.align 8
line_count:
    .zero 8
word_count:
    .zero 8
byte_count:
    .zero 8
in_word:
    .zero 1

.align 16
buffer:
    .zero 65536

output_buffer:
    .zero 128

.text
_start:
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buffer]
    mov rdx, 65536
    syscall
    
    test rax, rax
    jz done_reading
    
    mov rcx, rax
    lea rsi, [rip + buffer]
    
process_loop:
    movzx eax, byte ptr [rsi]
    
    lea rdi, [rip + byte_count]
    inc qword ptr [rdi]
    
    cmp al, 0x0A
    jne check_word
    lea rdi, [rip + line_count]
    inc qword ptr [rdi]
    
check_word:
    cmp al, 0x09
    jb not_whitespace
    cmp al, 0x0D
    jbe is_whitespace
    cmp al, 0x20
    jne not_whitespace
    
is_whitespace:
    lea rdi, [rip + in_word]
    mov byte ptr [rdi], 0
    jmp next_byte
    
not_whitespace:
    lea rdi, [rip + in_word]
    movzx edx, byte ptr [rdi]
    test dl, dl
    jnz next_byte
    
    mov byte ptr [rdi], 1
    lea rdi, [rip + word_count]
    inc qword ptr [rdi]
    
next_byte:
    inc rsi
    dec rcx
    jnz process_loop
    
    jmp read_loop
    
done_reading:
    lea rdi, [rip + output_buffer]
    
    lea rsi, [rip + line_count]
    mov rax, qword ptr [rsi]
    call convert_to_decimal
    
    mov byte ptr [rdi], 0x20
    inc rdi
    
    lea rsi, [rip + word_count]
    mov rax, qword ptr [rsi]
    call convert_to_decimal
    
    mov byte ptr [rdi], 0x20
    inc rdi
    
    lea rsi, [rip + byte_count]
    mov rax, qword ptr [rsi]
    call convert_to_decimal
    
    mov byte ptr [rdi], 0x0A
    inc rdi
    
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

convert_to_decimal:
    push rbx
    push rcx
    push rdx
    push r9
    
    test rax, rax
    jnz not_zero
    mov byte ptr [rdi], 0x30
    inc rdi
    pop r9
    pop rdx
    pop rcx
    pop rbx
    ret
    
not_zero:
    mov rbx, rdi
    
divide_loop:
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, 0x30
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz divide_loop
    
    mov r9, rdi
    dec rdi
    mov rsi, rbx
    
reverse_loop:
    cmp rsi, rdi
    jge reverse_done
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp reverse_loop
    
reverse_done:
    mov rdi, r9
    pop r9
    pop rdx
    pop rcx
    pop rbx
    ret
```
