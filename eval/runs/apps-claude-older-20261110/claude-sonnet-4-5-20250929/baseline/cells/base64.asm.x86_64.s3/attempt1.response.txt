.globl _start

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .space 49152
output_buffer:
    .space 70000

.text
_start:
    xor rbx, rbx
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [input_buffer + rbx]
    mov rdx, 49152
    sub rdx, rbx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    test rbx, rbx
    jz exit_success
    
    xor r12, r12
    xor r13, r13
    xor r14, r14
    
encode_loop:
    mov rcx, rbx
    sub rcx, r12
    jz encode_done
    
    cmp rcx, 3
    jae encode_3bytes
    cmp rcx, 2
    je encode_2bytes
    jmp encode_1byte

encode_3bytes:
    movzx r10d, byte ptr [input_buffer + r12]
    shl r10d, 16
    movzx ecx, byte ptr [input_buffer + r12 + 1]
    shl ecx, 8
    or r10d, ecx
    movzx ecx, byte ptr [input_buffer + r12 + 2]
    or r10d, ecx
    
    mov ecx, r10d
    shr ecx, 18
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, r10d
    shr ecx, 12
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, r10d
    shr ecx, 6
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, r10d
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    add r12, 3
    jmp encode_loop

encode_2bytes:
    movzx r10d, byte ptr [input_buffer + r12]
    shl r10d, 16
    movzx ecx, byte ptr [input_buffer + r12 + 1]
    shl ecx, 8
    or r10d, ecx
    
    mov ecx, r10d
    shr ecx, 18
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, r10d
    shr ecx, 12
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, r10d
    shr ecx, 6
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, 61
    call output_char
    
    add r12, 2
    jmp encode_done

encode_1byte:
    movzx r10d, byte ptr [input_buffer + r12]
    shl r10d, 16
    
    mov ecx, r10d
    shr ecx, 18
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, r10d
    shr ecx, 12
    and ecx, 63
    movzx ecx, byte ptr [base64_alphabet + rcx]
    call output_char
    
    mov ecx, 61
    call output_char
    mov ecx, 61
    call output_char
    
    add r12, 1
    jmp encode_done

output_char:
    mov byte ptr [output_buffer + r13], cl
    inc r13
    inc r14
    cmp r14, 76
    jb 1f
    mov byte ptr [output_buffer + r13], 10
    inc r13
    xor r14, r14
1:
    ret

encode_done:
    test r14, r14
    jz write_output
    mov byte ptr [output_buffer + r13], 10
    inc r13

write_output:
    test r13, r13
    jz exit_success
    
    xor r15, r15
write_loop:
    mov rax, 1
    mov rdi, 1
    lea rsi, [output_buffer + r15]
    mov rdx, r13
    sub rdx, r15
    syscall
    add r15, rax
    cmp r15, r13
    jb write_loop

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
