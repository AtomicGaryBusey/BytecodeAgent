```asm
.globl _start

.section .bss
.align 16
input_buffer:
    .space 32768
line_array:
    .space 16000

.section .rodata
newline:
    .byte 0x0A

.section .text
_start:
    xor r12d, r12d
    
read_loop:
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 32768
    sub edx, r12d
    jle read_done
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12d, eax
    jmp read_loop

read_done:
    test r12d, r12d
    jz exit_ok
    
    lea r13, [rip + input_buffer]
    lea r14, [rip + line_array]
    xor r15d, r15d
    xor ebx, ebx
    
parse_next_line:
    cmp ebx, r12d
    jge done_parsing
    
    lea rsi, [r13 + rbx]
    mov rdi, rsi
    mov ecx, r12d
    sub ecx, ebx
    mov al, 0x0A
    repne scasb
    
    je found_newline
    
    sub rdi, rsi
    mov [r14], rsi
    mov [r14 + 8], rdi
    add r14, 16
    inc r15d
    jmp done_parsing
    
found_newline:
    mov rdx, rdi
    sub rdx, rsi
    dec rdx
    mov [r14], rsi
    mov [r14 + 8], rdx
    add r14, 16
    inc r15d
    
    sub rdi, r13
    mov ebx, edi
    jmp parse_next_line

done_parsing:
    test r15d, r15d
    jz exit_ok
    
    lea rdi, [rip + line_array]
    mov esi, r15d
    call sort_lines_func
    
    lea r13, [rip + line_array]
    xor r14d, r14d
    
output_loop:
    cmp r14d, r15d
    jge exit_ok
    
    mov rsi, [r13 + r14*16]
    mov rdx, [r13 + r14*16 + 8]
    mov edi, 1
    mov eax, 1
    syscall
    
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    
    inc r14d
    jmp output_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

compare_lines:
    push rbx
    
    mov r8, [rdi]
    mov r9, [rdi + 8]
    mov r10, [rsi]
    mov r11, [rsi + 8]
    
    mov rcx, r9
    cmp rcx, r11
    cmovg rcx, r11
    
    mov rdi, r8
    mov rsi, r10
    
    test rcx, rcx
    jz compare_by_length
    
    repe cmpsb
    jne not_equal
    
compare_by_length:
    mov rax, r9
    sub rax, r11
    pop rbx
    ret
    
not_equal:
    sbb rax, rax
    or rax, 1
    pop rbx
    ret

sort_lines_func:
    test esi, esi
    jz sort_done
    
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi
    mov r13d, esi
    
    xor r14d, r14d
outer_loop:
    mov r15d, r13d
    dec r15d
    cmp r14d, r15d
    jge sort_done_pop
    
    mov r15d, r14d
    
    mov ebx, r14d
    inc ebx
inner_loop:
    cmp ebx, r13d
    jge found_min
    
    lea rdi, [r12 + rbx*16]
    lea rsi, [r12 + r15*16]
    call compare_lines
    
    test rax, rax
    jge no_update
    
    mov r15d, ebx
    
no_update:
    inc ebx
    jmp inner_loop

found_min:
    cmp r14d, r15d
    je no_swap
    
    lea rsi, [r12 + r14*16]
    lea rdi, [r12 + r15*16]
    
    mov rax, [rsi]
    mov rdx, [rdi]
    mov [rsi], rdx
    mov [rdi], rax
    
    mov rax, [rsi + 8]
    mov rdx, [rdi + 8]
    mov [rsi + 8], rdx
    mov [rdi + 8], rax
    
no_swap:
    inc r14d
    jmp outer_loop

sort_done_pop:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
sort_done:
    ret
```
