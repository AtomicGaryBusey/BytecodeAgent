```asm
.globl _start

.bss
.align 16
input_buffer: .space 65536
tape: .space 30000
jump_table: .space 262144
bracket_stack: .space 4000

.text
_start:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor rcx, rcx
    lea rsi, [rip + input_buffer]
find_sep_loop:
    cmp rcx, r12
    jge no_separator
    cmp byte ptr [rsi + rcx], 0x21
    je found_separator
    inc rcx
    jmp find_sep_loop
found_separator:
    mov r13, rcx
    lea r14, [rsi + rcx + 1]
    mov r15, r12
    sub r15, rcx
    dec r15
    jmp build_jump_table
no_separator:
    mov r13, r12
    xor r14, r14
    xor r15, r15

build_jump_table:
    xor rcx, rcx
    xor r8, r8
    lea rsi, [rip + input_buffer]
    lea rdi, [rip + jump_table]
    lea r9, [rip + bracket_stack]
bracket_loop:
    cmp rcx, r13
    jge bracket_done
    movzx rax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je found_open
    cmp al, 0x5d
    je found_close
    inc rcx
    jmp bracket_loop
found_open:
    mov [r9 + r8*4], ecx
    inc r8
    inc rcx
    jmp bracket_loop
found_close:
    test r8, r8
    jz unbalanced
    dec r8
    mov eax, [r9 + r8*4]
    mov [rdi + rax*4], ecx
    mov [rdi + rcx*4], eax
    inc rcx
    jmp bracket_loop
bracket_done:
    test r8, r8
    jnz unbalanced
    jmp execute

unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall

execute:
    xor r12, r12
    xor rbx, rbx
    xor rbp, rbp
    lea r8, [rip + tape]
    lea r9, [rip + input_buffer]
    lea r10, [rip + jump_table]
exec_loop:
    cmp r12, r13
    jge exec_done
    movzx rax, byte ptr [r9 + r12]
    cmp al, 0x3e
    je cmd_right
    cmp al, 0x3c
    je cmd_left
    cmp al, 0x2b
    je cmd_inc
    cmp al, 0x2d
    je cmd_dec
    cmp al, 0x2e
    je cmd_output
    cmp al, 0x2c
    je cmd_input
    cmp al, 0x5b
    je cmd_open
    cmp al, 0x5d
    je cmd_close
    inc r12
    jmp exec_loop

cmd_right:
    inc rbx
    inc r12
    jmp exec_loop

cmd_left:
    dec rbx
    inc r12
    jmp exec_loop

cmd_inc:
    inc byte ptr [r8 + rbx]
    inc r12
    jmp exec_loop

cmd_dec:
    dec byte ptr [r8 + rbx]
    inc r12
    jmp exec_loop

cmd_output:
    mov rax, 1
    mov rdi, 1
    lea rsi, [r8 + rbx]
    mov rdx, 1
    syscall
    inc r12
    jmp exec_loop

cmd_input:
    cmp rbp, r15
    jge input_eof
    movzx rax, byte ptr [r14 + rbp]
    mov [r8 + rbx], al
    inc rbp
    inc r12
    jmp exec_loop
input_eof:
    mov byte ptr [r8 + rbx], 0
    inc r12
    jmp exec_loop

cmd_open:
    cmp byte ptr [r8 + rbx], 0
    jne cmd_open_nonzero
    mov r12d, [r10 + r12*4]
    inc r12
    jmp exec_loop
cmd_open_nonzero:
    inc r12
    jmp exec_loop

cmd_close:
    cmp byte ptr [r8 + rbx], 0
    je cmd_close_zero
    mov r12d, [r10 + r12*4]
    inc r12
    jmp exec_loop
cmd_close_zero:
    inc r12
    jmp exec_loop

exec_done:
    mov rax, 60
    xor rdi, rdi
    syscall
```
