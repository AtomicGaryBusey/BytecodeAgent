.globl _start

.section .rodata
error_msg:
    .ascii "error\n"
error_msg_len = . - error_msg

min_int_str:
    .ascii "-9223372036854775808\n"
min_int_len = . - min_int_str

.bss
    .align 16
input_buffer:
    .zero 65536

value_stack:
    .zero 8192

output_buffer:
    .zero 256

.text
_start:
    xor r12, r12
    
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
    
read_done:
    xor r13, r13
    
process_lines:
    cmp r13, r12
    jge exit_program
    
    call process_line
    jmp process_lines

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

process_line:
    mov r14, r13
    
find_eol:
    cmp r14, r12
    jge eol_found
    lea rax, [rip + input_buffer]
    cmp byte ptr [rax + r14], 0x0A
    je eol_found
    inc r14
    jmp find_eol
    
eol_found:
    lea rbx, [rip + value_stack]
    mov rbp, rbx
    mov r15, r13
    
token_loop:
skip_ws:
    cmp r15, r14
    jge check_result
    lea rax, [rip + input_buffer]
    movzx ecx, byte ptr [rax + r15]
    cmp cl, 0x20
    je next_ws
    cmp cl, 0x09
    je next_ws
    jmp token_start
next_ws:
    inc r15
    jmp skip_ws
    
token_start:
    mov rdi, r15
find_tok_end:
    cmp r15, r14
    jge tok_end
    lea rax, [rip + input_buffer]
    movzx ecx, byte ptr [rax + r15]
    cmp cl, 0x20
    je tok_end
    cmp cl, 0x09
    je tok_end
    inc r15
    jmp find_tok_end
    
tok_end:
    mov rsi, r15
    sub rsi, rdi
    
    call parse_token
    test rax, rax
    jz line_error
    
    jmp token_loop

check_result:
    sub rbp, rbx
    cmp rbp, 8
    jne line_error
    
    mov rax, qword ptr [rbx]
    call print_number
    
    mov r13, r14
    cmp r13, r12
    jge line_done
    inc r13
    
line_done:
    ret

line_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_msg_len
    syscall
    
    mov r13, r14
    cmp r13, r12
    jge line_done
    inc r13
    ret

parse_token:
    cmp rsi, 1
    jne try_number
    
    lea rax, [rip + input_buffer]
    movzx ecx, byte ptr [rax + rdi]
    
    cmp cl, '+'
    je op_add
    cmp cl, '-'
    je op_sub
    cmp cl, '*'
    je op_mul
    cmp cl, '/'
    je op_div
    cmp cl, '%'
    je op_mod
    
try_number:
    call parse_number
    test rax, rax
    jz tok_invalid
    
    mov qword ptr [rbp], rdx
    add rbp, 8
    mov rax, 1
    ret

op_add:
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r8, qword ptr [rbp]
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r9, qword ptr [rbp]
    add r9, r8
    mov qword ptr [rbp], r9
    add rbp, 8
    mov rax, 1
    ret

op_sub:
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r8, qword ptr [rbp]
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r9, qword ptr [rbp]
    sub r9, r8
    mov qword ptr [rbp], r9
    add rbp, 8
    mov rax, 1
    ret

op_mul:
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r8, qword ptr [rbp]
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r9, qword ptr [rbp]
    imul r9, r8
    mov qword ptr [rbp], r9
    add rbp, 8
    mov rax, 1
    ret

op_div:
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r8, qword ptr [rbp]
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r9, qword ptr [rbp]
    
    test r8, r8
    jz tok_invalid
    
    mov rax, 0x8000000000000000
    cmp r9, rax
    jne div_safe
    cmp r8, -1
    je tok_invalid
    
div_safe:
    mov rax, r9
    cqo
    idiv r8
    mov qword ptr [rbp], rax
    add rbp, 8
    mov rax, 1
    ret

op_mod:
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r8, qword ptr [rbp]
    cmp rbp, rbx
    jle tok_invalid
    sub rbp, 8
    mov r9, qword ptr [rbp]
    
    test r8, r8
    jz tok_invalid
    
    mov rax, 0x8000000000000000
    cmp r9, rax
    jne mod_safe
    cmp r8, -1
    je tok_invalid
    
mod_safe:
    mov rax, r9
    cqo
    idiv r8
    mov qword ptr [rbp], rdx
    add rbp, 8
    mov rax, 1
    ret

tok_invalid:
    xor rax, rax
    ret

parse_number:
    test rsi, rsi
    jz num_invalid
    
    lea rax, [rip + input_buffer]
    movzx ecx, byte ptr [rax + rdi]
    xor r10, r10
    cmp cl, '-'
    jne parse_digits
    
    mov r10, 1
    inc rdi
    dec rsi
    jz num_invalid
    
parse_digits:
    cmp rsi, 18
    jg num_invalid
    
    lea rax, [rip + input_buffer]
    add rax, rdi
    
    xor rdx, rdx
    xor r8, r8
    
digit_loop:
    cmp r8, rsi
    jge num_valid_check
    
    movzx ecx, byte ptr [rax + r8]
    
    cmp cl, '0'
    jl num_invalid
    cmp cl, '9'
    jg num_invalid
    
    sub cl, '0'
    imul rdx, 10
    movzx eax, cl
    add rdx, rax
    
    inc r8
    jmp digit_loop

num_valid_check:
    test r10, r10
    jz num_valid
    neg rdx

num_valid:
    mov rax, 1
    ret

num_invalid:
    xor rax, rax
    ret

print_number:
    mov r8, 0x8000000000000000
    cmp rax, r8
    je print_min_int
    
    lea rbx, [rip + output_buffer]
    xor rcx, rcx
    
    mov r8, rax
    test r8, r8
    jns pos_num
    
    mov byte ptr [rbx], '-'
    inc rcx
    neg r8
    
pos_num:
    test r8, r8
    jnz convert
    
    mov byte ptr [rbx + rcx], '0'
    inc rcx
    jmp write_out
    
convert:
    mov r9, rcx
    
conv_loop:
    test r8, r8
    jz reverse_digits
    
    xor rdx, rdx
    mov rax, r8
    mov r10, 10
    div r10
    
    add dl, '0'
    mov byte ptr [rbx + rcx], dl
    inc rcx
    mov r8, rax
    jmp conv_loop
    
reverse_digits:
    mov r10, r9
    mov r11, rcx
    dec r11
    
rev_loop:
    cmp r10, r11
    jge write_out
    
    movzx eax, byte ptr [rbx + r10]
    movzx edx, byte ptr [rbx + r11]
    mov byte ptr [rbx + r10], dl
    mov byte ptr [rbx + r11], al
    
    inc r10
    dec r11
    jmp rev_loop
    
write_out:
    mov byte ptr [rbx + rcx], 0x0A
    inc rcx
    
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, rcx
    syscall
    ret

print_min_int:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_int_str]
    mov rdx, min_int_len
    syscall
    ret
