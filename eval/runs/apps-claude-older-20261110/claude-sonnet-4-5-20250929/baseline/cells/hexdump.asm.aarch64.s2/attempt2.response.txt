.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buffer:
    .space 65536
output_buffer:
    .space 8192
output_pos:
    .space 8

.text
_start:
    adrp x0, output_pos
    add x0, x0, :lo12:output_pos
    str xzr, [x0]
    
    bl read_all
    mov x20, x0
    
    mov x19, #0
    
process_loop:
    cmp x19, x20
    bge print_final
    
    sub x0, x20, x19
    cmp x0, #16
    mov x1, #16
    csel x21, x0, x1, lo
    
    mov x0, x19
    mov x1, x21
    bl format_row
    
    add x19, x19, #16
    b process_loop
    
print_final:
    mov x0, x20
    bl print_offset_line
    bl flush_output
    
    mov x0, #0
    mov x8, #93
    svc #0

read_all:
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
    
1:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cmp x2, #0
    ble 2f
    
    mov x8, #63
    svc #0
    
    cmp x0, #0
    ble 2f
    
    add x20, x20, x0
    b 1b
    
2:
    mov x0, x20
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
    ret

format_row:
    stp x29, x30, [sp, #-64]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    
    mov x19, x0
    mov x20, x1
    
    mov x0, x19
    mov x1, #8
    bl print_hex_digits
    
    mov x0, #32
    bl put_char
    mov x0, #32
    bl put_char
    
    mov x21, #0
1:
    cmp x21, #16
    bge 2f
    
    cmp x21, x20
    bge 4f
    
    adrp x0, input_buffer
    add x0, x0, :lo12:input_buffer
    add x0, x0, x19
    ldrb w0, [x0, x21]
    bl print_hex_byte
    mov x0, #32
    bl put_char
    b 5f
    
4:
    mov x0, #32
    bl put_char
    mov x0, #32
    bl put_char
    mov x0, #32
    bl put_char
    
5:
    cmp x21, #7
    bne 3f
    mov x0, #32
    bl put_char
    
3:
    add x21, x21, #1
    b 1b
    
2:
    mov x0, #32
    bl put_char
    mov x0, #124
    bl put_char
    
    mov x21, #0
6:
    cmp x21, x20
    bge 7f
    
    adrp x0, input_buffer
    add x0, x0, :lo12:input_buffer
    add x0, x0, x19
    ldrb w0, [x0, x21]
    
    cmp w0, #0x20
    blt 8f
    cmp w0, #0x7E
    bgt 8f
    bl put_char
    b 9f
    
8:
    mov x0, #46
    bl put_char
    
9:
    add x21, x21, #1
    b 6b
    
7:
    mov x0, #124
    bl put_char
    mov x0, #10
    bl put_char
    
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #64
    ret

print_offset_line:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    
    mov x1, #8
    bl print_hex_digits
    mov x0, #10
    bl put_char
    
    ldp x29, x30, [sp], #16
    ret

print_hex_digits:
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    stp x19, x20, [sp, #16]
    
    mov x19, x0
    mov x20, x1
    
1:
    sub x20, x20, #1
    lsl x0, x20, #2
    lsr x0, x19, x0
    and x0, x0, #0xF
    
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits
    ldrb w0, [x1, x0]
    bl put_char
    
    cmp x20, #0
    bgt 1b
    
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
    ret

print_hex_byte:
    stp x29, x30, [sp, #-32]!
    mov x29, sp
    str x19, [sp, #16]
    
    mov x19, x0
    
    lsr x0, x19, #4
    and x0, x0, #0xF
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits

    and x0, x19, #0xF
    adrp x1, hex_digits
    add x1, x1, :lo12:hex_digits

    ldp x29, x30, [sp], #32

    mov x29, sp
    stp x19, x20, [sp, #16]
    
    mov x19, x0
    
    adrp x0, output_pos
    add x0, x0, :lo12:output_pos
    ldr x20, [x0]
    
    adrp x1, output_buffer
    add x1, x1, :lo12:output_buffer

    add x20, x20, #1
    str x20, [x0]
    
    cmp x20, #8192
    blt 1f
    bl flush_output

    ldp x19, x20, [sp, #16]

    add x1, x1, :lo12:output_buffer

    add x0, x0, :lo12:output_pos

    ldp x29, x30, [sp], #16
    ret
