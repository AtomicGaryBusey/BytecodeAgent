.globl _start

.section .bss
    .align 16
    input_buffer: .space 65536
    output_buffer: .space 65536

.section .text
_start:
    # Read all input into input_buffer
    xor r12d, r12d            # total bytes read = 0
    
read_loop:
    # read(0, input_buffer + r12, 65536 - r12)
    xor eax, eax              # syscall 0 = read
    xor edi, edi              # fd = 0
    lea rsi, [rip + input_buffer]
    add rsi, r12              # buffer + offset
    mov edx, 65536
    sub edx, r12d             # count = 65536 - total_read
    jz read_done              # if no space left, done
    syscall
    
    test rax, rax
    jz read_done              # if read returned 0, EOF
    
    add r12, rax              # total_read += bytes_read
    cmp r12, 65536
    jb read_loop              # continue if more space
    
read_done:
    # r12 = total input size
    test r12, r12
    jz exit_program           # if no input, exit
    
    # Process lines
    xor r13, r13              # current position in input
    
process_lines:
    cmp r13, r12
    jae exit_program          # if at end, exit
    
    # Find end of current line (newline or end of input)
    mov r14, r13              # search position
    lea rsi, [rip + input_buffer]
    
find_newline:
    cmp r14, r12
    jae found_end             # reached end without newline
    
    movzx eax, byte ptr [rsi + r14]
    inc r14
    cmp al, 0x0A
    jne find_newline
    
    # Found newline at r14-1
    # Line is from r13 to r14-1 (exclusive)
    mov r15, r14
    dec r15                   # r15 = position of newline
    jmp reverse_and_output
    
found_end:
    # Line is from r13 to r12 (exclusive)
    mov r15, r12
    
reverse_and_output:
    # Reverse line from r13 (inclusive) to r15 (exclusive) into output_buffer
    
    lea rdi, [rip + output_buffer]
    lea rsi, [rip + input_buffer]
    xor r8, r8                # output position
    
    # Copy in reverse
    mov rax, r15              # source position (going backwards from end)
    
reverse_copy_loop:
    cmp rax, r13
    jbe reverse_copy_done
    dec rax
    movzx ebx, byte ptr [rsi + rax]
    mov byte ptr [rdi + r8], bl
    inc r8
    jmp reverse_copy_loop
    
reverse_copy_done:
    # Add newline
    mov byte ptr [rdi + r8], 0x0A
    inc r8
    
    # Write output
    # write(1, output_buffer, r8)
    mov eax, 1                # syscall 1 = write
    mov edi, 1                # fd = 1 (stdout)
    lea rsi, [rip + output_buffer]
    mov rdx, r8               # count
    syscall
    
    # Move to next line
    mov r13, r14              # next line starts after newline (or at end)
    jmp process_lines
    
exit_program:
    # exit(0)
    mov eax, 60               # syscall 60 = exit
    xor edi, edi              # status = 0
    syscall
