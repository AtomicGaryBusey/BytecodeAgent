.globl _start

.bss
.align 16
buffer:
    .zero 65536

output_buffer:
    .zero 100

.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
    
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    
    test rax, rax
    jz done_reading
    
    add r14, rax
    
    lea rdi, [rip + buffer]
    mov rcx, rax
    
process_loop:
    movzx eax, byte ptr [rdi]
    inc rdi
    
    cmp al, 0x0A
    je is_newline
    
    cmp al, 0x20
    je is_whitespace
    cmp al, 0x09
    je is_whitespace
    cmp al, 0x0B
    je is_whitespace
    cmp al, 0x0C
    je is_whitespace
    cmp al, 0x0D
    je is_whitespace
    
    test r15, r15
    jnz next_byte
    
    inc r13
    mov r15, 1
    jmp next_byte
    
is_newline:
    inc r12
    
is_whitespace:
    xor r15, r15
    
next_byte:
    dec rcx
    jnz process_loop
    
    jmp read_loop
    
done_reading:
    mov rax, r12
    lea rdi, [rip + output_buffer]
    call uint_to_decimal
    
    mov byte ptr [rdi], 0x20
    inc rdi
    
    mov rax, r13
    call uint_to_decimal
    
    mov byte ptr [rdi], 0x20
    inc rdi
    
    mov rax, r14
    call uint_to_decimal
    
    mov byte ptr [rdi], 0x0A
    inc rdi
    
    lea rsi, [rip + output_buffer]
    sub rdi, rsi
    mov rdx, rdi
    
    mov eax, 1
    mov edi, 1
    syscall
    
    mov eax, 60
    xor edi, edi
    syscall

uint_to_decimal:
    push rbx
    push rcx
    push rdx
    push r8
    
    test rax, rax
    jnz 1f
    mov byte ptr [rdi], 0x30
    inc rdi
    pop r8
    pop rdx
    pop rcx
    pop rbx
    ret
    
1:
    mov rbx, rdi
    mov rcx, 10
    
2:
    xor edx, edx
    div rcx
    add dl, 0x30
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz 2b
    
    mov r8, rdi
    
    mov rsi, rbx
    dec rdi
    
3:
    cmp rsi, rdi
    jge 4f
    mov al, byte ptr [rsi]
    mov cl, byte ptr [rdi]
    mov byte ptr [rsi], cl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp 3b
    
4:
    mov rdi, r8
    
    pop r8
    pop rdx
    pop rcx
    pop rbx
    ret
