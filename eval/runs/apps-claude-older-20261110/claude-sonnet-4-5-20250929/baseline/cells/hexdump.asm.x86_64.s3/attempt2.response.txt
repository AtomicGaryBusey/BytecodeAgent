.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .align 16
input_buffer:
    .space 65536
output_buffer:
    .space 4096

.text
_start:
    xor r12, r12
    
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
    
read_done:
    xor r13, r13
    
process_loop:
    cmp r13, r12
    jge print_final_offset
    
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    lea rbx, [rip + output_buffer]
    
    mov rax, r13
    mov rdi, rbx
    call format_hex_8
    mov rbx, rdi
    
    mov word ptr [rbx], 0x2020
    add rbx, 2
    
    xor r15, r15
    
byte_loop:
    cmp r15, 16
    jge after_bytes
    
    cmp r15, r14
    jge empty_slot
    
    lea rsi, [rip + input_buffer]
    add rsi, r13
    add rsi, r15
    movzx eax, byte ptr [rsi]
    
    mov ecx, eax
    shr ecx, 4
    lea rsi, [rip + hex_digits]
    movzx ecx, byte ptr [rsi + rcx]
    mov byte ptr [rbx], cl
    
    and eax, 0x0f
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [rbx + 1], al
    
    mov byte ptr [rbx + 2], ' '
    add rbx, 3
    
    cmp r15, 7
    jne 1f
    mov byte ptr [rbx], ' '
    inc rbx
1:
    inc r15
    jmp byte_loop
    
empty_slot:
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], ' '
    mov byte ptr [rbx + 2], ' '
    add rbx, 3
    
    cmp r15, 7
    jne 1f
    mov byte ptr [rbx], ' '
    inc rbx
1:
    inc r15
    jmp byte_loop
    
after_bytes:
    mov byte ptr [rbx], ' '
    mov byte ptr [rbx + 1], '|'
    add rbx, 2
    
    xor r15, r15
    
ascii_loop:
    cmp r15, r14
    jge after_ascii
    
    lea rsi, [rip + input_buffer]
    add rsi, r13
    add rsi, r15
    movzx eax, byte ptr [rsi]
    
    cmp eax, 0x20
    jb print_dot
    cmp eax, 0x7e
    ja print_dot
    
    mov byte ptr [rbx], al
    inc rbx
    inc r15
    jmp ascii_loop
    
print_dot:
    mov byte ptr [rbx], '.'
    inc rbx
    inc r15
    jmp ascii_loop
    
after_ascii:
    mov byte ptr [rbx], '|'
    mov byte ptr [rbx + 1], 10
    add rbx, 2
    
    lea rsi, [rip + output_buffer]
    sub rbx, rsi
    mov rdx, rbx
    mov rax, 1
    mov rdi, 1
    syscall
    
    add r13, r14
    jmp process_loop
    
print_final_offset:
    lea rbx, [rip + output_buffer]
    mov rax, r12
    mov rdi, rbx
    call format_hex_8
    mov rbx, rdi
    
    mov byte ptr [rbx], 10
    inc rbx
    
    lea rsi, [rip + output_buffer]
    sub rbx, rsi
    mov rdx, rbx
    mov rax, 1
    mov rdi, 1
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

format_hex_8:
    push rcx
    push rdx
    push rsi
    
    lea rsi, [rip + hex_digits]
    shl rax, 28
    mov rcx, 8
    
1:
    rol rax, 4
    mov rdx, rax
    shr rdx, 60
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec rcx
    jnz 1b
    
    pop rsi
    pop rdx
    pop rcx
    ret
