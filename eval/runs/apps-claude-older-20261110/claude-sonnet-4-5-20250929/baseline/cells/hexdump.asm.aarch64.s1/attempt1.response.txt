.section .rodata

output_line:

.text

    svc #0

    add x19, x19, x0

    mov x21, #0

    bge print_final
    
    sub x22, x19, x20
    cmp x22, #16

    stp x19, x20, [sp, #16]

    mov x20, x1
    mov x21, x2

    add x22, x22, :lo12:output_line
    mov x23, x22
    
    mov x0, x21
    mov x1, x22
    bl write_hex8
    add x22, x22, #8
    
    mov w0, #' '
    strb w0, [x22], #1
    strb w0, [x22], #1

hex_loop:
    cmp x24, #16
    bge hex_done
    
    cmp x24, #8
    bne 1f
    mov w0, #' '
    strb w0, [x22], #1

    bge hex_blank

    mov x1, x22
    bl write_hex2
    add x22, x22, #2
    mov w0, #' '
    strb w0, [x22], #1

    mov w0, #' '
    strb w0, [x22], #1
    strb w0, [x22], #1
    strb w0, [x22], #1
    
hex_next:
    add x24, x24, #1

hex_done:

    strb w0, [x22], #1
    mov w0, #'|'
    strb w0, [x22], #1

ascii_loop:

    bge ascii_done
    
    ldrb w0, [x19, x24]
    cmp w0, #0x20
    blo ascii_dot
    cmp w0, #0x7e

    strb w0, [x22], #1

ascii_dot:

    strb w0, [x22], #1

    add x24, x24, #1
    b ascii_loop
    
ascii_done:

    strb w0, [x22], #1
    mov w0, #10
    strb w0, [x22], #1

    mov x1, x23

    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #64

    stp x19, x20, [sp, #16]

    mov x20, x1

    and w0, w0, #0xff
    add x1, x20, #4
    bl write_hex2

    add x2, x2, :lo12:hex_digits

    and w3, w3, #0xf
    ldrb w3, [x2, w3, uxtw]
    strb w3, [x1]
    
    and w0, w0, #0xf
    ldrb w0, [x2, w0, uxtw]
    strb w0, [x1, #1]
    
    ret

    mov x29, sp

    mov x8, #64
