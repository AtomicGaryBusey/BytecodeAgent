```asm
.globl _start

.text

_start:
    # Read all input
    xor r12, r12
    
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
    
read_done:
    test r12, r12
    jz exit_program
    
    # Parse lines
    xor r13, r13
    xor rdi, rdi
    lea r14, [rip + input_buffer]
    lea r15, [rip + lines]
    
parse_loop:
    cmp rdi, r12
    jge parse_done
    
    mov rsi, rdi
    
find_eol:
    cmp rdi, r12
    jge line_found
    movzx eax, byte ptr [r14 + rdi]
    inc rdi
    cmp al, 10
    jne find_eol
    dec rdi
    
line_found:
    lea rax, [r14 + rsi]
    mov qword ptr [r15 + r13*16], rax
    
    mov rax, rdi
    sub rax, rsi
    mov qword ptr [r15 + r13*16 + 8], rax
    
    inc r13
    
    cmp rdi, r12
    jge parse_loop
    inc rdi
    jmp parse_loop
    
parse_done:
    test r13, r13
    jz exit_program
    
    # Sort
    xor rdi, rdi
    mov rsi, r13
    dec rsi
    call quicksort
    
    # Output
    xor r12, r12
    
output_loop:
    cmp r12, r13
    jge exit_program
    
    lea r15, [rip + lines]
    mov rsi, qword ptr [r15 + r12*16]
    mov rdx, qword ptr [r15 + r12*16 + 8]
    
    test rdx, rdx
    jz write_newline
    
    mov rax, 1
    mov rdi, 1
    syscall
    
write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    
    inc r12
    jmp output_loop

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

quicksort:
    cmp rdi, rsi
    jge qs_ret
    
    push rbx
    push r12
    push r13
    
    mov r12, rdi
    mov r13, rsi
    
    call partition
    mov rbx, rax
    
    mov rdi, r12
    mov rsi, rbx
    dec rsi
    call quicksort
    
    mov rdi, rbx
    inc rdi
    mov rsi, r13
    call quicksort
    
    pop r13
    pop r12
    pop rbx
qs_ret:
    ret

partition:
    push rbx
    push r12
    push r13
    push r14
    
    mov r12, rdi
    dec r12
    mov r13, rdi
    mov r14, rsi
    
part_loop:
    cmp r13, r14
    jge part_done
    
    mov rdi, r13
    mov rsi, r14
    call compare_lines
    
    test rax, rax
    jg part_next
    
    inc r12
    mov rdi, r12
    mov rsi, r13
    call swap_lines
    
part_next:
    inc r13
    jmp part_loop
    
part_done:
    inc r12
    mov rdi, r12
    mov rsi, r14
    call swap_lines
    
    mov rax, r12
    
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

compare_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    lea rbx, [rip + lines]
    
    mov r12, qword ptr [rbx + rdi*16]
    mov r13, qword ptr [rbx + rdi*16 + 8]
    mov r14, qword ptr [rbx + rsi*16]
    mov r15, qword ptr [rbx + rsi*16 + 8]
    
    xor rcx, rcx
    
cmp_loop:
    cmp rcx, r13
    jge cmp_end1
    cmp rcx, r15
    jge cmp_end2
    
    movzx eax, byte ptr [r12 + rcx]
    movzx edx, byte ptr [r14 + rcx]
    
    cmp eax, edx
    jb cmp_less
    ja cmp_greater
    
    inc rcx
    jmp cmp_loop
    
cmp_end1:
    cmp rcx, r15
    jb cmp_less
    xor rax, rax
    jmp cmp_done
    
cmp_end2:
    mov rax, 1
    jmp cmp_done
    
cmp_less:
    mov rax, -1
    jmp cmp_done
    
cmp_greater:
    mov rax, 1
    
cmp_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

swap_lines:
    lea r8, [rip + lines]
    
    mov rax, qword ptr [r8 + rdi*16]
    mov rcx, qword ptr [r8 + rdi*16 + 8]
    mov rdx, qword ptr [r8 + rsi*16]
    mov r9, qword ptr [r8 + rsi*16 + 8]
    
    mov qword ptr [r8 + rdi*16], rdx
    mov qword ptr [r8 + rdi*16 + 8], r9
    mov qword ptr [r8 + rsi*16], rax
    mov qword ptr [r8 + rsi*16 + 8], rcx
    
    ret

.section .rodata
newline:
    .byte 10

.bss
    .align 16
input_buffer:
    .space 32768
    
    .align 16
lines:
    .space 16000
```
