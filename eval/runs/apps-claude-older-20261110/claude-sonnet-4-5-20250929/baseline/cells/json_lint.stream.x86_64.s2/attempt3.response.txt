.globl _start

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg

err_msg:
    .ascii "error at "
err_len = . - err_msg

newline:
    .byte 10

.bss
.align 16
input_buf:
    .zero 65536

num_buf:
    .zero 24

.text

_start:
    call read_all
    
    lea r12, [rip + input_buf]
    lea r13, [r12 + rax]
    mov rsi, r12
    
    call parse_doc
    
    test rax, rax
    jnz do_error
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    syscall
    
    xor edi, edi
    jmp do_exit

do_error:
    dec rax
    mov r14, rax
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, err_len
    syscall
    
    mov rax, r14
    call print_num
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    
    mov edi, 1

do_exit:
    mov rax, 60
    syscall

read_all:
    push rbx
    push r14
    
    xor rbx, rbx
    lea r14, [rip + input_buf]
    
1:
    cmp rbx, 65536
    jge 2f
    
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + rbx]
    mov rdx, 65536
    sub rdx, rbx
    syscall
    
    test rax, rax
    jle 2f
    
    add rbx, rax
    jmp 1b
    
2:
    mov rax, rbx
    pop r14
    pop rbx
    ret

print_num:
    push rbx
    push r12
    
    lea r12, [rip + num_buf]
    lea rbx, [r12 + 23]
    
    mov rcx, 10
1:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rbx
    mov byte ptr [rbx], dl
    test rax, rax
    jnz 1b
    
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    lea rdx, [r12 + 23]
    sub rdx, rbx
    syscall
    
    pop r12
    pop rbx
    ret

parse_doc:
    push rbx
    
    call skip_ws
    
    cmp rsi, r13
    jge 1f
    
    call parse_val
    test rax, rax
    jnz 2f
    
    call skip_ws
    
    cmp rsi, r13
    je 3f
    
    mov rax, rsi
    sub rax, r12
    inc rax
    jmp 2f
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    jmp 2f
    
3:
    xor rax, rax
    
2:
    pop rbx
    ret

skip_ws:
1:
    cmp rsi, r13
    jge 2f
    
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je 3f
    cmp al, 9
    je 3f
    cmp al, 10
    je 3f
    cmp al, 13
    jne 2f
    
3:
    inc rsi
    jmp 1b
    
2:
    ret

parse_val:
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    
    cmp al, '{'
    je parse_obj
    cmp al, '['
    je parse_arr
    cmp al, '"'
    je parse_str
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_num
    cmp al, '0'
    jl 1f
    cmp al, '9'
    jle parse_num
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret

parse_true:
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'r'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'u'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'e'
    jne 1f
    
    inc rsi
    xor rax, rax
    ret
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret

parse_false:
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'a'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'l'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 's'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'e'
    jne 1f
    
    inc rsi
    xor rax, rax
    ret
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret

parse_null:
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'u'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'l'
    jne 1f
    
    inc rsi
    cmp rsi, r13
    jge 1f
    cmp byte ptr [rsi], 'l'
    jne 1f
    
    inc rsi
    xor rax, rax
    ret
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret

parse_obj:
    inc rsi
    
    call skip_ws
    
    cmp rsi, r13
    jge 1f
    
    cmp byte ptr [rsi], '}'
    je 2f
    
    call parse_member
    test rax, rax
    jnz 3f
    
4:
    call skip_ws
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    cmp al, '}'
    je 2f
    cmp al, ','
    jne 1f
    
    inc rsi
    call skip_ws
    
    call parse_member
    test rax, rax
    jnz 3f
    
    jmp 4b
    
2:
    inc rsi
    xor rax, rax
    ret
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret
    
3:
    ret

parse_member:
    call parse_str
    test rax, rax
    jnz 1f
    
    call skip_ws
    
    cmp rsi, r13
    jge 2f
    
    cmp byte ptr [rsi], ':'
    jne 2f
    
    inc rsi
    call skip_ws
    
    call parse_val
    ret
    
2:
    mov rax, rsi
    sub rax, r12
    inc rax
1:
    ret

parse_arr:
    inc rsi
    
    call skip_ws
    
    cmp rsi, r13
    jge 1f
    
    cmp byte ptr [rsi], ']'
    je 2f
    
    call parse_val
    test rax, rax
    jnz 3f
    
4:
    call skip_ws
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    cmp al, ']'
    je 2f
    cmp al, ','
    jne 1f
    
    inc rsi
    call skip_ws
    
    call parse_val
    test rax, rax
    jnz 3f
    
    jmp 4b
    
2:
    inc rsi
    xor rax, rax
    ret
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret
    
3:
    ret

parse_str:
    cmp rsi, r13
    jge 1f
    
    cmp byte ptr [rsi], '"'
    jne 1f
    
    inc rsi
    
2:
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    
    cmp al, '"'
    je 3f
    
    cmp al, '\\'
    je 4f
    
    cmp al, 0x20
    jb 1f
    
    inc rsi
    jmp 2b
    
3:
    inc rsi
    xor rax, rax
    ret
    
4:
    inc rsi
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    
    cmp al, '"'
    je 5f
    cmp al, '\\'
    je 5f
    cmp al, '/'
    je 5f
    cmp al, 'b'
    je 5f
    cmp al, 'f'
    je 5f
    cmp al, 'n'
    je 5f
    cmp al, 'r'
    je 5f
    cmp al, 't'
    je 5f
    cmp al, 'u'
    je 6f
    
    jmp 1f
    
5:
    inc rsi
    jmp 2b
    
6:
    inc rsi
    
    mov ecx, 4
7:
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    
    cmp al, '0'
    jl 1f
    cmp al, '9'
    jle 8f
    cmp al, 'A'

    cmp al, 'F'

    cmp al, 'a'

    cmp al, 'f'
    jg 1f
    
8:
    inc rsi
    dec ecx
    jnz 7b
    jmp 2b
    
9:
    cmp al, 'a'

    cmp al, 'f'
    jle 8b
    
1:
    mov rax, rsi

    ret

parse_num:
    cmp rsi, r13
    jge 1f
    
    cmp byte ptr [rsi], '-'
    jne 2f
    inc rsi
    
2:
    cmp rsi, r13
    jge 1f
    
    movzx eax, byte ptr [rsi]
    
    cmp al, '0'
    je 3f
    
    cmp al, '1'
    jl 1f
    cmp al, '9'

    inc rsi

    cmp rsi, r13

    movzx eax, byte ptr [rsi]

    movzx eax, byte ptr [rsi]

    cmp al, '9'

12:

    cmp al, '0'
    jl 9f
    cmp al, '9'
    jg 9f

9:
    xor rax, rax
    ret
    
1:
    mov rax, rsi
    sub rax, r12
    inc rax
    ret
