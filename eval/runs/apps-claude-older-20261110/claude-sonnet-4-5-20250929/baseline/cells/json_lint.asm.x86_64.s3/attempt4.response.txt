.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

msg_error:
    .ascii "error at "
msg_error_len = . - msg_error

.bss
    .align 16
input_buffer:
    .zero 65536
input_len:
    .zero 8
current_pos:
    .zero 8
decimal_buf:
    .zero 32

.text

_start:
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov qword ptr [rip + input_len], r12
    mov qword ptr [rip + current_pos], 0
    
    call skip_ws
    call parse_value
    test rax, rax
    jz error_exit
    call skip_ws
    
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jne error_exit
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

error_exit:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_error]
    mov rdx, msg_error_len
    syscall
    
    mov rax, qword ptr [rip + current_pos]
    call print_decimal
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + decimal_buf]
    mov byte ptr [rsi], 10
    mov rdx, 1
    syscall
    
    mov rax, 60
    mov rdi, 1
    syscall

skip_ws:
1:
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 2f
    
    movzx ecx, byte ptr [input_buffer + rax]
    
    cmp cl, 0x20
    je 3f
    cmp cl, 0x09
    je 3f
    cmp cl, 0x0A
    je 3f
    cmp cl, 0x0D
    jne 2f
3:
    inc qword ptr [rip + current_pos]
    jmp 1b
2:
    ret

parse_value:
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 1f
    
    movzx ecx, byte ptr [input_buffer + rax]
    
    cmp cl, '{'
    je parse_object
    cmp cl, '['
    je parse_array
    cmp cl, '"'
    je parse_string
    cmp cl, 't'
    je parse_true
    cmp cl, 'f'
    je parse_false
    cmp cl, 'n'
    je parse_null
    cmp cl, '-'
    je parse_number
    cmp cl, '0'
    jl 1f
    cmp cl, '9'
    jg 1f
    jmp parse_number

1:
    xor rax, rax
    ret

parse_object:
    inc qword ptr [rip + current_pos]
    call skip_ws
    
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    cmp cl, '}'
    je 1f
    
2:
    call parse_string
    test rax, rax
    jz 9f
    
    call skip_ws
    
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    cmp cl, ':'
    jne 9f
    inc qword ptr [rip + current_pos]
    
    call skip_ws
    call parse_value
    test rax, rax
    jz 9f
    call skip_ws
    
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    cmp cl, '}'
    je 1f
    cmp cl, ','
    jne 9f
    
    inc qword ptr [rip + current_pos]
    call skip_ws
    jmp 2b
    
1:
    inc qword ptr [rip + current_pos]
    mov rax, 1
    ret

9:
    xor rax, rax
    ret

parse_array:
    inc qword ptr [rip + current_pos]
    call skip_ws
    
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    cmp cl, ']'
    je 1f
    
2:
    call parse_value
    test rax, rax

    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    cmp cl, ']'

    cmp cl, ','
    jne 9f
    
    inc qword ptr [rip + current_pos]

1:
    inc qword ptr [rip + current_pos]
    mov rax, 1
    ret

9:
    xor rax, rax
    ret

parse_string:
    push r12
    
    inc qword ptr [rip + current_pos]
    
1:
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    
    cmp cl, 0x20
    jb 9f
    
    cmp cl, '"'
    je 8f
    
    cmp cl, '\\'
    je 2f
    
    inc qword ptr [rip + current_pos]
    jmp 1b

2:
    inc qword ptr [rip + current_pos]
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx ecx, byte ptr [input_buffer + rax]
    
    cmp cl, '"'
    je 3f
    cmp cl, '\\'
    je 3f
    cmp cl, '/'
    je 3f
    cmp cl, 'b'
    je 3f
    cmp cl, 'f'
    je 3f
    cmp cl, 'n'
    je 3f
    cmp cl, 'r'
    je 3f
    cmp cl, 't'
    je 3f
    cmp cl, 'u'
    je 4f
    jmp 9f

3:
    inc qword ptr [rip + current_pos]
    jmp 1b

4:
    inc qword ptr [rip + current_pos]
    mov r12d, 4
5:
    mov rax, qword ptr [rip + current_pos]
    cmp rax, qword ptr [rip + input_len]
    jge 9f
    
    movzx edx, byte ptr [input_buffer + rax]
    
    cmp dl, '0'
    jl 9f
    cmp dl, '9'
    jle 6f
    cmp dl, 'A'
    jl 9f
    cmp dl, 'F'
    jle 6f
    cmp dl, 'a'
    jl 9f
    cmp dl, 'f'
    jg 9f

6:
    inc qword ptr [rip + current_pos]
    dec r12d
    jnz 5b
    jmp 1b

8:
    inc qword ptr [rip + current_pos]
    mov rax, 1
    pop r12
    ret

9:
    xor rax, rax
    pop r12
    ret

    movzx ecx, byte ptr [input_buffer + rax]
    
    cmp cl, '-'
    jne 1f
    inc qword ptr [rip + current_pos]

1:

print_decimal:
    push rbx
    push r12
    push r13
    
    test rax, rax
    jnz 1f
    lea rsi, [rip + decimal_buf]
    mov byte ptr [rsi], '0'
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    pop r13
    pop r12
    pop rbx
    ret

1:
    mov rbx, rax
    lea r12, [rip + decimal_buf]
    add r12, 31
    mov r13, r12
    
2:
    test rbx, rbx
    jz 3f
    mov rax, rbx
    xor rdx, rdx
    mov rcx, 10
    div rcx
    mov rbx, rax
    add dl, '0'
    dec r13
    mov byte ptr [r13], dl
    jmp 2b

3:
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    mov rdx, r12
    sub rdx, r13
    syscall
    
    pop r13
    pop r12
    pop rbx
    ret
