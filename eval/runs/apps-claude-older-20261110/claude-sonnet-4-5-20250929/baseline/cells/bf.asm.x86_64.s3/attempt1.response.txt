```assembly
.globl _start

.text
_start:
    # Read all of stdin into input_buffer
    xor r12, r12
    lea rsi, [rip + input_buffer]
    mov rdx, 65536
    
read_loop:
    mov rdi, 0
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add rsi, rax
    sub rdx, rax
    jz read_done
    jmp read_loop
    
read_done:
    # Find the '!' separator
    lea rdi, [rip + input_buffer]
    mov rcx, r12
    mov al, 0x21
    repne scasb
    jne no_separator
    
    # Found '!'
    lea rax, [rip + input_buffer]
    sub rdi, rax
    dec rdi
    mov r13, rdi
    inc rdi
    mov r14, rdi
    mov r15, r12
    sub r15, rdi
    jmp separator_done
    
no_separator:
    mov r13, r12
    xor r14, r14
    xor r15, r15
    
separator_done:
    # Check bracket balance and build jump table
    xor rbx, rbx
    xor rcx, rcx
    
bracket_check_loop:
    cmp rbx, r13
    jge bracket_check_done
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rbx]
    
    cmp al, '['
    je push_bracket
    cmp al, ']'
    je pop_bracket
    inc rbx
    jmp bracket_check_loop
    
push_bracket:
    lea rsi, [rip + bracket_stack]
    mov dword ptr [rsi + rcx * 4], ebx
    inc rcx
    inc rbx
    jmp bracket_check_loop
    
pop_bracket:
    test rcx, rcx
    jz unbalanced
    dec rcx
    lea rsi, [rip + bracket_stack]
    mov edi, dword ptr [rsi + rcx * 4]
    
    lea rsi, [rip + jump_table]
    lea eax, [rbx + 1]
    mov dword ptr [rsi + rdi * 4], eax
    lea eax, [rdi + 1]
    mov dword ptr [rsi + rbx * 4], eax
    
    inc rbx
    jmp bracket_check_loop
    
bracket_check_done:
    test rcx, rcx
    jnz unbalanced
    
    # Execute the program
    xor rbx, rbx
    xor r8, r8
    xor r9, r9
    
execute_loop:
    cmp rbx, r13
    jge execute_done
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rbx]
    
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_inc
    cmp al, '-'
    je cmd_dec
    cmp al, '.'
    je cmd_output
    cmp al, ','
    je cmd_input
    cmp al, '['
    je cmd_loop_start
    cmp al, ']'
    je cmd_loop_end
    
    inc rbx
    jmp execute_loop
    
cmd_right:
    inc r8
    inc rbx
    jmp execute_loop
    
cmd_left:
    dec r8
    inc rbx
    jmp execute_loop
    
cmd_inc:
    lea rsi, [rip + tape]
    inc byte ptr [rsi + r8]
    inc rbx
    jmp execute_loop
    
cmd_dec:
    lea rsi, [rip + tape]
    dec byte ptr [rsi + r8]
    inc rbx
    jmp execute_loop
    
cmd_output:
    lea rsi, [rip + tape]
    add rsi, r8
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc rbx
    jmp execute_loop
    
cmd_input:
    lea rsi, [rip + tape]
    cmp r9, r15
    jge input_exhausted
    
    lea rdi, [rip + input_buffer]
    add rdi, r14
    movzx eax, byte ptr [rdi + r9]
    mov byte ptr [rsi + r8], al
    inc r9
    inc rbx
    jmp execute_loop
    
input_exhausted:
    mov byte ptr [rsi + r8], 0
    inc rbx
    jmp execute_loop
    
cmd_loop_start:
    lea rsi, [rip + tape]
    cmp byte ptr [rsi + r8], 0
    jne loop_start_continue
    
    lea rsi, [rip + jump_table]
    mov ebx, dword ptr [rsi + rbx * 4]
    jmp execute_loop
    
loop_start_continue:
    inc rbx
    jmp execute_loop
    
cmd_loop_end:
    lea rsi, [rip + tape]
    cmp byte ptr [rsi + r8], 0
    je loop_end_continue
    
    lea rsi, [rip + jump_table]
    mov ebx, dword ptr [rsi + rbx * 4]
    jmp execute_loop
    
loop_end_continue:
    inc rbx
    jmp execute_loop
    
execute_done:
    mov rdi, 0
    mov rax, 60
    syscall
    
unbalanced:
    mov rdi, 1
    mov rax, 60
    syscall

.bss
.align 8
input_buffer:
    .zero 65536
tape:
    .zero 30000
jump_table:
    .zero 262144
bracket_stack:
    .zero 4000
```
