.globl _start

.section .bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65536

.text
_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jle read_done
    
    add r12, rax
    cmp r12, 65536
    jb read_loop
    
read_done:
    test r12, r12
    jz exit_program
    
    xor r13, r13
    
process_line:
    cmp r13, r12
    jae exit_program
    
    mov r14, r13
    
find_line_end:
    cmp r13, r12
    jae found_end_of_input
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x0A
    je found_newline
    
    inc r13
    jmp find_line_end
    
found_newline:
    mov r15, r13
    sub r15, r14
    
    call reverse_and_output
    
    inc r13
    jmp process_line
    
found_end_of_input:
    mov r15, r13
    sub r15, r14
    
    test r15, r15
    jz exit_program
    
    call reverse_and_output
    jmp exit_program

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

reverse_and_output:
    xor rcx, rcx
    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    
    mov rbx, r15
reverse_loop:
    test rbx, rbx
    jz reverse_done
    
    dec rbx
    movzx eax, byte ptr [rsi + r14 + rbx]
    mov byte ptr [rdi + rcx], al
    inc rcx
    jmp reverse_loop
    
reverse_done:
    mov byte ptr [rdi + rcx], 0x0A
    inc rcx
    
    push r12
    push r13
    push r14
    push r15
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, rcx
    syscall
    
    pop r15
    pop r14
    pop r13
    pop r12
    ret
