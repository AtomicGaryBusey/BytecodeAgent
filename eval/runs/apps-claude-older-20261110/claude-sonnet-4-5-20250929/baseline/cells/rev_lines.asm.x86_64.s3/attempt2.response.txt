.globl _start

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_ok
    
    xor r14, r14
    lea r15, [rip + output_buffer]

process_lines:
    cmp r14, r12
    jge exit_ok
    
    mov rdi, r14

find_newline:
    cmp rdi, r12
    jge line_without_newline
    
    movzx eax, byte ptr [r13 + rdi]
    cmp al, 10
    je found_newline
    
    inc rdi
    jmp find_newline

found_newline:
    mov rcx, r14
    mov rdx, rdi
    xor rbx, rbx

reverse_with_newline:
    cmp rdx, rcx
    jle reverse_with_newline_done
    dec rdx
    movzx eax, byte ptr [r13 + rdx]
    mov byte ptr [r15 + rbx], al
    inc rbx
    jmp reverse_with_newline

reverse_with_newline_done:
    mov byte ptr [r15 + rbx], 10
    inc rbx
    
    mov rax, 1
    push rdi
    mov rdi, 1
    mov rsi, r15
    mov rdx, rbx
    syscall
    pop rdi
    
    lea r14, [rdi + 1]
    jmp process_lines

line_without_newline:
    mov rcx, r14
    mov rdx, rdi
    xor rbx, rbx

reverse_without_newline:
    cmp rdx, rcx
    jle reverse_without_newline_done
    dec rdx
    movzx eax, byte ptr [r13 + rdx]
    mov byte ptr [r15 + rbx], al
    inc rbx
    jmp reverse_without_newline

reverse_without_newline_done:
    mov byte ptr [r15 + rbx], 10
    inc rbx
    
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, rbx
    syscall

exit_ok:
    mov rax, 60
    xor rdi, rdi
    syscall
