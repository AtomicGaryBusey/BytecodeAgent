.globl _start

.text

_start:
    adrp x0, input_buffer
    add x0, x0, :lo12:input_buffer
    mov x19, x0
    mov x20, #0

read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x19
    add x1, x1, x20
    mov x2, #32768
    sub x2, x2, x20
    svc #0
    
    cmp x0, #0
    beq read_done
    add x20, x20, x0
    cmp x20, #32768
    blt read_loop
    
read_done:
    mov x21, #0
    mov x22, #0
    
    adrp x23, line_starts
    add x23, x23, :lo12:line_starts
    adrp x24, line_lengths
    add x24, x24, :lo12:line_lengths
    
    cmp x20, #0
    beq skip_parsing
    
    mov x0, x19
    mov x1, #0
    
parse_loop:
    cmp x22, x20
    bge end_of_input
    
    ldrb w2, [x19, x22]
    add x22, x22, #1
    
    cmp w2, #10
    beq found_newline
    
    add x1, x1, #1
    b parse_loop
    
found_newline:
    str x0, [x23, x21, lsl #3]
    str w1, [x24, x21, lsl #2]
    add x21, x21, #1
    
    add x0, x19, x22
    mov x1, #0
    b parse_loop
    
end_of_input:
    cmp x1, #0
    beq skip_parsing
    str x0, [x23, x21, lsl #3]
    str w1, [x24, x21, lsl #2]
    add x21, x21, #1
    
skip_parsing:
    cmp x21, #0
    beq done
    
    mov x0, #0
    sub x1, x21, #1
    bl quicksort
    
    mov x22, #0
output_loop:
    cmp x22, x21
    bge done
    
    ldr x0, [x23, x22, lsl #3]
    ldr w1, [x24, x22, lsl #2]
    
    cmp x1, #0
    beq skip_write
    mov x8, #64
    mov x2, x1
    mov x1, x0
    mov x0, #1
    svc #0
    
skip_write:
    adrp x0, newline
    add x1, x0, :lo12:newline
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0
    
    add x22, x22, #1
    b output_loop
    
done:
    mov x8, #93
    mov x0, #0
    svc #0

quicksort:
    cmp x0, x1
    bge quicksort_ret
    
    stp x29, x30, [sp, #-64]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x0, x1, [sp, #48]
    
    mov x2, x0
    mov x3, x1
    bl partition
    
    ldp x2, x3, [sp, #48]
    mov x19, x0
    
    mov x0, x2
    sub x1, x19, #1
    bl quicksort
    
    ldp x2, x3, [sp, #48]
    add x0, x19, #1
    mov x1, x3
    bl quicksort
    
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x29, x30, [sp], #64
quicksort_ret:
    ret

partition:
    stp x29, x30, [sp, #-64]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x25, x26, [sp, #48]
    
    mov x19, x2
    mov x20, x2
    mov x21, x3
    
    ldr x22, [x23, x21, lsl #3]
    ldr w25, [x24, x21, lsl #2]
    
partition_loop:
    cmp x20, x21
    bge partition_done
    
    ldr x0, [x23, x20, lsl #3]
    ldr w1, [x24, x20, lsl #2]
    mov x2, x22
    mov x3, x25
    bl compare_lines
    
    cmp x0, #0
    bge partition_skip_swap
    
    mov x0, x19
    mov x1, x20
    bl swap_lines
    add x19, x19, #1
    
partition_skip_swap:
    add x20, x20, #1
    b partition_loop
    
partition_done:
    mov x0, x19
    mov x1, x21
    bl swap_lines
    
    mov x0, x19
    
    ldp x25, x26, [sp, #48]
    ldp x19, x20, [sp, #16]
    ldp x21, x22, [sp, #32]
    ldp x29, x30, [sp], #64
    ret

compare_lines:
    mov x4, x0
    mov x5, x2
    mov x6, x1
    mov x7, x3
    
    mov x8, #0
compare_loop:
    cmp x8, x6
    bge compare_end1
    cmp x8, x7
    bge compare_end2
    
    ldrb w9, [x4, x8]
    ldrb w10, [x5, x8]
    
    cmp w9, w10
    bne compare_diff
    
    add x8, x8, #1
    b compare_loop
    
compare_diff:
    sub x0, x9, x10
    ret
    
compare_end1:
    cmp x8, x7
    beq compare_equal
    mov x0, #-1
    ret
    
compare_end2:
    mov x0, #1
    ret
    
compare_equal:
    mov x0, #0
    ret

swap_lines:
    ldr x2, [x23, x0, lsl #3]
    ldr w3, [x24, x0, lsl #2]
    ldr x4, [x23, x1, lsl #3]
    ldr w5, [x24, x1, lsl #2]
    
    str x4, [x23, x0, lsl #3]
    str w5, [x24, x0, lsl #2]
    str x2, [x23, x1, lsl #3]
    str w3, [x24, x1, lsl #2]
    ret
