```asm
.globl _start

.section .bss
.align 16
input_buffer: .zero 65536
output_buffer: .zero 65536

.text
_start:
    # Read all input
    xor r12, r12                # r12 = total bytes read
    
read_loop:
    mov rax, 0                  # syscall: read
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + input_buffer]
    add rsi, r12                # buf = input_buffer + bytes_read
    mov rdx, 65536
    sub rdx, r12                # count = 65536 - bytes_read
    jz read_done                # if no space left, done
    syscall
    
    test rax, rax
    jle read_done               # if <= 0, done reading
    add r12, rax                # total_bytes_read += bytes_read
    cmp r12, 65536
    jl read_loop
    
read_done:
    # r12 = total bytes in input
    test r12, r12
    jz exit_program             # if no input, exit
    
    # Process lines
    xor r13, r13                # r13 = current position in input
    
process_lines:
    cmp r13, r12
    jge exit_program            # if we've processed all input, exit
    
    # Find end of line (newline or end of input)
    mov r14, r13                # r14 = start of line
    mov r15, r13                # r15 = current search position
    lea rsi, [rip + input_buffer]
    
find_eol:
    cmp r15, r12
    jge found_eol               # reached end of input
    movzx eax, byte ptr [rsi + r15]
    cmp al, 0x0A
    je found_eol
    inc r15
    jmp find_eol
    
found_eol:
    # r14 = start of line
    # r15 = position of newline or end of input
    # line length = r15 - r14
    
    # Reverse the line into output_buffer
    mov rcx, r15
    sub rcx, r14                # rcx = line length
    lea rdi, [rip + output_buffer]
    
    # Copy bytes in reverse
    xor rbx, rbx                # rbx = output position
    
reverse_loop:
    test rcx, rcx
    jz reverse_done
    dec rcx
    lea r8, [r14 + rcx]
    movzx eax, byte ptr [rsi + r8]
    mov byte ptr [rdi + rbx], al
    inc rbx
    jmp reverse_loop
    
reverse_done:
    # Add newline
    mov byte ptr [rdi + rbx], 0x0A
    inc rbx
    
    # Write output
    mov rax, 1                  # syscall: write
    mov rdi, 1                  # fd = stdout
    lea rsi, [rip + output_buffer]
    mov rdx, rbx                # count = bytes to write
    syscall
    
    # Move to next line
    mov r13, r15
    cmp r13, r12
    jge exit_program
    # At this point, input_buffer[r13] is definitely a newline, so skip it
    inc r13
    jmp process_lines
    
exit_program:
    mov rax, 60                 # syscall: exit
    xor rdi, rdi                # status = 0
    syscall
```
