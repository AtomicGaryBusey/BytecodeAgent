.text
.globl _start

_start:
    adrp x0, input_buffer
    add x19, x0, :lo12:input_buffer
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #32768, lsl #0
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x0, line_array
    add x21, x0, :lo12:line_array
    mov x22, #0
    mov x23, #0
    cbz x20, write_output

parse_loop:
    cmp x23, x20
    b.ge parse_done
    mov x1, x23

find_newline:
    cmp x1, x20
    b.ge last_line
    add x2, x19, x1
    ldrb w3, [x2]
    cmp w3, #10
    b.eq found_newline
    add x1, x1, #1
    b find_newline

found_newline:
    sub x2, x1, x23
    add x3, x19, x23
    lsl x4, x22, #4
    str x3, [x21, x4]
    add x4, x4, #8
    str x2, [x21, x4]
    add x22, x22, #1
    add x23, x1, #1
    b parse_loop

last_line:
    sub x2, x20, x23
    cbz x2, parse_done
    add x3, x19, x23
    lsl x4, x22, #4
    str x3, [x21, x4]
    add x4, x4, #8
    str x2, [x21, x4]
    add x22, x22, #1

parse_done:
    cbz x22, write_output
    mov x0, x21
    mov x1, #0
    sub x2, x22, #1
    bl quicksort

write_output:
    mov x23, #0

write_loop:
    cmp x23, x22
    b.ge exit_program
    lsl x4, x23, #4
    ldr x5, [x21, x4]
    add x4, x4, #8
    ldr x6, [x21, x4]
    cbz x6, write_newline
    mov x0, #1
    mov x1, x5
    mov x2, x6
    mov x8, #64
    svc #0

write_newline:
    adrp x0, newline
    add x1, x0, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    add x23, x23, #1
    b write_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

quicksort:
    cmp x1, x2
    b.ge quicksort_ret
    stp x29, x30, [sp, #-64]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x0, x19
    mov x1, x20
    mov x2, x21
    bl partition
    mov x22, x0
    mov x0, x19
    mov x1, x20
    sub x2, x22, #1
    bl quicksort
    mov x0, x19
    add x1, x22, #1
    mov x2, x21
    bl quicksort
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #64

quicksort_ret:
    ret

partition:
    stp x29, x30, [sp, #-80]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    mov x19, x0
    mov x20, x1
    mov x21, x2
    lsl x4, x21, #4
    ldr x22, [x19, x4]
    add x4, x4, #8
    ldr x23, [x19, x4]
    sub x24, x20, #1
    mov x25, x20

partition_loop:
    cmp x25, x21
    b.ge partition_done
    lsl x4, x25, #4
    ldr x0, [x19, x4]
    add x4, x4, #8
    ldr x1, [x19, x4]
    mov x2, x22
    mov x3, x23
    bl compare_lines
    cmp x0, #0
    b.gt partition_next
    add x24, x24, #1
    mov x0, x19
    mov x1, x24
    mov x2, x25
    bl swap_lines

partition_next:
    add x25, x25, #1
    b partition_loop

partition_done:
    add x24, x24, #1
    mov x0, x19
    mov x1, x24
    mov x2, x21
    bl swap_lines
    mov x0, x24
    ldp x25, x26, [sp, #64]
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #80
    ret

compare_lines:
    mov x4, #0

compare_loop:
    cmp x4, x1
    b.ge compare_line1_end
    cmp x4, x3
    b.ge compare_line2_end
    ldrb w5, [x0, x4]
    ldrb w6, [x2, x4]
    cmp w5, w6
    b.lt compare_less
    b.gt compare_greater
    add x4, x4, #1
    b compare_loop

compare_line1_end:
    cmp x4, x3
    b.ge compare_equal
    b compare_less

compare_line2_end:
    b compare_greater

compare_less:
    mov x0, #-1
    ret

compare_equal:
    mov x0, #0
    ret

compare_greater:
    mov x0, #1
    ret

swap_lines:
    cmp x1, x2
    b.eq swap_ret
    lsl x5, x1, #4
    ldr x3, [x0, x5]
    add x5, x5, #8
    ldr x4, [x0, x5]
    lsl x5, x2, #4
    ldr x6, [x0, x5]
    add x5, x5, #8
    ldr x7, [x0, x5]
    lsl x5, x1, #4
    str x6, [x0, x5]
    add x5, x5, #8
    str x7, [x0, x5]
    lsl x5, x2, #4
    str x3, [x0, x5]
    add x5, x5, #8
    str x4, [x0, x5]

swap_ret:
    ret
