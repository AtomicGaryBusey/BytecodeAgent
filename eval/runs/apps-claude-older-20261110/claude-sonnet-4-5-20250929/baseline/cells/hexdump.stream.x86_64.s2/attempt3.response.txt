.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .space 65536
output_buffer:
    .space 8192

.text
_start:
    xor r12, r12
    lea r13, [rip + input_buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r14, r14
    
process_loop:
    cmp r14, r12
    jge print_final_offset
    
    mov rcx, r12
    sub rcx, r14
    cmp rcx, 16
    jbe 1f
    mov rcx, 16
1:
    push rcx
    
    lea r15, [rip + output_buffer]
    mov rdi, r14
    call print_hex8
    
    mov word ptr [r15], 0x2020
    add r15, 2
    
    pop rcx
    push rcx
    xor rsi, rsi
    
hex_loop:
    cmp rsi, 16
    jge hex_done
    
    cmp rsi, 8
    jne 2f
    mov byte ptr [r15], ' '
    inc r15
2:
    
    cmp rsi, rcx
    jge print_empty_byte
    
    lea rdx, [r14 + rsi]
    movzx rax, byte ptr [r13 + rdx]
    mov rbx, rax
    shr rax, 4
    and rbx, 15
    lea rdi, [rip + hex_digits]
    movzx rax, byte ptr [rdi + rax]
    mov byte ptr [r15], al
    inc r15
    movzx rbx, byte ptr [rdi + rbx]
    mov byte ptr [r15], bl
    inc r15
    mov byte ptr [r15], ' '
    inc r15
    inc rsi
    jmp hex_loop
    
print_empty_byte:
    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], ' '
    inc r15
    inc rsi
    jmp hex_loop

hex_done:
    mov byte ptr [r15], ' '
    inc r15
    mov byte ptr [r15], '|'
    inc r15
    
    pop rcx
    push rcx
    xor rsi, rsi
    
ascii_loop:
    cmp rsi, rcx
    jge ascii_done
    
    lea rdx, [r14 + rsi]
    movzx rax, byte ptr [r13 + rdx]
    cmp rax, 0x20
    jb print_dot
    cmp rax, 0x7e
    ja print_dot
    mov byte ptr [r15], al
    inc r15
    inc rsi
    jmp ascii_loop
    
print_dot:
    mov byte ptr [r15], '.'
    inc r15
    inc rsi
    jmp ascii_loop

ascii_done:
    mov byte ptr [r15], '|'
    inc r15
    mov byte ptr [r15], 10
    inc r15
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r15
    sub rdx, rsi
    syscall
    
    pop rcx
    add r14, rcx
    jmp process_loop

print_final_offset:
    lea r15, [rip + output_buffer]
    mov rdi, r12
    call print_hex8
    mov byte ptr [r15], 10
    inc r15
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    mov rdx, r15
    sub rdx, rsi
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

print_hex8:
    lea rsi, [rip + hex_digits]
    mov r8, 7
1:
    mov rax, rdi
    mov rcx, r8
    shl rcx, 2
    shr rax, cl
    and rax, 15
    movzx rax, byte ptr [rsi + rax]
    mov byte ptr [r15], al
    inc r15
    dec r8
    jns 1b
    ret
