```asm
.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .space 65536
output_buffer:
    .space 8192

_start:
    xor r12, r12
    lea r13, [rip + input_buffer]

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [r13 + r12]
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r14, r14
    lea r15, [rip + output_buffer]
    
process_loop:
    cmp r14, r12
    jge print_final_offset
    
    mov rcx, r12
    sub rcx, r14
    cmp rcx, 16
    jbe 1f
    mov rcx, 16
1:

    call print_hex8

    add r15, 2
    
    pop rcx
    push rcx

    cmp rsi, 16
    jge hex_done

    jge print_empty_byte
