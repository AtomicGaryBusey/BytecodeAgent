```asm
.globl _start

.section .rodata
error_msg:
    .ascii "error\n"
error_msg_len = . - error_msg

.bss
    .align 16
input_buffer:
    .zero 65536
    
value_stack:
    .zero 8192
    
digit_buffer:
    .zero 32

.text
_start:
    xor r12, r12
    
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    
    test rax, rax
    jle read_done
    
    add r12, rax
    cmp r12, 65536
    jb read_loop
    
read_done:
    xor r13, r13
    
process_lines:
    cmp r13, r12
    jae all_done
    
    xor r14, r14
    xor r15, r15
    
process_line:
    cmp r13, r12
    jae line_end
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + r13]
    
    cmp bl, 0x0A
    je line_end
    
    cmp bl, 0x20
    je skip_ws
    cmp bl, 0x09
    je skip_ws
    
    jmp parse_token_start
    
skip_ws:
    inc r13
    jmp process_line
    
parse_token_start:
    mov rsi, r13
    
find_token_end:
    cmp r13, r12
    jae token_end_found
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + r13]
    
    cmp bl, 0x20
    je token_end_found
    cmp bl, 0x09
    je token_end_found
    cmp bl, 0x0A
    je token_end_found
    
    inc r13
    jmp find_token_end
    
token_end_found:
    mov rcx, r13
    sub rcx, rsi
    
    cmp rcx, 1
    jne try_integer
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + rsi]
    
    cmp bl, '+'
    je do_add
    cmp bl, '-'
    je do_sub
    cmp bl, '*'
    je do_mul
    cmp bl, '/'
    je do_div
    cmp bl, '%'
    je do_mod
    
try_integer:
    call parse_integer
    test al, al
    jz token_error
    
    lea rax, [rip + value_stack]
    mov [rax + r14*8], rdx
    inc r14
    
    jmp process_line
    
do_add:
    cmp r14, 2
    jb token_error
    
    lea rax, [rip + value_stack]
    dec r14
    mov rbx, [rax + r14*8]
    dec r14
    mov rcx, [rax + r14*8]
    add rcx, rbx
    mov [rax + r14*8], rcx
    inc r14
    
    jmp process_line
    
do_sub:
    cmp r14, 2
    jb token_error
    
    lea rax, [rip + value_stack]
    dec r14
    mov rbx, [rax + r14*8]
    dec r14
    mov rcx, [rax + r14*8]
    sub rcx, rbx
    mov [rax + r14*8], rcx
    inc r14
    
    jmp process_line
    
do_mul:
    cmp r14, 2
    jb token_error
    
    lea rax, [rip + value_stack]
    dec r14
    mov rbx, [rax + r14*8]
    dec r14
    mov rcx, [rax + r14*8]
    imul rcx, rbx
    mov [rax + r14*8], rcx
    inc r14
    
    jmp process_line
    
do_div:
    cmp r14, 2
    jb token_error
    
    lea rax, [rip + value_stack]
    dec r14
    mov rbx, [rax + r14*8]
    
    test rbx, rbx
    jz token_error
    
    dec r14
    mov rax, [rax + r14*8]
    
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne 1f
    cmp rbx, -1
    je token_error
1:
    cqo
    idiv rbx
    
    lea rcx, [rip + value_stack]
    mov [rcx + r14*8], rax
    inc r14
    
    jmp process_line
    
do_mod:
    cmp r14, 2
    jb token_error
    
    lea rax, [rip + value_stack]
    dec r14
    mov rbx, [rax + r14*8]
    
    test rbx, rbx
    jz token_error
    
    dec r14
    mov rax, [rax + r14*8]
    
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne 1f
    cmp rbx, -1
    je token_error
1:
    cqo
    idiv rbx
    
    lea rcx, [rip + value_stack]
    mov [rcx + r14*8], rdx
    inc r14
    
    jmp process_line
    
token_error:
    mov r15, 1
    
skip_to_eol:
    cmp r13, r12
    jae line_end
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + r13]
    inc r13
    
    cmp bl, 0x0A
    jne skip_to_eol
    
    jmp line_end_after_newline
    
line_end:
    cmp r13, r12
    jae line_end_after_newline
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + r13]
    cmp bl, 0x0A
    jne line_end_after_newline
    
    inc r13
    
line_end_after_newline:
    test r15, r15
    jnz output_error
    
    cmp r14, 1
    jne output_error
    
    lea rax, [rip + value_stack]
    mov rdi, [rax]
    call output_number
    
    jmp process_lines
    
output_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_msg_len
    syscall
    
    jmp process_lines
    
all_done:
    mov rax, 60
    xor rdi, rdi
    syscall

parse_integer:
    push rbx
    push rcx
    push rsi
    push r8
    push r9
    
    xor r8, r8
    xor r9, r9
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + rsi]
    cmp bl, '-'
    jne 1f
    
    mov r9, 1
    inc rsi
    dec rcx
1:
    test rcx, rcx
    jz parse_int_error
    
    cmp rcx, 18
    ja parse_int_error
    
parse_digit_loop:
    test rcx, rcx
    jz parse_int_done
    
    lea rax, [rip + input_buffer]
    movzx ebx, byte ptr [rax + rsi]
    
    cmp bl, '0'
    jb parse_int_error
    cmp bl, '9'
    ja parse_int_error
    
    sub bl, '0'
    movzx ebx, bl
    
    mov rax, r8
    mov rdx, 10
    mul rdx
    add rax, rbx
    mov r8, rax
    
    inc rsi
    dec rcx
    jmp parse_digit_loop
    
parse_int_done:
    test r9, r9
    jz 1f
    neg r8
1:
    mov rdx, r8
    mov al, 1
    
    pop r9
    pop r8
    pop rsi
    pop rcx
    pop rbx
    ret
    
parse_int_error:
    xor al, al
    pop r9
    pop r8
    pop rsi
    pop rcx
    pop rbx
    ret

output_number:
    push rax
    push rbx
    push rcx
    push rdx
    push rsi
    push rdi
    push r8
    push r9
    
    mov r8, rdi
    xor r9, r9
    
    test r8, r8
    jns 1f
    
    mov r9, 1
    neg r8
    
1:
    lea rbx, [rip + digit_buffer]
    add rbx, 31
    mov rcx, rbx
    
    mov rax, r8
2:
    xor rdx, rdx
    mov rbx, 10
    div rbx
    
    add dl, '0'
    dec rcx
    mov [rcx], dl
    
    test rax, rax
    jnz 2b
    
    test r9, r9
    jz 3f
    
    dec rcx
    mov byte ptr [rcx], '-'
    
3:
    mov rax, 1
    mov rdi, 1
    mov rsi, rcx
    lea rbx, [rip + digit_buffer]
    add rbx, 31
    sub rbx, rcx
    mov rdx, rbx
    syscall
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buffer]
    mov byte ptr [rsi], 0x0A
    mov rdx, 1
    syscall
    
    pop r9
    pop r8
    pop rdi
    pop rsi
    pop rdx
    pop rcx
    pop rbx
    pop rax
    ret
```
