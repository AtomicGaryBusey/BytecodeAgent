hex_chars:

.align 4

    .zero 65536

    .zero 256

    b.ge print_final_offset
    
    sub x22, x19, x21
    mov x23, #16
    cmp x22, x23
    csel x22, x22, x23, lt

    mov x0, x19
    mov x1, x25

    mov x2, #9

    mov x0, x21
    mov x1, x26

    strb w0, [x26], #1

    strb w0, [x26], #1
    strb w0, [x26], #1

    strb w0, [x26], #1

    strb w0, [x26], #1
    mov w0, #10
    strb w0, [x26], #1

    sub x4, x4, #1
    cbnz x4, format_hex_8_loop
