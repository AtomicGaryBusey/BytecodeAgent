```asm
.globl _start

.text

_start:
    # Read all input into buffer
    xor r12, r12                    # current position in buffer
    
read_loop:
    mov rax, 0                      # read syscall
    mov rdi, 0                      # fd = 0 (stdin)
    lea rsi, [rip + input_buffer]
    add rsi, r12                    # buf + current_pos
    mov rdx, 32768
    sub rdx, r12                    # count = 32768 - current_pos
    jle read_done                   # if no space left, done
    syscall
    test rax, rax
    jz read_done                    # if 0 bytes read, EOF
    add r12, rax                    # update current_pos
    jmp read_loop

read_done:
    # r12 now contains total bytes read (input_size)
    
    # Parse lines
    xor r13, r13                    # line count
    xor r14, r14                    # current position in input
    
parse_loop:
    cmp r14, r12                    # if pos >= input_size
    jge parse_done
    
    # Find end of line (next newline or end of input)
    mov r15, r14                    # start of line
    
find_newline:
    cmp r14, r12
    jge line_end_no_newline
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r14]
    inc r14
    cmp al, 10                      # newline
    jne find_newline
    
    # Found newline at r14-1
    # Line is from r15 to r14-1 (not including newline)
    mov rbx, r14
    dec rbx
    sub rbx, r15                    # length (excluding newline)
    jmp store_line
    
line_end_no_newline:
    # End of input without newline
    # Line is from r15 to r14
    mov rbx, r14
    sub rbx, r15                    # length
    
store_line:
    # Store line: pointer and length
    lea rsi, [rip + input_buffer]
    add rsi, r15                    # pointer to line start
    
    lea rax, [rip + lines]
    mov rdx, r13
    shl rdx, 4                      # r13 * 16
    add rax, rdx
    
    mov qword ptr [rax], rsi        # store pointer
    mov qword ptr [rax + 8], rbx    # store length
    
    inc r13
    jmp parse_loop
    
parse_done:
    # r13 contains line count
    mov qword ptr [rip + line_count], r13
    
    # Sort lines - quicksort
    test r13, r13
    jz output_lines                 # if no lines, skip to output
    
    # quicksort(0, line_count - 1)
    xor rdi, rdi                    # left = 0
    mov rsi, r13
    dec rsi                         # right = line_count - 1
    call quicksort
    
output_lines:
    # Output all lines
    xor r12, r12                    # current line index
    
output_loop:
    cmp r12, qword ptr [rip + line_count]
    jge done
    
    # Get line pointer and length
    lea rax, [rip + lines]
    mov rdx, r12
    shl rdx, 4
    add rax, rdx
    
    mov r13, qword ptr [rax]        # line pointer
    mov r14, qword ptr [rax + 8]    # line length
    
    # Write line
    test r14, r14
    jz write_newline                # if empty line, just write newline
    
    mov rax, 1                      # write syscall
    mov rdi, 1                      # fd = 1 (stdout)
    mov rsi, r13                    # buf
    mov rdx, r14                    # count
    syscall
    
write_newline:
    # Write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    
    inc r12
    jmp output_loop
    
done:
    mov rax, 60                     # exit syscall
    xor rdi, rdi                    # status = 0
    syscall

# Quicksort implementation
# rdi = left, rsi = right
quicksort:
    cmp rdi, rsi
    jge quicksort_ret               # if left >= right, return
    
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi                    # save left
    mov r13, rsi                    # save right
    
    # Partition
    call partition                  # returns pivot in rax
    mov r14, rax                    # pivot index
    
    # quicksort(left, pivot - 1)
    mov rdi, r12
    mov rsi, r14
    dec rsi
    call quicksort
    
    # quicksort(pivot + 1, right)
    mov rdi, r14
    inc rdi
    mov rsi, r13
    call quicksort
    
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    
quicksort_ret:
    ret

# Partition for quicksort
# rdi = left, rsi = right
# Returns pivot index in rax
partition:
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi                    # left
    mov r13, rsi                    # right
    
    # Choose pivot as rightmost element
    # i = left - 1
    mov r14, r12
    dec r14
    
    # j = left
    mov r15, r12
    
partition_loop:
    cmp r15, r13
    jge partition_done
    
    # Compare lines[j] with lines[right]
    mov rdi, r15                    # index j
    mov rsi, r13                    # index right (pivot)
    call compare_lines
    
    # if lines[j] <= lines[right]
    test rax, rax
    jg partition_next
    
    # Swap lines[++i] with lines[j]
    inc r14
    mov rdi, r14
    mov rsi, r15
    call swap_lines
    
partition_next:
    inc r15
    jmp partition_loop
    
partition_done:
    # Swap lines[i+1] with lines[right]
    inc r14
    mov rdi, r14
    mov rsi, r13
    call swap_lines
    
    mov rax, r14                    # return pivot index
    
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# Compare two lines
# rdi = index1, rsi = index2
# Returns: <0 if line1 < line2, 0 if equal, >0 if line1 > line2
compare_lines:
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    # Get line1 pointer and length
    lea rax, [rip + lines]
    mov rdx, rdi
    shl rdx, 4
    add rax, rdx
    mov r12, qword ptr [rax]        # line1 pointer
    mov r13, qword ptr [rax + 8]    # line1 length
    
    # Get line2 pointer and length
    lea rax, [rip + lines]
    mov rdx, rsi
    shl rdx, 4
    add rax, rdx
    mov r14, qword ptr [rax]        # line2 pointer
    mov r15, qword ptr [rax + 8]    # line2 length
    
    # Compare bytes
    xor rcx, rcx                    # index
    
compare_loop:
    cmp rcx, r13
    jge compare_line1_end
    cmp rcx, r15
    jge compare_line2_end
    
    movzx rax, byte ptr [r12 + rcx]
    movzx rdx, byte ptr [r14 + rcx]
    sub rax, rdx
    jnz compare_done
    
    inc rcx
    jmp compare_loop
    
compare_line1_end:
    # line1 ended, check if line2 also ended
    cmp rcx, r15
    jge compare_equal               # both ended, equal
    mov rax, -1                     # line1 shorter, line1 < line2
    jmp compare_done
    
compare_line2_end:
    # line2 ended but line1 didn't
    mov rax, 1                      # line1 longer, line1 > line2
    jmp compare_done
    
compare_equal:
    xor rax, rax
    
compare_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# Swap two lines in the array
# rdi = index1, rsi = index2
swap_lines:
    cmp rdi, rsi
    je swap_ret                     # if same index, nothing to do
    
    push rbx
    push r12
    
    # Get pointers to both line entries
    lea rax, [rip + lines]
    mov rdx, rdi
    shl rdx, 4
    add rax, rdx
    mov r8, rax                     # pointer to line1 entry
    
    lea rax, [rip + lines]
    mov rdx, rsi
    shl rdx, 4
    add rax, rdx
    mov r9, rax                     # pointer to line2 entry
    
    # Swap pointer
    mov r10, qword ptr [r8]
    mov r11, qword ptr [r9]
    mov qword ptr [r8], r11
    mov qword ptr [r9], r10
    
    # Swap length
    mov r10, qword ptr [r8 + 8]
    mov r11, qword ptr [r9 + 8]
    mov qword ptr [r8 + 8], r11
    mov qword ptr [r9 + 8], r10
    
    pop r12
    pop rbx
    
swap_ret:
    ret

.section .rodata
newline:
    .byte 10

.bss
input_buffer:
    .zero 32768

lines:
    .zero 16000

line_count:
    .zero 8
```
