.globl _start

.text
_start:
    # Read all input into buffer
    xor r12, r12                    # r12 = bytes read so far
    
read_loop:
    lea rsi, [rip + input_buffer]
    add rsi, r12                    # buffer + offset
    mov rdi, 0                      # stdin
    mov rdx, 65536
    sub rdx, r12                    # remaining space
    jz read_done
    xor rax, rax                    # read syscall
    syscall
    test rax, rax
    jle read_done                   # EOF or error
    add r12, rax
    jmp read_loop
    
read_done:
    # r12 = total input size
    # Find '!' separator
    lea rdi, [rip + input_buffer]
    mov rcx, r12
    mov al, 0x21                    # '!'
    xor r13, r13                    # r13 = index
    
find_bang:
    test rcx, rcx
    jz no_bang
    cmp byte ptr [rdi + r13], al
    je found_bang
    inc r13
    dec rcx
    jmp find_bang
    
found_bang:
    # r13 = index of '!'
    mov r14, r12
    sub r14, r13
    dec r14                         # r14 = program input size
    jmp validate
    
no_bang:
    mov r13, r12                    # program size = total
    xor r14, r14                    # program input size = 0
    
validate:
    # Validate bracket matching
    lea rsi, [rip + input_buffer]
    xor rcx, rcx                    # bracket depth counter
    xor rdi, rdi                    # index
    
validate_loop:
    cmp rdi, r13
    jge validate_done
    movzx rax, byte ptr [rsi + rdi]
    cmp al, 0x5b                    # '['
    je open_bracket
    cmp al, 0x5d                    # ']'
    je close_bracket
    inc rdi
    jmp validate_loop
    
open_bracket:
    inc rcx
    inc rdi
    jmp validate_loop
    
close_bracket:
    dec rcx
    test rcx, rcx
    js validation_failed
    inc rdi
    jmp validate_loop
    
validate_done:
    test rcx, rcx
    jnz validation_failed
    
    # Execute program
    lea r15, [rip + input_buffer]   # r15 = program start
    lea r12, [r15 + r13 + 1]        # r12 = program input start
    xor rbx, rbx                    # rbx = program input position
    lea rbp, [rip + bf_tape]        # rbp = tape pointer
    xor rsi, rsi                    # rsi = program counter
    
execute_loop:
    cmp rsi, r13
    jge execute_done
    
    movzx rax, byte ptr [r15 + rsi]
    
    cmp al, 0x3e                    # '>'
    je cmd_right
    cmp al, 0x3c                    # '<'
    je cmd_left
    cmp al, 0x2b                    # '+'
    je cmd_inc
    cmp al, 0x2d                    # '-'
    je cmd_dec
    cmp al, 0x2e                    # '.'
    je cmd_output
    cmp al, 0x2c                    # ','
    je cmd_input
    cmp al, 0x5b                    # '['
    je cmd_loop_start
    cmp al, 0x5d                    # ']'
    je cmd_loop_end
    
    inc rsi
    jmp execute_loop
    
cmd_right:
    inc rbp
    inc rsi
    jmp execute_loop
    
cmd_left:
    dec rbp
    inc rsi
    jmp execute_loop
    
cmd_inc:
    inc byte ptr [rbp]
    inc rsi
    jmp execute_loop
    
cmd_dec:
    dec byte ptr [rbp]
    inc rsi
    jmp execute_loop
    
cmd_output:
    push rsi
    mov rdi, 1
    mov rsi, rbp
    mov rdx, 1
    mov rax, 1
    syscall
    pop rsi
    inc rsi
    jmp execute_loop
    
cmd_input:
    cmp rbx, r14
    jge input_exhausted
    movzx rax, byte ptr [r12 + rbx]
    mov byte ptr [rbp], al
    inc rbx
    inc rsi
    jmp execute_loop
    
input_exhausted:
    mov byte ptr [rbp], 0
    inc rsi
    jmp execute_loop
    
cmd_loop_start:
    cmp byte ptr [rbp], 0
    jne loop_start_continue
    
    mov rcx, 1
    mov rdi, rsi
    inc rdi
    
find_close:
    cmp rdi, r13
    jge execute_done
    movzx rax, byte ptr [r15 + rdi]
    cmp al, 0x5b
    je found_open_in_search
    cmp al, 0x5d
    je found_close_in_search
    inc rdi
    jmp find_close
    
found_open_in_search:
    inc rcx
    inc rdi
    jmp find_close
    
found_close_in_search:
    dec rcx
    jz found_matching_close
    inc rdi
    jmp find_close
    
found_matching_close:
    mov rsi, rdi
    inc rsi
    jmp execute_loop
    
loop_start_continue:
    inc rsi
    jmp execute_loop
    
cmd_loop_end:
    cmp byte ptr [rbp], 0
    je loop_end_continue
    
    mov rcx, 1
    mov rdi, rsi
    dec rdi
    
find_open:
    test rdi, rdi
    js execute_done
    movzx rax, byte ptr [r15 + rdi]
    cmp al, 0x5d
    je found_close_in_search2
    cmp al, 0x5b
    je found_open_in_search2
    dec rdi
    jmp find_open
    
found_close_in_search2:
    inc rcx
    dec rdi
    jmp find_open
    
found_open_in_search2:
    dec rcx
    jz found_matching_open
    dec rdi
    jmp find_open
    
found_matching_open:
    mov rsi, rdi
    inc rsi
    jmp execute_loop
    
loop_end_continue:
    inc rsi
    jmp execute_loop
    
execute_done:
    xor rdi, rdi
    mov rax, 60
    syscall
    
validation_failed:
    mov rdi, 1
    mov rax, 60
    syscall

.bss
.align 16
input_buffer: .space 65536
bf_tape: .space 30000
