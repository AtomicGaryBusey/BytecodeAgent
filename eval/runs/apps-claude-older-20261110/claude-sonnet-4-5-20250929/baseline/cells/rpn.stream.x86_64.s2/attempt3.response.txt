.globl _start

.section .rodata
error_msg:
    .ascii "error\n"
error_msg_len = . - error_msg
int64_min_str:
    .ascii "-9223372036854775808\n"

.bss
    .align 16
input_buffer:
    .zero 65536
input_len:
    .zero 8
stack:
    .zero 4000
output_buffer:
    .zero 32

.text

_start:
    call read_input
    
    lea rdi, [rip + input_buffer]
    mov rsi, qword ptr [rip + input_len]
    call process_all
    
    mov rax, 60
    xor rdi, rdi
    syscall

read_input:
    xor r12, r12
1:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle 2f
    add r12, rax
    jmp 1b
2:
    mov qword ptr [rip + input_len], r12
    ret

process_all:
    push r12
    push r13
    push r14
    
    mov r12, rdi
    mov r13, rsi
    lea r13, [r12 + r13]
    
1:
    cmp r12, r13
    jge 2f
    
    mov r14, r12
3:
    cmp r14, r13
    jge 4f
    cmp byte ptr [r14], 0x0a
    je 4f
    inc r14
    jmp 3b
    
4:
    mov rdi, r12
    mov rsi, r14
    call process_line
    
    mov r12, r14
    cmp r12, r13
    jge 2f
    inc r12
    jmp 1b
    
2:
    pop r14
    pop r13
    pop r12
    ret

process_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi
    mov r13, rsi
    lea r14, [rip + stack]
    
1:
    cmp r12, r13
    jge 9f
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je 2f
    cmp al, 0x09
    je 2f
    jmp 3f
2:
    inc r12
    jmp 1b
    
3:
    mov r15, r12
4:
    cmp r15, r13
    jge 5f
    movzx eax, byte ptr [r15]
    cmp al, 0x20
    je 5f
    cmp al, 0x09
    je 5f
    inc r15
    jmp 4b
    
5:
    mov rdi, r12
    mov rsi, r15
    call classify_token
    
    test rax, rax
    jz 99f
    
    cmp rax, 1
    je 6f
    
    mov rdi, r12
    mov rsi, r14
    call do_operator
    mov r14, rax
    test rdx, rdx
    jnz 99f
    jmp 8f
    
6:
    mov rdi, r12
    mov rsi, r15
    call parse_int
    mov qword ptr [r14], rax
    add r14, 8
    
8:
    mov r12, r15
    jmp 1b
    
9:
    lea rax, [rip + stack]
    add rax, 8
    cmp r14, rax
    jne 99f
    
    mov rdi, qword ptr [rip + stack]
    call print_int
    jmp 100f
    
99:
    call print_error
    
100:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

classify_token:
    mov rcx, rsi
    sub rcx, rdi
    jz 1f
    
    cmp rcx, 1
    jne 2f
    movzx eax, byte ptr [rdi]
    cmp al, '+'
    je 3f
    cmp al, '-'
    je 3f
    cmp al, '*'
    je 3f
    cmp al, '/'
    je 3f
    cmp al, '%'
    je 3f
    
2:
    push rdi
    push rsi
    call is_literal
    pop rsi
    pop rdi
    test rax, rax
    jz 1f
    mov rax, 1
    ret
    
3:
    mov rax, 2
    ret
    
1:
    xor rax, rax
    ret

is_literal:
    push rbx
    mov rbx, rdi
    
    cmp rbx, rsi
    jge 1f
    cmp byte ptr [rbx], '-'
    jne 2f
    inc rbx
    
2:
    cmp rbx, rsi
    jge 1f
    
    mov rcx, rbx
3:
    cmp rcx, rsi
    jge 4f
    movzx eax, byte ptr [rcx]
    cmp al, '0'
    jb 1f
    cmp al, '9'
    ja 1f
    inc rcx
    jmp 3b
    
4:
    sub rcx, rbx
    cmp rcx, 19
    jge 1f
    
    mov rax, 1
    pop rbx
    ret
    
1:
    xor rax, rax
    pop rbx
    ret

parse_int:
    push rbx
    
    xor rax, rax
    xor r11, r11
    
    cmp rdi, rsi
    jge 3f
    cmp byte ptr [rdi], '-'
    jne 1f
    mov r11, 1
    inc rdi
    
1:
    cmp rdi, rsi
    jge 2f
    movzx ebx, byte ptr [rdi]
    sub bl, '0'
    movzx rbx, bl
    imul rax, 10
    add rax, rbx
    inc rdi
    jmp 1b
    
2:
    test r11, r11
    jz 3f
    neg rax
    
3:
    pop rbx
    ret

do_operator:
    push rbx
    push r12
    
    mov r12, rsi
    
    lea rax, [rip + stack]
    add rax, 16
    cmp r12, rax
    jl 99f
    
    sub r12, 8
    mov r11, qword ptr [r12]
    
    sub r12, 8
    mov r10, qword ptr [r12]
    
    movzx ecx, byte ptr [rdi]
    
    cmp cl, '+'
    je 1f
    cmp cl, '-'
    je 2f
    cmp cl, '*'
    je 3f
    cmp cl, '/'
    je 4f
    cmp cl, '%'
    je 5f
    jmp 99f
    
1:
    add r10, r11
    jmp 90f
    
2:
    sub r10, r11
    jmp 90f
    
3:
    mov rax, r10
    imul rax, r11
    mov r10, rax
    jmp 90f
    
4:
    test r11, r11
    jz 99f
    mov rax, 0x8000000000000000
    cmp r10, rax
    jne 6f
    cmp r11, -1
    je 99f
6:
    mov rax, r10
    cqo
    idiv r11
    mov r10, rax
    jmp 90f
    
5:
    test r11, r11
    jz 99f
    mov rax, 0x8000000000000000
    cmp r10, rax
    jne 7f
    cmp r11, -1
    je 99f
7:
    mov rax, r10
    cqo
    idiv r11
    mov r10, rdx
    jmp 90f
    
90:
    mov qword ptr [r12], r10
    add r12, 8
    mov rax, r12
    xor rdx, rdx
    pop r12
    pop rbx
    ret
    
99:
    mov rax, r12
    mov rdx, 1
    pop r12
    pop rbx
    ret

print_int:
    push rbx
    push r12
    push r13
    
    mov rbx, rdi
    
    mov rax, 0x8000000000000000
    cmp rbx, rax
    jne 1f
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + int64_min_str]
    mov rdx, 21
    syscall
    jmp 100f
    
1:
    lea r12, [rip + output_buffer]
    add r12, 31
    mov byte ptr [r12], 0x0a
    dec r12
    
    mov rax, rbx
    xor r13, r13
    test rax, rax
    jns 2f
    mov r13, 1
    neg rax
    
2:
    mov rcx, 10
3:
    xor rdx, rdx
    div rcx
    add dl, '0'
    mov byte ptr [r12], dl
    dec r12
    test rax, rax
    jnz 3b
    
    test r13, r13
    jz 4f
    mov byte ptr [r12], '-'
    dec r12
    
4:
    inc r12
    lea r13, [rip + output_buffer]
    add r13, 32
    sub r13, r12
    
    mov rax, 1
    mov rdi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    
100:
    pop r13
    pop r12
    pop rbx
    ret

print_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_msg_len
    syscall
    ret
