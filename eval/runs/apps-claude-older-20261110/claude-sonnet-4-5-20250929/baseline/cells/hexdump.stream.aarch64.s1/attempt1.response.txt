```asm
.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buffer:
    .zero 65536
output_line:
    .zero 128

.text
_start:
    mov x19, #0
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    
read_loop:

    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    svc #0

    add x19, x19, x0
    b read_loop

    mov x21, #0

    cmp x21, x19
    bge print_final_offset
    
    sub x22, x19, x21
    mov x23, #16
    cmp x22, x23
    csel x22, x22, x23, lt

    mov x1, x22
    add x2, x20, x21

    mov x8, #64
    mov x0, #1
    adrp x1, output_line
    add x1, x1, :lo12:output_line
    svc #0

    adrp x0, output_line
    add x0, x0, :lo12:output_line
    mov x1, x19
    bl format_offset_8
    
    adrp x0, output_line
    add x0, x0, :lo12:output_line

    mov x8, #64
    mov x0, #1
    adrp x1, output_line
    add x1, x1, :lo12:output_line
    mov x2, #9
    svc #0
    
    mov x8, #93
    mov x0, #0
    svc #0

    stp x29, x30, [sp, #-64]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    
    mov x19, x0
    mov x20, x1
    mov x21, x2
    
    adrp x22, output_line
    add x22, x22, :lo12:output_line
    mov x23, x22
    
    mov x0, x23
    mov x1, x19
    bl format_offset_8
    add x23, x23, #8
    
    mov w0, #0x20
    strb w0, [x23], #1
    strb w0, [x23], #1

    bge hex_done
    
    cmp x24, x20
    bge hex_padding

    lsr w1, w0, #4
    and w2, w0, #0x0f
    
    adrp x3, hex_digits
    add x3, x3, :lo12:hex_digits
    ldrb w1, [x3, w1, uxtw]
    ldrb w2, [x3, w2, uxtw]
    
    strb w1, [x23], #1
    strb w2, [x23], #1
    mov w1, #0x20
    strb w1, [x23], #1
    
    cmp x24, #7
    bne 1f
    mov w1, #0x20
    strb w1, [x23], #1

    b hex_loop
    
hex_padding:
    mov w0, #0x20
    strb w0, [x23], #1
    strb w0, [x23], #1
    strb w0, [x23], #1
    
    cmp x24, #7
    bne 1f
    mov w0, #0x20
    strb w0, [x23], #1

    add x24, x24, #1
    b hex_loop
    
hex_done:
    mov w0, #0x20
    strb w0, [x23], #1
    mov w0, #0x7c
    strb w0, [x23], #1
    
    mov x24, #0
ascii_loop:
    cmp x24, x20
    bge ascii_done
    
    ldrb w0, [x21, x24]
    
    cmp w0, #0x20
    blt not_printable
    cmp w0, #0x7e
    bgt not_printable
    
    strb w0, [x23], #1
    b ascii_next
    
not_printable:
    mov w0, #0x2e
    strb w0, [x23], #1
    
ascii_next:
    add x24, x24, #1
    b ascii_loop
    
ascii_done:
    mov w0, #0x7c
    strb w0, [x23], #1
    mov w0, #0x0a
    strb w0, [x23], #1
    
    sub x0, x23, x22
    
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]

    stp x19, x20, [sp, #16]
    
    mov x19, x0
    mov x20, x1

    mov x3, #28
format_offset_loop:
    lsr x4, x20, x3
    and x4, x4, #0x0f
    ldrb w4, [x2, x4]
    strb w4, [x19], #1
    subs x3, x3, #4
    bge format_offset_loop
    
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #32
    ret
```
