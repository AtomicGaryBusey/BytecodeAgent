.globl _start

.section .bss
.align 16
input_buffer:
    .space 32768
line_array:
    .space 16000

.section .rodata
newline:
    .byte 0x0A

.section .text
_start:
    xor ebx, ebx
    
read_loop:
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 32768
    sub edx, ebx
    jz read_done
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add ebx, eax
    cmp ebx, 32768
    jl read_loop
    
read_done:
    test ebx, ebx
    jz exit_program
    
    xor ecx, ecx
    xor esi, esi
    lea rdi, [rip + line_array]
    lea r8, [rip + input_buffer]
    
parse_loop:
    cmp esi, ebx
    jge parse_done
    
    lea rax, [r8 + rsi]
    mov qword ptr [rdi], rax
    
    mov r9d, esi
    
find_newline:
    cmp esi, ebx
    jge line_at_eof
    cmp byte ptr [r8 + rsi], 0x0A
    je found_newline
    inc esi
    jmp find_newline
    
found_newline:
    mov eax, esi
    sub eax, r9d
    mov qword ptr [rdi + 8], rax
    inc esi
    add rdi, 16
    inc ecx
    jmp parse_loop
    
line_at_eof:
    mov eax, esi
    sub eax, r9d
    mov qword ptr [rdi + 8], rax
    inc ecx
    
parse_done:
    test ecx, ecx
    jz exit_program
    
    lea rdi, [rip + line_array]
    xor esi, esi
    lea edx, [ecx - 1]
    push rcx
    call quicksort
    pop rcx
    
    xor ebx, ebx
    
output_loop:
    cmp ebx, ecx
    jge exit_program
    
    mov rax, rbx
    shl rax, 4
    lea rdx, [rip + line_array]
    add rax, rdx
    mov rsi, qword ptr [rax]
    mov rdx, qword ptr [rax + 8]
    
    test rdx, rdx
    jz write_newline
    push rbx
    push rcx
    mov edi, 1
    mov eax, 1
    syscall
    pop rcx
    pop rbx
    
write_newline:
    push rbx
    push rcx
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    syscall
    pop rcx
    pop rbx
    
    inc ebx
    jmp output_loop
    
exit_program:
    xor edi, edi
    mov eax, 60
    syscall

quicksort:
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    
    cmp r13, r14
    jge qs_done
    
    call partition
    mov r15, rax
    
    mov rdi, r12
    mov rsi, r13
    lea rdx, [r15 - 1]
    call quicksort
    
    mov rdi, r12
    lea rsi, [r15 + 1]
    mov rdx, r14
    call quicksort
    
qs_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

partition:
    push rbx
    push rbp
    
    mov r10, r12
    mov r11, r13
    mov rbp, r14
    
    mov rax, rbp
    shl rax, 4
    add rax, r10
    mov r8, qword ptr [rax]
    mov r9, qword ptr [rax + 8]
    
    lea rbx, [r11 - 1]
    
part_loop:
    cmp r11, rbp
    jge part_done
    
    mov rax, r11
    shl rax, 4
    add rax, r10
    mov rdi, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    mov rdx, r8
    mov rcx, r9
    push r8
    push r9
    push r10
    push r11
    push rbx
    push rbp
    call compare_lines
    pop rbp
    pop rbx
    pop r11
    pop r10
    pop r9
    pop r8
    
    cmp rax, 0
    jg part_next
    
    inc rbx
    mov rax, rbx
    shl rax, 4
    add rax, r10
    mov rcx, r11
    shl rcx, 4
    add rcx, r10
    
    mov rdx, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    mov rdi, qword ptr [rcx]
    mov r15, qword ptr [rcx + 8]
    mov qword ptr [rax], rdi
    mov qword ptr [rax + 8], r15
    mov qword ptr [rcx], rdx
    mov qword ptr [rcx + 8], rsi
    
part_next:
    inc r11
    jmp part_loop
    
part_done:
    inc rbx
    mov rax, rbx
    shl rax, 4
    add rax, r10
    mov rcx, rbp
    shl rcx, 4
    add rcx, r10
    
    mov rdx, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    mov rdi, qword ptr [rcx]
    mov r15, qword ptr [rcx + 8]
    mov qword ptr [rax], rdi
    mov qword ptr [rax + 8], r15
    mov qword ptr [rcx], rdx
    mov qword ptr [rcx + 8], rsi
    
    mov rax, rbx
    pop rbp
    pop rbx
    ret

compare_lines:
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    mov r11, rcx
    
    mov rax, rsi
    cmp rax, rcx
    cmova rax, rcx
    
    xor ecx, ecx
    
cmp_loop:
    cmp rcx, rax
    jge cmp_len
    
    movzx edx, byte ptr [r8 + rcx]
    movzx esi, byte ptr [r10 + rcx]
    sub edx, esi
    jnz cmp_diff
    
    inc rcx
    jmp cmp_loop
    
cmp_diff:
    mov eax, edx
    ret
    
cmp_len:
    mov rax, r9
    sub rax, r11
    ret
