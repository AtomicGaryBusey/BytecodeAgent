.globl _start

.section .bss
.align 16
input_buffer:
    .zero 32768
lines:
    .zero 16000

.section .text

_start:
    xor ebx, ebx
read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 32768
    sub edx, ebx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add ebx, eax
    jmp read_loop

read_done:
    test ebx, ebx
    jz exit_program
    
    xor r12d, r12d
    xor edi, edi
    lea r8, [rip + input_buffer]
    lea r9, [rip + lines]

parse_loop:
    cmp edi, ebx
    jge parse_done
    
    lea rsi, [r8 + rdi]
    mov [r9], rsi
    
    mov r10d, edi
find_newline:
    cmp edi, ebx
    jge found_eof
    movzx eax, byte ptr [r8 + rdi]
    inc edi
    cmp al, 10
    jne find_newline
    
    mov eax, edi
    dec eax
    sub eax, r10d
    mov [r9 + 8], rax
    inc r12d
    add r9, 16
    jmp parse_loop

found_eof:
    mov eax, edi
    sub eax, r10d
    mov [r9 + 8], rax
    inc r12d

parse_done:
    test r12d, r12d
    jz exit_program
    
    cmp r12d, 1
    jle sort_done
    
    mov r10d, 1
outer:
    cmp r10d, r12d
    jge sort_done
    
    lea r9, [rip + lines]
    mov eax, r10d
    shl rax, 4
    add r9, rax
    mov r13, [r9]
    mov r14, [r9 + 8]
    
    mov r11d, r10d
    dec r11d

inner:
    test r11d, r11d
    js insert
    
    lea r9, [rip + lines]
    mov eax, r11d
    shl rax, 4
    add r9, rax
    mov r15, [r9]
    mov rbp, [r9 + 8]
    
    mov rdi, r15
    mov rsi, r13
    mov rdx, rbp
    mov rcx, r14
    call cmp_lines
    
    cmp eax, 0
    jle insert
    
    lea r9, [rip + lines]
    mov eax, r11d
    inc eax
    shl rax, 4
    add r9, rax
    mov [r9], r15
    mov [r9 + 8], rbp
    
    dec r11d
    jmp inner

insert:
    inc r11d
    lea r9, [rip + lines]
    mov eax, r11d
    shl rax, 4
    add r9, rax
    mov [r9], r13
    mov [r9 + 8], r14
    
    inc r10d
    jmp outer

sort_done:
    xor r10d, r10d
write_loop:
    cmp r10d, r12d
    jge exit_program
    
    lea r9, [rip + lines]
    mov eax, r10d
    shl rax, 4
    add r9, rax
    mov rsi, [r9]
    mov rdx, [r9 + 8]
    
    test rdx, rdx
    jz write_newline
    
    mov rax, 1
    mov rdi, 1
    syscall

write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    
    inc r10d
    jmp write_loop

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

cmp_lines:
    push rbx
    push r12
    push r13
    
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    mov r12, rcx
    
    mov r13, rdx
    cmp r13, rcx
    jle 1f
    mov r13, rcx
1:
    xor rbx, rbx
cmp_loop:
    cmp rbx, r13
    jge check_len
    
    movzx eax, byte ptr [r8 + rbx]
    movzx edx, byte ptr [r10 + rbx]
    sub eax, edx
    jnz cmp_ret
    inc rbx
    jmp cmp_loop

check_len:
    mov rax, rsi
    sub rax, r12

cmp_ret:
    pop r13
    pop r12
    pop rbx
    ret

.section .rodata
nl:
    .byte 10
