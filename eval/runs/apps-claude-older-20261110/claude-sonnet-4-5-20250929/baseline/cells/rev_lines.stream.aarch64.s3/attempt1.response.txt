.globl _start

.bss
.align 4
input_buffer:
    .zero 65536
output_buffer:
    .zero 65536

.text
_start:
    // Read all input into input_buffer
    mov x19, #0                    // Total bytes read
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer

read_loop:
    mov x8, #63                    // read syscall
    mov x0, #0                     // stdin
    add x1, x20, x19               // buffer + offset
    mov x2, #65536
    sub x2, x2, x19                // remaining space
    svc #0
    
    cmp x0, #0
    beq read_done
    add x19, x19, x0               // Update total bytes read
    cmp x19, #65536
    blt read_loop
    
read_done:
    // x19 = total bytes read
    // x20 = input_buffer address
    
    mov x21, #0                    // Current position in input
    
process_next_line:
    cmp x21, x19
    bge all_done                   // All input processed
    
    // Find the next newline or end of input
    mov x22, x21                   // Start of current line
    
find_newline_loop:
    cmp x21, x19
    bge handle_final_line          // Reached end without newline
    
    ldrb w0, [x20, x21]
    cmp w0, #10                    // Check for newline
    beq handle_line_with_newline
    
    add x21, x21, #1
    b find_newline_loop
    
handle_line_with_newline:
    // Line from x22 to x21-1, newline at x21
    // Reverse bytes from x22 to x21-1 and output with newline
    
    mov x23, x22                   // Start of line
    sub x24, x21, #1               // End of line
    
    adrp x25, output_buffer
    add x25, x25, :lo12:output_buffer
    mov x26, #0                    // Output position
    
reverse_with_newline:
    cmp x24, x23
    blt write_reversed_with_newline
    
    ldrb w0, [x20, x24]
    strb w0, [x25, x26]
    add x26, x26, #1
    sub x24, x24, #1
    b reverse_with_newline
    
write_reversed_with_newline:
    // Add newline
    mov w0, #10
    strb w0, [x25, x26]
    add x26, x26, #1
    
    // Write output
    mov x8, #64                    // write syscall
    mov x0, #1                     // stdout
    mov x1, x25                    // output_buffer
    mov x2, x26                    // length
    svc #0
    
    // Move to next line (skip the newline)
    add x21, x21, #1
    b process_next_line
    
handle_final_line:
    // Line from x22 to x19-1 (end of input, no newline)
    // Only process if there are bytes
    
    cmp x22, x19
    bge all_done                   // No bytes left
    
    mov x23, x22                   // Start of line
    sub x24, x19, #1               // End of line
    
    adrp x25, output_buffer
    add x25, x25, :lo12:output_buffer
    mov x26, #0                    // Output position
    
reverse_without_newline:
    cmp x24, x23
    blt write_reversed_without_newline
    
    ldrb w0, [x20, x24]
    strb w0, [x25, x26]
    add x26, x26, #1
    sub x24, x24, #1
    b reverse_without_newline
    
write_reversed_without_newline:
    // Add newline
    mov w0, #10
    strb w0, [x25, x26]
    add x26, x26, #1
    
    // Write output
    mov x8, #64                    // write syscall
    mov x0, #1                     // stdout
    mov x1, x25                    // output_buffer
    mov x2, x26                    // length
    svc #0
    
all_done:
    mov x8, #93                    // exit syscall
    mov x0, #0                     // status 0
    svc #0
