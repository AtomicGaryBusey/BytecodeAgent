.globl _start

.text

_start:
    xor r12, r12
    
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 32768
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    
    add r12, rax
    jmp read_loop
    
read_done:
    test r12, r12
    jz exit_success
    
    xor r13, r13
    xor r14, r14
    lea rdi, [rip + input_buffer]
    mov r15, rdi
    
parse_loop:
    cmp r14, r12
    jge parse_done
    
    movzx eax, byte ptr [rdi + r14]
    inc r14
    
    cmp al, 0x0A
    jne parse_loop
    
    lea rsi, [rip + line_array]
    mov rax, r13
    shl rax, 4
    add rsi, rax
    
    mov qword ptr [rsi], r15
    lea rax, [rdi + r14 - 1]
    sub rax, r15
    mov qword ptr [rsi + 8], rax
    
    inc r13
    lea r15, [rdi + r14]
    jmp parse_loop
    
parse_done:
    lea rax, [rdi + r12]
    cmp r15, rax
    jge no_final_line
    
    lea rsi, [rip + line_array]
    mov rax, r13
    shl rax, 4
    add rsi, rax
    
    mov qword ptr [rsi], r15
    lea rax, [rdi + r12]
    sub rax, r15
    mov qword ptr [rsi + 8], rax
    
    inc r13
    
no_final_line:
    test r13, r13
    jz exit_success
    
    lea rdi, [rip + line_array]
    xor rsi, rsi
    mov rdx, r13
    dec rdx
    call quicksort
    
    xor r14, r14
    
output_loop:
    cmp r14, r13
    jge exit_success
    
    lea rsi, [rip + line_array]
    mov rax, r14
    shl rax, 4
    add rsi, rax
    
    mov r15, qword ptr [rsi]
    mov r12, qword ptr [rsi + 8]
    
    test r12, r12
    jz write_newline
    
    mov rax, 1
    mov rdi, 1
    mov rsi, r15
    mov rdx, r12
    syscall
    
write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    
    inc r14
    jmp output_loop
    
exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

quicksort:
    cmp rsi, rdx
    jge quicksort_done
    
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    
    call partition
    mov r15, rax
    
    mov rdi, r12
    mov rsi, r13
    mov rdx, r15
    dec rdx
    call quicksort
    
    mov rdi, r12
    mov rsi, r15
    inc rsi
    mov rdx, r14
    call quicksort
    
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    
quicksort_done:
    ret

partition:
    push rbx
    push r12
    push r13
    push r14
    push r15
    
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    
    mov r15, r13
    dec r15
    
    mov rbx, r13
    
partition_loop:
    cmp rbx, r14
    jge partition_done
    
    mov rax, rbx
    shl rax, 4
    add rax, r12
    mov rdi, qword ptr [rax]
    mov rsi, qword ptr [rax + 8]
    
    mov rax, r14
    shl rax, 4
    add rax, r12
    mov rdx, qword ptr [rax]
    mov rcx, qword ptr [rax + 8]
    
    call compare_lines
    
    test rax, rax
    jg skip_swap
    
    inc r15
    
    cmp r15, rbx
    je skip_swap
    
    mov rax, r15
    shl rax, 4
    lea rdi, [r12 + rax]
    
    mov rax, rbx
    shl rax, 4
    lea rsi, [r12 + rax]
    
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    
    mov qword ptr [rdi], r10
    mov qword ptr [rdi + 8], r11
    mov qword ptr [rsi], r8
    mov qword ptr [rsi + 8], r9
    
skip_swap:
    inc rbx
    jmp partition_loop
    
partition_done:
    inc r15
    
    cmp r15, r14
    je partition_return
    
    mov rax, r15
    shl rax, 4
    lea rdi, [r12 + rax]
    
    mov rax, r14
    shl rax, 4
    lea rsi, [r12 + rax]
    
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]
    mov r10, qword ptr [rsi]
    mov r11, qword ptr [rsi + 8]
    
    mov qword ptr [rdi], r10
    mov qword ptr [rdi + 8], r11
    mov qword ptr [rsi], r8
    mov qword ptr [rsi + 8], r9
    
partition_return:
    mov rax, r15
    
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

compare_lines:
    push rbx
    push r12
    push r13
    push r14
    
    mov r8, rdi
    mov r9, rsi
    mov r10, rdx
    mov r11, rcx
    
    mov r12, r9
    cmp r11, r12
    cmovb r12, r11
    
    xor rbx, rbx
    
compare_loop:
    cmp rbx, r12
    jge compare_length
    
    movzx rax, byte ptr [r8 + rbx]
    movzx rdx, byte ptr [r10 + rbx]
    
    cmp rax, rdx
    jne compare_diff
    
    inc rbx
    jmp compare_loop
    
compare_diff:
    sub rax, rdx
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
    
compare_length:
    mov rax, r9
    sub rax, r11
    
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

.section .rodata
newline:
    .byte 0x0A

.bss
    .align 16
input_buffer:
    .zero 32768
    
    .align 16
line_array:
    .zero 16000
