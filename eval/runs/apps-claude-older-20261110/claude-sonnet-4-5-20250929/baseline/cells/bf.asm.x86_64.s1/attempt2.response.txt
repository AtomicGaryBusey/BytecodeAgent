.globl _start

.bss
.align 16
input_buffer: .space 65536
bf_tape: .space 30000

.text
_start:
    xor r12, r12

read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    js read_done
    
    add r12, rax
    cmp r12, 65536
    jb read_loop
    
read_done:
    xor rcx, rcx
    mov r13, r12
    xor r14, r14
    xor r15, r15
    
find_separator:
    cmp rcx, r12
    jae separator_done
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x21
    je found_separator
    inc rcx
    jmp find_separator
    
found_separator:
    mov r13, rcx
    inc rcx
    mov r14, rcx
    mov r15, r12
    sub r15, rcx
    
separator_done:
    xor rcx, rcx
    xor r8, r8
    
check_brackets:
    cmp rcx, r13
    jae brackets_ok
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    
    cmp al, 0x5b
    je push_bracket
    cmp al, 0x5d
    je pop_bracket
    inc rcx
    jmp check_brackets
    
push_bracket:
    inc r8
    inc rcx
    jmp check_brackets
    
pop_bracket:
    test r8, r8
    jz bracket_error
    dec r8
    inc rcx
    jmp check_brackets
    
brackets_ok:
    test r8, r8
    jnz bracket_error
    
    xor rbx, rbx
    xor rcx, rcx
    xor r10, r10
    
execute_loop:
    cmp rcx, r13
    jae execute_done
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    
    cmp al, 0x3e
    je cmd_right
    cmp al, 0x3c
    je cmd_left
    cmp al, 0x2b
    je cmd_inc
    cmp al, 0x2d
    je cmd_dec
    cmp al, 0x2e
    je cmd_out
    cmp al, 0x2c
    je cmd_in
    cmp al, 0x5b
    je cmd_loop_start
    cmp al, 0x5d
    je cmd_loop_end
    
    inc rcx
    jmp execute_loop
    
cmd_right:
    inc rbx
    inc rcx
    jmp execute_loop
    
cmd_left:
    dec rbx
    inc rcx
    jmp execute_loop
    
cmd_inc:
    lea rdi, [rip + bf_tape]
    inc byte ptr [rdi + rbx]
    inc rcx
    jmp execute_loop
    
cmd_dec:
    lea rdi, [rip + bf_tape]
    dec byte ptr [rdi + rbx]
    inc rcx
    jmp execute_loop
    
cmd_out:
    mov r9, rcx
    lea rdi, [rip + bf_tape]
    lea rsi, [rdi + rbx]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    mov rcx, r9
    inc rcx
    jmp execute_loop
    
cmd_in:
    lea rdi, [rip + bf_tape]
    cmp r10, r15
    jae cmd_in_zero
    
    lea rsi, [rip + input_buffer]
    add rsi, r14
    movzx eax, byte ptr [rsi + r10]
    mov byte ptr [rdi + rbx], al
    inc r10
    inc rcx
    jmp execute_loop
    
cmd_in_zero:
    mov byte ptr [rdi + rbx], 0
    inc rcx
    jmp execute_loop
    
cmd_loop_start:
    lea rdi, [rip + bf_tape]
    movzx eax, byte ptr [rdi + rbx]
    test al, al
    jnz loop_start_continue
    
    inc rcx
    mov r8, 1
    
find_close:
    cmp rcx, r13
    jae execute_done
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je inc_depth
    cmp al, 0x5d
    je dec_depth
    inc rcx
    jmp find_close
    
inc_depth:
    inc r8
    inc rcx
    jmp find_close
    
dec_depth:
    dec r8
    jz found_close
    inc rcx
    jmp find_close
    
found_close:
    inc rcx
    jmp execute_loop
    
loop_start_continue:
    inc rcx
    jmp execute_loop
    
cmd_loop_end:
    lea rdi, [rip + bf_tape]
    movzx eax, byte ptr [rdi + rbx]
    test al, al
    jz loop_end_continue
    
    mov r8, 1
    
find_open:
    test rcx, rcx
    jz execute_done
    dec rcx
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5d
    je inc_depth_back
    cmp al, 0x5b
    je dec_depth_back
    jmp find_open
    
inc_depth_back:
    inc r8
    jmp find_open
    
dec_depth_back:
    dec r8
    jz found_open
    jmp find_open
    
found_open:
    inc rcx
    jmp execute_loop
    
loop_end_continue:
    inc rcx
    jmp execute_loop
    
execute_done:
    mov rax, 60
    xor rdi, rdi
    syscall
    
bracket_error:
    mov rax, 60
    mov rdi, 1
    syscall
