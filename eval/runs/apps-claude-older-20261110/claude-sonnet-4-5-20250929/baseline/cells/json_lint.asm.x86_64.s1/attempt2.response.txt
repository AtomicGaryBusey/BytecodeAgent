.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

msg_error:
    .ascii "error at "
msg_error_len = . - msg_error

msg_newline:
    .ascii "\n"

.bss
.align 16
input_buffer:
    .zero 65536
input_size:
    .zero 8

.text
_start:
    xor r12, r12
1:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    
    test rax, rax
    jz 2f
    
    add r12, rax
    cmp r12, 65536
    jl 1b
    
2:
    mov qword ptr [rip + input_size], r12
    
    xor r13, r13
    
    call skip_ws
    call parse_value
    call skip_ws
    
    cmp r13, r12
    jne error
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

skip_ws:
    mov r15, qword ptr [rip + input_size]
    lea rsi, [rip + input_buffer]
1:
    cmp r13, r15
    jge 2f
    
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x20
    je 3f
    cmp al, 0x09
    je 3f
    cmp al, 0x0A
    je 3f
    cmp al, 0x0D
    je 3f
    jmp 2f
    
3:
    inc r13
    jmp 1b
2:
    ret

parse_value:
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6E
    je parse_null
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb error
    cmp al, 0x39
    jbe parse_number
    jmp error

parse_object:
    inc r13
    call skip_ws
    
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x7D
    je 1f
    
2:
    call parse_member
    
    call skip_ws
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x7D
    je 1f
    
    cmp al, 0x2C
    jne error
    inc r13
    
    call skip_ws
    jmp 2b
    
1:
    inc r13
    ret

parse_member:
    call parse_string
    call skip_ws
    
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x3A
    jne error
    inc r13
    
    call skip_ws
    call parse_value
    ret

parse_array:
    inc r13
    call skip_ws
    
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x5D
    je 1f
    
2:
    call parse_value
    
    call skip_ws
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x5D
    je 1f
    
    cmp al, 0x2C
    jne error
    inc r13
    
    call skip_ws
    jmp 2b
    
1:
    inc r13
    ret

parse_string:
    mov r15, qword ptr [rip + input_size]
    lea rsi, [rip + input_buffer]
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x22
    jne error
    inc r13
    
1:
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x22
    je 2f
    
    cmp al, 0x5C
    je 3f
    
    cmp al, 0x20
    jb error
    
    inc r13
    jmp 1b
    
3:
    inc r13
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x22
    je 4f
    cmp al, 0x5C
    je 4f
    cmp al, 0x2F
    je 4f
    cmp al, 0x62
    je 4f
    cmp al, 0x66
    je 4f
    cmp al, 0x6E
    je 4f
    cmp al, 0x72
    je 4f
    cmp al, 0x74
    je 4f
    cmp al, 0x75
    je 5f
    jmp error
    
4:
    inc r13
    jmp 1b
    
5:
    inc r13
    mov rcx, 4
6:
    cmp r13, r15
    jge error
    movzx eax, byte ptr [rsi + r13]
    
    cmp al, 0x30
    jb error
    cmp al, 0x39
    jbe 7f
    cmp al, 0x41
    jb error
    cmp al, 0x46
    jbe 7f
    cmp al, 0x61
    jb error
    cmp al, 0x66
    ja error
7:
    inc r13
    dec rcx
    jnz 6b
    jmp 1b
    
2:
    inc r13
    ret

parse_true:
    inc r13
    
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x72
    jne error
    inc r13
    
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x75
    jne error
    inc r13
    
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x65
    jne error

    ret

    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]

    cmp r13, r15
    jge error

    cmp al, 0x73
    jne error
    inc r13
    
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x65
    jne error

    inc r13
    
    mov r15, qword ptr [rip + input_size]
    cmp r13, r15
    jge error
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x75
    jne error
    inc r13
    
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x6C
    jne error
    inc r13
    
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]

    ret

parse_number:
    mov r15, qword ptr [rip + input_size]
    lea rsi, [rip + input_buffer]
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x2D
    jne 1f
    inc r13
    cmp r13, r15
    jge error
    
1:
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x30
    je 2f
    
    cmp al, 0x31
    jb error
    cmp al, 0x39
    ja error
    
    inc r13
    
3:
    cmp r13, r15
    jge 4f
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x30
    jb 4f
    cmp al, 0x39
    ja 4f
    inc r13
    jmp 3b
    
2:
    inc r13
    
4:
    cmp r13, r15
    jge 7f
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x2E
    jne 6f
    
    inc r13
    cmp r13, r15
    jge error
    
    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x30
    jb error
    cmp al, 0x39
    ja error
    inc r13
    
5:
    cmp r13, r15

    movzx eax, byte ptr [rsi + r13]
    cmp al, 0x30
    jb 6f
    cmp al, 0x39
    ja 6f
    inc r13
    jmp 5b

    cmp r13, r15

    movzx eax, byte ptr [rsi + r13]

    cmp al, 0x45

    inc r13

error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_error]
    mov rdx, msg_error_len
    syscall
    
    mov rax, r13
    call print_decimal
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_newline]
    mov rdx, 1
    syscall
    
    mov rax, 60
    mov rdi, 1
    syscall

print_decimal:
    mov rcx, 10
    mov rbx, rax
    xor r10, r10
    
    test rax, rax
    jnz 1f
    
    dec rsp
    mov byte ptr [rsp], 0x30
    inc r10
    jmp 3f
    
1:
    xor rdx, rdx
    mov rax, rbx
    div rcx
    add dl, 0x30
    dec rsp
    mov byte ptr [rsp], dl
    inc r10
    mov rbx, rax
    test rax, rax
    jnz 1b
    
3:
    mov rax, 1
    mov rdi, 1
    mov rsi, rsp
    mov rdx, r10
    syscall
    
    add rsp, r10
    ret
