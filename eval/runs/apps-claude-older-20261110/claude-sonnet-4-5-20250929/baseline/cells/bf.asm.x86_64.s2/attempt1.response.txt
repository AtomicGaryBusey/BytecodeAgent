```asm
.globl _start

.text
_start:
    # Read all of stdin
    xor r12, r12
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    # Find '!' separator
    xor rcx, rcx
    mov r13, r12
find_sep:
    cmp rcx, r12
    jge sep_done
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x21
    jne 1f
    mov r13, rcx
    jmp sep_done
1:
    inc rcx
    jmp find_sep

sep_done:
    # Set up program input start
    lea rbx, [r13 + 1]
    cmp r13, r12
    jl 1f
    mov rbx, r12
1:
    
    # Check bracket matching
    xor r15, r15
    xor rcx, rcx
check_brackets:
    cmp rcx, r13
    jge check_done
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rcx]
    cmp al, '['
    jne 1f
    inc r15
    cmp r15, 1000
    jg bracket_error
    jmp 2f
1:
    cmp al, ']'
    jne 2f
    test r15, r15
    jz bracket_error
    dec r15
2:
    inc rcx
    jmp check_brackets

check_done:
    test r15, r15
    jnz bracket_error
    
    # Execute the program
    xor r15, r15
    xor r14, r14
    
execute_loop:
    cmp r15, r13
    jge exit_success
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r15]
    
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_inc
    cmp al, '-'
    je cmd_dec
    cmp al, '.'
    je cmd_output
    cmp al, ','
    je cmd_input
    cmp al, '['
    je cmd_loop_start
    cmp al, ']'
    je cmd_loop_end
    
    inc r15
    jmp execute_loop

cmd_right:
    inc r14
    inc r15
    jmp execute_loop

cmd_left:
    dec r14
    inc r15
    jmp execute_loop

cmd_inc:
    lea rsi, [rip + tape]
    inc byte ptr [rsi + r14]
    inc r15
    jmp execute_loop

cmd_dec:
    lea rsi, [rip + tape]
    dec byte ptr [rsi + r14]
    inc r15
    jmp execute_loop

cmd_output:
    lea rsi, [rip + tape]
    add rsi, r14
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    inc r15
    jmp execute_loop

cmd_input:
    cmp rbx, r12
    jge input_eof
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rbx]
    inc rbx
    jmp store_input
input_eof:
    xor eax, eax
store_input:
    lea rsi, [rip + tape]
    mov byte ptr [rsi + r14], al
    inc r15
    jmp execute_loop

cmd_loop_start:
    lea rsi, [rip + tape]
    movzx eax, byte ptr [rsi + r14]
    test al, al
    jnz 1f
    mov rcx, 1
    inc r15
find_close:
    cmp r15, r13
    jge execute_loop
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r15]
    cmp al, '['
    jne 2f
    inc rcx
2:
    cmp al, ']'
    jne 3f
    dec rcx
    jz 1f
3:
    inc r15
    jmp find_close
1:
    inc r15
    jmp execute_loop

cmd_loop_end:
    lea rsi, [rip + tape]
    movzx eax, byte ptr [rsi + r14]
    test al, al
    jz 1f
    mov rcx, 1
    dec r15
find_open:
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r15]
    cmp al, ']'
    jne 2f
    inc rcx
2:
    cmp al, '['
    jne 3f
    dec rcx
    jz 4f
3:
    dec r15
    jmp find_open
4:
    inc r15
    jmp execute_loop
1:
    inc r15
    jmp execute_loop

bracket_error:
    mov rax, 60
    mov rdi, 1
    syscall

exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall

.bss
.align 8
input_buffer:
    .zero 65536
tape:
    .zero 30000
```
