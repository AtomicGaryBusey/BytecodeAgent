.globl _start

.text

_start:
    adrp x9, input_buffer
    add x9, x9, :lo12:input_buffer
    mov x10, x9

read_loop:
    adrp x11, input_buffer
    add x11, x11, :lo12:input_buffer
    sub x1, x10, x11
    mov x2, #65536
    sub x2, x2, x1
    cmp x2, #0
    b.le read_done
    
    mov x0, #0
    mov x1, x10
    mov x8, #63
    svc #0
    
    cmp x0, #0
    b.le read_done
    
    add x10, x10, x0
    b read_loop

read_done:
    adrp x9, input_end
    add x9, x9, :lo12:input_end
    str x10, [x9]
    
    adrp x0, input_buffer
    add x0, x0, :lo12:input_buffer
    mov x1, x10

find_sep:
    cmp x0, x1
    b.ge sep_not_found
    ldrb w2, [x0]
    cmp w2, #0x21
    b.eq sep_found
    add x0, x0, #1
    b find_sep

sep_found:
    adrp x9, program_end
    add x9, x9, :lo12:program_end
    str x0, [x9]
    
    add x0, x0, #1
    adrp x9, prog_input
    add x9, x9, :lo12:prog_input
    str x0, [x9]
    b validate

sep_not_found:
    adrp x9, program_end
    add x9, x9, :lo12:program_end
    str x1, [x9]
    
    adrp x9, prog_input
    add x9, x9, :lo12:prog_input
    str x1, [x9]

validate:
    adrp x0, input_buffer
    add x0, x0, :lo12:input_buffer
    adrp x9, program_end
    add x9, x9, :lo12:program_end
    ldr x1, [x9]
    
    mov x2, #0

val_loop:
    cmp x0, x1
    b.ge val_done
    
    ldrb w3, [x0]
    cmp w3, #0x5b
    b.eq val_open
    cmp w3, #0x5d
    b.eq val_close
    add x0, x0, #1
    b val_loop

val_open:
    add x2, x2, #1
    add x0, x0, #1
    b val_loop

val_close:
    sub x2, x2, #1
    cmp x2, #0
    b.lt error_exit
    add x0, x0, #1
    b val_loop

val_done:
    cmp x2, #0
    b.ne error_exit
    
    adrp x19, tape
    add x19, x19, :lo12:tape
    
    adrp x20, input_buffer
    add x20, x20, :lo12:input_buffer
    
    adrp x9, program_end
    add x9, x9, :lo12:program_end
    ldr x21, [x9]
    
    adrp x9, prog_input
    add x9, x9, :lo12:prog_input
    ldr x22, [x9]
    
    adrp x9, input_end
    add x9, x9, :lo12:input_end
    ldr x23, [x9]

exec_loop:
    cmp x20, x21
    b.ge exit_ok
    
    ldrb w0, [x20]
    
    cmp w0, #0x3e
    b.eq op_right
    cmp w0, #0x3c
    b.eq op_left
    cmp w0, #0x2b
    b.eq op_inc
    cmp w0, #0x2d
    b.eq op_dec
    cmp w0, #0x2e
    b.eq op_out
    cmp w0, #0x2c
    b.eq op_in
    cmp w0, #0x5b
    b.eq op_lbr
    cmp w0, #0x5d
    b.eq op_rbr
    
    add x20, x20, #1
    b exec_loop

op_right:
    add x19, x19, #1
    add x20, x20, #1
    b exec_loop

op_left:
    sub x19, x19, #1
    add x20, x20, #1
    b exec_loop

op_inc:
    ldrb w0, [x19]
    add w0, w0, #1
    strb w0, [x19]
    add x20, x20, #1
    b exec_loop

op_dec:
    ldrb w0, [x19]
    sub w0, w0, #1
    strb w0, [x19]
    add x20, x20, #1
    b exec_loop

op_out:
    mov x0, #1
    mov x1, x19
    mov x2, #1
    mov x8, #64
    svc #0
    add x20, x20, #1
    b exec_loop

op_in:
    cmp x22, x23
    b.ge in_eof
    ldrb w0, [x22]
    strb w0, [x19]
    add x22, x22, #1
    add x20, x20, #1
    b exec_loop

in_eof:
    strb wzr, [x19]
    add x20, x20, #1
    b exec_loop

op_lbr:
    ldrb w0, [x19]
    cmp w0, #0
    b.ne op_lbr_skip
    
    mov x1, #1
    add x20, x20, #1
find_rbr:
    ldrb w0, [x20]
    cmp w0, #0x5b
    b.eq inc_depth
    cmp w0, #0x5d
    b.eq dec_depth
    add x20, x20, #1
    b find_rbr

inc_depth:
    add x1, x1, #1
    add x20, x20, #1
    b find_rbr

dec_depth:
    sub x1, x1, #1
    cmp x1, #0
    b.eq found_rbr
    add x20, x20, #1
    b find_rbr

found_rbr:
    add x20, x20, #1
    b exec_loop

op_lbr_skip:
    add x20, x20, #1
    b exec_loop

op_rbr:
    ldrb w0, [x19]
    cmp w0, #0
    b.eq op_rbr_skip
    
    mov x1, #1
    sub x20, x20, #1
find_lbr:
    ldrb w0, [x20]
    cmp w0, #0x5d
    b.eq inc_depth_b
    cmp w0, #0x5b
    b.eq dec_depth_b
    sub x20, x20, #1
    b find_lbr

inc_depth_b:
    add x1, x1, #1
    sub x20, x20, #1
    b find_lbr

dec_depth_b:
    sub x1, x1, #1
    cmp x1, #0
    b.eq found_lbr
    sub x20, x20, #1
    b find_lbr

found_lbr:
    add x20, x20, #1
    b exec_loop

op_rbr_skip:
    add x20, x20, #1
    b exec_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

error_exit:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.align 4
input_buffer:
    .zero 65536
input_end:
    .zero 8
program_end:
    .zero 8
prog_input:
    .zero 8
tape:
    .zero 30000
