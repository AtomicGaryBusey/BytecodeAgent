.globl _start

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

msg_error:
    .ascii "error at "
msg_error_len = . - msg_error

newline_char:
    .byte 0x0a

.bss
.align 16
input_buffer:
    .space 65536
input_len:
    .space 8
current_pos:
    .space 8
num_buffer:
    .space 32

.text
_start:
    call read_input
    mov qword ptr [rip + current_pos], 0
    call parse_document
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_ok]
    mov rdx, msg_ok_len
    syscall
    
    mov rax, 60
    xor rdi, rdi
    syscall

read_input:
    xor r12, r12
.read_loop:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    
    test rax, rax
    jz .read_done
    
    add r12, rax
    cmp r12, 65536
    jb .read_loop
.read_done:
    mov qword ptr [rip + input_len], r12
    ret

error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + msg_error]
    mov rdx, msg_error_len
    syscall
    
    mov rax, qword ptr [rip + current_pos]
    call print_number
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    
    mov rax, 60
    mov rdi, 1
    syscall

print_number:
    push rbx
    push r12
    
    mov rbx, rax
    lea r12, [rip + num_buffer]
    add r12, 31
    mov byte ptr [r12], 0
.pn_loop:
    dec r12
    mov rax, rbx
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add dl, '0'
    mov byte ptr [r12], dl
    mov rbx, rax
    test rax, rax
    jnz .pn_loop
    
    lea rax, [rip + num_buffer]
    add rax, 31
    sub rax, r12
    mov rdx, rax
    
    mov rax, 1
    mov rdi, 1
    mov rsi, r12
    syscall
    
    pop r12
    pop rbx
    ret

get_byte:
    mov rsi, qword ptr [rip + current_pos]
    cmp rsi, qword ptr [rip + input_len]
    jae .at_end
    lea rdi, [rip + input_buffer]
    movzx eax, byte ptr [rdi + rsi]
    or eax, 0x100
    ret
.at_end:
    xor eax, eax
    ret

advance:
    inc qword ptr [rip + current_pos]
    ret

skip_ws:
.loop:
    call get_byte
    jz .done
    cmp al, 0x20
    je .skip
    cmp al, 0x09
    je .skip
    cmp al, 0x0a
    je .skip
    cmp al, 0x0d
    je .skip
    ret
.skip:
    call advance
    jmp .loop
.done:
    ret

parse_document:
    call skip_ws
    call parse_value
    call skip_ws
    mov rsi, qword ptr [rip + current_pos]
    cmp rsi, qword ptr [rip + input_len]
    jne error
    ret

parse_value:
    call get_byte
    jz error
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe parse_number
    jmp error

parse_object:
    call advance
    call skip_ws
    call get_byte
    jz error
    cmp al, '}'
    je .obj_empty
.obj_member_loop:
    call parse_string
    call skip_ws
    call get_byte
    jz error
    cmp al, ':'
    jne error
    call advance
    call skip_ws
    call parse_value
    call skip_ws
    call get_byte
    jz error
    cmp al, ','
    je .obj_comma
    cmp al, '}'
    je .obj_close
    jmp error
.obj_comma:
    call advance
    call skip_ws
    jmp .obj_member_loop
.obj_empty:
.obj_close:
    call advance
    ret

parse_array:
    call advance
    call skip_ws
    call get_byte
    jz error
    cmp al, ']'
    je .arr_empty
.arr_element_loop:
    call parse_value
    call skip_ws
    call get_byte
    jz error
    cmp al, ','
    je .arr_comma
    cmp al, ']'
    je .arr_close
    jmp error
.arr_comma:
    call advance
    call skip_ws
    jmp .arr_element_loop
.arr_empty:
.arr_close:
    call advance
    ret

parse_string:
    call get_byte
    jz error
    cmp al, '"'
    jne error
    call advance
.str_loop:
    call get_byte
    jz error
    cmp al, '"'
    je .str_done
    cmp al, '\\'
    je .str_escape
    cmp al, 0x20
    jb error
    call advance
    jmp .str_loop
.str_escape:
    call advance
    call get_byte
    jz error
    cmp al, '"'
    je .str_simple_escape
    cmp al, '\\'
    je .str_simple_escape
    cmp al, '/'
    je .str_simple_escape
    cmp al, 'b'
    je .str_simple_escape
    cmp al, 'f'
    je .str_simple_escape
    cmp al, 'n'
    je .str_simple_escape
    cmp al, 'r'
    je .str_simple_escape
    cmp al, 't'
    je .str_simple_escape
    cmp al, 'u'
    je .str_unicode
    jmp error
.str_simple_escape:
    call advance
    jmp .str_loop
.str_unicode:
    call advance
    call parse_hex_digit
    call advance
    call parse_hex_digit
    call advance
    call parse_hex_digit
    call advance
    call parse_hex_digit
    call advance
    jmp .str_loop
.str_done:
    call advance
    ret

parse_hex_digit:
    call get_byte
    jz error
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe .hex_ok
    cmp al, 'a'
    jb .hex_check_upper
    cmp al, 'f'
    jbe .hex_ok
    jmp error
.hex_check_upper:
    cmp al, 'A'
    jb error
    cmp al, 'F'
    jbe .hex_ok
    jmp error
.hex_ok:
    ret

parse_true:
    call advance
    call get_byte
    jz error
    cmp al, 'r'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 'u'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 'e'
    jne error
    call advance
    ret

parse_false:
    call advance
    call get_byte
    jz error
    cmp al, 'a'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 'l'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 's'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 'e'
    jne error
    call advance
    ret

parse_null:
    call advance
    call get_byte
    jz error
    cmp al, 'u'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 'l'
    jne error
    call advance
    call get_byte
    jz error
    cmp al, 'l'
    jne error
    call advance
    ret

parse_number:
    call get_byte
    jz error
    cmp al, '-'
    je .num_negative
    jmp .num_int
.num_negative:
    call advance
.num_int:
    call get_byte
    jz error
    cmp al, '0'
    je .num_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
.num_digits:
    call advance
    call get_byte
    jz .num_done_no_frac
    cmp al, '0'
    jb .num_done_no_frac
    cmp al, '9'
    ja .num_done_no_frac
    jmp .num_digits
.num_zero:
    call advance
.num_done_no_frac:
    call get_byte
    jz .num_end
    cmp al, '.'
    je .num_frac
    cmp al, 'e'
    je .num_exp
    cmp al, 'E'
    je .num_exp
    ret
.num_frac:
    call advance
    call get_byte
    jz error
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
.num_frac_digits:
    call advance
    call get_byte
    jz .num_end
    cmp al, '0'
    jb .num_check_exp
    cmp al, '9'
    ja .num_check_exp
    jmp .num_frac_digits
.num_check_exp:
    cmp al, 'e'
    je .num_exp
    cmp al, 'E'
    je .num_exp
    ret
.num_exp:
    call advance
    call get_byte
    jz error
    cmp al, '+'
    je .num_exp_sign
    cmp al, '-'
    je .num_exp_sign
    jmp .num_exp_digits_start
.num_exp_sign:
    call advance
.num_exp_digits_start:
    call get_byte
    jz error
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
.num_exp_digits:
    call advance
    call get_byte
    jz .num_end
    cmp al, '0'
    jb .num_end
    cmp al, '9'
    ja .num_end
    jmp .num_exp_digits
.num_end:
    ret
