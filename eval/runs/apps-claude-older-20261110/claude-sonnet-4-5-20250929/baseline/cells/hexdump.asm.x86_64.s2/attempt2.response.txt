.globl _start

.section .rodata
hex_chars:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 256

.text

_start:
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    jz read_done
    syscall
    
    test rax, rax
    jle read_done
    add r12d, eax
    jmp read_loop
    
read_done:
    xor r13d, r13d
    
line_loop:
    cmp r13d, r12d
    jae final_offset
    
    mov r14d, r12d
    sub r14d, r13d
    cmp r14d, 16
    jbe 1f
    mov r14d, 16
1:
    lea rbx, [rip + input_buffer]
    add rbx, r13
    
    lea rdi, [rip + output_buffer]
    
    mov r8d, r13d
    mov ecx, 8
1:
    rol r8d, 4
    movzx edx, r8b
    and edx, 0x0f
    lea rsi, [rip + hex_chars]
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    
    mov word ptr [rdi], 0x2020
    add rdi, 2
    
    xor r15d, r15d
    
hex_byte_loop:
    cmp r15d, r14d
    jae hex_padding
    
    movzx eax, byte ptr [rbx + r15]
    
    mov edx, eax
    shr edx, 4
    lea rsi, [rip + hex_chars]
    mov cl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], cl
    inc rdi
    
    and eax, 0x0f
    lea rsi, [rip + hex_chars]
    mov cl, byte ptr [rsi + rax]
    mov byte ptr [rdi], cl
    inc rdi
    
    mov byte ptr [rdi], 0x20
    inc rdi
    
    inc r15d
    cmp r15d, 8
    jne hex_byte_loop
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_byte_loop
    
hex_padding:
    cmp r15d, 16
    jae hex_done
    
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    
    inc r15d
    cmp r15d, 8
    jne hex_padding
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_padding
    
hex_done:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2
    
    xor r15d, r15d
    
ascii_loop:
    cmp r15d, r14d
    jae ascii_done
    
    movzx eax, byte ptr [rbx + r15]
    
    cmp al, 0x20
    jb 1f
    cmp al, 0x7e
    jbe 2f
1:
    mov al, 0x2e
2:
    mov byte ptr [rdi], al
    inc rdi
    inc r15d
    jmp ascii_loop
    
ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    
    add r13d, r14d
    jmp line_loop
    
final_offset:
    lea rdi, [rip + output_buffer]
    
    mov r8d, r12d
    mov ecx, 8
1:
    rol r8d, 4
    movzx edx, r8b
    and edx, 0x0f
    lea rsi, [rip + hex_chars]
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    
    mov byte ptr [rdi], 0x0a
    
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output_buffer]
    mov edx, 9
    syscall
    
    mov eax, 60
    xor edi, edi
    syscall
