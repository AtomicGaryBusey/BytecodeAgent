```asm
.globl _start

.section .rodata
ok_msg: .ascii "ok\n"
ok_msg_len = . - ok_msg
error_msg: .ascii "error at "
error_msg_len = . - error_msg

.bss
.align 16
input_buf: .space 65536
input_len: .space 8
pos: .space 8
depth: .space 8
digit_buf: .space 32

.text

_start:
    xor ebx, ebx
1:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jle 2f
    add rbx, rax
    cmp rbx, 65536
    jl 1b
2:
    mov qword ptr [rip + input_len], rbx
    xor eax, eax
    mov qword ptr [rip + pos], rax
    mov qword ptr [rip + depth], rax
    
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jne error
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_msg_len
    syscall
    xor edi, edi
    jmp do_exit

error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_msg_len
    syscall
    mov rax, qword ptr [rip + pos]
    call print_num
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + digit_buf]
    mov byte ptr [rsi], 10
    mov rdx, 1
    syscall
    mov rdi, 1

do_exit:
    mov rax, 60
    syscall
    ud2

skip_ws:
    push rbx
1:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jge 2f
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 0x20
    je 3f
    cmp cl, 0x09
    je 3f
    cmp cl, 0x0A
    je 3f
    cmp cl, 0x0D
    jne 2f
3:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp 1b
2:
    pop rbx
    ret

get_char:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jge error
    lea rbx, [rip + input_buf]
    movzx eax, byte ptr [rbx + rax]
    ret

advance:
    mov rax, qword ptr [rip + pos]
    inc rax
    mov qword ptr [rip + pos], rax
    ret

parse_value:
    call get_char
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jl error
    cmp al, '9'
    jle parse_number
    jmp error

parse_object:
    mov rax, qword ptr [rip + depth]
    inc rax
    mov qword ptr [rip + depth], rax
    call advance
    call skip_ws
    call get_char
    cmp al, '}'
    je 1f
2:
    call parse_string
    call skip_ws
    call get_char
    cmp al, ':'
    jne error
    call advance
    call skip_ws
    call parse_value
    call skip_ws
    call get_char
    cmp al, ','
    je 3f
    cmp al, '}'
    jne error
    jmp 1f
3:
    call advance
    call skip_ws
    jmp 2b
1:
    call advance
    mov rax, qword ptr [rip + depth]
    dec rax
    mov qword ptr [rip + depth], rax
    ret

parse_array:
    mov rax, qword ptr [rip + depth]
    inc rax
    mov qword ptr [rip + depth], rax
    call advance
    call skip_ws
    call get_char
    cmp al, ']'
    je 1f
2:
    call parse_value
    call skip_ws
    call get_char
    cmp al, ','
    je 3f
    cmp al, ']'
    jne error
    jmp 1f
3:
    call advance
    call skip_ws
    jmp 2b
1:
    call advance
    mov rax, qword ptr [rip + depth]
    dec rax
    mov qword ptr [rip + depth], rax
    ret

parse_string:
    call advance
1:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jge error
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 0x20
    jl error
    cmp cl, '"'
    je 2f
    cmp cl, 0x5C
    je 3f
    inc rax
    mov qword ptr [rip + pos], rax
    jmp 1b
2:
    call advance
    ret
3:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jge error
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '"'
    je 4f
    cmp cl, 0x5C
    je 4f
    cmp cl, '/'
    je 4f
    cmp cl, 'b'
    je 4f
    cmp cl, 'f'
    je 4f
    cmp cl, 'n'
    je 4f
    cmp cl, 'r'
    je 4f
    cmp cl, 't'
    je 4f
    cmp cl, 'u'
    jne error
    inc rax
    mov qword ptr [rip + pos], rax
    mov r8d, 4
5:
    cmp rax, qword ptr [rip + input_len]
    jge error
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jl error
    cmp cl, '9'
    jle 6f
    cmp cl, 'a'
    jl 7f
    cmp cl, 'f'
    jle 6f
7:
    cmp cl, 'A'
    jl error
    cmp cl, 'F'
    jg error
6:
    inc rax
    mov qword ptr [rip + pos], rax
    dec r8d
    jnz 5b
    jmp 1b
4:
    inc rax
    mov qword ptr [rip + pos], rax
    jmp 1b

parse_true:
    call advance
    call get_char
    cmp al, 'r'
    jne error
    call advance
    call get_char
    cmp al, 'u'
    jne error
    call advance
    call get_char
    cmp al, 'e'
    jne error
    call advance
    ret

parse_false:
    call advance
    call get_char
    cmp al, 'a'
    jne error
    call advance
    call get_char
    cmp al, 'l'
    jne error
    call advance
    call get_char
    cmp al, 's'
    jne error
    call advance
    call get_char
    cmp al, 'e'
    jne error
    call advance
    ret

parse_null:
    call advance
    call get_char
    cmp al, 'u'
    jne error
    call advance
    call get_char
    cmp al, 'l'
    jne error
    call advance
    call get_char
    cmp al, 'l'
    jne error
    call advance
    ret

parse_number:
    push rbx
    call get_char
    cmp al, '-'
    jne 1f
    call advance
1:
    call get_char
    cmp al, '0'
    jl error
    cmp al, '9'
    jg error
    cmp al, '0'
    je 2f
    call advance
3:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jge 4f
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jl 4f
    cmp cl, '9'
    jg 4f
    inc rax
    mov qword ptr [rip + pos], rax
    jmp 3b
2:
    call advance
4:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jge 5f
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '.'
    jne 5f
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jge error
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jl error
    cmp cl, '9'
    jg error
6:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jge 5f
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jl 5f
    cmp cl, '9'
    jle 6b
5:
    mov rax, qword ptr [rip + pos]
    cmp rax, qword ptr [rip + input_len]
    jge 7f
    lea rbx, [rip + input_buf]
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, 'e'
    je 8f
    cmp cl, 'E'
    jne 7f
8:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jge error
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '+'
    je 9f
    cmp cl, '-'
    jne 10f
9:
    inc rax
    mov qword ptr [rip + pos], rax
10:
    cmp rax, qword ptr [rip + input_len]
    jge error
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jl error
    cmp cl, '9'
    jg error
11:
    inc rax
    mov qword ptr [rip + pos], rax
    cmp rax, qword ptr [rip + input_len]
    jge 7f
    movzx ecx, byte ptr [rbx + rax]
    cmp cl, '0'
    jl 7f
    cmp cl, '9'
    jle 11b
7:
    pop rbx
    ret

print_num:
    push rbx
    push r12
    mov rbx, rax
    lea r12, [rip + digit_buf]
    add r12, 31
    mov byte ptr [r12], 0
    dec r12
    test rbx, rbx
    jnz 1f
    mov byte ptr [r12], '0'
    jmp 3f
1:
    test rbx, rbx
    jz 2f
    mov rax, rbx
    xor edx, edx
    mov rcx, 10
    div rcx
    add dl, '0'
    mov byte ptr [r12], dl
    dec r12
    mov rbx, rax
    jmp 1b
2:
    inc r12
3:
    lea rax, [rip + digit_buf]
    add rax, 32
    sub rax, r12
    mov rdx, rax
    mov rax, 1
    mov rdi, 1
    mov rsi, r12
    syscall
    pop r12
    pop rbx
    ret
```
