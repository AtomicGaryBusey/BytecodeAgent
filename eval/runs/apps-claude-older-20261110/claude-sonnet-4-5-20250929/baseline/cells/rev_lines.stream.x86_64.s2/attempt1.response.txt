.globl _start

.section .rodata
newline:
    .byte 0x0A

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65536

.text
_start:
    xor r12, r12
    
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    
    test rax, rax
    jz read_done
    
    add r12, rax
    cmp r12, 65536
    jb read_loop
    
read_done:
    test r12, r12
    jz exit_success
    
    xor r13, r13
    
process_lines:
    cmp r13, r12
    jge exit_success
    
    mov r14, r13
    
find_newline:
    cmp r14, r12
    jge line_at_eof
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r14]
    cmp al, 0x0A
    je reverse_and_write
    
    inc r14
    jmp find_newline
    
line_at_eof:
    mov r14, r12
    
reverse_and_write:
    mov r15, r13
    
reverse_loop:
    cmp r15, r14
    jge reverse_done
    
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + r15]
    
    mov rcx, r14
    sub rcx, r15
    dec rcx
    
    lea rdi, [rip + output_buffer]
    mov byte ptr [rdi + rcx], al
    
    inc r15
    jmp reverse_loop
    
reverse_done:
    mov rax, r14
    sub rax, r13
    mov rdx, rax
    
    test rdx, rdx
    jz write_newline
    
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + output_buffer]
    syscall
    
write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    
    mov r13, r14
    inc r13
    jmp process_lines
    
exit_success:
    mov rax, 60
    xor rdi, rdi
    syscall
