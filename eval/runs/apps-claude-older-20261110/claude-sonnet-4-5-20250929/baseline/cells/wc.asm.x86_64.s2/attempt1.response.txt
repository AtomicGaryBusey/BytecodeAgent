```asm
.globl _start

.section .bss
.align 16
input_buffer:
    .space 65536

output_buffer:
    .space 128

.text

_start:
    # Read all input
    xor r12, r12              # r12 = total bytes read
    
read_loop:
    mov rax, 0                # syscall: read
    mov rdi, 0                # stdin
    lea rsi, [rip + input_buffer]
    add rsi, r12              # buffer + offset
    mov rdx, 65536
    sub rdx, r12              # remaining space
    jz read_done              # if no space, done
    syscall
    
    test rax, rax
    jle read_done             # if <= 0, done
    add r12, rax              # add bytes read
    jmp read_loop
    
read_done:
    # Process the input
    xor r14, r14              # r14 = line count
    xor r15, r15              # r15 = word count
    xor rbx, rbx              # rbx = current position
    xor ecx, ecx              # ecx = in_word flag
    
    test r12, r12
    jz process_done
    
    lea r13, [rip + input_buffer]
    
process_loop:
    cmp rbx, r12
    jge process_done
    
    movzx eax, byte ptr [r13 + rbx]
    
    # Check if newline
    cmp al, 0x0A
    jne check_whitespace
    inc r14
    
check_whitespace:
    # Check if whitespace: 0x20, 0x09, 0x0A, 0x0B, 0x0C, 0x0D
    cmp al, 0x20
    je is_whitespace
    cmp al, 0x09
    je is_whitespace
    cmp al, 0x0A
    je is_whitespace
    cmp al, 0x0B
    je is_whitespace
    cmp al, 0x0C
    je is_whitespace
    cmp al, 0x0D
    je is_whitespace
    
    # Not whitespace - it's a word character
    test ecx, ecx
    jnz already_in_word
    inc r15                   # start new word
    mov ecx, 1                # set in_word flag
already_in_word:
    inc rbx
    jmp process_loop
    
is_whitespace:
    xor ecx, ecx              # clear in_word flag
    inc rbx
    jmp process_loop
    
process_done:
    # Build output string
    lea rdi, [rip + output_buffer]
    
    # Write line count
    mov rax, r14
    call uint_to_decimal
    mov byte ptr [rdi], ' '
    inc rdi
    
    # Write word count
    mov rax, r15
    call uint_to_decimal
    mov byte ptr [rdi], ' '
    inc rdi
    
    # Write byte count
    mov rax, r12
    call uint_to_decimal
    mov byte ptr [rdi], 0x0A
    inc rdi
    
    # Write output
    lea rsi, [rip + output_buffer]
    sub rdi, rsi              # length
    mov rdx, rdi
    mov rax, 1                # syscall: write
    mov rdi, 1                # stdout
    syscall
    
    # Exit
    mov rax, 60               # syscall: exit
    xor rdi, rdi              # status 0
    syscall

# Convert unsigned integer in rax to decimal string at rdi
# Updates rdi to point after the last digit
uint_to_decimal:
    push rbx
    push rcx
    push rdx
    push r8
    
    test rax, rax
    jnz non_zero
    
    # Special case: 0
    mov byte ptr [rdi], '0'
    inc rdi
    jmp decimal_done
    
non_zero:
    mov r8, rax               # save original value
    
    # Count digits
    xor rcx, rcx              # digit count
    mov rbx, rax
digit_count_loop:
    inc rcx
    xor rdx, rdx
    mov rax, rbx
    mov rbx, 10
    div rbx
    mov rbx, rax
    test rbx, rbx
    jnz digit_count_loop
    
    # rcx = number of digits
    add rdi, rcx              # point past the last digit
    push rdi                  # save for return
    
    # Convert digits (write backwards)
    mov rax, r8               # restore original value
digit_convert_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx                   # rax = quotient, rdx = remainder
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz digit_convert_loop
    
    pop rdi                   # restore end pointer
    
decimal_done:
    pop r8
    pop rdx
    pop rcx
    pop rbx
    ret
```
