

export function $vm_run(l code, l len, l stack, l cap): l {
  @main:
    %pc = l 0
    %cur_addr = code
    %stack_ptr = stack
.Lloop:
    %less = csltl %pc, len
    jz .Ldone
    %op = loadl %cur_addr
    %eq0 = ceql %op, l 0
    jz .Lhalt
    %eq1 = ceql %op, l 1
    jz .Lpush
    %eq2 = ceql %op, l 2
    jz .Ladd
    %eq3 = ceql %op, l 3
    jz .Lsub
    %eq4 = ceql %op, l 4
    jz .Lmul
    %eq5 = ceql %op, l 5
    jz .Ldup
    %eq6 = ceql %op, l 6
    jz .Lswap
    %eq7 = ceql %op, l 7
    jz .Ldrop
    %eq8 = ceql %op, l 8
    jz .Lover
    %eq9 = ceql %op, l 9
    jz .Llt
    %eq10 = ceql %op, l 10
    jz .Ljz
    %eq11 = ceql %op, l 11
    jz .Ljmp
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Lhalt:
    jmp .Ldone
.Lpush:
    %operand_addr = addl %cur_addr, l 8
    %v = loadl %operand_addr
    storel %v, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 2
    %cur_addr = addl %cur_addr, l 16
    jmp .Lloop
.Ladd:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %sum = addl %a, %b
    storel %sum, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Lsub:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %diff = subl %a, %b
    storel %diff, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Lmul:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %prod = imull %a, %b
    storel %prod, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Ldup:
    %top_addr = subl %stack_ptr, l 8
    %top = loadl %top_addr
    storel %top, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Lswap:
    %addr_b = subl %stack_ptr, l 8
    %b = loadl %addr_b
    %addr_a = subl %stack_ptr, l 8
    %a = loadl %addr_a
    storel %b, %addr_a
    storel %a, %addr_b
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Ldrop:
    %stack_ptr = subl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Lover:
    %below_addr = subl %stack_ptr, l 16
    %below = loadl %below_addr
    storel %below, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Llt:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %lt = csltl %a, %b
    storel %lt, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp .Lloop
.Ljz:
    %operand_addr = addl %cur_addr, l 8
    %t = loadl %operand_addr
    %stack_ptr = subl %stack_ptr, l 8
    %v = loadl %stack_ptr
    %zero = ceql %v, l 0
    jnz .Ljz_target
    %pc = addl %pc, l 2
    %cur_addr = addl %cur_addr, l 16
    jmp .Lloop
.Ljz_target:
    %shifted = shll %t, l 3
    %cur_addr = addl code, %shifted
    %pc = %t
    jmp .Lloop
.Ljmp:
    %operand_addr = addl %cur_addr, l 8
    %t = loadl %operand_addr
    %shifted = shll %t, l 3
    %cur_addr = addl code, %shifted
    %pc = %t
    jmp .Lloop
.Ldone:
    %result = l 0
    %equal = ceql %stack_ptr, stack
    jz .Lzero_stack
    %top_addr = subl %stack_ptr, l 8
    %result = loadl %top_addr
    jmp .Lzero_stack
.Lzero_stack:
    %cap8 = shll cap, l 3
    %end_addr = addl stack, %cap8
    %current = %stack_ptr
.Lzero_loop:
    %less_end = csltl %current, %end_addr
    jz .Lzero_done
    storel l 0, %current
    %current = addl %current, l 8
    jmp .Lzero_loop
.Lzero_done:
    ret
}
