

export function $vm_run(l prog, l len, l stack, l cap): l {
  @main:
    %pc = l 0
    %cur_addr = prog
    %stack_ptr = stack
    jmp @loop
  @loop:
    %less = csltl %pc, len
    jz @done
    %op = loadl %cur_addr
    %eq0 = ceql %op, l 0
    jz @halt
    %eq1 = ceql %op, l 1
    jz @push
    %eq2 = ceql %op, l 2
    jz @add
    %eq3 = ceql %op, l 3
    jz @sub
    %eq4 = ceql %op, l 4
    jz @mul
    %eq5 = ceql %op, l 5
    jz @dup
    %eq6 = ceql %op, l 6
    jz @swap
    %eq7 = ceql %op, l 7
    jz @drop
    %eq8 = ceql %op, l 8
    jz @over
    %eq9 = ceql %op, l 9
    jz @lt
    %eq10 = ceql %op, l 10
    jz @jz
    %eq11 = ceql %op, l 11
    jz @jmp
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @halt:
    jmp @done
  @push:
    %operand_addr = addl %cur_addr, l 8
    %v = loadl %operand_addr
    storel %v, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 2
    %cur_addr = addl %cur_addr, l 16
    jmp @loop
  @add:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %sum = addl %a, %b
    storel %sum, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @sub:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %diff = subl %a, %b
    storel %diff, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @mul:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %prod = imull %a, %b
    storel %prod, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @dup:
    %top_addr = subl %stack_ptr, l 8
    %top = loadl %top_addr
    storel %top, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @swap:
    %addr_b = subl %stack_ptr, l 8
    %b = loadl %addr_b
    %addr_a = subl %stack_ptr, l 16
    %a = loadl %addr_a
    storel %b, %addr_a
    storel %a, %addr_b
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @drop:
    %stack_ptr = subl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @over:
    %below_addr = subl %stack_ptr, l 16
    %below = loadl %below_addr
    storel %below, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @lt:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %lt_val = csltl %a, %b
    storel %lt_val, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %pc = addl %pc, l 1
    %cur_addr = addl %cur_addr, l 8
    jmp @loop
  @jz:
    %operand_addr = addl %cur_addr, l 8
    %t = loadl %operand_addr
    %stack_ptr = subl %stack_ptr, l 8
    %v = loadl %stack_ptr
    %zero = ceql %v, l 0
    jz @jz_target
    %pc = addl %pc, l 2
    %cur_addr = addl %cur_addr, l 16
    jmp @loop
  @jz_target:
    %target_addr = addl prog, shll %t, l 3
    %pc = %t
    %cur_addr = %target_addr
    jmp @loop
  @jmp:
    %operand_addr = addl %cur_addr, l 8
    %t = loadl %operand_addr
    %target_addr = addl prog, shll %t, l 3
    %pc = %t
    %cur_addr = %target_addr
    jmp @loop
  @done:
    %result = l 0
    %equal = ceql %stack_ptr, stack
    jz @zero_stack
    %top_addr = subl %stack_ptr, l 8
    %result = loadl %top_addr
    jmp @zero_stack
  @zero_stack:
    %cap8 = shll cap, l 3
    %end_addr = addl stack, %cap8
    %current = %stack_ptr
  @zero_loop:
    %less_end = csltl %current, %end_addr
    jz @zero_done
    storel l 0, %current
    %current = addl %current, l 8
    jmp @zero_loop
  @zero_done:
    ret
}
