

export function l $vm_run(l code, l len, l stack, l cap) {
  @main:
    %pc = l 0
    %depth = l 0
    %code_ptr = code
    %stack_ptr = stack
  .Lcheck:
    %less = csltl %pc, %len
    jz .Ldone
    %op = loadl %code_ptr
    %eq0 = ceql %op, l 0
    jz .Lhalt
    %eq1 = ceql %op, l 1
    jz .Lpush
    %eq2 = ceql %op, l 2
    jz .Ladd
    jmp .Lnext
  .Lhalt:
    jmp .Lnext
  .Lpush:
    %operand_addr = addl %code_ptr, l 8
    %v = loadl %operand_addr
    storel %v, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %depth = addl %depth, l 1
    %pc = addl %pc, l 2
    %code_ptr = addl %code_ptr, l 16
    jmp .Lcheck
  .Ladd:
    %stack_ptr = subl %stack_ptr, l 8
    %b = loadl %stack_ptr
    %depth = subl %depth, l 1
    %stack_ptr = subl %stack_ptr, l 8
    %a = loadl %stack_ptr
    %depth = subl %depth, l 1
    %sum = addl %a, %b
    storel %sum, %stack_ptr
    %stack_ptr = addl %stack_ptr, l 8
    %depth = addl %depth, l 1
    %pc = addl %pc, l 1
    %code_ptr = addl %code_ptr, l 8
    jmp .Lcheck
  .Lnext:
    jmp .Lcheck
  .Ldone:
    %result = l 0
    ret
}
