

export function l $mp_mul(l a, l na, l b, l nb, l r) {
@start:
%cmp_na = ceql %na, l 0
jnz @na_not_zero
%zero = l 0
storel %zero, [%r]
ret l 0
@na_not_zero:
%cmp_nb = ceql %nb, l 0
jnz @nb_not_zero
%zero = l 0
storel %zero, [%r]
ret l 0
@nb_not_zero:
%a0 = loadl [%a]
%b0 = loadl [%b]
%a_low = andl %a0, l 0xFFFFFFFF
%a_high = shrl %a0, l 32
%b_low = andl %b0, l 0xFFFFFFFF
%b_high = shrl %b0, l 32
%p00 = mull %a_low, %b_low
%p11 = mull %a_high, %b_high
%p01 = mull %a_high, %b_low
%p10 = mull %a_low, %b_high
%mid = addl %p01, %p10
%carry_mid = cultl %mid, %p01
%mid_low = andl %mid, l 0xFFFFFFFF
%mid_high = shrl %mid, l 32
%mid_low_shifted = shll %mid_low, l 32
%low = addl %p00, %mid_low_shifted
%carry_low = cultl %low, %p00
%carry_mid_shifted = shll %carry_mid, l 32
%high = addl %p11, %mid_high
%high1 = addl %high, %carry_mid_shifted
%high2 = addl %high1, %carry_low
storel %low, [%r]
%r1 = addl %r, l 8
storel %high2, [%r1]
%cmp_high = ceql %high2, l 0
jnz @high_is_zero
ret l 2
@high_is_zero:
%cmp_low = ceql %low, l 0
jnz @low_is_zero
ret l 1
@low_is_zero:
ret l 0
}
