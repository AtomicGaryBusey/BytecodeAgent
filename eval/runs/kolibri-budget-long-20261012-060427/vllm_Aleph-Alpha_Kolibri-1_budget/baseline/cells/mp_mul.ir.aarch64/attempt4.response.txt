

export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r)
@start {
  l %zero = 0
  l %two = 2
  l %mask32 = 4294967295
  l %shl32 = 4294967296
  storel [ %r ], %zero
  storel [ %r, #8 ], %zero
  w %eq_na0 = ceql %na, %zero
  jnz @zero_case, %eq_na0
  w %eq_nb0 = ceql %nb, %zero
  jnz @zero_case, %eq_nb0
  jmp @mul
}
@zero_case {
  retl %zero
}
@mul {
  l %a1 = loadl [ %a ]
  l %b1 = loadl [ %b ]
  l %a0 = andl %a1, %mask32
  l %a1 = shrl %a1, 32
  l %b0 = andl %b1, %mask32
  l %b1 = shrl %b1, 32
  l %p00 = mull %a0, %b0
  l %p01 = mull %a0, %b1
  l %p10 = mull %a1, %b0
  l %p11 = mull %a1, %b1
  l %sum1 = addl %p01, %p10
  l %carry_mid = shrl %sum1, 32
  l %low_mid = andl %sum1, %mask32
  l %carry0 = shrl %p00, 32
  l %low0 = andl %p00, %mask32
  l %sum1b = addl %carry0, %low_mid
  l %l1 = andl %sum1b, %mask32
  l %carry_out1 = shrl %sum1b, 32
  l %low11 = andl %p11, %mask32
  l %carry11 = shrl %p11, 32
  l %sum2 = addl %carry_mid, %low11
  l %sum2 = addl %sum2, %carry_out1
  l %l2 = andl %sum2, %mask32
  l %carry_out2 = shrl %sum2, 32
  l %sum3 = addl %carry11, %carry_out2
  l %l3 = andl %sum3, %mask32
  l %prod_r0 = mull %l1, %shl32
  l %r0 = addl %prod_r0, %low0
  storel [ %r ], %r0
  l %prod_r1 = mull %l3, %shl32
  l %r1 = addl %prod_r1, %l2
  storel [ %r, #8 ], %r1
  jmp @done
}
@done {
  retl %two
}
