

qsort_pairs:
push rbp
mov rbp, rsp
mov qword ptr [rbp - 8], rdi
mov qword ptr [rbp - 16], rsi
mov rax, qword ptr [rbp - 16]
cmp rax, 1
jle .base_case
mov r8, 0
mov rdx, qword ptr [rbp - 8]
mov rax, qword ptr [rbp - 16]
xor rcx, rcx
.Lminloop:
cmp rcx, rax
jge .Lminend
lea r10, [rdx + rcx*2]
mov r11, qword ptr [r10]
mov rsi, qword ptr [r10 + 8]
lea r10, [rdx + r8*2]
mov rdi, qword ptr [r10]
mov r9, qword ptr [r10 + 8]
cmp r11, rdi
jg .Lmin_next
jl .Lmin_update
cmp rsi, r9
jg .Lmin_next
jl .Lmin_update
.Lmin_next:
add rcx, 1
jmp .Lminloop
.Lmin_update:
mov r8, rcx
jmp .Lmin_next
.Lminend:
mov rdx, qword ptr [rbp - 8]
lea r10, [rdx + r8*2]
mov rax, qword ptr [rdx]
mov rsi, qword ptr [rdx + 8]
mov rdi, qword ptr [r10]
mov r9, qword ptr [r10 + 8]
mov qword ptr [rdx], rdi
mov qword ptr [r10], rax
mov qword ptr [rdx + 8], r9
mov qword ptr [r10 + 8], rsi
mov rax, qword ptr [rbp - 16]
sub rax, 1
lea rdi, [rdx + 2]
call qsort_pairs
pop rbp
xor rax, rax
ret
.base_case:
pop rbp
xor rax, rax
ret
