

qsort_pairs:
push rbp
mov rbp, rsp
mov qword ptr [rbp - 8], rdi
mov qword ptr [rbp - 16], rsi
mov rcx, qword ptr [rbp - 16]
cmp rcx, 1
jle .base_case

mov r10, rcx
imul r10, r10, 16
sub r10, 16
add r10, rdi
mov r12, qword ptr [r10]
mov r13, qword ptr [r10 + 8]

mov r11, rdi
mov r8, rdi
dec rcx

.Lscan_loop:
test rcx, rcx
jz .Lscan_end
mov rax, qword ptr [r8]
mov rdx, qword ptr [r8 + 8]
cmp rax, r12
jl .Lswap
jg .Lscan_next
cmp rdx, r13
jle .Lswap
jmp .Lscan_next
.Lswap:
movups xmm0, [r11]
movups xmm1, [r8]
movups [r11], xmm1
movups [r8], xmm0
add r11, 16
.Lscan_next:
add r8, 16
dec rcx
jmp .Lscan_loop

.Lscan_end:
movups xmm0, [r11]
movups xmm1, [r10]
movups [r11], xmm1
movups [r10], xmm0
mov rax, r11
sub rax, qword ptr [rbp - 8]
shr rax, 4
mov qword ptr [rbp - 24], rax

mov qword ptr [rbp - 32], r12
mov qword ptr [rbp - 40], r13

mov rdi, qword ptr [rbp - 8]
mov rsi, qword ptr [rbp - 24]
call qsort_pairs

mov r12, qword ptr [rbp - 32]
mov r13, qword ptr [rbp - 40]

mov rax, qword ptr [rbp - 16]
sub rax, qword ptr [rbp - 24]
dec rax
mov qword ptr [rbp - 48], rax

mov rax, qword ptr [rbp - 8]
mov rcx, qword ptr [rbp - 24]
add rcx, 1
imul rcx, rcx, 16
add rax, rcx
mov rdi, rax
mov rsi, qword ptr [rbp - 48]
call qsort_pairs

pop rbp
xor rax, rax
ret

.base_case:
pop rbp
xor rax, rax
ret
