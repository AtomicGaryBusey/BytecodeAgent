

packbits_decode:
  mov x4, #0
  mov x5, x0
  mov x6, x2
  mov x7, x1
  mov x8, x3
.Lloop:
  cbnz x7, .Lhas_source
  b .Ldone
.Lhas_source:
  cbnz x8, .Lread_header
  b .Loutput_full
.Lread_header:
  ldr w18, [x5]
  add x5, x5, #1
  sub x7, x7, #1
  sxtw x18, w18
  cmp x18, #-128
  b.eq .Lnoop
  cmp x18, #0
  b.lt .Lrepeat
  b .Lliteral
.Lnoop:
  b .Lloop
.Lrepeat:
  cbnz x7, .Lrepeat_src_ok
  b .Lsrc_underflow
.Lrepeat_src_ok:
  ldr w18, [x5]
  add x5, x5, #1
  sub x7, x7, #1
  mov x9, #1
  sub x9, x9, x18
  b .Lrepeat_write_loop
.Lrepeat_write_loop:
  cbnz x9, .Lrepeat_write_loop_body
  b .Lloop
.Lrepeat_write_loop_body:
  cbnz x8, .Lrepeat_write_byte
  b .Loutput_full
.Lrepeat_write_byte:
  str w18, [x6]
  add x6, x6, #1
  sub x8, x8, #1
  add x4, x4, #1
  sub x9, x9, #1
  b .Lrepeat_write_loop
.Lliteral:
  add x9, x18, #1
  cmp x7, x9
  b.lt .Lsrc_underflow
  b .Llit_write_loop
.Llit_write_loop:
  cbnz x9, .Llit_write_loop_body
  b .Lloop
.Llit_write_loop_body:
  cbnz x8, .Llit_write_byte
  b .Loutput_full
.Llit_write_byte:
  ldr w19, [x5]
  add x5, x5, #1
  sub x7, x7, #1
  str w19, [x6]
  add x6, x6, #1
  sub x8, x8, #1
  add x4, x4, #1
  sub x9, x9, #1
  b .Llit_write_loop
.Lsrc_underflow:
  mov x0, #-2
  ret
.Loutput_full:
  mov x0, #-1
  ret
.Ldone:
  mov x0, x4
  ret
