

export function l $lcs_len(l %x, l %m, l %y, l %n, l %row)
@start:
%n1 = add %n, $1;
%j = $0;
%ptr = %row;
@zero_loop:
%lt = csltl %j, %n1;
jnz %lt, @zero_body;
jmp @zero_done;
@zero_body:
storel $0, %ptr;
%j = add %j, $1;
%ptr = add %ptr, $8;
jmp @zero_loop;
@zero_done:
%i = $0;
jmp @outer_loop;
@outer_loop:
%lt_i = csltl %i, %m;
jnz %lt_i, @inner_start;
jmp @done;
@inner_start:
%j = $1;
%ptr_row_curr = add %row, $8;
%ptr_row_prev = %row;
%ptr_x = add %x, %i;
%ptr_y = %y;
%diag = $0;
jmp @inner_loop;
@inner_loop:
%lt_j = csltl %j, %n1;
jnz %lt_j, @inner_body;
jmp @inner_done;
@inner_body:
%bx =w loadub %ptr_x;
%by =w loadub %ptr_y;
%eq = ceql %bx, %by;
%up = loadl %ptr_row_curr;
%left = loadl %ptr_row_prev;
jnz %eq, @eq_case;
jmp @else_max;
@eq_case:
%new = add %diag, $1;
jmp @after_max;
@else_max:
%lt_up = csltl %up, %left;
jnz %lt_up, @max_left;
%new = %up;
jmp @after_max;
@max_left:
%new = %left;
jmp @after_max;
@after_max:
storel %new, %ptr_row_curr;
%diag = %up;
%ptr_row_prev = %ptr_row_curr;
%ptr_row_curr = add %ptr_row_curr, $8;
%ptr_y = add %ptr_y, $1;
%j = add %j, $1;
jmp @inner_loop;
@inner_done:
%i = add %i, $1;
jmp @outer_loop;
@done:
%prod = mul %n, $8;
%result_addr = add %row, %prod;
%result = loadl %result_addr;
ret %result
