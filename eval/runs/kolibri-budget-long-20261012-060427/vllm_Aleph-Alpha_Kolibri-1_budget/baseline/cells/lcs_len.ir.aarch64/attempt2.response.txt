

export function l $lcs_len(%x, %m, %y, %n, %row) {
@start:
%n1 = add %n, $1;
%j = $0;
%ptr = %row;
jmp @zero_loop;

@zero_loop:
%lt = w csltl %j, %n1;
jnz %lt, @zero_body;
jmp @zero_done;

@zero_body:
storel $0, %ptr;
%j = add %j, $1;
%ptr = add %ptr, $8;
jmp @zero_loop;

@zero_done:
%i = $0;
jmp @outer_loop;

@outer_loop:
%lt_i = w csltl %i, %m;
jnz %lt_i, @inner_start;
jmp @done;

@inner_start:
%j = $1;
%ptr_y = %y;
%ptr_row = add %row, $8;
%ptr_x = add %x, %i;
%diag = $0;
%prev = $0;
jmp @inner_loop;

@inner_loop:
%lt_j = w csltl %j, %n1;
jnz %lt_j, @inner_body;
jmp @inner_done;

@inner_body:
%by = w loadub %ptr_y;
%bx = w loadub %ptr_x;
%eq = w ceql %bx, %by;
%up = loadl %ptr_row;
jnz %eq, @eq_case;
jmp @else_max;

@eq_case:
%new = add %diag, $1;
jmp @inner_after;

@else_max:
%lt_up = w csltl %up, %prev;
jnz %lt_up, @ne_prev;
%new = %up;
jmp @inner_after;

@ne_prev:
%new = %prev;
jmp @inner_after;

@inner_after:
storel %new, %ptr_row;
%diag = %up;
%prev = %new;
%j = add %j, $1;
%ptr_y = add %ptr_y, $1;
%ptr_row = add %ptr_row, $8;
jmp @inner_loop;

@inner_done:
%i = add %i, $1;
jmp @outer_loop;

@done:
%prod = mul %n, $8;
%result_addr = add %row, %prod;
%result = loadl %result_addr;
ret %result;
}
