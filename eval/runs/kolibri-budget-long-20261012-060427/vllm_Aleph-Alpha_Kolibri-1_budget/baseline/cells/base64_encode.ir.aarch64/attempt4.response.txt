

export function l $base64_encode(src, len, out) {
    @start:
    %zero =l movl $0
    %one =l movl $1
    %two =l movl $2
    %three =l movl $3
    %rem =l movl %len
    %group_count =l movl %zero
@loop:
    %cmp =w csltl %rem, %three
    jnz %cmp, @after_loop
    %rem =l subll %rem, %three
    %group_count =l addl %group_count, %one
    jmp @loop
@after_loop:
    %total_groups =l movl %group_count
    %eq0 =w ceql %rem, %zero
    jnz %eq0, @no_add
    %total_groups =l addl %total_groups, %one
    jmp @compute_total
@no_add:
    jmp @compute_total
@compute_total:
    %temp1 =l movl %total_groups
    %temp2 =l addl %temp1, %temp1
    %temp3 =l addl %temp2, %temp2
    %total =l movl %temp3
    %rax =l movl %total
    ret
}
