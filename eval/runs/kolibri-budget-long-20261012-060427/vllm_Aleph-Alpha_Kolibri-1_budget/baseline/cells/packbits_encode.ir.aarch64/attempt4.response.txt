

export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
  @entry:
  %i = 0
  %pos = 0
  jmp @loop
  @loop:
  %lt_n = csltl %i, %n
  jnz %lt_n, @done
  %ptr_i = add %src, %i
  %c_i = w loadub %ptr_i
  %run = 1
  %k = add %i, 1
  @runloop:
  %lt_k_n = csltl %k, %n
  jnz %lt_k_n, @rundone
  %lt_run128 = csltl %run, 128
  jnz %lt_run128, @rundone
  %ptr_k = add %src, %k
  %c_k = w loadub %ptr_k
  %eq = ceql %c_k, %c_i
  jnz %eq, @rundone
  %run = add %run, 1
  %k = add %k, 1
  jmp @runloop
  @rundone:
  %lt_3 = csltl %run, 3
  jnz %lt_3, @literal
  %room = sub %cap, %pos
  %lt_2 = csltl %room, 2
  jnz %lt_2, @notenough
  %header = sub 257, %run
  %ptr_h = add %out, %pos
  storeb %header, %ptr_h
  %ptr_d = add %ptr_h, 1
  storeb %c_i, %ptr_d
  %pos = add %pos, 2
  %i = add %i, %run
  jmp @loop
  @literal:
  %j = %i
  %L = 0
  @litloop:
  %lt_j_n = csltl %j, %n
  jnz %lt_j_n, @litdone
  %diff = sub %j, %i
  %lt_128 = csltl %diff, 128
  jnz %lt_128, @litdone
  %j2 = add %j, 2
  %lt_j2_n = csltl %j2, %n
  jnz %lt_j2_n, @litinclude
  %ptr_j = add %src, %j
  %c_j = w loadub %ptr_j
  %ptr_j1 = add %ptr_j, 1
  %c_j1 = w loadub %ptr_j1
  %ptr_j2 = add %ptr_j, 2
  %c_j2 = w loadub %ptr_j2
  %eq1 = ceql %c_j, %c_j1
  jnz %eq1, @litinclude
  %eq2 = ceql %c_j, %c_j2
  jnz %eq2, @litinclude
  jmp @litdone
  @litinclude:
  %L = add %L, 1
  %j = add %j, 1
  jmp @litloop
  @litdone:
  %needed = add %L, 1
  %room = sub %cap, %pos
  %lt_needed = csltl %room, %needed
  jnz %lt_needed, @notenough
  %header = sub %L, 1
  %ptr_h2 = add %out, %pos
  storeb %header, %ptr_h2
  %copy_i = 0
  @copyloop:
  %lt_copy = csltl %copy_i, %L
  jnz %lt_copy, @copydone
  %ptr_s = add %src, %i
  %ptr_s = add %ptr_s, %copy_i
  %ptr_d2 = add %out, %pos
  %ptr_d2 = add %ptr_d2, %copy_i
  %ptr_d2 = add %ptr_d2, 1
  %b_s = w loadub %ptr_s
  storeb %b_s, %ptr_d2
  %copy_i = add %copy_i, 1
  jmp @copyloop
  @copydone:
  %pos = add %pos, %needed
  %i = %j
  jmp @loop
  @notenough:
  %pos = -1
  ret %pos
  @done:
  ret %pos
}
