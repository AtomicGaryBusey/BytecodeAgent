

export function l $packbits_encode(l %0, l %1, l %2, l %3) {
  %i = 0
  %pos = 0
  @loop:
  %lt_n = csltl %i, %1
  jnz %lt_n, @done
  %ptr_i = add %0, %i
  %c_i = w loadub %ptr_i
  %run_i = 0
  %k = %i
  @runloop_i:
  %lt_k_n = csltl %k, %1
  jnz %lt_k_n, @rundone_i
  %lt_cnt = csltl %run_i, 128
  jnz %lt_cnt, @rundone_i
  %ptr_k = add %0, %k
  %c_k = w loadub %ptr_k
  %eq = ceql %c_k, %c_i
  jnz %eq, @rundone_i
  %run_i = add %run_i, 1
  %k = add %k, 1
  jmp @runloop_i
  @rundone_i:
  %lt_3 = csltl %run_i, 3
  jnz %lt_3, @literal
  %room = sub %3, %pos
  %lt_2 = csltl %room, 2
  jnz %lt_2, @notenough
  %header = sub 257, %run_i
  %ptr_h = add %2, %pos
  storeb %header, %ptr_h
  %ptr_d1 = add %2, %pos
  %ptr_d = add %ptr_d1, 1
  storeb %c_i, %ptr_d
  %pos = add %pos, 2
  %i = add %i, %run_i
  jmp @loop
  @literal:
  %j = %i
  %L = 0
  @litloop:
  %lt_j_n = csltl %j, %1
  jnz %lt_j_n, @litdone
  %diff = sub %j, %i
  %lt_128 = csltl %diff, 128
  jnz %lt_128, @litdone
  %j2 = add %j, 2
  %lt_j2_n = csltl %j2, %1
  jnz %lt_j2_n, @litrun3
  @litinclude:
  %L = add %L, 1
  %j = add %j, 1
  jmp @litloop
  @litrun3:
  %ptr_j = add %0, %j
  %c_j = w loadub %ptr_j
  %ptr_j1 = add %ptr_j, 1
  %c_j1 = w loadub %ptr_j1
  %ptr_j2 = add %ptr_j, 2
  %c_j2 = w loadub %ptr_j2
  %eq1 = ceql %c_j, %c_j1
  jnz %eq1, @litinclude
  %eq2 = ceql %c_j, %c_j2
  jnz %eq2, @litinclude
  jmp @litdone
  @litdone:
  %needed = add 1, %L
  %room = sub %3, %pos
  %lt_needed = csltl %room, %needed
  jnz %lt_needed, @notenough
  %header = sub %L, 1
  %ptr_h2 = add %2, %pos
  storeb %header, %ptr_h2
  %copy_i = 0
  @copyloop:
  %lt_copy = csltl %copy_i, %L
  jnz %lt_copy, @copydone
  %ptr_s = add %0, %i
  %ptr_s = add %ptr_s, %copy_i
  %ptr_d_base = add %2, %pos
  %ptr_d2 = add %ptr_d_base, %copy_i
  %ptr_d2 = add %ptr_d2, 1
  %b_s = w loadub %ptr_s
  storeb %b_s, %ptr_d2
  %copy_i = add %copy_i, 1
  jmp @copyloop
  @copydone:
  %pos = add %pos, %needed
  %i = %j
  jmp @loop
  @notenough:
  ret -1
  @done:
  ret %pos
}
