

export function l $packbits_encode(l src, l n, l out, l cap) {
  %i = 0
  %wpos = 0
  @loop:
  %lt_n = csltl %i, %n
  jnz %lt_n, @done
  %ptr_i = add %src, %i
  %c = w loadub %ptr_i
  %count = 0
  %j = %i
  @runloop:
  %lt_j_n = csltl %j, %n
  jnz %lt_j_n, @rundone
  %lt_cnt = csltl %count, 128
  jnz %lt_cnt, @rundone
  %ptr_j = add %src, %j
  %bj = w loadub %ptr_j
  %eq = ceql %bj, %c
  jnz %eq, @rundone
  %count = add %count, 1
  %j = add %j, 1
  jmp @runloop
  @rundone:
  %run_i = %count
  %lt_3 = csltl %run_i, 3
  jnz %lt_3, @literal
  %room = sub %cap, %wpos
  %lt_2 = csltl %room, 2
  jnz %lt_2, @notenough
  %header = sub 257, %run_i
  %ptr_h = add %out, %wpos
  storeb %header, %ptr_h
  %ptr_d = add %out, add %wpos, 1
  %c2 = w loadub (add %src, %i)
  storeb %c2, %ptr_d
  %wpos = add %wpos, 2
  %i = add %i, %run_i
  jmp @loop
  @literal:
  %j = %i
  %L = 0
  @litloop:
  %lt_j_n = csltl %j, %n
  jnz %lt_j_n, @litdone
  %diff = sub %j, %i
  %lt_128 = csltl %diff, 128
  jnz %lt_128, @litdone
  %ptr_j = add %src, %j
  %c_j = w loadub %ptr_j
  %cnt = 0
  %k = %j
  @runjloop:
  %lt_k_n = csltl %k, %n
  jnz %lt_k_n, @runjdone
  %lt_cnt_j = csltl %cnt, 128
  jnz %lt_cnt_j, @runjdone
  %ptr_k = add %src, %k
  %bk = w loadub %ptr_k
  %eq2 = ceql %bk, %c_j
  jnz %eq2, @runjdone
  %cnt = add %cnt, 1
  %k = add %k, 1
  jmp @runjloop
  @runjdone:
  %run_j = %cnt
  %lt_runj_3 = csltl %run_j, 3
  jnz %lt_runj_3, @litdone
  %L = add %L, 1
  %j = add %j, 1
  jmp @litloop
  @litdone:
  %needed = add 1, %L
  %room = sub %cap, %wpos
  %lt_needed = csltl %room, %needed
  jnz %lt_needed, @notenough
  %header = sub %L, 1
  %ptr_h2 = add %out, %wpos
  storeb %header, %ptr_h2
  %copy_i = 0
  @copyloop:
  %lt_copy = csltl %copy_i, %L
  jnz %lt_copy, @copydone
  %ptr_s = add (add %src, %i), %copy_i
  %ptr_d2 = add (add %out, %wpos), add 1, %copy_i
  %b_s = w loadub %ptr_s
  storeb %b_s, %ptr_d2
  %copy_i = add %copy_i, 1
  jmp @copyloop
  @copydone:
  %wpos = add %wpos, %needed
  %i = %j
  jmp @loop
  @notenough:
  ret -1
  @done:
  ret %wpos
}
