

packbits_encode:
    push rbx
    xor r8, r8
    xor r9, r9
.Lloop:
    cmp r8, rsi
    jge .Ldone
    mov r10, 1
.Lrunloop:
    lea rax, [r8 + r10]
    cmp rax, rsi
    jge .Lrunend
    movzx eax, byte ptr [rdi + r8]
    movzx ebx, byte ptr [rdi + rax]
    cmp al, bl
    je .Lruncontinue
    jmp .Lrunend
.Lruncontinue:
    inc r10
    jmp .Lrunloop
.Lrunend:
    cmp r10, 128
    jle .Lruncapok
    mov r10, 128
.Lruncapok:
    cmp r10, 3
    jl .Lliteral
    cmp rcx, 2
    jl .Lreturn_minus1
    mov r11, 128
    sub r11, r10
    add r11, 129
    mov byte ptr [rdx], r11b
    movzx eax, byte ptr [rdi + r8]
    mov byte ptr [rdx + 1], al
    add rdx, 2
    sub rcx, 2
    add r9, 2
    add r8, r10
    jmp .Lloop
.Lliteral:
    mov r11, r8
    xor r10, r10
.Llitloop:
    cmp r11, rsi
    jge .Llitend
    cmp r10, 128
    jge .Llitend
    lea rax, [r11 + 2]
    cmp rax, rsi
    jge .Llitcontinue
    movzx eax, byte ptr [rdi + r11]
    movzx ebx, byte ptr [rdi + r11 + 1]
    cmp al, bl
    je .Lcheck3lit
    jmp .Llitcontinue
.Lcheck3lit:
    movzx ecx, byte ptr [rdi + r11 + 2]
    cmp al, cl
    je .Lrun3lit
    jmp .Llitcontinue
.Lrun3lit:
    jmp .Llitend
.Llitcontinue:
    inc r10
    inc r11
    jmp .Llitloop
.Llitend:
    sub r11, r10
    mov r11, 1
    sub r11, r10
    mov byte ptr [rdx], r11b
    ; copy r10 bytes from src+i to out
