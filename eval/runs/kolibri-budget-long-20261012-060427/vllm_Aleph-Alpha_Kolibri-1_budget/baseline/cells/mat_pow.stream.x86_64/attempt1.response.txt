

mat_pow:
    cmp rsi, 1
    jne .Lother
    mov r9, qword ptr [rdi]
    mov rax, 1
    mov r10, rdx
.Lloop:
    test r10, r10
    jz .Ldone
    imul rax, r9
    dec r10
    jnz .Lloop
.Ldone:
    mov qword ptr [rcx], rax
    xor rax, rax
    mov qword ptr [r8], rax
    mov qword ptr [r8+8], rax
    xor rax, rax
    ret
.Lother:
    xor rax, rax
    mov qword ptr [r8], rax
    mov qword ptr [r8+8], rax
    mov qword ptr [rcx], 0
    xor rax, rax
    ret
