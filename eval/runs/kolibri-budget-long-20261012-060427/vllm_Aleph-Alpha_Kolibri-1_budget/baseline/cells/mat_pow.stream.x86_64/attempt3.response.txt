

mat_pow:
    cmp rsi, 1
    je .Ln1
    cmp rsi, 2
    je .Ln2
    jmp .Ldefault
.Ln1:
    test rdx, rdx
    jz .Ln1_id
    mov rax, 1
    mov r9, rdx
.Lpow1:
    test r9, r9
    jz .Lpow1_done
    imul rax, qword ptr [rdi]
    dec r9
    jnz .Lpow1
.Lpow1_done:
    mov qword ptr [rcx], rax
    xor rax, rax
    mov qword ptr [r8], rax
    mov qword ptr [r8+8], rax
    xor eax, eax
    ret
.Ln1_id:
    mov qword ptr [rcx], 1
    xor rax, rax
    mov qword ptr [r8], rax
    mov qword ptr [r8+8], rax
    xor eax, eax
    ret
.Ln2:
    test rdx, rdx
    jz .Ln2_id
    mov qword ptr [rcx], 1
    mov qword ptr [rcx+8], 0
    mov qword ptr [rcx+16], 0
    mov qword ptr [rcx+24], 1
    mov r9, rdx
.Ln2_loop:
    test r9, r9
    jz .Ln2_done
    mov rax, qword ptr [rdi]
    mov qword ptr [r8], rax
    mov rax, qword ptr [rdi+8]
    mov qword ptr [r8+8], rax
    mov rax, qword ptr [rdi+16]
    mov qword ptr [r8+16], rax
    mov rax, qword ptr [rdi+24]
    mov qword ptr [r8+24], rax
    mov rax, qword ptr [rcx]
    imul rax, qword ptr [r8]
    mov r10, qword ptr [rcx+8]
    imul r10, qword ptr [r8+16]
    add rax, r10
    mov qword ptr [r8+32], rax
    mov rax, qword ptr [rcx]
    imul rax, qword ptr [r8+8]
    mov r10, qword ptr [rcx+8]
    imul r10, qword ptr [r8+24]
    add rax, r10
    mov qword ptr [r8+40], rax
    mov rax, qword ptr [rcx+16]
    imul rax, qword ptr [r8]
    mov r10, qword ptr [rcx+24]
    imul r10, qword ptr [r8+16]
    add rax, r10
    mov qword ptr [r8+48], rax
    mov rax, qword ptr [rcx+16]
    imul rax, qword ptr [r8+8]
    mov r10, qword ptr [rcx+24]
    imul r10, qword ptr [r8+24]
    add rax, r10
    mov qword ptr [r8+56], rax
    mov rax, qword ptr [r8+32]
    mov qword ptr [rcx], rax
    mov rax, qword ptr [r8+40]
    mov qword ptr [rcx+8], rax
    mov rax, qword ptr [r8+48]
    mov qword ptr [rcx+16], rax
    mov rax, qword ptr [r8+56]
    mov qword ptr [rcx+24], rax
    dec r9
    jnz .Ln2_loop
.Ln2_done:
    xor rax, rax
    mov qword ptr [r8], rax
    mov qword ptr [r8+8], rax
    mov qword ptr [r8+16], rax
    mov qword ptr [r8+24], rax
    mov qword ptr [r8+32], rax
    mov qword ptr [r8+40], rax
    mov qword ptr [r8+48], rax
    mov qword ptr [r8+56], rax
    xor eax, eax
    ret
.Ln2_id:
    mov qword ptr [rcx], 1
    mov qword ptr [rcx+8], 0
    mov qword ptr [rcx+16], 0
    mov qword ptr [rcx+24], 1
    xor rax, rax
    mov qword ptr [r8], rax
    mov qword ptr [r8+8], rax
    mov qword ptr [r8+16], rax
    mov qword ptr [r8+24], rax
    mov qword ptr [r8+32], rax
    mov qword ptr [r8+40], rax
    mov qword ptr [r8+48], rax
    mov qword ptr [r8+56], rax
    xor eax, eax
    ret
.Ldefault:
    mov rax, rsi
    imul rax, rsi
    lea rax, [rax+rax]
    mov r9, rax
.Lzero_tmp:
    test r9, r9
    jz .Lzero_tmp_done
    xor rax, rax
    mov qword ptr [r8], rax
    add r8, 8
    dec r9
    jnz .Lzero_tmp
.Lzero_tmp_done:
    mov rax, rsi
    imul rax, rsi
    mov r9, rax
.Lzero_out:
    test r9, r9
    jz .Lzero_out_done
    xor rax, rax
    mov qword ptr [rcx], rax
    add rcx, 8
    dec r9
    jnz .Lzero_out
.Lzero_out_done:
    xor eax, eax
    ret
