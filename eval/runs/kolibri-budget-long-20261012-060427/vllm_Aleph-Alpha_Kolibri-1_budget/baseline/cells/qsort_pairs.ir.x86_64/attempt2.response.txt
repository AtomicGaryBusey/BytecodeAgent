

export function l $qsort_pairs(l %p, l %m) {
    @start
    %cmp1 = w csltl l %m, 1
    jnz @done if %cmp1
    %n1 = l subl l %m, 1
    %cmp2 = w csltl l %n1, 1
    jnz @done if %cmp2
    %p_right = l addl l %p, 2
    %r1 =l call $qsort_pairs(l %p, 1)
    %r2 =l call $qsort_pairs(l %p_right, %n1)
    retl 0
@done
    retl 0
}
