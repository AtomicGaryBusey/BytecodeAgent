

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
    @start:
    %rem_src = %n
    %rem_cap = %cap
    %src_ptr = %src
    %out_ptr = %out
    %written = 0l
    %ret = 0l
    @main_loop:
    %lt1 = csltl %rem_src 1l
    jnz @has_source
    @no_source:
    %ret = %written
    jmp @done
    @has_source:
    %rem_cap_gt0 = csltl 0l %rem_cap
    jnz @has_cap
    @no_cap:
    %ret = -1l
    jmp @done
    @has_cap:
    %hdr_w = w loadsb %src_ptr
    %src_ptr = addl %src_ptr 1l
    %rem_src = sub l %rem_src 1l
    %hdr_l = extsl %hdr_w
    %cmpNonNeg = csgeql %hdr_l 0l
    %cmpLe127 = csleql %hdr_l 127l
    %isLiteral = andw %cmpNonNeg %cmpLe127
    jnz @not_literal
    @literal_packet:
    %N = addl %hdr_l 1l
    %ltN = csltl %rem_src %N
    jnz @literal_source_error
    %i = 0l
    @literal_copy_loop:
    %copied_ltN = csltl %i %N
    jnz @literal_cap_check
    @literal_done:
    jmp @main_loop
    @literal_cap_check:
    %rem_cap_gt0 = csltl 0l %rem_cap
    jnz @literal_copy_one
    @literal_cap:
    %ret = -1l
    jmp @done
    @literal_copy_one:
    %b = w loadub %src_ptr
    storeb %b %out_ptr
    %src_ptr = addl %src_ptr 1l
    %out_ptr = addl %out_ptr 1l
    %rem_cap = sub l %rem_cap 1l
    %written = addl %written 1l
    %i = addl %i 1l
    jmp @literal_copy_loop
    @literal_source_error:
    %ret = -2l
    jmp @done
    @not_literal:
    %cmpNeg = csltl %hdr_l 0l
    %tmp2 = addl %hdr_l 127l
    %cmpGeNeg127 = csgeql %tmp2 0l
    %isRepeat = andw %cmpNeg %cmpGeNeg127
    jnz @not_repeat
    @repeat_packet:
    %lt1r = csltl %rem_src 1l
    jnz @repeat_source_error
    %val_w = w loadub %src_ptr
    %src_ptr = addl %src_ptr 1l
    %rem_src = sub l %rem_src 1l
    %count = sub l 1l %hdr_l
    %i = 0l
    @repeat_copy_loop:
    %copied_ltCount = csltl %i %count
    jnz @repeat_cap_check
    @repeat_done:
    jmp @main_loop
    @repeat_cap_check:
    %rem_cap_gt0 = csltl 0l %rem_cap
    jnz @repeat_copy_one
    @repeat_cap:
    %ret = -1l
    jmp @done
    @repeat_copy_one:
    storeb %val_w %out_ptr
    %out_ptr = addl %out_ptr 1l
    %rem_cap = sub l %rem_cap 1l
    %written = addl %written 1l
    %i = addl %i 1l
    jmp @repeat_copy_loop
    @repeat_source_error:
    %ret = -2l
    jmp @done
    @not_repeat:
    @noop:
    jmp @main_loop
    @done:
    retl %ret
}
