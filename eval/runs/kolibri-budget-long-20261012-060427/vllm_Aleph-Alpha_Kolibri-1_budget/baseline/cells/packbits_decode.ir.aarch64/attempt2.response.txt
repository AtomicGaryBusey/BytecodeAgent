

export function l $packbits_decode(l $1, l $2, l $3, l $4) {
    @start:
    %src_ptr = $1
    %out_ptr = $3
    %rem_src = $2
    %rem_cap = $4
    %written = l 0
    %ret = %written
    @main_loop:
    %lt1 = csltl %rem_src, l 1
    jnz @done
    %hdr_w = w loadub %src_ptr
    %hdr_l = extul %hdr_w
    %src_ptr = addl %src_ptr, l 1
    %rem_src = sub l %rem_src, l 1
    %isLiteral = csltl %hdr_l, l 128
    jnz @literal_body
    %isRepeat = csltl l 128, %hdr_l
    jnz @repeat_body
    @noop_body:
    jmp @main_loop
    @literal_body:
    %N = %hdr_l
    %ltN = csltl %rem_src, %N
    jnz @source_error
    %i = l 0
    @copy_loop:
    %c = csltl %i, %N
    jnz @copy_one
    jmp @copy_done
    @literal_cap:
    %ret = l -1
    jmp @done
    @copy_one:
    %c2 = csltl %rem_cap, l 1
    jnz @literal_cap
    %b = w loadub %src_ptr
    storeb %b, %out_ptr
    %src_ptr = addl %src_ptr, l 1
    %out_ptr = addl %out_ptr, l 1
    %rem_src = sub l %rem_src, l 1
    %rem_cap = sub l %rem_cap, l 1
    %written = add l %written, l 1
    %i = add l %i, l 1
    jmp @copy_loop
    @copy_done:
    %ret = %written
    jmp @main_loop
    @source_error:
    %ret = l -2
    jmp @done
    @repeat_body:
    %lt1r = csltl %rem_src, l 1
    jnz @source_error_repeat
    %val_w = w loadub %src_ptr
    %src_ptr = addl %src_ptr, l 1
    %rem_src = sub l %rem_src, l 1
    %count = sub l l 257, %hdr_l
    %i = l 0
    @repeat_loop:
    %c = csltl %i, %count
    jnz @repeat_one
    jmp @repeat_done
    @repeat_cap:
    %ret = l -1
    jmp @done
    @repeat_one:
    %c2 = csltl %rem_cap, l 1
    jnz @repeat_cap
    storeb %val_w, %out_ptr
    %out_ptr = addl %out_ptr, l 1
    %rem_cap = sub l %rem_cap, l 1
    %written = add l %written, l 1
    %i = add l %i, l 1
    jmp @repeat_loop
    @repeat_done:
    %ret = %written
    jmp @main_loop
    @source_error_repeat:
    %ret = l -2
    jmp @done
    @done:
    retl %ret
}
