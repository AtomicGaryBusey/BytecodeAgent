

export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
    @start:
    %rem_src = %n
    %rem_cap = %cap
    %src_ptr = %src
    %out_ptr = %out
    %written = 0
    %ret = 0
    @main_loop:
    %lt1 = csltl %rem_src, 1
    jnz @done
    %hdr_w = w loadub %src_ptr
    %hdr_l = extul %hdr_w
    %src_ptr = addl %src_ptr, 1
    %rem_src = sub l %rem_src, 1
    %isLiteral = csltl %hdr_l, 128
    jnz @literal_body
    %isRepeat = csltl 128, %hdr_l
    jnz @repeat_body
    @noop_body:
    jmp @main_loop
    @literal_body:
    %N = add l %hdr_l, 1
    %ltN = csltl %rem_src, %N
    jnz @source_error
    %i = 0
    @copy_loop:
    %rem_cap_gt0 = csltl 0, %rem_cap
    jnz @copy_one
    @literal_cap:
    %ret = -1
    jmp @done
    @copy_one:
    %b = w loadub %src_ptr
    storeb %b, %out_ptr
    %src_ptr = addl %src_ptr, 1
    %out_ptr = addl %out_ptr, 1
    %rem_src = sub l %rem_src, 1
    %rem_cap = sub l %rem_cap, 1
    %written = add l %written, 1
    %i = add l %i, 1
    %copied_ltN = csltl %i, %N
    jnz @copy_loop
    @copy_done:
    %ret = %written
    jmp @main_loop
    @source_error:
    %ret = -2
    jmp @done
    @repeat_body:
    %lt1r = csltl %rem_src, 1
    jnz @source_error_repeat
    %val_w = w loadub %src_ptr
    %src_ptr = addl %src_ptr, 1
    %rem_src = sub l %rem_src, 1
    %count = sub l 257, %hdr_l
    %i = 0
    @repeat_loop:
    %rem_cap_gt0 = csltl 0, %rem_cap
    jnz @repeat_one
    @repeat_cap:
    %ret = -1
    jmp @done
    @repeat_one:
    storeb %val_w, %out_ptr
    %out_ptr = addl %out_ptr, 1
    %rem_cap = sub l %rem_cap, 1
    %written = add l %written, 1
    %i = add l %i, 1
    %copied_ltCount = csltl %i, %count
    jnz @repeat_loop
    @repeat_done:
    %ret = %written
    jmp @main_loop
    @source_error_repeat:
    %ret = -2
    jmp @done
    @done:
    retl %ret
}
