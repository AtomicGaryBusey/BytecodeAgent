    ldr x4, [x3]

    cmp x9, x4
    b.lt .Lless
    b.gt .Lnotless

    subs x8, x8, #1

    subs x14, x14, #1
    add x19, x6, #16

    mov x0, x19
    mov x1, x20

.Lbasecase:

.Lend:
    ldp x21, x22, [sp], #16
