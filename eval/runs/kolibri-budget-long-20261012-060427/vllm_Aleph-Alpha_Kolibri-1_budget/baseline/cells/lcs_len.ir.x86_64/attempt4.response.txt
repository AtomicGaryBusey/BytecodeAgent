

export function l $lcs_len(%l %x, %l %m, %l %y, %l %n, %l %row) {
%zero = 0
%one = 1
%i = 0
%j = 0
@init_loop:
%init_gt_j_n = csltl %n %j
jnz @init_done
%row_j_addr = addl %row %j
storel %zero %row_j_addr
%j = addl %j %one
jmp @init_loop
@init_done:
%i = 0
@outer_loop:
%outer_lt_i_m = csltl %i %m
jnz @outer_body
jmp @done
@outer_body:
%diag = 0
%j = 1
@inner_loop:
%inner_gt_j_n = csltl %n %j
jnz @inner_exit
%j_minus_1 = subtl %j %one
%row_j_addr = addl %row %j
%row_j_minus_1_addr = addl %row %j_minus_1
%up = loadl %row_j_addr
%prev = loadl %row_j_minus_1_addr
%x_addr = addl %x %i
%x_char = loadub %x_addr
%y_addr = addl %y %j_minus_1
%y_char = loadub %y_addr
%eq = ceql %x_char %y_char
jnz @match
%inner_lt_up_prev = csltl %up %prev
jnz @use_prev
storel %up %row_j_addr
jmp @next
@use_prev:
storel %prev %row_j_addr
jmp @next
@match:
%match_val = addl %diag %one
storel %match_val %row_j_addr
jmp @next
@next:
%diag = %up
%j = addl %j %one
jmp @inner_loop
@inner_exit:
%i = addl %i %one
jmp @outer_loop
@done:
%row_n_addr = addl %row %n
%result = loadl %row_n_addr
ret %result
}
