

export function l $lcs_len(%l %x, %l %m, %l %y, %l %n, %l %row) {
%one =i 1
%zero =i 0
%j =i 0
@zero_loop:
%gt =w csltl %n %j
jnz @zero_done
%row_j_addr =i addl %row %j
storel %zero %row_j_addr
%j =i addl %j %one
jmp @zero_loop
@zero_done:
%i =i 0
jmp @outer_loop
@outer_loop:
%lt =w csltl %i %m
jnz @inner_init
jmp @done
@inner_init:
%diag =i %zero
%j =i %one
jmp @inner_loop
@inner_loop:
%gt =w csltl %n %j
jnz @inner_exit
%row_j_addr =i addl %row %j
%up =i loadl %row_j_addr
%j_minus_1 =i subtl %j %one
%row_j_minus_1_addr =i addl %row %j_minus_1
%prev =i loadl %row_j_minus_1_addr
%x_addr =i addl %x %i
%x_char =w loadub %x_addr
%y_addr =i addl %y %j_minus_1
%y_char =w loadub %y_addr
%eq =w ceql %x_char %y_char
jnz @match
@no_match:
%gt_up =w csltl %prev %up
jnz @use_up
%new_val =i %prev
jmp @store
@use_up:
%new_val =i %up
jmp @store
@match:
%new_val =i addl %diag %one
jmp @store
@store:
storel %new_val %row_j_addr
%diag =i %up
%j =i addl %j %one
jmp @inner_loop
@inner_exit:
%i =i addl %i %one
jmp @outer_loop
@done:
%row_n_addr =i addl %row %n
%result =i loadl %row_n_addr
ret %result
}
