

export function l $lcs_len(%l %x, %l %m, %l %y, %l %n, %l %row) {
%one =l 1
%zero =l 0
%j =l 0
@zero_loop:
%gt =w csltl %n %j
jnz @zero_done
%row_j_addr =l addl %row %j
storel %zero %row_j_addr
%j =l addl %j %one
jmp @zero_loop
@zero_done:
%i =l 0
jmp @outer_loop
@outer_loop:
%lt =w csltl %i %m
jnz @inner_init
jmp @done
@inner_init:
%diag =l 0
%j =l 1
jmp @inner_loop
@inner_loop:
%gt =w csltl %n %j
jnz @inner_exit
%row_j_addr =l addl %row %j
%up =l loadl %row_j_addr
%j_minus_1 =l subtl %j %one
%row_j_minus_1_addr =l addl %row %j_minus_1
%prev =l loadl %row_j_minus_1_addr
%x_addr =l addl %x %i
%x_char =w loadub %x_addr
%y_addr =l addl %y %j_minus_1
%y_char =w loadub %y_addr
%eq =w ceql %x_char %y_char
jnz @match
@no_match:
%gt_up =w csltl %prev %up
jnz @use_up
storel %prev %row_j_addr
jmp @store_end
@use_up:
storel %up %row_j_addr
jmp @store_end
@match:
%new_val =l addl %diag %one
storel %new_val %row_j_addr
@store_end:
%diag =l addl %up %zero
%j =l addl %j %one
jmp @inner_loop
@inner_exit:
%i =l addl %i %one
jmp @outer_loop
@done:
%row_n_addr =l addl %row %n
%result =l loadl %row_n_addr
ret %result
}
