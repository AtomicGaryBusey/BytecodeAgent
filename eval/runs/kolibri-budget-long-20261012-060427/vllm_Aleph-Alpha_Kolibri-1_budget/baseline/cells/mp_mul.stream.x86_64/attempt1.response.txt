

mp_mul:
 test rsi, rsi
 jz .Lzero
 test rcx, rcx
 jz .Lzero
 cmp rsi, 1
 jne .Lnot_one
 cmp rcx, 1
 jne .Lnot_one
.Lone_one:
 mov rax, qword ptr [rdi]
 mul qword ptr [rdx]
 mov qword ptr [r8], rax
 mov qword ptr [r8+8], rdx
 test rdx, rdx
 jz .Lone_one_ret1
 mov rax, 2
 jmp .Ldone
.Lone_one_ret1:
 mov rax, 1
 jmp .Ldone
.Lnot_one:
 jmp .Lzero_setup
.Lzero:
 jmp .Lzero_setup
.Lzero_setup:
 mov rax, rsi
 add rax, rcx
 mov r9, rax
 mov r10, r8
.Lzero_loop:
 xor rax, rax
 mov qword ptr [r10], rax
 add r10, 8
 dec r9
 jnz .Lzero_loop
 mov rax, 0
 jmp .Ldone
.Ldone:
 ret
