

vm_run:
xor r8, r8
xor r9, r9
test rsi, rsi
jz .return
.loop_start:
cmp r9, rsi
jge .return
lea rax, [rdi + r9*8]
mov rax, qword ptr [rax]
cmp rax, 0
je .halt
cmp rax, 1
je .push
cmp rax, 2
je .add
cmp rax, 3
je .sub
cmp rax, 4
je .mul
cmp rax, 5
je .dup
cmp rax, 6
je .swap
cmp rax, 7
je .drop
cmp rax, 8
je .over
cmp rax, 9
je .lt
cmp rax, 10
je .jz
cmp rax, 11
je .jmp
add r9, 1
jmp .check_len
.halt:
jmp .return
.push:
lea r10, [rdi + r9*8 + 8]
mov r10, qword ptr [r10]
mov qword ptr [rdx + r8*8], r10
inc r8
add r9, 2
jmp .check_len
.add:
mov rax, qword ptr [rdx + r8*8 - 16]
mov r10, qword ptr [rdx + r8*8 - 8]

mov qword ptr [rdx + r8*8 - 16], rax
dec r8
add r9, 1
jmp .check_len
.sub:
mov rax, qword ptr [rdx + r8*8 - 16]
mov r10, qword ptr [rdx + r8*8 - 8]
sub rax, r10
mov qword ptr [rdx + r8*8 - 16], rax
dec r8
add r9, 1
jmp .check_len
.mul:
mov r11, rdx
mov rax, qword ptr [rdx + r8*8 - 16]
mov r10, qword ptr [rdx + r8*8 - 8]
mul r10
mov rdx, r11
mov qword ptr [rdx + r8*8 - 16], rax
dec r8
add r9, 1
jmp .check_len
.dup:
mov rax, qword ptr [rdx + r8*8 - 8]
mov qword ptr [rdx + r8*8], rax
inc r8
add r9, 1
jmp .check_len
.swap:
mov rax, qword ptr [rdx + r8*8 - 8]
mov r10, qword ptr [rdx + r8*8 - 16]
mov qword ptr [rdx + r8*8 - 8], r10
mov qword ptr [rdx + r8*8 - 16], rax
add r9, 1
jmp .check_len
.drop:
dec r8
add r9, 1
jmp .check_len
.over:
mov rax, qword ptr [rdx + r8*8 - 16]
mov qword ptr [rdx + r8*8], rax
inc r8
add r9, 1
jmp .check_len
.lt:
mov rax, qword ptr [rdx + r8*8 - 16]
mov r10, qword ptr [rdx + r8*8 - 8]
cmp rax, r10
setl al
movzx rax, al
mov qword ptr [rdx + r8*8 - 16], rax
dec r8
add r9, 1
jmp .check_len
.jz:
mov rax, qword ptr [rdx + r8*8 - 8]
dec r8
lea r10, [rdi + r9*8 + 8]
mov r10, qword ptr [r10]
test rax, rax
jz .jz_zero
add r9, 2
jmp .check_len
.jz_zero:
mov r9, r10
jmp .check_len
.jmp:
lea r10, [rdi + r9*8 + 8]
mov r10, qword ptr [r10]
mov r9, r10
jmp .check_len
.check_len:
cmp r9, rsi
jge .return
jmp .loop_start
.return:
test r8, r8
jz .set_rax_zero
mov rax, qword ptr [rdx + r8*8 - 8]
jmp .zero_loop
.set_rax_zero:
xor rax, rax
.zero_loop:
mov r10, r8
cmp r10, rcx
jge .zero_done
mov qword ptr [rdx + r10*8], 0
inc r10
jmp .zero_loop
.zero_done:
ret
