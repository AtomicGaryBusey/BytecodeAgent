

packbits_decode:
    xor r9, r9
    test rsi, rsi
    jz .Lnormal_return
    jmp .Lloop
.Lnormal_return:
    mov rax, r9
    jmp .Ldone
.Lloop:
    test rsi, rsi
    jz .Lnormal_return
    movsx rax, byte ptr [rdi]
    inc rdi
    dec rsi
    mov r8, rax
    test r8, r8
    jns .Lliteral
    mov rax, -128
    cmp r8, rax
    je .Lnoop
    jmp .Lrepeat
.Lliteral:
    mov rax, r8
    inc rax
    cmp rsi, rax
    jl .Lliteral_src_err
    cmp rcx, rax
    jl .Lliteral_outoverflow
    mov r10, rax
    xor r11, r11
    jmp .Lliteral_write
.Lliteral_src_err:
    mov rax, -2
    jmp .Ldone
.Lliteral_outoverflow:
    mov r10, rcx
    mov r11, 1
    jmp .Lliteral_write
.Lliteral_write:
    test r10, r10
    jz .Lliteral_done
.Llit_loop:
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec rsi
    dec rcx
    inc r9
    dec r10
    jnz .Llit_loop
.Lliteral_done:
    test r11, r11
    jz .Lloop
    mov rax, -1
    jmp .Ldone
.Lrepeat:
    mov rax, 1
    sub rax, r8
    cmp rsi, 1
    jl .Lrepeat_src_err
    cmp rcx, rax
    jl .Lrepeat_outoverflow
    mov r10, rax
    xor r11, r11
    jmp .Lread_repeat
.Lrepeat_src_err:
    mov rax, -2
    jmp .Ldone
.Lrepeat_outoverflow:
    mov r10, rcx
    mov r11, 1
    jmp .Lread_repeat
.Lread_repeat:
    xor eax, eax
    mov al, byte ptr [rdi]
    inc rdi
    dec rsi
    test r10, r10
    jz .Lrepeat_done
.Lrepeat_loop:
    mov byte ptr [rdx], al
    inc rdx
    dec rcx
    inc r9
    dec r10
    jnz .Lrepeat_loop
.Lrepeat_done:
    test r11, r11
    jz .Lloop
    mov rax, -1
    jmp .Ldone
.Lnoop:
    jmp .Lloop
.Ldone:
    ret
