

export function l $qsort_pairs(l %p, l %n)
{
    @start:
        %two =l const l 2
        %cmp = w csltl l %n, l %two
        jnz %cmp, @return
        %n_minus_1 =l sub l %n, l 1
        %r =l call $qsort_pairs(l %p, l %n_minus_1)
        jmp @return
    @return:
        %ret_val =l const l 0
        ret
}
