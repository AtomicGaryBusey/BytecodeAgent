.text
.align 2
.globl _start
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    adrp    x24, hextab
    add     x24, x24, :lo12:hextab
    adrp    x25, outbuf
    add     x25, x25, :lo12:outbuf
    mov     x20, x25

    mov     x26, x19
    mov     x21, #65536
read_loop:
    cbz     x21, read_done
    mov     x8, #63
    mov     x0, #0
    mov     x1, x26
    mov     x2, x21
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x26, x26, x0
    sub     x21, x21, x0
    b       read_loop

read_done:
    sub     x21, x26, x19
    mov     x22, #0

row_loop:
    cmp     x22, x21
    b.hs    print_total

    sub     x23, x21, x22
    cmp     x23, #16
    b.ls    1f
    mov     x23, #16
1:
    mov     w0, w22
    bl      puthex8

    mov     w0, #' '
    strb    w0, [x20], #1
    strb    w0, [x20], #1

    mov     x27, #0
hex_loop:
    cmp     x27, x23
    b.hs    2f
    add     x0, x19, x22
    ldrb    w0, [x0, x27]
    lsr     w1, w0, #4
    ldrb    w1, [x24, x1]
    strb    w1, [x20], #1
    and     w1, w0, #15
    ldrb    w1, [x24, x1]
    strb    w1, [x20], #1
    mov     w1, #' '
    strb    w1, [x20], #1
    b       3f
2:
    mov     w1, #' '
    strb    w1, [x20], #1
    strb    w1, [x20], #1
    strb    w1, [x20], #1
3:
    cmp     x27, #7
    b.ne    4f
    mov     w1, #' '
    strb    w1, [x20], #1
4:
    add     x27, x27, #1
    cmp     x27, #16
    b.lo    hex_loop

    mov     w0, #' '
    strb    w0, [x20], #1
    mov     w0, #'|'
    strb    w0, [x20], #1

    mov     x27, #0
ascii_loop:
    cmp     x27, x23
    b.hs    ascii_done
    add     x0, x19, x22
    ldrb    w0, [x0, x27]
    cmp     w0, #0x20
    b.lo    5f
    cmp     w0, #0x7e
    b.ls    6f
5:
    mov     w0, #'.'
6:
    strb    w0, [x20], #1
    add     x27, x27, #1
    b       ascii_loop

ascii_done:
    mov     w0, #'|'
    strb    w0, [x20], #1
    mov     w0, #'\n'
    strb    w0, [x20], #1

    add     x22, x22, #16
    b       row_loop

print_total:
    mov     w0, w21
    bl      puthex8
    mov     w0, #'\n'
    strb    w0, [x20], #1

    sub     x2, x20, x25
    mov     x8, #64
    mov     x0, #1
    mov     x1, x25
    svc     #0

    mov     x8, #93
    mov     x0, #0
    svc     #0

puthex8:
    mov     w3, #8
7:
    lsr     w2, w0, #28
    ldrb    w2, [x24, x2]
    strb    w2, [x20], #1
    lsl     w0, w0, #4
    subs    w3, w3, #1
    b.ne    7b
    ret

.section .rodata
