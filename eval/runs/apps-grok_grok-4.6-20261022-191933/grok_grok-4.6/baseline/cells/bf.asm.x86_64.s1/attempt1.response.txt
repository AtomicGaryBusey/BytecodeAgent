.globl _start

.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rbp, [r12 + r13]
    mov r8, rbp
    mov r14, r13
    xor ecx, ecx
find_bang:
    cmp rcx, r13
    jae split_done
    cmp byte ptr [r12 + rcx], '!'
    je got_bang
    inc rcx
    jmp find_bang
got_bang:
    lea r8, [r12 + rcx + 1]
    mov r14, rcx
split_done:
    lea rbx, [rip + jumps]
    xor r13, r13
    xor edx, edx
match_loop:
    cmp r13, r14
    jae match_done
    mov al, byte ptr [r12 + r13]
    cmp al, '['
    je match_open
    cmp al, ']'
    je match_close
    inc r13
    jmp match_loop
match_open:
    push r13
    inc edx
    inc r13
    jmp match_loop
match_close:
    test edx, edx
    jz unbalanced
    pop rsi
    dec edx
    mov qword ptr [rbx + r13*8], rsi
    mov qword ptr [rbx + rsi*8], r13
    inc r13
    jmp match_loop
match_done:
    test edx, edx
    jnz unbalanced
    lea r11, [rip + cmdtab]
    lea rax, [rip + cmd_nop]
    mov rdi, r11
    mov rcx, 256
    rep stosq
    lea rax, [rip + cmd_right]
    mov qword ptr [r11 + '>' * 8], rax
    lea rax, [rip + cmd_left]
    mov qword ptr [r11 + '<' * 8], rax
    lea rax, [rip + cmd_plus]
    mov qword ptr [r11 + '+' * 8], rax
    lea rax, [rip + cmd_minus]
    mov qword ptr [r11 + '-' * 8], rax
    lea rax, [rip + cmd_dot]
    mov qword ptr [r11 + '.' * 8], rax
    lea rax, [rip + cmd_comma]
    mov qword ptr [r11 + ',' * 8], rax
    lea rax, [rip + cmd_lbrack]
    mov qword ptr [r11 + '[' * 8], rax
    lea rax, [rip + cmd_rbrack]
    mov qword ptr [r11 + ']' * 8], rax
    lea r15, [rip + tape]
    lea r9, [rip + outbuf]
    xor r13, r13
    xor r10, r10
interp_loop:
    cmp r13, r14
    jae interp_done
    movzx eax, byte ptr [r12 + r13]
    jmp qword ptr [r11 + rax*8]
cmd_nop:
    inc r13
    jmp interp_loop
cmd_right:
    inc r10
    jmp cmd_nop
cmd_left:
    dec r10
    jmp cmd_nop
cmd_plus:
    inc byte ptr [r15 + r10]
    jmp cmd_nop
cmd_minus:
    dec byte ptr [r15 + r10]
    jmp cmd_nop
cmd_dot:
    mov al, byte ptr [r15 + r10]
    mov byte ptr [r9], al
    inc r9
    jmp cmd_nop
cmd_comma:
    cmp r8, rbp
    jae comma_eof
    mov al, byte ptr [r8]
    inc r8
    mov byte ptr [r15 + r10], al
    jmp cmd_nop
comma_eof:
    mov byte ptr [r15 + r10], 0
    jmp cmd_nop
cmd_lbrack:
    cmp byte ptr [r15 + r10], 0
    jne cmd_nop
    mov r13, qword ptr [rbx + r13*8]
    jmp cmd_nop
cmd_rbrack:
    cmp byte ptr [r15 + r10], 0
    je cmd_nop
    mov r13, qword ptr [rbx + r13*8]
    jmp cmd_nop
interp_done:
    lea rsi, [rip + outbuf]
    mov rdx, r9
    sub rdx, rsi
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
do_exit:
    xor edi, edi
    mov eax, 60
    syscall
unbalanced:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.p2align 3
inbuf:
    .zero 65536
.p2align 3
jumps:
    .zero 524288
.p2align 3
cmdtab:
    .zero 2048
tape:
    .zero 30000
outbuf:
    .zero 65536
