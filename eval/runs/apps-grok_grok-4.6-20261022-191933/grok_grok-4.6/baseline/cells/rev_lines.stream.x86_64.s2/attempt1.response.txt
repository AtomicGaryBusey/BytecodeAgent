.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFSIZE, 65536

.text
_start:
    lea r12, [rip + buffer]
    xor r13, r13

read_loop:
    mov rdx, BUFSIZE
    sub rdx, r13
    jz process
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle process
    add r13, rax
    jmp read_loop

process:
    xor r14, r14

line_loop:
    cmp r14, r13
    jge done
    mov r15, r14
find_nl:
    cmp r15, r13
    jge got_line
    cmp byte ptr [r12 + r15], 0x0A
    je got_line
    inc r15
    jmp find_nl

got_line:
    mov r8, r14
    mov r9, r15
    cmp r8, r9
    jae reverse_done
    dec r9
reverse_loop:
    cmp r8, r9
    jae reverse_done
    mov al, byte ptr [r12 + r8]
    mov dl, byte ptr [r12 + r9]
    mov byte ptr [r12 + r8], dl
    mov byte ptr [r12 + r9], al
    inc r8
    dec r9
    jmp reverse_loop

reverse_done:
    mov rdx, r15
    sub rdx, r14
    jz write_nl
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [r12 + r14]
    syscall

write_nl:
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    cmp r15, r13
    jge done
    lea r14, [r15 + 1]
    jmp line_loop

done:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

.section .rodata
nl:
    .byte 0x0A

.bss
.balign 16
buffer:
    .space 65536
