.globl _start

.text
_start:
    adrp    x19, input
    add     x19, x19, :lo12:input
    adrp    x22, linebuf
    add     x22, x22, :lo12:linebuf
    adrp    x26, hexdigits
    add     x26, x26, :lo12:hexdigits

    mov     x20, #0
read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    mov     x21, #0
row_loop:
    cmp     x21, x20
    b.hs    rows_done

    sub     x24, x20, x21
    cmp     x24, #16
    b.ls    n_ok
    mov     x24, #16
n_ok:
    add     x27, x19, x21

    mov     x23, x22
    mov     w0, w21
    bl      print_hex8

    mov     w0, #0x20
    strb    w0, [x23], #1
    strb    w0, [x23], #1

    mov     x25, #0
hex_slot_loop:
    cmp     x25, #8
    b.ne    no_gap
    mov     w0, #0x20
    strb    w0, [x23], #1
no_gap:
    cmp     x25, x24
    b.hs    hex_empty
    ldrb    w0, [x27, x25]
    lsr     w1, w0, #4
    ldrb    w1, [x26, x1]
    strb    w1, [x23], #1
    and     w1, w0, #15
    ldrb    w1, [x26, x1]
    strb    w1, [x23], #1
    mov     w0, #0x20
    strb    w0, [x23], #1
    b       hex_slot_next
hex_empty:
    mov     w0, #0x20
    strb    w0, [x23], #1
    strb    w0, [x23], #1
    strb    w0, [x23], #1
hex_slot_next:
    add     x25, x25, #1
    cmp     x25, #16
    b.lo    hex_slot_loop

    mov     w0, #0x20
    strb    w0, [x23], #1
    mov     w0, #0x7c
    strb    w0, [x23], #1

    mov     x25, #0
ascii_loop:
    cmp     x25, x24
    b.hs    ascii_done
    ldrb    w0, [x27, x25]
    cmp     w0, #0x20
    b.lo    ascii_dot
    cmp     w0, #0x7e
    b.ls    ascii_store
ascii_dot:
    mov     w0, #0x2e
ascii_store:
    strb    w0, [x23], #1
    add     x25, x25, #1
    b       ascii_loop
ascii_done:
    mov     w0, #0x7c
    strb    w0, [x23], #1
    mov     w0, #0x0a
    strb    w0, [x23], #1

    mov     x0, #1
    mov     x1, x22
    sub     x2, x23, x22
    mov     x8, #64
    svc     #0

    add     x21, x21, x24
    b       row_loop

rows_done:
    mov     x23, x22
    mov     w0, w20
    bl      print_hex8
    mov     w0, #0x0a
    strb    w0, [x23], #1
    mov     x0, #1
    mov     x1, x22
    sub     x2, x23, x22
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0

print_hex8:
    mov     w1, #32
print_hex8_loop:
    sub     w1, w1, #4
    lsr     w2, w0, w1
    and     w2, w2, #15
    ldrb    w2, [x26, x2]
    strb    w2, [x23], #1
    cbnz    w1, print_hex8_loop
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
input:
    .zero 65536
linebuf:
    .zero 96
