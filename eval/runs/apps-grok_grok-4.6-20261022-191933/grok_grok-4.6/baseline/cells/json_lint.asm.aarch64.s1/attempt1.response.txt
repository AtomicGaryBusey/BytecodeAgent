.globl _start

.text
_start:
    adr x19, input_buf
    mov x20, x19
    mov x0, #65536
    add x21, x19, x0
read_loop:
    sub x2, x21, x20
    cbz x2, read_done
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, x20
    mov x20, x19
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne error_exit
    mov x0, #1
    adr x1, msg_ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

skip_ws:
    cmp x20, x21
    b.hs skip_ws_done
    ldrb w0, [x20]
    cmp w0, #0x20
    b.eq skip_ws_eat
    cmp w0, #0x09
    b.eq skip_ws_eat
    cmp w0, #0x0A
    b.eq skip_ws_eat
    cmp w0, #0x0D
    b.eq skip_ws_eat
skip_ws_done:
    ret
skip_ws_eat:
    add x20, x20, #1
    b skip_ws

parse_value:
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x7B
    b.eq parse_object
    cmp w0, #0x5B
    b.eq parse_array
    cmp w0, #0x22
    b.eq parse_string
    cmp w0, #0x2D
    b.eq parse_number
    cmp w0, #'0'
    b.lo pv_lit
    cmp w0, #'9'
    b.ls parse_number
pv_lit:
    cmp w0, #'t'
    b.eq parse_true
    cmp w0, #'f'
    b.eq parse_false
    cmp w0, #'n'
    b.eq parse_null
    b error_exit

parse_object:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x7D
    b.eq obj_end
obj_member:
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x3A
    b.ne error_exit
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x7D
    b.eq obj_end
    cmp w0, #0x2C
    b.ne error_exit
    add x20, x20, #1
    bl skip_ws
    b obj_member
obj_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x5D
    b.eq arr_end
arr_elem:
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x5D
    b.eq arr_end
    cmp w0, #0x2C
    b.ne error_exit
    add x20, x20, #1
    bl skip_ws
    b arr_elem
arr_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

parse_string:
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x22
    b.ne error_exit
    add x20, x20, #1
str_loop:
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq str_end
    cmp w0, #0x5C
    b.eq str_esc
    cmp w0, #0x20
    b.lo error_exit
    add x20, x20, #1
    b str_loop
str_end:
    add x20, x20, #1
    ret
str_esc:
    add x20, x20, #1
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq esc_ok
    cmp w0, #0x5C
    b.eq esc_ok
    cmp w0, #'/'
    b.eq esc_ok
    cmp w0, #'b'
    b.eq esc_ok
    cmp w0, #'f'
    b.eq esc_ok
    cmp w0, #'n'
    b.eq esc_ok
    cmp w0, #'r'
    b.eq esc_ok
    cmp w0, #'t'
    b.eq esc_ok
    cmp w0, #'u'
    b.eq esc_u
    b error_exit
esc_ok:
    add x20, x20, #1
    b str_loop
esc_u:
    add x20, x20, #1
    mov w2, #4
uni_loop:
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo error_exit
    cmp w0, #'9'
    b.ls uni_ok
    cmp w0, #'A'
    b.lo error_exit
    cmp w0, #'F'
    b.ls uni_ok
    cmp w0, #'a'
    b.lo error_exit
    cmp w0, #'f'
    b.hi error_exit
uni_ok:
    add x20, x20, #1
    subs w2, w2, #1
    b.ne uni_loop
    b str_loop

parse_number:
    ldrb w0, [x20]
    cmp w0, #0x2D
    b.ne pn_int
    add x20, x20, #1
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
pn_int:
    cmp w0, #'0'
    b.eq pn_zero
    cmp w0, #'1'
    b.lo error_exit
    cmp w0, #'9'
    b.hi error_exit
    add x20, x20, #1
pn_more_int:
    cmp x20, x21
    b.hs pn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo pn_frac
    cmp w0, #'9'
    b.hi pn_frac
    add x20, x20, #1
    b pn_more_int
pn_zero:
    add x20, x20, #1
pn_frac:
    cmp x20, x21
    b.hs pn_done
    ldrb w0, [x20]
    cmp w0, #0x2E
    b.ne pn_exp
    add x20, x20, #1
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo error_exit
    cmp w0, #'9'
    b.hi error_exit
    add x20, x20, #1
pn_frac_more:
    cmp x20, x21
    b.hs pn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo pn_exp
    cmp w0, #'9'
    b.hi pn_exp
    add x20, x20, #1
    b pn_frac_more
pn_exp:
    cmp x20, x21
    b.hs pn_done
    ldrb w0, [x20]
    cmp w0, #'e'
    b.eq pn_exp_go
    cmp w0, #'E'
    b.ne pn_done
pn_exp_go:
    add x20, x20, #1
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    cmp w0, #0x2B
    b.eq pn_exp_sign
    cmp w0, #0x2D
    b.eq pn_exp_sign
    b pn_exp_need
pn_exp_sign:
    add x20, x20, #1
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
pn_exp_need:
    cmp w0, #'0'
    b.lo error_exit
    cmp w0, #'9'
    b.hi error_exit
    add x20, x20, #1
pn_exp_more:
    cmp x20, x21
    b.hs pn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo pn_done
    cmp w0, #'9'
    b.hi pn_done
    add x20, x20, #1
    b pn_exp_more
pn_done:
    ret

parse_true:
    adr x1, lit_true
    mov w2, #4
    b match_lit
parse_false:
    adr x1, lit_false
    mov w2, #5
    b match_lit
parse_null:
    adr x1, lit_null
    mov w2, #4
match_lit:
    cmp x20, x21
    b.hs error_exit
    ldrb w0, [x20]
    ldrb w3, [x1], #1
    cmp w0, w3
    b.ne error_exit
    add x20, x20, #1
    subs w2, w2, #1
    b.ne match_lit
    ret

error_exit:
    cmp x20, x21
    csel x20, x21, x20, hi
    sub x0, x20, x19
    adr x1, decbuf
    add x1, x1, #16
    mov x2, x1
    mov x3, #10
itoa_loop:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #'0'
    sub x1, x1, #1
    strb w5, [x1]
    mov x0, x4
    cbnz x0, itoa_loop
    mov x4, x1
    sub x5, x2, x1
    mov x0, #1
    adr x1, msg_error
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #1
    mov x1, x4
    mov x2, x5
    mov x8, #64
    svc #0
    mov x0, #1
    adr x1, msg_nl
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

.section .rodata
