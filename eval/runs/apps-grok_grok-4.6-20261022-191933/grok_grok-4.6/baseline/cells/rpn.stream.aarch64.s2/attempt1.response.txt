.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAX_INPUT, 65536

.text
.balign 4
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    mov x21, MAX_INPUT
read_loop:
    cmp x20, x21
    b.ge read_done
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    add x20, x19, x20
process_lines:
    cmp x19, x20
    b.ge all_done
    mov x21, x19
find_nl:
    cmp x19, x20
    b.ge line_no_nl
    ldrb w0, [x19]
    cmp w0, #10
    b.eq line_got_nl
    add x19, x19, #1
    b find_nl
line_got_nl:
    mov x0, x21
    mov x1, x19
    add x19, x19, #1
    bl process_line
    b process_lines
line_no_nl:
    mov x0, x21
    mov x1, x19
    bl process_line
all_done:
    mov x0, #0
    mov x8, SYS_exit
    svc #0

.balign 4
process_line:
    stp x29, x30, [sp, #-16]!
    mov x21, x0
    mov x22, x1
    adrp x23, rpn_stack
    add x23, x23, :lo12:rpn_stack
    mov x24, x23
token_loop:
skip_ws:
    cmp x21, x22
    b.ge tokens_done
    ldrb w0, [x21]
    cmp w0, #32
    b.eq ws_one
    cmp w0, #9
    b.ne have_token
ws_one:
    add x21, x21, #1
    b skip_ws
have_token:
    mov x25, x21
find_tok_end:
    cmp x21, x22
    b.ge tok_end_found
    ldrb w0, [x21]
    cmp w0, #32
    b.eq tok_end_found
    cmp w0, #9
    b.eq tok_end_found
    add x21, x21, #1
    b find_tok_end
tok_end_found:
    sub x26, x21, x25
    cmp x26, #1
    b.ne try_number
    ldrb w0, [x25]
    cmp w0, #'+'
    b.eq do_op
    cmp w0, #'-'
    b.eq do_op
    cmp w0, #'*'
    b.eq do_op
    cmp w0, #'/'
    b.eq do_op
    cmp w0, #'%'
    b.eq do_op
    b try_number
do_op:
    sub x0, x24, x23
    cmp x0, #16
    b.lt line_error
    ldr x2, [x24, #-16]
    ldr x1, [x24, #-8]
    sub x24, x24, #16
    ldrb w0, [x25]
    cmp w0, #'+'
    b.eq op_add
    cmp w0, #'-'
    b.eq op_sub
    cmp w0, #'*'
    b.eq op_mul
    cmp w0, #'/'
    b.eq op_div
op_mod:
    cbz x1, line_error
    mov x3, #1
    lsl x3, x3, #63
    cmp x2, x3
    b.ne mod_do
    cmp x1, #-1
    b.eq line_error
mod_do:
    sdiv x0, x2, x1
    msub x0, x0, x1, x2
    b op_push
op_add:
    add x0, x2, x1
    b op_push
op_sub:
    sub x0, x2, x1
    b op_push
op_mul:
    mul x0, x2, x1
    b op_push
op_div:
    cbz x1, line_error
    mov x3, #1
    lsl x3, x3, #63
    cmp x2, x3
    b.ne div_do
    cmp x1, #-1
    b.eq line_error
div_do:
    sdiv x0, x2, x1
op_push:
    str x0, [x24]
    add x24, x24, #8
    b token_loop
try_number:
    mov x27, x25
    mov x28, #0
    ldrb w0, [x27]
    cmp w0, #'-'
    b.ne num_digits
    mov x28, #1
    add x27, x27, #1
num_digits:
    sub x0, x21, x27
    cmp x0, #1
    b.lt line_error
    cmp x0, #18
    b.gt line_error
    mov x0, #0
    mov x2, #10
num_loop:
    cmp x27, x21
    b.ge num_done
    ldrb w1, [x27]
    sub w1, w1, #'0'
    cmp w1, #9
    b.hi line_error
    madd x0, x0, x2, x1
    add x27, x27, #1
    b num_loop
num_done:
    cmp x28, #0
    b.eq num_push
    neg x0, x0
num_push:
    str x0, [x24]
    add x24, x24, #8
    b token_loop
tokens_done:
    sub x0, x24, x23
    cmp x0, #8
    b.ne line_error
    ldr x0, [x23]
    bl print_num
    ldp x29, x30, [sp], #16
    ret
line_error:
    bl print_error
    ldp x29, x30, [sp], #16
    ret

.balign 4
print_num:
    stp x29, x30, [sp, #-64]!
    add x2, sp, #63
    mov w3, #10
    strb w3, [x2]
    mov x3, x0
    cmp x0, #0
    b.ge 1f
    neg x3, x3
1:
    mov x4, #10
2:
    udiv x5, x3, x4
    msub x6, x5, x4, x3
    add w6, w6, #'0'
    sub x2, x2, #1
    strb w6, [x2]
    mov x3, x5
    cbnz x3, 2b
    cmp x0, #0
    b.ge 3f
    sub x2, x2, #1
    mov w6, #'-'
    strb w6, [x2]
3:
    add x3, sp, #64
    mov x0, #1
    mov x1, x2
    sub x2, x3, x2
    mov x8, SYS_write
    svc #0
    ldp x29, x30, [sp], #64
    ret

.balign 4
print_error:
    mov x0, #1
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x2, #6
    mov x8, SYS_write
    svc #0
    ret

.section .rodata
error_msg:
    .ascii "error\n"

.bss
.balign 16
input_buf:
    .space 65536
.balign 8
rpn_stack:
    .space 8192
