.globl _start
.text
_start:
	lea r12, [rip + input_buf]
	xor r13, r13
read_loop:
	mov edx, 65536
	sub rdx, r13
	jz read_done
	xor eax, eax
	xor edi, edi
	lea rsi, [r12 + r13]
	syscall
	test rax, rax
	jle read_done
	add r13, rax
	jmp read_loop
read_done:
	xor ecx, ecx
find_bang:
	cmp rcx, r13
	jae no_bang
	cmp byte ptr [r12 + rcx], '!'
	je found_bang
	inc rcx
	jmp find_bang
found_bang:
	mov rbx, rcx
	lea rbp, [r12 + rcx + 1]
	lea r8, [r12 + r13]
	jmp check_brackets
no_bang:
	mov rbx, r13
	lea rbp, [r12 + r13]
	mov r8, rbp
check_brackets:
	lea r15, [rip + jump_table]
	lea rsi, [rip + bracket_stack]
	xor edx, edx
	xor ecx, ecx
bracket_loop:
	cmp rcx, rbx
	jae bracket_end
	mov al, byte ptr [r12 + rcx]
	cmp al, '['
	je open_br
	cmp al, ']'
	je close_br
	inc rcx
	jmp bracket_loop
open_br:
	cmp rdx, 1000
	jae unbalanced
	mov qword ptr [rsi + rdx*8], rcx
	inc rdx
	inc rcx
	jmp bracket_loop
close_br:
	test rdx, rdx
	jz unbalanced
	dec rdx
	mov rdi, qword ptr [rsi + rdx*8]
	mov qword ptr [r15 + rcx*8], rdi
	mov qword ptr [r15 + rdi*8], rcx
	inc rcx
	jmp bracket_loop
bracket_end:
	test rdx, rdx
	jnz unbalanced
	lea r10, [rip + handler_table]
	lea r14, [rip + tape]
	xor r13, r13
	jmp dispatch
h_gt:
	inc r14
	jmp next
h_lt:
	dec r14
	jmp next
h_plus:
	inc byte ptr [r14]
	jmp next
h_minus:
	dec byte ptr [r14]
	jmp next
h_dot:
	mov eax, 1
	mov edi, 1
	mov rsi, r14
	mov edx, 1
	syscall
	jmp next
h_comma:
	cmp rbp, r8
	jae comma_eof
	mov al, byte ptr [rbp]
	mov byte ptr [r14], al
	inc rbp
	jmp next
comma_eof:
	mov byte ptr [r14], 0
	jmp next
h_lbrack:
	cmp byte ptr [r14], 0
	jne next
	mov r13, qword ptr [r15 + r13*8]
	jmp next
h_rbrack:
	cmp byte ptr [r14], 0
	je next
	mov r13, qword ptr [r15 + r13*8]
h_other:
next:
	inc r13
dispatch:
	cmp r13, rbx
	jae finished
	movzx eax, byte ptr [r12 + r13]
	jmp qword ptr [r10 + rax*8]
finished:
	mov eax, 60
	xor edi, edi
	syscall
unbalanced:
	mov eax, 60
	mov edi, 1
	syscall
