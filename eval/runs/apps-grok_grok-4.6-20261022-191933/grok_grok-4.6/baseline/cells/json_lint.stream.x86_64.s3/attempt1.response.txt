.globl _start
.text
_start:
    lea r12, [rip + input_buf]
    mov rbx, r12
    lea r13, [r12 + 65536]
read_loop:
    mov rdx, r13
    sub rdx, rbx
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov r13, rbx
    mov rbx, r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    je do_ok
    jmp do_error

skip_ws:
    cmp rbx, r13
    jae skip_ws_done
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    je skip_ws_adv
    cmp al, 0x09
    je skip_ws_adv
    cmp al, 0x0A
    je skip_ws_adv
    cmp al, 0x0D
    je skip_ws_adv
skip_ws_done:
    ret
skip_ws_adv:
    inc rbx
    jmp skip_ws

parse_value:
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb do_error
    cmp al, '9'
    jbe parse_number
    jmp do_error

parse_object:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae do_error
    cmp byte ptr [rbx], '}'
    je obj_end
obj_member:
    cmp rbx, r13
    jae do_error
    cmp byte ptr [rbx], '"'
    jne do_error
    call parse_string
    call skip_ws
    cmp rbx, r13
    jae do_error
    cmp byte ptr [rbx], ':'
    jne do_error
    inc rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je obj_comma
    cmp al, '}'
    je obj_end
    jmp do_error
obj_comma:
    inc rbx
    call skip_ws
    jmp obj_member
obj_end:
    inc rbx
    ret

parse_array:
    inc rbx
    call skip_ws
    cmp rbx, r13
    jae do_error
    cmp byte ptr [rbx], ']'
    je arr_end
arr_value:
    call parse_value
    call skip_ws
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, ','
    je arr_comma
    cmp al, ']'
    je arr_end
    jmp do_error
arr_comma:
    inc rbx
    call skip_ws
    jmp arr_value
arr_end:
    inc rbx
    ret

parse_string:
    inc rbx
str_loop:
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, 0x20
    jb do_error
    cmp al, '"'
    je str_end
    cmp al, 0x5C
    je str_esc
    inc rbx
    jmp str_loop
str_end:
    inc rbx
    ret
str_esc:
    inc rbx
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, '"'
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    je str_unicode
    jmp do_error
str_esc_ok:
    inc rbx
    jmp str_loop
str_unicode:
    inc rbx
    mov ecx, 4
str_hex:
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb do_error
    cmp al, '9'
    jbe str_hex_ok
    or al, 0x20
    cmp al, 'a'
    jb do_error
    cmp al, 'f'
    ja do_error
str_hex_ok:
    inc rbx
    dec ecx
    jnz str_hex
    jmp str_loop

parse_number:
    movzx eax, byte ptr [rbx]
    cmp al, '-'
    jne num_int
    inc rbx
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
num_int:
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb do_error
    cmp al, '9'
    ja do_error
num_int_digits:
    inc rbx
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_after_int
    cmp al, '9'
    jbe num_int_digits
    jmp num_after_int
num_zero:
    inc rbx
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_after_int
    cmp al, '9'
    jbe do_error
num_after_int:
    cmp al, '.'
    je num_frac
num_check_exp:
    cmp al, 'e'
    je num_exp
    cmp al, 'E'
    je num_exp
num_done:
    ret
num_frac:
    inc rbx
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb do_error
    cmp al, '9'
    ja do_error
num_frac_digits:
    inc rbx
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_check_exp
    cmp al, '9'
    jbe num_frac_digits
    jmp num_check_exp
num_exp:
    inc rbx
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je num_exp_sign
    cmp al, '-'
    je num_exp_sign
num_exp_need_digit:
    cmp al, '0'
    jb do_error
    cmp al, '9'
    ja do_error
num_exp_digits:
    inc rbx
    cmp rbx, r13
    jae num_done
    movzx eax, byte ptr [rbx]
    cmp al, '0'
    jb num_done
    cmp al, '9'
    jbe num_exp_digits
    jmp num_done
num_exp_sign:
    inc rbx
    cmp rbx, r13
    jae do_error
    movzx eax, byte ptr [rbx]
    jmp num_exp_need_digit

parse_true:
    inc rbx
    lea rsi, [rip + str_rue]
    jmp match_rest
parse_false:
    inc rbx
    lea rsi, [rip + str_alse]
    jmp match_rest
parse_null:
    inc rbx
    lea rsi, [rip + str_ull]
    jmp match_rest

match_rest:
    movzx eax, byte ptr [rsi]
    test al, al
    jz match_done
    cmp rbx, r13
    jae do_error
    cmp al, byte ptr [rbx]
    jne do_error
    inc rbx
    inc rsi
    jmp match_rest
match_done:
    ret

do_ok:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

do_error:
    mov rax, rbx
    sub rax, r12
    lea rdi, [rip + out_buf]
    lea rsi, [rip + msg_err]
    mov ecx, 9
    rep movsb
    lea rsi, [rip + tmp_buf + 16]
    mov r8d, 10
itoa_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz itoa_loop
    lea rcx, [rip + tmp_buf + 16]
    sub rcx, rsi
    rep movsb
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + out_buf]
    sub rdi, rsi
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
str_rue:
    .asciz "rue"
str_alse:
    .asciz "alse"
str_ull:
    .asciz "ull"

.bss
.align 16
input_buf:
    .space 65536
out_buf:
    .space 32
tmp_buf:
    .space 32
