.globl _start

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x21, x19
    mov x23, #65536
    add x23, x19, x23

read_loop:
    cmp x21, x23
    b.hs read_done
    mov x0, #0
    mov x1, x21
    sub x2, x23, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x20, x21
    mov x21, x19

line_loop:
    cmp x21, x20
    b.hs all_done
    mov x22, x21

find_nl:
    cmp x22, x20
    b.hs no_nl
    ldrb w0, [x22]
    cmp w0, #10
    b.eq got_nl
    add x22, x22, #1
    b find_nl

got_nl:
    bl reverse
    mov x0, #1
    mov x1, x21
    sub x2, x22, x21
    add x2, x2, #1
    mov x8, #64
    svc #0
    add x21, x22, #1
    b line_loop

no_nl:
    bl reverse
    mov x0, #1
    mov x1, x21
    sub x2, x20, x21
    mov x8, #64
    svc #0
    adrp x1, nlbyte
    add x1, x1, :lo12:nlbyte
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0

all_done:
    mov x0, #0
    mov x8, #93
    svc #0

// reverse bytes in [x21, x22)
reverse:
    mov x0, x21
    mov x1, x22
    cmp x0, x1
    b.hs 2f
    sub x1, x1, #1
1:
    cmp x0, x1
    b.hs 2f
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w3, [x0]
    strb w2, [x1]
    add x0, x0, #1
    sub x1, x1, #1
    b 1b
2:
    ret

.section .rodata
nlbyte:
    .byte 10

.bss
.balign 16
buf:
    .space 65536
