.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536
.equ JT_SIZE, 262144
.equ MAX_NEST, 1000
.equ BSTACK_SIZE, 8000

.text
.balign 4
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf

    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    mov x0, #0
find_bang:
    cmp x0, x20
    b.hs no_bang
    ldrb w1, [x19, x0]
    cmp w1, #'!'
    b.eq found_bang
    add x0, x0, #1
    b find_bang
found_bang:
    add x23, x19, x0
    add x23, x23, #1
    add x24, x19, x20
    mov x20, x0
    b do_compact
no_bang:
    add x23, x19, x20
    mov x24, x23

do_compact:
    adrp x22, jumptable
    add x22, x22, :lo12:jumptable
    adrp x25, bstack
    add x25, x25, :lo12:bstack

    mov x0, #0
    mov x1, #0
    mov x2, #0
compact_loop:
    cmp x0, x20
    b.hs compact_done
    ldrb w3, [x19, x0]
    add x0, x0, #1
    cmp w3, #'>'
    b.eq copy_cmd
    cmp w3, #'<'
    b.eq copy_cmd
    cmp w3, #'+'
    b.eq copy_cmd
    cmp w3, #'-'
    b.eq copy_cmd
    cmp w3, #'.'
    b.eq copy_cmd
    cmp w3, #','
    b.eq copy_cmd
    cmp w3, #'['
    b.eq compact_open
    cmp w3, #']'
    b.eq compact_close
    b compact_loop
copy_cmd:
    strb w3, [x19, x1]
    add x1, x1, #1
    b compact_loop
compact_open:
    cmp x2, #MAX_NEST
    b.hs unbalanced
    str x1, [x25, x2, lsl #3]
    add x2, x2, #1
    strb w3, [x19, x1]
    add x1, x1, #1
    b compact_loop
compact_close:
    cbz x2, unbalanced
    sub x2, x2, #1
    ldr x5, [x25, x2, lsl #3]
    str w1, [x22, x5, lsl #2]
    str w5, [x22, x1, lsl #2]
    strb w3, [x19, x1]
    add x1, x1, #1
    b compact_loop
compact_done:
    cbnz x2, unbalanced
    mov x20, x1

    adrp x21, tape
    add x21, x21, :lo12:tape
    mov x26, #0

interp_loop:
    cmp x26, x20
    b.hs interp_done
    ldrb w0, [x19, x26]
    cmp w0, #'>'
    b.eq cmd_gt
    cmp w0, #'<'
    b.eq cmd_lt
    cmp w0, #'+'
    b.eq cmd_plus
    cmp w0, #'-'
    b.eq cmd_minus
    cmp w0, #'.'
    b.eq cmd_dot
    cmp w0, #','
    b.eq cmd_comma
    cmp w0, #'['
    b.eq cmd_lb
    cmp w0, #']'
    b.eq cmd_rb
next_cmd:
    add x26, x26, #1
    b interp_loop

cmd_gt:
    add x21, x21, #1
    b next_cmd
cmd_lt:
    sub x21, x21, #1
    b next_cmd
cmd_plus:
    ldrb w0, [x21]
    add w0, w0, #1
    strb w0, [x21]
    b next_cmd
cmd_minus:
    ldrb w0, [x21]
    sub w0, w0, #1
    strb w0, [x21]
    b next_cmd
cmd_dot:
    mov x0, #1
    mov x1, x21
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    b next_cmd
cmd_comma:
    cmp x23, x24
    b.hs comma_eof
    ldrb w1, [x23], #1
    strb w1, [x21]
    b next_cmd
comma_eof:
    strb wzr, [x21]
    b next_cmd
cmd_lb:
    ldrb w0, [x21]
    cbnz w0, next_cmd
    ldr w26, [x22, x26, lsl #2]
    b next_cmd
cmd_rb:
    ldrb w0, [x21]
    cbz w0, next_cmd
    ldr w26, [x22, x26, lsl #2]
    b next_cmd

interp_done:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

unbalanced:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0

.bss
.balign 16
buf:
    .space BUF_SIZE
tape:
    .space 30000
.balign 4
jumptable:
    .space JT_SIZE
.balign 8
bstack:
    .space BSTACK_SIZE
