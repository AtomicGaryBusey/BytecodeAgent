.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAX_INPUT, 32768

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x22, line_ptrs
    add x22, x22, :lo12:line_ptrs
    adrp x23, line_lens
    add x23, x23, :lo12:line_lens

    mov x20, #0
read_loop:
    mov x2, #MAX_INPUT
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov w0, #10
    strb w0, [x19, x20]

    mov x21, #0
    mov x0, #0
parse_loop:
    cmp x0, x20
    b.hs parse_done
    add x1, x19, x0
    mov x2, x0
parse_scan:
    cmp x0, x20
    b.hs parse_eof
    ldrb w3, [x19, x0]
    add x0, x0, #1
    cmp w3, #10
    b.ne parse_scan
    sub x3, x0, x2
    sub x3, x3, #1
    b parse_store
parse_eof:
    sub x3, x0, x2
parse_store:
    str x1, [x22, x21, lsl #3]
    str w3, [x23, x21, lsl #2]
    add x21, x21, #1
    b parse_loop
parse_done:

    mov x9, #1
sort_outer:
    cmp x9, x21
    b.hs sort_done
    ldr x10, [x22, x9, lsl #3]
    ldr w11, [x23, x9, lsl #2]
    mov x12, x9
sort_inner:
    cbz x12, sort_insert
    sub x13, x12, #1
    ldr x14, [x22, x13, lsl #3]
    ldr w15, [x23, x13, lsl #2]
    mov x0, x14
    mov w1, w15
    mov x2, x10
    mov w3, w11
    bl compare_lines
    cmp w0, #0
    b.le sort_insert
    str x14, [x22, x12, lsl #3]
    str w15, [x23, x12, lsl #2]
    mov x12, x13
    b sort_inner
sort_insert:
    str x10, [x22, x12, lsl #3]
    str w11, [x23, x12, lsl #2]
    add x9, x9, #1
    b sort_outer
sort_done:

    mov x9, #0
print_loop:
    cmp x9, x21
    b.hs do_exit
    ldr x1, [x22, x9, lsl #3]
    ldr w2, [x23, x9, lsl #2]
    add x2, x2, #1
    mov x0, #1
    mov x8, #SYS_write
    svc #0
    add x9, x9, #1
    b print_loop
do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

compare_lines:
    cmp w1, w3
    csel w4, w1, w3, lo
    mov x5, #0
1:
    cmp x5, x4
    b.hs 2f
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.ne 3f
    add x5, x5, #1
    b 1b
3:
    sub w0, w6, w7
    ret
2:
    sub w0, w1, w3
    ret

.bss
.balign 8
input_buf:
    .space 32769
.balign 8
line_ptrs:
    .space 8000
.balign 4
line_lens:
    .space 4000
