.text
.globl _start
_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, #0
    movz    x21, #1, lsl #16

read_loop:
    sub     x2, x21, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    add     x21, x19, x20

line_loop:
    cmp     x19, x21
    b.hs    all_done
    mov     x20, x19
find_nl:
    cmp     x20, x21
    b.hs    found_end
    ldrb    w0, [x20]
    cmp     w0, #10
    b.eq    found_end
    add     x20, x20, #1
    b       find_nl

found_end:
    adrp    x23, rpn_stack
    add     x23, x23, :lo12:rpn_stack
    mov     x22, x23
    mov     x24, x19

tok_loop:
skip_ws:
    cmp     x24, x20
    b.hs    tokens_done
    ldrb    w0, [x24]
    cmp     w0, #32
    b.eq    skip_one
    cmp     w0, #9
    b.eq    skip_one
    b       have_token
skip_one:
    add     x24, x24, #1
    b       skip_ws

have_token:
    mov     x25, x24
find_tok_end:
    cmp     x25, x20
    b.hs    got_tok_end
    ldrb    w0, [x25]
    cmp     w0, #32
    b.eq    got_tok_end
    cmp     w0, #9
    b.eq    got_tok_end
    add     x25, x25, #1
    b       find_tok_end

got_tok_end:
    sub     x3, x25, x24
    cmp     x3, #1
    b.ne    try_number
    ldrb    w0, [x24]
    cmp     w0, #'+'
    b.eq    do_op
    cmp     w0, #'-'
    b.eq    do_op
    cmp     w0, #'*'
    b.eq    do_op
    cmp     w0, #'/'
    b.eq    do_op
    cmp     w0, #'%'
    b.eq    do_op

try_number:
    ldrb    w0, [x24]
    cmp     w0, #'-'
    b.ne    num_pos
    add     x26, x24, #1
    mov     x27, #1
    b       num_count
num_pos:
    mov     x26, x24
    mov     x27, #0
num_count:
    sub     x3, x25, x26
    cbz     x3, line_error
    cmp     x3, #18
    b.hi    line_error
    mov     x28, #0
    mov     x9, #10
num_digits:
    cmp     x26, x25
    b.hs    num_finish
    ldrb    w6, [x26], #1
    sub     w6, w6, #'0'
    cmp     w6, #9
    b.hi    line_error
    madd    x28, x28, x9, x6
    b       num_digits
num_finish:
    cbz     x27, num_push
    neg     x28, x28
num_push:
    str     x28, [x22], #8
    mov     x24, x25
    b       tok_loop

do_op:
    sub     x3, x22, x23
    cmp     x3, #16
    b.lo    line_error
    ldr     x5, [x22, #-8]!
    ldr     x4, [x22, #-8]!
    cmp     w0, #'+'
    b.eq    op_add
    cmp     w0, #'-'
    b.eq    op_sub
    cmp     w0, #'*'
    b.eq    op_mul
    cmp     w0, #'/'
    b.eq    op_div
    cbz     x5, line_error
    movz    x3, #0x8000, lsl #48
    cmp     x4, x3
    b.ne    1f
    cmn     x5, #1
    b.eq    line_error
1:
    sdiv    x3, x4, x5
    msub    x4, x3, x5, x4
    b       op_push
op_div:
    cbz     x5, line_error
    movz    x3, #0x8000, lsl #48
    cmp     x4, x3
    b.ne    2f
    cmn     x5, #1
    b.eq    line_error
2:
    sdiv    x4, x4, x5
    b       op_push
op_add:
    add     x4, x4, x5
    b       op_push
op_sub:
    sub     x4, x4, x5
    b       op_push
op_mul:
    mul     x4, x4, x5
op_push:
    str     x4, [x22], #8
    mov     x24, x25
    b       tok_loop

tokens_done:
    sub     x3, x22, x23
    cmp     x3, #8
    b.ne    line_error
    ldr     x0, [x23]
    bl      print_num
    b       next_line

line_error:
    bl      print_error

next_line:
    cmp     x20, x21
    b.hs    all_done
    add     x19, x20, #1
    b       line_loop

all_done:
    mov     x0, #0
    mov     x8, #93
    svc     #0

print_error:
    adrp    x1, error_msg
    add     x1, x1, :lo12:error_msg
    mov     x2, #6
    mov     x0, #1
    mov     x8, #64
    svc     #0
    ret

print_num:
    adrp    x7, num_buf
    add     x7, x7, :lo12:num_buf
    add     x1, x7, #31
    mov     w2, #10
    strb    w2, [x1]
    mov     x3, #0
    cmp     x0, #0
    b.eq    pn_zero
    b.gt    pn_setup
    mov     x3, #1
    neg     x0, x0
pn_setup:
    mov     x4, #10
pn_loop:
    udiv    x5, x0, x4
    msub    x6, x5, x4, x0
    add     w6, w6, #'0'
    strb    w6, [x1, #-1]!
    mov     x0, x5
    cbnz    x0, pn_loop
    cbz     x3, pn_write
    mov     w2, #'-'
    strb    w2, [x1, #-1]!
    b       pn_write
pn_zero:
    mov     w2, #'0'
    strb    w2, [x1, #-1]!
pn_write:
    add     x2, x7, #32
    sub     x2, x2, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    ret

.section .rodata
error_msg:
    .ascii  "error\n"

.bss
    .balign 8
input_buf:
    .space  65536
    .balign 8
rpn_stack:
    .space  8192
    .balign 8
num_buf:
    .space  32
