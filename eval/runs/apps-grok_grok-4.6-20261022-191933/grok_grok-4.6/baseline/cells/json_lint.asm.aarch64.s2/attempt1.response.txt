.section .rodata
msg_ok:
    .ascii "ok\n"
.equ MSG_OK_LEN, . - msg_ok
msg_err:
    .ascii "error at "
.equ MSG_ERR_LEN, . - msg_err
msg_nl:
    .ascii "\n"
.equ MSG_NL_LEN, . - msg_nl

.bss
.balign 8
input:
    .space 65536
numbuf:
    .space 32

.text
.p2align 2
.globl _start
_start:
    adrp    x19, input
    add     x19, x19, :lo12:input
    mov     x20, #0
    mov     x21, #65536
read_loop:
    cmp     x20, x21
    b.hs    read_done
    mov     x0, #0
    add     x1, x19, x20
    sub     x2, x21, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:
    mov     x21, x20
    mov     x20, #0
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.ne    error
success:
    mov     x0, #1
    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #MSG_OK_LEN
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0

skip_ws:
    cmp     x20, x21
    b.ge    2f
    ldrb    w0, [x19, x20]
    cmp     w0, #0x20
    b.eq    1f
    cmp     w0, #0x09
    b.eq    1f
    cmp     w0, #0x0A
    b.eq    1f
    cmp     w0, #0x0D
    b.eq    1f
    ret
1:
    add     x20, x20, #1
    b       skip_ws
2:
    ret

parse_value:
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #'{'
    b.eq    parse_object
    cmp     w0, #'['
    b.eq    parse_array
    cmp     w0, #0x22
    b.eq    parse_string
    cmp     w0, #'t'
    b.eq    parse_true
    cmp     w0, #'f'
    b.eq    parse_false
    cmp     w0, #'n'
    b.eq    parse_null
    cmp     w0, #'-'
    b.eq    parse_number
    cmp     w0, #'0'
    b.lo    error
    cmp     w0, #'9'
    b.hi    error
    b       parse_number

expect:
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, w1
    b.ne    error
    add     x20, x20, #1
    ret

parse_true:
    add     x20, x20, #1
    stp     x29, x30, [sp, #-16]!
    mov     w1, #'r'
    bl      expect
    mov     w1, #'u'
    bl      expect
    mov     w1, #'e'
    bl      expect
    ldp     x29, x30, [sp], #16
    ret

parse_false:
    add     x20, x20, #1
    stp     x29, x30, [sp, #-16]!
    mov     w1, #'a'
    bl      expect
    mov     w1, #'l'
    bl      expect
    mov     w1, #'s'
    bl      expect
    mov     w1, #'e'
    bl      expect
    ldp     x29, x30, [sp], #16
    ret

parse_null:
    add     x20, x20, #1
    stp     x29, x30, [sp, #-16]!
    mov     w1, #'u'
    bl      expect
    mov     w1, #'l'
    bl      expect
    mov     w1, #'l'
    bl      expect
    ldp     x29, x30, [sp], #16
    ret

parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #'}'
    b.eq    po_done
po_member:
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #0x22
    b.ne    error
    bl      parse_string
    bl      skip_ws
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #':'
    b.ne    error
    add     x20, x20, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #','
    b.eq    po_next
    cmp     w0, #'}'
    b.eq    po_done
    b       error
po_next:
    add     x20, x20, #1
    bl      skip_ws
    b       po_member
po_done:
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret

parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #']'
    b.eq    pa_done
pa_value:
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #','
    b.eq    pa_next
    cmp     w0, #']'
    b.eq    pa_done
    b       error
pa_next:
    add     x20, x20, #1
    bl      skip_ws
    b       pa_value
pa_done:
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret

parse_string:
    add     x20, x20, #1
ps_loop:
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #0x22
    b.eq    ps_end
    cmp     w0, #0x5C
    b.eq    ps_esc
    cmp     w0, #0x20
    b.lo    error
    add     x20, x20, #1
    b       ps_loop
ps_end:
    add     x20, x20, #1
    ret
ps_esc:
    add     x20, x20, #1
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #0x22
    b.eq    ps_esc_ok
    cmp     w0, #0x5C
    b.eq    ps_esc_ok
    cmp     w0, #'/'
    b.eq    ps_esc_ok
    cmp     w0, #'b'
    b.eq    ps_esc_ok
    cmp     w0, #'f'
    b.eq    ps_esc_ok
    cmp     w0, #'n'
    b.eq    ps_esc_ok
    cmp     w0, #'r'
    b.eq    ps_esc_ok
    cmp     w0, #'t'
    b.eq    ps_esc_ok
    cmp     w0, #'u'
    b.eq    ps_uni
    b       error
ps_esc_ok:
    add     x20, x20, #1
    b       ps_loop
ps_uni:
    add     x20, x20, #1
    mov     x1, #4
ps_uni_loop:
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #'0'
    b.lo    error
    cmp     w0, #'9'
    b.ls    ps_hex_ok
    cmp     w0, #'A'
    b.lo    error
    cmp     w0, #'F'
    b.ls    ps_hex_ok
    cmp     w0, #'a'
    b.lo    error
    cmp     w0, #'f'
    b.ls    ps_hex_ok
    b       error
ps_hex_ok:
    add     x20, x20, #1
    subs    x1, x1, #1
    b.ne    ps_uni_loop
    b       ps_loop

parse_number:
    ldrb    w0, [x19, x20]
    cmp     w0, #'-'
    b.ne    pn_int
    add     x20, x20, #1
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
pn_int:
    cmp     w0, #'0'
    b.eq    pn_zero
    cmp     w0, #'1'
    b.lo    error
    cmp     w0, #'9'
    b.hi    error
    add     x20, x20, #1
pn_int_more:
    cmp     x20, x21
    b.ge    pn_done
    ldrb    w0, [x19, x20]
    cmp     w0, #'0'
    b.lo    pn_frac
    cmp     w0, #'9'
    b.hi    pn_frac
    add     x20, x20, #1
    b       pn_int_more
pn_zero:
    add     x20, x20, #1
pn_frac:
    cmp     x20, x21
    b.ge    pn_done
    ldrb    w0, [x19, x20]
    cmp     w0, #'.'
    b.ne    pn_exp
    add     x20, x20, #1
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #'0'
    b.lo    error
    cmp     w0, #'9'
    b.hi    error
    add     x20, x20, #1
pn_frac_more:
    cmp     x20, x21
    b.ge    pn_done
    ldrb    w0, [x19, x20]
    cmp     w0, #'0'
    b.lo    pn_exp
    cmp     w0, #'9'
    b.hi    pn_exp
    add     x20, x20, #1
    b       pn_frac_more
pn_exp:
    cmp     x20, x21
    b.ge    pn_done
    ldrb    w0, [x19, x20]
    cmp     w0, #'e'
    b.eq    pn_exp_go
    cmp     w0, #'E'
    b.ne    pn_done
pn_exp_go:
    add     x20, x20, #1
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
    cmp     w0, #'+'
    b.eq    pn_exp_sign
    cmp     w0, #'-'
    b.ne    pn_exp_digs
pn_exp_sign:
    add     x20, x20, #1
    cmp     x20, x21
    b.ge    error
    ldrb    w0, [x19, x20]
pn_exp_digs:
    cmp     w0, #'0'
    b.lo    error
    cmp     w0, #'9'
    b.hi    error
    add     x20, x20, #1
pn_exp_more:
    cmp     x20, x21
    b.ge    pn_done
    ldrb    w0, [x19, x20]
    cmp     w0, #'0'
    b.lo    pn_done
    cmp     w0, #'9'
    b.hi    pn_done
    add     x20, x20, #1
    b       pn_exp_more
pn_done:
    ret

error:
    mov     x0, #1
    adrp    x1, msg_err
    add     x1, x1, :lo12:msg_err
    mov     x2, #MSG_ERR_LEN
    mov     x8, #64
    svc     #0
    mov     x0, x20
    adrp    x1, numbuf
    add     x1, x1, :lo12:numbuf
    add     x1, x1, #20
    mov     x3, #0
    cbnz    x0, conv_loop
    sub     x1, x1, #1
    mov     w2, #'0'
    strb    w2, [x1]
    mov     x3, #1
    b       conv_write
conv_loop:
    mov     x4, #10
conv_iter:
    cbz     x0, conv_write
    udiv    x5, x0, x4
    msub    x6, x5, x4, x0
    add     w6, w6, #'0'
    sub     x1, x1, #1
    strb    w6, [x1]
    add     x3, x3, #1
    mov     x0, x5
    b       conv_iter
conv_write:
    mov     x0, #1
    mov     x2, x3
    mov     x8, #64
    svc     #0
    mov     x0, #1
    adrp    x1, msg_nl
    add     x1, x1, :lo12:msg_nl
    mov     x2, #MSG_NL_LEN
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0
