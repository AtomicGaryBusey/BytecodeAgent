.globl _start
.text
_start:
    lea r15, [rip + input_buf]
    xor r12, r12
.Lread:
    mov rdx, 65536
    sub rdx, r12
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r15 + r12]
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread
.Lread_done:
    lea rbx, [rip + hex_table]
    lea r14, [rip + line_buf]
    xor r13, r13
    test r12, r12
    jz .Lfinal
.Ldump:
    mov ebp, 16
    mov rdx, r12
    sub rdx, r13
    cmp rdx, rbp
    cmovb rbp, rdx
    lea r10, [r15 + r13]
    mov rdi, r14
    mov eax, 0x20
    mov ecx, 60
    rep stosb
    mov eax, r13d
    mov rdi, r14
    mov ecx, 8
.Loff:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, [rbx + rdx]
    mov [rdi], dl
    inc rdi
    dec ecx
    jnz .Loff
    xor r8d, r8d
.Lhex:
    cmp r8d, ebp
    jae .Lhex_done
    movzx eax, byte ptr [r10 + r8]
    lea r9, [r8 + r8*2]
    add r9, r14
    add r9, 10
    cmp r8d, 8
    jb .Lhex_pos
    inc r9
.Lhex_pos:
    mov ecx, eax
    shr ecx, 4
    mov cl, [rbx + rcx]
    mov [r9], cl
    and eax, 15
    mov al, [rbx + rax]
    mov [r9 + 1], al
    inc r8d
    jmp .Lhex
.Lhex_done:
    mov byte ptr [r14 + 60], '|'
    xor r8d, r8d
.Lascii:
    cmp r8d, ebp
    jae .Lascii_done
    mov al, [r10 + r8]
    cmp al, 0x20
    jb .Ldot
    cmp al, 0x7e
    jbe .Lputch
.Ldot:
    mov al, '.'
.Lputch:
    mov [r14 + r8 + 61], al
    inc r8d
    jmp .Lascii
.Lascii_done:
    mov byte ptr [r14 + rbp + 61], '|'
    mov byte ptr [r14 + rbp + 62], 10
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    lea rdx, [rbp + 63]
    syscall
    add r13, rbp
    cmp r13, r12
    jb .Ldump
.Lfinal:
    mov eax, r12d
    mov rdi, r14
    mov ecx, 8
.Lfinal_hex:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, [rbx + rdx]
    mov [rdi], dl
    inc rdi
    dec ecx
    jnz .Lfinal_hex
    mov byte ptr [r14 + 8], 10
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, 9
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .space 65536
line_buf:
    .space 80
