.globl _start

.text
_start:
    lea r13, [rip + input]
    xor r14, r14
read_loop:
    mov edx, 65536
    sub edx, r14d
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r14]
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r12, r12
    call skip_ws
    cmp r12, r14
    jae do_error
    call parse_value
    test eax, eax
    jz do_error
    call skip_ws
    cmp r12, r14
    jne do_error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

do_error:
    lea rsi, [rip + msg_err]
    lea rdi, [rip + outbuf]
    mov ecx, 9
    rep movsb
    mov eax, r12d
    mov r8d, 10
    xor ecx, ecx
itoa_loop:
    xor edx, edx
    div r8d
    add edx, 48
    push rdx
    inc ecx
    test eax, eax
    jnz itoa_loop
itoa_store:
    pop rax
    stosb
    dec ecx
    jnz itoa_store
    mov al, 10
    stosb
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r12, r14
    jae skip_ws_done
    movzx eax, byte ptr [r13 + r12]
    cmp al, 0x20
    ja skip_ws_done
    je skip_ws_eat
    cmp al, 0x09
    je skip_ws_eat
    cmp al, 0x0A
    je skip_ws_eat
    cmp al, 0x0D
    jne skip_ws_done
skip_ws_eat:
    inc r12
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp r12, r14
    jae pv_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb pv_lit
    cmp al, '9'
    jbe parse_number
pv_lit:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
pv_fail:
    xor eax, eax
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r14
    jae obj_fail
    cmp byte ptr [r13 + r12], '}'
    je obj_end
obj_member:
    cmp r12, r14
    jae obj_fail
    cmp byte ptr [r13 + r12], '"'
    jne obj_fail
    call parse_string
    test eax, eax
    jz obj_fail
    call skip_ws
    cmp r12, r14
    jae obj_fail
    cmp byte ptr [r13 + r12], ':'
    jne obj_fail
    inc r12
    call skip_ws
    call parse_value
    test eax, eax
    jz obj_fail
    call skip_ws
    cmp r12, r14
    jae obj_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, ','
    je obj_comma
    cmp al, '}'
    je obj_end
obj_fail:
    xor eax, eax
    ret
obj_comma:
    inc r12
    call skip_ws
    jmp obj_member
obj_end:
    inc r12
    mov eax, 1
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r14
    jae arr_fail
    cmp byte ptr [r13 + r12], ']'
    je arr_end
arr_elem:
    call parse_value
    test eax, eax
    jz arr_fail
    call skip_ws
    cmp r12, r14
    jae arr_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, ','
    je arr_comma
    cmp al, ']'
    je arr_end
arr_fail:
    xor eax, eax
    ret
arr_comma:
    inc r12
    call skip_ws
    jmp arr_elem
arr_end:
    inc r12
    mov eax, 1
    ret

parse_string:
    inc r12
str_loop:
    cmp r12, r14
    jae str_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '"'
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb str_fail
    inc r12
    jmp str_loop
str_end:
    inc r12
    mov eax, 1
    ret
str_fail:
    xor eax, eax
    ret
str_esc:
    inc r12
    cmp r12, r14
    jae str_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '"'
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    je str_uni
    jmp str_fail
str_esc_ok:
    inc r12
    jmp str_loop
str_uni:
    inc r12
    mov ecx, 4
str_hex:
    cmp r12, r14
    jae str_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    jb str_fail
    cmp al, '9'
    jbe str_hex_ok
    or al, 0x20
    cmp al, 'a'
    jb str_fail
    cmp al, 'f'
    ja str_fail
str_hex_ok:
    inc r12
    dec ecx
    jnz str_hex
    jmp str_loop

parse_number:
    cmp byte ptr [r13 + r12], '-'
    jne num_int
    inc r12
num_int:
    cmp r12, r14
    jae num_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb num_fail
    cmp al, '9'
    ja num_fail
num_int_loop:
    inc r12
    cmp r12, r14
    jae num_frac
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    jb num_frac
    cmp al, '9'
    ja num_frac
    jmp num_int_loop
num_zero:
    inc r12
num_frac:
    cmp r12, r14
    jae num_ok
    cmp byte ptr [r13 + r12], '.'
    jne num_exp
    inc r12
    cmp r12, r14
    jae num_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    jb num_fail
    cmp al, '9'
    ja num_fail
num_frac_loop:
    inc r12
    cmp r12, r14
    jae num_ok
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    jb num_exp
    cmp al, '9'
    ja num_exp
    jmp num_frac_loop
num_exp:
    cmp r12, r14
    jae num_ok
    movzx eax, byte ptr [r13 + r12]
    cmp al, 'e'
    je num_exp_go
    cmp al, 'E'
    jne num_ok
num_exp_go:
    inc r12
    cmp r12, r14
    jae num_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '+'
    je num_sign
    cmp al, '-'
    jne num_edig
num_sign:
    inc r12
num_edig:
    cmp r12, r14
    jae num_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    jb num_fail
    cmp al, '9'
    ja num_fail
num_edig_loop:
    inc r12
    cmp r12, r14
    jae num_ok
    movzx eax, byte ptr [r13 + r12]
    cmp al, '0'
    jb num_ok
    cmp al, '9'
    ja num_ok
    jmp num_edig_loop
num_ok:
    mov eax, 1
    ret
num_fail:
    xor eax, eax
    ret

parse_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp match_lit
parse_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp match_lit
parse_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4
    jmp match_lit
match_lit:
    cmp r12, r14
    jae match_fail
    movzx eax, byte ptr [r13 + r12]
    cmp al, byte ptr [rsi]
    jne match_fail
    inc r12
    inc rsi
    dec ecx
    jnz match_lit
    mov eax, 1
    ret
match_fail:
    xor eax, eax
    ret

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.balign 16
input:
    .space 65536
outbuf:
    .space 32
