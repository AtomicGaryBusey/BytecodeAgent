.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFSIZE, 65536

.text
.globl _start
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, BUFSIZE
    syscall
    test rax, rax
    jle done_reading
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buffer]

process_loop:
    movzx edx, byte ptr [rsi]
    inc rsi
    cmp dl, 10
    jne not_nl
    inc r12
not_nl:
    cmp dl, 32
    je is_ws
    cmp dl, 9
    jb not_ws
    cmp dl, 13
    ja not_ws
is_ws:
    xor r15, r15
    jmp next_byte
not_ws:
    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
next_byte:
    dec rcx
    jnz process_loop
    jmp read_loop

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call append_uint
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call append_uint
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call append_uint
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    mov rdx, rdi
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

append_uint:
    push rbx
    lea rbx, [rip + numbuf + 20]
    mov rsi, rbx
    mov ecx, 10
    test rax, rax
    jnz append_loop
    dec rsi
    mov byte ptr [rsi], 48
    jmp append_copy
append_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz append_loop
append_copy:
    mov rcx, rbx
    sub rcx, rsi
append_cpy_loop:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec rcx
    jnz append_cpy_loop
    pop rbx
    ret

.bss
.align 16
buffer:
    .space 65536
numbuf:
    .space 32
outbuf:
    .space 32
