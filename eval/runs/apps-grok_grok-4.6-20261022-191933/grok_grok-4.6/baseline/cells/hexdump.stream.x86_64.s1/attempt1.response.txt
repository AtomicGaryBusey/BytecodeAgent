.globl _start
.text
_start:
    xor r12, r12
read_loop:
    mov edx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
row_loop:
    cmp r13, r12
    jae print_final
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    lea r15, [rip + inbuf]
    add r15, r13
    lea rdi, [rip + linebuf]
    mov ecx, 60
    mov al, 0x20
    rep stosb
    lea rdi, [rip + linebuf]
    mov eax, r13d
    call put_hex8
    xor ebx, ebx
hex_loop:
    cmp rbx, r14
    jae hex_done
    lea rdi, [rbx + rbx*2 + 10]
    cmp ebx, 8
    jb 2f
    inc rdi
2:
    lea rax, [rip + linebuf]
    add rdi, rax
    movzx eax, byte ptr [r15 + rbx]
    call put_hex2
    inc ebx
    jmp hex_loop
hex_done:
    lea rdi, [rip + linebuf]
    mov byte ptr [rdi + 60], '|'
    xor ebx, ebx
asc_loop:
    cmp rbx, r14
    jae asc_done
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 0x20
    jb 3f
    cmp al, 0x7e
    jbe 4f
3:
    mov al, 0x2e
4:
    mov byte ptr [rdi + rbx + 61], al
    inc ebx
    jmp asc_loop
asc_done:
    mov byte ptr [rdi + r14 + 61], '|'
    mov byte ptr [rdi + r14 + 62], 0x0a
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    lea rdx, [r14 + 63]
    syscall
    add r13, r14
    jmp row_loop
print_final:
    lea rdi, [rip + linebuf]
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 0x0a
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov edx, 9
    syscall
    mov eax, 60
    xor edi, edi
    syscall
5:
    jmp 5b

put_hex8:
    lea r8, [rip + hexdigits]
    mov ecx, 8
6:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 6b
    ret

put_hex2:
    lea r8, [rip + hexdigits]
    movzx ecx, al
    shr ecx, 4
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi], cl
    movzx ecx, al
    and ecx, 15
    mov cl, byte ptr [r8 + rcx]
    mov byte ptr [rdi + 1], cl
    ret

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .space 65536
linebuf:
    .space 80
