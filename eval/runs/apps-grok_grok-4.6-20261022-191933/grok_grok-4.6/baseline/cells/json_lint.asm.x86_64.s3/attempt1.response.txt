.globl _start
.text
_start:
    lea rbx, [rip + buffer]
    mov rsi, rbx
    mov edx, 65536
read_loop:
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    sub rdx, rax
    jnz read_loop
read_done:
    mov r14, rsi
    mov rsi, rbx
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r14
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp rsi, r14
    jae skip_ws_done
    movzx eax, byte ptr [rsi]
    cmp al, 0x20
    je skip_ws_one
    ja skip_ws_done
    cmp al, 0x09
    je skip_ws_one
    cmp al, 0x0A
    je skip_ws_one
    cmp al, 0x0D
    je skip_ws_one
skip_ws_done:
    ret
skip_ws_one:
    inc rsi
    jmp skip_ws

parse_value:
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp parse_number

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r14
    jae error
    cmp byte ptr [rsi], '}'
    je obj_end
obj_member:
    cmp rsi, r14
    jae error
    cmp byte ptr [rsi], '"'
    jne error
    call parse_string
    call skip_ws
    cmp rsi, r14
    jae error
    cmp byte ptr [rsi], ':'
    jne error
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je obj_comma
    cmp al, '}'
    je obj_end
    jmp error
obj_comma:
    inc rsi
    call skip_ws
    jmp obj_member
obj_end:
    inc rsi
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r14
    jae error
    cmp byte ptr [rsi], ']'
    je arr_end
arr_value:
    call parse_value
    call skip_ws
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, ','
    je arr_comma
    cmp al, ']'
    je arr_end
    jmp error
arr_comma:
    inc rsi
    call skip_ws
    jmp arr_value
arr_end:
    inc rsi
    ret

parse_string:
    inc rsi
str_loop:
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb error
    inc rsi
    jmp str_loop
str_end:
    inc rsi
    ret
str_esc:
    inc rsi
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    je str_unicode
    jmp error
str_esc_ok:
    inc rsi
    jmp str_loop
str_unicode:
    inc rsi
    mov ecx, 4
str_hex:
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe str_hex_ok
    or al, 0x20
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
str_hex_ok:
    inc rsi
    dec ecx
    jnz str_hex
    jmp str_loop

parse_number:
    cmp byte ptr [rsi], '-'
    jne num_int
    inc rsi
    cmp rsi, r14
    jae error
num_int:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
    inc rsi
num_int_digits:
    cmp rsi, r14
    jae num_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb num_after_int
    cmp al, '9'
    ja num_after_int
    inc rsi
    jmp num_int_digits
num_zero:
    inc rsi
num_after_int:
    cmp rsi, r14
    jae num_done
    cmp byte ptr [rsi], '.'
    jne num_check_exp
    inc rsi
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc rsi
num_frac_digits:
    cmp rsi, r14
    jae num_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb num_check_exp
    cmp al, '9'
    ja num_check_exp
    inc rsi
    jmp num_frac_digits
num_check_exp:
    cmp rsi, r14
    jae num_done
    movzx eax, byte ptr [rsi]
    cmp al, 'e'
    je num_exp
    cmp al, 'E'
    je num_exp
    jmp num_done
num_exp:
    inc rsi
    cmp rsi, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je num_exp_sign
    cmp al, '-'
    je num_exp_sign
    jmp num_exp_need_digit
num_exp_sign:
    inc rsi
    cmp rsi, r14
    jae error
num_exp_need_digit:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc rsi
num_exp_digits:
    cmp rsi, r14
    jae num_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb num_done
    cmp al, '9'
    ja num_done
    inc rsi
    jmp num_exp_digits
num_done:
    ret

parse_true:
    inc rsi
    mov dl, 'r'
    call expect
    mov dl, 'u'
    call expect
    mov dl, 'e'
    call expect
    ret

parse_false:
    inc rsi
    mov dl, 'a'
    call expect
    mov dl, 'l'
    call expect
    mov dl, 's'
    call expect
    mov dl, 'e'
    call expect
    ret

parse_null:
    inc rsi
    mov dl, 'u'
    call expect
    mov dl, 'l'
    call expect
    mov dl, 'l'
    call expect
    ret

expect:
    cmp rsi, r14
    jae error
    cmp byte ptr [rsi], dl
    jne error
    inc rsi
    ret

error:
    cmp rsi, r14
    cmova rsi, r14
    mov rax, rsi
    sub rax, rbx
    lea rdi, [rip + numbuf + 15]
    mov byte ptr [rdi], 10
    mov ecx, 10
    mov r8d, eax
err_conv:
    mov eax, r8d
    xor edx, edx
    div ecx
    mov r8d, eax
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test r8d, r8d
    jnz err_conv
    mov r8, rdi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rdx, [rip + numbuf + 16]
    sub rdx, r8
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "

.bss
.balign 16
buffer:
    .space 65536
    .space 16
numbuf:
    .space 16
