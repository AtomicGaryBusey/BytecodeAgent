.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ MAXIN, 65536
.equ TAPESIZE, 30000

.text
_start:
    lea r12, [rip + input_buf]
    xor r15d, r15d
read_loop:
    cmp r15, MAXIN
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r15]
    mov edx, MAXIN
    sub edx, r15d
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    mov rdi, r12
    mov rcx, r15
    test rcx, rcx
    jz no_bang
    mov al, 0x21
find_bang:
    cmp byte ptr [rdi], al
    je found_bang
    inc rdi
    dec rcx
    jnz find_bang
no_bang:
    lea r13, [r12 + r15]
    mov rbx, r13
    mov r8, r13
    jmp check_brackets
found_bang:
    mov r13, rdi
    lea rbx, [rdi + 1]
    lea r8, [r12 + r15]
check_brackets:
    lea r10, [rip + jump_table]
    mov rbp, rsp
    xor ecx, ecx
    mov r14, r13
    sub r14, r12
br_loop:
    cmp rcx, r14
    jae br_done
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x5B
    je br_open
    cmp al, 0x5D
    je br_close
    inc rcx
    jmp br_loop
br_open:
    push rcx
    inc rcx
    jmp br_loop
br_close:
    cmp rsp, rbp
    je unbalanced
    pop rdx
    mov dword ptr [r10 + rdx*4], ecx
    mov dword ptr [r10 + rcx*4], edx
    inc rcx
    jmp br_loop
br_done:
    cmp rsp, rbp
    jne unbalanced
    lea rdi, [rip + dispatch]
    mov r11, rdi
    lea rax, [rip + cmd_other]
    mov ecx, 256
    rep stosq
    lea rax, [rip + cmd_right]
    mov qword ptr [r11 + 0x3E*8], rax
    lea rax, [rip + cmd_left]
    mov qword ptr [r11 + 0x3C*8], rax
    lea rax, [rip + cmd_plus]
    mov qword ptr [r11 + 0x2B*8], rax
    lea rax, [rip + cmd_minus]
    mov qword ptr [r11 + 0x2D*8], rax
    lea rax, [rip + cmd_output]
    mov qword ptr [r11 + 0x2E*8], rax
    lea rax, [rip + cmd_input]
    mov qword ptr [r11 + 0x2C*8], rax
    lea rax, [rip + cmd_lbrack]
    mov qword ptr [r11 + 0x5B*8], rax
    lea rax, [rip + cmd_rbrack]
    mov qword ptr [r11 + 0x5D*8], rax
    mov r14, r12
    lea r15, [rip + tape]
    lea r9, [rip + outbuf]
interp_loop:
    cmp r14, r13
    jae interp_done
    movzx eax, byte ptr [r14]
    jmp qword ptr [r11 + rax*8]
cmd_other:
    inc r14
    jmp interp_loop
cmd_right:
    inc r15
    inc r14
    jmp interp_loop
cmd_left:
    dec r15
    inc r14
    jmp interp_loop
cmd_plus:
    inc byte ptr [r15]
    inc r14
    jmp interp_loop
cmd_minus:
    dec byte ptr [r15]
    inc r14
    jmp interp_loop
cmd_output:
    mov al, byte ptr [r15]
    mov byte ptr [r9], al
    inc r9
    inc r14
    jmp interp_loop
cmd_input:
    xor eax, eax
    cmp rbx, r8
    jae 1f
    mov al, byte ptr [rbx]
    inc rbx
1:
    mov byte ptr [r15], al
    inc r14
    jmp interp_loop
cmd_lbrack:
    cmp byte ptr [r15], 0
    jne 1f
    mov rax, r14
    sub rax, r12
    mov eax, dword ptr [r10 + rax*4]
    lea r14, [r12 + rax]
1:
    inc r14
    jmp interp_loop
cmd_rbrack:
    cmp byte ptr [r15], 0
    je 1f
    mov rax, r14
    sub rax, r12
    mov eax, dword ptr [r10 + rax*4]
    lea r14, [r12 + rax]
1:
    inc r14
    jmp interp_loop
interp_done:
    lea rsi, [rip + outbuf]
    mov rdx, r9
    sub rdx, rsi
    jz exit0
write_more:
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    test rax, rax
    jle exit0
    add rsi, rax
    sub rdx, rax
    jnz write_more
exit0:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall
    jmp exit0
unbalanced:
    mov rsp, rbp
    mov eax, SYS_EXIT
    mov edi, 1
    syscall
    jmp unbalanced

.bss
.balign 8
input_buf:
    .space MAXIN
tape:
    .space TAPESIZE
.balign 4
jump_table:
    .space MAXIN * 4
.balign 8
dispatch:
    .space 256 * 8
outbuf:
    .space MAXIN
