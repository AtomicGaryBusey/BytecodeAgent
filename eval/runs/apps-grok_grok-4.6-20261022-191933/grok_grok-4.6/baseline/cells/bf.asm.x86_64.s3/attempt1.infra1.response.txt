.globl _start

.text
_start:
    lea r12, [rip + input_buf]
    xor r13, r13
    mov r14d, 65536

read_loop:
    mov rdx, r14
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor ecx, ecx

find_bang:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov r14, rcx
    lea r8, [r12 + rcx + 1]
    lea r9, [r12 + r13]
    jmp check_brackets

no_bang:
    mov r14, r13
    lea
