.globl _start

SYS_READ = 0
SYS_WRITE = 1
SYS_EXIT = 60

.text
_start:
  lea r12, [rip + input_buf]
  xor ebp, ebp
read_loop:
  xor eax, eax
  xor edi, edi
  lea rsi, [r12 + rbp]
  mov edx, 65536
  sub edx, ebp
  jz read_done
  syscall
  test rax, rax
  jle read_done
  add rbp, rax
  cmp rbp, 65536
  jb read_loop
read_done:
  lea r14, [r12 + rbp]
  mov r13, r12
  call skip_ws
  call parse_value
  call skip_ws
  cmp r13, r14
  je ok_exit
  jmp error_exit

skip_ws:
  cmp r13, r14
  jae skip_ws_ret
  movzx eax, byte ptr [r13]
  cmp al, 0x20
  je skip_ws_eat
  cmp al, 0x09
  je skip_ws_eat
  cmp al, 0x0A
  je skip_ws_eat
  cmp al, 0x0D
  je skip_ws_eat
skip_ws_ret:
  ret
skip_ws_eat:
  inc r13
  jmp skip_ws

parse_value:
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, '{'
  je parse_object
  cmp al, '['
  je parse_array
  cmp al, '"'
  je parse_string
  cmp al, '-'
  je parse_number
  cmp al, '0'
  jb parse_value_lit
  cmp al, '9'
  jbe parse_number
parse_value_lit:
  cmp al, 't'
  je parse_true
  cmp al, 'f'
  je parse_false
  cmp al, 'n'
  je parse_null
  jmp error_exit

parse_true:
  lea rsi, [rip + lit_true]
  jmp match_literal
parse_false:
  lea rsi, [rip + lit_false]
  jmp match_literal
parse_null:
  lea rsi, [rip + lit_null]
  jmp match_literal

match_literal:
  movzx eax, byte ptr [rsi]
  test al, al
  jz match_lit_done
  cmp r13, r14
  jae error_exit
  cmp al, byte ptr [r13]
  jne error_exit
  inc rsi
  inc r13
  jmp match_literal
match_lit_done:
  ret

parse_object:
  inc r13
  call skip_ws
  cmp r13, r14
  jae error_exit
  cmp byte ptr [r13], '}'
  jne obj_member
  inc r13
  ret
obj_member:
  cmp r13, r14
  jae error_exit
  cmp byte ptr [r13], '"'
  jne error_exit
  call parse_string
  call skip_ws
  cmp r13, r14
  jae error_exit
  cmp byte ptr [r13], ':'
  jne error_exit
  inc r13
  call skip_ws
  call parse_value
  call skip_ws
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, ','
  je obj_comma
  cmp al, '}'
  jne error_exit
  inc r13
  ret
obj_comma:
  inc r13
  call skip_ws
  jmp obj_member

parse_array:
  inc r13
  call skip_ws
  cmp r13, r14
  jae error_exit
  cmp byte ptr [r13], ']'
  jne arr_value
  inc r13
  ret
arr_value:
  call parse_value
  call skip_ws
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, ','
  je arr_comma
  cmp al, ']'
  jne error_exit
  inc r13
  ret
arr_comma:
  inc r13
  call skip_ws
  jmp arr_value

parse_string:
  inc r13
str_loop:
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, '"'
  je str_end
  cmp al, 0x20
  jb error_exit
  inc r13
  cmp al, 0x5C
  jne str_loop
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  inc r13
  cmp al, '"'
  je str_loop
  cmp al, 0x5C
  je str_loop
  cmp al, '/'
  je str_loop
  cmp al, 'b'
  je str_loop
  cmp al, 'f'
  je str_loop
  cmp al, 'n'
  je str_loop
  cmp al, 'r'
  je str_loop
  cmp al, 't'
  je str_loop
  cmp al, 'u'
  je str_unicode
  dec r13
  jmp error_exit
str_unicode:
  mov ecx, 4
str_hex:
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, '0'
  jb error_exit
  cmp al, '9'
  jbe str_hex_ok
  cmp al, 'A'
  jb error_exit
  cmp al, 'F'
  jbe str_hex_ok
  cmp al, 'a'
  jb error_exit
  cmp al, 'f'
  ja error_exit
str_hex_ok:
  inc r13
  dec ecx
  jnz str_hex
  jmp str_loop
str_end:
  inc r13
  ret

parse_number:
  cmp byte ptr [r13], '-'
  jne num_int
  inc r13
  cmp r13, r14
  jae error_exit
num_int:
  movzx eax, byte ptr [r13]
  cmp al, '0'
  je num_zero
  cmp al, '1'
  jb error_exit
  cmp al, '9'
  ja error_exit
  inc r13
num_int_digits:
  cmp r13, r14
  jae num_done
  movzx eax, byte ptr [r13]
  cmp al, '0'
  jb num_frac
  cmp al, '9'
  ja num_frac
  inc r13
  jmp num_int_digits
num_zero:
  inc r13
num_frac:
  cmp r13, r14
  jae num_done
  cmp byte ptr [r13], '.'
  jne num_exp
  inc r13
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, '0'
  jb error_exit
  cmp al, '9'
  ja error_exit
  inc r13
num_frac_digits:
  cmp r13, r14
  jae num_done
  movzx eax, byte ptr [r13]
  cmp al, '0'
  jb num_exp
  cmp al, '9'
  ja num_exp
  inc r13
  jmp num_frac_digits
num_exp:
  cmp r13, r14
  jae num_done
  movzx eax, byte ptr [r13]
  cmp al, 'e'
  je num_do_exp
  cmp al, 'E'
  je num_do_exp
  jmp num_done
num_do_exp:
  inc r13
  cmp r13, r14
  jae error_exit
  movzx eax, byte ptr [r13]
  cmp al, '+'
  je num_exp_sign
  cmp al, '-'
  je num_exp_sign
  jmp num_exp_need_digit
num_exp_sign:
  inc r13
  cmp r13, r14
  jae error_exit
num_exp_need_digit:
  movzx eax, byte ptr [r13]
  cmp al, '0'
  jb error_exit
  cmp al, '9'
  ja error_exit
  inc r13
num_exp_digits:
  cmp r13, r14
  jae num_done
  movzx eax, byte ptr [r13]
  cmp al, '0'
  jb num_done
  cmp al, '9'
  ja num_done
  inc r13
  jmp num_exp_digits
num_done:
  ret

ok_exit:
  mov eax, SYS_WRITE
  mov edi, 1
  lea rsi, [rip + msg_ok]
  mov edx, 3
  syscall
  mov eax, SYS_EXIT
  xor edi, edi
  syscall

error_exit:
  mov rax, r13
  sub rax, r12
  lea rbx, [rip + numbuf + 16]
  mov byte ptr [rbx], 10
  mov ecx, 10
  test rax, rax
  jz err_zero
err_conv:
  xor rdx, rdx
  div rcx
  add dl, '0'
  dec rbx
  mov byte ptr [rbx], dl
  test rax, rax
  jnz err_conv
  jmp err_print
err_zero:
  dec rbx
  mov byte ptr [rbx], '0'
err_print:
  mov eax, SYS_WRITE
  mov edi, 1
  lea rsi, [rip + msg_err]
  mov edx, 9
  syscall
  mov eax, SYS_WRITE
  mov edi, 1
  mov rsi, rbx
  lea rdx, [rip + numbuf + 17]
  sub rdx, rbx
  syscall
  mov eax, SYS_EXIT
  mov edi, 1
  syscall

.section .rodata
msg_ok:
  .ascii "ok\n"
msg_err:
  .ascii "error at "
lit_true:
  .asciz "true"
lit_false:
  .asciz "false"
lit_null:
  .asciz "null"

.bss
.balign 8
input_buf:
  .space 65536
numbuf:
  .space 32
