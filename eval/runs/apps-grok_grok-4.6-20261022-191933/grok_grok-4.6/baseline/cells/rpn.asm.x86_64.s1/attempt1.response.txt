.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INPUT_MAX, 65536

.text
_start:
    lea r12, [rip + input_buf]
    mov r13, r12
    lea r14, [r12 + INPUT_MAX]
Lread:
    mov rdx, r14
    sub rdx, r13
    jz Lread_done
    mov eax, SYS_READ
    xor edi, edi
    mov rsi, r13
    syscall
    test rax, rax
    jle Lread_done
    add r13, rax
    jmp Lread
Lread_done:
    mov rbp, r13
    mov rbx, r12
Lline_loop:
    cmp rbx, rbp
    je Lexit
    mov rsi, rbx
Lfind_nl:
    cmp rsi, rbp
    je Llast_line
    cmp byte ptr [rsi], 10
    je Lfound_nl
    inc rsi
    jmp Lfind_nl
Lfound_nl:
    mov rdi, rbx
    mov rdx, rsi
    push rsi
    call process_line
    pop rsi
    lea rbx, [rsi + 1]
    jmp Lline_loop
Llast_line:
    mov rdi, rbx
    mov rdx, rbp
    call process_line
Lexit:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

process_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rdx
    lea r14, [rip + rpn_stack]
    xor r15, r15
Lpl_token_loop:
Lpl_skip_ws:
    cmp r12, r13
    jae Lpl_tokens_done
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je Lpl_is_ws
    cmp al, 0x09
    jne Lpl_got_tok
Lpl_is_ws:
    inc r12
    jmp Lpl_skip_ws
Lpl_got_tok:
    mov r8, r12
Lpl_scan_tok:
    inc r12
    cmp r12, r13
    jae Lpl_tok_end
    movzx eax, byte ptr [r12]
    cmp al, 0x20
    je Lpl_tok_end
    cmp al, 0x09
    jne Lpl_scan_tok
Lpl_tok_end:
    mov rcx, r12
    sub rcx, r8
    cmp rcx, 1
    jne Lpl_check_num
    movzx eax, byte ptr [r8]
    cmp al, '+'
    je Lpl_do_op
    cmp al, '-'
    je Lpl_do_op
    cmp al, '*'
    je Lpl_do_op
    cmp al, '/'
    je Lpl_do_op
    cmp al, '%'
    je Lpl_do_op
Lpl_check_num:
    mov rsi, r8
    xor r10, r10
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne Lpl_digits
    mov r10, 1
    inc rsi
Lpl_digits:
    mov rcx, r12
    sub rcx, rsi
    cmp rcx, 1
    jb Lpl_error
    cmp rcx, 18
    ja Lpl_error
    xor r11, r11
Lpl_dig_loop:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb Lpl_error
    cmp al, '9'
    ja Lpl_error
    imul r11, r11, 10
    sub eax, '0'
    add r11, rax
    inc rsi
    cmp rsi, r12
    jb Lpl_dig_loop
    test r10, r10
    jz Lpl_push
    neg r11
Lpl_push:
    mov qword ptr [r14 + r15*8], r11
    inc r15
    jmp Lpl_token_loop
Lpl_do_op:
    cmp r15, 2
    jb Lpl_error
    dec r15
    mov rcx, qword ptr [r14 + r15*8]
    dec r15
    mov rax, qword ptr [r14 + r15*8]
    movzx ebx, byte ptr [r8]
    cmp bl, '+'
    je Lpl_add
    cmp bl, '-'
    je Lpl_sub
    cmp bl, '*'
    je Lpl_mul
    cmp bl, '/'
    je Lpl_div
Lpl_mod:
    test rcx, rcx
    jz Lpl_error
    mov rbx, 1
    shl rbx, 63
    cmp rax, rbx
    jne Lpl_mod_ok
    cmp rcx, -1
    je Lpl_error
Lpl_mod_ok:
    cqo
    idiv rcx
    mov rax, rdx
    jmp Lpl_push_op
Lpl_div:
    test rcx, rcx
    jz Lpl_error
    mov rbx, 1
    shl rbx, 63
    cmp rax, rbx
    jne Lpl_div_ok
    cmp rcx, -1
    je Lpl_error
Lpl_div_ok:
    cqo
    idiv rcx
    jmp Lpl_push_op
Lpl_add:
    add rax, rcx
    jmp Lpl_push_op
Lpl_sub:
    sub rax, rcx
    jmp Lpl_push_op
Lpl_mul:
    imul rax, rcx
Lpl_push_op:
    mov qword ptr [r14 + r15*8], rax
    inc r15
    jmp Lpl_token_loop
Lpl_tokens_done:
    cmp r15, 1
    jne Lpl_error
    mov rax, qword ptr [r14]
    call print_i64
    jmp Lpl_ret
Lpl_error:
    call print_error
Lpl_ret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

print_i64:
    push rbx
    push r8
    lea rdi, [rip + numbuf]
    add rdi, 30
    mov byte ptr [rdi], 10
    test rax, rax
    jnz Lpr_nonzero
    dec rdi
    mov byte ptr [rdi], '0'
    jmp Lpr_write
Lpr_nonzero:
    mov rbx, rax
    xor r8, r8
    test rbx, rbx
    jns Lpr_conv
    mov r8, 1
    neg rbx
Lpr_conv:
    mov rcx, 10
Lpr_loop:
    xor rdx, rdx
    mov rax, rbx
    div rcx
    add edx, '0'
    dec rdi
    mov byte ptr [rdi], dl
    mov rbx, rax
    test rbx, rbx
    jnz Lpr_loop
    test r8, r8
    jz Lpr_write
    dec rdi
    mov byte ptr [rdi], '-'
Lpr_write:
    lea rdx, [rip + numbuf]
    add rdx, 30
    sub rdx, rdi
    inc rdx
    mov rsi, rdi
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    pop r8
    pop rbx
    ret

print_error:
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 6
    syscall
    ret

.section .rodata
err_msg:
    .ascii "error\n"

.bss
.balign 8
input_buf:
    .space 65536
rpn_stack:
    .space 8192
numbuf:
    .space 32
