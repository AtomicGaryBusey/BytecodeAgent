.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFSIZE, 4096

.text
_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x0, xzr
    mov x1, x23
    mov x2, #BUFSIZE
    mov x8, #SYS_READ
    svc #0
    cmp x0, xzr
    b.le done_read
    add x21, x21, x0
    mov x4, x23
    add x5, x23, x0

proc_loop:
    cmp x4, x5
    b.eq read_loop
    ldrb w3, [x4], #1
    cmp w3, #32
    b.eq is_ws
    sub w6, w3, #9
    cmp w6, #4
    b.ls is_ws
    cbnz x22, proc_loop
    add x20, x20, #1
    mov x22, #1
    b proc_loop

is_ws:
    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    mov x22, xzr
    b proc_loop

done_read:
    sub sp, sp, #64
    add x1, sp, #64
    mov w2, #10
    sub x1, x1, #1
    strb w2, [x1]
    mov x0, x21
    bl utoa
    mov w2, #32
    sub x1, x1, #1
    strb w2, [x1]
    mov x0, x20
    bl utoa
    mov w2, #32
    sub x1, x1, #1
    strb w2, [x1]
    mov x0, x19
    bl utoa
    add x2, sp, #64
    sub x2, x2, x1
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    mov x0, xzr
    mov x8, #SYS_EXIT
    svc #0

utoa:
    mov x3, #10
1:
    udiv x2, x0, x3
    msub x4, x2, x3, x0
    add w4, w4, #48
    sub x1, x1, #1
    strb w4, [x1]
    mov x0, x2
    cbnz x0, 1b
    ret

.bss
.balign 16
buf:
    .zero 4096
