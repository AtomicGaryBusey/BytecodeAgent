.globl _start
.text
_start:
    lea r12, [rip + buf]
    lea r15, [rip + line_ptr]
    lea r10, [rip + line_len]
    xor r13, r13
read_loop:
    mov rdx, 32768
    sub rdx, r13
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor ebp, ebp
    xor ebx, ebx
    xor r14d, r14d
parse_loop:
    cmp rbp, r13
    jae parse_tail
    cmp byte ptr [r12 + rbp], 10
    je parse_nl
    inc rbp
    jmp parse_loop
parse_nl:
    mov rax, r12
    add rax, rbx
    mov [r15 + r14*8], rax
    mov rax, rbp
    sub rax, rbx
    mov [r10 + r14*8], rax
    inc r14
    inc rbp
    mov rbx, rbp
    jmp parse_loop
parse_tail:
    cmp rbx, r13
    jae parse_done
    mov rax, r12
    add rax, rbx
    mov [r15 + r14*8], rax
    mov rax, r13
    sub rax, rbx
    mov [r10 + r14*8], rax
    inc r14
parse_done:
    mov ebx, 1
sort_outer:
    cmp rbx, r14
    jae sort_done
    mov r8, [r15 + rbx*8]
    mov r9, [r10 + rbx*8]
    mov rbp, rbx
sort_inner:
    test rbp, rbp
    jz sort_place
    mov rsi, [r15 + rbp*8 - 8]
    mov rax, [r10 + rbp*8 - 8]
    mov rdi, r8
    mov rdx, r9
    mov rcx, rax
    cmp rcx, rdx
    cmova rcx, rdx
    test rcx, rcx
    jz sort_cmplens
    repe cmpsb
    jne sort_cmpflags
sort_cmplens:
    cmp rax, rdx
sort_cmpflags:
    jbe sort_place
    mov rax, [r15 + rbp*8 - 8]
    mov [r15 + rbp*8], rax
    mov rax, [r10 + rbp*8 - 8]
    mov [r10 + rbp*8], rax
    dec rbp
    jmp sort_inner
sort_place:
    mov [r15 + rbp*8], r8
    mov [r10 + rbp*8], r9
    inc rbx
    jmp sort_outer
sort_done:
    xor ebx, ebx
print_loop:
    cmp rbx, r14
    jae print_done
    mov rsi, [r15 + rbx*8]
    mov rdx, [r10 + rbx*8]
    test rdx, rdx
    jz print_nl
    mov eax, 1
    mov edi, 1
    syscall
print_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    inc rbx
    jmp print_loop
print_done:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
nl:
    .byte 10

.bss
.balign 8
buf:
    .space 32768
.balign 8
line_ptr:
    .space 8000
.balign 8
line_len:
    .space 8000
