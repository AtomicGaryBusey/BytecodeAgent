.globl _start
.text
_start:
    lea rbx, [rip + input_buf]
    mov r13, rbx
    lea r8, [rbx + 65536]
read_loop:
    mov rdx, r8
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rip + value_stack]
check_more:
    cmp rbx, r13
    jge exit_ok
    mov r14, r12
token_loop:
    cmp rbx, r13
    jge line_end
    movzx eax, byte ptr [rbx]
    cmp al, 10
    je line_end
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp have_token
skip_one:
    inc rbx
    jmp token_loop
have_token:
    mov r15, rbx
scan_token:
    inc rbx
    cmp rbx, r13
    jge got_token
    movzx eax, byte ptr [rbx]
    cmp al, 32
    je got_token
    cmp al, 9
    je got_token
    cmp al, 10
    je got_token
    jmp scan_token
got_token:
    mov rcx, rbx
    sub rcx, r15
    cmp rcx, 1
    je single_char
    xor r8, r8
    xor r9, r9
    mov rsi, r15
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne num_digits
    mov r8, 1
    inc rsi
    dec rcx
num_digits:
    cmp rcx, 18
    ja do_error
num_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    sub al, '0'
    cmp al, 9
    ja do_error
    imul r9, r9, 10
    add r9, rax
    dec rcx
    jnz num_loop
    test r8, r8
    jz num_push
    neg r9
num_push:
    mov qword ptr [r14], r9
    add r14, 8
    jmp token_loop
single_char:
    movzx r10d, byte ptr [r15]
    cmp r10b, '0'
    jb maybe_op
    cmp r10b, '9'
    ja maybe_op
    sub r10, '0'
    mov qword ptr [r14], r10
    add r14, 8
    jmp token_loop
maybe_op:
    cmp r10b, '+'
    je do_op
    cmp r10b, '-'
    je do_op
    cmp r10b, '*'
    je do_op
    cmp r10b, '/'
    je do_op
    cmp r10b, '%'
    je do_op
    jmp do_error
do_op:
    mov rdx, r14
    sub rdx, r12
    cmp rdx, 16
    jb do_error
    sub r14, 16
    mov rax, qword ptr [r14]
    mov rcx, qword ptr [r14 + 8]
    cmp r10b, '+'
    je op_add
    cmp r10b, '-'
    je op_sub
    cmp r10b, '*'
    je op_mul
    jmp op_divmod
op_add:
    add rax, rcx
    jmp push_result
op_sub:
    sub rax, rcx
    jmp push_result
op_mul:
    imul rax, rcx
    jmp push_result
op_divmod:
    test rcx, rcx
    jz do_error
    cmp rcx, -1
    jne op_div_go
    mov r8, 1
    shl r8, 63
    cmp rax, r8
    je do_error
op_div_go:
    cqo
    idiv rcx
    cmp r10b, '/'
    je push_result
    mov rax, rdx
push_result:
    mov qword ptr [r14], rax
    add r14, 8
    jmp token_loop
line_end:
    mov rax, r14
    sub rax, r12
    cmp rax, 8
    jne do_error
    mov rax, qword ptr [r12]
    call print_int
    cmp rbx, r13
    jge check_more
    inc rbx
    jmp check_more
do_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 6
    syscall
skip_line:
    cmp rbx, r13
    jge check_more
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 10
    jne skip_line
    jmp check_more
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
print_int:
    lea r10, [rip + numbuf]
    add r10, 31
    mov byte ptr [r10], 10
    xor r8, r8
    test rax, rax
    jns pi_pos
    mov r8, 1
    neg rax
pi_pos:
    test rax, rax
    jnz pi_conv
    dec r10
    mov byte ptr [r10], '0'
    jmp pi_sign
pi_conv:
    mov r9, 10
pi_loop:
    xor rdx, rdx
    div r9
    add dl, '0'
    dec r10
    mov byte ptr [r10], dl
    test rax, rax
    jnz pi_loop
pi_sign:
    test r8, r8
    jz pi_write
    dec r10
    mov byte ptr [r10], '-'
pi_write:
    lea rdx, [rip + numbuf]
    add rdx, 32
    sub rdx, r10
    mov rsi, r10
    mov eax, 1
    mov edi, 1
    syscall
    ret
.section .rodata
error_msg:
    .ascii "error\n"
.bss
.p2align 3
input_buf:
    .space 65536
value_stack:
    .space 8192
numbuf:
    .space 32
