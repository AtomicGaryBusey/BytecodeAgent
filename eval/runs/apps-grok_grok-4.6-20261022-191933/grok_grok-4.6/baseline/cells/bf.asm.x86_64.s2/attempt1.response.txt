.globl _start
.text
_start:
    lea rbx, [rip + input_buf]
    xor r13, r13
    mov r12, 65536
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    mov rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    sub r12, rax
    jnz read_loop
read_done:
    xor r14, r14
split_loop:
    cmp r14, r13
    jae no_bang
    cmp byte ptr [rbx + r14], '!'
    je got_bang
    inc r14
    jmp split_loop
got_bang:
    lea r15, [rbx + r14 + 1]
    mov r12, r13
    sub r12, r14
    dec r12
    jmp after_split
no_bang:
    mov r14, r13
    xor r12, r12
after_split:
    xor ecx, ecx
    xor edx, edx
    lea r8, [rip + jump_table]
match_loop:
    cmp rcx, r14
    jae match_end
    mov al, byte ptr [rbx + rcx]
    cmp al, '['
    je match_open
    cmp al, ']'
    je match_close
match_next:
    inc rcx
    jmp match_loop
match_open:
    push rcx
    inc edx
    jmp match_next
match_close:
    test edx, edx
    jz unbalanced
    pop rsi
    dec edx
    lea eax, [ecx + 1]
    mov dword ptr [r8 + rsi*4], eax
    lea eax, [esi + 1]
    mov dword ptr [r8 + rcx*4], eax
    jmp match_next
match_end:
    test edx, edx
    jnz unbalanced
    lea r13, [rip + tape]
    xor ebp, ebp
interp_loop:
    cmp rbp, r14
    jae interp_done
    movzx eax, byte ptr [rbx + rbp]
    cmp al, '>'
    je cmd_gt
    cmp al, '<'
    je cmd_lt
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_lbrack
    cmp al, ']'
    je cmd_rbrack
next_cmd:
    inc rbp
    jmp interp_loop
cmd_gt:
    inc r13
    jmp next_cmd
cmd_lt:
    dec r13
    jmp next_cmd
cmd_plus:
    inc byte ptr [r13]
    jmp next_cmd
cmd_minus:
    dec byte ptr [r13]
    jmp next_cmd
cmd_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, r13
    mov edx, 1
    syscall
    jmp next_cmd
cmd_comma:
    xor eax, eax
    test r12, r12
    jz comma_store
    mov al, byte ptr [r15]
    inc r15
    dec r12
comma_store:
    mov byte ptr [r13], al
    jmp next_cmd
cmd_lbrack:
    cmp byte ptr [r13], 0
    je do_jump
    jmp next_cmd
cmd_rbrack:
    cmp byte ptr [r13], 0
    je next_cmd
do_jump:
    mov eax, dword ptr [r8 + rbp*4]
    mov ebp, eax
    jmp interp_loop
interp_done:
    mov eax, 60
    xor edi, edi
    syscall
unbalanced:
    mov eax, 60
    mov edi, 1
    syscall

.bss
input_buf:
    .zero 65536
tape:
    .zero 30000
.p2align 2
jump_table:
    .zero 262144
