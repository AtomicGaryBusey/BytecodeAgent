.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
.zero 49152
outbuf:
.zero 66560

.text
_start:
    xor r12, r12
read_loop:
    mov edx, 49152
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    xor r8, r8
encode_loop:
    cmp r12, 3
    jb encode_tail
    movzx ebx, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]
    add rsi, 3
    sub r12, 3
    mov eax, ebx
    shr eax, 2
    call emit_idx
    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov r9d, ecx
    shr r9d, 4
    or eax, r9d
    call emit_idx
    mov eax, ecx
    and eax, 15
    shl eax, 2
    mov r9d, edx
    shr r9d, 6
    or eax, r9d
    call emit_idx
    mov eax, edx
    and eax, 63
    call emit_idx
    jmp encode_loop
encode_tail:
    cmp r12, 2
    je encode_2
    cmp r12, 1
    je encode_1
    jmp encode_done
encode_2:
    movzx ebx, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    mov eax, ebx
    shr eax, 2
    call emit_idx
    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov r9d, ecx
    shr r9d, 4
    or eax, r9d
    call emit_idx
    mov eax, ecx
    and eax, 15
    shl eax, 2
    call emit_idx
    mov al, '='
    call emit_char
    jmp encode_done
encode_1:
    movzx ebx, byte ptr [rsi]
    mov eax, ebx
    shr eax, 2
    call emit_idx
    mov eax, ebx
    and eax, 3
    shl eax, 4
    call emit_idx
    mov al, '='
    call emit_char
    mov al, '='
    call emit_char
encode_done:
    test r8, r8
    jz 2f
    mov al, 10
    stosb
2:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit_idx:
    lea r10, [rip + alphabet]
    mov al, byte ptr [r10 + rax]
emit_char:
    stosb
    inc r8
    cmp r8, 76
    jne 1f
    mov al, 10
    stosb
    xor r8, r8
1:
    ret
