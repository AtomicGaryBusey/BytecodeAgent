.globl _start
.text
_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    adrp    x22, output_buf
    add     x22, x22, :lo12:output_buf
    adrp    x23, hex_table
    add     x23, x23, :lo12:hex_table

    mov     x20, #0
read_loop:
    mov     x0, #65536
    cmp     x20, x0
    b.hs    read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #65536
    sub     x2, x2, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop
read_done:

    mov     x21, #0
line_loop:
    cmp     x21, x20
    b.hs    lines_done
    sub     x0, x20, x21
    cmp     x0, #16
    mov     x1, #16
    csel    x24, x0, x1, lo
    mov     x25, x22

    mov     w0, w21
    mov     w1, #28
off_hex:
    lsr     w2, w0, w1
    and     w2, w2, #15
    ldrb    w2, [x23, x2]
    strb    w2, [x25], #1
    subs    w1, w1, #4
    b.ge    off_hex
    mov     w2, #32
    strb    w2, [x25], #1
    strb    w2, [x25], #1

    mov     x26, #0
hex_loop:
    cmp     x26, x24
    b.hs    empty_slot
    add     x0, x19, x21
    ldrb    w1, [x0, x26]
    lsr     w2, w1, #4
    ldrb    w2, [x23, x2]
    strb    w2, [x25], #1
    and     w2, w1, #15
    ldrb    w2, [x23, x2]
    strb    w2, [x25], #1
    mov     w2, #32
    strb    w2, [x25], #1
    b       after_slot
empty_slot:
    mov     w2, #32
    strb    w2, [x25], #1
    strb    w2, [x25], #1
    strb    w2, [x25], #1
after_slot:
    add     x26, x26, #1
    cmp     x26, #8
    b.ne    1f
    mov     w2, #32
    strb    w2, [x25], #1
1:
    cmp     x26, #16
    b.lo    hex_loop

    mov     w2, #32
    strb    w2, [x25], #1
    mov     w2, #124
    strb    w2, [x25], #1

    mov     x26, #0
ascii_loop:
    cmp     x26, x24
    b.hs    ascii_done
    add     x0, x19, x21
    ldrb    w1, [x0, x26]
    mov     w3, #46
    cmp     w1, #32
    csel    w2, w1, w3, hs
    cmp     w1, #127
    csel    w2, w3, w2, hs
    strb    w2, [x25], #1
    add     x26, x26, #1
    b       ascii_loop
ascii_done:
    mov     w2, #124
    strb    w2, [x25], #1
    mov     w2, #10
    strb    w2, [x25], #1

    mov     x0, #1
    mov     x1, x22
    sub     x2, x25, x22
    mov     x8, #64
    svc     #0

    add     x21, x21, #16
    b       line_loop

lines_done:
    mov     x25, x22
    mov     w0, w20
    mov     w1, #28
final_hex:
    lsr     w2, w0, w1
    and     w2, w2, #15
    ldrb    w2, [x23, x2]
    strb    w2, [x25], #1
    subs    w1, w1, #4
    b.ge    final_hex
    mov     w2, #10
    strb    w2, [x25], #1
    mov     x0, #1
    mov     x1, x22
    sub     x2, x25, x22
    mov     x8, #64
    svc     #0

    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
hex_table:
    .ascii  "0123456789abcdef"

.bss
.balign 16
input_buf:
    .space  65536
output_buf:
    .space  80
