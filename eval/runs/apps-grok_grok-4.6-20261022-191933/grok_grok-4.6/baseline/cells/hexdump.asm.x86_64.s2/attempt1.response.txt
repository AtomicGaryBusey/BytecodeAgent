.globl _start

.equ BUFSIZE, 65536

.text

_start:
    lea r12, [rip + inbuf]
    xor r13, r13

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, BUFSIZE
    sub edx, r13d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r14, r14

dump_loop:
    cmp r14, r13
    jae dump_done
    mov r15, r13
    sub r15, r14
    cmp r15, 16
    jbe 1f
    mov r15, 16
1:
    call format_line
    mov edx, eax
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    syscall
    add r14, r15
    jmp dump_loop

dump_done:
    lea rdi, [rip + linebuf]
    mov eax, r13d
    call put_hex8
    mov byte ptr [rdi], 10
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + linebuf]
    mov edx, 9
    syscall
    mov eax, 60
    xor edi, edi
    syscall

put_hex8:
    mov ecx, 8
1:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [hex_digits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 1b
    ret

format_line:
    lea rdi, [rip + linebuf]
    mov eax, r14d
    call put_hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    lea rsi, [r12 + r14]
    xor ecx, ecx
hex_loop:
    cmp ecx, 16
    je hex_done
    cmp ecx, r15d
    jae hex_pad
    movzx eax, byte ptr [rsi + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [hex_digits + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [hex_digits + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp hex_after
hex_pad:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
hex_after:
    inc ecx
    cmp ecx, 8
    jne hex_loop
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_loop
hex_done:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7C
    add rdi, 2
    xor ecx, ecx
ascii_loop:
    cmp ecx, r15d
    jae ascii_done
    movzx eax, byte ptr [rsi + rcx]
    mov edx, 0x2E
    cmp al, 0x20
    jb ascii_store
    cmp al, 0x7E
    ja ascii_store
    mov edx, eax
ascii_store:
    mov byte ptr [rdi], dl
    inc rdi
    inc ecx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], 0x7C
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    lea rax, [rip + linebuf]
    sub rdi, rax
    mov eax, edi
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .space BUFSIZE
linebuf:
    .space 80
