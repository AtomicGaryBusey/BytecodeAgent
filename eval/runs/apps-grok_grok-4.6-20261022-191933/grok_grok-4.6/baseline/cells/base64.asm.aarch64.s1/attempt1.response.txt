.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INSIZ, 49152
.equ LINELEN, 76

.globl _start
.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0
    mov x21, #INSIZ

read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, x21
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, read_loop

read_done:
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x23, x22
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet
    mov x25, x19
    add x28, x19, x20
    mov w26, #0

encode_loop:
    sub x0, x28, x25
    cmp x0, #3
    b.lt remain
    ldrb w0, [x25], #1
    ldrb w1, [x25], #1
    ldrb w2, [x25], #1
    lsl w3, w0, #16
    orr w3, w3, w1, lsl #8
    orr w3, w3, w2
    bl emit_four
    b encode_loop

remain:
    cbz x0, encode_done
    cmp x0, #1
    b.eq remain1
    ldrb w0, [x25], #1
    ldrb w1, [x25], #1
    lsl w3, w0, #16
    orr w3, w3, w1, lsl #8
    lsr w4, w3, #18
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    lsr w4, w3, #12
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    lsr w4, w3, #6
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    mov w4, #61
    strb w4, [x23], #1
    b remain_wrap

remain1:
    ldrb w0, [x25], #1
    lsl w3, w0, #16
    lsr w4, w3, #18
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    lsr w4, w3, #12
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    mov w4, #61
    strb w4, [x23], #1
    strb w4, [x23], #1

remain_wrap:
    add w26, w26, #4
    cmp w26, #LINELEN
    b.ne encode_done
    mov w4, #10
    strb w4, [x23], #1
    mov w26, #0

encode_done:
    cbz w26, write_out
    mov w4, #10
    strb w4, [x23], #1

write_out:
    sub x2, x23, x22
    cbz x2, do_exit
    mov x0, #1
    mov x1, x22
    mov x8, #SYS_WRITE
    svc #0

do_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

emit_four:
    lsr w4, w3, #18
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    lsr w4, w3, #12
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    lsr w4, w3, #6
    and w4, w4, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    and w4, w3, #63
    ldrb w4, [x24, x4]
    strb w4, [x23], #1
    add w26, w26, #4
    cmp w26, #LINELEN
    b.ne 1f
    mov w4, #10
    strb w4, [x23], #1
    mov w26, #0
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .space INSIZ
.balign 16
outbuf:
    .space 67000
