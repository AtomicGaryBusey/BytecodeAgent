.globl _start

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0

read_loop:
    mov x2, #65536
    sub x2, x2, x20
    cbz x2, process
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le process
    add x20, x20, x0
    b read_loop

process:
    mov x21, #0

line_loop:
    cmp x21, x20
    b.ge done
    mov x22, x21

scan:
    cmp x21, x20
    b.ge reverse
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq reverse
    add x21, x21, #1
    b scan

reverse:
    mov x0, x22
    cmp x0, x21
    b.ge wr_line
    sub x1, x21, #1

rev_loop:
    cmp x0, x1
    b.ge wr_line
    ldrb w2, [x19, x0]
    ldrb w3, [x19, x1]
    strb w3, [x19, x0]
    strb w2, [x19, x1]
    add x0, x0, #1
    sub x1, x1, #1
    b rev_loop

wr_line:
    sub x2, x21, x22
    cbz x2, wr_nl
    mov x0, #1
    add x1, x19, x22
    mov x8, #64
    svc #0

wr_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #64
    svc #0
    cmp x21, x20
    b.ge done
    add x21, x21, #1
    b line_loop

done:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
nl:
    .byte 10

.bss
buf:
    .space 65536
