.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.text
.p2align 2
.globl _start
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, x19
    mov x0, #BUF_SIZE
    add x21, x19, x0

.Lread_loop:
    cmp x20, x21
    b.hs .Lread_done
    mov x8, #SYS_read
    mov x0, #0
    mov x1, x20
    sub x2, x21, x20
    svc #0
    cmp x0, #0
    b.le .Lread_done
    add x20, x20, x0
    b .Lread_loop

.Lread_done:
    mov x21, x20
    mov x20, x19
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne error_exit

    mov x8, #SYS_write
    mov x0, #1
    adr x1, msg_ok
    mov x2, #3
    svc #0
    mov x8, #SYS_exit
    mov x0, #0
    svc #0
.Lhalt:
    b .Lhalt

.p2align 2
skip_ws:
    cmp x20, x21
    b.hs .Lsw_done
    ldrb w0, [x20]
    cmp w0, #0x20
    b.eq .Lsw_adv
    cmp w0, #0x09
    b.eq .Lsw_adv
    cmp w0, #0x0A
    b.eq .Lsw_adv
    cmp w0, #0x0D
    b.eq .Lsw_adv
.Lsw_done:
    ret
.Lsw_adv:
    add x20, x20, #1
    b skip_ws

.p2align 2
parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #'{'
    b.eq .Lpv_obj
    cmp w0, #'['
    b.eq .Lpv_arr
    cmp w0, #0x22
    b.eq .Lpv_str
    cmp w0, #'-'
    b.eq .Lpv_num
    cmp w0, #'0'
    b.lo .Lpv_lit
    cmp w0, #'9'
    b.ls .Lpv_num
.Lpv_lit:
    cmp w0, #'t'
    b.eq .Lpv_true
    cmp w0, #'f'
    b.eq .Lpv_false
    cmp w0, #'n'
    b.eq .Lpv_null
    b error_exit
.Lpv_obj:
    bl parse_object
    b .Lpv_done
.Lpv_arr:
    bl parse_array
    b .Lpv_done
.Lpv_str:
    bl parse_string
    b .Lpv_done
.Lpv_num:
    bl parse_number
    b .Lpv_done
.Lpv_true:
    adr x1, lit_true
    mov x2, #4
    bl match_str
    b .Lpv_done
.Lpv_false:
    adr x1, lit_false
    mov x2, #5
    bl match_str
    b .Lpv_done
.Lpv_null:
    adr x1, lit_null
    mov x2, #4
    bl match_str
.Lpv_done:
    ldp x29, x30, [sp], #16
    ret

.p2align 2
match_str:
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    ldrb w3, [x1], #1
    cmp w0, w3
    b.ne error_exit
    add x20, x20, #1
    subs x2, x2, #1
    b.ne match_str
    ret

.p2align 2
parse_object:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #'}'
    b.eq .Lobj_end
.Lobj_member:
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #':'
    b.ne error_exit
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #'}'
    b.eq .Lobj_end
    cmp w0, #','
    b.ne error_exit
    add x20, x20, #1
    bl skip_ws
    b .Lobj_member
.Lobj_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

.p2align 2
parse_array:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #']'
    b.eq .Larr_end
.Larr_val:
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #']'
    b.eq .Larr_end
    cmp w0, #','
    b.ne error_exit
    add x20, x20, #1
    bl skip_ws
    b .Larr_val
.Larr_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

.p2align 2
parse_string:
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #0x22
    b.ne error_exit
    add x20, x20, #1
.Lps_loop:
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq .Lps_end
    cmp w0, #0x5C
    b.eq .Lps_esc
    cmp w0, #0x20
    b.lo error_exit
    add x20, x20, #1
    b .Lps_loop
.Lps_end:
    add x20, x20, #1
    ret
.Lps_esc:
    add x20, x20, #1
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #0x22
    b.eq .Lps_esc_ok
    cmp w0, #0x5C
    b.eq .Lps_esc_ok
    cmp w0, #'/'
    b.eq .Lps_esc_ok
    cmp w0, #'b'
    b.eq .Lps_esc_ok
    cmp w0, #'f'
    b.eq .Lps_esc_ok
    cmp w0, #'n'
    b.eq .Lps_esc_ok
    cmp w0, #'r'
    b.eq .Lps_esc_ok
    cmp w0, #'t'
    b.eq .Lps_esc_ok
    cmp w0, #'u'
    b.eq .Lps_uni
    b error_exit
.Lps_esc_ok:
    add x20, x20, #1
    b .Lps_loop
.Lps_uni:
    add x20, x20, #1
    mov x1, #4
.Lps_hex:
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo error_exit
    cmp w0, #'9'
    b.ls .Lps_hex_ok
    cmp w0, #'A'
    b.lo error_exit
    cmp w0, #'F'
    b.ls .Lps_hex_ok
    cmp w0, #'a'
    b.lo error_exit
    cmp w0, #'f'
    b.hi error_exit
.Lps_hex_ok:
    add x20, x20, #1
    subs x1, x1, #1
    b.ne .Lps_hex
    b .Lps_loop

.p2align 2
parse_number:
    ldrb w0, [x20]
    cmp w0, #'-'
    b.ne .Lpn_int
    add x20, x20, #1
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
.Lpn_int:
    cmp w0, #'0'
    b.eq .Lpn_zero
    cmp w0, #'1'
    b.lo error_exit
    cmp w0, #'9'
    b.hi error_exit
    add x20, x20, #1
.Lpn_digits:
    cmp x20, x21
    b.eq .Lpn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo .Lpn_after_int
    cmp w0, #'9'
    b.hi .Lpn_after_int
    add x20, x20, #1
    b .Lpn_digits
.Lpn_zero:
    add x20, x20, #1
    cmp x20, x21
    b.eq .Lpn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo .Lpn_after_int
    cmp w0, #'9'
    b.ls error_exit
.Lpn_after_int:
    cmp w0, #'.'
    b.eq .Lpn_frac
    b .Lpn_check_exp
.Lpn_frac:
    add x20, x20, #1
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo error_exit
    cmp w0, #'9'
    b.hi error_exit
.Lpn_frac_loop:
    add x20, x20, #1
    cmp x20, x21
    b.eq .Lpn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo .Lpn_check_exp
    cmp w0, #'9'
    b.hi .Lpn_check_exp
    b .Lpn_frac_loop
.Lpn_check_exp:
    cmp w0, #'e'
    b.eq .Lpn_exp
    cmp w0, #'E'
    b.ne .Lpn_done
.Lpn_exp:
    add x20, x20, #1
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
    cmp w0, #'+'
    b.eq .Lpn_exp_sign
    cmp w0, #'-'
    b.ne .Lpn_exp_digit
.Lpn_exp_sign:
    add x20, x20, #1
    cmp x20, x21
    b.eq error_exit
    ldrb w0, [x20]
.Lpn_exp_digit:
    cmp w0, #'0'
    b.lo error_exit
    cmp w0, #'9'
    b.hi error_exit
.Lpn_exp_loop:
    add x20, x20, #1
    cmp x20, x21
    b.eq .Lpn_done
    ldrb w0, [x20]
    cmp w0, #'0'
    b.lo .Lpn_done
    cmp w0, #'9'
    b.hi .Lpn_done
    b .Lpn_exp_loop
.Lpn_done:
    ret

.p2align 2
error_exit:
    cmp x20, x21
    csel x20, x21, x20, hi
    sub x22, x20, x19
    mov x8, #SYS_write
    mov x0, #1
    adr x1, msg_err
    mov x2, #9
    svc #0
    sub sp, sp, #16
    add x10, sp, #16
    mov x11, x10
    mov x0, x22
.Litoa:
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #'0'
    sub x11, x11, #1
    strb w5, [x11]
    mov x0, x4
    cbnz x0, .Litoa
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x11
    sub x2, x10, x11
    svc #0
    mov x8, #SYS_write
    mov x0, #1
    adr x1, msg_nl
    mov x2, #1
    svc #0
    mov x8, #SYS_exit
    mov x0, #1
    svc #0
    b .Lhalt

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
msg_nl:
    .ascii "\n"
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.p2align 4
buf:
    .space 65536
