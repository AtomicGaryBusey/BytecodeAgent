.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.text
.p2align 2
.globl _start
_start:
    adrp x19, stdin_buf
    add x19, x19, :lo12:stdin_buf
    mov x20, #0

read_loop:
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x0, #0
find_bang:
    cmp x0, x20
    b.hs no_bang
    ldrb w1, [x19, x0]
    cmp w1, #33
    b.eq found_bang
    add x0, x0, #1
    b find_bang

found_bang:
    mov x21, x0
    add x22, x19, x0
    add x22, x22, #1
    sub x23, x20, x0
    sub x23, x23, #1
    b setup_tables

no_bang:
    mov x21, x20
    add x22, x19, x20
    mov x23, #0

setup_tables:
    adrp x24, match_table
    add x24, x24, :lo12:match_table
    adrp x25, tape
    add x25, x25, :lo12:tape
    mov x2, #0
    mov x3, #0

bracket_loop:
    cmp x2, x21
    b.hs bracket_end
    ldrb w0, [x19, x2]
    cmp w0, #91
    b.eq br_open
    cmp w0, #93
    b.eq br_close
br_cont:
    add x2, x2, #1
    b bracket_loop

br_open:
    str x2, [sp, #-16]!
    add x3, x3, #1
    b br_cont

br_close:
    cbz x3, unbalanced
    ldr x0, [sp], #16
    sub x3, x3, #1
    strh w2, [x24, x0, lsl #1]
    strh w0, [x24, x2, lsl #1]
    b br_cont

bracket_end:
    cbnz x3, unbalanced
    mov x26, #0
    mov x27, #0
    mov x28, #0

exec_loop:
    cmp x27, x21
    b.hs exec_done
    ldrb w0, [x19, x27]
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_inc
    cmp w0, #45
    b.eq cmd_dec
    cmp w0, #46
    b.eq cmd_dot
    cmp w0, #44
    b.eq cmd_comma
    cmp w0, #91
    b.eq cmd_lbrack
    cmp w0, #93
    b.eq cmd_rbrack
next:
    add x27, x27, #1
    b exec_loop

cmd_right:
    add x26, x26, #1
    b next

cmd_left:
    sub x26, x26, #1
    b next

cmd_inc:
    ldrb w0, [x25, x26]
    add w0, w0, #1
    strb w0, [x25, x26]
    b next

cmd_dec:
    ldrb w0, [x25, x26]
    sub w0, w0, #1
    strb w0, [x25, x26]
    b next

cmd_dot:
    mov x0, #1
    add x1, x25, x26
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    b next

cmd_comma:
    cmp x28, x23
    b.hs comma_eof
    ldrb w0, [x22, x28]
    add x28, x28, #1
    strb w0, [x25, x26]
    b next
comma_eof:
    strb wzr, [x25, x26]
    b next

cmd_lbrack:
    ldrb w0, [x25, x26]
    cbnz w0, next
    ldrh w0, [x24, x27, lsl #1]
    mov x27, x0
    b next

cmd_rbrack:
    ldrb w0, [x25, x26]
    cbz w0, next
    ldrh w0, [x24, x27, lsl #1]
    mov x27, x0
    b next

exec_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0
    b exec_done

unbalanced:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0
    b unbalanced

.bss
.balign 8
stdin_buf:
    .space 65536
tape:
    .space 30000
.balign 2
match_table:
    .space 131072
