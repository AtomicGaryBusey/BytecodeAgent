.text
.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0

read_loop:
    cmp x20, #BUFSIZE
    b.hs read_done
    mov x0, #0
    add x1, x19, x20
    mov x2, #BUFSIZE
    sub x2, x2, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

process_loop:
    cmp x21, x20
    b.hs exit_ok
    mov x22, x21

scan:
    cmp x21, x20
    b.hs scan_done
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq scan_done
    add x21, x21, #1
    b scan

scan_done:
    add x0, x19, x22
    add x1, x19, x21
    cmp x0, x1
    b.eq rev_done
    sub x1, x1, #1
rev:
    cmp x0, x1
    b.hs rev_done
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w3, [x0], #1
    strb w2, [x1], #-1
    b rev

rev_done:
    mov w2, #10
    strb w2, [x19, x21]
    mov x0, #1
    add x1, x19, x22
    sub x2, x21, x22
    add x2, x2, #1
    mov x8, #SYS_write
    svc #0
    cmp x21, x20
    b.hs exit_ok
    add x21, x21, #1
    b process_loop

exit_ok:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.bss
.balign 16
buf:
    .space 65537
