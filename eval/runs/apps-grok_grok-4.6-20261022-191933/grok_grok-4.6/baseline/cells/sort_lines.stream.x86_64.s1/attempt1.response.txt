.globl _start
.text
_start:
    lea r12, [rip + buffer]
    mov r13, r12
    mov r14, 32768
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r14
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    sub r14, rax
    jnz read_loop
read_done:
    mov rsi, r13
    mov rdi, r13
    xor r15, r15
parse_loop:
    cmp rdi, r12
    jae parse_end
    mov al, byte ptr [rdi]
    inc rdi
    cmp al, 10
    jne parse_loop
    mov qword ptr [line_ptrs + r15*8], rsi
    mov rax, rdi
    sub rax, rsi
    dec rax
    mov dword ptr [line_lens + r15*4], eax
    inc r15
    mov rsi, rdi
    jmp parse_loop
parse_end:
    cmp rsi, r12
    jae parse_done
    mov qword ptr [line_ptrs + r15*8], rsi
    mov rax, r12
    sub rax, rsi
    mov dword ptr [line_lens + r15*4], eax
    inc r15
parse_done:
    test r15, r15
    jz do_exit
    mov ebx, 1
ms_outer:
    cmp rbx, r15
    jae ms_done
    xor r14, r14
ms_inner:
    cmp r14, r15
    jae ms_next_width
    lea r12, [r14 + rbx]
    cmp r12, r15
    cmovae r12, r15
    lea r13, [r14 + rbx*2]
    cmp r13, r15
    cmovae r13, r15
    mov rcx, r14
    mov rdx, r12
    mov rdi, r14
merge_loop:
    cmp rcx, r12
    jae merge_right_remain
    cmp rdx, r13
    jae merge_left_remain
    mov r8, qword ptr [line_ptrs + rcx*8]
    mov r9, qword ptr [line_ptrs + rdx*8]
    mov r10d, dword ptr [line_lens + rcx*4]
    mov r11d, dword ptr [line_lens + rdx*4]
    mov eax, r10d
    cmp eax, r11d
    cmova eax, r11d
    test eax, eax
    jz cmp_lens
cmp_bytes:
    movzx esi, byte ptr [r8]
    movzx ebp, byte ptr [r9]
    cmp esi, ebp
    jne cmp_bytes_diff
    inc r8
    inc r9
    dec eax
    jnz cmp_bytes
    jmp cmp_lens
cmp_bytes_diff:
    ja take_right
    jmp take_left
cmp_lens:
    cmp r10d, r11d
    ja take_right
take_left:
    mov r8, qword ptr [line_ptrs + rcx*8]
    mov qword ptr [temp_ptrs + rdi*8], r8
    mov r8d, dword ptr [line_lens + rcx*4]
    mov dword ptr [temp_lens + rdi*4], r8d
    inc rcx
    inc rdi
    jmp merge_loop
take_right:
    mov r8, qword ptr [line_ptrs + rdx*8]
    mov qword ptr [temp_ptrs + rdi*8], r8
    mov r8d, dword ptr [line_lens + rdx*4]
    mov dword ptr [temp_lens + rdi*4], r8d
    inc rdx
    inc rdi
    jmp merge_loop
merge_left_remain:
    cmp rcx, r12
    jae merge_copy_back
    mov r8, qword ptr [line_ptrs + rcx*8]
    mov qword ptr [temp_ptrs + rdi*8], r8
    mov r8d, dword ptr [line_lens + rcx*4]
    mov dword ptr [temp_lens + rdi*4], r8d
    inc rcx
    inc rdi
    jmp merge_left_remain
merge_right_remain:
    cmp rdx, r13
    jae merge_copy_back
    mov r8, qword ptr [line_ptrs + rdx*8]
    mov qword ptr [temp_ptrs + rdi*8], r8
    mov r8d, dword ptr [line_lens + rdx*4]
    mov dword ptr [temp_lens + rdi*4], r8d
    inc rdx
    inc rdi
    jmp merge_right_remain
merge_copy_back:
    mov rcx, r14
merge_copy_loop:
    cmp rcx, r13
    jae merge_next
    mov r8, qword ptr [temp_ptrs + rcx*8]
    mov qword ptr [line_ptrs + rcx*8], r8
    mov r8d, dword ptr [temp_lens + rcx*4]
    mov dword ptr [line_lens + rcx*4], r8d
    inc rcx
    jmp merge_copy_loop
merge_next:
    mov r14, r13
    jmp ms_inner
ms_next_width:
    shl rbx, 1
    jmp ms_outer
ms_done:
    xor ebx, ebx
print_loop:
    cmp rbx, r15
    jae do_exit
    mov eax, 1
    mov edi, 1
    mov rsi, qword ptr [line_ptrs + rbx*8]
    mov edx, dword ptr [line_lens + rbx*4]
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rbx
    jmp print_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 8
buffer:
    .space 32768
.align 8
line_ptrs:
    .space 8000
temp_ptrs:
    .space 8000
line_lens:
    .space 4000
temp_lens:
    .space 4000
