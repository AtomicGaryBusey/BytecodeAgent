.globl _start

.text

_start:
    lea r12, [rip + input_buf]
    mov r13, r12

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    lea rdx, [rip + input_buf]
    add rdx, 65536
    sub rdx, r13
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    mov r14, r12

line_loop:
    cmp r14, r13
    jae all_done
    mov r15, r14

find_nl:
    cmp r14, r13
    jae last_line
    cmp byte ptr [r14], 10
    je have_nl
    inc r14
    jmp find_nl

have_nl:
    call eval_line
    inc r14
    jmp line_loop

last_line:
    call eval_line

all_done:
    mov eax, 60
    xor edi, edi
    syscall

eval_line:
    lea rbx, [rip + rpn_stack]
    mov r9, r15

skip_ws:
    cmp r9, r14
    jae tokens_done
    movzx eax, byte ptr [r9]
    cmp al, 32
    je is_ws
    cmp al, 9
    jne found_tok
is_ws:
    inc r9
    jmp skip_ws

found_tok:
    mov r10, r9
scan_tok:
    inc r9
    cmp r9, r14
    jae have_tok
    movzx eax, byte ptr [r9]
    cmp al, 32
    je have_tok
    cmp al, 9
    je have_tok
    jmp scan_tok

have_tok:
    mov rcx, r9
    sub rcx, r10
    cmp rcx, 1
    jne try_num
    mov al, byte ptr [r10]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod

try_num:
    mov rdi, r10
    xor r8d, r8d
    cmp byte ptr [rdi], '-'
    jne num_digits
    inc rdi
    mov r8d, 1
num_digits:
    cmp rdi, r9
    jae line_error
    mov rcx, r9
    sub rcx, rdi
    cmp rcx, 18
    ja line_error
    xor eax, eax
num_loop:
    movzx ecx, byte ptr [rdi]
    sub ecx, '0'
    cmp ecx, 9
    ja line_error
    imul rax, 10
    add rax, rcx
    inc rdi
    cmp rdi, r9
    jb num_loop
    test r8d, r8d
    jz num_push
    neg rax
num_push:
    mov qword ptr [rbx], rax
    add rbx, 8
    jmp skip_ws

op_add:
    lea rax, [rip + rpn_stack]
    add rax, 16
    cmp rbx, rax
    jb line_error
    sub rbx, 8
    mov rax, qword ptr [rbx - 8]
    add rax, qword ptr [rbx]
    mov qword ptr [rbx - 8], rax
    jmp skip_ws

op_sub:
    lea rax, [rip + rpn_stack]
    add rax, 16
    cmp rbx, rax
    jb line_error
    sub rbx, 8
    mov rax, qword ptr [rbx - 8]
    sub rax, qword ptr [rbx]
    mov qword ptr [rbx - 8], rax
    jmp skip_ws

op_mul:
    lea rax, [rip + rpn_stack]
    add rax, 16
    cmp rbx, rax
    jb line_error
    sub rbx, 8
    mov rax, qword ptr [rbx - 8]
    imul rax, qword ptr [rbx]
    mov qword ptr [rbx - 8], rax
    jmp skip_ws

op_div:
    lea rax, [rip + rpn_stack]
    add rax, 16
    cmp rbx, rax
    jb line_error
    mov rcx, qword ptr [rbx - 8]
    test rcx, rcx
    jz line_error
    mov rax, qword ptr [rbx - 16]
    mov r8, 1
    shl r8, 63
    cmp rax, r8
    jne div_go
    cmp rcx, -1
    je line_error
div_go:
    cqo
    idiv rcx
    sub rbx, 8
    mov qword ptr [rbx - 8], rax
    jmp skip_ws

op_mod:
    lea rax, [rip + rpn_stack]
    add rax, 16
    cmp rbx, rax
    jb line_error
    mov rcx, qword ptr [rbx - 8]
    test rcx, rcx
    jz line_error
    mov rax, qword ptr [rbx - 16]
    mov r8, 1
    shl r8, 63
    cmp rax, r8
    jne mod_go
    cmp rcx, -1
    je line_error
mod_go:
    cqo
    idiv rcx
    sub rbx, 8
    mov qword ptr [rbx - 8], rdx
    jmp skip_ws

tokens_done:
    lea rax, [rip + rpn_stack]
    add rax, 8
    cmp rbx, rax
    jne line_error
    mov rax, qword ptr [rip + rpn_stack]
    lea rdi, [rip + numbuf]
    add rdi, 31
    mov byte ptr [rdi], 10
    mov r8, rax
    test rax, rax
    jns abs_ready
    neg rax
abs_ready:
    test rax, rax
    jnz conv_loop
    dec rdi
    mov byte ptr [rdi], '0'
    jmp put_sign
conv_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz conv_loop
put_sign:
    test r8, r8
    jns do_write
    dec rdi
    mov byte ptr [rdi], '-'
do_write:
    lea rdx, [rip + numbuf]
    add rdx, 32
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    ret

line_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_str]
    mov edx, 6
    syscall
    ret

.section .rodata
error_str:
    .ascii "error\n"

.bss
.balign 8
input_buf:
    .zero 65536
.balign 8
rpn_stack:
    .zero 8192
numbuf:
    .zero 32
