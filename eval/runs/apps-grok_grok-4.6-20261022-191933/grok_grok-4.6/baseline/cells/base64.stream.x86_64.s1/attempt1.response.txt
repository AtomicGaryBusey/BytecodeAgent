.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    lea r12, [rip + input_buf]
    mov r13, r12
    mov r14, 65536

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r14
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    sub r14, rax
    jnz read_loop

read_done:
    mov r14, r12
    sub r14, r13
    mov rsi, r13
    lea rdi, [rip + output_buf]
    lea r15, [rip + alphabet]
    xor r8, r8
    test r14, r14
    jz finish_encode

encode_loop:
    cmp r14, 3
    jb encode_tail
    movzx r9d, byte ptr [rsi]
    movzx r10d, byte ptr [rsi + 1]
    movzx r11d, byte ptr [rsi + 2]
    add rsi, 3
    sub r14, 3
    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [r15 + rax]
    call emit
    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov edx, r10d
    shr edx, 4
    or eax, edx
    mov al, byte ptr [r15 + rax]
    call emit
    mov eax, r10d
    and eax, 15
    shl eax, 2
    mov edx, r11d
    shr edx, 6
    or eax, edx
    mov al, byte ptr [r15 + rax]
    call emit
    mov eax, r11d
    and eax, 63
    mov al, byte ptr [r15 + rax]
    call emit
    jmp encode_loop

encode_tail:
    test r14, r14
    jz finish_encode
    cmp r14, 1
    je tail_1
    movzx r9d, byte ptr [rsi]
    movzx r10d, byte ptr [rsi + 1]
    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [r15 + rax]
    call emit
    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov edx, r10d
    shr edx, 4
    or eax, edx
    mov al, byte ptr [r15 + rax]
    call emit
    mov eax, r10d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [r15 + rax]
    call emit
    mov al, '='
    call emit
    jmp finish_encode

tail_1:
    movzx r9d, byte ptr [rsi]
    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [r15 + rax]
    call emit
    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov al, byte ptr [r15 + rax]
    call emit
    mov al, '='
    call emit
    mov al, '='
    call emit

finish_encode:
    test r8, r8
    jz do_write
    mov byte ptr [rdi], 10
    inc rdi

do_write:
    mov rdx, rdi
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    jz do_exit
    mov eax, 1
    mov edi, 1
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r8, r8
1:
    ret

.bss
.align 16
input_buf:
    .space 65536
output_buf:
    .space 131072
