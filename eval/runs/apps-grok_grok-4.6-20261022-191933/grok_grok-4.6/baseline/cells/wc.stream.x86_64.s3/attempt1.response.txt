.globl _start

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle do_print
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buffer]

process:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je 2f
    sub al, 9
    cmp al, 4
    jbe 2f
    test r15, r15
    jnz 3f
    inc r13
    mov r15, 1
    jmp 3f
2:
    xor r15, r15
3:
    dec rcx
    jnz process
    jmp read_loop

do_print:
    sub rsp, 64
    lea rdi, [rsp + 63]
    mov byte ptr [rdi], 10
    mov rax, r14
    call u64_to_dec
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r13
    call u64_to_dec
    dec rdi
    mov byte ptr [rdi], 32
    mov rax, r12
    call u64_to_dec
    mov rsi, rdi
    lea rdx, [rsp + 64]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

u64_to_dec:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz 1b
    ret

.bss
buffer:
    .space 65536
