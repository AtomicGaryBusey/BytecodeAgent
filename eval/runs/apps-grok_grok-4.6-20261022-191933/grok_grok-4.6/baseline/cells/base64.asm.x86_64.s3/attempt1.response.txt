.globl _start

.text
_start:
    lea rbx, [rip + inbuf]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    cmp r12, 65536
    jb read_loop
read_done:
    mov rsi, rbx
    lea rdi, [rip + outbuf]
    xor r14, r14
    lea r15, [rip + alphabet]
encode_loop:
    cmp r12, 3
    jb encode_tail
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]
    add rsi, 3
    sub r12, 3
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov ecx, r10d
    shr ecx, 6
    or eax, ecx
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov eax, r10d
    and eax, 63
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    jmp encode_loop
encode_tail:
    test r12, r12
    jz encode_done
    cmp r12, 1
    je encode_one
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov ecx, r9d
    shr ecx, 4
    or eax, ecx
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov eax, r9d
    and eax, 15
    shl eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov al, '='
    call emit_char
    jmp encode_done
encode_one:
    movzx r8d, byte ptr [rsi]
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov eax, r8d
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [r15 + rax]
    call emit_char
    mov al, '='
    call emit_char
    mov al, '='
    call emit_char
encode_done:
    test r14, r14
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi
write_out:
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    jz do_exit
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne emit_done
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
emit_done:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 131072
