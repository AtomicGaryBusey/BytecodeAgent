.globl _start

.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFSIZE, 32768

.text
_start:
    lea r12, [rip + input_buf]
    mov rbx, r12
    mov r13d, BUFSIZE

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov rdx, r13
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    sub r13, rax
    jnz read_loop

read_done:
    xor r14, r14
    mov r13, r12

parse_loop:
    cmp r13, rbx
    jae parse_done
    mov r15, r13

parse_scan:
    cmp r13, rbx
    jae parse_last
    cmp byte ptr [r13], 10
    je parse_found
    inc r13
    jmp parse_scan

parse_found:
    mov rax, r13
    sub rax, r15
    lea rcx, [rip + line_ptrs]
    mov qword ptr [rcx + r14*8], r15
    lea rcx, [rip + line_lens]
    mov qword ptr [rcx + r14*8], rax
    inc r14
    inc r13
    jmp parse_loop

parse_last:
    mov rax, r13
    sub rax, r15
    lea rcx, [rip + line_ptrs]
    mov qword ptr [rcx + r14*8], r15
    lea rcx, [rip + line_lens]
    mov qword ptr [rcx + r14*8], rax
    inc r14

parse_done:
    mov r8, 1
    jmp sort_outer_cmp

sort_outer:
    lea rax, [rip + line_ptrs]
    mov r9, qword ptr [rax + r8*8]
    lea rax, [rip + line_lens]
    mov r10, qword ptr [rax + r8*8]
    mov r11, r8

sort_inner:
    test r11, r11
    jz sort_place
    lea rax, [rip + line_ptrs]
    mov rsi, qword ptr [rax + r11*8 - 8]
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + r11*8 - 8]
    call compare_lines
    test rax, rax
    jle sort_place
    lea rax, [rip + line_ptrs]
    mov rcx, qword ptr [rax + r11*8 - 8]
    mov qword ptr [rax + r11*8], rcx
    lea rax, [rip + line_lens]
    mov rcx, qword ptr [rax + r11*8 - 8]
    mov qword ptr [rax + r11*8], rcx
    dec r11
    jmp sort_inner

sort_place:
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + r11*8], r9
    lea rax, [rip + line_lens]
    mov qword ptr [rax + r11*8], r10
    inc r8

sort_outer_cmp:
    cmp r8, r14
    jb sort_outer

    xor r8, r8
    jmp print_cmp

print_loop:
    lea rax, [rip + line_ptrs]
    mov rsi, qword ptr [rax + r8*8]
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + r8*8]
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    inc r8

print_cmp:
    cmp r8, r14
    jb print_loop

    mov eax, SYS_EXIT
    xor edi, edi
    syscall

compare_lines:
    xor ecx, ecx
    mov rax, rdx
    cmp rax, r10
    cmova rax, r10
cmp_loop:
    cmp rcx, rax
    jae cmp_len
    movzx edi, byte ptr [rsi + rcx]
    cmp dil, byte ptr [r9 + rcx]
    jne cmp_diff
    inc rcx
    jmp cmp_loop
cmp_diff:
    movzx eax, byte ptr [rsi + rcx]
    movzx edi, byte ptr [r9 + rcx]
    sub rax, rdi
    ret
cmp_len:
    mov rax, rdx
    sub rax, r10
    ret

.section .rodata
nl:
    .byte 10

.bss
.align 8
input_buf:
    .space 32768
.align 8
line_ptrs:
    .space 8000
.align 8
line_lens:
    .space 8000
