.globl _start

.text
.p2align 2
_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    mov     x20, x19
.Lread:
    adrp    x1, input_buf_end
    add     x1, x1, :lo12:input_buf_end
    sub     x2, x1, x20
    cbz     x2, .Lread_done
    mov     x0, #0
    mov     x1, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    .Lread_done
    add     x20, x20, x0
    b       .Lread
.Lread_done:
    mov     x21, x20
    mov     x20, x19
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.ne    error_here
print_ok:
    mov     x0, #1
    adrp    x1, msg_ok
    add     x1, x1, :lo12:msg_ok
    mov     x2, #3
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0
.Lhalt:
    b       .Lhalt

skip_ws:
    cmp     x20, x21
    b.hs    .Lskip_done
    ldrb    w0, [x20]
    cmp     w0, #0x20
    b.eq    .Lskip_adv
    cmp     w0, #0x09
    b.eq    .Lskip_adv
    cmp     w0, #0x0a
    b.eq    .Lskip_adv
    cmp     w0, #0x0d
    b.eq    .Lskip_adv
.Lskip_done:
    ret
.Lskip_adv:
    add     x20, x20, #1
    b       skip_ws

parse_value:
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #'{'
    b.eq    do_parse_object
    cmp     w0, #'['
    b.eq    do_parse_array
    cmp     w0, #0x22
    b.eq    do_parse_string
    cmp     w0, #'-'
    b.eq    do_parse_number
    cmp     w0, #'0'
    b.lo    .Lpv_kw
    cmp     w0, #'9'
    b.ls    do_parse_number
.Lpv_kw:
    cmp     w0, #'t'
    b.eq    parse_true
    cmp     w0, #'f'
    b.eq    parse_false
    cmp     w0, #'n'
    b.eq    parse_null
    b       error_here

parse_true:
    adrp    x1, lit_true
    add     x1, x1, :lo12:lit_true
    mov     x2, #4
    b       match_lit

parse_false:
    adrp    x1, lit_false
    add     x1, x1, :lo12:lit_false
    mov     x2, #5
    b       match_lit

parse_null:
    adrp    x1, lit_null
    add     x1, x1, :lo12:lit_null
    mov     x2, #4
    b       match_lit

match_lit:
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    ldrb    w3, [x1], #1
    cmp     w0, w3
    b.ne    error_here
    add     x20, x20, #1
    subs    x2, x2, #1
    b.ne    match_lit
    ret

do_parse_object:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #'}'
    b.ne    .Lobj_member
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
.Lobj_member:
    bl      do_parse_string
    bl      skip_ws
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #':'
    b.ne    error_here
    add     x20, x20, #1
    bl      skip_ws
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #','
    b.eq    .Lobj_comma
    cmp     w0, #'}'
    b.ne    error_here
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
.Lobj_comma:
    add     x20, x20, #1
    bl      skip_ws
    b       .Lobj_member

do_parse_array:
    stp     x29, x30, [sp, #-16]!
    add     x20, x20, #1
    bl      skip_ws
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #']'
    b.ne    .Larr_value
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
.Larr_value:
    bl      parse_value
    bl      skip_ws
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #','
    b.eq    .Larr_comma
    cmp     w0, #']'
    b.ne    error_here
    add     x20, x20, #1
    ldp     x29, x30, [sp], #16
    ret
.Larr_comma:
    add     x20, x20, #1
    bl      skip_ws
    b       .Larr_value

do_parse_string:
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #0x22
    b.ne    error_here
    add     x20, x20, #1
.Lstr_loop:
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    add     x20, x20, #1
    cmp     w0, #0x22
    b.eq    .Lstr_done
    cmp     w0, #0x5c
    b.eq    .Lstr_esc
    cmp     w0, #0x20
    b.lo    error_prev
    b       .Lstr_loop
.Lstr_done:
    ret
.Lstr_esc:
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    add     x20, x20, #1
    cmp     w0, #0x22
    b.eq    .Lstr_loop
    cmp     w0, #0x5c
    b.eq    .Lstr_loop
    cmp     w0, #'/'
    b.eq    .Lstr_loop
    cmp     w0, #'b'
    b.eq    .Lstr_loop
    cmp     w0, #'f'
    b.eq    .Lstr_loop
    cmp     w0, #'n'
    b.eq    .Lstr_loop
    cmp     w0, #'r'
    b.eq    .Lstr_loop
    cmp     w0, #'t'
    b.eq    .Lstr_loop
    cmp     w0, #'u'
    b.eq    .Lstr_uni
    b       error_prev
.Lstr_uni:
    mov     w4, #4
.Lstr_uni_loop:
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    add     x20, x20, #1
    sub     w1, w0, #'0'
    cmp     w1, #9
    b.ls    .Lstr_uni_ok
    sub     w1, w0, #'A'
    cmp     w1, #5
    b.ls    .Lstr_uni_ok
    sub     w1, w0, #'a'
    cmp     w1, #5
    b.ls    .Lstr_uni_ok
    b       error_prev
.Lstr_uni_ok:
    subs    w4, w4, #1
    b.ne    .Lstr_uni_loop
    b       .Lstr_loop

do_parse_number:
    ldrb    w0, [x20]
    cmp     w0, #'-'
    b.ne    .Lnum_int
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
.Lnum_int:
    cmp     w0, #'0'
    b.eq    .Lnum_zero
    sub     w1, w0, #'1'
    cmp     w1, #8
    b.hi    error_here
    add     x20, x20, #1
.Lnum_intloop:
    cmp     x20, x21
    b.hs    .Lnum_after_int
    ldrb    w0, [x20]
    sub     w1, w0, #'0'
    cmp     w1, #9
    b.hi    .Lnum_after_int
    add     x20, x20, #1
    b       .Lnum_intloop
.Lnum_zero:
    add     x20, x20, #1
.Lnum_after_int:
    cmp     x20, x21
    b.hs    .Lnum_done
    ldrb    w0, [x20]
    cmp     w0, #'.'
    b.eq    .Lnum_frac
.Lnum_check_exp:
    cmp     x20, x21
    b.hs    .Lnum_done
    ldrb    w0, [x20]
    cmp     w0, #'e'
    b.eq    .Lnum_exp
    cmp     w0, #'E'
    b.eq    .Lnum_exp
.Lnum_done:
    ret
.Lnum_frac:
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    sub     w1, w0, #'0'
    cmp     w1, #9
    b.hi    error_here
    add     x20, x20, #1
.Lfrac_loop:
    cmp     x20, x21
    b.hs    .Lnum_done
    ldrb    w0, [x20]
    sub     w1, w0, #'0'
    cmp     w1, #9
    b.hi    .Lnum_check_exp
    add     x20, x20, #1
    b       .Lfrac_loop
.Lnum_exp:
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
    cmp     w0, #'+'
    b.eq    .Lexp_sign
    cmp     w0, #'-'
    b.ne    .Lexp_need_digit
.Lexp_sign:
    add     x20, x20, #1
    cmp     x20, x21
    b.hs    error_eof
    ldrb    w0, [x20]
.Lexp_need_digit:
    sub     w1, w0, #'0'
    cmp     w1, #9
    b.hi    error_here
    add     x20, x20, #1
.Lexp_loop:
    cmp     x20, x21
    b.hs    .Lnum_done
    ldrb    w0, [x20]
    sub     w1, w0, #'0'
    cmp     w1, #9
    b.hi    .Lnum_done
    add     x20, x20, #1
    b       .Lexp_loop

error_prev:
    sub     x20, x20, #1
    b       error_here
error_eof:
    mov     x20, x21
error_here:
    sub     x22, x20, x19
    mov     x0, #1
    adrp    x1, msg_err
    add     x1, x1, :lo12:msg_err
    mov     x2, #9
    mov     x8, #64
    svc     #0
    sub     sp, sp, #16
    add     x1, sp, #14
    mov     w2, #0x0a
    strb    w2, [sp, #15]
    mov     x0, x22
    mov     x3, #10
.Litoa:
    udiv    x4, x0, x3
    msub    x5, x4, x3, x0
    add     w5, w5, #0x30
    strb    w5, [x1]
    sub     x1, x1, #1
    mov     x0, x4
    cbnz    x0, .Litoa
    add     x1, x1, #1
    add     x2, sp, #16
    sub     x2, x2, x1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    mov     x0, #1
    mov     x8, #93
    svc     #0
    b       .Lhalt

.section .rodata
