.globl _start

.equ SYS_read, 0
.equ SYS_write, 1
.equ SYS_exit, 60
.equ BUFLEN, 8192

.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, BUFLEN
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]

process:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 0x20
    je is_ws
    cmp al, 0x09
    jb is_word
    cmp al, 0x0D
    ja is_word
is_ws:
    cmp al, 0x0A
    jne clear_word
    inc r12
clear_word:
    xor r15, r15
    jmp proc_next
is_word:
    test r15, r15
    jnz proc_next
    inc r13
    mov r15, 1
proc_next:
    dec rcx
    jnz process
    jmp read_loop

read_done:
    lea rdi, [rip + outend]
    dec rdi
    mov byte ptr [rdi], 10
    mov rax, r14
    call utoa
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r13
    call utoa
    dec rdi
    mov byte ptr [rdi], 0x20
    mov rax, r12
    call utoa
    mov rsi, rdi
    lea rdx, [rip + outend]
    sub rdx, rsi
    mov eax, SYS_write
    mov edi, 1
    syscall
    mov eax, SYS_exit
    xor edi, edi
    syscall

utoa:
    mov r8, 10
    test rax, rax
    jnz utoa_loop
    dec rdi
    mov byte ptr [rdi], '0'
    ret
utoa_loop:
    xor rdx, rdx
    div r8
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz utoa_loop
    ret

.bss
buf:
    .space 8192
    .space 32
outend:
