.globl _start
.text
_start:
	lea rbx, [rip + input_buf]
	mov rbp, rbx
	add rbp, 65536
read_loop:
	mov rdx, rbp
	sub rdx, rbx
	jz read_done
	xor eax, eax
	xor edi, edi
	mov rsi, rbx
	syscall
	test rax, rax
	jle read_done
	add rbx, rax
	jmp read_loop
read_done:
	mov r12, rbx
	lea r14, [rip + input_buf]
process_lines:
	cmp r14, r12
	jae finish
	mov r15, r14
find_nl:
	cmp r15, r12
	jae found_end
	cmp byte ptr [r15], 0x0A
	je found_end
	inc r15
	jmp find_nl
found_end:
	lea r13, [rip + rpn_stack]
	mov rbx, r13
	mov rbp, r14
token_loop:
skip_sep:
	cmp rbp, r15
	jae tokens_done
	movzx eax, byte ptr [rbp]
	cmp al, 0x20
	je 1f
	cmp al, 0x09
	jne have_token
1:
	inc rbp
	jmp skip_sep
have_token:
	mov rsi, rbp
scan_token:
	inc rbp
	cmp rbp, r15
	jae token_got
	movzx eax, byte ptr [rbp]
	cmp al, 0x20
	je token_got
	cmp al, 0x09
	jne scan_token
token_got:
	mov rcx, rbp
	sub rcx, rsi
	cmp rcx, 1
	jne try_number
	movzx eax, byte ptr [rsi]
	cmp al, '+'
	jne 1f
	xor r8, r8
	jmp do_binop
1:
	cmp al, '-'
	jne 1f
	mov r8, 1
	jmp do_binop
1:
	cmp al, '*'
	jne 1f
	mov r8, 2
	jmp do_binop
1:
	cmp al, '/'
	jne 1f
	mov r8, 3
	jmp do_binop
1:
	cmp al, '%'
	jne try_number
	mov r8, 4
	jmp do_binop
try_number:
	xor r8, r8
	xor eax, eax
	mov rdi, rsi
	cmp byte ptr [rdi], '-'
	jne 1f
	inc rdi
	mov r8, 1
1:
	mov rdx, rbp
	sub rdx, rdi
	cmp rdx, 1
	jl do_error
	cmp rdx, 18
	jg do_error
2:
	movzx ecx, byte ptr [rdi]
	sub ecx, '0'
	cmp ecx, 9
	ja do_error
	imul rax, 10
	add rax, rcx
	inc rdi
	cmp rdi, rbp
	jb 2b
	test r8, r8
	jz 3f
	neg rax
3:
	lea rdx, [r13 + 8192]
	cmp rbx, rdx
	jae do_error
	mov qword ptr [rbx], rax
	add rbx, 8
	jmp token_loop
do_binop:
	mov rax, rbx
	sub rax, r13
	cmp rax, 16
	jb do_error
	mov rcx, qword ptr [rbx - 8]
	mov rax, qword ptr [rbx - 16]
	sub rbx, 16
	cmp r8, 3
	jae op_divmod
	cmp r8, 1
	jb op_add
	je op_sub
	imul rax, rcx
	jmp push_result
op_add:
	add rax, rcx
	jmp push_result
op_sub:
	sub rax, rcx
	jmp push_result
op_divmod:
	test rcx, rcx
	jz do_error
	cmp rcx, -1
	jne 1f
	mov r9, 0x8000000000000000
	cmp rax, r9
	je do_error
1:
	cqo
	idiv rcx
	cmp r8, 3
	je push_result
	mov rax, rdx
push_result:
	mov qword ptr [rbx], rax
	add rbx, 8
	jmp token_loop
tokens_done:
	lea rax, [r13 + 8]
	cmp rbx, rax
	jne do_error
	mov rax, qword ptr [rbx - 8]
	call print_num
	jmp next_line
do_error:
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + err_msg]
	mov edx, 6
	syscall
next_line:
	mov r14, r15
	cmp r14, r12
	jae finish
	inc r14
	jmp process_lines
finish:
	mov eax, 60
	xor edi, edi
	syscall
print_num:
	mov r8, rax
	lea rdi, [rip + num_buf + 31]
	mov byte ptr [rdi], 0x0A
	test rax, rax
	jns 1f
	neg rax
1:
	mov ecx, 10
2:
	xor rdx, rdx
	div rcx
	add edx, '0'
	dec rdi
	mov byte ptr [rdi], dl
	test rax, rax
	jnz 2b
	test r8, r8
	jns 3f
	dec rdi
	mov byte ptr [rdi], '-'
3:
	lea rdx, [rip + num_buf + 32]
	sub rdx, rdi
	mov rsi, rdi
	mov eax, 1
	mov edi, 1
	syscall
	ret

.section .rodata
err_msg:
	.ascii "error\n"

.bss
.align 8
input_buf:
	.space 65536
.align 8
rpn_stack:
	.space 8192
num_buf:
	.space 32
