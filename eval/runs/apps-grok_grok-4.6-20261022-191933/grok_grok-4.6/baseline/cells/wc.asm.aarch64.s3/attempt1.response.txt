.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFSIZE, 4096

.text
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x8, #SYS_READ
    mov x0, #0
    mov x1, x23
    mov x2, #BUFSIZE
    svc #0

    cmp x0, #0
    b.le done_read

    add x21, x21, x0

    mov x1, x23
    mov x2, x0
process:
    ldrb w3, [x1], #1

    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws

    cbnz x22, next
    add x20, x20, #1
    mov x22, #1
    b next
is_ws:
    mov x22, #0
next:
    subs x2, x2, #1
    b.ne process
    b read_loop

done_read:
    sub sp, sp, #64
    add x1, sp, #64

    mov w3, #10
    sub x1, x1, #1
    strb w3, [x1]

    mov x0, x21
    bl utoa

    mov w3, #32
    sub x1, x1, #1
    strb w3, [x1]

    mov x0, x20
    bl utoa

    mov w3, #32
    sub x1, x1, #1
    strb w3, [x1]

    mov x0, x19
    bl utoa

    add x2, sp, #64
    sub x2, x2, x1

    mov x8, #SYS_WRITE
    mov x0, #1
    svc #0

    mov x8, #SYS_EXIT
    mov x0, #0
    svc #0

utoa:
    mov x5, #10
utoa_loop:
    udiv x3, x0, x5
    msub x4, x3, x5, x0
    add w4, w4, #48
    sub x1, x1, #1
    strb w4, [x1]
    mov x0, x3
    cbnz x0, utoa_loop
    ret

.bss
buf:
    .space 4096
