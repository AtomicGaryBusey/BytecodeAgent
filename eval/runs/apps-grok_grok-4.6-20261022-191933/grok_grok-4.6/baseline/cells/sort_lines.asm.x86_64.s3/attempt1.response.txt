.globl _start

.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ BUFSIZE, 32768

.text
_start:
    lea rbx, [rip + buffer]
    xor r12, r12
    mov r13d, BUFSIZE

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    sub r13, rax
    jnz read_loop

read_done:
    xor ecx, ecx
    xor r15, r15
    xor ebp, ebp

parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [rbx + rcx], 10
    jne parse_cont
    lea rax, [rbx + r15]
    lea rdx, [rip + line_ptrs]
    mov qword ptr [rdx + rbp*8], rax
    mov rax, rcx
    sub rax, r15
    lea rdx, [rip + line_lens]
    mov qword ptr [rdx + rbp*8], rax
    inc ebp
    lea r15, [rcx + 1]
parse_cont:
    inc rcx
    jmp parse_loop

parse_end:
    cmp r15, r12
    jae parse_done
    lea rax, [rbx + r15]
    lea rdx, [rip + line_ptrs]
    mov qword ptr [rdx + rbp*8], rax
    mov rax, r12
    sub rax, r15
    lea rdx, [rip + line_lens]
    mov qword ptr [rdx + rbp*8], rax
    inc ebp

parse_done:
    mov r12d, 1
sort_outer:
    cmp r12, rbp
    jae sort_done
    lea rdx, [rip + line_ptrs]
    mov r13, qword ptr [rdx + r12*8]
    lea rdx, [rip + line_lens]
    mov r14, qword ptr [rdx + r12*8]
    mov r15, r12
sort_inner:
    test r15, r15
    jz sort_place
    lea rdx, [rip + line_ptrs]
    mov rdi, qword ptr [rdx + r15*8 - 8]
    lea rdx, [rip + line_lens]
    mov rsi, qword ptr [rdx + r15*8 - 8]
    mov rdx, r13
    mov rcx, r14
    call compare
    test rax, rax
    jle sort_place
    lea rdx, [rip + line_ptrs]
    mov rax, qword ptr [rdx + r15*8 - 8]
    mov qword ptr [rdx + r15*8], rax
    lea rdx, [rip + line_lens]
    mov rax, qword ptr [rdx + r15*8 - 8]
    mov qword ptr [rdx + r15*8], rax
    dec r15
    jmp sort_inner
sort_place:
    lea rdx, [rip + line_ptrs]
    mov qword ptr [rdx + r15*8], r13
    lea rdx, [rip + line_lens]
    mov qword ptr [rdx + r15*8], r14
    inc r12
    jmp sort_outer

sort_done:
    xor r12, r12
print_loop:
    cmp r12, rbp
    jae print_done
    lea rdx, [rip + line_ptrs]
    mov rsi, qword ptr [rdx + r12*8]
    lea rdx, [rip + line_lens]
    mov rdx, qword ptr [rdx + r12*8]
    test rdx, rdx
    jz print_nl
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
print_nl:
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r12
    jmp print_loop

print_done:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

compare:
    mov r8, rsi
    cmp r8, rcx
    cmova r8, rcx
    xor r9, r9
cmp_loop:
    cmp r9, r8
    jae cmp_len
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    sub eax, r10d
    jnz cmp_diff
    inc r9
    jmp cmp_loop
cmp_diff:
    movsxd rax, eax
    ret
cmp_len:
    mov rax, rsi
    sub rax, rcx
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 8
buffer:
    .space 32768
line_ptrs:
    .space 8000
line_lens:
    .space 8000
