.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.text

print_hex8:
    mov w3, #32
1:
    sub w3, w3, #4
    lsr w2, w0, w3
    and w2, w2, #15
    ldrb w2, [x22, x2]
    strb w2, [x1], #1
    cbnz w3, 1b
    ret

_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    mov x0, #0
    add x1, x19, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, hex_table
    add x22, x22, :lo12:hex_table
    adrp x23, linebuf
    add x23, x23, :lo12:linebuf
    mov x21, #0

dump_loop:
    cmp x21, x20
    b.hs dump_done
    sub x25, x20, x21
    cmp x25, #16
    mov x26, #16
    csel x25, x25, x26, lo
    add x28, x19, x21
    mov x1, x23
    mov w0, w21
    bl print_hex8
    mov w0, #0x20
    strb w0, [x1], #1
    strb w0, [x1], #1
    mov x27, #0

hex_loop:
    cmp x27, #8
    b.ne 1f
    mov w0, #0x20
    strb w0, [x1], #1
1:
    cmp x27, x25
    b.hs 2f
    ldrb w0, [x28, x27]
    lsr w2, w0, #4
    ldrb w2, [x22, x2]
    strb w2, [x1], #1
    and w2, w0, #15
    ldrb w2, [x22, x2]
    strb w2, [x1], #1
    mov w0, #0x20
    strb w0, [x1], #1
    b 3f
2:
    mov w0, #0x20
    strb w0, [x1], #1
    strb w0, [x1], #1
    strb w0, [x1], #1
3:
    add x27, x27, #1
    cmp x27, #16
    b.lo hex_loop
    mov w0, #0x20
    strb w0, [x1], #1
    mov w0, #0x7c
    strb w0, [x1], #1
    mov x27, #0

ascii_loop:
    cmp x27, x25
    b.eq ascii_done
    ldrb w0, [x28, x27]
    sub w2, w0, #32
    cmp w2, #95
    mov w3, #0x2e
    csel w0, w0, w3, lo
    strb w0, [x1], #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w0, #0x7c
    strb w0, [x1], #1
    mov w0, #0x0a
    strb w0, [x1], #1
    sub x2, x1, x23
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x23
    svc #0
    add x21, x21, x25
    b dump_loop

dump_done:
    mov w0, w20
    mov x1, x23
    bl print_hex8
    mov w0, #0x0a
    strb w0, [x23, #8]
    mov x8, #SYS_write
    mov x0, #1
    mov x1, x23
    mov x2, #9
    svc #0
    mov x8, #SYS_exit
    mov x0, #0
    svc #0

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .space 65536
linebuf:
    .space 80
