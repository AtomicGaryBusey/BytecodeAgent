.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 32768

.globl _start
.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    adrp x21, lines
    add x21, x21, :lo12:lines
    adrp x23, lens
    add x23, x23, :lo12:lens

    mov x20, #0
read_loop:
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    ble read_done
    add x20, x20, x0
    b read_loop
read_done:

    mov x22, #0
    mov x24, #0
    mov x25, #0
parse_loop:
    cmp x24, x20
    bge parse_tail
    ldrb w0, [x19, x24]
    cmp w0, #10
    bne parse_next
    add x0, x19, x25
    str x0, [x21, x22, lsl #3]
    sub x0, x24, x25
    str x0, [x23, x22, lsl #3]
    add x22, x22, #1
    add x25, x24, #1
parse_next:
    add x24, x24, #1
    b parse_loop
parse_tail:
    cmp x25, x20
    bge parse_done
    add x0, x19, x25
    str x0, [x21, x22, lsl #3]
    sub x0, x20, x25
    str x0, [x23, x22, lsl #3]
    add x22, x22, #1
parse_done:

    mov x26, #1
sort_outer:
    cmp x26, x22
    bge sort_done
    ldr x27, [x21, x26, lsl #3]
    ldr x28, [x23, x26, lsl #3]
    sub x25, x26, #1
sort_inner:
    cmp x25, #0
    blt sort_insert
    ldr x0, [x21, x25, lsl #3]
    ldr x1, [x23, x25, lsl #3]
    mov x2, x27
    mov x3, x28
    bl compare
    cmp x0, #0
    ble sort_insert
    add x24, x25, #1
    ldr x0, [x21, x25, lsl #3]
    str x0, [x21, x24, lsl #3]
    ldr x0, [x23, x25, lsl #3]
    str x0, [x23, x24, lsl #3]
    sub x25, x25, #1
    b sort_inner
sort_insert:
    add x24, x25, #1
    str x27, [x21, x24, lsl #3]
    str x28, [x23, x24, lsl #3]
    add x26, x26, #1
    b sort_outer
sort_done:

    adrp x20, nl
    add x20, x20, :lo12:nl
    mov x26, #0
write_loop:
    cmp x26, x22
    bge write_done
    ldr x1, [x21, x26, lsl #3]
    ldr x2, [x23, x26, lsl #3]
    cbz x2, write_nl
    mov x0, #1
    mov x8, #SYS_write
    svc #0
write_nl:
    mov x0, #1
    mov x1, x20
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    add x26, x26, #1
    b write_loop
write_done:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

compare:
    cmp x1, x3
    csel x4, x1, x3, lo
    mov x5, #0
cmp_loop:
    cmp x5, x4
    bge cmp_len
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    bne cmp_diff
    add x5, x5, #1
    b cmp_loop
cmp_len:
    sub x0, x1, x3
    ret
cmp_diff:
    sub x0, x6, x7
    ret

.section .rodata
nl:
    .byte 10

.bss
.balign 8
buf:
    .space 32768
.balign 8
lines:
    .space 8000
.balign 8
lens:
    .space 8000
