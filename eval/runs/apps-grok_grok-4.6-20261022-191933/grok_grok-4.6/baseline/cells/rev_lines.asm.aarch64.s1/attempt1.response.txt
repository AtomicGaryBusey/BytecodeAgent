.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.text
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, #0

read_loop:
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #SYS_read
    mov x0, #0
    add x1, x19, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0

line_loop:
    cmp x21, x20
    b.ge lines_done
    mov x22, x21

find_nl:
    cmp x21, x20
    b.ge do_reverse
    ldrb w0, [x19, x21]
    cmp w0, #10
    b.eq do_reverse
    add x21, x21, #1
    b find_nl

do_reverse:
    add x0, x19, x22
    add x1, x19, x21
    sub x1, x1, #1
rev_loop:
    cmp x0, x1
    b.ge rev_done
    ldrb w2, [x0]
    ldrb w3, [x1]
    strb w3, [x0], #1
    strb w2, [x1], #-1
    b rev_loop

rev_done:
    cmp x21, x20
    b.ge lines_done
    add x21, x21, #1
    b line_loop

lines_done:
    cbz x20, do_exit
    sub x0, x20, #1
    ldrb w1, [x19, x0]
    cmp w1, #10
    b.eq do_write
    mov w1, #10
    strb w1, [x19, x20]
    add x20, x20, #1

do_write:
    mov x21, #0
write_loop:
    cmp x21, x20
    b.ge do_exit
    mov x8, #SYS_write
    mov x0, #1
    add x1, x19, x21
    sub x2, x20, x21
    svc #0
    cmp x0, #0
    b.le do_exit
    add x21, x21, x0
    b write_loop

do_exit:
    mov x8, #SYS_exit
    mov x0, #0
    svc #0

.bss
.balign 16
buf:
    .space 65537
