.globl _start
.text
_start:
    xor r13, r13
read_loop:
    mov edx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rip + inbuf]
    lea r15, [rip + outbuf]
    xor r14, r14
    xor rbx, rbx
process_loop:
    cmp r14, r13
    jae process_done
    mov r8, r14
find_eol:
    cmp r14, r13
    jae found_eol
    cmp byte ptr [r12 + r14], 10
    je found_eol
    inc r14
    jmp find_eol
found_eol:
    mov rcx, r14
    sub rcx, r8
    jz add_nl
    lea rsi, [r12 + r14]
rev_loop:
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [r15 + rbx], al
    inc rbx
    dec rcx
    jnz rev_loop
add_nl:
    mov byte ptr [r15 + rbx], 10
    inc rbx
    cmp r14, r13
    jae process_done
    inc r14
    jmp process_loop
process_done:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov rdx, rbx
    syscall
    mov eax, 60
    xor edi, edi
    syscall
.bss

outbuf:
