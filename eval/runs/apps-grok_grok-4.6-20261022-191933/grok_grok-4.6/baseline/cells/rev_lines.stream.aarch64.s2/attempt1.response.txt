.globl _start

.text

_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0
    movz    x21, #1, lsl #16

read_loop:
    cbz     x21, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, x21
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    ble     read_done
    add     x20, x20, x0
    sub     x21, x21, x0
    b       read_loop

read_done:
    mov     x21, #0
    mov     x22, #0

scan:
    cmp     x21, x20
    bge     scan_done
    ldrb    w0, [x19, x21]
    cmp     w0, #10
    bne     scan_next
    add     x0, x19, x22
    sub     x1, x21, x22
    bl      emit_line
    add     x22, x21, #1
scan_next:
    add     x21, x21, #1
    b       scan

scan_done:
    cmp     x22, x20
    bge     do_exit
    add     x0, x19, x22
    sub     x1, x20, x22
    bl      emit_line

do_exit:
    mov     x0, #0
    mov     x8, #93
    svc     #0

emit_line:
    cbz     x1, emit_nl
    mov     x2, x0
    add     x3, x0, x1
    sub     x3, x3, #1
rev_loop:
    cmp     x2, x3
    bge     rev_done
    ldrb    w4, [x2]
    ldrb    w5, [x3]
    strb    w5, [x2]
    strb    w4, [x3]
    add     x2, x2, #1
    sub     x3, x3, #1
    b       rev_loop
rev_done:
    mov     x2, x1
    mov     x1, x0
    mov     x0, #1
    mov     x8, #64
    svc     #0
emit_nl:
    mov     x0, #1
    adrp    x1, nl
    add     x1, x1, :lo12:nl
    mov     x2, #1
    mov     x8, #64
    svc     #0
    ret
