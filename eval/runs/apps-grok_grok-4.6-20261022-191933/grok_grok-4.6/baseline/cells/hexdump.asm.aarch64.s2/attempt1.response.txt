.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAX_IN, 65536

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x23, hextab
    add x23, x23, :lo12:hextab
    adrp x26, outbuf
    add x26, x26, :lo12:outbuf

    mov x20, #0
read_loop:
    mov x2, #MAX_IN
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_read
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x1, x26
    mov x21, #0

dump_loop:
    cmp x21, x20
    b.hs dump_final

    sub x22, x20, x21
    cmp x22, #16
    b.ls got_n
    mov x22, #16
got_n:
    add x25, x19, x21

    mov w0, w21
    bl put8hex

    mov w2, #32
    strb w2, [x1], #1
    strb w2, [x1], #1

    mov x24, #0
hex_slots:
    cmp x24, x22
    b.hs hex_pad
    ldrb w0, [x25, x24]
    bl put2hex
    mov w2, #32
    strb w2, [x1], #1
    b hex_slot_end
hex_pad:
    mov w2, #32
    strb w2, [x1], #1
    strb w2, [x1], #1
    strb w2, [x1], #1
hex_slot_end:
    cmp x24, #7
    b.ne hex_no_mid
    mov w2, #32
    strb w2, [x1], #1
hex_no_mid:
    add x24, x24, #1
    cmp x24, #16
    b.lo hex_slots

    mov w2, #32
    strb w2, [x1], #1
    mov w2, #124
    strb w2, [x1], #1

    mov x24, #0
ascii_loop:
    cmp x24, x22
    b.hs ascii_done
    ldrb w0, [x25, x24]
    cmp w0, #32
    b.lo ascii_dot
    cmp w0, #126
    b.ls ascii_store
ascii_dot:
    mov w0, #46
ascii_store:
    strb w0, [x1], #1
    add x24, x24, #1
    b ascii_loop
ascii_done:

    mov w2, #124
    strb w2, [x1], #1
    mov w2, #10
    strb w2, [x1], #1

    add x21, x21, x22
    b dump_loop

dump_final:
    mov w0, w20
    bl put8hex
    mov w2, #10
    strb w2, [x1], #1

    mov x27, x1
write_loop:
    cmp x26, x27
    b.hs do_exit
    mov x0, #1
    mov x1, x26
    sub x2, x27, x26
    mov x8, #SYS_write
    svc #0
    cmp x0, #0
    b.le do_exit
    add x26, x26, x0
    b write_loop

do_exit:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

put8hex:
    mov w3, #8
put8_loop:
    lsr w2, w0, #28
    ldrb w2, [x23, x2]
    strb w2, [x1], #1
    lsl w0, w0, #4
    subs w3, w3, #1
    b.ne put8_loop
    ret

put2hex:
    lsr w2, w0, #4
    ldrb w2, [x23, x2]
    strb w2, [x1], #1
    and w2, w0, #15
    ldrb w2, [x23, x2]
    strb w2, [x1], #1
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 327680
