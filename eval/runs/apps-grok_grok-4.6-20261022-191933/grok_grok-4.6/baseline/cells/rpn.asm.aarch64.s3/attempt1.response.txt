.text
.p2align 2
.globl _start
_start:
    adrp    x20, input_buf
    add     x20, x20, :lo12:input_buf
    mov     x21, #0

read_loop:
    mov     x2, #65536
    sub     x2, x2, x21
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x20, x21
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x21, x21, x0
    b       read_loop

read_done:
    add     x21, x20, x21
    mov     x22, x20

line_loop:
    cmp     x22, x21
    b.ge    all_done
    mov     x23, x22

find_nl:
    cmp     x22, x21
    b.ge    last_line
    ldrb    w0, [x22]
    cmp     w0, #10
    b.eq    found_nl
    add     x22, x22, #1
    b       find_nl

found_nl:
    mov     x25, x22
    add     x22, x22, #1
    bl      process_line
    b       line_loop

last_line:
    mov     x25, x21
    bl      process_line

all_done:
    mov     x0, #0
    mov     x8, #93
    svc     #0

process_line:
    str     x30, [sp, #-16]!
    mov     x19, x23
    adrp    x26, rpn_stack
    add     x26, x26, :lo12:rpn_stack
    mov     x27, #0

skip_ws:
    cmp     x19, x25
    b.ge    tokens_done
    ldrb    w0, [x19]
    cmp     w0, #32
    b.eq    skip_one
    cmp     w0, #9
    b.ne    have_token
skip_one:
    add     x19, x19, #1
    b       skip_ws

have_token:
    mov     x1, x19
find_tok_end:
    add     x19, x19, #1
    cmp     x19, x25
    b.ge    token_got
    ldrb    w0, [x19]
    cmp     w0, #32
    b.eq    token_got
    cmp     w0, #9
    b.ne    find_tok_end

token_got:
    sub     x2, x19, x1
    cmp     x2, #1
    b.ne    check_number
    ldrb    w10, [x1]
    cmp     w10, #'+'
    b.eq    do_binop
    cmp     w10, #'-'
    b.eq    do_binop
    cmp     w10, #'*'
    b.eq    do_binop
    cmp     w10, #'/'
    b.eq    do_binop
    cmp     w10, #'%'
    b.eq    do_binop

check_number:
    mov     x3, x1
    mov     x4, #0
    ldrb    w0, [x3]
    cmp     w0, #'-'
    b.ne    parse_digits
    mov     x4, #1
    add     x3, x3, #1
parse_digits:
    sub     x5, x19, x3
    cmp     x5, #1
    b.lt    line_error
    cmp     x5, #18
    b.gt    line_error
    mov     x6, #0
    mov     x7, #10
parse_loop:
    ldrb    w0, [x3], #1
    sub     w0, w0, #'0'
    cmp     w0, #9
    b.hi    line_error
    madd    x6, x6, x7, x0
    cmp     x3, x19
    b.lt    parse_loop
    cbz     x4, push_val
    neg     x6, x6

push_val:
    str     x6, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       skip_ws

do_binop:
    cmp     x27, #2
    b.lt    line_error
    sub     x27, x27, #1
    ldr     x9, [x26, x27, lsl #3]
    sub     x27, x27, #1
    ldr     x8, [x26, x27, lsl #3]
    cmp     w10, #'+'
    b.eq    op_add
    cmp     w10, #'-'
    b.eq    op_sub
    cmp     w10, #'*'
    b.eq    op_mul
    b       op_divmod

op_add:
    add     x6, x8, x9
    b       push_val

op_sub:
    sub     x6, x8, x9
    b       push_val

op_mul:
    mul     x6, x8, x9
    b       push_val

op_divmod:
    cbz     x9, line_error
    mov     x0, #1
    lsl     x0, x0, #63
    cmp     x8, x0
    b.ne    do_sdiv
    cmn     x9, #1
    b.eq    line_error
do_sdiv:
    sdiv    x6, x8, x9
    cmp     w10, #'/'
    b.eq    push_val
    msub    x6, x6, x9, x8
    b       push_val

tokens_done:
    cmp     x27, #1
    b.ne    line_error
    ldr     x0, [x26]
    bl      print_int64
    ldr     x30, [sp], #16
    ret

line_error:
    adrp    x1, error_msg
    add     x1, x1, :lo12:error_msg
    mov     x0, #1
    mov     x2, #6
    mov     x8, #64
    svc     #0
    ldr     x30, [sp], #16
    ret

print_int64:
    adrp    x11, out_buf
    add     x11, x11, :lo12:out_buf
    add     x12, x11, #31
    mov     w1, #10
    strb    w1, [x12]
    mov     x14, #10
    mov     x13, #0
    cmp     x0, #0
    b.ge    conv_loop
    mov     x13, #1
    neg     x0, x0
conv_loop:
    udiv    x15, x0, x14
    msub    x1, x15, x14, x0
    add     w1, w1, #'0'
    sub     x12, x12, #1
    strb    w1, [x12]
    mov     x0, x15
    cbnz    x0, conv_loop
    cbz     x13, conv_write
    sub     x12, x12, #1
    mov     w1, #'-'
    strb    w1, [x12]
conv_write:
    mov     x0, #1
    mov     x1, x12
    add     x2, x11, #32
    sub     x2, x2, x12
    mov     x8, #64
    svc     #0
    ret

.section .rodata
error_msg:
    .ascii  "error\n"

.bss
.balign 8
input_buf:
    .space  65536
rpn_stack:
    .space  8192
out_buf:
    .space  32
