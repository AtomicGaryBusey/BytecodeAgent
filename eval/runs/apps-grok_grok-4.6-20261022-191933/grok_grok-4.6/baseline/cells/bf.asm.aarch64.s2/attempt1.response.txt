.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_BYTES, 524288
.equ STACK_BYTES, 8000

.text
.balign 4
.globl _start
_start:
    // Read all of stdin into stdin_buf (at most BUF_SIZE bytes)
    adrp x19, stdin_buf
    add x19, x19, :lo12:stdin_buf
    mov x20, x19
    mov x0, #BUF_SIZE
    add x21, x19, x0

read_loop:
    cmp x20, x21
    b.hs read_done
    mov x0, #0
    mov x1, x20
    sub x2, x21, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    // x19 = start of input, x20 = end of input
    // Split on the first '!' into program and program-input
    mov x0, x19
find_bang:
    cmp x0, x20
    b.eq no_bang
    ldrb w1, [x0]
    cmp w1, #0x21
    b.eq found_bang
    add x0, x0, #1
    b find_bang

found_bang:
    mov x21, x0
    add x24, x0, #1
    mov x25, x20
    b setup_brackets

no_bang:
    mov x21, x20
    mov x24, x20
    mov x25, x20

setup_brackets:
    // x19 = program start, x21 = program end
    // x24 = input start, x25 = input end
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    adrp x9, bracket_stack
    add x9, x9, :lo12:bracket_stack
    mov x10, x9
    mov x0, x19

bracket_loop:
    cmp x0, x21
    b.eq bracket_done
    ldrb w1, [x0]
    cmp w1, #0x5b
    b.eq bracket_open
    cmp w1, #0x5d
    b.eq bracket_close
    add x0, x0, #1
    b bracket_loop

bracket_open:
    str x0, [x10], #8
    add x0, x0, #1
    b bracket_loop

bracket_close:
    cmp x10, x9
    b.eq unbalanced
    ldr x1, [x10, #-8]!
    sub x2, x0, x19
    sub x3, x1, x19
    str x1, [x26, x2, lsl #3]
    str x0, [x26, x3, lsl #3]
    add x0, x0, #1
    b bracket_loop

bracket_done:
    cmp x10, x9
    b.ne unbalanced

    // Interpreter setup
    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x28, x27
    mov x20, x19

interp_loop:
    cmp x20, x21
    b.hs interp_done
    ldrb w0, [x20]
    cmp w0, #0x2b
    b.eq do_plus
    cmp w0, #0x2d
    b.eq do_minus
    cmp w0, #0x3e
    b.eq do_right
    cmp w0, #0x3c
    b.eq do_left
    cmp w0, #0x2e
    b.eq do_dot
    cmp w0, #0x2c
    b.eq do_comma
    cmp w0, #0x5b
    b.eq do_lbrack
    cmp w0, #0x5d
    b.eq do_rbrack
    add x20, x20, #1
    b interp_loop

do_plus:
    ldrb w0, [x22]
    add w0, w0, #1
    strb w0, [x22]
    add x20, x20, #1
    b interp_loop

do_minus:
    ldrb w0, [x22]
    sub w0, w0, #1
    strb w0, [x22]
    add x20, x20, #1
    b interp_loop

do_right:
    add x22, x22, #1
    add x20, x20, #1
    b interp_loop

do_left:
    sub x22, x22, #1
    add x20, x20, #1
    b interp_loop

do_dot:
    ldrb w0, [x22]
    strb w0, [x28], #1
    add x20, x20, #1
    b interp_loop

do_comma:
    cmp x24, x25
    b.hs comma_eof
    ldrb w0, [x24], #1
    strb w0, [x22]
    add x20, x20, #1
    b interp_loop
comma_eof:
    strb wzr, [x22]
    add x20, x20, #1
    b interp_loop

do_lbrack:
    ldrb w0, [x22]
    cbnz w0, lbrack_cont
    sub x2, x20, x19
    ldr x20, [x26, x2, lsl #3]
lbrack_cont:
    add x20, x20, #1
    b interp_loop

do_rbrack:
    ldrb w0, [x22]
    cbz w0, rbrack_cont
    sub x2, x20, x19
    ldr x20, [x26, x2, lsl #3]
rbrack_cont:
    add x20, x20, #1
    b interp_loop

interp_done:
    sub x2, x28, x27
    cbz x2, do_exit
    mov x0, #1
    mov x1, x27
    mov x8, #SYS_WRITE
    svc #0

do_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
    b do_exit

unbalanced:
    mov x0, #1
    mov x8, #SYS_EXIT
    svc #0
    b unbalanced

.bss
.balign 16
stdin_buf:
    .zero BUF_SIZE
tape:
    .zero TAPE_SIZE
.balign 8
jump_table:
    .zero JUMP_BYTES
output_buf:
    .zero BUF_SIZE
.balign 8
bracket_stack:
    .zero STACK_BYTES
