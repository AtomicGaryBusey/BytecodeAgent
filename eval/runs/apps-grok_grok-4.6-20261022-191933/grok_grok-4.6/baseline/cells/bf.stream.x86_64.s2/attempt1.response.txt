.globl _start
.text
_start:
    lea rdi, [rip + dispatch_table]
    lea rax, [rip + ign_cmd]
    mov ecx, 256
    rep stosq
    lea rdi, [rip + dispatch_table]
    lea rax, [rip + cmd_gt]
    mov qword ptr [rdi + 62*8], rax
    lea rax, [rip + cmd_lt]
    mov qword ptr [rdi + 60*8], rax
    lea rax, [rip + cmd_plus]
    mov qword ptr [rdi + 43*8], rax
    lea rax, [rip + cmd_minus]
    mov qword ptr [rdi + 45*8], rax
    lea rax, [rip + cmd_dot]
    mov qword ptr [rdi + 46*8], rax
    lea rax, [rip + cmd_comma]
    mov qword ptr [rdi + 44*8], rax
    lea rax, [rip + cmd_lbrack]
    mov qword ptr [rdi + 91*8], rax
    lea rax, [rip + cmd_rbrack]
    mov qword ptr [rdi + 93*8], rax

    lea r12, [rip + input_buf]
    mov edx, 65536
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    sub rdx, rax
    jnz read_loop
read_done:
    lea rbx, [rip + input_buf]
    mov r14, r12
    sub r14, rbx

    xor ecx, ecx
find_bang:
    cmp rcx, r14
    jae no_bang
    cmp byte ptr [rbx + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
found_bang:
    mov r15, rcx
    lea r8, [rbx + rcx + 1]
    mov r9, r14
    sub r9, rcx
    dec r9
    jmp got_split
no_bang:
    mov r15, r14
    lea r8, [rbx + r14]
    xor r9, r9
got_split:

    lea rsi, [rip + bracket_stack]
    lea rdi, [rip + jump_table]
    lea r10, [rip + input_buf]
    xor ecx, ecx
    xor edx, edx
validate_loop:
    cmp rcx, r15
    jae validate_end
    movzx eax, byte ptr [r10 + rcx]
    cmp al, 0x5b
    je va_open
    cmp al, 0x5d
    je va_close
va_next:
    inc rcx
    jmp validate_loop
va_open:
    cmp edx, 1000
    jae unbalanced
    mov dword ptr [rsi + rdx*4], ecx
    inc edx
    jmp va_next
va_close:
    test edx, edx
    jz unbalanced
    dec edx
    mov eax, dword ptr [rsi + rdx*4]
    mov dword ptr [rdi + rax*4], ecx
    mov dword ptr [rdi + rcx*4], eax
    jmp va_next
validate_end:
    test edx, edx
    jnz unbalanced

    mov rbx, r8
    add r9, r8
    mov rbp, r9
    lea r8, [rip + jump_table]
    lea r9, [rip + out_buf]
    lea r10, [rip + input_buf]
    lea r11, [rip + dispatch_table]
    lea r12, [rip + tape]
    xor r13, r13
    xor r14, r14
    test r15, r15
    jz exec_done
dispatch:
    movzx eax, byte ptr [r10 + r13]
    jmp qword ptr [r11 + rax*8]

ign_cmd:
    inc r13
    cmp r13, r15
    jb dispatch
    jmp exec_done

cmd_gt:
    inc r12
    jmp ign_cmd
cmd_lt:
    dec r12
    jmp ign_cmd
cmd_plus:
    inc byte ptr [r12]
    jmp ign_cmd
cmd_minus:
    dec byte ptr [r12]
    jmp ign_cmd
cmd_dot:
    mov al, byte ptr [r12]
    mov byte ptr [r9 + r14], al
    inc r14
    jmp ign_cmd
cmd_comma:
    cmp rbx, rbp
    jae comma_eof
    mov al, byte ptr [rbx]
    inc rbx
    mov byte ptr [r12], al
    jmp ign_cmd
comma_eof:
    mov byte ptr [r12], 0
    jmp ign_cmd
cmd_lbrack:
    cmp byte ptr [r12], 0
    jne ign_cmd
    mov r13d, dword ptr [r8 + r13*4]
    jmp ign_cmd
cmd_rbrack:
    cmp byte ptr [r12], 0
    je ign_cmd
    mov r13d, dword ptr [r8 + r13*4]
    jmp ign_cmd

exec_done:
    test r14, r14
    jz do_exit
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + out_buf]
    mov rdx, r14
    syscall
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

unbalanced:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 8
input_buf:
    .space 65536
tape:
    .space 30000
jump_table:
    .space 262144
out_buf:
    .space 65536
dispatch_table:
    .space 2048
bracket_stack:
    .space 4000
