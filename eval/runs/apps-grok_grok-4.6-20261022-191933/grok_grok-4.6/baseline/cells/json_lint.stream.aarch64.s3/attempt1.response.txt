.text
.globl _start
_start:
    adrp x19, buf
    add x19, x19, :lo12:buf
    mov x20, x19
    mov x21, #65536
read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, read_loop
read_done:
    mov x21, x20
    mov x20, x19
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.ne error
    adrp x1, msg_ok
    add x1, x1, :lo12:msg_ok
    mov x0, #1
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

error:
    sub x22, x20, x19
    adrp x1, msg_err
    add x1, x1, :lo12:msg_err
    mov x0, #1
    mov x2, #9
    mov x8, #64
    svc #0
    sub sp, sp, #16
    add x1, sp, #16
    mov x0, x22
itoa_loop:
    mov x3, #10
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #'0'
    sub x1, x1, #1
    strb w5, [x1]
    mov x0, x4
    cbnz x0, itoa_loop
    add x2, sp, #16
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    add sp, sp, #16
    adrp x1, msg_nl
    add x1, x1, :lo12:msg_nl
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    mov x0, #1
    mov x8, #93
    svc #0

skip_ws:
    cmp x20, x21
    b.eq skip_ws_done
    ldrb w0, [x20]
    cmp w0, #' '
    b.eq skip_ws_eat
    cmp w0, #9
    b.eq skip_ws_eat
    cmp w0, #10
    b.eq skip_ws_eat
    cmp w0, #13
    b.ne skip_ws_done
skip_ws_eat:
    add x20, x20, #1
    b skip_ws
skip_ws_done:
    ret

parse_value:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'"'
    b.eq parse_string
    cmp w0, #'{'
    b.eq parse_object
    cmp w0, #'['
    b.eq parse_array
    cmp w0, #'t'
    b.eq parse_true
    cmp w0, #'f'
    b.eq parse_false
    cmp w0, #'n'
    b.eq parse_null
    cmp w0, #'-'
    b.eq parse_number
    sub w1, w0, #'0'
    cmp w1, #10
    b.lo parse_number
    b error

parse_true:
    add x20, x20, #1
    adrp x0, str_rue
    add x0, x0, :lo12:str_rue
    b match
parse_false:
    add x20, x20, #1
    adrp x0, str_alse
    add x0, x0, :lo12:str_alse
    b match
parse_null:
    add x20, x20, #1
    adrp x0, str_ull
    add x0, x0, :lo12:str_ull
    b match
match:
    ldrb w1, [x0], #1
    cbz w1, match_done
    cmp x20, x21
    b.eq error
    ldrb w2, [x20]
    cmp w1, w2
    b.ne error
    add x20, x20, #1
    b match
match_done:
    ret

parse_string:
    add x20, x20, #1
string_loop:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'"'
    b.eq string_end
    cmp w0, #'\\'
    b.eq string_escape
    cmp w0, #0x20
    b.lo error
    add x20, x20, #1
    b string_loop
string_end:
    add x20, x20, #1
    ret
string_escape:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'"'
    b.eq esc_ok
    cmp w0, #'\\'
    b.eq esc_ok
    cmp w0, #'/'
    b.eq esc_ok
    cmp w0, #'b'
    b.eq esc_ok
    cmp w0, #'f'
    b.eq esc_ok
    cmp w0, #'n'
    b.eq esc_ok
    cmp w0, #'r'
    b.eq esc_ok
    cmp w0, #'t'
    b.eq esc_ok
    cmp w0, #'u'
    b.eq string_unicode
    b error
esc_ok:
    add x20, x20, #1
    b string_loop
string_unicode:
    add x20, x20, #1
    mov w1, #4
uni_loop:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    sub w2, w0, #'0'
    cmp w2, #10
    b.lo uni_ok
    orr w2, w0, #32
    sub w2, w2, #'a'
    cmp w2, #6
    b.hs error
uni_ok:
    add x20, x20, #1
    subs w1, w1, #1
    b.ne uni_loop
    b string_loop

parse_number:
    ldrb w0, [x20]
    cmp w0, #'-'
    b.ne num_int
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
num_int:
    cmp w0, #'0'
    b.eq num_zero
    sub w1, w0, #'1'
    cmp w1, #9
    b.hs error
    add x20, x20, #1
num_int_digits:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs num_frac_or_exp
    add x20, x20, #1
    b num_int_digits
num_zero:
    add x20, x20, #1
num_frac_or_exp:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x20]
    cmp w0, #'.'
    b.eq num_frac
    cmp w0, #'e'
    b.eq num_exp
    cmp w0, #'E'
    b.eq num_exp
num_done:
    ret
num_frac:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs error
    add x20, x20, #1
num_frac_digits:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs num_exp_check
    add x20, x20, #1
    b num_frac_digits
num_exp_check:
    cmp w0, #'e'
    b.eq num_exp
    cmp w0, #'E'
    b.eq num_exp
    ret
num_exp:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'+'
    b.eq num_exp_sign
    cmp w0, #'-'
    b.ne num_exp_need_digit
num_exp_sign:
    add x20, x20, #1
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
num_exp_need_digit:
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs error
    add x20, x20, #1
num_exp_digits:
    cmp x20, x21
    b.eq num_done
    ldrb w0, [x20]
    sub w1, w0, #'0'
    cmp w1, #10
    b.hs num_done
    add x20, x20, #1
    b num_exp_digits

parse_object:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'}'
    b.eq obj_end
obj_loop:
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #'"'
    b.ne error
    bl parse_string
    bl skip_ws
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #':'
    b.ne error
    add x20, x20, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #','
    b.eq obj_comma
    cmp w0, #'}'
    b.eq obj_end
    b error
obj_comma:
    add x20, x20, #1
    bl skip_ws
    b obj_loop
obj_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x20, x20, #1
    bl skip_ws
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #']'
    b.eq arr_end
arr_loop:
    bl parse_value
    bl skip_ws
    cmp x20, x21
    b.eq error
    ldrb w0, [x20]
    cmp w0, #','
    b.eq arr_comma
    cmp w0, #']'
    b.eq arr_end
    b error
arr_comma:
    add x20, x20, #1
    bl skip_ws
    b arr_loop
arr_end:
    add x20, x20, #1
    ldp x29, x30, [sp], #16
    ret
