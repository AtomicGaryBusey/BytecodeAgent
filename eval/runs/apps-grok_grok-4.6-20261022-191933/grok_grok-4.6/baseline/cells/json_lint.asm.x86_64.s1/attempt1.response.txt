.globl _start
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INBUF_SIZE, 65536

.text
_start:
    lea r12, [rip + input_buf]
    mov r14, r12
read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r14
    lea rdx, [r12 + INBUF_SIZE]
    sub rdx, r14
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    mov r13, r14
    mov r14, r12
    call skip_ws
    cmp r14, r13
    jae error_eof
    call parse_value
    call skip_ws
    cmp r14, r13
    jne error_here
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

skip_ws:
    cmp r14, r13
    jae skip_ws_done
    movzx eax, byte ptr [r14]
    cmp al, 0x20
    je skip_ws_eat
    cmp al, 9
    je skip_ws_eat
    cmp al, 10
    je skip_ws_eat
    cmp al, 13
    je skip_ws_eat
skip_ws_done:
    ret
skip_ws_eat:
    inc r14
    jmp skip_ws

parse_value:
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb pv_kw
    cmp al, '9'
    jbe parse_number
pv_kw:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp error_here

parse_object:
    inc r14
    call skip_ws
    cmp r14, r13
    jae error_eof
    cmp byte ptr [r14], '}'
    jne po_member
    inc r14
    ret
po_member:
    cmp r14, r13
    jae error_eof
    cmp byte ptr [r14], '"'
    jne error_here
    call parse_string
    call skip_ws
    cmp r14, r13
    jae error_eof
    cmp byte ptr [r14], ':'
    jne error_here
    inc r14
    call skip_ws
    call parse_value
    call skip_ws
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '}'
    je po_close
    cmp al, ','
    jne error_here
    inc r14
    call skip_ws
    jmp po_member
po_close:
    inc r14
    ret

parse_array:
    inc r14
    call skip_ws
    cmp r14, r13
    jae error_eof
    cmp byte ptr [r14], ']'
    jne pa_value
    inc r14
    ret
pa_value:
    call parse_value
    call skip_ws
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, ']'
    je pa_close
    cmp al, ','
    jne error_here
    inc r14
    call skip_ws
    jmp pa_value
pa_close:
    inc r14
    ret

parse_string:
    inc r14
ps_loop:
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '"'
    je ps_end
    cmp al, 0x20
    jb error_here
    cmp al, 0x5C
    je ps_escape
    inc r14
    jmp ps_loop
ps_end:
    inc r14
    ret
ps_escape:
    inc r14
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '"'
    je ps_esc_ok
    cmp al, 0x5C
    je ps_esc_ok
    cmp al, '/'
    je ps_esc_ok
    cmp al, 'b'
    je ps_esc_ok
    cmp al, 'f'
    je ps_esc_ok
    cmp al, 'n'
    je ps_esc_ok
    cmp al, 'r'
    je ps_esc_ok
    cmp al, 't'
    je ps_esc_ok
    cmp al, 'u'
    je ps_unicode
    jmp error_here
ps_esc_ok:
    inc r14
    jmp ps_loop
ps_unicode:
    inc r14
    mov ecx, 4
ps_hex:
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '0'
    jb error_here
    cmp al, '9'
    jbe ps_hex_ok
    cmp al, 'A'
    jb error_here
    cmp al, 'F'
    jbe ps_hex_ok
    cmp al, 'a'
    jb error_here
    cmp al, 'f'
    jbe ps_hex_ok
    jmp error_here
ps_hex_ok:
    inc r14
    dec ecx
    jnz ps_hex
    jmp ps_loop

parse_number:
    cmp byte ptr [r14], '-'
    jne pn_int
    inc r14
    cmp r14, r13
    jae error_eof
pn_int:
    movzx eax, byte ptr [r14]
    cmp al, '0'
    je pn_zero
    cmp al, '1'
    jb error_here
    cmp al, '9'
    ja error_here
pn_int_digits:
    inc r14
    cmp r14, r13
    jae pn_frac
    movzx eax, byte ptr [r14]
    cmp al, '0'
    jb pn_frac
    cmp al, '9'
    jbe pn_int_digits
    jmp pn_frac
pn_zero:
    inc r14
pn_frac:
    cmp r14, r13
    jae pn_done
    cmp byte ptr [r14], '.'
    jne pn_exp
    inc r14
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '0'
    jb error_here
    cmp al, '9'
    ja error_here
pn_frac_digits:
    inc r14
    cmp r14, r13
    jae pn_done
    movzx eax, byte ptr [r14]
    cmp al, '0'
    jb pn_exp
    cmp al, '9'
    jbe pn_frac_digits
pn_exp:
    cmp r14, r13
    jae pn_done
    movzx eax, byte ptr [r14]
    cmp al, 'e'
    je pn_do_exp
    cmp al, 'E'
    jne pn_done
pn_do_exp:
    inc r14
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, '+'
    je pn_exp_sign
    cmp al, '-'
    jne pn_exp_need_digit
pn_exp_sign:
    inc r14
    cmp r14, r13
    jae error_eof
pn_exp_need_digit:
    movzx eax, byte ptr [r14]
    cmp al, '0'
    jb error_here
    cmp al, '9'
    ja error_here
pn_exp_digits:
    inc r14
    cmp r14, r13
    jae pn_done
    movzx eax, byte ptr [r14]
    cmp al, '0'
    jb pn_done
    cmp al, '9'
    jbe pn_exp_digits
pn_done:
    ret

parse_true:
    lea rsi, [rip + str_true]
    mov ecx, 4
    jmp match_lit
parse_false:
    lea rsi, [rip + str_false]
    mov ecx, 5
    jmp match_lit
parse_null:
    lea rsi, [rip + str_null]
    mov ecx, 4
match_lit:
    cmp r14, r13
    jae error_eof
    movzx eax, byte ptr [r14]
    cmp al, byte ptr [rsi]
    jne error_here
    inc r14
    inc rsi
    dec ecx
    jnz match_lit
    ret

error_eof:
    mov r8, r13
    sub r8, r12
    jmp print_error
error_here:
    mov r8, r14
    sub r8, r12
print_error:
    mov eax, SYS_WRITE
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall
    mov rax, r8
    lea rdi, [rip + numbuf + 16]
    mov byte ptr [rdi], 10
    mov rcx, 10
conv_loop:
    xor rdx, rdx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz conv_loop
    mov eax, SYS_WRITE
    lea rdx, [rip + numbuf + 17]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    syscall
    mov eax, SYS_EXIT
    mov edi, 1
    syscall

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
str_true:
    .ascii "true"
str_false:
    .ascii "false"
str_null:
    .ascii "null"

.bss
.balign 16
input_buf:
    .space 65536
numbuf:
    .space 32
