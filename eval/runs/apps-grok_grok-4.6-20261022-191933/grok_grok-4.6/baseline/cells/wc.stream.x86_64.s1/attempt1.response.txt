.globl _start

.equ BUFSIZE, 8192

.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, BUFSIZE
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    mov rcx, rax
    lea rsi, [rip + buf]

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne 1f
    inc r13
1:
    cmp al, 32
    je is_ws
    cmp al, 9
    jb not_ws
    cmp al, 13
    ja not_ws
is_ws:
    xor r15, r15
    jmp next_byte
not_ws:
    test r15, r15
    jnz next_byte
    inc r14
    mov r15, 1
next_byte:
    dec rcx
    jnz process_loop
    jmp read_loop

done_read:
    lea rbp, [rip + outbuf]
    lea rbx, [rbp + 63]
    mov byte ptr [rbx], 10
    dec rbx
    mov rax, r12
    call utoa
    mov byte ptr [rbx], 32
    dec rbx
    mov rax, r14
    call utoa
    mov byte ptr [rbx], 32
    dec rbx
    mov rax, r13
    call utoa
    inc rbx
    lea rdx, [rbp + 64]
    sub rdx, rbx
    mov rsi, rbx
    mov edi, 1
    mov eax, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
hang:
    jmp hang

utoa:
    mov ecx, 10
1:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [rbx], dl
    dec rbx
    test rax, rax
    jnz 1b
    ret

.bss
buf:
    .space 8192
outbuf:
    .space 64
