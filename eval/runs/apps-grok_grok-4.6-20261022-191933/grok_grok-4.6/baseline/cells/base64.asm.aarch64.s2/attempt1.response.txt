.globl _start

.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAXIN, 49152
.equ LINELEN, 76

.text
_start:
    adrp    x19, inbuf
    add     x19, x19, :lo12:inbuf
    mov     x20, #0

read_loop:
    mov     x0, #0
    add     x1, x19, x20
    mov     x2, #MAXIN
    sub     x2, x2, x20
    cbz     x2, read_done
    mov     x8, #SYS_read
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    cbz     x20, do_exit

    adrp    x22, outbuf
    add     x22, x22, :lo12:outbuf
    adrp    x23, alphabet
    add     x23, x23, :lo12:alphabet
    add     x20, x19, x20
    mov     x24, x19
    mov     x25, #0
    mov     x26, #0

encode_loop:
    sub     x0, x20, x24
    cmp     x0, #3
    b.lt    encode_tail

    ldrb    w1, [x24], #1
    ldrb    w2, [x24], #1
    ldrb    w3, [x24], #1

    lsr     w4, w1, #2
    ldrb    w4, [x23, x4]
    bl      emit_char

    and     w4, w1, #3
    lsl     w4, w4, #4
    lsr     w5, w2, #4
    orr     w4, w4, w5
    ldrb    w4, [x23, x4]
    bl      emit_char

    and     w4, w2, #15
    lsl     w4, w4, #2
    lsr     w5, w3, #6
    orr     w4, w4, w5
    ldrb    w4, [x23, x4]
    bl      emit_char

    and     w4, w3, #63
    ldrb    w4, [x23, x4]
    bl      emit_char

    b       encode_loop

encode_tail:
    cbz     x0, encode_done
    cmp     x0, #1
    b.eq    encode_1

    ldrb    w1, [x24], #1
    ldrb    w2, [x24], #1

    lsr     w4, w1, #2
    ldrb    w4, [x23, x4]
    bl      emit_char

    and     w4, w1, #3
    lsl     w4, w4, #4
    lsr     w5, w2, #4
    orr     w4, w4, w5
    ldrb    w4, [x23, x4]
    bl      emit_char

    and     w4, w2, #15
    lsl     w4, w4, #2
    ldrb    w4, [x23, x4]
    bl      emit_char

    mov     w4, #'='
    bl      emit_char
    b       encode_done

encode_1:
    ldrb    w1, [x24], #1

    lsr     w4, w1, #2
    ldrb    w4, [x23, x4]
    bl      emit_char

    and     w4, w1, #3
    lsl     w4, w4, #4
    ldrb    w4, [x23, x4]
    bl      emit_char

    mov     w4, #'='
    bl      emit_char
    mov     w4, #'='
    bl      emit_char

encode_done:
    cbz     x26, do_write
    mov     w4, #10
    strb    w4, [x22, x25]
    add     x25, x25, #1

do_write:
    mov     x0, #1
    mov     x1, x22
    mov     x2, x25
    mov     x8, #SYS_write
    svc     #0

do_exit:
    mov     x0, #0
    mov     x8, #SYS_exit
    svc     #0

emit_char:
    strb    w4, [x22, x25]
    add     x25, x25, #1
    add     x26, x26, #1
    cmp     x26, #LINELEN
    b.ne    1f
    mov     w4, #10
    strb    w4, [x22, x25]
    add     x25, x25, #1
    mov     x26, #0
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .balign 16
inbuf:
    .space 49152
outbuf:
    .space 66560
