.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUFSIZE, 65536

.globl _start
.text
_start:
    adrp x23, buf
    add x23, x23, :lo12:buf

read_loop:
    mov x8, SYS_read
    mov x0, #0
    mov x1, x23
    mov x2, BUFSIZE
    svc #0
    cmp x0, #0
    b.le finish
    add x21, x21, x0
    mov x9, x23
    add x10, x23, x0

process_loop:
    cmp x9, x10
    b.eq read_loop
    ldrb w11, [x9], #1
    cmp w11, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w11, #32
    b.eq 2f
    sub w12, w11, #9
    cmp w12, #4
    b.ls 2f
    cbnz x22, process_loop
    add x20, x20, #1
    mov x22, #1
    b process_loop
2:
    mov x22, #0
    b process_loop

finish:
    sub sp, sp, #64
    mov x25, sp
    mov x1, sp
    mov x0, x19
    bl utoa
    mov w2, #' '
    strb w2, [x1], #1
    mov x0, x20
    bl utoa
    mov w2, #' '
    strb w2, [x1], #1
    mov x0, x21
    bl utoa
    mov w2, #10
    strb w2, [x1], #1
    sub x2, x1, x25
    mov x8, SYS_write
    mov x0, #1
    mov x1, x25
    svc #0
    mov x8, SYS_exit
    mov x0, #0
    svc #0

utoa:
    cbnz x0, 1f
    mov w2, #'0'
    strb w2, [x1], #1
    ret
1:
    sub sp, sp, #16
    mov x3, #0
2:
    mov x4, #10
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #'0'
    strb w6, [sp, x3]
    add x3, x3, #1
    mov x0, x5
    cbnz x0, 2b
3:
    sub x3, x3, #1
    ldrb w6, [sp, x3]
    strb w6, [x1], #1
    cbnz x3, 3b
    add sp, sp, #16
    ret

.bss
.balign 16
buf:
    .space BUFSIZE
