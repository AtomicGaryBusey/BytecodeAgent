.globl _start
.text
_start:
    lea rbx, [rip + buffer]
    xor r12, r12
read_loop:
    mov edx, 32768
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    xor r14, r14
parse_loop:
    cmp r13, r12
    jae parse_done
    mov r8, r13
scan_nl:
    cmp r13, r12
    jae got_line
    cmp byte ptr [rbx + r13], 10
    je got_line
    inc r13
    jmp scan_nl
got_line:
    lea rcx, [rbx + r8]
    mov rax, r13
    sub rax, r8
    lea rdx, [rip + line_ptrs]
    mov qword ptr [rdx + r14*8], rcx
    lea rdx, [rip + line_lens]
    mov qword ptr [rdx + r14*8], rax
    inc r14
    cmp r13, r12
    jae parse_done
    inc r13
    jmp parse_loop
parse_done:
    mov rbx, 1
sort_outer:
    cmp rbx, r14
    jae sort_done
    lea rax, [rip + line_ptrs]
    mov r12, qword ptr [rax + rbx*8]
    lea rax, [rip + line_lens]
    mov r13, qword ptr [rax + rbx*8]
    mov rbp, rbx
sort_inner:
    test rbp, rbp
    jz sort_place
    lea rax, [rip + line_ptrs]
    mov rdi, qword ptr [rax + rbp*8 - 8]
    lea rax, [rip + line_lens]
    mov rsi, qword ptr [rax + rbp*8 - 8]
    mov rdx, r12
    mov rcx, r13
    call compare
    test rax, rax
    jle sort_place
    lea rax, [rip + line_ptrs]
    mov rdx, qword ptr [rax + rbp*8 - 8]
    mov qword ptr [rax + rbp*8], rdx
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + rbp*8 - 8]
    mov qword ptr [rax + rbp*8], rdx
    dec rbp
    jmp sort_inner
sort_place:
    lea rax, [rip + line_ptrs]
    mov qword ptr [rax + rbp*8], r12
    lea rax, [rip + line_lens]
    mov qword ptr [rax + rbp*8], r13
    inc rbx
    jmp sort_outer
sort_done:
    xor ebx, ebx
print_loop:
    cmp rbx, r14
    jae print_done
    lea rax, [rip + line_ptrs]
    mov rsi, qword ptr [rax + rbx*8]
    lea rax, [rip + line_lens]
    mov rdx, qword ptr [rax + rbx*8]
    test rdx, rdx
    jz print_nl
    mov eax, 1
    mov edi, 1
    syscall
print_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
    inc rbx
    jmp print_loop
print_done:
    mov eax, 60
    xor edi, edi
    syscall

compare:
    mov r8, rsi
    mov r9, rcx
    cmp rsi, rcx
    cmova rsi, rcx
    mov rcx, rsi
    mov rsi, rdi
    mov rdi, rdx
    test rcx, rcx
    jz 1f
    repe cmpsb
    jne 2f
1:
    mov rax, r8
    sub rax, r9
    ret
2:
    movzx eax, byte ptr [rsi - 1]
    movzx edx, byte ptr [rdi - 1]
    sub rax, rdx
    ret

.section .rodata
nl:
    .byte 10

.bss
.align 8
buffer:
    .space 32768
line_ptrs:
    .space 8000
line_lens:
    .space 8000
