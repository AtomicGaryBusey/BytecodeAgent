.globl _start

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    lea r12, [rip + inbuf]
    xor r13, r13

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 49152
    sub edx, r13d
    jle read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    mov rsi, r12
    lea rdi, [rip + outbuf]
    lea r15, [rip + alphabet]
    xor r14, r14

encode_loop:
    cmp r13, 3
    jb encode_tail
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]
    add rsi, 3
    sub r13, 3
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov eax, r10d
    and eax, 63
    movzx eax, byte ptr [r15 + rax]
    call emit
    jmp encode_loop

encode_tail:
    cmp r13, 2
    je encode_two
    cmp r13, 1
    je encode_one
    jmp encode_done

encode_two:
    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov eax, r9d
    and eax, 15
    shl eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov al, 61
    call emit
    jmp encode_done

encode_one:
    movzx r8d, byte ptr [rsi]
    mov eax, r8d
    shr eax, 2
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov eax, r8d
    and eax, 3
    shl eax, 4
    movzx eax, byte ptr [r15 + rax]
    call emit
    mov al, 61
    call emit
    mov al, 61
    call emit

encode_done:
    test r14, r14
    jz write_out
    mov byte ptr [rdi], 10
    inc rdi

write_out:
    lea rsi, [rip + outbuf]
    sub rdi, rsi
    jz do_exit
    mov rdx, rdi
    mov eax, 1
    mov edi, 1
    syscall

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

emit:
    mov byte ptr [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne 1f
    mov byte ptr [rdi], 10
    inc rdi
    xor r14, r14
1:
    ret

.bss
.balign 16
inbuf:
    .space 49152
outbuf:
    .space 66560
