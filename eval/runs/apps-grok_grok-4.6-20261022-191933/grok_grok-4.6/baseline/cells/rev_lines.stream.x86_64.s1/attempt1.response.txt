.equ BUFSIZE, 65536

.globl _start
.text
_start:
    xor ebx, ebx
.Lread:
    mov edx, BUFSIZE
    sub edx, ebx
    jz .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle .Lread_done
    add rbx, rax
    jmp .Lread
.Lread_done:
    xor r12d, r12d
    xor r13d, r13d
.Lscan:
    cmp r13, rbx
    jae .Lscan_done
    lea rax, [rip + inbuf]
    cmp byte ptr [rax + r13], 10
    je .Lfound_nl
    inc r13
    jmp .Lscan
.Lfound_nl:
    mov rdi, r12
    mov rsi, r13
    call reverse
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, r13
    sub rdx, r12
    inc rdx
    syscall
    inc r13
    mov r12, r13
    jmp .Lscan
.Lscan_done:
    cmp r12, rbx
    je .Lexit
    mov rdi, r12
    mov rsi, rbx
    call reverse
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, rbx
    sub rdx, r12
    syscall
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + nl]
    mov edx, 1
    syscall
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

reverse:
    cmp rdi, rsi
    jae 2f
    lea r8, [rip + inbuf]
    add rdi, r8
    add rsi, r8
    dec rsi
1:
    cmp rdi, rsi
    jae 2f
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp 1b
2:
    ret

.section .rodata
nl:
    .byte 10

.bss
inbuf:
    .space BUFSIZE
