.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INBUF_SIZE, 65536
.equ LINELEN, 76

.globl _start
.text
_start:
    lea r14, [rip + inbuf]
    mov rsi, r14
    lea r15, [r14 + INBUF_SIZE]

read_loop:
    mov rdx, r15
    sub rdx, rsi
    jz read_done
    mov eax, SYS_READ
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add rsi, rax
    jmp read_loop

read_done:
    mov r13, rsi
    sub r13, r14
    mov rsi, r14

    lea rdi, [rip + outbuf]
    mov r8, rdi
    xor r9d, r9d
    lea rbx, [rip + alphabet]

    mov rax, r13
    xor rdx, rdx
    mov rcx, 3
    div rcx
    mov r12, rdx
    mov rcx, rax
    test rcx, rcx
    jz do_remainder

encode_loop:
    movzx r13d, byte ptr [rsi]
    movzx r14d, byte ptr [rsi + 1]
    movzx r15d, byte ptr [rsi + 2]
    add rsi, 3

    mov eax, r13d
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call emit

    mov eax, r13d
    and eax, 3
    shl eax, 4
    mov edx, r14d
    shr edx, 4
    or eax, edx
    mov al, byte ptr [rbx + rax]
    call emit

    mov eax, r14d
    and eax, 15
    shl eax, 2
    mov edx, r15d
    shr edx, 6
    or eax, edx
    mov al, byte ptr [rbx + rax]
    call emit

    mov eax, r15d
    and eax, 63
    mov al, byte ptr [rbx + rax]
    call emit

    dec rcx
    jnz encode_loop

do_remainder:
    test r12, r12
    jz finish
    cmp r12, 1
    je rem1

    movzx r13d, byte ptr [rsi]
    movzx r14d, byte ptr [rsi + 1]

    mov eax, r13d
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call emit

    mov eax, r13d
    and eax, 3
    shl eax, 4
    mov edx, r14d
    shr edx, 4
    or eax, edx
    mov al, byte ptr [rbx + rax]
    call emit

    mov eax, r14d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [rbx + rax]
    call emit

    mov al, '='
    call emit
    jmp finish

rem1:
    movzx r13d, byte ptr [rsi]

    mov eax, r13d
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    call emit

    mov eax, r13d
    and eax, 3
    shl eax, 4
    mov al, byte ptr [rbx + rax]
    call emit

    mov al, '='
    call emit
    mov al, '='
    call emit

finish:
    test r9d, r9d
    jz write_out
    mov al, 10
    stosb

write_out:
    mov rdx, rdi
    sub rdx, r8
    jz do_exit
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r8
    syscall

do_exit:
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

emit:
    stosb
    inc r9d
    cmp r9d, LINELEN
    jne 1f
    mov al, 10
    stosb
    xor r9d, r9d
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .space 65536
outbuf:
    .space 131072
