.globl _start

.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFSIZE, 32768

.text
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x22, line_arr
    add x22, x22, :lo12:line_arr

    mov x20, #0
read_loop:
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    add x26, x19, x20
    mov x24, x19
    mov x25, x19
    mov x21, #0
parse_loop:
    cmp x24, x26
    b.eq parse_tail
    ldrb w0, [x24]
    cmp w0, #10
    b.eq parse_nl
    add x24, x24, #1
    b parse_loop
parse_nl:
    add x0, x22, x21, lsl #4
    sub x1, x24, x25
    stp x25, x1, [x0]
    add x21, x21, #1
    add x24, x24, #1
    mov x25, x24
    b parse_loop
parse_tail:
    cmp x25, x26
    b.eq parse_done
    add x0, x22, x21, lsl #4
    sub x1, x26, x25
    stp x25, x1, [x0]
    add x21, x21, #1
parse_done:

    mov x24, #1
sort_outer:
    cmp x24, x21
    b.ge sort_done
    add x0, x22, x24, lsl #4
    ldp x25, x26, [x0]
    sub x27, x24, #1
sort_inner:
    cmp x27, #0
    b.lt sort_ins
    add x4, x22, x27, lsl #4
    ldp x0, x1, [x4]
    mov x2, x25
    mov x3, x26
    bl compare
    cmp x0, #0
    b.le sort_ins
    add x4, x22, x27, lsl #4
    ldp x0, x1, [x4]
    stp x0, x1, [x4, #16]
    sub x27, x27, #1
    b sort_inner
sort_ins:
    add x0, x22, x27, lsl #4
    stp x25, x26, [x0, #16]
    add x24, x24, #1
    b sort_outer
sort_done:

    mov x24, #0
print_loop:
    cmp x24, x21
    b.ge print_done
    add x4, x22, x24, lsl #4
    ldp x1, x2, [x4]
    mov x0, #1
    bl write_fully
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    bl write_fully
    add x24, x24, #1
    b print_loop
print_done:

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0
hang:
    b hang

compare:
    cmp x1, x3
    csel x4, x3, x1, hi
    mov x5, #0
1:
    cmp x5, x4
    b.eq 2f
    ldrb w6, [x0, x5]
    ldrb w7, [x2, x5]
    cmp w6, w7
    b.ne 3f
    add x5, x5, #1
    b 1b
3:
    sub w0, w6, w7
    sxtw x0, w0
    ret
2:
    sub x0, x1, x3
    ret

write_fully:
    cbz x2, 1f
    mov x3, x0
2:
    mov x0, x3
    mov x8, #SYS_WRITE
    svc #0
    cmp x0, #0
    b.le 1f
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, 2b
1:
    ret

.section .rodata
nl:
    .byte 10

.bss
.balign 16
input_buf:
    .space 32768
line_arr:
    .space 16000
