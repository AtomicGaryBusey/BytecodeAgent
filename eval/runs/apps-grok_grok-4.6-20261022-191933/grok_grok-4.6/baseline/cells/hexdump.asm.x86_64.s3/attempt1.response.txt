.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ INPUT_MAX, 65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
    .balign 16
input:
    .space 65536
linebuf:
    .space 128

.text
.globl _start
_start:
    lea r8, [rip + hex_digits]
    lea r9, [rip + input]
    xor r12, r12

read_loop:
    mov edx, INPUT_MAX
    sub edx, r12d
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r9 + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r13, r13

print_loop:
    cmp r13, r12
    jae print_final
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_len_ok
    mov r14, 16
row_len_ok:
    lea rdi, [rip + linebuf]
    mov eax, r13d
    call put_hex8
    mov ax, 0x2020
    mov word ptr [rdi], ax
    add rdi, 2
    lea r15, [r9 + r13]
    xor ebx, ebx

hex_loop:
    cmp ebx, 16
    jge hex_done
    cmp ebx, r14d
    jae hex_pad
    movzx eax, byte ptr [r15 + rbx]
    call put_hex2
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp hex_after
hex_pad:
    mov ax, 0x2020
    mov word ptr [rdi], ax
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
hex_after:
    cmp ebx, 7
    jne hex_no_extra
    mov byte ptr [rdi], 0x20
    inc rdi
hex_no_extra:
    inc ebx
    jmp hex_loop

hex_done:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi
    xor ebx, ebx

ascii_loop:
    cmp ebx, r14d
    jae ascii_done
    movzx eax, byte ptr [r15 + rbx]
    mov dl, 0x2e
    cmp al, 0x20
    jb ascii_store
    cmp al, 0x7e
    ja ascii_store
    mov dl, al
ascii_store:
    mov byte ptr [rdi], dl
    inc rdi
    inc ebx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + linebuf]
    sub rdx, rsi
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    add r13, r14
    jmp print_loop

print_final:
    lea rdi, [rip + linebuf]
    mov eax, r12d
    call put_hex8
    mov byte ptr [rdi], 0x0a
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + linebuf]
    sub rdx, rsi
    mov eax, SYS_WRITE
    mov edi, 1
    syscall
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

put_hex8:
    mov ecx, 8
put_hex8_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz put_hex8_loop
    ret

put_hex2:
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    ret
