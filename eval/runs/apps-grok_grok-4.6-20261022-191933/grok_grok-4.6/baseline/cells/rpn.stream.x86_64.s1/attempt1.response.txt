.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13, r13
rd_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    jz rd_done
    syscall
    test rax, rax
    jle rd_done
    add r13, rax
    jmp rd_loop
rd_done:
    add r13, r12
main_loop:
    cmp r12, r13
    jae exit_ok
    mov r15, r12
find_nl:
    cmp r12, r13
    jae got_line
    cmp byte ptr [r12], 10
    je got_line
    inc r12
    jmp find_nl
got_line:
    push r12
    push r13
    call process_line
    pop r13
    pop r12
    cmp r12, r13
    jae exit_ok
    inc r12
    jmp main_loop
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

process_line:
    mov rbp, r12
    mov r8, r15
    lea r14, [rip + rpn_stack]
pl_skip:
    cmp r8, rbp
    jae pl_done
    movzx eax, byte ptr [r8]
    cmp al, 32
    je pl_ws
    cmp al, 9
    je pl_ws
    jmp pl_got
pl_ws:
    inc r8
    jmp pl_skip
pl_got:
    mov r9, r8
pl_scan:
    inc r8
    cmp r8, rbp
    jae pl_tend
    movzx eax, byte ptr [r8]
    cmp al, 32
    je pl_tend
    cmp al, 9
    je pl_tend
    jmp pl_scan
pl_tend:
    mov rcx, r8
    sub rcx, r9
    cmp rcx, 1
    jne pl_num
    movzx eax, byte ptr [r9]
    cmp al, '+'
    je pl_add
    cmp al, '-'
    je pl_sub
    cmp al, '*'
    je pl_mul
    cmp al, '/'
    je pl_div
    cmp al, '%'
    je pl_mod
pl_num:
    mov rsi, r9
    xor edi, edi
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne pl_ndig
    inc rsi
    cmp rsi, r8
    jae pl_err
    mov edi, 1
pl_ndig:
    mov rcx, r8
    sub rcx, rsi
    cmp rcx, 1
    jb pl_err
    cmp rcx, 18
    ja pl_err
    xor rax, rax
pl_nloop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja pl_err
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    cmp rsi, r8
    jb pl_nloop
    test edi, edi
    jz pl_npush
    neg rax
pl_npush:
    mov [r14], rax
    add r14, 8
    jmp pl_skip
pl_add:
    xor r10d, r10d
    jmp pl_bin
pl_sub:
    mov r10d, 1
    jmp pl_bin
pl_mul:
    mov r10d, 2
    jmp pl_bin
pl_div:
    mov r10d, 3
    jmp pl_bin
pl_mod:
    mov r10d, 4
pl_bin:
    lea rax, [rip + rpn_stack]
    add rax, 16
    cmp r14, rax
    jb pl_err
    sub r14, 8
    mov rbx, [r14]
    sub r14, 8
    mov rax, [r14]
    cmp r10d, 0
    je pl_opadd
    cmp r10d, 1
    je pl_opsub
    cmp r10d, 2
    je pl_opmul
    test rbx, rbx
    jz pl_err
    mov rcx, 1
    shl rcx, 63
    cmp rax, rcx
    jne pl_idiv
    cmp rbx, -1
    je pl_err
pl_idiv:
    cqo
    idiv rbx
    cmp r10d, 3
    je pl_bstore
    mov rax, rdx
    jmp pl_bstore
pl_opadd:
    add rax, rbx
    jmp pl_bstore
pl_opsub:
    sub rax, rbx
    jmp pl_bstore
pl_opmul:
    imul rax, rbx
pl_bstore:
    mov [r14], rax
    add r14, 8
    jmp pl_skip
pl_done:
    lea rax, [rip + rpn_stack]
    add rax, 8
    cmp r14, rax
    jne pl_err
    mov rax, [rip + rpn_stack]
    call print_num
    ret
pl_err:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 6
    syscall
    ret

print_num:
    lea r9, [rip + outbuf]
    lea rdi, [r9 + 31]
    mov byte ptr [rdi], 10
    xor r8d, r8d
    test rax, rax
    jns pn_cv
    mov r8d, 1
    neg rax
pn_cv:
    mov rcx, 10
pn_lp:
    xor rdx, rdx
    div rcx
    add edx, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pn_lp
    test r8d, r8d
    jz pn_wr
    dec rdi
    mov byte ptr [rdi], '-'
pn_wr:
    lea rdx, [r9 + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.balign 8
input:
    .space 65536
rpn_stack:
    .space 8192
outbuf:
    .space 32
