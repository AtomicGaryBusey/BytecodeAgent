.text
.globl _start
_start:
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf
    adrp    x22, hex_digits
    add     x22, x22, :lo12:hex_digits
    adrp    x23, line_buf
    add     x23, x23, :lo12:line_buf
    mov     x20, #0

read_loop:
    mov     x0, #65536
    sub     x2, x0, x20
    cbz     x2, read_done
    mov     x0, #0
    add     x1, x19, x20
    mov     x8, #63
    svc     #0
    cmp     x0, #0
    b.le    read_done
    add     x20, x20, x0
    b       read_loop

read_done:
    movz    x26, #0x2020
    movk    x26, #0x2020, lsl #16
    movk    x26, #0x2020, lsl #32
    movk    x26, #0x2020, lsl #48
    mov     x21, #0

print_loop:
    cmp     x21, x20
    b.hs    print_final
    sub     x24, x20, x21
    cmp     x24, #16
    b.ls    n_ok
    mov     x24, #16
n_ok:
    mov     w0, w21
    mov     x1, x23
    bl      write_hex8
    add     x0, x23, #8
    str     x26, [x0]
    str     x26, [x0, #8]
    str     x26, [x0, #16]
    str     x26, [x0, #24]
    str     x26, [x0, #32]
    str     x26, [x0, #40]
    str     w26, [x0, #48]
    mov     w0, #'|'
    strb    w0, [x23, #60]
    mov     x25, #0
hex_loop:
    cmp     x25, x24
    b.hs    hex_done
    add     x0, x23, #10
    add     x0, x0, x25, lsl #1
    add     x0, x0, x25
    cmp     x25, #8
    cinc    x0, x0, hs
    add     x1, x19, x21
    ldrb    w1, [x1, x25]
    lsr     w2, w1, #4
    ldrb    w2, [x22, x2]
    strb    w2, [x0]
    and     w1, w1, #15
    ldrb    w1, [x22, x1]
    strb    w1, [x0, #1]
    add     x25, x25, #1
    b       hex_loop
hex_done:
    add     x0, x23, #61
    mov     x25, #0
ascii_loop:
    cmp     x25, x24
    b.hs    ascii_done
    add     x1, x19, x21
    ldrb    w1, [x1, x25]
    sub     w2, w1, #32
    cmp     w2, #94
    b.ls    ascii_store
    mov     w1, #'.'
ascii_store:
    strb    w1, [x0], #1
    add     x25, x25, #1
    b       ascii_loop
ascii_done:
    mov     w1, #'|'
    strb    w1, [x0], #1
    mov     w1, #10
    strb    w1, [x0]
    mov     x0, #1
    mov     x1, x23
    add     x2, x24, #63
    mov     x8, #64
    svc     #0
    add     x21, x21, x24
    b       print_loop

print_final:
    mov     w0, w20
    mov     x1, x23
    bl      write_hex8
    mov     w0, #10
    strb    w0, [x23, #8]
    mov     x0, #1
    mov     x1, x23
    mov     x2, #9
    mov     x8, #64
    svc     #0
    mov     x0, #0
    mov     x8, #93
    svc     #0

write_hex8:
    mov     w2, #8
1:
    lsr     w3, w0, #28
    ldrb    w3, [x22, x3]
    strb    w3, [x1], #1
    lsl     w0, w0, #4
    subs    w2, w2, #1
    b.ne    1b
    ret

.section .rodata
hex_digits:
    .ascii  "0123456789abcdef"

.bss
.balign 16
line_buf:
    .space  128
.balign 16
input_buf:
    .space  65536
