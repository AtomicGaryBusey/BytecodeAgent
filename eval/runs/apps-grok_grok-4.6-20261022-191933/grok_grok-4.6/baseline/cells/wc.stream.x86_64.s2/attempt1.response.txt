.globl _start

.text

_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15

.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle .Lprint
    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buf]

.Lbyte:
    movzx r8d, byte ptr [rsi]
    inc rsi
    cmp r8b, 32
    je .Lws
    cmp r8b, 9
    jb .Lnw
    cmp r8b, 13
    ja .Lnw
.Lws:
    cmp r8b, 10
    jne .Lnotnl
    inc r12
.Lnotnl:
    xor r15, r15
    jmp .Lnext
.Lnw:
    test r15, r15
    jnz .Lnext
    inc r13
    mov r15, 1
.Lnext:
    dec rcx
    jnz .Lbyte
    jmp .Lread

.Lprint:
    mov rax, r12
    mov r8b, 32
    call .Lemit
    mov rax, r13
    mov r8b, 32
    call .Lemit
    mov rax, r14
    mov r8b, 10
    call .Lemit
    mov eax, 60
    xor edi, edi
    syscall

.Lemit:
    sub rsp, 40
    lea rdi, [rsp + 32]
    mov byte ptr [rdi], r8b
    mov ecx, 10
    test rax, rax
    jnz .Lconv
    dec rdi
    mov byte ptr [rdi], 48
    jmp .Lwrite
.Lconv:
    xor rdx, rdx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz .Lconv
.Lwrite:
    lea rdx, [rsp + 33]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall
    add rsp, 40
    ret

.bss
buf:
    .space 65536
