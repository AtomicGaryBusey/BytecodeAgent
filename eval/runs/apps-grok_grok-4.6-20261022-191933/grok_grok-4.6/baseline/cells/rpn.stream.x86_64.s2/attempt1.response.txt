.text
.globl _start
_start:
    lea r15, [rip + input_buf]
    lea r13, [r15 + 65536]
read_loop:
    mov rdx, r13
    sub rdx, r15
    jz read_done
    xor eax, eax
    xor edi, edi
    mov rsi, r15
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    lea r12, [rip + input_buf]
    mov r13, r15
    lea rbp, [rip + rpn_stack]
line_loop:
    cmp r12, r13
    jae all_done
    mov rbx, r12
find_nl_loop:
    cmp rbx, r13
    jae last_line
    cmp byte ptr [rbx], 10
    je got_nl
    inc rbx
    jmp find_nl_loop
got_nl:
    call process_line
    lea r12, [rbx + 1]
    jmp line_loop
last_line:
    call process_line
all_done:
    mov eax, 60
    xor edi, edi
    syscall

process_line:
    mov r14, rbp
    mov rsi, r12
parse_loop:
    cmp rsi, rbx
    jae line_finish
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je skip_one
    cmp al, 9
    jne scan_start
skip_one:
    inc rsi
    jmp parse_loop
scan_start:
    mov r8, rsi
scan_tok:
    inc r8
    cmp r8, rbx
    jae scan_tok_done
    movzx eax, byte ptr [r8]
    cmp al, 32
    je scan_tok_done
    cmp al, 9
    je scan_tok_done
    jmp scan_tok
scan_tok_done:
    mov rdx, r8
    sub rdx, rsi
    cmp rdx, 1
    jne try_number
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je is_op
    cmp al, '-'
    je is_op
    cmp al, '*'
    je is_op
    cmp al, '/'
    je is_op
    cmp al, '%'
    je is_op
try_number:
    mov r9, rsi
    xor r15d, r15d
    cmp byte ptr [r9], '-'
    jne parse_digits
    mov r15d, 1
    inc r9
parse_digits:
    xor eax, eax
    xor ecx, ecx
digit_loop:
    cmp r9, r8
    jae digits_done
    movzx edx, byte ptr [r9]
    sub edx, 48
    cmp edx, 9
    ja invalid
    inc ecx
    cmp ecx, 18
    ja invalid
    imul rax, rax, 10
    add rax, rdx
    inc r9
    jmp digit_loop
digits_done:
    test ecx, ecx
    jz invalid
    test r15d, r15d
    jz push_num
    neg rax
push_num:
    mov qword ptr [r14], rax
    add r14, 8
    mov rsi, r8
    jmp parse_loop
is_op:
    mov r9d, eax
    mov rax, r14
    sub rax, rbp
    cmp rax, 16
    jb invalid
    sub r14, 16
    mov rax, qword ptr [r14]
    mov rcx, qword ptr [r14 + 8]
    cmp r9b, '+'
    je op_add
    cmp r9b, '-'
    je op_sub
    cmp r9b, '*'
    je op_mul
    test rcx, rcx
    jz invalid
    cmp rcx, -1
    jne div_ok
    mov rdx, 1
    shl rdx, 63
    cmp rax, rdx
    je invalid
div_ok:
    cqo
    idiv rcx
    cmp r9b, '%'
    jne push_result
    mov rax, rdx
    jmp push_result
op_add:
    add rax, rcx
    jmp push_result
op_sub:
    sub rax, rcx
    jmp push_result
op_mul:
    imul rax, rcx
push_result:
    mov qword ptr [r14], rax
    add r14, 8
    mov rsi, r8
    jmp parse_loop
line_finish:
    mov rax, r14
    sub rax, rbp
    cmp rax, 8
    jne invalid
    mov rax, qword ptr [rbp]
    call print_num
    ret
invalid:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 6
    syscall
    ret

print_num:
    lea rdi, [rip + num_buf]
    add rdi, 30
    mov byte ptr [rdi], 10
    test rax, rax
    jnz pn_nonzero
    dec rdi
    mov byte ptr [rdi], '0'
    jmp pn_write
pn_nonzero:
    xor r9d, r9d
    test rax, rax
    jns pn_conv
    mov r9d, 1
    neg rax
pn_conv:
    mov ecx, 10
pn_conv_loop:
    xor rdx, rdx
    div rcx
    add edx, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pn_conv_loop
    test r9d, r9d
    jz pn_write
    dec rdi
    mov byte ptr [rdi], '-'
pn_write:
    lea rdx, [rip + num_buf]
    add rdx, 31
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    ret

.section .rodata
err_msg:
    .ascii "error\n"

.bss
.balign 16
input_buf:
    .space 65536
.balign 8
rpn_stack:
    .space 8192
num_buf:
    .space 32
