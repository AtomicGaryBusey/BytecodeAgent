.globl _start

.equ INPUT_MAX, 65536

.text
.p2align 2
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0

read_loop:
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    mov x0, #0
    add x1, x19, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, #0
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ne report_error

    mov x8, #64
    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0

report_error:
    mov x8, #64
    mov x0, #1
    adrp x1, error_prefix
    add x1, x1, :lo12:error_prefix
    mov x2, #9
    svc #0
    sub sp, sp, #16
    add x1, sp, #16
    mov x3, x21
conv_loop:
    mov x4, #10
    udiv x5, x3, x4
    msub x6, x5, x4, x3
    add w6, w6, #'0'
    sub x1, x1, #1
    strb w6, [x1]
    mov x3, x5
    cbnz x3, conv_loop
    add x2, sp, #16
    sub x2, x2, x1
    mov x8, #64
    mov x0, #1
    svc #0
    mov w0, #10
    strb w0, [sp]
    mov x8, #64
    mov x0, #1
    mov x1, sp
    mov x2, #1
    svc #0
    mov x8, #93
    mov x0, #1
    svc #0

skip_ws:
    cmp x21, x20
    b.ge skip_ws_done
    ldrb w0, [x19, x21]
    cmp w0, #0x20
    b.eq skip_ws_eat
    cmp w0, #0x09
    b.eq skip_ws_eat
    cmp w0, #0x0A
    b.eq skip_ws_eat
    cmp w0, #0x0D
    b.eq skip_ws_eat
skip_ws_done:
    ret
skip_ws_eat:
    add x21, x21, #1
    b skip_ws

parse_value:
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'{'
    b.eq parse_object
    cmp w0, #'['
    b.eq parse_array
    cmp w0, #'"'
    b.eq parse_string
    cmp w0, #'-'
    b.eq parse_number
    cmp w0, #'0'
    b.lt parse_value_kw
    cmp w0, #'9'
    b.le parse_number
parse_value_kw:
    cmp w0, #'t'
    b.eq parse_true
    cmp w0, #'f'
    b.eq parse_false
    cmp w0, #'n'
    b.eq parse_null
    b report_error

parse_object:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'}'
    b.eq object_empty
object_member:
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.ne report_error
    bl parse_string
    bl skip_ws
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #':'
    b.ne report_error
    add x21, x21, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #','
    b.eq object_comma
    cmp w0, #'}'
    b.ne report_error
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret
object_comma:
    add x21, x21, #1
    bl skip_ws
    b object_member
object_empty:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
    bl skip_ws
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #']'
    b.eq array_empty
array_element:
    bl parse_value
    bl skip_ws
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #','
    b.eq array_comma
    cmp w0, #']'
    b.ne report_error
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret
array_comma:
    add x21, x21, #1
    bl skip_ws
    b array_element
array_empty:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret

parse_string:
    add x21, x21, #1
string_loop:
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq string_end
    cmp w0, #0x5C
    b.eq string_esc
    cmp w0, #0x20
    b.lt report_error
    add x21, x21, #1
    b string_loop
string_end:
    add x21, x21, #1
    ret
string_esc:
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'"'
    b.eq string_esc_ok
    cmp w0, #0x5C
    b.eq string_esc_ok
    cmp w0, #'/'
    b.eq string_esc_ok
    cmp w0, #'b'
    b.eq string_esc_ok
    cmp w0, #'f'
    b.eq string_esc_ok
    cmp w0, #'n'
    b.eq string_esc_ok
    cmp w0, #'r'
    b.eq string_esc_ok
    cmp w0, #'t'
    b.eq string_esc_ok
    cmp w0, #'u'
    b.eq string_unicode
    b report_error
string_esc_ok:
    add x21, x21, #1
    b string_loop
string_unicode:
    add x21, x21, #1
    mov x3, #4
string_hex:
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    sub w2, w0, #'0'
    cmp w2, #10
    b.lo string_hex_ok
    sub w2, w0, #'A'
    cmp w2, #6
    b.lo string_hex_ok
    sub w2, w0, #'a'
    cmp w2, #6
    b.lo string_hex_ok
    b report_error
string_hex_ok:
    add x21, x21, #1
    subs x3, x3, #1
    b.ne string_hex
    b string_loop

parse_number:
    ldrb w0, [x19, x21]
    cmp w0, #'-'
    b.ne num_int
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
num_int:
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
    add x21, x21, #1
    cmp w0, #'0'
    b.eq num_after_int
num_more_int:
    cmp x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt num_after_int
    cmp w0, #'9'
    b.gt num_after_int
    add x21, x21, #1
    b num_more_int
num_after_int:
    cmp x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp w0, #'.'
    b.ne num_try_exp
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
num_frac_loop:
    add x21, x21, #1
    cmp x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt num_try_exp
    cmp w0, #'9'
    b.gt num_try_exp
    b num_frac_loop
num_try_exp:
    cmp x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.eq num_exp
    cmp w0, #'E'
    b.eq num_exp
    b num_done
num_exp:
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'+'
    b.eq num_exp_sign
    cmp w0, #'-'
    b.ne num_exp_need_digit
num_exp_sign:
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
num_exp_need_digit:
    cmp w0, #'0'
    b.lt report_error
    cmp w0, #'9'
    b.gt report_error
num_exp_loop:
    add x21, x21, #1
    cmp x21, x20
    b.ge num_done
    ldrb w0, [x19, x21]
    cmp w0, #'0'
    b.lt num_done
    cmp w0, #'9'
    b.gt num_done
    b num_exp_loop
num_done:
    ret

parse_true:
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'r'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'u'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.ne report_error
    add x21, x21, #1
    ret

parse_false:
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'a'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'l'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'s'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'e'
    b.ne report_error
    add x21, x21, #1
    ret

parse_null:
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'u'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'l'
    b.ne report_error
    add x21, x21, #1
    cmp x21, x20
    b.ge report_error
    ldrb w0, [x19, x21]
    cmp w0, #'l'
    b.ne report_error
    add x21, x21, #1
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
error_prefix:
    .ascii "error at "

.bss
.p2align 4
input:
    .space 65536
