.globl _start
.text
_start:
    xor r12, r12
    xor r13, r13
    xor r14, r14
    xor r15, r15
.Lread:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 4096
    syscall
    test rax, rax
    jle .Ldone
    add r14, rax
    lea rsi, [rip + buf]
    mov rcx, rax
.Lbyte:
    movzx eax, byte ptr [rsi]
    cmp al, 10
    jne .Lnotnl
    inc r12
.Lnotnl:
    cmp al, 32
    je .Lws
    cmp al, 9
    jb .Lnotws
    cmp al, 13
    ja .Lnotws
.Lws:
    xor r15, r15
    jmp .Lnext
.Lnotws:
    test r15, r15
    jnz .Lnext
    inc r13
    mov r15, 1
.Lnext:
    inc rsi
    dec rcx
    jnz .Lbyte
    jmp .Lread
.Ldone:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call itoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call itoa
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call itoa
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
itoa:
    lea r8, [rip + numtmp + 20]
    mov rsi, r8
    mov ecx, 10
    test rax, rax
    jnz .Ldiv
    dec rsi
    mov byte ptr [rsi], 48
    jmp .Lcopy
.Ldiv:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz .Ldiv
.Lcopy:
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    cmp rsi, r8
    jne .Lcopy
    ret
.bss
buf:
    .space 4096
numtmp:
    .space 32
outbuf:
    .space 64
