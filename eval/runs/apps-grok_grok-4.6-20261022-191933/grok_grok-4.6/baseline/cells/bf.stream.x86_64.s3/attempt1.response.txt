.globl _start
.text
_start:
    lea r12, [rip + input_buf]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub edx, r13d
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor ecx, ecx
find_bang:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [r12 + rcx], '!'
    je got_bang
    inc rcx
    jmp find_bang
got_bang:
    mov r15, rcx
    lea rbx, [r12 + rcx + 1]
    lea r8, [r12 + r13]
    jmp check_br
no_bang:
    mov r15, r13
    lea rbx, [r12 + r13]
    mov r8, rbx
check_br:
    lea r9, [rip + jump_table]
    mov rbp, rsp
    xor ecx, ecx
br_loop:
    cmp rcx, r15
    jae br_end
    mov al, byte ptr [r12 + rcx]
    cmp al, '['
    je br_open
    cmp al, ']'
    je br_close
    inc rcx
    jmp br_loop
br_open:
    push rcx
    inc rcx
    jmp br_loop
br_close:
    cmp rsp, rbp
    je fail
    pop rdx
    mov dword ptr [r9 + rdx*4], ecx
    mov dword ptr [r9 + rcx*4], edx
    inc rcx
    jmp br_loop
br_end:
    cmp rsp, rbp
    jne fail
    xor r13d, r13d
    lea r14, [rip + tape]
.p2align 4
run:
    cmp r13, r15
    jae ok
    movzx eax, byte ptr [r12 + r13]
    cmp al, '>'
    je op_gt
    cmp al, '<'
    je op_lt
    cmp al, '+'
    je op_plus
    cmp al, '-'
    je op_minus
    cmp al, '.'
    je op_dot
    cmp al, ','
    je op_comma
    cmp al, '['
    je op_lbr
    cmp al, ']'
    je op_rbr
    inc r13
    jmp run
op_gt:
    inc r14
    inc r13
    jmp run
op_lt:
    dec r14
    inc r13
    jmp run
op_plus:
    inc byte ptr [r14]
    inc r13
    jmp run
op_minus:
    dec byte ptr [r14]
    inc r13
    jmp run
op_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, 1
    syscall
    inc r13
    jmp run
op_comma:
    cmp rbx, r8
    jae comma_zero
    mov al, byte ptr [rbx]
    mov byte ptr [r14], al
    inc rbx
    inc r13
    jmp run
comma_zero:
    mov byte ptr [r14], 0
    inc r13
    jmp run
op_lbr:
    cmp byte ptr [r14], 0
    jne lbr_skip
    mov r13d, dword ptr [r9 + r13*4]
lbr_skip:
    inc r13
    jmp run
op_rbr:
    cmp byte ptr [r14], 0
    je rbr_skip
    mov r13d, dword ptr [r9 + r13*4]
rbr_skip:
    inc r13
    jmp run
ok:
    mov eax, 60
    xor edi, edi
    syscall
fail:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
input_buf:
    .space 65536
tape:
    .space 30000
jump_table:
    .space 262144
