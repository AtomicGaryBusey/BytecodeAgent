.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536
.balign 16
outbuf:
    .space 327680

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, hexdigits
    add x22, x22, :lo12:hexdigits
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x27, x23
    mov w28, #0x20
    orr x28, x28, x28, lsl #8
    orr x28, x28, x28, lsl #16
    orr x28, x28, x28, lsl #32
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs print_total

    sub x24, x20, x21
    mov x0, #16
    cmp x24, x0
    csel x24, x24, x0, lo

    stp x28, x28, [x23]
    stp x28, x28, [x23, #16]
    stp x28, x28, [x23, #32]
    stp x28, x28, [x23, #48]
    stp x28, x28, [x23, #64]

    mov w0, w21
    mov x1, x23
    bl write_hex8

    add x26, x19, x21
    mov x25, #0

byte_loop:
    cmp x25, x24
    b.hs bytes_done
    ldrb w0, [x26, x25]
    add x1, x23, #10
    add x1, x1, x25, lsl #1
    add x1, x1, x25
    cmp x25, #8
    csinc x1, x1, x1, lo
    lsr w3, w0, #4
    ldrb w3, [x22, x3]
    strb w3, [x1]
    and w3, w0, #15
    ldrb w3, [x22, x3]
    strb w3, [x1, #1]
    mov w3, #0x2e
    cmp w0, #0x20
    b.lo 2f
    cmp w0, #0x7e
    b.hi 2f
    mov w3, w0
2:
    add x1, x23, #61
    strb w3, [x1, x25]
    add x25, x25, #1
    b byte_loop

bytes_done:
    mov w3, #0x7c
    strb w3, [x23, #60]
    add x1, x23, #61
    add x1, x1, x24
    strb w3, [x1]
    mov w3, #0x0a
    strb w3, [x1, #1]
    add x23, x23, #63
    add x23, x23, x24
    add x21, x21, x24
    b row_loop

print_total:
    mov w0, w20
    mov x1, x23
    bl write_hex8
    mov w3, #0x0a
    strb w3, [x23, #8]
    add x23, x23, #9
    mov x0, #1
    mov x1, x27
    sub x2, x23, x27
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

write_hex8:
    mov w2, #28
1:
    lsr w3, w0, w2
    and w3, w3, #15
    ldrb w3, [x22, x3]
    strb w3, [x1], #1
    subs w2, w2, #4
    b.ge 1b
    ret
