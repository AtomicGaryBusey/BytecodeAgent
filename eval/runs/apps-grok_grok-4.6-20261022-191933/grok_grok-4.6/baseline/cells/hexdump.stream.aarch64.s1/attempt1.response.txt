.globl _start

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 327680

.text
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, hexdigits
    add x22, x22, :lo12:hexdigits
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x27, x23
    movz x28, #0x2020
    movk x28, #0x2020, lsl #16
    movk x28, #0x2020, lsl #32
    movk x28, #0x2020, lsl #48
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs print_total

    sub x24, x20, x21
    mov x0, #16
    cmp x24, x0
    csel x24, x24, x0, lo

    str x28, [x23]
    str x28, [x23, #8]
    str x28, [x23, #16]
    str x28, [x23, #24]
    str x28, [x23, #32]
    str x28, [x23, #40]
    str x28, [x23, #48]
    str x28, [x23, #56]

    mov w0, w21
    mov x1, x23
    bl write_hex8

    add x26, x19, x21
    mov x25, #0
byte_loop:
    cmp x25, x24
    b.hs bytes_done
    ldrb w0, [x26, x25]
    add x1, x23, #10
    add x1, x1, x25, lsl #1
    add x1, x1, x25
    cmp x25, #8
    csinc x1, x1, x1, lo
    lsr w3, w0, #4
    ldrb w3, [x22, x3]
    strb w3, [x1]
    and w3, w0, #15
    ldrb w3, [x22, x3]
    strb w3, [x1, #1]
    mov w3, #0x2e
    cmp w0, #0x20
    ccmp w0, #0x7f, #2, hs
    csel w0, w0, w3, lo
    add x1, x23, x25
    strb w0, [x1, #61]
    add x25, x25, #1
    b byte_loop

bytes_done:
    mov w0, #0x7c
    strb w0, [x23, #60]
    add x1, x23, x24
    strb w0, [x1, #61]
    mov w0, #0x0a
    strb w0, [x1, #62]
    add x23, x23, x24
    add x23, x23, #63
    add x21, x21, #16
    b row_loop

print_total:
    mov w0, w20
    mov x1, x23
    bl write_hex8
    mov w0, #0x0a
    strb w0, [x1]
    mov x0, #1
    mov x1, x27
    sub x2, x23, x27
    add x2, x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

write_hex8:
    mov w2, #32
1:
    sub w2, w2, #4
    lsr w3, w0, w2
    and w3, w3, #15
    ldrb w3, [x22, x3]
    strb w3, [x1], #1
    cbnz w2, 1b
    ret
