.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFSIZE, 32768
.equ MAXLINES, 1000

.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    adrp x23, line_ptrs
    add x23, x23, :lo12:line_ptrs
    adrp x24, line_lens
    add x24, x24, :lo12:line_lens

    mov x20, #0
read_loop:
    mov x2, #BUFSIZE
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    add x1, x19, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    mov x22, #0
    mov x25, #0
    mov x26, #0
parse_loop:
    cmp x25, x20
    b.ge parse_end
    ldrb w0, [x19, x25]
    cmp w0, #10
    b.ne parse_next
    cmp x22, #MAXLINES
    b.ge parse_after_save
    add x0, x19, x26
    str x0, [x23, x22, lsl #3]
    sub x0, x25, x26
    str x0, [x24, x22, lsl #3]
    add x22, x22, #1
parse_after_save:
    add x26, x25, #1
parse_next:
    add x25, x25, #1
    b parse_loop
parse_end:
    cmp x26, x20
    b.ge parse_done
    cmp x22, #MAXLINES
    b.ge parse_done
    add x0, x19, x26
    str x0, [x23, x22, lsl #3]
    sub x0, x20, x26
    str x0, [x24, x22, lsl #3]
    add x22, x22, #1
parse_done:

    mov x25, #1
sort_outer:
    cmp x25, x22
    b.ge sort_done
    ldr x9, [x23, x25, lsl #3]
    ldr x10, [x24, x25, lsl #3]
    mov x26, x25
sort_inner:
    cbz x26, sort_insert
    sub x11, x26, #1
    ldr x0, [x23, x11, lsl #3]
    ldr x1, [x24, x11, lsl #3]
    cmp x1, x10
    csel x4, x1, x10, lo
    mov x5, #0
cmp_loop:
    cmp x5, x4
    b.eq cmp_prefix_eq
    ldrb w6, [x0, x5]
    ldrb w7, [x9, x5]
    cmp w6, w7
    b.ne cmp_diff
    add x5, x5, #1
    b cmp_loop
cmp_diff:
    cmp w6, w7
    b.ls sort_insert
    b sort_shift
cmp_prefix_eq:
    cmp x1, x10
    b.ls sort_insert
sort_shift:
    ldr x12, [x23, x11, lsl #3]
    str x12, [x23, x26, lsl #3]
    ldr x12, [x24, x11, lsl #3]
    str x12, [x24, x26, lsl #3]
    mov x26, x11
    b sort_inner
sort_insert:
    str x9, [x23, x26, lsl #3]
    str x10, [x24, x26, lsl #3]
    add x25, x25, #1
    b sort_outer
sort_done:

    adr x27, newline
    mov x25, #0
print_loop:
    cmp x25, x22
    b.ge print_done
    ldr x2, [x24, x25, lsl #3]
    cbz x2, print_nl
    ldr x1, [x23, x25, lsl #3]
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
print_nl:
    mov x0, #1
    mov x1, x27
    mov x2, #1
    mov x8, #SYS_WRITE
    svc #0
    add x25, x25, #1
    b print_loop
print_done:

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 8
input:
    .space 32768
line_ptrs:
    .space 8000
line_lens:
    .space 8000
