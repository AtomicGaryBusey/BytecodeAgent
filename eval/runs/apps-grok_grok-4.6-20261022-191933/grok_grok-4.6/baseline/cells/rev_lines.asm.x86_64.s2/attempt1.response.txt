.globl _start

.text

_start:
    lea r12, [rip + buf]
    xor ebx, ebx

.Lread:
    mov edx, 65536
    sub edx, ebx
    jz .Lprocess
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    syscall
    test rax, rax
    jle .Lprocess
    add rbx, rax
    jmp .Lread

.Lprocess:
    xor r13d, r13d
    xor r14d, r14d

.Lscan:
    cmp r14, rbx
    je .Lafter_scan
    cmp byte ptr [r12 + r14], 0x0A
    je .Ldo_rev
    inc r14
    jmp .Lscan

.Ldo_rev:
    call reverse
    inc r14
    mov r13, r14
    jmp .Lscan

.Lafter_scan:
    mov r14, rbx
    call reverse

    test rbx, rbx
    jz .Lexit

    mov eax, 1
    mov edi, 1
    mov rsi, r12
    mov rdx, rbx
    syscall

    cmp byte ptr [r12 + rbx - 1], 0x0A
    je .Lexit

    push 0x0A
    mov eax, 1
    mov edi, 1
    mov rsi, rsp
    mov edx, 1
    syscall

.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

reverse:
    mov r8, r13
    mov r9, r14
    cmp r8, r9
    jae .Lrev_done
    dec r9
.Lrev_loop:
    cmp r8, r9
    jae .Lrev_done
    mov al, byte ptr [r12 + r8]
    mov dl, byte ptr [r12 + r9]
    mov byte ptr [r12 + r8], dl
    mov byte ptr [r12 + r9], al
    inc r8
    dec r9
    jmp .Lrev_loop
.Lrev_done:
    ret

.bss
buf:
    .space 65536
