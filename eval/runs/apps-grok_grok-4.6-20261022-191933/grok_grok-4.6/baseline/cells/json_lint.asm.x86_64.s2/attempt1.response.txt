.globl _start

.equ BUF_SIZE, 65536

.text
_start:
    lea r12, [rip + input_buf]
    xor r15, r15
read_loop:
    mov rdx, BUF_SIZE
    sub rdx, r15
    jbe read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r15]
    syscall
    test rax, rax
    jle read_done
    add r15, rax
    jmp read_loop
read_done:
    mov r13, r12
    lea r14, [r12 + r15]
    call skip_ws
    cmp r13, r14
    je error
    call parse_value
    call skip_ws
    cmp r13, r14
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

skip_ws:
    cmp r13, r14
    jae skip_ws_done
    movzx eax, byte ptr [r13]
    cmp al, 0x20
    je skip_ws_eat
    cmp al, 0x09
    je skip_ws_eat
    cmp al, 0x0A
    je skip_ws_eat
    cmp al, 0x0D
    je skip_ws_eat
skip_ws_done:
    ret
skip_ws_eat:
    inc r13
    jmp skip_ws

parse_value:
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    jmp parse_number

parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae error
    cmp byte ptr [r13], '}'
    je obj_end
obj_member:
    cmp r13, r14
    jae error
    cmp byte ptr [r13], 0x22
    jne error
    call parse_string
    call skip_ws
    cmp r13, r14
    jae error
    cmp byte ptr [r13], ':'
    jne error
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, '}'
    je obj_end
    cmp al, ','
    jne error
    inc r13
    call skip_ws
    jmp obj_member
obj_end:
    inc r13
    ret

parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae error
    cmp byte ptr [r13], ']'
    je arr_end
arr_item:
    call parse_value
    call skip_ws
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, ']'
    je arr_end
    cmp al, ','
    jne error
    inc r13
    call skip_ws
    jmp arr_item
arr_end:
    inc r13
    ret

parse_string:
    inc r13
str_loop:
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, 0x22
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb error
    inc r13
    jmp str_loop
str_end:
    inc r13
    ret
str_esc:
    inc r13
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, 0x22
    je str_esc_ok
    cmp al, 0x5C
    je str_esc_ok
    cmp al, '/'
    je str_esc_ok
    cmp al, 'b'
    je str_esc_ok
    cmp al, 'f'
    je str_esc_ok
    cmp al, 'n'
    je str_esc_ok
    cmp al, 'r'
    je str_esc_ok
    cmp al, 't'
    je str_esc_ok
    cmp al, 'u'
    je str_uni
    jmp error
str_esc_ok:
    inc r13
    jmp str_loop
str_uni:
    inc r13
    mov ecx, 4
str_hex:
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb error
    cmp al, '9'
    jbe str_hex_ok
    or al, 0x20
    cmp al, 'a'
    jb error
    cmp al, 'f'
    ja error
str_hex_ok:
    inc r13
    dec ecx
    jnz str_hex
    jmp str_loop

parse_number:
    cmp r13, r14
    jae error
    cmp byte ptr [r13], '-'
    jne num_int
    inc r13
    cmp r13, r14
    jae error
num_int:
    movzx eax, byte ptr [r13]
    cmp al, '0'
    je num_zero
    cmp al, '1'
    jb error
    cmp al, '9'
    ja error
    inc r13
num_intd:
    cmp r13, r14
    jae num_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb num_frac
    cmp al, '9'
    ja num_frac
    inc r13
    jmp num_intd
num_zero:
    inc r13
num_frac:
    cmp r13, r14
    jae num_done
    cmp byte ptr [r13], '.'
    jne num_exp
    inc r13
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc r13
num_fracd:
    cmp r13, r14
    jae num_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb num_exp
    cmp al, '9'
    ja num_exp
    inc r13
    jmp num_fracd
num_exp:
    cmp r13, r14
    jae num_done
    movzx eax, byte ptr [r13]
    cmp al, 'e'
    je num_expdo
    cmp al, 'E'
    je num_expdo
    jmp num_done
num_expdo:
    inc r13
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, '+'
    je num_exps
    cmp al, '-'
    je num_exps
    jmp num_expneed
num_exps:
    inc r13
num_expneed:
    cmp r13, r14
    jae error
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb error
    cmp al, '9'
    ja error
    inc r13
num_expd:
    cmp r13, r14
    jae num_done
    movzx eax, byte ptr [r13]
    cmp al, '0'
    jb num_done
    cmp al, '9'
    ja num_done
    inc r13
    jmp num_expd
num_done:
    ret

parse_true:
    lea rsi, [rip + lit_rue]
    mov ecx, 3
    jmp parse_lit
parse_false:
    lea rsi, [rip + lit_alse]
    mov ecx, 4
    jmp parse_lit
parse_null:
    lea rsi, [rip + lit_ull]
    mov ecx, 3
parse_lit:
    inc r13
parse_lit_loop:
    cmp r13, r14
    jae error
    movzx eax, byte ptr [rsi]
    cmp byte ptr [r13], al
    jne error
    inc r13
    inc rsi
    dec ecx
    jnz parse_lit_loop
    ret

error:
    mov rax, r13
    sub rax, r12
    lea rbx, [rip + out_buf + 15]
    mov byte ptr [rbx], 10
    dec rbx
    mov ecx, 10
cvt:
    xor edx, edx
    div rcx
    add dl, '0'
    mov byte ptr [rbx], dl
    dec rbx
    test rax, rax
    jnz cvt
    inc rbx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + msg_err]
    mov edx, 9
    syscall
    lea rdx, [rip + out_buf + 16]
    sub rdx, rbx
    mov rsi, rbx
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_rue:
    .ascii "rue"
lit_alse:
    .ascii "alse"
lit_ull:
    .ascii "ull"

.bss
.balign 16
input_buf:
    .space BUF_SIZE
out_buf:
    .space 16
