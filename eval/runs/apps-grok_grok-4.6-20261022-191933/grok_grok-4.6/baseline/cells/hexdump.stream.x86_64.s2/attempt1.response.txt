.globl _start
.text
_start:
    lea r8, [rip + hexdigits]
    xor r12, r12
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
row_loop:
    cmp r13, r12
    jae print_total
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe 1f
    mov r14, 16
1:
    lea r15, [rip + input_buf]
    add r15, r13
    lea rdi, [rip + line_buf]
    mov eax, r13d
    mov ecx, 8
2:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 2b
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    add rdi, 2
    xor ebx, ebx
hex_slot:
    cmp ebx, r14d
    jae blank_slot
    movzx eax, byte ptr [r15 + rbx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    and eax, 0xf
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    jmp slot_done
blank_slot:
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
slot_done:
    inc ebx
    cmp ebx, 8
    jne 3f
    mov byte ptr [rdi], ' '
    inc rdi
3:
    cmp ebx, 16
    jb hex_slot
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi + 1], '|'
    add rdi, 2
    xor ebx, ebx
ascii_loop:
    cmp ebx, r14d
    jae ascii_done
    movzx eax, byte ptr [r15 + rbx]
    mov edx, '.'
    cmp eax, 0x20
    jb 4f
    cmp eax, 0x7e
    ja 4f
    mov edx, eax
4:
    mov byte ptr [rdi], dl
    inc rdi
    inc ebx
    jmp ascii_loop
ascii_done:
    mov byte ptr [rdi], '|'
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2
    mov rdx, rdi
    lea rsi, [rip + line_buf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    add r13, r14
    jmp row_loop
print_total:
    lea rdi, [rip + line_buf]
    mov eax, r12d
    mov ecx, 8
5:
    rol eax, 4
    mov edx, eax
    and edx, 0xf
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 5b
    mov byte ptr [rdi], 0x0a
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov edx, 9
    syscall
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hexdigits:
    .ascii "0123456789abcdef"

.bss
input_buf:
    .space 65536
line_buf:
    .space 80
