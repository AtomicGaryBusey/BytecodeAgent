.globl _start

.text
_start:
    lea r12, [rip + buf]
    xor r13, r13

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r14, r14

process_loop:
    cmp r14, r13
    jge finished
    mov r15, r14

find_nl:
    cmp r14, r13
    jge last_line
    cmp byte ptr [r12 + r14], 10
    je found_nl
    inc r14
    jmp find_nl

found_nl:
    call emit_reversed
    inc r14
    jmp process_loop

last_line:
    cmp r15, r13
    je finished
    call emit_reversed

finished:
    mov eax, 60
    xor edi, edi
    syscall

emit_reversed:
    lea rdi, [r12 + r15]
    lea rsi, [r12 + r14]
    cmp rdi, rsi
    je emit_nl
    dec rsi
rev_loop:
    cmp rdi, rsi
    jge write_line
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp rev_loop

write_line:
    mov eax, 1
    mov edi, 1
    lea rsi, [r12 + r15]
    mov rdx, r14
    sub rdx, r15
    syscall

emit_nl:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    ret

.section .rodata
newline:
    .byte 10

.bss
buf:
    .space 65536
