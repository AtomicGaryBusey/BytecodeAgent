.globl _start

.text
_start:
    xor ebx, ebx
read_loop:
    mov edx, 65536
    sub rdx, rbx
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r15, [rip + hextab]
    lea rdi, [rip + outbuf]
    xor r12, r12

print_rows:
    cmp r12, rbx
    jae print_total

    mov r13, rbx
    sub r13, r12
    cmp r13, 16
    jbe 1f
    mov r13, 16
1:
    lea r14, [rip + inbuf]
    add r14, r12

    mov eax, r12d
    call put_hex8

    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor ecx, ecx
hex_slots:
    cmp ecx, 16
    jae hex_slots_done

    cmp ecx, r13d
    jae slot_empty

    movzx eax, byte ptr [r14 + rcx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    movzx eax, byte ptr [r15 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp slot_after

slot_empty:
    mov dword ptr [rdi], 0x20202020
    add rdi, 3

slot_after:
    cmp ecx, 7
    jne 2f
    mov byte ptr [rdi], 32
    inc rdi
2:
    inc ecx
    jmp hex_slots

hex_slots_done:
    mov byte ptr [rdi], 32
    inc rdi
    mov byte ptr [rdi], 124
    inc rdi

    xor ecx, ecx
ascii_loop:
    cmp ecx, r13d
    jae ascii_done
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    jmp ascii_put
ascii_dot:
    mov al, 46
ascii_put:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 124
    inc rdi
    mov byte ptr [rdi], 10
    inc rdi

    add r12, r13
    jmp print_rows

print_total:
    mov eax, ebx
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

put_hex8:
    mov ecx, 8
3:
    rol eax, 4
    mov edx, eax
    and edx, 15
    movzx edx, byte ptr [r15 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz 3b
    ret

.section .rodata
hextab:
    .ascii "0123456789abcdef"

.bss
.balign 16
inbuf:
    .space 65536
outbuf:
    .space 327680
