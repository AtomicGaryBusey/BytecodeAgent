export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %ahi, %blo
	%p2 =l mul %alo, %bhi
	%p3 =l mul %ahi, %bhi
	%t0 =l shr %p0, 32
	%m1 =l and %p1, 4294967295
	%m2 =l and %p2, 4294967295
	%t1 =l add %t0, %m1
	%t2 =l add %t1, %m2
	%carry =l shr %t2, 32
	%h1 =l shr %p1, 32
	%h2 =l shr %p2, 32
	%r0 =l add %p3, %h1
	%r1 =l add %r0, %h2
	%r =l add %r1, %carry
	ret %r
}
