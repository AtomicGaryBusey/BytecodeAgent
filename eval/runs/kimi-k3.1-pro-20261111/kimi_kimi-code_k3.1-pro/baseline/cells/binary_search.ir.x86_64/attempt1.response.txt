export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	%hi0 =l copy %n
	jmp @loop
@loop
	%lo =l phi @start %lo0, @setlow %lo1, @sethigh %lo2
	%hi =l phi @start %hi0, @setlow %hi1, @sethigh %hi2
	%nonempty =w csltl %lo, %hi
	jnz %nonempty, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @setlow, @sethigh
@setlow
	%lo1 =l add %mid, 1
	%hi1 =l copy %hi
	jmp @loop
@sethigh
	%lo2 =l copy %lo
	%hi2 =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
