mp_mul:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov r12, rdi
mov r15, rsi
mov r13, rdx
mov rbp, rcx
mov r14, r8
lea r11, [r15 + rbp]
xor ecx, ecx
.Lzero:
cmp rcx, r11
jae .Lzero_done
mov qword ptr [r14 + rcx*8], 0
inc rcx
jmp .Lzero
.Lzero_done:
xor ebx, ebx
.Louter:
cmp rbx, r15
jae .Lscan_setup
mov r10, qword ptr [r12 + rbx*8]
xor ecx, ecx
xor r9d, r9d
.Linner:
cmp rcx, rbp
jae .Linner_done
mov rax, r10
mul qword ptr [r13 + rcx*8]
lea r11, [rbx + rcx]
add rax, qword ptr [r14 + r11*8]
adc rdx, 0
add rax, r9
adc rdx, 0
mov qword ptr [r14 + r11*8], rax
mov r9, rdx
inc rcx
jmp .Linner
.Linner_done:
lea r11, [rbx + rbp]
mov qword ptr [r14 + r11*8], r9
inc rbx
jmp .Louter
.Lscan_setup:
lea rax, [r15 + rbp]
.Lscan:
test rax, rax
jz .Ldone
cmp qword ptr [r14 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lscan
.Ldone:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
