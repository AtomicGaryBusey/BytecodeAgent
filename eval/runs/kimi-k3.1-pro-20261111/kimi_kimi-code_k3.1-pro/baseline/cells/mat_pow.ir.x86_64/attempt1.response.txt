export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
	%nn =l mul %n, %n
	%nnbytes =l shl %nn, 3
	%prod =l add %tmp, %nnbytes
	jmp @copy_base_test

@copy_base_test
	%cbi =l phi @entry 0, @copy_base_body %cbinext
	%cbcond =w csltl %cbi, %nn
	jnz %cbcond, @copy_base_body, @identity_test

@copy_base_body
	%cboffset =l shl %cbi, 3
	%cbsrc =l add %a, %cboffset
	%cbdst =l add %tmp, %cboffset
	%cbvalue =l loadl %cbsrc
	storel %cbvalue, %cbdst
	%cbinext =l add %cbi, 1
	jmp @copy_base_test

@identity_test
	%ii =l phi @copy_base_test 0, @identity_next %iinext
	%icond =w csltl %ii, %nn
	jnz %icond, @identity_body, @power_test

@identity_body
	%irow =l div %ii, %n
	%irown =l mul %irow, %n
	%icol =l sub %ii, %irown
	%idoffset =l shl %ii, 3
	%idaddr =l add %out, %idoffset
	%idiag =w ceql %irow, %icol
	jnz %idiag, @identity_one, @identity_zero

@identity_one
	storel 1, %idaddr
	jmp @identity_next

@identity_zero
	storel 0, %idaddr
	jmp @identity_next

@identity_next
	%iinext =l add %ii, 1
	jmp @identity_test

@power_test
	%count =l phi @identity_test %k, @iteration_done %countnext
	%powercond =w csltl 0, %count
	jnz %powercond, @multiply_element_test, @zero_test

@multiply_element_test
	%me =l phi @power_test 0, @multiply_element_store %menext
	%mecond =w csltl %me, %nn
	jnz %mecond, @multiply_element_init, @copy_product_test

@multiply_element_init
	%merow =l div %me, %n
	%merown =l mul %merow, %n
	%mecol =l sub %me, %merown
	jmp @multiply_inner_test

@multiply_inner_test
	%mit =l phi @multiply_element_init 0, @multiply_inner_body %mitnext
	%misum =l phi @multiply_element_init 0, @multiply_inner_body %misumnext
	%micond =w csltl %mit, %n
	jnz %micond, @multiply_inner_body, @multiply_element_store

@multiply_inner_body
	%mixbase =l mul %merow, %n
	%mixindex =l add %mixbase, %mit
	%mixoffset =l shl %mixindex, 3
	%mixaddr =l add %out, %mixoffset
	%mix =l loadl %mixaddr
	%miybase =l mul %mit, %n
	%miyindex =l add %miybase, %mecol
	%miyoffset =l shl %miyindex, 3
	%miyaddr =l add %tmp, %miyoffset
	%miy =l loadl %miyaddr
	%miterm =l mul %mix, %miy
	%misumnext =l add %misum, %miterm
	%mitnext =l add %mit, 1
	jmp @multiply_inner_test

@multiply_element_store
	%meoffset =l shl %me, 3
	%meaddr =l add %prod, %meoffset
	storel %misum, %meaddr
	%menext =l add %me, 1
	jmp @multiply_element_test

@copy_product_test
	%cpi =l phi @multiply_element_test 0, @copy_product_body %cpinext
	%cpcond =w csltl %cpi, %nn
	jnz %cpcond, @copy_product_body, @iteration_done

@copy_product_body
	%cpoffset =l shl %cpi, 3
	%cpsrc =l add %prod, %cpoffset
	%cpdst =l add %out, %cpoffset
	%cpvalue =l loadl %cpsrc
	storel %cpvalue, %cpdst
	%cpinext =l add %cpi, 1
	jmp @copy_product_test

@iteration_done
	%countnext =l sub %count, 1
	jmp @power_test

@zero_test
	%zi =l phi @power_test 0, @zero_body %zinext
	%zlimit =l shl %nn, 1
	%zcond =w csltl %zi, %zlimit
	jnz %zcond, @zero_body, @return

@zero_body
	%zoffset =l shl %zi, 3
	%zaddr =l add %tmp, %zoffset
	storel 0, %zaddr
	%zinext =l add %zi, 1
	jmp @zero_test

@return
	ret 0
}
