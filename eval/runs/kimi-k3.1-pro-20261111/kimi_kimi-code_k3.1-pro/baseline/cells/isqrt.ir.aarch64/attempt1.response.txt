export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @next %lo_next
	%hi =l phi @start 4294967295, @next %hi_next
	%done =w ceql %lo, %hi
	jnz %done, @exit, @body
@body
	%sum =l add %lo, %hi
	%sum1 =l add %sum, 1
	%mid =l shr %sum1, 1
	%square =l mul %mid, %mid
	%ok =w culel %square, %n
	jnz %ok, @take, @drop
@take
	jmp @next
@drop
	%mid_minus_one =l sub %mid, 1
	jmp @next
@next
	%lo_next =l phi @take %mid, @drop %lo
	%hi_next =l phi @take %hi, @drop %mid_minus_one
	jmp @loop
@exit
	ret %lo
}
