export function l $collatz_max(l %n) {
@start
	%k0 =l copy 1
	%max0 =l copy 0
	%steps0 =l copy 0
	jmp @outer
@outer
	%k =l phi @start %k0, @merge %k1
	%max =l phi @start %max0, @merge %maxnext
	%done =w csltl %n, %k
	jnz %done, @return, @body
@body
	jmp @inner
@inner
	%x =l phi @body %k, @next %xnext
	%steps =l phi @body %steps0, @next %stepsnext
	%isone =w ceql %x, 1
	jnz %isone, @update, @step
@step
	%parity =l and %x, 1
	%isodd =w ceql %parity, 1
	jnz %isodd, @odd, @even
@odd
	%x3 =l mul %x, 3
	%xo =l add %x3, 1
	jmp @next
@even
	%xe =l div %x, 2
	jmp @next
@next
	%xnext =l phi @odd %xo, @even %xe
	%stepsnext =l add %steps, 1
	jmp @inner
@update
	%greater =w csltl %max, %steps
	jnz %greater, @newmax, @keepmax
@newmax
	jmp @merge
@keepmax
	jmp @merge
@merge
	%maxnext =l phi @newmax %steps, @keepmax %max
	%k1 =l add %k, 1
	jmp @outer
@return
	ret %max
}
