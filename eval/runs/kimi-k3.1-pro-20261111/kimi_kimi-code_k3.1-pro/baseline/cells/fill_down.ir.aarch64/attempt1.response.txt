export function l $fill_down(l %a, l %n, l %start) {
@entry
	jmp @loop
@loop
	%i =l phi @entry 0, @body %next
	%done =w ceql %i, %n
	jnz %done, @exit, @body
@body
	%value =l sub %start, %i
	%offset =l mul %i, 8
	%address =l add %a, %offset
	storel %value, %address
	%next =l add %i, 1
	jmp @loop
@exit
	ret %n
}
