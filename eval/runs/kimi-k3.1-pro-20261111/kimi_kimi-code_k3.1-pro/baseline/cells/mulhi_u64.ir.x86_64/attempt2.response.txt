export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%t0 =l shr %p00, 32
	%t1 =l and %p01, 4294967295
	%t2 =l and %p10, 4294967295
	%t3 =l add %t0, %t1
	%t4 =l add %t3, %t2
	%carry =l shr %t4, 32
	%h0 =l shr %p01, 32
	%h1 =l shr %p10, 32
	%h2 =l add %p11, %h0
	%h3 =l add %h2, %h1
	%h4 =l add %h3, %carry
	ret %h4
}
