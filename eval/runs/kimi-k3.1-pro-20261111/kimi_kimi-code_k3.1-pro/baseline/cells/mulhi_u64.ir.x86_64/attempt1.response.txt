export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 = and %a, 4294967295
	%a1 = shr %a, 32
	%b0 = and %b, 4294967295
	%b1 = shr %b, 32
	%p00 = mul %a0, %b0
	%p01 = mul %a0, %b1
	%p10 = mul %a1, %b0
	%p11 = mul %a1, %b1
	%t0 = shr %p00, 32
	%t1 = and %p01, 4294967295
	%t2 = and %p10, 4294967295
	%t3 = add %t0, %t1
	%t4 = add %t3, %t2
	%carry = shr %t4, 32
	%h0 = shr %p01, 32
	%h1 = shr %p10, 32
	%h2 = add %p11, %h0
	%h3 = add %h2, %h1
	%h4 = add %h3, %carry
	ret %h4
}
