export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @loop
@loop
	%i =l phi @start 1, @body %nexti
	%min =l phi @start %y0, @body %nextmin
	%max =l phi @start %y0, @body %nextmax
	%sum =l phi @start %y0, @body %nextsum
	%count =l phi @start %odd0, @body %nextcount
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l shl %i, 3
	%ptr =l add %a, %offset
	%x =l loadl %ptr
	%y =l call %f(l %x)
	%ismin =w csltl %y, %min
	%nextmin =l select %ismin, %y, %min
	%ismax =w csltl %max, %y
	%nextmax =l select %ismax, %y, %max
	%nextsum =l add %sum, %y
	%odd =l and %y, 1
	%nextcount =l add %count, %odd
	%nexti =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%maxptr =l add %out, 8
	storel %max, %maxptr
	%sumptr =l add %out, 16
	storel %sum, %sumptr
	ret %count
}
