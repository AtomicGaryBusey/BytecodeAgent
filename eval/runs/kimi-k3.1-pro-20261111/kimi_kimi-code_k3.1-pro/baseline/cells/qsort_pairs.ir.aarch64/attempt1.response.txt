export function l $qsort_pairs(l %a, l %n) {
@entry
	jmp @outer
@outer
	%pa =l phi @entry %a, @set_right %a_next
	%pn =l phi @entry %n, @set_right %n_next
	%base =w csltl %pn, 2
	jnz %base, @done, @init
@init
	%last =l sub %pn, 1
	%lastoff =l mul %last, 16
	%plast =l add %pa, %lastoff
	%plastv =l add %plast, 8
	%pk =l loadl %plast
	%pv =l loadl %plastv
	jmp @part
@part
	%i =l phi @init 0, @cont %i_after
	%j =l phi @init 0, @cont %j_next
	%joff =l mul %j, 16
	%pj =l add %pa, %joff
	%pjv =l add %pj, 8
	%kj =l loadl %pj
	%vj =l loadl %pjv
	%key_less =w csltl %kj, %pk
	jnz %key_less, @take, @equal_key
@equal_key
	%key_equal =w ceql %kj, %pk
	jnz %key_equal, @value_check, @skip
@value_check
	%value_less =w csltl %vj, %pv
	jnz %value_less, @take, @skip
@take
	%ioff =l mul %i, 16
	%pi =l add %pa, %ioff
	%piv =l add %pi, 8
	%ki =l loadl %pi
	%vi =l loadl %piv
	storel %kj, %pi
	storel %vj, %piv
	storel %ki, %pj
	storel %vi, %pjv
	%i_inc =l add %i, 1
	jmp @cont
@skip
	jmp @cont
@cont
	%i_after =l phi @take %i_inc, @skip %i
	%j_next =l add %j, 1
	%more =w csltl %j_next, %last
	jnz %more, @part, @finish
@finish
	%fioff =l mul %i_after, 16
	%fpi =l add %pa, %fioff
	%fpiv =l add %fpi, 8
	%fki =l loadl %fpi
	%fvi =l loadl %fpiv
	storel %pk, %fpi
	storel %pv, %fpiv
	storel %fki, %plast
	storel %fvi, %plastv
	%left_big =w csgtl %i_after, 1
	jnz %left_big, @call_left, @set_right
@call_left
	%r =l call $qsort_pairs(l %pa, l %i_after)
	jmp @set_right
@set_right
	%a_next =l add %plast, 16
	%n_next =l sub %last, %i_after
	jmp @outer
@done
	ret 0
}
