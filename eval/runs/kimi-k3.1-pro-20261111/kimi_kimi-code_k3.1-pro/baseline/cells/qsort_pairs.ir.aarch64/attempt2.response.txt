export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @done, @setup
@setup
	%last =l sub %n, 1
	%lastoff =l mul %last, 16
	%pivot =l add %a, %lastoff
	%pivotv =l add %pivot, 8
	%pk =l loadl %pivot
	%pv =l loadl %pivotv
	jmp @loop
@loop
	%i =l phi @setup 0, @next %i_next
	%j =l phi @setup 0, @next %j_next
	%joff =l mul %j, 16
	%pj =l add %a, %joff
	%pjv =l add %pj, 8
	%kj =l loadl %pj
	%vj =l loadl %pjv
	%less =w csltl %kj, %pk
	jnz %less, @take, @equal
@equal
	%eq =w ceql %kj, %pk
	jnz %eq, @value, @skip
@value
	%vless =w csltl %vj, %pv
	jnz %vless, @take, @skip
@take
	%ioff =l mul %i, 16
	%pi =l add %a, %ioff
	%piv =l add %pi, 8
	%ki =l loadl %pi
	%vi =l loadl %piv
	storel %kj, %pi
	storel %vj, %piv
	storel %ki, %pj
	storel %vi, %pjv
	%i_inc =l add %i, 1
	jmp @next
@skip
	jmp @next
@next
	%i_next =l phi @take %i_inc, @skip %i
	%j_next =l add %j, 1
	%more =w csltl %j_next, %last
	jnz %more, @loop, @partitioned
@partitioned
	%pioff =l mul %i_next, 16
	%ppi =l add %a, %pioff
	%ppiv =l add %ppi, 8
	%pki =l loadl %ppi
	%pvi =l loadl %ppiv
	storel %pk, %ppi
	storel %pv, %ppiv
	storel %pki, %pivot
	storel %pvi, %pivotv
	%right_n =l sub %last, %i_next
	%right_index =l add %i_next, 1
	%right_off =l mul %right_index, 16
	%right_a =l add %a, %right_off
	%left_needed =w csgtl %i_next, 1
	jnz %left_needed, @left_call, @check_right
@left_call
	%r1 =l call $qsort_pairs(l %a, l %i_next)
	jmp @check_right
@check_right
	%right_needed =w csgtl %right_n, 1
	jnz %right_needed, @right_call, @done
@right_call
	%r2 =l call $qsort_pairs(l %right_a, l %right_n)
	jmp @done
@done
	ret 0
}
