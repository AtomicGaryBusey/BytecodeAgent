bubble_sort:
cmp x1, #2
b.lt .Ldone
sub x1, x1, #1
.Louter:
mov x2, #0
.Linner:
ldr x3, [x0, x2, lsl #3]
add x4, x2, #1
ldr x5, [x0, x4, lsl #3]
cmp x3, x5
b.le .Lnoswap
str x5, [x0, x2, lsl #3]
str x3, [x0, x4, lsl #3]
.Lnoswap:
add x2, x2, #1
cmp x2, x1
b.lt .Linner
subs x1, x1, #1
b.ne .Louter
.Ldone:
mov x0, #0
ret
