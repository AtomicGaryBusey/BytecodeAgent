export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @init_cond

@init_cond
	%i0 =l phi @start 0, @init_body %i0next
	%init_done =w ceql %i0, %n
	jnz %init_done, @multiply_start, @init_body

@init_body
	%off0 =l mul %i0, 8
	%ptr0 =l add %r, %off0
	storel 0, %ptr0
	%i0next =l add %i0, 1
	jmp @init_cond

@multiply_start
	jmp @outer_cond

@outer_cond
	%i =l phi @multiply_start 0, @inner_done %inext
	%outer_end =w ceql %i, %na
	jnz %outer_end, @scan_setup, @outer_body

@outer_body
	%aoff =l mul %i, 8
	%aptr =l add %a, %aoff
	%av =l loadl %aptr
	jmp @inner_cond

@inner_cond
	%j =l phi @outer_body 0, @inner_body %jnext
	%carry =l phi @outer_body 0, @inner_body %carry_next
	%inner_end =w ceql %j, %nb
	jnz %inner_end, @inner_done, @inner_body

@inner_body
	%boff =l mul %j, 8
	%bptr =l add %b, %boff
	%bv =l loadl %bptr
	%index =l add %i, %j
	%roff =l mul %index, 8
	%rptr =l add %r, %roff
	%old =l loadl %rptr

	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1

	%p01low =l shl %p01, 32
	%lo0 =l add %p00, %p01low
	%c01 =w cultl %lo0, %p00
	%c01l =l extuw %c01
	%p01high =l shr %p01, 32
	%hi0 =l add %p11, %p01high
	%hi1 =l add %hi0, %c01l

	%p10low =l shl %p10, 32
	%product_low =l add %lo0, %p10low
	%c10 =w cultl %product_low, %lo0
	%c10l =l extuw %c10
	%p10high =l shr %p10, 32
	%hi2 =l add %hi1, %p10high
	%product_high =l add %hi2, %c10l

	%sum0 =l add %product_low, %old
	%c0 =w cultl %sum0, %product_low
	%c0l =l extuw %c0
	%sum =l add %sum0, %carry
	%c1 =w cultl %sum, %sum0
	%c1l =l extuw %c1
	%newcarry0 =l add %product_high, %c0l
	%carry_next =l add %newcarry0, %c1l
	storel %sum, %rptr
	%jnext =l add %j, 1
	jmp @inner_cond

@inner_done
	%endindex =l add %i, %nb
	%endoff =l mul %endindex, 8
	%endptr =l add %r, %endoff
	storel %carry, %endptr
	%inext =l add %i, 1
	jmp @outer_cond

@scan_setup
	jmp @scan_cond

@scan_cond
	%k =l phi @scan_setup %n, @scan_body %kprev
	%zero_limbs =w ceql %k, 0
	jnz %zero_limbs, @return, @scan_check

@scan_check
	%km1 =l sub %k, 1
	%scanoff =l mul %km1, 8
	%scanptr =l add %r, %scanoff
	%scanvalue =l loadl %scanptr
	%iszero =w ceql %scanvalue, 0
	jnz %iszero, @scan_body, @return

@scan_body
	%kprev =l sub %k, 1
	jmp @scan_cond

@return
	ret %k
}
