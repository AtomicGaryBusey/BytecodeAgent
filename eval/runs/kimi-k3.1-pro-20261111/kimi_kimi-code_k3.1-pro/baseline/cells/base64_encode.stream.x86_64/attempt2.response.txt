base64_encode:
mov r8, rdx
test rsi, rsi
jz .Lfinish
.Lgroup:
cmp rsi, 3
jb .Ltail
movzx r10d, byte ptr [rdi]
shl r10d, 16
movzx eax, byte ptr [rdi + 1]
shl eax, 8
or r10d, eax
movzx eax, byte ptr [rdi + 2]
or r10d, eax
add rdi, 3
sub rsi, 3
mov ecx, 4
xor r11d, r11d
jmp .Lencode_chars
.Ltail:
movzx r10d, byte ptr [rdi]
shl r10d, 16
cmp rsi, 2
jne .Ltail_ready
movzx eax, byte ptr [rdi + 1]
shl eax, 8
or r10d, eax
.Ltail_ready:
lea ecx, [rsi + 1]
mov r11d, 1
.Lencode_chars:
mov eax, r10d
shr eax, 18
and eax, 63
shl r10d, 6
cmp eax, 25
jbe .Lupper
cmp eax, 51
jbe .Llower
cmp eax, 61
jbe .Ldigit
cmp eax, 62
je .Lplus
mov eax, 47
jmp .Lstore
.Lplus:
mov eax, 43
jmp .Lstore
.Ldigit:
sub eax, 4
jmp .Lstore
.Llower:
add eax, 71
jmp .Lstore
.Lupper:
add eax, 65
.Lstore:
mov byte ptr [r8], al
inc r8
dec ecx
jnz .Lencode_chars
test r11d, r11d
jnz .Lpad
test rsi, rsi
jnz .Lgroup
jmp .Lfinish
.Lpad:
cmp rsi, 1
jne .Lone_pad
mov byte ptr [r8], 61
inc r8
.Lone_pad:
mov byte ptr [r8], 61
inc r8
.Lfinish:
mov rax, r8
sub rax, rdx
ret
