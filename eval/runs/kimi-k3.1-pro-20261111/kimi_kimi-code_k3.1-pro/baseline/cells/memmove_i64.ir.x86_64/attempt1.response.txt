export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @return, @direction
@direction
	%backward =w cultl %src, %dst
	jnz %backward, @backward_setup, @forward_setup
@forward_setup
	jmp @forward_cond
@forward_cond
	%i =l phi @forward_setup 0, @forward_body %nexti
	%more =w csltl %i, %n
	jnz %more, @forward_body, @return
@forward_body
	%offset =l mul %i, 8
	%source =l add %src, %offset
	%value =l loadl %source
	%dest =l add %dst, %offset
	storel %value, %dest
	%nexti =l add %i, 1
	jmp @forward_cond
@backward_setup
	%initial =l sub %n, 1
	jmp @backward_body
@backward_body
	%j =l phi @backward_setup %initial, @backward_next %prevj
	%boffset =l mul %j, 8
	%bsource =l add %src, %boffset
	%bvalue =l loadl %bsource
	%bdest =l add %dst, %boffset
	storel %bvalue, %bdest
	%last =w ceql %j, 0
	jnz %last, @return, @backward_next
@backward_next
	%prevj =l sub %j, 1
	jmp @backward_body
@return
	ret 0
}
