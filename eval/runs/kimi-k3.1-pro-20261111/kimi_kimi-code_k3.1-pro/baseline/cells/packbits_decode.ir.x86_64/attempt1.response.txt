export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet

@packet
	%i =l phi @start 0, @nop %i_head, @lit_full_done %li, @rep_full_done %i_data
	%pos =l phi @start 0, @nop %pos, @lit_full_done %lp, @rep_full_done %rp
	%have =w csltl %i, %n
	jnz %have, @header, @success

@header
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i_head =l add %i, 1
	%is_nop =w ceqw %h, -128
	jnz %is_nop, @nop, @classify

@nop
	jmp @packet

@classify
	%negative =w csltw %h, 0
	jnz %negative, @repeat, @literal

@literal
	%hl =l extsw %h
	%count =l add %hl, 1
	%remaining =l sub %n, %i_head
	%src_ok =w cslel %count, %remaining
	jnz %src_ok, @lit_src_ok, @source_error

@lit_src_ok
	%space =l sub %cap, %pos
	%fits =w cslel %count, %space
	jnz %fits, @lit_full_setup, @lit_trunc_setup

@lit_full_setup
	%src_end =l add %i_head, %count
	jmp @lit_full_loop

@lit_full_loop
	%li =l phi @lit_full_setup %i_head, @lit_full_body %li_next
	%lp =l phi @lit_full_setup %pos, @lit_full_body %lp_next
	%lit_more =w csltl %li, %src_end
	jnz %lit_more, @lit_full_body, @lit_full_done

@lit_full_body
	%lsp =l add %src, %li
	%lop =l add %out, %lp
	%lv =w loadub %lsp
	storeb %lv, %lop
	%li_next =l add %li, 1
	%lp_next =l add %lp, 1
	jmp @lit_full_loop

@lit_full_done
	jmp @packet

@lit_trunc_setup
	jmp @lit_trunc_loop

@lit_trunc_loop
	%lti =l phi @lit_trunc_setup %i_head, @lit_trunc_body %lti_next
	%ltp =l phi @lit_trunc_setup %pos, @lit_trunc_body %ltp_next
	%lt_more =w csltl %ltp, %cap
	jnz %lt_more, @lit_trunc_body, @overflow

@lit_trunc_body
	%ltsp =l add %src, %lti
	%ltop =l add %out, %ltp
	%ltv =w loadub %ltsp
	storeb %ltv, %ltop
	%lti_next =l add %lti, 1
	%ltp_next =l add %ltp, 1
	jmp @lit_trunc_loop

@repeat
	%rhl =l extsw %h
	%rcount =l sub 1, %rhl
	%rep_has_src =w csltl %i_head, %n
	jnz %rep_has_src, @rep_src_ok, @source_error

@rep_src_ok
	%rvp =l add %src, %i_head
	%rv =w loadub %rvp
	%i_data =l add %i_head, 1
	%rspace =l sub %cap, %pos
	%rfits =w cslel %rcount, %rspace
	jnz %rfits, @rep_full_setup, @rep_trunc_setup

@rep_full_setup
	%rep_end =l add %pos, %rcount
	jmp @rep_full_loop

@rep_full_loop
	%rp =l phi @rep_full_setup %pos, @rep_full_body %rp_next
	%rep_more =w csltl %rp, %rep_end
	jnz %rep_more, @rep_full_body, @rep_full_done

@rep_full_body
	%rop =l add %out, %rp
	storeb %rv, %rop
	%rp_next =l add %rp, 1
	jmp @rep_full_loop

@rep_full_done
	jmp @packet

@rep_trunc_setup
	jmp @rep_trunc_loop

@rep_trunc_loop
	%rtp =l phi @rep_trunc_setup %pos, @rep_trunc_body %rtp_next
	%rt_more =w csltl %rtp, %cap
	jnz %rt_more, @rep_trunc_body, @overflow

@rep_trunc_body
	%rtop =l add %out, %rtp
	storeb %rv, %rtop
	%rtp_next =l add %rtp, 1
	jmp @rep_trunc_loop

@success
	ret %pos

@source_error
	ret -2

@overflow
	ret -1
}
