packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movsx r10, byte ptr [rdi + r8]
    inc r8
    test r10, r10
    js .Lnegative
    inc r10
    mov rax, rsi
    sub rax, r8
    cmp rax, r10
    jl .Lsource_error
    mov rax, rcx
    sub rax, r9
    cmp rax, r10
    jl .Lliteral_short
.Lliteral_copy:
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], r11b
    inc r8
    inc r9
    dec r10
    jnz .Lliteral_copy
    jmp .Lloop
.Lliteral_short:
    test rax, rax
    jz .Loverflow
.Lliteral_short_copy:
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], r11b
    inc r8
    inc r9
    dec rax
    jnz .Lliteral_short_copy
    jmp .Loverflow
.Lnegative:
    cmp r10, -128
    je .Lloop
    cmp r8, rsi
    jae .Lsource_error
    mov r11, 1
    sub r11, r10
    mov r10b, byte ptr [rdi + r8]
    inc r8
    mov rax, rcx
    sub rax, r9
    cmp rax, r11
    jl .Lrepeat_short
.Lrepeat_copy:
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec r11
    jnz .Lrepeat_copy
    jmp .Lloop
.Lrepeat_short:
    test rax, rax
    jz .Loverflow
.Lrepeat_short_copy:
    mov byte ptr [rdx + r9], r10b
    inc r9
    dec rax
    jnz .Lrepeat_short_copy
.Loverflow:
    mov rax, -1
    ret
.Lsource_error:
    mov rax, -2
    ret
.Ldone:
    mov rax, r9
    ret
