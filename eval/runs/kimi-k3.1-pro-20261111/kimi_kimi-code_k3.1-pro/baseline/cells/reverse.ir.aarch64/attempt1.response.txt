export function l $reverse(l %a, l %n) {
@entry
	%last =l sub %n, 1
@loop
	%i =l phi @entry 0, @swap %inext
	%j =l phi @entry %last, @swap %jnext
	%go =w csltl %i, %j
	jnz %go, @swap, @done
@swap
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%iptr =l add %a, %ioff
	%jptr =l add %a, %joff
	%ival =l loadl %iptr
	%jval =l loadl %jptr
	storel %jval, %iptr
	storel %ival, %jptr
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
