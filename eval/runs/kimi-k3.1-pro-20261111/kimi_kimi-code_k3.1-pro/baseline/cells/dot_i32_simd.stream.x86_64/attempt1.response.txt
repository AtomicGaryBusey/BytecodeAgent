dot_i32_simd:
pxor xmm4, xmm4
xor ecx, ecx
cmp rdx, 4
jl .Lreduce
.Lsimd_loop:
movdqu xmm0, [rdi + rcx*4]
movdqu xmm1, [rsi + rcx*4]
movdqa xmm2, xmm0
movdqa xmm3, xmm1
pmovsxdq xmm0, xmm0
pmovsxdq xmm1, xmm1
psrldq xmm2, 8
psrldq xmm3, 8
pmovsxdq xmm2, xmm2
pmovsxdq xmm3, xmm3
pmuldq xmm0, xmm1
pmuldq xmm2, xmm3
paddq xmm0, xmm2
paddq xmm4, xmm0
add rcx, 4
sub rdx, 4
cmp rdx, 4
jge .Lsimd_loop
.Lreduce:
movdqa xmm0, xmm4
psrldq xmm0, 8
paddq xmm4, xmm0
movq rax, xmm4
test rdx, rdx
jz .Ldone
.Lscalar_loop:
movsxd r8, dword ptr [rdi + rcx*4]
movsxd r9, dword ptr [rsi + rcx*4]
imul r8, r9
add rax, r8
inc rcx
dec rdx
jnz .Lscalar_loop
.Ldone:
ret
