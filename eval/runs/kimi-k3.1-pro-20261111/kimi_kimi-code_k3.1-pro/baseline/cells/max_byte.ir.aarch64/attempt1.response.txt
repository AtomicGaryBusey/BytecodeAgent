export function l $max_byte(l %p, l %n) {
@start
	%max =l copy 0
	%i =l copy 0
	%empty =w ceql %n, 0
	jnz %empty, @done, @loop
@loop
	%addr =l add %p, %i
	%b =l loadub %addr
	%greater =w csltl %max, %b
	jnz %greater, @update, @next
@update
	%max =l copy %b
	jmp @next
@next
	%i =l add %i, 1
	%more =w csltl %i, %n
	jnz %more, @loop, @done
@done
	ret %max
}
