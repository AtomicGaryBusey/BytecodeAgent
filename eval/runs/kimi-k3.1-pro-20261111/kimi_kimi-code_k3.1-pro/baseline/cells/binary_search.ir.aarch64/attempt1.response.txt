export function l $binary_search(l %a, l %n, l %key) {
@entry
	jmp @loop
@loop
	%lo =l phi @entry 0, @low %next, @high %lo
	%hi =l phi @entry %n, @low %hi, @high %mid
	%more =w csltl %lo, %hi
	jnz %more, @body, @notfound
@body
	%delta =l sub %hi, %lo
	%half =l shr %delta, 1
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%less =w csltl %value, %key
	jnz %less, @low, @checkhigh
@checkhigh
	%greater =w csltl %key, %value
	jnz %greater, @high, @found
@low
	%next =l add %mid, 1
	jmp @loop
@high
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
