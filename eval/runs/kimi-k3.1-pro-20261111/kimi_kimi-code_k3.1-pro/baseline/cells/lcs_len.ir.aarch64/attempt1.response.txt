export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
	jmp @init_test
@init_test
	%j =l phi @start %j0, @init_body %j_next
	%init_done =w csgtl %j, %n
	jnz %init_done, @init_done, @init_body
@init_body
	%off =l shl %j, 3
	%p =l add %row, %off
	storel 0, %p
	%j_next =l add %j, 1
	jmp @init_test
@init_done
	%i0 =l copy 0
	jmp @outer_test
@outer_test
	%i =l phi @init_done %i0, @outer_next %i_next
	%outer_done =w csgel %i, %m
	jnz %outer_done, @return, @outer_body
@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	%diag0 =l copy 0
	%k0 =l copy 1
	jmp @inner_test
@inner_test
	%k =l phi @outer_body %k0, @inner_next %k_next
	%diag =l phi @outer_body %diag0, @inner_next %diag_next
	%inner_done =w csgtl %k, %n
	jnz %inner_done, @outer_next, @inner_body
@inner_body
	%kprev =l sub %k, 1
	%koff =l shl %k, 3
	%rp =l add %row, %koff
	%up =l loadl %rp
	%yp =l add %y, %kprev
	%yb =w loadub %yp
	%same =w ceqw %xb, %yb
	jnz %same, @match, @mismatch
@match
	%match_val =l add %diag, 1
	storel %match_val, %rp
	jmp @inner_next
@mismatch
	%prev_off =l shl %kprev, 3
	%prev_p =l add %row, %prev_off
	%left =l loadl %prev_p
	%up_bigger =w csgtl %up, %left
	jnz %up_bigger, @store_up, @store_left
@store_up
	storel %up, %rp
	jmp @inner_next
@store_left
	storel %left, %rp
	jmp @inner_next
@inner_next
	%diag_next =l copy %up
	%k_next =l add %k, 1
	jmp @inner_test
@outer_next
	%i_next =l add %i, 1
	jmp @outer_test
@return
	%result_off =l shl %n, 3
	%result_p =l add %row, %result_off
	%result =l loadl %result_p
	ret %result
}
