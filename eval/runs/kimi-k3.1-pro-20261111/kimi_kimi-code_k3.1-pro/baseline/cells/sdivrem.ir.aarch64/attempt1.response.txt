export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bzero =w ceql %b, 0
	jnz %bzero, @divzero, @check
@check
	%amin =w ceql %a, 0x8000000000000000
	%bm1 =w ceql %b, -1
	%overflow =w and %amin, %bm1
	jnz %overflow, @overflow, @normal
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@overflow
	%zr =l copy 0
	storel %zr, %rem
	ret %a
@divzero
	storel %a, %rem
	%zq =l copy 0
	ret %zq
}
