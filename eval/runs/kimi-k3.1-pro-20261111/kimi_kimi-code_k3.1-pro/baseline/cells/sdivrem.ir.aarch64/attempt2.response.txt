export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%bzero =w ceql %b, 0
	jnz %bzero, @divzero, @check
@check
	%intmin =l shl 1, 63
	%amin =w ceql %a, %intmin
	%bm1 =w ceql %b, -1
	%overflow =w and %amin, %bm1
	jnz %overflow, @overflow, @normal
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@overflow
	storel 0, %rem
	ret %a
@divzero
	storel %a, %rem
	ret 0
}
