export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @after %i_next
	%crc =w phi @start 4294967295, @after %crc_next
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc0 =w xor %crc, %b
	jmp @inner
@inner
	%j =w phi @body 0, @bit_end %j_next
	%c =w phi @body %crc0, @bit_end %c_next
	%jcond =w csltw %j, 8
	jnz %jcond, @shift, @after
@shift
	%bit =w and %c, 1
	%shr =w shr %c, 1
	%iszero =w ceqw %bit, 0
	jnz %iszero, @no_xor, @yes_xor
@yes_xor
	%cx =w xor %shr, 3988292384
	jmp @bit_end
@no_xor
	jmp @bit_end
@bit_end
	%c_next =w phi @yes_xor %cx, @no_xor %shr
	%j_next =w add %j, 1
	jmp @inner
@after
	%crc_next =w copy %c
	%i_next =l add %i, 1
	jmp @outer
@done
	%outw =w xor %crc, 4294967295
	%out =l extuw %outw
	ret %out
}
