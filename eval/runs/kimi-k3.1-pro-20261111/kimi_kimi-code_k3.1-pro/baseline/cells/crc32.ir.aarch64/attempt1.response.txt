export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @next %i1
	%crc =w phi @start 4294967295, @next %crci
	%more =w csltl %i, %n
	jnz %more, @byte, @finish
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc0 =w xor %crc, %b
	jmp @inner
@inner
	%j =w phi @byte 0, @step %j1
	%crci =w phi @byte %crc0, @step %crcnext
	%jmore =w csltw %j, 8
	jnz %jmore, @step, @next
@step
	%lsb =w and %crci, 1
	%mask =w sub 0, %lsb
	%shift =w shr %crci, 1
	%poly =w and %mask, 3988292384
	%crcnext =w xor %shift, %poly
	%j1 =w add %j, 1
	jmp @inner
@next
	%i1 =l add %i, 1
	jmp @outer
@finish
	%out =w xor %crc, 4294967295
	%res =l extuw %out
	ret %res
}
