export function l $base64_encode(l %src, l %n, l %out) {
@start
    jmp @loop

@loop
    %i =l phi @start 0, @group_done %i_next
    %op =l phi @start %out, @group_done %op_next
    %left =l sub %n, %i
    %partial =w csltl %left, 3
    jnz %partial, @tail, @full

@full
    %p0 =l add %src, %i
    %p1 =l add %p0, 1
    %p2 =l add %p0, 2
    %b0 =l loadub %p0
    %b1 =l loadub %p1
    %b2 =l loadub %p2
    %b0s =l shl %b0, 16
    %b1s =l shl %b1, 8
    %val01 =l or %b0s, %b1s
    %val =l or %val01, %b2
    jmp @emit

@emit
    %j =l phi @full 0, @store %j_next
    %sixj =l mul %j, 6
    %shift =l sub 18, %sixj
    %shifted =l shr %val, %shift
    %v =l and %shifted, 63
    %is_upper =w csltl %v, 26
    jnz %is_upper, @upper, @check_lower

@upper
    %c_upper =l add %v, 65
    jmp @store

@check_lower
    %is_lower =w csltl %v, 52
    jnz %is_lower, @lower, @check_digit

@lower
    %c_lower =l add %v, 71
    jmp @store

@check_digit
    %is_digit =w csltl %v, 62
    jnz %is_digit, @digit, @check_special

@digit
    %c_digit =l sub %v, 4
    jmp @store

@check_special
    %is_plus =w ceql %v, 62
    jnz %is_plus, @plus, @slash

@plus
    %c_plus =l copy 43
    jmp @store

@slash
    %c_slash =l copy 47
    jmp @store

@store
    %c =l phi @upper %c_upper, @lower %c_lower, @digit %c_digit, @plus %c_plus, @slash %c_slash
    %addr =l add %op, %j
    storeb %c, %addr
    %j_next =l add %j, 1
    %group_emitted =w ceql %j_next, 4
    jnz %group_emitted, @group_done, @emit

@group_done
    %i_next =l add %i, 3
    %op_next =l add %op, 4
    jmp @loop

@tail
    %empty =w ceql %left, 0
    jnz %empty, @return_empty, @tail_nonempty

@return_empty
    %empty_result =l sub %op, %out
    ret %empty_result

@tail_nonempty
    %tp0 =l add %src, %i
    %tb0 =l loadub %tp0
    %tval0 =l shl %tb0, 16
    %one_byte =w ceql %left, 1
    jnz %one_byte, @tail_one, @tail_two

@tail_one
    jmp @tail_setup

@tail_two
    %tp1 =l add %tp0, 1
    %tb1 =l loadub %tp1
    %tb1s =l shl %tb1, 8
    %tval1 =l or %tval0, %tb1s
    jmp @tail_setup

@tail_setup
    %tval =l phi @tail_one %tval0, @tail_two %tval1
    %nchars =l phi @tail_one 2, @tail_two 3
    jmp @tail_emit

@tail_emit
    %tj =l phi @tail_setup 0, @tail_store %tj_next
    %tsixj =l mul %tj, 6
    %tshift =l sub 18, %tsixj
    %tshifted =l shr %tval, %tshift
    %tv =l and %tshifted, 63
    %tis_upper =w csltl %tv, 26
    jnz %tis_upper, @tail_upper, @tail_check_lower

@tail_upper
    %tc_upper =l add %tv, 65
    jmp @tail_store

@tail_check_lower
    %tis_lower =w csltl %tv, 52
    jnz %tis_lower, @tail_lower, @tail_check_digit

@tail_lower
    %tc_lower =l add %tv, 71
    jmp @tail_store

@tail_check_digit
    %tis_digit =w csltl %tv, 62
    jnz %tis_digit, @tail_digit, @tail_check_special

@tail_digit
    %tc_digit =l sub %tv, 4
    jmp @tail_store

@tail_check_special
    %tis_plus =w ceql %tv, 62
    jnz %tis_plus, @tail_plus, @tail_slash

@tail_plus
    %tc_plus =l copy 43
    jmp @tail_store

@tail_slash
    %tc_slash =l copy 47
    jmp @tail_store

@tail_store
    %tc =l phi @tail_upper %tc_upper, @tail_lower %tc_lower, @tail_digit %tc_digit, @tail_plus %tc_plus, @tail_slash %tc_slash
    %taddr =l add %op, %tj
    storeb %tc, %taddr
    %tj_next =l add %tj, 1
    %tail_emitted =w ceql %tj_next, %nchars
    jnz %tail_emitted, @pad, @tail_emit

@pad
    %pad_pos =l add %op, %nchars
    %equals =l copy 61
    storeb %equals, %pad_pos
    %two_pads =w ceql %nchars, 2
    jnz %two_pads, @pad_second, @finish

@pad_second
    %pad_pos_second =l add %pad_pos, 1
    storeb %equals, %pad_pos_second
    jmp @finish

@finish
    %base =l sub %op, %out
    %result =l add %base, 4
    ret %result
}
