export function l $isqrt(l %n0) {
@start
	%n =l alloc8 8
	%res =l alloc8 8
	%bit =l alloc8 8
	storel %n0, %n
	storel 0, %res
	storel 4611686018427387904, %bit
@find
	%nv =l loadl %n
	%bv =l loadl %bit
	%skip =w cultl %nv, %bv
	jnz %skip, @find_shift, @test
@find_shift
	%bs =l shr %bv, 2
	storel %bs, %bit
	jmp @find
@test
	%bv0 =l loadl %bit
	%done =w ceql %bv0, 0
	jnz %done, @return, @body
@body
	%rv =l loadl %res
	%bv1 =l loadl %bit
	%sum =l add %rv, %bv1
	%nv1 =l loadl %n
	%less =w cultl %nv1, %sum
	jnz %less, @else, @then
@then
	%nv2 =l sub %nv1, %sum
	storel %nv2, %n
	%half =l shr %rv, 1
	%rv2 =l add %half, %bv1
	storel %rv2, %res
	jmp @next
@else
	%rv3 =l shr %rv, 1
	storel %rv3, %res
	jmp @next
@next
	%bv2 =l loadl %bit
	%bv3 =l shr %bv2, 2
	storel %bv3, %bit
	jmp @test
@return
	%out =l loadl %res
	ret %out
}
