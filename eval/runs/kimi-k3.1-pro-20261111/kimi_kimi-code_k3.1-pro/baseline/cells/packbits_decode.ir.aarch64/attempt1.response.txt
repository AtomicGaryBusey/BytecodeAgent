export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop

@loop
	%i =l phi @start 0, @nop %i1, @lit_done %lit_inext, @rep_done %rep_inext
	%o =l phi @start 0, @nop %o, @lit_done %lit_onext, @rep_done %rep_onext
	%have_header =w csltl %i, %n
	jnz %have_header, @header, @success

@header
	%hp =l add %src, %i
	%h =w loadsb %hp
	%i1 =l add %i, 1
	%negative =w csltw %h, 0
	jnz %negative, @negative, @literal

@negative
	%is_nop =w ceqw %h, -128
	jnz %is_nop, @nop, @repeat

@nop
	jmp @loop

@literal
	%lit_countw =w add %h, 1
	%lit_count =l extuw %lit_countw
	%lit_srcleft =l sub %n, %i1
	%lit_srcok =w cslel %lit_count, %lit_srcleft
	jnz %lit_srcok, @lit_source_ok, @source_error

@lit_source_ok
	%lit_outleft =l sub %cap, %o
	%lit_fits =w cslel %lit_count, %lit_outleft
	jnz %lit_fits, @lit_full, @lit_truncated

@lit_full
	jmp @lit_setup

@lit_truncated
	jmp @lit_setup

@lit_setup
	%lit_write =l phi @lit_full %lit_count, @lit_truncated %lit_outleft
	%lit_overflow =w phi @lit_full 0, @lit_truncated 1
	jmp @lit_copy

@lit_copy
	%lit_j =l phi @lit_setup 0, @lit_body %lit_jnext
	%lit_more =w csltl %lit_j, %lit_write
	jnz %lit_more, @lit_body, @lit_done

@lit_body
	%lit_sbase =l add %src, %i1
	%lit_sp =l add %lit_sbase, %lit_j
	%lit_dbase =l add %out, %o
	%lit_dp =l add %lit_dbase, %lit_j
	%lit_byte =w loadub %lit_sp
	storeb %lit_byte, %lit_dp
	%lit_jnext =l add %lit_j, 1
	jmp @lit_copy

@lit_done
	%lit_inext =l add %i1, %lit_count
	%lit_onext =l add %o, %lit_write
	jnz %lit_overflow, @output_error, @loop

@repeat
	%rep_srcok =w csltl %i1, %n
	jnz %rep_srcok, @rep_source_ok, @source_error

@rep_source_ok
	%rep_countw =w sub 1, %h
	%rep_count =l extuw %rep_countw
	%rep_vp =l add %src, %i1
	%rep_byte =w loadub %rep_vp
	%rep_outleft =l sub %cap, %o
	%rep_fits =w cslel %rep_count, %rep_outleft
	jnz %rep_fits, @rep_full, @rep_truncated

@rep_full
	jmp @rep_setup

@rep_truncated
	jmp @rep_setup

@rep_setup
	%rep_write =l phi @rep_full %rep_count, @rep_truncated %rep_outleft
	%rep_overflow =w phi @rep_full 0, @rep_truncated 1
	jmp @rep_fill

@rep_fill
	%rep_j =l phi @rep_setup 0, @rep_body %rep_jnext
	%rep_more =w csltl %rep_j, %rep_write
	jnz %rep_more, @rep_body, @rep_done

@rep_body
	%rep_dbase =l add %out, %o
	%rep_dp =l add %rep_dbase, %rep_j
	storeb %rep_byte, %rep_dp
	%rep_jnext =l add %rep_j, 1
	jmp @rep_fill

@rep_done
	%rep_inext =l add %i1, 1
	%rep_onext =l add %o, %rep_write
	jnz %rep_overflow, @output_error, @loop

@success
	ret %o

@source_error
	ret -2

@output_error
	ret -1
}
