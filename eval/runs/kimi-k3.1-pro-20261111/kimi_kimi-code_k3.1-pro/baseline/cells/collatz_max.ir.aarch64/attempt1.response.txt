export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @advance %knext
	%max =l phi @start 0, @advance %maxnext
	jmp @inner
@inner
	%x =l phi @outer %k, @next %xnext
	%steps =l phi @outer 0, @next %stepsnext
	%done =w ceql %x, 1
	jnz %done, @after, @testeven
@testeven
	%lowbit =l and %x, 1
	%iseven =w ceql %lowbit, 0
	jnz %iseven, @even, @odd
@even
	%nexteven =l div %x, 2
	jmp @next
@odd
	%triple =l mul %x, 3
	%nextodd =l add %triple, 1
	jmp @next
@next
	%xnext =l phi @even %nexteven, @odd %nextodd
	%stepsnext =l add %steps, 1
	jmp @inner
@after
	%islarger =w csltl %max, %steps
	jnz %islarger, @update, @advance
@update
	%newmax =l copy %steps
	jmp @advance
@advance
	%maxnext =l phi @after %max, @update %newmax
	%knext =l add %k, 1
	%more =w cslel %knext, %n
	jnz %more, @outer, @return
@return
	ret %maxnext
}
