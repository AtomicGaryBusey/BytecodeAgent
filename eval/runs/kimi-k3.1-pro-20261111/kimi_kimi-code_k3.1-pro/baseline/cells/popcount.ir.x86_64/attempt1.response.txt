export function l $popcount(l %x) {
@start
	%t0 =l shr %x, 1
	%t1 =l and %t0, 0x5555555555555555
	%t2 =l sub %x, %t1
	%t3 =l and %t2, 0x3333333333333333
	%t4 =l shr %t2, 2
	%t5 =l and %t4, 0x3333333333333333
	%t6 =l add %t3, %t5
	%t7 =l shr %t6, 4
	%t8 =l add %t6, %t7
	%t9 =l and %t8, 0x0f0f0f0f0f0f0f0f
	%t10 =l mul %t9, 0x0101010101010101
	%count =l shr %t10, 56
	ret %count
}
