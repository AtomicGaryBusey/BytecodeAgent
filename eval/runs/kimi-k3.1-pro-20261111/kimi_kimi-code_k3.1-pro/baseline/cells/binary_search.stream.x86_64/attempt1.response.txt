binary_search:
    xor eax, eax
    mov rcx, rsi
.Lloop:
    cmp rax, rcx
    jae .Lnot_found
    mov r8, rcx
    sub r8, rax
    shr r8, 1
    add r8, rax
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .Lfound
    jl .Lgo_right
    mov rcx, r8
    jmp .Lloop
.Lgo_right:
    lea rax, [r8 + 1]
    jmp .Lloop
.Lfound:
    mov rax, r8
    ret
.Lnot_found:
    mov rax, -1
    ret
