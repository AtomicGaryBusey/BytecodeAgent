export function l $count_desc(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @zero, @init
@init
	%i0 =l copy 1
	%count0 =l copy 0
	jmp @loop
@loop
	%i =l phi @init %i0, @body %inext
	%count =l phi @init %count0, @body %countnext
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%nextptr =l add %a, %offset
	%next =l loadl %nextptr
	%prevoffset =l sub %offset, 8
	%prevptr =l add %a, %prevoffset
	%prev =l loadl %prevptr
	%desc =w csgtl %prev, %next
	%increment =l extuw %desc
	%countnext =l add %count, %increment
	%inext =l add %i, 1
	jmp @loop
@done
	ret %count
@zero
	ret 0
}
