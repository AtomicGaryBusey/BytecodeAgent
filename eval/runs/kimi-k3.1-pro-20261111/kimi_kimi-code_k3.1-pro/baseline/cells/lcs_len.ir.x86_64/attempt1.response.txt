export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @zero
@zero
	%jz =l phi @start 0, @zero %jz1
	%zoff =l mul %jz, 8
	%zp =l add %row, %zoff
	storel 0, %zp
	%jz1 =l add %jz, 1
	%zmore =w cslel %jz1, %n
	jnz %zmore, @zero, @init_outer
@init_outer
	jmp @outer
@outer
	%i =l phi @init_outer 0, @outer_next %i1
	%im =w csltl %i, %m
	jnz %im, @outer_body, @done
@outer_body
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%j =l phi @outer_body 1, @inner_store %j1
	%diag =l phi @outer_body 0, @inner_store %up
	%jn =w cslel %j, %n
	jnz %jn, @inner_body, @outer_next
@inner_body
	%joff =l mul %j, 8
	%jp =l add %row, %joff
	%up =l loadl %jp
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @mismatch
@mismatch
	%lmoff =l mul %jm1, 8
	%lmp =l add %row, %lmoff
	%left =l loadl %lmp
	%upgt =w csgtl %up, %left
	jnz %upgt, @take_up, @take_left
@match
	%matchval =l add %diag, 1
	jmp @inner_store
@take_up
	jmp @inner_store
@take_left
	jmp @inner_store
@inner_store
	%value =l phi @match %matchval, @take_up %up, @take_left %left
	storel %value, %jp
	%j1 =l add %j, 1
	jmp @inner
@outer_next
	%i1 =l add %i, 1
	jmp @outer
@done
	%noff =l mul %n, 8
	%np =l add %row, %noff
	%result =l loadl %np
	ret %result
}
