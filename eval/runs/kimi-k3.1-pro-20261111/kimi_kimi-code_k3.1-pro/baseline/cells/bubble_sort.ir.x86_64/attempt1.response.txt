export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @setup
@setup
	%initial =l sub %n, 1
	jmp @outer
@outer
	%remaining =l phi @setup %initial, @pass_done %next_remaining
	%zero =l copy 0
	jmp @cond
@cond
	%j =l phi @outer %zero, @advance %jnext
	%go =w csltl %j, %remaining
	jnz %go, @body, @pass_done
@body
	%offset =l mul %j, 8
	%p =l add %a, %offset
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csltl %y, %x
	jnz %swap, @do_swap, @advance
@do_swap
	storel %y, %p
	storel %x, %q
	jmp @advance
@advance
	%jnext =l add %j, 1
	jmp @cond
@pass_done
	%next_remaining =l sub %remaining, 1
	%more =w csltl 0, %next_remaining
	jnz %more, @outer, @done
@done
	ret 0
}
