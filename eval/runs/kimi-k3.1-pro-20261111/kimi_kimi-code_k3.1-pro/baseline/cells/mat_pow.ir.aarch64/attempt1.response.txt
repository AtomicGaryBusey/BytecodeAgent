export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%twonn =l add %nn, %nn
	jmp @init_i_check

@init_i_check
	%i =l phi @start 0, @init_i_next %i_next
	%i_more =w csltl %i, %n
	jnz %i_more, @init_j_check, @exp_check

@init_j_check
	%j =l phi @init_i_check 0, @init_j_next %j_next
	%j_more =w csltl %j, %n
	jnz %j_more, @init_store, @init_i_next

@init_store
	%row =l mul %i, %n
	%idx =l add %row, %j
	%off =l mul %idx, 8
	%addr =l add %out, %off
	%diag =w ceql %i, %j
	jnz %diag, @init_one, @init_zero

@init_one
	storel 1, %addr
	jmp @init_j_next

@init_zero
	storel 0, %addr
	jmp @init_j_next

@init_j_next
	%j_next =l add %j, 1
	jmp @init_j_check

@init_i_next
	%i_next =l add %i, 1
	jmp @init_i_check

@exp_check
	%e =l phi @init_i_check 0, @copy_done %e_next
	%e_more =w csltl %e, %k
	jnz %e_more, @mul_i_check, @zero_check

@mul_i_check
	%mi =l phi @exp_check 0, @mul_i_next %mi_next
	%mi_more =w csltl %mi, %n
	jnz %mi_more, @mul_j_check, @copy_check

@mul_j_check
	%mj =l phi @mul_i_check 0, @mul_j_next %mj_next
	%mj_more =w csltl %mj, %n
	jnz %mj_more, @mul_t_check, @mul_i_next

@mul_t_check
	%mt =l phi @mul_j_check 0, @mul_t_next %mt_next
	%sum =l phi @mul_j_check 0, @mul_t_next %sum_next
	%mt_more =w csltl %mt, %n
	jnz %mt_more, @mul_body, @mul_store

@mul_body
	%mrow =l mul %mi, %n
	%xidx =l add %mrow, %mt
	%xoff =l mul %xidx, 8
	%xaddr =l add %out, %xoff
	%x =l loadl %xaddr
	%trow =l mul %mt, %n
	%yidx =l add %trow, %mj
	%yoff =l mul %yidx, 8
	%yaddr =l add %a, %yoff
	%y =l loadl %yaddr
	%prod =l mul %x, %y
	jmp @mul_t_next

@mul_t_next
	%sum_next =l add %sum, %prod
	%mt_next =l add %mt, 1
	jmp @mul_t_check

@mul_store
	%srow =l mul %mi, %n
	%sidx =l add %srow, %mj
	%soff =l mul %sidx, 8
	%saddr =l add %tmp, %soff
	storel %sum, %saddr
	jmp @mul_j_next

@mul_j_next
	%mj_next =l add %mj, 1
	jmp @mul_j_check

@mul_i_next
	%mi_next =l add %mi, 1
	jmp @mul_i_check

@copy_check
	%c =l phi @mul_i_check 0, @copy_next %c_next
	%c_more =w csltl %c, %nn
	jnz %c_more, @copy_body, @copy_done

@copy_body
	%coff =l mul %c, 8
	%ctmp =l add %tmp, %coff
	%cout =l add %out, %coff
	%cval =l loadl %ctmp
	storel %cval, %cout
	jmp @copy_next

@copy_next
	%c_next =l add %c, 1
	jmp @copy_check

@copy_done
	%e_next =l add %e, 1
	jmp @exp_check

@zero_check
	%z =l phi @exp_check 0, @zero_next %z_next
	%z_more =w csltl %z, %twonn
	jnz %z_more, @zero_body, @done

@zero_body
	%zoff =l mul %z, 8
	%zaddr =l add %tmp, %zoff
	storel 0, %zaddr
	jmp @zero_next

@zero_next
	%z_next =l add %z, 1
	jmp @zero_check

@done
	ret 0
}
