bubble_sort:
mov rax, rsi
dec rax
.Louter:
test rax, rax
jle .Ldone
xor ecx, ecx
.Linner:
cmp rcx, rax
jge .Lnext
mov rdx, qword ptr [rdi + rcx*8]
mov r8, qword ptr [rdi + rcx*8 + 8]
cmp rdx, r8
jle .Lskip
mov qword ptr [rdi + rcx*8], r8
mov qword ptr [rdi + rcx*8 + 8], rdx
.Lskip:
inc rcx
jmp .Linner
.Lnext:
dec rax
jmp .Louter
.Ldone:
xor eax, eax
ret
