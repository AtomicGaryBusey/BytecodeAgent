export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loophead
@loophead
	%i =l phi @start 0, @next %inext
	%o =l phi @start 0, @next %onext
	%active =w csltl %i, %n
	jnz %active, @group, @done
@group
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%b0s =w shl %b0, 16
	%rem =l sub %n, %i
	%lt3 =w csltl %rem, 3
	jnz %lt3, @partial, @full
@full
	%p1 =l add %p0, 1
	%b1 =w loadub %p1
	%p2 =l add %p0, 2
	%b2 =w loadub %p2
	%b1s =w shl %b1, 8
	%vtmp =w or %b0s, %b1s
	%vfull =w or %vtmp, %b2
	jmp @groupmerge
@partial
	%isone =w ceql %rem, 1
	jnz %isone, @one, @two
@two
	%p1t =l add %p0, 1
	%b1t =w loadub %p1t
	%b1ts =w shl %b1t, 8
	%vtwo =w or %b0s, %b1ts
	jmp @groupmerge
@one
	jmp @groupmerge
@groupmerge
	%v =w phi @full %vfull, @two %vtwo, @one %b0s
	%count =w phi @full 4, @two 3, @one 2
	%ogroup =l add %out, %o
	jmp @charloop
@charloop
	%j =w phi @groupmerge 0, @charnext %jnext
	%op =l phi @groupmerge %ogroup, @charnext %opnext
	%ischar =w csltw %j, %count
	jnz %ischar, @mapchar, @padchar
@mapchar
	%six =w mul %j, 6
	%shift =w sub 18, %six
	%idxraw =w shr %v, %shift
	%idx =w and %idxraw, 63
	%lt26 =w csltw %idx, 26
	jnz %lt26, @upper, @notupper
@upper
	%chupper =w add %idx, 65
	jmp @storechar
@notupper
	%lt52 =w csltw %idx, 52
	jnz %lt52, @lower, @notlower
@lower
	%chlower =w add %idx, 71
	jmp @storechar
@notlower
	%lt62 =w csltw %idx, 62
	jnz %lt62, @digit, @special
@digit
	%chdigit =w sub %idx, 4
	jmp @storechar
@special
	%isplus =w ceqw %idx, 62
	jnz %isplus, @plus, @slash
@plus
	%chplus =w copy 43
	jmp @storechar
@slash
	%chslash =w copy 47
	jmp @storechar
@padchar
	jmp @storechar
@storechar
	%ch =w phi @upper %chupper, @lower %chlower, @digit %chdigit, @plus %chplus, @slash %chslash, @padchar 61
	storeb %ch, %op
	jmp @charnext
@charnext
	%opnext =l add %op, 1
	%jnext =w add %j, 1
	%jdone =w ceqw %jnext, 4
	jnz %jdone, @next, @charloop
@next
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @loophead
@done
	ret %o
}
