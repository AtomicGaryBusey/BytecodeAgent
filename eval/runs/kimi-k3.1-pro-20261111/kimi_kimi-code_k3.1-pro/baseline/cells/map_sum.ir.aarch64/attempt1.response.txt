export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%sum =l phi @start 0, @body %sum1
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%offset =l mul %i, 8
	%addr =l add %a, %offset
	%x =l loadl %addr
	%r =l call %f(l %x)
	%sum1 =l add %sum, %r
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %sum
}
