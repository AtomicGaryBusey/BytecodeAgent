export function l $bmax(l %a, l %n) {
@start
	%max0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %i1
	%max =l phi @start %max0, @next %max1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%greater =w csltl %max, %v
	jnz %greater, @update, @next
@update
	jmp @next
@next
	%max1 =l phi @body %max, @update %v
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %max
}
