export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop

@loop
	%pc =l phi @start 0, @push %push_pc, @add %add_pc, @sub %sub_pc, @mul %mul_pc, @dup %dup_pc, @swap %swap_pc, @drop %drop_pc, @over %over_pc, @lt %lt_pc, @jz_take %jz_target, @jz_fall %jz_pc, @jump %jump_target, @unknown %unknown_pc
	%depth =l phi @start 0, @push %push_depth, @add %add_depth, @sub %sub_depth, @mul %mul_depth, @dup %dup_depth, @swap %depth, @drop %drop_depth, @over %over_depth, @lt %lt_depth, @jz_take %jz_depth, @jz_fall %jz_depth, @jump %depth, @unknown %depth
	%finished =w ceql %pc, %len
	jnz %finished, @finish, @fetch

@fetch
	%pc_offset =l mul %pc, 8
	%instruction_addr =l add %code, %pc_offset
	%op =l loadl %instruction_addr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @finish, @check_push

@check_push
	%is_push =w ceql %op, 1
	jnz %is_push, @push, @check_add

@check_add
	%is_add =w ceql %op, 2
	jnz %is_add, @add, @check_sub

@check_sub
	%is_sub =w ceql %op, 3
	jnz %is_sub, @sub, @check_mul

@check_mul
	%is_mul =w ceql %op, 4
	jnz %is_mul, @mul, @check_dup

@check_dup
	%is_dup =w ceql %op, 5
	jnz %is_dup, @dup, @check_swap

@check_swap
	%is_swap =w ceql %op, 6
	jnz %is_swap, @swap, @check_drop

@check_drop
	%is_drop =w ceql %op, 7
	jnz %is_drop, @drop, @check_over

@check_over
	%is_over =w ceql %op, 8
	jnz %is_over, @over, @check_lt

@check_lt
	%is_lt =w ceql %op, 9
	jnz %is_lt, @lt, @check_jz

@check_jz
	%is_jz =w ceql %op, 10
	jnz %is_jz, @jz, @check_jmp

@check_jmp
	%is_jmp =w ceql %op, 11
	jnz %is_jmp, @jump, @unknown

@push
	%push_operand_pc =l add %pc, 1
	%push_operand_offset =l mul %push_operand_pc, 8
	%push_operand_addr =l add %code, %push_operand_offset
	%push_value =l loadl %push_operand_addr
	%push_stack_offset =l mul %depth, 8
	%push_stack_addr =l add %stack, %push_stack_offset
	storel %push_value, %push_stack_addr
	%push_depth =l add %depth, 1
	%push_pc =l add %pc, 2
	jmp @loop

@add
	%add_depth =l sub %depth, 1
	%add_a_index =l sub %depth, 2
	%add_b_index =l sub %depth, 1
	%add_a_offset =l mul %add_a_index, 8
	%add_b_offset =l mul %add_b_index, 8
	%add_a_addr =l add %stack, %add_a_offset
	%add_b_addr =l add %stack, %add_b_offset
	%add_a =l loadl %add_a_addr
	%add_b =l loadl %add_b_addr
	%add_value =l add %add_a, %add_b
	storel %add_value, %add_a_addr
	%add_pc =l add %pc, 1
	jmp @loop

@sub
	%sub_depth =l sub %depth, 1
	%sub_a_index =l sub %depth, 2
	%sub_b_index =l sub %depth, 1
	%sub_a_offset =l mul %sub_a_index, 8
	%sub_b_offset =l mul %sub_b_index, 8
	%sub_a_addr =l add %stack, %sub_a_offset
	%sub_b_addr =l add %stack, %sub_b_offset
	%sub_a =l loadl %sub_a_addr
	%sub_b =l loadl %sub_b_addr
	%sub_value =l sub %sub_a, %sub_b
	storel %sub_value, %sub_a_addr
	%sub_pc =l add %pc, 1
	jmp @loop

@mul
	%mul_depth =l sub %depth, 1
	%mul_a_index =l sub %depth, 2
	%mul_b_index =l sub %depth, 1
	%mul_a_offset =l mul %mul_a_index, 8
	%mul_b_offset =l mul %mul_b_index, 8
	%mul_a_addr =l add %stack, %mul_a_offset
	%mul_b_addr =l add %stack, %mul_b_offset
	%mul_a =l loadl %mul_a_addr
	%mul_b =l loadl %mul_b_addr
	%mul_value =l mul %mul_a, %mul_b
	storel %mul_value, %mul_a_addr
	%mul_pc =l add %pc, 1
	jmp @loop

@dup
	%dup_source_index =l sub %depth, 1
	%dup_source_offset =l mul %dup_source_index, 8
	%dup_dest_offset =l mul %depth, 8
	%dup_source_addr =l add %stack, %dup_source_offset
	%dup_dest_addr =l add %stack, %dup_dest_offset
	%dup_value =l loadl %dup_source_addr
	storel %dup_value, %dup_dest_addr
	%dup_depth =l add %depth, 1
	%dup_pc =l add %pc, 1
	jmp @loop

@swap
	%swap_a_index =l sub %depth, 2
	%swap_b_index =l sub %depth, 1
	%swap_a_offset =l mul %swap_a_index, 8
	%swap_b_offset =l mul %swap_b_index, 8
	%swap_a_addr =l add %stack, %swap_a_offset
	%swap_b_addr =l add %stack, %swap_b_offset
	%swap_a =l loadl %swap_a_addr
	%swap_b =l loadl %swap_b_addr
	storel %swap_b, %swap_a_addr
	storel %swap_a, %swap_b_addr
	%swap_pc =l add %pc, 1
	jmp @loop

@drop
	%drop_depth =l sub %depth, 1
	%drop_pc =l add %pc, 1
	jmp @loop

@over
	%over_source_index =l sub %depth, 2
	%over_source_offset =l mul %over_source_index, 8
	%over_dest_offset =l mul %depth, 8
	%over_source_addr =l add %stack, %over_source_offset
	%over_dest_addr =l add %stack, %over_dest_offset
	%over_value =l loadl %over_source_addr
	storel %over_value, %over_dest_addr
	%over_depth =l add %depth, 1
	%over_pc =l add %pc, 1
	jmp @loop

@lt
	%lt_depth =l sub %depth, 1
	%lt_a_index =l sub %depth, 2
	%lt_b_index =l sub %depth, 1
	%lt_a_offset =l mul %lt_a_index, 8
	%lt_b_offset =l mul %lt_b_index, 8
	%lt_a_addr =l add %stack, %lt_a_offset
	%lt_b_addr =l add %stack, %lt_b_offset
	%lt_a =l loadl %lt_a_addr
	%lt_b =l loadl %lt_b_addr
	%lt_comparison =w csltl %lt_a, %lt_b
	%lt_value =l extuw %lt_comparison
	storel %lt_value, %lt_a_addr
	%lt_pc =l add %pc, 1
	jmp @loop

@jz
	%jz_depth =l sub %depth, 1
	%jz_value_offset =l mul %jz_depth, 8
	%jz_value_addr =l add %stack, %jz_value_offset
	%jz_value =l loadl %jz_value_addr
	%jz_is_zero =w ceql %jz_value, 0
	jnz %jz_is_zero, @jz_take, @jz_fall

@jz_take
	%jz_operand_pc =l add %pc, 1
	%jz_operand_offset =l mul %jz_operand_pc, 8
	%jz_operand_addr =l add %code, %jz_operand_offset
	%jz_target =l loadl %jz_operand_addr
	jmp @loop

@jz_fall
	%jz_pc =l add %pc, 2
	jmp @loop

@jump
	%jump_operand_pc =l add %pc, 1
	%jump_operand_offset =l mul %jump_operand_pc, 8
	%jump_operand_addr =l add %code, %jump_operand_offset
	%jump_target =l loadl %jump_operand_addr
	jmp @loop

@unknown
	%unknown_pc =l add %pc, 1
	jmp @loop

@finish
	%stack_empty =w ceql %depth, 0
	jnz %stack_empty, @empty, @nonempty

@nonempty
	%result_index =l sub %depth, 1
	%result_offset =l mul %result_index, 8
	%result_addr =l add %stack, %result_offset
	%top =l loadl %result_addr
	jmp @clear

@empty
	jmp @clear

@clear
	%result =l phi @nonempty %top, @empty 0
	jmp @clear_loop

@clear_loop
	%clear_index =l phi @clear %depth, @clear_body %clear_next
	%clear_done =w ceql %clear_index, %cap
	jnz %clear_done, @return, @clear_body

@clear_body
	%clear_offset =l mul %clear_index, 8
	%clear_addr =l add %stack, %clear_offset
	storel 0, %clear_addr
	%clear_next =l add %clear_index, 1
	jmp @clear_loop

@return
	ret %result
}
