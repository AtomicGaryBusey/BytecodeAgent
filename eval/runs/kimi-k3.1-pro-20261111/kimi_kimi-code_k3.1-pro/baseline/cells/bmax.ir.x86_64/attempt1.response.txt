export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%i =l copy 1
@loop
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w csltl %max, %value
	jnz %greater, @update, @next
@update
	%max =l copy %value
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %max
}
