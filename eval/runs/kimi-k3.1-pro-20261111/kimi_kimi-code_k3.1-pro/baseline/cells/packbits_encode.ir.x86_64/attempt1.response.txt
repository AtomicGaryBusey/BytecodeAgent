export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @repeat_write %i_after_repeat, @literal_done %j
	%o =l phi @start 0, @repeat_write %o_after_repeat, @literal_done %o_after_literal
	%more =w csltl %i, %n
	jnz %more, @run_init, @done
@run_init
	%basep =l add %src, %i
	%baseb =w loadub %basep
	jmp @run_test
@run_test
	%r =l phi @run_init 1, @run_inc %r_next
	%below128 =w csltl %r, 128
	jnz %below128, @run_check_end, @run_finish
@run_check_end
	%rpos =l add %i, %r
	%rinbounds =w csltl %rpos, %n
	jnz %rinbounds, @run_check_byte, @run_finish
@run_check_byte
	%rp =l add %src, %rpos
	%rb =w loadub %rp
	%same =w ceqw %rb, %baseb
	jnz %same, @run_inc, @run_finish
@run_inc
	%r_next =l add %r, 1
	jmp @run_test
@run_finish
	%shortrun =w csltl %r, 3
	jnz %shortrun, @literal_init, @repeat
@repeat
	%repeat_end =l add %o, 2
	%repeat_fits =w cslel %repeat_end, %cap
	jnz %repeat_fits, @repeat_write, @fail
@repeat_write
	%repeat_header =l sub 257, %r
	%repeat_out =l add %out, %o
	storeb %repeat_header, %repeat_out
	%repeat_value_out =l add %repeat_out, 1
	storeb %baseb, %repeat_value_out
	%o_after_repeat =l add %o, 2
	%i_after_repeat =l add %i, %r
	jmp @outer
@literal_init
	jmp @literal_test
@literal_test
	%j =l phi @literal_init %i, @literal_advance %j_next
	%j_inbounds =w csltl %j, %n
	jnz %j_inbounds, @literal_check_length, @literal_finish
@literal_check_length
	%current_length =l sub %j, %i
	%length_ok =w csltl %current_length, 128
	jnz %length_ok, @literal_run_init, @literal_finish
@literal_run_init
	%lit_basep =l add %src, %j
	%lit_baseb =w loadub %lit_basep
	jmp @literal_run_test
@literal_run_test
	%rj =l phi @literal_run_init 1, @literal_run_inc %rj_next
	%rj_below128 =w csltl %rj, 128
	jnz %rj_below128, @literal_run_check_end, @literal_run_finish
@literal_run_check_end
	%rjpos =l add %j, %rj
	%rj_inbounds =w csltl %rjpos, %n
	jnz %rj_inbounds, @literal_run_check_byte, @literal_run_finish
@literal_run_check_byte
	%rjp =l add %src, %rjpos
	%rjb =w loadub %rjp
	%lit_same =w ceqw %rjb, %lit_baseb
	jnz %lit_same, @literal_run_inc, @literal_run_finish
@literal_run_inc
	%rj_next =l add %rj, 1
	jmp @literal_run_test
@literal_run_finish
	%literal_run_short =w csltl %rj, 3
	jnz %literal_run_short, @literal_advance, @literal_finish
@literal_advance
	%j_next =l add %j, 1
	jmp @literal_test
@literal_finish
	%length =l sub %j, %i
	%packet_size =l add %length, 1
	%literal_end =l add %o, %packet_size
	%literal_fits =w cslel %literal_end, %cap
	jnz %literal_fits, @literal_write_header, @fail
@literal_write_header
	%literal_header =l sub %length, 1
	%literal_out =l add %out, %o
	storeb %literal_header, %literal_out
	jmp @literal_copy_test
@literal_copy_test
	%k =l phi @literal_write_header 0, @literal_copy_body %k_next
	%copy_more =w csltl %k, %length
	jnz %copy_more, @literal_copy_body, @literal_done
@literal_copy_body
	%src_index =l add %i, %k
	%copy_src =l add %src, %src_index
	%copy_byte =w loadub %copy_src
	%out_data_start =l add %o, 1
	%out_index =l add %out_data_start, %k
	%copy_out =l add %out, %out_index
	storeb %copy_byte, %copy_out
	%k_next =l add %k, 1
	jmp @literal_copy_test
@literal_done
	%o_after_literal =l add %o, %packet_size
	jmp @outer
@fail
	ret -1
@done
	ret %o
}
