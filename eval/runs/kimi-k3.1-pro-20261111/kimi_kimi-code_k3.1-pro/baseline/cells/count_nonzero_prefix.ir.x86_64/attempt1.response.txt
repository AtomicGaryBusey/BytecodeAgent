export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i_next
	%p =l phi @start %a, @next %p_next
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%v =l loadl %p
	%zero =w ceql %v, 0
	jnz %zero, @done, @next
@next
	%i_next =l add %i, 1
	%p_next =l add %p, 8
	jmp @loop
@done
	ret %i
}
