export function l $map_into(l %dst, l %src, l %n, l %f) {
@entry
	jmp @test
@test
	%i =l phi @entry 0, @body %i1
	%count =l phi @entry 0, @body %count1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%ps =l add %src, %off
	%x =l loadl %ps
	%r =l call %f(l %x)
	%pd =l add %dst, %off
	storel %r, %pd
	%less =w csltl %r, %x
	%inc =l extsw %less
	%count1 =l add %count, %inc
	%i1 =l add %i, 1
	jmp @test
@done
	ret %count
}
