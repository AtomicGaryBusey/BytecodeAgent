export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i_next
	%count =l phi @start 0, @body %count_next
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w cugtl %value, %t
	%increment =l extuw %greater
	%count_next =l add %count, %increment
	%i_next =l add %i, 1
	jmp @loop
@end
	ret %count
}
