mat_pow:
push rbx
push rbp
push r12
push r13
push r14
mov r12, rcx
mov r13, r8
mov r14, rsi
mov rbx, rdx
mov rbp, rsi
imul rbp, rsi
xor ecx, ecx
.Lzero_out:
cmp rcx, rbp
jge .Lidentity_setup
mov qword ptr [r12 + rcx*8], 0
inc rcx
jmp .Lzero_out
.Lidentity_setup:
xor edx, edx
xor eax, eax
.Lidentity_diag:
cmp rdx, r14
jge .Lcopy_base_setup
mov qword ptr [r12 + rax*8], 1
lea rax, [rax + r14 + 1]
inc rdx
jmp .Lidentity_diag
.Lcopy_base_setup:
xor ecx, ecx
.Lcopy_base:
cmp rcx, rbp
jge .Lpower_setup
mov rax, qword ptr [rdi + rcx*8]
mov qword ptr [r13 + rcx*8], rax
inc rcx
jmp .Lcopy_base
.Lpower_setup:
test rbx, rbx
je .Lclear_setup
lea r11, [r13 + rbp*8]
.Lmultiply:
xor esi, esi
xor r10d, r10d
.Lrow:
xor edi, edi
.Lcol:
xor eax, eax
xor ecx, ecx
mov r8, rdi
mov r9, rsi
imul r9, r14
.Linner:
mov rdx, qword ptr [r12 + r9*8]
imul rdx, qword ptr [r13 + r8*8]
add rax, rdx
inc r9
add r8, r14
inc rcx
cmp rcx, r14
jl .Linner
mov qword ptr [r11 + r10*8], rax
inc r10
inc rdi
cmp rdi, r14
jl .Lcol
inc rsi
cmp rsi, r14
jl .Lrow
xor ecx, ecx
.Lcopy_product:
cmp rcx, rbp
jge .Lnext_power
mov rax, qword ptr [r11 + rcx*8]
mov qword ptr [r12 + rcx*8], rax
inc rcx
jmp .Lcopy_product
.Lnext_power:
dec rbx
jne .Lmultiply
.Lclear_setup:
lea r8, [rbp + rbp]
xor ecx, ecx
.Lclear_tmp:
cmp rcx, r8
jge .Ldone
mov qword ptr [r13 + rcx*8], 0
inc rcx
jmp .Lclear_tmp
.Ldone:
xor eax, eax
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
