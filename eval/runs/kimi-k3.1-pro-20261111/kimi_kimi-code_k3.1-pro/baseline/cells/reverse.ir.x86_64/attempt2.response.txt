export function l $reverse(l %a, l %n) {
@start
	%jinit =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %inext
	%j =l phi @start %jinit, @body %jnext
	%cond =w csltl %i, %j
	jnz %cond, @body, @done
@body
	%ioff =l mul %i, 8
	%joff =l mul %j, 8
	%p =l add %a, %ioff
	%q =l add %a, %joff
	%x =l loadl %p
	%y =l loadl %q
	storel %y, %p
	storel %x, %q
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	jmp @loop
@done
	ret 0
}
