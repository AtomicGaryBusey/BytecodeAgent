export function l $max_byte(l %p0, l %n0) {
@start
	jmp @loop
@loop
	%p =l phi @start %p0, @next %p1
	%n =l phi @start %n0, @next %n1
	%m =l phi @start 0, @next %m1
	%done =w ceql %n, 0
	jnz %done, @exit, @body
@body
	%b =l loadub %p
	%gt =w csgtl %b, %m
	jnz %gt, @update, @skip
@update
	jmp @next
@skip
	jmp @next
@next
	%m1 =l phi @update %b, @skip %m
	%p1 =l add %p, 1
	%n1 =l sub %n, 1
	jmp @loop
@exit
	ret %m
}
