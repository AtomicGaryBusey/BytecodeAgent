qsort_pairs:
push rbx
push rbp
push r12
push r13
push r14
mov rbx, rdi
mov r12, rsi
cmp r12, 1
jle .Ldone
mov rax, r12
shl rax, 4
lea rbp, [rbx + rax - 16]
xor r13d, r13d
xor r14d, r14d
.Lpartition_loop:
lea rax, [r12 - 1]
cmp r14, rax
jge .Lpartition_done
mov rdx, r14
shl rdx, 4
mov rcx, qword ptr [rbx + rdx]
cmp rcx, qword ptr [rbp]
jg .Lnext
jl .Lless
mov rcx, qword ptr [rbx + rdx + 8]
cmp rcx, qword ptr [rbp + 8]
jge .Lnext
.Lless:
mov rax, r13
shl rax, 4
mov rcx, qword ptr [rbx + rax]
mov rsi, qword ptr [rbx + rdx]
mov qword ptr [rbx + rax], rsi
mov qword ptr [rbx + rdx], rcx
mov rcx, qword ptr [rbx + rax + 8]
mov rsi, qword ptr [rbx + rdx + 8]
mov qword ptr [rbx + rax + 8], rsi
mov qword ptr [rbx + rdx + 8], rcx
inc r13
.Lnext:
inc r14
jmp .Lpartition_loop
.Lpartition_done:
mov rax, r13
shl rax, 4
mov rcx, qword ptr [rbx + rax]
mov rdx, qword ptr [rbp]
mov qword ptr [rbx + rax], rdx
mov qword ptr [rbp], rcx
mov rcx, qword ptr [rbx + rax + 8]
mov rdx, qword ptr [rbp + 8]
mov qword ptr [rbx + rax + 8], rdx
mov qword ptr [rbp + 8], rcx
mov rdi, rbx
mov rsi, r13
call qsort_pairs
lea rdi, [r13 + 1]
shl rdi, 4
add rdi, rbx
mov rsi, r12
sub rsi, r13
dec rsi
call qsort_pairs
.Ldone:
xor eax, eax
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
