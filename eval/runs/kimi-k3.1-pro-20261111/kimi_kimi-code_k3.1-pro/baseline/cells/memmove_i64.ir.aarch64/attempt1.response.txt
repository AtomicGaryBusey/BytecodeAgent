export function l $memmove_i64(l %dst, l %src, l %n) {
@entry
	%iszero =w ceql %n, 0
	jnz %iszero, @done, @choose
@choose
	%forward =w culel %dst, %src
	jnz %forward, @fwd_setup, @bwd_setup
@fwd_setup
	%nbytes =l mul %n, 8
	%end =l add %src, %nbytes
	jmp @fwd_loop
@fwd_loop
	%sp =l phi @fwd_setup %src, @fwd_body %spnext
	%dp =l phi @fwd_setup %dst, @fwd_body %dpnext
	%fdone =w ceql %sp, %end
	jnz %fdone, @done, @fwd_body
@fwd_body
	%value =l loadl %sp
	storel %value, %dp
	%spnext =l add %sp, 8
	%dpnext =l add %dp, 8
	jmp @fwd_loop
@bwd_setup
	%nbytes2 =l mul %n, 8
	%sp0 =l add %src, %nbytes2
	%dp0 =l add %dst, %nbytes2
	jmp @bwd_loop
@bwd_loop
	%bsp =l phi @bwd_setup %sp0, @bwd_body %bspnext
	%bdp =l phi @bwd_setup %dp0, @bwd_body %bdpnext
	%bdone =w ceql %bsp, %src
	jnz %bdone, @done, @bwd_body
@bwd_body
	%bspnext =l sub %bsp, 8
	%bdpnext =l sub %bdp, 8
	%bvalue =l loadl %bspnext
	storel %bvalue, %bdpnext
	jmp @bwd_loop
@done
	ret 0
}
