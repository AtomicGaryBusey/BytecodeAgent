export function l $count_desc(l %a, l %n) {
@start
	%last = sub %n, 1
	jmp @loop
@loop
	%i = phi @start 0, @body %i_next
	%count = phi @start 0, @body %count_next
	%more = csltl %i, %last
	jnz %more, @body, @done
@body
	%offset = mul %i, 8
	%p = add %a, %offset
	%x = loadl %p
	%p_next = add %p, 8
	%y = loadl %p_next
	%desc = csgtl %x, %y
	%inc = extuw %desc
	%count_next = add %count, %inc
	%i_next = add %i, 1
	jmp @loop
@done
	ret %count
}
