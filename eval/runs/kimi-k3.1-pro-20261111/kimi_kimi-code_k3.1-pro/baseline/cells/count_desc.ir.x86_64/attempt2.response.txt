export function l $count_desc(l %a, l %n) {
@start
	%last =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %i_next
	%count =l phi @start 0, @body %count_next
	%more =w csltl %i, %last
	jnz %more, @body, @done
@body
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%p_next =l add %p, 8
	%y =l loadl %p_next
	%desc =w csgtl %x, %y
	%inc =l extuw %desc
	%count_next =l add %count, %inc
	%i_next =l add %i, 1
	jmp @loop
@done
	ret %count
}
