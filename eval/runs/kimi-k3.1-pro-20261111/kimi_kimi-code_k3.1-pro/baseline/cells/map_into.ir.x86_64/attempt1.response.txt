export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i_next
	%count =l phi @start 0, @body %count_next
	%more =w csltl %i, %n
	jnz %more, @body, @end
@body
	%offset =l mul %i, 8
	%src_ptr =l add %src, %offset
	%x =l loadl %src_ptr
	%r =l call %f(l %x)
	%dst_ptr =l add %dst, %offset
	storel %r, %dst_ptr
	%less =w csltl %r, %x
	%increment =l extuw %less
	%count_next =l add %count, %increment
	%i_next =l add %i, 1
	jmp @loop
@end
	ret %count
}
