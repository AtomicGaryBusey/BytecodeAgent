max_byte:
    xor eax, eax
    test rsi, rsi
    jz .Ldone
    pxor xmm0, xmm0
    xor edx, edx
    cmp rsi, 16
    jb .Ltail
    mov rcx, rsi
    and rcx, -16
.Lvector:
    movdqu xmm1, xmmword ptr [rdi + rdx]
    pmaxub xmm0, xmm1
    add rdx, 16
    cmp rdx, rcx
    jb .Lvector
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    pextrb eax, xmm0, 0
.Ltail:
    cmp rdx, rsi
    jae .Ldone
    movzx ecx, byte ptr [rdi + rdx]
    cmp ecx, eax
    cmova eax, ecx
    inc rdx
    jmp .Ltail
.Ldone:
    ret
