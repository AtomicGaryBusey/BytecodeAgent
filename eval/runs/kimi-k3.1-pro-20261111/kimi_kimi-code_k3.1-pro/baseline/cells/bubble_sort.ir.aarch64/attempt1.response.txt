export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @outer_init
@outer_init
	%last0 =l sub %n, 1
	jmp @outer
@outer
	%last =l phi @outer_init %last0, @outer_next %last_next
	%active =w csgtl %last, 0
	jnz %active, @inner, @done
@inner
	%j =l phi @outer 0, @inner_next %j_next
	%more =w csltl %j, %last
	jnz %more, @body, @outer_next
@body
	%offset =l mul %j, 8
	%p =l add %a, %offset
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%greater =w csgtl %x, %y
	jnz %greater, @swap, @inner_next
@swap
	storel %y, %p
	storel %x, %p1
	jmp @inner_next
@inner_next
	%j_next =l add %j, 1
	jmp @inner
@outer_next
	%last_next =l sub %last, 1
	jmp @outer
@done
	ret 0
}
