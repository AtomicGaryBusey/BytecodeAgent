export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
	jmp @test
@test
	%i =l phi @start 1, @maxend %i1
	%mn =l phi @start %y0, @maxend %mn1
	%mx =l phi @start %y0, @maxend %mx1
	%sum =l phi @start %y0, @maxend %sum1
	%cnt =l phi @start %odd0, @maxend %cnt1
	%more =w csltl %i, %n
	jnz %more, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%sum1 =l add %sum, %y
	%bit =l and %y, 1
	%cnt1 =l add %cnt, %bit
	%i1 =l add %i, 1
	%ltmin =w csltl %y, %mn
	jnz %ltmin, @setmin, @minend
@setmin
	jmp @minend
@minend
	%mn1 =l phi @body %mn, @setmin %y
	%gtmax =w csltl %mx, %y
	jnz %gtmax, @setmax, @maxend
@setmax
	jmp @maxend
@maxend
	%mx1 =l phi @minend %mx, @setmax %y
	jmp @test
@done
	storel %mn, %out
	%out1 =l add %out, 8
	storel %mx, %out1
	%out2 =l add %out, 16
	storel %sum, %out2
	ret %cnt
}
