export function l $qsort_pairs(l %a, l %n) {
@entry
	%base =w csltl %n, 2
	jnz %base, @done, @setup
@done
	ret 0
@setup
	%last =l sub %n, 1
	%lastoff =l mul %last, 16
	%pivot =l add %a, %lastoff
	%pivotkey =l loadl %pivot
	%pivotvalptr =l add %pivot, 8
	%pivotval =l loadl %pivotvalptr
	jmp @loop
@loop
	%i =l phi @setup 0, @advance %inext
	%j =l phi @setup 0, @advance %jnext
	%more =w csltl %j, %last
	jnz %more, @inspect, @place
@inspect
	%joff =l mul %j, 16
	%jptr =l add %a, %joff
	%jkey =l loadl %jptr
	%keyless =w csltl %jkey, %pivotkey
	jnz %keyless, @swap, @keynotless
@keynotless
	%keyequal =w ceql %jkey, %pivotkey
	jnz %keyequal, @comparevalue, @stay
@comparevalue
	%cmpvalptr =l add %jptr, 8
	%cmpval =l loadl %cmpvalptr
	%valless =w csltl %cmpval, %pivotval
	jnz %valless, @swap, @valnotless
@valnotless
	%valequal =w ceql %cmpval, %pivotval
	jnz %valequal, @swap, @stay
@swap
	%ioff =l mul %i, 16
	%iptr =l add %a, %ioff
	%ikey =l loadl %iptr
	%ivalptr =l add %iptr, 8
	%ival =l loadl %ivalptr
	%jvalptr =l add %jptr, 8
	%jval =l loadl %jvalptr
	storel %jkey, %iptr
	storel %jval, %ivalptr
	storel %ikey, %jptr
	storel %ival, %jvalptr
	%inew =l add %i, 1
	jmp @advance
@stay
	jmp @advance
@advance
	%inext =l phi @stay %i, @swap %inew
	%jnext =l add %j, 1
	jmp @loop
@place
	%finaloff =l mul %i, 16
	%finalptr =l add %a, %finaloff
	%finalvalptr =l add %finalptr, 8
	%finalkey =l loadl %finalptr
	%finalval =l loadl %finalvalptr
	storel %pivotkey, %finalptr
	storel %pivotval, %finalvalptr
	storel %finalkey, %pivot
	storel %finalval, %pivotvalptr
	%leftresult =l call $qsort_pairs(l %a, l %i)
	%rightindex =l add %i, 1
	%rightoff =l mul %rightindex, 16
	%rightptr =l add %a, %rightoff
	%rightn =l sub %n, %rightindex
	%rightresult =l call $qsort_pairs(l %rightptr, l %rightn)
	ret 0
}
