collatz_max:
xor r8d, r8d
mov esi, 1
.Louter:
cmp rsi, rdi
jg .Ldone
mov rcx, rsi
xor edx, edx
.Linner:
cmp rcx, 1
je .Lfinish
test rcx, 1
jz .Leven
lea rcx, [rcx + rcx*2 + 1]
jmp .Lstep
.Leven:
shr rcx, 1
.Lstep:
inc rdx
jmp .Linner
.Lfinish:
cmp rdx, r8
cmovg r8, rdx
inc rsi
jmp .Louter
.Ldone:
mov rax, r8
ret
