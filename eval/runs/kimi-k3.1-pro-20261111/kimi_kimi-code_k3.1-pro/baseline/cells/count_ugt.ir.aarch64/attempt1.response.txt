export function l $count_ugt(l %a, l %n, l %t) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @zero, @loop
@loop
	%left =l phi @start %n, @loop %left1
	%ptr =l phi @start %a, @loop %ptr1
	%count =l phi @start 0, @loop %count1
	%value =l loadl %ptr
	%gt =w cugtl %value, %t
	%gtl =l extuw %gt
	%count1 =l add %count, %gtl
	%ptr1 =l add %ptr, 8
	%left1 =l sub %left, 1
	%more =w cnel %left1, 0
	jnz %more, @loop, @exit
@exit
	ret %count1
@zero
	ret 0
}
