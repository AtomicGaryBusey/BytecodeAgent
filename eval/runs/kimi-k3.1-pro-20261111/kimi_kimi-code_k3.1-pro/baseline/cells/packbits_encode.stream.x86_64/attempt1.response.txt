packbits_encode:
push rbx
push rbp
push r12
push r13
push r14
push r15
xor ebx, ebx
mov r12, rsi
mov r13, rdx
mov r14, rcx
xor r15d, r15d
.Louter:
cmp rbx, r12
jge .Ldone
lea r8, [rbx + 128]
cmp r8, r12
cmovg r8, r12
mov r9, rbx
mov r10b, byte ptr [rdi + rbx]
.Lfirst_run_loop:
inc r9
cmp r9, r8
jge .Lfirst_run_done
cmp byte ptr [rdi + r9], r10b
je .Lfirst_run_loop
.Lfirst_run_done:
sub r9, rbx
cmp r9, 3
jl .Lliteral_start
lea rax, [r15 + 2]
cmp rax, r14
jg .Lfail
mov eax, 257
sub rax, r9
mov byte ptr [r13 + r15], al
mov byte ptr [r13 + r15 + 1], r10b
add r15, 2
add rbx, r9
jmp .Louter
.Lliteral_start:
mov rbp, rbx
.Lliteral_loop:
cmp rbp, r12
jge .Lliteral_emit
mov rax, rbp
sub rax, rbx
cmp rax, 128
jge .Lliteral_emit
lea r8, [rbp + 128]
cmp r8, r12
cmovg r8, r12
mov r9, rbp
mov r10b, byte ptr [rdi + rbp]
.Lliteral_run_loop:
inc r9
cmp r9, r8
jge .Lliteral_run_done
cmp byte ptr [rdi + r9], r10b
je .Lliteral_run_loop
.Lliteral_run_done:
sub r9, rbp
cmp r9, 3
jge .Lliteral_emit
inc rbp
jmp .Lliteral_loop
.Lliteral_emit:
mov r9, rbp
sub r9, rbx
lea rax, [r15 + r9 + 1]
cmp rax, r14
jg .Lfail
lea rax, [r9 - 1]
mov byte ptr [r13 + r15], al
inc r15
lea r11, [rdi + rbx]
lea r8, [r13 + r15]
xor eax, eax
.Lliteral_copy:
cmp rax, r9
jge .Lliteral_copy_done
mov r10b, byte ptr [r11 + rax]
mov byte ptr [r8 + rax], r10b
inc rax
jmp .Lliteral_copy
.Lliteral_copy_done:
add r15, r9
mov rbx, rbp
jmp .Louter
.Ldone:
mov rax, r15
jmp .Lreturn
.Lfail:
mov rax, -1
.Lreturn:
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
