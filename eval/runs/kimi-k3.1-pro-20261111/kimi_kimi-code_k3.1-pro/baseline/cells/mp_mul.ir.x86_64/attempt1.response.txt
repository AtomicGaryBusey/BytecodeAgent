export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%n =l add %na, %nb
	%nempty =w ceql %n, 0
	jnz %nempty, @init_done, @init_loop
@init_loop
	%ii =l phi @entry 0, @init_loop %iin
	%iioff =l mul %ii, 8
	%iip =l add %r, %iioff
	storel 0, %iip
	%iin =l add %ii, 1
	%iimore =w csltl %iin, %n
	jnz %iimore, @init_loop, @init_done
@init_done
	%naempty =w ceql %na, 0
	jnz %naempty, @scan_setup, @outer_loop
@outer_loop
	%i =l phi @init_done 0, @store_carry %inext
	%aoff =l mul %i, 8
	%ap =l add %a, %aoff
	%av =l loadl %ap
	%a0 =l and %av, 4294967295
	%a1 =l shr %av, 32
	%nbempty =w ceql %nb, 0
	jnz %nbempty, @store_carry, @inner_loop
@inner_loop
	%j =l phi @outer_loop 0, @inner_loop %jn
	%carry =l phi @outer_loop 0, @inner_loop %ncarry
	%boff =l mul %j, 8
	%bp =l add %b, %boff
	%bv =l loadl %bp
	%b0 =l and %bv, 4294967295
	%b1 =l shr %bv, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%p00hi =l shr %p00, 32
	%p01lo =l and %p01, 4294967295
	%p10lo =l and %p10, 4294967295
	%mid0 =l add %p00hi, %p01lo
	%mid =l add %mid0, %p10lo
	%midshift =l shl %mid, 32
	%p00lo =l and %p00, 4294967295
	%lo =l add %midshift, %p00lo
	%p01hi =l shr %p01, 32
	%p10hi =l shr %p10, 32
	%midhi =l shr %mid, 32
	%hi0 =l add %p11, %p01hi
	%hi1 =l add %hi0, %p10hi
	%hi =l add %hi1, %midhi
	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%rp =l add %r, %roff
	%old =l loadl %rp
	%s0 =l add %old, %lo
	%c0 =w cultl %s0, %old
	%s1 =l add %s0, %carry
	%c1 =w cultl %s1, %s0
	storel %s1, %rp
	%c0l =l extuw %c0
	%c1l =l extuw %c1
	%ncarry0 =l add %hi, %c0l
	%ncarry =l add %ncarry0, %c1l
	%jn =l add %j, 1
	%jmore =w csltl %jn, %nb
	jnz %jmore, @inner_loop, @store_carry
@store_carry
	%cout =l phi @outer_loop 0, @inner_loop %ncarry
	%ci =l add %i, %nb
	%coff =l mul %ci, 8
	%cp =l add %r, %coff
	storel %cout, %cp
	%inext =l add %i, 1
	%imore =w csltl %inext, %na
	jnz %imore, @outer_loop, @scan_setup
@scan_setup
	%scanempty =w ceql %n, 0
	jnz %scanempty, @ret_zero, @scan_loop
@scan_loop
	%k =l phi @scan_setup %n, @scan_zero %kdec
	%km1 =l add %k, -1
	%koff =l mul %km1, 8
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kzero =w ceql %kv, 0
	jnz %kzero, @scan_zero, @ret_k
@scan_zero
	%kdec =l add %k, -1
	%kdempty =w ceql %kdec, 0
	jnz %kdempty, @ret_zero, @scan_loop
@ret_k
	ret %k
@ret_zero
	ret 0
}
