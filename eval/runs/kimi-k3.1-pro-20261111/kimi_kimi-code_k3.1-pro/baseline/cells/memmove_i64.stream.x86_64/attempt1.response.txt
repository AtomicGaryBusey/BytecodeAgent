memmove_i64:
test rdx, rdx
jz .Ldone
cmp rdi, rsi
jbe .Lforward
mov rcx, rdx
.Lbackward:
dec rcx
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
test rcx, rcx
jnz .Lbackward
jmp .Ldone
.Lforward:
xor ecx, ecx
.Lforward_loop:
mov rax, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], rax
inc rcx
cmp rcx, rdx
jb .Lforward_loop
.Ldone:
xor eax, eax
ret
