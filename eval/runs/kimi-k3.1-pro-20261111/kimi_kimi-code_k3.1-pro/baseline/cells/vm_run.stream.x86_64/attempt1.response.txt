vm_run:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    je .Lstop
    mov r11, qword ptr [rdi + r8*8]
    cmp r11, 0
    je .Lstop
    cmp r11, 1
    je .Lpush
    cmp r11, 2
    je .Ladd
    cmp r11, 3
    je .Lsub
    cmp r11, 4
    je .Lmul
    cmp r11, 5
    je .Ldup
    cmp r11, 6
    je .Lswap
    cmp r11, 7
    je .Ldrop
    cmp r11, 8
    je .Lover
    cmp r11, 9
    je .Llt
    cmp r11, 10
    je .Ljz
    cmp r11, 11
    je .Ljmp
    inc r8
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [rdi + r8*8 + 8]
    mov qword ptr [rdx + r9*8], rax
    inc r9
    add r8, 2
    jmp .Lloop
.Ladd:
    mov rax, qword ptr [rdx + r9*8 - 8]
    add qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Lsub:
    mov rax, qword ptr [rdx + r9*8 - 8]
    sub qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Lmul:
    mov rax, qword ptr [rdx + r9*8 - 8]
    imul rax, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [rdx + r9*8 - 8]
    mov qword ptr [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [rdx + r9*8 - 8]
    mov r10, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 8], r10
    mov qword ptr [rdx + r9*8 - 16], rax
    inc r8
    jmp .Lloop
.Ldrop:
    dec r9
    inc r8
    jmp .Lloop
.Lover:
    mov rax, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8], rax
    inc r9
    inc r8
    jmp .Lloop
.Llt:
    mov rax, qword ptr [rdx + r9*8 - 16]
    cmp rax, qword ptr [rdx + r9*8 - 8]
    setl al
    movzx rax, al
    mov qword ptr [rdx + r9*8 - 16], rax
    dec r9
    inc r8
    jmp .Lloop
.Ljz:
    mov rax, qword ptr [rdx + r9*8 - 8]
    dec r9
    test rax, rax
    jz .Ljz_taken
    add r8, 2
    jmp .Lloop
.Ljz_taken:
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Ljmp:
    mov r8, qword ptr [rdi + r8*8 + 8]
    jmp .Lloop
.Lstop:
    xor eax, eax
    test r9, r9
    jz .Lzero_start
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lzero_start:
    mov r10, r9
.Lzero_loop:
    cmp r10, rcx
    jae .Lreturn
    mov qword ptr [rdx + r10*8], 0
    inc r10
    jmp .Lzero_loop
.Lreturn:
    ret
