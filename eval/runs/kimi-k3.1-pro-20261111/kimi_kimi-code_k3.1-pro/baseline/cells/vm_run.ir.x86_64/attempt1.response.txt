export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%dp =l alloc8 8
	%retp =l alloc8 8
	storel 0, %pcp
	storel 0, %dp
	jmp @loop

@loop
	%pc =l loadl %pcp
	%at_end =w ceql %pc, %len
	jnz %at_end, @finish, @fetch

@fetch
	%pc_off =l mul %pc, 8
	%insn_addr =l add %code, %pc_off
	%op =l loadl %insn_addr
	%is_halt =w ceql %op, 0
	jnz %is_halt, @finish, @check_push

@check_push
	%is_push =w ceql %op, 1
	jnz %is_push, @push, @check_add

@check_add
	%is_add =w ceql %op, 2
	jnz %is_add, @add, @check_sub

@check_sub
	%is_sub =w ceql %op, 3
	jnz %is_sub, @sub, @check_mul

@check_mul
	%is_mul =w ceql %op, 4
	jnz %is_mul, @mul, @check_dup

@check_dup
	%is_dup =w ceql %op, 5
	jnz %is_dup, @dup, @check_swap

@check_swap
	%is_swap =w ceql %op, 6
	jnz %is_swap, @swap, @check_drop

@check_drop
	%is_drop =w ceql %op, 7
	jnz %is_drop, @drop, @check_over

@check_over
	%is_over =w ceql %op, 8
	jnz %is_over, @over, @check_lt

@check_lt
	%is_lt =w ceql %op, 9
	jnz %is_lt, @lt, @check_jz

@check_jz
	%is_jz =w ceql %op, 10
	jnz %is_jz, @jz, @check_jmp

@check_jmp
	%is_jmp =w ceql %op, 11
	jnz %is_jmp, @jump, @advance

@push
	%push_operand_pc =l add %pc, 1
	%push_operand_off =l mul %push_operand_pc, 8
	%push_operand_addr =l add %code, %push_operand_off
	%push_value =l loadl %push_operand_addr
	%push_d =l loadl %dp
	%push_slot_off =l mul %push_d, 8
	%push_slot =l add %stack, %push_slot_off
	storel %push_value, %push_slot
	%push_new_d =l add %push_d, 1
	storel %push_new_d, %dp
	%push_new_pc =l add %pc, 2
	storel %push_new_pc, %pcp
	jmp @loop

@add
	%add_d =l loadl %dp
	%add_b_index =l sub %add_d, 1
	%add_a_index =l sub %add_d, 2
	%add_b_off =l mul %add_b_index, 8
	%add_a_off =l mul %add_a_index, 8
	%add_b_addr =l add %stack, %add_b_off
	%add_a_addr =l add %stack, %add_a_off
	%add_b =l loadl %add_b_addr
	%add_a =l loadl %add_a_addr
	%add_result =l add %add_a, %add_b
	storel %add_result, %add_a_addr
	storel %add_b_index, %dp
	%add_new_pc =l add %pc, 1
	storel %add_new_pc, %pcp
	jmp @loop

@sub
	%sub_d =l loadl %dp
	%sub_b_index =l sub %sub_d, 1
	%sub_a_index =l sub %sub_d, 2
	%sub_b_off =l mul %sub_b_index, 8
	%sub_a_off =l mul %sub_a_index, 8
	%sub_b_addr =l add %stack, %sub_b_off
	%sub_a_addr =l add %stack, %sub_a_off
	%sub_b =l loadl %sub_b_addr
	%sub_a =l loadl %sub_a_addr
	%sub_result =l sub %sub_a, %sub_b
	storel %sub_result, %sub_a_addr
	storel %sub_b_index, %dp
	%sub_new_pc =l add %pc, 1
	storel %sub_new_pc, %pcp
	jmp @loop

@mul
	%mul_d =l loadl %dp
	%mul_b_index =l sub %mul_d, 1
	%mul_a_index =l sub %mul_d, 2
	%mul_b_off =l mul %mul_b_index, 8
	%mul_a_off =l mul %mul_a_index, 8
	%mul_b_addr =l add %stack, %mul_b_off
	%mul_a_addr =l add %stack, %mul_a_off
	%mul_b =l loadl %mul_b_addr
	%mul_a =l loadl %mul_a_addr
	%mul_result =l mul %mul_a, %mul_b
	storel %mul_result, %mul_a_addr
	storel %mul_b_index, %dp
	%mul_new_pc =l add %pc, 1
	storel %mul_new_pc, %pcp
	jmp @loop

@dup
	%dup_d =l loadl %dp
	%dup_top_index =l sub %dup_d, 1
	%dup_top_off =l mul %dup_top_index, 8
	%dup_dest_off =l mul %dup_d, 8
	%dup_top_addr =l add %stack, %dup_top_off
	%dup_dest_addr =l add %stack, %dup_dest_off
	%dup_value =l loadl %dup_top_addr
	storel %dup_value, %dup_dest_addr
	%dup_new_d =l add %dup_d, 1
	storel %dup_new_d, %dp
	%dup_new_pc =l add %pc, 1
	storel %dup_new_pc, %pcp
	jmp @loop

@swap
	%swap_d =l loadl %dp
	%swap_top_index =l sub %swap_d, 1
	%swap_below_index =l sub %swap_d, 2
	%swap_top_off =l mul %swap_top_index, 8
	%swap_below_off =l mul %swap_below_index, 8
	%swap_top_addr =l add %stack, %swap_top_off
	%swap_below_addr =l add %stack, %swap_below_off
	%swap_top =l loadl %swap_top_addr
	%swap_below =l loadl %swap_below_addr
	storel %swap_below, %swap_top_addr
	storel %swap_top, %swap_below_addr
	%swap_new_pc =l add %pc, 1
	storel %swap_new_pc, %pcp
	jmp @loop

@drop
	%drop_d =l loadl %dp
	%drop_new_d =l sub %drop_d, 1
	storel %drop_new_d, %dp
	%drop_new_pc =l add %pc, 1
	storel %drop_new_pc, %pcp
	jmp @loop

@over
	%over_d =l loadl %dp
	%over_source_index =l sub %over_d, 2
	%over_source_off =l mul %over_source_index, 8
	%over_dest_off =l mul %over_d, 8
	%over_source_addr =l add %stack, %over_source_off
	%over_dest_addr =l add %stack, %over_dest_off
	%over_value =l loadl %over_source_addr
	storel %over_value, %over_dest_addr
	%over_new_d =l add %over_d, 1
	storel %over_new_d, %dp
	%over_new_pc =l add %pc, 1
	storel %over_new_pc, %pcp
	jmp @loop

@lt
	%lt_d =l loadl %dp
	%lt_b_index =l sub %lt_d, 1
	%lt_a_index =l sub %lt_d, 2
	%lt_b_off =l mul %lt_b_index, 8
	%lt_a_off =l mul %lt_a_index, 8
	%lt_b_addr =l add %stack, %lt_b_off
	%lt_a_addr =l add %stack, %lt_a_off
	%lt_b =l loadl %lt_b_addr
	%lt_a =l loadl %lt_a_addr
	%lt_comparison =w csltl %lt_a, %lt_b
	%lt_result =l extuw %lt_comparison
	storel %lt_result, %lt_a_addr
	storel %lt_b_index, %dp
	%lt_new_pc =l add %pc, 1
	storel %lt_new_pc, %pcp
	jmp @loop

@jz
	%jz_d =l loadl %dp
	%jz_value_index =l sub %jz_d, 1
	%jz_value_off =l mul %jz_value_index, 8
	%jz_value_addr =l add %stack, %jz_value_off
	%jz_value =l loadl %jz_value_addr
	storel %jz_value_index, %dp
	%jz_operand_pc =l add %pc, 1
	%jz_operand_off =l mul %jz_operand_pc, 8
	%jz_operand_addr =l add %code, %jz_operand_off
	%jz_target =l loadl %jz_operand_addr
	%jz_is_zero =w ceql %jz_value, 0
	jnz %jz_is_zero, @jz_taken, @jz_not_taken

@jz_taken
	storel %jz_target, %pcp
	jmp @loop

@jz_not_taken
	%jz_new_pc =l add %pc, 2
	storel %jz_new_pc, %pcp
	jmp @loop

@jump
	%jump_operand_pc =l add %pc, 1
	%jump_operand_off =l mul %jump_operand_pc, 8
	%jump_operand_addr =l add %code, %jump_operand_off
	%jump_target =l loadl %jump_operand_addr
	storel %jump_target, %pcp
	jmp @loop

@advance
	%advance_pc =l add %pc, 1
	storel %advance_pc, %pcp
	jmp @loop

@finish
	%final_d =l loadl %dp
	%stack_empty =w ceql %final_d, 0
	jnz %stack_empty, @return_zero, @return_top

@return_zero
	storel 0, %retp
	jmp @zero_start

@return_top
	%return_index =l sub %final_d, 1
	%return_off =l mul %return_index, 8
	%return_addr =l add %stack, %return_off
	%return_value =l loadl %return_addr
	storel %return_value, %retp
	jmp @zero_start

@zero_start
	storel %final_d, %pcp
	jmp @zero_loop

@zero_loop
	%zero_index =l loadl %pcp
	%zero_done =w ceql %zero_index, %cap
	jnz %zero_done, @done, @zero_body

@zero_body
	%zero_off =l mul %zero_index, 8
	%zero_addr =l add %stack, %zero_off
	storel 0, %zero_addr
	%zero_next =l add %zero_index, 1
	storel %zero_next, %pcp
	jmp @zero_loop

@done
	%result =l loadl %retp
	ret %result
}
