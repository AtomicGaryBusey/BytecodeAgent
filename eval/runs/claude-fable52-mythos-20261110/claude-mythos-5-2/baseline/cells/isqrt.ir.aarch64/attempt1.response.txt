export function l $isqrt(l %n) {
@start
	%m =l copy %n
	%res =l copy 0
	%bit =l copy 4611686018427387904
@shrink
	%c =w cugtl %bit, %m
	jnz %c, @shrinkbody, @loop
@shrinkbody
	%bit =l shr %bit, 2
	jmp @shrink
@loop
	%c2 =w cnel %bit, 0
	jnz %c2, @body, @done
@body
	%t =l add %res, %bit
	%res =l shr %res, 1
	%c3 =w cugel %m, %t
	jnz %c3, @take, @next
@take
	%m =l sub %m, %t
	%res =l add %res, %bit
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
