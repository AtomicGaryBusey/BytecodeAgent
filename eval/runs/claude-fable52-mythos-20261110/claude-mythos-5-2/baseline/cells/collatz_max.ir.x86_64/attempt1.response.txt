export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%max =l copy 0
@outer
	%c =w csgtl %k, %n
	jnz %c, @done, @init
@init
	%x =l copy %k
	%steps =l copy 0
@inner
	%e =w ceql %x, 1
	jnz %e, @endinner, @step
@step
	%steps =l add %steps, 1
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @isodd, @iseven
@isodd
	%x =l mul %x, 3
	%x =l add %x, 1
	jmp @inner
@iseven
	%x =l shr %x, 1
	jmp @inner
@endinner
	%g =w csgtl %steps, %max
	jnz %g, @upd, @next
@upd
	%max =l copy %steps
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %max
}
