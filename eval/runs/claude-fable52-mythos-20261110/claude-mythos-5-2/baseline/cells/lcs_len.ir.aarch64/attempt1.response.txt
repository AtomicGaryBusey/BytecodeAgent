export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
@zloop
	%c =w cslel %j, %n
	jnz %c, @zbody, @zdone
@zbody
	%o =l shl %j, 3
	%p =l add %row, %o
	storel 0, %p
	%j =l add %j, 1
	jmp @zloop
@zdone
	%i =l copy 0
@oloop
	%c1 =w csltl %i, %m
	jnz %c1, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%diag =l copy 0
	%k =l copy 1
@iloop
	%c2 =w cslel %k, %n
	jnz %c2, @ibody, @idone
@ibody
	%o2 =l shl %k, 3
	%rp =l add %row, %o2
	%up =l loadl %rp
	%yp0 =l add %y, %k
	%yp =l sub %yp0, 1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%v =l add %diag, 1
	jmp @store
@nomatch
	%lp =l sub %rp, 8
	%left =l loadl %lp
	%gt =w csgtl %left, %up
	jnz %gt, @useleft, @useup
@useleft
	%v =l copy %left
	jmp @store
@useup
	%v =l copy %up
@store
	storel %v, %rp
	%diag =l copy %up
	%k =l add %k, 1
	jmp @iloop
@idone
	%i =l add %i, 1
	jmp @oloop
@done
	%o3 =l shl %n, 3
	%fp =l add %row, %o3
	%r =l loadl %fp
	ret %r
}
