packbits_encode:
    xor r8d, r8d
    xor r9d, r9d
.Lmain:
    cmp r8, rsi
    jge .Ldone
    mov r11, r8
.Llit:
    cmp r11, rsi
    jge .Llitend
    mov rax, r11
    sub rax, r8
    cmp rax, 128
    jge .Llitend
    lea rax, [r11 + 2]
    cmp rax, rsi
    jge .Lnext
    mov al, byte ptr [rdi + r11]
    cmp al, byte ptr [rdi + r11 + 1]
    jne .Lnext
    cmp al, byte ptr [rdi + r11 + 2]
    je .Llitend
.Lnext:
    inc r11
    jmp .Llit
.Llitend:
    mov r10, r11
    sub r10, r8
    jz .Lrepeat
    lea rax, [r9 + r10 + 1]
    cmp rax, rcx
    jg .Lfail
    lea eax, [r10 - 1]
    mov byte ptr [rdx + r9], al
    inc r9
.Lcopy:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    cmp r8, r11
    jl .Lcopy
    jmp .Lmain
.Lrepeat:
    mov al, byte ptr [rdi + r8]
    mov r10d, 3
.Lcnt:
    cmp r10, 128
    jge .Lcntend
    lea r11, [r8 + r10]
    cmp r11, rsi
    jge .Lcntend
    cmp al, byte ptr [rdi + r11]
    jne .Lcntend
    inc r10
    jmp .Lcnt
.Lcntend:
    lea r11, [r9 + 2]
    cmp r11, rcx
    jg .Lfail
    mov byte ptr [rdx + r9 + 1], al
    mov eax, 1
    sub eax, r10d
    mov byte ptr [rdx + r9], al
    mov r9, r11
    add r8, r10
    jmp .Lmain
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
