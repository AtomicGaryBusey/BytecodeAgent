binary_search:
    xor ecx, ecx                # lo = 0
    mov r8, rsi                 # hi = n
.Lloop:
    cmp rcx, r8
    jge .Lnotfound
    mov rax, rcx
    add rax, r8
    shr rax, 1                  # mid = (lo + hi) / 2
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    je .Ldone
    jl .Lless
    mov r8, rax                 # a[mid] > key: hi = mid
    jmp .Lloop
.Lless:
    lea rcx, [rax + 1]          # a[mid] < key: lo = mid + 1
    jmp .Lloop
.Lnotfound:
    mov rax, -1
.Ldone:
    ret
