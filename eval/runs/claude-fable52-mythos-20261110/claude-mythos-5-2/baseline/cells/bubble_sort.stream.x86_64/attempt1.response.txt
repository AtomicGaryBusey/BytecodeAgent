bubble_sort:
    mov rcx, rsi
.Louter:
    cmp rcx, 2
    jl .Ldone
    dec rcx
    xor edx, edx
.Linner:
    mov rax, qword ptr [rdi + rdx*8]
    mov r8, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r8
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r8
    mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
    inc rdx
    cmp rdx, rcx
    jl .Linner
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
