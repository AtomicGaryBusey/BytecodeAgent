mp_mul:
push rbx
push r12
push r13
mov r9, rdx
lea r10, [rsi + rcx]
xor eax, eax
xor r11d, r11d
.Lzero:
cmp r11, r10
jge .Lzdone
mov qword ptr [r8 + r11*8], rax
inc r11
jmp .Lzero
.Lzdone:
test rsi, rsi
jle .Lcount
test rcx, rcx
jle .Lcount
xor r11d, r11d
.Louter:
lea r12, [r8 + r11*8]
xor r13d, r13d
xor ebx, ebx
.Linner:
mov rax, qword ptr [r9 + rbx*8]
mul qword ptr [rdi + r11*8]
add rax, r13
adc rdx, 0
add rax, qword ptr [r12 + rbx*8]
adc rdx, 0
mov qword ptr [r12 + rbx*8], rax
mov r13, rdx
inc rbx
cmp rbx, rcx
jl .Linner
mov qword ptr [r12 + rcx*8], r13
inc r11
cmp r11, rsi
jl .Louter
.Lcount:
mov rax, r10
.Lc:
test rax, rax
jle .Ldone
cmp qword ptr [r8 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lc
.Ldone:
pop r13
pop r12
pop rbx
ret
