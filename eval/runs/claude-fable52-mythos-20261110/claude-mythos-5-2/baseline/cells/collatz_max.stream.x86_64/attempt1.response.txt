collatz_max:
    xor eax, eax
    mov rcx, 1
.Louter:
    cmp rcx, rdi
    jg .Ldone
    mov rdx, rcx
    xor esi, esi
.Linner:
    cmp rdx, 1
    je .Lend
    inc rsi
    test dl, 1
    jnz .Lodd
    shr rdx, 1
    jmp .Linner
.Lodd:
    lea rdx, [rdx + rdx*2 + 1]
    jmp .Linner
.Lend:
    cmp rsi, rax
    cmovg rax, rsi
    inc rcx
    jmp .Louter
.Ldone:
    ret
