export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%n1 =l add %n, 1
	%i =l copy 0
@idloop
	%r =l urem %i, %n1
	%c =w ceql %r, 0
	%v =l extuw %c
	%o =l shl %i, 3
	%p =l add %out, %o
	storel %v, %p
	%i =l add %i, 1
	%c =w csltl %i, %nn
	jnz %c, @idloop, @kcheck
@kcheck
	%c =w csgtl %k, 0
	jnz %c, @mul, @zero
@mul
	%i =l copy 0
@iloop
	%j =l copy 0
@jloop
	%s =l copy 0
	%t =l copy 0
@tloop
	%in =l mul %i, %n
	%xi =l add %in, %t
	%xo =l shl %xi, 3
	%xp =l add %out, %xo
	%x =l loadl %xp
	%tn =l mul %t, %n
	%yi =l add %tn, %j
	%yo =l shl %yi, 3
	%yp =l add %a, %yo
	%y =l loadl %yp
	%m =l mul %x, %y
	%s =l add %s, %m
	%t =l add %t, 1
	%c =w csltl %t, %n
	jnz %c, @tloop, @jend
@jend
	%in2 =l mul %i, %n
	%di =l add %in2, %j
	%do =l shl %di, 3
	%dp =l add %tmp, %do
	storel %s, %dp
	%j =l add %j, 1
	%c =w csltl %j, %n
	jnz %c, @jloop, @iend
@iend
	%i =l add %i, 1
	%c =w csltl %i, %n
	jnz %c, @iloop, @copy
@copy
	%i =l copy 0
@cloop
	%o =l shl %i, 3
	%sp =l add %tmp, %o
	%v =l loadl %sp
	%p =l add %out, %o
	storel %v, %p
	%i =l add %i, 1
	%c =w csltl %i, %nn
	jnz %c, @cloop, @cend
@cend
	%k =l sub %k, 1
	jmp @kcheck
@zero
	%nn2 =l add %nn, %nn
	%i =l copy 0
@zloop
	%o =l shl %i, 3
	%p =l add %tmp, %o
	storel 0, %p
	%i =l add %i, 1
	%c =w csltl %i, %nn2
	jnz %c, @zloop, @done
@done
	ret 0
}
