export function l $bubble_sort(l %a, l %n) {
@start
	%end =l copy %n
@outer
	%c =w csgtl %end, 1
	jnz %c, @init, @done
@init
	%i =l copy 1
@inner
	%c2 =w csltl %i, %end
	jnz %c2, @body, @next
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%q =l sub %p, 8
	%x =l loadl %q
	%y =l loadl %p
	%c3 =w csgtl %x, %y
	jnz %c3, @swap, @cont
@swap
	storel %y, %q
	storel %x, %p
@cont
	%i =l add %i, 1
	jmp @inner
@next
	%end =l sub %end, 1
	jmp @outer
@done
	ret 0
}
