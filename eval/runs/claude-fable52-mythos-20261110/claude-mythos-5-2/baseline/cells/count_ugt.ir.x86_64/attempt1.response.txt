export function l $count_ugt(l %a, l %n, l %t) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%c =l phi @start 0, @body %c2
	%lt =w csltl %i, %n
	jnz %lt, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%g =w cugtl %v, %t
	%ge =l extuw %g
	%c2 =l add %c, %ge
	%i2 =l add %i, 1
	jmp @loop
@done
	ret %c
}
