export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%i =l copy 0
@zloop
	%c =w csltl %i, %total
	jnz %c, @zbody, @zdone
@zbody
	%off =l shl %i, 3
	%p =l add %r, %off
	storel 0, %p
	%i =l add %i, 1
	jmp @zloop
@zdone
	%i =l copy 0
@oloop
	%c =w csltl %i, %na
	jnz %c, @obody, @odone
@obody
	%off =l shl %i, 3
	%p =l add %a, %off
	%ai =l loadl %p
	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	%carry =l copy 0
	%j =l copy 0
@iloop
	%c =w csltl %j, %nb
	jnz %c, @ibody, @idone
@ibody
	%off =l shl %j, 3
	%p =l add %b, %off
	%bj =l loadl %p
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%m0 =l shr %p00, 32
	%m1 =l and %p10, 4294967295
	%m2 =l and %p01, 4294967295
	%mid =l add %m0, %m1
	%mid =l add %mid, %m2
	%lo0 =l and %p00, 4294967295
	%midl =l shl %mid, 32
	%lo =l or %lo0, %midl
	%h1 =l shr %p10, 32
	%h2 =l shr %p01, 32
	%h3 =l shr %mid, 32
	%hi =l add %p11, %h1
	%hi =l add %hi, %h2
	%hi =l add %hi, %h3
	%ij =l add %i, %j
	%off =l shl %ij, 3
	%rp =l add %r, %off
	%rv =l loadl %rp
	%s1 =l add %lo, %rv
	%c1 =w cultl %s1, %lo
	%c1l =l extuw %c1
	%hi =l add %hi, %c1l
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	%c2l =l extuw %c2
	%hi =l add %hi, %c2l
	storel %s2, %rp
	%carry =l copy %hi
	%j =l add %j, 1
	jmp @iloop
@idone
	%ij =l add %i, %nb
	%off =l shl %ij, 3
	%rp =l add %r, %off
	storel %carry, %rp
	%i =l add %i, 1
	jmp @oloop
@odone
	%k =l copy %total
@tloop
	%c =w csgtl %k, 0
	jnz %c, @tcheck, @tdone
@tcheck
	%k1 =l sub %k, 1
	%off =l shl %k1, 3
	%p =l add %r, %off
	%v =l loadl %p
	%c =w ceql %v, 0
	jnz %c, @tdec, @tdone
@tdec
	%k =l copy %k1
	jmp @tloop
@tdone
	ret %k
}
