export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	jmp @loop
@loop
	%m =l phi @start %m0, @next %m2
	%i =l phi @start 1, @next %i1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%v =l loadl %p
	%g =w csgtl %v, %m
	jnz %g, @take, @keep
@take
	jmp @next
@keep
	jmp @next
@next
	%m2 =l phi @take %v, @keep %m
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %m
}
