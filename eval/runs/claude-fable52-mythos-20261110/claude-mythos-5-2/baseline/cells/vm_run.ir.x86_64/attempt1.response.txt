export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
@loop
	%c =w ceql %pc, %len
	jnz %c, @end, @fetch
@fetch
	%o =l mul %pc, 8
	%p =l add %code, %o
	%op =l loadl %p
	%p2 =l add %p, 8
	%to =l mul %d, 8
	%tp =l add %stack, %to
	%bp =l sub %tp, 8
	%ap =l sub %tp, 16
	%c =w ceql %op, 0
	jnz %c, @end, @n1
@n1
	%c =w ceql %op, 1
	jnz %c, @push, @n2
@n2
	%c =w ceql %op, 2
	jnz %c, @add, @n3
@n3
	%c =w ceql %op, 3
	jnz %c, @sub, @n4
@n4
	%c =w ceql %op, 4
	jnz %c, @mul, @n5
@n5
	%c =w ceql %op, 5
	jnz %c, @dup, @n6
@n6
	%c =w ceql %op, 6
	jnz %c, @swap, @n7
@n7
	%c =w ceql %op, 7
	jnz %c, @drop, @n8
@n8
	%c =w ceql %op, 8
	jnz %c, @over, @n9
@n9
	%c =w ceql %op, 9
	jnz %c, @lt, @n10
@n10
	%c =w ceql %op, 10
	jnz %c, @jz, @jmp
@push
	%v =l loadl %p2
	storel %v, %tp
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%b =l loadl %bp
	%a =l loadl %ap
	%r =l add %a, %b
	storel %r, %ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@sub
	%b =l loadl %bp
	%a =l loadl %ap
	%r =l sub %a, %b
	storel %r, %ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@mul
	%b =l loadl %bp
	%a =l loadl %ap
	%r =l mul %a, %b
	storel %r, %ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@dup
	%b =l loadl %bp
	storel %b, %tp
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%b =l loadl %bp
	%a =l loadl %ap
	storel %b, %ap
	storel %a, %bp
	%pc =l add %pc, 1
	jmp @loop
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%a =l loadl %ap
	storel %a, %tp
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@lt
	%b =l loadl %bp
	%a =l loadl %ap
	%rw =w csltl %a, %b
	%r =l extuw %rw
	storel %r, %ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%v =l loadl %bp
	%d =l sub %d, 1
	%c =w ceql %v, 0
	jnz %c, @jzt, @jzf
@jzt
	%pc =l loadl %p2
	jmp @loop
@jzf
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%pc =l loadl %p2
	jmp @loop
@end
	%i =l copy %d
@zl
	%c =w csltl %i, %cap
	jnz %c, @zb, @fin
@zb
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i =l add %i, 1
	jmp @zl
@fin
	%c =w ceql %d, 0
	jnz %c, @r0, @r1
@r0
	ret 0
@r1
	%ro =l mul %d, 8
	%rp =l add %stack, %ro
	%rp =l sub %rp, 8
	%res =l loadl %rp
	ret %res
}
