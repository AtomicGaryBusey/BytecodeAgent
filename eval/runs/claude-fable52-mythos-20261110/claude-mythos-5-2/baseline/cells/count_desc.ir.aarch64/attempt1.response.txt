export function l $count_desc(l %a, l %n) {
@start
	%m =l sub %n, 1
@loop
	%i =l phi @start 0, @body %i2
	%c =l phi @start 0, @body %c2
	%t =w csltl %i, %m
	jnz %t, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%c2 =l add %c, %ge
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %c
}
