export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j =l copy 0
@zloop
	%c =w cslel %j, %n
	jnz %c, @zbody, @zdone
@zbody
	%o =l mul %j, 8
	%p =l add %row, %o
	storel 0, %p
	%j =l add %j, 1
	jmp @zloop
@zdone
	%i =l copy 0
@oloop
	%c =w csltl %i, %m
	jnz %c, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%diag =l copy 0
	%j =l copy 1
@iloop
	%c =w cslel %j, %n
	jnz %c, @ibody, @idone
@ibody
	%o =l mul %j, 8
	%p =l add %row, %o
	%up =l loadl %p
	%yp =l add %y, %j
	%yp =l sub %yp, 1
	%yb =w loadub %yp
	%e =w ceqw %xb, %yb
	jnz %e, @match, @nomatch
@match
	%v =l add %diag, 1
	jmp @store
@nomatch
	%pl =l sub %p, 8
	%left =l loadl %pl
	%g =w csgtl %left, %up
	jnz %g, @useleft, @useup
@useleft
	%v =l copy %left
	jmp @store
@useup
	%v =l copy %up
@store
	storel %v, %p
	%diag =l copy %up
	%j =l add %j, 1
	jmp @iloop
@idone
	%i =l add %i, 1
	jmp @oloop
@done
	%o =l mul %n, 8
	%p =l add %row, %o
	%r =l loadl %p
	ret %r
}
