crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
    add rsi, rdi
.Lbyte:
    movzx edx, byte ptr [rdi]
    xor eax, edx
    mov ecx, 8
.Lbit:
    mov edx, eax
    and edx, 1
    neg edx
    and edx, 0xEDB88320
    shr eax, 1
    xor eax, edx
    dec ecx
    jnz .Lbit
    inc rdi
    cmp rdi, rsi
    jb .Lbyte
.Ldone:
    not eax
    ret
