dot_i32_simd:
    xor eax, eax
    xor ecx, ecx
    pxor xmm0, xmm0
    mov r8, rdx
    and r8, -4
    jz .Ltail
.Lloop:
    pmovsxdq xmm1, qword ptr [rdi + rcx*4]
    pmovsxdq xmm2, qword ptr [rsi + rcx*4]
    pmuldq xmm1, xmm2
    paddq xmm0, xmm1
    pmovsxdq xmm3, qword ptr [rdi + rcx*4 + 8]
    pmovsxdq xmm4, qword ptr [rsi + rcx*4 + 8]
    pmuldq xmm3, xmm4
    paddq xmm0, xmm3
    add rcx, 4
    cmp rcx, r8
    jl .Lloop
    movq rax, xmm0
    psrldq xmm0, 8
    movq r9, xmm0
    add rax, r9
.Ltail:
    cmp rcx, rdx
    jge .Ldone
.Lscalar:
    movsxd r9, dword ptr [rdi + rcx*4]
    movsxd r10, dword ptr [rsi + rcx*4]
    imul r9, r10
    add rax, r9
    inc rcx
    cmp rcx, rdx
    jl .Lscalar
.Ldone:
    ret
