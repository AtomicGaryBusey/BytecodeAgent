export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @gnext %i2
	%o =l phi @start 0, @gnext %o2
	%c =w csltl %i, %n
	jnz %c, @group, @done
@group
	%p =l add %src, %i
	%b0 =l loadub %p
	%v0 =l shl %b0, 16
	%r =l sub %n, %i
	%c1 =w csgtl %r, 1
	jnz %c1, @two, @build
@two
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%v1 =l or %v0, %s1
	%c2 =w csgtl %r, 2
	jnz %c2, @three, @build
@three
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @build
@build
	%v =l phi @group %v0, @two %v1, @three %v2
	%k =l phi @group 1, @two 2, @three 3
	jmp @emit
@emit
	%j =l phi @build 0, @store %j2
	%sh =w phi @build 18, @store %sh2
	%cp =w csgtl %j, %k
	jnz %cp, @pad, @map
@map
	%x0 =l shr %v, %sh
	%x1 =l and %x0, 63
	%x =w copy %x1
	%ch0 =w add %x, 65
	%g26 =w csgew %x, 26
	%a1 =w mul %g26, 6
	%ch1 =w add %ch0, %a1
	%g52 =w csgew %x, 52
	%a2 =w mul %g52, 75
	%ch2 =w sub %ch1, %a2
	%e62 =w ceqw %x, 62
	%a3 =w mul %e62, 15
	%ch3 =w sub %ch2, %a3
	%e63 =w ceqw %x, 63
	%a4 =w mul %e63, 12
	%ch4 =w sub %ch3, %a4
	jmp @store
@pad
	jmp @store
@store
	%ch =w phi @map %ch4, @pad 61
	%q0 =l add %out, %o
	%q =l add %q0, %j
	storeb %ch, %q
	%j2 =l add %j, 1
	%sh2 =w sub %sh, 6
	%cj =w csltl %j2, 4
	jnz %cj, @emit, @gnext
@gnext
	%i2 =l add %i, 3
	%o2 =l add %o, 4
	jmp @loop
@done
	ret %o
}
