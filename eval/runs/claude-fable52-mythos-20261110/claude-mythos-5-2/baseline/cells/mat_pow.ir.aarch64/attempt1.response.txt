export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%n1 =l add %n, 1
	%i =l copy 0
@z
	%c =w csltl %i, %nn
	jnz %c, @zb, @zd
@zb
	%o =l shl %i, 3
	%p =l add %out, %o
	storel 0, %p
	%i =l add %i, 1
	jmp @z
@zd
	%i =l copy 0
@d
	%c =w csltl %i, %n
	jnz %c, @db, @dd
@db
	%o =l mul %i, %n1
	%o =l shl %o, 3
	%p =l add %out, %o
	storel 1, %p
	%i =l add %i, 1
	jmp @d
@dd
	%cnt =l copy 0
@m
	%c =w csltl %cnt, %k
	jnz %c, @mi, @fin
@mi
	%i =l copy 0
@il
	%c =w csltl %i, %n
	jnz %c, @jinit, @cp
@jinit
	%j =l copy 0
	%in =l mul %i, %n
@jl
	%c =w csltl %j, %n
	jnz %c, @tinit, @inext
@tinit
	%t =l copy 0
	%s =l copy 0
@tl
	%c =w csltl %t, %n
	jnz %c, @tb, @td
@tb
	%o =l add %in, %t
	%o =l shl %o, 3
	%p =l add %out, %o
	%x =l loadl %p
	%o2 =l mul %t, %n
	%o2 =l add %o2, %j
	%o2 =l shl %o2, 3
	%p2 =l add %a, %o2
	%y =l loadl %p2
	%pr =l mul %x, %y
	%s =l add %s, %pr
	%t =l add %t, 1
	jmp @tl
@td
	%o =l add %in, %j
	%o =l shl %o, 3
	%p =l add %tmp, %o
	storel %s, %p
	%j =l add %j, 1
	jmp @jl
@inext
	%i =l add %i, 1
	jmp @il
@cp
	%i =l copy 0
@cl
	%c =w csltl %i, %nn
	jnz %c, @cb, @mnext
@cb
	%o =l shl %i, 3
	%p =l add %tmp, %o
	%x =l loadl %p
	%p2 =l add %out, %o
	storel %x, %p2
	%i =l add %i, 1
	jmp @cl
@mnext
	%cnt =l add %cnt, 1
	jmp @m
@fin
	%nn2 =l shl %nn, 1
	%i =l copy 0
@fl
	%c =w csltl %i, %nn2
	jnz %c, @fb, @done
@fb
	%o =l shl %i, 3
	%p =l add %tmp, %o
	storel 0, %p
	%i =l add %i, 1
	jmp @fl
@done
	ret 0
}
