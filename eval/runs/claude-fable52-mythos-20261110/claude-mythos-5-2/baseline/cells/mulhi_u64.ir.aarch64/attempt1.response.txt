export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%ll =l mul %alo, %blo
	%lh =l mul %alo, %bhi
	%hl =l mul %ahi, %blo
	%hh =l mul %ahi, %bhi
	%llhi =l shr %ll, 32
	%t =l add %hl, %llhi
	%tlo =l and %t, 4294967295
	%thi =l shr %t, 32
	%u =l add %lh, %tlo
	%uhi =l shr %u, 32
	%r0 =l add %hh, %thi
	%r =l add %r0, %uhi
	ret %r
}
