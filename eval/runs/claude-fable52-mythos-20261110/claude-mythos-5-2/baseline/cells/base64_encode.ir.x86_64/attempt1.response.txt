export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @gdone %i3
	%o =l phi @start 0, @gdone %o4
	%rem =l sub %n, %i
	%c0 =w csgtl %rem, 0
	jnz %c0, @grp, @end
@grp
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%c1 =w csgtl %rem, 1
	jnz %c1, @two, @build
@two
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%v1 =l or %v0, %s1
	%c2 =w csgtl %rem, 2
	jnz %c2, @three, @build
@three
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @build
@build
	%v =l phi @grp %v0, @two %v1, @three %v2
	%k =l phi @grp 2, @two 3, @three 4
	jmp @inner
@inner
	%j =l phi @build 0, @store %j1
	%sh =l phi @build 18, @store %sh1
	%cj =w csltl %j, 4
	jnz %cj, @body, @gdone
@body
	%ck =w csltl %j, %k
	jnz %ck, @mapc, @pad
@mapc
	%shw =w copy %sh
	%t =l shr %v, %shw
	%x =l and %t, 63
	%g1 =w csgtl %x, 25
	%e1 =l extuw %g1
	%a1 =l mul %e1, 6
	%g2 =w csgtl %x, 51
	%e2 =l extuw %g2
	%a2 =l mul %e2, 75
	%g3 =w csgtl %x, 61
	%e3 =l extuw %g3
	%a3 =l mul %e3, 15
	%g4 =w csgtl %x, 62
	%e4 =l extuw %g4
	%a4 =l mul %e4, 3
	%m0 =l add %x, 65
	%m1 =l add %m0, %a1
	%m2 =l sub %m1, %a2
	%m3 =l sub %m2, %a3
	%m =l add %m3, %a4
	jmp @store
@pad
	jmp @store
@store
	%ch =l phi @mapc %m, @pad 61
	%chw =w copy %ch
	%op0 =l add %out, %o
	%op =l add %op0, %j
	storeb %chw, %op
	%j1 =l add %j, 1
	%sh1 =l sub %sh, 6
	jmp @inner
@gdone
	%i3 =l add %i, 3
	%o4 =l add %o, 4
	jmp @outer
@end
	ret %o
}
