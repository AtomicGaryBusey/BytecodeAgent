export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%j =l copy %i
@lit
	%c =w csltl %j, %n
	jnz %c, @lit2, @litend
@lit2
	%d =l sub %j, %i
	%c =w csltl %d, 128
	jnz %c, @lit3, @litend
@lit3
	%j2 =l add %j, 2
	%c =w csltl %j2, %n
	jnz %c, @lit4, @litadv
@lit4
	%p =l add %src, %j
	%b0 =w loadub %p
	%p1 =l add %p, 1
	%b1 =w loadub %p1
	%c =w ceqw %b0, %b1
	jnz %c, @lit5, @litadv
@lit5
	%p2 =l add %p, 2
	%b2 =w loadub %p2
	%c =w ceqw %b0, %b2
	jnz %c, @litend, @litadv
@litadv
	%j =l add %j, 1
	jmp @lit
@litend
	%L =l sub %j, %i
	%c =w cnel %L, 0
	jnz %c, @emitlit, @run
@emitlit
	%need =l add %o, %L
	%need =l add %need, 1
	%c =w csgtl %need, %cap
	jnz %c, @fail, @litok
@litok
	%h =l sub %L, 1
	%hw =w copy %h
	%q =l add %out, %o
	storeb %hw, %q
	%o =l add %o, 1
	%k =l copy %i
@copy
	%c =w csltl %k, %j
	jnz %c, @copybody, @copyend
@copybody
	%sp =l add %src, %k
	%bv =w loadub %sp
	%q =l add %out, %o
	storeb %bv, %q
	%o =l add %o, 1
	%k =l add %k, 1
	jmp @copy
@copyend
	%i =l copy %j
	jmp @loop
@run
	%sp =l add %src, %i
	%b =w loadub %sp
	%r =l copy 3
@runloop
	%e =l add %i, %r
	%c =w csltl %e, %n
	jnz %c, @run2, @runend
@run2
	%c =w csltl %r, 128
	jnz %c, @run3, @runend
@run3
	%sp =l add %src, %e
	%bv =w loadub %sp
	%c =w ceqw %bv, %b
	jnz %c, @runinc, @runend
@runinc
	%r =l add %r, 1
	jmp @runloop
@runend
	%need =l add %o, 2
	%c =w csgtl %need, %cap
	jnz %c, @fail, @runok
@runok
	%h =l sub 257, %r
	%hw =w copy %h
	%q =l add %out, %o
	storeb %hw, %q
	%q =l add %q, 1
	storeb %b, %q
	%o =l add %o, 2
	%i =l add %i, %r
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
