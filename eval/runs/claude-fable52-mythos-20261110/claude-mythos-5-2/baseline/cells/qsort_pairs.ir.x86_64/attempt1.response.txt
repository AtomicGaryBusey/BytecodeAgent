export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @init
@init
	%last =l sub %n, 1
	%lo =l mul %last, 16
	%lp =l add %a, %lo
	%lp8 =l add %lp, 8
	%mid =l shr %n, 1
	%mo =l mul %mid, 16
	%mp =l add %a, %mo
	%mp8 =l add %mp, 8
	%mk =l loadl %mp
	%mv =l loadl %mp8
	%lk =l loadl %lp
	%lv =l loadl %lp8
	storel %lk, %mp
	storel %lv, %mp8
	storel %mk, %lp
	storel %mv, %lp8
@loop
	%i =l phi @init 0, @next %i2
	%j =l phi @init 0, @next %j2
	%cj =w csltl %j, %last
	jnz %cj, @body, @fin
@body
	%jo =l mul %j, 16
	%jp =l add %a, %jo
	%jp8 =l add %jp, 8
	%kj =l loadl %jp
	%vj =l loadl %jp8
	%lt =w csltl %kj, %mk
	jnz %lt, @swap, @chkeq
@chkeq
	%eq =w ceql %kj, %mk
	jnz %eq, @chkv, @noswap
@chkv
	%vlt =w csltl %vj, %mv
	jnz %vlt, @swap, @noswap
@swap
	%io =l mul %i, 16
	%ip =l add %a, %io
	%ip8 =l add %ip, 8
	%ki =l loadl %ip
	%vi =l loadl %ip8
	storel %kj, %ip
	storel %vj, %ip8
	storel %ki, %jp
	storel %vi, %jp8
	%i1 =l add %i, 1
	jmp @next
@noswap
	jmp @next
@next
	%i2 =l phi @swap %i1, @noswap %i
	%j2 =l add %j, 1
	jmp @loop
@fin
	%fo =l mul %i, 16
	%fp =l add %a, %fo
	%fp8 =l add %fp, 8
	%fk =l loadl %fp
	%fv =l loadl %fp8
	storel %fk, %lp
	storel %fv, %lp8
	storel %mk, %fp
	storel %mv, %fp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%rp =l add %fp, 16
	%rn0 =l sub %n, %i
	%rn =l sub %rn0, 1
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	jmp @done
@done
	ret 0
}
