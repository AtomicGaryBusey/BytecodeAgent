packbits_decode:
    xor eax, eax
    xor r8d, r8d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx r9, byte ptr [rdi + r8]
    inc r8
    test r9, r9
    js .Lneg
    inc r9
    mov r10, rsi
    sub r10, r8
    cmp r9, r10
    jg .Lerr2
    mov r10, rcx
    sub r10, rax
    cmp r9, r10
    jg .Llit_over
.Llit_copy:
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + rax], r11b
    inc r8
    inc rax
    dec r9
    jnz .Llit_copy
    jmp .Lloop
.Llit_over:
    test r10, r10
    jz .Lerr1
.Llo:
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + rax], r11b
    inc r8
    inc rax
    dec r10
    jnz .Llo
.Lerr1:
    mov rax, -1
    ret
.Lneg:
    cmp r9, -128
    je .Lloop
    cmp r8, rsi
    jge .Lerr2
    movzx r11d, byte ptr [rdi + r8]
    inc r8
    neg r9
    inc r9
    mov r10, rcx
    sub r10, rax
    cmp r9, r10
    jg .Lrep_over
.Lrf:
    mov byte ptr [rdx + rax], r11b
    inc rax
    dec r9
    jnz .Lrf
    jmp .Lloop
.Lrep_over:
    test r10, r10
    jz .Lerr1
.Lro:
    mov byte ptr [rdx + rax], r11b
    inc rax
    dec r10
    jnz .Lro
    jmp .Lerr1
.Lerr2:
    mov rax, -2
    ret
.Ldone:
    ret
