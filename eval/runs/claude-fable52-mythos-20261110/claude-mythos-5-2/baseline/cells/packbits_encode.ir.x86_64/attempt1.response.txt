export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@top
	%c =w csgel %i, %n
	jnz %c, @done, @scaninit
@scaninit
	%j =l copy %i
@scan
	%c =w csgel %j, %n
	jnz %c, @scanend, @scan2
@scan2
	%d =l sub %j, %i
	%c =w csgel %d, 128
	jnz %c, @scanend, @runinit
@runinit
	%r =l copy 1
	%p =l add %src, %j
	%b =w loadub %p
@runloop
	%k =l add %j, %r
	%c =w csgel %k, %n
	jnz %c, @runend, @run2
@run2
	%c =w csgel %r, 128
	jnz %c, @runend, @run3
@run3
	%q =l add %src, %k
	%b2 =w loadub %q
	%c =w ceqw %b2, %b
	jnz %c, @runinc, @runend
@runinc
	%r =l add %r, 1
	jmp @runloop
@runend
	%c =w csgel %r, 3
	jnz %c, @scanend, @scaninc
@scaninc
	%j =l add %j, 1
	jmp @scan
@scanend
	%c =w ceql %j, %i
	jnz %c, @repeat, @literal
@repeat
	%t =l add %o, 2
	%c =w csgtl %t, %cap
	jnz %c, @fail, @rep2
@rep2
	%h =l sub 257, %r
	%hw =w copy %h
	%po =l add %out, %o
	storeb %hw, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o =l copy %t
	%i =l add %i, %r
	jmp @top
@literal
	%len =l sub %j, %i
	%t =l add %o, 1
	%t =l add %t, %len
	%c =w csgtl %t, %cap
	jnz %c, @fail, @lit2
@lit2
	%h =l sub %len, 1
	%hw =w copy %h
	%po =l add %out, %o
	storeb %hw, %po
	%o =l add %o, 1
@copy
	%c =w csgel %i, %j
	jnz %c, @top, @copy2
@copy2
	%sp =l add %src, %i
	%v =w loadub %sp
	%dp =l add %out, %o
	storeb %v, %dp
	%i =l add %i, 1
	%o =l add %o, 1
	jmp @copy
@fail
	ret -1
@done
	ret %o
}
