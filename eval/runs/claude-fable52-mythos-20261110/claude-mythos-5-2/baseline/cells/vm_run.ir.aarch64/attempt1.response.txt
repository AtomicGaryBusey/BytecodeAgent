export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
@loop
	%c =w ceql %pc, %len
	jnz %c, @done, @fetch
@fetch
	%o =l mul %pc, 8
	%p =l add %code, %o
	%op =l loadl %p
	%p1 =l add %p, 8
	%do =l mul %d, 8
	%tp =l add %stack, %do
	%t1 =l sub %tp, 8
	%t2 =l sub %tp, 16
	%c =w ceql %op, 0
	jnz %c, @done, @n1
@n1
	%c =w ceql %op, 1
	jnz %c, @push, @n5
@n5
	%c =w ceql %op, 5
	jnz %c, @dup, @n6
@n6
	%c =w ceql %op, 6
	jnz %c, @swap, @n7
@n7
	%c =w ceql %op, 7
	jnz %c, @drop, @n8
@n8
	%c =w ceql %op, 8
	jnz %c, @over, @n10
@n10
	%c =w ceql %op, 10
	jnz %c, @jz, @n11
@n11
	%c =w ceql %op, 11
	jnz %c, @jmp, @bin
@push
	%v =l loadl %p1
	storel %v, %tp
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@dup
	%v =l loadl %t1
	storel %v, %tp
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%b =l loadl %t1
	%a =l loadl %t2
	storel %b, %t2
	storel %a, %t1
	%pc =l add %pc, 1
	jmp @loop
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%v =l loadl %t2
	storel %v, %tp
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%d =l sub %d, 1
	%v =l loadl %t1
	%c =w ceql %v, 0
	jnz %c, @jmp, @jznt
@jznt
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%pc =l loadl %p1
	jmp @loop
@bin
	%b =l loadl %t1
	%a =l loadl %t2
	%d =l sub %d, 1
	%pc =l add %pc, 1
	%c =w ceql %op, 2
	jnz %c, @add, @b3
@b3
	%c =w ceql %op, 3
	jnz %c, @sub, @b4
@b4
	%c =w ceql %op, 4
	jnz %c, @mul, @lt
@add
	%r =l add %a, %b
	jmp @binst
@sub
	%r =l sub %a, %b
	jmp @binst
@mul
	%r =l mul %a, %b
	jmp @binst
@lt
	%rw =w csltl %a, %b
	%r =l extuw %rw
@binst
	storel %r, %t2
	jmp @loop
@done
	%i =l copy %d
@zloop
	%c =w csltl %i, %cap
	jnz %c, @zbody, @zend
@zbody
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i =l add %i, 1
	jmp @zloop
@zend
	%c =w ceql %d, 0
	jnz %c, @empty, @top
@empty
	ret 0
@top
	%fo =l mul %d, 8
	%fp =l add %stack, %fo
	%fp =l sub %fp, 8
	%res =l loadl %fp
	ret %res
}
