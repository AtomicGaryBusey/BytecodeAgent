isqrt:
    mov x1, x0                      // x1 = remaining n
    mov x0, #0                      // x0 = result
    mov x2, #0x4000000000000000     // x2 = bit (highest power of 4)
.Lshrink:
    cmp x2, x1
    b.ls .Lloop
    lsr x2, x2, #2
    cbnz x2, .Lshrink
    ret
.Lloop:
    cbz x2, .Ldone
    add x3, x0, x2                  // x3 = res + bit
    lsr x0, x0, #1                  // res >>= 1
    cmp x1, x3
    b.lo .Lskip
    sub x1, x1, x3                  // n -= res + bit
    add x0, x0, x2                  // res += bit
.Lskip:
    lsr x2, x2, #2
    b .Lloop
.Ldone:
    ret
