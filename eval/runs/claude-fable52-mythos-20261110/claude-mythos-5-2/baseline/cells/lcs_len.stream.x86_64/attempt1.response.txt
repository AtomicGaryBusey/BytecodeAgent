lcs_len:
    # rdi=x, rsi=m, rdx=y, rcx=n, r8=row
    xor eax, eax
.Lzero:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, rcx
    jle .Lzero
    xor r9, r9                      # i
.Louter:
    cmp r9, rsi
    jge .Ldone
    movzx r10d, byte ptr [rdi + r9]  # x[i]
    xor r11d, r11d                  # diag
    mov eax, 1                      # j
.Linner:
    cmp rax, rcx
    jg .Lnext
    push rbx
    mov rbx, qword ptr [r8 + rax*8]  # up
    cmp r10b, byte ptr [rdx + rax - 1]
    jne .Lneq
    inc r11
    jmp .Lstore
.Lneq:
    mov r11, qword ptr [r8 + rax*8 - 8]
    cmp r11, rbx
    cmovl r11, rbx
.Lstore:
    mov qword ptr [r8 + rax*8], r11
    mov r11, rbx
    pop rbx
    inc rax
    jmp .Linner
.Lnext:
    inc r9
    jmp .Louter
.Ldone:
    mov rax, qword ptr [r8 + rcx*8]
    ret
