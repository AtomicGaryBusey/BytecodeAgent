export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @merge %k1
	%max =l phi @start 0, @merge %max1
	%c =w cslel %k, %n
	jnz %c, @inner, @done
@inner
	%x =l phi @outer %k, @step %x1
	%s =l phi @outer 0, @step %s1
	%e =w ceql %x, 1
	jnz %e, @next, @body
@body
	%b =l and %x, 1
	%isodd =w cnel %b, 0
	jnz %isodd, @oddb, @evenb
@oddb
	%t =l mul %x, 3
	%xo =l add %t, 1
	jmp @step
@evenb
	%xe =l shr %x, 1
	jmp @step
@step
	%x1 =l phi @oddb %xo, @evenb %xe
	%s1 =l add %s, 1
	jmp @inner
@next
	%g =w csgtl %s, %max
	jnz %g, @upd, @keep
@upd
	jmp @merge
@keep
	jmp @merge
@merge
	%max1 =l phi @upd %s, @keep %max
	%k1 =l add %k, 1
	jmp @outer
@done
	ret %max
}
