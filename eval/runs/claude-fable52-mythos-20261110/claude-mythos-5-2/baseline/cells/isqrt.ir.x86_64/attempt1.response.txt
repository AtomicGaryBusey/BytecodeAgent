export function l $isqrt(l %n) {
@start
	%res =l copy 0
	%bit =l copy 4611686018427387904
@shrink
	%c =w cugtl %bit, %n
	jnz %c, @shrinkbody, @loop
@shrinkbody
	%bit =l shr %bit, 2
	jmp @shrink
@loop
	%nz =w cnel %bit, 0
	jnz %nz, @body, @done
@body
	%t =l add %res, %bit
	%res =l shr %res, 1
	%c2 =w cugel %n, %t
	jnz %c2, @take, @next
@take
	%n =l sub %n, %t
	%res =l add %res, %bit
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
