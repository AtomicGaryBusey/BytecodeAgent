export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @init
@init
	%last =l sub %n, 1
	%lo =l mul %last, 16
	%pp =l add %a, %lo
	%pk =l loadl %pp
	%pp8 =l add %pp, 8
	%pv =l loadl %pp8
	jmp @loop
@loop
	%i =l phi @init 0, @next %i2
	%j =l phi @init 0, @next %j2
	%c1 =w csltl %j, %last
	jnz %c1, @body, @fin
@body
	%jo =l mul %j, 16
	%jp =l add %a, %jo
	%kj =l loadl %jp
	%jp8 =l add %jp, 8
	%vj =l loadl %jp8
	%lt =w csltl %kj, %pk
	jnz %lt, @swap, @chkeq
@chkeq
	%eq =w ceql %kj, %pk
	jnz %eq, @chkv, @noswap
@chkv
	%lv =w csltl %vj, %pv
	jnz %lv, @swap, @noswap
@swap
	%io =l mul %i, 16
	%ip =l add %a, %io
	%ip8 =l add %ip, 8
	%ki =l loadl %ip
	%vi =l loadl %ip8
	storel %kj, %ip
	storel %vj, %ip8
	storel %ki, %jp
	storel %vi, %jp8
	%i1 =l add %i, 1
	jmp @next
@noswap
	jmp @next
@next
	%i2 =l phi @swap %i1, @noswap %i
	%j2 =l add %j, 1
	jmp @loop
@fin
	%fo =l mul %i, 16
	%fp =l add %a, %fo
	%fp8 =l add %fp, 8
	%fk =l loadl %fp
	%fv =l loadl %fp8
	storel %pk, %fp
	storel %pv, %fp8
	storel %fk, %pp
	storel %fv, %pp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%rp =l add %fp, 16
	%rn =l sub %last, %i
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	jmp @done
@done
	ret 0
}
