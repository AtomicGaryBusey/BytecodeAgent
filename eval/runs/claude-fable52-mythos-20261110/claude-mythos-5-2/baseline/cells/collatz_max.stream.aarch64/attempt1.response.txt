collatz_max:
    mov x9, #0              // max
    mov x10, #1             // k
.Louter:
    cmp x10, x0
    b.gt .Ldone
    mov x11, x10            // x
    mov x12, #0             // steps
.Linner:
    cmp x11, #1
    b.eq .Lend
    tbnz x11, #0, .Lodd
    lsr x11, x11, #1
    b .Lstep
.Lodd:
    add x11, x11, x11, lsl #1
    add x11, x11, #1
.Lstep:
    add x12, x12, #1
    b .Linner
.Lend:
    cmp x12, x9
    csel x9, x12, x9, gt
    add x10, x10, #1
    b .Louter
.Ldone:
    mov x0, x9
    ret
