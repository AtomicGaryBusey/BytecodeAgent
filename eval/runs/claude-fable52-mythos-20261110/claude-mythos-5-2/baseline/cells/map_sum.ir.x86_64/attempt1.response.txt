export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%s =l phi @start 0, @body %s2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%s2 =l add %s, %r
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %s
}
