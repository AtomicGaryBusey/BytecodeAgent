export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
@loop
	%i =l phi @start 1, @body 	%i2
	%min =l phi @start %y0, @body %min2
	%max =l phi @start %y0, @body %max2
	%sum =l phi @start %y0, @body %sum2
	%cnt =l phi @start %odd0, @body %cnt2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%ltl =l extuw %lt
	%m1 =l neg %ltl
	%d1 =l xor %y, %min
	%d1m =l and %d1, %m1
	%min2 =l xor %min, %d1m
	%gt =w csgtl %y, %max
	%gtl =l extuw %gt
	%m2 =l neg %gtl
	%d2 =l xor %y, %max
	%d2m =l and %d2, %m2
	%max2 =l xor %max, %d2m
	%sum2 =l add %sum, %y
	%o =l and %y, 1
	%cnt2 =l add %cnt, %o
	%i2 =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
