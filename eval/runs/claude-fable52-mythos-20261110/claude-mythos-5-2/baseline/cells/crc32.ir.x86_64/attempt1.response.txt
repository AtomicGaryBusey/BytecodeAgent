export function l $crc32(l %p, l %n) {
@start
	%c0 =w copy 4294967295
	%i0 =l copy 0
@loop
	%c =w phi @start %c0, @bitdone %c3
	%i =l phi @start %i0, @bitdone %i1
	%t =w csltl %i, %n
	jnz %t, @body, @end
@body
	%a =l add %p, %i
	%b =w loadub %a
	%cx =w xor %c, %b
	%k0 =w copy 0
@bit
	%c1 =w phi @body %cx, @bit %c3
	%k =w phi @body %k0, @bit %k1
	%lo =w and %c1, 1
	%m =w neg %lo
	%mask =w and %m, 3988292384
	%sh =w shr %c1, 1
	%c3 =w xor %sh, %mask
	%k1 =w add %k, 1
	%tk =w csltw %k1, 8
	jnz %tk, @bit, @bitdone
@bitdone
	%i1 =l add %i, 1
	jmp @loop
@end
	%f =w xor %c, 4294967295
	%r =l extuw %f
	ret %r
}
