vm_run:
  xor r8d, r8d                     # pc
  xor r9d, r9d                     # depth
.Lloop:
  cmp r8, rsi
  jae .Ldone
  mov rax, qword ptr [rdi + r8*8]
  test rax, rax
  jz .Ldone
  cmp rax, 1
  je .Lpush
  cmp rax, 2
  je .Ladd
  cmp rax, 3
  je .Lsub
  cmp rax, 4
  je .Lmul
  cmp rax, 5
  je .Ldup
  cmp rax, 6
  je .Lswap
  cmp rax, 7
  je .Ldrop
  cmp rax, 8
  je .Lover
  cmp rax, 9
  je .Llt
  cmp rax, 10
  je .Ljz
  cmp rax, 11
  je .Ljmp
.Lnext:
  inc r8
  jmp .Lloop
.Lpush:
  mov rax, qword ptr [rdi + r8*8 + 8]
  mov qword ptr [rdx + r9*8], rax
  inc r9
  add r8, 2
  jmp .Lloop
.Ladd:
  mov rax, qword ptr [rdx + r9*8 - 8]
  dec r9
  add qword ptr [rdx + r9*8 - 8], rax
  jmp .Lnext
.Lsub:
  mov rax, qword ptr [rdx + r9*8 - 8]
  dec r9
  sub qword ptr [rdx + r9*8 - 8], rax
  jmp .Lnext
.Lmul:
  mov rax, qword ptr [rdx + r9*8 - 8]
  dec r9
  imul rax, qword ptr [rdx + r9*8 - 8]
  mov qword ptr [rdx + r9*8 - 8], rax
  jmp .Lnext
.Ldup:
  mov rax, qword ptr [rdx + r9*8 - 8]
  mov qword ptr [rdx + r9*8], rax
  inc r9
  jmp .Lnext
.Lswap:
  mov rax, qword ptr [rdx + r9*8 - 8]
  mov r10, qword ptr [rdx + r9*8 - 16]
  mov qword ptr [rdx + r9*8 - 8], r10
  mov qword ptr [rdx + r9*8 - 16], rax
  jmp .Lnext
.Ldrop:
  dec r9
  jmp .Lnext
.Lover:
  mov rax, qword ptr [rdx + r9*8 - 16]
  mov qword ptr [rdx + r9*8], rax
  inc r9
  jmp .Lnext
.Llt:
  mov rax, qword ptr [rdx + r9*8 - 8]
  dec r9
  xor r10d, r10d
  cmp qword ptr [rdx + r9*8 - 8], rax
  setl r10b
  mov qword ptr [rdx + r9*8 - 8], r10
  jmp .Lnext
.Ljz:
  dec r9
  mov rax, qword ptr [rdx + r9*8]
  test rax, rax
  jnz .Lskip2
  mov r8, qword ptr [rdi + r8*8 + 8]
  jmp .Lloop
.Lskip2:
  add r8, 2
  jmp .Lloop
.Ljmp:
  mov r8, qword ptr [rdi + r8*8 + 8]
  jmp .Lloop
.Ldone:
  xor eax, eax
  test r9, r9
  jz .Lzero
  mov rax, qword ptr [rdx + r9*8 - 8]
.Lzero:
  mov r10, r9
.Lzloop:
  cmp r10, rcx
  jae .Lret
  mov qword ptr [rdx + r10*8], 0
  inc r10
  jmp .Lzloop
.Lret:
  ret
