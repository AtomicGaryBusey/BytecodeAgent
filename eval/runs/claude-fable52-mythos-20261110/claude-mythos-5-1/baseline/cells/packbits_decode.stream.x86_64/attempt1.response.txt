packbits_decode:
    xor eax, eax
    xor r8d, r8d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx r9, byte ptr [rdi + r8]
    inc r8
    cmp r9, -128
    je .Lloop
    test r9, r9
    js .Lrepeat
    inc r9
    lea r10, [r8 + r9]
    cmp r10, rsi
    jg .Lerr2
.Lcopy:
    cmp rax, rcx
    jge .Lerr1
    movzx r11d, byte ptr [rdi + r8]
    mov byte ptr [rdx + rax], r11b
    inc r8
    inc rax
    dec r9
    jnz .Lcopy
    jmp .Lloop
.Lrepeat:
    cmp r8, rsi
    jge .Lerr2
    mov r10d, 1
    sub r10, r9
    movzx r11d, byte ptr [rdi + r8]
    inc r8
.Lrep:
    cmp rax, rcx
    jge .Lerr1
    mov byte ptr [rdx + rax], r11b
    inc rax
    dec r10
    jnz .Lrep
    jmp .Lloop
.Lerr1:
    mov rax, -1
    ret
.Lerr2:
    mov rax, -2
    ret
.Ldone:
    ret
