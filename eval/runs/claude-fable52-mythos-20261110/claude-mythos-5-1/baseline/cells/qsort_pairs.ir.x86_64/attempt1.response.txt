export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csgtl %n, 1
	jnz %c, @body, @ret
@body
	%mid =l shr %n, 1
	%midoff =l shl %mid, 4
	%pm =l add %a, %midoff
	%nm1 =l sub %n, 1
	%lastoff =l shl %nm1, 4
	%pl =l add %a, %lastoff
	%pm1 =l add %pm, 8
	%pl1 =l add %pl, 8
	%mk =l loadl %pm
	%mv =l loadl %pm1
	%lk =l loadl %pl
	%lv =l loadl %pl1
	storel %lk, %pm
	storel %lv, %pm1
	storel %mk, %pl
	storel %mv, %pl1
	%pk =l copy %mk
	%pv =l copy %mv
	%i =l copy 0
	%j =l copy 0
@loop
	%cj =w csltl %j, %nm1
	jnz %cj, @cmp, @done
@cmp
	%joff =l shl %j, 4
	%pj =l add %a, %joff
	%pj1 =l add %pj, 8
	%jk =l loadl %pj
	%jv =l loadl %pj1
	%lt =w csltl %jk, %pk
	%eq =w ceql %jk, %pk
	%vlt =w csltl %jv, %pv
	%both =w and %eq, %vlt
	%less =w or %lt, %both
	jnz %less, @swap, @next
@swap
	%ioff =l shl %i, 4
	%pi =l add %a, %ioff
	%pi1 =l add %pi, 8
	%ik =l loadl %pi
	%iv =l loadl %pi1
	storel %jk, %pi
	storel %jv, %pi1
	storel %ik, %pj
	storel %iv, %pj1
	%i =l add %i, 1
@next
	%j =l add %j, 1
	jmp @loop
@done
	%ioff2 =l shl %i, 4
	%pi2 =l add %a, %ioff2
	%pi21 =l add %pi2, 8
	%ik2 =l loadl %pi2
	%iv2 =l loadl %pi21
	storel %pk, %pi2
	storel %pv, %pi21
	storel %ik2, %pl
	storel %iv2, %pl1
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%ip1off =l shl %ip1, 4
	%a2 =l add %a, %ip1off
	%n2 =l sub %n, %ip1
	%r2 =l call $qsort_pairs(l %a2, l %n2)
@ret
	ret 0
}
