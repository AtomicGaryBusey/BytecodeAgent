export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
    %i0 =l copy 0
    %o0 =l copy 0
    jmp @outer
@outer
    %i =l phi @start %i0, @rep_w %i_rep, @lit_fin %j
    %o =l phi @start %o0, @rep_w %o2, @lit_fin %oend
    %c0 =w csltl %i, %n
    jnz %c0, @run_i, @ret
@run_i
    %pi =l add %src, %i
    %bi =w loadub %pi
    %lim0 =l add %i, 128
    %c1 =w csltl %lim0, %n
    jnz %c1, @lim_a, @lim_b
@lim_a
    jmp @runi_hdr
@lim_b
    jmp @runi_hdr
@runi_hdr
    %limi =l phi @lim_a %lim0, @lim_b %n
    jmp @runi_loop
@runi_loop
    %k =l phi @runi_hdr %i, @runi_next %k1
    %c2 =w csltl %k, %limi
    jnz %c2, @runi_chk, @runi_end
@runi_chk
    %pk =l add %src, %k
    %bk =w loadub %pk
    %c3 =w ceqw %bk, %bi
    jnz %c3, @runi_next, @runi_end
@runi_next
    %k1 =l add %k, 1
    jmp @runi_loop
@runi_end
    %r =l sub %k, %i
    %c4 =w csgel %r, 3
    jnz %c4, @rep, @lit
@rep
    %o2 =l add %o, 2
    %c5 =w csgtl %o2, %cap
    jnz %c5, @fail, @rep_w
@rep_w
    %rw =w copy %r
    %h =w sub 257, %rw
    %po =l add %out, %o
    storeb %h, %po
    %po1 =l add %po, 1
    storeb %bi, %po1
    %i_rep =l add %i, %r
    jmp @outer
@lit
    %j0 =l add %i, 1
    jmp @lit_loop
@lit_loop
    %j =l phi @lit %j0, @lit_inc %j1
    %c6 =w csltl %j, %n
    jnz %c6, @lit_c2, @lit_done
@lit_c2
    %d =l sub %j, %i
    %c7 =w csltl %d, 128
    jnz %c7, @runj_setup, @lit_done
@runj_setup
    %pj =l add %src, %j
    %bj =w loadub %pj
    %limj0 =l add %j, 3
    %c8 =w csltl %limj0, %n
    jnz %c8, @limj_a, @limj_b
@limj_a
    jmp @runj_hdr
@limj_b
    jmp @runj_hdr
@runj_hdr
    %limj =l phi @limj_a %limj0, @limj_b %n
    jmp @runj_loop
@runj_loop
    %m =l phi @runj_hdr %j, @runj_next %m1
    %c9 =w csltl %m, %limj
    jnz %c9, @runj_chk, @runj_end
@runj_chk
    %pm =l add %src, %m
    %bm =w loadub %pm
    %c10 =w ceqw %bm, %bj
    jnz %c10, @runj_next, @runj_end
@runj_next
    %m1 =l add %m, 1
    jmp @runj_loop
@runj_end
    %rj =l sub %m, %j
    %c11 =w csgel %rj, 3
    jnz %c11, @lit_done, @lit_inc
@lit_inc
    %j1 =l add %j, 1
    jmp @lit_loop
@lit_done
    %L =l sub %j, %i
    %o1 =l add %o, 1
    %oend =l add %o1, %L
    %c12 =w csgtl %oend, %cap
    jnz %c12, @fail, @lit_w
@lit_w
    %Lw =w copy %L
    %Lm1 =w sub %Lw, 1
    %poh =l add %out, %o
    storeb %Lm1, %poh
    %pd0 =l add %out, %o1
    jmp @copy_loop
@copy_loop
    %t =l phi @lit_w 0, @copy_next %t1
    %c13 =w csltl %t, %L
    jnz %c13, @copy_next, @lit_fin
@copy_next
    %ps =l add %pi, %t
    %bs =w loadub %ps
    %pd =l add %pd0, %t
    storeb %bs, %pd
    %t1 =l add %t, 1
    jmp @copy_loop
@lit_fin
    jmp @outer
@fail
    ret -1
@ret
    ret %o
}
