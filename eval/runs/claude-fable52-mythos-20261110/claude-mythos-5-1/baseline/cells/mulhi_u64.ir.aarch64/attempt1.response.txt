export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p00 =l mul %alo, %blo
	%p01 =l mul %alo, %bhi
	%p10 =l mul %ahi, %blo
	%p11 =l mul %ahi, %bhi
	%p00h =l shr %p00, 32
	%p01l =l and %p01, 4294967295
	%p01h =l shr %p01, 32
	%p10l =l and %p10, 4294967295
	%p10h =l shr %p10, 32
	%mid0 =l add %p00h, %p01l
	%mid =l add %mid0, %p10l
	%midh =l shr %mid, 32
	%h0 =l add %p11, %p01h
	%h1 =l add %h0, %p10h
	%h2 =l add %h1, %midh
	ret %h2
}
