mat_pow:
    push rbx
    push r12
    push r13
    push rbp
    mov r10, rsi
    imul r10, rsi            # nn = n*n
    xor eax, eax
.Lid0:
    mov qword ptr [rcx + rax*8], 0
    inc rax
    cmp rax, r10
    jb .Lid0
    xor eax, eax
    lea r11, [rsi + 1]
.Lid1:
    mov qword ptr [rcx + rax*8], 1
    add rax, r11
    cmp rax, r10
    jb .Lid1
    test rdx, rdx
    jz .Lfinish
.Lpow:
    xor r9d, r9d             # i
.Li:
    xor r11d, r11d           # j
.Lj:
    xor ebx, ebx             # acc
    mov r12, rsi             # t counter
    mov r13, r9
    imul r13, rsi            # i*n + t
    mov rbp, r11             # t*n + j
.Lt:
    mov rax, qword ptr [rcx + r13*8]
    imul rax, qword ptr [rdi + rbp*8]
    add rbx, rax
    inc r13
    add rbp, rsi
    dec r12
    jnz .Lt
    mov rax, r13
    sub rax, rsi
    add rax, r11
    mov qword ptr [r8 + rax*8], rbx
    inc r11
    cmp r11, rsi
    jb .Lj
    inc r9
    cmp r9, rsi
    jb .Li
    xor eax, eax
.Lcp:
    mov rbx, qword ptr [r8 + rax*8]
    mov qword ptr [rcx + rax*8], rbx
    inc rax
    cmp rax, r10
    jb .Lcp
    dec rdx
    jnz .Lpow
.Lfinish:
    lea r11, [r10 + r10]
    xor eax, eax
.Lz:
    mov qword ptr [r8 + rax*8], 0
    inc rax
    cmp rax, r11
    jb .Lz
    xor eax, eax
    pop rbp
    pop r13
    pop r12
    pop rbx
    ret
