lcs_len:
  push rbx
  push r12
  xor r10d, r10d
.Lzero:
  mov qword ptr [r8 + r10*8], 0
  inc r10
  cmp r10, rcx
  jle .Lzero
  xor r10d, r10d
  test rsi, rsi
  jle .Ldone
.Louter:
  xor r11d, r11d
  xor ebx, ebx
  mov r9d, 1
  cmp r9, rcx
  jg .Lnexti
.Linner:
  mov r12, qword ptr [r8 + r9*8]
  movzx eax, byte ptr [rdx + r9 - 1]
  cmp al, byte ptr [rdi + r10]
  jne .Lnomatch
  lea rax, [r11 + 1]
  jmp .Lstore
.Lnomatch:
  mov rax, r12
  cmp rax, rbx
  cmovl rax, rbx
.Lstore:
  mov qword ptr [r8 + r9*8], rax
  mov rbx, rax
  mov r11, r12
  inc r9
  cmp r9, rcx
  jle .Linner
.Lnexti:
  inc r10
  cmp r10, rsi
  jl .Louter
.Ldone:
  mov rax, qword ptr [r8 + rcx*8]
  pop r12
  pop rbx
  ret
