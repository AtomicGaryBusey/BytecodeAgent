max_byte:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    cmp rsi, 16
    jb .Ltail
    pxor xmm0, xmm0
    mov rcx, rsi
    and rcx, -16
    xor edx, edx
.Lvec:
    pmaxub xmm0, xmmword ptr [rdi + rdx]
    add rdx, 16
    cmp rdx, rcx
    jb .Lvec
    pshufd xmm1, xmm0, 0x4e
    pmaxub xmm0, xmm1
    pshufd xmm1, xmm0, 0xb1
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrld xmm1, 16
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrlw xmm1, 8
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
    add rdi, rcx
    sub rsi, rcx
    jz .Ldone
.Ltail:
    movzx ecx, byte ptr [rdi]
    cmp ecx, eax
    cmova eax, ecx
    inc rdi
    dec rsi
    jnz .Ltail
.Ldone:
    ret
