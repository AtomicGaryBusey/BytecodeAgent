export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@outer
	%c1 =w csltl %i, %n
	jnz %c1, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc1 =w xor %crc, %b
	%k =w copy 0
	%cc =w copy %crc1
@inner
	%lsb =w and %cc, 1
	%sh =w shr %cc, 1
	jnz %lsb, @odd, @even
@odd
	%cc2 =w xor %sh, 3988292384
	jmp @next
@even
	%cc2 =w copy %sh
@next
	%cc =w copy %cc2
	%k =w add %k, 1
	%c2 =w cultw %k, 8
	jnz %c2, @inner, @endinner
@endinner
	%crc =w copy %cc
	%i =l add %i, 1
	jmp @outer
@done
	%r =w xor %crc, 4294967295
	%res =l extuw %r
	ret %res
}
