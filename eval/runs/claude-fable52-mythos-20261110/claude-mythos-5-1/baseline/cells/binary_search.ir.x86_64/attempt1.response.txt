export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lt %lo2, @gt %lo
	%hi =l phi @start %n, @lt %hi, @gt %mid
	%c =w csltl %lo, %hi
	jnz %c, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%ltc =w csltl %v, %key
	jnz %ltc, @lt, @gt
@lt
	%lo2 =l add %mid, 1
	jmp @loop
@gt
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
