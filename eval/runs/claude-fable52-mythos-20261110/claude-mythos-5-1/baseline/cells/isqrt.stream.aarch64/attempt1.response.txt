isqrt:
    mov x1, #0
    mov x2, #1
    lsl x2, x2, #62
.Lshrink:
    cmp x2, x0
    b.ls .Lloop
    lsr x2, x2, #2
    cbnz x2, .Lshrink
.Lloop:
    cbz x2, .Ldone
    add x3, x1, x2
    lsr x1, x1, #1
    cmp x0, x3
    b.lo .Lskip
    sub x0, x0, x3
    add x1, x1, x2
.Lskip:
    lsr x2, x2, #2
    b .Lloop
.Ldone:
    mov x0, x1
    ret
