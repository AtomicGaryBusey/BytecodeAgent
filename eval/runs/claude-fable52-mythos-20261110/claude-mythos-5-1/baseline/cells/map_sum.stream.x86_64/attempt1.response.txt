map_sum:
	push rbx
	push r12
	push r13
	push r14
	push r15
	mov r12, rdi
	mov r13, rsi
	mov r14, rdx
	xor ebx, ebx
	xor r15d, r15d
	test r13, r13
	jle .Ldone
.Lloop:
	mov rdi, qword ptr [r12 + r15*8]
	call r14
	add rbx, rax
	inc r15
	cmp r15, r13
	jl .Lloop
.Ldone:
	mov rax, rbx
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbx
	ret
