export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%i0 =l copy 0
@id_loop
	%c0 =w csltl %i0, %nn
	jnz %c0, @id_body, @id_done
@id_body
	%r0 =l div %i0, %n
	%q0 =l rem %i0, %n
	%e0 =w ceql %r0, %q0
	%v0 =l extuw %e0
	%o0 =l mul %i0, 8
	%p0 =l add %out, %o0
	storel %v0, %p0
	%i0 =l add %i0, 1
	jmp @id_loop
@id_done
	%kk =l copy %k
@pow_loop
	%ck =w csgtl %kk, 0
	jnz %ck, @mul_start, @finish
@mul_start
	%i =l copy 0
@mul_i
	%ci =w csltl %i, %n
	jnz %ci, @mul_j_init, @copy_init
@mul_j_init
	%j =l copy 0
@mul_j
	%cj =w csltl %j, %n
	jnz %cj, @mul_t_init, @mul_i_next
@mul_t_init
	%t =l copy 0
	%acc =l copy 0
@mul_t
	%ct =w csltl %t, %n
	jnz %ct, @mul_t_body, @mul_t_done
@mul_t_body
	%xi =l mul %i, %n
	%xit =l add %xi, %t
	%xoff =l mul %xit, 8
	%xp =l add %out, %xoff
	%x =l loadl %xp
	%yt =l mul %t, %n
	%ytj =l add %yt, %j
	%yoff =l mul %ytj, 8
	%yp =l add %a, %yoff
	%y =l loadl %yp
	%prod =l mul %x, %y
	%acc =l add %acc, %prod
	%t =l add %t, 1
	jmp @mul_t
@mul_t_done
	%zi =l mul %i, %n
	%zij =l add %zi, %j
	%zoff =l mul %zij, 8
	%zp =l add %tmp, %zoff
	storel %acc, %zp
	%j =l add %j, 1
	jmp @mul_j
@mul_i_next
	%i =l add %i, 1
	jmp @mul_i
@copy_init
	%m =l copy 0
@copy_loop
	%cm =w csltl %m, %nn
	jnz %cm, @copy_body, @copy_done
@copy_body
	%moff =l mul %m, 8
	%sp =l add %tmp, %moff
	%sv =l loadl %sp
	%dp =l add %out, %moff
	storel %sv, %dp
	%m =l add %m, 1
	jmp @copy_loop
@copy_done
	%kk =l sub %kk, 1
	jmp @pow_loop
@finish
	%nn2 =l mul %nn, 2
	%z =l copy 0
@zero_loop
	%cz =w csltl %z, %nn2
	jnz %cz, @zero_body, @done
@zero_body
	%zo =l mul %z, 8
	%zq =l add %tmp, %zo
	storel 0, %zq
	%z =l add %z, 1
	jmp @zero_loop
@done
	ret 0
}
