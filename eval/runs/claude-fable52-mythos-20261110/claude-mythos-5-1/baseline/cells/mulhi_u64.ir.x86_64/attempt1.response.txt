export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%alo =l and %a, %mask
	%ahi =l shr %a, 32
	%blo =l and %b, %mask
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0hi =l shr %p0, 32
	%p1lo =l and %p1, %mask
	%p1hi =l shr %p1, 32
	%p2lo =l and %p2, %mask
	%p2hi =l shr %p2, 32
	%mid0 =l add %p0hi, %p1lo
	%mid =l add %mid0, %p2lo
	%midhi =l shr %mid, 32
	%h0 =l add %p3, %p1hi
	%h1 =l add %h0, %p2hi
	%h =l add %h1, %midhi
	ret %h
}
