export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i3
	%o =l phi @start 0, @next %o3
	%c =w csltl %i, %n
	jnz %c, @group, @done
@group
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%i1 =l add %i, 1
	%c1 =w csltl %i1, %n
	jnz %c1, @hasb1, @emit
@hasb1
	%p1 =l add %src, %i1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%v1 =l or %v0, %s1
	%i2 =l add %i, 2
	%c2 =w csltl %i2, %n
	jnz %c2, @hasb2, @emit
@hasb2
	%p2 =l add %src, %i2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @emit
@emit
	%val =l phi @group %v0, @hasb1 %v1, @hasb2 %v2
	%em =l phi @group 2, @hasb1 3, @hasb2 4
	jmp @eloop
@eloop
	%k =l phi @emit 0, @ebody %k1
	%o1 =l phi @emit %o, @ebody %o2
	%sh =l phi @emit 18, @ebody %sh1
	%ck =w csltl %k, %em
	jnz %ck, @ebody, @pad
@ebody
	%shv =l shr %val, %sh
	%v =l and %shv, 63
	%ge26 =w csgel %v, 26
	%ge52 =w csgel %v, 52
	%eq62 =w ceql %v, 62
	%eq63 =w ceql %v, 63
	%e1 =l extuw %ge26
	%e2 =l extuw %ge52
	%e3 =l extuw %eq62
	%e4 =l extuw %eq63
	%t1 =l mul %e1, 6
	%t2 =l mul %e2, 75
	%t3 =l mul %e3, 15
	%t4 =l mul %e4, 12
	%ch0 =l add %v, 65
	%ch1 =l add %ch0, %t1
	%ch2 =l sub %ch1, %t2
	%ch3 =l sub %ch2, %t3
	%ch =l sub %ch3, %t4
	%op =l add %out, %o1
	storeb %ch, %op
	%o2 =l add %o1, 1
	%k1 =l add %k, 1
	%sh1 =l sub %sh, 6
	jmp @eloop
@pad
	%k2 =l phi @eloop %k, @pbody %k3
	%o4 =l phi @eloop %o1, @pbody %o5
	%cp =w csltl %k2, 4
	jnz %cp, @pbody, @next
@pbody
	%opp =l add %out, %o4
	storeb 61, %opp
	%o5 =l add %o4, 1
	%k3 =l add %k2, 1
	jmp @pad
@next
	%i3 =l add %i, 3
	%o3 =l copy %o4
	jmp @loop
@done
	ret %o
}
