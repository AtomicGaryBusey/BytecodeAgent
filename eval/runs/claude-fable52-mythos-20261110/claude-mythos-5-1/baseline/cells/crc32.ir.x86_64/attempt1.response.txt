export function l $crc32(l %p, l %n) {
@start
	%crc0 =w copy 4294967295
	%i0 =l copy 0
@outer
	%crc =w phi @start %crc0, @bitdone %crc2
	%i =l phi @start %i0, @bitdone %i1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crcx =w xor %crc, %b
	%k0 =w copy 0
@bit
	%crcb =w phi @body %crcx, @bitnext %crcn
	%k =w phi @body %k0, @bitnext %k1
	%lsb =w and %crcb, 1
	%sh =w shr %crcb, 1
	jnz %lsb, @odd, @even
@odd
	%crco =w xor %sh, 3988292384
	jmp @bitnext
@even
	jmp @bitnext
@bitnext
	%crcn =w phi @odd %crco, @even %sh
	%k1 =w add %k, 1
	%kc =w cultw %k1, 8
	jnz %kc, @bit, @bitdone
@bitdone
	%crc2 =w copy %crcn
	%i1 =l add %i, 1
	jmp @outer
@end
	%r =w xor %crc, 4294967295
	%rl =l extuw %r
	ret %rl
}
