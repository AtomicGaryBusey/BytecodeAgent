export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%ip =l alloc8 8
	%op =l alloc8 8
	%cp =l alloc8 8
	storel 0, %ip
	storel 0, %op
	storel 0, %cp
	jmp @loop
@loop
	%i =l loadl %ip
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%p =l add %src, %i
	%h =w loadsb %p
	%i1 =l add %i, 1
	storel %i1, %ip
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @notnop
@notnop
	%neg =w csltw %h, 0
	jnz %neg, @rep, @lit
@lit
	%len0 =w add %h, 1
	%len =l extsw %len0
	%end =l add %i1, %len
	%ok =w cslel %end, %n
	jnz %ok, @litloop, @err2
@litloop
	%li =l loadl %ip
	%lc =w csltl %li, %end
	jnz %lc, @litcopy, @loop
@litcopy
	%lo =l loadl %op
	%lfull =w csgel %lo, %cap
	jnz %lfull, @err1, @litwrite
@litwrite
	%lsp =l add %src, %li
	%lb =w loadub %lsp
	%ldp =l add %out, %lo
	storeb %lb, %ldp
	%lo1 =l add %lo, 1
	storel %lo1, %op
	%li1 =l add %li, 1
	storel %li1, %ip
	jmp @litloop
@rep
	%rok =w csltl %i1, %n
	jnz %rok, @repread, @err2
@repread
	%rsp =l add %src, %i1
	%rb =w loadub %rsp
	%i2 =l add %i1, 1
	storel %i2, %ip
	%cnt0 =w sub 1, %h
	%cnt =l extsw %cnt0
	storel %cnt, %cp
	jmp @reploop
@reploop
	%rc =l loadl %cp
	%rnz =w csgtl %rc, 0
	jnz %rnz, @repwrite, @loop
@repwrite
	%ro =l loadl %op
	%rfull =w csgel %ro, %cap
	jnz %rfull, @err1, @repstore
@repstore
	%rdp =l add %out, %ro
	storeb %rb, %rdp
	%ro1 =l add %ro, 1
	storel %ro1, %op
	%rc1 =l sub %rc, 1
	storel %rc1, %cp
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
@done
	%o =l loadl %op
	ret %o
}
