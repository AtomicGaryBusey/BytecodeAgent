export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pk =l alloc8 8
	%pc =l alloc8 8
	%n =l add %na, %nb
	storel 0, %pi
	jmp @zcond
@zcond
	%zi =l loadl %pi
	%zc =w csltl %zi, %n
	jnz %zc, @zbody, @ocond_init
@zbody
	%zo =l mul %zi, 8
	%zp =l add %r, %zo
	storel 0, %zp
	%zi1 =l add %zi, 1
	storel %zi1, %pi
	jmp @zcond
@ocond_init
	storel 0, %pi
	jmp @ocond
@ocond
	%i =l loadl %pi
	%oc =w csltl %i, %na
	jnz %oc, @obody, @count_init
@obody
	storel 0, %pj
	storel 0, %pc
	%ao =l mul %i, 8
	%ap =l add %a, %ao
	%ai =l loadl %ap
	%al =l and %ai, 4294967295
	%ah =l shr %ai, 32
	jmp @icond
@icond
	%j =l loadl %pj
	%ic =w csltl %j, %nb
	jnz %ic, @ibody, @iend
@ibody
	%bo =l mul %j, 8
	%bp =l add %b, %bo
	%bj =l loadl %bp
	%bl =l and %bj, 4294967295
	%bh =l shr %bj, 32
	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%llh =l shr %ll, 32
	%mid =l add %lh, %llh
	%mid2 =l add %mid, %hl
	%mc =w cultl %mid2, %mid
	%mcl =l extuw %mc
	%lo1 =l shl %mid2, 32
	%lll =l and %ll, 4294967295
	%lo =l or %lo1, %lll
	%midh =l shr %mid2, 32
	%hi1 =l add %hh, %midh
	%mcs =l shl %mcl, 32
	%hi =l add %hi1, %mcs
	%ij =l add %i, %j
	%ro =l mul %ij, 8
	%rp =l add %r, %ro
	%rv =l loadl %rp
	%t =l add %rv, %lo
	%c1 =w cultl %t, %rv
	%carry =l loadl %pc
	%t2 =l add %t, %carry
	%c2 =w cultl %t2, %t
	storel %t2, %rp
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%nc1 =l add %hi, %c1l
	%nc =l add %nc1, %c2l
	storel %nc, %pc
	%j1 =l add %j, 1
	storel %j1, %pj
	jmp @icond
@iend
	%fc =l loadl %pc
	%inb =l add %i, %nb
	%fo =l mul %inb, 8
	%fp =l add %r, %fo
	storel %fc, %fp
	%i1 =l add %i, 1
	storel %i1, %pi
	jmp @ocond
@count_init
	storel %n, %pk
	jmp @kcond
@kcond
	%k =l loadl %pk
	%kc =w csgtl %k, 0
	jnz %kc, @kchk, @done
@kchk
	%k1 =l sub %k, 1
	%ko =l mul %k1, 8
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%kz =w ceql %kv, 0
	jnz %kz, @kdec, @done
@kdec
	storel %k1, %pk
	jmp @kcond
@done
	%res =l loadl %pk
	ret %res
}
