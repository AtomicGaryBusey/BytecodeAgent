export function l $isqrt(l %n0) {
@start
	jmp @l1
@l1
	%bit1 =l phi @start 4611686018427387904, @l1b %bit2
	%c1 =w cugtl %bit1, %n0
	jnz %c1, @l1b, @l2
@l1b
	%bit2 =l shr %bit1, 2
	jmp @l1
@l2
	%bit3 =l phi @l1 %bit1, @l2e %bit5
	%n3 =l phi @l1 %n0, @l2e %n5
	%res3 =l phi @l1 0, @l2e %res5
	%c2 =w cnel %bit3, 0
	jnz %c2, @l2a, @end
@l2a
	%s =l add %res3, %bit3
	%c3 =w cugel %n3, %s
	jnz %c3, @l2b, @l2c
@l2b
	%n4 =l sub %n3, %s
	%h =l shr %res3, 1
	%res4 =l add %h, %bit3
	jmp @l2e
@l2c
	%res6 =l shr %res3, 1
	jmp @l2e
@l2e
	%n5 =l phi @l2b %n4, @l2c %n3
	%res5 =l phi @l2b %res4, @l2c %res6
	%bit5 =l shr %bit3, 2
	jmp @l2
@end
	ret %res3
}
