export function l $isqrt(l %n0) {
@start
	%bit0 =l copy 4611686018427387904
	jmp @loop1
@loop1
	%bit1 =l phi @start %bit0, @shift1 %bit2
	%c1 =w cugtl %bit1, %n0
	jnz %c1, @shift1, @loop2init
@shift1
	%bit2 =l shr %bit1, 2
	jmp @loop1
@loop2init
	jmp @loop2
@loop2
	%n =l phi @loop2init %n0, @join %n2
	%res =l phi @loop2init 0, @join %res2
	%bit =l phi @loop2init %bit1, @join %bitn
	%c2 =w cnel %bit, 0
	jnz %c2, @body, @done
@body
	%s =l add %res, %bit
	%c3 =w cugel %n, %s
	jnz %c3, @take, @skip
@take
	%n_t =l sub %n, %s
	%h =l shr %res, 1
	%res_t =l add %h, %bit
	jmp @join
@skip
	%res_s =l shr %res, 1
	jmp @join
@join
	%n2 =l phi @take %n_t, @skip %n
	%res2 =l phi @take %res_t, @skip %res_s
	%bitn =l shr %bit, 2
	jmp @loop2
@done
	ret %res
}
