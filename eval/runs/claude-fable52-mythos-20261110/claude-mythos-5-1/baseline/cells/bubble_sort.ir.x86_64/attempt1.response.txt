export function l $bubble_sort(l %a, l %n) {
@start
	%i =l copy 0
	jmp @outer
@outer
	%nm1 =l sub %n, 1
	%c =w csltl %i, %nm1
	jnz %c, @inner_init, @done
@inner_init
	%j =l copy 0
	jmp @inner
@inner
	%lim =l sub %nm1, %i
	%c2 =w csltl %j, %lim
	jnz %c2, @body, @inext
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%gt =w csgtl %x, %y
	jnz %gt, @swap, @jnext
@swap
	storel %y, %p
	storel %x, %q
	jmp @jnext
@jnext
	%j =l add %j, 1
	jmp @inner
@inext
	%i =l add %i, 1
	jmp @outer
@done
	ret 0
}
