export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init
@init
	%j =l phi @start 0, @initbody %j1
	%c =w cslel %j, %n
	jnz %c, @initbody, @outer
@initbody
	%p =l mul %j, 8
	%p2 =l add %row, %p
	storel 0, %p2
	%j1 =l add %j, 1
	jmp @init
@outer
	%i =l phi @init 0, @innerend %i1
	%ci =w csltl %i, %m
	jnz %ci, @outerbody, @ret
@outerbody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @inner
@inner
	%k =l phi @outerbody 1, @store %k1
	%diag =l phi @outerbody 0, @store %up
	%ck =w cslel %k, %n
	jnz %ck, @innerbody, @innerend
@innerbody
	%yo =l sub %k, 1
	%yp =l add %y, %yo
	%yb =w loadub %yp
	%ko =l mul %k, 8
	%rp =l add %row, %ko
	%up =l loadl %rp
	%lp =l sub %rp, 8
	%left =l loadl %lp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @store
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @useup, @useleft
@useup
	jmp @store
@useleft
	jmp @store
@store
	%v =l phi @match %v1, @useup %up, @useleft %left
	storel %v, %rp
	%k1 =l add %k, 1
	jmp @inner
@innerend
	%i1 =l add %i, 1
	jmp @outer
@ret
	%rn =l mul %n, 8
	%rpn =l add %row, %rn
	%res =l loadl %rpn
	ret %res
}
