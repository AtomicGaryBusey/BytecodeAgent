export function l $collatz_max(l %n) {
@start
	%k =l copy 1
	%max =l copy 0
	jmp @outer
@outer
	%c =w cslel %k, %n
	jnz %c, @body, @done
@body
	%x =l copy %k
	%steps =l copy 0
	jmp @inner
@inner
	%e =w ceql %x, 1
	jnz %e, @iend, @istep
@istep
	%r =l and %x, 1
	jnz %r, @odd, @even
@odd
	%t =l mul %x, 3
	%x =l add %t, 1
	%steps =l add %steps, 1
	jmp @inner
@even
	%x =l shr %x, 1
	%steps =l add %steps, 1
	jmp @inner
@iend
	%g =w csgtl %steps, %max
	jnz %g, @upd, @next
@upd
	%max =l copy %steps
	jmp @next
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %max
}
