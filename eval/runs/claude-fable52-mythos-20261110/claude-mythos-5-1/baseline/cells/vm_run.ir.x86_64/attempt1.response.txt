export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcs =l alloc8 8
	%ds =l alloc8 8
	storel 0, %pcs
	storel 0, %ds
	%i0 =l copy 0
	jmp @zloop
@zloop
	%i =l phi @start %i0, @zbody %i1
	%zc =w csltl %i, %cap
	jnz %zc, @zbody, @loop
@zbody
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i1 =l add %i, 1
	jmp @zloop
@loop
	%pc =l loadl %pcs
	%end =w ceql %pc, %len
	jnz %end, @done, @fetch
@fetch
	%o =l mul %pc, 8
	%ip =l add %code, %o
	%op =l loadl %ip
	%d =l loadl %ds
	%o2 =l mul %d, 8
	%sp =l add %stack, %o2
	%top =l sub %sp, 8
	%below =l sub %sp, 16
	%pc1 =l add %pc, 1
	%pc2 =l add %pc, 2
	%ip1 =l add %ip, 8
	%dp =l add %d, 1
	%dm =l sub %d, 1
	%isHalt =w ceql %op, 0
	jnz %isHalt, @done, @c1
@c1
	%t1 =w ceql %op, 1
	jnz %t1, @push, @c2
@push
	%pv =l loadl %ip1
	storel %pv, %sp
	storel %dp, %ds
	storel %pc2, %pcs
	jmp @loop
@c2
	%t2 =w ceql %op, 2
	jnz %t2, @add, @c3
@add
	%aa =l loadl %below
	%ab =l loadl %top
	%ar =l add %aa, %ab
	storel %ar, %below
	storel %dm, %ds
	storel %pc1, %pcs
	jmp @loop
@c3
	%t3 =w ceql %op, 3
	jnz %t3, @sub, @c4
@sub
	%sa =l loadl %below
	%sb =l loadl %top
	%sr =l sub %sa, %sb
	storel %sr, %below
	storel %dm, %ds
	storel %pc1, %pcs
	jmp @loop
@c4
	%t4 =w ceql %op, 4
	jnz %t4, @mul, @c5
@mul
	%ma =l loadl %below
	%mb =l loadl %top
	%mr =l mul %ma, %mb
	storel %mr, %below
	storel %dm, %ds
	storel %pc1, %pcs
	jmp @loop
@c5
	%t5 =w ceql %op, 5
	jnz %t5, @dup, @c6
@dup
	%dv =l loadl %top
	storel %dv, %sp
	storel %dp, %ds
	storel %pc1, %pcs
	jmp @loop
@c6
	%t6 =w ceql %op, 6
	jnz %t6, @swap, @c7
@swap
	%wa =l loadl %below
	%wb =l loadl %top
	storel %wb, %below
	storel %wa, %top
	storel %pc1, %pcs
	jmp @loop
@c7
	%t7 =w ceql %op, 7
	jnz %t7, @drop, @c8
@drop
	storel %dm, %ds
	storel %pc1, %pcs
	jmp @loop
@c8
	%t8 =w ceql %op, 8
	jnz %t8, @over, @c9
@over
	%ov =l loadl %below
	storel %ov, %sp
	storel %dp, %ds
	storel %pc1, %pcs
	jmp @loop
@c9
	%t9 =w ceql %op, 9
	jnz %t9, @lt, @c10
@lt
	%la =l loadl %below
	%lb =l loadl %top
	%lw =w csltl %la, %lb
	%lr =l extsw %lw
	storel %lr, %below
	storel %dm, %ds
	storel %pc1, %pcs
	jmp @loop
@c10
	%t10 =w ceql %op, 10
	jnz %t10, @jz, @c11
@jz
	%jv =l loadl %top
	storel %dm, %ds
	%jzc =w ceql %jv, 0
	jnz %jzc, @jztake, @jzskip
@jztake
	%jt =l loadl %ip1
	storel %jt, %pcs
	jmp @loop
@jzskip
	storel %pc2, %pcs
	jmp @loop
@c11
	%t11 =w ceql %op, 11
	jnz %t11, @jmp, @other
@jmp
	%jmt =l loadl %ip1
	storel %jmt, %pcs
	jmp @loop
@other
	storel %pc1, %pcs
	jmp @loop
@done
	%fd =l loadl %ds
	%fe =w ceql %fd, 0
	jnz %fe, @empty, @nonempty
@empty
	%r0 =l copy 0
	jmp @clear
@nonempty
	%fo =l mul %fd, 8
	%fp =l add %stack, %fo
	%ft =l sub %fp, 8
	%r1 =l loadl %ft
	jmp @clear
@clear
	%res =l phi @empty %r0, @nonempty %r1
	jmp @cloop
@cloop
	%k =l phi @clear %fd, @cbody %k1
	%cc =w csltl %k, %cap
	jnz %cc, @cbody, @ret
@cbody
	%ko =l mul %k, 8
	%kp =l add %stack, %ko
	storel 0, %kp
	%k1 =l add %k, 1
	jmp @cloop
@ret
	ret %res
}
