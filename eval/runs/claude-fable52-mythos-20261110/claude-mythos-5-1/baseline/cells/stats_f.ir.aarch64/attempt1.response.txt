export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%min =l copy %y0
	%max =l copy %y0
	%sum =l copy %y0
	%cnt =l and %y0, 1
	%i =l copy 1
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	%min =l copy %y
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @acc
@setmax
	%max =l copy %y
@acc
	%sum =l add %sum, %y
	%b =l and %y, 1
	%cnt =l add %cnt, %b
	%i =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
