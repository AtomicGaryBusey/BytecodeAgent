export function l $count_ugt(l %a, l %n, l %t) {
@start
	%cnt0 =l copy 0
	%i0 =l copy 0
	jmp @loop
@loop
	%cnt =l phi @start %cnt0, @body %cnt1
	%i =l phi @start %i0, @body %i1
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w cugtl %v, %t
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
