export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init
@init
	%j0 =l phi @start 0, @init %j1
	%p0 =l mul %j0, 8
	%p1 =l add %row, %p0
	storel 0, %p1
	%j1 =l add %j0, 1
	%c0 =w csltl %n, %j1
	jnz %c0, @ocheck, @init
@ocheck
	%i0 =l phi @init 0, @oend %i1
	%c1 =w csltl %i0, %m
	jnz %c1, @obody, @ret
@obody
	%xp =l add %x, %i0
	%xi =w loadub %xp
	jmp @icheck
@icheck
	%j2 =l phi @obody 1, @store %j3
	%diag0 =l phi @obody 0, @store %up
	%c2 =w csltl %n, %j2
	jnz %c2, @oend, @ibody
@ibody
	%p2 =l mul %j2, 8
	%p3 =l add %row, %p2
	%up =l loadl %p3
	%p4 =l sub %p3, 8
	%left =l loadl %p4
	%jm1 =l sub %j2, 1
	%yp =l add %y, %jm1
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @match, @nomatch
@match
	%v0 =l add %diag0, 1
	jmp @store
@nomatch
	%c3 =w csltl %up, %left
	jnz %c3, @useleft, @useup
@useleft
	jmp @store
@useup
	jmp @store
@store
	%v =l phi @match %v0, @useleft %left, @useup %up
	storel %v, %p3
	%j3 =l add %j2, 1
	jmp @icheck
@oend
	%i1 =l add %i0, 1
	jmp @ocheck
@ret
	%p5 =l mul %n, 8
	%p6 =l add %row, %p5
	%r =l loadl %p6
	ret %r
}
