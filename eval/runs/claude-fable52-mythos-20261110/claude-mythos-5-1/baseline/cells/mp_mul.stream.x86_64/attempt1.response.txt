mp_mul:
    push rbx
    push r12
    push r13
    mov r9, rdx                     # b
    lea r10, [rsi + rcx]            # n = na + nb
    xor r11d, r11d
.Lzero:
    cmp r11, r10
    jge .Lzdone
    mov qword ptr [r8 + r11*8], 0
    inc r11
    jmp .Lzero
.Lzdone:
    xor r10d, r10d                  # i
.Louter:
    cmp r10, rsi
    jge .Ldone_mul
    mov rdx, qword ptr [rdi + r10*8]
    xor eax, eax                    # carry
    lea r13, [r8 + r10*8]           # &r[i]
    xor r11d, r11d                  # j
.Linner:
    cmp r11, rcx
    jge .Lrow_end
    mulx rbx, r12, qword ptr [r9 + r11*8]
    add r12, rax
    adc rbx, 0
    add qword ptr [r13 + r11*8], r12
    adc rbx, 0
    mov rax, rbx
    inc r11
    jmp .Linner
.Lrow_end:
    mov qword ptr [r13 + r11*8], rax
    inc r10
    jmp .Louter
.Ldone_mul:
    lea rax, [rsi + rcx]
.Lcount:
    test rax, rax
    jz .Lret
    cmp qword ptr [r8 + rax*8 - 8], 0
    jne .Lret
    dec rax
    jmp .Lcount
.Lret:
    pop r13
    pop r12
    pop rbx
    ret
