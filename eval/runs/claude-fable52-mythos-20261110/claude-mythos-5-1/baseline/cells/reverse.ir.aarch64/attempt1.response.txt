export function l $reverse(l %a, l %n) {
@start
	%lo =l copy %a
	%nm1 =l sub %n, 1
	%off =l mul %nm1, 8
	%hi =l add %a, %off
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @done
@body
	%x =l loadl %lo
	%y =l loadl %hi
	storel %y, %lo
	storel %x, %hi
	%lo =l add %lo, 8
	%hi =l sub %hi, 8
	jmp @loop
@done
	ret 0
}
