isqrt:
    xor eax, eax
    mov rcx, 1
    shl rcx, 62
.Lfind:
    cmp rcx, rdi
    jbe .Lloop
    shr rcx, 2
    jmp .Lfind
.Lloop:
    test rcx, rcx
    jz .Ldone
    lea rdx, [rax + rcx]
    cmp rdi, rdx
    jb .Lskip
    sub rdi, rdx
    shr rax, 1
    add rax, rcx
    shr rcx, 2
    jmp .Lloop
.Lskip:
    shr rax, 1
    shr rcx, 2
    jmp .Lloop
.Ldone:
    ret
