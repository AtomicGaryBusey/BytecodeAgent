collatz_max:
	xor eax, eax                 # max steps
	mov rcx, 1                   # k
.Louter:
	cmp rcx, rdi
	jg .Ldone
	mov rdx, rcx                 # x
	xor esi, esi                 # step count
.Linner:
	cmp rdx, 1
	je .Lend
	test dl, 1
	jnz .Lodd
	shr rdx, 1
	inc rsi
	jmp .Linner
.Lodd:
	lea rdx, [rdx + rdx*2 + 1]
	inc rsi
	jmp .Linner
.Lend:
	cmp rsi, rax
	cmovg rax, rsi
	inc rcx
	jmp .Louter
.Ldone:
	ret
