export function l $count_desc(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	%c0 =w csgtl %nm1, 0
	jnz %c0, @loop, @done0
@done0
	ret 0
@loop
	%i =l phi @start 0, @next %i1
	%cnt =l phi @start 0, @next %cnt1
	%off =l mul %i, 8
	%p =l add %a, %off
	%p1 =l add %p, 8
	%x =l loadl %p
	%y =l loadl %p1
	%gt =w csgtl %x, %y
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	%c1 =w csltl %i1, %nm1
	jnz %c1, @next, @done
@next
	jmp @loop
@done
	ret %cnt1
}
