    mov x0, #0          // max
    mov x10, #1         // k

    b.gt .Ldone
    mov x11, x10        // x
    mov x12, #0         // steps
.Linner:

    b.eq .Linner_done
    tbnz x11, #0, .Lodd
    lsr x11, x11, #1
    add x12, x12, #1
    b .Linner
.Lodd:

    add x11, x13, #1
    add x12, x12, #1

    csel x0, x12, x0, gt
    add x10, x10, #1
    b .Louter
.Ldone:
    ret
