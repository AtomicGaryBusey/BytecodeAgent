collatz_max:
	mov	x9, #1			// k
	mov	x0, #0			// max
.Louter:
	cmp	x9, x0
	mov	x10, x9			// x = k
	mov	x11, #0			// steps
.Linner:
	cmp	x10, #1
	b.eq	.Ldone_k
	tbnz	x10, #0, .Lodd
	lsr	x10, x10, #1
	add	x11, x11, #1
	b	.Linner
.Lodd:
	add	x10, x10, x10, lsl #1	// 3x
	add	x10, x10, #1
	add	x11, x11, #1
	b	.Linner
.Ldone_k:
	cmp	x11, x0
	csel	x0, x11, x0, gt
	add	x9, x9, #1
	cmp	x9, x1
	b.le	.Louter_fix
	ret
.Louter_fix:
	b	.Louter
