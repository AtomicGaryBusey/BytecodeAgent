export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
    %ppc =l alloc8 8
    %pd =l alloc8 8
    %pi =l alloc8 8
    storel 0, %ppc
    storel 0, %pd
    jmp @loop
@loop
    %pc =l loadl %ppc
    %atend =w ceql %pc, %len
    jnz %atend, @done, @exec
@exec
    %pcoff =l mul %pc, 8
    %ip =l add %code, %pcoff
    %op =l loadl %ip
    %operaddr =l add %ip, 8
    %d =l loadl %pd
    %doff =l mul %d, 8
    %top =l add %stack, %doff
    %topm1 =l sub %top, 8
    %topm2 =l sub %top, 16
    %pc1 =l add %pc, 1
    %pc2 =l add %pc, 2
    %dp1 =l add %d, 1
    %dm1 =l sub %d, 1
    %c0 =w ceql %op, 0
    jnz %c0, @done, @n1
@n1
    %c1 =w ceql %op, 1
    jnz %c1, @push, @n2
@n2
    %c2 =w ceql %op, 2
    jnz %c2, @addop, @n3
@n3
    %c3 =w ceql %op, 3
    jnz %c3, @subop, @n4
@n4
    %c4 =w ceql %op, 4
    jnz %c4, @mulop, @n5
@n5
    %c5 =w ceql %op, 5
    jnz %c5, @dupop, @n6
@n6
    %c6 =w ceql %op, 6
    jnz %c6, @swapop, @n7
@n7
    %c7 =w ceql %op, 7
    jnz %c7, @dropop, @n8
@n8
    %c8 =w ceql %op, 8
    jnz %c8, @overop, @n9
@n9
    %c9 =w ceql %op, 9
    jnz %c9, @ltop, @n10
@n10
    %c10 =w ceql %op, 10
    jnz %c10, @jzop, @n11
@n11
    %c11 =w ceql %op, 11
    jnz %c11, @jmpop, @other
@other
    storel %pc1, %ppc
    jmp @loop
@push
    %pv =l loadl %operaddr
    storel %pv, %top
    storel %dp1, %pd
    storel %pc2, %ppc
    jmp @loop
@addop
    %ab =l loadl %topm1
    %aa =l loadl %topm2
    %ar =l add %aa, %ab
    storel %ar, %topm2
    storel %dm1, %pd
    storel %pc1, %ppc
    jmp @loop
@subop
    %sb =l loadl %topm1
    %sa =l loadl %topm2
    %sr =l sub %sa, %sb
    storel %sr, %topm2
    storel %dm1, %pd
    storel %pc1, %ppc
    jmp @loop
@mulop
    %mb =l loadl %topm1
    %ma =l loadl %topm2
    %mr =l mul %ma, %mb
    storel %mr, %topm2
    storel %dm1, %pd
    storel %pc1, %ppc
    jmp @loop
@dupop
    %dv =l loadl %topm1
    storel %dv, %top
    storel %dp1, %pd
    storel %pc1, %ppc
    jmp @loop
@swapop
    %wb =l loadl %topm1
    %wa =l loadl %topm2
    storel %wa, %topm1
    storel %wb, %topm2
    storel %pc1, %ppc
    jmp @loop
@dropop
    storel %dm1, %pd
    storel %pc1, %ppc
    jmp @loop
@overop
    %ov =l loadl %topm2
    storel %ov, %top
    storel %dp1, %pd
    storel %pc1, %ppc
    jmp @loop
@ltop
    %lb =l loadl %topm1
    %la =l loadl %topm2
    %lc =w csltl %la, %lb
    %lr =l extuw %lc
    storel %lr, %topm2
    storel %dm1, %pd
    storel %pc1, %ppc
    jmp @loop
@jzop
    %zv =l loadl %topm1
    %zt =l loadl %operaddr
    storel %dm1, %pd
    %zz =w ceql %zv, 0
    jnz %zz, @jztaken, @jznot
@jztaken
    storel %zt, %ppc
    jmp @loop
@jznot
    storel %pc2, %ppc
    jmp @loop
@jmpop
    %jt =l loadl %operaddr
    storel %jt, %ppc
    jmp @loop
@done
    %fd =l loadl %pd
    storel %fd, %pi
    jmp @zloop
@zloop
    %zi =l loadl %pi
    %zc =w csltl %zi, %cap
    jnz %zc, @zbody, @ret
@zbody
    %zoff =l mul %zi, 8
    %zaddr =l add %stack, %zoff
    storel 0, %zaddr
    %zi1 =l add %zi, 1
    storel %zi1, %pi
    jmp @zloop
@ret
    %fz =w ceql %fd, 0
    jnz %fz, @ret0, @rettop
@ret0
    ret 0
@rettop
    %fdm1 =l sub %fd, 1
    %foff =l mul %fdm1, 8
    %faddr =l add %stack, %foff
    %fr =l loadl %faddr
    ret %fr
}
