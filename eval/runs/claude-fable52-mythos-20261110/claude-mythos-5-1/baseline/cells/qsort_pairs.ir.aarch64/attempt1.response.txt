export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @body
@body
	%nm1 =l sub %n, 1
	%mid =l shr %n, 1
	%moff =l mul %mid, 16
	%pm =l add %a, %moff
	%pm1 =l add %pm, 8
	%loff =l mul %nm1, 16
	%pl =l add %a, %loff
	%pl1 =l add %pl, 8
	%mk =l loadl %pm
	%mv =l loadl %pm1
	%lk =l loadl %pl
	%lv =l loadl %pl1
	storel %lk, %pm
	storel %lv, %pm1
	storel %mk, %pl
	storel %mv, %pl1
	%pk =l copy %mk
	%pv =l copy %mv
	%i =l copy 0
	%j =l copy 0
@loop
	%cj =w csltl %j, %nm1
	jnz %cj, @cmp, @end
@cmp
	%joff =l mul %j, 16
	%pj =l add %a, %joff
	%pj1 =l add %pj, 8
	%kj =l loadl %pj
	%vj =l loadl %pj1
	%lt =w csltl %kj, %pk
	jnz %lt, @swap, @eqk
@eqk
	%eq =w ceql %kj, %pk
	jnz %eq, @cmpv, @next
@cmpv
	%ltv =w csltl %vj, %pv
	jnz %ltv, @swap, @next
@swap
	%ioff =l mul %i, 16
	%pi =l add %a, %ioff
	%pi1 =l add %pi, 8
	%ki =l loadl %pi
	%vi =l loadl %pi1
	storel %kj, %pi
	storel %vj, %pi1
	storel %ki, %pj
	storel %vi, %pj1
	%i =l add %i, 1
	jmp @next
@next
	%j =l add %j, 1
	jmp @loop
@end
	%ioff2 =l mul %i, 16
	%pi2 =l add %a, %ioff2
	%pi3 =l add %pi2, 8
	%ki2 =l loadl %pi2
	%vi2 =l loadl %pi3
	storel %pk, %pi2
	storel %pv, %pi3
	storel %ki2, %pl
	storel %vi2, %pl1
	%r1 =l call $qsort_pairs(l %a, l %i)
	%p2 =l add %pi2, 16
	%n2 =l sub %nm1, %i
	%r2 =l call $qsort_pairs(l %p2, l %n2)
	jmp @done
@done
	ret 0
}
