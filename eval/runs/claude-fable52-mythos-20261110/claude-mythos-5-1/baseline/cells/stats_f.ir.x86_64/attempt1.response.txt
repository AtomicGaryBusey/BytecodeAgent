export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%y0 =l call %f(l 0)
	%p0 =l loadl %a
	jmp @init
@init
	%ya =l call %f(l %p0)
	%min0 =l copy %ya
	%max0 =l copy %ya
	%sum0 =l copy %ya
	%odd0 =l and %ya, 1
	%i0 =l copy 1
	jmp @loop
@loop
	%min =l phi @init %min0, @body %min2
	%max =l phi @init %max0, @body %max2
	%sum =l phi @init %sum0, @body %sum2
	%odd =l phi @init %odd0, @body %odd2
	%i =l phi @init %i0, @body %i2
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%lt =w csltl %y, %min
	%gt =w csgtl %y, %max
	%nlt =w ceqw %lt, 0
	%ngt =w ceqw %gt, 0
	%mina =l mul %min, 1
	%nlt64 =l extuw %nlt
	%lt64 =l extuw %lt
	%ngt64 =l extuw %ngt
	%gt64 =l extuw %gt
	%nlt64n =l sub 0, %nlt64
	%lt64n =l sub 0, %lt64
	%ngt64n =l sub 0, %ngt64
	%gt64n =l sub 0, %gt64
	%mk =l and %min, %nlt64n
	%my =l and %y, %lt64n
	%min2 =l or %mk, %my
	%xk =l and %max, %ngt64n
	%xy =l and %y, %gt64n
	%max2 =l or %xk, %xy
	%sum2 =l add %sum, %y
	%o =l and %y, 1
	%odd2 =l add %odd, %o
	%i2 =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %odd
}
