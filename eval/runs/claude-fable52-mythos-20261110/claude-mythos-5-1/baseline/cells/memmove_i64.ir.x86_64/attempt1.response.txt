export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%ule =l cultl %dst, %src
	jnz %ule, @fwd, @bwd
@fwd
	%i =l copy 0
@floop
	%c =w csltl %i, %n
	jnz %c, @fbody, @done
@fbody
	%off =l mul %i, 8
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%v =l loadl %sp
	storel %v, %dp
	%i =l add %i, 1
	jmp @floop
@bwd
	%j =l copy %n
@bloop
	%c2 =w csgtl %j, 0
	jnz %c2, @bbody, @done
@bbody
	%j =l sub %j, 1
	%off2 =l mul %j, 8
	%sp2 =l add %src, %off2
	%dp2 =l add %dst, %off2
	%v2 =l loadl %sp2
	storel %v2, %dp2
	jmp @bloop
@done
	ret 0
}
