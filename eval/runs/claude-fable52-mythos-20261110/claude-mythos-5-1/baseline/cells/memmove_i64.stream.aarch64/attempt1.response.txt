memmove_i64:
	cmp x2, #0
	b.le .Ldone
	cmp x0, x1
	b.ls .Lforward
	// dst > src: copy backward
	mov x9, x2
.Lback:
	sub x9, x9, #1
	ldr x10, [x1, x9, lsl #3]
	str x10, [x0, x9, lsl #3]
	cbnz x9, .Lback
	b .Ldone
.Lforward:
	mov x9, #0
.Lfwd:
	ldr x10, [x1, x9, lsl #3]
	str x10, [x0, x9, lsl #3]
	add x9, x9, #1
	cmp x9, x2
	b.lt .Lfwd
.Ldone:
	mov x0, #0
	ret
