memmove_i64:
	xor eax, eax
	test rdx, rdx
	jle .Ldone
	cmp rdi, rsi
	jbe .Lforward
	# dst > src: copy backward
	xor ecx, ecx
	mov rcx, rdx
.Lback:
	mov r8, qword ptr [rsi + rcx*8 - 8]
	mov qword ptr [rdi + rcx*8 - 8], r8
	dec rcx
	jnz .Lback
	ret
.Lforward:
	xor ecx, ecx
.Lfwd:
	mov r8, qword ptr [rsi + rcx*8]
	mov qword ptr [rdi + rcx*8], r8
	inc rcx
	cmp rcx, rdx
	jne .Lfwd
.Ldone:
	ret
