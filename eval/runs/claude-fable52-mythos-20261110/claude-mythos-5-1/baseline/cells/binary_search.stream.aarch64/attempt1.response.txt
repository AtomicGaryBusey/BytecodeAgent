binary_search:
	mov x3, #0              // lo
	sub x4, x1, #1          // hi = n - 1
.Lloop:
	cmp x3, x4
	b.gt .Lnotfound
	add x5, x3, x4
	lsr x5, x5, #1          // mid = (lo + hi) >> 1 (both non-negative, no overflow)
	ldr x6, [x0, x5, lsl #3]
	cmp x6, x2
	b.eq .Lfound
	b.lt .Lgoright
	sub x4, x5, #1          // hi = mid - 1
	b .Lloop
.Lgoright:
	add x3, x5, #1          // lo = mid + 1
	b .Lloop
.Lfound:
	mov x0, x5
	ret
.Lnotfound:
	mov x0, #-1
	ret
