export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%w =l copy 0
@outer
	%c0 =w csltl %i, %n
	jnz %c0, @run_i, @done
@done
	ret %w
@run_i
	%pi =l add %src, %i
	%bi =w loadub %pi
	%lim1 =l add %i, 128
	%cl1 =w csltl %n, %lim1
	jnz %cl1, @lim1_n, @lim1_ok
@lim1_n
	%lim1 =l copy %n
@lim1_ok
	%k =l add %i, 1
@runi_loop
	%c1 =w csltl %k, %lim1
	jnz %c1, @runi_check, @runi_end
@runi_check
	%pk =l add %src, %k
	%bk =w loadub %pk
	%c2 =w ceqw %bk, %bi
	jnz %c2, @runi_inc, @runi_end
@runi_inc
	%k =l add %k, 1
	jmp @runi_loop
@runi_end
	%r =l sub %k, %i
	%c3 =w csgel %r, 3
	jnz %c3, @repeat, @literal
@repeat
	%w2 =l add %w, 2
	%c4 =w cslel %w2, %cap
	jnz %c4, @repeat_ok, @fail
@fail
	ret -1
@repeat_ok
	%hdr =l sub 257, %r
	%po =l add %out, %w
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %bi, %po1
	%w =l copy %w2
	%i =l add %i, %r
	jmp @outer
@literal
	%j =l copy %i
@lit_loop
	%c5 =w csltl %j, %n
	jnz %c5, @lit_chk128, @lit_end
@lit_chk128
	%d =l sub %j, %i
	%c6 =w csltl %d, 128
	jnz %c6, @run_j, @lit_end
@run_j
	%pj =l add %src, %j
	%bj =w loadub %pj
	%lim2 =l add %j, 128
	%cl2 =w csltl %n, %lim2
	jnz %cl2, @lim2_n, @lim2_ok
@lim2_n
	%lim2 =l copy %n
@lim2_ok
	%k2 =l add %j, 1
@runj_loop
	%c10 =w csltl %k2, %lim2
	jnz %c10, @runj_check, @runj_end
@runj_check
	%pk2 =l add %src, %k2
	%bk2 =w loadub %pk2
	%c11 =w ceqw %bk2, %bj
	jnz %c11, @runj_inc, @runj_end
@runj_inc
	%k2 =l add %k2, 1
	jmp @runj_loop
@runj_end
	%rj =l sub %k2, %j
	%c7 =w csgel %rj, 3
	jnz %c7, @lit_end, @lit_inc
@lit_inc
	%j =l add %j, 1
	jmp @lit_loop
@lit_end
	%L =l sub %j, %i
	%need =l add %L, 1
	%wn =l add %w, %need
	%c8 =w cslel %wn, %cap
	jnz %c8, @lit_ok, @fail
@lit_ok
	%hdr2 =l sub %L, 1
	%po2 =l add %out, %w
	storeb %hdr2, %po2
	%cp =l copy %i
	%dst =l add %po2, 1
@copy_loop
	%c9 =w csltl %cp, %j
	jnz %c9, @copy_body, @copy_end
@copy_body
	%ps =l add %src, %cp
	%bb =w loadub %ps
	storeb %bb, %dst
	%cp =l add %cp, 1
	%dst =l add %dst, 1
	jmp @copy_loop
@copy_end
	%w =l copy %wn
	%i =l copy %j
	jmp @outer
}
