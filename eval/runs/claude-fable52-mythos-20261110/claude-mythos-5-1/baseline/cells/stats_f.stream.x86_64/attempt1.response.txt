stats_f:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 24
    mov rbx, rdi                  # a
    mov rbp, rsi                  # n
    mov r12, rdx                  # f
    mov r13, rcx                  # out
    mov rdi, qword ptr [rbx]
    call r12
    mov r14, rax                  # min
    mov r15, rax                  # max
    mov qword ptr [rsp + 8], rax  # sum
    and eax, 1
    mov qword ptr [rsp + 16], rax # odd count
    mov qword ptr [rsp], 1        # index
.Lloop:
    mov rcx, qword ptr [rsp]
    cmp rcx, rbp
    jge .Ldone
    mov rdi, qword ptr [rbx + rcx*8]
    call r12
    cmp rax, r14
    cmovl r14, rax
    cmp rax, r15
    cmovg r15, rax
    add qword ptr [rsp + 8], rax
    and eax, 1
    add qword ptr [rsp + 16], rax
    inc qword ptr [rsp]
    jmp .Lloop
.Ldone:
    mov qword ptr [r13], r14
    mov qword ptr [r13 + 8], r15
    mov rax, qword ptr [rsp + 8]
    mov qword ptr [r13 + 16], rax
    mov rax, qword ptr [rsp + 16]
    add rsp, 24
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
