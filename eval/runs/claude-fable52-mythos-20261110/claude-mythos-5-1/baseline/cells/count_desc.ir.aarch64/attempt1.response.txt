export function l $count_desc(l %a, l %n) {
@start
	%c0 =w csltl %n, 2
	jnz %c0, @ret0, @init
@init
	%m =l sub %n, 1
	%i0 =l copy 0
	%cnt0 =l copy 0
	%prev0 =l loadl %a
	jmp @loop
@loop
	%i =l phi @init %i0, @body %i1
	%cnt =l phi @init %cnt0, @body %cnt1
	%prev =l phi @init %prev0, @body %cur
	%off =l mul %i, 8
	%p =l add %a, %off
	%p1 =l add %p, 8
	%cur =l loadl %p1
	%gt =w csgtl %prev, %cur
	%gtl =l extuw %gt
	%cnt1 =l add %cnt, %gtl
	%i1 =l add %i, 1
	%more =w csltl %i1, %m
	jnz %more, @body, @done
@body
	jmp @loop
@done
	ret %cnt1
@ret0
	ret 0
}
