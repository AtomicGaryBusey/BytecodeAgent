export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @hdr, @done
@done
	ret %o
@hdr
	%p =l add %src, %i
	%h =w loadsb %p
	%i =l add %i, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @notnop
@notnop
	%neg =w csltw %h, 0
	jnz %neg, @rep, @lit
@lit
	%len =l extsw %h
	%len =l add %len, 1
	%end =l add %i, %len
	%bad =w csgtl %end, %n
	jnz %bad, @err2, @litloop
@litloop
	%c2 =w csltl %i, %end
	jnz %c2, @litbody, @loop
@litbody
	%c3 =w csltl %o, %cap
	jnz %c3, @litwrite, @err1
@litwrite
	%sp1 =l add %src, %i
	%b =w loadub %sp1
	%dp =l add %out, %o
	storeb %b, %dp
	%i =l add %i, 1
	%o =l add %o, 1
	jmp @litloop
@rep
	%bad2 =w csgel %i, %n
	jnz %bad2, @err2, @repok
@repok
	%hl =l extsw %h
	%cnt =l sub 1, %hl
	%sp2 =l add %src, %i
	%b2 =w loadub %sp2
	%i =l add %i, 1
	jmp @reploop
@reploop
	%c4 =w csgtl %cnt, 0
	jnz %c4, @repbody, @loop
@repbody
	%c5 =w csltl %o, %cap
	jnz %c5, @repwrite, @err1
@repwrite
	%dp2 =l add %out, %o
	storeb %b2, %dp2
	%o =l add %o, 1
	%cnt =l sub %cnt, 1
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
}
