export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%k =l copy 0
@zloop
	%zc =w csltl %k, %n
	jnz %zc, @zbody, @zdone
@zbody
	%zoff =l mul %k, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%k =l add %k, 1
	jmp @zloop
@zdone
	%i =l copy 0
@iloop
	%ic =w csltl %i, %na
	jnz %ic, @ibody, @idone
@ibody
	%ioff =l mul %i, 8
	%ap =l add %a, %ioff
	%ai =l loadl %ap
	%al =l and %ai, 4294967295
	%ah =l shr %ai, 32
	%carry =l copy 0
	%j =l copy 0
@jloop
	%jc =w csltl %j, %nb
	jnz %jc, @jbody, @jdone
@jbody
	%joff =l mul %j, 8
	%bp =l add %b, %joff
	%bj =l loadl %bp
	%bl =l and %bj, 4294967295
	%bh =l shr %bj, 32
	%p0 =l mul %al, %bl
	%p1 =l mul %al, %bh
	%p2 =l mul %ah, %bl
	%p3 =l mul %ah, %bh
	%mid =l add %p1, %p2
	%c1w =w cultl %mid, %p1
	%c1 =l extuw %c1w
	%midlo =l shl %mid, 32
	%midhi =l shr %mid, 32
	%lo =l add %p0, %midlo
	%c2w =w cultl %lo, %p0
	%c2 =l extuw %c2w
	%c1s =l shl %c1, 32
	%hi0 =l add %p3, %midhi
	%hi1 =l add %hi0, %c1s
	%hi =l add %hi1, %c2
	%ij =l add %i, %j
	%ijoff =l mul %ij, 8
	%rp =l add %r, %ijoff
	%rv =l loadl %rp
	%s1 =l add %rv, %lo
	%c3w =w cultl %s1, %lo
	%c3 =l extuw %c3w
	%s2 =l add %s1, %carry
	%c4w =w cultl %s2, %s1
	%c4 =l extuw %c4w
	storel %s2, %rp
	%cr0 =l add %hi, %c3
	%carry =l add %cr0, %c4
	%j =l add %j, 1
	jmp @jloop
@jdone
	%inb =l add %i, %nb
	%inboff =l mul %inb, 8
	%rp2 =l add %r, %inboff
	storel %carry, %rp2
	%i =l add %i, 1
	jmp @iloop
@idone
	%k =l copy %n
@kloop
	%kc =w csgtl %k, 0
	jnz %kc, @kcheck, @kdone
@kcheck
	%km1 =l sub %k, 1
	%kmoff =l mul %km1, 8
	%kp =l add %r, %kmoff
	%kv =l loadl %kp
	%knz =w cnel %kv, 0
	jnz %knz, @kdone, @kdec
@kdec
	%k =l copy %km1
	jmp @kloop
@kdone
	ret %k
}
