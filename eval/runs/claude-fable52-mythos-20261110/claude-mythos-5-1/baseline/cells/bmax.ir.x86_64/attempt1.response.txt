export function l $bmax(l %a, l %n) {
@start
	%best =l loadl %a
	%i =l copy 1
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%gt =w csgtl %v, %best
	jnz %gt, @upd, @next
@upd
	%best =l copy %v
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %best
}
