export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l mul %nn, 2
	%n8 =l mul %n, 8
	%np1 =l add %n, 1
	%stride =l mul %np1, 8
	%nn8 =l mul %nn, 8
	%outend =l add %out, %nn8
@zero_init
	%p =l phi @start %out, @zero_loop %p1
	%c1 =w cultl %p, %outend
	jnz %c1, @zero_loop, @diag
@zero_loop
	storel 0, %p
	%p1 =l add %p, 8
	jmp @zero_init
@diag
	%d =l phi @zero_init %out, @diag_loop %d1
	%c2 =w cultl %d, %outend
	jnz %c2, @diag_loop, @pow_init
@diag_loop
	storel 1, %d
	%d1 =l add %d, %stride
	jmp @diag
@pow_init
	%i =l phi @diag 0, @copy_done %i1
	%c3 =w csltl %i, %k
	jnz %c3, @mul_i_init, @final_zero
@mul_i_init
	%ri =l phi @pow_init 0, @mul_i_next %ri1
	%c4 =w csltl %ri, %n
	jnz %c4, @mul_j_init, @copy_init
@mul_j_init
	%rj =l phi @mul_i_init 0, @mul_j_next %rj1
	%c5 =w csltl %rj, %n
	jnz %c5, @mul_t_init, @mul_i_next
@mul_t_init
	%rowb =l mul %ri, %n8
	%xp0 =l add %out, %rowb
	%j8 =l mul %rj, 8
	%yp0 =l add %a, %j8
@mul_t
	%t =l phi @mul_t_init 0, @mul_t_body %t1
	%xp =l phi @mul_t_init %xp0, @mul_t_body %xp1
	%yp =l phi @mul_t_init %yp0, @mul_t_body %yp1
	%sum =l phi @mul_t_init 0, @mul_t_body %sum1
	%c6 =w csltl %t, %n
	jnz %c6, @mul_t_body, @mul_t_done
@mul_t_body
	%xv =l loadl %xp
	%yv =l loadl %yp
	%pr =l mul %xv, %yv
	%sum1 =l add %sum, %pr
	%xp1 =l add %xp, 8
	%yp1 =l add %yp, %n8
	%t1 =l add %t, 1
	jmp @mul_t
@mul_t_done
	%ij =l add %rowb, %j8
	%pp =l add %tmp, %ij
	storel %sum, %pp
	jmp @mul_j_next
@mul_j_next
	%rj1 =l add %rj, 1
	jmp @mul_j_init
@mul_i_next
	%ri1 =l add %ri, 1
	jmp @mul_i_init
@copy_init
	%cp =l phi @mul_i_init 0, @copy_body %cp1
	%c7 =w cultl %cp, %nn8
	jnz %c7, @copy_body, @copy_done
@copy_body
	%src =l add %tmp, %cp
	%dst =l add %out, %cp
	%v =l loadl %src
	storel %v, %dst
	%cp1 =l add %cp, 8
	jmp @copy_init
@copy_done
	%i1 =l add %i, 1
	jmp @pow_init
@final_zero
	%nn28 =l mul %nn2, 8
	%tend =l add %tmp, %nn28
@fz
	%z =l phi @final_zero %tmp, @fz_body %z1
	%c8 =w cultl %z, %tend
	jnz %c8, @fz_body, @ret
@fz_body
	storel 0, %z
	%z1 =l add %z, 8
	jmp @fz
@ret
	ret 0
}
