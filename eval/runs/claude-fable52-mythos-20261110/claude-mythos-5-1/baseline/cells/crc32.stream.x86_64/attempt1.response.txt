crc32:
	mov eax, 0xFFFFFFFF
	test rsi, rsi
	jle .Ldone
	lea rdx, [rdi + rsi]
.Lbyte:
	movzx ecx, byte ptr [rdi]
	xor eax, ecx
	mov ecx, 8
.Lbit:
	mov r8d, eax
	shr eax, 1
	and r8d, 1
	neg r8d
	and r8d, 0xEDB88320
	xor eax, r8d
	dec ecx
	jnz .Lbit
	inc rdi
	cmp rdi, rdx
	jne .Lbyte
.Ldone:
	not eax
	ret
