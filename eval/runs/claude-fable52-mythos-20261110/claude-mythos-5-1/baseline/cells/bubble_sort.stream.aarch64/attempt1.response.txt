bubble_sort:
    cmp     x1, #2
    b.lt    .Ldone
    sub     x9, x1, #1          // x9 = number of comparisons in this pass (n-1)
.Louter:
    mov     x10, #0             // i = 0
    mov     x11, #0             // swapped flag
.Linner:
    ldr     x12, [x0, x10, lsl #3]
    add     x13, x10, #1
    ldr     x14, [x0, x13, lsl #3]
    cmp     x12, x14
    b.le    .Lnoswap
    str     x14, [x0, x10, lsl #3]
    str     x12, [x0, x13, lsl #3]
    mov     x11, #1
.Lnoswap:
    mov     x10, x13
    cmp     x10, x9
    b.lt    .Linner
    cbz     x11, .Ldone
    subs    x9, x9, #1
    b.gt    .Louter
.Ldone:
    mov     x0, #0
    ret
