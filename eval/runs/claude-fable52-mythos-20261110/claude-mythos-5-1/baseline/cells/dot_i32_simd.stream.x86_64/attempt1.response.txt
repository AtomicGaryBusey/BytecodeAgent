dot_i32_simd:
    pxor xmm2, xmm2
    pxor xmm3, xmm3
    xor eax, eax
    xor r8d, r8d
    mov rcx, rdx
    and rcx, -4
    jz .Ltail
.Lvec:
    pmovsxdq xmm0, qword ptr [rdi + r8*4]
    pmovsxdq xmm1, qword ptr [rsi + r8*4]
    pmuldq xmm0, xmm1
    paddq xmm2, xmm0
    pmovsxdq xmm0, qword ptr [rdi + r8*4 + 8]
    pmovsxdq xmm1, qword ptr [rsi + r8*4 + 8]
    pmuldq xmm0, xmm1
    paddq xmm3, xmm0
    add r8, 4
    cmp r8, rcx
    jb .Lvec
    paddq xmm2, xmm3
    movq rax, xmm2
    pshufd xmm2, xmm2, 0xee
    movq r9, xmm2
    add rax, r9
.Ltail:
    cmp r8, rdx
    jae .Ldone
    movsxd r9, dword ptr [rdi + r8*4]
    movsxd r10, dword ptr [rsi + r8*4]
    imul r9, r10
    add rax, r9
    inc r8
    jmp .Ltail
.Ldone:
    ret
