fib_rec:
    cmp rdi, 2
    jb .Lbase
    push rbx
    push rbp
    push r12            # 3 pushes: rsp now 0 mod 16
    mov rbx, rdi
    lea rdi, [rbx - 1]
    call fib_rec
    mov rbp, rax
    lea rdi, [rbx - 2]
    call fib_rec
    add rax, rbp
    pop r12
    pop rbp
    pop rbx
    ret
.Lbase:
    mov rax, rdi
    ret
