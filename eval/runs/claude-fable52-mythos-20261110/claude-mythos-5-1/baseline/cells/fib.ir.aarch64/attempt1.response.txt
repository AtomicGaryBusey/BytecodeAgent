export function l $fib(l %n) {
@start
	%a0 =l copy 0
	%b0 =l copy 1
	%i0 =l copy 0
	jmp @loop
@loop
	%a =l phi @start %a0, @body %b
	%b =l phi @start %b0, @body %c
	%i =l phi @start %i0, @body %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%c =l add %a, %b
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %a
}
