bubble_sort:
    cmp rsi, 2
    jl .Ldone
    mov rcx, rsi
.Louter:
    dec rcx
    jz .Ldone
    xor rdx, rdx
    xor r8d, r8d
.Linner:
    mov rax, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r9
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], rax
    mov r8d, 1
.Lnoswap:
    inc rdx
    cmp rdx, rcx
    jb .Linner
    test r8d, r8d
    jnz .Louter
.Ldone:
    xor eax, eax
    ret
