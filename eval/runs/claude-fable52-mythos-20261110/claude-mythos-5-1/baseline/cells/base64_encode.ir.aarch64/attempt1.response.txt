export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%o =l copy 0
	jmp @loop
@loop
	%rem =l sub %n, %i
	%c =w csgel %rem, 3
	jnz %c, @full, @tail
@full
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	%t0 =l shl %b0, 16
	%t1 =l shl %b1, 8
	%t2 =l or %t0, %t1
	%val =l or %t2, %b2
	%cnt =l copy 4
	%pad =l copy 0
	%i =l add %i, 3
	jmp @emit
@tail
	%z =w ceql %rem, 0
	jnz %z, @done, @partial
@partial
	%q0 =l add %src, %i
	%c0 =l loadub %q0
	%u0 =l shl %c0, 16
	%one =w ceql %rem, 1
	jnz %one, @tail1, @tail2
@tail1
	%val =l copy %u0
	%cnt =l copy 2
	%pad =l copy 2
	%i =l copy %n
	jmp @emit
@tail2
	%q1 =l add %q0, 1
	%c1 =l loadub %q1
	%u1 =l shl %c1, 8
	%val =l or %u0, %u1
	%cnt =l copy 3
	%pad =l copy 1
	%i =l copy %n
	jmp @emit
@emit
	%sh =l copy 18
	%k =l copy 0
	jmp @eloop
@eloop
	%v0 =l shr %val, %sh
	%v =l and %v0, 63
	%lt26 =w csltl %v, 26
	jnz %lt26, @mA, @m2
@mA
	%ch =l add %v, 65
	jmp @store
@m2
	%lt52 =w csltl %v, 52
	jnz %lt52, @ma, @m3
@ma
	%ch =l add %v, 71
	jmp @store
@m3
	%lt62 =w csltl %v, 62
	jnz %lt62, @md, @m4
@md
	%ch =l sub %v, 4
	jmp @store
@m4
	%is62 =w ceql %v, 62
	jnz %is62, @mp, @ms
@mp
	%ch =l copy 43
	jmp @store
@ms
	%ch =l copy 47
	jmp @store
@store
	%op =l add %out, %o
	storeb %ch, %op
	%o =l add %o, 1
	%sh =l sub %sh, 6
	%k =l add %k, 1
	%more =w csltl %k, %cnt
	jnz %more, @eloop, @padloop
@padloop
	%pz =w ceql %pad, 0
	jnz %pz, @loop, @padw
@padw
	%op2 =l add %out, %o
	storeb 61, %op2
	%o =l add %o, 1
	%pad =l sub %pad, 1
	jmp @padloop
@done
	ret %o
}
