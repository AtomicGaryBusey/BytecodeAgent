memmove_i64:
    test rdx, rdx
    jle .Ldone
    cmp rdi, rsi
    je .Ldone
    ja .Lbackward
    xor rcx, rcx
.Lfwd:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    inc rcx
    cmp rcx, rdx
    jl .Lfwd
    jmp .Ldone
.Lbackward:
    mov rcx, rdx
.Lbwd:
    dec rcx
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    test rcx, rcx
    jnz .Lbwd
.Ldone:
    xor eax, eax
    ret
