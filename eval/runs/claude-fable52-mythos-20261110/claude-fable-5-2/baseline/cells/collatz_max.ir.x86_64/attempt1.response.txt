export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @next %k1
	%max =l phi @start 0, @next %max1
	%c =w cslel %k, %n
	jnz %c, @inner_init, @done
@inner_init
	jmp @inner
@inner
	%x =l phi @inner_init %k, @step %x1
	%s =l phi @inner_init 0, @step %s1
	%e =w ceql %x, 1
	jnz %e, @finish, @body
@body
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @isodd, @iseven
@isodd
	%t =l mul %x, 3
	%xo =l add %t, 1
	jmp @step
@iseven
	%xe =l shr %x, 1
	jmp @step
@step
	%x1 =l phi @isodd %xo, @iseven %xe
	%s1 =l add %s, 1
	jmp @inner
@finish
	%g =w csgtl %s, %max
	jnz %g, @upd, @next
@upd
	jmp @next
@next
	%max1 =l phi @finish %max, @upd %s
	%k1 =l add %k, 1
	jmp @outer
@done
	ret %max
}
