export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @init
@init
	%mid =l shr %n, 1
	%moff =l shl %mid, 4
	%pm =l add %a, %moff
	%last =l sub %n, 1
	%loff =l shl %last, 4
	%pl =l add %a, %loff
	%pl8 =l add %pl, 8
	%pm8 =l add %pm, 8
	%pk =l loadl %pm
	%pv =l loadl %pm8
	%lk =l loadl %pl
	%lv =l loadl %pl8
	storel %lk, %pm
	storel %lv, %pm8
	storel %pk, %pl
	storel %pv, %pl8
	%i =l copy 0
	%j =l copy 0
	jmp @loop
@loop
	%c2 =w csltl %j, %last
	jnz %c2, @body, @end
@body
	%joff =l shl %j, 4
	%pj =l add %a, %joff
	%pj8 =l add %pj, 8
	%jk =l loadl %pj
	%jv =l loadl %pj8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @chkeq
@chkeq
	%eq =w ceql %jk, %pk
	jnz %eq, @chkv, @next
@chkv
	%ltv =w csltl %jv, %pv
	jnz %ltv, @swap, @next
@swap
	%ioff =l shl %i, 4
	%pi =l add %a, %ioff
	%pi8 =l add %pi, 8
	%ik =l loadl %pi
	%iv =l loadl %pi8
	storel %jk, %pi
	storel %jv, %pi8
	storel %ik, %pj
	storel %iv, %pj8
	%i =l add %i, 1
	jmp @next
@next
	%j =l add %j, 1
	jmp @loop
@end
	%eoff =l shl %i, 4
	%pe =l add %a, %eoff
	%pe8 =l add %pe, 8
	%ek =l loadl %pe
	%ev =l loadl %pe8
	storel %ek, %pl
	storel %ev, %pl8
	storel %pk, %pe
	storel %pv, %pe8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%right =l add %pe, 16
	%rn =l sub %last, %i
	%r2 =l call $qsort_pairs(l %right, l %rn)
	jmp @done
@done
	ret 0
}
