export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
@loop
	%c =w csltl %pc, %len
	jnz %c, @fetch, @done
@fetch
	%o =l mul %pc, 8
	%p =l add %code, %o
	%op =l loadl %p
	%t =l mul %d, 8
	%tp =l add %stack, %t
	%bp =l sub %tp, 8
	%ap =l sub %tp, 16
	%c =w ceql %op, 0
	jnz %c, @done, @n1
@n1
	%c =w ceql %op, 1
	jnz %c, @push, @n2
@n2
	%c =w ceql %op, 2
	jnz %c, @add, @n3
@n3
	%c =w ceql %op, 3
	jnz %c, @sub, @n4
@n4
	%c =w ceql %op, 4
	jnz %c, @mul, @n5
@n5
	%c =w ceql %op, 5
	jnz %c, @dup, @n6
@n6
	%c =w ceql %op, 6
	jnz %c, @swap, @n7
@n7
	%c =w ceql %op, 7
	jnz %c, @drop, @n8
@n8
	%c =w ceql %op, 8
	jnz %c, @over, @n9
@n9
	%c =w ceql %op, 9
	jnz %c, @lt, @n10
@n10
	%c =w ceql %op, 10
	jnz %c, @jz, @jump
@push
	%q =l add %p, 8
	%v =l loadl %q
	storel %v, %tp
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@add
	%a =l loadl %ap
	%b =l loadl %bp
	%r =l add %a, %b
	jmp @bin
@sub
	%a =l loadl %ap
	%b =l loadl %bp
	%r =l sub %a, %b
	jmp @bin
@mul
	%a =l loadl %ap
	%b =l loadl %bp
	%r =l mul %a, %b
	jmp @bin
@lt
	%a =l loadl %ap
	%b =l loadl %bp
	%rw =w csltl %a, %b
	%r =l extuw %rw
	jmp @bin
@bin
	storel %r, %ap
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@dup
	%b =l loadl %bp
	storel %b, %tp
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@swap
	%a =l loadl %ap
	%b =l loadl %bp
	storel %b, %ap
	storel %a, %bp
	%pc =l add %pc, 1
	jmp @loop
@drop
	%d =l sub %d, 1
	%pc =l add %pc, 1
	jmp @loop
@over
	%a =l loadl %ap
	storel %a, %tp
	%d =l add %d, 1
	%pc =l add %pc, 1
	jmp @loop
@jz
	%v =l loadl %bp
	%d =l sub %d, 1
	%c =w ceql %v, 0
	jnz %c, @jump, @skip2
@skip2
	%pc =l add %pc, 2
	jmp @loop
@jump
	%q =l add %p, 8
	%pc =l loadl %q
	jmp @loop
@done
	%i =l copy %d
@zloop
	%c =w csltl %i, %cap
	jnz %c, @zbody, @fin
@zbody
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i =l add %i, 1
	jmp @zloop
@fin
	%c =w ceql %d, 0
	jnz %c, @ret0, @rettop
@ret0
	ret 0
@rettop
	%fo =l mul %d, 8
	%fp =l add %stack, %fo
	%fp =l sub %fp, 8
	%res =l loadl %fp
	ret %res
}
