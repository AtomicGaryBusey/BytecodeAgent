export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn2 =l add %nn, %nn
	%n1 =l add %n, 1
	%i =l copy 0
@zl
	%o =l shl %i, 3
	%p =l add %out, %o
	storel 0, %p
	%i =l add %i, 1
	%c =w csltl %i, %nn
	jnz %c, @zl, @zd
@zd
	%i =l copy 0
@dl
	%o =l mul %i, %n1
	%o =l shl %o, 3
	%p =l add %out, %o
	storel 1, %p
	%i =l add %i, 1
	%c =w csltl %i, %n
	jnz %c, @dl, @dd
@dd
	%s =l copy 0
@kl
	%c =w csltl %s, %k
	jnz %c, @mul, @fin
@mul
	%i =l copy 0
@il
	%j =l copy 0
@jl
	%t =l copy 0
	%acc =l copy 0
@tl
	%x =l mul %i, %n
	%x =l add %x, %t
	%x =l shl %x, 3
	%x =l add %out, %x
	%xv =l loadl %x
	%y =l mul %t, %n
	%y =l add %y, %j
	%y =l shl %y, 3
	%y =l add %a, %y
	%yv =l loadl %y
	%m =l mul %xv, %yv
	%acc =l add %acc, %m
	%t =l add %t, 1
	%c =w csltl %t, %n
	jnz %c, @tl, @td
@td
	%z =l mul %i, %n
	%z =l add %z, %j
	%z =l shl %z, 3
	%z =l add %tmp, %z
	storel %acc, %z
	%j =l add %j, 1
	%c =w csltl %j, %n
	jnz %c, @jl, @jd
@jd
	%i =l add %i, 1
	%c =w csltl %i, %n
	jnz %c, @il, @id
@id
	%i =l copy 0
@cl
	%o =l shl %i, 3
	%p =l add %tmp, %o
	%v =l loadl %p
	%q =l add %out, %o
	storel %v, %q
	%i =l add %i, 1
	%c =w csltl %i, %nn
	jnz %c, @cl, @cd
@cd
	%s =l add %s, 1
	jmp @kl
@fin
	%i =l copy 0
@fl
	%o =l shl %i, 3
	%p =l add %tmp, %o
	storel 0, %p
	%i =l add %i, 1
	%c =w csltl %i, %nn2
	jnz %c, @fl, @done
@done
	ret 0
}
