export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%j0 =l copy 0
@zloop
	%jz =l phi @start %j0, @zbody %jz1
	%c =w cslel %jz, %n
	jnz %c, @zbody, @zdone
@zbody
	%off =l shl %jz, 3
	%p =l add %row, %off
	storel 0, %p
	%jz1 =l add %jz, 1
	jmp @zloop
@zdone
	%i0 =l copy 0
@oloop
	%i =l phi @zdone %i0, @oend %i1
	%c2 =w csltl %i, %m
	jnz %c2, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%d0 =l copy 0
	%j1 =l copy 1
@iloop
	%j =l phi @obody %j1, @istore %jn
	%diag =l phi @obody %d0, @istore %up
	%c3 =w cslel %j, %n
	jnz %c3, @ibody, @oend
@ibody
	%o2 =l shl %j, 3
	%rp =l add %row, %o2
	%up =l loadl %rp
	%rpm =l sub %rp, 8
	%left =l loadl %rpm
	%ym =l add %y, %j
	%yp =l sub %ym, 1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%v1 =l add %diag, 1
	jmp @istore
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @useup, @useleft
@useup
	jmp @istore
@useleft
	jmp @istore
@istore
	%v =l phi @match %v1, @useup %up, @useleft %left
	storel %v, %rp
	%jn =l add %j, 1
	jmp @iloop
@oend
	%i1 =l add %i, 1
	jmp @oloop
@done
	%o3 =l shl %n, 3
	%rr =l add %row, %o3
	%res =l loadl %rr
	ret %res
}
