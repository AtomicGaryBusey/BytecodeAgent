bubble_sort:
    cmp x1, #2
    b.lt .Ldone
    sub x9, x1, #1              // x9 = number of comparisons in this pass
.Louter:
    mov x10, #0                 // x10 = index
    mov x11, #0                 // x11 = swapped flag
    mov x12, x0                 // x12 = pointer to a[j]
.Linner:
    ldp x13, x14, [x12]
    cmp x13, x14
    b.le .Lnoswap
    stp x14, x13, [x12]
    mov x11, #1
.Lnoswap:
    add x12, x12, #8
    add x10, x10, #1
    cmp x10, x9
    b.lt .Linner
    cbz x11, .Ldone
    subs x9, x9, #1
    b.gt .Louter
.Ldone:
    mov x0, #0
    ret
