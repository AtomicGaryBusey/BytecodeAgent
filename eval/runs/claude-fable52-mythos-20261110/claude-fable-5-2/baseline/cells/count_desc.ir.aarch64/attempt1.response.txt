export function l $count_desc(l %a, l %n) {
@start
	%m =l sub %n, 1
@loop
	%i =l phi @start 0, @body %i1
	%cnt =l phi @start 0, @body %cnt1
	%c =w csltl %i, %m
	jnz %c, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%g =w csgtl %x, %y
	%ge =l extuw %g
	%cnt1 =l add %cnt, %ge
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
}
