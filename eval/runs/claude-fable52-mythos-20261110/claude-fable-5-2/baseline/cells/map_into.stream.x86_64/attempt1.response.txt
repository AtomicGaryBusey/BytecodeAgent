map_into:
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov rbx, rdi
mov rbp, rsi
mov r12, rdx
mov r13, rcx
xor r14d, r14d
xor r15d, r15d
.Lloop:
cmp r14, r12
jge .Ldone
mov rdi, qword ptr [rbp + r14*8]
call r13
mov qword ptr [rbx + r14*8], rax
cmp rax, qword ptr [rbp + r14*8]
jge .Lskip
inc r15
.Lskip:
inc r14
jmp .Lloop
.Ldone:
mov rax, r15
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
