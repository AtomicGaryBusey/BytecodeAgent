export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@top
	%c =w csltl %i, %n
	jnz %c, @begin, @done
@begin
	%j =l copy %i
@runinit
	%pj =l add %src, %j
	%b =w loadub %pj
	%r =l copy 1
@runloop
	%k =l add %j, %r
	%c1 =w csltl %k, %n
	jnz %c1, @rl2, @rundone
@rl2
	%c2 =w csltl %r, 128
	jnz %c2, @rl3, @rundone
@rl3
	%pk =l add %src, %k
	%b2 =w loadub %pk
	%c3 =w ceqw %b2, %b
	jnz %c3, @rinc, @rundone
@rinc
	%r =l add %r, 1
	jmp @runloop
@rundone
	%c4 =w csgel %r, 3
	jnz %c4, @isrun, @norun
@isrun
	%c5 =w ceql %j, %i
	jnz %c5, @emitrep, @emitlit
@norun
	%j =l add %j, 1
	%c6 =w csltl %j, %n
	jnz %c6, @nr2, @emitlit
@nr2
	%d =l sub %j, %i
	%c7 =w csltl %d, 128
	jnz %c7, @runinit, @emitlit
@emitrep
	%e =l add %o, 2
	%c8 =w csgtl %e, %cap
	jnz %c8, @fail, @rep2
@rep2
	%po =l add %out, %o
	%rw =w copy %r
	%h =w sub 257, %rw
	storeb %h, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%o =l copy %e
	%i =l add %i, %r
	jmp @top
@emitlit
	%len =l sub %j, %i
	%e2 =l add %o, %len
	%e3 =l add %e2, 1
	%c9 =w csgtl %e3, %cap
	jnz %c9, @fail, @lit2
@lit2
	%pl =l add %out, %o
	%lw =w copy %len
	%hl =w sub %lw, 1
	storeb %hl, %pl
	%o =l add %o, 1
@copyloop
	%c10 =w csltl %i, %j
	jnz %c10, @copybody, @top
@copybody
	%ps =l add %src, %i
	%v =w loadub %ps
	%pd =l add %out, %o
	storeb %v, %pd
	%i =l add %i, 1
	%o =l add %o, 1
	jmp @copyloop
@fail
	ret -1
@done
	ret %o
}
