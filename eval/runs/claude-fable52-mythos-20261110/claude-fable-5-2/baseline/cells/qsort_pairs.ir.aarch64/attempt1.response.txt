export function l $qsort_pairs(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @done, @work
@work
	%last =l sub %n, 1
	%lo =l shl %last, 4
	%lp =l add %a, %lo
	%lp8 =l add %lp, 8
	%mid =l shr %n, 1
	%mo =l shl %mid, 4
	%mp =l add %a, %mo
	%mp8 =l add %mp, 8
	%pk =l loadl %mp
	%pv =l loadl %mp8
	%tk =l loadl %lp
	%tv =l loadl %lp8
	storel %tk, %mp
	storel %tv, %mp8
	storel %pk, %lp
	storel %pv, %lp8
	jmp @loop
@loop
	%i =l phi @work 0, @next %i2
	%j =l phi @work 0, @next %j2
	%c2 =w csltl %j, %last
	jnz %c2, @body, @fin
@body
	%jo =l shl %j, 4
	%jp =l add %a, %jo
	%jp8 =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jp8
	%lt =w csltl %jk, %pk
	jnz %lt, @swap, @cheq
@cheq
	%eq =w ceql %jk, %pk
	jnz %eq, @chv, @noswap
@chv
	%ltv =w csltl %jv, %pv
	jnz %ltv, @swap, @noswap
@swap
	%io =l shl %i, 4
	%ip =l add %a, %io
	%ip8 =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ip8
	storel %jk, %ip
	storel %jv, %ip8
	storel %ik, %jp
	storel %iv, %jp8
	%i1 =l add %i, 1
	jmp @next
@noswap
	jmp @next
@next
	%i2 =l phi @swap %i1, @noswap %i
	%j2 =l add %j, 1
	jmp @loop
@fin
	%fo =l shl %i, 4
	%fp =l add %a, %fo
	%fp8 =l add %fp, 8
	%fk =l loadl %fp
	%fv =l loadl %fp8
	storel %pk, %fp
	storel %pv, %fp8
	storel %fk, %lp
	storel %fv, %lp8
	%r1 =l call $qsort_pairs(l %a, l %i)
	%rp =l add %fp, 16
	%rn =l sub %last, %i
	%r2 =l call $qsort_pairs(l %rp, l %rn)
	jmp @done
@done
	ret 0
}
