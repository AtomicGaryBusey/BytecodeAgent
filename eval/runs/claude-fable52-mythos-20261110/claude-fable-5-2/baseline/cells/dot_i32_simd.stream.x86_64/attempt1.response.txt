dot_i32_simd:
    pxor xmm0, xmm0
    xor ecx, ecx
    mov r8, rdx
    and r8, -4
    jz .Lreduce
.Lloop:
    movdqu xmm1, xmmword ptr [rdi + rcx*4]
    movdqu xmm2, xmmword ptr [rsi + rcx*4]
    pshufd xmm3, xmm1, 0xF5
    pshufd xmm4, xmm2, 0xF5
    pmuldq xmm1, xmm2
    pmuldq xmm3, xmm4
    paddq xmm0, xmm1
    paddq xmm0, xmm3
    add rcx, 4
    cmp rcx, r8
    jl .Lloop
.Lreduce:
    movq rax, xmm0
    psrldq xmm0, 8
    movq r9, xmm0
    add rax, r9
.Ltail:
    cmp rcx, rdx
    jge .Ldone
    movsxd r9, dword ptr [rdi + rcx*4]
    movsxd r10, dword ptr [rsi + rcx*4]
    imul r9, r10
    add rax, r9
    inc rcx
    jmp .Ltail
.Ldone:
    ret
