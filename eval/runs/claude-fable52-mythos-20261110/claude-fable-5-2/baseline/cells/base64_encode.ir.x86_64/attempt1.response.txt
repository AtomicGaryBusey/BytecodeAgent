export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i2
	%o =l phi @start 0, @next %o2
	%rem =l sub %n, %i
	%c =w csgtl %rem, 0
	jnz %c, @body, @done
@body
	%p =l add %src, %i
	%b0 =l loadub %p
	%c1 =w csgtl %rem, 1
	jnz %c1, @l1, @j1
@l1
	%p1 =l add %p, 1
	%b1x =l loadub %p1
	jmp @j1
@j1
	%b1 =l phi @body 0, @l1 %b1x
	%c2 =w csgtl %rem, 2
	jnz %c2, @l2, @j2
@l2
	%p2 =l add %p, 2
	%b2x =l loadub %p2
	jmp @j2
@j2
	%b2 =l phi @j1 0, @l2 %b2x
	%s0 =l shl %b0, 16
	%s1 =l shl %b1, 8
	%v0 =l or %s0, %s1
	%v =l or %v0, %b2
	jmp @inner
@inner
	%k =l phi @j2 0, @istore %k2
	%k6 =l mul %k, 6
	%sh =l sub 18, %k6
	%shw =w copy %sh
	%vs =l shr %v, %shw
	%idx =l and %vs, 63
	%ch0 =l add %idx, 65
	%t1w =w csgel %idx, 26
	%t1 =l extuw %t1w
	%m1 =l mul %t1, 6
	%ch1 =l add %ch0, %m1
	%t2w =w csgel %idx, 52
	%t2 =l extuw %t2w
	%m2 =l mul %t2, 75
	%ch2 =l sub %ch1, %m2
	%t3w =w csgel %idx, 62
	%t3 =l extuw %t3w
	%m3 =l mul %t3, 15
	%ch3 =l sub %ch2, %m3
	%t4w =w csgel %idx, 63
	%t4 =l extuw %t4w
	%m4 =l mul %t4, 3
	%ch =l add %ch3, %m4
	%cp =w cslel %k, %rem
	jnz %cp, @istore, @pad
@pad
	jmp @istore
@istore
	%chf =l phi @inner %ch, @pad 61
	%q0 =l add %out, %o
	%q =l add %q0, %k
	storeb %chf, %q
	%k2 =l add %k, 1
	%ck =w csltl %k2, 4
	jnz %ck, @inner, @next
@next
	%i2 =l add %i, 3
	%o2 =l add %o, 4
	jmp @loop
@done
	ret %o
}
