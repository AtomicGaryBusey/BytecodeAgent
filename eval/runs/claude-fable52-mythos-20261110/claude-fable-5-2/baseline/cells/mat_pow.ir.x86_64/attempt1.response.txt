export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%n1 =l add %n, 1
	%i =l copy 0
@init
	%r =l urem %i, %n1
	%c =w ceql %r, 0
	%v =l extuw %c
	%o =l shl %i, 3
	%p =l add %out, %o
	storel %v, %p
	%i =l add %i, 1
	%c =w csltl %i, %nn
	jnz %c, @init, @initdone
@initdone
	%s =l copy 0
@kloop
	%c =w csltl %s, %k
	jnz %c, @mul, @clear
@mul
	%i =l copy 0
@iloop
	%j =l copy 0
@jloop
	%acc =l copy 0
	%t =l copy 0
@tloop
	%x1 =l mul %i, %n
	%x1 =l add %x1, %t
	%x1 =l shl %x1, 3
	%p =l add %out, %x1
	%x =l loadl %p
	%y1 =l mul %t, %n
	%y1 =l add %y1, %j
	%y1 =l shl %y1, 3
	%p =l add %a, %y1
	%y =l loadl %p
	%m =l mul %x, %y
	%acc =l add %acc, %m
	%t =l add %t, 1
	%c =w csltl %t, %n
	jnz %c, @tloop, @tdone
@tdone
	%z1 =l mul %i, %n
	%z1 =l add %z1, %j
	%z1 =l shl %z1, 3
	%p =l add %tmp, %z1
	storel %acc, %p
	%j =l add %j, 1
	%c =w csltl %j, %n
	jnz %c, @jloop, @jdone
@jdone
	%i =l add %i, 1
	%c =w csltl %i, %n
	jnz %c, @iloop, @copy
@copy
	%i =l copy 0
@cloop
	%o =l shl %i, 3
	%p =l add %tmp, %o
	%v =l loadl %p
	%q =l add %out, %o
	storel %v, %q
	%i =l add %i, 1
	%c =w csltl %i, %nn
	jnz %c, @cloop, @cdone
@cdone
	%s =l add %s, 1
	jmp @kloop
@clear
	%n2 =l shl %nn, 1
	%i =l copy 0
@zloop
	%o =l shl %i, 3
	%p =l add %tmp, %o
	storel 0, %p
	%i =l add %i, 1
	%c =w csltl %i, %n2
	jnz %c, @zloop, @done
@done
	ret 0
}
