export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @hdr, @done
@hdr
	%p =l add %src, %i
	%hw =w loadsb %p
	%h =l extsw %hw
	%i =l add %i, 1
	%c =w ceql %h, -128
	jnz %c, @loop, @chk
@chk
	%rem =l sub %n, %i
	%c =w csltl %h, 0
	jnz %c, @rep, @lit
@lit
	%cnt =l add %h, 1
	%c =w csltl %rem, %cnt
	jnz %c, @err2, @litloop
@litloop
	%c =w csgtl %cnt, 0
	jnz %c, @litbody, @loop
@litbody
	%c =w csltl %o, %cap
	jnz %c, @litw, @err1
@litw
	%p =l add %src, %i
	%b =w loadub %p
	%q =l add %out, %o
	storeb %b, %q
	%i =l add %i, 1
	%o =l add %o, 1
	%cnt =l sub %cnt, 1
	jmp @litloop
@rep
	%c =w csltl %rem, 1
	jnz %c, @err2, @rep2
@rep2
	%p =l add %src, %i
	%b =w loadub %p
	%i =l add %i, 1
	%cnt =l sub 1, %h
@reploop
	%c =w csgtl %cnt, 0
	jnz %c, @repbody, @loop
@repbody
	%c =w csltl %o, %cap
	jnz %c, @repw, @err1
@repw
	%q =l add %out, %o
	storeb %b, %q
	%o =l add %o, 1
	%cnt =l sub %cnt, 1
	jmp @reploop
@err1
	ret -1
@err2
	ret -2
@done
	ret %o
}
