qsort_pairs:
push rbx
push r12
push r13
mov rbx, rdi
mov r12, rsi
.Lloop:
cmp r12, 1
jle .Ldone
mov rax, r12
dec rax
shl rax, 4
add rax, rbx
mov r8, qword ptr [rax]
mov r9, qword ptr [rax + 8]
mov rcx, rbx
mov rdx, rbx
.Lpart:
cmp rdx, rax
jae .Lpartdone
mov r10, qword ptr [rdx]
mov r11, qword ptr [rdx + 8]
cmp r10, r8
jl .Lswap
jg .Lnext
cmp r11, r9
jge .Lnext
.Lswap:
mov rsi, qword ptr [rcx]
mov rdi, qword ptr [rcx + 8]
mov qword ptr [rcx], r10
mov qword ptr [rcx + 8], r11
mov qword ptr [rdx], rsi
mov qword ptr [rdx + 8], rdi
add rcx, 16
.Lnext:
add rdx, 16
jmp .Lpart
.Lpartdone:
mov rsi, qword ptr [rcx]
mov rdi, qword ptr [rcx + 8]
mov qword ptr [rcx], r8
mov qword ptr [rcx + 8], r9
mov qword ptr [rax], rsi
mov qword ptr [rax + 8], rdi
mov rsi, rcx
sub rsi, rbx
shr rsi, 4
mov r13, rsi
mov rdi, rbx
call qsort_pairs
lea rax, [r13 + 1]
sub r12, rax
shl rax, 4
add rbx, rax
jmp .Lloop
.Ldone:
xor eax, eax
pop r13
pop r12
pop rbx
ret
