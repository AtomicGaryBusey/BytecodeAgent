bubble_sort:
    cmp rsi, 2
    jl .Ldone
    lea rcx, [rsi - 1]          # rcx = number of comparisons in this pass
.Louter:
    xor edx, edx                # rdx = index
.Linner:
    mov rax, qword ptr [rdi + rdx*8]
    mov r8, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r8
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r8
    mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
    inc rdx
    cmp rdx, rcx
    jl .Linner
    dec rcx
    jnz .Louter
.Ldone:
    xor eax, eax
    ret
