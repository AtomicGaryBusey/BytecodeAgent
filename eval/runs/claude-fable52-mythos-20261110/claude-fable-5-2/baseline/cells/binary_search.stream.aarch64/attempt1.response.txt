binary_search:
    mov x9, #0              // lo
    mov x10, x1             // hi (exclusive)
.Lloop:
    cmp x9, x10
    b.ge .Lnotfound
    add x11, x9, x10
    lsr x11, x11, #1        // mid
    ldr x12, [x0, x11, lsl #3]
    cmp x12, x2
    b.eq .Lfound
    b.lt .Lless
    mov x10, x11            // hi = mid
    b .Lloop
.Lless:
    add x9, x11, #1         // lo = mid + 1
    b .Lloop
.Lfound:
    mov x0, x11
    ret
.Lnotfound:
    mov x0, #-1
    ret
