export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	%odd0 =l and %y0, 1
@loop
	%i =l phi @start 1, @body %i2
	%min =l phi @start %y0, @body %min2
	%max =l phi @start %y0, @body %max2
	%sum =l phi @start %y0, @body %sum2
	%cnt =l phi @start %odd0, @body %cnt2
	%c =w csltl %i, %n
	jnz %c, @call, @done
@call
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%clt =w csltl %y, %min
	jnz %clt, @setmin, @keepmin
@setmin
	jmp @minjoin
@keepmin
	jmp @minjoin
@minjoin
	%min2 =l phi @setmin %y, @keepmin %min
	%cgt =w csgtl %y, %max
	jnz %cgt, @setmax, @keepmax
@setmax
	jmp @body
@keepmax
	jmp @body
@body
	%max2 =l phi @setmax %y, @keepmax %max
	%sum2 =l add %sum, %y
	%odd =l and %y, 1
	%cnt2 =l add %cnt, %odd
	%i2 =l add %i, 1
	jmp @loop
@done
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
