export function l $isqrt(l %n) {
@start
	%x =l copy %n
	%res =l copy 0
	%bit =l copy 4611686018427387904
@shrink
	%c =w cugtl %bit, %x
	jnz %c, @shrinkbody, @loop
@shrinkbody
	%bit =l shr %bit, 2
	jmp @shrink
@loop
	%z =w ceql %bit, 0
	jnz %z, @done, @body
@body
	%t =l add %res, %bit
	%res =l shr %res, 1
	%ge =w cugel %x, %t
	jnz %ge, @take, @next
@take
	%x =l sub %x, %t
	%res =l add %res, %bit
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
