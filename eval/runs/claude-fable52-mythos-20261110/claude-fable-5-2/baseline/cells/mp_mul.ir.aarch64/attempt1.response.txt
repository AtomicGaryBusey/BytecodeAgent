export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zloop
@zloop
	%z =l phi @start 0, @zbody %z1
	%zc =w csltl %z, %n
	jnz %zc, @zbody, @outer
@zbody
	%zo =l shl %z, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%z1 =l add %z, 1
	jmp @zloop
@outer
	%i =l phi @zloop 0, @iend %i1
	%ic =w csltl %i, %na
	jnz %ic, @ibody, @trim
@ibody
	%ao =l shl %i, 3
	%ap =l add %a, %ao
	%x =l loadl %ap
	%x0 =l and %x, 4294967295
	%x1 =l shr %x, 32
	%rb =l add %r, %ao
	jmp @inner
@inner
	%j =l phi @ibody 0, @jbody %j1
	%carry =l phi @ibody 0, @jbody %hi3
	%jc =w csltl %j, %nb
	jnz %jc, @jbody, @iend
@jbody
	%bo =l shl %j, 3
	%bp =l add %b, %bo
	%y =l loadl %bp
	%y0 =l and %y, 4294967295
	%y1 =l shr %y, 32
	%p00 =l mul %x0, %y0
	%p01 =l mul %x0, %y1
	%p10 =l mul %x1, %y0
	%p11 =l mul %x1, %y1
	%p00h =l shr %p00, 32
	%p01l =l and %p01, 4294967295
	%p10l =l and %p10, 4294967295
	%mid0 =l add %p00h, %p01l
	%mid =l add %mid0, %p10l
	%p00l =l and %p00, 4294967295
	%midl =l shl %mid, 32
	%lo =l or %p00l, %midl
	%p01h =l shr %p01, 32
	%p10h =l shr %p10, 32
	%midh =l shr %mid, 32
	%hi0 =l add %p11, %p01h
	%hi1 =l add %hi0, %p10h
	%hi =l add %hi1, %midh
	%rp =l add %rb, %bo
	%rv =l loadl %rp
	%lo2 =l add %lo, %rv
	%c1 =w cultl %lo2, %lo
	%c1l =l extuw %c1
	%hi2 =l add %hi, %c1l
	%lo3 =l add %lo2, %carry
	%c2 =w cultl %lo3, %lo2
	%c2l =l extuw %c2
	%hi3 =l add %hi2, %c2l
	storel %lo3, %rp
	%j1 =l add %j, 1
	jmp @inner
@iend
	%no =l shl %nb, 3
	%cp =l add %rb, %no
	storel %carry, %cp
	%i1 =l add %i, 1
	jmp @outer
@trim
	%k =l phi @outer %n, @tdec %k1
	%kc =w csgtl %k, 0
	jnz %kc, @tchk, @done
@tchk
	%k1 =l sub %k, 1
	%to =l shl %k1, 3
	%tp =l add %r, %to
	%tv =l loadl %tp
	%tz =w ceql %tv, 0
	jnz %tz, @tdec, @done
@tdec
	jmp @trim
@done
	ret %k
}
