binary_search:
    xor ecx, ecx                # lo = 0
    mov r8, rsi                 # hi = n
.Lloop:
    cmp rcx, r8
    jge .Lnotfound
    lea rax, [rcx + r8]
    shr rax, 1                  # mid = (lo + hi) / 2
    mov r9, qword ptr [rdi + rax*8]
    cmp r9, rdx
    je .Ldone
    jl .Lright
    mov r8, rax                 # hi = mid
    jmp .Lloop
.Lright:
    lea rcx, [rax + 1]          # lo = mid + 1
    jmp .Lloop
.Lnotfound:
    mov rax, -1
.Ldone:
    ret
