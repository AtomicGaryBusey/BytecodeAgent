export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
@loop
	%c =w csltl %i, %j
	jnz %c, @body, @done
@body
	%io =l shl %i, 3
	%jo =l shl %j, 3
	%pi =l add %a, %io
	%pj =l add %a, %jo
	%x =l loadl %pi
	%y =l loadl %pj
	storel %y, %pi
	storel %x, %pj
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @loop
@done
	ret 0
}
