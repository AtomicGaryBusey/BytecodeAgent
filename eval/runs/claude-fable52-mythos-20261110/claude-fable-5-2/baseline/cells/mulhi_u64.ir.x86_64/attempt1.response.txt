export function l $mulhi_u64(l %a, l %b) {
@start
	%a0 =l and %a, 4294967295
	%a1 =l shr %a, 32
	%b0 =l and %b, 4294967295
	%b1 =l shr %b, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%c0 =l shr %p00, 32
	%t1 =l add %p10, %c0
	%t1lo =l and %t1, 4294967295
	%t1hi =l shr %t1, 32
	%t2 =l add %p01, %t1lo
	%t2hi =l shr %t2, 32
	%r0 =l add %p11, %t1hi
	%r =l add %r0, %t2hi
	ret %r
}
