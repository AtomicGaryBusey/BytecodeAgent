export function l $max_byte(l %p, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i1
	%m =l phi @start 0, @next %m1
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%a =l add %p, %i
	%b =l loadub %a
	%g =w cugtl %b, %m
	jnz %g, @upd, @keep
@upd
	jmp @next
@keep
	jmp @next
@next
	%m1 =l phi @upd %b, @keep %m
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %m
}
