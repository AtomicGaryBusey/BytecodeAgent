export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @next %i2
	%o =l phi @start 0, @next %o2
	%rem =l sub %n, %i
	%c =w csgtl %rem, 0
	jnz %c, @body, @done
@body
	%p =l add %src, %i
	%b0 =l loadub %p
	%v0 =l shl %b0, 16
	%c1 =w csgtl %rem, 1
	jnz %c1, @two, @merge
@two
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	%s1 =l shl %b1, 8
	%v1 =l or %v0, %s1
	%c2 =w csgtl %rem, 2
	jnz %c2, @three, @merge
@three
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @merge
@merge
	%v =l phi @body %v0, @two %v1, @three %v2
	%k =l phi @body 2, @two 3, @three 4
	jmp @inner
@inner
	%j =l phi @merge 0, @store %j2
	%sh =l phi @merge 18, @store %sh2
	%q =l add %o, %j
	%op =l add %out, %q
	%cj =w csltl %j, %k
	jnz %cj, @mapc, @pad
@mapc
	%shw =w copy %sh
	%t =l shr %v, %shw
	%x =l and %t, 63
	%g1 =l csgel %x, 26
	%g2 =l csgel %x, 52
	%g3 =l csgel %x, 62
	%g4 =l csgel %x, 63
	%m1 =l mul %g1, 6
	%m2 =l mul %g2, 75
	%m3 =l mul %g3, 15
	%m4 =l mul %g4, 3
	%a0 =l add %x, 65
	%a1 =l add %a0, %m1
	%a2 =l sub %a1, %m2
	%a3 =l sub %a2, %m3
	%ch =l add %a3, %m4
	jmp @store
@pad
	jmp @store
@store
	%cc =l phi @mapc %ch, @pad 61
	%ccw =w copy %cc
	storeb %ccw, %op
	%j2 =l add %j, 1
	%sh2 =l sub %sh, 6
	%cl =w csltl %j2, 4
	jnz %cl, @inner, @next
@next
	%i2 =l add %i, 3
	%o2 =l add %o, 4
	jmp @loop
@done
	ret %o
}
