packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jge .Ldone
    movsx eax, byte ptr [rdi + r8]
    inc r8
    cmp eax, -128
    je .Lloop
    test eax, eax
    js .Lrep
    inc eax
    lea r10, [r8 + rax]
    cmp r10, rsi
    jg .Lerr2
.Llit:
    cmp r9, rcx
    jge .Lerr1
    mov r11b, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], r11b
    inc r8
    inc r9
    dec eax
    jnz .Llit
    jmp .Lloop
.Lrep:
    cmp r8, rsi
    jge .Lerr2
    neg eax
    inc eax
    mov r11b, byte ptr [rdi + r8]
    inc r8
.Lrl:
    cmp r9, rcx
    jge .Lerr1
    mov byte ptr [rdx + r9], r11b
    inc r9
    dec eax
    jnz .Lrl
    jmp .Lloop
.Ldone:
    mov rax, r9
    ret
.Lerr1:
    mov rax, -1
    ret
.Lerr2:
    mov rax, -2
    ret
