export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%p =l add %src, %i
	%b =w loadub %p
	%r =l copy 1
@runloop
	%k =l add %i, %r
	%c =w csltl %k, %n
	jnz %c, @run2, @runend
@run2
	%c =w csltl %r, 128
	jnz %c, @run3, @runend
@run3
	%q =l add %src, %k
	%b2 =w loadub %q
	%c =w ceqw %b2, %b
	jnz %c, @runinc, @runend
@runinc
	%r =l add %r, 1
	jmp @runloop
@runend
	%c =w csgel %r, 3
	jnz %c, @rep, @lit
@rep
	%t =l add %o, 2
	%c =w csgtl %t, %cap
	jnz %c, @fail, @repw
@repw
	%rw =w copy %r
	%h =w sub 257, %rw
	%d =l add %out, %o
	storeb %h, %d
	%d1 =l add %d, 1
	storeb %b, %d1
	%o =l copy %t
	%i =l add %i, %r
	jmp @loop
@lit
	%j =l copy %i
@litloop
	%c =w csltl %j, %n
	jnz %c, @lit2, @litend
@lit2
	%len =l sub %j, %i
	%c =w csltl %len, 128
	jnz %c, @lit3, @litend
@lit3
	%j2 =l add %j, 2
	%c =w csltl %j2, %n
	jnz %c, @lit4, @litinc
@lit4
	%pj =l add %src, %j
	%x0 =w loadub %pj
	%pj1 =l add %pj, 1
	%x1 =w loadub %pj1
	%c =w ceqw %x0, %x1
	jnz %c, @lit5, @litinc
@lit5
	%pj2 =l add %pj, 2
	%x2 =w loadub %pj2
	%c =w ceqw %x0, %x2
	jnz %c, @litend, @litinc
@litinc
	%j =l add %j, 1
	jmp @litloop
@litend
	%len =l sub %j, %i
	%t =l add %o, 1
	%t =l add %t, %len
	%c =w csgtl %t, %cap
	jnz %c, @fail, @litw
@litw
	%lw =w copy %len
	%h =w sub %lw, 1
	%d =l add %out, %o
	storeb %h, %d
	%d =l add %d, 1
	%s =l add %src, %i
	%m =l copy 0
@cpy
	%c =w csltl %m, %len
	jnz %c, @cpyb, @cpyend
@cpyb
	%sp =l add %s, %m
	%v =w loadub %sp
	%dp =l add %d, %m
	storeb %v, %dp
	%m =l add %m, 1
	jmp @cpy
@cpyend
	%o =l copy %t
	%i =l copy %j
	jmp @loop
@fail
	ret -1
@done
	ret %o
}
