export function l $crc32(l %p, l %n) {
@start
	%c0 =w copy 4294967295
	%i0 =l copy 0
@loop
	%c =w phi @start %c0, @bitend %c3
	%i =l phi @start %i0, @bitend %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %c, %b
	%k0 =w copy 0
@bit
	%c1 =w phi @body %cx, @bit %c3
	%k =w phi @body %k0, @bit %k1
	%low =w and %c1, 1
	%mask =w neg %low
	%poly =w and %mask, 3988292384
	%sh =w shr %c1, 1
	%c3 =w xor %sh, %poly
	%k1 =w add %k, 1
	%kc =w csltw %k1, 8
	jnz %kc, @bit, @bitend
@bitend
	%i1 =l add %i, 1
	jmp @loop
@done
	%r =w xor %c, 4294967295
	%rl =l extuw %r
	ret %rl
}
