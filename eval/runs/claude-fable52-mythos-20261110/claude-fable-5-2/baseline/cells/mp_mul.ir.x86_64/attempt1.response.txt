export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%total =l add %na, %nb
	%k =l copy 0
@zloop
	%c =w csltl %k, %total
	jnz %c, @zbody, @zdone
@zbody
	%off =l shl %k, 3
	%p =l add %r, %off
	storel 0, %p
	%k =l add %k, 1
	jmp @zloop
@zdone
	%i =l copy 0
@oloop
	%c =w csltl %i, %na
	jnz %c, @obody, @odone
@obody
	%off =l shl %i, 3
	%p =l add %a, %off
	%ai =l loadl %p
	%x0 =l and %ai, 4294967295
	%x1 =l shr %ai, 32
	%carry =l copy 0
	%j =l copy 0
@iloop
	%c =w csltl %j, %nb
	jnz %c, @ibody, @idone
@ibody
	%off =l shl %j, 3
	%p =l add %b, %off
	%bj =l loadl %p
	%y0 =l and %bj, 4294967295
	%y1 =l shr %bj, 32
	%ll =l mul %x0, %y0
	%lh =l mul %x0, %y1
	%hl =l mul %x1, %y0
	%hh =l mul %x1, %y1
	%t =l shr %ll, 32
	%m1 =l and %lh, 4294967295
	%t =l add %t, %m1
	%m2 =l and %hl, 4294967295
	%t =l add %t, %m2
	%lo =l and %ll, 4294967295
	%ts =l shl %t, 32
	%lo =l or %lo, %ts
	%hi =l shr %lh, 32
	%hi =l add %hi, %hh
	%u =l shr %hl, 32
	%hi =l add %hi, %u
	%u =l shr %t, 32
	%hi =l add %hi, %u
	%ij =l add %i, %j
	%off =l shl %ij, 3
	%p =l add %r, %off
	%rv =l loadl %p
	%s =l add %lo, %rv
	%c1 =w cultl %s, %lo
	%s2 =l add %s, %carry
	%c2 =w cultl %s2, %s
	storel %s2, %p
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%carry =l add %hi, %c1l
	%carry =l add %carry, %c2l
	%j =l add %j, 1
	jmp @iloop
@idone
	%ij =l add %i, %nb
	%off =l shl %ij, 3
	%p =l add %r, %off
	storel %carry, %p
	%i =l add %i, 1
	jmp @oloop
@odone
	%k =l copy %total
@cloop
	%c =w csgtl %k, 0
	jnz %c, @ccheck, @fin
@ccheck
	%k1 =l sub %k, 1
	%off =l shl %k1, 3
	%p =l add %r, %off
	%v =l loadl %p
	%c =w ceql %v, 0
	jnz %c, @cdec, @fin
@cdec
	%k =l copy %k1
	jmp @cloop
@fin
	ret %k
}
