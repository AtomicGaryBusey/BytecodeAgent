export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%off =l shl %mid, 3
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%lo =l add %mid, 1
	jmp @loop
@left
	%hi =l copy %mid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
