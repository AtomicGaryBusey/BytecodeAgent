mp_mul:
push rbx
push r12
push r13
mov r9, rdx
lea r10, [rsi + rcx]
xor eax, eax
xor r11d, r11d
.Lzero:
cmp r11, r10
jae .Lzdone
mov qword ptr [r8 + r11*8], rax
inc r11
jmp .Lzero
.Lzdone:
test rsi, rsi
jz .Lcount
test rcx, rcx
jz .Lcount
xor ebx, ebx
.Louter:
lea r12, [r8 + rbx*8]
xor r13d, r13d
xor r11d, r11d
.Linner:
mov rax, qword ptr [rdi + rbx*8]
mul qword ptr [r9 + r11*8]
add rax, r13
adc rdx, 0
add rax, qword ptr [r12 + r11*8]
adc rdx, 0
mov qword ptr [r12 + r11*8], rax
mov r13, rdx
inc r11
cmp r11, rcx
jb .Linner
mov qword ptr [r12 + rcx*8], r13
inc rbx
cmp rbx, rsi
jb .Louter
.Lcount:
mov rax, r10
.Lcloop:
test rax, rax
jz .Ldone
cmp qword ptr [r8 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lcloop
.Ldone:
pop r13
pop r12
pop rbx
ret
