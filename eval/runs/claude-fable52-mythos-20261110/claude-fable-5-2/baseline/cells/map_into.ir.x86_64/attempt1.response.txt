export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%cnt =l phi @start 0, @body %cnt2
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l shl %i, 3
	%sp =l add %src, %off
	%dp =l add %dst, %off
	%x =l loadl %sp
	%r =l call %f(l %x)
	storel %r, %dp
	%lt =w csltl %r, %x
	%ltl =l extuw %lt
	%cnt2 =l add %cnt, %ltl
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
