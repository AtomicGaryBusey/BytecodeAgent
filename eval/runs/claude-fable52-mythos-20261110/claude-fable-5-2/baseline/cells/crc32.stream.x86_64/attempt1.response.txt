crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
.Lbyte:
    movzx ecx, byte ptr [rdi]
    xor eax, ecx
    mov ecx, 8
.Lbit:
    mov edx, eax
    shr eax, 1
    and edx, 1
    neg edx
    and edx, 0xEDB88320
    xor eax, edx
    dec ecx
    jnz .Lbit
    inc rdi
    dec rsi
    jnz .Lbyte
.Ldone:
    not eax
    ret
