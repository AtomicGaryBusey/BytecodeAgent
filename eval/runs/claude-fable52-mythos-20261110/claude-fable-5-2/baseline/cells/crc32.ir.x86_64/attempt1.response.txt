export function l $crc32(l %p, l %n) {
@start
	%c0 =w copy 4294967295
@loop
	%c =w phi @start %c0, @bodyend %c3
	%i =l phi @start 0, @bodyend %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %c, %b
@bit
	%c1 =w phi @body %cx, @bit %c2
	%k =w phi @body 0, @bit %k1
	%low =w and %c1, 1
	%mask =w neg %low
	%poly =w and %mask, 3988292384
	%sh =w shr %c1, 1
	%c2 =w xor %sh, %poly
	%k1 =w add %k, 1
	%kc =w csltw %k1, 8
	jnz %kc, @bit, @bodyend
@bodyend
	%c3 =w copy %c2
	%i1 =l add %i, 1
	jmp @loop
@done
	%f =w xor %c, 4294967295
	%r =l extuw %f
	ret %r
}
