export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
@loop
	%c =w csltl %pc, %len
	jnz %c, @fetch, @done
@fetch
	%o =l mul %pc, 8
	%p =l add %code, %o
	%op =l loadl %p
	%p8 =l add %p, 8
	%do =l mul %d, 8
	%tp =l add %stack, %do
	%p1 =l sub %tp, 8
	%p2 =l sub %tp, 16
	%c =w ceql %op, 0
	jnz %c, @done, @n1
@n1
	%c =w ceql %op, 1
	jnz %c, @push, @n10
@n10
	%c =w ceql %op, 10
	jnz %c, @jz, @n11
@n11
	%c =w ceql %op, 11
	jnz %c, @jmp, @n5
@push
	%v =l loadl %p8
	storel %v, %tp
	%d =l add %d, 1
	%pc =l add %pc, 2
	jmp @loop
@jz
	%v =l loadl %p8
	%x =l loadl %p1
	%d =l sub %d, 1
	%c =w ceql %x, 0
	jnz %c, @jztaken, @jznot
@jztaken
	%pc =l copy %v
	jmp @loop
@jznot
	%pc =l add %pc, 2
	jmp @loop
@jmp
	%v =l loadl %p8
	%pc =l copy %v
	jmp @loop
@n5
	%c =w ceql %op, 5
	jnz %c, @dup, @n6
@dup
	%b =l loadl %p1
	storel %b, %tp
	%d =l add %d, 1
	jmp @next
@n6
	%c =w ceql %op, 6
	jnz %c, @swap, @n7
@swap
	%b =l loadl %p1
	%a =l loadl %p2
	storel %a, %p1
	storel %b, %p2
	jmp @next
@n7
	%c =w ceql %op, 7
	jnz %c, @drop, @n8
@drop
	%d =l sub %d, 1
	jmp @next
@n8
	%c =w ceql %op, 8
	jnz %c, @over, @bin
@over
	%a =l loadl %p2
	storel %a, %tp
	%d =l add %d, 1
	jmp @next
@bin
	%b =l loadl %p1
	%a =l loadl %p2
	%d =l sub %d, 1
	%c =w ceql %op, 2
	jnz %c, @add, @b3
@add
	%r =l add %a, %b
	jmp @binst
@b3
	%c =w ceql %op, 3
	jnz %c, @sub, @b4
@sub
	%r =l sub %a, %b
	jmp @binst
@b4
	%c =w ceql %op, 4
	jnz %c, @mul, @lt
@mul
	%r =l mul %a, %b
	jmp @binst
@lt
	%cw =w csltl %a, %b
	%r =l extuw %cw
	jmp @binst
@binst
	storel %r, %p2
	jmp @next
@next
	%pc =l add %pc, 1
	jmp @loop
@done
	%i =l copy %d
@zloop
	%c =w csltl %i, %cap
	jnz %c, @zbody, @fin
@zbody
	%zo =l mul %i, 8
	%zp =l add %stack, %zo
	storel 0, %zp
	%i =l add %i, 1
	jmp @zloop
@fin
	%c =w ceql %d, 0
	jnz %c, @empty, @top
@empty
	ret 0
@top
	%fo =l mul %d, 8
	%fp =l add %stack, %fo
	%fp =l sub %fp, 8
	%res =l loadl %fp
	ret %res
}
