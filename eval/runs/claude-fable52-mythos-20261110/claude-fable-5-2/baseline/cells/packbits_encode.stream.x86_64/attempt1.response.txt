xor r9d, r9d
.Lmain:

jge .Ldone
movzx eax, byte ptr [rdi + r8]
mov r10d, 1
.Lrun:

cmp r11, rsi
jge .Lrund
cmp r10, 128
jge .Lrund
cmp al, byte ptr [rdi + r11]
jne .Lrund
inc r10
jmp .Lrun
.Lrund:

jl .Llit
lea r11, [r9 + 2]
cmp r11, rcx
jg .Lfail
mov r11d, 1
sub r11, r10
mov byte ptr [rdx + r9], r11b
mov byte ptr [rdx + r9 + 1], al

.Llit:

.Lscan:

jge .Lscand
mov rax, r11
sub rax, r8
cmp rax, 128
jge .Lscand

cmp rax, rsi
jge .Lnext
mov al, byte ptr [rdi + r11]
cmp al, byte ptr [rdi + r11 + 1]

cmp al, byte ptr [rdi + r11 + 2]
je .Lscand
.Lnext:

jmp .Lscan
.Lscand:

lea rax, [r9 + r10 + 1]
cmp rax, rcx
jg .Lfail

.Lcopy:

mov byte ptr [rdx + r9], al
inc r8
inc r9
cmp r8, r11
jl .Lcopy
jmp .Lmain
.Ldone:

ret
.Lfail:
