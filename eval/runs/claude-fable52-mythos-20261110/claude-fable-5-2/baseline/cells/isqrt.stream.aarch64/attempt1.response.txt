isqrt:
    mov x1, #0                  // result
    mov x2, #0x4000000000000000 // bit
.Lloop:
    add x3, x1, x2              // result + bit
    lsr x1, x1, #1
    cmp x0, x3
    b.lo .Lskip
    sub x0, x0, x3
    add x1, x1, x2
.Lskip:
    lsr x2, x2, #2
    cbnz x2, .Lloop
    mov x0, x1
    ret
