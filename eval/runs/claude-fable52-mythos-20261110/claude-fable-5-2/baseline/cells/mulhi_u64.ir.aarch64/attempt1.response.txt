export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32
	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%llh =l shr %ll, 32
	%t1 =l add %hl, %llh
	%t1l =l and %t1, 4294967295
	%t1h =l shr %t1, 32
	%t2 =l add %lh, %t1l
	%t2h =l shr %t2, 32
	%r1 =l add %hh, %t1h
	%r =l add %r1, %t2h
	ret %r
}
