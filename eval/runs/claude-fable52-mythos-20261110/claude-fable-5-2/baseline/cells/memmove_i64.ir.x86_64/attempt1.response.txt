export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%back =w cultl %src, %dst
	jnz %back, @binit, @finit
@finit
	%i =l copy 0
	jmp @fcond
@fcond
	%c =w csltl %i, %n
	jnz %c, @fbody, @done
@fbody
	%off =l shl %i, 3
	%s =l add %src, %off
	%d =l add %dst, %off
	%v =l loadl %s
	storel %v, %d
	%i =l add %i, 1
	jmp @fcond
@binit
	%j =l copy %n
	jmp @bcond
@bcond
	%c2 =w csgtl %j, 0
	jnz %c2, @bbody, @done
@bbody
	%j =l sub %j, 1
	%off2 =l shl %j, 3
	%s2 =l add %src, %off2
	%d2 =l add %dst, %off2
	%v2 =l loadl %s2
	storel %v2, %d2
	jmp @bcond
@done
	ret 0
}
