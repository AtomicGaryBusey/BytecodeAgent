    mov x6, #1                  // r

    b.ge .Lrundone
    add x8, x4, x6
    cmp x8, x1
    b.ge .Lrundone
    ldrb w9, [x0, x8]

    add x8, x5, #2

    strb w7, [x2, x10]

    add x4, x4, x6

    mov x8, x4                  // j
.Lscan:

    b.ge .Lscandone
    sub x9, x8, x4
    cmp x9, #128
    b.ge .Lscandone
    add x10, x8, #2
    cmp x10, x1
    b.ge .Lnext
    ldrb w11, [x0, x8]
    ldrb w12, [x0, x10]
    cmp w11, w12
    b.ne .Lnext
    sub x10, x10, #1
    ldrb w12, [x0, x10]
    cmp w11, w12
    b.eq .Lscandone
.Lnext:
    add x8, x8, #1
    b .Lscan
.Lscandone:
    sub x9, x8, x4              // L
    add x10, x5, x9
    add x10, x10, #1
    cmp x10, x3
    b.gt .Lfail
    sub w11, w9, #1
    strb w11, [x2, x5]
    add x5, x5, #1
.Lcopy:

    strb w11, [x2, x5]
    add x4, x4, #1
    add x5, x5, #1
    cmp x4, x8
    b.lt .Lcopy
    b .Lmain
.Ldone:
    mov x0, x5
    ret
.Lfail:
    mov x0, #-1
    ret
