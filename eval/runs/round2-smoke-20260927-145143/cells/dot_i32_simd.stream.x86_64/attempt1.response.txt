dot_i32_simd:
    xor eax, eax
    pxor xmm0, xmm0
    pxor xmm5, xmm5
    xor ecx, ecx
    mov r8, rdx
    and r8, -4
.Lvec:
    cmp rcx, r8
    jge .Lreduce
    pmovsxdq xmm1, qword ptr [rdi + rcx*4]
    pmovsxdq xmm2, qword ptr [rsi + rcx*4]
    pmuldq xmm1, xmm2
    paddq xmm0, xmm1
    pmovsxdq xmm3, qword ptr [rdi + rcx*4 + 8]
    pmovsxdq xmm4, qword ptr [rsi + rcx*4 + 8]
    pmuldq xmm3, xmm4
    paddq xmm5, xmm3
    add rcx, 4
    jmp .Lvec
.Lreduce:
    paddq xmm0, xmm5
    pshufd xmm1, xmm0, 0x4e
    paddq xmm0, xmm1
    movq rax, xmm0
.Lscalar:
    cmp rcx, rdx
    jge .Ldone
    movsxd r9, dword ptr [rdi + rcx*4]
    movsxd r10, dword ptr [rsi + rcx*4]
    imul r9, r10
    add rax, r9
    inc rcx
    jmp .Lscalar
.Ldone:
    ret
