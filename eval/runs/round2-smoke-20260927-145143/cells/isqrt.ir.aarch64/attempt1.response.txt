export function l $isqrt(l %n0) {
@start
	%n =l copy %n0
	%res =l copy 0
	%bit =l copy 4611686018427387904
@shrink
	%c =w cugtl %bit, %n
	jnz %c, @doshr, @loop
@doshr
	%bit =l shr %bit, 2
	jmp @shrink
@loop
	%z =w ceql %bit, 0
	jnz %z, @done, @body
@body
	%t =l add %res, %bit
	%ge =w cugel %n, %t
	jnz %ge, @take, @skip
@take
	%n =l sub %n, %t
	%h =l shr %res, 1
	%res =l add %h, %bit
	jmp @next
@skip
	%res =l shr %res, 1
@next
	%bit =l shr %bit, 2
	jmp @loop
@done
	ret %res
}
