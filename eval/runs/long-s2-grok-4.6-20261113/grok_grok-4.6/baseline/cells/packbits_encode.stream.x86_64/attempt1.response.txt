packbits_encode:
	mov r10, rdx
	lea r11, [rdx + rcx]
	xor r8, r8
.Lloop:
	cmp r8, rsi
	jge .Ldone
	movzx eax, byte ptr [rdi + r8]
	mov r9, 1
.Lrunloop:
	cmp r9, 128
	jge .Lrunend
	lea rcx, [r8 + r9]
	cmp rcx, rsi
	jge .Lrunend
	cmp al, byte ptr [rdi + rcx]
	jne .Lrunend
	inc r9
	jmp .Lrunloop
.Lrunend:
	cmp r9, 3
	jl .Lliteral
	lea rcx, [rdx + 2]
	cmp rcx, r11
	ja .Lfail
	mov ecx, 1
	sub ecx, r9d
	mov byte ptr [rdx], cl
	mov byte ptr [rdx + 1], al
	add rdx, 2
	add r8, r9
	jmp .Lloop
.Lliteral:
	mov rcx, r8
.Llitloop:
	cmp rcx, rsi
	jge .Llitend
	mov r9, rcx
	sub r9, r8
	cmp r9, 128
	jge .Llitend
	lea r9, [rcx + 2]
	cmp r9, rsi
	jge .Llitinc
	movzx eax, byte ptr [rdi + rcx]
	cmp al, byte ptr [rdi + rcx + 1]
	jne .Llitinc
	cmp al, byte ptr [rdi + rcx + 2]
	je .Llitend
.Llitinc:
	inc rcx
	jmp .Llitloop
.Llitend:
	mov r9, rcx
	sub r9, r8
	lea rax, [rdx + r9 + 1]
	cmp rax, r11
	ja .Lfail
	lea eax, [r9 - 1]
	mov byte ptr [rdx], al
	inc rdx
.Lcopy:
	movzx eax, byte ptr [rdi + r8]
	mov byte ptr [rdx], al
	inc r8
	inc rdx
	dec r9
	jnz .Lcopy
	jmp .Lloop
.Ldone:
	mov rax, rdx
	sub rax, r10
	ret
.Lfail:
	mov rax, -1
	ret
