export function l $qsort_pairs(l %a, l %n) {
@start
	%le1 =w cslel %n, 1
	jnz %le1, @ret0, @work
@ret0
	ret 0
@work
	%nm1 =l sub %n, 1
	%pivoff =l mul %nm1, 16
	%pp =l add %a, %pivoff
	%pk =l loadl %pp
	%pp1 =l add %pp, 8
	%pv =l loadl %pp1
	jmp @ploop
@ploop
	%i =l phi @work 0, @pinc %i_new
	%j =l phi @work 0, @pinc %j_new
	%jlt =w csltl %j, %nm1
	jnz %jlt, @pbody, @pdone
@pbody
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jp1 =l add %jp, 8
	%jv =l loadl %jp1
	%klt =w csltl %jk, %pk
	jnz %klt, @do_swap, @not_klt
@not_klt
	%kgt =w csgtl %jk, %pk
	jnz %kgt, @no_swap, @keys_eq
@keys_eq
	%vlt =w csltl %jv, %pv
	jnz %vlt, @do_swap, @no_swap
@do_swap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ipv =l add %ip, 8
	%iv =l loadl %ipv
	storel %jk, %ip
	storel %jv, %ipv
	storel %ik, %jp
	storel %iv, %jp1
	%i_plus =l add %i, 1
	jmp @pinc
@no_swap
	jmp @pinc
@pinc
	%i_new =l phi @do_swap %i_plus, @no_swap %i
	%j_new =l add %j, 1
	jmp @ploop
@pdone
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ip2v =l add %ip2, 8
	%iv2 =l loadl %ip2v
	storel %pk, %ip2
	storel %pv, %ip2v
	storel %ik2, %pp
	storel %iv2, %pp1
	%r1 =l call $qsort_pairs(l %a, l %i)
	%ip1rec =l add %i, 1
	%roff =l mul %ip1rec, 16
	%ra =l add %a, %roff
	%rn =l sub %n, %ip1rec
	%r2 =l call $qsort_pairs(l %ra, l %rn)
	ret 0
}
