packbits_decode:
    lea rsi, [rdi + rsi]
    mov r11, rdx
    lea r8, [rdx + rcx]
.Lloop:
    cmp rdi, rsi
    jae .Ldone
    movsx eax, byte ptr [rdi]
    inc rdi
    cmp eax, -128
    je .Lloop
    test eax, eax
    js .Lrepeat
    inc eax
    mov ecx, eax
    lea r10, [rdi + rcx]
    cmp r10, rsi
    ja .Lerr_src
    mov r10, r8
    sub r10, rdx
    cmp rcx, r10
    ja .Lovf_lit
.Lcopy:
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec ecx
    jnz .Lcopy
    jmp .Lloop
.Lovf_lit:
    test r10, r10
    jz .Lerr_cap
.Lcopy_ovf:
    mov al, byte ptr [rdi]
    mov byte ptr [rdx], al
    inc rdi
    inc rdx
    dec r10
    jnz .Lcopy_ovf
    jmp .Lerr_cap
.Lrepeat:
    neg eax
    inc eax
    mov ecx, eax
    cmp rdi, rsi
    jae .Lerr_src
    movzx r9d, byte ptr [rdi]
    inc rdi
    mov r10, r8
    sub r10, rdx
    cmp rcx, r10
    ja .Lovf_rep
.Lfill:
    mov byte ptr [rdx], r9b
    inc rdx
    dec ecx
    jnz .Lfill
    jmp .Lloop
.Lovf_rep:
    test r10, r10
    jz .Lerr_cap
.Lfill_ovf:
    mov byte ptr [rdx], r9b
    inc rdx
    dec r10
    jnz .Lfill_ovf
.Lerr_cap:
    mov rax, -1
    ret
.Ldone:
    mov rax, rdx
    sub rax, r11
    ret
.Lerr_src:
    mov rax, -2
    ret
