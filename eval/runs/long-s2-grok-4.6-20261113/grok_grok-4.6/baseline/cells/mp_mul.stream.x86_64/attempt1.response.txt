mp_mul:
	push rbx
	push rbp
	push r12
	push r13
	push r14
	push r15
	mov r12, rdi
	mov r13, rsi
	mov r14, rdx
	mov r15, rcx
	lea rax, [r13 + r15]
	test rax, rax
	jz .Lret0
	xor ebx, ebx
.Lzero:
	mov qword ptr [r8 + rbx*8], 0
	inc rbx
	cmp rbx, rax
	jb .Lzero
	test r13, r13
	jz .Lret0
	test r15, r15
	jz .Lret0
	xor r9d, r9d
.Louter:
	mov rdx, qword ptr [r12 + r9*8]
	xor r10d, r10d
	xor r11d, r11d
.Linner:
	lea rbx, [r9 + r10]
	mulx rbp, rdi, qword ptr [r14 + r10*8]
	add rdi, r11
	adc rbp, 0
	add rdi, qword ptr [r8 + rbx*8]
	adc rbp, 0
	mov qword ptr [r8 + rbx*8], rdi
	mov r11, rbp
	inc r10
	cmp r10, r15
	jb .Linner
	lea rbx, [r9 + r15]
	mov qword ptr [r8 + rbx*8], r11
	inc r9
	cmp r9, r13
	jb .Louter
	lea rax, [r13 + r15]
.Ltrim:
	test rax, rax
	jz .Lret
	cmp qword ptr [r8 + rax*8 - 8], 0
	jne .Lret
	dec rax
	jmp .Ltrim
.Lret0:
	xor eax, eax
.Lret:
	pop r15
	pop r14
	pop r13
	pop r12
	pop rbp
	pop rbx
	ret
