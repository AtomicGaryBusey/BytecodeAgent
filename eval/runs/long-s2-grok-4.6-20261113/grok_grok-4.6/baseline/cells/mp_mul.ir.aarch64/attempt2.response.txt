export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%all1 =l copy -1
	%m32 =l shr %all1, 32
	jmp @zloop
@zloop
	%zi =l phi @start 0, @zbody %zi1
	%zcmp =w csltl %zi, %n
	jnz %zcmp, @zbody, @iloop
@zbody
	%zoff =l shl %zi, 3
	%zaddr =l add %r, %zoff
	storel 0, %zaddr
	%zi1 =l add %zi, 1
	jmp @zloop
@iloop
	%i =l phi @zloop 0, @jafter %i1
	%icmp =w csltl %i, %na
	jnz %icmp, @ibody, @kloop
@ibody
	%aoff =l shl %i, 3
	%aaddr =l add %a, %aoff
	%ai =l loadl %aaddr
	jmp @jloop
@jloop
	%j =l phi @ibody 0, @jbody %j1
	%carry =l phi @ibody 0, @jbody %newcarry
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @jbody, @jafter
@jbody
	%boff =l shl %j, 3
	%baddr =l add %b, %boff
	%bj =l loadl %baddr
	%ij =l add %i, %j
	%roff =l shl %ij, 3
	%raddr =l add %r, %roff
	%rij =l loadl %raddr
	%ai0 =l and %ai, %m32
	%ai1 =l shr %ai, 32
	%bj0 =l and %bj, %m32
	%bj1 =l shr %bj, 32
	%p00 =l mul %ai0, %bj0
	%p01 =l mul %ai0, %bj1
	%p10 =l mul %ai1, %bj0
	%p11 =l mul %ai1, %bj1
	%p00h =l shr %p00, 32
	%p01l =l and %p01, %m32
	%p10l =l and %p10, %m32
	%mid0 =l add %p00h, %p01l
	%mid =l add %mid0, %p10l
	%p00l =l and %p00, %m32
	%midl =l and %mid, %m32
	%mids =l shl %midl, 32
	%lo =l add %p00l, %mids
	%p01h =l shr %p01, 32
	%p10h =l shr %p10, 32
	%midh =l shr %mid, 32
	%hi0 =l add %p11, %p01h
	%hi1 =l add %hi0, %p10h
	%hi =l add %hi1, %midh
	%s =l add %lo, %rij
	%c1 =w cultl %s, %lo
	%c1l =l extuw %c1
	%s2 =l add %s, %carry
	%c2 =w cultl %s2, %s
	%c2l =l extuw %c2
	storel %s2, %raddr
	%hic =l add %hi, %c1l
	%newcarry =l add %hic, %c2l
	%j1 =l add %j, 1
	jmp @jloop
@jafter
	%inb =l add %i, %nb
	%coff =l shl %inb, 3
	%caddr =l add %r, %coff
	storel %carry, %caddr
	%i1 =l add %i, 1
	jmp @iloop
@kloop
	%k =l phi @iloop %n, @kcheck %km1
	%kgt =w csltl 0, %k
	jnz %kgt, @kcheck, @kdone
@kcheck
	%km1 =l sub %k, 1
	%koff =l shl %km1, 3
	%kaddr =l add %r, %koff
	%kval =l loadl %kaddr
	%kzero =w ceql %kval, 0
	jnz %kzero, @kloop, @kdone
@kdone
	ret %k
}
