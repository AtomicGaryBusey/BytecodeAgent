export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%src_end =l add %src, %n
	%out_end =l add %out, %cap
	jmp @loop
@loop
	%sc =l phi @start %src, @not_lit %sc_h, @lit_done %sc_lit, @rep_done %sc_rep
	%oc =l phi @start %out, @not_lit %oc, @lit_done %oc_lit, @rep_done %oc_rep
	%src_left =l sub %src_end, %sc
	%has =w csgtl %src_left, 0
	jnz %has, @have_hdr, @done
@have_hdr
	%h =w loadsb %sc
	%sc_h =l add %sc, 1
	%src_left2 =l sub %src_end, %sc_h
	%is_lit =w csgew %h, 0
	jnz %is_lit, @literal, @not_lit
@not_lit
	%is_nop =w ceqw %h, -128
	jnz %is_nop, @loop, @repeat
@literal
	%lit_cnt_w =w add %h, 1
	%lit_cnt =l extsw %lit_cnt_w
	%lit_src_ok =w cslel %lit_cnt, %src_left2
	jnz %lit_src_ok, @lit_check_out, @err_src
@lit_check_out
	%out_left =l sub %out_end, %oc
	%lit_fits =w cslel %lit_cnt, %out_left
	jnz %lit_fits, @lit_copy_loop, @lit_ovf_loop
@lit_copy_loop
	%li =l phi @lit_check_out 0, @lit_copy_body %li_n
	%lit_more =w csltl %li, %lit_cnt
	jnz %lit_more, @lit_copy_body, @lit_done
@lit_copy_body
	%lp =l add %sc_h, %li
	%lb =w loadub %lp
	%lq =l add %oc, %li
	storeb %lb, %lq
	%li_n =l add %li, 1
	jmp @lit_copy_loop
@lit_done
	%sc_lit =l add %sc_h, %lit_cnt
	%oc_lit =l add %oc, %lit_cnt
	jmp @loop
@lit_ovf_loop
	%loi =l phi @lit_check_out 0, @lit_ovf_body %loi_n
	%lit_ovf_more =w csltl %loi, %out_left
	jnz %lit_ovf_more, @lit_ovf_body, @err_cap
@lit_ovf_body
	%lop =l add %sc_h, %loi
	%lob =w loadub %lop
	%loq =l add %oc, %loi
	storeb %lob, %loq
	%loi_n =l add %loi, 1
	jmp @lit_ovf_loop
@repeat
	%rep_has_byte =w csgtl %src_left2, 0
	jnz %rep_has_byte, @rep_check_out, @err_src
@rep_check_out
	%rep_byte =w loadub %sc_h
	%sc_rep =l add %sc_h, 1
	%rep_cnt_w =w sub 1, %h
	%rep_cnt =l extsw %rep_cnt_w
	%out_left_r =l sub %out_end, %oc
	%rep_fits =w cslel %rep_cnt, %out_left_r
	jnz %rep_fits, @rep_fill_loop, @rep_ovf_loop
@rep_fill_loop
	%ri =l phi @rep_check_out 0, @rep_fill_body %ri_n
	%rep_more =w csltl %ri, %rep_cnt
	jnz %rep_more, @rep_fill_body, @rep_done
@rep_fill_body
	%rq =l add %oc, %ri
	storeb %rep_byte, %rq
	%ri_n =l add %ri, 1
	jmp @rep_fill_loop
@rep_done
	%oc_rep =l add %oc, %rep_cnt
	jmp @loop
@rep_ovf_loop
	%roi =l phi @rep_check_out 0, @rep_ovf_body %roi_n
	%rep_ovf_more =w csltl %roi, %out_left_r
	jnz %rep_ovf_more, @rep_ovf_body, @err_cap
@rep_ovf_body
	%roq =l add %oc, %roi
	storeb %rep_byte, %roq
	%roi_n =l add %roi, 1
	jmp @rep_ovf_loop
@done
	%written =l sub %oc, %out
	ret %written
@err_src
	ret -2
@err_cap
	ret -1
}
