export function l $qsort_pairs(l %a, l %n) {
@start
	%chk =w csltl %n, 2
	jnz %chk, @done, @work
@done
	ret 0
@work
	%nm1 =l sub %n, 1
	%half =l div %n, 2
	%hoff =l mul %half, 16
	%hp =l add %a, %hoff
	%hk =l loadl %hp
	%hvp =l add %hp, 8
	%hv =l loadl %hvp
	%poff =l mul %nm1, 16
	%pp =l add %a, %poff
	%pk =l loadl %pp
	%pvp =l add %pp, 8
	%pv =l loadl %pvp
	storel %hk, %pp
	storel %hv, %pvp
	storel %pk, %hp
	storel %pv, %hvp
	jmp @loop
@loop
	%i =l phi @work 0, @next %i2
	%j =l phi @work 0, @next %j2
	%jcond =w csltl %j, %nm1
	jnz %jcond, @body, @finish
@body
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jvp =l add %jp, 8
	%jv =l loadl %jvp
	%lt1 =w csltl %jk, %hk
	jnz %lt1, @swap, @eqk
@eqk
	%eq1 =w ceql %jk, %hk
	jnz %eq1, @cmpv, @noswap
@cmpv
	%lt2 =w csltl %jv, %hv
	jnz %lt2, @swap, @noswap
@swap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	storel %jk, %ip
	storel %jv, %ivp
	storel %ik, %jp
	storel %iv, %jvp
	%i1 =l add %i, 1
	jmp @next
@noswap
	jmp @next
@next
	%i2 =l phi @swap %i1, @noswap %i
	%j2 =l add %j, 1
	jmp @loop
@finish
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ivp2 =l add %ip2, 8
	%iv2 =l loadl %ivp2
	storel %hk, %ip2
	storel %hv, %ivp2
	storel %ik2, %pp
	storel %iv2, %pvp
	%left =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%roff =l mul %ip1, 16
	%ra =l add %a, %roff
	%rn =l sub %n, %ip1
	%right =l call $qsort_pairs(l %ra, l %rn)
	ret 0
}
