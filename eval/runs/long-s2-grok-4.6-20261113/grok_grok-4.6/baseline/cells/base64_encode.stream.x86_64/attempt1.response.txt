base64_encode:
    mov r8, rdx
.Lloop:
    cmp rsi, 3
    jb .Lremain
    movzx eax, byte ptr [rdi]
    shl eax, 16
    movzx ecx, byte ptr [rdi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rdi + 2]
    or eax, ecx
    mov r9d, eax
    add rdi, 3
    sub rsi, 3
    lea r11, [rip + .Lloop]
    jmp .Lencode
.Lremain:
    test rsi, rsi
    jz .Ldone
    movzx eax, byte ptr [rdi]
    shl eax, 16
    cmp rsi, 1
    je .Lhave
    movzx ecx, byte ptr [rdi + 1]
    shl ecx, 8
    or eax, ecx
.Lhave:
    mov r9d, eax
    lea r11, [rip + .Lpad]
    jmp .Lencode
.Lencode:
    mov r10d, 18
.Lchars:
    shrx eax, r9d, r10d
    and eax, 63
    cmp eax, 26
    jb .LcA
    cmp eax, 52
    jb .Lca
    cmp eax, 62
    jb .Lc0
    je .LcPlus
    mov al, '/'
    jmp .LcOut
.LcPlus:
    mov al, '+'
    jmp .LcOut
.Lc0:
    sub al, 4
    jmp .LcOut
.Lca:
    add al, 71
    jmp .LcOut
.LcA:
    add al, 65
.LcOut:
    mov byte ptr [rdx], al
    inc rdx
    sub r10d, 6
    jns .Lchars
    jmp r11
.Lpad:
    mov byte ptr [rdx - 1], '='
    cmp rsi, 1
    jne .Ldone
    mov byte ptr [rdx - 2], '='
.Ldone:
    mov rax, rdx
    sub rax, r8
    ret
