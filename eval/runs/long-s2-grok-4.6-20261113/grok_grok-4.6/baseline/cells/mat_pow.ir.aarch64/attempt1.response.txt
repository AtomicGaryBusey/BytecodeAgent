export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%s1p =l alloc8 8
	%s2p =l alloc8 8
	%dstp =l alloc8 8
	%phasep =l alloc8 8
	%kp =l alloc8 8
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%scratch =l add %tmp, %nn8
	%n2 =l add %nn, %nn
	storel %k, %kp
	jmp @copya_init
@copya_init
	jmp @copya_cond
@copya_cond
	%cai =l phi @copya_init 0, @copya_inc %cai1
	%ca_ok =w csltl %cai, %nn
	jnz %ca_ok, @copya_body, @id_init
@copya_body
	%ca_off =l mul %cai, 8
	%ca_ap =l add %a, %ca_off
	%ca_tp =l add %tmp, %ca_off
	%ca_v =l loadl %ca_ap
	storel %ca_v, %ca_tp
	jmp @copya_inc
@copya_inc
	%cai1 =l add %cai, 1
	jmp @copya_cond
@id_init
	jmp @id_cond
@id_cond
	%idi =l phi @id_init 0, @id_inc %idi1
	%id_ok =w csltl %idi, %nn
	jnz %id_ok, @id_body, @pow_init
@id_body
	%id_row =l udiv %idi, %n
	%id_col =l urem %idi, %n
	%id_eq =w ceql %id_row, %id_col
	%id_v =l extuw %id_eq
	%id_off =l mul %idi, 8
	%id_p =l add %out, %id_off
	storel %id_v, %id_p
	jmp @id_inc
@id_inc
	%idi1 =l add %idi, 1
	jmp @id_cond
@pow_init
	jmp @pow_cond
@pow_cond
	%pk =l loadl %kp
	%pk_nz =w cnel %pk, 0
	jnz %pk_nz, @pow_oddcheck, @cleanup
@pow_oddcheck
	%pk_and =l and %pk, 1
	%pk_odd =w cnel %pk_and, 0
	jnz %pk_odd, @mul_res, @mul_base
@mul_res
	storel %out, %s1p
	storel %tmp, %s2p
	storel %out, %dstp
	storel 0, %phasep
	jmp @mm_init
@mul_base
	storel %tmp, %s1p
	storel %tmp, %s2p
	storel %tmp, %dstp
	storel 1, %phasep
	jmp @mm_init
@mm_init
	%s1 =l loadl %s1p
	%s2 =l loadl %s2p
	jmp @mm_i_init
@mm_i_init
	jmp @mm_i_cond
@mm_i_cond
	%mi =l phi @mm_i_init 0, @mm_i_inc %mi1
	%mi_ok =w csltl %mi, %n
	jnz %mi_ok, @mm_j_init, @mm_copy_init
@mm_j_init
	jmp @mm_j_cond
@mm_j_cond
	%mj =l phi @mm_j_init 0, @mm_j_inc %mj1
	%mj_ok =w csltl %mj, %n
	jnz %mj_ok, @mm_t_init, @mm_i_inc
@mm_t_init
	jmp @mm_t_cond
@mm_t_cond
	%mt =l phi @mm_t_init 0, @mm_t_inc %mt1
	%msum =l phi @mm_t_init 0, @mm_t_inc %msum1
	%mt_ok =w csltl %mt, %n
	jnz %mt_ok, @mm_t_body, @mm_t_done
@mm_t_body
	%ait_i =l mul %mi, %n
	%ait_ix =l add %ait_i, %mt
	%ait_o =l mul %ait_ix, 8
	%ait_p =l add %s1, %ait_o
	%ait =l loadl %ait_p
	%btj_i =l mul %mt, %n
	%btj_ix =l add %btj_i, %mj
	%btj_o =l mul %btj_ix, 8
	%btj_p =l add %s2, %btj_o
	%btj =l loadl %btj_p
	%prod =l mul %ait, %btj
	%nsum =l add %msum, %prod
	jmp @mm_t_inc
@mm_t_inc
	%mt1 =l add %mt, 1
	%msum1 =l copy %nsum
	jmp @mm_t_cond
@mm_t_done
	%cij_i =l mul %mi, %n
	%cij_ix =l add %cij_i, %mj
	%cij_o =l mul %cij_ix, 8
	%cij_p =l add %scratch, %cij_o
	storel %msum, %cij_p
	jmp @mm_j_inc
@mm_j_inc
	%mj1 =l add %mj, 1
	jmp @mm_j_cond
@mm_i_inc
	%mi1 =l add %mi, 1
	jmp @mm_i_cond
@mm_copy_init
	%cdst =l loadl %dstp
	jmp @mm_copy_cond
@mm_copy_cond
	%cci =l phi @mm_copy_init 0, @mm_copy_inc %cci1
	%cc_ok =w csltl %cci, %nn
	jnz %cc_ok, @mm_copy_body, @after_mm
@mm_copy_body
	%cc_off =l mul %cci, 8
	%cc_sp =l add %scratch, %cc_off
	%cc_dp =l add %cdst, %cc_off
	%cc_v =l loadl %cc_sp
	storel %cc_v, %cc_dp
	jmp @mm_copy_inc
@mm_copy_inc
	%cci1 =l add %cci, 1
	jmp @mm_copy_cond
@after_mm
	%ph =l loadl %phasep
	%ph_is1 =w ceql %ph, 1
	jnz %ph_is1, @pow_shift, @mul_base
@pow_shift
	%pk_old =l loadl %kp
	%pk1 =l shr %pk_old, 1
	storel %pk1, %kp
	jmp @pow_cond
@cleanup
	jmp @zero_init
@zero_init
	jmp @zero_cond
@zero_cond
	%zi =l phi @zero_init 0, @zero_inc %zi1
	%z_ok =w csltl %zi, %n2
	jnz %z_ok, @zero_body, @ret
@zero_body
	%z_off =l mul %zi, 8
	%z_p =l add %tmp, %z_off
	storel 0, %z_p
	jmp @zero_inc
@zero_inc
	%zi1 =l add %zi, 1
	jmp @zero_cond
@ret
	ret 0
}
