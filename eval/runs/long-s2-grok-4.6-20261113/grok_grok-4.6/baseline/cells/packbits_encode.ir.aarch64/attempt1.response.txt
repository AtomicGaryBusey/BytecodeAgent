export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @repwrite %i_rep, @copydone %i_lit
	%wr =l phi @start 0, @repwrite %wr_rep, @copydone %wr_lit
	%i_lt_n =w csltl %i, %n
	jnz %i_lt_n, @havebyte, @done
@havebyte
	%iplus1 =l add %i, 1
	%ok1 =w csltl %iplus1, %n
	jnz %ok1, @chk2, @dolit
@chk2
	%iplus2 =l add %i, 2
	%ok2 =w csltl %iplus2, %n
	jnz %ok2, @chk3eq, @dolit
@chk3eq
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%p1 =l add %src, %iplus1
	%b1 =w loadub %p1
	%p2 =l add %src, %iplus2
	%b2 =w loadub %p2
	%eq01 =w ceqw %b0, %b1
	jnz %eq01, @chk02, @dolit
@chk02
	%eq02 =w ceqw %b0, %b2
	jnz %eq02, @dorep, @dolit
@dorep
	%rp0 =l add %src, %i
	%rb =w loadub %rp0
	jmp @runcnt
@runcnt
	%r =w phi @dorep 3, @runinc %rnext
	%r_lt_128 =w csltw %r, 128
	jnz %r_lt_128, @runmore, @repemit
@runmore
	%rext =l extuw %r
	%ir =l add %i, %rext
	%ir_lt_n =w csltl %ir, %n
	jnz %ir_lt_n, @runeq, @repemit
@runeq
	%pr =l add %src, %ir
	%br =w loadub %pr
	%eqr =w ceqw %rb, %br
	jnz %eqr, @runinc, @repemit
@runinc
	%rnext =w add %r, 1
	jmp @runcnt
@repemit
	%need2 =l add %wr, 2
	%fit2 =w cslel %need2, %cap
	jnz %fit2, @repwrite, @fail
@repwrite
	%hdr =w sub 257, %r
	%op =l add %out, %wr
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %rb, %op1
	%wr_rep =l add %wr, 2
	%rext2 =l extuw %r
	%i_rep =l add %i, %rext2
	jmp @outer
@dolit
	jmp @litcnt
@litcnt
	%j =l phi @dolit %i, @litinc %jnext
	%Lacc =w phi @dolit 0, @litinc %Lnext
	%j_lt_n =w csltl %j, %n
	jnz %j_lt_n, @litlen, @litemit
@litlen
	%L_lt_128 =w csltw %Lacc, 128
	jnz %L_lt_128, @litrun, @litemit
@litrun
	%j1 =l add %j, 1
	%j1_lt_n =w csltl %j1, %n
	jnz %j1_lt_n, @litc2, @litinc
@litc2
	%j2 =l add %j, 2
	%j2_lt_n =w csltl %j2, %n
	jnz %j2_lt_n, @liteq, @litinc
@liteq
	%qj0 =l add %src, %j
	%cj0 =w loadub %qj0
	%qj1 =l add %src, %j1
	%cj1 =w loadub %qj1
	%qj2 =l add %src, %j2
	%cj2 =w loadub %qj2
	%ceq01 =w ceqw %cj0, %cj1
	jnz %ceq01, @liteq2, @litinc
@liteq2
	%ceq02 =w ceqw %cj0, %cj2
	jnz %ceq02, @litemit, @litinc
@litinc
	%jnext =l add %j, 1
	%Lnext =w add %Lacc, 1
	jmp @litcnt
@litemit
	%Lext =l extuw %Lacc
	%needL =l add %Lext, 1
	%needtot =l add %wr, %needL
	%fitL =w cslel %needtot, %cap
	jnz %fitL, @litwrite, @fail
@litwrite
	%hdrL =w sub %Lacc, 1
	%opl =l add %out, %wr
	storeb %hdrL, %opl
	%dst_start =l add %opl, 1
	%src_i =l add %src, %i
	jmp @copy
@copy
	%cs =l phi @litwrite %src_i, @copydo %csn
	%cd =l phi @litwrite %dst_start, @copydo %cdn
	%kk =w phi @litwrite %Lacc, @copydo %kk1
	jnz %kk, @copydo, @copydone
@copydo
	%cb =w loadub %cs
	storeb %cb, %cd
	%csn =l add %cs, 1
	%cd =l add %cd, 1
	%kk1 =w sub %kk, 1
	jmp @copy
@copydone
	%wr_lit =l add %wr, %needL
	%i_lit =l copy %j
	jmp @outer
@done
	ret %wr
@fail
	ret -1
}
