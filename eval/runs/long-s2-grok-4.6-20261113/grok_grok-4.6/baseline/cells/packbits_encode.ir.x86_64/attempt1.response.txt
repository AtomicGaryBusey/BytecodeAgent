export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main
@main
	%i =l phi @start 0, @emitrep %i_rep, @emitlit %i_lit
	%wr =l phi @start 0, @emitrep %wr_rep, @emitlit %wr_lit
	%cont =w csltl %i, %n
	jnz %cont, @rloop, @done
@rloop
	%r_r =w phi @main 1, @rinc %r_new
	%r_l =l extuw %r_r
	%idx2 =l add %i, %r_l
	%okn =w csltl %idx2, %n
	%ok128 =w csltw %r_r, 128
	%ok =w and %okn, %ok128
	jnz %ok, @rcmp, @rdone
@rcmp
	%pa =l add %src, %i
	%ba =w loadub %pa
	%pb =l add %src, %idx2
	%bb =w loadub %pb
	%same =w ceqw %ba, %bb
	jnz %same, @rinc, @rdone
@rinc
	%r_new =w add %r_r, 1
	jmp @rloop
@rdone
	%isrep =w cslew 3, %r_r
	jnz %isrep, @dorep, @jloop
@dorep
	%need2 =l add %wr, 2
	%fits =w cslel %need2, %cap
	jnz %fits, @emitrep, @fail
@emitrep
	%hrepw =w sub 257, %r_r
	%op =l add %out, %wr
	storeb %hrepw, %op
	%pa2 =l add %src, %i
	%bval =w loadub %pa2
	%op1 =l add %op, 1
	storeb %bval, %op1
	%wr_rep =l add %wr, 2
	%i_rep =l add %i, %r_l
	jmp @main
@jloop
	%j =l phi @rdone %i, @jinc %jnext
	%Lacc =w phi @rdone 0, @jinc %Lnext
	%jn =w csltl %j, %n
	%s128 =w csltw %Lacc, 128
	%jok =w and %jn, %s128
	jnz %jok, @rjloop, @jdone
@rjloop
	%rj_r =w phi @jloop 1, @rjinc %rj_new
	%rj_l =l extuw %rj_r
	%j2 =l add %j, %rj_l
	%j2n =w csltl %j2, %n
	%rj128 =w csltw %rj_r, 128
	%rjok =w and %j2n, %rj128
	jnz %rjok, @rjcmp, @rjdone
@rjcmp
	%pc =l add %src, %j
	%bc =w loadub %pc
	%pd =l add %src, %j2
	%bd =w loadub %pd
	%same2 =w ceqw %bc, %bd
	jnz %same2, @rjinc, @rjdone
@rjinc
	%rj_new =w add %rj_r, 1
	jmp @rjloop
@rjdone
	%lt3 =w csltw %rj_r, 3
	jnz %lt3, @jinc, @jdone
@jinc
	%jnext =l add %j, 1
	%Lnext =w add %Lacc, 1
	jmp @jloop
@jdone
	%needLw =w add %Lacc, 1
	%needL =l extuw %needLw
	%needtot =l add %wr, %needL
	%fitsL =w cslel %needtot, %cap
	jnz %fitsL, @emitlith, @fail
@emitlith
	%hlitw =w sub %Lacc, 1
	%opl =l add %out, %wr
	storeb %hlitw, %opl
	jmp @cloop
@cloop
	%k =w phi @emitlith 0, @cbyte %knext
	%kL =w csltw %k, %Lacc
	jnz %kL, @cbyte, @emitlit
@cbyte
	%k_l =l extuw %k
	%soff =l add %i, %k_l
	%sp =l add %src, %soff
	%byt =w loadub %sp
	%k_l1 =l add %k_l, 1
	%daddr =l add %opl, %k_l1
	storeb %byt, %daddr
	%knext =w add %k, 1
	jmp @cloop
@emitlit
	%wr_lit =l add %wr, %needL
	%i_lit =l copy %j
	jmp @main
@fail
	ret -1
@done
	ret %wr
}
