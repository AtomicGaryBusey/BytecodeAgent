export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcp =l alloc8 8
	%spp =l alloc8 8
	storel 0, %pcp
	storel 0, %spp
	jmp @loop
@loop
	%pc =l loadl %pcp
	%sp =l loadl %spp
	%done =w ceql %pc, %len
	jnz %done, @halt, @fetch
@fetch
	%ioff =l mul %pc, 8
	%iaddr =l add %code, %ioff
	%op =l loadl %iaddr
	%ishalt =w ceql %op, 0
	jnz %ishalt, @halt, @c1
@c1
	%ispush =w ceql %op, 1
	jnz %ispush, @op_push, @c2
@c2
	%isadd =w ceql %op, 2
	jnz %isadd, @op_add, @c3
@c3
	%issub =w ceql %op, 3
	jnz %issub, @op_sub, @c4
@c4
	%ismul =w ceql %op, 4
	jnz %ismul, @op_mul, @c5
@c5
	%isdup =w ceql %op, 5
	jnz %isdup, @op_dup, @c6
@c6
	%isswap =w ceql %op, 6
	jnz %isswap, @op_swap, @c7
@c7
	%isdrop =w ceql %op, 7
	jnz %isdrop, @op_drop, @c8
@c8
	%isover =w ceql %op, 8
	jnz %isover, @op_over, @c9
@c9
	%islt =w ceql %op, 9
	jnz %islt, @op_lt, @c10
@c10
	%isjz =w ceql %op, 10
	jnz %isjz, @op_jz, @c11
@c11
	%isjmp =w ceql %op, 11
	jnz %isjmp, @op_jmp, @op_other
@op_push
	%p1 =l add %pc, 1
	%poff =l mul %p1, 8
	%paddr =l add %code, %poff
	%pval =l loadl %paddr
	%qoff =l mul %sp, 8
	%qaddr =l add %stack, %qoff
	storel %pval, %qaddr
	%psp =l add %sp, 1
	storel %psp, %spp
	%ppc =l add %pc, 2
	storel %ppc, %pcp
	jmp @loop
@op_add
	%asp1 =l sub %sp, 1
	%aoffb =l mul %asp1, 8
	%aaddrb =l add %stack, %aoffb
	%badd =l loadl %aaddrb
	%asp2 =l sub %sp, 2
	%aoffa =l mul %asp2, 8
	%aaddra =l add %stack, %aoffa
	%aadd =l loadl %aaddra
	%asum =l add %aadd, %badd
	storel %asum, %aaddra
	storel %asp1, %spp
	%apc =l add %pc, 1
	storel %apc, %pcp
	jmp @loop
@op_sub
	%sbsp1 =l sub %sp, 1
	%sboffb =l mul %sbsp1, 8
	%sbaddrb =l add %stack, %sboffb
	%sb_b =l loadl %sbaddrb
	%sbsp2 =l sub %sp, 2
	%sboffa =l mul %sbsp2, 8
	%sbaddra =l add %stack, %sboffa
	%sb_a =l loadl %sbaddra
	%sbdiff =l sub %sb_a, %sb_b
	storel %sbdiff, %sbaddra
	storel %sbsp1, %spp
	%sbpc =l add %pc, 1
	storel %sbpc, %pcp
	jmp @loop
@op_mul
	%musp1 =l sub %sp, 1
	%muoffb =l mul %musp1, 8
	%muaddrb =l add %stack, %muoffb
	%mu_b =l loadl %muaddrb
	%musp2 =l sub %sp, 2
	%muoffa =l mul %musp2, 8
	%muaddra =l add %stack, %muoffa
	%mu_a =l loadl %muaddra
	%muprod =l mul %mu_a, %mu_b
	storel %muprod, %muaddra
	storel %musp1, %spp
	%mupc =l add %pc, 1
	storel %mupc, %pcp
	jmp @loop
@op_dup
	%dtopi =l sub %sp, 1
	%doff =l mul %dtopi, 8
	%daddr =l add %stack, %doff
	%dval =l loadl %daddr
	%dnoff =l mul %sp, 8
	%dnaddr =l add %stack, %dnoff
	storel %dval, %dnaddr
	%dsp =l add %sp, 1
	storel %dsp, %spp
	%dpc =l add %pc, 1
	storel %dpc, %pcp
	jmp @loop
@op_swap
	%wtopi =l sub %sp, 1
	%woff1 =l mul %wtopi, 8
	%waddr1 =l add %stack, %woff1
	%wvalb =l loadl %waddr1
	%wtopj =l sub %sp, 2
	%woff0 =l mul %wtopj, 8
	%waddr0 =l add %stack, %woff0
	%wvala =l loadl %waddr0
	storel %wvalb, %waddr0
	storel %wvala, %waddr1
	%wpc =l add %pc, 1
	storel %wpc, %pcp
	jmp @loop
@op_drop
	%drsp =l sub %sp, 1
	storel %drsp, %spp
	%drpc =l add %pc, 1
	storel %drpc, %pcp
	jmp @loop
@op_over
	%ovti =l sub %sp, 2
	%ovoff =l mul %ovti, 8
	%ovaddr =l add %stack, %ovoff
	%ovval =l loadl %ovaddr
	%ovnoff =l mul %sp, 8
	%ovnaddr =l add %stack, %ovnoff
	storel %ovval, %ovnaddr
	%ovsp =l add %sp, 1
	storel %ovsp, %spp
	%ovpc =l add %pc, 1
	storel %ovpc, %pcp
	jmp @loop
@op_lt
	%ltsp1 =l sub %sp, 1
	%ltoffb =l mul %ltsp1, 8
	%ltaddrb =l add %stack, %ltoffb
	%ltb =l loadl %ltaddrb
	%ltsp2 =l sub %sp, 2
	%ltoffa =l mul %ltsp2, 8
	%ltaddra =l add %stack, %ltoffa
	%lta =l loadl %ltaddra
	%ltw =w csltl %lta, %ltb
	%ltv =l extuw %ltw
	storel %ltv, %ltaddra
	storel %ltsp1, %spp
	%ltpc =l add %pc, 1
	storel %ltpc, %pcp
	jmp @loop
@op_jz
	%jzsp =l sub %sp, 1
	storel %jzsp, %spp
	%jzoff =l mul %jzsp, 8
	%jzaddr =l add %stack, %jzoff
	%jzv =l loadl %jzaddr
	%jzp1 =l add %pc, 1
	%jzofft =l mul %jzp1, 8
	%jzaddrt =l add %code, %jzofft
	%jzt =l loadl %jzaddrt
	%jzzero =w ceql %jzv, 0
	jnz %jzzero, @jz_take, @jz_skip
@jz_take
	storel %jzt, %pcp
	jmp @loop
@jz_skip
	%jzpc =l add %pc, 2
	storel %jzpc, %pcp
	jmp @loop
@op_jmp
	%jp1 =l add %pc, 1
	%joff =l mul %jp1, 8
	%jaddr =l add %code, %joff
	%jt =l loadl %jaddr
	storel %jt, %pcp
	jmp @loop
@op_other
	%opc2 =l add %pc, 1
	storel %opc2, %pcp
	jmp @loop
@halt
	%hsp =l loadl %spp
	jmp @zloop
@zloop
	%zi =l phi @halt %hsp, @zbody %zini
	%zcont =w csltl %zi, %cap
	jnz %zcont, @zbody, @ret
@zbody
	%zoff =l mul %zi, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%zini =l add %zi, 1
	jmp @zloop
@ret
	%isempty =w ceql %hsp, 0
	jnz %isempty, @ret0, @rettop
@ret0
	ret 0
@rettop
	%rti =l sub %hsp, 1
	%rtoff =l mul %rti, 8
	%rtaddr =l add %stack, %rtoff
	%rtval =l loadl %rtaddr
	ret %rtval
}
