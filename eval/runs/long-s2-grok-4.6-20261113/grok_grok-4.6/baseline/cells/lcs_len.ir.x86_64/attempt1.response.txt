export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @inith
@inith
	%ji =l phi @start 0, @initb %jin
	%ci =w cslel %ji, %n
	jnz %ci, @initb, @initd
@initb
	%io =l mul %ji, 8
	%ip =l add %row, %io
	storel 0, %ip
	%jin =l add %ji, 1
	jmp @inith
@initd
	jmp @outerh
@outerh
	%i =l phi @initd 0, @innerd %in
	%co =w csltl %i, %m
	jnz %co, @outerb, @done
@outerb
	%xp =l add %x, %i
	%xi =w loadub %xp
	jmp @innerh
@innerh
	%j =l phi @outerb 1, @innerc %jn
	%diag =l phi @outerb 0, @innerc %upout
	%cj =w cslel %j, %n
	jnz %cj, @innerb, @innerd
@innerb
	%jo =l mul %j, 8
	%jp =l add %row, %jo
	%up =l loadl %jp
	%jm =l sub %j, 1
	%yp =l add %y, %jm
	%yj =w loadub %yp
	%eq =w ceqw %xi, %yj
	jnz %eq, @eqc, @neqc
@eqc
	%veq =l add %diag, 1
	jmp @store
@neqc
	%jmo =l mul %jm, 8
	%jmptr =l add %row, %jmo
	%left =l loadl %jmptr
	%gt =w csgtl %up, %left
	jnz %gt, @gtrue, @gfalse
@gtrue
	jmp @store
@gfalse
	jmp @store
@store
	%nv =l phi @eqc %veq, @gtrue %up, @gfalse %left
	storel %nv, %jp
	jmp @innerc
@innerc
	%upout =l copy %up
	%jn =l add %j, 1
	jmp @innerh
@innerd
	%in =l add %i, 1
	jmp @outerh
@done
	%no =l mul %n, 8
	%np =l add %row, %no
	%res =l loadl %np
	ret %res
}
