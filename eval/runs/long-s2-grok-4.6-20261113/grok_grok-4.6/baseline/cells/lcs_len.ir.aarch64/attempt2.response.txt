export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @initloop
@initloop
	%j =l phi @start 0, @initbody %jnext
	%cmpinit =w csgtl %j, %n
	jnz %cmpinit, @iloop, @initbody
@initbody
	%offj =l mul %j, 8
	%addrj =l add %row, %offj
	storel 0, %addrj
	%jnext =l add %j, 1
	jmp @initloop
@iloop
	%i =l phi @initloop 0, @innerdone %inext
	%cmpi =w csltl %i, %m
	jnz %cmpi, @jloop, @done
@jloop
	%jj =l phi @iloop 1, @jcont %jjnext
	%diag =l phi @iloop 0, @jcont %up
	%cmpj =w cslel %jj, %n
	jnz %cmpj, @jbody, @innerdone
@jbody
	%offjj =l mul %jj, 8
	%addrjj =l add %row, %offjj
	%up =l loadl %addrjj
	%xaddr =l add %x, %i
	%xb =w loadub %xaddr
	%jprev =l sub %jj, 1
	%yaddr =l add %y, %jprev
	%yb =w loadub %yaddr
	%eq =w ceqw %xb, %yb
	jnz %eq, @eqcase, @neqcase
@eqcase
	%valeq =l add %diag, 1
	storel %valeq, %addrjj
	jmp @jcont
@neqcase
	%offprev =l mul %jprev, 8
	%addrprev =l add %row, %offprev
	%left =l loadl %addrprev
	%isgt =w csgtl %up, %left
	jnz %isgt, @takeup, @takeleft
@takeup
	storel %up, %addrjj
	jmp @jcont
@takeleft
	storel %left, %addrjj
	jmp @jcont
@jcont
	%jjnext =l add %jj, 1
	jmp @jloop
@innerdone
	%inext =l add %i, 1
	jmp @iloop
@done
	%offn =l mul %n, 8
	%addrn =l add %row, %offn
	%res =l loadl %addrn
	ret %res
}
