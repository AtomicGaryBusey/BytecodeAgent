export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc.addr =l alloc8 8
	%sp.addr =l alloc8 8
	storel 0, %pc.addr
	storel 0, %sp.addr
	jmp @loop
@loop
	%pc.cur =l loadl %pc.addr
	%atend =w ceql %pc.cur, %len
	jnz %atend, @halt, @exec
@exec
	%pc.off =l mul %pc.cur, 8
	%ip =l add %code, %pc.off
	%op =l loadl %ip
	%is.halt =w ceql %op, 0
	jnz %is.halt, @halt, @n0
@n0
	%is.push =w ceql %op, 1
	jnz %is.push, @op.push, @n1
@n1
	%is.add =w ceql %op, 2
	jnz %is.add, @op.add, @n2
@n2
	%is.sub =w ceql %op, 3
	jnz %is.sub, @op.sub, @n3
@n3
	%is.mul =w ceql %op, 4
	jnz %is.mul, @op.mul, @n4
@n4
	%is.dup =w ceql %op, 5
	jnz %is.dup, @op.dup, @n5
@n5
	%is.swap =w ceql %op, 6
	jnz %is.swap, @op.swap, @n6
@n6
	%is.drop =w ceql %op, 7
	jnz %is.drop, @op.drop, @n7
@n7
	%is.over =w ceql %op, 8
	jnz %is.over, @op.over, @n8
@n8
	%is.lt =w ceql %op, 9
	jnz %is.lt, @op.lt, @n9
@n9
	%is.jz =w ceql %op, 10
	jnz %is.jz, @op.jz, @n10
@n10
	%is.jmp =w ceql %op, 11
	jnz %is.jmp, @op.jmp, @incpc
@op.push
	%push.pc1 =l add %pc.cur, 1
	%push.off =l mul %push.pc1, 8
	%push.vaddr =l add %code, %push.off
	%push.v =l loadl %push.vaddr
	%push.sp =l loadl %sp.addr
	%push.soff =l mul %push.sp, 8
	%push.saddr =l add %stack, %push.soff
	storel %push.v, %push.saddr
	%push.sp1 =l add %push.sp, 1
	storel %push.sp1, %sp.addr
	%push.pc2 =l add %pc.cur, 2
	storel %push.pc2, %pc.addr
	jmp @loop
@op.add
	%add.sp =l loadl %sp.addr
	%add.sp1 =l sub %add.sp, 1
	%add.boff =l mul %add.sp1, 8
	%add.baddr =l add %stack, %add.boff
	%add.b =l loadl %add.baddr
	%add.sp2 =l sub %add.sp1, 1
	%add.aoff =l mul %add.sp2, 8
	%add.aaddr =l add %stack, %add.aoff
	%add.a =l loadl %add.aaddr
	%add.res =l add %add.a, %add.b
	storel %add.res, %add.aaddr
	storel %add.sp1, %sp.addr
	jmp @incpc
@op.sub
	%sub.sp =l loadl %sp.addr
	%sub.sp1 =l sub %sub.sp, 1
	%sub.boff =l mul %sub.sp1, 8
	%sub.baddr =l add %stack, %sub.boff
	%sub.b =l loadl %sub.baddr
	%sub.sp2 =l sub %sub.sp1, 1
	%sub.aoff =l mul %sub.sp2, 8
	%sub.aaddr =l add %stack, %sub.aoff
	%sub.a =l loadl %sub.aaddr
	%sub.res =l sub %sub.a, %sub.b
	storel %sub.res, %sub.aaddr
	storel %sub.sp1, %sp.addr
	jmp @incpc
@op.mul
	%mul.sp =l loadl %sp.addr
	%mul.sp1 =l sub %mul.sp, 1
	%mul.boff =l mul %mul.sp1, 8
	%mul.baddr =l add %stack, %mul.boff
	%mul.b =l loadl %mul.baddr
	%mul.sp2 =l sub %mul.sp1, 1
	%mul.aoff =l mul %mul.sp2, 8
	%mul.aaddr =l add %stack, %mul.aoff
	%mul.a =l loadl %mul.aaddr
	%mul.res =l mul %mul.a, %mul.b
	storel %mul.res, %mul.aaddr
	storel %mul.sp1, %sp.addr
	jmp @incpc
@op.dup
	%dup.sp =l loadl %sp.addr
	%dup.sp1 =l sub %dup.sp, 1
	%dup.off =l mul %dup.sp1, 8
	%dup.addr =l add %stack, %dup.off
	%dup.v =l loadl %dup.addr
	%dup.noff =l mul %dup.sp, 8
	%dup.naddr =l add %stack, %dup.noff
	storel %dup.v, %dup.naddr
	%dup.spn =l add %dup.sp, 1
	storel %dup.spn, %sp.addr
	jmp @incpc
@op.swap
	%sw.sp =l loadl %sp.addr
	%sw.i1 =l sub %sw.sp, 1
	%sw.i0 =l sub %sw.sp, 2
	%sw.off1 =l mul %sw.i1, 8
	%sw.off0 =l mul %sw.i0, 8
	%sw.a1 =l add %stack, %sw.off1
	%sw.a0 =l add %stack, %sw.off0
	%sw.v1 =l loadl %sw.a1
	%sw.v0 =l loadl %sw.a0
	storel %sw.v0, %sw.a1
	storel %sw.v1, %sw.a0
	jmp @incpc
@op.drop
	%dr.sp =l loadl %sp.addr
	%dr.sp1 =l sub %dr.sp, 1
	storel %dr.sp1, %sp.addr
	jmp @incpc
@op.over
	%ov.sp =l loadl %sp.addr
	%ov.i =l sub %ov.sp, 2
	%ov.off =l mul %ov.i, 8
	%ov.addr =l add %stack, %ov.off
	%ov.v =l loadl %ov.addr
	%ov.noff =l mul %ov.sp, 8
	%ov.naddr =l add %stack, %ov.noff
	storel %ov.v, %ov.naddr
	%ov.spn =l add %ov.sp, 1
	storel %ov.spn, %sp.addr
	jmp @incpc
@op.lt
	%lt.sp =l loadl %sp.addr
	%lt.sp1 =l sub %lt.sp, 1
	%lt.boff =l mul %lt.sp1, 8
	%lt.baddr =l add %stack, %lt.boff
	%lt.b =l loadl %lt.baddr
	%lt.sp2 =l sub %lt.sp1, 1
	%lt.aoff =l mul %lt.sp2, 8
	%lt.aaddr =l add %stack, %lt.aoff
	%lt.a =l loadl %lt.aaddr
	%lt.cmp =w csltl %lt.a, %lt.b
	%lt.res =l extuw %lt.cmp
	storel %lt.res, %lt.aaddr
	storel %lt.sp1, %sp.addr
	jmp @incpc
@op.jz
	%jz.sp =l loadl %sp.addr
	%jz.sp1 =l sub %jz.sp, 1
	storel %jz.sp1, %sp.addr
	%jz.off =l mul %jz.sp1, 8
	%jz.addr =l add %stack, %jz.off
	%jz.v =l loadl %jz.addr
	%jz.pc1 =l add %pc.cur, 1
	%jz.toff =l mul %jz.pc1, 8
	%jz.taddr =l add %code, %jz.toff
	%jz.t =l loadl %jz.taddr
	%jz.zero =w ceql %jz.v, 0
	jnz %jz.zero, @jz.take, @jz.skip
@jz.take
	storel %jz.t, %pc.addr
	jmp @loop
@jz.skip
	%jz.pc2 =l add %pc.cur, 2
	storel %jz.pc2, %pc.addr
	jmp @loop
@op.jmp
	%jmp.pc1 =l add %pc.cur, 1
	%jmp.off =l mul %jmp.pc1, 8
	%jmp.addr =l add %code, %jmp.off
	%jmp.t =l loadl %jmp.addr
	storel %jmp.t, %pc.addr
	jmp @loop
@incpc
	%inc.pc =l add %pc.cur, 1
	storel %inc.pc, %pc.addr
	jmp @loop
@halt
	%sp.final =l loadl %sp.addr
	jmp @zero.loop
@zero.loop
	%zi =l phi @halt %sp.final, @zero.body %zi.next
	%zcmp =w csltl %zi, %cap
	jnz %zcmp, @zero.body, @do.ret
@zero.body
	%zi.off =l mul %zi, 8
	%zi.addr =l add %stack, %zi.off
	storel 0, %zi.addr
	%zi.next =l add %zi, 1
	jmp @zero.loop
@do.ret
	%sp.empty =w ceql %sp.final, 0
	jnz %sp.empty, @ret.0, @ret.top
@ret.0
	ret 0
@ret.top
	%top.idx =l sub %sp.final, 1
	%top.off =l mul %top.idx, 8
	%top.addr =l add %stack, %top.off
	%retval =l loadl %top.addr
	ret %retval
}
