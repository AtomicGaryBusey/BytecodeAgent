export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%mask =l copy 4294967295
	%n0 =w ceql %n, 0
	jnz %n0, @ret0, @zloop
@ret0
	ret 0
@zloop
	%zi =l phi @start 0, @zbody %zi1
	%zcmp =w csltl %zi, %n
	jnz %zcmp, @zbody, @iloop
@zbody
	%zi8 =l shl %zi, 3
	%zrp =l add %r, %zi8
	storel 0, %zrp
	%zi1 =l add %zi, 1
	jmp @zloop
@iloop
	%i =l phi @zloop 0, @jdone %i_next
	%icmp =w csltl %i, %na
	jnz %icmp, @ibody, @kloop
@ibody
	%i8 =l shl %i, 3
	%ap =l add %a, %i8
	%aval =l loadl %ap
	%a0 =l and %aval, %mask
	%a1 =l shr %aval, 32
	jmp @jloop
@jloop
	%j =l phi @ibody 0, @jbody %j_next
	%carry =l phi @ibody 0, @jbody %newcarry
	%jcmp =w csltl %j, %nb
	jnz %jcmp, @jbody, @jdone
@jbody
	%j8 =l shl %j, 3
	%bp =l add %b, %j8
	%bval =l loadl %bp
	%b0 =l and %bval, %mask
	%b1 =l shr %bval, 32
	%u0 =l mul %a0, %b0
	%u1 =l mul %a0, %b1
	%u2 =l mul %a1, %b0
	%u3 =l mul %a1, %b1
	%tmid =l add %u1, %u2
	%c1 =w cultl %tmid, %u1
	%c1l =l extuw %c1
	%tlo =l shl %tmid, 32
	%thi =l shr %tmid, 32
	%plo =l add %u0, %tlo
	%c2 =w cultl %plo, %u0
	%c2l =l extuw %c2
	%hi1 =l add %u3, %thi
	%hi2 =l add %hi1, %c2l
	%c1s =l shl %c1l, 32
	%prodh =l add %hi2, %c1s
	%ij =l add %i, %j
	%ij8 =l shl %ij, 3
	%rp =l add %r, %ij8
	%rval =l loadl %rp
	%s1 =l add %plo, %rval
	%ca =w cultl %s1, %plo
	%cal =l extuw %ca
	%s2 =l add %s1, %carry
	%cb =w cultl %s2, %s1
	%cbl =l extuw %cb
	%csum =l add %cal, %cbl
	%newcarry =l add %prodh, %csum
	storel %s2, %rp
	%j_next =l add %j, 1
	jmp @jloop
@jdone
	%inb =l add %i, %nb
	%inb8 =l shl %inb, 3
	%rp2 =l add %r, %inb8
	storel %carry, %rp2
	%i_next =l add %i, 1
	jmp @iloop
@kloop
	%k =l phi @iloop %n, @kbody %k1
	%k0 =w ceql %k, 0
	jnz %k0, @retk, @kcheck
@kcheck
	%km1 =l sub %k, 1
	%km18 =l shl %km1, 3
	%rpk =l add %r, %km18
	%rvk =l loadl %rpk
	%isz =w ceql %rvk, 0
	jnz %isz, @kbody, @retk
@kbody
	%k1 =l copy %km1
	jmp @kloop
@retk
	ret %k
}
