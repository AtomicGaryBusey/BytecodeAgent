export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @cont %i_next
	%o =l phi @start 0, @cont %o_next
	%more =w csltl %i, %n
	jnz %more, @hdr, @success
@success
	ret %o
@hdr
	%src_i =l add %src, %i
	%h =w loadsb %src_i
	%i1 =l add %i, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @notnop
@nop
	%i_nop =l copy %i1
	%o_nop =l copy %o
	jmp @cont
@notnop
	%islit =w csgew %h, 0
	jnz %islit, @literal, @repeat
@literal
	%hl =l extsw %h
	%litcnt =l add %hl, 1
	%srem =l sub %n, %i1
	%srcok =w cslel %litcnt, %srem
	jnz %srcok, @lit_srcok, @err2
@lit_srcok
	%orem =l sub %cap, %o
	%lt =w csltl %litcnt, %orem
	%ltl =l extuw %lt
	%diff =l sub %litcnt, %orem
	%adj =l mul %diff, %ltl
	%tocopy =l add %orem, %adj
	%p_src =l add %src, %i1
	%p_dst =l add %out, %o
	jmp @lcopy
@lcopy
	%cleft =l phi @lit_srcok %tocopy, @lcopy1 %cleft1
	%ps =l phi @lit_srcok %p_src, @lcopy1 %ps1
	%pd =l phi @lit_srcok %p_dst, @lcopy1 %pd1
	%has =w cnel %cleft, 0
	jnz %has, @lcopy1, @lcopy_done
@lcopy1
	%byte =w loadub %ps
	storeb %byte, %pd
	%ps1 =l add %ps, 1
	%pd1 =l add %pd, 1
	%cleft1 =l sub %cleft, 1
	jmp @lcopy
@lcopy_done
	%partial =w csltl %tocopy, %litcnt
	jnz %partial, @err1, @lit_done
@lit_done
	%i_lit =l add %i1, %tocopy
	%o_lit =l add %o, %tocopy
	jmp @cont
@repeat
	%sremr =l sub %n, %i1
	%srcokr =w cslel 1, %sremr
	jnz %srcokr, @rep_srcok, @err2
@rep_srcok
	%hr =l extsw %h
	%repcnt =l sub 1, %hr
	%bp =l add %src, %i1
	%repb =w loadub %bp
	%i_rep0 =l add %i1, 1
	%oremr =l sub %cap, %o
	%ltr =w csltl %repcnt, %oremr
	%ltlr =l extuw %ltr
	%diffr =l sub %repcnt, %oremr
	%adjr =l mul %diffr, %ltlr
	%tofill =l add %oremr, %adjr
	%p_dstr =l add %out, %o
	jmp @rfill
@rfill
	%fleft =l phi @rep_srcok %tofill, @rfill1 %fleft1
	%pdr =l phi @rep_srcok %p_dstr, @rfill1 %pdr1
	%hasr =w cnel %fleft, 0
	jnz %hasr, @rfill1, @rfill_done
@rfill1
	storeb %repb, %pdr
	%pdr1 =l add %pdr, 1
	%fleft1 =l sub %fleft, 1
	jmp @rfill
@rfill_done
	%partialr =w csltl %tofill, %repcnt
	jnz %partialr, @err1, @rep_done
@rep_done
	%i_rep =l copy %i_rep0
	%o_rep =l add %o, %tofill
	jmp @cont
@cont
	%i_next =l phi @nop %i_nop, @lit_done %i_lit, @rep_done %i_rep
	%o_next =l phi @nop %o_nop, @lit_done %o_lit, @rep_done %o_rep
	jmp @loop
@err1
	ret -1
@err2
	ret -2
}
