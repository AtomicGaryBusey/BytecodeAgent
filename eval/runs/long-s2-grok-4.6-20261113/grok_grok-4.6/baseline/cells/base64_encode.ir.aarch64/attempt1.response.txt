export function l $base64_encode(l %src, l %n, l %out) {
@start
	%psrc =l alloc8 8
	%pout =l alloc8 8
	%pleft =l alloc8 8
	%pval =l alloc8 8
	%pidx =l alloc8 8
	%ppad =l alloc8 8
	%pchar =l alloc8 8
	storel %src, %psrc
	storel %out, %pout
	storel %n, %pleft
	jmp @loop
@loop
	%left =l loadl %pleft
	%is0 =w ceql %left, 0
	jnz %is0, @done, @check3
@check3
	%ge3 =w csgel %left, 3
	jnz %ge3, @do3, @check2
@do3
	%s3 =l loadl %psrc
	%b03 =l loadub %s3
	%s3a =l add %s3, 1
	%b13 =l loadub %s3a
	%s3b =l add %s3, 2
	%b23 =l loadub %s3b
	%s3c =l add %s3, 3
	storel %s3c, %psrc
	%left3 =l sub %left, 3
	storel %left3, %pleft
	storel 0, %ppad
	%sh03 =l shl %b03, 16
	%sh13 =l shl %b13, 8
	%or013 =l or %sh03, %sh13
	%val3 =l or %or013, %b23
	storel %val3, %pval
	jmp @encode
@check2
	%is2 =w ceql %left, 2
	jnz %is2, @do2, @do1
@do2
	%s2 =l loadl %psrc
	%b02 =l loadub %s2
	%s2a =l add %s2, 1
	%b12 =l loadub %s2a
	storel 0, %pleft
	storel 1, %ppad
	%sh02 =l shl %b02, 16
	%sh12 =l shl %b12, 8
	%val2 =l or %sh02, %sh12
	storel %val2, %pval
	jmp @encode
@do1
	%s1 =l loadl %psrc
	%b01 =l loadub %s1
	storel 0, %pleft
	storel 2, %ppad
	%val1 =l shl %b01, 16
	storel %val1, %pval
	jmp @encode
@encode
	storel 0, %pidx
	jmp @encchar
@encchar
	%idx =l loadl %pidx
	%idx6 =l mul %idx, 6
	%shamt =l sub 18, %idx6
	%val =l loadl %pval
	%shf =l shr %val, %shamt
	%v =l and %shf, 63
	%lt26 =w csltl %v, 26
	jnz %lt26, @m_A, @m_ge26
@m_A
	%cA =l add %v, 65
	storel %cA, %pchar
	jmp @mapped
@m_ge26
	%lt52 =w csltl %v, 52
	jnz %lt52, @m_a, @m_ge52
@m_a
	%ca =l add %v, 71
	storel %ca, %pchar
	jmp @mapped
@m_ge52
	%lt62 =w csltl %v, 62
	jnz %lt62, @m_dig, @m_sym
@m_dig
	%cd =l sub %v, 4
	storel %cd, %pchar
	jmp @mapped
@m_sym
	%eq62 =w ceql %v, 62
	jnz %eq62, @m_plus, @m_slash
@m_plus
	storel 43, %pchar
	jmp @mapped
@m_slash
	storel 47, %pchar
	jmp @mapped
@mapped
	%ch =l loadl %pchar
	%op =l loadl %pout
	storeb %ch, %op
	%op1 =l add %op, 1
	storel %op1, %pout
	%idxn =l add %idx, 1
	storel %idxn, %pidx
	%more =w csltl %idxn, 4
	jnz %more, @encchar, @dopad
@dopad
	%pad =l loadl %ppad
	%haspad =w cnel %pad, 0
	jnz %haspad, @writepad, @loop
@writepad
	%op2 =l loadl %pout
	%plast =l sub %op2, 1
	storeb 61, %plast
	%ispad2 =w ceql %pad, 2
	jnz %ispad2, @writepad2, @loop
@writepad2
	%plast2 =l sub %op2, 2
	storeb 61, %plast2
	jmp @loop
@done
	%opend =l loadl %pout
	%res =l sub %opend, %out
	ret %res
}
