export function l $base64_encode(l %src, l %n, l %out) {
@start
	%z =w ceql %n, 0
	jnz %z, @ret0, @loop
@ret0
	ret 0
@loop
	%ip =l phi @start %src, @aftergrp %ipn
	%op =l phi @start %out, @aftergrp %opn
	%left =l phi @start %n, @aftergrp %leftn
	%ge3 =w csgel %left, 3
	jnz %ge3, @full, @tail
@full
	%b0 =l loadub %ip
	%ip1 =l add %ip, 1
	%b1 =l loadub %ip1
	%ip2 =l add %ip, 2
	%b2 =l loadub %ip2
	%s0 =l shl %b0, 16
	%s1 =l shl %b1, 8
	%s01 =l or %s0, %s1
	%bits =l or %s01, %b2
	jmp @emit
@tail
	%is0 =w ceql %left, 0
	jnz %is0, @done, @tailnz
@tailnz
	%is1 =w ceql %left, 1
	jnz %is1, @tail1, @tail2
@tail1
	%b0t1 =l loadub %ip
	%bitst1 =l shl %b0t1, 16
	jmp @emit
@tail2
	%b0t2 =l loadub %ip
	%ip1t2 =l add %ip, 1
	%b1t2 =l loadub %ip1t2
	%s0t2 =l shl %b0t2, 16
	%s1t2 =l shl %b1t2, 8
	%bitst2 =l or %s0t2, %s1t2
	jmp @emit
@emit
	%ebits =l phi @full %bits, @emit %ebits, @tail1 %bitst1, @tail2 %bitst2
	%eshift =l phi @full 18, @emit %eshiftn, @tail1 18, @tail2 18
	%eop =l phi @full %op, @emit %eopn, @tail1 %op, @tail2 %op
	%ecnt =l phi @full 4, @emit %ecntn, @tail1 2, @tail2 3
	%epad =l phi @full 0, @emit %epad, @tail1 2, @tail2 1
	%eip =l phi @full %ip, @emit %eip, @tail1 %ip, @tail2 %ip
	%eleft =l phi @full %left, @emit %eleft, @tail1 %left, @tail2 %left
	%eadd =l phi @full 3, @emit %eadd, @tail1 1, @tail2 2
	%tmpsh =l shr %ebits, %eshift
	%idx =l and %tmpsh, 63
	%lt26 =w csltl %idx, 26
	%lt26l =l extuw %lt26
	%not26 =l sub 1, %lt26l
	%mul6 =l mul %not26, 6
	%off1 =l add 65, %mul6
	%lt52 =w csltl %idx, 52
	%lt52l =l extuw %lt52
	%not52 =l sub 1, %lt52l
	%mul75 =l mul %not52, 75
	%off2 =l sub %off1, %mul75
	%lt62 =w csltl %idx, 62
	%lt62l =l extuw %lt62
	%chn =l add %idx, %off2
	%ch1 =l mul %chn, %lt62l
	%eq62 =w ceql %idx, 62
	%eq62l =l extuw %eq62
	%ch2 =l mul %eq62l, 43
	%eq63 =w ceql %idx, 63
	%eq63l =l extuw %eq63
	%ch3 =l mul %eq63l, 47
	%ch12 =l add %ch1, %ch2
	%ch =l add %ch12, %ch3
	storeb %ch, %eop
	%eopn =l add %eop, 1
	%eshiftn =l sub %eshift, 6
	%ecntn =l sub %ecnt, 1
	%more =w csgtl %ecntn, 0
	jnz %more, @emit, @pad
@pad
	%ppad =l phi @emit %epad, @padw %ppadn
	%pop =l phi @emit %eopn, @padw %popn
	%pip =l phi @emit %eip, @padw %pip
	%pleft =l phi @emit %eleft, @padw %pleft
	%padd =l phi @emit %eadd, @padw %padd
	%need =w csgtl %ppad, 0
	jnz %need, @padw, @aftergrp
@padw
	%eqsign =l copy 61
	storeb %eqsign, %pop
	%popn =l add %pop, 1
	%ppadn =l sub %ppad, 1
	jmp @pad
@aftergrp
	%ipn =l add %pip, %padd
	%leftn =l sub %pleft, %padd
	%opn =l copy %pop
	jmp @loop
@done
	%len =l sub %op, %out
	ret %len
}
