export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%pi =l alloc8 8
	%pj =l alloc8 8
	%pt =l alloc8 8
	%psum =l alloc8 8
	%pk =l alloc8 8
	%psrc1 =l alloc8 8
	%psrc2 =l alloc8 8
	%pdst =l alloc8 8
	%pcont =l alloc8 8
	%pbase =l alloc8 8
	%pscr =l alloc8 8
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%n2 =l add %nn, %nn
	%tmp1 =l add %tmp, %nn8
	storel %k, %pk
	storel %tmp, %pbase
	storel %tmp1, %pscr
	storel 0, %pi
	jmp @idi
@idi
	%idi_i =l loadl %pi
	%idi_cmp =w csltl %idi_i, %n
	jnz %idi_cmp, @idibody, @iddone
@idibody
	storel 0, %pj
	jmp @idj
@idj
	%idj_j =l loadl %pj
	%idj_cmp =w csltl %idj_j, %n
	jnz %idj_cmp, @idjbody, @idinext
@idjbody
	%idjb_i =l loadl %pi
	%idjb_j =l loadl %pj
	%idjb_eq =w ceql %idjb_i, %idjb_j
	jnz %idjb_eq, @idoneval, @idzeroval
@idoneval
	%idov_i =l loadl %pi
	%idov_j =l loadl %pj
	%idov_in =l mul %idov_i, %n
	%idov_idx =l add %idov_in, %idov_j
	%idov_off =l mul %idov_idx, 8
	%idov_p =l add %out, %idov_off
	storel 1, %idov_p
	jmp @idjinc
@idzeroval
	%idzv_i =l loadl %pi
	%idzv_j =l loadl %pj
	%idzv_in =l mul %idzv_i, %n
	%idzv_idx =l add %idzv_in, %idzv_j
	%idzv_off =l mul %idzv_idx, 8
	%idzv_p =l add %out, %idzv_off
	storel 0, %idzv_p
	jmp @idjinc
@idjinc
	%idji_j =l loadl %pj
	%idji_j1 =l add %idji_j, 1
	storel %idji_j1, %pj
	jmp @idj
@idinext
	%idin_i =l loadl %pi
	%idin_i1 =l add %idin_i, 1
	storel %idin_i1, %pi
	jmp @idi
@iddone
	%idd_k =l loadl %pk
	%idd_kz =w ceql %idd_k, 0
	jnz %idd_kz, @zero_tmp, @copya_start
@copya_start
	storel 0, %pi
	jmp @copya
@copya
	%ca_i =l loadl %pi
	%ca_cmp =w csltl %ca_i, %nn
	jnz %ca_cmp, @copyabody, @powloop
@copyabody
	%cab_i =l loadl %pi
	%cab_off =l mul %cab_i, 8
	%cab_ps =l add %a, %cab_off
	%cab_pd =l add %tmp, %cab_off
	%cab_v =l loadl %cab_ps
	storel %cab_v, %cab_pd
	%cab_i1 =l add %cab_i, 1
	storel %cab_i1, %pi
	jmp @copya
@powloop
	%pl_k =l loadl %pk
	%pl_kz =w ceql %pl_k, 0
	jnz %pl_kz, @zero_tmp, @powbody
@powbody
	%pb_k =l loadl %pk
	%pb_bit =l and %pb_k, 1
	%pb_bz =w ceql %pb_bit, 0
	jnz %pb_bz, @do_square, @do_mul
@do_mul
	%dm_base =l loadl %pbase
	%dm_scr =l loadl %pscr
	storel %out, %psrc1
	storel %dm_base, %psrc2
	storel %dm_scr, %pdst
	storel 1, %pcont
	jmp @matmul
@do_square
	%ds_base =l loadl %pbase
	%ds_scr =l loadl %pscr
	storel %ds_base, %psrc1
	storel %ds_base, %psrc2
	storel %ds_scr, %pdst
	storel 0, %pcont
	jmp @matmul
@matmul
	storel 0, %pi
	jmp @mi
@mi
	%mi_i =l loadl %pi
	%mi_cmp =w csltl %mi_i, %n
	jnz %mi_cmp, @mibody, @matmul_done
@mibody
	storel 0, %pj
	jmp @mj
@mj
	%mj_j =l loadl %pj
	%mj_cmp =w csltl %mj_j, %n
	jnz %mj_cmp, @mjbody, @minext
@mjbody
	storel 0, %psum
	storel 0, %pt
	jmp @mt
@mt
	%mt_t =l loadl %pt
	%mt_cmp =w csltl %mt_t, %n
	jnz %mt_cmp, @mtbody, @mstore
@mtbody
	%mtb_src1 =l loadl %psrc1
	%mtb_i =l loadl %pi
	%mtb_t =l loadl %pt
	%mtb_in =l mul %mtb_i, %n
	%mtb_idxa =l add %mtb_in, %mtb_t
	%mtb_offa =l mul %mtb_idxa, 8
	%mtb_pa =l add %mtb_src1, %mtb_offa
	%mtb_va =l loadl %mtb_pa
	%mtb_src2 =l loadl %psrc2
	%mtb_j =l loadl %pj
	%mtb_tn =l mul %mtb_t, %n
	%mtb_idxb =l add %mtb_tn, %mtb_j
	%mtb_offb =l mul %mtb_idxb, 8
	%mtb_pb =l add %mtb_src2, %mtb_offb
	%mtb_vb =l loadl %mtb_pb
	%mtb_prod =l mul %mtb_va, %mtb_vb
	%mtb_s =l loadl %psum
	%mtb_s1 =l add %mtb_s, %mtb_prod
	storel %mtb_s1, %psum
	%mtb_t1 =l add %mtb_t, 1
	storel %mtb_t1, %pt
	jmp @mt
@mstore
	%ms_dst =l loadl %pdst
	%ms_i =l loadl %pi
	%ms_j =l loadl %pj
	%ms_in =l mul %ms_i, %n
	%ms_idxc =l add %ms_in, %ms_j
	%ms_offc =l mul %ms_idxc, 8
	%ms_pc =l add %ms_dst, %ms_offc
	%ms_s =l loadl %psum
	storel %ms_s, %ms_pc
	%ms_j1 =l add %ms_j, 1
	storel %ms_j1, %pj
	jmp @mj
@minext
	%mn_i =l loadl %pi
	%mn_i1 =l add %mn_i, 1
	storel %mn_i1, %pi
	jmp @mi
@matmul_done
	%md_cont =l loadl %pcont
	%md_is1 =w ceql %md_cont, 1
	jnz %md_is1, @after_mul, @after_square
@after_mul
	storel 0, %pi
	jmp @copyout
@copyout
	%co_i =l loadl %pi
	%co_cmp =w csltl %co_i, %nn
	jnz %co_cmp, @copyoutbody, @do_square
@copyoutbody
	%cob_i =l loadl %pi
	%cob_scr =l loadl %pscr
	%cob_off =l mul %cob_i, 8
	%cob_ps =l add %cob_scr, %cob_off
	%cob_pd =l add %out, %cob_off
	%cob_v =l loadl %cob_ps
	storel %cob_v, %cob_pd
	%cob_i1 =l add %cob_i, 1
	storel %cob_i1, %pi
	jmp @copyout
@after_square
	%as_base =l loadl %pbase
	%as_scr =l loadl %pscr
	storel %as_scr, %pbase
	storel %as_base, %pscr
	%as_k =l loadl %pk
	%as_k1 =l shr %as_k, 1
	storel %as_k1, %pk
	jmp @powloop
@zero_tmp
	storel 0, %pi
	jmp @zt
@zt
	%zt_i =l loadl %pi
	%zt_cmp =w csltl %zt_i, %n2
	jnz %zt_cmp, @ztbody, @done
@ztbody
	%ztb_i =l loadl %pi
	%ztb_off =l mul %ztb_i, 8
	%ztb_p =l add %tmp, %ztb_off
	storel 0, %ztb_p
	%ztb_i1 =l add %ztb_i, 1
	storel %ztb_i1, %pi
	jmp @zt
@done
	ret 0
}
