.text
.globl _start

_start:
    adrp x21, inbuf
    add  x21, x21, :lo12:inbuf
    mov  x20, #0
rd_loop:
    movz x2, #1, lsl #16
    sub  x2, x2, x20
    cbz  x2, rd_done
    mov  x0, #0
    add  x1, x21, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le rd_done
    add  x20, x20, x0
    b    rd_loop
rd_done:
    adrp x22, stk
    add  x22, x22, :lo12:stk
    mov  x19, #0
line_start:
    cmp  x19, x20
    b.ge done_all
    mov  x23, #0
    mov  x24, #0
tok_loop:
    cmp  x19, x20
    b.ge finish_eof
    ldrb w10, [x21, x19]
    cmp  w10, #10
    b.eq finish_nl
    cmp  w10, #32
    b.eq skip_sep
    cmp  w10, #9
    b.eq skip_sep
    mov  x12, x19
tok_scan:
    add  x19, x19, #1
    cmp  x19, x20
    b.ge tok_end
    ldrb w10, [x21, x19]
    cmp  w10, #10
    b.eq tok_end
    cmp  w10, #32
    b.eq tok_end
    cmp  w10, #9
    b.ne tok_scan
tok_end:
    bl   process_token
    b    tok_loop
skip_sep:
    add  x19, x19, #1
    b    tok_loop
finish_nl:
    add  x19, x19, #1
finish_eof:
    bl   emit_result
    b    line_start
done_all:
    mov  x0, #0
    mov  x8, #93
    svc  #0

// process_token: token is bytes [x21+x12, x21+x19); uses x22 stack, x23 depth, x24 error
process_token:
    cbnz x24, pt_ret
    sub  x14, x19, x12
    add  x13, x21, x12
    ldrb w10, [x13]
    cmp  x14, #1
    b.ne pt_lit
    cmp  w10, #43
    b.eq pt_isop
    cmp  w10, #45
    b.eq pt_isop
    cmp  w10, #42
    b.eq pt_isop
    cmp  w10, #47
    b.eq pt_isop
    cmp  w10, #37
    b.eq pt_isop
    b    pt_lit
pt_isop:
    mov  w15, w10
    cmp  x23, #2
    b.lt pt_bad
    sub  x9, x23, #1
    ldr  x11, [x22, x9, lsl #3]
    sub  x9, x23, #2
    ldr  x10, [x22, x9, lsl #3]
    cmp  w15, #43
    b.eq pt_add
    cmp  w15, #45
    b.eq pt_sub
    cmp  w15, #42
    b.eq pt_mul
    cmp  w15, #47
    b.eq pt_div
    b    pt_mod
pt_add:
    add  x16, x10, x11
    b    pt_pop_store
pt_sub:
    sub  x16, x10, x11
    b    pt_pop_store
pt_mul:
    mul  x16, x10, x11
    b    pt_pop_store
pt_div:
    cbz  x11, pt_bad
    cmn  x11, #1
    b.ne pt_dv_ok
    mov  x9, #1
    lsl  x9, x9, #63
    cmp  x10, x9
    b.eq pt_bad
pt_dv_ok:
    sdiv x16, x10, x11
    b    pt_pop_store
pt_mod:
    cbz  x11, pt_bad
    cmn  x11, #1
    b.ne pt_md_ok
    mov  x9, #1
    lsl  x9, x9, #63
    cmp  x10, x9
    b.eq pt_bad
pt_md_ok:
    sdiv x16, x10, x11
    msub x16, x16, x11, x10
    b    pt_pop_store
pt_pop_store:
    sub  x23, x23, #1
    sub  x9, x23, #1
    str  x16, [x22, x9, lsl #3]
    ret
pt_lit:
    ldrb w10, [x13]
    mov  x15, #0
    cmp  w10, #45
    b.ne pt_digits
    mov  x15, #1
    add  x13, x13, #1
    sub  x14, x14, #1
pt_digits:
    cmp  x14, #1
    b.lt pt_bad
    cmp  x14, #18
    b.gt pt_bad
    mov  x16, #0
    mov  x9, #10
pt_dloop:
    ldrb w10, [x13], #1
    sub  w10, w10, #48
    cmp  w10, #9
    b.hi pt_bad
    madd x16, x16, x9, x10
    sub  x14, x14, #1
    cbnz x14, pt_dloop
    cbz  x15, pt_push
    neg  x16, x16
pt_push:
    str  x16, [x22, x23, lsl #3]
    add  x23, x23, #1
    ret
pt_bad:
    mov  x24, #1
pt_ret:
    ret

// emit_result: prints stk[0] if valid single value, else "error\n"
emit_result:
    stp  x30, xzr, [sp, #-16]!
    cbnz x24, er_err
    cmp  x23, #1
    b.ne er_err
    ldr  x16, [x22]
    adrp x17, numbuf
    add  x17, x17, :lo12:numbuf
    add  x17, x17, #32
    mov  x15, #0
    cmp  x16, #0
    b.ge er_mag
    mov  x15, #1
    neg  x16, x16
er_mag:
    mov  x9, #10
er_dig:
    udiv x10, x16, x9
    msub x11, x10, x9, x16
    add  x11, x11, #48
    sub  x17, x17, #1
    strb w11, [x17]
    mov  x16, x10
    cbnz x16, er_dig
    cbz  x15, er_nosign
    mov  w11, #45
    sub  x17, x17, #1
    strb w11, [x17]
er_nosign:
    mov  w11, #10
    sub  x17, x17, #1
    strb w11, [x17]
    adrp x9, numbuf
    add  x9, x9, :lo12:numbuf
    add  x9, x9, #32
    sub  x1, x9, x17
    mov  x0, x17
    bl   write_buf
    b    er_done
er_err:
    adrp x0, errmsg
    add  x0, x0, :lo12:errmsg
    mov  x1, #6
    bl   write_buf
er_done:
    ldp  x30, xzr, [sp], #16
    ret

// write_buf: x0 = pointer, x1 = length; writes all to stdout
write_buf:
    mov  x9, x0
    mov  x10, x1
wb_loop:
    cbz  x10, wb_ret
    mov  x0, #1
    mov  x1, x9
    mov  x2, x10
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le wb_ret
    add  x9, x9, x0
    sub  x10, x10, x0
    b    wb_loop
wb_ret:
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.section .bss
.balign 8
inbuf:
    .zero 65536
.balign 8
stk:
    .zero 32768
numbuf:
    .zero 32
