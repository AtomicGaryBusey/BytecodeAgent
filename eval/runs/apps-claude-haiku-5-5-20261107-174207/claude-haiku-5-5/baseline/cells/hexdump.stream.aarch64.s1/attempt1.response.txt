```
// hexdump: canonical hex+ASCII dump of standard input (AArch64 Linux, no libc)
    .text
    .globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    adrp x27, hextab
    add x27, x27, :lo12:hextab
    mov x20, #0                 // total bytes read
read_loop:
    add x1, x19, x20
    movz x2, #1, lsl #16        // 65536
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    adrp x22, linebuf
    add x22, x22, :lo12:linebuf
    mov x21, #0                 // current offset
row_loop:
    cmp x21, x20
    b.hs done_rows
    sub x23, x20, x21           // remaining
    mov x24, #16
    cmp x23, #16
    csel x24, x23, x24, cc      // n = min(remaining, 16)
    add x26, x19, x21           // source pointer
    mov x25, x22                // line pointer
    // offset: 8 hex digits
    lsr w0, w21, #24
    bl hex2
    lsr w0, w21, #16
    bl hex2
    lsr w0, w21, #8
    bl hex2
    mov w0, w21
    bl hex2
    // two spaces
    mov w0, #32
    bl emit
    bl emit
    // hex bytes
    mov x28, #0
hex_loop:
    cmp x28, x24
    b.hs hex_pad
    ldrb w0, [x26, x28]
    bl hex2
    mov w0, #32
    bl emit
    b hex_after
hex_pad:
    mov w0, #32
    bl emit
    bl emit
    bl emit
hex_after:
    cmp x28, #7
    b.ne hex_next
    mov w0, #32
    bl emit
hex_next:
    add x28, x28, #1
    cmp x28, #16
    b.lo hex_loop
    // separator " |"
    mov w0, #32
    bl emit
    mov w0, #124
    bl emit
    // ASCII column
    mov x28, #0
chr_loop:
    cmp x28, x24
    b.hs chr_done
    ldrb w0, [x26, x28]
    cmp w0, #32
    b.lo chr_dot
    cmp w0, #126
    b.hi chr_dot
    b chr_put
chr_dot:
    mov w0, #46
chr_put:
    bl emit
    add x28, x28, #1
    b chr_loop
chr_done:
    mov w0, #124
    bl emit
    mov w0, #10
    bl emit
    bl flush
    add x21, x21, x24
    b row_loop
done_rows:
    mov x25, x22
    lsr w0, w20, #24
    bl hex2
    lsr w0, w20, #16
    bl hex2
    lsr w0, w20, #8
    bl hex2
    mov w0, w20
    bl hex2
    mov w0, #10
    bl emit
    bl flush
    mov x0, #0
    mov x8, #93
    svc #0

// emit: store byte w0 at [x25], advance x25
emit:
    strb w0, [x25], #1
    ret

// hex2: store two lowercase hex digits of low byte of w0 at [x25], advance x25
hex2:
    and w0, w0, #255
    lsr w1, w0, #4
    ldrb w1, [x27, x1]
    strb w1, [x25], #1
    and w1, w0, #15
    ldrb w1, [x27, x1]
    strb w1, [x25], #1
    ret

// flush: write bytes from x22 up to x25 to stdout, handling partial writes
flush:
    mov x9, x22
    sub x10, x25, x22
flush_loop:
    cbz x10, flush_done
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le fail
    add x9, x9, x0
    sub x10, x10, x0
    b flush_loop
flush_done:
    ret
fail:
    mov x0, #1
    mov x8, #93
    svc #0

    .section .rodata
hextab:
    .ascii "0123456789abcdef"

    .bss
    .align 4
inbuf:
    .zero 65536
linebuf:
    .zero 128
```
