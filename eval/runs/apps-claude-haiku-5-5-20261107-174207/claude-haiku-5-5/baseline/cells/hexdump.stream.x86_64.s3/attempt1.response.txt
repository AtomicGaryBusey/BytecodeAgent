.section .rodata
hexdig:
  .ascii "0123456789abcdef"
.bss
inbuf:
  .zero 65536
linebuf:
  .zero 128
.text
.globl _start
_start:
  xor r12d, r12d
  lea r13, [rip + inbuf]
read_loop:
  mov rdx, 65536
  sub rdx, r12
  jz read_done
  lea rsi, [r13 + r12]
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jle read_done
  add r12, rax
  jmp read_loop
read_done:
  xor r14d, r14d
row_loop:
  cmp r14, r12
  jae rows_done
  mov rbx, r12
  sub rbx, r14
  cmp rbx, 16
  jbe row_n_ok
  mov ebx, 16
row_n_ok:
  lea rdi, [rip + linebuf]
  mov rax, r14
  mov ecx, 8
  call hex_out
  mov byte ptr [rdi], 0x20
  mov byte ptr [rdi + 1], 0x20
  add rdi, 2
  lea rsi, [r13 + r14]
  xor r10d, r10d
hex_loop:
  cmp r10, rbx
  jae hex_blank
  movzx eax, byte ptr [rsi + r10]
  mov ecx, 2
  call hex_out
  mov byte ptr [rdi], 0x20
  inc rdi
  jmp hex_after
hex_blank:
  mov byte ptr [rdi], 0x20
  mov byte ptr [rdi + 1], 0x20
  mov byte ptr [rdi + 2], 0x20
  add rdi, 3
hex_after:
  cmp r10, 7
  jne hex_nomid
  mov byte ptr [rdi], 0x20
  inc rdi
hex_nomid:
  inc r10
  cmp r10, 16
  jb hex_loop
  mov byte ptr [rdi], 0x20
  mov byte ptr [rdi + 1], 0x7c
  add rdi, 2
  xor r10d, r10d
chr_loop:
  cmp r10, rbx
  jae chr_done
  movzx eax, byte ptr [rsi + r10]
  cmp eax, 0x20
  jb chr_dot
  cmp eax, 0x7e
  ja chr_dot
  jmp chr_store
chr_dot:
  mov eax, 0x2e
chr_store:
  mov byte ptr [rdi], al
  inc rdi
  inc r10
  jmp chr_loop
chr_done:
  mov byte ptr [rdi], 0x7c
  mov byte ptr [rdi + 1], 0x0a
  add rdi, 2
  lea rsi, [rip + linebuf]
  mov rdx, rdi
  sub rdx, rsi
  call write_all
  add r14, rbx
  jmp row_loop
rows_done:
  lea rdi, [rip + linebuf]
  mov rax, r12
  mov ecx, 8
  call hex_out
  mov byte ptr [rdi], 0x0a
  inc rdi
  lea rsi, [rip + linebuf]
  mov rdx, rdi
  sub rdx, rsi
  call write_all
  mov eax, 60
  xor edi, edi
  syscall

# rax = value, ecx = digit count, rdi = destination; advances rdi
hex_out:
  lea r8, [rip + hexdig]
  mov r9d, ecx
ho_loop:
  dec r9d
  mov ecx, r9d
  shl ecx, 2
  mov rdx, rax
  shr rdx, cl
  and edx, 15
  movzx edx, byte ptr [r8 + rdx]
  mov byte ptr [rdi], dl
  inc rdi
  test r9d, r9d
  jnz ho_loop
  ret

# rsi = buffer, rdx = length, writes all to stdout
write_all:
wa_loop:
  test rdx, rdx
  jz wa_ret
  mov eax, 1
  mov edi, 1
  syscall
  test rax, rax
  jle wa_ret
  add rsi, rax
  sub rdx, rax
  jmp wa_loop
wa_ret:
  ret
