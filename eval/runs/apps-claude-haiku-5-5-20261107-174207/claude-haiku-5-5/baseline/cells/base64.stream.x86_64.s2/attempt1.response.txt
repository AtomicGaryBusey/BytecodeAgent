```
.intel_syntax noprefix
.globl _start

.section .rodata
b64:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 70000

.text
_start:
    # read all of standard input into inbuf
    xor r12, r12
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    lea r15, [rip + outbuf]
    xor ebp, ebp

enc_loop:
    cmp r13, r12
    jae enc_done
    mov rdi, r12
    sub rdi, r13
    lea rsi, [rip + inbuf]
    add rsi, r13
    movzx ebx, byte ptr [rsi]
    xor ecx, ecx
    xor edx, edx
    cmp rdi, 2
    jb load_done
    movzx ecx, byte ptr [rsi + 1]
    cmp rdi, 3
    jb load_done
    movzx edx, byte ptr [rsi + 2]
load_done:
    # c0 = b0 >> 2
    mov eax, ebx
    shr eax, 2
    call lut
    call emit
    # c1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov esi, ecx
    shr esi, 4
    or eax, esi
    call lut
    call emit
    # c2 = ((b1 & 15) << 2) | (b2 >> 6), or '=' if no second byte
    cmp rdi, 2
    jb pad2
    mov eax, ecx
    and eax, 15
    shl eax, 2
    mov esi, edx
    shr esi, 6
    or eax, esi
    call lut
    call emit
    jmp c3_part
pad2:
    mov al, 61
    call emit
c3_part:
    # c3 = b2 & 63, or '=' if no third byte
    cmp rdi, 3
    jb pad3
    mov eax, edx
    and eax, 63
    call lut
    call emit
    jmp next_group
pad3:
    mov al, 61
    call emit
next_group:
    add r13, 3
    jmp enc_loop

enc_done:
    # final newline if the last line is non-empty
    test rbp, rbp
    jz write_out
    mov byte ptr [r15], 10
    inc r15

write_out:
    lea rsi, [rip + outbuf]
    mov rdx, r15
    sub rdx, rsi
wloop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle fail
    add rsi, rax
    sub rdx, rax
    jmp wloop

finish:
    mov eax, 60
    xor edi, edi
    syscall

fail:
    mov eax, 60
    mov edi, 1
    syscall

# lut: eax = 6-bit value -> al = base64 character
lut:
    lea r10, [rip + b64]
    movzx eax, byte ptr [r10 + rax]
    ret

# emit: store al at [r15], advance r15, insert newline after 76 chars
emit:
    mov byte ptr [r15], al
    inc r15
    inc rbp
    cmp rbp, 76
    jne emit_done
    mov byte ptr [r15], 10
    inc r15
    xor ebp, ebp
emit_done:
    ret
```
