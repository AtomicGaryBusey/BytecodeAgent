.section .bss
.align 4
inbuf:  .zero 65536
outbuf: .zero 65600

.text
.globl _start
_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0                    // n = bytes read so far

read_loop:
    mov  x0, #0
    add  x1, x19, x20
    mov  x2, #65536
    sub  x2, x2, x20
    mov  x8, #63
    svc  #0
    cbz  x0, read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    adrp x21, outbuf
    add  x21, x21, :lo12:outbuf
    mov  x22, #0                    // out length
    mov  x23, #0                    // line start
    mov  x24, #0                    // i

scan:
    cmp  x24, x20
    b.ge scan_end
    ldrb w0, [x19, x24]
    cmp  w0, #10
    b.ne next

    mov  x25, x24                   // j = i
rev:
    cmp  x25, x23
    b.le rev_done
    sub  x25, x25, #1
    ldrb w1, [x19, x25]
    strb w1, [x21, x22]
    add  x22, x22, #1
    b    rev
rev_done:
    mov  w1, #10
    strb w1, [x21, x22]
    add  x22, x22, #1
    add  x23, x24, #1

next:
    add  x24, x24, #1
    b    scan

scan_end:
    cmp  x23, x20
    b.ge emit

    mov  x25, x20                   // j = n
rev2:
    cmp  x25, x23
    b.le rev2_done
    sub  x25, x25, #1
    ldrb w1, [x19, x25]
    strb w1, [x21, x22]
    add  x22, x22, #1
    b    rev2
rev2_done:
    mov  w1, #10
    strb w1, [x21, x22]
    add  x22, x22, #1

emit:
    mov  x26, #0                    // offset written
wloop:
    cmp  x26, x22
    b.ge done
    mov  x0, #1
    add  x1, x21, x26
    sub  x2, x22, x26
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le done
    add  x26, x26, x0
    b    wloop

done:
    mov  x0, #0
    mov  x8, #93
    svc  #0
