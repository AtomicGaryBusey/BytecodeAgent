.text
.globl _start

.equ ST_TOPB, 0
.equ ST_TOPA, 1
.equ ST_ARRVC, 2
.equ ST_ARRV, 3
.equ ST_OBJKC, 4
.equ ST_OBJK, 5
.equ ST_OBJC, 6
.equ ST_OBJV, 7
.equ ST_AFT, 8
.equ ST_STR, 9
.equ ST_ESC, 10
.equ ST_U1, 11
.equ ST_LIT, 15
.equ ST_NEG, 16
.equ ST_ZERO, 17
.equ ST_INT, 18
.equ ST_DOT, 19
.equ ST_FRAC, 20
.equ ST_EXPM, 21
.equ ST_EXPS, 22
.equ ST_EXPD, 23

_start:
	mov x21, #0
	adrp x22, inbuf
	add x22, x22, :lo12:inbuf
read_loop:
	mov x0, #0
	mov x1, x22
	mov x2, #4096
	mov x8, #63
	svc #0
	cmp x0, #0
	b.le at_eof
	mov x20, x0
	mov x19, #0
byte_loop:
	cmp x19, x20
	b.ge read_loop
	ldrb w0, [x22, x19]
	mov x1, x21
	bl process_byte
	add x19, x19, #1
	add x21, x21, #1
	b byte_loop

at_eof:
	adrp x10, state_var
	add x10, x10, :lo12:state_var
	ldr w3, [x10]
	cmp w3, #ST_TOPA
	b.eq do_ok
	cmp w3, #ST_ZERO
	b.eq eof_num
	cmp w3, #ST_INT
	b.eq eof_num
	cmp w3, #ST_FRAC
	b.eq eof_num
	cmp w3, #ST_EXPD
	b.eq eof_num
	b eof_fail
eof_num:
	adrp x11, depth_var
	add x11, x11, :lo12:depth_var
	ldr w4, [x11]
	cbnz w4, eof_fail
	b do_ok
eof_fail:
	mov x0, x21
	b fail
do_ok:
	mov x0, #1
	adrp x1, okmsg
	add x1, x1, :lo12:okmsg
	mov x2, #3
	mov x8, #64
	svc #0
	mov x0, #0
	mov x8, #93
	svc #0

// x0 = byte, x1 = position
process_byte:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	str x0, [sp, #16]
	str x1, [sp, #24]
pb_again:
	ldr x0, [sp, #16]
	ldr x1, [sp, #24]
	bl step
	cbnz w0, pb_again
	ldp x29, x30, [sp], #32
	ret

// x0 = byte, x1 = position (preserved). returns w0: 0 consumed, 1 reprocess
step:
	adrp x10, state_var
	add x10, x10, :lo12:state_var
	ldr w3, [x10]
	cmp w3, #ST_TOPB
	b.eq s_topb
	cmp w3, #ST_TOPA
	b.eq s_topa
	cmp w3, #ST_ARRVC
	b.eq s_arrvc
	cmp w3, #ST_ARRV
	b.eq s_arrv
	cmp w3, #ST_OBJKC
	b.eq s_objkc
	cmp w3, #ST_OBJK
	b.eq s_objk
	cmp w3, #ST_OBJC
	b.eq s_objc
	cmp w3, #ST_OBJV
	b.eq s_objv
	cmp w3, #ST_AFT
	b.eq s_aft
	cmp w3, #ST_STR
	b.eq s_str
	cmp w3, #ST_ESC
	b.eq s_esc
	sub w2, w3, #ST_U1
	cmp w2, #3
	b.ls s_u
	cmp w3, #ST_LIT
	b.eq s_lit
	cmp w3, #ST_NEG
	b.eq s_neg
	cmp w3, #ST_ZERO
	b.eq s_zero
	cmp w3, #ST_INT
	b.eq s_int
	cmp w3, #ST_DOT
	b.eq s_dot
	cmp w3, #ST_FRAC
	b.eq s_frac
	cmp w3, #ST_EXPM
	b.eq s_expm
	cmp w3, #ST_EXPS
	b.eq s_exps
	b s_expd

s_topb:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	b vstart

s_topa:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	b fail_c

s_arrvc:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	cmp w0, #0x5D
	b.eq do_pop
	b vstart

s_arrv:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	b vstart

s_objkc:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	cmp w0, #0x7D
	b.eq do_pop
	cmp w0, #0x22
	b.eq key_start
	b fail_c

s_objk:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	cmp w0, #0x22
	b.eq key_start
	b fail_c

s_objc:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	cmp w0, #0x3A
	b.ne fail_c
	mov w6, #ST_OBJV
	b setret0

s_objv:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	b vstart

s_aft:
	cmp w0, #0x20
	b.eq ret0
	cmp w0, #0x09
	b.eq ret0
	cmp w0, #0x0A
	b.eq ret0
	cmp w0, #0x0D
	b.eq ret0
	adrp x11, depth_var
	add x11, x11, :lo12:depth_var
	ldr w4, [x11]
	sub w4, w4, #1
	adrp x12, stk
	add x12, x12, :lo12:stk
	ldrb w5, [x12, x4]
	cmp w5, #0x5B
	b.ne aft_obj
	cmp w0, #0x2C
	b.eq aft_comma_arr
	cmp w0, #0x5D
	b.eq do_pop
	b fail_c
aft_obj:
	cmp w0, #0x2C
	b.eq aft_comma_obj
	cmp w0, #0x7D
	b.eq do_pop
	b fail_c
aft_comma_arr:
	mov w6, #ST_ARRV
	b setret0
aft_comma_obj:
	mov w6, #ST_OBJK
	b setret0

s_str:
	cmp w0, #0x22
	b.eq str_end
	cmp w0, #0x5C
	b.eq str_bs
	cmp w0, #0x20
	b.lo fail_c
	b ret0
str_bs:
	mov w6, #ST_ESC
	b setret0
str_end:
	adrp x11, strkey_var
	add x11, x11, :lo12:strkey_var
	ldr w4, [x11]
	cbnz w4, str_endkey
	b done_ret
str_endkey:
	mov w6, #ST_OBJC
	b setret0

s_esc:
	cmp w0, #0x75
	b.eq esc_u
	cmp w0, #0x22
	b.eq esc_ok
	cmp w0, #0x5C
	b.eq esc_ok
	cmp w0, #0x2F
	b.eq esc_ok
	cmp w0, #0x62
	b.eq esc_ok
	cmp w0, #0x66
	b.eq esc_ok
	cmp w0, #0x6E
	b.eq esc_ok
	cmp w0, #0x72
	b.eq esc_ok
	cmp w0, #0x74
	b.eq esc_ok
	b fail_c
esc_ok:
	mov w6, #ST_STR
	b setret0
esc_u:
	mov w6, #ST_U1
	b setret0

// states 11..14: expect hex digits
s_u:
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls u_hex_ok
	orr w2, w0, #0x20
	sub w2, w2, #0x61
	cmp w2, #5
	b.hi fail_c
u_hex_ok:
	add w6, w3, #1
	cmp w6, #15
	b.ne u_store
	mov w6, #ST_STR
u_store:
	b setret0

s_lit:
	adrp x11, litp_var
	add x11, x11, :lo12:litp_var
	ldr x12, [x11]
	ldrb w13, [x12]
	cmp w13, w0
	b.ne fail_c
	add x12, x12, #1
	str x12, [x11]
	ldrb w13, [x12]
	cbz w13, lit_done
	b ret0
lit_done:
	b done_ret

s_neg:
	cmp w0, #0x30
	b.eq to_zero
	sub w2, w0, #0x31
	cmp w2, #8
	b.ls to_int
	b fail_c

s_zero:
	cmp w0, #0x2E
	b.eq to_dot
	cmp w0, #0x65
	b.eq to_expm
	cmp w0, #0x45
	b.eq to_expm
	b done_again

s_int:
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls ret0
	cmp w0, #0x2E
	b.eq to_dot
	cmp w0, #0x65
	b.eq to_expm
	cmp w0, #0x45
	b.eq to_expm
	b done_again

s_dot:
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls to_frac
	b fail_c

s_frac:
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls ret0
	cmp w0, #0x65
	b.eq to_expm
	cmp w0, #0x45
	b.eq to_expm
	b done_again

s_expm:
	cmp w0, #0x2B
	b.eq to_exps
	cmp w0, #0x2D
	b.eq to_exps
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls to_expd
	b fail_c

s_exps:
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls to_expd
	b fail_c

s_expd:
	sub w2, w0, #0x30
	cmp w2, #9
	b.ls ret0
	b done_again

to_zero:
	mov w6, #ST_ZERO
	b setret0
to_int:
	mov w6, #ST_INT
	b setret0
to_dot:
	mov w6, #ST_DOT
	b setret0
to_frac:
	mov w6, #ST_FRAC
	b setret0
to_expm:
	mov w6, #ST_EXPM
	b setret0
to_exps:
	mov w6, #ST_EXPS
	b setret0
to_expd:
	mov w6, #ST_EXPD
	b setret0

// value start dispatch: x0 = byte
vstart:
	cmp w0, #0x7B
	b.eq v_obj
	cmp w0, #0x5B
	b.eq v_arr
	cmp w0, #0x22
	b.eq v_str
	cmp w0, #0x2D
	b.eq v_neg
	cmp w0, #0x30
	b.eq v_zero
	sub w2, w0, #0x31
	cmp w2, #8
	b.ls v_int
	cmp w0, #0x74
	b.eq v_t
	cmp w0, #0x66
	b.eq v_f
	cmp w0, #0x6E
	b.eq v_n
	b fail_c
v_obj:
	mov w5, #0x7B
	mov w6, #ST_OBJKC
	b do_push
v_arr:
	mov w5, #0x5B
	mov w6, #ST_ARRVC
	b do_push
v_str:
	adrp x11, strkey_var
	add x11, x11, :lo12:strkey_var
	str wzr, [x11]
	mov w6, #ST_STR
	b setret0
v_neg:
	mov w6, #ST_NEG
	b setret0
v_zero:
	mov w6, #ST_ZERO
	b setret0
v_int:
	mov w6, #ST_INT
	b setret0
v_t:
	adrp x12, str_rue
	add x12, x12, :lo12:str_rue
	b lit_start
v_f:
	adrp x12, str_alse
	add x12, x12, :lo12:str_alse
	b lit_start
v_n:
	adrp x12, str_ull
	add x12, x12, :lo12:str_ull
	b lit_start
lit_start:
	adrp x11, litp_var
	add x11, x11, :lo12:litp_var
	str x12, [x11]
	mov w6, #ST_LIT
	b setret0

key_start:
	adrp x11, strkey_var
	add x11, x11, :lo12:strkey_var
	mov w5, #1
	str w5, [x11]
	mov w6, #ST_STR
	b setret0

// w5 = char, w6 = new state
do_push:
	adrp x11, depth_var
	add x11, x11, :lo12:depth_var
	ldr w4, [x11]
	adrp x12, stk
	add x12, x12, :lo12:stk
	strb w5, [x12, x4]
	add w4, w4, #1
	str w4, [x11]
	b setret0

do_pop:
	adrp x11, depth_var
	add x11, x11, :lo12:depth_var
	ldr w4, [x11]
	sub w4, w4, #1
	str w4, [x11]
	b done_ret

// done_again: value ended, byte must be reprocessed (returns 1)
done_again:
	mov w14, #1
	b done_common
// done_ret: value ended, byte consumed (returns 0)
done_ret:
	mov w14, #0
done_common:
	adrp x10, state_var
	add x10, x10, :lo12:state_var
	adrp x11, depth_var
	add x11, x11, :lo12:depth_var
	ldr w4, [x11]
	mov w5, #ST_AFT
	cbnz w4, done_store
	mov w5, #ST_TOPA
done_store:
	str w5, [x10]
	mov w0, w14
	ret

// w6 = new state; store and return 0
setret0:
	adrp x10, state_var
	add x10, x10, :lo12:state_var
	str w6, [x10]
ret0:
	mov w0, #0
	ret

fail_c:
	mov x0, x1
fail:
	// x0 = position; print "error at N\n" and exit 1
	adrp x10, numbuf
	add x10, x10, :lo12:numbuf
	mov x9, x0
	mov x12, #10
	add x11, x10, #24
	mov w13, #10
	strb w13, [x11]
fail_digits:
	udiv x14, x9, x12
	msub x15, x14, x12, x9
	add w15, w15, #0x30
	sub x11, x11, #1
	strb w15, [x11]
	mov x9, x14
	cbnz x9, fail_digits
	mov x0, #1
	adrp x1, errmsg
	add x1, x1, :lo12:errmsg
	mov x2, #9
	mov x8, #64
	svc #0
	add x2, x10, #25
	sub x2, x2, x11
	mov x1, x11
	mov x0, #1
	mov x8, #64
	svc #0
	mov x0, #1
	mov x8, #93
	svc #0

.section .rodata
okmsg:
	.ascii "ok\n"
errmsg:
	.ascii "error at "
str_rue:
	.asciz "rue"
str_alse:
	.asciz "alse"
str_ull:
	.asciz "ull"

.section .bss
.align 3
state_var:
	.zero 8
depth_var:
	.zero 8
strkey_var:
	.zero 8
litp_var:
	.zero 8
stk:
	.zero 4096
inbuf:
	.zero 4096
numbuf:
	.zero 32
