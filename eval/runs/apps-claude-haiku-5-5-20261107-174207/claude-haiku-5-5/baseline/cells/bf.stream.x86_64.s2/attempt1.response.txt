.text
.globl _start
_start:
    # read all of standard input into inbuf
    xor r12d, r12d
rd:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov edx, 65536
    sub rdx, r12
    jz rdone
    xor edi, edi
    xor eax, eax
    syscall
    cmp rax, -4
    je rd
    test rax, rax
    jle rdone
    add r12, rax
    jmp rd
rdone:
    # find first '!' (0x21)
    lea rsi, [rip + inbuf]
    xor ecx, ecx
bs:
    cmp rcx, r12
    jae bnone
    cmp byte ptr [rsi + rcx], 0x21
    je bfound
    inc rcx
    jmp bs
bnone:
    mov qword ptr [rip + progend], rcx
    mov qword ptr [rip + inpstart], r12
    mov qword ptr [rip + inpend], r12
    jmp parse_init
bfound:
    mov qword ptr [rip + progend], rcx
    lea rdx, [rcx + 1]
    mov qword ptr [rip + inpstart], rdx
    mov qword ptr [rip + inpend], r12

    # parse program: keep only commands as codes 0..7, match brackets
parse_init:
    lea rbx, [rip + ops]
    lea r10, [rip + bstk]
    lea r11, [rip + jumps]
    lea rsi, [rip + inbuf]
    xor ecx, ecx
    xor r8d, r8d
    xor r9d, r9d
    mov rdx, qword ptr [rip + progend]
ploop:
    cmp rcx, rdx
    jae pend
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 0x3E
    jne c1
    xor eax, eax
    jmp emit
c1:
    cmp al, 0x3C
    jne c2
    mov eax, 1
    jmp emit
c2:
    cmp al, 0x2B
    jne c3
    mov eax, 2
    jmp emit
c3:
    cmp al, 0x2D
    jne c4
    mov eax, 3
    jmp emit
c4:
    cmp al, 0x2E
    jne c5
    mov eax, 4
    jmp emit
c5:
    cmp al, 0x2C
    jne c6
    mov eax, 5
    jmp emit
c6:
    cmp al, 0x5B
    jne c7
    mov eax, 6
    jmp emit
c7:
    cmp al, 0x5D
    jne ploop
    mov eax, 7
emit:
    mov byte ptr [rbx + r8], al
    cmp eax, 6
    je pushb
    cmp eax, 7
    je closeb
    inc r8
    jmp ploop
pushb:
    mov dword ptr [r10 + r9*4], r8d
    inc r9
    inc r8
    jmp ploop
closeb:
    test r9, r9
    jz fail
    dec r9
    mov eax, dword ptr [r10 + r9*4]
    mov dword ptr [r11 + rax*4], r8d
    mov dword ptr [r11 + r8*4], eax
    inc r8
    jmp ploop
pend:
    test r9, r9
    jnz fail
    mov qword ptr [rip + nops], r8

    # run setup
run_init:
    lea rbx, [rip + ops]
    xor r12d, r12d
    lea r13, [rip + tape]
    lea r14, [rip + inbuf]
    add r14, qword ptr [rip + inpstart]
    lea r15, [rip + inbuf]
    add r15, qword ptr [rip + inpend]
    xor ebp, ebp

dispatch:
    cmp r12, qword ptr [rip + nops]
    jae finish
    movzx eax, byte ptr [rbx + r12]
    lea rcx, [rip + jtab]
    jmp qword ptr [rcx + rax*8]

next:
    inc r12
    jmp dispatch

h_right:
    inc r13
    jmp next
h_left:
    dec r13
    jmp next
h_plus:
    inc byte ptr [r13]
    jmp next
h_minus:
    dec byte ptr [r13]
    jmp next
h_out:
    mov al, byte ptr [r13]
    lea rcx, [rip + outbuf]
    mov byte ptr [rcx + rbp], al
    inc rbp
    cmp rbp, 65536
    jne next
    call flush
    jmp next
h_in:
    xor eax, eax
    cmp r14, r15
    jae hi_store
    mov al, byte ptr [r14]
    inc r14
hi_store:
    mov byte ptr [r13], al
    jmp next
h_open:
    cmp byte ptr [r13], 0
    jne next
    lea rcx, [rip + jumps]
    mov r12d, dword ptr [rcx + r12*4]
    inc r12
    jmp dispatch
h_close:
    cmp byte ptr [r13], 0
    je next
    lea rcx, [rip + jumps]
    mov r12d, dword ptr [rcx + r12*4]
    inc r12
    jmp dispatch

finish:
    call flush
    xor edi, edi
    mov eax, 60
    syscall

fail:
    mov edi, 1
    mov eax, 60
    syscall

# write outbuf[0..rbp) to stdout, then rbp = 0
flush:
    xor r8d, r8d
fl1:
    cmp r8, rbp
    jae fl2
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r8
    mov rdx, rbp
    sub rdx, r8
    mov eax, 1
    syscall
    cmp rax, -4
    je fl1
    test rax, rax
    jle fl2
    add r8, rax
    jmp fl1
fl2:
    xor ebp, ebp
    ret

.section .rodata
.align 8
jtab:
    .quad h_right, h_left, h_plus, h_minus, h_out, h_in, h_open, h_close

.bss
.align 16
inbuf:
    .zero 65536
ops:
    .zero 65536
jumps:
    .zero 262144
bstk:
    .zero 262144
outbuf:
    .zero 65536
tape:
    .zero 30000
nops:
    .zero 8
progend:
    .zero 8
inpstart:
    .zero 8
inpend:
    .zero 8
