```
.intel_syntax noprefix
.section .text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_reading
    mov r9, rax
    add r14, rax
    lea rsi, [rip + buf]
    xor r8d, r8d

scan:
    cmp r8, r9
    jge read_loop
    movzx eax, byte ptr [rsi + r8]
    cmp al, 10
    jne 2f
    inc r12
2:
    cmp al, 32
    je ws
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    jbe ws
    test r15, r15
    jnz next
    mov r15, 1
    inc r13
    jmp next
ws:
    xor r15d, r15d
next:
    inc r8
    jmp scan

done_reading:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call store_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call store_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call store_num
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, rdi = destination; writes decimal digits, returns rdi past them
store_num:
    lea r9, [rip + tmp + 32]
    mov r8, 10
conv:
    xor edx, edx
    div r8
    dec r9
    add dl, 48
    mov byte ptr [r9], dl
    test rax, rax
    jnz conv
    lea rcx, [rip + tmp + 32]
    sub rcx, r9
copy:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    dec rcx
    jnz copy
    ret

.section .bss
.align 16
buf:
    .zero 65536
tmp:
    .zero 32
outbuf:
    .zero 128
```
