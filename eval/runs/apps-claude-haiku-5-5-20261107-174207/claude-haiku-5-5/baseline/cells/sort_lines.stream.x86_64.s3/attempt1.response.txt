.text
.globl _start
_start:
    # read all of standard input into inbuf
    xor r12d, r12d
readloop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle readdone
    add r12, rax
    jmp readloop
readdone:

    # split into lines: r8 = scan index, r9 = line start, r10 = line count
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    lea r11, [rip + inbuf]
scanloop:
    cmp r8, r12
    jae scanend
    cmp byte ptr [r11 + r8], 10
    jne scannext
    lea rax, [rip + lstart]
    mov dword ptr [rax + r10*4], r9d
    lea rax, [rip + lend]
    mov dword ptr [rax + r10*4], r8d
    inc r10
    lea r9, [r8 + 1]
scannext:
    inc r8
    jmp scanloop
scanend:
    cmp r9, r12
    jae splitdone
    lea rax, [rip + lstart]
    mov dword ptr [rax + r10*4], r9d
    lea rax, [rip + lend]
    mov dword ptr [rax + r10*4], r12d
    inc r10
splitdone:
    mov qword ptr [rip + nlines], r10

    # initialise source index array bufA with identity
    lea rbx, [rip + bufA]
    lea rbp, [rip + bufB]
    xor ecx, ecx
initloop:
    cmp rcx, qword ptr [rip + nlines]
    jae initdone
    mov dword ptr [rbx + rcx*4], ecx
    inc rcx
    jmp initloop
initdone:

    # bottom-up merge sort
    mov qword ptr [rip + width], 1
passloop:
    mov rax, qword ptr [rip + width]
    mov rcx, qword ptr [rip + nlines]
    cmp rax, rcx
    jae sortdone
    mov qword ptr [rip + lo_v], 0
mergeouter:
    mov rcx, qword ptr [rip + nlines]
    mov rdx, qword ptr [rip + lo_v]
    cmp rdx, rcx
    jae passend
    mov rax, qword ptr [rip + width]
    # mid = min(lo + width, n)
    lea r15, [rdx + rax]
    cmp r15, rcx
    jbe midok
    mov r15, rcx
midok:
    # hi = min(lo + 2*width, n)
    lea r9, [rdx + rax*2]
    cmp r9, rcx
    jbe hiok
    mov r9, rcx
hiok:
    mov qword ptr [rip + hi_v], r9
    mov r14, rdx
    mov r12, rdx
    mov r13, r15
    call mergerange
    mov rax, qword ptr [rip + width]
    mov rdx, qword ptr [rip + lo_v]
    lea rdx, [rdx + rax*2]
    mov qword ptr [rip + lo_v], rdx
    jmp mergeouter
passend:
    xchg rbx, rbp
    mov rax, qword ptr [rip + width]
    shl rax, 1
    mov qword ptr [rip + width], rax
    jmp passloop
sortdone:

    # output: r14 = outbuf, r15 = inbuf, r12 = k, r13 = output length
    lea r14, [rip + outbuf]
    lea r15, [rip + inbuf]
    xor r12d, r12d
    xor r13d, r13d
outloop:
    mov r11, qword ptr [rip + nlines]
    cmp r12, r11
    jae outdone
    mov eax, dword ptr [rbx + r12*4]
    lea rcx, [rip + lstart]
    mov r8d, dword ptr [rcx + rax*4]
    lea rcx, [rip + lend]
    mov r9d, dword ptr [rcx + rax*4]
outcopy:
    cmp r8, r9
    jae outcopied
    movzx ecx, byte ptr [r15 + r8]
    mov byte ptr [r14 + r13], cl
    inc r8
    inc r13
    jmp outcopy
outcopied:
    mov byte ptr [r14 + r13], 10
    inc r13
    inc r12
    jmp outloop
outdone:

    # write output completely
    xor r12d, r12d
writeloop:
    cmp r12, r13
    jae writedone
    mov eax, 1
    mov edi, 1
    lea rsi, [r14 + r12]
    mov rdx, r13
    sub rdx, r12
    syscall
    test rax, rax
    jle writedone
    add r12, rax
    jmp writeloop
writedone:
    mov eax, 60
    xor edi, edi
    syscall

# merge the sorted ranges src[lo..mid) and src[mid..hi) into dst[lo..hi)
# rbx = src, rbp = dst, r12 = i, r13 = j, r14 = k, r15 = mid, hi_v = hi
mergerange:
mloop:
    cmp r12, r15
    jae rightonly
    mov rax, qword ptr [rip + hi_v]
    cmp r13, rax
    jae leftonly
    mov edi, dword ptr [rbx + r12*4]
    mov esi, dword ptr [rbx + r13*4]
    call lessequal
    test eax, eax
    jnz takeleft
    mov eax, dword ptr [rbx + r13*4]
    mov dword ptr [rbp + r14*4], eax
    inc r13
    inc r14
    jmp mloop
takeleft:
    mov eax, dword ptr [rbx + r12*4]
    mov dword ptr [rbp + r14*4], eax
    inc r12
    inc r14
    jmp mloop
leftonly:
    cmp r12, r15
    jae mergedone
    mov eax, dword ptr [rbx + r12*4]
    mov dword ptr [rbp + r14*4], eax
    inc r12
    inc r14
    jmp leftonly
rightonly:
    cmp r13, qword ptr [rip + hi_v]
    jae mergedone
    mov eax, dword ptr [rbx + r13*4]
    mov dword ptr [rbp + r14*4], eax
    inc r13
    inc r14
    jmp rightonly
mergedone:
    ret

# input: rdi = id of line a, rsi = id of line b
# output: rax = 1 if line a <= line b (unsigned bytewise, prefix shorter first), else 0
# clobbers rax, rcx, rdx, rdi, rsi, r8-r11
lessequal:
    lea r10, [rip + lstart]
    mov r8d, dword ptr [r10 + rdi*4]
    mov r9d, dword ptr [r10 + rsi*4]
    lea r10, [rip + lend]
    mov ecx, dword ptr [r10 + rdi*4]
    sub ecx, r8d
    mov edx, dword ptr [r10 + rsi*4]
    sub edx, r9d
    mov r11d, ecx
    cmp ecx, edx
    jbe minok
    mov r11d, edx
minok:
    lea rax, [rip + inbuf]
    add r8, rax
    add r9, rax
    xor r10d, r10d
cmploop:
    cmp r10, r11
    jae prefixequal
    movzx eax, byte ptr [r8 + r10]
    movzx edi, byte ptr [r9 + r10]
    cmp eax, edi
    jne bytediffer
    inc r10
    jmp cmploop
bytediffer:
    setb al
    movzx eax, al
    ret
prefixequal:
    cmp ecx, edx
    setbe al
    movzx eax, al
    ret

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65536
lstart:
    .zero 4096
lend:
    .zero 4096
bufA:
    .zero 4096
bufB:
    .zero 4096
nlines:
    .zero 8
hi_v:
    .zero 8
lo_v:
    .zero 8
width:
    .zero 8
