.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d

main_read:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jle eof
    mov r14, rax
    xor r15d, r15d

proc_loop:
    cmp r15, r14
    jge main_read
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + r15]
    lea rdx, [rip + rowbuf]
    mov byte ptr [rdx + r13], al
    inc r13
    inc r15
    cmp r13, 16
    jne proc_loop
    call emit_row
    add r12, 16
    xor r13d, r13d
    jmp proc_loop

eof:
    test r13, r13
    jz eof_total
    call emit_row

eof_total:
    add r12, r13
    lea rdi, [rip + outbuf]
    mov ebx, r12d
    mov eax, ebx
    shr eax, 24
    call hexbyte
    mov eax, ebx
    shr eax, 16
    call hexbyte
    mov eax, ebx
    shr eax, 8
    call hexbyte
    mov eax, ebx
    call hexbyte
    mov byte ptr [rdi], 10
    lea rsi, [rip + outbuf]
    mov edx, 9
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

# emit_row: r12 = row offset, r13 = row length (1..16)
emit_row:
    lea rdi, [rip + outbuf]
    mov ecx, 80
    mov eax, 0x20
    rep stosb

    lea rdi, [rip + outbuf]
    mov ebx, r12d
    mov eax, ebx
    shr eax, 24
    call hexbyte
    mov eax, ebx
    shr eax, 16
    call hexbyte
    mov eax, ebx
    shr eax, 8
    call hexbyte
    mov eax, ebx
    call hexbyte

    lea rdi, [rip + outbuf + 10]
    lea rsi, [rip + rowbuf]
    xor r8d, r8d
hex_loop:
    cmp r8, r13
    jge hex_done
    movzx eax, byte ptr [rsi + r8]
    call hexbyte
    add rdi, 1
    cmp r8, 7
    jne hex_next
    add rdi, 1
hex_next:
    inc r8
    jmp hex_loop
hex_done:

    lea rdi, [rip + outbuf + 60]
    mov byte ptr [rdi], 124
    inc rdi
    lea rsi, [rip + rowbuf]
    xor r8d, r8d
asc_loop:
    cmp r8, r13
    jge asc_done
    movzx eax, byte ptr [rsi + r8]
    cmp eax, 0x20
    jb asc_dot
    cmp eax, 0x7e
    ja asc_dot
    jmp asc_store
asc_dot:
    mov eax, 46
asc_store:
    mov byte ptr [rdi + r8], al
    inc r8
    jmp asc_loop
asc_done:

    lea rax, [rip + outbuf + 61]
    add rax, r13
    mov byte ptr [rax], 124
    mov byte ptr [rax + 1], 10

    lea rsi, [rip + outbuf]
    lea rdx, [r13 + 63]
    call write_all
    ret

# hexbyte: al = value, rdi = destination; writes 2 hex digits, advances rdi by 2
hexbyte:
    lea rdx, [rip + hexd]
    movzx ecx, al
    shr ecx, 4
    movzx ecx, byte ptr [rdx + rcx]
    mov byte ptr [rdi], cl
    movzx ecx, al
    and ecx, 15
    movzx ecx, byte ptr [rdx + rcx]
    mov byte ptr [rdi + 1], cl
    add rdi, 2
    ret

# write_all: rsi = buffer, rdx = length
write_all:
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_fail
    add rsi, rax
    sub rdx, rax
    jnz write_loop
    ret
write_fail:
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 4096
rowbuf:
    .zero 16
outbuf:
    .zero 80
