.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor r14d, r14d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r14]
    mov edx, 65536
    sub edx, r14d
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jne fail
    lea rsi, [rip + msg_ok]
    mov edx, 3
    call write_all
    mov edi, 0
    mov eax, 60
    syscall

# ---------- whitespace: r13 = pos ----------
skip_ws:
ws_loop:
    cmp r13, r14
    jae ws_ret
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x20
    je ws_inc
    cmp al, 0x09
    je ws_inc
    cmp al, 0x0A
    je ws_inc
    cmp al, 0x0D
    jne ws_ret
ws_inc:
    inc r13
    jmp ws_loop
ws_ret:
    ret

# ---------- value ----------
parse_value:
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x7B
    je parse_object
    cmp al, 0x5B
    je parse_array
    cmp al, 0x22
    je parse_string
    cmp al, 0x2D
    je parse_number
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe parse_number
    cmp al, 0x74
    je parse_true
    cmp al, 0x66
    je parse_false
    cmp al, 0x6E
    je parse_null
    jmp fail

parse_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp parse_lit
parse_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp parse_lit
parse_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4
parse_lit:
lit_loop:
    test ecx, ecx
    jz lit_ret
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, byte ptr [rsi]
    jne fail
    inc r13
    inc rsi
    dec ecx
    jmp lit_loop
lit_ret:
    ret

# ---------- object ----------
parse_object:
    inc r13
    call skip_ws
    cmp r13, r14
    jae fail
    cmp byte ptr [r12 + r13], 0x7D
    jne obj_member
    inc r13
    ret
obj_member:
    cmp r13, r14
    jae fail
    cmp byte ptr [r12 + r13], 0x22
    jne fail
    call parse_string
    call skip_ws
    cmp r13, r14
    jae fail
    cmp byte ptr [r12 + r13], 0x3A
    jne fail
    inc r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2C
    je obj_comma
    cmp al, 0x7D
    jne fail
    inc r13
    ret
obj_comma:
    inc r13
    call skip_ws
    jmp obj_member

# ---------- array ----------
parse_array:
    inc r13
    call skip_ws
    cmp r13, r14
    jae fail
    cmp byte ptr [r12 + r13], 0x5D
    jne arr_value
    inc r13
    ret
arr_value:
    call parse_value
    call skip_ws
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2C
    je arr_comma
    cmp al, 0x5D
    jne fail
    inc r13
    ret
arr_comma:
    inc r13
    call skip_ws
    jmp arr_value

# ---------- string ----------
parse_string:
    inc r13
str_loop:
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x22
    je str_end
    cmp al, 0x5C
    je str_esc
    cmp al, 0x20
    jb fail
    inc r13
    jmp str_loop
str_end:
    inc r13
    ret
str_esc:
    inc r13
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x22
    je str_simple
    cmp al, 0x5C
    je str_simple
    cmp al, 0x2F
    je str_simple
    cmp al, 0x62
    je str_simple
    cmp al, 0x66
    je str_simple
    cmp al, 0x6E
    je str_simple
    cmp al, 0x72
    je str_simple
    cmp al, 0x74
    je str_simple
    cmp al, 0x75
    jne fail
    inc r13
    mov edx, 4
hex_loop:
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe hex_ok
    or al, 0x20
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
hex_ok:
    inc r13
    dec edx
    jnz hex_loop
    jmp str_loop
str_simple:
    inc r13
    jmp str_loop

# ---------- number ----------
parse_number:
    cmp byte ptr [r12 + r13], 0x2D
    jne num_int
    inc r13
num_int:
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    inc r13
    cmp al, 0x30
    je num_frac
num_idig:
    cmp r13, r14
    jae num_frac
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb num_frac
    cmp al, 0x39
    ja num_frac
    inc r13
    jmp num_idig
num_frac:
    cmp r13, r14
    jae num_done
    cmp byte ptr [r12 + r13], 0x2E
    jne num_exp
    inc r13
    call digits_req
num_exp:
    cmp r13, r14
    jae num_done
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x65
    je exp_start
    cmp al, 0x45
    jne num_done
exp_start:
    inc r13
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x2B
    je exp_sign
    cmp al, 0x2D
    jne exp_digits
exp_sign:
    inc r13
exp_digits:
    call digits_req
num_done:
    ret

# requires at least one digit at r13, consumes all digits
digits_req:
    cmp r13, r14
    jae fail
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
dr_loop:
    inc r13
    cmp r13, r14
    jae dr_ret
    movzx eax, byte ptr [r12 + r13]
    cmp al, 0x30
    jb dr_ret
    cmp al, 0x39
    ja dr_ret
    jmp dr_loop
dr_ret:
    ret

# ---------- write rsi/rdx fully to stdout ----------
write_all:
wa_loop:
    test rdx, rdx
    jz wa_ret
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wa_ret
    add rsi, rax
    sub rdx, rax
    jmp wa_loop
wa_ret:
    ret

# ---------- error path: r13 = offset ----------
fail:
    lea rsi, [rip + msg_err]
    mov edx, 9
    call write_all
    mov rax, r13
    lea rdi, [rip + numbuf + 31]
    mov byte ptr [rdi], 10
    mov r8d, 10
conv_loop:
    dec rdi
    xor edx, edx
    div r8
    add dl, 0x30
    mov byte ptr [rdi], dl
    test rax, rax
    jnz conv_loop
    lea rdx, [rip + numbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    call write_all
    mov edi, 1
    mov eax, 60
    syscall

.section .rodata
lit_true:  .ascii "true"
lit_false: .ascii "false"
lit_null:  .ascii "null"
msg_ok:    .ascii "ok\n"
msg_err:   .ascii "error at "

.section .bss
.align 16
inbuf:  .zero 65536
numbuf: .zero 32
