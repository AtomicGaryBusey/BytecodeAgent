.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea r8, [rip + buf]

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    mov r9, rax
    add r14, rax
    xor ebx, ebx

proc:
    cmp rbx, r9
    jae read_loop
    movzx eax, byte ptr [r8 + rbx]
    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je ws
    cmp al, 9
    jb nonws
    cmp al, 13
    jbe ws
nonws:
    test r15, r15
    jnz next
    inc r13
    mov r15d, 1
    jmp next
ws:
    xor r15d, r15d
next:
    inc rbx
    jmp proc

done_read:
    xor r10, r10
    mov rax, r12
    call emit_num
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r10], 32
    inc r10
    mov rax, r13
    call emit_num
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r10], 32
    inc r10
    mov rax, r14
    call emit_num
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r10], 10
    inc r10

    lea rsi, [rip + outbuf]
    mov rdx, r10
wloop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp wloop

finish:
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, appends decimal digits to outbuf at index r10
emit_num:
    mov rcx, 10
    xor esi, esi
dloop:
    xor edx, edx
    div rcx
    add dl, 48
    push rdx
    inc esi
    test rax, rax
    jnz dloop
ploop:
    pop rdx
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r10], dl
    inc r10
    dec esi
    jnz ploop
    ret

.bss
buf:
    .zero 65536
outbuf:
    .zero 64
