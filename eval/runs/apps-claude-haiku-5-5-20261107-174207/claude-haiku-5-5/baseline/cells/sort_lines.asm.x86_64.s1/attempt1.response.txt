.text
.globl _start
_start:
  xor r12d, r12d
rd_loop:
  lea rsi, [rip + inbuf]
  add rsi, r12
  mov rdx, 65536
  sub rdx, r12
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jle rd_done
  add r12, rax
  jmp rd_loop
rd_done:
  xor r13d, r13d
  xor r14d, r14d
  xor r15d, r15d
  lea rbx, [rip + inbuf]
pz_loop:
  cmp r15, r12
  jae pz_end
  cmp byte ptr [rbx + r15], 10
  jne pz_next
  lea rax, [rip + starts]
  mov qword ptr [rax + r13*8], r14
  lea rax, [rip + ends]
  mov qword ptr [rax + r13*8], r15
  inc r13
  lea r14, [r15 + 1]
pz_next:
  inc r15
  jmp pz_loop
pz_end:
  cmp r14, r12
  jae pz_fin
  lea rax, [rip + starts]
  mov qword ptr [rax + r13*8], r14
  lea rax, [rip + ends]
  mov qword ptr [rax + r13*8], r12
  inc r13
pz_fin:
  xor ecx, ecx
ix_loop:
  cmp rcx, r13
  jae ix_done
  lea rax, [rip + idxA]
  mov qword ptr [rax + rcx*8], rcx
  inc rcx
  jmp ix_loop
ix_done:
  lea r8, [rip + idxA]
  lea r9, [rip + idxB]
  mov r10d, 1
ms_outer:
  cmp r10, r13
  jae ms_done
  xor r12d, r12d
ms_block:
  cmp r12, r13
  jae ms_pass_end
  lea rbp, [r12 + r10]
  cmp rbp, r13
  jbe ms_mid_ok
  mov rbp, r13
ms_mid_ok:
  lea rax, [r12 + r10*2]
  cmp rax, r13
  jbe ms_hi_ok
  mov rax, r13
ms_hi_ok:
  mov qword ptr [rip + vhi], rax
  mov r14, r12
  mov r15, rbp
  mov rbx, r12
m_main:
  cmp r14, rbp
  jae m_leftover
  cmp r15, qword ptr [rip + vhi]
  jae m_leftover
  mov rax, qword ptr [r8 + r14*8]
  mov rcx, qword ptr [r8 + r15*8]
  call cmp_gt
  test al, al
  jnz m_take_right
  mov rax, qword ptr [r8 + r14*8]
  mov qword ptr [r9 + rbx*8], rax
  inc r14
  jmp m_next
m_take_right:
  mov rax, qword ptr [r8 + r15*8]
  mov qword ptr [r9 + rbx*8], rax
  inc r15
m_next:
  inc rbx
  jmp m_main
m_leftover:
  cmp r14, rbp
  jae m_right
  mov rax, qword ptr [r8 + r14*8]
  mov qword ptr [r9 + rbx*8], rax
  inc r14
  inc rbx
  jmp m_leftover
m_right:
  cmp r15, qword ptr [rip + vhi]
  jae m_block_done
  mov rax, qword ptr [r8 + r15*8]
  mov qword ptr [r9 + rbx*8], rax
  inc r15
  inc rbx
  jmp m_right
m_block_done:
  lea r12, [r12 + r10*2]
  jmp ms_block
ms_pass_end:
  xchg r8, r9
  shl r10, 1
  jmp ms_outer
ms_done:
  xor ebx, ebx
  xor r15d, r15d
ot_loop:
  cmp r15, r13
  jae ot_done
  mov rcx, qword ptr [r8 + r15*8]
  lea rsi, [rip + starts]
  mov rdx, qword ptr [rsi + rcx*8]
  lea rsi, [rip + ends]
  mov rdi, qword ptr [rsi + rcx*8]
ot_copy:
  cmp rdx, rdi
  jae ot_nl
  movzx eax, byte ptr [rdx + inbuf]
  mov byte ptr [rbx + outbuf], al
  inc rdx
  inc rbx
  jmp ot_copy
ot_nl:
  mov byte ptr [rbx + outbuf], 10
  inc rbx
  inc r15
  jmp ot_loop
ot_done:
  xor r12d, r12d
wr_loop:
  cmp r12, rbx
  jae fin
  lea rsi, [rip + outbuf]
  add rsi, r12
  mov rdx, rbx
  sub rdx, r12
  mov edi, 1
  mov eax, 1
  syscall
  test rax, rax
  jle fin
  add r12, rax
  jmp wr_loop
fin:
  xor edi, edi
  mov eax, 60
  syscall

# Compare lines: rax = line a, rcx = line b. Returns al = 1 if a > b, else 0.
cmp_gt:
  lea rsi, [rip + starts]
  mov rdi, qword ptr [rsi + rax*8]
  mov rdx, qword ptr [rsi + rcx*8]
  lea rsi, [rip + ends]
  mov r11, qword ptr [rsi + rax*8]
  mov rax, qword ptr [rsi + rcx*8]
  lea rsi, [rip + inbuf]
  add rdi, rsi
  add rdx, rsi
  add r11, rsi
  add rax, rsi
cmp_loop:
  cmp rdi, r11
  jae cmp_le
  cmp rdx, rax
  jae cmp_gt_ret
  movzx ecx, byte ptr [rdi]
  movzx esi, byte ptr [rdx]
  cmp ecx, esi
  jne cmp_diff
  inc rdi
  inc rdx
  jmp cmp_loop
cmp_diff:
  seta al
  movzx eax, al
  ret
cmp_le:
  xor eax, eax
  ret
cmp_gt_ret:
  mov eax, 1
  ret

  .zero 65536
starts:

  .zero 8192

  .zero 8192
