.globl _start

_start:
    mov     x19, #0            // lines
    mov     x20, #0            // words
    mov     x21, #0            // bytes
    mov     x22, #0            // in-word flag
    adrp    x26, buf
    add     x26, x26, :lo12:buf

read_loop:
    mov     x8, #63
    mov     x0, #0
    mov     x1, x26
    mov     x2, #4096
    svc     #0
    cmp     x0, #0
    b.le    done_read
    mov     x23, x0            // count read
    add     x21, x21, x0
    mov     x24, #0            // index

process_loop:
    cmp     x24, x23
    b.ge    read_loop
    ldrb    w0, [x26, x24]
    add     x24, x24, #1

    cmp     w0, #0x0A
    b.ne    1f
    add     x19, x19, #1
1:
    cmp     w0, #0x20
    b.eq    is_ws
    sub     w1, w0, #9
    cmp     w1, #4
    b.ls    is_ws

    // non-whitespace byte
    cbnz    x22, process_loop
    mov     x22, #1
    add     x20, x20, #1
    b       process_loop

is_ws:
    mov     x22, #0
    b       process_loop

done_read:
    adrp    x9, outbuf
    add     x9, x9, :lo12:outbuf

    mov     x0, x19
    mov     x1, x9
    bl      fmt
    mov     x9, x0
    mov     w3, #32
    strb    w3, [x9], #1

    mov     x0, x20
    mov     x1, x9
    bl      fmt
    mov     x9, x0
    mov     w3, #32
    strb    w3, [x9], #1

    mov     x0, x21
    mov     x1, x9
    bl      fmt
    mov     x9, x0
    mov     w3, #10
    strb    w3, [x9], #1

    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    sub     x2, x9, x1

wloop:
    cbz     x2, wdone
    mov     x0, #1
    mov     x8, #64
    svc     #0
    cmp     x0, #0
    b.le    exit_err
    add     x1, x1, x0
    sub     x2, x2, x0
    b       wloop

wdone:
    mov     x0, #0
    mov     x8, #93
    svc     #0

exit_err:
    mov     x0, #1
    mov     x8, #93
    svc     #0

// fmt: x0 = value, x1 = destination; writes decimal digits, returns x0 = next destination
fmt:
    sub     sp, sp, #32
    add     x4, sp, #32
    mov     x10, #10
1:
    udiv    x2, x0, x10
    msub    x3, x2, x10, x0
    add     w3, w3, #48
    sub     x4, x4, #1
    strb    w3, [x4]
    mov     x0, x2
    cbnz    x0, 1b
    add     x5, sp, #32
2:
    cmp     x4, x5
    b.hs    3f
    ldrb    w3, [x4], #1
    strb    w3, [x1], #1
    b       2b
3:
    mov     x0, x1
    add     sp, sp, #32
    ret

.bss
.balign 16
