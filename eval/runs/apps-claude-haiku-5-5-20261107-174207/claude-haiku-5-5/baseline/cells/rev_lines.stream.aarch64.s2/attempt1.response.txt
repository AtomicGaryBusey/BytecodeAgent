.text
.globl _start
_start:
    adrp x19, buf
    add  x19, x19, :lo12:buf
    mov  x21, #0                // len read so far
read_loop:
    mov  x0, #0
    add  x1, x19, x21
    mov  x2, #65536
    sub  x2, x2, x21
    mov  x8, #63
    svc  #0
    cbz  x0, read_done
    add  x21, x21, x0
    b    read_loop
read_done:
    mov  x22, x21               // n
    adrp x20, outbuf
    add  x20, x20, :lo12:outbuf
    mov  x23, #0                // i: line start
    mov  x24, #0                // o: output length
line_loop:
    cmp  x23, x22
    b.hs write_out
    mov  x25, x23               // e: scan for newline
find_nl:
    cmp  x25, x22
    b.hs found
    ldrb w0, [x19, x25]
    cmp  w0, #10
    b.eq found
    add  x25, x25, #1
    b    find_nl
found:
    sub  x26, x25, x23          // line length L
    mov  x27, #0                // k
rev_loop:
    cmp  x27, x26
    b.hs rev_done
    sub  x28, x25, #1
    sub  x28, x28, x27          // index e-1-k
    ldrb w0, [x19, x28]
    add  x3, x24, x27
    strb w0, [x20, x3]
    add  x27, x27, #1
    b    rev_loop
rev_done:
    add  x3, x24, x26
    mov  w0, #10
    strb w0, [x20, x3]
    add  x24, x24, x26
    add  x24, x24, #1
    add  x23, x25, #1           // i = e + 1
    b    line_loop
write_out:
    mov  x25, #0                // bytes written
wloop:
    cmp  x25, x24
    b.hs done
    mov  x0, #1
    add  x1, x20, x25
    sub  x2, x24, x25
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le done
    add  x25, x25, x0
    b    wloop
done:
    mov  x0, #0
    mov  x8, #93
    svc  #0

.bss
.balign 16
buf:
    .zero 65536
.balign 16
outbuf:
    .zero 65552
