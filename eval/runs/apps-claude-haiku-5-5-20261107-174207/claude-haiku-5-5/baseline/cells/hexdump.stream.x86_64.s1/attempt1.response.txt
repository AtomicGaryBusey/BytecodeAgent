```
.intel_syntax noprefix
.text
.globl _start
_start:
    lea r13, [rip + inbuf]
    xor r12d, r12d
read_loop:
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r14d, r14d
row_loop:
    cmp r14, r12
    jae rows_done
    lea r15, [r13 + r14]
    mov rbp, r12
    sub rbp, r14
    cmp rbp, 16
    jbe 1f
    mov ebp, 16
1:
    lea rdi, [rip + linebuf]
    mov rdx, r14
    call put_hex8
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    add rdi, 2
    lea rsi, [rip + hexdig]
    xor ebx, ebx
hex_loop:
    cmp rbx, rbp
    jae hex_pad
    movzx eax, byte ptr [r15 + rbx]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [rsi + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    movzx eax, byte ptr [rsi + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp hex_next
hex_pad:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
hex_next:
    cmp rbx, 7
    jne 2f
    mov byte ptr [rdi], 32
    inc rdi
2:
    inc rbx
    cmp rbx, 16
    jb hex_loop
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2
    xor ebx, ebx
chr_loop:
    cmp rbx, rbp
    jae chr_done
    movzx eax, byte ptr [r15 + rbx]
    cmp eax, 32
    jb chr_dot
    cmp eax, 126
    ja chr_dot
    jmp chr_store
chr_dot:
    mov eax, 46
chr_store:
    mov byte ptr [rdi], al
    inc rdi
    inc rbx
    jmp chr_loop
chr_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    call write_all
    add r14, 16
    jmp row_loop
rows_done:
    lea rdi, [rip + linebuf]
    mov rdx, r12
    call put_hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + linebuf]
    mov rdx, rdi
    sub rdx, rsi
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

# put_hex8: edx = value, rdi = destination; writes 8 hex digits, advances rdi
put_hex8:
    lea r8, [rip + hexdig]
    mov ecx, 28
ph_loop:
    mov eax, edx
    shr eax, cl
    and eax, 15
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    sub ecx, 4
    jge ph_loop
    ret

# write_all: rsi = buffer, rdx = length (> 0)
write_all:
wr_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wr_fail
    add rsi, rax
    sub rdx, rax
    jnz wr_loop
    ret
wr_fail:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hexdig:
    .ascii "0123456789abcdef"

.bss
.align 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
```
