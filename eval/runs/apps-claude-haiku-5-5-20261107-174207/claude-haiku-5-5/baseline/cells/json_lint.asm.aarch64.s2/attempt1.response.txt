```
.equ S_VAL, 0
.equ S_OBJ_FIRST, 1
.equ S_OBJ_KEY, 2
.equ S_KEY_COLON, 3
.equ S_ARR_FIRST, 4
.equ S_AFTER_VALUE, 5
.equ S_STR, 6
.equ S_ESC, 7
.equ S_HEX, 8
.equ S_LIT, 9
.equ S_NEG, 10
.equ S_ZERO, 11
.equ S_INT, 12
.equ S_FRAC0, 13
.equ S_FRAC, 14
.equ S_EXP0, 15
.equ S_EXPD, 16
.equ S_EXP, 17
.equ S_DONE, 18

.section .rodata
okmsg:
    .ascii "ok\n"
errmsg:
    .ascii "error at "
nlmsg:
    .ascii "\n"
lit_rue:
    .asciz "rue"
lit_alse:
    .asciz "alse"
lit_ull:
    .asciz "ull"

.bss
    .align 4
inbuf:
    .zero 65536
stackarr:
    .zero 1024
numbuf:
    .zero 32

.text
.globl _start
_start:
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf
    mov x20, #0
read_loop:
    movz x10, #1, lsl #16
    cmp x20, x10
    b.hs read_done
    mov x8, #63
    mov x0, #0
    add x1, x23, x20
    movz x2, #1, lsl #16
    sub x2, x2, x20
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x19, #0
    mov x21, #S_VAL
    mov x22, #0
    adrp x27, stackarr
    add x27, x27, :lo12:stackarr

main_loop:
    cmp x19, x20
    b.hs at_eof
    ldrb w9, [x23, x19]
    cmp x21, #S_VAL
    b.eq st_val
    cmp x21, #S_AFTER_VALUE
    b.eq st_after_value
    cmp x21, #S_STR
    b.eq st_str
    cmp x21, #S_OBJ_FIRST
    b.eq st_obj_first
    cmp x21, #S_OBJ_KEY
    b.eq st_obj_key
    cmp x21, #S_KEY_COLON
    b.eq st_key_colon
    cmp x21, #S_ARR_FIRST
    b.eq st_arr_first
    cmp x21, #S_ESC
    b.eq st_esc
    cmp x21, #S_HEX
    b.eq st_hex
    cmp x21, #S_LIT
    b.eq st_lit
    cmp x21, #S_NEG
    b.eq st_neg
    cmp x21, #S_ZERO
    b.eq st_zero
    cmp x21, #S_INT
    b.eq st_int
    cmp x21, #S_FRAC0
    b.eq st_frac0
    cmp x21, #S_FRAC
    b.eq st_frac
    cmp x21, #S_EXP0
    b.eq st_exp0
    cmp x21, #S_EXPD
    b.eq st_expd
    cmp x21, #S_EXP
    b.eq st_exp
    b st_done

// ---------- value start ----------
st_val:
    bl is_ws
    cbnz w10, val_ws
    cmp w9, #0x7B
    b.eq val_open_obj
    cmp w9, #0x5B
    b.eq val_open_arr
    cmp w9, #0x22
    b.eq val_str
    cmp w9, #0x2D
    b.eq val_neg
    cmp w9, #0x30
    b.eq val_zero
    sub w10, w9, #0x31
    cmp w10, #8
    b.ls val_int
    cmp w9, #0x74
    b.eq val_t
    cmp w9, #0x66
    b.eq val_f
    cmp w9, #0x6E
    b.eq val_n
    b report_error
val_ws:
    add x19, x19, #1
    b main_loop
val_open_obj:
    strb w9, [x27, x22]
    add x22, x22, #1
    add x19, x19, #1
    mov x21, #S_OBJ_FIRST
    b main_loop
val_open_arr:
    strb w9, [x27, x22]
    add x22, x22, #1
    add x19, x19, #1
    mov x21, #S_ARR_FIRST
    b main_loop
val_str:
    mov x25, #0
    add x19, x19, #1
    mov x21, #S_STR
    b main_loop
val_neg:
    add x19, x19, #1
    mov x21, #S_NEG
    b main_loop
val_zero:
    add x19, x19, #1
    mov x21, #S_ZERO
    b main_loop
val_int:
    add x19, x19, #1
    mov x21, #S_INT
    b main_loop
val_t:
    adrp x26, lit_rue
    add x26, x26, :lo12:lit_rue
    add x19, x19, #1
    mov x21, #S_LIT
    b main_loop
val_f:
    adrp x26, lit_alse
    add x26, x26, :lo12:lit_alse
    add x19, x19, #1
    mov x21, #S_LIT
    b main_loop
val_n:
    adrp x26, lit_ull
    add x26, x26, :lo12:lit_ull
    add x19, x19, #1
    mov x21, #S_LIT
    b main_loop

// ---------- array / object start ----------
st_arr_first:
    bl is_ws
    cbnz w10, arr_first_ws
    cmp w9, #0x5D
    b.ne arr_first_val
    sub x22, x22, #1
    add x19, x19, #1
    mov x21, #S_AFTER_VALUE
    b main_loop
arr_first_ws:
    add x19, x19, #1
    b main_loop
arr_first_val:
    mov x21, #S_VAL
    b main_loop

st_obj_first:
    bl is_ws
    cbnz w10, obj_first_ws
    cmp w9, #0x7D
    b.ne obj_first_key
    sub x22, x22, #1
    add x19, x19, #1
    mov x21, #S_AFTER_VALUE
    b main_loop
obj_first_ws:
    add x19, x19, #1
    b main_loop
obj_first_key:
    mov x21, #S_OBJ_KEY
    b main_loop

st_obj_key:
    bl is_ws
    cbnz w10, okey_ws
    cmp w9, #0x22
    b.ne report_error
    add x19, x19, #1
    mov x25, #1
    mov x21, #S_STR
    b main_loop
okey_ws:
    add x19, x19, #1
    b main_loop

st_key_colon:
    bl is_ws
    cbnz w10, kc_ws
    cmp w9, #0x3A
    b.ne report_error
    add x19, x19, #1
    mov x21, #S_VAL
    b main_loop
kc_ws:
    add x19, x19, #1
    b main_loop

// ---------- after a complete value ----------
st_after_value:
    cbnz x22, av_nonzero
    mov x21, #S_DONE
    b main_loop
av_nonzero:
    sub x10, x22, #1
    ldrb w11, [x27, x10]
    cmp w11, #0x5B
    b.eq av_arr
    bl is_ws
    cbnz w10, av_ws
    cmp w9, #0x2C
    b.eq av_comma_obj
    cmp w9, #0x7D
    b.eq av_close
    b report_error
av_arr:
    bl is_ws
    cbnz w10, av_ws
    cmp w9, #0x2C
    b.eq av_comma_arr
    cmp w9, #0x5D
    b.eq av_close
    b report_error
av_ws:
    add x19, x19, #1
    b main_loop
av_comma_obj:
    add x19, x19, #1
    mov x21, #S_OBJ_KEY
    b main_loop
av_comma_arr:
    add x19, x19, #1
    mov x21, #S_VAL
    b main_loop
av_close:
    sub x22, x22, #1
    add x19, x19, #1
    mov x21, #S_AFTER_VALUE
    b main_loop

// ---------- strings ----------
st_str:
    cmp w9, #0x22
    b.eq str_end
    cmp w9, #0x5C
    b.eq str_bs
    cmp w9, #0x20
    b.lo report_error
    add x19, x19, #1
    b main_loop
str_end:
    add x19, x19, #1
    cbnz x25, str_end_key
    mov x21, #S_AFTER_VALUE
    b main_loop
str_end_key:
    mov x21, #S_KEY_COLON
    b main_loop
str_bs:
    add x19, x19, #1
    mov x21, #S_ESC
    b main_loop

st_esc:
    cmp w9, #0x75
    b.eq esc_u
    cmp w9, #0x22
    b.eq esc_ok
    cmp w9, #0x5C
    b.eq esc_ok
    cmp w9, #0x2F
    b.eq esc_ok
    cmp w9, #0x62
    b.eq esc_ok
    cmp w9, #0x66
    b.eq esc_ok
    cmp w9, #0x6E
    b.eq esc_ok
    cmp w9, #0x72
    b.eq esc_ok
    cmp w9, #0x74
    b.eq esc_ok
    b report_error
esc_u:
    add x19, x19, #1
    mov x24, #4
    mov x21, #S_HEX
    b main_loop
esc_ok:
    add x19, x19, #1
    mov x21, #S_STR
    b main_loop

st_hex:
    sub w10, w9, #0x30
    cmp w10, #9
    b.ls hex_ok
    orr w10, w9, #0x20
    sub w10, w10, #0x61
    cmp w10, #5
    b.hi report_error
hex_ok:
    add x19, x19, #1
    sub x24, x24, #1
    cbnz x24, main_loop
    mov x21, #S_STR
    b main_loop

// ---------- literals ----------
st_lit:
    ldrb w10, [x26]
    cmp w9, w10
    b.ne report_error
    add x19, x19, #1
    add x26, x26, #1
    ldrb w10, [x26]
    cbnz w10, main_loop
    mov x21, #S_AFTER_VALUE
    b main_loop

// ---------- numbers ----------
st_neg:
    cmp w9, #0x30
    b.eq neg_zero
    sub w10, w9, #0x31
    cmp w10, #8
    b.ls neg_int
    b report_error
neg_zero:
    add x19, x19, #1
    mov x21, #S_ZERO
    b main_loop
neg_int:
    add x19, x19, #1
    mov x21, #S_INT
    b main_loop

st_zero:
    b int_nd

st_int:
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi int_nd
    add x19, x19, #1
    b main_loop
int_nd:
    cmp w9, #0x2E
    b.eq num_dot
    cmp w9, #0x65
    b.eq num_e
    cmp w9, #0x45
    b.eq num_e
    mov x21, #S_AFTER_VALUE
    b main_loop
num_dot:
    add x19, x19, #1
    mov x21, #S_FRAC0
    b main_loop
num_e:
    add x19, x19, #1
    mov x21, #S_EXP0
    b main_loop

st_frac0:
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi report_error
    add x19, x19, #1
    mov x21, #S_FRAC
    b main_loop

st_frac:
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi frac_nd
    add x19, x19, #1
    b main_loop
frac_nd:
    cmp w9, #0x65
    b.eq num_e
    cmp w9, #0x45
    b.eq num_e
    mov x21, #S_AFTER_VALUE
    b main_loop

st_exp0:
    cmp w9, #0x2B
    b.eq exp0_sign
    cmp w9, #0x2D
    b.eq exp0_sign
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi report_error
    add x19, x19, #1
    mov x21, #S_EXP
    b main_loop
exp0_sign:
    add x19, x19, #1
    mov x21, #S_EXPD
    b main_loop

st_expd:
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi report_error
    add x19, x19, #1
    mov x21, #S_EXP
    b main_loop

st_exp:
    sub w10, w9, #0x30
    cmp w10, #9
    b.hi exp_nd
    add x19, x19, #1
    b main_loop
exp_nd:
    mov x21, #S_AFTER_VALUE
    b main_loop

// ---------- trailing whitespace ----------
st_done:
    bl is_ws
    cbnz w10, done_ws
    b report_error
done_ws:
    add x19, x19, #1
    b main_loop

// ---------- end of input ----------
at_eof:
    cmp x21, #S_INT
    b.eq eof_fin
    cmp x21, #S_ZERO
    b.eq eof_fin
    cmp x21, #S_FRAC
    b.eq eof_fin
    cmp x21, #S_EXP
    b.eq eof_fin
    b eof_check
eof_fin:
    mov x21, #S_AFTER_VALUE
eof_check:
    cmp x21, #S_DONE
    b.eq success
    cmp x21, #S_AFTER_VALUE
    b.ne report_error
    cbnz x22, report_error
    b success

// ---------- helper: w9 is whitespace? result in w10 ----------
is_ws:
    cmp w9, #0x20
    cset w10, eq
    cmp w9, #0x09
    cset w11, eq
    orr w10, w10, w11
    cmp w9, #0x0A
    cset w11, eq
    orr w10, w10, w11
    cmp w9, #0x0D
    cset w11, eq
    orr w10, w10, w11
    ret

// ---------- output ----------
success:
    mov x8, #64
    mov x0, #1
    adrp x1, okmsg
    add x1, x1, :lo12:okmsg
    mov x2, #3
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0

report_error:
    adrp x1, numbuf
    add x1, x1, :lo12:numbuf
    add x1, x1, #24
    mov x5, x1
    mov x0, x19
    mov x2, #10
conv_loop:
    udiv x3, x0, x2
    msub x4, x3, x2, x0
    add w4, w4, #0x30
    strb w4, [x1, #-1]!
    mov x0, x3
    cbnz x0, conv_loop
    mov x12, x1
    sub x13, x5, x1
    mov x8, #64
    mov x0, #1
    adrp x1, errmsg
    add x1, x1, :lo12:errmsg
    mov x2, #9
    svc #0
    mov x8, #64
    mov x0, #1
    mov x1, x12
    mov x2, x13
    svc #0
    mov x8, #64
    mov x0, #1
    adrp x1, nlmsg
    add x1, x1, :lo12:nlmsg
    mov x2, #1
    svc #0
    mov x8, #93
    mov x0, #1
    svc #0
```
