```
.intel_syntax noprefix

.section .bss
.align 16
inbuf:
    .zero 131072
.align 16
matchtab:
    .zero 262144
.align 16
stacktab:
    .zero 262144
.align 16
tape:
    .zero 30000

.section .text
.globl _start
_start:
    # Read all of standard input into inbuf
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov rdx, 131072
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:

    # Find first '!': program = [0, i), input = (i, total)
    lea r10, [r12 + r13]
    xor ecx, ecx
scan_bang:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp scan_bang
found_bang:
    mov r13, rcx
    lea r9, [r12 + rcx + 1]
    jmp have_input
no_bang:
    mov r9, r10
have_input:

    # Bracket check and matching
    lea r15, [rip + matchtab]
    lea r14, [rip + stacktab]
    xor ebx, ebx
    xor r8d, r8d
bloop:
    cmp rbx, r13
    jae bdone
    movzx eax, byte ptr [r12 + rbx]
    cmp eax, 0x5b
    je bpush
    cmp eax, 0x5d
    je bpop
    jmp bnext
bpush:
    mov dword ptr [r14 + r8*4], ebx
    inc r8
    jmp bnext
bpop:
    test r8, r8
    jz bfail
    dec r8
    mov eax, dword ptr [r14 + r8*4]
    mov dword ptr [r15 + rax*4], ebx
    mov dword ptr [r15 + rbx*4], eax
bnext:
    inc rbx
    jmp bloop
bdone:
    test r8, r8
    jnz bfail

    # Interpreter
    lea r14, [rip + tape]
    xor ebx, ebx
run:
    cmp rbx, r13
    jae done
    movzx eax, byte ptr [r12 + rbx]
    cmp eax, 0x3e
    je op_right
    cmp eax, 0x3c
    je op_left
    cmp eax, 0x2b
    je op_inc
    cmp eax, 0x2d
    je op_dec
    cmp eax, 0x2e
    je op_out
    cmp eax, 0x2c
    je op_in
    cmp eax, 0x5b
    je op_open
    cmp eax, 0x5d
    je op_close
    jmp next
op_right:
    inc r14
    jmp next
op_left:
    dec r14
    jmp next
op_inc:
    inc byte ptr [r14]
    jmp next
op_dec:
    dec byte ptr [r14]
    jmp next
op_out:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, 1
    syscall
    jmp next
op_in:
    cmp r9, r10
    jae in_eof
    mov al, byte ptr [r9]
    mov byte ptr [r14], al
    inc r9
    jmp next
in_eof:
    mov byte ptr [r14], 0
    jmp next
op_open:
    cmp byte ptr [r14], 0
    jne next
    mov ebx, dword ptr [r15 + rbx*4]
    jmp next
op_close:
    cmp byte ptr [r14], 0
    je next
    mov ebx, dword ptr [r15 + rbx*4]
    jmp next
next:
    inc rbx
    jmp run
done:
    mov eax, 60
    xor edi, edi
    syscall
bfail:
    mov eax, 60
    mov edi, 1
    syscall
```
