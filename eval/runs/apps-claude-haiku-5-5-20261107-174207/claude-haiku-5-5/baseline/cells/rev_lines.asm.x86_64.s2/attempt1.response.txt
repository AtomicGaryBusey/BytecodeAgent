```
.intel_syntax noprefix
.equ INSIZE, 131072
.equ OUTSIZE, 131072

.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, INSIZE
    sub rdx, r12
    jz done_read
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle done_read
    add r12, rax
    jmp read_loop
done_read:
    xor r13d, r13d
    xor r14d, r14d
    lea r15, [rip + inbuf]
    lea rbp, [rip + outbuf]
line_loop:
    cmp r13, r12
    jae write_out
    mov rcx, r13
scan:
    cmp rcx, r12
    jae scan_end
    cmp byte ptr [r15 + rcx], 10
    je scan_end
    inc rcx
    jmp scan
scan_end:
    mov rdx, rcx
rev:
    cmp rdx, r13
    jbe rev_end
    dec rdx
    movzx eax, byte ptr [r15 + rdx]
    mov byte ptr [rbp + r14], al
    inc r14
    jmp rev
rev_end:
    mov byte ptr [rbp + r14], 10
    inc r14
    lea r13, [rcx + 1]
    jmp line_loop
write_out:
    lea rsi, [rip + outbuf]
    mov rdx, r14
wloop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp wloop
finish:
    xor edi, edi
    mov eax, 60
    syscall

.bss
.align 16
inbuf:
    .zero INSIZE
outbuf:
    .zero OUTSIZE
```
