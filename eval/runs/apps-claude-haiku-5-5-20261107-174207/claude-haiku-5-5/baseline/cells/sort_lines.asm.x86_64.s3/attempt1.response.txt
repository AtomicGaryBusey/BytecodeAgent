.globl _start

_start:
    # read all of standard input into inbuf; r14 = total bytes
    xor r14d, r14d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r14
    mov rdx, 65536
    sub rdx, r14
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:

    # split into lines: r15 = line count, rbx = scan index, r12 = line start
    xor r15d, r15d
    xor ebx, ebx
    xor r12d, r12d
scan:
    cmp rbx, r14
    jae scan_end
    cmp byte ptr [inbuf + rbx], 10
    jne scan_next
    mov rax, r15
    mov qword ptr [lstart + rax*8], r12
    mov rcx, rbx
    sub rcx, r12
    mov qword ptr [llen + rax*8], rcx
    mov qword ptr [idxa + rax*8], rax
    inc r15
    lea r12, [rbx + 1]
scan_next:
    inc rbx
    jmp scan
scan_end:
    cmp r12, r14
    jae no_tail
    mov rax, r15
    mov qword ptr [lstart + rax*8], r12
    mov rcx, r14
    sub rcx, r12
    mov qword ptr [llen + rax*8], rcx
    mov qword ptr [idxa + rax*8], rax
    inc r15
no_tail:

    # sort index array by bottom-up merge sort
    lea rax, [rip + idxa]
    mov qword ptr [rip + srcp], rax
    lea rax, [rip + idxb]
    mov qword ptr [rip + dstp], rax
    mov r14, r15
    mov r12, 1
sort_w_loop:
    cmp r12, r14
    jae sort_done
    xor r13d, r13d
lo_loop:
    cmp r13, r14
    jae lo_done
    lea rsi, [r13 + r12]
    cmp rsi, r14
    jbe mid_ok
    mov rsi, r14
mid_ok:
    lea rdx, [r13 + r12*2]
    cmp rdx, r14
    jbe hi_ok
    mov rdx, r14
hi_ok:
    mov rdi, r13
    call merge
    lea r13, [r13 + r12*2]
    jmp lo_loop
lo_done:
    mov rax, qword ptr [rip + srcp]
    mov rcx, qword ptr [rip + dstp]
    mov qword ptr [rip + srcp], rcx
    mov qword ptr [rip + dstp], rax
    shl r12, 1
    jmp sort_w_loop
sort_done:

    # build output in outbuf; r13 = output length, r12 = k, r15 = count
    xor r13d, r13d
    xor r12d, r12d
out_loop:
    cmp r12, r15
    jae out_done
    mov rax, qword ptr [rip + srcp]
    mov rbx, qword ptr [rax + r12*8]
    mov rsi, qword ptr [lstart + rbx*8]
    mov rcx, qword ptr [llen + rbx*8]
    xor edx, edx
cp_loop:
    cmp rdx, rcx
    jae cp_done
    movzx eax, byte ptr [inbuf + rsi + rdx]
    mov byte ptr [outbuf + r13 + rdx], al
    inc rdx
    jmp cp_loop
cp_done:
    mov byte ptr [outbuf + r13 + rcx], 10
    add r13, rcx
    inc r13
    inc r12
    jmp out_loop
out_done:

    # write output buffer completely
    xor r12d, r12d
w_loop:
    cmp r12, r13
    jae w_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r12
    mov rdx, r13
    sub rdx, r12
    syscall
    test rax, rax
    jle w_done
    add r12, rax
    jmp w_loop
w_done:
    mov eax, 60
    xor edi, edi
    syscall

# merge(rdi = lo, rsi = mid, rdx = hi): merges src[lo..mid) and src[mid..hi) into dst
merge:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdi
    mov r14, rsi
    mov r15, rdx
m_loop:
    cmp rbx, r14
    jae m_tail_j
    cmp r12, r15
    jae m_tail_i
    mov rax, qword ptr [rip + srcp]
    mov rdi, qword ptr [rax + r12*8]
    mov rsi, qword ptr [rax + rbx*8]
    call cmp_lt
    test eax, eax
    jz take_a
    mov rax, qword ptr [rip + srcp]
    mov rcx, qword ptr [rax + r12*8]
    mov rdx, qword ptr [rip + dstp]
    mov qword ptr [rdx + r13*8], rcx
    inc r12
    jmp m_next
take_a:
    mov rax, qword ptr [rip + srcp]
    mov rcx, qword ptr [rax + rbx*8]
    mov rdx, qword ptr [rip + dstp]
    mov qword ptr [rdx + r13*8], rcx
    inc rbx
m_next:
    inc r13
    jmp m_loop
m_tail_i:
    cmp rbx, r14
    jae m_done
    mov rax, qword ptr [rip + srcp]
    mov rcx, qword ptr [rax + rbx*8]
    mov rdx, qword ptr [rip + dstp]
    mov qword ptr [rdx + r13*8], rcx
    inc rbx
    inc r13
    jmp m_tail_i
m_tail_j:
    cmp r12, r15
    jae m_done
    mov rax, qword ptr [rip + srcp]
    mov rcx, qword ptr [rax + r12*8]
    mov rdx, qword ptr [rip + dstp]
    mov qword ptr [rdx + r13*8], rcx
    inc r12
    inc r13
    jmp m_tail_j
m_done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

# cmp_lt(rdi = line a, rsi = line b): eax = 1 if line a < line b, else 0
cmp_lt:
    mov r8, qword ptr [lstart + rdi*8]
    mov r9, qword ptr [llen + rdi*8]
    mov r10, qword ptr [lstart + rsi*8]
    mov r11, qword ptr [llen + rsi*8]
    xor ecx, ecx
c_loop:
    cmp rcx, r9
    je c_aend
    cmp rcx, r11
    je c_zero
    movzx eax, byte ptr [inbuf + r8 + rcx]
    movzx edx, byte ptr [inbuf + r10 + rcx]
    cmp eax, edx
    jne c_diff
    inc rcx
    jmp c_loop
c_aend:
    cmp rcx, r11
    setb al
    movzx eax, al
    ret
c_diff:
    setb al
    movzx eax, al
    ret
c_zero:
    xor eax, eax
    ret

.bss
.align 8
inbuf:
    .zero 65536
outbuf:
    .zero 65536
lstart:
    .zero 16384
llen:
    .zero 16384
idxa:
    .zero 16384
idxb:
    .zero 16384
srcp:
    .zero 8
dstp:
    .zero 8
