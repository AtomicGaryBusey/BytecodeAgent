.equ INBUF_MAX, 131072

.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    cmp r13, INBUF_MAX
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov rdx, INBUF_MAX
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r10, r13
    xor ecx, ecx
find_loop:
    cmp rcx, r13
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_loop
found_bang:
    mov r13, rcx
    lea r8, [r12 + rcx + 1]
    jmp have_split
no_bang:
    lea r8, [r12 + r13]
have_split:
    lea r9, [r12 + r10]

    xor ecx, ecx
    xor edx, edx
    lea rbx, [rip + jtab]
    lea rsi, [rip + bstack]
scan:
    cmp rcx, r13
    jae scan_done
    movzx eax, byte ptr [r12 + rcx]
    cmp eax, 0x5B
    je push_b
    cmp eax, 0x5D
    je pop_b
    jmp scan_next
push_b:
    mov dword ptr [rsi + rdx*4], ecx
    inc rdx
    jmp scan_next
pop_b:
    test rdx, rdx
    jz fail
    dec rdx
    mov edi, dword ptr [rsi + rdx*4]
    mov dword ptr [rbx + rdi*4], ecx
    mov dword ptr [rbx + rcx*4], edi
scan_next:
    inc rcx
    jmp scan
scan_done:
    test rdx, rdx
    jnz fail

    xor r14d, r14d
    lea r15, [rip + tape]
run:
    cmp r14, r13
    jae run_done
    movzx eax, byte ptr [r12 + r14]
    cmp eax, 0x3E
    je op_right
    cmp eax, 0x3C
    je op_left
    cmp eax, 0x2B
    je op_inc
    cmp eax, 0x2D
    je op_dec
    cmp eax, 0x2E
    je op_out
    cmp eax, 0x2C
    je op_in
    cmp eax, 0x5B
    je op_open
    cmp eax, 0x5D
    je op_close
    jmp op_next
op_right:
    inc r15
    jmp op_next
op_left:
    dec r15
    jmp op_next
op_inc:
    inc byte ptr [r15]
    jmp op_next
op_dec:
    dec byte ptr [r15]
    jmp op_next
op_out:
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov edx, 1
    syscall
    jmp op_next
op_in:
    cmp r8, r9
    jae in_eof
    movzx eax, byte ptr [r8]
    mov byte ptr [r15], al
    inc r8
    jmp op_next
in_eof:
    mov byte ptr [r15], 0
    jmp op_next
op_open:
    cmp byte ptr [r15], 0
    jne op_next
    mov eax, dword ptr [rbx + r14*4]
    lea r14, [rax + 1]
    jmp run
op_close:
    cmp byte ptr [r15], 0
    je op_next
    mov eax, dword ptr [rbx + r14*4]
    lea r14, [rax + 1]
    jmp run
op_next:
    inc r14
    jmp run
run_done:
    mov eax, 60
    xor edi, edi
    syscall
fail:
    mov eax, 60
    mov edi, 1
    syscall

.bss
.align 16
inbuf:
    .zero 131072
jtab:
    .zero 262144
bstack:
    .zero 262144
tape:
    .zero 30000
