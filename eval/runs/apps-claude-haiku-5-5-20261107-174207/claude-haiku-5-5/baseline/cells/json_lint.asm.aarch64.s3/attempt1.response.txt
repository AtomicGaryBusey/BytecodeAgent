.text
.globl _start

_start:
	adrp	x21, inbuf
	add	x21, x21, :lo12:inbuf
	mov	x20, #0
rd_loop:
	mov	x0, #0
	add	x1, x21, x20
	movz	x2, #1, lsl #16
	sub	x2, x2, x20
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	rd_done
	add	x20, x20, x0
	b	rd_loop
rd_done:
	mov	x19, #0
	bl	skip_ws
	bl	parse_value
	bl	skip_ws
	cmp	x19, x20
	b.ne	err
	mov	x0, #1
	adrp	x1, msg_ok
	add	x1, x1, :lo12:msg_ok
	mov	x2, #3
	mov	x8, #64
	svc	#0
	mov	x0, #0
	mov	x8, #93
	svc	#0

err:
	adrp	x22, outbuf
	add	x22, x22, :lo12:outbuf
	mov	x1, x19
	mov	x3, #10
	add	x2, x22, #31
	mov	w4, #0x0A
	strb	w4, [x2]
err_digit:
	udiv	x4, x1, x3
	msub	x5, x4, x3, x1
	add	w5, w5, #0x30
	sub	x2, x2, #1
	strb	w5, [x2]
	mov	x1, x4
	cbnz	x1, err_digit
	mov	x23, x2
	mov	x0, #1
	adrp	x1, msg_err
	add	x1, x1, :lo12:msg_err
	mov	x2, #9
	mov	x8, #64
	svc	#0
	mov	x0, #1
	mov	x1, x23
	add	x2, x22, #32
	sub	x2, x2, x23
	mov	x8, #64
	svc	#0
	mov	x0, #1
	mov	x8, #93
	svc	#0

skip_ws:
	cmp	x19, x20
	b.hs	sw_done
	ldrb	w0, [x21, x19]
	cmp	w0, #0x20
	b.eq	sw_inc
	cmp	w0, #0x0A
	b.eq	sw_inc
	cmp	w0, #0x0D
	b.eq	sw_inc
	cmp	w0, #0x09
	b.ne	sw_done
sw_inc:
	add	x19, x19, #1
	b	skip_ws
sw_done:
	ret

skip_digits:
	cmp	x19, x20
	b.hs	sd_done
	ldrb	w0, [x21, x19]
	sub	w1, w0, #0x30
	cmp	w1, #9
	b.hi	sd_done
	add	x19, x19, #1
	b	skip_digits
sd_done:
	ret

match_lit:
	mov	x3, #0
ml_loop:
	cmp	x19, x20
	b.hs	err
	ldrb	w4, [x21, x19]
	ldrb	w5, [x1, x3]
	cmp	w4, w5
	b.ne	err
	add	x19, x19, #1
	add	x3, x3, #1
	cmp	x3, x2
	b.lo	ml_loop
	ret

parse_value:
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x7B
	b.eq	parse_object
	cmp	w0, #0x5B
	b.eq	parse_array
	cmp	w0, #0x22
	b.eq	parse_string
	cmp	w0, #0x2D
	b.eq	parse_number
	sub	w1, w0, #0x30
	cmp	w1, #9
	b.ls	parse_number
	cmp	w0, #0x74
	b.eq	lit_true
	cmp	w0, #0x66
	b.eq	lit_false
	cmp	w0, #0x6E
	b.eq	lit_null
	b	err

lit_true:
	adrp	x1, str_true
	add	x1, x1, :lo12:str_true
	mov	x2, #4
	b	match_lit
lit_false:
	adrp	x1, str_false
	add	x1, x1, :lo12:str_false
	mov	x2, #5
	b	match_lit
lit_null:
	adrp	x1, str_null
	add	x1, x1, :lo12:str_null
	mov	x2, #4
	b	match_lit

parse_string:
	add	x19, x19, #1
ps_loop:
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x22
	b.eq	ps_close
	cmp	w0, #0x5C
	b.eq	ps_esc
	cmp	w0, #0x20
	b.lo	err
	add	x19, x19, #1
	b	ps_loop
ps_close:
	add	x19, x19, #1
	ret
ps_esc:
	add	x19, x19, #1
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x22
	b.eq	ps_simple
	cmp	w0, #0x5C
	b.eq	ps_simple
	cmp	w0, #0x2F
	b.eq	ps_simple
	cmp	w0, #0x62
	b.eq	ps_simple
	cmp	w0, #0x66
	b.eq	ps_simple
	cmp	w0, #0x6E
	b.eq	ps_simple
	cmp	w0, #0x72
	b.eq	ps_simple
	cmp	w0, #0x74
	b.eq	ps_simple
	cmp	w0, #0x75
	b.ne	err
	mov	w3, #4
ps_hex:
	add	x19, x19, #1
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	sub	w1, w0, #0x30
	cmp	w1, #9
	b.ls	ps_hexok
	orr	w1, w0, #0x20
	sub	w1, w1, #0x61
	cmp	w1, #5
	b.hi	err
ps_hexok:
	subs	w3, w3, #1
	b.ne	ps_hex
ps_simple:
	add	x19, x19, #1
	b	ps_loop

parse_number:
	stp	x29, x30, [sp, #-16]!
	mov	x29, sp
	ldrb	w0, [x21, x19]
	cmp	w0, #0x2D
	b.ne	num_int
	add	x19, x19, #1
num_int:
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	sub	w1, w0, #0x30
	cmp	w1, #9
	b.hi	err
	cmp	w0, #0x30
	b.ne	num_digits
	add	x19, x19, #1
	b	num_frac
num_digits:
	bl	skip_digits
num_frac:
	cmp	x19, x20
	b.hs	num_exp
	ldrb	w0, [x21, x19]
	cmp	w0, #0x2E
	b.ne	num_exp
	add	x19, x19, #1
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	sub	w1, w0, #0x30
	cmp	w1, #9
	b.hi	err
	bl	skip_digits
num_exp:
	cmp	x19, x20
	b.hs	num_end
	ldrb	w0, [x21, x19]
	orr	w1, w0, #0x20
	cmp	w1, #0x65
	b.ne	num_end
	add	x19, x19, #1
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x2B
	b.eq	num_sign
	cmp	w0, #0x2D
	b.ne	num_expdig
num_sign:
	add	x19, x19, #1
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
num_expdig:
	sub	w1, w0, #0x30
	cmp	w1, #9
	b.hi	err
	bl	skip_digits
num_end:
	ldp	x29, x30, [sp], #16
	ret

parse_array:
	stp	x29, x30, [sp, #-16]!
	mov	x29, sp
	add	x19, x19, #1
	bl	skip_ws
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x5D
	b.eq	arr_end
arr_elem:
	bl	parse_value
	bl	skip_ws
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x2C
	b.ne	arr_tail
	add	x19, x19, #1
	bl	skip_ws
	b	arr_elem
arr_tail:
	cmp	w0, #0x5D
	b.ne	err
arr_end:
	add	x19, x19, #1
	ldp	x29, x30, [sp], #16
	ret

parse_object:
	stp	x29, x30, [sp, #-16]!
	mov	x29, sp
	add	x19, x19, #1
	bl	skip_ws
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x7D
	b.eq	obj_end
obj_mem:
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x22
	b.ne	err
	bl	parse_string
	bl	skip_ws
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x3A
	b.ne	err
	add	x19, x19, #1
	bl	skip_ws
	bl	parse_value
	bl	skip_ws
	cmp	x19, x20
	b.hs	err
	ldrb	w0, [x21, x19]
	cmp	w0, #0x2C
	b.ne	obj_tail
	add	x19, x19, #1
	bl	skip_ws
	b	obj_mem
obj_tail:
	cmp	w0, #0x7D
	b.ne	err
obj_end:
	add	x19, x19, #1
	ldp	x29, x30, [sp], #16
	ret

.section .rodata

	.ascii "ok\n"

	.ascii "error at "

	.ascii "true"

	.ascii "false"

	.ascii "null"

	.zero 65536

	.zero 32
