.text
.globl _start
_start:
	xor r12d, r12d
	xor r13d, r13d
	lea r14, [rip + inbuf]
read_loop:
	xor eax, eax
	xor edi, edi
	lea rsi, [rip + inbuf]
	add rsi, r13
	mov rdx, 131072
	sub rdx, r13
	syscall
	test rax, rax
	jle read_done
	add r13, rax
	jmp read_loop
read_done:
	xor r12d, r12d
	call ws
	call value
	call ws
	cmp r12, r13
	jne fail
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + okmsg]
	mov edx, 3
	syscall
	xor edi, edi
	mov eax, 60
	syscall

# ws: skip space, tab, LF, CR
ws:
ws_loop:
	cmp r12, r13
	jae ws_done
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x20
	je ws_adv
	cmp al, 0x09
	je ws_adv
	cmp al, 0x0A
	je ws_adv
	cmp al, 0x0D
	je ws_adv
ws_done:
	ret
ws_adv:
	inc r12
	jmp ws_loop

# value
value:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x7B
	je object
	cmp al, 0x5B
	je array
	cmp al, 0x22
	je string
	cmp al, 0x2D
	je number
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	jbe number
	cmp al, 0x74
	je lit_t
	cmp al, 0x66
	je lit_f
	cmp al, 0x6E
	je lit_n
	jmp fail
lit_t:
	lea rsi, [rip + s_true]
	jmp literal
lit_f:
	lea rsi, [rip + s_false]
	jmp literal
lit_n:
	lea rsi, [rip + s_null]
	jmp literal

# literal: rsi points to NUL-terminated expected text, r12 at first byte
literal:
lit_loop:
	movzx ecx, byte ptr [rsi]
	test ecx, ecx
	jz lit_done
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp eax, ecx
	jne fail
	inc r12
	inc rsi
	jmp lit_loop
lit_done:
	ret

# object
object:
	inc r12
	call ws
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x7D
	jne obj_member
	inc r12
	ret
obj_member:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x22
	jne fail
	call string
	call ws
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x3A
	jne fail
	inc r12
	call ws
	call value
	call ws
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x2C
	je obj_comma
	cmp al, 0x7D
	jne fail
	inc r12
	ret
obj_comma:
	inc r12
	call ws
	jmp obj_member

# array
array:
	inc r12
	call ws
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x5D
	jne arr_value
	inc r12
	ret
arr_value:
	call value
	call ws
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x2C
	je arr_comma
	cmp al, 0x5D
	jne fail
	inc r12
	ret
arr_comma:
	inc r12
	call ws
	jmp arr_value

# string: r12 at opening quote
string:
	inc r12
str_loop:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x22
	je str_end
	cmp al, 0x5C
	je str_esc
	cmp al, 0x20
	jb fail
	inc r12
	jmp str_loop
str_end:
	inc r12
	ret
str_esc:
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x22
	je str_simple
	cmp al, 0x5C
	je str_simple
	cmp al, 0x2F
	je str_simple
	cmp al, 0x62
	je str_simple
	cmp al, 0x66
	je str_simple
	cmp al, 0x6E
	je str_simple
	cmp al, 0x72
	je str_simple
	cmp al, 0x74
	je str_simple
	cmp al, 0x75
	je str_u
	jmp fail
str_simple:
	inc r12
	jmp str_loop
str_u:
	inc r12
	mov edx, 4
str_hex:
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	jbe str_hex_ok
	or al, 0x20
	cmp al, 0x61
	jb fail
	cmp al, 0x66
	ja fail
str_hex_ok:
	inc r12
	dec edx
	jnz str_hex
	jmp str_loop

# number
number:
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x2D
	jne num_nosign
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
num_nosign:
	cmp al, 0x30
	jne num_nz
	inc r12
	jmp num_frac
num_nz:
	cmp al, 0x31
	jb fail
	cmp al, 0x39
	ja fail
num_idig:
	inc r12
	cmp r12, r13
	jae num_frac
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x30
	jb num_frac
	cmp al, 0x39
	jbe num_idig
num_frac:
	cmp r12, r13
	jae num_exp
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x2E
	jne num_exp
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	ja fail
num_fdig:
	inc r12
	cmp r12, r13
	jae num_exp
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x30
	jb num_exp
	cmp al, 0x39
	jbe num_fdig
num_exp:
	cmp r12, r13
	jae num_done
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x65
	je num_e
	cmp al, 0x45
	jne num_done
num_e:
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x2B
	je num_esign
	cmp al, 0x2D
	jne num_edig
num_esign:
	inc r12
	cmp r12, r13
	jae fail
	movzx eax, byte ptr [r14 + r12]
num_edig:
	cmp al, 0x30
	jb fail
	cmp al, 0x39
	ja fail
num_xdig:
	inc r12
	cmp r12, r13
	jae num_done
	movzx eax, byte ptr [r14 + r12]
	cmp al, 0x30
	jb num_done
	cmp al, 0x39
	jbe num_xdig
num_done:
	ret

# fail: r12 = offset of error; print "error at N\n" and exit 1
fail:
	lea rsi, [rip + numbuf + 20]
	mov rax, r12
	mov rcx, 10
fail_conv:
	xor edx, edx
	div rcx
	add dl, 0x30
	dec rsi
	mov byte ptr [rsi], dl
	test rax, rax
	jnz fail_conv
	mov r8, rsi
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + errmsg]
	mov edx, 9
	syscall
	mov eax, 1
	mov edi, 1
	mov rsi, r8
	lea rdx, [rip + numbuf + 20]
	sub rdx, r8
	syscall
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + nlmsg]
	mov edx, 1
	syscall
	mov eax, 60
	mov edi, 1
	syscall

.section .rodata
errmsg:
	.ascii "error at "
nlmsg:
	.byte 0x0A
okmsg:
	.byte 0x6F, 0x6B, 0x0A
s_true:
	.asciz "true"
s_false:
	.asciz "false"
s_null:
	.asciz "null"

.bss
inbuf:
	.zero 131072
numbuf:
	.zero 32
