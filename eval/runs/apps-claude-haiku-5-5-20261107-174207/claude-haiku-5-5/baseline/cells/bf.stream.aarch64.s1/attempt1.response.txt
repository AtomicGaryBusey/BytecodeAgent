.globl _start

_start:
    // read all of standard input into inbuf (x19 = base, x20 = length N)
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0
read_loop:
    movz x2, #1, lsl #16
    sub  x2, x2, x20
    cbz  x2, read_done
    add  x1, x19, x20
    mov  x0, #0
    mov  x8, #63
    svc  #0
    tbnz x0, #63, read_done
    cbz  x0, read_done
    add  x20, x20, x0
    b    read_loop
read_done:

    // find first '!' : x21 = P (program length)
    mov  x21, #0
find_loop:
    cmp  x21, x20
    b.hs no_bang
    ldrb w0, [x19, x21]
    cmp  w0, #33
    b.eq found_bang
    add  x21, x21, #1
    b    find_loop
no_bang:
    add  x27, x19, x20
    mov  x28, x27
    b    have_input
found_bang:
    add  x27, x19, x21
    add  x27, x27, #1
    add  x28, x19, x20
have_input:

    // bracket check and jump table build
    adrp x24, jt
    add  x24, x24, :lo12:jt
    adrp x22, stk
    add  x22, x22, :lo12:stk
    mov  x23, #0
    mov  x9, #0
bracket_loop:
    cmp  x9, x21
    b.hs bracket_done
    ldrb w0, [x19, x9]
    cmp  w0, #91
    b.ne check_close
    str  w9, [x22, x23, lsl #2]
    add  x23, x23, #1
    b    bracket_next
check_close:
    cmp  w0, #93
    b.ne bracket_next
    cbz  x23, fail
    sub  x23, x23, #1
    ldr  w10, [x22, x23, lsl #2]
    str  w9, [x24, x10, lsl #2]
    str  w10, [x24, x9, lsl #2]
bracket_next:
    add  x9, x9, #1
    b    bracket_loop
bracket_done:
    cbnz x23, fail

    // set up interpreter state
    adrp x22, tape
    add  x22, x22, :lo12:tape
    mov  x23, x22                 // data pointer
    adrp x25, outbuf
    add  x25, x25, :lo12:outbuf   // output buffer
    mov  x26, #0                  // output length
    mov  x9, #0                   // pc

run_loop:
    cmp  x9, x21
    b.hs run_done
    ldrb w0, [x19, x9]
    cmp  w0, #62
    b.eq op_right
    cmp  w0, #60
    b.eq op_left
    cmp  w0, #43
    b.eq op_inc
    cmp  w0, #45
    b.eq op_dec
    cmp  w0, #46
    b.eq op_out
    cmp  w0, #44
    b.eq op_in
    cmp  w0, #91
    b.eq op_open
    cmp  w0, #93
    b.eq op_close
    b    op_next

op_right:
    add  x23, x23, #1
    b    op_next
op_left:
    sub  x23, x23, #1
    b    op_next
op_inc:
    ldrb w1, [x23]
    add  w1, w1, #1
    strb w1, [x23]
    b    op_next
op_dec:
    ldrb w1, [x23]
    sub  w1, w1, #1
    strb w1, [x23]
    b    op_next
op_out:
    ldrb w1, [x23]
    strb w1, [x25, x26]
    add  x26, x26, #1
    b    op_next
op_in:
    cmp  x27, x28
    b.hs in_eof
    ldrb w1, [x27], #1
    b    in_store
in_eof:
    mov  w1, #0
in_store:
    strb w1, [x23]
    b    op_next
op_open:
    ldrb w1, [x23]
    cbnz w1, op_next
    ldr  w1, [x24, x9, lsl #2]
    add  x9, x1, #1
    b    run_loop
op_close:
    ldrb w1, [x23]
    cbz  w1, op_next
    ldr  w1, [x24, x9, lsl #2]
    add  x9, x1, #1
    b    run_loop
op_next:
    add  x9, x9, #1
    b    run_loop

run_done:
    // write output buffer
    mov  x9, #0
write_loop:
    cmp  x9, x26
    b.hs exit_ok
    mov  x0, #1
    add  x1, x25, x9
    sub  x2, x26, x9
    mov  x8, #64
    svc  #0
    tbnz x0, #63, exit_ok
    cbz  x0, exit_ok
    add  x9, x9, x0
    b    write_loop

exit_ok:
    mov  x0, #0
    mov  x8, #93
    svc  #0

fail:
    mov  x0, #1
    mov  x8, #93
    svc  #0

.bss
.balign 16

    .zero 65536

    .zero 30000
.balign 16

    .zero 262144
.balign 16

.balign 16

    .zero 65536
