// hexdump: canonical hex+ASCII dump of standard input (AArch64 Linux, no libc)

.section .bss
.balign 16
inbuf:
    .zero 65536
linebuf:
    .zero 128

.text
.globl _start
_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    adrp x25, linebuf
    add  x25, x25, :lo12:linebuf
    mov  x20, #0                    // total length

// read all of stdin into inbuf
read_loop:
    mov  x0, #0
    add  x1, x19, x20
    mov  x2, #0x10000
    sub  x2, x2, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    mov  x22, #0                    // row offset

row_loop:
    cmp  x22, x20
    b.hs finish
    sub  x23, x20, x22              // remaining
    cmp  x23, #16
    mov  x9, #16
    csel x23, x23, x9, lo           // n = min(16, remaining)

    mov  x1, x25
    // 8 hex digits of offset
    ubfx w0, w22, #24, #8
    bl   hexbyte
    ubfx w0, w22, #16, #8
    bl   hexbyte
    ubfx w0, w22, #8, #8
    bl   hexbyte
    ubfx w0, w22, #0, #8
    bl   hexbyte
    mov  w2, #0x2020               // two spaces
    strh w2, [x1], #2

    mov  x24, #0
hex_loop:
    cmp  x24, #16
    b.hs hex_done
    cmp  x24, x23
    b.hs hex_pad
    add  x9, x22, x24
    ldrb w0, [x19, x9]
    bl   hexbyte
    mov  w2, #0x20
    strb w2, [x1], #1
    b    hex_sep
hex_pad:
    mov  w2, #0x2020               // three spaces
    strh w2, [x1], #2
    mov  w2, #0x20
    strb w2, [x1], #1
hex_sep:
    cmp  x24, #7
    b.ne hex_next
    mov  w2, #0x20                 // extra space after 8th slot
    strb w2, [x1], #1
hex_next:
    add  x24, x24, #1
    b    hex_loop

hex_done:
    mov  w2, #0x7c20               // " |"
    strh w2, [x1], #2

    mov  x24, #0
chr_loop:
    cmp  x24, x23
    b.hs chr_done
    add  x9, x22, x24
    ldrb w0, [x19, x9]
    sub  w2, w0, #0x20
    cmp  w2, #0x5e
    b.ls chr_store
    mov  w0, #0x2e                 // '.'
chr_store:
    strb w0, [x1], #1
    add  x24, x24, #1
    b    chr_loop

chr_done:
    mov  w2, #0x0a7c               // "|\n"
    strh w2, [x1], #2
    bl   flush
    add  x22, x22, x23
    b    row_loop

finish:
    mov  x1, x25
    ubfx w0, w20, #24, #8
    bl   hexbyte
    ubfx w0, w20, #16, #8
    bl   hexbyte
    ubfx w0, w20, #8, #8
    bl   hexbyte
    ubfx w0, w20, #0, #8
    bl   hexbyte
    mov  w2, #0x0a
    strb w2, [x1], #1
    bl   flush
    mov  x0, #0
    mov  x8, #93
    svc  #0

// flush: write linebuf[0 .. x1) to stdout completely
flush:
    sub  x27, x1, x25
    mov  x26, x25
flush_loop:
    cbz  x27, flush_done
    mov  x0, #1
    mov  x1, x26
    mov  x2, x27
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le flush_err
    add  x26, x26, x0
    sub  x27, x27, x0
    b    flush_loop
flush_done:
    ret
flush_err:
    mov  x0, #1
    mov  x8, #93
    svc  #0

// hexbyte: w0 = byte (0..255), x1 = destination; writes 2 hex chars, advances x1
hexbyte:
    and  w3, w0, #0xf
    lsr  w2, w0, #4
    add  w2, w2, #0x30
    cmp  w2, #0x39
    b.ls hb_hi
    add  w2, w2, #39
hb_hi:
    strb w2, [x1], #1
    add  w3, w3, #0x30
    cmp  w3, #0x39
    b.ls hb_lo
    add  w3, w3, #39
hb_lo:
    strb w3, [x1], #1
    ret
