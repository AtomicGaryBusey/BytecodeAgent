.intel_syntax noprefix is not needed here.
.text
.globl _start

.equ S_V, 0
.equ S_AV, 1
.equ S_AO, 2
.equ S_AK, 3
.equ S_COLON, 4
.equ S_AFTER, 5
.equ S_STR, 6
.equ S_ESC, 7
.equ S_U, 8
.equ S_LIT, 9
.equ S_MINUS, 10
.equ S_ZERO, 11
.equ S_INT, 12
.equ S_FRAC0, 13
.equ S_FRAC, 14
.equ S_EXP0, 15
.equ S_EXPSIGN, 16
.equ S_EXPDIG, 17

_start:
    xor r13d, r13d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov rdx, 131072
    sub rdx, r13
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea rbx, [rip + inbuf]
    lea rbp, [rip + stk]
    xor r12d, r12d
    xor r14d, r14d
    xor r15d, r15d
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

main_loop:
    cmp r12, r13
    jae at_eof
    movzx eax, byte ptr [rbx + r12]
    cmp r14, S_V
    je st_v
    cmp r14, S_AV
    je st_av
    cmp r14, S_AO
    je st_ao
    cmp r14, S_AK
    je st_ak
    cmp r14, S_COLON
    je st_colon
    cmp r14, S_AFTER
    je st_after
    cmp r14, S_STR
    je st_str
    cmp r14, S_ESC
    je st_esc
    cmp r14, S_U
    je st_u
    cmp r14, S_LIT
    je st_lit
    cmp r14, S_MINUS
    je st_minus
    cmp r14, S_ZERO
    je st_zero
    cmp r14, S_INT
    je st_int
    cmp r14, S_FRAC0
    je st_frac0
    cmp r14, S_FRAC
    je st_frac
    cmp r14, S_EXP0
    je st_exp0
    cmp r14, S_EXPSIGN
    je st_expsign
    cmp r14, S_EXPDIG
    je st_expdig
    jmp fail

adv:
    inc r12
    jmp main_loop

num_complete:
    mov r14d, S_AFTER
    jmp main_loop

# value start, al = non-whitespace byte, not consumed
vstart:
    cmp al, 0x7B
    je vs_obj
    cmp al, 0x5B
    je vs_arr
    cmp al, 0x22
    je vs_str
    cmp al, 0x2D
    je vs_minus
    cmp al, 0x30
    je vs_zero
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja vs_check_lit
    mov r14d, S_INT
    jmp adv
vs_check_lit:
    cmp al, 0x74
    je vs_t
    cmp al, 0x66
    je vs_f
    cmp al, 0x6E
    je vs_n
    jmp fail
vs_obj:
    mov byte ptr [rbp + r15], 0x7B
    inc r15
    mov r14d, S_AO
    jmp adv
vs_arr:
    mov byte ptr [rbp + r15], 0x5B
    inc r15
    mov r14d, S_AV
    jmp adv
vs_str:
    xor r9d, r9d
    mov r14d, S_STR
    jmp adv
vs_minus:
    mov r14d, S_MINUS
    jmp adv
vs_zero:
    mov r14d, S_ZERO
    jmp adv
vs_t:
    lea r8, [rip + lit_rue]
    mov r14d, S_LIT
    jmp adv
vs_f:
    lea r8, [rip + lit_alse]
    mov r14d, S_LIT
    jmp adv
vs_n:
    lea r8, [rip + lit_ull]
    mov r14d, S_LIT
    jmp adv

# V: expecting value, whitespace allowed
st_v:
    cmp al, 0x20
    je adv
    cmp al, 0x09
    je adv
    cmp al, 0x0A
    je adv
    cmp al, 0x0D
    je adv
    jmp vstart

# AV: after '[', whitespace, ']' or value
st_av:
    cmp al, 0x20
    je adv
    cmp al, 0x09
    je adv
    cmp al, 0x0A
    je adv
    cmp al, 0x0D
    je adv
    cmp al, 0x5D
    jne vstart
    dec r15
    mov r14d, S_AFTER
    jmp adv

# AO: after '{', whitespace, '}' or key
st_ao:
    cmp al, 0x20
    je adv
    cmp al, 0x09
    je adv
    cmp al, 0x0A
    je adv
    cmp al, 0x0D
    je adv
    cmp al, 0x7D
    jne st_ak_chk
    dec r15
    mov r14d, S_AFTER
    jmp adv

# AK: key required after ','
st_ak:
    cmp al, 0x20
    je adv
    cmp al, 0x09
    je adv
    cmp al, 0x0A
    je adv
    cmp al, 0x0D
    je adv
st_ak_chk:
    cmp al, 0x22
    jne fail
    mov r9d, 1
    mov r14d, S_STR
    jmp adv

st_colon:
    cmp al, 0x20
    je adv
    cmp al, 0x09
    je adv
    cmp al, 0x0A
    je adv
    cmp al, 0x0D
    je adv
    cmp al, 0x3A
    jne fail
    mov r14d, S_V
    jmp adv

st_after:
    cmp al, 0x20
    je adv
    cmp al, 0x09
    je adv
    cmp al, 0x0A
    je adv
    cmp al, 0x0D
    je adv
    test r15, r15
    jz fail
    movzx ecx, byte ptr [rbp + r15 - 1]
    cmp al, 0x2C
    jne sa_close
    cmp cl, 0x7B
    jne sa_arr
    mov r14d, S_AK
    jmp adv
sa_arr:
    mov r14d, S_V
    jmp adv
sa_close:
    cmp al, 0x5D
    jne sa_cb
    cmp cl, 0x5B
    jne fail
    dec r15
    jmp adv
sa_cb:
    cmp al, 0x7D
    jne fail
    cmp cl, 0x7B
    jne fail
    dec r15
    jmp adv

# string
st_str:
    cmp al, 0x22
    jne st_str2
    mov r14d, S_AFTER
    test r9d, r9d
    jz adv
    mov r14d, S_COLON
    jmp adv
st_str2:
    cmp al, 0x5C
    jne st_str3
    mov r14d, S_ESC
    jmp adv
st_str3:
    cmp al, 0x20
    jb fail
    jmp adv

st_esc:
    cmp al, 0x75
    jne esc_simple
    mov r10d, 4
    mov r14d, S_U
    jmp adv
esc_simple:
    cmp al, 0x22
    je esc_ok
    cmp al, 0x5C
    je esc_ok
    cmp al, 0x2F
    je esc_ok
    cmp al, 0x62
    je esc_ok
    cmp al, 0x66
    je esc_ok
    cmp al, 0x6E
    je esc_ok
    cmp al, 0x72
    je esc_ok
    cmp al, 0x74
    je esc_ok
    jmp fail
esc_ok:
    mov r14d, S_STR
    jmp adv

st_u:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    jbe u_ok
    cmp al, 0x41
    jb fail
    cmp al, 0x46
    jbe u_ok
    cmp al, 0x61
    jb fail
    cmp al, 0x66
    ja fail
u_ok:
    dec r10d
    jnz adv
    mov r14d, S_STR
    jmp adv

# literals
st_lit:
    cmp al, byte ptr [r8]
    jne fail
    inc r8
    cmp byte ptr [r8], 0
    jne adv
    mov r14d, S_AFTER
    jmp adv

# numbers
st_minus:
    cmp al, 0x30
    jne sm_nz
    mov r14d, S_ZERO
    jmp adv
sm_nz:
    cmp al, 0x31
    jb fail
    cmp al, 0x39
    ja fail
    mov r14d, S_INT
    jmp adv

st_zero:
    cmp al, 0x2E
    jne sz2
    mov r14d, S_FRAC0
    jmp adv
sz2:
    cmp al, 0x65
    je sz_e
    cmp al, 0x45
    je sz_e
    jmp num_complete
sz_e:
    mov r14d, S_EXP0
    jmp adv

st_int:
    cmp al, 0x30
    jb si2
    cmp al, 0x39
    ja si2
    jmp adv
si2:
    cmp al, 0x2E
    jne si3
    mov r14d, S_FRAC0
    jmp adv
si3:
    cmp al, 0x65
    je si_e
    cmp al, 0x45
    je si_e
    jmp num_complete
si_e:
    mov r14d, S_EXP0
    jmp adv

st_frac0:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    mov r14d, S_FRAC
    jmp adv

st_frac:
    cmp al, 0x30
    jb sf2
    cmp al, 0x39
    ja sf2
    jmp adv
sf2:
    cmp al, 0x65
    je sf_e
    cmp al, 0x45
    je sf_e
    jmp num_complete
sf_e:
    mov r14d, S_EXP0
    jmp adv

st_exp0:
    cmp al, 0x2B
    je se_sign
    cmp al, 0x2D
    je se_sign
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    mov r14d, S_EXPDIG
    jmp adv
se_sign:
    mov r14d, S_EXPSIGN
    jmp adv

st_expsign:
    cmp al, 0x30
    jb fail
    cmp al, 0x39
    ja fail
    mov r14d, S_EXPDIG
    jmp adv

st_expdig:
    cmp al, 0x30
    jb num_complete
    cmp al, 0x39
    ja num_complete
    jmp adv

# end of input
at_eof:
    cmp r14, S_ZERO
    je eof_acc
    cmp r14, S_INT
    je eof_acc
    cmp r14, S_FRAC
    je eof_acc
    cmp r14, S_EXPDIG
    je eof_acc
    cmp r14, S_AFTER
    jne eof_bad
eof_acc:
    test r15, r15
    jnz eof_bad
    jmp success
eof_bad:
    mov r12, r13
    jmp fail

success:
    lea rsi, [rip + msg_ok]
    mov edx, 3
    call write_all
    xor eax, eax
    jmp do_exit

# prints "error at N\n" with N = r12, exits 1
fail:
    mov rax, r12
    lea rsi, [rip + numbuf + 31]
    mov byte ptr [rsi], 10
    mov r11, 10
f_conv:
    xor edx, edx
    div r11
    add dl, 0x30
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz f_conv
    mov r10, rsi
    lea rsi, [rip + msg_err]
    mov edx, 9
    call write_all
    mov rsi, r10
    lea rdx, [rip + numbuf + 32]
    sub rdx, r10
    call write_all
    mov eax, 1
    jmp do_exit

# write_all: rsi = buffer, rdx = count
write_all:
    test rdx, rdx
    jz wa_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wa_done
    add rsi, rax
    sub rdx, rax
    jmp write_all
wa_done:
    ret

do_exit:
    mov edi, eax
    mov eax, 60
    syscall

.section .rodata
msg_ok:
    .ascii "ok\n"
msg_err:
    .ascii "error at "
lit_rue:
    .asciz "rue"
lit_alse:
    .asciz "alse"
lit_ull:
    .asciz "ull"

.bss
inbuf:
    .zero 131072
stk:
    .zero 2048
numbuf:
    .zero 64
