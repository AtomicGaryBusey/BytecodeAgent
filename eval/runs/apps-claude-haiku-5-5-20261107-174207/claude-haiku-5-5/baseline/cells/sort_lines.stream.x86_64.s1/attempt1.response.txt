```
.globl _start

.equ INCAP, 32832
.equ MAXLINES, 1100

.text
_start:
    # read all of standard input into inbuf
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov rdx, INCAP
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:

    # split into lines: ents[i] = (start, len)
    xor r12d, r12d
    xor r15d, r15d
    xor r14d, r14d
    lea rbp, [rip + inbuf]
    lea rbx, [rip + ents]
parse_loop:
    cmp r12, r13
    jae parse_tail
    cmp byte ptr [rbp + r12], 10
    jne parse_next
    mov rax, r14
    shl rax, 4
    mov [rbx + rax], r15
    mov rdx, r12
    sub rdx, r15
    mov [rbx + rax + 8], rdx
    inc r14
    lea r15, [r12 + 1]
parse_next:
    inc r12
    jmp parse_loop
parse_tail:
    cmp r15, r13
    jae parse_done
    mov rax, r14
    shl rax, 4
    mov [rbx + rax], r15
    mov rdx, r13
    sub rdx, r15
    mov [rbx + rax + 8], rdx
    inc r14
parse_done:

    # insertion sort of ents by line contents
    lea rbp, [rip + ents]
    mov r12, 1
outer_loop:
    cmp r12, r14
    jae sort_done
    mov rax, r12
    shl rax, 4
    mov r15, [rbp + rax]
    mov rbx, [rbp + rax + 8]
    mov r13, r12
inner_loop:
    test r13, r13
    jz place
    mov rax, r13
    dec rax
    shl rax, 4
    mov rsi, [rbp + rax + 8]
    mov rdi, [rbp + rax]
    lea r8, [rip + inbuf]
    add rdi, r8
    mov rcx, rbx
    lea rdx, [rip + inbuf]
    add rdx, r15
    call cmp_gt
    test eax, eax
    jz place
    mov rax, r13
    dec rax
    shl rax, 4
    mov rcx, r13
    shl rcx, 4
    mov rdx, [rbp + rax]
    mov [rbp + rcx], rdx
    mov rdx, [rbp + rax + 8]
    mov [rbp + rcx + 8], rdx
    dec r13
    jmp inner_loop
place:
    mov rax, r13
    shl rax, 4
    mov [rbp + rax], r15
    mov [rbp + rax + 8], rbx
    inc r12
    jmp outer_loop
sort_done:

    # build output buffer: each line followed by newline
    xor r12d, r12d
    xor r13d, r13d
    lea rbp, [rip + ents]
out_loop:
    cmp r12, r14
    jae out_done
    mov rax, r12
    shl rax, 4
    mov rdx, [rbp + rax]
    mov rcx, [rbp + rax + 8]
    lea rsi, [rip + inbuf]
    add rsi, rdx
    lea rdi, [rip + outbuf]
    add rdi, r13
    add r13, rcx
    cld
    rep movsb
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r13], 10
    inc r13
    inc r12
    jmp out_loop
out_done:

    # write output buffer completely
    lea rsi, [rip + outbuf]
    mov rdx, r13
write_loop:
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

# cmp_gt: rdi = A ptr, rsi = A len, rdx = B ptr, rcx = B len
# returns eax = 1 if A > B (lexicographic unsigned), else 0
cmp_gt:
    mov r8, rsi
    cmp rcx, rsi
    jae 1f
    mov r8, rcx
1:
    xor r9d, r9d
2:
    cmp r9, r8
    je 3f
    movzx eax, byte ptr [rdi + r9]
    movzx r10d, byte ptr [rdx + r9]
    cmp eax, r10d
    jne 4f
    inc r9
    jmp 2b
3:
    cmp rsi, rcx
    seta al
    movzx eax, al
    ret
4:
    seta al
    movzx eax, al
    ret

.bss
.align 16
ents:
    .zero MAXLINES * 16
inbuf:
    .zero INCAP
outbuf:
    .zero 33024
```
