.globl _start
_start:
    adrp x25, inbuf
    add x25, x25, :lo12:inbuf
    adrp x26, stk
    add x26, x26, :lo12:stk
    mov x19, #0
read_loop:
    add x1, x25, x19
    movz x2, #2, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    ble read_done
    add x19, x19, x0
    b read_loop
read_done:
    mov x20, #0
line_loop:
    cmp x20, x19
    bge exit_ok
    mov x21, x20
find_nl:
    cmp x21, x19
    bge line_found
    ldrb w0, [x25, x21]
    cmp w0, #10
    beq line_found
    add x21, x21, #1
    b find_nl
line_found:
    mov x22, x20
    mov x23, #0
    mov x24, #0
tok_loop:
    cmp x22, x21
    bge line_done
    ldrb w0, [x25, x22]
    cmp w0, #32
    beq skip_sep
    cmp w0, #9
    beq skip_sep
    mov x27, x22
tok_scan:
    cmp x27, x21
    bge tok_found
    ldrb w0, [x25, x27]
    cmp w0, #32
    beq tok_found
    cmp w0, #9
    beq tok_found
    add x27, x27, #1
    b tok_scan
tok_found:
    bl process_token
    cbnz x24, line_done
    mov x22, x27
    b tok_loop
skip_sep:
    add x22, x22, #1
    b tok_loop
line_done:
    cbnz x24, print_err
    cmp x23, #1
    bne print_err
    ldr x0, [x26]
    bl print_int
    b next_line
print_err:
    adrp x1, msg_error
    add x1, x1, :lo12:msg_error
    mov x2, #6
    bl write_buf
next_line:
    add x20, x21, #1
    b line_loop
exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

// process_token: token = [x22, x27), state x23 (depth), x24 (error)
process_token:
    ldrb w2, [x25, x22]
    sub x1, x27, x22
    cmp x1, #1
    bne pt_literal
    cmp w2, #43
    beq pt_op
    cmp w2, #45
    beq pt_op
    cmp w2, #42
    beq pt_op
    cmp w2, #47
    beq pt_op
    cmp w2, #37
    beq pt_op
pt_literal:
    mov x3, x22
    mov x4, #0
    cmp w2, #45
    bne pt_nosign
    mov x4, #1
    add x3, x3, #1
pt_nosign:
    sub x5, x27, x3
    cmp x5, #1
    blt pt_err
    cmp x5, #18
    bgt pt_err
    mov x6, #0
pt_dloop:
    cmp x3, x27
    bge pt_ddone
    ldrb w7, [x25, x3]
    sub w7, w7, #48
    cmp w7, #9
    bhi pt_err
    mov x8, #10
    madd x6, x6, x8, x7
    add x3, x3, #1
    b pt_dloop
pt_ddone:
    cbz x4, pt_pos
    neg x6, x6
pt_pos:
    cmp x23, #1024
    bge pt_err
    str x6, [x26, x23, lsl #3]
    add x23, x23, #1
    ret
pt_err:
    mov x24, #1
    ret
pt_op:
    cmp x23, #2
    blt pt_err
    sub x23, x23, #1
    ldr x4, [x26, x23, lsl #3]
    sub x23, x23, #1
    ldr x3, [x26, x23, lsl #3]
    cmp w2, #43
    beq pt_add
    cmp w2, #45
    beq pt_sub
    cmp w2, #42
    beq pt_mul
    cmp w2, #47
    beq pt_divop
    b pt_modop
pt_add:
    add x6, x3, x4
    b pt_push_res
pt_sub:
    sub x6, x3, x4
    b pt_push_res
pt_mul:
    mul x6, x3, x4
    b pt_push_res
pt_divop:
    cbz x4, pt_err
    cmn x4, #1
    bne pt_dv
    mov x5, #1
    lsl x5, x5, #63
    cmp x3, x5
    beq pt_err
pt_dv:
    sdiv x6, x3, x4
    b pt_push_res
pt_modop:
    cbz x4, pt_err
    cmn x4, #1
    bne pt_mv
    mov x5, #1
    lsl x5, x5, #63
    cmp x3, x5
    beq pt_err
pt_mv:
    sdiv x6, x3, x4
    msub x6, x6, x4, x3
    b pt_push_res
pt_push_res:
    str x6, [x26, x23, lsl #3]
    add x23, x23, #1
    ret

// write_buf: write x2 bytes at x1 to fd 1
write_buf:
wb_loop:
    cbz x2, wb_done
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    ble wb_done
    add x1, x1, x0
    sub x2, x2, x0
    b wb_loop
wb_done:
    ret

// print_int: print signed x0 in decimal followed by newline
print_int:
    adrp x9, outbuf
    add x9, x9, :lo12:outbuf
    add x10, x9, #32
    mov x11, x10
    mov x12, #0
    cmp x0, #0
    bge pi_pos
    mov x12, #1
    neg x0, x0
pi_pos:
    mov x13, #10
    sub x11, x11, #1
    mov w15, #10
    strb w15, [x11]
pi_loop:
    udiv x14, x0, x13
    msub x15, x14, x13, x0
    add x15, x15, #48
    sub x11, x11, #1
    strb w15, [x11]
    mov x0, x14
    cbnz x0, pi_loop
    cbz x12, pi_nominus
    sub x11, x11, #1
    mov w15, #45
    strb w15, [x11]
pi_nominus:
    mov x1, x11
    sub x2, x10, x11
    b write_buf

.section .rodata
