.globl _start
.section .rodata
msg_error:
    .ascii "error\n"

.section .bss
.align 16
inbuf:
    .zero 65536
stack:
    .zero 8192
outbuf:
    .zero 4096
numbuf:
    .zero 40
inlen:
    .zero 8
outpos:
    .zero 8

.text
_start:
    xor r12d, r12d
rd_loop:
    mov rdx, 65536
    sub rdx, r12
    jz rd_done
    lea rsi, [rip + inbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle rd_done
    add r12, rax
    jmp rd_loop
rd_done:
    mov qword ptr [rip + inlen], r12
    mov r12, qword ptr [rip + inlen]
    xor r13d, r13d
line_start:
    cmp r13, r12
    jae all_done
    xor r14d, r14d
    xor r15d, r15d
tok_loop:
skip:
    cmp r13, r12
    jae end_line
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + r13]
    cmp eax, 32
    je skip_adv
    cmp eax, 9
    je skip_adv
    cmp eax, 10
    je nl_found
    jmp tok_scan_start
skip_adv:
    inc r13
    jmp skip
nl_found:
    inc r13
    jmp end_line
tok_scan_start:
    mov rsi, r13
    mov rdi, r13
tok_scan:
    cmp rdi, r12
    jae tok_end
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + rdi]
    cmp eax, 32
    je tok_end
    cmp eax, 9
    je tok_end
    cmp eax, 10
    je tok_end
    inc rdi
    jmp tok_scan
tok_end:
    mov r13, rdi
    mov rcx, rdi
    sub rcx, rsi
    cmp r15, 0
    jne tok_loop
    cmp rcx, 1
    jne lit_start
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + rsi]
    cmp eax, 43
    je op_dispatch
    cmp eax, 45
    je op_dispatch
    cmp eax, 42
    je op_dispatch
    cmp eax, 47
    je op_dispatch
    cmp eax, 37
    je op_dispatch
    jmp lit_start

op_dispatch:
    cmp r14, 2
    jb set_err
    lea rdx, [rip + stack]
    mov r8, qword ptr [rdx + r14*8 - 8]
    mov r9, qword ptr [rdx + r14*8 - 16]
    sub r14, 2
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + rsi]
    cmp eax, 43
    je op_add
    cmp eax, 45
    je op_sub
    cmp eax, 42
    je op_mul
    cmp eax, 47
    je op_div
    jmp op_mod

op_add:
    mov rax, r9
    add rax, r8
    mov r8, rax
    jmp push_r8
op_sub:
    mov rax, r9
    sub rax, r8
    mov r8, rax
    jmp push_r8
op_mul:
    mov rax, r9
    imul rax, r8
    mov r8, rax
    jmp push_r8
op_div:
    test r8, r8
    jz set_err
    cmp r8, -1
    jne dv_go
    movabs rax, 0x8000000000000000
    cmp r9, rax
    je set_err
dv_go:
    mov rax, r9
    cqo
    idiv r8
    mov r8, rax
    jmp push_r8
op_mod:
    test r8, r8
    jz set_err
    cmp r8, -1
    jne md_go
    movabs rax, 0x8000000000000000
    cmp r9, rax
    je set_err
md_go:
    mov rax, r9
    cqo
    idiv r8
    mov r8, rdx
    jmp push_r8

lit_start:
    lea rbx, [rip + inbuf]
    add rbx, rsi
    xor r8d, r8d
    xor r9d, r9d
    xor edi, edi
    movzx eax, byte ptr [rbx]
    cmp eax, 45
    jne lit_loop
    mov r9d, 1
    mov edi, 1
lit_loop:
    cmp rdi, rcx
    jae lit_done
    movzx eax, byte ptr [rbx + rdi]
    sub eax, 48
    cmp eax, 9
    ja set_err
    imul r8, r8, 10
    add r8, rax
    inc rdi
    jmp lit_loop
lit_done:
    mov rax, rcx
    sub rax, r9
    cmp rax, 18
    ja set_err
    test r9, r9
    jz push_r8
    neg r8
    jmp push_r8

push_r8:
    cmp r14, 1024
    jae set_err
    lea rdx, [rip + stack]
    mov qword ptr [rdx + r14*8], r8
    inc r14
    jmp tok_loop

set_err:
    mov r15d, 1
    jmp tok_loop

end_line:
    cmp qword ptr [rip + outpos], 4000
    jbe el_go
    call flush
el_go:
    test r15, r15
    jnz el_err
    cmp r14, 1
    jne el_err
    lea rdx, [rip + stack]
    mov rax, qword ptr [rdx]
    call print_num
    call emit_nl
    jmp line_start
el_err:
    lea rsi, [rip + msg_error]
    mov ecx, 6
    call emit
    jmp line_start

# rsi = source, rcx = count; appends to outbuf
emit:
    lea rdi, [rip + outbuf]
    add rdi, qword ptr [rip + outpos]
    add qword ptr [rip + outpos], rcx
    rep movsb
    ret

emit_nl:
    lea rdi, [rip + outbuf]
    add rdi, qword ptr [rip + outpos]
    mov byte ptr [rdi], 10
    inc qword ptr [rip + outpos]
    ret

# rax = signed value; appends decimal and nothing else
print_num:
    lea rdi, [rip + numbuf + 32]
    xor r10d, r10d
    test rax, rax
    jns pn_pos
    neg rax
    mov r10d, 1
pn_pos:
    mov rcx, 10
pn_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz pn_loop
    test r10, r10
    jz pn_copy
    dec rdi
    mov byte ptr [rdi], 45
pn_copy:
    lea rax, [rip + numbuf + 32]
    sub rax, rdi
    mov rcx, rax
    mov rsi, rdi
    call emit
    ret

flush:
    lea rsi, [rip + outbuf]
    mov rdx, qword ptr [rip + outpos]
fl_loop:
    test rdx, rdx
    jz fl_done
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jle fl_done
    add rsi, rax
    sub rdx, rax
    jmp fl_loop
fl_done:
    mov qword ptr [rip + outpos], 0
    ret

all_done:
    call flush
    xor edi, edi
    mov eax, 60
    syscall
