.text
.globl _start
.equ msg_err_len, 9
_start:
    mov qword ptr [rip + inlen], 0
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, qword ptr [rip + inlen]
    mov edx, 131072
    sub rdx, qword ptr [rip + inlen]
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add qword ptr [rip + inlen], rax
    jmp read_loop
read_done:
    mov qword ptr [rip + g_pos], 0
    call skip_ws
    call parse_value
    call skip_ws
    mov rax, qword ptr [rip + g_pos]
    cmp rax, qword ptr [rip + inlen]
    jne fail
    lea rsi, [rip + msg_ok]
    mov edx, 3
    mov edi, 1
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

# ---------------------------------------------------------------
# fail: print "error at N\n" for N = g_pos, exit 1
# ---------------------------------------------------------------
fail:
    mov rax, qword ptr [rip + g_pos]
    lea r8, [rip + tmpbuf + 24]
    mov ecx, 10
fail_dig:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz fail_dig
    lea rdi, [rip + outbuf]
    lea rsi, [rip + msg_err]
    mov ecx, msg_err_len
    rep movsb
    mov rsi, r8
    lea rax, [rip + tmpbuf + 24]
    sub rax, r8
    mov rcx, rax
    rep movsb
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    call write_all
    mov eax, 60
    mov edi, 1
    syscall

# ---------------------------------------------------------------
# write_all: rsi = buf, rdx = count, rdi = fd
# ---------------------------------------------------------------
write_all:
    test rdx, rdx
    jz wa_done
    mov eax, 1
    syscall
    test rax, rax
    jle wa_done
    add rsi, rax
    sub rdx, rax
    jmp write_all
wa_done:
    ret

# ---------------------------------------------------------------
# peek: eax = byte at g_pos, or 256 at end of input
# ---------------------------------------------------------------
peek:
    mov rax, qword ptr [rip + g_pos]
    cmp rax, qword ptr [rip + inlen]
    jae peek_eof
    lea rdx, [rip + inbuf]
    movzx eax, byte ptr [rdx + rax]
    ret
peek_eof:
    mov eax, 256
    ret

# ---------------------------------------------------------------
# skip_ws
# ---------------------------------------------------------------
skip_ws:
    mov rax, qword ptr [rip + g_pos]
sw_loop:
    cmp rax, qword ptr [rip + inlen]
    jae sw_done
    lea rdx, [rip + inbuf]
    movzx ecx, byte ptr [rdx + rax]
    cmp ecx, 0x20
    je sw_next
    cmp ecx, 0x09
    je sw_next
    cmp ecx, 0x0A
    je sw_next
    cmp ecx, 0x0D
    jne sw_done
sw_next:
    inc rax
    jmp sw_loop
sw_done:
    mov qword ptr [rip + g_pos], rax
    ret

# ---------------------------------------------------------------
# parse_value
# ---------------------------------------------------------------
parse_value:
    call peek
    cmp eax, 0x7B
    je parse_object
    cmp eax, 0x5B
    je parse_array
    cmp eax, 0x22
    je parse_string
    cmp eax, 0x2D
    je parse_number
    cmp eax, 0x30
    jb pv_word
    cmp eax, 0x39
    jbe parse_number
pv_word:
    cmp eax, 0x74
    je parse_true
    cmp eax, 0x66
    je parse_false
    cmp eax, 0x6E
    je parse_null
    jmp fail

parse_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp expect_lit
parse_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp expect_lit
parse_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4
    jmp expect_lit

# expect_lit: rsi = literal, ecx = length; consumes matching bytes
expect_lit:
    xor edx, edx
el_loop:
    cmp edx, ecx
    jae el_done
    mov rax, qword ptr [rip + g_pos]
    cmp rax, qword ptr [rip + inlen]
    jae fail
    lea r8, [rip + inbuf]
    movzx r9d, byte ptr [r8 + rax]
    cmp r9b, byte ptr [rsi + rdx]
    jne fail
    add qword ptr [rip + g_pos], 1
    inc edx
    jmp el_loop
el_done:
    ret

# ---------------------------------------------------------------
# parse_string: g_pos at opening quote
# ---------------------------------------------------------------
parse_string:
    add qword ptr [rip + g_pos], 1
ps_loop:
    call peek
    cmp eax, 256
    je fail
    cmp eax, 0x22
    je ps_close
    cmp eax, 0x5C
    je ps_esc
    cmp eax, 0x20
    jb fail
    add qword ptr [rip + g_pos], 1
    jmp ps_loop
ps_close:
    add qword ptr [rip + g_pos], 1
    ret
ps_esc:
    add qword ptr [rip + g_pos], 1
    call peek
    cmp eax, 0x22
    je ps_simple
    cmp eax, 0x5C
    je ps_simple
    cmp eax, 0x2F
    je ps_simple
    cmp eax, 0x62
    je ps_simple
    cmp eax, 0x66
    je ps_simple
    cmp eax, 0x6E
    je ps_simple
    cmp eax, 0x72
    je ps_simple
    cmp eax, 0x74
    je ps_simple
    cmp eax, 0x75
    jne fail
    add qword ptr [rip + g_pos], 1
    mov ecx, 4
ps_hex:
    call peek
    cmp eax, 0x30
    jb fail
    cmp eax, 0x39
    jbe ps_hexok
    cmp eax, 0x41
    jb fail
    cmp eax, 0x46
    jbe ps_hexok
    cmp eax, 0x61
    jb fail
    cmp eax, 0x66
    ja fail
ps_hexok:
    add qword ptr [rip + g_pos], 1
    dec ecx
    jnz ps_hex
    jmp ps_loop
ps_simple:
    add qword ptr [rip + g_pos], 1
    jmp ps_loop

# ---------------------------------------------------------------
# parse_number
# ---------------------------------------------------------------
parse_number:
    call peek
    cmp eax, 0x2D
    jne pn_start
    add qword ptr [rip + g_pos], 1
pn_start:
    call peek
    cmp eax, 0x30
    je pn_zero
    cmp eax, 0x31
    jb fail
    cmp eax, 0x39
    ja fail
    add qword ptr [rip + g_pos], 1
pn_int_loop:
    call peek
    cmp eax, 0x30
    jb pn_frac
    cmp eax, 0x39
    ja pn_frac
    add qword ptr [rip + g_pos], 1
    jmp pn_int_loop
pn_zero:
    add qword ptr [rip + g_pos], 1
pn_frac:
    call peek
    cmp eax, 0x2E
    jne pn_exp
    add qword ptr [rip + g_pos], 1
    call peek
    cmp eax, 0x30
    jb fail
    cmp eax, 0x39
    ja fail
pn_frac_loop:
    add qword ptr [rip + g_pos], 1
    call peek
    cmp eax, 0x30
    jb pn_exp
    cmp eax, 0x39
    jbe pn_frac_loop
pn_exp:
    call peek
    cmp eax, 0x65
    je pn_e
    cmp eax, 0x45
    jne pn_end
pn_e:
    add qword ptr [rip + g_pos], 1
    call peek
    cmp eax, 0x2B
    je pn_esign
    cmp eax, 0x2D
    jne pn_expdig
pn_esign:
    add qword ptr [rip + g_pos], 1
    call peek
pn_expdig:
    cmp eax, 0x30
    jb fail
    cmp eax, 0x39
    ja fail
pn_exp_loop:
    add qword ptr [rip + g_pos], 1
    call peek
    cmp eax, 0x30
    jb pn_end
    cmp eax, 0x39
    jbe pn_exp_loop
pn_end:
    ret

# ---------------------------------------------------------------
# parse_object: g_pos at '{'
# ---------------------------------------------------------------
parse_object:
    add qword ptr [rip + g_pos], 1
    call skip_ws
    call peek
    cmp eax, 0x7D
    je po_close_empty
    cmp eax, 0x22
    jne fail
po_member:
    call parse_string
    call skip_ws
    call peek
    cmp eax, 0x3A
    jne fail
    add qword ptr [rip + g_pos], 1
    call skip_ws
    call parse_value
    call skip_ws
    call peek
    cmp eax, 0x2C
    je po_comma
    cmp eax, 0x7D
    jne fail
    add qword ptr [rip + g_pos], 1
    ret
po_comma:
    add qword ptr [rip + g_pos], 1
    call skip_ws
    call peek
    cmp eax, 0x22
    jne fail
    jmp po_member
po_close_empty:
    add qword ptr [rip + g_pos], 1
    ret

# ---------------------------------------------------------------
# parse_array: g_pos at '['
# ---------------------------------------------------------------
parse_array:
    add qword ptr [rip + g_pos], 1
    call skip_ws
    call peek
    cmp eax, 0x5D
    je pa_empty
pa_elem:
    call parse_value
    call skip_ws
    call peek
    cmp eax, 0x2C
    je pa_comma
    cmp eax, 0x5D
    jne fail
    add qword ptr [rip + g_pos], 1
    ret
pa_comma:
    add qword ptr [rip + g_pos], 1
    call skip_ws
    jmp pa_elem
pa_empty:
    add qword ptr [rip + g_pos], 1
    ret

# ---------------------------------------------------------------
# read-only data
# ---------------------------------------------------------------
.section .rodata
msg_err:
    .ascii "error at "
msg_ok:
    .ascii "ok\n"
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

# ---------------------------------------------------------------
# zero-initialised working storage
# ---------------------------------------------------------------
.section .bss
.align 16
g_pos:
    .zero 8
inlen:
    .zero 8
tmpbuf:
    .zero 32
outbuf:
    .zero 64
inbuf:
    .zero 131072
