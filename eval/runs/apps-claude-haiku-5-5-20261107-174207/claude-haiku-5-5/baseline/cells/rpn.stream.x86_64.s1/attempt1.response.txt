.globl _start
.text
_start:
    xor r13d, r13d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov rdx, 131072
    sub rdx, r13
    jz read_done
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r8, [rip + inbuf]
    lea r9, [rip + vstack]
    xor r12d, r12d
    xor r14d, r14d
line_loop:
    cmp r12, r13
    jae finish
    cmp r14, 4000
    jbe 1f
    call flush
1:
    mov r15, r12
find_nl:
    cmp r15, r13
    jae line_found
    cmp byte ptr [r8 + r15], 10
    je line_found
    inc r15
    jmp find_nl
line_found:
    mov r10, r12
    xor ebx, ebx
    xor ebp, ebp
tok_loop:
    cmp r10, r15
    jae line_done
    movzx eax, byte ptr [r8 + r10]
    cmp eax, 32
    je skip_sep
    cmp eax, 9
    je skip_sep
    mov r11, r10
tok_scan:
    inc r10
    cmp r10, r15
    jae tok_end
    movzx eax, byte ptr [r8 + r10]
    cmp eax, 32
    je tok_end
    cmp eax, 9
    jne tok_scan
tok_end:
    mov rcx, r10
    sub rcx, r11
    cmp rcx, 1
    jne literal
    movzx eax, byte ptr [r8 + r11]
    cmp eax, 43
    je operator
    cmp eax, 45
    je operator
    cmp eax, 42
    je operator
    cmp eax, 47
    je operator
    cmp eax, 37
    je operator
    jmp literal
literal:
    xor esi, esi
    movzx eax, byte ptr [r8 + r11]
    cmp eax, 45
    jne lit_nodash
    mov esi, 1
    inc r11
lit_nodash:
    mov rcx, r10
    sub rcx, r11
    test rcx, rcx
    jz bad
    cmp rcx, 18
    ja bad
    xor eax, eax
    mov rdx, r11
dig_loop:
    cmp rdx, r10
    jae dig_done
    movzx edi, byte ptr [r8 + rdx]
    sub edi, 48
    cmp edi, 9
    ja bad
    imul rax, rax, 10
    add rax, rdi
    inc rdx
    jmp dig_loop
dig_done:
    test esi, esi
    jz lit_push
    neg rax
lit_push:
    mov [r9 + rbx*8], rax
    inc rbx
    jmp tok_loop
operator:
    cmp rbx, 2
    jb bad
    mov rax, [r9 + rbx*8 - 16]
    mov rcx, [r9 + rbx*8 - 8]
    sub rbx, 2
    movzx edx, byte ptr [r8 + r11]
    cmp edx, 43
    je op_add
    cmp edx, 45
    je op_sub
    cmp edx, 42
    je op_mul
    cmp edx, 47
    je op_div
    jmp op_mod
op_add:
    add rax, rcx
    jmp op_push
op_sub:
    sub rax, rcx
    jmp op_push
op_mul:
    imul rax, rcx
    jmp op_push
op_div:
    test rcx, rcx
    jz bad
    cmp rcx, -1
    jne op_div_go
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    je bad
op_div_go:
    cqo
    idiv rcx
    jmp op_push
op_mod:
    test rcx, rcx
    jz bad
    cmp rcx, -1
    jne op_mod_go
    movabs rdx, 0x8000000000000000
    cmp rax, rdx
    je bad
op_mod_go:
    cqo
    idiv rcx
    mov rax, rdx
op_push:
    mov [r9 + rbx*8], rax
    inc rbx
    jmp tok_loop
skip_sep:
    inc r10
    jmp tok_loop
bad:
    mov ebp, 1
line_done:
    test ebp, ebp
    jnz do_err
    cmp rbx, 1
    jne do_err
    mov rax, [r9]
    call emit_int
    jmp next_line
do_err:
    call emit_err
next_line:
    lea r12, [r15 + 1]
    jmp line_loop
finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

flush:
    lea rsi, [rip + outbuf]
    mov rdx, r14
fl_loop:
    test rdx, rdx
    jz fl_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle fl_fail
    add rsi, rax
    sub rdx, rax
    jmp fl_loop
fl_fail:
    mov eax, 60
    xor edi, edi
    syscall
fl_done:
    xor r14d, r14d
    ret

emit_int:
    lea rdi, [rip + outbuf]
    add rdi, r14
    test rax, rax
    jns ei_pos
    mov byte ptr [rdi], 45
    inc rdi
    neg rax
ei_pos:
    lea rsi, [rip + numtmp + 24]
    mov rcx, 10
ei_loop:
    xor edx, edx
    div rcx
    add edx, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz ei_loop
    lea rcx, [rip + numtmp + 24]
    sub rcx, rsi
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    lea rax, [rip + outbuf]
    sub rdi, rax
    mov r14, rdi
    ret

emit_err:
    lea rsi, [rip + errstr]
    lea rdi, [rip + outbuf]
    add rdi, r14
    mov ecx, 6
    rep movsb
    lea rax, [rip + outbuf]
    sub rdi, rax
    mov r14, rdi
    ret

.section .rodata
errstr:
    .ascii "error\n"

.bss
inbuf:
    .zero 131072
vstack:
    .zero 8192
outbuf:
    .zero 4096

    .zero 32
