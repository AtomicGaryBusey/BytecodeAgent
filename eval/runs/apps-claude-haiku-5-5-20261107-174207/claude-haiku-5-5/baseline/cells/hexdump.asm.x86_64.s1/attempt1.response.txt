.text
.globl _start
_start:
  xor r12d, r12d
read_loop:
  mov rdx, 65536
  sub rdx, r12
  jz read_done
  lea rsi, [rip + inbuf]
  add rsi, r12
  xor edi, edi
  xor eax, eax
  syscall
  test rax, rax
  jle read_done
  add r12, rax
  jmp read_loop
read_done:
  xor r13d, r13d
row_loop:
  cmp r13, r12
  jae finish
  mov r14, r12
  sub r14, r13
  cmp r14, 16
  jbe 1f
  mov r14d, 16
1:
  lea rbp, [rip + inbuf]
  add rbp, r13
  lea rdi, [rip + linebuf]
  mov eax, r13d
  call hex32
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  add rdi, 2
  xor ebx, ebx
hex_loop:
  cmp ebx, r14d
  jae pad_slot
  movzx eax, byte ptr [rbp + rbx]
  call hex_byte
  mov byte ptr [rdi], 32
  inc rdi
  jmp slot_done
pad_slot:
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 32
  mov byte ptr [rdi + 2], 32
  add rdi, 3
slot_done:
  cmp ebx, 7
  jne no_extra
  mov byte ptr [rdi], 32
  inc rdi
no_extra:
  inc ebx
  cmp ebx, 16
  jb hex_loop
  mov byte ptr [rdi], 32
  mov byte ptr [rdi + 1], 124
  add rdi, 2
  xor ebx, ebx
char_loop:
  cmp ebx, r14d
  jae chars_done
  movzx eax, byte ptr [rbp + rbx]
  cmp eax, 32
  jb char_dot
  cmp eax, 126
  ja char_dot
  jmp char_put
char_dot:
  mov eax, 46
char_put:
  mov byte ptr [rdi], al
  inc rdi
  inc ebx
  jmp char_loop
chars_done:
  mov byte ptr [rdi], 124
  mov byte ptr [rdi + 1], 10
  add rdi, 2
  lea rsi, [rip + linebuf]
  mov rdx, rdi
  sub rdx, rsi
  call write_all
  add r13, 16
  jmp row_loop
finish:
  lea rdi, [rip + linebuf]
  mov eax, r12d
  call hex32
  mov byte ptr [rdi], 10
  inc rdi
  lea rsi, [rip + linebuf]
  mov rdx, rdi
  sub rdx, rsi
  call write_all
  xor edi, edi
  mov eax, 60
  syscall

# hex32: eax = 32-bit value, writes 8 hex digits at rdi, advances rdi
hex32:
  mov r8d, eax
  mov eax, r8d
  shr eax, 24
  call hex_byte
  mov eax, r8d
  shr eax, 16
  call hex_byte
  mov eax, r8d
  shr eax, 8
  call hex_byte
  mov eax, r8d
  call hex_byte
  ret

# hex_byte: low byte of eax as 2 hex digits at rdi, advances rdi by 2
hex_byte:
  movzx eax, al
  mov ecx, eax
  shr ecx, 4
  movzx edx, byte ptr [hexdig + rcx]
  mov byte ptr [rdi], dl
  and eax, 15
  movzx edx, byte ptr [hexdig + rax]
  mov byte ptr [rdi + 1], dl
  add rdi, 2
  ret

# write_all: write rdx bytes from rsi to fd 1, looping on partial writes
write_all:
  test rdx, rdx
  jz write_done
wloop:
  test rdx, rdx
  jz write_done
  mov eax, 1
  mov edi, 1
  syscall
  test rax, rax
  jle write_done
  add rsi, rax
  sub rdx, rax
  jmp wloop
write_done:
  ret

.section .rodata
hexdig:
  .ascii "0123456789abcdef"

.bss
inbuf:
  .zero 65536
linebuf:
  .zero 128
