```
.intel_syntax noprefix
.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    add r14, rax
    lea rbx, [rip + inbuf]
    lea rdi, [rbx + rax]
scan:
    cmp rbx, rdi
    je read_loop
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 10
    jne 2f
    inc r12
2:
    cmp al, 32
    je 3f
    mov edx, eax
    sub edx, 9
    cmp edx, 4
    ja 4f
3:
    xor r15d, r15d
    jmp scan
4:
    test r15d, r15d
    jnz scan
    mov r15d, 1
    inc r13
    jmp scan
done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call emit_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call emit_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call emit_num
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
write_loop:
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub rdx, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# emit_num: rax = value, rdi = output pointer; writes decimal digits, advances rdi
emit_num:
    mov rcx, 10
    xor r9d, r9d
5:
    xor edx, edx
    div rcx
    add dl, 48
    push rdx
    inc r9
    test rax, rax
    jnz 5b
6:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec r9
    jnz 6b
    ret

.bss
inbuf:
    .zero 65536
outbuf:
    .zero 64
```
