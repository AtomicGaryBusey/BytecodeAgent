.text
.globl _start

_start:
    adrp x20, inbuf
    add  x20, x20, :lo12:inbuf
    mov  x21, #0

read_loop:
    mov  x8, #63
    mov  x0, #0
    add  x1, x20, x21
    movz x2, #1, lsl #16
    sub  x2, x2, x21
    svc  #0
    cbz  x0, read_done
    add  x21, x21, x0
    b    read_loop

read_done:
    mov  x22, #0
    mov  x23, #0
    mov  x24, #0

scan_loop:
    cmp  x23, x21
    b.hs scan_end
    ldrb w9, [x20, x23]
    add  x23, x23, #1
    cmp  w9, #10
    b.ne scan_loop
    sub  x25, x23, #1
    sub  x25, x25, x24
    adrp x10, starts
    add  x10, x10, :lo12:starts
    str  x24, [x10, x22, lsl #3]
    adrp x11, lens
    add  x11, x11, :lo12:lens
    str  x25, [x11, x22, lsl #3]
    add  x22, x22, #1
    mov  x24, x23
    b    scan_loop

scan_end:
    cmp  x24, x21
    b.hs parse_done
    sub  x25, x21, x24
    adrp x10, starts
    add  x10, x10, :lo12:starts
    str  x24, [x10, x22, lsl #3]
    adrp x11, lens
    add  x11, x11, :lo12:lens
    str  x25, [x11, x22, lsl #3]
    add  x22, x22, #1

parse_done:
    adrp x26, arrA
    add  x26, x26, :lo12:arrA
    adrp x27, arrB
    add  x27, x27, :lo12:arrB
    mov  x23, #0

init_loop:
    cmp  x23, x22
    b.hs init_done
    str  x23, [x26, x23, lsl #3]
    add  x23, x23, #1
    b    init_loop

init_done:
    mov  x28, #1

pass_loop:
    cmp  x28, x22
    b.hs sort_done
    mov  x19, #0

block_loop:
    cmp  x19, x22
    b.hs block_done
    add  x20, x19, x28
    cmp  x20, x22
    csel x20, x22, x20, hi
    add  x21, x19, x28, lsl #1
    cmp  x21, x22
    csel x21, x22, x21, hi
    mov  x23, x19
    mov  x24, x20
    mov  x25, x19

merge_loop:
    cmp  x23, x20
    b.hs take_right_rest
    cmp  x24, x21
    b.hs take_left_rest
    ldr  x9, [x26, x23, lsl #3]
    ldr  x10, [x26, x24, lsl #3]
    adrp x11, starts
    add  x11, x11, :lo12:starts
    adrp x12, lens
    add  x12, x12, :lo12:lens
    adrp x13, inbuf
    add  x13, x13, :lo12:inbuf
    ldr  x14, [x11, x9, lsl #3]
    add  x0, x13, x14
    ldr  x1, [x12, x9, lsl #3]
    ldr  x14, [x11, x10, lsl #3]
    add  x2, x13, x14
    ldr  x3, [x12, x10, lsl #3]
    bl   cmp_gt
    cbnz x0, take_b
    ldr  x9, [x26, x23, lsl #3]
    str  x9, [x27, x25, lsl #3]
    add  x23, x23, #1
    b    merge_next

take_b:
    ldr  x9, [x26, x24, lsl #3]
    str  x9, [x27, x25, lsl #3]
    add  x24, x24, #1

merge_next:
    add  x25, x25, #1
    b    merge_loop

take_right_rest:
    cmp  x24, x21
    b.hs merge_done
    ldr  x9, [x26, x24, lsl #3]
    str  x9, [x27, x25, lsl #3]
    add  x24, x24, #1
    add  x25, x25, #1
    b    take_right_rest

take_left_rest:
    cmp  x23, x20
    b.hs merge_done
    ldr  x9, [x26, x23, lsl #3]
    str  x9, [x27, x25, lsl #3]
    add  x23, x23, #1
    add  x25, x25, #1
    b    take_left_rest

merge_done:
    add  x19, x19, x28, lsl #1
    b    block_loop

block_done:
    mov  x9, x26
    mov  x26, x27
    mov  x27, x9
    lsl  x28, x28, #1
    b    pass_loop

sort_done:
    mov  x23, #0
    mov  x24, #0

out_loop:
    cmp  x23, x22
    b.hs out_done
    ldr  x9, [x26, x23, lsl #3]
    adrp x10, starts
    add  x10, x10, :lo12:starts
    ldr  x11, [x10, x9, lsl #3]
    adrp x12, lens
    add  x12, x12, :lo12:lens
    ldr  x13, [x12, x9, lsl #3]
    adrp x14, inbuf
    add  x14, x14, :lo12:inbuf
    add  x11, x14, x11
    adrp x15, outbuf
    add  x15, x15, :lo12:outbuf
    add  x15, x15, x24

copy_loop:
    cbz  x13, copy_done
    ldrb w9, [x11], #1
    strb w9, [x15], #1
    sub  x13, x13, #1
    b    copy_loop

copy_done:
    mov  w9, #10
    strb w9, [x15], #1
    adrp x16, outbuf
    add  x16, x16, :lo12:outbuf
    sub  x24, x15, x16
    add  x23, x23, #1
    b    out_loop

out_done:
    mov  x19, #0

w_loop:
    cmp  x19, x24
    b.hs w_done
    mov  x8, #64
    mov  x0, #1
    adrp x1, outbuf
    add  x1, x1, :lo12:outbuf
    add  x1, x1, x19
    sub  x2, x24, x19
    svc  #0
    cmp  x0, #0
    b.le w_done
    add  x19, x19, x0
    b    w_loop

w_done:
    mov  x8, #93
    mov  x0, #0
    svc  #0

// cmp_gt: x0=pa, x1=la, x2=pb, x3=lb; returns x0=1 if a>b (memcmp order), else 0.
// Uses only x0-x6.
cmp_gt:
    cmp  x1, x3
    csel x4, x1, x3, lo

cg_loop:
    cbz  x4, cg_prefix
    ldrb w5, [x0], #1
    ldrb w6, [x2], #1
    cmp  w5, w6
    b.ne cg_diff
    sub  x4, x4, #1
    b    cg_loop

cg_diff:
    cset x0, hi
    ret

cg_prefix:
    cmp  x1, x3
    cset x0, hi
    ret
