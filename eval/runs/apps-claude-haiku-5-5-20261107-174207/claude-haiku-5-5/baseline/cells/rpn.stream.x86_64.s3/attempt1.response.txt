.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r8d, r8d
    lea rbx, [rip + valstack]
    lea rbp, [rip + inbuf]

read_loop:
    cmp r13, 65536
    jae read_done
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r13
    mov rdx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:

main_loop:
    cmp r12, r13
    jae finish
    mov r14, r12
find_nl:
    cmp r14, r13
    jae found_nl
    cmp byte ptr [rbp + r14], 10
    je found_nl
    inc r14
    jmp find_nl
found_nl:
    xor r15d, r15d

tok_loop:
skip:
    cmp r12, r14
    jae line_end
    movzx eax, byte ptr [rbp + r12]
    cmp eax, 32
    je skip_adv
    cmp eax, 9
    jne tok_start
skip_adv:
    inc r12
    jmp skip

tok_start:
    mov rsi, r12
    mov rdi, r12
tok_scan:
    cmp rdi, r14
    jae tok_end
    movzx eax, byte ptr [rbp + rdi]
    cmp eax, 32
    je tok_end
    cmp eax, 9
    je tok_end
    inc rdi
    jmp tok_scan
tok_end:
    mov r12, rdi
    mov rdx, rdi
    sub rdx, rsi
    cmp rdx, 1
    jne try_literal
    movzx eax, byte ptr [rbp + rsi]
    cmp eax, 43
    je is_op
    cmp eax, 45
    je is_op
    cmp eax, 42
    je is_op
    cmp eax, 47
    je is_op
    cmp eax, 37
    je is_op
    jmp try_literal

is_op:
    mov ecx, eax
    cmp r15, 2
    jb line_error
    dec r15
    mov rsi, qword ptr [rbx + r15*8]
    mov rax, qword ptr [rbx + r15*8 - 8]
    cmp ecx, 43
    je op_add
    cmp ecx, 45
    je op_sub
    cmp ecx, 42
    je op_mul
    jmp divmod
op_add:
    add rax, rsi
    jmp op_store
op_sub:
    sub rax, rsi
    jmp op_store
op_mul:
    imul rax, rsi
op_store:
    mov qword ptr [rbx + r15*8 - 8], rax
    jmp tok_loop
divmod:
    test rsi, rsi
    jz line_error
    cmp rsi, -1
    jne dm_go
    movabs rdi, 0x8000000000000000
    cmp rax, rdi
    je line_error
dm_go:
    cqo
    idiv rsi
    cmp ecx, 47
    je op_store
    mov rax, rdx
    jmp op_store

try_literal:
    mov r9, rsi
    xor ecx, ecx
    movzx eax, byte ptr [rbp + r9]
    cmp eax, 45
    jne tl_digits
    mov ecx, 1
    inc r9
tl_digits:
    mov r10, rdi
    sub r10, r9
    jz line_error
    cmp r10, 18
    ja line_error
    xor eax, eax
tl_loop:
    cmp r9, rdi
    jae tl_done
    movzx edx, byte ptr [rbp + r9]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r9
    jmp tl_loop
tl_done:
    test ecx, ecx
    jz tl_push
    neg rax
tl_push:
    mov qword ptr [rbx + r15*8], rax
    inc r15
    jmp tok_loop

line_error:
    mov r12, r14
    lea rsi, [rip + err_str]
    mov ecx, 6
    call emit_mem
    jmp line_finish

line_end:
    cmp r15, 1
    jne line_error
    mov rax, qword ptr [rbx]
    call emit_int
    mov al, 10
    call emit_al
line_finish:
    lea r12, [r14 + 1]
    jmp main_loop

# emit al to outbuf[r8], r8++ ; clobbers rdx only
emit_al:
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + r8], al
    inc r8
    ret

# emit ecx bytes from rsi
emit_mem:
    test rcx, rcx
    jz em_done
    movzx eax, byte ptr [rsi]
    call emit_al
    inc rsi
    dec rcx
    jmp emit_mem
em_done:
    ret

# emit signed 64-bit rax in decimal
emit_int:
    test rax, rax
    jns ei_pos
    push rax
    mov al, 45
    call emit_al
    pop rax
    neg rax
ei_pos:
    sub rsp, 32
    lea rdi, [rsp + 32]
    mov rcx, 10
ei_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz ei_loop
ei_out:
    lea rdx, [rsp + 32]
    cmp rdi, rdx
    jae ei_ret
    movzx eax, byte ptr [rdi]
    call emit_al
    inc rdi
    jmp ei_out
ei_ret:
    add rsp, 32
    ret

finish:
    xor r9d, r9d
wr_loop:
    cmp r9, r8
    jae do_exit
    mov rax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r9
    mov rdx, r8
    sub rdx, r9
    syscall
    test rax, rax
    jle do_exit
    add r9, rax
    jmp wr_loop
do_exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
err_str:
    .ascii "error\n"

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 400000
.balign 16
valstack:
    .zero 262144
