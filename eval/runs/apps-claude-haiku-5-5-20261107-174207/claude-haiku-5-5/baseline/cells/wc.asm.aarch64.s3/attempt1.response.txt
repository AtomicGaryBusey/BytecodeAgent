.section .bss
.align 4
buf:
    .zero 65536
tmp:
    .zero 32
out:
    .zero 64

.text
.globl _start
_start:
    mov x19, #0             // lines
    mov x20, #0             // words
    mov x21, #0             // bytes
    mov x22, #0             // in-word flag

read_loop:
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #65536
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq done_read
    add x21, x21, x0
    adrp x1, buf
    add x1, x1, :lo12:buf
    add x25, x1, x0

byte_loop:
    cmp x1, x25
    b.hs read_loop
    ldrb w3, [x1], #1
    cmp w3, #10
    b.ne 2f
    add x19, x19, #1
2:
    cmp w3, #32
    b.eq 3f
    sub w4, w3, #9
    cmp w4, #4
    b.ls 3f
    // non-whitespace byte
    cbnz x22, byte_loop
    mov x22, #1
    add x20, x20, #1
    b byte_loop
3:
    // whitespace byte
    mov x22, #0
    b byte_loop

done_read:
    adrp x26, out
    add x26, x26, :lo12:out

    mov x0, x19
    bl put_num
    mov w14, #32
    strb w14, [x26], #1

    mov x0, x20
    bl put_num
    mov w14, #32
    strb w14, [x26], #1

    mov x0, x21
    bl put_num
    mov w14, #10
    strb w14, [x26], #1

    adrp x1, out
    add x1, x1, :lo12:out
    sub x2, x26, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

// put_num: x0 = value, x26 = destination pointer (advanced on return)
put_num:
    adrp x10, tmp
    add x10, x10, :lo12:tmp
    add x10, x10, #32
    mov x11, #10
    mov x12, x10
1:
    udiv x13, x0, x11
    msub x14, x13, x11, x0
    add w14, w14, #48
    sub x12, x12, #1
    strb w14, [x12]
    mov x0, x13
    cbnz x0, 1b
2:
    cmp x12, x10
    b.hs 3f
    ldrb w14, [x12], #1
    strb w14, [x26], #1
    b 2b
3:
    ret
