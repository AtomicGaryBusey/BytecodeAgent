```
.intel_syntax noprefix
.text
.globl _start

_start:
    xor ebx, ebx
    xor ebp, ebp
    xor r12d, r12d
    xor r13d, r13d

main_read:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jle done
    mov r14, rax
    add r12, rax
    lea rsi, [rip + inbuf]
    xor ecx, ecx

proc_loop:
    cmp rcx, r14
    jae main_read
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp al, 10
    jne 3f
    inc rbx
3:
    cmp al, 32
    je ws_byte
    sub al, 9
    cmp al, 4
    jbe ws_byte
    test r13, r13
    jnz proc_loop
    mov r13, 1
    inc rbp
    jmp proc_loop

ws_byte:
    xor r13d, r13d
    jmp proc_loop

done:
    lea r15, [rip + outbuf]

    mov rax, rbx
    call emit_num
    mov byte ptr [r15], 32
    inc r15

    mov rax, rbp
    call emit_num
    mov byte ptr [r15], 32
    inc r15

    mov rax, r12
    call emit_num
    mov byte ptr [r15], 10
    inc r15

    lea rsi, [rip + outbuf]
write_loop:
    mov rdx, r15
    sub rdx, rsi
    jbe exit_prog
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_prog
    add rsi, rax
    jmp write_loop

exit_prog:
    mov eax, 60
    xor edi, edi
    syscall

# in: rax = value, r15 = output pointer; out: digits appended, r15 advanced
emit_num:
    lea rsi, [rip + numtmp + 24]
    mov ecx, 10
4:
    xor edx, edx
    div rcx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz 4b
    lea rdi, [rip + numtmp + 24]
5:
    cmp rsi, rdi
    jae 6f
    mov dl, byte ptr [rsi]
    mov byte ptr [r15], dl
    inc rsi
    inc r15
    jmp 5b
6:
    ret

.bss
.align 16
inbuf:
    .zero 4096
outbuf:
    .zero 128
numtmp:
    .zero 32
```
