.text
.globl _start

_start:
    // read all of stdin into inbuf
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0
read_loop:
    mov  x0, #0
    add  x1, x19, x20
    movz x2, #1, lsl #16
    sub  x2, x2, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop
read_done:

    // parse lines: x21 = lstart base, x22 = llen base, x23 = count
    adrp x21, lstart
    add  x21, x21, :lo12:lstart
    adrp x22, llen
    add  x22, x22, :lo12:llen
    mov  x23, #0
    mov  x24, #0
    mov  x25, #0
parse_loop:
    cmp  x25, x20
    b.ge parse_end
    add  x0, x19, x25
    ldrb w1, [x0]
    cmp  w1, #10
    b.ne parse_next
    sub  x2, x25, x24
    str  w24, [x21, x23, lsl #2]
    str  w2, [x22, x23, lsl #2]
    add  x23, x23, #1
    add  x24, x25, #1
parse_next:
    add  x25, x25, #1
    b    parse_loop
parse_end:
    cmp  x24, x20
    b.hs no_final
    sub  x2, x20, x24
    str  w24, [x21, x23, lsl #2]
    str  w2, [x22, x23, lsl #2]
    add  x23, x23, #1
no_final:

    // initialise idxA (x24 = src) and idxB (x25 = dst)
    adrp x24, idxA
    add  x24, x24, :lo12:idxA
    adrp x25, idxB
    add  x25, x25, :lo12:idxB
    mov  x28, #0
init_loop:
    cmp  x28, x23
    b.hs init_done
    str  w28, [x24, x28, lsl #2]
    add  x28, x28, #1
    b    init_loop
init_done:

    // bottom-up merge sort: x26 = width
    mov  x26, #1
pass_loop:
    cmp  x26, x23
    b.hs pass_done
    mov  x27, #0
lo_loop:
    cmp  x27, x23
    b.hs lo_done
    // mid = min(lo + width, count)
    add  x19, x27, x26
    cmp  x19, x23
    csel x19, x19, x23, ls
    // hi = min(mid + width, count)
    add  x20, x19, x26
    cmp  x20, x23
    csel x20, x20, x23, ls
    // i = lo, j = mid
    mov  x21, x27
    mov  x22, x19
merge_loop:
    cmp  x21, x19
    b.hs copy_right
    cmp  x22, x20
    b.hs copy_left
    ldr  w0, [x24, x21, lsl #2]
    ldr  w1, [x24, x22, lsl #2]
    bl   cmp_gt
    cbnz x0, pick_right
    // take left
    ldr  w0, [x24, x21, lsl #2]
    add  x28, x21, x22
    sub  x28, x28, x19
    str  w0, [x25, x28, lsl #2]
    add  x21, x21, #1
    b    merge_loop
pick_right:
    ldr  w0, [x24, x22, lsl #2]
    add  x28, x21, x22
    sub  x28, x28, x19
    str  w0, [x25, x28, lsl #2]
    add  x22, x22, #1
    b    merge_loop
copy_left:
    cmp  x21, x19
    b.hs copy_right
    ldr  w0, [x24, x21, lsl #2]
    add  x28, x21, x22
    sub  x28, x28, x19
    str  w0, [x25, x28, lsl #2]
    add  x21, x21, #1
    b    copy_left
copy_right:
    cmp  x22, x20
    b.hs merge_done
    ldr  w0, [x24, x22, lsl #2]
    add  x28, x21, x22
    sub  x28, x28, x19
    str  w0, [x25, x28, lsl #2]
    add  x22, x22, #1
    b    copy_right
merge_done:
    add  x27, x27, x26, lsl #1
    b    lo_loop
lo_done:
    // swap src and dst
    mov  x9, x24
    mov  x24, x25
    mov  x25, x9
    lsl  x26, x26, #1
    b    pass_loop
pass_done:

    // build output in outbuf: x19 = inbuf, x21 = lstart, x22 = llen,
    // x25 = outbuf, x20 = out offset, x28 = i, x24 = sorted indices
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    adrp x21, lstart
    add  x21, x21, :lo12:lstart
    adrp x22, llen
    add  x22, x22, :lo12:llen
    adrp x25, outbuf
    add  x25, x25, :lo12:outbuf
    mov  x20, #0
    mov  x28, #0
out_loop:
    cmp  x28, x23
    b.hs out_done
    ldr  w0, [x24, x28, lsl #2]
    ldr  w1, [x21, x0, lsl #2]
    ldr  w2, [x22, x0, lsl #2]
    add  x3, x19, x1
    add  x4, x25, x20
    mov  x5, #0
copy_line:
    cmp  x5, x2
    b.hs copy_nl
    ldrb w6, [x3, x5]
    strb w6, [x4, x5]
    add  x5, x5, #1
    b    copy_line
copy_nl:
    mov  w6, #10
    strb w6, [x4, x2]
    add  x20, x20, x2
    add  x20, x20, #1
    add  x28, x28, #1
    b    out_loop
out_done:

    // write outbuf to stdout
    mov  x27, #0
write_loop:
    cmp  x27, x20
    b.hs write_done
    mov  x0, #1
    add  x1, x25, x27
    sub  x2, x20, x27
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le write_done
    add  x27, x27, x0
    b    write_loop
write_done:
    mov  x0, #0
    mov  x8, #93
    svc  #0

// cmp_gt: x0 = index a, x1 = index b; returns x0 = 1 if line a > line b, else 0
// leaf; clobbers only x0-x11
cmp_gt:
    adrp x9, lstart
    add  x9, x9, :lo12:lstart
    adrp x10, llen
    add  x10, x10, :lo12:llen
    ldr  w2, [x9, x0, lsl #2]
    ldr  w3, [x10, x0, lsl #2]
    ldr  w4, [x9, x1, lsl #2]
    ldr  w5, [x10, x1, lsl #2]
    adrp x11, inbuf
    add  x11, x11, :lo12:inbuf
    add  x2, x11, x2
    add  x4, x11, x4
    cmp  x3, x5
    csel x6, x3, x5, ls
    mov  x7, #0
cg_loop:
    cmp  x7, x6
    b.hs cg_eq
    ldrb w9, [x2, x7]
    ldrb w10, [x4, x7]
    cmp  w9, w10
    b.ne cg_diff
    add  x7, x7, #1
    b    cg_loop
cg_diff:
    cset x0, hi
    ret
cg_eq:
    cmp  x3, x5
    cset x0, hi
    ret

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 65600
.balign 16
lstart:
    .zero 4096
.balign 16
llen:
    .zero 4096
.balign 16
idxA:
    .zero 4096
.balign 16
idxB:
    .zero 4096
