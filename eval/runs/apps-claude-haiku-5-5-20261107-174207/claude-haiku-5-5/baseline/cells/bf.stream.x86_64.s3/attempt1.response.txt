.equ INBUF_SIZE, 131072
.equ OUTSZ, 65536

.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor r13d, r13d
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + r13]
    mov rdx, INBUF_SIZE
    sub rdx, r13
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    mov r14, r13
    xor ecx, ecx
find_bang:
    cmp rcx, r14
    jae no_bang
    cmp byte ptr [r12 + rcx], 0x21
    je found_bang
    inc rcx
    jmp find_bang
no_bang:
    mov r13, r14
    mov rbx, r14
    mov rbp, r14
    jmp check_brackets
found_bang:
    mov r13, rcx
    lea rbx, [rcx + 1]
    mov rbp, r14
check_brackets:
    xor r8d, r8d
    xor r15d, r15d
    lea r9, [rip + bstack]
    lea r10, [rip + jt]
scan:
    cmp r15, r13
    jae scan_done
    movzx eax, byte ptr [r12 + r15]
    cmp al, 0x5b
    je push_b
    cmp al, 0x5d
    je pop_b
next_scan:
    inc r15
    jmp scan
push_b:
    mov dword ptr [r9 + r8*4], r15d
    inc r8
    jmp next_scan
pop_b:
    test r8, r8
    jz fail
    dec r8
    mov eax, dword ptr [r9 + r8*4]
    mov dword ptr [r10 + rax*4], r15d
    mov dword ptr [r10 + r15*4], eax
    jmp next_scan
scan_done:
    test r8, r8
    jnz fail

    xor r8d, r8d
    xor r15d, r15d
    lea r14, [rip + tape]
    lea r10, [rip + jt]
run:
    cmp r15, r13
    jae finish
    movzx eax, byte ptr [r12 + r15]
    cmp al, 0x3e
    je op_right
    cmp al, 0x3c
    je op_left
    cmp al, 0x2b
    je op_inc
    cmp al, 0x2d
    je op_dec
    cmp al, 0x2e
    je op_out
    cmp al, 0x2c
    je op_in
    cmp al, 0x5b
    je op_open
    cmp al, 0x5d
    je op_close
    jmp next
op_right:
    inc r14
    jmp next
op_left:
    dec r14
    jmp next
op_inc:
    inc byte ptr [r14]
    jmp next
op_dec:
    dec byte ptr [r14]
    jmp next
op_out:
    cmp r8, OUTSZ
    jb out_room
    call flush
out_room:
    lea rdx, [rip + outbuf]
    mov al, byte ptr [r14]
    mov byte ptr [rdx + r8], al
    inc r8
    jmp next
op_in:
    cmp rbx, rbp
    jae in_eof
    movzx eax, byte ptr [r12 + rbx]
    inc rbx
    mov byte ptr [r14], al
    jmp next
in_eof:
    mov byte ptr [r14], 0
    jmp next
op_open:
    cmp byte ptr [r14], 0
    jne next
    mov eax, dword ptr [r10 + r15*4]
    mov r15, rax
    jmp next
op_close:
    cmp byte ptr [r14], 0
    je next
    mov eax, dword ptr [r10 + r15*4]
    mov r15, rax
    jmp next
next:
    inc r15
    jmp run

finish:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

fail:
    mov eax, 60
    mov edi, 1
    syscall

flush:
    xor r9d, r9d
flush_loop:
    cmp r9, r8
    jae flush_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r9
    mov rdx, r8
    sub rdx, r9
    syscall
    test rax, rax
    jle flush_done
    add r9, rax
    jmp flush_loop
flush_done:
    xor r8d, r8d
    ret

.bss
.align 16
inbuf:
    .zero 131072
bstack:
    .zero 262144
jt:
    .zero 262144
tape:
    .zero 30000
outbuf:
    .zero 65536
