.text
.globl _start
_start:
    xor r12, r12
read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    jz read_done
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea rbx, [rip + inbuf]
    lea rbp, [rip + ents]
parse_loop:
    cmp r13, r12
    jae parse_end
    cmp byte ptr [rbx + r13], 10
    jne parse_next
    mov rdx, r15
    shl rdx, 4
    add rdx, rbp
    mov [rdx], r14
    mov rax, r13
    sub rax, r14
    mov [rdx + 8], rax
    inc r15
    lea r14, [r13 + 1]
parse_next:
    inc r13
    jmp parse_loop
parse_end:
    cmp r14, r12
    jae parse_done2
    mov rdx, r15
    shl rdx, 4
    add rdx, rbp
    mov [rdx], r14
    mov rax, r12
    sub rax, r14
    mov [rdx + 8], rax
    inc r15
parse_done2:
    # insertion sort
    mov r13, 1
outer:
    cmp r13, r15
    jae sort_done
    mov rdx, r13
    shl rdx, 4
    add rdx, rbp
    mov rbx, [rdx]
    mov r14, [rdx + 8]
    mov r12, r13
inner:
    test r12, r12
    jz place
    mov rdx, r12
    dec rdx
    shl rdx, 4
    add rdx, rbp
    mov r8, [rdx]
    mov r9, [rdx + 8]
    mov r10, rbx
    mov r11, r14
    call cmp_lines
    test rax, rax
    jle place
    mov rdx, r12
    shl rdx, 4
    add rdx, rbp
    mov rax, [rdx - 16]
    mov [rdx], rax
    mov rax, [rdx - 8]
    mov [rdx + 8], rax
    dec r12
    jmp inner
place:
    mov rdx, r12
    shl rdx, 4
    add rdx, rbp
    mov [rdx], rbx
    mov [rdx + 8], r14
    inc r13
    jmp outer
sort_done:
    # build output
    xor r12, r12
    xor r13, r13
out_loop:
    cmp r13, r15
    jae out_done
    mov rdx, r13
    shl rdx, 4
    add rdx, rbp
    mov r8, [rdx]
    mov r9, [rdx + 8]
    lea rsi, [rip + inbuf]
    add rsi, r8
    lea rdi, [rip + outbuf]
    add rdi, r12
    mov rcx, r9
    rep movsb
    add r12, r9
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r12], 10
    inc r12
    inc r13
    jmp out_loop
out_done:
    # write output
    xor r13, r13
wr_loop:
    cmp r13, r12
    jae wr_done
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r13
    mov rdx, r12
    sub rdx, r13
    syscall
    test rax, rax
    jle wr_done
    add r13, rax
    jmp wr_loop
wr_done:
    mov eax, 60
    xor edi, edi
    syscall

# compare lines: r8 = startA, r9 = lenA, r10 = startB, r11 = lenB
# returns rax < 0 if A < B, 0 if equal, > 0 if A > B
cmp_lines:
    lea rdi, [rip + inbuf]
    add r8, rdi
    add r10, rdi
    mov rcx, r9
    cmp rcx, r11
    cmovg rcx, r11
    xor edx, edx
cmp_loop:
    cmp rdx, rcx
    je cmp_tail
    movzx eax, byte ptr [r8 + rdx]
    movzx esi, byte ptr [r10 + rdx]
    cmp eax, esi
    jne cmp_diff
    inc rdx
    jmp cmp_loop
cmp_diff:
    sub rax, rsi
    ret
cmp_tail:
    mov rax, r9
    sub rax, r11
    ret

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65536
ents:
    .zero 16384
