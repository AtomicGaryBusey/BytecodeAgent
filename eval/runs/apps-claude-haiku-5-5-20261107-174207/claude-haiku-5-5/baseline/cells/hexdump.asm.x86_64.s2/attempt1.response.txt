.text
.globl _start
_start:
    lea r12, [rip + inbuf]
    xor r13, r13
read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14, r14
row_loop:
    cmp r14, r13
    jae final
    lea r9, [r12 + r14]
    mov rbx, r13
    sub rbx, r14
    cmp rbx, 16
    jbe n_ok
    mov ebx, 16
n_ok:
    lea r15, [rip + linebuf]
    mov edx, r14d
    call put_hex8
    mov byte ptr [r15], 32
    mov byte ptr [r15 + 1], 32
    add r15, 2
    xor r8d, r8d
hex_loop:
    cmp r8d, 16
    jae hex_done
    cmp r8, rbx
    jae pad_slot
    movzx edx, byte ptr [r9 + r8]
    call put_hex2
    mov byte ptr [r15], 32
    inc r15
    jmp after_slot
pad_slot:
    mov byte ptr [r15], 32
    mov byte ptr [r15 + 1], 32
    mov byte ptr [r15 + 2], 32
    add r15, 3
after_slot:
    cmp r8d, 7
    jne no_extra
    mov byte ptr [r15], 32
    inc r15
no_extra:
    inc r8d
    jmp hex_loop
hex_done:
    mov byte ptr [r15], 32
    mov byte ptr [r15 + 1], 124
    add r15, 2
    xor r8d, r8d
chr_loop:
    cmp r8, rbx
    jae chr_done
    movzx eax, byte ptr [r9 + r8]
    cmp eax, 32
    jb chr_dot
    cmp eax, 126
    ja chr_dot
    jmp chr_store
chr_dot:
    mov eax, 46
chr_store:
    mov byte ptr [r15], al
    inc r15
    inc r8
    jmp chr_loop
chr_done:
    mov byte ptr [r15], 124
    mov byte ptr [r15 + 1], 10
    add r15, 2
    lea rsi, [rip + linebuf]
    mov rdx, r15
    sub rdx, rsi
    call write_all
    add r14, 16
    jmp row_loop
final:
    lea r15, [rip + linebuf]
    mov edx, r13d
    call put_hex8
    mov byte ptr [r15], 10
    inc r15
    lea rsi, [rip + linebuf]
    mov rdx, r15
    sub rdx, rsi
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

# in: dl = byte; writes 2 hex digits at [r15], advances r15; clobbers eax, edx
put_hex2:
    mov al, dl
    shr al, 4
    call put_nib
    mov al, dl
    and al, 15
    call put_nib
    ret

# in: edx = value; writes 8 hex digits at [r15], advances r15; clobbers eax, ecx, edx
put_hex8:
    mov ecx, 8
put_hex8_loop:
    rol edx, 4
    mov eax, edx
    and eax, 15
    call put_nib
    dec ecx
    jnz put_hex8_loop
    ret

# in: al = 0..15; writes one lowercase hex digit at [r15], advances r15
put_nib:
    cmp al, 10
    jb put_nib_digit
    add al, 87
    jmp put_nib_store
put_nib_digit:
    add al, 48
put_nib_store:
    mov byte ptr [r15], al
    inc r15
    ret

# in: rsi = buffer, rdx = count; writes all bytes to fd 1
write_all:
write_loop:
    test rdx, rdx
    jz write_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jmp write_loop
write_done:
    ret

.bss
.align 16
inbuf:
    .zero 65536
linebuf:
    .zero 128
