.text
.globl _start
_start:
    xor r12d, r12d
.Lread:
    cmp r12, 65536
    jae .Lread_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    syscall
    test rax, rax
    jle .Lread_done
    add r12, rax
    jmp .Lread
.Lread_done:
    xor r13d, r13d
.Lrow:
    cmp r13, r12
    jae .Lfinal
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe .Lnok
    mov r14d, 16
.Lnok:
    lea rbx, [rip + inbuf]
    add rbx, r13
    lea rdi, [rip + line]
    mov rax, r13
    mov ecx, 8
    call emit_hex
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor r15d, r15d
.Lbyte:
    cmp r15, 16
    jae .Lbyte_end
    cmp r15, r14
    jae .Lbyte_pad
    movzx eax, byte ptr [rbx + r15]
    mov ecx, 2
    call emit_hex
    jmp .Lbyte_sp
.Lbyte_pad:
    mov word ptr [rdi], 0x2020
    add rdi, 2
.Lbyte_sp:
    mov byte ptr [rdi], 32
    inc rdi
    cmp r15, 7
    jne .Lbyte_next
    mov byte ptr [rdi], 32
    inc rdi
.Lbyte_next:
    inc r15
    jmp .Lbyte
.Lbyte_end:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2
    xor r15d, r15d
.Lchr:
    cmp r15, r14
    jae .Lchr_end
    movzx eax, byte ptr [rbx + r15]
    cmp eax, 32
    jb .Ldot
    cmp eax, 126
    ja .Ldot
    jmp .Lput
.Ldot:
    mov eax, 46
.Lput:
    mov byte ptr [rdi], al
    inc rdi
    inc r15
    jmp .Lchr
.Lchr_end:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    mov rdx, rdi
    lea rsi, [rip + line]
    sub rdx, rsi
    call write_all
    add r13, 16
    jmp .Lrow
.Lfinal:
    lea rdi, [rip + line]
    mov rax, r12
    mov ecx, 8
    call emit_hex
    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    lea rsi, [rip + line]
    sub rdx, rsi
    call write_all
.Lexit:
    mov eax, 60
    xor edi, edi
    syscall

# in: rax = value, ecx = number of hex digits, rdi = destination
# out: rdi advanced past the digits; clobbers rcx, rdx, r8, r9, r10
emit_hex:
    lea r8, [rip + hexd]
    mov r10d, ecx
    mov r9d, ecx
    shl r9d, 2
    sub r9d, 4
.Leh:
    mov rdx, rax
    mov ecx, r9d
    shr rdx, cl
    and edx, 15
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub r9d, 4
    dec r10d
    jnz .Leh
    ret

# in: rsi = buffer, rdx = length
# clobbers: rax, rcx, rdi, rsi, rdx, r11
write_all:
.Lw:
    test rdx, rdx
    jz .Lwd
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle .Lexit
    add rsi, rax
    sub rdx, rax
    jmp .Lw
.Lwd:
    ret

.section .rodata
hexd:
    .ascii "0123456789abcdef"

.bss
inbuf:
    .zero 65536
line:
    .zero 128
