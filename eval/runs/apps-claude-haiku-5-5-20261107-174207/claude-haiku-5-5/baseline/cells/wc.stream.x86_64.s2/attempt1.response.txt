```
.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall
    test rax, rax
    jle done_read
    mov rcx, rax
    add r14, rax
    lea rbx, [rip + inbuf]

process:
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 10
    jne check_space
    inc r12

check_space:
    cmp al, 32
    je is_ws
    mov edx, eax
    sub dl, 9
    cmp dl, 4
    jbe is_ws
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

is_ws:
    xor r15d, r15d

next_byte:
    dec rcx
    jnz process
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call conv
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call conv
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call conv
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_ok
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

# Convert unsigned rax to decimal at rdi; advances rdi.
conv:
    mov r9, rdi
    mov rcx, 10
    test rax, rax
    jnz cv_digits
    mov byte ptr [rdi], 48
    inc rdi
    ret

cv_digits:
cv_loop:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    test rax, rax
    jnz cv_loop

    mov rsi, rdi
    dec rsi
    mov r8, r9
cv_rev:
    cmp r8, rsi
    jae cv_done
    mov al, byte ptr [r8]
    mov dl, byte ptr [rsi]
    mov byte ptr [r8], dl
    mov byte ptr [rsi], al
    inc r8
    dec rsi
    jmp cv_rev
cv_done:
    ret

.bss
inbuf:
    .zero 65536
outbuf:
    .zero 64
```
