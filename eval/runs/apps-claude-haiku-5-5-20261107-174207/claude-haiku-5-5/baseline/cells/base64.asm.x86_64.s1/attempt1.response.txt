```
.intel_syntax noprefix
.text
.globl _start

_start:
    xor r14, r14
read_loop:
    mov edi, 0
    lea rsi, [rip + inbuf]
    add rsi, r14
    mov rdx, 49152
    sub rdx, r14
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r14, rax
    jmp read_loop
read_done:

    xor r15, r15
    xor r12, r12
    xor r13, r13
enc_loop:
    mov rax, r14
    sub rax, r15
    cmp rax, 3
    jb tail

    lea rsi, [rip + inbuf]
    movzx r8d, byte ptr [rsi + r15]
    shl r8d, 16
    movzx eax, byte ptr [rsi + r15 + 1]
    shl eax, 8
    or r8d, eax
    movzx eax, byte ptr [rsi + r15 + 2]
    or r8d, eax

    mov ebx, r8d
    shr ebx, 18
    call emit_idx
    mov ebx, r8d
    shr ebx, 12
    call emit_idx
    mov ebx, r8d
    shr ebx, 6
    call emit_idx
    mov ebx, r8d
    call emit_idx

    add r15, 3
    jmp enc_loop

tail:
    mov rcx, r14
    sub rcx, r15
    jz finish

    lea rsi, [rip + inbuf]
    movzx r8d, byte ptr [rsi + r15]
    shl r8d, 16
    cmp rcx, 2
    jne tail1

    movzx eax, byte ptr [rsi + r15 + 1]
    shl eax, 8
    or r8d, eax
    mov ebx, r8d
    shr ebx, 18
    call emit_idx
    mov ebx, r8d
    shr ebx, 12
    call emit_idx
    mov ebx, r8d
    shr ebx, 6
    call emit_idx
    mov al, 61
    call emit
    jmp finish

tail1:
    mov ebx, r8d
    shr ebx, 18
    call emit_idx
    mov ebx, r8d
    shr ebx, 12
    call emit_idx
    mov al, 61
    call emit
    mov al, 61
    call emit

finish:
    test r13, r13
    jz output
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + r12], 10
    inc r12

output:
    xor r15, r15
wloop:
    cmp r15, r12
    jae done
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r15
    mov rdx, r12
    sub rdx, r15
    mov eax, 1
    syscall
    test rax, rax
    jle done
    add r15, rax
    jmp wloop

done:
    xor edi, edi
    mov eax, 60
    syscall

# ebx = 6-bit index (masked here); emits alphabet character
emit_idx:
    and ebx, 63
    lea rdx, [rip + alphabet]
    movzx eax, byte ptr [rdx + rbx]
    call emit
    ret

# al = character to append; handles 76-column line wrapping
emit:
    lea rdx, [rip + outbuf]
    mov byte ptr [rdx + r12], al
    inc r12
    inc r13
    cmp r13, 76
    jne emit_ret
    mov byte ptr [rdx + r12], 10
    inc r12
    xor r13d, r13d
emit_ret:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero 49152
outbuf:
    .zero 70000
```
