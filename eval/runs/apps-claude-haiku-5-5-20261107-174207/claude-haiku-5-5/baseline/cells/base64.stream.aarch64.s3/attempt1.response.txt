.text
.globl _start
_start:
    adrp x20, inbuf
    add  x20, x20, :lo12:inbuf
    mov  x19, #0
read_loop:
    mov  x0, #0
    add  x1, x20, x19
    mov  x2, #49152
    sub  x2, x2, x19
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.eq read_done
    add  x19, x19, x0
    b    read_loop
read_done:
    adrp x26, alphabet
    add  x26, x26, :lo12:alphabet
    adrp x21, outbuf
    add  x21, x21, :lo12:outbuf
    mov  x22, #0
    mov  x23, #0
    mov  x27, #0
group_loop:
    sub  x0, x19, x27
    cmp  x0, #3
    b.ge full
    cbz  x0, fin
    cmp  x0, #1
    b.eq one
    b    two
full:
    add  x1, x20, x27
    ldrb w3, [x1]
    ldrb w4, [x1, #1]
    ldrb w5, [x1, #2]
    add  x27, x27, #3
    lsr  w0, w3, #2
    bl   emitidx
    and  w0, w3, #3
    lsl  w0, w0, #4
    orr  w0, w0, w4, lsr #4
    bl   emitidx
    and  w0, w4, #15
    lsl  w0, w0, #2
    orr  w0, w0, w5, lsr #6
    bl   emitidx
    and  w0, w5, #63
    bl   emitidx
    b    group_loop
one:
    add  x1, x20, x27
    ldrb w3, [x1]
    add  x27, x27, #1
    lsr  w0, w3, #2
    bl   emitidx
    and  w0, w3, #3
    lsl  w0, w0, #4
    bl   emitidx
    mov  w0, #61
    bl   emit
    mov  w0, #61
    bl   emit
    b    group_loop
two:
    add  x1, x20, x27
    ldrb w3, [x1]
    ldrb w4, [x1, #1]
    add  x27, x27, #2
    lsr  w0, w3, #2
    bl   emitidx
    and  w0, w3, #3
    lsl  w0, w0, #4
    orr  w0, w0, w4, lsr #4
    bl   emitidx
    and  w0, w4, #15
    lsl  w0, w0, #2
    bl   emitidx
    mov  w0, #61
    bl   emit
    b    group_loop
fin:
    cbz  x23, write_out
    mov  w0, #10
    strb w0, [x21, x22]
    add  x22, x22, #1
write_out:
    mov  x27, #0
wloop:
    cmp  x27, x22
    b.ge exit_ok
    mov  x0, #1
    add  x1, x21, x27
    sub  x2, x22, x27
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit_ok
    add  x27, x27, x0
    b    wloop
exit_ok:
    mov  x0, #0
    mov  x8, #93
    svc  #0

// emitidx: w0 = 6-bit index; emits alphabet character
emitidx:
    ldrb w0, [x26, x0]
    b    emit

// emit: w0 = character; appends to outbuf with a newline every 76 characters
emit:
    strb w0, [x21, x22]
    add  x22, x22, #1
    add  x23, x23, #1
    cmp  x23, #76
    b.ne 1f
    mov  w0, #10
    strb w0, [x21, x22]
    add  x22, x22, #1
    mov  x23, #0
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
inbuf:
    .space 49152
outbuf:
    .space 70000
