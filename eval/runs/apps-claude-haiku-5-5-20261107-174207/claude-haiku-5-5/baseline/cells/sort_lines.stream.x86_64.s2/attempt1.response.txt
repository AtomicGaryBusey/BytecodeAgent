```
.intel_syntax noprefix
.text
.globl _start

_start:
    mov qword ptr [rip + total], 0
read_loop:
    mov rax, 0
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, qword ptr [rip + total]
    mov rdx, 65536
    sub rdx, qword ptr [rip + total]
    syscall
    test rax, rax
    jle read_done
    add qword ptr [rip + total], rax
    jmp read_loop
read_done:

    xor ecx, ecx
    xor edx, edx
    xor ebx, ebx
    lea rdi, [rip + inbuf]
    mov r8, qword ptr [rip + total]
parse_loop:
    cmp rcx, r8
    jae parse_end
    cmp byte ptr [rdi + rcx], 10
    jne parse_next
    mov rax, rcx
    sub rax, rdx
    mov r9, rax
    shl r9, 32
    or r9, rdx
    lea r10, [rip + entries]
    mov qword ptr [r10 + rbx*8], r9
    inc rbx
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rdx, r8
    jae parse_done
    mov rax, r8
    sub rax, rdx
    mov r9, rax
    shl r9, 32
    or r9, rdx
    lea r10, [rip + entries]
    mov qword ptr [r10 + rbx*8], r9
    inc rbx
parse_done:

    # bottom-up merge sort: rbx = n, r12 = src, r13 = dst, r14 = width
    lea r12, [rip + entries]
    lea r13, [rip + temp]
    mov r14, 1
ms_outer:
    cmp r14, rbx
    jae ms_done
    xor r15d, r15d
ms_block:
    cmp r15, rbx
    jae ms_block_done
    lea rbp, [r15 + r14]
    cmp rbp, rbx
    cmova rbp, rbx
    lea r11, [r15 + r14*2]
    cmp r11, rbx
    cmova r11, rbx
    mov r8, r15
    mov r9, rbp
    mov r10, r15
merge_loop:
    cmp r10, r11
    jae merge_done
    cmp r8, rbp
    jae take_right
    cmp r9, r11
    jae take_left
    mov rdi, qword ptr [r12 + r9*8]
    mov rsi, qword ptr [r12 + r8*8]
    call less
    test al, al
    jnz take_right
take_left:
    mov rax, qword ptr [r12 + r8*8]
    mov qword ptr [r13 + r10*8], rax
    inc r8
    inc r10
    jmp merge_loop
take_right:
    mov rax, qword ptr [r12 + r9*8]
    mov qword ptr [r13 + r10*8], rax
    inc r9
    inc r10
    jmp merge_loop
merge_done:
    mov r15, r11
    jmp ms_block
ms_block_done:
    xchg r12, r13
    shl r14, 1
    jmp ms_outer
ms_done:

    # output
    xor r9d, r9d
    xor ecx, ecx
out_line:
    cmp rcx, rbx
    jae out_done
    mov rsi, qword ptr [r12 + rcx*8]
    mov rdx, rsi
    shr rdx, 32
    mov esi, esi
    lea rax, [rip + inbuf]
    add rsi, rax
    lea rdi, [rip + outbuf]
    add rdi, r9
    xor r10d, r10d
copy_loop:
    cmp r10, rdx
    jae copy_done
    movzx eax, byte ptr [rsi + r10]
    mov byte ptr [rdi + r10], al
    inc r10
    jmp copy_loop
copy_done:
    mov byte ptr [rdi + rdx], 10
    lea r9, [r9 + rdx + 1]
    inc rcx
    jmp out_line
out_done:
    xor r10d, r10d
write_loop:
    cmp r10, r9
    jae finish
    mov rax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    add rsi, r10
    mov rdx, r9
    sub rdx, r10
    syscall
    test rax, rax
    jle finish
    add r10, rax
    jmp write_loop
finish:
    mov eax, 60
    xor edi, edi
    syscall

# less: rdi = entry a, rsi = entry b; al = 1 if line a < line b
# preserves all registers except rax
less:
    push rcx
    push rdx
    push rsi
    push rdi
    push r8
    push r9
    push r10
    push r11
    mov ecx, edi
    shr rdi, 32
    mov edx, esi
    shr rsi, 32
    mov r8, rdi
    cmp rsi, rdi
    cmovb r8, rsi
    lea r9, [rip + inbuf]
    add rcx, r9
    add rdx, r9
    xor r10d, r10d
lt_prefix:
    cmp r10, r8
    jae lt_tail
    movzx eax, byte ptr [rcx + r10]
    movzx r11d, byte ptr [rdx + r10]
    cmp eax, r11d
    jne lt_diff
    inc r10
    jmp lt_prefix
lt_tail:
    cmp rdi, rsi
    setb al
    movzx eax, al
    jmp lt_done
lt_diff:
    cmp eax, r11d
    setb al
    movzx eax, al
lt_done:
    pop r11
    pop r10
    pop r9
    pop r8
    pop rdi
    pop rsi
    pop rdx
    pop rcx
    ret

.bss
.align 16
total:
    .zero 8
inbuf:
    .zero 65536
outbuf:
    .zero 65536
.align 16
entries:
    .zero 16384
temp:
    .zero 16384
```
