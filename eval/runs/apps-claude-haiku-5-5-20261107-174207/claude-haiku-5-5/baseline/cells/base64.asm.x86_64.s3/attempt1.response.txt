.globl _start
.text
_start:
    xor r12d, r12d
read_loop:
    lea rsi, [rip + inbuf]
    add rsi, r12
    mov rdx, 65536
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
enc_loop:
    cmp r13, r12
    jae enc_done
    mov r10, r12
    sub r10, r13
    lea rbx, [rip + inbuf]
    movzx eax, byte ptr [rbx + r13]
    xor r8d, r8d
    xor r9d, r9d
    cmp r10, 2
    jb no_b1
    movzx r8d, byte ptr [rbx + r13 + 1]
no_b1:
    cmp r10, 3
    jb no_b2
    movzx r9d, byte ptr [rbx + r13 + 2]
no_b2:
    mov ecx, eax
    shr ecx, 2
    mov byte ptr [rip + grp], cl
    mov ecx, eax
    and ecx, 3
    shl ecx, 4
    mov edx, r8d
    shr edx, 4
    or ecx, edx
    mov byte ptr [rip + grp + 1], cl
    mov ecx, r8d
    and ecx, 15
    shl ecx, 2
    mov edx, r9d
    shr edx, 6
    or ecx, edx
    mov byte ptr [rip + grp + 2], cl
    mov ecx, r9d
    and ecx, 63
    mov byte ptr [rip + grp + 3], cl
    mov r15d, 2
    cmp r10, 2
    jb pad_set
    mov r15d, 3
    cmp r10, 2
    je pad_set
    mov r15d, 4
pad_set:
    xor r14d, r14d
emit_group_loop:
    cmp r14, 4
    jae group_done
    cmp r14, r15
    jb emit_real
    mov eax, 61
    jmp emit_do
emit_real:
    lea rdx, [rip + grp]
    movzx ecx, byte ptr [rdx + r14]
    lea rdx, [rip + alpha]
    movzx eax, byte ptr [rdx + rcx]
emit_do:
    call emit
    inc r14
    jmp emit_group_loop
group_done:
    add r13, 3
    jmp enc_loop
enc_done:
    cmp qword ptr [rip + col], 0
    je finish
    call flush
finish:
    mov eax, 60
    xor edi, edi
    syscall

# al = character; appends to current line, flushes when line reaches 76
emit:
    mov rcx, qword ptr [rip + col]
    lea rdx, [rip + linebuf]
    mov byte ptr [rdx + rcx], al
    inc rcx
    mov qword ptr [rip + col], rcx
    cmp rcx, 76
    jb emit_ret
    call flush
emit_ret:
    ret

# writes current line plus newline, resets column
flush:
    mov rdx, qword ptr [rip + col]
    lea rsi, [rip + linebuf]
    mov byte ptr [rsi + rdx], 10
    inc rdx
    mov qword ptr [rip + col], 0
    call write_all
    ret

# rsi = buffer, rdx = count
write_all:
wa_loop:
    test rdx, rdx
    jz wa_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle wa_err
    add rsi, rax
    sub rdx, rax
    jmp wa_loop
wa_err:
    mov eax, 60
    mov edi, 1
    syscall
wa_done:
    ret

.section .rodata
alpha:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
inbuf:
    .zero 65536
grp:
    .zero 4
linebuf:
    .zero 80
col:
    .zero 8
