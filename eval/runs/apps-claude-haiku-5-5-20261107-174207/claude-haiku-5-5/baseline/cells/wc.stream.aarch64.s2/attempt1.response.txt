.globl _start

_start:
    mov x19, #0                 // lines
    mov x20, #0                 // words
    mov x21, #0                 // bytes
    mov x22, #0                 // in-word flag

read_loop:
    mov x8, #63
    mov x0, #0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov x2, #4096
    svc #0
    cmp x0, #0
    b.le done_read
    mov x23, x0                 // count of bytes in this chunk
    add x21, x21, x23
    adrp x24, buf
    add x24, x24, :lo12:buf

proc_loop:
    ldrb w3, [x24], #1
    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    cmp w3, #32
    b.eq is_ws
    sub w4, w3, #9
    cmp w4, #4
    b.ls is_ws
    // non-whitespace byte
    cbnz x22, 2f
    add x20, x20, #1
    mov x22, #1
2:
    b 3f
is_ws:
    mov x22, #0
3:
    subs x23, x23, #1
    b.ne proc_loop
    b read_loop

done_read:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf

    mov x0, x19
    mov x1, x25
    bl fmt
    mov x25, x0
    mov w3, #32
    strb w3, [x25], #1

    mov x0, x20
    mov x1, x25
    bl fmt
    mov x25, x0
    mov w3, #32
    strb w3, [x25], #1

    mov x0, x21
    mov x1, x25
    bl fmt
    mov x25, x0
    mov w3, #10
    strb w3, [x25], #1

    // write output, looping on partial writes
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x25, x1
write_loop:
    cbz x2, write_done
    mov x0, #1
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_done
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop
write_done:
    mov x0, #0
    mov x8, #93
    svc #0

// fmt: x0 = value, x1 = destination; writes decimal digits, returns x0 = pointer past digits
fmt:
    adrp x9, tmp
    add x9, x9, :lo12:tmp
    add x10, x9, #32            // end of tmp
    mov x11, #10
    mov x12, x10
fmt_loop:
    udiv x13, x0, x11
    msub x14, x13, x11, x0      // remainder
    sub x12, x12, #1
    add w14, w14, #48
    strb w14, [x12]
    mov x0, x13
    cbnz x0, fmt_loop
fmt_copy:
    cmp x12, x10
    b.hs fmt_done
    ldrb w14, [x12], #1
    strb w14, [x1], #1
    b fmt_copy
fmt_done:
    mov x0, x1
    ret

.bss
.align 4
buf:
    .zero 4096
outbuf:
    .zero 64
tmp:
    .zero 32
