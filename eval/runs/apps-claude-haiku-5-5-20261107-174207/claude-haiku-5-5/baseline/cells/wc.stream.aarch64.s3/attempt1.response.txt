.section .bss
.align 4
inbuf:
    .zero 4096
outbuf:
    .zero 64
numtmp:
    .zero 24

.text
.globl _start

_start:
    adrp x23, inbuf
    add  x23, x23, :lo12:inbuf
    mov  x19, #0            // lines
    mov  x20, #0            // words
    mov  x21, #0            // bytes
    mov  x22, #0            // in_word flag

read_loop:
    mov  x0, #0
    mov  x1, x23
    mov  x2, #4096
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    mov  x24, x0            // n bytes read
    add  x21, x21, x0
    mov  x25, #0            // index

scan_loop:
    cmp  x25, x24
    b.ge read_loop
    ldrb w0, [x23, x25]
    add  x25, x25, #1
    cmp  w0, #0x0A
    b.ne check_ws
    add  x19, x19, #1
check_ws:
    cmp  w0, #0x20
    b.eq is_ws
    sub  w1, w0, #0x09
    cmp  w1, #4
    b.ls is_ws
    // non-whitespace byte
    cbnz x22, scan_loop
    add  x20, x20, #1
    mov  x22, #1
    b    scan_loop
is_ws:
    mov  x22, #0
    b    scan_loop

read_done:
    adrp x27, outbuf
    add  x27, x27, :lo12:outbuf
    adrp x28, outbuf
    add  x28, x28, :lo12:outbuf   // write start
    mov  x0, x19
    bl   put_num
    mov  w0, #0x20
    strb w0, [x27], #1
    mov  x0, x20
    bl   put_num
    mov  w0, #0x20
    strb w0, [x27], #1
    mov  x0, x21
    bl   put_num
    mov  w0, #0x0A
    strb w0, [x27], #1
    mov  x26, x27           // end of output

write_loop:
    cmp  x28, x26
    b.hs exit_prog
    mov  x0, #1
    mov  x1, x28
    sub  x2, x26, x28
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit_prog
    add  x28, x28, x0
    b    write_loop

exit_prog:
    mov  x0, #0
    mov  x8, #93
    svc  #0

// put_num: x0 = value, x27 = destination pointer; advances x27
put_num:
    adrp x9, numtmp
    add  x9, x9, :lo12:numtmp
    add  x10, x9, #24       // end of temp
    mov  x11, x10           // current start of digits
    mov  x12, #10
pn_digit:
    udiv x13, x0, x12
    msub x14, x13, x12, x0  // remainder
    add  w14, w14, #0x30
    sub  x11, x11, #1
    strb w14, [x11]
    mov  x0, x13
    cbnz x0, pn_digit
pn_copy:
    cmp  x11, x10
    b.hs pn_done
    ldrb w14, [x11], #1
    strb w14, [x27], #1
    b    pn_copy
pn_done:
    ret
