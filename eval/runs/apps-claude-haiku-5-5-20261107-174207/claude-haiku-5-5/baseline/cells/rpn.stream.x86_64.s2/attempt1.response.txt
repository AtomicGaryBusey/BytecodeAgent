.text
.globl _start
_start:
    lea rbx, [rip + inbuf]
    lea rbp, [rip + vstack]
    xor r12d, r12d
read_loop:
    lea rsi, [rbx + r12]
    mov edx, 65536
    sub rdx, r12
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
line_start:
    cmp r13, r12
    jae done
    xor r14d, r14d
    xor r15d, r15d
token_loop:
skip_sep:
    cmp r13, r12
    jae end_line
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x20
    je skip_inc
    cmp al, 0x09
    je skip_inc
    jmp not_sep
skip_inc:
    inc r13
    jmp skip_sep
not_sep:
    cmp al, 0x0A
    jne tok_start
    inc r13
    jmp end_line
tok_start:
    mov r8, r13
tok_scan:
    inc r13
    cmp r13, r12
    jae tok_done
    movzx eax, byte ptr [rbx + r13]
    cmp al, 0x20
    je tok_done
    cmp al, 0x09
    je tok_done
    cmp al, 0x0A
    je tok_done
    jmp tok_scan
tok_done:
    test r15, r15
    jnz token_loop
    call process_token
    jmp token_loop

end_line:
    test r15, r15
    jnz emit_error
    cmp r14, 1
    jne emit_error
    mov rax, [rbp]
    call out_number
    jmp line_start
emit_error:
    lea rsi, [rip + err_msg]
    mov edx, 6
    call write_all
    jmp line_start

done:
    xor edi, edi
    mov eax, 60
    syscall

# process token [r8, r13)
process_token:
    mov rcx, r13
    sub rcx, r8
    movzx r9d, byte ptr [rbx + r8]
    cmp rcx, 1
    jne pt_literal
    cmp r9b, 0x2B
    je pt_op
    cmp r9b, 0x2D
    je pt_op
    cmp r9b, 0x2A
    je pt_op
    cmp r9b, 0x2F
    je pt_op
    cmp r9b, 0x25
    je pt_op
    jmp pt_literal

pt_literal:
    mov rsi, r8
    xor r10d, r10d
    cmp byte ptr [rbx + rsi], 0x2D
    jne pl_digits
    mov r10d, 1
    inc rsi
pl_digits:
    mov rcx, r13
    sub rcx, rsi
    jz set_err
    cmp rcx, 18
    ja set_err
    xor eax, eax
pl_loop:
    cmp rsi, r13
    jae pl_end
    movzx edx, byte ptr [rbx + rsi]
    sub edx, 0x30
    cmp edx, 9
    ja set_err
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    jmp pl_loop
pl_end:
    test r10, r10
    jz pl_push
    neg rax
pl_push:
    mov [rbp + r14*8], rax
    inc r14
    ret

pt_op:
    cmp r14, 2
    jb set_err
    lea rsi, [rbp + r14*8 - 16]
    mov rdi, [rsi]
    mov rdx, [rsi + 8]
    dec r14
    cmp r9b, 0x2B
    je op_add
    cmp r9b, 0x2D
    je op_sub
    cmp r9b, 0x2A
    je op_mul
    test rdx, rdx
    jz set_err
    cmp rdx, -1
    jne op_divok
    movabs rax, 0x8000000000000000
    cmp rdi, rax
    je set_err
op_divok:
    mov rcx, rdx
    mov rax, rdi
    cqo
    idiv rcx
    cmp r9b, 0x2F
    je op_store
    mov rax, rdx
    jmp op_store
op_add:
    mov rax, rdi
    add rax, rdx
    jmp op_store
op_sub:
    mov rax, rdi
    sub rax, rdx
    jmp op_store
op_mul:
    mov rax, rdi
    imul rax, rdx
op_store:
    mov [rsi], rax
    ret

set_err:
    mov r15d, 1
    ret

# rax = signed value; writes decimal and newline
out_number:
    lea rsi, [rip + obuf + 32]
    mov byte ptr [rsi - 1], 10
    dec rsi
    xor r9d, r9d
    test rax, rax
    jns on_pos
    mov r9d, 1
    neg rax
on_pos:
    mov rcx, 10
on_digit:
    xor edx, edx
    div rcx
    add dl, 0x30
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz on_digit
    test r9d, r9d
    jz on_write
    dec rsi
    mov byte ptr [rsi], 0x2D
on_write:
    lea rdx, [rip + obuf + 32]
    sub rdx, rsi
    call write_all
    ret

# rsi = buffer, rdx = length
write_all:
    test rdx, rdx
    jz wa_done
wa_loop:
    mov eax, 1
    mov edi, 1
    syscall
    cmp rax, 0
    jle wa_done
    add rsi, rax
    sub rdx, rax
    jnz wa_loop
wa_done:
    ret

.section .rodata
err_msg:
    .ascii "error\n"

.bss
.align 16
inbuf:
    .zero 65536
obuf:
    .zero 32
.align 16
vstack:
    .zero 262144
