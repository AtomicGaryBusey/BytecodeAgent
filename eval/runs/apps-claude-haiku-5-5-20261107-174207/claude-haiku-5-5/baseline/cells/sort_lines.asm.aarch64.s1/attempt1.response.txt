```
// sort_lines: sort stdin lines bytewise ascending and write them to stdout.

.equ CAP, 32768
.equ MAXLINES, 1000

.section .bss
.balign 16
inbuf:
    .zero CAP
.balign 8
starts:
    .zero MAXLINES*8
lens:
    .zero MAXLINES*8
order:
    .zero MAXLINES*8

.section .rodata
nl:
    .byte 10

.section .text
.globl _start

_start:
    // read all of stdin into inbuf; x20 = total bytes read
    mov x20, #0
read_loop:
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x20
    mov x2, #CAP
    sub x2, x2, x20
    cbz x2, read_done
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le read_done
    add x20, x20, x0
    b read_loop
read_done:

    // split into lines: starts[k], lens[k]; x22 = line count
    adrp x23, inbuf
    add x23, x23, :lo12:inbuf
    adrp x24, starts
    add x24, x24, :lo12:starts
    adrp x25, lens
    add x25, x25, :lo12:lens
    mov x21, #0
    mov x22, #0
parse_loop:
    cmp x21, x20
    b.hs parse_done
    mov x26, x21
    mov x27, x21
scan_loop:
    cmp x27, x20
    b.hs scan_end
    ldrb w9, [x23, x27]
    cmp w9, #10
    b.eq scan_end
    add x27, x27, #1
    b scan_loop
scan_end:
    sub x9, x27, x26
    str x26, [x24, x22, lsl #3]
    str x9, [x25, x22, lsl #3]
    add x22, x22, #1
    add x21, x27, #1
    b parse_loop
parse_done:

    // order[k] = k
    mov x9, #0
init_loop:
    cmp x9, x22
    b.hs init_done
    adrp x10, order
    add x10, x10, :lo12:order
    str x9, [x10, x9, lsl #3]
    add x9, x9, #1
    b init_loop
init_done:

    // insertion sort of order[] by line content
    mov x9, #1
sort_outer:
    cmp x9, x22
    b.hs sort_done
    adrp x10, order
    add x10, x10, :lo12:order
    ldr x12, [x10, x9, lsl #3]   // key
    mov x13, x9                  // j
sort_inner:
    cbz x13, sort_place
    sub x14, x13, #1
    ldr x15, [x10, x14, lsl #3]  // prev = order[j-1]
    mov x0, x12
    mov x1, x15
    bl lt_line
    cbz x0, sort_place
    str x15, [x10, x13, lsl #3]  // order[j] = prev
    mov x13, x14
    b sort_inner
sort_place:
    str x12, [x10, x13, lsl #3]
    add x9, x9, #1
    b sort_outer
sort_done:

    // output each line followed by newline
    mov x9, #0
out_loop:
    cmp x9, x22
    b.hs out_done
    adrp x10, order
    add x10, x10, :lo12:order
    ldr x11, [x10, x9, lsl #3]   // idx
    adrp x12, starts
    add x12, x12, :lo12:starts
    ldr x13, [x12, x11, lsl #3]  // start
    adrp x14, lens
    add x14, x14, :lo12:lens
    ldr x15, [x14, x11, lsl #3]  // len
    cbz x15, out_nl
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x13
    mov x2, x15
    mov x0, #1
    mov x8, #64
    svc #0
out_nl:
    mov x0, #1
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x2, #1
    mov x8, #64
    svc #0
    add x9, x9, #1
    b out_loop
out_done:

    mov x0, #0
    mov x8, #93
    svc #0

// lt_line: x0 = index a, x1 = index b; returns x0 = 1 if line a < line b, else 0.
// Uses only x0-x7.
lt_line:
    adrp x2, starts
    add x2, x2, :lo12:starts
    adrp x3, lens
    add x3, x3, :lo12:lens
    ldr x4, [x2, x0, lsl #3]     // start a
    ldr x5, [x3, x0, lsl #3]     // len a
    ldr x6, [x2, x1, lsl #3]     // start b
    ldr x7, [x3, x1, lsl #3]     // len b
    adrp x2, inbuf
    add x2, x2, :lo12:inbuf
    add x4, x2, x4               // ptr a
    add x6, x2, x6               // ptr b
    cmp x5, x7
    csel x1, x5, x7, lo          // x1 = min(len a, len b)
    mov x0, #0                   // i
lt_loop:
    cmp x0, x1
    b.hs lt_end
    ldrb w2, [x4, x0]
    ldrb w3, [x6, x0]
    cmp w2, w3
    b.ne lt_diff
    add x0, x0, #1
    b lt_loop
lt_diff:
    cset x0, lo
    ret
lt_end:
    cmp x5, x7
    cset x0, lo
    ret
```
