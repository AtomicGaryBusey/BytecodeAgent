	adrp	x19, inbuf
	add	x19, x19, :lo12:inbuf
	mov	x21, #0

	add	x1, x19, x21
	mov	x2, #65536
	sub	x2, x2, x21
	mov	x8, #63
	svc	#0

	add	x21, x21, x0
	b	read_loop

	cmp	x20, x21
	b.ge	no_bang
	ldrb	w0, [x19, x20]
	cmp	w0, #0x21
	b.eq	found_bang
	add	x20, x20, #1
	b	find_loop

	mov	x25, x21		// input cursor = N (empty)
	mov	x26, x21		// input end = N
	b	bounds_done

	mov	x26, x21		// input end = N
bounds_done:

	add	x21, x21, :lo12:jt
	adrp	x22, bstack
	add	x22, x22, :lo12:bstack

	mov	x24, #0			// stack depth

	b.ge	scan_done
	ldrb	w0, [x19, x23]
	cmp	w0, #0x5b
	b.ne	1f
	str	w23, [x22, x24, lsl #2]

1:

	b.ne	next_scan
	cbz	x24, fail
	sub	x24, x24, #1
	ldr	w1, [x22, x24, lsl #2]
	str	w23, [x21, x1, lsl #2]
	str	w1, [x21, x23, lsl #2]

	b	scan_loop

	add	x22, x22, :lo12:tape

	b.ge	done
	ldrb	w0, [x19, x24]

	b.eq	op_right
	cmp	w0, #0x3c
	b.eq	op_left
	cmp	w0, #0x2b
	b.eq	op_inc
	cmp	w0, #0x2d
	b.eq	op_dec
	cmp	w0, #0x2e
	b.eq	op_out
	cmp	w0, #0x2c
	b.eq	op_in
	cmp	w0, #0x5b
	b.eq	op_lb
	cmp	w0, #0x5d
	b.eq	op_rb
	b	next
op_right:
	add	x23, x23, #1
	b	next
op_left:
	sub	x23, x23, #1
	b	next
op_inc:
	ldrb	w0, [x22, x23]
	add	w0, w0, #1
	strb	w0, [x22, x23]
	b	next
op_dec:
	ldrb	w0, [x22, x23]
	sub	w0, w0, #1
	strb	w0, [x22, x23]
	b	next
op_out:

	strb	w0, [x28, x27]
	add	x27, x27, #1
	b	next

	ldrb	w0, [x19, x25]
	add	x25, x25, #1
	b	in_store

	b	next

	b	next

	b	run

	mov	x9, x28
	mov	x10, x27

	mov	x0, #1
	mov	x1, x9
	mov	x2, x10
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	b.le	flush_done
	add	x9, x9, x0
	sub	x10, x10, x0
	b	flush_loop

	mov	x0, #0
	mov	x8, #93
	svc	#0

	mov	x8, #93
	svc	#0
