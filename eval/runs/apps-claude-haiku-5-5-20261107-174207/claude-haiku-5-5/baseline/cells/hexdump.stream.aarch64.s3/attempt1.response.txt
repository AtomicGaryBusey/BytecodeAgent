// hexdump: canonical hex+ASCII dump of standard input (AArch64 Linux, no libc)

	.text
	.globl _start

_start:
	// read all of standard input into inbuf
	adrp	x20, inbuf
	add	x20, x20, :lo12:inbuf
	mov	x21, #0				// total length
rd_loop:
	mov	x0, #0
	add	x1, x20, x21
	movz	x2, #1, lsl #16			// 65536
	sub	x2, x2, x21
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	b.le	rd_done
	add	x21, x21, x0
	b	rd_loop
rd_done:

	// dump rows
	mov	x22, #0				// offset
row_loop:
	cmp	x22, x21
	b.hs	rows_done
	sub	x23, x21, x22
	mov	x24, #16
	cmp	x23, #16
	csel	x24, x23, x24, lo		// n = min(remaining, 16)
	add	x25, x20, x22			// row pointer
	bl	format_row
	mov	x2, x0
	adrp	x1, obuf
	add	x1, x1, :lo12:obuf
	bl	write_all
	add	x22, x22, #16
	b	row_loop
rows_done:

	// final offset line: total length, 8 hex digits, newline
	adrp	x16, obuf
	add	x16, x16, :lo12:obuf
	mov	x9, x16
	adrp	x10, hexdig
	add	x10, x10, :lo12:hexdig
	mov	x11, #28
	mov	x12, x21
fin_loop:
	lsr	x13, x12, x11
	and	x13, x13, #15
	ldrb	w14, [x10, x13]
	strb	w14, [x9], #1
	subs	x11, x11, #4
	b.pl	fin_loop
	mov	w14, #10
	strb	w14, [x9], #1
	mov	x1, x16
	sub	x2, x9, x16
	bl	write_all

	// exit(0)
	mov	x0, #0
	mov	x8, #93
	svc	#0

// format_row: x22 = offset, x24 = n (1..16), x25 = row bytes
// writes one formatted line into obuf, returns its length in x0
format_row:
	adrp	x16, obuf
	add	x16, x16, :lo12:obuf
	mov	x9, x16
	adrp	x10, hexdig
	add	x10, x10, :lo12:hexdig
	// offset: 8 hex digits
	mov	x11, #28
	mov	x12, x22
fr_off:
	lsr	x13, x12, x11
	and	x13, x13, #15
	ldrb	w14, [x10, x13]
	strb	w14, [x9], #1
	subs	x11, x11, #4
	b.pl	fr_off
	// two spaces
	mov	w14, #0x2020
	strh	w14, [x9], #2
	// 16 byte slots
	mov	x11, #0
fr_hex:
	cmp	x11, x24
	b.hs	fr_pad
	ldrb	w13, [x25, x11]
	lsr	w14, w13, #4
	ldrb	w15, [x10, x14]
	strb	w15, [x9], #1
	and	w14, w13, #15
	ldrb	w15, [x10, x14]
	strb	w15, [x9], #1
	mov	w15, #32
	strb	w15, [x9], #1
	b	fr_next
fr_pad:
	mov	w15, #0x2020
	strh	w15, [x9], #2
	mov	w15, #32
	strb	w15, [x9], #1
fr_next:
	cmp	x11, #7
	b.ne	1f
	mov	w15, #32
	strb	w15, [x9], #1
1:
	add	x11, x11, #1
	cmp	x11, #16
	b.lo	fr_hex
	// separator: space, '|'
	mov	w15, #32
	strb	w15, [x9], #1
	mov	w15, #124
	strb	w15, [x9], #1
	// ASCII column
	mov	x11, #0
fr_chr:
	cmp	x11, x24
	b.hs	fr_chr_end
	ldrb	w13, [x25, x11]
	sub	w14, w13, #32
	cmp	w14, #94
	b.ls	2f
	mov	w13, #46
2:
	strb	w13, [x9], #1
	add	x11, x11, #1
	b	fr_chr
fr_chr_end:
	mov	w15, #124
	strb	w15, [x9], #1
	mov	w15, #10
	strb	w15, [x9], #1
	sub	x0, x9, x16
	ret

// write_all: x1 = buffer, x2 = length; writes everything to fd 1
write_all:
wa_loop:
	cbz	x2, wa_done
	mov	x0, #1
	mov	x8, #64
	svc	#0
	cmp	x0, #0
	b.le	wa_done
	add	x1, x1, x0
	sub	x2, x2, x0
	b	wa_loop
wa_done:
	ret

	.section .rodata
hexdig:
	.ascii "0123456789abcdef"

	.bss
	.balign 16
inbuf:
	.zero 65536
obuf:
	.zero 128
