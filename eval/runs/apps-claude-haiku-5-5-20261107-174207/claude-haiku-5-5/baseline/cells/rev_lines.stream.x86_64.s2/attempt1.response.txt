.text
.globl _start
_start:
    xor r13d, r13d
    lea r8, [rip + inbuf]
    lea r9, [rip + outbuf]
read_loop:
    mov rdx, 65536
    sub rdx, r13
    jz read_done
    lea rsi, [r8 + r13]
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r14d, r14d
    xor r15d, r15d
proc_loop:
    cmp r14, r13
    jae write_phase
    mov rbx, r14
scan:
    cmp rbx, r13
    jae found
    cmp byte ptr [r8 + rbx], 10
    je found
    inc rbx
    jmp scan
found:
    mov rcx, rbx
    sub rcx, r14
    xor edx, edx
copy:
    cmp rdx, rcx
    jae copy_done
    mov rax, rbx
    sub rax, rdx
    dec rax
    movzx eax, byte ptr [r8 + rax]
    lea rsi, [r15 + rdx]
    mov byte ptr [r9 + rsi], al
    inc rdx
    jmp copy
copy_done:
    add r15, rcx
    mov byte ptr [r9 + r15], 10
    inc r15
    lea r14, [rbx + 1]
    jmp proc_loop
write_phase:
    xor r14d, r14d
wloop:
    cmp r14, r15
    jae done
    mov eax, 1
    mov edi, 1
    lea rsi, [r9 + r14]
    mov rdx, r15
    sub rdx, r14
    syscall
    test rax, rax
    jle done
    add r14, rax
    jmp wloop
done:
    mov eax, 60
    xor edi, edi
    syscall
