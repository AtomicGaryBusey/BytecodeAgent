```
.intel_syntax noprefix
.section .text
.globl _start

_start:
    xor ebx, ebx
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 65536
    syscall
    test rax, rax
    jle input_done
    mov r15, rax
    add rbx, rax
    lea rsi, [rip + inbuf]
    xor ecx, ecx

proc_loop:
    cmp rcx, r15
    jae read_loop
    movzx eax, byte ptr [rsi + rcx]
    inc rcx
    cmp eax, 10
    jne 2f
    inc r12
2:
    cmp eax, 32
    je 3f
    lea edx, [rax - 9]
    cmp edx, 4
    jbe 3f
    test r14d, r14d
    jnz proc_loop
    inc r13
    mov r14d, 1
    jmp proc_loop
3:
    xor r14d, r14d
    jmp proc_loop

input_done:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call put_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call put_num
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, rbx
    call put_num
    mov byte ptr [rdi], 10
    inc rdi

    lea r14, [rip + outbuf]
    mov r15, rdi
    sub r15, r14

write_loop:
    test r15, r15
    jz finish
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    mov eax, 1
    syscall
    test rax, rax
    jle finish
    add r14, rax
    sub r15, rax
    jmp write_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

# rax = value, rdi = destination; writes decimal digits, returns rdi advanced
put_num:
    xor ecx, ecx
    mov r8, 10
pn_digits:
    xor edx, edx
    div r8
    push rdx
    inc ecx
    test rax, rax
    jnz pn_digits
pn_write:
    pop rax
    add al, 48
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz pn_write
    ret

.section .bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 64
```
