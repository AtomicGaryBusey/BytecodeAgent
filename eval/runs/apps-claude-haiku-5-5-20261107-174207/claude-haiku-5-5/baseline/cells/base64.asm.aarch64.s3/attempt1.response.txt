    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    // x19 = input buffer, x20 = input length
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0
read_loop:
    mov  x9, #65536
    sub  x2, x9, x20
    cbz  x2, read_done
    add  x1, x19, x20
    mov  x0, #0
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop

    adrp x24, alphabet
    add  x24, x24, :lo12:alphabet
    adrp x22, outbuf
    add  x22, x22, :lo12:outbuf
    mov  x23, #0
    mov  x21, #0

loop_full:

    cmp  x9, x20
    b.hi tail
    add  x10, x19, x21
    ldrb w0, [x10]
    ldrb w1, [x10, #1]
    ldrb w2, [x10, #2]
    lsr  w3, w0, #2
    and  w4, w0, #3
    lsl  w4, w4, #4
    lsr  w5, w1, #4
    orr  w4, w4, w5
    and  w5, w1, #15

    bl   emit
    ldrb w9, [x24, x4]
    mov  w0, w9
    bl   emit
    ldrb w9, [x24, x5]
    mov  w0, w9
    bl   emit
    ldrb w9, [x24, x6]
    mov  w0, w9
    bl   emit
    add  x21, x21, #3
    b    loop_full

    sub  x25, x20, x21
    cbz  x25, finish
    add  x10, x19, x21
    ldrb w0, [x10]
    mov  w1, #0
    cmp  x25, #2

    and  w4, w0, #3

    lsr  w5, w1, #4

    and  w5, w1, #15
    lsl  w5, w5, #2
    ldrb w9, [x24, x3]
    mov  w0, w9
    bl   emit
    ldrb w9, [x24, x4]
    mov  w0, w9
    bl   emit
    cmp  x25, #2
    b.ne tail_eq
    ldrb w9, [x24, x5]
    mov  w0, w9
    bl   emit
    b    tail_last

    mov  w0, #61
    bl   emit
tail_last:
    mov  w0, #61
    bl   emit

    cbz  x23, write_out
    mov  w9, #10
    strb w9, [x22], #1

write_out:
    adrp x26, outbuf
    add  x26, x26, :lo12:outbuf
    sub  x27, x22, x26
write_loop:
    cbz  x27, exit_prog
    mov  x0, #1
    mov  x1, x26
    mov  x2, x27
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit_prog
    add  x26, x26, x0
    sub  x27, x27, x0
    b    write_loop

exit_prog:
    mov  x0, #0
    mov  x8, #93
    svc  #0

// emit: store the character in w0 at [x22], advance x22,
// track the column in x23, and write a newline after 76 characters.
// Clobbers only w0, x22, x23 (and x30 via the call).
emit:
    strb w0, [x22], #1
    add  x23, x23, #1
    cmp  x23, #76
    b.ne emit_ret
    mov  w0, #10
    strb w0, [x22], #1
    mov  x23, #0
emit_ret:
    ret

.bss
.balign 16
inbuf:
    .zero 65536
outbuf:
    .zero 98304
