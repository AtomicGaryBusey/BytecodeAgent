.text
.globl _start
_start:
  adrp x19, inbuf
  add x19, x19, :lo12:inbuf
  adrp x25, linebuf
  add x25, x25, :lo12:linebuf
  mov x20, #0
read_loop:
  mov x2, #0x10000
  sub x2, x2, x20
  cbz x2, read_done
  add x1, x19, x20
  mov x0, #0
  mov x8, #63
  svc #0
  cmp x0, #0
  b.le read_done
  add x20, x20, x0
  b read_loop
read_done:
  mov x22, #0
row_loop:
  cmp x22, x20
  b.ge final
  sub x23, x20, x22
  mov x24, #16
  cmp x23, x24
  csel x23, x23, x24, lt
  mov x26, x25
  mov x0, x22
  mov x1, #8
  mov x2, x26
  bl hex_emit
  mov x26, x0
  mov w4, #32
  strb w4, [x26], #1
  strb w4, [x26], #1
  add x21, x19, x22
  mov x27, #0
bytes_loop:
  cmp x27, #16
  b.ge bytes_done
  cmp x27, x23
  b.ge pad
  ldrb w0, [x21, x27]
  mov x1, #2
  mov x2, x26
  bl hex_emit
  mov x26, x0
  mov w4, #32
  strb w4, [x26], #1
  b after_byte
pad:
  mov w4, #32
  strb w4, [x26], #1
  strb w4, [x26], #1
  strb w4, [x26], #1
after_byte:
  cmp x27, #7
  b.ne 3f
  mov w4, #32
  strb w4, [x26], #1
3:
  add x27, x27, #1
  b bytes_loop
bytes_done:
  mov w4, #32
  strb w4, [x26], #1
  mov w4, #124
  strb w4, [x26], #1
  mov x27, #0
chars_loop:
  cmp x27, x23
  b.ge chars_done
  ldrb w0, [x21, x27]
  sub w1, w0, #32
  cmp w1, #94
  mov w2, #46
  csel w0, w0, w2, ls
  strb w0, [x26], #1
  add x27, x27, #1
  b chars_loop
chars_done:
  mov w4, #124
  strb w4, [x26], #1
  mov w4, #10
  strb w4, [x26], #1
  mov x1, x25
  sub x2, x26, x25
  bl write_all
  add x22, x22, #16
  b row_loop
final:
  mov x0, x20
  mov x1, #8
  mov x2, x25
  bl hex_emit
  mov x26, x0
  mov w4, #10
  strb w4, [x26]
  add x26, x26, #1
  mov x1, x25
  sub x2, x26, x25
  bl write_all
  mov x0, #0
  mov x8, #93
  svc #0

// hex_emit: x0 = value, x1 = digit count, x2 = destination
// returns x0 = destination + digit count; clobbers x3-x7
hex_emit:
  add x3, x2, x1
1:
  sub x3, x3, #1
  and x4, x0, #15
  cmp x4, #10
  add x6, x4, #48
  add x7, x4, #87
  csel x6, x6, x7, lo
  strb w6, [x3]
  lsr x0, x0, #4
  cmp x3, x2
  b.hi 1b
  add x0, x2, x1
  ret

// write_all: x1 = buffer, x2 = length; writes to fd 1 until done
write_all:
1:
  cbz x2, 2f
  mov x0, #1
  mov x8, #64
  svc #0
  cmp x0, #0
  b.le die
  add x1, x1, x0
  sub x2, x2, x0
  b 1b
2:
  ret

die:
  mov x0, #1
  mov x8, #93
  svc #0

.bss
.balign 16
inbuf:
  .zero 65536
linebuf:
  .zero 128
