.text
.globl _start

_start:
    adrp x19, inbuf
    add  x19, x19, :lo12:inbuf
    mov  x20, #0

read_loop:
    mov  x0, #0
    add  x1, x19, x20
    movz x2, #1, lsl #16
    sub  x2, x2, x20
    mov  x8, #63
    svc  #0
    cmp  x0, #0
    b.le read_done
    add  x20, x20, x0
    b    read_loop

read_done:
    adrp x21, outbuf
    add  x21, x21, :lo12:outbuf
    adrp x24, alphabet
    add  x24, x24, :lo12:alphabet
    mov  x22, #0
    mov  x23, #0
    mov  x25, #0

enc_loop:
    sub  x0, x20, x25
    cmp  x0, #3
    b.lt enc_tail

    ldrb w1, [x19, x25]
    add  x2, x25, #1
    ldrb w2, [x19, x2]
    add  x3, x25, #2
    ldrb w3, [x19, x3]
    add  x25, x25, #3

    lsl  w1, w1, #16
    orr  w1, w1, w2, lsl #8
    orr  w1, w1, w3

    ubfx w4, w1, #18, #6
    ubfx w5, w1, #12, #6
    ubfx w6, w1, #6, #6
    and  w7, w1, #63

    ldrb w0, [x24, x4]
    bl   emit
    ldrb w0, [x24, x5]
    bl   emit
    ldrb w0, [x24, x6]
    bl   emit
    ldrb w0, [x24, x7]
    bl   emit
    b    enc_loop

enc_tail:
    sub  x0, x20, x25
    cmp  x0, #0
    b.eq done_enc
    cmp  x0, #1
    b.eq tail1

    ldrb w1, [x19, x25]
    add  x2, x25, #1
    ldrb w2, [x19, x2]
    lsl  w1, w1, #16
    orr  w1, w1, w2, lsl #8
    ubfx w4, w1, #18, #6
    ubfx w5, w1, #12, #6
    ubfx w6, w1, #6, #6

    ldrb w0, [x24, x4]
    bl   emit
    ldrb w0, [x24, x5]
    bl   emit
    ldrb w0, [x24, x6]
    bl   emit
    mov  w0, #61
    bl   emit
    b    done_enc

tail1:
    ldrb w1, [x19, x25]
    lsl  w1, w1, #16
    ubfx w4, w1, #18, #6
    ubfx w5, w1, #12, #6

    ldrb w0, [x24, x4]
    bl   emit
    ldrb w0, [x24, x5]
    bl   emit
    mov  w0, #61
    bl   emit
    mov  w0, #61
    bl   emit

done_enc:
    cmp  x23, #0
    b.eq write_out
    mov  w0, #10
    strb w0, [x21, x22]
    add  x22, x22, #1

write_out:
    mov  x25, #0
wloop:
    cmp  x25, x22
    b.ge exit_ok
    mov  x0, #1
    add  x1, x21, x25
    sub  x2, x22, x25
    mov  x8, #64
    svc  #0
    cmp  x0, #0
    b.le exit_ok
    add  x25, x25, x0
    b    wloop

exit_ok:
    mov  x0, #0
    mov  x8, #93
    svc  #0

// emit: w0 = character to append to outbuf; inserts newline after 76 chars
// uses x21 (outbuf), x22 (out length), x23 (column), w9
emit:
    strb w0, [x21, x22]
    add  x22, x22, #1
    add  x23, x23, #1
    cmp  x23, #76
    b.ne 1f
    mov  w9, #10
    strb w9, [x21, x22]
    add  x22, x22, #1
    mov  x23, #0
1:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
outbuf:
    .zero 100000
