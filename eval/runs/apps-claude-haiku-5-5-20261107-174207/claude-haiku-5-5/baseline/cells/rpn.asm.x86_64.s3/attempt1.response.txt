.text
.globl _start
_start:
read_loop:
    mov rdx, 65536
    sub rdx, qword ptr [rip + in_len]
    jle read_done
    lea rsi, [rip + in_buf]
    add rsi, qword ptr [rip + in_len]
    xor eax, eax
    xor edi, edi
    syscall
    test rax, rax
    jle read_done
    add qword ptr [rip + in_len], rax
    jmp read_loop
read_done:
    lea rbx, [rip + in_buf]
    mov r13, qword ptr [rip + in_len]
    xor r12d, r12d
line_loop:
    cmp r12, r13
    jae done_all
    xor r14d, r14d
    xor r15d, r15d
tok_loop:
    cmp r12, r13
    jae finish_line
    movzx eax, byte ptr [rbx + r12]
    cmp al, 10
    je line_nl
    cmp al, 32
    je skip_sep
    cmp al, 9
    je skip_sep
    mov r8, r12
scan_tok:
    inc r12
    cmp r12, r13
    jae tok_end
    movzx eax, byte ptr [rbx + r12]
    cmp al, 10
    je tok_end
    cmp al, 32
    je tok_end
    cmp al, 9
    je tok_end
    jmp scan_tok
tok_end:
    call process_token
    jmp tok_loop
skip_sep:
    inc r12
    jmp tok_loop
line_nl:
    inc r12
finish_line:
    call emit_result
    jmp line_loop
done_all:
    call flush
    mov eax, 60
    xor edi, edi
    syscall

process_token:
    mov rax, r12
    sub rax, r8
    test r15, r15
    jnz pt_ret
    movzx ecx, byte ptr [rbx + r8]
    cmp rax, 1
    jne pt_lit
    cmp cl, 43
    je pt_op
    cmp cl, 45
    je pt_op
    cmp cl, 42
    je pt_op
    cmp cl, 47
    je pt_op
    cmp cl, 37
    je pt_op
    jmp pt_lit
pt_op:
    movzx r9d, cl
    cmp r14, 2
    jb pt_err
    lea rdx, [rip + stk]
    mov rsi, qword ptr [rdx + r14*8 - 8]
    mov rax, qword ptr [rdx + r14*8 - 16]
    dec r14
    cmp r9d, 43
    je op_add
    cmp r9d, 45
    je op_sub
    cmp r9d, 42
    je op_mul
    jmp op_divmod
op_add:
    add rax, rsi
    jmp store_res
op_sub:
    sub rax, rsi
    jmp store_res
op_mul:
    imul rax, rsi
    jmp store_res
op_divmod:
    test rsi, rsi
    jz pt_err
    cmp rsi, -1
    jne dm_ok
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    je pt_err
dm_ok:
    cqo
    idiv rsi
    cmp r9d, 37
    jne store_res
    mov rax, rdx
store_res:
    lea rdx, [rip + stk]
    mov qword ptr [rdx + r14*8 - 8], rax
    ret
pt_lit:
    mov rdi, r8
    xor r10d, r10d
    cmp byte ptr [rbx + rdi], 45
    jne pl_digits
    mov r10d, 1
    inc rdi
pl_digits:
    mov rdx, r12
    sub rdx, rdi
    jz pt_err
    cmp rdx, 18
    ja pt_err
    xor eax, eax
pl_loop:
    cmp rdi, r12
    jae pl_done
    movzx ecx, byte ptr [rbx + rdi]
    sub ecx, 48
    cmp ecx, 9
    ja pt_err
    lea rax, [rax + rax*4]
    lea rax, [rcx + rax*2]
    inc rdi
    jmp pl_loop
pl_done:
    test r10d, r10d
    jz pl_push
    neg rax
pl_push:
    cmp r14, 1024
    jae pt_err
    lea rdx, [rip + stk]
    mov qword ptr [rdx + r14*8], rax
    inc r14
    ret
pt_err:
    mov r15d, 1
pt_ret:
    ret

emit_result:
    test r15, r15
    jnz er_err
    cmp r14, 1
    jne er_err
    mov rax, qword ptr [rip + stk]
    call emit_int
    lea rsi, [rip + nl_str]
    mov edx, 1
    jmp emit_bytes
er_err:
    lea rsi, [rip + err_str]
    mov edx, 6
    jmp emit_bytes

emit_int:
    lea r8, [rip + tmp + 32]
    xor ecx, ecx
    test rax, rax
    jns ei_pos
    neg rax
    mov ecx, 1
ei_pos:
    mov r9, 10
ei_loop:
    xor edx, edx
    div r9
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz ei_loop
    test ecx, ecx
    jz ei_nosign
    dec r8
    mov byte ptr [r8], 45
ei_nosign:
    mov rsi, r8
    lea rdx, [rip + tmp + 32]
    sub rdx, r8
    jmp emit_bytes

emit_bytes:
    mov rax, qword ptr [rip + out_len]
    add rax, rdx
    cmp rax, 8192
    jbe eb_ok
    push rsi
    push rdx
    call flush
    pop rdx
    pop rsi
eb_ok:
    lea rdi, [rip + out_buf]
    add rdi, qword ptr [rip + out_len]
    mov rcx, rdx
    rep movsb
    add qword ptr [rip + out_len], rdx
    ret

flush:
    lea rsi, [rip + out_buf]
    mov rdx, qword ptr [rip + out_len]
fl_loop:
    test rdx, rdx
    jz fl_done
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle fl_fail
    add rsi, rax
    sub rdx, rax
    jmp fl_loop
fl_done:
    mov qword ptr [rip + out_len], 0
    ret
fl_fail:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
err_str:
    .ascii "error\n"
nl_str:
    .byte 10

.bss
in_len:
    .zero 8
out_len:
    .zero 8
tmp:
    .zero 32
stk:
    .zero 8192
out_buf:
    .zero 8192
in_buf:
    .zero 65536
