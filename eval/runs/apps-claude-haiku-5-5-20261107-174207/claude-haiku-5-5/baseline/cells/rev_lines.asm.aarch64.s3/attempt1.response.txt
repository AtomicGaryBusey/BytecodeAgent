.globl _start

.text
_start:
    adrp x24, inbuf
    add x24, x24, :lo12:inbuf
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x19, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, read_done
    add x1, x24, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    tbnz x0, #63, read_done
    add x19, x19, x0
    b read_loop
read_done:

    mov x20, #0
    mov x21, #0

line_loop:
    cmp x20, x19
    b.ge done_lines
    mov x22, x20

find_nl:
    cmp x22, x19
    b.ge found
    ldrb w26, [x24, x22]
    cmp w26, #10
    b.eq found
    add x22, x22, #1
    b find_nl

found:
    sub x23, x22, #1
copy_loop:
    cmp x23, x20
    b.lt copy_done
    ldrb w26, [x24, x23]
    strb w26, [x25, x21]
    add x21, x21, #1
    sub x23, x23, #1
    b copy_loop
copy_done:
    mov w26, #10
    strb w26, [x25, x21]
    add x21, x21, #1
    add x20, x22, #1
    b line_loop
done_lines:

    mov x20, x25
    mov x23, x21
write_loop:
    cbz x23, write_done
    mov x0, #1
    mov x1, x20
    mov x2, x23
    mov x8, #64
    svc #0
    cmp x0, #0
    b.le write_done
    add x20, x20, x0
    sub x23, x23, x0
    b write_loop
write_done:

    mov x0, #0
    mov x8, #93
    svc #0

.bss
