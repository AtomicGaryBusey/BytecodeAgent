export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zero
@zero
	%zi =l phi @start 0, @zerobody %zinext
	%zlt =w csltl %zi, %n
	jnz %zlt, @zerobody, @zerodone
@zerobody
	%zoff =l mul %zi, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zinext =l add %zi, 1
	jmp @zero
@zerodone
	%naz =w ceql %na, 0
	jnz %naz, @scan, @checkb
@checkb
	%nbz =w ceql %nb, 0
	jnz %nbz, @scan, @outer
@outer
	%i =l phi @checkb 0, @ostore %inext
	%ilt =w csltl %i, %na
	jnz %ilt, @obody, @scan
@obody
	%aoff =l mul %i, 8
	%ap =l add %a, %aoff
	%ai =l loadl %ap
	jmp @inner
@inner
	%j =l phi @obody 0, @ibody %jnext
	%cy =l phi @obody 0, @ibody %cynext
	%jlt =w csltl %j, %nb
	jnz %jlt, @ibody, @ostore
@ibody
	%boff =l mul %j, 8
	%bp =l add %b, %boff
	%bj =l loadl %bp
	%alo =l and %ai, 4294967295
	%ahi =l shr %ai, 32
	%blo =l and %bj, 4294967295
	%bhi =l shr %bj, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0h =l shr %p0, 32
	%p1l =l and %p1, 4294967295
	%p2l =l and %p2, 4294967295
	%md0 =l add %p0h, %p1l
	%mid =l add %md0, %p2l
	%p0l =l and %p0, 4294967295
	%mdl =l and %mid, 4294967295
	%mds =l shl %mdl, 32
	%lo =l add %p0l, %mds
	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%mdh =l shr %mid, 32
	%hi0 =l add %p3, %p1h
	%hi1 =l add %hi0, %p2h
	%hi =l add %hi1, %mdh
	%ij =l add %i, %j
	%roff =l mul %ij, 8
	%rp =l add %r, %roff
	%rv =l loadl %rp
	%s1 =l add %rv, %lo
	%c1w =w cultl %s1, %rv
	%c1 =l extuw %c1w
	%s2 =l add %s1, %cy
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %rp
	%hc =l add %hi, %c1
	%cynext =l add %hc, %c2
	%jnext =l add %j, 1
	jmp @inner
@ostore
	%inb =l add %i, %nb
	%coff =l mul %inb, 8
	%cp =l add %r, %coff
	storel %cy, %cp
	%inext =l add %i, 1
	jmp @outer
@scan
	%k =l phi @zerodone %n, @checkb %n, @outer %n, @scandec %knext
	%kz =w ceql %k, 0
	jnz %kz, @done, @stest
@stest
	%k1 =l sub %k, 1
	%koff =l mul %k1, 8
	%kp =l add %r, %koff
	%kv =l loadl %kp
	%kvz =w ceql %kv, 0
	jnz %kvz, @scandec, @done
@scandec
	%knext =l sub %k, 1
	jmp @scan
@done
	ret %k
}
