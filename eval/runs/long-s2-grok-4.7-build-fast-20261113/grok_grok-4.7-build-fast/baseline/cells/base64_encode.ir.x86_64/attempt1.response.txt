export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @advance %inext
	%o =l phi @start 0, @advance %onext
	%rem =l sub %n, %i
	%is0 =w ceql %rem, 0
	jnz %is0, @ret, @more
@more
	%lt3 =w csltl %rem, 3
	jnz %lt3, @partial, @triplet
@triplet
	%tp0 =l add %src, %i
	%tb0 =w loadub %tp0
	%ti1 =l add %i, 1
	%tp1 =l add %src, %ti1
	%tb1 =w loadub %tp1
	%ti2 =l add %i, 2
	%tp2 =l add %src, %ti2
	%tb2 =w loadub %tp2
	%tb0s =w shl %tb0, 16
	%tb1s =w shl %tb1, 8
	%tv01 =w or %tb0s, %tb1s
	%val3 =w or %tv01, %tb2
	jmp @emit
@partial
	%is1 =w ceql %rem, 1
	jnz %is1, @one, @two
@one
	%op0 =l add %src, %i
	%ob0 =w loadub %op0
	%val1 =w shl %ob0, 16
	jmp @emit
@two
	%wp0 =l add %src, %i
	%wb0 =w loadub %wp0
	%wi1 =l add %i, 1
	%wp1 =l add %src, %wi1
	%wb1 =w loadub %wp1
	%wb0s =w shl %wb0, 16
	%wb1s =w shl %wb1, 8
	%val2 =w or %wb0s, %wb1s
	jmp @emit
@emit
	%val =w phi @triplet %val3, @one %val1, @two %val2
	%pad =w phi @triplet 0, @one 2, @two 1
	%iadd =l phi @triplet 3, @one 1, @two 2
	jmp @kloop
@kloop
	%k =l phi @emit 0, @kstore %knext
	%sh =w phi @emit 18, @kstore %shnext
	%idxsh =w shr %val, %sh
	%idx =w and %idxsh, 63
	%lt26 =w csltw %idx, 26
	jnz %lt26, @maz, @notaz
@maz
	%chaz =w add %idx, 65
	jmp @kstore
@notaz
	%lt52 =w csltw %idx, 52
	jnz %lt52, @malow, @notlow
@malow
	%chalow =w add %idx, 71
	jmp @kstore
@notlow
	%lt62 =w csltw %idx, 62
	jnz %lt62, @m09, @not09
@m09
	%ch09 =w sub %idx, 4
	jmp @kstore
@not09
	%is62 =w ceqw %idx, 62
	jnz %is62, @mplus, @mslash
@mplus
	%chplus =w copy 43
	jmp @kstore
@mslash
	%chslash =w copy 47
	jmp @kstore
@kstore
	%ch =w phi @maz %chaz, @malow %chalow, @m09 %ch09, @mplus %chplus, @mslash %chslash
	%off =l add %o, %k
	%dp =l add %out, %off
	storeb %ch, %dp
	%knext =l add %k, 1
	%shnext =w sub %sh, 6
	%kdone =w ceql %knext, 4
	jnz %kdone, @dopad, @kloop
@dopad
	%nopad =w ceqw %pad, 0
	jnz %nopad, @advance, @putpad
@putpad
	%p3 =l add %o, 3
	%dp3 =l add %out, %p3
	%eq1 =w copy 61
	storeb %eq1, %dp3
	%twopad =w ceqw %pad, 2
	jnz %twopad, @putpad2, @advance
@putpad2
	%p2 =l add %o, 2
	%dp2 =l add %out, %p2
	%eq2 =w copy 61
	storeb %eq2, %dp2
	jmp @advance
@advance
	%inext =l add %i, %iadd
	%onext =l add %o, 4
	%waspad =w ceqw %pad, 0
	jnz %waspad, @loop, @retpad
@retpad
	ret %onext
@ret
	ret %o
}
