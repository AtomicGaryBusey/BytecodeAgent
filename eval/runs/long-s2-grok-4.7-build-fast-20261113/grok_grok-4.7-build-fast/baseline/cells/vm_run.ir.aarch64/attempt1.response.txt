export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pcslot =l alloc8 8
	%spslot =l alloc8 8
	storel 0, %pcslot
	storel 0, %spslot
	jmp @loop
@loop
	%pc =l loadl %pcslot
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @decode
@decode
	%pcoff =l mul %pc, 8
	%pcaddr =l add %code, %pcoff
	%op =l loadl %pcaddr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @ck1
@ck1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @ck2
@ck2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @ck3
@ck3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @ck4
@ck4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @ck5
@ck5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @ck6
@ck6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @ck7
@ck7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @ck8
@ck8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @ck9
@ck9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @ck10
@ck10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @ck11
@ck11
	%is11 =w ceql %op, 11
	jnz %is11, @dojmp, @other
@other
	%pcn =l add %pc, 1
	storel %pcn, %pcslot
	jmp @loop
@push
	%ppc =l add %pc, 1
	%ppoff =l mul %ppc, 8
	%ppaddr =l add %code, %ppoff
	%pval =l loadl %ppaddr
	%psp =l loadl %spslot
	%pspoff =l mul %psp, 8
	%pspaddr =l add %stack, %pspoff
	storel %pval, %pspaddr
	%psp1 =l add %psp, 1
	storel %psp1, %spslot
	%ppc2 =l add %pc, 2
	storel %ppc2, %pcslot
	jmp @loop
@add
	%asp =l loadl %spslot
	%ai1 =l sub %asp, 1
	%ai2 =l sub %asp, 2
	%ao1 =l mul %ai1, 8
	%ao2 =l mul %ai2, 8
	%aa1 =l add %stack, %ao1
	%aa2 =l add %stack, %ao2
	%ab =l loadl %aa1
	%aa =l loadl %aa2
	%asum =l add %aa, %ab
	storel %asum, %aa2
	storel %ai1, %spslot
	%apc =l add %pc, 1
	storel %apc, %pcslot
	jmp @loop
@sub
	%ssp =l loadl %spslot
	%si1 =l sub %ssp, 1
	%si2 =l sub %ssp, 2
	%so1 =l mul %si1, 8
	%so2 =l mul %si2, 8
	%sa1 =l add %stack, %so1
	%sa2 =l add %stack, %so2
	%sb =l loadl %sa1
	%sa =l loadl %sa2
	%sdiff =l sub %sa, %sb
	storel %sdiff, %sa2
	storel %si1, %spslot
	%spc =l add %pc, 1
	storel %spc, %pcslot
	jmp @loop
@mul
	%msp =l loadl %spslot
	%mi1 =l sub %msp, 1
	%mi2 =l sub %msp, 2
	%mo1 =l mul %mi1, 8
	%mo2 =l mul %mi2, 8
	%ma1 =l add %stack, %mo1
	%ma2 =l add %stack, %mo2
	%mb =l loadl %ma1
	%ma =l loadl %ma2
	%mprod =l mul %ma, %mb
	storel %mprod, %ma2
	storel %mi1, %spslot
	%mpc =l add %pc, 1
	storel %mpc, %pcslot
	jmp @loop
@dup
	%dsp =l loadl %spslot
	%di =l sub %dsp, 1
	%do =l mul %di, 8
	%da =l add %stack, %do
	%dv =l loadl %da
	%dno =l mul %dsp, 8
	%dna =l add %stack, %dno
	storel %dv, %dna
	%dsp1 =l add %dsp, 1
	storel %dsp1, %spslot
	%dpc =l add %pc, 1
	storel %dpc, %pcslot
	jmp @loop
@swap
	%wsp =l loadl %spslot
	%wi1 =l sub %wsp, 1
	%wi2 =l sub %wsp, 2
	%wo1 =l mul %wi1, 8
	%wo2 =l mul %wi2, 8
	%wa1 =l add %stack, %wo1
	%wa2 =l add %stack, %wo2
	%wv1 =l loadl %wa1
	%wv2 =l loadl %wa2
	storel %wv2, %wa1
	storel %wv1, %wa2
	%wpc =l add %pc, 1
	storel %wpc, %pcslot
	jmp @loop
@drop
	%rsp =l loadl %spslot
	%rsp1 =l sub %rsp, 1
	storel %rsp1, %spslot
	%rpc =l add %pc, 1
	storel %rpc, %pcslot
	jmp @loop
@over
	%osp =l loadl %spslot
	%oi =l sub %osp, 2
	%oo =l mul %oi, 8
	%oa =l add %stack, %oo
	%ov =l loadl %oa
	%ono =l mul %osp, 8
	%ona =l add %stack, %ono
	storel %ov, %ona
	%osp1 =l add %osp, 1
	storel %osp1, %spslot
	%opc =l add %pc, 1
	storel %opc, %pcslot
	jmp @loop
@lt
	%lsp =l loadl %spslot
	%li1 =l sub %lsp, 1
	%li2 =l sub %lsp, 2
	%lo1 =l mul %li1, 8
	%lo2 =l mul %li2, 8
	%la1 =l add %stack, %lo1
	%la2 =l add %stack, %lo2
	%lb =l loadl %la1
	%la =l loadl %la2
	%lcmp =w csltl %la, %lb
	%lres =l extuw %lcmp
	storel %lres, %la2
	storel %li1, %spslot
	%lpc =l add %pc, 1
	storel %lpc, %pcslot
	jmp @loop
@jz
	%jsp =l loadl %spslot
	%ji =l sub %jsp, 1
	%jo =l mul %ji, 8
	%ja =l add %stack, %jo
	%jv =l loadl %ja
	storel %ji, %spslot
	%jpc1 =l add %pc, 1
	%joff =l mul %jpc1, 8
	%jaddr =l add %code, %joff
	%jt =l loadl %jaddr
	%jzero =w ceql %jv, 0
	jnz %jzero, @jztake, @jzskip
@jztake
	storel %jt, %pcslot
	jmp @loop
@jzskip
	%jpc2 =l add %pc, 2
	storel %jpc2, %pcslot
	jmp @loop
@dojmp
	%gpc1 =l add %pc, 1
	%goff =l mul %gpc1, 8
	%gaddr =l add %code, %goff
	%gt =l loadl %gaddr
	storel %gt, %pcslot
	jmp @loop
@finish
	%depth =l loadl %spslot
	jmp @zloop
@zloop
	%zi =l phi @finish %depth, @zstep %zi1
	%zdone =w csgel %zi, %cap
	jnz %zdone, @doret, @zstep
@zstep
	%zoff =l mul %zi, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%zi1 =l add %zi, 1
	jmp @zloop
@doret
	%empty =w ceql %depth, 0
	jnz %empty, @ret0, @rettop
@ret0
	ret 0
@rettop
	%topi =l sub %depth, 1
	%topoff =l mul %topi, 8
	%topaddr =l add %stack, %topoff
	%topv =l loadl %topaddr
	ret %topv
}
