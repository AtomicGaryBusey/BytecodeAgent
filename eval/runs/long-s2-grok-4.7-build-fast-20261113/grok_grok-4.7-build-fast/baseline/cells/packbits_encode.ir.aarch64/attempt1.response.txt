export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i0 =l copy 0
	%o0 =l copy 0
	jmp @outer
@outer
	%i =l phi @start %i0, @write_rep %i_rep, @lit_done %i_lit
	%o =l phi @start %o0, @write_rep %o_rep, @lit_done %o_lit
	%more =w csltl %i, %n
	jnz %more, @do_run, @done
@done
	ret %o
@do_run
	%sp =l add %src, %i
	%b =w loadub %sp
	%cnt0 =l copy 1
	%k0 =l add %i, 1
	jmp @runloop
@runloop
	%cnt =l phi @do_run %cnt0, @run_inc %cnt1
	%k =l phi @do_run %k0, @run_inc %k1
	%cntlt =w csltl %cnt, 128
	jnz %cntlt, @run_chk_n, @run_done
@run_chk_n
	%klt =w csltl %k, %n
	jnz %klt, @run_chk_eq, @run_done
@run_chk_eq
	%kp =l add %src, %k
	%bk =w loadub %kp
	%eq =w ceqw %bk, %b
	jnz %eq, @run_inc, @run_done
@run_inc
	%cnt1 =l add %cnt, 1
	%k1 =l add %k, 1
	jmp @runloop
@run_done
	%lt3 =w csltl %cnt, 3
	jnz %lt3, @start_lit, @emit_rep
@emit_rep
	%need =l add %o, 2
	%fit =w cslel %need, %cap
	jnz %fit, @write_rep, @fail
@write_rep
	%hdrl =l sub 257, %cnt
	%hdrw =w copy %hdrl
	%op =l add %out, %o
	storeb %hdrw, %op
	%o1 =l add %o, 1
	%op1 =l add %out, %o1
	storeb %b, %op1
	%o_rep =l add %o, 2
	%i_rep =l add %i, %cnt
	jmp @outer
@fail
	%neg1 =l copy -1
	ret %neg1
@start_lit
	%j0 =l copy %i
	jmp @jloop
@jloop
	%j =l phi @start_lit %j0, @j_adv %j1
	%jltn =w csltl %j, %n
	jnz %jltn, @j_chk_len, @lit_emit
@j_chk_len
	%diff =l sub %j, %i
	%dlt =w csltl %diff, 128
	jnz %dlt, @j_run, @lit_emit
@j_run
	%jp =l add %src, %j
	%jb =w loadub %jp
	%jcnt0 =l copy 1
	%jk0 =l add %j, 1
	jmp @jrunloop
@jrunloop
	%jcnt =l phi @j_run %jcnt0, @jrun_inc %jcnt1
	%jk =l phi @j_run %jk0, @jrun_inc %jk1
	%jclt =w csltl %jcnt, 128
	jnz %jclt, @jrun_chk_n, @jrun_done
@jrun_chk_n
	%jklt =w csltl %jk, %n
	jnz %jklt, @jrun_chk_eq, @jrun_done
@jrun_chk_eq
	%jkp =l add %src, %jk
	%jbk =w loadub %jkp
	%jeq =w ceqw %jbk, %jb
	jnz %jeq, @jrun_inc, @jrun_done
@jrun_inc
	%jcnt1 =l add %jcnt, 1
	%jk1 =l add %jk, 1
	jmp @jrunloop
@jrun_done
	%jlt3 =w csltl %jcnt, 3
	jnz %jlt3, @j_adv, @lit_emit
@j_adv
	%j1 =l add %j, 1
	jmp @jloop
@lit_emit
	%len =l sub %j, %i
	%sz =l add %len, 1
	%end =l add %o, %sz
	%lfit =w cslel %end, %cap
	jnz %lfit, @lit_write, @fail
@lit_write
	%hm1 =l sub %len, 1
	%hw =w copy %hm1
	%hp =l add %out, %o
	storeb %hw, %hp
	%c0 =l copy 0
	jmp @copyloop
@copyloop
	%c =l phi @lit_write %c0, @copy_step %c1
	%clt =w csltl %c, %len
	jnz %clt, @copy_step, @lit_done
@copy_step
	%si =l add %i, %c
	%scp =l add %src, %si
	%cb =w loadub %scp
	%di0 =l add %o, 1
	%di =l add %di0, %c
	%dp =l add %out, %di
	storeb %cb, %dp
	%c1 =l add %c, 1
	jmp @copyloop
@lit_done
	%o_lit =l add %o, %sz
	%i_lit =l copy %j
	jmp @outer
}
