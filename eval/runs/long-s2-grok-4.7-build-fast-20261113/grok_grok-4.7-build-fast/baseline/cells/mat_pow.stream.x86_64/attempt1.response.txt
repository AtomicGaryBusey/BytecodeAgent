mat_pow:
push rbx
push rbp
push r12
push r13
push r14
push r15
mov rbx, rsi
mov r14, rdx
mov r12, rcx
mov r13, r8
mov r15, rbx
imul r15, rbx
xor r8, r8
.Lcopy:
cmp r8, r15
jge .Lcopy_done
mov rax, qword ptr [rdi + r8*8]
mov qword ptr [r13 + r8*8], rax
inc r8
jmp .Lcopy
.Lcopy_done:
xor r8, r8
.Lzero_out:
cmp r8, r15
jge .Lzero_out_done
mov qword ptr [r12 + r8*8], 0
inc r8
jmp .Lzero_out
.Lzero_out_done:
xor r8, r8
.Lident:
cmp r8, rbx
jge .Lident_done
mov r9, r8
imul r9, rbx
add r9, r8
mov qword ptr [r12 + r9*8], 1
inc r8
jmp .Lident
.Lident_done:
test r14, r14
jz .Lfinish
.Lpow_loop:
test r14, 1
jz .Lskip_mul
mov rsi, r12
mov rdi, r13
mov rdx, r15
shl rdx, 3
add rdx, r13
lea r11, [rip + .Lret1]
jmp .Lmul
.Lret1:
mov r9, r15
shl r9, 3
add r9, r13
xor r8, r8
.Lcopy_out:
cmp r8, r15
jge .Lskip_mul
mov rax, qword ptr [r9 + r8*8]
mov qword ptr [r12 + r8*8], rax
inc r8
jmp .Lcopy_out
.Lskip_mul:
shr r14, 1
jz .Lfinish
mov rsi, r13
mov rdi, r13
mov rdx, r15
shl rdx, 3
add rdx, r13
lea r11, [rip + .Lret2]
jmp .Lmul
.Lret2:
mov r9, r15
shl r9, 3
add r9, r13
xor r8, r8
.Lcopy_base:
cmp r8, r15
jge .Lpow_loop
mov rax, qword ptr [r9 + r8*8]
mov qword ptr [r13 + r8*8], rax
inc r8
jmp .Lcopy_base
.Lfinish:
mov r9, r15
shl r9, 1
xor r8, r8
.Lzero_tmp:
cmp r8, r9
jge .Ldone
mov qword ptr [r13 + r8*8], 0
inc r8
jmp .Lzero_tmp
.Ldone:
xor rax, rax
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
.Lmul:
xor r8, r8
.Lmul_i:
cmp r8, rbx
jge .Lmul_done
xor r9, r9
.Lmul_j:
cmp r9, rbx
jge .Lmul_next_i
xor rax, rax
xor r10, r10
.Lmul_t:
cmp r10, rbx
jge .Lmul_store
mov rbp, r8
imul rbp, rbx
add rbp, r10
mov rcx, qword ptr [rsi + rbp*8]
mov rbp, r10
imul rbp, rbx
add rbp, r9
imul rcx, qword ptr [rdi + rbp*8]
add rax, rcx
inc r10
jmp .Lmul_t
.Lmul_store:
mov rbp, r8
imul rbp, rbx
add rbp, r9
mov qword ptr [rdx + rbp*8], rax
inc r9
jmp .Lmul_j
.Lmul_next_i:
inc r8
jmp .Lmul_i
.Lmul_done:
jmp r11
