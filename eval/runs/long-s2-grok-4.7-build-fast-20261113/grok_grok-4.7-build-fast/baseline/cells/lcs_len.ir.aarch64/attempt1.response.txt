export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @izero
@izero
	%iz =l phi @start 0, @izbody %izn
	%izc =w cslel %iz, %n
	jnz %izc, @izbody, @iout
@izbody
	%izo =l mul %iz, 8
	%izp =l add %row, %izo
	storel 0, %izp
	%izn =l add %iz, 1
	jmp @izero
@iout
	%i =l phi @izero 0, @inc %in
	%ic =w csltl %i, %m
	jnz %ic, @obody, @end
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @jin
@jin
	%j =l phi @obody 1, @jnext %jn
	%diag =l phi @obody 0, @jnext %dn
	%jc =w cslel %j, %n
	jnz %jc, @jbody, @inc
@jbody
	%jo =l mul %j, 8
	%jp =l add %row, %jo
	%up =l loadl %jp
	%jm =l sub %j, 1
	%jmo =l mul %jm, 8
	%jpl =l add %row, %jmo
	%left =l loadl %jpl
	%yp =l add %y, %jm
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @eqb, @neb
@eqb
	%nv =l add %diag, 1
	storel %nv, %jp
	jmp @jnext
@neb
	%gt =w csgtl %up, %left
	jnz %gt, @gtb, @leb
@gtb
	storel %up, %jp
	jmp @jnext
@leb
	storel %left, %jp
	jmp @jnext
@jnext
	%dn =l copy %up
	%jn =l add %j, 1
	jmp @jin
@inc
	%in =l add %i, 1
	jmp @iout
@end
	%no =l mul %n, 8
	%np =l add %row, %no
	%res =l loadl %np
	ret %res
}
