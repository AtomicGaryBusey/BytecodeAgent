export function l $base64_encode(l %src, l %n, l %out) {
@start
	%iz =l copy 0
	%oz =l copy 0
	%c1 =l copy 1
	%c2 =l copy 2
	%c3 =l copy 3
	%c4 =l copy 4
	%c5 =l copy 5
	%c6 =l copy 6
	%c26 =l copy 26
	%c52 =l copy 52
	%c62 =l copy 62
	%c63 =l copy 63
	%c65 =l copy 65
	%c71 =l copy 71
	%c43 =l copy 43
	%c47 =l copy 47
	%c61 =l copy 61
	%w6 =w copy 6
	%w8 =w copy 8
	%w12 =w copy 12
	%w16 =w copy 16
	%w18 =w copy 18
	%wz =w copy 0
	jmp @loop
@loop
	%i =l phi @start %iz, @bodyend %inext
	%o =l phi @start %oz, @bodyend %onext
	%ip3 =l add %i, %c3
	%more =w cslel %ip3, %n
	jnz %more, @body, @rem
@body
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%i1 =l add %i, %c1
	%p1 =l add %src, %i1
	%b1 =l loadub %p1
	%i2 =l add %i, %c2
	%p2 =l add %src, %i2
	%b2 =l loadub %p2
	%b0s =l shl %b0, %w16
	%b1s =l shl %b1, %w8
	%or01 =l or %b0s, %b1s
	%val =l or %or01, %b2
	jmp @emit
@emit
	%k =l phi @body %iz, @emitnext %k1
	%s =w phi @body %w18, @emitnext %s1
	%shifted =l shr %val, %s
	%ev =l and %shifted, %c63
	%off =l add %o, %k
	%edst =l add %out, %off
	jmp @map
@map
	%v =l phi @emit %ev, @one %ov0, @oneb %oneb_v, @two %tv0, @twob %twob_v, @twoc %twoc_v
	%dst =l phi @emit %edst, @one %odst0, @oneb %oneb_dst, @two %tdst0, @twob %twob_dst, @twoc %twoc_dst
	%cont =l phi @emit %c1, @one %c2, @oneb %c3, @two %c4, @twob %c5, @twoc %c6
	%nv1 =l phi @emit %iz, @one %ov1, @oneb %iz, @two %tv1, @twob %twob_nv, @twoc %iz
	%nv2 =l phi @emit %iz, @one %iz, @oneb %iz, @two %tv2, @twob %iz, @twoc %iz
	%fk =l phi @emit %k, @one %iz, @oneb %iz, @two %iz, @twob %iz, @twoc %iz
	%fs =w phi @emit %s, @one %wz, @oneb %wz, @two %wz, @twob %wz, @twoc %wz
	%lt26 =w csltl %v, %c26
	%lt52 =w csltl %v, %c52
	%lt62 =w csltl %v, %c62
	%eq62 =w ceql %v, %c62
	%u =l extuw %lt26
	%p52 =l extuw %lt52
	%p62 =l extuw %lt62
	%e62 =l extuw %eq62
	%islow =l sub %p52, %u
	%isdig =l sub %p62, %p52
	%ge62 =l sub %c1, %p62
	%issl =l sub %ge62, %e62
	%cu =l add %v, %c65
	%cl =l add %v, %c71
	%cd =l sub %v, %c4
	%mu =l mul %u, %cu
	%ml =l mul %islow, %cl
	%md =l mul %isdig, %cd
	%mp =l mul %e62, %c43
	%ms =l mul %issl, %c47
	%t1 =l add %mu, %ml
	%t2 =l add %t1, %md
	%t3 =l add %t2, %mp
	%ch =l add %t3, %ms
	storeb %ch, %dst
	%c_is1 =w ceql %cont, %c1
	jnz %c_is1, @emitnext, @dc2
@dc2
	%c_is2 =w ceql %cont, %c2
	jnz %c_is2, @oneb, @dc3
@dc3
	%c_is3 =w ceql %cont, %c3
	jnz %c_is3, @onepad, @dc4
@dc4
	%c_is4 =w ceql %cont, %c4
	jnz %c_is4, @twob, @dc5
@dc5
	%c_is5 =w ceql %cont, %c5
	jnz %c_is5, @twoc, @twopad
@emitnext
	%k1 =l add %fk, %c1
	%s1 =w sub %fs, %w6
	%edone =w ceql %k1, %c4
	jnz %edone, @bodyend, @emit
@bodyend
	%inext =l add %i, %c3
	%onext =l add %o, %c4
	jmp @loop
@rem
	%left =l sub %n, %i
	%is1 =w ceql %left, %c1
	jnz %is1, @one, @chk2
@chk2
	%is2 =w ceql %left, %c2
	jnz %is2, @two, @done
@done
	ret %o
@one
	%op0 =l add %src, %i
	%ob0 =l loadub %op0
	%oval =l shl %ob0, %w16
	%osh0 =l shr %oval, %w18
	%ov0 =l and %osh0, %c63
	%osh1 =l shr %oval, %w12
	%ov1 =l and %osh1, %c63
	%odst0 =l add %out, %o
	jmp @map
@oneb
	%oneb_v =l copy %nv1
	%oneb_off =l add %o, %c1
	%oneb_dst =l add %out, %oneb_off
	jmp @map
@onepad
	%op2 =l add %o, %c2
	%od2 =l add %out, %op2
	storeb %c61, %od2
	%op3 =l add %o, %c3
	%od3 =l add %out, %op3
	storeb %c61, %od3
	%ret1 =l add %o, %c4
	ret %ret1
@two
	%tp0 =l add %src, %i
	%tb0 =l loadub %tp0
	%ti1 =l add %i, %c1
	%tp1 =l add %src, %ti1
	%tb1 =l loadub %tp1
	%tb0s =l shl %tb0, %w16
	%tb1s =l shl %tb1, %w8
	%tval =l or %tb0s, %tb1s
	%tsh0 =l shr %tval, %w18
	%tv0 =l and %tsh0, %c63
	%tsh1 =l shr %tval, %w12
	%tv1 =l and %tsh1, %c63
	%tsh2 =l shr %tval, %w6
	%tv2 =l and %tsh2, %c63
	%tdst0 =l add %out, %o
	jmp @map
@twob
	%twob_v =l copy %nv1
	%twob_nv =l copy %nv2
	%twob_off =l add %o, %c1
	%twob_dst =l add %out, %twob_off
	jmp @map
@twoc
	%twoc_v =l copy %nv1
	%twoc_off =l add %o, %c2
	%twoc_dst =l add %out, %twoc_off
	jmp @map
@twopad
	%tpoff =l add %o, %c3
	%tpd =l add %out, %tpoff
	storeb %c61, %tpd
	%ret2 =l add %o, %c4
	ret %ret2
}
