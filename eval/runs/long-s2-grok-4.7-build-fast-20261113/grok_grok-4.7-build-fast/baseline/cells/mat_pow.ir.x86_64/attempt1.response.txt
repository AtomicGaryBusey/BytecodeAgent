export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%nn8 =l mul %nn, 8
	%prod =l add %tmp, %nn8
	%zero =l copy 0
	%kzero =w ceql %k, 0
	jnz %kzero, @id_i, @cp_loop
@cp_loop
	%ci =l phi @start 0, @cp_body %ci1
	%cidone =w ceql %ci, %nn
	jnz %cidone, @pow_init, @cp_body
@cp_body
	%ci8 =l mul %ci, 8
	%cap =l add %a, %ci8
	%cop =l add %out, %ci8
	%cav =l loadl %cap
	storel %cav, %cop
	%ci1 =l add %ci, 1
	jmp @cp_loop
@pow_init
	%rem0 =l sub %k, 1
	jmp @pow_loop
@pow_loop
	%rem =l phi @pow_init %rem0, @pow_dec %rem1
	%remdone =w ceql %rem, 0
	jnz %remdone, @zero_init, @mm_i
@mm_i
	%i =l phi @pow_loop 0, @mm_i_inc %i1
	%idone =w ceql %i, %n
	jnz %idone, @pb, @mm_j
@mm_j
	%j =l phi @mm_i 0, @mm_j_inc %j1
	%jdone =w ceql %j, %n
	jnz %jdone, @mm_i_inc, @mm_t
@mm_t
	%t =l phi @mm_j 0, @mm_t_body %t1
	%s =l phi @mm_j 0, @mm_t_body %s1
	%tdone =w ceql %t, %n
	jnz %tdone, @mm_store, @mm_t_body
@mm_t_body
	%in =l mul %i, %n
	%outi =l add %in, %t
	%outoff =l mul %outi, 8
	%outp =l add %out, %outoff
	%ov =l loadl %outp
	%tn =l mul %t, %n
	%ai =l add %tn, %j
	%aoff =l mul %ai, 8
	%ap =l add %a, %aoff
	%av =l loadl %ap
	%ml =l mul %ov, %av
	%s1 =l add %s, %ml
	%t1 =l add %t, 1
	jmp @mm_t
@mm_store
	%in2 =l mul %i, %n
	%pij =l add %in2, %j
	%poff =l mul %pij, 8
	%pp =l add %prod, %poff
	storel %s, %pp
	jmp @mm_j_inc
@mm_j_inc
	%j1 =l add %j, 1
	jmp @mm_j
@mm_i_inc
	%i1 =l add %i, 1
	jmp @mm_i
@pb
	%pi =l phi @mm_i 0, @pb_body %pi1
	%pidone =w ceql %pi, %nn
	jnz %pidone, @pow_dec, @pb_body
@pb_body
	%pi8 =l mul %pi, 8
	%psp =l add %prod, %pi8
	%pdp =l add %out, %pi8
	%pv =l loadl %psp
	storel %pv, %pdp
	%pi1 =l add %pi, 1
	jmp @pb
@pow_dec
	%rem1 =l sub %rem, 1
	jmp @pow_loop
@id_i
	%ii =l phi @start 0, @id_i_inc %ii1
	%iidone =w ceql %ii, %n
	jnz %iidone, @zero_init, @id_j
@id_j
	%jj =l phi @id_i 0, @id_j_inc %jj1
	%jjdone =w ceql %jj, %n
	jnz %jjdone, @id_i_inc, @id_body
@id_body
	%eq =w ceql %ii, %jj
	%ival =l extuw %eq
	%iin =l mul %ii, %n
	%iij =l add %iin, %jj
	%iioff =l mul %iij, 8
	%iip =l add %out, %iioff
	storel %ival, %iip
	jmp @id_j_inc
@id_j_inc
	%jj1 =l add %jj, 1
	jmp @id_j
@id_i_inc
	%ii1 =l add %ii, 1
	jmp @id_i
@zero_init
	%nz =l mul %nn, 2
	jmp @zero
@zero
	%zi =l phi @zero_init 0, @zero_body %zi1
	%zdone =w ceql %zi, %nz
	jnz %zdone, @end, @zero_body
@zero_body
	%zi8 =l mul %zi, 8
	%zp =l add %tmp, %zi8
	storel %zero, %zp
	%zi1 =l add %zi, 1
	jmp @zero
@end
	ret %zero
}
