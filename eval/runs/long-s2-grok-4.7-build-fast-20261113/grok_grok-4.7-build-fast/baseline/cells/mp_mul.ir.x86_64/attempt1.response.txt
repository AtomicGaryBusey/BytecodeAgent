export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%zero =l copy 0
	%one =l copy 1
	%two32 =l shl %one, 32
	%mask =l sub %two32, %one
	%n =l add %na, %nb
	jmp @zhead
@zhead
	%zi =l phi @start %zero, @zbody %zi1
	%zc =w csltl %zi, %n
	jnz %zc, @zbody, @zdone
@zbody
	%zoff =l shl %zi, 3
	%zp =l add %r, %zoff
	storel %zero, %zp
	%zi1 =l add %zi, 1
	jmp @zhead
@zdone
	%naz =w ceql %na, 0
	jnz %naz, @r0, @cknb
@cknb
	%nbz =w ceql %nb, 0
	jnz %nbz, @r0, @minit
@r0
	ret %zero
@minit
	%i0 =l copy 0
	jmp @ihead
@ihead
	%i =l phi @minit %i0, @jdone %i1
	%ic =w csltl %i, %na
	jnz %ic, @ibody, @sinit
@ibody
	%ioff =l shl %i, 3
	%ap =l add %a, %ioff
	%ai =l loadl %ap
	%j0 =l copy 0
	%c0 =l copy 0
	jmp @jhead
@jhead
	%j =l phi @ibody %j0, @jbody %j1
	%carry =l phi @ibody %c0, @jbody %cnext
	%jc =w csltl %j, %nb
	jnz %jc, @jbody, @jdone
@jbody
	%joff =l shl %j, 3
	%bp =l add %b, %joff
	%bj =l loadl %bp
	%ij =l add %i, %j
	%ijoff =l shl %ij, 3
	%rp =l add %r, %ijoff
	%rij =l loadl %rp
	%a0 =l and %ai, %mask
	%b0 =l and %bj, %mask
	%a1 =l shr %ai, 32
	%b1 =l shr %bj, 32
	%p00 =l mul %a0, %b0
	%p01 =l mul %a0, %b1
	%p10 =l mul %a1, %b0
	%p11 =l mul %a1, %b1
	%mid =l add %p01, %p10
	%midcw =w cultl %mid, %p01
	%midc =l extuw %midcw
	%ml32 =l and %mid, %mask
	%mlsh =l shl %ml32, 32
	%lo =l add %p00, %mlsh
	%locw =w cultl %lo, %p00
	%loc =l extuw %locw
	%mtop =l shr %mid, 32
	%mcsh =l shl %midc, 32
	%hi0 =l add %p11, %mtop
	%hi1 =l add %hi0, %mcsh
	%hi =l add %hi1, %loc
	%s1 =l add %lo, %rij
	%c1w =w cultl %s1, %lo
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	storel %s2, %rp
	%hs =l add %hi, %c1
	%cnext =l add %hs, %c2
	%j1 =l add %j, 1
	jmp @jhead
@jdone
	%inb =l add %i, %nb
	%inbo =l shl %inb, 3
	%rp2 =l add %r, %inbo
	storel %carry, %rp2
	%i1 =l add %i, 1
	jmp @ihead
@sinit
	%k0 =l copy %n
	jmp @shead
@shead
	%k =l phi @sinit %k0, @sdec %k1
	%kg =w csgtl %k, 0
	jnz %kg, @sbody, @r0
@sbody
	%km1 =l sub %k, 1
	%ko =l shl %km1, 3
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%kz =w ceql %kv, 0
	jnz %kz, @sdec, @rk
@sdec
	%k1 =l copy %km1
	jmp @shead
@rk
	ret %k
}
