qsort_pairs:
cmp rsi, 1
jg .Lqs_start
xor eax, eax
ret
.Lqs_start:
push rbx
push r12
push r13
mov rbx, rdi
mov r12, rsi
mov rax, r12
dec rax
shl rax, 4
mov r8, qword ptr [rbx + rax]
mov r9, qword ptr [rbx + rax + 8]
xor r10d, r10d
xor r11d, r11d
.Lpart_loop:
mov rax, r12
dec rax
cmp r11, rax
jge .Lpart_place
mov rax, r11
shl rax, 4
mov rcx, qword ptr [rbx + rax]
mov rdx, qword ptr [rbx + rax + 8]
cmp rcx, r8
jl .Lpart_less
jg .Lpart_next
cmp rdx, r9
jl .Lpart_less
jmp .Lpart_next
.Lpart_less:
mov rsi, r10
shl rsi, 4
mov rdi, qword ptr [rbx + rsi]
mov qword ptr [rbx + rsi], rcx
mov qword ptr [rbx + rax], rdi
mov rdi, qword ptr [rbx + rsi + 8]
mov qword ptr [rbx + rsi + 8], rdx
mov qword ptr [rbx + rax + 8], rdi
inc r10
.Lpart_next:
inc r11
jmp .Lpart_loop
.Lpart_place:
mov rax, r12
dec rax
shl rax, 4
mov rsi, r10
shl rsi, 4
mov rcx, qword ptr [rbx + rsi]
mov rdx, qword ptr [rbx + rax]
mov qword ptr [rbx + rsi], rdx
mov qword ptr [rbx + rax], rcx
mov rcx, qword ptr [rbx + rsi + 8]
mov rdx, qword ptr [rbx + rax + 8]
mov qword ptr [rbx + rsi + 8], rdx
mov qword ptr [rbx + rax + 8], rcx
mov r13, r10
mov rdi, rbx
mov rsi, r13
call qsort_pairs
lea rdi, [r13 + 1]
shl rdi, 4
add rdi, rbx
mov rsi, r12
sub rsi, r13
dec rsi
call qsort_pairs
xor eax, eax
pop r13
pop r12
pop rbx
ret
