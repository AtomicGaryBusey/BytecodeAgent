export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @part
@done
	%z =l copy 0
	ret %z
@part
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%paddr =l add %a, %poff
	%pkey =l loadl %paddr
	%pvaddr =l add %paddr, 8
	%pval =l loadl %pvaddr
	jmp @loop
@loop
	%j =l phi @part 0, @incj %jnext
	%i =l phi @part 0, @incj %inext
	%jdone =w csgel %j, %nm1
	jnz %jdone, @finish, @body
@body
	%joff =l mul %j, 16
	%jaddr =l add %a, %joff
	%jkey =l loadl %jaddr
	%jvaddr =l add %jaddr, 8
	%jval =l loadl %jvaddr
	%klt =w csltl %jkey, %pkey
	jnz %klt, @doswap, @checkgt
@checkgt
	%kgt =w csgtl %jkey, %pkey
	jnz %kgt, @noswap, @checkval
@checkval
	%vle =w cslel %jval, %pval
	jnz %vle, @doswap, @noswap
@doswap
	%ioff =l mul %i, 16
	%iaddr =l add %a, %ioff
	%ikey =l loadl %iaddr
	%ivaddr =l add %iaddr, 8
	%ival =l loadl %ivaddr
	storel %jkey, %iaddr
	storel %jval, %ivaddr
	storel %ikey, %jaddr
	storel %ival, %jvaddr
	%i1 =l add %i, 1
	jmp @incj
@noswap
	jmp @incj
@incj
	%inext =l phi @doswap %i1, @noswap %i
	%jnext =l add %j, 1
	jmp @loop
@finish
	%ioff2 =l mul %i, 16
	%iaddr2 =l add %a, %ioff2
	%ikey2 =l loadl %iaddr2
	%ivaddr2 =l add %iaddr2, 8
	%ival2 =l loadl %ivaddr2
	storel %pkey, %iaddr2
	storel %pval, %ivaddr2
	storel %ikey2, %paddr
	storel %ival2, %pvaddr
	%left =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%roff =l mul %ip1, 16
	%rptr =l add %a, %roff
	%rcount =l sub %n, %ip1
	%right =l call $qsort_pairs(l %rptr, l %rcount)
	%z2 =l copy 0
	ret %z2
}
