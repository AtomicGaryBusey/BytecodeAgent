packbits_encode:
xor r8, r8
xor r9, r9
.Lloop:
cmp r8, rsi
jge .Ldone
mov r10, r8
xor r11, r11
.Lrun_i:
cmp r11, 128
jge .Lrun_i_end
lea rax, [r10 + r11]
cmp rax, rsi
jge .Lrun_i_end
movzx eax, byte ptr [rdi + rax]
cmp al, byte ptr [rdi + r10]
jne .Lrun_i_end
inc r11
jmp .Lrun_i
.Lrun_i_end:
cmp r11, 3
jl .Lliteral
mov rax, rcx
sub rax, r9
cmp rax, 2
jl .Lfail
mov eax, r11d
dec eax
neg al
mov byte ptr [rdx + r9], al
movzx eax, byte ptr [rdi + r8]
mov byte ptr [rdx + r9 + 1], al
add r9, 2
add r8, r11
jmp .Lloop
.Lliteral:
mov r10, r8
.Llit_scan:
cmp r10, rsi
jge .Llit_emit
mov rax, r10
sub rax, r8
cmp rax, 128
jge .Llit_emit
xor r11, r11
.Lrun_j:
cmp r11, 128
jge .Lrun_j_end
lea rax, [r10 + r11]
cmp rax, rsi
jge .Lrun_j_end
movzx eax, byte ptr [rdi + rax]
cmp al, byte ptr [rdi + r10]
jne .Lrun_j_end
inc r11
jmp .Lrun_j
.Lrun_j_end:
cmp r11, 3
jge .Llit_emit
inc r10
jmp .Llit_scan
.Llit_emit:
mov r11, r10
sub r11, r8
mov rax, r9
add rax, r11
inc rax
cmp rax, rcx
jg .Lfail
mov eax, r11d
dec eax
mov byte ptr [rdx + r9], al
inc r9
.Lcopy:
test r11, r11
jz .Lloop
mov al, byte ptr [rdi + r8]
mov byte ptr [rdx + r9], al
inc r8
inc r9
dec r11
jmp .Lcopy
.Ldone:
mov rax, r9
ret
.Lfail:
mov rax, -1
ret
