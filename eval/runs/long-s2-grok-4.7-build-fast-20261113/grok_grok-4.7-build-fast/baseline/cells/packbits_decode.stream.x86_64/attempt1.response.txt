packbits_decode:
    lea r8, [rdi + rsi]
    lea r9, [rdx + rcx]
    mov r10, rdx
.Lnext:
    cmp rdi, r8
    jae .Lsuccess
    movzx eax, byte ptr [rdi]
    add rdi, 1
    cmp eax, 128
    je .Lnext
    ja .Lrepeat
    add eax, 1
    mov r11d, eax
    mov rcx, r8
    sub rcx, rdi
    cmp rcx, r11
    jb .Lsrc_err
    mov rcx, r9
    sub rcx, rdx
    cmp rcx, r11
    jb .Llit_partial
.Llit_full:
    movzx eax, byte ptr [rdi]
    mov byte ptr [rdx], al
    add rdi, 1
    add rdx, 1
    sub r11, 1
    jnz .Llit_full
    jmp .Lnext
.Llit_partial:
    test rcx, rcx
    jz .Loverflow
.Llit_part_loop:
    movzx eax, byte ptr [rdi]
    mov byte ptr [rdx], al
    add rdi, 1
    add rdx, 1
    sub rcx, 1
    jnz .Llit_part_loop
    jmp .Loverflow
.Lrepeat:
    mov r11d, 257
    sub r11d, eax
    cmp rdi, r8
    jae .Lsrc_err
    movzx eax, byte ptr [rdi]
    add rdi, 1
    mov rcx, r9
    sub rcx, rdx
    cmp rcx, r11
    jb .Lrep_partial
.Lrep_full:
    mov byte ptr [rdx], al
    add rdx, 1
    sub r11, 1
    jnz .Lrep_full
    jmp .Lnext
.Lrep_partial:
    test rcx, rcx
    jz .Loverflow
.Lrep_part_loop:
    mov byte ptr [rdx], al
    add rdx, 1
    sub rcx, 1
    jnz .Lrep_part_loop
.Loverflow:
    mov rax, -1
    ret
.Lsuccess:
    mov rax, rdx
    sub rax, r10
    ret
.Lsrc_err:
    mov rax, -2
    ret
