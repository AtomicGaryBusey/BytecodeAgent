lcs_len:
push rbx
push r12
push r13
xor rax, rax
.Lzero:
mov qword ptr [r8 + rax*8], 0
inc rax
cmp rax, rcx
jle .Lzero
xor r9, r9
cmp r9, rsi
jge .Ldone
.Louter:
movzx ebx, byte ptr [rdi + r9]
xor r11, r11
mov r10, 1
cmp r10, rcx
jg .Lnexti
.Linner:
mov r12, qword ptr [r8 + r10*8]
mov r13, qword ptr [r8 + r10*8 - 8]
movzx eax, byte ptr [rdx + r10 - 1]
cmp eax, ebx
jne .Lno
lea rax, [r11 + 1]
mov qword ptr [r8 + r10*8], rax
jmp .Lsetd
.Lno:
cmp r12, r13
cmovg r13, r12
mov qword ptr [r8 + r10*8], r13
.Lsetd:
mov r11, r12
inc r10
cmp r10, rcx
jle .Linner
.Lnexti:
inc r9
cmp r9, rsi
jl .Louter
.Ldone:
mov rax, qword ptr [r8 + rcx*8]
pop r13
pop r12
pop rbx
ret
