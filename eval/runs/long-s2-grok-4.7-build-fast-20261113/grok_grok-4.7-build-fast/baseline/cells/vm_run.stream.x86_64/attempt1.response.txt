vm_run:
    push rbx
    push rbp
    xor ebx, ebx
    xor ebp, ebp
.Lloop:
    cmp rbx, rsi
    jae .Ldone
    mov rax, qword ptr [rdi + rbx*8]
    cmp rax, 1
    jb .Ldone
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    inc rbx
    jmp .Lloop
.Lpush:
    mov rax, qword ptr [rdi + rbx*8 + 8]
    mov qword ptr [rdx + rbp*8], rax
    inc rbp
    add rbx, 2
    jmp .Lloop
.Ladd:
    mov rax, qword ptr [rdx + rbp*8 - 8]
    add qword ptr [rdx + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Lsub:
    mov rax, qword ptr [rdx + rbp*8 - 8]
    sub qword ptr [rdx + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Lmul:
    mov rax, qword ptr [rdx + rbp*8 - 16]
    imul rax, qword ptr [rdx + rbp*8 - 8]
    mov qword ptr [rdx + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Ldup:
    mov rax, qword ptr [rdx + rbp*8 - 8]
    mov qword ptr [rdx + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Lswap:
    mov rax, qword ptr [rdx + rbp*8 - 8]
    mov r8, qword ptr [rdx + rbp*8 - 16]
    mov qword ptr [rdx + rbp*8 - 8], r8
    mov qword ptr [rdx + rbp*8 - 16], rax
    inc rbx
    jmp .Lloop
.Ldrop:
    dec rbp
    inc rbx
    jmp .Lloop
.Lover:
    mov rax, qword ptr [rdx + rbp*8 - 16]
    mov qword ptr [rdx + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Llt:
    mov rax, qword ptr [rdx + rbp*8 - 16]
    cmp rax, qword ptr [rdx + rbp*8 - 8]
    setl al
    movzx rax, al
    mov qword ptr [rdx + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Ljz:
    dec rbp
    mov r8, qword ptr [rdx + rbp*8]
    mov r9, qword ptr [rdi + rbx*8 + 8]
    add rbx, 2
    test r8, r8
    jnz .Lloop
    mov rbx, r9
    jmp .Lloop
.Ljmp:
    mov rbx, qword ptr [rdi + rbx*8 + 8]
    jmp .Lloop
.Ldone:
    xor eax, eax
    test rbp, rbp
    jz .Lzero
    mov rax, qword ptr [rdx + rbp*8 - 8]
.Lzero:
    cmp rbp, rcx
    jge .Lret
    mov qword ptr [rdx + rbp*8], 0
    inc rbp
    jmp .Lzero
.Lret:
    pop rbp
    pop rbx
    ret
