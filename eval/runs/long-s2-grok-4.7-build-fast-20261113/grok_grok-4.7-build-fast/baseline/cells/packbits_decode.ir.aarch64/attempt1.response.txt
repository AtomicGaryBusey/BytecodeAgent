export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%si =l phi @start 0, @nop %si_nop, @litdone %si_lit, @repdone %si_rep
	%oi =l phi @start 0, @nop %oi_nop, @litdone %oi_lit, @repdone %oi_rep
	%more =w csltl %si, %n
	jnz %more, @readh, @done
@done
	ret %oi
@readh
	%hp =l add %src, %si
	%h =w loadsb %hp
	%si1 =l add %si, 1
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @checklit
@nop
	%si_nop =l copy %si1
	%oi_nop =l copy %oi
	jmp @loop
@checklit
	%isneg =w csltw %h, 0
	jnz %isneg, @repeat, @literal
@literal
	%hp1 =w add %h, 1
	%lcnt0 =l extuw %hp1
	%remain =l sub %n, %si1
	%srcok =w cslel %lcnt0, %remain
	jnz %srcok, @litinit, @errsrc
@litinit
	%si_li =l copy %si1
	%oi_li =l copy %oi
	%lc_li =l copy %lcnt0
	jmp @litloop
@litloop
	%si_l =l phi @litinit %si_li, @litwr %si_l2
	%oi_l =l phi @litinit %oi_li, @litwr %oi_l2
	%lc =l phi @litinit %lc_li, @litwr %lc2
	%cdone =w ceql %lc, 0
	jnz %cdone, @litdone, @litbyte
@litdone
	%si_lit =l copy %si_l
	%oi_lit =l copy %oi_l
	jmp @loop
@litbyte
	%capok =w csltl %oi_l, %cap
	jnz %capok, @litwr, @errcap
@litwr
	%sp =l add %src, %si_l
	%bv =w loadub %sp
	%dp =l add %out, %oi_l
	storeb %bv, %dp
	%si_l2 =l add %si_l, 1
	%oi_l2 =l add %oi_l, 1
	%lc2 =l sub %lc, 1
	jmp @litloop
@repeat
	%hasb =w csltl %si1, %n
	jnz %hasb, @repinit, @errsrc
@repinit
	%rp =l add %src, %si1
	%rb =w loadub %rp
	%si_r0 =l add %si1, 1
	%hs =l extsw %h
	%rcnt0 =l sub 1, %hs
	%oi_ri =l copy %oi
	jmp @reploop
@reploop
	%oi_r =l phi @repinit %oi_ri, @repwr %oi_r2
	%rc =l phi @repinit %rcnt0, @repwr %rc2
	%rdone =w ceql %rc, 0
	jnz %rdone, @repdone, @repwrchk
@repdone
	%si_rep =l copy %si_r0
	%oi_rep =l copy %oi_r
	jmp @loop
@repwrchk
	%capok2 =w csltl %oi_r, %cap
	jnz %capok2, @repwr, @errcap
@repwr
	%dp2 =l add %out, %oi_r
	storeb %rb, %dp2
	%oi_r2 =l add %oi_r, 1
	%rc2 =l sub %rc, 1
	jmp @reploop
@errsrc
	ret -2
@errcap
	ret -1
}
