export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%iz =l copy 0
	jmp @zcond
@zcond
	%zi =l phi @start %iz, @zbody %zi1
	%zc =w cslel %zi, %n
	jnz %zc, @zbody, @zdone
@zbody
	%zo =l mul %zi, 8
	%za =l add %row, %zo
	storel 0, %za
	%zi1 =l add %zi, 1
	jmp @zcond
@zdone
	%i0 =l copy 0
	jmp @ocond
@ocond
	%i =l phi @zdone %i0, @oinc %i1
	%oc =w csltl %i, %m
	jnz %oc, @obody, @done
@obody
	%xp =l add %x, %i
	%xb =w loadub %xp
	%d0 =l copy 0
	%j0 =l copy 1
	jmp @icond
@icond
	%j =l phi @obody %j0, @istore %j1
	%diag =l phi @obody %d0, @istore %dup
	%jc =w cslel %j, %n
	jnz %jc, @ibody, @oinc
@ibody
	%jo =l mul %j, 8
	%ja =l add %row, %jo
	%up =l loadl %ja
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%nvm =l add %diag, 1
	jmp @istore
@nomatch
	%jmo =l mul %jm1, 8
	%jma =l add %row, %jmo
	%left =l loadl %jma
	%gt =w csgtl %up, %left
	jnz %gt, @takeup, @takeleft
@takeup
	%nvu =l copy %up
	jmp @istore
@takeleft
	%nvl =l copy %left
	jmp @istore
@istore
	%nv =l phi @match %nvm, @takeup %nvu, @takeleft %nvl
	storel %nv, %ja
	%j1 =l add %j, 1
	%dup =l copy %up
	jmp @icond
@oinc
	%i1 =l add %i, 1
	jmp @ocond
@done
	%no =l mul %n, 8
	%na =l add %row, %no
	%res =l loadl %na
	ret %res
}
