export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @nop %i_nop, @litdone %i_lit, @repdone %i_rep
	%w =l phi @start 0, @nop %w_nop, @litdone %w_lit, @repdone %w_rep
	%cont =w csltl %i, %n
	jnz %cont, @readh, @done
@done
	ret %w
@readh
	%hp =l add %src, %i
	%hs =w loadsb %hp
	%i1 =l add %i, 1
	%hcmp =w copy -128
	%isnop =w ceqw %hs, %hcmp
	jnz %isnop, @nop, @chkpos
@nop
	%i_nop =l copy %i1
	%w_nop =l copy %w
	jmp @loop
@chkpos
	%ispos =w csgew %hs, 0
	jnz %ispos, @literal, @repeat
@literal
	%cntw =w add %hs, 1
	%count =l extuw %cntw
	%iend =l add %i1, %count
	%sok =w cslel %iend, %n
	jnz %sok, @litloop, @badsrc
@litloop
	%j =l phi @literal 0, @litst %j1
	%lw =l phi @literal %w, @litst %lw1
	%li =l phi @literal %i1, @litst %li1
	%jmore =w csltl %j, %count
	jnz %jmore, @litcap, @litdone
@litcap
	%lroom =w csltl %lw, %cap
	jnz %lroom, @litst, @badcap
@litst
	%sp =l add %src, %li
	%bv =w loadub %sp
	%dp =l add %out, %lw
	storeb %bv, %dp
	%lw1 =l add %lw, 1
	%li1 =l add %li, 1
	%j1 =l add %j, 1
	jmp @litloop
@litdone
	%i_lit =l copy %li
	%w_lit =l copy %lw
	jmp @loop
@repeat
	%has =w csltl %i1, %n
	jnz %has, @repload, @badsrc
@repload
	%rp =l add %src, %i1
	%rb =w loadub %rp
	%ribase =l add %i1, 1
	%one =w copy 1
	%rcw =w sub %one, %hs
	%rcount =l extuw %rcw
	jmp @reploop
@reploop
	%k =l phi @repload 0, @repst %k1
	%rw =l phi @repload %w, @repst %rw1
	%kmore =w csltl %k, %rcount
	jnz %kmore, @repcap, @repdone
@repcap
	%rroom =w csltl %rw, %cap
	jnz %rroom, @repst, @badcap
@repst
	%rdp =l add %out, %rw
	storeb %rb, %rdp
	%rw1 =l add %rw, 1
	%k1 =l add %k, 1
	jmp @reploop
@repdone
	%i_rep =l copy %ribase
	%w_rep =l copy %rw
	jmp @loop
@badsrc
	%em2 =l copy -2
	ret %em2
@badcap
	%em1 =l copy -1
	ret %em1
}
