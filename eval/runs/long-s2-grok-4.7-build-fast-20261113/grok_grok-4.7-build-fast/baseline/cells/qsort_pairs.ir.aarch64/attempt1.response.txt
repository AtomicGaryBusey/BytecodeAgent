export function l $qsort_pairs(l %a, l %n) {
@start
	%le1 =w cslel %n, 1
	jnz %le1, @ret0, @body
@ret0
	ret 0
@body
	%nm1 =l sub %n, 1
	%poff =l mul %nm1, 16
	%pp =l add %a, %poff
	%pk =l loadl %pp
	%pvaddr =l add %pp, 8
	%pv =l loadl %pvaddr
	jmp @part
@part
	%i =l phi @body 0, @incj %inext
	%j =l phi @body 0, @incj %jnext
	%jlt =w csltl %j, %nm1
	jnz %jlt, @docmp, @partdone
@docmp
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jvaddr =l add %jp, 8
	%jv =l loadl %jvaddr
	%klt =w csltl %jk, %pk
	jnz %klt, @less, @checkeq
@checkeq
	%keq =w ceql %jk, %pk
	jnz %keq, @cmpval, @incj
@cmpval
	%vlt =w csltl %jv, %pv
	jnz %vlt, @less, @incj
@less
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivaddr =l add %ip, 8
	%iv =l loadl %ivaddr
	storel %jk, %ip
	storel %jv, %ivaddr
	storel %ik, %jp
	storel %iv, %jvaddr
	%iplus =l add %i, 1
	jmp @incj
@incj
	%inext =l phi @less %iplus, @checkeq %i, @cmpval %i
	%jnext =l add %j, 1
	jmp @part
@partdone
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ivaddr2 =l add %ip2, 8
	%iv2 =l loadl %ivaddr2
	storel %pk, %ip2
	storel %pv, %ivaddr2
	storel %ik2, %pp
	storel %iv2, %pvaddr
	%left =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%rightn =l sub %n, %ip1
	%roff =l mul %ip1, 16
	%rp =l add %a, %roff
	%right =l call $qsort_pairs(l %rp, l %rightn)
	%sum =l add %left, %right
	ret %sum
}
