qsort_pairs:
    cmp x1, #1
    b.le .Lret0
    stp x29, x30, [sp, #-16]!
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    mov x19, x0
    mov x20, x1
.Louter:
    sub x15, x20, #1
    add x10, x19, x15, lsl #4
    ldp x11, x12, [x10]
    mov x13, #0
    mov x14, #0
.Lscan:
    cmp x14, x15
    b.ge .Lfinish
    add x10, x19, x14, lsl #4
    ldp x5, x6, [x10]
    cmp x5, x11
    b.lt .Ltake
    b.gt .Lskip
    cmp x6, x12
    b.gt .Lskip
.Ltake:
    add x4, x19, x13, lsl #4
    ldp x2, x3, [x4]
    stp x5, x6, [x4]
    stp x2, x3, [x10]
    add x13, x13, #1
.Lskip:
    add x14, x14, #1
    b .Lscan
.Lfinish:
    add x4, x19, x13, lsl #4
    ldp x2, x3, [x4]
    stp x11, x12, [x4]
    add x10, x19, x15, lsl #4
    stp x2, x3, [x10]
    mov x21, x13
    sub x22, x20, x21
    sub x22, x22, #1
    cmp x21, x22
    b.ge .Lright
    mov x0, x19
    mov x1, x21
    bl qsort_pairs
    cmp x22, #1
    b.le .Lepilogue
    add x9, x21, #1
    add x19, x19, x9, lsl #4
    mov x20, x22
    b .Louter
.Lright:
    add x9, x21, #1
    add x0, x19, x9, lsl #4
    mov x1, x22
    bl qsort_pairs
    cmp x21, #1
    b.le .Lepilogue
    mov x20, x21
    b .Louter
.Lepilogue:
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ldp x29, x30, [sp], #16
.Lret0:
    mov x0, #0
    ret
