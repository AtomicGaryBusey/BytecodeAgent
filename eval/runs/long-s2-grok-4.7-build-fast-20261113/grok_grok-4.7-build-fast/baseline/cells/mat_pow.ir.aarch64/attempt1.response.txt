export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%si =l alloc8 8
	%sj =l alloc8 8
	%st =l alloc8 8
	%ss =l alloc8 8
	%se =l alloc8 8
	storel 0, %si
	jmp @idi
@idi
	%i0 =l loadl %si
	%ci =w csltl %i0, %n
	jnz %ci, @idib, @iddone
@idib
	storel 0, %sj
	jmp @idj
@idj
	%j0 =l loadl %sj
	%cj =w csltl %j0, %n
	jnz %cj, @idjb, @idjdone
@idjb
	%i1 =l loadl %si
	%j1 =l loadl %sj
	%in0 =l mul %i1, %n
	%idx0 =l add %in0, %j1
	%off0 =l shl %idx0, 3
	%p0 =l add %out, %off0
	%eq0 =w ceql %i1, %j1
	%v0 =l extuw %eq0
	storel %v0, %p0
	%j2 =l loadl %sj
	%j3 =l add %j2, 1
	storel %j3, %sj
	jmp @idj
@idjdone
	%i2 =l loadl %si
	%i3 =l add %i2, 1
	storel %i3, %si
	jmp @idi
@iddone
	storel 0, %se
	jmp @ek
@ek
	%e0 =l loadl %se
	%ce =w csltl %e0, %k
	jnz %ce, @eb, @edone
@eb
	storel 0, %si
	jmp @mi
@mi
	%i4 =l loadl %si
	%cmi =w csltl %i4, %n
	jnz %cmi, @mib, @midone
@mib
	storel 0, %sj
	jmp @mj
@mj
	%j4 =l loadl %sj
	%cmj =w csltl %j4, %n
	jnz %cmj, @mjb, @mjdone
@mjb
	storel 0, %ss
	storel 0, %st
	jmp @mt
@mt
	%t0 =l loadl %st
	%cmt =w csltl %t0, %n
	jnz %cmt, @mtb, @mtdone
@mtb
	%i5 =l loadl %si
	%t1 =l loadl %st
	%in1 =l mul %i5, %n
	%idx1 =l add %in1, %t1
	%off1 =l shl %idx1, 3
	%po =l add %out, %off1
	%xv =l loadl %po
	%j5 =l loadl %sj
	%tn =l mul %t1, %n
	%idx2 =l add %tn, %j5
	%off2 =l shl %idx2, 3
	%pa =l add %a, %off2
	%yv =l loadl %pa
	%prod =l mul %xv, %yv
	%s0 =l loadl %ss
	%s1 =l add %s0, %prod
	storel %s1, %ss
	%t2 =l loadl %st
	%t3 =l add %t2, 1
	storel %t3, %st
	jmp @mt
@mtdone
	%i6 =l loadl %si
	%j6 =l loadl %sj
	%in2 =l mul %i6, %n
	%idx3 =l add %in2, %j6
	%off3 =l shl %idx3, 3
	%pt =l add %tmp, %off3
	%s2 =l loadl %ss
	storel %s2, %pt
	%j7 =l loadl %sj
	%j8 =l add %j7, 1
	storel %j8, %sj
	jmp @mj
@mjdone
	%i7 =l loadl %si
	%i8 =l add %i7, 1
	storel %i8, %si
	jmp @mi
@midone
	storel 0, %si
	jmp @cp
@cp
	%c0 =l loadl %si
	%cc =w csltl %c0, %nn
	jnz %cc, @cpb, @cpdone
@cpb
	%c1 =l loadl %si
	%off4 =l shl %c1, 3
	%ps =l add %tmp, %off4
	%pd =l add %out, %off4
	%cv =l loadl %ps
	storel %cv, %pd
	%c2 =l loadl %si
	%c3 =l add %c2, 1
	storel %c3, %si
	jmp @cp
@cpdone
	%e1 =l loadl %se
	%e2 =l add %e1, 1
	storel %e2, %se
	jmp @ek
@edone
	%two =l add %nn, %nn
	storel 0, %si
	jmp @z
@z
	%z0 =l loadl %si
	%cz =w csltl %z0, %two
	jnz %cz, @zb, @zd
@zb
	%z1 =l loadl %si
	%off5 =l shl %z1, 3
	%pz =l add %tmp, %off5
	storel 0, %pz
	%z2 =l loadl %si
	%z3 =l add %z2, 1
	storel %z3, %si
	jmp @z
@zd
	%rv =l copy 0
	ret %rv
}
