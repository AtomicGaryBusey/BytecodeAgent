export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @continue %pc2
	%sp =l phi @start 0, @continue %sp2
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch
@fetch
	%pc_off =l mul %pc, 8
	%pc_ptr =l add %code, %pc_off
	%op =l loadl %pc_ptr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @t1
@t1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @t2
@t2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @t3
@t3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @t4
@t4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @t5
@t5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @t6
@t6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @t7
@t7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @t8
@t8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @t9
@t9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @t10
@t10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @t11
@t11
	%is11 =w ceql %op, 11
	jnz %is11, @dojmp, @other
@push
	%pu_pc1 =l add %pc, 1
	%pu_off =l mul %pu_pc1, 8
	%pu_addr =l add %code, %pu_off
	%pu_v =l loadl %pu_addr
	%pu_soff =l mul %sp, 8
	%pu_saddr =l add %stack, %pu_soff
	storel %pu_v, %pu_saddr
	%sp_push =l add %sp, 1
	%pc_push =l add %pc, 2
	jmp @continue
@add
	%add_ib =l sub %sp, 1
	%add_ia =l sub %sp, 2
	%add_offb =l mul %add_ib, 8
	%add_offa =l mul %add_ia, 8
	%add_addrb =l add %stack, %add_offb
	%add_addra =l add %stack, %add_offa
	%add_b =l loadl %add_addrb
	%add_a =l loadl %add_addra
	%add_r =l add %add_a, %add_b
	storel %add_r, %add_addra
	%pc_add =l add %pc, 1
	jmp @continue
@sub
	%sub_ib =l sub %sp, 1
	%sub_ia =l sub %sp, 2
	%sub_offb =l mul %sub_ib, 8
	%sub_offa =l mul %sub_ia, 8
	%sub_addrb =l add %stack, %sub_offb
	%sub_addra =l add %stack, %sub_offa
	%sub_b =l loadl %sub_addrb
	%sub_a =l loadl %sub_addra
	%sub_r =l sub %sub_a, %sub_b
	storel %sub_r, %sub_addra
	%pc_sub =l add %pc, 1
	jmp @continue
@mul
	%mul_ib =l sub %sp, 1
	%mul_ia =l sub %sp, 2
	%mul_offb =l mul %mul_ib, 8
	%mul_offa =l mul %mul_ia, 8
	%mul_addrb =l add %stack, %mul_offb
	%mul_addra =l add %stack, %mul_offa
	%mul_b =l loadl %mul_addrb
	%mul_a =l loadl %mul_addra
	%mul_r =l mul %mul_a, %mul_b
	storel %mul_r, %mul_addra
	%pc_mul =l add %pc, 1
	jmp @continue
@dup
	%dup_i =l sub %sp, 1
	%dup_off =l mul %dup_i, 8
	%dup_addr =l add %stack, %dup_off
	%dup_v =l loadl %dup_addr
	%dup_doff =l mul %sp, 8
	%dup_daddr =l add %stack, %dup_doff
	storel %dup_v, %dup_daddr
	%sp_dup =l add %sp, 1
	%pc_dup =l add %pc, 1
	jmp @continue
@swap
	%sw_ib =l sub %sp, 1
	%sw_ia =l sub %sp, 2
	%sw_offb =l mul %sw_ib, 8
	%sw_offa =l mul %sw_ia, 8
	%sw_addrb =l add %stack, %sw_offb
	%sw_addra =l add %stack, %sw_offa
	%sw_b =l loadl %sw_addrb
	%sw_a =l loadl %sw_addra
	storel %sw_a, %sw_addrb
	storel %sw_b, %sw_addra
	%pc_swap =l add %pc, 1
	jmp @continue
@drop
	%sp_drop =l sub %sp, 1
	%pc_drop =l add %pc, 1
	jmp @continue
@over
	%over_i =l sub %sp, 2
	%over_off =l mul %over_i, 8
	%over_addr =l add %stack, %over_off
	%over_v =l loadl %over_addr
	%over_doff =l mul %sp, 8
	%over_daddr =l add %stack, %over_doff
	storel %over_v, %over_daddr
	%sp_over =l add %sp, 1
	%pc_over =l add %pc, 1
	jmp @continue
@lt
	%lt_ib =l sub %sp, 1
	%lt_ia =l sub %sp, 2
	%lt_offb =l mul %lt_ib, 8
	%lt_offa =l mul %lt_ia, 8
	%lt_addrb =l add %stack, %lt_offb
	%lt_addra =l add %stack, %lt_offa
	%lt_b =l loadl %lt_addrb
	%lt_a =l loadl %lt_addra
	%lt_c =w csltl %lt_a, %lt_b
	%lt_r =l extsw %lt_c
	storel %lt_r, %lt_addra
	%pc_lt =l add %pc, 1
	jmp @continue
@jz
	%jz_sp1 =l sub %sp, 1
	%jz_off =l mul %jz_sp1, 8
	%jz_addr =l add %stack, %jz_off
	%jz_v =l loadl %jz_addr
	%jz_pc1 =l add %pc, 1
	%jz_toff =l mul %jz_pc1, 8
	%jz_taddr =l add %code, %jz_toff
	%jz_t =l loadl %jz_taddr
	%jz_z =w ceql %jz_v, 0
	jnz %jz_z, @jz_yes, @jz_no
@jz_yes
	jmp @continue
@jz_no
	%pc_jzno =l add %pc, 2
	jmp @continue
@dojmp
	%jm_pc1 =l add %pc, 1
	%jm_off =l mul %jm_pc1, 8
	%jm_addr =l add %code, %jm_off
	%jm_t =l loadl %jm_addr
	jmp @continue
@other
	%pc_other =l add %pc, 1
	jmp @continue
@continue
	%pc2 =l phi @push %pc_push, @add %pc_add, @sub %pc_sub, @mul %pc_mul, @dup %pc_dup, @swap %pc_swap, @drop %pc_drop, @over %pc_over, @lt %pc_lt, @jz_yes %jz_t, @jz_no %pc_jzno, @dojmp %jm_t, @other %pc_other
	%sp2 =l phi @push %sp_push, @add %add_ib, @sub %sub_ib, @mul %mul_ib, @dup %sp_dup, @swap %sp, @drop %sp_drop, @over %sp_over, @lt %lt_ib, @jz_yes %jz_sp1, @jz_no %jz_sp1, @dojmp %sp, @other %sp
	jmp @loop
@finish
	%sp0 =w ceql %sp, 0
	jnz %sp0, @ret_zero, @ret_top
@ret_zero
	jmp @zero_init
@ret_top
	%rt_i =l sub %sp, 1
	%rt_off =l mul %rt_i, 8
	%rt_addr =l add %stack, %rt_off
	%rt_v =l loadl %rt_addr
	jmp @zero_init
@zero_init
	%rv =l phi @ret_zero 0, @ret_top %rt_v
	jmp @zero
@zero
	%zi =l phi @zero_init %sp, @zstep %zi_n
	%zmore =w csltl %zi, %cap
	jnz %zmore, @zstep, @done
@zstep
	%zs_off =l mul %zi, 8
	%zs_addr =l add %stack, %zs_off
	storel 0, %zs_addr
	%zi_n =l add %zi, 1
	jmp @zero
@done
	ret %rv
}
