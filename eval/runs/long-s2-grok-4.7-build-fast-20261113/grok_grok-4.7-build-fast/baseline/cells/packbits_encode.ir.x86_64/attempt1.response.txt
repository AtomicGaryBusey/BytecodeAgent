export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @rep_write %i_rep, @copyend %j
	%w =l phi @start 0, @rep_write %w_rep, @copyend %w_lit
	%at_end =w csgel %i, %n
	jnz %at_end, @success, @body
@body
	%ip =l add %src, %i
	%ib =w loadub %ip
	%k1 =l add %i, 1
	jmp @run1
@run1
	%r1 =w phi @body 1, @run1inc %r1n
	%k =l phi @body %k1, @run1inc %kn
	%rfull =w csgew %r1, 128
	jnz %rfull, @run1done, @run1n
@run1n
	%kpast =w csgel %k, %n
	jnz %kpast, @run1done, @run1cmp
@run1cmp
	%kp =l add %src, %k
	%kb =w loadub %kp
	%same =w ceqw %kb, %ib
	jnz %same, @run1inc, @run1done
@run1inc
	%r1n =w add %r1, 1
	%kn =l add %k, 1
	jmp @run1
@run1done
	%ge3 =w csgew %r1, 3
	jnz %ge3, @rep, @lit_loop
@rep
	%need =l add %w, 2
	%nofit =w csgtl %need, %cap
	jnz %nofit, @fail, @rep_write
@rep_write
	%hdr =w sub 257, %r1
	%op =l add %out, %w
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %ib, %op1
	%r1l =l extuw %r1
	%i_rep =l add %i, %r1l
	%w_rep =l add %w, 2
	jmp @outer
@lit_loop
	%j =l phi @run1done %i, @lit_adv %jnext
	%lc =w phi @run1done 0, @lit_adv %lcnext
	%jpast =w csgel %j, %n
	jnz %jpast, @lit_emit, @lit_chk128
@lit_chk128
	%jlim =l add %i, 128
	%toolong =w csgel %j, %jlim
	jnz %toolong, @lit_emit, @lit_run
@lit_run
	%jp =l add %src, %j
	%jb =w loadub %jp
	%m1 =l add %j, 1
	jmp @run2
@run2
	%r2 =w phi @lit_run 1, @run2inc %r2n
	%m =l phi @lit_run %m1, @run2inc %mn
	%r2full =w csgew %r2, 128
	jnz %r2full, @run2done, @run2n
@run2n
	%mpast =w csgel %m, %n
	jnz %mpast, @run2done, @run2cmp
@run2cmp
	%mp =l add %src, %m
	%mb =w loadub %mp
	%same2 =w ceqw %mb, %jb
	jnz %same2, @run2inc, @run2done
@run2inc
	%r2n =w add %r2, 1
	%mn =l add %m, 1
	jmp @run2
@run2done
	%r2ge3 =w csgew %r2, 3
	jnz %r2ge3, @lit_emit, @lit_adv
@lit_adv
	%jnext =l add %j, 1
	%lcnext =w add %lc, 1
	jmp @lit_loop
@lit_emit
	%lcl =l extuw %lc
	%lplus =l add %lcl, 1
	%need2 =l add %w, %lplus
	%nofit2 =w csgtl %need2, %cap
	jnz %nofit2, @fail, @lit_write
@lit_write
	%hm1 =w sub %lc, 1
	%opb =l add %out, %w
	storeb %hm1, %opb
	%dest0 =l add %opb, 1
	jmp @copyloop
@copyloop
	%c =l phi @lit_write 0, @copyone %cnext
	%cdone =w csgel %c, %lcl
	jnz %cdone, @copyend, @copyone
@copyone
	%sp =l add %i, %c
	%spp =l add %src, %sp
	%byte =w loadub %spp
	%dp =l add %dest0, %c
	storeb %byte, %dp
	%cnext =l add %c, 1
	jmp @copyloop
@copyend
	%w_lit =l add %w, %lplus
	jmp @outer
@success
	ret %w
@fail
	%err =l sub 0, 1
	ret %err
}
