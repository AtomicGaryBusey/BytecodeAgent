bubble_sort:
    cmp x1, #2
    b.lt .Lret0
    mov x2, #0
.Louter:
    mov x7, #0
    sub x8, x1, #1
    sub x8, x8, x2
    lsl x8, x8, #3
    add x3, x0, x8
    mov x4, x0
.Linner:
    cmp x4, x3
    b.ge .Linner_done
    ldr x5, [x4]
    ldr x6, [x4, #8]
    cmp x5, x6
    b.le .Lno_swap
    str x6, [x4]
    str x5, [x4, #8]
    mov x7, #1
.Lno_swap:
    add x4, x4, #8
    b .Linner
.Linner_done:
    cbz x7, .Lret0
    add x2, x2, #1
    sub x8, x1, #1
    cmp x2, x8
    b.lt .Louter
.Lret0:
    mov x0, #0
    ret
