export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo_0 =l copy 0
	%hi_0 =l copy %n
	jmp @loop
@loop
	%lo =l phi @start %lo_0, @ltmid %lo_1, @gtmid %lo
	%hi =l phi @start %hi_0, @ltmid %mid, @gtmid %mid
	%cond =w csltl %lo, %hi
	jnz %cond, @body, @notfound
@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @ltmid, @gtmid
@ltmid
	%lo_1 =l add %mid, 1
	jmp @loop
@gtmid
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
