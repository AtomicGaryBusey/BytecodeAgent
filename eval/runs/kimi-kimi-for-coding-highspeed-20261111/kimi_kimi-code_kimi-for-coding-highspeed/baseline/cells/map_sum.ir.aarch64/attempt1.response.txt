export function l $map_sum(l %a, l %n, l %f) {
@start
	%cond =w csltl 0, %n
	jnz %cond, @loop, @end
@loop
	%i =l phi @start 0, @loop %inext
	%sum =l phi @start 0, @loop %sum2
	%p =l phi @start %a, @loop %p2
	%x =l loadl %p
	%fx =l call %f(l %x)
	%sum2 =l add %sum, %fx
	%inext =l add %i, 1
	%p2 =l add %p, 8
	%cond2 =w csltl %inext, %n
	jnz %cond2, @loop, @end
@end
	%res =l phi @start 0, @loop %sum2
	ret %res
}
