memmove_i64:
    test rdx, rdx
    je .Ldone
    cmp rdi, rsi
    jbe .Lforward
    lea rcx, [rdx - 1]
.Lback_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    sub rcx, 1
    jnc .Lback_loop
    xor rax, rax
    ret
.Lforward:
    xor rcx, rcx
.Lfwd_loop:
    mov rax, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], rax
    add rcx, 1
    cmp rcx, rdx
    jb .Lfwd_loop
.Ldone:
    xor rax, rax
    ret
