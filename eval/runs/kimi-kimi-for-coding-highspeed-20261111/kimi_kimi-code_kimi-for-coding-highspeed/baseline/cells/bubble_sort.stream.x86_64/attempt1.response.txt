bubble_sort:
    mov rax, rsi
    sub rax, 1
    jle .Ldone
    mov rcx, rax
.Louter:
    xor rdx, rdx
.Linner:
    mov r8, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp r8, r9
    jle .Lskip
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], r8
.Lskip:
    add rdx, 1
    cmp rdx, rcx
    jl .Linner
    sub rcx, 1
    jnz .Louter
.Ldone:
    xor rax, rax
    ret
