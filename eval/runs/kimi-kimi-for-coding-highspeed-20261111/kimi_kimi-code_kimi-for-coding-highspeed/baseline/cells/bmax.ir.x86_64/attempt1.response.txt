export function l $bmax(l %a, l %n) {
@start
    %max =l loadl %a
    %p =l add %a, 8
    %i =l copy 1
    %c =w csltl %i, %n
    jnz %c, @loop(%max, %p, %i), @end(%max)
@loop(l %max, l %p, l %i)
    %cur =l loadl %p
    %gt =w csgtl %cur, %max
    jnz %gt, @then, @else
@then
    jmp @next(%cur)
@else
    jmp @next(%max)
@next(l %m)
    %p1 =l add %p, 8
    %i1 =l add %i, 1
    %c1 =w csltl %i1, %n
    jnz %c1, @loop(%m, %p1, %i1), @end(%m)
@end(l %r)
    ret %r
}
