export function l $bmax(l %a, l %n) {
@start
    %max =l loadl %a
    %p =l add %a, 8
    %i =l copy 1
    jmp @check
@check
    %curmax =l phi @start %max, @next %nextmax
    %curp =l phi @start %p, @next %nextp
    %curi =l phi @start %i, @next %nexti
    %c =w csltl %curi, %n
    jnz %c, @loop, @end
@loop
    %v =l loadl %curp
    %gt =w csgtl %v, %curmax
    jnz %gt, @then, @else
@then
    %mt =l copy %v
    jmp @next
@else
    %mf =l copy %curmax
    jmp @next
@next
    %nextmax =l phi @then %mt, @else %mf
    %nextp =l add %curp, 8
    %nexti =l add %curi, 1
    jmp @check
@end
    ret %curmax
}
