binary_search:
    mov     x3, #0
    mov     x4, x1
.Lloop:
    cmp     x3, x4
    bge     .Lnotfound
    add     x5, x3, x4
    lsr     x5, x5, #1
    ldr     x6, [x0, x5, lsl #3]
    cmp     x6, x2
    beq     .Lfound
    bgt     .Lgreater
    add     x3, x5, #1
    b       .Lloop
.Lgreater:
    mov     x4, x5
    b       .Lloop
.Lfound:
    mov     x0, x5
    ret
.Lnotfound:
    mov     x0, #-1
    ret
