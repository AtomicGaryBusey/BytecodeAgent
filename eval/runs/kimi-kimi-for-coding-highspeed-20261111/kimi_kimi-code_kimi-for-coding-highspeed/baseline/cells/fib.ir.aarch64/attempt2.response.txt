export function l $fib(l %n) {
@start
	%c0 =w ceql %n, 0
	jnz %c0, @ret0, @chk1
@chk1
	%c1 =w ceql %n, 1
	jnz %c1, @ret1, @init
@init
	%a =l copy 0
	%b =l copy 1
	%i =l copy 2
	jmp @loop
@loop
	%a1 =l phi @init %a, @cont %newa
	%b1 =l phi @init %b, @cont %newb
	%i1 =l phi @init %i, @cont %i2
	%c =l add %a1, %b1
	%cond =w csltl %i1, %n
	jnz %cond, @cont, @done
@cont
	%newa =l copy %b1
	%newb =l copy %c
	%i2 =l add %i1, 1
	jmp @loop
@done
	ret %c
@ret0
	ret 0
@ret1
	ret 1
}
