export function l $fib(l %n) {
@start
	%c0 =w ceql %n, 0
	jnz %c0, @ret0, @chk1
@chk1
	%c1 =w ceql %n, 1
	jnz %c1, @ret1, @loop(0, 1, 2)
@loop(l %a, l %b, l %i)
	%c =l add %a, %b
	%cond =w csltl %i, %n
	jnz %cond, @cont(%b, %c, %i), @done(%c)
@cont(l %a1, l %b1, l %i1)
	%i2 =l add %i1, 1
	jmp @loop(%a1, %b1, %i2)
@done(l %r)
	ret %r
@ret0
	ret 0
@ret1
	ret 1
}
