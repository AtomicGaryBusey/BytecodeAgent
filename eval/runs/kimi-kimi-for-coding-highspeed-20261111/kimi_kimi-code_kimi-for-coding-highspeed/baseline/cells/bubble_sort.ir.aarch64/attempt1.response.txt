export function l $bubble_sort(l %a, l %n) {
@start
	%nm1 =l sub %n, 1
	jmp @outer(0)

@outer(l %i)
	%oc =w csltl %i, %nm1
	jnz %oc, @outer_body, @end

@outer_body
	%ilim =l sub %nm1, %i
	jmp @inner(0)

@inner(l %j)
	%ic =w csltl %j, %ilim
	jnz %ic, @inner_body, @outer_next

@inner_body
	%off =l mul %j, 8
	%p1 =l add %a, %off
	%pj =l loadl %p1
	%p2 =l add %p1, 8
	%pn =l loadl %p2
	%gt =w csgtl %pj, %pn
	jnz %gt, @swap, @inner_next

@swap
	storel %pn, %p1
	storel %pj, %p2
	jmp @inner_next

@inner_next
	%jn =l add %j, 1
	jmp @inner(%jn)

@outer_next
	%in =l add %i, 1
	jmp @outer(%in)

@end
	ret 0
}
