export function l $bubble_sort(l %a, l %n) {
@start
	%cmpn =w csltl %n, 1
	jnz %cmpn, @done, @outer

@outer
	%i =l phi @start [ 0 ], @outer_next [ %inext ]
	%nminus1 =l sub %n, 1
	%iend =l sub %nminus1, %i
	%cmpsi =w csltl %i, %nminus1
	jnz %cmpsi, @inner, @done

@inner
	%j =l phi @outer [ 0 ], @inner_next [ %jnext ]
	%cmpsj =w csltl %j, %iend
	jnz %cmpsj, @swap_test, @outer_next

@swap_test
	%offj =l mul %j, 8
	%pj =l add %a, %offj
	%xj =l loadl %pj
	%jp1 =l add %j, 1
	%offjp1 =l mul %jp1, 8
	%pjp1 =l add %a, %offjp1
	%xjp1 =l loadl %pjp1
	%cmpgt =w csgtl %xj, %xjp1
	jnz %cmpgt, @swap, @inner_next

@swap
	storel %xjp1, %pj
	storel %xj, %pjp1
	jmp @inner_next

@inner_next
	%jnext =l add %j, 1
	jmp @inner

@outer_next
	%inext =l add %i, 1
	jmp @outer

@done
	ret 0
}
