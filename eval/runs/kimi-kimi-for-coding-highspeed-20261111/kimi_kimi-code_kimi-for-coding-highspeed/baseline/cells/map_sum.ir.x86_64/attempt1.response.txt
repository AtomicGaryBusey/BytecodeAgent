export function l $map_sum(l %a, l %n, l %f) {
@start
	%zero =l copy 0
	%cond =w csltl %zero, %n
	jnz %cond, @loop, @done

@loop
	%sum =l phi @start %zero, @loop %newsum
	%i =l phi @start %zero, @loop %nexti
	%off =l mul %i, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%mapped =l call %f(l %val)
	%newsum =l add %sum, %mapped
	%nexti =l add %i, 1
	%again =w csltl %nexti, %n
	jnz %again, @loop, @done

@done
	%result =l phi @start %zero, @loop %newsum
	ret %result
}
