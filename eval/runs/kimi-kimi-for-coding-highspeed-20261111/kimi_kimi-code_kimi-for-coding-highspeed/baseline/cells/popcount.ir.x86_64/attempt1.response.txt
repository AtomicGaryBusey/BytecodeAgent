export function l $popcount(l %x) {
@start
	%s1 =l shr %x, 1
	%m1 =l and %s1, 6148914691236517205
	%a =l sub %x, %m1

	%s2 =l shr %a, 2
	%m2a =l and %s2, 3689348814741910323
	%m2b =l and %a, 3689348814741910323
	%b =l add %m2a, %m2b

	%s4 =l shr %b, 4
	%c =l add %b, %s4
	%d =l and %c, 1085102592571150095

	%s8 =l shr %d, 8
	%e =l add %d, %s8

	%s16 =l shr %e, 16

	%s32 =l shr %f, 32
	%g =l add %f, %s32

	%r =l and %g, 127
	ret %r
}
