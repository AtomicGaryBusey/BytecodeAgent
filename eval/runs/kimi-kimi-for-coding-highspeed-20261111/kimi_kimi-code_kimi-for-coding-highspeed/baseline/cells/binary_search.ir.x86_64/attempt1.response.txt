export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo0 =l copy 0
	%hi0 =l sub %n, 1
	jmp @loop(l %lo0, l %hi0)

@loop(l %lo, l %hi)
	%ok =w cslel %lo, %hi
	jnz %ok, @body, @notfound

@body
	%diff =l sub %hi, %lo
	%half =l div %diff, 2
	%mid =l add %lo, %half
	%off =l mul %mid, 8
	%ptr =l add %a, %off
	%val =l loadl %ptr
	%eq =w ceql %val, %key
	jnz %eq, @found, @compare

@found
	ret %mid

@compare
	%lt =w csltl %key, %val
	jnz %lt, @left, @right

@left
	%hi1 =l sub %mid, 1
	jmp @loop(l %lo, l %hi1)

@right
	%lo1 =l add %mid, 1
	jmp @loop(l %lo1, l %hi)

@notfound
	%res =l copy -1
	ret %res
}
