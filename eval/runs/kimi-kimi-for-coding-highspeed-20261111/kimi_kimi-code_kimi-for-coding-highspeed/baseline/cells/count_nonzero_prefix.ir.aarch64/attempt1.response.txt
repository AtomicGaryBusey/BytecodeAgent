export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @inc %i1
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%zero =w ceql %v, 0
	jnz %zero, @end, @inc
@inc
	%i1 =l add %i, 1
	jmp @loop
@end
	ret %i
}
