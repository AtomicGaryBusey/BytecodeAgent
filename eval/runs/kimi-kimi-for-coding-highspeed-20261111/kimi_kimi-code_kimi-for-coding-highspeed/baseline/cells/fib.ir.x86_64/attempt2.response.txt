export function l $fib(l %n) {
@start
	%zero =w ceql %n, 0
	jnz %zero, @retzero, @loop
@loop
	%prev =l phi @start 0, @body %curr
	%curr =l phi @start 1, @body %next
	%i =l phi @start 1, @body %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%next =l add %prev, %curr
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %curr
@retzero
	ret 0
}
