export function l $fib(l %n) {
@start
	%zero =w ceql %n, 0
	jnz %zero, @retzero, @init
@init
	jmp @loop
@loop
	%prev =l phi @init 0, @body %prev1
	%curr =l phi @init 1, @body %curr1
	%i =l phi @init 1, @body %i1
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%next =l addl %prev, %curr
	%prev1 =l copy %curr
	%curr1 =l copy %next
	%i1 =l addl %i, 1
	jmp @loop
@done
	ret %curr
@retzero
	ret 0
}
