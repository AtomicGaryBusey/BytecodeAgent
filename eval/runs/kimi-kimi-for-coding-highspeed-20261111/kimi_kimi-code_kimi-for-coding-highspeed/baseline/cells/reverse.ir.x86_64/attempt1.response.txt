export function l $reverse(l %a, l %n) {
@start
	%i =l copy 0
	%j =l sub %n, 1
	%c0 =w csltl %i, %j
	jnz %c0, @loop, @end
@loop
	%oi =l mul %i, 8
	%pi =l add %a, %oi
	%xi =l loadl %pi
	%oj =l mul %j, 8
	%pj =l add %a, %oj
	%xj =l loadl %pj
	storel %xj, %pi
	storel %xi, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	%c1 =w csltl %i1, %j1
	jnz %c1, @loop, @end
@end
	%zero =l copy 0
	ret %zero
}
