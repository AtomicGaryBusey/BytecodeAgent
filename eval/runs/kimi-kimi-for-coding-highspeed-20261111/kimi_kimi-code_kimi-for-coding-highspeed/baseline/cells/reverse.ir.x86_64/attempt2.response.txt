export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop(0, %j0)
@loop(l %i, l %j)
	%c =w csltl %i, %j
	jnz %c, @body, @end
@body
	%oi =l mul %i, 8
	%pi =l add %a, %oi
	%xi =l loadl %pi
	%oj =l mul %j, 8
	%pj =l add %a, %oj
	%xj =l loadl %pj
	storel %xj, %pi
	storel %xi, %pj
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop(%i1, %j1)
@end
	%zero =l copy 0
	ret %zero
}
