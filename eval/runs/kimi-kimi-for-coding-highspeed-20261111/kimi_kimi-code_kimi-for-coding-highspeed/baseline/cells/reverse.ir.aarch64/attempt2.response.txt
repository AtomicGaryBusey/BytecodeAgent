export function l $reverse(l %a, l %n) {
@start
  %half =l shr %n, 1
  %skip =w csltl %n, 2
  jnz %skip, @done, @loop
@loop
  %i =l phi @start 0, @body %i1
  %cond =w csltl %i, %half
  jnz %cond, @body, @done
@body
  %offl =l mul %i, 8
  %left =l add %a, %offl
  %t =l sub %n, %i
  %ir =l sub %t, 1
  %offr =l mul %ir, 8
  %right =l add %a, %offr
  %vl =l loadl %left
  %vr =l loadl %right
  storel %vr, %left
  storel %vl, %right
  %i1 =l add %i, 1
  jmp @loop
@done
  ret 0
}
