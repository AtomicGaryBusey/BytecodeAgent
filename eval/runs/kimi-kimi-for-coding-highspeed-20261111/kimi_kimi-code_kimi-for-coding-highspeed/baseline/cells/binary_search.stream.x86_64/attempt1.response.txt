binary_search:
    test rsi, rsi
    jz .Lnot_found
    mov rax, 0
    mov rcx, rsi
    dec rcx
.Lloop:
    cmp rax, rcx
    jg .Lnot_found
    mov r8, rax
    add r8, rcx
    sar r8, 1
    mov r9, qword ptr [rdi + r8*8]
    cmp r9, rdx
    je .Lfound
    jl .Llow
    mov rcx, r8
    dec rcx
    jmp .Lloop
.Llow:
    mov rax, r8
    inc rax
    jmp .Lloop
.Lfound:
    mov rax, r8
    ret
.Lnot_found:
    mov rax, -1
    ret
