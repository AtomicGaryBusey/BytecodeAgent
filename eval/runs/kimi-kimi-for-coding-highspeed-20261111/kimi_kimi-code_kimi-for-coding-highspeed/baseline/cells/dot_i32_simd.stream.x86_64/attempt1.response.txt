dot_i32_simd:
    pxor xmm4, xmm4
    xor rax, rax
    cmp rdx, 4
    jl .Lscalar_loop
.Lsimd_loop:
    movdqu xmm0, [rdi]
    movdqu xmm1, [rsi]
    movdqa xmm2, xmm0
    pmuldq xmm2, xmm1
    psrldq xmm0, 4
    psrldq xmm1, 4
    pmuldq xmm0, xmm1
    paddq xmm4, xmm2
    paddq xmm4, xmm0
    add rdi, 16
    add rsi, 16
    sub rdx, 4
    cmp rdx, 4
    jge .Lsimd_loop
.Lscalar_loop:
    test rdx, rdx
    jz .Lreduce
    movsxd rcx, dword ptr [rdi]
    movsxd r8, dword ptr [rsi]
    imul rcx, r8
    add rax, rcx
    add rdi, 4
    add rsi, 4
    dec rdx
    jmp .Lscalar_loop
.Lreduce:
    movq rcx, xmm4
    pextrq rdx, xmm4, 1
    add rax, rcx
    add rax, rdx
    ret
