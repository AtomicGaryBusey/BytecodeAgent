export function l $popcount(l %x) {
@start
	%a =l shr %x, l 1
	%b =l and %a, l 0x5555555555555555
	%c =l sub %x, %b

	%d =l and %c, l 0x3333333333333333
	%e =l shr %c, l 2
	%f =l and %e, l 0x3333333333333333
	%g =l add %d, %f

	%h =l shr %g, l 4
	%i =l add %g, %h
	%j =l and %i, l 0x0F0F0F0F0F0F0F0F

	%k =l shr %j, l 8
	%m =l add %j, %k

	%n =l shr %m, l 16
	%o =l add %m, %n

	%p =l shr %o, l 32
	%q =l add %o, %p

	%r =l and %q, l 0xFF
	ret %r
}
