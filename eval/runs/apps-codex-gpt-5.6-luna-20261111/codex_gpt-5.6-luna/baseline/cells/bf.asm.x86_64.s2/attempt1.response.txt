.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ STACK_SIZE, 8000
.equ MATCH_SIZE, 524288

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13, r13

read_more:
    mov rax, 0
    xor rdi, rdi
    lea rsi, [r12 + r13]
    mov rdx, INPUT_SIZE
    sub rdx, r13
    syscall
    test rax, rax
    jz input_done
    add r13, rax
    jmp read_more

input_done:
    mov rbx, r12
    mov r14, r13
    lea r15, [r12 + r13]
    xor rcx, rcx

find_delimiter:
    cmp rcx, r13
    jae delimiter_done
    cmp byte ptr [rbx + rcx], 33
    je delimiter_found
    inc rcx
    jmp find_delimiter

delimiter_found:
    mov r14, rcx
    lea r15, [rbx + rcx + 1]

delimiter_done:
    lea r8, [rip + bracket_stack]
    xor r9, r9
    xor rcx, rcx

validate_loop:
    cmp rcx, r14
    jae validate_done
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rcx
    jmp validate_loop

validate_open:
    mov qword ptr [r8 + r9*8], rcx
    inc r9
    inc rcx
    jmp validate_loop

validate_close:
    test r9, r9
    jz invalid_program
    dec r9
    mov rdx, qword ptr [r8 + r9*8]
    lea rax, [rip + match_table]
    mov qword ptr [rax + rdx*8], rcx
    mov qword ptr [rax + rcx*8], rdx
    inc rcx
    jmp validate_loop

validate_done:
    test r9, r9
    jnz invalid_program

    lea r8, [rip + tape]
    xor r9, r9

execute_loop:
    cmp r9, r14
    jae successful_exit
    movzx eax, byte ptr [rbx + r9]
    inc r9

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close
    jmp execute_loop

command_right:
    inc r8
    jmp execute_loop

command_left:
    dec r8
    jmp execute_loop

command_plus:
    inc byte ptr [r8]
    jmp execute_loop

command_minus:
    dec byte ptr [r8]
    jmp execute_loop

command_output:
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, 1
    syscall
    jmp execute_loop

command_input:
    cmp r15, r12
    jb input_store_zero
    cmp r15, r12
    jae input_compare_end
input_compare_end:
    cmp r15, r10
    jae input_store_zero
    mov al, byte ptr [r15]
    inc r15
    mov byte ptr [r8], al
    jmp execute_loop

input_store_zero:
    mov byte ptr [r8], 0
    jmp execute_loop

command_open:
    cmp byte ptr [r8], 0
    jne execute_loop
    mov rdx, r9
    dec rdx
    lea rax, [rip + match_table]
    mov r9, qword ptr [rax + rdx*8]
    inc r9
    jmp execute_loop

command_close:
    cmp byte ptr [r8], 0
    je execute_loop
    mov rdx, r9
    dec rdx
    lea rax, [rip + match_table]
    mov r9, qword ptr [rax + rdx*8]
    inc r9
    jmp execute_loop

invalid_program:
    mov rax, 60
    mov rdi, 1
    syscall

successful_exit:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .bss
.balign 8
input_buffer:
    .zero INPUT_SIZE
.balign 8
tape:
    .zero TAPE_SIZE
.balign 8
bracket_stack:
    .zero STACK_SIZE
.balign 8
match_table:
    .zero MATCH_SIZE
