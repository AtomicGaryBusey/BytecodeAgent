.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_SIZE, 524288
.equ BRACKET_SIZE, 8000
.equ OUTPUT_SIZE, 65536

.section .bss
.align 8
input_buffer:
.zero INPUT_SIZE
.align 8
tape:
.zero TAPE_SIZE
.align 8
jump_table:
.zero JUMP_SIZE
.align 8
bracket_stack:
.zero BRACKET_SIZE
.align 8
output_buffer:
.zero OUTPUT_SIZE

.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor ebx, ebx

read_input:
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, INPUT_SIZE
    sub edx, ebx
    syscall
    test rax, rax
    jz input_done
    add rbx, rax
    cmp ebx, INPUT_SIZE
    jb read_input

input_done:
    lea r15, [r12 + rbx]
    mov r14, r15
    mov rsi, r12

find_delimiter:
    cmp rsi, r15
    jae delimiter_done
    cmp byte ptr [rsi], 33
    je delimiter_found
    inc rsi
    jmp find_delimiter

delimiter_found:
    mov r13, rsi
    lea r14, [rsi + 1]
    jmp validate_start

delimiter_done:
    mov r13, r15

validate_start:
    xor r8d, r8d
    lea rbx, [rip + bracket_stack]
    lea r11, [rip + jump_table]
    mov rsi, r12

validate_loop:
    cmp rsi, r13
    jae validate_end
    movzx eax, byte ptr [rsi]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rsi
    jmp validate_loop

validate_open:
    mov rax, rsi
    sub rax, r12
    mov qword ptr [rbx + r8*8], rax
    inc r8
    inc rsi
    jmp validate_loop

validate_close:
    test r8, r8
    jz invalid_program
    dec r8
    mov rax, qword ptr [rbx + r8*8]
    mov rcx, rsi
    sub rcx, r12
    mov qword ptr [r11 + rax*8], rcx
    mov qword ptr [r11 + rcx*8], rax
    inc rsi
    jmp validate_loop

validate_end:
    test r8, r8
    jnz invalid_program

    lea rdi, [rip + tape]
    lea r9, [rip + output_buffer]
    xor ebx, ebx
    lea r10, [rip + jump_table]
    mov rsi, r12

execute_loop:
    cmp rsi, r13
    jae execution_done
    movzx eax, byte ptr [rsi]
    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close
    inc rsi
    jmp execute_loop

command_right:
    inc rdi
    inc rsi
    jmp execute_loop

command_left:
    dec rdi
    inc rsi
    jmp execute_loop

command_plus:
    inc byte ptr [rdi]
    inc rsi
    jmp execute_loop

command_minus:
    dec byte ptr [rdi]
    inc rsi
    jmp execute_loop

command_output:
    mov al, byte ptr [rdi]
    mov byte ptr [r9 + rbx], al
    inc rbx
    inc rsi
    jmp execute_loop

command_input:
    cmp r14, r15
    jae input_exhausted
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rdi], al
    inc rsi
    jmp execute_loop

input_exhausted:
    mov byte ptr [rdi], 0
    inc rsi
    jmp execute_loop

command_open:
    cmp byte ptr [rdi], 0
    jne open_continue
    mov rax, rsi
    sub rax, r12
    mov rax, qword ptr [r10 + rax*8]
    lea rsi, [r12 + rax + 1]
    jmp execute_loop

open_continue:
    inc rsi
    jmp execute_loop

command_close:
    cmp byte ptr [rdi], 0
    je close_continue
    mov rax, rsi
    sub rax, r12
    mov rax, qword ptr [r10 + rax*8]
    lea rsi, [r12 + rax + 1]
    jmp execute_loop

close_continue:
    inc rsi
    jmp execute_loop

execution_done:
    test rbx, rbx
    jz success

    lea rsi, [rip + output_buffer]

write_loop:
    mov eax, 1
    mov edi, 1
    mov rdx, rbx
    syscall
    add rsi, rax
    sub rbx, rax
    jnz write_loop

success:
    mov eax, 60
    xor edi, edi
    syscall

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall
