.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_SIZE, 262144
.equ STACK_SIZE, 8000

.globl _start

.text
_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x20, tape
    add  x20, x20, :lo12:tape
    adrp x21, jump_table
    add  x21, x21, :lo12:jump_table
    adrp x22, output_buf
    add  x22, x22, :lo12:output_buf
    adrp x18, bracket_stack
    add  x18, x18, :lo12:bracket_stack

    mov  x25, #0

read_loop:
    mov  x0, #0
    add  x1, x19, x25
    mov  x2, #INPUT_SIZE
    sub  x2, x2, x25
    mov  x8, #63
    svc  #0
    cbz  x0, read_done
    add  x25, x25, x0
    b    read_loop

read_done:
    mov  x15, #0

find_delimiter:
    cmp  x15, x25
    b.eq no_delimiter
    ldrb w0, [x19, x15]
    cmp  w0, #33
    b.eq delimiter_found
    add  x15, x15, #1
    b    find_delimiter

delimiter_found:
    mov  x23, x15
    add  x11, x15, #1
    b    validate_start

no_delimiter:
    mov  x23, x25
    mov  x11, x25

validate_start:
    mov  x15, #0
    mov  x14, #0

validate_loop:
    cmp  x15, x23
    b.eq validate_done
    ldrb w0, [x19, x15]
    cmp  w0, #91
    b.eq validate_open
    cmp  w0, #93
    b.eq validate_close
    add  x15, x15, #1
    b    validate_loop

validate_open:
    lsl  x1, x14, #2
    str  w15, [x18, x1]
    add  x14, x14, #1
    add  x15, x15, #1
    b    validate_loop

validate_close:
    cbz  x14, invalid_program
    sub  x14, x14, #1
    lsl  x1, x14, #2
    ldr  w2, [x18, x1]
    lsl  x3, x15, #2
    str  w2, [x21, x3]
    str  w15, [x21, x1]
    add  x15, x15, #1
    b    validate_loop

validate_done:
    cbnz x14, invalid_program

    mov  x9, #0
    mov  x12, #0
    mov  x13, #0
    mov  x10, x23

execute_loop:
    cmp  x9, x10
    b.eq flush_output
    ldrb w0, [x19, x9]

    cmp  w0, #62
    b.eq command_right
    cmp  w0, #60
    b.eq command_left
    cmp  w0, #43
    b.eq command_inc
    cmp  w0, #45
    b.eq command_dec
    cmp  w0, #46
    b.eq command_output
    cmp  w0, #44
    b.eq command_input
    cmp  w0, #91
    b.eq command_open
    cmp  w0, #93
    b.eq command_close

    add  x9, x9, #1
    b    execute_loop

command_right:
    add  x12, x12, #1
    add  x9, x9, #1
    b    execute_loop

command_left:
    sub  x12, x12, #1
    add  x9, x9, #1
    b    execute_loop

command_inc:
    ldrb w1, [x20, x12]
    add  w1, w1, #1
    strb w1, [x20, x12]
    add  x9, x9, #1
    b    execute_loop

command_dec:
    ldrb w1, [x20, x12]
    sub  w1, w1, #1
    strb w1, [x20, x12]
    add  x9, x9, #1
    b    execute_loop

command_output:
    ldrb w1, [x20, x12]
    strb w1, [x22, x13]
    add  x13, x13, #1
    add  x9, x9, #1
    b    execute_loop

command_input:
    cmp  x11, x25
    b.eq input_exhausted
    ldrb w1, [x19, x11]
    add  x11, x11, #1
    strb w1, [x20, x12]
    add  x9, x9, #1
    b    execute_loop

input_exhausted:
    strb wzr, [x20, x12]
    add  x9, x9, #1
    b    execute_loop

command_open:
    ldrb w1, [x20, x12]
    cbnz w1, open_continue
    lsl  x2, x9, #2
    ldr  w3, [x21, x2]
    add  x9, x3, #1
    b    execute_loop

open_continue:
    add  x9, x9, #1
    b    execute_loop

command_close:
    ldrb w1, [x20, x12]
    cbz  w1, close_continue
    lsl  x2, x9, #2
    ldr  w3, [x21, x2]
    add  x9, x3, #1
    b    execute_loop

close_continue:
    add  x9, x9, #1
    b    execute_loop

flush_output:
    mov  x14, #0

write_loop:
    cmp  x14, x13
    b.eq exit_success
    mov  x0, #1
    add  x1, x22, x14
    sub  x2, x13, x14
    mov  x8, #64
    svc  #0
    add  x14, x14, x0
    b    write_loop

invalid_program:
    mov  x0, #1
    mov  x8, #93
    svc  #0

exit_success:
    mov  x0, #0
    mov  x8, #93
    svc  #0

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE

.balign 16
tape:
    .zero TAPE_SIZE

.balign 16
jump_table:
    .zero JUMP_SIZE

.balign 16
bracket_stack:
    .zero STACK_SIZE

.balign 16
output_buf:
    .zero INPUT_SIZE
