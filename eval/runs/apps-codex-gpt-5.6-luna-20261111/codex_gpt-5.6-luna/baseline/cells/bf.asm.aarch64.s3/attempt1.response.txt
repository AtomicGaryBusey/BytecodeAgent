.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_SIZE, 262144
.equ STACK_SIZE, 8000

.globl _start

.text
_start:
    adrp x19, input_buf
    add  x19, x19, :lo12:input_buf
    adrp x20, tape
    add  x20, x20, :lo12:tape
    adrp x21, jump_table
    add  x21, x21, :lo12:jump_table
    adrp x22, output_buf
    add  x22, x22, :lo12:output_buf
    adrp x18, bracket_stack
    add  x18, x18, :lo12:bracket_stack

    mov  x24, #0

read_loop:
    mov  x0, #0
    add  x1, x19, x24
    mov  x2, #INPUT_SIZE
    sub  x2, x2, x24
    mov  x8, #63
    svc  #0
    cbz  x0, read_done
    add  x24, x24, x0
    b    read_loop

read_done:
    mov  x0, #0

find_delimiter:
    cmp  x0, x24
    b.eq no_delimiter
    ldrb w1, [x19, x0]
    cmp  w1, #33
    b.eq delimiter_found
    add  x0, x0, #1
    b    find_delimiter

delimiter_found:
    mov  x23, x0
    add  x24, x0, #1
    mov  x25, x24
    b    validate_start

no_delimiter:
    mov  x23, x24
    mov  x25, x24
    mov  x24, x24
    b    validate_start

validate_start:
    mov  x0, #0
    mov  x28, #0

validate_loop:
    cmp  x0, x23
    b.eq validate_done
    ldrb w1, [x19, x0]
    cmp  w1, #91
    b.eq validate_open
    cmp  w1, #93
    b.eq validate_close
    add  x0, x0, #1
    b    validate_loop

validate_open:
    lsl  x3, x28, #2
    str  w0, [x18, x3]
    add  x28, x28, #1
    add  x0, x0, #1
    b    validate_loop

validate_close:
    cbz  x28, invalid_program
    sub  x28, x28, #1
    lsl  x3, x28, #2
    ldr  w4, [x18, x3]
    lsl  x5, x0, #2
    str  w4, [x21, x5]
    str  w0, [x21, x3]
    add  x0, x0, #1
    b    validate_loop

validate_done:
    cbnz x28, invalid_program

    mov  x0, #0
    mov  x26, #0
    mov  x27, #0

execute_loop:
    cmp  x0, x23
    b.eq flush_output
    ldrb w1, [x19, x0]

    cmp  w1, #62
    b.eq command_right
    cmp  w1, #60
    b.eq command_left
    cmp  w1, #43
    b.eq command_inc
    cmp  w1, #45
    b.eq command_dec
    cmp  w1, #46
    b.eq command_output
    cmp  w1, #44
    b.eq command_input
    cmp  w1, #91
    b.eq command_open
    cmp  w1, #93
    b.eq command_close

    add  x0, x0, #1
    b    execute_loop

command_right:
    add  x26, x26, #1
    add  x0, x0, #1
    b    execute_loop

command_left:
    sub  x26, x26, #1
    add  x0, x0, #1
    b    execute_loop

command_inc:
    ldrb w2, [x20, x26]
    add  w2, w2, #1
    strb w2, [x20, x26]
    add  x0, x0, #1
    b    execute_loop

command_dec:
    ldrb w2, [x20, x26]
    sub  w2, w2, #1
    strb w2, [x20, x26]
    add  x0, x0, #1
    b    execute_loop

command_output:
    ldrb w2, [x20, x26]
    strb w2, [x22, x27]
    add  x27, x27, #1
    add  x0, x0, #1
    b    execute_loop

command_input:
    cmp  x24, x25
    b.eq input_exhausted
    ldrb w2, [x19, x24]
    add  x24, x24, #1
    strb w2, [x20, x26]
    add  x0, x0, #1
    b    execute_loop

input_exhausted:
    strb wzr, [x20, x26]
    add  x0, x0, #1
    b    execute_loop

command_open:
    ldrb w2, [x20, x26]
    cbnz w2, open_continue
    lsl  x3, x0, #2
    ldr  w4, [x21, x3]
    add  x0, x4, #1
    b    execute_loop

open_continue:
    add  x0, x0, #1
    b    execute_loop

command_close:
    ldrb w2, [x20, x26]
    cbz  w2, close_continue
    lsl  x3, x0, #2
    ldr  w4, [x21, x3]
    add  x0, x4, #1
    b    execute_loop

close_continue:
    add  x0, x0, #1
    b    execute_loop

flush_output:
    mov  x3, #0

write_loop:
    cmp  x3, x27
    b.eq exit_success
    mov  x0, #1
    add  x1, x22, x3
    sub  x2, x27, x3
    mov  x8, #64
    svc  #0
    add  x3, x3, x0
    b    write_loop

invalid_program:
    mov  x0, #1
    mov  x8, #93
    svc  #0

exit_success:
    mov  x0, #0
    mov  x8, #93
    svc  #0

.bss
.balign 16
input_buf:
    .zero INPUT_SIZE

.balign 16
tape:
    .zero TAPE_SIZE

.balign 16
jump_table:
    .zero JUMP_SIZE

.balign 16
bracket_stack:
    .zero STACK_SIZE

.balign 16
output_buf:
    .zero INPUT_SIZE
