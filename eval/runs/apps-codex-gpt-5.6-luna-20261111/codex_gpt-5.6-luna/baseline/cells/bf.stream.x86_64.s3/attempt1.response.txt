.globl _start

.section .text
_start:
    lea r14, [rip + input]
    xor r12, r12

read_more:
    xor edi, edi
    lea rsi, [r14 + r12]
    mov edx, 65536
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    xor rcx, rcx
    xor rbx, rbx

find_bang:
    cmp rcx, r12
    jae no_bang
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 33
    je found_bang
    inc rcx
    jmp find_bang

found_bang:
    mov rbx, rcx
    lea r15, [rcx + 1]
    jmp split_done

no_bang:
    mov rbx, r12
    mov r15, r12

split_done:
    lea rbp, [rip + pairs]
    lea rsi, [rip + bracket_stack]
    xor r9, r9
    xor rcx, rcx

check_brackets:
    cmp rcx, rbx
    jae brackets_done
    movzx eax, byte ptr [r14 + rcx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    inc rcx
    jmp check_brackets

open_bracket:
    mov qword ptr [rsi + r9*8], rcx
    inc r9
    inc rcx
    jmp check_brackets

close_bracket:
    test r9, r9
    jz invalid
    dec r9
    mov rax, qword ptr [rsi + r9*8]
    mov qword ptr [rbp + rax*8], rcx
    mov qword ptr [rbp + rcx*8], rax
    inc rcx
    jmp check_brackets

brackets_done:
    test r9, r9
    jnz invalid

    lea r13, [rip + tape]
    xor r10, r10
    xor r12, r12
    mov r9, qword ptr [rip + input_total]

execute:
    cmp r12, rbx
    jae success
    movzx eax, byte ptr [r14 + r12]

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close

    inc r12
    jmp execute

command_right:
    inc r10
    inc r12
    jmp execute

command_left:
    dec r10
    inc r12
    jmp execute

command_plus:
    inc byte ptr [r13 + r10]
    inc r12
    jmp execute

command_minus:
    dec byte ptr [r13 + r10]
    inc r12
    jmp execute

command_output:
    mov edi, 1
    lea rsi, [r13 + r10]
    mov edx, 1
    mov eax, 1
    syscall
    inc r12
    jmp execute

command_input:
    cmp r15, r9
    jae input_exhausted
    movzx eax, byte ptr [r14 + r15]
    inc r15
    mov byte ptr [r13 + r10], al
    inc r12
    jmp execute

input_exhausted:
    mov byte ptr [r13 + r10], 0
    inc r12
    jmp execute

command_open:
    cmp byte ptr [r13 + r10], 0
    jne open_nonzero
    mov rax, qword ptr [rbp + r12*8]
    inc rax
    mov r12, rax
    jmp execute

open_nonzero:
    inc r12
    jmp execute

command_close:
    cmp byte ptr [r13 + r10], 0
    je close_zero
    mov rax, qword ptr [rbp + r12*8]
    inc rax
    mov r12, rax
    jmp execute

close_zero:
    inc r12
    jmp execute

success:
    mov edi, 0
    mov eax, 60
    syscall

invalid:
    mov edi, 1
    mov eax, 60
    syscall

.section .bss
.align 8
input_total:
    .quad 0
input:
    .zero 131072
tape:
    .zero 30000
bracket_stack:
    .zero 8000
pairs:
    .zero 524288
