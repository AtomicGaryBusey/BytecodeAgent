.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000

.globl _start

.section .text
_start:
    lea r10, [rip + input]
    xor r9d, r9d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [r10 + r9]
    mov edx, INPUT_SIZE
    sub rdx, r9
    syscall
    test rax, rax
    jz read_done
    add r9, rax
    jmp read_loop

read_done:
    lea r10, [rip + input]
    mov r11, r9
    lea rdi, [rip + output]
    xor r8d, r8d
    lea r9, [rip + alphabet]

encode_loop:
    cmp r11, 3
    jb tail_check

    movzx eax, byte ptr [r10]
    shl eax, 16
    movzx edx, byte ptr [r10 + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [r10 + 2]
    or eax, edx

    mov ecx, eax
    shr ecx, 18
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi + 1], cl

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi + 2], cl

    and eax, 63
    movzx eax, byte ptr [r9 + rax]
    mov byte ptr [rdi + 3], al

    add rdi, 4
    add r10, 3
    sub r11, 3
    add r8, 4
    cmp r8, 76
    jne encode_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp encode_loop

tail_check:
    test r11, r11
    jz write_output
    cmp r11, 1
    je tail_one

tail_two:
    movzx eax, byte ptr [r10]
    shl eax, 16
    movzx edx, byte ptr [r10 + 1]
    shl edx, 8
    or eax, edx

    mov ecx, eax
    shr ecx, 18
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi + 1], cl

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi + 2], cl

    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi
    jmp write_output

tail_one:
    movzx eax, byte ptr [r10]
    shl eax, 16

    mov ecx, eax
    shr ecx, 18
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi], cl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    movzx ecx, byte ptr [r9 + rcx]
    mov byte ptr [rdi + 1], cl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input:
    .zero INPUT_SIZE
output:
    .zero OUTPUT_SIZE
