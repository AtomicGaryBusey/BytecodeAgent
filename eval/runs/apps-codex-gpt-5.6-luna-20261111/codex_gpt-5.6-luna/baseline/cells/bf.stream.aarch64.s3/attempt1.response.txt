.section .text
.globl _start

_start:
	adrp x19, input_buf
	add x19, x19, :lo12:input_buf
	mov x20, #0

read_more:
	movz x27, #1, lsl #16
	cmp x20, x27
	b.eq read_done
	mov x0, #0
	add x1, x19, x20
	movz x2, #1, lsl #16
	sub x2, x2, x20
	mov x8, #63
	svc #0
	cbz x0, read_done
	add x20, x20, x0
	b read_more

read_done:
	add x24, x19, x20
	mov x26, #0

find_delimiter:
	cmp x26, x20
	b.eq no_delimiter
	ldrb w27, [x19, x26]
	cmp w27, #33
	b.eq delimiter_found
	add x26, x26, #1
	b find_delimiter

delimiter_found:
	add x27, x26, #1
	add x23, x19, x27
	mov x20, x26
	b validate

no_delimiter:
	mov x23, x24

validate:
	adrp x25, jump_table
	add x25, x25, :lo12:jump_table
	adrp x28, bracket_stack
	add x28, x28, :lo12:bracket_stack
	mov x29, #0
	mov x26, #0

validate_loop:
	cmp x26, x20
	b.eq validate_done
	ldrb w27, [x19, x26]
	cmp w27, #91
	b.eq open_bracket
	cmp w27, #93
	b.eq close_bracket
	add x26, x26, #1
	b validate_loop

open_bracket:
	str w26, [x28, x29, lsl #2]
	add x29, x29, #1
	add x26, x26, #1
	b validate_loop

close_bracket:
	cbz x29, bad_program
	sub x29, x29, #1
	ldr w27, [x28, x29, lsl #2]
	str w26, [x25, x27, lsl #2]
	str w27, [x25, x26, lsl #2]
	add x26, x26, #1
	b validate_loop

validate_done:
	cbnz x29, bad_program

	adrp x21, tape
	add x21, x21, :lo12:tape
	mov x22, #0
	mov x26, #0

execute:
	cmp x26, x20
	b.eq success
	ldrb w27, [x19, x26]

	cmp w27, #62
	b.eq command_right
	cmp w27, #60
	b.eq command_left
	cmp w27, #43
	b.eq command_plus
	cmp w27, #45
	b.eq command_minus
	cmp w27, #46
	b.eq command_output
	cmp w27, #44
	b.eq command_input
	cmp w27, #91
	b.eq command_open
	cmp w27, #93
	b.eq command_close

	add x26, x26, #1
	b execute

command_right:
	add x22, x22, #1
	add x26, x26, #1
	b execute

command_left:
	sub x22, x22, #1
	add x26, x26, #1
	b execute

command_plus:
	ldrb w28, [x21, x22]
	add w28, w28, #1
	strb w28, [x21, x22]
	add x26, x26, #1
	b execute

command_minus:
	ldrb w28, [x21, x22]
	sub w28, w28, #1
	strb w28, [x21, x22]
	add x26, x26, #1
	b execute

command_output:
	ldrb w28, [x21, x22]
	mov x0, #1
	add x1, x21, x22
	mov x2, #1
	mov x8, #64
write_again:
	svc #0
	cmp x0, #1
	b.ne write_again
	add x26, x26, #1
	b execute

command_input:
	cmp x23, x24
	b.cs input_empty
	ldrb w28, [x23]
	add x23, x23, #1
	b store_input

input_empty:
	mov w28, #0

store_input:
	strb w28, [x21, x22]
	add x26, x26, #1
	b execute

command_open:
	ldrb w28, [x21, x22]
	cbnz w28, open_continue
	ldr w28, [x25, x26, lsl #2]
	add x26, x28, #1
	b execute

open_continue:
	add x26, x26, #1
	b execute

command_close:
	ldrb w28, [x21, x22]
	cbz w28, close_continue
	ldr w28, [x25, x26, lsl #2]
	add x26, x28, #1
	b execute

close_continue:
	add x26, x26, #1
	b execute

bad_program:
	mov x0, #1
	mov x8, #93
	svc #0

success:
	mov x0, #0
	mov x8, #93
	svc #0

.section .bss
.align 16
input_buf:
.zero 65536
tape:
.zero 30000
jump_table:
.zero 262144
bracket_stack:
.zero 4000
