.equ INPUT_SIZE, 65536
.equ VALUE_STACK_SIZE, 8192
.equ OUTPUT_SIZE, 64
.equ ERROR_LEN, 6

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_more:
    mov x0, #0
    add x1, x19, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    ble input_ready
    add x20, x20, x0
    cmp x20, #INPUT_SIZE
    blt read_more

input_ready:
    mov x21, #0

next_line:
    cmp x21, x20
    beq finish
    add x22, x19, x21
    mov x23, x22

find_newline:
    cmp x23, x19
    add x24, x19, x20
    bhs final_line
    ldrb w9, [x23]
    cmp w9, #10
    beq found_newline
    add x23, x23, #1
    b find_newline

found_newline:
    add x21, x23, #1
    b begin_line

final_line:
    mov x23, x24
    mov x21, x20

begin_line:
    adrp x25, value_stack
    add x25, x25, :lo12:value_stack
    mov x26, #0
    mov x27, x22

parse_loop:
    cmp x27, x23
    beq line_done
    ldrb w9, [x27]
    cmp w9, #32
    beq skip_separator
    cmp w9, #9
    beq skip_separator
    mov x28, x27
    b scan_token

skip_separator:
    add x27, x27, #1
    b parse_loop

scan_token:
    cmp x27, x23
    beq token_scanned
    ldrb w9, [x27]
    cmp w9, #32
    beq token_scanned
    cmp w9, #9
    beq token_scanned
    add x27, x27, #1
    b scan_token

token_scanned:
    sub x29, x27, x28
    cmp x29, #1
    bne numeric_token
    ldrb w9, [x28]
    cmp w9, #43
    beq operator_token
    cmp w9, #45
    beq operator_token
    cmp w9, #42
    beq operator_token
    cmp w9, #47
    beq operator_token
    cmp w9, #37
    beq operator_token

numeric_token:
    mov x15, x28
    mov x16, #0
    ldrb w9, [x15]
    cmp w9, #45
    bne numeric_digits
    mov x16, #1
    add x15, x15, #1
    sub x17, x29, #1
    b check_digit_count

numeric_digits:
    mov x17, x29

check_digit_count:
    cmp x17, #1
    blt line_error
    cmp x17, #18
    bgt line_error
    mov x14, #0
    mov x18, #0

digit_loop:
    cmp x18, x17
    beq digits_done
    ldrb w9, [x15, x18]
    cmp w9, #48
    blt line_error
    cmp w9, #57
    bgt line_error
    sub w9, w9, #48
    mov x13, #10
    mul x14, x14, x13
    add x14, x14, x9
    add x18, x18, #1
    b digit_loop

digits_done:
    cbz x16, push_value
    neg x14, x14

push_value:
    str x14, [x25, x26, lsl #3]
    add x26, x26, #1
    b parse_loop

operator_token:
    cmp x26, #2
    blt line_error
    sub x17, x26, #1
    ldr x11, [x25, x17, lsl #3]
    sub x17, x26, #2
    ldr x10, [x25, x17, lsl #3]
    sub x26, x26, #1
    ldrb w9, [x28]
    cmp w9, #43
    beq do_add
    cmp w9, #45
    beq do_sub
    cmp w9, #42
    beq do_mul
    cmp w9, #47
    beq do_div
    cmp w9, #37
    beq do_rem
    b line_error

do_add:
    add x14, x10, x11
    b store_result

do_sub:
    sub x14, x10, x11
    b store_result

do_mul:
    mul x14, x10, x11
    b store_result

do_div:
    cbz x11, line_error
    mov x12, #1
    lsl x12, x12, #63
    cmp x10, x12
    bne division_safe
    cmn x11, #1
    beq line_error
division_safe:
    sdiv x14, x10, x11
    b store_result

do_rem:
    cbz x11, line_error
    mov x12, #1
    lsl x12, x12, #63
    cmp x10, x12
    bne remainder_safe
    cmn x11, #1
    beq line_error
remainder_safe:
    sdiv x13, x10, x11
    msub x14, x13, x11, x10
    b store_result

store_result:
    sub x17, x26, #1
    str x14, [x25, x17, lsl #3]
    b parse_loop

line_done:
    cmp x26, #1
    bne line_error
    ldr x0, [x25]
    b print_number

line_error:
    adrp x1, error_string
    add x1, x1, :lo12:error_string
    mov x2, #ERROR_LEN
    b write_output

print_number:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x18, x1, #OUTPUT_SIZE
    mov x15, x18
    mov x16, #0
    cmp x0, #0
    bge positive_number
    mov x16, #1
    neg x14, x0
    b convert_number

positive_number:
    mov x14, x0

convert_number:
    mov x13, #10

convert_loop:
    udiv x17, x14, x13
    msub x9, x17, x13, x14
    add w9, w9, #48
    strb w9, [x15, #-1]!
    mov x14, x17
    cbnz x14, convert_loop
    cbz x16, append_newline
    mov w9, #45
    strb w9, [x15, #-1]!

append_newline:
    mov w9, #10
    strb w9, [x18]
    sub x2, x18, x15
    mov x1, x15

write_output:
    mov x0, #1
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_output
    b next_line

finish:
    mov x0, #0
    mov x8, #93
    svc #0
    b finish

.section .rodata
error_string:
    .ascii "error\n"

.section .bss
.align 3
input_buf:
    .zero INPUT_SIZE
value_stack:
    .zero VALUE_STACK_SIZE
output_buf:
    .zero OUTPUT_SIZE
