.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1001

.section .text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0

read_loop:
    mov x2, #INPUT_SIZE
    sub x2, x2, x21
    cbz x2, parse_input
    mov x0, #0
    add x1, x20, x21
    mov x8, #63
    svc #0
    cbz x0, parse_input
    add x21, x21, x0
    b read_loop

parse_input:
    adrp x27, line_starts
    add x27, x27, :lo12:line_starts
    adrp x28, line_lengths
    add x28, x28, :lo12:line_lengths
    mov x22, x20
    add x23, x20, x21
    mov x24, #0
    mov x26, x20

parse_loop:
    cmp x22, x23
    b.eq parse_done
    ldrb w0, [x22], #1
    cmp w0, #10
    b.ne parse_loop
    sub x1, x22, x26
    sub x1, x1, #1
    str x26, [x27, x24, lsl #3]
    str x1, [x28, x24, lsl #3]
    add x24, x24, #1
    mov x26, x22
    b parse_loop

parse_done:
    cmp x26, x23
    b.eq sort_start
    sub x1, x23, x26
    str x26, [x27, x24, lsl #3]
    str x1, [x28, x24, lsl #3]
    add x24, x24, #1

sort_start:
    mov x25, x24
    mov x19, #0

sort_outer:
    add x17, x19, #1
    cmp x17, x25
    b.hs output_start
    mov x18, x19

sort_inner:
    cmp x17, x25
    b.hs select_done

    ldr x10, [x27, x18, lsl #3]
    ldr x11, [x28, x18, lsl #3]
    ldr x12, [x27, x17, lsl #3]
    ldr x13, [x28, x17, lsl #3]

    cmp x11, x13
    csel x15, x11, x13, lo
    mov x14, #0

compare_bytes:
    cmp x14, x15
    b.hs compare_lengths
    ldrb w8, [x10, x14]
    ldrb w9, [x12, x14]
    cmp w8, w9
    b.ne bytes_differ
    add x14, x14, #1
    b compare_bytes

bytes_differ:
    cmp w9, w8
    b.lo candidate_less
    b candidate_not_less

compare_lengths:
    cmp x13, x11
    b.lo candidate_less
    b candidate_not_less

candidate_less:
    mov x18, x17

candidate_not_less:
    add x17, x17, #1
    b sort_inner

select_done:
    cmp x18, x19
    b.eq next_outer

    ldr x10, [x27, x19, lsl #3]
    ldr x11, [x28, x19, lsl #3]
    ldr x12, [x27, x18, lsl #3]
    ldr x13, [x28, x18, lsl #3]
    str x12, [x27, x19, lsl #3]
    str x13, [x28, x19, lsl #3]
    str x10, [x27, x18, lsl #3]
    str x11, [x28, x18, lsl #3]

next_outer:
    add x19, x19, #1
    b sort_outer

output_start:
    adrp x29, newline
    add x29, x29, :lo12:newline
    mov x19, #0

output_loop:
    cmp x19, x25
    b.hs program_exit
    ldr x10, [x27, x19, lsl #3]
    ldr x11, [x28, x19, lsl #3]

write_line:
    cbz x11, write_newline
    mov x0, #1
    mov x1, x10
    mov x2, x11
    mov x8, #64
    svc #0
    cbz x0, write_line
    add x10, x10, x0
    sub x11, x11, x0
    b write_line

write_newline:
    mov x0, #1
    mov x1, x29
    mov x2, #1
    mov x8, #64
    svc #0
    cbz x0, write_newline
    add x19, x19, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.section .bss
.balign 8
input_buf:
    .zero INPUT_SIZE
line_starts:
    .zero MAX_LINES * 8
line_lengths:
    .zero MAX_LINES * 8
