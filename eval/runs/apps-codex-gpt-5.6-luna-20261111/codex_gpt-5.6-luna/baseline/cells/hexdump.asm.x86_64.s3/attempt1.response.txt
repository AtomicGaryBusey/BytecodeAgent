.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 128

.globl _start

.section .rodata
hex:
.ascii "0123456789abcdef"

.section .bss
.align 16
input:
.zero INPUT_SIZE
extra:
.zero 1
output:
.zero OUTPUT_SIZE

.text
_start:
    lea r10, [rip + input]
    xor ebx, ebx

read_loop:
    mov edx, INPUT_SIZE
    sub edx, ebx
    jz read_extra
    xor edi, edi
    mov rsi, r10
    add rsi, rbx
    xor eax, eax
    syscall
    test eax, eax
    jz input_done
    add ebx, eax
    cmp ebx, INPUT_SIZE
    jb read_loop

read_extra:
    xor edi, edi
    lea rsi, [rip + extra]
    mov edx, 1
    xor eax, eax
    syscall

input_done:
    mov r12d, ebx
    xor r13d, r13d
    lea r11, [rip + hex]

row_loop:
    cmp r13d, r12d
    jae final_offset

    mov eax, r13d
    mov ecx, 28
    lea r15, [rip + output]

offset_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r11 + rdx]
    mov byte ptr [r15], dl
    inc r15
    sub ecx, 4
    jns offset_loop

    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 32
    inc r15

    mov eax, r12d
    sub eax, r13d
    mov r14d, 16
    cmp eax, 16
    cmovb r14d, eax

    xor r8d, r8d
slot_loop:
    cmp r8d, r14d
    jae empty_slot

    mov edx, r13d
    add edx, r8d
    movzx eax, byte ptr [r10 + rdx]

    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r11 + rdx]
    mov byte ptr [r15], dl
    inc r15

    and eax, 15
    movzx eax, byte ptr [r11 + rax]
    mov byte ptr [r15], al
    inc r15

    mov byte ptr [r15], 32
    inc r15
    jmp slot_next

empty_slot:
    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 32
    inc r15

slot_next:
    inc r8d
    cmp r8d, 8
    jne no_middle_space
    mov byte ptr [r15], 32
    inc r15

no_middle_space:
    cmp r8d, 16
    jb slot_loop

    mov byte ptr [r15], 32
    inc r15
    mov byte ptr [r15], 124
    inc r15

    xor r8d, r8d
ascii_loop:
    cmp r8d, r14d
    jae ascii_done

    mov edx, r13d
    add edx, r8d
    movzx eax, byte ptr [r10 + rdx]
    cmp eax, 32
    jb nonprintable
    cmp eax, 126
    ja nonprintable
    mov byte ptr [r15], al
    inc r15
    inc r8d
    jmp ascii_loop

nonprintable:
    mov byte ptr [r15], 46
    inc r15
    inc r8d
    jmp ascii_loop

ascii_done:
    mov byte ptr [r15], 124
    inc r15
    mov byte ptr [r15], 10
    inc r15

    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r13d, r14d
    jmp row_loop

final_offset:
    lea r15, [rip + output]
    mov eax, r12d
    mov ecx, 28

final_hex_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    movzx edx, byte ptr [r11 + rdx]
    mov byte ptr [r15], dl
    inc r15
    sub ecx, 4
    jns final_hex_loop

    mov byte ptr [r15], 10
    inc r15

    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall
