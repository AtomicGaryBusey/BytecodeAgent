.equ input_capacity, 65537
.equ read_chunk, 4096

.globl _start

.section .text

_start:
    xor r13, r13

read_loop:
    mov eax, input_capacity
    sub rax, r13
    mov edx, read_chunk
    cmp rax, rdx
    cmovb rdx, rax
    lea rsi, [rip + input]
    add rsi, r13
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    mov r12, r13
    xor r13, r13
    xor r14, r14
    lea r9, [rip + hex]

row_loop:
    cmp r13, r12
    je final_offset

    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe row_length_ready
    mov r15d, 16

row_length_ready:
    lea rbx, [rip + row]
    mov r14, r13
    mov ecx, 7

offset_digits:
    mov rdx, r14
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rbx + rcx], al
    shr r14, 4
    dec ecx
    jns offset_digits

    mov byte ptr [rbx + 8], 32
    mov byte ptr [rbx + 9], 32
    xor r8, r8

byte_loop:
    cmp r8, 16
    je bytes_done
    cmp r8, r15
    jae empty_slot

    lea rdx, [r8 * 3 + 10]
    cmp r8, 8
    jb slot_position_ready
    add rdx, 1

slot_position_ready:
    lea rsi, [rip + input]
    movzx r10d, byte ptr [rsi + r13 + r8]

    mov eax, r10d
    shr eax, 4
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rbx + rdx], al

    mov eax, r10d
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rbx + rdx + 1], al
    mov byte ptr [rbx + rdx + 2], 32

    cmp r10d, 32
    jb nonprintable
    cmp r10d, 126
    ja nonprintable
    mov byte ptr [rbx + r8 + 61], r10b
    jmp next_byte

nonprintable:
    mov byte ptr [rbx + r8 + 61], 46
    jmp next_byte

empty_slot:
    lea rdx, [r8 * 3 + 10]
    cmp r8, 8
    jb empty_position_ready
    add rdx, 1

empty_position_ready:
    mov byte ptr [rbx + rdx], 32
    mov byte ptr [rbx + rdx + 1], 32
    mov byte ptr [rbx + rdx + 2], 32

next_byte:
    inc r8
    jmp byte_loop

bytes_done:
    mov byte ptr [rbx + 59], 32
    mov byte ptr [rbx + 60], 124
    mov byte ptr [rbx + r15 + 61], 124
    mov byte ptr [rbx + r15 + 62], 10

    lea rsi, [rip + row]
    lea rdx, [r15 + 63]
    call write_all

    add r13, r15
    jmp row_loop

final_offset:
    lea rbx, [rip + final]
    mov r14, r12
    mov ecx, 7

final_digits:
    mov rdx, r14
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rbx + rcx], al
    shr r14, 4
    dec ecx
    jns final_digits

    mov byte ptr [rbx + 8], 10
    lea rsi, [rip + final]
    mov edx, 9
    call write_all

    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_done

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jz write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop

write_done:
    ret

.section .rodata

hex:
    .ascii "0123456789abcdef"

.section .bss

.align 8
input:
    .zero input_capacity

.align 8
row:
    .zero 80

.align 8
final:
    .zero 9
