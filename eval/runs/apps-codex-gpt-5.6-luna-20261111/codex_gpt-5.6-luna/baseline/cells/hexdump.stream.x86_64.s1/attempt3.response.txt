.equ input_capacity, 65536
.equ read_chunk, 4096

.globl _start

.section .text

_start:
    xor r13, r13

read_loop:
    cmp r13, input_capacity
    jae read_done
    mov edx, read_chunk
    mov eax, input_capacity
    sub rax, r13
    cmp rax, rdx
    cmovb rdx, rax
    lea rsi, [rip + input]
    add rsi, r13
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    mov r12, r13
    xor r13, r13
    lea r9, [rip + hex]
    lea r10, [rip + input]

row_loop:
    cmp r13, r12
    je final_offset

    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe row_length_ready
    mov r15d, 16

row_length_ready:
    lea rbx, [rip + row]
    mov r14, r13
    mov ecx, 7

offset_digits:
    mov rdx, r14
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rbx + rcx], al
    shr r14, 4
    dec ecx
    jns offset_digits

    mov byte ptr [rbx + 59], 32
    mov byte ptr [rbx + 60], 124
    xor r8, r8

byte_loop:
    cmp r8, 16
    je bytes_done
    cmp r8, r15
    jae empty_slot

    lea rdx, [r8 + r8*2 + 10]
    cmp r8, 8
    jb byte_position_ready
    inc rdx

byte_position_ready:
    movzx eax, byte ptr [r10 + r8]
    mov r11d, eax
    shr eax, 4
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rbx + rdx], al
    and r11d, 15
    mov al, byte ptr [r9 + r11]
    mov byte ptr [rbx + rdx + 1], al
    mov byte ptr [rbx + rdx + 2], 32
    inc r8
    jmp byte_loop

empty_slot:
    lea rdx, [r8 + r8*2 + 10]
    cmp r8, 8
    jb empty_position_ready
    inc rdx

empty_position_ready:
    mov byte ptr [rbx + rdx], 32
    mov byte ptr [rbx + rdx + 1], 32
    mov byte ptr [rbx + rdx + 2], 32
    inc r8
    jmp byte_loop

bytes_done:
    xor r8, r8

ascii_loop:
    cmp r8, r15
    jae ascii_done
    movzx eax, byte ptr [r10 + r13 + r8]
    cmp eax, 32
    jb nonprintable
    cmp eax, 126
    ja nonprintable
    jmp ascii_store

nonprintable:
    mov eax, 46

ascii_store:
    lea rdx, [r8 + 61]
    mov byte ptr [rbx + rdx], al
    inc r8
    jmp ascii_loop

ascii_done:
    lea rdx, [r15 + 61]
    mov byte ptr [rbx + rdx], 124
    inc rdx
    mov byte ptr [rbx + rdx], 10

    mov r14, r15
    add r14, 63
    lea rsi, [rip + row]

write_row:
    test r14, r14
    jz row_written
    mov edx, r14d
    mov edi, 1
    mov eax, 1
    syscall
    add rsi, rax
    sub r14, rax
    jmp write_row

row_written:
    add r13, r15
    jmp row_loop

final_offset:
    lea rbx, [rip + row]
    mov r14, r12
    mov ecx, 7

final_digits:
    mov rdx, r14
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rbx + rcx], al
    shr r14, 4
    dec ecx
    jns final_digits

    mov byte ptr [rbx + 8], 10
    lea rsi, [rip + row]
    mov r14d, 9

write_final:
    test r14, r14
    jz exit_program
    mov edx, r14d
    mov edi, 1
    mov eax, 1
    syscall
    add rsi, rax
    sub r14, rax
    jmp write_final

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hex:
    .ascii "0123456789abcdef"

.section .bss
.align 8
input:
    .zero 65536
row:
    .zero 80
