.equ input_capacity, 65536
.equ read_chunk, 4096

.globl _start

.section .text

_start:
    xor r13, r13

read_loop:
    cmp r13, input_capacity
    jae read_done
    mov eax, input_capacity
    sub rax, r13
    mov edx, read_chunk
    cmp rax, rdx
    cmovb rdx, rax
    lea rsi, [rip + input]
    add rsi, r13
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    mov r12, r13
    xor r13, r13
    lea r9, [rip + hex]
    lea r10, [rip + input]

row_loop:
    cmp r13, r12
    jae final_offset

    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe row_length_ready
    mov r15d, 16

row_length_ready:
    lea rbx, [rip + row]
    mov r14, r13
    mov ecx, 7

offset_digits:
    mov rdx, r14
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rbx + rcx], al
    shr r14, 4
    dec ecx
    jns offset_digits

    mov byte ptr [rbx + 59], 32
    mov byte ptr [rbx + 60], 124
    lea r14, [r10 + r13]
    xor r8, r8

byte_loop:
    cmp r8, 16
    jae bytes_done
    cmp r8, r15
    jae empty_slot

    lea rdx, [r8 + r8*2 + 10]
    cmp r8, 8
    jb byte_position_ready
    inc rdx

byte_position_ready:
    movzx eax, byte ptr [r14 + r8]
    mov r11d, eax
    shr eax, 4
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rbx + rdx], al
    and r11d, 15
    mov al, byte ptr [r9 + r11]
    mov byte ptr [rbx + rdx + 1], al
    mov byte ptr [rbx + rdx + 2], 32
    inc r8
    jmp byte_loop

empty_slot:
    lea rdx, [r8 + r8*2 + 10]
    cmp r8, 8
    jb empty_position_ready
    inc rdx

empty_position_ready:
    mov byte ptr [rbx + rdx], 32
    mov byte ptr [rbx + rdx + 1], 32
    mov byte ptr [rbx + rdx + 2], 32
    inc r8
    jmp byte_loop

bytes_done:
    xor r8, r8

ascii_loop:
    cmp r8, r15
    jae ascii_done
    movzx eax, byte ptr [r14 + r8]
    cmp eax, 32
    jb ascii_dot
    cmp eax, 126
    ja ascii_dot
    mov byte ptr [rbx + r8 + 61], al
    inc r8
    jmp ascii_loop

ascii_dot:
    mov byte ptr [rbx + r8 + 61], 46
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rbx + r15 + 61], 124
    mov byte ptr [rbx + r15 + 62], 10
    add r13, r15
    lea r15, [r15 + 62]
    xor r8, r8

write_row:
    cmp r8, r15
    jae row_loop
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r8]
    mov rdx, r15
    sub rdx, r8
    syscall
    add r8, rax
    jmp write_row

final_offset:
    lea rbx, [rip + row]
    mov r14, r12
    mov ecx, 7

final_digits:
    mov rdx, r14
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rbx + rcx], al
    shr r14, 4
    dec ecx
    jns final_digits

    mov byte ptr [rbx + 8], 10
    xor r8, r8

write_final:
    cmp r8, 9
    jae exit_ok
    mov eax, 1
    mov edi, 1
    lea rsi, [rbx + r8]
    mov edx, 9
    sub rdx, r8
    syscall
    add r8, rax
    jmp write_final

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input:
    .zero 65536
row:
    .zero 80
