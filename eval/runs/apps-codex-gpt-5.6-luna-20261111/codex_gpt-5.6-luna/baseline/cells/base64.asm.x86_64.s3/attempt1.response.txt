.equ input_size, 49152
.equ output_size, 70000

.section .text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_loop:
    mov edx, input_size
    sub rdx, r13
    jz input_done
    lea rsi, [r12 + r13]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add r13, rax
    jmp read_loop

input_done:
    lea r14, [rip + output_buffer]
    lea r10, [rip + alphabet]
    xor ebx, ebx
    xor r15d, r15d
    xor ebp, ebp

encode_loop:
    cmp rbx, r13
    jae encode_finish

    mov eax, r13d
    sub eax, ebx
    cmp eax, 3
    jb partial_group

    movzx eax, byte ptr [r12 + rbx]
    shl eax, 16
    movzx edx, byte ptr [r12 + rbx + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [r12 + rbx + 2]
    or eax, edx
    mov r8d, eax

    mov eax, r8d
    shr eax, 18
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov eax, r8d
    shr eax, 12
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov eax, r8d
    shr eax, 6
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov eax, r8d
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15

    add rbx, 3
    add ebp, 4
    cmp ebp, 76
    jne encode_loop
    mov byte ptr [r14 + r15], 10
    inc r15
    xor ebp, ebp
    jmp encode_loop

partial_group:
    mov r9d, eax
    movzx eax, byte ptr [r12 + rbx]
    shl eax, 16
    cmp r9d, 1
    je partial_one
    movzx edx, byte ptr [r12 + rbx + 1]
    shl edx, 8
    or eax, edx

partial_one:
    mov r8d, eax

    mov eax, r8d
    shr eax, 18
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15

    mov eax, r8d
    shr eax, 12
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15

    cmp r9d, 1
    je partial_one_padding

    mov eax, r8d
    shr eax, 6
    and eax, 63
    mov dl, byte ptr [r10 + rax]
    mov byte ptr [r14 + r15], dl
    inc r15
    mov byte ptr [r14 + r15], 61
    inc r15
    jmp partial_newline

partial_one_padding:
    mov byte ptr [r14 + r15], 61
    inc r15
    mov byte ptr [r14 + r15], 61
    inc r15

partial_newline:
    mov byte ptr [r14 + r15], 10
    inc r15
    jmp output_prepare

encode_finish:
    test ebp, ebp
    jz output_prepare
    mov byte ptr [r14 + r15], 10
    inc r15

output_prepare:
    test r15, r15
    jz program_exit
    mov rsi, r14
    mov rdx, r15

write_loop:
    mov edi, 1
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop

program_exit:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 8
input_buffer:
    .zero input_size
output_buffer:
    .zero output_size
