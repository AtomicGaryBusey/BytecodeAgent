.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 65537

.globl _start

.section .bss
.align 16
input_buffer:
.zero INPUT_SIZE
output_buffer:
.zero OUTPUT_SIZE

.text
_start:
    lea r12, [rip + input_buffer]
    xor r13, r13

read_more:
    cmp r13, INPUT_SIZE
    je read_full
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, INPUT_SIZE
    sub rdx, r13
    syscall
    test rax, rax
    je input_done
    add r13, rax
    jmp read_more

read_full:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    xor edx, edx
    syscall

input_done:
    xor r14, r14

find_line:
    cmp r14, r13
    jae final_line
    mov rsi, r14

scan_line:
    cmp rsi, r13
    jae final_line
    cmp byte ptr [r12 + rsi], 10
    je newline_found
    inc rsi
    jmp scan_line

newline_found:
    mov r15, rsi
    lea rbx, [rip + output_buffer]
    mov rdx, rsi
    cmp rdx, r14
    je append_newline
    dec rdx

reverse_line:
    mov al, byte ptr [r12 + rdx]
    mov byte ptr [rbx], al
    inc rbx
    cmp rdx, r14
    je append_newline
    dec rdx
    jmp reverse_line

append_newline:
    mov byte ptr [rbx], 10
    inc rbx
    lea rsi, [rip + output_buffer]
    mov rdx, rbx
    sub rdx, rsi
    call write_buffer
    mov r14, r15
    inc r14
    jmp find_line

final_line:
    cmp r14, r13
    je finish
    lea rbx, [rip + output_buffer]
    mov rdx, r13
    cmp rdx, r14
    je final_append
    dec rdx

final_reverse:
    mov al, byte ptr [r12 + rdx]
    mov byte ptr [rbx], al
    inc rbx
    cmp rdx, r14
    je final_append
    dec rdx
    jmp final_reverse

final_append:
    mov byte ptr [rbx], 10
    inc rbx
    lea rsi, [rip + output_buffer]
    mov rdx, rbx
    sub rdx, rsi
    call write_buffer

finish:
    mov eax, 60
    xor edi, edi
    syscall

write_buffer:
    test rdx, rdx
    je write_done
write_again:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jne write_again
write_done:
    ret
