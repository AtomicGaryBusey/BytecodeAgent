.equ INPUT_SIZE, 32768

.section .rodata
newline:
    .byte 10

.section .bss
.align 8
input:
    .zero INPUT_SIZE
offsets:
    .zero 8000
lengths:
    .zero 8000

.text
.globl _start
_start:
    mov r15, offset input
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_SIZE
    jae parse_start
    mov edi, 0
    lea rsi, [r15 + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    mov eax, 0
    syscall
    test rax, rax
    jz parse_start
    add r12, rax
    jmp read_loop

parse_start:
    xor r14d, r14d
    xor ebx, ebx
    xor ecx, ecx

parse_loop:
    cmp rcx, r12
    jae parse_end
    movzx eax, byte ptr [r15 + rcx]
    cmp al, 10
    je parse_newline
    inc rcx
    jmp parse_loop

parse_newline:
    mov qword ptr [offsets + r14*8], rbx
    mov rax, rcx
    sub rax, rbx
    mov qword ptr [lengths + r14*8], rax
    inc r14
    lea rbx, [rcx + 1]
    inc rcx
    jmp parse_loop

parse_end:
    cmp rbx, r12
    jae sort_start
    mov qword ptr [offsets + r14*8], rbx
    mov rax, r12
    sub rax, rbx
    mov qword ptr [lengths + r14*8], rax
    inc r14

sort_start:
    cmp r14, 1
    jbe output_start
    mov edx, 1

sort_outer:
    cmp rdx, r14
    jae output_start
    mov r8, qword ptr [offsets + rdx*8]
    mov r9, qword ptr [lengths + rdx*8]
    mov rcx, rdx
    dec rcx

compare_record:
    mov r10, qword ptr [offsets + rcx*8]
    mov r11, qword ptr [lengths + rcx*8]
    mov r12, r9
    cmp r11, r12
    cmovb r12, r11
    lea rdi, [r15 + r8]
    lea rsi, [r15 + r10]
    xor ebx, ebx

compare_bytes:
    cmp rbx, r12
    jae compare_lengths
    movzx eax, byte ptr [rdi + rbx]
    movzx r13d, byte ptr [rsi + rbx]
    cmp eax, r13d
    jb shift_record
    ja insert_record
    inc rbx
    jmp compare_bytes

compare_lengths:
    cmp r9, r11
    jb shift_record
    jmp insert_record

shift_record:
    mov rax, qword ptr [offsets + rcx*8]
    mov qword ptr [offsets + rcx*8 + 8], rax
    mov rax, qword ptr [lengths + rcx*8]
    mov qword ptr [lengths + rcx*8 + 8], rax
    test rcx, rcx
    jz insert_zero
    dec rcx
    jmp compare_record

insert_zero:
    mov qword ptr [offsets], r8
    mov qword ptr [lengths], r9
    inc rdx
    jmp sort_outer

insert_record:
    lea rax, [rcx + 1]
    mov qword ptr [offsets + rax*8], r8
    mov qword ptr [lengths + rax*8], r9
    inc rdx
    jmp sort_outer

output_start:
    xor ebx, ebx

output_line:
    cmp rbx, r14
    jae finish
    mov r8, qword ptr [offsets + rbx*8]
    mov r9, qword ptr [lengths + rbx*8]
    add r8, r15

write_line:
    test r9, r9
    jz write_newline
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    mov eax, 1
    syscall
    test rax, rax
    jz write_line
    add r8, rax
    sub r9, rax
    jmp write_line

write_newline:
    mov r8, offset newline
    mov r9d, 1

write_nl_loop:
    test r9, r9
    jz next_line
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    mov eax, 1
    syscall
    test rax, rax
    jz write_nl_loop
    add r8, rax
    sub r9, rax
    jmp write_nl_loop

next_line:
    inc rbx
    jmp output_line

finish:
    xor edi, edi
    mov eax, 60
    syscall
    ud2
