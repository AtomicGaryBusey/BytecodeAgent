.equ READ_SIZE, 65536

.section .text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, READ_SIZE
    syscall
    test rax, rax
    jle input_done

    add r14, rax
    lea rsi, [rip + buffer]
    mov rcx, rax

process_loop:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    je whitespace
    cmp al, 9
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace
    cmp al, 32
    je whitespace

    test r15, r15
    jnz next_byte
    inc r13
    mov r15, 1
    jmp next_byte

whitespace:
    cmp al, 10
    jne next_byte
    inc r12

next_byte:
    dec rcx
    jnz process_loop
    jmp read_loop

input_done:
    lea r9, [rip + output]

    mov rax, r12
    lea rdi, [rip + lines_tmp_end]
    call convert_number
    mov rsi, rax
    mov rcx, rdx
    mov rdi, r9
    rep movsb
    add r9, rdx
    mov byte ptr [r9], 32
    inc r9

    mov rax, r13
    lea rdi, [rip + words_tmp_end]
    call convert_number
    mov rsi, rax
    mov rcx, rdx
    mov rdi, r9
    rep movsb
    add r9, rdx
    mov byte ptr [r9], 32
    inc r9

    mov rax, r14
    lea rdi, [rip + bytes_tmp_end]
    call convert_number
    mov rsi, rax
    mov rcx, rdx
    mov rdi, r9
    rep movsb
    add r9, rdx
    mov byte ptr [r9], 10
    inc r9

    lea rsi, [rip + output]
    mov r14, r9
    sub r14, rsi

write_loop:
    mov eax, 1
    mov edi, 1
    mov rdx, r14
    syscall
    test rax, rax
    jle finish
    add rsi, rax
    sub r14, rax
    jnz write_loop

finish:
    mov eax, 60
    xor edi, edi
    syscall

convert_number:
    mov r8d, 10
    xor ecx, ecx
    test rax, rax
    jnz convert_loop
    dec rdi
    mov byte ptr [rdi], 48
    mov rax, rdi
    mov edx, 1
    ret

convert_loop:
    xor edx, edx
    div r8
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    inc rcx
    test rax, rax
    jnz convert_loop
    mov rax, rdi
    mov rdx, rcx
    ret

.section .bss
.align 16
buffer:
    .zero READ_SIZE

lines_tmp:
    .zero 32
lines_tmp_end:

words_tmp:
    .zero 32
words_tmp_end:

bytes_tmp:
    .zero 32
bytes_tmp_end:

output:
    .zero 128
