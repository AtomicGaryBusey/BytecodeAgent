.equ BUF_SIZE, 65537

.globl _start

.section .rodata
newline:
	.byte 0x0a

.section .bss
.align 3
input_buffer:
	.zero BUF_SIZE

.text
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	adrp x29, newline
	add x29, x29, :lo12:newline

	mov x20, #0
	movz x21, #BUF_SIZE

read_loop:
	cmp x20, x21
	b.eq input_done
	mov x0, #0
	add x1, x19, x20
	sub x2, x21, x20
	mov x8, #63
	svc #0
	add x20, x20, x0
	cbnz x0, read_loop

input_done:
	mov x22, #0

next_line:
	cmp x22, x20
	b.eq program_exit
	mov x23, x22

find_newline:
	cmp x23, x20
	b.eq final_line
	ldrb w24, [x19, x23]
	cmp w24, #10
	b.eq found_newline
	add x23, x23, #1
	b find_newline

found_newline:
	mov x24, x23
	mov x17, #0
	b process_line

final_line:
	mov x24, x20
	mov x17, #1

process_line:
	add x25, x19, x22
	add x26, x19, x24
	sub x26, x26, #1

reverse_loop:
	cmp x25, x26
	b.hs reversed
	ldrb w27, [x25]
	ldrb w28, [x26]
	strb w28, [x25]
	strb w27, [x26]
	add x25, x25, #1
	sub x26, x26, #1
	b reverse_loop

reversed:
	add x1, x19, x22
	sub x2, x24, x22
	bl write_all

	mov x1, x29
	mov x2, #1
	bl write_all

	cbnz x17, program_exit
	add x22, x24, #1
	b next_line

write_all:
write_loop:
	cbz x2, write_return
	mov x0, #1
	mov x8, #64
	svc #0
	add x1, x1, x0
	sub x2, x2, x0
	b write_loop

write_return:
	ret

program_exit:
	mov x0, #0
	mov x8, #93
	svc #0
fatal:
	b fatal
