.equ BUF_SIZE, 65536

.globl _start

.section .rodata
newline:
	.byte 0x0a

.section .bss
.align 3
input_buffer:
	.zero BUF_SIZE

.text
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer

	adrp x18, newline
	add x18, x18, :lo12:newline

	mov x20, #0

read_loop:
	mov x2, #BUF_SIZE
	sub x2, x2, x20
	cbz x2, process_input
	add x1, x19, x20
	mov x0, #0
	mov x8, #63
	svc #0
	cbz x0, process_input
	add x20, x20, x0
	b read_loop

process_input:
	mov x21, #0

find_line:
	cmp x21, x20
	beq final_line

	mov x22, x21

scan_line:
	cmp x22, x20
	beq final_line
	ldrb w23, [x19, x22]
	cmp w23, #10
	beq found_newline
	add x22, x22, #1
	b scan_line

found_newline:
	mov x23, x22
	sub x24, x23, x21
	mov x29, #0
	b reverse_line

final_line:
	mov x23, x20
	sub x24, x23, x21
	cbz x24, exit_program
	mov x29, #1

reverse_line:
	cmp x24, #1
	b.ls reversed

	mov x25, x21
	add x26, x21, x24
	sub x26, x26, #1

reverse_loop:
	cmp x25, x26
	b.hs reversed
	ldrb w27, [x19, x25]
	ldrb w28, [x19, x26]
	strb w28, [x19, x25]
	strb w27, [x19, x26]
	add x25, x25, #1
	sub x26, x26, #1
	b reverse_loop

reversed:
	add x1, x19, x21
	mov x2, x24
	bl write_all

	mov x1, x18
	mov x2, #1
	bl write_all

	cbnz x29, exit_program
	add x21, x23, #1
	b find_line

exit_program:
	mov x0, #0
	mov x8, #93
	svc #0

write_all:
	mov x27, x1
	mov x28, x2

write_loop:
	cbz x28, write_done
	mov x0, #1
	mov x1, x27
	mov x2, x28
	mov x8, #64
	svc #0
	add x27, x27, x0
	sub x28, x28, x0
	b write_loop

write_done:
	ret
