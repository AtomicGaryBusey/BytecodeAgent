.equ INPUT_CAP, 65536
.equ OUTPUT_CAP, 65537

.section .bss
.balign 16
buffer:
.zero INPUT_CAP
output:
.zero OUTPUT_CAP

.text
.globl _start
_start:
	lea rbx, [rip + buffer]
	xor r12d, r12d

read_more:
	cmp r12, INPUT_CAP
	jae input_done
	mov eax, 0
	xor edi, edi
	lea rsi, [rbx + r12]
	mov edx, INPUT_CAP
	sub rdx, r12
	syscall
	test rax, rax
	jz input_done
	add r12, rax
	jmp read_more

input_done:
	xor r8d, r8d
	xor r9d, r9d
	lea r10, [rip + output]

parse_loop:
	cmp r8, r12
	jae final_line
	cmp byte ptr [rbx + r8], 10
	je newline_found
	inc r8
	jmp parse_loop

newline_found:
	mov r11, r8

reverse_line:
	cmp r11, r9
	je reversed_line
	dec r11
	mov al, byte ptr [rbx + r11]
	mov byte ptr [r10], al
	inc r10
	jmp reverse_line

reversed_line:
	mov byte ptr [r10], 10
	inc r10
	inc r8
	mov r9, r8
	jmp parse_loop

final_line:
	cmp r9, r12
	je output_ready
	mov r11, r12

reverse_final:
	cmp r11, r9
	je final_reversed
	dec r11
	mov al, byte ptr [rbx + r11]
	mov byte ptr [r10], al
	inc r10
	jmp reverse_final

final_reversed:
	mov byte ptr [r10], 10
	inc r10

output_ready:
	lea r14, [rip + output]
	mov r15, r10
	sub r15, r14

write_loop:
	test r15, r15
	jz done
	mov eax, 1
	mov edi, 1
	mov rsi, r14
	mov rdx, r15
	syscall
	add r14, rax
	sub r15, rax
	jmp write_loop

done:
	mov eax, 60
	xor edi, edi
	syscall
