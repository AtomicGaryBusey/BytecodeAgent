.equ BUF_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_SIZE, 524288
.equ STACK_SIZE, 8000

.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    movz x19, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x19
    cbz x2, input_done
    movz x0, #0
    add x1, x20, x19
    movz x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    b read_loop

input_done:
    movz x27, #0
find_delimiter:
    cmp x27, x19
    b.ge no_delimiter
    ldrb w0, [x20, x27]
    cmp w0, #0x21
    b.eq delimiter_found
    add x27, x27, #1
    b find_delimiter

delimiter_found:
    mov x21, x27
    add x22, x20, x27
    add x22, x22, #1
    add x23, x20, x19
    b validate_setup

no_delimiter:
    mov x21, x19
    add x22, x20, x19
    mov x23, x22

validate_setup:
    adrp x24, bracket_stack
    add x24, x24, :lo12:bracket_stack
    adrp x25, jump_table
    add x25, x25, :lo12:jump_table
    movz x26, #0
    movz x27, #0

validate_loop:
    cmp x27, x21
    b.ge validate_done
    ldrb w0, [x20, x27]
    cmp w0, #0x5b
    b.eq validate_open
    cmp w0, #0x5d
    b.eq validate_close
    add x27, x27, #1
    b validate_loop

validate_open:
    str x27, [x24, x26, lsl #3]
    add x26, x26, #1
    add x27, x27, #1
    b validate_loop

validate_close:
    cbz x26, invalid_program
    sub x26, x26, #1
    ldr x28, [x24, x26, lsl #3]
    add x0, x27, #1
    str x0, [x25, x28, lsl #3]
    add x0, x28, #1
    str x0, [x25, x27, lsl #3]
    add x27, x27, #1
    b validate_loop

validate_done:
    cbnz x26, invalid_program

    adrp x26, tape
    add x26, x26, :lo12:tape
    adrp x29, output_buf
    add x29, x29, :lo12:output_buf
    movz x27, #0
    movz x28, #0
    movz x30, #0

execute_loop:
    cmp x27, x21
    b.ge program_done
    ldrb w0, [x20, x27]
    cmp w0, #0x3e
    b.eq command_right
    cmp w0, #0x3c
    b.eq command_left
    cmp w0, #0x2b
    b.eq command_add
    cmp w0, #0x2d
    b.eq command_sub
    cmp w0, #0x2e
    b.eq command_output
    cmp w0, #0x2c
    b.eq command_input
    cmp w0, #0x5b
    b.eq command_open
    cmp w0, #0x5d
    b.eq command_close
    add x27, x27, #1
    b execute_loop

command_right:
    add x28, x28, #1
    add x27, x27, #1
    b execute_loop

command_left:
    sub x28, x28, #1
    add x27, x27, #1
    b execute_loop

command_add:
    ldrb w1, [x26, x28]
    add w1, w1, #1
    strb w1, [x26, x28]
    add x27, x27, #1
    b execute_loop

command_sub:
    ldrb w1, [x26, x28]
    sub w1, w1, #1
    strb w1, [x26, x28]
    add x27, x27, #1
    b execute_loop

command_output:
    ldrb w1, [x26, x28]
    strb w1, [x29, x30]
    add x30, x30, #1
    add x27, x27, #1
    b execute_loop

command_input:
    cmp x22, x23
    b.ge input_exhausted
    ldrb w1, [x22], #1
    strb w1, [x26, x28]
    add x27, x27, #1
    b execute_loop

input_exhausted:
    strb wzr, [x26, x28]
    add x27, x27, #1
    b execute_loop

command_open:
    ldrb w1, [x26, x28]
    cbnz w1, open_continue
    ldr x27, [x25, x27, lsl #3]
    b execute_loop

open_continue:
    add x27, x27, #1
    b execute_loop

command_close:
    ldrb w1, [x26, x28]
    cbz w1, close_continue
    ldr x27, [x25, x27, lsl #3]
    b execute_loop

close_continue:
    add x27, x27, #1
    b execute_loop

program_done:
    cbz x30, successful_exit
    movz x19, #0

write_loop:
    cmp x19, x30
    b.ge successful_exit
    movz x0, #1
    add x1, x29, x19
    sub x2, x30, x19
    movz x8, #64
    svc #0
    cbz x0, successful_exit
    add x19, x19, x0
    b write_loop

invalid_program:
    movz x0, #1
    movz x8, #93
    svc #0

successful_exit:
    movz x0, #0
    movz x8, #93
    svc #0

.section .bss
.balign 16
input_buf:
    .zero BUF_SIZE
jump_table:
    .zero JUMP_SIZE
bracket_stack:
    .zero STACK_SIZE
tape:
    .zero TAPE_SIZE
output_buf:
    .zero BUF_SIZE
