.section .rodata
hex_digits:
	.ascii "0123456789abcdef"

.section .bss
	.align 4
input_buffer:
	.zero 65536
output_buffer:
	.zero 128

.text
.globl _start
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	mov x20, #0

read_loop:
	movz x24, #1, lsl #16
	cmp x20, x24
	b.eq input_done
	sub x2, x24, x20
	mov x0, #0
	add x1, x19, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	b.le input_done
	add x20, x20, x0
	b read_loop

input_done:
	mov x21, #0

row_loop:
	cmp x21, x20
	b.hs rows_done

	adrp x22, output_buffer
	add x22, x22, :lo12:output_buffer
	mov x25, x22

	mov x23, #0
	mov x24, #28
	adrp x27, hex_digits
	add x27, x27, :lo12:hex_digits

offset_digits:
	lsr x26, x21, x24
	and x26, x26, #15
	ldrb w26, [x27, x26]
	strb w26, [x22], #1
	subs x24, x24, #4
	b.ge offset_digits

	mov w26, #32
	strb w26, [x22], #1
	strb w26, [x22], #1

	sub x24, x20, x21
	cmp x24, #16
	b.ls row_length_ready
	mov x24, #16
row_length_ready:

hex_loop:
	cmp x23, #16
	b.hs hex_done
	cmp x23, x24
	b.hs hex_blank

	add x26, x19, x21
	add x26, x26, x23
	ldrb w26, [x26]
	lsr w28, w26, #4
	ldrb w28, [x27, x28]
	strb w28, [x22], #1
	and w28, w26, #15
	ldrb w28, [x27, x28]
	strb w28, [x22], #1
	mov w28, #32
	strb w28, [x22], #1
	b hex_slot_done

hex_blank:
	mov w28, #32
	strb w28, [x22], #1
	strb w28, [x22], #1
	strb w28, [x22], #1

hex_slot_done:
	cmp x23, #7
	b.ne no_group_space
	mov w28, #32
	strb w28, [x22], #1
no_group_space:
	add x23, x23, #1
	b hex_loop

hex_done:
	mov w28, #32
	strb w28, [x22], #1
	mov w28, #124
	strb w28, [x22], #1

	mov x23, #0
ascii_loop:
	cmp x23, x24
	b.hs ascii_done
	add x26, x19, x21
	add x26, x26, x23
	ldrb w28, [x26]
	cmp w28, #32
	b.lo ascii_dot
	cmp w28, #126
	b.hi ascii_dot
	strb w28, [x22], #1
	b ascii_next
ascii_dot:
	mov w28, #46
	strb w28, [x22], #1
ascii_next:
	add x23, x23, #1
	b ascii_loop

ascii_done:
	mov w28, #124
	strb w28, [x22], #1
	mov w28, #10
	strb w28, [x22], #1

	sub x1, x22, x25
	mov x0, x25
	bl write_all

	add x21, x21, x24
	b row_loop

rows_done:
	adrp x22, output_buffer
	add x22, x22, :lo12:output_buffer
	mov x25, x22
	mov x23, #28
	adrp x27, hex_digits
	add x27, x27, :lo12:hex_digits

final_digits:
	lsr x26, x20, x23
	and x26, x26, #15
	ldrb w26, [x27, x26]
	strb w26, [x22], #1
	subs x23, x23, #4
	b.ge final_digits

	mov w26, #10
	strb w26, [x22], #1
	sub x1, x22, x25
	mov x0, x25
	bl write_all

	mov x0, #0
	mov x8, #93
	svc #0

write_all:
	mov x4, x0
	mov x3, x1
write_loop:
	cbz x3, write_done
	mov x0, #1
	mov x1, x4
	mov x2, x3
	mov x8, #64
	svc #0
	cmp x0, #0
	b.le write_done
	add x4, x4, x0
	sub x3, x3, x0
	b write_loop
write_done:
	ret
