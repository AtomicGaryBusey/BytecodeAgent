.equ IN_SIZE, 65536
.equ STACK_SIZE, 135168

.globl _start

.text

_start:
    sub sp, sp, #STACK_SIZE
    mov x21, sp
    add x22, x21, #IN_SIZE
    mov x20, #0

read_loop:
    cmp x20, #IN_SIZE
    b.eq read_done

    mov x0, #0
    add x1, x21, x20
    mov x2, #IN_SIZE
    sub x2, x2, x20
    mov x8, #63
    svc #0

    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x23, #0
    mov x24, #0

line_scan:
    cmp x23, x20
    b.eq output_done
    mov x25, x23

scan_bytes:
    cmp x25, x20
    b.eq no_newline
    ldrb w26, [x21, x25]
    cmp w26, #10
    b.eq found_newline
    add x25, x25, #1
    b scan_bytes

found_newline:
    mov x26, x25

reverse_line:
    cmp x26, x23
    b.eq append_newline
    sub x26, x26, #1
    ldrb w27, [x21, x26]
    strb w27, [x22, x24]
    add x24, x24, #1
    b reverse_line

append_newline:
    mov w27, #10
    strb w27, [x22, x24]
    add x24, x24, #1
    add x23, x25, #1
    b line_scan

no_newline:
    mov x26, x25
    b reverse_line

output_done:
    mov x27, #0

write_loop:
    cmp x27, x24
    b.eq exit_program

    mov x0, #1
    add x1, x22, x27
    sub x2, x24, x27
    mov x8, #64
    svc #0
    add x27, x27, x0
    b write_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0
