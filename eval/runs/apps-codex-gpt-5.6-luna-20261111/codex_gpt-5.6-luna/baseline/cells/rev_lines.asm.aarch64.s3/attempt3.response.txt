.equ IN_SIZE, 65536
.equ OUT_SIZE, 65537

.globl _start

.text

_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x20, #0

read_loop:
    mov x9, #IN_SIZE
    cmp x20, x9
    b.eq read_done

    mov x0, #0
    add x1, x21, x20
    sub x2, x9, x20
    mov x8, #63
    svc #0

    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x23, #0
    mov x24, #0

line_scan:
    cmp x23, x20
    b.eq output_done
    mov x25, x23

scan_bytes:
    cmp x25, x20
    b.eq reverse_line_start
    ldrb w26, [x21, x25]
    cmp w26, #10
    b.eq reverse_line_start
    add x25, x25, #1
    b scan_bytes

reverse_line_start:
    mov x26, x25

reverse_line:
    cmp x26, x23
    b.eq append_newline
    sub x26, x26, #1
    ldrb w27, [x21, x26]
    strb w27, [x22, x24]
    add x24, x24, #1
    b reverse_line

append_newline:
    mov w27, #10
    strb w27, [x22, x24]
    add x24, x24, #1

    cmp x25, x20
    b.eq output_done
    add x23, x25, #1
    b line_scan

output_done:
    mov x27, #0

write_loop:
    cmp x27, x24
    b.eq exit_program

    mov x0, #1
    add x1, x22, x27
    sub x2, x24, x27
    mov x8, #64
    svc #0
    add x27, x27, x0
    b write_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.align 4
input_buf:
.zero IN_SIZE
output_buf:
.zero OUT_SIZE
