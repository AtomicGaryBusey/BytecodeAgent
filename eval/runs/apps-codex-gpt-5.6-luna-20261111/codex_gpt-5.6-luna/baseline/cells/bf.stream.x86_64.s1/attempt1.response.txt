.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ STACK_SIZE, 4000
.equ JUMP_SIZE, 262144

.section .text
.globl _start
_start:
    lea rsi, [rip + input]
    xor r12d, r12d

read_loop:
    mov rdx, INPUT_SIZE
    sub rdx, r12
    jz read_done
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov eax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    cmp r12, INPUT_SIZE
    jb read_loop

read_done:
    mov r8, r12
    lea rsi, [rip + input]
    xor ecx, ecx

find_delimiter:
    cmp rcx, r12
    jae no_delimiter
    cmp byte ptr [rsi + rcx], 0x21
    je delimiter_found
    inc rcx
    jmp find_delimiter

delimiter_found:
    mov r15, rcx
    lea rbx, [rcx + 1]
    jmp validate_start

no_delimiter:
    mov r15, r12
    mov rbx, r12

validate_start:
    lea rdi, [rip + bracket_stack]
    lea r10, [rip + jump_table]
    xor ecx, ecx
    xor r11d, r11d

validate_loop:
    cmp rcx, r15
    jae validate_done
    movzx eax, byte ptr [rsi + rcx]
    cmp al, 0x5b
    je open_bracket
    cmp al, 0x5d
    je close_bracket
    inc rcx
    jmp validate_loop

open_bracket:
    mov dword ptr [rdi + r11*4], ecx
    inc r11
    inc rcx
    jmp validate_loop

close_bracket:
    test r11, r11
    jz invalid_program
    dec r11
    mov eax, dword ptr [rdi + r11*4]
    lea edx, [rcx + 1]
    mov dword ptr [r10 + rax*4], edx
    lea eax, [rax + 1]
    mov dword ptr [r10 + rcx*4], eax
    inc rcx
    jmp validate_loop

validate_done:
    test r11, r11
    jnz invalid_program

    lea r14, [rip + tape]
    mov r13, rbx
    xor r12d, r12d

execute_loop:
    cmp r12, r15
    jae success_exit
    movzx eax, byte ptr [rsi + r12]
    cmp al, 0x3e
    je command_right
    cmp al, 0x3c
    je command_left
    cmp al, 0x2b
    je command_plus
    cmp al, 0x2d
    je command_minus
    cmp al, 0x2e
    je command_output
    cmp al, 0x2c
    je command_input
    cmp al, 0x5b
    je command_open
    cmp al, 0x5d
    je command_close
    inc r12
    jmp execute_loop

command_right:
    inc r14
    inc r12
    jmp execute_loop

command_left:
    dec r14
    inc r12
    jmp execute_loop

command_plus:
    inc byte ptr [r14]
    inc r12
    jmp execute_loop

command_minus:
    dec byte ptr [r14]
    inc r12
    jmp execute_loop

command_output:
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, 1
    syscall
    inc r12
    lea rsi, [rip + input]
    jmp execute_loop

command_input:
    cmp r13, r8
    jae input_empty
    mov al, byte ptr [rsi + r13]
    inc r13
    mov byte ptr [r14], al
    inc r12
    jmp execute_loop

input_empty:
    mov byte ptr [r14], 0
    inc r12
    jmp execute_loop

command_open:
    cmp byte ptr [r14], 0
    jne open_continue
    mov eax, dword ptr [r10 + r12*4]
    mov r12d, eax
    jmp execute_loop

open_continue:
    inc r12
    jmp execute_loop

command_close:
    cmp byte ptr [r14], 0
    je close_continue
    mov eax, dword ptr [r10 + r12*4]
    mov r12d, eax
    jmp execute_loop

close_continue:
    inc r12
    jmp execute_loop

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

success_exit:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.balign 8
input:
    .zero INPUT_SIZE
tape:
    .zero TAPE_SIZE
bracket_stack:
    .zero STACK_SIZE
jump_table:
    .zero JUMP_SIZE
