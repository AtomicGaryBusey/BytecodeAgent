.equ IN_SIZE, 65536

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    mov x21, #0

read_more:
    cmp x21, #IN_SIZE
    beq input_ready
    add x1, x19, x21
    mov x2, #IN_SIZE
    sub x2, x2, x21
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, input_ready
    add x21, x21, x0
    b read_more

input_ready:
    add x21, x19, x21
    adrp x22, values
    add x22, x22, :lo12:values
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf

next_line:
    cmp x20, x21
    beq finish
    mov x23, #0

token_loop:
    cmp x20, x21
    beq line_finish
    ldrb w27, [x20]
    cmp w27, #10
    beq line_end
    cmp w27, #32
    beq skip_separator
    cmp w27, #9
    beq skip_separator

    mov x25, x20
    mov x26, #0

scan_token:
    cmp x20, x21
    beq token_done
    ldrb w27, [x20]
    cmp w27, #10
    beq token_done
    cmp w27, #32
    beq token_done
    cmp w27, #9
    beq token_done
    add x20, x20, #1
    add x26, x26, #1
    b scan_token

token_done:
    cmp x26, #1
    bne parse_literal
    ldrb w27, [x25]
    cmp w27, #43
    beq do_operator
    cmp w27, #45
    beq do_operator
    cmp w27, #42
    beq do_operator
    cmp w27, #47
    beq do_operator
    cmp w27, #37
    beq do_operator

parse_literal:
    mov x28, #0
    ldrb w27, [x25]
    cmp w27, #45
    bne literal_digits
    cmp x26, #1
    beq do_operator
    mov x28, #1
    add x25, x25, #1
    sub x26, x26, #1

literal_digits:
    cmp x26, #1
    blt bad_line
    cmp x26, #18
    bgt bad_line
    mov x0, #0
    mov x29, #0
    mov x30, #10

literal_loop:
    ldrb w27, [x25, x29]
    cmp w27, #48
    blt bad_line
    cmp w27, #57
    bgt bad_line
    sub w27, w27, #48
    uxtw x2, w27
    mul x0, x0, x30
    add x0, x0, x2
    add x29, x29, #1
    cmp x29, x26
    bne literal_loop
    cbz x28, push_value
    neg x0, x0

push_value:
    str x0, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_loop

do_operator:
    cmp x23, #2
    blt bad_line
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]
    sub x23, x23, #1
    ldr x0, [x22, x23, lsl #3]

    cmp w27, #43
    beq operator_add
    cmp w27, #45
    beq operator_sub
    cmp w27, #42
    beq operator_mul
    cmp w27, #47
    beq operator_div
    b operator_mod

operator_add:
    add x0, x0, x1
    b operator_push

operator_sub:
    sub x0, x0, x1
    b operator_push

operator_mul:
    mul x0, x0, x1
    b operator_push

operator_div:
    cbz x1, bad_line
    mov x2, #1
    lsl x2, x2, #63
    cmp x0, x2
    bne division_safe
    mov x2, #-1
    cmp x1, x2
    beq bad_line
division_safe:
    sdiv x2, x0, x1
    mov x0, x2
    b operator_push

operator_mod:
    cbz x1, bad_line
    mov x2, #1
    lsl x2, x2, #63
    cmp x0, x2
    bne remainder_safe
    mov x2, #-1
    cmp x1, x2
    beq bad_line
remainder_safe:
    sdiv x2, x0, x1
    msub x0, x2, x1, x0

operator_push:
    str x0, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_loop

skip_separator:
    add x20, x20, #1
    b token_loop

line_end:
    add x20, x20, #1

line_finish:
    cmp x23, #1
    bne emit_error
    ldr x0, [x22]
    b emit_value

bad_line:
    cmp x20, x21
    beq emit_error
    ldrb w27, [x20]
    cmp w27, #10
    beq bad_line_end
    add x20, x20, #1
    b bad_line

bad_line_end:
    add x20, x20, #1

emit_error:
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    b write_output

emit_value:
    add x29, x24, #64
    mov x1, x29
    mov w2, #10
    strb w2, [x1, #-1]!
    mov x3, x0
    mov x4, #0
    cmp x3, #0
    bge convert_value
    neg x3, x3
    mov x4, #1

convert_value:
    mov x2, #10
    cbnz x3, convert_digits
    mov w5, #48
    strb w5, [x1, #-1]!
    b convert_sign

convert_digits:
    udiv x5, x3, x2
    msub x6, x5, x2, x3
    add w6, w6, #48
    strb w6, [x1, #-1]!
    mov x3, x5
    cbnz x3, convert_digits

convert_sign:
    cbz x4, value_ready
    mov w5, #45
    strb w5, [x1, #-1]!

value_ready:
    sub x2, x29, x1

write_output:
    mov x0, #1

write_loop:
    mov x8, #64
    svc #0
    cmp x0, x2
    beq write_done
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

write_done:
    cmp x20, x21
    bne next_line

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_text:
    .ascii "error\n"

.section .bss
.align 4
input_buf:
    .zero IN_SIZE

.align 3
values:
    .zero 8000

.align 3
output_buf:
    .zero 64
