.equ IN_SIZE, 65536

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    mov x21, #0

read_more:
    cmp x21, #IN_SIZE
    beq input_ready
    add x1, x19, x21
    mov x2, #IN_SIZE
    sub x2, x2, x21
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    beq input_ready
    add x21, x21, x0
    b read_more

input_ready:
    add x21, x19, x21
    mov x20, x19
    adrp x22, values
    add x22, x22, :lo12:values
    adrp x28, output_buf
    add x28, x28, :lo12:output_buf

next_line:
    cmp x20, x21
    beq finish
    mov x23, #0
    mov x24, #0

token_loop:
    cmp x20, x21
    beq line_finish
    ldrb w27, [x20]
    cmp w27, #10
    beq line_end
    cmp w27, #32
    beq skip_sep
    cmp w27, #9
    beq skip_sep

    mov x25, x20
    mov x26, #0

scan_token:
    cmp x20, x21
    beq token_done
    ldrb w27, [x20]
    cmp w27, #10
    beq token_done
    cmp w27, #32
    beq token_done
    cmp w27, #9
    beq token_done
    add x20, x20, #1
    add x26, x26, #1
    b scan_token

token_done:
    cmp x26, #1
    bne literal_check
    ldrb w27, [x25]
    cmp w27, #43
    beq do_operator
    cmp w27, #45
    beq do_operator
    cmp w27, #42
    beq do_operator
    cmp w27, #47
    beq do_operator
    cmp w27, #37
    beq do_operator

literal_check:
    mov x28, #0
    mov x29, #0
    ldrb w27, [x25]
    cmp w27, #45
    bne literal_start
    cmp x26, #1
    beq do_operator
    mov x28, #1
    mov x29, #1

literal_start:
    sub x26, x26, x28
    cmp x26, #1
    blt bad_line
    cmp x26, #18
    bgt bad_line
    add x25, x25, x28
    mov x0, #0
    mov x30, #10
    mov x28, #0

literal_loop:
    ldrb w27, [x25, x28]
    cmp w27, #48
    blt bad_line
    cmp w27, #57
    bgt bad_line
    sub w27, w27, #48
    mul x0, x0, x30
    add x0, x0, w27, uxtw
    add x28, x28, #1
    cmp x28, x26
    bne literal_loop
    cbz x29, push_value
    neg x0, x0

push_value:
    str x0, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_loop

do_operator:
    cmp x23, #2
    blt bad_line
    sub x23, x23, #1
    ldr x1, [x22, x23, lsl #3]
    sub x23, x23, #1
    ldr x0, [x22, x23, lsl #3]

    cmp w27, #43
    beq op_add
    cmp w27, #45
    beq op_sub
    cmp w27, #42
    beq op_mul
    cmp w27, #47
    beq op_div
    b op_mod

op_add:
    add x0, x0, x1
    b op_push

op_sub:
    sub x0, x0, x1
    b op_push

op_mul:
    mul x0, x0, x1
    b op_push

op_div:
    cbz x1, bad_line
    mov x2, #1
    lsl x2, x2, #63
    cmp x0, x2
    bne div_safe
    mov x2, #-1
    cmp x1, x2
    beq bad_line
div_safe:
    sdiv x2, x0, x1
    mov x0, x2
    b op_push

op_mod:
    cbz x1, bad_line
    mov x2, #1
    lsl x2, x2, #63
    cmp x0, x2
    bne mod_safe
    mov x2, #-1
    cmp x1, x2
    beq bad_line
mod_safe:
    sdiv x2, x0, x1
    msub x0, x2, x1, x0

op_push:
    str x0, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_loop

skip_sep:
    add x20, x20, #1
    b token_loop

line_end:
    add x20, x20, #1

line_finish:
    cmp x23, #1
    bne emit_error
    ldr x0, [x22]
    b emit_value

bad_line:
    cmp x20, x21
    beq emit_error
    ldrb w27, [x20]
    cmp w27, #10
    beq consume_bad_end
    add x20, x20, #1
    b bad_line

consume_bad_end:
    add x20, x20, #1

emit_error:
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    b write_output

emit_value:
    add x29, x28, #32
    mov x1, x29
    mov w2, #10
    strb w2, [x1, #-1]!
    mov x3, x0
    mov x4, #0
    cmp x3, #0
    bge convert_start
    neg x3, x3
    mov x4, #1

convert_start:
    mov x2, #10
    cbnz x3, convert_loop
    mov w5, #48
    strb w5, [x1, #-1]!
    b convert_sign

convert_loop:
    udiv x5, x3, x2
    msub x6, x5, x2, x3
    add w6, w6, #48
    strb w6, [x1, #-1]!
    mov x3, x5
    cbnz x3, convert_loop

convert_sign:
    cbz x4, value_ready
    mov w5, #45
    strb w5, [x1, #-1]!

value_ready:
    add x2, x29, #0
    sub x2, x2, x1

write_output:
    mov x0, #1
write_loop:
    mov x8, #64
    svc #0
    cmp x0, x2
    beq write_done
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

write_done:
    cmp x20, x21
    bne next_line

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_text:
    .ascii "error\n"

.section .bss
.align 4
input_buf:
    .zero IN_SIZE
.align 3
values:
    .zero 8000
.align 3
output_buf:
    .zero 64
