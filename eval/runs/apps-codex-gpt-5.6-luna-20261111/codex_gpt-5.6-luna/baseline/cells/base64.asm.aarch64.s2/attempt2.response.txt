.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #INPUT_SIZE
    mov x21, #0

read_loop:
    cbz x20, read_probe
    mov x0, #0
    mov x1, x19
    mov x2, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    add x21, x21, x0
    sub x20, x20, x0
    b read_loop

read_probe:
    adrp x1, probe_byte
    add x1, x1, :lo12:probe_byte
    mov x0, #0
    mov x2, #1
    mov x8, #SYS_READ
    svc #0

input_done:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x22, output_buffer
    add x22, x22, :lo12:output_buffer
    mov x25, x22
    mov x20, x21
    mov x23, #0
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet

encode_loop:
    cmp x20, #3
    b.lo encode_tail

    ldrb w0, [x19], #1
    ldrb w1, [x19], #1
    ldrb w2, [x19], #1

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    and w5, w1, #15
    lsl w5, w5, #2
    lsr w6, w2, #6
    orr w5, w5, w6
    and w6, w2, #63

    mov w0, w3
    bl emit_char
    mov w0, w4
    bl emit_char
    mov w0, w5
    bl emit_char
    mov w0, w6
    bl emit_char

    sub x20, x20, #3
    b encode_loop

encode_tail:
    cbz x20, finish_output
    cmp x20, #1
    b.eq encode_one

    ldrb w0, [x19], #1
    ldrb w1, [x19]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    lsr w5, w1, #4
    orr w4, w4, w5
    and w5, w1, #15
    lsl w5, w5, #2

    mov w0, w3
    bl emit_char
    mov w0, w4
    bl emit_char
    mov w0, w5
    bl emit_char
    mov w0, #64
    bl emit_char
    b finish_output

encode_one:
    ldrb w0, [x19]
    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4

    mov w0, w3
    bl emit_char
    mov w0, w4
    bl emit_char
    mov w0, #64
    bl emit_char
    mov w0, #64
    bl emit_char

finish_output:
    cbz x23, output_done
    mov w1, #10
    strb w1, [x22], #1

output_done:
    sub x26, x22, x25
    cbz x26, exit_program

write_loop:
    mov x0, #1
    mov x1, x25
    mov x2, x26
    mov x8, #SYS_WRITE
    svc #0
    sub x26, x26, x0
    add x25, x25, x0
    cbnz x26, write_loop

exit_program:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

emit_char:
    ldrb w1, [x24, x0]
    strb w1, [x22], #1
    add x23, x23, #1
    cmp x23, #76
    b.ne emit_done
    mov w1, #10
    strb w1, [x22], #1
    mov x23, #0

emit_done:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="

.section .bss
.balign 16
input_buffer:
    .zero INPUT_SIZE
probe_byte:
    .zero 1
.balign 16
output_buffer:
    .zero OUTPUT_SIZE
