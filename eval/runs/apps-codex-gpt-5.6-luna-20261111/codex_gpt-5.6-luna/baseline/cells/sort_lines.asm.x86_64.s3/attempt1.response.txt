.equ INPUT_SIZE, 32768

.section .text
.globl _start

_start:
    lea r12, [rip + input_buffer]
    xor r13, r13

.read_loop:
    mov edi, 0
    lea rsi, [r12 + r13]
    mov edx, INPUT_SIZE
    sub rdx, r13
    mov eax, 0
    syscall
    test rax, rax
    jz .read_done
    add r13, rax
    cmp r13, INPUT_SIZE
    jb .read_loop

.read_done:
    xor rbx, rbx
    xor r13, r13
    xor r14, r14

.parse_loop:
    cmp rbx, r12
    jae .parse_done
    cmp rbx, qword ptr [rip + input_size]
    jae .parse_done
    movzx eax, byte ptr [r12 + rbx]
    cmp al, 10
    je .found_newline
    inc rbx
    jmp .parse_loop

.found_newline:
    lea rdx, [rip + starts]
    mov [rdx + r14*8], r13
    lea rdx, [rip + lengths]
    mov rax, rbx
    sub rax, r13
    mov [rdx + r14*8], rax
    inc r14
    lea r13, [rbx + 1]
    inc rbx
    jmp .parse_loop

.parse_done:
    mov rax, qword ptr [rip + input_size]
    cmp r13, rax
    jae .sort_start
    lea rdx, [rip + starts]
    mov [rdx + r14*8], r13
    lea rdx, [rip + lengths]
    sub rax, r13
    mov [rdx + r14*8], rax
    inc r14

.sort_start:
    test r14, r14
    jz .program_done

    lea rbx, [rip + order]
    mov qword ptr [rbx], 0
    mov r15, 1

.sort_next:
    cmp r15, r14
    jae .output_start
    mov r13, r14

.shift_check:
    test r13, r13
    jz .insert_item
    dec r13
    mov rax, [rbx + r13*8]
    mov edi, r15d
    mov esi, eax
    call compare_lines
    test eax, eax
    jz .insert_item
    mov rax, [rbx + r13*8]
    mov [rbx + r13*8 + 8], rax
    jmp .shift_check

.insert_item:
    mov [rbx + r13*8], r15
    inc r15
    inc r14
    dec r14
    jmp .sort_next

.output_start:
    xor rbx, rbx
    mov r14, r15

.output_line:
    cmp rbx, r14
    jae .program_done
    mov rax, [rip + order + rbx*8]
    lea rdx, [rip + starts]
    mov r8, [rdx + rax*8]
    lea rdx, [rip + lengths]
    mov r9, [rdx + rax*8]

.write_line:
    test r9, r9
    jz .write_newline
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    mov eax, 1
    syscall
    add r8, rax
    sub r9, rax
    jmp .write_line

.write_newline:
    lea r8, [rip + newline]
    mov r9, 1

.write_nl_loop:
    test r9, r9
    jz .next_output
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    mov eax, 1
    syscall
    add r8, rax
    sub r9, rax
    jmp .write_nl_loop

.next_output:
    inc rbx
    jmp .output_line

.program_done:
    xor edi, edi
    mov eax, 60
    syscall

compare_lines:
    lea rdx, [rip + starts]
    mov r8, [rdx + rdi*8]
    mov r10, [rdx + rsi*8]
    lea rdx, [rip + lengths]
    mov r9, [rdx + rdi*8]
    mov r11, [rdx + rsi*8]
    xor ecx, ecx

.compare_loop:
    cmp rcx, r9
    jae .compare_lengths
    cmp rcx, r11
    jae .compare_not_less
    movzx eax, byte ptr [r8 + rcx]
    movzx edx, byte ptr [r10 + rcx]
    cmp eax, edx
    jb .compare_less
    ja .compare_not_less
    inc rcx
    jmp .compare_loop

.compare_lengths:
    cmp r9, r11
    jb .compare_less

.compare_not_less:
    xor eax, eax
    ret

.compare_less:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 10

.section .bss
    .align 8
input_buffer:
    .zero INPUT_SIZE
input_size:
    .zero 8
starts:
    .zero 8008
lengths:
    .zero 8008
order:
    .zero 8000
