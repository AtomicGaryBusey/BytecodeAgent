.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.globl _start

.section .text
_start:
    lea r8, [rip + input]
    lea r10, [rip + starts]
    lea r12, [rip + lengths]
    xor r9d, r9d

read_more:
    cmp r9, INPUT_SIZE
    jae parse_input
    mov eax, 0
    xor edi, edi
    lea rsi, [r8 + r9]
    mov edx, INPUT_SIZE
    sub rdx, r9
    syscall
    test rax, rax
    jz parse_input
    add r9, rax
    jmp read_more

parse_input:
    xor ebx, ebx
    xor esi, esi
    xor r13d, r13d

parse_loop:
    cmp rbx, r9
    jae parse_final
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    cmp al, 10
    jne parse_loop

    mov rax, rbx
    dec rax
    sub rax, rsi
    mov qword ptr [r10 + r13*8], rsi
    mov qword ptr [r12 + r13*8], rax
    inc r13
    mov rsi, rbx
    jmp parse_loop

parse_final:
    cmp rsi, r9
    je sort_input
    mov qword ptr [r10 + r13*8], rsi
    mov rax, r9
    sub rax, rsi
    mov qword ptr [r12 + r13*8], rax
    inc r13

sort_input:
    cmp r13, 1
    jbe output_start
    mov ebx, 1

sort_outer:
    cmp rbx, r13
    jae output_start

    mov r14, qword ptr [r10 + rbx*8]
    mov r15, qword ptr [r12 + rbx*8]
    mov rax, rbx
    dec rax
    mov ebp, eax

compare_previous:
    mov rax, qword ptr [r10 + rbp*8]
    mov rcx, qword ptr [r12 + rbp*8]
    lea rdi, [r8 + r14]
    lea rdx, [r8 + rax]
    xor esi, esi

compare_bytes:
    cmp rsi, r15
    jae candidate_exhausted
    cmp rsi, rcx
    jae previous_exhausted
    movzx eax, byte ptr [rdi + rsi]
    movzx edx, byte ptr [rdx + rsi]
    cmp eax, edx
    jb shift_previous
    jae candidate_not_smaller
    inc rsi
    jmp compare_bytes

candidate_not_smaller:
    ja place_candidate
    inc rsi
    jmp compare_bytes

candidate_exhausted:
    cmp r15, rcx
    jb shift_previous
    jmp place_candidate

previous_exhausted:
    jmp place_candidate

shift_previous:
    mov rax, qword ptr [r10 + rbp*8]
    mov qword ptr [r10 + rbp*8 + 8], rax
    mov rax, qword ptr [r12 + rbp*8]
    mov qword ptr [r12 + rbp*8 + 8], rax
    test ebp, ebp
    jz insert_candidate
    dec ebp
    jmp compare_previous

insert_candidate:
    mov qword ptr [r10], r14
    mov qword ptr [r12], r15
    inc rbx
    jmp sort_outer

place_candidate:
    mov qword ptr [r10 + rbx*8], r14
    mov qword ptr [r12 + rbx*8], r15
    inc rbx
    jmp sort_outer

output_start:
    xor ebx, ebx

output_line:
    cmp rbx, r13
    jae finish
    mov r14, qword ptr [r10 + rbx*8]
    mov r15, qword ptr [r12 + rbx*8]
    xor r9d, r9d

write_content:
    cmp r9, r15
    jae write_newline
    mov eax, 1
    mov edi, 1
    lea rsi, [r8 + r14]
    add rsi, r9
    mov rdx, r15
    sub rdx, r9
    syscall
    add r9, rax
    jmp write_content

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    cmp rax, 1
    jne write_newline
    inc rbx
    jmp output_line

finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
.align 8
input:
    .zero INPUT_SIZE
starts:
    .zero MAX_LINES * 8
lengths:
    .zero MAX_LINES * 8
