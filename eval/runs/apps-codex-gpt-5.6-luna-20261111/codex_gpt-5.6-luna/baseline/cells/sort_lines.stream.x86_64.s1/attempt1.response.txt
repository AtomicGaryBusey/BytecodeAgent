.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.globl _start

.section .text
_start:
    lea r8, [rip + input]
    lea r11, [rip + starts]
    lea r10, [rip + lengths]
    xor r12d, r12d

read_more:
    mov eax, 0
    xor edi, edi
    lea rsi, [r8 + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_more

parse_input:
    xor ebx, ebx
    xor r13d, r13d

parse_outer:
    cmp rbx, r12
    jae sort_input
    mov r14, rbx

parse_scan:
    cmp rbx, r12
    jae parse_record
    movzx eax, byte ptr [r8 + rbx]
    cmp al, 10
    je parse_record
    inc rbx
    jmp parse_scan

parse_record:
    mov r15, rbx
    sub r15, r14
    mov qword ptr [r11 + r13*8], r14
    mov qword ptr [r10 + r13*8], r15
    inc r13
    cmp rbx, r12
    jae sort_input
    inc rbx
    jmp parse_outer

sort_input:
    cmp r13, 1
    jbe output_start
    mov ebx, 1

sort_outer:
    cmp rbx, r13
    jae output_start
    mov r14, qword ptr [r11 + rbx*8]
    mov r15, qword ptr [r10 + rbx*8]
    mov rax, rbx
    dec rax
    mov rdx, qword ptr [r11 + rax*8]
    mov rcx, qword ptr [r10 + rax*8]
    lea rdi, [r8 + r14]
    lea rdx, [r8 + rdx]
    xor esi, esi

compare_records:
    cmp rsi, r15
    jae key_exhausted
    cmp rsi, rcx
    jae previous_exhausted
    movzx eax, byte ptr [rdi + rsi]
    movzx ebp, byte ptr [rdx + rsi]
    cmp eax, ebp
    jb shift_record
    ja no_shift
    inc rsi
    jmp compare_records

key_exhausted:
    cmp r15, rcx
    jb shift_record
    jmp no_shift

previous_exhausted:
    jmp no_shift

shift_record:
    mov rax, qword ptr [r11 + rbx*8 - 8]
    mov qword ptr [r11 + rbx*8], rax
    mov rax, qword ptr [r10 + rbx*8 - 8]
    mov qword ptr [r10 + rbx*8], rax
    cmp rbx, 1
    je insert_zero
    dec rbx
    mov rdx, qword ptr [r11 + rbx*8]
    mov rcx, qword ptr [r10 + rbx*8]
    lea rdi, [r8 + r14]
    lea rdx, [r8 + rdx]
    xor esi, esi
    jmp compare_records

insert_zero:
    mov qword ptr [r11], r14
    mov qword ptr [r10], r15
    inc rbx
    jmp sort_outer

no_shift:
    mov qword ptr [r11 + rbx*8], r14
    mov qword ptr [r10 + rbx*8], r15
    inc rbx
    jmp sort_outer

output_start:
    xor ebx, ebx

output_line:
    cmp rbx, r13
    jae finish
    mov r12, qword ptr [r11 + rbx*8]
    mov r15, qword ptr [r10 + rbx*8]
    xor r14d, r14d

write_line:
    cmp r14, r15
    jae write_newline
    mov eax, 1
    mov edi, 1
    lea rsi, [r8 + r12 + r14]
    mov rdx, r15
    sub rdx, r14
    syscall
    add r14, rax
    jmp write_line

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    cmp rax, 1
    jne write_newline
    inc rbx
    jmp output_line

finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
.align 8
input:
    .zero INPUT_SIZE
.align 8
starts:
    .zero MAX_LINES * 8
.align 8
lengths:
    .zero MAX_LINES * 8
