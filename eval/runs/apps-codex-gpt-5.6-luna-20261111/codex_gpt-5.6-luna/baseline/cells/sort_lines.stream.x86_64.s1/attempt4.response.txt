.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.globl _start

.section .text
_start:
    lea r8, [rip + input]
    lea r10, [rip + starts]
    lea r12, [rip + lengths]
    xor r9d, r9d

read_more:
    cmp r9, INPUT_SIZE
    jae parse_input
    xor eax, eax
    xor edi, edi
    lea rsi, [r8 + r9]
    mov edx, INPUT_SIZE
    sub rdx, r9
    syscall
    test rax, rax
    jz parse_input
    add r9, rax
    jmp read_more

parse_input:
    xor ebx, ebx
    xor esi, esi
    xor r13d, r13d

parse_loop:
    cmp rbx, r9
    jae parse_final
    movzx eax, byte ptr [r8 + rbx]
    inc rbx
    cmp al, 10
    jne parse_loop

    mov rax, rbx
    dec rax
    sub rax, rsi
    mov qword ptr [r10 + r13*8], rsi
    mov qword ptr [r12 + r13*8], rax
    inc r13
    mov rsi, rbx
    jmp parse_loop

parse_final:
    cmp rsi, r9
    je sort_start
    mov qword ptr [r10 + r13*8], rsi
    mov rax, r9
    sub rax, rsi
    mov qword ptr [r12 + r13*8], rax
    inc r13

sort_start:
    cmp r13, 1
    jbe output_start
    xor ebx, ebx

sort_outer:
    mov qword ptr [sort_i], rbx
    mov qword ptr [sort_min], rbx
    mov r14, rbx
    inc r14

sort_find:
    cmp r14, r13
    jae sort_swap

    mov rax, qword ptr [sort_min]
    mov qword ptr [sort_j], r14

    mov rdx, qword ptr [r10 + r14*8]
    mov rcx, qword ptr [r12 + r14*8]
    mov r11, qword ptr [r10 + rax*8]
    mov rsi, qword ptr [r12 + rax*8]
    xor r9d, r9d

compare_lines:
    cmp r9, rcx
    jae candidate_end
    cmp r9, rsi
    jae minimum_end
    movzx eax, byte ptr [r8 + rdx + r9]
    movzx edi, byte ptr [r8 + r11 + r9]
    cmp eax, edi
    jb candidate_less
    ja minimum_end
    inc r9
    jmp compare_lines

candidate_end:
    cmp rcx, rsi
    jb candidate_less
    jmp minimum_end

candidate_less:
    mov rax, qword ptr [sort_j]
    mov qword ptr [sort_min], rax

minimum_end:
    inc r14
    jmp sort_find

sort_swap:
    mov rbx, qword ptr [sort_i]
    mov rax, qword ptr [sort_min]
    cmp rax, rbx
    je sort_next

    mov rdx, qword ptr [r10 + rbx*8]
    mov rcx, qword ptr [r10 + rax*8]
    mov qword ptr [r10 + rbx*8], rcx
    mov qword ptr [r10 + rax*8], rdx

    mov rdx, qword ptr [r12 + rbx*8]
    mov rcx, qword ptr [r12 + rax*8]
    mov qword ptr [r12 + rbx*8], rcx
    mov qword ptr [r12 + rax*8], rdx

sort_next:
    inc rbx
    cmp rbx, r13
    jb sort_outer

output_start:
    xor ebx, ebx

output_line:
    cmp rbx, r13
    jae finish
    mov r14, qword ptr [r10 + rbx*8]
    mov r15, qword ptr [r12 + rbx*8]
    xor r9d, r9d

write_content:
    cmp r9, r15
    jae write_newline
    mov eax, 1
    mov edi, 1
    lea rsi, [r8 + r14 + r9]
    mov rdx, r15
    sub rdx, r9
    syscall
    add r9, rax
    jmp write_content

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc rbx
    jmp output_line

finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
.align 8
input:
    .zero INPUT_SIZE
starts:
    .zero MAX_LINES * 8
lengths:
    .zero MAX_LINES * 8
sort_i:
    .zero 8
sort_min:
    .zero 8
sort_j:
    .zero 8
