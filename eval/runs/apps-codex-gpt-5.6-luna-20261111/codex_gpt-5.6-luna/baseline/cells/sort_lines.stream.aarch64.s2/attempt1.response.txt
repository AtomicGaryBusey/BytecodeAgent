.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.section .rodata
newline:
	.byte 10

.bss
	.balign 16
input_buf:
	.zero INPUT_SIZE
line_starts:
	.zero MAX_LINES * 8
line_lengths:
	.zero MAX_LINES * 8

.text
.globl _start
_start:
	adrp x20, input_buf
	add x20, x20, :lo12:input_buf
	mov x19, #0

read_loop:
	mov x0, #0
	add x1, x20, x19
	movz x10, #INPUT_SIZE
	sub x2, x10, x19
	mov x8, #63
	svc #0
	cbz x0, parse_start
	add x19, x19, x0
	cmp x19, x10
	b.lo read_loop

parse_start:
	adrp x24, line_starts
	add x24, x24, :lo12:line_starts
	adrp x25, line_lengths
	add x25, x25, :lo12:line_lengths
	mov x21, #0
	mov x3, #0
	mov x4, #0

parse_loop:
	cmp x4, x19
	b.hs parse_done
	ldrb w5, [x20, x4]
	cmp w5, #10
	b.ne parse_next
	sub x6, x4, x3
	str x3, [x24, x21, lsl #3]
	str x6, [x25, x21, lsl #3]
	add x21, x21, #1
	add x3, x4, #1

parse_next:
	add x4, x4, #1
	b parse_loop

parse_done:
	cmp x3, x19
	b.eq sort_start
	sub x6, x19, x3
	str x3, [x24, x21, lsl #3]
	str x6, [x25, x21, lsl #3]
	add x21, x21, #1

sort_start:
	mov x22, #0

sort_outer:
	cmp x22, x21
	b.hs output_start
	mov x23, x22
	add x26, x22, #1

sort_inner:
	cmp x26, x21
	b.hs sort_swap
	ldr x10, [x24, x26, lsl #3]
	ldr x11, [x24, x23, lsl #3]
	ldr x12, [x25, x26, lsl #3]
	ldr x13, [x25, x23, lsl #3]

	cmp x12, x13
	b.lo j_shorter
	mov x14, x13
	b compare_length_ready

j_shorter:
	mov x14, x12

compare_length_ready:
	mov x15, #0

compare_bytes:
	cmp x15, x14
	b.hs compare_lengths
	ldrb w16, [x20, x10, lsl #0]
	ldrb w17, [x20, x11, lsl #0]
	cmp w16, w17
	b.lo j_is_less
	b.hi j_is_not_less
	add x10, x10, #1
	add x11, x11, #1
	add x15, x15, #1
	b compare_bytes

compare_lengths:
	cmp x12, x13
	b.lo j_is_less
	b j_is_not_less

j_is_less:
	mov x23, x26

j_is_not_less:
	add x26, x26, #1
	b sort_inner

sort_swap:
	cmp x23, x22
	b.eq sort_next
	ldr x10, [x24, x22, lsl #3]
	ldr x11, [x24, x23, lsl #3]
	str x11, [x24, x22, lsl #3]
	str x10, [x24, x23, lsl #3]
	ldr x12, [x25, x22, lsl #3]
	ldr x13, [x25, x23, lsl #3]
	str x13, [x25, x22, lsl #3]
	str x12, [x25, x23, lsl #3]

sort_next:
	add x22, x22, #1
	b sort_outer

output_start:
	adrp x28, newline
	add x28, x28, :lo12:newline
	mov x22, #0

output_line:
	cmp x22, x21
	b.hs program_exit
	ldr x10, [x24, x22, lsl #3]
	ldr x27, [x25, x22, lsl #3]
	add x26, x20, x10

write_line:
	cbz x27, write_newline
	mov x0, #1
	mov x1, x26
	mov x2, x27
	mov x8, #64
	svc #0
	cbz x0, write_line
	add x26, x26, x0
	sub x27, x27, x0
	b write_line

write_newline:
	mov x0, #1
	mov x1, x28
	mov x2, #1
	mov x8, #64
	svc #0
	cbz x0, write_newline
	add x22, x22, #1
	b output_line

program_exit:
	mov x0, #0
	mov x8, #93
	svc #0
