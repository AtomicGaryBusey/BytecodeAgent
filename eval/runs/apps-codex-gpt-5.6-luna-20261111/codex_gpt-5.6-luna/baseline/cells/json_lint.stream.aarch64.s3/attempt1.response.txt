.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ BUF_SIZE, 65536

.section .rodata
ok_msg:
	.ascii "ok\n"
ok_msg_len = . - ok_msg
err_msg:
	.ascii "error at "
err_msg_len = . - err_msg
nl:
	.ascii "\n"

.section .bss
	.align 4
input_buf:
	.zero BUF_SIZE
out_buf:
	.zero 64

.section .text
.globl _start
_start:
	adrp x20, input_buf
	add x20, x20, :lo12:input_buf
	mov x21, #0

read_loop:
	add x1, x20, x21
	mov x0, #0
	mov x2, #BUF_SIZE
	sub x2, x2, x21
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	beq read_done
	add x21, x21, x0
	b read_loop

read_done:
	mov x19, #0
	bl parse_value
	cmp x0, #0
	bne invalid_result
	bl skip_ws
	cmp x19, x21
	blo trailing_error

	adrp x1, ok_msg
	add x1, x1, :lo12:ok_msg
	mov x2, #ok_msg_len
	bl write_all
	mov x0, #0
	mov x8, #SYS_exit
	svc #0

trailing_error:
	mov x22, x19
	b report_error

invalid_result:
	cmp x0, #2
	bne use_error_position
	mov x22, x21
	b report_error

use_error_position:
	mov x22, x1

report_error:
	adrp x1, err_msg
	add x1, x1, :lo12:err_msg
	mov x2, #err_msg_len
	bl write_all

	adrp x24, out_buf
	add x24, x24, :lo12:out_buf
	add x24, x24, #63
	mov x23, #10
	mov x25, #0
	cbnz x22, convert_number
	sub x24, x24, #1
	mov w26, #'0'
	strb w26, [x24]
	mov x25, #1
	b number_ready

convert_number:
	udiv x26, x22, x23
	msub x27, x26, x23, x22
	add w27, w27, #'0'
	sub x24, x24, #1
	strb w27, [x24]
	add x25, x25, #1
	mov x22, x26
	cbnz x22, convert_number

number_ready:
	mov x1, x24
	mov x2, x25
	bl write_all
	adrp x1, nl
	add x1, x1, :lo12:nl
	mov x2, #1
	bl write_all
	mov x0, #1
	mov x8, #SYS_exit
	svc #0

write_all:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	mov x4, x1
	mov x5, x2
write_loop:
	cbz x5, write_done
	mov x0, #1
	mov x1, x4
	mov x2, x5
	mov x8, #SYS_write
	svc #0
	add x4, x4, x0
	sub x5, x5, x0
	b write_loop
write_done:
	ldp x29, x30, [sp], #32
	ret

skip_ws:
skip_ws_loop:
	cmp x19, x21
	bhs skip_ws_done
	ldrb w0, [x20, x19]
	cmp w0, #0x20
	beq skip_ws_one
	cmp w0, #9
	beq skip_ws_one
	cmp w0, #10
	beq skip_ws_one
	cmp w0, #13
	bne skip_ws_done
skip_ws_one:
	add x19, x19, #1
	b skip_ws_loop
skip_ws_done:
	ret

parse_value:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	cmp x19, x21
	bhs value_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'{'
	beq value_object
	cmp w2, #'['
	beq value_array
	cmp w2, #'"'
	beq value_string
	cmp w2, #'-'
	beq value_number
	cmp w2, #'0'
	blo value_keyword
	cmp w2, #'9'
	bls value_number
value_keyword:
	cmp w2, #'t'
	beq value_true
	cmp w2, #'f'
	beq value_false
	cmp w2, #'n'
	beq value_null
	mov x0, #1
	mov x1, x19
	b value_done
value_object:
	bl parse_object
	b value_done
value_array:
	bl parse_array
	b value_done
value_string:
	bl parse_string
	b value_done
value_number:
	bl parse_number
	b value_done
value_true:
	bl parse_true
	b value_done
value_false:
	bl parse_false
	b value_done
value_null:
	bl parse_null
	b value_done
value_incomplete:
	mov x0, #2
value_done:
	ldp x29, x30, [sp], #32
	ret

parse_true:
	mov w2, #'t'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'r'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'u'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'e'
	bl literal_char
literal_done:
	ret

parse_false:
	mov w2, #'f'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'a'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'l'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'s'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'e'
	bl literal_char
	b literal_done

parse_null:
	mov w2, #'n'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'u'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'l'
	bl literal_char
	cmp x0, #0
	bne literal_done
	mov w2, #'l'
	bl literal_char
	b literal_done

literal_char:
	cmp x19, x21
	bhs literal_incomplete
	ldrb w3, [x20, x19]
	cmp w3, w2
	bne literal_error
	add x19, x19, #1
	mov x0, #0
	ret
literal_incomplete:
	mov x0, #2
	ret
literal_error:
	mov x0, #1
	mov x1, x19
	ret

parse_string:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	add x19, x19, #1
string_loop:
	cmp x19, x21
	bhs string_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'"'
	beq string_end
	cmp w2, #'\\'
	beq string_escape
	cmp w2, #0x20
	blo string_error
	add x19, x19, #1
	b string_loop
string_escape:
	add x19, x19, #1
	cmp x19, x21
	bhs string_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'u'
	beq unicode_escape
	cmp w2, #'"'
	beq simple_escape
	cmp w2, #'\\'
	beq simple_escape
	cmp w2, #'/'
	beq simple_escape
	cmp w2, #'b'
	beq simple_escape
	cmp w2, #'f'
	beq simple_escape
	cmp w2, #'n'
	beq simple_escape
	cmp w2, #'r'
	beq simple_escape
	cmp w2, #'t'
	beq simple_escape
	b string_error
simple_escape:
	add x19, x19, #1
	b string_loop
unicode_escape:
	add x19, x19, #1
	mov x3, #4
unicode_loop:
	cmp x19, x21
	bhs string_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'0'
	blo unicode_check_upper
	cmp w2, #'9'
	bls unicode_digit
unicode_check_upper:
	cmp w2, #'A'
	blo unicode_check_lower
	cmp w2, #'F'
	bls unicode_digit
unicode_check_lower:
	cmp w2, #'a'
	blo string_error
	cmp w2, #'f'
	bhi string_error
unicode_digit:
	add x19, x19, #1
	sub x3, x3, #1
	cbnz x3, unicode_loop
	b string_loop
string_end:
	add x19, x19, #1
	mov x0, #0
	b string_done
string_incomplete:
	mov x0, #2
	b string_done
string_error:
	mov x0, #1
	mov x1, x19
string_done:
	ldp x29, x30, [sp], #32
	ret

parse_number:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	ldrb w2, [x20, x19]
	cmp w2, #'-'
	bne number_first
	add x19, x19, #1
	cmp x19, x21
	bhs number_incomplete
	ldrb w2, [x20, x19]
number_first:
	cmp w2, #'0'
	beq number_zero
	cmp w2, #'1'
	blo number_error
	cmp w2, #'9'
	bhi number_error
number_digits:
	add x19, x19, #1
	cmp x19, x21
	bhs number_done
	ldrb w2, [x20, x19]
	cmp w2, #'0'
	blo number_fraction
	cmp w2, #'9'
	bls number_digits
	b number_fraction
number_zero:
	add x19, x19, #1
	cmp x19, x21
	bhs number_done
	ldrb w2, [x20, x19]
	cmp w2, #'0'
	blo number_fraction
	cmp w2, #'9'
	bls number_error
	b number_fraction
number_fraction:
	cmp w2, #'.'
	bne number_exponent
	add x19, x19, #1
	cmp x19, x21
	bhs number_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'0'
	blo number_error
	cmp w2, #'9'
	bhi number_error
frac_digits:
	add x19, x19, #1
	cmp x19, x21
	bhs number_done
	ldrb w2, [x20, x19]
	cmp w2, #'0'
	blo number_exponent
	cmp w2, #'9'
	bls frac_digits
number_exponent:
	cmp w2, #'e'
	beq exponent_start
	cmp w2, #'E'
	bne number_done
exponent_start:
	add x19, x19, #1
	cmp x19, x21
	bhs number_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'+'
	beq exponent_sign
	cmp w2, #'-'
	bne exponent_digit_first
exponent_sign:
	add x19, x19, #1
	cmp x19, x21
	bhs number_incomplete
	ldrb w2, [x20, x19]
exponent_digit_first:
	cmp w2, #'0'
	blo number_error
	cmp w2, #'9'
	bhi number_error
exponent_digits:
	add x19, x19, #1
	cmp x19, x21
	bhs number_done
	ldrb w2, [x20, x19]
	cmp w2, #'0'
	blo number_done
	cmp w2, #'9'
	bls exponent_digits
number_done:
	mov x0, #0
	b number_return
number_incomplete:
	mov x0, #2
	b number_return
number_error:
	mov x0, #1
	mov x1, x19
number_return:
	ldp x29, x30, [sp], #32
	ret

parse_array:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	add x19, x19, #1
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #']'
	beq empty_array
array_value:
	bl parse_value
	cmp x0, #0
	bne compound_return
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #']'
	beq empty_array
	cmp w2, #','
	bne compound_error
	add x19, x19, #1
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #']'
	beq compound_error
	b array_value
empty_array:
	add x19, x19, #1
	mov x0, #0
	b compound_finish
parse_object:
	stp x29, x30, [sp, #-32]!
	mov x29, sp
	add x19, x19, #1
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'}'
	beq empty_object
object_member:
	bl parse_string
	cmp x0, #0
	bne compound_return
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #':'
	bne compound_error
	add x19, x19, #1
	bl skip_ws
	bl parse_value
	cmp x0, #0
	bne compound_return
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'}'
	beq empty_object
	cmp w2, #','
	bne compound_error
	add x19, x19, #1
	bl skip_ws
	cmp x19, x21
	bhs compound_incomplete
	ldrb w2, [x20, x19]
	cmp w2, #'}'
	beq compound_error
	b object_member
empty_object:
	add x19, x19, #1
	mov x0, #0
	b compound_finish
compound_incomplete:
	mov x0, #2
	b compound_finish
compound_error:
	mov x0, #1
	mov x1, x19
	b compound_finish
compound_return:
compound_finish:
	ldp x29, x30, [sp], #32
	ret
