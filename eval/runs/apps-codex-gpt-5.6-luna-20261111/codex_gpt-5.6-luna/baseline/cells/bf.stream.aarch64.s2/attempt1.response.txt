.section .text
.globl _start

_start:
    adrp x20, input
    add x20, x20, :lo12:input
    adrp x21, tape
    add x21, x21, :lo12:tape
    adrp x22, brackets
    add x22, x22, :lo12:brackets
    adrp x23, output
    add x23, x23, :lo12:output
    adrp x28, matches
    add x28, x28, :lo12:matches

    mov x19, #0

read_loop:
    cmp x19, #65536
    b.eq read_done
    mov x0, #0
    add x1, x20, x19
    mov x2, #65536
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x24, #0

find_delimiter:
    cmp x24, x19
    b.eq no_delimiter
    ldrb w0, [x20, x24]
    cmp w0, #33
    b.eq delimiter_found
    add x24, x24, #1
    b find_delimiter

delimiter_found:
    mov x25, x24
    add x24, x24, #1
    b validate_start

no_delimiter:
    mov x25, x19
    mov x24, x19

validate_start:
    mov x26, #0
    mov x27, #0

validate_loop:
    cmp x27, x25
    b.eq validate_done
    ldrb w0, [x20, x27]
    cmp w0, #91
    b.eq validate_open
    cmp w0, #93
    b.eq validate_close
    add x27, x27, #1
    b validate_loop

validate_open:
    str w27, [x22, x26, lsl #2]
    add x26, x26, #1
    add x27, x27, #1
    b validate_loop

validate_close:
    cbz x26, invalid
    sub x26, x26, #1
    ldr w0, [x22, x26, lsl #2]
    str w0, [x28, x27, lsl #2]
    str w27, [x28, x0, lsl #2]
    add x27, x27, #1
    b validate_loop

validate_done:
    cbnz x26, invalid

    mov x19, x21
    mov x27, #0
    mov x26, #0

execute_loop:
    cmp x27, x25
    b.eq flush_output
    ldrb w0, [x20, x27]

    cmp w0, #62
    b.eq op_right
    cmp w0, #60
    b.eq op_left
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #46
    b.eq op_output
    cmp w0, #44
    b.eq op_input
    cmp w0, #91
    b.eq op_open
    cmp w0, #93
    b.eq op_close

    add x27, x27, #1
    b execute_loop

op_right:
    add x19, x19, #1
    add x27, x27, #1
    b execute_loop

op_left:
    sub x19, x19, #1
    add x27, x27, #1
    b execute_loop

op_add:
    ldrb w1, [x19]
    add w1, w1, #1
    strb w1, [x19]
    add x27, x27, #1
    b execute_loop

op_sub:
    ldrb w1, [x19]
    sub w1, w1, #1
    strb w1, [x19]
    add x27, x27, #1
    b execute_loop

op_output:
    ldrb w1, [x19]
    strb w1, [x23, x26]
    add x26, x26, #1
    add x27, x27, #1
    b execute_loop

op_input:
    cmp x24, x19
    b.eq input_empty
    ldrb w1, [x20, x24]
    add x24, x24, #1
    strb w1, [x19]
    add x27, x27, #1
    b execute_loop

input_empty:
    strb wzr, [x19]
    add x27, x27, #1
    b execute_loop

op_open:
    ldrb w1, [x19]
    cbnz w1, open_continue
    ldr w27, [x28, x27, lsl #2]
    add x27, x27, #1
    b execute_loop

open_continue:
    add x27, x27, #1
    b execute_loop

op_close:
    ldrb w1, [x19]
    cbz w1, close_continue
    ldr w27, [x28, x27, lsl #2]
    add x27, x27, #1
    b execute_loop

close_continue:
    add x27, x27, #1
    b execute_loop

flush_output:
    mov x24, #0

write_loop:
    cmp x24, x26
    b.eq success
    mov x0, #1
    add x1, x23, x24
    sub x2, x26, x24
    mov x8, #64
    svc #0
    add x24, x24, x0
    b write_loop

invalid:
    mov x0, #1
    mov x8, #93
    svc #0

success:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.align 4
input:
    .zero 65536
tape:
    .zero 30000
.align 4
output:
    .zero 65536
matches:
    .zero 262144
brackets:
    .zero 4096
