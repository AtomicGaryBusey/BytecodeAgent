.equ INPUT_SIZE, 65536
.equ LINE_SIZE, 80

.globl _start

.section .text
_start:
    lea r13, [rip + input]
    xor r12, r12

read_loop:
    cmp r12, INPUT_SIZE
    je read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r13 + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r14, r14

row_loop:
    cmp r14, r12
    jae final_offset

    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_size_done
    mov r15, 16

row_size_done:
    lea rdi, [rip + line]
    mov ecx, LINE_SIZE
    mov eax, 32
    rep stosb

    lea rdi, [rip + line]
    mov eax, r14d
    mov ecx, 28
    lea r11, [rip + hex]

offset_digits:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns offset_digits

    lea rdi, [rip + line + 10]
    lea r9, [rip + line + 61]
    lea r10, [r13 + r14]
    xor ebx, ebx

byte_loop:
    cmp rbx, r15
    jae bytes_done

    movzx eax, byte ptr [r10 + rbx]
    mov r8d, eax

    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl

    and eax, 15
    mov al, byte ptr [r11 + rax]
    mov byte ptr [rdi + 1], al

    cmp r8d, 32
    jb nonprintable
    cmp r8d, 126
    ja nonprintable
    mov byte ptr [r9 + rbx], r8b
    jmp char_done

nonprintable:
    mov byte ptr [r9 + rbx], 46

char_done:
    add rdi, 3
    inc rbx
    jmp byte_loop

bytes_done:
    lea rdi, [rip + line]
    mov byte ptr [rdi + 59], 32
    mov byte ptr [rdi + 60], 124
    mov byte ptr [rdi + r15 + 61], 124
    mov byte ptr [rdi + r15 + 62], 10

    lea rsi, [rip + line]
    mov edx, 63
    add rdx, r15

write_row:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_row

    add r14, r15
    jmp row_loop

final_offset:
    lea rdi, [rip + line]
    mov eax, r12d
    mov ecx, 28
    lea r11, [rip + hex]

final_digits:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [r11 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns final_digits

    mov byte ptr [rdi], 10
    lea rsi, [rip + line]
    mov edx, 9

write_final:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_final

    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
hex:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input:
    .zero INPUT_SIZE
line:
    .zero LINE_SIZE
