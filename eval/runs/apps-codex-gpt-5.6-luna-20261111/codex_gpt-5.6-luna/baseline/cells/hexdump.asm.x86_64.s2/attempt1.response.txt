.equ INPUT_SIZE, 65536

.section .text
.globl _start

_start:
    xor r12d, r12d
    xor r14d, r14d
    lea rbx, [rip + input_buffer]

read_loop:
    cmp r12, INPUT_SIZE
    jae input_done
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    lea r13, [rip + output_buffer]
    lea r8, [rip + hex_digits]

row_loop:
    cmp r14, r12
    jae final_offset

    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_size_done
    mov r15, 16

row_size_done:
    mov rdi, r13
    mov rax, r14
    mov ecx, 28

offset_hex_loop:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jnc offset_hex_loop

    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r9d, r9d

hex_row_loop:
    cmp r9, 16
    jae hex_row_done
    cmp r9, r15
    jae missing_byte

    mov r10, r14
    add r10, r9
    movzx eax, byte ptr [rbx + r10]
    mov edx, eax
    shr edx, 4
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 32
    inc rdi
    inc r9
    cmp r9, 8
    jne hex_row_loop
    mov byte ptr [rdi], 32
    inc rdi
    jmp hex_row_loop

missing_byte:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    inc r9
    cmp r9, 8
    jne hex_row_loop
    mov byte ptr [rdi], 32
    inc rdi
    jmp hex_row_loop

hex_row_done:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2

    xor r9d, r9d

ascii_loop:
    cmp r9, r15
    jae ascii_done
    mov r10, r14
    add r10, r9
    movzx eax, byte ptr [rbx + r10]
    cmp eax, 32
    jb non_printable
    cmp eax, 126
    ja non_printable
    mov byte ptr [rdi], al
    jmp ascii_store_done

non_printable:
    mov byte ptr [rdi], 46

ascii_store_done:
    inc rdi
    inc r9
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2
    mov rdx, rdi
    sub rdx, r13
    mov rsi, r13
    call write_all

    add r14, r15
    jmp row_loop

final_offset:
    mov rdi, r13
    mov rax, r12
    mov ecx, 28

final_hex_loop:
    mov rdx, rax
    shr rdx, cl
    and edx, 15
    movzx edx, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jnc final_hex_loop

    mov byte ptr [rdi], 10
    inc rdi
    mov rdx, rdi
    sub rdx, r13
    mov rsi, r13
    call write_all

    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_done

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jz write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop

write_done:
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero 80
