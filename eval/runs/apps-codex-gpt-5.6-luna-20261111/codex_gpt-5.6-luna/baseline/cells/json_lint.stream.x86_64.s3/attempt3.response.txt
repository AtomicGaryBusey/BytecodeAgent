.equ IN_SIZE, 65536

.section .rodata
ok_msg:
    .ascii "ok\n"
.equ OK_LEN, 3
err_prefix:
    .ascii "error at "
.equ ERR_PREFIX_LEN, 9
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.section .bss
.align 16
input:
    .zero IN_SIZE
outbuf:
    .zero 128

.text
.globl _start
_start:
    xor ebx, ebx

read_loop:
    cmp ebx, IN_SIZE
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [input + rbx]
    mov edx, IN_SIZE
    sub edx, ebx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov r13, rbx
    xor r12d, r12d
    call parse_ws
    call parse_value
    test eax, eax
    jz report_error
    call parse_ws
    cmp r12, r13
    jne syntax_error

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, OK_LEN
    call write_all
    xor edi, edi
    mov eax, 60
    syscall

syntax_error:
    mov r14, r12

report_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, ERR_PREFIX_LEN
    call write_all

    lea rdi, [outbuf + 127]
    mov byte ptr [rdi], 10
    mov rcx, r14
    test rcx, rcx
    jnz decimal_loop
    dec rdi
    mov byte ptr [rdi], '0'
    jmp decimal_ready

decimal_loop:
    xor edx, edx
    mov rax, rcx
    mov ebx, 10
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    mov rcx, rax
    test rcx, rcx
    jnz decimal_loop

decimal_ready:
    mov rsi, rdi
    lea rdx, [outbuf + 128]
    sub rdx, rdi
    mov edi, 1
    call write_all
    mov edi, 1
    mov eax, 60
    syscall

write_all:
    test rdx, rdx
    jz write_done
write_again:
    mov eax, 1
    syscall
    test rax, rax
    jz write_done
    add rsi, rax
    sub rdx, rax
    jnz write_again
write_done:
    ret

parse_ws:
ws_loop:
    cmp r12, r13
    jae ws_done
    movzx eax, byte ptr [input + r12]
    cmp al, 32
    je ws_advance
    cmp al, 9
    je ws_advance
    cmp al, 10
    je ws_advance
    cmp al, 13
    je ws_advance
    jmp ws_done
ws_advance:
    inc r12
    jmp ws_loop
ws_done:
    ret

parse_value:
    cmp r12, r13
    jae value_eof
    movzx eax, byte ptr [input + r12]
    cmp al, '{'
    je value_object
    cmp al, '['
    je value_array
    cmp al, '"'
    je value_string
    cmp al, '-'
    je value_number
    cmp al, '0'
    jb value_literal_check
    cmp al, '9'
    jbe value_number

value_literal_check:
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    mov r14, r12
    xor eax, eax
    ret

value_eof:
    mov r14, r13
    xor eax, eax
    ret

value_object:
    call parse_object
    ret

value_array:
    call parse_array
    ret

value_string:
    call parse_string
    ret

value_number:
    call parse_number
    ret

value_true:
    lea rsi, [rip + true_text]
    mov ecx, 4
    call parse_literal
    ret

value_false:
    lea rsi, [rip + false_text]
    mov ecx, 5
    call parse_literal
    ret

value_null:
    lea rsi, [rip + null_text]
    mov ecx, 4
    call parse_literal
    ret

parse_literal:
    xor edx, edx
literal_loop:
    cmp rdx, rcx
    jae literal_ok
    cmp r12, r13
    jae literal_eof
    movzx eax, byte ptr [input + r12]
    cmp al, byte ptr [rsi + rdx]
    jne literal_bad
    inc r12
    inc rdx
    jmp literal_loop

literal_ok:
    mov eax, 1
    ret

literal_eof:
    mov r14, r13
    xor eax, eax
    ret

literal_bad:
    mov r14, r12
    xor eax, eax
    ret

parse_string:
    inc r12

string_loop:
    cmp r12, r13
    jae string_eof
    movzx eax, byte ptr [input + r12]
    cmp al, '"'
    je string_done
    cmp al, 32
    jb string_bad
    cmp al, 92
    je string_escape
    inc r12
    jmp string_loop

string_escape:
    inc r12
    cmp r12, r13
    jae string_eof
    movzx eax, byte ptr [input + r12]
    cmp al, '"'
    je string_simple
    cmp al, 92
    je string_simple
    cmp al, '/'
    je string_simple
    cmp al, 'b'
    je string_simple
    cmp al, 'f'
    je string_simple
    cmp al, 'n'
    je string_simple
    cmp al, 'r'
    je string_simple
    cmp al, 't'
    je string_simple
    cmp al, 'u'
    je string_unicode
    jmp string_bad

string_simple:
    inc r12
    jmp string_loop

string_unicode:
    inc r12
    mov ecx, 4

unicode_loop:
    cmp r12, r13
    jae string_eof
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    jb unicode_upper
    cmp al, '9'
    jbe unicode_ok

unicode_upper:
    cmp al, 'A'
    jb unicode_lower
    cmp al, 'F'
    jbe unicode_ok

unicode_lower:
    cmp al, 'a'
    jb string_bad
    cmp al, 'f'
    ja string_bad

unicode_ok:
    inc r12
    dec ecx
    jnz unicode_loop
    jmp string_loop

string_done:
    inc r12
    mov eax, 1
    ret

string_eof:
    mov r14, r13
    xor eax, eax
    ret

string_bad:
    mov r14, r12
    xor eax, eax
    ret

parse_number:
    cmp byte ptr [input + r12], '-'
    jne number_start
    inc r12
    cmp r12, r13
    jae number_eof

number_start:
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb number_bad
    cmp al, '9'
    ja number_bad

number_digits:
    inc r12
    cmp r12, r13
    jae number_end
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    jb number_end
    cmp al, '9'
    jbe number_digits
    jmp number_end

number_zero:
    inc r12

number_end:
    cmp r12, r13
    jae number_ok
    cmp byte ptr [input + r12], '.'
    jne number_exp
    inc r12
    cmp r12, r13
    jae number_eof
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    jb number_bad
    cmp al, '9'
    ja number_bad

frac_digits:
    inc r12
    cmp r12, r13
    jae number_ok
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    jb number_exp
    cmp al, '9'
    jbe frac_digits

number_exp:
    cmp r12, r13
    jae number_ok
    movzx eax, byte ptr [input + r12]
    cmp al, 'e'
    je exp_start
    cmp al, 'E'
    jne number_ok

exp_start:
    inc r12
    cmp r12, r13
    jae number_eof
    movzx eax, byte ptr [input + r12]
    cmp al, '+'
    je exp_sign
    cmp al, '-'
    jne exp_digit_check

exp_sign:
    inc r12
    cmp r12, r13
    jae number_eof

exp_digit_check:
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    jb number_bad
    cmp al, '9'
    ja number_bad

exp_digits:
    inc r12
    cmp r12, r13
    jae number_ok
    movzx eax, byte ptr [input + r12]
    cmp al, '0'
    jb number_ok
    cmp al, '9'
    jbe exp_digits

number_ok:
    mov eax, 1
    ret

number_eof:
    mov r14, r13
    xor eax, eax
    ret

number_bad:
    mov r14, r12
    xor eax, eax
    ret

parse_array:
    inc r12
    call parse_ws
    cmp r12, r13
    jae container_eof
    cmp byte ptr [input + r12], ']'
    je array_empty

array_value:
    call parse_value
    test eax, eax
    jz container_fail
    call parse_ws
    cmp r12, r13
    jae container_eof
    cmp byte ptr [input + r12], ']'
    je array_done
    cmp byte ptr [input + r12], ','
    jne container_bad
    inc r12
    call parse_ws
    jmp array_value

array_empty:
    inc r12

array_done:
    cmp byte ptr [input + r12], ']'
    jne array_done_return
    inc r12
array_done_return:
    mov eax, 1
    ret

parse_object:
    inc r12
    call parse_ws
    cmp r12, r13
    jae container_eof
    cmp byte ptr [input + r12], '}'
    je object_empty

object_member:
    cmp byte ptr [input + r12], '"'
    jne container_bad
    call parse_string
    test eax, eax
    jz container_fail
    call parse_ws
    cmp r12, r13
    jae container_eof
    cmp byte ptr [input + r12], ':'
    jne container_bad
    inc r12
    call parse_ws
    call parse_value
    test eax, eax
    jz container_fail
    call parse_ws
    cmp r12, r13
    jae container_eof
    cmp byte ptr [input + r12], '}'
    je object_done
    cmp byte ptr [input + r12], ','
    jne container_bad
    inc r12
    call parse_ws
    jmp object_member

object_empty:
    inc r12

object_done:
    cmp byte ptr [input + r12], '}'
    jne object_done_return
    inc r12
object_done_return:
    mov eax, 1
    ret

container_eof:
    mov r14, r13

container_fail:
    xor eax, eax
    ret

container_bad:
    mov r14, r12
    xor eax, eax
    ret
