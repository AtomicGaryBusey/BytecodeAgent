.equ INPUT_SIZE, 32768
.equ MAX_LINES, 1000

.section .text
.globl _start

_start:
    lea rbx, [rip + input]
    xor r12, r12

read_more:
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    cmp r12, INPUT_SIZE
    jb read_more
    jmp read_more

parse_input:
    xor r13, r13
    xor r14, r14
    xor r15, r15
    lea r8, [rip + starts]
    lea r9, [rip + lengths]

parse_loop:
    cmp r14, r12
    jae parse_end
    cmp byte ptr [rbx + r14], 10
    jne parse_next
    mov qword ptr [r8 + r15*8], r13
    mov rax, r14
    sub rax, r13
    mov qword ptr [r9 + r15*8], rax
    inc r15
    lea r13, [r14 + 1]

parse_next:
    inc r14
    jmp parse_loop

parse_end:
    cmp r13, r12
    jae sort_begin
    mov qword ptr [r8 + r15*8], r13
    mov rax, r12
    sub rax, r13
    mov qword ptr [r9 + r15*8], rax
    inc r15

sort_begin:
    mov r13, r15
    xor r12, r12

sort_outer:
    cmp r12, r13
    jae output_begin
    mov r14, r12
    lea r15, [r12 + 1]

sort_inner:
    cmp r15, r13
    jae sort_swap
    mov rdi, r14
    mov rsi, r15
    call compare_records
    cmp eax, 0
    jle sort_next
    mov r14, r15

sort_next:
    inc r15
    jmp sort_inner

sort_swap:
    cmp r14, r12
    je sort_advance
    lea r8, [rip + starts]
    lea r9, [rip + lengths]

    mov rax, qword ptr [r8 + r12*8]
    mov rdx, qword ptr [r8 + r14*8]
    mov qword ptr [r8 + r12*8], rdx
    mov qword ptr [r8 + r14*8], rax

    mov rax, qword ptr [r9 + r12*8]
    mov rdx, qword ptr [r9 + r14*8]
    mov qword ptr [r9 + r12*8], rdx
    mov qword ptr [r9 + r14*8], rax

sort_advance:
    inc r12
    jmp sort_outer

output_begin:
    xor r12, r12

output_loop:
    cmp r12, r13
    jae program_exit

    lea r8, [rip + starts]
    lea r9, [rip + lengths]
    mov r14, qword ptr [r8 + r12*8]
    add r14, rbx
    mov r15, qword ptr [r9 + r12*8]

write_line:
    test r15, r15
    jz write_newline
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r15
    syscall
    add r14, rax
    sub r15, rax
    jmp write_line

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r12
    jmp output_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

compare_records:
    lea r8, [rip + starts]
    lea r9, [rip + lengths]
    mov r10, qword ptr [r8 + rdi*8]
    mov r11, qword ptr [r9 + rdi*8]
    mov r8, qword ptr [r8 + rsi*8]
    mov r9, qword ptr [r9 + rsi*8]

    lea rdx, [rip + input]
    add r10, rdx
    add r8, rdx
    mov rdx, qword ptr [rip + lengths]
    mov r11, qword ptr [rip + lengths + 8]
    add rdx, rdx

    lea rdx, [rip + lengths]
    mov rax, qword ptr [rdx + rdi*8]
    mov rcx, qword ptr [rdx + rsi*8]
    xor rdx, rdx

compare_loop:
    cmp rdx, rax
    jae compare_lengths
    cmp rdx, rcx
    jae compare_lengths
    movzx edi, byte ptr [r10 + rdx]
    movzx esi, byte ptr [r8 + rdx]
    cmp edi, esi
    jb compare_less
    ja compare_greater
    inc rdx
    jmp compare_loop

compare_lengths:
    cmp rax, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.balign 8
input:
    .zero INPUT_SIZE
starts:
    .zero MAX_LINES * 8
lengths:
    .zero MAX_LINES * 8
