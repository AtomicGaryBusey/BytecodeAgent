.equ INPUT_SIZE, 65536
.equ LINE_SIZE, 128

.section .text
.globl _start
_start:
    lea r14, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    mov edx, INPUT_SIZE
    sub rdx, r12
    lea rsi, [r14 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    js read_loop
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    lea rbx, [rip + line_buffer]
    xor r13d, r13d

row_loop:
    lea r11, [rip + hex_table]
    cmp r13, r12
    jae final_offset

    mov r15, r12
    sub r15, r13
    cmp r15, 16
    jbe row_size_ready
    mov r15d, 16

row_size_ready:
    xor ecx, ecx

fill_spaces:
    mov byte ptr [rbx + rcx + 10], 32
    inc ecx
    cmp ecx, 49
    jb fill_spaces

    mov byte ptr [rbx + 59], 32
    mov byte ptr [rbx + 60], 124

    mov rdx, r13
    xor ecx, ecx

offset_digits:
    mov rax, rdx
    shr rax, 28
    and eax, 15
    mov al, byte ptr [r11 + rax]
    mov byte ptr [rbx + rcx], al
    shl rdx, 4
    inc ecx
    cmp ecx, 8
    jb offset_digits

    lea r10, [r14 + r13]
    lea r9, [rbx + 10]
    xor ecx, ecx

byte_loop:
    cmp rcx, r15
    jae row_ready

    cmp ecx, 8
    jne hex_position_ready
    inc r9

hex_position_ready:
    movzx eax, byte ptr [r10]
    mov r8d, eax

    mov edx, eax
    shr edx, 4
    movzx eax, byte ptr [r11 + rdx]
    mov byte ptr [r9], al

    mov edx, r8d
    and edx, 15
    movzx eax, byte ptr [r11 + rdx]
    mov byte ptr [r9 + 1], al

    mov eax, r8d
    cmp eax, 32
    jb nonprintable
    cmp eax, 126
    ja nonprintable
    mov byte ptr [rbx + rcx + 61], al
    jmp ascii_done

nonprintable:
    mov byte ptr [rbx + rcx + 61], 46

ascii_done:
    inc r10
    add r9, 3
    inc rcx
    jmp byte_loop

row_ready:
    mov byte ptr [rbx + r15 + 61], 124
    mov byte ptr [rbx + r15 + 62], 10

    lea rsi, [rip + line_buffer]
    mov edx, 63
    add rdx, r15

write_row:
    mov edi, 1
    mov eax, 1
    syscall
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz write_row

    add r13, r15
    jmp row_loop

final_offset:
    lea r11, [rip + hex_table]
    mov rdx, r13
    xor ecx, ecx

final_digits:
    mov rax, rdx
    shr rax, 28
    and eax, 15
    mov al, byte ptr [r11 + rax]
    mov byte ptr [rbx + rcx], al
    shl rdx, 4
    inc ecx
    cmp ecx, 8
    jb final_digits

    mov byte ptr [rbx + 8], 10
    lea rsi, [rip + line_buffer]
    mov edx, 9

write_final:
    mov edi, 1
    mov eax, 1
    syscall
    sub rdx, rax
    add rsi, rax
    test rdx, rdx
    jnz write_final

    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buffer:
    .zero INPUT_SIZE

line_buffer:
    .zero LINE_SIZE
