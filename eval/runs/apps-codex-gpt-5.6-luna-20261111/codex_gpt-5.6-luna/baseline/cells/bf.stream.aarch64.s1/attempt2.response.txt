.equ INMAX, 65536
.equ MATCH_SIZE, 262144
.equ STACK_SIZE, 4000
.equ TAPE_SIZE, 30000

.section .text
.globl _start

_start:
	adrp x19, input_buf
	add x19, x19, :lo12:input_buf
	mov x20, #0

read_more:
	mov x0, #0
	add x1, x19, x20
	mov x2, #INMAX
	sub x2, x2, x20
	mov x8, #63
	svc #0
	cbz x0, read_finished
	add x20, x20, x0
	cmp x20, #INMAX
	b.lo read_more

read_finished:
	mov x22, x20
	mov x23, #0

find_bang:
	cmp x23, x22
	b.hs no_bang
	ldrb w0, [x19, x23]
	cmp w0, #33
	b.eq found_bang
	add x23, x23, #1
	b find_bang

found_bang:
	mov x20, x23
	add x21, x23, #1
	b validate_setup

no_bang:
	mov x20, x22
	mov x21, x22

validate_setup:
	adrp x25, bracket_stack
	add x25, x25, :lo12:bracket_stack
	adrp x26, bracket_match
	add x26, x26, :lo12:bracket_match
	mov x23, #0
	mov x24, #0

validate:
	cmp x23, x20
	b.hs validate_end
	ldrb w0, [x19, x23]
	cmp w0, #91
	b.eq validate_open
	cmp w0, #93
	b.eq validate_close
	add x23, x23, #1
	b validate

validate_open:
	str w23, [x25, x24, lsl #2]
	add x24, x24, #1
	add x23, x23, #1
	b validate

validate_close:
	cbz x24, invalid
	sub x24, x24, #1
	ldr w1, [x25, x24, lsl #2]
	str w23, [x26, x1, lsl #2]
	str w23, [x26, x23, lsl #2]
	add x23, x23, #1
	b validate

validate_end:
	cbnz x24, invalid

	adrp x27, tape
	add x27, x27, :lo12:tape
	mov x23, #0
	mov x24, #0

execute:
	cmp x23, x20
	b.hs success
	ldrb w0, [x19, x23]

	cmp w0, #62
	b.eq do_right
	cmp w0, #60
	b.eq do_left
	cmp w0, #43
	b.eq do_plus
	cmp w0, #45
	b.eq do_minus
	cmp w0, #46
	b.eq do_output
	cmp w0, #44
	b.eq do_input
	cmp w0, #91
	b.eq do_open
	cmp w0, #93
	b.eq do_close

	add x23, x23, #1
	b execute

do_right:
	add x24, x24, #1
	add x23, x23, #1
	b execute

do_left:
	sub x24, x24, #1
	add x23, x23, #1
	b execute

do_plus:
	ldrb w1, [x27, x24]
	add w1, w1, #1
	strb w1, [x27, x24]
	add x23, x23, #1
	b execute

do_minus:
	ldrb w1, [x27, x24]
	sub w1, w1, #1
	strb w1, [x27, x24]
	add x23, x23, #1
	b execute

do_output:
	mov x0, #1
	add x1, x27, x24
	mov x2, #1
	mov x8, #64
	svc #0
	add x23, x23, #1
	b execute

do_input:
	cmp x21, x22
	b.hs input_zero
	ldrb w1, [x19, x21]
	add x21, x21, #1
	strb w1, [x27, x24]
	add x23, x23, #1
	b execute

input_zero:
	strb wzr, [x27, x24]
	add x23, x23, #1
	b execute

do_open:
	ldrb w0, [x27, x24]
	cbnz w0, open_next
	ldr w1, [x26, x23, lsl #2]
	add x23, x1, #1
	b execute

open_next:
	add x23, x23, #1
	b execute

do_close:
	ldrb w0, [x27, x24]
	cbz w0, close_next
	ldr w1, [x26, x23, lsl #2]
	add x23, x1, #1
	b execute

close_next:
	add x23, x23, #1
	b execute

invalid:
	mov x0, #1
	mov x8, #93
	svc #0

success:
	mov x0, #0
	mov x8, #93
	svc #0

.section .bss
.align 4
input_buf:
.zero INMAX

.align 4
bracket_match:
.zero MATCH_SIZE

.align 4
bracket_stack:
.zero STACK_SIZE

.align 4
tape:
.zero TAPE_SIZE
