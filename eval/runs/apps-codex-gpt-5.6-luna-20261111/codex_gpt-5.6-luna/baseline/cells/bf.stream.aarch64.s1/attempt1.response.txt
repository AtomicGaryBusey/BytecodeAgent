.section .text
.globl _start

_start:
	adrp x19, input_buf
	add x19, x19, :lo12:input_buf
	mov x20, #0

read_loop:
	mov x2, #65536
	sub x2, x2, x20
	add x1, x19, x20
	mov x0, #0
	mov x8, #63
	svc #0
	cbz x0, read_done
	add x20, x20, x0
	cmp x20, #65536
	b.lo read_loop

read_done:
	mov x29, x20
	mov x23, #0

find_delim:
	cmp x23, x29
	b.hs no_delim
	ldrb w0, [x19, x23]
	cmp w0, #33
	b.eq delim_found
	add x23, x23, #1
	b find_delim

delim_found:
	mov x20, x23
	add x21, x23, #1
	b setup_validate

no_delim:
	mov x20, x29
	mov x21, x29

setup_validate:
	mov x22, x29
	adrp x25, bracket_stack
	add x25, x25, :lo12:bracket_stack
	adrp x26, bracket_match
	add x26, x26, :lo12:bracket_match
	mov x23, #0
	mov x24, #0

validate_loop:
	cmp x23, x20
	b.hs validate_done
	ldrb w0, [x19, x23]
	cmp w0, #91
	b.eq validate_open
	cmp w0, #93
	b.eq validate_close
	add x23, x23, #1
	b validate_loop

validate_open:
	str w23, [x25, x24, lsl #2]
	add x24, x24, #1
	add x23, x23, #1
	b validate_loop

validate_close:
	cbz x24, invalid
	sub x24, x24, #1
	ldr w1, [x25, x24, lsl #2]
	str w23, [x26, x1, lsl #2]
	str w23, [x26, x23, lsl #2]
	add x23, x23, #1
	b validate_loop

validate_done:
	cbnz x24, invalid

	adrp x27, tape
	add x27, x27, :lo12:tape
	mov x23, #0
	mov x24, #0

execute_loop:
	cmp x23, x20
	b.hs success
	ldrb w0, [x19, x23]
	cmp w0, #62
	b.eq command_right
	cmp w0, #60
	b.eq command_left
	cmp w0, #43
	b.eq command_plus
	cmp w0, #45
	b.eq command_minus
	cmp w0, #46
	b.eq command_output
	cmp w0, #44
	b.eq command_input
	cmp w0, #91
	b.eq command_open
	cmp w0, #93
	b.eq command_close
	add x23, x23, #1
	b execute_loop

command_right:
	add x24, x24, #1
	add x23, x23, #1
	b execute_loop

command_left:
	sub x24, x24, #1
	add x23, x23, #1
	b execute_loop

command_plus:
	ldrb w1, [x27, x24]
	add w1, w1, #1
	strb w1, [x27, x24]
	add x23, x23, #1
	b execute_loop

command_minus:
	ldrb w1, [x27, x24]
	sub w1, w1, #1
	strb w1, [x27, x24]
	add x23, x23, #1
	b execute_loop

command_output:
	mov x0, #1
	add x1, x27, x24
	mov x2, #1
	mov x8, #64
	svc #0
	add x23, x23, #1
	b execute_loop

command_input:
	cmp x21, x22
	b.hs input_empty
	ldrb w1, [x19, x21]
	add x21, x21, #1
	strb w1, [x27, x24]
	add x23, x23, #1
	b execute_loop

input_empty:
	strb wzr, [x27, x24]
	add x23, x23, #1
	b execute_loop

command_open:
	ldrb w0, [x27, x24]
	cbnz w0, open_continue
	ldr w1, [x26, x23, lsl #2]
	add x23, x1, #1
	b execute_loop

open_continue:
	add x23, x23, #1
	b execute_loop

command_close:
	ldrb w0, [x27, x24]
	cbz w0, close_continue
	ldr w1, [x26, x23, lsl #2]
	add x23, x1, #1
	b execute_loop

close_continue:
	add x23, x23, #1
	b execute_loop

invalid:
	mov x0, #1
	mov x8, #93
	svc #0

success:
	mov x0, #0
	mov x8, #93
	svc #0

.section .bss
.align 4
input_buf:
.zero 65536

.align 4
bracket_match:
.zero 262144

.align 4
bracket_stack:
.zero 4000

.align 4

.zero 30000
