.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ BRACKET_STACK_SIZE, 8000
.equ BRACKET_MAP_SIZE, 524288
.equ OUTPUT_SIZE, 65536

.globl _start

.text
_start:
	adrp x19, input_buf
	add x19, x19, :lo12:input_buf
	mov x20, #0

read_more:
	mov x0, #0
	add x1, x19, x20
	mov x2, #INPUT_SIZE
	sub x2, x2, x20
	cbz x2, read_done
	mov x8, #63
	svc #0
	cbz x0, read_done
	add x20, x20, x0
	b read_more

read_done:
	mov x27, #0

find_delimiter:
	cmp x27, x20
	b.hs no_delimiter
	ldrb w0, [x19, x27]
	cmp w0, #33
	b.eq delimiter_found
	add x27, x27, #1
	b find_delimiter

delimiter_found:
	mov x21, x27
	add x18, x19, x27
	add x18, x18, #1
	sub x24, x20, x27
	sub x24, x24, #1
	b validate

no_delimiter:
	mov x21, x20
	add x18, x19, x20
	mov x24, #0

validate:
	adrp x25, bracket_map
	add x25, x25, :lo12:bracket_map
	adrp x28, bracket_stack
	add x28, x28, :lo12:bracket_stack
	mov x27, #0
	mov x29, #0

validate_loop:
	cmp x27, x21
	b.hs validate_end
	ldrb w0, [x19, x27]
	cmp w0, #91
	b.eq validate_open
	cmp w0, #93
	b.eq validate_close
	add x27, x27, #1
	b validate_loop

validate_open:
	add x1, x28, x29, lsl #3
	str x27, [x1]
	add x29, x29, #1
	add x27, x27, #1
	b validate_loop

validate_close:
	cbz x29, invalid
	sub x29, x29, #1
	add x1, x28, x29, lsl #3
	ldr x2, [x1]
	add x3, x25, x27, lsl #3
	str x2, [x3]
	add x3, x25, x2, lsl #3
	str x27, [x3]
	add x27, x27, #1
	b validate_loop

validate_end:
	cbnz x29, invalid

	adrp x23, tape
	add x23, x23, :lo12:tape
	adrp x25, bracket_map
	add x25, x25, :lo12:bracket_map
	adrp x28, output_buf
	add x28, x28, :lo12:output_buf
	mov x22, #0
	mov x26, #0
	mov x27, #0

execute_loop:
	cmp x27, x21
	b.hs write_output
	ldrb w0, [x19, x27]
	add x27, x27, #1

	cmp w0, #62
	b.eq command_right
	cmp w0, #60
	b.eq command_left
	cmp w0, #43
	b.eq command_plus
	cmp w0, #45
	b.eq command_minus
	cmp w0, #46
	b.eq command_output
	cmp w0, #44
	b.eq command_input
	cmp w0, #91
	b.eq command_open
	cmp w0, #93
	b.eq command_close
	b execute_loop

command_right:
	add x23, x23, #1
	b execute_loop

command_left:
	sub x23, x23, #1
	b execute_loop

command_plus:
	ldrb w1, [x23]
	add w1, w1, #1
	strb w1, [x23]
	b execute_loop

command_minus:
	ldrb w1, [x23]
	sub w1, w1, #1
	strb w1, [x23]
	b execute_loop

command_output:
	ldrb w1, [x23]
	strb w1, [x28, x26]
	add x26, x26, #1
	b execute_loop

command_input:
	cmp x22, x24
	b.hs input_empty
	ldrb w1, [x18, x22]
	add x22, x22, #1
	strb w1, [x23]
	b execute_loop

input_empty:
	strb wzr, [x23]
	b execute_loop

command_open:
	ldrb w1, [x23]
	cbnz w1, execute_loop
	sub x1, x27, #1
	ldr x2, [x25, x1, lsl #3]
	add x27, x2, #1
	b execute_loop

command_close:
	ldrb w1, [x23]
	cbz w1, execute_loop
	sub x1, x27, #1
	ldr x2, [x25, x1, lsl #3]
	add x27, x2, #1
	b execute_loop

write_output:
	cbz x26, success

write_loop:
	mov x0, #1
	mov x1, x28
	mov x2, x26
	mov x8, #64
	svc #0
	add x28, x28, x0
	sub x26, x26, x0
	cbnz x26, write_loop

success:
	mov x0, #0
	mov x8, #93
	svc #0
	b success

invalid:
	mov x0, #1
	mov x8, #93
	svc #0
	b invalid

.bss
.balign 16
input_buf:
	.zero INPUT_SIZE

.balign 16
tape:
	.zero TAPE_SIZE

.balign 16
bracket_stack:
	.zero BRACKET_STACK_SIZE

.balign 16
bracket_map:
	.zero BRACKET_MAP_SIZE

.balign 16
output_buf:
	.zero OUTPUT_SIZE
