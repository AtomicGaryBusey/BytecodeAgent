.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ BRACKET_SIZE, 8000
.equ MAP_SIZE, 524288
.equ OUTPUT_SIZE, 65536

.globl _start

.text
_start:
	adrp x19, input_buf
	add x19, x19, :lo12:input_buf
	mov x20, #0

bf_read:
	add x1, x19, x20
	mov x2, #INPUT_SIZE
	sub x2, x2, x20
	mov x0, #0
	mov x8, #63
	svc #0
	cbz x0, bf_read_done
	add x20, x20, x0
	b bf_read

bf_read_done:
	mov x25, #0

bf_find_delim:
	cmp x25, x20
	b.hs bf_no_delim
	ldrb w0, [x19, x25]
	cmp w0, #33
	b.eq bf_have_delim
	add x25, x25, #1
	b bf_find_delim

bf_have_delim:
	mov x22, x25
	add x23, x19, x25
	add x23, x23, #1
	sub x24, x20, x25
	sub x24, x24, #1
	b bf_validate_start

bf_no_delim:
	mov x22, x20
	add x23, x19, x20
	mov x24, #0

bf_validate_start:
	adrp x27, bracket_map
	add x27, x27, :lo12:bracket_map
	adrp x28, bracket_stack
	add x28, x28, :lo12:bracket_stack
	mov x25, #0
	mov x26, #0

bf_validate_loop:
	cmp x25, x22
	b.hs bf_validate_done
	ldrb w0, [x19, x25]
	cmp w0, #91
	b.eq bf_validate_open
	cmp w0, #93
	b.eq bf_validate_close
	add x25, x25, #1
	b bf_validate_loop

bf_validate_open:
	add x1, x28, x26, lsl #3
	str x25, [x1]
	add x26, x26, #1
	add x25, x25, #1
	b bf_validate_loop

bf_validate_close:
	cbz x26, bf_invalid
	sub x26, x26, #1
	add x1, x28, x26, lsl #3
	ldr x2, [x1]
	add x3, x27, x25, lsl #3
	str x2, [x3]
	add x3, x27, x2, lsl #3
	str x25, [x3]
	add x25, x25, #1
	b bf_validate_loop

bf_validate_done:
	cbnz x26, bf_invalid

	adrp x26, tape
	add x26, x26, :lo12:tape
	adrp x20, bracket_map
	add x20, x20, :lo12:bracket_map
	adrp x28, output_buf
	add x28, x28, :lo12:output_buf
	mov x25, #0
	mov x27, #0
	mov x29, #0

bf_exec_loop:
	cmp x27, x22
	b.hs bf_write_output
	ldrb w0, [x19, x27]
	add x27, x27, #1

	cmp w0, #62
	b.eq bf_cmd_right
	cmp w0, #60
	b.eq bf_cmd_left
	cmp w0, #43
	b.eq bf_cmd_plus
	cmp w0, #45
	b.eq bf_cmd_minus
	cmp w0, #46
	b.eq bf_cmd_output
	cmp w0, #44
	b.eq bf_cmd_input
	cmp w0, #91
	b.eq bf_cmd_open
	cmp w0, #93
	b.eq bf_cmd_close
	b bf_exec_loop

bf_cmd_right:
	add x26, x26, #1
	b bf_exec_loop

bf_cmd_left:
	sub x26, x26, #1
	b bf_exec_loop

bf_cmd_plus:
	ldrb w1, [x26]
	add w1, w1, #1
	strb w1, [x26]
	b bf_exec_loop

bf_cmd_minus:
	ldrb w1, [x26]
	sub w1, w1, #1
	strb w1, [x26]
	b bf_exec_loop

bf_cmd_output:
	ldrb w1, [x26]
	strb w1, [x28], #1
	add x29, x29, #1
	b bf_exec_loop

bf_cmd_input:
	cmp x25, x24
	b.hs bf_input_empty
	ldrb w1, [x23, x25]
	add x25, x25, #1
	strb w1, [x26]
	b bf_exec_loop

bf_input_empty:
	mov w1, #0
	strb w1, [x26]
	b bf_exec_loop

bf_cmd_open:
	ldrb w1, [x26]
	cbnz w1, bf_exec_loop
	sub x1, x27, #1
	ldr x2, [x20, x1, lsl #3]
	add x27, x2, #1
	b bf_exec_loop

bf_cmd_close:
	ldrb w1, [x26]
	cbz w1, bf_exec_loop
	sub x1, x27, #1
	ldr x2, [x20, x1, lsl #3]
	add x27, x2, #1
	b bf_exec_loop

bf_write_output:
	cbz x29, bf_success

bf_write_loop:
	mov x0, #1
	mov x1, x28
	mov x2, x29
	mov x8, #64
	svc #0
	cbz x0, bf_write_loop
	add x28, x28, x0
	sub x29, x29, x0
	cbnz x29, bf_write_loop

bf_success:
	mov x0, #0
	mov x8, #93
	svc #0
	b bf_success

bf_invalid:
	mov x0, #1
	mov x8, #93
	svc #0
	b bf_invalid

.bss
