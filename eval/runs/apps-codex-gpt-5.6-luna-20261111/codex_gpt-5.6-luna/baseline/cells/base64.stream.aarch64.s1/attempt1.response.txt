.equ IN_SIZE, 49152
.equ OUT_SIZE, 70000

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #IN_SIZE
    mov x19, #0

read_loop:
    sub x2, x21, x19
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le input_done
    add x19, x19, x0
    cmp x19, x21
    b.lo read_loop

input_done:
    mov x21, x19
    mov x19, #0
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x23, x22
    mov x25, #0
    adrp x26, alphabet
    add x26, x26, :lo12:alphabet

encode_loop:
    sub x27, x21, x19
    cmp x27, #3
    b.lo remainder

    ldrb w27, [x20, x19]
    add x28, x19, #1
    ldrb w28, [x20, x28]
    add x29, x19, #2
    ldrb w29, [x20, x29]

    lsr x0, x27, #2
    ldrb w0, [x26, x0]
    bl emit

    and x0, x27, #3
    lsl x0, x0, #4
    lsr x1, x28, #4
    orr x0, x0, x1
    ldrb w0, [x26, x0]
    bl emit

    and x0, x28, #15
    lsl x0, x0, #2
    lsr x1, x29, #6
    orr x0, x0, x1
    ldrb w0, [x26, x0]
    bl emit

    and x0, x29, #63
    ldrb w0, [x26, x0]
    bl emit

    add x19, x19, #3
    b encode_loop

remainder:
    cbz x27, encoding_done

    ldrb w27, [x20, x19]
    lsr x0, x27, #2
    ldrb w0, [x26, x0]
    bl emit

    and x0, x27, #3
    lsl x0, x0, #4
    cmp x27, #2
    b.lo one_byte
    ldrb w28, [x20, x19, lsl #0]
    add x28, x19, #1
    ldrb w28, [x20, x28]
    lsr x1, x28, #4
    orr x0, x0, x1
    ldrb w0, [x26, x0]
    bl emit

    and x0, x28, #15
    lsl x0, x0, #2
    ldrb w0, [x26, x0]
    bl emit
    mov w0, #'=
    bl emit
    b encoding_done

one_byte:
    ldrb w0, [x26, x0]
    bl emit
    mov w0, #'=
    bl emit
    mov w0, #'=
    bl emit

encoding_done:
    cbz x25, prepare_write
    mov w0, #10
    bl emit

prepare_write:
    sub x24, x23, x22
    mov x25, #0

write_loop:
    cmp x25, x24
    b.hs exit_success
    sub x2, x24, x25
    add x1, x22, x25
    mov x0, #1
    mov x8, #64
    svc #0
    add x25, x25, x0
    b write_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

emit:
    strb w0, [x23], #1
    add x25, x25, #1
    cmp x25, #76
    b.ne 1f
    mov w0, #10
    strb w0, [x23], #1
    mov x25, #0
1:
    ret

.bss
.align 3
input_buf:
.zero IN_SIZE
.align 3
output_buf:
.zero OUT_SIZE
