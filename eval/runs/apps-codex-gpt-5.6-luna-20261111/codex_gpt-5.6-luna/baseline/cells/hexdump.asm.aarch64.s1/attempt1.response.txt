.equ READ_NR, 63
.equ WRITE_NR, 64
.equ EXIT_NR, 93
.equ MAX_INPUT, 65536
.equ ROW_LEN, 79

.text
.globl _start
_start:
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    mov x19, #0

read_loop:
    mov x10, #MAX_INPUT
    cmp x19, x10
    b.eq read_full

    sub x2, x10, x19
    add x1, x24, x19
    mov x0, #0
    mov x8, #READ_NR
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_full:
    adrp x1, extra_byte
    add x1, x1, :lo12:extra_byte
    mov x0, #0
    mov x2, #1
    mov x8, #READ_NR
    svc #0

read_done:
    mov x21, #0
    mov x20, #0
    adrp x22, hex_digits
    add x22, x22, :lo12:hex_digits
    adrp x23, output_row
    add x23, x23, :lo12:output_row

row_loop:
    cmp x21, x19
    b.eq final_line

    sub x26, x19, x21
    mov x27, #16
    cmp x26, x27
    csel x26, x27, x26, hi

    mov x25, #0
    mov x27, #28
offset_loop:
    lsr x28, x20, x27
    and x28, x28, #15
    ldrb w28, [x22, x28]
    strb w28, [x23, x25]
    add x25, x25, #1
    sub x27, x27, #4
    cbnz x27, offset_loop

    mov w28, #32
    strb w28, [x23, #8]
    strb w28, [x23, #9]

    mov x25, #0
hex_loop:
    cmp x25, #16
    b.eq ascii_start
    cmp x25, x26
    b.hs hex_padding

    add x27, x21, x25
    ldrb w28, [x24, x27]
    lsr w29, w28, #4
    ldrb w29, [x22, x29]
    strb w29, [x23, x25, lsl #2]

    and w29, w28, #15
    ldrb w29, [x22, x29]
    add x27, x25, #1
    add x27, x27, x25, lsl #1
    strb w29, [x23, x27]

    mov w28, #32
    add x27, x27, #1
    strb w28, [x23, x27]
    add x25, x25, #1
    cmp x25, #8
    b.ne hex_loop
    strb w28, [x23, #34]
    b hex_loop

hex_padding:
    add x27, x25, x25, lsl #1
    add x27, x27, x25
    mov w28, #32
    strb w28, [x23, x27]
    add x27, x27, #1
    strb w28, [x23, x27]
    add x27, x27, #2
    strb w28, [x23, x27]
    add x25, x25, #1
    cmp x25, #8
    b.ne hex_loop
    strb w28, [x23, #34]
    b hex_loop

ascii_start:
    mov w28, #32
    strb w28, [x23, #59]
    mov w28, #124
    strb w28, [x23, #60]

    mov x25, #0
ascii_loop:
    cmp x25, x26
    b.eq ascii_end
    add x27, x21, x25
    ldrb w28, [x24, x27]
    cmp w28, #32
    b.lo ascii_dot
    cmp w28, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w28, #46
ascii_store:
    add x27, x25, #61
    strb w28, [x23, x27]
    add x25, x25, #1
    b ascii_loop

ascii_end:
    add x27, x26, #61
    mov w28, #124
    strb w28, [x23, x27]
    add x27, x27, #1
    mov w28, #10
    strb w28, [x23, x27]

    mov x0, x23
    mov x1, #ROW_LEN
    bl write_all

    add x21, x21, x26
    add x20, x20, x26
    b row_loop

final_line:
    mov x25, #0
    mov x27, #28
final_offset_loop:
    lsr x28, x20, x27
    and x28, x28, #15
    ldrb w28, [x22, x28]
    strb w28, [x23, x25]
    add x25, x25, #1
    sub x27, x27, #4
    cbnz x27, final_offset_loop

    mov w28, #10
    strb w28, [x23, #8]
    mov x0, x23
    mov x1, #9
    bl write_all

    mov x0, #0
    mov x8, #EXIT_NR
    svc #0

write_all:
    mov x9, x0
    mov x10, x1
write_loop:
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #WRITE_NR
    svc #0
    add x9, x9, x0
    sub x10, x10, x0
    cbnz x10, write_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .zero 65536
output_row:
    .zero 79
extra_byte:
    .zero 1
