.equ INPUT_SIZE, 65536
.equ OUTPUT_SIZE, 65537

.section .text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_more:
    cmp x19, #INPUT_SIZE
    b.eq input_done
    add x1, x20, x19
    mov x2, #INPUT_SIZE
    sub x2, x2, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq input_done
    add x19, x19, x0
    b read_more

input_done:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x23, #0
    mov x24, #0
    mov x25, x22

scan:
    cmp x24, x19
    b.eq final_line
    ldrb w26, [x21, x24]
    add x24, x24, #1
    cmp w26, #10
    b.eq newline_line
    b scan

newline_line:
    sub x26, x24, #1

reverse_line:
    cmp x26, x23
    b.eq line_reversed
    sub x26, x26, #1
    ldrb w27, [x21, x26]
    strb w27, [x25], #1
    b reverse_line

line_reversed:
    mov w27, #10
    strb w27, [x25], #1
    mov x23, x24
    b scan

final_line:
    cmp x23, x19
    b.eq output_ready
    mov x26, x19

reverse_final:
    cmp x26, x23
    b.eq final_reversed
    sub x26, x26, #1
    ldrb w27, [x21, x26]
    strb w27, [x25], #1
    b reverse_final

final_reversed:
    mov w27, #10
    strb w27, [x25], #1

output_ready:
    sub x26, x25, x22
    mov x24, x22

write_more:
    cmp x26, #0
    b.eq done
    mov x0, #1
    mov x1, x24
    mov x2, x26
    mov x8, #64
    svc #0
    add x24, x24, x0
    sub x26, x26, x0
    b write_more

done:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.balign 16
input_buf:
    .zero INPUT_SIZE
output_buf:
    .zero OUTPUT_SIZE
