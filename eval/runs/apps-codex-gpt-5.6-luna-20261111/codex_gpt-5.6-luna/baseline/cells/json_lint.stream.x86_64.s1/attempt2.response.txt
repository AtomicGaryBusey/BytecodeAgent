.equ BUF_SIZE, 65536

.section .rodata
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
err_prefix:
    .ascii "error at "
err_prefix_len = . - err_prefix
ok_msg:
    .ascii "ok\n"
ok_msg_len = . - ok_msg
newline:
    .ascii "\n"

.section .bss
.align 16
input:
    .zero BUF_SIZE
outbuf:
    .zero 64

.text
.globl _start
_start:
    xor r13d, r13d
    mov rbx, offset input

read_loop:
    mov eax, BUF_SIZE
    sub rax, r13
    mov rdx, rax
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    xor r12d, r12d
    call skip_ws
    call parse_value
    test eax, eax
    jnz invalid
    call skip_ws
    cmp r12, r13
    jne invalid

valid:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_msg_len
    call write_all
    mov eax, 60
    xor edi, edi
    syscall

invalid:
    mov r14, r12
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, err_prefix_len
    call write_all

    lea r9, [rip + outbuf + 64]
    mov r8, r14
    test r8, r8
    jnz decimal_loop
    dec r9
    mov byte ptr [r9], '0'
    mov rsi, r9
    mov edx, 1
    jmp decimal_write

decimal_loop:
    xor edx, edx
    mov rax, r8
    mov ecx, 10
    div rcx
    add dl, '0'
    dec r9
    mov byte ptr [r9], dl
    mov r8, rax
    test r8, r8
    jnz decimal_loop
    mov rsi, r9
    lea rdx, [rip + outbuf + 64]
    sub rdx, rsi

decimal_write:
    call write_all
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    mov eax, 60
    mov edi, 1
    syscall

write_all:
    mov r9, rsi
    mov r8, rdx
write_loop:
    test r8, r8
    jz write_done
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    syscall
    add r9, rax
    sub r8, rax
    jmp write_loop
write_done:
    ret

skip_ws:
skip_ws_loop:
    cmp r12, r13
    jae skip_ws_done
    movzx eax, byte ptr [rbx + r12]
    cmp al, 32
    je skip_ws_one
    cmp al, 9
    je skip_ws_one
    cmp al, 10
    je skip_ws_one
    cmp al, 13
    je skip_ws_one
    ret
skip_ws_one:
    inc r12
    jmp skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp r12, r13
    jae parse_value_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_value_keyword
    cmp al, '9'
    jbe parse_number

parse_value_keyword:
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null

parse_value_error:
    mov eax, 1
    ret

value_true:
    mov r15, offset lit_true
    mov ecx, 4
    call check_literal
    ret

value_false:
    mov r15, offset lit_false
    mov ecx, 5
    call check_literal
    ret

value_null:
    mov r15, offset lit_null
    mov ecx, 4
    call check_literal
    ret

check_literal:
    xor edx, edx
literal_loop:
    cmp rdx, rcx
    jae literal_ok
    cmp r12, r13
    jae literal_bad
    movzx eax, byte ptr [r15 + rdx]
    cmp byte ptr [rbx + r12], al
    jne literal_bad
    inc r12
    inc rdx
    jmp literal_loop

literal_ok:
    xor eax, eax
    ret

literal_bad:
    mov eax, 1
    ret

parse_string:
    inc r12
string_loop:
    cmp r12, r13
    jae string_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '"'
    je string_done
    cmp al, 92
    je string_escape
    cmp al, 32
    jb string_error
    inc r12
    jmp string_loop

string_escape:
    inc r12
    cmp r12, r13
    jae string_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '"'
    je simple_escape
    cmp al, 92
    je simple_escape
    cmp al, '/'
    je simple_escape
    cmp al, 'b'
    je simple_escape
    cmp al, 'f'
    je simple_escape
    cmp al, 'n'
    je simple_escape
    cmp al, 'r'
    je simple_escape
    cmp al, 't'
    je simple_escape
    cmp al, 'u'
    jne string_error
    inc r12
    mov edx, 4

unicode_loop:
    cmp r12, r13
    jae string_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb unicode_alpha
    cmp al, '9'
    jbe unicode_digit

unicode_alpha:
    cmp al, 'a'
    jb unicode_upper
    cmp al, 'f'
    jbe unicode_digit

unicode_upper:
    cmp al, 'A'
    jb string_error
    cmp al, 'F'
    ja string_error

unicode_digit:
    inc r12
    dec edx
    jnz unicode_loop
    jmp string_loop

simple_escape:
    inc r12
    jmp string_loop

string_done:
    inc r12
    xor eax, eax
    ret

string_error:
    mov eax, 1
    ret

parse_number:
    cmp byte ptr [rbx + r12], '-'
    jne number_start
    inc r12
    cmp r12, r13
    jae number_error

number_start:
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb number_error
    cmp al, '9'
    ja number_error

number_digits:
    inc r12
    cmp r12, r13
    jae number_suffix
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb number_suffix
    cmp al, '9'
    jbe number_digits
    jmp number_suffix

number_zero:
    inc r12

number_suffix:
    cmp r12, r13
    jae number_ok
    movzx eax, byte ptr [rbx + r12]
    cmp al, '.'
    je number_fraction
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent
    jmp number_ok

number_fraction:
    inc r12
    cmp r12, r13
    jae number_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb number_error
    cmp al, '9'
    ja number_error

fraction_digits:
    inc r12
    cmp r12, r13
    jae number_ok
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb fraction_suffix
    cmp al, '9'
    jbe fraction_digits

fraction_suffix:
    cmp al, 'e'
    je number_exponent
    cmp al, 'E'
    je number_exponent

number_ok:
    xor eax, eax
    ret

number_exponent:
    inc r12
    cmp r12, r13
    jae number_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '+'
    je exponent_sign
    cmp al, '-'
    jne exponent_digit_check

exponent_sign:
    inc r12
    cmp r12, r13
    jae number_error

exponent_digit_check:
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb number_error
    cmp al, '9'
    ja number_error

exponent_digits:
    inc r12
    cmp r12, r13
    jae number_ok
    movzx eax, byte ptr [rbx + r12]
    cmp al, '0'
    jb number_ok
    cmp al, '9'
    jbe exponent_digits
    jmp number_ok

number_error:
    mov eax, 1
    ret

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    jae array_error
    cmp byte ptr [rbx + r12], ']'
    je array_empty_done

array_item:
    call parse_value
    test eax, eax
    jnz array_error
    call skip_ws
    cmp r12, r13
    jae array_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, ']'
    je array_done
    cmp al, ','
    jne array_error
    inc r12
    call skip_ws
    cmp r12, r13
    jae array_error
    jmp array_item

array_empty_done:
array_done:
    inc r12
    xor eax, eax
    ret

array_error:
    mov eax, 1
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    jae object_error
    cmp byte ptr [rbx + r12], '}'
    je object_empty_done

object_member:
    cmp byte ptr [rbx + r12], '"'
    jne object_error
    call parse_string
    test eax, eax
    jnz object_error
    call skip_ws
    cmp r12, r13
    jae object_error
    cmp byte ptr [rbx + r12], ':'
    jne object_error
    inc r12
    call skip_ws
    call parse_value
    test eax, eax
    jnz object_error
    call skip_ws
    cmp r12, r13
    jae object_error
    movzx eax, byte ptr [rbx + r12]
    cmp al, '}'
    je object_done
    cmp al, ','
    jne object_error
    inc r12
    call skip_ws
    cmp r12, r13
    jae object_error
    jmp object_member

object_empty_done:
object_done:
    inc r12
    xor eax, eax
    ret

object_error:
    mov eax, 1
    ret
