.equ BUFFER_SIZE, 65536

.globl _start

.text

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, BUFFER_SIZE
    syscall
    test rax, rax
    jz input_done

    add r14, rax
    mov rdx, rax
    lea rbx, [rip + input_buffer]

scan_buffer:
    test rdx, rdx
    jz read_more

    movzx eax, byte ptr [rbx]
    inc rbx
    dec rdx

    cmp eax, 10
    je newline_byte
    cmp eax, 9
    jb nonwhite_byte
    cmp eax, 13
    jbe whitespace_byte
    cmp eax, 32
    je whitespace_byte

nonwhite_byte:
    test r15d, r15d
    jnz scan_buffer
    inc r13
    mov r15d, 1
    jmp scan_buffer

whitespace_byte:
    xor r15d, r15d
    jmp scan_buffer

newline_byte:
    inc r12
    xor r15d, r15d
    jmp scan_buffer

input_done:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rbx, [rip + output_buffer]
    mov rsi, rbx
    mov rdx, rdi
    sub rdx, rbx

write_more:
    mov eax, 1
    mov edi, 1
    syscall
    sub rdx, rax
    add rsi, rax
    jnz write_more

    mov eax, 60
    xor edi, edi
    syscall

decimal:
    test rax, rax
    jnz decimal_nonzero

    mov byte ptr [rdi], 48
    inc rdi
    ret

decimal_nonzero:
    xor r8d, r8d
    mov ebx, 10

decimal_divide:
    xor edx, edx
    div rbx
    push rdx
    inc r8
    test rax, rax
    jnz decimal_divide

decimal_emit:
    pop rax
    add al, 48
    mov byte ptr [rdi], al
    inc rdi
    dec r8
    jnz decimal_emit
    ret

.section .bss
.balign 16
input_buffer:
    .zero BUFFER_SIZE
output_buffer:
    .zero 64
