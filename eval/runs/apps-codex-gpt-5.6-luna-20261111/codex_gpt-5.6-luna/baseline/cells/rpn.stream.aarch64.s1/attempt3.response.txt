.equ INPUT_SIZE, 65536

.section .text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_more:
    cmp x20, #INPUT_SIZE
    beq read_done
    add x1, x19, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_more

read_done:
    mov x22, x20
    mov x20, x19
    add x21, x19, x22

next_line:
    cmp x20, x21
    b.ge finish
    mov x24, x20

find_newline:
    cmp x24, x21
    b.ge final_line
    ldrb w25, [x24], #1
    cmp w25, #10
    b.ne find_newline
    stp x20, x24, [sp, #-16]!
    mov x0, x20
    sub x1, x24, #1
    bl process_line
    ldp x20, x24, [sp], #16
    mov x20, x24
    b next_line

final_line:
    mov x0, x20
    mov x1, x21
    bl process_line

finish:
    mov x0, #0
    mov x8, #93
    svc #0

process_line:
    stp x29, x30, [sp, #-16]!
    adrp x19, value_stack
    add x19, x19, :lo12:value_stack
    mov x23, #0
    mov x24, x0

token_loop:
    cmp x24, x1
    b.ge line_done
    ldrb w25, [x24]
    cmp w25, #32
    b.eq skip_separator
    cmp w25, #9
    b.eq skip_separator
    b token_start

skip_separator:
    add x24, x24, #1
    b token_loop

token_start:
    mov x0, x24

scan_token:
    cmp x24, x1
    b.ge token_ended
    ldrb w25, [x24]
    cmp w25, #32
    b.eq token_ended
    cmp w25, #9
    b.eq token_ended
    add x24, x24, #1
    b scan_token

token_ended:
    sub x25, x24, x0
    cbz x25, line_error
    cmp x25, #1
    b.ne parse_number

    ldrb w26, [x0]
    cmp w26, #43
    beq op_add_select
    cmp w26, #45
    beq op_sub_select
    cmp w26, #42
    beq op_mul_select
    cmp w26, #47
    beq op_div_select
    cmp w26, #37
    beq op_rem_select
    b parse_number

op_add_select:
    mov x22, #1
    b do_operator

op_sub_select:
    mov x22, #2
    b do_operator

op_mul_select:
    mov x22, #3
    b do_operator

op_div_select:
    mov x22, #4
    b do_operator

op_rem_select:
    mov x22, #5

do_operator:
    cmp x23, #2
    b.lo line_error
    sub x26, x23, #2
    ldr x27, [x19, x26, lsl #3]
    add x28, x26, #1
    ldr x29, [x19, x28, lsl #3]
    cmp x22, #1
    beq operator_add
    cmp x22, #2
    beq operator_sub
    cmp x22, #3
    beq operator_mul
    cmp x22, #4
    beq operator_div

    cbz x29, line_error
    mov x30, #1
    lsl x30, x30, #63
    cmp x27, x30
    b.ne remainder_do
    mov x30, #-1
    cmp x29, x30
    beq line_error

remainder_do:
    sdiv x30, x27, x29
    msub x27, x30, x29, x27
    b operator_store

operator_add:
    add x27, x27, x29
    b operator_store

operator_sub:
    sub x27, x27, x29
    b operator_store

operator_mul:
    mul x27, x27, x29
    b operator_store

operator_div:
    cbz x29, line_error
    mov x30, #1
    lsl x30, x30, #63
    cmp x27, x30
    b.ne division_do
    mov x30, #-1
    cmp x29, x30
    beq line_error

division_do:
    sdiv x27, x27, x29

operator_store:
    str x27, [x19, x26, lsl #3]
    sub x23, x23, #1
    b token_loop

parse_number:
    mov x27, x0
    mov x28, #0
    mov x22, #0
    ldrb w29, [x27]
    cmp w29, #45
    b.ne number_digits
    mov x22, #1
    add x27, x27, #1

number_digits:
    mov x26, #0

number_loop:
    cmp x27, x24
    b.ge number_finished
    ldrb w29, [x27]
    cmp w29, #48
    b.lo line_error
    cmp w29, #57
    b.hi line_error
    cmp x26, #18
    b.hs line_error
    add x28, x28, x28, lsl #2
    lsl x28, x28, #1
    sub w29, w29, #48
    add x28, x28, x29
    add x26, x26, #1
    add x27, x27, #1
    b number_loop

number_finished:
    cbz x26, line_error
    cbz x22, number_store
    neg x28, x28

number_store:
    str x28, [x19, x23, lsl #3]
    add x23, x23, #1
    b token_loop

line_done:
    cmp x23, #1
    b.ne line_error
    ldr x0, [x19]
    bl emit_value
    ldp x29, x30, [sp], #16
    ret

line_error:
    bl emit_error
    ldp x29, x30, [sp], #16
    ret

emit_error:
    stp x29, x30, [sp, #-16]!
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
    bl write_all
    ldp x29, x30, [sp], #16
    ret

emit_value:
    stp x29, x30, [sp, #-16]!
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x26, x1, #31
    mov x25, #0
    cmp x0, #0
    b.ge value_positive
    mov x25, #1
    neg x0, x0

value_positive:
    mov x27, #10

decimal_loop:
    udiv x28, x0, x27
    msub x29, x28, x27, x0
    add x29, x29, #48
    sub x26, x26, #1
    strb w29, [x26]
    mov x0, x28
    cbnz x0, decimal_loop
    cbz x25, add_newline
    sub x26, x26, #1
    mov w29, #45
    strb w29, [x26]

add_newline:
    mov w27, #10
    strb w27, [x1, #31]
    sub x2, x1, x26
    add x2, x2, #1
    mov x1, x26
    bl write_all
    ldp x29, x30, [sp], #16
    ret

write_all:
    mov x0, #1

write_loop:
    cbz x2, write_done
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    b write_loop

write_done:
    ret

.section .rodata
error_text:
    .ascii "error\n"

.section .bss
.balign 8
input_buf:
    .zero INPUT_SIZE

value_stack:
    .zero 8192

output_buf:
    .zero 32
