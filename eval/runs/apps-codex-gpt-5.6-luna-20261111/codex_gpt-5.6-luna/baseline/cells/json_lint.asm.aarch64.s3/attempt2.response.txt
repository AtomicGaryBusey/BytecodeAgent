.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUFSIZE, 65536

.section .rodata
ok_msg:
	.ascii "ok\n"
ok_len = . - ok_msg
err_msg:
	.ascii "error at "
err_len = . - err_msg
newline:
	.ascii "\n"

.section .bss
	.align 4
input_buf:
	.zero BUFSIZE
out_buf:
	.zero 64

.text
.globl _start
_start:
	adrp x19, input_buf
	add x19, x19, :lo12:input_buf
	mov x20, #0

read_more:
	mov x0, #0
	add x1, x19, x20
	mov x2, #BUFSIZE
	sub x2, x2, x20
	mov x8, #SYS_READ
	svc #0
	cbz x0, input_done
	add x20, x20, x0
	b read_more

input_done:
	mov x21, #0
	bl parse_ws
	bl parse_value
	cbz x0, report_error
	bl parse_ws
	cmp x21, x20
	b.ne report_error

	mov x0, #1
	adrp x1, ok_msg
	add x1, x1, :lo12:ok_msg
	mov x2, #ok_len
	bl write_all
	mov x0, #0
	mov x8, #SYS_EXIT
	svc #0

report_error:
	mov x0, #1
	adrp x1, err_msg
	add x1, x1, :lo12:err_msg
	mov x2, #err_len
	bl write_all

	adrp x28, out_buf
	add x28, x28, :lo12:out_buf
	add x23, x28, #64
	mov x24, x21
	mov x26, #10
	cbnz x24, decimal_loop
	sub x23, x23, #1
	mov w25, #'0'
	strb w25, [x23]
	mov x24, #1
	b decimal_done

decimal_loop:
	udiv x25, x24, x26
	msub x27, x25, x26, x24
	add w27, w27, #'0'
	sub x23, x23, #1
	strb w27, [x23]
	mov x24, x25
	cbnz x24, decimal_loop
	add x24, x28, #64
	sub x24, x24, x23

decimal_done:
	mov x0, #1
	mov x1, x23
	mov x2, x24
	bl write_all
	mov x0, #1
	adrp x1, newline
	add x1, x1, :lo12:newline
	mov x2, #1
	bl write_all
	mov x0, #1
	mov x8, #SYS_EXIT
	svc #0

write_all:
	stp x29, x30, [sp, #-16]!
	mov x29, sp
write_loop:
	cbz x2, write_done
	mov x0, #1
	mov x8, #SYS_WRITE
	svc #0
	add x1, x1, x0
	sub x2, x2, x0
	b write_loop
write_done:
	ldp x29, x30, [sp], #16
	ret

parse_ws:
parse_ws_loop:
	cmp x21, x20
	b.hs parse_ws_done
	ldrb w0, [x19, x21]
	cmp w0, #0x20
	b.eq parse_ws_inc
	cmp w0, #0x09
	b.eq parse_ws_inc
	cmp w0, #0x0a
	b.eq parse_ws_inc
	cmp w0, #0x0d
	b.eq parse_ws_inc
	b parse_ws_done
parse_ws_inc:
	add x21, x21, #1
	b parse_ws_loop
parse_ws_done:
	ret

parse_value:
	stp x29, x30, [sp, #-16]!
	mov x29, sp
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'{'
	b.eq parse_object_call
	cmp w0, #'['
	b.eq parse_array_call
	cmp w0, #'"'
	b.eq parse_string_call
	cmp w0, #'-'
	b.eq parse_number_call
	cmp w0, #'0'
	b.lo parse_value_fail
	cmp w0, #'9'
	b.ls parse_number_call
	cmp w0, #'t'
	b.eq parse_true
	cmp w0, #'f'
	b.eq parse_false
	cmp w0, #'n'
	b.eq parse_null
	b parse_value_fail

parse_object_call:
	bl parse_object
	b parse_value_return
parse_array_call:
	bl parse_array
	b parse_value_return
parse_string_call:
	bl parse_string
	b parse_value_return
parse_number_call:
	bl parse_number
	b parse_value_return

parse_true:
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'r'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'u'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'e'
	b.ne parse_value_fail
	add x21, x21, #1
	mov x0, #1
	b parse_value_return

parse_false:
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'a'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'l'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'s'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'e'
	b.ne parse_value_fail
	add x21, x21, #1
	mov x0, #1
	b parse_value_return

parse_null:
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'u'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'l'
	b.ne parse_value_fail
	add x21, x21, #1
	cmp x21, x20
	b.hs parse_value_fail
	ldrb w0, [x19, x21]
	cmp w0, #'l'
	b.ne parse_value_fail
	add x21, x21, #1
	mov x0, #1
	b parse_value_return

parse_value_fail:
	mov x0, #0
parse_value_return:
	ldp x29, x30, [sp], #16
	ret

parse_string:
	stp x29, x30, [sp, #-16]!
	mov x29, sp
	add x21, x21, #1
string_loop:
	cmp x21, x20
	b.hs string_fail
	ldrb w0, [x19, x21]
	cmp w0, #'"'
	b.eq string_done
	cmp w0, #0x20
	b.lo string_fail
	cmp w0, #'\\'
	b.eq string_escape
	add x21, x21, #1
	b string_loop
string_escape:
	add x21, x21, #1
	cmp x21, x20
	b.hs string_fail
	ldrb w0, [x19, x21]
	cmp w0, #'u'
	b.eq string_unicode
	cmp w0, #'"'
	b.eq string_escaped
	cmp w0, #'\\'
	b.eq string_escaped
	cmp w0, #'/'
	b.eq string_escaped
	cmp w0, #'b'
	b.eq string_escaped
	cmp w0, #'f'
	b.eq string_escaped
	cmp w0, #'n'
	b.eq string_escaped
	cmp w0, #'r'
	b.eq string_escaped
	cmp w0, #'t'
	b.ne string_fail
string_escaped:
	add x21, x21, #1
	b string_loop
string_unicode:
	add x21, x21, #1
	mov x1, #4
string_hex_loop:
	cmp x21, x20
	b.hs string_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo string_hex_alpha
	cmp w0, #'9'
	b.ls string_hex_ok
string_hex_alpha:
	cmp w0, #'a'
	b.lo string_hex_upper
	cmp w0, #'f'
	b.ls string_hex_ok
string_hex_upper:
	cmp w0, #'A'
	b.lo string_fail
	cmp w0, #'F'
	b.hi string_fail
string_hex_ok:
	add x21, x21, #1
	subs x1, x1, #1
	b.ne string_hex_loop
	b string_loop
string_done:
	add x21, x21, #1
	mov x0, #1
	b string_return
string_fail:
	mov x0, #0
string_return:
	ldp x29, x30, [sp], #16
	ret

parse_number:
	stp x29, x30, [sp, #-16]!
	mov x29, sp
	ldrb w0, [x19, x21]
	cmp w0, #'-'
	b.ne number_int
	add x21, x21, #1
	cmp x21, x20
	b.hs number_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo number_fail
	cmp w0, #'9'
	b.hi number_fail
number_int:
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.ne number_nonzero
	add x21, x21, #1
	b number_frac
number_nonzero:
	cmp w0, #'1'
	b.lo number_fail
	cmp w0, #'9'
	b.hi number_fail
	add x21, x21, #1
number_digits:
	cmp x21, x20
	b.hs number_frac
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo number_frac
	cmp w0, #'9'
	b.hi number_frac
	add x21, x21, #1
	b number_digits
number_frac:
	cmp x21, x20
	b.hs number_done
	ldrb w0, [x19, x21]
	cmp w0, #'.'
	b.ne number_exp
	add x21, x21, #1
	cmp x21, x20
	b.hs number_fail
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo number_fail
	cmp w0, #'9'
	b.hi number_fail
number_frac_digits:
	add x21, x21, #1
	cmp x21, x20
	b.hs number_done
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo number_done
	cmp w0, #'9'
	b.hi number_done
	b number_frac_digits
number_exp:
	cmp w0, #'e'
	b.eq number_exp_start
	cmp w0, #'E'
	b.ne number_done
number_exp_start:
	add x21, x21, #1
	cmp x21, x20
	b.hs number_fail
	ldrb w0, [x19, x21]
	cmp w0, #'+'
	b.eq number_exp_sign
	cmp w0, #'-'
	b.ne number_exp_digit
number_exp_sign:
	add x21, x21, #1
	cmp x21, x20
	b.hs number_fail
	ldrb w0, [x19, x21]
number_exp_digit:
	cmp w0, #'0'
	b.lo number_fail
	cmp w0, #'9'
	b.hi number_fail
number_exp_digits:
	add x21, x21, #1
	cmp x21, x20
	b.hs number_done
	ldrb w0, [x19, x21]
	cmp w0, #'0'
	b.lo number_done
	cmp w0, #'9'
	b.hi number_done
	b number_exp_digits
number_done:
	mov x0, #1
	b number_return
number_fail:
	mov x0, #0
number_return:
	ldp x29, x30, [sp], #16
	ret

parse_array:
	stp x29, x30, [sp, #-16]!
	mov x29, sp
	add x21, x21, #1
	bl parse_ws
	cmp x21, x20
	b.hs array_fail
	ldrb w0, [x19, x21]
	cmp w0, #']'
	b.eq array_empty
	bl parse_value
	cbz x0, array_return
array_loop:
	bl parse_ws
	cmp x21, x20
	b.hs array_fail
	ldrb w0, [x19, x21]
	cmp w0, #']'
	b.eq array_done
	cmp w0, #','
	b.ne array_fail
	add x21, x21, #1
	bl parse_ws
	bl parse_value
	cbz x0, array_return
	b array_loop
array_empty:
	add x21, x21, #1
array_done:
	add x21, x21, #1
	mov x0, #1
	b array_return
array_fail:
	mov x0, #0
array_return:
	ldp x29, x30, [sp], #16
	ret

parse_object:
	stp x29, x30, [sp, #-16]!
	mov x29, sp
	add x21, x21, #1
	bl parse_ws
	cmp x21, x20
	b.hs object_fail
	ldrb w0, [x19, x21]
	cmp w0, #'}'
	b.eq object_empty
object_member:
	bl parse_string
	cbz x0, object_return
	bl parse_ws
	cmp x21, x20
	b.hs object_fail
	ldrb w0, [x19, x21]
	cmp w0, #':'
	b.ne object_fail
	add x21, x21, #1
	bl parse_ws
	bl parse_value
	cbz x0, object_return
object_loop:
	bl parse_ws
	cmp x21, x20
	b.hs object_fail
	ldrb w0, [x19, x21]
	cmp w0, #'}'
	b.eq object_done
	cmp w0, #','
	b.ne object_fail
	add x21, x21, #1
	bl parse_ws
	b object_member
object_empty:
	add x21, x21, #1
object_done:
	add x21, x21, #1
	mov x0, #1
	b object_return
object_fail:
	mov x0, #0
object_return:
	ldp x29, x30, [sp], #16
	ret
