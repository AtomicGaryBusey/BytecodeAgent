.section .text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x24, line_starts
    add x24, x24, :lo12:line_starts
    adrp x25, line_lengths
    add x25, x25, :lo12:line_lengths

    mov x20, #0

read_more:
    mov x0, #0
    add x1, x19, x20
    mov x2, #32768
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    beq parse_input
    add x20, x20, x0
    b read_more

parse_input:
    mov x21, #0
    mov x22, #0
    mov x23, #0

parse_loop:
    cmp x22, x20
    b.hs parse_done
    ldrb w26, [x19, x22]
    add x22, x22, #1
    cmp w26, #10
    b.ne parse_loop

    sub x27, x22, x21
    sub x27, x27, #1
    str x21, [x24, x23, lsl #3]
    str x27, [x25, x23, lsl #3]
    add x23, x23, #1
    mov x21, x22
    b parse_loop

parse_done:
    cmp x22, x21
    beq sort_start
    sub x27, x22, x21
    str x21, [x24, x23, lsl #3]
    str x27, [x25, x23, lsl #3]
    add x23, x23, #1

sort_start:
    mov x26, x19
    mov x22, #0

sort_outer:
    cmp x22, x23
    b.hs output_start
    mov x24, x22
    add x25, x22, #1

sort_inner:
    cmp x25, x23
    b.hs sort_swap

    ldr x5, [x19, x25, lsl #3]
    ldr x6, [x19, x24, lsl #3]
    ldr x7, [x20, x25, lsl #3]
    ldr x8, [x20, x24, lsl #3]
    add x5, x26, x5
    add x6, x26, x6
    mov x9, #0

compare_loop:
    cmp x9, x7
    b.hs compare_lengths
    cmp x9, x8
    b.hs compare_lengths
    ldrb w10, [x5, x9]
    ldrb w11, [x6, x9]
    cmp w10, w11
    b.lo choose_j
    b.hi next_j
    add x9, x9, #1
    b compare_loop

compare_lengths:
    cmp x7, x8
    b.lo choose_j
    b next_j

choose_j:
    mov x24, x25

next_j:
    add x25, x25, #1
    b sort_inner

sort_swap:
    cmp x24, x22
    b.eq next_outer

    ldr x5, [x19, x22, lsl #3]
    ldr x6, [x19, x24, lsl #3]
    str x6, [x19, x22, lsl #3]
    str x5, [x19, x24, lsl #3]

    ldr x5, [x20, x22, lsl #3]
    ldr x6, [x20, x24, lsl #3]
    str x6, [x20, x22, lsl #3]
    str x5, [x20, x24, lsl #3]

next_outer:
    add x22, x22, #1
    b sort_outer

output_start:
    mov x22, #0

output_loop:
    cmp x22, x23
    b.hs normal_exit

    ldr x5, [x19, x22, lsl #3]
    ldr x10, [x20, x22, lsl #3]
    add x9, x26, x5

write_line:
    cbz x10, write_newline
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    cbz x0, error_exit
    add x9, x9, x0
    sub x10, x10, x0
    b write_line

write_newline:
    adr x9, newline

write_nl:
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    cbz x0, error_exit
    cmp x0, #1
    b.lo error_exit
    add x22, x22, #1
    b output_loop

normal_exit:
    mov x0, #0
    mov x8, #93
    svc #0

error_exit:
    mov x0, #1
    mov x8, #93
    svc #0

newline:
    .byte 10

.section .bss
.align 4
input_buf:
    .zero 32768
.align 3
line_starts:
    .zero 8000
.align 3
line_lengths:
    .zero 8000
