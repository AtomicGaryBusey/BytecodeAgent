.section .rodata
newline:
    .byte 10

.section .text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x21, line_starts
    add x21, x21, :lo12:line_starts
    adrp x22, line_lengths
    add x22, x22, :lo12:line_lengths

    mov x20, #0

read_more:
    cmp x20, #32768
    b.eq parse_input
    mov x0, #0
    add x1, x19, x20
    mov x2, #32768
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, parse_input
    add x20, x20, x0
    b read_more

parse_input:
    mov x23, #0
    mov x24, #0
    mov x25, #0

parse_loop:
    cmp x24, x20
    b.hs parse_done
    ldrb w26, [x19, x24]
    add x24, x24, #1
    cmp w26, #10
    b.ne parse_loop

    sub x27, x24, x25
    sub x27, x27, #1
    str x25, [x21, x23, lsl #3]
    str x27, [x22, x23, lsl #3]
    add x23, x23, #1
    mov x25, x24
    b parse_loop

parse_done:
    cmp x24, x25
    b.eq sort_start
    sub x27, x24, x25
    str x25, [x21, x23, lsl #3]
    str x27, [x22, x23, lsl #3]
    add x23, x23, #1

sort_start:
    mov x24, #0

sort_outer:
    cmp x24, x23
    b.hs output_start
    mov x25, x24
    add x26, x24, #1

sort_inner:
    cmp x26, x23
    b.hs sort_swap

    ldr x5, [x21, x26, lsl #3]
    ldr x6, [x21, x25, lsl #3]
    ldr x7, [x22, x26, lsl #3]
    ldr x8, [x22, x25, lsl #3]
    add x5, x19, x5
    add x6, x19, x6
    mov x9, #0

compare_loop:
    cmp x9, x7
    b.hs compare_lengths
    cmp x9, x8
    b.hs compare_lengths
    ldrb w10, [x5, x9]
    ldrb w11, [x6, x9]
    cmp w10, w11
    b.lo choose_candidate
    b.hi next_candidate
    add x9, x9, #1
    b compare_loop

compare_lengths:
    cmp x7, x8
    b.lo choose_candidate
    b next_candidate

choose_candidate:
    mov x25, x26

next_candidate:
    add x26, x26, #1
    b sort_inner

sort_swap:
    cmp x25, x24
    b.eq next_outer

    ldr x5, [x21, x24, lsl #3]
    ldr x6, [x21, x25, lsl #3]
    str x6, [x21, x24, lsl #3]
    str x5, [x21, x25, lsl #3]

    ldr x5, [x22, x24, lsl #3]
    ldr x6, [x22, x25, lsl #3]
    str x6, [x22, x24, lsl #3]
    str x5, [x22, x25, lsl #3]

next_outer:
    add x24, x24, #1
    b sort_outer

output_start:
    mov x24, #0

output_loop:
    cmp x24, x23
    b.hs normal_exit

    ldr x5, [x21, x24, lsl #3]
    ldr x10, [x22, x24, lsl #3]
    add x9, x19, x5

write_line:
    cbz x10, write_newline
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    cbz x0, write_line
    add x9, x9, x0
    sub x10, x10, x0
    b write_line

write_newline:
    adrp x9, newline
    add x9, x9, :lo12:newline

write_nl:
    mov x0, #1
    mov x1, x9
    mov x2, #1
    mov x8, #64
    svc #0
    cbz x0, write_nl
    add x24, x24, #1
    b output_loop

normal_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.align 3
input_buf:
    .zero 32768

line_starts:
    .zero 8000

line_lengths:
    .zero 8000
