.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60
.equ MAX_INPUT, 65536

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_msg_len = . - ok_msg
err_msg:
    .ascii "error at "
err_msg_len = . - err_msg
true_msg:
    .ascii "true"
false_msg:
    .ascii "false"
null_msg:
    .ascii "null"

.bss
.align 8
buffer:
    .zero MAX_INPUT
input_len:
    .zero 8
input_pos:
    .zero 8
error_pos:
    .zero 8
outbuf:
    .zero 128

.text
.globl _start
_start:
    xor eax, eax
    mov qword ptr [rip + input_len], rax

read_loop:
    mov rax, MAX_INPUT
    sub rax, qword ptr [rip + input_len]
    jz read_done
    mov rdx, rax
    mov rax, SYS_READ
    xor edi, edi
    mov rsi, qword ptr [rip + input_len]
    lea rsi, [rip + buffer + rsi]
    syscall
    test rax, rax
    jz read_done
    add qword ptr [rip + input_len], rax
    jmp read_loop

read_done:
    mov qword ptr [rip + input_pos], 0
    call skip_ws
    call parse_value
    test eax, eax
    jz print_error
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jne syntax_failure
    lea rsi, [rip + ok_msg]
    mov edx, ok_msg_len
    call write_all
    mov eax, SYS_EXIT
    xor edi, edi
    syscall

syntax_failure:
    call fail
    jmp print_error

print_error:
    lea rsi, [rip + err_msg]
    mov edx, err_msg_len
    call write_all

    mov rax, qword ptr [rip + error_pos]
    lea rdi, [rip + outbuf + 127]
    xor ecx, ecx
    mov ebx, 10
    test rax, rax
    jnz convert_digits
    dec rdi
    mov byte ptr [rdi], '0'
    mov ecx, 1
    jmp digits_ready

convert_digits:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc ecx
    test rax, rax
    jnz convert_digits

digits_ready:
    mov rsi, rdi
    mov edx, ecx
    call write_all
    mov byte ptr [rip + outbuf], 10
    lea rsi, [rip + outbuf]
    mov edx, 1
    call write_all
    mov eax, SYS_EXIT
    mov edi, 1
    syscall

write_all:
    push r12
    push r13
    mov r12, rsi
    mov r13, rdx
write_loop:
    test r13, r13
    jz write_done
    mov eax, SYS_WRITE
    mov edi, 1
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jle write_done
    add r12, rax
    sub r13, rax
    jmp write_loop
write_done:
    pop r13
    pop r12
    ret

fail:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jbe fail_store
    mov rax, qword ptr [rip + input_len]
fail_store:
    mov qword ptr [rip + error_pos], rax
    xor eax, eax
    ret

skip_ws:
skip_ws_loop:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae skip_ws_done
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, 32
    je skip_ws_one
    cmp dl, 9
    je skip_ws_one
    cmp dl, 10
    je skip_ws_one
    cmp dl, 13
    jne skip_ws_done
skip_ws_one:
    inc rax
    mov qword ptr [rip + input_pos], rax
    jmp skip_ws_loop
skip_ws_done:
    ret

parse_value:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae parse_value_fail
    movzx eax, byte ptr [rip + buffer + rax]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_value_word
    cmp al, '9'
    jbe parse_number
parse_value_word:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
parse_value_fail:
    call fail
    ret

parse_true:
    lea rsi, [rip + true_msg]
    mov edx, 4
    jmp parse_literal

parse_false:
    lea rsi, [rip + false_msg]
    mov edx, 5
    jmp parse_literal

parse_null:
    lea rsi, [rip + null_msg]
    mov edx, 4

parse_literal:
    xor ecx, ecx
literal_loop:
    cmp ecx, edx
    jae literal_done
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae literal_fail
    movzx eax, byte ptr [rip + buffer + rax]
    cmp al, byte ptr [rsi + rcx]
    jne literal_fail
    inc qword ptr [rip + input_pos]
    inc ecx
    jmp literal_loop
literal_done:
    mov eax, 1
    ret
literal_fail:
    call fail
    ret

parse_string:
    inc qword ptr [rip + input_pos]
string_loop:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae string_fail
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '"'
    je string_done
    cmp dl, '\'
    je string_escape
    cmp dl, 32
    jb string_fail
    inc rax
    mov qword ptr [rip + input_pos], rax
    jmp string_loop

string_escape:
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae string_fail
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '"'
    je string_escape_one
    cmp dl, '\'
    je string_escape_one
    cmp dl, '/'
    je string_escape_one
    cmp dl, 'b'
    je string_escape_one
    cmp dl, 'f'
    je string_escape_one
    cmp dl, 'n'
    je string_escape_one
    cmp dl, 'r'
    je string_escape_one
    cmp dl, 't'
    je string_escape_one
    cmp dl, 'u'
    jne string_fail
    inc rax
    mov qword ptr [rip + input_pos], rax
    mov ecx, 4
unicode_loop:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae string_fail
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb unicode_alpha
    cmp dl, '9'
    jbe unicode_one
unicode_alpha:
    cmp dl, 'A'
    jb unicode_lower
    cmp dl, 'F'
    jbe unicode_one
unicode_lower:
    cmp dl, 'a'
    jb string_fail
    cmp dl, 'f'
    ja string_fail
unicode_one:
    inc rax
    mov qword ptr [rip + input_pos], rax
    dec ecx
    jnz unicode_loop
    jmp string_loop

string_escape_one:
    inc rax
    mov qword ptr [rip + input_pos], rax
    jmp string_loop

string_done:
    inc rax
    mov qword ptr [rip + input_pos], rax
    mov eax, 1
    ret
string_fail:
    call fail
    ret

parse_number:
    mov rax, qword ptr [rip + input_pos]
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '-'
    jne number_digits
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb number_fail
    cmp dl, '9'
    ja number_fail

number_digits:
    mov rax, qword ptr [rip + input_pos]
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jne number_nonzero
    inc rax
    mov qword ptr [rip + input_pos], rax
    jmp number_fraction

number_nonzero:
    cmp dl, '1'
    jb number_fail
    cmp dl, '9'
    ja number_fail
number_digit_loop:
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_fraction
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb number_fraction
    cmp dl, '9'
    jbe number_digit_loop

number_fraction:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '.'
    jne number_exponent
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb number_fail
    cmp dl, '9'
    ja number_fail
number_frac_loop:
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb number_exponent
    cmp dl, '9'
    jbe number_frac_loop

number_exponent:
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, 'e'
    je exponent_start
    cmp dl, 'E'
    jne number_done
exponent_start:
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '+'
    je exponent_sign
    cmp dl, '-'
    jne exponent_digit_check
exponent_sign:
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_fail
exponent_digit_check:
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb number_fail
    cmp dl, '9'
    ja number_fail
exponent_loop:
    inc rax
    mov qword ptr [rip + input_pos], rax
    cmp rax, qword ptr [rip + input_len]
    jae number_done
    movzx edx, byte ptr [rip + buffer + rax]
    cmp dl, '0'
    jb number_done
    cmp dl, '9'
    jbe exponent_loop
number_done:
    mov eax, 1
    ret
number_fail:
    call fail
    ret

parse_object:
    inc qword ptr [rip + input_pos]
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    cmp byte ptr [rip + buffer + rax], '}'
    je object_close
object_member:
    call parse_string
    test eax, eax
    jz object_return
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    cmp byte ptr [rip + buffer + rax], ':'
    jne object_fail
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    call parse_value
    test eax, eax
    jz object_return
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae object_fail
    mov dl, byte ptr [rip + buffer + rax]
    cmp dl, '}'
    je object_close
    cmp dl, ','
    jne object_fail
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    jmp object_member
object_close:
    inc qword ptr [rip + input_pos]
    mov eax, 1
    ret
object_fail:
    call fail
object_return:
    ret

parse_array:
    inc qword ptr [rip + input_pos]
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae array_fail
    cmp byte ptr [rip + buffer + rax], ']'
    je array_close
array_value:
    call parse_value
    test eax, eax
    jz array_return
    call skip_ws
    mov rax, qword ptr [rip + input_pos]
    cmp rax, qword ptr [rip + input_len]
    jae array_fail
    mov dl, byte ptr [rip + buffer + rax]
    cmp dl, ']'
    je array_close
    cmp dl, ','
    jne array_fail
    inc rax
    mov qword ptr [rip + input_pos], rax
    call skip_ws
    jmp array_value
array_close:
    inc qword ptr [rip + input_pos]
    mov eax, 1
    ret
array_fail:
    call fail
array_return:
    ret
