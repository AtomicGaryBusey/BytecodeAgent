.equ INPUT_SIZE, 65536

.section .text
.globl _start

_start:
    lea rbx, [rip + buffer]
    xor r12d, r12d

read_loop:
    cmp r12, INPUT_SIZE
    je input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov rdx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    xor r13d, r13d

scan:
    mov r15, r13

scan_loop:
    cmp r15, r12
    je final_line
    cmp byte ptr [rbx + r15], 10
    je line_found
    inc r15
    jmp scan_loop

line_found:
    mov rdx, r15
    sub rdx, r13
    jz line_newline

    lea r8, [rbx + r13]
    lea r9, [rbx + r15 - 1]

reverse_line:
    cmp r8, r9
    jae reversed
    mov al, byte ptr [r8]
    mov r11b, byte ptr [r9]
    mov byte ptr [r8], r11b
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_line

reversed:
    lea rsi, [rbx + r13]
    call write_all

line_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    lea r13, [r15 + 1]
    cmp r13, r12
    jne scan
    jmp finish

final_line:
    cmp r13, r12
    je finish

    mov rdx, r12
    sub rdx, r13
    lea r8, [rbx + r13]
    lea r9, [rbx + r12 - 1]

reverse_final:
    cmp r8, r9
    jae final_reversed
    mov al, byte ptr [r8]
    mov r11b, byte ptr [r9]
    mov byte ptr [r8], r11b
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_final

final_reversed:
    lea rsi, [rbx + r13]
    call write_all
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all

finish:
    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_done
    mov r8, rdx
    mov r9, rsi

write_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    syscall
    add r9, rax
    sub r8, rax
    jnz write_loop

write_done:
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.align 16
buffer:
    .zero INPUT_SIZE
