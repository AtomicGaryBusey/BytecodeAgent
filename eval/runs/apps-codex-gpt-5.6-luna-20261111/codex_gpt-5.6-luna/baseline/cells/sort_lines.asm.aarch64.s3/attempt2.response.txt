.text
.globl _start

_start:
    adrp x25, input_buf
    add  x25, x25, :lo12:input_buf
    adrp x24, records
    add  x24, x24, :lo12:records

    mov  x27, #0
    mov  x28, #32768

read_loop:
    cmp  x27, x28
    b.hs parse_input
    mov  x0, #0
    add  x1, x25, x27
    sub  x2, x28, x27
    mov  x8, #63
    svc  #0
    cbz  x0, parse_input
    add  x27, x27, x0
    b    read_loop

parse_input:
    mov  x19, #0
    mov  x20, #0
    mov  x21, #0

parse_loop:
    cmp  x19, x27
    b.hs parse_done
    ldrb w0, [x25, x19]
    cmp  w0, #10
    b.ne parse_next
    add  x1, x24, x21, lsl #4
    str  x20, [x1]
    sub  x2, x19, x20
    str  x2, [x1, #8]
    add  x21, x21, #1
    add  x20, x19, #1

parse_next:
    add  x19, x19, #1
    b    parse_loop

parse_done:
    cmp  x19, x20
    b.eq sort_start
    add  x1, x24, x21, lsl #4
    str  x20, [x1]
    sub  x2, x19, x20
    str  x2, [x1, #8]
    add  x21, x21, #1

sort_start:
    mov  x19, #0

sort_outer:
    cmp  x19, x21
    b.hs output_start
    mov  x22, x19
    add  x23, x19, #1

sort_inner:
    cmp  x23, x21
    b.hs sort_swap

    add  x9, x24, x23, lsl #4
    ldr  x1, [x9]
    ldr  x2, [x9, #8]

    add  x10, x24, x22, lsl #4
    ldr  x3, [x10]
    ldr  x4, [x10, #8]

    cmp  x2, x4
    csel x5, x2, x4, lo
    mov  x6, #0

compare_loop:
    cmp  x6, x5
    b.hs compare_prefix
    ldrb w7, [x25, x1]
    ldrb w8, [x25, x3]
    add  x1, x1, #1
    add  x3, x3, #1
    add  x6, x6, #1
    cmp  w7, w8
    b.eq compare_loop
    b.lo candidate_less
    b    compare_done

compare_prefix:
    cmp  x2, x4
    b.lo candidate_less
    b    compare_done

candidate_less:
    mov  x22, x23

compare_done:
    add  x23, x23, #1
    b    sort_inner

sort_swap:
    cmp  x22, x19
    b.eq sort_next
    add  x0, x24, x19, lsl #4
    add  x1, x24, x22, lsl #4
    ldp  x2, x3, [x0]
    ldp  x4, x5, [x1]
    stp  x4, x5, [x0]
    stp  x2, x3, [x1]

sort_next:
    add  x19, x19, #1
    b    sort_outer

output_start:
    mov  x19, #0
    adrp x26, newline
    add  x26, x26, :lo12:newline

output_loop:
    cmp  x19, x21
    b.hs program_exit
    add  x0, x24, x19, lsl #4
    ldr  x1, [x0]
    ldr  x2, [x0, #8]
    add  x0, x25, x1
    mov  x1, x2
    bl   write_all
    mov  x0, x26
    mov  x1, #1
    bl   write_all
    add  x19, x19, #1
    b    output_loop

program_exit:
    mov  x0, #0
    mov  x8, #93
    svc  #0

write_all:
    cbz  x1, write_done
    mov  x3, x0
    mov  x4, x1

write_loop:
    mov  x0, #1
    mov  x1, x3
    mov  x2, x4
    mov  x8, #64
    svc  #0
    add  x3, x3, x0
    sub  x4, x4, x0
    cbnz x4, write_loop

write_done:
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 4
input_buf:
    .zero 32768

.align 4
records:
    .zero 16000
