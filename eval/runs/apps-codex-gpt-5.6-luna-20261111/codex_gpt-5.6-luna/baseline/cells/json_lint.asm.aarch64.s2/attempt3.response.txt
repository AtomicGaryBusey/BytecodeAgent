.equ BUF_SIZE, 65536

.section .rodata
ok_text:
	.ascii "ok\n"
ok_len = . - ok_text
err_text:
	.ascii "error at "
err_len = . - err_text
true_text:
	.ascii "true"
false_text:
	.ascii "false"
null_text:
	.ascii "null"

.section .bss
	.align 4
input_buf:
	.zero BUF_SIZE
output_buf:
	.zero 64

.text
.globl _start
_start:
	adrp x20, input_buf
	add x20, x20, :lo12:input_buf
	mov x21, #0

read_loop:
	mov x0, #0
	add x1, x20, x21
	mov x2, #BUF_SIZE
	sub x2, x2, x21
	mov x8, #63
	svc #0
	cmp x0, #0
	beq input_done
	add x21, x21, x0
	b read_loop

input_done:
	mov x22, #0
	bl skip_ws
	bl parse_value
	cmp x0, #0
	beq error
	bl skip_ws
	cmp x22, x21
	bne error

	adrp x1, ok_text
	add x1, x1, :lo12:ok_text
	mov x2, #ok_len
	bl write_all
	mov x0, #0
	mov x8, #93
	svc #0

error:
	adrp x1, err_text
	add x1, x1, :lo12:err_text
	adrp x0, output_buf
	add x0, x0, :lo12:output_buf
	mov x2, #err_len
	mov x3, #0

copy_error:
	cmp x3, x2
	bge error_digits
	ldrb w4, [x1, x3]
	strb w4, [x0, x3]
	add x3, x3, #1
	b copy_error

error_digits:
	mov x4, x22
	cmp x4, #0
	bne make_digits
	mov w5, #'0'
	strb w5, [x0, x3]
	add x3, x3, #1
	b finish_digits

make_digits:
	mov x6, #10
	mov x7, x3

digit_loop:
	udiv x5, x4, x6
	msub x8, x5, x6, x4
	add w8, w8, #'0'
	strb w8, [x0, x3]
	add x3, x3, #1
	mov x4, x5
	cmp x4, #0
	bne digit_loop

	mov x4, x7
	sub x5, x3, #1

reverse_loop:
	cmp x4, x5
	bge finish_digits
	ldrb w6, [x0, x4]
	ldrb w7, [x0, x5]
	strb w7, [x0, x4]
	strb w6, [x0, x5]
	add x4, x4, #1
	sub x5, x5, #1
	b reverse_loop

finish_digits:
	mov w4, #10
	strb w4, [x0, x3]
	add x3, x3, #1
	adrp x1, output_buf
	add x1, x1, :lo12:output_buf
	mov x2, x3
	bl write_all
	mov x0, #1
	mov x8, #93
	svc #0

write_all:
	mov x3, x1

write_loop:
	cmp x2, #0
	beq write_done
	mov x0, #1
	mov x1, x3
	mov x8, #64
	svc #0
	add x3, x3, x0
	sub x2, x2, x0
	b write_loop

write_done:
	ret

skip_ws:
	cmp x22, x21
	bge skip_done
	ldrb w0, [x20, x22]
	cmp w0, #0x20
	beq skip_one
	cmp w0, #9
	beq skip_one
	cmp w0, #10
	beq skip_one
	cmp w0, #13
	bne skip_done

skip_one:
	add x22, x22, #1
	b skip_ws

skip_done:
	ret

parse_value:
	cmp x22, x21
	bge value_bad
	ldrb w0, [x20, x22]
	cmp w0, #'{'
	beq parse_object
	cmp w0, #'['
	beq parse_array
	cmp w0, #'"'
	beq parse_string
	cmp w0, #'-'
	beq parse_number
	cmp w0, #'0'
	blo value_literal
	cmp w0, #'9'
	bls parse_number

value_literal:
	cmp w0, #'t'
	beq parse_true
	cmp w0, #'f'
	beq parse_false
	cmp w0, #'n'
	beq parse_null

value_bad:
	mov x0, #0
	ret

parse_true:
	adrp x4, true_text
	add x4, x4, :lo12:true_text
	mov x3, #4
	b match_literal

parse_false:
	adrp x4, false_text
	add x4, x4, :lo12:false_text
	mov x3, #5
	b match_literal

parse_null:
	adrp x4, null_text
	add x4, x4, :lo12:null_text
	mov x3, #4

match_literal:
	mov x5, #0

match_loop:
	cmp x5, x3
	bge match_ok
	cmp x22, x21
	bge match_bad
	ldrb w6, [x20, x22]
	ldrb w7, [x4, x5]
	cmp w6, w7
	bne match_bad
	add x22, x22, #1
	add x5, x5, #1
	b match_loop

match_ok:
	mov x0, #1
	ret

match_bad:
	mov x0, #0
	ret

parse_string:
	add x22, x22, #1

string_loop:
	cmp x22, x21
	bge string_bad
	ldrb w0, [x20, x22]
	cmp w0, #'"'
	beq string_end
	cmp w0, #'\\'
	beq string_escape
	cmp w0, #0x20
	blo string_bad
	add x22, x22, #1
	b string_loop

string_escape:
	add x22, x22, #1
	cmp x22, x21
	bge string_bad
	ldrb w0, [x20, x22]
	cmp w0, #'"'
	beq simple_escape
	cmp w0, #'\\'
	beq simple_escape
	cmp w0, #'/'
	beq simple_escape
	cmp w0, #'b'
	beq simple_escape
	cmp w0, #'f'
	beq simple_escape
	cmp w0, #'n'
	beq simple_escape
	cmp w0, #'r'
	beq simple_escape
	cmp w0, #'t'
	beq simple_escape
	cmp w0, #'u'
	bne string_bad
	add x22, x22, #1
	mov x3, #4

unicode_loop:
	cmp x3, #0
	beq string_loop
	cmp x22, x21
	bge string_bad
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo unicode_alpha
	cmp w0, #'9'
	bls unicode_ok
unicode_alpha:
	cmp w0, #'a'
	blo unicode_upper
	cmp w0, #'f'
	bls unicode_ok
unicode_upper:
	cmp w0, #'A'
	blo string_bad
	cmp w0, #'F'
	bhi string_bad
unicode_ok:
	add x22, x22, #1
	sub x3, x3, #1
	b unicode_loop

simple_escape:
	add x22, x22, #1
	b string_loop

string_end:
	add x22, x22, #1
	mov x0, #1
	ret

string_bad:
	mov x0, #0
	ret

parse_number:
	ldrb w0, [x20, x22]
	cmp w0, #'-'
	bne number_int
	add x22, x22, #1
	cmp x22, x21
	bge number_bad

number_int:
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	beq number_zero
	cmp w0, #'1'
	blo number_bad
	cmp w0, #'9'
	bhi number_bad

number_digits:
	add x22, x22, #1
	cmp x22, x21
	bge number_fraction
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_fraction
	cmp w0, #'9'
	bls number_digits
	b number_fraction

number_zero:
	add x22, x22, #1

number_fraction:
	cmp x22, x21
	bge number_ok
	ldrb w0, [x20, x22]
	cmp w0, #'.'
	bne number_exponent
	add x22, x22, #1
	cmp x22, x21
	bge number_bad
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_bad
	cmp w0, #'9'
	bhi number_bad

fraction_digits:
	add x22, x22, #1
	cmp x22, x21
	bge number_exponent
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_exponent
	cmp w0, #'9'
	bls fraction_digits

number_exponent:
	cmp x22, x21
	bge number_ok
	ldrb w0, [x20, x22]
	cmp w0, #'e'
	beq exponent_start
	cmp w0, #'E'
	bne number_ok

exponent_start:
	add x22, x22, #1
	cmp x22, x21
	bge number_bad
	ldrb w0, [x20, x22]
	cmp w0, #'+'
	beq exponent_sign
	cmp w0, #'-'
	bne exponent_check

exponent_sign:
	add x22, x22, #1
	cmp x22, x21
	bge number_bad

exponent_check:
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_bad
	cmp w0, #'9'
	bhi number_bad

exponent_digits:
	add x22, x22, #1
	cmp x22, x21
	bge number_ok
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_ok
	cmp w0, #'9'
	bls exponent_digits

number_ok:
	mov x0, #1
	ret

number_bad:
	mov x0, #0
	ret

parse_array:
	stp x29, x30, [sp, #-16]!
	add x22, x22, #1
	bl skip_ws
	cmp x22, x21
	bge array_bad
	ldrb w0, [x20, x22]
	cmp w0, #']'
	beq array_empty

array_item:
	bl parse_value
	cmp x0, #0
	beq array_bad
	bl skip_ws
	cmp x22, x21
	bge array_bad
	ldrb w0, [x20, x22]
	cmp w0, #']'
	beq array_done
	cmp w0, #','
	bne array_bad
	add x22, x22, #1
	bl skip_ws
	b array_item

array_empty:
	add x22, x22, #1

array_done:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret

array_bad:
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret

parse_object:
	stp x29, x30, [sp, #-16]!
	add x22, x22, #1
	bl skip_ws
	cmp x22, x21
	bge object_bad
	ldrb w0, [x20, x22]
	cmp w0, #'}'
	beq object_empty

object_member:
	bl parse_string
	cmp x0, #0
	beq object_bad
	bl skip_ws
	cmp x22, x21
	bge object_bad
	ldrb w0, [x20, x22]
	cmp w0, #':'
	bne object_bad
	add x22, x22, #1
	bl skip_ws
	bl parse_value
	cmp x0, #0
	beq object_bad
	bl skip_ws
	cmp x22, x21
	bge object_bad
	ldrb w0, [x20, x22]
	cmp w0, #'}'
	beq object_done
	cmp w0, #','
	bne object_bad
	add x22, x22, #1
	bl skip_ws
	b object_member

object_empty:
	add x22, x22, #1

object_done:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret

object_bad:
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret
