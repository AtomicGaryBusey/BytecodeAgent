.equ BUF_SIZE, 65536

.section .rodata
true_text:
	.ascii "true"
false_text:
	.ascii "false"
null_text:
	.ascii "null"
ok_text:
	.ascii "ok\n"
ok_len = . - ok_text
err_prefix:
	.ascii "error at "
err_prefix_len = . - err_prefix

.section .bss
	.align 4
input_buf:
	.zero BUF_SIZE
output_buf:
	.zero 64

.text
.globl _start
_start:
	adrp x20, input_buf
	add x20, x20, :lo12:input_buf
	mov x21, #0

read_more:
	mov x0, #0
	add x1, x20, x21
	mov x2, #BUF_SIZE
	sub x2, x2, x21
	mov x8, #63
	svc #0
	cmp x0, #0
	beq input_done
	add x21, x21, x0
	b read_more

input_done:
	mov x22, #0
	bl skip_ws
	bl parse_value
	cmp x0, #0
	beq report_error
	bl skip_ws
	cmp x22, x21
	bne report_error

	adrp x1, ok_text
	add x1, x1, :lo12:ok_text
	mov x2, #ok_len
	bl write_all
	mov x0, #0
	mov x8, #93
	svc #0

report_error:
	adrp x1, err_prefix
	add x1, x1, :lo12:err_prefix
	adrp x0, output_buf
	add x0, x0, :lo12:output_buf
	mov x2, #err_prefix_len
	mov x3, #0

copy_prefix:
	cmp x3, x2
	bge prefix_done
	ldrb w4, [x1, x3]
	strb w4, [x0, x3]
	add x3, x3, #1
	b copy_prefix

prefix_done:
	mov x3, #err_prefix_len
	mov x4, x22
	cmp x4, #0
	bne convert_digits
	mov w5, #'0'
	strb w5, [x0, x3]
	add x3, x3, #1
	b digits_done

convert_digits:
	mov x6, #10
	mov x7, x3
digit_loop:
	udiv x5, x4, x6
	msub x8, x5, x6, x4
	add w8, w8, #'0'
	strb w8, [x0, x3]
	add x3, x3, #1
	mov x4, x5
	cmp x4, #0
	bne digit_loop

	mov x4, x7
	sub x5, x3, #1
reverse_digits:
	cmp x4, x5
	bge digits_done
	ldrb w6, [x0, x4]
	ldrb w7, [x0, x5]
	strb w7, [x0, x4]
	strb w6, [x0, x5]
	add x4, x4, #1
	sub x5, x5, #1
	b reverse_digits

digits_done:
	mov w4, #10
	strb w4, [x0, x3]
	add x3, x3, #1
	adrp x1, output_buf
	add x1, x1, :lo12:output_buf
	mov x2, x3
	bl write_all
	mov x0, #1
	mov x8, #93
	svc #0

write_all:
	mov x3, x0
write_loop:
	cmp x2, #0
	beq write_done
	mov x0, x3
	mov x8, #64
	svc #0
	cmp x0, #0
	beq write_done
	add x1, x1, x0
	sub x2, x2, x0
	b write_loop
write_done:
	ret

skip_ws:
skip_ws_loop:
	cmp x22, x21
	bge skip_ws_done
	ldrb w0, [x20, x22]
	cmp w0, #0x20
	beq skip_one
	cmp w0, #9
	beq skip_one
	cmp w0, #10
	beq skip_one
	cmp w0, #13
	bne skip_ws_done
skip_one:
	add x22, x22, #1
	b skip_ws_loop
skip_ws_done:
	ret

parse_value:
	cmp x22, x21
	bge value_fail
	ldrb w0, [x20, x22]
	cmp w0, #'{'
	beq parse_object
	cmp w0, #'['
	beq parse_array
	cmp w0, #'"'
	beq parse_string
	cmp w0, #'-'
	beq parse_number
	cmp w0, #'0'
	blo value_check_literals
	cmp w0, #'9'
	bls parse_number

value_check_literals:
	cmp w0, #'t'
	beq parse_true
	cmp w0, #'f'
	beq parse_false
	cmp w0, #'n'
	beq parse_null

value_fail:
	mov x0, #0
	ret

parse_true:
	adrp x1, true_text
	add x1, x1, :lo12:true_text
	mov x2, #4
	b parse_literal

parse_false:
	adrp x1, false_text
	add x1, x1, :lo12:false_text
	mov x2, #5
	b parse_literal

parse_null:
	adrp x1, null_text
	add x1, x1, :lo12:null_text
	mov x2, #4
	b parse_literal

parse_literal:
	mov x3, #0
literal_loop:
	cmp x3, x2
	bge literal_success
	cmp x22, x21
	bge literal_fail
	ldrb w4, [x20, x22]
	ldrb w5, [x1, x3]
	cmp w4, w5
	bne literal_fail
	add x22, x22, #1
	add x3, x3, #1
	b literal_loop
literal_success:
	mov x0, #1
	ret
literal_fail:
	mov x0, #0
	ret

parse_string:
	add x22, x22, #1
string_loop:
	cmp x22, x21
	bge string_fail
	ldrb w0, [x20, x22]
	cmp w0, #'"'
	beq string_done
	cmp w0, #'\\'
	beq string_escape
	cmp w0, #0x20
	blo string_fail
	add x22, x22, #1
	b string_loop

string_escape:
	add x22, x22, #1
	cmp x22, x21
	bge string_fail
	ldrb w0, [x20, x22]
	cmp w0, #'"'
	beq simple_escape
	cmp w0, #'\\'
	beq simple_escape
	cmp w0, #'/'
	beq simple_escape
	cmp w0, #'b'
	beq simple_escape
	cmp w0, #'f'
	beq simple_escape
	cmp w0, #'n'
	beq simple_escape
	cmp w0, #'r'
	beq simple_escape
	cmp w0, #'t'
	beq simple_escape
	cmp w0, #'u'
	bne string_fail
	add x22, x22, #1
	mov x3, #0
unicode_loop:
	cmp x3, #4
	bge string_loop
	cmp x22, x21
	bge string_fail
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo unicode_alpha
	cmp w0, #'9'
	bls unicode_digit
unicode_alpha:
	cmp w0, #'a'
	blo unicode_upper
	cmp w0, #'f'
	bls unicode_digit
unicode_upper:
	cmp w0, #'A'
	blo string_fail
	cmp w0, #'F'
	bhi string_fail
unicode_digit:
	add x22, x22, #1
	add x3, x3, #1
	b unicode_loop

simple_escape:
	add x22, x22, #1
	b string_loop

string_done:
	add x22, x22, #1
	mov x0, #1
	ret
string_fail:
	mov x0, #0
	ret

parse_number:
	ldrb w0, [x20, x22]
	cmp w0, #'-'
	bne number_int
	add x22, x22, #1
	cmp x22, x21
	bge number_fail
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_fail
	cmp w0, #'9'
	bhi number_fail

number_int:
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	bne number_nonzero
	add x22, x22, #1
	b number_fraction

number_nonzero:
	cmp w0, #'1'
	blo number_fail
	cmp w0, #'9'
	bhi number_fail
number_digits:
	add x22, x22, #1
	cmp x22, x21
	bge number_fraction
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_fraction
	cmp w0, #'9'
	bls number_digits

number_fraction:
	cmp x22, x21
	bge number_success
	ldrb w0, [x20, x22]
	cmp w0, #'.'
	bne number_exponent
	add x22, x22, #1
	cmp x22, x21
	bge number_fail
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_fail
	cmp w0, #'9'
	bhi number_fail
frac_digits:
	add x22, x22, #1
	cmp x22, x21
	bge number_exponent
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_exponent
	cmp w0, #'9'
	bls frac_digits

number_exponent:
	cmp x22, x21
	bge number_success
	ldrb w0, [x20, x22]
	cmp w0, #'e'
	beq exponent_start
	cmp w0, #'E'
	bne number_success
exponent_start:
	add x22, x22, #1
	cmp x22, x21
	bge number_fail
	ldrb w0, [x20, x22]
	cmp w0, #'+'
	beq exponent_sign
	cmp w0, #'-'
	bne exponent_digit_check
exponent_sign:
	add x22, x22, #1
	cmp x22, x21
	bge number_fail
exponent_digit_check:
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_fail
	cmp w0, #'9'
	bhi number_fail
exponent_digits:
	add x22, x22, #1
	cmp x22, x21
	bge number_success
	ldrb w0, [x20, x22]
	cmp w0, #'0'
	blo number_success
	cmp w0, #'9'
	bls exponent_digits
number_success:
	mov x0, #1
	ret
number_fail:
	mov x0, #0
	ret

parse_array:
	stp x29, x30, [sp, #-16]!
	add x22, x22, #1
	bl skip_ws
	cmp x22, x21
	bge array_fail
	ldrb w0, [x20, x22]
	cmp w0, #']'
	beq array_empty
array_item:
	bl parse_value
	cmp x0, #0
	beq array_fail
	bl skip_ws
	cmp x22, x21
	bge array_fail
	ldrb w0, [x20, x22]
	cmp w0, #']'
	beq array_done
	cmp w0, #','
	bne array_fail
	add x22, x22, #1
	bl skip_ws
	b array_item
array_empty:
	add x22, x22, #1
array_done:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret
array_fail:
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret

parse_object:
	stp x29, x30, [sp, #-16]!
	add x22, x22, #1
	bl skip_ws
	cmp x22, x21
	bge object_fail
	ldrb w0, [x20, x22]
	cmp w0, #'}'
	beq object_empty
object_member:
	bl parse_string
	cmp x0, #0
	beq object_fail
	bl skip_ws
	cmp x22, x21
	bge object_fail
	ldrb w0, [x20, x22]
	cmp w0, #':'
	bne object_fail
	add x22, x22, #1
	bl skip_ws
	bl parse_value
	cmp x0, #0
	beq object_fail
	bl skip_ws
	cmp x22, x21
	bge object_fail
	ldrb w0, [x20, x22]
	cmp w0, #'}'
	beq object_done
	cmp w0, #','
	bne object_fail
	add x22, x22, #1
	bl skip_ws
	b object_member
object_empty:
	add x22, x22, #1
object_done:
	mov x0, #1
	ldp x29, x30, [sp], #16
	ret
object_fail:
	mov x0, #0
	ldp x29, x30, [sp], #16
	ret
