.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x23, buffer
    add x23, x23, :lo12:buffer

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done

    mov x24, x0
    add x21, x21, x24
    mov x27, x23
    add x28, x27, x24

process_loop:
    ldrb w0, [x27], #1

    cmp w0, #10
    b.eq got_newline
    cmp w0, #9
    b.eq got_space
    cmp w0, #11
    b.eq got_space
    cmp w0, #12
    b.eq got_space
    cmp w0, #13
    b.eq got_space
    cmp w0, #32
    b.eq got_space

    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte

got_newline:
    add x19, x19, #1
    mov x22, #0
    b next_byte

got_space:
    mov x22, #0

next_byte:
    cmp x27, x28
    b.ne process_loop
    b read_loop

input_done:
    adrp x26, output
    add x26, x26, :lo12:output
    mov x25, x26

    mov x0, x19
    mov x1, x25
    bl emit_num
    mov x25, x0

    mov w0, #32
    strb w0, [x25], #1

    mov x0, x20
    mov x1, x25
    bl emit_num
    mov x25, x0

    mov w0, #32
    strb w0, [x25], #1

    mov x0, x21
    mov x1, x25
    bl emit_num
    mov x25, x0

    mov w0, #10
    strb w0, [x25], #1

    sub x27, x25, x26
    mov x1, x26

write_loop:
    mov x0, #1
    mov x2, x27
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x27, x27, x0
    cbnz x27, write_loop

    mov x0, #0
    mov x8, #93
    svc #0

emit_num:
    cbnz x0, emit_nonzero
    mov w2, #48
    strb w2, [x1], #1
    mov x0, x1
    ret

emit_nonzero:
    adrp x2, digits
    add x2, x2, :lo12:digits
    add x2, x2, #32
    mov x3, #0
    mov x7, #10

emit_digits:
    udiv x4, x0, x7
    msub x5, x4, x7, x0
    add x5, x5, #48
    strb w5, [x2, #-1]!
    add x3, x3, #1
    mov x0, x4
    cbnz x0, emit_digits

    mov x6, x2

copy_digits:
    ldrb w5, [x6], #1
    strb w5, [x1], #1
    subs x3, x3, #1
    b.ne copy_digits

    mov x0, x1
    ret

.bss
.balign 16
buffer:
    .zero 4096
output:
    .zero 128
digits:
    .zero 32
