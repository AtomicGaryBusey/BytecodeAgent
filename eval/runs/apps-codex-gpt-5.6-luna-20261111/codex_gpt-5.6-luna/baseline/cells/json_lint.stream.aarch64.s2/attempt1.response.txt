.section .rodata

	.ascii "false"

	.zero 65536

	strb w4, [x2, x3]

	add x23, x2, #63
	mov x24, #0
	mov x25, #10

	msub x27, x26, x25, x22
	sub x23, x23, #1
	add w27, w27, #48
	strb w27, [x23]
	add x24, x24, #1
	mov x22, x26
	cbnz x22, digit_loop
	b copy_digits

	mov w27, #48
	strb w27, [x23]
	mov x24, #1

copy_digits:
	add x2, x2, #err_len
	mov x28, #0

	cmp x28, x24
	b.ge finish_error
	ldrb w27, [x23, x28]
	strb w27, [x2, x28]
	add x28, x28, #1
	b copy_digit_loop

	add x2, x2, x24
	mov w27, #10
	strb w27, [x2]
	add x2, x2, #1
	adrp x1, out_buf
	add x1, x1, :lo12:out_buf
	sub x2, x2, x1
	mov x0, #1
	mov x8, #64
	svc #0
	mov x0, #1
	mov x8, #93
	svc #0

skip_ws:

	b.ge skip_ws_done
	ldrb w0, [x19, x21]
	cmp w0, #32
	b.eq skip_ws_one
	cmp w0, #9
	b.eq skip_ws_one
	cmp w0, #10
	b.eq skip_ws_one
	cmp w0, #13
	b.ne skip_ws_done

	bl skip_ws
	cmp x21, x20
