.section .text
.globl _start

_start:
    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf

    mov x19, #0

read_more:
    mov x0, #0
    add x1, x24, x19
    movz x2, #1, lsl #16
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    movz x0, #1, lsl #16
    cmp x19, x0
    b.lo read_more

input_done:
    mov x21, #0
    mov x22, #0
    mov x23, #0

scan_input:
    cmp x22, x19
    b.hs input_end
    ldrb w0, [x24, x22]
    cmp w0, #10
    b.eq found_newline
    add x22, x22, #1
    b scan_input

found_newline:
    mov x27, x22

reverse_line:
    cmp x27, x21
    b.ls reversed_line
    sub x27, x27, #1
    ldrb w0, [x24, x27]
    strb w0, [x25, x23]
    add x23, x23, #1
    b reverse_line

reversed_line:
    mov w0, #10
    strb w0, [x25, x23]
    add x23, x23, #1
    add x22, x22, #1
    mov x21, x22
    b scan_input

input_end:
    cmp x21, x19
    b.eq prepare_write
    mov x27, x19

reverse_final:
    cmp x27, x21
    b.ls reversed_final
    sub x27, x27, #1
    ldrb w0, [x24, x27]
    strb w0, [x25, x23]
    add x23, x23, #1
    b reverse_final

reversed_final:
    mov w0, #10
    strb w0, [x25, x23]
    add x23, x23, #1

prepare_write:
    mov x26, #0
    mov x27, x23

write_loop:
    cmp x26, x27
    b.hs finish
    mov x0, #1
    add x1, x25, x26
    sub x2, x27, x26
    mov x8, #64
    svc #0
    cbz x0, write_loop
    add x26, x26, x0
    b write_loop

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.align 4
input_buf:
    .zero 65536
output_buf:
    .zero 131072
