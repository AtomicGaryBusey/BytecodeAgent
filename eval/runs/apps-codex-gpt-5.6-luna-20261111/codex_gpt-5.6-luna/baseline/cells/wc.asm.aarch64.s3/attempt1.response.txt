.equ BUF_SIZE, 65536
.equ OUT_SIZE, 128

.globl _start

.text

_start:
    mov x20, #0
    mov x21, #0
    mov x22, #0
    mov x23, #0

    adrp x24, input_buffer
    add x24, x24, :lo12:input_buffer

read_loop:
    mov x0, #0
    mov x1, x24
    mov x2, #BUF_SIZE
    mov x8, #63
    svc #0
    cbz x0, make_output

    add x22, x22, x0
    mov x25, x24
    add x26, x25, x0

scan_loop:
    cmp x25, x26
    b.eq read_loop

    ldrb w27, [x25], #1

    cmp w27, #10
    b.eq byte_newline
    cmp w27, #32
    b.eq byte_whitespace
    cmp w27, #9
    b.eq byte_whitespace
    cmp w27, #11
    b.eq byte_whitespace
    cmp w27, #12
    b.eq byte_whitespace
    cmp w27, #13
    b.eq byte_whitespace

    cbnz x23, scan_loop
    add x21, x21, #1
    mov x23, #1
    b scan_loop

byte_newline:
    add x20, x20, #1

byte_whitespace:
    mov x23, #0
    b scan_loop

make_output:
    adrp x19, output_buffer
    add x19, x19, :lo12:output_buffer
    mov x1, x19

    mov x0, x20
    bl num_to_ascii
    mov x1, x0
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl num_to_ascii
    mov x1, x0
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x22
    bl num_to_ascii
    mov x1, x0
    mov w2, #10
    strb w2, [x1], #1

    mov x2, x1
    sub x2, x2, x19
    mov x1, x19

write_loop:
    mov x0, #1
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_loop

    mov x0, #0
    mov x8, #93
    svc #0

num_to_ascii:
    cbnz x0, num_nonzero
    mov w6, #48
    strb w6, [x1], #1
    mov x0, x1
    ret

num_nonzero:
    mov x2, #10
    mov x3, #1
    mov x5, x0

num_find_divisor:
    udiv x4, x5, x3
    cmp x4, #10
    b.lo num_digits
    mul x3, x3, x2
    b num_find_divisor

num_digits:
    udiv x4, x5, x3
    add w6, w4, #48
    strb w6, [x1], #1
    msub x5, x4, x3, x5
    cmp x3, #1
    b.eq num_done
    udiv x3, x3, x2
    b num_digits

num_done:
    mov x0, x1
    ret

.bss
.align 4
input_buffer:
.zero BUF_SIZE

.align 4
output_buffer:
.zero OUT_SIZE
