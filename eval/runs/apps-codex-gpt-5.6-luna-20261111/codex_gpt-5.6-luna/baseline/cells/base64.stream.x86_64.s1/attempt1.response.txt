.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000

.globl _start

.text
_start:
    xor r12d, r12d
    lea r15, [rip + output]

read_loop:
    cmp r12, INPUT_SIZE
    je encode_start
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz encode_start
    add r12, rax
    jmp read_loop

encode_start:
    lea rsi, [rip + input]
    mov rdx, r12
    mov rdi, r15
    lea r8, [rip + alphabet]
    xor r14d, r14d

encode_loop:
    cmp rdx, 3
    jb remainder

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx ecx, byte ptr [rsi + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rsi + 2]
    or eax, ecx
    mov ebx, eax

    mov eax, ebx
    shr eax, 18
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al

    mov eax, ebx
    shr eax, 12
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al

    mov eax, ebx
    shr eax, 6
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], al

    mov eax, ebx
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 3], al

    add rdi, 4
    add rsi, 3
    sub rdx, 3
    add r14d, 4
    cmp r14d, 76
    jne encode_loop

    mov byte ptr [rdi], 10
    inc rdi
    xor r14d, r14d
    jmp encode_loop

remainder:
    test rdx, rdx
    jz finish_lines
    cmp rdx, 1
    je remainder_one

    movzx ebx, byte ptr [rsi]
    shl ebx, 8
    movzx eax, byte ptr [rsi + 1]
    or ebx, eax

    mov eax, ebx
    shr eax, 10
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al

    mov eax, ebx
    shr eax, 4
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al

    mov eax, ebx
    shl eax, 2
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 2], al

    mov byte ptr [rdi + 3], '='
    add rdi, 4
    jmp final_newline

remainder_one:
    movzx ebx, byte ptr [rsi]

    mov eax, ebx
    shr eax, 2
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi], al

    mov eax, ebx
    shl eax, 4
    and eax, 63
    movzx eax, byte ptr [r8 + rax]
    mov byte ptr [rdi + 1], al

    mov word ptr [rdi + 2], 0x3d3d
    add rdi, 4

final_newline:
    mov byte ptr [rdi], 10
    inc rdi
    jmp prepare_write

finish_lines:
    cmp r14d, 0
    je prepare_write
    mov byte ptr [rdi], 10
    inc rdi

prepare_write:
    mov rdx, rdi
    sub rdx, r15
    test rdx, rdx
    jz exit_program
    mov rsi, r15

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input:
    .zero INPUT_SIZE
output:
    .zero OUTPUT_SIZE
