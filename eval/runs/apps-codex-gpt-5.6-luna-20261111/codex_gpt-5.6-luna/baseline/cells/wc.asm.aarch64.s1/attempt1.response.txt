.equ BUF_SIZE, 65536

.globl _start

.text

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_more:
    mov x0, #0
    mov x1, x23
    mov x2, #BUF_SIZE
    mov x8, #63
    svc #0
    cmp x0, #0
    b.le finish_input

    mov x26, x0
    add x21, x21, x26
    mov x24, x23
    add x25, x23, x26

process_input:
    cmp x24, x25
    b.hs read_more

    ldrb w27, [x24], #1
    cmp w27, #10
    b.eq newline_byte
    cmp w27, #32
    b.eq whitespace_byte
    cmp w27, #9
    b.eq whitespace_byte
    cmp w27, #11
    b.eq whitespace_byte
    cmp w27, #12
    b.eq whitespace_byte
    cmp w27, #13
    b.eq whitespace_byte

    cbnz x22, process_input
    add x20, x20, #1
    mov x22, #1
    b process_input

newline_byte:
    add x19, x19, #1
    mov x22, #0
    b process_input

whitespace_byte:
    mov x22, #0
    b process_input

finish_input:
    adrp x26, output_buffer
    add x26, x26, :lo12:output_buffer
    mov x1, x26

    mov x0, x19
    bl print_num
    mov x1, x0
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x20
    bl print_num
    mov x1, x0
    mov w2, #32
    strb w2, [x1], #1

    mov x0, x21
    bl print_num
    mov x1, x0
    mov w2, #10
    strb w2, [x1], #1

    mov x27, x1
    mov x1, x26
    sub x2, x27, x26

write_loop:
    mov x0, #1
    mov x8, #64
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    cbnz x2, write_loop

    mov x0, #0
    mov x8, #93
    svc #0

print_num:
    mov x2, x1
    cbnz x0, print_num_loop

    mov w3, #48
    strb w3, [x1], #1
    mov x0, x1
    ret

print_num_loop:
    mov x5, #10
    udiv x4, x0, x5
    msub x6, x4, x5, x0
    add x6, x6, #48
    strb w6, [x1], #1
    mov x0, x4
    cbnz x0, print_num_loop

    sub x7, x1, #1
    mov x8, x2

print_num_reverse:
    cmp x8, x7
    b.hs print_num_done
    ldrb w3, [x8]
    ldrb w4, [x7]
    strb w4, [x8]
    strb w3, [x7]
    add x8, x8, #1
    sub x7, x7, #1
    b print_num_reverse

print_num_done:
    mov x0, x1
    ret

.bss
.balign 16
input_buffer:
    .zero BUF_SIZE
output_buffer:
    .zero 64
