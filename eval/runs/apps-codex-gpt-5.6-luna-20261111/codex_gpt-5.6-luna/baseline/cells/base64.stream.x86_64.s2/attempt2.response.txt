.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000
.equ LINE_SIZE, 76

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12, r12

read_loop:
    cmp r12, INPUT_SIZE
    jae read_done
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rsi, [rip + alphabet]
    lea r13, [rip + output_buffer]
    xor r14, r14
    xor r15, r15

encode_loop:
    mov rax, r12
    sub rax, r15
    cmp rax, 3
    jb remainder

    movzx eax, byte ptr [rbx + r15]
    shl eax, 16
    movzx ecx, byte ptr [rbx + r15 + 1]
    shl ecx, 8
    or eax, ecx
    movzx ecx, byte ptr [rbx + r15 + 2]
    or eax, ecx
    mov r10d, eax

    mov edx, r10d
    shr edx, 18
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov edx, r10d
    shr edx, 12
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov edx, r10d
    shr edx, 6
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov edx, r10d
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    add r15, 3
    jmp encode_loop

remainder:
    cmp r15, r12
    je finish_output

    mov rax, r12
    sub rax, r15
    cmp rax, 1
    je one_byte

    movzx eax, byte ptr [rbx + r15]
    shl eax, 16
    movzx ecx, byte ptr [rbx + r15 + 1]
    shl ecx, 8
    or eax, ecx
    mov r10d, eax

    mov edx, r10d
    shr edx, 18
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov edx, r10d
    shr edx, 12
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov edx, r10d
    shr edx, 6
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov al, 61
    call emit_char
    jmp finish_output

one_byte:
    movzx eax, byte ptr [rbx + r15]
    shl eax, 16
    mov r10d, eax

    mov edx, r10d
    shr edx, 18
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov edx, r10d
    shr edx, 12
    and edx, 63
    mov al, byte ptr [rsi + rdx]
    call emit_char

    mov al, 61
    call emit_char
    mov al, 61
    call emit_char

finish_output:
    test r14, r14
    jz write_output
    mov byte ptr [r13], 10
    inc r13

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, r13
    sub rdx, rsi
    jz exit_program

write_loop:
    mov edi, 1
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

emit_char:
    mov byte ptr [r13], al
    inc r13
    inc r14
    cmp r14, LINE_SIZE
    jne emit_done
    mov byte ptr [r13], 10
    inc r13
    xor r14, r14
emit_done:
    ret

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero OUTPUT_SIZE
