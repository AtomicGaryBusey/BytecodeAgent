.equ input_size, 49152
.equ output_size, 70000

.globl _start

.section .text
_start:
    lea rbx, [rip + input_buffer]
    xor r12, r12

read_loop:
    cmp r12, input_size
    je have_input
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r12]
    mov edx, input_size
    sub edx, r12d
    syscall
    add r12, rax
    test rax, rax
    jne read_loop

have_input:
    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    lea r11, [rip + alphabet]
    xor r13, r13

encode_loop:
    cmp r12, 3
    jb encode_final

    movzx r8d, byte ptr [rsi]
    movzx r9d, byte ptr [rsi + 1]
    movzx r10d, byte ptr [rsi + 2]

    mov eax, r8d
    shr eax, 2
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi], dl

    mov eax, r8d
    and eax, 3
    shl eax, 4
    mov edx, r9d
    shr edx, 4
    or eax, edx
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi + 1], dl

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov edx, r10d
    shr edx, 6
    or eax, edx
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi + 2], dl

    mov eax, r10d
    and eax, 63
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi + 3], dl

    add rdi, 4
    add r13, 4
    cmp r13, 76
    jne no_line_break
    mov byte ptr [rdi], 10
    inc rdi
    xor r13, r13

no_line_break:
    add rsi, 3
    sub r12, 3
    jmp encode_loop

encode_final:
    test r12, r12
    jz finish_line

    movzx r8d, byte ptr [rsi]
    mov eax, r8d
    shr eax, 2
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi], dl

    mov eax, r8d
    and eax, 3
    shl eax, 4

    cmp r12, 1
    je final_one

    movzx r9d, byte ptr [rsi + 1]
    mov edx, r9d
    shr edx, 4
    or eax, edx
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi + 1], dl

    mov eax, r9d
    and eax, 15
    shl eax, 2
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi + 2], dl
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r13, 4
    jmp finish_line

final_one:
    mov dl, byte ptr [r11 + rax]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    add r13, 4

finish_line:
    test r13, r13
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_program

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buffer:
    .zero input_size
output_buffer:
    .zero output_size
