.equ INPUT_SIZE, 65536

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x21, #0

read_loop:
    mov x0, #INPUT_SIZE
    sub x2, x0, x21
    cbz x2, input_done
    mov x0, #0
    add x1, x19, x21
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x21, x21, x0
    b read_loop

input_done:
    adrp x25, hexchars
    add x25, x25, :lo12:hexchars
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x20, #0

row_loop:
    cmp x20, x21
    b.eq final_offset

    sub x22, x21, x20
    mov x27, #16
    cmp x22, x27
    csel x22, x27, x22, hi

    mov x23, x24
    mov x12, #28

row_offset_loop:
    lsr x13, x20, x12
    and x13, x13, #15
    ldrb w14, [x25, x13]
    strb w14, [x23], #1
    subs x12, x12, #4
    b.ge row_offset_loop

    mov w14, #32
    strb w14, [x23], #1
    strb w14, [x23], #1

    add x26, x19, x20
    mov x12, #0

hex_slot_loop:
    cmp x12, x22
    b.hs hex_padding

    ldrb w14, [x26, x12]
    lsr w13, w14, #4
    ldrb w15, [x25, x13]
    strb w15, [x23], #1
    and w13, w14, #15
    ldrb w15, [x25, x13]
    strb w15, [x23], #1
    mov w15, #32
    strb w15, [x23], #1
    b hex_slot_next

hex_padding:
    mov w15, #32
    strb w15, [x23], #1
    strb w15, [x23], #1
    strb w15, [x23], #1

hex_slot_next:
    add x12, x12, #1
    cmp x12, #8
    b.ne hex_slot_check
    mov w15, #32
    strb w15, [x23], #1

hex_slot_check:
    cmp x12, #16
    b.lo hex_slot_loop

    mov w15, #32
    strb w15, [x23], #1
    mov w15, #124
    strb w15, [x23], #1

    mov x12, #0

ascii_loop:
    cmp x12, x22
    b.hs ascii_done
    ldrb w14, [x26, x12]
    cmp w14, #32
    b.lo ascii_dot
    cmp w14, #126
    b.hi ascii_dot
    strb w14, [x23], #1
    b ascii_next

ascii_dot:
    mov w15, #46
    strb w15, [x23], #1

ascii_next:
    add x12, x12, #1
    b ascii_loop

ascii_done:
    mov w15, #124
    strb w15, [x23], #1
    mov w15, #10
    strb w15, [x23], #1

    sub x1, x23, x24
    mov x0, x24
    bl write_all

    add x20, x20, x22
    b row_loop

final_offset:
    mov x23, x24
    mov x12, #28

final_offset_loop:
    lsr x13, x20, x12
    and x13, x13, #15
    ldrb w14, [x25, x13]
    strb w14, [x23], #1
    subs x12, x12, #4
    b.ge final_offset_loop

    mov w14, #10
    strb w14, [x23]
    sub x1, x23, x24
    add x1, x1, #1
    mov x0, x24
    bl write_all

    mov x0, #0
    mov x8, #93
    svc #0

write_all:
    mov x9, x0
    mov x10, x1
    cbz x10, write_done

write_loop:
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    add x9, x9, x0
    subs x10, x10, x0
    b.ne write_loop

write_done:
    ret

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.section .bss
.align 4
input_buf:
    .zero INPUT_SIZE
output_buf:
    .zero 80
