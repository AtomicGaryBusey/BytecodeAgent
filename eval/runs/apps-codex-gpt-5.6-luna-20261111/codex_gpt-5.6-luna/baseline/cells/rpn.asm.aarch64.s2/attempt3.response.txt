.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ INPUT_SIZE, 65536

.section .text
.globl _start
_start:
	adrp x19, input
	add x19, x19, :lo12:input
	adrp x24, values
	add x24, x24, :lo12:values
	mov x20, #0

read_loop:
	mov x21, #INPUT_SIZE
	cmp x20, x21
	beq process_input
	sub x2, x21, x20
	add x1, x19, x20
	mov x0, #0
	mov x8, #SYS_read
	svc #0
	cmp x0, #0
	beq process_input
	add x20, x20, x0
	b read_loop

process_input:
	mov x22, #0
	cmp x22, x20
	beq finish

line_begin:
	add x27, x19, x22
	mov x28, x27
	add x29, x19, x20

line_scan:
	cmp x28, x29
	beq line_final
	ldrb w0, [x28]
	add x28, x28, #1
	cmp w0, #10
	b.ne line_scan
	sub x28, x28, #1
	add x23, x28, #1
	b parse_setup

line_final:
	mov x28, x29
	mov x23, x29

parse_setup:
	mov x25, #0

parse_loop:
	cmp x27, x28
	beq line_done
	ldrb w0, [x27]
	cmp w0, #32
	beq skip_separator
	cmp w0, #9
	beq skip_separator

	mov x0, x27
	mov x1, x27

token_scan:
	cmp x1, x28
	beq token_found
	ldrb w3, [x1]
	cmp w3, #32
	beq token_found
	cmp w3, #9
	beq token_found
	add x1, x1, #1
	b token_scan

token_found:
	sub x2, x1, x0
	mov x27, x1
	cmp x2, #1
	b.ne literal_check
	ldrb w3, [x0]
	cmp w3, #43
	beq do_operator
	cmp w3, #45
	beq do_operator
	cmp w3, #42
	beq do_operator
	cmp w3, #47
	beq do_operator
	cmp w3, #37
	beq do_operator

literal_check:
	ldrb w3, [x0]
	cmp w3, #45
	b.ne positive_literal
	cmp x2, #1
	beq invalid_line
	cmp x2, #19
	b.hs invalid_line
	add x3, x0, #1
	sub x4, x2, #1
	mov x6, #1
	b parse_digits

positive_literal:
	cmp x2, #1
	b.lo invalid_line
	cmp x2, #18
	b.hi invalid_line
	mov x3, x0
	mov x4, x2
	mov x6, #0

parse_digits:
	mov x5, #0
	mov x7, #0

digit_loop:
	cmp x7, x4
	beq digits_done
	ldrb w8, [x3, x7]
	cmp w8, #48
	b.lo invalid_line
	cmp w8, #57
	b.hi invalid_line
	sub w8, w8, #48
	mov x9, #10
	mul x5, x5, x9
	add x5, x5, x8
	add x7, x7, #1
	b digit_loop

digits_done:
	cmp x6, #0
	beq push_literal
	neg x5, x5

push_literal:
	str x5, [x24, x25, lsl #3]
	add x25, x25, #1
	b parse_loop

do_operator:
	cmp x25, #2
	b.lo invalid_line
	sub x25, x25, #1
	ldr x1, [x24, x25, lsl #3]
	sub x7, x25, #1
	ldr x0, [x24, x7, lsl #3]

	cmp w3, #43
	beq op_add
	cmp w3, #45
	beq op_sub
	cmp w3, #42
	beq op_mul
	cmp w3, #47
	beq op_div
	b op_mod

op_add:
	add x0, x0, x1
	b store_result

op_sub:
	sub x0, x0, x1
	b store_result

op_mul:
	mul x0, x0, x1
	b store_result

op_div:
	cmp x1, #0
	beq invalid_line
	movz x6, #0x8000, lsl #48
	cmp x0, x6
	b.ne signed_div
	mov x6, #-1
	cmp x1, x6
	beq invalid_line

signed_div:
	sdiv x0, x0, x1
	b store_result

op_mod:
	cmp x1, #0
	beq invalid_line
	movz x6, #0x8000, lsl #48
	cmp x0, x6
	b.ne signed_mod
	mov x6, #-1
	cmp x1, x6
	beq invalid_line

signed_mod:
	sdiv x6, x0, x1
	msub x0, x6, x1, x0
	b store_result

store_result:
	str x0, [x24, x7, lsl #3]
	b parse_loop

skip_separator:
	add x27, x27, #1
	b parse_loop

line_done:
	cmp x25, #1
	b.ne invalid_line
	ldr x0, [x24]
	b print_number

invalid_line:
	adrp x1, error_string
	add x1, x1, :lo12:error_string
	mov x2, #6
	b write_output

print_number:
	adrp x1, output
	add x1, x1, :lo12:output
	add x2, x1, #63
	mov x3, #0
	tst x0, x0
	b.ge decimal_loop_start
	mov x3, #1
	neg x0, x0

decimal_loop_start:
	mov x6, #10

decimal_loop:
	udiv x4, x0, x6
	msub x5, x4, x6, x0
	add w5, w5, #48
	sub x2, x2, #1
	strb w5, [x2]
	mov x0, x4
	cmp x0, #0
	b.ne decimal_loop
	cmp x3, #0
	beq decimal_sign_done
	mov w5, #45
	sub x2, x2, #1
	strb w5, [x2]

decimal_sign_done:
	mov w5, #10
	adrp x1, output
	add x1, x1, :lo12:output
	add x3, x1, #63
	strb w5, [x3]
	add x3, x3, #1
	sub x2, x3, x2

write_output:
	mov x0, #1
	mov x8, #SYS_write
	svc #0
	mov x4, x2
	sub x2, x4, x0
	add x1, x1, x0
	cmp x2, #0
	b.ne write_output

next_line:
	sub x22, x23, x19
	cmp x22, x20
	b.ne line_begin

finish:
	mov x0, #0
	mov x8, #SYS_exit
	svc #0

.section .rodata
error_string:
	.ascii "error\n"

.section .bss
.balign 16
input:
	.zero 65536
.balign 16
values:
	.zero 8192
.balign 16
output:
	.zero 64
