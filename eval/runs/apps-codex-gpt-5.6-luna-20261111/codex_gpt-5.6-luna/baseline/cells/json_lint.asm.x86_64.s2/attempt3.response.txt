.section .rodata
true_lit:
.ascii "true"
false_lit:
.ascii "false"
null_lit:
.ascii "null"
ok_msg:
.ascii "ok\n"
err_msg:
.ascii "error at "

.equ true_len, 4
.equ false_len, 5
.equ null_len, 4
.equ ok_len, 3
.equ err_len, 9
.equ input_size, 65536

.section .bss
.align 16
buffer:
.zero input_size
outbuf:
.zero 128

.text
.globl _start
_start:
	lea r12, [rip + buffer]
	xor r13d, r13d

read_loop:
	xor eax, eax
	xor edi, edi
	lea rsi, [r12 + r13]
	mov edx, input_size
	sub rdx, r13
	syscall
	test rax, rax
	jz input_done
	add r13, rax
	jmp read_loop

input_done:
	xor edi, edi
	call skip_ws
	mov rdi, rax
	call parse_value
	jc report_error
	mov rdi, rax
	call skip_ws
	cmp rax, r13
	jne report_error

	mov eax, 1
	mov edi, 1
	lea rsi, [rip + ok_msg]
	mov edx, ok_len
	syscall
	mov eax, 60
	xor edi, edi
	syscall

report_error:
	mov r14, rax
	lea rsi, [rip + err_msg]
	lea rdi, [rip + outbuf]
	mov ecx, err_len
1:
	mov al, byte ptr [rsi]
	mov byte ptr [rdi], al
	inc rsi
	inc rdi
	dec ecx
	jnz 1b

	lea r15, [rip + outbuf + 63]
	mov rax, r14
	test rax, rax
	jnz 2f
	mov byte ptr [r15], '0'
	dec r15
	jmp 3f
2:
	xor edx, edx
	mov ebx, 10
	div rbx
	add dl, '0'
	mov byte ptr [r15], dl
	dec r15
	test rax, rax
	jnz 2b
3:
	inc r15
	lea rsi, [rip + outbuf + 9]
	lea rdx, [rip + outbuf + 64]
4:
	cmp r15, rdx
	je 5f
	mov al, byte ptr [r15]
	mov byte ptr [rsi], al
	inc r15
	inc rsi
	jmp 4b
5:
	mov byte ptr [rsi], 10
	inc rsi
	lea rdx, [rip + outbuf]
	sub rsi, rdx
	mov edx, esi
	mov eax, 1
	mov edi, 1
	lea rsi, [rip + outbuf]
	syscall
	mov eax, 60
	mov edi, 1
	syscall

skip_ws:
	mov r8, rdi
1:
	cmp r8, r13
	jae 2f
	movzx eax, byte ptr [r12 + r8]
	cmp al, 32
	je 3f
	cmp al, 9
	je 3f
	cmp al, 10
	je 3f
	cmp al, 13
	jne 2f
3:
	inc r8
	jmp 1b
2:
	mov rax, r8
	ret

parse_value:
	cmp rdi, r13
	jae value_eof
	movzx eax, byte ptr [r12 + rdi]
	cmp al, '{'
	je value_object
	cmp al, '['
	je value_array
	cmp al, '"'
	je value_string
	cmp al, '-'
	je value_number
	cmp al, '0'
	jb value_literal
	cmp al, '9'
	jbe value_number
value_literal:
	cmp al, 't'
	je value_true
	cmp al, 'f'
	je value_false
	cmp al, 'n'
	je value_null
	mov rax, rdi
	stc
	ret
value_eof:
	mov rax, r13
	stc
	ret
value_object:
	jmp parse_object
value_array:
	jmp parse_array
value_string:
	jmp parse_string
value_number:
	jmp parse_number
value_true:
	lea rsi, [rip + true_lit]
	mov ecx, true_len
	jmp parse_literal
value_false:
	lea rsi, [rip + false_lit]
	mov ecx, false_len
	jmp parse_literal
value_null:
	lea rsi, [rip + null_lit]
	mov ecx, null_len
	jmp parse_literal

parse_literal:
	mov r8, rdi
	xor r9d, r9d
1:
	cmp r9d, ecx
	jae 2f
	cmp r8, r13
	jae 3f
	movzx eax, byte ptr [r12 + r8]
	cmp al, byte ptr [rsi + r9]
	jne 4f
	inc r8
	inc r9
	jmp 1b
2:
	mov rax, r8
	clc
	ret
3:
	mov rax, r13
	stc
	ret
4:
	mov rax, r8
	stc
	ret

parse_string:
	mov r8, rdi
	inc r8
1:
	cmp r8, r13
	jae 2f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '"'
	je 3f
	cmp al, 92
	je 4f
	cmp al, 32
	jb 5f
	inc r8
	jmp 1b
4:
	inc r8
	cmp r8, r13
	jae 2f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '"'
	je 6f
	cmp al, 92
	je 6f
	cmp al, '/'
	je 6f
	cmp al, 'b'
	je 6f
	cmp al, 'f'
	je 6f
	cmp al, 'n'
	je 6f
	cmp al, 'r'
	je 6f
	cmp al, 't'
	je 6f
	cmp al, 'u'
	jne 5f
	inc r8
	mov edx, 4
7:
	cmp r8, r13
	jae 2f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '0'
	jb 8f
	cmp al, '9'
	jbe 9f
8:
	cmp al, 'A'
	jb 10f
	cmp al, 'F'
	jbe 9f
10:
	cmp al, 'a'
	jb 5f
	cmp al, 'f'
	ja 5f
9:
	inc r8
	dec edx
	jnz 7b
	jmp 1b
6:
	inc r8
	jmp 1b
3:
	inc r8
	mov rax, r8
	clc
	ret
2:
	mov rax, r13
	stc
	ret
5:
	mov rax, r8
	stc
	ret

parse_number:
	mov r8, rdi
	cmp r8, r13
	jae 1f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '-'
	jne 2f
	inc r8
	cmp r8, r13
	jae 1f
	movzx eax, byte ptr [r12 + r8]
2:
	cmp al, '0'
	je 3f
	cmp al, '1'
	jb 4f
	cmp al, '9'
	ja 4f
5:
	inc r8
	cmp r8, r13
	jae 6f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '0'
	jb 7f
	cmp al, '9'
	jbe 5b
	jmp 7f
3:
	inc r8
	cmp r8, r13
	jae 6f
	movzx eax, byte ptr [r12 + r8]
7:
	cmp al, '.'
	jne 8f
	inc r8
	cmp r8, r13
	jae 1f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '0'
	jb 4f
	cmp al, '9'
	ja 4f
9:
	inc r8
	cmp r8, r13
	jae 6f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '0'
	jb 8f
	cmp al, '9'
	jbe 9b
8:
	cmp al, 'e'
	je 10f
	cmp al, 'E'
	jne 6f
10:
	inc r8
	cmp r8, r13
	jae 1f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '+'
	je 11f
	cmp al, '-'
	jne 12f
11:
	inc r8
	cmp r8, r13
	jae 1f
	movzx eax, byte ptr [r12 + r8]
12:
	cmp al, '0'
	jb 4f
	cmp al, '9'
	ja 4f
13:
	inc r8
	cmp r8, r13
	jae 6f
	movzx eax, byte ptr [r12 + r8]
	cmp al, '0'
	jb 6f
	cmp al, '9'
	jbe 13b
6:
	mov rax, r8
	clc
	ret
1:
	mov rax, r13
	stc
	ret
4:
	mov rax, r8
	stc
	ret

parse_array:
	inc rdi
	call skip_ws
	mov r8, rax
	cmp r8, r13
	jae 1f
	cmp byte ptr [r12 + r8], ']'
	je 2f
3:
	push r8
	mov rdi, r8
	call parse_value
	jc 4f
	pop r8
	mov r8, rax
	mov rdi, r8
	call skip_ws
	mov r8, rax
	cmp r8, r13
	jae 1f
	cmp byte ptr [r12 + r8], ']'
	je 2f
	cmp byte ptr [r12 + r8], ','
	jne 5f
	inc r8
	mov rdi, r8
	call skip_ws
	mov r8, rax
	jmp 3b
4:
	pop r8
	ret
2:
	inc r8
	mov rax, r8
	clc
	ret
1:
	mov rax, r13
	stc
	ret
5:
	mov rax, r8
	stc
	ret

parse_object:
	inc rdi
	call skip_ws
	mov r8, rax
	cmp r8, r13
	jae 1f
	cmp byte ptr [r12 + r8], '}'
	je 2f
3:
	cmp byte ptr [r12 + r8], '"'
	jne 5f
	push r8
	mov rdi, r8
	call parse_string
	jc 4f
	pop r8
	mov r8, rax
	mov rdi, r8
	call skip_ws
	mov r8, rax
	cmp r8, r13
	jae 1f
	cmp byte ptr [r12 + r8], ':'
	jne 5f
	inc r8
	mov rdi, r8
	call skip_ws
	mov r8, rax
	push r8
	mov rdi, r8
	call parse_value
	jc 4f
	pop r8
	mov r8, rax
	mov rdi, r8
	call skip_ws
	mov r8, rax
	cmp r8, r13
	jae 1f
	cmp byte ptr [r12 + r8], '}'
	je 2f
	cmp byte ptr [r12 + r8], ','
	jne 5f
	inc r8
	mov rdi, r8
	call skip_ws
	mov r8, rax
	jmp 3b
4:
	pop r8
	ret
2:
	inc r8
	mov rax, r8
	clc
	ret
1:
	mov rax, r13
	stc
	ret
5:
	mov rax, r8
	stc
	ret
