.equ INPUT_SIZE, 32768
.equ MAX_RECORDS, 1000

.section .bss
.align 8
input_buffer:
.zero INPUT_SIZE
records:
.zero 16000
record_count:
.zero 8

.section .rodata
newline:
.byte 10

.text
.globl _start
_start:
    xor r12, r12

read_loop:
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    xor rbx, rbx
    xor r10, r10
    xor r11, r11

parse_loop:
    cmp rbx, r12
    jae parse_done
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 10
    jne parse_next

    mov rax, r11
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov qword ptr [rdi], r10
    mov rax, rbx
    sub rax, r10
    mov qword ptr [rdi + 8], rax
    inc r11
    lea r10, [rbx + 1]

parse_next:
    inc rbx
    jmp parse_loop

parse_done:
    cmp r10, r12
    je save_count
    mov rax, r11
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov qword ptr [rdi], r10
    mov rax, r12
    sub rax, r10
    mov qword ptr [rdi + 8], rax
    inc r11

save_count:
    mov qword ptr [rip + record_count], r11
    cmp r11, 1
    jbe output_start

    mov rbx, 1

sort_outer:
    cmp rbx, r11
    jae output_start

    mov rax, rbx
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]

    mov rdx, rbx

find_position:
    test rdx, rdx
    jz insert_zero
    dec rdx

    mov rax, rdx
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov r10, qword ptr [rdi]
    mov r11, qword ptr [rdi + 8]

    xor eax, eax
    lea rdi, [rip + input_buffer]
    add rdi, r8
    lea rsi, [rip + input_buffer]
    add rsi, r10

compare_bytes:
    cmp rax, r9
    jae key_ended
    cmp rax, r11
    jae existing_ended
    movzx ecx, byte ptr [rdi + rax]
    movzx esi, byte ptr [rsi + rax]
    cmp ecx, esi
    jb insert_here
    ja no_insert
    inc rax
    jmp compare_bytes

key_ended:
    cmp rax, r11
    jb insert_here
    jmp no_insert

existing_ended:
    jmp no_insert

insert_here:
    inc rdx
    jmp perform_insert

insert_zero:
    xor edx, edx

perform_insert:
    mov rax, rbx

shift_loop:
    cmp rax, rdx
    je place_key
    dec rax
    mov rcx, rax
    shl rcx, 4
    lea rdi, [rip + records]
    add rdi, rcx
    mov rcx, qword ptr [rdi]
    mov qword ptr [rdi + 16], rcx
    mov rcx, qword ptr [rdi + 8]
    mov qword ptr [rdi + 24], rcx
    jmp shift_loop

place_key:
    mov rax, rdx
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov qword ptr [rdi], r8
    mov qword ptr [rdi + 8], r9
    inc rbx
    jmp sort_outer

no_insert:
    inc rbx
    jmp sort_outer

output_start:
    xor rbx, rbx
    mov r12, qword ptr [rip + record_count]

output_record:
    cmp rbx, r12
    jae exit_zero

    mov rax, rbx
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]

write_line:
    test r9, r9
    jz write_newline
    lea rsi, [rip + input_buffer]
    add rsi, r8
    mov edx, r9d
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jz write_line
    add r8, rax
    sub r9, rax
    jmp write_line

write_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc rbx
    jmp output_record

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
    hlt
