.equ INPUT_SIZE, 32768
.equ MAX_RECORDS, 1000
.equ RECORD_SIZE, 16

.section .bss
.align 8
input_buffer:
.zero INPUT_SIZE
records:
.zero MAX_RECORDS * RECORD_SIZE
record_count:
.zero 8

.section .rodata
newline:
.byte 10

.text
.globl _start
_start:
    xor r12, r12

read_loop:
    cmp r12, INPUT_SIZE
    jae parse_input
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov rdx, INPUT_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    xor rbx, rbx
    xor r10, r10
    xor r11, r11

parse_loop:
    cmp rbx, r12
    jae parse_done
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 10
    jne parse_next

    mov rax, r11
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov qword ptr [rdi], r10
    mov rax, rbx
    sub rax, r10
    mov qword ptr [rdi + 8], rax
    inc r11
    lea r10, [rbx + 1]

parse_next:
    inc rbx
    jmp parse_loop

parse_done:
    cmp r10, r12
    je save_count

    mov rax, r11
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov qword ptr [rdi], r10
    mov rax, r12
    sub rax, r10
    mov qword ptr [rdi + 8], rax
    inc r11

save_count:
    mov qword ptr [rip + record_count], r11
    cmp r11, 1
    jbe output_start

    mov rbx, 1

sort_outer:
    mov r11, qword ptr [rip + record_count]
    cmp rbx, r11
    jae output_start

    mov rax, rbx
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]

    lea rdi, [rip + input_buffer]
    add rdi, r8
    mov r14, rbx

find_position:
    test r14, r14
    jz insert_zero
    dec r14

    mov rax, r14
    shl rax, 4
    lea rsi, [rip + records]
    add rsi, rax
    mov r10, qword ptr [rsi]
    mov r15, qword ptr [rsi + 8]

    lea rsi, [rip + input_buffer]
    add rsi, r10
    xor rcx, rcx

compare_bytes:
    cmp rcx, r9
    jae key_ended
    cmp rcx, r15
    jae existing_ended

    movzx edx, byte ptr [rdi + rcx]
    movzx eax, byte ptr [rsi + rcx]
    cmp edx, eax
    jb insert_before
    ja continue_search
    inc rcx
    jmp compare_bytes

key_ended:
    cmp rcx, r15
    jb insert_before
    jmp no_insert

existing_ended:
    jmp no_insert

continue_search:
    test r14, r14
    jz no_insert
    jmp find_position

insert_before:
    mov r15, r14
    jmp perform_insert

insert_zero:
    xor r15, r15

perform_insert:
    mov rax, rbx

shift_loop:
    cmp rax, r15
    je place_key
    dec rax
    mov rcx, rax
    shl rcx, 4
    lea rdi, [rip + records]
    add rdi, rcx
    mov rsi, qword ptr [rdi]
    mov qword ptr [rdi + 16], rsi
    mov rsi, qword ptr [rdi + 8]
    mov qword ptr [rdi + 24], rsi
    jmp shift_loop

place_key:
    mov rax, r15
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov qword ptr [rdi], r8
    mov qword ptr [rdi + 8], r9
    inc rbx
    jmp sort_outer

no_insert:
    inc rbx
    jmp sort_outer

output_start:
    xor rbx, rbx
    mov r12, qword ptr [rip + record_count]

output_record:
    cmp rbx, r12
    jae exit_zero

    mov rax, rbx
    shl rax, 4
    lea rdi, [rip + records]
    add rdi, rax
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdi + 8]

write_line:
    test r9, r9
    jz write_newline
    lea rsi, [rip + input_buffer]
    add rsi, r8
    mov edx, r9d
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jz write_line
    add r8, rax
    sub r9, rax
    jmp write_line

write_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    test rax, rax
    jz write_newline
    inc rbx
    jmp output_record

exit_zero:
    xor edi, edi
    mov eax, 60
    syscall
    hlt
