.equ INPUT_SIZE, 32768
.equ MAX_TASKS, 2048

.section .bss
.align 8
input_buffer:
.zero INPUT_SIZE
records:
.zero 16000
aux_records:
.zero 16000
counts:
.zero 2056
positions:
.zero 2056
task_stack:
.zero 49152
task_count:
.zero 8

.section .rodata
newline:
.byte 10

.text
.globl _start
_start:
    lea rsi, [rip + input_buffer]
    xor r12, r12

read_loop:
    mov rdx, INPUT_SIZE
    sub rdx, r12
    mov eax, 0
    xor edi, edi
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    xor rbx, rbx
    xor r10, r10
    xor r9, r9
    lea rdi, [rip + records]

parse_loop:
    cmp rbx, r12
    jae parse_done
    movzx eax, byte ptr [rsi + rbx]
    cmp al, 10
    jne parse_next
    mov qword ptr [rdi + r9*16], r10
    mov rax, rbx
    sub rax, r10
    mov qword ptr [rdi + r9*16 + 8], rax
    inc r9
    lea r10, [rbx + 1]

parse_next:
    inc rbx
    jmp parse_loop

parse_done:
    cmp r10, r12
    je start_sort
    mov qword ptr [rdi + r9*16], r10
    mov rax, r12
    sub rax, r10
    mov qword ptr [rdi + r9*16 + 8], rax
    inc r9

start_sort:
    test r9, r9
    jz exit_zero
    mov qword ptr [rip + task_count], 1
    lea rdi, [rip + task_stack]
    mov qword ptr [rdi], 0
    mov qword ptr [rdi + 8], r9
    mov qword ptr [rdi + 16], 0

sort_task:
    mov rax, qword ptr [rip + task_count]
    test rax, rax
    jz output_start
    dec rax
    mov qword ptr [rip + task_count], rax
    lea rbx, [rip + task_stack]
    lea rbx, [rbx + rax*24]
    mov r8, qword ptr [rbx]
    mov r9, qword ptr [rbx + 8]
    mov r10, qword ptr [rbx + 16]
    mov rax, r9
    sub rax, r8
    cmp rax, 1
    jbe sort_task

    lea rdi, [rip + counts]
    xor ecx, ecx
clear_counts:
    mov qword ptr [rdi + rcx*8], 0
    inc rcx
    cmp rcx, 257
    jb clear_counts

    lea rdi, [rip + records]
    xor r11, r11
count_records:
    cmp r11, r9
    jae make_positions
    mov r12, qword ptr [rdi + r11*16]
    mov r13, qword ptr [rdi + r11*16 + 8]
    cmp r10, r13
    jae record_terminated
    mov rdx, r12
    add rdx, r10
    lea rsi, [rip + input_buffer]
    movzx eax, byte ptr [rsi + rdx]
    inc eax
    jmp record_key_ready
record_terminated:
    xor eax, eax
record_key_ready:
    lea rsi, [rip + counts]
    inc qword ptr [rsi + rax*8]
    inc r11
    jmp count_records

make_positions:
    lea rsi, [rip + positions]
    lea rdi, [rip + counts]
    xor ecx, ecx
    mov rdx, r8
make_positions_loop:
    mov qword ptr [rsi + rcx*8], rdx
    add rdx, qword ptr [rdi + rcx*8]
    inc rcx
    cmp rcx, 257
    jb make_positions_loop

    lea rdi, [rip + records]
    lea rsi, [rip + aux_records]
    xor r11, r11
distribute_records:
    cmp r11, r9
    jae copy_records
    mov r12, qword ptr [rdi + r11*16]
    mov r13, qword ptr [rdi + r11*16 + 8]
    cmp r10, r13
    jae distribute_terminated
    mov rdx, r12
    add rdx, r10
    lea rbx, [rip + input_buffer]
    movzx eax, byte ptr [rbx + rdx]
    inc eax
    jmp distribute_key_ready
distribute_terminated:
    xor eax, eax
distribute_key_ready:
    lea rbx, [rip + positions]
    mov rdx, qword ptr [rbx + rax*8]
    inc qword ptr [rbx + rax*8]
    mov qword ptr [rsi + rdx*16], r12
    mov qword ptr [rsi + rdx*16 + 8], r13
    inc r11
    jmp distribute_records

copy_records:
    lea rdi, [rip + records]
    lea rsi, [rip + aux_records]
    mov r11, r8
copy_loop:
    cmp r11, r9
    jae push_groups
    mov r12, qword ptr [rsi + r11*16]
    mov r13, qword ptr [rsi + r11*16 + 8]
    mov qword ptr [rdi + r11*16], r12
    mov qword ptr [rdi + r11*16 + 8], r13
    inc r11
    jmp copy_loop

push_groups:
    lea rdi, [rip + counts]
    lea rsi, [rip + positions]
    mov r14, 1
push_group_loop:
    cmp r14, 257
    jae sort_task
    mov r15, qword ptr [rdi + r14*8]
    cmp r15, 1
    jbe next_group
    mov rax, qword ptr [rip + task_count]
    lea rbx, [rip + task_stack]
    lea rbx, [rbx + rax*24]
    mov rdx, qword ptr [rsi + r14*8]
    sub rdx, r15
    mov qword ptr [rbx], rdx
    mov qword ptr [rbx + 8], qword ptr [rsi + r14*8]
    lea rdx, [r10 + 1]
    mov qword ptr [rbx + 16], rdx
    inc rax
    mov qword ptr [rip + task_count], rax
next_group:
    inc r14
    jmp push_group_loop

output_start:
    xor r12, r12
    lea rdi, [rip + records]

output_record:
    cmp r12, r9
    jae exit_zero
    mov r8, qword ptr [rdi + r12*16]
    mov r9, qword ptr [rdi + r12*16 + 8]

write_line:
    test r9, r9
    jz write_newline
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + input_buffer]
    add rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jz write_line
    add r8, rax
    sub r9, rax
    jmp write_line

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    inc r12
    lea rdi, [rip + records]
    jmp output_record

exit_zero:
    mov eax, 60
    xor edi, edi
    syscall
    hlt
