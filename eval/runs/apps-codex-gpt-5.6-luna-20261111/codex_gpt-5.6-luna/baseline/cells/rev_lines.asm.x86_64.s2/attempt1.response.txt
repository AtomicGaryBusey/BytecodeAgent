.equ INPUT_SIZE, 65536

.globl _start

.section .text
_start:
    lea rbx, [rip + input]
    xor r15, r15

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [rbx + r15]
    mov edx, INPUT_SIZE
    sub rdx, r15
    syscall
    test rax, rax
    jz process_start
    add r15, rax
    jmp read_loop

process_start:
    xor r14, r14
    xor r13, r13

scan_loop:
    cmp r13, r15
    jae final_line
    cmp byte ptr [rbx + r13], 10
    je found_newline
    inc r13
    jmp scan_loop

found_newline:
    mov r8, r14
    mov r9, r13
reverse_line:
    cmp r8, r9
    jae line_reversed
    dec r9
    cmp r8, r9
    jae line_reversed
    mov al, byte ptr [rbx + r8]
    mov r12b, byte ptr [rbx + r9]
    mov byte ptr [rbx + r8], r12b
    mov byte ptr [rbx + r9], al
    inc r8
    jmp reverse_line

line_reversed:
    mov r9, r13
    sub r9, r14
    lea r8, [rbx + r14]
    call write_bytes

    lea r8, [rbx + r13]
    mov r9, 1
    call write_bytes

    inc r13
    mov r14, r13
    cmp r14, r15
    jb scan_loop
    jmp finish

final_line:
    cmp r14, r15
    jae finish
    mov r8, r14
    mov r9, r15
reverse_final:
    cmp r8, r9
    jae final_reversed
    dec r9
    cmp r8, r9
    jae final_reversed
    mov al, byte ptr [rbx + r8]
    mov r12b, byte ptr [rbx + r9]
    mov byte ptr [rbx + r8], r12b
    mov byte ptr [rbx + r9], al
    inc r8
    jmp reverse_final

final_reversed:
    mov r9, r15
    sub r9, r14
    lea r8, [rbx + r14]
    call write_bytes

    lea r8, [rip + newline]
    mov r9, 1
    call write_bytes

finish:
    mov eax, 60
    xor edi, edi
    syscall

write_bytes:
    test r9, r9
    jz write_done
write_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    add r8, rax
    sub r9, rax
    jnz write_loop
write_done:
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.align 16
input:
    .zero INPUT_SIZE
