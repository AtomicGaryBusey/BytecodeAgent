.text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz input_done

    add r14, rax
    lea rbx, [rip + buffer]
    lea rsi, [rbx + rax]

process:
    cmp rbx, rsi
    jae read_more

    movzx eax, byte ptr [rbx]
    inc rbx

    cmp al, 10
    je newline
    cmp al, 9
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace
    cmp al, 32
    je whitespace

    test r15, r15
    jnz process
    inc r13
    mov r15d, 1
    jmp process

newline:
    inc r12

whitespace:
    xor r15d, r15d
    jmp process

input_done:
    lea rdi, [rip + output_end]

    dec rdi
    mov byte ptr [rdi], 10

    mov rax, r14
    call convert
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call convert
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call convert

    mov rsi, rdi
    lea rdx, [rip + output_end]
    sub rdx, rsi

write_more:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_more

    mov eax, 60
    xor edi, edi
    syscall

convert:
    test rax, rax
    jnz convert_loop
    dec rdi
    mov byte ptr [rdi], 48
    ret

convert_loop:
    xor edx, edx
    mov ecx, 10
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz convert_loop
    ret

.bss
.align 8
buffer:
.zero 65536
output:
.zero 128
output_end:
