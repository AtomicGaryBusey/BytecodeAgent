.equ input_size, 65536

.globl _start

.text
_start:
    lea rbx, [rip + input_buffer]
    xor r13, r13

read_more:
    cmp r13, input_size
    je process
    xor eax, eax
    xor edi, edi
    lea rsi, [rbx + r13]
    mov edx, input_size
    sub edx, r13d
    syscall
    test rax, rax
    jz process
    add r13, rax
    jmp read_more

process:
    xor r14, r14
    xor r15, r15

scan:
    cmp r14, r13
    jae final_line
    movzx eax, byte ptr [rbx + r14]
    cmp al, 10
    je found_newline
    inc r14
    jmp scan

found_newline:
    mov r12, 1
    jmp reverse_prepare

final_line:
    cmp r15, r13
    je exit_program
    xor r12, r12
    mov r14, r13

reverse_prepare:
    cmp r15, r14
    je output_line
    mov r8, r15
    mov r9, r14
    dec r9

reverse_loop:
    cmp r8, r9
    jae output_line
    mov r10b, byte ptr [rbx + r8]
    mov r11b, byte ptr [rbx + r9]
    mov byte ptr [rbx + r8], r11b
    mov byte ptr [rbx + r9], r10b
    inc r8
    dec r9
    jmp reverse_loop

output_line:
    mov rdx, r14
    sub rdx, r15
    test rdx, rdx
    jz emit_newline
    mov edi, 1
    lea rsi, [rbx + r15]

write_line:
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_line

emit_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    test r12, r12
    jz exit_program
    lea r15, [r14 + 1]
    mov r14, r15
    jmp scan

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero input_size
