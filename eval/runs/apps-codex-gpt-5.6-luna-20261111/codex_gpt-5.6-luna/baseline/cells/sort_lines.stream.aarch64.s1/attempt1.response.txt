.equ BUF_SIZE, 32768

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x20, records
    add x20, x20, :lo12:records

    mov x24, #0
read_loop:
    mov x0, #0
    add x1, x19, x24
    mov x2, #BUF_SIZE
    sub x2, x2, x24
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x24, x24, x0
    cmp x24, #BUF_SIZE
    b.lo read_loop

read_done:
    mov x21, #0
    mov x25, #0
    mov x26, #0

parse_loop:
    cmp x25, x24
    b.hs parse_end
    ldrb w0, [x19, x25]
    add x25, x25, #1
    cmp w0, #10
    b.ne parse_loop

    add x27, x20, x21, lsl #4
    str x26, [x27]
    sub x28, x25, x26
    sub x28, x28, #1
    str x28, [x27, #8]
    add x21, x21, #1
    mov x26, x25
    b parse_loop

parse_end:
    cmp x26, x24
    b.hs sort_start
    add x27, x20, x21, lsl #4
    str x26, [x27]
    sub x28, x24, x26
    str x28, [x27, #8]
    add x21, x21, #1

sort_start:
    mov x25, #1
sort_outer:
    cmp x25, x21
    b.hs output_start

    add x27, x20, x25, lsl #4
    ldr x22, [x27]
    ldr x23, [x27, #8]
    sub x26, x25, #1

sort_inner:
    cmp x26, #0
    b.lt insert_key

    add x27, x20, x26, lsl #4
    ldr x0, [x27]
    ldr x1, [x27, #8]
    add x0, x19, x0
    add x2, x19, x22
    mov x3, x23
    bl compare_lines
    cmp x0, #0
    ble insert_key

    add x28, x27, #16
    ldp x0, x1, [x27]
    stp x0, x1, [x28]
    sub x26, x26, #1
    b sort_inner

insert_key:
    add x28, x26, #1
    add x27, x20, x28, lsl #4
    str x22, [x27]
    str x23, [x27, #8]
    add x25, x25, #1
    b sort_outer

output_start:
    mov x25, #0

output_loop:
    cmp x25, x21
    b.hs program_exit
    add x27, x20, x25, lsl #4
    ldr x0, [x27]
    ldr x1, [x27, #8]
    add x0, x19, x0
    cbz x1, output_newline
    bl write_all

output_newline:
    adrp x0, newline
    add x0, x0, :lo12:newline
    mov x1, #1
    bl write_all
    add x25, x25, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

compare_lines:
    cmp x1, x3
    csel x5, x1, x3, lo
    mov x4, #0

compare_loop:
    cmp x4, x5
    b.hs compare_lengths
    ldrb w6, [x0, x4]
    ldrb w7, [x2, x4]
    cmp w6, w7
    b.ne compare_bytes
    add x4, x4, #1
    b compare_loop

compare_lengths:
    cmp x1, x3
    b.lo compare_less
    b.hi compare_greater
    mov x0, #0
    ret

compare_bytes:
    cmp w6, w7
    b.lo compare_less
    b compare_greater

compare_less:
    mov x0, #0
    sub x0, x0, #1
    ret

compare_greater:
    mov x0, #1
    ret

write_all:
    cbz x1, write_done
    mov x2, x1
    mov x8, #64
    mov x1, x0
    mov x0, #1
    svc #0
    cbz x0, write_done
    add x1, x1, x0
    sub x2, x2, x0
    mov x0, x1
    mov x1, x2
    b write_all

write_done:
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.balign 16
input_buf:
    .zero BUF_SIZE
.balign 16
records:
    .zero 16000
