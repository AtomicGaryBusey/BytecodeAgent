.equ input_size, 65536
.equ stack_size, 8192

.section .text
.globl _start
_start:
    xor ebx, ebx

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, input_size
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, input_size
    jb read_more

read_done:
    lea r12, [rip + input_buffer]
    lea r13, [rip + input_buffer]
    add r13, rbx
    lea r14, [rip + value_stack]

next_line:
    cmp r12, r13
    je program_exit
    xor r15d, r15d

token_loop:
skip_separators:
    cmp r12, r13
    je finish_line
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    cmp al, 10
    je finish_newline
    jmp scan_token

skip_one:
    inc r12
    jmp skip_separators

scan_token:
    mov rbx, r12

scan_token_loop:
    cmp r12, r13
    je token_scanned
    movzx eax, byte ptr [r12]
    cmp al, 32
    je token_scanned
    cmp al, 9
    je token_scanned
    cmp al, 10
    je token_scanned
    inc r12
    jmp scan_token_loop

token_scanned:
    mov rcx, r12
    sub rcx, rbx
    cmp rcx, 1
    jne maybe_number

    movzx eax, byte ptr [rbx]
    cmp al, '+'
    je do_add
    cmp al, '-'
    je do_sub
    cmp al, '*'
    je do_mul
    cmp al, '/'
    je do_div
    cmp al, '%'
    je do_mod

maybe_number:
    mov r9, rbx
    mov r8, rcx
    cmp byte ptr [r9], '-'
    jne number_digits
    inc r9
    dec r8
    cmp r8, 0
    je line_error

number_digits:
    cmp r8, 18
    ja line_error
    xor eax, eax
    xor edx, edx

parse_digits:
    cmp rdx, r8
    je number_done
    movzx ecx, byte ptr [r9 + rdx]
    cmp cl, '0'
    jb line_error
    cmp cl, '9'
    ja line_error
    imul rax, rax, 10
    sub ecx, '0'
    add rax, rcx
    inc rdx
    jmp parse_digits

number_done:
    cmp byte ptr [rbx], '-'
    jne push_number
    neg rax

push_number:
    mov [r14 + r15*8], rax
    inc r15
    jmp token_loop

do_add:
    mov r10d, 1
    jmp binary_op

do_sub:
    mov r10d, 2
    jmp binary_op

do_mul:
    mov r10d, 3
    jmp binary_op

do_div:
    mov r10d, 4
    jmp binary_op

do_mod:
    mov r10d, 5

binary_op:
    cmp r15, 2
    jb line_error
    dec r15
    mov rbx, [r14 + r15*8]
    dec r15
    mov rax, [r14 + r15*8]
    cmp r10d, 1
    je op_add
    cmp r10d, 2
    je op_sub
    cmp r10d, 3
    je op_mul
    cmp r10d, 4
    je op_div
    jmp op_mod

op_add:
    add rax, rbx
    jmp store_result

op_sub:
    sub rax, rbx
    jmp store_result

op_mul:
    imul rax, rbx
    jmp store_result

op_div:
    test rbx, rbx
    jz line_error
    mov r8, 0x8000000000000000
    cmp rax, r8
    jne divide_ok
    cmp rbx, -1
    je line_error

divide_ok:
    cqo
    idiv rbx
    jmp store_result

op_mod:
    test rbx, rbx
    jz line_error
    mov r8, 0x8000000000000000
    cmp rax, r8
    jne modulo_ok
    cmp rbx, -1
    je line_error

modulo_ok:
    cqo
    idiv rbx
    mov rax, rdx

store_result:
    mov [r14 + r15*8], rax
    inc r15
    jmp token_loop

finish_newline:
    inc r12

finish_line:
    cmp r15, 1
    jne line_error_output

    mov rax, [r14]
    lea rdi, [rip + output_buffer_end]
    mov byte ptr [rdi], 10
    mov r8, 10
    test rax, rax
    jns convert_positive
    neg rax
    mov ecx, 1
    jmp convert_digits

convert_positive:
    xor ecx, ecx

convert_digits:
    test rax, rax
    jnz convert_loop
    dec rdi
    mov byte ptr [rdi], '0'
    jmp add_sign

convert_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz convert_loop

add_sign:
    test ecx, ecx
    jz output_value
    dec rdi
    mov byte ptr [rdi], '-'

output_value:
    mov rsi, rdi
    lea rdx, [rip + output_buffer_end]
    sub rdx, rsi
    inc rdx
    call write_all
    jmp next_line

line_error:
    cmp r12, r13
    je line_error_output

skip_error_line:
    cmp r12, r13
    je line_error_output
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 10
    jne skip_error_line

line_error_output:
    lea rsi, [rip + error_message]
    mov edx, 6
    call write_all
    jmp next_line

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

write_all:
    mov r8, rsi
    mov r9, rdx

write_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    mov rdx, r9
    syscall
    test rax, rax
    jz write_loop
    add r8, rax
    sub r9, rax
    jnz write_loop
    ret

.section .rodata
error_message:
    .ascii "error\n"

.section .bss
.align 8
input_buffer:
    .zero input_size
value_stack:
    .zero stack_size
output_buffer:
    .zero 32
output_buffer_end:
    .zero 1
