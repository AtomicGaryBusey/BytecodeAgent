.equ buf_size, 65536

.section .text
.globl _start

_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, buf_size
    syscall
    test rax, rax
    jz output_result

    add r14, rax
    mov rbp, rax
    lea rbx, [rip + input_buffer]

process_loop:
    movzx eax, byte ptr [rbx]
    inc rbx
    dec rbp

    cmp al, 10
    je newline_byte
    cmp al, 9
    je whitespace_byte
    cmp al, 11
    je whitespace_byte
    cmp al, 12
    je whitespace_byte
    cmp al, 13
    je whitespace_byte
    cmp al, 32
    je whitespace_byte

    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

newline_byte:
    inc r12
    xor r15d, r15d
    jmp next_byte

whitespace_byte:
    xor r15d, r15d

next_byte:
    test rbp, rbp
    jnz process_loop
    jmp read_loop

output_result:
    lea rdi, [rip + output_buffer]
    mov rax, r12
    call emit_num
    mov byte ptr [rax], 32
    inc rax

    mov rdi, rax
    mov rax, r13
    call emit_num
    mov byte ptr [rax], 32
    inc rax

    mov rdi, rax
    mov rax, r14
    call emit_num
    mov byte ptr [rax], 10
    inc rax

    lea rsi, [rip + output_buffer]
    mov rdx, rax
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

emit_num:
    test rax, rax
    jnz emit_digits

    mov byte ptr [rdi], 48
    inc rdi
    mov rax, rdi
    ret

emit_digits:
    xor ecx, ecx
    mov r8d, 10

emit_divide:
    xor edx, edx
    div r8
    add dl, 48
    push rdx
    inc ecx
    test rax, rax
    jnz emit_divide

emit_write:
    pop rax
    mov byte ptr [rdi], al
    inc rdi
    loop emit_write

    mov rax, rdi
    ret

.section .bss
.align 16
input_buffer:
.zero buf_size

.align 16
output_buffer:
.zero 64
