.equ BUF_SIZE, 65536
.equ OUT_SIZE, 64

.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + input_buffer]

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jz finish
    js finish

    add r14, rax
    mov rdx, rax
    xor ecx, ecx

scan_loop:
    cmp rcx, rdx
    jae read_loop

    movzx eax, byte ptr [rbx + rcx]
    inc rcx

    cmp al, 10
    je newline

    cmp al, 32
    je whitespace
    cmp al, 9
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace

    test r15b, r15b
    jnz scan_loop
    inc r13
    mov r15b, 1
    jmp scan_loop

newline:
    inc r12

whitespace:
    xor r15d, r15d
    jmp scan_loop

finish:
    lea rdi, [rip + output_buffer + OUT_SIZE]
    mov rax, r14
    call decimal
    mov r8, rax

    dec r8
    mov byte ptr [r8], 32
    mov rdi, r8
    mov rax, r13
    call decimal
    mov r9, rax

    dec r9
    mov byte ptr [r9], 32
    mov rdi, r9
    mov rax, r12
    call decimal
    mov r10, rax

    lea r11, [rip + output_buffer + OUT_SIZE]

write_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r11
    sub rdx, r10
    syscall
    test rax, rax
    jle exit_program
    add r10, rax
    cmp r10, r11
    jb write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

decimal:
    mov r8d, 10
    test rax, rax
    jnz decimal_loop
    dec rdi
    mov byte ptr [rdi], 48
    mov rax, rdi
    ret

decimal_loop:
    xor edx, edx
    div r8
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz decimal_loop
    mov rax, rdi
    ret

.bss
.balign 8
input_buffer:
    .zero BUF_SIZE

output_buffer:
    .zero OUT_SIZE
