.equ BUFFER_SIZE, 65536

.globl _start

.section .text

_start:
    xor r12d, r12d

read_loop:
    cmp r12, BUFFER_SIZE
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, BUFFER_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r11d, r11d
    lea rsi, [rip + buffer]
    mov rcx, r12

parse_loop:
    test rcx, rcx
    jz format_output
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r10
    cmp al, 10
    jne check_space
    inc r8

check_space:
    cmp al, 32
    je whitespace
    cmp al, 9
    je whitespace
    cmp al, 10
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace
    test r11d, r11d
    jnz parse_loop
    inc r9
    mov r11d, 1
    jmp parse_loop

whitespace:
    xor r11d, r11d
    jmp parse_loop

format_output:
    lea rdi, [rip + output + 10]
    mov rax, r8
    call number_to_decimal
    mov r12, rax
    mov r15, rdi

    lea rdi, [rip + output + 25]
    mov rax, r9
    call number_to_decimal
    mov r13, rax
    mov r14, rdi

    lea rdi, [rip + output + 40]
    mov rax, r10
    call number_to_decimal
    mov rbx, rdi

    lea rdi, [rip + output]

    mov rsi, r12
    mov rdx, r15
    sub rdx, rsi
    mov rcx, rdx
    rep movsb

    mov byte ptr [rdi], 32
    inc rdi

    mov rsi, r13
    mov rdx, r14
    sub rdx, rsi
    mov rcx, rdx
    rep movsb

    mov byte ptr [rdi], 32
    inc rdi

    mov rsi, rbx
    lea rdx, [rip + output + 40]
    sub rdx, rsi
    mov rcx, rdx
    rep movsb

    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

number_to_decimal:
    mov ebx, 10

decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz decimal_loop
    mov rax, rdi
    ret

.section .bss
.balign 8
buffer:
    .zero BUFFER_SIZE
output:
    .zero 64
