.equ BUFFER_SIZE, 65536

.globl _start

.section .text

_start:
    xor r12d, r12d

read_loop:
    cmp r12, BUFFER_SIZE
    jae input_done
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov rdx, BUFFER_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    xor r11d, r11d
    lea rsi, [rip + buffer]
    mov rcx, r12

parse_loop:
    test rcx, rcx
    jz format_output
    movzx eax, byte ptr [rsi]
    inc rsi
    dec rcx
    inc r10
    cmp al, 10
    jne check_whitespace
    inc r8

check_whitespace:
    cmp al, 32
    je whitespace
    cmp al, 9
    je whitespace
    cmp al, 10
    je whitespace
    cmp al, 11
    je whitespace
    cmp al, 12
    je whitespace
    cmp al, 13
    je whitespace
    test r11d, r11d
    jnz parse_loop
    inc r9
    mov r11d, 1
    jmp parse_loop

whitespace:
    xor r11d, r11d
    jmp parse_loop

format_output:
    mov rax, r8
    lea rdi, [rip + number1 + 32]
    call decimal
    mov r12, rax

    mov rax, r9
    lea rdi, [rip + number2 + 32]
    call decimal
    mov r13, rax

    mov rax, r10
    lea rdi, [rip + number3 + 32]
    call decimal
    mov r14, rax

    lea rdi, [rip + output]

    lea rsi, [rip + number1 + 32]
    mov rdx, rsi
    sub rdx, r12
    mov rsi, r12
    mov rcx, rdx
    rep movsb

    mov byte ptr [rdi], 32
    inc rdi

    lea rsi, [rip + number2 + 32]
    mov rdx, rsi
    sub rdx, r13
    mov rsi, r13
    mov rcx, rdx
    rep movsb

    mov byte ptr [rdi], 32
    inc rdi

    lea rsi, [rip + number3 + 32]
    mov rdx, rsi
    sub rdx, r14
    mov rsi, r14
    mov rcx, rdx
    rep movsb

    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle exit_program
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

decimal:
    mov ebx, 10
    test rax, rax
    jnz decimal_loop
    dec rdi
    mov byte ptr [rdi], 48
    mov rax, rdi
    ret

decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz decimal_loop
    mov rax, rdi
    ret

.section .bss
.balign 8
buffer:
    .zero BUFFER_SIZE
number1:
    .zero 32
number2:
    .zero 32
number3:
    .zero 32
output:
    .zero 64
