.equ INPUT_SIZE, 65536

.section .text
.globl _start

_start:
    lea rbx, [rip + hex_digits]
    lea r13, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r13
    mov edx, INPUT_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz render_start
    add r12, rax
    add r13, rax
    jmp read_loop

render_start:
    lea r13, [rip + input_buffer]
    mov r14, r12
    xor r15d, r15d

row_loop:
    test r14, r14
    jz final_offset
    cmp r14, 16
    jbe row_size_ready
    mov r8d, 16
    jmp row_size_done

row_size_ready:
    mov r8d, r14d

row_size_done:
    lea rdi, [rip + line_buffer]
    mov eax, r15d
    mov ecx, 28
    mov r10, rdi

offset_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r10], dl
    inc r10
    sub ecx, 4
    jns offset_loop

    mov byte ptr [r10], 32
    inc r10
    mov byte ptr [r10], 32
    inc r10

    xor r11d, r11d

slot_loop:
    cmp r11d, 16
    jae slots_done
    cmp r11d, r8d
    jae empty_slot

    movzx eax, byte ptr [r13 + r11]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r10], dl
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r10 + 1], dl
    mov byte ptr [r10 + 2], 32
    jmp slot_done

empty_slot:
    mov byte ptr [r10], 32
    mov byte ptr [r10 + 1], 32
    mov byte ptr [r10 + 2], 32

slot_done:
    inc r11
    add r10, 3
    cmp r11d, 8
    jne slot_loop
    inc r10
    jmp slot_loop

slots_done:
    mov byte ptr [r10], 32
    inc r10
    mov byte ptr [r10], 124

    lea r10, [rdi + 61]
    xor r11d, r11d

ascii_loop:
    cmp r11d, r8d
    jae ascii_done
    movzx eax, byte ptr [r13 + r11]
    cmp al, 32
    jb nonprintable
    cmp al, 126
    ja nonprintable
    mov byte ptr [r10], al
    jmp ascii_stored

nonprintable:
    mov byte ptr [r10], 46

ascii_stored:
    inc r10
    inc r11
    jmp ascii_loop

ascii_done:
    mov byte ptr [r10], 124
    inc r10
    mov byte ptr [r10], 10

    add r13, r8
    sub r14, r8
    add r15, r8

    lea rsi, [rip + line_buffer]
    mov edx, 63
    add edx, r8d
    call write_all
    jmp row_loop

final_offset:
    lea rdi, [rip + line_buffer]
    mov eax, r12d
    mov ecx, 28
    mov r10, rdi

final_hex_loop:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [r10], dl
    inc r10
    sub ecx, 4
    jns final_hex_loop

    mov byte ptr [r10], 10
    lea rsi, [rip + line_buffer]
    mov edx, 9
    call write_all

    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_done

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop

write_done:
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 16
input_buffer:
    .zero INPUT_SIZE

.align 16
line_buffer:
    .zero 80
