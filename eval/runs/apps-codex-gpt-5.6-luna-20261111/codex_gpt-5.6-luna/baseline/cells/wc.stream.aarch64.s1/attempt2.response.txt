.equ BUF_SIZE, 65536

.section .bss
.balign 16
input_buffer:
	.zero BUF_SIZE
output_buffer:
	.zero 64

.text
.globl _start
_start:
	mov x19, #0
	mov x20, #0
	mov x21, #0
	mov x22, #0

	adrp x23, input_buffer
	add x23, x23, :lo12:input_buffer

read_again:
	mov x0, #0
	mov x1, x23
	mov x2, #BUF_SIZE
	mov x8, #63
	svc #0
	cbz x0, input_done

	add x21, x21, x0
	mov x25, x23
	add x26, x23, x0

process_byte:
	cmp x25, x26
	b.eq read_again
	ldrb w27, [x25], #1

	cmp w27, #10
	b.eq got_newline
	cmp w27, #32
	b.eq got_whitespace
	cmp w27, #9
	b.eq got_whitespace
	cmp w27, #11
	b.eq got_whitespace
	cmp w27, #12
	b.eq got_whitespace
	cmp w27, #13
	b.eq got_whitespace

	cbnz x22, process_byte
	add x20, x20, #1
	mov x22, #1
	b process_byte

got_newline:
	add x19, x19, #1
	mov x22, #0
	b process_byte

got_whitespace:
	mov x22, #0
	b process_byte

input_done:
	adrp x24, output_buffer
	add x24, x24, :lo12:output_buffer

	mov x0, x19
	mov x1, x24
	bl decimal

	mov w2, #32
	strb w2, [x0], #1

	mov x0, x20
	mov x1, x0
	mov x1, x24
	bl decimal

	mov w2, #32
	strb w2, [x0], #1

	mov x0, x21
	mov x1, x0
	mov x1, x24
	bl decimal

	mov w2, #10
	strb w2, [x0], #1
	mov x25, x0

	mov x1, x24
	sub x2, x25, x24

write_again:
	mov x0, #1
	mov x8, #64
	svc #0
	add x1, x1, x0
	sub x2, x2, x0
	cbnz x2, write_again

	mov x0, #0
	mov x8, #93
	svc #0

decimal:
	mov x9, x0
	mov x10, x1
	mov x11, x1
	cbnz x9, decimal_loop

	mov w14, #48
	strb w14, [x11], #1
	b decimal_reverse_start

decimal_loop:
	mov x12, #10
	udiv x13, x9, x12
	msub x14, x13, x12, x9
	add w14, w14, #48
	strb w14, [x11], #1
	mov x9, x13
	cbnz x9, decimal_loop

decimal_reverse_start:
	mov x15, x10
	sub x16, x11, #1

decimal_reverse:
	cmp x15, x16
	b.hs decimal_done
	ldrb w14, [x15]
	ldrb w17, [x16]
	strb w17, [x15]
	strb w14, [x16]
	add x15, x15, #1
	sub x16, x16, #1
	b decimal_reverse

decimal_done:
	mov x0, x11
	ret
