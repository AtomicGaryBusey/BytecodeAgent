.equ BUF_SIZE, 65536

.section .rodata
dec_divisors:
	.quad 10000
	.quad 1000
	.quad 100
	.quad 10
	.quad 1

.section .bss
.balign 16
input_buffer:
	.zero BUF_SIZE
output_buffer:
	.zero 64

.text
.globl _start
_start:
	mov x19, #0
	mov x20, #0
	mov x21, #0
	mov x22, #0

	adrp x23, input_buffer
	add x23, x23, :lo12:input_buffer

read_again:
	mov x0, #0
	mov x1, x23
	mov x2, #BUF_SIZE
	mov x8, #63
	svc #0
	cbz x0, input_done

	add x21, x21, x0
	mov x25, x23
	add x26, x23, x0

process_byte:
	cmp x25, x26
	b.eq read_again
	ldrb w27, [x25], #1

	cmp w27, #10
	b.eq got_newline
	cmp w27, #32
	b.eq got_whitespace
	cmp w27, #9
	b.eq got_whitespace
	cmp w27, #11
	b.eq got_whitespace
	cmp w27, #12
	b.eq got_whitespace
	cmp w27, #13
	b.eq got_whitespace

	cbnz x22, process_byte
	add x20, x20, #1
	mov x22, #1
	b process_byte

got_newline:
	add x19, x19, #1
	mov x22, #0
	b process_byte

got_whitespace:
	mov x22, #0
	b process_byte

input_done:
	adrp x24, output_buffer
	add x24, x24, :lo12:output_buffer
	mov x1, x24
	mov x0, x19
	bl decimal

	mov w2, #32
	strb w2, [x0], #1
	mov x1, x0
	mov x0, x20
	bl decimal

	mov w2, #32
	strb w2, [x0], #1
	mov x1, x0
	mov x0, x21
	bl decimal

	mov w2, #10
	strb w2, [x0], #1
	mov x25, x0

	mov x1, x24
	sub x2, x25, x24

write_again:
	mov x0, #1
	mov x8, #64
	svc #0
	add x1, x1, x0
	sub x2, x2, x0
	cbnz x2, write_again

	mov x0, #0
	mov x8, #93
	svc #0

decimal:
	adrp x7, dec_divisors
	add x7, x7, :lo12:dec_divisors
	mov x4, #0
	mov x5, #0

decimal_loop:
	ldr x6, [x7, x4, lsl #3]
	udiv x8, x0, x6
	msub x0, x8, x6, x0
	cbnz x8, decimal_emit
	cbnz x5, decimal_next
	cmp x6, #1
	b.eq decimal_emit
	b decimal_next

decimal_emit:
	add x8, x8, #48
	strb w8, [x1], #1
	mov x5, #1

decimal_next:
	add x4, x4, #1
	cmp x4, #5
	b.ne decimal_loop
	mov x0, x1
	ret
