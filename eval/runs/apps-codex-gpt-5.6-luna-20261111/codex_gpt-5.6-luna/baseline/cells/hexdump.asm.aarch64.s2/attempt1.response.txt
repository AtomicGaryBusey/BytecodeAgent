.equ BUF_SIZE, 65536

.section .text
.align 4
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    adrp x26, hexchars
    add x26, x26, :lo12:hexchars

    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x20, x21
    mov x2, #BUF_SIZE
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cbz x0, process_start
    add x21, x21, x0
    b read_loop

process_start:
    mov x22, #0
    mov x23, #0

row_loop:
    cmp x22, x21
    b.hs final_offset

    sub x25, x21, x22
    mov x27, #16
    cmp x25, #16
    csel x25, x25, x27, lo

    mov x0, x24
    mov x1, x23
    mov x2, #28
    mov x6, #0

offset_digits:
    lsr x3, x1, x2
    and x3, x3, #15
    ldrb w4, [x26, x3]
    strb w4, [x0, x6]
    add x6, x6, #1
    subs x2, x2, #4
    b.ge offset_digits

    mov w4, #32
    strb w4, [x0, #8]
    strb w4, [x0, #9]

    mov x6, #0
    add x7, x0, #10

slot_loop:
    cmp x6, #8
    b.ne slot_after_extra
    mov w4, #32
    strb w4, [x7]
    add x7, x7, #1

slot_after_extra:
    cmp x6, #16
    b.hs slots_done
    cmp x6, x25
    b.hs pad_slot

    add x3, x22, x6
    ldrb w1, [x20, x3]
    lsr w2, w1, #4
    and w2, w2, #15
    ldrb w4, [x26, x2]
    strb w4, [x7], #1
    and w2, w1, #15
    ldrb w4, [x26, x2]
    strb w4, [x7], #1
    mov w4, #32
    strb w4, [x7], #1
    add x6, x6, #1
    b slot_loop

pad_slot:
    mov w4, #32
    strb w4, [x7], #1
    strb w4, [x7], #1
    strb w4, [x7], #1
    add x6, x6, #1
    b slot_loop

slots_done:
    mov w4, #32
    strb w4, [x0, #59]
    mov w4, #124
    strb w4, [x0, #60]

    add x7, x0, #61
    mov x6, #0

ascii_loop:
    cmp x6, x25
    b.hs ascii_done
    add x3, x22, x6
    ldrb w1, [x20, x3]
    cmp w1, #32
    b.lo ascii_dot
    cmp w1, #126
    b.hi ascii_dot
    strb w1, [x7, x6]
    add x6, x6, #1
    b ascii_loop

ascii_dot:
    mov w1, #46
    strb w1, [x7, x6]
    add x6, x6, #1
    b ascii_loop

ascii_done:
    add x3, x25, #61
    mov w4, #124
    strb w4, [x0, x3]
    add x3, x3, #1
    mov w4, #10
    strb w4, [x0, x3]

    add x1, x24, #0
    add x2, x25, #63
    bl write_all

    add x22, x22, x25
    add x23, x23, x25
    b row_loop

final_offset:
    mov x0, x24
    mov x1, x21
    mov x2, #28
    mov x6, #0

final_digits:
    lsr x3, x1, x2
    and x3, x3, #15
    ldrb w4, [x26, x3]
    strb w4, [x0, x6]
    add x6, x6, #1
    subs x2, x2, #4
    b.ge final_digits

    mov w4, #10
    strb w4, [x0, #8]
    mov x1, x24
    mov x2, #9
    bl write_all

    mov x0, #0
    mov x8, #93
    svc #0

write_all:
    mov x9, x1
    mov x10, x2

write_loop:
    cbz x10, write_done
    mov x0, #1
    mov x1, x9
    mov x2, x10
    mov x8, #64
    svc #0
    cbz x0, write_loop
    add x9, x9, x0
    sub x10, x10, x0
    b write_loop

write_done:
    ret

.section .rodata
.align 4
hexchars:
    .ascii "0123456789abcdef"

.section .bss
.align 4
input_buf:
    .zero BUF_SIZE
output_buf:
    .zero 80
