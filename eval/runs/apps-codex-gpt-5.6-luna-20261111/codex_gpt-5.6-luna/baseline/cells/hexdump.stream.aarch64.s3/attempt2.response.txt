.equ INPUT_SIZE, 65536

.section .rodata
hex_digits:
	.ascii "0123456789abcdef"

.section .bss
.align 4
input_buffer:
	.zero INPUT_SIZE
output_buffer:
	.zero 128

.text
.globl _start
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	adrp x23, output_buffer
	add x23, x23, :lo12:output_buffer
	movz x22, #1, lsl #16
	mov x20, #0

read_loop:
	cmp x20, x22
	beq process_input
	mov x0, #0
	add x1, x19, x20
	sub x2, x22, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	beq process_input
	add x20, x20, x0
	b read_loop

process_input:
	adrp x27, hex_digits
	add x27, x27, :lo12:hex_digits
	mov x21, #0

row_loop:
	cmp x21, x20
	bhs final_offset

	sub x24, x20, x21
	cmp x24, #16
	bls row_length_ready
	mov x24, #16

row_length_ready:
	mov x26, #0
	mov x25, #0

offset_loop:
	mov x30, #7
	sub x30, x30, x25
	lsl x30, x30, #2
	lsr x29, x21, x30
	and x29, x29, #15
	ldrb w28, [x27, x29]
	strb w28, [x23, x25]
	add x25, x25, #1
	cmp x25, #8
	blo offset_loop

	mov w28, #32
	strb w28, [x23, #8]
	strb w28, [x23, #9]
	mov x26, #10
	mov x25, #0

slot_loop:
	cmp x25, #16
	bhs chars_start
	cmp x25, x24
	bhs slot_padding

	add x30, x25, x25, lsl #1
	add x30, x30, #10
	cmp x25, #8
	blo slot_position_ready
	add x30, x30, #1

slot_position_ready:
	add x29, x19, x21
	add x29, x29, x25
	ldrb w28, [x29]
	lsr x29, x28, #4
	ldrb w29, [x27, x29]
	strb w29, [x23, x30]
	and x29, x28, #15
	ldrb w29, [x27, x29]
	add x30, x30, #1
	strb w29, [x23, x30]
	add x30, x30, #1
	mov w29, #32
	strb w29, [x23, x30]
	add x26, x30, #1
	b slot_done

slot_padding:
	add x29, x23, x26
	mov w28, #32
	strb w28, [x29]
	strb w28, [x29, #1]
	strb w28, [x29, #2]
	add x26, x26, #3

slot_done:
	cmp x25, #7
	bne slot_next
	mov w28, #32
	strb w28, [x23, x26]
	add x26, x26, #1

slot_next:
	add x25, x25, #1
	b slot_loop

chars_start:
	mov w28, #32
	strb w28, [x23, x26]
	add x26, x26, #1
	mov w28, #124
	strb w28, [x23, x26]
	add x26, x26, #1
	mov x25, #0

chars_loop:
	cmp x25, x24
	bhs chars_done
	add x29, x19, x21
	add x29, x29, x25
	ldrb w28, [x29]
	cmp w28, #32
	blo nonprintable
	cmp w28, #126
	bls printable
nonprintable:
	mov w28, #46
printable:
	strb w28, [x23, x26]
	add x26, x26, #1
	add x25, x25, #1
	b chars_loop

chars_done:
	mov w28, #124
	strb w28, [x23, x26]
	add x26, x26, #1
	mov w28, #10
	strb w28, [x23, x26]
	add x26, x26, #1

write_row:
	cmp x26, #0
	beq row_written
	mov x0, #1
	mov x1, x23
	mov x2, x26
	mov x8, #64
	svc #0
	add x1, x1, x0
	sub x2, x2, x0
	mov x26, x2
	b write_row

row_written:
	add x21, x21, x24
	b row_loop

final_offset:
	mov x25, #0

final_digit_loop:
	mov x30, #7
	sub x30, x30, x25
	lsl x30, x30, #2
	lsr x29, x20, x30
	and x29, x29, #15
	ldrb w28, [x27, x29]
	strb w28, [x23, x25]
	add x25, x25, #1
	cmp x25, #8
	blo final_digit_loop

	mov w28, #10
	strb w28, [x23, #8]

	mov x0, #1
	mov x1, x23
	mov x2, #9
	mov x8, #64
	svc #0

	mov x0, #0
	mov x8, #93
	svc #0
