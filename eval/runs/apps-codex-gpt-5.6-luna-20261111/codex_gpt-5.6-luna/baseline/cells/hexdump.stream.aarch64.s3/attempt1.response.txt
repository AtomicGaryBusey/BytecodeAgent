.equ INPUT_SIZE, 65536
.equ ROW_SIZE, 128

.section .rodata
hex_digits:
	.ascii "0123456789abcdef"

.section .bss
.align 4
input_buffer:
	.zero INPUT_SIZE
output_buffer:
	.zero ROW_SIZE

.text
.globl _start
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	mov x20, #0

read_loop:
	mov x0, #0
	add x1, x19, x20
	mov x2, #INPUT_SIZE
	sub x2, x2, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	beq process_input
	add x20, x20, x0
	cmp x20, #INPUT_SIZE
	blo read_loop

process_input:
	adrp x27, hex_digits
	add x27, x27, :lo12:hex_digits
	adrp x23, output_buffer
	add x23, x23, :lo12:output_buffer
	mov x21, #0

row_loop:
	cmp x21, x20
	bhs final_offset

	sub x24, x20, x21
	cmp x24, #16
	bls row_length_done
	mov x24, #16

row_length_done:
	mov x26, #0
	mov x28, x21

offset_digits:
	mov x29, #7
	sub x29, x29, x26
	lsl x29, x29, #2
	lsr x29, x28, x29
	and x29, x29, #15
	ldrb w30, [x27, x29]
	strb w30, [x23, x26]
	add x26, x26, #1
	cmp x26, #8
	blo offset_digits

	mov w30, #32
	strb w30, [x23, #8]
	strb w30, [x23, #9]

	mov x25, #0

hex_slots:
	cmp x25, x24
	bhs hex_padding
	add x29, x25, x25, lsl #1
	add x29, x29, #10
	cmp x25, #8
	blo slot_position_ready
	add x29, x29, #1

slot_position_ready:
	add x30, x19, x21
	add x30, x30, x25
	ldrb w28, [x30]
	lsr x30, x28, #4
	and x30, x30, #15
	ldrb w30, [x27, x30]
	strb w30, [x23, x29]
	and x30, x28, #15
	ldrb w30, [x27, x30]
	strb w30, [x23, x29, #1]
	mov w30, #32
	strb w30, [x23, x29, #2]
	b next_slot

hex_padding:
	add x29, x25, x25, lsl #1
	add x29, x29, #10
	cmp x25, #8
	blo padding_position_ready
	add x29, x29, #1

padding_position_ready:
	mov w30, #32
	strb w30, [x23, x29]
	strb w30, [x23, x29, #1]
	strb w30, [x23, x29, #2]

next_slot:
	add x25, x25, #1
	cmp x25, #16
	blo hex_slots

	mov w30, #32
	strb w30, [x23, #59]
	mov w30, #124
	strb w30, [x23, #60]

	mov x25, #0
	add x29, x24, #61

ascii_loop:
	cmp x25, x24
	bhs ascii_done
	add x30, x19, x21
	add x30, x30, x25
	ldrb w28, [x30]
	cmp w28, #32
	blo nonprintable
	cmp w28, #126
	bhi nonprintable
	mov w30, w28
	b store_ascii

nonprintable:
	mov w30, #46

store_ascii:
	strb w30, [x23, x29]
	add x25, x25, #1
	add x29, x29, #1
	b ascii_loop

ascii_done:
	mov w30, #124
	strb w30, [x23, x29]
	add x29, x29, #1
	mov w30, #10
	strb w30, [x23, x29]

	mov x0, x23
	add x1, x24, #63
	bl write_all

	add x21, x21, x24
	b row_loop

final_offset:
	mov x26, #0
	mov x28, x21

final_digits:
	mov x29, #7
	sub x29, x29, x26
	lsl x29, x29, #2
	lsr x29, x28, x29
	and x29, x29, #15
	ldrb w30, [x27, x29]
	strb w30, [x23, x26]
	add x26, x26, #1
	cmp x26, #8
	blo final_digits

	mov w30, #10
	strb w30, [x23, #8]
	mov x0, x23
	mov x1, #9
	bl write_all

	mov x0, #0
	mov x8, #93
	svc #0

write_all:
	mov x9, x0
	mov x10, x1

write_loop:
	cmp x10, #0
	beq write_done
	mov x0, #1
	mov x1, x9
	mov x2, x10
	mov x8, #64
	svc #0
	add x9, x9, x0
	sub x10, x10, x0
	b write_loop

write_done:
	ret
