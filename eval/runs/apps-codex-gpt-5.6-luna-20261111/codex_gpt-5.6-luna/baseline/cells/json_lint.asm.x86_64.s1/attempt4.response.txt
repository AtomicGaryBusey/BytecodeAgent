.globl _start

.section .rodata
true_l: .ascii "true"
false_l: .ascii "false"
null_l: .ascii "null"
ok_s: .ascii "ok\n"
ok_n = . - ok_s
err_s: .ascii "error at "
err_n = . - err_s

.section .bss
.align 8
buf: .zero 65536
len: .zero 8
cur: .zero 8
epos: .zero 8
out: .zero 128

.section .text
_start:
    lea r12, [rip + buf]
    xor ebx, ebx
read_loop:
    cmp ebx, 65536
    jae read_done
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 65536
    sub edx, ebx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov [rip + len], rbx
    mov qword ptr [rip + cur], 0
    call skip_ws
    call parse_value
    test eax, eax
    jnz bad
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jne bad_at_cur
    lea rsi, [rip + ok_s]
    mov edx, ok_n
    call write_all
    xor edi, edi
    mov eax, 60
    syscall

bad_at_cur:
    mov rax, [rip + cur]
    mov [rip + epos], rax
bad:
    lea rdi, [rip + out]
    lea rsi, [rip + err_s]
    mov ecx, err_n
1:
    test ecx, ecx
    jz 2f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec ecx
    jmp 1b
2:
    mov rax, [rip + epos]
    lea rdi, [rip + out + 127]
    xor ecx, ecx
    mov ebx, 10
    test rax, rax
    jnz 3f
    dec rdi
    mov byte ptr [rdi], '0'
    inc ecx
    jmp 4f
3:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc ecx
    test rax, rax
    jnz 3b
4:
    mov r8d, ecx
    mov rsi, rdi
    lea rdi, [rip + out + err_n]
5:
    test r8d, r8d
    jz 6f
    mov al, byte ptr [rsi]
    mov byte ptr [rdi], al
    inc rsi
    inc rdi
    dec r8d
    jmp 5b
6:
    mov byte ptr [rdi], 10
    lea rsi, [rip + out]
    mov edx, err_n
    add edx, ecx
    inc edx
    call write_all
    mov edi, 1
    mov eax, 60
    syscall

write_all:
    test rdx, rdx
    jz 2f
1:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz 1b
2:
    ret

fail:
    mov rax, [rip + cur]
    mov [rip + epos], rax
    mov eax, 1
    ret

skip_ws:
1:
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae 3f
    movzx ecx, byte ptr [r12 + rax]
    cmp ecx, 32
    je 2f
    cmp ecx, 9
    je 2f
    cmp ecx, 10
    je 2f
    cmp ecx, 13
    jne 3f
2:
    inc rax
    mov [rip + cur], rax
    jmp 1b
3:
    ret

parse_value:
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    movzx eax, byte ptr [r12 + rax]
    cmp al, '"'
    je parse_string
    cmp al, '['
    je parse_array
    cmp al, '{'
    je parse_object
    cmp al, 't'
    je literal_true
    cmp al, 'f'
    je literal_false
    cmp al, 'n'
    je literal_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp parse_number

literal_true:
    lea rsi, [rip + true_l]
    mov ecx, 4
    jmp literal
literal_false:
    lea rsi, [rip + false_l]
    mov ecx, 5
    jmp literal
literal_null:
    lea rsi, [rip + null_l]
    mov ecx, 4
literal:
    xor edx, edx
1:
    mov rax, [rip + cur]
    add rax, rdx
    cmp rax, [rip + len]
    jae 3f
    movzx eax, byte ptr [r12 + rax]
    cmp al, byte ptr [rsi + rdx]
    jne 3f
    inc edx
    cmp edx, ecx
    jb 1b
    mov rax, [rip + cur]
    add rax, rcx
    mov [rip + cur], rax
    xor eax, eax
    ret
3:
    mov rax, [rip + cur]
    add rax, rdx
    mov [rip + epos], rax
    mov eax, 1
    ret

parse_string:
    mov rax, [rip + cur]
    inc rax
    mov [rip + cur], rax
1:
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '"'
    je 8f
    cmp cl, 92
    je 2f
    cmp cl, 32
    jb fail
    inc rax
    mov [rip + cur], rax
    jmp 1b
2:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '"'
    je 7f
    cmp cl, 92
    je 7f
    cmp cl, '/'
    je 7f
    cmp cl, 'b'
    je 7f
    cmp cl, 'f'
    je 7f
    cmp cl, 'n'
    je 7f
    cmp cl, 'r'
    je 7f
    cmp cl, 't'
    je 7f
    cmp cl, 'u'
    jne fail
    inc rax
    mov [rip + cur], rax
    mov edx, 4
3:
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '0'
    jb 4f
    cmp cl, '9'
    jbe 6f
4:
    cmp cl, 'a'
    jb 5f
    cmp cl, 'f'
    jbe 6f
5:
    cmp cl, 'A'
    jb fail
    cmp cl, 'F'
    ja fail
6:
    inc rax
    mov [rip + cur], rax
    dec edx
    jnz 3b
    jmp 1b
7:
    inc rax
    mov [rip + cur], rax
    jmp 1b
8:
    inc rax
    mov [rip + cur], rax
    xor eax, eax
    ret

parse_number:
    mov rax, [rip + cur]
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '-'
    jne 1f
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
1:
    cmp cl, '0'
    je 2f
    cmp cl, '1'
    jb fail
    cmp cl, '9'
    ja fail
3:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae 9f
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '0'
    jb 4f
    cmp cl, '9'
    jbe 3b
    jmp 4f
2:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae 9f
    movzx ecx, byte ptr [r12 + rax]
4:
    cmp cl, '.'
    jne 6f
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
5:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae 9f
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '0'
    jb 6f
    cmp cl, '9'
    jbe 5b
6:
    cmp cl, 'e'
    je 7f
    cmp cl, 'E'
    jne 9f
7:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '+'
    je 8f
    cmp cl, '-'
    jne 10f
8:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae fail
    movzx ecx, byte ptr [r12 + rax]
10:
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
11:
    inc rax
    mov [rip + cur], rax
    cmp rax, [rip + len]
    jae 9f
    movzx ecx, byte ptr [r12 + rax]
    cmp cl, '0'
    jb 9f
    cmp cl, '9'
    jbe 11b
9:
    xor eax, eax
    ret

parse_array:
    inc qword ptr [rip + cur]
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    cmp byte ptr [r12 + rax], ']'
    je 4f
    call parse_value
    test eax, eax
    jnz 5f
1:
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    cmp byte ptr [r12 + rax], ']'
    je 4f
    cmp byte ptr [r12 + rax], ','
    jne fail
    inc rax
    mov [rip + cur], rax
    call skip_ws
    call parse_value
    test eax, eax
    jnz 5f
    jmp 1b
4:
    inc qword ptr [rip + cur]
    xor eax, eax
    ret
5:
    mov eax, 1
    ret

parse_object:
    inc qword ptr [rip + cur]
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    cmp byte ptr [r12 + rax], '}'
    je 4f
    cmp byte ptr [r12 + rax], '"'
    jne fail
    call parse_string
    test eax, eax
    jnz 5f
1:
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    cmp byte ptr [r12 + rax], ':'
    jne fail
    inc rax
    mov [rip + cur], rax
    call skip_ws
    call parse_value
    test eax, eax
    jnz 5f
2:
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    cmp byte ptr [r12 + rax], '}'
    je 4f
    cmp byte ptr [r12 + rax], ','
    jne fail
    inc rax
    mov [rip + cur], rax
    call skip_ws
    mov rax, [rip + cur]
    cmp rax, [rip + len]
    jae fail
    cmp byte ptr [r12 + rax], '"'
    jne fail
    call parse_string
    test eax, eax
    jnz 5f
    jmp 1b
4:
    inc qword ptr [rip + cur]
    xor eax, eax
    ret
5:
    mov eax, 1
    ret
