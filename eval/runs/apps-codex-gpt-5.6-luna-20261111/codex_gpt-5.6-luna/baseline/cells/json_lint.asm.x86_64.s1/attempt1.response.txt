.globl _start

.section .rodata
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
err_msg:
    .ascii "error at "
err_len = . - err_msg
newline:
    .ascii "\n"

.section .bss
.align 8
buffer:
    .zero 65536
buf_len:
    .zero 8
pos:
    .zero 8
err_pos:
    .zero 8
outbuf:
    .zero 128

.section .text

_start:
    xor eax, eax
    lea rsi, [rip + buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jz read_done
    mov [rip + buf_len], rax
    mov rbx, rax

read_more:
    cmp rbx, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    mov [rip + buf_len], rbx
    jmp read_more

read_done:
    mov qword ptr [rip + pos], 0
    call skip_ws
    call parse_value
    test eax, eax
    jnz invalid
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jne invalid_at_pos

    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    call write_all
    xor edi, edi
    mov eax, 60
    syscall

invalid_at_pos:
    mov rax, [rip + pos]
    mov [rip + err_pos], rax

invalid:
    lea rsi, [rip + outbuf]
    mov rdi, rsi
    lea rdx, [rip + err_msg]
    mov ecx, err_len
copy_err:
    test ecx, ecx
    jz make_number
    mov al, byte ptr [rdx]
    mov byte ptr [rdi], al
    inc rdx
    inc rdi
    dec ecx
    jmp copy_err

make_number:
    mov rax, [rip + err_pos]
    lea rdi, [rip + outbuf + 127]
    xor ecx, ecx
    mov ebx, 10
    test rax, rax
    jnz number_loop
    dec rdi
    mov byte ptr [rdi], '0'
    inc ecx
    jmp number_ready

number_loop:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc ecx
    test rax, rax
    jnz number_loop

number_ready:
    lea rsi, [rip + outbuf]
    lea rdx, [rip + err_msg]
    mov eax, err_len
    add rsi, rax
    sub rsi, rdi
    mov r8, rdi
    lea rdi, [rip + outbuf]
    mov eax, err_len
    add rdi, rax
    mov rax, rsi
    mov rsi, r8
    mov rdx, rax
    call write_all

    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    mov edi, 1
    mov eax, 60
    syscall

write_all:
    test rdx, rdx
    jz write_done
write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    test rax, rax
    jle write_done
    add rsi, rax
    sub rdx, rax
    jnz write_loop
write_done:
    ret

fail:
    mov rax, [rip + pos]
    mov [rip + err_pos], rax
    mov eax, 1
    ret

skip_ws:
skip_ws_loop:
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae skip_ws_done
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp ecx, 32
    je skip_ws_inc
    cmp ecx, 9
    je skip_ws_inc
    cmp ecx, 10
    je skip_ws_inc
    cmp ecx, 13
    je skip_ws_inc
    ret
skip_ws_inc:
    inc rax
    mov [rip + pos], rax
    jmp skip_ws_loop
skip_ws_done:
    ret

parse_value:
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je parse_string
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, 't'
    je value_true
    cmp al, 'f'
    je value_false
    cmp al, 'n'
    je value_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp parse_number

value_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp parse_literal
value_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp parse_literal
value_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4

parse_literal:
    xor edx, edx
literal_loop:
    mov rax, [rip + pos]
    add rax, rdx
    cmp rax, [rip + buf_len]
    jae literal_bad
    lea rdi, [rip + buffer]
    add rdi, rax
    movzx eax, byte ptr [rdi]
    cmp al, byte ptr [rsi + rdx]
    jne literal_bad
    inc rdx
    cmp edx, ecx
    jb literal_loop
    mov rax, [rip + pos]
    add rax, rcx
    mov [rip + pos], rax
    xor eax, eax
    ret
literal_bad:
    mov rax, [rip + pos]
    add rax, rdx
    mov [rip + err_pos], rax
    mov eax, 1
    ret

parse_string:
    mov rax, [rip + pos]
    inc rax
    mov [rip + pos], rax
string_loop:
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '"'
    je string_done
    cmp cl, '\'
    je string_escape
    cmp cl, 32
    jb fail
    inc rax
    mov [rip + pos], rax
    jmp string_loop

string_done:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret

string_escape:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '"'
    je simple_escape
    cmp cl, '\'
    je simple_escape
    cmp cl, '/'
    je simple_escape
    cmp cl, 'b'
    je simple_escape
    cmp cl, 'f'
    je simple_escape
    cmp cl, 'n'
    je simple_escape
    cmp cl, 'r'
    je simple_escape
    cmp cl, 't'
    je simple_escape
    cmp cl, 'u'
    jne fail
    inc rax
    mov [rip + pos], rax
    mov edx, 4
unicode_loop:
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb unicode_alpha
    cmp cl, '9'
    jbe unicode_digit
unicode_alpha:
    cmp cl, 'a'
    jb unicode_upper
    cmp cl, 'f'
    jbe unicode_ok
unicode_upper:
    cmp cl, 'A'
    jb fail
    cmp cl, 'F'
    ja fail
unicode_ok:
unicode_digit:
    inc rax
    mov [rip + pos], rax
    dec edx
    jnz unicode_loop
    jmp string_loop
simple_escape:
    inc rax
    mov [rip + pos], rax
    jmp string_loop

parse_number:
    mov rax, [rip + pos]
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '-'
    jne number_digits
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail

number_digits:
    mov rax, [rip + pos]
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jne number_nonzero
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae number_end
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb number_fraction
    cmp cl, '9'
    jbe fail
    jmp number_fraction

number_nonzero:
    cmp cl, '1'
    jb fail
    cmp cl, '9'
    ja fail
number_digit_loop:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae number_end
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb number_fraction
    cmp cl, '9'
    jbe number_digit_loop

number_fraction:
    cmp cl, '.'
    jne number_exponent
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
fraction_loop:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae number_end
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb number_exponent
    cmp cl, '9'
    jbe fraction_loop

number_exponent:
    cmp cl, 'e'
    je exponent_start
    cmp cl, 'E'
    jne number_end
exponent_start:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '+'
    je exponent_sign
    cmp cl, '-'
    jne exponent_digit
exponent_sign:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
exponent_digit:
    cmp cl, '0'
    jb fail
    cmp cl, '9'
    ja fail
exponent_loop:
    inc rax
    mov [rip + pos], rax
    cmp rax, [rip + buf_len]
    jae number_end
    lea rsi, [rip + buffer]
    add rsi, rax
    movzx ecx, byte ptr [rsi]
    cmp cl, '0'
    jb number_end
    cmp cl, '9'
    jbe exponent_loop

number_end:
    xor eax, eax
    ret

parse_array:
    mov rax, [rip + pos]
    inc rax
    mov [rip + pos], rax
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    cmp byte ptr [rsi], ']'
    je array_done
    call parse_value
    test eax, eax
    jnz array_fail
array_loop:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    cmp byte ptr [rsi], ']'
    je array_done
    cmp byte ptr [rsi], ','
    jne fail
    inc rax
    mov [rip + pos], rax
    call skip_ws
    call parse_value
    test eax, eax
    jnz array_fail
    jmp array_loop
array_done:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret
array_fail:
    mov eax, 1
    ret

parse_object:
    mov rax, [rip + pos]
    inc rax
    mov [rip + pos], rax
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    cmp byte ptr [rsi], '}'
    je object_done
    call parse_string
    test eax, eax
    jnz object_fail
object_member:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    cmp byte ptr [rsi], ':'
    jne fail
    inc rax
    mov [rip + pos], rax
    call skip_ws
    call parse_value
    test eax, eax
    jnz object_fail
object_loop:
    call skip_ws
    mov rax, [rip + pos]
    cmp rax, [rip + buf_len]
    jae fail
    lea rsi, [rip + buffer]
    add rsi, rax
    cmp byte ptr [rsi], '}'
    je object_done
    cmp byte ptr [rsi], ','
    jne fail
    inc rax
    mov [rip + pos], rax
    call skip_ws
    call parse_string
    test eax, eax
    jnz object_fail
    jmp object_member
object_done:
    inc rax
    mov [rip + pos], rax
    xor eax, eax
    ret
object_fail:
    mov eax, 1
    ret
