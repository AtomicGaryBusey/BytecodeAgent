.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000
.equ LINE_SIZE, 76

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
input_buffer:
.zero INPUT_SIZE
output_buffer:
.zero OUTPUT_SIZE

.text
.globl _start
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	mov x20, #0

read_loop:
	cmp x20, #INPUT_SIZE
	b.eq input_done
	mov x0, #0
	add x1, x19, x20
	mov x2, #INPUT_SIZE
	sub x2, x2, x20
	mov x8, #SYS_READ
	svc #0
	cbz x0, input_done
	add x20, x20, x0
	b read_loop

input_done:
	adrp x24, alphabet
	add x24, x24, :lo12:alphabet
	adrp x25, output_buffer
	add x25, x25, :lo12:output_buffer
	mov x21, #0
	mov x22, #0
	mov x23, #0

encode_loop:
	sub x0, x20, x21
	cmp x0, #3
	b.lo remainder

	add x1, x19, x21
	ldrb w4, [x1]
	ldrb w5, [x1, #1]
	ldrb w11, [x1, #2]
	add x21, x21, #3

	lsr w6, w4, #2
	and w7, w4, #3
	lsl w7, w7, #4
	lsr w12, w5, #4
	orr w7, w7, w12
	and w8, w5, #15
	lsl w8, w8, #2
	lsr w13, w11, #6
	orr w8, w8, w13
	and w9, w11, #63

	ldrb w10, [x24, w6, uxtw]
	bl emit_char
	ldrb w10, [x24, w7, uxtw]
	bl emit_char
	ldrb w10, [x24, w8, uxtw]
	bl emit_char
	ldrb w10, [x24, w9, uxtw]
	bl emit_char
	b encode_loop

remainder:
	cbz x0, finish_output
	cmp x0, #1
	b.eq remainder_one

	add x1, x19, x21
	ldrb w4, [x1]
	ldrb w5, [x1, #1]
	lsr w6, w4, #2
	and w7, w4, #3
	lsl w7, w7, #4
	lsr w12, w5, #4
	orr w7, w7, w12
	and w8, w5, #15
	lsl w8, w8, #2

	ldrb w10, [x24, w6, uxtw]
	bl emit_char
	ldrb w10, [x24, w7, uxtw]
	bl emit_char
	ldrb w10, [x24, w8, uxtw]
	bl emit_char
	mov w10, #61
	bl emit_char
	b finish_output

remainder_one:
	add x1, x19, x21
	ldrb w4, [x1]
	lsr w6, w4, #2
	and w7, w4, #3
	lsl w7, w7, #4

	ldrb w10, [x24, w6, uxtw]
	bl emit_char
	ldrb w10, [x24, w7, uxtw]
	bl emit_char
	mov w10, #61
	bl emit_char
	mov w10, #61
	bl emit_char

finish_output:
	cbz x23, write_output
	mov w10, #10
	strb w10, [x25, x22]
	add x22, x22, #1
	mov x23, #0

write_output:
	mov x21, #0

write_loop:
	cmp x21, x22
	b.eq program_exit
	mov x0, #1
	add x1, x25, x21
	sub x2, x22, x21
	mov x8, #SYS_WRITE
	svc #0
	add x21, x21, x0
	b write_loop

program_exit:
	mov x0, #0
	mov x8, #SYS_EXIT
	svc #0

emit_char:
	strb w10, [x25, x22]
	add x22, x22, #1
	add x23, x23, #1
	cmp x23, #LINE_SIZE
	b.ne emit_return
	mov w10, #10
	strb w10, [x25, x22]
	add x22, x22, #1
	mov x23, #0

emit_return:
	ret
