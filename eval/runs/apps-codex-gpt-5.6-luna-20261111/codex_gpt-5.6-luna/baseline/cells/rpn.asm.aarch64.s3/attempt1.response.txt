.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ INPUT_SIZE, 65536

.section .rodata
error_msg:
    .ascii "error\n"
error_len = . - error_msg

.section .bss
    .align 4
input_buf:
    .zero INPUT_SIZE
eval_stack:
    .zero 8192
out_buf:
    .zero 32

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    adrp x28, eval_stack
    add x28, x28, :lo12:eval_stack
    mov x20, #0

read_more:
    mov x0, #0
    add x1, x19, x20
    mov x2, #INPUT_SIZE
    sub x2, x2, x20
    mov x8, #SYS_READ
    svc #0
    cmp x0, #0
    beq input_done
    add x20, x20, x0
    b read_more

input_done:
    mov x24, #0
    mov x25, #0

scan_input:
    cmp x24, x20
    beq final_remainder
    ldrb w0, [x19, x24]
    cmp w0, #10
    beq found_newline
    add x24, x24, #1
    b scan_input

found_newline:
    add x0, x19, x25
    sub x1, x24, x25
    bl process_line
    add x24, x24, #1
    mov x25, x24
    b scan_input

final_remainder:
    cmp x24, x25
    beq program_exit
    add x0, x19, x25
    sub x1, x24, x25
    bl process_line

program_exit:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

write_all:
    cbz x1, write_done
write_loop:
    mov x2, x1
    mov x1, x0
    mov x0, #1
    mov x8, #SYS_WRITE
    svc #0
    add x1, x1, x0
    sub x2, x2, x0
    mov x0, x1
    mov x1, x2
    cbnz x1, write_loop
write_done:
    ret

process_line:
    add x20, x0, x1
    mov x21, x0
    mov x22, #0

next_token:
    cmp x21, x20
    beq line_finish
    ldrb w0, [x21]
    cmp w0, #32
    beq skip_separator
    cmp w0, #9
    beq skip_separator
    mov x24, x21

find_token_end:
    cmp x21, x20
    beq token_ended
    ldrb w0, [x21]
    cmp w0, #32
    beq token_ended
    cmp w0, #9
    beq token_ended
    add x21, x21, #1
    b find_token_end

token_ended:
    sub x23, x21, x24
    ldrb w26, [x24]
    cmp x23, #1
    bne not_single_operator
    cmp w26, #43
    beq operator_ready
    cmp w26, #45
    beq operator_ready
    cmp w26, #42
    beq operator_ready
    cmp w26, #47
    beq operator_ready
    cmp w26, #37
    beq operator_ready

not_single_operator:
    cmp w26, #45
    beq negative_literal
    cmp x23, #18
    bhi token_error
    mov x24, x21
    sub x24, x24, x23
    mov x25, #0
    mov x27, #0
    b parse_digits

negative_literal:
    cmp x23, #1
    beq operator_ready
    cmp x23, #19
    bhi token_error
    add x24, x24, #1
    sub x27, x23, #1
    mov x25, #0

parse_negative_digits:
    cbz x27, negative_done
    ldrb w0, [x24]
    cmp w0, #48
    blo token_error
    cmp w0, #57
    bhi token_error
    sub w0, w0, #48
    add x25, x25, x25, lsl #3
    add x25, x25, x25
    add x25, x25, x0
    add x24, x24, #1
    sub x27, x27, #1
    b parse_negative_digits

negative_done:
    neg x25, x25
    str x25, [x28, x22, lsl #3]
    add x22, x22, #1
    b next_token

parse_digits:
    cmp x27, x23
    beq digits_done
    ldrb w0, [x24, x27]
    cmp w0, #48
    blo token_error
    cmp w0, #57
    bhi token_error
    sub w0, w0, #48
    add x25, x25, x25, lsl #3
    add x25, x25, x25
    add x25, x25, x0
    add x27, x27, #1
    b parse_digits

digits_done:
    str x25, [x28, x22, lsl #3]
    add x22, x22, #1
    b next_token

operator_ready:
    cmp x22, #2
    blo token_error
    sub x3, x22, #1
    ldr x1, [x28, x3, lsl #3]
    sub x3, x22, #2
    ldr x2, [x28, x3, lsl #3]

    cmp w26, #43
    beq do_add
    cmp w26, #45
    beq do_sub
    cmp w26, #42
    beq do_mul
    cmp w26, #47
    beq do_div

    cmp x1, #0
    beq token_error
    mov x0, x2
    sdiv x3, x2, x1
    msub x0, x3, x1, x2
    b store_operation

do_add:
    add x0, x2, x1
    b store_operation

do_sub:
    sub x0, x2, x1
    b store_operation

do_mul:
    mul x0, x2, x1
    b store_operation

do_div:
    cmp x1, #0
    beq token_error
    movz x3, #0
    movk x3, #0x8000, lsl #48
    cmp x2, x3
    bne division_safe
    cmn x1, #1
    beq token_error
division_safe:
    sdiv x0, x2, x1

store_operation:
    sub x3, x22, #2
    str x0, [x28, x3, lsl #3]
    sub x22, x22, #1
    b next_token

skip_separator:
    add x21, x21, #1
    b next_token

line_finish:
    cmp x22, #1
    bne token_error
    ldr x24, [x28]
    adrp x27, out_buf
    add x27, x27, :lo12:out_buf
    add x25, x27, #32
    sub x26, x25, #1
    mov w0, #10
    strb w0, [x26]
    mov x0, x24
    mov x1, #0
    cmp x0, #0
    bge magnitude_ready
    mov x1, #1
    neg x0, x0

magnitude_ready:
    mov x24, x0
    mov x27, #10

decimal_loop:
    udiv x0, x24, x27
    msub x3, x0, x27, x24
    add w3, w3, #48
    sub x26, x26, #1
    strb w3, [x26]
    mov x24, x0
    cbnz x24, decimal_loop

    cbz x1, sign_done
    sub x26, x26, #1
    mov w0, #45
    strb w0, [x26]

sign_done:
    mov x0, x26
    sub x1, x25, x26
    bl write_all
    ret

token_error:
    adrp x0, error_msg
    add x0, x0, :lo12:error_msg
    mov x1, #error_len
    bl write_all
    ret
