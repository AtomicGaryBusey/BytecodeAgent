.globl _start

.section .text
_start:
    lea rsi, [rip + inputbuf]
    xor rbx, rbx

read_more:
    mov rdx, 65536
    sub rdx, rbx
    lea rsi, [rip + inputbuf]
    add rsi, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_more

read_done:
    mov r9, rbx
    lea r15, [rip + inputbuf]
    mov r12, r9
    mov r13, r9
    xor rbx, rbx

find_delimiter:
    cmp rbx, r9
    jae delimiter_done
    cmp byte ptr [r15 + rbx], 33
    je delimiter_found
    inc rbx
    jmp find_delimiter

delimiter_found:
    mov r12, rbx
    inc rbx
    mov r13, rbx

delimiter_done:
    lea rdi, [rip + bracket_stack]
    lea rsi, [rip + jump_table]
    xor r8, r8
    xor rbx, rbx

validate_loop:
    cmp rbx, r12
    jae validate_done
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rbx
    jmp validate_loop

validate_open:
    mov dword ptr [rdi + r8*4], ebx
    inc r8
    inc rbx
    jmp validate_loop

validate_close:
    test r8, r8
    jz invalid_program
    dec r8
    mov eax, dword ptr [rdi + r8*4]
    mov ecx, ebx
    mov dword ptr [rsi + rax*4], ecx
    mov dword ptr [rsi + rcx*4], eax
    inc rbx
    jmp validate_loop

validate_done:
    test r8, r8
    jnz invalid_program

    lea r14, [rip + tape]
    xor rbx, rbx
    xor r8, r8

execute_loop:
    cmp rbx, r12
    jae program_done
    movzx eax, byte ptr [r15 + rbx]

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close

    inc rbx
    jmp execute_loop

command_right:
    inc r14
    inc rbx
    jmp execute_loop

command_left:
    dec r14
    inc rbx
    jmp execute_loop

command_plus:
    inc byte ptr [r14]
    inc rbx
    jmp execute_loop

command_minus:
    dec byte ptr [r14]
    inc rbx
    jmp execute_loop

command_output:
    mov al, byte ptr [r14]
    mov byte ptr [rip + outputbuf + r8], al
    inc r8
    inc rbx
    jmp execute_loop

command_input:
    cmp r13, r9
    jae input_exhausted
    mov al, byte ptr [r15 + r13]
    inc r13
    mov byte ptr [r14], al
    inc rbx
    jmp execute_loop

input_exhausted:
    mov byte ptr [r14], 0
    inc rbx
    jmp execute_loop

command_open:
    cmp byte ptr [r14], 0
    jne open_nonzero
    mov ebx, dword ptr [rsi + rbx*4]
    inc rbx
    jmp execute_loop

open_nonzero:
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [r14], 0
    je close_zero
    mov ebx, dword ptr [rsi + rbx*4]
    inc rbx
    jmp execute_loop

close_zero:
    inc rbx
    jmp execute_loop

invalid_program:
    mov edi, 1
    mov eax, 60
    syscall

program_done:
    mov r12, r8
    xor rbx, rbx

write_loop:
    cmp rbx, r12
    jae success_exit
    mov edi, 1
    lea rsi, [rip + outputbuf]
    add rsi, rbx
    mov rdx, r12
    sub rdx, rbx
    mov eax, 1
    syscall
    test rax, rax
    jz write_loop
    add rbx, rax
    jmp write_loop

success_exit:
    xor edi, edi
    mov eax, 60
    syscall

.section .bss
.align 8
inputbuf:
    .zero 65536
outputbuf:
    .zero 65536
tape:
    .zero 30000
.align 8
jump_table:
    .zero 262144
bracket_stack:
    .zero 4096
