.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ JUMP_SIZE, 262144
.equ STACK_SIZE, 4000
.equ OUTPUT_SIZE, 65536

.globl _start

.section .text
_start:
    lea r15, [rip + inputbuf]
    xor r12, r12

read_more:
    cmp r12, INPUT_SIZE
    jae read_done
    mov edx, INPUT_SIZE
    sub rdx, r12
    lea rsi, [rip + inputbuf]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    mov r13, r12
    mov r10, r12
    xor rbx, rbx

find_delimiter:
    cmp rbx, r12
    jae delimiter_done
    cmp byte ptr [r15 + rbx], 33
    je delimiter_found
    inc rbx
    jmp find_delimiter

delimiter_found:
    mov r13, rbx
    lea r10, [rbx + 1]

delimiter_done:
    lea rdi, [rip + bracket_stack]
    lea rsi, [rip + jump_table]
    xor r8, r8
    xor rbx, rbx

validate_loop:
    cmp rbx, r13
    jae validate_done
    movzx eax, byte ptr [r15 + rbx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rbx
    jmp validate_loop

validate_open:
    mov dword ptr [rdi + r8*4], ebx
    inc r8
    inc rbx
    jmp validate_loop

validate_close:
    test r8, r8
    jz invalid_program
    dec r8
    mov eax, dword ptr [rdi + r8*4]
    mov ecx, ebx
    mov dword ptr [rsi + rax*4], ecx
    mov dword ptr [rsi + rcx*4], eax
    inc rbx
    jmp validate_loop

validate_done:
    test r8, r8
    jnz invalid_program

    lea r14, [rip + tape]
    lea rdi, [rip + outputbuf]
    xor rbx, rbx
    xor r9, r9

execute_loop:
    cmp rbx, r13
    jae program_done
    movzx eax, byte ptr [r15 + rbx]

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close

    inc rbx
    jmp execute_loop

command_right:
    inc r14
    inc rbx
    jmp execute_loop

command_left:
    dec r14
    inc rbx
    jmp execute_loop

command_plus:
    inc byte ptr [r14]
    inc rbx
    jmp execute_loop

command_minus:
    dec byte ptr [r14]
    inc rbx
    jmp execute_loop

command_output:
    mov al, byte ptr [r14]
    mov byte ptr [rdi + r9], al
    inc r9
    inc rbx
    jmp execute_loop

command_input:
    cmp r10, r12
    jae input_empty
    mov al, byte ptr [r15 + r10]
    mov byte ptr [r14], al
    inc r10
    inc rbx
    jmp execute_loop

input_empty:
    mov byte ptr [r14], 0
    inc rbx
    jmp execute_loop

command_open:
    cmp byte ptr [r14], 0
    jne open_continue
    lea rax, [rip + jump_table]
    mov eax, dword ptr [rax + rbx*4]
    inc eax
    mov ebx, eax
    jmp execute_loop

open_continue:
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [r14], 0
    je close_continue
    lea rax, [rip + jump_table]
    mov eax, dword ptr [rax + rbx*4]
    inc eax
    mov ebx, eax
    jmp execute_loop

close_continue:
    inc rbx
    jmp execute_loop

program_done:
    test r9, r9
    jz exit_success
    lea rsi, [rip + outputbuf]

write_loop:
    test r9, r9
    jz exit_success
    mov eax, 1
    mov edi, 1
    mov rdx, r9
    syscall
    test rax, rax
    jz exit_success
    sub r9, rax
    add rsi, rax
    jmp write_loop

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

exit_success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 8
inputbuf:
.zero INPUT_SIZE

.align 8
tape:
.zero TAPE_SIZE

.align 8
jump_table:
.zero JUMP_SIZE

.align 8
bracket_stack:
.zero STACK_SIZE

.align 8
outputbuf:
.zero OUTPUT_SIZE
