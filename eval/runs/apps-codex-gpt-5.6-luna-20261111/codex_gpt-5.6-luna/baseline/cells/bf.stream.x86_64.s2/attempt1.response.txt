.globl _start

.section .text
_start:
    lea rsi, [rip + inputbuf]
    xor rbx, rbx
read_more:
    mov rdx, 65536
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_more

read_done:
    mov r9, rbx
    mov r12, r9
    mov r13, r9
    xor rbx, rbx
find_delimiter:
    cmp rbx, r9
    jae delimiter_done
    cmp byte ptr [rsi + rbx], 33
    je delimiter_found
    inc rbx
    jmp find_delimiter

delimiter_found:
    mov r12, rbx
    lea r13, [rbx + 1]

delimiter_done:
    mov r14, r9
    lea rdi, [rip + bracket_stack]
    lea rsi, [rip + jump_table]
    xor r8, r8
    xor rbx, rbx

validate_loop:
    cmp rbx, r12
    jae validate_done
    movzx eax, byte ptr [rip + inputbuf + rbx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    inc rbx
    jmp validate_loop

open_bracket:
    cmp r8, 1000
    jae invalid_program
    mov qword ptr [rdi + r8*8], rbx
    inc r8
    inc rbx
    jmp validate_loop

close_bracket:
    test r8, r8
    jz invalid_program
    dec r8
    mov rax, qword ptr [rdi + r8*8]
    mov qword ptr [rsi + rax*8], rbx
    mov qword ptr [rsi + rbx*8], rax
    inc rbx
    jmp validate_loop

validate_done:
    test r8, r8
    jnz invalid_program

    lea r10, [rip + inputbuf]
    lea r8, [rip + inputbuf]
    lea r9, [rip + jump_table]
    lea r15, [rip + tape]
    xor rbx, rbx

execute_loop:
    cmp rbx, r12
    jae success
    movzx eax, byte ptr [r10 + rbx]
    inc rbx

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close
    jmp execute_loop

command_right:
    inc r15
    jmp execute_loop

command_left:
    dec r15
    jmp execute_loop

command_plus:
    inc byte ptr [r15]
    jmp execute_loop

command_minus:
    dec byte ptr [r15]
    jmp execute_loop

command_output:
    mov al, byte ptr [r15]
    mov byte ptr [rip + outbyte], al
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbyte]
    mov edx, 1
    syscall
    jmp execute_loop

command_input:
    cmp r13, r14
    jae input_exhausted
    mov al, byte ptr [r8 + r13]
    inc r13
    mov byte ptr [r15], al
    jmp execute_loop

input_exhausted:
    mov byte ptr [r15], 0
    jmp execute_loop

command_open:
    cmp byte ptr [r15], 0
    jne execute_loop
    mov rbx, qword ptr [r9 + rbx*8]
    inc rbx
    jmp execute_loop

command_close:
    cmp byte ptr [r15], 0
    je execute_loop
    mov rbx, qword ptr [r9 + rbx*8]
    inc rbx
    jmp execute_loop

invalid_program:
    mov eax, 60
    mov edi, 1
    syscall

success:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata

.section .data

.section .bss
.align 8
inputbuf:
    .zero 65536
jump_table:
    .zero 524288
tape:
    .zero 30000
bracket_stack:
    .zero 8000
outbyte:
    .zero 1
