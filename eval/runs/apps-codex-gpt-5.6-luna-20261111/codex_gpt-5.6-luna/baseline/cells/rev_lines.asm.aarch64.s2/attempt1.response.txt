.equ BUFSIZE, 65536
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

.section .text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, xzr
    mov x24, #BUFSIZE

read_loop:
    cmp x20, x24
    b.eq process_input
    mov x0, #0
    add x1, x19, x20
    sub x2, x24, x20
    mov x8, #SYS_READ
    svc #0
    cbz x0, process_input
    add x20, x20, x0
    b read_loop

process_input:
    cbz x20, exit_program
    mov x21, xzr

next_line:
    cmp x21, x20
    b.eq exit_program
    mov x22, x21

find_end:
    cmp x22, x20
    b.eq final_line
    ldrb w0, [x19, x22]
    cmp w0, #10
    b.eq terminated_line
    add x22, x22, #1
    b find_end

terminated_line:
    mov x25, #0
    b reverse_line

final_line:
    mov x25, #1

reverse_line:
    mov x23, x22

reverse_loop:
    cmp x23, x21
    b.eq emit_newline
    sub x23, x23, #1
    add x1, x19, x23
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_WRITE

write_byte:
    svc #0
    cmp x0, #1
    b.ne write_byte
    b reverse_loop

emit_newline:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #SYS_WRITE

write_newline:
    svc #0
    cmp x0, #1
    b.ne write_newline
    cbnz x25, exit_program
    add x21, x22, #1
    b next_line

exit_program:
    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

.section .rodata
newline:
    .byte 10

.section .bss
.balign 16
input_buffer:
    .zero BUFSIZE
