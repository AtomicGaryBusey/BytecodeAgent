.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

.section .rodata
newline:
    .byte 0x0a

.bss
.align 3
input_buffer:
    .zero BUF_SIZE
.align 2
line_offsets:
    .zero 4000
.align 2
line_lengths:
    .zero 4000

.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    adrp x16, line_offsets
    add x16, x16, :lo12:line_offsets
    adrp x17, line_lengths
    add x17, x17, :lo12:line_lengths
    adrp x18, newline
    add x18, x18, :lo12:newline

    mov x20, #0
    mov x9, #BUF_SIZE

read_loop:
    add x1, x19, x20
    sub x2, x9, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, parse_start
    add x20, x20, x0
    b read_loop

parse_start:
    mov x21, #0
    mov x22, #0
    mov x23, #0

parse_loop:
    cmp x21, x20
    b.hs parse_done
    ldrb w24, [x19, x21]
    cmp w24, #10
    b.eq parse_newline
    add x21, x21, #1
    b parse_loop

parse_newline:
    str w22, [x16, x23, lsl #2]
    sub w25, w21, w22
    str w25, [x17, x23, lsl #2]
    add x23, x23, #1
    add x21, x21, #1
    mov x22, x21
    b parse_loop

parse_done:
    cmp x20, x22
    b.eq sort_start
    str w22, [x16, x23, lsl #2]
    sub w25, w20, w22
    str w25, [x17, x23, lsl #2]
    add x23, x23, #1

sort_start:
    mov x22, #1

sort_outer:
    cmp x22, x23
    b.hs output_start

    ldr w20, [x16, x22, lsl #2]
    ldr w21, [x17, x22, lsl #2]
    mov x24, x22

sort_inner:
    cbz x24, sort_insert
    sub x25, x24, #1
    ldr w26, [x16, x25, lsl #2]
    ldr w27, [x17, x25, lsl #2]

    mov x0, x26
    mov x1, x27
    mov x2, x20
    mov x3, x21
    bl compare_lines

    cmp x0, #0
    b.le sort_insert

    str w26, [x16, x24, lsl #2]
    str w27, [x17, x24, lsl #2]
    sub x24, x24, #1
    b sort_inner

sort_insert:
    str w20, [x16, x24, lsl #2]
    str w21, [x17, x24, lsl #2]
    add x22, x22, #1
    b sort_outer

output_start:
    mov x22, #0

output_loop:
    cmp x22, x23
    b.hs program_exit

    ldr w24, [x16, x22, lsl #2]
    ldr w25, [x17, x22, lsl #2]
    add x0, x19, x24
    mov x1, x25
    bl write_all

    mov x0, x18
    mov x1, #1
    bl write_all

    add x22, x22, #1
    b output_loop

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

compare_lines:
    add x6, x19, x0
    add x7, x19, x2
    mov x4, #0

compare_loop:
    cmp x4, x1
    b.hs compare_lengths
    cmp x4, x3
    b.hs compare_lengths

    ldrb w5, [x6, x4]
    ldrb w8, [x7, x4]
    cmp w5, w8
    b.lo compare_less
    b.hi compare_greater

    add x4, x4, #1
    b compare_loop

compare_lengths:
    cmp x1, x3
    b.lo compare_less
    b.hi compare_greater
    mov x0, #0
    ret

compare_less:
    mov x0, #-1
    ret

compare_greater:
    mov x0, #1
    ret

write_all:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    cbz x1, write_done
    mov x4, x0
    mov x5, x1

write_loop:
    mov x0, #1
    mov x1, x4
    mov x2, x5
    mov x8, #64
    svc #0
    add x4, x4, x0
    sub x5, x5, x0
    cbnz x5, write_loop

write_done:
    ldp x29, x30, [sp], #16
    ret
