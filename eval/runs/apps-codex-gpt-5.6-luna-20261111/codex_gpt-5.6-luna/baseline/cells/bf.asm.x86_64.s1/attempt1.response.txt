.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ MATCH_SIZE, 524288
.equ STACK_SIZE, 8000

.section .text
.globl _start
_start:
    lea r15, [rip + buffer]
    xor ebx, ebx

read_more:
    mov eax, INPUT_SIZE
    sub rax, rbx
    mov rdx, rax
    lea rsi, [r15 + rbx]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add rbx, rax
    jmp read_more

input_done:
    mov qword ptr [rip + prog_end], rbx
    mov qword ptr [rip + input_cur], rbx
    mov qword ptr [rip + input_end], rbx

    xor ecx, ecx
find_delimiter:
    cmp rcx, rbx
    jae delimiter_done
    cmp byte ptr [r15 + rcx], 33
    je delimiter_found
    inc rcx
    jmp find_delimiter

delimiter_found:
    mov qword ptr [rip + prog_end], rcx
    lea rax, [rcx + 1]
    mov qword ptr [rip + input_cur], rax

delimiter_done:
    lea r14, [rip + matches]
    lea r10, [rip + bracket_stack]
    xor ebx, ebx
    xor edx, edx

validate:
    cmp rbx, qword ptr [rip + prog_end]
    jae validate_done
    mov al, byte ptr [r15 + rbx]
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    inc rbx
    jmp validate

open_bracket:
    mov qword ptr [r10 + rdx*8], rbx
    inc rdx
    inc rbx
    jmp validate

close_bracket:
    test rdx, rdx
    jz invalid
    dec rdx
    mov rax, qword ptr [r10 + rdx*8]
    mov qword ptr [r14 + rax*8], rbx
    mov qword ptr [r14 + rbx*8], rax
    inc rbx
    jmp validate

validate_done:
    test rdx, rdx
    jnz invalid

    xor ebx, ebx
    xor r12d, r12d
    lea r13, [rip + tape]

execute:
    cmp rbx, qword ptr [rip + prog_end]
    jae success
    mov al, byte ptr [r15 + rbx]

    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_add
    cmp al, 45
    je command_sub
    cmp al, 46
    je command_output
    cmp al, 44
    je command_input
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close

    inc rbx
    jmp execute

command_right:
    inc r12
    inc rbx
    jmp execute

command_left:
    dec r12
    inc rbx
    jmp execute

command_add:
    inc byte ptr [r13 + r12]
    inc rbx
    jmp execute

command_sub:
    dec byte ptr [r13 + r12]
    inc rbx
    jmp execute

command_output:
    lea rsi, [r13 + r12]
output_retry:
    mov eax, 1
    mov edi, 1
    mov edx, 1
    syscall
    cmp eax, 1
    jne output_retry
    inc rbx
    jmp execute

command_input:
    mov rax, qword ptr [rip + input_cur]
    cmp rax, qword ptr [rip + input_end]
    jae input_empty
    movzx ecx, byte ptr [r15 + rax]
    inc rax
    mov qword ptr [rip + input_cur], rax
    mov byte ptr [r13 + r12], cl
    inc rbx
    jmp execute

input_empty:
    mov byte ptr [r13 + r12], 0
    inc rbx
    jmp execute

command_open:
    cmp byte ptr [r13 + r12], 0
    jne open_continue
    mov rax, qword ptr [r14 + rbx*8]
    inc rax
    mov rbx, rax
    jmp execute

open_continue:
    inc rbx
    jmp execute

command_close:
    cmp byte ptr [r13 + r12], 0
    je close_continue
    mov rax, qword ptr [r14 + rbx*8]
    inc rax
    mov rbx, rax
    jmp execute

close_continue:
    inc rbx
    jmp execute

invalid:
    mov eax, 60
    mov edi, 1
    syscall

success:
    mov eax, 60
    xor edi, edi
    syscall

.section .bss
.align 8
buffer:
    .zero INPUT_SIZE

.align 8
tape:
    .zero TAPE_SIZE

.align 8
matches:
    .zero MATCH_SIZE

.align 8
bracket_stack:
    .zero STACK_SIZE

.align 8
prog_end:
    .zero 8

input_cur:
    .zero 8

input_end:
    .zero 8
