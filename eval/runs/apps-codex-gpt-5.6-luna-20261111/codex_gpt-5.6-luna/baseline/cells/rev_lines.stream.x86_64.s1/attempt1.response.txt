.equ INPUT_SIZE, 65536

.section .text
.globl _start

_start:
    lea r12, [rip + buffer]
    lea r13, [rip + buffer]
    add r13, INPUT_SIZE

read_loop:
    cmp r12, r13
    jae input_done
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r13
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    mov r14, r12
    lea r12, [rip + buffer]

next_line:
    cmp r12, r14
    jae finish

    mov r15, r12

find_newline:
    cmp r15, r14
    jae final_line
    cmp byte ptr [r15], 10
    je found_newline
    inc r15
    jmp find_newline

found_newline:
    mov r8, r12
    lea r9, [r15 - 1]

reverse_line:
    cmp r8, r9
    jae reversed
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_line

reversed:
    mov rsi, r12
    mov rdx, r15
    sub rdx, r12
    call write_all

    mov rsi, r15
    mov edx, 1
    call write_all

    lea r12, [r15 + 1]
    jmp next_line

final_line:
    mov r8, r12
    lea r9, [r14 - 1]

reverse_final:
    cmp r8, r9
    jae final_reversed
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_final

final_reversed:
    mov rsi, r12
    mov rdx, r14
    sub rdx, r12
    call write_all

    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    jmp finish

write_all:
    test rdx, rdx
    jz write_done

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jnz write_loop

write_done:
    ret

finish:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.section .bss
.align 16
buffer:
    .zero INPUT_SIZE
