.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

.section .text
.globl _start

_start:
    xor r12d, r12d

read_loop:
    mov eax, 0
    xor edi, edi
    lea rsi, [buffer + r12]
    mov edx, BUF_SIZE + 1
    sub rdx, r12
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    xor rbx, rbx
    xor r13d, r13d
    xor r14d, r14d

parse_loop:
    cmp rbx, r12
    jae parse_final
    movzx eax, byte ptr [buffer + rbx]
    cmp al, 10
    jne parse_next
    mov [line_starts + r14*8], r13
    mov rax, rbx
    sub rax, r13
    mov [line_lengths + r14*8], rax
    inc r14
    lea r13, [rbx + 1]

parse_next:
    inc rbx
    jmp parse_loop

parse_final:
    cmp r13, r12
    je sort_start
    mov [line_starts + r14*8], r13
    mov rax, r12
    sub rax, r13
    mov [line_lengths + r14*8], rax
    inc r14

sort_start:
    mov [line_count], r14
    mov rbx, 1

sort_outer:
    cmp rbx, [line_count]
    jae output_start
    mov r12, [line_starts + rbx*8]
    mov r13, [line_lengths + rbx*8]
    mov rcx, rbx
    dec rcx

sort_compare:
    test rcx, rcx
    js sort_place_zero

    mov r14, [line_starts + rcx*8]
    mov r15, [line_lengths + rcx*8]
    xor rdx, rdx

sort_bytes:
    cmp rdx, r15
    je old_end
    cmp rdx, r13
    je sort_shift
    movzx eax, byte ptr [buffer + r14 + rdx]
    movzx r8d, byte ptr [buffer + r12 + rdx]
    cmp eax, r8d
    ja sort_shift
    jb sort_not_greater
    inc rdx
    jmp sort_bytes

old_end:
    cmp r15, r13
    jb sort_not_greater
    jmp sort_shift

sort_not_greater:
    lea rdx, [rcx + 1]
    mov [line_starts + rdx*8], r12
    mov [line_lengths + rdx*8], r13
    inc rbx
    jmp sort_outer

sort_shift:
    lea rdx, [rcx + 1]
    mov [line_starts + rdx*8], r14
    mov [line_lengths + rdx*8], r15
    dec rcx
    jmp sort_compare

sort_place_zero:
    mov [line_starts], r12
    mov [line_lengths], r13
    inc rbx
    jmp sort_outer

output_start:
    xor rbx, rbx

output_loop:
    cmp rbx, [line_count]
    jae program_exit
    mov r12, [line_starts + rbx*8]
    mov r13, [line_lengths + rbx*8]
    test r13, r13
    jz output_newline
    lea rsi, [buffer + r12]
    mov rdx, r13
    call write_all

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    call write_all
    inc rbx
    jmp output_loop

program_exit:
    mov eax, 60
    xor edi, edi
    syscall

write_all:
    test rdx, rdx
    jz write_done
    mov r8, rdx
    mov r9, rsi

write_loop:
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov rdx, r8
    syscall
    add r9, rax
    sub r8, rax
    jnz write_loop

write_done:
    ret

.section .rodata
newline:
    .byte 10

.section .bss
.align 16
buffer:
    .zero BUF_SIZE + 1

.align 8
line_starts:
    .zero MAX_LINES * 8

.align 8
line_lengths:
    .zero MAX_LINES * 8

.align 8
line_count:
    .zero 8
