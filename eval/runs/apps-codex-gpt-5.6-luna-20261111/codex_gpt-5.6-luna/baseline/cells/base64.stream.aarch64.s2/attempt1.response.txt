.equ input_size, 49152
.equ output_size, 70000

.section .rodata
alphabet:
	.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
	.align 3
input_buffer:
	.zero input_size
output_buffer:
	.zero output_size

.text
.globl _start
_start:
	adrp x19, input_buffer
	add x19, x19, :lo12:input_buffer
	mov x20, #0

read_loop:
	add x1, x19, x20
	mov x0, #0
	mov x2, #input_size
	sub x2, x2, x20
	mov x8, #63
	svc #0
	cmp x0, #0
	beq encode_start
	add x20, x20, x0
	b read_loop

encode_start:
	adrp x21, output_buffer
	add x21, x21, :lo12:output_buffer
	adrp x25, alphabet
	add x25, x25, :lo12:alphabet
	mov x22, #0
	mov x23, #0
	mov x24, #0

group_loop:
	sub x26, x20, x24
	cmp x26, #3
	b.lo remainder

	ldrb w4, [x19, x24]
	add x27, x24, #1
	ldrb w5, [x19, x27]
	add x27, x24, #2
	ldrb w6, [x19, x27]

	lsr w7, w4, #2
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	and w7, w4, #3
	lsl w7, w7, #4
	lsr w8, w5, #4
	orr w7, w7, w8
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	and w7, w5, #15
	lsl w7, w7, #2
	lsr w8, w6, #6
	orr w7, w7, w8
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	and w7, w6, #63
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	add x24, x24, #3
	b group_loop

remainder:
	cmp x24, x20
	beq finish_encode
	sub x26, x20, x24
	cmp x26, #1
	beq one_byte

	ldrb w4, [x19, x24]
	add x27, x24, #1
	ldrb w5, [x19, x27]

	lsr w7, w4, #2
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	and w7, w4, #3
	lsl w7, w7, #4
	lsr w8, w5, #4
	orr w7, w7, w8
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	and w7, w5, #15
	lsl w7, w7, #2
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	mov w0, #61
	bl emit_char
	b finish_encode

one_byte:
	ldrb w4, [x19, x24]

	lsr w7, w4, #2
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	and w7, w4, #3
	lsl w7, w7, #4
	ldrb w0, [x25, w7, uxtw]
	bl emit_char

	mov w0, #61
	bl emit_char
	mov w0, #61
	bl emit_char

finish_encode:
	cmp x23, #0
	beq write_start
	mov w0, #10
	bl emit_char

write_start:
	mov x24, #0

write_loop:
	cmp x24, x22
	beq exit_success
	add x1, x21, x24
	sub x2, x22, x24
	mov x0, #1
	mov x8, #64
	svc #0
	add x24, x24, x0
	b write_loop

exit_success:
	mov x0, #0
	mov x8, #93
	svc #0
	b exit_success

emit_char:
	strb w0, [x21, x22]
	add x22, x22, #1
	add x23, x23, #1
	cmp x23, #76
	b.ne emit_done
	mov w0, #10
	strb w0, [x21, x22]
	add x22, x22, #1
	mov x23, #0
emit_done:
	ret
