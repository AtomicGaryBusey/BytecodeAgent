.section .bss
.align 4
input_buf:
.zero 65536
.align 3
output_buf:
.zero 128

.text
.globl _start
_start:
    adrp x23, input_buf
    add x23, x23, :lo12:input_buf
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #65536
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x21, x21, x0
    mov x1, x23
    mov x2, x0

process_loop:
    ldrb w25, [x1], #1
    subs x2, x2, #1
    cmp w25, #10
    b.eq newline_byte
    cmp w25, #32
    b.eq whitespace_byte
    cmp w25, #9
    b.eq whitespace_byte
    cmp w25, #11
    b.eq whitespace_byte
    cmp w25, #12
    b.eq whitespace_byte
    cmp w25, #13
    b.eq whitespace_byte
    cbnz x22, next_byte
    add x20, x20, #1
    mov x22, #1
    b next_byte

newline_byte:
    add x19, x19, #1
    mov x22, #0
    b next_byte

whitespace_byte:
    mov x22, #0

next_byte:
    cbnz x2, process_loop
    b read_loop

input_done:
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    mov x26, x27

    mov x0, x19
    mov x1, x26
    bl decimal
    add x26, x26, x0
    mov w0, #32
    strb w0, [x26], #1

    mov x0, x20
    mov x1, x26
    bl decimal
    add x26, x26, x0
    mov w0, #32
    strb w0, [x26], #1

    mov x0, x21
    mov x1, x26
    bl decimal
    add x26, x26, x0
    mov w0, #10
    strb w0, [x26], #1

    mov x28, x26
    sub x28, x28, x27

write_loop:
    cbz x28, exit_program
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
    add x27, x27, x0
    sub x28, x28, x0
    b write_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

decimal:
    mov x9, x1
    cbnz x0, decimal_nonzero
    mov w4, #48
    strb w4, [x1]
    mov x0, #1
    ret

decimal_nonzero:
    mov x2, #10
    mov x3, #0

decimal_build:
    udiv x4, x0, x2
    msub x5, x4, x2, x0
    add w5, w5, #48
    strb w5, [x1], #1
    add x3, x3, #1
    mov x0, x4
    cbnz x0, decimal_build

    mov x6, x9
    sub x7, x1, #1

decimal_reverse:
    cmp x6, x7
    b.hs decimal_done
    ldrb w4, [x6]
    ldrb w5, [x7]
    strb w5, [x6]
    strb w4, [x7]
    add x6, x6, #1
    sub x7, x7, #1
    b decimal_reverse

decimal_done:
    mov x0, x3
    ret
