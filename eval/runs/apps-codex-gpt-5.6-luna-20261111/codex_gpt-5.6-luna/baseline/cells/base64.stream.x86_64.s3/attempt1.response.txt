.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 67000

.section .text
.globl _start
_start:
    xor ebx, ebx

read_loop:
    mov edi, 0
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, INPUT_SIZE
    sub edx, ebx
    mov eax, 0
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + input_buffer]
    lea rdx, [rsi + rbx]
    lea rdi, [rip + output_buffer]
    lea r12, [rip + alphabet]
    xor r8d, r8d

encode_full:
    lea rcx, [rsi + 3]
    cmp rcx, rdx
    ja encode_remainder

    movzx r9d, byte ptr [rsi]
    movzx r10d, byte ptr [rsi + 1]
    movzx r11d, byte ptr [rsi + 2]

    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi], al

    mov eax, r9d
    and eax, 3
    shl eax, 4
    mov ecx, r10d
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi + 1], al

    mov eax, r10d
    and eax, 15
    shl eax, 2
    mov ecx, r11d
    shr ecx, 6
    or eax, ecx
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi + 2], al

    mov eax, r11d
    and eax, 63
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi + 3], al

    add rsi, 3
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne encode_full
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp encode_full

encode_remainder:
    mov ecx, edx
    sub ecx, esi
    test ecx, ecx
    jz finish_full

    movzx r9d, byte ptr [rsi]

    mov eax, r9d
    shr eax, 2
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi], al

    mov eax, r9d
    and eax, 3
    shl eax, 4

    cmp ecx, 1
    je remainder_one

    movzx r10d, byte ptr [rsi + 1]
    mov ecx, r10d
    shr ecx, 4
    or eax, ecx
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi + 1], al

    mov eax, r10d
    and eax, 15
    shl eax, 2
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi + 2], al
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    jmp finish_remainder

remainder_one:
    mov al, byte ptr [r12 + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4

finish_remainder:
    mov byte ptr [rdi], 10
    inc rdi
    jmp output_ready

finish_full:
    test r8d, r8d
    jz output_ready
    mov byte ptr [rdi], 10
    inc rdi

output_ready:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi

write_loop:
    test rdx, rdx
    jz exit_program
    mov edi, 1
    mov eax, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jmp write_loop

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
input_buffer:
    .zero INPUT_SIZE
output_buffer:
    .zero OUTPUT_SIZE
