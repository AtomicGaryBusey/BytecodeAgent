.equ BUF_SIZE, 65536
.equ STACK_SIZE, 65536

.section .text
.globl _start
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #BUF_SIZE
    b.eq read_done
    mov x0, #0
    add x1, x21, x20
    mov x2, #BUF_SIZE
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x21, x21, x20
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x22, value_stack
    add x22, x22, :lo12:value_stack

line_start:
    cmp x20, x21
    b.eq program_exit
    mov x23, #0
    mov x24, #0

token_loop:
    cmp x20, x21
    b.eq line_finish
    ldrb w0, [x20]
    cmp w0, #10
    b.eq consume_line
    cmp w0, #32
    b.eq skip_separator
    cmp w0, #9
    b.eq skip_separator

    mov x25, x20
scan_token:
    cmp x20, x21
    b.eq token_scanned
    ldrb w0, [x20]
    cmp w0, #10
    b.eq token_scanned
    cmp w0, #32
    b.eq token_scanned
    cmp w0, #9
    b.eq token_scanned
    add x20, x20, #1
    b scan_token

token_scanned:
    sub x27, x20, x25
    cbz x27, token_loop

    cmp x27, #1
    b.ne maybe_integer
    ldrb w0, [x25]
    cmp w0, #43
    b.eq operator_token
    cmp w0, #45
    b.eq operator_token
    cmp w0, #42
    b.eq operator_token
    cmp w0, #47
    b.eq operator_token
    cmp w0, #37
    b.eq operator_token

maybe_integer:
    mov x19, #0
    mov x28, #0
    mov x29, x25
    mov x30, x27
    ldrb w0, [x29]
    cmp w0, #45
    b.ne integer_digits
    mov x19, #1
    add x29, x29, #1
    sub x30, x30, #1
    cbz x30, invalid_token

integer_digits:
    cmp x30, #0
    b.eq integer_done
    ldrb w0, [x29], #1
    cmp w0, #48
    b.lo invalid_token
    cmp w0, #57
    b.hi invalid_token
    sub w0, w0, #48
    mov x1, #10
    mul x28, x28, x1
    add x28, x28, x0
    sub x30, x30, #1
    b integer_digits

integer_done:
    cbz x19, push_integer
    neg x28, x28

push_integer:
    str x28, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_loop

operator_token:
    cmp x23, #2
    b.lo invalid_token
    sub x23, x23, #1
    ldr x30, [x22, x23, lsl #3]
    sub x23, x23, #1
    ldr x29, [x22, x23, lsl #3]
    ldrb w0, [x25]

    cmp w0, #43
    b.eq do_add
    cmp w0, #45
    b.eq do_sub
    cmp w0, #42
    b.eq do_mul
    cmp w0, #47
    b.eq do_div
    b do_mod

do_add:
    add x28, x29, x30
    b push_result

do_sub:
    sub x28, x29, x30
    b push_result

do_mul:
    mul x28, x29, x30
    b push_result

do_div:
    cbz x30, invalid_token
    mov x1, #1
    lsl x1, x1, #63
    cmp x29, x1
    b.ne division_ok
    mov x2, #-1
    cmp x30, x2
    b.eq invalid_token
division_ok:
    sdiv x28, x29, x30
    b push_result

do_mod:
    cbz x30, invalid_token
    mov x1, #1
    lsl x1, x1, #63
    cmp x29, x1
    b.ne modulo_ok
    mov x2, #-1
    cmp x30, x2
    b.eq invalid_token
modulo_ok:
    sdiv x28, x29, x30
    msub x28, x28, x30, x29

push_result:
    str x28, [x22, x23, lsl #3]
    add x23, x23, #1
    b token_loop

skip_separator:
    add x20, x20, #1
    b token_loop

invalid_token:
    mov x24, #1
skip_line:
    cmp x20, x21
    b.eq line_finish
    ldrb w0, [x20], #1
    cmp w0, #10
    b.ne skip_line
    b output_line

consume_line:
    add x20, x20, #1

line_finish:
    cmp x23, #1
    b.ne output_error
    cbnz x24, output_error
    ldr x28, [x22]
    b output_value

output_line:
    cmp x23, #1
    b.ne output_error
    cbnz x24, output_error
    ldr x28, [x22]

output_value:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x1, x1, #32
    mov x2, x1
    mov x19, #0
    cmp x28, #0
    b.ge format_digits
    mov x19, #1
    neg x29, x28
    b format_loop

format_digits:
    mov x29, x28
    cbnz x29, format_loop
    sub x1, x1, #1
    mov w0, #48
    strb w0, [x1]
    b format_sign

format_loop:
    mov x0, #10
    udiv x30, x29, x0
    msub x0, x30, x0, x29
    add w0, w0, #48
    sub x1, x1, #1
    strb w0, [x1]
    mov x29, x30
    cbnz x29, format_loop

format_sign:
    cbz x19, append_newline
    sub x1, x1, #1
    mov w0, #45
    strb w0, [x1]

append_newline:
    mov w0, #10
    strb w0, [x2]
    add x2, x2, #1
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0
    b line_start

output_error:
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x0, #1
    mov x2, #6
    mov x8, #64
    svc #0
    b line_start

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
error_msg:
    .ascii "error\n"

.section .bss
.align 3
input_buf:
    .zero BUF_SIZE
.align 3
value_stack:
    .zero STACK_SIZE
.align 3
output_buf:
    .zero 64
