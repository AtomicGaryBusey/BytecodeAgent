.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93
.equ BUF_SIZE, 65536

.section .text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr
    mov x23, xzr

read_more:
    mov x0, #0
    mov x1, x19
    mov x2, #BUF_SIZE
    mov x8, #SYS_READ
    svc #0
    cbz x0, format_output
    add x22, x22, x0
    mov x24, x19
    add x25, x24, x0

scan:
    cmp x24, x25
    b.eq read_more
    ldrb w26, [x24], #1

    cmp w26, #0x0a
    b.ne no_line
    add x20, x20, #1
no_line:
    cmp w26, #0x20
    b.eq whitespace
    cmp w26, #0x09
    b.eq whitespace
    cmp w26, #0x0b
    b.eq whitespace
    cmp w26, #0x0c
    b.eq whitespace
    cmp w26, #0x0d
    b.eq whitespace

    cbnz x23, scan
    add x21, x21, #1
    mov x23, #1
    b scan

whitespace:
    mov x23, xzr
    b scan

format_output:
    adrp x18, output_buf
    add x18, x18, :lo12:output_buf
    mov x24, x18
    mov x5, #10

    mov x0, x20
    mov x1, x24
    bl decimal
    mov x24, x0
    mov w2, #0x20
    strb w2, [x24], #1

    mov x0, x21
    mov x1, x24
    bl decimal
    mov x24, x0
    mov w2, #0x20
    strb w2, [x24], #1

    mov x0, x22
    mov x1, x24
    bl decimal
    mov x24, x0
    mov w2, #0x0a
    strb w2, [x24], #1

    sub x25, x24, x18
    mov x24, x18

write_more:
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #SYS_WRITE
    svc #0
    add x24, x24, x0
    sub x25, x25, x0
    cbnz x25, write_more

    mov x0, #0
    mov x8, #SYS_EXIT
    svc #0

decimal:
    mov x2, sp
    sub sp, sp, #16
    mov x6, #0
    cbnz x0, decimal_nonzero
    mov w4, #0x30
    strb w4, [sp]
    mov x6, #1
    b decimal_reverse

decimal_nonzero:
    udiv x3, x0, x5
    msub x4, x3, x5, x0
    add x4, x4, #0x30
    strb w4, [sp, x6]
    add x6, x6, #1
    mov x0, x3
    cbnz x0, decimal_nonzero

decimal_reverse:
    cbz x6, decimal_done
    sub x6, x6, #1
    ldrb w4, [sp, x6]
    strb w4, [x1], #1
    b decimal_reverse

decimal_done:
    mov sp, x2
    mov x0, x1
    ret

.section .bss
.align 4
input_buf:
    .zero BUF_SIZE
.align 4
output_buf:
    .zero 64
