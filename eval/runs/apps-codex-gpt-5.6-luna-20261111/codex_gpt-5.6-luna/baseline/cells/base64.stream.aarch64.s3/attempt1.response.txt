.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000
.equ LINE_LENGTH, 76

.section .text
.align 4
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #INPUT_SIZE
    b.hs encode_start
    sub x3, xzr, x20
    add x3, x3, #INPUT_SIZE
    mov x2, #4096
    cmp x3, x2
    b.hs read_ready
    mov x2, x3
read_ready:
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, encode_start
    add x20, x20, x0
    b read_loop

encode_start:
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x24, x22
    adrp x26, alphabet
    add x26, x26, :lo12:alphabet
    mov x21, #0
    mov x23, #0

encode_loop:
    cmp x21, x20
    b.hs write_start
    sub x0, x20, x21
    cmp x0, #3
    b.hs encode_three
    cmp x0, #2
    b.hs encode_two
    b encode_one

encode_three:
    ldrb w0, [x19, x21]
    add x1, x21, #1
    ldrb w1, [x19, x1]
    add x1, x21, #2
    ldrb w2, [x19, x1]

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    orr w4, w4, w1, lsr #4
    and w5, w1, #15
    lsl w5, w5, #2
    orr w5, w5, w2, lsr #6
    and w6, w2, #63

    ldrb w0, [x26, x3, uxtw]
    bl emit_char
    ldrb w0, [x26, x4, uxtw]
    bl emit_char
    ldrb w0, [x26, x5, uxtw]
    bl emit_char
    ldrb w0, [x26, x6, uxtw]
    bl emit_char

    add x21, x21, #3
    b encode_loop

encode_two:
    ldrb w0, [x19, x21]
    add x1, x21, #1
    ldrb w1, [x19, x1]

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    orr w4, w4, w1, lsr #4
    and w5, w1, #15
    lsl w5, w5, #2

    ldrb w0, [x26, x3, uxtw]
    bl emit_char
    ldrb w0, [x26, x4, uxtw]
    bl emit_char
    ldrb w0, [x26, x5, uxtw]
    bl emit_char
    mov w0, #61
    bl emit_char

    add x21, x21, #2
    b encode_loop

encode_one:
    ldrb w0, [x19, x21]

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4

    ldrb w0, [x26, x3, uxtw]
    bl emit_char
    ldrb w0, [x26, x4, uxtw]
    bl emit_char
    mov w0, #61
    bl emit_char
    mov w0, #61
    bl emit_char

    add x21, x21, #1
    b encode_loop

write_start:
    sub x25, x22, x24
    cbz x25, exit_success

write_loop:
    mov x0, #1
    mov x1, x24
    mov x2, x25
    mov x8, #64
    svc #0
    add x24, x24, x0
    sub x25, x25, x0
    cbnz x25, write_loop

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

emit_char:
    strb w0, [x22], #1
    add x23, x23, #1
    cmp x23, #LINE_LENGTH
    b.ne emit_done
    mov w0, #10
    strb w0, [x22], #1
    mov x23, #0
emit_done:
    ret

.section .rodata
.align 4
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4
input_buf:
    .zero INPUT_SIZE
output_buf:
    .zero OUTPUT_SIZE
