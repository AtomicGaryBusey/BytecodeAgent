.equ INPUT_SIZE, 49152
.equ OUTPUT_SIZE, 70000

.section .text
.align 4
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    cmp x20, #INPUT_SIZE
    b.hs encode_start

    mov x2, #4096
    mov x3, #INPUT_SIZE
    sub x3, x3, x20
    cmp x3, x2
    csel x2, x3, x2, lo

    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, encode_start
    add x20, x20, x0
    b read_loop

encode_start:
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    mov x23, x22

    adrp x24, alphabet
    add x24, x24, :lo12:alphabet

    add x20, x19, x20
    mov x21, x19
    mov x25, #0

encode_loop:
    cmp x21, x20
    b.hs encode_done

    sub x0, x20, x21
    cmp x0, #3
    b.hs encode_three
    cmp x0, #2
    b.hs encode_two
    b encode_one

encode_three:
    ldrb w0, [x21]
    ldrb w1, [x21, #1]
    ldrb w2, [x21, #2]
    add x21, x21, #3

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    orr w4, w4, w1, lsr #4
    and w5, w1, #15
    lsl w5, w5, #2
    orr w5, w5, w2, lsr #6
    and w6, w2, #63

    add x7, x24, x3
    ldrb w3, [x7]
    strb w3, [x23], #1

    add x7, x24, x4
    ldrb w4, [x7]
    strb w4, [x23], #1

    add x7, x24, x5
    ldrb w5, [x7]
    strb w5, [x23], #1

    add x7, x24, x6
    ldrb w6, [x7]
    strb w6, [x23], #1

    add x25, x25, #4
    cmp x25, #76
    b.ne encode_loop

    mov w0, #10
    strb w0, [x23], #1
    mov x25, #0
    b encode_loop

encode_two:
    ldrb w0, [x21]
    ldrb w1, [x21, #1]
    add x21, x21, #2

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    orr w4, w4, w1, lsr #4
    and w5, w1, #15
    lsl w5, w5, #2

    add x7, x24, x3
    ldrb w3, [x7]
    strb w3, [x23], #1

    add x7, x24, x4
    ldrb w4, [x7]
    strb w4, [x23], #1

    add x7, x24, x5
    ldrb w5, [x7]
    strb w5, [x23], #1

    mov w0, #61
    strb w0, [x23], #1
    b encode_done

encode_one:
    ldrb w0, [x21]
    add x21, x21, #1

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4

    add x7, x24, x3
    ldrb w3, [x7]
    strb w3, [x23], #1

    add x7, x24, x4
    ldrb w4, [x7]
    strb w4, [x23], #1

    mov w0, #61
    strb w0, [x23], #1
    strb w0, [x23], #1

encode_done:
    cmp x23, x22
    b.eq exit_program

    cmp x25, #0
    b.eq prepare_write

    mov w0, #10
    strb w0, [x23], #1

prepare_write:
    sub x20, x23, x22
    mov x21, x22

write_loop:
    mov x0, #1
    mov x1, x21
    mov x2, x20
    mov x8, #64
    svc #0
    add x21, x21, x0
    sub x20, x20, x0
    cbnz x20, write_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
.align 4
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 4
input_buf:
    .zero INPUT_SIZE
output_buf:
    .zero OUTPUT_SIZE
