.equ INPUT_CAP, 65536

.section .bss
.balign 16
input_buffer:
.zero 65537

.section .rodata
newline:
.byte 10

.text
.globl _start
_start:
adrp x19, input_buffer
add x19, x19, :lo12:input_buffer
mov x20, #0
mov x21, #INPUT_CAP

read_more:
cmp x20, x21
b.eq probe_eof
sub x2, x21, x20
add x1, x19, x20
mov x0, #0
mov x8, #63
svc #0
cbz x0, process_input
add x20, x20, x0
b read_more

probe_eof:
add x1, x19, x20
mov x2, #1
mov x0, #0
mov x8, #63
svc #0
cbz x0, process_input
add x20, x20, x0

process_input:
mov x21, #0
mov x22, #0

scan_input:
cmp x22, x20
b.hs final_line
ldrb w23, [x19, x22]
add x22, x22, #1
cmp w23, #10
b.ne scan_input

sub x1, x22, x21
sub x1, x1, #1
add x0, x19, x21
bl reverse_line
add x0, x19, x21
sub x1, x22, x21
sub x1, x1, #1
bl write_line
mov x21, x22
b scan_input

final_line:
cmp x21, x20
b.eq program_exit
add x0, x19, x21
sub x1, x20, x21
bl reverse_line
add x0, x19, x21
sub x1, x20, x21
bl write_line

program_exit:
mov x0, #0
mov x8, #93
svc #0

reverse_line:
subs x2, x1, #1
b.mi reverse_done
mov x3, #0

reverse_loop:
cmp x3, x2
b.hs reverse_done
ldrb w4, [x0, x3]
ldrb w5, [x0, x2]
strb w5, [x0, x3]
strb w4, [x0, x2]
add x3, x3, #1
sub x2, x2, #1
b reverse_loop

reverse_done:
ret

write_line:
mov x26, x0
mov x27, x1

write_bytes:
cbz x27, write_newline
mov x0, #1
mov x1, x26
mov x2, x27
mov x8, #64
svc #0
add x26, x26, x0
sub x27, x27, x0
b write_bytes

write_newline:
adrp x26, newline
add x26, x26, :lo12:newline
mov x27, #1

write_newline_loop:
mov x0, #1
mov x1, x26
mov x2, #1
mov x8, #64
svc #0
cbz x0, write_newline_loop
ret
