.equ INPUT_SIZE, 65536

.section .text
.globl _start

_start:
    adrp x21, input_buffer
    add x21, x21, :lo12:input_buffer
    adrp x22, output_buffer
    add x22, x22, :lo12:output_buffer
    adrp x23, hex_digits
    add x23, x23, :lo12:hex_digits

    mov x19, #0

read_more:
    mov x2, #INPUT_SIZE
    sub x2, x2, x19
    mov x0, #0
    add x1, x21, x19
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x19, x19, x0
    b read_more

input_done:
    mov x20, #0

row_next:
    cmp x20, x19
    b.hs final_offset

    sub x24, x19, x20
    mov x27, #16
    cmp x24, #16
    csel x24, x27, x24, hi

    mov x25, #0
    mov w26, #0x20

clear_row:
    strb w26, [x22, x25]
    add x25, x25, #1
    cmp x25, #80
    b.lo clear_row

    mov x25, #0
    mov x26, #28

offset_digits:
    lsr x27, x20, x26
    and x27, x27, #15
    ldrb w27, [x23, x27]
    strb w27, [x22, x25]
    add x25, x25, #1
    subs x26, x26, #4
    b.ge offset_digits

    mov x25, #0

byte_next:
    cmp x25, x24
    b.hs row_ready

    add x27, x20, x25
    ldrb w26, [x21, x27]

    lsr w27, w26, #4
    ldrb w27, [x23, x27]
    add x28, x25, x25, lsl #1
    add x28, x28, #10
    strb w27, [x22, x28]

    and w27, w26, #15
    ldrb w27, [x23, x27]
    add x28, x28, #1
    strb w27, [x22, x28]

    cmp w26, #0x20
    b.lo nonprintable
    cmp w26, #0x7e
    b.hi nonprintable
    add x28, x25, #61
    strb w26, [x22, x28]
    b byte_done

nonprintable:
    mov w27, #'.'
    add x28, x25, #61
    strb w27, [x22, x28]

byte_done:
    add x25, x25, #1
    b byte_next

row_ready:
    mov w26, #0x20
    strb w26, [x22, #59]
    mov w26, #'|'
    strb w26, [x22, #60]
    add x27, x24, #61
    strb w26, [x22, x27]
    add x27, x27, #1
    mov w26, #'\n'
    strb w26, [x22, x27]

    add x2, x24, #63
    mov x1, x22
    bl write_all

    add x20, x20, x24
    b row_next

final_offset:
    mov x20, x19
    mov x25, #0
    mov x26, #28

final_digits:
    lsr x27, x20, x26
    and x27, x27, #15
    ldrb w27, [x23, x27]
    strb w27, [x22, x25]
    add x25, x25, #1
    subs x26, x26, #4
    b.ge final_digits

    mov w26, #'\n'
    strb w26, [x22, #8]
    mov x1, x22
    mov x2, #9
    bl write_all

    mov x0, #0
    mov x8, #93
    svc #0

write_all:
    mov x27, x1
    mov x28, x2

write_again:
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
    cbz x0, write_again
    add x27, x27, x0
    sub x28, x28, x0
    cbnz x28, write_again
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 4
input_buffer:
    .zero 65536
output_buffer:
    .zero 80
