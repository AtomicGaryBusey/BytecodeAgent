.global _start
.equ BUFFER_SIZE, 65536

.section .rodata
space_char: .byte ' '
newline_char: .byte 0x0A
zero_char: .byte '0'

.section .bss
buf: .space BUFFER_SIZE

.section .data
lines_qty: .quad 0
words_qty: .quad 0
bytes_qty: .quad 0
num_buf: .space 20

.section .text

_start:
    xor r9d, r9d                # in_word flag = 0
read_loop:
    mov rdi, 0                  # fd 0 stdin
    lea rsi, buf
    mov rdx, BUFFER_SIZE
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done
    mov rcx, rax                # bytes read
    lea rdi, buf                # pointer to buffer
process_bytes:
    test rcx, rcx
    jz read_loop                # no more bytes in this chunk, read next
    movzx r10d, byte ptr [rdi]  # current byte
    inc qword ptr [bytes_qty]   # total bytes
    cmp r10d, 0x0A
    je newline
    cmp r10d, 0x20
    je whitespace
    cmp r10d, 0x09
    je whitespace
    cmp r10d, 0x0B
    je whitespace
    cmp r10d, 0x0C
    je whitespace
    cmp r10d, 0x0D
    je whitespace
    test r9d, r9d
    jne nonws_cont
    inc qword ptr [words_qty]
    mov r9d, 1
nonws_cont:
    jmp after_char
whitespace:
    xor r9d, r9d
    jmp after_char
newline:
    inc qword ptr [lines_qty]
    xor r9d, r9d
    jmp after_char
after_char:
    inc rdi
    dec rcx
    jnz process_bytes
    jmp read_loop

read_done:
    lea rdi, [lines_qty]
    call print_number
    mov rdi, 1
    mov rsi, space_char
    mov rdx, 1
    mov rax, 1
    syscall
    lea rdi, [words_qty]
    call print_number
    mov rdi, 1
    mov rsi, space_char
    mov rdx, 1
    mov rax, 1
    syscall
    lea rdi, [bytes_qty]
    call print_number
    mov rdi, 1
    mov rsi, newline_char
    mov rdx, 1
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

print_number:
    mov rax, qword ptr [rdi]   # load number
    xor rdx, rdx
    test rax, rax
    jnz .divide_loop
    mov rdi, 1
    mov rsi, zero_char
    mov rdx, 1
    mov rax, 1
    syscall
    ret
.divide_loop:
    mov rcx, 10
    xor r8d, r8d
div_loop:
    test rax, rax
    jz .done_div
    xor rdx, rdx
    div rcx
    add rdx, 48
    lea rsi, [rip + num_buf]
    add rsi, r8d
    mov byte ptr [rsi], dl
    inc r8d
    jmp div_loop
.done_div:
    mov rdi, 1
    mov r8, r8d
    lea rsi, [rip + num_buf]
    add rsi, r8
    dec rsi
.output_digit:
    mov rdx, 1
    mov rax, 1
    syscall
    dec rsi
    dec r8
    jnz .output_digit
    ret
