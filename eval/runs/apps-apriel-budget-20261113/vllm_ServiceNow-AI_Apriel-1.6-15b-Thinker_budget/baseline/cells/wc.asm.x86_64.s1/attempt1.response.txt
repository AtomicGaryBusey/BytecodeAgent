.global _start
.equ buffer_len, 65536

.section .rodata
space_char: .byte ' '
newline_char: .byte 0x0A
zero_char: .byte '0'

.section .bss
buf: .space 65536
num_buf: .space 6

.section .data
lines_qty: .quad 0
words_qty: .quad 0
bytes_qty: .quad 0

.section .text

_start:
    xor r9d, r9d                ; in_word flag = 0
.read_loop:
    lea rsi, [buf]              ; buffer address
    mov rdx, buffer_len         ; number of bytes to read
    mov rax, 0                  ; sys_read
    syscall
    test rax, rax
    jz end_read
    mov rcx, rax                ; bytes read
    lea rdi, [buf]              ; pointer to buffer

process_bytes:
    movzx r10d, byte ptr [rdi]     ; current byte
    inc qword ptr [bytes_qty]       ; total bytes
    cmp r10d, 0x0A
    je .newline_char
    cmp r10d, 0x20
    je .whitespace_char
    cmp r10d, 0x09
    je .whitespace_char
    cmp r10d, 0x0B
    je .whitespace_char
    cmp r10d, 0x0C
    je .whitespace_char
    cmp r10d, 0x0D
    je .whitespace_char
    ; non-whitespace
    test r9d, r9d
    jne .nonws_continue
    inc qword ptr [words_qty]
    mov r9d, 1
.nonws_continue:
    jmp .after_char

.whitespace_char:
    mov r9d, 0
    jmp .after_char

.newline_char:
    inc qword ptr [lines_qty]
    mov r9d, 0
    jmp .after_char

.after_char:
    inc rdi
    dec rcx
    jnz process_bytes
    jmp read_loop

end_read:
    ; print lines
    lea rdi, [lines_qty]
    call print_number
    ; space
    mov rdi, 1
    mov rsi, space_char
    mov rdx, 1
    mov rax, 1
    syscall
    ; print words
    lea rdi, [words_qty]
    call print_number
    ; space
    mov rdi, 1
    mov rsi, space_char
    mov rdx, 1
    mov rax, 1
    syscall
    ; print bytes
    lea rdi, [bytes_qty]
    call print_number
    ; newline
    mov rdi, 1
    mov rsi, newline_char
    mov rdx, 1
    mov rax, 1
    syscall
    ; exit
    mov rax, 60
    xor rdi, rdi
    syscall

print_number:
    mov rax, qword ptr [rdi]      ; load value
    xor rdx, rdx                  ; high half zero
    test rax, rax
    jnz .divide_loop
    ; zero case: write '0'
    mov rdi, 1
    mov rsi, zero_char
    mov rdx, 1
    mov rax, 1
    syscall
    ret
.divide_loop:
    mov rcx, 10                   ; divisor
    xor r8d, r8d                  ; digit index
.div_loop:
    div rcx                       ; rdx:rax / rcx
    add al, 48                    ; ascii digit
    lea rsi, [rip + num_buf]
    add rsi, r8d
    mov [rsi], al
    inc r8d
    test rdx, rdx
    jnz .div_loop
    ; output digits in reverse order
    mov rcx, r8d                 ; count
    lea rsi, [rip + num_buf]
    add rsi, r8d
    dec rsi
.output_digit:
    mov rdx, 1
    mov rax, 1
    syscall
    dec rsi
    dec rcx
    jnz .output_digit
    ret
