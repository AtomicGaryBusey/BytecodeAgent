.global _start
.equ BUFFER_SIZE, 65536

.section .rodata
space_char: .byte ' '
newline_char: .byte 0x0A
zero_char: .byte '0'

.section .bss
buf: .space BUFFER_SIZE
.align 8
lines_qty: .space 8
words_qty: .space 8
bytes_qty: .space 8
.align 8
num_buf: .space 20

.section .text
_start:
    xor r9d, r9d                # in_word flag = 0
read_loop:
    mov rdi, 0                  # fd 0 stdin
    lea rsi, [rip + buf]
    mov rdx, BUFFER_SIZE
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done
    mov rcx, rax                # bytes read
    lea rdi, [rip + buf]
process_bytes:
    test rcx, rcx
    jz read_loop                # no more bytes in this chunk, read next
    mov al, byte ptr [rdi]      # current byte
    lea rsi, [rip + bytes_qty]
    inc qword ptr [rsi]
    cmp al, 0x0A
    je newline
    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    test r9d, r9d
    jne nonws_cont
    lea rsi, [rip + words_qty]
    inc qword ptr [rsi]
    mov r9d, 1
nonws_cont:
    jmp after_char
whitespace:
    xor r9d, r9d
    jmp after_char
newline:
    lea rsi, [rip + lines_qty]
    inc qword ptr [rsi]
    xor r9d, r9d
    jmp after_char
after_char:
    inc rdi
    dec rcx
    jnz process_bytes
    jmp read_loop

read_done:
    lea rdi, [rip + lines_qty]
    call print_number
    mov rdi, 1
    lea rsi, [rip + space_char]
    mov rdx, 1
    mov rax, 1
    syscall
    lea rdi, [rip + words_qty]
    call print_number
    mov rdi, 1
    lea rsi, [rip + space_char]
    mov rdx, 1
    mov rax, 1
    syscall
    lea rdi, [rip + bytes_qty]
    call print_number
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

print_number:
    mov rax, qword ptr [rdi]   # load number
    test rax, rax
    jnz .divide_loop
    mov rdi, 1
    lea rsi, [rip + zero_char]
    mov rdx, 1
    mov rax, 1
    syscall
    ret
.divide_loop:
    mov rcx, 10
    xor rdx, rdx
    xor r9d, r9d
    .div_loop:
    test rax, rax
    jz .done_div
    xor rdx, rdx
    div rcx
    add dl, 48
    lea rsi, [rip + num_buf]
    mov byte ptr [rsi + r9d], dl
    inc r9d
    jmp .div_loop
.done_div:
    mov rdi, 1
    mov r8, r9d                # length of digit string
    lea rsi, [rip + num_buf]
    add rsi, r8
    dec rsi
    .output_digit:
    mov rdx, 1
    mov rax, 1
    syscall
    dec rsi
    dec r8
    jnz .output_digit
    ret
