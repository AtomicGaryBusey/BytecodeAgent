global _start
.text
_start:
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    mov x0, #0          // fd = 0 (stdin)
    mov x2, #65536      // buffer size
    mov x3, #0          // bytes read
read_loop:
    sub x4, x2, x3      // remaining space
    mov x0, #0          // fd
    mov x1, x1          // buffer address
    mov x2, x4          // count
    mov x8, #63         // sys_read
    svc 0
    cmp x0, #0
    beq read_done
    add x3, x3, x0
    b read_loop
read_done:
    mov x8, #93         // sys_exit
    mov x0, #0
    svc 0

.bss
inbuf:
    .space 65536
stack:
    .space 4096
