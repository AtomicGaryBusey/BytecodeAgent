.global _start
.equ BUF_SIZE, 65536

.section .rodata
nl: .byte 0x0A

.section .bss
input_buf: .space 65536

.section .text
_start:
    xor r8, r8          ; offset = 0

.read_loop:
    mov rdi, 0          ; fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, r8
    mov eax, BUF_SIZE
    sub eax, r8
    mov rdx, eax
    mov eax, 0
    syscall
    test eax, eax
    jz .read_done
    add r8, eax
    jmp .read_loop

.read_done:
    mov r9, r8
    xor r10, r10
    lea rdx, [rip + input_buf]
    jmp .process

.process:
    cmp r10, r9
    je .final_line
    mov r11, r10
    ; find newline
.find_newline:
    cmp r11, r9
    jge .final_line
    mov al, byte ptr [rdx + r11]
    cmp al, 0x0A
    je .found_newline
    inc r11
    jmp .find_newline
.found_newline:
    ; reverse bytes from start (r10) to newline-1
    mov rax, r11
    dec rax
    mov rcx, r10
    cmp rcx, rax
    jbe .no_reverse
.reverse:
    mov rsi, rdx
    add rsi, rcx
    mov rdi, rdx
    add rdi, rax
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rdi]
    mov byte ptr [rsi], al
    mov byte ptr [rdi], bl
    inc rcx
    dec rax
    cmp rcx, rax
    jle .reverse_done
    jmp .reverse
.reverse_done:
.no_reverse:
    mov r12, r11
    mov rax, r12
    sub rax, r10
    mov r8, rax
    test r8, r8
    jz .skip_line_bytes
    mov rdi, 1
    mov rsi, rdx
    add rsi, r10
.write_line_bytes:
    mov rdx, 1
    mov al, byte ptr [rsi]
    mov eax, 1
    syscall
    inc rsi
    dec r8
    jnz .write_line_bytes
.skip_line_bytes:
    lea rsi, [rip + nl]
    mov rdx, 1
    mov eax, 1
    syscall
    mov r10, r12
    inc r10
    jmp .process

.final_line:
    cmp r10, r9
    je .exit_program
    mov rax, r9
    dec rax
    mov rcx, r10
    cmp rcx, rax
    jbe .no_reverse_final
.reverse_final:
    mov rsi, rdx
    add rsi, rcx
    mov rdi, rdx
    add rdi, rax
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rdi]
    mov byte ptr [rsi], al
    mov byte ptr [rdi], bl
    inc rcx
    dec rax
    cmp rcx, rax
    jle .reverse_final_done
    jmp .reverse_final
.reverse_final_done:
.no_reverse_final:
    mov rax, r9
    sub rax, r10
    mov r8, rax
    mov rdi, 1
    mov rsi, rdx
    add rsi, r10
.write_final_bytes:
    mov rdx, 1
    mov al, byte ptr [rsi]
    mov eax, 1
    syscall
    inc rsi
    dec r8
    jnz .write_final_bytes
    lea rsi, [rip + nl]
    mov rdx, 1
    mov eax, 1
    syscall
.exit_program:
    mov eax, 60
    xor edi, edi
    syscall
