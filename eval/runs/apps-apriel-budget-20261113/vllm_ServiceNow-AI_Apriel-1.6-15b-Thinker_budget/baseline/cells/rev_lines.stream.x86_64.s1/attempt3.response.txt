.global _start
.equ BUF_SIZE, 65536

.section .rodata
nl: .byte 0x0A

.section .data
input_len: .zero 4

.section .bss
input_buf: .space 65536

.section .text
_start:
    xor rax, rax
    ; read loop
read_loop:
    mov r8, BUF_SIZE
    mov eax, [input_len]
    sub r8, eax
    mov rdx, r8
    mov rdi, 0
    mov rsi, input_buf
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add [input_len], rax
    jmp read_loop

read_done:
    mov rbx, [input_len]
    lea r9, [input_buf + rbx]
    mov rdi, input_buf

process_loop:
    cmp rdi, r9
    jge process_done

    mov r8, rdi
scan_loop:
    cmp r8, r9
    jge scan_done
    mov al, byte ptr [r8]
    cmp al, 0x0A
    jne scan_next
    jmp newline_found
scan_next:
    inc r8
    jmp scan_loop
scan_done:
    mov r8, r9
newline_found:
    mov rax, r8
    sub rax, rdi
    mov rdx, rax
    push r8
    mov r10, rdi
    lea r11, [r8 - 1]
rev_loop:
    movzx eax, byte ptr [r10]
    movzx ebx, byte ptr [r11]
    mov byte ptr [r10], bl
    mov byte ptr [r11], al
    inc r10
    dec r11
    cmp r10, r11
    jl rev_loop
    mov rsi, rdi
    mov rdi, 1
    mov rax, 1
    syscall
    mov rax, 1
    mov rdi, 1
    mov rsi, nl
    mov rdx, 1
    syscall
    pop r8
    cmp r8, r9
    jne newline_was_found
    mov rdi, r9
    jmp process_loop
newline_was_found:
    lea rdi, [r8 + 1]
    jmp process_loop

process_done:
    mov rax, 60
    xor rdi, rdi
    syscall
