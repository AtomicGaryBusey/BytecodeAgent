.global _start
.equ BUF_SIZE, 65536

.section .rodata
nl: .byte 0x0A

.section .bss
input_buf: .space 65536

.section .text
_start:
    xor eax, eax
    mov rbx, 0
read_top:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov rcx, BUF_SIZE
    sub rcx, rbx
    mov rdx, rcx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_top

read_done:
    mov r15, rbx
    xor r8, r8
    lea r14, [rip + input_buf]

process_loop:
    cmp r8, r15
    jge exit_program

    mov r9, r8
scan_loop:
    cmp r9, r15
    jge newline_not_found
    movzx rdx, byte ptr [r14 + r9]
    cmp rdx, 0x0A
    jne scan_adv
    jmp newline_found
scan_adv:
    inc r9
    jmp scan_loop

newline_found:
    mov r10, r9
    dec r10
reverse_loop:
    cmp r10, r8
    jl reverse_done
    mov rax, 1
    mov rdi, 1
    lea rsi, [r14 + r10]
    mov rdx, 1
    syscall
    dec r10
    jmp reverse_loop
reverse_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    inc r9
    mov r8, r9
    jmp process_loop

newline_not_found:
    mov r10, r15
    dec r10
rem_reverse_loop:
    cmp r10, r8
    jl rem_reverse_done
    mov rax, 1
    mov rdi, 1
    lea rsi, [r14 + r10]
    mov rdx, 1
    syscall
    dec r10
    jmp rem_reverse_loop
rem_reverse_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall
    jmp exit_program

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
