.global _start
.section .rodata
nl: .byte 0x0A

.section .bss
input_buf: .space 65536

.section .text
_start:
    xor r8, r8          ; offset for read
    xor r9, r9          ; total_len

read_loop:
    mov rax, 0          ; sys_read
    mov rdi, 0          ; stdin
    mov rsi, input_buf
    add rsi, r8
    mov rdx, 65536
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    add r9, rax
    jmp read_loop

read_done:
    test r9, r9
    jz exit_program

    xor r8, r8          ; scanning offset
    xor r10, r10        ; start offset

process_loop:
    cmp r8, r9
    jge process_done

    mov al, byte [rel input_buf + r8]
    cmp al, 0x0A
    je line_found

    inc r8
    jmp process_loop

line_found:
    mov rdx, r8
    sub rdx, r10        ; length = r8 - r10

    test rdx, rdx
    jz reverse_skip

    lea r12, [rel input_buf + r10]   ; start_addr
    mov rax, r8
    dec rax
    lea r13, [rel input_buf + rax]   ; end_addr

    rev_loop:
        cmp r12, r13
        jge rev_done
        mov al, byte [r12]
        mov bl, byte [r13]
        mov byte [r12], bl
        mov byte [r13], al
        inc r12
        dec r13
        jmp rev_loop

reverse_skip:
    jmp rev_done

rev_done:
    test rdx, rdx
    jz write_skip

write_line:
    mov rax, 1          ; sys_write
    mov rdi, 1          ; stdout
    lea rsi, [rel input_buf + r10]
    mov rdx, rdx
    syscall

write_skip:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel nl]
    mov rdx, 1
    syscall

    mov r10, r8
    inc r10
    mov r8, r10

    jmp process_loop

process_done:
    cmp r10, r9
    jge exit_program

    mov rdx, r9
    sub rdx, r10        ; length = r9 - r10

    test rdx, rdx
    jz reverse_skip_final

    lea r12, [rel input_buf + r10]
    mov rax, r9
    dec rax
    lea r13, [rel input_buf + rax]

    revfinal_loop:
        cmp r12, r13
        jge revfinal_done
        mov al, byte [r12]
        mov bl, byte [r13]
        mov byte [r12], bl
        mov byte [r13], al
        inc r12
        dec r13
        jmp revfinal_loop

reverse_skip_final:
    jmp revfinal_done

revfinal_done:
    test rdx, rdx
    jz write_skip_final

    mov rax, 1
    mov rdi, 1
    lea rsi, [rel input_buf + r10]
    mov rdx, rdx
    syscall

write_skip_final:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel nl]
    mov rdx, 1
    syscall

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
