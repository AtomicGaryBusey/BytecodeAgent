.global _start
.text
.section .rodata
hexchars: .ascii "0123456789abcdef"
.section .bss
input: .space 65536
row_buf: .space 80
.text
_start:
    xor rbx, rbx                ; total_len = 0
    lea rsi, [input]            ; input pointer
read_loop:
    mov rax, 0                  ; sys_read
    mov rdi, 0                  ; fd 0
    mov rdx, 65536              ; max read
    syscall
    test rax, rax
    jz .read_done
    add rbx, rax                ; total_len += bytes read
    lea rsi, [rsi + rax]        ; advance input pointer
    jmp .read_loop
.read_done:
    xor rcx, rcx                ; offset = 0
.row_process:
    cmp rcx, rbx
    jge .print_total_len
    mov rax, rbx
    sub rax, rcx                ; remaining = total_len - offset
    mov r8d, ax                 ; r8d = remaining
    mov r11d, 16
    cmp r8d, r11d
    jge .n_full
    mov r11d, r8d               ; r11d = n = min(16, remaining)
.n_full:
    lea rdx, [input + rcx]      ; pointer to start of this row
    lea rdi, [row_buf]          ; destination buffer

    ; offset as 8 hex digits
    mov rax, rcx
    shr rax, 28
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    ; two spaces
    mov byte [rdi], ' '
    inc rdi
    mov byte [rdi], ' '
    inc rdi
    ; hex bytes loop
    xor r9d, r9d                ; i = 0
.hex_loop:
    cmp r9d, 8
    je .extra_space_before
    cmp r9d, r11d               ; i < n ?
    jge .empty_slot
    movzx r10d, byte ptr [rdx + r9]   ; byte value
    ; high nibble
    mov r11d, r10d
    shr r11d, 4
    movzx r12d, byte ptr [hexchars + r11d]
    mov byte [rdi], r12b
    inc rdi
    ; low nibble
    mov r11d, r10d
    and r11d, 0xF
    movzx r12d, byte ptr [hexchars + r11d]
    mov byte [rdi], r12b
    inc rdi
    ; space after byte
    mov byte [rdi], ' '
    inc rdi
    jmp .hex_slot_done
.extra_space_before:
    mov byte [rdi], ' '
    inc rdi
    jmp .hex_slot_done
.empty_slot:
    mov byte [rdi], ' '
    inc rdi
    mov byte [rdi], ' '
    inc rdi
    mov byte [rdi], ' '
    inc rdi
.hex_slot_done:
    inc r9d
    cmp r9d, 16
    jl .hex_loop
    ; space and '|'
    mov byte [rdi], ' '
    inc rdi
    mov byte [rdi], '|'
    inc rdi
    ; ASCII part
    xor r9d, r9d                ; i = 0
.ascii_loop:
    cmp r9d, r11d
    jge .ascii_done
    movzx r10d, byte ptr [rdx + r9]   ; original byte
    cmp r10d, 0x20
    jl .ascii_dot
    cmp r10d, 0x7e
    jg .ascii_dot
    mov byte [rdi], r10b
    jmp .ascii_inc
.ascii_dot:
    mov byte [rdi], '.'
.ascii_inc:
    inc rdi
    inc r9d
    jmp .ascii_loop
.ascii_done:
    mov byte [rdi], '|'
    inc rdi
    mov byte [rdi], 10
    inc rdi
    ; write line
    mov r8, rcx                 ; save offset for after syscall
    mov rax, rdi
    sub rax, row_buf
    mov rdx, rax                ; length
    mov rax, 1                  ; sys_write
    mov rdi, 1                  ; fd 1
    mov rsi, row_buf
    syscall
    mov rcx, r8                 ; restore offset
    add rcx, 16                 ; next row
    jmp .row_process
.print_total_len:
    lea rdi, [row_buf]
    mov rax, rbx                ; total_len
    shr rax, 28
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    shr rax, 4
    and rax, 0xF
    movzx r10d, byte ptr [hexchars + rax]
    mov byte [rdi], r10b
    inc rdi
    mov byte [rdi], 10
    inc rdi
    mov rax, 1
    mov rdi, 1
    mov rsi, row_buf
    mov rdx, 9
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
