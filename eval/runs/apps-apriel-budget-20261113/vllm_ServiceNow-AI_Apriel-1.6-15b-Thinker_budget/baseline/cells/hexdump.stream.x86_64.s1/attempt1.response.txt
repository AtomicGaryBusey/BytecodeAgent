.global _start
.section .rodata
hex_chars:
    .byte "0123456789abcdef"
.section .bss
input_buf:
    .space 65536
out_buf:
    .space 128
.section .text
_start:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, 65536
    syscall
    mov r14, rax
    xor rcx, rcx
    sub rsp, 16
loop_start:
    cmp rcx, r14
    jge loop_end
    mov r9, r14
    sub r9, rcx
    mov r10, 16
    cmp r9, r10
    jge row_len_full
    mov r10, r9
row_len_full:
    lea rsi, [rip + out_buf]
    xor rdx, rdx
    mov rax, rcx
    shr rax, 28
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 24
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 20
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 16
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 12
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 8
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 4
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov byte [rsi + rdx], ' '
    inc rdx
    mov byte [rsi + rdx], ' '
    inc rdx
    lea r8, [rip + input_buf]
    add r8, rcx
    xor r11, r11
inner_loop:
    cmp r11, 16
    jge inner_done
    cmp r11, r10
    jge missing_slot
    movzx r12d, byte [r8 + r11]
    mov rax, r12
    shr rax, 4
    and rax, 0F
    movzx r13d, al
    mov al, byte [rip + hex_chars + r13]
    mov [rsi + rdx], al
    inc rdx
    mov rax, r12
    and rax, 0F
    movzx r13d, al
    mov al, byte [rip + hex_chars + r13]
    mov [rsi + rdx], al
    inc rdx
    mov byte [rsi + rdx], ' '
    inc rdx
    cmp r11, 7
    jne after_present
    mov byte [rsi + rdx], ' '
    inc rdx
after_present:
    inc r11
    jmp inner_loop
missing_slot:
    mov byte [rsi + rdx], ' '
    inc rdx
    mov byte [rsi + rdx], ' '
    inc rdx
    mov byte [rsi + rdx], ' '
    inc rdx
    cmp r11, 7
    jne after_missing
    mov byte [rsi + rdx], ' '
    inc rdx
after_missing:
    inc r11
    jmp inner_loop
inner_done:
    mov byte [rsi + rdx], ' '
    inc rdx
    mov byte [rsi + rdx], '|'
    inc rdx
    xor r12, r12
ascii_loop:
    cmp r12, r10
    jge ascii_done
    movzx r13d, byte [r8 + r12]
    cmp r13d, 0x20
    jb dot_ascii
    cmp r13d, 0x7e
    bg dot_ascii
    mov al, byte [r8 + r12]
    jmp store_ascii
dot_ascii:
    mov al, '.'
store_ascii:
    mov [rsi + rdx], al
    inc rdx
    inc r12
    jmp ascii_loop
ascii_done:
    mov byte [rsi + rdx], '|'
    inc rdx
    mov byte [rsi + rdx], 0x0a
    inc rdx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    syscall
    mov [rsp - 8], rcx
    mov [rsp - 4], r10
    mov rcx, [rsp - 8]
    mov r10, [rsp - 4]
    add rcx, r10
    add rsp, 8
    jmp loop_start
loop_end:
    lea rsi, [rip + out_buf]
    xor rdx, rdx
    mov rax, rcx
    shr rax, 28
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 24
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 20
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 16
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 12
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 8
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    shr rax, 4
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov rax, rcx
    and rax, 0F
    movzx r8d, al
    mov al, byte [rip + hex_chars + r8]
    mov [rsi + rdx], al
    inc rdx
    mov byte [rsi + rdx], 0x0a
    inc rdx
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
