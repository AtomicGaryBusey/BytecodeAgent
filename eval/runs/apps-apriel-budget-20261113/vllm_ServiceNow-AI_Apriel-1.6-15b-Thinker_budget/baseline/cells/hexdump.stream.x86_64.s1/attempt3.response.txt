.global _start
.text

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.section .bss
input: .space 65536
row_buf: .space 80

.text

_start:
    xor    r9, r9                ; total_len = 0
read_input:
    mov    rax, 0                ; sys_read
    mov    rdi, 0                ; fd 0
    mov    rsi, input
    lea    rsi, [rsi + r9]       ; ptr = input + total_len
    mov    rdx, 65536            ; request size
    syscall
    test   rax, rax
    jz     .done_read
    add    r9, rax               ; total_len += bytes read
    jmp    .read_input
.done_read:
    mov    rbx, r9               ; total_len in rbx
    xor    rdi, rdi              ; row_offset = 0

.row_loop:
    cmp    rdi, rbx
    jge    .print_len

    mov    rdx, rbx
    sub    rdx, rdi              ; remaining
    mov    rcx, 16
    cmp    rdx, rcx
    jge    .row_full
    mov    rcx, rdx
    jmp    .row_process
.row_full:
    ; rcx already 16
.row_process:
    lea    r8, [rip + input]     ; base address of input
    lea    r9, [r8 + rdi]        ; row_start = input + row_offset

    xor    r10, r10              ; idx = 0

    ; offset as 8 hex digits (most significant first)
    mov    r11, rdi
    shr    r11, 28
    mov    r12, 8
.offset_outer:
    mov    ecx, r11
    and    ecx, 0xF
    mov    al, cl
    cmp    al, 10
    jl     .offset_low
    add    al, 0x5B
    jmp    .offset_out
.offset_low:
    add    al, 0x30
.offset_out:
    mov    byte ptr [row_buf + r10], al
    inc    r10
    shr    r11, 4
    dec    r12
    jnz    .offset_outer

    ; two spaces after offset
    mov    byte ptr [row_buf + r10], ' '
    inc    r10
    mov    byte ptr [row_buf + r10], ' '
    inc    r10

    ; hex bytes loop (i = 0..15)
    xor    r13, r13              ; i = 0
.byte_loop:
    cmp    r13, rcx
    jl     .byte_print
    ; missing bytes: three spaces
    mov    byte ptr [row_buf + r10], ' '
    inc    r10
    mov    byte ptr [row_buf + r10], ' '
    inc    r10
    mov    byte ptr [row_buf + r10], ' '
    inc    r10
    jmp    .byte_after
.byte_print:
    movzx  r14d, byte ptr [r9 + r13]
    ; high nibble
    mov    r15d, r14d
    shr    r15d, 4
    and    r15d, 0xF
    mov    al, r15b
    cmp    al, 10
    jl     .high_low
    add    al, 0x5B
    jmp    .high_out
.high_low:
    add    al, 0x30
.high_out:
    mov    byte ptr [row_buf + r10], al
    inc    r10
    ; low nibble
    mov    r15d, r14d
    and    r15d, 0xF
    mov    al, r15b
    cmp    al, 10
    jl     .low_low
    add    al, 0x5B
    jmp    .low_out
.low_low:
    add    al, 0x30
.low_out:
    mov    byte ptr [row_buf + r10], al
    inc    r10
    ; space after byte
    mov    byte ptr [row_buf + r10], ' '
    inc    r10
.byte_after:
    cmp    r13, 7
    jne    .byte_next
    mov    byte ptr [row_buf + r10], ' '
    inc    r10
.byte_next:
    inc    r13
    cmp    r13, 16
    jl     .byte_loop

    ; pipe before ASCII
    mov    byte ptr [row_buf + r10], '|'
    inc    r10
    xor    r13, r13              ; ascii_i = 0
.ascii_loop:
    cmp    r13, rcx
    jge    .ascii_done
    movzx  r14d, byte ptr [r9 + r13]
    mov    al, r14b
    cmp    al, 0x20
    jb     .ascii_dot
    cmp    al, 0x7e
    ja     .ascii_dot
    mov    byte ptr [row_buf + r10], al
    jmp    .ascii_next
.ascii_dot:
    mov    byte ptr [row_buf + r10], '.'
.ascii_next:
    inc    r10
    inc    r13
    jmp    .ascii_loop
.ascii_done:
    mov    byte ptr [row_buf + r10], '|'
    inc    r10
    mov    byte ptr [row_buf + r10], 10   ; newline
    inc    r10

    ; write the row
    mov    rax, 1
    mov    rdi, 1
    lea    rsi, [rip + row_buf]
    mov    rdx, r10
    syscall

    add    rdi, 16
    jmp    .row_loop

.print_len:
    xor    r10, r10
    mov    r11, rbx
    shr    r11, 28
    mov    r12, 8
.len_outer:
    mov    ecx, r11
    and    ecx, 0xF
    mov    al, cl
    cmp    al, 10
    jl     .len_low
    add    al, 0x5B
    jmp    .len_out
.len_low:
    add    al, 0x30
.len_out:
    mov    byte ptr [row_buf + r10], al
    inc    r10
    shr    r11, 4
    dec    r12
    jnz    .len_outer
    mov    byte ptr [row_buf + r10], 10   ; newline
    inc    r10
    mov    rax, 1
    mov    rdi, 1
    lea    rsi, [rip + row_buf]
    mov    rdx, r10
    syscall

    mov    rax, 60
    xor    rdi, rdi
    syscall
