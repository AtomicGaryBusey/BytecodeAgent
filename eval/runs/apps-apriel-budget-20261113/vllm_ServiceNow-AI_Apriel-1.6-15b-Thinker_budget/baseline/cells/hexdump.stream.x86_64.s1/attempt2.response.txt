.global _start
.text

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.section .bss
input: .space 65536
row_buf: .space 16
outbuf: .space 80

.section .data
input_len: .long 0

.section .text

_start:
    xor    r8, r8                 ; r8 = 0, bytes read so far
read_loop:
    mov    rax, 0                 ; sys_read
    mov    rdi, 0                 ; stdin
    mov    rsi, input             ; buffer
    mov    rdx, 65536             ; max per read
    syscall
    test   eax, eax
    jz    read_done
    add    r8, eax                ; increase total
    mov    [input_len], r8
    jmp    read_loop
read_done:
    mov    rbx, r8                ; total length

    xor    rax, rax               ; current offset = 0
row_loop:
    cmp    rax, rbx
    jge    done_rows

    mov    r15, rax               ; save offset for this row

    mov    rdx, rbx
    sub    rdx, rax               ; remaining = total - offset
    mov    rcx, 16
    cmp    rdx, rcx
    jge    have_full_row
    mov    rcx, rdx               ; rcx = remaining (<16)
have_full_row:
    mov    r13, rcx               ; n = bytes to process (0..16)

    lea    rdi, [rel row_buf]     ; dest buffer
    mov    rsi, input
    add    rsi, rax               ; src = input + offset
    mov    rcx, r13               ; count
    rep movsb

    lea    rsi, [rel outbuf]      ; output buffer
    xor    r9, r9                 ; outpos = 0

    ; offset hex digits (8)
    mov    r10, r15
    shr    r10, 28
    mov    r11, 8
off_digit_loop:
    and    r10, 15
    movzx  rcx, r10b
    mov    al, byte ptr [hexchars + rcx]
    mov    byte ptr [rsi + r9], al
    inc    r9
    shr    r10, 4
    dec    r11
    jnz    off_digit_loop

    mov    byte ptr [rsi + r9], ' '
    inc    r9
    mov    byte ptr [rsi + r9], ' '
    inc    r9

    xor    r8, r8                 ; i = 0 for hex bytes
hex_byte_loop:
    cmp    r8, 15
    jg    after_hex
    cmp    r8, r13                ; i < n ?
    jl    have_byte
    ; no byte: three spaces
    mov    byte ptr [rsi + r9], ' '
    inc    r9
    mov    byte ptr [rsi + r9], ' '
    inc    r9
    mov    byte ptr [rsi + r9], ' '
    inc    r9
    jmp    after_byte
have_byte:
    mov    al, byte ptr [rdi + r8]
    mov    bl, al
    ; hi nibble
    shr    bl, 4
    movzx  rcx, bl
    mov    al, byte ptr [hexchars + rcx]
    mov    byte ptr [rsi + r9], al
    inc    r9
    ; lo nibble
    and    bl, 15
    movzx  rcx, bl
    mov    al, byte ptr [hexchars + rcx]
    mov    byte ptr [rsi + r9], al
    inc    r9
    mov    byte ptr [rsi + r9], ' '
    inc    r9
after_byte:
    cmp    r8, 7
    je    extra_space_hex
    jmp    inc_hex_i
extra_space_hex:
    mov    byte ptr [rsi + r9], ' '
    inc    r9
inc_hex_i:
    inc    r8
    jmp    hex_byte_loop
after_hex:
    ; ascii column header
    mov    byte ptr [rsi + r9], ' '
    inc    r9
    mov    byte ptr [rsi + r9], '|'
    inc    r9
    xor    r10, r10               ; ascii index = 0
ascii_loop:
    cmp    r10, r13
    jge    after_ascii
    mov    al, byte ptr [rdi + r10]
    cmp    al, 0x20
    jb    unprintable
    cmp    al, 0x7e
    jg    unprintable
    mov    byte ptr [rsi + r9], al
    jmp    inc_ascii
unprintable:
    mov    byte ptr [rsi + r9], '.'
inc_ascii:
    inc    r9
    inc    r10
    jmp    ascii_loop
after_ascii:
    mov    byte ptr [rsi + r9], '|'
    inc    r9
    mov    byte ptr [rsi + r9], '\n'
    inc    r9

    ; write the line
    mov    rax, 1                ; sys_write
    mov    rdi, 1                ; stdout
    mov    rsi, outbuf
    mov    rdx, r9               ; length
    syscall

    add    r15, r13              ; next offset
    mov    rax, r15              ; set for next iteration
    jmp    row_loop

done_rows:
    ; output total length line
    lea    rsi, [rel outbuf]
    xor    r9, r9
    mov    r10, rbx
    shr    r10, 28
    mov    r11, 8
len_digit_loop:
    and    r10, 15
    movzx  rcx, r10b
    mov    al, byte ptr [hexchars + rcx]
    mov    byte ptr [rsi + r9], al
    inc    r9
    shr    r10, 4
    dec    r11
    jnz    len_digit_loop
    mov    byte ptr [rsi + r9], '\n'
    mov    rax, 1
    mov    rdi, 1
    mov    rsi, outbuf
    mov    rdx, r9
    syscall

    xor    rdi, rdi
    mov    rax, 60              ; sys_exit
    syscall
