.globl _start
    .section .rodata
input_buf:
    .space 65536
    .align 4
    .section .text
_start:
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf   // x5 = buffer base
    mov x2, #0                    // offset = 0
.read_loop:
    mov #65536, x3               // MAX_INPUT
    sub x3, x2, x3               // remaining = MAX_INPUT - offset
    cmp x3, #0
    beq .done_reading
    // start address of chunk
    add x1, x5, x2               // x1 = base + offset
    mov #0, x0                  // fd = 0 (stdin)
    mov x2, x3                  // count = remaining
    mov #63, x8                 // sys_read
    svc
    mov x9, x0                  // bytes read
    cmp x9, #0
    beq .done_reading
    // write the bytes just read
    mov #1, x0                  // fd = 1 (stdout)
    // x1 = start address (unchanged)
    mov x2, x9                  // count = bytes read
    mov #64, x8                 // sys_write
    svc
    add x2, x9, x2              // offset += bytes read
    b .read_loop
.done_reading:
    mov #93, x8                 // sys_exit
    mov #0, x0                  // status 0
    svc
