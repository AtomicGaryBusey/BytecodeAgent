.section .rodata
space_char:
    .byte ' '
newline_char:
    .byte 0x0A
.bss
buf:
    .space 65536
in_word_flag:
    .zero 1
num_buf:
    .space 6
num_len:
    .space 4
.section .text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    xor r11, r11
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, buf
    mov rdx, 65536
    syscall
    mov rbx, rax
    test rbx, rbx
    jz done_reading
process_bytes:
    mov rdx, buf
    mov rcx, rbx
process_one:
    test rcx, rcx
    jz done_bytes
    mov al, byte ptr [rdx]
    inc r10
    cmp al, 0x0A
    jne .no_newline
    inc r8
=no_newline:
    mov bl, al
    cmp bl, ' '
    je .whitespace
    cmp bl, 9
    je .whitespace
    cmp bl, 10
    je .whitespace
    cmp bl, 11
    je .whitespace
    cmp bl, 12
    je .whitespace
    cmp bl, 13
    je .whitespace
    test r11, r11
    jne .nonword_update
    inc r9
    mov r11, 1
    jmp .advance
.nonword_update:
    mov r11, 1
    jmp .advance
.whitespace:
    xor r11, r11
    jmp .advance
.advance:
    inc rdx
    dec rcx
    jmp process_one
done_bytes:
    jmp read_loop
done_reading:
    mov rax, r8
    xor ecx, ecx
    test rax, rax
    jnz conv_loop
    mov byte ptr [num_buf], '0'
    mov dword ptr [num_len], 1
    mov ecx, 1
    jmp after_conv
conv_loop:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    inc ecx
    mov byte ptr [num_buf + ecx - 1], dl
    test rax, rax
    jnz conv_loop
after_conv:
    mov dword ptr [num_len], ecx
    mov rdi, 1
    mov rbx, rcx
    lea rsi, [num_buf + rcx - 1]
    mov rax, 1
out_loop:
    test rbx, rbx
    jz out_done
    mov rdx, 1
    syscall
    dec rbx
    inc rsi
    jmp out_loop
out_done:
    lea rsi, [rip + space_char]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    mov r8, r9
    mov rax, r8
    xor ecx, ecx
    test rax, rax
    jnz conv_loop2
    mov byte ptr [num_buf], '0'
    mov dword ptr [num_len], 1
    mov ecx, 1
    jmp after_conv2
conv_loop2:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    inc ecx
    mov byte ptr [num_buf + ecx - 1], dl
    test rax, rax
    jnz conv_loop2
after_conv2:
    mov dword ptr [num_len], ecx
    mov rdi, 1
    mov rbx, rcx
    lea rsi, [num_buf + rcx - 1]
    mov rax, 1
out_loop2:
    test rbx, rbx
    jz out_done2
    mov rdx, 1
    syscall
    dec rbx
    inc rsi
    jmp out_loop2
out_done2:
    lea rsi, [rip + space_char]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    mov r8, r10
    mov rax, r8
    xor ecx, ecx
    test rax, rax
    jnz conv_loop3
    mov byte ptr [num_buf], '0'
    mov dword ptr [num_len], 1
    mov ecx, 1
    jmp after_conv3
conv_loop3:
    xor edx, edx
    mov rbx, 10
    div rbx
    add dl, '0'
    inc ecx
    mov byte ptr [num_buf + ecx - 1], dl
    test rax, rax
    jnz conv_loop3
after_conv3:
    mov dword ptr [num_len], ecx
    mov rdi, 1
    mov rbx, rcx
    lea rsi, [num_buf + rcx - 1]
    mov rax, 1
out_loop3:
    test rbx, rbx
    jz out_done3
    mov rdx, 1
    syscall
    dec rbx
    inc rsi
    jmp out_loop3
out_done3:
    lea rsi, [rip + newline_char]
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
