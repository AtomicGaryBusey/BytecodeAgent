.section .rodata
space:
    .byte ' '
newline_char:
    .byte 0x0A
.section .bss
buf:
    .space 65536
digits:
    .space 6
.section .text
    .globl _start
_start:
    xor eax, eax
    xor r8, r8
    xor r9, r9
    xor r10, r10
read_loop:
    mov eax, 0
    mov rdi, 0
    lea rsi, [buf]
    mov rdx, 65536
    syscall
    mov rcx, eax
    test rcx, rcx
    jz .done_reading
    add r8, rcx
    mov rsi, buf
    mov rdx, rcx
    xor r11, r11
scan_loop:
    test rdx, rdx
    jz .after_scan
    mov al, byte [rsi]
    cmp al, 0x0A
    je .is_newline
    cmp al, 0x20
    je .whitespace
    cmp al, 0x09
    je .whitespace
    cmp al, 0x0B
    je .whitespace
    cmp al, 0x0C
    je .whitespace
    cmp al, 0x0D
    je .whitespace
    test r11, r11
    jne .nonword_continue
    inc r9
    mov r11, 1
    jmp .scan_next
.nonword_continue:
    mov r11, 1
    jmp .scan_next
.whitespace:
    xor r11, r11
    jmp .scan_next
.is_newline:
    inc r10
    xor r11, r11
    jmp .scan_next
scan_next:
    inc rsi
    dec rdx
    jmp scan_loop
.after_scan:
    jmp read_loop
.done_reading:
    mov rax, r10
    xor rdx, rdx
    lea rsi, [digits + 5]
    test rax, rax
    jnz .div_loop1
    dec rsi
    mov al, '0'
    mov [rsi], al
    jmp .calc_len1
.div_loop1:
    mov rcx, 10
    div rcx
    add rdx, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz .div_loop1
.calc_len1:
    lea rdx, [digits + 5]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [space]
    mov rdx, 1
    syscall
    mov rax, r9
    xor rdx, rdx
    lea rsi, [digits + 5]
    test rax, rax
    jnz .div_loop2
    dec rsi
    mov al, '0'
    mov [rsi], al
    jmp .calc_len2
.div_loop2:
    mov rcx, 10
    div rcx
    add rdx, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz .div_loop2
.calc_len2:
    lea rdx, [digits + 5]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [space]
    mov rdx, 1
    syscall
    mov rax, r8
    xor rdx, rdx
    lea rsi, [digits + 5]
    test rax, rax
    jnz .div_loop3
    dec rsi
    mov al, '0'
    mov [rsi], al
    jmp .calc_len3
.div_loop3:
    mov rcx, 10
    div rcx
    add rdx, 48
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz .div_loop3
.calc_len3:
    lea rdx, [digits + 5]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [newline_char]
    mov rdx, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
