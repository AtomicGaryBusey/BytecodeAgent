.section .rodata
space_char:
    .byte ' '
newline_char:
    .byte 0x0A
.section .data
in_word_flag:
    .zero 1
.section .bss
buf:
    .space 65536
.section .text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    mov byte [in_word_flag], 0
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buf]
    mov rdx, 65536
    syscall
    mov rcx, rax
    test rcx, rcx
    jz done_reading
    mov rdx, rcx
    lea rbx, [rip + buf]
    movzx r11d, byte ptr [in_word_flag]
process_loop:
    test rdx, rdx
    jz after_bytes
    movzx eax, byte ptr [rbx]
    inc r8
    cmp eax, 0x0A
    jne .not_newline
    inc r10
.not_newline:
    cmp eax, 0x20
    je .whitespace
    cmp eax, 0x09
    je .whitespace
    cmp eax, 0x0B
    je .whitespace
    cmp eax, 0x0C
    je .whitespace
    cmp eax, 0x0D
    je .whitespace
    test r11, r11
    jnz .already_in_word
    inc r9
    mov byte [in_word_flag], 1
.already_in_word:
    jmp .advance
.whitespace:
    mov byte [in_word_flag], 0
.advance:
    inc rbx
    dec rdx
    jmp process_loop
after_bytes:
    jmp read_loop
done_reading:
    mov rax, r10
    call write_decimal
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space_char]
    mov rdx, 1
    syscall
    mov rax, r9
    call write_decimal
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space_char]
    mov rdx, 1
    syscall
    mov rax, r8
    call write_decimal
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall
write_decimal:
    sub rsp, 10
    lea r9, [rsp+9]
    lea rdi, [rsp+9]
    xor r8, r8
    test rax, rax
    jnz .non_zero
    mov al, '0'
    mov [rdi], al
    inc r8
    sub rdi, 1
    jmp .output
.non_zero:
    mov rbx, 10
.divide_loop:
    test rax, rax
    jz .done_div
    cqo
    idiv rbx
    add al, '0'
    mov [rdi], al
    inc r8
    sub rdi, 1
    jmp .divide_loop
.done_div:
    lea rsi, [rdi+1]
.output:
    test r8, r8
    jz .output_done
    mov rax, 1
    mov rdi, 1
    mov rsi, rsi
    mov rdx, 1
    syscall
    inc rsi
    dec r8
    jmp .output
.output_done:
    add rsp, 10
    ret
