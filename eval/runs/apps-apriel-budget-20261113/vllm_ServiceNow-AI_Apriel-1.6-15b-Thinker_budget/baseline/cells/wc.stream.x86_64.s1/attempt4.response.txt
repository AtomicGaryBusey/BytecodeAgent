.section .rodata
ten:
    .int 10
space_char:
    .byte ' '
newline_char:
    .byte 0x0A
.bss
buf:
    .space 65536
in_word:
    .zero 1
num_buf1:
    .space 6
num_buf2:
    .space 6
num_buf3:
    .space 6
.text
.globl _start
_start:
    xor r8, r8
    xor r9, r9
    xor r10, r10
read_loop:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rip + buf]
    mov rdx, 65536
    syscall
    mov r11, rax
    test r11, r11
    jz read_done
    mov rcx, r11
    lea rdx, [rip + buf]
process_loop:
    test rcx, rcx
    jz read_done
    mov al, byte ptr [rdx]
    inc r10
    cmp al, 0x0A
    jne not_newline
    inc r8
not_newline:
    cmp al, 0x09
    je whitespace
    cmp al, 0x0A
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    cmp al, 0x20
    je whitespace
    movzx eax, byte ptr [rel in_word]
    test eax, 1
    jne word_already
    inc r9
word_already:
    mov byte ptr [rel in_word], 1
    jmp advance
whitespace:
    mov byte ptr [rel in_word], 0
advance:
    inc rdx
    dec rcx
    jmp process_loop
read_done:
    ; convert lines to buffer1 and write
    mov rdx, r8
    lea rsi, [rip + num_buf1]
    call convert
    mov rax, 1
    mov rdi, 1
    mov rdx, rcx
    syscall
    ; write space
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space_char]
    mov rdx, 1
    syscall
    ; convert words to buffer2 and write
    mov rdx, r9
    lea rsi, [rip + num_buf2]
    call convert
    mov rax, 1
    mov rdi, 1
    mov rdx, rcx
    syscall
    ; write space
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + space_char]
    mov rdx, 1
    syscall
    ; convert bytes to buffer3 and write
    mov rdx, r10
    lea rsi, [rip + num_buf3]
    call convert
    mov rax, 1
    mov rdi, 1
    mov rdx, rcx
    syscall
    ; write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    ; exit
    mov rax, 60
    xor rdi, rdi
    syscall
convert:
    xor rcx, rcx
    mov rax, rsi
    add rax, 5
    mov rsi, rax
conv_loop:
    test rdx, rdx
    jz conv_zero
    xor edx, edx
    mov eax, rdx
    div dword ptr [ten]
    inc rcx
    add dl, 0x30
    mov [rsi], dl
    dec rsi
    mov rdx, eax
    jmp conv_loop
conv_zero:
    inc rcx
    mov byte ptr [rsi], '0'
    dec rsi
    lea rsi, [rsi + 1]
    ret
