.global _start
.section .rodata
newline: .byte 10

.section .bss
buf: .space 65536
revbuf: .space 65536

.section .text
_start:
	adrp x20, buf
	add x20, x20, #0
	mov x21, #0
	mov x22, #4096

read_loop:
	mov x8, #63
	mov x0, #0
	mov x1, x20
	add x1, x1, x21
	mov x2, #4096
	svc #0
	cmp x0, #0
	beq read_done
	add x21, x21, x0
	b read_loop

read_done:
	mov x23, x21
	cmp x23, #0
	beq exit0

	adrp x25, revbuf
	add x25, x25, #0
	mov x26, #0

read_process:
	mov x24, x26

process_loop:
	cmp x24, x23
	beq after_input

	edit: // debug label
	ldrb w0, [x20, x24]
	cmp w0, #0x0A
	bne not_newline

	sub x27, x24, x26
	cmp x27, #0
	beq newline_only

	mov x28, #0
	sub x29, x24, #1

reverse_loop:
	cmp x29, x26
	blt reverse_done
	lnb: // debug label
	ldrb w1, [x20, x29]
	strb w1, [x25, x28]
	add x28, x28, #1
	sub x29, x29, #1
	b reverse_loop

reverse_done:
	mov x8, #64
	mov x0, #1
	mov x1, x25
	mov x2, x28
	svc #0

	adrp x30, newline
	add x30, x30, #0
	mov x8, #64
	mov x0, #1
	mov x1, x30
	mov x2, #1
	svc #0

	add x26, x24, #1
	mov x24, x26
	b process_loop

newline_only:
	adrp x30, newline
	add x30, x30, #0
	mov x8, #64
	mov x0, #1
	mov x1, x30
	mov x2, #1
	svc #0

	add x26, x24, #1
	mov x24, x26
	b process_loop

not_newline:
	add x24, x24, #1
	b process_loop

after_input:
	cmp x26, x23
	beq exit0

	sub x27, x23, x26
	mov x28, #0
	sub x29, x23, #1

final_reverse_loop:
	cmp x29, x26
	blt final_reverse_done
	ldrb w1, [x20, x29]
	strb w1, [x25, x28]
	add x28, x28, #1
	sub x29, x29, #1
	b final_reverse_loop

final_reverse_done:
	mov x8, #64
	mov x0, #1
	mov x1, x25
	mov x2, x28
	svc #0

	adrp x30, newline
	add x30, x30, #0
	mov x8, #64
	mov x0, #1
	mov x1, x30
	mov x2, #1
	svc #0

exit0:
	mov x8, #93
	mov x0, #0
	svc #0
