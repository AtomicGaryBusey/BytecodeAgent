.global _start
.section .rodata
newline: .byte 10
.section .bss
linebuf: .space 65536
tmp: .zero 1
.section .text
_start:
    mov w0, #0
    adrp x5, linebuf
    add x5, x5, #0
    adrp x6, tmp
    add x6, x6, #0
read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x6
    mov x2, #1
    svc #0
    cmp x0, #0
    beq eof
    ldrb w1, [x6]
    cmp w1, #0x0A
    beq newline_case
    strb w1, [x5, #0]
    add x5, x5, #1
    add w0, w0, #1
    b read_loop
newline_case:
    cmp w0, #0
    beq write_newline
    mov x3, w0
    sub x3, x3, #1
rev_loop:
    adrp x1, linebuf
    add x1, x1, x3, lsl #0
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0
    sub x3, x3, #1
    cmp x3, #0
    bgt rev_loop
write_newline:
    mov x8, #64
    mov x0, #1
    adrp x1, newline
    add x1, x1, #0
    mov x2, #1
    svc #0
    mov w0, #0
    adrp x5, linebuf
    add x5, x5, #0
    b read_loop
eof:
    cmp w0, #0
    beq exit_program
    mov x3, w0
    sub x3, x3, #1
rev_loop_eof:
    adrp x1, linebuf
    add x1, x1, x3, lsl #0
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0
    sub x3, x3, #1
    cmp x3, #0
    bgt rev_loop_eof
    mov x8, #64
    mov x0, #1
    adrp x1, newline
    add x1, x1, #0
    mov x2, #1
    svc #0
exit_program:
    mov x8, #93
    mov x0, #0
    svc #0
