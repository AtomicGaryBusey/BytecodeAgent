.global _start
.section .rodata
newline: .byte 10
.section .bss
linebuf: .space 65536
revbuf: .space 65536
readbuf: .space 4096
.section .text
_start:
    adrp x5, linebuf
    add x5, x5, #0
    adrp x6, readbuf
    add x6, x6, #0
    mov w4, #0        // current line length

read_main:
    mov x8, #63
    mov x0, #0
    mov x1, x6
    mov x2, #4096
    svc #0
    cmp x0, #0
    beq eof

    mov x7, x6         // pointer to readbuf
    mov x3, x0         // bytes read count

process_byte:
    cmp x3, #0
    beq process_done

    ldrb w1, [x7]      // load next byte
    add x7, x7, #1
    sub x3, x3, #1

    cmp w1, #0x0A
    beq newline_flush

    // store byte in linebuf
    add x2, x5, w4
    strb w1, [x2]
    add w4, w4, #1

    b process_byte

process_done:
    b read_main

newline_flush:
    // prepare reversal copy
    adrp x12, revbuf
    add x12, x12, #0
    mov x9, w4         // count
    mov x10, w4
    sub x10, x10, #1   // source index
    mov x11, #0         // destination index

rev_copy:
    cmp x9, #0
    beq rev_copy_done
    add x2, x5, x10    // source address
    ldrb w1, [x2]
    add x3, x12, x11   // destination address
    strb w1, [x3]
    sub x9, x9, #1
    sub x10, x10, #1
    add x11, x11, #1
    b rev_copy

rev_copy_done:
    // write reversed line (may be zero length)
    mov x8, #64
    mov x0, #1
    mov x1, x12
    mov x2, w4
    svc #0

    // write newline
    adrp x13, newline
    add x13, x13, #0
    mov x8, #64
    mov x0, #1
    mov x1, x13
    mov x2, #1
    svc #0

    mov w4, #0
    b read_main

eof:
    cmp w4, #0
    beq exit_clean

    // flush final line
    adrp x12, revbuf
    add x12, x12, #0
    mov x9, w4
    mov x10, w4
    sub x10, x10, #1
    mov x11, #0

rev_copy_final:
    cmp x9, #0
    beq rev_copy_final_done
    add x2, x5, x10
    ldrb w1, [x2]
    add x3, x12, x11
    strb w1, [x3]
    sub x9, x9, #1
    sub x10, x10, #1
    add x11, x11, #1
    b rev_copy_final

rev_copy_final_done:
    // write reversed line
    mov x8, #64
    mov x0, #1
    mov x1, x12
    mov x2, w4
    svc #0

    // write newline
    adrp x13, newline
    add x13, x13, #0
    mov x8, #64
    mov x0, #1
    mov x1, x13
    mov x2, #1
    svc #0

    // exit
    mov x8, #93
    mov x0, #0
    svc #0

exit_clean:
    mov x8, #93
    mov x0, #0
    svc #0
