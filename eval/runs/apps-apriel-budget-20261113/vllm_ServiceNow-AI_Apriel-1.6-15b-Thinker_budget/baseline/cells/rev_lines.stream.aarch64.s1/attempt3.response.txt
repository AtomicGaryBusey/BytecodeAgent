.global _start
.section .rodata
newline: .byte 10

.section .bss
buf: .space 65536
revbuf: .space 65536

.section .text
_start:
    adrp x19, buf
    add x19, x19, #0
    mov x21, #0

read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x19
    add x1, x1, x21
    mov x2, #4096
    svc #0
    cmp x0, #0
    beq read_done

    add x21, x21, x0
    b read_loop

read_done:
    cmp x21, #0
    beq exit_prog

    adrp x25, revbuf
    add x25, x25, #0
    mov x22, #0
    mov x23, #0
    mov x24, #0

scan_loop:
    cmp x22, x21
    beq scan_done

    ldrb w2, [x19, x22]
    cmp w2, #0x0A
    bne scan_not_newline

    sub x26, x22, x23
    cmp x26, #0
    beq newline_empty

    sub x27, x22, #1
    mov x28, x23
    mov x24, #0

reverse_copy:
    cmp x27, x28
    blt reverse_copy_done

    ldrb w3, [x19, x27]
    strb w3, [x25, x24]
    add x24, x24, #1
    sub x27, x27, #1
    b reverse_copy

reverse_copy_done:
    mov x8, #64
    mov x0, #1
    mov x1, x25
    mov x2, x24
    svc #0

    adrp x1, newline
    add x1, x1, #0
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0

    mov x24, #0
    add x23, x22, #1
    mov x22, x23
    b scan_not_newline

newline_empty:
    adrp x1, newline
    add x1, x1, #0
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0

    add x23, x22, #1
    mov x22, x23
    b scan_not_newline

scan_not_newline:
    add x22, x22, #1
    b scan_loop

scan_done:
    cmp x23, x21
    beq exit_prog

    sub x26, x21, x23
    cmp x26, #0
    beq exit_prog

    sub x27, x21, #1
    mov x28, x23
    mov x24, #0

reverse_copy_final:
    cmp x27, x28
    blt reverse_copy_final_done

    ldrb w3, [x19, x27]
    strb w3, [x25, x24]
    add x24, x24, #1
    sub x27, x27, #1
    b reverse_copy_final

reverse_copy_final_done:
    mov x8, #64
    mov x0, #1
    mov x1, x25
    mov x2, x24
    svc #0

    adrp x1, newline
    add x1, x1, #0
    mov x8, #64
    mov x0, #1
    mov x2, #1
    svc #0

exit_prog:
    mov x8, #93
    mov x0, #0
    svc #0

    mov x8, #94
    mov x0, #0
    svc #0
