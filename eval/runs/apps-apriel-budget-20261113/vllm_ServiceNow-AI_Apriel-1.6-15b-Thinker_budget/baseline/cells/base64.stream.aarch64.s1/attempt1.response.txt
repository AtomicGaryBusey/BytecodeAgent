.equ INPUT_MAX, 49152
.equ READ_BUF, 4096
.equ LINE_LEN, 76

.section .rodata
alphabet:
    .string "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
input_buf:
    .space INPUT_MAX
output_buf:
    .space 66400

.global _start
.text
.align 16
_start:
    adrp x13, input_buf
    add x13, x13, :lo12:input_buf
    mov w9, #0          // total_in = 0
    mov w6, #INPUT_MAX // remaining max
    mov w4, #READ_BUF  // chunk size

.read_loop:
    sub w3, w6, w9
    cmp w3, w4
    b.ge .set_n_readbuf
    mov w2, w3
    b .read
.set_n_readbuf:
    mov w2, w4
.read:
    add x1, x13, w9
    mov w0, #0
    mov w8, #63
    svc #0
    mov w5, x0
    cbnz w5, .read_update
    b .read_done
.read_update:
    add w9, w9, w5
    b .read_loop

.read_done:
    cbnz w9, .encode_start
    mov w0, #0
    mov w8, #93
    svc #0

.encode_start:
    adrp x14, output_buf
    add x14, x14, :lo12:output_buf
    adrp x31, alphabet
    add x31, x31, :lo12:alphabet
    mov w10, #0
    mov w11, #0
    mov w12, #0

.encode_main:
    sub w0, w9, w10
    cmp w0, #3
    ble .final_group

    add x0, x13, x10
    ldrb w16, [x0]
    ldrb w17, [x0, #1]
    ldrb w18, [x0, #2]

    // v0 = (b0 >> 2) & 0x3F
    lsr w19, w16, #2
    and w19, w19, #0x3F

    // v1 = ((b0 & 0x3) << 4) | ((b1 >> 4) & 0x3F)
    and w20, w16, #0x3
    shl w20, w20, #4
    lsr w21, w17, #4
    and w21, w21, #0x3F
    orr w22, w20, w21

    // v2 = ((b1 & 0xF) << 2) | ((b2 >> 6) & 0x3F)
    and w23, w17, #0xF
    shl w23, w23, #2
    lsr w24, w18, #6
    and w24, w24, #0x3F
    orr w25, w23, w24

    // v3 = (b2 & 0x3F)
    and w26, w18, #0x3F

    // output v0
    add x2, x31, w19
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new0
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new0:

    // output v1
    add x2, x31, w22
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new1
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new1:

    // output v2
    add x2, x31, w25
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new2
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new2:

    // output v3
    add x2, x31, w26
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new3
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new3:

    add w10, w10, #3
    b .encode_main

.final_group:
    cmp w0, #1
    beq .final_one
    cmp w0, #2
    beq .final_two
    b .finalize

.final_one:
    add x0, x13, x10
    ldrb w16, [x0]

    // v0 = (b0 >> 2) & 0x3F
    lsr w19, w16, #2
    and w19, w19, #0x3F

    // output v0
    add x2, x31, w19
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_one0
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_one0:

    // v1 = ((b0 & 0x3) << 4) & 0x3F
    and w20, w16, #0x3
    shl w20, w20, #4
    and w20, w20, #0x3F

    // output v1
    add x2, x31, w20
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_one1
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_one1:

    // padding '='
    mov w27, #61
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_one2
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_one2:

    // second padding '='
    mov w27, #61
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_one3
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_one3:

    add w10, w10, #1
    b .finalize

.final_two:
    add x0, x13, x10
    ldrb w16, [x0]
    ldrb w17, [x0, #1]

    // v0 = (b0 >> 2) & 0x3F
    lsr w19, w16, #2
    and w19, w19, #0x3F

    // output v0
    add x2, x31, w19
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_two0
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_two0:

    // v1 = ((b0 & 0x3) << 4) | ((b1 >> 4) & 0x3F)
    and w20, w16, #0x3
    shl w20, w20, #4
    lsr w21, w17, #4
    and w21, w21, #0x3F
    orr w22, w20, w21

    // output v1
    add x2, x31, w22
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_two1
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_two1:

    // v2 = ((b1 & 0xF) << 2) & 0x3F
    and w23, w17, #0xF
    shl w23, w23, #2
    and w23, w23, #0x3F

    // output v2
    add x2, x31, w23
    ldrb w27, [x2, #0]
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_two2
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_two2:

    // padding '='
    mov w27, #61
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    add w12, w12, #1
    cmp w12, #LINE_LEN
    bne .no_new_two3
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w12, #0
.no_new_two3:

    add w10, w10, #2
    b .finalize

.finalize:
    cbnz w12, .add_last_newline
    mov w0, #1
    mov x1, x14
    mov x2, x11
    mov w8, #64
    svc #0
    mov w0, #0
    mov w8, #93
    svc #0

.add_last_newline:
    mov w27, #0x0A
    strb w27, [x14, x11, LSL #0]
    add w11, w11, #1
    mov w0, #1
    mov x1, x14
    mov x2, x11
    mov w8, #64
    svc #0
    mov w0, #0
    mov w8, #93
    svc #0
