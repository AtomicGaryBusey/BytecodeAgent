.global _start
.equ INPUT_BUF_SIZE, 65536
.equ STACK_SIZE, 8192
.equ ERROR_MSG_LEN, 6
.section .text
_start:
    xor r12, r12          ; buf_pos = 0
    xor r13, r13          ; total_len = 0
read_loop:
    mov rax, 0            ; sys_read
    mov rdi, 0            ; stdin
    mov rsi, input_buf
    add rsi, r12
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r12
    syscall
    test rax, rax
    jz .read_done
    mov r14, rax
    add r12, r14
    add r13, r14
    cmp r12, INPUT_BUF_SIZE
    jl .read_loop
.read_done:
    mov r8, input_buf     ; line pointer
    mov r9, r13           ; total_len
    jmp .outer_loop

.outer_loop:
    cmp r8, r9
    je .exit_program

    lea r10, [rip + .stack_end]   ; stack_ptr = top
    xor r11, r11                 ; stack_sz = 0

skip_sep_loop:
    mov al, byte [r8]
    test al, al
    jz .line_end
    cmp al, ' '
    je .skip_sep_next
    cmp al, '\t'
    je .skip_sep_next
    cmp al, '\n'
    je .empty_line_error
    jmp .token_start

.skip_sep_next:
    inc r8
    jmp skip_sep_loop

.empty_line_error:
    jmp .handle_error

.token_start:
    mov r12, r8                 ; token_start
.token_loop:
    mov al, byte [r8]
    cmp al, ' '
    je .token_done
    cmp al, '\t'
    je .token_done
    cmp al, '\n'
    je .token_done
    cmp r8, r9
    je .token_done
    inc r8
    jmp .token_loop

.token_done:
    mov r13, r8
    sub r13, r12                ; token_len in r13
    mov al, byte [r12]
    cmp r13, 1
    jne .proc_literal
    cmp al, '+'
    je .proc_operator_plus
    cmp al, '-'
    je .proc_operator_minus
    cmp al, '*'
    je .proc_operator_mul
    cmp al, '/'
    je .proc_operator_div
    cmp al, '%'
    je .proc_operator_mod
    jmp .handle_error

.proc_literal:
    mov bl, byte [r12]
    cmp bl, '-'
    je .literal_with_sign
    mov r14, r12                ; start of digits
    jmp .literal_parse

.literal_with_sign:
    cmp r13, 1
    je .invalid_literal
    mov r14, r12
    inc r14
    jmp .literal_parse

.literal_parse:
    cmp r14, r13
    je .invalid_literal
    xor rdx, rdx                ; digit count
    xor rax, rax                ; result
.literal_digit_loop:
    cmp rdx, 18
    jge .invalid_literal
    cmp r14, r13
    jge .literal_done
    mov cl, byte [r14]
    cmp cl, '0'
    jb .invalid_token
    cmp cl, '9'
    ja .invalid_token
    sub cl, '0'
    movzx rcx, cl
    imul rax, 10
    add rax, rcx
    inc rdx
    inc r14
    jmp .literal_digit_loop
.literal_done:
    mov bl, byte [r12]
    cmp bl, '-'
    je .apply_sign_negative
    jmp .push_literal

.apply_sign_negative:
    neg rax
    jmp .push_literal

.push_literal:
    dec r10
    mov qword [r10], rax
    inc r11
    jmp .after_token

.proc_operator_plus:
    cmp r11, 2
    jl .handle_error
    mov rbx, [r10]
    add r10, 8
    sub r11, 1
    mov rax, [r10]
    add r10, 8
    sub r11, 1
    add rax, rbx
    dec r10
    mov qword [r10], rax
    inc r11
    jmp .after_token

.proc_operator_minus:
    cmp r11, 2
    jl .handle_error
    mov rbx, [r10]
    add r10, 8
    sub r11, 1
    mov rax, [r10]
    add r10, 8
    sub r11, 1
    sub rax, rbx
    dec r10
    mov qword [r10], rax
    inc r11
    jmp .after_token

.proc_operator_mul:
    cmp r11, 2
    jl .handle_error
    mov rbx, [r10]
    add r10, 8
    sub r11, 1
    mov rax, [r10]
    add r10, 8
    sub r11, 1
    imul rax, rbx
    dec r10
    mov qword [r10], rax
    inc r11
    jmp .after_token

.proc_operator_div:
    cmp r11, 2
    jl .handle_error
    mov rbx, [r10]
    add r10, 8
    sub r11, 1
    mov rax, [r10]
    add r10, 8
    sub r11, 1
    cmp rbx, 0
    je .handle_error
    cmp rax, -9223372036854775808
    jne .div_normal
    cmp rbx, -1
    je .handle_error
.div_normal:
    mov rdi, rbx
    mov rax, rax
    cqo
    idiv rdi
    dec r10
    mov qword [r10], rax
    inc r11
    jmp .after_token

.proc_operator_mod:
    cmp r11, 2
    jl .handle_error
    mov rbx, [r10]
    add r10, 8
    sub r11, 1
    mov rax, [r10]
    add r10, 8
    sub r11, 1
    cmp rbx, 0
    je .handle_error
    cmp rax, -9223372036854775808
    jne .mod_normal
    cmp rbx, -1
    je .handle_error
.mod_normal:
    mov rdi, rbx
    mov rax, rax
    cqo
    idiv rdi
    mov rax, rdx
    dec r10
    mov qword [r10], rax
    inc r11
    jmp .after_token

.after_token:
    cmp r8, r9
    je .line_end
    mov al, byte [r8]
    cmp al, '\n'
    je .line_end
    cmp al, ' '
    je .skip_spaces
    cmp al, '\t'
    je .skip_spaces
    jmp .token_start

.skip_spaces:
    inc r8
    jmp .skip_spaces

.line_end:
    cmp r11, 1
    jne .handle_error
    mov rax, [r10]
    test rax, rax
    jge .print_non_negative
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + dash_char]
    mov rdx, 1
    syscall
    mov rsi, rax
    neg rsi
    jmp .convert_abs
.print_non_negative:
    mov rsi, rax
.convert_abs:
    xor r8, r8                 ; digit count = 0
    test rsi, rsi
    jz .single_zero
.convert_loop:
    xor rcx, rcx
    mov rax, rsi
    mov rbx, 10
    div rbx
    mov cl, rdx
    add cl, '0'
    mov byte [digit_buf + r8*1], cl
    inc r8
    mov rsi, rax
    test rsi, rsi
    jnz .convert_loop
.single_zero:
    mov byte [digit_buf], '0'
    inc r8
    mov r9, r8
    dec r9
    xor r10, r10               ; loop index
.output_digits_loop:
    cmp r10, r9
    jg .output_newline
    lea rsi, [digit_buf + r9*1]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r9
    inc r10
    jmp .output_digits_loop
.output_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    cmp r8, r9
    jge .skip_nl_check
    mov al, byte [r8]
    cmp al, '\n'
    jne .skip_nl_check
skip_nl_check:
    inc r8
    jmp .outer_loop

.handle_error:
    inc r8
    cmp r8, r9
    jge .skip_done_err
skip_err_char:
    mov al, byte [r8]
    cmp al, '\n'
    jne .skip_err_char
skip_done_err:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, ERROR_MSG_LEN
    syscall
    cmp r8, r9
    jge .skip_nl_err
    mov al, byte [r8]
    cmp al, '\n'
    jne .skip_nl_err
skip_nl_err:
    inc r8
    jmp .outer_loop

.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.invalid_literal:
    jmp .handle_error
.invalid_token:
    jmp .handle_error

.section .rodata
error_msg: .asciz "error\n"
dash_char: .byte '-'
newline_char: .byte 0x0A

.section .bss
.align 8
input_buf: .space INPUT_BUF_SIZE
.stack: .space STACK_SIZE
.stack_end:
digit_buf: .space 20
