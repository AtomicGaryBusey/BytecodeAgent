.global _start
.equ INPUT_BUF_SIZE, 65536
.equ STACK_SIZE, 4096
.equ ERROR_MSG_LEN, 6
.equ MIN_INT_STR_LEN, 19

.section .text
_start:
    ;--- read all input into input_buf ---------------------------------
    xor rdx, rdx
    mov r8, input_buf          ; start of buffer
    mov r9, input_buf          ; current write pointer
    mov rax, 0                 ; sys_read
    mov rdi, 0
    mov r10, INPUT_BUF_SIZE    ; remaining space
read_loop:
    mov rsi, r9
    mov rdx, r10
    syscall
    test rax, rax
    jz .read_done
    mov r11, rax
    add r9, rax
    sub r10, rax
    cmp r10, 0
    jne .read_loop
.read_done:
    ; r8 = start, r9 = end (r9 points just after last byte)
    jmp .outer_loop

.outer_loop:
    cmp r8, r9
    je .exit_program

    ; reset stack for this line
    lea r10, [rel stack_end]   ; r10 = stack pointer (one past)
    xor r11, r11               ; stack size = 0

skip_leading:
    cmp r8, r9
    je .exit_program
    mov al, byte ptr [r8]
    cmp al, 0x0A
    je .empty_line_error
    cmp al, ' '
    je .skip_leading_next
    cmp al, 0x09
    je .skip_leading_next
.skip_leading_next:
    inc r8
    jmp skip_leading

    ; token start address
    mov r12, r8                ; r12 = token_start

.token_scan:
    cmp r8, r9
    je .token_done
    mov al, byte ptr [r8]
    cmp al, 0x0A
    je .token_done
    cmp al, ' '
    je .token_done
    cmp al, 0x09
    je .token_done
    inc r8
    jmp .token_scan

.token_done:
    ; token length = r8 - r12
    mov r13, r8
    sub r13, r12               ; r13 = len

    ; first character
    mov al, byte ptr [r12]
    cmp al, '+'
    je .proc_op_add
    cmp al, '-'
    ; could be minus operator or negative literal
    cmp r13, 1
    je .proc_op_minus
    ; length >1 and starts with '-': literal
    jmp .proc_literal

    ;--- process literal ------------------------------------------------
proc_literal:
    ; r12 = start, r13 = length
    ; check that first char is digit or '-'
    mov al, byte ptr [r12]
    cmp al, '-'
    jne .literal_no_sign
    ; sign present
    mov r15, 1                 ; sign flag
    lea r14, [r12+1]           ; r14 = start of digits
    cmp r14, r8                ; at least one digit?
    jge .invalid_token
    xor r9, r9                 ; value
.literal_no_sign:
    xor r15, r15               ; sign flag = 0
    mov r14, r12               ; start of digits
    xor r9, r9                 ; value

    ; parse digits
    xor r16, r16               ; digit count
.parse_digit:
    cmp r14, r8
    jge .parse_done
    mov al, byte ptr [r14]
    cmp al, '0'
    jb .invalid_token
    cmp al, '9'
    jb .invalid_token
    sub al, '0'
    movzx rbx, al
    imul r9, 10
    add r9, rbx
    inc r16
    cmp r16, 19                ; >18 digits -> error
    jl .parse_digit
    jmp .invalid_token
.parse_done:
    test r15, r15
    jz .push_literal
    neg r9
    jmp .push_literal

.push_literal:
    sub r10, 8
    mov qword ptr [r10], r9
    inc r11
    jmp .after_token

    ;--- process operators ---------------------------------------------
proc_op_add:
    cmp r11, 2
    jl .handle_error
    ; pop b
    sub r11, 1
    sub r10, 8
    mov rbx, qword ptr [r10]
    ; pop a
    sub r11, 1
    sub r10, 8
    mov rax, qword ptr [r10]
    add rax, rbx
    ; push result
    sub r10, 8
    mov qword ptr [r10], rax
    inc r11
    jmp .after_token

proc_op_minus:
    cmp r11, 2
    jl .handle_error
    sub r11, 1
    sub r10, 8
    mov rbx, qword ptr [r10]
    sub r11, 1
    sub r10, 8
    mov rax, qword ptr [r10]
    sub rax, rbx
    sub r10, 8
    mov qword ptr [r10], rax
    inc r11
    jmp .after_token

proc_op_mul:
    cmp r11, 2
    jl .handle_error
    sub r11, 1
    sub r10, 8
    mov rbx, qword ptr [r10]
    sub r11, 1
    sub r10, 8
    mov rax, qword ptr [r10]
    imul rax, rbx
    sub r10, 8
    mov qword ptr [r10], rax
    inc r11
    jmp .after_token

proc_op_div:
    cmp r11, 2
    jl .handle_error
    sub r11, 1
    sub r10, 8
    mov rbx, qword ptr [r10]          ; divisor b
    cmp rbx, 0
    je .handle_error
    sub r11, 1
    sub r10, 8
    mov rax, qword ptr [r10]          ; dividend a
    cmp rax, -9223372036854775808
    jne .div_ok
    cmp rbx, -1
    je .handle_error
.div_ok:
    xor rdx, rdx
    mov rdi, rbx
    idiv rdi                         ; rax = a / b
    sub r10, 8
    mov qword ptr [r10], rax
    inc r11
    jmp .after_token

proc_op_mod:
    cmp r11, 2
    jl .handle_error
    sub r11, 1
    sub r10, 8
    mov rbx, qword ptr [r10]          ; divisor b
    cmp rbx, 0
    je .handle_error
    sub r11, 1
    sub r10, 8
    mov rax, qword ptr [r10]          ; dividend a
    cmp rax, -9223372036854775808
    jne .mod_ok
    cmp rbx, -1
    je .handle_error
.mod_ok:
    xor rdx, rdx
    mov rdi, rbx
    idiv rdi                         ; rax = a / b, rdx = a % b
    sub r10, 8
    mov qword ptr [r10], rdx          ; remainder
    inc r11
    jmp .after_token

.after_token:
    ; skip spaces/tabs after token
    cmp r8, r9
    je .line_end
    mov al, byte ptr [r8]
    cmp al, 0x0A
    je .line_end
    cmp al, ' '
    je .skip_spaces_next
    cmp al, 0x09
    je .skip_spaces_next
.skip_spaces_next:
    inc r8
    jmp .after_token

.line_end:
    cmp r11, 1
    jne .handle_error
    ; result is on stack top
    sub r10, 8
    mov rax, qword ptr [r10]

    ;--- print result ---------------------------------------------------
    cmp rax, -9223372036854775808
    je .print_min_int
    test rax, rax
    jns .print_nonneg
    ; negative non-min
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel dash_char]
    mov rdx, 1
    syscall
    neg rax
    jmp .print_nonneg

.print_nonneg:
    ; print number (or zero)
    xor r13, r13
    test rax, rax
    jz .print_zero
.print_digit_loop:
    xor rdx, rdx
    mov rbx, 10
    idiv rbx
    mov rcx, rdx
    add rcx, '0'
    mov byte ptr [digit_buf + r13], cl
    inc r13
    test rax, rax
    jnz .print_digit_loop
.print_zero:
    mov byte ptr [digit_buf], '0'
    mov r13, 1
    ; output digits reversed
    mov r14, r13
    dec r14
.output_one_digit:
    test r14, r14
    jz .output_newline
    lea rsi, [digit_buf + r14]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r14
    jmp .output_one_digit
.output_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel newline_char]
    mov rdx, 1
    syscall
    jmp .after_print

.print_min_int:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel min_int_str]
    mov rdx, MIN_INT_STR_LEN
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel newline_char]
    mov rdx, 1
    syscall
    jmp .after_print

.after_print:
    ; skip newline in input if present
    cmp r8, r9
    je .line_done
    mov al, byte ptr [r8]
    cmp al, 0x0A
    je .skip_input_nl
    jmp .line_done
.skip_input_nl:
    inc r8
.line_done:
    jmp .outer_loop

.handle_error:
    ; skip to end of line (including newline)
    .skip_err_loop:
        cmp r8, r9
        je .skip_err_done
        mov al, byte ptr [r8]
        cmp al, 0x0A
        jne .skip_err_next
        jmp .skip_err_done
    .skip_err_next:
        inc r8
        jmp .skip_err_loop
    .skip_err_done:
    ; write "error\n"
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel error_msg]
    mov rdx, ERROR_MSG_LEN
    syscall
    jmp .outer_loop

.empty_line_error:
    inc r8
    jmp .handle_error

.invalid_token:
    jmp .handle_error

.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
.error_msg: .asciz "error\n"
.dash_char: .byte '-'
.newline_char: .byte 0x0A
.min_int_str: .asciz "9223372036854775808"

.section .bss
.align 8
input_buf: .space INPUT_BUF_SIZE
stack: .space STACK_SIZE
.stack_end:

.section .data
.digit_buf: .space 20
