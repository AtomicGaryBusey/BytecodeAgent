.global _start
.equ INPUT_BUF_SIZE, 65536
.equ STACK_SIZE, 8192
.equ ERROR_MSG_LEN, 6
.equ MIN_INT_STR_LEN, 19
.section .text
_start:
    xor r8, r8
    xor r9, r9
read_input:
    mov rax, 0
    mov rdi, 0
    mov rsi, input_buf
    add rsi, r8
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r8
    syscall
    test rax, rax
    jz .read_done
    mov r10, rax
    add r8, r10
    add r9, r10
    cmp r8, INPUT_BUF_SIZE
    jl .read_input
.read_done:
    lea r8, [rip + input_buf]
    jmp .outer_loop
.outer_loop:
    cmp r8, r9
    je .exit_program
    lea r10, [rip + stack_end]
    xor r11, r11
skip_leading:
    cmp r8, r9
    je .exit_program
    mov al, byte [r8]
    cmp al, '\n'
    je .empty_line_error
    cmp al, ' '
    je .skip_leading_next
    cmp al, '\t'
    je .skip_leading_next
.skip_leading_next:
    inc r8
    jmp skip_leading
    mov r13, r8
.token_scan:
    cmp r8, r9
    je .token_done
    mov al, byte [r8]
    cmp al, '\n'
    je .token_done
    cmp al, ' '
    je .token_done
    cmp al, '\t'
    je .token_done
    inc r8
    jmp .token_scan
.token_done:
    lea rdx, [r13+1]
    cmp r8, rdx
    jne .proc_literal
    mov al, byte [r13]
    cmp al, '+'
    je .proc_operator_plus
    cmp al, '-'
    je .proc_operator_minus
    cmp al, '*'
    je .proc_operator_mul
    cmp al, '/'
    je .proc_operator_div
    cmp al, '%'
    je .proc_operator_mod
    jmp .handle_error
.proc_literal:
    cmp r8, r13
    je .invalid_literal
    mov al, byte [r13]
    cmp al, '-'
    je .literal_has_sign
    xor r15, r15
    mov r14, r13
    xor rdx, rdx
    xor rax, rax
    jmp .literal_parse
.literal_has_sign:
    mov r15, 1
    lea rdx, [r13+1]
    cmp r8, rdx
    jle .invalid_literal
    mov r14, rdx
    xor rdx, rdx
    xor rax, rax
    jmp .literal_parse
.literal_parse:
    cmp rdx, 18
    jge .invalid_literal
    cmp r14, r8
    jge .literal_done
    mov cl, byte [r14]
    cmp cl, '0'
    jb .invalid_literal
    cmp cl, '9'
    jb .invalid_literal
    sub cl, '0'
    movzx rcx, cl
    imul rax, 10
    add rax, rcx
    inc rdx
    inc r14
    jmp .literal_parse
.literal_done:
    test r15, r15
    jz .push_literal
    neg rax
    jmp .push_literal
.push_literal:
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_plus:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    add rax, rbx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_minus:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    sub rax, rbx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_mul:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    imul rax, rbx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_div:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    cmp rbx, 0
    je .handle_error
    cmp rax, -9223372036854775808
    jne .div_ok
    cmp rbx, -1
    je .handle_error
.div_ok:
    xor rdx, rdx
    mov rdi, rbx
    idiv rdi
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_mod:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    cmp rbx, 0
    je .handle_error
    cmp rax, -9223372036854775808
    jne .mod_ok
    cmp rbx, -1
    je .handle_error
.mod_ok:
    xor rdx, rdx
    mov rdi, rbx
    idiv rdi
    sub r10, 8
    mov qword [r10], rdx
    inc r11
    jmp .after_token
.after_token:
    cmp r8, r9
    je .line_end
    mov al, byte [r8]
    cmp al, '\n'
    je .line_end
    cmp al, ' '
    je .skip_spaces_next
    cmp al, '\t'
    je .skip_spaces_next
.skip_spaces_next:
    inc r8
    jmp .after_token
.line_end:
    cmp r11, 1
    jne .handle_error
    mov rax, qword [r10]
    jmp .print_result
.print_result:
    test rax, rax
    jns .print_nonneg
    cmp rax, -9223372036854775808
    jne .print_negative_normal
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_int_str]
    mov rdx, MIN_INT_STR_LEN
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    jmp .after_print
.print_negative_normal:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + dash_char]
    mov rdx, 1
    syscall
    neg rax
    jmp .print_number
.print_nonneg:
    jmp .print_number
.print_number:
    xor r15, r15
    test rax, rax
    jz .print_zero
    .print_digit_loop:
        mov rbx, 10
        xor rdx, rdx
        div rbx
        mov rcx, rdx
        add cl, '0'
        mov byte [digit_buf + r15], cl
        inc r15
        test rax, rax
        jnz .print_digit_loop
    .print_zero:
        mov byte [digit_buf], '0'
        mov r15, 1
    .output_digits:
        mov r16, r15
        dec r16
        jmp .output_one_digit_loop
    .output_one_digit_loop:
        cmp r16, 0
        je .output_newline
        lea rsi, [digit_buf + r16]
        mov rax, 1
        mov rdi, 1
        mov rdx, 1
        syscall
        dec r16
        jmp .output_one_digit_loop
    .output_newline:
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + newline_char]
        mov rdx, 1
        syscall
.after_print:
    cmp r8, r9
    je .line_done
    mov al, byte [r8]
    cmp al, '\n'
    je .skip_line_nl
    jmp .line_done
.skip_line_nl:
    inc r8
.line_done:
    jmp .outer_loop
.handle_error:
    .skip_err_loop:
        cmp r8, r9
        je .skip_err_done
        mov al, byte [r8]
        cmp al, '\n'
        jne .skip_err_next
        jmp .skip_err_done
    .skip_err_next:
        inc r8
        jmp .skip_err_loop
    .skip_err_done:
        mov rax, 1
        mov rdi, 1
        lea rsi, [rip + error_msg]
        mov rdx, ERROR_MSG_LEN
        syscall
        cmp r8, r9
        je .handle_error_done
        mov al, byte [r8]
        cmp al, '\n'
        je .handle_error_skip_nl
        jmp .handle_error_done
    .handle_error_skip_nl:
        inc r8
    .handle_error_done:
        jmp .outer_loop
.empty_line_error:
    inc r8
    jmp .handle_error
.invalid_literal:
    jmp .handle_error
.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
.section .rodata
error_msg: .asciz "error\n"
dash_char: .byte '-'
newline_char: .byte 0x0A
min_int_str: .asciz "9223372036854775808"
.section .bss
.align 8
input_buf: .space INPUT_BUF_SIZE
stack: .space STACK_SIZE
.stack_end:
.section .data
digit_buf: .space 20
