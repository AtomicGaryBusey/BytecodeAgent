.global _start
.equ INPUT_BUF_SIZE, 65536
.equ STACK_SIZE, 8192
.equ ERROR_MSG_LEN, 6
.equ MIN_INT_STR_LEN, 19
.section .text
_start:
    xor r8, r8          ; buf offset
    xor r9, r9          ; total read
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, input_buf
    add rsi, r8
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r8
    syscall
    test rax, rax
    jz .read_done
    mov r10, rax
    add r8, r10
    add r9, r10
    cmp r8, INPUT_BUF_SIZE
    jl .read_loop
.read_done:
    lea r8, [rip + input_buf]   ; r8 = buffer start
    jmp .outer_loop
.outer_loop:
    cmp r8, r9
    je .exit_program
    lea r10, [rip + stack_end]   ; stack ptr
    xor r11, r11                 ; stack size
skip_leading:
    mov al, byte [r8]
    test al, al
    jz .outer_loop
    cmp al, ' '
    je .skip_leading_next
    cmp al, '\t'
    je .skip_leading_next
    cmp al, '\n'
    je .empty_line_error
    jmp .token_start
.skip_leading_next:
    inc r8
    jmp skip_leading
.empty_line_error:
    jmp .handle_error
.token_start:
    mov r13, r8                 ; token start
.token_loop:
    cmp r8, r9
    je .token_done
    mov al, byte [r8]
    cmp al, ' '
    je .token_done
    cmp al, '\t'
    je .token_done
    cmp al, '\n'
    je .token_done
    inc r8
    jmp .token_loop
.token_done:
    cmp r8, r13
    jne .not_single_char
    mov al, byte [r13]
    cmp al, '+'
    je .proc_operator_plus
    cmp al, '-'
    je .proc_operator_minus
    cmp al, '*'
    je .proc_operator_mul
    cmp al, '/'
    je .proc_operator_div
    cmp al, '%'
    je .proc_operator_mod
    jmp .handle_error
.not_single_char:
    jmp .proc_literal
.proc_literal:
    mov al, byte [r13]
    cmp al, '-'
    jne .literal_no_sign
    mov r15, 1
    cmp r8, r13+1
    jle .invalid_literal
    mov r14, r13+1
    xor rdx, rdx
    xor rax, rax
    jmp .literal_parse
.literal_no_sign:
    xor r15, r15
    mov r14, r13
    xor rdx, rdx
    xor rax, rax
.literal_parse:
    cmp rdx, 18
    jge .invalid_literal
    cmp r14, r8
    jge .literal_done
    mov cl, byte [r14]
    cmp cl, '0'
    jb .invalid_literal
    cmp cl, '9'
    jb .invalid_literal
    sub cl, '0'
    movzx rcx, cl
    imul rax, 10
    add rax, rcx
    inc rdx
    inc r14
    jmp .literal_parse
.literal_done:
    test r15, r15
    jz .push_literal
    neg rax
    jmp .push_literal
.push_literal:
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_plus:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    add rax, rbx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_minus:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    sub rax, rbx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_mul:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    imul rax, rbx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_div:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    cmp rbx, 0
    je .handle_error
    cmp rax, -9223372036854775808
    jne .div_normal
    cmp rbx, -1
    je .handle_error
.div_normal:
    xor rdx, rdx
    mov rdi, rbx
    idiv rdi
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.proc_operator_mod:
    cmp r11, 2
    jl .handle_error
    mov rbx, qword [r10]
    sub r10, 8
    sub r11, 1
    mov rax, qword [r10]
    sub r10, 8
    sub r11, 1
    cmp rbx, 0
    je .handle_error
    cmp rax, -9223372036854775808
    jne .mod_normal
    cmp rbx, -1
    je .handle_error
.mod_normal:
    xor rdx, rdx
    mov rdi, rbx
    idiv rdi
    mov rax, rdx
    sub r10, 8
    mov qword [r10], rax
    inc r11
    jmp .after_token
.after_token:
    cmp r8, r9
    je .line_end
    mov al, byte [r8]
    cmp al, '\n'
    je .line_end
    cmp al, ' '
    je .skip_spaces
    cmp al, '\t'
    je .skip_spaces
    jmp .token_start
.skip_spaces:
    inc r8
    jmp .skip_spaces
.line_end:
    cmp r11, 1
    jne .handle_error
    mov rax, qword [r10]
    jmp .print_result
.print_result:
    test rax, rax
    jge .print_nonneg
    mov r12, rax
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + dash_char]
    mov rdx, 1
    syscall
    mov r13, r12
    cmp r13, -9223372036854775808
    je .print_min_int
    neg r13
    mov r14, r13
    jmp .print_number
.print_nonneg:
    mov r14, rax
.print_number:
    xor r15, r15
    test r14, r14
    jz .print_zero
.print_loop:
    mov rax, r14
    mov rbx, 10
    xor rdx, rdx
    div rbx
    mov rcx, rdx
    add cl, '0'
    mov byte [digit_buf + r15], cl
    inc r15
    mov r14, rax
    test r14, r14
    jnz .print_loop
.print_zero:
    mov byte [digit_buf], '0'
    mov r15, 1
    mov r16, r15
    dec r16
.output_digits:
    cmp r16, 0
    je .output_newline
    lea rsi, [digit_buf + r16]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    dec r16
    jmp .output_digits
.output_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    cmp r8, r9
    je .line_done
    mov al, byte [r8]
    cmp al, '\n'
    je .skip_line_nl
    jmp .line_done
.skip_line_nl:
    inc r8
.line_done:
    jmp .outer_loop
.print_min_int:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + min_int_str]
    mov rdx, MIN_INT_STR_LEN
    syscall
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall
    cmp r8, r9
    je .line_done
    mov al, byte [r8]
    cmp al, '\n'
    je .skip_line_nl
    jmp .line_done
.handle_error:
    .skip_err_loop:
        cmp r8, r9
        je .skip_err_done
        mov al, byte [r8]
        cmp al, '\n'
        jne .inc_err_char
        jmp .skip_err_done
    .inc_err_char:
        inc r8
        jmp .skip_err_loop
    .skip_err_done:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, ERROR_MSG_LEN
    syscall
    cmp r8, r9
    je .handle_error_done
    mov al, byte [r8]
    cmp al, '\n'
    je .handle_error_done
    jmp .handle_error_done
.handle_error_done:
    inc r8
    jmp .outer_loop
.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
.invalid_literal:
    jmp .handle_error
.invalid_token:
    jmp .handle_error
.section .rodata
error_msg: .asciz "error\n"
dash_char: .byte '-'
newline_char: .byte 0x0A
min_int_str: .asciz "9223372036854775808"
.section .bss
.align 8
input_buf: .space INPUT_BUF_SIZE
stack: .space STACK_SIZE
.stack_end:
.section .data
digit_buf: .space 20
