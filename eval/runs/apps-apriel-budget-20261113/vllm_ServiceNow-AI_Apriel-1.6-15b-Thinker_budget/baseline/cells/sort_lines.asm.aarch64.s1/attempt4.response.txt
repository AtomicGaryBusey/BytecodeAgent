.section .rodata
.newline:
        .byte 10

        .section .bss
.align 4
input_buf:
        .space 32768

.align 4
line_offset:
        .space 4000

.align 2
line_len:
        .space 2000

        .section .text
        .global _start
        .align 16
_start:
        stp x29, x30, [sp, #-16]!
        mov x29, sp

        adrp x19, input_buf
        add x19, x19, :lo12:input_buf

        mov x20, #0
        mov x21, #32768
        add x22, x19, x20

read_loop:
        mov x8, #63
        mov x0, #0
        mov x1, x22
        mov x2, x21
        svc #0
        mov x23, x0
        cmp x23, #0
        beq read_done
        add x20, x20, x23
        sub x21, x21, x23
        add x22, x22, x23
        b read_loop

read_done:
        mov x24, #0
        mov x25, #0
        mov x26, x20
        mov x27, x19

        mov x28, #0
parse_loop:
        cmp x28, x26
        beq parse_done

        add x30, x27, x28
        ldrb w31, [x30]
        cmp w31, #10
        beq newline_found
        add x28, x28, #1
        b parse_loop

newline_found:
        sub x32, x28, x25
        add x33, x27, x25
        lsl x34, x24, #2
        str x33, [line_offset, x34]
        strh w32, [line_len, x24, lsl #1]
        add x24, x24, #1
        add x25, x28, #1
        add x28, x28, #1
        b parse_loop

parse_done:
        sub x35, x28, x25
        cmp x35, #0
        beq no_final_line
        add x36, x27, x25
        lsl x37, x24, #2
        str x36, [line_offset, x37]
        strh w35, [line_len, x24, lsl #1]
        add x24, x24, #1

no_final_line:
        mov x38, #1
        cmp x38, x24
        ble sort_done

outer_i_loop:
        cmp x38, x24
        beq outer_done

        lsl x39, x38, #2
        ldr x40, [line_offset, x39]
        ldrh w41, [line_len, x38, lsl #1]

        mov x42, x38
inner_j_loop:
        cmp x42, #0
        beq inner_done

        sub x43, x42, #1
        lsl x44, x43, #2
        ldr x45, [line_offset, x44]
        ldrh w46, [line_len, x43, lsl #1]

        mov x0, x45
        mov w1, w46
        mov x2, x40
        mov w3, w41
        bl cmp_line
        cmp w0, #1
        beq shift_right
        b inner_done

shift_right:
        str x45, [line_offset, x42, lsl #2]
        strh w46, [line_len, x42, lsl #1]
        sub x42, x42, #1
        b inner_j_loop

inner_done:
        str x40, [line_offset, x42, lsl #2]
        strh w41, [line_len, x42, lsl #1]

        add x38, x38, #1
        b outer_i_loop

outer_done:
sort_done:
        mov x48, #0
out_loop:
        cmp x48, x24
        beq out_done

        lsl x49, x48, #2
        ldr x50, [line_offset, x49]
        ldrh w51, [line_len, x48, lsl #1]

        mov x8, #64
        mov x0, #1
        mov x1, x50
        mov x2, w51
        svc #0

        adrp x52, .newline
        add x52, x52, :lo12:.newline
        mov x8, #64
        mov x0, #1
        mov x1, x52
        mov x2, #1
        svc #0

        add x48, x48, #1
        b out_loop

out_done:
        mov x8, #93
        mov x0, #0
        svc #0

        .global cmp_line
cmp_line:
        mov x4, #0
compare_loop:
        cmp x4, x1
        beq compare_end
        cmp x4, x3
        beq compare_end
        add x5, x0, x4
        ldrb w6, [x5]
        add x5, x2, x4
        ldrb w7, [x5]
        cmp w6, w7
        blt less
        bgt greater
        add x4, x4, #1
        b compare_loop
compare_end:
        cmp x1, x3
        blt less
        beq equal
        b greater
less:
        mov x0, #-1
        ret
equal:
        mov x0, #0
        ret
greater:
        mov x0, #1
        ret
