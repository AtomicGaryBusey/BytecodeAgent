.section .rodata
nl_byte:
    .byte 10
out_buf:
    .space 1
    .equ max_input, 32768
    .equ max_lines, 1000

    .section .bss
input_buf:
    .space 32768
line_offset:
    .space 4000
line_len:
    .space 2000

    .section .text
    .global _start
_start:
    stp x29, x30, [sp, #-16]!
    mov x29, sp
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
    mov x3, #0
read_loop:
    sub x2, #32768, x3
    add x1, x5, x3
    mov x0, #0
    svc 63
    cmp x0, #0
    beq read_done
    add x3, x0
    b read_loop
read_done:
    mov x4, #0
    mov x6, #0
    mov x7, #0
    adrp x11, line_offset
    add x11, x11, :lo12:line_offset
    adrp x12, line_len
    add x12, x12, :lo12:line_len
parse_loop:
    cmp x4, x3
    beq parse_done
    add x0, x5, x4
    ldrb w9, [x0]
    cmp w9, #10
    bne not_newline
    sub x10, x4, x7
    lsl x13, x6, #2
    add x13, x11, x13
    str x7, [x13]
    lsl x14, x6, #1
    add x14, x12, x14
    strh w10, [x14]
    add x6, x6, #1
    add x4, x4, #1
    mov x7, x4
    b parse_loop
not_newline:
    add x4, x4, #1
    b parse_loop
parse_done:
    cmp x4, x7
    bne parse_final_line
    b sort_start
parse_final_line:
    sub x10, x4, x7
    lsl x13, x6, #2
    add x13, x11, x13
    str x7, [x13]
    lsl x14, x6, #1
    add x14, x12, x14
    strh w10, [x14]
    add x6, x6, #1
sort_start:
    mov x16, #1
    cmp x16, x6
    beq sort_done
sort_i_loop:
    lsl x17, x16, #2
    add x17, x11, x17
    ldr x18, [x17]
    lsl x19, x16, #1
    add x19, x12, x19
    ldh w20, [x19]
    sub x21, x16, #1
sort_j_loop:
    cmp x21, #0
    beq insert_done
    lsl x22, x21, #2
    add x22, x11, x22
    ldr x23, [x22]
    lsl x24, x21, #1
    add x24, x12, x24
    ldh w25, [x24]
    mov x0, x23
    mov x1, w25
    mov x2, x18
    mov x3, w20
    bl cmp_line
    cmp x0, #0
    ble insert_done
    lsl x17, x16, #2
    add x17, x11, x17
    str x23, [x17]
    lsl x19, x16, #1
    add x19, x12, x19
    strh w25, [x19]
    sub x21, x21, #1
    b sort_j_loop
insert_done:
    add x21, x21, #1
    lsl x13, x21, #2
    add x13, x11, x13
    str x18, [x13]
    lsl x14, x21, #1
    add x14, x12, x14
    strh w20, [x14]
    add x16, x16, #1
    cmp x16, x6
    bge sort_done
    b sort_i_loop
sort_done:
    adrp x30, out_buf
    add x30, x30, :lo12:out_buf
    adrp x27, nl_byte
    add x27, x27, :lo12:nl_byte
    mov x21, #0
output_loop:
    cmp x21, x6
    beq output_done
    lsl x22, x21, #2
    add x22, x11, x22
    ldr x23, [x22]
    lsl x24, x21, #1
    add x24, x12, x24
    ldh w25, [x24]
    add x26, x5, x23
    mov x15, #0
copy_byte_loop:
    cmp x15, w25
    beq byte_copy_done
    add x28, x26, x15
    ldrb w29, [x28]
    stb w29, [x30]
    mov x0, #1
    mov x1, x30
    mov x2, #1
    svc 64
    add x15, x15, #1
    b copy_byte_loop
byte_copy_done:
    ldrb w29, [x27]
    stb w29, [x30]
    mov x0, #1
    mov x1, x30
    mov x2, #1
    svc 64
    add x21, x21, #1
    b output_loop
output_done:
    mov x0, #0
    svc 93
cmp_line:
    mov x4, #0
compare_loop:
    cmp x4, x1
    bge compare_done
    cmp x4, x3
    bge compare_done
    add x5, x0, x4
    ldrb w6, [x5]
    add x5, x2, x4
    ldrb w7, [x5]
    cmp w6, w7
    blt less
    bgt greater
    add x4, x4, #1
    b compare_loop
compare_done:
    cmp x1, x3
    blt less
    beq equal
    b greater
less:
    mov x0, #-1
    ret
equal:
    mov x0, #0
    ret
greater:
    mov x0, #1
    ret
