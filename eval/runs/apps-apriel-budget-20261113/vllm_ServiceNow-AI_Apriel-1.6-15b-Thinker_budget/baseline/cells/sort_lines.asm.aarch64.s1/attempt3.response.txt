.section .rodata
.newline:
    .byte 10
.equ max_input, 32768

.section .bss
input_buf:
    .space 32768
line_offset:
    .space 8000
line_len:
    .space 2000

.section .text
.globl _start
.align 16
_start:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    adrp x9, input_buf
    add x9, x9, :lo12:input_buf

    mov x10, #0
    mov x11, x9

read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x11
    sub x12, #32768, x10
    mov x2, x12
    svc 0
    mov x13, x0
    cmp x13, #0
    beq read_done
    add x11, x13
    add x10, x13
    b read_loop
read_done:

    mov x14, #0
    mov x15, #0
    mov x16, #0

    adrp x17, line_offset
    add x17, x17, :lo12:line_offset
    adrp x18, line_len
    add x18, x18, :lo12:line_len

parse_loop:
    cmp x16, x10
    beq parse_done

    add x19, x9, x16
    ldrb w20, [x19]
    cmp w20, #10
    beq newline_found
    add x16, x16, #1
    b parse_loop

newline_found:
    sub x21, x16, x15
    lsl x22, x14, #3
    add x22, x22, x17
    str x15, [x22]
    lsl x23, x14, #1
    add x23, x23, x18
    strh w21, [x23]
    add x14, x14, #1
    add x15, x16, #1
    add x16, x16, #1
    b parse_loop

parse_done:
    cmp x15, x10
    bge no_final_line
    sub x21, x10, x15
    lsl x22, x14, #3
    add x22, x22, x17
    str x15, [x22]
    lsl x23, x14, #1
    add x23, x23, x18
    strh w21, [x23]
    add x14, x14, #1
no_final_line:

    mov x24, #1
    cmp x24, x14
    beq sort_done

sort_i_loop:
    lsl x25, x24, #3
    add x25, x25, x17
    ldr x26, [x25]
    lsl x27, x24, #1
    add x27, x27, x18
    ldrh w28, [x27]

    sub x29, x24, #1

sort_j_loop:
    cmp x29, #0
    beq insert_done

    lsl x30, x29, #3
    add x30, x30, x17
    ldr x31, [x30]
    lsl x32, x29, #1
    add x32, x32, x18
    ldrh w33, [x32]

    mov x0, x31
    mov x1, w33
    mov x2, x26
    mov x3, w28
    bl cmp_line
    cmp x0, #1
    beq shift_right
    b insert_done

shift_right:
    add x34, x29, #1
    lsl x34, x34, #3
    add x34, x34, x17
    str x31, [x34]
    add x35, x29, #1
    lsl x35, x35, #1
    add x35, x35, x18
    strh w33, [x35]
    sub x29, x29, #1
    b sort_j_loop

insert_done:
    add x29, x29, #1
    lsl x36, x29, #3
    add x36, x36, x17
    str x26, [x36]
    lsl x37, x29, #1
    add x37, x37, x18
    strh w28, [x37]

    add x24, x24, #1
    cmp x24, x14
    bge sort_done
    b sort_i_loop

sort_done:

    mov x38, #0

output_loop:
    cmp x38, x14
    beq output_done

    lsl x39, x38, #3
    add x39, x39, x17
    ldr x40, [x39]
    lsl x41, x38, #1
    add x41, x41, x18
    ldrh w42, [x41]

    mov x8, #64
    mov x0, #1
    add x43, x9, x40
    mov x1, x43
    mov x2, w42
    svc 0

    adrp x44, .newline
    add x44, x44, :lo12:.newline
    mov x8, #64
    mov x0, #1
    mov x1, x44
    mov x2, #1
    svc 0

    add x38, x38, #1
    b output_loop

output_done:
    mov x8, #93
    mov x0, #0
    svc 0

.global cmp_line
cmp_line:
    mov x4, #0
compare_loop:
    cmp x4, x1
    beq compare_done
    cmp x4, x3
    beq compare_done
    add x5, x0, x4
    ldrb w6, [x5]
    add x5, x2, x4
    ldrb w7, [x5]
    cmp w6, w7
    blt less
    bgt greater
    add x4, x4, #1
    b compare_loop
compare_done:
    cmp x1, x3
    blt less
    beq equal
    b greater
less:
    mov x0, #-1
    ret
equal:
    mov x0, #0
    ret
greater:
    mov x0, #1
    ret
