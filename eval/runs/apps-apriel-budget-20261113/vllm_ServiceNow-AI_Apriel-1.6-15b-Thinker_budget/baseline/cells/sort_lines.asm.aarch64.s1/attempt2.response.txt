.section .rodata
    .equ max_input, 32768

    .section .data
out_buf:
    .space 32768

    .section .bss
input_buf:
    .space 32768
line_offset:
    .space 4000
line_len:
    .space 2000

    .section .text
    .global _start
    .align 16
_start:
    stp x29, x30, [sp, #-16]!
    mov x29, sp

    // read input
    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x7, #0          // write offset in input_buf
    mov x3, #0          // total bytes read
read_loop:
    mov x0, #63
    mov x1, x9
    add x1, x1, x7
    mov x2, max_input
    svc 0
    cmp x0, #0
    beq read_done
    add x7, x0
    add x3, x0
    b read_loop
read_done:

    // parse lines
    mov x4, #0          // index i
    mov x5, #0          // start of current line
    mov x6, #0          // line count

    adrp x11, line_offset
    add x11, x11, :lo12:line_offset
    adrp x12, line_len
    add x12, x12, :lo12:line_len

parse_loop:
    cmp x4, x3
    beq parse_check_final
    add x13, x9, x4
    ldrb w14, [x13]
    cmp w14, #10
    beq newline_found
    add x4, x4, #1
    b parse_loop
newline_found:
    sub x15, x4, x5
    lsl x16, x6, #2
    add x16, x16, x11
    str x5, [x16]
    lsl x17, x6, #1
    add x17, x17, x12
    strh w15, [x17]
    add x6, x6, #1
    add x5, x4, #1
    add x4, x4, #1
    b parse_loop

parse_check_final:
    cmp x5, x3
    bge no_final_line
    sub x15, x3, x5
    lsl x16, x6, #2
    add x16, x16, x11
    str x5, [x16]
    lsl x17, x6, #1
    add x17, x17, x12
    strh w15, [x17]
    add x6, x6, #1
no_final_line:

    // sort lines (insertion sort)
    mov x18, #1
    cmp x18, x6
    beq sort_done
sort_i_loop:
    lsl x19, x18, #2
    add x19, x19, x11
    ldr x20, [x19]          // offset_i
    lsl x21, x18, #1
    add x21, x21, x12
    ldrh w22, [x21]         // len_i
    sub x23, x18, #1
sort_j_loop:
    cmp x23, #0
    beq insert_done
    lsl x24, x23, #2
    add x24, x24, x11
    ldr x25, [x24]          // offset_j
    lsl x26, x23, #1
    add x26, x26, x12
    ldrh w27, [x26]         // len_j

    mov x0, x25
    mov x1, w27
    mov x2, x20
    mov x3, w22
    bl cmp_line
    cmp x0, #1
    bne insert_done
    str x25, [x19]
    strh w27, [x21]
    sub x23, x23, #1
    b sort_j_loop
insert_done:
    str x20, [x19]
    strh w22, [x21]
    add x18, x18, #1
    cmp x18, x6
    bge sort_done
    b sort_i_loop
sort_done:

    // output sorted lines
    mov x28, #0
    adrp x30, out_buf
    add x30, x30, :lo12:out_buf
output_loop:
    cmp x28, x6
    beq output_done
    lsl x31, x28, #2
    add x31, x31, x11
    ldr x32, [x31]          // offset
    lsl x33, x28, #1
    add x33, x33, x12
    ldrh w34, [x33]         // len

    mov x31, x30            // start_out
    mov x35, #0
copy_bytes:
    cmp x35, w34
    beq copy_done
    add x36, x9, x32
    add x36, x36, x35
    ldrb w37, [x36]
    add x30, x30, x35
    strb w37, [x30]
    add x35, x35, #1
    b copy_bytes
copy_done:
    // write line bytes
    mov x0, #64
    mov x1, #1
    mov x2, x31
    mov x3, w34
    svc 0
    // store newline
    mov w37, #10
    // x30 points to start_out + len
    strb w37, [x30]
    // write newline
    mov x0, #64
    mov x1, #1
    mov x2, x30
    mov x3, #1
    svc 0
    // reset out_buf pointer
    adrp x30, out_buf
    add x30, x30, :lo12:out_buf

    add x28, x28, #1
    b output_loop
output_done:
    mov x0, #93
    mov x1, #0
    svc 0

    .global cmp_line
cmp_line:
    // x0 = offset_a, x1 = len_a, x2 = offset_b, x3 = len_b
    mov x4, #0
compare_loop:
    cmp x4, x1
    beq compare_done
    cmp x4, x3
    beq compare_done
    add x5, x0, x4
    ldrb w6, [x5]
    add x5, x2, x4
    ldrb w7, [x5]
    cmp w6, w7
    blt less
    bgt greater
    add x4, x4, #1
    b compare_loop
compare_done:
    cmp x1, x3
    blt less
    beq equal
    b greater
less:
    mov x0, #-1
    ret
equal:
    mov x0, #0
    ret
greater:
    mov x0, #1
    ret
