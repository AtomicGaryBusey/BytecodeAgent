.global _start
    .equ BUF_SIZE, 65536

    .section .bss
buf:
    .space BUF_SIZE
    .balign 8

    .section .rodata
newline:
    .byte 0x0A

    .section .text
_start:
    xor rax, rax          # sys_read
    mov rdi, 0
    lea rsi, [buf]
    xor r8, r8             # offset = 0
.read_loop:
    mov rdx, BUF_SIZE
    sub rdx, r8
    lea rsi, [buf + r8]
    syscall
    test rax, rax
    jz .read_done
    add r8, rax
    jmp .read_loop

.read_done:
    mov rdx, r8            # total bytes read
    test rdx, rdx
    jz .exit_program

    xor rbx, rbx           # start of current line
.process_loop:
    cmp rbx, rdx
    je .exit_program

    mov rcx, rbx
.scan_loop:
    cmp rcx, rdx
    jge .final_line_no_newline
    mov al, byte ptr [buf + rcx]
    cmp al, 0x0A
    jne .next_char
    jmp .newline_found

.next_char:
    inc rcx
    jmp .scan_loop

.newline_found:
    mov r8, rbx
    mov r9, rcx
    dec r9
.rev_loop:
    movzx eax, byte ptr [buf + r8]
    movzx edx, byte ptr [buf + r9]
    mov byte ptr [buf + r8], dl
    mov byte ptr [buf + r9], al
    inc r8
    dec r9
    cmp r8, r9
    jl .rev_loop

    mov rax, 1
    mov rdi, 1
    lea rsi, [buf + rbx]
    mov rdx, rcx
    sub rdx, rbx
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    mov rbx, rcx
    inc rbx
    jmp .process_loop

.final_line_no_newline:
    mov r8, rbx
    mov r9, rdx
    dec r9
.rev_loop2:
    movzx eax, byte ptr [buf + r8]
    movzx edx, byte ptr [buf + r9]
    mov byte ptr [buf + r8], dl
    mov byte ptr [buf + r9], al
    inc r8
    dec r9
    cmp r8, r9
    jl .rev_loop2

    mov rax, 1
    mov rdi, 1
    lea rsi, [buf + rbx]
    mov rdx, rdx
    sub rdx, rbx
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    jmp .exit_program

.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
