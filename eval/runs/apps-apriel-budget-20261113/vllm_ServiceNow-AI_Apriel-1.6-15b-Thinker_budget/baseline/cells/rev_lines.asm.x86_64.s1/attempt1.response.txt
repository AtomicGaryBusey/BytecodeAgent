.global _start
.section .bss
buf:
    .space 65536
    .balign 8
.section .rodata
newline:
    .byte 0x0A
.section .text

_start:
    xor rax, rax
    mov rdi, 0
    xor rbx, rbx
.read_loop:
    mov rdx, 65536
    sub rdx, rbx
    lea rsi, [buf + rbx]
    syscall
    test rax, rax
    jnz .read_cont
    jmp .read_done

.read_cont:
    add rbx, rax
    jmp .read_loop

.read_done:
    test rbx, rbx
    jz .exit_program

    mov rdx, rbx
    xor rbx, rbx

.process_loop:
    cmp rbx, rdx
    je .exit_program

    mov rcx, rbx
.scan_loop:
    cmp rcx, rdx
    jge .final_line_no_newline
    mov al, byte ptr [buf + rcx]
    cmp al, 0x0A
    je .found_newline
    inc rcx
    jmp .scan_loop

.found_newline:
    cmp rcx, rbx
    je .empty_line

    mov r8, rbx
    dec rcx
    mov r9, r8
    mov r10, rcx
.rev_loop:
    movzx eax, byte ptr [buf + r9]
    movzx edx, byte ptr [buf + r10]
    mov byte ptr [buf + r9], dl
    mov byte ptr [buf + r10], al
    inc r9
    dec r10
    cmp r9, r10
    jl .rev_loop

    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, rcx
    sub rdx, r8
    inc rdx
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    inc rcx
    mov rbx, rcx
    jmp .process_loop

.empty_line:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc rcx
    mov rbx, rcx
    jmp .process_loop

.final_line_no_newline:
    cmp rbx, rdx
    je .exit_program

    mov r8, rbx
    mov rcx, rdx
    dec rcx
    mov r9, r8
    mov r10, rcx
.rev_loop2:
    movzx eax, byte ptr [buf + r9]
    movzx edx, byte ptr [buf + r10]
    mov byte ptr [buf + r9], dl
    mov byte ptr [buf + r10], al
    inc r9
    dec r10
    cmp r9, r10
    jl .rev_loop2

    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    mov rdx, rcx
    sub rdx, r8
    inc rdx
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    jmp .exit_program

.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
