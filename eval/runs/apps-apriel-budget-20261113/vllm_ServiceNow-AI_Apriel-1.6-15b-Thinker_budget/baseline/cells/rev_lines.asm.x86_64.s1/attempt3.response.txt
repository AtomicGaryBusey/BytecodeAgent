.global _start
.section .bss
buf:
    .space 65536
    .balign 8
.section .rodata
newline:
    .byte 0x0A
.section .text

_start:
    xor rax, rax          # sys_read
    mov rdi, 0
    lea rsi, [buf]
    xor rbx, rbx          # offset = 0
.read_loop:
    mov rdx, 65536
    sub rdx, rbx
    lea rsi, [buf + rbx]
    xor rax, rax
    syscall
    test rax, rax
    jz .read_done
    add rbx, rax
    jmp .read_loop

.read_done:
    test rbx, rbx
    jz .exit_program

    mov rdx, rbx          # total bytes
    xor rbx, rbx          # start of line = 0

.process_loop:
    cmp rbx, rdx
    je .exit_program

    mov rcx, rbx          # scanning pointer
.scan_loop:
    cmp rcx, rdx
    jge .final_line_no_newline
    mov al, byte ptr [buf + rcx]
    cmp al, 0x0A
    je .newline_found
    inc rcx
    jmp .scan_loop

.newline_found:
    cmp rcx, rbx
    je .empty_line

    mov r8, rbx
    mov r9, rcx
    dec r9
.rev_loop:
    movzx eax, byte ptr [buf + r8]
    movzx edx, byte ptr [buf + r9]
    mov byte ptr [buf + r8], dl
    mov byte ptr [buf + r9], al
    inc r8
    dec r9
    cmp r8, r9
    jl .rev_loop

    mov r10, rcx
    sub r10, rbx
    mov rax, 1            # sys_write
    mov rdi, 1
    mov rsi, rbx
    mov rdx, r10
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    mov rbx, rcx
    inc rbx
    jmp .process_loop

.empty_line:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    mov rbx, rcx
    inc rbx
    jmp .process_loop

.final_line_no_newline:
    mov r8, rbx
    mov r9, rdx
    dec r9
.rev_loop2:
    movzx eax, byte ptr [buf + r8]
    movzx edx, byte ptr [buf + r9]
    mov byte ptr [buf + r8], dl
    mov byte ptr [buf + r9], al
    inc r8
    dec r9
    cmp r8, r9
    jl .rev_loop2

    mov r10, rdx
    sub r10, rbx
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    mov rdx, r10
    syscall

    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    jmp .exit_program

.exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall
