.section .rodata
base64_table: .string "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 8
inbuf: .space 49152
.align 8
outbuf: .space 66500

.section .text
.global _start
_start:
    xor edi, edi
    mov esi, inbuf
    mov edx, 49152
    xor r12, r12
.read_loop:
    mov rax, 0
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    jmp .read_loop
.read_done:
    test r12, r12
    jz .exit_zero

    xor r13, r13
    xor r14, r14
    mov r8, inbuf
    mov r9, outbuf

    mov rax, r12
    xor rdx, rdx
    mov rcx, 3
    div rcx
    mov rbx, rax
    mov r11, rdx
    mov r10, rbx

.encode_full:
    test r10, r10
    jz .process_remainder

    movzx eax, byte ptr [r8]
    movzx ebx, byte ptr [r8+1]
    movzx ecx, byte ptr [r8+2]

    movzx edx, al
    shr edx, 2
    and edx, 0x3F
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .no_newline_c0
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.no_newline_c0:

    movzx edx, al
    and edx, 0x3
    shl edx, 4
    movzx esi, byte ptr [r8+1]
    shr esi, 4
    and esi, 0x0F
    or edx, esi
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .no_newline_c1
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.no_newline_c1:

    movzx edx, ebx
    and edx, 0x0F
    shl edx, 2
    movzx esi, byte ptr [r8+2]
    shr esi, 6
    and esi, 0x03
    or edx, esi
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .no_newline_c2
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.no_newline_c2:

    movzx edx, byte ptr [r8+2]
    and edx, 0x3F
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .no_newline_c3
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.no_newline_c3:

    add r8, 3
    dec r10
    jmp .encode_full

.process_remainder:
    mov rax, r11
    cmp rax, 2
    je .rem2
    cmp rax, 1
    jne .post_encode

    movzx eax, byte ptr [r8]

    movzx edx, al
    shr edx, 2
    and edx, 0x3F
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem1_nl0
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem1_nl0:

    movzx edx, al
    and edx, 0x3
    shl edx, 4
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem1_nl1
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem1_nl1:

    mov dl, '='
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem1_nl2
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem1_nl2:

    mov dl, '='
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem1_done
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem1_done:
    jmp .post_encode

.rem2:
    movzx eax, byte ptr [r8]      ; b0
    movzx ebx, byte ptr [r8+1]    ; b1

    movzx edx, al
    shr edx, 2
    and edx, 0x3F
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem2_nl0
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem2_nl0:

    movzx edx, al
    and edx, 0x3
    shl edx, 4
    movzx esi, byte ptr [r8+1]
    shr esi, 4
    and esi, 0x0F
    or edx, esi
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem2_nl1
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem2_nl1:

    movzx edx, bl
    and edx, 0x0F
    shl edx, 2
    mov dl, byte ptr [rel base64_table + rdx]
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem2_nl2
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem2_nl2:

    mov dl, '='
    mov byte ptr [r9 + r13], dl
    inc r13
    inc r14
    cmp r14, 76
    jne .rem2_done
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    xor r14, r14
.rem2_done:
    jmp .post_encode

.post_encode:
    test r14, r14
    jz .write_output
    mov dl, 0x0A
    mov byte ptr [r9 + r13], dl
    inc r13
    jmp .write_output

.write_output:
    mov rdi, 1
    mov rsi, outbuf
    mov rdx, r13
    mov rax, 1
    syscall
    jmp .exit

.exit:
    mov rax, 60
    xor edi, edi
    syscall

.exit_zero:
    mov rax, 60
    xor edi, edi
    syscall
