.section .rodata
base64_table: .string "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 8
inbuf: .space 49152
.align 8
outbuf: .space 66500

.section .text
.global _start
_start:
    xor edi, edi
    mov rsi, inbuf
    mov edx, 49152
    xor eax, eax
.read_loop:
    mov rax, 0
    syscall
    test rax, rax
    jz .read_done
    add r12, rax
    add rsi, rax
    jmp .read_loop
.read_done:
    test r12, r12
    jz .exit_zero

    mov rsi, inbuf
    mov rdi, outbuf
    xor r10, r10
    xor r14, r14

    mov rax, r12
    xor rdx, rdx
    mov rbx, 3
    div rbx
    mov r8, rax
    mov r9, rdx

.encode_full:
    test r8, r8
    jz .enc_full_done

    movzx r11d, byte ptr [rsi]
    movzx r12d, byte ptr [rsi+1]
    movzx r13d, byte ptr [rsi+2]

    mov eax, r11d
    shr eax, 2
    and eax, 0x3F
    movzx eax, byte ptr [rip + base64_table + rax]
    mov byte [rdi], al
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .enc_full_no_nl1
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.enc_full_no_nl1:

    mov ebx, r11d
    and ebx, 0x3
    shl ebx, 4
    mov ecx, r12d
    shr ecx, 4
    and ecx, 0x0F
    or ebx, ecx
    movzx ebx, byte ptr [rip + base64_table + ebx]
    mov byte [rdi], bl
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .enc_full_no_nl2
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.enc_full_no_nl2:

    mov edx, r12d
    and edx, 0x0F
    shl edx, 2
    mov esi, r13d
    shr esi, 6
    and esi, 0x03
    or edx, esi
    movzx edx, byte ptr [rip + base64_table + edx]
    mov byte [rdi], dl
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .enc_full_no_nl3
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.enc_full_no_nl3:

    mov edx, r13d
    and edx, 0x3F
    movzx edx, byte ptr [rip + base64_table + edx]
    mov byte [rdi], dl
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .enc_full_no_nl4
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.enc_full_no_nl4:

    add rsi, 3
    dec r8
    jmp .encode_full

.enc_full_done:
    test r9, r9
    jz .skip_rem
    cmp r9, 1
    jne .rem2

.rem1:
    movzx r11d, byte ptr [rsi]

    mov eax, r11d
    shr eax, 2
    and eax, 0x3F
    movzx eax, byte ptr [rip + base64_table + eax]
    mov byte [rdi], al
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem1_no_nl1
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem1_no_nl1:

    mov ebx, r11d
    and ebx, 0x3
    shl ebx, 4
    movzx ebx, byte ptr [rip + base64_table + ebx]
    mov byte [rdi], bl
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem1_no_nl2
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem1_no_nl2:

    mov byte [rdi], '='
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem1_nl3
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem1_nl3:

    mov byte [rdi], '='
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem1_done
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem1_done:

    jmp .post_encode

.rem2:
    movzx r11d, byte ptr [rsi]
    movzx r12d, byte ptr [rsi+1]

    mov eax, r11d
    shr eax, 2
    and eax, 0x3F
    movzx eax, byte ptr [rip + base64_table + eax]
    mov byte [rdi], al
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem2_no_nl1
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem2_no_nl1:

    mov ebx, r11d
    and ebx, 0x3
    shl ebx, 4
    mov ecx, r12d
    shr ecx, 4
    and ecx, 0x0F
    or ebx, ecx
    movzx ebx, byte ptr [rip + base64_table + ebx]
    mov byte [rdi], bl
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem2_no_nl2
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem2_no_nl2:

    mov byte [rdi], '='
    inc rdi
    inc r14
    inc r10
    cmp r10, 76
    jne .rem2_done
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
    xor r10, r10
.rem2_done:

    jmp .post_encode

.skip_rem:

.post_encode:
    test r10, r10
    jz .no_final_nl
    mov byte [rdi], 0x0A
    inc rdi
    inc r14
.no_final_nl:

    mov rax, 1
    mov rdi, 1
    mov rsi, outbuf
    mov rdx, r14
    syscall

.exit_zero:
    mov rax, 60
    xor edi, edi
    syscall
