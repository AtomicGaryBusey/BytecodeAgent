.section .rodata
base64_table: .string "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
.section .bss
.align 8
inbuf: .space 49152
.align 8
outbuf: .space 66500
.section .text
.global _start
_start:
    xor r12, r12
    xor r8, r8
read_loop:
    mov rdi, 0
    mov rsi, r8
    mov rdx, 49152
    mov rax, 0
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    add r8, rax
    jmp read_loop
read_done:
    test r12, r12
    jz exit_zero
    mov rax, r12
    xor rdx, rdx
    mov rbx, 3
    div rbx
    mov r8, rax
    mov r9, rdx
    mov rsi, inbuf
    mov rdi, outbuf
    xor r10, r10
    xor r15, r15
    lea r11, [rip + base64_table]
enc_full:
    test r8, r8
    jz enc_full_done
    movzx r12d, byte [rsi]
    movzx r13d, byte [rsi+1]
    movzx r14d, byte [rsi+2]
    mov eax, r12d
    shr eax, 2
    and eax, 0x3F
    movzx eax, byte [r11 + eax]
    mov byte [rdi], al
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .no_nl_c0
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.no_nl_c0:
    mov ebx, r12d
    and ebx, 0x3
    shl ebx, 4
    mov ecx, r13d
    shr ecx, 4
    and ecx, 0x0F
    or ebx, ecx
    movzx ebx, byte [r11 + ebx]
    mov byte [rdi], bl
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .no_nl_c1
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.no_nl_c1:
    mov edx, r13d
    and edx, 0x0F
    shl edx, 4
    mov esi, r14d
    shr esi, 6
    and esi, 0x03
    or edx, esi
    movzx edx, byte [r11 + edx]
    mov byte [rdi], dl
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .no_nl_c2
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.no_nl_c2:
    mov esi, r14d
    and esi, 0x3F
    movzx esi, byte [r11 + esi]
    mov byte [rdi], sil
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .no_nl_c3
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.no_nl_c3:
    add rsi, 3
    dec r8
    jmp enc_full
enc_full_done:
    test r9, r9
    jz .skip_rem
    cmp r9, 1
    jne .rem2
.rem1:
    movzx r12d, byte [rsi]
    mov eax, r12d
    shr eax, 2
    and eax, 0x3F
    movzx eax, byte [r11 + eax]
    mov byte [rdi], al
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem1_no_nl_c0
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem1_no_nl_c0:
    mov ebx, r12d
    and ebx, 0x3
    shl ebx, 4
    movzx ebx, byte [r11 + ebx]
    mov byte [rdi], bl
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem1_no_nl_c1
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem1_no_nl_c1:
    mov byte [rdi], '='
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem1_no_nl_c2
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem1_no_nl_c2:
    mov byte [rdi], '='
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem1_done
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem1_done:
    jmp .post_encode
.rem2:
    movzx r12d, byte [rsi]
    movzx r13d, byte [rsi+1]
    mov eax, r12d
    shr eax, 2
    and eax, 0x3F
    movzx eax, byte [r11 + eax]
    mov byte [rdi], al
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem2_no_nl_c0
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem2_no_nl_c0:
    mov ebx, r12d
    and ebx, 0x3
    shl ebx, 4
    mov ecx, r13d
    shr ecx, 4
    and ecx, 0x0F
    or ebx, ecx
    movzx ebx, byte [r11 + ebx]
    mov byte [rdi], bl
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem2_no_nl_c1
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem2_no_nl_c1:
    mov byte [rdi], '='
    inc rdi
    inc r10
    inc r15
    cmp r10, 76
    jne .rem2_done
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
    xor r10, r10
.rem2_done:
    jmp .post_encode
.skip_rem:
.post_encode:
    test r10, r10
    jz .no_final_nl
    mov byte [rdi], 0x0A
    inc rdi
    inc r15
.no_final_nl:
    mov rax, 1
    mov rdi, 1
    mov rsi, outbuf
    mov rdx, r15
    syscall
exit_zero:
    mov rax, 60
    xor edi, edi
    syscall
