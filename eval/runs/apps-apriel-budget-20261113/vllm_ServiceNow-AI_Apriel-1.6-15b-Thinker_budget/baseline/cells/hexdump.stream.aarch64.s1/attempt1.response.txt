.global _start
    .section .rodata
hex_chars:
    .string "0123456789abcdef"
space_char:
    .byte ' '
pipe_char:
    .byte '|'
dot_char:
    .byte '.'
newline_char:
    .byte '\n'
    .data
row_buf:
    .space 80
    .bss
input_buf:
    .space 65536
    .section .text
_start:
    adrp x3, input_buf
    add x3, x3, :lo12:input_buf
    mov x2, #0          // total_len = 0
read_loop:
    mov x4, #65536
    subs x4, x4, x2     // x4 = remaining
    add x5, x3, x2      // buf = input_buf + total_len
    mov x0, #0
    mov x1, x5
    mov x2, x4
    mov x8, #63
    svc #0
    cbnz x0, read_loop
    adrp x6, row_buf
    add x6, x6, :lo12:row_buf
    mov x7, #0          // row_offset
row_loop:
    cmp x7, x2
    b.ge row_done
    sub x8, x2, x7       // bytes left in input after this row
    mov x9, #16
    cmp x8, x9
    ble short_row
    mov x10, #16
    b row_process
short_row:
    mov x10, x8
    mov x12, x6          // start of buffer
    adrp x15, hex_chars
    add x15, x15, :lo12:hex_chars
    adrp x18, space_char
    add x18, x18, :lo12:space_char
    ldrb w19, [x18, #0]
    // ----- output offset -----
    shiftr x13, x7, #24
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    shiftr x13, x7, #16
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    shiftr x13, x7, #8
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    mov x13, x7
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    strb w19, [x6], #1
    strb w19, [x6], #1
    // ----- process 16 byte slots -----
    mov x11, #0
slot_loop:
    cmp x11, x10
    b.lt real_slot
    // placeholder: three spaces
    strb w19, [x6], #1
    strb w19, [x6], #1
    strb w19, [x6], #1
    b after_slot
real_slot:
    add x20, x7, x11
    add x20, x20, x3
    ldrb w22, [x20]
    lsr x14, w22, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, w22, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    strb w19, [x6], #1
after_slot:
    cmp x11, #7
    beq extra_space
    add x11, x11, #1
    cmp x11, #16
    b.lt slot_loop
    // extra space after the 8th byte
extra_space:
    strb w19, [x6], #1
    add x11, x11, #1
    cmp x11, #16
    b.lt slot_loop
    // ----- ascii column -----
    strb w19, [x6], #1
    adrp x22, pipe_char
    add x22, x22, :lo12:pipe_char
    ldrb w23, [x22, #0]
    strb w23, [x6], #1
    adrp x24, dot_char
    add x24, x24, :lo12:dot_char
    ldrb w25, [x24, #0]
    mov x26, #0
ascii_loop:
    cmp x26, x10
    b.lt ascii_proc
    b ascii_done
ascii_proc:
    add x27, x7, x26
    add x27, x27, x3
    ldrb w28, [x27]
    cmp w28, #0x20
    blt ascii_dot
    cmp w28, #0x7e
    bgt ascii_dot
    strb w28, [x6], #1
    b ascii_next
ascii_dot:
    strb w25, [x6], #1
ascii_next:
    add x26, x26, #1
    b ascii_loop
ascii_done:
    strb w23, [x6], #1
    adrp x29, newline_char
    add x29, x29, :lo12:newline_char
    ldrb w30, [x29, #0]
    strb w30, [x6], #1
    sub x31, x6, x12
    mov x0, #1
    mov x1, x12
    mov x2, x31
    mov x8, #64
    svc #0
    add x7, x7, #16
    b row_loop
row_done:
    mov x6, x12
    adrp x15, hex_chars
    add x15, x15, :lo12:hex_chars
    shiftr x13, x2, #24
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    shiftr x13, x2, #16
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    shiftr x13, x2, #8
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    mov x13, x2
    lsr x14, x13, #4
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    and x14, x13, #0xF
    add x16, x15, x14
    ldrb w17, [x16, #0]
    strb w17, [x6], #1
    adrp x29, newline_char
    add x29, x29, :lo12:newline_char
    ldrb w30, [x29, #0]
    strb w30, [x6], #1
    sub x31, x6, x12
    mov x0, #1
    mov x1, x12
    mov x2, x31
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0
