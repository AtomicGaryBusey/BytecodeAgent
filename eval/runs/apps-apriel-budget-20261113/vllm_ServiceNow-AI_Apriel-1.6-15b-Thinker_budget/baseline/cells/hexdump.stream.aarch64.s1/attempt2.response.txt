.global _start
.section .rodata
hex_chars:
    .string "0123456789abcdef"
space_char:
    .byte ' '
pipe_char:
    .byte '|'
dot_char:
    .byte '.'
newline_char:
    .byte '\n'

.data
row_buf:
    .space 80

.bss
input_buf:
    .space 65536

.section .text
_start:
    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    mov w1, #0          // total_len = 0

read_loop:
    movw w2, #65536
    sub w2, w2, w1
    add x3, x0, w1
    mov x4, #0
    mov x5, x3
    mov x6, w2
    mov x8, #63
    svc #0
    mov w7, x0
    cbnz x0, read_cont
    b read_done
read_cont:
    add w1, w1, w7
    b read_loop
read_done:
    // w1 = total_len

    adrp x9, row_buf
    add x9, x9, :lo12:row_buf
    mov w10, #0          // offset = 0

row_loop:
    cmp w10, w1
    b.ge row_done

    sub w11, w1, w10      // remaining
    mov w12, w11
    cmp w12, #16
    b.le n_is_remaining
    mov w12, #16
    b n_after
n_is_remaining:
    b n_after
n_after:
    // w12 = n (bytes in row)

    mov x13, x9           // buffer pointer

    // print offset as 8 hex digits
    lsr w14, w10, #8      // high byte
    mov w15, w10          // low byte

    adrp x16, hex_chars
    add x16, x16, :lo12:hex_chars

    // high byte high nibble
    lsr w17, w14, #4
    and w18, w14, #0xF
    add x17, x16, w17
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1
    add x17, x16, w18
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1

    // low byte high nibble
    lsr w20, w15, #4
    and w21, w15, #0xF
    add x17, x16, w20
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1
    add x17, x16, w21
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1

    // two spaces
    ldrb w19, [space_char]
    stb w19, [x13]
    add x13, x13, #1
    ldrb w19, [space_char]
    stb w19, [x13]
    add x13, x13, #1

    // iterate over 16 byte slots
    mov w17, #0
byte_loop:
    cmp w17, #16
    beq byte_done

    cmp w17, w12
    b.ge byte_missing

    // load byte
    add x18, x0, w10
    add x18, x18, w17
    ldrb w22, [x18]

    // hex pair
    lsr w23, w22, #4
    and w24, w22, #0xF
    add x17, x16, w23
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1
    add x17, x16, w24
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1

    // space after hex pair
    ldrb w19, [space_char]
    stb w19, [x13]
    add x13, x13, #1

    b byte_next
byte_missing:
    // three spaces
    mov w19, #32
    stb w19, [x13]
    add x13, x13, #1
    stb w19, [x13]
    add x13, x13, #1
    stb w19, [x13]
    add x13, x13, #1
byte_next:
    // extra space after slot 7
    cmp w17, #7
    beq extra_space_done
    b byte_inc
extra_space_done:
    ldrb w19, [space_char]
    stb w19, [x13]
    add x13, x13, #1
byte_inc:
    add w17, w17, #1
    b byte_loop

byte_done:
    // space before '|'
    ldrb w19, [space_char]
    stb w19, [x13]
    add x13, x13, #1

    // '|'
    ldrb w19, [pipe_char]
    stb w19, [x13]
    add x13, x13, #1

    // ASCII column
    mov w17, #0
ascii_loop:
    cmp w17, w12
    beq ascii_done

    // load same byte
    add x18, x0, w10
    add x18, x18, w17
    ldrb w22, [x18]

    cmp w22, #32
    b.lt ascii_dot
    cmp w22, #126
    b.gt ascii_dot
    // printable
    stb w22, [x13]
    add x13, x13, #1
    b ascii_next
ascii_dot:
    ldrb w22, [dot_char]
    stb w22, [x13]
    add x13, x13, #1
ascii_next:
    add w17, w17, #1
    b ascii_loop

ascii_done:
    // closing '|'
    ldrb w19, [pipe_char]
    stb w19, [x13]
    add x13, x13, #1

    // newline
    ldrb w19, [newline_char]
    stb w19, [x13]
    add x13, x13, #1

    // line length = 63 + n
    mov w25, w12
    add w25, w25, #63

    // write line
    mov x0, #1
    mov x1, x9
    mov x2, w25
    mov x8, #64
    svc #0

    // next row
    add w10, w10, #16
    b row_loop

row_done:
    // final total length line
    mov x13, x9

    // total_len = w1
    lsr w14, w1, #8
    mov w15, w1

    adrp x16, hex_chars
    add x16, x16, :lo12:hex_chars

    // high byte hex
    lsr w17, w14, #4
    and w18, w14, #0xF
    add x17, x16, w17
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1
    add x17, x16, w18
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1

    // low byte hex
    lsr w20, w15, #4
    and w21, w15, #0xF
    add x17, x16, w20
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1
    add x17, x16, w21
    ldrb w19, [x17]
    stb w19, [x13]
    add x13, x13, #1

    // newline
    ldrb w19, [newline_char]
    stb w19, [x13]
    add x13, x13, #1

    // length = 9
    mov w26, #9

    // write final line
    mov x0, #1
    mov x1, x9
    mov x2, w26
    mov x8, #64
    svc #0

    // exit
    mov x0, #0
    mov x8, #93
    svc #0
