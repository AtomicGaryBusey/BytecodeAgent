.global _start
.section .rodata
hex_chars:
    .string "0123456789abcdef"
.section .data
row_buf:
    .space 80
.section .bss
input_buf:
    .space 65536
.section .text
_start:
    adrp x0, input_buf
    add x0, x0, :lo12:input_buf
    mov x1, x0
    mov x2, #65536
    mov x3, #0
read_loop:
    mov x8, #63
    svc
    cmp x0, #0
    beq read_done
    add x3, x3, x0
    b read_loop
read_done:
    mov x4, #0
process_rows:
    cmp x4, x3
    b.ge print_total_len
    sub x5, x3, x4
    mov x6, #16
    cmp x5, x6
    ble set_n
    mov x7, #16
    b set_n_after
set_n:
    mov x7, x5
set_n_after:
    adrp x8, input_buf
    add x8, x8, :lo12:input_buf
    add x8, x8, x4
    adrp x9, row_buf
    add x9, x9, :lo12:row_buf
    mov x10, x9
    adrp x11, hex_chars
    add x11, x11, :lo12:hex_chars
    mov x12, x4
    lsr x12, x12, #28
    mov x13, #8
print_offset_nibble:
    add x14, x11, x12
    ldrb w15, [x14]
    strb w15, [x10]
    add x10, x10, #1
    lsr x12, x12, #4
    sub x13, x13, #1
    b.ne print_offset_nibble
    mov w15, #32
    strb w15, [x10]
    add x10, x10, #1
    strb w15, [x10]
    add x10, x10, #1
    mov x16, #0
row_slot:
    cmp x16, #16
    b.ge write_row
    cmp x16, x7
    b.lt slot_has_byte
    mov w15, #32
    strb w15, [x10]
    add x10, x10, #1
    strb w15, [x10]
    add x10, x10, #1
    strb w15, [x10]
    add x10, x10, #1
    cmp x16, #7
    beq extra_space_missing
    b inc_slot
extra_space_missing:
    mov w15, #32
    strb w15, [x10]
    add x10, x10, #1
inc_slot:
    add x16, x16, #1
    b row_slot
slot_has_byte:
    add x17, x8, x16
    ldrb w17, [x17]
    lsr w18, w17, #4
    add x14, x11, w18
    ldrb w15, [x14]
    strb w15, [x10]
    add x10, x10, #1
    and w18, w17, #0xF
    add x14, x11, w18
    ldrb w15, [x14]
    strb w15, [x10]
    add x10, x10, #1
    mov w15, #32
    strb w15, [x10]
    add x10, x10, #1
    cmp x16, #7
    beq extra_space
    b inc_slot2
extra_space:
    mov w15, #32
    strb w15, [x10]
    add x10, x10, #1
inc_slot2:
    add x16, x16, #1
    b row_slot
write_row:
    mov w15, #32
    strb w15, [x10]
    add x10, x10, #1
    mov w15, #124
    strb w15, [x10]
    add x10, x10, #1
    mov x16, #0
ascii_loop:
    cmp x16, x7
    b.ge ascii_done
    add x17, x8, x16
    ldrb w17, [x17]
    movw x18, #0x20
    cmp x17, x18
    b.lt ascii_dot
    movw x18, #0x7E
    cmp x17, x18
    b.gt ascii_dot
    mov x19, x17
    b ascii_store
ascii_dot:
    mov w19, #46
ascii_store:
    strb w19, [x10]
    add x10, x10, #1
    add x16, x16, #1
    b ascii_loop
ascii_done:
    mov w15, #124
    strb w15, [x10]
    add x10, x10, #1
    mov w15, #10
    strb w15, [x10]
    add x10, x10, #1
    mov x20, #62
    add x20, x20, x7
    mov x8, #64
    mov x0, #1
    adrp x1, row_buf
    add x1, x1, :lo12:row_buf
    mov x2, x20
    svc
    add x4, x4, #16
    b process_rows
print_total_len:
    adrp x9, row_buf
    add x9, x9, :lo12:row_buf
    mov x10, x9
    mov x12, x3
    lsr x12, x12, #28
    mov x13, #8
print_total_nibble:
    add x14, x11, x12
    ldrb w15, [x14]
    strb w15, [x10]
    add x10, x10, #1
    lsr x12, x12, #4
    sub x13, x13, #1
    b.ne print_total_nibble
    mov w15, #10
    strb w15, [x10]
    add x10, x10, #1
    mov x8, #64
    mov x0, #1
    adrp x1, row_buf
    add x1, x1, :lo12:row_buf
    mov x2, #9
    svc
    mov x8, #93
    mov x0, #0
    svc
