.global _start
.section .rodata
hex_chars:
    .string "0123456789abcdef"
.section .bss
input_buf:
    .space 65536
out_buf:
    .space 80
.section .text
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x22, #0
    mov x23, #0
read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x21
    mov x24, #65536
    sub x2, x24, x22
    svc #0
    cmp x0, #0
    beq read_done
    add x22, x22, x0
    add x21, x21, x0
    b read_loop
read_done:
    adrp x19, hex_chars
    add x19, x19, :lo12:hex_chars
process_rows:
    cmp x23, x22
    bge final_offset
    sub x4, x22, x23
    mov x5, x4
    cmp x5, #16
    ble row_len_done
    mov x5, #16
row_len_done:
    adrp x6, input_buf
    add x6, x6, :lo12:input_buf
    add x6, x6, x23
    adrp x7, out_buf
    add x7, x7, :lo12:out_buf
    mov x8, #0
    mov x9, x23
    lsr x10, x9, #28
    mov x11, #8
.offset_loop:
    ldrb w12, [x19, x10, LSL #0]
    strb w12, [x7, x8]
    add x8, x8, #1
    lsr x10, x10, #4
    subs x11, x11, #1
    bne .offset_loop
    mov w12, #32
    strb w12, [x7, x8]; add x8, #1
    strb w12, [x7, x8]; add x8, #1
    mov x12, #0
hex_slot:
    cmp x12, x5
    blt slot_has_data
    mov w13, #32
    strb w13, [x7, x8]; add x8, #1
    strb w13, [x7, x8]; add x8, #1
    strb w13, [x7, x8]; add x8, #1
    b slot_common
slot_has_data:
    ldrb w13, [x6, x12]
    lsr x14, w13, #4
    and x15, w13, #15
    ldrb w16, [x19, x14, LSL #0]
    ldrb w17, [x19, x15, LSL #0]
    mov w18, #32
    strb w16, [x7, x8]; add x8, #1
    strb w17, [x7, x8]; add x8, #1
    strb w18, [x7, x8]; add x8, #1
    b slot_common
slot_common:
    cmp x12, #7
    beq extra_space
    add x12, x12, #1
    cmp x12, #16
    blo hex_slot
extra_space:
    mov w18, #32
    strb w18, [x7, x8]; add x8, #1
    add x12, x12, #1
    cmp x12, #16
    blo hex_slot
    mov w18, #32
    strb w18, [x7, x8]; add x8, #1
    mov w18, #124
    strb w18, [x7, x8]; add x8, #1
    mov x12, #0
.ascii_loop:
    cmp x12, x5
    blt ascii_have
    b ascii_done
ascii_have:
    ldrb w13, [x6, x12]
    cmp w13, #32
    bge ascii_print
    cmp w13, #126
    ble ascii_print
    mov w13, #46
ascii_print:
    strb w13, [x7, x8]; add x8, #1
    add x12, x12, #1
    cmp x12, x5
    blo ascii_loop
ascii_done:
    mov w13, #124
    strb w13, [x7, x8]; add x8, #1
    mov w13, #10
    strb w13, [x7, x8]; add x8, #1
    mov x9, x8
    mov x8, #64
    mov x0, #1
    mov x1, x7
    mov x2, x9
    svc #0
    add x23, x23, x5
    b process_rows
final_offset:
    mov x9, x22
    lsr x10, x9, #28
    mov x11, #8
.offset_final_loop:
    ldrb w12, [x19, x10, LSL #0]
    strb w12, [x7, x8]; add x8, #1
    lsr x10, x10, #4
    subs x11, x11, #1
    bne .offset_final_loop
    mov w12, #10
    strb w12, [x7, x8]; add x8, #1
    mov x9, x8
    mov x8, #64
    mov x0, #1
    mov x1, x7
    mov x2, x9
    svc #0
    mov x8, #93
    mov x0, #0
    svc #0
