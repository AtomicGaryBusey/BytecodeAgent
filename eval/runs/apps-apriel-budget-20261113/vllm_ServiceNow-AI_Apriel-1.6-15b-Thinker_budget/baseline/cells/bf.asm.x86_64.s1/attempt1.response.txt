.equ MAX_INPUT, 65536

.section .bss
input_buf: .space 65536
total_len: .int 0
prog_len: .int 0
input_len2: .int 0
tape: .space 30000
bracket_map: .space 262144
out_buf: .byte 0

.section .text
.global _start
_start:
    xor r10, r10
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, input_buf
    add rsi, r10
    mov rdx, MAX_INPUT
    sub rdx, r10
    syscall
    test rax, rax
    jz read_done
    add r10, rax
    jmp read_loop
read_done:
    mov [total_len], r10

    xor r8, r8
    mov rcx, [total_len]
find_sep:
    cmp r8, rcx
    jge no_sep
    movzx rdx, byte ptr [input_buf + r8]
    cmp rdx, 33
    je sep_found
    inc r8
    jmp find_sep
sep_found:
    mov [prog_len], r8
    mov rax, [total_len]
    mov rbx, r8
    inc rbx
    sub rax, rbx
    mov [input_len2], rax
    jmp after_find
no_sep:
    mov [prog_len], [total_len]
    mov [input_len2], 0
after_find:
    xor r10, r10
    mov r8, 0
    mov r9, [prog_len]
    lea r12, [rip + bracket_map]
build_loop:
    cmp r8, r9
    jge build_done
    movzx rdx, byte ptr [input_buf + r8]
    cmp rdx, 91
    je push_case
    cmp rdx, 93
    je pop_case
    inc r8
    jmp build_loop
push_case:
    push r8
    inc r10
    inc r8
    jmp build_loop
pop_case:
    test r10, r10
    jz unbalanced
    dec r10
    pop rdx
    lea rsi, [r12 + r8*4]
    mov dword ptr [rsi], rdx
    lea rsi, [r12 + rdx*4]
    mov dword ptr [rsi], r8
    inc r8
    jmp build_loop
build_done:
    test r10, r10
    jz balanced
    jmp unbalanced
balanced:
    xor r8, r8
    xor r9, r9
    xor r10, r10
    mov r14, [prog_len]
interp_loop:
    cmp r8, r14
    jge interp_end
    lea rdx, [rip + tape]
    add rdx, r9
    movzx rax, byte ptr [rdx]
    lea rdx, [rip + input_buf]
    add rdx, r8
    movzx rbx, byte ptr [rdx]
    cmp bl, 62
    je cmd_gt
    cmp bl, 60
    je cmd_lt
    cmp bl, 43
    je cmd_plus
    cmp bl, 45
    je cmd_minus
    cmp bl, 46
    je cmd_dot
    cmp bl, 44
    je cmd_comma
    cmp bl, 91
    je cmd_left
    cmp bl, 93
    je cmd_right
    inc r8
    jmp interp_loop
cmd_gt:
    inc r9
    inc r8
    jmp interp_loop
cmd_lt:
    dec r9
    inc r8
    jmp interp_loop
cmd_plus:
    inc al
    lea rdx, [rip + tape]
    add rdx, r9
    mov byte ptr [rdx], al
    inc r8
    jmp interp_loop
cmd_minus:
    dec al
    lea rdx, [rip + tape]
    add rdx, r9
    mov byte ptr [rdx], al
    inc r8
    jmp interp_loop
cmd_dot:
    mov byte ptr [out_buf], al
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + out_buf]
    mov rdx, 1
    syscall
    inc r8
    jmp interp_loop
cmd_comma:
    mov rax, [input_len2]
    cmp r10, rax
    jge comma_exhausted
    lea rsi, [rip + input_buf]
    add rsi, [prog_len]
    inc rsi
    add rsi, r10
    movzx al, byte ptr [rsi]
    inc r10
    jmp comma_done
comma_exhausted:
    xor al, al
comma_done:
    lea rdx, [rip + tape]
    add rdx, r9
    mov byte ptr [rdx], al
    inc r8
    jmp interp_loop
cmd_left:
    test al, al
    jz left_jump
    inc r8
    jmp interp_loop
left_jump:
    lea r13, [rip + bracket_map]
    lea rsi, [r13 + r8*4]
    mov eax, [rsi]
    inc eax
    mov r8, eax
    jmp interp_loop
cmd_right:
    test al, al
    jnz right_jump
    inc r8
    jmp interp_loop
right_jump:
    lea r13, [rip + bracket_map]
    lea rsi, [r13 + r8*4]
    mov eax, [rsi]
    inc eax
    mov r8, eax
    jmp interp_loop
interp_end:
    mov rax, 60
    xor rdi, rdi
    syscall
unbalanced:
    mov rax, 60
    mov rdi, 1
    syscall
