.global _start
.equ bufsiz, 65536
.section .rodata
newline:
    .byte 10
.section .bss
.balign 4
buf:
    .space 65536
.section .text
_start:
    adrp x4, buf
    add x4, x4, :lo12:buf
    mov x2, #0

read_loop:
    sub x3, bufsiz, x2
    mov x0, #0
    add x1, x4, x2
    mov x2, x3
    mov x8, #63
    svc 0
    cmp x0, #0
    beq .eof

    add x2, x2, x0
    b read_loop

.eof:
    mov x5, #0
    mov x6, #0

process_loop:
    cmp x6, x2
    beq .process_done

    ldrb w7, [x4, x6]
    cmp w7, #10
    b.eq .newline_processed

    add x6, x6, #1
    b process_loop

newline_processed:
    sub x0, x6, x5
    mov x3, x0

    sub x9, x6, #1
    mov x11, #0

reverse_loop:
    cmp x11, x3
    b.ge .reverse_done

    sub x10, x9, x11
    add x1, x4, x10
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0
    add x11, x11, #1
    b reverse_loop

.reverse_done:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0

    add x6, x6, #1
    mov x5, x6
    b process_loop

.process_done:
    sub x8, x2, x5
    mov x9, #0

copy_loop:
    cmp x9, x8
    b.ge .copy_done

    add x10, x5, x9
    ldrb w7, [x4, x10]
    strb w7, [x4, x9]
    add x9, x9, #1
    b copy_loop

.copy_done:
    mov x2, x8

    cmp x2, #0
    beq .finalize_done

    mov x0, x2
    mov x3, x0
    sub x9, x2, #1
    mov x11, #0

final_reverse_loop:
    cmp x11, x3
    b.ge .final_reverse_done

    sub x10, x9, x11
    add x1, x4, x10
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0
    add x11, x11, #1
    b final_reverse_loop

.final_reverse_done:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0

.finalize_done:
    mov x0, #0
    mov x8, #93
    svc 0
