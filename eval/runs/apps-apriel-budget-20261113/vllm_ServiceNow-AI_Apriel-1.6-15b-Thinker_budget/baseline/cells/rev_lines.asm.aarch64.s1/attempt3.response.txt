.global _start
.equ BUFFER_SIZE, 65536

.section .rodata
.align 3
buffer_size:
    .quad BUFFER_SIZE
.align 2
nl:
    .byte 10

.section .bss
.balign 4
buf:
    .space BUFFER_SIZE

.section .text
_start:
    adrp x4, buf
    add x4, x4, :lo12:buf
    mov x5, #0

read_loop:
    adrp x6, buffer_size
    add x6, x6, :lo12:buffer_size
    ldr x6, [x6]
    sub x6, x6, x5
    mov x0, #0
    add x1, x4, x5
    mov x2, x6
    mov x8, #63
    svc 0
    cmp x0, #0
    beq .eof
    add x5, x5, x0
    b read_loop

.eof:
    mov x6, #0

process_loop:
    cmp x6, x5
    beq .process_done

    mov x2, x6
scan_start:
    cmp x2, x5
    beq .final_line_no_newline
    ldrb w3, [x4, x2]
    cmp w3, #10
    beq .newline_found
    add x2, x2, #1
    b scan_start

.newline_found:
    mov x10, x2
    mov x9, x2
    sub x9, x9, #1

rev_loop:
    cmp x9, x6
    b.lt .write_newline
    mov x1, x4
    add x1, x1, x9
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0
    sub x9, x9, #1
    cmp x9, x6
    b.ge rev_loop

.write_newline:
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0
    add x6, x10, #1
    b process_loop

.final_line_no_newline:
    mov x9, x5
    sub x9, x9, #1

rev_final_loop:
    cmp x9, x6
    b.lt .write_newline_final
    mov x1, x4
    add x1, x1, x9
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0
    sub x9, x9, #1
    cmp x9, x6
    b.ge rev_final_loop

.write_newline_final:
    adrp x1, nl
    add x1, x1, :lo12:nl
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc 0

.process_done:
    mov x0, #0
    mov x8, #93
    svc 0
