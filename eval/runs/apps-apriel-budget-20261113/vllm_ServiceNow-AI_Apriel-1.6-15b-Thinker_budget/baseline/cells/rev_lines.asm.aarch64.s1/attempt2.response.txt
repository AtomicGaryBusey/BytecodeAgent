.global _start
.equ BUFFER_SIZE, 65536

.section .rodata
.align 3
bufsize:
    .int 65536
.align 2
newline:
    .byte 10

.section .bss
.balign 4
buf:
    .space 65536

.section .text
_start:
    adrp x4, buf
    add x4, x4, :lo12:buf

    mov x5, #0

read_loop:
    adrp x7, bufsize
    add x7, x7, :lo12:bufsize
    ldr w7, [x7]
    mov x7, w7
    sub x8, x7, x5

    mov x0, #0
    add x1, x4, x5
    mov x2, x8
    mov x8, #63
    svc 0

    cmp x0, #0
    beq .eof

    add x5, x5, x0
    b read_loop

.eof:
    mov x2, #0
    mov x6, #0

process_loop:
    cmp x2, x5
    beq .process_done

    ldrb w3, [x4, x2]
    cmp w3, #10
    bne .not_newline

    mov x12, x2
    cmp x12, x6
    beq .empty_line

    sub x9, x12, #1
rev_loop:
    add x10, x4, x9
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc 0
    sub x9, x9, #1
    cmp x9, x6
    bge rev_loop

.empty_line:
    adrp x10, newline
    add x10, x10, :lo12:newline
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc 0

    add x6, x12, #1
    mov x2, x6
    b process_loop

.not_newline:
    add x2, x2, #1
    b process_loop

.process_done:
    cmp x6, x5
    beq .final_done

    mov x12, x5
    sub x9, x12, #1
final_rev_loop:
    add x10, x4, x9
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc 0
    sub x9, x9, #1
    cmp x9, x6
    bge final_rev_loop

    adrp x10, newline
    add x10, x10, :lo12:newline
    mov x0, #1
    mov x1, x10
    mov x2, #1
    mov x8, #64
    svc 0

.final_done:
    mov x0, #0
    mov x8, #93
    svc 0
