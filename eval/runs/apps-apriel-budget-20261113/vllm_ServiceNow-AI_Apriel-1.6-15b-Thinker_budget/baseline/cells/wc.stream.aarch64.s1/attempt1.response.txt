.globl _start
    .text
_start:
    // read loop
read_loop:
    adrp x1, buffer
    add x1, x1, x3
    adrp x5, BufSize
    add x5, x5, :lo12:BufSize
    ldr w4, [x5]
    sub w4, w4, x3
    mov x8, #63
    mov x0, #0
    mov x2, w4
    svc #0
    cbnz x0, read_loop_cont
    b scan_start
read_loop_cont:
    add x3, x3, x0
    b read_loop
scan_start:
    adrp x1, buffer
    add x1, x1, :lo12:buffer
    mov w9, x3
    // scanning loop
scan_loop:
    cbz w9, scan_done
    ldrb w10, [x1], #1
    cmp w10, #0x0A
    beq inc_line
inc_line:
    add w6, w6, #1
    cmp w10, #0x20
    beq ws
    cmp w10, #0x09
    beq ws
    cmp w10, #0x0A
    beq ws
    cmp w10, #0x0B
    beq ws
    cmp w10, #0x0C
    beq ws
    cmp w10, #0x0D
    beq ws
    cbz w8, new_word
    b continue_scan
new_word:
    add w7, w7, #1
    mov w8, #1
continue_scan:
    sub w9, w9, #1
    b scan_loop
ws:
    mov w8, #0
scan_done:
    // print lines
    mov w12, w6
    mov w13, #0
    mov w14, #0
    adrp x15, tmp_digits
    add x15, x15, :lo12:tmp_digits
div_loop:
    cbz w12, after_div
    mov w17, #10
    udiv w16, w12, w17
    add w12, w12, #0x30
    add x16, x15, w14
    strb w12, [x16]
    add w14, w14, #1
    mov w12, w16
    b div_loop
after_div:
    cbnz w14, print_digits
    mov w12, #0x30
    adrp x16, out_char
    add x16, x16, :lo12:out_char
    strb w12, [x16]
    mov x8, #64
    mov x0, #1
    mov x1, x16
    mov x2, #1
    svc #0
    b after_print_line
print_digits:
    sub w15, w14, #1
    adrp x16, tmp_digits
    add x16, x16, :lo12:tmp_digits
print_loop:
    cbnz w15, load_digit
    b done_printing
load_digit:
    add x17, x16, w15
    ldrb w12, [x17]
    adrp x18, out_char
    add x18, x18, :lo12:out_char
    strb w12, [x18]
    mov x8, #64
    mov x0, #1
    mov x1, x18
    mov x2, #1
    svc #0
    sub w15, w15, #1
    b print_loop
after_print_line:
    // write space
    mov w12, #0x20
    adrp x16, out_char
    add x16, x16, :lo12:out_char
    strb w12, [x16]
    mov x8, #64
    mov x0, #1
    mov x1, x16
    mov x2, #1
    svc #0
    // print words
    mov w12, w7
    mov w13, #0
    mov w14, #0
    adrp x15, tmp_digits
    add x15, x15, :lo12:tmp_digits
div_loop2:
    cbz w12, after_div2
    mov w17, #10
    udiv w16, w12, w17
    add w12, w12, #0x30
    add x16, x15, w14
    strb w12, [x16]
    add w14, w14, #1
    mov w12, w16
    b div_loop2
after_div2:
    cbnz w14, print_digits2
    mov w12, #0x30
    adrp x16, out_char
    add x16, x16, :lo12:out_char
    strb w12, [x16]
    mov x8, #64
    mov x0, #1
    mov x1, x16
    mov x2, #1
    svc #0
    b after_print_word
print_digits2:
    sub w15, w14, #1
    adrp x16, tmp_digits
    add x16, x16, :lo12:tmp_digits
print_loop2:
    cbnz w15, load_digit2
    b done_printing2
load_digit2:
    add x17, x16, w15
    ldrb w12, [x17]
    adrp x18, out_char
    add x18, x18, :lo12:out_char
    strb w12, [x18]
    mov x8, #64
    mov x0, #1
    mov x1, x18
    mov x2, #1
    svc #0
    sub w15, w15, #1
    b print_loop2
after_print_word:
    // write space
    mov w12, #0x20
    adrp x16, out_char
    add x16, x16, :lo12:out_char
    strb w12, [x16]
    mov x8, #64
    mov x0, #1
    mov x1, x16
    mov x2, #1
    svc #0
    // print bytes
    mov w12, x3
    mov w13, #0
    mov w14, #0
    adrp x15, tmp_digits
    add x15, x15, :lo12:tmp_digits
div_loop3:
    cbz w12, after_div3
    mov w17, #10
    udiv w16, w12, w17
    add w12, w12, #0x30
    add x16, x15, w14
    strb w12, [x16]
    add w14, w14, #1
    mov w12, w16
    b div_loop3
after_div3:
    cbnz w14, print_digits3
    mov w12, #0x30
    adrp x16, out_char
    add x16, x16, :lo12:out_char
    strb w12, [x16]
    mov x8, #64
    mov x0, #1
    mov x1, x16
    mov x2, #1
    svc #0
    b after_print_bytes
print_digits3:
    sub w15, w14, #1
    adrp x16, tmp_digits
    add x16, x16, :lo12:tmp_digits
print_loop3:
    cbnz w15, load_digit3
    b done_printing3

    add x17, x16, w15
    ldrb w12, [x17]
    adrp x18, out_char
    add x18, x18, :lo12:out_char
    strb w12, [x18]

    mov x1, x18

    svc #0

    // write newline
    mov w12, #0x0A
    adrp x16, out_char
    add x16, x16, :lo12:out_char
    strb w12, [x16]
    mov x8, #64
    mov x0, #1
    mov x1, x16
    mov x2, #1
    svc #0
    // exit
    mov x8, #93
    mov x0, #0
    svc #0
    .section .rodata

    .int 65536
    .bss
