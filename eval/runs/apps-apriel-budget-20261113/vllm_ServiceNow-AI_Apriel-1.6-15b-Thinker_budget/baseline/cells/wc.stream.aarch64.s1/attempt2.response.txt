.globl _start
.text
_start:
    // initialize counters
    mov w9, #0      // lines
    mov w10, #0     // words
    mov w11, #0     // bytes
    mov w12, #0     // in_word flag

read_loop:
    // sys_read
    mov w8, #63          // syscall read
    mov w1, #0           // fd = 0
    adrp x2, buf
    add x2, x2, :lo12:buf
    mov w3, #4096        // read up to 4096 bytes
    svc #0
    // x0 = bytes read
    cmp x0, #0
    beq end_input

    // process bytes
    mov x4, x2           // pointer to buffer
    mov x5, x0           // bytes to process
process_bytes:
    cmp x5, #0
    beq continue_read

    ldrb w6, [x4], #1   // w6 = byte, advance pointer
    add x11, x11, #1     // total bytes

    cmp w6, #10          // newline 0x0A
    beq newline_case

    cmp w6, #32          // space
    beq whitespace_case
    cmp w6, #9           // tab
    beq whitespace_case
    cmp w6, #11          // vertical tab
    beq whitespace_case
    cmp w6, #12          // form feed
    beq whitespace_case
    cmp w6, #13          // carriage return
    beq whitespace_case

    // non-whitespace
    cmp w12, #0
    beq start_word
    // already in word: nothing

    sub x5, x5, #1
    b process_bytes

newline_case:
    add x9, x9, #1       // lines++
    mov w12, #0          // end word
    sub x5, x5, #1
    b process_bytes

whitespace_case:
    mov w12, #0
    sub x5, x5, #1
    b process_bytes

start_word:
    add x10, x10, #1     // words++
    mov w12, #1
    sub x5, x5, #1
    b process_bytes

continue_read:
    b read_loop

end_input:
    // output lines
    // convert lines (x9) to decimal string in num_buf, write it
    adrp x6, num_buf_end
    add x6, x6, :lo12:num_buf_end
    mov x7, #0
    mov x5, x9
    cmp x5, #0
    bne convert_nonzero
    sub x6, x6, #1
    mov w0, #48
    strb w0, [x6]
    mov x7, #1
convert_nonzero:
    mov x15, #10
convert_loop:
    cmp x5, #0
    beq convert_done
    mov x8, x5
    udiv x5, x5, x15
    mul x9, x5, x15
    sub x8, x8, x9
    sub x6, x6, #1
    add w0, w8, #48
    strb w0, [x6]
    add x7, x7, #1
    b convert_loop
convert_done:
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0

    // write space
    adrp x6, space_char
    add x6, x6, :lo12:space_char
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, #1
    svc #0

    // output words (x10)
    adrp x6, num_buf_end
    add x6, x6, :lo12:num_buf_end
    mov x7, #0
    mov x5, x10
    cmp x5, #0
    bne convert_words_nonzero
    sub x6, x6, #1
    mov w0, #48
    strb w0, [x6]
    mov x7, #1
    b convert_words_done
convert_words_nonzero:
    mov x15, #10
conv_loop_w:
    cmp x5, #0
    beq conv_done_w
    mov x8, x5
    udiv x5, x5, x15
    mul x9, x5, x15
    sub x8, x8, x9
    sub x6, x6, #1
    add w0, w8, #48
    strb w0, [x6]
    add x7, x7, #1
    b conv_loop_w
conv_done_w:
convert_words_done:
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0

    // write space
    adrp x6, space_char
    add x6, x6, :lo12:space_char
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, #1
    svc #0

    // output bytes (x11)
    adrp x6, num_buf_end
    add x6, x6, :lo12:num_buf_end
    mov x7, #0
    mov x5, x11
    cmp x5, #0
    bne convert_bytes_nonzero
    sub x6, x6, #1
    mov w0, #48
    strb w0, [x6]
    mov x7, #1
    b convert_bytes_done
convert_bytes_nonzero:
    mov x15, #10
conv_loop_b:
    cmp x5, #0
    beq conv_done_b
    mov x8, x5
    udiv x5, x5, x15
    mul x9, x5, x15
    sub x8, x8, x9
    sub x6, x6, #1
    add w0, w8, #48
    strb w0, [x6]
    add x7, x7, #1
    b conv_loop_b
conv_done_b:
convert_bytes_done:
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0

    // write newline
    adrp x6, newline_char
    add x6, x6, :lo12:newline_char
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, #1
    svc #0

    // exit
    mov w8, #93
    mov x0, #0
    svc #0

    .section .rodata
space_char:
    .byte 32
newline_char:
    .byte 10

    .section .bss
buf:
    .space 4096
    .align 4
num_buf:
    .space 12
num_buf_end:
