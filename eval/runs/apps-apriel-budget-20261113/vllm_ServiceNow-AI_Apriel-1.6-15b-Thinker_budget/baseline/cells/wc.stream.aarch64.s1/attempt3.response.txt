.globl _start
.text
_start:
    mov w9, #0      // lines
    mov w10, #0     // words
    mov w11, #0     // bytes
    mov w12, #0     // in_word flag

read_loop:
    mov w8, #63          // sys_read
    mov x0, #0           // fd 0
    adrp x1, buf
    add x1, x1, :lo12:buf
    mov w2, #4096        // buffer size
    svc #0
    cmp x0, #0
    beq end_input

    mov x4, x1           // pointer to buffer
    mov x5, x0           // bytes to process

process_bytes:
    cmp x5, #0
    beq read_loop
    ldrb w6, [x4], #1
    sub x5, x5, #1
    add w11, w11, #1

    cmp w6, #10
    beq newline_case

    cmp w6, #32
    beq whitespace_case
    cmp w6, #9
    beq whitespace_case
    cmp w6, #11
    beq whitespace_case
    cmp w6, #12
    beq whitespace_case
    cmp w6, #13
    beq whitespace_case

    cmp w12, #0
    beq start_word
    b process_bytes

newline_case:
    add w9, w9, #1
    mov w12, #0
    b process_bytes

whitespace_case:
    mov w12, #0
    b process_bytes

start_word:
    add w10, w10, #1
    mov w12, #1
    b process_bytes

end_input:
    // output lines
    mov x5, w9
    adrp x6, num_buf_end
    add x6, x6, :lo12:num_buf_end
    sub x6, x6, #1
    mov x7, #0
    mov x15, #10
    cmp x5, #0
    beq convert_zero
convert_loop:
    mov x16, x5
    udiv x5, x5, x15
    mul x17, x5, x15
    sub x16, x16, x17
    mov w0, x16
    add w0, w0, #48
    strb w0, [x6]
    sub x6, x6, #1
    add x7, x7, #1
    cmp x5, #0
    bne convert_loop
    b convert_done

convert_zero:
    mov w0, #48
    strb w0, [x6]
    sub x6, x6, #1
    mov x7, #1

convert_done:
    add x6, x6, #1
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0

    // write space
    adrp x6, space_char
    add x6, x6, :lo12:space_char
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, #1
    svc #0

    // output words
    mov x5, w10
    adrp x6, num_buf_end
    add x6, x6, :lo12:num_buf_end
    sub x6, x6, #1
    mov x7, #0
    mov x15, #10
    cmp x5, #0
    beq convert_zero2
convert_loop2:
    mov x16, x5
    udiv x5, x5, x15
    mul x17, x5, x15
    sub x16, x16, x17
    mov w0, x16
    add w0, w0, #48
    strb w0, [x6]
    sub x6, x6, #1
    add x7, x7, #1
    cmp x5, #0
    bne convert_loop2
    b convert_done2

convert_zero2:
    mov w0, #48
    strb w0, [x6]
    sub x6, x6, #1
    mov x7, #1

convert_done2:
    add x6, x6, #1
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0

    // write space
    adrp x6, space_char
    add x6, x6, :lo12:space_char
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, #1
    svc #0

    // output bytes
    mov x5, w11
    adrp x6, num_buf_end
    add x6, x6, :lo12:num_buf_end
    sub x6, x6, #1
    mov x7, #0
    mov x15, #10
    cmp x5, #0
    beq convert_zero3
convert_loop3:
    mov x16, x5
    udiv x5, x5, x15
    mul x17, x5, x15
    sub x16, x16, x17
    mov w0, x16
    add w0, w0, #48
    strb w0, [x6]
    sub x6, x6, #1
    add x7, x7, #1
    cmp x5, #0
    bne convert_loop3
    b convert_done3

convert_zero3:
    mov w0, #48
    strb w0, [x6]
    sub x6, x6, #1
    mov x7, #1

convert_done3:
    add x6, x6, #1
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0

    // write newline
    adrp x6, newline_char
    add x6, x6, :lo12:newline_char
    mov w8, #64
    mov x0, #1
    mov x1, x6
    mov x2, #1
    svc #0

    // exit
    mov w8, #93
    mov x0, #0
    svc #0

    .section .rodata
space_char:
    .byte 32
newline_char:
    .byte 10

    .section .bss
buf:
    .space 4096
    .align 4
num_buf:
    .space 12
num_buf_end:
