.global _start
.text
_start:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rel .input]
    mov rdx, 65536
    xor r10, r10
    mov [rel .buf_len], r10
read_loop:
    syscall
    test rax, rax
    jz read_done
    mov r10, rax
    mov r9, [rel .buf_len]
    lea rsi, [rel .input + r9]
    sub rdx, r9
    syscall
    mov [rel .buf_len], rax
    jmp read_loop
read_done:
    lea rsi, [rel .input]
    mov r10, [rel .buf_len]
    lea rdx, [rel .input + r10]
    xor r14, r14
    lea r12, [rel .stack]
    xor r13, r13
    jmp skip_sep

skip_sep:
    cmp rsi, rdx
    jge line_end
    mov al, byte [rsi]
    cmp al, 0x09
    je skip_sep
    cmp al, 0x20
    je skip_sep
    cmp al, 0x0A
    je line_end
    mov r9, rsi
    mov r8, r9
token_scan:
    cmp r8, rdx
    jge token_done
    mov al, byte [r8]
    cmp al, 0x09
    je token_done
    cmp al, 0x20
    je token_done
    cmp al, 0x0A
    je token_done
    inc r8
    jmp token_scan
token_done:
    sub r8, r9
    cmp r8, 19
    jg token_error
    mov al, byte [r9]
    cmp r8, 1
    jne token_literal
    cmp al, 43
    je op_plus
    cmp al, 45
    je op_minus
    cmp al, 42
    je op_mul
    cmp al, 47
    je op_div
    cmp al, 37
    je op_mod
    jmp token_error
token_literal:
    cmp al, '-'
    je literal_negative
    cmp al, 48
    jl token_error
    cmp al, 57
    jg token_error
    mov r10, r9
    mov r11, r9
    add r11, r8
    xor rax, rax
literal_digit_loop:
    cmp r10, r11
    jge literal_done
    movzx rcx, byte [r10]
    cmp rcx, 48
    jl token_error
    cmp rcx, 57
    jg token_error
    sub rcx, 48
    imul rax, 10
    add rax, rcx
    inc r10
    jmp literal_digit_loop
literal_done:
    jmp push_result
literal_negative:
    cmp r8, 2
    jl token_error
    mov r10, r9
    add r10, 1
    mov r11, r9
    add r11, r8
    xor rax, rax
literal_digit_loop_neg:
    cmp r10, r11
    jge literal_done_neg
    movzx rcx, byte [r10]
    cmp rcx, 48
    jl token_error
    cmp rcx, 57
    jg token_error
    sub rcx, 48
    imul rax, 10
    add rax, rcx
    inc r10
    jmp literal_digit_loop_neg
literal_done_neg:
    neg rax
    jmp push_result
push_result:
    mov [r12], rax
    lea r12, [r12 + 8]
    inc r13
    mov rsi, r9
    add rsi, r8
    jmp skip_sep

op_plus:
    cmp r13, 2
    jl operator_error
    mov rax, [r12 - 16]
    mov rbx, [r12 - 8]
    add rax, rbx
    jmp push_result
op_minus:
    cmp r13, 2
    jl operator_error
    mov rax, [r12 - 16]
    mov rbx, [r12 - 8]
    sub rax, rbx
    jmp push_result
op_mul:
    cmp r13, 2
    jl operator_error
    mov rax, [r12 - 16]
    mov rbx, [r12 - 8]
    imul rax, rbx
    jmp push_result
op_div:
    cmp r13, 2
    jl operator_error
    mov rax, [r12 - 16]   ; a
    mov rbx, [r12 - 8]    ; b
    test rbx, rbx
    jz division_by_zero
    cmp rax, MIN_VAL
    jne div_ok
    cmp rbx, -1
    jne div_ok
div_error:
    jmp operator_error
div_ok:
    movsx rdx, rax
    idiv rbx
    mov rax, rdx
    jmp push_result
division_by_zero:
    jmp operator_error

op_mod:
    cmp r13, 2
    jl operator_error
    mov rax, [r12 - 16]   ; a
    mov rbx, [r12 - 8]    ; b
    test rbx, rbx
    jz modulo_by_zero
    cmp rax, MIN_VAL
    jne mod_ok
    cmp rbx, -1
    jne mod_ok
mod_error:
    jmp operator_error
mod_ok:
    movsx rdx, rax
    idiv rbx
    jmp push_result
modulo_by_zero:
    jmp operator_error

operator_error:
    mov r14, 1
    jmp line_end
token_error:
    mov r14, 1
    jmp line_end

line_end:
    test r14, r14
    jz success_check
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .error_msg]
    mov rdx, .error_msg_len
    syscall
    lea r12, [rel .stack]
    xor r13, r13
    jmp skip_newline

success_check:
    cmp r13, 1
    jne error_after_success
    mov rax, [r12 - 8]
    cmp rax, MIN_VAL
    je output_min_seq
    test rax, rax
    jge output_pos_seq
    neg rax
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .dash]
    mov rdx, 1
    syscall
output_pos_seq:
    xor r8, r8
digit_loop_seq:
    test rax, rax
    jz digit_done_seq
    xor rdx, rdx
    mov r9, 10
    div r9
    add rdx, 48
    mov byte [rel .digit_buf + r8], dl
    inc r8
    jmp digit_loop_seq
digit_done_seq:
    mov r9, r8
    dec r9
    mov r10, r9
output_digit_seq:
    test r10, r10
    jge output_done_seq
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .digit_buf + r10]
    mov rdx, 1
    syscall
    dec r10
    jmp output_digit_seq
output_done_seq:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .nl]
    mov rdx, 1
    syscall
    lea r12, [rel .stack]
    xor r13, r13
    jmp skip_newline
output_min_seq:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .min_str]
    mov rdx, .min_str_len
    syscall
    lea r12, [rel .stack]
    xor r13, r13
    jmp skip_newline

error_after_success:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .error_msg]
    mov rdx, .error_msg_len
    syscall
    lea r12, [rel .stack]
    xor r13, r13
    jmp skip_newline

skip_newline:
    cmp rsi, rdx
    jge continue_next_line
    mov al, byte [rsi]
    cmp al, 0x0A
    je newline_skip
    jmp continue_next_line
newline_skip:
    inc rsi
    jmp skip_newline

continue_next_line:
    cmp rsi, rdx
    jge end_program
    jmp skip_sep

end_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.bss
.stack: .skip 16384
.buf_len: .quad 0
.input: .skip 65536

.rodata
.error_msg: .asciz "error\n"
.error_msg_len = . - .error_msg
.min_str: .asciz "-9223372036854775808\n"
.min_str_len = . - .min_str
.dash: .byte '-'
.nl: .byte 0x0A
.digit_buf: .skip 20

    .equ MIN_VAL, -9223372036854775808
