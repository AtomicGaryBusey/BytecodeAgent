.global _start
.text
_start:
    xor r8, r8
read:
    mov rax, 0
    mov rdi, 0
    lea rsi, [rel .input + r8*8]
    mov rdx, BUF_SIZE
    sub rdx, r8
    syscall
    test rax, rax
    jnz read_cont
    jmp read_done
read_cont:
    add r8, rax
    jmp read
read_done:
    mov [rel .input_len], r8
    mov rcx, .input
    mov rax, [rel .input_len]
    lea rdx, [rcx + rax]
    mov rsi, rcx
process:
    cmp rsi, rdx
    jge done
    mov rax, rsi
find_nl:
    cmp byte ptr [rax], 0x0A
    je nl_found
    cmp rax, rdx
    jge nl_end
    inc rax
    jmp find_nl
nl_found:
    mov r8, rax
    jmp parse_line
nl_end:
    mov r8, rdx
    jmp parse_line
parse_line:
    xor r10, r10
skip_sep:
    cmp rsi, r8
    jge line_check
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je skip_sep
    cmp al, 0x09
    je skip_sep
    mov bl, byte ptr [rsi]
    inc rsi
    cmp bl, '-'
    je token_minus
    cmp bl, '+'
    je token_op
    cmp bl, '*'
    je token_op
    cmp bl, '/'
    je token_op
    cmp bl, '%'
    je token_op
    jmp line_error
token_op:
    cmp rsi, r8
    jge apply_op
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je apply_op
    cmp al, 0x09
    je apply_op
    jmp line_error
apply_op:
    jmp operator_handler
token_minus:
    cmp rsi, r8
    jge op_minus
    mov al, byte ptr [rsi]
    cmp al, 0x20
    je op_minus
    cmp al, 0x09
    je op_minus
    mov r12, 1
    xor r13, r13
    xor r14, r14
    jmp literal_digits
op_minus:
    jmp operator_handler
literal_digits:
digit_loop:
    cmp rsi, r8
    jge literal_error_no_digit
    mov al, byte ptr [rsi]
    cmp al, '0'
    jb literal_error_invalid
    cmp al, '9'
    ja literal_error_invalid
    sub al, '0'
    imul r13, r13, 10
    add r13, rax
    inc r14
    cmp r14, 19
    jja literal_error_too_long
    inc rsi
    jmp digit_loop
literal_error_no_digit:
    jmp line_error
literal_error_invalid:
    jmp line_error
literal_error_too_long:
    jmp line_error
    test r12, r12
    jz literal_push
    neg r13
literal_push:
    lea rcx, [rel .stack + r10*8]
    mov qword ptr [rcx], r13
    inc r10
    jmp skip_sep
operator_handler:
    cmp r10, 2
    jb operator_error
    dec r10
    mov rbx, qword ptr [rel .stack + r10*8]
    dec r10
    mov rax, qword ptr [rel .stack + r10*8]
    cmp bl, '+'
    je op_add
    cmp bl, '-'
    je op_sub
    cmp bl, '*'
    je op_mul
    cmp bl, '/'
    je op_div
    cmp bl, '%'
    je op_mod
    jmp line_error
op_add:
    add rax, rbx
    jmp op_store
op_sub:
    sub rax, rbx
    jmp op_store
op_mul:
    imul rax, rbx
    jmp op_store
op_div:
    test rbx, rbx
    jnz div_ok
    jmp line_error
    cmp rax, -9223372036854775808
    jne div_ok
    cmp rbx, -1
    jne div_ok
    jmp line_error
div_ok:
    cqo
    idiv rbx
    jmp op_store
op_mod:
    test rbx, rbx
    jnz mod_ok
    jmp line_error
    cmp rax, -9223372036854775808
    jne mod_ok
    cmp rbx, -1
    jne mod_ok
    jmp line_error
mod_ok:
    cqo
    idiv rbx
    mov rax, rdx
    jmp op_store
op_store:
    lea rcx, [rel .stack + r10*8]
    mov qword ptr [rcx], rax
    inc r10
    jmp skip_sep
operator_error:
    jmp line_error
line_check:
    cmp r10, 1
    jne line_error
    mov rax, qword ptr [rel .stack]
    mov r11, rdx
    jmp print_int_and_nl
print_int_and_nl:
    mov rbx, rax
    xor r15, r15
    test rax, rax
    jge print_loop
    mov r12, 1
    neg rbx
    jmp print_loop
print_loop:
    test rbx, rbx
    jz print_done
    mov rax, rbx
    xor rdx, rdx
    mov rcx, 10
    div rcx
    add rdx, 48
    inc r15
    push rdx
    mov rbx, rax
    jmp print_loop
print_done:
    test r12, r12
    jz no_minus
    push byte 45
    inc r15
no_minus:
    test r15, r15
    jnz print_emit
    push byte 48
    mov r15, 1
print_emit:
    mov rcx, r15
pop_digit_loop:
    lea rsi, [rsp]
    mov al, byte ptr [rsp]
    mov rax, 1
    mov rdi, 1
    mov rdx, 1
    syscall
    add rsp, 8
    dec rcx
    jnz pop_digit_loop
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .nl]
    mov rdx, 1
    syscall
    mov rdx, r11
    jmp parse_line_done
parse_line_done:
    cmp r8, rdx
    jne have_newline
    jmp done
have_newline:
    lea rsi, [r8+1]
    jmp process
line_error:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rel .error_msg]
    mov rdx, .error_len
    syscall
    mov rsi, r8
    cmp r8, rdx
    jne have_newline_err
    jmp done
have_newline_err:
    lea rsi, [r8+1]
    jmp process
done:
    mov rax, 60
    xor rdi, rdi
    syscall
.section .rodata
.error_msg: .asciz "error\n"
.error_len = 6
.nl: .byte 0x0A
.section .bss
.input: .space 65536
.input_len: .zero 8
.stack: .space 2048
.equ BUF_SIZE, 65536
