.section .rodata
nibble:
    .byte "0123456789abcdef"

    .section .bss
off: .quad 0
inbuf: .space 16
outbuf: .space 128

    .section .text
    .globl _start
_start:
    // main loop
L_loop:
    // read 16 bytes
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, #0
    mov x2, #16
    svc #0
    // x0 = bytes read
    cbnz x0, L_has_data
    // no more data: format final offset line
    mov x4, #off
    ldr x5, [x4]          // current offset
    adrp x6, outbuf
    add x6, x6, #0
    movz x7, xzr
    adrp x8, nibble
    add x8, x8, #0
    // format offset hex
    mov x9, x5
    // nibble 0 shift 28
    lsr x10, x9, #28
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 1 shift 24
    lsr x10, x9, #24
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 2 shift 20
    lsr x10, x9, #20
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 3 shift 16
    lsr x10, x9, #16
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 4 shift 12
    lsr x10, x9, #12
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 5 shift 8
    lsr x10, x9, #8
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 6 shift 4
    lsr x10, x9, #4
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 7 shift 0
    lsr x10, x9, #0
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // newline
    movz w11, #10
    strb w11, [x6, x7]
    add x7, x7, #1
    // write outbuf
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0
    // exit
    mov x0, #0
    svc #93
    // end
L_has_data:
    // store n
    mov x14, x0          // n = bytes read
    // load current offset
    mov x4, #off
    ldr x5, [x4]         // offset before increment
    // outbuf address
    adrp x6, outbuf
    add x6, x6, #0
    movz x7, xzr
    // nibble table base
    adrp x8, nibble
    add x8, x8, #0
    // format offset hex
    mov x9, x5
    // nibble 0 shift 28
    lsr x10, x9, #28
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 1 shift 24
    lsr x10, x9, #24
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 2 shift 20
    lsr x10, x9, #20
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 3 shift 16
    lsr x10, x9, #16
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 4 shift 12
    lsr x10, x9, #12
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 5 shift 8
    lsr x10, x9, #8
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 6 shift 4
    lsr x10, x9, #4
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // nibble 7 shift 0
    lsr x10, x9, #0
    and x10, x10, #0xF
    ldrb w11, [x8, x10, lsl #2]
    strb w11, [x6, x7]
    add x7, x7, #1
    // two spaces
    movz w11, #32
    strb w11, [x6, x7]
    add x7, x7, #1
    strb w11, [x6, x7]
    add x7, x7, #1
    // hex bytes loop
    mov x9, #0
L_byte_loop:
    cmp x9, #15
    b.gt L_after_bytes
    cmp x9, x14
    blt L_has_byte
L_missing_byte:
    // three spaces
    movz w11, #32
    strb w11, [x6, x7]
    add x7, x7, #1
    strb w11, [x6, x7]
    add x7, x7, #1
    strb w11, [x6, x7]
    add x7, x7, #1
    // extra space after byte 7?
    cmp x9, #7
    b.ne L_byte_continue
    // extra space
    strb w11, [x6, x7]
    add x7, x7, #1
L_byte_continue:
    add x9, x9, #1
    b L_byte_loop
L_has_byte:
    // address of byte in inbuf
    add x10, x1, x9, lsl #0
    ldrb w11, [x10]
    // high nibble
    lsr w12, w11, #4
    and w12, w12, #0xF
    ldrb w13, [x8, w12, lsl #2]
    strb w13, [x6, x7]
    add x7, x7, #1
    // low nibble
    and w11, w11, #0xF
    ldrb w13, [x8, w11, lsl #2]
    strb w13, [x6, x7]
    add x7, x7, #1
    // space after two hex digits
    movz w13, #32
    strb w13, [x6, x7]
    add x7, x7, #1
    // extra space after byte 7?
    cmp x9, #7
    b.ne L_byte_continue2
    // extra space
    strb w13, [x6, x7]
    add x7, x7, #1
L_byte_continue2:
    add x9, x9, #1
    b L_byte_loop
L_after_bytes:
    // space and '|'
    movz w13, #32
    strb w13, [x6, x7]
    add x7, x7, #1
    movz w13, #124
    strb w13, [x6, x7]
    add x7, x7, #1
    // ascii part
    mov x9, #0
L_ascii_loop:
    cmp x9, x14
    b.lt L_ascii_body
    // reached end
    b L_end_ascii
L_ascii_body:
    // address of ascii byte
    add x10, x1, x9, lsl #0
    ldrb w11, [x10]
    // printable?
    cmp w11, #0x20
    b.lt L_ascii_dot
    cmp w11, #0x7e
    b.gt L_ascii_dot
    // printable
    strb w11, [x6, x7]
    add x7, x7, #1
    b L_ascii_continue
L_ascii_dot:
    movz w11, #46
    strb w11, [x6, x7]
    add x7, x7, #1
L_ascii_continue:
    add x9, x9, #1
    b L_ascii_loop
L_end_ascii:
    // '|' and newline
    movz w13, #124
    strb w13, [x6, x7]
    add x7, x7, #1
    movz w13, #10
    strb w13, [x6, x7]
    add x7, x7, #1
    // write outbuf
    mov x0, #1
    mov x1, x6
    mov x2, x7
    svc #0
    // update offset
    add x5, x5, x14
    // store back
    str x5, [x4]
    // continue loop
    b L_loop
