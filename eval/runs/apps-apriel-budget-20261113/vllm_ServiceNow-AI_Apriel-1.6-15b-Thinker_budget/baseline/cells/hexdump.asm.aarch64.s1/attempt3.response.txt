newline: .byte 10

.section .bss
.align 4

.align 4
inbuf: .space 65536
.align 4

.section .text
.global _start
_start:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, #0
    mov x3, x1
    mov x2, #65536
    mov x8, #63
    svc #0
    mov x14, x0
    cmp x0, #0
    beq done_reading
    mov x2, x3
    b process_rows

process_rows:
    cmp x14, #0
    beq done_reading

    cmp x14, #16
    blt set_row_bytes
    mov x3, #16
    b after_row_bytes

set_row_bytes:
    mov x3, x14

after_row_bytes:
    adrp x9, outbuf
    add x9, x9, #0
    mov x7, x9

    adrp x4, off
    add x4, x4, #0
    ldr w5, [x4]

    adrp x6, nibble
    add x6, x6, #0

    lsr x10, x5, #28
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #24
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #20
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #16
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #12

    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #4
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #0
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1

    adrp x11, space
    add x11, x11, #0
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    strb w12, [x7, #0]
    add x7, x7, #1

    mov x15, #0
L_byte_loop:
    cmp x15, #16
    beq L_after_byte_loop
    cmp x15, x3
    blt L_has_byte

L_missing_byte:
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    strb w12, [x7, #0]
    add x7, x7, #1
    strb w12, [x7, #0]
    add x7, x7, #1

    cmp x15, #7
    beq L_extra_space
    add x15, x15, #1
    b L_byte_loop

L_has_byte:
    add x12, x2, x15
    ldrb w12, [x12]

    lsr w13, w12, #4
    mov x13, w13
    ldrb w14, [x6, x13]
    strb w14, [x7, #0]
    add x7, x7, #1

    and w12, w12, #0xF
    mov x13, w12
    ldrb w14, [x6, x13]
    strb w14, [x7, #0]
    add x7, x7, #1

    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1

    cmp x15, #7
    beq L_extra_space2
    add x15, x15, #1
    b L_byte_loop

L_extra_space:
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    add x15, x15, #1
    b L_byte_loop

L_extra_space2:
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    add x15, x15, #1
    b L_byte_loop

    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1

    adrp x12, pipe
    add x12, x12, #0
    ldrb w12, [x12]
    strb w12, [x7, #0]
    add x7, x7, #1

    mov x15, #0
L_ascii_loop:
    cmp x15, x3
    beq L_ascii_done
    add x12, x2, x15
    ldrb w12, [x12]
    cmp w12, #0x20
    blt L_ascii_dot
    cmp w12, #0x7e
    bgt L_ascii_dot
    strb w12, [x7, #0]
    add x7, x7, #1
    b L_ascii_next

    strb w12, [x7, #0]
    add x7, x7, #1

    add x15, x15, #1
    b L_ascii_loop
L_ascii_done:

    strb w12, [x7, #0]
    add x7, x7, #1

    adrp x12, newline
    add x12, x12, #0
    ldrb w12, [x12]
    strb w12, [x7, #0]
    add x7, x7, #1

    sub x20, x7, x9
    mov x0, #1
    mov x1, x7
    mov x2, x20
    mov x8, #64
    svc #0

    add w5, w5, x3

    add x2, x2, x3

    b process_rows

    adrp x4, off

    add x6, x6, #0

    lsr x10, x5, #28

    lsr x10, x5, #20

    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #16
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #12

    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1

    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #4

    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1
    lsr x10, x5, #0
    and x10, x10, #0xF
    ldrb w11, [x6, x10]
    strb w11, [x7, #0]
    add x7, x7, #1

    adrp x12, newline
    add x12, x12, #0
    ldrb w12, [x12]
    strb w12, [x7, #0]
    add x7, x7, #1

    sub x20, x7, x9
    mov x0, #1
    mov x1, x7
    mov x2, x20
    mov x8, #64
    svc #0
