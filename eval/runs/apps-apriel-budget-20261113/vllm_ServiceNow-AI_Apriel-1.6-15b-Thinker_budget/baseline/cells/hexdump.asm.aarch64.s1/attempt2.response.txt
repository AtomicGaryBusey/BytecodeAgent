.section .rodata
nibble: .byte "0123456789abcdef"
space_char: .byte 32
pipe_char: .byte 124
newline_char: .byte 10

.section .bss
.align 4
off: .space 4
.align 4
inbuf: .space 16
.align 4
outbuf: .space 100

.section .text
.global _start
_start:
    // read loop
read_loop:
    mov x0, #0
    adrp x1, inbuf
    add x1, x1, #0
    mov x2, #16
    mov x8, #63
    svc #0
    cbnz x0, have_data

    // no more data: output final offset line
    adrp x4, off
    add x4, x4, #0
    ldr x5, [x4]        // offset

    adrp x9, outbuf
    add x9, x9, #0
    mov x7, x9         // outbuf pointer

    adrp x6, nibble
    add x6, x6, #0

    // nibble 7 (most significant)
    lsr x10, x5, #28
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 6
    lsr x10, x5, #24
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 5
    lsr x10, x5, #20
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 4
    lsr x10, x5, #16
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 3
    lsr x10, x5, #12
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 2
    lsr x10, x5, #8
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 1
    lsr x10, x5, #4
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 0
    lsr x10, x5, #0
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1

    // newline
    adrp x11, newline_char
    add x11, x11, #0
    ldrb w11, [x11]
    strb w11, [x7, #0]
    add x7, x7, #1

    // write
    sub x2, x7, x9
    mov x0, #1
    mov x1, x7
    mov x8, #64
    svc #0

    // exit
    mov x0, #0
    mov x8, #93
    svc #0

have_data:
    mov x14, x0          // n = bytes read

    // load current offset
    adrp x4, off
    add x4, x4, #0
    ldr x5, [x4]

    // outbuf base and pointer
    adrp x9, outbuf
    add x9, x9, #0
    mov x7, x9

    // nibble table
    adrp x6, nibble
    add x6, x6, #0

    // offset hex digits (8 nibbles)
    lsr x10, x5, #28
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 6
    lsr x10, x5, #24
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 5
    lsr x10, x5, #20
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 4
    lsr x10, x5, #16
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 3
    lsr x10, x5, #12
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 2
    lsr x10, x5, #8
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 1
    lsr x10, x5, #4
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1
    // nibble 0
    lsr x10, x5, #0
    and x10, x10, #0xF
    ldrb w11, [x6, x10, lsl #2]
    strb w11, [x7, #0]
    add x7, x7, #1

    // two spaces after offset
    adrp x11, space_char
    add x11, x11, #0
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    strb w12, [x7, #0]
    add x7, x7, #1

    // byte loop
    mov x15, #0          // i

L_byte_loop:
    cmp x15, #16
    beq L_after_bytes

    cmp x15, x14
    blt L_has_byte

L_missing_byte:
    // three spaces
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    strb w12, [x7, #0]
    add x7, x7, #1
    strb w12, [x7, #0]
    add x7, x7, #1

    // extra space after byte 7
    cmp x15, #7
    beq L_extra_space
    add x15, x15, #1
    b L_byte_loop

L_has_byte:
    // address of byte
    add x12, x1, x15
    ldrb w12, [x12]

    // high nibble
    lsr w13, w12, #4
    and w13, w13, #0xF
    ldrb w14, [x6, w13, lsl #2]
    strb w14, [x7, #0]
    add x7, x7, #1

    // low nibble
    and w12, w12, #0xF
    ldrb w14, [x6, w12, lsl #2]
    strb w14, [x7, #0]
    add x7, x7, #1

    // space after hex
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1

    // extra space after byte 7
    cmp x15, #7
    beq L_extra_space2
    add x15, x15, #1
    b L_byte_loop

L_extra_space:
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    add x15, x15, #1
    b L_byte_loop

L_extra_space2:
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1
    add x15, x15, #1
    b L_byte_loop

L_after_bytes:
    // space before '|'
    ldrb w12, [x11]
    strb w12, [x7, #0]
    add x7, x7, #1

    // pipe char
    adrp x12, pipe_char
    add x12, x12, #0
    ldrb w12, [x12]
    strb w12, [x7, #0]
    add x7, x7, #1

    // ascii loop
    mov x15, #0
L_ascii_loop:
    cmp x15, x14
    beq L_ascii_done
    add x12, x1, x15
    ldrb w12, [x12]
    cmp w12, #0x20
    blt L_ascii_dot
    cmp w12, #0x7e
    bgt L_ascii_dot
    strb w12, [x7, #0]
    add x7, x7, #1
    b L_ascii_next
L_ascii_dot:
    movz w12, #46
    strb w12, [x7, #0]
    add x7, x7, #1
L_ascii_next:
    add x15, x15, #1
    b L_ascii_loop
L_ascii_done:
    // pipe after ascii
    ldrb w12, [x12]
    strb w12, [x7, #0]
    add x7, x7, #1
    // newline
    adrp x12, newline_char
    add x12, x12, #0
    ldrb w12, [x12]
    strb w12, [x7, #0]
    add x7, x7, #1

    // write line
    sub x2, x7, x9
    mov x0, #1
    mov x1, x7
    mov x8, #64
    svc #0

    // update offset
    add x5, x5, x14
    str x5, [x4]

    b read_loop
