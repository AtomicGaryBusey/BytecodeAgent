.equ INPUT_BUF_SIZE, 32768
.equ MAX_LINES, 1000
.equ LINE_STRUCT_SIZE, 16
.equ LINES_SIZE, MAX_LINES * LINE_STRUCT_SIZE

.section .rodata
new_line: .byte 0x0A

.section .bss
.align 8
input_buf: .space INPUT_BUF_SIZE
.align 8
line_cnt: .space 8
.align 8
lines: .space LINES_SIZE

.section .text
.global _start
_start:
    xor r8, r8                 ; total_read = 0
read_loop:
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r8                ; remaining
    mov r9, 4096
    cmp r9, rdx
    jge .read_small
    mov rcx, 4096
    jmp .sys_read
.read_small:
    mov rcx, rdx
.sys_read:
    mov rdi, 0                 ; fd = stdin
    mov rsi, input_buf
    add rsi, r8                ; dest = input_buf + total_read
    mov rax, 0                 ; sys_read
    syscall
    test rax, rax
    jz .read_done
    add r8, rax                ; total_read += bytes_read
    jmp read_loop
.read_done:
    ; parse lines
    mov rsi, input_buf
    mov rdx, r8                ; total_len
    xor rcx, rcx               ; scan offset
    xor r9, r9                 ; line_start
parse_loop:
    cmp rcx, rdx
    jge parse_done
    mov al, byte ptr [rsi + rcx]
    test al, 0x0A
    jne scan_next
    ; newline found
    mov r10, rcx
    sub r10, r9                ; length
    mov r11, qword ptr [rel line_cnt]
    lea r12, [rel lines + r11*LINE_STRUCT_SIZE]
    mov qword ptr [r12], r9    ; offset
    mov qword ptr [r12+8], r10
    inc qword ptr [rel line_cnt]
    inc rcx
    mov r9, rcx
    jmp parse_loop
scan_next:
    inc rcx
    jmp parse_loop
parse_done:
    cmp r9, rdx
    jge no_final_line
    mov r10, rdx
    sub r10, r9                ; length
    mov r11, qword ptr [rel line_cnt]
    lea r12, [rel lines + r11*LINE_STRUCT_SIZE]
    mov qword ptr [r12], r9
    mov qword ptr [r12+8], r10
    inc qword ptr [rel line_cnt]
.no_final_line:
    ; insertion sort
    mov rbx, qword ptr [rel line_cnt]   ; n
    xor rdi, rdi                    ; i = 0
outer_loop:
    cmp rdi, rbx
    jge sort_done
    ; load key
    lea r8, [rel lines + rdi*LINE_STRUCT_SIZE]
    mov r9, qword ptr [r8]          ; key_offset
    mov r10, qword ptr [r8+8]       ; key_len
    mov rcx, rdi
    dec rcx                         ; j = i-1
inner_loop:
    test rcx, rcx
    jz inner_break
    ; load line_j
    lea r11, [rel lines + rcx*LINE_STRUCT_SIZE]
    mov r12, qword ptr [r11]       ; offset_j
    mov r13, qword ptr [r11+8]     ; len_j
    ; compute min_len
    mov r14, r10
    cmp r14, r13
    jle min_len_key
    mov r14, r13
min_len_key:
    mov r15, rcx                    ; save j
    test r14, r14
    jz strings_equal
    mov rcx, r14                    ; count for rep cmpsb
    ; set up for rep cmpsb
    mov rsi, input_buf
    add rsi, r9                     ; rsi = key start
    mov rdi, input_buf
    add rdi, r12                    ; rdi = line_j start
    rep cmpsb
    mov rcx, r15                    ; restore j
    je strings_equal
    jc inner_break                 ; key > line_j
    jmp key_less
strings_equal:
    ; compare lengths
    mov rax, r10
    cmp rax, r13
    jb key_less
    jmp inner_break
key_less:
    ; shift line_j up by one
    lea rsi, [rel lines + rdi*LINE_STRUCT_SIZE]   ; dest = i
    lea rdx, [rel lines + rcx*LINE_STRUCT_SIZE]   ; src = j
    mov rax, qword ptr [rdx]
    mov qword ptr [rsi], rax
    mov rax, qword ptr [rdx+8]
    mov qword ptr [rsi+8], rax
    dec rcx
    jmp inner_loop
inner_break:
    ; store key at position i
    lea rsi, [rel lines + rdi*LINE_STRUCT_SIZE]
    mov qword ptr [rsi], r9
    mov qword ptr [rsi+8], r10
    inc rdi
    jmp outer_loop
sort_done:
    ; output sorted lines
    mov rbx, qword ptr [rel line_cnt]
    xor rcx, rcx
output_loop:
    cmp rcx, rbx
    jge output_done
    lea r8, [rel lines + rcx*LINE_STRUCT_SIZE]
    mov r9, qword ptr [r8]        ; offset
    mov r10, qword ptr [r8+8]     ; length
    mov rdi, 1
    mov rsi, input_buf
    add rsi, r9
    mov rdx, r10
    mov rax, 1
    syscall
    mov rdi, 1
    mov rdx, 1
    mov rsi, new_line
    mov rax, 1
    syscall
    inc rcx
    jmp output_loop
output_done:
    mov rax, 60
    xor rdi, rdi
    syscall
