.equ INPUT_BUF_SIZE, 32768
.equ MAX_LINES, 1000
.equ LINE_STRUCT_SIZE, 8
.equ LINES_SIZE, MAX_LINES * LINE_STRUCT_SIZE

.section .rodata
new_line: .byte 0x0A

.section .bss
.align 8
input_buf: .space INPUT_BUF_SIZE
.align 8
line_cnt: .space 8
.align 8
lines: .space LINES_SIZE

.section .text
.globl _start
_start:
    xor r8, r8                ; total_read = 0

read_loop:
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r8               ; remaining
    mov rdi, 0                ; fd = stdin
    mov rsi, input_buf
    add rsi, r8               ; dest = input_buf + total_read
    mov rax, 0                ; sys_read
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    ; parse lines
    mov rsi, input_buf
    mov rdx, r8               ; total_len
    xor rcx, rcx              ; scan_offset = 0
    xor r9, r9                ; line_start = 0

parse_loop:
    cmp rcx, rdx
    jge parse_done
    mov al, byte ptr [rsi + rcx]
    cmp al, 0x0A
    jne scan_next
    ; newline found
    mov rax, rcx
    sub rax, r9               ; length
    mov rbx, [rel line_cnt]   ; current count
    lea rdx, [rel lines + rbx*LINE_STRUCT_SIZE]
    mov [rdx], r9             ; offset
    mov [rdx+8], rax          ; length
    inc qword ptr [rel line_cnt]
    inc rcx
    mov r9, rcx
    jmp parse_loop

scan_next:
    inc rcx
    jmp parse_loop

parse_done:
    cmp rcx, r9
    jge no_final_line
    ; final line
    mov rax, rcx
    sub rax, r9
    mov rbx, [rel line_cnt]
    lea rdx, [rel lines + rbx*LINE_STRUCT_SIZE]
    mov [rdx], r9
    mov [rdx+8], rax
    inc qword ptr [rel line_cnt]
no_final_line:

    ; insertion sort
    mov rbx, [rel line_cnt]   ; n
    xor rdi, rdi              ; i = 0

outer_loop:
    cmp rdi, rbx
    jge sort_done
    ; load key
    lea r8, [rel lines + rdi*LINE_STRUCT_SIZE]
    mov r9, [r8]              ; key_offset
    mov r10, [r8+8]           ; key_len
    ; j = i-1
    mov rcx, rdi
    dec rcx

inner_loop:
    test rcx, rcx
    jz inner_break
    ; load line j
    lea r11, [rel lines + rcx*LINE_STRUCT_SIZE]
    mov r12, [r11]            ; offset_j
    mov r13, [r11+8]          ; len_j
    ; min_len = min(key_len, len_j)
    mov r14, r10
    cmp r14, r13
    jle min_len
    mov r14, r13
min_len:
    ; compare up to min_len bytes
    xor rax, rax              ; idx = 0
    mov rsi, input_buf
    add rsi, r9               ; key_start
    mov rdx, input_buf
    add rdx, r12              ; linej_start

compare_loop:
    cmp rax, r14
    jge compare_done
    mov al, byte ptr [rsi + rax]
    mov bl, byte ptr [rdx + rax]
    cmp al, bl
    jb key_less
    ja key_greater
    inc rax
    jmp compare_loop

compare_done:
    ; strings equal up to min_len; compare lengths
    cmp r10, r13
    jl key_less
    jge key_greater

key_less:
    mov r15, rcx
    ; move line j down one
    lea rsi, [rel lines + rdi*LINE_STRUCT_SIZE]
    lea rdx, [rel lines + r15*LINE_STRUCT_SIZE]
    mov rax, [rdx]
    mov [rsi], rax
    mov rax, [rdx+8]
    mov [rsi+8], rax
    dec rcx
    jmp inner_loop

key_greater:
    ; stop shifting
    mov r15, rcx
    jmp inner_break

inner_break:
    ; insert key at position i
    lea rsi, [rel lines + rdi*LINE_STRUCT_SIZE]
    mov [rsi], r9
    mov [rsi+8], r10
    inc rdi
    jmp outer_loop

sort_done:

    ; output sorted lines
    mov rbx, [rel line_cnt]   ; number of lines
    xor r15, r15              ; index

output_loop:
    cmp r15, rbx
    jge output_done
    lea r8, [rel lines + r15*LINE_STRUCT_SIZE]
    mov r9, [r8]              ; offset
    mov r10, [r8+8]           ; length
    mov rdi, 1                ; fd = stdout
    mov rsi, input_buf
    add rsi, r9
    mov rdx, r10
    mov rax, 1                ; write
    syscall
    ; write newline
    mov rdi, 1
    mov rdx, 1
    lea rsi, [rip + new_line]
    mov rax, 1
    syscall
    inc r15
    jmp output_loop

output_done:
    mov rax, 60
    xor rdi, rdi
    syscall
