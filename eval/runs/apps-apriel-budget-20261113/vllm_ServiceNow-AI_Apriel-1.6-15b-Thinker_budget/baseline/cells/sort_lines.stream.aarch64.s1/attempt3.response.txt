.global _start
    .section .rodata
newline:
    .byte 10
    .align 4
READ:
    .equ 63
WRITE:
    .equ 64
EXIT:
    .equ 93
IN_BUF_SIZE:
    .equ 32768
MAX_LINES:
    .equ 1000
LINES_STRUCT_SIZE:
    .equ 8
LINES_BUF_SIZE:
    .equ 8000

    .section .bss
line_starts:
    .space 4000
    .align 4
line_lens:
    .space 4000
    .align 4

    .section .data
input_buf:
    .space 32768
    .align 4

    .section .text
_start:
    mov w2, #IN_BUF_SIZE
    mov w3, #0
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
read_loop:
    sub w4, w2, w3
    cbz w4, read_done
    mov w8, #READ
    mov x0, #0
    add x1, x5, x3
    mov x2, w4
    svc #0
    cmp x0, #0
    beq read_done
    add w3, w3, x0
    b read_loop

read_done:
    cbnz w3, parse_lines
    mov w0, #0
    mov w8, #EXIT
    svc #0
    b .

parse_lines:
    adrp x9, line_starts
    add x9, x9, :lo12:line_starts
    adrp x10, line_lens
    add x10, x10, :lo12:line_lens
    mov w5, #0
    mov w6, #0

parse_start:
    cmp w6, w3
    b.ge parse_end

    mov w7, w6
scan_loop:
    cmp w7, w3
    b.ge scan_no_newline
    add x0, x5, w7
    ldrb w8, [x0]
    cmp w8, #10
    beq scan_newline
    add w7, w7, #1
    b scan_loop

scan_newline:
    sub w8, w7, w6
    add x11, x9, w5, lsl #2
    str w6, [x11]
    add x12, x10, w5, lsl #2
    str w8, [x12]
    add w5, w5, #1
    add w6, w7, #1
    b parse_start

scan_no_newline:
    sub w8, w3, w6
    add x11, x9, w5, lsl #2
    str w6, [x11]
    add x12, x10, w5, lsl #2
    str w8, [x12]
    add w5, w5, #1
    mov w6, w3
    b parse_start

parse_end:
    cmp w5, #1
    ble sort_done
    mov w1, w5

    mov w9, #1
sort_outer:
    cmp w9, w5
    bge sort_done
    add x13, x9, w9, lsl #2
    ldw w10, [x13]
    add x14, x10, w9, lsl #2
    ldw w11, [x14]
    mov w8, w9

sort_inner:
    cmp w8, #1
    ble inner_break
    sub w12, w8, #1
    add x15, x9, w12, lsl #2
    ldw w13, [x15]
    add x16, x10, w12, lsl #2
    ldw w14, [x16]

    cmp w14, w11
    ble min_len_assign
    mov w2, w11
    bra min_len_set
min_len_assign:
    mov w2, w14
min_len_set:
    mov w3, #0
cmp_loop_start:
    cmp w3, w2
    b.ge cmp_done
    add x0, x5, w13
    add x0, x0, w3
    ldrb w4, [x0]
    add x1, x5, w10
    add x1, x1, w3
    ldrb w5, [x1]
    cmp w4, w5
    beq byte_equal
    cmp w4, w5
    bgt shift_continue
    b shift_false
byte_equal:
    add w3, w3, #1
    b cmp_loop_start
cmp_done:
    cmp w14, w11
    bgt shift_continue
    b shift_false

shift_continue:
    add x17, x9, w12, lsl #2
    str w10, [x17]
    add x18, x10, w12, lsl #2
    str w11, [x18]
    sub w8, w8, #1
    b sort_inner

shift_false:
    add x19, x9, w8, lsl #2
    str w10, [x19]
    add x20, x10, w8, lsl #2
    str w11, [x20]
    add w9, w9, #1
    b sort_outer

inner_break:
    b shift_false

sort_done:
    mov w5, w1
    mov w9, #0
output_loop:
    cmp w9, w5
    bge output_done
    add x21, x9, w9, lsl #2
    ldw w22, [x21]
    add x22, x10, w9, lsl #2
    ldw w23, [x22]
    cmp w23, #0
    beq write_newline_only
    mov w8, #WRITE
    mov x0, #1
    add x1, x5, w22
    mov x2, w23
    svc #0
write_newline_only:
    mov w8, #WRITE
    mov x0, #1
    adrp x23, newline
    add x23, x23, :lo12:newline
    mov x1, x23
    mov x2, #1
    svc #0
    add w9, w9, #1
    b output_loop

output_done:
    mov w0, #0
    mov w8, #EXIT
    svc #0
    b .
