.global _start
    .section .rodata
newline:
    .byte 10
    .align 4
    .equ READ, 63
    .equ WRITE, 64
    .equ EXIT, 93
    .equ IN_BUF_SIZE, 32768
    .section .bss
lines:
    .space 4000
    .align 4
    .section .data
input_buf:
    .space 32768
    .align 4
    .section .text
_start:
    /* READ INPUT */
    mov x9, #0                     // total_len = 0
    mov x10, #32768                // buffer size
read_loop:
    mov x0, #0                     // fd = 0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    sub x2, x10, x9                // count = buffer - total_len
    mov x8, #63                    // sys_read
    svc #0
    mov x9, x0                     // total_len = bytes read
    cmp x0, #0
    beq read_done
    b read_loop
read_done:
    /* PARSE LINES */
    mov w0, #0                     // line_start = 0
    mov w1, #0                     // consumed = 0
    mov w2, #0                     // line_idx = 0
    adrp x3, lines
    add x3, x3, :lo12:lines        // lines_base
    adrp x4, input_buf
    add x4, x4, :lo12:input_buf    // buffer_ptr (will be incremented)
scan_loop:
    cmp w1, x9
    beq scan_end
    ldrb w5, [x4], #1
    add w1, w1, #1
    cmp w5, #10
    b.ne scan_continue
    sub w6, w1, w0                 // length
    lsl w7, w2, #2
    add x8, x3, x7
    strh w0, [x8]
    strh w6, [x8, #2]
    add w2, w2, #1
    mov w0, w1                     // start = consumed
scan_continue:
    b scan_loop
scan_end:
    sub w6, w1, w0                 // final length
    cmp w6, #0
    beq parse_done
    lsl w7, w2, #2
    add x8, x3, x7
    strh w0, [x8]
    strh w6, [x8, #2]
parse_done:
    mov x0, w2                     // line count in x0
    cmp w0, #2
    blt output_start
    /* BUBBLE SORT */
    sub w5, w0, #2                 // inner_limit = n-2
    mov x1, #0                     // pass counter
pass_loop:
    cmp x1, w0
    beq pass_done
    mov x2, #0                     // j
inner_loop:
    cmp x2, w5
    b.gt inner_done
    add x7, x3, x2, lsl #2
    ldrh w8, [x7]                  // start_j
    ldrh w9, [x7, #2]              // len_j
    add x6, x2, #1
    add x11, x3, x6, lsl #2
    ldrh w12, [x11]                // start_j1
    ldrh w13, [x11, #2]            // len_j1
    /* compare lines */
    cmp w9, w13
    b.le min_len_eq
    mov w14, w13                   // min_len = len_j1
    b min_len_set
min_len_eq:
    mov w14, w9                    // min_len = len_j
min_len_set:
    mov w15, w14                   // loop counter = min_len
    /* pointers to line data */
    adrp x12, input_buf
    add x12, x12, :lo12:input_buf
    mov x13, w8
    add x12, x12, x13
    adrp x13, input_buf
    add x13, x13, :lo12:input_buf
    mov x14, w12
    add x13, x13, x14
compare_loop:
    cbnz w15, compare_body
    /* all min_len bytes equal, compare lengths */
    cmp w9, w13
    beq cmp_equal_len
    cmp w9, w13
    b.lt cmp_len_lt
    b cmp_len_gt
cmp_equal_len:
    movi w0, #0
    b after_compare
cmp_len_lt:
    movi w0, #-1
    b after_compare
cmp_len_gt:
    movi w0, #1
    b after_compare
compare_body:
    ldrb w16, [x12], #1
    ldrb w17, [x13], #1
    cmp w16, w17
    b.ne diff_found
    sub w15, w15, #1
    b compare_loop
diff_found:
    cmp w16, w17
    b.lt cmp_lt
    b cmp_gt
cmp_lt:
    movi w0, #-1
    b after_compare
cmp_gt:
    movi w0, #1
    b after_compare
after_compare:
    cmp w0, #1
    b.eq swap_needed
    add x2, x2, #1
    b inner_loop
swap_needed:
    strh w12, [x7]
    strh w13, [x7, #2]
    strh w8, [x11]
    strh w9, [x11, #2]
    add x2, x2, #1
    b inner_loop
inner_done:
    add x1, x1, #1
    b pass_loop
pass_done:
output_start:
    mov x2, #0                     // index for output
output_loop:
    cmp x2, w0
    beq output_done
    add x7, x3, x2, lsl #2
    ldrh w8, [x7]                  // start
    ldrh w9, [x7, #2]              // length
    cmp w9, #0
    beq write_newline
    adrp x5, input_buf
    add x5, x5, :lo12:input_buf
    mov x6, w8
    add x5, x5, x6
    mov x0, #1
    mov x1, x5
    mov x2, w9
    mov x8, #64
    svc #0
write_newline:
    adrp x5, newline
    add x5, x5, :lo12:newline
    mov x0, #1
    mov x1, x5
    mov x2, #1
    mov x8, #64
    svc #0
    add x2, x2, #1
    b output_loop
output_done:
    mov x0, #0
    mov x8, #93
    svc #0
