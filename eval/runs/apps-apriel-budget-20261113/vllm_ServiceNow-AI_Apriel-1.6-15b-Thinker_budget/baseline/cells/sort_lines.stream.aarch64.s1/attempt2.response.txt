.global _start
    .section .rodata
newline:
    .byte 10
    .align 4
    .equ READ, 63
    .equ WRITE, 64
    .equ EXIT, 93
    .equ IN_BUF_SIZE, 32768
    .equ LINES_BUF_SIZE, 8000

    .section .bss
lines:
    .space LINES_BUF_SIZE
    .align 4

    .section .data
input_buf:
    .space IN_BUF_SIZE
    .align 4

    .section .text
_start:
    // read input
    mov w9, #0          // total_len
    mov w10, #IN_BUF_SIZE
read_loop:
    mov x0, #0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    sub x2, x10, w9
    mov x8, #READ
    svc #0
    mov w9, x0
    cmp x0, #0
    beq read_done
    b read_loop
read_done:
    // parse lines
    mov w0, #0          // line_cnt
    mov w1, #0          // line_start
    mov w2, #0          // i
    adrp x3, lines
    add x3, x3, :lo12:lines
    adrp x4, input_buf
    add x4, x4, :lo12:input_buf
scan_loop:
    cmp w2, w9
    beq scan_done
    ldrb w5, [x4], #1
    add w2, w2, #1
    cmp w5, #10
    b .ne scan_loop
    // newline found: length = i - start - 1
    sub w4, w2, w1
    sub w4, w4, #1
    // store descriptor
    add x5, x3, w0, LSL #3
    str w1, [x5]
    str w4, [x5, #4]
    add w0, w0, #1
    // next start is current i
    mov w1, w2
    b scan_loop
scan_done:
    // final line if start < total_len
    cmp w1, w9
    b .ge no_final_line
    sub w4, w9, w1
    add x5, x3, w0, LSL #3
    str w1, [x5]
    str w4, [x5, #4]
    add w0, w0, #1
no_final_line:
    // bubble sort
    mov w1, #0          // pass index
bubble_pass:
    cmp w1, w0
    b .ge bubble_done
    add w2, w1, #1
bubble_inner:
    cmp w2, w0
    b .lt bubble_compare
    b bubble_pass_next
bubble_compare:
    sub w3, w2, #1
    add x5, x3, w3, LSL #3
    ldr w6, [x5]
    ldr w7, [x5, #4]
    add x5, x5, #8
    ldr w8, [x5]
    ldr w9, [x5, #4]
    mov x0, w6
    mov x1, w7
    mov x2, w8
    mov x3, w9
    bl cmp_line
    cmp x0, #1
    b .eq bubble_swap
    b bubble_inner_inc
bubble_swap:
    sub x5, x5, #8
    str w8, [x5]
    str w9, [x5, #4]
    add x5, x5, #8
    str w6, [x5]
    str w7, [x5, #4]
bubble_inner_inc:
    add w2, w2, #1
    b bubble_inner
bubble_pass_next:
    add w1, w1, #1
    b bubble_pass
bubble_done:
    // output
    mov w1, #0          // line index
write_loop:
    cmp w1, w0
    beq write_done
    add x5, x3, w1, LSL #3
    ldr w6, [x5]       // start
    ldr w7, [x5, #4]   // len
    cmp w7, #0
    b .eq write_newline
    mov x0, #1
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x1, x1, w6
    mov x2, w7
    mov x8, #WRITE
    svc #0
write_newline:
    mov x0, #1
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x2, #1
    mov x8, #WRITE
    svc #0
    add w1, w1, #1
    b write_loop
write_done:
    mov x0, #0
    mov x8, #EXIT
    svc #0

    // cmp_line function
cmp_line:
    // x0=start1, x1=len1, x2=start2, x3=len2
    mov w4, x1
    mov w5, x3
    cmp w4, w5
    b .lt min_len
    mov w6, w5
    b min_len_end
min_len:
    mov w6, w4
min_len_end:
    // pointers to strings
    adrp x7, input_buf
    add x7, x7, :lo12:input_buf
    add x7, x7, x0
    adrp x8, input_buf
    add x8, x8, :lo12:input_buf
    add x8, x8, x2
    mov x9, w6
compare_loop:
    cmp x9, #0
    beq compare_done_bytes
    ldrb w10, [x7], #1
    ldrb w11, [x8], #1
    cmp w10, w11
    b .le compare_loop
    mov x0, #1
    ret
compare_done_bytes:
    cmp w4, w5
    beq compare_equal_len
    // lengths differ
    cmp w4, w5
    b .lt cmp_len1_lt_len2
    mov x0, #1
    ret
compare_equal_len:
    mov x0, #0
    ret
cmp_len1_lt_len2:
    mov x0, #0
    ret
