.global _start
    .section .rodata
newline:
    .byte 10
    .align 4
    .section .bss
input_buf:
    .space 32768
    .align 4
line_starts:
    .space 4000
    .align 4
line_lens:
    .space 4000
    .align 4
line_count:
    .word 0
    .section .text
    .global _start
_start:
    adrp x13, line_starts
    add x13, x13, :lo12:line_starts
    adrp x14, line_lens
    add x14, x14, :lo12:line_lens
    adrp x15, input_buf
    add x15, x15, :lo12:input_buf
    mov x5, #0          // total_len
    mov x1, x15         // next pointer
read_loop:
    mov x8, #63
    mov x0, #0
    mov x1, x1
    mov x2, #32768
    svc #0
    cmp x0, #0
    beq read_done
    add x5, x5, x0
    add x1, x1, x0
    b read_loop
read_done:
    mov x0, #0          // count
    mov x1, #0          // start
    mov x2, #0          // i
parse_loop:
    cmp x2, x5
    bge parse_done
    ldrb w3, [x15, x2]
    cmp w3, #10
    bne not_newline
    sub x4, x2, x1
    lsl x6, x0, #2
    add x7, x13, x6
    str x1, [x7]
    add x8, x14, x6
    str x4, [x8]
    add x0, x0, #1
    add x2, x2, #1
    mov x1, x2
    b parse_loop
not_newline:
    add x2, x2, #1
    b parse_loop
parse_done:
    cmp x2, x5
    bge parse_finish
    sub x4, x5, x1
    lsl x6, x0, #2
    add x7, x13, x6
    str x1, [x7]
    add x8, x14, x6
    str x4, [x8]
    add x0, x0, #1
parse_finish:
    adrp x16, line_count
    add x16, x16, :lo12:line_count
    str x0, [x16]
    ldr x0, [x16]                     // count
    cmp x0, #1
    ble sorted_done
    mov x2, x0                         // copy count
    sub x3, x2, #1                     // max index
    mov x1, #0                         // outer i
outer_loop:
    cmp x1, x3
    bge sorted_done
    mov x4, #0                         // inner j
inner_loop:
    cmp x4, x3
    bge inner_done
    lsl x5, x4, #2
    add x6, x13, x5
    ldr x7, [x6]                       // start_j
    add x8, x14, x5
    ldr x9, [x8]                       // len_j
    add x5, x4, #1
    lsl x5, x5, #2
    add x10, x13, x5
    ldr x11, [x10]                     // start_jp1
    add x12, x14, x5
    ldr x13, [x12]                     // len_jp1
    mov x0, x7
    mov x1, x9
    mov x2, x11
    mov x3, x13
    bl cmp_lines
    cmp x0, #0
    ble inner_no_swap
    str x11, [x6]
    str x13, [x8]
    str x7, [x10]
    str x9, [x12]
    b inner_loop
inner_no_swap:
    add x4, x4, #1
    b inner_loop
inner_done:
    add x1, x1, #1
    b outer_loop
sorted_done:
    adrp x16, line_count
    add x16, x16, :lo12:line_count
    ldr x0, [x16]                     // count
    mov x1, #0
output_loop:
    cmp x1, x0
    bge output_done
    lsl x2, x1, #2
    add x3, x13, x2
    ldr x4, [x3]                      // start
    add x5, x14, x2
    ldr x6, [x5]                      // len
    mov x8, #64
    mov x0, #1
    mov x1, x4
    mov x2, x6
    svc #0
    adrp x7, newline
    add x7, x7, :lo12:newline
    mov x8, #64
    mov x0, #1
    mov x1, x7
    mov x2, #1
    svc #0
    add x1, x1, #1
    b output_loop
output_done:
    mov x8, #93
    mov x0, #0
    svc #0
    .global cmp_lines
cmp_lines:
    // args: start1=x0, len1=x1, start2=x2, len2=x3
    mov x4, x1
    mov x5, x3
    cmp x4, x5
    ble min_len
    mov x4, x5
min_len:
    mov x6, #0          // offset
compare_loop:
    cmp x6, x4
    bge compare_done_min
    ldrb w7, [x0, x6]
    ldrb w8, [x2, x6]
    cmp w7, w8
    bne compare_diff
    add x6, x6, #1
    b compare_loop
compare_done_min:
    cmp x1, x3
    beq equal_len
    cmp x1, x3
    blt less_len
    mov x0, #1
    ret
equal_len:
    mov x0, #0
    ret
compare_diff:
    cmp w7, w8
    blt less_bytes
    mov x0, #1
    ret
less_bytes:
    mov x0, #-1
    ret
less_len:
    mov x0, #-1
    ret
