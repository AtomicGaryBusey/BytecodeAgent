.section .bss
.align 4
buffer: .space 32768
used: .quad 0
line_start: .space 1000*4
line_len: .space 1000*4
line_cnt: .quad 0

.section .rodata
newline: .byte 10

.section .text
.global _start
.align 4

_start:
    mov rax, 0          ; sys_read
read_loop:
    mov rdi, 0
    mov rsi, buffer
    mov rcx, [used]
    add rsi, rcx
    mov rdx, 32768
    sub rdx, rcx
    syscall
    test rax, rax
    jz read_done
    mov rcx, rax
    add [used], rcx
    jmp read_loop

read_done:
    mov rax, [used]     ; total bytes
    xor rdx, rdx        ; start offset = 0
    mov [line_cnt], rdx

parse_loop:
    cmp rdx, rax
    jge parse_done

    mov rcx, rdx        ; scan offset
search_newline:
    cmp rcx, rax
    jge newline_not_found
    movzx r8d, byte ptr [buffer + rcx]
    cmp r8d, 10
    jne next_scan
    jmp newline_found
next_scan:
    inc rcx
    jmp search_newline

newline_not_found:
    mov r8, rax
    sub r8, rdx
    mov r9d, r8d
    mov r10, [line_cnt]
    mov [line_start + r10*4], rdx
    mov [line_len + r10*4], r9d
    inc [line_cnt]
    jmp parse_done

newline_found:
    mov r8, rcx
    sub r8, rdx
    mov r9d, r8d
    mov r10, [line_cnt]
    mov [line_start + r10*4], rdx
    mov [line_len + r10*4], r9d
    inc [line_cnt]
    mov rdx, rcx
    inc rdx
    jmp parse_loop

parse_done:
    mov rax, [line_cnt]      ; count
    xor rbx, rbx             ; i = 0

sel_outer:
    cmp rbx, rax
    jge sel_done
    mov rcx, rbx
    inc rcx
    mov rdx, rcx
    inc rdx
inner_loop:
    cmp rdx, rax
    jge inner_done
    mov r8, rdx
    mov r9, rcx
    mov rdi, r8
    mov rsi, r9
    call cmp_line
    jl min_found
    inc rdx
    jmp inner_loop
min_found:
    mov rcx, r8
    inc rdx
    jmp inner_loop
inner_done:
    mov r10, [line_start + rbx*4]
    mov r11, [line_start + rcx*4]
    mov [line_start + rbx*4], r11
    mov [line_start + rcx*4], r10
    mov r10, [line_len + rbx*4]
    mov r11, [line_len + rcx*4]
    mov [line_len + rbx*4], r11
    mov [line_len + rcx*4], r10
    inc rbx
    jmp sel_outer
sel_done:
    mov r12, [line_cnt]    ; count
    xor r14, r14           ; index = 0

output_loop:
    cmp r14, r12
    jge output_done

    mov r8, [line_start + r14*4]
    mov r9d, [line_len + r14*4]
    xor r10d, r10d

write_bytes:
    cmp r10d, r9d
    jge write_newline
    mov rsi, r8
    add rsi, r10d
    mov rdi, 1
    mov rdx, 1
    mov rax, 1
    syscall
    inc r10d
    jmp write_bytes
write_newline:
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r14
    jmp output_loop

output_done:
    mov rax, 60
    xor rdi, rdi
    syscall

cmp_line:
    mov rax, [line_start + rdi*4]
    mov rbx, [line_len + rdi*4]
    mov rcx, [line_start + rsi*4]
    mov rdx, [line_len + rsi*4]

    mov r8d, rbx
    mov r9d, rdx
    cmp r8d, r9d
    jge min_len_adj
    mov r9d, r8d
min_len_adj:
    xor r10d, r10d
byte_loop:
    movzx r11d, byte ptr [buffer + rax + r10d]
    movzx r12d, byte ptr [buffer + rcx + r10d]
    sub r11d, r12d
    test r11d, r11d
    jz byte_next
    mov eax, r11d
    ret
byte_next:
    inc r10d
    cmp r10d, r9d
    jl byte_loop
    cmp rbx, rdx
    jl len_i_lt
    jg len_i_gt
    xor eax, eax
    ret
len_i_lt:
    mov eax, -1
    ret
len_i_gt:
    mov eax, 1
    ret
