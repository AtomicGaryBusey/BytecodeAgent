.global _start
.section .text
_start:
    xor eax,eax
    mov dword ptr [used],eax
    mov dword ptr [cur_line_start],eax
    mov dword ptr [line_cnt],eax
read_loop:
    mov eax,[used]
    mov ebx,32768
    sub ebx,eax
    mov eax,0
    mov edi,0
    mov esi,buffer
    mov edx,ebx
    syscall
    test eax,eax
    jz read_done
    mov ecx,[used]
    add ecx,eax
    mov [used],ecx
    mov edx,[used]
    sub edx,eax
    mov rcx,eax
    mov r8,[cur_line_start]
process_bytes_loop:
    test rcx,rcx
    jz bytes_done
    mov al,byte ptr [buffer+rdx]
    cmp al,0x0A
    jne not_newline
    mov r9,rdx
    sub r9,r8
    mov edi,[line_cnt]
    lea rsi,[offset_arr+rdi*4]
    mov dword [rsi],r8d
    lea rsi,[len_arr+ rdi*4]
    mov dword [rsi],r9d
    inc dword ptr [line_cnt]
    inc rdx
    mov [cur_line_start],rdx
    dec rcx
    jmp process_bytes_loop
not_newline:
    inc rdx
    dec rcx
    jmp process_bytes_loop
bytes_done:
    jmp read_loop
read_done:
    mov rax,[cur_line_start]
    mov rbx,[used]
    cmp rax,rbx
    jge no_final_line
    mov rcx,rbx
    sub rcx,rax
    mov edi,[line_cnt]
    lea rsi,[offset_arr+ rdi*4]
    mov dword [rsi],rax
    lea rsi,[len_arr+ rdi*4]
    mov dword [rsi],rcx
    inc dword ptr [line_cnt]
no_final_line:
    mov eax,[line_cnt]
    mov ebx,eax
    mov esi,1
    cmp esi,ebx
    jge sort_done
sort_outer:
    mov edi,esi
    lea r8,[offset_arr+rdi*4]
    mov r8d,dword [r8]
    lea r9,[len_arr+rdi*4]
    mov r9d,dword [r9]
    mov edx,esi
    sub edx,1
sort_inner:
    cmp edx,0
    jle sort_inner_break
    mov edi,edx
    lea r10,[offset_arr+rdi*4]
    mov r10d,dword [r10]
    lea r11,[len_arr+rdi*4]
    mov r11d,dword [r11]
    mov r12,r11
    mov r13,r9
    mov r14,r10
    mov r15,r8
    mov rbx,r11
    mov rcx,r9
compare_loop:
    test rbx,rbx
    jz compare_len
    test rcx,rcx
    jz compare_len
    mov al,byte ptr [r14]
    mov bl,byte ptr [r15]
    cmp al,bl
    jne compare_result
    inc r14
    inc r15
    dec rbx
    dec rcx
    jmp compare_loop
compare_len:
    mov rbx,r12
    mov rcx,r13
    cmp rbx,rcx
    jl compare_result_minus
    jg compare_result_plus
    jmp compare_result_zero
compare_result:
    cmp al,bl
    jl compare_result_minus
    jg compare_result_plus
    jmp compare_result_zero
compare_result_minus:
    mov eax,-1
    jmp compare_done
compare_result_plus:
    mov eax,1
    jmp compare_done
compare_result_zero:
    xor eax,eax
    jmp compare_done
compare_done:
    cmp eax,1
    jne sort_inner_break
    mov edi,esi
    lea rsi,[offset_arr+rdi*4]
    mov dword [rsi],r10d
    lea rsi,[len_arr+rdi*4]
    mov dword [rsi],r11d
    mov edi,edx
    lea rsi,[offset_arr+rdi*4]
    mov dword [rsi],r8d
    lea rsi,[len_arr+rdi*4]
    mov dword [rsi],r9d
    dec edx
    jmp sort_inner
sort_inner_break:
    mov edi,edx
    lea rsi,[offset_arr+rdi*4]
    mov dword [rsi],r8d
    lea rsi,[len_arr+rdi*4]
    mov dword [rsi],r9d
    inc esi
    cmp esi,ebx
    jl sort_outer
sort_done:
    mov eax,[line_cnt]
    mov ebx,eax
    mov esi,0
    cmp esi,ebx
    jge output_done
output_loop:
    mov edi,esi
    lea r8,[offset_arr+rdi*4]
    mov r8d,dword [r8]
    lea r9,[len_arr+rdi*4]
    mov r9d,dword [r9]
    mov eax,1
    mov edi,1
    lea rsi,[buffer+r8]
    mov rdx,r9d
    syscall
    lea rsi,[rip+newline]
    mov eax,1
    mov edi,1
    mov rdx,1
    syscall
    inc esi
    cmp esi,ebx
    jl output_loop
output_done:
    mov eax,60
    xor edi,edi
    syscall
.section .rodata
newline: .byte 0x0A
.section .bss
.align 4
buffer: .space 32768
used: .space 4
cur_line_start: .space 4
line_cnt: .space 4
offset_arr: .space 4000
len_arr: .space 4000
