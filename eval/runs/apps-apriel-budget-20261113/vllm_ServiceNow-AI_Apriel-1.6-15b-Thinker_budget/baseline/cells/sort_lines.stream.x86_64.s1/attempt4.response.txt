.globl _start
.section .rodata
newline: .byte 0x0A

.section .bss
.align 8
buffer: .space 32768
linestrt: .space 4000
linelen: .space 2000

.section .data
linecnt: .zero 4

.section .text
.align 16
_start:
    lea rbx, [rel buffer]        ; buffer base
    xor r8, r8                   ; total_bytes = 0

read_loop:
    mov rax, 0                   ; sys_read
    mov rdi, 0                   ; fd 0
    mov rsi, rbx
    add rsi, r8
    mov rdx, 32768
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    mov rcx, r8                  ; total_bytes
    xor r8, r8                   ; linecnt = 0
    xor r9, r9                   ; start offset

parse_loop:
    cmp r9, rcx
    jge parse_done

    mov r10, r9
find_nl:
    cmp r10, rcx
    jge no_nl
    movzx r11, byte ptr [rbx + r10]
    cmp r11, 0x0A
    jne inc_nl
    mov eax, r10
    sub eax, r9
    lea rdx, [linestrt + r8*4]
    mov dword ptr [rdx], r9
    lea rdx, [linelen + r8*2]
    mov word ptr [rdx], ax
    inc r8
    inc r10
    mov r9, r10
    jmp parse_loop
inc_nl:
    inc r10
    jmp find_nl

no_nl:
    cmp r9, rcx
    jne final_line_store
    jmp parse_done

final_line_store:
    mov eax, rcx
    sub eax, r9
    lea rdx, [linestrt + r8*4]
    mov dword ptr [rdx], r9
    lea rdx, [linelen + r8*2]
    mov word ptr [rdx], ax
    inc r8

parse_done:
    mov dword ptr [linecnt], r8

    mov r11, r8                 ; linecnt for sorting

    mov r9, 1                   ; i = 1

sort_loop:
    cmp r9, r11
    jge sort_done

    mov r12, r9                 ; save i

    mov eax, dword ptr [linestrt + r9*4]   ; key offset
    mov edx, word ptr [linelen + r9*2]    ; key length
    mov r15, edx               ; save key length

    mov r10, r12
    dec r10                     ; j = i-1

inner_loop:
    cmp r10, 0
    jle inner_done

    mov ecx, dword ptr [linestrt + r10*4]  ; offset_j
    mov ebp, word ptr [linelen + r10*2]   ; len_j

    mov r13, rcx               ; save offset_j
    mov r14, ebp               ; save len_j

    mov rdi, rcx               ; offset_a = offset_j
    mov rsi, rax               ; offset_b = key offset
    mov rdx, ebp               ; len_a = len_j
    mov rcx, r15               ; len_b = key length
    jmp compare_lines

inner_done:
    mov rdx, r10
    inc rdx
    mov dword ptr [linestrt + rdx*4], eax
    mov word ptr [linelen + rdx*2], r15w
    mov r9, r12                ; restore i
    inc r9
    jmp sort_loop

compare_lines:
    ; inputs: rdi = offset_a, rsi = offset_b, rdx = len_a, rcx = len_b
    ; output: al = 1 if line_a > line_b else 0
    mov r8, rdx
    mov r9, rcx
    mov r10, rdi
    mov r11, rsi

compare_loop:
    test r8, r8
    jz compare_a_done
    test r9, r9
    jz compare_b_done

    movzx eax, byte ptr [rbx + r10]
    movzx edx, byte ptr [rbx + r11]
    cmp eax, edx
    jne compare_result

    inc r10
    inc r11
    dec r8
    dec r9
    jmp compare_loop

compare_a_done:
    xor eax, eax
    jmp compare_done

compare_b_done:
    mov al, 1
    jmp compare_done

compare_result:
    setg al
compare_done:

    test al, al
    jz inner_done

    mov rdx, r10
    inc rdx
    mov dword ptr [linestrt + rdx*4], r13
    mov rdx, r10
    inc rdx
    mov word ptr [linelen + rdx*2], r14w

    dec r10
    jmp inner_loop

sort_done:
    mov rdi, 1                  ; stdout
    xor r9, r9                  ; index = 0

output_loop:
    cmp r9, r11
    jge output_done

    mov eax, dword ptr [linestrt + r9*4]
    mov edx, word ptr [linelen + r9*2]

    mov rsi, rbx
    add rsi, rax
    mov rdx, edx
    mov rax, 1
    syscall

    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall

    inc r9
    jmp output_loop

output_done:
    mov rax, 60
    xor rdi, rdi
    syscall
