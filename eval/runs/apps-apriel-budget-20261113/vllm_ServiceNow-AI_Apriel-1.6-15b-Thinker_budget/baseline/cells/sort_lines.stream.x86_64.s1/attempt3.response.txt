.globl _start
.section .rodata
newline: .byte 0x0A
.section .bss
.align 8
buffer: .space 32768
line_start: .space 4000
line_len: .space 4000
line_cnt: .space 4
.section .text
.align 16
_start:
    lea rbx, [rip + buffer]
    xor r8, r8
read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    add rsi, r8
    mov rdx, 32768
    sub rdx, r8
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop
read_done:
    mov rcx, r8
    xor r8, r8
    xor r9, r9
parse_loop:
    cmp r9, rcx
    jge parse_done
    mov r10, r9
find_nl:
    cmp r10, rcx
    jge no_nl
    movzx r11, byte ptr [r10]
    cmp r11, 0x0A
    jne inc_scan
    jmp nl_found
inc_scan:
    inc r10
    jmp find_nl
no_nl:
    mov r11, rcx
    sub r11, r9
    mov r12, rcx
    jmp after_find
nl_found:
    mov r11, r10
    sub r11, r9
    inc r10
    mov r12, r10
after_find:
    lea rdx, [rel line_start + r8*4]
    mov dword ptr [rdx], r9
    lea rdx, [rel line_len + r8*4]
    mov dword ptr [rdx], r11
    inc r8
    mov r9, r12
    cmp r8, 1000
    jge parse_done
    jmp parse_loop
parse_done:
    lea rdx, [rel line_cnt]
    mov dword ptr [rdx], r8
    mov r8, dword ptr [rel line_cnt]
    test r8, r8
    jle sort_end
    mov r9, 1
sort_outer:
    cmp r9, r8
    jge sort_end
    mov r10, r9
    mov r11, r9
    dec r11
inner_loop:
    test r11, r11
    jz inner_done
    lea rdx, [rel line_start + r10*4]
    mov eax, dword ptr [rdx]
    lea rdx, [rel line_len + r10*4]
    mov ecx, dword ptr [rdx]
    lea rdx, [rel line_start + r11*4]
    mov edx, dword ptr [rdx]
    lea rdx, [rel line_len + r11*4]
    mov esi, dword ptr [rdx]
    xor r14, r14
compare_bytes:
    cmp r14, ecx
    jge compare_len
    cmp r14, esi
    jge compare_len
    mov dl, byte ptr [eax + r14]
    mov bl, byte ptr [edx + r14]
    cmp dl, bl
    jl line_j_smaller
    jg line_j_greater
    inc r14
    jmp compare_bytes
compare_len:
    cmp ecx, esi
    jl line_j_smaller
    jg line_j_greater
    jmp line_j_greater
line_j_smaller:
    mov r12, eax
    mov r13, ecx
    lea rdx, [rel line_start + r10*4]
    mov dword ptr [rdx], edx
    lea rdx, [rel line_len + r10*4]
    mov dword ptr [rdx], esi
    lea rdx, [rel line_start + r11*4]
    mov dword ptr [rdx], r12
    lea rdx, [rel line_len + r11*4]
    mov dword ptr [rdx], r13
    dec r10
    mov r11, r10
    dec r11
    jmp inner_loop
line_j_greater:
    jmp inner_done
inner_done:
    inc r9
    jmp sort_outer
sort_end:
    mov r8, dword ptr [rel line_cnt]
    xor r9, r9
output_loop:
    cmp r9, r8
    jge output_done
    lea rdx, [rel line_start + r9*4]
    mov eax, dword ptr [rdx]
    lea rdx, [rel line_len + r9*4]
    mov ecx, dword ptr [rdx]
    mov rdi, 1
    mov rsi, rbx
    add rsi, rax
    mov rdx, rcx
    mov rax, 1
    syscall
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    inc r9
    jmp output_loop
output_done:
    mov rdi, 0
    mov rax, 60
    syscall
