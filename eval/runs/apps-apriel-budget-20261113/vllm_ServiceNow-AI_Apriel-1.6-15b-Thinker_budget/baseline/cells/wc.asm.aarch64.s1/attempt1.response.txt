.text
.global _start
_start:
    adrp x19, BufSize
    add x19, x19, :lo12:BufSize
    ldr x19, [x19]      // x19 = 65536

    mov x20, #0          // total_len = 0
    mov x21, #0          // buf_ptr = 0

read_loop:
    sub x22, x19, x20    // remaining = BufSize - total_len
    mov x0, #0           // fd
    mov x1, x21          // buffer address
    mov x2, x22          // count
    msr x8, #63          // read
    svc #0
    cmp x0, #0
    beq read_done

    add x20, x20, x0     // total_len += n
    add x21, x21, x0     // buf_ptr += n
    b read_loop

read_done:
    adrp x23, buf
    add x23, x23, :lo12:buf
    mov x24, x20         // remaining bytes to scan
    mov x25, #0          // lines
    mov x26, #0          // words
    mov x27, #0          // in_word flag

scan_loop:
    cmp x24, #0
    beq scan_done

    ldrb w28, [x23]
    add x23, x23, #1
    sub x24, x24, #1

    cmp w28, #0x0A
    b.ne .not_nl
    add x25, x25, #1       // lines++
    mov x27, #0           // in_word = 0
    b .scan_cont

.not_nl:
    cmp w28, #0x20
    beq .ws
    cmp w28, #0x09
    beq .ws
    cmp w28, #0x0B
    beq .ws
    cmp w28, #0x0C
    beq .ws
    cmp w28, #0x0D
    beq .ws

    cbz x27, .start_word
    b .scan_cont

.start_word:
    add x26, x26, #1       // words++
    mov x27, #1

.scan_cont:
    b scan_loop

.ws:
    mov x27, #0

scan_done:
    mov x21, x25          // x21 = lines
    mov x22, x26          // x22 = words
    // x20 = bytes

    adrp x23, out_buf
    add x23, x23, :lo12:out_buf   // out_buf_start
    add x24, x23, #32             // out_buf_end

    mov x25, x24                  // outp = end

    sub x25, x25, #1
    strb 10, [x25]

    mov x0, x20
    adrp x1, tmp_buf
    add x1, x1, :lo12:tmp_buf
    mov x2, x25
    bl int_to_str
    mov x25, x2

    sub x25, x25, #1
    strb 32, [x25]

    mov x0, x22
    mov x2, x25
    bl int_to_str
    mov x25, x2

    sub x25, x25, #1
    strb 32, [x25]

    mov x0, x21
    mov x2, x25
    bl int_to_str
    mov x25, x2

    sub x3, x24, x25

    mov x0, #1
    mov x1, x23
    mov x2, x3
    msr x8, #64
    svc #0

    mov x0, #0
    msr x8, #93
    svc #0

int_to_str:
    mov x3, #0
    cbnz x0, .div_loop
    strb 48, [x1, x3]
    add x3, x3, #1
    b .digit_copy

.div_loop:
    udiv x4, x0, #10
    add w5, x0, #48
    strb w5, [x1, x3]
    add x3, x3, #1
    mov x0, x4
    cbnz x0, .div_loop

.digit_copy:
    sub x3, x3, #1
.copy_loop:
    cmp x3, #0
    beq .copy_done
    sub x2, x2, #1
    ldrb w5, [x1, x3]
    strb w5, [x2]
    sub x3, x3, #1
    b .copy_loop
.copy_done:
    ret

    .section .rodata
BufSize:
    .int 65536

    .section .bss
buf:
    .space 65536
out_buf:
    .space 32
tmp_buf:
    .space 6
