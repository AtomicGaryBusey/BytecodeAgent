.text
.global _start

    adrp x21, buf
    add x21, x21, :lo12:buf

    mov x0, #0
    mov x1, x21
    mov x2, #65536
    mov x8, #63

    mov x24, x20
    mov x25, #0
    mov x26, #0

scan_loop:
    cmp x24, #0
    beq scan_done

    ldrb w28, [x23]
    add x23, x23, #1
    sub x24, x24, #1

    cmp w28, #0x0A
    beq .nl

    cmp w28, #0x20
    beq .ws
    cmp w28, #0x09
    beq .ws
    cmp w28, #0x0B
    beq .ws
    cmp w28, #0x0C
    beq .ws
    cmp w28, #0x0D
    beq .ws

    cbz x27, .start_word
    b .scan_cont

.start_word:
    add x26, x26, #1
    mov x27, #1

.scan_cont:
    b scan_loop

.nl:
    add x25, x25, #1
    mov x27, #0

.ws:
    mov x27, #0

    adrp x9, out_buf
    add x9, x9, :lo12:out_buf
    mov x28, x9

    mov x0, x25
    adrp x1, tmp_buf
    add x1, x1, :lo12:tmp_buf
    bl int_to_str
    add x2, x2, #1
    mov x3, x0
copy_loop:
    cmp x3, #0
    beq copy_loop_end
    ldrb w4, [x2]
    strb w4, [x28]
    add x2, x2, #1
    add x28, x28, #1
    sub x3, x3, #1
    b copy_loop
copy_loop_end:
    mov w5, #32
    strb w5, [x28]
    add x28, x28, #1

    mov x0, x26
    bl int_to_str
    add x2, x2, #1
    mov x3, x0
    cmp x3, #0
    beq copy_loop_end2
copy_loop2:
    ldrb w4, [x2]
    strb w4, [x28]
    add x2, x2, #1
    add x28, x28, #1
    sub x3, x3, #1
    b copy_loop2
copy_loop_end2:
    mov w5, #32
    strb w5, [x28]
    add x28, x28, #1

    mov x0, x20
    bl int_to_str
    add x2, x2, #1
    mov x3, x0
    cmp x3, #0
    beq copy_loop_end3
copy_loop3:
    ldrb w4, [x2]
    strb w4, [x28]
    add x2, x2, #1
    add x28, x28, #1
    sub x3, x3, #1
    b copy_loop3

    strb w5, [x28]
    add x28, x28, #1

    sub x2, x28, x9

    mov x0, #1
    mov x1, x9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

int_to_str:
    adrp x2, tmp_buf_end
    add x2, x2, :lo12:tmp_buf_end
    sub x2, x2, #1
    mov x3, #0
    movz x4, #10

    cbnz x0, .outer_loop

    add w5, w0, #48
    strb w5, [x2]
    sub x2, x2, #1
    add x3, x3, #1
    mov x0, #1
    ret

.outer_loop:
    mov x6, #0
    mov x7, x0
.inner_loop:
    cmp x7, x4
    blt .inner_done
    sub x7, x7, x4
    add x6, x6, #1
    b .inner_loop

    strb w5, [x2]
    sub x2, x2, #1
    add x3, x3, #1
    mov x0, x6
    cbnz x0, .outer_loop
    mov x0, x3
    ret
