.text
.global _start
_start:
    adrp x3, buf
    add x3, x3, #:lo12:buf
    mov x1, #0          // bytes_read
    mov x2, #65536      // read size
read_loop:
    mov x8, #63         // sys_read
    mov x0, #0          // fd 0
    lsl x4, x1, #3
    add x5, x3, x4
    mov x2, #65536
    svc #0
    mov x6, x0
    cmp x6, #0
    beq read_done
    add x1, x1, x6
    b read_loop
read_done:
    mov x3, x3          // pointer to buffer start
    mov x4, x1          // remaining bytes
    mov x5, #0          // lines count
    mov x6, #0          // words count
    mov x7, #0          // in_word flag
scan_loop:
    cmp x4, #0
    beq scan_done
    ldrb w8, [x3]
    add x3, x3, #4
    sub x4, x4, #1
    cmp w8, #0x0A
    beq inc_line
    cmp w8, #0x20
    beq inc_word_flag
    cmp w8, #0x09
    beq inc_word_flag
    cmp w8, #0x0B
    beq inc_word_flag
    cmp w8, #0x0C
    beq inc_word_flag
    cmp w8, #0x0D
    beq inc_word_flag
    cbnz x7, non_word
    add x6, x6, #1
    mov x7, #1
non_word:
    b scan_loop
inc_line:
    add x5, x5, #1
    mov x7, #0
    b scan_loop
inc_word_flag:
    mov x7, #0
    b scan_loop
scan_done:
    adrp x8, out
    add x8, x8, #:lo12:out
    mov x9, x8          // out_ptr
    adrp x10, tmp
    add x10, x10, #:lo12:tmp
    mov x11, x10        // tmp_ptr
    mov x12, x10        // tmp_start
    // convert lines (x5)
    mov x13, x5
    mov x14, #0
    mov x15, #0
    cmp x13, #0
    beq conv_lines_zero_flag
    mov x15, #1
conv_lines_loop:
    mov w16, x13
    mov w17, #10
    udiv w16, w17
    add w17, w17, #48
    strb w17, [x11]
    add x11, x11, #1
    add x14, x14, #1
    mov x13, w16
    cmp x13, #0
    bne conv_lines_loop
conv_lines_zero_flag:
    cmp x15, #0
    beq conv_lines_zero_store
conv_lines_zero_store:
    mov w17, #48
    strb w17, [x11]
    add x11, x11, #1
    add x14, x14, #1
    // copy reversed digits
    mov x18, x12
    add x18, x18, x14
    sub x18, x18, #1
copy_lines:
    cmp x14, #0
    beq copy_lines_done
    ldrb w17, [x18]
    strb w17, [x9]
    add x9, x9, #1
    sub x18, x18, #1
    sub x14, x14, #1
    b copy_lines
copy_lines_done:
    mov w17, #32
    strb w17, [x9]
    add x9, x9, #1
    mov x11, x12
    // convert words (x6)
    mov x13, x6
    mov x14, #0
    mov x15, #0
    cmp x13, #0
    beq conv_words_zero_flag
    mov x15, #1
conv_words_loop:
    mov w16, x13
    mov w17, #10
    udiv w16, w17
    add w17, w17, #48
    strb w17, [x11]
    add x11, x11, #1
    add x14, x14, #1
    mov x13, w16
    cmp x13, #0
    bne conv_words_loop
conv_words_zero_flag:
    cmp x15, #0
    beq conv_words_zero_store
conv_words_zero_store:
    mov w17, #48
    strb w17, [x11]
    add x11, x11, #1
    add x14, x14, #1
    // copy reversed digits
    mov x18, x12
    add x18, x18, x14
    sub x18, x18, #1
copy_words:
    cmp x14, #0
    beq copy_words_done
    ldrb w17, [x18]
    strb w17, [x9]
    add x9, x9, #1
    sub x18, x18, #1
    sub x14, x14, #1
    b copy_words
copy_words_done:
    mov w17, #32
    strb w17, [x9]
    add x9, x9, #1
    mov x11, x12
    // convert bytes (x1)
    mov x13, x1
    mov x14, #0
    mov x15, #0
    cmp x13, #0
    beq conv_bytes_zero_flag
    mov x15, #1
conv_bytes_loop:
    mov w16, x13
    mov w17, #10
    udiv w16, w17
    add w17, w17, #48
    strb w17, [x11]
    add x11, x11, #1
    add x14, x14, #1
    mov x13, w16
    cmp x13, #0
    bne conv_bytes_loop
conv_bytes_zero_flag:
    cmp x15, #0
    beq conv_bytes_zero_store
conv_bytes_zero_store:
    mov w17, #48
    strb w17, [x11]
    add x11, x11, #1
    add x14, x14, #1
    // copy reversed digits
    mov x18, x12
    add x18, x18, x14
    sub x18, x18, #1
copy_bytes:
    cmp x14, #0
    beq copy_bytes_done
    ldrb w17, [x18]
    strb w17, [x9]
    add x9, x9, #1
    sub x18, x18, #1
    sub x14, x14, #1
    b copy_bytes
copy_bytes_done:
    mov w17, #10
    strb w17, [x9]
    add x9, x9, #1
    // write output
    mov x0, #1
    mov x1, x8
    mov x2, x9
    sub x2, x2, x8
    mov x8, #64
    svc #0
    // exit
    mov x0, #0
    mov x8, #93
    svc #0
.bss
buf:
    .space 65536
tmp:
    .space 8
out:
    .space 32
