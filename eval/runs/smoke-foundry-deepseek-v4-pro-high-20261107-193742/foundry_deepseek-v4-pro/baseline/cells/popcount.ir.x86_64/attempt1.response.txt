export function l $popcount(l %x) {
@start
    %mask1 =l copy 0x5555555555555555
    %mask2 =l copy 0x3333333333333333
    %mask3 =l copy 0x0f0f0f0f0f0f0f0f
    %mult =l copy 0x0101010101010101
    %t1 =l shr %x, 1
    %t1 =l and %t1, %mask1
    %x =l and %x, %mask1
    %x =l add %x, %t1
    %t2 =l shr %x, 2
    %t2 =l and %t2, %mask2
    %x =l and %x, %mask2
    %x =l add %x, %t2
    %t3 =l shr %x, 4
    %t3 =l and %t3, %mask3
    %x =l and %x, %mask3
    %x =l add %x, %t3
    %x =l mul %x, %mult
    %x =l shr %x, 56
    ret %x
}
