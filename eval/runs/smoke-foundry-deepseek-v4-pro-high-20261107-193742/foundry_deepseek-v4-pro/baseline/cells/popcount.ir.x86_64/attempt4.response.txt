export function l $popcount(l %x) {
@start
    %mask1 =l copy 0x5555555555555555
    %t0 =l shr %x, 1
    %t1 =l and %t0, %mask1
    %t2 =l and %x, %mask1
    %x1 =l add %t2, %t1
    %mask2 =l copy 0x3333333333333333
    %t3 =l shr %x1, 2
    %t4 =l and %t3, %mask2
    %t5 =l and %x1, %mask2
    %x2 =l add %t5, %t4
    %mask3 =l copy 0x0f0f0f0f0f0f0f0f
    %t6 =l shr %x2, 4
    %t7 =l and %t6, %mask3
    %t8 =l and %x2, %mask3

    %x4 =l mul %x3, %mult

    ret %x5
}
