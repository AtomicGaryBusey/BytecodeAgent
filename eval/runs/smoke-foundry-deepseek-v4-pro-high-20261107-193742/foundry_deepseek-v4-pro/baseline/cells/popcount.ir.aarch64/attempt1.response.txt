export function l $popcount(l %x) {
@start
    %t1 =l shr %x, 1
    %c1 =l copy 0x5555555555555555
    %t2 =l and %t1, %c1
    %t3 =l sub %x, %t2
    %t4 =l shr %t3, 2
    %c2 =l copy 0x3333333333333333
    %t5 =l and %t4, %c2
    %t6 =l and %t3, %c2
    %t7 =l add %t6, %t5
    %t8 =l shr %t7, 4
    %t9 =l add %t7, %t8
    %c3 =l copy 0x0F0F0F0F0F0F0F0F
    %t10 =l and %t9, %c3
    %c4 =l copy 0x0101010101010101
    %t11 =l mul %t10, %c4
    %t12 =l shr %t11, 56
    ret %t12
}
