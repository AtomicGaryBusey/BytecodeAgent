export function l $popcount(l %x) {
@start
        %c55_8 =l copy 0x55
        %c55_16 =l shl %c55_8, 8
        %tmp55_1 =l or %c55_8, %c55_16
        %tmp55_2 =l shl %tmp55_1, 16
        %tmp55_3 =l or %tmp55_1, %tmp55_2
        %tmp55_4 =l shl %tmp55_3, 32
        %m55 =l or %tmp55_3, %tmp55_4

        %c33_8 =l copy 0x33
        %c33_16 =l shl %c33_8, 8
        %tmp33_1 =l or %c33_8, %c33_16
        %tmp33_2 =l shl %tmp33_1, 16
        %tmp33_3 =l or %tmp33_1, %tmp33_2
        %tmp33_4 =l shl %tmp33_3, 32
        %m33 =l or %tmp33_3, %tmp33_4

        %c0F_8 =l copy 0x0F
        %c0F_16 =l shl %c0F_8, 8
        %tmp0F_1 =l or %c0F_8, %c0F_16
        %tmp0F_2 =l shl %tmp0F_1, 16
        %tmp0F_3 =l or %tmp0F_1, %tmp0F_2
        %tmp0F_4 =l shl %tmp0F_3, 32
        %m0F =l or %tmp0F_3, %tmp0F_4

        %c01_8 =l copy 0x01
        %c01_16 =l shl %c01_8, 8
        %tmp01_1 =l or %c01_8, %c01_16
        %tmp01_2 =l shl %tmp01_1, 16
        %tmp01_3 =l or %tmp01_1, %tmp01_2
        %tmp01_4 =l shl %tmp01_3, 32
        %m01 =l or %tmp01_3, %tmp01_4

        %shr1 =l shr %x, 1
        %and1 =l and %shr1, %m55
        %sub1 =l sub %x, %and1
        %shr2 =l shr %sub1, 2
        %and2 =l and %shr2, %m33
        %and3 =l and %sub1, %m33
        %add1 =l add %and3, %and2
        %shr4 =l shr %add1, 4
        %add2 =l add %add1, %shr4
        %and4 =l and %add2, %m0F
        %mul =l mul %and4, %m01
        %result =l shr %mul, 56
        ret %result
}
