export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w csltl %n, 2
	jnz %small, @ret, @go
@go
	%m1 =l sub %n, 1
	%pm =l mul %m1, 16
	%pp =l add %a, %pm
	%pk =l loadl %pp
	%pvp =l add %pp, 8
	%pv =l loadl %pvp
	%i =l copy 0
	%j =l copy 0
@loop
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%ck =w csltl %jk, %pk
	jnz %ck, @swap, @ckeq
@ckeq
	%ceq =w ceql %jk, %pk
	jnz %ceq, @ckv, @next
@ckv
	%jvp =l add %jp, 8
	%jv =l loadl %jvp
	%cv =w csltl %jv, %pv
	jnz %cv, @swap, @next
@swap
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	%jvp2 =l add %jp, 8
	%jv2 =l loadl %jvp2
	storel %jk, %ip
	storel %jv2, %ivp
	storel %ik, %jp
	storel %iv, %jvp2
	%i =l add %i, 1
	jmp @next
@next
	%j =l add %j, 1
	%more =w csltl %j, %m1
	jnz %more, @loop, @fin
@fin
	%ioff2 =l mul %i, 16
	%ip2 =l add %a, %ioff2
	%ik2 =l loadl %ip2
	%ivp2 =l add %ip2, 8
	%iv2 =l loadl %ivp2
	storel %pk, %ip2
	storel %pv, %ivp2
	storel %ik2, %pp
	storel %iv2, %pvp
	%r =l call $qsort_pairs(l %a, l %i)
	%ip1 =l add %i, 1
	%roff =l mul %ip1, 16
	%a =l add %a, %roff
	%n =l sub %n, %ip1
	%small2 =w csltl %n, 2
	jnz %small2, @ret, @go
@ret
	ret 0
}
