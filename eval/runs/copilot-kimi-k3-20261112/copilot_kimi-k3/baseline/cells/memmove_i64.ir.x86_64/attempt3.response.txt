export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt =w cultl %dst, %src
	jnz %lt, @finit, @btest
@btest
	%eq =w ceql %dst, %src
	jnz %eq, @ret, @binit
@finit
	%i =l copy 0
@floop
	%fd =w csgel %i, %n
	jnz %fd, @ret, @fbody
@fbody
	%fo =l mul %i, 8
	%fsp =l add %src, %fo
	%fv =l loadl %fsp
	%fdp =l add %dst, %fo
	storel %fv, %fdp
	%i =l add %i, 1
	jmp @floop
@binit
	%j =l sub %n, 1
@bloop
	%bd =w csltl %j, 0
	jnz %bd, @ret, @bbody
@bbody
	%bo =l mul %j, 8
	%bsp =l add %src, %bo
	%bv =l loadl %bsp
	%bdp =l add %dst, %bo
	storel %bv, %bdp
	%j =l sub %j, 1
	jmp @bloop
@ret
	ret 0
}
