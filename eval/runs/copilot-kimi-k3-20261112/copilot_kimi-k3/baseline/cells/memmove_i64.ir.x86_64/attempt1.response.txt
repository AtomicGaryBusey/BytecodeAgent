export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%lt = cultl %dst, %src
	jnz %lt, @finit, @btest
@btest
	%eq = ceql %dst, %src
	jnz %eq, @ret, @binit
@finit
	jmp @floop(0)
@floop(l %i)
	%fd = csgel %i, %n
	jnz %fd, @ret, @fbody
@fbody
	%fo = mul %i, 8
	%fsp = add %src, %fo
	%fv = loadl %fsp
	%fdp = add %dst, %fo
	storel %fv, %fdp
	%fi = add %i, 1
	jmp @floop(%fi)
@binit
	%last = sub %n, 1
	jmp @bloop(%last)
@bloop(l %j)
	%bd = csltl %j, 0
	jnz %bd, @ret, @bbody
@bbody
	%bo = mul %j, 8
	%bsp = add %src, %bo
	%bv = loadl %bsp
	%bdp = add %dst, %bo
	storel %bv, %bdp
	%bj = sub %j, 1
	jmp @bloop(%bj)
@ret
	ret 0
}
