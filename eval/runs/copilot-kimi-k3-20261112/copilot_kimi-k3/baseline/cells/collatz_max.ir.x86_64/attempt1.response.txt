export function l $collatz_max(l %n) {
@start
	%slots =l alloc8 32
	%one =l copy 1
	%zero =l copy 0
	storel %one, %slots
	%maxp0 =l add %slots, 8
	storel %zero, %maxp0
	jmp @outer
@outer
	%k =l loadl %slots
	%in =w cslel %k, %n
	jnz %in, @body, @done
@body
	%xp0 =l add %slots, 16
	storel %k, %xp0
	%sp0 =l add %slots, 24
	%z0 =l copy 0
	storel %z0, %sp0
	jmp @inner
@inner
	%xp1 =l add %slots, 16
	%x =l loadl %xp1
	%is1 =w ceql %x, 1
	jnz %is1, @after, @step
@step
	%parity =l and %x, 1
	%even =w ceql %parity, 0
	jnz %even, @even, @odd
@even
	%half =l div %x, 2
	%xp2 =l add %slots, 16
	storel %half, %xp2
	jmp @inc
@odd
	%tri =l mul %x, 3
	%nextx =l add %tri, 1
	%xp3 =l add %slots, 16
	storel %nextx, %xp3
	jmp @inc
@inc
	%sp1 =l add %slots, 24
	%s =l loadl %sp1
	%s1 =l add %s, 1
	storel %s1, %sp1
	jmp @inner
@after
	%sp2 =l add %slots, 24
	%steps =l loadl %sp2
	%maxp1 =l add %slots, 8
	%max =l loadl %maxp1
	%gt =w csgtl %steps, %max
	jnz %gt, @update, @next
@update
	%maxp2 =l add %slots, 8
	storel %steps, %maxp2
	jmp @next
@next
	%k1 =l add %k, 1
	storel %k1, %slots
	jmp @outer
@done
	%maxp3 =l add %slots, 8
	%ret =l loadl %maxp3
	ret %ret
}
