export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i0 =l copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @inc %inext
	%lt =w csltl %i, %n
	jnz %lt, @check, @done
@check
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @done, @inc
@inc
	%inext =l add %i, 1
	jmp @loop
@done
	ret %i
}
