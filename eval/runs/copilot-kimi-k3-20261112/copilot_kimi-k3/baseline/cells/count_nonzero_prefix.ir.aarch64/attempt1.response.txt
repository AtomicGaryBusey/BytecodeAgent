export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%pi =l alloc8 8
	storel 0, %pi
@loop
	%i =l loadl %pi
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @done, @next
@next
	%i2 =l add %i, 1
	storel %i2, %pi
	jmp @loop
@done
	%r =l loadl %pi
	ret %r
}
