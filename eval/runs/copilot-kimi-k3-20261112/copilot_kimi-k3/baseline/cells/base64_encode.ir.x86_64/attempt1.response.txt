export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%o =l copy 0
@loop
	%rem =l sub %n, %i
	%cdone =w csltl %rem, 1
	jnz %cdone, @end, @body
@body
	%p0 =l add %src, %i
	%b0 =w loadub %p0
	%val =w shl %b0, 16
	%ch2 =w csltl 1, %rem
	jnz %ch2, @ld1, @emitinit
@ld1
	%i1 =l add %i, 1
	%p1 =l add %src, %i1
	%b1 =w loadub %p1
	%t1 =w shl %b1, 8
	%val =w or %val, %t1
	%ch3 =w csltl 2, %rem
	jnz %ch3, @ld2, @emitinit
@ld2
	%i2 =l add %i, 2
	%p2 =l add %src, %i2
	%b2 =w loadub %p2
	%val =w or %val, %b2
	jmp @emitinit
@emitinit
	%ck3 =w csltl 2, %rem
	jnz %ck3, @k4, @krest
@krest
	%ck2 =w csltl 1, %rem
	jnz %ck2, @k3, @k2
@k4
	%k =w copy 4
	%j =w copy 0
	jmp @eloop
@k3
	%k =w copy 3
	%j =w copy 0
	jmp @eloop
@k2
	%k =w copy 2
	%j =w copy 0
	jmp @eloop
@eloop
	%cj =w csltw %j, %k
	jnz %cj, @ebody, @pad
@ebody
	%sixj =w mul %j, 6
	%sh =w sub 18, %sixj
	%vv =w shr %val, %sh
	%v =w and %vv, 63
	%cm1 =w csltw %v, 26
	jnz %cm1, @mA, @cm2
@mA
	%c =w add %v, 65
	jmp @store
@cm2
	%cm3 =w csltw %v, 52
	jnz %cm3, @ma, @cm4
@ma
	%c =w add %v, 71
	jmp @store
@cm4
	%cm5 =w csltw %v, 62
	jnz %cm5, @m0, @cm6
@m0
	%c =w sub %v, 4
	jmp @store
@cm6
	%cm7 =w ceqw %v, 62
	jnz %cm7, @mplus, @mslash
@mplus
	%c =w copy 43
	jmp @store
@mslash
	%c =w copy 47
@store
	%po =l add %out, %o
	storeb %c, %po
	%o =l add %o, 1
	%j =w add %j, 1
	jmp @eloop
@pad
	%cp =w csltw %j, 4
	jnz %cp, @padbody, @next
@padbody
	%pop =l add %out, %o
	%eq =w copy 61
	storeb %eq, %pop
	%o =l add %o, 1
	%j =w add %j, 1
	jmp @pad
@next
	%i =l add %i, 3
	jmp @loop
@end
	ret %o
}
