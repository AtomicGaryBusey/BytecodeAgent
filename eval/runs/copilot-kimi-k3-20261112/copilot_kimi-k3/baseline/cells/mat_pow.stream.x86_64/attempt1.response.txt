mat_pow:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r10, rsi
    imul r10, rsi
    xor r9, r9
.Lzero_out:
    cmp r9, r10
    jge .Ldiag
    mov qword ptr [rcx + r9*8], 0
    inc r9
    jmp .Lzero_out
.Ldiag:
    xor rbx, rbx
.Ldiag_loop:
    cmp rbx, rsi
    jge .Lpow_loop
    mov rax, rbx
    imul rax, rsi
    add rax, rbx
    mov qword ptr [rcx + rax*8], 1
    inc rbx
    jmp .Ldiag_loop
.Lpow_loop:
    test rdx, rdx
    jz .Lzero_tmp
    xor rbx, rbx
.Li_loop:
    cmp rbx, rsi
    jge .Lcopy
    mov r9, rbx
    imul r9, rsi
    xor rbp, rbp
.Lj_loop:
    cmp rbp, rsi
    jge .Lnext_i
    xor r13, r13
    xor r12, r12
    xor r11, r11
.Lt_loop:
    cmp r12, rsi
    jge .Lstore
    mov rax, r9
    add rax, r12
    mov r14, qword ptr [rcx + rax*8]
    mov rax, r11
    add rax, rbp
    mov rax, qword ptr [rdi + rax*8]
    imul r14, rax
    add r13, r14
    inc r12
    add r11, rsi
    jmp .Lt_loop
.Lstore:
    mov rax, r9
    add rax, rbp
    mov qword ptr [r8 + rax*8], r13
    inc rbp
    jmp .Lj_loop
.Lnext_i:
    inc rbx
    jmp .Li_loop
.Lcopy:
    xor r9, r9
.Lcopy_loop:
    cmp r9, r10
    jge .Lnext_pow
    mov rax, qword ptr [r8 + r9*8]
    mov qword ptr [rcx + r9*8], rax
    inc r9
    jmp .Lcopy_loop
.Lnext_pow:
    dec rdx
    jmp .Lpow_loop
.Lzero_tmp:
    lea r9, [r10 + r10]
    xor r11, r11
.Lzt_loop:
    cmp r11, r9
    jge .Ldone
    mov qword ptr [r8 + r11*8], 0
    inc r11
    jmp .Lzt_loop
.Ldone:
    xor rax, rax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
