export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i0 =l copy 0
	%c0 =w copy 0
	jmp @loop
@loop
	%i =l phi @start %i0, @body %i1
	%cnt =w phi @start %c0, @body %cnt1
	%go =w csltl %i, %n
	jnz %go, @body, @done
@body
	%off =l mul %i, 8
	%sp =l add %src, %off
	%s =l loadl %sp
	%r =l call %f(l %s)
	%dp =l add %dst, %off
	storel %r, %dp
	%lt =w csltl %r, %s
	%cnt1 =w add %cnt, %lt
	%i1 =l add %i, 1
	jmp @loop
@done
	%out =l extuw %cnt
	ret %out
}
