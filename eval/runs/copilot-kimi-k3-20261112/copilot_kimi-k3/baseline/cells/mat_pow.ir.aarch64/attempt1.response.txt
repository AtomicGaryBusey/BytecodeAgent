export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	jmp @ir
@ir
	%ir =l phi @start 0, @icd %irn
	%irc =w csltl %ir, %n
	jnz %irc, @ic0, @idone
@ic0
	jmp @ic
@ic
	%ic =l phi @ic0 0, @icb %icn
	%icc =w csltl %ic, %n
	jnz %icc, @icb, @icd
@icb
	%i1 =l mul %ir, %n
	%i2 =l add %i1, %ic
	%i3 =l shl %i2, 3
	%i4 =l add %out, %i3
	%iw =w ceql %ir, %ic
	%iv =l extsw %iw
	storel %iv, %i4
	%icn =l add %ic, 1
	jmp @ic
@icd
	%irn =l add %ir, 1
	jmp @ir
@idone
	jmp @el
@el
	%e =l phi @idone 0, @cpd %en
	%ec =w csltl %e, %k
	jnz %ec, @mr0, @edoneall
@mr0
	jmp @mr
@mr
	%mr =l phi @mr0 0, @mcd %mrn
	%mrc =w csltl %mr, %n
	jnz %mrc, @mc0, @mrdone
@mc0
	jmp @mc
@mc
	%mc =l phi @mc0 0, @mtd %mcn
	%mcc =w csltl %mc, %n
	jnz %mcc, @mt0, @mcd
@mt0
	jmp @mt
@mt
	%mt =l phi @mt0 0, @mtb %mtn
	%acc =l phi @mt0 0, @mtb %accn
	%mtc =w csltl %mt, %n
	jnz %mtc, @mtb, @mtd
@mtb
	%x1 =l mul %mr, %n
	%x2 =l add %x1, %mt
	%x3 =l shl %x2, 3
	%x4 =l add %out, %x3
	%xv =l loadl %x4
	%y1 =l mul %mt, %n
	%y2 =l add %y1, %mc
	%y3 =l shl %y2, 3
	%y4 =l add %a, %y3
	%yv =l loadl %y4
	%p =l mul %xv, %yv
	%accn =l add %acc, %p
	%mtn =l add %mt, 1
	jmp @mt
@mtd
	%s1 =l mul %mr, %n
	%s2 =l add %s1, %mc
	%s3 =l shl %s2, 3
	%s4 =l add %tmp, %s3
	storel %acc, %s4
	%mcn =l add %mc, 1
	jmp @mc
@mcd
	%mrn =l add %mr, 1
	jmp @mr
@mrdone
	jmp @cp
@cp
	%cpi =l phi @mrdone 0, @cpb %cpin
	%cpc =w csltl %cpi, %nn
	jnz %cpc, @cpb, @cpd
@cpb
	%co =l shl %cpi, 3
	%cta =l add %tmp, %co
	%cv =l loadl %cta
	%coa =l add %out, %co
	storel %cv, %coa
	%cpin =l add %cpi, 1
	jmp @cp
@cpd
	%en =l add %e, 1
	jmp @el
@edoneall
	%nn2 =l add %nn, %nn
	jmp @zl
@zl
	%zi =l phi @edoneall 0, @zb %zin
	%zc =w csltl %zi, %nn2
	jnz %zc, @zb, @zd
@zb
	%zo =l shl %zi, 3
	%za =l add %tmp, %zo
	storel 0, %za
	%zin =l add %zi, 1
	jmp @zl
@zd
	ret 0
}
