export function l $bmax(l %a, l %n) {
@start
	%m0 =l loadl %a
	%i0 =l copy 1
	jmp @loop
@loop
	%i =l phi @start %i0, @next %i1
	%m =l phi @start %m0, @next %m2
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%take =w csltl %m, %v
	jnz %take, @upd, @next0
@upd
	jmp @next
@next0
	jmp @next
@next
	%m2 =l phi @upd %v, @next0 %m
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %m
}
