binary_search:
    mov x3, x0
    mov x4, xzr
    mov x5, x1
.Lloop:
    cmp x4, x5
    b.ge .Lnotfound
    add x6, x4, x5
    lsr x6, x6, #1
    ldr x7, [x3, x6, lsl #3]
    cmp x7, x2
    b.eq .Lfound
    b.lt .Lright
    mov x5, x6
    b .Lloop
.Lright:
    add x4, x6, #1
    b .Lloop
.Lfound:
    mov x0, x6
    ret
.Lnotfound:
    movn x0, #0
    ret
