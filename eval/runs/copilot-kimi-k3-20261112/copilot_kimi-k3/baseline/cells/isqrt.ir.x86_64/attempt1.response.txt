export function l $isqrt(l %n) {
@start
	%lo0 =l copy 0
	%hi0 =l copy 4294967296
	%slot =l alloc8 16
	%hiaddr =l add %slot, 8
	storel %lo0, %slot
	storel %hi0, %hiaddr
	%one =l copy 1
@loop
	%lo1 =l loadl %slot
	%hi1 =l loadl %hiaddr
	%diff =l sub %hi1, %lo1
	%cont =w cugtl %diff, %one
	jnz %cont, @body, @done
@body
	%sum =l add %lo1, %hi1
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	storel %mid, %slot
	jmp @loop
@sethi
	storel %mid, %hiaddr
	jmp @loop
@done
	%res =l loadl %slot
	ret %res
}
