export function l $reverse(l %a, l %n) {
@start
	%j0 =l sub %n, 1
	jmp @loop
@loop
	%i =l phi @start 0, @body %i1
	%j =l phi @start %j0, @body %j1
	%cont =w csltl %i, %j
	jnz %cont, @body, @done
@body
	%io =l mul %i, 8
	%ip =l add %a, %io
	%jo =l mul %j, 8
	%jp =l add %a, %jo
	%iv =l loadl %ip
	%jv =l loadl %jp
	storel %jv, %ip
	storel %iv, %jp
	%i1 =l add %i, 1
	%j1 =l sub %j, 1
	jmp @loop
@done
	ret 0
}
