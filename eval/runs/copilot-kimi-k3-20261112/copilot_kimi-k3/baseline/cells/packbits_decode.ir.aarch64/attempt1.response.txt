export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop
@loop
	%si =l phi @start 0, @next %si_n
	%oi =l phi @start 0, @next %oi_n
	%end =w csgel %si, %n
	jnz %end, @retn, @hdr
@hdr
	%ph =l add %src, %si
	%h =w loadsb %ph
	%si1 =l add %si, 1
	%neg =w csltw %h, 0
	jnz %neg, @neg, @lit
@lit
	%lcw =w add %h, 1
	%lc =l extsw %lcw
	%lrem =l sub %n, %si1
	%lshort =w csltl %lrem, %lc
	jnz %lshort, @err2, @lcap
@lcap
	%lneed =l add %oi, %lc
	%lover =w csgtl %lneed, %cap
	jnz %lover, @lfill, @lcloop
@lcloop
	%i =l phi @lcap 0, @lcbody %i1
	%icd =w csgel %i, %lc
	jnz %icd, @ldone, @lcbody
@lcbody
	%lsa =l add %src, %si1
	%lsp =l add %lsa, %i
	%lb =w loadub %lsp
	%lda =l add %out, %oi
	%ldp =l add %lda, %i
	storeb %lb, %ldp
	%i1 =l add %i, 1
	jmp @lcloop
@ldone
	%si_l =l add %si1, %lc
	%oi_l =l add %oi, %lc
	jmp @next
@lfill
	%lroom =l sub %cap, %oi
	jmp @lfloop
@lfloop
	%j =l phi @lfill 0, @lfbody %j1
	%jd =w csgel %j, %lroom
	jnz %jd, @err1, @lfbody
@lfbody
	%fsa =l add %src, %si1
	%fsp =l add %fsa, %j
	%fb =w loadub %fsp
	%fda =l add %out, %oi
	%fdp =l add %fda, %j
	storeb %fb, %fdp
	%j1 =l add %j, 1
	jmp @lfloop
@neg
	%nop =w ceqw %h, -128
	jnz %nop, @nopdone, @rep
@nopdone
	jmp @next
@rep
	%rcw =w sub 1, %h
	%rc =l extsw %rcw
	%rrem =l sub %n, %si1
	%rshort =w csltl %rrem, 1
	jnz %rshort, @err2, @rcap
@rcap
	%rneed =l add %oi, %rc
	%rover =w csgtl %rneed, %cap
	jnz %rover, @rfill, @rnorm
@rnorm
	%rbp =l add %src, %si1
	%rb =w loadub %rbp
	jmp @rnloop
@rnloop
	%k =l phi @rnorm 0, @rnbody %k1
	%kd =w csgel %k, %rc
	jnz %kd, @rdone, @rnbody
@rnbody
	%rda =l add %out, %oi
	%rdp =l add %rda, %k
	storeb %rb, %rdp
	%k1 =l add %k, 1
	jmp @rnloop
@rdone
	%si_r =l add %si1, 1
	%oi_r =l add %oi, %rc
	jmp @next
@rfill
	%rroom =l sub %cap, %oi
	%fbp =l add %src, %si1
	%fbb =w loadub %fbp
	jmp @rfloop
@rfloop
	%m =l phi @rfill 0, @rfbody %m1
	%md =w csgel %m, %rroom
	jnz %md, @err1, @rfbody
@rfbody
	%mda =l add %out, %oi
	%mdp =l add %mda, %m
	storeb %fbb, %mdp
	%m1 =l add %m, 1
	jmp @rfloop
@next
	%si_n =l phi @ldone %si_l, @rdone %si_r, @nopdone %si1
	%oi_n =l phi @ldone %oi_l, @rdone %oi_r, @nopdone %oi
	jmp @loop
@retn
	ret %oi
@err1
	ret -1
@err2
	ret -2
}
