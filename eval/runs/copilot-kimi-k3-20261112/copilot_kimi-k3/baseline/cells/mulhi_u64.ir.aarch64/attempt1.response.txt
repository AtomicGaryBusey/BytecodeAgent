export function l $mulhi_u64(l %a, l %b) {
@start
	%mask =l copy 4294967295
	%al =l and %a, %mask
	%ah =l shr %a, 32
	%bl =l and %b, %mask
	%bh =l shr %b, 32
	%p0 =l mul %al, %bl
	%p1 =l mul %al, %bh
	%p2 =l mul %ah, %bl
	%p3 =l mul %ah, %bh
	%p0h =l shr %p0, 32
	%p1l =l and %p1, %mask
	%p2l =l and %p2, %mask
	%s0 =l add %p0h, %p1l
	%s1 =l add %s0, %p2l
	%sc =l shr %s1, 32
	%p1h =l shr %p1, 32
	%p2h =l shr %p2, 32
	%h0 =l add %p3, %p1h
	%h1 =l add %h0, %p2h
	%hi =l add %h1, %sc
	ret %hi
}
