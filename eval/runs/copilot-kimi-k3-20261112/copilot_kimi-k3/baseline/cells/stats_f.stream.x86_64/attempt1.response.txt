stats_f:
push rbx
push rbp
push r12
push r13
push r14
push r15
sub rsp, 8
mov r12, rdi
mov r13, rsi
mov r14, rdx
mov r15, rcx
xor ebp, ebp
mov rdi, qword ptr [r12]
call r14
mov qword ptr [r15], rax
mov qword ptr [r15 + 8], rax
mov qword ptr [r15 + 16], rax
test rax, 1
je .Lfirst_even
inc rbp
.Lfirst_even:
mov ebx, 1
.Lloop:
cmp rbx, r13
jge .Ldone
mov rdi, qword ptr [r12 + rbx*8]
call r14
cmp rax, qword ptr [r15]
jge .Lnot_min
mov qword ptr [r15], rax
.Lnot_min:
cmp rax, qword ptr [r15 + 8]
jle .Lnot_max
mov qword ptr [r15 + 8], rax
.Lnot_max:
add qword ptr [r15 + 16], rax
test rax, 1
je .Leven
inc rbp
.Leven:
inc rbx
jmp .Lloop
.Ldone:
mov rax, rbp
add rsp, 8
pop r15
pop r14
pop r13
pop r12
pop rbp
pop rbx
ret
