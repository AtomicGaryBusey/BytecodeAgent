export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%cnt =l copy 0
@loop
	%go =w csltl %i, %n
	jnz %go, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%cmp =w cugtl %v, %t
	%cmpl =l extsw %cmp
	%cnt =l add %cnt, %cmpl
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
