export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%i0 =l copy 0
@id_loop
	%cid =w csltl %i0, %nn
	jnz %cid, @id_body, @id_done
@id_body
	%row =l div %i0, %n
	%col =l rem %i0, %n
	%eq =w ceql %row, %col
	%ib =l mul %i0, 8
	%ip =l add %out, %ib
	jnz %eq, @id_one, @id_zero
@id_one
	storel 1, %ip
	jmp @id_next
@id_zero
	storel 0, %ip
	jmp @id_next
@id_next
	%i0 =l add %i0, 1
	jmp @id_loop
@id_done
	%kk =l copy %k
@outer
	%ck =w csgtl %kk, 0
	jnz %ck, @outer_body, @outer_done
@outer_body
	%i =l copy 0
@m_loop
	%ci =w csltl %i, %nn
	jnz %ci, @m_body, @m_done
@m_body
	%r =l div %i, %n
	%c =l rem %i, %n
	%acc =l copy 0
	%t =l copy 0
@d_loop
	%ct =w csltl %t, %n
	jnz %ct, @d_body, @d_done
@d_body
	%rn =l mul %r, %n
	%xidx =l add %rn, %t
	%xb =l mul %xidx, 8
	%xp =l add %out, %xb
	%xv =l loadl %xp
	%tn =l mul %t, %n
	%yidx =l add %tn, %c
	%yb =l mul %yidx, 8
	%yp =l add %a, %yb
	%yv =l loadl %yp
	%prod =l mul %xv, %yv
	%acc =l add %acc, %prod
	%t =l add %t, 1
	jmp @d_loop
@d_done
	%mb =l mul %i, 8
	%mp =l add %tmp, %mb
	storel %acc, %mp
	%i =l add %i, 1
	jmp @m_loop
@m_done
	%j =l copy 0
@c_loop
	%cj =w csltl %j, %nn
	jnz %cj, @c_body, @c_done
@c_body
	%jb =l mul %j, 8
	%src =l add %tmp, %jb
	%dst =l add %out, %jb
	%v =l loadl %src
	storel %v, %dst
	%j =l add %j, 1
	jmp @c_loop
@c_done
	%kk =l sub %kk, 1
	jmp @outer
@outer_done
	%nn2 =l mul %nn, 2
	%z =l copy 0
@z_loop
	%cz =w csltl %z, %nn2
	jnz %cz, @z_body, @z_done
@z_body
	%zb =l mul %z, 8
	%zp =l add %tmp, %zb
	storel 0, %zp
	%z =l add %z, 1
	jmp @z_loop
@z_done
	ret 0
}
