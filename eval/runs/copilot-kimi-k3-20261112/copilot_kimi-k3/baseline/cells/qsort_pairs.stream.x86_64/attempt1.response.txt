qsort_pairs:
    cmp rsi, 2
    jl .Lret
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 16
    mov rbx, rdi
    mov r12, rsi
.Louter:
    mov r15, qword ptr [rbx]
    mov rax, qword ptr [rbx + 8]
    mov qword ptr [rsp], rax
    mov r13, -1
    mov r14, r12
.Lscan_i:
    inc r13
    mov rax, r13
    shl rax, 4
    mov rcx, qword ptr [rbx + rax]
    mov rdx, qword ptr [rbx + rax + 8]
    cmp rcx, r15
    jl .Lscan_i
    jg .Lscan_j
    cmp rdx, qword ptr [rsp]
    jl .Lscan_i
.Lscan_j:
    dec r14
    mov rax, r14
    shl rax, 4
    mov rcx, qword ptr [rbx + rax]
    mov rdx, qword ptr [rbx + rax + 8]
    cmp rcx, r15
    jg .Lscan_j
    jl .Lcheck
    cmp rdx, qword ptr [rsp]
    jg .Lscan_j
.Lcheck:
    cmp r13, r14
    jge .Lpart_done
    mov rax, r13
    shl rax, 4
    mov r8, r14
    shl r8, 4
    mov rcx, qword ptr [rbx + rax]
    mov rdx, qword ptr [rbx + rax + 8]
    mov r9, qword ptr [rbx + r8]
    mov r10, qword ptr [rbx + r8 + 8]
    mov qword ptr [rbx + rax], r9
    mov qword ptr [rbx + rax + 8], r10
    mov qword ptr [rbx + r8], rcx
    mov qword ptr [rbx + r8 + 8], rdx
    jmp .Lscan_i
.Lpart_done:
    mov rdi, rbx
    lea rsi, [r14 + 1]
    call qsort_pairs
    lea rax, [r14 + 1]
    shl rax, 4
    add rbx, rax
    sub r12, r14
    dec r12
    cmp r12, 2
    jge .Louter
    add rsp, 16
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
.Lret:
    xor eax, eax
    ret
