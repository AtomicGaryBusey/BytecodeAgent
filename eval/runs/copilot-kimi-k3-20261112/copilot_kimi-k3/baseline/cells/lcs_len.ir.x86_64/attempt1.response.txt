export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @inithead
@inithead
	%j =l phi @start 0, @initbody %jnext
	%iz =w csgtl %j, %n
	jnz %iz, @main, @initbody
@initbody
	%o =l mul %j, 8
	%p =l add %row, %o
	storel 0, %p
	%jnext =l add %j, 1
	jmp @inithead
@main
	jmp @ihead
@ihead
	%i =l phi @main 0, @iend %inext
	%id =w csgel %i, %m
	jnz %id, @ret, @ibody
@ibody
	%xa =l add %x, %i
	%xi =w loadub %xa
	jmp @jhead
@jhead
	%jj =l phi @ibody 1, @jend %jjnext
	%diag =l phi @ibody 0, @jend %up
	%jd =w csgtl %jj, %n
	jnz %jd, @iend, @jbody
@jbody
	%off =l mul %jj, 8
	%pj =l add %row, %off
	%up =l loadl %pj
	%offm =l sub %off, 8
	%pjm =l add %row, %offm
	%left =l loadl %pjm
	%ya =l add %y, %jj
	%yaa =l sub %ya, 1
	%yb =w loadub %yaa
	%eq =w ceqw %xi, %yb
	jnz %eq, @match, @nomatch
@match
	%mv =l add %diag, 1
	jmp @jend
@nomatch
	%gt =w csgtl %up, %left
	jnz %gt, @kup, @uleft
@kup
	jmp @jend
@uleft
	jmp @jend
@jend
	%nv =l phi @match %mv, @kup %up, @uleft %left
	storel %nv, %pj
	%jjnext =l add %jj, 1
	jmp @jhead
@iend
	%inext =l add %i, 1
	jmp @ihead
@ret
	%offn =l mul %n, 8
	%pn =l add %row, %offn
	%res =l loadl %pn
	ret %res
}
