export function l $binary_search(l %a, l %n, l %key) {
@entry
	%lo0 =l copy 0
	%hi0 =l sub %n, 1
	jmp @loop
@loop
	%lo =l phi @entry %lo0, @left %loL, @right %loR
	%hi =l phi @entry %hi0, @left %hiL, @right %hiR
	%in =w cslel %lo, %hi
	jnz %in, @body, @notfound
@body
	%d =l sub %hi, %lo
	%h =l sar %d, 1
	%m =l add %lo, %h
	%o =l mul %m, 8
	%p =l add %a, %o
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@left
	%loL =l copy %lo
	%hiL =l sub %m, 1
	jmp @loop
@right
	%loR =l add %m, 1
	%hiR =l copy %hi
	jmp @loop
@found
	ret %m
@notfound
	ret -1
}
