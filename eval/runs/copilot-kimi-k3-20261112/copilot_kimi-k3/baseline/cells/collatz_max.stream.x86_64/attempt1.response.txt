collatz_max:
    xor eax, eax
    mov ecx, 1
.Louter:
    cmp rcx, rdi
    jg .Ldone
    mov rdx, rcx
    xor r8d, r8d
.Linner:
    cmp rdx, 1
    je .Lafter
    test dl, 1
    jnz .Lodd
    shr rdx, 1
    jmp .Lstep
.Lodd:
    lea rdx, [rdx + rdx*2 + 1]
.Lstep:
    inc r8
    jmp .Linner
.Lafter:
    cmp r8, rax
    jle .Lnot_new
    mov rax, r8
.Lnot_new:
    inc rcx
    jmp .Louter
.Ldone:
    ret
