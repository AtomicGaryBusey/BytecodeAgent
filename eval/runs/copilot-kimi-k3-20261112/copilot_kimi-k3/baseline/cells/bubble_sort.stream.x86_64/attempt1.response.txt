bubble_sort:
	cmp rsi, 2
	jl .Ldone
	lea r9, [rsi - 1]
.Louter:
	test r9, r9
	jle .Ldone
	xor ecx, ecx
.Linner:
	cmp rcx, r9
	jge .Louter_next
	mov rdx, qword ptr [rdi + rcx*8]
	mov r8, qword ptr [rdi + rcx*8 + 8]
	cmp rdx, r8
	jle .Lnoswap
	mov qword ptr [rdi + rcx*8], r8
	mov qword ptr [rdi + rcx*8 + 8], rdx
.Lnoswap:
	inc rcx
	jmp .Linner
.Louter_next:
	dec r9
	jmp .Louter
.Ldone:
	xor eax, eax
	ret
