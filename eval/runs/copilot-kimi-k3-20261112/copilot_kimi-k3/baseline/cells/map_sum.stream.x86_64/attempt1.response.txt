map_sum:
push rbx
push r12
push r13
push r14
push r15
mov rbx, rdi
mov r12, rsi
mov r13, rdx
xor r14, r14
xor r15, r15
test r12, r12
je .Ldone
.Lloop:
mov rdi, qword ptr [rbx + r14*8]
call r13
add r15, rax
inc r14
cmp r14, r12
jl .Lloop
.Ldone:
mov rax, r15
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
