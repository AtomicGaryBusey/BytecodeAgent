map_into:
push rbx
push r12
push r13
push r14
push r15
mov r13, rdi
mov r12, rsi
lea r14, [rsi + rdx*8]
mov r15, rcx
xor ebx, ebx
.Lloop:
cmp r12, r14
jae .Ldone
mov rdi, qword ptr [r12]
call r15
mov qword ptr [r13], rax
cmp rax, qword ptr [r12]
setl cl
movzx ecx, cl
add rbx, rcx
add r12, 8
add r13, 8
jmp .Lloop
.Ldone:
mov rax, rbx
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
