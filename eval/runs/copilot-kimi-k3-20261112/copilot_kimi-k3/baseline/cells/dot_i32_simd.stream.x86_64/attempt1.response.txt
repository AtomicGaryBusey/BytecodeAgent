dot_i32_simd:
    xor ecx, ecx
    pxor xmm7, xmm7
    lea r10, [rdx - 4]
.Lvec:
    cmp rcx, r10
    jg .Ltail
    movdqu xmm0, xmmword ptr [rdi + rcx*4]
    movdqu xmm1, xmmword ptr [rsi + rcx*4]
    movdqa xmm2, xmm0
    pmuldq xmm2, xmm1
    pshufd xmm3, xmm0, 0x31
    pshufd xmm4, xmm1, 0x31
    pmuldq xmm3, xmm4
    paddq xmm2, xmm3
    paddq xmm7, xmm2
    add rcx, 4
    jmp .Lvec
.Ltail:
    movdqa xmm6, xmm7
    psrldq xmm6, 8
    paddq xmm7, xmm6
    movq rax, xmm7
.Lscalar:
    cmp rcx, rdx
    jge .Ldone
    movsxd r8, dword ptr [rdi + rcx*4]
    movsxd r9, dword ptr [rsi + rcx*4]
    imul r8, r9
    add rax, r8
    inc rcx
    jmp .Lscalar
.Ldone:
    ret
