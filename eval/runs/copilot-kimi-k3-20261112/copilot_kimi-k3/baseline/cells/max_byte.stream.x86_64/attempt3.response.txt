max_byte:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    pxor xmm0, xmm0
    mov rcx, rsi
    and rcx, -16
    xor edx, edx
    cmp rdx, rcx
    jae .Ltail
.Lloop16:
    movdqu xmm1, [rdi + rdx]
    pmaxub xmm0, xmm1
    add rdx, 16
    cmp rdx, rcx
    jb .Lloop16
.Ltail:
    cmp rdx, rsi
    jae .Lreduce
    movzx ecx, byte ptr [rdi + rdx]
    movd xmm1, ecx
    pmaxub xmm0, xmm1
    add rdx, 1
    jmp .Ltail
.Lreduce:
    pshufd xmm1, xmm0, 0x4e
    pmaxub xmm0, xmm1
    pshufd xmm1, xmm0, 0xb1
    pmaxub xmm0, xmm1
    pshuflw xmm1, xmm0, 0xb1
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    movd eax, xmm0
    movzx eax, al
.Ldone:
    ret
