max_byte:
    xor eax, eax
    test rsi, rsi
    jle .Ldone
    pxor xmm0, xmm0
    xor r8d, r8d
    mov rcx, rsi
    and rcx, -16
    xor edx, edx
    cmp rdx, rcx
    jae .Ltail
.Lloop16:
    movdqu xmm1, [rdi + rdx]
    pmaxub xmm0, xmm1
    add rdx, 16
    cmp rdx, rcx
    jb .Lloop16
.Ltail:
    cmp rdx, rsi
    jae .Lreduce
    movzx ecx, byte [rdi + rdx]
    cmp r8d, ecx
    cmovb r8d, ecx
    inc rdx
    jmp .Ltail
.Lreduce:
    movdqa xmm1, xmm0
    psrldq xmm1, 8
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 4
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 2
    pmaxub xmm0, xmm1
    movdqa xmm1, xmm0
    psrldq xmm1, 1
    pmaxub xmm0, xmm1
    pextrb eax, xmm0, 0
    cmp eax, r8d
    cmovb eax, r8d
.Ldone:
    ret
