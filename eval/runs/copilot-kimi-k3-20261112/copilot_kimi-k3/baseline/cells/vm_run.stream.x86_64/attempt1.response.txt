    xor r9d, r9d
.Lloop:

    je .Ldone
    mov rax, qword ptr [rdi + r8*8]
    cmp rax, 0
    je .Ldone
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9

    mov qword ptr [rdx + r9*8], rax

    jmp .Lloop

.Ldrop:

    jmp .Lloop
.Lover:

    mov qword ptr [rdx + r9*8], rax
    inc r9
    inc r8

    dec r9

    add r8, 2
    test r10, r10
    jne .Lloop
    mov r8, rax
    jmp .Lloop
.Ljmp:

    jmp .Lloop
.Ldone:

    test r9, r9
    je .Lzfill
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lzfill:

.Lzfloop:

    jae .Lret
    mov qword ptr [rdx + r10*8], 0
    inc r10
    jmp .Lzfloop
.Lret:
