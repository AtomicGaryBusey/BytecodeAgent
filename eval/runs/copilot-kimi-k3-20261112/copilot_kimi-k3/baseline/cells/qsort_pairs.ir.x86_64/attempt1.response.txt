export function l $qsort_pairs(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @ret0, @setup
@ret0
	%z0 =l copy 0
	ret %z0
@setup
	%nm1 =l sub %n, 1
	%mid =l div %nm1, 2
	%moff =l mul %mid, 16
	%mp =l add %a, %moff
	%pk =l loadl %mp
	%mvp =l add %mp, 8
	%pv =l loadl %mvp
	%i =l copy 0
	%j =l copy %nm1
	jmp @while1
@while1
	%ioff =l mul %i, 16
	%ip =l add %a, %ioff
	%ik =l loadl %ip
	%ivp =l add %ip, 8
	%iv =l loadl %ivp
	%ikl =w csltl %ik, %pk
	jnz %ikl, @inci, @while1eq
@while1eq
	%ike =w ceql %ik, %pk
	jnz %ike, @while1val, @while2
@while1val
	%ivl =w csltl %iv, %pv
	jnz %ivl, @inci, @while2
@inci
	%i =l add %i, 1
	jmp @while1
@while2
	%joff =l mul %j, 16
	%jp =l add %a, %joff
	%jk =l loadl %jp
	%jvp =l add %jp, 8
	%jv =l loadl %jvp
	%jkg =w csltl %pk, %jk
	jnz %jkg, @decj, @while2eq
@while2eq
	%jke =w ceql %jk, %pk
	jnz %jke, @while2val, @check
@while2val
	%jvg =w csltl %pv, %jv
	jnz %jvg, @decj, @check
@decj
	%j =l sub %j, 1
	jmp @while2
@check
	%lt =w csltl %i, %j
	jnz %lt, @swap, @partdone
@swap
	%sioff =l mul %i, 16
	%sip =l add %a, %sioff
	%sivp =l add %sip, 8
	%sik =l loadl %sip
	%siv =l loadl %sivp
	%sjoff =l mul %j, 16
	%sjp =l add %a, %sjoff
	%sjvp =l add %sjp, 8
	%sjk =l loadl %sjp
	%sjv =l loadl %sjvp
	storel %sjk, %sip
	storel %sjv, %sivp
	storel %sik, %sjp
	storel %siv, %sjvp
	%i =l add %i, 1
	%j =l sub %j, 1
	jmp @while1
@partdone
	%left =l add %j, 1
	%right =l sub %n, %left
	%leftsmall =w csltl %left, 2
	jnz %leftsmall, @afterleft, @callleft
@callleft
	%rl =l call $qsort_pairs(l %a, l %left)
	jmp @afterleft
@afterleft
	%rightsmall =w csltl %right, 2
	jnz %rightsmall, @ret1, @callright
@callright
	%roff =l mul %left, 16
	%rptr =l add %a, %roff
	%rr =l call $qsort_pairs(l %rptr, l %right)
	jmp @ret1
@ret1
	%z1 =l copy 0
	ret %z1
}
