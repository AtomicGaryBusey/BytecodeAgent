export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	%zero =l copy 0
	%one =l copy 1
	%m32 =l copy 4294967295
	jmp @zhead
@zhead
	%zi =l phi @start %zero, @zbody %zi1
	%zc =w csltl %zi, %n
	jnz %zc, @zbody, @zero_done
@zbody
	%zoff =l shl %zi, 3
	%zaddr =l add %r, %zoff
	storel %zero, %zaddr
	%zi1 =l add %zi, %one
	jmp @zhead
@zero_done
	jmp @ohead
@ohead
	%oi =l phi @zero_done %zero, @ostore %oi1
	%oc =w csltl %oi, %na
	jnz %oc, @obody, @scan_init
@obody
	%aoff =l shl %oi, 3
	%aaddr =l add %a, %aoff
	%ai =l loadl %aaddr
	%x0 =l and %ai, %m32
	%x1 =l shr %ai, 32
	jmp @ihead
@ihead
	%j =l phi @obody %zero, @ibody %j1
	%carry =l phi @obody %zero, @ibody %carry1
	%jc =w csltl %j, %nb
	jnz %jc, @ibody, @ostore
@ibody
	%boff =l shl %j, 3
	%baddr =l add %b, %boff
	%bj =l loadl %baddr
	%y0 =l and %bj, %m32
	%y1 =l shr %bj, 32
	%p0 =l mul %x0, %y0
	%p1 =l mul %x0, %y1
	%p2 =l mul %x1, %y0
	%p3 =l mul %x1, %y1
	%mid =l add %p1, %p2
	%cmidw =w cultl %mid, %p1
	%cmid =l extuw %cmidw
	%z1lo =l shr %mid, 32
	%cmid32 =l shl %cmid, 32
	%z1hi =l add %z1lo, %cmid32
	%midsh =l shl %mid, 32
	%plo =l add %p0, %midsh
	%cloww =w cultl %plo, %p0
	%clow =l extuw %cloww
	%phit =l add %p3, %z1hi
	%phi =l add %phit, %clow
	%ij =l add %oi, %j
	%roff =l shl %ij, 3
	%raddr =l add %r, %roff
	%old =l loadl %raddr
	%s1 =l add %plo, %old
	%c1w =w cultl %s1, %plo
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	storel %s2, %raddr
	%c1 =l extuw %c1w
	%c2 =l extuw %c2w
	%cadd =l add %c1, %c2
	%carry1 =l add %phi, %cadd
	%j1 =l add %j, %one
	jmp @ihead
@ostore
	%idx =l add %oi, %nb
	%off2 =l shl %idx, 3
	%addr2 =l add %r, %off2
	storel %carry, %addr2
	%oi1 =l add %oi, %one
	jmp @ohead
@scan_init
	%kinit =l copy %n
	jmp @shead
@shead
	%k =l phi @scan_init %kinit, @sdec %k1
	%kzero =w ceql %k, %zero
	jnz %kzero, @done, @sbody
@sbody
	%km1 =l sub %k, %one
	%soff =l shl %km1, 3
	%saddr =l add %r, %soff
	%val =l loadl %saddr
	%vzero =w ceql %val, %zero
	jnz %vzero, @sdec, @done
@sdec
	%k1 =l copy %km1
	jmp @shead
@done
	ret %k
}
