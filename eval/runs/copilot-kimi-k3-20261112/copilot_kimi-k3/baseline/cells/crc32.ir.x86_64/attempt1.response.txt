export function l $crc32(l %p, l %n) {
@start
	jmp @outer_test
@outer_test
	%i =l phi @start 0, @inner_done %i_next
	%crc0 =w phi @start -1, @inner_done %crc
	%cond =w csltl %i, %n
	jnz %cond, @outer_body, @done
@outer_body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc1 =w xor %crc0, %b
	jmp @inner_test
@inner_test
	%j =w phi @outer_body 0, @inner_join %j_next
	%crc =w phi @outer_body %crc1, @inner_join %crc_next
	%jcond =w csltw %j, 8
	jnz %jcond, @inner_body, @inner_done
@inner_body
	%low =w and %crc, 1
	jnz %low, @odd, @even
@even
	%crc_even =w shr %crc, 1
	jmp @inner_join
@odd
	%shr =w shr %crc, 1
	%crc_odd =w xor %shr, -306674912
	jmp @inner_join
@inner_join
	%crc_next =w phi @even %crc_even, @odd %crc_odd
	%j_next =w add %j, 1
	jmp @inner_test
@inner_done
	%i_next =l add %i, 1
	jmp @outer_test
@done
	%outw =w xor %crc0, -1
	%out =l extuw %outw
	ret %out
}
