export function l $bubble_sort(l %a, l %n) {
@start
	%i =l alloc8 8
	%j =l alloc8 8
	%lim =l alloc8 8
	%zero =l copy 0
	%nm1 =l sub %n, 1
	storel %zero, %i
	storel %nm1, %lim
@outer_cond
	%iv =l loadl %i
	%lv =l loadl %lim
	%oc =w csltl %iv, %lv
	jnz %oc, @outer_body, @done
@outer_body
	storel %zero, %j
	%innermax =l sub %lv, %iv
	jmp @inner_cond
@inner_cond
	%jv =l loadl %j
	%ic =w csltl %jv, %innermax
	jnz %ic, @inner_body, @outer_next
@inner_body
	%off =l mul %jv, 8
	%p0 =l add %a, %off
	%p1 =l add %p0, 8
	%v0 =l loadl %p0
	%v1 =l loadl %p1
	%sw =w csltl %v1, %v0
	jnz %sw, @do_swap, @inner_next
@do_swap
	storel %v1, %p0
	storel %v0, %p1
	jmp @inner_next
@inner_next
	%jv2 =l loadl %j
	%jnext =l add %jv2, 1
	storel %jnext, %j
	jmp @inner_cond
@outer_next
	%iv2 =l loadl %i
	%inext =l add %iv2, 1
	storel %inext, %i
	jmp @outer_cond
@done
	ret 0
}
