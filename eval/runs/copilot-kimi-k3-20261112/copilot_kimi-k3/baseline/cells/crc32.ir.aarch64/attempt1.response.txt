export function l $crc32(l %p, l %n) {
@start
	%crc =w copy 4294967295
	%i =l copy 0
@loop
	%done =w csgel %i, %n
	jnz %done, @final, @body
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%crc =w xor %crc, %b
	%k =w copy 0
@bitloop
	%kd =w csgew %k, 8
	jnz %kd, @next, @bitbody
@bitbody
	%low =w and %crc, 1
	%shr =w shr %crc, 1
	%mask =w sub 0, %low
	%t =w and %mask, 3988292384
	%crc =w xor %shr, %t
	%k =w add %k, 1
	jmp @bitloop
@next
	%i =l add %i, 1
	jmp @loop
@final
	%r =w xor %crc, 4294967295
	%res =l extuw %r
	ret %res
}
