export function l $bmin_empty(l %a, l %n) {
@start
	%iszero =w ceql %n, 0
	jnz %iszero, @empty, @nonempty
@empty
	%maxv =l copy 9223372036854775807
	ret %maxv
@nonempty
	%min =l loadl %a
	%i =l copy 1
@loop
	%cont =w csltl %i, %n
	jnz %cont, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%lt =w csltl %v, %min
	jnz %lt, @update, @next
@update
	%min =l copy %v
	jmp @next
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %min
}
