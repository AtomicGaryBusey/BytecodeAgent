mp_mul:
    push rbx
    push rbp
    push r12
    push r13
    mov rbx, rdx
    mov r9, rsi
    add r9, rcx
    xor r10d, r10d
.Lzero:
    cmp r10, r9
    jge .Lzero_done
    mov qword ptr [r8 + r10*8], 0
    inc r10
    jmp .Lzero
.Lzero_done:
    xor r10d, r10d
.Louter:
    cmp r10, rsi
    jge .Lnorm_setup
    mov rbp, qword ptr [rdi + r10*8]
    xor r12d, r12d
    xor r11d, r11d
.Linner:
    cmp r11, rcx
    jge .Linner_done
    mov rax, rbp
    mul qword ptr [rbx + r11*8]
    add rax, r12
    adc rdx, 0
    lea r13, [r10 + r11]
    add rax, qword ptr [r8 + r13*8]
    adc rdx, 0
    mov qword ptr [r8 + r13*8], rax
    mov r12, rdx
    inc r11
    jmp .Linner
.Linner_done:
    lea r13, [r10 + rcx]
    mov qword ptr [r8 + r13*8], r12
    inc r10
    jmp .Louter
.Lnorm_setup:
    mov rax, r9
.Lnorm:
    test rax, rax
    je .Ldone
    cmp qword ptr [r8 + rax*8 - 8], 0
    jne .Ldone
    dec rax
    jmp .Lnorm
.Ldone:
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
