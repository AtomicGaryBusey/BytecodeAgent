binary_search:
	xor r8d, r8d
	lea r9, [rsi - 1]
.Lloop:
	cmp r8, r9
	jg .Lnotfound
	mov rcx, r9
	sub rcx, r8
	shr rcx, 1
	add rcx, r8
	mov rax, qword ptr [rdi + rcx*8]
	cmp rax, rdx
	je .Lfound
	jl .Llower
	lea r9, [rcx - 1]
	jmp .Lloop
.Llower:
	lea r8, [rcx + 1]
	jmp .Lloop
.Lfound:
	mov rax, rcx
	ret
.Lnotfound:
	mov rax, -1
	ret
