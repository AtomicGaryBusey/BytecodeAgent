packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lnext:
    cmp r8, rsi
    jge .Ldone
    movsx eax, byte ptr [rdi + r8]
    inc r8
    test eax, eax
    js .Lneg
    lea r10d, [eax + 1]
    mov r11, rsi
    sub r11, r8
    cmp r11, r10
    jl .Lsrcerr
    mov r11, rcx
    sub r11, r9
    cmp r10, r11
    jg .Llitover
.Llitcopy:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r10
    jnz .Llitcopy
    jmp .Lnext
.Llitover:
    test r11, r11
    jz .Lover
.Llitover_copy:
    mov al, byte ptr [rdi + r8]
    mov byte ptr [rdx + r9], al
    inc r8
    inc r9
    dec r11
    jnz .Llitover_copy
    jmp .Lover
.Lneg:
    cmp eax, -128
    je .Lnext
    mov r10d, 1
    sub r10d, eax
    cmp r8, rsi
    jge .Lsrcerr
    mov al, byte ptr [rdi + r8]
    inc r8
    mov r11, rcx
    sub r11, r9
    cmp r10, r11
    jg .Lrepover
.Lrepfill:
    mov byte ptr [rdx + r9], al
    inc r9
    dec r10
    jnz .Lrepfill
    jmp .Lnext
.Lrepover:
    test r11, r11
    jz .Lover
.Lrepover_fill:
    mov byte ptr [rdx + r9], al
    inc r9
    dec r11
    jnz .Lrepover_fill
.Lover:
    mov rax, -1
    ret
.Lsrcerr:
    mov rax, -2
    ret
.Ldone:
    mov rax, r9
    ret
