export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%i =l copy 0
	%min =l copy 0
	%max =l copy 0
	%sum =l copy 0
	%cnt =l copy 0
	jmp @loop
@loop
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	%first =w ceql %i, 0
	jnz %first, @init, @upd
@init
	%min =l copy %y
	%max =l copy %y
	jmp @acc
@upd
	%lt =w csltl %y, %min
	jnz %lt, @setmin, @chkmax
@setmin
	%min =l copy %y
	jmp @chkmax
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @setmax, @acc
@setmax
	%max =l copy %y
	jmp @acc
@acc
	%sum =l add %sum, %y
	%odd =l and %y, 1
	%cnt =l add %cnt, %odd
	%i =l add %i, 1
	jmp @loop
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
