export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	jmp @loop(0, 0, 0, 0, 0, 1)
@loop(l %i, l %min, l %max, l %sum, l %cnt, w %first)
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%y =l call %f(l %x)
	jnz %first, @init, @upd
@init
	jmp @minmax(%y, %y)
@upd
	%lt =w csltl %y, %min
	jnz %lt, @minmax_ymin, @chkmax
@minmax_ymin
	jmp @minmax(%y, %max)
@chkmax
	%gt =w csgtl %y, %max
	jnz %gt, @minmax_ymax, @minmax_keep
@minmax_ymax
	jmp @minmax(%min, %y)
@minmax_keep
	jmp @minmax(%min, %max)
@minmax(l %mn, l %mx)
	%nsum =l add %sum, %y
	%odd =l and %y, 1
	%ncnt =l add %cnt, %odd
	%ni =l add %i, 1
	jmp @loop(%ni, %mn, %mx, %nsum, %ncnt, 0)
@end
	storel %min, %out
	%o1 =l add %out, 8
	storel %max, %o1
	%o2 =l add %out, 16
	storel %sum, %o2
	ret %cnt
}
