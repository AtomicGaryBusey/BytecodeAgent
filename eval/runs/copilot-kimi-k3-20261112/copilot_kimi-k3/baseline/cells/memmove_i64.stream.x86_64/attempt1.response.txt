memmove_i64:
mov rax, 0
cmp rdx, 0
jle .Ldone
cmp rdi, rsi
je .Ldone
jb .Lforward
mov rcx, rdx
.Lbackward_loop:
mov r8, qword ptr [rsi + rcx*8 - 8]
mov qword ptr [rdi + rcx*8 - 8], r8
sub rcx, 1
jnz .Lbackward_loop
jmp .Ldone
.Lforward:
mov rcx, 0
.Lforward_loop:
mov r8, qword ptr [rsi + rcx*8]
mov qword ptr [rdi + rcx*8], r8
add rcx, 1
cmp rcx, rdx
jb .Lforward_loop
.Ldone:
ret
