packbits_encode:
xor r8d, r8d
xor r9d, r9d
.Louter:
cmp r8, rsi
jge .Ldone
movzx eax, byte ptr [rdi + r8]
lea r10, [r8 + 1]
lea r11, [r8 + 128]
cmp r11, rsi
cmovg r11, rsi
.Lrun:
cmp r10, r11
jge .Lrundone
cmp al, byte ptr [rdi + r10]
jne .Lrundone
inc r10
jmp .Lrun
.Lrundone:
sub r10, r8
cmp r10, 3
jl .Lliteral
lea r11, [r9 + 2]
cmp r11, rcx
jg .Lfail
mov r11d, eax
mov eax, 257
sub rax, r10
mov byte ptr [rdx + r9], al
mov byte ptr [rdx + r9 + 1], r11b
add r9, 2
add r8, r10
jmp .Louter
.Lliteral:
lea r10, [r8 + 1]
.Llitloop:
cmp r10, rsi
jge .Llitdone
mov rax, r10
sub rax, r8
cmp rax, 128
jge .Llitdone
lea rax, [r10 + 2]
cmp rax, rsi
jge .Llitnext
movzx eax, byte ptr [rdi + r10]
cmp al, byte ptr [rdi + r10 + 1]
jne .Llitnext
cmp al, byte ptr [rdi + r10 + 2]
jne .Llitnext
jmp .Llitdone
.Llitnext:
inc r10
jmp .Llitloop
.Llitdone:
sub r10, r8
lea rax, [r9 + r10 + 1]
cmp rax, rcx
jg .Lfail
lea rax, [r10 - 1]
mov byte ptr [rdx + r9], al
inc r9
xor r11d, r11d
.Lcopy:
cmp r11, r10
jge .Lcopydone
lea rax, [r8 + r11]
movzx eax, byte ptr [rdi + rax]
mov byte ptr [rdx + r9], al
inc r9
inc r11
jmp .Lcopy
.Lcopydone:
add r8, r10
jmp .Louter
.Ldone:
mov rax, r9
ret
.Lfail:
mov rax, -1
ret
