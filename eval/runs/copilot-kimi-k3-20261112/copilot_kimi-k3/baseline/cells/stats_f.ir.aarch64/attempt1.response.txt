export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
	%ip =l alloc8 8
	%minp =l alloc8 8
	%maxp =l alloc8 8
	%sump =l alloc8 8
	%cntp =l alloc8 8
	%x0 =l loadl %a
	%y0 =l call %f(l %x0)
	storel %y0, %minp
	storel %y0, %maxp
	storel %y0, %sump
	%o0 =l and %y0, 1
	storel %o0, %cntp
	storel 1, %ip
	jmp @loop
@loop
	%i =l loadl %ip
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%ap =l add %a, %off
	%x =l loadl %ap
	%y =l call %f(l %x)
	%mn =l loadl %minp
	%c1 =w csltl %y, %mn
	jnz %c1, @setmin, @chkmax
@setmin
	storel %y, %minp
	jmp @sum
@chkmax
	%mx =l loadl %maxp
	%c2 =w csltl %mx, %y
	jnz %c2, @setmax, @sum
@setmax
	storel %y, %maxp
	jmp @sum
@sum
	%s =l loadl %sump
	%s2 =l add %s, %y
	storel %s2, %sump
	%odd =l and %y, 1
	%cnt =l loadl %cntp
	%cnt2 =l add %cnt, %odd
	storel %cnt2, %cntp
	%i2 =l add %i, 1
	storel %i2, %ip
	jmp @loop
@end
	%rmn =l loadl %minp
	storel %rmn, %out
	%rmx =l loadl %maxp
	%o1 =l add %out, 8
	storel %rmx, %o1
	%rs =l loadl %sump
	%o2 =l add %out, 16
	storel %rs, %o2
	%rc =l loadl %cntp
	ret %rc
}
