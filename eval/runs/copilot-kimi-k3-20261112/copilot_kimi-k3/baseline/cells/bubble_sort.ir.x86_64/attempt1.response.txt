export function l $bubble_sort(l %a, l %n) {
@start
	%ip =l alloc8 8
	%jp =l alloc8 8
	storel 0, %ip
@outer
	%bound =l sub %n, 1
	%iv =l loadl %ip
	%done =w csgel %iv, %bound
	jnz %done, @ret, @inner
@inner
	storel 0, %jp
@inloop
	%jv =l loadl %jp
	%limit =l sub %bound, %iv
	%idone =w csgel %jv, %limit
	jnz %idone, @onext, @body
@body
	%j8 =l mul %jv, 8
	%p =l add %a, %j8
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%gt =w csgtl %x, %y
	jnz %gt, @swap, @inext
@swap
	storel %y, %p
	storel %x, %p1
	jmp @inext
@inext
	%jv2 =l add %jv, 1
	storel %jv2, %jp
	jmp @inloop
@onext
	%iv2 =l add %iv, 1
	storel %iv2, %ip
	jmp @outer
@ret
	ret 0
}
