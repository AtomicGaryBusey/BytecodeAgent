export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%pos =l alloc8 8
	%o =l alloc8 8
	%rem =l alloc8 8
	storel 0, %pos
	storel 0, %o
@loop
	%posv =l loadl %pos
	%cmain =w csltl %posv, %n
	jnz %cmain, @body, @done
@body
	%hp =l add %src, %posv
	%h =w loadsb %hp
	%posa =l add %posv, 1
	storel %posa, %pos
	%isneg =w csltw %h, 0
	jnz %isneg, @neg, @lit
@lit
	%hl =l extsw %h
	%count =l add %hl, 1
	%end =l add %posa, %count
	%cerr =w csgtl %end, %n
	jnz %cerr, @err2, @litloop
@litloop
	%posl =l loadl %pos
	%clit =w csltl %posl, %end
	jnz %clit, @litbody, @loop
@litbody
	%ov =l loadl %o
	%cfull =w csgel %ov, %cap
	jnz %cfull, @err1, @litwrite
@litwrite
	%sp =l add %src, %posl
	%b =w loadub %sp
	%op =l add %out, %ov
	storeb %b, %op
	%o1 =l add %ov, 1
	storel %o1, %o
	%posb =l add %posl, 1
	storel %posb, %pos
	jmp @litloop
@neg
	%isnop =w ceqw %h, -128
	jnz %isnop, @loop, @rep
@rep
	%posr =l loadl %pos
	%csrc =w csgel %posr, %n
	jnz %csrc, @err2, @repok
@repok
	%rp =l add %src, %posr
	%rb =w loadub %rp
	%hr =l extsw %h
	%rc =l sub 1, %hr
	storel %rc, %rem
	%posc =l add %posr, 1
	storel %posc, %pos
@reploop
	%remv =l loadl %rem
	%crem =w csgtl %remv, 0
	jnz %crem, @repbody, @loop
@repbody
	%ovr =l loadl %o
	%cfull2 =w csgel %ovr, %cap
	jnz %cfull2, @err1, @repwrite
@repwrite
	%opr =l add %out, %ovr
	storeb %rb, %opr
	%o2 =l add %ovr, 1
	storel %o2, %o
	%rem1 =l sub %remv, 1
	storel %rem1, %rem
	jmp @reploop
@done
	%res =l loadl %o
	ret %res
@err1
	ret -1
@err2
	ret -2
}
