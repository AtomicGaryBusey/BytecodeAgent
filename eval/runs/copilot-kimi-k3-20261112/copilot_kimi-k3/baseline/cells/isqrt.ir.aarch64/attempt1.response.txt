export function l $isqrt(l %n) {
@start
	%lo =l alloc8 8
	%hi =l alloc8 8
	storel 0, %lo
	storel 4294967296, %hi
@loop
	%lov =l loadl %lo
	%hiv =l loadl %hi
	%diff =l sub %hiv, %lov
	%c =w cugtl %diff, 1
	jnz %c, @body, @done
@body
	%sum =l add %lov, %hiv
	%mid =l shr %sum, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @setlo, @sethi
@setlo
	storel %mid, %lo
	jmp @loop
@sethi
	storel %mid, %hi
	jmp @loop
@done
	%r =l loadl %lo
	ret %r
}
