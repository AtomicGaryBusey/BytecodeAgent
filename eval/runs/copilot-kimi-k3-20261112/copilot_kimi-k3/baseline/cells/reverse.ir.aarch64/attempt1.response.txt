export function l $reverse(l %a, l %n) {
@entry
	%zero =l copy 0
	%j0 =l sub %n, 1
	%c0 =w csltl %zero, %j0
	jnz %c0, @loop, @ret
@loop
	%i =l phi @entry %zero, @body %inext
	%j =l phi @entry %j0, @body %jnext
	%io =l mul %i, 8
	%jo =l mul %j, 8
	%pi =l add %a, %io
	%pj =l add %a, %jo
	%vi =l loadl %pi
	%vj =l loadl %pj
	storel %vj, %pi
	storel %vi, %pj
	jmp @body
@body
	%inext =l add %i, 1
	%jnext =l sub %j, 1
	%c =w csltl %inext, %jnext
	jnz %c, @loop, @ret
@ret
	ret 0
}
