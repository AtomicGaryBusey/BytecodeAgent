export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	%frame =l alloc8 24
	%spp =l add %frame, 8
	%zp =l add %frame, 16
	storel 0, %frame
	storel 0, %spp
	jmp @loop

@loop
	%pc =l loadl %frame
	%sp =l loadl %spp
	%atend =w ceql %pc, %len
	jnz %atend, @finish, @fetch

@fetch
	%pcoff =l mul %pc, 8
	%codeaddr =l add %code, %pcoff
	%op =l loadl %codeaddr
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @check1

@check1
	%is1 =w ceql %op, 1
	jnz %is1, @push, @check2

@check2
	%is2 =w ceql %op, 2
	jnz %is2, @add, @check3

@check3
	%is3 =w ceql %op, 3
	jnz %is3, @sub, @check4

@check4
	%is4 =w ceql %op, 4
	jnz %is4, @mul, @check5

@check5
	%is5 =w ceql %op, 5
	jnz %is5, @dup, @check6

@check6
	%is6 =w ceql %op, 6
	jnz %is6, @swap, @check7

@check7
	%is7 =w ceql %op, 7
	jnz %is7, @drop, @check8

@check8
	%is8 =w ceql %op, 8
	jnz %is8, @over, @check9

@check9
	%is9 =w ceql %op, 9
	jnz %is9, @lt, @check10

@check10
	%is10 =w ceql %op, 10
	jnz %is10, @jz, @check11

@check11
	%is11 =w ceql %op, 11
	jnz %is11, @jmpop, @other

@push
	%push_pc1 =l add %pc, 1
	%push_opoff =l mul %push_pc1, 8
	%push_opaddr =l add %code, %push_opoff
	%push_val =l loadl %push_opaddr
	%push_off =l mul %sp, 8
	%push_addr =l add %stack, %push_off
	storel %push_val, %push_addr
	%push_newsp =l add %sp, 1
	storel %push_newsp, %spp
	%push_newpc =l add %pc, 2
	storel %push_newpc, %frame
	jmp @loop

@add
	%add_topi =l sub %sp, 1
	%add_ai =l sub %sp, 2
	%add_topoff =l mul %add_topi, 8
	%add_aoff =l mul %add_ai, 8
	%add_topaddr =l add %stack, %add_topoff
	%add_aaddr =l add %stack, %add_aoff
	%add_b =l loadl %add_topaddr
	%add_a =l loadl %add_aaddr
	%add_res =l add %add_a, %add_b
	storel %add_res, %add_aaddr
	storel %add_topi, %spp
	%add_newpc =l add %pc, 1
	storel %add_newpc, %frame
	jmp @loop

@sub
	%sub_topi =l sub %sp, 1
	%sub_ai =l sub %sp, 2
	%sub_topoff =l mul %sub_topi, 8
	%sub_aoff =l mul %sub_ai, 8
	%sub_topaddr =l add %stack, %sub_topoff
	%sub_aaddr =l add %stack, %sub_aoff
	%sub_b =l loadl %sub_topaddr
	%sub_a =l loadl %sub_aaddr
	%sub_res =l sub %sub_a, %sub_b
	storel %sub_res, %sub_aaddr
	storel %sub_topi, %spp
	%sub_newpc =l add %pc, 1
	storel %sub_newpc, %frame
	jmp @loop

@mul
	%mul_topi =l sub %sp, 1
	%mul_ai =l sub %sp, 2
	%mul_topoff =l mul %mul_topi, 8
	%mul_aoff =l mul %mul_ai, 8
	%mul_topaddr =l add %stack, %mul_topoff
	%mul_aaddr =l add %stack, %mul_aoff
	%mul_b =l loadl %mul_topaddr
	%mul_a =l loadl %mul_aaddr
	%mul_res =l mul %mul_a, %mul_b
	storel %mul_res, %mul_aaddr
	storel %mul_topi, %spp
	%mul_newpc =l add %pc, 1
	storel %mul_newpc, %frame
	jmp @loop

@dup
	%dup_topi =l sub %sp, 1
	%dup_topoff =l mul %dup_topi, 8
	%dup_newoff =l mul %sp, 8
	%dup_topaddr =l add %stack, %dup_topoff
	%dup_newaddr =l add %stack, %dup_newoff
	%dup_val =l loadl %dup_topaddr
	storel %dup_val, %dup_newaddr
	%dup_newsp =l add %sp, 1
	storel %dup_newsp, %spp
	%dup_newpc =l add %pc, 1
	storel %dup_newpc, %frame
	jmp @loop

@swap
	%swap_topi =l sub %sp, 1
	%swap_ai =l sub %sp, 2
	%swap_topoff =l mul %swap_topi, 8
	%swap_aoff =l mul %swap_ai, 8
	%swap_topaddr =l add %stack, %swap_topoff
	%swap_aaddr =l add %stack, %swap_aoff
	%swap_b =l loadl %swap_topaddr
	%swap_a =l loadl %swap_aaddr
	storel %swap_b, %swap_aaddr
	storel %swap_a, %swap_topaddr
	%swap_newpc =l add %pc, 1
	storel %swap_newpc, %frame
	jmp @loop

@drop
	%drop_newsp =l sub %sp, 1
	storel %drop_newsp, %spp
	%drop_newpc =l add %pc, 1
	storel %drop_newpc, %frame
	jmp @loop

@over
	%over_ai =l sub %sp, 2
	%over_aoff =l mul %over_ai, 8
	%over_newoff =l mul %sp, 8
	%over_aaddr =l add %stack, %over_aoff
	%over_newaddr =l add %stack, %over_newoff
	%over_val =l loadl %over_aaddr
	storel %over_val, %over_newaddr
	%over_newsp =l add %sp, 1
	storel %over_newsp, %spp
	%over_newpc =l add %pc, 1
	storel %over_newpc, %frame
	jmp @loop

@lt
	%lt_topi =l sub %sp, 1
	%lt_ai =l sub %sp, 2
	%lt_topoff =l mul %lt_topi, 8
	%lt_aoff =l mul %lt_ai, 8
	%lt_topaddr =l add %stack, %lt_topoff
	%lt_aaddr =l add %stack, %lt_aoff
	%lt_b =l loadl %lt_topaddr
	%lt_a =l loadl %lt_aaddr
	%lt_cmp =w csltl %lt_a, %lt_b
	%lt_res =l extsw %lt_cmp
	storel %lt_res, %lt_aaddr
	storel %lt_topi, %spp
	%lt_newpc =l add %pc, 1
	storel %lt_newpc, %frame
	jmp @loop

@jz
	%jz_pc1 =l add %pc, 1
	%jz_toff =l mul %jz_pc1, 8
	%jz_taddr =l add %code, %jz_toff
	%jz_target =l loadl %jz_taddr
	%jz_topi =l sub %sp, 1
	%jz_voff =l mul %jz_topi, 8
	%jz_vaddr =l add %stack, %jz_voff
	%jz_val =l loadl %jz_vaddr
	storel %jz_topi, %spp
	%jz_iszero =w ceql %jz_val, 0
	jnz %jz_iszero, @jz_take, @jz_fall

@jz_take
	storel %jz_target, %frame
	jmp @loop

@jz_fall
	%jz_fallpc =l add %pc, 2
	storel %jz_fallpc, %frame
	jmp @loop

@jmpop
	%jmp_pc1 =l add %pc, 1
	%jmp_toff =l mul %jmp_pc1, 8
	%jmp_taddr =l add %code, %jmp_toff
	%jmp_target =l loadl %jmp_taddr
	storel %jmp_target, %frame
	jmp @loop

@other
	%other_newpc =l add %pc, 1
	storel %other_newpc, %frame
	jmp @loop

@finish
	%final_sp =l loadl %spp
	storel %final_sp, %zp
	jmp @zero

@zero
	%zi =l loadl %zp
	%zdone =w ceql %zi, %cap
	jnz %zdone, @ret, @zero_body

@zero_body
	%zoff =l mul %zi, 8
	%zaddr =l add %stack, %zoff
	storel 0, %zaddr
	%znext =l add %zi, 1
	storel %znext, %zp
	jmp @zero

@ret
	%ret_sp =l loadl %spp
	%ret_empty =w ceql %ret_sp, 0
	jnz %ret_empty, @ret_zero, @ret_top

@ret_top
	%ret_i =l sub %ret_sp, 1
	%ret_off =l mul %ret_i, 8
	%ret_addr =l add %stack, %ret_off
	%ret_val =l loadl %ret_addr
	ret %ret_val

@ret_zero
	ret 0
}
