export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%pi =l alloc8 8
	%pw =l alloc8 8
	%pr =l alloc8 8
	%pj =l alloc8 8
	%prj =l alloc8 8
	%pk =l alloc8 8
	storel 0, %pi
	storel 0, %pw
@loop
	%i0 =l loadl %pi
	%c0 =w csltl %i0, %n
	jnz %c0, @body, @done
@body
	%sp0 =l add %src, %i0
	%b0 =w loadub %sp0
	storel 1, %pr
@rloop
	%r0 =l loadl %pr
	%ir =l add %i0, %r0
	%c1 =w csltl %ir, %n
	jnz %c1, @rchk2, @rend
@rchk2
	%c2 =w csltl %r0, 128
	jnz %c2, @rchk3, @rend
@rchk3
	%sp1 =l add %src, %ir
	%b1 =w loadub %sp1
	%c3 =w ceqw %b1, %b0
	jnz %c3, @rinc, @rend
@rinc
	%r1 =l add %r0, 1
	storel %r1, %pr
	jmp @rloop
@rend
	%r2 =l loadl %pr
	%c4 =w csltl %r2, 3
	jnz %c4, @lit, @rep
@rep
	%w0 =l loadl %pw
	%wn =l add %w0, 2
	%c5 =w csgtl %wn, %cap
	jnz %c5, @fail, @repw
@repw
	%hdr =l sub 257, %r2
	%op0 =l add %out, %w0
	storeb %hdr, %op0
	%op1 =l add %op0, 1
	storeb %b0, %op1
	storel %wn, %pw
	%in =l add %i0, %r2
	storel %in, %pi
	jmp @loop
@lit
	storel %i0, %pj
@lloop
	%j0 =l loadl %pj
	%c6 =w csltl %j0, %n
	jnz %c6, @lchk2, @lend
@lchk2
	%d0 =l sub %j0, %i0
	%c7 =w csltl %d0, 128
	jnz %c7, @lrun, @lend
@lrun
	%sp2 =l add %src, %j0
	%bj =w loadub %sp2
	storel 1, %prj
@lrloop
	%rj0 =l loadl %prj
	%jr =l add %j0, %rj0
	%c8 =w csltl %jr, %n
	jnz %c8, @lrchk2, @lrend
@lrchk2
	%c9 =w csltl %rj0, 128
	jnz %c9, @lrchk3, @lrend
@lrchk3
	%sp3 =l add %src, %jr
	%b2 =w loadub %sp3
	%c10 =w ceqw %b2, %bj
	jnz %c10, @lrinc, @lrend
@lrinc
	%rj1 =l add %rj0, 1
	storel %rj1, %prj
	jmp @lrloop
@lrend
	%rj2 =l loadl %prj
	%c11 =w csltl %rj2, 3
	jnz %c11, @linc, @lend
@linc
	%j1 =l add %j0, 1
	storel %j1, %pj
	jmp @lloop
@lend
	%j2 =l loadl %pj
	%L =l sub %j2, %i0
	%need =l add %L, 1
	%w1 =l loadl %pw
	%wt =l add %w1, %need
	%c12 =w csgtl %wt, %cap
	jnz %c12, @fail, @lwr
@lwr
	%hdr2 =l sub %L, 1
	%op2 =l add %out, %w1
	storeb %hdr2, %op2
	%wsrc =l add %w1, 1
	storel 0, %pk
@cloop
	%k0 =l loadl %pk
	%c13 =w csltl %k0, %L
	jnz %c13, @cbody, @cend
@cbody
	%ik =l add %i0, %k0
	%sp4 =l add %src, %ik
	%bv =w loadub %sp4
	%wk =l add %wsrc, %k0
	%op3 =l add %out, %wk
	storeb %bv, %op3
	%k1 =l add %k0, 1
	storel %k1, %pk
	jmp @cloop
@cend
	storel %wt, %pw
	storel %j2, %pi
	jmp @loop
@fail
	ret -1
@done
	%wf =l loadl %pw
	ret %wf
}
