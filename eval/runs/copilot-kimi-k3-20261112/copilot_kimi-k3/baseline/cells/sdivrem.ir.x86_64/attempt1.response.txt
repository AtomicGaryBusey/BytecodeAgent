export function l $sdivrem(l %a, l %b, l %rem) {
@entry
	%bz =w ceql %b, 0
	jnz %bz, @bzero, @chkminus
@bzero
	storel %a, %rem
	ret 0
@chkminus
	%bp1 =l add %b, 1
	%bm1 =w ceql %bp1, 0
	jnz %bm1, @chkmin, @divide
@chkmin
	%az =w ceql %a, 0
	jnz %az, @divide, @selfneg
@selfneg
	%nega =l sub 0, %a
	%same =w ceql %a, %nega
	jnz %same, @overflow, @divide
@overflow
	%zero =l copy 0
	storel %zero, %rem
	ret %a
@divide
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
