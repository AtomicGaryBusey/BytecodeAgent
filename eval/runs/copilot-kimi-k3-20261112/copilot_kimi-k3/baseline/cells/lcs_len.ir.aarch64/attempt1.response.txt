export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	%zero =l copy 0
	%one =l copy 1
	%eight =l copy 8
	%np1 =l add %n, %one
	storel %zero, %row
	jmp @zero_head
@zero_head
	%j0 =l phi @start %one, @zero_body %j0next
	%zcond =w csltl %j0, %np1
	jnz %zcond, @zero_body, @outer_head
@zero_body
	%zoff =l mul %j0, %eight
	%zaddr =l add %row, %zoff
	storel %zero, %zaddr
	%j0next =l add %j0, %one
	jmp @zero_head
@outer_head
	%i =l phi @zero_head %zero, @outer_cont %inext
	%icond =w csltl %i, %m
	jnz %icond, @outer_body, @done
@outer_body
	%xaddr =l add %x, %i
	%xb =w loadub %xaddr
	jmp @inner_head
@inner_head
	%j =l phi @outer_body %one, @inner_cont %jnext
	%diag =l phi @outer_body %zero, @inner_cont %dnext
	%jcond =w csltl %j, %np1
	jnz %jcond, @inner_body, @outer_cont
@inner_body
	%off =l mul %j, %eight
	%addr =l add %row, %off
	%up =l loadl %addr
	%prevoff =l sub %off, %eight
	%prevaddr =l add %row, %prevoff
	%left =l loadl %prevaddr
	%ym1 =l sub %j, %one
	%yaddr =l add %y, %ym1
	%yb =w loadub %yaddr
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%new =l add %diag, %one
	storel %new, %addr
	jmp @inner_cont
@nomatch
	storel %up, %addr
	%leftgt =w csgtl %left, %up
	jnz %leftgt, @storeleft, @inner_cont
@storeleft
	storel %left, %addr
	jmp @inner_cont
@inner_cont
	%dnext =l copy %up
	%jnext =l add %j, %one
	jmp @inner_head
@outer_cont
	%inext =l add %i, %one
	jmp @outer_head
@done
	%noff =l mul %n, %eight
	%naddr =l add %row, %noff
	%res =l loadl %naddr
	ret %res
}
