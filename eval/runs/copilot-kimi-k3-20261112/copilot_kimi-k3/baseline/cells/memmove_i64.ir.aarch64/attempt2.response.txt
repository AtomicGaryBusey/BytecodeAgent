export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%is0 =w ceql %n, 0
	jnz %is0, @ret, @dir
@dir
	%fwd =w cultl %dst, %src
	jnz %fwd, @f_init, @b_init
@f_init
	jmp @f_head
@f_head
	%i =l phi @f_init 0, @f_body %inext
	%fdone =w ceql %i, %n
	jnz %fdone, @ret, @f_body
@f_body
	%foff =l mul %i, 8
	%fsrc =l add %src, %foff
	%fdst =l add %dst, %foff
	%fv =l loadl %fsrc
	storel %fv, %fdst
	%inext =l add %i, 1
	jmp @f_head
@b_init
	jmp @b_head
@b_head
	%j =l phi @b_init %n, @b_body %idx
	%bdone =w ceql %j, 0
	jnz %bdone, @ret, @b_body
@b_body
	%idx =l sub %j, 1
	%boff =l mul %idx, 8
	%bsrc =l add %src, %boff
	%bdst =l add %dst, %boff
	%bv =l loadl %bsrc
	storel %bv, %bdst
	jmp @b_head
@ret
	ret 0
}
