lcs_len:
push rbx
xor r9d, r9d
.Lzero:
cmp r9, rcx
jg .Lzero_done
mov qword ptr [r8 + r9*8], 0
inc r9
jmp .Lzero
.Lzero_done:
xor r10d, r10d
.Louter:
cmp r10, rsi
jge .Ldone
movzx r11d, byte ptr [rdi + r10]
mov r9, 1
xor eax, eax
.Linner:
cmp r9, rcx
jg .Linner_done
mov rbx, qword ptr [r8 + r9*8]
cmp byte ptr [rdx + r9 - 1], r11b
je .Lmatch
mov rax, rbx
cmp rbx, qword ptr [r8 + r9*8 - 8]
jge .Lmism_store
mov rbx, qword ptr [r8 + r9*8 - 8]
.Lmism_store:
mov qword ptr [r8 + r9*8], rbx
jmp .Lnext
.Lmatch:
inc rax
mov qword ptr [r8 + r9*8], rax
mov rax, rbx
.Lnext:
inc r9
jmp .Linner
.Linner_done:
inc r10
jmp .Louter
.Ldone:
mov rax, qword ptr [r8 + rcx*8]
pop rbx
ret
