export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%i =l alloc8 8
	%written =l alloc8 8
	%j =l alloc8 8
	%rp =l alloc8 8
	%rr =l alloc8 8
	%rk =l alloc8 8
	%cp =l alloc8 8
	%dst =l alloc8 8
	storel 0, %i
	storel 0, %written
	jmp @main_loop

@main_loop
	%iv =l loadl %i
	%more =w csltl %iv, %n
	jnz %more, @have, @ret_written

@have
	storel %iv, %rp
	storel 1, %rr
	%k0 =l add %iv, 1
	storel %k0, %rk
	jmp @run_loop

@run_loop
	%k =l loadl %rk
	%kin =w csltl %k, %n
	jnz %kin, @run_chkcap, @run_done

@run_chkcap
	%r0 =l loadl %rr
	%rcap =w csltl %r0, 128
	jnz %rcap, @run_cmp, @run_done

@run_cmp
	%rv0 =l loadl %rp
	%pb =l add %src, %rv0
	%b0 =w loadub %pb
	%pk =l add %src, %k
	%bk =w loadub %pk
	%beq =w ceqw %bk, %b0
	jnz %beq, @run_inc, @run_done

@run_inc
	%r1 =l loadl %rr
	%r2 =l add %r1, 1
	storel %r2, %rr
	%k1 =l add %k, 1
	storel %k1, %rk
	jmp @run_loop

@run_done
	%rv =l loadl %rr
	%islit =w csltl %rv, 3
	jnz %islit, @literal, @repeat

@repeat
	%wv =l loadl %written
	%needr =l add %wv, 2
	%nofitr =w csltl %cap, %needr
	jnz %nofitr, @ret_neg, @rep_fit

@rep_fit
	%rw =w copy %rv
	%hdr =w sub 257, %rw
	%p0 =l add %out, %wv
	storeb %hdr, %p0
	%ps =l add %src, %iv
	%bb =w loadub %ps
	%p1 =l add %p0, 1
	storeb %bb, %p1
	storel %needr, %written
	%in =l add %iv, %rv
	storel %in, %i
	jmp @main_loop

@literal
	storel %iv, %j
	jmp @lit_loop

@lit_loop
	%jv =l loadl %j
	%jin =w csltl %jv, %n
	jnz %jin, @lit_chk_len, @lit_end

@lit_chk_len
	%len =l sub %jv, %iv
	%lcap =w csltl %len, 128
	jnz %lcap, @lit_run, @lit_end

@lit_run
	storel %jv, %rp
	storel 1, %rr
	%lk0 =l add %jv, 1
	storel %lk0, %rk
	jmp @lrun_loop

@lrun_loop
	%lk =l loadl %rk
	%lkin =w csltl %lk, %n
	jnz %lkin, @lrun_chkcap, @lrun_done

@lrun_chkcap
	%lr0 =l loadl %rr
	%lrcap =w csltl %lr0, 128
	jnz %lrcap, @lrun_cmp, @lrun_done

@lrun_cmp
	%lrv0 =l loadl %rp
	%lpb =l add %src, %lrv0
	%lb0 =w loadub %lpb
	%lpk =l add %src, %lk
	%lbk =w loadub %lpk
	%lbeq =w ceqw %lbk, %lb0
	jnz %lbeq, @lrun_inc, @lrun_done

@lrun_inc
	%lr1 =l loadl %rr
	%lr2 =l add %lr1, 1
	storel %lr2, %rr
	%lk1 =l add %lk, 1
	storel %lk1, %rk
	jmp @lrun_loop

@lrun_done
	%lrv =l loadl %rr
	%cont =w csltl %lrv, 3
	jnz %cont, @lit_inc, @lit_end

@lit_inc
	%jv2 =l loadl %j
	%jv3 =l add %jv2, 1
	storel %jv3, %j
	jmp @lit_loop

@lit_end
	%je =l loadl %j
	%L =l sub %je, %iv
	%needl =l add %L, 1
	%wv2 =l loadl %written
	%sum =l add %wv2, %needl
	%nofitl =w csltl %cap, %sum
	jnz %nofitl, @ret_neg, @lit_fit

@lit_fit
	%Lw =w copy %L
	%lhdr =w sub %Lw, 1
	%base =l add %out, %wv2
	storeb %lhdr, %base
	%d0 =l add %base, 1
	storel %d0, %dst
	storel %iv, %cp
	jmp @copy_loop

@copy_loop
	%cv =l loadl %cp
	%je2 =l loadl %j
	%cdo =w csltl %cv, %je2
	jnz %cdo, @copy_one, @copy_done

@copy_one
	%sptr =l add %src, %cv
	%sb =w loadub %sptr
	%dv =l loadl %dst
	storeb %sb, %dv
	%cvn =l add %cv, 1
	storel %cvn, %cp
	%dvn =l add %dv, 1
	storel %dvn, %dst
	jmp @copy_loop

@copy_done
	storel %sum, %written
	storel %je2, %i
	jmp @main_loop

@ret_written
	%res =l loadl %written
	ret %res

@ret_neg
	ret -1
}
