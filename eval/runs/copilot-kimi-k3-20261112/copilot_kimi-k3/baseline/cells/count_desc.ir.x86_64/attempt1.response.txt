export function l $count_desc(l %a, l %n) {
@start
	%cnt =l copy 0
	%i =l copy 0
	%lim =l sub %n, 1
	jmp @cond
@cond
	%c0 =w csltl %i, %lim
	jnz %c0, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p2 =l add %p, 8
	%y =l loadl %p2
	%c1 =w csgtl %x, %y
	%c1e =l extsw %c1
	%cnt =l add %cnt, %c1e
	%i =l add %i, 1
	jmp @cond
@done
	ret %cnt
}
