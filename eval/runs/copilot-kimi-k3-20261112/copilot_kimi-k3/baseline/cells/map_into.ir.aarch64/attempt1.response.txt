export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%count =l copy 0
@loop
	%inb =w csltl %i, %n
	jnz %inb, @body, @end
@body
	%off =l mul %i, 8
	%saddr =l add %src, %off
	%x =l loadl %saddr
	%r =l call %f(l %x)
	%daddr =l add %dst, %off
	storel %r, %daddr
	%lt =w csltl %r, %x
	%ltl =l extsw %lt
	%count =l add %count, %ltl
	%i =l add %i, 1
	jmp @loop
@end
	ret %count
}
