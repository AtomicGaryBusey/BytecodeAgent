export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @op_push %pc_push, @op_add %pc_add, @op_sub %pc_sub, @op_mul %pc_mul, @op_dup %pc_dup, @op_swap %pc_swap, @op_drop %pc_drop, @op_over %pc_over, @op_lt %pc_lt, @jz_zero %pc_jz0, @jz_nonzero %pc_jz1, @op_jmp %pc_jmp, @op_default %pc_def
	%d =l phi @start 0, @op_push %d_push, @op_add %d_add, @op_sub %d_sub, @op_mul %d_mul, @op_dup %d_dup, @op_swap %d, @op_drop %d_drop, @op_over %d_over, @op_lt %d_lt, @jz_zero %d_jz, @jz_nonzero %d_jz, @op_jmp %d, @op_default %d
	%end =w ceql %pc, %len
	jnz %end, @stop, @fetch
@fetch
	%f_off =l shl %pc, 3
	%f_addr =l add %code, %f_off
	%op =l loadl %f_addr
	%is0 =w ceql %op, 0
	jnz %is0, @stop, @chk1
@chk1
	%is1 =w ceql %op, 1
	jnz %is1, @op_push, @chk2
@chk2
	%is2 =w ceql %op, 2
	jnz %is2, @op_add, @chk3
@chk3
	%is3 =w ceql %op, 3
	jnz %is3, @op_sub, @chk4
@chk4
	%is4 =w ceql %op, 4
	jnz %is4, @op_mul, @chk5
@chk5
	%is5 =w ceql %op, 5
	jnz %is5, @op_dup, @chk6
@chk6
	%is6 =w ceql %op, 6
	jnz %is6, @op_swap, @chk7
@chk7
	%is7 =w ceql %op, 7
	jnz %is7, @op_drop, @chk8
@chk8
	%is8 =w ceql %op, 8
	jnz %is8, @op_over, @chk9
@chk9
	%is9 =w ceql %op, 9
	jnz %is9, @op_lt, @chk10
@chk10
	%is10 =w ceql %op, 10
	jnz %is10, @op_jz, @chk11
@chk11
	%is11 =w ceql %op, 11
	jnz %is11, @op_jmp, @op_default
@op_push
	%pu_pc1 =l add %pc, 1
	%pu_off =l shl %pu_pc1, 3
	%pu_addr =l add %code, %pu_off
	%pu_val =l loadl %pu_addr
	%pu_soff =l shl %d, 3
	%pu_saddr =l add %stack, %pu_soff
	storel %pu_val, %pu_saddr
	%d_push =l add %d, 1
	%pc_push =l add %pc, 2
	jmp @loop
@op_add
	%ad_i1 =l sub %d, 1
	%ad_i2 =l sub %d, 2
	%ad_o1 =l shl %ad_i1, 3
	%ad_o2 =l shl %ad_i2, 3
	%ad_a1 =l add %stack, %ad_o1
	%ad_a2 =l add %stack, %ad_o2
	%ad_b =l loadl %ad_a1
	%ad_a =l loadl %ad_a2
	%ad_r =l add %ad_a, %ad_b
	storel %ad_r, %ad_a2
	%d_add =l sub %d, 1
	%pc_add =l add %pc, 1
	jmp @loop
@op_sub
	%su_i1 =l sub %d, 1
	%su_i2 =l sub %d, 2
	%su_o1 =l shl %su_i1, 3
	%su_o2 =l shl %su_i2, 3
	%su_a1 =l add %stack, %su_o1
	%su_a2 =l add %stack, %su_o2
	%su_b =l loadl %su_a1
	%su_a =l loadl %su_a2
	%su_r =l sub %su_a, %su_b
	storel %su_r, %su_a2
	%d_sub =l sub %d, 1
	%pc_sub =l add %pc, 1
	jmp @loop
@op_mul
	%mu_i1 =l sub %d, 1
	%mu_i2 =l sub %d, 2
	%mu_o1 =l shl %mu_i1, 3
	%mu_o2 =l shl %mu_i2, 3
	%mu_a1 =l add %stack, %mu_o1
	%mu_a2 =l add %stack, %mu_o2
	%mu_b =l loadl %mu_a1
	%mu_a =l loadl %mu_a2
	%mu_r =l mul %mu_a, %mu_b
	storel %mu_r, %mu_a2
	%d_mul =l sub %d, 1
	%pc_mul =l add %pc, 1
	jmp @loop
@op_dup
	%du_i1 =l sub %d, 1
	%du_o1 =l shl %du_i1, 3
	%du_o2 =l shl %d, 3
	%du_a1 =l add %stack, %du_o1
	%du_a2 =l add %stack, %du_o2
	%du_v =l loadl %du_a1
	storel %du_v, %du_a2
	%d_dup =l add %d, 1
	%pc_dup =l add %pc, 1
	jmp @loop
@op_swap
	%sw_i1 =l sub %d, 1
	%sw_i2 =l sub %d, 2
	%sw_o1 =l shl %sw_i1, 3
	%sw_o2 =l shl %sw_i2, 3
	%sw_a1 =l add %stack, %sw_o1
	%sw_a2 =l add %stack, %sw_o2
	%sw_b =l loadl %sw_a1
	%sw_a =l loadl %sw_a2
	storel %sw_b, %sw_a2
	storel %sw_a, %sw_a1
	%pc_swap =l add %pc, 1
	jmp @loop
@op_drop
	%d_drop =l sub %d, 1
	%pc_drop =l add %pc, 1
	jmp @loop
@op_over
	%ov_i2 =l sub %d, 2
	%ov_o2 =l shl %ov_i2, 3
	%ov_od =l shl %d, 3
	%ov_a2 =l add %stack, %ov_o2
	%ov_ad =l add %stack, %ov_od
	%ov_v =l loadl %ov_a2
	storel %ov_v, %ov_ad
	%d_over =l add %d, 1
	%pc_over =l add %pc, 1
	jmp @loop
@op_lt
	%lt_i1 =l sub %d, 1
	%lt_i2 =l sub %d, 2
	%lt_o1 =l shl %lt_i1, 3
	%lt_o2 =l shl %lt_i2, 3
	%lt_a1 =l add %stack, %lt_o1
	%lt_a2 =l add %stack, %lt_o2
	%lt_b =l loadl %lt_a1
	%lt_a =l loadl %lt_a2
	%lt_w =w csltl %lt_a, %lt_b
	%lt_r =l extsw %lt_w
	storel %lt_r, %lt_a2
	%d_lt =l sub %d, 1
	%pc_lt =l add %pc, 1
	jmp @loop
@op_jz
	%jz_pc1 =l add %pc, 1
	%jz_to =l shl %jz_pc1, 3
	%jz_ta =l add %code, %jz_to
	%jz_t =l loadl %jz_ta
	%jz_i1 =l sub %d, 1
	%jz_o1 =l shl %jz_i1, 3
	%jz_a1 =l add %stack, %jz_o1
	%jz_v =l loadl %jz_a1
	%d_jz =l sub %d, 1
	%jz_z =w ceql %jz_v, 0
	jnz %jz_z, @jz_zero, @jz_nonzero
@jz_zero
	%pc_jz0 =l copy %jz_t
	jmp @loop
@jz_nonzero
	%pc_jz1 =l add %pc, 2
	jmp @loop
@op_jmp
	%jm_pc1 =l add %pc, 1
	%jm_to =l shl %jm_pc1, 3
	%jm_ta =l add %code, %jm_to
	%pc_jmp =l loadl %jm_ta
	jmp @loop
@op_default
	%pc_def =l add %pc, 1
	jmp @loop
@stop
	%st_empty =w ceql %d, 0
	jnz %st_empty, @st_zero, @st_top
@st_top
	%st_dm1 =l sub %d, 1
	%st_off =l shl %st_dm1, 3
	%st_addr =l add %stack, %st_off
	%st_res =l loadl %st_addr
	jmp @st_merge
@st_zero
	%st_res0 =l copy 0
	jmp @st_merge
@st_merge
	%rv =l phi @st_top %st_res, @st_zero %st_res0
	%zi0 =l copy %d
	jmp @zloop
@zloop
	%zi =l phi @st_merge %zi0, @zbody %zi_next
	%zdone =w ceql %zi, %cap
	jnz %zdone, @ret, @zbody
@zbody
	%z_off =l shl %zi, 3
	%z_addr =l add %stack, %z_off
	storel 0, %z_addr
	%zi_next =l add %zi, 1
	jmp @zloop
@ret
	ret %rv
}
