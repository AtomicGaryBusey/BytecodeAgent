export function l $map_sum(l %a, l %n, l %f) {
@start
	jmp @loop
@loop
	%i =l phi @start 0, @body %i2
	%sum =l phi @start 0, @body %sum2
	%done =w csgel %i, %n
	jnz %done, @end, @body
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%r =l call %f(l %x)
	%sum2 =l add %sum, %r
	%i2 =l add %i, 1
	jmp @loop
@end
	ret %sum
}
