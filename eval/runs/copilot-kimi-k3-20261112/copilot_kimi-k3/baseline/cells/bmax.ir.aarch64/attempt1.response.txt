export function l $bmax(l %a, l %n) {
@start
	%init =l loadl %a
	jmp @loop
@loop
	%i =l phi @start 1, @next %inext
	%max =l phi @start %init, @next %maxnext
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%cmp =w csgtl %v, %max
	jnz %cmp, @upd, @next
@upd
	jmp @next
@next
	%maxnext =l phi @upd %v, @body %max
	%inext =l add %i, 1
	jmp @loop
@done
	ret %max
}
