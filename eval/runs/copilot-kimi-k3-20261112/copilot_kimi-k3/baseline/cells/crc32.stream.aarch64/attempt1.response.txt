crc32:
	mov w2, #0xFFFFFFFF
	mov x3, #0
	mov w6, #0x8320
	movk w6, #0xEDB8, lsl #16
.Lbyte_loop:
	cmp x3, x1
	b.ge .Ldone
	ldrb w4, [x0, x3]
	eor w2, w2, w4
	mov w5, #8
.Lbit_loop:
	tbz w2, #0, .Lno_xor
	lsr w2, w2, #1
	eor w2, w2, w6
	b .Lnext_bit
.Lno_xor:
	lsr w2, w2, #1
.Lnext_bit:
	subs w5, w5, #1
	b.ne .Lbit_loop
	add x3, x3, #1
	b .Lbyte_loop
.Ldone:
	mvn w0, w2
	ret
