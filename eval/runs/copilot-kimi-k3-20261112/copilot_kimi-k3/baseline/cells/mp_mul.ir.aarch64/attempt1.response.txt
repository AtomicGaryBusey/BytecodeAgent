export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
	%n =l add %na, %nb
	jmp @zhead
@zhead
	%zk =l phi @start 0, @zbody %zk1
	%zcmp =w cultl %zk, %n
	jnz %zcmp, @zbody, @mcheck
@zbody
	%zoff =l mul %zk, 8
	%zp =l add %r, %zoff
	storel 0, %zp
	%zk1 =l add %zk, 1
	jmp @zhead
@mcheck
	%az =w ceql %na, 0
	jnz %az, @siginit, @checkb
@checkb
	%bz =w ceql %nb, 0
	jnz %bz, @siginit, @ihead
@ihead
	%i =l phi @checkb 0, @jdone %i1
	%icmp =w cultl %i, %na
	jnz %icmp, @ibody, @siginit
@ibody
	%ioff =l mul %i, 8
	%iap =l add %a, %ioff
	%av =l loadl %iap
	jmp @jhead
@jhead
	%j =l phi @ibody 0, @jbody %j1
	%carry =l phi @ibody 0, @jbody %carry1
	%jcmp =w cultl %j, %nb
	jnz %jcmp, @jbody, @jdone
@jbody
	%joff =l mul %j, 8
	%jbp =l add %b, %joff
	%bv =l loadl %jbp
	%ij =l add %i, %j
	%ijoff =l mul %ij, 8
	%rp =l add %r, %ijoff
	%rv =l loadl %rp
	%x0 =l and %av, 4294967295
	%x1 =l shr %av, 32
	%y0 =l and %bv, 4294967295
	%y1 =l shr %bv, 32
	%p0 =l mul %x0, %y0
	%p1 =l mul %x0, %y1
	%p2 =l mul %x1, %y0
	%p3 =l mul %x1, %y1
	%mid =l add %p1, %p2
	%cmw =w cultl %mid, %p1
	%cm =l extuw %cmw
	%mlo =l shl %mid, 32
	%mhi =l shr %mid, 32
	%low0 =l add %p0, %mlo
	%clw =w cultl %low0, %p0
	%cl =l extuw %clw
	%cms =l shl %cm, 32
	%h1 =l add %p3, %mhi
	%h2 =l add %h1, %cms
	%ph =l add %h2, %cl
	%s1 =l add %low0, %rv
	%c1w =w cultl %s1, %low0
	%c1 =l extuw %c1w
	%s2 =l add %s1, %carry
	%c2w =w cultl %s2, %s1
	%c2 =l extuw %c2w
	%cs =l add %c1, %c2
	%carry1 =l add %ph, %cs
	storel %s2, %rp
	%j1 =l add %j, 1
	jmp @jhead
@jdone
	%eidx =l add %i, %nb
	%eoff =l mul %eidx, 8
	%ep =l add %r, %eoff
	storel %carry, %ep
	%i1 =l add %i, 1
	jmp @ihead
@siginit
	jmp @sighead
@sighead
	%sk =l phi @siginit %n, @sigbody %sk1
	%spos =w cultl 0, %sk
	jnz %spos, @sigcheck, @ret
@sigcheck
	%sidx =l sub %sk, 1
	%soff =l mul %sidx, 8
	%srp =l add %r, %soff
	%sv =l loadl %srp
	%sz =w ceql %sv, 0
	jnz %sz, @sigbody, @ret
@sigbody
	%sk1 =l sub %sk, 1
	jmp @sighead
@ret
	ret %sk
}
