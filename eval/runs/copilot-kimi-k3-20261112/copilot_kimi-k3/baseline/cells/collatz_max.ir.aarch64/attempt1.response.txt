export function l $collatz_max(l %n) {
@start
	%kp =l alloc8 8
	%mp =l alloc8 8
	%xp =l alloc8 8
	%stepp =l alloc8 8
	storel 1, %kp
	storel 0, %mp
	jmp @outer
@outer
	%k =l loadl %kp
	%done =w csgtl %k, %n
	jnz %done, @ret, @init
@init
	storel %k, %xp
	storel 0, %stepp
	jmp @inner
@inner
	%x =l loadl %xp
	%is1 =w ceql %x, 1
	jnz %is1, @check, @step
@step
	%b =l and %x, 1
	%odd =w ceql %b, 1
	jnz %odd, @odd, @even
@even
	%xe =l shr %x, 1
	storel %xe, %xp
	jmp @inc
@odd
	%t =l mul %x, 3
	%xo =l add %t, 1
	storel %xo, %xp
	jmp @inc
@inc
	%s =l loadl %stepp
	%s1 =l add %s, 1
	storel %s1, %stepp
	jmp @inner
@check
	%sc =l loadl %stepp
	%mc =l loadl %mp
	%gt =w csgtl %sc, %mc
	jnz %gt, @setmax, @next
@setmax
	storel %sc, %mp
	jmp @next
@next
	%kn =l loadl %kp
	%ki =l add %kn, 1
	storel %ki, %kp
	jmp @outer
@ret
	%r =l loadl %mp
	ret %r
}
