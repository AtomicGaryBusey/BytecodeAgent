export function l $count_desc(l %a, l %n) {
@start
	%c =w csltl %n, 2
	jnz %c, @ret0, @loop
@loop
	%i =l phi @start 0, @body %in
	%cnt =l phi @start 0, @body %cn
	%nm1 =l sub %n, 1
	%go =w csltl %i, %nm1
	jnz %go, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%gt =w csltl %y, %x
	%g =l extsw %gt
	%cn =l add %cnt, %g
	%in =l add %i, 1
	jmp @loop
@done
	ret %cnt
@ret0
	ret 0
}
