export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @up %lo1, @down %lo
	%hi =l phi @start %n, @up %hi, @down %mid
	%c =w csltl %lo, %hi
	jnz %c, @body, @notfound
@body
	%sum =l add %lo, %hi
	%mid =l div %sum, 2
	%off =l mul %mid, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @up, @down
@up
	%lo1 =l add %mid, 1
	jmp @loop
@down
	jmp @loop
@found
	ret %mid
@notfound
	ret -1
}
