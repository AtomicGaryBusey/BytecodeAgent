.globl _start

.bss
.balign 16
input_buffer:
.zero 65537
output_buffer:
.zero 330000

.section .rodata
hex_digits:
.ascii "0123456789abcdef"

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output_buffer]
    lea r15, [rip + input_buffer]
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16
row_size_ready:
    mov eax, r13d
    call emit_hex8
    mov ax, 0x2020
    stosw

    xor r8d, r8d
hex_loop:
    cmp r8, r14
    jae empty_slot
    movzx r9d, byte ptr [r15 + r8]
    mov edx, r9d
    shr edx, 4
    mov al, byte ptr [hex_digits + rdx]
    stosb
    and r9d, 15
    mov al, byte ptr [hex_digits + r9]
    stosb
    mov al, 32
    stosb
    jmp slot_done
empty_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb
slot_done:
    cmp r8d, 7
    jne no_extra_space
    mov al, 32
    stosb
no_extra_space:
    inc r8d
    cmp r8d, 16
    jb hex_loop

    mov al, 32
    stosb
    mov al, 124
    stosb
    xor r8d, r8d
ascii_loop:
    mov al, byte ptr [r15 + r8]
    cmp al, 32
    jb replacement
    cmp al, 126
    jbe emit_character
replacement:
    mov al, 46
emit_character:
    stosb
    inc r8
    cmp r8, r14
    jb ascii_loop
    mov al, 124
    stosb
    mov al, 10
    stosb
    add r15, r14
    add r13, r14
    jmp row_loop

final_offset:
    mov eax, r12d
    call emit_hex8
    mov al, 10
    stosb

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_hex8:
    mov ebx, eax
    mov ecx, 28
hex8_loop:
    mov edx, ebx
    shr edx, cl
    and edx, 15
    mov al, byte ptr [hex_digits + rdx]
    stosb
    sub ecx, 4
    jns hex8_loop
    ret
