.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
.zero 49153
output:
.zero 66400

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz encode
    add r12, rax
    jmp read_loop

encode:
    lea rsi, [rip + input]
    lea rdi, [rip + output]
    lea rbx, [rip + alphabet]
    xor r8d, r8d
    test r12, r12
    jz finish

group:
    mov r9d, 1
    movzx eax, byte ptr [rsi]
    inc rsi
    dec r12
    shl eax, 16
    test r12, r12
    jz emit
    movzx edx, byte ptr [rsi]
    inc rsi
    dec r12
    shl edx, 8
    or eax, edx
    inc r9d
    test r12, r12
    jz emit
    movzx edx, byte ptr [rsi]
    inc rsi
    dec r12
    or eax, edx
    inc r9d

emit:
    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi], dl
    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    cmp r9d, 1
    je group_done
    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], dl
    cmp r9d, 2
    je group_done
    and eax, 63
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], al

group_done:
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne next_group
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
next_group:
    test r12, r12
    jnz group
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

finish:
    mov eax, 60
    xor edi, edi
    syscall
