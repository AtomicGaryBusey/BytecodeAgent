.globl _start

.bss
input:
    .zero 65537
output:
    .zero 65537

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz process
    add r13, rax
    jmp read_loop

process:
    lea r14, [rip + output]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
scan:
    cmp r9, r13
    je remainder
    cmp byte ptr [r12 + r9], 10
    je line
    inc r9
    jmp scan

line:
    mov rbx, r9
reverse:
    cmp rbx, r8
    je newline
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse
newline:
    mov byte ptr [r14 + r10], 10
    inc r10
    inc r9
    mov r8, r9
    jmp scan

remainder:
    cmp r8, r13
    je emit
    mov rbx, r13
reverse_final:
    cmp rbx, r8
    je final_newline
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_final
final_newline:
    mov byte ptr [r14 + r10], 10
    inc r10

emit:
    test r10, r10
    jz done
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    syscall
done:
    mov eax, 60
    xor edi, edi
    syscall
