.globl _start
.bss
buffer:
.zero 65537
outbuf:
.zero 32
.text
_start:
    lea r14, [rip + buffer]
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    mov edx, 65537
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop
read_done:
    mov r12, r14
    lea r13, [r14 + r15]
    call whitespace
    call value
    call whitespace
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok]
    mov edx, 3
    syscall
    xor edi, edi
    jmp terminate

peek:
    mov eax, -1
    cmp r12, r13
    jae peek_done
    movzx eax, byte ptr [r12]
peek_done:
    ret

whitespace:
    call peek
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    je ws_next
    ret
ws_next:
    inc r12
    jmp whitespace

value:
    call peek
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je true_value
    cmp eax, 102
    je false_value
    cmp eax, 110
    je null_value
    cmp eax, 45
    je number
    sub eax, 48
    cmp eax, 9
    jbe number
    jmp error

true_value:
    lea rbx, [rip + true_text]
    jmp literal
false_value:
    lea rbx, [rip + false_text]
    jmp literal
null_value:
    lea rbx, [rip + null_text]
literal:
    movzx edx, byte ptr [rbx]
    test edx, edx
    jz literal_done
    call peek
    cmp eax, edx
    jne error
    inc r12
    inc rbx
    jmp literal
literal_done:
    ret

string:
    inc r12
string_loop:
    call peek
    cmp eax, -1
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_done
    cmp eax, 92
    je escape
    inc r12
    jmp string_loop
string_done:
    inc r12
    ret
escape:
    inc r12
    call peek
    cmp eax, 117
    je unicode
    cmp eax, 34
    je escape_done
    cmp eax, 92
    je escape_done
    cmp eax, 47
    je escape_done
    cmp eax, 98
    je escape_done
    cmp eax, 102
    je escape_done
    cmp eax, 110
    je escape_done
    cmp eax, 114
    je escape_done
    cmp eax, 116
    jne error
escape_done:
    inc r12
    jmp string_loop
unicode:
    inc r12
    mov r8d, 4
hex_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_good
    or eax, 32
    cmp eax, 97
    jb error
    cmp eax, 102
    ja error
hex_good:
    inc r12
    dec r8d
    jnz hex_loop
    jmp string_loop

number:
    call peek
    cmp eax, 45
    jne integer
    inc r12
    call peek
integer:
    cmp eax, 48
    je zero_integer
    cmp eax, 49
    jb error
    cmp eax, 57
    ja error
integer_loop:
    inc r12
    call peek
    cmp eax, 48
    jb fraction
    cmp eax, 57
    jbe integer_loop
    jmp fraction
zero_integer:
    inc r12
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r12
    call require_digit
fraction_loop:
    inc r12
    call peek
    cmp eax, 48
    jb exponent
    cmp eax, 57
    jbe fraction_loop
exponent:
    cmp eax, 101
    je exponent_start
    cmp eax, 69
    jne number_done
exponent_start:
    inc r12
    call peek
    cmp eax, 43
    je exponent_sign
    cmp eax, 45
    jne exponent_digits
exponent_sign:
    inc r12
exponent_digits:
    call require_digit
exponent_loop:
    inc r12
    call peek
    cmp eax, 48
    jb number_done
    cmp eax, 57
    jbe exponent_loop
number_done:
    ret
require_digit:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    ret

array:
    inc r12
    call whitespace
    cmp eax, 93
    je container_done
array_value:
    call value
    call whitespace
    cmp eax, 93
    je container_done
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp array_value

object:
    inc r12
    call whitespace
    cmp eax, 125
    je container_done
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r12
    call whitespace
    call value
    call whitespace
    cmp eax, 125
    je container_done
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp object_member
container_done:
    inc r12
    ret

error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_text]
    mov edx, 9
    syscall
    mov rax, r12
    sub rax, r14
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ebx, 10
decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
terminate:
    mov eax, 60
    syscall
.section .rodata
ok:
.ascii "ok\n"
error_text:
.ascii "error at "
true_text:
.asciz "true"
false_text:
.asciz "false"
null_text:
.asciz "null"
