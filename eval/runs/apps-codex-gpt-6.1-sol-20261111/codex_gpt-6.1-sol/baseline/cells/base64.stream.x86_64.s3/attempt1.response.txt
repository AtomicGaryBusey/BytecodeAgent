.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
.zero 49155
output:
.zero 66400

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz encode_start
    add r12, rax
    jmp read_loop

encode_start:
    test r12, r12
    jz exit_ok
    lea rsi, [rip + input]
    lea rdi, [rip + output]
    lea r10, [rip + alphabet]
    xor r9d, r9d

encode_loop:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

    mov edx, eax
    shr edx, 18
    movzx ebx, byte ptr [r10 + rdx]
    mov byte ptr [rdi], bl

    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx ebx, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 1], bl

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    cmp r12, 1
    je group_done

    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx ebx, byte ptr [r10 + rdx]
    mov byte ptr [rdi + 2], bl
    cmp r12, 2
    je group_done

    and eax, 63
    movzx ebx, byte ptr [r10 + rax]
    mov byte ptr [rdi + 3], bl

group_done:
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne check_remaining
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d

check_remaining:
    cmp r12, 3
    jbe encode_done
    sub r12, 3
    add rsi, 3
    jmp encode_loop

encode_done:
    test r9d, r9d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall
