.globl _start
.bss
input:
.zero 65537
output:
.zero 65537

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea r14, [rip + output]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d

scan_loop:
    cmp r9, r13
    je final_line
    cmp byte ptr [r12 + r9], 10
    je terminated_line
    inc r9
    jmp scan_loop

terminated_line:
    mov rbx, r9
reverse_terminated:
    cmp rbx, r8
    je terminate_output
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_terminated
terminate_output:
    mov byte ptr [r14 + r10], 10
    inc r10
    inc r9
    mov r8, r9
    jmp scan_loop

final_line:
    cmp r8, r13
    je write_output
    mov rbx, r13
reverse_final:
    cmp rbx, r8
    je finish_final
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_final
finish_final:
    mov byte ptr [r14 + r10], 10
    inc r10

write_output:
    test r10, r10
    jz done
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    syscall
done:
    mov eax, 60
    xor edi, edi
    syscall
