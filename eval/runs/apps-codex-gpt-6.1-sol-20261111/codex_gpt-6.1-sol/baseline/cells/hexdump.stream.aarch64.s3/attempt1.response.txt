.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    adrp x20, output
    add x20, x20, :lo12:output
    adrp x21, digits
    add x21, x21, :lo12:digits
    mov x22, #0
    mov x24, #0

next_row:
    mov x23, #0
read_more:
    mov x0, #0
    add x1, x19, x23
    mov x2, #16
    sub x2, x2, x23
    mov x8, #63
    svc #0
    cbz x0, reached_eof
    add x23, x23, x0
    cmp x23, #16
    b.lo read_more
    b format_row

reached_eof:
    mov x24, #1
    cbz x23, final_offset

format_row:
    mov x0, x22
    mov x1, x20
    bl emit_offset
    mov w2, #32
    strb w2, [x1], #1
    strb w2, [x1], #1
    mov x3, #0

hex_slots:
    cmp x3, x23
    b.hs blank_slot
    ldrb w4, [x19, x3]
    lsr w5, w4, #4
    ldrb w5, [x21, x5]
    strb w5, [x1], #1
    and w4, w4, #15
    ldrb w5, [x21, x4]
    strb w5, [x1], #1
    strb w2, [x1], #1
    b slot_done
blank_slot:
    strb w2, [x1], #1
    strb w2, [x1], #1
    strb w2, [x1], #1
slot_done:
    cmp x3, #7
    b.ne no_extra_space
    strb w2, [x1], #1
no_extra_space:
    add x3, x3, #1
    cmp x3, #16
    b.lo hex_slots

    strb w2, [x1], #1
    mov w2, #124
    strb w2, [x1], #1
    mov x3, #0
ascii_loop:
    ldrb w4, [x19, x3]
    cmp w4, #32
    b.lo dot
    cmp w4, #126
    b.ls ascii_store
dot:
    mov w4, #46
ascii_store:
    strb w4, [x1], #1
    add x3, x3, #1
    cmp x3, x23
    b.lo ascii_loop
    strb w2, [x1], #1
    mov w2, #10
    strb w2, [x1], #1
    sub x2, x1, x20
    mov x1, x20
    mov x0, #1
    mov x8, #64
    svc #0
    add x22, x22, x23
    cbz x24, next_row

final_offset:
    mov x0, x22
    mov x1, x20
    bl emit_offset
    mov w2, #10
    strb w2, [x1]
    mov x0, #1
    mov x1, x20
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

emit_offset:
    mov x4, #28
offset_digit:
    lsr x5, x0, x4
    and x5, x5, #15
    ldrb w5, [x21, x5]
    strb w5, [x1], #1
    subs x4, x4, #4
    b.ge offset_digit
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 16
output:
    .zero 80
