.globl _start

.bss
.balign 16
input:
.zero 32769
records_a:
.zero 16000
records_b:
.zero 16000
output:
.zero 33768

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz parse
    add r13, rax
    jmp read_loop

parse:
    lea rdi, [rip + records_a]
    xor r14d, r14d
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r13
    jae parse_tail
    cmp byte ptr [r12 + rcx], 10
    jne parse_next
    lea rax, [r12 + rdx]
    mov qword ptr [rdi], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r14
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_tail:
    cmp rdx, r13
    jae sort_setup
    lea rax, [r12 + rdx]
    mov qword ptr [rdi], rax
    mov rax, r13
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    inc r14

sort_setup:
    lea r12, [rip + records_a]
    lea r13, [rip + records_b]
    mov r15d, 1
sort_pass:
    cmp r15, r14
    jae emit
    xor r8d, r8d
merge_group:
    cmp r8, r14
    jae pass_done
    lea r9, [r8 + r15]
    cmp r9, r14
    cmova r9, r14
    lea r10, [r9 + r15]
    cmp r10, r14
    cmova r10, r14
    mov rbx, r8
    mov rbp, r9
    mov rdx, r8
merge_loop:
    cmp rdx, r10
    jae group_done
    cmp rbx, r9
    jae take_right
    cmp rbp, r10
    jae take_left
    mov rdi, rbx
    shl rdi, 4
    add rdi, r12
    mov rsi, rbp
    shl rsi, 4
    add rsi, r12
    call compare
    jbe take_left
take_right:
    mov rax, rbp
    inc rbp
    jmp copy_record
take_left:
    mov rax, rbx
    inc rbx
copy_record:
    shl rax, 4
    movdqu xmm0, xmmword ptr [r12 + rax]
    mov rax, rdx
    shl rax, 4
    movdqu xmmword ptr [r13 + rax], xmm0
    inc rdx
    jmp merge_loop
group_done:
    mov r8, r10
    jmp merge_group
pass_done:
    xchg r12, r13
    shl r15, 1
    jmp sort_pass

compare:
    mov rax, qword ptr [rdi + 8]
    mov r11, qword ptr [rsi + 8]
    mov rcx, rax
    cmp rcx, r11
    cmova rcx, r11
    mov rdi, qword ptr [rdi]
    mov rsi, qword ptr [rsi]
    cmp eax, eax
    repe cmpsb
    jne compare_return
    cmp rax, r11
compare_return:
    ret

emit:
    lea rdi, [rip + output]
    xor ebx, ebx
emit_loop:
    cmp rbx, r14
    jae write_output
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rcx, qword ptr [r12 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp emit_loop
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
