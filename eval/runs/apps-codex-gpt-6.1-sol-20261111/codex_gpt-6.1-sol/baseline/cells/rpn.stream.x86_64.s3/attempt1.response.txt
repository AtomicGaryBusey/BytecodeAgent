.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    add r13, r12
    mov rbx, r12
    lea r8, [rip + output]
    xor r14d, r14d
    xor r15d, r15d
next_byte:
    cmp r12, r13
    je input_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je line_end
    test r15d, r15d
    jnz skip_byte
    cmp al, 32
    je skip_byte
    cmp al, 9
    je skip_byte
    mov rsi, r12
scan_token:
    inc r12
    cmp r12, r13
    je token_ready
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_ready
    cmp al, 32
    je token_ready
    cmp al, 9
    jne scan_token
token_ready:
    mov rcx, r12
    sub rcx, rsi
    cmp rcx, 1
    jne literal
    movzx edi, byte ptr [rsi]
    cmp dil, '+'
    je operator
    cmp dil, '-'
    je operator
    cmp dil, '*'
    je operator
    cmp dil, '/'
    je operator
    cmp dil, '%'
    je operator
literal:
    xor r9d, r9d
    cmp byte ptr [rsi], '-'
    jne digits_begin
    mov r9d, 1
    inc rsi
    dec rcx
digits_begin:
    test rcx, rcx
    jz bad
    cmp rcx, 18
    ja bad
    xor eax, eax
digits_loop:
    movzx edx, byte ptr [rsi]
    sub edx, '0'
    cmp edx, 9
    ja bad
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digits_loop
    test r9d, r9d
    jz push_value
    neg rax
push_value:
    lea rsi, [rip + values]
    mov qword ptr [rsi + r14*8], rax
    inc r14
    jmp next_byte
operator:
    cmp r14, 2
    jb bad
    lea rsi, [rip + values]
    dec r14
    mov rcx, qword ptr [rsi + r14*8]
    mov rax, qword ptr [rsi + r14*8 - 8]
    cmp dil, '+'
    je add_op
    cmp dil, '-'
    je sub_op
    cmp dil, '*'
    je mul_op
    test rcx, rcx
    jz bad
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne divide
    cmp rcx, -1
    je bad
divide:
    cqo
    idiv rcx
    cmp dil, '%'
    jne store_op
    mov rax, rdx
    jmp store_op
add_op:
    add rax, rcx
    jmp store_op
sub_op:
    sub rax, rcx
    jmp store_op
mul_op:
    imul rax, rcx
store_op:
    mov qword ptr [rsi + r14*8 - 8], rax
    jmp next_byte
bad:
    mov r15d, 1
    jmp next_byte
skip_byte:
    inc r12
    jmp next_byte
line_end:
    call emit_line
    inc r12
    mov rbx, r12
    xor r14d, r14d
    xor r15d, r15d
    jmp next_byte
input_end:
    cmp r12, rbx
    je write_output
    call emit_line
write_output:
    lea rsi, [rip + output]
    mov rdx, r8
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

emit_line:
    test r15d, r15d
    jnz emit_error
    cmp r14, 1
    jne emit_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns magnitude
    mov byte ptr [r8], '-'
    inc r8
    neg rax
magnitude:
    lea rdi, [rip + scratch + 32]
    mov ecx, 10
    xor r10d, r10d
decimal_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc r10
    test rax, rax
    jnz decimal_loop
copy_digits:
    mov al, byte ptr [rdi]
    mov byte ptr [r8], al
    inc rdi
    inc r8
    dec r10
    jnz copy_digits
    mov byte ptr [r8], 10
    inc r8
    ret
emit_error:
    mov dword ptr [r8], 0x6f727265
    mov word ptr [r8 + 4], 0x0a72
    add r8, 6
    ret

.bss
.balign 16
input:
    .zero 65537
values:
    .zero 8000
scratch:
    .zero 32
output:
    .zero 400000
