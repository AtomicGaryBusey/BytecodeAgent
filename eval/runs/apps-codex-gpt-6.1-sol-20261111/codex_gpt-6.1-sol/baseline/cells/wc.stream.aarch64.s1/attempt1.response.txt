.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, buffer
    add x23, x23, :lo12:buffer

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, output
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0

count_loop:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace
    cbnz w22, next_byte
    add x20, x20, #1
    mov w22, #1
    b next_byte

whitespace:
    mov w22, #0

next_byte:
    cmp x1, x2
    b.ne count_loop
    b read_loop

output:
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x25, x24
    mov x0, x19
    bl decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x20
    bl decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x21
    bl decimal
    mov w3, #10
    strb w3, [x25], #1

    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

decimal:
    sub sp, sp, #32
    add x5, sp, #32
    mov x6, x5
    mov x7, #10
decimal_digits:
    udiv x3, x0, x7
    msub x4, x3, x7, x0
    add w4, w4, #48
    strb w4, [x5, #-1]!
    mov x0, x3
    cbnz x0, decimal_digits
decimal_copy:
    ldrb w4, [x5], #1
    strb w4, [x25], #1
    cmp x5, x6
    b.ne decimal_copy
    add sp, sp, #32
    ret

.bss
.balign 16
buffer:
    .zero 4096
output_buffer:
    .zero 64
