.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x21, #65536

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    cbnz x2, read_call
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, process
    add x20, x20, x0
    b read_loop

process:
    adrp x22, output
    add x22, x22, :lo12:output
    mov x23, #0
    mov x24, #0
    mov x25, #0

scan:
    cmp x24, x20
    b.eq end_input
    ldrb w9, [x19, x24]
    cmp w9, #10
    b.eq terminated_line
    add x24, x24, #1
    b scan

terminated_line:
    mov x26, x24
reverse_line:
    cmp x26, x23
    b.eq append_newline
    sub x26, x26, #1
    ldrb w9, [x19, x26]
    strb w9, [x22, x25]
    add x25, x25, #1
    b reverse_line

append_newline:
    mov w9, #10
    strb w9, [x22, x25]
    add x25, x25, #1
    add x24, x24, #1
    mov x23, x24
    b scan

end_input:
    cmp x23, x20
    b.eq write_output
    mov x26, x20
reverse_final:
    cmp x26, x23
    b.eq final_newline
    sub x26, x26, #1
    ldrb w9, [x19, x26]
    strb w9, [x22, x25]
    add x25, x25, #1
    b reverse_final

final_newline:
    mov w9, #10
    strb w9, [x22, x25]
    add x25, x25, #1

write_output:
    cbz x25, done
    mov x0, #1
    mov x1, x22
    mov x2, x25
    mov x8, #64
    svc #0

done:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65537
output:
    .zero 65537
