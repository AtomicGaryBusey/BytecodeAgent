.globl _start

.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
input:
.zero 49153
output:
.zero 66400

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 49153
    sub rdx, r12
    syscall
    test rax, rax
    jz encode
    add r12, rax
    jmp read_loop

encode:
    lea rsi, [rip + input]
    lea rdi, [rip + output]
    lea rbx, [rip + alphabet]
    xor r8d, r8d
    test r12, r12
    jz finish

group:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r12, 1
    je packed
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    cmp r12, 2
    je packed
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
packed:
    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [rdi], cl
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], cl

    mov word ptr [rdi + 2], 0x3d3d
    cmp r12, 1
    je emitted
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx ecx, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 2], cl
    cmp r12, 2
    je emitted
    and eax, 63
    movzx ecx, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], cl
emitted:
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne advance
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
advance:
    cmp r12, 3
    jbe last
    sub r12, 3
    add rsi, 3
    jmp group
last:
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall
