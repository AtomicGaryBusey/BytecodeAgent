.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x25, #1
    movk x25, #1, lsl #16

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x25, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, output
    add x23, x23, :lo12:output
    mov x21, #0
    mov x22, #0
    mov x24, #0

scan_loop:
    cmp x21, x20
    b.eq final_remainder
    ldrb w9, [x19, x21]
    cmp w9, #10
    b.eq complete_line
    add x21, x21, #1
    b scan_loop

complete_line:
    mov x10, x21
reverse_line:
    cmp x10, x22
    b.eq append_newline
    sub x10, x10, #1
    ldrb w9, [x19, x10]
    strb w9, [x23, x24]
    add x24, x24, #1
    b reverse_line

append_newline:
    mov w9, #10
    strb w9, [x23, x24]
    add x24, x24, #1
    add x21, x21, #1
    mov x22, x21
    b scan_loop

final_remainder:
    cmp x22, x20
    b.eq write_output
    mov x10, x20
reverse_final:
    cmp x10, x22
    b.eq final_newline
    sub x10, x10, #1
    ldrb w9, [x19, x10]
    strb w9, [x23, x24]
    add x24, x24, #1
    b reverse_final

final_newline:
    mov w9, #10
    strb w9, [x23, x24]
    add x24, x24, #1

write_output:
    cbz x24, finish
    mov x0, #1
    mov x1, x23
    mov x2, x24
    mov x8, #64
    svc #0

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 65537
