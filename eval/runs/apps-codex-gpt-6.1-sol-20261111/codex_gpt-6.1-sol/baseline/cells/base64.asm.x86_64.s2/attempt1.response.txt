.section .rodata
alphabet:
.ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input:
.zero 65536
output:
.zero 66560

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz encode_start
    add r13, rax
    jmp read_loop

encode_start:
    lea r14, [rip + output]
    lea r15, [rip + alphabet]
    xor ebx, ebx
    xor ebp, ebp
    xor r8d, r8d

encode_loop:
    cmp rbx, r13
    jae finish
    movzx eax, byte ptr [r12 + rbx]
    inc rbx
    shl eax, 16
    mov r9d, 1
    cmp rbx, r13
    jae encode_group
    movzx edx, byte ptr [r12 + rbx]
    inc rbx
    shl edx, 8
    or eax, edx
    inc r9d
    cmp rbx, r13
    jae encode_group
    movzx edx, byte ptr [r12 + rbx]
    inc rbx
    or eax, edx
    inc r9d

encode_group:
    mov edx, eax
    shr edx, 18
    movzx ecx, byte ptr [r15 + rdx]
    mov byte ptr [r14 + rbp], cl
    mov edx, eax
    shr edx, 12
    and edx, 63
    movzx ecx, byte ptr [r15 + rdx]
    mov byte ptr [r14 + rbp + 1], cl
    mov ecx, 61
    cmp r9d, 1
    je third_ready
    mov edx, eax
    shr edx, 6
    and edx, 63
    movzx ecx, byte ptr [r15 + rdx]
third_ready:
    mov byte ptr [r14 + rbp + 2], cl
    mov ecx, 61
    cmp r9d, 3
    jne fourth_ready
    and eax, 63
    movzx ecx, byte ptr [r15 + rax]
fourth_ready:
    mov byte ptr [r14 + rbp + 3], cl
    add rbp, 4
    add r8d, 4
    cmp r8d, 76
    jne encode_loop
    mov byte ptr [r14 + rbp], 10
    inc rbp
    xor r8d, r8d
    jmp encode_loop

finish:
    test r8d, r8d
    jz write_output
    mov byte ptr [r14 + rbp], 10
    inc rbp
write_output:
    test rbp, rbp
    jz exit_program
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, rbp
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
