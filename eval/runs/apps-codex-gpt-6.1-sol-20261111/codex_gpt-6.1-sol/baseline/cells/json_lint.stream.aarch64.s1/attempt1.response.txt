.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop
read_done:
    mov x21, #0
    bl value
    bl whitespace
    cmp x21, x20
    b.ne error
    mov x0, #1
    adrp x1, ok
    add x1, x1, :lo12:ok
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

peek:
    cmp x21, x20
    b.hs peek_eof
    ldrb w0, [x19, x21]
    ret
peek_eof:
    mov w0, #256
    ret

whitespace:
    cmp x21, x20
    b.hs peek_eof
    ldrb w0, [x19, x21]
    cmp w0, #32
    b.eq ws_next
    cmp w0, #9
    b.eq ws_next
    cmp w0, #10
    b.eq ws_next
    cmp w0, #13
    b.ne ws_return
ws_next:
    add x21, x21, #1
    b whitespace
ws_return:
    ret

value:
    stp x29, x30, [sp, #-16]!
    bl whitespace
    cmp w0, #123
    b.eq object
    cmp w0, #91
    b.eq array
    cmp w0, #34
    b.eq string_value
    cmp w0, #116
    b.eq true_value
    cmp w0, #102
    b.eq false_value
    cmp w0, #110
    b.eq null_value
    cmp w0, #45
    b.eq number
    sub w1, w0, #48
    cmp w1, #9
    b.ls number
    b error
value_done:
    ldp x29, x30, [sp], #16
    ret

true_value:
    adrp x3, true_text
    add x3, x3, :lo12:true_text
    b literal
false_value:
    adrp x3, false_text
    add x3, x3, :lo12:false_text
    b literal
null_value:
    adrp x3, null_text
    add x3, x3, :lo12:null_text
literal:
    ldrb w4, [x3], #1
    cbz w4, value_done
    bl peek
    cmp w0, w4
    b.ne error
    add x21, x21, #1
    b literal

object:
    add x21, x21, #1
    bl whitespace
    cmp w0, #125
    b.eq container_end
object_member:
    cmp w0, #34
    b.ne error
    bl parse_string
    bl whitespace
    cmp w0, #58
    b.ne error
    add x21, x21, #1
    bl value
    bl whitespace
    cmp w0, #125
    b.eq container_end
    cmp w0, #44
    b.ne error
    add x21, x21, #1
    bl whitespace
    b object_member

array:
    add x21, x21, #1
    bl whitespace
    cmp w0, #93
    b.eq container_end
array_element:
    bl value
    bl whitespace
    cmp w0, #93
    b.eq container_end
    cmp w0, #44
    b.ne error
    add x21, x21, #1
    b array_element
container_end:
    add x21, x21, #1
    b value_done

string_value:
    bl parse_string
    b value_done

parse_string:
    stp x29, x30, [sp, #-16]!
    add x21, x21, #1
string_loop:
    bl peek
    cmp w0, #256
    b.eq error
    cmp w0, #32
    b.lo error
    cmp w0, #34
    b.eq string_end
    cmp w0, #92
    b.eq escape
    add x21, x21, #1
    b string_loop
string_end:
    add x21, x21, #1
    ldp x29, x30, [sp], #16
    ret
escape:
    add x21, x21, #1
    bl peek
    cmp w0, #117
    b.eq unicode
    cmp w0, #34
    b.eq escape_ok
    cmp w0, #92
    b.eq escape_ok
    cmp w0, #47
    b.eq escape_ok
    cmp w0, #98
    b.eq escape_ok
    cmp w0, #102
    b.eq escape_ok
    cmp w0, #110
    b.eq escape_ok
    cmp w0, #114
    b.eq escape_ok
    cmp w0, #116
    b.ne error
escape_ok:
    add x21, x21, #1
    b string_loop
unicode:
    add x21, x21, #1
    mov w5, #4
hex_loop:
    bl peek
    sub w1, w0, #48
    cmp w1, #9
    b.ls hex_ok
    orr w1, w0, #32
    sub w1, w1, #97
    cmp w1, #5
    b.hi error

    add x21, x21, #1
    subs w5, w5, #1
    b.ne hex_loop

    b.ne integer
    add x21, x21, #1
    bl peek

    cmp w0, #48
    b.eq zero_integer
    sub w1, w0, #49
    cmp w1, #8
    b.hi error

    bl peek
    sub w1, w0, #48
    cmp w1, #9

    add x21, x21, #1

exponent_digits:

    sub w1, w0, #48
    cmp w1, #9

    mov w11, #114

    mov w11, #114
