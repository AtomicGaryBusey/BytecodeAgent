.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x23, jumps
    add x23, x23, :lo12:jumps
    adrp x25, brackets
    add x25, x25, :lo12:brackets
    mov x22, #0
    mov x24, #0
    mov x26, #0

parse_loop:
    cmp x26, x20
    b.hs parse_done
    ldrb w3, [x19, x26]
    add x26, x26, #1
    cmp w3, #33
    b.eq parse_done
    cmp w3, #91
    b.eq open_bracket
    cmp w3, #93
    b.eq close_bracket
    cmp w3, #62
    b.eq save_command
    cmp w3, #60
    b.eq save_command
    cmp w3, #43
    b.eq save_command
    cmp w3, #45
    b.eq save_command
    cmp w3, #46
    b.eq save_command
    cmp w3, #44
    b.ne parse_loop
save_command:
    strb w3, [x21, x22]
    add x22, x22, #1
    b parse_loop

open_bracket:
    str w22, [x25, x24, lsl #2]
    add x24, x24, #1
    b save_command

close_bracket:
    cbz x24, invalid
    sub x24, x24, #1
    ldr w4, [x25, x24, lsl #2]
    add w5, w22, #1
    str w5, [x23, x4, lsl #2]
    add w5, w4, #1
    str w5, [x23, x22, lsl #2]
    b save_command

parse_done:
    cbnz x24, invalid
    adrp x27, tape
    add x27, x27, :lo12:tape
    adrp x28, output
    add x28, x28, :lo12:output
    mov x9, #0
    mov x10, #0
    mov x11, #0

execute:
    cmp x9, x22
    b.hs finish
    ldrb w3, [x21, x9]
    add x9, x9, #1
    cmp w3, #43
    b.eq increment
    cmp w3, #45
    b.eq decrement
    cmp w3, #62
    b.eq right
    cmp w3, #60
    b.eq left
    cmp w3, #91
    b.eq loop_open
    cmp w3, #93
    b.eq loop_close
    cmp w3, #46
    b.eq emit

    mov w4, #0
    cmp x26, x20
    b.hs store_input
    ldrb w4, [x19, x26]
    add x26, x26, #1
store_input:
    strb w4, [x27, x10]
    b execute

increment:
    ldrb w4, [x27, x10]
    add w4, w4, #1
    strb w4, [x27, x10]
    b execute
decrement:
    ldrb w4, [x27, x10]
    sub w4, w4, #1
    strb w4, [x27, x10]
    b execute
right:
    add x10, x10, #1
    b execute
left:
    sub x10, x10, #1
    b execute
loop_open:
    ldrb w4, [x27, x10]
    cbnz w4, execute
    sub x4, x9, #1
    ldr w9, [x23, x4, lsl #2]
    b execute
loop_close:
    ldrb w4, [x27, x10]
    cbz w4, execute
    sub x4, x9, #1
    ldr w9, [x23, x4, lsl #2]
    b execute
emit:
    ldrb w4, [x27, x10]
    strb w4, [x28, x11]
    add x11, x11, #1
    b execute

finish:
    cbz x11, success
    mov x0, #1
    mov x1, x28
    mov x2, x11
    mov x8, #64
    svc #0
success:
    mov x0, #0
    mov x8, #93
    svc #0
invalid:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 16
input:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
brackets:
    .zero 4000
tape:
    .zero 30000
output:
    .zero 65536
