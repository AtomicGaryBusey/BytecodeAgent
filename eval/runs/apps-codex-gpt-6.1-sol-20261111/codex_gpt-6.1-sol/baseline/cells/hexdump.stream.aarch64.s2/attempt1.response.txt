.text
.globl _start
_start:
    adrp x22, input
    add x22, x22, :lo12:input
    adrp x23, output
    add x23, x23, :lo12:output
    adrp x24, digits
    add x24, x24, :lo12:digits
    mov x19, #0

read_loop:
    mov x0, #0
    add x1, x22, x19
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x19
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x20, #0

row_loop:
    cmp x20, x19
    b.hs final_offset
    sub x21, x19, x20
    mov x0, #16
    cmp x21, x0
    csel x21, x21, x0, lo

    mov x0, x20
    mov x1, x23
    bl hex8
    mov w5, #32
    strb w5, [x1], #1
    strb w5, [x1], #1
    mov x6, #0
    add x7, x22, x20

slot_loop:
    cmp x6, x21
    b.hs empty_slot
    ldrb w9, [x7, x6]
    lsr w10, w9, #4
    ldrb w10, [x24, x10]
    strb w10, [x1], #1
    and w10, w9, #15
    ldrb w10, [x24, x10]
    strb w10, [x1], #1
    strb w5, [x1], #1
    b slot_done

empty_slot:
    strb w5, [x1], #1
    strb w5, [x1], #1
    strb w5, [x1], #1

slot_done:
    cmp x6, #7
    b.ne no_gap
    strb w5, [x1], #1
no_gap:
    add x6, x6, #1
    cmp x6, #16
    b.lo slot_loop

    strb w5, [x1], #1
    mov w9, #124
    strb w9, [x1], #1
    mov x6, #0

ascii_loop:
    ldrb w9, [x7, x6]
    sub w10, w9, #32
    cmp w10, #94
    mov w11, #46
    csel w9, w9, w11, ls
    strb w9, [x1], #1
    add x6, x6, #1
    cmp x6, x21
    b.lo ascii_loop

    mov w9, #124
    strb w9, [x1], #1
    mov w9, #10
    strb w9, [x1], #1
    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0
    add x20, x20, x21
    b row_loop

final_offset:
    mov x0, x19
    mov x1, x23
    bl hex8
    mov w9, #10
    strb w9, [x1]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

hex8:
    mov x2, #28
hex8_loop:
    lsrv x3, x0, x2
    and x3, x3, #15
    ldrb w4, [x24, x3]
    strb w4, [x1], #1
    subs x2, x2, #4
    b.ge hex8_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 80
