.globl _start
.section .rodata
okmsg:
.ascii "ok\n"
errmsg:
.ascii "error at "
littrue:
.asciz "true"
litfalse:
.asciz "false"
litnull:
.asciz "null"

.bss
buffer:
.zero 65537
outbuf:
.zero 32

.text
_start:
lea r12, [rip + buffer]
xor r14d, r14d
read_loop:
xor eax, eax
xor edi, edi
lea rsi, [r12 + r14]
mov edx, 65537
sub rdx, r14
syscall
test rax, rax
jz read_done
add r14, rax
jmp read_loop
read_done:
lea r13, [r12 + r14]
call whitespace
call value
call whitespace
cmp r12, r13
jne error
mov eax, 1
mov edi, 1
lea rsi, [rip + okmsg]
mov edx, 3
syscall
xor edi, edi
jmp finish

peek:
mov eax, -1
cmp r12, r13
jae peek_done
movzx eax, byte ptr [r12]
peek_done:
ret

whitespace:
call peek
cmp eax, 32
je ws_next
cmp eax, 9
je ws_next
cmp eax, 10
je ws_next
cmp eax, 13
je ws_next
ret
ws_next:
inc r12
jmp whitespace

value:
call peek
cmp eax, 34
je string
cmp eax, 123
je object
cmp eax, 91
je array
cmp eax, 116
je true_value
cmp eax, 102
je false_value
cmp eax, 110
je null_value
cmp eax, 45
je number
cmp eax, 48
jb error
cmp eax, 57
jbe number
jmp error

true_value:
lea r8, [rip + littrue]
jmp literal
false_value:
lea r8, [rip + litfalse]
jmp literal
null_value:
lea r8, [rip + litnull]
literal:
movzx edx, byte ptr [r8]
test edx, edx
jz literal_done
call peek
cmp eax, edx
jne error
inc r8
inc r12
jmp literal
literal_done:
ret

string:
inc r12
string_loop:
call peek
cmp eax, -1
je error
cmp eax, 32
jb error
cmp eax, 34
je string_done
cmp eax, 92
je escape
inc r12
jmp string_loop
string_done:
inc r12
ret
escape:
inc r12
call peek
cmp eax, 117
je unicode
cmp eax, 34
je escape_done
cmp eax, 92
je escape_done
cmp eax, 47
je escape_done
cmp eax, 98
je escape_done
cmp eax, 102
je escape_done
cmp eax, 110
je escape_done
cmp eax, 114
je escape_done
cmp eax, 116
jne error
escape_done:
inc r12
jmp string_loop
unicode:
inc r12
mov r9d, 4
hex_loop:
call peek
cmp eax, 48
jb error
cmp eax, 57
jbe hex_ok
cmp eax, 65
jb error
cmp eax, 70
jbe hex_ok
cmp eax, 97
jb error
cmp eax, 102
ja error
hex_ok:
inc r12
dec r9d
jnz hex_loop
jmp string_loop

number:
cmp eax, 45
jne integer
inc r12
call peek
integer:
cmp eax, 48
je zero_integer
cmp eax, 49
jb error
cmp eax, 57
ja error
integer_loop:
inc r12
call peek
cmp eax, 48
jb fraction
cmp eax, 57
jbe integer_loop
jmp fraction
zero_integer:
inc r12
call peek
fraction:
cmp eax, 46
jne exponent
inc r12
call peek
cmp eax, 48
jb error
cmp eax, 57
ja error
fraction_loop:
inc r12
call peek
cmp eax, 48
jb exponent
cmp eax, 57
jbe fraction_loop
exponent:
cmp eax, 101
je exponent_start
cmp eax, 69
jne number_done
exponent_start:
inc r12
call peek
cmp eax, 43
je exponent_sign
cmp eax, 45
jne exponent_digit
exponent_sign:
inc r12
call peek
exponent_digit:
cmp eax, 48
jb error
cmp eax, 57
ja error
exponent_loop:
inc r12
call peek
cmp eax, 48
jb number_done
cmp eax, 57
jbe exponent_loop
number_done:
ret

array:
inc r12
call whitespace
cmp eax, 93
je container_done
array_value:
call value
call whitespace
cmp eax, 93
je container_done
cmp eax, 44
jne error
inc r12
call whitespace
jmp array_value

object:
inc r12
call whitespace
cmp eax, 125
je container_done
object_member:
cmp eax, 34
jne error
call string
call whitespace
cmp eax, 58
jne error
inc r12
call whitespace
call value
call whitespace
cmp eax, 125
je container_done
cmp eax, 44
jne error
inc r12
call whitespace
jmp object_member
container_done:
inc r12
ret

error:
lea rax, [rip + buffer]
sub r12, rax
lea r8, [rip + outbuf + 31]
mov byte ptr [r8], 10
mov rax, r12
mov r9d, 10
decimal_loop:
xor edx, edx
div r9
add dl, 48
dec r8
mov byte ptr [r8], dl
test rax, rax
jnz decimal_loop
mov eax, 1
mov edi, 1
lea rsi, [rip + errmsg]
mov edx, 9
syscall
mov rsi, r8
lea rdx, [rip + outbuf + 32]
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
mov edi, 1
finish:
mov eax, 60
syscall
