.bss
.balign 16
input:
.zero 32769
lines:
.zero 16000
output:
.zero 33768

.text
.globl _start
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    js read_more
    add r13, rax
    jmp read_more

read_done:
    lea r14, [rip + lines]
    xor r15d, r15d
    xor ecx, ecx
    xor edx, edx
parse:
    cmp rcx, r13
    jae parse_end
    cmp byte ptr [r12 + rcx], 10
    jne next_byte
    mov rax, r15
    shl rax, 4
    lea rbx, [r12 + rdx]
    mov qword ptr [r14 + rax], rbx
    mov rbx, rcx
    sub rbx, rdx
    mov qword ptr [r14 + rax + 8], rbx
    inc r15
    lea rdx, [rcx + 1]
next_byte:
    inc rcx
    jmp parse
parse_end:
    cmp rdx, r13
    je sort_begin
    mov rax, r15
    shl rax, 4
    lea rbx, [r12 + rdx]
    mov qword ptr [r14 + rax], rbx
    sub r13, rdx
    mov qword ptr [r14 + rax + 8], r13
    inc r15

sort_begin:
    mov r8d, 1
sort_outer:
    cmp r8, r15
    jae emit
    mov rax, r8
    shl rax, 4
    mov r9, qword ptr [r14 + rax]
    mov r10, qword ptr [r14 + rax + 8]
    mov rbx, r8
sort_inner:
    test rbx, rbx
    jz insert
    mov rdx, rbx
    dec rdx
    shl rdx, 4
    mov rsi, qword ptr [r14 + rdx]
    mov rdi, r9
    mov rax, qword ptr [r14 + rdx + 8]
    mov rcx, rax
    cmp rcx, r10
    cmova rcx, r10
    test rcx, rcx
    jz length_compare
    repe cmpsb
    ja shift
    jb insert
length_compare:
    cmp rax, r10
    jbe insert
shift:
    mov rsi, qword ptr [r14 + rdx]
    mov qword ptr [r14 + rdx + 16], rsi
    mov qword ptr [r14 + rdx + 24], rax
    dec rbx
    jmp sort_inner
insert:
    mov rdx, rbx
    shl rdx, 4
    mov qword ptr [r14 + rdx], r9
    mov qword ptr [r14 + rdx + 8], r10
    inc r8
    jmp sort_outer

emit:
    lea rdi, [rip + output]
    xor ebx, ebx
copy_line:
    cmp rbx, r15
    jae write_output
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r14 + rax]
    mov rcx, qword ptr [r14 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp copy_line

write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz done
    mov eax, 1
    mov edi, 1
    syscall
done:
    mov eax, 60
    xor edi, edi
    syscall
