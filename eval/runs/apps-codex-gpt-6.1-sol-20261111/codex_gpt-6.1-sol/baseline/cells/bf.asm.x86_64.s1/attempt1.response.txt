.globl _start
.bss
input:
.zero 65537
code:
.zero 65536
.balign 4
jumps:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    lea rbx, [rip + input]
    lea rbp, [rip + code]
    lea r13, [rip + jumps]
    lea r14, [rip + brackets]
parse:
    cmp r8, r12
    jae no_separator
    movzx eax, byte ptr [rbx + r8]
    inc r8
    cmp al, 33
    je parsed
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne parse
save_command:
    mov byte ptr [rbp + r9], al
    inc r9
    jmp parse
open_bracket:
    mov dword ptr [r14 + r10*4], r9d
    inc r10d
    jmp save_command
close_bracket:
    test r10d, r10d
    jz invalid
    dec r10d
    mov edx, dword ptr [r14 + r10*4]
    lea ecx, [r9 + 1]
    mov dword ptr [r13 + rdx*4], ecx
    inc edx
    mov dword ptr [r13 + r9*4], edx
    jmp save_command
no_separator:
    mov r8, r12
parsed:
    test r10d, r10d
    jnz invalid
    xor r10d, r10d
    xor r15d, r15d
    lea r14, [rip + tape]
    lea rdi, [rip + output]
execute:
    cmp r10, r9
    jae finished
    movzx eax, byte ptr [rbp + r10]
    inc r10
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 46
    je emit
    cmp al, 44
    je consume
    cmp al, 91
    je loop_open
    cmp byte ptr [r14], 0
    je execute
    mov r10d, dword ptr [r13 + r10*4 - 4]
    jmp execute
loop_open:
    cmp byte ptr [r14], 0
    jne execute
    mov r10d, dword ptr [r13 + r10*4 - 4]
    jmp execute
increment:
    inc byte ptr [r14]
    jmp execute
decrement:
    dec byte ptr [r14]
    jmp execute
right:
    inc r14
    jmp execute
left:
    dec r14
    jmp execute
emit:
    mov al, byte ptr [r14]
    mov byte ptr [rdi + r15], al
    inc r15
    jmp execute
consume:
    xor eax, eax
    cmp r8, r12
    jae store_input
    mov al, byte ptr [rbx + r8]
    inc r8
store_input:
    mov byte ptr [r14], al
    jmp execute
finished:
    mov rdx, r15
    lea rsi, [rip + output]
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall
