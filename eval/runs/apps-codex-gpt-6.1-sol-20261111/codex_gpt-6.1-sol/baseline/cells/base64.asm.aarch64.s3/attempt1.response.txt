.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    mov x21, #65536
read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, encode_start
    add x20, x20, x0
    b read_loop

encode_start:
    adrp x22, alphabet
    add x22, x22, :lo12:alphabet
    adrp x23, output
    add x23, x23, :lo12:output
    mov x24, x23
    mov x25, #0
    mov w26, #10
    mov w27, #61
    cbz x20, finish

encode_loop:
    ldrb w3, [x19], #1
    sub x20, x20, #1
    lsr w4, w3, #2
    ldrb w4, [x22, w4, uxtw]
    strb w4, [x24], #1
    and w5, w3, #3
    lsl w5, w5, #4
    cbz x20, one_byte

    ldrb w6, [x19], #1
    sub x20, x20, #1
    lsr w7, w6, #4
    orr w5, w5, w7
    ldrb w5, [x22, w5, uxtw]
    strb w5, [x24], #1
    and w6, w6, #15
    lsl w6, w6, #2
    cbz x20, two_bytes

    ldrb w7, [x19], #1
    sub x20, x20, #1
    lsr w9, w7, #6
    orr w6, w6, w9
    ldrb w6, [x22, w6, uxtw]
    strb w6, [x24], #1
    and w7, w7, #63
    ldrb w7, [x22, w7, uxtw]
    strb w7, [x24], #1
    b group_done

one_byte:
    ldrb w5, [x22, w5, uxtw]
    strb w5, [x24], #1
    strb w27, [x24], #1
    strb w27, [x24], #1
    b group_done

two_bytes:
    ldrb w6, [x22, w6, uxtw]
    strb w6, [x24], #1
    strb w27, [x24], #1

group_done:
    add x25, x25, #4
    cmp x25, #76
    b.ne line_pending
    strb w26, [x24], #1
    mov x25, #0
line_pending:
    cbnz x20, encode_loop
    cbz x25, write_output
    strb w26, [x24], #1

write_output:
    mov x0, #1
    mov x1, x23
    sub x2, x24, x23
    mov x8, #64
    svc #0

finish:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input:
    .zero 65536
output:
    .zero 66560
