.globl _start

.bss
input:
.zero 65537
output:
.zero 393216
.p2align 3
values:
.zero 8000

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    lea r14, [r12 + r13]
    lea r15, [rip + output]

next_line:
    cmp r12, r14
    je finish
    mov r13, r12
find_end:
    cmp r13, r14
    je line_ready
    cmp byte ptr [r13], 10
    je line_ready
    inc r13
    jmp find_end

line_ready:
    mov rsi, r12
    xor r8d, r8d
    lea rbx, [rip + values]

next_token:
    cmp rsi, r13
    je end_line
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator
    mov rdi, rsi
token_end:
    cmp rdi, r13
    je token_ready
    movzx eax, byte ptr [rdi]
    cmp al, 32
    je token_ready
    cmp al, 9
    je token_ready
    inc rdi
    jmp token_end

skip_separator:
    inc rsi
    jmp next_token

token_ready:
    mov rcx, rdi
    sub rcx, rsi
    cmp rcx, 1
    jne literal
    movzx r9d, byte ptr [rsi]
    cmp r9b, 43
    je operator
    cmp r9b, 45
    je operator
    cmp r9b, 42
    je operator
    cmp r9b, 47
    je operator
    cmp r9b, 37
    je operator

literal:
    xor r10d, r10d
    cmp byte ptr [rsi], 45
    jne literal_digits
    mov r10d, 1
    inc rsi
    dec rcx
literal_digits:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    cmp rsi, rdi
    jb digit_loop
    test r10d, r10d
    jz push_literal
    neg rax
push_literal:
    mov qword ptr [rbx + r8*8], rax
    inc r8
    jmp next_token

operator:
    cmp r8, 2
    jb line_error
    dec r8
    mov r10, qword ptr [rbx + r8*8]
    mov rax, qword ptr [rbx + r8*8 - 8]
    cmp r9b, 43
    je do_add
    cmp r9b, 45
    je do_sub
    cmp r9b, 42
    je do_mul
    test r10, r10
    jz line_error
    cmp r10, -1
    jne do_div
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je line_error
do_div:
    cqo
    idiv r10
    cmp r9b, 37
    jne store_result
    mov rax, rdx
    jmp store_result
do_add:
    add rax, r10
    jmp store_result
do_sub:
    sub rax, r10
    jmp store_result
do_mul:
    imul rax, r10
store_result:
    mov qword ptr [rbx + r8*8 - 8], rax
    mov rsi, rdi
    jmp next_token

end_line:
    cmp r8, 1
    jne line_error
    mov rax, qword ptr [rbx]
    test rax, rax
    jns magnitude_ready
    mov byte ptr [r15], 45
    inc r15
    neg rax
magnitude_ready:
    mov rdi, r15
    mov ecx, 10
decimal_loop:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [r15], dl
    inc r15
    test rax, rax
    jnz decimal_loop
    lea rsi, [r15 - 1]
reverse_digits:
    cmp rdi, rsi
    jae digits_done
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp reverse_digits
digits_done:
    mov byte ptr [r15], 10
    inc r15
    jmp advance_line

line_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6

advance_line:
    mov r12, r13
    cmp r12, r14
    je finish
    inc r12
    jmp next_line

finish:
    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
