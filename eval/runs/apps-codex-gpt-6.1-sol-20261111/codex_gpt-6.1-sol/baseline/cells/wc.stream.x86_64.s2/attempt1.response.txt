.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz output
    js read_more
    add r14, rax
    lea r8, [rsi + rax]
scan:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp eax, 10
    jne not_newline
    inc r12
not_newline:
    cmp eax, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, r8
    jb scan
    jmp read_more

output:
    lea rdi, [rip + result]
    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + result]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

decimal:
    lea r9, [rip + digits + 32]
    mov r10, r9
    mov ecx, 10
digit_loop:
    xor edx, edx
    div rcx
    add dl, 48
    dec r9
    mov byte ptr [r9], dl
    test rax, rax
    jnz digit_loop
copy_digits:
    mov al, byte ptr [r9]
    mov byte ptr [rdi], al
    inc r9
    inc rdi
    cmp r9, r10
    jb copy_digits
    ret

.bss
.balign 16
buffer:
    .zero 4096
result:
    .zero 64
digits:
    .zero 32
