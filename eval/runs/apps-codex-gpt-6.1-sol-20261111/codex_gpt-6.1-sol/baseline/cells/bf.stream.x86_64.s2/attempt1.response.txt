.globl _start
.text
_start:
    lea r12, [rip + source]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    lea r14, [rip + commands]
    lea r15, [rip + jumps]
    xor ebx, ebx
    xor ecx, ecx
    xor r8d, r8d
    mov r9, r13
parse:
    cmp rbx, r13
    jae parsed
    movzx eax, byte ptr [r12 + rbx]
    inc rbx
    cmp al, 33
    je delimiter
    cmp al, 62
    je p_right
    cmp al, 60
    je p_left
    cmp al, 43
    je p_plus
    cmp al, 45
    je p_minus
    cmp al, 46
    je p_out
    cmp al, 44
    je p_in
    cmp al, 91
    je p_open
    cmp al, 93
    je p_close
    jmp parse
p_right:
    xor eax, eax
    jmp store
p_left:
    mov eax, 1
    jmp store
p_plus:
    mov eax, 2
    jmp store
p_minus:
    mov eax, 3
    jmp store
p_out:
    mov eax, 4
    jmp store
p_in:
    mov eax, 5
    jmp store
p_open:
    mov dword ptr [brackets + r8*4], ecx
    inc r8d
    mov eax, 6
    jmp store
p_close:
    test r8d, r8d
    jz invalid
    dec r8d
    mov edx, dword ptr [brackets + r8*4]
    lea eax, [rcx + 1]
    mov dword ptr [r15 + rdx*4], eax
    inc edx
    mov dword ptr [r15 + rcx*4], edx
    mov eax, 7
store:
    mov byte ptr [r14 + rcx], al
    inc ecx
    jmp parse
delimiter:
    mov r9, rbx
parsed:
    test r8d, r8d
    jnz invalid
    mov r11d, ecx
    mov dword ptr [rip + program_length], ecx
    mov r10, r13
    lea r12, [rip + commands]
    lea r13, [rip + jumps]
    lea r14, [rip + tape]
    lea r15, [rip + output]
    xor ebx, ebx
    xor r8d, r8d
    xor ebp, ebp
execute:
    cmp ebx, dword ptr [rip + program_length]
    jae finished
    movzx eax, byte ptr [r12 + rbx]
    inc ebx
    jmp qword ptr [dispatch + rax*8]
right:
    inc r8d
    jmp execute
left:
    dec r8d
    jmp execute
plus:
    inc byte ptr [r14 + r8]
    jmp execute
minus:
    dec byte ptr [r14 + r8]
    jmp execute
put:
    mov al, byte ptr [r14 + r8]
    mov byte ptr [r15 + rbp], al
    inc ebp
    jmp execute
get:
    xor eax, eax
    cmp r9, r10
    jae get_store
    mov al, byte ptr [source + r9]
    inc r9
get_store:
    mov byte ptr [r14 + r8], al
    jmp execute
open:
    cmp byte ptr [r14 + r8], 0
    jne execute
    mov ebx, dword ptr [r13 + rbx*4 - 4]
    jmp execute
close:
    cmp byte ptr [r14 + r8], 0
    je execute
    mov ebx, dword ptr [r13 + rbx*4 - 4]
    jmp execute
finished:
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, rbp
    syscall
    xor edi, edi
    jmp quit
invalid:
    mov edi, 1
quit:
    mov eax, 60
    syscall

.section .rodata
.balign 8
dispatch:
    .quad right, left, plus, minus, put, get, open, close

.bss
.balign 8
source:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
brackets:
    .zero 4000
program_length:
    .zero 4
tape:
    .zero 30000
output:
    .zero 65536
