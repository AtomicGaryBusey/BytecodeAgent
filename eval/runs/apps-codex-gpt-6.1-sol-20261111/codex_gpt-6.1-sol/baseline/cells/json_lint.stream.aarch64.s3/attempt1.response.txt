.globl _start

    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    add x2, x2, #1
    mov x8, #63

    add x20, x20, x0

    mov x21, #0
    bl whitespace
    bl value
    bl whitespace
    cmp x21, x20
    b.ne error
    mov x0, #1
    adrp x1, ok_message
    add x1, x1, :lo12:ok_message
    mov x2, #3
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

peek:
    mov w9, #256
    cmp x21, x20
    b.hs 1f
    ldrb w9, [x19, x21]

    ret

whitespace:
    cmp x21, x20
    b.hs 2f
    ldrb w9, [x19, x21]
    cmp w9, #32
    b.eq 1f
    cmp w9, #9
    b.eq 1f
    cmp w9, #10
    b.eq 1f
    cmp w9, #13
    b.ne 2f

    add x21, x21, #1
    b whitespace

    ret

value:
    stp x29, x30, [sp, #-16]!
    bl peek
    cmp w9, #34
    b.eq value_string
    cmp w9, #123
    b.eq object
    cmp w9, #91
    b.eq array
    cmp w9, #116
    b.eq literal_true
    cmp w9, #102
    b.eq literal_false
    cmp w9, #110
    b.eq literal_null
    bl number
value_done:
    ldp x29, x30, [sp], #16
    ret
value_string:
    bl string
    b value_done

    adrp x10, true_text
    add x10, x10, :lo12:true_text
    b literal
literal_false:
    adrp x10, false_text
    add x10, x10, :lo12:false_text
    b literal

    add x10, x10, :lo12:null_text

    bl peek
    cmp w9, w11
    b.ne error
    add x21, x21, #1
    b literal

    bl peek
    cmp w9, #125
    b.eq container_end

    b.ne error

    bl whitespace
    bl peek
    cmp w9, #58
    b.ne error
    add x21, x21, #1
    bl whitespace
    bl value
    bl whitespace
    bl peek
    cmp w9, #125
    b.eq container_end
    cmp w9, #44
    b.ne error
    add x21, x21, #1
    bl whitespace
    bl peek
    b object_member

array:

    bl whitespace
    bl peek
    cmp w9, #93
    b.eq container_end

    bl whitespace
    bl peek
    cmp w9, #93
    b.eq container_end
    cmp w9, #44
    b.ne error
    add x21, x21, #1

    add x21, x21, #1

    b.eq error
    cmp w9, #32

    cmp w9, #34
    b.eq string_end

    b string_loop

    bl peek
    cmp w9, #117
    b.eq unicode_escape
    cmp w9, #34
    b.eq escape_done
    cmp w9, #92
    b.eq escape_done
    cmp w9, #47
    b.eq escape_done
    cmp w9, #98
    b.eq escape_done
    cmp w9, #102
    b.eq escape_done
    cmp w9, #110
    b.eq escape_done
    cmp w9, #114
    b.eq escape_done
    cmp w9, #116

hex_loop:

    bl peek
    sub w13, w9, #48
