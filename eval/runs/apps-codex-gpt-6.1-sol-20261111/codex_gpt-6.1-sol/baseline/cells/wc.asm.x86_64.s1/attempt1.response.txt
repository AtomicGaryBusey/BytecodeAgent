.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 8192
    syscall
    test rax, rax
    jz output
    js read_loop
    add r14, rax
    lea rdx, [rsi + rax]

count_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne check_space
    inc r12
check_space:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte
whitespace:
    xor r15d, r15d
next_byte:
    cmp rsi, rdx
    jne count_loop
    jmp read_loop

output:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

decimal:
    lea r8, [rip + scratch + 32]
    mov r9, r8
    mov r10d, 10
decimal_loop:
    xor edx, edx
    div r10
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz decimal_loop
decimal_copy:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jne decimal_copy
    ret

.bss
.balign 16
buffer:
    .zero 8192
outbuf:
    .zero 80
scratch:
    .zero 32
