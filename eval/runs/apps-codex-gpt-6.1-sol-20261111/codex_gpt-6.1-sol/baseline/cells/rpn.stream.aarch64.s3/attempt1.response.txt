.text

    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, x19
    mov x21, #0

    mov x0, #0
    mov x1, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x21
    mov x8, #63

    add x20, x20, x0
    add x21, x21, x0

    add x21, x21, :lo12:values
    adrp x27, output
    add x27, x27, :lo12:output
    mov x28, x27

    cmp x19, x20
    b.hs finish
    mov x22, #0
    mov x23, #0

line_loop:
    cmp x19, x20
    b.hs end_line
    ldrb w0, [x19]
    cmp w0, #10
    b.eq newline
    cbnz x23, skip_byte
    cmp w0, #32
    b.eq skip_byte
    cmp w0, #9
    b.eq skip_byte

    mov x25, x19

    add x19, x19, #1
    cmp x19, x20
    b.hs token_ready
    ldrb w0, [x19]
    cmp w0, #10
    b.eq token_ready
    cmp w0, #32
    b.eq token_ready
    cmp w0, #9
    b.ne token_scan

token_ready:
    sub x26, x19, x25
    ldrb w0, [x25]
    cmp x26, #1
    b.ne literal
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42
    b.eq operator
    cmp w0, #47
    b.eq operator
    cmp w0, #37
    b.eq operator

literal:
    mov x1, x25
    mov x2, x26
    mov x3, #0
    cmp w0, #45
    b.ne literal_digits
    mov x3, #1
    add x1, x1, #1
    sub x2, x2, #1
literal_digits:
    cbz x2, bad_line
    cmp x2, #18
    b.hi bad_line
    mov x4, #0
    mov x5, #10

    ldrb w6, [x1], #1
    sub w6, w6, #48
    cmp w6, #9
    b.hi bad_line
    madd x4, x4, x5, x6
    subs x2, x2, #1
    b.ne digit_loop
    cmp x3, #0
    cneg x4, x4, ne
    str x4, [x21, x22, lsl #3]
    add x22, x22, #1
    b line_loop

operator:
    cmp x22, #2
    b.lo bad_line
    sub x22, x22, #1
    ldr x2, [x21, x22, lsl #3]
    sub x22, x22, #1
    ldr x1, [x21, x22, lsl #3]
    cmp w0, #43
    b.eq do_add
    cmp w0, #45
    b.eq do_sub
    cmp w0, #42
    b.eq do_mul
    cbz x2, bad_line
    mov x3, #0x8000000000000000
    cmp x1, x3
    b.ne division
    cmn x2, #1
    b.eq bad_line
division:

    cmp w0, #47
    b.eq store_result
    msub x3, x3, x2, x1
    b store_result
do_add:
    add x3, x1, x2
    b store_result
do_sub:
    sub x3, x1, x2
    b store_result
do_mul:
    mul x3, x1, x2

    str x3, [x21, x22, lsl #3]
    add x22, x22, #1
    b line_loop

bad_line:
    mov x23, #1
    b line_loop

    add x19, x19, #1
    b line_loop

    add x19, x19, #1
end_line:
    cbnz x23, emit_error
    cmp x22, #1
    b.ne emit_error
    ldr x0, [x21]
    tbz x0, #63, positive
    mov w1, #45
    strb w1, [x28], #1
    neg x0, x0

    add x1, x1, :lo12:digits
    mov x2, x1
    mov x3, #10

    msub x5, x4, x3, x0
    add w5, w5, #48
    strb w5, [x2], #1
    mov x0, x4
    cbnz x0, decimal_loop

    ldrb w5, [x2, #-1]!
    strb w5, [x28], #1
    cmp x2, x1
    b.ne copy_digits
    mov w0, #10
    strb w0, [x28], #1
    b next_line

    strb w0, [x28], #1
    mov w0, #114
    strb w0, [x28], #1
    strb w0, [x28], #1
    mov w0, #111
    strb w0, [x28], #1
    mov w0, #114
    strb w0, [x28], #1
    mov w0, #10
    strb w0, [x28], #1

    cbz x2, exit_program
    mov x0, #1
    mov x1, x27
    mov x8, #64

    mov x0, #0
    mov x8, #93
