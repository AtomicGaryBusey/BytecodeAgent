.globl _start
.bss
.balign 16
input:
.zero 32769
output:
.zero 33769
array_a:
.zero 16000
array_b:
.zero 16000

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop

read_done:
    lea rdi, [rip + array_a]
    xor r15d, r15d
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r13
    jae parse_end
    cmp byte ptr [r12 + rcx], 10
    jne parse_next
    lea rax, [r12 + rdx]
    mov qword ptr [rdi], rax
    mov rax, rcx
    sub rax, rdx
    mov qword ptr [rdi + 8], rax
    add rdi, 16
    inc r15
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_end:
    cmp rdx, r13
    jae parsed
    lea rax, [r12 + rdx]
    mov qword ptr [rdi], rax
    sub r13, rdx
    mov qword ptr [rdi + 8], r13
    inc r15
parsed:
    lea r12, [rip + array_a]
    lea r13, [rip + array_b]
    mov r14d, 1
sort_pass:
    cmp r14, r15
    jae sorted
    xor ebx, ebx
    xor ebp, ebp
merge_run:
    cmp rbx, r15
    jae pass_done
    mov r8, rbx
    lea r10, [rbx + r14]
    cmp r10, r15
    cmova r10, r15
    mov r9, r10
    lea r11, [r10 + r14]
    cmp r11, r15
    cmova r11, r15
merge_loop:
    cmp r8, r10
    jae take_right_check
    cmp r9, r11
    jae take_left
    mov rsi, r8
    shl rsi, 4
    add rsi, r12
    mov rdi, r9
    shl rdi, 4
    add rdi, r12
    call compare_lines
    test eax, eax
    jle take_left
take_right:
    mov rax, r9
    inc r9
    jmp copy_item
take_right_check:
    cmp r9, r11
    jb take_right
    mov rbx, r11
    jmp merge_run
take_left:
    mov rax, r8
    inc r8
copy_item:
    shl rax, 4
    movdqu xmm0, xmmword ptr [r12 + rax]
    mov rdx, rbp
    shl rdx, 4
    movdqu xmmword ptr [r13 + rdx], xmm0
    inc rbp
    jmp merge_loop
pass_done:
    xchg r12, r13
    shl r14, 1
    jmp sort_pass

sorted:
    lea rdi, [rip + output]
    xor ebx, ebx
output_loop:
    cmp rbx, r15
    jae output_done
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r12 + rax]
    mov rcx, qword ptr [r12 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp output_loop
output_done:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall

compare_lines:
    push r8
    push r9
    mov r8, qword ptr [rsi + 8]
    mov r9, qword ptr [rdi + 8]
    mov rsi, qword ptr [rsi]
    mov rdi, qword ptr [rdi]
    mov rcx, r8
    cmp rcx, r9
    cmova rcx, r9
compare_loop:
    test rcx, rcx
    jz compare_lengths
    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rdi]
    cmp eax, edx
    jb compare_less
    ja compare_greater
    inc rsi
    inc rdi
    dec rcx
    jmp compare_loop
compare_lengths:
    cmp r8, r9
    jb compare_less
    ja compare_greater
    xor eax, eax
    jmp compare_return
compare_less:
    mov eax, -1
    jmp compare_return
compare_greater:
    mov eax, 1
compare_return:
    pop r9
    pop r8
    ret
