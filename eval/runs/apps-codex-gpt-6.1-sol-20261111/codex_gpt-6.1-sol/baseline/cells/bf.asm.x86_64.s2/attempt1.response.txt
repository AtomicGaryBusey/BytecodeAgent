.globl _start
.bss
.balign 8
input:
.zero 65537
commands:
.zero 65536
jumps:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea rbx, [rip + input]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
    mov r13, r12

parse:
    cmp r8, r12
    jae parsed
    movzx eax, byte ptr [rbx + r8]
    inc r8
    cmp al, 33
    je delimiter
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne parse
save_command:
    mov byte ptr [commands + r9], al
    inc r9
    jmp parse

open_bracket:
    mov dword ptr [brackets + r10*4], r9d
    inc r10d
    jmp save_command

close_bracket:
    test r10d, r10d
    jz invalid
    dec r10d
    mov edx, dword ptr [brackets + r10*4]
    lea ecx, [r9 + 1]
    mov dword ptr [jumps + rdx*4], ecx
    inc edx
    mov dword ptr [jumps + r9*4], edx
    jmp save_command

delimiter:
    mov r13, r8
parsed:
    test r10d, r10d
    jnz invalid
    xor r8d, r8d
    xor r14d, r14d
    xor r15d, r15d
    lea rbx, [rip + tape]

execute:
    cmp r8, r9
    jae finished
    movzx eax, byte ptr [commands + r8]
    inc r8
    cmp al, 43
    je plus
    cmp al, 45
    je minus
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 46
    je emit
    cmp al, 44
    je consume
    cmp al, 91
    je loop_open
    cmp byte ptr [rbx + r14], 0
    je execute
    mov r8d, dword ptr [jumps + r8*4 - 4]
    jmp execute

loop_open:
    cmp byte ptr [rbx + r14], 0
    jne execute
    mov r8d, dword ptr [jumps + r8*4 - 4]
    jmp execute
plus:
    inc byte ptr [rbx + r14]
    jmp execute
minus:
    dec byte ptr [rbx + r14]
    jmp execute
right:
    inc r14d
    jmp execute
left:
    dec r14d
    jmp execute
emit:
    mov al, byte ptr [rbx + r14]
    mov byte ptr [output + r15], al
    inc r15d
    jmp execute
consume:
    xor eax, eax
    cmp r13, r12
    jae store_input
    mov al, byte ptr [input + r13]
    inc r13
store_input:
    mov byte ptr [rbx + r14], al
    jmp execute

finished:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov rdx, r15
    syscall
    xor edi, edi
    mov eax, 60
    syscall

invalid:
    mov edi, 1
    mov eax, 60
    syscall
