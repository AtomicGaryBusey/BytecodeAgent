.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x23, output
    add x23, x23, :lo12:output
    adrp x24, digits
    add x24, x24, :lo12:digits
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_offset
    sub x22, x20, x21
    mov x0, #16
    cmp x22, x0
    csel x22, x22, x0, lo

    mov x0, x21
    mov x1, x23
    bl hex8
    mov w5, #32
    strb w5, [x1], #1
    strb w5, [x1], #1
    add x25, x19, x21
    mov x6, #0

hex_slots:
    cmp x6, x22
    b.hs blank_slot
    ldrb w7, [x25, x6]
    lsr w9, w7, #4
    ldrb w9, [x24, x9]
    strb w9, [x1], #1
    and w9, w7, #15
    ldrb w9, [x24, x9]
    strb w9, [x1], #1
    strb w5, [x1], #1
    b slot_done
blank_slot:
    strb w5, [x1], #1
    strb w5, [x1], #1
    strb w5, [x1], #1
slot_done:
    cmp x6, #7
    b.ne no_gap
    strb w5, [x1], #1
no_gap:
    add x6, x6, #1
    cmp x6, #16
    b.lo hex_slots

    strb w5, [x1], #1
    mov w7, #124
    strb w7, [x1], #1
    mov x6, #0
ascii_loop:
    ldrb w7, [x25, x6]
    sub w9, w7, #32
    cmp w9, #94
    mov w10, #46
    csel w7, w7, w10, ls
    strb w7, [x1], #1
    add x6, x6, #1
    cmp x6, x22
    b.lo ascii_loop
    mov w7, #124
    strb w7, [x1], #1
    mov w7, #10
    strb w7, [x1], #1

    sub x2, x1, x23
    mov x1, x23
    mov x0, #1
    mov x8, #64
    svc #0
    add x21, x21, x22
    b row_loop

final_offset:
    mov x0, x20
    mov x1, x23
    bl hex8
    mov w3, #10
    strb w3, [x1]
    mov x0, #1
    mov x1, x23
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

hex8:
    mov x2, #28
hex8_loop:
    lsr x3, x0, x2
    and x3, x3, #15
    ldrb w4, [x24, x3]
    strb w4, [x1], #1
    subs x2, x2, #4
    b.ge hex8_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 69632
output:
    .zero 80
