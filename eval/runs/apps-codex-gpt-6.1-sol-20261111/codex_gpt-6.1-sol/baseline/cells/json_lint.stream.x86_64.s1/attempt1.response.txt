.globl _start
.bss
buffer:
.zero 65537
outbuf:
.zero 32

.section .rodata
okmsg:
.ascii "ok\n"
errmsg:
.ascii "error at "
trueword:
.asciz "true"
falseword:
.asciz "false"
nullword:
.asciz "null"

.text
_start:
    lea r14, [rip + buffer]
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    mov edx, 65537
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    add r15, rax
    jmp read_loop
read_done:
    mov r12, r14
    lea r13, [r14 + r15]
    call value
    call whitespace
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    xor edi, edi
    mov eax, 60
    syscall

peek:
    mov eax, 256
    cmp r12, r13
    je peek_done
    movzx eax, byte ptr [r12]
peek_done:
    ret

whitespace:
    call peek
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    jne ws_done
ws_next:
    inc r12
    jmp whitespace
ws_done:
    ret

value:
    call whitespace
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je literal_true
    cmp eax, 102
    je literal_false
    cmp eax, 110
    je literal_null
    cmp eax, 45
    je number
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    jmp number

literal_true:
    lea rsi, [rip + trueword]
    jmp literal
literal_false:
    lea rsi, [rip + falseword]
    jmp literal
literal_null:
    lea rsi, [rip + nullword]
literal:
    movzx edx, byte ptr [rsi]
    test edx, edx
    jz literal_done
    call peek
    cmp eax, edx
    jne error
    inc r12
    inc rsi
    jmp literal
literal_done:
    ret

string:
    inc r12
string_loop:
    call peek
    cmp eax, 256
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_done
    cmp eax, 92
    je escape
    inc r12
    jmp string_loop
string_done:
    inc r12
    ret
escape:
    inc r12
    call peek
    cmp eax, 117
    je unicode
    cmp eax, 34
    je escape_ok
    cmp eax, 92
    je escape_ok
    cmp eax, 47
    je escape_ok
    cmp eax, 98
    je escape_ok
    cmp eax, 102
    je escape_ok
    cmp eax, 110
    je escape_ok
    cmp eax, 114
    je escape_ok
    cmp eax, 116
    jne error
escape_ok:
    inc r12
    jmp string_loop
unicode:
    inc r12
    mov ecx, 4
unicode_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_ok
    or eax, 32
    cmp eax, 97
    jb error
    cmp eax, 102
    ja error
hex_ok:
    inc r12
    dec ecx
    jnz unicode_loop
    jmp string_loop

number:
    cmp eax, 45
    jne number_int
    inc r12
    call peek
number_int:
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    cmp eax, 48
    je zero_int
int_loop:
    inc r12
    call peek
    cmp eax, 48
    jb fraction
    cmp eax, 57
    jbe int_loop
    jmp fraction
zero_int:
    inc r12
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r12
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
frac_loop:
    inc r12
    call peek
    cmp eax, 48
    jb exponent
    cmp eax, 57
    jbe frac_loop
exponent:
    cmp eax, 101
    je exp_start
    cmp eax, 69
    jne number_done
exp_start:
    inc r12
    call peek
    cmp eax, 43
    je exp_sign
    cmp eax, 45
    jne exp_digit
exp_sign:
    inc r12
    call peek
exp_digit:
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
exp_loop:
    inc r12
    call peek
    cmp eax, 48
    jb number_done
    cmp eax, 57
    jbe exp_loop
number_done:
    ret

array:
    inc r12
    call whitespace
    cmp eax, 93
    je container_done
array_value:
    call value
    call whitespace
    cmp eax, 93
    je container_done
    cmp eax, 44
    jne error
    inc r12
    jmp array_value

object:
    inc r12
    call whitespace
    cmp eax, 125
    je container_done
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r12
    call value
    call whitespace
    cmp eax, 125
    je container_done
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp object_member
container_done:
    inc r12
    ret

error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 9
    syscall
    mov rax, r12
    sub rax, r14
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ebx, 10
decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall
