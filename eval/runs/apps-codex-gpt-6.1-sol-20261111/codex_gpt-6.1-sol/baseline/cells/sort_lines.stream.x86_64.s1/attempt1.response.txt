.globl _start
.bss
.balign 16
buffer:
.zero 32769
array_a:
.zero 16000
array_b:
.zero 16000
output:
.zero 33768

.text
_start:
    lea rsi, [rip + buffer]
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    add rsi, r12
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz parse
    add r12, rax
    jmp read_more

parse:
    lea r13, [rip + array_a]
    lea r14, [rip + array_b]
    lea rsi, [rip + buffer]
    xor ecx, ecx
    xor edx, edx
    xor r8d, r8d
parse_loop:
    cmp rcx, r12
    jae parse_tail
    cmp byte ptr [rsi + rcx], 10
    jne parse_next
    mov rax, r8
    shl rax, 4
    lea rdi, [rsi + rdx]
    mov qword ptr [r13 + rax], rdi
    mov rdi, rcx
    sub rdi, rdx
    mov qword ptr [r13 + rax + 8], rdi
    inc r8
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop
parse_tail:
    cmp rdx, r12
    je parsed
    mov rax, r8
    shl rax, 4
    lea rdi, [rsi + rdx]
    mov qword ptr [r13 + rax], rdi
    sub r12, rdx
    mov qword ptr [r13 + rax + 8], r12
    inc r8
parsed:
    mov r12, r8
    mov r15d, 1
sort_pass:
    cmp r15, r12
    jae emit
    xor r8d, r8d
merge_group:
    cmp r8, r12
    jae pass_done
    lea r9, [r8 + r15]
    cmp r9, r12
    cmova r9, r12
    lea r10, [r9 + r15]
    cmp r10, r12
    cmova r10, r12
    mov rbx, r8
    mov rbp, r9
    mov rdx, r8
merge_loop:
    cmp rdx, r10
    jae group_done
    cmp rbx, r9
    jae take_right
    cmp rbp, r10
    jae take_left
    call compare
    test eax, eax
    jg take_right
take_left:
    mov rax, rbx
    inc rbx
    jmp copy_item
take_right:
    mov rax, rbp
    inc rbp
copy_item:
    shl rax, 4
    movdqu xmm0, xmmword ptr [r13 + rax]
    mov rax, rdx
    shl rax, 4
    movdqu xmmword ptr [r14 + rax], xmm0
    inc rdx
    jmp merge_loop
group_done:
    mov r8, r10
    jmp merge_group
pass_done:
    xchg r13, r14
    shl r15, 1
    jmp sort_pass

compare:
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    mov rax, rbp
    shl rax, 4
    mov rdi, qword ptr [r13 + rax]
    mov rax, qword ptr [r13 + rax + 8]
    mov r11, rcx
    cmp rcx, rax
    cmova rcx, rax
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    jb compare_less
    ja compare_greater
compare_lengths:
    cmp r11, rax
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret
compare_less:
    mov eax, -1
    ret
compare_greater:
    mov eax, 1
    ret

emit:
    lea rdi, [rip + output]
    xor ebx, ebx
emit_loop:
    cmp rbx, r12
    jae write_output
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp emit_loop
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
