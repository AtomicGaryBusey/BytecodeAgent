.bss
.balign 16
buffer:
.zero 32769
records:
.zero 16000
output:
.zero 33768

.text
.globl _start
_start:
    lea r14, [rip + buffer]
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r12]
    mov edx, 32769
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js read_loop
    add r12, rax
    jmp read_loop

read_done:
    lea r13, [rip + records]
    xor ebx, ebx
    xor ecx, ecx
    xor edx, edx
parse_loop:
    cmp rcx, r12
    jae parse_end
    cmp byte ptr [r14 + rcx], 10
    jne parse_next
    mov rax, rbx
    shl rax, 4
    lea rsi, [r14 + rdx]
    mov qword ptr [r13 + rax], rsi
    mov rsi, rcx
    sub rsi, rdx
    mov qword ptr [r13 + rax + 8], rsi
    inc rbx
    lea rdx, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_end:
    cmp rdx, r12
    je sort_begin
    mov rax, rbx
    shl rax, 4
    lea rsi, [r14 + rdx]
    mov qword ptr [r13 + rax], rsi
    sub r12, rdx
    mov qword ptr [r13 + rax + 8], r12
    inc rbx

sort_begin:
    mov r12, rbx
    mov ebx, 1
sort_outer:
    cmp rbx, r12
    jae build_output
    mov r10, rbx
    shl r10, 4
    mov r8, qword ptr [r13 + r10]
    mov r9, qword ptr [r13 + r10 + 8]
sort_inner:
    test r10, r10
    jz place_key
    mov rsi, qword ptr [r13 + r10 - 16]
    mov rdx, qword ptr [r13 + r10 - 8]
    mov rdi, r8
    mov rcx, rdx
    cmp rcx, r9
    cmova rcx, r9
    test rcx, rcx
    jz compare_lengths
    repe cmpsb
    ja shift_record
    jb place_key
compare_lengths:
    cmp rdx, r9
    jbe place_key
shift_record:
    mov rax, qword ptr [r13 + r10 - 16]
    mov qword ptr [r13 + r10], rax
    mov qword ptr [r13 + r10 + 8], rdx
    sub r10, 16
    jmp sort_inner
place_key:
    mov qword ptr [r13 + r10], r8
    mov qword ptr [r13 + r10 + 8], r9
    inc rbx
    jmp sort_outer

build_output:
    lea rdi, [rip + output]
    xor ebx, ebx
output_loop:
    cmp rbx, r12
    jae write_output
    mov rax, rbx
    shl rax, 4
    mov rsi, qword ptr [r13 + rax]
    mov rcx, qword ptr [r13 + rax + 8]
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp output_loop
write_output:
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    jz finish
    mov eax, 1
    mov edi, 1
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall
