.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more
read_done:
    add r13, r12
    lea r14, [rip + output]
next_line:
    cmp r12, r13
    jae finish
    mov r15, r12
find_end:
    cmp r15, r13
    jae line_ready
    cmp byte ptr [r15], 10
    je line_ready
    inc r15
    jmp find_end
line_ready:
    xor ebp, ebp
next_token:
    cmp r12, r15
    jae line_done
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_space
    cmp al, 9
    je skip_space
    mov r8, r12
scan_token:
    inc r12
    cmp r12, r15
    jae token_ready
    movzx eax, byte ptr [r12]
    cmp al, 32
    je token_ready
    cmp al, 9
    jne scan_token
token_ready:
    mov rcx, r12
    sub rcx, r8
    movzx ebx, byte ptr [r8]
    cmp rcx, 1
    jne literal
    cmp bl, 43
    je operator
    cmp bl, 45
    je operator
    cmp bl, 42
    je operator
    cmp bl, 47
    je operator
    cmp bl, 37
    je operator
literal:
    xor r10d, r10d
    cmp bl, 45
    jne literal_digits
    mov r10d, 1
    inc r8
    dec rcx
literal_digits:
    test rcx, rcx
    jz line_error
    cmp rcx, 18
    ja line_error
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [r8]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc r8
    dec rcx
    jnz digit_loop
    test r10d, r10d
    jz push_value
    neg rax
push_value:
    lea rdi, [rip + values]
    mov qword ptr [rdi + rbp*8], rax
    inc ebp
    jmp next_token
skip_space:
    inc r12
    jmp next_token
operator:
    cmp ebp, 2
    jb line_error
    lea rdi, [rip + values]
    dec ebp
    mov rsi, qword ptr [rdi + rbp*8]
    mov rax, qword ptr [rdi + rbp*8 - 8]
    cmp bl, 43
    je add_values
    cmp bl, 45
    je sub_values
    cmp bl, 42
    je mul_values
    test rsi, rsi
    jz line_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne divide_values
    cmp rsi, -1
    je line_error
divide_values:
    cqo
    idiv rsi
    cmp bl, 37
    jne store_value
    mov rax, rdx
    jmp store_value
add_values:
    add rax, rsi
    jmp store_value
sub_values:
    sub rax, rsi
    jmp store_value
mul_values:
    imul rax, rsi
store_value:
    mov qword ptr [rdi + rbp*8 - 8], rax
    jmp next_token
line_done:
    cmp ebp, 1
    jne line_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns print_magnitude
    mov byte ptr [r14], 45
    inc r14
    neg rax
print_magnitude:
    lea rsi, [rip + digits + 32]
    mov ebx, 10
    xor ecx, ecx
convert_digit:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz convert_digit
copy_digits:
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc rsi
    inc r14
    dec ecx
    jnz copy_digits
    mov byte ptr [r14], 10
    inc r14
    jmp advance_line
line_error:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
advance_line:
    mov r12, r15
    cmp r12, r13
    jae finish
    inc r12
    jmp next_line
finish:
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    jz exit_ok
    mov eax, 1
    mov edi, 1
    syscall
exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.balign 8
input:
    .zero 65537
values:
    .zero 8008
digits:
    .zero 32
output:
    .zero 393216
