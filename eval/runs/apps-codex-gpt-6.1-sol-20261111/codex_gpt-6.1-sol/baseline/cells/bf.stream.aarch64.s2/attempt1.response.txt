.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x22, jumps
    add x22, x22, :lo12:jumps
    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x24, output
    add x24, x24, :lo12:output
    adrp x15, brackets
    add x15, x15, :lo12:brackets
    mov x25, #0
    mov x26, x20
    mov x27, #0
    mov x9, #0

parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w10, [x19, x9]
    add x9, x9, #1
    cmp w10, #33
    b.eq delimiter
    cmp w10, #91
    b.eq open_bracket
    cmp w10, #93
    b.eq close_bracket
    cmp w10, #62
    b.eq save_command
    cmp w10, #60
    b.eq save_command
    cmp w10, #43
    b.eq save_command
    cmp w10, #45
    b.eq save_command
    cmp w10, #46
    b.eq save_command
    cmp w10, #44
    b.ne parse_loop
save_command:
    strb w10, [x21, x25]
    add x25, x25, #1
    b parse_loop

open_bracket:
    str w25, [x15, x27, lsl #2]
    add x27, x27, #1
    b save_command

close_bracket:
    cbz x27, invalid
    sub x27, x27, #1
    ldr w11, [x15, x27, lsl #2]
    add w12, w25, #1
    str w12, [x22, x11, lsl #2]
    add w12, w11, #1
    str w12, [x22, x25, lsl #2]
    b save_command

delimiter:
    mov x26, x9
parse_done:
    cbnz x27, invalid
    mov x28, #0
    mov x13, #0
    mov x14, #0

execute:
    cmp x13, x25
    b.hs finish
    mov x9, x13
    ldrb w10, [x21, x13]
    add x13, x13, #1
    cmp w10, #62
    b.eq right
    cmp w10, #60
    b.eq left
    cmp w10, #43
    b.eq increment
    cmp w10, #45
    b.eq decrement
    cmp w10, #46
    b.eq emit
    cmp w10, #44
    b.eq consume
    ldrb w11, [x23, x14]
    cmp w10, #91
    b.eq test_open
    cbz w11, execute
    ldr w13, [x22, x9, lsl #2]
    b execute

test_open:
    cbnz w11, execute
    ldr w13, [x22, x9, lsl #2]
    b execute

right:
    add x14, x14, #1
    b execute
left:
    sub x14, x14, #1
    b execute
increment:
    ldrb w11, [x23, x14]
    add w11, w11, #1
    strb w11, [x23, x14]
    b execute
decrement:
    ldrb w11, [x23, x14]
    sub w11, w11, #1
    strb w11, [x23, x14]
    b execute
emit:
    ldrb w11, [x23, x14]
    strb w11, [x24, x28]
    add x28, x28, #1
    b execute
consume:
    mov w11, #0
    cmp x26, x20
    b.hs store_input
    ldrb w11, [x19, x26]
    add x26, x26, #1
store_input:
    strb w11, [x23, x14]
    b execute

finish:
    cbz x28, success
    mov x0, #1
    mov x1, x24
    mov x2, x28
    mov x8, #64
    svc #0
success:
    mov x0, #0
    b terminate
invalid:
    mov x0, #1
terminate:
    mov x8, #93
    svc #0
