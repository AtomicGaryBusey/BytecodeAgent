.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65537
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output
    add x22, x22, :lo12:output
    adrp x23, digits
    add x23, x23, :lo12:digits
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_offset
    sub x24, x20, x21
    cmp x24, #16
    mov x3, #16
    csel x24, x24, x3, lo

    mov x1, x22
    mov w0, w21
    bl format_offset

    mov w3, #32
    mov x4, #8
space_loop:
    strb w3, [x22, x4]
    add x4, x4, #1
    cmp x4, #60
    b.lo space_loop

    add x5, x22, #10
    add x6, x19, x21
    mov x7, #0
hex_loop:
    cmp x7, x24
    b.hs hex_slot_done
    ldrb w9, [x6, x7]
    lsr w10, w9, #4
    ldrb w10, [x23, w10, uxtw]
    strb w10, [x5]
    and w9, w9, #15
    ldrb w10, [x23, w9, uxtw]
    strb w10, [x5, #1]
hex_slot_done:
    add x5, x5, #3
    cmp x7, #7
    b.ne no_gap
    add x5, x5, #1
no_gap:
    add x7, x7, #1
    cmp x7, #16
    b.lo hex_loop

    mov w9, #124
    strb w9, [x22, #60]
    add x5, x22, #61
    mov x7, #0
ascii_loop:
    ldrb w9, [x6, x7]
    sub w10, w9, #32
    cmp w10, #94
    mov w11, #46
    csel w9, w9, w11, ls
    strb w9, [x5], #1
    add x7, x7, #1
    cmp x7, x24
    b.lo ascii_loop
    mov w9, #124
    strb w9, [x5], #1
    mov w9, #10
    strb w9, [x5], #1

    mov x0, #1
    mov x1, x22
    sub x2, x5, x22
    mov x8, #64
    svc #0
    add x21, x21, x24
    b row_loop

final_offset:
    mov w0, w20
    mov x1, x22
    bl format_offset
    mov w3, #10
    strb w3, [x22, #8]
    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

format_offset:
    mov w3, #28
offset_digit:
    lsr w4, w0, w3
    and w4, w4, #15
    ldrb w4, [x23, w4, uxtw]
    strb w4, [x1], #1
    subs w3, w3, #4
    b.ge offset_digit
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 80
