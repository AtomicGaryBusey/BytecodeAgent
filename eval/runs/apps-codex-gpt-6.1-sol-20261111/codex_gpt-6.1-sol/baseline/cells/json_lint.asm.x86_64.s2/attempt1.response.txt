.globl _start
.bss
buffer:
.zero 65537
outbuf:
.zero 32
.text
_start:
    lea r14, [rip + buffer]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more
read_done:
    add r13, r14
    mov r12, r14
    call value
    call whitespace
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

peek:
    mov eax, -1
    cmp r12, r13
    jae peek_done
    movzx eax, byte ptr [r12]
peek_done:
    ret

whitespace:
    call peek
    cmp eax, 32
    je ws_advance
    cmp eax, 9
    je ws_advance
    cmp eax, 10
    je ws_advance
    cmp eax, 13
    jne ws_done
ws_advance:
    inc r12
    jmp whitespace
ws_done:
    ret

value:
    call whitespace
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je true_value
    cmp eax, 102
    je false_value
    cmp eax, 110
    je null_value
    cmp eax, 45
    je number
    sub eax, 48
    cmp eax, 9
    jbe number
    jmp error

true_value:
    lea rsi, [rip + true_text]
    jmp literal
false_value:
    lea rsi, [rip + false_text]
    jmp literal
null_value:
    lea rsi, [rip + null_text]
literal:
    movzx edx, byte ptr [rsi]
    test edx, edx
    jz literal_done
    call peek
    cmp eax, edx
    jne error
    inc r12
    inc rsi
    jmp literal
literal_done:
    ret

string:
    inc r12
string_loop:
    call peek
    cmp eax, -1
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_done
    cmp eax, 92
    je escape
    inc r12
    jmp string_loop
string_done:
    inc r12
    ret
escape:
    inc r12
    call peek
    cmp eax, 117
    je unicode
    cmp eax, 34
    je escape_done
    cmp eax, 92
    je escape_done
    cmp eax, 47
    je escape_done
    cmp eax, 98
    je escape_done
    cmp eax, 102
    je escape_done
    cmp eax, 110
    je escape_done
    cmp eax, 114
    je escape_done
    cmp eax, 116
    jne error
escape_done:
    inc r12
    jmp string_loop
unicode:
    inc r12
    mov ecx, 4
hex_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_ok
    or eax, 32
    sub eax, 97
    cmp eax, 5
    ja error
hex_ok:
    inc r12
    dec ecx
    jnz hex_loop
    jmp string_loop

number:
    call peek
    cmp eax, 45
    jne integer
    inc r12
    call peek
integer:
    cmp eax, 48
    je zero_integer
    sub eax, 49
    cmp eax, 8
    ja error
integer_loop:
    inc r12
    call peek
    mov edx, eax
    sub edx, 48
    cmp edx, 9
    jbe integer_loop
    jmp fraction
zero_integer:
    inc r12
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r12
    call require_digit
fraction_loop:
    inc r12
    call peek
    mov edx, eax
    sub edx, 48
    cmp edx, 9
    jbe fraction_loop
exponent:
    cmp eax, 101
    je exponent_start
    cmp eax, 69
    jne number_done
exponent_start:
    inc r12
    call peek
    cmp eax, 43
    je exponent_sign
    cmp eax, 45
    jne exponent_digits
exponent_sign:
    inc r12
exponent_digits:
    call require_digit
exponent_loop:
    inc r12
    call peek
    sub eax, 48
    cmp eax, 9
    jbe exponent_loop
number_done:
    ret
require_digit:
    call peek
    sub eax, 48
    cmp eax, 9
    ja error
    ret

array:
    inc r12
    call whitespace
    cmp eax, 93
    je close_container
array_value:
    call value
    call whitespace
    cmp eax, 93
    je close_container
    cmp eax, 44
    jne error
    inc r12
    jmp array_value
close_container:
    inc r12
    ret

object:
    inc r12
    call whitespace
    cmp eax, 125
    je close_container
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r12
    call value
    call whitespace
    cmp eax, 125
    je close_container
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp object_member

error:
    mov rax, r12
    sub rax, r14
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ecx, 1
    mov r8d, 10
decimal:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz decimal
    mov r9, rsi
    mov r10d, ecx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov edx, r10d
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok_msg:
.ascii "ok\n"
error_msg:
.ascii "error at "
true_text:
.asciz "true"
false_text:
.asciz "false"
null_text:
.asciz "null"
