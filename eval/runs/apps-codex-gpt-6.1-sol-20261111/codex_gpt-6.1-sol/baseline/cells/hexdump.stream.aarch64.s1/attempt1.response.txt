.globl _start

.text
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    movz x2, #16
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x22, output
    add x22, x22, :lo12:output
    adrp x24, digits
    add x24, x24, :lo12:digits
    mov x21, #0

row_loop:
    cmp x21, x20
    b.hs final_offset
    sub x23, x20, x21
    cmp x23, #16
    mov x0, #16
    csel x23, x23, x0, lo
    mov x0, x21
    mov x1, x22
    bl hex8
    mov w5, #32
    strb w5, [x1], #1
    strb w5, [x1], #1
    add x25, x19, x21
    mov x6, #0

slot_loop:
    cmp x6, x23
    b.hs blank_slot
    ldrb w7, [x25, x6]
    lsr w9, w7, #4
    ldrb w9, [x24, x9]
    strb w9, [x1], #1
    and w7, w7, #15
    ldrb w9, [x24, x7]
    strb w9, [x1], #1
    strb w5, [x1], #1
    b slot_done
blank_slot:
    strb w5, [x1], #1
    strb w5, [x1], #1
    strb w5, [x1], #1
slot_done:
    cmp x6, #7
    b.ne no_extra
    strb w5, [x1], #1
no_extra:
    add x6, x6, #1
    cmp x6, #16
    b.lo slot_loop
    strb w5, [x1], #1
    mov w5, #124
    strb w5, [x1], #1
    mov x6, #0
ascii_loop:
    ldrb w7, [x25, x6]
    sub w9, w7, #32
    cmp w9, #94
    mov w10, #46
    csel w7, w7, w10, ls
    strb w7, [x1], #1
    add x6, x6, #1
    cmp x6, x23
    b.lo ascii_loop
    strb w5, [x1], #1
    mov w5, #10
    strb w5, [x1], #1
    sub x2, x1, x22
    mov x1, x22
    mov x0, #1
    mov x8, #64
    svc #0
    add x21, x21, x23
    b row_loop

final_offset:
    mov x0, x20
    mov x1, x22
    bl hex8
    mov w5, #10
    strb w5, [x1], #1
    mov x0, #1
    mov x1, x22
    mov x2, #9
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

hex8:
    mov x2, #28
hex8_loop:
    lsr x3, x0, x2
    and x3, x3, #15
    ldrb w4, [x24, x3]
    strb w4, [x1], #1
    subs x2, x2, #4
    b.ge hex8_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65552
output:
    .zero 80
