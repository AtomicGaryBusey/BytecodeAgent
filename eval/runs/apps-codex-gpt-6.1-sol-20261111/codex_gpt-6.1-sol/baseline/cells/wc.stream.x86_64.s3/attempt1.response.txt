.globl _start
.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz output
    js read_loop
    add r14, rax
    lea rdi, [rsi + rax]

scan_loop:
    movzx eax, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne classify
    inc r12
classify:
    cmp al, 32
    je whitespace
    sub eax, 9
    cmp eax, 4
    jbe whitespace
    test r15d, r15d
    jnz scan_next
    inc r13
    mov r15d, 1
    jmp scan_next
whitespace:
    xor r15d, r15d
scan_next:
    cmp rsi, rdi
    jb scan_loop
    jmp read_loop

output:
    lea rdi, [rip + outbuf]
    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi
    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

decimal:
    lea r8, [rip + digits + 32]
    mov r9, r8
    mov r10d, 10
decimal_divide:
    xor edx, edx
    div r10
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz decimal_divide
decimal_copy:
    mov al, byte ptr [r8]
    mov byte ptr [rdi], al
    inc r8
    inc rdi
    cmp r8, r9
    jb decimal_copy
    ret

.bss
.balign 16
buffer:
    .zero 4096
outbuf:
    .zero 64
digits:
    .zero 32
