.text
_start:
    adrp x19, raw
    add x19, x19, :lo12:raw
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    add x2, x2, #1
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0

    adrp x21, commands
    add x21, x21, :lo12:commands
    adrp x23, jumps
    add x23, x23, :lo12:jumps
    adrp x24, brackets
    add x24, x24, :lo12:brackets
    mov x22, #0
    mov x25, #0
    mov x26, #0

parse:
    cmp x26, x20
    b.hs parse_done
    ldrb w3, [x19, x26]
    add x26, x26, #1
    cmp w3, #33
    b.eq parse_done
    cmp w3, #91
    b.eq open_bracket
    cmp w3, #93
    b.eq close_bracket
    cmp w3, #43
    b.eq save_command
    cmp w3, #45
    b.eq save_command
    cmp w3, #62
    b.eq save_command
    cmp w3, #60
    b.eq save_command
    cmp w3, #46
    b.eq save_command
    cmp w3, #44
    b.ne parse
save_command:
    strb w3, [x21, x22]
    add x22, x22, #1
    b parse

open_bracket:
    str w22, [x24, x25, lsl #2]
    add x25, x25, #1
    b save_command

close_bracket:
    cbz x25, invalid
    sub x25, x25, #1
    ldr w4, [x24, x25, lsl #2]
    add w5, w22, #1
    str w5, [x23, x4, lsl #2]
    add w5, w4, #1
    str w5, [x23, x22, lsl #2]
    b save_command

parse_done:
    cbnz x25, invalid
    adrp x27, tape
    add x27, x27, :lo12:tape
    adrp x28, output
    add x28, x28, :lo12:output
    mov x29, #0
    mov x25, #0

execute:
    cmp x25, x22
    b.hs finished
    ldrb w3, [x21, x25]
    add x25, x25, #1
    cmp w3, #43
    b.eq increment
    cmp w3, #45
    b.eq decrement
    cmp w3, #62
    b.eq right
    cmp w3, #60
    b.eq left
    cmp w3, #46
    b.eq emit
    cmp w3, #44
    b.eq consume
    ldrb w4, [x27]
    cmp w3, #91
    b.eq loop_open
    cbz w4, execute
    b jump

loop_open:
    cbnz w4, execute
jump:

    ldr w25, [x23, x4, lsl #2]
    b execute

increment:
    ldrb w4, [x27]
    add w4, w4, #1
    strb w4, [x27]
    b execute

decrement:
    ldrb w4, [x27]
    sub w4, w4, #1
    strb w4, [x27]
    b execute

right:
    add x27, x27, #1
    b execute

left:

    b execute

emit:
    ldrb w4, [x27]
    strb w4, [x28, x29]
    add x29, x29, #1
    b execute

consume:

    cmp x26, x20
    b.hs consume_store
    ldrb w4, [x19, x26]
    add x26, x26, #1
consume_store:

    b execute

finished:
    cbz x29, success
    mov x0, #1
    mov x1, x28
    mov x2, x29
    mov x8, #64
    svc #0

    b terminate
