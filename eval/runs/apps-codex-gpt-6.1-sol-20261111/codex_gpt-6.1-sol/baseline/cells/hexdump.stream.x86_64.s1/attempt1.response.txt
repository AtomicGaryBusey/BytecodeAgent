.globl _start

.text
_start:
    xor r15d, r15d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r15
    mov edx, 65537
    sub rdx, r15
    syscall
    test rax, rax
    js read_loop
    jz read_done
    add r15, rax
    jmp read_loop

read_done:
    lea rdi, [rip + output]
    lea r14, [rip + input]
    xor r12d, r12d

row_loop:
    cmp r12, r15
    jae final_offset
    mov r13, r15
    sub r13, r12
    cmp r13, 16
    jbe row_length_ready
    mov r13d, 16

row_length_ready:
    mov eax, r12d
    call hex8
    mov word ptr [rdi], 0x2020
    add rdi, 2
    xor r8d, r8d

hex_slots:
    cmp r8, r13
    jae blank_slot
    movzx eax, byte ptr [r14 + r8]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [rdi], dl
    and eax, 15
    mov al, byte ptr [digits + rax]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 32
    jmp slot_done

blank_slot:
    mov word ptr [rdi], 0x2020
    mov byte ptr [rdi + 2], 32

slot_done:
    add rdi, 3
    cmp r8d, 7
    jne no_middle_space
    mov byte ptr [rdi], 32
    inc rdi

no_middle_space:
    inc r8d
    cmp r8d, 16
    jb hex_slots

    mov word ptr [rdi], 0x7c20
    add rdi, 2
    xor r8d, r8d

ascii_loop:
    movzx eax, byte ptr [r14 + r8]
    cmp eax, 32
    jb replace_char
    cmp eax, 126
    jbe store_char
replace_char:
    mov eax, 46
store_char:
    mov byte ptr [rdi], al
    inc rdi
    inc r8
    cmp r8, r13
    jb ascii_loop

    mov word ptr [rdi], 0x0a7c
    add rdi, 2
    add r14, r13
    add r12, r13
    jmp row_loop

final_offset:
    mov eax, r15d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

hex8:
    mov ecx, 8
hex8_loop:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [digits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jnz hex8_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
input:
    .zero 65537
output:
    .zero 324000
