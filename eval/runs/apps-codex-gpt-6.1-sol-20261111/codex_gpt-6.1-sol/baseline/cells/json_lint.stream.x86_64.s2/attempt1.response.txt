.text
.globl _start
_start:
    lea r14, [rip + buffer]
    xor r15d, r15d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r15]
    mov edx, 65537
    sub rdx, r15
    syscall
    test rax, rax
    jz read_done
    js read_more
    add r15, rax
    jmp read_more
read_done:
    mov r12, r14
    lea r13, [r14 + r15]
    call whitespace
    call value
    call whitespace
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

peek:
    mov eax, -1
    cmp r12, r13
    jae peek_done
    movzx eax, byte ptr [r12]
peek_done:
    ret

whitespace:
    call peek
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    jne ws_done
ws_next:
    inc r12
    jmp whitespace
ws_done:
    ret

value:
    call peek
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je literal_true
    cmp eax, 102
    je literal_false
    cmp eax, 110
    je literal_null
    cmp eax, 45
    je number
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    jmp number

literal_true:
    lea rbx, [rip + true_text]
    mov ebp, 4
    jmp literal
literal_false:
    lea rbx, [rip + false_text]
    mov ebp, 5
    jmp literal
literal_null:
    lea rbx, [rip + null_text]
    mov ebp, 4
literal:
    call peek
    movzx edx, byte ptr [rbx]
    cmp eax, edx
    jne error
    inc r12
    inc rbx
    dec ebp
    jnz literal
    ret

string:
    inc r12
string_loop:
    call peek
    cmp eax, -1
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_done
    cmp eax, 92
    je escape
    inc r12
    jmp string_loop
string_done:
    inc r12
    ret
escape:
    inc r12
    call peek
    cmp eax, 117
    je unicode
    cmp eax, 34
    je escape_done
    cmp eax, 92
    je escape_done
    cmp eax, 47
    je escape_done
    cmp eax, 98
    je escape_done
    cmp eax, 102
    je escape_done
    cmp eax, 110
    je escape_done
    cmp eax, 114
    je escape_done
    cmp eax, 116
    jne error
escape_done:
    inc r12
    jmp string_loop
unicode:
    inc r12
    mov r8d, 4
hex_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_ok
    cmp eax, 65
    jb error
    cmp eax, 70
    jbe hex_ok
    cmp eax, 97
    jb error
    cmp eax, 102
    ja error
hex_ok:
    inc r12
    dec r8d
    jnz hex_loop
    jmp string_loop

number:
    cmp eax, 45
    jne integer
    inc r12
    call peek
integer:
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    cmp eax, 48
    je zero_integer
    call digits
    jmp fraction
zero_integer:
    inc r12
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r12
    call require_digits
exponent:
    cmp eax, 101
    je exponent_start
    cmp eax, 69
    jne number_done
exponent_start:
    inc r12
    call peek
    cmp eax, 43
    je exponent_sign
    cmp eax, 45
    jne exponent_digits
exponent_sign:
    inc r12
exponent_digits:
    call require_digits
number_done:
    ret
require_digits:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
digits:
    inc r12
    call peek
    cmp eax, 48
    jb digits_done
    cmp eax, 57
    jbe digits
digits_done:
    ret

array:
    inc r12
    call whitespace
    cmp eax, 93
    je container_done
array_value:
    call value
    call whitespace
    cmp eax, 93
    je container_done
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp array_value

object:
    inc r12
    call whitespace
    cmp eax, 125
    je container_done
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r12
    call whitespace
    call value
    call whitespace
    cmp eax, 125
    je container_done
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp object_member
container_done:
    inc r12
    ret

error:
    mov rax, r12
    sub rax, r14
    lea rsi, [rip + output_end]
    dec rsi
    mov byte ptr [rsi], 10
    mov ebx, 10
decimal_loop:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    sub rsi, 9
    mov byte ptr [rsi], 101
    mov byte ptr [rsi + 1], 114
    mov byte ptr [rsi + 2], 114
    mov byte ptr [rsi + 3], 111
    mov byte ptr [rsi + 4], 114
    mov byte ptr [rsi + 5], 32
    mov byte ptr [rsi + 6], 97
    mov byte ptr [rsi + 7], 116
    mov byte ptr [rsi + 8], 32
    lea rdx, [rip + output_end]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    mov edi, 1
    syscall

.section .rodata
ok:
    .ascii "ok\n"
true_text:
    .ascii "true"
false_text:
    .ascii "false"
null_text:
    .ascii "null"

.bss
buffer:
    .zero 65537
output:
    .zero 32
output_end:
