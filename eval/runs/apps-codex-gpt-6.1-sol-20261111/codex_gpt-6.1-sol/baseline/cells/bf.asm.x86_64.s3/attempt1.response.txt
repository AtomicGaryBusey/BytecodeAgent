.globl _start
.bss
.balign 8
source:
.zero 65537
jumps:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js fail
    add r12, rax
    jmp read_more

read_done:
    lea rbx, [rip + source]
    mov r13, r12
    mov r14, r12
    xor ecx, ecx
    xor r8d, r8d
    lea r9, [rip + brackets]
    lea r10, [rip + jumps]
validate:
    cmp rcx, r12
    jae validated
    movzx eax, byte ptr [rbx + rcx]
    cmp al, 33
    je separator
    cmp al, 91
    je opening
    cmp al, 93
    je closing
next_validate:
    inc ecx
    jmp validate
opening:
    mov dword ptr [r9 + r8*4], ecx
    inc r8d
    jmp next_validate
closing:
    test r8d, r8d
    jz fail
    dec r8d
    mov edx, dword ptr [r9 + r8*4]
    mov dword ptr [r10 + rcx*4], edx
    mov dword ptr [r10 + rdx*4], ecx
    jmp next_validate
separator:
    mov r13, rcx
    lea r14, [rcx + 1]
validated:
    test r8d, r8d
    jnz fail
    xor r8d, r8d
    lea r9, [rip + tape]
    lea r15, [rip + output]
    xor ebp, ebp

execute:
    cmp r8, r13
    jae finish
    movzx eax, byte ptr [rbx + r8]
    inc r8
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 43
    je plus
    cmp al, 45
    je minus
    cmp al, 46
    je emit
    cmp al, 44
    je consume
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close
    jmp execute
right:
    inc r9
    jmp execute
left:
    dec r9
    jmp execute
plus:
    inc byte ptr [r9]
    jmp execute
minus:
    dec byte ptr [r9]
    jmp execute
emit:
    mov al, byte ptr [r9]
    mov byte ptr [r15 + rbp], al
    inc ebp
    jmp execute
consume:
    xor eax, eax
    cmp r14, r12
    jae store_input
    mov al, byte ptr [rbx + r14]
    inc r14
store_input:
    mov byte ptr [r9], al
    jmp execute
loop_open:
    cmp byte ptr [r9], 0
    jne execute
    mov r8d, dword ptr [r10 + r8*4 - 4]
    inc r8
    jmp execute
loop_close:
    cmp byte ptr [r9], 0
    je execute
    mov r8d, dword ptr [r10 + r8*4 - 4]
    inc r8
    jmp execute
finish:
    mov eax, 1
    mov edi, 1
    mov rsi, r15
    mov rdx, rbp
    syscall
    xor edi, edi
    jmp terminate
fail:
    mov edi, 1
terminate:
    mov eax, 60
    syscall
