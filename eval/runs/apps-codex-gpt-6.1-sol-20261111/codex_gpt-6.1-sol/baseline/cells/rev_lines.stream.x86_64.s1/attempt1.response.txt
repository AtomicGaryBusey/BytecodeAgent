.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz process
    add r13, rax
    jmp read_loop

process:
    lea r14, [rip + output]
    xor r8d, r8d
    xor r9d, r9d
    xor r10d, r10d
scan:
    cmp r9, r13
    jae final_line
    cmp byte ptr [r12 + r9], 10
    je line_found
    inc r9
    jmp scan

line_found:
    mov rbx, r9
reverse_line:
    cmp rbx, r8
    jbe append_newline
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_line
append_newline:
    mov byte ptr [r14 + r10], 10
    inc r10
    inc r9
    mov r8, r9
    jmp scan

final_line:
    cmp r8, r13
    jae write_output
    mov rbx, r13
reverse_final:
    cmp rbx, r8
    jbe final_newline
    dec rbx
    mov al, byte ptr [r12 + rbx]
    mov byte ptr [r14 + r10], al
    inc r10
    jmp reverse_final
final_newline:
    mov byte ptr [r14 + r10], 10
    inc r10

write_output:
    test r10, r10
    jz finish
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov rdx, r10
    syscall
finish:
    mov eax, 60
    xor edi, edi
    syscall

.bss
input:
    .zero 65537
output:
    .zero 65537
