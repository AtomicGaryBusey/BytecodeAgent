.globl _start
.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop
read_done:
    xor r13d, r13d
    lea rdi, [rip + output]
row_loop:
    cmp r13, r12
    jae final_offset
    mov eax, r13d
    call hex_offset
    mov ax, 0x2020
    stosw
    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14d, 16
row_size_ready:
    xor r8d, r8d
hex_slots:
    cmp r8, r14
    jae blank_slot
    lea rsi, [rip + input]
    add rsi, r13
    movzx edx, byte ptr [rsi + r8]
    mov eax, edx
    shr eax, 4
    lea rbx, [rip + digits]
    mov al, byte ptr [rbx + rax]
    stosb
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done
blank_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb
slot_done:
    cmp r8d, 7
    jne no_gap
    mov al, 32
    stosb
no_gap:
    inc r8d
    cmp r8d, 16
    jb hex_slots
    mov al, 32
    stosb
    mov al, 124
    stosb
    xor r8d, r8d
ascii_loop:
    lea rsi, [rip + input]
    add rsi, r13
    mov al, byte ptr [rsi + r8]
    cmp al, 32
    jb dot
    cmp al, 126
    jbe ascii_store
dot:
    mov al, 46
ascii_store:
    stosb
    inc r8
    cmp r8, r14
    jb ascii_loop
    mov al, 124
    stosb
    mov al, 10
    stosb
    add r13, r14
    jmp row_loop
final_offset:
    mov eax, r12d
    call hex_offset
    mov al, 10
    stosb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

hex_offset:
    mov edx, eax
    mov ecx, 8
    lea rbx, [rip + digits]
offset_digit:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [rbx + rax]
    stosb
    dec ecx
    jnz offset_digit
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
input:
    .zero 65537
output:
    .zero 323593
