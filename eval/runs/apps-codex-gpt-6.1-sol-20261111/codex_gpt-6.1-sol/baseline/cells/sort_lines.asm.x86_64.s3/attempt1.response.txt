.globl _start
.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 32769
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    lea r12, [rip + records_a]
    xor r15d, r15d
    xor ecx, ecx
    xor edx, edx
scan_loop:
    cmp rcx, r13
    jae scan_done
    cmp byte ptr [input + rcx], 10
    jne scan_next
    mov rax, rcx
    sub rax, rdx
    shl rax, 32
    or rax, rdx
    mov qword ptr [r12 + r15*8], rax
    inc r15
    lea rdx, [rcx + 1]
scan_next:
    inc rcx
    jmp scan_loop
scan_done:
    cmp rdx, r13
    je parsed
    mov rax, r13
    sub rax, rdx
    shl rax, 32
    or rax, rdx
    mov qword ptr [r12 + r15*8], rax
    inc r15
parsed:
    lea r13, [rip + records_b]
    mov r14d, 1
    sub rsp, 32
sort_pass:
    cmp r14, r15
    jae sorted
    mov qword ptr [rsp + 24], 0
merge_run:
    mov rbx, qword ptr [rsp + 24]
    cmp rbx, r15
    jae pass_done
    mov qword ptr [rsp + 16], rbx
    lea rbp, [rbx + r14]
    cmp rbp, r15
    cmova rbp, r15
    mov qword ptr [rsp], rbp
    lea rax, [rbp + r14]
    cmp rax, r15
    cmova rax, r15
    mov qword ptr [rsp + 8], rax
    mov qword ptr [rsp + 24], rax
merge_loop:
    cmp rbx, qword ptr [rsp]
    jae take_right_tail
    cmp rbp, qword ptr [rsp + 8]
    jae take_left
    mov rdi, qword ptr [r12 + rbx*8]
    mov rsi, qword ptr [r12 + rbp*8]
    call compare
    test eax, eax
    jle take_left
take_right:
    mov rax, qword ptr [r12 + rbp*8]
    inc rbp
    jmp store_record
take_right_tail:
    cmp rbp, qword ptr [rsp + 8]
    jae merge_run
    jmp take_right
take_left:
    mov rax, qword ptr [r12 + rbx*8]
    inc rbx
store_record:
    mov rcx, qword ptr [rsp + 16]
    mov qword ptr [r13 + rcx*8], rax
    inc qword ptr [rsp + 16]
    jmp merge_loop
pass_done:
    xchg r12, r13
    add r14, r14
    jmp sort_pass
sorted:
    lea rdi, [rip + output]
    xor ebx, ebx
output_loop:
    cmp rbx, r15
    jae output_done
    mov rax, qword ptr [r12 + rbx*8]
    mov esi, eax
    lea rsi, [input + rsi]
    shr rax, 32
    mov rcx, rax
    rep movsb
    mov byte ptr [rdi], 10
    inc rdi
    inc rbx
    jmp output_loop
output_done:
    mov rdx, rdi
    lea rsi, [rip + output]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

compare:
    mov r8, rdi
    shr r8, 32
    mov r9, rsi
    shr r9, 32
    mov edi, edi
    mov esi, esi
    lea rdi, [input + rdi]
    lea rsi, [input + rsi]
    mov r10, r8
    cmp r10, r9
    cmova r10, r9
    xor ecx, ecx
compare_loop:
    cmp rcx, r10
    jae compare_lengths
    movzx eax, byte ptr [rdi + rcx]
    movzx edx, byte ptr [rsi + rcx]
    sub eax, edx
    jne compare_return
    inc rcx
    jmp compare_loop
compare_lengths:
    mov rax, r8
    sub rax, r9
compare_return:
    ret

.balign 16

records_a:

records_b:

output:
