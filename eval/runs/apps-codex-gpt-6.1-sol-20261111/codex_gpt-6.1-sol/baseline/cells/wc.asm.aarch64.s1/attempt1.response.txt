.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, input_buffer
    add x23, x23, :lo12:input_buffer

read_next:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, print_counts
    add x21, x21, x0
    mov x1, x23
    add x2, x23, x0

count_next:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace
    cbnz w22, counted
    add x20, x20, #1
    mov w22, #1
    b counted

whitespace:
    mov w22, #0
counted:
    cmp x1, x2
    b.ne count_next
    b read_next

print_counts:
    adrp x24, output_buffer
    add x24, x24, :lo12:output_buffer
    mov x25, x24
    mov x0, x19
    bl append_decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x20
    bl append_decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x21
    bl append_decimal
    mov w3, #10
    strb w3, [x25], #1

    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

append_decimal:
    sub sp, sp, #32
    mov x4, sp
    mov x5, #10
1:
    udiv x6, x0, x5
    msub x7, x6, x5, x0
    add w7, w7, #48
    strb w7, [x4], #1
    mov x0, x6
    cbnz x0, 1b
2:
    ldrb w7, [x4, #-1]!
    strb w7, [x25], #1
    cmp x4, sp
    b.ne 2b
    add sp, sp, #32
    ret

.bss
.balign 16
input_buffer:
    .zero 4096
output_buffer:
    .zero 64
