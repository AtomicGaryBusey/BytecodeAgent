.globl _start

.bss
.balign 16
source:
.zero 65537
commands:
.zero 65536
jumps:
.zero 262144
brackets:
.zero 4000
tape:
.zero 30000
output:
.zero 65536

.text
_start:
    xor r14d, r14d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + source]
    add rsi, r14
    mov edx, 65537
    sub rdx, r14
    syscall
    test rax, rax
    jz parse_start
    add r14, rax
    jmp read_more

parse_start:
    mov r13, r14
    xor ecx, ecx
    xor edx, edx
    xor edi, edi
parse_loop:
    cmp rcx, r14
    jae parse_done
    movzx eax, byte ptr [source + rcx]
    inc ecx
    cmp al, 33
    je delimiter
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne parse_loop
save_command:
    mov byte ptr [commands + rdx], al
    inc edx
    jmp parse_loop

open_bracket:
    mov dword ptr [brackets + rdi*4], edx
    inc edi
    jmp save_command

close_bracket:
    test edi, edi
    jz invalid
    dec edi
    mov r8d, dword ptr [brackets + rdi*4]
    lea r9d, [rdx + 1]
    mov dword ptr [jumps + r8*4], r9d
    inc r8d
    mov dword ptr [jumps + rdx*4], r8d
    jmp save_command

delimiter:
    mov r13, rcx
parse_done:
    test edi, edi
    jnz invalid
    mov r12d, edx
    xor ebx, ebx
    xor r8d, r8d
    lea r15, [rip + tape]

execute:
    cmp ebx, r12d
    jae finish
    movzx eax, byte ptr [commands + rbx]
    inc ebx
    cmp al, 43
    je increment
    cmp al, 45
    je decrement
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 46
    je emit
    cmp al, 44
    je consume
    cmp al, 91
    je loop_open
    cmp byte ptr [r15], 0
    je execute
    mov ebx, dword ptr [jumps + rbx*4 - 4]
    jmp execute

loop_open:
    cmp byte ptr [r15], 0
    jne execute
    mov ebx, dword ptr [jumps + rbx*4 - 4]
    jmp execute

increment:
    inc byte ptr [r15]
    jmp execute
decrement:
    dec byte ptr [r15]
    jmp execute
right:
    inc r15
    jmp execute
left:
    dec r15
    jmp execute
emit:
    mov al, byte ptr [r15]
    mov byte ptr [output + r8], al
    inc r8d
    jmp execute
consume:
    xor eax, eax
    cmp r13, r14
    jae store_input
    mov al, byte ptr [source + r13]
    inc r13
store_input:
    mov byte ptr [r15], al
    jmp execute

finish:
    test r8d, r8d
    jz success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + output]
    mov edx, r8d
    syscall
success:
    xor edi, edi
    mov eax, 60
    syscall

invalid:
    mov edi, 1
    mov eax, 60
    syscall
