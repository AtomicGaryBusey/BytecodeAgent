.globl _start
.bss
.align 8
input:
.zero 65537
values:
.zero 8008
output:
.zero 400000
.text
_start:
    xor r12d, r12d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more
read_done:
    lea r13, [rip + input]
    add r13, r12
    lea r12, [rip + input]
    lea rbx, [rip + output]
    xor r14d, r14d
    xor r15d, r15d
    xor ebp, ebp
next_byte:
    cmp r12, r13
    je input_done
    movzx eax, byte ptr [r12]
    inc r12
    cmp al, 10
    je newline
    mov ebp, 1
    test r15d, r15d
    jnz next_byte
    cmp al, 32
    je next_byte
    cmp al, 9
    je next_byte
    lea rsi, [r12 - 1]
scan_token:
    cmp r12, r13
    je token_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_end
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    inc r12
    jmp scan_token
token_end:
    mov rcx, r12
    sub rcx, rsi
    movzx edi, byte ptr [rsi]
    cmp rcx, 1
    jne literal
    cmp dil, 43
    je operator
    cmp dil, 45
    je operator
    cmp dil, 42
    je operator
    cmp dil, 47
    je operator
    cmp dil, 37
    je operator
literal:
    xor r8d, r8d
    cmp dil, 45
    jne digits
    mov r8d, 1
    inc rsi
    dec rcx
digits:
    test rcx, rcx
    jz bad
    cmp rcx, 18
    ja bad
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja bad
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digit_loop
    test r8d, r8d
    jz push_value
    neg rax
push_value:
    lea rdx, [rip + values]
    mov qword ptr [rdx + r14*8], rax
    inc r14
    jmp next_byte
operator:
    cmp r14, 2
    jb bad
    lea rsi, [rip + values]
    dec r14
    mov r8, qword ptr [rsi + r14*8]
    mov rax, qword ptr [rsi + r14*8 - 8]
    cmp dil, 43
    je add_values
    cmp dil, 45
    je sub_values
    cmp dil, 42
    je mul_values
    test r8, r8
    jz bad
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne divide_values
    cmp r8, -1
    je bad
divide_values:
    cqo
    idiv r8
    cmp dil, 37
    jne store_value
    mov rax, rdx
    jmp store_value
add_values:
    add rax, r8
    jmp store_value
sub_values:
    sub rax, r8
    jmp store_value
mul_values:
    imul rax, r8
store_value:
    mov qword ptr [rsi + r14*8 - 8], rax
    jmp next_byte
bad:
    mov r15d, 1
    jmp next_byte
newline:
    call finish_line
    jmp next_byte
input_done:
    test ebp, ebp
    jz write_output
    call finish_line
write_output:
    lea rsi, [rip + output]
    mov rdx, rbx
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
finish_line:
    test r15d, r15d
    jnz print_error
    cmp r14, 1
    jne print_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns magnitude
    mov byte ptr [rbx], 45
    inc rbx
    neg rax
magnitude:
    sub rsp, 32
    lea rsi, [rsp + 32]
    xor ecx, ecx
    mov r8d, 10
convert:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz convert
copy_digits:
    mov al, byte ptr [rsi]
    mov byte ptr [rbx], al
    inc rsi
    inc rbx
    dec ecx
    jnz copy_digits
    add rsp, 32
    mov byte ptr [rbx], 10
    inc rbx
    jmp reset_line
print_error:
    mov dword ptr [rbx], 0x6f727265
    mov word ptr [rbx + 4], 0x0a72
    add rbx, 6
reset_line:
    xor r14d, r14d
    xor r15d, r15d
    xor ebp, ebp
    ret
