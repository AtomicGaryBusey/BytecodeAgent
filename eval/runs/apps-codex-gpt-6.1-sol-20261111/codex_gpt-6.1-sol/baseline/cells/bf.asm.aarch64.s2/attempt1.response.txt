.text
.globl _start
_start:
    adrp x19, input_buffer
    add x19, x19, :lo12:input_buffer
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, program
    add x21, x21, :lo12:program
    adrp x23, jumps
    add x23, x23, :lo12:jumps
    adrp x24, bracket_stack
    add x24, x24, :lo12:bracket_stack
    mov x22, #0
    mov x25, #0
    mov x26, x20
    mov x9, #0

parse_loop:
    cmp x9, x20
    b.hs parse_done
    ldrb w10, [x19, x9]
    add x9, x9, #1
    cmp w10, #33
    b.eq separator
    cmp w10, #91
    b.eq open_bracket
    cmp w10, #93
    b.eq close_bracket
    cmp w10, #62
    b.eq save_command
    cmp w10, #60
    b.eq save_command
    cmp w10, #43
    b.eq save_command
    cmp w10, #45
    b.eq save_command
    cmp w10, #46
    b.eq save_command
    cmp w10, #44
    b.ne parse_loop

save_command:
    strb w10, [x21, x22]
    add x22, x22, #1
    b parse_loop

open_bracket:
    str w22, [x24, x25, lsl #2]
    add x25, x25, #1
    b save_command

close_bracket:
    cbz x25, invalid
    sub x25, x25, #1
    ldr w11, [x24, x25, lsl #2]
    add w12, w22, #1
    str w12, [x23, x11, lsl #2]
    add w12, w11, #1
    str w12, [x23, x22, lsl #2]
    b save_command

separator:
    mov x26, x9

parse_done:
    cbnz x25, invalid
    adrp x27, output_buffer
    add x27, x27, :lo12:output_buffer
    adrp x18, tape
    add x18, x18, :lo12:tape
    mov x28, #0
    mov x17, #0
    mov x16, #0

execute:
    cmp x16, x22
    b.hs finished
    ldrb w9, [x21, x16]
    add x16, x16, #1
    cmp w9, #43
    b.eq increment
    cmp w9, #45
    b.eq decrement
    cmp w9, #62
    b.eq right
    cmp w9, #60
    b.eq left
    cmp w9, #46
    b.eq output_byte
    cmp w9, #44
    b.eq input_byte
    ldrb w10, [x18, x17]
    cmp w9, #91
    b.eq loop_open
    cbz w10, execute
    b jump

loop_open:
    cbnz w10, execute
jump:
    sub x11, x16, #1
    ldr w16, [x23, x11, lsl #2]
    b execute

increment:
    ldrb w10, [x18, x17]
    add w10, w10, #1
    strb w10, [x18, x17]
    b execute

decrement:
    ldrb w10, [x18, x17]
    sub w10, w10, #1
    strb w10, [x18, x17]
    b execute

right:
    add x17, x17, #1
    b execute

left:
    sub x17, x17, #1
    b execute

output_byte:
    ldrb w10, [x18, x17]
    strb w10, [x27, x28]
    add x28, x28, #1
    b execute

input_byte:
    mov w10, #0
    cmp x26, x20
    b.hs store_input
    ldrb w10, [x19, x26]
    add x26, x26, #1
store_input:
    strb w10, [x18, x17]
    b execute

finished:
    cbz x28, success
    mov x0, #1
    mov x1, x27
    mov x2, x28
    mov x8, #64
    svc #0
success:
    mov x0, #0
    b exit_program
invalid:
    mov x0, #1
exit_program:
    mov x8, #93
    svc #0

.bss
.balign 16
input_buffer:
    .zero 65537
program:
    .zero 65536
.balign 4
jumps:
    .zero 262144
bracket_stack:
    .zero 4000
output_buffer:
    .zero 65536
tape:
    .zero 30000
