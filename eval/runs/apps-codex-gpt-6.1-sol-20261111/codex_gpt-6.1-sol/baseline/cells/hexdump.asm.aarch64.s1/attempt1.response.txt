.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #65536
    sub x2, x2, x20
    cbnz x2, read_call
    adrp x1, extra
    add x1, x1, :lo12:extra
    mov x2, #1
read_call:
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, output
    add x21, x21, :lo12:output
    adrp x22, digits
    add x22, x22, :lo12:digits
    mov x23, #0

row_loop:
    cmp x23, x20
    b.hs final_offset
    mov x0, x23
    bl emit_offset
    mov w3, #32
    strb w3, [x21], #1
    strb w3, [x21], #1
    sub x24, x20, x23
    cmp x24, #16
    mov x3, #16
    csel x24, x24, x3, lo
    mov x25, #0

hex_loop:
    cmp x25, x24
    b.hs empty_slot
    add x3, x19, x23
    ldrb w4, [x3, x25]
    lsr w5, w4, #4
    ldrb w5, [x22, x5]
    strb w5, [x21], #1
    and w4, w4, #15
    ldrb w5, [x22, x4]
    strb w5, [x21], #1
    mov w3, #32
    strb w3, [x21], #1
    b slot_done
empty_slot:
    mov w3, #32
    strb w3, [x21], #1
    strb w3, [x21], #1
    strb w3, [x21], #1
slot_done:
    cmp x25, #7
    b.ne no_gap
    mov w3, #32
    strb w3, [x21], #1
no_gap:
    add x25, x25, #1
    cmp x25, #16
    b.lo hex_loop
    mov w3, #32
    strb w3, [x21], #1
    mov w3, #124
    strb w3, [x21], #1
    mov x25, #0
ascii_loop:
    add x3, x19, x23
    ldrb w4, [x3, x25]
    sub w5, w4, #32
    cmp w5, #94
    mov w6, #46
    csel w4, w4, w6, ls
    strb w4, [x21], #1
    add x25, x25, #1
    cmp x25, x24
    b.lo ascii_loop
    mov w3, #124
    strb w3, [x21], #1
    mov w3, #10
    strb w3, [x21], #1
    add x23, x23, x24
    b row_loop

final_offset:
    mov x0, x20
    bl emit_offset
    mov w3, #10
    strb w3, [x21], #1
    adrp x1, output
    add x1, x1, :lo12:output
    sub x2, x21, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

emit_offset:
    mov x3, #28
offset_loop:
    lsr x4, x0, x3
    and x4, x4, #15
    ldrb w5, [x22, x4]
    strb w5, [x21], #1
    subs x3, x3, #4
    b.ge offset_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65536
extra:
    .zero 1
.balign 16
output:
    .zero 323593
