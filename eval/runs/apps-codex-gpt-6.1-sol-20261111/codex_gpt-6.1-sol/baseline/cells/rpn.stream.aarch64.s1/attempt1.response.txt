.globl _start

.bss
.balign 16
input:
.space 65538
values:
.space 8000
outbuf:
.space 32

.section .rodata
error_text:
.ascii "error\n"

.text
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
read_loop:
    mov x0, #0
    add x1, x19, x20
    mov x2, #1
    movk x2, #1, lsl #16
    sub x2, x2, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    cbz x20, finish
    add x20, x19, x20
    ldrb w0, [x20, #-1]
    cmp w0, #10
    b.eq input_ready
    mov w0, #10
    strb w0, [x20], #1
input_ready:
    adrp x21, values
    add x21, x21, :lo12:values
new_line:
    mov x22, #0
    mov x23, #0
next_token:
    cmp x19, x20
    b.eq finish
    ldrb w0, [x19], #1
    cmp w0, #10
    b.eq line_done
    cbnz x23, next_token
    cmp w0, #32
    b.eq next_token
    cmp w0, #9
    b.eq next_token
    sub x24, x19, #1
scan_token:
    ldrb w1, [x19]
    cmp w1, #32
    b.eq token_ready
    cmp w1, #9
    b.eq token_ready
    cmp w1, #10
    b.eq token_ready
    add x19, x19, #1
    b scan_token

token_ready:
    sub x2, x19, x24
    cmp x2, #1
    b.ne literal
    cmp w0, #43
    b.eq operator
    cmp w0, #45
    b.eq operator
    cmp w0, #42
    b.eq operator
    cmp w0, #47
    b.eq operator
    cmp w0, #37
    b.eq operator

literal:
    mov x3, x24
    mov x4, #0
    cmp w0, #45
    b.ne literal_digits
    mov x4, #1
    add x3, x3, #1
literal_digits:
    sub x2, x19, x3
    cbz x2, bad_line
    cmp x2, #18
    b.hi bad_line
    mov x5, #0
    mov x6, #10
digit_loop:
    ldrb w7, [x3], #1
    sub w7, w7, #48
    cmp w7, #9
    b.hi bad_line
    madd x5, x5, x6, x7
    cmp x3, x19
    b.ne digit_loop
    cmp x4, #0
    cneg x5, x5, ne
    str x5, [x21, x22, lsl #3]
    add x22, x22, #1
    b next_token

operator:
    cmp x22, #2
    b.lo bad_line
    sub x22, x22, #1
    ldr x3, [x21, x22, lsl #3]
    sub x22, x22, #1
    ldr x4, [x21, x22, lsl #3]
    cmp w0, #43
    b.eq op_add
    cmp w0, #45
    b.eq op_sub
    cmp w0, #42
    b.eq op_mul
    cbz x3, bad_line
    mov x5, #0x8000000000000000
    cmp x4, x5
    b.ne divide
    cmn x3, #1
    b.eq bad_line
divide:
    sdiv x5, x4, x3
    cmp w0, #47
    b.eq push_result
    msub x5, x5, x3, x4
    b push_result
op_add:
    add x5, x4, x3
    b push_result
op_sub:
    sub x5, x4, x3
    b push_result
op_mul:
    mul x5, x4, x3
push_result:
    str x5, [x21, x22, lsl #3]
    add x22, x22, #1
    b next_token

bad_line:
    mov x23, #1
    b next_token

line_done:
    cbnz x23, print_error
    cmp x22, #1
    b.ne print_error
    ldr x3, [x21]
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x1, x1, #32
    mov w0, #10
    strb w0, [x1, #-1]!
    mov x2, #1
    cmp x3, #0
    cset x7, lt
    cneg x3, x3, lt
    mov x6, #10
decimal_loop:
    udiv x4, x3, x6
    msub x5, x4, x6, x3
    add w5, w5, #48
    strb w5, [x1, #-1]!
    add x2, x2, #1
    mov x3, x4
    cbnz x3, decimal_loop
    cbz x7, write_line
    mov w0, #45
    strb w0, [x1, #-1]!
    add x2, x2, #1
    b write_line
print_error:
    adrp x1, error_text
    add x1, x1, :lo12:error_text
    mov x2, #6
write_line:
    mov x0, #1
    mov x8, #64
    svc #0
    b new_line

finish:
    mov x0, #0
    mov x8, #93
    svc #0
