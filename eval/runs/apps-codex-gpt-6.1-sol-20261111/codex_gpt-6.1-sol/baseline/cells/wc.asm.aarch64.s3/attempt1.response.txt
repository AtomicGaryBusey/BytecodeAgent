.text
.globl _start
_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov w22, #0
    adrp x23, input
    add x23, x23, :lo12:input
read_next:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, print_counts
    add x21, x21, x0
    mov x1, x23
    add x2, x1, x0
count_next:
    ldrb w3, [x1], #1
    cmp w3, #10
    cinc x19, x19, eq
    cmp w3, #32
    b.eq whitespace
    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace
    cbnz w22, advance
    add x20, x20, #1
    mov w22, #1
    b advance
whitespace:
    mov w22, #0
advance:
    cmp x1, x2
    b.ne count_next
    b read_next

print_counts:
    adrp x24, output
    add x24, x24, :lo12:output
    mov x25, x24
    mov x0, x19
    bl decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x20
    bl decimal
    mov w3, #32
    strb w3, [x25], #1
    mov x0, x21
    bl decimal
    mov w3, #10
    strb w3, [x25], #1
    mov x0, #1
    mov x1, x24
    sub x2, x25, x24
    mov x8, #64
    svc #0
    mov x0, #0
    mov x8, #93
    svc #0

decimal:
    sub sp, sp, #32
    add x5, sp, #32
    mov x6, x5
    mov x7, #10
decimal_digit:
    udiv x3, x0, x7
    msub x4, x3, x7, x0
    add w4, w4, #48
    strb w4, [x5, #-1]!
    mov x0, x3
    cbnz x0, decimal_digit
decimal_copy:
    ldrb w4, [x5], #1
    strb w4, [x25], #1
    cmp x5, x6
    b.ne decimal_copy
    add sp, sp, #32
    ret

.bss
.balign 16
input:
    .zero 4096
output:
    .zero 64
