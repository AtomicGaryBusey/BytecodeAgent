.globl _start
.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_loop

input_done:
    lea rdi, [rip + output]
    lea r13, [rip + input]
    lea r14, [rip + digits]
    xor ebx, ebx

row_loop:
    cmp rbx, r12
    jae final_offset
    mov eax, ebx
    call print_offset
    mov ax, 0x2020
    stosw
    mov r15, r12
    sub r15, rbx
    cmp r15, 16
    jbe row_size_ready
    mov r15d, 16
row_size_ready:
    xor r8d, r8d
hex_loop:
    cmp r8, r15
    jae empty_slot
    movzx edx, byte ptr [r13 + r8]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [r14 + rax]
    stosb
    and edx, 15
    mov al, byte ptr [r14 + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done
empty_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb
slot_done:
    cmp r8d, 7
    jne no_extra_space
    mov al, 32
    stosb
no_extra_space:
    inc r8d
    cmp r8d, 16
    jb hex_loop
    mov al, 32
    stosb
    mov al, '|'
    stosb
    xor r8d, r8d
ascii_loop:
    mov al, byte ptr [r13 + r8]
    cmp al, 32
    jb replace_char
    cmp al, 126
    jbe emit_char
replace_char:
    mov al, '.'
emit_char:
    stosb
    inc r8
    cmp r8, r15
    jb ascii_loop
    mov al, '|'
    stosb
    mov al, 10
    stosb
    add r13, r15
    add rbx, r15
    jmp row_loop

final_offset:
    mov eax, r12d
    call print_offset
    mov al, 10
    stosb
    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall

print_offset:
    mov edx, eax
    mov ecx, 8
offset_digit:
    rol edx, 4
    mov eax, edx
    and eax, 15
    mov al, byte ptr [r14 + rax]
    stosb
    dec ecx
    jnz offset_digit
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
output:
    .zero 323593
