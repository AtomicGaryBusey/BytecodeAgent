.globl _start

.bss
.balign 16
input:
.zero 65537
values:
.zero 8008
output:
.zero 393216
digits:
.zero 32

.text
_start:
    lea r12, [rip + input]
    xor r13d, r13d
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    add r13, r12
    lea r14, [rip + output]

line_start:
    cmp r12, r13
    jae finish
    xor r15d, r15d

next_token:
    cmp r12, r13
    jae line_end
    movzx eax, byte ptr [r12]
    cmp al, 10
    je newline_end
    cmp al, 32
    je skip_separator
    cmp al, 9
    je skip_separator

    mov rsi, r12
scan_token:
    inc r12
    cmp r12, r13
    jae token_ready
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_ready
    cmp al, 32
    je token_ready
    cmp al, 9
    jne scan_token

token_ready:
    mov rdi, r12
    sub rdi, rsi
    cmp rdi, 1
    jne literal
    movzx ebx, byte ptr [rsi]
    cmp bl, 43
    je operator
    cmp bl, 45
    je operator
    cmp bl, 42
    je operator
    cmp bl, 47
    je operator
    cmp bl, 37
    je operator

literal:
    xor r8d, r8d
    cmp byte ptr [rsi], 45
    jne literal_digits
    inc rsi
    dec rdi
    mov r8d, 1
literal_digits:
    test rdi, rdi
    jz line_error
    cmp rdi, 18
    ja line_error
    xor eax, eax
parse_digit:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja line_error
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rdi
    jnz parse_digit
    test r8d, r8d
    jz push_value
    neg rax
push_value:
    lea rdx, [rip + values]
    mov qword ptr [rdx + r15*8], rax
    inc r15
    jmp next_token

operator:
    cmp r15, 2
    jb line_error
    lea rsi, [rip + values]
    dec r15
    mov rdi, qword ptr [rsi + r15*8]
    mov rax, qword ptr [rsi + r15*8 - 8]
    cmp bl, 43
    je add_values
    cmp bl, 45
    je subtract_values
    cmp bl, 42
    je multiply_values
    test rdi, rdi
    jz line_error
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne divide_values
    cmp rdi, -1
    je line_error
divide_values:
    cqo
    idiv rdi
    cmp bl, 37
    jne store_result
    mov rax, rdx
    jmp store_result
add_values:
    add rax, rdi
    jmp store_result
subtract_values:
    sub rax, rdi
    jmp store_result
multiply_values:
    imul rax, rdi
store_result:
    mov qword ptr [rsi + r15*8 - 8], rax
    jmp next_token

skip_separator:
    inc r12
    jmp next_token

newline_end:
    inc r12
line_end:
    cmp r15, 1
    jne emit_error
    mov rax, qword ptr [rip + values]
    test rax, rax
    jns format_number
    mov byte ptr [r14], 45
    inc r14
    neg rax
format_number:
    lea rsi, [rip + digits + 32]
    mov ebx, 10
    xor ecx, ecx
decimal_digit:
    xor edx, edx
    div rbx
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    inc ecx
    test rax, rax
    jnz decimal_digit
copy_digits:
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc rsi
    inc r14
    dec ecx
    jnz copy_digits
    mov byte ptr [r14], 10
    inc r14
    jmp line_start

line_error:
    cmp r12, r13
    jae emit_error
    mov al, byte ptr [r12]
    inc r12
    cmp al, 10
    jne line_error
emit_error:
    mov dword ptr [r14], 0x6f727265
    mov word ptr [r14 + 4], 0x0a72
    add r14, 6
    jmp line_start

finish:
    lea rsi, [rip + output]
    mov rdx, r14
    sub rdx, rsi
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall
exit_program:
    mov eax, 60
    xor edi, edi
    syscall
