.text
.globl _start
_start:
    adrp x19, input
    add x19, x19, :lo12:input
    mov x20, #0
    movz x21, #1
    movk x21, #1, lsl #16

read_loop:
    mov x0, #0
    add x1, x19, x20
    sub x2, x21, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x28, output
    add x28, x28, :lo12:output
    mov x25, x28
    mov x22, x19
    mov x23, x19
    add x24, x19, x20

scan_loop:
    cmp x23, x24
    b.eq scan_done
    ldrb w9, [x23]
    cmp w9, #10
    b.ne scan_next
    bl emit_line
    add x22, x23, #1
scan_next:
    add x23, x23, #1
    b scan_loop

scan_done:
    cmp x22, x24
    b.eq write_output
    bl emit_line

write_output:
    sub x2, x25, x28
    cbz x2, finish
    mov x0, #1
    mov x1, x28
    mov x8, #64
    svc #0

finish:
    mov x0, #0
    mov x8, #93
    svc #0

emit_line:
    mov x26, x23
reverse_loop:
    cmp x26, x22
    b.eq append_newline
    sub x26, x26, #1
    ldrb w9, [x26]
    strb w9, [x25], #1
    b reverse_loop
append_newline:
    mov w9, #10
    strb w9, [x25], #1
    ret

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 65537
