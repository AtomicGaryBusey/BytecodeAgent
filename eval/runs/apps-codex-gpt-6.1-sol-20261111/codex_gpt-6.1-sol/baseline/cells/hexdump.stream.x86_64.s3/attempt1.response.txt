.globl _start

.text
_start:
    xor r12d, r12d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input]
    add rsi, r12
    mov edx, 65537
    sub rdx, r12
    syscall
    test rax, rax
    jz read_done
    js read_loop
    add r12, rax
    jmp read_loop

read_done:
    mov r13, r12
    xor r12d, r12d
    lea rdi, [rip + output]
    lea rbx, [rip + digits]
    lea r14, [rip + input]

row_loop:
    cmp r12, r13
    jae final_offset
    mov eax, r12d
    call hex_offset
    mov ax, 0x2020
    stosw

    mov r15, r13
    sub r15, r12
    cmp r15, 16
    jbe row_size_ready
    mov r15d, 16
row_size_ready:
    xor r8d, r8d

hex_slots:
    cmp r8, r15
    jae blank_slot
    lea r9, [r14 + r12]
    movzx edx, byte ptr [r9 + r8]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [rbx + rax]
    stosb
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    mov al, 32
    stosb
    jmp slot_done

blank_slot:
    mov ax, 0x2020
    stosw
    mov al, 32
    stosb

slot_done:
    cmp r8d, 7
    jne no_middle_space
    mov al, 32
    stosb
no_middle_space:
    inc r8d
    cmp r8d, 16
    jb hex_slots

    mov ax, 0x7c20
    stosw
    xor r8d, r8d
    lea r9, [r14 + r12]

ascii_loop:
    mov al, byte ptr [r9 + r8]
    cmp al, 32
    jb replacement
    cmp al, 126
    jbe ascii_store
replacement:
    mov al, 46
ascii_store:
    stosb
    inc r8
    cmp r8, r15
    jb ascii_loop

    mov ax, 0x0a7c
    stosw
    add r12, r15
    jmp row_loop

final_offset:
    mov eax, r13d
    call hex_offset
    mov al, 10
    stosb

    lea rsi, [rip + output]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

hex_offset:
    mov ecx, 8
    mov r10d, eax
hex_offset_loop:
    rol r10d, 4
    mov edx, r10d
    and edx, 15
    mov al, byte ptr [rbx + rdx]
    stosb
    dec ecx
    jnz hex_offset_loop
    ret

.section .rodata
digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input:
    .zero 65537
.balign 16
output:
    .zero 400000
