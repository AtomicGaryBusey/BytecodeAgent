.globl _start
.bss
.align 16
input:
.zero 65537
values:
.zero 8008
output:
.zero 393216
.text
_start:
    lea r12, [rip + input]
    xor ebx, ebx
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_more
read_done:
    lea r13, [r12 + rbx]
    lea r15, [rip + output]
    lea rbp, [rip + values]
new_line:
    cmp r12, r13
    jae finish
    xor r14d, r14d
next_token:
    cmp r12, r13
    jae line_done
    movzx eax, byte ptr [r12]
    cmp al, 10
    je newline_done
    cmp al, 32
    je separator
    cmp al, 9
    je separator
    mov rsi, r12
scan_token:
    inc r12
    cmp r12, r13
    jae token_ready
    movzx eax, byte ptr [r12]
    cmp al, 10
    je token_ready
    cmp al, 32
    je token_ready
    cmp al, 9
    jne scan_token
token_ready:
    mov rcx, r12
    sub rcx, rsi
    movzx edi, byte ptr [rsi]
    cmp rcx, 1
    jne literal
    cmp dil, 43
    je operator
    cmp dil, 45
    je operator
    cmp dil, 42
    je operator
    cmp dil, 47
    je operator
    cmp dil, 37
    je operator
literal:
    xor r8d, r8d
    cmp dil, 45
    jne literal_digits
    mov r8d, 1
    inc rsi
    dec rcx
literal_digits:
    test rcx, rcx
    jz error_line
    cmp rcx, 18
    ja error_line
    xor eax, eax
digit_loop:
    movzx edx, byte ptr [rsi]
    sub edx, 48
    cmp edx, 9
    ja error_line
    imul rax, rax, 10
    add rax, rdx
    inc rsi
    dec rcx
    jnz digit_loop
    test r8d, r8d
    jz push_value
    neg rax
push_value:
    mov qword ptr [rbp + r14*8], rax
    inc r14
    jmp next_token
operator:
    cmp r14, 2
    jb error_line
    dec r14
    mov rbx, qword ptr [rbp + r14*8]
    mov rax, qword ptr [rbp + r14*8 - 8]
    cmp dil, 43
    je op_add
    cmp dil, 45
    je op_sub
    cmp dil, 42
    je op_mul
    test rbx, rbx
    jz error_line
    cmp rbx, -1
    jne divide
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    je error_line
divide:
    cqo
    idiv rbx
    cmp dil, 37
    jne store_result
    mov rax, rdx
    jmp store_result
op_add:
    add rax, rbx
    jmp store_result
op_sub:
    sub rax, rbx
    jmp store_result
op_mul:
    imul rax, rbx
store_result:
    mov qword ptr [rbp + r14*8 - 8], rax
    jmp next_token
separator:
    inc r12
    jmp next_token
newline_done:
    inc r12
line_done:
    cmp r14, 1
    jne emit_error
    mov rax, qword ptr [rbp]
    test rax, rax
    jns format_number
    mov byte ptr [r15], 45
    inc r15
    neg rax
format_number:
    mov rsi, r15
    mov ebx, 10
format_digit:
    xor edx, edx
    div rbx
    add dl, 48
    mov byte ptr [r15], dl
    inc r15
    test rax, rax
    jnz format_digit
    lea rdi, [r15 - 1]
reverse_digits:
    cmp rsi, rdi
    jae number_done
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    mov byte ptr [rsi], dl
    mov byte ptr [rdi], al
    inc rsi
    dec rdi
    jmp reverse_digits
number_done:
    mov byte ptr [r15], 10
    inc r15
    jmp new_line
error_line:
    cmp r12, r13
    jae emit_error
    mov al, byte ptr [r12]
    inc r12
    cmp al, 10
    jne error_line
emit_error:
    mov dword ptr [r15], 0x6f727265
    mov word ptr [r15 + 4], 0x0a72
    add r15, 6
    jmp new_line
finish:
    lea rsi, [rip + output]
    mov rdx, r15
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov eax, 60
    xor edi, edi
    syscall
