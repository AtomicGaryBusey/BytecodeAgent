.globl _start
.bss
.balign 16
buffer:
.zero 65537
outbuf:
.zero 32
.section .rodata
okmsg:
.ascii "ok\n"
errmsg:
.ascii "error at "
lit_true:
.asciz "true"
lit_false:
.asciz "false"
lit_null:
.asciz "null"
.text
_start:
    lea r14, [rip + buffer]
    xor r13d, r13d
read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r14 + r13]
    mov edx, 65537
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_loop
read_done:
    xor r12d, r12d
    call value
    call whitespace
    cmp r12, r13
    jne error
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + okmsg]
    mov edx, 3
    syscall
    xor edi, edi
    jmp quit
peek:
    mov eax, 256
    cmp r12, r13
    jae peek_done
    movzx eax, byte ptr [r14 + r12]
peek_done:
    ret
whitespace:
    call peek
    cmp eax, 32
    je ws_next
    cmp eax, 9
    je ws_next
    cmp eax, 10
    je ws_next
    cmp eax, 13
    jne ws_done
ws_next:
    inc r12
    jmp whitespace
ws_done:
    ret
value:
    call whitespace
    cmp eax, 34
    je string
    cmp eax, 123
    je object
    cmp eax, 91
    je array
    cmp eax, 116
    je true_value
    cmp eax, 102
    je false_value
    cmp eax, 110
    je null_value
    cmp eax, 45
    je number
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
    jmp number
true_value:
    lea rsi, [rip + lit_true]
    jmp literal
false_value:
    lea rsi, [rip + lit_false]
    jmp literal
null_value:
    lea rsi, [rip + lit_null]
literal:
    movzx edx, byte ptr [rsi]
    test edx, edx
    jz literal_done
    call peek
    cmp eax, edx
    jne error
    inc rsi
    inc r12
    jmp literal
literal_done:
    ret
string:
    inc r12
string_loop:
    call peek
    cmp eax, 256
    je error
    cmp eax, 32
    jb error
    cmp eax, 34
    je string_end
    cmp eax, 92
    je escape
    inc r12
    jmp string_loop
string_end:
    inc r12
    ret
escape:
    inc r12
    call peek
    cmp eax, 117
    je unicode
    cmp eax, 34
    je escape_ok
    cmp eax, 92
    je escape_ok
    cmp eax, 47
    je escape_ok
    cmp eax, 98
    je escape_ok
    cmp eax, 102
    je escape_ok
    cmp eax, 110
    je escape_ok
    cmp eax, 114
    je escape_ok
    cmp eax, 116
    jne error
escape_ok:
    inc r12
    jmp string_loop
unicode:
    inc r12
    mov edx, 4
hex_loop:
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    jbe hex_ok
    or eax, 32
    cmp eax, 97
    jb error
    cmp eax, 102
    ja error
hex_ok:
    inc r12
    dec edx
    jnz hex_loop
    jmp string_loop
number:
    cmp eax, 45
    jne integer
    inc r12
    call peek
integer:
    cmp eax, 48
    je zero_integer
    cmp eax, 49
    jb error
    cmp eax, 57
    ja error
integer_loop:
    inc r12
    call peek
    cmp eax, 48
    jb fraction
    cmp eax, 57
    jbe integer_loop
    jmp fraction
zero_integer:
    inc r12
    call peek
fraction:
    cmp eax, 46
    jne exponent
    inc r12
    call peek
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
fraction_loop:
    inc r12
    call peek
    cmp eax, 48
    jb exponent
    cmp eax, 57
    jbe fraction_loop
exponent:
    cmp eax, 101
    je exponent_start
    cmp eax, 69
    jne number_done
exponent_start:
    inc r12
    call peek
    cmp eax, 43
    je exponent_sign
    cmp eax, 45
    jne exponent_digit
exponent_sign:
    inc r12
    call peek
exponent_digit:
    cmp eax, 48
    jb error
    cmp eax, 57
    ja error
exponent_loop:
    inc r12
    call peek
    cmp eax, 48
    jb number_done
    cmp eax, 57
    jbe exponent_loop
number_done:
    ret
array:
    inc r12
    call whitespace
    cmp eax, 93
    je container_end
array_value:
    call value
    call whitespace
    cmp eax, 93
    je container_end
    cmp eax, 44
    jne error
    inc r12
    jmp array_value
object:
    inc r12
    call whitespace
    cmp eax, 125
    je container_end
object_member:
    cmp eax, 34
    jne error
    call string
    call whitespace
    cmp eax, 58
    jne error
    inc r12
    call value
    call whitespace
    cmp eax, 125
    je container_end
    cmp eax, 44
    jne error
    inc r12
    call whitespace
    jmp object_member
container_end:
    inc r12
    ret
error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 9
    syscall
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov rax, r12
    mov r8d, 10
decimal_loop:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz decimal_loop
    lea rdx, [rip + outbuf + 32]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    mov edi, 1
quit:
    mov eax, 60
    syscall
