.globl _start
.text
_start:
    lea r12, [rip + source]
    xor ebx, ebx
read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + rbx]
    mov edx, 65537
    sub rdx, rbx
    syscall
    test rax, rax
    jz read_done
    js read_more
    add rbx, rax
    jmp read_more

read_done:
    lea r13, [rip + jumps]
    lea r14, [rip + commands]
    xor esi, esi
    xor ecx, ecx
    xor ebp, ebp
compile:
    cmp rsi, rbx
    jae program_end
    movzx eax, byte ptr [r12 + rsi]
    inc esi
    cmp al, 33
    je separator
    cmp al, 91
    je open_bracket
    cmp al, 93
    je close_bracket
    cmp al, 62
    je save_command
    cmp al, 60
    je save_command
    cmp al, 43
    je save_command
    cmp al, 45
    je save_command
    cmp al, 46
    je save_command
    cmp al, 44
    jne compile
save_command:
    mov byte ptr [r14 + rcx], al
    inc ecx
    jmp compile

open_bracket:
    lea rdi, [rip + brackets]
    mov dword ptr [rdi + rbp*4], ecx
    inc ebp
    jmp save_command

close_bracket:
    test ebp, ebp
    jz invalid
    dec ebp
    lea rdi, [rip + brackets]
    mov edi, dword ptr [rdi + rbp*4]
    lea edx, [rcx + 1]
    mov dword ptr [r13 + rdi*4], edx
    inc edi
    mov dword ptr [r13 + rcx*4], edi
    jmp save_command

program_end:
    mov rsi, rbx
separator:
    test ebp, ebp
    jnz invalid
    lea r8, [r12 + rsi]
    lea r9, [r12 + rbx]
    mov ebp, ecx
    mov r12, r14
    lea r14, [rip + tape]
    xor ebx, ebx
    xor r15d, r15d
    xor r10d, r10d

execute:
    cmp r15d, ebp
    jae finished
    movzx eax, byte ptr [r12 + r15]
    inc r15d
    cmp al, 43
    je plus
    cmp al, 45
    je minus
    cmp al, 62
    je right
    cmp al, 60
    je left
    cmp al, 91
    je loop_open
    cmp al, 93
    je loop_close
    cmp al, 46
    je output
    xor eax, eax
    cmp r8, r9
    jae input_store
    mov al, byte ptr [r8]
    inc r8
input_store:
    mov byte ptr [r14 + rbx], al
    jmp execute

plus:
    inc byte ptr [r14 + rbx]
    jmp execute
minus:
    dec byte ptr [r14 + rbx]
    jmp execute
right:
    inc ebx
    jmp execute
left:
    dec ebx
    jmp execute
loop_open:
    cmp byte ptr [r14 + rbx], 0
    jne execute
    mov r15d, dword ptr [r13 + r15*4 - 4]
    jmp execute
loop_close:
    cmp byte ptr [r14 + rbx], 0
    je execute
    mov r15d, dword ptr [r13 + r15*4 - 4]
    jmp execute
output:
    mov al, byte ptr [r14 + rbx]
    lea rdi, [rip + outbuf]
    mov byte ptr [rdi + r10], al
    inc r10d
    jmp execute

finished:
    test r10d, r10d
    jz success
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbuf]
    mov edx, r10d
    syscall
success:
    xor edi, edi
    mov eax, 60
    syscall
invalid:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.balign 16
source:
    .zero 65537
commands:
    .zero 65536
.balign 4
jumps:
    .zero 262144
brackets:
    .zero 4000
tape:
    .zero 30000
outbuf:
    .zero 65536
