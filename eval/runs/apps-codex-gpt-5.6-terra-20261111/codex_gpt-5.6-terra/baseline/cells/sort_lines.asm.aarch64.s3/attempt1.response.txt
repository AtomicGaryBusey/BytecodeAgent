.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x22, #32769

read_loop:
    sub x2, x22, x20
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    add x25, x19, x20
    mov x23, x19
    mov x24, x19
    mov x26, #0

    adrp x12, line_ptrs
    add x12, x12, :lo12:line_ptrs
    adrp x13, line_lens
    add x13, x13, :lo12:line_lens

parse_loop:
    cmp x23, x25
    b.hs parse_done
    ldrb w0, [x23]
    cmp w0, #10
    b.ne parse_not_newline
    sub x1, x23, x24
    str x24, [x12, x26, lsl #3]
    str w1, [x13, x26, lsl #2]
    add x26, x26, #1
    add x23, x23, #1
    mov x24, x23
    b parse_loop

parse_not_newline:
    add x23, x23, #1
    b parse_loop

parse_done:
    cmp x24, x25
    b.hs sort_begin
    sub x1, x25, x24
    str x24, [x12, x26, lsl #3]
    str w1, [x13, x26, lsl #2]
    add x26, x26, #1

sort_begin:
    mov x27, #1

sort_outer:
    cmp x27, x26
    b.hs output_begin
    ldr x10, [x12, x27, lsl #3]
    ldr w11, [x13, x27, lsl #2]
    mov x14, x27

sort_inner:
    cbz x14, sort_place
    sub x19, x14, #1
    ldr x15, [x12, x19, lsl #3]
    ldr w16, [x13, x19, lsl #2]
    mov w17, #0

compare_bytes:
    cmp w17, w16
    b.hs compare_prefix
    cmp w17, w11
    b.hs compare_prefix
    ldrb w18, [x15, x17]
    ldrb w0, [x10, x17]
    cmp w18, w0
    b.ne compare_different
    add w17, w17, #1
    b compare_bytes

compare_different:
    b.hi sort_shift
    b sort_place

compare_prefix:
    cmp w16, w11
    b.hi sort_shift
    b sort_place

sort_shift:
    str x15, [x12, x14, lsl #3]
    str w16, [x13, x14, lsl #2]
    mov x14, x19
    b sort_inner

sort_place:
    str x10, [x12, x14, lsl #3]
    str w11, [x13, x14, lsl #2]
    add x27, x27, #1
    b sort_outer

output_begin:
    mov x27, #0

output_loop:
    cmp x27, x26
    b.hs exit_program
    ldr x1, [x12, x27, lsl #3]
    ldr w2, [x13, x27, lsl #2]
    cbz w2, output_newline
    mov x0, #1
    mov x8, #64
    svc #0

output_newline:
    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    add x27, x27, #1
    b output_loop

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.align 3
input_buf:
    .zero 32769
.align 3
line_ptrs:
    .zero 8000
.align 2
line_lens:
    .zero 4000
