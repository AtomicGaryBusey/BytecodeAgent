.text
.equ INPUT_MAX, 49152

.globl _start
_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, #0

read_loop:
    mov x2, #INPUT_MAX
    sub x2, x2, x20
    cbz x2, begin_encode
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, begin_encode
    add x20, x20, x0
    b read_loop

begin_encode:
    mov x21, x19
    add x22, x19, x20
    adrp x23, outbuf
    add x23, x23, :lo12:outbuf
    mov x27, x23
    adrp x26, base64_table
    add x26, x26, :lo12:base64_table
    mov w24, #0

full_group_loop:
    sub x0, x22, x21
    cmp x0, #3
    b.lo final_group

    ldrb w0, [x21], #1
    ldrb w1, [x21], #1
    ldrb w2, [x21], #1

    lsr w3, w0, #2

    and w4, w0, #3
    lsl w4, w4, #4
    orr w4, w4, w1, lsr #4

    and w5, w1, #15
    lsl w5, w5, #2
    orr w5, w5, w2, lsr #6

    and w6, w2, #63

    ldrb w7, [x26, w3, uxtw]
    strb w7, [x23], #1
    ldrb w7, [x26, w4, uxtw]
    strb w7, [x23], #1
    ldrb w7, [x26, w5, uxtw]
    strb w7, [x23], #1
    ldrb w7, [x26, w6, uxtw]
    strb w7, [x23], #1

    add w24, w24, #4
    cmp w24, #76
    b.ne full_group_loop
    mov w7, #10
    strb w7, [x23], #1
    mov w24, #0
    b full_group_loop

final_group:
    sub x0, x22, x21
    cbz x0, final_line
    cmp x0, #1
    b.eq one_byte

two_bytes:
    ldrb w0, [x21], #1
    ldrb w1, [x21], #1

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4
    orr w4, w4, w1, lsr #4
    and w5, w1, #15
    lsl w5, w5, #2

    ldrb w7, [x26, w3, uxtw]
    strb w7, [x23], #1
    ldrb w7, [x26, w4, uxtw]
    strb w7, [x23], #1
    ldrb w7, [x26, w5, uxtw]
    strb w7, [x23], #1
    mov w7, #61
    strb w7, [x23], #1
    mov w7, #10
    strb w7, [x23], #1
    b write_output

one_byte:
    ldrb w0, [x21], #1

    lsr w3, w0, #2
    and w4, w0, #3
    lsl w4, w4, #4

    ldrb w7, [x26, w3, uxtw]
    strb w7, [x23], #1
    ldrb w7, [x26, w4, uxtw]
    strb w7, [x23], #1
    mov w7, #61
    strb w7, [x23], #1
    strb w7, [x23], #1
    mov w7, #10
    strb w7, [x23], #1
    b write_output

final_line:
    cbz w24, write_output
    mov w7, #10
    strb w7, [x23], #1

write_output:
    sub x2, x23, x27
    cbz x2, exit_ok
    mov x0, #1
    mov x1, x27
    mov x8, #64
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
inbuf:
    .zero 49152
.align 4
outbuf:
    .zero 66560
