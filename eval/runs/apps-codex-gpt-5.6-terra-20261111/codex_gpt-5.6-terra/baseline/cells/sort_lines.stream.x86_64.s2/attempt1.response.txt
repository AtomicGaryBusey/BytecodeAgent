.equ BUF_SIZE, 32768

.text
.globl _start
_start:
    xor r12d, r12d

read_loop:
    cmp r12, BUF_SIZE
    je read_full
    lea rsi, [rip + buffer]
    add rsi, r12
    mov edx, BUF_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

read_full:
    lea rsi, [rip + buffer + BUF_SIZE]
    mov edx, 1
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    jmp read_full

parse_input:
    lea rsi, [rip + buffer]
    lea r15, [rip + buffer]
    add r15, r12
    mov rdi, rsi
    xor r13d, r13d

parse_loop:
    cmp rsi, r15
    jae parse_done
    cmp byte ptr [rsi], 10
    jne parse_next
    mov rax, rsi
    sub rax, rdi
    mov qword ptr [lines + r13*8], rdi
    mov dword ptr [lengths + r13*4], eax
    inc r13
    inc rsi
    mov rdi, rsi
    jmp parse_loop

parse_next:
    inc rsi
    jmp parse_loop

parse_done:
    cmp rdi, r15
    je sort_start
    mov rax, r15
    sub rax, rdi
    mov qword ptr [lines + r13*8], rdi
    mov dword ptr [lengths + r13*4], eax
    inc r13

sort_start:
    mov r14, 1

sort_outer:
    cmp r14, r13
    jae output_start
    mov r8, qword ptr [lines + r14*8]
    mov r9d, dword ptr [lengths + r14*4]
    mov rbx, r14

sort_inner:
    test rbx, rbx
    jz sort_insert
    mov rax, rbx
    dec rax
    mov r10, qword ptr [lines + rax*8]
    mov r11d, dword ptr [lengths + rax*4]

    mov rsi, r10
    mov rdi, r8
    mov ecx, r11d
    cmp ecx, r9d
    cmova ecx, r9d
    test ecx, ecx
    jz compare_prefix
    repe cmpsb
    jne compare_diff

compare_prefix:
    cmp r11d, r9d
    ja sort_shift
    jmp sort_insert

compare_diff:
    ja sort_shift
    jmp sort_insert

sort_shift:
    mov rax, rbx
    dec rax
    mov rcx, qword ptr [lines + rax*8]
    mov qword ptr [lines + rbx*8], rcx
    mov ecx, dword ptr [lengths + rax*4]
    mov dword ptr [lengths + rbx*4], ecx
    dec rbx
    jmp sort_inner

sort_insert:
    mov qword ptr [lines + rbx*8], r8
    mov dword ptr [lengths + rbx*4], r9d
    inc r14
    jmp sort_outer

output_start:
    xor r14d, r14d

output_loop:
    cmp r14, r13
    jae exit_ok
    mov rsi, qword ptr [lines + r14*8]
    mov edx, dword ptr [lengths + r14*4]
    test edx, edx
    jz output_newline
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc r14
    jmp output_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 8
buffer:
    .zero 32769
.align 8
lines:
    .zero 8000
.align 4
lengths:
    .zero 4000
