.text
.globl _start

_start:
    sub sp, sp, #16

    adrp x9, input_buf
    add x9, x9, :lo12:input_buf
    mov x10, x9
    add x11, x9, #16, lsl #12

read_loop:
    cmp x10, x11
    b.eq read_full
    mov x0, #0
    mov x1, x10
    sub x2, x11, x10
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x10, x10, x0
    b read_loop

read_full:
    mov x0, #0
    mov x1, sp
    mov x2, #1
    mov x8, #63
    svc #0
    cbz x0, read_done
    b bad_exit

read_done:
    mov x12, x9
find_separator:
    cmp x12, x10
    b.eq no_separator
    ldrb w14, [x12], #1
    cmp w14, #33
    b.ne find_separator
    sub x20, x12, #1
    mov x22, x12
    mov x23, x10
    b validate_start

no_separator:
    mov x20, x10
    mov x22, x10
    mov x23, x10

validate_start:
    adrp x26, jump_table
    add x26, x26, :lo12:jump_table
    adrp x15, bracket_stack
    add x15, x15, :lo12:bracket_stack
    mov x16, x15
    mov x17, x9

validate_loop:
    cmp x17, x20
    b.eq validate_end
    sub x27, x17, x9
    ldrb w14, [x17], #1
    cmp w14, #91
    b.eq validate_open
    cmp w14, #93
    b.eq validate_close
    b validate_loop

validate_open:
    str w27, [x16], #4
    b validate_loop

validate_close:
    cmp x16, x15
    b.eq bad_exit
    ldr w0, [x16, #-4]!
    str w27, [x26, x0, lsl #2]
    str w0, [x26, x27, lsl #2]
    b validate_loop

validate_end:
    cmp x16, x15
    b.ne bad_exit

    mov x19, x9
    adrp x21, tape
    add x21, x21, :lo12:tape
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf
    mov x24, x25
    add x28, x25, #16, lsl #12

run_loop:
    cmp x19, x20
    b.eq run_done
    sub x27, x19, x9
    ldrb w14, [x19], #1
    cmp w14, #62
    b.eq cmd_right
    cmp w14, #60
    b.eq cmd_left
    cmp w14, #43
    b.eq cmd_plus
    cmp w14, #45
    b.eq cmd_minus
    cmp w14, #46
    b.eq cmd_dot
    cmp w14, #44
    b.eq cmd_comma
    cmp w14, #91
    b.eq cmd_open
    cmp w14, #93
    b.eq cmd_close
    b run_loop

cmd_right:
    add x21, x21, #1
    b run_loop

cmd_left:
    sub x21, x21, #1
    b run_loop

cmd_plus:
    ldrb w0, [x21]
    add w0, w0, #1
    strb w0, [x21]
    b run_loop

cmd_minus:
    ldrb w0, [x21]
    sub w0, w0, #1
    strb w0, [x21]
    b run_loop

cmd_dot:
    ldrb w0, [x21]
    strb w0, [x24], #1
    cmp x24, x28
    b.ne run_loop
    bl flush_output
    b run_loop

cmd_comma:
    cmp x22, x23
    b.hs comma_empty
    ldrb w0, [x22], #1
    strb w0, [x21]
    b run_loop

comma_empty:
    strb wzr, [x21]
    b run_loop

cmd_open:
    ldrb w0, [x21]
    cbnz w0, run_loop
    ldr w0, [x26, x27, lsl #2]
    add x19, x9, x0
    add x19, x19, #1
    b run_loop

cmd_close:
    ldrb w0, [x21]
    cbz w0, run_loop
    ldr w0, [x26, x27, lsl #2]
    add x19, x9, x0
    add x19, x19, #1
    b run_loop

run_done:
    bl flush_output
    mov x0, #0
    mov x8, #93
    svc #0

bad_exit:
    mov x0, #1
    mov x8, #93
    svc #0

flush_output:
    cmp x24, x25
    b.eq flush_return
    mov x0, #1
    mov x1, x25
    sub x2, x24, x25
    mov x8, #64
    svc #0
    mov x24, x25
flush_return:
    ret
