.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    mov eax, 0
    mov edi, 0
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz output

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + buffer]

scan:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne not_newline
    inc r12
not_newline:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15d, r15d
    jnz already_word
    inc r13
    mov r15d, 1
already_word:
    dec rcx
    jnz scan
    jmp read_more

whitespace:
    xor r15d, r15d
    dec rcx
    jnz scan
    jmp read_more

output:
    lea rdi, [rip + out_end]
    dec rdi
    mov byte ptr [rdi], 10

    mov rax, r14
    call emit_number
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call emit_number
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call emit_number

    mov rsi, rdi
    lea rdx, [rip + out_end]
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

emit_number:
    test rax, rax
    jnz emit_loop
    dec rdi
    mov byte ptr [rdi], 48
    ret

emit_loop:
    mov ecx, 10
    xor edx, edx
    div rcx
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz emit_loop
    ret

.bss
.align 16
buffer:
.zero 4096
out:
.zero 64
out_end:
