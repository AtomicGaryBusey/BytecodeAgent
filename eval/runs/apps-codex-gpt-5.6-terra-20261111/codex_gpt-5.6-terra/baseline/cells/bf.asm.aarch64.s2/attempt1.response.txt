.text
.globl _start

_start:
	adrp x20, input_buf
	add x20, x20, :lo12:input_buf
	mov x19, xzr
	movz x22, #1, lsl #16

read_loop:
	cmp x19, x22
	b.eq read_done
	mov x0, xzr
	add x1, x20, x19
	sub x2, x22, x19
	mov x8, #63
	svc #0
	cbz x0, read_done
	add x19, x19, x0
	b read_loop

read_done:
	mov x21, x19
	mov x24, x19
	mov x0, xzr

find_separator:
	cmp x0, x19
	b.hs separator_done
	ldrb w1, [x20, x0]
	cmp w1, #33
	b.ne no_separator
	mov x21, x0
	add x24, x0, #1
	b separator_done

no_separator:
	add x0, x0, #1
	b find_separator

separator_done:
	adrp x22, bracket_map
	add x22, x22, :lo12:bracket_map
	adrp x23, bracket_stack
	add x23, x23, :lo12:bracket_stack
	mov w25, wzr
	mov w0, wzr

check_brackets:
	cmp w0, w21
	b.hs brackets_done
	ldrb w1, [x20, w0, uxtw]
	cmp w1, #91
	b.eq open_bracket
	cmp w1, #93
	b.eq close_bracket
	add w0, w0, #1
	b check_brackets

open_bracket:
	str w0, [x23, w25, uxtw #2]
	add w25, w25, #1
	add w0, w0, #1
	b check_brackets

close_bracket:
	cbz w25, bracket_error
	sub w25, w25, #1
	ldr w2, [x23, w25, uxtw #2]
	str w0, [x22, w2, uxtw #2]
	str w2, [x22, w0, uxtw #2]
	add w0, w0, #1
	b check_brackets

brackets_done:
	cbnz w25, bracket_error

	adrp x25, tape
	add x25, x25, :lo12:tape
	adrp x26, output_buf
	add x26, x26, :lo12:output_buf
	mov w23, wzr

execute_loop:
	cmp w23, w21
	b.hs execution_done
	ldrb w0, [x20, w23, uxtw]
	add w23, w23, #1

	cmp w0, #62
	b.eq command_right
	cmp w0, #60
	b.eq command_left
	cmp w0, #43
	b.eq command_plus
	cmp w0, #45
	b.eq command_minus
	cmp w0, #46
	b.eq command_dot
	cmp w0, #44
	b.eq command_comma
	cmp w0, #91
	b.eq command_open
	cmp w0, #93
	b.eq command_close
	b execute_loop

command_right:
	add x25, x25, #1
	b execute_loop

command_left:
	sub x25, x25, #1
	b execute_loop

command_plus:
	ldrb w1, [x25]
	add w1, w1, #1
	strb w1, [x25]
	b execute_loop

command_minus:
	ldrb w1, [x25]
	sub w1, w1, #1
	strb w1, [x25]
	b execute_loop

command_dot:
	ldrb w1, [x25]
	strb w1, [x26], #1
	b execute_loop

command_comma:
	cmp x24, x19
	b.hs comma_zero
	ldrb w1, [x20, x24]
	add x24, x24, #1
	strb w1, [x25]
	b execute_loop

comma_zero:
	strb wzr, [x25]
	b execute_loop

command_open:
	ldrb w1, [x25]
	cbnz w1, execute_loop
	sub w2, w23, #1
	ldr w2, [x22, w2, uxtw #2]
	add w23, w2, #1
	b execute_loop

command_close:
	ldrb w1, [x25]
	cbz w1, execute_loop
	sub w2, w23, #1
	ldr w2, [x22, w2, uxtw #2]
	add w23, w2, #1
	b execute_loop

execution_done:
	adrp x1, output_buf
	add x1, x1, :lo12:output_buf
	sub x2, x26, x1
	cbz x2, exit_success
	mov x0, #1
	mov x8, #64
	svc #0

exit_success:
	mov x0, xzr
	mov x8, #93
	svc #0

bracket_error:
	mov x0, #1
	mov x8, #93
	svc #0

.bss

.align 4

.align 4
