.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    lea r8, [rip + output_buffer]
    xor r13d, r13d

read_loop:
    xor eax, eax
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    je input_done
    add r13, rax
    jmp read_loop

input_done:
    xor r14d, r14d

line_loop:
    cmp r14, r13
    jae done
    mov rbx, r14

find_newline:
    cmp rbx, r13
    jae final_line
    mov al, byte ptr [r12 + rbx]
    cmp al, 10
    je newline_line
    inc rbx
    jmp find_newline

newline_line:
    mov r10d, 1
    jmp emit_line

final_line:
    xor r10d, r10d

emit_line:
    mov r9, r8
    mov rax, rbx

reverse_loop:
    cmp rax, r14
    je reverse_done
    dec rax
    mov cl, byte ptr [r12 + rax]
    mov byte ptr [r9], cl
    inc r9
    jmp reverse_loop

reverse_done:
    mov byte ptr [r9], 10
    inc r9
    mov rdx, r9
    sub rdx, r8
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    syscall
    test r10d, r10d
    je done
    lea r14, [rbx + 1]
    jmp line_loop

done:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 65537
