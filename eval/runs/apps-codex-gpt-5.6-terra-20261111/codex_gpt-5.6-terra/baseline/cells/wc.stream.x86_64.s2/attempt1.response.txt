.globl _start

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz make_output

    lea rbp, [rsi + rax]

process_byte:
    mov al, byte ptr [rsi]
    inc rsi
    inc r14

    cmp al, 10
    jne not_newline
    inc r12

not_newline:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb nonwhitespace
    cmp al, 13
    jbe whitespace

nonwhitespace:
    test r15, r15
    jnz byte_done
    inc r13
    mov r15d, 1
    jmp byte_done

whitespace:
    xor r15d, r15d

byte_done:
    cmp rsi, rbp
    jb process_byte
    jmp read_more

make_output:
    lea rdi, [rip + outbuf]
    mov rbx, 10

    mov rax, r12
    call decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

decimal:
    test rax, rax
    jnz decimal_nonzero
    mov byte ptr [rdi], 48
    inc rdi
    ret

decimal_nonzero:
    xor r8d, r8d
decimal_divide:
    xor edx, edx
    div rbx
    push rdx
    inc r8
    test rax, rax
    jnz decimal_divide

decimal_store:
    pop rdx
    add dl, 48
    mov byte ptr [rdi], dl
    inc rdi
    dec r8
    jnz decimal_store
    ret

.bss
.align 16
buffer:
.zero 4096
outbuf:
.zero 64
