.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_more:
    movz x9, #1, lsl #16
    add x9, x9, #1
    sub x2, x9, x20
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_more

read_done:
    add x3, x19, x20
    mov x4, x19

find_separator:
    cmp x4, x3
    b.hs no_separator
    ldrb w5, [x4], #1
    cmp w5, #33
    b.eq separator_found
    b find_separator

separator_found:
    sub x20, x4, #1
    mov x23, x4
    mov x24, x3
    b setup

no_separator:
    mov x20, x3
    mov x23, x3
    mov x24, x3

setup:
    adrp x22, tape
    add x22, x22, :lo12:tape
    adrp x25, bracket_map
    add x25, x25, :lo12:bracket_map
    adrp x26, bracket_stack
    add x26, x26, :lo12:bracket_stack
    mov x27, #0
    mov x21, x19

validate_loop:
    cmp x21, x20
    b.hs validate_done
    ldrb w0, [x21]
    cmp w0, #91
    b.eq validate_open
    cmp w0, #93
    b.eq validate_close
    add x21, x21, #1
    b validate_loop

validate_open:
    sub x1, x21, x19
    str w1, [x26, x27, lsl #2]
    add x27, x27, #1
    add x21, x21, #1
    b validate_loop

validate_close:
    cbz x27, invalid_program
    sub x27, x27, #1
    ldr w2, [x26, x27, lsl #2]
    sub x1, x21, x19
    str w1, [x25, w2, uxtw #2]
    str w2, [x25, w1, uxtw #2]
    add x21, x21, #1
    b validate_loop

validate_done:
    cbnz x27, invalid_program
    mov x21, x19

exec_loop:
    cmp x21, x20
    b.hs success
    ldrb w0, [x21], #1
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_plus
    cmp w0, #45
    b.eq cmd_minus
    cmp w0, #46
    b.eq cmd_dot
    cmp w0, #44
    b.eq cmd_comma
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close
    b exec_loop

cmd_right:
    add x22, x22, #1
    b exec_loop

cmd_left:
    sub x22, x22, #1
    b exec_loop

cmd_plus:
    ldrb w1, [x22]
    add w1, w1, #1
    strb w1, [x22]
    b exec_loop

cmd_minus:
    ldrb w1, [x22]
    sub w1, w1, #1
    strb w1, [x22]
    b exec_loop

cmd_dot:
    mov x0, #1
    mov x1, x22
    mov x2, #1
    mov x8, #64
    svc #0
    b exec_loop

cmd_comma:
    cmp x23, x24
    b.hs cmd_comma_zero
    ldrb w1, [x23], #1
    strb w1, [x22]
    b exec_loop

cmd_comma_zero:
    strb wzr, [x22]
    b exec_loop

cmd_open:
    ldrb w1, [x22]
    cbnz w1, exec_loop
    sub x1, x21, x19
    sub x1, x1, #1
    ldr w2, [x25, w1, uxtw #2]
    add x21, x19, w2, uxtw
    add x21, x21, #1
    b exec_loop

cmd_close:
    ldrb w1, [x22]
    cbz w1, exec_loop
    sub x1, x21, x19
    sub x1, x1, #1
    ldr w2, [x25, w1, uxtw #2]
    add x21, x19, w2, uxtw
    add x21, x21, #1
    b exec_loop

invalid_program:
    mov x0, #1
    mov x8, #93
    svc #0

success:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.p2align 4
input_buf:
    .zero 65537
.p2align 4
tape:
    .zero 30000
.p2align 2
bracket_map:
    .zero 262144
.p2align 2
bracket_stack:
    .zero 4000
