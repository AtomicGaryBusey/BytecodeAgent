.equ BUF_SIZE, 65536

.section .rodata
okmsg:
.ascii "ok\n"
oklen = . - okmsg
errmsg:
.ascii "error at "
errlen = . - errmsg

.bss
.align 16
buffer:
.zero 65536
.align 16
outbuf:
.zero 64

.text
.globl _start
_start:
    lea r12, [rip + buffer]
    xor r13, r13
read_loop:
    mov rdx, BUF_SIZE
    sub rdx, r13
    lea rsi, [r12 + r13]
    xor rdi, rdi
    xor rax, rax
    syscall
    test rax, rax
    jz input_done
    add r13, rax
    cmp r13, BUF_SIZE
    jb read_loop

input_done:
    mov rsi, r12
    lea rdi, [r12 + r13]
    call skip_ws
    cmp rsi, rdi
    je bad
    call parse_value
    call skip_ws
    cmp rsi, rdi
    jne bad

good:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + okmsg]
    mov rdx, oklen
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

bad:
    mov rbx, rsi
    sub rbx, r12
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + errmsg]
    mov rdx, errlen
    syscall

    lea r8, [rip + outbuf + 48]
    mov byte ptr [r8], 10
    mov rax, rbx
    test rax, rax
    jnz decimal_loop
    dec r8
    mov byte ptr [r8], 48
    jmp decimal_done

decimal_loop:
    mov rcx, 10
decimal_digit:
    xor rdx, rdx
    div rcx
    add dl, 48
    dec r8
    mov byte ptr [r8], dl
    test rax, rax
    jnz decimal_digit

decimal_done:
    lea rdx, [rip + outbuf + 49]
    sub rdx, r8
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    syscall
    mov rax, 60
    mov rdi, 1
    syscall

skip_ws:
    cmp rsi, rdi
    je skip_ws_done
skip_ws_loop:
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je skip_ws_take
    cmp al, 9
    je skip_ws_take
    cmp al, 10
    je skip_ws_take
    cmp al, 13
    jne skip_ws_done
skip_ws_take:
    inc rsi
    cmp rsi, rdi
    jne skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp rsi, rdi
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 116
    je parse_true
    cmp al, 102
    je parse_false
    cmp al, 110
    je parse_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb bad
    cmp al, 57
    jbe parse_number
    jmp bad

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 125
    jne object_member
    inc rsi
    ret

object_member:
    cmp byte ptr [rsi], 34
    jne bad
    call parse_string
    call skip_ws
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 58
    jne bad
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 125
    je object_close
    cmp byte ptr [rsi], 44
    jne bad
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je bad
    jmp object_member

object_close:
    inc rsi
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 93
    jne array_value
    inc rsi
    ret

array_value:
    call parse_value
    call skip_ws
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 93
    je array_close
    cmp byte ptr [rsi], 44
    jne bad
    inc rsi
    call skip_ws
    cmp rsi, rdi
    je bad
    jmp array_value

array_close:
    inc rsi
    ret

parse_string:
    inc rsi
string_loop:
    cmp rsi, rdi
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 34
    je string_end
    cmp al, 92
    je string_escape
    cmp al, 32
    jb bad
    inc rsi
    jmp string_loop

string_end:
    inc rsi
    ret

string_escape:
    inc rsi
    cmp rsi, rdi
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 34
    je string_simple
    cmp al, 92
    je string_simple
    cmp al, 47
    je string_simple
    cmp al, 98
    je string_simple
    cmp al, 102
    je string_simple
    cmp al, 110
    je string_simple
    cmp al, 114
    je string_simple
    cmp al, 116
    je string_simple
    cmp al, 117
    jne bad
    inc rsi
    mov ecx, 4
string_hex_loop:
    cmp rsi, rdi
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb string_hex_bad
    cmp al, 57
    jbe string_hex_ok
    cmp al, 65
    jb string_hex_lower
    cmp al, 70
    jbe string_hex_ok
string_hex_lower:
    cmp al, 97
    jb string_hex_bad
    cmp al, 102
    ja string_hex_bad
string_hex_ok:
    inc rsi
    dec ecx
    jnz string_hex_loop
    jmp string_loop
string_hex_bad:
    jmp bad

string_simple:
    inc rsi
    jmp string_loop

parse_true:
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 116
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 114
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 117
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 101
    jne bad
    inc rsi
    ret

parse_false:
    cmp byte ptr [rsi], 102
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 97
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 108
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 115
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 101
    jne bad
    inc rsi
    ret

parse_null:
    cmp byte ptr [rsi], 110
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 117
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 108
    jne bad
    inc rsi
    cmp rsi, rdi
    je bad
    cmp byte ptr [rsi], 108
    jne bad
    inc rsi
    ret

parse_number:
    cmp byte ptr [rsi], 45
    jne number_int
    inc rsi
    cmp rsi, rdi
    je bad

number_int:
    movzx eax, byte ptr [rsi]
    cmp al, 48
    je number_zero
    cmp al, 49
    jb bad
    cmp al, 57
    ja bad

number_digits:
    inc rsi
    cmp rsi, rdi
    je number_finish
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb number_after_int
    cmp al, 57
    jbe number_digits

number_after_int:
    cmp al, 46
    je number_frac
    cmp al, 101
    je number_exp
    cmp al, 69
    je number_exp
    ret

number_zero:
    inc rsi
    cmp rsi, rdi
    je number_finish
    movzx eax, byte ptr [rsi]
    cmp al, 46
    je number_frac
    cmp al, 101
    je number_exp
    cmp al, 69
    je number_exp
    ret

number_frac:
    inc rsi
    cmp rsi, rdi
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb bad
    cmp al, 57
    ja bad
number_frac_digits:
    inc rsi
    cmp rsi, rdi
    je number_finish
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb number_after_frac
    cmp al, 57
    jbe number_frac_digits

number_after_frac:
    cmp al, 101
    je number_exp
    cmp al, 69
    je number_exp
    ret

number_exp:
    inc rsi
    cmp rsi, rdi
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 43
    je number_exp_sign
    cmp al, 45
    jne number_exp_first
number_exp_sign:
    inc rsi
    cmp rsi, rdi
    je bad
number_exp_first:
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb bad
    cmp al, 57
    ja bad
number_exp_digits:
    inc rsi
    cmp rsi, rdi
    je number_finish
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb number_finish
    cmp al, 57
    jbe number_exp_digits
    ret

number_finish:
    ret
