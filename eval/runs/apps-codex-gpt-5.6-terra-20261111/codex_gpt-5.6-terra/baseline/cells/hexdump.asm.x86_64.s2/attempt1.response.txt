.text
.globl _start
_start:
    lea rbx, [rip + input]
    xor r12d, r12d

read_more:
    cmp r12, 65536
    jne read_space
    xor edx, edx
    lea rsi, [rip + input]
    jmp do_read

read_space:
    mov rdx, 65536
    sub rdx, r12
    lea rsi, [rbx + r12]

do_read:
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz begin_dump
    add r12, rax
    jmp read_more

begin_dump:
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae final_offset

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ok
    mov r14, 16

row_size_ok:
    lea r10, [rip + output]
    mov rdi, r10

    mov eax, r13d
    mov ecx, 28
offset_hex:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [hex + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns offset_hex

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi

    lea r15, [rbx + r13]
    xor r8d, r8d

hex_slots:
    cmp r8, r14
    jae empty_slot

    movzx edx, byte ptr [r15 + r8]
    mov eax, edx
    shr eax, 4
    mov al, byte ptr [hex + rax]
    mov byte ptr [rdi], al
    inc rdi
    and edx, 15
    mov dl, byte ptr [hex + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp slot_done

empty_slot:
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi

slot_done:
    cmp r8, 7
    jne no_middle_space
    mov byte ptr [rdi], 0x20
    inc rdi

no_middle_space:
    inc r8
    cmp r8, 16
    jb hex_slots

    mov byte ptr [rdi], 0x20
    inc rdi
    mov byte ptr [rdi], 0x7c
    inc rdi

    xor r8d, r8d
ascii_loop:
    cmp r8, r14
    jae ascii_done
    movzx edx, byte ptr [r15 + r8]
    cmp edx, 0x20
    jb ascii_dot
    cmp edx, 0x7e
    ja ascii_dot
    mov byte ptr [rdi], dl
    jmp ascii_store_done

ascii_dot:
    mov byte ptr [rdi], 0x2e

ascii_store_done:
    inc rdi
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    inc rdi
    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rdx, rdi
    sub rdx, r10
    mov edi, 1
    mov rsi, r10
    mov eax, 1
    syscall

    add r13, r14
    jmp row_loop

final_offset:
    lea r10, [rip + output]
    mov rdi, r10
    mov eax, r12d
    mov ecx, 28

final_hex:
    mov edx, eax
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [hex + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns final_hex

    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rdx, rdi
    sub rdx, r10
    mov edi, 1
    mov rsi, r10
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hex:
    .ascii "0123456789abcdef"

.bss
.align 16
input:
    .zero 65536
.align 16
output:
    .zero 128
