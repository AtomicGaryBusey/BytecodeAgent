.globl _start

.text
_start:
    call reset_line

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jz eof
    lea rbp, [rip + inbuf]
    lea r12, [rbp + rax]

byte_loop:
    cmp rbp, r12
    jae read_more
    movzx eax, byte ptr [rbp]
    inc rbp
    cmp al, 10
    je newline
    mov byte ptr [rip + lineseen], 1
    cmp al, 32
    je separator
    cmp al, 9
    je separator
    cmp byte ptr [rip + bad], 0
    jne byte_loop
    mov rcx, qword ptr [rip + toklen]
    cmp rcx, 19
    jae token_too_long
    lea rdx, [rip + token]
    mov byte ptr [rdx + rcx], al
    inc rcx
    mov qword ptr [rip + toklen], rcx
    jmp byte_loop

token_too_long:
    mov byte ptr [rip + bad], 1
    jmp byte_loop

separator:
    call finish_token
    jmp byte_loop

newline:
    call finish_token
    call finish_line
    call reset_line
    jmp byte_loop

eof:
    cmp byte ptr [rip + lineseen], 0
    je exit_program
    call finish_token
    call finish_line

exit_program:
    mov eax, 60
    xor edi, edi
    syscall

reset_line:
    mov qword ptr [rip + toklen], 0
    mov qword ptr [rip + valcount], 0
    mov byte ptr [rip + bad], 0
    mov byte ptr [rip + lineseen], 0
    ret

finish_token:
    mov r8, qword ptr [rip + toklen]
    test r8, r8
    jz ft_done
    mov qword ptr [rip + toklen], 0
    cmp byte ptr [rip + bad], 0
    jne ft_done

    cmp r8, 1
    jne parse_literal
    mov al, byte ptr [rip + token]
    cmp al, '+'
    je do_operator
    cmp al, '-'
    je do_operator
    cmp al, '*'
    je do_operator
    cmp al, '/'
    je do_operator
    cmp al, '%'
    je do_operator

parse_literal:
    lea rsi, [rip + token]
    xor ecx, ecx
    xor edi, edi
    mov al, byte ptr [rsi]
    cmp al, '-'
    jne literal_digits
    mov edi, 1
    inc rcx
    cmp rcx, r8
    jae ft_bad

literal_digits:
    mov rdx, r8
    sub rdx, rcx
    cmp rdx, 18
    ja ft_bad
    xor rax, rax

digit_loop:
    cmp rcx, r8
    jae digits_done
    movzx edx, byte ptr [rsi + rcx]
    cmp dl, '0'
    jb ft_bad
    cmp dl, '9'
    ja ft_bad
    sub edx, '0'
    imul rax, rax, 10
    add rax, rdx
    inc rcx
    jmp digit_loop

digits_done:
    test edi, edi
    jz push_value
    neg rax

push_value:
    mov rcx, qword ptr [rip + valcount]
    lea rdx, [rip + values]
    mov qword ptr [rdx + rcx*8], rax
    inc rcx
    mov qword ptr [rip + valcount], rcx
    ret

do_operator:
    mov dl, al
    mov rcx, qword ptr [rip + valcount]
    cmp rcx, 2
    jb ft_bad
    lea r8, [rip + values]
    mov rbx, qword ptr [r8 + rcx*8 - 8]
    mov rax, qword ptr [r8 + rcx*8 - 16]
    cmp dl, '+'
    je op_add
    cmp dl, '-'
    je op_sub
    cmp dl, '*'
    je op_mul
    cmp dl, '/'
    je op_div
    jmp op_mod

op_add:
    add rax, rbx
    jmp op_store

op_sub:
    sub rax, rbx
    jmp op_store

op_mul:
    imul rax, rbx
    jmp op_store

op_div:
    test rbx, rbx
    jz ft_bad
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne div_ok
    cmp rbx, -1
    je ft_bad
div_ok:
    cqo
    idiv rbx
    jmp op_store

op_mod:
    test rbx, rbx
    jz ft_bad
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne mod_ok
    cmp rbx, -1
    je ft_bad
mod_ok:
    cqo
    idiv rbx
    mov rax, rdx

op_store:
    mov qword ptr [r8 + rcx*8 - 16], rax
    dec rcx
    mov qword ptr [rip + valcount], rcx
    ret

ft_bad:
    mov byte ptr [rip + bad], 1
ft_done:
    ret

finish_line:
    cmp byte ptr [rip + bad], 0
    jne print_error
    cmp qword ptr [rip + valcount], 1
    jne print_error
    lea rsi, [rip + values]
    mov rax, qword ptr [rsi]
    jmp print_number

print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errmsg]
    mov edx, 6
    syscall
    ret

print_number:
    lea rsi, [rip + outbuf + 31]
    mov byte ptr [rsi], 10
    mov ecx, 1
    xor edi, edi
    test rax, rax
    jns number_abs
    mov edi, 1
    neg rax

number_abs:
    mov r8, 10

number_loop:
    xor edx, edx
    div r8
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    inc rcx
    test rax, rax
    jne number_loop
    test edi, edi
    jz number_write
    dec rsi
    mov byte ptr [rsi], '-'
    inc rcx

number_write:
    mov rdx, rcx
    mov eax, 1
    mov edi, 1
    syscall
    ret

.section .rodata
errmsg:
    .ascii "error\n"

.bss
.align 8
inbuf:
    .zero 4096
token:
    .zero 32
values:
    .zero 8192
outbuf:
    .zero 32
toklen:
    .zero 8
valcount:
    .zero 8
bad:
    .zero 1
lineseen:
    .zero 1
