.section .rodata
errmsg:
.ascii "error\n"
errmsg_len = . - errmsg

.bss
.align 16
inbuf:
.zero 4096
linebuf:
.zero 1001
.align 16
valstack:
.zero 8192
outbuf:
.zero 32
line_len:
.zero 4

.text
.globl _start
_start:
lea r14, [rip + inbuf]

read_more:
xor eax, eax
xor edi, edi
mov rsi, r14
mov edx, 4096
syscall
test rax, rax
jle eof_input
mov r13, rax
xor r12d, r12d

input_loop:
cmp r12, r13
jae read_more
mov al, byte ptr [r14 + r12]
inc r12
cmp al, 10
je end_line
mov ecx, dword ptr [rip + line_len]
lea rdx, [rip + linebuf]
mov byte ptr [rdx + rcx], al
inc ecx
mov dword ptr [rip + line_len], ecx
jmp input_loop

end_line:
lea rdi, [rip + linebuf]
mov esi, dword ptr [rip + line_len]
call process_line
mov dword ptr [rip + line_len], 0
jmp input_loop

eof_input:
mov esi, dword ptr [rip + line_len]
test esi, esi
jz exit_ok
lea rdi, [rip + linebuf]
call process_line

exit_ok:
mov eax, 60
xor edi, edi
syscall

process_line:
mov r15, rsi
xor r8d, r8d
xor r9d, r9d
xor r10d, r10d

skip_seps:
cmp r8, r15
jae finish_line
movzx eax, byte ptr [rdi + r8]
cmp al, 32
je skip_one
cmp al, 9
je skip_one
jmp token_start
skip_one:
inc r8
jmp skip_seps

token_start:
mov rdx, r8
token_scan:
cmp r8, r15
jae token_end
movzx eax, byte ptr [rdi + r8]
cmp al, 32
je token_end
cmp al, 9
je token_end
inc r8
jmp token_scan

token_end:
mov rcx, r8
mov rax, rcx
sub rax, rdx
cmp rax, 1
jne literal_try
movzx eax, byte ptr [rdi + rdx]
cmp al, '+'
je op_plus
cmp al, '-'
je op_minus
cmp al, '*'
je op_mul
cmp al, '/'
je op_div
cmp al, '%'
je op_mod
cmp al, '0'
jb bad_line
cmp al, '9'
ja bad_line
jmp literal_try

literal_try:
xor ebx, ebx
movzx eax, byte ptr [rdi + rdx]
cmp al, '-'
jne literal_digits
mov ebx, 1
inc rdx

literal_digits:
mov r11, rcx
sub r11, rdx
test r11, r11
jz bad_line
cmp r11, 18
ja bad_line
xor eax, eax
mov rsi, rdx

digit_loop:
cmp rsi, rcx
jae digits_done
movzx edx, byte ptr [rdi + rsi]
cmp dl, '0'
jb bad_line
cmp dl, '9'
ja bad_line
sub edx, '0'
imul rax, rax, 10
add rax, rdx
inc rsi
jmp digit_loop

digits_done:
test ebx, ebx
jz push_literal
neg rax
push_literal:
mov qword ptr [rip + valstack + r9*8], rax
inc r9
jmp skip_seps

need_two:
cmp r9, 2
jb bad_line
dec r9
mov rbx, qword ptr [rip + valstack + r9*8]
dec r9
mov rax, qword ptr [rip + valstack + r9*8]
ret

op_plus:
call need_two
add rax, rbx
jmp push_result
op_minus:
call need_two
sub rax, rbx
jmp push_result
op_mul:
call need_two
imul rax, rbx
jmp push_result
op_div:
call need_two
test rbx, rbx
jz bad_line
mov rcx, 0x8000000000000000
cmp rax, rcx
jne div_do
cmp rbx, -1
je bad_line
div_do:
cqo
idiv rbx
jmp push_result
op_mod:
call need_two
test rbx, rbx
jz bad_line
mov rcx, 0x8000000000000000
cmp rax, rcx
jne mod_do
cmp rbx, -1
je bad_line
mod_do:
cqo
idiv rbx
mov rax, rdx

push_result:
mov qword ptr [rip + valstack + r9*8], rax
inc r9
jmp skip_seps

finish_line:
cmp r9, 1
jne bad_line
mov rax, qword ptr [rip + valstack]
call print_value
ret

bad_line:
mov eax, 1
mov edi, 1
lea rsi, [rip + errmsg]
mov edx, errmsg_len
syscall
ret

print_value:
lea rsi, [rip + outbuf + 32]
dec rsi
mov byte ptr [rsi], 10
xor r8d, r8d
test rax, rax
jns print_digits
mov r8d, 1
neg rax

print_digits:
mov ebx, 10
print_digit_loop:
xor edx, edx
div rbx
add dl, '0'
dec rsi
mov byte ptr [rsi], dl
test rax, rax
jnz print_digit_loop
test r8d, r8d
jz print_write
dec rsi
mov byte ptr [rsi], '-'

print_write:
lea rdx, [rip + outbuf + 32]
sub rdx, rsi
mov eax, 1
mov edi, 1
syscall
ret
