.text
.globl _start

.equ INPUT_MAX, 49152

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    mov x3, #INPUT_MAX
    cmp x20, x3
    b.ne read_more

    add x1, x19, x20
    mov x0, #0
    mov x2, #1
    mov x8, #63
    svc #0
    b encode_start

read_more:
    add x1, x19, x20
    sub x2, x3, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, encode_start
    add x20, x20, x0
    b read_loop

encode_start:
    cbz x20, exit_ok

    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    adrp x22, output_buf
    add x22, x22, :lo12:output_buf
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet
    mov x23, #0

    mov x0, #3
    udiv x25, x20, x0
    msub x26, x25, x0, x20

    cbz x25, partial_check

full_loop:
    ldrb w3, [x21], #1
    ldrb w4, [x21], #1
    ldrb w5, [x21], #1

    lsr w7, w3, #2
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w3, #3
    lsl w7, w7, #4
    lsr w8, w4, #4
    orr w7, w7, w8
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w4, #15
    lsl w7, w7, #2
    lsr w8, w5, #6
    orr w7, w7, w8
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w5, #63
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    add x23, x23, #4
    cmp x23, #76
    b.ne full_no_newline
    mov w9, #10
    strb w9, [x22], #1
    mov x23, #0

full_no_newline:
    subs x25, x25, #1
    b.ne full_loop

partial_check:
    cbz x26, finish_full

    ldrb w3, [x21]
    lsr w7, w3, #2
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w3, #3
    lsl w7, w7, #4

    cmp x26, #1
    b.ne partial_two

    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1
    mov w9, #61
    strb w9, [x22], #1
    strb w9, [x22], #1
    b final_quartet

partial_two:
    ldrb w4, [x21, #1]
    lsr w8, w4, #4
    orr w7, w7, w8
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    and w7, w4, #15
    lsl w7, w7, #2
    ldrb w9, [x24, w7, uxtw]
    strb w9, [x22], #1

    mov w9, #61
    strb w9, [x22], #1

final_quartet:
    add x23, x23, #4
    cmp x23, #76
    b.ne final_short
    mov w9, #10
    strb w9, [x22], #1
    b write_output

final_short:
    mov w9, #10
    strb w9, [x22], #1
    b write_output

finish_full:
    cbz x23, write_output
    mov w9, #10
    strb w9, [x22], #1

write_output:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x22, x1
    mov x0, #1
    mov x8, #64
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buf:
    .zero 49153
.balign 16
output_buf:
    .zero 66400
