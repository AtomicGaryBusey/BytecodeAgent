.text
.globl _start

_start:
    lea r15, [rip + inputbuf]
read_loop:
    lea rbx, [rip + inputbuf + 65536]
    mov rdx, rbx
    sub rdx, r15
    xor eax, eax
    xor edi, edi
    mov rsi, r15
    syscall
    test rax, rax
    jz input_done
    add r15, rax
    jmp read_loop

input_done:
    mov rbp, r15
    lea rsi, [rip + inputbuf]

line_start:
    cmp rsi, rbp
    jae done
    mov r14, rsi
find_newline:
    cmp r14, rbp
    jae line_found
    cmp byte ptr [r14], 10
    je line_found
    inc r14
    jmp find_newline

line_found:
    mov rbx, rsi
    xor r12d, r12d
    xor r13d, r13d

parse_next:
skip_sep:
    cmp rbx, r14
    jae parse_done
    mov al, byte ptr [rbx]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    jmp token_start
skip_one:
    inc rbx
    jmp skip_sep

token_start:
    mov r8, rbx
scan_token:
    cmp rbx, r14
    jae token_end
    mov al, byte ptr [rbx]
    cmp al, 32
    je token_end
    cmp al, 9
    je token_end
    inc rbx
    jmp scan_token

token_end:
    mov r9, rbx
    test r13d, r13d
    jnz parse_next
    mov rcx, r9
    sub rcx, r8
    cmp rcx, 1
    jne literal

single_token:
    mov al, byte ptr [r8]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    jmp literal

literal:
    mov rdx, r8
    xor r10d, r10d
    cmp byte ptr [rdx], '-'
    jne literal_digits
    inc rdx
    mov r10d, 1
literal_digits:
    cmp rdx, r9
    jae invalid
    mov rcx, r9
    sub rcx, rdx
    cmp rcx, 18
    ja invalid
    xor eax, eax
digit_loop:
    movzx ecx, byte ptr [rdx]
    sub ecx, '0'
    cmp ecx, 9
    ja invalid
    imul rax, rax, 10
    add rax, rcx
    inc rdx
    cmp rdx, r9
    jb digit_loop
    test r10d, r10d
    jz push_value
    neg rax
push_value:
    mov qword ptr [values + r12*8], rax
    inc r12
    jmp parse_next

op_add:
    cmp r12, 2
    jb invalid
    mov rcx, qword ptr [values + r12*8 - 8]
    mov rax, qword ptr [values + r12*8 - 16]
    add rax, rcx
    mov qword ptr [values + r12*8 - 16], rax
    dec r12
    jmp parse_next

op_sub:
    cmp r12, 2
    jb invalid
    mov rcx, qword ptr [values + r12*8 - 8]
    mov rax, qword ptr [values + r12*8 - 16]
    sub rax, rcx
    mov qword ptr [values + r12*8 - 16], rax
    dec r12
    jmp parse_next

op_mul:
    cmp r12, 2
    jb invalid
    mov rcx, qword ptr [values + r12*8 - 8]
    mov rax, qword ptr [values + r12*8 - 16]
    imul rax, rcx
    mov qword ptr [values + r12*8 - 16], rax
    dec r12
    jmp parse_next

op_div:
    cmp r12, 2
    jb invalid
    mov rcx, qword ptr [values + r12*8 - 8]
    test rcx, rcx
    jz invalid
    mov rax, qword ptr [values + r12*8 - 16]
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne div_ok
    cmp rcx, -1
    je invalid
div_ok:
    cqo
    idiv rcx
    mov qword ptr [values + r12*8 - 16], rax
    dec r12
    jmp parse_next

op_mod:
    cmp r12, 2
    jb invalid
    mov rcx, qword ptr [values + r12*8 - 8]
    test rcx, rcx
    jz invalid
    mov rax, qword ptr [values + r12*8 - 16]
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne mod_ok
    cmp rcx, -1
    je invalid
mod_ok:
    cqo
    idiv rcx
    mov qword ptr [values + r12*8 - 16], rdx
    dec r12
    jmp parse_next

invalid:
    mov r13d, 1
    jmp parse_next

parse_done:
    test r13d, r13d
    jnz print_error
    cmp r12, 1
    jne print_error

    mov rax, qword ptr [values]
    lea rdi, [rip + outputbuf + 32]
    dec rdi
    mov byte ptr [rdi], 10
    xor r8d, r8d
    test rax, rax
    jns output_digits
    mov r8d, 1
    neg rax
output_digits:
    mov rcx, 10
output_digit_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jnz output_digit_loop
    test r8d, r8d
    jz write_number
    dec rdi
    mov byte ptr [rdi], '-'
write_number:
    lea rdx, [rip + outputbuf + 32]
    sub rdx, rdi
    mov rsi, rdi
    mov edi, 1
    mov eax, 1
    syscall
    jmp next_line

print_error:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + errorstr]
    mov edx, 6
    syscall

next_line:
    cmp r14, rbp
    jae line_start
    lea rsi, [r14 + 1]
    jmp line_start

done:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
errorstr:
    .ascii "error\n"

.bss
.align 8
inputbuf:
    .zero 65536
values:
    .zero 8192
outputbuf:
    .zero 32
