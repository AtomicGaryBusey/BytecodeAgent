.equ INPUT_CAP, 32769

.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    mov eax, INPUT_CAP
    sub rax, r12
    lea rsi, [rbx + r12]
    mov rdx, rax
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    lea rbp, [rip + line_lengths]
    lea rbx, [rip + line_starts]
    lea r15, [rip + input_buffer]
    add r15, r12
    lea rsi, [rip + input_buffer]
    mov r14, rsi
    xor r13d, r13d

parse_loop:
    cmp rsi, r15
    jae parse_done
    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 10
    jne parse_loop
    mov qword ptr [rbx + r13*8], r14
    mov rax, rsi
    sub rax, r14
    dec rax
    mov dword ptr [rbp + r13*4], eax
    inc r13
    mov r14, rsi
    jmp parse_loop

parse_done:
    cmp r14, r15
    jae sort_begin
    mov qword ptr [rbx + r13*8], r14
    mov rax, r15
    sub rax, r14
    mov dword ptr [rbp + r13*4], eax
    inc r13

sort_begin:
    mov r12d, 1

sort_outer:
    cmp r12, r13
    jae output_begin
    mov r14, qword ptr [rbx + r12*8]
    mov r15d, dword ptr [rbp + r12*4]
    mov r11, r12

sort_inner:
    test r11, r11
    jz sort_insert
    mov rdi, qword ptr [rbx + r11*8 - 8]
    mov esi, dword ptr [rbp + r11*4 - 4]
    mov rdx, r14
    mov ecx, r15d
    call compare_lines
    test eax, eax
    jle sort_insert
    mov rax, qword ptr [rbx + r11*8 - 8]
    mov qword ptr [rbx + r11*8], rax
    mov eax, dword ptr [rbp + r11*4 - 4]
    mov dword ptr [rbp + r11*4], eax
    dec r11
    jmp sort_inner

sort_insert:
    mov qword ptr [rbx + r11*8], r14
    mov dword ptr [rbp + r11*4], r15d
    inc r12
    jmp sort_outer

output_begin:
    xor r12d, r12d

output_loop:
    cmp r12, r13
    jae exit_ok
    mov edx, dword ptr [rbp + r12*4]
    test edx, edx
    jz output_newline
    mov rsi, qword ptr [rbx + r12*8]
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc r12
    jmp output_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

compare_lines:
    xor r8d, r8d
    mov r9d, esi
    cmp esi, ecx
    jbe compare_min_ready
    mov r9d, ecx

compare_min_ready:
    cmp r8d, r9d
    jae compare_prefix
    movzx eax, byte ptr [rdi + r8]
    movzx r10d, byte ptr [rdx + r8]
    cmp eax, r10d
    jb compare_less
    ja compare_greater
    inc r8d
    jmp compare_min_ready

compare_prefix:
    cmp esi, ecx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero INPUT_CAP
.align 8
line_starts:
    .zero 8000
.align 4
line_lengths:
    .zero 4000
