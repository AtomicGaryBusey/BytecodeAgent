.text
.globl _start
_start:
    xor r12d, r12d
read_more:
    mov eax, 32768
    sub eax, r12d
    lea rsi, [rip + input_buffer]
    add rsi, r12
    xor edi, edi
    mov edx, eax
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_more

parse_input:
    lea rbp, [rip + line_ptrs]
    lea r15, [rip + line_lens]
    lea rsi, [rip + input_buffer]
    lea rdi, [rsi + r12]
    mov rbx, rsi
    xor r13d, r13d

parse_loop:
    cmp rsi, rdi
    je parse_done
    cmp byte ptr [rsi], 10
    jne parse_next
    mov qword ptr [rbp + r13*8], rbx
    mov rax, rsi
    sub rax, rbx
    mov dword ptr [r15 + r13*4], eax
    inc r13
    inc rsi
    mov rbx, rsi
    jmp parse_loop

parse_next:
    inc rsi
    jmp parse_loop

parse_done:
    cmp rbx, rdi
    je sort_start
    mov qword ptr [rbp + r13*8], rbx
    mov rax, rdi
    sub rax, rbx
    mov dword ptr [r15 + r13*4], eax
    inc r13

sort_start:
    mov r14, 1

sort_outer:
    cmp r14, r13
    jae output_start
    mov r8, qword ptr [rbp + r14*8]
    mov r9d, dword ptr [r15 + r14*4]
    mov rbx, r14

sort_inner:
    test rbx, rbx
    jz sort_place
    mov r10, qword ptr [rbp + rbx*8 - 8]
    mov r11d, dword ptr [r15 + rbx*4 - 4]
    mov ecx, r9d
    cmp ecx, r11d
    jbe sort_min_ready
    mov ecx, r11d

sort_min_ready:
    mov rsi, r8
    mov rdi, r10
    test ecx, ecx
    jz sort_length_compare

sort_compare_loop:
    mov al, byte ptr [rsi]
    mov dl, byte ptr [rdi]
    cmp al, dl
    jb sort_shift
    ja sort_place
    inc rsi
    inc rdi
    dec ecx
    jnz sort_compare_loop

sort_length_compare:
    cmp r9d, r11d
    jb sort_shift
    jmp sort_place

sort_shift:
    mov qword ptr [rbp + rbx*8], r10
    mov dword ptr [r15 + rbx*4], r11d
    dec rbx
    jmp sort_inner

sort_place:
    mov qword ptr [rbp + rbx*8], r8
    mov dword ptr [r15 + rbx*4], r9d
    inc r14
    jmp sort_outer

output_start:
    xor r14d, r14d

output_loop:
    cmp r14, r13
    jae exit_program
    mov rdx, qword ptr [r15 + r14*4]
    test rdx, rdx
    jz output_newline
    mov rsi, qword ptr [rbp + r14*8]
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc r14
    jmp output_loop

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero 32768
.align 8
line_ptrs:
    .zero 8000
.align 4
line_lens:
    .zero 4000
