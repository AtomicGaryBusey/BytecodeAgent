.text
.globl _start

_start:
    adrp x20, input
    add x20, x20, :lo12:input
    mov x21, x20

read_loop:
    movz x3, #1, lsl #16
    add x3, x3, #1
    add x3, x20, x3
    sub x2, x3, x21
    mov x0, #0
    mov x1, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x22, x20
    mov x23, x20

scan_loop:
    cmp x23, x21
    b.hs scan_done
    ldrb w24, [x23]
    cmp w24, #10
    b.eq line_found
    add x23, x23, #1
    b scan_loop

line_found:
    mov x25, x23
    bl output_line
    add x23, x23, #1
    mov x22, x23
    b scan_loop

scan_done:
    cmp x22, x21
    b.hs exit_ok
    mov x25, x21
    bl output_line

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

output_line:
    mov x26, x22
    mov x27, x25

reverse_loop:
    cmp x26, x27
    b.hs reverse_done
    sub x27, x27, #1
    cmp x26, x27
    b.hs reverse_done
    ldrb w28, [x26]
    ldrb w29, [x27]
    strb w29, [x26]
    strb w28, [x27]
    add x26, x26, #1
    b reverse_loop

reverse_done:
    sub x2, x25, x22
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0

    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0
    ret

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input:
    .space 65537
