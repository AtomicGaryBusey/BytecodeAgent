.equ IN_SIZE, 65536

.text
.globl _start
_start:
    adrp x21, input_buf
    add x21, x21, :lo12:input_buf
    mov x20, #0

.Lread:
    cmp x20, #IN_SIZE
    b.eq .Lread_done
    movz x22, #1, lsl #16
    sub x2, x22, x20
    add x1, x21, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, .Lread_done
    add x20, x20, x0
    b .Lread

.Lread_done:
    mov x19, #0

.Lrow:
    cmp x19, x20
    b.hs .Lfinal

    sub x22, x20, x19
    mov x9, #16
    cmp x22, #16
    csel x22, x22, x9, ls

    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x23, x24

    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

    add x26, x21, x19

    mov w28, #28
.Loffset_hex:
    lsr w9, w19, w28
    and w9, w9, #15
    ldrb w10, [x25, w9, uxtw]
    strb w10, [x23], #1
    subs w28, w28, #4
    b.pl .Loffset_hex

    mov w10, #32
    strb w10, [x23], #1
    strb w10, [x23], #1

    mov x27, #0
.Lslots:
    cmp x27, x22
    b.hs .Lblank_slot

    ldrb w9, [x26, x27]
    lsr w10, w9, #4
    ldrb w10, [x25, w10, uxtw]
    strb w10, [x23], #1
    and w10, w9, #15
    ldrb w10, [x25, w10, uxtw]
    strb w10, [x23], #1
    mov w10, #32
    strb w10, [x23], #1
    b .Lslot_done

.Lblank_slot:
    mov w10, #32
    strb w10, [x23], #1
    strb w10, [x23], #1
    strb w10, [x23], #1

.Lslot_done:
    cmp x27, #7
    b.ne .Lno_extra_space
    mov w10, #32
    strb w10, [x23], #1
.Lno_extra_space:
    add x27, x27, #1
    cmp x27, #16
    b.lo .Lslots

    mov w10, #32
    strb w10, [x23], #1
    mov w10, #124
    strb w10, [x23], #1

    mov x27, #0
.Lascii:
    cmp x27, x22
    b.hs .Lascii_done
    ldrb w9, [x26, x27]
    cmp w9, #32
    b.lo .Ldot
    cmp w9, #126
    b.hi .Ldot
    b .Lput_ascii
.Ldot:
    mov w9, #46
.Lput_ascii:
    strb w9, [x23], #1
    add x27, x27, #1
    b .Lascii

.Lascii_done:
    mov w10, #124
    strb w10, [x23], #1
    mov w10, #10
    strb w10, [x23], #1

    mov x0, #1
    mov x1, x24
    sub x2, x23, x24
    mov x8, #64
    svc #0

    add x19, x19, x22
    b .Lrow

.Lfinal:
    adrp x24, output_buf
    add x24, x24, :lo12:output_buf
    mov x23, x24

    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits

    mov w28, #28
.Lfinal_hex:
    lsr w9, w20, w28
    and w9, w9, #15
    ldrb w10, [x25, w9, uxtw]
    strb w10, [x23], #1
    subs w28, w28, #4
    b.pl .Lfinal_hex

    mov w10, #10
    strb w10, [x23], #1

    mov x0, #1
    mov x1, x24
    sub x2, x23, x24
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 80
