.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0
    movz x3, #1, lsl #16

read_loop:
    cmp x21, x3
    b.eq read_done
    add x1, x20, x21
    sub x2, x3, x21
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    b read_loop

read_done:
    mov x19, x20
    add x22, x20, x21
    adrp x26, value_stack
    add x26, x26, :lo12:value_stack

line_begin:
    cmp x19, x22
    b.eq program_exit

    mov x23, x19
find_newline:
    cmp x23, x22
    b.eq final_line
    ldrb w0, [x23]
    cmp w0, #10
    b.eq found_newline
    add x23, x23, #1
    b find_newline

found_newline:
    add x27, x23, #1
    b evaluate_line

final_line:
    mov x27, x22

evaluate_line:
    mov x0, x19
    mov x1, x23
    mov x24, #0
    mov x25, #0

token_next:
    cmp x0, x1
    b.eq line_finish

skip_separators:
    cmp x0, x1
    b.eq line_finish
    ldrb w2, [x0]
    cmp w2, #32
    b.eq skip_one
    cmp w2, #9
    b.ne token_start
skip_one:
    add x0, x0, #1
    b skip_separators

token_start:
    mov x3, x0
scan_token:
    cmp x0, x1
    b.eq token_scanned
    ldrb w2, [x0]
    cmp w2, #32
    b.eq token_scanned
    cmp w2, #9
    b.eq token_scanned
    add x0, x0, #1
    b scan_token

token_scanned:
    sub x4, x0, x3
    cbnz x25, token_next

    cmp x4, #1
    b.ne literal_token
    ldrb w2, [x3]
    cmp w2, #43
    b.eq op_add
    cmp w2, #45
    b.eq op_sub
    cmp w2, #42
    b.eq op_mul
    cmp w2, #47
    b.eq op_div
    cmp w2, #37
    b.eq op_mod

literal_token:
    mov x5, x3
    mov w10, #0
    ldrb w2, [x5]
    cmp w2, #45
    b.ne literal_digits
    mov w10, #1
    add x5, x5, #1
    sub x6, x4, #1
    cbz x6, token_error
    b literal_count

literal_digits:
    mov x6, x4

literal_count:
    cmp x6, #18
    b.hi token_error
    mov x7, #0
    mov x8, #10
    add x9, x3, x4

digit_loop:
    cmp x5, x9
    b.eq digits_done
    ldrb w11, [x5], #1
    cmp w11, #48
    b.lo token_error
    cmp w11, #57
    b.hi token_error
    sub w11, w11, #48
    madd x7, x7, x8, x11
    b digit_loop

digits_done:
    cbz w10, push_literal
    neg x7, x7

push_literal:
    str x7, [x26, x24, lsl #3]
    add x24, x24, #1
    b token_next

op_add:
    cmp x24, #2
    b.lo token_error
    sub x24, x24, #2
    ldr x6, [x26, x24, lsl #3]
    add x7, x24, #1
    ldr x7, [x26, x7, lsl #3]
    add x6, x6, x7
    str x6, [x26, x24, lsl #3]
    add x24, x24, #1
    b token_next

op_sub:
    cmp x24, #2
    b.lo token_error
    sub x24, x24, #2
    ldr x6, [x26, x24, lsl #3]
    add x7, x24, #1
    ldr x7, [x26, x7, lsl #3]
    sub x6, x6, x7
    str x6, [x26, x24, lsl #3]
    add x24, x24, #1
    b token_next

op_mul:
    cmp x24, #2
    b.lo token_error
    sub x24, x24, #2
    ldr x6, [x26, x24, lsl #3]
    add x7, x24, #1
    ldr x7, [x26, x7, lsl #3]
    mul x6, x6, x7
    str x6, [x26, x24, lsl #3]
    add x24, x24, #1
    b token_next

op_div:
    cmp x24, #2
    b.lo token_error
    sub x24, x24, #2
    ldr x6, [x26, x24, lsl #3]
    add x7, x24, #1
    ldr x7, [x26, x7, lsl #3]
    cbz x7, token_error
    cmp x7, #-1
    b.ne div_do
    movz x8, #0x8000, lsl #48
    cmp x6, x8
    b.eq token_error
div_do:
    sdiv x6, x6, x7
    str x6, [x26, x24, lsl #3]
    add x24, x24, #1
    b token_next

op_mod:
    cmp x24, #2
    b.lo token_error
    sub x24, x24, #2
    ldr x6, [x26, x24, lsl #3]
    add x7, x24, #1
    ldr x7, [x26, x7, lsl #3]
    cbz x7, token_error
    cmp x7, #-1
    b.ne mod_do
    movz x8, #0x8000, lsl #48
    cmp x6, x8
    b.eq token_error
mod_do:
    sdiv x8, x6, x7
    msub x6, x8, x7, x6
    str x6, [x26, x24, lsl #3]
    add x24, x24, #1
    b token_next

token_error:
    mov x25, #1
    b token_next

line_finish:
    cbnz x25, output_error
    cmp x24, #1
    b.ne output_error
    ldr x0, [x26]
    bl emit_value
    b next_line

output_error:
    bl emit_error

next_line:
    mov x19, x27
    b line_begin

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

emit_error:
    adrp x1, error_msg
    add x1, x1, :lo12:error_msg
    mov x0, #1
    mov x2, #6
    mov x8, #64
    svc #0
    ret

emit_value:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x1, x1, #32
    sub x1, x1, #1
    mov w2, #10
    strb w2, [x1]

    mov x3, #0
    cmp x0, #0
    b.ge value_positive
    mov x3, #1
    neg x0, x0

value_positive:
    mov x4, #10

value_digit:
    udiv x5, x0, x4
    msub x6, x5, x4, x0
    add w6, w6, #48
    sub x1, x1, #1
    strb w6, [x1]
    mov x0, x5
    cbnz x0, value_digit

    cbz x3, value_write
    sub x1, x1, #1
    mov w6, #45
    strb w6, [x1]

value_write:
    adrp x7, output_buf
    add x7, x7, :lo12:output_buf
    add x7, x7, #32
    sub x2, x7, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .rodata

    .ascii "error\n"

.bss

    .zero 65536

    .zero 8192
