.globl _start

.text
_start:
    lea rsi, [rip + input_buffer]
    xor r8d, r8d

read_loop:
    cmp r8d, 65536
    je read_done
    lea rsi, [rip + input_buffer]
    add rsi, r8
    mov edx, 65536
    sub edx, r8d
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    jmp read_loop

read_done:
    lea r10, [rip + input_buffer]
    lea r13, [r10 + r8]
    mov r15, r13
    mov r14, r13
    mov rsi, r10

find_separator:
    cmp rsi, r13
    jae separator_done
    cmp byte ptr [rsi], 33
    je separator_found
    inc rsi
    jmp find_separator

separator_found:
    mov r13, rsi
    lea r14, [rsi + 1]

separator_done:
    lea rbp, [rip + bracket_map]
    lea r11, [rip + bracket_stack]
    mov rsi, r10
    xor edi, edi

validate_loop:
    cmp rsi, r13
    jae validate_done
    mov al, byte ptr [rsi]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rsi
    jmp validate_loop

validate_open:
    mov ecx, esi
    sub ecx, r10d
    mov dword ptr [r11 + rdi*4], ecx
    inc edi
    inc rsi
    jmp validate_loop

validate_close:
    test edi, edi
    jz error_exit
    dec edi
    mov ecx, dword ptr [r11 + rdi*4]
    mov eax, esi
    sub eax, r10d
    lea edx, [rcx + 1]
    mov dword ptr [rbp + rax*4], edx
    lea edx, [rax + 1]
    mov dword ptr [rbp + rcx*4], edx
    inc rsi
    jmp validate_loop

validate_done:
    test edi, edi
    jnz error_exit

    mov r12, r10
    lea rbx, [rip + tape]
    lea r8, [rip + output_buffer]
    mov r9, r8

execute_loop:
    cmp r12, r13
    jae execute_done
    mov al, byte ptr [r12]
    inc r12
    cmp al, 62
    je command_right
    cmp al, 60
    je command_left
    cmp al, 43
    je command_plus
    cmp al, 45
    je command_minus
    cmp al, 46
    je command_dot
    cmp al, 44
    je command_comma
    cmp al, 91
    je command_open
    cmp al, 93
    je command_close
    jmp execute_loop

command_right:
    inc rbx
    jmp execute_loop

command_left:
    dec rbx
    jmp execute_loop

command_plus:
    inc byte ptr [rbx]
    jmp execute_loop

command_minus:
    dec byte ptr [rbx]
    jmp execute_loop

command_dot:
    mov al, byte ptr [rbx]
    mov byte ptr [r9], al
    inc r9
    lea rax, [rip + output_buffer + 65536]
    cmp r9, rax
    jb execute_loop
    mov rdx, r9
    sub rdx, r8
    mov edi, 1
    mov rsi, r8
    mov eax, 1
    syscall
    mov r9, r8
    jmp execute_loop

command_comma:
    cmp r14, r15
    jae comma_zero
    mov al, byte ptr [r14]
    inc r14
    mov byte ptr [rbx], al
    jmp execute_loop

comma_zero:
    mov byte ptr [rbx], 0
    jmp execute_loop

command_open:
    cmp byte ptr [rbx], 0
    jne execute_loop
    mov rax, r12
    sub rax, r10
    dec rax
    mov eax, dword ptr [rbp + rax*4]
    lea r12, [r10 + rax]
    jmp execute_loop

command_close:
    cmp byte ptr [rbx], 0
    je execute_loop
    mov rax, r12
    sub rax, r10
    dec rax
    mov eax, dword ptr [rbp + rax*4]
    lea r12, [r10 + rax]
    jmp execute_loop

execute_done:
    cmp r9, r8
    je success_exit
    mov rdx, r9
    sub rdx, r8
    mov edi, 1
    mov rsi, r8
    mov eax, 1
    syscall

success_exit:
    xor edi, edi
    mov eax, 60
    syscall
    jmp success_exit

error_exit:
    mov edi, 1
    mov eax, 60
    syscall
    jmp error_exit

.bss
.align 16
input_buffer:
    .zero 65536
.align 16
bracket_map:
    .zero 262144
.align 16
bracket_stack:
    .zero 4000
.align 16
tape:
    .zero 30000
.align 16
output_buffer:
    .zero 65536
