.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz make_output

    add r14, rax
    lea rbx, [rip + buffer]
    lea rbp, [rbx + rax]

scan_byte:
    movzx edx, byte ptr [rbx]
    inc rbx
    cmp dl, 10
    jne 1f
    inc r12
1:
    cmp dl, 32
    je whitespace
    cmp dl, 9
    jb nonwhitespace
    cmp dl, 13
    jbe whitespace

nonwhitespace:
    test r15d, r15d
    jnz 2f
    inc r13
    mov r15d, 1
2:
    cmp rbx, rbp
    jne scan_byte
    jmp read_more

whitespace:
    xor r15d, r15d
    cmp rbx, rbp
    jne scan_byte
    jmp read_more

make_output:
    lea rsi, [rip + output]
    mov rdi, rsi

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    sub rsp, 32
    mov r9, rsp
    mov r8, rsp
    test rax, rax
    jnz 3f
    mov byte ptr [rdi], 48
    inc rdi
    add rsp, 32
    ret

3:
    mov ecx, 10
4:
    xor edx, edx
    div rcx
    add dl, 48
    mov byte ptr [r8], dl
    inc r8
    test rax, rax
    jnz 4b

5:
    dec r8
    mov dl, byte ptr [r8]
    mov byte ptr [rdi], dl
    inc rdi
    cmp r8, r9
    jne 5b
    add rsp, 32
    ret

.bss
.align 16
buffer:
.zero 4096
output:
.zero 64
