.equ BUF_SIZE, 65536

.text
.globl _start
_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    mov x20, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, begin_process
    mov x0, #0
    add x1, x19, x20
    mov x8, #63
    svc #0
    cbz x0, begin_process
    add x20, x20, x0
    b read_loop

begin_process:
    add x21, x19, x20
    mov x22, x19
    mov x23, x19

scan_loop:
    cmp x22, x21
    b.hs final_line
    ldrb w0, [x22]
    cmp w0, #10
    b.ne scan_next

    cmp x23, x22
    b.eq write_complete_line
    mov x24, x23
    sub x25, x22, #1

reverse_line:
    cmp x24, x25
    b.hs write_complete_line
    ldrb w3, [x24]
    ldrb w4, [x25]
    strb w4, [x24]
    strb w3, [x25]
    add x24, x24, #1
    sub x25, x25, #1
    b reverse_line

write_complete_line:
    sub x2, x22, x23
    add x2, x2, #1
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0
    add x22, x22, #1
    mov x23, x22
    b scan_loop

scan_next:
    add x22, x22, #1
    b scan_loop

final_line:
    cmp x23, x21
    b.hs exit_program
    mov x24, x23
    sub x25, x21, #1

reverse_final:
    cmp x24, x25
    b.hs write_final
    ldrb w3, [x24]
    ldrb w4, [x25]
    strb w4, [x24]
    strb w3, [x25]
    add x24, x24, #1
    sub x25, x25, #1
    b reverse_final

write_final:
    sub x2, x21, x23
    mov x0, #1
    mov x1, x23
    mov x8, #64
    svc #0

    adrp x1, newline_byte
    add x1, x1, :lo12:newline_byte
    mov w0, #10
    strb w0, [x1]
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.balign 16
buffer:
    .zero BUF_SIZE
newline_byte:
    .zero 1
