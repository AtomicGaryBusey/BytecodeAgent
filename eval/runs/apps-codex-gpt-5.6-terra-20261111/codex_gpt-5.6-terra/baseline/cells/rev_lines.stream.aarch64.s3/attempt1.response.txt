.text
.globl _start

.equ READSZ, 4096

_start:
    adrp x19, linebuf
    add x19, x19, :lo12:linebuf
    adrp x21, readbuf
    add x21, x21, :lo12:readbuf
    mov x20, #0

read_more:
    mov x0, #0
    mov x1, x21
    mov x2, #READSZ
    mov x8, #63
    svc #0
    cbz x0, end_input

    mov x22, x21
    add x23, x21, x0

process_bytes:
    cmp x22, x23
    b.hs read_more

    ldrb w5, [x22], #1
    cmp w5, #10
    b.eq found_newline

    strb w5, [x19, x20]
    add x20, x20, #1
    b process_bytes

found_newline:
    bl emit_line
    b process_bytes

end_input:
    cbz x20, exit_ok
    bl emit_line

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

emit_line:
    cbz x20, emit_newline

    mov x3, #0
    sub x4, x20, #1

reverse_loop:
    cmp x3, x4
    b.hs reverse_done

    ldrb w5, [x19, x3]
    ldrb w6, [x19, x4]
    strb w6, [x19, x3]
    strb w5, [x19, x4]
    add x3, x3, #1
    sub x4, x4, #1
    b reverse_loop

reverse_done:
emit_newline:
    mov w5, #10
    strb w5, [x19, x20]
    add x2, x20, #1
    mov x0, #1
    mov x1, x19
    mov x8, #64
    svc #0
    mov x20, #0
    ret

.bss
.balign 16
linebuf:
    .zero 65537

.balign 16
readbuf:
    .zero 4096
