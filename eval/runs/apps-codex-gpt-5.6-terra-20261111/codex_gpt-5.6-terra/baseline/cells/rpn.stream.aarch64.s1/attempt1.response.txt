.equ INPUT_CAP, 65536

.section .rodata
errmsg:
.ascii "error\n"
.equ errmsg_len, . - errmsg

.text
.globl _start
_start:
    adrp x28, inputbuf
    add x28, x28, :lo12:inputbuf
    mov x20, x28
    movz x22, #1, lsl #16

read_loop:
    cbz x22, read_done
    mov x0, #0
    mov x1, x20
    mov x2, x22
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    sub x22, x22, x0
    b read_loop

read_done:
    mov x21, x20
    mov x20, x28
    adrp x19, valstack
    add x19, x19, :lo12:valstack

line_begin:
    cmp x20, x21
    b.hs program_exit

    mov x22, x20
find_newline:
    cmp x22, x21
    b.hs line_found
    ldrb w0, [x22]
    cmp w0, #10
    b.eq line_found
    add x22, x22, #1
    b find_newline

line_found:
    mov x25, x20
    mov x24, #0

token_skip:
    cmp x25, x22
    b.hs line_finish
    ldrb w0, [x25]
    cmp w0, #32
    b.eq token_skip_one
    cmp w0, #9
    b.ne token_start
token_skip_one:
    add x25, x25, #1
    b token_skip

token_start:
    mov x26, x25
token_find_end:
    cmp x25, x22
    b.hs token_end
    ldrb w0, [x25]
    cmp w0, #32
    b.eq token_end
    cmp w0, #9
    b.eq token_end
    add x25, x25, #1
    b token_find_end

token_end:
    mov x27, x25
    sub x0, x27, x26
    cmp x0, #1
    b.ne try_integer

    ldrb w1, [x26]
    cmp w1, #43
    b.eq do_operator
    cmp w1, #45
    b.eq do_operator
    cmp w1, #42
    b.eq do_operator
    cmp w1, #47
    b.eq do_operator
    cmp w1, #37
    b.eq do_operator

try_integer:
    mov x2, x26
    mov w7, #0
    ldrb w1, [x2]
    cmp w1, #45
    b.ne integer_digits_begin
    mov w7, #1
    add x2, x2, #1
    cmp x2, x27
    b.hs line_error

integer_digits_begin:
    mov x3, #0
    mov x4, #0
    mov x10, #10
integer_digit_loop:
    cmp x2, x27
    b.hs integer_done
    ldrb w5, [x2], #1
    sub w6, w5, #48
    cmp w6, #9
    b.hi line_error
    add x4, x4, #1
    cmp x4, #18
    b.hi line_error
    madd x3, x3, x10, x6
    b integer_digit_loop

integer_done:
    cbz w7, integer_push
    neg x3, x3
integer_push:
    str x3, [x19, x24, lsl #3]
    add x24, x24, #1
    b token_skip

do_operator:
    cmp x24, #2
    b.lo line_error
    sub x24, x24, #1
    ldr x5, [x19, x24, lsl #3]
    sub x24, x24, #1
    ldr x4, [x19, x24, lsl #3]

    cmp w1, #43
    b.eq op_add
    cmp w1, #45
    b.eq op_sub
    cmp w1, #42
    b.eq op_mul
    cmp w1, #47
    b.eq op_div
    b op_mod

op_add:
    add x3, x4, x5
    b op_push
op_sub:
    sub x3, x4, x5
    b op_push
op_mul:
    mul x3, x4, x5
    b op_push

op_div:
    cbz x5, line_error
    movz x6, #0x8000, lsl #48
    cmp x4, x6
    b.ne op_div_ok
    cmn x5, #1
    b.eq line_error
op_div_ok:
    sdiv x3, x4, x5
    b op_push

op_mod:
    cbz x5, line_error
    movz x6, #0x8000, lsl #48
    cmp x4, x6
    b.ne op_mod_ok
    cmn x5, #1
    b.eq line_error
op_mod_ok:
    sdiv x6, x4, x5
    msub x3, x6, x5, x4

op_push:
    str x3, [x19, x24, lsl #3]
    add x24, x24, #1
    b token_skip

line_finish:
    cmp x24, #1
    b.ne line_error
    ldr x0, [x19]
    bl print_number
    b line_advance

line_error:
    bl print_error

line_advance:
    cmp x22, x21
    b.eq line_advance_end
    add x20, x22, #1
    b line_begin
line_advance_end:
    mov x20, x22
    b line_begin

print_error:
    mov x0, #1
    adrp x1, errmsg
    add x1, x1, :lo12:errmsg
    mov x2, #errmsg_len
    mov x8, #64
    svc #0
    ret

print_number:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    add x9, x1, #32
    sub x1, x9, #1
    mov w4, #10
    strb w4, [x1]

    cmp x0, #0
    b.ge print_positive
    neg x2, x0
    mov w5, #1
    b print_digits_begin
print_positive:
    mov x2, x0
    mov w5, #0

print_digits_begin:
    mov x3, #10
print_digit_loop:
    udiv x6, x2, x3
    msub x7, x6, x3, x2
    add w7, w7, #48
    sub x1, x1, #1
    strb w7, [x1]
    mov x2, x6
    cbnz x2, print_digit_loop

    cbz w5, print_write
    sub x1, x1, #1
    mov w7, #45
    strb w7, [x1]

print_write:
    sub x2, x9, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.align 3
inputbuf:
.zero 65536
.align 3
valstack:
.zero 8192
.align 3
outbuf:
.zero 32
