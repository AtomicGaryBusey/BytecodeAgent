.text
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ MAX_INPUT, 65536

.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    add x1, x20, x19
    movz x2, #1, lsl #16
    cmp x19, x2
    b.eq read_full
    sub x2, x2, x19
    b read_call
read_full:
    mov x2, #1
read_call:
    mov x0, #0
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x25, hex_digits
    add x25, x25, :lo12:hex_digits
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf
    mov x22, #0

row_loop:
    cmp x22, x19
    b.hs final_offset

    sub x23, x19, x22
    mov x24, #16
    cmp x23, x24
    csel x23, x23, x24, lo

    mov x21, x26
    mov x0, x22
    mov x1, #28
offset_hex_loop:
    lsr x2, x0, x1
    and x2, x2, #15
    ldrb w3, [x25, x2]
    strb w3, [x21], #1
    subs x1, x1, #4
    b.pl offset_hex_loop

    mov w3, #32
    strb w3, [x21], #1
    strb w3, [x21], #1

    mov x24, #0
slot_loop:
    cmp x24, x23
    b.hs blank_slot

    add x0, x22, x24
    ldrb w1, [x20, x0]
    lsr w2, w1, #4
    ldrb w3, [x25, x2]
    strb w3, [x21], #1
    and w2, w1, #15
    ldrb w3, [x25, x2]
    strb w3, [x21], #1
    mov w3, #32
    strb w3, [x21], #1
    b slot_done

blank_slot:
    mov w3, #32
    strb w3, [x21], #1
    strb w3, [x21], #1
    strb w3, [x21], #1

slot_done:
    cmp x24, #7
    b.ne no_group_space
    mov w3, #32
    strb w3, [x21], #1
no_group_space:
    add x24, x24, #1
    cmp x24, #16
    b.lo slot_loop

    mov w3, #32
    strb w3, [x21], #1
    mov w3, #124
    strb w3, [x21], #1

    mov x24, #0
ascii_loop:
    cmp x24, x23
    b.hs ascii_done
    add x0, x22, x24
    ldrb w1, [x20, x0]
    cmp w1, #32
    b.lo ascii_dot
    cmp w1, #126
    b.hi ascii_dot
    b ascii_store
ascii_dot:
    mov w1, #46
ascii_store:
    strb w1, [x21], #1
    add x24, x24, #1
    b ascii_loop

ascii_done:
    mov w3, #124
    strb w3, [x21], #1
    mov w3, #10
    strb w3, [x21], #1

    sub x2, x21, x26
    mov x0, #1
    mov x1, x26
    mov x8, #SYS_write
    svc #0

    add x22, x22, x23
    b row_loop

final_offset:
    mov x21, x26
    mov x0, x19
    mov x1, #28
final_hex_loop:
    lsr x2, x0, x1
    and x2, x2, #15
    ldrb w3, [x25, x2]
    strb w3, [x21], #1
    subs x1, x1, #4
    b.pl final_hex_loop

    mov w3, #10
    strb w3, [x21], #1

    sub x2, x21, x26
    mov x0, #1
    mov x1, x26
    mov x8, #SYS_write
    svc #0

    mov x0, #0
    mov x8, #SYS_exit
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65537
.balign 16
output_buf:
    .zero 128
