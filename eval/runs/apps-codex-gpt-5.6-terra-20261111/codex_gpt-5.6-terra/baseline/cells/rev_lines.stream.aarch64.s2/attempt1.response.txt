.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    cmp x20, x21
    b.ne read_space
    mov x1, x19
    mov x2, #0
    b do_read

read_space:
    add x1, x19, x20
    sub x2, x21, x20

do_read:
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x20, x20, x0
    b read_loop

input_done:
    mov x9, x19
    mov x1, x19
    add x11, x19, x20

scan_line:
    cmp x1, x11
    b.hs final_line
    ldrb w4, [x1], #1
    cmp w4, #10
    b.ne scan_line

    mov x10, x1
    sub x5, x1, #1
    mov x3, x9

reverse_complete_line:
    cmp x3, x5
    b.hs write_complete_line
    ldrb w6, [x3]
    ldrb w7, [x5]
    strb w7, [x3], #1
    strb w6, [x5], #-1
    b reverse_complete_line

write_complete_line:
    mov x0, #1
    mov x1, x9
    sub x2, x10, x9
    mov x8, #64
    svc #0
    mov x1, x10
    mov x9, x10
    b scan_line

final_line:
    cmp x9, x11
    b.hs exit_success

    mov x3, x9
    sub x5, x11, #1

reverse_final_line:
    cmp x3, x5
    b.hs write_final_line
    ldrb w6, [x3]
    ldrb w7, [x5]
    strb w7, [x3], #1
    strb w6, [x5], #-1
    b reverse_final_line

write_final_line:
    mov x0, #1
    mov x1, x9
    sub x2, x11, x9
    mov x8, #64
    svc #0

    adrp x1, newline
    add x1, x1, :lo12:newline
    mov x0, #1
    mov x2, #1
    mov x8, #64
    svc #0

exit_success:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 16
input_buf:
    .zero 65536
