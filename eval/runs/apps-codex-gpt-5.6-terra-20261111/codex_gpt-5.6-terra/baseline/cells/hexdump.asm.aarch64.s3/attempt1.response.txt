.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    adrp x27, output_buf
    add x27, x27, :lo12:output_buf
    adrp x10, hex_digits
    add x10, x10, :lo12:hex_digits

    mov x19, #0
    movz x28, #1, lsl #16

read_loop:
    cmp x19, x28
    b.eq read_full
    sub x2, x28, x19
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_full:
    sub sp, sp, #16
    mov x0, #0
    mov x1, sp
    mov x2, #1
    mov x8, #63
    svc #0
    add sp, sp, #16
    cbz x0, read_done

read_done:
    mov x21, #0

row_loop:
    cmp x21, x19
    b.eq final_offset

    sub x22, x19, x21
    mov x9, #16
    cmp x22, #16
    csel x22, x22, x9, lo

    mov x23, x27
    add x24, x20, x21

    mov w25, #28
offset_digits:
    lsr w4, w21, w25
    and w4, w4, #15
    ldrb w5, [x10, w4, uxtw]
    strb w5, [x23], #1
    subs w25, w25, #4
    b.pl offset_digits

    mov w5, #32
    strb w5, [x23], #1
    strb w5, [x23], #1

    mov x26, #0
hex_slots:
    cmp x26, x22
    b.hs hex_blank

    ldrb w4, [x24, x26]
    lsr w6, w4, #4
    ldrb w5, [x10, w6, uxtw]
    strb w5, [x23], #1
    and w6, w4, #15
    ldrb w5, [x10, w6, uxtw]
    strb w5, [x23], #1
    mov w5, #32
    strb w5, [x23], #1
    b hex_after

hex_blank:
    mov w5, #32
    strb w5, [x23], #1
    strb w5, [x23], #1
    strb w5, [x23], #1

hex_after:
    add x26, x26, #1
    cmp x26, #8
    b.ne no_extra_space
    mov w5, #32
    strb w5, [x23], #1
no_extra_space:
    cmp x26, #16
    b.lo hex_slots

    mov w5, #32
    strb w5, [x23], #1
    mov w5, #124
    strb w5, [x23], #1

    mov x26, #0
ascii_loop:
    cmp x26, x22
    b.hs ascii_done
    ldrb w4, [x24, x26]
    cmp w4, #32
    b.lo ascii_dot
    cmp w4, #126
    b.hi ascii_dot
    strb w4, [x23], #1
    b ascii_next
ascii_dot:
    mov w5, #46
    strb w5, [x23], #1
ascii_next:
    add x26, x26, #1
    b ascii_loop

ascii_done:
    mov w5, #124
    strb w5, [x23], #1
    mov w5, #10
    strb w5, [x23], #1

    mov x0, #1
    mov x1, x27
    sub x2, x23, x27
    mov x8, #64
    svc #0

    add x21, x21, x22
    b row_loop

final_offset:
    mov x23, x27
    mov w25, #28
final_digits:
    lsr w4, w21, w25
    and w4, w4, #15
    ldrb w5, [x10, w4, uxtw]
    strb w5, [x23], #1
    subs w25, w25, #4
    b.pl final_digits

    mov w5, #10
    strb w5, [x23], #1

    mov x0, #1
    mov x1, x27
    mov x2, #9
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 4
input_buf:
    .space 65536
.align 4
output_buf:
    .space 80
