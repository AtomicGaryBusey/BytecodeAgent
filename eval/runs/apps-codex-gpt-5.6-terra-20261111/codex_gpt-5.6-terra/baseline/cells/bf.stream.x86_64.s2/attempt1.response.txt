.globl _start

.equ INPUT_SIZE, 65536
.equ TAPE_SIZE, 30000
.equ STACK_SIZE, 1001

.text
_start:
    xor r12d, r12d
read_more:
    lea rsi, [rip + inputbuf]
    add rsi, r12
    mov edx, INPUT_SIZE
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r8, [rip + inputbuf]
    add r12, r8

    mov rbx, r8
    mov rbp, r12
    mov r13, r12
find_bang:
    cmp rbx, r12
    jae compact_begin
    cmp byte ptr [rbx], '!'
    je found_bang
    inc rbx
    jmp find_bang

found_bang:
    mov rbp, rbx
    lea r13, [rbx + 1]

compact_begin:
    lea rdi, [rip + codebuf]
    mov rbx, r8
compact_loop:
    cmp rbx, rbp
    jae compact_done
    mov al, byte ptr [rbx]
    inc rbx
    cmp al, '>'
    je compact_keep
    cmp al, '<'
    je compact_keep
    cmp al, '+'
    je compact_keep
    cmp al, '-'
    je compact_keep
    cmp al, '.'
    je compact_keep
    cmp al, ','
    je compact_keep
    cmp al, '['
    je compact_keep
    cmp al, ']'
    jne compact_loop
compact_keep:
    mov byte ptr [rdi], al
    inc rdi
    jmp compact_loop

compact_done:
    lea r8, [rip + codebuf]
    mov rbp, rdi
    lea r9, [rip + jumps]
    lea rdi, [rip + bracket_stack]
    xor ecx, ecx
    mov rbx, r8

check_loop:
    cmp rbx, rbp
    jae check_finish
    mov al, byte ptr [rbx]
    cmp al, '['
    je check_open
    cmp al, ']'
    je check_close
    inc rbx
    jmp check_loop

check_open:
    mov qword ptr [rdi + rcx*8], rbx
    inc rcx
    inc rbx
    jmp check_loop

check_close:
    test rcx, rcx
    jz bad_program
    dec rcx
    mov rax, qword ptr [rdi + rcx*8]
    mov rdx, rax
    sub rdx, r8
    mov qword ptr [r9 + rdx*8], rbx
    mov rdx, rbx
    sub rdx, r8
    mov qword ptr [r9 + rdx*8], rax
    inc rbx
    jmp check_loop

check_finish:
    test rcx, rcx
    jnz bad_program

    mov rbx, r8
    lea r14, [rip + tape]
    lea r10, [rip + outbuf]
    xor r15d, r15d

run_loop:
    cmp rbx, rbp
    jae run_done
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_output
    cmp al, ','
    je cmd_input
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    jmp run_loop

cmd_right:
    inc r14
    jmp run_loop

cmd_left:
    dec r14
    jmp run_loop

cmd_plus:
    inc byte ptr [r14]
    jmp run_loop

cmd_minus:
    dec byte ptr [r14]
    jmp run_loop

cmd_output:
    mov al, byte ptr [r14]
    mov byte ptr [r10 + r15], al
    inc r15
    cmp r15, INPUT_SIZE
    jb run_loop
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov edx, INPUT_SIZE
    syscall
    xor r15d, r15d
    jmp run_loop

cmd_input:
    cmp r13, r12
    jae input_zero
    mov al, byte ptr [r13]
    inc r13
    mov byte ptr [r14], al
    jmp run_loop

input_zero:
    mov byte ptr [r14], 0
    jmp run_loop

cmd_open:
    cmp byte ptr [r14], 0
    jne run_loop
    lea rax, [rbx - 1]
    sub rax, r8
    mov rbx, qword ptr [r9 + rax*8]
    inc rbx
    jmp run_loop

cmd_close:
    cmp byte ptr [r14], 0
    je run_loop
    lea rax, [rbx - 1]
    sub rax, r8
    mov rbx, qword ptr [r9 + rax*8]
    inc rbx
    jmp run_loop

run_done:
    test r15, r15
    jz good_exit
    mov eax, 1
    mov edi, 1
    mov rsi, r10
    mov rdx, r15
    syscall

good_exit:
    xor edi, edi
    mov eax, 60
    syscall

bad_program:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 16
inputbuf:
    .zero INPUT_SIZE
codebuf:
    .zero INPUT_SIZE
.align 8
jumps:
    .zero INPUT_SIZE * 8
tape:
    .zero TAPE_SIZE
outbuf:
    .zero INPUT_SIZE
.align 8
bracket_stack:
    .zero STACK_SIZE * 8
