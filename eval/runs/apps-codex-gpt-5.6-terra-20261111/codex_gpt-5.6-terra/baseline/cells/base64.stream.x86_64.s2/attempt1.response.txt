.globl _start

.text
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + inputbuf]
    add rsi, rbx
    mov edx, 49153
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inputbuf]
    lea r8, [rsi + rbx]
    lea rdi, [rip + outputbuf]
    lea rbx, [rip + alphabet]
    xor r9d, r9d

full_loop:
    lea rax, [rsi + 3]
    cmp rax, r8
    jbe encode_full
    jmp tail_check

encode_full:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx

    mov ecx, eax
    shr ecx, 18
    mov dl, byte ptr [rbx + rcx]
    mov byte ptr [rdi], dl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 1], dl

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov dl, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 2], dl

    and eax, 63
    mov dl, byte ptr [rbx + rax]
    mov byte ptr [rdi + 3], dl

    add rsi, 3
    add rdi, 4
    add r9d, 4
    cmp r9d, 76
    jne full_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp full_loop

tail_check:
    mov rax, r8
    sub rax, rsi
    test rax, rax
    jz no_tail
    cmp rax, 1
    je tail_one

tail_two:
    movzx eax, byte ptr [rsi]
    mov edx, eax
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al

    and edx, 3
    shl edx, 4
    movzx ecx, byte ptr [rsi + 1]
    mov r10d, ecx
    shr r10d, 4
    or edx, r10d
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], al

    and ecx, 15
    shl ecx, 2
    mov al, byte ptr [rbx + rcx]
    mov byte ptr [rdi + 2], al

    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi
    jmp write_output

tail_one:
    movzx eax, byte ptr [rsi]
    mov edx, eax
    shr eax, 2
    mov al, byte ptr [rbx + rax]
    mov byte ptr [rdi], al

    and edx, 3
    shl edx, 4
    mov al, byte ptr [rbx + rdx]
    mov byte ptr [rdi + 1], al

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61
    add rdi, 4
    mov byte ptr [rdi], 10
    inc rdi
    jmp write_output

no_tail:
    test r9d, r9d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + outputbuf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_program
    mov eax, 1
    mov edi, 1
    syscall

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inputbuf:
    .zero 49153
.align 16
