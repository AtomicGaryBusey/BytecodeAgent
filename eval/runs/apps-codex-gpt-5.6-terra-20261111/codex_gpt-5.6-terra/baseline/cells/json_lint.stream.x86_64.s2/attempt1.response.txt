.section .rodata
ok_msg:
.ascii "ok\n"
ok_len = . - ok_msg
err_msg:
.ascii "error at "
err_len = . - err_msg

.section .bss
.align 16
input_buf:
.zero 65536
input_end:
.align 16
out_buf:
.zero 32

.text
.globl _start
_start:
    lea r15, [rip + input_buf]
    mov r12, r15
    lea r14, [rip + input_end]
read_loop:
    cmp r12, r14
    jae read_done
    mov rax, 0
    mov rdi, 0
    mov rsi, r12
    mov rdx, r14
    sub rdx, r12
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    mov r13, r12
    mov r12, r15
    call parse_ws
    call parse_value
    call parse_ws
    cmp r12, r13
    jne fatal

valid:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    syscall
    mov rax, 60
    xor rdi, rdi
    syscall

fatal:
    mov r8, r12
    sub r8, r15
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + err_msg]
    mov rdx, err_len
    syscall

    lea rsi, [rip + out_buf + 32]
    dec rsi
    mov byte ptr [rsi], 10
    mov rax, r8
    test rax, rax
    jnz fatal_digits
    dec rsi
    mov byte ptr [rsi], '0'
    jmp fatal_write

fatal_digits:
    mov rbx, 10
fatal_digit_loop:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz fatal_digit_loop

fatal_write:
    lea rdx, [rip + out_buf + 32]
    sub rdx, rsi
    mov rax, 1
    mov rdi, 1
    syscall
    mov rax, 60
    mov rdi, 1
    syscall

parse_ws:
    cmp r12, r13
    jae parse_ws_done
parse_ws_loop:
    movzx eax, byte ptr [r12]
    cmp al, 32
    je parse_ws_take
    cmp al, 9
    je parse_ws_take
    cmp al, 10
    je parse_ws_take
    cmp al, 13
    je parse_ws_take
    ret
parse_ws_take:
    inc r12
    cmp r12, r13
    jb parse_ws_loop
parse_ws_done:
    ret

parse_value:
    cmp r12, r13
    jae fatal
    movzx eax, byte ptr [r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    je parse_number
    cmp al, '1'
    jb parse_value_other
    cmp al, '9'
    jbe parse_number
parse_value_other:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp fatal

parse_true:
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'r'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'u'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'e'
    jne fatal
    inc r12
    ret

parse_false:
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'a'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'l'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 's'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'e'
    jne fatal
    inc r12
    ret

parse_null:
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'u'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'l'
    jne fatal
    inc r12
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], 'l'
    jne fatal
    inc r12
    ret

parse_number:
    cmp byte ptr [r12], '-'
    jne number_int
    inc r12
    cmp r12, r13
    jae fatal
number_int:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb fatal
    cmp al, '9'
    ja fatal
number_digits:
    inc r12
    cmp r12, r13
    jae number_frac
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_frac
    cmp al, '9'
    ja number_frac
    jmp number_digits

number_zero:
    inc r12
number_frac:
    cmp r12, r13
    jae number_done
    cmp byte ptr [r12], '.'
    jne number_exp
    inc r12
    cmp r12, r13
    jae fatal
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb fatal
    cmp al, '9'
    ja fatal
number_frac_digits:
    inc r12
    cmp r12, r13
    jae number_exp
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_exp
    cmp al, '9'
    ja number_exp
    jmp number_frac_digits

number_exp:
    cmp r12, r13
    jae number_done
    movzx eax, byte ptr [r12]
    cmp al, 'e'
    je number_exp_take
    cmp al, 'E'
    jne number_done
number_exp_take:
    inc r12
    cmp r12, r13
    jae fatal
    movzx eax, byte ptr [r12]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    jne number_exp_first
number_exp_sign:
    inc r12
    cmp r12, r13
    jae fatal
number_exp_first:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb fatal
    cmp al, '9'
    ja fatal
number_exp_digits:
    inc r12
    cmp r12, r13
    jae number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    ja number_done
    jmp number_exp_digits
number_done:
    ret

parse_string:
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], '"'
    jne fatal
    inc r12
string_loop:
    cmp r12, r13
    jae fatal
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_close
    cmp al, '\'
    je string_escape
    cmp al, 32
    jb fatal
    inc r12
    jmp string_loop

string_close:
    inc r12
    ret

string_escape:
    inc r12
    cmp r12, r13
    jae fatal
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_escape_ok
    cmp al, '\'
    je string_escape_ok
    cmp al, '/'
    je string_escape_ok
    cmp al, 'b'
    je string_escape_ok
    cmp al, 'f'
    je string_escape_ok
    cmp al, 'n'
    je string_escape_ok
    cmp al, 'r'
    je string_escape_ok
    cmp al, 't'
    je string_escape_ok
    cmp al, 'u'
    je string_unicode
    jmp fatal
string_escape_ok:
    inc r12
    jmp string_loop

string_unicode:
    inc r12
    mov ecx, 4
string_hex_loop:
    cmp r12, r13
    jae fatal
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb string_hex_alpha
    cmp al, '9'
    jbe string_hex_ok
string_hex_alpha:
    cmp al, 'a'
    jb string_hex_upper
    cmp al, 'f'
    jbe string_hex_ok
string_hex_upper:
    cmp al, 'A'
    jb fatal
    cmp al, 'F'
    ja fatal
string_hex_ok:
    inc r12
    dec ecx
    jnz string_hex_loop
    jmp string_loop

parse_array:
    inc r12
    call parse_ws
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], ']'
    je array_empty
array_value:
    call parse_value
    call parse_ws
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], ']'
    je array_end
    cmp byte ptr [r12], ','
    jne fatal
    inc r12
    call parse_ws
    jmp array_value
array_empty:
    inc r12
    ret
array_end:
    inc r12
    ret

parse_object:
    inc r12
    call parse_ws
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], '}'
    je object_empty
object_member:
    call parse_string
    call parse_ws
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], ':'
    jne fatal
    inc r12
    call parse_ws
    call parse_value
    call parse_ws
    cmp r12, r13
    jae fatal
    cmp byte ptr [r12], '}'
    je object_end
    cmp byte ptr [r12], ','
    jne fatal
    inc r12
    call parse_ws
    jmp object_member
object_empty:
    inc r12
    ret
object_end:
    inc r12
    ret
