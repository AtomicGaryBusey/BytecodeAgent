.text
.globl _start
_start:
    lea rbx, [rip + input_buffer]
    xor r12d, r12d

read_more:
    cmp r12, 32768
    jae read_done
    lea rsi, [rbx + r12]
    mov edx, 32768
    sub rdx, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_more

read_done:
    lea r13, [rip + records]
    xor r14d, r14d
    xor r15d, r15d
    xor ecx, ecx

parse_loop:
    cmp r14, r12
    jae parse_end
    mov al, byte ptr [rbx + r14]
    cmp al, 10
    jne parse_next
    mov dword ptr [r13 + rcx*8], r15d
    mov eax, r14d
    sub eax, r15d
    mov dword ptr [r13 + rcx*8 + 4], eax
    inc ecx
    lea r15, [r14 + 1]
parse_next:
    inc r14
    jmp parse_loop

parse_end:
    cmp r15, r12
    jae sort_begin
    mov dword ptr [r13 + rcx*8], r15d
    mov eax, r12d
    sub eax, r15d
    mov dword ptr [r13 + rcx*8 + 4], eax
    inc ecx

sort_begin:
    mov r12d, ecx
    lea r14, [rip + input_buffer]
    cmp r12d, 1
    jbe output_begin
    mov r15d, 1

sort_outer:
    cmp r15d, r12d
    jae output_begin
    mov r8d, dword ptr [r13 + r15*8]
    mov r9d, dword ptr [r13 + r15*8 + 4]
    mov r10d, r15d

sort_inner:
    test r10d, r10d
    jz sort_insert
    mov esi, dword ptr [r13 + r10*8 - 8]
    mov edi, dword ptr [r13 + r10*8 - 4]
    lea rsi, [r14 + rsi]
    lea rdx, [r14 + r8]
    xor eax, eax

compare_loop:
    cmp eax, edi
    jae sort_insert
    cmp eax, r9d
    jae sort_shift
    movzx ecx, byte ptr [rsi + rax]
    cmp cl, byte ptr [rdx + rax]
    ja sort_shift
    jb sort_insert
    inc eax
    jmp compare_loop

sort_shift:
    mov rax, qword ptr [r13 + r10*8 - 8]
    mov qword ptr [r13 + r10*8], rax
    dec r10d
    jmp sort_inner

sort_insert:
    mov dword ptr [r13 + r10*8], r8d
    mov dword ptr [r13 + r10*8 + 4], r9d
    inc r15d
    jmp sort_outer

output_begin:
    xor r15d, r15d

output_loop:
    cmp r15d, r12d
    jae done
    mov eax, dword ptr [r13 + r15*8]
    lea rsi, [r14 + rax]
    mov edx, dword ptr [r13 + r15*8 + 4]
    test edx, edx
    jz output_newline
    mov eax, 1
    mov edi, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    mov edi, 1
    syscall
    inc r15d
    jmp output_loop

done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 8
input_buffer:
    .zero 32768
.align 8
records:
    .zero 8000
