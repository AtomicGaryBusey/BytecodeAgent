.text
.globl _start

_start:
    adrp x19, inbuf
    add x19, x19, :lo12:inbuf
    mov x20, x19
    movz x22, #1, lsl #16
    add x22, x22, x19

read_loop:
    cmp x20, x22
    b.eq read_done
    mov x0, #0
    mov x1, x20
    sub x2, x22, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x24, x20
    mov x21, x19

find_bang:
    cmp x21, x24
    b.hs no_bang
    ldrb w0, [x21]
    cmp w0, #33
    b.eq got_bang
    add x21, x21, #1
    b find_bang

got_bang:
    mov x22, x21
    add x23, x21, #1
    b setup_validate

no_bang:
    mov x22, x24
    mov x23, x24

setup_validate:
    adrp x27, jumps
    add x27, x27, :lo12:jumps
    adrp x2, bracket_stack
    add x2, x2, :lo12:bracket_stack
    mov x0, x19
    mov w1, #0

validate_loop:
    cmp x0, x22
    b.hs validate_done
    ldrb w4, [x0], #1
    cmp w4, #91
    b.eq validate_open
    cmp w4, #93
    b.eq validate_close
    b validate_loop

validate_open:
    sub x3, x0, x19
    sub x3, x3, #1
    str w3, [x2, w1, uxtw #2]
    add w1, w1, #1
    b validate_loop

validate_close:
    cbz w1, bracket_error
    sub w1, w1, #1
    ldr w3, [x2, w1, uxtw #2]
    sub x4, x0, x19
    sub x4, x4, #1
    str w4, [x27, x3, uxtw #2]
    str w3, [x27, x4, uxtw #2]
    b validate_loop

validate_done:
    cbnz w1, bracket_error

    adrp x20, tape
    add x20, x20, :lo12:tape
    mov x21, x19
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    mov x26, x25
    movz x0, #1, lsl #16
    add x26, x26, x0

exec_loop:
    cmp x21, x22
    b.hs exec_done
    ldrb w0, [x21], #1
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_plus
    cmp w0, #45
    b.eq cmd_minus
    cmp w0, #46
    b.eq cmd_dot
    cmp w0, #44
    b.eq cmd_comma
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close
    b exec_loop

cmd_right:
    add x20, x20, #1
    b exec_loop

cmd_left:
    sub x20, x20, #1
    b exec_loop

cmd_plus:
    ldrb w1, [x20]
    add w1, w1, #1
    strb w1, [x20]
    b exec_loop

cmd_minus:
    ldrb w1, [x20]
    sub w1, w1, #1
    strb w1, [x20]
    b exec_loop

cmd_dot:
    ldrb w1, [x20]
    strb w1, [x25], #1
    cmp x25, x26
    b.eq flush_full
    b exec_loop

cmd_comma:
    cmp x23, x24
    b.hs comma_zero
    ldrb w1, [x23], #1
    strb w1, [x20]
    b exec_loop

comma_zero:
    strb wzr, [x20]
    b exec_loop

cmd_open:
    ldrb w1, [x20]
    cbnz w1, exec_loop
    sub x1, x21, x19
    sub x1, x1, #1
    ldr w1, [x27, x1, uxtw #2]
    add x21, x19, w1, uxtw
    add x21, x21, #1
    b exec_loop

cmd_close:
    ldrb w1, [x20]
    cbz w1, exec_loop
    sub x1, x21, x19
    sub x1, x1, #1
    ldr w1, [x27, x1, uxtw #2]
    add x21, x19, w1, uxtw
    add x21, x21, #1
    b exec_loop

flush_full:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x25, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x25, x1
    b exec_loop

exec_done:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    cmp x25, x1
    b.eq exit_ok
    sub x2, x25, x1
    mov x0, #1
    mov x8, #64
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

bracket_error:
    mov x0, #1
    mov x8, #93
    svc #0

.bss
.balign 16
inbuf:
    .zero 65536
.balign 16
jumps:
    .zero 262144
.balign 16
bracket_stack:
    .zero 4000
.balign 16
tape:
    .zero 30000
.balign 16
outbuf:
    .zero 65536
