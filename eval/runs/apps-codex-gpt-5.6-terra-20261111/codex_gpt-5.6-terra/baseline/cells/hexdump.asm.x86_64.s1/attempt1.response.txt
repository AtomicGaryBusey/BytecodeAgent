.text
.globl _start
_start:
    xor r12d, r12d
read_loop:
    cmp r12d, 65536
    jae read_done
    lea rsi, [rip + input_buf]
    add rsi, r12
    mov edx, 65536
    sub edx, r12d
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    mov r14, r12
    xor r12d, r12d
    lea r13, [rip + input_buf]

row_loop:
    cmp r12, r14
    jae final_offset

    mov r15, r14
    sub r15, r12
    cmp r15, 16
    jbe row_count_ready
    mov r15, 16
row_count_ready:
    lea rbx, [rip + line_buf]

    mov eax, r12d
    mov rdi, rbx
    call hex8
    mov rbx, rdi

    mov byte ptr [rbx], 32
    inc rbx
    mov byte ptr [rbx], 32
    inc rbx

    xor r10d, r10d
hex_slots:
    cmp r10, r15
    jae hex_blank
    movzx eax, byte ptr [r13 + r10]
    mov rdi, rbx
    call hex2
    mov rbx, rdi
    mov byte ptr [rbx], 32
    inc rbx
    jmp hex_after
hex_blank:
    mov byte ptr [rbx], 32
    mov byte ptr [rbx + 1], 32
    mov byte ptr [rbx + 2], 32
    add rbx, 3
hex_after:
    cmp r10, 7
    jne no_middle_space
    mov byte ptr [rbx], 32
    inc rbx
no_middle_space:
    inc r10
    cmp r10, 16
    jb hex_slots

    mov byte ptr [rbx], 32
    inc rbx
    mov byte ptr [rbx], 124
    inc rbx

    xor r10d, r10d
ascii_loop:
    cmp r10, r15
    jae ascii_done
    mov dl, byte ptr [r13 + r10]
    cmp dl, 32
    jb ascii_dot
    cmp dl, 126
    ja ascii_dot
    mov byte ptr [rbx], dl
    jmp ascii_next
ascii_dot:
    mov byte ptr [rbx], 46
ascii_next:
    inc rbx
    inc r10
    jmp ascii_loop

ascii_done:
    mov byte ptr [rbx], 124
    inc rbx
    mov byte ptr [rbx], 10
    inc rbx

    lea rsi, [rip + line_buf]
    mov rdx, rbx
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r13, r15
    add r12, r15
    jmp row_loop

final_offset:
    lea rdi, [rip + line_buf]
    mov eax, r14d
    call hex8
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

hex8:
    mov r8d, eax
    mov ecx, 28
hex8_loop:
    mov edx, r8d
    shr edx, cl
    and edx, 15
    mov dl, byte ptr [rip + hex_digits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    sub ecx, 4
    jns hex8_loop
    ret

hex2:
    mov edx, eax
    shr edx, 4
    and edx, 15
    mov dl, byte ptr [rip + hex_digits + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    mov al, byte ptr [rip + hex_digits + rax]
    mov byte ptr [rdi], al
    inc rdi
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buf:
    .zero 65536
.align 16
line_buf:
    .zero 80
