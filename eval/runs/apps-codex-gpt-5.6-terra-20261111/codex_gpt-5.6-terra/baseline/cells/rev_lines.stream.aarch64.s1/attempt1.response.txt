.text
.globl _start
_start:
    adrp x20, buffer
    add x20, x20, :lo12:buffer
    mov x21, xzr

read_loop:
    add x1, x20, x21
    movz x2, #1, lsl #16
    cmp x21, x2
    b.ne read_space
    mov x2, #1
    b read_call
read_space:
    sub x2, x2, x21
read_call:
    mov x0, xzr
    mov x8, #63
    svc #0
    cbz x0, input_done
    add x21, x21, x0
    b read_loop

input_done:
    mov x22, x20
    add x23, x20, x21

line_outer:
    cmp x22, x23
    b.eq finish
    mov x24, x22

find_newline:
    cmp x22, x23
    b.eq final_line
    ldrb w3, [x22], #1
    cmp w3, #10
    b.ne find_newline

    mov x25, x24
    sub x26, x22, #2
reverse_terminated:
    cmp x25, x26
    b.hs write_terminated
    ldrb w3, [x25]
    ldrb w4, [x26]
    strb w4, [x25]
    strb w3, [x26]
    add x25, x25, #1
    sub x26, x26, #1
    b reverse_terminated

write_terminated:
    sub x2, x22, x24
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0
    b line_outer

final_line:
    mov x25, x24
    sub x26, x23, #1
reverse_final:
    cmp x25, x26
    b.hs write_final
    ldrb w3, [x25]
    ldrb w4, [x26]
    strb w4, [x25]
    strb w3, [x26]
    add x25, x25, #1
    sub x26, x26, #1
    b reverse_final

write_final:
    mov w3, #10
    strb w3, [x23]
    sub x2, x23, x24
    add x2, x2, #1
    mov x0, #1
    mov x1, x24
    mov x8, #64
    svc #0

finish:
    mov x0, xzr
    mov x8, #93
    svc #0

.bss
.align 4
buffer:
    .zero 65537
