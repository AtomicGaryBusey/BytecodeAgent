.equ IN_SIZE, 49152

.text
.globl _start
_start:
    lea r13, [rip + input_buffer]
    mov rsi, r13
    xor r12d, r12d

read_loop:
    mov eax, IN_SIZE
    sub eax, r12d
    jz encode_begin
    mov edx, eax
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz encode_begin
    add r12, rax
    lea rsi, [r13 + r12]
    jmp read_loop

encode_begin:
    mov r8, r12
    mov rsi, r13
    lea rdi, [rip + output_buffer]
    lea r10, [rip + alphabet]
    xor r9d, r9d

encode_loop:
    cmp r8, 3
    jb encode_tail

    movzx ebx, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]
    movzx edx, byte ptr [rsi + 2]

    mov eax, ebx
    shr eax, 2
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov ebp, ecx
    shr ebp, 4
    or eax, ebp
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov eax, ecx
    and eax, 15
    shl eax, 2
    mov ebp, edx
    shr ebp, 6
    or eax, ebp
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov eax, edx
    and eax, 63
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    add rsi, 3
    sub r8, 3
    jmp quad_complete

encode_tail:
    test r8, r8
    jz encode_done
    cmp r8, 1
    je tail_one

    movzx ebx, byte ptr [rsi]
    movzx ecx, byte ptr [rsi + 1]

    mov eax, ebx
    shr eax, 2
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov ebp, ecx
    shr ebp, 4
    or eax, ebp
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov eax, ecx
    and eax, 15
    shl eax, 2
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov byte ptr [rdi], 61
    inc rdi
    xor r8d, r8d
    jmp quad_complete

tail_one:
    movzx ebx, byte ptr [rsi]

    mov eax, ebx
    shr eax, 2
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov eax, ebx
    and eax, 3
    shl eax, 4
    mov al, byte ptr [r10 + rax]
    mov byte ptr [rdi], al
    inc rdi

    mov byte ptr [rdi], 61
    inc rdi
    mov byte ptr [rdi], 61
    inc rdi
    xor r8d, r8d

quad_complete:
    add r9d, 4
    cmp r9d, 76
    jne encode_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r9d, r9d
    jmp encode_loop

encode_done:
    test r9d, r9d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    jz exit_program
    mov edi, 1
    mov eax, 1
    syscall

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero 49152
.align 16
output_buffer:
    .zero 66400
