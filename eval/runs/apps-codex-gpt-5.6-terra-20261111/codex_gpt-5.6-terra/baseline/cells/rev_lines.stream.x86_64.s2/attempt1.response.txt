.text
.globl _start
_start:
    xor ebx, ebx
read_more:
    cmp rbx, 65536
    je process_input
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz process_input
    add rbx, rax
    jmp read_more

process_input:
    lea r12, [rip + inbuf]
    lea r13, [r12 + rbx]
    lea r14, [rip + outbuf]
    mov r15, r12

scan_line:
    cmp r12, r13
    jae final_line
    mov al, byte ptr [r12]
    inc r12
    cmp al, 10
    jne scan_line

    mov rsi, r12
    dec rsi
reverse_line:
    cmp rsi, r15
    jbe finish_line
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc r14
    jmp reverse_line

finish_line:
    mov byte ptr [r14], 10
    inc r14
    mov r15, r12
    jmp scan_line

final_line:
    cmp r15, r13
    je write_output
    mov rsi, r13
reverse_final:
    cmp rsi, r15
    jbe finish_final
    dec rsi
    mov al, byte ptr [rsi]
    mov byte ptr [r14], al
    inc r14
    jmp reverse_final

finish_final:
    mov byte ptr [r14], 10
    inc r14

write_output:
    lea rsi, [rip + outbuf]
    mov rdx, r14
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

.bss
.align 16
inbuf:
    .zero 65536
outbuf:
    .zero 65537
