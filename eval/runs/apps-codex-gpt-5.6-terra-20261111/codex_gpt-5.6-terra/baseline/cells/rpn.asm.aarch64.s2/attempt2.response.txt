.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    movz x9, #1, lsl #16
    add x27, x19, x9

read_loop:
    sub x2, x27, x20
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x21, value_stack
    add x21, x21, :lo12:value_stack
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf

new_line:
    cmp x19, x20
    b.eq done
    mov x22, #0
    mov w23, #0

line_loop:
    cmp x19, x20
    b.eq finish_line
    cbnz w23, discard_line

    ldrb w9, [x19]
    cmp w9, #10
    b.eq got_newline
    cmp w9, #32
    b.eq skip_byte
    cmp w9, #9
    b.eq skip_byte

    mov x24, x19
scan_token:
    cmp x19, x20
    b.eq end_token
    ldrb w9, [x19]
    cmp w9, #10
    b.eq end_token
    cmp w9, #32
    b.eq end_token
    cmp w9, #9
    b.eq end_token
    add x19, x19, #1
    b scan_token

end_token:
    mov x25, x19
    b parse_token

skip_byte:
    add x19, x19, #1
    b line_loop

got_newline:
    add x19, x19, #1
    b finish_line

discard_line:
    cmp x19, x20
    b.eq finish_line
    ldrb w9, [x19], #1
    cmp w9, #10
    b.eq finish_line
    b discard_line

parse_token:
    sub x9, x25, x24
    cmp x9, #1
    b.ne parse_number

    ldrb w10, [x24]
    cmp w10, #43
    b.eq op_add
    cmp w10, #45
    b.eq op_sub
    cmp w10, #42
    b.eq op_mul
    cmp w10, #47
    b.eq op_div
    cmp w10, #37
    b.eq op_mod

parse_number:
    mov x10, x24
    mov w12, #0
    ldrb w11, [x10]
    cmp w11, #45
    b.ne number_digits
    mov w12, #1
    add x10, x10, #1

number_digits:
    cmp x10, x25
    b.eq bad_token
    mov x13, #0
    mov x14, #0
    mov x15, #10

number_loop:
    cmp x10, x25
    b.eq number_done
    ldrb w11, [x10], #1
    cmp w11, #48
    b.lo bad_token
    cmp w11, #57
    b.hi bad_token
    sub w11, w11, #48
    madd x14, x14, x15, x11
    add x13, x13, #1
    cmp x13, #18
    b.hi bad_token
    b number_loop

number_done:
    cbz w12, push_number
    neg x14, x14

push_number:
    str x14, [x21, x22, lsl #3]
    add x22, x22, #1
    b line_loop

op_add:
    cmp x22, #2
    b.lo bad_token
    sub x9, x22, #1
    ldr x10, [x21, x9, lsl #3]
    sub x9, x9, #1
    ldr x11, [x21, x9, lsl #3]
    add x12, x11, x10
    b store_result

op_sub:
    cmp x22, #2
    b.lo bad_token
    sub x9, x22, #1
    ldr x10, [x21, x9, lsl #3]
    sub x9, x9, #1
    ldr x11, [x21, x9, lsl #3]
    sub x12, x11, x10
    b store_result

op_mul:
    cmp x22, #2
    b.lo bad_token
    sub x9, x22, #1
    ldr x10, [x21, x9, lsl #3]
    sub x9, x9, #1
    ldr x11, [x21, x9, lsl #3]
    mul x12, x11, x10
    b store_result

op_div:
    cmp x22, #2
    b.lo bad_token
    sub x9, x22, #1
    ldr x10, [x21, x9, lsl #3]
    cbz x10, bad_token
    sub x9, x9, #1
    ldr x11, [x21, x9, lsl #3]
    cmn x10, #1
    b.ne do_div
    movz x13, #0x8000, lsl #48
    cmp x11, x13
    b.eq bad_token
do_div:
    sdiv x12, x11, x10
    b store_result

op_mod:
    cmp x22, #2
    b.lo bad_token
    sub x9, x22, #1
    ldr x10, [x21, x9, lsl #3]
    cbz x10, bad_token
    sub x9, x9, #1
    ldr x11, [x21, x9, lsl #3]
    cmn x10, #1
    b.ne do_mod
    movz x13, #0x8000, lsl #48
    cmp x11, x13
    b.eq bad_token
do_mod:
    sdiv x12, x11, x10
    msub x12, x12, x10, x11

store_result:
    str x12, [x21, x9, lsl #3]
    add x22, x9, #1
    b line_loop

bad_token:
    mov w23, #1
    b line_loop

finish_line:
    cbnz w23, print_error
    cmp x22, #1
    b.ne print_error
    ldr x0, [x21]
    bl print_number
    b new_line

print_error:
    bl print_error_line
    b new_line

done:
    mov x0, #0
    mov x8, #93
    svc #0

print_error_line:
    mov w9, #101
    strb w9, [x26]
    mov w9, #114
    strb w9, [x26, #1]
    strb w9, [x26, #2]
    mov w9, #111
    strb w9, [x26, #3]
    mov w9, #114
    strb w9, [x26, #4]
    mov w9, #10
    strb w9, [x26, #5]
    mov x0, #1
    mov x1, x26
    mov x2, #6
    mov x8, #64
    svc #0
    ret

print_number:
    mov x16, x26
    mov x17, x26
    tbnz x0, #63, number_negative
    b number_magnitude

number_negative:
    mov w9, #45
    strb w9, [x17], #1
    neg x0, x0

number_magnitude:
    cbnz x0, number_convert
    mov w9, #48
    strb w9, [x17], #1
    b number_end

number_convert:
    add x10, x26, #32
    mov x11, #10

convert_loop:
    udiv x12, x0, x11
    msub x13, x12, x11, x0
    add w13, w13, #48
    sub x10, x10, #1
    strb w13, [x10]
    mov x0, x12
    cbnz x0, convert_loop

copy_digits:
    ldrb w9, [x10], #1
    strb w9, [x17], #1
    cmp x10, x26
    b.ne copy_digits

number_end:
    mov w9, #10
    strb w9, [x17], #1
    sub x2, x17, x16
    mov x0, #1
    mov x1, x16
    mov x8, #64
    svc #0
    ret
