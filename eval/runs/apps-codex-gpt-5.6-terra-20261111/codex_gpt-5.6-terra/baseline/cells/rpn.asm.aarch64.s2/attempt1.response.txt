    mov x0, #0
    mov x1, x20
    mov x8, #63

    adrp x21, value_stack
    add x21, x21, :lo12:value_stack
    adrp x26, output_buf
    add x26, x26, :lo12:output_buf

    cmp x19, x20
    b.eq all_done
    mov x22, #0
    mov w23, #0

    cmp x19, x20
    b.eq line_finish
    cbnz w23, discard_line
    ldrb w9, [x19]
    cmp w9, #10
    b.eq line_newline
    cmp w9, #32
    b.eq skip_sep
    cmp w9, #9
    b.eq skip_sep

    mov x24, x19

    cmp x19, x20
    b.eq token_end
    ldrb w9, [x19]
    cmp w9, #10
    b.eq token_end
    cmp w9, #32
    b.eq token_end
    cmp w9, #9
    b.eq token_end
    add x19, x19, #1
    b token_scan

token_end:
    mov x25, x19
    b parse_token

skip_sep:
    add x19, x19, #1
    b line_loop

line_newline:
    add x19, x19, #1
    b line_finish

    cmp x19, x20
    b.eq line_finish
    ldrb w9, [x19]
    add x19, x19, #1
    cmp w9, #10
    b.eq line_finish
    b discard_line

parse_token:
    sub x9, x25, x24
    cmp x9, #1
    b.ne parse_literal
    ldrb w10, [x24]
    cmp w10, #43
    b.eq op_add
    cmp w10, #45
    b.eq op_sub
    cmp w10, #42
    b.eq op_mul
    cmp w10, #47
    b.eq op_div
    cmp w10, #37
    b.eq op_mod
    b token_bad

parse_literal:
    mov x10, x24
    mov w12, #0
    ldrb w15, [x10]
    cmp w15, #45
    b.ne literal_digits_start
    mov w12, #1
    add x10, x10, #1

    cmp x10, x25
    b.eq token_bad
    mov x13, #0
    mov x14, #0
    mov x16, #10
literal_digit_loop:
    cmp x10, x25
    b.eq literal_done
    ldrb w15, [x10], #1
    cmp w15, #48
    b.lo token_bad
    cmp w15, #57
    b.hi token_bad
    sub w15, w15, #48
    madd x14, x14, x16, x15
    add x13, x13, #1
    cmp x13, #18
    b.hi token_bad
    b literal_digit_loop

    neg x14, x14

    str x14, [x21, x22, lsl #3]
    add x22, x22, #1
    b line_loop

op_add:
    cmp x22, #2
    b.lo token_bad
    sub x9, x22, #1
    ldr x11, [x21, x9, lsl #3]
    sub x9, x9, #1
    ldr x10, [x21, x9, lsl #3]
    add x12, x10, x11
    b op_store

op_sub:
    cmp x22, #2
    b.lo token_bad
    sub x9, x22, #1
    ldr x11, [x21, x9, lsl #3]
    sub x9, x9, #1
    ldr x10, [x21, x9, lsl #3]
    sub x12, x10, x11
    b op_store

op_mul:
    cmp x22, #2
    b.lo token_bad
    sub x9, x22, #1
    ldr x11, [x21, x9, lsl #3]
    sub x9, x9, #1
    ldr x10, [x21, x9, lsl #3]
    mul x12, x10, x11
    b op_store

op_div:

    b.lo token_bad
    sub x9, x22, #1
    ldr x11, [x21, x9, lsl #3]
    cbz x11, token_bad
    sub x9, x9, #1
    ldr x10, [x21, x9, lsl #3]
    cmn x11, #1
    b.ne div_do
    movz x13, #0x8000, lsl #48
    cmp x10, x13
    b.eq token_bad

    sdiv x12, x10, x11
    b op_store

    cmp x22, #2
    b.lo token_bad
    sub x9, x22, #1
    ldr x11, [x21, x9, lsl #3]
    cbz x11, token_bad
    sub x9, x9, #1
    ldr x10, [x21, x9, lsl #3]
    cmn x11, #1
    b.ne mod_do
    movz x13, #0x8000, lsl #48
    cmp x10, x13
    b.eq token_bad

    sdiv x12, x10, x11
    msub x12, x12, x11, x10

    str x12, [x21, x9, lsl #3]
    add x22, x9, #1
    b line_loop

    b line_loop

    cmp x22, #1
    b.ne output_error_line
    ldr x0, [x21]

    add x1, x1, :lo12:output_buf
    sub x2, x26, x1
    cbz x2, exit_ok

    mov x8, #64

    mov x8, #93

    strb w0, [x26], #1
    mov w0, #114
    strb w0, [x26], #1
    strb w0, [x26], #1
    mov w0, #111
    strb w0, [x26], #1
    mov w0, #114
