.globl _start

.text
_start:
    lea r13, [rip + inputbuf]
    mov rbx, r13
read_loop:
    lea rdx, [rip + input_end]
    sub rdx, rbx
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    mov r12, rbx
    mov rsi, r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne bad
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

bad:
    mov r8, rsi
    sub r8, r13
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall
    mov rax, r8
    lea rbp, [rip + outbuf + 32]
    dec rbp
    mov byte ptr [rbp], 10
    test rax, rax
    jnz bad_digits
    dec rbp
    mov byte ptr [rbp], 48
    jmp bad_write

bad_digits:
    mov r9d, 10
bad_digit_loop:
    xor edx, edx
    div r9
    add dl, 48
    dec rbp
    mov byte ptr [rbp], dl
    test rax, rax
    jnz bad_digit_loop

bad_write:
    lea rdx, [rip + outbuf + 32]
    sub rdx, rbp
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp rsi, r12
    je skip_ws_done
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je skip_ws_one
    cmp al, 9
    je skip_ws_one
    cmp al, 10
    je skip_ws_one
    cmp al, 13
    jne skip_ws_done
skip_ws_one:
    inc rsi
    jmp skip_ws
skip_ws_done:
    ret

parse_value:
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 45
    je parse_number
    cmp al, 48
    jb parse_value_other
    cmp al, 57
    jbe parse_number
parse_value_other:
    cmp al, 116
    je parse_true
    cmp al, 102
    je parse_false
    cmp al, 110
    je parse_null
    jmp bad

parse_true:
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 114
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 117
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 101
    jne bad
    inc rsi
    ret

parse_false:
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 97
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 108
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 115
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 101
    jne bad
    inc rsi
    ret

parse_null:
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 117
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 108
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 108
    jne bad
    inc rsi
    ret

parse_string:
    inc rsi
string_loop:
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 34
    je string_end
    cmp al, 92
    je string_escape
    cmp al, 32
    jb bad
    inc rsi
    jmp string_loop

string_end:
    inc rsi
    ret

string_escape:
    inc rsi
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 34
    je string_simple
    cmp al, 92
    je string_simple
    cmp al, 47
    je string_simple
    cmp al, 98
    je string_simple
    cmp al, 102
    je string_simple
    cmp al, 110
    je string_simple
    cmp al, 114
    je string_simple
    cmp al, 116
    je string_simple
    cmp al, 117
    jne bad
    inc rsi
    mov ecx, 4
string_hex_loop:
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb string_hex_alpha
    cmp al, 57
    jbe string_hex_ok
string_hex_alpha:
    cmp al, 97
    jb string_hex_upper
    cmp al, 102
    jbe string_hex_ok
string_hex_upper:
    cmp al, 65
    jb bad
    cmp al, 70
    ja bad
string_hex_ok:
    inc rsi
    dec ecx
    jnz string_hex_loop
    jmp string_loop

string_simple:
    inc rsi
    jmp string_loop

parse_number:
    movzx eax, byte ptr [rsi]
    cmp al, 45
    jne number_first
    inc rsi
    cmp rsi, r12
    je bad
number_first:
    movzx eax, byte ptr [rsi]
    cmp al, 48
    je number_zero
    cmp al, 49
    jb bad
    cmp al, 57
    ja bad
number_integer:
    inc rsi
number_integer_digits:
    cmp rsi, r12
    je number_suffix
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb number_suffix
    cmp al, 57
    ja number_suffix
    inc rsi
    jmp number_integer_digits

number_zero:
    inc rsi
number_suffix:
    cmp rsi, r12
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, 46
    jne number_exp_check
    inc rsi
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb bad
    cmp al, 57
    ja bad
    inc rsi
number_frac_digits:
    cmp rsi, r12
    je number_exp_check
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb number_exp_check
    cmp al, 57
    ja number_exp_check
    inc rsi
    jmp number_frac_digits

number_exp_check:
    cmp rsi, r12
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, 101
    je number_exp
    cmp al, 69
    jne number_done
number_exp:
    inc rsi
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, 43
    je number_exp_sign
    cmp al, 45
    jne number_exp_digit
number_exp_sign:
    inc rsi
    cmp rsi, r12
    je bad
number_exp_digit:
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb bad
    cmp al, 57
    ja bad
    inc rsi
number_exp_digits:
    cmp rsi, r12
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, 48
    jb number_done
    cmp al, 57
    ja number_done
    inc rsi
    jmp number_exp_digits
number_done:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 93
    je array_empty
array_value:
    call parse_value
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 93
    je array_end
    cmp byte ptr [rsi], 44
    jne bad
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    jmp array_value
array_empty:
    inc rsi
    ret
array_end:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 125
    je object_empty
object_member:
    cmp byte ptr [rsi], 34
    jne bad
    call parse_string
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 58
    jne bad
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 125
    je object_end
    cmp byte ptr [rsi], 44
    jne bad
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    jmp object_member
object_empty:
    inc rsi
    ret
object_end:
    inc rsi
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "

.bss
.align 16
inputbuf:
    .zero 65537
input_end:
.align 16
outbuf:
    .zero 32
