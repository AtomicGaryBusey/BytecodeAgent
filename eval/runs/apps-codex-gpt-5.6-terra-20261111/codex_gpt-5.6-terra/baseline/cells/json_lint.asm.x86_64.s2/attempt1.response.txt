.globl _start

.text
_start:
    lea r13, [rip + inputbuf]
    mov rbx, r13
read_more:
    lea rdx, [rip + input_end]
    sub rdx, rbx
    xor eax, eax
    xor edi, edi
    mov rsi, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_more
read_done:
    mov r12, rbx
    mov rsi, r13
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    jne bad
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

bad:
    mov r8, rsi
    sub r8, r13
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall
    mov rax, r8
    lea rbp, [rip + outbuf + 32]
    dec rbp
    mov byte ptr [rbp], 10
    test rax, rax
    jnz bad_digits
    dec rbp
    mov byte ptr [rbp], '0'
    jmp bad_write
bad_digits:
    mov r9d, 10
bad_digit_loop:
    xor edx, edx
    div r9
    add dl, '0'
    dec rbp
    mov byte ptr [rbp], dl
    test rax, rax
    jnz bad_digit_loop
bad_write:
    lea rdx, [rip + outbuf + 32]
    sub rdx, rbp
    mov eax, 1
    mov edi, 1
    mov rsi, rbp
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp rsi, r12
    je skip_done
    movzx eax, byte ptr [rsi]
    cmp al, 32
    je skip_one
    cmp al, 9
    je skip_one
    cmp al, 10
    je skip_one
    cmp al, 13
    jne skip_done
skip_one:
    inc rsi
    jmp skip_ws
skip_done:
    ret

parse_value:
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, '{'
    je value_object
    cmp al, '['
    je value_array
    cmp al, '"'
    je value_string
    cmp al, '-'
    je value_number
    cmp al, '0'
    jb value_other
    cmp al, '9'
    jbe value_number
value_other:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp bad
value_object:
    jmp parse_object
value_array:
    jmp parse_array
value_string:
    jmp parse_string
value_number:
    jmp parse_number

parse_true:
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'r'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'u'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'e'
    jne bad
    inc rsi
    ret

parse_false:
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'a'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'l'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 's'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'e'
    jne bad
    inc rsi
    ret

parse_null:
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'u'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'l'
    jne bad
    inc rsi
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], 'l'
    jne bad
    inc rsi
    ret

parse_string:
    inc rsi
string_loop:
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je string_end
    cmp al, '\'
    je string_escape
    cmp al, 32
    jb bad
    inc rsi
    jmp string_loop
string_end:
    inc rsi
    ret
string_escape:
    inc rsi
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, '"'
    je string_simple
    cmp al, '\'
    je string_simple
    cmp al, '/'
    je string_simple
    cmp al, 'b'
    je string_simple
    cmp al, 'f'
    je string_simple
    cmp al, 'n'
    je string_simple
    cmp al, 'r'
    je string_simple
    cmp al, 't'
    je string_simple
    cmp al, 'u'
    jne bad
    inc rsi
    mov ecx, 4
string_hex:
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb string_hex_alpha
    cmp al, '9'
    jbe string_hex_ok
string_hex_alpha:
    cmp al, 'a'
    jb string_hex_upper
    cmp al, 'f'
    jbe string_hex_ok
string_hex_upper:
    cmp al, 'A'
    jb bad
    cmp al, 'F'
    ja bad
string_hex_ok:
    inc rsi
    dec ecx
    jnz string_hex
    jmp string_loop
string_simple:
    inc rsi
    jmp string_loop

parse_number:
    movzx eax, byte ptr [rsi]
    cmp al, '-'
    jne number_first
    inc rsi
    cmp rsi, r12
    je bad
number_first:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb bad
    cmp al, '9'
    ja bad
number_integer:
    inc rsi
number_digits:
    cmp rsi, r12
    je number_suffix
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb number_suffix
    cmp al, '9'
    ja number_suffix
    inc rsi
    jmp number_digits
number_zero:
    inc rsi
number_suffix:
    cmp rsi, r12
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, '.'
    jne number_exp_check
    inc rsi
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb bad
    cmp al, '9'
    ja bad
    inc rsi
number_frac_digits:
    cmp rsi, r12
    je number_exp_check
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb number_exp_check
    cmp al, '9'
    ja number_exp_check
    inc rsi
    jmp number_frac_digits
number_exp_check:
    cmp rsi, r12
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, 'e'
    je number_exp
    cmp al, 'E'
    jne number_done
number_exp:
    inc rsi
    cmp rsi, r12
    je bad
    movzx eax, byte ptr [rsi]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    jne number_exp_digit
number_exp_sign:
    inc rsi
    cmp rsi, r12
    je bad
number_exp_digit:
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb bad
    cmp al, '9'
    ja bad
    inc rsi
number_exp_digits:
    cmp rsi, r12
    je number_done
    movzx eax, byte ptr [rsi]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    ja number_done
    inc rsi
    jmp number_exp_digits
number_done:
    ret

parse_array:
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], ']'
    je array_empty
array_value:
    call parse_value
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], ']'
    je array_end
    cmp byte ptr [rsi], ','
    jne bad
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    jmp array_value
array_empty:
    inc rsi
    ret
array_end:
    inc rsi
    ret

parse_object:
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], '}'
    je object_empty
object_member:
    cmp byte ptr [rsi], '"'
    jne bad
    call parse_string
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], ':'
    jne bad
    inc rsi
    call skip_ws
    call parse_value
    call skip_ws
    cmp rsi, r12
    je bad
    cmp byte ptr [rsi], '}'
    je object_end
    cmp byte ptr [rsi], ','
    jne bad
    inc rsi
    call skip_ws
    cmp rsi, r12
    je bad
    jmp object_member
object_empty:
    inc rsi
    ret
object_end:
    inc rsi
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "

.bss
.align 16
inputbuf:
    .zero 65537
input_end:
.align 16
outbuf:
    .zero 32
