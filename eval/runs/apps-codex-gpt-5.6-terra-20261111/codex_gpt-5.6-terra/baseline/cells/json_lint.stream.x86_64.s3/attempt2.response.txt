.globl _start

.text
_start:
    xor r14d, r14d
read_more:
    lea rsi, [rip + input]
    add rsi, r14
    mov edx, 65536
    sub rdx, r14
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r14, rax
    jmp read_more

read_done:
    lea r12, [rip + input]
    mov r13, r12
    add r13, r14
    call skip_ws
    cmp r12, r13
    je fail
    call parse_value
    call skip_ws
    cmp r12, r13
    jne fail

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, 3
    syscall
    mov eax, 60
    xor edi, edi
    syscall

fail:
    mov rax, r12
    lea rcx, [rip + input]
    sub rax, rcx
    lea rsi, [rip + outbuf + 33]
    mov byte ptr [rsi - 1], 10
    dec rsi
    test rax, rax
    jnz fail_digits
    dec rsi
    mov byte ptr [rsi], '0'
    jmp fail_print

fail_digits:
    mov ecx, 10
fail_digit_loop:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz fail_digit_loop

fail_print:
    mov rbx, rsi
    lea rbp, [rip + outbuf + 33]
    sub rbp, rbx
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, 9
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, rbp
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
    cmp r12, r13
    je skip_ws_done
skip_ws_loop:
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_ws_take
    cmp al, 9
    je skip_ws_take
    cmp al, 10
    je skip_ws_take
    cmp al, 13
    jne skip_ws_done
skip_ws_take:
    inc r12
    cmp r12, r13
    jne skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
    jmp parse_number

parse_true:
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 't'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'r'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'u'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'e'
    jne fail
    inc r12
    ret

parse_false:
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'f'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'a'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'l'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 's'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'e'
    jne fail
    inc r12
    ret

parse_null:
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'n'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'u'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'l'
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 'l'
    jne fail
    inc r12
    ret

parse_number:
    cmp byte ptr [r12], '-'
    jne number_int
    inc r12
    cmp r12, r13
    je fail
number_int:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb fail
    cmp al, '9'
    ja fail
number_digits:
    inc r12
    cmp r12, r13
    je number_after_int
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_after_int
    cmp al, '9'
    ja number_after_int
    jmp number_digits

number_zero:
    inc r12
number_after_int:
    cmp r12, r13
    je number_done
    cmp byte ptr [r12], '.'
    jne number_exp
    inc r12
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
number_frac_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_exp
    cmp al, '9'
    ja number_exp
    jmp number_frac_digits

number_exp:
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, 'e'
    je number_exp_take
    cmp al, 'E'
    jne number_done
number_exp_take:
    inc r12
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    jne number_exp_first
number_exp_sign:
    inc r12
    cmp r12, r13
    je fail
number_exp_first:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb fail
    cmp al, '9'
    ja fail
number_exp_digits:
    inc r12
    cmp r12, r13
    je number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    ja number_done
    jmp number_exp_digits
number_done:
    ret

parse_string:
    cmp r12, r13
    je fail
    cmp byte ptr [r12], '"'
    jne fail
    inc r12
string_loop:
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_end
    cmp al, '\'
    je string_escape
    cmp al, 32
    jb fail
    inc r12
    jmp string_loop
string_end:
    inc r12
    ret
string_escape:
    inc r12
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je string_esc_one
    cmp al, '\'
    je string_esc_one
    cmp al, '/'
    je string_esc_one
    cmp al, 'b'
    je string_esc_one
    cmp al, 'f'
    je string_esc_one
    cmp al, 'n'
    je string_esc_one
    cmp al, 'r'
    je string_esc_one
    cmp al, 't'
    je string_esc_one
    cmp al, 'u'
    jne fail
    inc r12
    mov ecx, 4
string_hex_loop:
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb string_hex_bad
    cmp al, '9'
    jbe string_hex_ok
    cmp al, 'A'
    jb string_hex_bad
    cmp al, 'F'
    jbe string_hex_ok
    cmp al, 'a'
    jb string_hex_bad
    cmp al, 'f'
    ja string_hex_bad
string_hex_ok:
    inc r12
    dec ecx
    jnz string_hex_loop
    jmp string_loop
string_hex_bad:
    jmp fail
string_esc_one:
    inc r12
    jmp string_loop

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], ']'
    jne array_value
    inc r12
    ret
array_value:
    call parse_value
array_after_value:
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], ']'
    je array_close
    cmp byte ptr [r12], ','
    jne fail
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    call parse_value
    jmp array_after_value
array_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], '}'
    jne object_member
    inc r12
    ret
object_member:
    cmp byte ptr [r12], '"'
    jne fail
    call parse_string
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], ':'
    jne fail
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    call parse_value
object_after_value:
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], '}'
    je object_close
    cmp byte ptr [r12], ','
    jne fail
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], '"'
    jne fail
    jmp object_member
object_close:
    inc r12
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "

.bss
.align 16
input:
    .zero 65536
outbuf:
    .zero 64
