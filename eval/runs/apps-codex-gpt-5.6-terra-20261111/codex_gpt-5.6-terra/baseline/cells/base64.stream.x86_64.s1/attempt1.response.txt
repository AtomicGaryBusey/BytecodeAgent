.text
.globl _start
_start:
    lea rbx, [rip + input]
    mov rsi, rbx
    mov r8d, 49152

read_loop:
    test r8, r8
    jz read_full
    xor edi, edi
    mov rdx, r8
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rsi, rax
    sub r8, rax
    jmp read_loop

read_full:
    xor edi, edi
    lea rsi, [rip + scratch]
    mov edx, 1
    xor eax, eax
    syscall
    lea r12, [rbx + 49152]
    jmp encode_start

read_done:
    mov r12, rsi

encode_start:
    lea r13, [rip + output]
    lea r15, [rip + alphabet]
    xor r14d, r14d

encode_loop:
    lea rax, [rbx + 3]
    cmp rax, r12
    ja encode_final

    movzx eax, byte ptr [rbx]
    shl eax, 16
    movzx edx, byte ptr [rbx + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rbx + 2]
    or eax, edx

    mov ecx, eax
    shr ecx, 18
    mov dl, byte ptr [r15 + rcx]
    mov byte ptr [r13], dl

    mov ecx, eax
    shr ecx, 12
    and ecx, 63
    mov dl, byte ptr [r15 + rcx]
    mov byte ptr [r13 + 1], dl

    mov ecx, eax
    shr ecx, 6
    and ecx, 63
    mov dl, byte ptr [r15 + rcx]
    mov byte ptr [r13 + 2], dl

    and eax, 63
    mov dl, byte ptr [r15 + rax]
    mov byte ptr [r13 + 3], dl

    add rbx, 3
    add r13, 4
    add r14d, 4
    cmp r14d, 76
    jne encode_loop
    mov byte ptr [r13], 10
    inc r13
    xor r14d, r14d
    jmp encode_loop

encode_final:
    cmp rbx, r12
    je finish_encode
    mov rax, r12
    sub rax, rbx
    cmp rax, 1
    je final_one

final_two:
    movzx eax, byte ptr [rbx]
    movzx edx, byte ptr [rbx + 1]

    mov ecx, eax
    shr ecx, 2
    mov sil, byte ptr [r15 + rcx]
    mov byte ptr [r13], sil

    and eax, 3
    shl eax, 4
    mov esi, edx
    shr esi, 4
    or eax, esi
    mov sil, byte ptr [r15 + rax]
    mov byte ptr [r13 + 1], sil

    and edx, 15
    shl edx, 2
    mov sil, byte ptr [r15 + rdx]
    mov byte ptr [r13 + 2], sil
    mov byte ptr [r13 + 3], 61
    jmp final_emit

final_one:
    movzx eax, byte ptr [rbx]

    mov ecx, eax
    shr ecx, 2
    mov dl, byte ptr [r15 + rcx]
    mov byte ptr [r13], dl

    and eax, 3
    shl eax, 4
    mov dl, byte ptr [r15 + rax]
    mov byte ptr [r13 + 1], dl
    mov byte ptr [r13 + 2], 61
    mov byte ptr [r13 + 3], 61

final_emit:
    add r13, 4
    add r14d, 4
    cmp r14d, 76
    jne finish_encode
    mov byte ptr [r13], 10
    inc r13
    xor r14d, r14d

finish_encode:
    test r14d, r14d
    jz write_output
    mov byte ptr [r13], 10
    inc r13

write_output:
    lea rsi, [rip + output]
    mov rdx, r13
    sub rdx, rsi
    test rdx, rdx
    jz exit_program
    mov edi, 1
    mov eax, 1
    syscall

exit_program:
    xor edi, edi
    mov eax, 60
    syscall
