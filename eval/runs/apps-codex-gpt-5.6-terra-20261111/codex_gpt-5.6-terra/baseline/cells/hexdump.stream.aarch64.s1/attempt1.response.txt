.text
.equ BUF_SIZE, 65536

.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0
    movz x26, #1, lsl #16

read_loop:
    cmp x19, x26
    b.hs read_done
    add x1, x20, x19
    sub x2, x26, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    mov x21, x19
    mov x22, #0

row_loop:
    cbz x21, rows_done
    cmp x21, #16
    b.ls short_row
    mov x25, #16
    b have_row_count

short_row:
    mov x25, x21

have_row_count:
    mov x0, x20
    mov x1, x25
    mov x2, x22
    bl emit_row
    add x20, x20, x25
    sub x21, x21, x25
    add x22, x22, x25
    b row_loop

rows_done:
    mov x0, x19
    bl emit_final
    mov x0, #0
    mov x8, #93
    svc #0

emit_row:
    adrp x3, output_buf
    add x3, x3, :lo12:output_buf
    adrp x7, hex_digits
    add x7, x7, :lo12:hex_digits

    mov w4, #28
row_offset_loop:
    lsr w6, w2, w4
    and w6, w6, #15
    ldrb w6, [x7, w6, uxtw]
    strb w6, [x3], #1
    subs w4, w4, #4
    b.ge row_offset_loop

    mov w5, #32
    strb w5, [x3], #1
    strb w5, [x3], #1

    mov x4, #0
row_hex_loop:
    cmp x4, x1
    b.hs row_blank_slot

    ldrb w5, [x0, x4]
    lsr w6, w5, #4
    ldrb w6, [x7, w6, uxtw]
    strb w6, [x3], #1
    and w6, w5, #15
    ldrb w6, [x7, w6, uxtw]
    strb w6, [x3], #1
    mov w6, #32
    strb w6, [x3], #1
    b row_slot_done

row_blank_slot:
    mov w6, #32
    strb w6, [x3], #1
    strb w6, [x3], #1
    strb w6, [x3], #1

row_slot_done:
    cmp x4, #7
    b.ne row_no_group_space
    mov w6, #32
    strb w6, [x3], #1

row_no_group_space:
    add x4, x4, #1
    cmp x4, #16
    b.lo row_hex_loop

    mov w5, #32
    strb w5, [x3], #1
    mov w5, #124
    strb w5, [x3], #1

    mov x4, #0
row_ascii_loop:
    cmp x4, x1
    b.hs row_ascii_done
    ldrb w5, [x0, x4]
    cmp w5, #32
    b.lo row_dot
    cmp w5, #126
    b.hi row_dot
    b row_ascii_store

row_dot:
    mov w5, #46

row_ascii_store:
    strb w5, [x3], #1
    add x4, x4, #1
    b row_ascii_loop

row_ascii_done:
    mov w5, #124
    strb w5, [x3], #1
    mov w5, #10
    strb w5, [x3], #1

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x3, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

emit_final:
    adrp x3, output_buf
    add x3, x3, :lo12:output_buf
    adrp x7, hex_digits
    add x7, x7, :lo12:hex_digits

    mov w4, #28
final_offset_loop:
    lsr w6, w0, w4
    and w6, w6, #15
    ldrb w6, [x7, w6, uxtw]
    strb w6, [x3], #1
    subs w4, w4, #4
    b.ge final_offset_loop

    mov w5, #10
    strb w5, [x3], #1

    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x3, x1
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero BUF_SIZE
.balign 16
output_buf:
    .zero 128
