.text
.globl _start
_start:
    lea rbx, [rip + inputbuf]
    lea r8, [rip + inputbuf + 65536]
read_loop:
    mov rdx, r8
    sub rdx, rbx
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r9, [rip + inputbuf]
    mov r14, rbx
    mov rbx, r9
find_bang:
    cmp rbx, r14
    jae no_bang
    cmp byte ptr [rbx], 33
    je have_bang
    inc rbx
    jmp find_bang
have_bang:
    mov r12, rbx
    lea r13, [rbx + 1]
    jmp validate_start
no_bang:
    mov r12, r14
    mov r13, r14

validate_start:
    lea r15, [rip + match_table]
    lea r11, [rip + bracket_stack]
    xor r8d, r8d
    mov rbx, r9
validate_loop:
    cmp rbx, r12
    jae validate_done
    mov al, byte ptr [rbx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rbx
    jmp validate_loop

validate_open:
    mov rcx, rbx
    sub rcx, r9
    mov dword ptr [r11 + r8*4], ecx
    inc r8d
    inc rbx
    jmp validate_loop

validate_close:
    test r8d, r8d
    jz bad_program
    dec r8d
    mov ecx, dword ptr [r11 + r8*4]
    mov rdx, rbx
    sub rdx, r9
    mov dword ptr [r15 + rcx*4], edx
    mov dword ptr [r15 + rdx*4], ecx
    inc rbx
    jmp validate_loop

validate_done:
    test r8d, r8d
    jnz bad_program

    lea rdi, [rip + tape]
    lea r8, [rip + outputbuf]
    lea r10, [rip + outputbuf + 65536]
    mov rsi, r8
    mov rbx, r9

run_loop:
    cmp rbx, r12
    jae run_done
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, 62
    je cmd_right
    cmp al, 60
    je cmd_left
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_dot
    cmp al, 44
    je cmd_comma
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    jmp run_loop

cmd_right:
    inc rdi
    jmp run_loop
cmd_left:
    dec rdi
    jmp run_loop
cmd_plus:
    inc byte ptr [rdi]
    jmp run_loop
cmd_minus:
    dec byte ptr [rdi]
    jmp run_loop

cmd_dot:
    mov al, byte ptr [rdi]
    mov byte ptr [rsi], al
    inc rsi
    cmp rsi, r10
    jb run_loop
    call flush_output
    jmp run_loop

cmd_comma:
    cmp r13, r14
    jae comma_zero
    mov al, byte ptr [r13]
    inc r13
    mov byte ptr [rdi], al
    jmp run_loop
comma_zero:
    mov byte ptr [rdi], 0
    jmp run_loop

cmd_open:
    cmp byte ptr [rdi], 0
    jne run_loop
    mov rcx, rbx
    sub rcx, r9
    dec rcx
    mov edx, dword ptr [r15 + rcx*4]
    lea rbx, [r9 + rdx + 1]
    jmp run_loop

cmd_close:
    cmp byte ptr [rdi], 0
    je run_loop
    mov rcx, rbx
    sub rcx, r9
    dec rcx
    mov edx, dword ptr [r15 + rcx*4]
    lea rbx, [r9 + rdx + 1]
    jmp run_loop

run_done:
    call flush_output
    mov rax, 60
    xor rdi, rdi
    syscall

bad_program:
    mov rax, 60
    mov rdi, 1
    syscall

flush_output:
    cmp rsi, r8
    je flush_return
    push rdi
    mov rdx, rsi
    sub rdx, r8
    mov rax, 1
    mov rdi, 1
    mov rsi, r8
    syscall
    lea rsi, [rip + outputbuf]
    pop rdi
flush_return:
    ret

.bss
.align 16
inputbuf:
    .zero 65537
.align 16
outputbuf:
    .zero 65536
.align 16
tape:
    .zero 30000
.align 16
match_table:
    .zero 262144
.align 16
bracket_stack:
    .zero 4000
