.text
.globl _start
_start:
    xor r12d, r12d
    lea rbx, [rip + inputbuf]

read_loop:
    cmp r12d, 32768
    jae parse_input
    mov edx, 32768
    sub edx, r12d
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea r10, [rip + descriptors]
    xor r8d, r8d
    xor r9d, r9d
    xor ecx, ecx

parse_loop:
    cmp rcx, r12
    jae parse_done
    cmp byte ptr [rbx + rcx], 10
    jne parse_next
    mov dword ptr [r10 + r8*8], r9d
    mov eax, ecx
    sub eax, r9d
    mov dword ptr [r10 + r8*8 + 4], eax
    inc r8
    lea r9, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_done:
    cmp r9, r12
    jae sort_begin
    mov dword ptr [r10 + r8*8], r9d
    mov eax, r12d
    sub eax, r9d
    mov dword ptr [r10 + r8*8 + 4], eax
    inc r8

sort_begin:
    mov r15, r8
    cmp r15, 1
    jbe output_begin
    lea r11, [rip + tempdescriptors]
    mov r14d, 1

sort_pass:
    xor r8d, r8d

merge_group:
    cmp r8, r15
    jae copy_back

    mov eax, r8d
    add eax, r14d
    cmp rax, r15
    jbe mid_ok
    mov eax, r15d
mid_ok:
    mov r9d, eax

    mov ebp, r9d
    add ebp, r14d
    cmp rbp, r15
    jbe right_ok
    mov ebp, r15d
right_ok:
    mov esi, r8d
    mov edi, r9d
    mov edx, r8d

merge_compare:
    cmp esi, r9d
    jae merge_rest_b
    cmp edi, ebp
    jae merge_rest_a

    mov eax, dword ptr [r10 + rsi*8]
    mov r12d, dword ptr [r10 + rsi*8 + 4]
    mov r8d, dword ptr [r10 + rdi*8]
    mov r13d, dword ptr [r10 + rdi*8 + 4]
    lea rax, [rbx + rax]
    lea r8, [rbx + r8]

compare_bytes:
    test r12d, r12d
    jz compare_prefix
    test r13d, r13d
    jz take_b
    movzx ecx, byte ptr [rax]
    cmp cl, byte ptr [r8]
    ja take_b
    jb take_a
    inc rax
    inc r8
    dec r12d
    dec r13d
    jmp compare_bytes

compare_prefix:
    test r13d, r13d
    jz take_a
    jmp take_a

take_a:
    mov rax, qword ptr [r10 + rsi*8]
    mov qword ptr [r11 + rdx*8], rax
    inc esi
    inc edx
    jmp merge_compare

take_b:
    mov rax, qword ptr [r10 + rdi*8]
    mov qword ptr [r11 + rdx*8], rax
    inc edi
    inc edx
    jmp merge_compare

merge_rest_a:
    cmp esi, r9d
    jae merge_done
    mov rax, qword ptr [r10 + rsi*8]
    mov qword ptr [r11 + rdx*8], rax
    inc esi
    inc edx
    jmp merge_rest_a

merge_rest_b:
    cmp edi, ebp
    jae merge_done
    mov rax, qword ptr [r10 + rdi*8]
    mov qword ptr [r11 + rdx*8], rax
    inc edi
    inc edx
    jmp merge_rest_b

merge_done:
    mov r8d, ebp
    jmp merge_group

copy_back:
    xor ecx, ecx
copy_loop:
    cmp rcx, r15
    jae next_pass
    mov rax, qword ptr [r11 + rcx*8]
    mov qword ptr [r10 + rcx*8], rax
    inc rcx
    jmp copy_loop

next_pass:
    shl r14, 1
    cmp r14, r15
    jb sort_pass

output_begin:
    xor r14d, r14d

output_loop:
    cmp r14, r15
    jae exit_program
    mov eax, dword ptr [r10 + r14*8]
    mov edx, dword ptr [r10 + r14*8 + 4]
    test edx, edx
    jz output_newline
    lea rsi, [rbx + rax]
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc r14
    jmp output_loop

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
inputbuf:
    .zero 32768
.align 16
descriptors:
    .zero 8000
.align 16
tempdescriptors:
    .zero 8000
