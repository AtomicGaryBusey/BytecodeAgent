.text
.globl _start
_start:
    xor r12d, r12d
    lea rbx, [rip + inputbuf]

read_loop:
    mov edx, 32768
    sub edx, r12d
    lea rsi, [rbx + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz parse_input
    add r12, rax
    jmp read_loop

parse_input:
    lea r10, [rip + descriptors]
    xor r8d, r8d
    xor r9d, r9d
    xor ecx, ecx

parse_loop:
    cmp rcx, r12
    jae parse_done
    cmp byte ptr [rbx + rcx], 10
    jne parse_next
    mov dword ptr [r10 + r8*8], r9d
    mov eax, ecx
    sub eax, r9d
    mov dword ptr [r10 + r8*8 + 4], eax
    inc r8
    lea r9, [rcx + 1]
parse_next:
    inc rcx
    jmp parse_loop

parse_done:
    cmp r9, r12
    jae sort_begin
    mov dword ptr [r10 + r8*8], r9d
    mov eax, r12d
    sub eax, r9d
    mov dword ptr [r10 + r8*8 + 4], eax
    inc r8

sort_begin:
    mov r15, r8
    cmp r15, 1
    jbe output_begin
    mov edi, 1

sort_outer:
    cmp rdi, r15
    jae output_begin
    mov r8d, dword ptr [r10 + rdi*8]
    mov r9d, dword ptr [r10 + rdi*8 + 4]
    mov rsi, rdi

sort_inner:
    test rsi, rsi
    jz sort_insert
    mov r12d, dword ptr [r10 + rsi*8 - 8]
    mov r13d, dword ptr [r10 + rsi*8 - 4]
    xor r14d, r14d

compare_loop:
    cmp r14d, r13d
    jae compare_prefix
    cmp r14d, r9d
    jae compare_prefix
    movzx eax, byte ptr [rbx + r12 + r14]
    movzx ecx, byte ptr [rbx + r8 + r14]
    cmp eax, ecx
    ja sort_shift
    jb sort_insert
    inc r14
    jmp compare_loop

compare_prefix:
    cmp r13d, r9d
    ja sort_shift
    jmp sort_insert

sort_shift:
    mov eax, dword ptr [r10 + rsi*8 - 8]
    mov edx, dword ptr [r10 + rsi*8 - 4]
    mov dword ptr [r10 + rsi*8], eax
    mov dword ptr [r10 + rsi*8 + 4], edx
    dec rsi
    jmp sort_inner

sort_insert:
    mov dword ptr [r10 + rsi*8], r8d
    mov dword ptr [r10 + rsi*8 + 4], r9d
    inc rdi
    jmp sort_outer

output_begin:
    xor r14d, r14d

output_loop:
    cmp r14, r15
    jae exit_program
    mov eax, dword ptr [r10 + r14*8]
    mov edx, dword ptr [r10 + r14*8 + 4]
    test edx, edx
    jz output_newline
    lea rsi, [rbx + rax]
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall
    inc r14
    jmp output_loop

exit_program:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
inputbuf:
    .zero 32768
.align 16
descriptors:
    .zero 8000
