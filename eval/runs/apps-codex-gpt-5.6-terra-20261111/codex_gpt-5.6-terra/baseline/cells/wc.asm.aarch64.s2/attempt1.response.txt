.text
.globl _start

_start:
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x23, inputbuf
    add x23, x23, :lo12:inputbuf

read_loop:
    mov x0, #0
    mov x1, x23
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done

    add x21, x21, x0
    mov x1, x23
    add x2, x1, x0

scan_loop:
    cmp x1, x2
    b.hs read_loop

    ldrb w3, [x1], #1
    cmp w3, #0x20
    b.eq whitespace

    sub w4, w3, #9
    cmp w4, #4
    b.ls whitespace

    cbnz x22, scan_loop
    add x20, x20, #1
    mov x22, #1
    b scan_loop

whitespace:
    cmp w3, #10
    b.ne 1f
    add x19, x19, #1
1:
    mov x22, #0
    b scan_loop

input_done:
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    mov x25, x24

    mov x0, x19
    mov x1, x24
    bl append_decimal
    mov x24, x1
    mov w0, #0x20
    strb w0, [x24], #1

    mov x0, x20
    mov x1, x24
    bl append_decimal
    mov x24, x1
    mov w0, #0x20
    strb w0, [x24], #1

    mov x0, x21
    mov x1, x24
    bl append_decimal
    mov x24, x1
    mov w0, #10
    strb w0, [x24], #1

    sub x2, x24, x25
    mov x0, #1
    mov x1, x25
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

append_decimal:
    cbnz x0, 2f
    mov w5, #48
    strb w5, [x1], #1
    ret

2:
    sub sp, sp, #32
    mov x2, sp
    mov x6, sp
    mov x4, #10

3:
    udiv x3, x0, x4
    msub x5, x3, x4, x0
    add w5, w5, #48
    strb w5, [x2], #1
    mov x0, x3
    cbnz x0, 3b

4:
    cmp x6, x2
    b.eq 5f
    ldrb w5, [x6], #1
    strb w5, [x1], #1
    b 4b

5:
    add sp, sp, #32
    ret

.bss
.balign 16
inputbuf:
    .zero 4096
.balign 16
outbuf:
    .zero 64
