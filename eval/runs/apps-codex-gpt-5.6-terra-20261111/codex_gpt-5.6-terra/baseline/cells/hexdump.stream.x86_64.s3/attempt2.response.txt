.equ INPUT_CAP, 65537

.text
.globl _start
_start:
    lea r13, [rip + input_buffer]
    xor r12d, r12d

read_loop:
    mov rdx, INPUT_CAP
    sub rdx, r12
    lea rsi, [r13 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add r12, rax
    jmp read_loop

read_done:
    xor r14d, r14d

row_loop:
    cmp r14, r12
    jae final_offset

    mov r15, r12
    sub r15, r14
    cmp r15, 16
    jbe row_count_ready
    mov r15, 16

row_count_ready:
    lea r9, [r13 + r14]
    lea rbx, [rip + output_buffer]
    lea rsi, [rip + hex_digits]

    mov ecx, 28
row_offset_hex:
    mov rax, r14
    shr rax, cl
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx], al
    inc rbx
    sub ecx, 4
    jns row_offset_hex

    mov byte ptr [rbx], ' '
    inc rbx
    mov byte ptr [rbx], ' '
    inc rbx

    xor r8d, r8d
hex_slots:
    cmp r8, r15
    jae hex_padding

    movzx eax, byte ptr [r9 + r8]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [rsi + rdx]
    mov byte ptr [rbx], dl
    inc rbx
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx], al
    inc rbx
    mov byte ptr [rbx], ' '
    inc rbx
    jmp hex_slot_done

hex_padding:
    mov byte ptr [rbx], ' '
    inc rbx
    mov byte ptr [rbx], ' '
    inc rbx
    mov byte ptr [rbx], ' '
    inc rbx

hex_slot_done:
    cmp r8, 7
    jne no_middle_space
    mov byte ptr [rbx], ' '
    inc rbx

no_middle_space:
    inc r8
    cmp r8, 16
    jb hex_slots

    mov byte ptr [rbx], ' '
    inc rbx
    mov byte ptr [rbx], '|'
    inc rbx

    xor r8d, r8d
ascii_loop:
    cmp r8, r15
    jae ascii_done
    mov al, byte ptr [r9 + r8]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    jmp ascii_store

ascii_dot:
    mov al, '.'

ascii_store:
    mov byte ptr [rbx], al
    inc rbx
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rbx], '|'
    inc rbx
    mov byte ptr [rbx], 10
    inc rbx

    mov rdx, rbx
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r14, r15
    jmp row_loop

final_offset:
    lea rbx, [rip + output_buffer]
    lea rsi, [rip + hex_digits]
    mov ecx, 28

final_hex:
    mov rax, r14
    shr rax, cl
    and eax, 15
    mov al, byte ptr [rsi + rax]
    mov byte ptr [rbx], al
    inc rbx
    sub ecx, 4
    jns final_hex

    mov byte ptr [rbx], 10

    lea rsi, [rip + output_buffer]
    mov edx, 9
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero INPUT_CAP
.align 16
output_buffer:
    .zero 128
