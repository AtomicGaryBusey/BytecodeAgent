.text
.globl _start
_start:
    lea rbx, [rip + input_buf]
    lea rdx, [rip + input_end]

read_loop:
    cmp rbx, rdx
    je read_full
    mov rsi, rbx
    mov rdx, input_end - input_buf
    lea rax, [rip + input_end]
    sub rax, rbx
    mov rdx, rax
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    add rbx, rax
    lea rdx, [rip + input_end]
    jmp read_loop

read_full:
    lea rsi, [rip + extra_byte]
    mov edx, 1
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jle read_done
    jmp read_done

read_done:
    lea rsi, [rip + input_buf]

find_bang:
    cmp rsi, rbx
    je no_bang
    cmp byte ptr [rsi], 0x21
    je got_bang
    inc rsi
    jmp find_bang

got_bang:
    mov r12, rsi
    lea r13, [rsi + 1]
    mov r14, rbx
    jmp brackets_begin

no_bang:
    mov r12, rbx
    mov r13, rbx
    mov r14, rbx

brackets_begin:
    lea rdi, [rip + input_buf]
    lea r8, [rip + jump_map]
    lea r9, [rip + bracket_stack]
    xor r10d, r10d
    lea rsi, [rip + input_buf]

bracket_loop:
    cmp rsi, r12
    je bracket_done
    movzx ecx, byte ptr [rsi]
    cmp cl, '['
    je open_bracket
    cmp cl, ']'
    je close_bracket
    inc rsi
    jmp bracket_loop

open_bracket:
    mov rax, rsi
    sub rax, rdi
    mov word ptr [r9 + r10*2], ax
    inc r10
    inc rsi
    jmp bracket_loop

close_bracket:
    test r10, r10
    jz bracket_error
    dec r10
    movzx edx, word ptr [r9 + r10*2]
    mov rax, rsi
    sub rax, rdi
    mov word ptr [r8 + rax*2], dx
    mov word ptr [r8 + rdx*2], ax
    inc rsi
    jmp bracket_loop

bracket_done:
    test r10, r10
    jnz bracket_error

    mov r9, rdi
    lea rbx, [rip + tape]
    mov r15, r9

run_loop:
    cmp r15, r12
    je success_exit
    movzx eax, byte ptr [r15]
    inc r15

    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    jmp run_loop

cmd_right:
    inc rbx
    jmp run_loop

cmd_left:
    dec rbx
    jmp run_loop

cmd_plus:
    inc byte ptr [rbx]
    jmp run_loop

cmd_minus:
    dec byte ptr [rbx]
    jmp run_loop

cmd_dot:
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov edx, 1
    syscall
    jmp run_loop

cmd_comma:
    cmp r13, r14
    je comma_zero
    mov al, byte ptr [r13]
    mov byte ptr [rbx], al
    inc r13
    jmp run_loop

comma_zero:
    mov byte ptr [rbx], 0
    jmp run_loop

cmd_open:
    cmp byte ptr [rbx], 0
    jne run_loop
    lea rax, [r15 - 1]
    sub rax, r9
    movzx eax, word ptr [r8 + rax*2]
    lea r15, [r9 + rax + 1]
    jmp run_loop

cmd_close:
    cmp byte ptr [rbx], 0
    je run_loop
    lea rax, [r15 - 1]
    sub rax, r9
    movzx eax, word ptr [r8 + rax*2]
    lea r15, [r9 + rax + 1]
    jmp run_loop

bracket_error:
    mov eax, 60
    mov edi, 1
    syscall

success_exit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
input_buf:
    .zero 65536
input_end:
extra_byte:
    .zero 1
.align 16
jump_map:
    .zero 131072
.align 16
bracket_stack:
    .zero 2000
.align 16
tape:
    .zero 30000
