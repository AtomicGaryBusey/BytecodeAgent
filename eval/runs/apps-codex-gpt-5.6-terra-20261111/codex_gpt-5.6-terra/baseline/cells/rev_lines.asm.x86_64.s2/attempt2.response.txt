.equ BUFSZ, 65537

.section .text
.globl _start
_start:
    lea r12, [rip + buffer]
    mov r13d, BUFSZ

read_loop:
    xor eax, eax
    xor edi, edi
    mov rsi, r12
    mov rdx, r13
    syscall
    test rax, rax
    jz process_input
    add r12, rax
    sub r13, rax
    jmp read_loop

process_input:
    lea rbx, [rip + buffer]
    cmp rbx, r12
    je exit_ok

line_start:
    mov r8, rbx

find_newline:
    cmp r8, r12
    je final_line
    cmp byte ptr [r8], 10
    je complete_line
    inc r8
    jmp find_newline

complete_line:
    mov rdi, rbx
    mov rsi, r8
    dec rsi

reverse_complete:
    cmp rdi, rsi
    jae write_complete
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp reverse_complete

write_complete:
    mov rdx, r8
    sub rdx, rbx
    test rdx, rdx
    jz write_complete_newline
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall

write_complete_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    lea rbx, [r8 + 1]
    cmp rbx, r12
    jne line_start
    jmp exit_ok

final_line:
    mov rdi, rbx
    mov rsi, r12
    dec rsi

reverse_final:
    cmp rdi, rsi
    jae write_final
    mov al, byte ptr [rdi]
    mov dl, byte ptr [rsi]
    mov byte ptr [rdi], dl
    mov byte ptr [rsi], al
    inc rdi
    dec rsi
    jmp reverse_final

write_final:
    mov rdx, r12
    sub rdx, rbx
    test rdx, rdx
    jz write_final_newline
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    syscall

write_final_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
buffer:
    .zero BUFSZ
