.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    movz x21, #1
    movk x21, #1, lsl #16

read_loop:
    mov x0, #0
    mov x1, x20
    mov x2, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    sub x21, x21, x0
    cbnz x21, read_loop

read_done:
    sub x25, x20, x19
    mov x24, x19
    mov x23, #0
    adrp x22, hex_digits
    add x22, x22, :lo12:hex_digits
    adrp x20, output_buf
    add x20, x20, :lo12:output_buf

row_loop:
    cbz x25, final_line
    mov x26, x25
    cmp x26, #16
    b.ls row_count_ready
    mov x26, #16

row_count_ready:
    mov x21, x20
    mov x27, #28

row_offset_loop:
    lsr x0, x23, x27
    and x0, x0, #15
    ldrb w1, [x22, x0]
    strb w1, [x21], #1
    subs x27, x27, #4
    b.ge row_offset_loop

    mov w0, #32
    strb w0, [x21], #1
    strb w0, [x21], #1
    mov x27, #0

row_slot_loop:
    cmp x27, x26
    b.hs row_blank_slot
    ldrb w0, [x24, x27]
    lsr w1, w0, #4
    ldrb w1, [x22, x1]
    strb w1, [x21], #1
    and w1, w0, #15
    ldrb w1, [x22, x1]
    strb w1, [x21], #1
    mov w0, #32
    strb w0, [x21], #1
    b row_slot_done

row_blank_slot:
    mov w0, #32
    strb w0, [x21], #1
    strb w0, [x21], #1
    strb w0, [x21], #1

row_slot_done:
    cmp x27, #7
    b.ne row_no_extra_space
    mov w0, #32
    strb w0, [x21], #1

row_no_extra_space:
    add x27, x27, #1
    cmp x27, #16
    b.lo row_slot_loop

    mov w0, #32
    strb w0, [x21], #1
    mov w0, #'|'
    strb w0, [x21], #1
    mov x27, #0

row_ascii_loop:
    cmp x27, x26
    b.hs row_ascii_done
    ldrb w0, [x24, x27]
    cmp w0, #0x20
    b.lo row_ascii_dot
    cmp w0, #0x7e
    b.hi row_ascii_dot
    b row_ascii_store

row_ascii_dot:
    mov w0, #'.'

row_ascii_store:
    strb w0, [x21], #1
    add x27, x27, #1
    b row_ascii_loop

row_ascii_done:
    mov w0, #'|'
    strb w0, [x21], #1
    mov w0, #10
    strb w0, [x21], #1

    mov x0, #1
    mov x1, x20
    sub x2, x21, x20
    mov x8, #64
    svc #0

    add x24, x24, x26
    add x23, x23, x26
    sub x25, x25, x26
    b row_loop

final_line:
    mov x21, x20
    mov x27, #28

final_offset_loop:
    lsr x0, x23, x27
    and x0, x0, #15
    ldrb w1, [x22, x0]
    strb w1, [x21], #1
    subs x27, x27, #4
    b.ge final_offset_loop

    mov w0, #10
    strb w0, [x21], #1

    mov x0, #1
    mov x1, x20
    sub x2, x21, x20
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65537
.balign 16
output_buf:
    .zero 128
