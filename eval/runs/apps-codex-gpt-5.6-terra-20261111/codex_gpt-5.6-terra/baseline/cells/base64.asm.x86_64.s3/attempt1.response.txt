.globl _start
.equ IN_SIZE, 49152
.equ OUT_SIZE, 66400

.text
_start:
    xor ebx, ebx

read_loop:
    mov eax, IN_SIZE
    sub eax, ebx
    jnz read_space

    xor edi, edi
    lea rsi, [rsp - 1]
    mov edx, 1
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    jmp read_loop

read_space:
    xor edi, edi
    lea rsi, [rip + inbuf]
    add rsi, rbx
    mov edx, eax
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea rsi, [rip + inbuf]
    lea rdi, [rip + outbuf]
    lea r9, [rip + alphabet]

    mov rax, rbx
    xor edx, edx
    mov ecx, 3
    div rcx
    mov rcx, rax
    mov r10d, edx
    xor r8d, r8d

    test rcx, rcx
    jz final_group

triple_loop:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    add rsi, 3

    mov ebx, eax
    shr ebx, 18
    movzx edx, byte ptr [r9 + rbx]
    mov byte ptr [rdi], dl
    inc rdi

    mov ebx, eax
    shr ebx, 12
    and ebx, 63
    movzx edx, byte ptr [r9 + rbx]
    mov byte ptr [rdi], dl
    inc rdi

    mov ebx, eax
    shr ebx, 6
    and ebx, 63
    movzx edx, byte ptr [r9 + rbx]
    mov byte ptr [rdi], dl
    inc rdi

    and eax, 63
    movzx edx, byte ptr [r9 + rax]
    mov byte ptr [rdi], dl
    inc rdi

    add r8d, 4
    cmp r8d, 76
    jne triple_next
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

triple_next:
    dec rcx
    jnz triple_loop

final_group:
    test r10d, r10d
    jz encoding_done

    movzx eax, byte ptr [rsi]
    shl eax, 16
    cmp r10d, 2
    jne final_bytes_ready
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx

final_bytes_ready:
    mov ebx, eax
    shr ebx, 18
    movzx edx, byte ptr [r9 + rbx]
    mov byte ptr [rdi], dl
    inc rdi

    mov ebx, eax
    shr ebx, 12
    and ebx, 63
    movzx edx, byte ptr [r9 + rbx]
    mov byte ptr [rdi], dl
    inc rdi

    cmp r10d, 1
    je final_third_pad
    mov ebx, eax
    shr ebx, 6
    and ebx, 63
    movzx edx, byte ptr [r9 + rbx]
    mov byte ptr [rdi], dl
    inc rdi
    jmp final_fourth

final_third_pad:
    mov byte ptr [rdi], '='
    inc rdi

final_fourth:
    mov byte ptr [rdi], '='
    inc rdi

    add r8d, 4
    cmp r8d, 76
    jne encoding_done
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

encoding_done:
    test r8d, r8d
    jz write_start
    mov byte ptr [rdi], 10
    inc rdi

write_start:
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    test rdx, rdx
    jz exit_ok

write_loop:
    mov eax, 1
    mov edi, 1
    syscall
    add rsi, rax
    sub rdx, rax
    jnz write_loop

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
inbuf:
    .zero IN_SIZE
.align 16
outbuf:
    .zero OUT_SIZE
