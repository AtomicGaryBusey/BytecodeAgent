.section .rodata
err_msg:
.asciz "error\n"
err_len = . - err_msg - 1

.text
.globl _start
_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, #0

read_loop:
    movz x2, #1, lsl #16
    sub x2, x2, x20
    cbz x2, read_done
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x21, x19
    add x22, x19, x20
    adrp x23, value_stack
    add x23, x23, :lo12:value_stack

line_outer:
    cmp x21, x22
    b.hs program_exit
    mov w24, #0
    mov w25, #0

line_scan:
    cmp x21, x22
    b.hs line_finish
    ldrb w8, [x21]
    cmp w8, #10
    b.eq line_newline
    cmp w8, #32
    b.eq skip_separator
    cmp w8, #9
    b.eq skip_separator

    mov x9, x21

token_scan:
    cmp x21, x22
    b.hs token_ready
    ldrb w10, [x21]
    cmp w10, #10
    b.eq token_ready
    cmp w10, #32
    b.eq token_ready
    cmp w10, #9
    b.eq token_ready
    add x21, x21, #1
    b token_scan

token_ready:
    cbnz w25, line_scan
    mov x0, x9
    sub x1, x21, x9
    bl eval_token
    b line_scan

skip_separator:
    add x21, x21, #1
    b line_scan

line_newline:
    add x21, x21, #1

line_finish:
    cbnz w25, output_error
    cmp w24, #1
    b.ne output_error
    ldr x0, [x23]
    bl print_number
    b line_outer

output_error:
    bl print_error
    b line_outer

program_exit:
    mov x0, #0
    mov x8, #93
    svc #0

eval_token:
    cmp x1, #1
    b.ne literal_start
    ldrb w2, [x0]
    cmp w2, #43
    b.eq op_add
    cmp w2, #45
    b.eq op_sub
    cmp w2, #42
    b.eq op_mul
    cmp w2, #47
    b.eq op_div
    cmp w2, #37
    b.eq op_mod

literal_start:
    mov x3, #0
    mov w4, #0
    mov x5, #0
    mov x6, #10
    ldrb w2, [x0]
    cmp w2, #45
    b.ne literal_digits
    mov w4, #1
    mov x3, #1
    cmp x3, x1
    b.hs token_bad

literal_digits:
    cmp x3, x1
    b.hs literal_done
    ldrb w2, [x0, x3]
    sub w7, w2, #48
    cmp w7, #9
    b.hi token_bad
    madd x5, x5, x6, x7
    add x3, x3, #1
    b literal_digits

literal_done:
    cbz w4, literal_push
    neg x5, x5

literal_push:
    add x8, x23, w24, uxtw #3
    str x5, [x8]
    add w24, w24, #1
    ret

op_add:
    cmp w24, #2
    b.lo token_bad
    sub w8, w24, #2
    add x9, x23, w8, uxtw #3
    ldr x10, [x9]
    ldr x11, [x9, #8]
    add x12, x10, x11
    b op_store

op_sub:
    cmp w24, #2
    b.lo token_bad
    sub w8, w24, #2
    add x9, x23, w8, uxtw #3
    ldr x10, [x9]
    ldr x11, [x9, #8]
    sub x12, x10, x11
    b op_store

op_mul:
    cmp w24, #2
    b.lo token_bad
    sub w8, w24, #2
    add x9, x23, w8, uxtw #3
    ldr x10, [x9]
    ldr x11, [x9, #8]
    mul x12, x10, x11
    b op_store

op_div:
    cmp w24, #2
    b.lo token_bad
    sub w8, w24, #2
    add x9, x23, w8, uxtw #3
    ldr x10, [x9]
    ldr x11, [x9, #8]
    cbz x11, token_bad
    mov x13, #1
    lsl x13, x13, #63
    cmp x10, x13
    b.ne div_normal
    mov x14, #-1
    cmp x11, x14
    b.eq token_bad
div_normal:
    sdiv x12, x10, x11
    b op_store

op_mod:
    cmp w24, #2
    b.lo token_bad
    sub w8, w24, #2
    add x9, x23, w8, uxtw #3
    ldr x10, [x9]
    ldr x11, [x9, #8]
    cbz x11, token_bad
    mov x13, #1
    lsl x13, x13, #63
    cmp x10, x13
    b.ne mod_normal
    mov x14, #-1
    cmp x11, x14
    b.eq token_bad
mod_normal:
    sdiv x13, x10, x11
    msub x12, x13, x11, x10

op_store:
    str x12, [x9]
    sub w24, w24, #1
    ret

token_bad:
    mov w25, #1
    ret

print_error:
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x0, #1
    mov x2, #err_len
    mov x8, #64
    svc #0
    ret

print_number:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    add x7, x1, #32
    sub x2, x7, #1
    mov w8, #0
    strb w8, [x2]
    mov x3, x0
    mov x5, #10
    mov w8, #0
    tbnz x3, #63, number_negative
    b number_digits

number_negative:
    mov w8, #1
    neg x3, x3

number_digits:
    udiv x4, x3, x5
    msub x6, x4, x5, x3
    sub x2, x2, #1
    add w6, w6, #48
    strb w6, [x2]
    mov x3, x4
    cbnz x3, number_digits
    cbz w8, number_write
    sub x2, x2, #1
    mov w6, #45
    strb w6, [x2]

number_write:
    sub x2, x7, x2
    mov x1, x2
    sub x1, x7, x2
    mov x0, #1
    mov x8, #64
    svc #0
    ret

.bss
.balign 16
input_buf:
.zero 65536
.balign 16
value_stack:
.zero 8192
.balign 16
output_buf:
.zero 32
