.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, #0

read_loop:
    mov x0, #0
    add x1, x20, x21
    movz x2, #0x8000
    sub x2, x2, x21
    mov x8, #63
    svc #0
    cmp x0, #0
    b.eq read_done
    add x21, x21, x0
    b read_loop

read_done:
    adrp x26, line_ptrs
    add x26, x26, :lo12:line_ptrs
    adrp x27, line_lens
    add x27, x27, :lo12:line_lens

    mov x22, #0
    mov x23, x20
    mov x24, x20
    add x25, x20, x21

parse_loop:
    cmp x23, x25
    b.hs parse_end
    ldrb w0, [x23]
    cmp w0, #10
    b.ne parse_next

    sub x1, x23, x24
    str x24, [x26, x22, lsl #3]
    str x1, [x27, x22, lsl #3]
    add x22, x22, #1
    add x23, x23, #1
    mov x24, x23
    b parse_loop

parse_next:
    add x23, x23, #1
    b parse_loop

parse_end:
    cmp x24, x25
    b.eq sort_begin
    sub x1, x25, x24
    str x24, [x26, x22, lsl #3]
    str x1, [x27, x22, lsl #3]
    add x22, x22, #1

sort_begin:
    adrp x20, line_ptrs
    add x20, x20, :lo12:line_ptrs
    adrp x21, line_lens
    add x21, x21, :lo12:line_lens

    mov x23, #1

sort_outer:
    cmp x23, x22
    b.hs output_begin

    ldr x3, [x20, x23, lsl #3]
    ldr x4, [x21, x23, lsl #3]
    mov x24, x23

sort_inner:
    cmp x24, #0
    b.eq sort_insert

    sub x10, x24, #1
    ldr x5, [x20, x10, lsl #3]
    ldr x6, [x21, x10, lsl #3]
    mov x7, #0

compare_bytes:
    cmp x7, x6
    b.hs compare_lengths
    cmp x7, x4
    b.hs compare_lengths
    ldrb w8, [x5, x7]
    ldrb w9, [x3, x7]
    cmp w8, w9
    b.ne compare_different
    add x7, x7, #1
    b compare_bytes

compare_different:
    b.hi sort_shift
    b sort_insert

compare_lengths:
    cmp x6, x4
    b.hi sort_shift
    b sort_insert

sort_shift:
    ldr x11, [x20, x10, lsl #3]
    ldr x12, [x21, x10, lsl #3]
    str x11, [x20, x24, lsl #3]
    str x12, [x21, x24, lsl #3]
    sub x24, x24, #1
    b sort_inner

sort_insert:
    str x3, [x20, x24, lsl #3]
    str x4, [x21, x24, lsl #3]
    add x23, x23, #1
    b sort_outer

output_begin:
    adrp x24, newline_byte
    add x24, x24, :lo12:newline_byte
    mov x23, #0

output_loop:
    cmp x23, x22
    b.hs exit_ok

    ldr x3, [x20, x23, lsl #3]
    ldr x4, [x21, x23, lsl #3]
    cbz x4, output_newline

    mov x0, #1
    mov x1, x3
    mov x2, x4
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    mov x1, x24
    mov x2, #1
    mov x8, #64
    svc #0

    add x23, x23, #1
    b output_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
newline_byte:
    .byte 10

.bss
.balign 8
input_buf:
    .zero 32768
.balign 8
line_ptrs:
    .zero 8000
.balign 8
line_lens:
    .zero 8000
