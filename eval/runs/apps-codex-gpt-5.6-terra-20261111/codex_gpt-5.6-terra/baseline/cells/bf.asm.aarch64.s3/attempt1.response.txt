.text
.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93
.equ IN_SIZE, 65536
.equ TAPE_SIZE, 30000

.globl _start
_start:
    adrp x9, inputbuf
    add x9, x9, :lo12:inputbuf
    mov x10, x9
    movz x11, #1, lsl #16
    add x11, x9, x11

read_loop:
    mov x0, #0
    mov x1, x10
    sub x2, x11, x10
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x10, x10, x0
    b read_loop

read_done:
    mov x27, x9
    mov x22, x10
    mov x20, x10
    mov x21, x10
    mov x12, x9

find_bang:
    cmp x12, x10
    b.hs no_bang
    ldrb w13, [x12], #1
    cmp w13, #33
    b.ne find_bang
    sub x20, x12, #1
    mov x21, x12
    b validate_setup

no_bang:
    mov x20, x10
    mov x21, x10

validate_setup:
    adrp x12, bracket_stack
    add x12, x12, :lo12:bracket_stack
    adrp x13, bracket_map
    add x13, x13, :lo12:bracket_map
    mov x10, x27
    mov x11, #0

validate_loop:
    cmp x10, x20
    b.hs validate_end
    ldrb w15, [x10]
    cmp w15, #91
    b.eq validate_open
    cmp w15, #93
    b.eq validate_close
    add x10, x10, #1
    b validate_loop

validate_open:
    str x10, [x12, x11, lsl #3]
    add x11, x11, #1
    add x10, x10, #1
    b validate_loop

validate_close:
    cbz x11, invalid
    sub x11, x11, #1
    ldr x16, [x12, x11, lsl #3]
    sub x17, x10, x27
    sub x18, x16, x27
    str w17, [x13, x18, lsl #2]
    str w18, [x13, x17, lsl #2]
    add x10, x10, #1
    b validate_loop

validate_end:
    cbnz x11, invalid

    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x25, outputbuf
    add x25, x25, :lo12:outputbuf
    mov x24, x25
    movz x26, #1, lsl #16
    add x26, x25, x26
    adrp x28, bracket_map
    add x28, x28, :lo12:bracket_map
    mov x19, x27

run_loop:
    cmp x19, x20
    b.hs finish
    ldrb w12, [x19], #1
    cmp w12, #62
    b.eq cmd_right
    cmp w12, #60
    b.eq cmd_left
    cmp w12, #43
    b.eq cmd_plus
    cmp w12, #45
    b.eq cmd_minus
    cmp w12, #46
    b.eq cmd_dot
    cmp w12, #44
    b.eq cmd_comma
    cmp w12, #91
    b.eq cmd_open
    cmp w12, #93
    b.eq cmd_close
    b run_loop

cmd_right:
    add x23, x23, #1
    b run_loop

cmd_left:
    sub x23, x23, #1
    b run_loop

cmd_plus:
    ldrb w13, [x23]
    add w13, w13, #1
    strb w13, [x23]
    b run_loop

cmd_minus:
    ldrb w13, [x23]
    sub w13, w13, #1
    strb w13, [x23]
    b run_loop

cmd_dot:
    ldrb w13, [x23]
    strb w13, [x24], #1
    cmp x24, x26
    b.ne run_loop
    mov x0, #1
    mov x1, x25
    movz x2, #1, lsl #16
    mov x8, #SYS_write
    svc #0
    mov x24, x25
    b run_loop

cmd_comma:
    cmp x21, x22
    b.hs comma_zero
    ldrb w13, [x21], #1
    strb w13, [x23]
    b run_loop

comma_zero:
    strb wzr, [x23]
    b run_loop

cmd_open:
    ldrb w13, [x23]
    cbnz w13, run_loop
    sub x10, x19, #1
    sub x10, x10, x27
    ldr w10, [x28, x10, lsl #2]
    add x19, x27, x10
    add x19, x19, #1
    b run_loop

cmd_close:
    ldrb w13, [x23]
    cbz w13, run_loop
    sub x10, x19, #1
    sub x10, x10, x27
    ldr w10, [x28, x10, lsl #2]
    add x19, x27, x10
    add x19, x19, #1
    b run_loop

finish:
    sub x2, x24, x25
    cbz x2, success
    mov x0, #1
    mov x1, x25
    mov x8, #SYS_write
    svc #0

success:
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

invalid:
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

.bss
.balign 16
inputbuf:
    .zero 65537
.balign 16
tape:
    .zero 30000
.balign 16
outputbuf:
    .zero 65536
.balign 16
bracket_stack:
    .zero 8000
.balign 16
bracket_map:
    .zero 262144
