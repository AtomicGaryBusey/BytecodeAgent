.text
.globl _start
_start:
    adrp x19, buffer
    add x19, x19, :lo12:buffer
    movz x9, #1, lsl #16
    add x9, x9, #1
    add x21, x19, x9
    mov x20, x19

read_loop:
    sub x2, x21, x20
    mov x0, #0
    mov x1, x20
    mov x8, #63
    svc #0
    cbz x0, process_start
    add x20, x20, x0
    b read_loop

process_start:
    mov x22, x19

line_scan:
    cmp x22, x20
    b.eq exit_ok
    mov x23, x22

scan_next:
    cmp x23, x20
    b.eq final_line
    ldrb w5, [x23]
    cmp w5, #10
    b.eq complete_line
    add x23, x23, #1
    b scan_next

complete_line:
    mov x3, x22
    sub x4, x23, #1
    cmp x3, x4
    b.hs write_complete

reverse_complete:
    ldrb w5, [x3]
    ldrb w6, [x4]
    strb w6, [x3]
    strb w5, [x4]
    add x3, x3, #1
    sub x4, x4, #1
    cmp x3, x4
    b.lo reverse_complete

write_complete:
    sub x2, x23, x22
    add x2, x2, #1
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0
    add x22, x23, #1
    b line_scan

final_line:
    mov x3, x22
    sub x4, x23, #1
    cmp x3, x4
    b.hs write_final

reverse_final:
    ldrb w5, [x3]
    ldrb w6, [x4]
    strb w6, [x3]
    strb w5, [x4]
    add x3, x3, #1
    sub x4, x4, #1
    cmp x3, x4
    b.lo reverse_final

write_final:
    mov w5, #10
    strb w5, [x23]
    sub x2, x23, x22
    add x2, x2, #1
    mov x0, #1
    mov x1, x22
    mov x8, #64
    svc #0

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.bss
.align 4
buffer:
    .zero 65538
