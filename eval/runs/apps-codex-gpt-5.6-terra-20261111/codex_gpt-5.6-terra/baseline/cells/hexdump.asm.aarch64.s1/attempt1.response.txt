.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, x20
    mov x19, #0
    movz x9, #1, lsl #16

read_loop:
    cmp x19, x9
    b.hs read_done
    sub x2, x9, x19
    mov x0, #0
    mov x1, x21
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    add x21, x21, x0
    b read_loop

read_done:
    adrp x26, hex_digits
    add x26, x26, :lo12:hex_digits
    adrp x28, output_buf
    add x28, x28, :lo12:output_buf
    mov x22, #0

row_loop:
    cmp x22, x19
    b.hs final_line

    sub x23, x19, x22
    mov x24, #16
    cmp x23, #16
    csel x23, x23, x24, lo

    mov x25, x28
    mov x27, #28
    mov x6, #8

row_offset_loop:
    lsr x4, x22, x27
    and x4, x4, #15
    ldrb w5, [x26, x4]
    strb w5, [x25], #1
    subs x6, x6, #1
    sub x27, x27, #4
    b.ne row_offset_loop

    mov w5, #0x20
    strb w5, [x25], #1
    strb w5, [x25], #1

    mov x27, #0

slot_loop:
    cmp x27, x23
    b.hs slot_blank

    add x4, x22, x27
    ldrb w5, [x20, x4]
    lsr w6, w5, #4
    ldrb w7, [x26, x6]
    strb w7, [x25], #1
    and w6, w5, #15
    ldrb w7, [x26, x6]
    strb w7, [x25], #1
    mov w7, #0x20
    strb w7, [x25], #1
    b slot_after

slot_blank:
    mov w7, #0x20
    strb w7, [x25], #1
    strb w7, [x25], #1
    strb w7, [x25], #1

slot_after:
    cmp x27, #7
    b.ne no_slot_gap
    mov w7, #0x20
    strb w7, [x25], #1

no_slot_gap:
    add x27, x27, #1
    cmp x27, #16
    b.lo slot_loop

    mov w7, #0x20
    strb w7, [x25], #1
    mov w7, #0x7c
    strb w7, [x25], #1

    mov x27, #0

ascii_loop:
    cmp x27, x23
    b.hs ascii_done
    add x4, x22, x27
    ldrb w5, [x20, x4]
    cmp w5, #0x20
    b.lo ascii_dot
    cmp w5, #0x7e
    b.hi ascii_dot
    b ascii_store

ascii_dot:
    mov w5, #0x2e

ascii_store:
    strb w5, [x25], #1
    add x27, x27, #1
    b ascii_loop

ascii_done:
    mov w5, #0x7c
    strb w5, [x25], #1
    mov w5, #0x0a
    strb w5, [x25], #1

    sub x2, x25, x28
    mov x0, #1
    mov x1, x28
    mov x8, #64
    svc #0

    add x22, x22, #16
    b row_loop

final_line:
    mov x25, x28
    mov x27, #28
    mov x6, #8

final_hex_loop:
    lsr x4, x19, x27
    and x4, x4, #15
    ldrb w5, [x26, x4]
    strb w5, [x25], #1
    subs x6, x6, #1
    sub x27, x27, #4
    b.ne final_hex_loop

    mov w5, #0x0a
    strb w5, [x25], #1

    sub x2, x25, x28
    mov x0, #1
    mov x1, x28
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.balign 16
input_buf:
    .zero 65536
.balign 16
output_buf:
    .zero 80
