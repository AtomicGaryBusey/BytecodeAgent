.text
.globl _start
_start:
    adrp x19, inputbuf
    add x19, x19, :lo12:inputbuf
    mov x20, #0
    movz x21, #1, lsl #16

read_loop:
    sub x2, x21, x20
    add x1, x19, x20
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    mov x27, x19
    add x22, x19, x20
    mov x9, x19

find_bang:
    cmp x9, x22
    b.hs no_bang
    ldrb w0, [x9]
    cmp w0, #33
    b.eq got_bang
    add x9, x9, #1
    b find_bang

got_bang:
    mov x20, x9
    add x21, x9, #1
    b setup_check

no_bang:
    mov x20, x22
    mov x21, x22

setup_check:
    adrp x26, bracket_map
    add x26, x26, :lo12:bracket_map
    adrp x11, bracket_stack
    add x11, x11, :lo12:bracket_stack
    mov x9, x27
    mov w10, #0

check_loop:
    cmp x9, x20
    b.hs check_done
    ldrb w0, [x9]
    cmp w0, #91
    b.eq check_open
    cmp w0, #93
    b.eq check_close
    add x9, x9, #1
    b check_loop

check_open:
    sub x12, x9, x27
    str w12, [x11, w10, uxtw #2]
    add w10, w10, #1
    add x9, x9, #1
    b check_loop

check_close:
    cbz w10, bad_program
    sub w10, w10, #1
    ldr w13, [x11, w10, uxtw #2]
    sub x12, x9, x27
    add w14, w12, #1
    str w14, [x26, w13, uxtw #2]
    add w14, w13, #1
    str w14, [x26, w12, uxtw #2]
    add x9, x9, #1
    b check_loop

check_done:
    cbnz w10, bad_program

    mov x19, x27
    adrp x23, tape
    add x23, x23, :lo12:tape
    adrp x24, outbuf
    add x24, x24, :lo12:outbuf
    adrp x28, outbuf_end
    add x28, x28, :lo12:outbuf_end

run_loop:
    cmp x19, x20
    b.hs run_done
    ldrb w0, [x19]
    cmp w0, #62
    b.eq cmd_right
    cmp w0, #60
    b.eq cmd_left
    cmp w0, #43
    b.eq cmd_plus
    cmp w0, #45
    b.eq cmd_minus
    cmp w0, #46
    b.eq cmd_dot
    cmp w0, #44
    b.eq cmd_comma
    cmp w0, #91
    b.eq cmd_open
    cmp w0, #93
    b.eq cmd_close
    add x19, x19, #1
    b run_loop

cmd_right:
    add x23, x23, #1
    add x19, x19, #1
    b run_loop

cmd_left:
    sub x23, x23, #1
    add x19, x19, #1
    b run_loop

cmd_plus:
    ldrb w1, [x23]
    add w1, w1, #1
    strb w1, [x23]
    add x19, x19, #1
    b run_loop

cmd_minus:
    ldrb w1, [x23]
    sub w1, w1, #1
    strb w1, [x23]
    add x19, x19, #1
    b run_loop

cmd_dot:
    ldrb w1, [x23]
    strb w1, [x24], #1
    cmp x24, x28
    b.ne dot_next
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    sub x2, x24, x1
    mov x0, #1
    mov x8, #64
    svc #0
    mov x24, x1
dot_next:
    add x19, x19, #1
    b run_loop

cmd_comma:
    cmp x21, x22
    b.hs comma_zero
    ldrb w1, [x21], #1
    strb w1, [x23]
    add x19, x19, #1
    b run_loop

comma_zero:
    strb wzr, [x23]
    add x19, x19, #1
    b run_loop

cmd_open:
    ldrb w1, [x23]
    cbnz w1, open_next
    sub x9, x19, x27
    ldr w10, [x26, x9, lsl #2]
    add x19, x27, w10, uxtw
    b run_loop
open_next:
    add x19, x19, #1
    b run_loop

cmd_close:
    ldrb w1, [x23]
    cbz w1, close_next
    sub x9, x19, x27
    ldr w10, [x26, x9, lsl #2]
    add x19, x27, w10, uxtw
    b run_loop
close_next:
    add x19, x19, #1
    b run_loop

run_done:
    adrp x1, outbuf
    add x1, x1, :lo12:outbuf
    cmp x24, x1
    b.eq good_exit
    sub x2, x24, x1
    mov x0, #1
    mov x8, #64
    svc #0

good_exit:
    mov x0, #0
    mov x8, #93
    svc #0

bad_program:
    mov x0, #1
    mov x8, #93
    svc #0

.bss

    .zero 4000
