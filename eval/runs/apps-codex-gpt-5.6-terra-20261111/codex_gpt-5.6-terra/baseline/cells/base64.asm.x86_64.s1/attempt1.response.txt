.text
.globl _start
_start:
    xor ebx, ebx

read_loop:
    mov eax, 49152
    sub eax, ebx
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, eax
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz encode_start
    add rbx, rax
    jmp read_loop

encode_start:
    lea rsi, [rip + input_buffer]
    lea rdi, [rip + output_buffer]
    mov r10, rdi
    lea r9, [rip + alphabet]
    mov rcx, rbx
    xor r8d, r8d

full_loop:
    cmp rcx, 3
    jb tail

    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx
    movzx edx, byte ptr [rsi + 2]
    or eax, edx
    add rsi, 3
    sub rcx, 3

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], dl

    mov edx, eax
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 3], dl

    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne full_loop
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d
    jmp full_loop

tail:
    test rcx, rcx
    jz final_line
    cmp rcx, 1
    je tail_one

tail_two:
    movzx eax, byte ptr [rsi]
    shl eax, 16
    movzx edx, byte ptr [rsi + 1]
    shl edx, 8
    or eax, edx

    mov edx, eax
    shr edx, 18
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl

    mov edx, eax
    shr edx, 12
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], dl

    mov edx, eax
    shr edx, 6
    and edx, 63
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 2], dl

    mov byte ptr [rdi + 3], 61
    jmp tail_done

tail_one:
    movzx eax, byte ptr [rsi]

    mov edx, eax
    shr edx, 2
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl

    and eax, 3
    shl eax, 4
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi + 1], al

    mov byte ptr [rdi + 2], 61
    mov byte ptr [rdi + 3], 61

tail_done:
    add rdi, 4
    add r8d, 4
    cmp r8d, 76
    jne final_line
    mov byte ptr [rdi], 10
    inc rdi
    xor r8d, r8d

final_line:
    test r8d, r8d
    jz write_output
    mov byte ptr [rdi], 10
    inc rdi

write_output:
    cmp rdi, r10
    je exit_ok
    mov rdx, rdi
    sub rdx, r10
    mov rsi, r10
    mov edi, 1
    mov eax, 1
    syscall

exit_ok:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
input_buffer:
    .zero 49152
output_buffer:
    .zero 66400
