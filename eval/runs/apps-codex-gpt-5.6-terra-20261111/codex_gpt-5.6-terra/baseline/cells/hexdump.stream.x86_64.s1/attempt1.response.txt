.equ BUF_SIZE, 65536

.text
.globl _start
_start:
    xor ebx, ebx
    lea r9, [rip + hex_digits]

read_more:
    cmp rbx, BUF_SIZE
    je read_full
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, BUF_SIZE
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    je input_done
    add rbx, rax
    jmp read_more

read_full:
    lea rsi, [rip + overflow_byte]
    mov edx, 1
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jne read_full

input_done:
    xor r12d, r12d

row_loop:
    cmp r12, rbx
    je final_offset

    mov r13, rbx
    sub r13, r12
    cmp r13, 16
    jbe row_count_ready
    mov r13d, 16

row_count_ready:
    lea r8, [rip + input_buffer]
    add r8, r12
    lea rdi, [rip + output_buffer]

    mov eax, r12d
    mov ecx, 8
offset_digits:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jne offset_digits

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    add rdi, 2

    xor ecx, ecx
hex_slots:
    cmp rcx, r13
    jae blank_slot

    movzx eax, byte ptr [r8 + rcx]
    mov edx, eax
    shr edx, 4
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi], al
    inc rdi
    mov byte ptr [rdi], 0x20
    inc rdi
    jmp slot_done

blank_slot:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

slot_done:
    cmp ecx, 7
    jne no_middle_space
    mov byte ptr [rdi], 0x20
    inc rdi
no_middle_space:
    inc ecx
    cmp ecx, 16
    jne hex_slots

    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2

    xor ecx, ecx
ascii_loop:
    cmp rcx, r13
    je ascii_done
    movzx eax, byte ptr [r8 + rcx]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    jmp ascii_store
ascii_dot:
    mov al, 0x2e
ascii_store:
    mov byte ptr [rdi], al
    inc rdi
    inc ecx
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 10
    add rdi, 2

    mov rdx, rdi
    lea rsi, [rip + output_buffer]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add r12, r13
    jmp row_loop

final_offset:
    lea rdi, [rip + output_buffer]
    mov eax, r12d
    mov ecx, 8
final_digits:
    rol eax, 4
    mov edx, eax
    and edx, 15
    mov dl, byte ptr [r9 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    dec ecx
    jne final_digits
    mov byte ptr [rdi], 10

    lea rsi, [rip + output_buffer]
    mov edx, 9
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 80
overflow_byte:
    .zero 1
