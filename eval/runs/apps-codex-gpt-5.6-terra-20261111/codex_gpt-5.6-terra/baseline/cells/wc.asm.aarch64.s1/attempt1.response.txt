.text
.globl _start

_start:
    sub sp, sp, #64
    mov x19, #0
    mov x20, #0
    mov x21, #0
    mov x22, #0

    adrp x24, input_buf
    add x24, x24, :lo12:input_buf
    adrp x25, output_buf
    add x25, x25, :lo12:output_buf

read_more:
    mov x0, #0
    mov x1, x24
    mov x2, #4096
    mov x8, #63
    svc #0
    cbz x0, input_done

    mov x23, x0
    add x21, x21, x23
    mov x1, x24
    add x2, x24, x23

scan_byte:
    cmp x1, x2
    b.hs read_more

    ldrb w4, [x1], #1

    cmp w4, #10
    b.ne not_newline
    add x19, x19, #1

not_newline:
    cmp w4, #32
    b.eq whitespace

    sub w5, w4, #9
    cmp w5, #4
    b.ls whitespace

    cbnz x22, scan_byte
    add x20, x20, #1
    mov x22, #1
    b scan_byte

whitespace:
    mov x22, #0
    b scan_byte

input_done:
    mov x0, x19
    mov x1, x25
    bl format_dec

    mov w2, #32
    strb w2, [x0], #1

    mov x1, x0
    mov x0, x20
    bl format_dec

    mov w2, #32
    strb w2, [x0], #1

    mov x1, x0
    mov x0, x21
    bl format_dec

    mov w2, #10
    strb w2, [x0], #1

    sub x2, x0, x25
    mov x1, x25
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

exit_loop:
    b exit_loop

format_dec:
    cbnz x0, format_nonzero
    mov w2, #48
    strb w2, [x1], #1
    mov x0, x1
    ret

format_nonzero:
    mov x6, sp
    mov x2, sp
    mov x3, #10

format_digits:
    udiv x4, x0, x3
    msub x5, x4, x3, x0
    add w5, w5, #48
    strb w5, [x2], #1
    mov x0, x4
    cbnz x0, format_digits

format_copy:
    sub x2, x2, #1
    ldrb w5, [x2]
    strb w5, [x1], #1
    cmp x2, x6
    b.ne format_copy

    mov x0, x1
    ret

.bss
.balign 16
input_buf:
    .zero 4096
.balign 16
output_buf:
    .zero 64
