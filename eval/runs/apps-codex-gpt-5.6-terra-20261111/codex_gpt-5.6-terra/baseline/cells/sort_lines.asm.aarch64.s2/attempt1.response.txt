.text
.globl _start

_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x19, #0

read_loop:
    cmp x19, #32768
    b.eq read_full
    mov x2, #32768
    sub x2, x2, x19
    b read_do

read_full:
    mov x2, #1

read_do:
    add x1, x20, x19
    mov x0, #0
    mov x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    adrp x22, records
    add x22, x22, :lo12:records
    mov x23, #0
    mov x24, #0
    mov x25, #0

parse_loop:
    cmp x24, x19
    b.hs parse_end
    ldrb w0, [x20, x24]
    cmp w0, #10
    b.ne parse_next

    add x1, x20, x25
    str x1, [x22]
    sub x2, x24, x25
    str x2, [x22, #8]
    add x22, x22, #16
    add x23, x23, #1
    add x25, x24, #1

parse_next:
    add x24, x24, #1
    b parse_loop

parse_end:
    cmp x25, x19
    b.hs sort_start
    add x1, x20, x25
    str x1, [x22]
    sub x2, x19, x25
    str x2, [x22, #8]
    add x23, x23, #1

sort_start:
    cmp x23, #1
    b.ls output_start

    sub x24, x23, #1

sort_outer:
    cbz x24, output_start
    mov w25, #0
    adrp x26, records
    add x26, x26, :lo12:records
    mov x27, #0

sort_inner:
    cmp x27, x24
    b.hs sort_pass_done

    ldr x4, [x26]
    ldr x5, [x26, #8]
    ldr x6, [x26, #16]
    ldr x7, [x26, #24]

    cmp x5, x7
    csel x9, x5, x7, ls
    mov x8, #0

compare_bytes:
    cmp x8, x9
    b.hs compare_prefix
    ldrb w11, [x4, x8]
    ldrb w12, [x6, x8]
    cmp w11, w12
    b.hi sort_swap
    b.lo sort_no_swap
    add x8, x8, #1
    b compare_bytes

compare_prefix:
    cmp x5, x7
    b.hi sort_swap
    b sort_no_swap

sort_swap:
    str x6, [x26]
    str x7, [x26, #8]
    str x4, [x26, #16]
    str x5, [x26, #24]
    mov w25, #1

sort_no_swap:
    add x26, x26, #16
    add x27, x27, #1
    b sort_inner

sort_pass_done:
    cbz w25, output_start
    sub x24, x24, #1
    b sort_outer

output_start:
    adrp x27, newline
    add x27, x27, :lo12:newline
    adrp x25, records
    add x25, x25, :lo12:records
    mov x24, #0

output_loop:
    cmp x24, x23
    b.hs exit_ok
    ldr x1, [x25]
    ldr x2, [x25, #8]
    cbz x2, output_newline
    mov x0, #1
    mov x8, #64
    svc #0

output_newline:
    mov x0, #1
    mov x1, x27
    mov x2, #1
    mov x8, #64
    svc #0

    add x25, x25, #16
    add x24, x24, #1
    b output_loop

exit_ok:
    mov x0, #0
    mov x8, #93
    svc #0

.section .rodata

    .byte 10
