.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
err_msg:
    .ascii "error at "
err_len = . - err_msg
nl_msg:
    .ascii "\n"

.bss
.align 16
input_buf:
    .zero 65536
out_buf:
    .zero 32

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r12, [rip + input_buf]
    lea r13, [rip + input_buf]
    add r13, rbx
    call skip_ws
    cmp r12, r13
    je fail
    call parse_value
    call skip_ws
    cmp r12, r13
    jne fail
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
    cmp r12, r13
    je sw_done
sw_loop:
    movzx eax, byte ptr [r12]
    cmp al, 32
    je sw_take
    cmp al, 9
    je sw_take
    cmp al, 10
    je sw_take
    cmp al, 13
    jne sw_done
sw_take:
    inc r12
    cmp r12, r13
    jne sw_loop
sw_done:
    ret

parse_value:
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, 123
    je parse_object
    cmp al, 91
    je parse_array
    cmp al, 34
    je parse_string
    cmp al, 116
    je parse_true
    cmp al, 102
    je parse_false
    cmp al, 110
    je parse_null
    cmp al, 45
    je parse_number
    cmp al, 48
    jb fail
    cmp al, 57
    jbe parse_number
    jmp fail

parse_true:
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 114
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 117
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 101
    jne fail
    inc r12
    ret

parse_false:
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 97
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 108
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 115
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 101
    jne fail
    inc r12
    ret

parse_null:
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 117
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 108
    jne fail
    inc r12
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 108
    jne fail
    inc r12
    ret

parse_number:
    cmp byte ptr [r12], 45
    jne num_int
    inc r12
    cmp r12, r13
    je fail
num_int:
    movzx eax, byte ptr [r12]
    cmp al, 48
    je num_zero
    cmp al, 49
    jb fail
    cmp al, 57
    ja fail
num_digits:
    inc r12
    cmp r12, r13
    je num_after_int
    movzx eax, byte ptr [r12]
    cmp al, 48
    jb num_after_int
    cmp al, 57
    jbe num_digits
    jmp num_after_int
num_zero:
    inc r12
num_after_int:
    cmp r12, r13
    je num_done
    cmp byte ptr [r12], 46
    jne num_exp_check
    inc r12
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
num_frac:
    inc r12
    cmp r12, r13
    je num_exp_check
    movzx eax, byte ptr [r12]
    cmp al, 48
    jb num_exp_check
    cmp al, 57
    jbe num_frac
num_exp_check:
    cmp r12, r13
    je num_done
    movzx eax, byte ptr [r12]
    cmp al, 101
    je num_exp
    cmp al, 69
    jne num_done
num_exp:
    inc r12
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, 43
    je num_exp_sign
    cmp al, 45
    jne num_exp_digit
num_exp_sign:
    inc r12
    cmp r12, r13
    je fail
num_exp_digit:
    movzx eax, byte ptr [r12]
    cmp al, 48
    jb fail
    cmp al, 57
    ja fail
num_exp_digits:
    inc r12
    cmp r12, r13
    je num_done
    movzx eax, byte ptr [r12]
    cmp al, 48
    jb num_done
    cmp al, 57
    jbe num_exp_digits
num_done:
    ret

parse_string:
    inc r12
str_loop:
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, 34
    je str_end
    cmp al, 92
    je str_escape
    cmp al, 32
    jb fail
    inc r12
    jmp str_loop
str_end:
    inc r12
    ret
str_escape:
    inc r12
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, 34
    je str_simple
    cmp al, 92
    je str_simple
    cmp al, 47
    je str_simple
    cmp al, 98
    je str_simple
    cmp al, 102
    je str_simple
    cmp al, 110
    je str_simple
    cmp al, 114
    je str_simple
    cmp al, 116
    je str_simple
    cmp al, 117
    jne fail
    inc r12
    mov ecx, 4
str_hex:
    cmp r12, r13
    je fail
    movzx eax, byte ptr [r12]
    cmp al, 48
    jb str_hex_alpha
    cmp al, 57
    jbe str_hex_ok
str_hex_alpha:
    cmp al, 97
    jb str_hex_upper
    cmp al, 102
    jbe str_hex_ok
str_hex_upper:
    cmp al, 65
    jb fail
    cmp al, 70
    ja fail
str_hex_ok:
    inc r12
    dec ecx
    jnz str_hex
    jmp str_loop
str_simple:
    inc r12
    jmp str_loop

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 93
    je arr_empty
arr_value:
    call parse_value
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 93
    je arr_close
    cmp byte ptr [r12], 44
    jne fail
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 93
    je fail
    jmp arr_value
arr_empty:
    inc r12
    ret
arr_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 125
    je obj_empty
obj_member:
    cmp byte ptr [r12], 34
    jne fail
    call parse_string
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 58
    jne fail
    inc r12
    call skip_ws
    call parse_value
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 125
    je obj_close
    cmp byte ptr [r12], 44
    jne fail
    inc r12
    call skip_ws
    cmp r12, r13
    je fail
    cmp byte ptr [r12], 125
    je fail
    jmp obj_member
obj_empty:
    inc r12
    ret
obj_close:
    inc r12
    ret

fail:
    lea rax, [rip + input_buf]
    mov rbx, r12
    sub rbx, rax
    mov edi, 1
    lea rsi, [rip + err_msg]
    mov edx, err_len
    mov eax, 1
    syscall
    lea rcx, [rip + out_buf + 32]
    mov rax, rbx
    test rax, rax
    jnz digit_loop_start
    dec rcx
    mov byte ptr [rcx], 48
    jmp write_digits
digit_loop_start:
    mov r8, 10
digit_loop:
    xor edx, edx
    div r8
    dec rcx
    add dl, 48
    mov byte ptr [rcx], dl
    test rax, rax
    jne digit_loop
write_digits:
    lea rdx, [rip + out_buf + 32]
    sub rdx, rcx
    mov edi, 1
    mov rsi, rcx
    mov eax, 1
    syscall
    mov edi, 1
    lea rsi, [rip + nl_msg]
    mov edx, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
