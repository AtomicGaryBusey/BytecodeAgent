.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
error_msg:
    .ascii "error at "
error_len = . - error_msg
newline:
    .ascii "\n"

.bss
.align 16
input_buffer:
    .zero 65536
output_buffer:
    .zero 32

.text
.globl _start
_start:
    xor ebx, ebx
read_more:
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 65536
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_more

read_done:
    lea r12, [rip + input_buffer]
    lea r13, [rip + input_buffer]
    add r13, rbx
    call skip_ws
    cmp r12, r13
    je parse_fail
    call parse_value
    call skip_ws
    cmp r12, r13
    jne parse_fail
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, ok_len
    mov eax, 1
    syscall
    xor edi, edi
    mov eax, 60
    syscall

skip_ws:
    cmp r12, r13
    je skip_ws_done
skip_ws_loop:
    mov al, byte ptr [r12]
    cmp al, 32
    je skip_ws_take
    cmp al, 9
    je skip_ws_take
    cmp al, 10
    je skip_ws_take
    cmp al, 13
    jne skip_ws_done
skip_ws_take:
    inc r12
    cmp r12, r13
    jne skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    jbe parse_number
    jmp parse_fail

parse_true:
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'r'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'u'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'e'
    jne parse_fail
    inc r12
    ret

parse_false:
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'a'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'l'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 's'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'e'
    jne parse_fail
    inc r12
    ret

parse_null:
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'u'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'l'
    jne parse_fail
    inc r12
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], 'l'
    jne parse_fail
    inc r12
    ret

parse_number:
    cmp byte ptr [r12], '-'
    jne number_int
    inc r12
    cmp r12, r13
    je parse_fail
number_int:
    mov al, byte ptr [r12]
    cmp al, '0'
    je number_zero
    cmp al, '1'
    jb parse_fail
    cmp al, '9'
    ja parse_fail
number_digits:
    inc r12
    cmp r12, r13
    je number_after_int
    mov al, byte ptr [r12]
    cmp al, '0'
    jb number_after_int
    cmp al, '9'
    ja number_after_int
    jmp number_digits
number_zero:
    inc r12
number_after_int:
    cmp r12, r13
    je number_done
    cmp byte ptr [r12], '.'
    jne number_exp_check
    inc r12
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    ja parse_fail
number_frac_digits:
    inc r12
    cmp r12, r13
    je number_exp_check
    mov al, byte ptr [r12]
    cmp al, '0'
    jb number_exp_check
    cmp al, '9'
    jbe number_frac_digits
number_exp_check:
    cmp r12, r13
    je number_done
    mov al, byte ptr [r12]
    cmp al, 'e'
    je number_exp
    cmp al, 'E'
    jne number_done
number_exp:
    inc r12
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, '+'
    je number_exp_sign
    cmp al, '-'
    jne number_exp_digit
number_exp_sign:
    inc r12
    cmp r12, r13
    je parse_fail
number_exp_digit:
    mov al, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    ja parse_fail
number_exp_digits:
    inc r12
    cmp r12, r13
    je number_done
    mov al, byte ptr [r12]
    cmp al, '0'
    jb number_done
    cmp al, '9'
    jbe number_exp_digits
number_done:
    ret

parse_string:
    inc r12
string_loop:
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, '"'
    je string_end
    cmp al, '\'
    je string_escape
    cmp al, 32
    jb parse_fail
    inc r12
    jmp string_loop
string_end:
    inc r12
    ret
string_escape:
    inc r12
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, '"'
    je string_simple
    cmp al, '\'
    je string_simple
    cmp al, '/'
    je string_simple
    cmp al, 'b'
    je string_simple
    cmp al, 'f'
    je string_simple
    cmp al, 'n'
    je string_simple
    cmp al, 'r'
    je string_simple
    cmp al, 't'
    je string_simple
    cmp al, 'u'
    jne parse_fail
    inc r12
    mov ecx, 4
string_hex_loop:
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, '0'
    jb string_hex_alpha
    cmp al, '9'
    jbe string_hex_ok
string_hex_alpha:
    cmp al, 'a'
    jb string_hex_upper
    cmp al, 'f'
    jbe string_hex_ok
string_hex_upper:
    cmp al, 'A'
    jb parse_fail
    cmp al, 'F'
    ja parse_fail
string_hex_ok:
    inc r12
    dec ecx
    jnz string_hex_loop
    jmp string_loop
string_simple:
    inc r12
    jmp string_loop

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ']'
    je array_empty
array_value:
    call parse_value
array_after_value:
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ']'
    je array_close
    cmp byte ptr [r12], ','
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ']'
    je parse_fail
    jmp array_value
array_empty:
    inc r12
    ret
array_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '}'
    je object_empty
object_member:
    cmp byte ptr [r12], '"'
    jne parse_fail
    call parse_string
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ':'
    jne parse_fail
    inc r12
    call skip_ws
    call parse_value
object_after_value:
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '}'
    je object_close
    cmp byte ptr [r12], ','
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '}'
    je parse_fail
    jmp object_member
object_empty:
    inc r12
    ret
object_close:
    inc r12
    ret

parse_fail:
    lea rax, [rip + input_buffer]
    mov rbx, r12
    sub rbx, rax
    mov edi, 1
    lea rsi, [rip + error_msg]
    mov edx, error_len
    mov eax, 1
    syscall
    lea rcx, [rip + output_buffer + 32]
    mov rax, rbx
    test rax, rax
    jne error_digits
    dec rcx
    mov byte ptr [rcx], '0'
    jmp error_write_digits
error_digits:
    mov r8, 10
error_digit_loop:
    xor edx, edx
    div r8
    dec rcx
    add dl, '0'
    mov byte ptr [rcx], dl
    test rax, rax
    jne error_digit_loop
error_write_digits:
    lea rdx, [rip + output_buffer + 32]
    sub rdx, rcx
    mov edi, 1
    mov rsi, rcx
    mov eax, 1
    syscall
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    mov eax, 1
    syscall
    mov edi, 1
    mov eax, 60
    syscall
