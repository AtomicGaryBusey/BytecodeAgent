.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 4096
    xor eax, eax
    syscall
    test rax, rax
    jz make_output

    add r14, rax
    mov rcx, rax
    lea rsi, [rip + input_buffer]

scan_byte:
    movzx eax, byte ptr [rsi]
    inc rsi

    cmp al, 10
    jne not_newline
    inc r12

not_newline:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15, r15
    jnz scan_next
    inc r13
    mov r15d, 1
    jmp scan_next

whitespace:
    xor r15d, r15d

scan_next:
    dec rcx
    jnz scan_byte
    jmp read_more

make_output:
    lea rsi, [rip + output_buffer + 64]
    mov rbx, 10

    dec rsi
    mov byte ptr [rsi], 10

    mov rax, r14
bytes_digits:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz bytes_digits

    dec rsi
    mov byte ptr [rsi], ' '

    mov rax, r13
words_digits:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz words_digits

    dec rsi
    mov byte ptr [rsi], ' '

    mov rax, r12
lines_digits:
    xor edx, edx
    div rbx
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz lines_digits

    lea rdx, [rip + output_buffer + 64]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

.bss
.align 16
input_buffer:
    .zero 4096
.align 16
output_buffer:
    .zero 64
