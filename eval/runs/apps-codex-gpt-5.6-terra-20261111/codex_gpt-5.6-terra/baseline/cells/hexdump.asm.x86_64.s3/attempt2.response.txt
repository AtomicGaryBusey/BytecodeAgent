.text
.globl _start

_start:
    xor r12d, r12d

read_more:
    mov rdx, 65536
    sub rdx, r12
    lea rsi, [rip + input_buffer]
    add rsi, r12
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add r12, rax
    jmp read_more

input_done:
    xor r13d, r13d

row_loop:
    cmp r13, r12
    jae rows_done

    mov r14, r12
    sub r14, r13
    cmp r14, 16
    jbe row_size_ready
    mov r14, 16

row_size_ready:
    lea r15, [rip + input_buffer]
    add r15, r13

    lea rdi, [rip + line_buffer]
    mov eax, r13d
    call hex8

    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    lea rsi, [rdi + 2]

    xor ecx, ecx
fill_slots:
    mov word ptr [rsi], 0x2020
    mov byte ptr [rsi + 2], 32
    add rsi, 3
    cmp ecx, 7
    jne no_fill_gap
    mov byte ptr [rsi], 32
    inc rsi
no_fill_gap:
    inc ecx
    cmp ecx, 16
    jb fill_slots

    mov byte ptr [rsi], 32
    mov byte ptr [rsi + 1], 124
    lea rdi, [rsi + 2]

    lea rsi, [rip + line_buffer + 10]
    lea r8, [rip + hex_digits]
    xor ecx, ecx

byte_loop:
    cmp rcx, r14
    jae bytes_done

    movzx ebx, byte ptr [r15 + rcx]

    mov eax, ebx
    shr eax, 4
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rsi], al

    mov eax, ebx
    and eax, 15
    mov al, byte ptr [r8 + rax]
    mov byte ptr [rsi + 1], al

    cmp ebx, 32
    jb ascii_dot
    cmp ebx, 126
    ja ascii_dot
    mov byte ptr [rdi], bl
    jmp ascii_done

ascii_dot:
    mov byte ptr [rdi], 46

ascii_done:
    inc rdi
    add rsi, 3
    cmp ecx, 7
    jne no_byte_gap
    inc rsi
no_byte_gap:
    inc ecx
    jmp byte_loop

bytes_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    lea rax, [rip + line_buffer]
    sub rdi, rax
    add rdi, 2

    mov rdx, rdi
    lea rsi, [rip + line_buffer]
    mov edi, 1
    mov eax, 1
    syscall

    add r13, r14
    jmp row_loop

rows_done:
    lea rdi, [rip + line_buffer]
    mov eax, r12d
    call hex8
    mov byte ptr [rdi], 10

    mov edx, 9
    lea rsi, [rip + line_buffer]
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

hex8:
    lea r8, [rip + hex_digits]
    mov ecx, 8
hex8_loop:
    mov edx, eax
    shr edx, 28
    mov dl, byte ptr [r8 + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl eax, 4
    dec ecx
    jnz hex8_loop
    ret

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
.align 16
input_buffer:
    .space 65536
.align 16
line_buffer:
    .space 128
