.text
.globl _start

_start:
	mov x19, #0
	mov x20, #0
	mov x21, #0
	mov x22, #0

read_loop:
	mov x0, #0
	adrp x1, input_buf
	add x1, x1, :lo12:input_buf
	mov x2, #4096
	mov x8, #63
	svc #0
	cbz x0, input_done

	add x21, x21, x0
	adrp x1, input_buf
	add x1, x1, :lo12:input_buf
	add x2, x1, x0

scan_loop:
	cmp x1, x2
	b.eq read_loop

	ldrb w3, [x1], #1

	cmp w3, #10
	b.ne not_newline
	add x19, x19, #1

not_newline:
	cmp w3, #32
	b.eq whitespace
	sub w4, w3, #9
	cmp w4, #4
	b.ls whitespace

	cbnz x22, scan_loop
	add x20, x20, #1
	mov x22, #1
	b scan_loop

whitespace:
	mov x22, #0
	b scan_loop

input_done:
	adrp x23, output_buf
	add x23, x23, :lo12:output_buf
	mov x1, x23

	mov x0, x19
	bl append_decimal
	mov x1, x0
	mov w2, #32
	strb w2, [x1], #1

	mov x0, x20
	bl append_decimal
	mov x1, x0
	mov w2, #32
	strb w2, [x1], #1

	mov x0, x21
	bl append_decimal
	mov x1, x0
	mov w2, #10
	strb w2, [x1], #1

	sub x2, x1, x23
	mov x0, #1
	mov x1, x23
	mov x8, #64
	svc #0

	mov x0, #0
	mov x8, #93
	svc #0

append_decimal:
	mov x5, x1
	cbnz x0, decimal_loop_start
	mov w4, #48
	strb w4, [x1], #1
	mov x0, x1
	ret

decimal_loop_start:
	mov x6, #10

decimal_loop:
	udiv x3, x0, x6
	msub x4, x3, x6, x0
	add w4, w4, #48
	strb w4, [x1], #1
	mov x0, x3
	cbnz x0, decimal_loop

	sub x6, x1, #1

reverse_loop:
	cmp x5, x6
	b.hs reverse_done
	ldrb w7, [x5]
	ldrb w8, [x6]
	strb w8, [x5]
	strb w7, [x6]
	add x5, x5, #1
	sub x6, x6, #1
	b reverse_loop

reverse_done:
	mov x0, x1
	ret

.bss
.align 4
input_buf:
.zero 4096
.align 4
output_buf:
.zero 64
