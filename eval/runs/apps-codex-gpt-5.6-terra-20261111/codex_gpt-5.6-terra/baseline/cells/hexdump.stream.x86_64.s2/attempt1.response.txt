.text
.globl _start
_start:
    lea r13, [rip + input]
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    je read_done
    mov edx, 65536
    sub rdx, r12
    lea rsi, [r13 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    mov r14, r12
    xor ebx, ebx

row_loop:
    cmp rbx, r14
    jae final_offset

    mov r15, r14
    sub r15, rbx
    cmp r15, 16
    jbe row_size_ready
    mov r15, 16

row_size_ready:
    lea r10, [r13 + rbx]
    lea rdi, [rip + outbuf]

    mov rax, rbx
    mov ecx, 8
    call hex_fixed

    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r8d, r8d
hex_slots:
    cmp r8, 16
    jae hex_slots_done
    cmp r8, r15
    jae blank_slot

    movzx eax, byte ptr [r10 + r8]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [rip + hexchars + rax]
    mov byte ptr [rdi], al
    and edx, 15
    mov dl, byte ptr [rip + hexchars + rdx]
    mov byte ptr [rdi + 1], dl
    mov byte ptr [rdi + 2], 32
    add rdi, 3
    jmp slot_extra

blank_slot:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 32
    mov byte ptr [rdi + 2], 32
    add rdi, 3

slot_extra:
    inc r8
    cmp r8, 8
    jne hex_slots
    mov byte ptr [rdi], 32
    inc rdi
    jmp hex_slots

hex_slots_done:
    mov byte ptr [rdi], 32
    mov byte ptr [rdi + 1], 124
    add rdi, 2

    xor r8d, r8d
ascii_loop:
    cmp r8, r15
    jae ascii_done
    mov al, byte ptr [r10 + r8]
    cmp al, 32
    jb ascii_dot
    cmp al, 126
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_next

ascii_dot:
    mov byte ptr [rdi], 46

ascii_next:
    inc rdi
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 124
    mov byte ptr [rdi + 1], 10
    add rdi, 2

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add rbx, r15
    jmp row_loop

final_offset:
    lea rdi, [rip + outbuf]
    mov rax, r14
    mov ecx, 8
    call hex_fixed
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

hex_fixed:
    mov rdx, rax
    shr rdx, 28
    mov dl, byte ptr [rip + hexchars + rdx]
    mov byte ptr [rdi], dl
    inc rdi
    shl eax, 4
    dec ecx
    jnz hex_fixed
    ret

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.bss
.align 16
input:
    .zero 65536
.align 16
outbuf:
    .zero 128
