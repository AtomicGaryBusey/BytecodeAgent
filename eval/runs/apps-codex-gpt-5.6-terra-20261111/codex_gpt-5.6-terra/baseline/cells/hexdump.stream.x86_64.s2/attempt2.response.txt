.text
.globl _start
_start:
    lea r13, [rip + input]
    xor r12d, r12d

read_loop:
    cmp r12, 65536
    je read_zero
    mov edx, 65536
    sub rdx, r12
    lea rsi, [r13 + r12]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_zero:
    lea rsi, [rip + input]
    xor edx, edx
    xor edi, edi
    xor eax, eax
    syscall

read_done:
    mov r14, r12
    xor ebx, ebx
    lea r9, [rip + hexchars]

row_loop:
    cmp rbx, r14
    jae final_offset

    mov r15, r14
    sub r15, rbx
    cmp r15, 16
    jbe row_size_ready
    mov r15, 16

row_size_ready:
    lea r10, [r13 + rbx]
    lea rdi, [rip + outbuf]

    mov rax, rbx
    call hex_fixed

    mov word ptr [rdi], 0x2020
    add rdi, 2

    xor r8d, r8d
hex_slots:
    cmp r8, 16
    jae hex_slots_done
    cmp r8, r15
    jae blank_slot

    movzx eax, byte ptr [r10 + r8]
    mov edx, eax
    shr eax, 4
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi], al
    and edx, 15
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rdi + 1], al
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3
    jmp slot_separator

blank_slot:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x20
    mov byte ptr [rdi + 2], 0x20
    add rdi, 3

slot_separator:
    cmp r8, 7
    jne next_slot
    mov byte ptr [rdi], 0x20
    inc rdi

next_slot:
    inc r8
    jmp hex_slots

hex_slots_done:
    mov byte ptr [rdi], 0x20
    mov byte ptr [rdi + 1], 0x7c
    add rdi, 2

    xor r8d, r8d
ascii_loop:
    cmp r8, r15
    jae ascii_done
    movzx eax, byte ptr [r10 + r8]
    cmp al, 0x20
    jb ascii_dot
    cmp al, 0x7e
    ja ascii_dot
    mov byte ptr [rdi], al
    jmp ascii_store_done

ascii_dot:
    mov byte ptr [rdi], 0x2e

ascii_store_done:
    inc rdi
    inc r8
    jmp ascii_loop

ascii_done:
    mov byte ptr [rdi], 0x7c
    mov byte ptr [rdi + 1], 0x0a
    add rdi, 2

    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    add rbx, r15
    jmp row_loop

final_offset:
    lea rdi, [rip + outbuf]
    mov rax, r14
    call hex_fixed
    mov byte ptr [rdi], 0x0a
    inc rdi

    mov rdx, rdi
    lea rsi, [rip + outbuf]
    sub rdx, rsi
    mov edi, 1
    mov eax, 1
    syscall

    xor edi, edi
    mov eax, 60
    syscall

hex_fixed:
    mov rdx, rax
    mov ecx, 28
hex_fixed_loop:
    mov rax, rdx
    shr rax, cl
    and eax, 15
    mov al, byte ptr [r9 + rax]
    mov byte ptr [rdi], al
    inc rdi
    sub ecx, 4
    jns hex_fixed_loop
    ret

.section .rodata
hexchars:
    .ascii "0123456789abcdef"

.bss
.align 16
input:
    .zero 65536
.align 16
outbuf:
    .zero 80
