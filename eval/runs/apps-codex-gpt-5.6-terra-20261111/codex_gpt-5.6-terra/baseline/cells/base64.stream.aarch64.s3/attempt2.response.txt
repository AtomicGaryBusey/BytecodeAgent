.text
.equ IN_SIZE, 49152
.equ OUT_SIZE, 66560

.globl _start
_start:
    adrp x20, inbuf
    add x20, x20, :lo12:inbuf
    mov x21, xzr

read_loop:
    movz x2, #IN_SIZE
    subs x2, x2, x21
    b.ne read_have_space
    mov x1, x20
    b read_call
read_have_space:
    add x1, x20, x21
read_call:
    mov x0, xzr
    movz x8, #63
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    b read_loop

read_done:
    adrp x22, outbuf
    add x22, x22, :lo12:outbuf
    mov x25, x22
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet
    mov x23, xzr

group_loop:
    cmp x21, #3
    b.lo final_group

    ldrb w0, [x20]
    ldrb w1, [x20, #1]
    ldrb w2, [x20, #2]
    add x20, x20, #3
    sub x21, x21, #3

    lsr w3, w0, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    and w3, w0, #3
    lsl w3, w3, #4
    orr w3, w3, w1, lsr #4
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    and w3, w1, #15
    lsl w3, w3, #2
    orr w3, w3, w2, lsr #6
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    and w3, w2, #63
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    add x23, x23, #4
    cmp x23, #76
    b.ne group_loop
    movz w3, #10
    strb w3, [x22], #1
    mov x23, xzr
    b group_loop

final_group:
    cbz x21, final_no_bytes
    cmp x21, #1
    b.eq one_byte

    ldrb w0, [x20]
    ldrb w1, [x20, #1]

    lsr w3, w0, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    and w3, w0, #3
    lsl w3, w3, #4
    orr w3, w3, w1, lsr #4
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    and w3, w1, #15
    lsl w3, w3, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    movz w3, #61
    strb w3, [x22], #1
    movz w3, #10
    strb w3, [x22], #1
    b write_output

one_byte:
    ldrb w0, [x20]

    lsr w3, w0, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    and w3, w0, #3
    lsl w3, w3, #4
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x22], #1

    movz w3, #61
    strb w3, [x22], #1
    strb w3, [x22], #1
    movz w3, #10
    strb w3, [x22], #1
    b write_output

final_no_bytes:
    cbz x23, write_output
    movz w3, #10
    strb w3, [x22], #1

write_output:
    sub x2, x22, x25
    cbz x2, exit_ok
    movz x0, #1
    mov x1, x25
    movz x8, #64
    svc #0

exit_ok:
    mov x0, xzr
    movz x8, #93
    svc #0

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16
inbuf:
    .zero IN_SIZE
.balign 16
outbuf:
    .zero OUT_SIZE
