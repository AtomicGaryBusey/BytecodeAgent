.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    mov eax, 0
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 4096
    syscall
    test rax, rax
    jz make_output

    add r14, rax
    lea rbx, [rip + input_buffer]
    mov r8, rax

scan_byte:
    movzx eax, byte ptr [rbx]
    inc rbx
    dec r8

    cmp al, 10
    jne not_newline
    inc r12

not_newline:
    cmp al, 32
    je whitespace
    cmp al, 9
    jb word_byte
    cmp al, 13
    jbe whitespace

word_byte:
    test r15, r15
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    test r8, r8
    jnz scan_byte
    jmp read_more

make_output:
    lea rdi, [rip + output_buffer]

    mov rax, r12
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r13
    call append_decimal
    mov byte ptr [rdi], 32
    inc rdi

    mov rax, r14
    call append_decimal
    mov byte ptr [rdi], 10
    inc rdi

    lea rsi, [rip + output_buffer]
    mov rdx, rdi
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

append_decimal:
    test rax, rax
    jnz decimal_nonzero
    mov byte ptr [rdi], 48
    inc rdi
    ret

decimal_nonzero:
    xor ecx, ecx
    mov r9d, 10

decimal_divide:
    xor edx, edx
    div r9
    push rdx
    inc ecx
    test rax, rax
    jnz decimal_divide

decimal_write:
    pop rax
    add al, 48
    mov byte ptr [rdi], al
    inc rdi
    dec ecx
    jnz decimal_write
    ret

.bss
.align 16
input_buffer:
    .zero 4096
output_buffer:
    .zero 64
