.equ READ_SIZE, 4096

.text
.globl _start
_start:
	mov	x19, #0
	mov	x20, #0
	mov	x21, #0
	mov	x22, #0

read_more:
	mov	x0, #0
	adrp	x1, input_buffer
	add	x1, x1, :lo12:input_buffer
	mov	x2, #READ_SIZE
	mov	x8, #63
	svc	#0
	cbz	x0, input_done

	add	x21, x21, x0
	adrp	x23, input_buffer
	add	x23, x23, :lo12:input_buffer
	add	x24, x23, x0

scan_loop:
	cmp	x23, x24
	b.hs	read_more

	ldrb	w5, [x23], #1
	cmp	w5, #0x20
	b.eq	whitespace
	sub	w6, w5, #9
	cmp	w6, #4
	b.ls	whitespace

	cbnz	x22, scan_loop
	add	x20, x20, #1
	mov	x22, #1
	b	scan_loop

whitespace:
	cmp	w5, #10
	b.ne	1f
	add	x19, x19, #1
1:
	mov	x22, #0
	b	scan_loop

input_done:
	adrp	x1, output_buffer
	add	x1, x1, :lo12:output_buffer

	mov	x0, x19
	bl	append_decimal
	mov	w2, #32
	strb	w2, [x1], #1

	mov	x0, x20
	bl	append_decimal
	mov	w2, #32
	strb	w2, [x1], #1

	mov	x0, x21
	bl	append_decimal
	mov	w2, #10
	strb	w2, [x1], #1

	mov	x2, x1
	adrp	x1, output_buffer
	add	x1, x1, :lo12:output_buffer
	sub	x2, x2, x1
	mov	x0, #1
	mov	x8, #64
	svc	#0

	mov	x0, #0
	mov	x8, #93
	svc	#0

append_decimal:
	stp	x29, x30, [sp, #-64]!
	add	x2, sp, #63
	add	x6, sp, #64
	mov	x3, #10

1:
	udiv	x4, x0, x3
	msub	x5, x4, x3, x0
	add	w5, w5, #48
	strb	w5, [x2], #-1
	mov	x0, x4
	cbnz	x0, 1b

	add	x2, x2, #1
2:
	cmp	x2, x6
	b.hs	3f
	ldrb	w5, [x2], #1
	strb	w5, [x1], #1
	b	2b
3:
	ldp	x29, x30, [sp], #64
	ret

.bss
.balign 16
input_buffer:
.space 4096
.balign 16
output_buffer:
.space 64
