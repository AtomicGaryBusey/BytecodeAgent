.globl _start

.equ ST_NONE, 0
.equ ST_POS, 1
.equ ST_MINUS, 2
.equ ST_OP, 3
.equ ST_BAD, 4
.equ ST_NEG, 5

.text
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    xor ebp, ebp
    xor ebx, ebx
    xor r10d, r10d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, 4096
    syscall
    test rax, rax
    jz at_eof
    mov rdx, rax
    lea rsi, [rip + input_buf]

next_byte:
    test rdx, rdx
    jz read_more
    movzx r8d, byte ptr [rsi]
    inc rsi
    dec rdx

    cmp r8b, 10
    je got_newline

    mov r10d, 1
    test r13d, r13d
    jnz next_byte

    cmp r8b, 32
    je got_separator
    cmp r8b, 9
    je got_separator

    cmp r14d, ST_NONE
    je token_start
    cmp r14d, ST_POS
    je token_pos
    cmp r14d, ST_NEG
    je token_neg
    cmp r14d, ST_MINUS
    je token_minus_more
    cmp r14d, ST_OP
    je token_bad
    jmp next_byte

got_separator:
    call finish_token
    jmp next_byte

got_newline:
    call emit_line
    jmp next_byte

token_start:
    cmp r8b, 45
    je start_minus
    cmp r8b, 48
    jb start_operator
    cmp r8b, 57
    ja start_operator
    movzx r15d, r8b
    sub r15, 48
    mov ebp, 1
    mov r14d, ST_POS
    jmp next_byte

start_minus:
    mov r14d, ST_MINUS
    jmp next_byte

start_operator:
    cmp r8b, 43
    je save_operator
    cmp r8b, 42
    je save_operator
    cmp r8b, 47
    je save_operator
    cmp r8b, 37
    je save_operator
    mov r14d, ST_BAD
    jmp next_byte

save_operator:
    mov ebx, r8d
    mov r14d, ST_OP
    jmp next_byte

token_pos:
    cmp r8b, 48
    jb token_bad
    cmp r8b, 57
    ja token_bad
    inc rbp
    cmp rbp, 18
    ja token_bad
    imul r15, r15, 10
    movzx eax, r8b
    sub eax, 48
    add r15, rax
    jmp next_byte

token_neg:
    cmp r8b, 48
    jb token_bad
    cmp r8b, 57
    ja token_bad
    inc rbp
    cmp rbp, 18
    ja token_bad
    imul r15, r15, 10
    movzx eax, r8b
    sub eax, 48
    add r15, rax
    jmp next_byte

token_minus_more:
    cmp r8b, 48
    jb token_bad
    cmp r8b, 57
    ja token_bad
    movzx r15d, r8b
    sub r15, 48
    mov ebp, 1
    mov r14d, ST_NEG
    jmp next_byte

token_bad:
    mov r14d, ST_BAD
    jmp next_byte

at_eof:
    test r10d, r10d
    jz do_exit
    call emit_line

do_exit:
    mov eax, 60
    xor edi, edi
    syscall

finish_token:
    cmp r14d, ST_NONE
    je finish_done
    cmp r14d, ST_POS
    je finish_push_pos
    cmp r14d, ST_NEG
    je finish_push_neg
    cmp r14d, ST_MINUS
    je finish_subtract
    cmp r14d, ST_OP
    je finish_operator
    mov r13d, 1
    jmp finish_done

finish_push_pos:
    mov qword ptr [vals + r12*8], r15
    inc r12
    jmp finish_done

finish_push_neg:
    neg r15
    mov qword ptr [vals + r12*8], r15
    inc r12
    jmp finish_done

finish_subtract:
    mov ebx, 45

finish_operator:
    cmp r12, 2
    jb finish_error
    mov r8, qword ptr [vals + r12*8 - 8]
    mov r9, qword ptr [vals + r12*8 - 16]

    cmp ebx, 43
    je op_add
    cmp ebx, 45
    je op_sub
    cmp ebx, 42
    je op_mul
    cmp ebx, 47
    je op_div
    cmp ebx, 37
    je op_mod
    jmp finish_error

op_add:
    add r9, r8
    jmp op_store
op_sub:
    sub r9, r8
    jmp op_store
op_mul:
    imul r9, r8
    jmp op_store

op_div:
    test r8, r8
    jz finish_error
    mov rax, r9
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne div_ok
    cmp r8, -1
    je finish_error
div_ok:
    cqo
    idiv r8
    mov r9, rax
    jmp op_store

op_mod:
    test r8, r8
    jz finish_error
    mov rax, r9
    mov rdx, 0x8000000000000000
    cmp rax, rdx
    jne mod_ok
    cmp r8, -1
    je finish_error
mod_ok:
    cqo
    idiv r8
    mov r9, rdx

op_store:
    mov qword ptr [vals + r12*8 - 16], r9
    dec r12
    jmp finish_done

finish_error:
    mov r13d, 1

finish_done:
    xor r14d, r14d
    ret

emit_line:
    test r13d, r13d
    jnz emit_error
    call finish_token
    test r13d, r13d
    jnz emit_error
    cmp r12, 1
    jne emit_error

    mov rax, qword ptr [vals]
    xor ebx, ebx
    test rax, rax
    jns number_abs_ready
    mov ebx, 1
    neg rax

number_abs_ready:
    lea r9, [rip + output_buf + 32]
    mov rsi, r9
    dec rsi
    mov byte ptr [rsi], 10
    mov r8, 10

number_loop:
    xor edx, edx
    div r8
    add dl, 48
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jnz number_loop

    test ebx, ebx
    jz number_write
    dec rsi
    mov byte ptr [rsi], 45

number_write:
    mov rdx, r9
    sub rdx, rsi
    mov eax, 1
    mov edi, 1
    syscall
    jmp emit_reset

emit_error:
    lea rsi, [rip + error_msg]
    mov edx, 6
    mov eax, 1
    mov edi, 1
    syscall

emit_reset:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d
    xor ebp, ebp
    xor ebx, ebx
    xor r10d, r10d
    ret

.section .rodata
error_msg:
    .ascii "error\n"

.bss
.align 16
input_buf:
    .zero 4096
.align 16
vals:
    .zero 4096
.align 16
output_buf:
    .zero 32
