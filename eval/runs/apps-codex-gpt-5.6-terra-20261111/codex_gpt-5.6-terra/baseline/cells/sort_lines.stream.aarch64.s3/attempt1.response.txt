    ldr x1, [x12, x18, lsl #3]

    ldr x1, [x12, x16, lsl #3]

    ldr x0, [x11, x16, lsl #3]
    ldr x1, [x12, x16, lsl #3]

    str x1, [x14, x20, lsl #3]

    ldr x1, [x12, x18, lsl #3]

    mov x11, x13
    mov x13, x0
    mov x0, x12
    mov x12, x14
