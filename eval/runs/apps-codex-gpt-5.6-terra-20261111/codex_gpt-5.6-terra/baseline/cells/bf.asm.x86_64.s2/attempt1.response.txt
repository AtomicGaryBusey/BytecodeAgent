.globl _start

.text
_start:
    lea r12, [rip + inputbuf]
    xor r13d, r13d

read_more:
    cmp r13, 65536
    jae read_done
    mov eax, 0
    xor edi, edi
    lea rsi, [r12 + r13]
    mov edx, 65536
    sub rdx, r13
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    mov r14, r13
    xor ebx, ebx

find_bang:
    cmp rbx, r13
    jae no_bang
    cmp byte ptr [r12 + rbx], 33
    je have_bang
    inc rbx
    jmp find_bang

have_bang:
    mov r14, rbx
    lea r15, [rbx + 1]
    jmp validate_start

no_bang:
    mov r15, r13

validate_start:
    lea r9, [rip + bracketmap]
    lea r10, [rip + bracketstack]
    xor ebx, ebx
    xor r8d, r8d

validate_loop:
    cmp rbx, r14
    jae validate_done
    mov al, byte ptr [r12 + rbx]
    cmp al, 91
    je validate_open
    cmp al, 93
    je validate_close
    inc rbx
    jmp validate_loop

validate_open:
    mov dword ptr [r10 + r8*4], ebx
    inc r8d
    inc rbx
    jmp validate_loop

validate_close:
    test r8d, r8d
    jz bad_program
    dec r8d
    mov edx, dword ptr [r10 + r8*4]
    mov dword ptr [r9 + rbx*4], edx
    mov dword ptr [r9 + rdx*4], ebx
    inc rbx
    jmp validate_loop

validate_done:
    test r8d, r8d
    jnz bad_program

    lea r10, [rip + tape]
    xor ebx, ebx

run_loop:
    cmp rbx, r14
    jae good_exit
    mov al, byte ptr [r12 + rbx]
    inc rbx
    cmp al, 62
    je cmd_right
    cmp al, 60
    je cmd_left
    cmp al, 43
    je cmd_plus
    cmp al, 45
    je cmd_minus
    cmp al, 46
    je cmd_dot
    cmp al, 44
    je cmd_comma
    cmp al, 91
    je cmd_open
    cmp al, 93
    je cmd_close
    jmp run_loop

cmd_right:
    inc r10
    jmp run_loop

cmd_left:
    dec r10
    jmp run_loop

cmd_plus:
    inc byte ptr [r10]
    jmp run_loop

cmd_minus:
    dec byte ptr [r10]
    jmp run_loop

cmd_dot:
    mov al, byte ptr [r10]
    mov byte ptr [rip + outbyte], al
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + outbyte]
    mov edx, 1
    syscall
    jmp run_loop

cmd_comma:
    cmp r15, r13
    jae comma_zero
    mov al, byte ptr [r12 + r15]
    inc r15
    mov byte ptr [r10], al
    jmp run_loop

comma_zero:
    mov byte ptr [r10], 0
    jmp run_loop

cmd_open:
    cmp byte ptr [r10], 0
    jne run_loop
    lea rax, [rbx - 1]
    mov ebx, dword ptr [r9 + rax*4]
    inc rbx
    jmp run_loop

cmd_close:
    cmp byte ptr [r10], 0
    je run_loop
    lea rax, [rbx - 1]
    mov ebx, dword ptr [r9 + rax*4]
    inc rbx
    jmp run_loop

bad_program:
    mov eax, 60
    mov edi, 1
    syscall

good_exit:
    mov eax, 60
    xor edi, edi
    syscall

.bss
.align 16
inputbuf:
    .zero 65536
.align 16
bracketmap:
    .zero 262144
.align 16
bracketstack:
    .zero 4000
.align 16
tape:
    .zero 30000
outbyte:
    .zero 1
