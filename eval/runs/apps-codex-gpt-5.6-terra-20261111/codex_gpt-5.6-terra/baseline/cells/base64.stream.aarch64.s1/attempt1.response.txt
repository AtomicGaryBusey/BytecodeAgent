.text
.globl _start

_start:
    adrp x25, outbuf
    add x25, x25, :lo12:outbuf
    adrp x22, inbuf
    add x22, x22, :lo12:inbuf
    mov x19, xzr
    movz x26, #0xc000

read_loop:
    cmp x19, x26
    b.eq read_done
    adrp x1, inbuf
    add x1, x1, :lo12:inbuf
    add x1, x1, x19
    sub x2, x26, x19
    mov x0, xzr
    movz x8, #63
    svc #0
    cbz x0, read_done
    add x19, x19, x0
    b read_loop

read_done:
    cbz x19, exit_ok

    add x23, x22, x19
    mov x20, x25
    mov w21, wzr
    adrp x24, alphabet
    add x24, x24, :lo12:alphabet

full_group_loop:
    sub x0, x23, x22
    cmp x0, #3
    b.lo final_group

    ldrb w0, [x22], #1
    ldrb w1, [x22], #1
    ldrb w2, [x22], #1

    lsr w3, w0, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    and w3, w0, #3
    lsl w3, w3, #4
    lsr w4, w1, #4
    orr w3, w3, w4
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    and w3, w1, #15
    lsl w3, w3, #2
    lsr w4, w2, #6
    orr w3, w3, w4
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    and w3, w2, #63
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    add w21, w21, #4
    cmp w21, #76
    b.ne full_group_loop
    mov w3, #10
    strb w3, [x20], #1
    mov w21, wzr
    b full_group_loop

final_group:
    cmp x22, x23
    b.eq finish_lines

    ldrb w0, [x22], #1
    lsr w3, w0, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    and w3, w0, #3
    lsl w3, w3, #4

    cmp x22, x23
    b.eq final_one

    ldrb w1, [x22], #1
    lsr w4, w1, #4
    orr w3, w3, w4
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    and w3, w1, #15
    lsl w3, w3, #2
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1

    mov w3, #61
    strb w3, [x20], #1
    b final_count

final_one:
    ldrb w3, [x24, w3, uxtw]
    strb w3, [x20], #1
    mov w3, #61
    strb w3, [x20], #1
    strb w3, [x20], #1

final_count:
    add w21, w21, #4
    cmp w21, #76
    b.ne finish_lines
    mov w3, #10
    strb w3, [x20], #1
    mov w21, wzr

finish_lines:
    cbz w21, write_output
    mov w3, #10
    strb w3, [x20], #1

write_output:
    sub x2, x20, x25
    mov x0, #1
    mov x1, x25
    movz x8, #64
    svc #0

exit_ok:
    mov x0, xzr
    movz x8, #93
    svc #0

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.balign 16

.balign 16
