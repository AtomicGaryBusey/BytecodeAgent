.globl _start

.text
_start:
    lea r12, [rip + input_buf]
    xor ebx, ebx

read_loop:
    cmp rbx, 65536
    je read_probe
    mov rdx, 65536
    sub rdx, rbx
    lea rsi, [r12 + rbx]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_probe:
    lea rsi, [r12 + rbx]
    mov edx, 1
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r14, [r12 + rbx]
    mov r8, r12
    mov rax, r12

find_bang:
    cmp rax, r14
    je no_bang
    cmp byte ptr [rax], '!'
    je got_bang
    inc rax
    jmp find_bang

got_bang:
    mov r12, rax
    lea r13, [rax + 1]
    jmp validate_start

no_bang:
    mov r12, r14
    mov r13, r14

validate_start:
    lea r9, [rip + match_table]
    lea r10, [rip + bracket_stack]
    xor r15d, r15d
    mov rbx, r8

validate_loop:
    cmp rbx, r12
    je validate_done
    movzx eax, byte ptr [rbx]
    cmp al, '['
    je validate_open
    cmp al, ']'
    je validate_close
    inc rbx
    jmp validate_loop

validate_open:
    mov qword ptr [r10 + r15*8], rbx
    inc r15
    inc rbx
    jmp validate_loop

validate_close:
    test r15, r15
    jz bad_exit
    dec r15
    mov rdx, qword ptr [r10 + r15*8]
    lea rcx, [rbx + 1]
    mov rax, rdx
    sub rax, r8
    mov qword ptr [r9 + rax*8], rcx
    lea rcx, [rdx + 1]
    mov rax, rbx
    sub rax, r8
    mov qword ptr [r9 + rax*8], rcx
    inc rbx
    jmp validate_loop

validate_done:
    test r15, r15
    jnz bad_exit

    lea r15, [rip + tape]
    lea rbp, [rip + output_buf]
    lea r10, [rip + output_buf + 65536]
    mov rbx, r8

run_loop:
    cmp rbx, r12
    je finish_output
    movzx eax, byte ptr [rbx]
    inc rbx
    cmp al, '>'
    je cmd_right
    cmp al, '<'
    je cmd_left
    cmp al, '+'
    je cmd_plus
    cmp al, '-'
    je cmd_minus
    cmp al, '.'
    je cmd_dot
    cmp al, ','
    je cmd_comma
    cmp al, '['
    je cmd_open
    cmp al, ']'
    je cmd_close
    jmp run_loop

cmd_right:
    inc r15
    jmp run_loop

cmd_left:
    dec r15
    jmp run_loop

cmd_plus:
    inc byte ptr [r15]
    jmp run_loop

cmd_minus:
    dec byte ptr [r15]
    jmp run_loop

cmd_dot:
    mov al, byte ptr [r15]
    mov byte ptr [rbp], al
    inc rbp
    cmp rbp, r10
    jne run_loop
    mov edi, 1
    lea rsi, [rip + output_buf]
    mov edx, 65536
    mov eax, 1
    syscall
    lea rbp, [rip + output_buf]
    jmp run_loop

cmd_comma:
    cmp r13, r14
    je comma_zero
    mov al, byte ptr [r13]
    inc r13
    mov byte ptr [r15], al
    jmp run_loop

comma_zero:
    mov byte ptr [r15], 0
    jmp run_loop

cmd_open:
    cmp byte ptr [r15], 0
    jne run_loop
    mov rax, rbx
    sub rax, r8
    dec rax
    mov rbx, qword ptr [r9 + rax*8]
    jmp run_loop

cmd_close:
    cmp byte ptr [r15], 0
    je run_loop
    mov rax, rbx
    sub rax, r8
    dec rax
    mov rbx, qword ptr [r9 + rax*8]
    jmp run_loop

finish_output:
    mov rdx, rbp
    lea rsi, [rip + output_buf]
    sub rdx, rsi
    jz good_exit
    mov edi, 1
    mov eax, 1
    syscall

good_exit:
    xor edi, edi
    mov eax, 60
    syscall

bad_exit:
    mov edi, 1
    mov eax, 60
    syscall

.bss
.align 8
input_buf:
    .zero 65537
.align 8
output_buf:
    .zero 65536
.align 8
tape:
    .zero 30000
.align 8
match_table:
    .zero 524288
.align 8
bracket_stack:
    .zero 8000
