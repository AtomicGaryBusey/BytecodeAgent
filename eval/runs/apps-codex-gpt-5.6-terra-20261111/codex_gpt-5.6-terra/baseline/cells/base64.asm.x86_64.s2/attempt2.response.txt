.text
.globl _start
_start:
    xor ebx, ebx

read_more:
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 49152
    sub edx, ebx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz begin_encode
    add rbx, rax
    jmp read_more

begin_encode:
    test rbx, rbx
    jz exit_ok
    lea r12, [rip + input_buffer]
    lea r13, [r12 + rbx]
    lea r14, [rip + line_buffer]
    lea r15, [rip + alphabet]
    xor r8d, r8d

full_group:
    mov rax, r13
    sub rax, r12
    cmp rax, 3
    jb final_group

    movzx eax, byte ptr [r12]
    movzx edx, byte ptr [r12 + 1]
    movzx ecx, byte ptr [r12 + 2]

    mov ebx, eax
    shr ebx, 2
    mov bl, byte ptr [r15 + rbx]
    mov byte ptr [r14 + r8], bl

    and eax, 3
    shl eax, 4
    mov ebx, edx
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r14 + r8 + 1], al

    and edx, 15
    shl edx, 2
    mov ebx, ecx
    shr ebx, 6
    or edx, ebx
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14 + r8 + 2], dl

    and ecx, 63
    mov cl, byte ptr [r15 + rcx]
    mov byte ptr [r14 + r8 + 3], cl

    add r12, 3
    add r8d, 4
    cmp r8d, 76
    jne full_group

    mov byte ptr [r14 + 76], 10
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    mov edx, 77
    syscall
    xor r8d, r8d
    jmp full_group

final_group:
    mov rax, r13
    sub rax, r12
    test rax, rax
    jz flush_last
    cmp rax, 1
    je one_byte

    movzx eax, byte ptr [r12]
    movzx edx, byte ptr [r12 + 1]

    mov ebx, eax
    shr ebx, 2
    mov bl, byte ptr [r15 + rbx]
    mov byte ptr [r14 + r8], bl

    and eax, 3
    shl eax, 4
    mov ebx, edx
    shr ebx, 4
    or eax, ebx
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r14 + r8 + 1], al

    and edx, 15
    shl edx, 2
    mov dl, byte ptr [r15 + rdx]
    mov byte ptr [r14 + r8 + 2], dl
    mov byte ptr [r14 + r8 + 3], '='
    add r8d, 4
    jmp flush_last

one_byte:
    movzx eax, byte ptr [r12]

    mov ebx, eax
    shr ebx, 2
    mov bl, byte ptr [r15 + rbx]
    mov byte ptr [r14 + r8], bl

    and eax, 3
    shl eax, 4
    mov al, byte ptr [r15 + rax]
    mov byte ptr [r14 + r8 + 1], al
    mov byte ptr [r14 + r8 + 2], '='
    mov byte ptr [r14 + r8 + 3], '='
    add r8d, 4

flush_last:
    test r8d, r8d
    jz exit_ok
    mov byte ptr [r14 + r8], 10
    mov edx, r8d
    add edx, 1
    mov eax, 1
    mov edi, 1
    mov rsi, r14
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero 49152
.align 16
line_buffer:
    .zero 77
