.text
.globl _start
_start:
    xor ebx, ebx
read_more:
    lea rsi, [rip + input_buffer]
    add rsi, rbx
    mov edx, 49152
    sub edx, ebx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz begin_encode
    add rbx, rax
    jmp read_more

begin_encode:
    test rbx, rbx
    jz exit_ok
    lea rsi, [rip + input_buffer]
    lea rbp, [rsi + rbx]
    lea r9, [rip + line_buffer]
    lea r10, [rip + alphabet]
    xor r8d, r8d

full_group:
    mov rax, rbp
    sub rax, rsi
    cmp rax, 3
    jb final_group

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]
    movzx ecx, byte ptr [rsi + 2]

    mov r11d, eax
    shr r11d, 2
    mov r11b, byte ptr [r10 + r11]
    mov byte ptr [r9 + r8], r11b

    and eax, 3
    shl eax, 4
    mov r11d, edx
    shr r11d, 4
    or eax, r11d
    mov al, byte ptr [r10 + rax]
    mov byte ptr [r9 + r8 + 1], al

    and edx, 15
    shl edx, 2
    mov r11d, ecx
    shr r11d, 6
    or edx, r11d
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [r9 + r8 + 2], dl

    and ecx, 63
    mov cl, byte ptr [r10 + rcx]
    mov byte ptr [r9 + r8 + 3], cl

    add rsi, 3
    add r8d, 4
    cmp r8d, 76
    jne full_group

    mov byte ptr [r9 + 76], 10
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    mov edx, 77
    syscall
    xor r8d, r8d
    jmp full_group

final_group:
    mov rax, rbp
    sub rax, rsi
    test rax, rax
    jz flush_last
    cmp rax, 1
    je one_byte

    movzx eax, byte ptr [rsi]
    movzx edx, byte ptr [rsi + 1]

    mov r11d, eax
    shr r11d, 2
    mov r11b, byte ptr [r10 + r11]
    mov byte ptr [r9 + r8], r11b

    and eax, 3
    shl eax, 4
    mov r11d, edx
    shr r11d, 4
    or eax, r11d
    mov al, byte ptr [r10 + rax]
    mov byte ptr [r9 + r8 + 1], al

    and edx, 15
    shl edx, 2
    mov dl, byte ptr [r10 + rdx]
    mov byte ptr [r9 + r8 + 2], dl
    mov byte ptr [r9 + r8 + 3], '='
    add r8d, 4
    jmp flush_last

one_byte:
    movzx eax, byte ptr [rsi]
    mov r11d, eax
    shr r11d, 2
    mov r11b, byte ptr [r10 + r11]
    mov byte ptr [r9 + r8], r11b

    and eax, 3
    shl eax, 4
    mov al, byte ptr [r10 + rax]
    mov byte ptr [r9 + r8 + 1], al
    mov byte ptr [r9 + r8 + 2], '='
    mov byte ptr [r9 + r8 + 3], '='
    add r8d, 4

flush_last:
    test r8d, r8d
    jz exit_ok
    mov byte ptr [r9 + r8], 10
    lea edx, [r8 + 1]
    mov eax, 1
    mov edi, 1
    mov rsi, r9
    syscall

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

.section .rodata
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
input_buffer:
    .zero 49152
.align 16
line_buffer:
    .zero 77
