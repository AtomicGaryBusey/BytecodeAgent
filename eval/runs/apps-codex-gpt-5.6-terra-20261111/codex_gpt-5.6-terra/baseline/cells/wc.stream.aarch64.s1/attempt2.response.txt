.text
.globl _start

_start:
	mov x19, #0
	mov x20, #0
	mov x21, #0
	mov x22, #0

read_loop:
	mov x0, #0
	adrp x1, input_buf
	add x1, x1, :lo12:input_buf
	mov x2, #4096
	mov x8, #63
	svc #0
	cbz x0, input_done

	add x21, x21, x0
	mov x24, x0
	adrp x23, input_buf
	add x23, x23, :lo12:input_buf

scan_loop:
	cbz x24, read_loop
	ldrb w4, [x23], #1
	sub x24, x24, #1

	cmp w4, #10
	b.ne not_newline
	add x19, x19, #1

not_newline:
	cmp w4, #32
	b.eq whitespace
	sub w5, w4, #9
	cmp w5, #4
	b.ls whitespace

	cbz x22, new_word
	b scan_loop

new_word:
	add x20, x20, #1
	mov x22, #1
	b scan_loop

whitespace:
	mov x22, #0
	b scan_loop

input_done:
	adrp x25, output_buf
	add x25, x25, :lo12:output_buf
	mov x1, x25

	mov x0, x19
	bl append_decimal
	mov w3, #32
	strb w3, [x1], #1

	mov x0, x20
	bl append_decimal
	mov w3, #32
	strb w3, [x1], #1

	mov x0, x21
	bl append_decimal
	mov w3, #10
	strb w3, [x1], #1

	sub x2, x1, x25
	mov x0, #1
	mov x1, x25
	mov x8, #64
	svc #0

	mov x0, #0
	mov x8, #93
	svc #0

append_decimal:
	sub sp, sp, #32
	mov x7, sp
	mov x2, sp
	mov x4, #10
	cbnz x0, decimal_loop

	mov w3, #48
	strb w3, [x1], #1
	add sp, sp, #32
	ret

decimal_loop:
	udiv x5, x0, x4
	msub x6, x5, x4, x0
	add w6, w6, #48
	strb w6, [x2], #1
	mov x0, x5
	cbnz x0, decimal_loop

decimal_copy:
	sub x2, x2, #1
	ldrb w3, [x2]
	strb w3, [x1], #1
	cmp x2, x7
	b.ne decimal_copy

	add sp, sp, #32
	ret

.bss
.balign 16
input_buf:
.space 4096
.balign 16
output_buf:
.space 64
