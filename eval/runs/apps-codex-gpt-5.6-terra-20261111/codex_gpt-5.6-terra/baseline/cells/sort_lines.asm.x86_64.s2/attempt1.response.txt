.text
.globl _start
_start:
    lea r12, [rip + input_buffer]
    xor r13d, r13d

read_more:
    mov rdx, 32768
    sub rdx, r13
    lea rsi, [r12 + r13]
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add r13, rax
    jmp read_more

read_done:
    lea r13, [r12 + r13]
    mov r14, r12
    mov rsi, r12
    xor ebp, ebp

parse_loop:
    cmp rsi, r13
    je parse_end
    movzx eax, byte ptr [rsi]
    cmp eax, 10
    jne parse_next

    mov rcx, rsi
    sub rcx, r14
    mov rax, rbp
    shl rax, 4
    lea rdx, [rip + records]
    mov qword ptr [rdx + rax], r14
    mov qword ptr [rdx + rax + 8], rcx
    inc rbp
    lea r14, [rsi + 1]

parse_next:
    inc rsi
    jmp parse_loop

parse_end:
    cmp r14, r13
    je sort_begin
    mov rcx, r13
    sub rcx, r14
    mov rax, rbp
    shl rax, 4
    lea rdx, [rip + records]
    mov qword ptr [rdx + rax], r14
    mov qword ptr [rdx + rax + 8], rcx
    inc rbp

sort_begin:
    lea rbx, [rip + records]
    mov r15, 1

sort_outer:
    cmp r15, rbp
    jae output_begin

    mov rax, r15
    shl rax, 4
    mov r8, qword ptr [rbx + rax]
    mov r9, qword ptr [rbx + rax + 8]
    mov r14, r15

sort_inner:
    test r14, r14
    jz sort_insert

    mov rax, r14
    dec rax
    shl rax, 4
    mov rdi, qword ptr [rbx + rax]
    mov rsi, qword ptr [rbx + rax + 8]
    mov rdx, r8
    mov rcx, r9
    call compare_lines
    test eax, eax
    jle sort_insert

    mov rax, r14
    shl rax, 4
    mov r10, r14
    dec r10
    shl r10, 4
    mov r11, qword ptr [rbx + r10]
    mov qword ptr [rbx + rax], r11
    mov r11, qword ptr [rbx + r10 + 8]
    mov qword ptr [rbx + rax + 8], r11
    dec r14
    jmp sort_inner

sort_insert:
    mov rax, r14
    shl rax, 4
    mov qword ptr [rbx + rax], r8
    mov qword ptr [rbx + rax + 8], r9
    inc r15
    jmp sort_outer

output_begin:
    xor r15d, r15

output_loop:
    cmp r15, rbp
    jae done

    mov rax, r15
    shl rax, 4
    mov rsi, qword ptr [rbx + rax]
    mov rdx, qword ptr [rbx + rax + 8]
    test rdx, rdx
    jz output_newline
    mov edi, 1
    mov eax, 1
    syscall

output_newline:
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall

    inc r15
    jmp output_loop

done:
    xor edi, edi
    mov eax, 60
    syscall

compare_lines:
    mov r10, rsi
    cmp r10, rcx
    cmova r10, rcx
    xor r11d, r11d

compare_loop:
    cmp r11, r10
    jae compare_length
    movzx eax, byte ptr [rdi + r11]
    cmp al, byte ptr [rdx + r11]
    jb compare_less
    ja compare_greater
    inc r11
    jmp compare_loop

compare_length:
    cmp rsi, rcx
    jb compare_less
    ja compare_greater
    xor eax, eax
    ret

compare_less:
    mov eax, -1
    ret

compare_greater:
    mov eax, 1
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
input_buffer:
    .zero 32768
.align 16
records:
    .zero 16000
