.equ SYS_read, 63
.equ SYS_write, 64
.equ SYS_exit, 93

.text
.globl _start
_start:
    adrp x20, input_buf
    add x20, x20, :lo12:input_buf
    mov x21, x20
    movz x22, #1, lsl #16
    add x22, x22, #1
read_loop:
    mov x0, #0
    mov x1, x21
    mov x2, x22
    mov x8, #SYS_read
    svc #0
    cbz x0, read_done
    add x21, x21, x0
    sub x22, x22, x0
    b read_loop

read_done:
    mov x0, x20
    mov x1, x21
    bl skip_ws
    cmp x0, x1
    b.hs fail
    bl parse_value
    bl skip_ws
    cmp x0, x1
    b.ne fail

    mov x0, #1
    adrp x1, ok_msg
    add x1, x1, :lo12:ok_msg
    mov x2, #3
    mov x8, #SYS_write
    svc #0
    mov x0, #0
    mov x8, #SYS_exit
    svc #0

fail:
    sub x3, x0, x20
    mov x0, #1
    adrp x1, err_msg
    add x1, x1, :lo12:err_msg
    mov x2, #9
    mov x8, #SYS_write
    svc #0

    adrp x4, out_buf
    add x4, x4, :lo12:out_buf
    add x4, x4, #32
    mov x5, #10
    cbnz x3, decimal_loop
    sub x4, x4, #1
    mov w6, #'0'
    strb w6, [x4]
    b decimal_done
decimal_loop:
    udiv x6, x3, x5
    msub x7, x6, x5, x3
    add w7, w7, #'0'
    sub x4, x4, #1
    strb w7, [x4]
    mov x3, x6
    cbnz x3, decimal_loop
decimal_done:
    adrp x7, out_buf
    add x7, x7, :lo12:out_buf
    add x7, x7, #32
    sub x2, x7, x4
    mov x0, #1
    mov x1, x4
    mov x8, #SYS_write
    svc #0

    mov x0, #1
    adrp x1, nl_msg
    add x1, x1, :lo12:nl_msg
    mov x2, #1
    mov x8, #SYS_write
    svc #0
    mov x0, #1
    mov x8, #SYS_exit
    svc #0

skip_ws:
    cmp x0, x1
    b.hs 2f
1:
    ldrb w2, [x0]
    cmp w2, #0x20
    b.eq 3f
    cmp w2, #0x09
    b.eq 3f
    cmp w2, #0x0a
    b.eq 3f
    cmp w2, #0x0d
    b.ne 2f
3:
    add x0, x0, #1
    cmp x0, x1
    b.lo 1b
2:
    ret

parse_value:
    stp x29, x30, [sp, #-16]!
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'{'
    b.eq pv_object
    cmp w2, #'['
    b.eq pv_array
    cmp w2, #'"'
    b.eq pv_string
    cmp w2, #'-'
    b.eq pv_number
    cmp w2, #'0'
    b.eq pv_number
    cmp w2, #'1'
    b.lo pv_other
    cmp w2, #'9'
    b.ls pv_number
pv_other:
    cmp w2, #'t'
    b.eq pv_true
    cmp w2, #'f'
    b.eq pv_false
    cmp w2, #'n'
    b.eq pv_null
    b fail
pv_object:
    bl parse_object
    b pv_done
pv_array:
    bl parse_array
    b pv_done
pv_string:
    bl parse_string
    b pv_done
pv_number:
    bl parse_number
    b pv_done
pv_true:
    bl parse_true
    b pv_done
pv_false:
    bl parse_false
    b pv_done
pv_null:
    bl parse_null
pv_done:
    ldp x29, x30, [sp], #16
    ret

parse_true:
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'t'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'r'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'u'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'e'
    b.ne fail
    add x0, x0, #1
    ret

parse_false:
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'f'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'a'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'l'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'s'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'e'
    b.ne fail
    add x0, x0, #1
    ret

parse_null:
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'n'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'u'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'l'
    b.ne fail
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'l'
    b.ne fail
    add x0, x0, #1
    ret

parse_number:
    ldrb w2, [x0]
    cmp w2, #'-'
    b.ne 1f
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
1:
    ldrb w2, [x0]
    cmp w2, #'0'
    b.eq 3f
    cmp w2, #'1'
    b.lo fail
    cmp w2, #'9'
    b.hi fail
2:
    add x0, x0, #1
    cmp x0, x1
    b.hs 4f
    ldrb w2, [x0]
    cmp w2, #'0'
    b.lo 4f
    cmp w2, #'9'
    b.hi 4f
    b 2b
3:
    add x0, x0, #1
4:
    cmp x0, x1
    b.hs 8f
    ldrb w2, [x0]
    cmp w2, #'.'
    b.ne 6f
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'0'
    b.lo fail
    cmp w2, #'9'
    b.hi fail
5:
    add x0, x0, #1
    cmp x0, x1
    b.hs 8f
    ldrb w2, [x0]
    cmp w2, #'0'
    b.lo 6f
    cmp w2, #'9'
    b.ls 5b
6:
    cmp x0, x1
    b.hs 8f
    ldrb w2, [x0]
    cmp w2, #'e'
    b.eq 7f
    cmp w2, #'E'
    b.ne 8f
7:
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'+'
    b.eq 9f
    cmp w2, #'-'
    b.ne 10f
9:
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
10:
    ldrb w2, [x0]
    cmp w2, #'0'
    b.lo fail
    cmp w2, #'9'
    b.hi fail
11:
    add x0, x0, #1
    cmp x0, x1
    b.hs 8f
    ldrb w2, [x0]
    cmp w2, #'0'
    b.lo 8f
    cmp w2, #'9'
    b.ls 11b
8:
    ret

parse_string:
    add x0, x0, #1
1:
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'"'
    b.eq 8f
    cmp w2, #'\\'
    b.eq 2f
    cmp w2, #0x20
    b.lo fail
    add x0, x0, #1
    b 1b
2:
    add x0, x0, #1
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'"'
    b.eq 7f
    cmp w2, #'\\'
    b.eq 7f
    cmp w2, #'/'
    b.eq 7f
    cmp w2, #'b'
    b.eq 7f
    cmp w2, #'f'
    b.eq 7f
    cmp w2, #'n'
    b.eq 7f
    cmp w2, #'r'
    b.eq 7f
    cmp w2, #'t'
    b.eq 7f
    cmp w2, #'u'
    b.ne fail
    add x0, x0, #1
    mov w3, #4
3:
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'0'
    b.lo 4f
    cmp w2, #'9'
    b.ls 6f
4:
    cmp w2, #'a'
    b.lo 5f
    cmp w2, #'f'
    b.ls 6f
5:
    cmp w2, #'A'
    b.lo fail
    cmp w2, #'F'
    b.hi fail
6:
    add x0, x0, #1
    subs w3, w3, #1
    b.ne 3b
    b 1b
7:
    add x0, x0, #1
    b 1b
8:
    add x0, x0, #1
    ret

parse_array:
    stp x29, x30, [sp, #-16]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #']'
    b.eq 4f
1:
    bl parse_value
    bl skip_ws
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #']'
    b.eq 4f
    cmp w2, #','
    b.ne fail
    add x0, x0, #1
    bl skip_ws
    b 1b
4:
    add x0, x0, #1
    ldp x29, x30, [sp], #16
    ret

parse_object:
    stp x29, x30, [sp, #-16]!
    add x0, x0, #1
    bl skip_ws
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'}'
    b.eq 5f
1:
    cmp w2, #'"'
    b.ne fail
    bl parse_string
    bl skip_ws
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #':'
    b.ne fail
    add x0, x0, #1
    bl skip_ws
    bl parse_value
    bl skip_ws
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    cmp w2, #'}'
    b.eq 5f
    cmp w2, #','
    b.ne fail
    add x0, x0, #1
    bl skip_ws
    cmp x0, x1
    b.hs fail
    ldrb w2, [x0]
    b 1b
5:
    add x0, x0, #1
    ldp x29, x30, [sp], #16
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_msg:
    .ascii "error at "
nl_msg:
    .ascii "\n"

.bss
.balign 16
input_buf:
    .zero 65537
.balign 16
out_buf:
    .zero 64
