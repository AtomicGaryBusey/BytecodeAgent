.equ READ_SIZE, 4096

.text
.globl _start

_start:
    mov x19, xzr
    mov x20, xzr
    mov x21, xzr
    mov x22, xzr

.Lread:
    mov x0, xzr
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    mov x2, #READ_SIZE
    mov x8, #63
    svc #0

    cbz x0, .Ldone_read

    add x21, x21, x0
    adrp x1, input_buf
    add x1, x1, :lo12:input_buf
    add x2, x1, x0

.Lscan:
    ldrb w3, [x1], #1

    cmp w3, #10
    b.ne .Lnot_newline
    add x19, x19, #1
    b .Lwhitespace

.Lnot_newline:
    sub w4, w3, #9
    cmp w4, #4
    b.ls .Lwhitespace
    cmp w3, #32
    b.eq .Lwhitespace

    cbnz x22, .Lscan_next
    add x20, x20, #1
    mov x22, #1
    b .Lscan_next

.Lwhitespace:
    mov x22, xzr

.Lscan_next:
    cmp x1, x2
    b.ne .Lscan
    b .Lread

.Ldone_read:
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf

    mov x0, x19
    bl append_decimal
    mov w3, #32
    strb w3, [x1], #1

    mov x0, x20
    bl append_decimal
    mov w3, #32
    strb w3, [x1], #1

    mov x0, x21
    bl append_decimal
    mov w3, #10
    strb w3, [x1], #1

    mov x2, x1
    adrp x1, output_buf
    add x1, x1, :lo12:output_buf
    sub x2, x2, x1
    mov x0, #1
    mov x8, #64
    svc #0

    mov x0, xzr
    mov x8, #93
    svc #0

append_decimal:
    sub sp, sp, #32
    mov x2, #10
    mov x3, xzr

    cbnz x0, .Lad_extract
    mov w5, #48
    strb w5, [x1], #1
    add sp, sp, #32
    ret

.Lad_extract:
    udiv x4, x0, x2
    msub x5, x4, x2, x0
    add w5, w5, #48
    strb w5, [sp, x3]
    add x3, x3, #1
    mov x0, x4
    cbnz x0, .Lad_extract

.Lad_emit:
    subs x3, x3, #1
    ldrb w5, [sp, x3]
    strb w5, [x1], #1
    cbnz x3, .Lad_emit

    add sp, sp, #32
    ret

.bss
.align 4
input_buf:
    .zero 4096

.align 4
output_buf:
    .zero 64
