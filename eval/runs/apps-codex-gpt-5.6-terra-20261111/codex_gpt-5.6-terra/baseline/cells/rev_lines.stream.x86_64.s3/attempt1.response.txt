.text
.globl _start

_start:
    lea rbx, [rip + linebuf]
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + inbuf]
    mov edx, 4096
    syscall
    test rax, rax
    jz end_input

    lea r13, [rip + inbuf]
    lea r14, [r13 + rax]

process_byte:
    cmp r13, r14
    je read_more
    mov al, byte ptr [r13]
    inc r13
    cmp al, 10
    je finish_line
    mov byte ptr [rbx + r12], al
    inc r12
    jmp process_byte

finish_line:
    call output_line
    xor r12d, r12d
    jmp process_byte

end_input:
    test r12, r12
    jz exit_ok
    call output_line

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

output_line:
    test r12, r12
    jz write_newline
    lea r8, [rbx]
    lea r9, [rbx + r12 - 1]

reverse_loop:
    cmp r8, r9
    jae reversed
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_loop

reversed:
    mov eax, 1
    mov edi, 1
    mov rsi, rbx
    mov rdx, r12
    syscall

write_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
linebuf:
    .zero 65536
.align 16
inbuf:
    .zero 4096
