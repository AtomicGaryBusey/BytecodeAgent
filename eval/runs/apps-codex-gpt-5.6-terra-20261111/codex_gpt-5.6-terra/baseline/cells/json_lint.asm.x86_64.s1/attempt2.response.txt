.equ ERRPRELEN, 9
.equ OKLEN, 3

.text
.globl _start
_start:
    xor ebx, ebx
read_loop:
    lea rsi, [rip + input_buf]
    add rsi, rbx
    mov edx, 65537
    sub rdx, rbx
    xor edi, edi
    xor eax, eax
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop

read_done:
    lea r12, [rip + input_buf]
    lea r13, [r12 + rbx]
    call skip_ws
    cmp r12, r13
    je parse_fail
    call parse_value
    call skip_ws
    cmp r12, r13
    jne parse_fail

    mov eax, 1
    mov edi, 1
    lea rsi, [rip + ok_msg]
    mov edx, OKLEN
    syscall
    mov eax, 60
    xor edi, edi
    syscall

parse_fail:
    mov rax, r12
    lea rcx, [rip + input_buf]
    sub rax, rcx
    lea rsi, [rip + out_buf + 31]
    mov byte ptr [rsi], 10
    test rax, rax
    jne error_digits
    dec rsi
    mov byte ptr [rsi], '0'
    jmp error_write

error_digits:
    mov r9, 10
error_digit_loop:
    xor edx, edx
    div r9
    add dl, '0'
    dec rsi
    mov byte ptr [rsi], dl
    test rax, rax
    jne error_digit_loop

error_write:
    mov r8, rsi
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + err_prefix]
    mov edx, ERRPRELEN
    syscall
    mov eax, 1
    mov edi, 1
    mov rsi, r8
    lea rdx, [rip + out_buf + 32]
    sub rdx, rsi
    syscall
    mov eax, 60
    mov edi, 1
    syscall

skip_ws:
skip_ws_loop:
    cmp r12, r13
    je skip_ws_done
    movzx eax, byte ptr [r12]
    cmp al, 32
    je skip_ws_take
    cmp al, 9
    je skip_ws_take
    cmp al, 10
    je skip_ws_take
    cmp al, 13
    jne skip_ws_done
skip_ws_take:
    inc r12
    jmp skip_ws_loop
skip_ws_done:
    ret

parse_value:
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '{'
    je parse_object
    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, '-'
    je parse_number
    cmp al, '0'
    jb parse_value_other
    cmp al, '9'
    jbe parse_number
parse_value_other:
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    jmp parse_fail

parse_true:
    lea rsi, [rip + lit_true]
    mov ecx, 4
    jmp parse_literal
parse_false:
    lea rsi, [rip + lit_false]
    mov ecx, 5
    jmp parse_literal
parse_null:
    lea rsi, [rip + lit_null]
    mov ecx, 4

parse_literal:
    test ecx, ecx
    jz parse_literal_done
    cmp r12, r13
    je parse_fail
    mov al, byte ptr [r12]
    cmp al, byte ptr [rsi]
    jne parse_fail
    inc r12
    inc rsi
    dec ecx
    jmp parse_literal
parse_literal_done:
    ret

parse_string:
    inc r12
parse_string_loop:
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je parse_string_end
    cmp al, 92
    je parse_string_escape
    cmp al, 32
    jb parse_fail
    inc r12
    jmp parse_string_loop

parse_string_end:
    inc r12
    ret

parse_string_escape:
    inc r12
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '"'
    je parse_escape_simple
    cmp al, 92
    je parse_escape_simple
    cmp al, '/'
    je parse_escape_simple
    cmp al, 'b'
    je parse_escape_simple
    cmp al, 'f'
    je parse_escape_simple
    cmp al, 'n'
    je parse_escape_simple
    cmp al, 'r'
    je parse_escape_simple
    cmp al, 't'
    je parse_escape_simple
    cmp al, 'u'
    jne parse_fail
    inc r12
    mov ecx, 4
parse_unicode_loop:
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_unicode_alpha
    cmp al, '9'
    jbe parse_unicode_ok
parse_unicode_alpha:
    cmp al, 'a'
    jb parse_unicode_upper
    cmp al, 'f'
    jbe parse_unicode_ok
parse_unicode_upper:
    cmp al, 'A'
    jb parse_fail
    cmp al, 'F'
    ja parse_fail
parse_unicode_ok:
    inc r12
    dec ecx
    jne parse_unicode_loop
    jmp parse_string_loop

parse_escape_simple:
    inc r12
    jmp parse_string_loop

parse_number:
    cmp byte ptr [r12], '-'
    jne parse_number_int
    inc r12
    cmp r12, r13
    je parse_fail
parse_number_int:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    je parse_number_zero
    cmp al, '1'
    jb parse_fail
    cmp al, '9'
    ja parse_fail
parse_number_digits:
    inc r12
    cmp r12, r13
    je parse_number_post
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_number_post
    cmp al, '9'
    ja parse_number_post
    jmp parse_number_digits

parse_number_zero:
    inc r12
parse_number_post:
    cmp r12, r13
    je parse_number_done
    movzx eax, byte ptr [r12]
    cmp al, '.'
    je parse_number_fraction
    cmp al, 'e'
    je parse_number_exponent
    cmp al, 'E'
    je parse_number_exponent
parse_number_done:
    ret

parse_number_fraction:
    inc r12
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    ja parse_fail
parse_fraction_digits:
    inc r12
    cmp r12, r13
    je parse_fraction_post
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_fraction_post
    cmp al, '9'
    ja parse_fraction_post
    jmp parse_fraction_digits
parse_fraction_post:
    cmp r12, r13
    je parse_number_done
    movzx eax, byte ptr [r12]
    cmp al, 'e'
    je parse_number_exponent
    cmp al, 'E'
    je parse_number_exponent
    ret

parse_number_exponent:
    inc r12
    cmp r12, r13
    je parse_fail
    movzx eax, byte ptr [r12]
    cmp al, '+'
    je parse_exponent_sign
    cmp al, '-'
    je parse_exponent_sign
    jmp parse_exponent_first
parse_exponent_sign:
    inc r12
    cmp r12, r13
    je parse_fail
parse_exponent_first:
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_fail
    cmp al, '9'
    ja parse_fail
parse_exponent_digits:
    inc r12
    cmp r12, r13
    je parse_number_done
    movzx eax, byte ptr [r12]
    cmp al, '0'
    jb parse_number_done
    cmp al, '9'
    ja parse_number_done
    jmp parse_exponent_digits

parse_array:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ']'
    je parse_array_empty
parse_array_value:
    call parse_value
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ']'
    je parse_array_close
    cmp byte ptr [r12], ','
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    jmp parse_array_value
parse_array_empty:
    inc r12
    ret
parse_array_close:
    inc r12
    ret

parse_object:
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '}'
    je parse_object_empty
parse_object_member:
    cmp byte ptr [r12], '"'
    jne parse_fail
    call parse_string
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], ':'
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    call parse_value
    call skip_ws
    cmp r12, r13
    je parse_fail
    cmp byte ptr [r12], '}'
    je parse_object_close
    cmp byte ptr [r12], ','
    jne parse_fail
    inc r12
    call skip_ws
    cmp r12, r13
    je parse_fail
    jmp parse_object_member
parse_object_empty:
    inc r12
    ret
parse_object_close:
    inc r12
    ret

.section .rodata
ok_msg:
    .ascii "ok\n"
err_prefix:
    .ascii "error at "
lit_true:
    .ascii "true"
lit_false:
    .ascii "false"
lit_null:
    .ascii "null"

.bss
.align 16
input_buf:
    .zero 65537
out_buf:
    .zero 32
