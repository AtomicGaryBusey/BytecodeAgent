    b.hs sort_done
    ldr x24, [x19, w23, uxtw #3]
    ldr x25, [x20, w23, uxtw #3]

    ldr x6, [x19, w9, uxtw #3]
    ldr x7, [x20, w9, uxtw #3]

    mov x11, x7
    mov x12, x24
    mov x13, x25

    cmp x12, x13
    b.eq sort_shift
    ldrb w14, [x10], #1
    ldrb w15, [x12], #1
    cmp w14, w15
    b.eq compare_loop
