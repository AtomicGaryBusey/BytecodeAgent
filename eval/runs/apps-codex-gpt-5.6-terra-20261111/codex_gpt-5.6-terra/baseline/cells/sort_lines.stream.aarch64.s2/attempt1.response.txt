.text
.globl _start

_start:
    adrp x19, input_buf
    add x19, x19, :lo12:input_buf
    mov x20, x19
    movz x2, #0x8000
    add x22, x19, x2

read_loop:
    cmp x20, x22
    b.eq read_done
    sub x2, x22, x20
    movz x0, #0
    mov x1, x20
    movz x8, #63
    svc #0
    cbz x0, read_done
    add x20, x20, x0
    b read_loop

read_done:
    adrp x3, line_starts
    add x3, x3, :lo12:line_starts
    adrp x4, line_ends
    add x4, x4, :lo12:line_ends
    mov x0, x19
    mov x1, x19
    movz w5, #0

parse_loop:
    cmp x1, x20
    b.hs parse_finish
    ldrb w6, [x1], #1
    cmp w6, #10
    b.ne parse_loop
    str x0, [x3, w5, uxtw #3]
    str x1, [x4, w5, uxtw #3]
    sub x7, x1, #1
    str x7, [x4, w5, uxtw #3]
    add w5, w5, #1
    mov x0, x1
    b parse_loop

parse_finish:
    cmp x0, x20
    b.hs parse_done
    str x0, [x3, w5, uxtw #3]
    str x20, [x4, w5, uxtw #3]
    add w5, w5, #1

parse_done:
    mov w22, w5
    movz w23, #1

sort_outer:
    cmp w23, w22
    b.hs sort_done
    ldr x24, [x3, w23, uxtw #3]
    ldr x25, [x4, w23, uxtw #3]
    mov w26, w23

sort_inner:
    cbz w26, sort_place
    sub w9, w26, #1
    ldr x6, [x3, w9, uxtw #3]
    ldr x7, [x4, w9, uxtw #3]
    mov x0, x6
    mov x1, x7
    mov x2, x24
    mov x3, x25

compare_loop:
    cmp x0, x1
    b.eq sort_place
    cmp x2, x3
    b.eq sort_shift
    ldrb w4, [x0], #1
    ldrb w5, [x2], #1
    cmp w4, w5
    b.eq compare_loop
    b.hi sort_shift
    b sort_place

sort_shift:
    adrp x20, line_starts
    add x20, x20, :lo12:line_starts
    adrp x21, line_ends
    add x21, x21, :lo12:line_ends
    str x6, [x20, w26, uxtw #3]
    str x7, [x21, w26, uxtw #3]
    mov w26, w9
    mov x3, x20
    b sort_inner

sort_place:
    adrp x20, line_starts
    add x20, x20, :lo12:line_starts
    adrp x21, line_ends
    add x21, x21, :lo12:line_ends
    str x24, [x20, w26, uxtw #3]
    str x25, [x21, w26, uxtw #3]
    add w23, w23, #1
    adrp x3, line_starts
    add x3, x3, :lo12:line_starts
    adrp x4, line_ends
    add x4, x4, :lo12:line_ends
    b sort_outer

sort_done:
    adrp x20, line_starts
    add x20, x20, :lo12:line_starts
    adrp x21, line_ends
    add x21, x21, :lo12:line_ends
    adrp x27, newline
    add x27, x27, :lo12:newline
    movz w23, #0

output_loop:
    cmp w23, w22
    b.hs exit_program
    ldr x1, [x20, w23, uxtw #3]
    ldr x2, [x21, w23, uxtw #3]
    sub x2, x2, x1
    cbz x2, output_newline
    movz x0, #1
    movz x8, #64
    svc #0

output_newline:
    movz x0, #1
    mov x1, x27
    movz x2, #1
    movz x8, #64
    svc #0
    add w23, w23, #1
    b output_loop

exit_program:
    movz x0, #0
    movz x8, #93
    svc #0

.section .rodata
newline:
    .byte 10

.bss
.balign 8
input_buf:
    .zero 32768
.balign 8
line_starts:
    .zero 8000
.balign 8
line_ends:
    .zero 8000
