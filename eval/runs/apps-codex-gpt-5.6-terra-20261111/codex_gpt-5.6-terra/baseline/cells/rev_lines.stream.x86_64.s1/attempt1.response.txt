.equ READ_SIZE, 4096

.text
.globl _start
_start:
    xor r12d, r12d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buf]
    mov edx, READ_SIZE
    syscall

    test rax, rax
    jz end_input

    mov r14, rax
    lea r13, [rip + input_buf]

process_byte:
    mov al, byte ptr [r13]
    inc r13
    dec r14

    cmp al, 10
    je line_end

    lea rdx, [rip + line_buf]
    mov byte ptr [rdx + r12], al
    inc r12
    jmp check_more

line_end:
    call emit_line

check_more:
    test r14, r14
    jnz process_byte
    jmp read_more

end_input:
    test r12, r12
    jz exit_ok
    call emit_line

exit_ok:
    mov eax, 60
    xor edi, edi
    syscall

emit_line:
    test r12, r12
    jz emit_newline

    lea r8, [rip + line_buf]
    lea r9, [r8 + r12 - 1]

reverse_loop:
    cmp r8, r9
    jae write_line
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_loop

write_line:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + line_buf]
    mov rdx, r12
    syscall

emit_newline:
    mov eax, 1
    mov edi, 1
    lea rsi, [rip + newline]
    mov edx, 1
    syscall
    xor r12d, r12d
    ret

.section .rodata
newline:
    .byte 10

.bss
.align 16
line_buf:
    .zero 65536
input_buf:
    .zero 4096
