.text
.globl _start
_start:
    xor r12d, r12d
    xor r13d, r13d
    xor r14d, r14d
    xor r15d, r15d

read_more:
    xor eax, eax
    xor edi, edi
    lea rsi, [rip + input_buffer]
    mov edx, 65536
    syscall
    test rax, rax
    jle make_output

    add r14, rax
    lea r8, [rip + input_buffer]
    lea r9, [r8 + rax]

scan_byte:
    movzx eax, byte ptr [r8]
    inc r8

    cmp al, 10
    jne 1f
    inc r12
1:
    cmp al, 32
    je whitespace
    sub al, 9
    cmp al, 4
    jbe whitespace

    test r15, r15
    jnz next_byte
    inc r13
    mov r15d, 1
    jmp next_byte

whitespace:
    xor r15d, r15d

next_byte:
    cmp r8, r9
    jne scan_byte
    jmp read_more

make_output:
    lea rdi, [rip + output_buffer + 64]

    dec rdi
    mov byte ptr [rdi], 10

    mov rax, r14
    call decimal_back
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r13
    call decimal_back
    dec rdi
    mov byte ptr [rdi], 32

    mov rax, r12
    call decimal_back

    lea rdx, [rip + output_buffer + 64]
    sub rdx, rdi
    mov rsi, rdi
    mov eax, 1
    mov edi, 1
    syscall

    mov eax, 60
    xor edi, edi
    syscall

decimal_back:
    mov r8d, 10
    test rax, rax
    jnz 1f
    dec rdi
    mov byte ptr [rdi], 48
    ret
1:
    xor edx, edx
    div r8
    add dl, 48
    dec rdi
    mov byte ptr [rdi], dl
    test rax, rax
    jne 1b
    ret

.bss
.align 16
input_buffer:
    .zero 65536
.align 16
output_buffer:
    .zero 64
