.text
.globl _start
_start:
    lea r12, [rip + buffer]
    xor r13d, r13d

read_more:
    mov rax, 65536
    sub rax, r13
    lea rsi, [rip + buffer]
    add rsi, r13
    xor edi, edi
    mov rdx, rax
    xor eax, eax
    syscall
    test rax, rax
    jz input_done
    add r13, rax
    jmp read_more

input_done:
    lea r12, [rip + buffer]
    lea r13, [r12 + r13]

line_loop:
    cmp r12, r13
    je done

    mov r14, r12
find_newline:
    cmp r14, r13
    je final_line
    cmp byte ptr [r14], 10
    je have_newline
    inc r14
    jmp find_newline

have_newline:
    mov r8, r12
    mov r9, r14
    cmp r8, r9
    je write_line
    dec r9
reverse_line:
    cmp r8, r9
    jae write_line
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_line

write_line:
    mov rdx, r14
    sub rdx, r12
    inc rdx
    mov rsi, r12
    mov edi, 1
    mov eax, 1
    syscall
    lea r12, [r14 + 1]
    jmp line_loop

final_line:
    mov r8, r12
    mov r9, r13
    cmp r8, r9
    je done
    dec r9
reverse_final:
    cmp r8, r9
    jae write_final
    mov al, byte ptr [r8]
    mov dl, byte ptr [r9]
    mov byte ptr [r8], dl
    mov byte ptr [r9], al
    inc r8
    dec r9
    jmp reverse_final

write_final:
    mov rdx, r13
    sub rdx, r12
    mov rsi, r12
    mov edi, 1
    mov eax, 1
    syscall
    lea rsi, [rip + newline]
    mov edx, 1
    mov edi, 1
    mov eax, 1
    syscall

done:
    xor edi, edi
    mov eax, 60
    syscall

.section .rodata
newline:
    .byte 10

.bss
.align 16
buffer:
    .zero 65536
