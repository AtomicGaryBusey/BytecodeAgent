vm_run:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
    xor ebp, ebp
.Lloop:
    cmp rbx, r13
    je .Ldone
    mov rax, [r12 + rbx*8]
    cmp rax, 1
    je .Lpush
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 9
    je .Llt
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    cmp rax, 0
    je .Ldone
    inc rbx
    jmp .Lloop
.Lpush:
    mov rax, [r12 + rbx*8 + 8]
    mov [r14 + rbp*8], rax
    inc rbp
    add rbx, 2
    jmp .Lloop
.Ladd:
    mov rax, [r14 + rbp*8 - 8]
    add rax, [r14 + rbp*8 - 16]
    mov [r14 + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Lsub:
    mov rax, [r14 + rbp*8 - 16]
    sub rax, [r14 + rbp*8 - 8]
    mov [r14 + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Lmul:
    mov rax, [r14 + rbp*8 - 16]
    imul rax, [r14 + rbp*8 - 8]
    mov [r14 + rbp*8 - 16], rax
    dec rbp
    inc rbx
    jmp .Lloop
.Ldup:
    mov rax, [r14 + rbp*8 - 8]
    mov [r14 + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Lswap:
    mov rax, [r14 + rbp*8 - 8]
    mov rcx, [r14 + rbp*8 - 16]
    mov [r14 + rbp*8 - 8], rcx
    mov [r14 + rbp*8 - 16], rax
    inc rbx
    jmp .Lloop
.Ldrop:
    dec rbp
    inc rbx
    jmp .Lloop
.Lover:
    mov rax, [r14 + rbp*8 - 16]
    mov [r14 + rbp*8], rax
    inc rbp
    inc rbx
    jmp .Lloop
.Llt:
    mov rax, [r14 + rbp*8 - 8]
    xor ecx, ecx
    cmp qword ptr [r14 + rbp*8 - 16], rax
    setl cl
    mov [r14 + rbp*8 - 16], rcx
    dec rbp
    inc rbx
    jmp .Lloop
.Ljz:
    dec rbp
    mov rax, [r14 + rbp*8]
    mov rcx, [r12 + rbx*8 + 8]
    test rax, rax
    jne .Ljznz
    mov rbx, rcx
    jmp .Lloop
.Ljznz:
    add rbx, 2
    jmp .Lloop
.Ljmp:
    mov rbx, [r12 + rbx*8 + 8]
    jmp .Lloop
.Ldone:
    xor eax, eax
    test rbp, rbp
    je .Lzero
    mov rax, [r14 + rbp*8 - 8]
.Lzero:
    mov rcx, rbp
.Lzloop:
    cmp rcx, r15
    jge .Lret
    mov qword ptr [r14 + rcx*8], 0
    inc rcx
    jmp .Lzloop
.Lret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
