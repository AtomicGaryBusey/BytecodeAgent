base64_encode:
    push rbx
    mov r8, rdx                 # out
    mov rax, rsi                # n
    xor edx, edx
    mov ecx, 3
    div rcx                     # rax = n/3, rdx = n%3
    mov r9, rax                 # full groups
    mov r10, rdx                # remainder
    mov rbx, rax
    shl rbx, 2                  # 4 * (n/3)
    test r10, r10
    jz .Llen_done
    add rbx, 4                  # +4 if remainder
.Llen_done:
    test r9, r9
    jz .Lremainder
.Lloop:
    movzx eax, byte ptr [rdi]       # b0
    movzx ecx, byte ptr [rdi+1]     # b1
    mov esi, eax
    shr esi, 2                      # v0 = b0 >> 2
    mov edx, eax
    and edx, 3
    shl edx, 4                      # (b0 & 3) << 4
    mov r11d, ecx
    and r11d, 15
    shl r11d, 2                     # (b1 & 15) << 2
    shr ecx, 4
    or edx, ecx                     # v1
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    mov esi, edx                    # v1
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    movzx edx, byte ptr [rdi+2]     # b2
    add rdi, 3
    mov eax, edx
    shr eax, 6
    or eax, r11d                    # v2
    mov esi, eax
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    and edx, 63                     # v3
    mov esi, edx
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    dec r9
    jnz .Lloop
.Lremainder:
    test r10, r10
    jz .Ldone
    movzx eax, byte ptr [rdi]
    cmp r10, 2
    je .Lrem2
    mov edx, eax
    and edx, 3
    shl edx, 4                      # v1 = (b0 & 3) << 4
    mov esi, eax
    shr esi, 2                      # v0
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    mov esi, edx
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    mov byte ptr [r8], '='
    mov byte ptr [r8+1], '='
    add r8, 2
    jmp .Ldone
.Lrem2:
    movzx ecx, byte ptr [rdi+1]
    mov esi, eax
    shr esi, 2                      # v0
    mov edx, eax
    and edx, 3
    shl edx, 4
    mov r11d, ecx
    and r11d, 15
    shl r11d, 2                     # v2 = (b1 & 15) << 2
    shr ecx, 4
    or edx, ecx                     # v1
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    mov esi, edx                    # v1
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    mov esi, r11d                   # v2
    lea eax, [rsi+65]
    lea ecx, [rsi+71]
    cmp esi, 26
    cmovae eax, ecx
    lea ecx, [rsi-4]
    cmp esi, 52
    cmovae eax, ecx
    lea ecx, [rsi-19]
    cmp esi, 62
    cmovae eax, ecx
    lea ecx, [rsi-16]
    cmp esi, 63
    cmovae eax, ecx
    mov byte ptr [r8], al
    inc r8
    mov byte ptr [r8], '='
    inc r8
.Ldone:
    mov rax, rbx
    pop rbx
    ret
