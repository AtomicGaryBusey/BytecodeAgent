export function l $isqrt(l %n) {
@entry
	%lo =l copy 0
	%hi =l copy 4294967295
@loop
	%go =w cultl %lo, %hi
	jnz %go, @body, @done
@body
	%s =l add %lo, %hi
	%s1 =l add %s, 1
	%mid =l udiv %s1, 2
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @take, @skip
@take
	%lo =l copy %mid
	jmp @loop
@skip
	%hm1 =l sub %mid, 1
	%hi =l copy %hm1
	jmp @loop
@done
	ret %lo
}
