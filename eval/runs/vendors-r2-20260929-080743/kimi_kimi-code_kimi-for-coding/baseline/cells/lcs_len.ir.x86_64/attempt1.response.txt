export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	%zj =l copy 0
	%zero =l copy 0
	jmp @zloop
@zloop
	%zc =w cslel %zj, %n
	jnz %zc, @zbody, @zdone
@zbody
	%za =l mul %zj, 8
	%zp =l add %row, %za
	storel %zero, %zp
	%zj =l add %zj, 1
	jmp @zloop
@zdone
	%i =l copy 0
	jmp @iloop
@iloop
	%ic =w csltl %i, %m
	jnz %ic, @ibody, @idone
@ibody
	%diag =l copy 0
	%j =l copy 1
	jmp @jloop
@jloop
	%jc =w cslel %j, %n
	jnz %jc, @jbody, @jdone
@jbody
	%ja =l mul %j, 8
	%p =l add %row, %ja
	%up =l loadl %p
	%xp =l add %x, %i
	%xb =w loadub %xp
	%jm1 =l sub %j, 1
	%yp =l add %y, %jm1
	%yb =w loadub %yp
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%nv =l add %diag, 1
	storel %nv, %p
	jmp @jnext
@nomatch
	%la =l mul %jm1, 8
	%lp =l add %row, %la
	%left =l loadl %lp
	%mc =w cslel %left, %up
	jnz %mc, @mup, @mleft
@mup
	storel %up, %p
	jmp @jnext
@mleft
	storel %left, %p
	jmp @jnext
@jnext
	%diag =l copy %up
	%j =l add %j, 1
	jmp @jloop
@jdone
	%i =l add %i, 1
	jmp @iloop
@idone
	%ra =l mul %n, 8
	%rp =l add %row, %ra
	%res =l loadl %rp
	ret %res
}
