export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	%izero =l copy 0
	%wzero =l copy 0
	jmp @main
@main
	%i =l phi @start %izero, @repeat %inew, @copydone %jout
	%written =l phi @start %wzero, @repeat %wnew, @copydone %wnew2
	%more =w csltl %i, %n
	jnz %more, @body, @ret
@body
	%bp =l add %src, %i
	%b =w loadub %bp
	%k0 =l copy %i
	%c0 =l copy 0
	jmp @runloop
@runloop
	%k =l phi @body %k0, @runinc %k1
	%c =l phi @body %c0, @runinc %c1
	%rc1 =w csltl %k, %n
	jz %rc1, @rundone
	%rc2 =w csltl %c, 128
	jz %rc2, @rundone
	%kp =l add %src, %k
	%kb =w loadub %kp
	%rc3 =w ceqw %kb, %b
	jz %rc3, @rundone
	%k1 =l add %k, 1
	%c1 =l add %c, 1
	jmp @runloop
@rundone
	%isrep =w cslel 3, %c
	jnz %isrep, @repeat, @literal
@repeat
	%need2 =l add %written, 2
	%fits =w cslel %need2, %cap
	jz %fits, @fail
	%op =l add %out, %written
	%cw =w cast %c
	%hdr =w sub 257, %cw
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %b, %op1
	%wnew =l add %written, 2
	%inew =l add %i, %c
	jmp @main
@literal
	%j0 =l copy %i
	jmp @litloop
@litloop
	%j =l phi @literal %j0, @jinc %j1
	%lc1 =w csltl %j, %n
	jz %lc1, @litdone
	%diff =l sub %j, %i
	%lc2 =w csltl %diff, 128
	jz %lc2, @litdone
	%bjp =l add %src, %j
	%bj =w loadub %bjp
	%rk0 =l copy %j
	%rc0 =l copy 0
	jmp @jrun
@jrun
	%rk =l phi @litloop %rk0, @jinc %rk1
	%rc =l phi @litloop %rc0, @jinc %rc1
	%jr1 =w csltl %rk, %n
	jz %jr1, @jdone
	%jr2 =w csltl %rc, 128
	jz %jr2, @jdone
	%jkp =l add %src, %rk
	%jkb =w loadub %jkp
	%jr3 =w ceqw %jkb, %bj
	jz %jr3, @jdone
	%rk1 =l add %rk, 1
	%rc1 =l add %rc, 1
	jmp @jrun
@jdone
	%jr4 =w csltl %rc, 3
	jz %jr4, @litdone
	%j1 =l add %j, 1
	jmp @litloop
@litdone
	%L =l sub %j, %i
	%tot =l add %L, 1
	%tot2 =l add %tot, %written
	%fits2 =w cslel %tot2, %cap
	jz %fits2, @fail
	%op2 =l add %out, %written
	%Lw =w cast %L
	%hdr2 =w sub %Lw, 1
	storeb %hdr2, %op2
	%srcp =l add %src, %i
	%dstp =l add %op2, 1
	%cp0 =l copy 0
	jmp @copy
@copy
	%cp =l phi @litdone %cp0, @copyinc %cp1
	%cc =w csltl %cp, %L
	jz %cc, @copydone
	%sp =l add %srcp, %cp
	%byte =w loadub %sp
	%dp =l add %dstp, %cp
	storeb %byte, %dp
	%cp1 =l add %cp, 1
	jmp @copy
@copydone
	%jout =l copy %j
	%wnew2 =l add %written, %tot
	jmp @main
@fail
	ret -1
@ret
	ret %written
}
