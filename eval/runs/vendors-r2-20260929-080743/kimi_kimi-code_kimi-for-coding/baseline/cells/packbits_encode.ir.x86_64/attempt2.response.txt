export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @main
@main
	%i =l phi @start 0, @repok %inew, @copydone %jout
	%written =l phi @start 0, @repok %wnew, @copydone %wnew2
	%more =w csltl %i, %n
	jnz %more, @body, @ret
@body
	%bp =l add %src, %i
	%b =w loadub %bp
	jmp @runloop
@runloop
	%k =l phi @body %i, @runinc %k1
	%c =l phi @body 0, @runinc %c1
	%rc1 =w csltl %k, %n
	jnz %rc1, @rchk2, @rundone
@rchk2
	%rc2 =w csltl %c, 128
	jnz %rc2, @rchk3, @rundone
@rchk3
	%kp =l add %src, %k
	%kb =w loadub %kp
	%rc3 =w ceqw %kb, %b
	jnz %rc3, @runinc, @rundone
@runinc
	%k1 =l add %k, 1
	%c1 =l add %c, 1
	jmp @runloop
@rundone
	%isrep =w cslel 3, %c
	jnz %isrep, @repeat, @literal
@repeat
	%need2 =l add %written, 2
	%fits =w cslel %need2, %cap
	jnz %fits, @repok, @fail
@repok
	%op =l add %out, %written
	%cw =w cast %c
	%hdr =w sub 257, %cw
	storeb %hdr, %op
	%op1 =l add %op, 1
	storeb %b, %op1
	%wnew =l add %written, 2
	%inew =l add %i, %c
	jmp @main
@literal
	jmp @litloop
@litloop
	%j =l phi @literal %i, @jinc %j1
	%lc1 =w csltl %j, %n
	jnz %lc1, @lchk2, @litdone
@lchk2
	%diff =l sub %j, %i
	%lc2 =w csltl %diff, 128
	jnz %lc2, @lruncheck, @litdone
@lruncheck
	%bjp =l add %src, %j
	%bj =w loadub %bjp
	jmp @jrun
@jrun
	%rk =l phi @lruncheck %j, @jruninc %rk1
	%rc =l phi @lruncheck 0, @jruninc %rc1
	%jr1 =w csltl %rk, %n
	jnz %jr1, @jchk2, @jdone
@jchk2
	%jr2 =w csltl %rc, 128
	jnz %jr2, @jchk3, @jdone
@jchk3
	%jkp =l add %src, %rk
	%jkb =w loadub %jkp
	%jr3 =w ceqw %jkb, %bj
	jnz %jr3, @jruninc, @jdone
@jruninc
	%rk1 =l add %rk, 1
	%rc1 =l add %rc, 1
	jmp @jrun
@jdone
	%jr4 =w csltl %rc, 3
	jnz %jr4, @litdone, @jinc
@jinc
	%j1 =l add %j, 1
	jmp @litloop
@litdone
	%L =l sub %j, %i
	%tot =l add %L, 1
	%tot2 =l add %tot, %written
	%fits2 =w cslel %tot2, %cap
	jnz %fits2, @litok, @fail
@litok
	%op2 =l add %out, %written
	%Lw =w cast %L
	%hdr2 =w sub %Lw, 1
	storeb %hdr2, %op2
	%srcp =l add %src, %i
	%dstp =l add %op2, 1
	jmp @copy
@copy
	%cp =l phi @litok 0, @copyinc %cp1
	%cc =w csltl %cp, %L
	jnz %cc, @copyinc, @copydone
@copyinc
	%sp =l add %srcp, %cp
	%byte =w loadub %sp
	%dp =l add %dstp, %cp
	storeb %byte, %dp
	%cp1 =l add %cp, 1
	jmp @copy
@copydone
	%jout =l copy %j
	%wnew2 =l add %written, %tot
	jmp @main
@fail
	ret -1
@ret
	ret %written
}
