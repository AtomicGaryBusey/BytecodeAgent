stats_f:
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 32
    mov rbx, rdi
    lea r12, [rdi + rsi*8]
    mov r13, rcx
    mov r14, rdx
    xor r15d, r15d
    mov qword ptr [rsp+16], 0
    mov rdi, qword ptr [rbx]
    call r14
    add rbx, 8
    mov qword ptr [rsp], rax
    mov qword ptr [rsp+8], rax
    add r15, rax
    mov rdx, rax
    and edx, 1
    add qword ptr [rsp+16], rdx
    cmp rbx, r12
    je .Ldone
.Lloop:
    mov rdi, qword ptr [rbx]
    call r14
    add rbx, 8
    add r15, rax
    mov rdx, rax
    and edx, 1
    add qword ptr [rsp+16], rdx
    cmp rax, qword ptr [rsp]
    jge .Lno_min
    mov qword ptr [rsp], rax
.Lno_min:
    cmp rax, qword ptr [rsp+8]
    jle .Lno_max
    mov qword ptr [rsp+8], rax
.Lno_max:
    cmp rbx, r12
    jne .Lloop
.Ldone:
    mov rax, qword ptr [rsp]
    mov qword ptr [r13], rax
    mov rax, qword ptr [rsp+8]
    mov qword ptr [r13+8], rax
    mov qword ptr [r13+16], r15
    mov rax, qword ptr [rsp+16]
    add rsp, 32
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
