export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	jmp @head(l 0, l 0)
@head(l %i, l %cnt)
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @fin
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%x =l loadl %sa
	%r =l call %f(l %x)
	%da =l add %dst, %off
	storel %r, %da
	%lt =w csltl %r, %x
	%e =l extsw %lt
	%cnt2 =l add %cnt, %e
	%i2 =l add %i, 1
	jmp @head(l %i2, l %cnt2)
@fin
	ret %cnt
}
