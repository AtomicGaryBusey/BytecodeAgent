export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
	jmp @head
@head
	%cmp =w csltl %i, %n
	jnz %cmp, @body, @fin
@body
	%off =l mul %i, 8
	%sa =l add %src, %off
	%x =l loadl %sa
	%r =l call %f(l %x)
	%da =l add %dst, %off
	storel %r, %da
	%lt =w csltl %r, %x
	%e =l extsw %lt
	%cnt =l add %cnt, %e
	%i =l add %i, 1
	jmp @head
@fin
	ret %cnt
}
