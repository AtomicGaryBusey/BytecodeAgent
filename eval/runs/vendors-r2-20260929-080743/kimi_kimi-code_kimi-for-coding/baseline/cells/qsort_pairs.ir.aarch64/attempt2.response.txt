export function l $qsort_pairs(l %a, l %n) {
@entry
	jmp @top
@top
	%aa =l phi @entry %a, @contR %ar, @contL %aa
	%nn =l phi @entry %n, @contR %rsize, @contL %p
	%t =w cslel %nn, 1
	jnz %t, @ret, @body
@ret
	ret 0
@body
	%nm1 =l sub %nn, 1
	%poff =l mul %nm1, 16
	%pka =l add %aa, %poff
	%pk =l loadl %pka
	%pva =l add %pka, 8
	%pv =l loadl %pva
	jmp @forc
@forc
	%i =l phi @body -1, @finc %in
	%j =l phi @body 0, @finc %jn
	%cj =w csltl %j, %nm1
	jnz %cj, @fbody, @after
@fbody
	%joff =l mul %j, 16
	%ja =l add %aa, %joff
	%jk =l loadl %ja
	%jva =l add %ja, 8
	%jv =l loadl %jva
	%klt =w csltl %jk, %pk
	jnz %klt, @swap, @ckeq
@ckeq
	%keq =w ceql %jk, %pk
	jnz %keq, @ckv, @finc
@ckv
	%vlt =w csltl %jv, %pv
	jnz %vlt, @swap, @finc
@swap
	%i1 =l add %i, 1
	%ioff =l mul %i1, 16
	%ia =l add %aa, %ioff
	%ik =l loadl %ia
	%iva =l add %ia, 8
	%iv =l loadl %iva
	storel %ik, %ja
	storel %jv, %iva
	storel %jk, %ia
	storel %iv, %jva
	jmp @finc
@finc
	%in =l phi @swap %i1, @ckeq %i, @ckv %i
	%jn =l add %j, 1
	jmp @forc
@after
	%p =l add %i, 1
	%poff2 =l mul %p, 16
	%pa =l add %aa, %poff2
	%pk2 =l loadl %pa
	%pva2 =l add %pa, 8
	%pv2 =l loadl %pva2
	storel %pk2, %pka
	storel %pv2, %pva
	storel %pk, %pa
	storel %pv, %pva2
	%rsize =l sub %nm1, %p
	%lse =w cslel %p, %rsize
	jnz %lse, @callL, @callR
@callL
	%r1 =l call $qsort_pairs(l %aa, l %p)
	jmp @contR
@contR
	%p1 =l add %p, 1
	%roff =l mul %p1, 16
	%ar =l add %aa, %roff
	jmp @top
@callR
	%p1b =l add %p, 1
	%roffb =l mul %p1b, 16
	%arb =l add %aa, %roffb
	%r2 =l call $qsort_pairs(l %arb, l %rsize)
	jmp @contL
@contL
	jmp @top
}
