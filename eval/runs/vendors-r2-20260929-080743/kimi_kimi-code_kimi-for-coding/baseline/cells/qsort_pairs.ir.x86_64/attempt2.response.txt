export function l $qsort_pairs(l %a, l %n) {
@start
    %small =w csltl %n, 2
    jnz %small, @done, @body
@body
    %nm1 =l sub %n, 1
    %poff =l mul %nm1, 16
    %pa =l add %a, %poff
    %pk =l loadl %pa
    %pa8 =l add %pa, 8
    %pv =l loadl %pa8
    %i =l copy 0
    %j =l copy 0
@loop
    %cont =w csltl %j, %nm1
    jnz %cont, @iter, @part
@iter
    %joff =l mul %j, 16
    %ja =l add %a, %joff
    %jk =l loadl %ja
    %ja8 =l add %ja, 8
    %jv =l loadl %ja8
    %clt =w csltl %jk, %pk
    %ceq =w ceql %jk, %pk
    %cvl =w csltl %jv, %pv
    %both =w and %ceq, %cvl
    %cond =w or %clt, %both
    jnz %cond, @swap, @next
@swap
    %ioff =l mul %i, 16
    %ia =l add %a, %ioff
    %ik =l loadl %ia
    %ia8 =l add %ia, 8
    %iv =l loadl %ia8
    storel %jk, %ia
    storel %jv, %ia8
    storel %ik, %ja
    storel %iv, %ja8
    %i =l add %i, 1
@next
    %j =l add %j, 1
    jmp @loop
@part
    %ioff2 =l mul %i, 16
    %ia2 =l add %a, %ioff2
    %ik2 =l loadl %ia2
    %ia28 =l add %ia2, 8
    %iv2 =l loadl %ia28
    storel %pk, %ia2
    storel %pv, %ia28
    storel %ik2, %pa
    storel %iv2, %pa8
    %r1 =l call $qsort_pairs(l %a, l %i)
    %ip1 =l add %i, 1
    %roff =l mul %ip1, 16
    %ra =l add %a, %roff
    %rn =l sub %nm1, %i
    %r2 =l call $qsort_pairs(l %ra, l %rn)
    ret 0
@done
    ret 0
}
