qsort_pairs:
    cmp rsi, 2
    jl .Lret0
    push rbx
    push r12
    push r13
    push r14
    push r15
    sub rsp, 16
    mov r12, rdi
    mov r13, rsi
    lea rax, [r13 - 1]
    mov rcx, rax
    shr rax, 1
    shl rax, 4
    shl rcx, 4
    mov r14, [r12 + rax]
    mov r15, [r12 + rax + 8]
    mov rdx, [r12 + rax]
    mov rsi, [r12 + rax + 8]
    mov rdi, [r12 + rcx]
    mov r8, [r12 + rcx + 8]
    mov [r12 + rax], rdi
    mov [r12 + rax + 8], r8
    mov [r12 + rcx], rdx
    mov [r12 + rcx + 8], rsi
    xor ebx, ebx
    xor r8d, r8d
.Lscan:
    cmp rbx, rcx
    jae .Lplace
    mov rax, [r12 + rbx]
    mov rdx, [r12 + rbx + 8]
    cmp rax, r14
    jl .Lless
    jg .Lnless
    cmp rdx, r15
    jl .Lless
.Lnless:
    add rbx, 16
    jmp .Lscan
.Lless:
    cmp r8, rbx
    je .Lnoswap
    mov rsi, [r12 + r8]
    mov rdi, [r12 + r8 + 8]
    mov [r12 + r8], rax
    mov [r12 + r8 + 8], rdx
    mov [r12 + rbx], rsi
    mov [r12 + rbx + 8], rdi
.Lnoswap:
    add r8, 16
    add rbx, 16
    jmp .Lscan
.Lplace:
    cmp r8, rcx
    je .Lcount
    mov rax, [r12 + r8]
    mov rdx, [r12 + r8 + 8]
    mov rsi, [r12 + rcx]
    mov rdi, [r12 + rcx + 8]
    mov [r12 + r8], rsi
    mov [r12 + r8 + 8], rdi
    mov [r12 + rcx], rax
    mov [r12 + rcx + 8], rdx
.Lcount:
    mov rax, r8
    shr rax, 4
    mov [rsp], rax
    cmp rax, 1
    jle .Lskip_left
    mov rdi, r12
    mov rsi, rax
    call qsort_pairs
.Lskip_left:
    mov rsi, [rsp]
    lea rax, [rsi + 1]
    mov rdx, r13
    sub rdx, rax
    cmp rdx, 1
    jle .Ldone
    mov rdi, r12
    shl rax, 4
    add rdi, rax
    mov rsi, rdx
    call qsort_pairs
.Ldone:
    add rsp, 16
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
.Lret0:
    xor eax, eax
    ret
