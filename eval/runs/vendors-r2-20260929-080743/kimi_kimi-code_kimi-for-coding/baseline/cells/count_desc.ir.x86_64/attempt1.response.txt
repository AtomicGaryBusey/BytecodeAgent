export function l $count_desc(l %a, l %n) {
@start
	%i = copy 0
	%count = copy 0
	%nm1 = sub %n, 1
	jmp @loop
@loop
	%cond = csltl %i, %nm1
	jnz %cond, @body, @done
@body
	%off = mul %i, 8
	%pa = add %a, %off
	%ai = loadl %pa
	%ai1 = loadl %pa, 8
	%gt = csgtl %ai, %ai1
	jnz %gt, @inc, @next
@inc
	%count = add %count, 1
@next
	%i = add %i, 1
	jmp @loop
@done
	ret %count
}
