export function l $count_desc(l %a, l %n) {
@start
	%i =l copy 0
	%count =l copy 0
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%cond =w csltl %i, %nm1
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%pa =l add %a, %off
	%pa1 =l add %pa, 8
	%ai =l loadl %pa
	%ai1 =l loadl %pa1
	%gt =w csgtl %ai, %ai1
	jnz %gt, @inc, @next
@inc
	%count =l add %count, 1
@next
	%i =l add %i, 1
	jmp @loop
@done
	ret %count
}
