export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%p0hi =l shr %p0, 32
	%s =l add %p1, %p2
	%c1 =w cultl %s, %p1
	%t =l add %p0hi, %s
	%c2 =w cultl %t, %p0hi
	%thi =l shr %t, 32
	%c1l =l extuw %c1
	%c2l =l extuw %c2
	%cc =l add %c1l, %c2l
	%r =l add %p3, %thi
	%r2 =l add %r, %cc
	ret %r2
}
