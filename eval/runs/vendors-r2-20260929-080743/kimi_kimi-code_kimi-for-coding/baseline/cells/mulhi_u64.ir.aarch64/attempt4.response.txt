export function l $mulhi_u64(l %a, l %b) {
@start
	%alo =l and %a, 4294967295
	%ahi =l shr %a, 32
	%blo =l and %b, 4294967295
	%bhi =l shr %b, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%cy =l shr %p0, 32
	%t =l add %p1, %cy
	%mid =l add %t, %p2
	%c =w cultl %mid, %t
	%cl =l extuw %c
	%csh =l shl %cl, 32
	%mhi =l shr %mid, 32
	%s =l add %p3, %mhi
	%res =l add %s, %csh
	ret %res
}
