export function l $crc32(l %p, l %n) {
@start
	%cs =l alloc4 4
	%is =l alloc8 8
	%js =l alloc4 4
	storew 4294967295, %cs
	storel 0, %is
	jmp @outer
@outer
	%i =l loadl %is
	%cont =w csltl %i, %n
	jnz %cont, @body, @fin
@body
	%c0 =w loadw %cs
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %c0, %b
	storew %cx, %cs
	storew 8, %js
	jmp @inner
@inner
	%cc =w loadw %cs
	%j =w loadw %js
	%bit =w and %cc, 1
	%shr =w shr %cc, 1
	%mask =w sub 0, %bit
	%poly =w and 3988292384, %mask
	%cn =w xor %shr, %poly
	storew %cn, %cs
	%j1 =w sub %j, 1
	storew %j1, %js
	%go =w csgtw %j1, 0
	jnz %go, @inner, @next
@next
	%i2 =l loadl %is
	%i3 =l add %i2, 1
	storel %i3, %is
	jmp @outer
@fin
	%cr =w loadw %cs
	%res =w xor %cr, 4294967295
	%resl =l extuw %res
	ret %resl
}
