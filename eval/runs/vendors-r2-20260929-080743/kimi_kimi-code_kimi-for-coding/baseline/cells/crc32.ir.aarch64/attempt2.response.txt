export function l $crc32(l %p, l %n) {
@start
	%c0 =w copy 4294967295
	%i0 =l copy 0
	jmp @outer(%c0, %i0)
@outer(%crc =w, %i =l)
	%cont =w csltl %i, %n
	jnz %cont, @body, @fin(%crc)
@body
	%addr =l add %p, %i
	%b =w loadub %addr
	%cx =w xor %crc, %b
	%j0 =w copy 8
	jmp @inner(%cx, %j0)
@inner(%c =w, %j =w)
	%bit =w and %c, 1
	%shr =w shr %c, 1
	%mask =w sub 0, %bit
	%poly =w and 3988292384, %mask
	%cn =w xor %shr, %poly
	%j1 =w sub %j, 1
	%go =w csgtw %j1, 0
	jnz %go, @inner(%cn, %j1), @next
@next
	%i1 =l add %i, 1
	jmp @outer(%cn, %i1)
@fin(%cr =w)
	%res =w xor %cr, 4294967295
	%resl =l extuw %res
	ret %resl
}
