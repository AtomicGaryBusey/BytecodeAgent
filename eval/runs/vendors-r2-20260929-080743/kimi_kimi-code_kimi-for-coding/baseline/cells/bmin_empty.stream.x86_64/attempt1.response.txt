bmin_empty:
    test rsi, rsi
    je .Lempty
    mov rax, qword ptr [rdi]
    mov ecx, 1
.Lloop:
    cmp rcx, rsi
    jge .Ldone
    mov rdx, qword ptr [rdi + rcx*8]
    cmp rdx, rax
    jge .Lskip
    mov rax, rdx
.Lskip:
    inc rcx
    jmp .Lloop
.Ldone:
    ret
.Lempty:
    movabs rax, 9223372036854775807
    ret
