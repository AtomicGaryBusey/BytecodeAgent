collatz_max:
	mov	x9, #0		// running max steps
	mov	x10, #1		// k = 1
.Louter:
	cmp	x10, x0
	bgt	.Ldone
	mov	x11, x10	// x = k
	mov	x12, #0		// steps = 0
.Linner:
	cmp	x11, #1
	beq	.Lcheck
	tst	x11, #1
	bne	.Lodd
	lsr	x11, x11, #1
	b	.Lstep
.Lodd:
	add	x11, x11, x11, lsl #1	// x = 3x
	add	x11, x11, #1		// x = 3x + 1
.Lstep:
	add	x12, x12, #1
	b	.Linner
.Lcheck:
	cmp	x12, x9
	csel	x9, x12, x9, gt	// keep larger of steps, max
	add	x10, x10, #1
	b	.Louter
.Ldone:
	mov	x0, x9
	ret
