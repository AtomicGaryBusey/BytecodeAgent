export function l $count_nonzero_prefix(l %a, l %n) {
@start
	%i =l copy 0
	%one =w copy 1
	jnz %one, @check, @done
@check
	%c =w csltl %i, %n
	jnz %c, @body, @done
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%z =w ceql %v, 0
	jnz %z, @done, @inc
@inc
	%i2 =l add %i, 1
	%i =l copy %i2
	jnz %one, @check, @done
@done
	ret %i
}
