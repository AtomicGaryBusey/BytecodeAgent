export function l $base64_encode(l %src, l %n, l %out) {
@start
	%i =l copy 0
	%w =l copy 0
	%pad =l copy 61
	br @group

@group
	%rem =l sub %n, %i
	%go =w csgel %rem, 1
	jnz %go, @load, @done

@load
	%p =l add %src, %i
	%b0 =l loadub %p
	%ge2 =w csgel %rem, 2
	jnz %ge2, @loadb1, @zero1

@loadb1
	%p1 =l add %p, 1
	%b1 =l loadub %p1
	br @haveb1

@zero1
	%b1 =l copy 0

@haveb1
	%ge3 =w csgel %rem, 3
	jnz %ge3, @loadb2, @zero2

@loadb2
	%p2 =l add %p, 2
	%b2 =l loadub %p2
	br @haveb2

@zero2
	%b2 =l copy 0

@haveb2
	%v16 =l shl %b0, 16
	%v8 =l shl %b1, 8
	%val0 =l or %v16, %v8
	%val =l or %val0, %b2
	%le3 =w csltl %rem, 4
	jnz %le3, @mIsRem, @mIs3

@mIsRem
	%m =l copy %rem
	br @havem

@mIs3
	%m =l copy 3

@havem
	%c =l copy 0
	%sh =l copy 18
	br @charcond

@charcond
	%creal =w csltl %c, %m
	jnz %creal, @realchar, @padchar

@realchar
	%vs =l shr %val, %sh
	%v6 =l and %vs, 63
	%g26 =w csgel %v6, 26
	%g52 =w csgel %v6, 52
	%g62 =w csgel %v6, 62
	%g63 =w csgel %v6, 63
	%e26 =l extsw %g26
	%e52 =l extsw %g52
	%e62 =l extsw %g62
	%e63 =l extsw %g63
	%f26 =l mul %e26, 6
	%f52 =l mul %e52, 75
	%f62 =l mul %e62, 15
	%f63 =l mul %e63, 3
	%ch0 =l add %v6, 65
	%ch1 =l add %ch0, %f26
	%ch2 =l sub %ch1, %f52
	%ch3 =l sub %ch2, %f62
	%ch =l add %ch3, %f63
	%dst =l add %out, %w
	%dstc =l add %dst, %c
	storeb %ch, %dstc
	br @charnext

@padchar
	%dstp =l add %out, %w
	%dstpc =l add %dstp, %c
	storeb %pad, %dstpc
	br @charnext

@charnext
	%c =l add %c, 1
	%sh =l sub %sh, 6
	%cmore =w csltl %c, 4
	jnz %cmore, @charcond, @groupnext

@groupnext
	%w =l add %w, 4
	%last =w csltl %rem, 4
	jnz %last, @done, @advance

@advance
	%i =l add %i, 3
	br @group

@done
	ret %w
}
