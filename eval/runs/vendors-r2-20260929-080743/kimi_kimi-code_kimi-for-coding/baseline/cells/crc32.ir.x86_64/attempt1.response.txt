export function l $crc32(l %p, l %n) {
@start
	jmp @outer
@outer
	%crc =w phi @start 4294967295, @next %crcn
	%i =l phi @start 0, @next %in
	%cmp =w csltl %i, %n
	jnz %cmp, @byte, @done
@byte
	%pa =l add %p, %i
	%b =w loadub %pa
	%cr =w xor %crc, %b
	jmp @bit
@bit
	%crk =w phi @byte %cr, @merge %crm
	%k =w phi @byte 0, @merge %km
	%kcmp =w csltl %k, 8
	jnz %kcmp, @dobit, @next
@dobit
	%lsb =w and %crk, 1
	%shr =w shr %crk, 1
	jnz %lsb, @odd, @even
@odd
	%cro =w xor %shr, 0xEDB88320
	jmp @merge
@even
	%cre =w copy %shr
@merge
	%crm =w phi @odd %cro, @even %cre
	%km =w add %k, 1
	jmp @bit
@next
	%crcn =w copy %crk
	%in =l add %i, 1
	jmp @outer
@done
	%res =w xor %crc, 4294967295
	%resl =l extuw %res
	ret %resl
}
