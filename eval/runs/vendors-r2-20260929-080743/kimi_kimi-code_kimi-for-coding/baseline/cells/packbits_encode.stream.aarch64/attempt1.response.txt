packbits_encode:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x22, x3
    mov x23, xzr
    mov x24, xzr
.Louter:
    cmp x23, x20
    bge .Ldone
    ldrb w9, [x19, x23]
    mov x13, x23
    add x14, x23, #128
    cmp x14, x20
    csel x14, x20, x14, hi
.Lruncount:
    add x13, x13, #1
    cmp x13, x14
    bge .Lrundone
    ldrb w10, [x19, x13]
    cmp w10, w9
    beq .Lruncount
.Lrundone:
    sub x15, x13, x23
    cmp x15, #3
    blt .Lliteral
    sub x10, x22, x24
    cmp x10, #2
    blt .Lfail
    mov w11, #257
    sub w11, w11, w15
    add x10, x21, x24
    strb w11, [x10]
    strb w9, [x10, #1]
    add x24, x24, #2
    add x23, x23, x15
    b .Louter
.Lliteral:
    mov x13, x23
.Llitscan:
    cmp x13, x20
    bge .Llitdone
    sub x10, x13, x23
    cmp x10, #128
    bge .Llitdone
    ldrb w9, [x19, x13]
    mov x11, x13
    add x12, x13, #128
    cmp x12, x20
    csel x12, x20, x12, hi
.Llitrun:
    add x11, x11, #1
    cmp x11, x12
    bge .Llitrundone
    ldrb w10, [x19, x11]
    cmp w10, w9
    beq .Llitrun
.Llitrundone:
    sub x10, x11, x13
    cmp x10, #3
    bge .Llitdone
    add x13, x13, #1
    b .Llitscan
.Llitdone:
    sub x15, x13, x23
    sub x10, x22, x24
    add x11, x15, #1
    cmp x10, x11
    blt .Lfail
    add x10, x21, x24
    sub w11, w15, #1
    strb w11, [x10]
    add x10, x10, #1
    mov x11, x23
.Lcopy:
    cmp x11, x13
    bge .Lcopydone
    ldrb w12, [x19, x11]
    strb w12, [x10]
    add x10, x10, #1
    add x11, x11, #1
    b .Lcopy
.Lcopydone:
    add x24, x24, x15
    add x24, x24, #1
    mov x23, x13
    b .Louter
.Lfail:
    mov x0, #-1
    b .Lepilogue
.Ldone:
    mov x0, x24
.Lepilogue:
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
