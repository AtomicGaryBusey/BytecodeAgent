export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer_init
@outer_init
	%i =l copy 0
	jmp @outer_cond
@outer_cond
	%c0 =w csltl %i, %n
	jnz %c0, @inner_init, @done
@inner_init
	%bound =l sub %n, %i
	%limit =l sub %bound, 1
	%j =l copy 0
	jmp @inner_cond
@inner_cond
	%c1 =w csltl %j, %limit
	jnz %c1, @body, @outer_inc
@body
	%off =l mul %j, 8
	%p =l add %a, %off
	%x =l loadl %p
	%q =l add %p, 8
	%y =l loadl %q
	%c2 =w csgtl %x, %y
	jnz %c2, @swap, @inner_inc
@swap
	storel %y, %p
	storel %x, %q
	jmp @inner_inc
@inner_inc
	%j =l add %j, 1
	jmp @inner_cond
@outer_inc
	%i =l add %i, 1
	jmp @outer_cond
@done
	ret 0
}
