max_byte:
    mov w2, wzr                 // running max = 0
    cbz x1, .Ldone              // n == 0: return 0, read nothing
    mov x3, x0                  // cursor = p
.Lloop:
    ldrb w5, [x3], #1           // load byte, advance cursor
    cmp w5, w2
    csel w2, w5, w2, hi         // keep larger (unsigned)
    subs x1, x1, #1
    b.ne .Lloop
.Ldone:
    mov w0, w2                  // zero-extend result into x0
    ret
