export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
	%i =l copy 0
	%cnt =l copy 0
	%go =w csltl %i, %n
	jnz %go, @loop, @done
@loop
	%off =l mul %i, 8
	%sa =l add %src, %off
	%x =l loadl %sa
	%r =l call %f(l %x)
	%da =l add %dst, %off
	storel %r, %da
	%lt =w csltl %r, %x
	%i =l add %i, 1
	jnz %lt, @inc, @cond
@inc
	%cnt =l add %cnt, 1
@cond
	%go =w csltl %i, %n
	jnz %go, @loop, @done
@done
	ret %cnt
}
