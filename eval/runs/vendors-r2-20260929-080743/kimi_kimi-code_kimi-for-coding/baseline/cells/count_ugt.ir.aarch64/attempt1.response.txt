export function l $count_ugt(l %a, l %n, l %t) {
@start
	%i =l copy 0
	%cnt =l copy 0
	jmp @loop
@loop
	%c =w csltl %i, %n
	jnz %c, @body, @end
@body
	%off =l mul %i, 8
	%p =l add %a, %off
	%v =l loadl %p
	%g =w cugtl %v, %t
	%e =l extuw %g
	%cnt =l add %cnt, %e
	%i =l add %i, 1
	jmp @loop
@end
	ret %cnt
}
