memmove_i64:
    cmp x0, x1
    beq .Ldone
    bhs .Lbackward
.Lforward:
    cbz x2, .Ldone
.Lfwd:
    ldr x3, [x1], #8
    str x3, [x0], #8
    subs x2, x2, #1
    bne .Lfwd
    b .Ldone
.Lbackward:
    cbz x2, .Ldone
    mov x4, x2
.Lbwd:
    sub x4, x4, #1
    ldr x3, [x1, x4, lsl #3]
    str x3, [x0, x4, lsl #3]
    cbnz x4, .Lbwd
.Ldone:
    mov x0, #0
    ret
