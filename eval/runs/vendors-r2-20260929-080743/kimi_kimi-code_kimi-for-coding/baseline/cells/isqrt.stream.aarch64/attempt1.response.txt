isqrt:
    mov     x10, xzr              // res = 0
    mov     x9, #1
    lsl     x9, x9, #62           // bit = 1 << 62, highest 4^k for 64-bit
.Lloop:
    add     x11, x10, x9          // res + bit (never overflows 64 bits)
    cmp     x0, x11
    b.lo    .Lelse                // n < res + bit
    sub     x0, x0, x11           // n -= res + bit
    add     x10, x9, x10, lsr #1  // res = bit + (res >> 1)
    b       .Lnext
.Lelse:
    lsr     x10, x10, #1          // res >>= 1
.Lnext:
    lsr     x9, x9, #2            // bit >>= 2
    cbnz    x9, .Lloop
    mov     x0, x10               // return res
    ret
