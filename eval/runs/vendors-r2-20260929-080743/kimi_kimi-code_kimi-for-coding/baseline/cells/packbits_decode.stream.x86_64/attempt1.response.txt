packbits_decode:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    jae .Ldone
    movsx r10d, byte ptr [rdi + r8]
    cmp r10d, -128
    je .Lnop
    test r10d, r10d
    js .Lrepeat
    lea r11, [r8 + r10 + 2]
    cmp r11, rsi
    jg .Lsrcerr
    lea r11, [r9 + r10 + 1]
    cmp r11, rcx
    jg .Llit_over
    inc r10d
    inc r8
    lea r11, [rdi + r8]
.Llit_copy:
    movzx eax, byte ptr [r11]
    mov byte ptr [rdx + r9], al
    inc r11
    inc r9
    inc r8
    dec r10d
    jnz .Llit_copy
    jmp .Lloop
.Lrepeat:
    lea r11, [r8 + 2]
    cmp r11, rsi
    jg .Lsrcerr
    mov r11d, 1
    sub r11d, r10d
    lea rax, [r9 + r11]
    cmp rax, rcx
    jg .Lrep_over
    movzx eax, byte ptr [rdi + r8 + 1]
    add r8, 2
.Lrep_fill:
    mov byte ptr [rdx + r9], al
    inc r9
    dec r11d
    jnz .Lrep_fill
    jmp .Lloop
.Lnop:
    inc r8
    jmp .Lloop
.Llit_over:
    mov r11, rcx
    sub r11, r9
    test r11, r11
    jz .Lcapret
    lea r10, [rdi + r8 + 1]
.Llit_over_copy:
    movzx eax, byte ptr [r10]
    mov byte ptr [rdx + r9], al
    inc r10
    inc r9
    dec r11
    jnz .Llit_over_copy
    jmp .Lcapret
.Lrep_over:
    mov r11, rcx
    sub r11, r9
    test r11, r11
    jz .Lcapret
    movzx eax, byte ptr [rdi + r8 + 1]
.Lrep_over_fill:
    mov byte ptr [rdx + r9], al
    inc r9
    dec r11
    jnz .Lrep_over_fill
    jmp .Lcapret
.Ldone:
    mov rax, r9
    ret
.Lsrcerr:
    mov rax, -2
    ret
.Lcapret:
    mov rax, -1
    ret
