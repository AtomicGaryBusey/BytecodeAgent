export function l $map_sum(l %a, l %n, l %f) {
@start
	%sum =l copy 0
	%i =l copy 0
	%cmp =w csltl %i, %n
	jnz %cmp, @loop, @ret
@loop
	%off =l mul %i, 8
	%addr =l add %a, %off
	%x =l loadl %addr
	%r =l call %f(l %x)
	%sum =l add %sum, %r
	%i =l add %i, 1
	%cmp =w csltl %i, %n
	jnz %cmp, @loop, @ret
@ret
	ret %sum
}
