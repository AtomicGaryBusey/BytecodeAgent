crc32:
    mov eax, 0xFFFFFFFF
    test rsi, rsi
    jle .Ldone
    mov ecx, 8
.Lbyte_loop:
    movzx edx, byte ptr [rdi]
    xor eax, edx
.Lbit_loop:
    test eax, 1
    jz .Lbit_shift
    shr eax, 1
    xor eax, 0xEDB88320
    jmp .Lbit_next
.Lbit_shift:
    shr eax, 1
.Lbit_next:
    dec ecx
    jnz .Lbit_loop
    mov ecx, 8
    inc rdi
    dec rsi
    jnz .Lbyte_loop
.Ldone:
    xor eax, 0xFFFFFFFF
    ret
