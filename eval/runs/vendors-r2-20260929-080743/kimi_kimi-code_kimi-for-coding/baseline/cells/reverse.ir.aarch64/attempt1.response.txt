export function l $reverse(l %a, l %n) {
@start
	%j =l sub %n, 1
	%i =l copy 0
	%c =w csltl %i, %j
	jnz %c, @loop, @done
@loop
	%oi =l shl %i, 3
	%oj =l shl %j, 3
	%pi =l add %a, %oi
	%pj =l add %a, %oj
	%t =l loadl %pi
	%u =l loadl %pj
	storel %u, %pi
	storel %t, %pj
	%i =l add %i, 1
	%j =l sub %j, 1
	%c =w csltl %i, %j
	jnz %c, @loop, @done
@done
	ret 0
}
