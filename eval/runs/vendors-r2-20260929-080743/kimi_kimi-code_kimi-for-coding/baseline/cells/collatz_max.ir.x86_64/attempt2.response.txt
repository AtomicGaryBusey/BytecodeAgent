export function l $collatz_max(l %n) {
@start
	%max =l copy 0
	%k =l copy 1
	jmp @outer
@outer
	%go =w cslel %k, %n
	jnz %go, @body, @done
@body
	%x =l copy %k
	%steps =l copy 0
	jmp @iter
@iter
	%one =w ceql %x, 1
	jnz %one, @record, @step
@step
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @odd, @even
@even
	%x =l div %x, 2
	jmp @count
@odd
	%t =l mul %x, 3
	%x =l add %t, 1
@count
	%steps =l add %steps, 1
	jmp @iter
@record
	%gt =w csgtl %steps, %max
	jnz %gt, @take, @next
@take
	%max =l copy %steps
@next
	%k =l add %k, 1
	jmp @outer
@done
	ret %max
}
