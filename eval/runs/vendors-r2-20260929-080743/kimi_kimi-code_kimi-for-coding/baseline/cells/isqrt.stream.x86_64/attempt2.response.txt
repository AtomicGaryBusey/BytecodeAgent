isqrt:
    xor rax, rax
    mov rcx, 1
    shl rcx, 62
.Lfind:
    cmp rcx, rdi
    jbe .Lloop
    shr rcx, 2
    jmp .Lfind
.Lloop:
    test rcx, rcx
    jz .Ldone
    lea rdx, [rax + rcx]
    cmp rdi, rdx
    jb .Lelse
    sub rdi, rdx
    shr rax, 1
    add rax, rcx
    jmp .Lnext
.Lelse:
    shr rax, 1
.Lnext:
    shr rcx, 2
    jmp .Lloop
.Ldone:
    ret
