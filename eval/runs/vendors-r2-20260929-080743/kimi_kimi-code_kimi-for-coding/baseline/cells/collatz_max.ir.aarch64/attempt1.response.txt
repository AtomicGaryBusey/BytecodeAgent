export function l $collatz_max(l %n) {
@start
	%max0 =l copy 0
	%k0 =l copy 1
	jmp @loop
@loop
	%max =l phi @start %max0, @cont %maxc
	%k =l phi @start %k0, @cont %k1
	%done =w csltl %n, %k
	jnz %done, @end, @body
@body
	%x0 =l copy %k
	%s0 =l copy 0
	jmp @inner
@inner
	%x =l phi @body %x0, @merge %xm
	%s =l phi @body %s0, @merge %sm
	%isone =w ceql %x, 1
	jnz %isone, @after, @chk
@chk
	%parity =l rem %x, 2
	%even =w ceql %parity, 0
	jnz %even, @ev, @od
@ev
	%xe =l div %x, 2
	%se =l add %s, 1
	jmp @merge
@od
	%xo =l mul %x, 3
	%xo1 =l add %xo, 1
	%so =l add %s, 1
	jmp @merge
@merge
	%xm =l phi @ev %xe, @od %xo1
	%sm =l phi @ev %se, @od %so
	jmp @inner
@after
	%gt =w csgtl %s, %max
	jnz %gt, @upd, @cont
@upd
	%maxu =l copy %s
	jmp @cont
@cont
	%maxc =l phi @after %max, @upd %maxu
	%k1 =l add %k, 1
	jmp @loop
@end
	ret %max
}
