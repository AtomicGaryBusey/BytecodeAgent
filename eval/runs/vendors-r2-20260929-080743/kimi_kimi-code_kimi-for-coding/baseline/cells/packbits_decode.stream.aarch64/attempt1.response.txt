packbits_decode:
    stp x19, x20, [sp, #-16]!
    stp x21, x22, [sp, #-16]!
    stp x23, x24, [sp, #-16]!
    mov x19, x0
    mov x20, x1
    mov x21, x2
    mov x22, x3
    mov x23, #0
    mov x24, #0
.Lloop:
    cmp x24, x20
    b.ge .Ldone
    ldrsb x0, [x19, x24]
    add x24, x24, #1
    cmn x0, #128
    b.eq .Lloop
    cmp x0, #0
    b.lt .Lrepeat
    add x1, x0, #1
    add x2, x24, x1
    cmp x2, x20
    b.gt .Lerr_src
    add x2, x23, x1
    cmp x2, x22
    b.gt .Ltrunc_lit
    add x3, x19, x24
    add x4, x21, x23
    mov x5, #0
.Llit_copy:
    ldrb w6, [x3, x5]
    strb w6, [x4, x5]
    add x5, x5, #1
    cmp x5, x1
    b.lt .Llit_copy
    add x23, x23, x1
    add x24, x24, x1
    b .Lloop
.Lrepeat:
    neg x1, x0
    add x1, x1, #1
    add x2, x24, #1
    cmp x2, x20
    b.gt .Lerr_src
    add x2, x23, x1
    cmp x2, x22
    b.gt .Ltrunc_rep
    ldrb w6, [x19, x24]
    add x4, x21, x23
    mov x5, #0
.Lrep_fill:
    strb w6, [x4, x5]
    add x5, x5, #1
    cmp x5, x1
    b.lt .Lrep_fill
    add x23, x23, x1
    add x24, x24, #1
    b .Lloop
.Ltrunc_lit:
    sub x1, x22, x23
    add x3, x19, x24
    add x4, x21, x23
    mov x5, #0
.Ltrunc_lit_copy:
    cmp x5, x1
    b.ge .Lretm1
    ldrb w6, [x3, x5]
    strb w6, [x4, x5]
    add x5, x5, #1
    b .Ltrunc_lit_copy
.Ltrunc_rep:
    sub x1, x22, x23
    ldrb w6, [x19, x24]
    add x4, x21, x23
    mov x5, #0
.Ltrunc_rep_fill:
    cmp x5, x1
    b.ge .Lretm1
    strb w6, [x4, x5]
    add x5, x5, #1
    b .Ltrunc_rep_fill
.Lretm1:
    mov x0, #-1
    b .Lepilogue
.Lerr_src:
    mov x0, #-2
    b .Lepilogue
.Ldone:
    mov x0, x23
.Lepilogue:
    ldp x23, x24, [sp], #16
    ldp x21, x22, [sp], #16
    ldp x19, x20, [sp], #16
    ret
