reverse:
    mov rax, rsi
    shr rax, 1
    test rax, rax
    je .Ldone
    lea rdx, [rdi + rsi*8]
    sub rdx, 8
.Lloop:
    mov r8, qword ptr [rdi]
    mov r9, qword ptr [rdx]
    mov qword ptr [rdi], r9
    mov qword ptr [rdx], r8
    add rdi, 8
    sub rdx, 8
    dec rax
    jnz .Lloop
.Ldone:
    xor eax, eax
    ret
