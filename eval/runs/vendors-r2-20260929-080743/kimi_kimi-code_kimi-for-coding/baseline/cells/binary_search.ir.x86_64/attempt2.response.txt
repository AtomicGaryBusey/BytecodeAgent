export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @back %nl
	%hi =l phi @start %n, @back %nh
	%c =w csltl %lo, %hi
	jnz %c, @body, @fail
@body
	%d =l sub %hi, %lo
	%h =l shr %d, 1
	%m =l add %lo, %h
	%o =l shl %m, 3
	%p =l add %a, %o
	%v =l loadl %p
	%e =w ceql %v, %key
	jnz %e, @found, @cmp
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @left, @right
@left
	%nl =l add %m, 1
	%nh =l copy %hi
	jmp @back
@right
	%nl =l copy %lo
	%nh =l copy %m
	jmp @back
@back
	jmp @loop
@found
	ret %m
@fail
	ret -1
}
