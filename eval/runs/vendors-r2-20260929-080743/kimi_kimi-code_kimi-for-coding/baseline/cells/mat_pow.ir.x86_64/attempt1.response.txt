export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%c =l copy 0
	jmp @zout
@zout
	%cond =w csltl %c, %nn
	jnz %cond, @zout.body, @ident
@zout.body
	%off =l mul %c, 8
	%addr =l add %out, %off
	storel 0, %addr
	%c =l add %c, 1
	jmp @zout
@ident
	%i =l copy 0
	jmp @ident.t
@ident.t
	%cond =w csltl %i, %n
	jnz %cond, @ident.body, @kinit
@ident.body
	%ix =l mul %i, %n
	%ix =l add %ix, %i
	%off =l mul %ix, 8
	%addr =l add %out, %off
	storel 1, %addr
	%i =l add %i, 1
	jmp @ident.t
@kinit
	%kk =l copy %k
	jmp @ktest
@ktest
	%cond =w csgtl %kk, 0
	jnz %cond, @mm.i.init, @ztmp
@mm.i.init
	%i =l copy 0
	jmp @mm.i
@mm.i
	%cond =w csltl %i, %n
	jnz %cond, @mm.j.init, @cp.init
@mm.j.init
	%j =l copy 0
	jmp @mm.j
@mm.j
	%cond =w csltl %j, %n
	jnz %cond, @mm.t.init, @mm.i.next
@mm.t.init
	%sum =l copy 0
	%t =l copy 0
	jmp @mm.t
@mm.t
	%cond =w csltl %t, %n
	jnz %cond, @mm.t.body, @mm.store
@mm.t.body
	%ix =l mul %i, %n
	%ix =l add %ix, %t
	%off =l mul %ix, 8
	%p =l add %out, %off
	%x =l loadl %p
	%jx =l mul %t, %n
	%jx =l add %jx, %j
	%off2 =l mul %jx, 8
	%p2 =l add %a, %off2
	%y =l loadl %p2
	%pr =l mul %x, %y
	%sum =l add %sum, %pr
	%t =l add %t, 1
	jmp @mm.t
@mm.store
	%ix =l mul %i, %n
	%ix =l add %ix, %j
	%off =l mul %ix, 8
	%p =l add %tmp, %off
	storel %sum, %p
	%j =l add %j, 1
	jmp @mm.j
@mm.i.next
	%i =l add %i, 1
	jmp @mm.i
@cp.init
	%c =l copy 0
	jmp @cp
@cp
	%cond =w csltl %c, %nn
	jnz %cond, @cp.body, @kdec
@cp.body
	%off =l mul %c, 8
	%pa =l add %tmp, %off
	%v =l loadl %pa
	%pb =l add %out, %off
	storel %v, %pb
	%c =l add %c, 1
	jmp @cp
@kdec
	%kk =l sub %kk, 1
	jmp @ktest
@ztmp
	%tot =l mul %nn, 2
	%c =l copy 0
	jmp @zt
@zt
	%cond =w csltl %c, %tot
	jnz %cond, @zt.body, @done
@zt.body
	%off =l mul %c, 8
	%p =l add %tmp, %off
	storel 0, %p
	%c =l add %c, 1
	jmp @zt
@done
	ret 0
}
