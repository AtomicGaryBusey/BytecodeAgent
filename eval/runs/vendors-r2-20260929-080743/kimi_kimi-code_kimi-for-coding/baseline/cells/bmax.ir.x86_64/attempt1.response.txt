export function l $bmax(l %a, l %n) {
@start
	%max =l loadl %a
	%bytes =l mul %n, 8
	%limit =l add %a, %bytes
	%p =l add %a, 8
@loop
	%cond =w csltl %p, %limit
	jnz %cond, @body, @done
@body
	%x =l loadl %p
	%gt =w csgtl %x, %max
	jnz %gt, @upd, @next
@upd
	%max =l copy %x
@next
	%p =l add %p, 8
	jmp @loop
@done
	ret %max
}
