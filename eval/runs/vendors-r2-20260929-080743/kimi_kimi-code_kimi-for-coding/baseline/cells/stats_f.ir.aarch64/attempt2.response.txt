export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %min_init =l copy 9223372036854775807
    %max_init =l copy -9223372036854775808
    %sum_init =l copy 0
    %odd_init =l copy 0
    %i_init =l copy 0
    jmp @loop
@loop
    %min =l phi @start %min_init, @iter %min_next
    %max =l phi @start %max_init, @iter %max_next
    %sum =l phi @start %sum_init, @iter %sum_next
    %odd =l phi @start %odd_init, @iter %odd_next
    %i =l phi @start %i_init, @iter %i_next
    %cont =w csltl %i, %n
    jnz %cont, @body, @done
@body
    %off =l mul %i, 8
    %addr =l add %a, %off
    %x =l loadl %addr
    %y =l call %f(l %x)
    %bit =l and %y, 1
    %lt =w csltl %y, %min
    jnz %lt, @min_take, @min_keep
@min_take
    jmp @min_merge
@min_keep
    jmp @min_merge
@min_merge
    %min_next =l phi @min_take %y, @min_keep %min
    %gt =w csgtl %y, %max
    jnz %gt, @max_take, @max_keep
@max_take
    jmp @max_merge
@max_keep
    jmp @max_merge
@max_merge
    %max_next =l phi @max_take %y, @max_keep %max
    %sum_next =l add %sum, %y
    %odd_next =l add %odd, %bit
    %i_next =l add %i, 1
    jmp @loop
@iter
    jmp @loop
@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %odd
}
