export function l $sdivrem(l %a, l %b, l %rem) {
@start
	%min =l shl 1, 63
	%iszero =w ceql %b, 0
	jnz %iszero, @zerodiv, @chkmin
@chkmin
	%ismin =w ceql %a, %min
	jnz %ismin, @negonechk, @normal
@negonechk
	%isneg1 =w ceql %b, -1
	jnz %isneg1, @overflow, @normal
@normal
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
@zerodiv
	storel %a, %rem
	ret 0
@overflow
	storel 0, %rem
	ret %min
}
