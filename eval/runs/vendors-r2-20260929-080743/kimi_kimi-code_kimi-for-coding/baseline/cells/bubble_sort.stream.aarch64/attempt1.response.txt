bubble_sort:
    cmp x1, #1
    b.le .Ldone
    sub x2, x1, #1
.Louter:
    mov x4, xzr
.Linner:
    ldr x6, [x0, x4, lsl #3]
    add x7, x4, #1
    ldr x8, [x0, x7, lsl #3]
    cmp x6, x8
    ble .Lnoswap
    str x8, [x0, x4, lsl #3]
    str x6, [x0, x7, lsl #3]
.Lnoswap:
    add x4, x4, #1
    cmp x4, x2
    b.lt .Linner
    subs x2, x2, #1
    b.ne .Louter
.Ldone:
    mov x0, xzr
    ret
