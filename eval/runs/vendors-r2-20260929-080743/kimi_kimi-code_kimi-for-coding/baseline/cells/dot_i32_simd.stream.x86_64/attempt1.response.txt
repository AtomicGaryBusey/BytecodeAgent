dot_i32_simd:
    pxor xmm6, xmm6
    pxor xmm7, xmm7
    mov rax, rdx
    shr rax, 3
    xor rcx, rcx
.Lloop8:
    test rax, rax
    jz .Lreduce
    movdqu xmm0, [rdi + rcx]
    movdqu xmm1, [rsi + rcx]
    movdqu xmm2, [rdi + rcx + 16]
    movdqu xmm3, [rsi + rcx + 16]
    pmovsxdq xmm4, xmm0
    pmovsxdq xmm5, xmm1
    pmuldq xmm4, xmm5
    paddq xmm6, xmm4
    pshufd xmm0, xmm0, 0xee
    pshufd xmm1, xmm1, 0xee
    pmovsxdq xmm4, xmm0
    pmovsxdq xmm5, xmm1
    pmuldq xmm4, xmm5
    paddq xmm6, xmm4
    pmovsxdq xmm4, xmm2
    pmovsxdq xmm5, xmm3
    pmuldq xmm4, xmm5
    paddq xmm7, xmm4
    pshufd xmm2, xmm2, 0xee
    pshufd xmm3, xmm3, 0xee
    pmovsxdq xmm4, xmm2
    pmovsxdq xmm5, xmm3
    pmuldq xmm4, xmm5
    paddq xmm7, xmm4
    add rcx, 32
    dec rax
    jnz .Lloop8
.Lreduce:
    paddq xmm6, xmm7
    pshufd xmm0, xmm6, 0x4e
    paddq xmm6, xmm0
    movq rax, xmm6
    mov r8, rdx
    and r8, 7
    test r8, r8
    jz .Lret
    lea r9, [rdi + rcx]
    lea r10, [rsi + rcx]
    xor rcx, rcx
.Lscalar:
    mov edx, dword ptr [r9 + rcx*4]
    movsxd rdx, edx
    mov r11d, dword ptr [r10 + rcx*4]
    movsxd r11, r11d
    imul rdx, r11
    add rax, rdx
    inc rcx
    cmp rcx, r8
    jb .Lscalar
.Lret:
    ret
