export function l $count_ugt(l %a, l %n, l %t) {
@start
        %zero =l copy 0
        %ap =l copy %a
        jmp @loop
@loop
        %i =l phi @start %zero, @next %i2
        %p =l phi @start %ap, @next %p2
        %c =l phi @start %zero, @next %c2
        %go =w csltl %i, %n
        jnz %go, @body, @done
@body
        %v =l loadl %p
        %gt =w cugtl %v, %t
        jnz %gt, @inc, @next
@inc
        %c1 =l add %c, 1
        jmp @next
@next
        %c2 =l phi @body %c, @inc %c1
        %p2 =l add %p, 8
        %i2 =l add %i, 1
        jmp @loop
@done
        ret %c
}
