export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%fwd =w csltl %dst, %src
	jnz %fwd, @fwd_init, @bwd_init

@fwd_init
	%i =l copy 0
	jmp @fwd_cond

@fwd_cond
	%fc =w csltl %i, %n
	jnz %fc, @fwd_body, @done

@fwd_body
	%foff =l mul %i, 8
	%fsa =l add %src, %foff
	%fv =l loadl %fsa
	%fda =l add %dst, %foff
	storel %fv, %fda
	%i =l add %i, 1
	jmp @fwd_cond

@bwd_init
	%j =l sub %n, 1
	jmp @bwd_cond

@bwd_cond
	%bc =w csgel %j, 0
	jnz %bc, @bwd_body, @done

@bwd_body
	%boff =l mul %j, 8
	%bsa =l add %src, %boff
	%bv =l loadl %bsa
	%bda =l add %dst, %boff
	storel %bv, %bda
	%j =l sub %j, 1
	jmp @bwd_cond

@done
	ret 0
}
