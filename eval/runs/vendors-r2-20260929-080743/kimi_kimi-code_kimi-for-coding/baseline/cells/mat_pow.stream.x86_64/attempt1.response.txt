    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    mov rbx, r8
    mov rbp, r13
    imul rbp, r13
    lea rbp, [rbx + rbp*8]
    mov rdi, rbx
    mov rsi, r12
    mov rcx, r13
    imul rcx, r13
    rep movsq
    mov rdi, r15
    mov rcx, r13
    imul rcx, r13
    xor eax, eax

    xor ecx, ecx
.Lident:
    mov rax, rcx
    imul rax, r13
    add rax, rcx
    mov qword ptr [r15 + rax*8], 1
    inc rcx
    cmp rcx, r13
    jl .Lident
.Lloop:
    test r14, r14
    jz .Ldone
    test r14, 1
    jz .Lshift
    mov rdi, r15
    mov rsi, rbx
    mov rdx, rbp
    lea r12, [rip + .Lret1]
    jmp .Lmul
.Lret1:
    mov rdi, r15
    mov rsi, rbp
    mov rcx, r13
    imul rcx, r13
    rep movsq
.Lshift:
    shr r14, 1
    jz .Ldone
    mov rdi, rbx
    mov rsi, rbx
    mov rdx, rbp
    lea r12, [rip + .Lret2]
    jmp .Lmul
.Lret2:
    mov rdi, rbx
    mov rsi, rbp
    mov rcx, r13
    imul rcx, r13
    rep movsq
    jmp .Lloop
.Ldone:
    mov rdi, rbx
    mov rcx, r13
    imul rcx, r13
    add rcx, rcx
    xor eax, eax
    rep stosq
    xor eax, eax
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
.Lmul:
    xor r8d, r8d
.Lmul_i:
    xor r9d, r9d
.Lmul_j:
    xor r10d, r10d
    xor r11d, r11d
.Lmul_t:
    mov rcx, r8
    imul rcx, r13
    add rcx, r11
    mov rax, qword ptr [rdi + rcx*8]
    mov rcx, r11
    imul rcx, r13
    add rcx, r9
    imul rax, qword ptr [rsi + rcx*8]
    add r10, rax
    inc r11
    cmp r11, r13
    jl .Lmul_t
    mov rcx, r8
    imul rcx, r13
    add rcx, r9
    mov qword ptr [rdx + rcx*8], r10
    inc r9
    cmp r9, r13
    jl .Lmul_j
    inc r8
    cmp r8, r13
    jl .Lmul_i
    jmp r12
