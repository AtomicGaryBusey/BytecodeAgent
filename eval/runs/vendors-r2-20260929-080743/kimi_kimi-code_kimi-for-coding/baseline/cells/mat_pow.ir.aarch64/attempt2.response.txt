export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@entry
	%nn =l mul %n, %n
	%i =l copy 0
@zloop
	%zc =w csltl %i, %nn
	jnz %zc, @zbody, @zdone
@zbody
	%za =l mul %i, 8
	%zp =l add %out, %za
	storel 0, %zp
	%i =l add %i, 1
	jmp @zloop
@zdone
	%d =l copy 0
@dloop
	%dc =w csltl %d, %n
	jnz %dc, @dbody, @ddone
@dbody
	%di =l mul %d, %n
	%di =l add %di, %d
	%da =l mul %di, 8
	%dp =l add %out, %da
	storel 1, %dp
	%d =l add %d, 1
	jmp @dloop
@ddone
	%e =l copy 0
@eloop
	%ec =w csltl %e, %k
	jnz %ec, @ebody, @edone
@ebody
	%mi =l copy 0
@mi_loop
	%mic =w csltl %mi, %n
	jnz %mic, @mj_init, @ebody_copy
@mj_init
	%mj =l copy 0
@mj_loop
	%mjc =w csltl %mj, %n
	jnz %mjc, @mt_init, @mj_next
@mt_init
	%acc =l copy 0
	%mt =l copy 0
@mt_loop
	%mtc =w csltl %mt, %n
	jnz %mtc, @mt_body, @mt_done
@mt_body
	%lidx =l mul %mi, %n
	%lidx =l add %lidx, %mt
	%laddr =l mul %lidx, 8
	%lp =l add %out, %laddr
	%lv =l loadl %lp
	%ridx =l mul %mt, %n
	%ridx =l add %ridx, %mj
	%raddr =l mul %ridx, 8
	%rp =l add %a, %raddr
	%rv =l loadl %rp
	%prod =l mul %lv, %rv
	%acc =l add %acc, %prod
	%mt =l add %mt, 1
	jmp @mt_loop
@mt_done
	%didx =l mul %mi, %n
	%didx =l add %didx, %mj
	%daddr =l mul %didx, 8
	%dp2 =l add %tmp, %daddr
	storel %acc, %dp2
	%mj =l add %mj, 1
	jmp @mj_loop
@mj_next
	%mi =l add %mi, 1
	jmp @mi_loop
@ebody_copy
	%ci =l copy 0
@cloop
	%cc =w csltl %ci, %nn
	jnz %cc, @cbody, @cdone
@cbody
	%ca =l mul %ci, 8
	%cps =l add %tmp, %ca
	%cv =l loadl %cps
	%cpd =l add %out, %ca
	storel %cv, %cpd
	%ci =l add %ci, 1
	jmp @cloop
@cdone
	%e =l add %e, 1
	jmp @eloop
@edone
	%zl =l mul %nn, 2
	%zi =l copy 0
@z2loop
	%z2c =w csltl %zi, %zl
	jnz %z2c, @z2body, @z2done
@z2body
	%z2a =l mul %zi, 8
	%z2p =l add %tmp, %z2a
	storel 0, %z2p
	%zi =l add %zi, 1
	jmp @z2loop
@z2done
	ret 0
}
