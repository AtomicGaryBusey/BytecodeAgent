export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%sp =l copy 0
	jmp @loop
@loop
	%done =w ceql %pc, %len
	jnz %done, @finish, @fetch
@fetch
	%pm =l mul %pc, 8
	%pa =l add %code, %pm
	%op =l loadl %pa
	%is0 =w ceql %op, 0
	jnz %is0, @finish, @k1
@k1
	%is1 =w ceql %op, 1
	jnz %is1, @opush, @k2
@k2
	%is2 =w ceql %op, 2
	jnz %is2, @oadd, @k3
@k3
	%is3 =w ceql %op, 3
	jnz %is3, @osub, @k4
@k4
	%is4 =w ceql %op, 4
	jnz %is4, @omul, @k5
@k5
	%is5 =w ceql %op, 5
	jnz %is5, @odup, @k6
@k6
	%is6 =w ceql %op, 6
	jnz %is6, @oswap, @k7
@k7
	%is7 =w ceql %op, 7
	jnz %is7, @odrop, @k8
@k8
	%is8 =w ceql %op, 8
	jnz %is8, @oover, @k9
@k9
	%is9 =w ceql %op, 9
	jnz %is9, @olt, @k10
@k10
	%is10 =w ceql %op, 10
	jnz %is10, @ojz, @k11
@k11
	%is11 =w ceql %op, 11
	jnz %is11, @ojmp, @other
@opush
	%po =l add %pc, 1
	%pom =l mul %po, 8
	%poa =l add %code, %pom
	%pv =l loadl %poa
	%sm =l mul %sp, 8
	%sa =l add %stack, %sm
	storel %pv, %sa
	%sp =l add %sp, 1
	%pc =l add %pc, 2
	jmp @loop
@oadd
	%ai =l sub %sp, 2
	%bi =l sub %sp, 1
	%am =l mul %ai, 8
	%aa =l add %stack, %am
	%bm =l mul %bi, 8
	%ba =l add %stack, %bm
	%a =l loadl %aa
	%b =l loadl %ba
	%r =l add %a, %b
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@osub
	%ai =l sub %sp, 2
	%bi =l sub %sp, 1
	%am =l mul %ai, 8
	%aa =l add %stack, %am
	%bm =l mul %bi, 8
	%ba =l add %stack, %bm
	%a =l loadl %aa
	%b =l loadl %ba
	%r =l sub %a, %b
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@omul
	%ai =l sub %sp, 2
	%bi =l sub %sp, 1
	%am =l mul %ai, 8
	%aa =l add %stack, %am
	%bm =l mul %bi, 8
	%ba =l add %stack, %bm
	%a =l loadl %aa
	%b =l loadl %ba
	%r =l mul %a, %b
	storel %r, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@odup
	%ti =l sub %sp, 1
	%tm =l mul %ti, 8
	%ta =l add %stack, %tm
	%tv =l loadl %ta
	%sm =l mul %sp, 8
	%sa =l add %stack, %sm
	storel %tv, %sa
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@oswap
	%ai =l sub %sp, 2
	%bi =l sub %sp, 1
	%am =l mul %ai, 8
	%aa =l add %stack, %am
	%bm =l mul %bi, 8
	%ba =l add %stack, %bm
	%a =l loadl %aa
	%b =l loadl %ba
	storel %b, %aa
	storel %a, %ba
	%pc =l add %pc, 1
	jmp @loop
@odrop
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@oover
	%ti =l sub %sp, 2
	%tm =l mul %ti, 8
	%ta =l add %stack, %tm
	%tv =l loadl %ta
	%sm =l mul %sp, 8
	%sa =l add %stack, %sm
	storel %tv, %sa
	%sp =l add %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@olt
	%ai =l sub %sp, 2
	%bi =l sub %sp, 1
	%am =l mul %ai, 8
	%aa =l add %stack, %am
	%bm =l mul %bi, 8
	%ba =l add %stack, %bm
	%a =l loadl %aa
	%b =l loadl %ba
	%c =w csltl %a, %b
	%cl =l extsw %c
	storel %cl, %aa
	%sp =l sub %sp, 1
	%pc =l add %pc, 1
	jmp @loop
@ojz
	%po =l add %pc, 1
	%pom =l mul %po, 8
	%poa =l add %code, %pom
	%t =l loadl %poa
	%ti =l sub %sp, 1
	%tm =l mul %ti, 8
	%ta =l add %stack, %tm
	%v =l loadl %ta
	%sp =l sub %sp, 1
	%isz =w ceql %v, 0
	%npc =l add %pc, 2
	jnz %isz, @jzt, @jzns
@jzt
	%pc =l copy %t
	jmp @loop
@jzns
	%pc =l copy %npc
	jmp @loop
@ojmp
	%po =l add %pc, 1
	%pom =l mul %po, 8
	%poa =l add %code, %pom
	%t =l loadl %poa
	%pc =l copy %t
	jmp @loop
@other
	%pc =l add %pc, 1
	jmp @loop
@finish
	%zi =l copy %sp
	jmp @ztest
@ztest
	%more =w csltl %zi, %cap
	jnz %more, @zbody, @ret
@zbody
	%zm =l mul %zi, 8
	%za =l add %stack, %zm
	storel 0, %za
	%zi =l add %zi, 1
	jmp @ztest
@ret
	%has =w csgtl %sp, 0
	jnz %has, @havetop, @empty
@havetop
	%rt =l sub %sp, 1
	%rm =l mul %rt, 8
	%ra =l add %stack, %rm
	%res =l loadl %ra
	ret %res
@empty
	ret 0
}
