export function l $binary_search(l %a, l %n, l %key) {
@start
	%lo =l copy 0
	%hi =l copy %n
	jmp @loop
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @fail
@body
	%s =l add %lo, %hi
	%m =l div %s, 2
	%off =l mul %m, 8
	%p =l add %a, %off
	%v =l loadl %p
	%eq =w ceql %v, %key
	jnz %eq, @found, @cmp
@found
	ret %m
@cmp
	%lt =w csltl %v, %key
	jnz %lt, @right, @left
@right
	%lo =l add %m, 1
	jmp @loop
@left
	%hi =l copy %m
	jmp @loop
@fail
	ret -1
}
