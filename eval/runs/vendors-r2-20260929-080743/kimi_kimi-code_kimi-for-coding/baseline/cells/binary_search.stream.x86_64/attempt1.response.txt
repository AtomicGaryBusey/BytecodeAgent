binary_search:
    xor r8, r8
    mov r9, rsi
.Lloop:
    cmp r8, r9
    jge .Lnot_found
    mov rcx, r9
    sub rcx, r8
    shr rcx, 1
    add rcx, r8
    mov rax, qword ptr [rdi + rcx*8]
    cmp rax, rdx
    je .Lfound
    jl .Lgo_right
    mov r9, rcx
    jmp .Lloop
.Lgo_right:
    lea r8, qword ptr [rcx + 1]
    jmp .Lloop
.Lfound:
    mov rax, rcx
    ret
.Lnot_found:
    mov rax, -1
    ret
