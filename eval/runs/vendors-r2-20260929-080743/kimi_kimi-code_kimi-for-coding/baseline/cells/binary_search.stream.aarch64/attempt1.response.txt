binary_search:
    mov x9, #0              // low = 0
    mov x10, x1             // high = n
.Lloop:
    cmp x9, x10             // while low < high
    b.ge .Lnot_found
    add x11, x9, x10        // mid = (low + high) / 2
    lsr x11, x11, #1
    ldr x12, [x0, x11, lsl #3]  // v = a[mid]
    cmp x12, x2
    beq .Lfound
    b.lt .Lgo_right         // a[mid] < key -> low = mid + 1
    mov x10, x11            // a[mid] > key -> high = mid
    b .Lloop
.Lgo_right:
    add x9, x11, #1
    b .Lloop
.Lfound:
    mov x0, x11
    ret
.Lnot_found:
    mov x0, #-1
    ret
