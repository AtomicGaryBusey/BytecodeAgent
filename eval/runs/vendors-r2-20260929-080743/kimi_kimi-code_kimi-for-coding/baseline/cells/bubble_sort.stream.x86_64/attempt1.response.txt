bubble_sort:
    lea rcx, qword ptr [rsi - 1]
.Louter:
    test rcx, rcx
    jle .Ldone
    xor edx, edx
    mov r8, rcx
.Linner:
    mov rax, qword ptr [rdi + rdx*8]
    mov r9, qword ptr [rdi + rdx*8 + 8]
    cmp rax, r9
    jle .Lnoswap
    mov qword ptr [rdi + rdx*8], r9
    mov qword ptr [rdi + rdx*8 + 8], rax
.Lnoswap:
    inc rdx
    dec r8
    jnz .Linner
    dec rcx
    jmp .Louter
.Ldone:
    xor eax, eax
    ret
