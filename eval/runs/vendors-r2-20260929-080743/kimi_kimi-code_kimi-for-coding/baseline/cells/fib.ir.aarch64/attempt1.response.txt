export function l $fib(l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @retn, @loop
@retn
	ret %n
@loop
	%i =l phi @start 1, @body %ni
	%prev =l phi @start 0, @body %np
	%cur =l phi @start 1, @body %nc
	%go =w csltl %i, %n
	jnz %go, @body, @done
@body
	%nc =l add %prev, %cur
	%ni =l add %i, 1
	%np =l copy %cur
	jmp @loop
@done
	ret %cur
}
