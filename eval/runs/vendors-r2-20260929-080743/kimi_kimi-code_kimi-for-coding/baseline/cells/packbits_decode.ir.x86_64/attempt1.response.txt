export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%pos =l copy 0
	%w =l copy 0
	jmp @check
@check
	%more =w csltl %pos, %n
	jnz %more, @head, @done
@head
	%hp =l add %src, %pos
	%hb =w loadsb %hp
	%pos =l add %pos, 1
	%h =l extsw %hb
	%isnop =w ceql %h, -128
	jnz %isnop, @check, @notnop
@notnop
	%islit =w csgel %h, 0
	jnz %islit, @lit, @rep
@lit
	%len =l add %h, 1
	%rem =l sub %n, %pos
	%srcck =w csgel %rem, %len
	jnz %srcck, @litcap, @errsrc
@litcap
	%room =l sub %cap, %w
	%capck =w csgel %room, %len
	jnz %capck, @litfull, @litpart
@litfull
	%i =l copy 0
	jmp @litfl
@litfl
	%ifin =w ceql %i, %len
	jnz %ifin, @litfd, @litfs
@litfs
	%sp =l add %src, %pos
	%sp =l add %sp, %i
	%lb =w loadub %sp
	%op =l add %out, %w
	%op =l add %op, %i
	storeb %lb, %op
	%i =l add %i, 1
	jmp @litfl
@litfd
	%pos =l add %pos, %len
	%w =l add %w, %len
	jmp @check
@litpart
	%i =l copy 0
	jmp @litpl
@litpl
	%ipfin =w ceql %i, %room
	jnz %ipfin, @errcap, @litps
@litps
	%sp =l add %src, %pos
	%sp =l add %sp, %i
	%lb =w loadub %sp
	%op =l add %out, %w
	%op =l add %op, %i
	storeb %lb, %op
	%i =l add %i, 1
	jmp @litpl
@rep
	%rem =l sub %n, %pos
	%srcck =w csgel %rem, 1
	jnz %srcck, @repc, @errsrc
@repc
	%bp =l add %src, %pos
	%cb =w loadub %bp
	%pos =l add %pos, 1
	%cnt =l sub 1, %h
	%room =l sub %cap, %w
	%capck =w csgel %room, %cnt
	jnz %capck, @repfull, @reppart
@repfull
	%j =l copy 0
	jmp @repfl
@repfl
	%jfin =w ceql %j, %cnt
	jnz %jfin, @repfd, @repfs
@repfs
	%op =l add %out, %w
	%op =l add %op, %j
	storeb %cb, %op
	%j =l add %j, 1
	jmp @repfl
@repfd
	%w =l add %w, %cnt
	jmp @check
@reppart
	%j =l copy 0
	jmp @reppl
@reppl
	%jpfin =w ceql %j, %room
	jnz %jpfin, @errcap, @repps
@repps
	%op =l add %out, %w
	%op =l add %op, %j
	storeb %cb, %op
	%j =l add %j, 1
	jmp @reppl
@errsrc
	ret -2
@errcap
	ret -1
@done
	ret %w
}
