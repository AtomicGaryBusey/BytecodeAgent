export function l $bubble_sort(l %a, l %n) {
@start
	jmp @outer(0)
@outer(%i =l)
	%c0 =w csltl %i, %n
	jnz %c0, @body, @ret
@body
	%t =l sub %n, 1
	%lim =l sub %t, %i
	jmp @inner(0)
@inner(%j =l)
	%c1 =w csltl %j, %lim
	jnz %c1, @elem, @next
@elem
	%off =l mul %j, 8
	%pa =l add %a, %off
	%pb =l add %pa, 8
	%x =l loadl %pa
	%y =l loadl %pb
	%c2 =w csgtl %x, %y
	jnz %c2, @swap, @cont
@swap
	storel %y, %pa
	storel %x, %pb
@cont
	%j1 =l add %j, 1
	jmp @inner(%j1)
@next
	%i1 =l add %i, 1
	jmp @outer(%i1)
@ret
	ret 0
}
