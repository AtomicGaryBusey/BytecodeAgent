export function l $bubble_sort(l %a, l %n) {
@start
	%c0 =w ceql %n, 0
	jnz %c0, @ret, @outer_init
@outer_init
	%i =l copy 0
	jmp @outer_test
@outer_test
	%c1 =w csltl %i, %n
	jnz %c1, @inner_init, @ret
@inner_init
	%j =l copy 0
	%lim =l sub %n, %i
	%lim =l sub %lim, 1
	jmp @inner_test
@inner_test
	%c2 =w csltl %j, %lim
	jnz %c2, @compare, @outer_inc
@compare
	%off =l mul %j, 8
	%pa =l add %a, %off
	%pb =l add %pa, 8
	%x =l loadl %pa
	%y =l loadl %pb
	%c3 =w csgtl %x, %y
	jnz %c3, @swap, @inner_inc
@swap
	storel %y, %pa
	storel %x, %pb
	jmp @inner_inc
@inner_inc
	%j =l add %j, 1
	jmp @inner_test
@outer_inc
	%i =l add %i, 1
	jmp @outer_test
@ret
	%r =l copy 0
	ret %r
}
