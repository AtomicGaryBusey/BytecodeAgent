export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@entry
	jmp @fetch

@fetch
	%pc =l phi @entry 0, @bp %pc_p, @ba %pc_a, @bs %pc_s, @bm %pc_m, @bd %pc_d, @bw %pc_w, @bx %pc_x, @bo %pc_o, @bl %pc_l, @jt %pc_jt, @jf %pc_jf, @bj %pc_bj, @bn %pc_n
	%sp =l phi @entry 0, @bp %sp_p, @ba %sp_a, @bs %sp_s, @bm %sp_m, @bd %sp_d, @bw %sp_w, @bx %sp_x, @bo %sp_o, @bl %sp_l, @jt %sp_j, @jf %sp_j, @bj %sp, @bn %sp
	%at_end =w ceql %pc, %len
	jnz %at_end, @done, @decode

@decode
	%ioff =l mul %pc, 8
	%iaddr =l add %code, %ioff
	%op =l loadl %iaddr
	%eq0 =w ceql %op, 0
	jnz %eq0, @done, @k1

@k1
	%eq1 =w ceql %op, 1
	jnz %eq1, @bp, @k2

@k2
	%eq2 =w ceql %op, 2
	jnz %eq2, @ba, @k3

@k3
	%eq3 =w ceql %op, 3
	jnz %eq3, @bs, @k4

@k4
	%eq4 =w ceql %op, 4
	jnz %eq4, @bm, @k5

@k5
	%eq5 =w ceql %op, 5
	jnz %eq5, @bd, @k6

@k6
	%eq6 =w ceql %op, 6
	jnz %eq6, @bw, @k7

@k7
	%eq7 =w ceql %op, 7
	jnz %eq7, @bx, @k8

@k8
	%eq8 =w ceql %op, 8
	jnz %eq8, @bo, @k9

@k9
	%eq9 =w ceql %op, 9
	jnz %eq9, @bl, @k10

@k10
	%eq10 =w ceql %op, 10
	jnz %eq10, @bz, @k11

@k11
	%eq11 =w ceql %op, 11
	jnz %eq11, @bj, @bn

@bp
	%poff =l add %ioff, 8
	%paddr =l add %code, %poff
	%val =l loadl %paddr
	%psoff =l mul %sp, 8
	%psaddr =l add %stack, %psoff
	storel %val, %psaddr
	%sp_p =l add %sp, 1
	%pc_p =l add %pc, 2
	jmp @fetch

@ba
	%ab1 =l sub %sp, 1
	%ab1o =l mul %ab1, 8
	%ab1a =l add %stack, %ab1o
	%bv =l loadl %ab1a
	%ab2 =l sub %sp, 2
	%ab2o =l mul %ab2, 8
	%ab2a =l add %stack, %ab2o
	%av =l loadl %ab2a
	%arv =l add %av, %bv
	storel %arv, %ab2a
	%sp_a =l sub %sp, 1
	%pc_a =l add %pc, 1
	jmp @fetch

@bs
	%sb1 =l sub %sp, 1
	%sb1o =l mul %sb1, 8
	%sb1a =l add %stack, %sb1o
	%bsv =l loadl %sb1a
	%sb2 =l sub %sp, 2
	%sb2o =l mul %sb2, 8
	%sb2a =l add %stack, %sb2o
	%asv =l loadl %sb2a
	%srv =l sub %asv, %bsv
	storel %srv, %sb2a
	%sp_s =l sub %sp, 1
	%pc_s =l add %pc, 1
	jmp @fetch

@bm
	%mb1 =l sub %sp, 1
	%mb1o =l mul %mb1, 8
	%mb1a =l add %stack, %mb1o
	%bmv =l loadl %mb1a
	%mb2 =l sub %sp, 2
	%mb2o =l mul %mb2, 8
	%mb2a =l add %stack, %mb2o
	%amv =l loadl %mb2a
	%mrv =l mul %amv, %bmv
	storel %mrv, %mb2a
	%sp_m =l sub %sp, 1
	%pc_m =l add %pc, 1
	jmp @fetch

@bd
	%db1 =l sub %sp, 1
	%db1o =l mul %db1, 8
	%db1a =l add %stack, %db1o
	%dv =l loadl %db1a
	%dso =l mul %sp, 8
	%dsa =l add %stack, %dso
	storel %dv, %dsa
	%sp_d =l add %sp, 1
	%pc_d =l add %pc, 1
	jmp @fetch

@bw
	%wb1 =l sub %sp, 1
	%wb1o =l mul %wb1, 8
	%wb1a =l add %stack, %wb1o
	%w1v =l loadl %wb1a
	%wb2 =l sub %sp, 2
	%wb2o =l mul %wb2, 8
	%wb2a =l add %stack, %wb2o
	%w2v =l loadl %wb2a
	storel %w2v, %wb1a
	storel %w1v, %wb2a
	%pc_w =l add %pc, 1
	%sp_w =l copy %sp
	jmp @fetch

@bx
	%sp_x =l sub %sp, 1
	%pc_x =l add %pc, 1
	jmp @fetch

@bo
	%ob2 =l sub %sp, 2
	%ob2o =l mul %ob2, 8
	%ob2a =l add %stack, %ob2o
	%ov =l loadl %ob2a
	%oso =l mul %sp, 8
	%osa =l add %stack, %oso
	storel %ov, %osa
	%sp_o =l add %sp, 1
	%pc_o =l add %pc, 1
	jmp @fetch

@bl
	%lb1 =l sub %sp, 1
	%lb1o =l mul %lb1, 8
	%lb1a =l add %stack, %lb1o
	%lbv =l loadl %lb1a
	%lb2 =l sub %sp, 2
	%lb2o =l mul %lb2, 8
	%lb2a =l add %stack, %lb2o
	%lav =l loadl %lb2a
	%ltw =w csltl %lav, %lbv
	%ltl =l extsw %ltw
	storel %ltl, %lb2a
	%sp_l =l sub %sp, 1
	%pc_l =l add %pc, 1
	jmp @fetch

@bz
	%zoff =l add %ioff, 8
	%zaddr =l add %code, %zoff
	%ztgt =l loadl %zaddr
	%zb1 =l sub %sp, 1
	%zb1o =l mul %zb1, 8
	%zb1a =l add %stack, %zb1o
	%zv =l loadl %zb1a
	%sp_j =l sub %sp, 1
	%isz =w ceql %zv, 0
	jnz %isz, @jt, @jf

@jt
	%pc_jt =l copy %ztgt
	jmp @fetch

@jf
	%pc_jf =l add %pc, 2
	jmp @fetch

@bj
	%joff =l add %ioff, 8
	%jaddr =l add %code, %joff
	%pc_bj =l loadl %jaddr
	jmp @fetch

@bn
	%pc_n =l add %pc, 1
	jmp @fetch

@done
	%dsp =l phi @fetch %sp, @decode %sp
	%empty =w ceql %dsp, 0
	jnz %empty, @rz, @rt

@rt
	%tidx =l sub %dsp, 1
	%toff =l mul %tidx, 8
	%taddr =l add %stack, %toff
	%res_t =l loadl %taddr
	jmp @zinit

@rz
	%res_z =l copy 0
	jmp @zinit

@zinit
	%res =l phi @rt %res_t, @rz %res_z
	%zi0 =l copy %dsp
	jmp @zcond

@zcond
	%zi =l phi @zinit %zi0, @zbody %zi2
	%zlt =w csltl %zi, %cap
	jnz %zlt, @zbody, @fin

@zbody
	%zboff =l mul %zi, 8
	%zbaddr =l add %stack, %zboff
	storel 0, %zbaddr
	%zi2 =l add %zi, 1
	jmp @zcond

@fin
	ret %res
}
