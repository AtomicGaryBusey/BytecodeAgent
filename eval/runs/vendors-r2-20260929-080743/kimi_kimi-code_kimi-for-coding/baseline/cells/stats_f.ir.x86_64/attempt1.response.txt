export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
        %i =l copy 0
        %one =l copy 1
        %x0 =l loadl %a
        %y0 =l call %f(l %x0)
        %min =l copy %y0
        %max =l copy %y0
        %sum =l copy %y0
        %odd0 =l and %y0, 1
        %cnt =l copy %odd0
        %i =l add %i, %one
        jmp @cond
@cond
        %more =w csltl %i, %n
        jnz %more, @body, @done
@body
        %off =l mul %i, 8
        %p =l add %a, %off
        %x =l loadl %p
        %y =l call %f(l %x)
        %cm =w csltl %y, %min
        jnz %cm, @setmin, @checkmax
@setmin
        %min =l copy %y
        jmp @checkmax
@checkmax
        %cx =w csgtl %y, %max
        jnz %cx, @setmax, @accum
@setmax
        %max =l copy %y
        jmp @accum
@accum
        %sum =l add %sum, %y
        %odd =l and %y, 1
        %cnt =l add %cnt, %odd
        %i =l add %i, %one
        jmp @cond
@done
        storel %min, %out
        %out1 =l add %out, 8
        storel %max, %out1
        %out2 =l add %out, 16
        storel %sum, %out2
        ret %cnt
}
