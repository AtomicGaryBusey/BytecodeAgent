stats_f:
    stp x29, x30, [sp, #-96]!
    stp x19, x20, [sp, #16]
    stp x21, x22, [sp, #32]
    stp x23, x24, [sp, #48]
    stp x25, x26, [sp, #64]
    stp x27, x28, [sp, #80]
    mov x19, x2             // f
    mov x20, x0             // a (advanced by 8 each iteration)
    mov x26, x3             // out
    sub x27, x1, #1         // remaining elements after a[0]
    // process a[0]
    ldr x0, [x20]
    blr x19
    mov x22, x0             // min
    mov x23, x0             // max
    mov x24, x0             // sum
    and x25, x0, #1         // odd count
    add x20, x20, #8
.Lloop:
    cbz x27, .Ldone
    ldr x0, [x20]
    blr x19                 // y = f(a[i]) in x0
    cmp x0, x22
    csel x22, x0, x22, lt   // min = y < min ? y : min
    cmp x0, x23
    csel x23, x0, x23, gt   // max = y > max ? y : max
    add x24, x24, x0        // sum += y (wrapping)
    and x9, x0, #1
    add x25, x25, x9        // count odd
    add x20, x20, #8
    sub x27, x27, #1
    b .Lloop
.Ldone:
    stp x22, x23, [x26]     // out[0] = min, out[1] = max
    str x24, [x26, #16]     // out[2] = sum
    mov x0, x25             // return odd count
    ldp x27, x28, [sp, #80]
    ldp x25, x26, [sp, #64]
    ldp x23, x24, [sp, #48]
    ldp x21, x22, [sp, #32]
    ldp x19, x20, [sp, #16]
    ldp x29, x30, [sp], #96
    ret
