export function l $isqrt(l %n) {
@start
	%lo =l copy 0
	%hi =l copy 4294967295
	jmp @loop
@loop
	%c =w csltl %lo, %hi
	jnz %c, @body, @ret
@body
	%t =l add %lo, %hi
	%t1 =l add %t, 1
	%mid =l shr %t1, 1
	%sq =l mul %mid, %mid
	%le =w culel %sq, %n
	jnz %le, @take, @skip
@take
	%lo =l copy %mid
	jmp @loop
@skip
	%hm =l sub %mid, 1
	%hi =l copy %hm
	jmp @loop
@ret
	ret %lo
}
