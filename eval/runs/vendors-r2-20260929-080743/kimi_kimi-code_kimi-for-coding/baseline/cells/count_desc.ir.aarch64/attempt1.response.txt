export function l $count_desc(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @zero, @entry
@entry
	%nm1 =l sub %n, 1
	jmp @loop
@loop
	%cnt =l phi @entry 0, @body %cnt1
	%i =l phi @entry 0, @body %i1
	%go =w csltl %i, %nm1
	jnz %go, @body, @done
@body
	%off =l shl %i, 3
	%p =l add %a, %off
	%x =l loadl %p
	%p1 =l add %p, 8
	%y =l loadl %p1
	%gt =w csgtl %x, %y
	%inc =l extsw %gt
	%cnt1 =l add %cnt, %inc
	%i1 =l add %i, 1
	jmp @loop
@done
	ret %cnt
@zero
	ret 0
}
