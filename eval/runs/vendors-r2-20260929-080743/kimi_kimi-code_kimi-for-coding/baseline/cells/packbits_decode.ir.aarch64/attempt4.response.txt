export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	%z =l copy 0
	%m1 =l sub %z, 1
	%m2 =l sub %z, 2
	jmp @loop

@loop
	%spos =l phi @start %z, @litcf %sposa, @repcf %sposb, @nop %sposc
	%opos =l phi @start %z, @litcf %oposa, @repcf %oposb, @nop %oposc
	%more =w csltl %spos, %n
	jnz %more, @head, @fin

@fin
	ret %opos

@head
	%hp =l add %src, %spos
	%h =w loadsb %hp
	%isneg =w csltw %h, 0
	jnz %isneg, @neg, @lit

@lit
	%hl =l extsw %h
	%need =l add %hl, 2
	%avail =l sub %n, %spos
	%srcok =w csgel %avail, %need
	jnz %srcok, @litroom, @errsrc

@litroom
	%len =l add %hl, 1
	%room =l sub %cap, %opos
	%fits =w csgel %room, %len
	jnz %fits, @litcopy, @littrunc

@litcopy
	%i =l phi @litroom 0, @litcb %in
	%go =w csltl %i, %len
	jnz %go, @litcb, @litcf

@litcb
	%lsa =l add %spos, 1
	%lsb =l add %lsa, %i
	%lsp =l add %src, %lsb
	%lby =w loadub %lsp
	%loa =l add %out, %opos
	%lob =l add %loa, %i
	storeb %lby, %lob
	%in =l add %i, 1
	jmp @litcopy

@litcf
	%adv =l add %len, 1
	%sposa =l add %spos, %adv
	%oposa =l add %opos, %len
	jmp @loop

@littrunc
	%j =l phi @litroom 0, @littb %jn
	%go2 =w csltl %j, %room
	jnz %go2, @littb, @littf

@littb
	%tsa =l add %spos, 1
	%tsb =l add %tsa, %j
	%tsp =l add %src, %tsb
	%tby =w loadub %tsp
	%toa =l add %out, %opos
	%tob =l add %toa, %j
	storeb %tby, %tob
	%jn =l add %j, 1
	jmp @littrunc

@littf
	ret %m1

@errsrc
	ret %m2

@neg
	%h128 =w add %h, 128
	%isnop =w ceqw %h128, 0
	jnz %isnop, @nop, @rep

@nop
	%sposc =l add %spos, 1
	%oposc =l copy %opos
	jmp @loop

@rep
	%avail2 =l sub %n, %spos
	%srcok2 =w csgel %avail2, 2
	jnz %srcok2, @reproom, @errsrc

@reproom
	%hl2 =l extsw %h
	%len2 =l sub 1, %hl2
	%room2 =l sub %cap, %opos
	%fits2 =w csgel %room2, %len2
	jnz %fits2, @repcopy, @reptrunc

@repcopy
	%k =l phi @reproom 0, @repcb %kn
	%vp =l add %hp, 1
	%val =w loadub %vp
	%go3 =w csltl %k, %len2
	jnz %go3, @repcb, @repcf

@repcb
	%roa =l add %out, %opos
	%rob =l add %roa, %k
	storeb %val, %rob
	%kn =l add %k, 1
	jmp @repcopy

@repcf
	%sposb =l add %spos, 2
	%oposb =l add %opos, %len2
	jmp @loop

@reptrunc
	%m =l phi @reproom 0, @reptb %mn
	%vp2 =l add %hp, 1
	%val2 =w loadub %vp2
	%go4 =w csltl %m, %room2
	jnz %go4, @reptb, @reptf

@reptb
	%uoa =l add %out, %opos
	%uob =l add %uoa, %m
	storeb %val2, %uob
	%mn =l add %m, 1
	jmp @reptrunc

@reptf
	ret %m1
}
