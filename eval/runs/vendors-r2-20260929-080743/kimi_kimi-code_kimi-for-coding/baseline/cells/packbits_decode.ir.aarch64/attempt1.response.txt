export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @loop(0, 0)

@loop(l %si, l %oi)
	%more =w csltl %si, %n
	jnz %more, @body, @done(%oi)

@done(l %res)
	ret %res

@body
	%sp =l add %src, %si
	%h =w loadsb %sp
	%neg =w csltw %h, 0
	jnz %neg, @ctrl, @lit

@lit
	%hb =w loadub %sp
	%hl =l extsw %hb
	%cnt =l add %hl, 1
	%need =l add %cnt, 1
	%rem =l sub %n, %si
	%srcok =w csgel %rem, %need
	jnz %srcok, @litcap, @errsrc

@errsrc
	ret -2

@litcap
	%room =l sub %cap, %oi
	%fits =w csgel %room, %cnt
	jnz %fits, @litcopy, @litfill

@litcopy
	%from =l add %sp, 1
	%to =l add %out, %oi
	jmp @litcploop(0)

@litcploop(l %i)
	%go =w csltl %i, %cnt
	jnz %go, @litcpset, @litcpend

@litcpset
	%p =l add %from, %i
	%c =w loadub %p
	%q =l add %to, %i
	storeb %c, %q
	%i1 =l add %i, 1
	jmp @litcploop(%i1)

@litcpend
	%si2 =l add %si, %need
	%oi2 =l add %oi, %cnt
	jmp @loop(%si2, %oi2)

@litfill
	%to1 =l add %out, %oi
	jmp @litfloop(0)

@litfloop(l %j)
	%go1 =w csltl %j, %room
	jnz %go1, @litfset, @errcap

@litfset
	%q1 =l add %to1, %j
	storeb %hb, %q1
	%j1 =l add %j, 1
	jmp @litfloop(%j1)

@errcap
	ret -1

@ctrl
	%isnop =w ceqw %h, -128
	jnz %isnop, @nop, @rep

@nop
	%si1 =l add %si, 1
	jmp @loop(%si1, %oi)

@rep
	%rem1 =l sub %n, %si
	%srcok1 =w csgel %rem1, 2
	jnz %srcok1, @repcap, @errsrc

@repcap
	%hs =l extsw %h
	%cntr =l sub 1, %hs
	%room1 =l sub %cap, %oi
	%fits1 =w csgel %room1, %cntr
	jnz %fits1, @repcopy, @repfill

@repcopy
	%bp =l add %sp, 1
	%b =w loadub %bp
	%to2 =l add %out, %oi
	jmp @repcloop(0)

@repcloop(l %k)
	%go2 =w csltl %k, %cntr
	jnz %go2, @repcset, @repcend

@repcset
	%q2 =l add %to2, %k
	storeb %b, %q2
	%k1 =l add %k, 1
	jmp @repcloop(%k1)

@repcend
	%si3 =l add %si, 2
	%oi3 =l add %oi, %cntr
	jmp @loop(%si3, %oi3)

@repfill
	%bp1 =l add %sp, 1
	%b1 =w loadub %bp1
	%to3 =l add %out, %oi
	jmp @repfloop(0)

@repfloop(l %m)
	%go3 =w csltl %m, %room1
	jnz %go3, @repfset, @errcap

@repfset
	%q3 =l add %to3, %m
	storeb %b1, %q3
	%m1 =l add %m, 1
	jmp @repfloop(%m1)
}
