collatz_max:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r15, rdi
    xor r12d, r12d
    mov ebx, 1
.Louter:
    cmp rbx, r15
    jg .Ldone
    mov r13, rbx
    xor r14d, r14d
.Linner:
    cmp r13, 1
    je .Lrecord
    test r13b, 1
    jz .Leven
    lea r13, [r13 + r13*2 + 1]
    inc r14
    jmp .Linner
.Leven:
    shr r13, 1
    inc r14
    jmp .Linner
.Lrecord:
    cmp r14, r12
    jle .Lnext
    mov r12, r14
.Lnext:
    inc rbx
    jmp .Louter
.Ldone:
    mov rax, r12
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret
