export function l $mulhi_u64(l %a, l %b) {
@start
	%al =l and %a, 4294967295
	%ah =l shr %a, 32
	%bl =l and %b, 4294967295
	%bh =l shr %b, 32
	%ll =l mul %al, %bl
	%lh =l mul %al, %bh
	%hl =l mul %ah, %bl
	%hh =l mul %ah, %bh
	%s =l add %lh, %hl
	%c0 =w cultl %s, %lh
	%sh =l shr %s, 32
	%sl =l shl %s, 32
	%low =l add %ll, %sl
	%c1 =w cultl %low, %ll
	%c0e =l extuw %c0
	%c1e =l extuw %c1
	%c0s =l shl %c0e, 32
	%h1 =l add %hh, %sh
	%h2 =l add %h1, %c0s
	%h3 =l add %h2, %c1e
	ret %h3
}
