export function l $memmove_i64(l %dst, l %src, l %n) {
@start
	%back =w csltl %src, %dst
	jnz %back, @backward, @forward

@forward
	%i =l copy 0
	jmp @fcond
@fbody
	%foff =l mul %i, 8
	%fsp =l add %src, %foff
	%fv =l loadl %fsp
	%fdp =l add %dst, %foff
	storel %fv, %fdp
	%fi =l add %i, 1
	%i =l copy %fi
@fcond
	%fc =w csltl %i, %n
	jnz %fc, @fbody, @done

@backward
	%j =l sub %n, 1
	jmp @bcond
@bbody
	%boff =l mul %j, 8
	%bsp =l add %src, %boff
	%bv =l loadl %bsp
	%bdp =l add %dst, %boff
	storel %bv, %bdp
	%bj =l sub %j, 1
	%j =l copy %bj
@bcond
	%bc =w csgel %j, 0
	jnz %bc, @bbody, @done

@done
	ret 0
}
