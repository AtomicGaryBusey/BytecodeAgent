bmax:
    mov x2, #0              // i = 0
    ldr x3, [x0]            // max = a[0]
.Lloop:
    add x2, x2, #1          // i++
    cmp x2, x1              // i >= n?
    b.ge .Ldone
    ldr x4, [x0, x2, lsl #3] // x4 = a[i]
    cmp x3, x4              // max >= a[i] (signed)?
    b.ge .Lloop
    mov x3, x4              // max = a[i]
    b .Lloop
.Ldone:
    mov x0, x3              // return max
    ret
