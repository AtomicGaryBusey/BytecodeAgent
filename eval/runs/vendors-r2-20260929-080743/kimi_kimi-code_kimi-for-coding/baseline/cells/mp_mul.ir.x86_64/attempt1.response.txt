export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%nab =l add %na, %nb
	jmp @zh
@zh
	%k =l phi @entry 0, @zb %kn
	%kc =w cultl %k, %nab
	jnz %kc, @zb, @zd
@zb
	%ko =l mul %k, 8
	%kp =l add %r, %ko
	storel 0, %kp
	%kn =l add %k, 1
	jmp @zh
@zd
	jmp @oh
@oh
	%i =l phi @zd 0, @ifn %in
	%ic =w cultl %i, %na
	jnz %ic, @ob, @sc
@ob
	%ao =l mul %i, 8
	%ap =l add %a, %ao
	%ai =l loadl %ap
	%io =l mul %i, 8
	jmp @ih
@ih
	%j =l phi @ob 0, @ib %jn
	%cy =l phi @ob 0, @ib %cyn
	%jc =w cultl %j, %nb
	jnz %jc, @ib, @ifn
@ib
	%jo =l mul %j, 8
	%ro =l add %io, %jo
	%rp =l add %r, %ro
	%rj =l loadl %rp
	%bp =l add %b, %jo
	%bj =l loadl %bp
	%alo =l and %ai, 0xffffffff
	%ahi =l shr %ai, 32
	%blo =l and %bj, 0xffffffff
	%bhi =l shr %bj, 32
	%p0 =l mul %alo, %blo
	%p1 =l mul %alo, %bhi
	%p2 =l mul %ahi, %blo
	%p3 =l mul %ahi, %bhi
	%ss =l add %p1, %p2
	%c1 =w cultl %ss, %p1
	%shi =l shr %ss, 32
	%slo =l mul %ss, 0x100000000
	%lo =l add %p0, %slo
	%c2 =w cultl %lo, %p0
	%c1l =l extuw %c1
	%big =l mul %c1l, 0x100000000
	%h1 =l add %p3, %shi
	%h2 =l add %h1, %big
	%c2l =l extuw %c2
	%h3 =l add %h2, %c2l
	%t =l add %lo, %rj
	%ca =w cultl %t, %lo
	%cal =l extuw %ca
	%h4 =l add %h3, %cal
	%t2 =l add %t, %cy
	%cb =w cultl %t2, %t
	%cbl =l extuw %cb
	%cyn =l add %h4, %cbl
	storel %t2, %rp
	%jn =l add %j, 1
	jmp @ih
@ifn
	%no =l mul %nb, 8
	%fo =l add %io, %no
	%fp =l add %r, %fo
	storel %cy, %fp
	%in =l add %i, 1
	jmp @oh
@sc
	jmp @sh
@sh
	%m =l phi @sc %nab, @sb %mn
	%mc =w ceql %m, 0
	jnz %mc, @r0, @sb
@sb
	%mn =l sub %m, 1
	%mo =l mul %mn, 8
	%mp =l add %r, %mo
	%mv =l loadl %mp
	%nz =w cnel %mv, 0
	jnz %nz, @fd, @sh
@fd
	%res =l add %mn, 1
	ret %res
@r0
	ret 0
}
