export function l $sdivrem(l %a, l %b, l %rem) {
@entry
	%isz =w ceql %b, 0
	jnz %isz, @zerodiv, @nonzero
@zerodiv
	storel %a, %rem
	ret 0
@nonzero
	%isneg1 =w ceql %b, -1
	jnz %isneg1, @negone, @divide
@negone
	%qneg =l sub 0, %a
	storel 0, %rem
	ret %qneg
@divide
	%q =l div %a, %b
	%r =l rem %a, %b
	storel %r, %rem
	ret %q
}
