lcs_len:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    mov r14, rcx
    mov r15, r8
    lea rdx, [r14 + 1]
    xor ecx, ecx
    xor eax, eax
.Lzero:
    cmp rcx, rdx
    jae .Lzero_done
    mov qword ptr [r15 + rcx*8], rax
    inc rcx
    jmp .Lzero
.Lzero_done:
    xor ebp, ebp
.Louter:
    cmp rbp, r12
    jge .Ldone
    movzx edi, byte ptr [rbx + rbp]
    xor eax, eax
    mov ecx, 1
.Linner:
    cmp rcx, r14
    jg .Linner_done
    mov rsi, qword ptr [r15 + rcx*8]
    cmp dil, byte ptr [r13 + rcx - 1]
    jne .Lno_match
    lea rdx, [rax + 1]
    jmp .Lstore
.Lno_match:
    mov rdx, qword ptr [r15 + rcx*8 - 8]
    cmp rdx, rsi
    cmovl rdx, rsi
.Lstore:
    mov qword ptr [r15 + rcx*8], rdx
    mov rax, rsi
    inc rcx
    jmp .Linner
.Linner_done:
    inc rbp
    jmp .Louter
.Ldone:
    mov rax, qword ptr [r15 + r14*8]
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
