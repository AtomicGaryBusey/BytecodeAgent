export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@entry
	jmp @main

@main
	%i =w phi @entry 0, @repok %inew, @cpydone %j
	%op =w phi @entry 0, @repok %opnew, @cpydone %oplit
	%il =l extuw %i
	%c0 =w csltl %il, %n
	jnz %c0, @body, @done

@done
	%rl =l extuw %op
	ret %rl

@body
	%pi =l add %src, %il
	%b =w loadub %pi
	%k0 =w add %i, 1
	jmp @run

@run
	%k =w phi @body %k0, @runc %k1
	%kl =l extuw %k
	%c1 =w csltl %kl, %n
	jnz %c1, @runcap, @rundone

@runcap
	%kd =w sub %k, %i
	%c2 =w csltw %kd, 128
	jnz %c2, @runeq, @rundone

@runeq
	%pk =l add %src, %kl
	%bk =w loadub %pk
	%c3 =w ceqw %bk, %b
	jnz %c3, @runc, @rundone

@runc
	%k1 =w add %k, 1
	jmp @run

@rundone
	%run =w sub %k, %i
	%isrep =w csgew %run, 3
	jnz %isrep, @rep, @litscan

@rep
	%opl =l extuw %op
	%room =l sub %cap, %opl
	%fits =w csgel %room, 2
	jnz %fits, @repok, @fail

@repok
	%hdr =w sub 257, %run
	%po =l add %out, %opl
	storeb %hdr, %po
	%po1 =l add %po, 1
	storeb %b, %po1
	%opnew =w add %op, 2
	%inew =w add %i, %run
	jmp @main

@litscan
	%j =w phi @rundone %i, @litnext %j1
	%jl =l extuw %j
	%c4 =w csltl %jl, %n
	jnz %c4, @litc2, @litdone

@litc2
	%jd =w sub %j, %i
	%c5 =w csltw %jd, 128
	jnz %c5, @litc3, @litdone

@litc3
	%pj =l add %src, %jl
	%bj =w loadub %pj
	%m0 =w add %j, 1
	jmp @lrun

@lrun
	%m =w phi @litc3 %m0, @lrunc %m1
	%ml =l extuw %m
	%c6 =w csltl %ml, %n
	jnz %c6, @lruncap, @lrdone

@lruncap
	%md =w sub %m, %j
	%c7 =w csltw %md, 3
	jnz %c7, @lruneq, @lrdone

@lruneq
	%pm =l add %src, %ml
	%bm =w loadub %pm
	%c8 =w ceqw %bm, %bj
	jnz %c8, @lrunc, @lrdone

@lrunc
	%m1 =w add %m, 1
	jmp @lrun

@lrdone
	%runj =w sub %m, %j
	%go =w csltw %runj, 3
	jnz %go, @litnext, @litdone

@litnext
	%j1 =w add %j, 1
	jmp @litscan

@litdone
	%len =w sub %j, %i
	%need =w add %len, 1
	%opl2 =l extuw %op
	%room2 =l sub %cap, %opl2
	%needl =l extuw %need
	%fits2 =w csgel %room2, %needl
	jnz %fits2, @litok, @fail

@litok
	%lh =w sub %len, 1
	%po2 =l add %out, %opl2
	storeb %lh, %po2
	jmp @cpy

@cpy
	%t =w phi @litok 0, @cpyinc %t1
	%c9 =w csltw %t, %len
	jnz %c9, @cpyb, @cpydone

@cpyb
	%tl =l extuw %t
	%sl =l extuw %i
	%ps =l add %src, %sl
	%pa =l add %ps, %tl
	%v =w loadub %pa
	%pd =l add %po2, 1
	%pb =l add %pd, %tl
	storeb %v, %pb
	jmp @cpyinc

@cpyinc
	%t1 =w add %t, 1
	jmp @cpy

@cpydone
	%oplit =w add %op, %need
	jmp @main

@fail
	%neg =l copy -1
	ret %neg
}
