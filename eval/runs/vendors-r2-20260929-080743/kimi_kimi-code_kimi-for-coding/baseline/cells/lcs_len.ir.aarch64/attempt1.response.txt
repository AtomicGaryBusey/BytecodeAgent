export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@entry
	%zero =l copy 0
	%j0 =l copy 0
	jmp @zloop
@zloop
	%zdone =w csgtl %j0, %n
	jnz %zdone, @znext, @zbody
@zbody
	%zoff =l mul %j0, 8
	%zaddr =l add %row, %zoff
	storel %zero, %zaddr
	%j0 =l add %j0, 1
	jmp @zloop
@znext
	%i =l copy 0
	jmp @outer
@outer
	%odone =w csgel %i, %m
	jnz %odone, @done, @obody
@obody
	%diag =l copy 0
	%j =l copy 1
	jmp @inner
@inner
	%idone =w csgtl %j, %n
	jnz %idone, @inext, @ibody
@ibody
	%off =l mul %j, 8
	%pj =l add %row, %off
	%up =l loadl %pj
	%px =l add %x, %i
	%xb =w loadub %px
	%jm1 =l sub %j, 1
	%py =l add %y, %jm1
	%yb =w loadub %py
	%eq =w ceqw %xb, %yb
	jnz %eq, @match, @nomatch
@match
	%new =l add %diag, 1
	storel %new, %pj
	jmp @after
@nomatch
	%offm =l mul %jm1, 8
	%pjm1 =l add %row, %offm
	%left =l loadl %pjm1
	%gt =w csgtl %up, %left
	jnz %gt, @useup, @useleft
@useup
	storel %up, %pj
	jmp @after
@useleft
	storel %left, %pj
	jmp @after
@after
	%diag =l copy %up
	%j =l add %j, 1
	jmp @inner
@inext
	%i =l add %i, 1
	jmp @outer
@done
	%noff =l mul %n, 8
	%pn =l add %row, %noff
	%res =l loadl %pn
	ret %res
}
