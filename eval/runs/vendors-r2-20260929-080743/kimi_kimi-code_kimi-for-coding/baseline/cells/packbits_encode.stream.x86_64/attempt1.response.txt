packbits_encode:
    xor r8d, r8d                  # i = 0
    xor r9d, r9d                  # written = 0
.Louter:
    cmp r8, rsi                   # while i < n
    jge .Ldone
    movzx eax, byte ptr [rdi + r8]    # b = src[i]
    lea r10, [r8 + 1]             # k = i + 1
    lea r11, [r8 + 128]           # limit = min(n, i + 128)
    cmp r11, rsi
    cmovg r11, rsi
.Lrun:
    cmp r10, r11
    jge .Lrundone
    cmp byte ptr [rdi + r10], al
    jne .Lrundone
    inc r10
    jmp .Lrun
.Lrundone:
    sub r10, r8                   # r = run length
    cmp r10, 3
    jl .Lliteral
    lea r11, [r9 + 2]             # repeat packet: need 2 bytes
    cmp r11, rcx
    jg .Lfail
    mov r11d, 257
    sub r11, r10                  # header = 257 - r
    mov byte ptr [rdx + r9], r11b
    mov byte ptr [rdx + r9 + 1], al
    add r9, 2
    add r8, r10
    jmp .Louter
.Lliteral:
    mov r10, r8                   # j = i
.Llitscan:
    cmp r10, rsi                  # j < n ?
    jge .Llitdone
    mov rax, r10
    sub rax, r8
    cmp rax, 128                  # j - i < 128 ?
    jge .Llitdone
    lea rax, [r10 + 2]
    cmp rax, rsi                  # fewer than 3 bytes left -> run(j) < 3
    jge .Llitadv
    movzx r11d, byte ptr [rdi + r10]
    cmp r11b, byte ptr [rdi + r10 + 1]
    jne .Llitadv
    cmp r11b, byte ptr [rdi + r10 + 2]
    jne .Llitadv
    jmp .Llitdone                 # run(j) >= 3: stop literal here
.Llitadv:
    inc r10
    jmp .Llitscan
.Llitdone:
    mov r11, r10
    sub r11, r8                   # L = j - i
    lea rax, [r9 + r11 + 1]       # packet size = L + 1
    cmp rax, rcx
    jg .Lfail
    lea rax, [r11 - 1]
    mov byte ptr [rdx + r9], al   # header = L - 1
    push rsi
    push rdi
    push rcx
    lea rsi, [rdi + r8]           # copy src[i..j-1] to out+written+1
    lea rdi, [rdx + r9 + 1]
    mov rcx, r11
    rep movsb
    pop rcx
    pop rdi
    pop rsi
    lea r9, [r9 + r11 + 1]
    mov r8, r10
    jmp .Louter
.Ldone:
    mov rax, r9
    ret
.Lfail:
    mov rax, -1
    ret
