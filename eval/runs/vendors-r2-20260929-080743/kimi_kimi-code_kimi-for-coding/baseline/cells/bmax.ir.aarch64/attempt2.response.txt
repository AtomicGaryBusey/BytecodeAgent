export function l $bmax(l %a, l %n) {
@entry
	%max0 =l loadl %a
	%off =l mul %n, 8
	%end =l add %a, %off
	%p0 =l add %a, 8
	jmp @loop
@loop
	%p = phi @entry %p0, @iter %p1
	%max = phi @entry %max0, @iter %max1
	%c =w csltl %p, %end
	jnz %c, @body, @done
@body
	%v =l loadl %p
	%m =w csgtl %v, %max
	%p1 =l add %p, 8
	jnz %m, @upd, @iter
@upd
	jmp @iter
@iter
	%max1 = phi @body %max, @upd %v
	jmp @loop
@done
	ret %max
}
