export function l $bmax(l %a, l %n) {
@entry
	%m0 =l loadl %a
	%off =l mul %n, 8
	%end =l add %a, %off
	%p0 =l add %a, 8
	jmp @loop
@loop
	%p =l phi @entry %p0, @iter %p1, @take %p2
	%m =l phi @entry %m0, @iter %m, @take %v
	%c =w csltl %p, %end
	jnz %c, @iter, @out
@iter
	%v =l loadl %p
	%g =w csgtl %v, %m
	%p1 =l add %p, 8
	jnz %g, @take, @loop
@take
	%p2 =l add %p, 8
	jmp @loop
@out
	ret %m
}
