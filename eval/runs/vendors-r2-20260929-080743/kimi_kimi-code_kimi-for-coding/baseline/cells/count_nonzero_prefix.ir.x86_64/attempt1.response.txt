export function l $count_nonzero_prefix(l %a, l %n) {
@entry
	%zero =l copy 0
	jmp @loop
@loop
	%i =l phi @entry %zero, @inc %inew
	%cond =w csltl %i, %n
	jnz %cond, @body, @done
@body
	%off =l mul %i, 8
	%addr =l add %a, %off
	%v =l loadl %addr
	%isz =w ceql %v, 0
	jnz %isz, @done, @inc
@inc
	%inew =l add %i, 1
	jmp @loop
@done
	ret %i
}
