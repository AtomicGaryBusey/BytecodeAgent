export function l $base64_encode(l %src, l %n, l %out) {
@start
    %p =l copy %src
    %q =l copy %out
    %groups =l div %n, 3
    %rem =l rem %n, 3
    jmp @grouptest

@grouptest
    %ggo =w csgtl %groups, 0
    jnz %ggo, @group, @aftergroups

@group
    %b0 =l loadub %p
    %pa =l add %p, 1
    %b1 =l loadub %pa
    %pb =l add %p, 2
    %b2 =l loadub %pb
    %x0 =l shl %b0, 16
    %x1 =l shl %b1, 8
    %xx =l or %x0, %x1
    %x =l or %xx, %b2
    %k =l copy 4
    %sh =l copy 18
    jmp @chartest

@chartest
    %kgo =w csgtl %k, 0
    jnz %kgo, @charmap, @groupdone

@charmap
    %xs =l shr %x, %sh
    %idx =l and %xs, 63
    %c65 =l add %idx, 65
    %ge26 =w csltl 25, %idx
    %e26 =l extsw %ge26
    %d1 =l mul %e26, 6
    %ge52 =w csltl 51, %idx
    %e52 =l extsw %ge52
    %d2 =l mul %e52, -75
    %eq62 =w ceql %idx, 62
    %e62 =l extsw %eq62
    %d3 =l mul %e62, -15
    %eq63 =w ceql %idx, 63
    %e63 =l extsw %eq63
    %d4 =l mul %e63, -12
    %c1 =l add %c65, %d1
    %c2 =l add %c1, %d2
    %c3 =l add %c2, %d3
    %c4 =l add %c3, %d4
    storeb %c4, %q
    %q =l add %q, 1
    %sh =l sub %sh, 6
    %k =l sub %k, 1
    jmp @chartest

@groupdone
    %p =l add %p, 3
    %groups =l sub %groups, 1
    jmp @grouptest

@aftergroups
    %rgo =w csgtl %rem, 0
    jnz %rgo, @remstart, @done

@remstart
    %rb0 =l loadub %p
    %rx =l shl %rb0, 16
    %r2 =w ceql %rem, 2
    jnz %r2, @remb1, @remx

@remb1
    %rp =l add %p, 1
    %rb1 =l loadub %rp
    %rxb =l shl %rb1, 8
    %rx =l or %rx, %rxb
    jmp @remx

@remx
    %rk =l add %rem, 1
    %rsh =l copy 18
    jmp @remchartest

@remchartest
    %rkgo =w csgtl %rk, 0
    jnz %rkgo, @remcharmap, @padding

@remcharmap
    %rxs =l shr %rx, %rsh
    %ridx =l and %rxs, 63
    %rc65 =l add %ridx, 65
    %rge26 =w csltl 25, %ridx
    %re26 =l extsw %rge26
    %rd1 =l mul %re26, 6
    %rge52 =w csltl 51, %ridx
    %re52 =l extsw %rge52
    %rd2 =l mul %re52, -75
    %req62 =w ceql %ridx, 62
    %re62 =l extsw %req62
    %rd3 =l mul %re62, -15
    %req63 =w ceql %ridx, 63
    %re63 =l extsw %req63
    %rd4 =l mul %re63, -12
    %rc1 =l add %rc65, %rd1
    %rc2 =l add %rc1, %rd2
    %rc3 =l add %rc2, %rd3
    %rc4 =l add %rc3, %rd4
    storeb %rc4, %q
    %q =l add %q, 1
    %rsh =l sub %rsh, 6
    %rk =l sub %rk, 1
    jmp @remchartest

@padding
    %npad =l sub 3, %rem
    jmp @padtest

@padtest
    %pgo =w csgtl %npad, 0
    jnz %pgo, @padwrite, @done

@padwrite
    storeb 61, %q
    %q =l add %q, 1
    %npad =l sub %npad, 1
    jmp @padtest

@done
    %len =l sub %q, %out
    ret %len
}
