memmove_i64:
    xor eax, eax
    test rdx, rdx
    je .Ldone
    cmp rdi, rsi
    je .Ldone
    jb .Lforward
    mov rcx, rdx
    dec rcx
.Lbackward:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    dec rcx
    jns .Lbackward
    ret
.Lforward:
    xor ecx, ecx
.Lforward_loop:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jb .Lforward_loop
.Ldone:
    ret
