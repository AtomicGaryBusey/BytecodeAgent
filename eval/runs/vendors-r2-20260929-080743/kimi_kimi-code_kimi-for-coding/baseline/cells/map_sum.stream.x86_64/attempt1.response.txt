map_sum:
    push rbx
    push rbp
    push r12
    push r13
    push r14
    mov rbx, rdi
    mov r12, rsi
    mov r13, rdx
    xor r14d, r14d
    test r12, r12
    jle .Ldone
.Lloop:
    mov rdi, qword ptr [rbx]
    call r13
    add r14, rax
    add rbx, 8
    dec r12
    jnz .Lloop
.Ldone:
    mov rax, r14
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret
