export function l $bmax(l %a, l %n) {
@start
    %first =l loadl %a
    jmp @loop
@loop
    %i =l phi @start 1, @next %ni
    %max =l phi @start %first, @next %nm
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %greater =w csgtl %value, %max
    jnz %greater, @replace, @keep
@replace
    jmp @next
@keep
    jmp @next
@next
    %nm =l phi @replace %value, @keep %max
    %ni =l add %i, 1
    jmp @loop
@done
    ret %max
}
