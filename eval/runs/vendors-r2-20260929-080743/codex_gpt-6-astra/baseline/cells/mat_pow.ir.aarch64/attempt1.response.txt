export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
	%nn =l mul %n, %n
	%bytes =l mul %nn, 8
	%total =l mul %nn, 2
	jmp @init
@init
	%p =l phi @start 0, @initnext %pnext
	%row =l div %p, %n
	%col =l rem %p, %n
	%diag =w ceql %row, %col
	%value =l extuw %diag
	%off =l mul %p, 8
	%dest =l add %out, %off
	storel %value, %dest
	%pnext =l add %p, 1
	%moreinit =w csltl %pnext, %nn
	jnz %moreinit, @initnext, @powers
@initnext
	jmp @init
@powers
	%power =l phi @init 0, @copied %nextpower
	%morepowers =w csltl %power, %k
	jnz %morepowers, @productstart, @clearstart
@productstart
	jmp @element
@element
	%e =l phi @productstart 0, @elementnext %enext
	%i =l div %e, %n
	%j =l rem %e, %n
	%rowbase =l mul %i, %n
	jmp @dot
@dot
	%t =l phi @element 0, @dotnext %tnext
	%sum =l phi @element 0, @dotnext %sumnew
	%li =l add %rowbase, %t
	%lo =l mul %li, 8
	%lp =l add %out, %lo
	%lv =l loadl %lp
	%rr =l mul %t, %n
	%ri =l add %rr, %j
	%ro =l mul %ri, 8
	%rp =l add %a, %ro
	%rv =l loadl %rp
	%term =l mul %lv, %rv
	%sumnew =l add %sum, %term
	%tnext =l add %t, 1
	%moredot =w csltl %tnext, %n
	jnz %moredot, @dotnext, @storedot
@dotnext
	jmp @dot
@storedot
	%eo =l mul %e, 8
	%ep =l add %tmp, %eo
	storel %sumnew, %ep
	%enext =l add %e, 1
	%moreelements =w csltl %enext, %nn
	jnz %moreelements, @elementnext, @copystart
@elementnext
	jmp @element
@copystart
	jmp @copy
@copy
	%c =l phi @copystart 0, @copynext %cnext
	%co =l mul %c, 8
	%cp =l add %tmp, %co
	%cv =l loadl %cp
	%op =l add %out, %co
	storel %cv, %op
	%cnext =l add %c, 1
	%morecopy =w csltl %cnext, %nn
	jnz %morecopy, @copynext, @copied
@copynext
	jmp @copy
@copied
	%nextpower =l add %power, 1
	jmp @powers
@clearstart
	jmp @clear
@clear
	%z =l phi @clearstart 0, @clearnext %znext
	%zo =l mul %z, 8
	%zp =l add %tmp, %zo
	storel 0, %zp
	%znext =l add %z, 1
	%moreclear =w csltl %znext, %total
	jnz %moreclear, @clearnext, @done
@clearnext
	jmp @clear
@done
	ret 0
}
