export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @raise %mid, @lower %lo
	%hi =l phi @start 4294967296, @raise %hi, @lower %mid
	%gap =l sub %hi, %lo
	%more =w cugtl %gap, 1
	jnz %more, @test, @done
@test
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%square =l mul %mid, %mid
	%fits =w culel %square, %n
	jnz %fits, @raise, @lower
@raise
	jmp @loop
@lower
	jmp @loop
@done
	ret %lo
}
