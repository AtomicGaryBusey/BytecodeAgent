export function l $mulhi_u64(l %a, l %b) {
@start
    %a0 =l and %a, 4294967295
    %a1 =l shr %a, 32
    %b0 =l and %b, 4294967295
    %b1 =l shr %b, 32
    %p00 =l mul %a0, %b0
    %p01 =l mul %a0, %b1
    %p10 =l mul %a1, %b0
    %p11 =l mul %a1, %b1
    %c00 =l shr %p00, 32
    %lo01 =l and %p01, 4294967295
    %lo10 =l and %p10, 4294967295
    %m0 =l add %c00, %lo01
    %m1 =l add %m0, %lo10
    %carry =l shr %m1, 32
    %hi01 =l shr %p01, 32
    %hi10 =l shr %p10, 32
    %h0 =l add %p11, %hi01
    %h1 =l add %h0, %hi10
    %result =l add %h1, %carry
    ret %result
}
