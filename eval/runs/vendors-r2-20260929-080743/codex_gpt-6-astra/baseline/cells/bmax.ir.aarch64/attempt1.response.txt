export function l $bmax(l %a, l %n) {
@start
    %first = loadl %a
    jmp @loop
@loop
    %i = phi @start 1, @body %next
    %max = phi @start %first, @body %updated
    %more = csltl %i, %n
    jnz %more, @load, @done
@load
    %offset = mul %i, 8
    %ptr = add %a, %offset
    %value = loadl %ptr
    %greater = csgtl %value, %max
    jnz %greater, @replace, @keep
@replace
    jmp @body
@keep
    jmp @body
@body
    %updated = phi @replace %value, @keep %max
    %next = add %i, 1
    jmp @loop
@done
    ret %max
}
