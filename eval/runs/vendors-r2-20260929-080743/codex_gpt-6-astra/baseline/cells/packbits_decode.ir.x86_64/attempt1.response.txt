export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet
@packet
	%i =l phi @start 0, @noop %next, @literal_done %lend, @repeat_done %rend
	%o =l phi @start 0, @noop %o, @literal_done %lo, @repeat_done %ro
	%finished =w ceql %i, %n
	jnz %finished, @success, @header
@header
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%h =l extsw %hw
	%next =l add %i, 1
	%isnoop =w ceql %h, -128
	jnz %isnoop, @noop, @classify
@noop
	jmp @packet
@classify
	%negative =w csltl %h, 0
	jnz %negative, @repeat_check, @literal_check
@literal_check
	%lc =l add %h, 1
	%lend =l add %next, %lc
	%ltruncated =w csgtl %lend, %n
	jnz %ltruncated, @truncated, @literal_loop
@literal_loop
	%li =l phi @literal_check %next, @literal_write %linext
	%lo =l phi @literal_check %o, @literal_write %lonext
	%ldone =w ceql %li, %lend
	jnz %ldone, @literal_done, @literal_room
@literal_room
	%lfull =w ceql %lo, %cap
	jnz %lfull, @overflow, @literal_write
@literal_write
	%lp =l add %src, %li
	%lb =w loadub %lp
	%lop =l add %out, %lo
	storeb %lb, %lop
	%linext =l add %li, 1
	%lonext =l add %lo, 1
	jmp @literal_loop
@literal_done
	jmp @packet
@repeat_check
	%rtruncated =w ceql %next, %n
	jnz %rtruncated, @truncated, @repeat_init
@repeat_init
	%rp =l add %src, %next
	%rb =w loadub %rp
	%rc =l sub 1, %h
	%rend =l add %next, 1
	jmp @repeat_loop
@repeat_loop
	%left =l phi @repeat_init %rc, @repeat_write %leftnext
	%ro =l phi @repeat_init %o, @repeat_write %ronext
	%rdone =w ceql %left, 0
	jnz %rdone, @repeat_done, @repeat_room
@repeat_room
	%rfull =w ceql %ro, %cap
	jnz %rfull, @overflow, @repeat_write
@repeat_write
	%rop =l add %out, %ro
	storeb %rb, %rop
	%ronext =l add %ro, 1
	%leftnext =l sub %left, 1
	jmp @repeat_loop
@repeat_done
	jmp @packet
@success
	ret %o
@truncated
	ret -2
@overflow
	ret -1
}
