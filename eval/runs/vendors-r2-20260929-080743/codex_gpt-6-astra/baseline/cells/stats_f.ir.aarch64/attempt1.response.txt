export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    jmp @loop
@loop
    %i =l phi @start 1, @next %inc
    %min =l phi @start %y0, @next %newmin
    %max =l phi @start %y0, @next %newmax
    %sum =l phi @start %y0, @next %newsum
    %count =l phi @start %odd0, @next %newcount
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %x =l loadl %ptr
    %y =l call %f(l %x)
    %less =w csltl %y, %min
    jnz %less, @minyes, @minno
@minyes
    jmp @minmerge
@minno
    jmp @minmerge
@minmerge
    %newmin =l phi @minyes %y, @minno %min
    %greater =w csgtl %y, %max
    jnz %greater, @maxyes, @maxno
@maxyes
    jmp @next
@maxno
    jmp @next
@next
    %newmax =l phi @maxyes %y, @maxno %max
    %newsum =l add %sum, %y
    %odd =l and %y, 1
    %newcount =l add %count, %odd
    %inc =l add %i, 1
    jmp @loop
@done
    storel %min, %out
    %out1 =l add %out, 8
    storel %max, %out1
    %out2 =l add %out, 16
    storel %sum, %out2
    ret %count
}
