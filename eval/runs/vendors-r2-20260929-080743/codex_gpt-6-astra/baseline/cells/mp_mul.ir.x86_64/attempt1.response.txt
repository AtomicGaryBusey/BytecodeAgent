export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@start
    %n =l add %na, %nb
    jmp @zero
@zero
    %z =l phi @start 0, @zero_body %znext
    %zmore =w cultl %z, %n
    jnz %zmore, @zero_body, @check
@zero_body
    %zoff =l shl %z, 3
    %zp =l add %r, %zoff
    storel 0, %zp
    %znext =l add %z, 1
    jmp @zero
@check
    %az =w ceql %na, 0
    %bz =w ceql %nb, 0
    %empty =w or %az, %bz
    jnz %empty, @trim_start, @outer
@outer
    %i =l phi @check 0, @row_end %inext
    %ioff =l shl %i, 3
    %ap =l add %a, %ioff
    %av =l loadl %ap
    %al =l and %av, 4294967295
    %ah =l shr %av, 32
    jmp @inner
@inner
    %j =l phi @outer 0, @body %jnext
    %carry =l phi @outer 0, @body %carrynext
    %more =w cultl %j, %nb
    jnz %more, @body, @row_end
@body
    %boff =l shl %j, 3
    %bp =l add %b, %boff
    %bv =l loadl %bp
    %bl =l and %bv, 4294967295
    %bh =l shr %bv, 32
    %p00 =l mul %al, %bl
    %p10 =l mul %ah, %bl
    %p01 =l mul %al, %bh
    %p11 =l mul %ah, %bh
    %p00hi =l shr %p00, 32
    %u =l add %p10, %p00hi
    %ulo =l and %u, 4294967295
    %uhi =l shr %u, 32
    %v =l add %p01, %ulo
    %vhi =l shr %v, 32
    %h0 =l add %p11, %uhi
    %hi =l add %h0, %vhi
    %vlo =l shl %v, 32
    %p00lo =l and %p00, 4294967295
    %lo =l or %vlo, %p00lo
    %index =l add %i, %j
    %roff =l shl %index, 3
    %rp =l add %r, %roff
    %old =l loadl %rp
    %s1 =l add %lo, %old
    %c1 =w cultl %s1, %lo
    %c1l =l extuw %c1
    %s2 =l add %s1, %carry
    %c2 =w cultl %s2, %s1
    %c2l =l extuw %c2
    %h1 =l add %hi, %c1l
    %carrynext =l add %h1, %c2l
    storel %s2, %rp
    %jnext =l add %j, 1
    jmp @inner
@row_end
    %last =l add %i, %nb
    %lastoff =l shl %last, 3
    %lastp =l add %r, %lastoff
    storel %carry, %lastp
    %inext =l add %i, 1
    %imore =w cultl %inext, %na
    jnz %imore, @outer, @trim_start
@trim_start
    jmp @trim
@trim
    %k =l phi @trim_start %n, @trim_load %prev
    %done =w ceql %k, 0
    jnz %done, @return, @trim_load
@trim_load
    %prev =l sub %k, 1
    %toff =l shl %prev, 3
    %tp =l add %r, %toff
    %tv =l loadl %tp
    %iszero =w ceql %tv, 0
    jnz %iszero, @trim, @return
@return
    ret %k
}
