bubble_sort:
    cmp x1, #1
    b.le .Ldone
    sub x1, x1, #1
.Louter:
    mov x2, #0
    mov x3, #0
.Linner:
    add x4, x0, x2, lsl #3
    ldr x5, [x4]
    ldr x6, [x4, #8]
    cmp x5, x6
    b.le .Lnext
    str x6, [x4]
    str x5, [x4, #8]
    mov x3, #1
.Lnext:
    add x2, x2, #1
    cmp x2, x1
    b.lt .Linner
    cbz x3, .Ldone
    subs x1, x1, #1
    b.ne .Louter
.Ldone:
    mov x0, #0
    ret
