map_sum:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    xor ebx, ebx
    test r13, r13
    jz .Ldone
.Lloop:
    mov rdi, qword ptr [r12]
    cld
    call r14
    add rbx, rax
    add r12, 8
    dec r13
    jnz .Lloop
.Ldone:
    mov rax, rbx
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    cld
    ret
