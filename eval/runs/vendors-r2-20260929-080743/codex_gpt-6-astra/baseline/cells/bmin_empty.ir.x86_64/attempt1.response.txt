export function l $bmin_empty(l %a, l %n) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @next %inc
    %min =l phi @start 9223372036854775807, @next %newmin
    %more =w csltl %i, %n
    jnz %more, @read, @done
@read
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %value =l loadl %ptr
    %less =w csltl %value, %min
    jnz %less, @smaller, @keep
@smaller
    jmp @next
@keep
    jmp @next
@next
    %newmin =l phi @smaller %value, @keep %min
    %inc =l add %i, 1
    jmp @loop
@done
    ret %min
}
