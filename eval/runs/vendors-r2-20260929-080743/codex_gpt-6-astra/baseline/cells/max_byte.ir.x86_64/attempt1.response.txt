export function l $max_byte(l %p, l %n) {
@start
    %empty =w ceql %n, 0
    jnz %empty, @zero, @loop
@loop
    %i =l phi @start 0, @next %inc
    %max =l phi @start 0, @next %best
    %addr =l add %p, %i
    %b =l loadub %addr
    %greater =w cugtl %b, %max
    jnz %greater, @update, @keep
@update
    jmp @next
@keep
    jmp @next
@next
    %best =l phi @update %b, @keep %max
    %inc =l add %i, 1
    %more =w cultl %inc, %n
    jnz %more, @loop, @done
@done
    ret %best
@zero
    ret 0
}
