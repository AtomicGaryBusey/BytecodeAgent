export function l $mp_mul(l %a, l %na, l %b, l %nb, l %r) {
@entry
	%n =l add %na, %nb
	jmp @zero_test
@zero_test
	%z =l phi @entry 0, @zero_body %zn
	%zmore =w cultl %z, %n
	jnz %zmore, @zero_body, @start
@zero_body
	%zo =l shl %z, 3
	%zp =l add %r, %zo
	storel 0, %zp
	%zn =l add %z, 1
	jmp @zero_test
@start
	%az =w ceql %na, 0
	%bz =w ceql %nb, 0
	%empty =w or %az, %bz
	jnz %empty, @scan_start, @outer
@outer
	%i =l phi @start 0, @outer_end %in
	%ao =l shl %i, 3
	%ap =l add %a, %ao
	%av =l loadl %ap
	%al =l and %av, 4294967295
	%ah =l shr %av, 32
	jmp @inner
@inner
	%j =l phi @outer 0, @inner %jn
	%carry =l phi @outer 0, @inner %nextcarry
	%bo =l shl %j, 3
	%bp =l add %b, %bo
	%bv =l loadl %bp
	%bl =l and %bv, 4294967295
	%bh =l shr %bv, 32
	%p00 =l mul %al, %bl
	%p10 =l mul %ah, %bl
	%p01 =l mul %al, %bh
	%p11 =l mul %ah, %bh
	%toph =l shr %p00, 32
	%u =l add %p10, %toph
	%ul =l and %u, 4294967295
	%uh =l shr %u, 32
	%v =l add %p01, %ul
	%vh =l shr %v, 32
	%h0 =l add %p11, %uh
	%hi =l add %h0, %vh
	%vl =l shl %v, 32
	%tl =l and %p00, 4294967295
	%lo =l or %vl, %tl
	%idx =l add %i, %j
	%ro =l shl %idx, 3
	%rp =l add %r, %ro
	%old =l loadl %rp
	%s1 =l add %lo, %old
	%c1 =w cultl %s1, %lo
	%c1l =l extuw %c1
	%s2 =l add %s1, %carry
	%c2 =w cultl %s2, %s1
	%c2l =l extuw %c2
	%hc =l add %hi, %c1l
	%nextcarry =l add %hc, %c2l
	storel %s2, %rp
	%jn =l add %j, 1
	%jmore =w cultl %jn, %nb
	jnz %jmore, @inner, @outer_end
@outer_end
	%ci =l add %i, %nb
	%co =l shl %ci, 3
	%cp =l add %r, %co
	storel %nextcarry, %cp
	%in =l add %i, 1
	%imore =w cultl %in, %na
	jnz %imore, @outer, @scan_start
@scan_start
	jmp @scan
@scan
	%k =l phi @scan_start %n, @scan_load %km
	%done =w ceql %k, 0
	jnz %done, @return, @scan_load
@scan_load
	%km =l sub %k, 1
	%ko =l shl %km, 3
	%kp =l add %r, %ko
	%kv =l loadl %kp
	%nonzero =w cnel %kv, 0
	jnz %nonzero, @return, @scan
@return
	ret %k
}
