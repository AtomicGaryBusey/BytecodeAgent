export function l $count_desc(l %a, l %n) {
@start
    %small =w csltl %n, 2
    jnz %small, @zero, @init
@zero
    ret 0
@init
    %first =l loadl %a
    jmp @loop
@loop
    %i =l phi @init 1, @loop %nexti
    %count =l phi @init 0, @loop %nextcount
    %prev =l phi @init %first, @loop %curr
    %offset =l mul %i, 8
    %ptr =l add %a, %offset
    %curr =l loadl %ptr
    %desc =w csgtl %prev, %curr
    %inc =l extuw %desc
    %nextcount =l add %count, %inc
    %nexti =l add %i, 1
    %more =w csltl %nexti, %n
    jnz %more, @loop, @done
@done
    ret %nextcount
}
