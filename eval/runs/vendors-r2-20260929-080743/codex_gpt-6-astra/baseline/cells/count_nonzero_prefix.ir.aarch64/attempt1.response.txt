export function l $count_nonzero_prefix(l %a, l %n) {
@start
	jmp @check
@check
	%i =l phi @start 0, @next %inc
	%inbounds =w csltl %i, %n
	jnz %inbounds, @read, @done
@read
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%zero =w ceql %value, 0
	jnz %zero, @done, @next
@next
	%inc =l add %i, 1
	jmp @check
@done
	ret %i
}
