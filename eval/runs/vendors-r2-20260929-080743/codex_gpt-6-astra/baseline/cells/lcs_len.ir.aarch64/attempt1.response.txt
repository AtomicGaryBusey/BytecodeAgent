export function l $lcs_len(l %x, l %m, l %y, l %n, l %row) {
@start
	jmp @init
@init
	%k =l phi @start 0, @init %knext
	%ko =l mul %k, 8
	%kp =l add %row, %ko
	storel 0, %kp
	%knext =l add %k, 1
	%moreinit =w cslel %knext, %n
	jnz %moreinit, @init, @outercheck
@outercheck
	%i =l phi @init 0, @outerdone %inext
	%morex =w csltl %i, %m
	jnz %morex, @outer, @done
@outer
	%xp =l add %x, %i
	%xb =w loadub %xp
	jmp @innercheck
@innercheck
	%j =l phi @outer 1, @update %jnext
	%diag =l phi @outer 0, @update %up
	%left =l phi @outer 0, @update %value
	%morey =w cslel %j, %n
	jnz %morey, @inner, @outerdone
@inner
	%offset =l mul %j, 8
	%rp =l add %row, %offset
	%up =l loadl %rp
	%yi =l sub %j, 1
	%yp =l add %y, %yi
	%yb =w loadub %yp
	%equal =w ceqw %xb, %yb
	jnz %equal, @match, @different
@match
	%matched =l add %diag, 1
	jmp @update
@different
	%upgreater =w csgtl %up, %left
	jnz %upgreater, @takeup, @takeleft
@takeup
	jmp @update
@takeleft
	jmp @update
@update
	%value =l phi @match %matched, @takeup %up, @takeleft %left
	storel %value, %rp
	%jnext =l add %j, 1
	jmp @innercheck
@outerdone
	%inext =l add %i, 1
	jmp @outercheck
@done
	%lastoffset =l mul %n, 8
	%lastp =l add %row, %lastoffset
	%result =l loadl %lastp
	ret %result
}
