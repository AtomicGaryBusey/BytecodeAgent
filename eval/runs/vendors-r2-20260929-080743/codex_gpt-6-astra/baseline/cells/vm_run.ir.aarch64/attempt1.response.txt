export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	jmp @loop
@loop
	%pc =l phi @start 0, @next %npc
	%d =l phi @start 0, @next %nd
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch
@fetch
	%coff =l shl %pc, 3
	%cp =l add %code, %coff
	%op =l loadl %cp
	%pc1 =l add %pc, 1
	%pc2 =l add %pc, 2
	%doff =l shl %d, 3
	%dp =l add %stack, %doff
	%halt =w ceql %op, 0
	jnz %halt, @finish, @testpush
@testpush
	%ispush =w ceql %op, 1
	jnz %ispush, @push, @testjump
@push
	%operandp =l add %cp, 8
	%imm =l loadl %operandp
	storel %imm, %dp
	%dpush =l add %d, 1
	jmp @next
@testjump
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jump, @top
@jump
	%jumpp =l add %cp, 8
	%target =l loadl %jumpp
	jmp @next
@top
	%tp =l sub %dp, 8
	%b =l loadl %tp
	%dm1 =l sub %d, 1
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @testdup
@jz
	%jzp =l add %cp, 8
	%ztarget =l loadl %jzp
	%zero =w ceql %b, 0
	jnz %zero, @taken, @untaken
@taken
	jmp @next
@untaken
	jmp @next
@testdup
	%isdup =w ceql %op, 5
	jnz %isdup, @dup, @testdrop
@dup
	storel %b, %dp
	%ddup =l add %d, 1
	jmp @next
@testdrop
	%isdrop =w ceql %op, 7
	jnz %isdrop, @drop, @second
@drop
	jmp @next
@second
	%ap =l sub %dp, 16
	%a =l loadl %ap
	%isswap =w ceql %op, 6
	jnz %isswap, @swap, @testover
@swap
	storel %b, %ap
	storel %a, %tp
	jmp @next
@testover
	%isover =w ceql %op, 8
	jnz %isover, @over, @testadd
@over
	storel %a, %dp
	%dover =l add %d, 1
	jmp @next
@testadd
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @testsub
@add
	%sum =l add %a, %b
	jmp @binary
@testsub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @testmul
@sub
	%diff =l sub %a, %b
	jmp @binary
@testmul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @lt
@mul
	%prod =l mul %a, %b
	jmp @binary
@lt
	%less =w csltl %a, %b
	%less64 =l extuw %less
	jmp @binary
@binary
	%result =l phi @add %sum, @sub %diff, @mul %prod, @lt %less64
	storel %result, %ap
	jmp @next
@next
	%npc =l phi @push %pc2, @jump %target, @taken %ztarget, @untaken %pc2, @dup %pc1, @drop %pc1, @swap %pc1, @over %pc1, @binary %pc1
	%nd =l phi @push %dpush, @jump %d, @taken %dm1, @untaken %dm1, @dup %ddup, @drop %dm1, @swap %d, @over %dover, @binary %dm1
	jmp @loop
@finish
	jmp @clear
@clear
	%i =l phi @finish %d, @clearone %i1
	%done =w ceql %i, %cap
	jnz %done, @returntest, @clearone
@clearone
	%ioff =l shl %i, 3
	%ip =l add %stack, %ioff
	storel 0, %ip
	%i1 =l add %i, 1
	jmp @clear
@returntest
	%empty =w ceql %d, 0
	jnz %empty, @returnzero, @returntop
@returnzero
	ret 0
@returntop
	%last =l sub %d, 1
	%lastoff =l shl %last, 3
	%lastp =l add %stack, %lastoff
	%value =l loadl %lastp
	ret %value
}
