isqrt:
    xor ecx, ecx
    mov r8, 4294967296
.Lloop:
    lea rsi, [rcx + 1]
    cmp rsi, r8
    jae .Ldone
    lea rsi, [rcx + r8]
    shr rsi, 1
    mov rax, rsi
    imul rax, rsi
    cmp rax, rdi
    ja .Lhigh
    mov rcx, rsi
    jmp .Lloop
.Lhigh:
    mov r8, rsi
    jmp .Lloop
.Ldone:
    mov rax, rcx
    ret
