push r13
push r14
push r15

mov r13, rsi
mov r14, rdx
mov r15, rcx
mov rbx, r8
mov rbp, r13
imul rbp, r13
xor eax, eax
.Linit_zero:
mov qword ptr [r15 + rax*8], 0
inc rax
cmp rax, rbp
jb .Linit_zero
mov rcx, r15
lea rdx, [r13*8 + 8]
xor eax, eax
.Linit_diag:
mov qword ptr [rcx], 1
add rcx, rdx
inc rax
cmp rax, r13
jb .Linit_diag
test r14, r14
jz .Lclear
.Lpower:
lea rdx, [r13*8]
mov rdi, r15
mov rcx, rbx
xor r8d, r8d
.Lrow:
xor r9d, r9d
.Lcolumn:

xor r10d, r10d
xor r11d, r11d
.Ldot:

imul rax, qword ptr [rsi]
add r11, rax
add rsi, rdx
inc r10
cmp r10, r13
jb .Ldot
mov qword ptr [rcx], r11
add rcx, 8
inc r9
cmp r9, r13
jb .Lcolumn
add rdi, rdx
inc r8
cmp r8, r13
jb .Lrow
xor eax, eax
.Lcopy:
mov rcx, qword ptr [rbx + rax*8]
mov qword ptr [r15 + rax*8], rcx
inc rax
cmp rax, rbp
jb .Lcopy
dec r14
jnz .Lpower
.Lclear:
lea rcx, [rbp + rbp]
xor eax, eax
.Lclear_loop:
mov qword ptr [rbx + rax*8], 0
inc rax
cmp rax, rcx
jb .Lclear_loop

pop r14
pop r13
pop r12
pop rbp
pop rbx
