export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @continue %next
	%best =l phi @start 0, @continue %updated
	jmp @inner
@inner
	%x =l phi @outer %k, @step %y
	%count =l phi @outer 0, @step %inc
	%finished =w ceql %x, 1
	jnz %finished, @done, @parity
@parity
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @oddstep, @evenstep
@oddstep
	%triple =l mul %x, 3
	%oddvalue =l add %triple, 1
	jmp @step
@evenstep
	%evenvalue =l shr %x, 1
	jmp @step
@step
	%y =l phi @oddstep %oddvalue, @evenstep %evenvalue
	%inc =l add %count, 1
	jmp @inner
@done
	%greater =w csgtl %count, %best
	jnz %greater, @replace, @keep
@replace
	jmp @advance
@keep
	jmp @advance
@advance
	%updated =l phi @replace %count, @keep %best
	%next =l add %k, 1
	%more =w cslel %next, %n
	jnz %more, @continue, @exit
@continue
	jmp @outer
@exit
	ret %updated
}
