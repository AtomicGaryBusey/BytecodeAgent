memmove_i64:

    cmp x0, x1
    b.eq .Ldone
    b.lo .Lforward
    add x9, x0, x2, lsl #3
    add x10, x1, x2, lsl #3
.Lbackward:
    ldr x11, [x10, #-8]!
    str x11, [x9, #-8]!
    subs x2, x2, #1
    b.ne .Lbackward
    b .Ldone
.Lforward:
    ldr x11, [x1], #8
    str x11, [x0], #8
    subs x2, x2, #1
    b.ne .Lforward
.Ldone:
    mov x0, #0
    ret
