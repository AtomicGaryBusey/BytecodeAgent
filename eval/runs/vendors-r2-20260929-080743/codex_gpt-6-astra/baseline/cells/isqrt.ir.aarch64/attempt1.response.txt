export function l $isqrt(l %n) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lower %mid, @upper %lo
	%hi =l phi @start 4294967296, @lower %hi, @upper %mid
	%gap =l sub %hi, %lo
	%done =w ceql %gap, 1
	jnz %done, @end, @test
@test
	%half =l shr %gap, 1
	%mid =l add %lo, %half
	%quot =l udiv %n, %mid
	%fits =w culel %mid, %quot
	jnz %fits, @lower, @upper
@lower
	jmp @loop
@upper
	jmp @loop
@end
	ret %lo
}
