export function l $map_into(l %dst, l %src, l %n, l %f) {
@start
    jmp @loop
@loop
    %i =l phi @start 0, @body %next
    %count =l phi @start 0, @body %updated
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %sptr =l add %src, %offset
    %dptr =l add %dst, %offset
    %x =l loadl %sptr
    %r =l call %f(l %x)
    storel %r, %dptr
    %less =w csltl %r, %x
    %increment =l extuw %less
    %updated =l add %count, %increment
    %next =l add %i, 1
    jmp @loop
@done
    ret %count
}
