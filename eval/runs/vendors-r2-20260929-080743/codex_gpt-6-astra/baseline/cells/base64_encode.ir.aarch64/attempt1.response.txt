export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%i =l phi @start 0, @groupdone %inext
	%o =l phi @start 0, @groupdone %onext
	%more =w csltl %i, %n
	jnz %more, @first, @done
@first
	%remaining =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%v0 =l shl %b0, 16
	%hassecond =w cugtl %remaining, 1
	jnz %hassecond, @second, @one
@one
	jmp @emit
@second
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%t1 =l shl %b1, 8
	%v1 =l or %v0, %t1
	%hasthird =w cugtl %remaining, 2
	jnz %hasthird, @third, @two
@two
	jmp @emit
@third
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	%v2 =l or %v1, %b2
	jmp @emit
@emit
	%bits =l phi @one %v0, @two %v1, @third %v2
	%valid =l phi @one 2, @two 3, @third 4
	jmp @character
@character
	%j =l phi @emit 0, @written %jnext
	%ispad =w cugel %j, %valid
	jnz %ispad, @padding, @index
@padding
	jmp @write
@index
	%shiftpart =l mul %j, 6
	%shift =l sub 18, %shiftpart
	%shifted =l shr %bits, %shift
	%idx =l and %shifted, 63
	%isupper =w cultl %idx, 26
	jnz %isupper, @upper, @testlower
@upper
	%uc =l add %idx, 65
	jmp @write
@testlower
	%islower =w cultl %idx, 52
	jnz %islower, @lower, @testdigit
@lower
	%lc =l add %idx, 71
	jmp @write
@testdigit
	%isdigit =w cultl %idx, 62
	jnz %isdigit, @digit, @special
@digit
	%dc =l sub %idx, 4
	jmp @write
@special
	%delta =l sub %idx, 62
	%scaled =l mul %delta, 4
	%sc =l add %scaled, 43
	jmp @write
@write
	%c =l phi @padding 61, @upper %uc, @lower %lc, @digit %dc, @special %sc
	%offset =l add %o, %j
	%dest =l add %out, %offset
	storeb %c, %dest
	jmp @written
@written
	%jnext =l add %j, 1
	%again =w cultl %jnext, 4
	jnz %again, @character, @groupdone
@groupdone
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @group
@done
	ret %o
}
