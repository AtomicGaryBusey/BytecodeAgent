export function l $binary_search(l %a, l %n, l %key) {
@entry
	jmp @loop
@loop
	%lo =l phi @entry 0, @left %lo, @right %next
	%hi =l phi @entry %n, @left %mid, @right %hi
	%active =w csltl %lo, %hi
	jnz %active, @probe, @missing
@probe
	%span =l sub %hi, %lo
	%half =l shr %span, 1
	%mid =l add %lo, %half
	%offset =l mul %mid, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @right, @left
@left
	jmp @loop
@right
	%next =l add %mid, 1
	jmp @loop
@found
	ret %mid
@missing
	ret -1
}
