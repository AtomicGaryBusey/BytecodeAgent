map_into:
    cld
    xor eax, eax
    test rdx, rdx
    jz .Lreturn
    push rbx
    push rbp
    push r12
    push r13
    push r14
    push r15
    sub rsp, 8
    mov r12, rdi
    mov r13, rsi
    mov r14, rdx
    mov r15, rcx
    xor ebx, ebx
.Lloop:
    mov rbp, qword ptr [r13]
    mov rdi, rbp
    call r15
    mov qword ptr [r12], rax
    cmp rax, rbp
    setl al
    movzx eax, al
    add rbx, rax
    add r12, 8
    add r13, 8
    dec r14
    jnz .Lloop
    mov rax, rbx
    add rsp, 8
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
.Lreturn:
    cld
    ret
