export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @next
@next
	%i =l phi @start 0, @repeat_done %ri, @literal_done %j
	%used =l phi @start 0, @repeat_done %ru, @literal_done %lu
	%end =w ceql %i, %n
	jnz %end, @success, @run_start
@run_start
	%ip =l add %src, %i
	%b =w loadub %ip
	jmp @run_check
@run_check
	%r =l phi @run_start 1, @run_more %nr
	%rp =l add %i, %r
	%run_end =w ceql %rp, %n
	%run_full =w ceql %r, 128
	%run_stop =w or %run_end, %run_full
	jnz %run_stop, @run_done, @run_load
@run_load
	%sp =l add %src, %rp
	%rb =w loadub %sp
	%same =w ceqw %rb, %b
	jnz %same, @run_more, @run_done
@run_more
	%nr =l add %r, 1
	jmp @run_check
@run_done
	%repeat =w csgel %r, 3
	jnz %repeat, @repeat_check, @literal_start
@repeat_check
	%ru =l add %used, 2
	%rfits =w cslel %ru, %cap
	jnz %rfits, @repeat_write, @failure
@repeat_write
	%rh =l sub 257, %r
	%op =l add %out, %used
	storeb %rh, %op
	%opb =l add %op, 1
	storeb %b, %opb
	jmp @repeat_done
@repeat_done
	%ri =l add %i, %r
	jmp @next
@literal_start
	%firstj =l add %i, 1
	jmp @literal_scan
@literal_scan
	%j =l phi @literal_start %firstj, @literal_advance %nj
	%len =l sub %j, %i
	%jend =w ceql %j, %n
	%lfull =w ceql %len, 128
	%lstop =w or %jend, %lfull
	jnz %lstop, @literal_check, @triple_bound
@triple_bound
	%j2 =l add %j, 2
	%has3 =w csltl %j2, %n
	jnz %has3, @triple_load, @literal_advance
@triple_load
	%jp =l add %src, %j
	%jp1 =l add %jp, 1
	%jp2 =l add %jp, 2
	%b0 =w loadub %jp
	%b1 =w loadub %jp1
	%b2 =w loadub %jp2
	%eq1 =w ceqw %b0, %b1
	%eq2 =w ceqw %b0, %b2
	%triple =w and %eq1, %eq2
	jnz %triple, @literal_check, @literal_advance
@literal_advance
	%nj =l add %j, 1
	jmp @literal_scan
@literal_check
	%payload_end =l add %used, %len
	%lu =l add %payload_end, 1
	%lfits =w cslel %lu, %cap
	jnz %lfits, @literal_header, @failure
@literal_header
	%lh =l sub %len, 1
	%lp =l add %out, %used
	storeb %lh, %lp
	%dest =l add %lp, 1
	jmp @copy
@copy
	%k =l phi @literal_header 0, @copy %nk
	%from =l add %ip, %k
	%to =l add %dest, %k
	%v =w loadub %from
	storeb %v, %to
	%nk =l add %k, 1
	%copied =w ceql %nk, %len
	jnz %copied, @literal_done, @copy
@literal_done
	jmp @next
@success
	ret %used
@failure
	ret -1
}
