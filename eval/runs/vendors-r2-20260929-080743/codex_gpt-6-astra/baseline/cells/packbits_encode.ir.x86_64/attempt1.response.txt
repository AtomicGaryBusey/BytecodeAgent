export function l $packbits_encode(l %src, l %n, l %out, l %cap) {
@start
	jmp @outer
@outer
	%i =l phi @start 0, @repeat_done %ri, @literal_done %j
	%used =l phi @start 0, @repeat_done %ru, @literal_done %lu
	%finished =w ceql %i, %n
	jnz %finished, @done, @run_start
@run_start
	%sp =l add %src, %i
	%b =w loadub %sp
	jmp @run_check
@run_check
	%r =l phi @run_start 1, @run_extend %next_r
	%at_limit =w ceql %r, 128
	jnz %at_limit, @run_ready, @run_bounds
@run_bounds
	%pos =l add %i, %r
	%at_end =w cugel %pos, %n
	jnz %at_end, @run_ready, @run_load
@run_load
	%rp =l add %src, %pos
	%rb =w loadub %rp
	%same =w ceqw %rb, %b
	jnz %same, @run_extend, @run_ready
@run_extend
	%next_r =l add %r, 1
	jmp @run_check
@run_ready
	%is_repeat =w cugel %r, 3
	jnz %is_repeat, @repeat, @literal_start
@repeat
	%ru =l add %used, 2
	%repeat_fits =w culel %ru, %cap
	jnz %repeat_fits, @repeat_write, @fail
@repeat_write
	%rh =l sub 257, %r
	%op =l add %out, %used
	storeb %rh, %op
	%ob =l add %op, 1
	storeb %b, %ob
	jmp @repeat_done
@repeat_done
	%ri =l add %i, %r
	jmp @outer
@literal_start
	%first_j =l add %i, 1
	jmp @literal_check
@literal_check
	%j =l phi @literal_start %first_j, @literal_advance %next_j
	%len =l sub %j, %i
	%literal_end =w ceql %j, %n
	%literal_full =w ceql %len, 128
	%literal_stop =w or %literal_end, %literal_full
	jnz %literal_stop, @literal_packet, @literal_bounds
@literal_bounds
	%j2 =l add %j, 2
	%has_three =w cultl %j2, %n
	jnz %has_three, @literal_probe, @literal_advance
@literal_probe
	%jp =l add %src, %j
	%jp1 =l add %jp, 1
	%jp2 =l add %jp, 2
	%v0 =w loadub %jp
	%v1 =w loadub %jp1
	%v2 =w loadub %jp2
	%eq01 =w ceqw %v0, %v1
	%eq02 =w ceqw %v0, %v2
	%triple =w and %eq01, %eq02
	jnz %triple, @literal_packet, @literal_advance
@literal_advance
	%next_j =l add %j, 1
	jmp @literal_check
@literal_packet
	%packet_size =l add %len, 1
	%lu =l add %used, %packet_size
	%literal_fits =w culel %lu, %cap
	jnz %literal_fits, @literal_header, @fail
@literal_header
	%lh =l sub %len, 1
	%lp =l add %out, %used
	storeb %lh, %lp
	%dest =l add %lp, 1
	jmp @copy
@copy
	%k =l phi @literal_header 0, @copy %next_k
	%cp =l add %sp, %k
	%cv =w loadub %cp
	%dp =l add %dest, %k
	storeb %cv, %dp
	%next_k =l add %k, 1
	%copy_done =w ceql %next_k, %len
	jnz %copy_done, @literal_done, @copy
@literal_done
	jmp @outer
@done
	ret %used
@fail
	ret -1
}
