export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %zero =w ceql %b, 0
    jnz %zero, @divzero, @checkoverflow
@divzero
    storel %a, %rem
    ret 0
@checkoverflow
    %ismin =w ceql %a, -9223372036854775808
    %isnegone =w ceql %b, -1
    %overflow =w and %ismin, %isnegone
    jnz %overflow, @overflowcase, @divide
@overflowcase
    storel 0, %rem
    ret -9223372036854775808
@divide
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
