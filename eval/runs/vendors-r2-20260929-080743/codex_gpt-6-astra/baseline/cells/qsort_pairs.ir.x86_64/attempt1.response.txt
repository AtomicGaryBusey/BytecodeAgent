export function l $qsort_pairs(l %a, l %n) {
@entry
    %small =w cslel %n, 1
    jnz %small, @done, @init

@init
    %last =l sub %n, 1
    %poff =l mul %last, 16
    %pp =l add %a, %poff
    %ppv =l add %pp, 8
    %pk =l loadl %pp
    %pv =l loadl %ppv
    jmp @loop

@loop
    %i =l phi @init 0, @next %ni
    %j =l phi @init 0, @next %nj
    %more =w csltl %j, %last
    jnz %more, @inspect, @partitioned

@inspect
    %jo =l mul %j, 16
    %jp =l add %a, %jo
    %jvp =l add %jp, 8
    %jk =l loadl %jp
    %jv =l loadl %jvp
    %kl =w csltl %jk, %pk
    %ke =w ceql %jk, %pk
    %vl =w csltl %jv, %pv
    %tie =w and %ke, %vl
    %less =w or %kl, %tie
    jnz %less, @swap, @keep

@swap
    %io =l mul %i, 16
    %ip =l add %a, %io
    %ivp =l add %ip, 8
    %ik =l loadl %ip
    %iv =l loadl %ivp
    storel %jk, %ip
    storel %jv, %ivp
    storel %ik, %jp
    storel %iv, %jvp
    %inc =l add %i, 1
    jmp @next

@keep
    jmp @next

@next
    %ni =l phi @swap %inc, @keep %i
    %nj =l add %j, 1
    jmp @loop

@partitioned
    %fo =l mul %i, 16
    %fp =l add %a, %fo
    %fvp =l add %fp, 8
    %fk =l loadl %fp
    %fv =l loadl %fvp
    storel %pk, %fp
    storel %pv, %fvp
    storel %fk, %pp
    storel %fv, %ppv
    %right =l add %fp, 16
    %rn =l sub %last, %i
    %r0 =l call $qsort_pairs(l %a, l %i)
    %r1 =l call $qsort_pairs(l %right, l %rn)
    jmp @done

@done
    ret 0
}
