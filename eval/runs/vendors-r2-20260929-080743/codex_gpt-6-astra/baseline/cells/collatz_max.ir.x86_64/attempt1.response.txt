export function l $collatz_max(l %n) {
@start
	jmp @outer
@outer
	%k =l phi @start 1, @done %next
	%best =l phi @start 0, @done %updated
	jmp @inner
@inner
	%x =l phi @outer %k, @step %nx
	%count =l phi @outer 0, @step %nc
	%finished =w ceql %x, 1
	jnz %finished, @done, @parity
@parity
	%bit =l and %x, 1
	%odd =w cnel %bit, 0
	jnz %odd, @odd, @even
@odd
	%triple =l mul %x, 3
	%ox =l add %triple, 1
	jmp @step
@even
	%ex =l shr %x, 1
	jmp @step
@step
	%nx =l phi @odd %ox, @even %ex
	%nc =l add %count, 1
	jmp @inner
@done
	%greater =w csgtl %count, %best
	%mask =l extsw %greater
	%delta =l sub %count, %best
	%gain =l mul %delta, %mask
	%updated =l add %best, %gain
	%next =l add %k, 1
	%more =w cslel %next, %n
	jnz %more, @outer, @exit
@exit
	ret %updated
}
