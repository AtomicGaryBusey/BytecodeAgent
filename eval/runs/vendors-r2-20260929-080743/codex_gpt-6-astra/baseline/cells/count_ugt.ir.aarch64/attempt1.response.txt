export function l $count_ugt(l %a, l %n, l %t) {
@start
	jnz 0, @loop, @check
@check
	%nonzero =w cnel %n, 0
	jnz %nonzero, @loop, @empty
@loop
	%i =l phi @start 0, @check 0, @body %next
	%count =l phi @start 0, @check 0, @body %total
	%offset =l mul %i, 8
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%greater =w cugtl %value, %t
	%increment =l extuw %greater
	%total =l add %count, %increment
	%next =l add %i, 1
	%more =w cultl %next, %n
	jnz %more, @body, @done
@body
	jmp @loop
@done
	ret %total
@empty
	ret 0
}
