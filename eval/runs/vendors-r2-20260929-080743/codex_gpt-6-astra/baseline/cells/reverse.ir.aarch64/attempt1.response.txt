export function l $reverse(l %a, l %n) {
@start
    %half =l shr %n, 1
    jmp @loop
@loop
    %i =l phi @start 0, @swap %next
    %more =w csltl %i, %half
    jnz %more, @swap, @done
@swap
    %last =l sub %n, 1
    %j =l sub %last, %i
    %ioff =l mul %i, 8
    %joff =l mul %j, 8
    %left =l add %a, %ioff
    %right =l add %a, %joff
    %lv =l loadl %left
    %rv =l loadl %right
    storel %rv, %left
    storel %lv, %right
    %next =l add %i, 1
    jmp @loop
@done
    ret 0
}
