export function l $sdivrem(l %a, l %b, l %rem) {
@start
    %zero =w ceql %b, 0
    jnz %zero, @divzero, @check
@divzero
    storel %a, %rem
    ret 0
@check
    %min =w ceql %a, -9223372036854775808
    %negone =w ceql %b, -1
    %overflow =w and %min, %negone
    jnz %overflow, @special, @normal
@special
    storel 0, %rem
    ret %a
@normal
    %q =l div %a, %b
    %r =l rem %a, %b
    storel %r, %rem
    ret %q
}
