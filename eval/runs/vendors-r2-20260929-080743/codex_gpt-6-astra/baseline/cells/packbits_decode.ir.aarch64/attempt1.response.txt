export function l $packbits_decode(l %src, l %n, l %out, l %cap) {
@start
	jmp @packet
@packet
	%i =l phi @start 0, @noop %after, @done %next
	%j =l phi @start 0, @noop %j, @done %written
	%end =w ceql %i, %n
	jnz %end, @success, @header
@header
	%hp =l add %src, %i
	%hw =w loadsb %hp
	%h =l extsw %hw
	%after =l add %i, 1
	%isnop =w ceql %h, -128
	jnz %isnop, @noop, @classify
@noop
	jmp @packet
@classify
	%literal =w csgel %h, 0
	jnz %literal, @lit, @rep
@lit
	%litlen =l add %h, 1
	jmp @validate
@rep
	%replen =l sub 1, %h
	jmp @validate
@validate
	%len =l phi @lit %litlen, @rep %replen
	%need =l phi @lit %litlen, @rep 1
	%step =l phi @lit 1, @rep 0
	%remaining =l sub %n, %after
	%short =w csgtl %need, %remaining
	jnz %short, @truncated, @prepare
@prepare
	%next =l add %after, %need
	%base =l add %src, %after
	jmp @copy
@copy
	%k =l phi @prepare 0, @store %knext
	%written =l phi @prepare %j, @store %wnext
	%readp =l phi @prepare %base, @store %rnext
	%complete =w ceql %k, %len
	jnz %complete, @done, @space
@space
	%full =w ceql %written, %cap
	jnz %full, @overflow, @store
@store
	%b =w loadub %readp
	%wp =l add %out, %written
	storeb %b, %wp
	%knext =l add %k, 1
	%wnext =l add %written, 1
	%rnext =l add %readp, %step
	jmp @copy
@done
	jmp @packet
@success
	ret %j
@truncated
	ret -2
@overflow
	ret -1
}
