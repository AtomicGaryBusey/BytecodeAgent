binary_search:
    xor ecx, ecx
.Lloop:
    cmp rcx, rsi
    jge .Lmissing
    mov rax, rsi
    sub rax, rcx
    shr rax, 1
    add rax, rcx
    cmp qword ptr [rdi + rax*8], rdx
    je .Ldone
    jl .Lright
    mov rsi, rax
    jmp .Lloop
.Lright:
    lea rcx, [rax + 1]
    jmp .Lloop
.Lmissing:
    mov rax, -1
.Ldone:
    ret
