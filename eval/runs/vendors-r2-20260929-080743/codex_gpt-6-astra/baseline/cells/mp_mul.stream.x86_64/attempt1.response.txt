mp_mul:
push rbx
push rbp
push r12
push r13
cld
mov r9, rdx
mov rbp, rcx
lea r10, [rsi + rcx]
xor eax, eax
.Lzero:
cmp rax, r10
jae .Lzero_done
mov qword ptr [r8 + rax*8], 0
inc rax
jmp .Lzero
.Lzero_done:
test rsi, rsi
jz .Lscan
test rbp, rbp
jz .Lscan
xor r11d, r11d
mov r13, r8
.Louter:
mov r12, qword ptr [rdi + r11*8]
xor ebx, ebx
xor ecx, ecx
.Linner:
mov rax, r12
mul qword ptr [r9 + rcx*8]
add rax, rbx
adc rdx, 0
add rax, qword ptr [r13 + rcx*8]
adc rdx, 0
mov qword ptr [r13 + rcx*8], rax
mov rbx, rdx
inc rcx
cmp rcx, rbp
jb .Linner
mov qword ptr [r13 + rbp*8], rbx
add r13, 8
inc r11
cmp r11, rsi
jb .Louter
.Lscan:
mov rax, r10
.Lscan_loop:
test rax, rax
jz .Ldone
cmp qword ptr [r8 + rax*8 - 8], 0
jne .Ldone
dec rax
jmp .Lscan_loop
.Ldone:
pop r13
pop r12
pop rbp
pop rbx
ret
