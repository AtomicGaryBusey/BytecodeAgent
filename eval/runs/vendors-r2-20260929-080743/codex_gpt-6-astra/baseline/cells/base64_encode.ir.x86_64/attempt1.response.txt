export function l $base64_encode(l %src, l %n, l %out) {
@start
	jmp @group
@group
	%i =l phi @start 0, @next %inext
	%o =l phi @start 0, @next %onext
	%done =w cugel %i, %n
	jnz %done, @end, @first
@first
	%rem =l sub %n, %i
	%p0 =l add %src, %i
	%b0 =l loadub %p0
	%has1 =w cugtl %rem, 1
	jnz %has1, @second, @one
@one
	jmp @pack
@second
	%p1 =l add %p0, 1
	%b1 =l loadub %p1
	%has2 =w cugtl %rem, 2
	jnz %has2, @third, @two
@two
	jmp @pack
@third
	%p2 =l add %p0, 2
	%b2 =l loadub %p2
	jmp @pack
@pack
	%x1 =l phi @one 0, @two %b1, @third %b1
	%x2 =l phi @one 0, @two 0, @third %b2
	%hi =l shl %b0, 16
	%mid =l shl %x1, 8
	%hm =l or %hi, %mid
	%bits =l or %hm, %x2
	%limit =l add %rem, 1
	jmp @character
@character
	%j =l phi @pack 0, @advance %jnext
	%pad =w cugel %j, %limit
	jnz %pad, @padding, @index
@padding
	jmp @write
@index
	%sixj =l mul %j, 6
	%shift =l sub 18, %sixj
	%shifted =l shr %bits, %shift
	%v =l and %shifted, 63
	%upper =w cultl %v, 26
	jnz %upper, @uppercase, @testlower
@uppercase
	%cu =l add %v, 65
	jmp @write
@testlower
	%lower =w cultl %v, 52
	jnz %lower, @lowercase, @testdigit
@lowercase
	%cl =l add %v, 71
	jmp @write
@testdigit
	%digit =w cultl %v, 62
	jnz %digit, @number, @symbol
@number
	%cn =l sub %v, 4
	jmp @write
@symbol
	%last =l sub %v, 62
	%delta =l mul %last, 4
	%cs =l add %delta, 43
	jmp @write
@write
	%c =l phi @padding 61, @uppercase %cu, @lowercase %cl, @number %cn, @symbol %cs
	%pos =l add %o, %j
	%dest =l add %out, %pos
	storeb %c, %dest
	%jnext =l add %j, 1
	%more =w cultl %jnext, 4
	jnz %more, @advance, @next
@advance
	jmp @character
@next
	%inext =l add %i, 3
	%onext =l add %o, 4
	jmp @group
@end
	ret %o
}
