export function l $stats_f(l %a, l %n, l %f, l %out) {
@start
    %x0 =l loadl %a
    %y0 =l call %f(l %x0)
    %odd0 =l and %y0, 1
    jmp @loop
@loop
    %i =l phi @start 1, @next %inext
    %min =l phi @start %y0, @next %minnext
    %max =l phi @start %y0, @next %maxnext
    %sum =l phi @start %y0, @next %sumnext
    %count =l phi @start %odd0, @next %countnext
    %more =w csltl %i, %n
    jnz %more, @body, @done
@body
    %offset =l mul %i, 8
    %p =l add %a, %offset
    %x =l loadl %p
    %y =l call %f(l %x)
    %less =w csltl %y, %min
    jnz %less, @newmin, @keepmin
@newmin
    jmp @checkmax
@keepmin
    jmp @checkmax
@checkmax
    %minnext =l phi @newmin %y, @keepmin %min
    %greater =w csgtl %y, %max
    jnz %greater, @newmax, @keepmax
@newmax
    jmp @next
@keepmax
    jmp @next
@next
    %maxnext =l phi @newmax %y, @keepmax %max
    %sumnext =l add %sum, %y
    %odd =l and %y, 1
    %countnext =l add %count, %odd
    %inext =l add %i, 1
    jmp @loop
@done
    %out1 =l add %out, 8
    %out2 =l add %out, 16
    storel %min, %out
    storel %max, %out1
    storel %sum, %out2
    ret %count
}
