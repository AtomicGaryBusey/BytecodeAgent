export function l $vm_run(l %code, l %len, l %stack, l %cap) {
@start
	%pc =l copy 0
	%d =l copy 0
	jmp @loop
@loop
	%end =w ceql %pc, %len
	jnz %end, @finish, @fetch
@fetch
	%off =l shl %pc, 3
	%ip =l add %code, %off
	%op =l loadl %ip
	%pc =l add %pc, 1
	%halt =w ceql %op, 0
	jnz %halt, @finish, @dispatch1
@dispatch1
	%ispush =w ceql %op, 1
	jnz %ispush, @push, @dispatch2
@dispatch2
	%isdup =w ceql %op, 5
	jnz %isdup, @dup, @dispatch3
@dispatch3
	%isswap =w ceql %op, 6
	jnz %isswap, @swap, @dispatch4
@dispatch4
	%isdrop =w ceql %op, 7
	jnz %isdrop, @drop, @dispatch5
@dispatch5
	%isover =w ceql %op, 8
	jnz %isover, @over, @dispatch6
@dispatch6
	%isjz =w ceql %op, 10
	jnz %isjz, @jz, @dispatch7
@dispatch7
	%isjmp =w ceql %op, 11
	jnz %isjmp, @jump, @binary
@push
	%operand =l add %ip, 8
	%v =l loadl %operand
	%pc =l add %pc, 1
	jmp @append
@dup
	%di =l sub %d, 1
	%doff =l shl %di, 3
	%dp =l add %stack, %doff
	%v =l loadl %dp
	jmp @append
@over
	%oi =l sub %d, 2
	%ooff =l shl %oi, 3
	%optr =l add %stack, %ooff
	%v =l loadl %optr
	jmp @append
@append
	%voff =l shl %d, 3
	%vp =l add %stack, %voff
	storel %v, %vp
	%d =l add %d, 1
	jmp @loop
@swap
	%si =l sub %d, 2
	%soff =l shl %si, 3
	%sa =l add %stack, %soff
	%sb =l add %sa, 8
	%sav =l loadl %sa
	%sbv =l loadl %sb
	storel %sbv, %sa
	storel %sav, %sb
	jmp @loop
@drop
	%d =l sub %d, 1
	jmp @loop
@jz
	%d =l sub %d, 1
	%zoff =l shl %d, 3
	%zp =l add %stack, %zoff
	%zv =l loadl %zp
	%zero =w ceql %zv, 0
	jnz %zero, @jump, @jznext
@jznext
	%pc =l add %pc, 1
	jmp @loop
@jump
	%jp =l add %ip, 8
	%pc =l loadl %jp
	jmp @loop
@binary
	%bi =l sub %d, 2
	%boff =l shl %bi, 3
	%ap =l add %stack, %boff
	%bp =l add %ap, 8
	%a =l loadl %ap
	%b =l loadl %bp
	%isadd =w ceql %op, 2
	jnz %isadd, @add, @binsub
@binsub
	%issub =w ceql %op, 3
	jnz %issub, @sub, @binmul
@binmul
	%ismul =w ceql %op, 4
	jnz %ismul, @mul, @lt
@add
	%r =l add %a, %b
	jmp @binaryend
@sub
	%r =l sub %a, %b
	jmp @binaryend
@mul
	%r =l mul %a, %b
	jmp @binaryend
@lt
	%less =w csltl %a, %b
	%r =l extuw %less
	jmp @binaryend
@binaryend
	storel %r, %ap
	%d =l sub %d, 1
	jmp @loop
@finish
	%i =l copy %d
	jmp @clearcheck
@clearcheck
	%done =w ceql %i, %cap
	jnz %done, @result, @clear
@clear
	%coff =l shl %i, 3
	%cp =l add %stack, %coff
	storel 0, %cp
	%i =l add %i, 1
	jmp @clearcheck
@result
	%empty =w ceql %d, 0
	jnz %empty, @retzero, @rettop
@retzero
	ret 0
@rettop
	%ti =l sub %d, 1
	%toff =l shl %ti, 3
	%tp =l add %stack, %toff
	%top =l loadl %tp
	ret %top
}
