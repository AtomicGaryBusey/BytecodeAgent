qsort_pairs:
cld
cmp rsi, 1
jle .Ldone
push rbx
push r12
push r13
mov rbx, rdi
mov r12, rsi
mov r13, rdi
mov rax, rsi
dec rax
shl rax, 4
lea rsi, [rdi + rax]
mov r8, qword ptr [rsi]
mov r9, qword ptr [rsi + 8]
mov rdx, rdi
.Lpartition:
cmp rdx, rsi
je .Lpivot
mov r10, qword ptr [rdx]
mov r11, qword ptr [rdx + 8]
cmp r10, r8
jl .Lswap
jg .Lnext
cmp r11, r9
jg .Lnext
.Lswap:
mov rax, qword ptr [r13]
mov rcx, qword ptr [r13 + 8]
mov qword ptr [r13], r10
mov qword ptr [r13 + 8], r11
mov qword ptr [rdx], rax
mov qword ptr [rdx + 8], rcx
add r13, 16
.Lnext:
add rdx, 16
jmp .Lpartition
.Lpivot:
mov rax, qword ptr [r13]
mov rcx, qword ptr [r13 + 8]
mov qword ptr [r13], r8
mov qword ptr [r13 + 8], r9
mov qword ptr [rsi], rax
mov qword ptr [rsi + 8], rcx
mov rdi, rbx
mov rsi, r13
sub rsi, rbx
shr rsi, 4
call qsort_pairs
mov rax, r13
sub rax, rbx
shr rax, 4
mov rsi, r12
sub rsi, rax
dec rsi
lea rdi, [r13 + 16]
call qsort_pairs
pop r13
pop r12
pop rbx
.Ldone:
xor eax, eax
ret
