packbits_encode:
push rbx
xor r8d, r8d
xor r9d, r9d
cld
.Lnext:
cmp r8, rsi
jae .Ldone
movzx eax, byte ptr [rdi + r8]
mov r10d, 1
.Lmeasure:
cmp r10, 128
jae .Lmeasured
lea r11, [r8 + r10]
cmp r11, rsi
jae .Lmeasured
cmp byte ptr [rdi + r11], al
jne .Lmeasured
inc r10
jmp .Lmeasure
.Lmeasured:
cmp r10, 3
jb .Lliteral
cmp rcx, 2
jb .Lfail
mov r11d, 257
sub r11d, r10d
mov byte ptr [rdx], r11b
mov byte ptr [rdx + 1], al
add r8, r10
add rdx, 2
sub rcx, 2
add r9, 2
jmp .Lnext
.Lliteral:
mov rbx, r8
.Lscan:
cmp rbx, rsi
jae .Lemit_literal
mov r10, rbx
sub r10, r8
cmp r10, 128
jae .Lemit_literal
lea r11, [rbx + 2]
cmp r11, rsi
jae .Linclude
movzx eax, byte ptr [rdi + rbx]
cmp byte ptr [rdi + rbx + 1], al
jne .Linclude
cmp byte ptr [rdi + rbx + 2], al
je .Lemit_literal
.Linclude:
inc rbx
jmp .Lscan
.Lemit_literal:
mov r10, rbx
sub r10, r8
lea r11, [r10 + 1]
cmp rcx, r11
jb .Lfail
sub rcx, r11
add r9, r11
lea eax, [r10 - 1]
mov byte ptr [rdx], al
inc rdx
.Lcopy:
movzx eax, byte ptr [rdi + r8]
mov byte ptr [rdx], al
inc r8
inc rdx
cmp r8, rbx
jb .Lcopy
jmp .Lnext
.Lfail:
mov rax, -1
pop rbx
ret
.Ldone:
mov rax, r9
pop rbx
ret
