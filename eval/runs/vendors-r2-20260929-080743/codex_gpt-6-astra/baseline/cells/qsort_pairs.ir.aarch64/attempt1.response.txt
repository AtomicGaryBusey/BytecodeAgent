export function l $qsort_pairs(l %a, l %n) {
@entry
	%small =w cslel %n, 1
	jnz %small, @done, @setup

@setup
	%last =l sub %n, 1
	%poff =l mul %last, 16
	%pp =l add %a, %poff
	%ppv =l add %pp, 8
	%pk =l loadl %pp
	%pv =l loadl %ppv
	jmp @loop

@loop
	%j =l phi @setup 0, @advance %nextj
	%i =l phi @setup 0, @advance %nexti
	%more =w csltl %j, %last
	jnz %more, @compare, @partitioned

@compare
	%jo =l mul %j, 16
	%jp =l add %a, %jo
	%jpv =l add %jp, 8
	%jk =l loadl %jp
	%jv =l loadl %jpv
	%kl =w csltl %jk, %pk
	%ke =w ceql %jk, %pk
	%vl =w csltl %jv, %pv
	%tie =w and %ke, %vl
	%less =w or %kl, %tie
	jnz %less, @swap, @keep

@swap
	%io =l mul %i, 16
	%ip =l add %a, %io
	%ipv =l add %ip, 8
	%ik =l loadl %ip
	%iv =l loadl %ipv
	storel %jk, %ip
	storel %jv, %ipv
	storel %ik, %jp
	storel %iv, %jpv
	%inc =l add %i, 1
	jmp @advance

@keep
	jmp @advance

@advance
	%nexti =l phi @swap %inc, @keep %i
	%nextj =l add %j, 1
	jmp @loop

@partitioned
	%fo =l mul %i, 16
	%fp =l add %a, %fo
	%fpv =l add %fp, 8
	%fk =l loadl %fp
	%fv =l loadl %fpv
	storel %pk, %fp
	storel %pv, %fpv
	storel %fk, %pp
	storel %fv, %ppv
	%right =l add %fp, 16
	%rn =l sub %last, %i
	%r1 =l call $qsort_pairs(l %a, l %i)
	%r2 =l call $qsort_pairs(l %right, l %rn)
	jmp @done

@done
	ret 0
}
