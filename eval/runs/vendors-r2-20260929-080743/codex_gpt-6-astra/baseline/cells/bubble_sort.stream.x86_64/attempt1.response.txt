bubble_sort:
xor eax, eax
cmp rsi, 1
jle .Ldone
dec rsi
.Lpass:
xor ecx, ecx
xor r8d, r8d
.Lloop:
mov rdx, qword ptr [rdi + rcx*8]
mov r9, qword ptr [rdi + rcx*8 + 8]
cmp rdx, r9
jle .Lnext
mov qword ptr [rdi + rcx*8], r9
mov qword ptr [rdi + rcx*8 + 8], rdx
mov r8d, 1
.Lnext:
inc rcx
cmp rcx, rsi
jl .Lloop
test r8d, r8d
jz .Ldone
dec rsi
jnz .Lpass
.Ldone:
ret
