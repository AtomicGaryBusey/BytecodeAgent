export function l $reverse(l %a, l %n) {
@start
    %half =l shr %n, 1
    %last =l sub %n, 1
    jmp @loop
@loop
    %i =l phi @start 0, @swap %next
    %more =w csltl %i, %half
    jnz %more, @swap, @done
@swap
    %j =l sub %last, %i
    %ioff =l mul %i, 8
    %joff =l mul %j, 8
    %ip =l add %a, %ioff
    %jp =l add %a, %joff
    %iv =l loadl %ip
    %jv =l loadl %jp
    storel %jv, %ip
    storel %iv, %jp
    %next =l add %i, 1
    jmp @loop
@done
    ret 0
}
