vm_run:
    xor r8d, r8d
    xor r9d, r9d
.Lloop:
    cmp r8, rsi
    je .Lstop
    mov rax, qword ptr [rdi + r8*8]
    inc r8
    test rax, rax
    je .Lstop
    cmp rax, 1
    je .Lpush
    cmp rax, 5
    je .Ldup
    cmp rax, 6
    je .Lswap
    cmp rax, 7
    je .Ldrop
    cmp rax, 8
    je .Lover
    cmp rax, 10
    je .Ljz
    cmp rax, 11
    je .Ljmp
    dec r9
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8]
    cmp rax, 2
    je .Ladd
    cmp rax, 3
    je .Lsub
    cmp rax, 4
    je .Lmul
    cmp r10, r11
    setl r10b
    movzx r10d, r10b
    jmp .Lstore
.Ladd:
    add r10, r11
    jmp .Lstore
.Lsub:
    sub r10, r11
    jmp .Lstore
.Lmul:
    imul r10, r11
.Lstore:
    mov qword ptr [rdx + r9*8 - 8], r10
    jmp .Lloop
.Lpush:
    mov r10, qword ptr [rdi + r8*8]
    inc r8
    jmp .Lappend
.Ldup:
    mov r10, qword ptr [rdx + r9*8 - 8]
    jmp .Lappend
.Lover:
    mov r10, qword ptr [rdx + r9*8 - 16]
.Lappend:
    mov qword ptr [rdx + r9*8], r10
    inc r9
    jmp .Lloop
.Lswap:
    mov r10, qword ptr [rdx + r9*8 - 8]
    mov r11, qword ptr [rdx + r9*8 - 16]
    mov qword ptr [rdx + r9*8 - 16], r10
    mov qword ptr [rdx + r9*8 - 8], r11
    jmp .Lloop
.Ldrop:
    dec r9
    jmp .Lloop
.Ljz:
    dec r9
    cmp qword ptr [rdx + r9*8], 0
    je .Ljmp
    inc r8
    jmp .Lloop
.Ljmp:
    mov r8, qword ptr [rdi + r8*8]
    jmp .Lloop
.Lstop:
    xor eax, eax
    test r9, r9
    je .Lclear
    mov rax, qword ptr [rdx + r9*8 - 8]
.Lclear:
    cmp r9, rcx
    je .Lreturn
    mov qword ptr [rdx + r9*8], 0
    inc r9
    jmp .Lclear
.Lreturn:
    cld
    ret
