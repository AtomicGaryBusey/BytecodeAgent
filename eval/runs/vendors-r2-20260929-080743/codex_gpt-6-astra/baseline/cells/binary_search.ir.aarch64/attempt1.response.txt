export function l $binary_search(l %a, l %n, l %key) {
@start
	jmp @loop
@loop
	%lo =l phi @start 0, @lower %next, @upper %lo
	%hi =l phi @start %n, @lower %hi, @upper %mid
	%nonempty =w csltl %lo, %hi
	jnz %nonempty, @probe, @missing
@probe
	%sum =l add %lo, %hi
	%mid =l shr %sum, 1
	%offset =l shl %mid, 3
	%ptr =l add %a, %offset
	%value =l loadl %ptr
	%equal =w ceql %value, %key
	jnz %equal, @found, @compare
@compare
	%less =w csltl %value, %key
	jnz %less, @lower, @upper
@lower
	%next =l add %mid, 1
	jmp @loop
@upper
	jmp @loop
@found
	ret %mid
@missing
	ret -1
}
