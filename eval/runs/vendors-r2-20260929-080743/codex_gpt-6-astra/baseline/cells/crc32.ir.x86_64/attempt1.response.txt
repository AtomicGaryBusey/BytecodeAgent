export function l $crc32(l %p, l %n) {
@start
	jmp @bytes
@bytes
	%i =l phi @start 0, @next %inext
	%crc =w phi @start 4294967295, @next %updated
	%more =w cultl %i, %n
	jnz %more, @byte, @done
@byte
	%addr =l add %p, %i
	%b =w loadub %addr
	%mixed =w xor %crc, %b
	jmp @bits
@bits
	%k =w phi @byte 0, @bits %knext
	%c =w phi @byte %mixed, @bits %updated
	%low =w and %c, 1
	%mask =w sub 0, %low
	%poly =w and %mask, 3988292384
	%shifted =w shr %c, 1
	%updated =w xor %shifted, %poly
	%knext =w add %k, 1
	%again =w cultw %knext, 8
	jnz %again, @bits, @next
@next
	%inext =l add %i, 1
	jmp @bytes
@done
	%final =w xor %crc, 4294967295
	%result =l extuw %final
	ret %result
}
