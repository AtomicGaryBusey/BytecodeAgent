memmove_i64:
    cld
    xor eax, eax
    test rdx, rdx
    jz .Ldone
    cmp rdi, rsi
    je .Ldone
    ja .Lbackward
    xor ecx, ecx
.Lforward:
    mov r8, qword ptr [rsi + rcx*8]
    mov qword ptr [rdi + rcx*8], r8
    inc rcx
    cmp rcx, rdx
    jb .Lforward
    ret
.Lbackward:
    dec rdx
    mov r8, qword ptr [rsi + rdx*8]
    mov qword ptr [rdi + rdx*8], r8
    test rdx, rdx
    jnz .Lbackward
.Ldone:
    ret
