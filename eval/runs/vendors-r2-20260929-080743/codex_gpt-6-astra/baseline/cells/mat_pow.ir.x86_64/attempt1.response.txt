export function l $mat_pow(l %a, l %n, l %k, l %out, l %tmp) {
@start
    %count =l mul %n, %n
    %bytes =l mul %count, 8
    %scratchcount =l mul %count, 2
    jmp @init
@init
    %idx =l phi @start 0, @initnext %idxnext
    %row =l udiv %idx, %n
    %col =l urem %idx, %n
    %diag =w ceql %row, %col
    %value =l extuw %diag
    %off =l mul %idx, 8
    %op =l add %out, %off
    storel %value, %op
    %idxnext =l add %idx, 1
    %initmore =w cultl %idxnext, %count
    jnz %initmore, @initnext, @initialized
@initnext
    jmp @init
@initialized
    jmp @power
@power
    %step =l phi @initialized 0, @copied %nextstep
    %more =w cultl %step, %k
    jnz %more, @productstart, @clearstart
@productstart
    jmp @element
@element
    %e =l phi @productstart 0, @elementnext %enext
    %i =l udiv %e, %n
    %j =l urem %e, %n
    %rowindex =l mul %i, %n
    jmp @dot
@dot
    %t =l phi @element 0, @dotnext %tnext
    %sum =l phi @element 0, @dotnext %sumnew
    %leftindex =l add %rowindex, %t
    %leftoffset =l mul %leftindex, 8
    %leftptr =l add %out, %leftoffset
    %left =l loadl %leftptr
    %rightrow =l mul %t, %n
    %rightindex =l add %rightrow, %j
    %rightoffset =l mul %rightindex, 8
    %rightptr =l add %a, %rightoffset
    %right =l loadl %rightptr
    %term =l mul %left, %right
    %sumnew =l add %sum, %term
    %tnext =l add %t, 1
    %dotmore =w cultl %tnext, %n
    jnz %dotmore, @dotnext, @storeproduct
@dotnext
    jmp @dot
@storeproduct
    %eoffset =l mul %e, 8
    %eptr =l add %tmp, %eoffset
    storel %sumnew, %eptr
    %enext =l add %e, 1
    %elementmore =w cultl %enext, %count
    jnz %elementmore, @elementnext, @copystart
@elementnext
    jmp @element
@copystart
    jmp @copy
@copy
    %c =l phi @copystart 0, @copynext %cnext
    %coffset =l mul %c, 8
    %src =l add %tmp, %coffset
    %dst =l add %out, %coffset
    %cv =l loadl %src
    storel %cv, %dst
    %cnext =l add %c, 1
    %copymore =w cultl %cnext, %count
    jnz %copymore, @copynext, @copied
@copynext
    jmp @copy
@copied
    %nextstep =l add %step, 1
    jmp @power
@clearstart
    jmp @clear
@clear
    %z =l phi @clearstart 0, @clearnext %znext
    %zoffset =l mul %z, 8
    %zptr =l add %tmp, %zoffset
    storel 0, %zptr
    %znext =l add %z, 1
    %clearmore =w cultl %znext, %scratchcount
    jnz %clearmore, @clearnext, @done
@clearnext
    jmp @clear
@done
    ret 0
}
