export function l $bubble_sort(l %a, l %n) {
@start
	%small =w csltl %n, 2
	jnz %small, @done, @init
@init
	%limit0 =l sub %n, 1
	jmp @outer
@outer
	%limit =l phi @init %limit0, @next %reduced
	jmp @inner
@inner
	%i =l phi @outer 0, @advance %j
	%offset =l mul %i, 8
	%p =l add %a, %offset
	%q =l add %p, 8
	%x =l loadl %p
	%y =l loadl %q
	%swap =w csgtl %x, %y
	jnz %swap, @exchange, @advance
@exchange
	storel %y, %p
	storel %x, %q
	jmp @advance
@advance
	%j =l add %i, 1
	%more =w csltl %j, %limit
	jnz %more, @inner, @next
@next
	%reduced =l sub %limit, 1
	%again =w csgtl %reduced, 0
	jnz %again, @outer, @done
@done
	ret 0
}
