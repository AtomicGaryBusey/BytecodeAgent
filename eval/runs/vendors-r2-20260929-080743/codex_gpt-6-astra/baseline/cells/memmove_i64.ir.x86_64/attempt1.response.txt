export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty =w ceql %n, 0
    jnz %empty, @done, @direction
@direction
    %before =w cultl %dst, %src
    jnz %before, @forward, @backward
@forward
    %i =l phi @direction 0, @forward %next
    %offset =l shl %i, 3
    %sp =l add %src, %offset
    %dp =l add %dst, %offset
    %value =l loadl %sp
    storel %value, %dp
    %next =l add %i, 1
    %more =w cultl %next, %n
    jnz %more, @forward, @done
@backward
    %remaining =l phi @direction %n, @backward %j
    %j =l sub %remaining, 1
    %backoffset =l shl %j, 3
    %backsp =l add %src, %backoffset
    %backdp =l add %dst, %backoffset
    %backvalue =l loadl %backsp
    storel %backvalue, %backdp
    %backmore =w cnel %j, 0
    jnz %backmore, @backward, @done
@done
    ret 0
}
