export function l $bubble_sort(l %a, l %n) {
@start
	%small =w cslel %n, 1
	jnz %small, @done, @init
@init
	%initial =l sub %n, 1
	jmp @outer
@outer
	%limit =l phi @init %initial, @pass_end %next_limit
	jmp @inner
@inner
	%i =l phi @outer 0, @advance %next_i
	%offset =l mul %i, 8
	%left_ptr =l add %a, %offset
	%right_ptr =l add %left_ptr, 8
	%left =l loadl %left_ptr
	%right =l loadl %right_ptr
	%swap =w csgtl %left, %right
	jnz %swap, @exchange, @advance
@exchange
	storel %right, %left_ptr
	storel %left, %right_ptr
	jmp @advance
@advance
	%next_i =l add %i, 1
	%more =w csltl %next_i, %limit
	jnz %more, @inner, @pass_end
@pass_end
	%next_limit =l sub %limit, 1
	%again =w csgtl %next_limit, 0
	jnz %again, @outer, @done
@done
	ret 0
}
