export function l $bubble_sort(l %a, l %n) {
@start
	%small = cslel %n, 1
	jnz %small, @done, @init
@init
	%initial = sub %n, 1
	jmp @outer
@outer
	%limit = phi @init %initial, @pass_end %next_limit
	jmp @inner
@inner
	%i = phi @outer 0, @advance %next_i
	%offset = mul %i, 8
	%left_ptr = add %a, %offset
	%right_ptr = add %left_ptr, 8
	%left = loadl %left_ptr
	%right = loadl %right_ptr
	%swap = csgtl %left, %right
	jnz %swap, @exchange, @advance
@exchange
	storel %right, %left_ptr
	storel %left, %right_ptr
	jmp @advance
@advance
	%next_i = add %i, 1
	%more = csltl %next_i, %limit
	jnz %more, @inner, @pass_end
@pass_end
	%next_limit = sub %limit, 1
	%again = csgtl %next_limit, 0
	jnz %again, @outer, @done
@done
	ret 0
}
