export function l $memmove_i64(l %dst, l %src, l %n) {
@start
    %empty =w ceql %n, 0
    jnz %empty, @done, @direction
@direction
    %before =w cultl %dst, %src
    jnz %before, @forward, @backward
@forward
    %i =l phi @direction 0, @forward %next
    %offset =l shl %i, 3
    %sptr =l add %src, %offset
    %dptr =l add %dst, %offset
    %value =l loadl %sptr
    storel %value, %dptr
    %next =l add %i, 1
    %more =w cultl %next, %n
    jnz %more, @forward, @done
@backward
    %remaining =l phi @direction %n, @backward %index
    %index =l sub %remaining, 1
    %boffset =l shl %index, 3
    %bsptr =l add %src, %boffset
    %bdptr =l add %dst, %boffset
    %bvalue =l loadl %bsptr
    storel %bvalue, %bdptr
    %bmore =w cnel %index, 0
    jnz %bmore, @backward, @done
@done
    ret 0
}
