.globl _start

.equ SYS_READ,      0
.equ SYS_WRITE,     1
.equ SYS_EXIT,     60
.equ INPUT_BUF_SIZE,65536
.equ STACK_CAP,    8192            # bytes, 1024 entries
.equ OUTBUF_SIZE,   32
.equ ERR_MSG_LEN,    6            # "error\n"
.equ INT64_MIN, 0x8000000000000000

.text
_start:
    xor     rbx, rbx              # offset = 0
read_loop:
    mov     rdi, 0                # fd = stdin
    lea     rsi, [rip + input_buf]
    add     rsi, rbx              # buf = input_buf + offset
    mov     rdx, INPUT_BUF_SIZE
    sub     rdx, rbx              # count = remaining space
    mov     rax, SYS_READ
    syscall
    test    rax, rax
    jz      read_done
    js      read_error
    add     rbx, rax
    cmp     rbx, INPUT_BUF_SIZE
    jae     read_done
    jmp     read_loop
read_error:
    mov     rax, SYS_EXIT
    xor     rdi, rdi
    syscall
read_done:
    mov     r13, rbx              # total input length
    lea     r14, [rip + input_buf]   # current pointer
    mov     r15, r13                 # remaining bytes

outer_loop:
    cmp     r15, 0
    je      exit_program

    xor     r12, r12                # stack count = 0
    xor     r10, r10                # line_error = 0

line_loop:
    # skip spaces/tabs
skip_spaces:
    cmp     r15, 0
    je      line_end
    mov     al, byte ptr [r14]
    cmp     al, ' '
    je      inc_ptr
    cmp     al, 9
    je      inc_ptr
    jmp     check_newline
inc_ptr:
    inc     r14
    dec     r15
    jmp     skip_spaces
check_newline:
    cmp     al, 10
    je      line_end

    # token start
    mov     rdx, r14            # token_start
    xor     rcx, rcx            # token_len = 0

token_scan:
    cmp     r15, 0
    je      token_done
    mov     al, byte ptr [r14]
    cmp     al, ' '
    je      token_done
    cmp     al, 9
    je      token_done
    cmp     al, 10
    je      token_done
    inc     r14
    dec     r15
    inc     rcx
    jmp     token_scan
token_done:
    cmp     r10, 1
    jne     token_process
    jmp     line_loop            # already error, skip rest of line

token_process:
    # single-char operators ?
    cmp     rcx, 1
    jne     check_number
    mov     al, byte ptr [rdx]
    cmp     al, '+'
    je      op_add
    cmp     al, '-'
    je      op_sub
    cmp     al, '*'
    je      op_mul
    cmp     al, '/'
    je      op_div
    cmp     al, '%'
    je      op_mod
    # not an operator, treat as number
check_number:
    # parse integer literal
    mov     rsi, rdx                # token ptr
    mov     rbx, rcx                # token length
    xor     r8d, r8d                # sign flag = 0
    mov     al, byte ptr [rsi]
    cmp     al, '-'
    jne     num_digits
    mov     r8d, 1
    inc     rsi
    dec     rbx
num_digits:
    cmp     rbx, 0
    je      invalid_token
    xor     rax, rax                # accumulator
num_loop:
    cmp     rbx, 0
    je      num_done
    mov     al, byte ptr [rsi]
    cmp     al, '0'
    jb      invalid_token
    cmp     al, '9'
    ja      invalid_token
    sub     al, '0'
    movzx   rdx, al
    imul    rax, rax, 10
    add     rax, rdx
    inc     rsi
    dec     rbx
    jmp     num_loop
num_done:
    cmp     r8d, 0
    je      push_number
    neg     rax
push_number:
    lea     rbx, [rip + stack]
    mov     [rbx + r12*8], rax
    inc     r12
    jmp     line_loop

invalid_token:
    mov     r10, 1                  # line_error = 1
    jmp     line_loop

# ---------- operators ----------
op_add:
    cmp     r12, 2
    jl      set_error
    lea     rbx, [rip + stack]
    dec     r12
    mov     rdx, [rbx + r12*8]      # b
    dec     r12
    mov     rcx, [rbx + r12*8]      # a
    add     rcx, rdx
    mov     [rbx + r12*8], rcx
    inc     r12
    jmp     line_loop

op_sub:
    cmp     r12, 2
    jl      set_error
    lea     rbx, [rip + stack]
    dec     r12
    mov     rdx, [rbx + r12*8]      # b
    dec     r12
    mov     rcx, [rbx + r12*8]      # a
    sub     rcx, rdx
    mov     [rbx + r12*8], rcx
    inc     r12
    jmp     line_loop

op_mul:
    cmp     r12, 2
    jl      set_error
    lea     rbx, [rip + stack]
    dec     r12
    mov     rdx, [rbx + r12*8]      # b
    dec     r12
    mov     rcx, [rbx + r12*8]      # a
    imul    rcx, rdx
    mov     [rbx + r12*8], rcx
    inc     r12
    jmp     line_loop

op_div:
    cmp     r12, 2
    jl      set_error
    lea     rbx, [rip + stack]
    dec     r12
    mov     rdx, [rbx + r12*8]      # b
    dec     r12
    mov     rcx, [rbx + r12*8]      # a
    cmp     rdx, 0
    je      set_error
    mov     rax, rcx
    mov     r9, qword ptr [rip + int64_min]
    cmp     rax, r9
    jne     .div_ok
    mov     r8, -1
    cmp     rdx, r8
    je      set_error
.div_ok:
    mov     r8, rdx                 # divisor
    cqo
    idiv    r8
    mov     [rbx + r12*8], rax
    inc     r12
    jmp     line_loop

op_mod:
    cmp     r12, 2
    jl      set_error
    lea     rbx, [rip + stack]
    dec     r12
    mov     rdx, [rbx + r12*8]      # b
    dec     r12
    mov     rcx, [rbx + r12*8]      # a
    cmp     rdx, 0
    je      set_error
    mov     rax, rcx
    mov     r9, qword ptr [rip + int64_min]
    cmp     rax, r9
    jne     .mod_ok
    mov     r8, -1
    cmp     rdx, r8
    je      set_error
.mod_ok:
    mov     r8, rdx                 # divisor
    cqo
    idiv    r8
    mov     [rbx + r12*8], rdx
    inc     r12
    jmp     line_loop

set_error:
    mov     r10, 1
    jmp     line_loop

line_end:
    cmp     r10, 1
    je      output_error
    cmp     r12, 1
    jne     output_error

    # success: write the number
    lea     rbx, [rip + stack]
    mov     rax, [rbx]          # value
    call    write_number
    # newline
    mov     rax, SYS_WRITE
    mov     rdi, 1
    lea     rsi, [rip + nl]
    mov     rdx, 1
    syscall
    jmp     after_line

output_error:
    mov     rax, SYS_WRITE
    mov     rdi, 1
    lea     rsi, [rip + err_msg]
    mov     rdx, ERR_MSG_LEN
    syscall

after_line:
    # consume newline if present
    cmp     r15, 0
    je      outer_loop
    mov     al, byte ptr [r14]
    cmp     al, 10
    jne     outer_loop
    inc     r14
    dec     r15
    jmp     outer_loop

exit_program:
    mov     rax, SYS_EXIT
    xor     rdi, rdi
    syscall

# -------------------------------------------------
# write_number: rax = signed 64-bit value, writes decimal without newline
# preserves r12‑r15
write_number:
    push    r12
    push    r13
    push    r14
    push    r15

    xor     rbx, rbx                # sign flag = 0
    test    rax, rax
    jns     .wnum_pos
    mov     rbx, 1
    neg     rax
.wnum_pos:
    lea     rsi, [rip + outbuf + OUTBUF_SIZE]   # point just after buffer
    xor     rcx, rcx                # length = 0

    cmp     rax, 0
    jne     .wnum_loop
    dec     rsi
    mov     byte ptr [rsi], '0'
    inc     rcx
    jmp     .wnum_done
.wnum_loop:
    xor     rdx, rdx
    mov     r8, 10
    div     r8
    dec     rsi
    add     dl, '0'
    mov     byte ptr [rsi], dl
    inc     rcx
    cmp     rax, 0
    jne     .wnum_loop
.wnum_done:
    cmp     rbx, 1
    jne     .wnum_no_sign
    dec     rsi
    mov     byte ptr [rsi], '-'
    inc     rcx
.wnum_no_sign:
    mov     rax, SYS_WRITE
    mov     rdi, 1
    mov     rdx, rcx
    syscall

    pop     r15
    pop     r14
    pop     r13
    pop     r12
    ret

# -------------------------------------------------
# Data
.section .rodata
err_msg:    .ascii "error\n"
nl:         .byte 0x0A
int64_min:  .quad INT64_MIN

.section .bss
.align 8
input_buf:  .zero INPUT_BUF_SIZE
stack:      .zero STACK_CAP
outbuf:     .zero OUTBUF_SIZE
