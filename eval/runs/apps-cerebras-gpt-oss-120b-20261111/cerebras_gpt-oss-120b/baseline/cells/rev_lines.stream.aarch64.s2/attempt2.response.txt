	.global	_start
	.text
	.align	3

// System call numbers
SYS_READ		= 63
SYS_WRITE		= 64
SYS_EXIT		= 93

// File descriptors
FD_STDIN		= 0
FD_STDOUT		= 1

// Buffer size (max input)
BUF_SIZE		= 65536

_start:
	// addresses of buffers
	adr	x9, input_buf          // x9 = input buffer base
	adr	x20, output_buf        // x20 = output buffer base

	// total bytes read so far -> x10 = 0
	mov	x10, #0

read_loop:
	// read(fd=0, buf = x9 + x10, count = BUF_SIZE - x10)
	mov	x0, #FD_STDIN
	add	x1, x9, x10
	mov	x2, #BUF_SIZE
	sub	x2, x2, x10
	mov	x8, #SYS_READ
	svc	#0

	// result in x0
	cmp	x0, #0
	beq	read_done
	add	x10, x10, x0
	b	read_loop

read_done:
	// x11 = start pointer of current line
	mov	x11, x9
	// x12 = end pointer = input_buf + total_len
	add	x12, x9, x10

process_loop:
	cmp	x11, x12
	b.ge	finish

	// scan for newline
	mov	x13, x11
scan_newline:
	cmp	x13, x12
	b.eq	no_newline_found
	ldrb	w0, [x13]
	cmp	w0, #10               // '\n'
	b.eq	newline_found
	add	x13, x13, #1
	b	scan_newline

newline_found:
	// line length = x13 - x11
	sub	x14, x13, x11          // x14 = length

	// preserve length in x14 for later write
	// reverse copy
	bl	reverse_copy

	// write reversed line if length > 0
	cmp	x14, #0
	beq	skip_write_line
	mov	x0, #FD_STDOUT
	mov	x1, x20
	mov	x2, x14
	mov	x8, #SYS_WRITE
	svc	#0
skip_write_line:
	// write newline
	mov	x0, #FD_STDOUT
	adr	x1, nl_char
	mov	x2, #1
	mov	x8, #SYS_WRITE
	svc	#0

	// advance start pointer past newline
	add	x11, x13, #1
	b	process_loop

no_newline_found:
	// final line without trailing newline
	sub	x14, x12, x11          // length
	cmp	x14, #0
	beq	finish

	bl	reverse_copy

	// write reversed final line
	mov	x0, #FD_STDOUT
	mov	x1, x20
	mov	x2, x14
	mov	x8, #SYS_WRITE
	svc	#0

	// write newline (always)
	mov	x0, #FD_STDOUT
	adr	x1, nl_char
	mov	x2, #1
	mov	x8, #SYS_WRITE
	svc	#0
	b	finish

// -------------------------------------------------------------
 // reverse_copy:
 //   Input: x11 = src start, x14 = length, x20 = dst start
 //   Clobbers: x5-x6, x15, w0
reverse_copy:
	// copy length to mutable register
	mov	x15, x14
	cmp	x15, #0
	beq	rev_done

	// src_end = src + len - 1
	add	x5, x11, x15
	sub	x5, x5, #1

	// dst_ptr = dst (x20)
	mov	x6, x20

rev_loop:
	ldrb	w0, [x5]          // load byte
	strb	w0, [x6], #1      // store and post-increment
	subs	x5, x5, #1
	subs	x15, x15, #1
	bne	rev_loop

rev_done:
	ret

// -------------------------------------------------------------
finish:
	mov	x0, #0
	mov	x8, #SYS_EXIT
	svc	#0

	.section .rodata
	.align	3
nl_char:
	.byte	0x0A

	.bss
	.align	3
input_buf:
	.zero	BUF_SIZE
output_buf:
	.zero	BUF_SIZE
