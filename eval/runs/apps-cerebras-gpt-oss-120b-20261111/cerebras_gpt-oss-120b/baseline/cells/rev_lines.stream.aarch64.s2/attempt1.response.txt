	.global	_start
	.text
	.align	2

// System call numbers
SYS_READ		= 63
SYS_WRITE		= 64
SYS_EXIT		= 93

// File descriptors
FD_STDIN		= 0
FD_STDOUT		= 1

// Buffer size (max input)
BUF_SIZE		= 65536

// ---------------------------------------------------------------------
// Entry point
_start:
	// -------------------------------------------------
	// Load addresses of buffers
	adr	x9, input_buf          // x9 = input buffer base
	adr	x20, output_buf        // x20 = output buffer base

	// -------------------------------------------------
	// Read loop: fill input_buf, total length in x10
	mov	x10, #0                // x10 = bytes read so far

read_loop:
	// x0 = fd
	mov	x0, #FD_STDIN
	// x1 = buffer pointer = input_buf + offset
	add	x1, x9, x10
	// x2 = remaining space = BUF_SIZE - offset
	mov	x2, #BUF_SIZE
	sub	x2, x2, x10
	// syscall number
	mov	x8, #SYS_READ
	svc	#0

	// result in x0
	cmp	x0, #0
	beq	read_done              // EOF
	add	x10, x10, x0           // increase total
	b	read_loop

read_done:
	// -------------------------------------------------
	// Set up pointers for processing
	// x11 = start pointer (current line start)
	mov	x11, x9                // x11 = input_buf
	// x12 = end pointer = input_buf + total_len
	add	x12, x9, x10

process_loop:
	// If start >= end, no more lines
	cmp	x11, x12
	b.ge	finish                 // reached end

	// Scan for newline
	mov	x13, x11              // x13 = scan pointer
scan_newline:
	cmp	x13, x12
	b.eq	no_newline_found     // reached end without newline
	ldrb	w0, [x13]
	cmp	w0, #10                // '\n'
	b.eq	newline_found
	add	x13, x13, #1
	b	scan_newline

newline_found:
	// line_len = newline_ptr - line_start
	sub	x14, x13, x11          // x14 = length of line (may be 0)

	// Reverse copy line to output_buf
	bl	reverse_copy          // uses x11 (src), x14 (len), x20 (dst)

	// Write reversed line if length >0
	cmp	x14, #0
	beq	skip_write_line
	mov	x0, #FD_STDOUT
	mov	x1, x20
	mov	x2, x14
	mov	x8, #SYS_WRITE
	svc	#0
skip_write_line:
	// Write newline
	mov	x0, #FD_STDOUT
	adr	x1, nl_char
	mov	x2, #1
	mov	x8, #SYS_WRITE
	svc	#0

	// Advance start pointer past newline
	add	x11, x13, #1
	b	process_loop

no_newline_found:
	// Remaining bytes without trailing newline form final line
	sub	x14, x12, x11          // x14 = remaining length
	cmp	x14, #0
	beq	finish                 // no data left

	// Reverse copy final line
	bl	reverse_copy

	// Write reversed final line
	mov	x0, #FD_STDOUT
	mov	x1, x20
	mov	x2, x14
	mov	x8, #SYS_WRITE
	svc	#0

	// Write newline (always)
	mov	x0, #FD_STDOUT
	adr	x1, nl_char
	mov	x2, #1
	mov	x8, #SYS_WRITE
	svc	#0
	b	finish

// ---------------------------------------------------------------------
// reverse_copy:
//   Input:
//     x11 = src pointer (start of line)
//     x14 = length (bytes)
//     x20 = dst pointer (output buffer)
//   Output: dst filled with reversed bytes, dst pointer unchanged
//   Clobbers: x5-x9, w0-w2
reverse_copy:
	// If length == 0, nothing to do
	cmp	x14, #0
	b.eq	rev_done

	// src_end = src + len - 1
	add	x5, x11, x14
	sub	x5, x5, #1            // x5 = src_end

	// dst_ptr = dst (x20)
	mov	x6, x20

	// Loop copy
rev_loop:
	ldrb	w0, [x5]              // load byte from src_end
	strb	w0, [x6], #1          // store to dst, post-increment
	subs	x5, x5, #1            // src_end--
	subs	x14, x14, #1
	bne	rev_loop

rev_done:
	ret

// ---------------------------------------------------------------------
// Finish: exit the program
finish:
	mov	x0, #0
	mov	x8, #SYS_EXIT
	svc	#0

	// -----------------------------------------------------------------
	.section .rodata
	.align	2
nl_char:
	.byte	0x0A

	// -----------------------------------------------------------------
	.bss
	.align	3
input_buf:
	.space	BUF_SIZE
output_buf:
	.space	BUF_SIZE
