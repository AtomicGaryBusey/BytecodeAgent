.globl _start
.text
_start:
    # read loop
read_loop:
    mov     rax, 0                     # sys_read
    mov     rdi, 0                     # fd 0
    lea     rsi, [rip + buf]
    mov     rdx, buf_size
    syscall
    test    rax, rax
    jz      read_done
    mov     rcx, rax                   # bytes read
    add     qword ptr [rip + byte_cnt], rax
    xor     rbx, rbx                   # index = 0
process_buf:
    cmp     rbx, rcx
    jae     after_buf
    mov     al, byte ptr [rip + buf + rbx]
    # count newlines
    cmp     al, 0x0A
    jne     check_ws
    inc     qword ptr [rip + line_cnt]
check_ws:
    # whitespace detection
    cmp     al, 0x09
    je      is_ws
    cmp     al, 0x0A
    je      is_ws
    cmp     al, 0x0B
    je      is_ws
    cmp     al, 0x0C
    je      is_ws
    cmp     al, 0x0D
    je      is_ws
    cmp     al, 0x20
    je      is_ws
    # not whitespace
    mov     byte ptr [rip + in_word], 1
    jmp     next_char
is_ws:
    movzx   eax, byte ptr [rip + in_word]
    test    al, al
    je      next_char
    inc     qword ptr [rip + word_cnt]
    mov     byte ptr [rip + in_word], 0
next_char:
    inc     rbx
    jmp     process_buf
after_buf:
    jmp     read_loop
read_done:
    movzx   eax, byte ptr [rip + in_word]
    test    al, al
    je      prepare_output
    inc     qword ptr [rip + word_cnt]
prepare_output:
    lea     rdi, [rip + outbuf]        # destination pointer

    mov     rax, qword ptr [rip + line_cnt]
    call    num_to_ascii
    mov     byte ptr [rdi], ' '
    inc     rdi

    mov     rax, qword ptr [rip + word_cnt]
    call    num_to_ascii
    mov     byte ptr [rdi], ' '
    inc     rdi

    mov     rax, qword ptr [rip + byte_cnt]
    call    num_to_ascii
    mov     byte ptr [rdi], 0x0A
    inc     rdi

    lea     rdx, [rdi - outbuf]
    mov     rax, 1                     # sys_write
    mov     rdi, 1                     # fd 1
    lea     rsi, [rip + outbuf]
    syscall

    mov     rax, 60                    # sys_exit
    xor     rdi, rdi
    syscall

#------------------------------------------------------------
# Convert unsigned 64‑bit integer in rax to decimal ASCII.
# Writes digits to buffer pointed by rdi, updates rdi to point
# after the written digits.
num_to_ascii:
    lea     rbx, [rip + numbuf + 31]   # end of temporary buffer
    xor     rcx, rcx                   # digit count
    test    rax, rax
    jne     .conv_loop
    # zero case
    mov     byte ptr [rbx], '0'
    dec     rbx
    inc     rcx
    jmp     .conv_done
.conv_loop:
    xor     rdx, rdx
    mov     rsi, 10
    div     rsi                        # rax = quotient, rdx = remainder
    add     dl, '0'
    mov     byte ptr [rbx], dl
    dec     rbx
    inc     rcx
    test    rax, rax
    jne     .conv_loop
.conv_done:
    inc     rbx                        # first digit
    mov     rsi, rbx
    mov     rdx, rcx
    mov     rcx, rdx
    rep     movsb
    ret

#------------------------------------------------------------
.section .bss
    .align 8
buf:        .zero buf_size
    .align 8
line_cnt:   .quad 0
word_cnt:   .quad 0
byte_cnt:   .quad 0
in_word:    .byte 0
    .align 8
numbuf:     .zero 32
outbuf:     .zero 64

#------------------------------------------------------------
.section .rodata
buf_size = 65536
