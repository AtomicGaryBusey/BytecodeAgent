.globl _start
.text
_start:
    // initialize counters
    mov     x19, #0          // lines
    mov     x20, #0          // words
    mov     x21, #0          // bytes
    mov     w22, #1          // prev_is_space = true

    // buffer address
    adrp    x5, input_buf
    add     x5, x5, :lo12:input_buf

read_loop:
    // read(fd=0, buf=x5, count=65536)
    mov     x0, #0
    mov     x1, x5
    mov     w2, #65536
    mov     x8, #63
    svc     #0
    cbz     x0, read_done    // EOF

    mov     x6, x0           // bytes left in this chunk
    mov     x7, #0           // offset

proc_loop:
    ldrb    w8, [x5, x7]     // load byte
    add     x21, x21, #1    // total bytes

    // line count
    cmp     w8, #10
    bne     not_nl
    add     x19, x19, #1
not_nl:

    // whitespace test
    mov     w9, w8
    mov     w10, #0
    cmp     w9, #0x20
    beq     ws_set
    cmp     w9, #0x09
    beq     ws_set
    cmp     w9, #0x0A
    beq     ws_set
    cmp     w9, #0x0B
    beq     ws_set
    cmp     w9, #0x0C
    beq     ws_set
    cmp     w9, #0x0D
    beq     ws_set
    b       ws_check_done
ws_set:
    mov     w10, #1
ws_check_done:

    // word counting
    cmp     w10, #0
    bne     is_ws
    // not whitespace
    cmp     w22, #1
    bne     not_new_word
    add     x20, x20, #1    // new word
    mov     w22, #0
    b       after_ws
not_new_word:
    mov     w22, #0
    b       after_ws
is_ws:
    mov     w22, #1
after_ws:

    // advance offset
    add     x7, x7, #1
    subs    x6, x6, #1
    bne     proc_loop

    b       read_loop

read_done:
    // output buffer address
    adrp    x4, output_buf
    add     x4, x4, :lo12:output_buf
    mov     x5, x4            // dest pointer

    // write lines
    mov     x0, x19
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // space
    mov     w8, #' '
    strb    w8, [x5], #1

    // write words
    mov     x0, x20
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // space
    mov     w8, #' '
    strb    w8, [x5], #1

    // write bytes
    mov     x0, x21
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // newline
    mov     w8, #10
    strb    w8, [x5], #1

    // length = x5 - x4
    sub     x2, x5, x4

    // write(fd=1, buf=x4, count=x2)
    mov     x0, #1
    mov     x1, x4
    mov     x8, #64
    svc     #0

    // exit(0)
    mov     x0, #0
    mov     x8, #93
    svc     #0

// -------------------------------------------------------------
write_number:                // x0 = value, x1 = dest ptr, returns updated dest in x1
    adrp    x2, conv_buf
    add     x2, x2, :lo12:conv_buf
    add     x3, x2, #24      // end pointer

    cmp     x0, #0
    bne     wn_convert
    // value == 0
    mov     w4, #'0'
    strb    w4, [x1], #1
    ret

wn_convert:
    mov     x4, x3           // write pointer starts at end

wn_loop:
    mov     x5, #10
    udiv    x6, x0, x5       // quotient
    mul     x7, x6, x5
    sub     x8, x0, x7       // remainder
    add     w9, w8, #'0'     // digit char
    sub     x4, x4, #1
    strb    w9, [x4]
    mov     x0, x6
    cbnz    x0, wn_loop

wn_copy:
    cmp     x4, x3
    bge     wn_end_copy
    ldrb    w9, [x4], #1
    strb    w9, [x1], #1
    b       wn_copy
wn_end_copy:
    ret

// -------------------------------------------------------------
.bss
.align 8
input_buf:
    .space 65536
output_buf:
    .space 32
.align 8
conv_buf:
    .space 24
