.global _start
.text
_start:
    // initialise counters
    mov     x19, #0          // lines
    mov     x20, #0          // words
    mov     x21, #0          // bytes
    mov     x22, #1          // prev_is_space = true

    // buffer address
    adrp    x5, input_buf
    add     x5, x5, :lo12:input_buf

read_loop:
    // read(fd=0, buf=x5, count=4096)
    mov     x0, #0
    mov     x1, x5
    mov     w2, #4096
    mov     x8, #63
    svc     #0
    cbz     x0, read_done    // EOF

    mov     x6, x0          // bytes left in this chunk
    mov     x7, #0          // offset

process_byte:
    ldrb    w4, [x5, x7]
    add     x21, x21, #1   // total bytes

    cmp     w4, #10
    bne     not_nl
    add     x19, x19, #1   // line count
not_nl:
    // whitespace test
    mov     w8, w4
    cmp     w8, #0x20
    beq     is_ws
    cmp     w8, #0x09
    beq     is_ws
    cmp     w8, #0x0A
    beq     is_ws
    cmp     w8, #0x0B
    beq     is_ws
    cmp     w8, #0x0C
    beq     is_ws
    cmp     w8, #0x0D
    beq     is_ws
    // not whitespace
    tst     x22, #1
    beq     not_word_start
    add     x20, x20, #1   // new word
    mov     x22, #0
    b       after_ws
is_ws:
    mov     x22, #1
after_ws:
    add     x7, x7, #1
    subs    x6, x6, #1
    bne     process_byte

    b       read_loop

read_done:
    // output buffer address
    adrp    x4, output_buf
    add     x4, x4, :lo12:output_buf
    mov     x5, x4

    // write lines
    mov     x0, x19
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // space
    mov     w2, #' '
    strb    w2, [x5], #1

    // write words
    mov     x0, x20
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // space
    mov     w2, #' '
    strb    w2, [x5], #1

    // write bytes
    mov     x0, x21
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // newline
    mov     w2, #10
    strb    w2, [x5], #1

    // length = x5 - x4
    sub     x2, x5, x4

    // write(fd=1, buf=x4, count=x2)
    mov     x0, #1
    mov     x1, x4
    mov     x8, #64
    svc     #0

    // exit(0)
    mov     x0, #0
    mov     x8, #93
    svc     #0

// ------------------------------------------------------------
write_number:
    // x0 = value, x1 = dest pointer
    sub     sp, sp, #32          // allocate temporary space
    mov     x9, sp               // digit stack pointer
    mov     x10, #0              // digit count

    cmp     x0, #0
    bne     .wn_loop
    // value == 0
    mov     w2, #'0'
    strb    w2, [x1], #1
    b       .wn_done

.wn_loop:
    mov     w5, #10
    udiv    x2, x0, x5          // quotient
    mul     x6, x2, x5          // x6 = quotient * 10
    sub     x3, x0, x6          // remainder
    add     w4, w3, #'0'
    sub     sp, sp, #1
    strb    w4, [sp]
    add     x10, x10, #1
    mov     x0, x2
    cbnz    x0, .wn_loop

.wn_out:
    ldrb    w4, [sp], #1
    strb    w4, [x1], #1
    subs    x10, x10, #1
    bne     .wn_out

.wn_done:
    add     sp, sp, #32
    ret

.bss
.align 8
input_buf:
    .space 65536
output_buf:
    .space 32
