.global _start
.text
_start:
    // counters: lines (x19), words (x20), bytes (x21), prev_is_space (x22)
    mov     x19, #0
    mov     x20, #0
    mov     x21, #0
    mov     x22, #1

    // buffer address
    adrp    x5, input_buf
    add     x5, x5, :lo12:input_buf

read_loop:
    // read(fd=0, buf=x5, count=65536)
    mov     x0, #0
    mov     x1, x5
    mov     w2, #65536
    mov     x8, #63
    svc     #0
    cbz     x0, read_done          // EOF

    mov     x6, x0                 // bytes left in this chunk
    mov     x7, #0                 // offset

process_byte:
    ldrb    w4, [x5, x7]           // load byte
    add     x21, x21, #1          // total bytes

    // line count
    cmp     w4, #10
    bne     not_nl
    add     x19, x19, #1
not_nl:

    // whitespace test
    mov     w8, w4
    cmp     w8, #0x20
    beq     is_ws
    cmp     w8, #0x09
    beq     is_ws
    cmp     w8, #0x0A
    beq     is_ws
    cmp     w8, #0x0B
    beq     is_ws
    cmp     w8, #0x0C
    beq     is_ws
    cmp     w8, #0x0D
    beq     is_ws
    // not whitespace
    cmp     x22, #1
    bne     not_ws_continue
    // start of new word
    add     x20, x20, #1
    mov     x22, #0
    b       after_ws
not_ws_continue:
    // keep prev_is_space = 0
    b       after_ws
is_ws:
    mov     x22, #1
after_ws:
    add     x7, x7, #1
    subs    x6, x6, #1
    bne     process_byte

    b       read_loop

read_done:
    // output buffer address
    adrp    x4, output_buf
    add     x4, x4, :lo12:output_buf
    mov     x5, x4

    // write lines
    mov     x0, x19
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // space
    mov     w2, #' '
    strb    w2, [x5], #1

    // write words
    mov     x0, x20
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // space
    mov     w2, #' '
    strb    w2, [x5], #1

    // write bytes
    mov     x0, x21
    mov     x1, x5
    bl      write_number
    mov     x5, x1

    // newline
    mov     w2, #10
    strb    w2, [x5], #1

    // length = x5 - x4
    sub     x2, x5, x4

    // write(fd=1, buf=x4, count=x2)
    mov     x0, #1
    mov     x1, x4
    mov     x8, #64
    svc     #0

    // exit(0)
    mov     x0, #0
    mov     x8, #93
    svc     #0

// write_number: x0 = value, x1 = dest pointer, returns updated dest in x1
write_number:
    // temporary buffer for digits (conv_buf, 24 bytes)
    adrp    x2, conv_buf
    add     x2, x2, :lo12:conv_buf
    add     x9, x2, #24          // end pointer

    cmp     x0, #0
    bne     .wn_convert
    // value == 0
    mov     w3, #'0'
    strb    w3, [x1], #1
    ret

.wn_convert:
    mov     x3, x9               // write pointer starts at end

.wn_loop:
    mov     x4, #10
    udiv    x5, x0, x4           // quotient
    mul     x6, x5, x4           // quotient * 10
    sub     x7, x0, x6           // remainder
    add     w8, w7, #'0'         // digit character
    sub     x3, x3, #1
    strb    w8, [x3]
    mov     x0, x5
    cbnz    x0, .wn_loop

    // copy digits forward
.wn_copy:
    cmp     x3, x9
    bge     .wn_end_copy
    ldrb    w8, [x3], #1
    strb    w8, [x1], #1
    b       .wn_copy
.wn_end_copy:
    ret

.bss
.align 8
input_buf:
    .space 65536
output_buf:
    .space 32
.align 8
conv_buf:
    .space 24
