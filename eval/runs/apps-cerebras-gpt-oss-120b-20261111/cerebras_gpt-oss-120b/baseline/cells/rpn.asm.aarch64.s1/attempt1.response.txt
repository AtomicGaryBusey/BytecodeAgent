    .section .bss
    .align 8
input_buf:
    .zero 65536                // buffer for input
input_len:
    .quad 0                    // total bytes read
stack:
    .zero 8192                 // 1024 * 8 bytes for RPN stack
temp_buf:
    .zero 32                   // temporary buffer for number conversion

    .section .rodata
    .align 8
err_msg:
    .ascii "error\n"

    .section .text
    .global _start
_start:
    // -------------------------------------------------
    // Read entire input into input_buf (max 65536)
    // -------------------------------------------------
    mov     x0, #0                     // fd = stdin
    adrp    x1, input_buf
    add     x1, x1, :lo12:input_buf
    mov     x2, #65536
    mov     x8, #63                    // sys_read
read_loop:
    svc     #0
    cbz     x0, read_done              // EOF when read returns 0
    // accumulate length
    ldr     x3, =input_len
    ldr     x4, [x3]
    add     x4, x4, x0
    str     x4, [x3]
    b       read_loop
read_done:
    // -------------------------------------------------
    // If no input, exit successfully
    // -------------------------------------------------
    ldr     x13, =input_len
    ldr     x13, [x13]                 // total_len in x13
    cbz     x13, exit_success

    // -------------------------------------------------
    // Process each line
    // -------------------------------------------------
    mov     x14, #0                    // offset into input_buf

line_loop:
    cmp     x14, x13
    bge     exit_success               // finished all input

    // Find end of current line (newline or end of buffer)
    mov     x15, x14                    // scanning pointer
find_newline:
    cmp     x15, x13
    bge     line_end_no_nl
    adrp    x2, input_buf
    add     x2, x2, :lo12:input_buf
    add     x3, x2, x15
    ldrb    w4, [x3]
    cmp     w4, #0x0A
    beq     line_end_found
    add     x15, x15, #1
    b       find_newline
line_end_found:
    sub     x16, x15, x14               // line_len = newline - start
    mov     x0, x14                    // start offset
    mov     x1, x16                    // length
    bl      parse_line
    add     x14, x15, #1               // skip newline
    b       line_loop
line_end_no_nl:
    sub     x16, x13, x14              // remaining bytes as line
    mov     x0, x14
    mov     x1, x16
    bl      parse_line
    mov     x14, x13
    b       line_loop

exit_success:
    mov     x0, #0
    mov     x8, #93                    // sys_exit
    svc     #0

// -----------------------------------------------------
// parse_line(start_offset = x0, length = x1)
// -----------------------------------------------------
parse_line:
    // Save callee‑saved registers we use
    stp     x19, x20, [sp, #-16]!
    stp     x21, x22, [sp, #-16]!
    stp     x23, x24, [sp, #-16]!

    // Base address of input buffer
    adrp    x5, input_buf
    add     x5, x5, :lo12:input_buf

    // RPN stack index (0 = empty)
    mov     x19, #0

    // error flag = 0 (no error)
    mov     x20, #0

    // cursor within line
    mov     x6, #0

token_loop:
    // ----- skip separators (space, tab) -----
skip_sep:
    cmp     x6, x1
    bge     after_tokens
    add     x7, x5, x0
    add     x7, x7, x6
    ldrb    w8, [x7]
    cmp     w8, #0x20
    beq     inc_sep
    cmp     w8, #0x09
    beq     inc_sep
    b       after_sep
inc_sep:
    add     x6, x6, #1
    b       skip_sep
after_sep:
    cmp     x6, x1
    bge     after_tokens

    // token start = x6
    mov     x9, x6

    // ----- find token end -----
find_tok_end:
    cmp     x6, x1
    bge     tok_end
    add     x7, x5, x0
    add     x7, x7, x6
    ldrb    w8, [x7]
    cmp     w8, #0x20
    beq     tok_end
    cmp     w8, #0x09
    beq     tok_end
    add     x6, x6, #1
    b       find_tok_end
tok_end:
    sub     x10, x6, x9               // token_len in x10

    // If an error already occurred, ignore remaining tokens
    cmp     x20, #0
    bne     token_loop

    // Load first character of token
    add     x7, x5, x0
    add     x7, x7, x9
    ldrb    w11, [x7]

    // ----- operator (single char) -----
    cmp     x10, #1
    bne     maybe_literal
    cmp     w11, #'+'                 // '+'
    beq     op_plus
    cmp     w11, #'-'                 // '-'
    beq     op_minus
    cmp     w11, #'*'                 // '*'
    beq     op_mul
    cmp     w11, #'/'                 // '/'
    beq     op_div
    cmp     w11, #'%'                 // '%'
    beq     op_mod
    b       maybe_literal

// -----------------------------------------------------
// Operators
// -----------------------------------------------------
op_plus:
    cmp     x19, #2
    blt     set_error
    // b = stack[--sp]
    sub     x19, x19, #1
    ldr     x12, =stack
    add     x12, x12, x19, LSL #3
    ldr     x12, [x12]
    // a = stack[--sp]
    sub     x19, x19, #1
    ldr     x13, =stack
    add     x13, x13, x19, LSL #3
    ldr     x13, [x13]
    add     x15, x13, x12            // result
    // push result
    ldr     x14, =stack
    add     x14, x14, x19, LSL #3
    str     x15, [x14]
    add     x19, x19, #1
    b       token_loop

op_minus:
    cmp     x19, #2
    blt     set_error
    sub     x19, x19, #1
    ldr     x12, =stack
    add     x12, x12, x19, LSL #3
    ldr     x12, [x12]               // b
    sub     x19, x19, #1
    ldr     x13, =stack
    add     x13, x13, x19, LSL #3
    ldr     x13, [x13]               // a
    sub     x15, x13, x12
    ldr     x14, =stack
    add     x14, x14, x19, LSL #3
    str     x15, [x14]
    add     x19, x19, #1
    b       token_loop

op_mul:
    cmp     x19, #2
    blt     set_error
    sub     x19, x19, #1
    ldr     x12, =stack
    add     x12, x12, x19, LSL #3
    ldr     x12, [x12]               // b
    sub     x19, x19, #1
    ldr     x13, =stack
    add     x13, x13, x19, LSL #3
    ldr     x13, [x13]               // a
    mul     x15, x13, x12
    ldr     x14, =stack
    add     x14, x14, x19, LSL #3
    str     x15, [x14]
    add     x19, x19, #1
    b       token_loop

op_div:
    cmp     x19, #2
    blt     set_error
    sub     x19, x19, #1
    ldr     x12, =stack
    add     x12, x12, x19, LSL #3   // b
    ldr     x12, [x12]
    sub     x19, x19, #1
    ldr     x13, =stack
    add     x13, x13, x19, LSL #3   // a
    ldr     x13, [x13]
    // b == 0 ?
    cmp     x12, #0
    beq     set_error
    // overflow a == INT64_MIN && b == -1 ?
    mov     x15, #0x8000000000000000
    cmp     x13, x15
    bne     div_ok
    cmp     x12, #-1
    beq     set_error
div_ok:
    sdiv    x15, x13, x12           // quotient
    ldr     x14, =stack
    add     x14, x14, x19, LSL #3
    str     x15, [x14]
    add     x19, x19, #1
    b       token_loop

op_mod:
    cmp     x19, #2
    blt     set_error
    sub     x19, x19, #1
    ldr     x12, =stack
    add     x12, x12, x19, LSL #3   // b
    ldr     x12, [x12]
    sub     x19, x19, #1
    ldr     x13, =stack
    add     x13, x13, x19, LSL #3   // a
    ldr     x13, [x13]
    // b == 0 ?
    cmp     x12, #0
    beq     set_error
    // overflow a == INT64_MIN && b == -1 ?
    mov     x15, #0x8000000000000000
    cmp     x13, x15
    bne     mod_ok
    cmp     x12, #-1
    beq     set_error
mod_ok:
    sdiv    x14, x13, x12           // q = a / b
    mul     x14, x14, x12           // q*b
    sub     x15, x13, x14           // remainder
    ldr     x14, =stack
    add     x14, x14, x19, LSL #3
    str     x15, [x14]
    add     x19, x19, #1
    b       token_loop

// -----------------------------------------------------
// Literal parsing
// -----------------------------------------------------
maybe_literal:
    // Verify token length (1..18) and characters
    // Reject if length > 18
    cmp     x10, #19
    bhs     set_error
    // Optional leading '-'
    mov     x21, #0                 // sign flag 0 = positive
    mov     x22, x9                 // index within token
    cmp     w11, #'-'
    bne     lit_no_sign
    // token is just "-" ? then invalid (already not operator)
    cmp     x10, #1
    beq     set_error
    mov     x21, #1                 // negative
    add     x22, x22, #1           // skip sign
lit_no_sign:
    // Parse digits
    mov     x23, #0                 // accumulator
parse_digits:
    cmp     x22, x10
    beq     lit_done
    // load character at token[base+offset]
    add     x7, x5, x0
    add     x7, x7, x9
    add     x7, x7, x22
    ldrb    w8, [x7]
    // char must be '0'..'9'
    cmp     w8, #'0'
    blt     set_error
    cmp     w8, #'9'
    bgt     set_error
    // digit = w8 - '0'
    sub     w8, w8, #'0'
    // accumulator = accumulator*10 + digit
    mov     x24, #10
    mul     x23, x23, x24
    add     x23, x23, x8
    add     x22, x22, #1
    b       parse_digits
lit_done:
    // apply sign if needed
    cmp     x21, #0
    beq     lit_push
    neg     x23, x23
lit_push:
    // push onto RPN stack
    ldr     x14, =stack
    add     x14, x14, x19, LSL #3
    str     x23, [x14]
    add     x19, x19, #1
    b       token_loop

// -----------------------------------------------------
// Error handling
// -----------------------------------------------------
set_error:
    mov     x20, #1
    b       token_loop

after_tokens:
    // ----- output result or error -----
    cmp     x20, #0
    bne     output_error

    cmp     x19, #1
    bne     output_error

    // get result from stack[0]
    ldr     x0, =stack
    ldr     x1, [x0]

    bl      print_int64
    b       parse_done

output_error:
    // write "error\n"
    adrp    x0, err_msg
    add     x0, x0, :lo12:err_msg
    mov     x1, #6
    mov     x2, #1
    mov     x8, #64
    svc     #0

parse_done:
    // Restore callee‑saved registers and return
    ldp     x23, x24, [sp], #16
    ldp     x21, x22, [sp], #16
    ldp     x19, x20, [sp], #16
    ret

// -----------------------------------------------------
// print_int64: x0 = signed 64‑bit value
// Writes decimal representation followed by '\n'
// -----------------------------------------------------
print_int64:
    // Save registers we use
    stp     x19, x20, [sp, #-16]!
    stp     x21, x22, [sp, #-16]!

    // temp buffer base
    adrp    x19, temp_buf
    add     x19, x19, :lo12:temp_buf
    add     x20, x19, #32           // end pointer

    // Determine sign
    mov     x21, #0                 // sign flag
    cmp     x0, #0
    bge     pos_number
    mov     x21, #1
    neg     x0, x0                  // absolute value (as unsigned)
pos_number:
    // Special case 0
    cmp     x0, #0
    bne     conv_loop
    sub     x20, x20, #1
    mov     w1, #'0'
    strb    w1, [x20]
    b       after_digits

conv_loop:
    mov     x2, #10
conv_div:
    udiv    x3, x0, x2              // quotient
    mul     x4, x3, x2
    sub     x4, x0, x4              // remainder 0..9
    add     x4, x4, #'0'
    sub     x20, x20, #1
    strb    w4, [x20]
    mov     x0, x3
    cmp     x0, #0
    bne     conv_div

after_digits:
    cmp     x21, #0
    beq     no_sign
    sub     x20, x20, #1
    mov     w1, #'-'
    strb    w1, [x20]
no_sign:
    // length = (temp_buf+32) - x20
    add     x1, x19, #32
    sub     x2, x1, x20

    // write string
    mov     x0, #1
    mov     x1, x20
    mov     x8, #64
    svc     #0

    // Restore registers and return
    ldp     x21, x22, [sp], #16
    ldp     x19, x20, [sp], #16
    ret
