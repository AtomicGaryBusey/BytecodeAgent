	.global _start
	.text
_start:
	// zero counters
	mov	x4, #0          // lines
	mov	x5, #0          // bytes
	mov	x7, #0          // words

	// addresses of buffers and flag
	adr	x10, input_buf   // buffer address
	adr	x11, in_word     // in_word flag address

read_loop:
	// sys_read(fd=0, buf, 4096)
	mov	x0, #0
	mov	x1, x10
	mov	x2, #4096
	mov	x8, #63
	svc	#0
	cbnz	x0, have_input
	// x0 == 0 => EOF
	b	done_read

have_input:
	// x0 = bytes read
	mov	x12, x0          // length remaining
	mov	x13, x10         // pointer to current byte

proc_loop:
	ldrb	w3, [x13], #1   // load byte, post-increment pointer
	add	x5, x5, #1       // byte count++

	// newline?
	cmp	w3, #0x0a
	bne	.not_nl
	add	x4, x4, #1       // line count++
.not_nl:

	// whitespace check
	cmp	w3, #0x20
	beq	.is_ws
	cmp	w3, #0x09
	beq	.is_ws
	cmp	w3, #0x0a
	beq	.is_ws
	cmp	w3, #0x0b
	beq	.is_ws
	cmp	w3, #0x0c
	beq	.is_ws
	cmp	w3, #0x0d
	beq	.is_ws
	// non‑whitespace
	ldrb	w6, [x11]       // load in_word flag
	cbnz	w6, .continue   // already in word
	// start of new word
	add	x7, x7, #1       // word count++
	mov	w6, #1
	strb	w6, [x11]       // set in_word = 1
	b	.continue

.is_ws:
	// set in_word = 0
	mov	w6, #0
	strb	w6, [x11]

.continue:
	subs	x12, x12, #1
	bne	proc_loop

	b	read_loop

done_read:
	// output: lines space words space bytes newline
	// write lines
	mov	x0, x4
	bl	write_number

	// write space
	mov	x0, #1
	adr	x1, space_char
	mov	x2, #1
	mov	x8, #64
	svc	#0

	// write words
	mov	x0, x7
	bl	write_number

	// write space
	mov	x0, #1
	adr	x1, space_char
	mov	x2, #1
	mov	x8, #64
	svc	#0

	// write bytes
	mov	x0, x5
	bl	write_number

	// write newline
	mov	x0, #1
	adr	x1, newline_char
	mov	x2, #1
	mov	x8, #64
	svc	#0

	// exit
	mov	x0, #0
	mov	x8, #93
	svc	#0

// ------------------------------------------------------------
// write_number: x0 = unsigned 64‑bit number, writes its decimal
// representation to fd 1. Clobbers x0‑x3, x8, x9, x10‑x15.
write_number:
	adr	x14, out_buf          // start of buffer
	add	x15, x14, #24         // end of buffer
	mov	x9, #10               // divisor
	mov	x12, x0               // value
	mov	x16, x15              // write pointer (start at end)

	cmp	x12, #0
	bne	.wnum_loop
	// zero case: write single '0'
	sub	x16, x16, #1
	mov	w2, #'0'
	strb	w2, [x16]
	b	.wnum_done

.wnum_loop:
	udiv	x2, x12, x9          // quotient
	mul	x3, x2, x9
	sub	x3, x12, x3          // remainder
	add	x3, x3, #'0'
	sub	x16, x16, #1
	strb	w3, [x16]
	mov	x12, x2
	cbnz	x12, .wnum_loop

.wnum_done:
	// length = end - ptr
	sub	x2, x15, x16           // length
	mov	x0, #1                // fd
	mov	x1, x16               // buffer address
	mov	x8, #64
	svc	#0
	ret

	.section .bss
	.align 8
input_buf:	.space	4096
in_word:	.byte	0
	.align 8
out_buf:	.space	24

	.section .ro
