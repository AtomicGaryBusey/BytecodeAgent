.globl _start

.section .bss
buffer:
    .zero 65536
tmpbuf:
    .zero 65536

.section .rodata
newline_char:
    .byte 0x0A

.text
_start:
    # rbx = input buffer pointer
    lea rbx, [rip + buffer]
    xor rcx, rcx                # rcx = total_len

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    lea rsi, [rbx + rcx]        # buf = buffer + total_len
    mov rdx, 65536
    sub rdx, rcx                # count = remaining space
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                # total_len += bytes_read
    jmp read_loop

read_done:
    xor rdx, rdx                # rdx = src_offset

process_line:
    cmp rdx, rcx
    jae done_processing         # src_offset >= total_len

    # Find newline
    mov rsi, rdx                # rsi = idx
find_nl:
    cmp rsi, rcx
    jae no_newline_found
    mov al, byte ptr [rbx + rsi]
    cmp al, 0x0A
    je newline_found
    inc rsi
    jmp find_nl

newline_found:
    mov r8, rsi                 # r8 = line_end (index of newline)
    jmp have_line_end

no_newline_found:
    mov r8, rcx                 # r8 = line_end = total_len

have_line_end:
    # line_len = line_end - src_offset
    mov r9, r8
    sub r9, rdx                 # r9 = line_len
    test r9, r9
    jz write_newline            # empty line

    # rax = line_len - 1 (last index)
    lea rax, [r9 - 1]
    mov rsi, rbx
    add rsi, rdx                # rsi = buffer + src_offset
    lea rdi, [rip + tmpbuf]     # rdi = tmpbuf

rev_copy_loop:
    mov r10b, byte ptr [rsi + rax]
    mov byte ptr [rdi + rax], r10b
    dec rax
    jns rev_copy_loop           # continue while rax >= 0

write_line:
    # write reversed line
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = stdout
    lea rsi, [rip + tmpbuf]
    mov rdx, r9                 # count = line_len
    call write_all

write_newline:
    # write newline character
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall

    # advance src_offset past line and possible newline
    mov rax, r8                 # rax = line_end
    cmp r8, rcx
    je advance_done            # no newline at end of input
    inc rax                     # skip newline
advance_done:
    mov rdx, rax                # src_offset = new position
    jmp process_line

done_processing:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status = 0
    syscall

# ------------------------------------------------------------
# write_all: write all bytes from [rsi] of length rdx to fd rdi
# Clobbers: rax, rdx, rsi, rdi
write_all:
    test rdx, rdx
    jz .wa_done
    mov rax, 1                  # sys_write
    syscall
    test rax, rax
    jz .wa_done
    sub rdx, rax
    add rsi, rax
    jmp write_all
.wa_done:
    ret
