.globl _start

.section .bss
buffer:
    .zero 65536
outbuf:
    .zero 131072

.text
_start:
    # r8 = buffer base, r9 = outbuf base
    lea r8, [rip + buffer]
    lea r9, [rip + outbuf]

    xor r10, r10          # total_len = 0
read_loop:
    mov rax, 0            # sys_read
    mov rdi, 0            # fd = stdin
    lea rsi, [r8 + r10]   # buf = buffer + total_len
    mov rdx, 65536
    sub rdx, r10          # count = remaining space
    syscall
    cmp rax, 0
    je read_done
    add r10, rax          # total_len += bytes_read
    jmp read_loop
read_done:
    xor r11, r11          # src_offset = 0
    xor rdx, rdx          # out_len = 0

proc_loop:
    cmp r11, r10
    jae proc_done

    mov rax, r11          # line_start = src_offset

    mov rbx, r11          # rbx = scan index
find_nl:
    cmp rbx, r10
    jae no_nl             # reached end of input
    mov al, byte ptr [r8 + rbx]
    cmp al, 0x0A
    je have_nl
    inc rbx
    jmp find_nl

have_nl:
    # line_end = rbx (index of newline)
    mov rcx, rbx
    sub rcx, rax          # rcx = length
    test rcx, rcx
    jz after_rev
    dec rcx               # rcx = last index
    mov rsi, r8
    add rsi, rax          # rsi = buffer + line_start
rev_copy_loop:
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [r9 + rdx], al
    inc rdx
    dec rcx
    cmp rcx, 0
    jge rev_copy_loop
after_rev:
    mov byte ptr [r9 + rdx], 0x0A   # newline
    inc rdx
    inc rbx                         # skip newline
    mov r11, rbx
    jmp proc_loop

no_nl:
    # final line without trailing newline
    mov rcx, r10
    sub rcx, rax          # rcx = length
    test rcx, rcx
    jz after_rev2
    dec rcx
    mov rsi, r8
    add rsi, rax          # rsi = buffer + line_start
rev_copy2_loop:
    mov al, byte ptr [rsi + rcx]
    mov byte ptr [r9 + rdx], al
    inc rdx
    dec rcx
    cmp rcx, 0
    jge rev_copy2_loop
after_rev2:
    mov byte ptr [r9 + rdx], 0x0A   # newline
    inc rdx
    jmp proc_done

proc_done:
proc_done:
