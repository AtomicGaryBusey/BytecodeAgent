.globl _start

.equ SYS_READ,   0
.equ SYS_WRITE,  1
.equ SYS_EXIT,   60
.equ BUF_SIZE,   65536

.section .bss
buffer:
    .zero BUF_SIZE
revbuf:
    .zero BUF_SIZE

.section .rodata
newline_char:
    .byte 0x0A

.text
_start:
    lea rbx, [rip + buffer]          # rbx = input buffer pointer
    lea r12, [rip + revbuf]          # r12 = reversed buffer pointer
    xor rcx, rcx                     # rcx = total_len = 0

read_loop:
    mov rax, SYS_READ
    xor rdi, rdi                     # fd = 0 (stdin)
    lea rsi, [rbx + rcx]             # buf = buffer + total_len
    mov rdx, BUF_SIZE
    sub rdx, rcx                     # count = remaining space
    test rdx, rdx
    jz read_done                     # no space left (should not happen)
    syscall
    cmp rax, 0
    je read_done
    add rcx, rax                     # total_len += bytes_read
    jmp read_loop

read_done:
    xor rdx, rdx                     # rdx = src_offset = 0

process_loop:
    cmp rdx, rcx
    jae done_processing

    mov r8, rdx                      # r8 = idx scanner

find_nl:
    cmp r8, rcx
    jae no_newline_found
    mov al, [rbx + r8]
    cmp al, 0x0A
    je newline_found
    inc r8
    jmp find_nl

newline_found:
    jmp have_line_end

no_newline_found:
    # r8 == rcx (end of input)
have_line_end:
    mov r9, r8                      # r9 = line_end
    sub r9, rdx                     # r9 = line_len

    test r9, r9
    jz write_newline_only

    # reverse copy line_len bytes
    lea rsi, [rbx + rdx]            # rsi = start of line
    mov rax, r9
    dec rax                         # rax = src_idx = line_len-1
rev_copy_loop:
    mov al, [rsi + rax]             # load byte from source
    mov r10, r9
    dec r10
    sub r10, rax                    # dest_idx = line_len-1 - src_idx
    mov [r12 + r10], al             # store reversed byte
    dec rax
    jns rev_copy_loop

    # write reversed line
    mov rdi, 1                      # stdout
    mov rsi, r12
    mov rdx, r9
    call write_all

write_newline_only:
    mov rax, SYS_WRITE
    mov rdi, 1
    lea rsi, [rip + newline_char]
    mov rdx, 1
    syscall

    # advance src_offset past line and optional newline
    mov rax, r8                     # rax = line_end
    cmp r8, rcx
    je advance_done                 # no newline at EOF
    inc rax                         # skip newline
advance_done:
    mov rdx, rax                    # src_offset = new position
    jmp process_loop

done_processing:
    mov rax, SYS_EXIT
    xor rdi, rdi
    syscall

# ------------------------------------------------------------
# write_all: write all bytes from [rsi] of length rdx to fd rdi
# Clobbers: rax, rdx, rsi, rdi
write_all:
    test rdx, rdx
    jz .wa_ret
    mov rax, SYS_WRITE
    syscall
    test rax, rax
    jz .wa_ret
    sub rdx, rax
    add rsi, rax
    jmp write_all
.wa_ret:
    ret
