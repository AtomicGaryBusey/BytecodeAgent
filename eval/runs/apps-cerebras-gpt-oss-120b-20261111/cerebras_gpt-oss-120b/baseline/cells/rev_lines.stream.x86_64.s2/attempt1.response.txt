.globl _start

.section .bss
buffer:
    .zero 65536
temp_byte:
    .zero 1

.section .rodata
nl:
    .byte 0x0A

.text
_start:
    xor     rbx, rbx                # total bytes read = 0

read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + buffer + rbx]
