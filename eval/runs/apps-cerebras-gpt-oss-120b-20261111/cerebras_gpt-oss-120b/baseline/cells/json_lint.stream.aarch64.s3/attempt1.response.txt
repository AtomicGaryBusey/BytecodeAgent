	.globl _start
	.text
_start:
	// read all input into buf, count in x19 (size)
	mov	x19, #0                // total bytes read
	adrp	x20, buf
	add	x20, x20, :lo12:buf   // x20 = buf pointer

read_loop:
	mov	x0, #0                // fd = stdin
	mov	x1, x20               // buf
	add	x2, x19, #65536      // remaining space (but we will cap)
	sub	x2, x2, x19           // x2 = 65536 - total_read
	cmp	x2, #0
	beq	read_done
	// syscall read
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x19, x19, x0          // total += ret
	add	x20, x20, x0          // buf += ret
	b	read_loop

read_done:
	// output "error at 0\n"
	adrp	x0, err_msg
	add	x0, x0, :lo12:err_msg
	mov	x1, #13               // length of err_msg
	mov	x2, #1                // fd = stdout
	mov	x8, #64
	svc	#0
	// exit with status 1
	mov	x0, #1
	mov	x8, #93
	svc	#0

	// data
	.section .bss
	.align 3
buf:
	.zero	65536

	.section .rodata
	.align 3
err_msg:
	.string	"error at 0\n"
