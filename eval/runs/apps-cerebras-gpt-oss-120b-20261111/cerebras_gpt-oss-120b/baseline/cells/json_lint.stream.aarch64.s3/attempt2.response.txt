	.globl	_start
	.text
_start:
	// read all input into buf, total length -> x21
	adrp	x20, buf
	add	x20, x20, :lo12:buf          // x20 = buf base
	mov	x19, #0                      // total bytes read
read_loop:
	// compute remaining space
	mov	x1, x20                      // x1 = current write pointer (buf + x19)
	mov	x2, #65536
	sub	x2, x2, x19                  // x2 = remaining space
	cmp	x2, #0
	beq	read_done
	// syscall read(fd=0, buf=x1, count=x2)
	mov	x0, #0
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x19, x19, x0                // total += ret
	add	x20, x20, x0                // advance write pointer
	b	read_loop
read_done:
	// set pointers
	adrp	x20, buf
	add	x20, x20, :lo12:buf          // x20 = base pointer
	mov	x21, x19                    // length
	add	x24, x20, x21               // x24 = end pointer (base+len)
	mov	x22, x20                    // x22 = current pointer

	// parse document
	bl	skip_ws
	bl	parse_value
	bl	skip_ws
	cmp	x22, x24
	beq	ok_path
	// extra bytes after value
	b	error_path

ok_path:
	// write "ok\n"
	adrp	x0, msg_ok
	add	x0, x0, :lo12:msg_ok
	mov	x1, #3
	mov	x2, #1
	mov	x8, #64
	svc	#0
	// exit 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

error_path:
	// compute offset = x22 - x20
	sub	x5, x22, x20                // x5 = offset
	// write "error at "
	adrp	x0, msg_err_prefix
	add	x0, x0, :lo12:msg_err_prefix
	mov	x1, #9
	mov	x2, #1
	mov	x8, #64
	svc	#0
	// convert x5 to decimal into buffer
	adrp	x6, dec_buf
	add	x6, x6, :lo12:dec_buf       // x6 = buffer end (dec_buf + 12)
	mov	x7, x6                     // x7 = cursor (start from end)
	cmp	x5, #0
	bne	conv_loop
	// number is zero
	subs	x7, x7, #1
	mov	w8, #'0'
	strb	w8, [x7]
	b	conv_done
conv_loop:
	// divide x5 by 10
	udiv	x9, x5, #10
