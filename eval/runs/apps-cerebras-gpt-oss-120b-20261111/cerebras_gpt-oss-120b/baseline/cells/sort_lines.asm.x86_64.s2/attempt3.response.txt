.globl _start
.section .bss
.align 8
input_buf:   .skip 32768
lines_ptrs: .skip 8000            # 1000 * 8
lines_len:  .skip 8000            # 1000 * 8
total_len:   .quad 0
line_count:  .quad 0

.section .rodata
newline: .byte 0x0A

.section .text
_start:
    # r12 = address of input_buf
    lea r12, [rip + input_buf]
    # r13 = address of lines_ptrs
    lea r13, [rip + lines_ptrs]
    # r14 = address of lines_len
    lea r14, [rip + lines_len]

    # ---------------------------------------------------
    # read entire input (max 32768 bytes)
    xor rbx, rbx                # total bytes read
read_loop:
    mov rdi, 0                 # fd = stdin
    lea rsi, [r12 + rbx]       # buffer pointer
    mov rdx, 32768
    sub rdx, rbx               # remaining space
    test rdx, rdx
    jz read_done
    xor rax, rax               # sys_read = 0
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    jmp read_loop
read_done:
    mov [total_len], rbx

    # ---------------------------------------------------
    # split into lines
    xor rsi, rsi               # current offset i
    xor r9, r9                 # start offset of current line
    xor rcx, rcx               # line index
    mov rdx, [total_len]       # total length
parse_loop:
    cmp rsi, rdx
    jge parse_end
    mov al, byte ptr [r12 + rsi]
    cmp al, 0x0A
    jne not_newline
    # newline at rsi
    mov rax, rsi
    sub rax, r9                # length = i - start
    lea r8, [r12 + r9]         # pointer to line start
    mov [r13 + rcx*8], r8
    mov [r14 + rcx*8], rax
    inc rcx
    inc rsi                    # skip newline
    mov r9, rsi                # new line start
    jmp parse_loop
not_newline:
    inc rsi
    jmp parse_loop
parse_end:
    # final partial line (if any)
    cmp r9, rdx
    jae parse_done
    mov rax, rdx
    sub rax, r9
    lea r8, [r12 + r9]
    mov [r13 + rcx*8], r8
    mov [r14 + rcx*8], rax
    inc rcx
parse_done:
    mov [line_count], rcx

    # ---------------------------------------------------
    # selection sort lines
    mov rax, [line_count]      # n
    cmp rax, 1
    jl sort_done
    dec rax                    # rax = n-1 (last index)
    xor r11, r11               # i = 0
outer_sel:
    cmp r11, rax
    jge sort_done
    mov r10, r11               # min_idx = i
    mov r9, r11
    inc r9                     # j = i+1
inner_sel:
    mov rcx, [line_count]
    cmp r9, rcx
    jge after_inner_sel
    # load line j
    mov r15, [r13 + r9*8]      # ptr_j
    mov rdx, [r14 + r9*8]      # len_j
    # load line min
    mov rsi, [r13 + r10*8]     # ptr_min
    mov rdi, [r14 + r10*8]     # len_min
    xor ebp, ebp               # result = 0
cmp_loop:
    test rdx, rdx
    jz cmp_len_check
    test rdi, rdi
    jz cmp_len_check
    mov al, byte ptr [r15]
    mov bl, byte ptr [rsi]
    cmp al, bl
    jne cmp_diff
    inc r15
    inc rsi
    dec rdx
    dec rdi
    jmp cmp_loop
cmp_len_check:
    cmp rdx, rdi
    je cmp_eq
    jl cmp_a_less
    mov ebp, 1
    jmp cmp_done
cmp_a_less:
    mov ebp, -1
    jmp cmp_done
cmp_eq:
    xor ebp, ebp
    jmp cmp_done
cmp_diff:
    movzx eax, al
    movzx ecx, bl
    sub eax, ecx
    mov ebp, eax
cmp_done:
    cmp ebp, 0
    jge no_update_min
    mov r10, r9                # new min = j
no_update_min:
    inc r9
    jmp inner_sel
after_inner_sel:
    # swap if min != i
    cmp r10, r11
    je skip_swap
    # swap pointers
    mov rax, [r13 + r11*8]
    mov rdx, [r13 + r10*8]
    mov [r13 + r11*8], rdx
    mov [r13 + r10*8], rax
    # swap lengths
    mov rax, [r14 + r11*8]
    mov rdx, [r14 + r10*8]
    mov [r14 + r11*8], rdx
    mov [r14 + r10*8], rax
skip_swap:
    inc r11
    jmp outer_sel
sort_done:

    # ---------------------------------------------------
    # output sorted lines with newline after each
    xor rax, rax               # line index = 0
output_loop:
    mov rbx, [line_count]
    cmp rax, rbx
    jge output_done
    mov rsi, [r13 + rax*8]     # line pointer
    mov rcx, [r14 + rax*8]     # length
    # write line
    mov rdi, 1
    mov rdx, rcx
    mov rax, 1                 # sys_write
    syscall
    # write newline
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall
    inc rax
    jmp output_loop
output_done:
    mov rdi, 0
    mov rax, 60                # sys_exit
    syscall
