.global _start
.text
_start:
    xor     r12, r12                # total bytes read = 0
    lea     rbx, [rip + buf]       # rbx = base of buffer

read_loop:
    mov     rax, 0                 # sys_read
    mov     rdi, 0                 # fd = stdin
    lea     rsi, [rbx + r12]       # buf + offset
    mov     rdx, 65536
    sub     rdx, r12               # remaining space
    cmp     rdx, 0
    je      read_done
    syscall
    test    rax, rax
    je      read_done
    add     r12, rax               # increase total
    jmp     read_loop

read_done:
    xor     r13, r13               # i = 0

proc_loop:
    cmp     r13, r12
    jae     proc_done

    mov     r14, r13               # start = i

scan_loop:
    cmp     r13, r12
    jae     scan_end
    mov     al, byte [rbx + r13]
    cmp     al, 0x0A
    je      scan_end
    inc     r13
    jmp     scan_loop

scan_end:                           # line = [r14, r13)
    mov     r15, r13
    sub     r15, r14               # line_len = r13 - r14
    cmp     r15, 0
    je      write_nl

    mov     r8, r13
    dec     r8                     # r8 = last index of line

rev_loop:
    cmp     r8, r14
    jb      rev_done
    mov     al, byte [rbx + r8]
    mov     byte [outbyte], al
    mov     rax, 1                 # sys_write
    mov     rdi, 1                 # fd = stdout
    lea     rsi, [rip + outbyte]
    mov     rdx, 1
    syscall
    dec     r8
    jmp     rev_loop

rev_done:
write_nl:
    mov     al, 0x0A
    mov     byte [outbyte], al
    mov     rax, 1                 # sys_write
    mov     rdi, 1
    lea     rsi, [rip + outbyte]
    mov     rdx, 1
    syscall

    cmp     r13, r12
    jae     proc_done
    mov     al, byte [rbx + r13]
    cmp     al, 0x0A
    jne     proc_continue
    inc     r13

proc_continue:
    jmp     proc_loop

proc_done:
    mov     rax, 60                # sys_exit
    xor     rdi, rdi
    syscall

.section .bss
    .align 8
buf:
    .zero 65536
outbyte:
    .zero 1
