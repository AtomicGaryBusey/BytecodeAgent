.globl _start

.equ BUF_SIZE, 65536
.equ LINE_BUF_SIZE, 80

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
buffer:     .zero BUF_SIZE
len:        .quad 0
line_buf:   .zero LINE_BUF_SIZE

.text
_start:
    xor rbx, rbx                # total bytes read = 0

read_loop:
    mov rdi, 0                  # fd 0
    lea rsi, [rip + buffer]
    add rsi, rbx                # buffer + total
    mov rdx, BUF_SIZE
    sub rdx, rbx                # remaining space
    xor rax, rax                # sys_read
    syscall
    test rax, rax
    jz read_done
    add rbx, rax                # increase total
    cmp rbx, BUF_SIZE
    jae read_done
    jmp read_loop

read_done:
    mov [rip + len], rbx

    xor rcx, rcx                # offset = 0

row_loop:
    mov rax, [rip + len]
    cmp rcx, rax
    jae final_len_line

    mov rdx, rax
    sub rdx, rcx                # remaining bytes
    mov r10, rdx                # save remaining
    cmp rdx, 16
    jl set_n
    mov rdx, 16                 # n = 16
set_n:
    mov r10, rdx                # n in r10

    # write offset (8 hex digits) into line_buf[0..7]
    mov rax, rcx
    mov rbx, rax
    # nibble 7
    mov rdx, rbx
    shr rdx, 28
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 0], al
    # nibble 6
    mov rdx, rbx
    shr rdx, 24
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 1], al
    # nibble 5
    mov rdx, rbx
    shr rdx, 20
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 2], al
    # nibble 4
    mov rdx, rbx
    shr rdx, 16
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 3], al
    # nibble 3
    mov rdx, rbx
    shr rdx, 12
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 4], al
    # nibble 2
    mov rdx, rbx
    shr rdx, 8
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 5], al
    # nibble 1
    mov rdx, rbx
    shr rdx, 4
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 6], al
    # nibble 0
    mov rdx, rbx
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 7], al

    # two spaces after offset
    mov byte ptr [rip + line_buf + 8], ' '
    mov byte ptr [rip + line_buf + 9], ' '

    lea rdi, [rip + line_buf + 10]   # dest pointer for hex area
    lea rsi_src, [rip + buffer]
    add rsi_src, rcx                  # source pointer for input bytes
    xor r8, r8                        # i = 0

hex_loop:
    cmp r8, 16
    je after_hex

    cmp r8, r10
    jl write_byte_hex

    # write three spaces for missing byte
    mov byte ptr [rdi + 0], ' '
    mov byte ptr [rdi + 1], ' '
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    inc r8
    jmp check_extra_space

write_byte_hex:
    mov al, [rsi_src + r8]
    mov bl, al
    shr bl, 4
    and bl, 0xF
    mov dl, [rip + hex_digits + rbx]
    mov [rdi + 0], dl
    and al, 0xF
    mov dl, [rip + hex_digits + rax]
    mov [rdi + 1], dl
    mov byte ptr [rdi + 2], ' '
    add rdi, 3
    inc r8

check_extra_space:
    cmp r8, 8
    jne hex_loop
    mov byte ptr [rdi], ' '
    inc rdi
    jmp hex_loop

after_hex:
    # space before '|'
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    xor r8, r8                # ascii counter
ascii_loop:
    cmp r8, r10
    je after_ascii
    mov al, [rsi_src + r8]
    cmp al, 0x20
    jl dot_char
    cmp al, 0x7e
    jg dot_char
    mov [rdi], al
    inc rdi
    inc r8
    jmp ascii_loop
dot_char:
    mov byte ptr [rdi], '.'
    inc rdi
    inc r8
    jmp ascii_loop

after_ascii:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10      # newline
    inc rdi

    # write the line
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi                # length
    mov rdi, 1                  # stdout
    mov rax, 1                  # sys_write
    syscall

    add rcx, r10                # offset += n
    jmp row_loop

final_len_line:
    # print final length as 8 hex digits + newline
    mov rax, [rip + len]
    mov rbx, rax
    # nibble 7
    mov rdx, rbx
    shr rdx, 28
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 0], al
    # nibble 6
    mov rdx, rbx
    shr rdx, 24
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 1], al
    # nibble 5
    mov rdx, rbx
    shr rdx, 20
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 2], al
    # nibble 4
    mov rdx, rbx
    shr rdx, 16
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 3], al
    # nibble 3
    mov rdx, rbx
    shr rdx, 12
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 4], al
    # nibble 2
    mov rdx, rbx
    shr rdx, 8
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 5], al
    # nibble 1
    mov rdx, rbx
    shr rdx, 4
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 6], al
    # nibble 0
    mov rdx, rbx
    and dl, 0xF
    mov al, [rip + hex_digits + rdx]
    mov byte ptr [rip + line_buf + 7], al

    mov byte ptr [rip + line_buf + 8], 10   # newline

    lea rsi, [rip + line_buf]
    mov rdx, 9
    mov rdi, 1
    mov rax, 1
    syscall

    mov rdi, 0
    mov rax, 60
    syscall
