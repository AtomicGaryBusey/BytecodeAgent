.globl _start
.equ BUF_SIZE, 65536
.equ LINE_BUF_SIZE, 96

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.bss
buffer:     .zero BUF_SIZE
len_var:    .zero 8
line_buf:   .zero LINE_BUF_SIZE

.text
_start:
    # ----- read all input into buffer -----
    xor rbx, rbx                     # total bytes read = 0

read_loop:
    mov rdi, 0                       # fd = stdin
    lea rsi, [rip + buffer]
    add rsi, rbx                     # buffer + total
    mov rdx, BUF_SIZE
    sub rdx, rbx                     # remaining space
    xor rax, rax                     # sys_read = 0
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, BUF_SIZE
    jae read_done
    jmp read_loop

read_done:
    mov [rip + len_var], rbx

    # ----- dump rows -----
    xor rcx, rcx                     # offset = 0

row_loop:
    mov rax, [rip + len_var]
    cmp rcx, rax
    jae final_len_line               # if offset >= len, done

    mov rdx, rax
    sub rdx, rcx                     # remaining bytes
    mov r8, rdx                      # n = remaining
    cmp r8, 16
    jl have_n
    mov r8, 16
have_n:

    # ---- write offset (8 hex digits) into line_buf[0..7] ----
    lea rdi, [rip + line_buf]        # dest pointer
    mov rax, rcx                     # value to print
    xor r9, r9                       # digit index = 0
offset_digit_loop:
    mov rdx, rax
    shr rdx, 28
    and dl, 0xF
    mov dl, [rip + hex_digits + rdx]
    mov [rdi + r9], dl
    inc r9
    shl rax, 4
    cmp r9, 8
    jne offset_digit_loop

    # two spaces after offset
    mov byte ptr [rip + line_buf + 8], ' '
    mov byte ptr [rip + line_buf + 9], ' '

    # ---- hex area starts at line_buf+10 ----
    lea rdi, [rip + line_buf + 10]   # current write pointer
    lea rsi, [rip + buffer]
    add rsi, rcx                     # source pointer

    xor r9, r9                       # i = 0
hex_byte_loop:
    cmp r9, 16
    je after_hex
    cmp r9, r8
    jl hex_present
    # missing byte: three spaces
    mov byte ptr [rdi], ' '
    mov byte ptr [rdi+1], ' '
    mov byte ptr [rdi+2], ' '
    add rdi, 3
    inc r9
    jmp check_extra_space

hex_present:
    mov al, [rsi + r9]
    mov bl, al
    shr bl, 4
    and bl, 0xF
    mov dl, [rip + hex_digits + rbx]   # high nibble
    mov [rdi], dl
    and al, 0xF
    mov dl, [rip + hex_digits + rax]   # low nibble
    mov [rdi+1], dl
    mov byte ptr [rdi+2], ' '
    add rdi, 3
    inc r9

check_extra_space:
    cmp r9, 8
    jne hex_byte_loop
    mov byte ptr [rdi], ' '
    inc rdi
    jmp hex_byte_loop

after_hex:
    # space before ASCII dump
    mov byte ptr [rdi], ' '
    inc rdi
    mov byte ptr [rdi], '|'
    inc rdi

    # ---- ASCII column ----
    xor r9, r9                       # j = 0
ascii_loop:
    cmp r9, r8
    je after_ascii
    mov al, [rsi + r9]
    cmp al, 0x20
    jl dot_char
    cmp al, 0x7e
    jg dot_char
    mov [rdi], al
    inc rdi
    inc r9
    jmp ascii_loop
dot_char:
    mov byte ptr [rdi], '.'
    inc rdi
    inc r9
    jmp ascii_loop

after_ascii:
    mov byte ptr [rdi], '|'
    inc rdi
    mov byte ptr [rdi], 10          # newline
    inc rdi

    # ---- write the line ----
    lea rsi, [rip + line_buf]
    mov rdx, rdi
    sub rdx, rsi                     # length = rdi - line_buf
    mov rdi, 1                       # stdout
    mov rax, 1                       # sys_write
    syscall

    add rcx, r8
    jmp row_loop

final_len_line:
    # ---- write total length (8 hex digits) + newline ----
    lea rdi, [rip + line_buf]        # dest pointer
    mov rax, [rip + len_var]         # total length
    xor r9, r9                       # digit index = 0
len_digit_loop:
    mov rdx, rax
    shr rdx, 28
    and dl, 0xF
    mov dl, [rip + hex_digits + rdx]
    mov [rdi + r9], dl
    inc r9
    shl rax, 4
    cmp r9, 8
    jne len_digit_loop

    mov byte ptr [rip + line_buf + 8], 10   # newline

    lea rsi, [rip + line_buf]
    mov rdx, 9
    mov rdi, 1
    mov rax, 1
    syscall

    # ---- exit ----
    xor rdi, rdi                     # status 0
    mov rax, 60                      # sys_exit
    syscall
