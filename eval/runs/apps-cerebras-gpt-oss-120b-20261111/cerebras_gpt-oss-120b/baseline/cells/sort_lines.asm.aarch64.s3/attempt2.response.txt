.global _start
    .text
_start:
    // Load addresses of buffers
    adrp    x19, buf
    add     x19, x19, :lo12:buf          // x19 = buf base
    adrp    x22, line_offs
    add     x22, x22, :lo12:line_offs    // x22 = line_offs base
    adrp    x23, line_lens
    add     x23, x23, :lo12:line_lens    // x23 = line_lens base

    // ---------- read input ----------
    mov     x20, #0                      // total_len = 0
read_loop:
    mov     x0, #0                       // fd = stdin
    add     x1, x19, x20                 // buf + offset
    mov     x2, #32768
    sub     x2, x2, x20                  // remaining space
    mov     x8, #63                      // sys_read
    svc     #0
    cbz     x0, read_done                // EOF
    add     x20, x20, x0                 // total_len += bytes read
    b       read_loop
read_done:
    // x20 now holds total_len

    // ---------- parse lines ----------
    mov     x21, #0                      // line_count = 0
    mov     x24, #0                      // i = 0
parse_loop:
    cmp     x24, x20
    bge     parse_done
    mov     x25, x24                     // start = i
find_nl:
    cmp     x24, x20
    bge     after_nl                     // reached end without newline
    add     x0, x19, x24
    ldrb    w0, [x0]
    cmp     w0, #0x0A
    beq     nl_found
    add     x24, x24, #1
    b       find_nl
nl_found:
    sub     x26, x24, x25                // len = i - start
    // store start offset
    lsl     x0, x21, #3
    add     x0, x22, x0
    str     x25, [x0]
    // store length
    lsl     x0, x21, #3
    add     x0, x23, x0
    str     x26, [x0]
    add     x21, x21, #1                 // line_count++
    add     x24, x24, #1                 // skip newline
    b       parse_loop
after_nl:
    sub     x26, x24, x25                // len = i - start (final line)
    // store start offset
    lsl     x0, x21, #3
    add     x0, x22, x0
    str     x25, [x0]
    // store length
    lsl     x0, x21, #3
    add     x0, x23, x0
    str     x26, [x0]
    add     x21, x21, #1                 // line_count++
    b       parse_done
parse_done:

    // ---------- insertion sort ----------
    mov     x27, #1                      // i = 1
sort_outer:
    cmp     x27, x21
    bge     sort_done
    // load key offset
    lsl     x0, x27, #3
    add     x0, x22, x0
    ldr     x28, [x0]                    // key_offset
    // load key length
    lsl     x0, x27, #3
    add     x0, x23, x0
    ldr     x29, [x0]                    // key_len
    sub     x9, x27, #1                  // j = i - 1
sort_inner:
    cmp     x9, #0
    blt     insert_position
    // load offset_j
    lsl     x0, x9, #3
    add     x0, x22, x0
    ldr     x5, [x0]                     // offset_j
    // load len_j
    lsl     x0, x9, #3
    add     x0, x23, x0
    ldr     x6, [x0]                     // len_j
    // compare key (x28,x29) with line j (x5,x6)
    mov     x0, x28
    mov     x1, x29
    mov     x2, x5
    mov     x3, x6
    bl      compare
    cmp     w0, #0
    bge     insert_position
    // shift line j up to j+1
    add     x10, x9, #1                  // dest = j+1
    // store offset
    lsl     x11, x10, #3
    add     x11, x22, x11
    str     x5, [x11]
    // store length
    lsl     x12, x10, #3
    add     x12, x23, x12
    str     x6, [x12]
    sub     x9, x9, #1
    b       sort_inner
insert_position:
    add     x10, x9, #1                  // dest = j+1 (or 0 if j==-1)
    // store key offset
    lsl     x11, x10, #3
    add     x11, x22, x11
    str     x28, [x11]
    // store key length
    lsl     x12, x10, #3
    add     x12, x23, x12
    str     x29, [x12]
    add     x27, x27, #1
    b       sort_outer
sort_done:

    // ---------- output ----------
    mov     x24, #0                     // out_idx = 0
out_loop:
    cmp     x24, x21
    bge     out_done
    // load offset
    lsl     x0, x24, #3
    add     x0, x22, x0
    ldr     x5, [x0]                    // offset
    // load length
    lsl     x0, x24, #3
    add     x0, x23, x0
    ldr     x6, [x0]                    // len
    // write line bytes
    add     x0, x19, x5                // addr = buf + offset
    mov     x1, x0                     // buf ptr
    mov     x2, x6                     // count
write_loop:
    cbz     x2, write_nl
    mov     x0, #1                     // fd = stdout
    mov     x8, #64
    svc     #0
    // x0 = bytes_written
    cmp     x0, x2
    beq     write_nl
    sub     x2, x2, x0
    add     x1, x1, x0
    b       write_loop
write_nl:
    // write newline
    adrp    x0, nl
    add     x0, x0, :lo12:nl
    mov     x1, x0
    mov     x2, #1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    // ignore partial write (should be 1)
    add     x24, x24, #1
    b       out_loop
out_done:
    // exit
    mov     x0, #0
    mov     x8, #93
    svc     #0

// ----------------------------------------------------
    // compare two lines
    // x0 = offset_a, x1 = len_a, x2 = offset_b, x3 = len_b
    // returns w0 = -1 if a<b, 0 if equal, 1 if a>b
compare:
    mov     x4, #0                     // idx = 0
cmp_loop:
    cmp     x4, x1
    bge     cmp_len
    cmp     x4, x3
    bge     cmp_len
    // load byte a
    add     x5, x19, x0
    add     x5, x5, x4
    ldrb    w5, [x5]
    // load byte b
    add     x6, x19, x2
    add     x6, x6, x4
    ldrb    w6, [x6]
    cmp     w5, w6
    beq     inc_idx
    // unsigned compare
    b.lo    less_than
    // a > b
    mov     w0, #1
    ret
less_than:
    mov     w0, #-1
    ret
inc_idx:
    add     x4, x4, #1
    b       cmp_loop
cmp_len:
    cmp     x1, x3
    beq     equal_len
    blt     a_less
    // a > b
    mov     w0, #1
    ret
a_less:
    mov     w0, #-1
    ret
equal_len:
    mov     w0, #0
    ret

    .section .bss
    .align  8
buf:            .zero   32768
line_offs:      .zero   8000        // 1000 * 8
line_lens:      .zero   8000        // 1000 * 8

    .section .rodata
    .align 1
nl:             .byte   0x0A
