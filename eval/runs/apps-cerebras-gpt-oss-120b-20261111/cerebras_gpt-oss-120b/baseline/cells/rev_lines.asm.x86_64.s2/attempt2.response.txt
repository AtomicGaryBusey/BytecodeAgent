.globl _start
.equ BUF_SIZE, 65536

.section .bss
buf:
    .zero BUF_SIZE
outbuf:
    .zero BUF_SIZE

.section .data
nl:
    .byte 0x0A

.text
_start:
    xor rbx, rbx                 # total bytes read = 0

read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buf]        # buffer start
    add rsi, rbx                # current write position
    mov rdx, BUF_SIZE
    sub rdx, rbx                # remaining space
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    je process_input            # EOF
    add rbx, rax                # accumulate total read
    cmp rbx, BUF_SIZE
    jae process_input           # buffer full
    jmp read_loop

process_input:
    lea rsi, [rip + buf]        # rsi = buffer start
    mov rcx, rbx                # rcx = total length
    lea rdx, [rsi + rcx]        # rdx = buffer end (exclusive)
    mov r8, rsi                 # r8 = line start pointer

line_loop:
    cmp r8, rdx
    jae done                    # no more data

    mov r9, r8                  # r9 = scan pointer

scan_nl:
    cmp r9, rdx
    je final_line_no_nl         # reached end without newline
    mov al, byte ptr [r9]
    cmp al, 0x0A
    je line_with_nl
    inc r9
    jmp scan_nl

line_with_nl:
    # r8 = line start, r9 = newline position
    mov r10, r9
    sub r10, r8                 # r10 = line length
    test r10, r10
    je write_newline            # empty line

    # copy reversed line to outbuf
    lea rcx, [rip + outbuf]     # destination pointer
    mov r11, r9
    dec r11                     # source pointer = last byte of line
copy_rev:
    mov al, byte ptr [r11]
    mov byte ptr [rcx], al
    inc rcx
    dec r11
    dec r10
    jne copy_rev

    # write reversed line
    mov rdx, r9
    sub rdx, r8                 # length
    lea rsi, [rip + outbuf]
    mov rdi, 1                  # stdout
    mov rax, 1                  # sys_write
    syscall

write_newline:
    # write newline character
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    # advance to next line
    inc r9                      # skip newline
    mov r8, r9
    jmp line_loop

final_line_no_nl:
    # r8 = line start, rdx = buffer end
    mov r10, rdx
    sub r10, r8                 # r10 = line length
    test r10, r10
    je done                     # no data left

    # copy reversed final line to outbuf
    lea rcx, [rip + outbuf]     # destination pointer
    mov r11, rdx
    dec r11                     # source pointer = last byte of line
copy_rev_last:
    mov al, byte ptr [r11]
    mov byte ptr [rcx], al
    inc rcx
    dec r11
    dec r10
    jne copy_rev_last

    # write reversed final line
    mov rdx, rdx
    sub rdx, r8                 # length
    lea rsi, [rip + outbuf]
    mov rdi, 1
    mov rax, 1
    syscall

    # write newline
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    jmp done

done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
