.globl _start
.equ BUF_SIZE, 65536

.section .bss
buf:
    .zero BUF_SIZE
outbuf:
    .zero BUF_SIZE

.section .rodata
nl:
    .byte 0x0A

.text
_start:
    xor rbx, rbx                # total bytes read = 0

read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buf]        # buffer start
    add rsi, rbx                # current write position
    mov rdx, BUF_SIZE
    sub rdx, rbx                # remaining space
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    je after_read               # EOF
    add rbx, rax                # accumulated total read
    cmp rbx, BUF_SIZE
    jae after_read              # buffer full
    jmp read_loop

after_read:
    lea rsi, [rip + buf]        # rsi = buffer start
    mov r15, rsi                # r15 = buffer start (preserve)
    add r15, rbx                # r15 = buffer end (exclusive)

    mov r8, rsi                 # r8 = line start pointer

line_loop:
    cmp r8, r15
    jae done                    # no more data

    mov r9, r8                  # r9 = scan pointer

scan_nl:
    cmp r9, r15
    je final_line_no_nl         # reached end without newline
    mov al, byte ptr [r9]
    cmp al, 0x0A
    je line_with_nl
    inc r9
    jmp scan_nl

line_with_nl:
    mov r10, r9
    sub r10, r8                 # r10 = line length
    test r10, r10
    je write_nl_only           # empty line

    lea rcx, [rip + outbuf]     # rcx = dest pointer

    mov r11, r9
    dec r11                     # r11 = source pointer (last byte of line)

copy_rev:
    mov al, byte ptr [r11]
    mov byte ptr [rcx], al
    inc rcx
    dec r11
    dec r10
    jne copy_rev

    # write reversed line
    mov rdx, r9
    sub rdx, r8                 # length = r9 - r8
    lea rsi, [rip + outbuf]
    mov rdi, 1                  # stdout
    mov rax, 1                  # sys_write
    syscall

write_nl_only:
    # write newline for this line (empty or after reversed line)
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    # advance past newline
    inc r9
    mov r8, r9
    jmp line_loop

final_line_no_nl:
    mov r10, r15
    sub r10, r8                 # r10 = line length
    test r10, r10
    je done                     # empty final remainder -> nothing to output

    lea rcx, [rip + outbuf]

    mov r11, r15
    dec r11                     # source = last byte of final line

copy_rev_last:
    mov al, byte ptr [r11]
    mov byte ptr [rcx], al
    inc rcx
    dec r11
    dec r10
    jne copy_rev_last

    # write reversed final line
    mov rdx, r15
    sub rdx, r8                 # length = r15 - r8
    lea rsi, [rip + outbuf]
    mov rdi, 1
    mov rax, 1
    syscall

    # write final newline
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    jmp done

done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
