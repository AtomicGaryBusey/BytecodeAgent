.globl _start

.equ BUF_SIZE, 65536

.section .bss
inbuf:
    .zero BUF_SIZE
outbuf:
    .zero BUF_SIZE

.section .data
nl_char:
    .byte 0x0A

.text
_start:
    xor rbx, rbx                 # total bytes read = 0

read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + inbuf]      # buffer start
    add rsi, rbx                # current write position
    mov rdx, BUF_SIZE
    sub rdx, rbx                # remaining space
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    je process_input            # EOF
    add rbx, rax                # accumulate total read
    cmp rbx, BUF_SIZE
    jae process_input           # buffer full
    jmp read_loop

process_input:
    lea rsi, [rip + inbuf]      # rsi = buffer start
    mov rcx, rbx                # rcx = total length
    mov rdi, rsi                # rdi = current line start
    lea r10, [rsi + rcx]        # r10 = buffer end (exclusive)

next_line:
    cmp rdi, r10
    jae done                    # no more data

    mov rax, rdi                # rax = line start
    mov rbx, rdi                # rbx = scan pointer

find_nl:
    cmp rbx, r10
    je no_newline               # reached end without newline
    mov al, byte ptr [rbx]
    cmp al, 0x0A
    je found_newline
    inc rbx
    jmp find_nl

found_newline:
    mov r8, rbx                 # r8 = pointer to newline
    sub r8, rax                 # r8 = line length
    mov r9, r8                  # r9 = saved length
    test r8, r8
    je write_newline_only       # empty line

    lea rsi, [rip + outbuf]     # destination pointer
    lea rdx, [rax + r8 - 1]     # source pointer (last byte of line)

copy_rev:
    mov al, byte ptr [rdx]
    mov byte ptr [rsi], al
    inc rsi
    dec rdx
    dec r8
    jne copy_rev

    # write reversed line
    mov rdx, r9                 # length
    lea rsi, [rip + outbuf]
    mov rdi, 1                  # stdout
    mov rax, 1                  # sys_write
    syscall

write_newline_only:
    # write newline
    lea rsi, [rip + nl_char]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    inc rbx                     # skip the newline character
    mov rdi, rbx                # next line start
    jmp next_line

no_newline:
    mov r8, r10                 # r8 = buffer end
    sub r8, rax                 # r8 = line length (final line)
    mov r9, r8
    test r8, r8
    je done                     # no data left

    lea rsi, [rip + outbuf]
    lea rdx, [rax + r8 - 1]

copy_rev_last:
    mov al, byte ptr [rdx]
    mov byte ptr [rsi], al
    inc rsi
    dec rdx
    dec r8
    jne copy_rev_last

    # write reversed final line
    mov rdx, r9
    lea rsi, [rip + outbuf]
    mov rdi, 1
    mov rax, 1
    syscall

    # write newline
    lea rsi, [rip + nl_char]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall

    jmp done

done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall
