.globl _start
.text
_start:
    # -------------------------------------------------
    # Read entire stdin into input_buf (max 65536)
    # -------------------------------------------------
    xor     rbx, rbx                # total bytes read = 0
read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]  # buffer base
    add     rsi, rbx                # current write pointer
    mov     rdx, INPUT_BUF_SIZE
    sub     rdx, rbx                # remaining space
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                # accumulate
    cmp     rbx, INPUT_BUF_SIZE
    jae     read_done
    jmp     read_loop
read_done:
    # rbx = total bytes read
    # -------------------------------------------------
    # Set pointers for line processing
    # -------------------------------------------------
    lea     r12, [rip + input_buf]          # r12 = current pointer (start of line)
    lea     r13, [rip + input_buf]
    add     r13, rbx                        # r13 = end pointer (input_buf + total)

line_loop:
    cmp     r12, r13
    jae     exit_program

    # r14 = line start
    mov     r14, r12

    # Find end of line (newline or end of input)
    mov     r15, r12                # scanner pointer
find_nl:
    cmp     r15, r13
    jae     no_newline
    mov     al, byte ptr [r15]
    cmp     al, 0x0A
    je      have_newline
    inc     r15
    jmp     find_nl
have_newline:
    mov     r10, r15                # r10 = line end (exclusive)
    inc     r15                     # r15 = start of next line
    jmp     line_start
no_newline:
    mov     r10, r13                # line end = end of input
    mov     r15, r13                # next line start = end (will exit after this)
line_start:
    # -------------------------------------------------
    # Evaluate the line [r14, r10)
    # -------------------------------------------------
    xor     r9d, r9d                # r9 = stack size (count)
    xor     r11d, r11d              # r11 = error flag (0 = no error)
    lea     r14, [rip + stack]      # r14 = stack base address

    mov     rdx, r14                # rdx = token pointer (reuse rdx)

token_loop:
    # skip separators (space 0x20, tab 0x09)
    cmp     rdx, r10
    jae     token_done
    mov     al, byte ptr [rdx]
    cmp     al, 0x20
    je      inc_token_ptr
    cmp     al, 0x09
    je      inc_token_ptr
    jmp     token_start
inc_token_ptr:
    inc     rdx
    jmp     token_loop

token_start:
    # rdx points to start of token
    mov     rsi, rdx                # rsi = token start
    mov     rcx, rdx                # rcx = token end finder
find_token_end:
    cmp     rcx, r10
    jae     token_end_found
    mov     al, byte ptr [rcx]
    cmp     al, 0x20
    je      token_end_found
    cmp     al, 0x09
    je      token_end_found
    inc     rcx
    jmp     find_token_end
token_end_found:
    mov     rbx, rcx                # rbx = token end (exclusive)
    # -------------------------------------------------
    # Check if token is an operator (single char and one of + - * / %)
    # -------------------------------------------------
    mov     rax, rbx
    sub     rax, rsi                # rax = token length
    cmp     rax, 1
    jne     maybe_literal
    mov     al, byte ptr [rsi]
    cmp     al, '+'
    je      op_add
    cmp     al, '-'
    je      op_sub
    cmp     al, '*'
    je      op_mul
    cmp     al, '/'
    je      op_div
    cmp     al, '%'
    je      op_mod
    # not a recognized operator => error
    jmp     set_error

maybe_literal:
    # -------------------------------------------------
    # Parse integer literal
    # -------------------------------------------------
    xor     r5d, r5d                # r5 = sign flag (0 = positive)
    mov     rcx, rsi                # rc
