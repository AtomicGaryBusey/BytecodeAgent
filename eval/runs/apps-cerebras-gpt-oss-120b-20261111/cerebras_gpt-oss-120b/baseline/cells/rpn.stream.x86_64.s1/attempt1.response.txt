.globl _start
.text
_start:
    # read entire stdin into buffer
    xor     rax, rax                # syscall: read
    mov     rdi, 0                  # fd 0
    lea     rsi, [rip + input_buf]
    mov     rdx, INPUT_BUF_SIZE
read_loop:
    syscall
    cmp     rax, 0
    je      read_done
    add     rbx, rax                # rbx = total bytes read
    lea     rsi, [rip + input_buf + rbx]
    mov
