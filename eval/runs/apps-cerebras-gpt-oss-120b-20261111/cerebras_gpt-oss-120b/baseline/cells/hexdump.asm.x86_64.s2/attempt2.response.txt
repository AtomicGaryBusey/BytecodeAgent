.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
buffer:
    .space 65536
line_buf:
    .space 80

.text
_start:
    # Read all input into buffer
    lea rbx, [rip + buffer]          # rbx = write pointer
    xor rcx, rcx                     # rcx = total bytes read

read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    mov rsi, rbx                     # buf = current pointer
    mov rdx, 65536
    sub rdx, rcx                     # remaining space
    cmp rdx, 0
    jle read_done
    syscall
    test rax, rax
    je read_done
    add rcx, rax                     # total += bytes_read
    add rbx, rax                     # advance pointer
    jmp read_loop

read_done:
    # rcx = total bytes read
    # Process rows
    lea rsi, [rip + buffer]          # rsi = data pointer
    xor rdx, rdx                     # rdx = offset

process_loop:
    cmp rdx, rcx
    jae final_offset

    mov rax, rcx
    sub rax, rdx                     # remaining bytes = total - offset
    cmp rax, 16
    jae set16
    mov r8, rax                      # n = remaining (<16)
    jmp have_n
set16:
    mov r8, 16                       # n = 16
have_n:
    mov rdi, rsi                     # data pointer for row_dump
    mov r9, r8                       # save n
    mov rsi, r8                      # n argument
    call row_dump

    add rsi, r9                      # advance data pointer
    add rdx, r9                      # advance offset
    jmp process_loop

final_offset:
    # Print total offset line: 8 hex digits + newline
    mov rax, rcx                     # total
    lea rbx, [rip + line_buf]        # line buffer pointer
    lea r8, [rip + hex_digits]

    mov rcx, 8
offset_final_loop:
    mov r9d, eax                     # move low 32 bits for shift
    shr r9d, 28                      # get top nibble (bits 60-63)
    and r9d, 0xF
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rbx], al
    inc rbx
    shl rax, 4
    dec rcx
    jnz offset_final_loop

    mov byte ptr [rbx], 0x0a
    inc rbx

    mov rdx, 9                       # length
    mov rax, 1                       # sys_write
    mov rdi, 1                       # fd = stdout
    lea rsi, [rip + line_buf]
    syscall

    # Exit
    mov rax, 60                      # sys_exit
    xor rdi, rdi
    syscall

# -------------------------------------------------
# row_dump:
#   rdi = data pointer
#   rsi = n (0..16)
#   rdx = offset
# -------------------------------------------------
row_dump:
    lea rbx, [rip + line_buf]        # rbx = line buffer pointer
    mov rax, rdx                     # offset
    mov rcx, 8
    lea r8, [rip + hex_digits]

offset_loop:
    mov r9d, eax                     # copy for shifting
    shr r9d, 28                      # top nibble
    and r9d, 0xF
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rbx], al
    inc rbx
    shl rax, 4
    dec rcx
    jnz offset_loop

    # two spaces after offset
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx

    # hex bytes (16 slots)
    xor rcx, rcx                     # i = 0
hex_loop:
    cmp rcx, 16
    je hex_done

    cmp rcx, rsi
    jae hex_space                    # i >= n -> spaces

    # byte present
    mov al, byte ptr [rdi + rcx]
    movzx r9d, al
    shr r9d, 4
    and r9d, 0xF
    movzx r10d, al
    and r10d, 0xF

    mov al, byte ptr [r8 + r9]
    mov byte ptr [rbx], al
    inc rbx
    mov al, byte ptr [r8 + r10]
    mov byte ptr [rbx], al
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
    jmp after_hex

hex_space:
    # three spaces for empty slot
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x20
    inc rbx

after_hex:
    inc rcx
    cmp rcx, 8
    jne hex_loop
    # extra space after the eighth byte slot
    mov byte ptr [rbx], 0x20
    inc rbx
    jmp hex_loop

hex_done:
    # space and '|'
    mov byte ptr [rbx], 0x20
    inc rbx
    mov byte ptr [rbx], 0x7c      # '|'
    inc rbx

    # ASCII representation
    xor rcx, rcx                 # i = 0
ascii_loop:
    cmp rcx, rsi
    je ascii_done
    mov al, byte ptr [rdi + rcx]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7e
    jg ascii_dot
    mov byte ptr [rbx], al
    inc rbx
    jmp ascii_next
ascii_dot:
    mov byte ptr [rbx], 0x2e    # '.'
    inc rbx
ascii_next:
    inc rcx
    jmp ascii_loop

ascii_done:
    # closing '|', newline
    mov byte ptr [rbx], 0x7c
    inc rbx
    mov byte ptr [rbx], 0x0a
    inc rbx

    # length = 63 + n
    mov rax, rsi
    add rax, 63

    # write line
    mov rax, 1                 # sys_write
    mov rdi, 1                 # fd = stdout
    lea rsi, [rip + line_buf]
    mov rdx, rax
    syscall
    ret
