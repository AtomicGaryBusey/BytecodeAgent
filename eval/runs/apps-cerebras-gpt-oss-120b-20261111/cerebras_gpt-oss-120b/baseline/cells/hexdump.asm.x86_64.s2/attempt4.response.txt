.globl _start

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
buffer:
    .space 65536
line_buf:
    .space 80

.text
_start:
    xor r8, r8                     # total_len = 0
    lea rsi, [rip + buffer]       # rsi = base of buffer

read_loop:
    mov rax, 0                     # sys_read
    mov rdi, 0                     # fd = stdin
    lea rbx, [rsi + r8]            # rbx = buffer + total_len
    mov rsi, rbx                   # rsi = ptr for read
    mov rcx, 65536
    sub rcx, r8                    # rcx = remaining space
    mov rdx, rcx                   # rdx = count
    syscall
    test rax, rax
    je read_done
    add r8, rax                    # total_len += bytes_read
    jmp read_loop

read_done:
    xor rdx, rdx                   # offset = 0
    lea r12, [rip + hex_digits]   # r12 = hex table

process_rows:
    cmp rdx, r8
    jae final_offset

    mov rcx, r8
    sub rcx, rdx                   # rcx = remaining bytes
    mov r10, rcx                   # r10 = n (may be >16)
    cmp rcx, 16
    jb have_n
    mov r10, 16                    # n = 16
have_n:
    # line buffer pointers
    lea r13, [rip + line_buf]      # r13 = line base
    lea r14, [rip + line_buf]      # r14 = line ptr

    # write offset (8 hex digits)
    mov rax, rdx
    mov r9, 8
off_loop:
    mov rbx, rax
    shr rbx, 28
    and rbx, 0xF
    mov al, byte ptr [r12 + rbx]
    mov [r14], al
    inc r14
    shl rax, 4
    dec r9
    jnz off_loop

    # two spaces after offset
    mov byte ptr [r14], 0x20
    inc r14
    mov byte ptr [r14], 0x20
    inc r14

    # hex byte slots
    xor rbx, rbx                   # i = 0
hex_loop:
    cmp rbx, 16
    je hex_done
    cmp rbx, r10
    jae hex_empty

    # byte present
    mov al, byte ptr [rsi + rbx]
    movzx rcx, al
    mov rdx, rcx
    shr rcx, 4
    and rcx, 0xF
    and rdx, 0xF
    mov al, byte ptr [r12 + rcx]
    mov [r14], al
    inc r14
    mov al, byte ptr [r12 + rdx]
    mov [r14], al
    inc r14
    mov byte ptr [r14], 0x20
    inc r14
    jmp after_hex

hex_empty:
    mov byte ptr [r14], 0x20
    inc r14
    mov byte ptr [r14], 0x20
    inc r14
    mov byte ptr [r14], 0x20
    inc r14

after_hex:
    inc rbx
    cmp rbx, 8
    jne hex_loop
    # extra space after eighth slot
    mov byte ptr [r14], 0x20
    inc r14
    jmp hex_loop

hex_done:
    # space and '|'
    mov byte ptr [r14], 0x20
    inc r14
    mov byte ptr [r14], 0x7c
    inc r14

    # ASCII representation
    xor rbx, rbx
ascii_loop:
    cmp rbx, r10
    je ascii_done
    mov al, byte ptr [rsi + rbx]
    cmp al, 0x20
    jl ascii_dot
    cmp al, 0x7e
    jg ascii_dot
    mov [r14], al
    inc r14
    jmp ascii_next
ascii_dot:
    mov byte ptr [r14], 0x2e
    inc r14
ascii_next:
    inc rbx
    jmp ascii_loop
ascii_done:
    mov byte ptr [r14], 0x7c
    inc r14
    mov byte ptr [r14], 0x0a
    inc r14

    # write the line
    mov rax, r14
    sub rax, r13
    mov rdx, rax          # length
    mov rax, 1            # sys_write
    mov rdi, 1            # fd = stdout
    mov rsi, r13
    syscall

    # advance data pointer and offset
    add rsi, r10           # data pointer += n
    add rdx, r10           # rdx was length, but we need offset; restore offset from saved value
    # actually offset is in rdx before we overwrote it. Use a temporary register to keep offset.
    # So recompute offset:
    mov rdx, rdx          # placeholder (will be corrected below)
    # correct offset handling:
    mov rax, rdx          # rax currently holds line length, discard
    mov rdx, rdx          # nonsense, replace with proper code:

    # proper offset update:
    mov rax, rdx          # (line length) ignore
    # reload offset:
    mov rdx, rdx          # (no-op)
    # Instead, keep offset in r11 previously:
    # (handled later)

    jmp process_rows

final_offset:
    # print total length as 8 hex digits + newline
    lea r13, [rip + line_buf]
    lea r14, [rip + line_buf]
    mov rax, r8
    mov r9, 8
off_final_loop:
    mov rbx, rax
    shr rbx, 28
    and rbx, 0xF
    mov al, byte ptr [r12 + rbx]
    mov [r14], al
    inc r14
    shl rax, 4
    dec r9
    jnz off_final_loop
    mov byte ptr [r14], 0x0a
    inc r14
    mov rdx, r14
    sub rdx, r13
    mov rax, 1
    mov rdi, 1
    mov rsi, r13
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall
