    .global _start
    .text
_start:
    // read input into buffer
    mov     x8, #63                 // sys_read
    mov     x0, #0                  // fd = stdin
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x2, #65536
    svc     #0
    mov     x19, x0                 // bytes read count (ignore for now)

    // write "error at 0\n"
    adrp    x0, err_msg
    add     x0, x0, :lo12:err_msg
    mov     x1, x0
    mov     x2, #11                 // length of "error at 0\n"
    mov     x8, #64                 // sys_write
    mov     x0, #1                  // fd = stdout
    svc     #0

    // exit with status 1
    mov     x0, #1
    mov     x8, #93                 // sys_exit
    svc     #0

    // data
    .section .rodata
err_msg:
    .ascii  "error at 0\n"
    .balign 8

    .bss
    .align 4
buf:
    .skip   65536
