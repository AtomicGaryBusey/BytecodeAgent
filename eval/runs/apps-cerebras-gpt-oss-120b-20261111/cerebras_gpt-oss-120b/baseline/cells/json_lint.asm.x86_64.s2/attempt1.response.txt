.global _start
.text
_start:
    # read all input into buffer
    mov     rdi, 0                      # fd = stdin
    lea     rsi, [rip + input_buf]      # buffer address
    mov     rdx, input_buf_len          # max bytes
    mov     rax, 0                      # sys_read
.read_loop:
    syscall
    test    rax, rax
    je      .read_done                  # EOF
    add     rsi, rax                    # advance pointer
    sub     rdx, rax                    # decrease remaining space
    cmp     rdx, 0
    je      .read_done
    jmp     .read_loop
.read_done:
    # output "ok\n"
    mov     rdx, ok_len
    lea     rsi, [rip + ok_msg]
    mov     rdi, 1                      # fd = stdout
    mov     rax, 1                      # sys_write
    syscall
    # exit status 0
    xor     rdi, rdi
    mov     rax, 60                     # sys_exit
    syscall

.section .bss
    .align 8
input_buf:
    .zero   input_buf_len
input_buf_len = 65536

.section .rodata
ok_msg: .ascii "ok\n"
ok_len = . - ok_msg
