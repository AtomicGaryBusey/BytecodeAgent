        .global _start
        .text
        .align 2

BUF_SIZE = 65536

_start:
        // read all input into inbuf
        mov     x19, #0                 // x19 = total length (len)

read_loop:
        mov     x0, #0                 // fd = stdin
        adr     x1, inbuf
        add     x1, x1, x19            // buf = inbuf + len
        mov     x2, #BUF_SIZE
        sub     x2, x2, x19            // count = BUF_SIZE - len
        mov     x8, #63                // sys_read
        svc     #0
        cbz     x0, read_done          // if read returns 0 -> EOF
        add     x19, x19, x0           // len += nread
        b       read_loop

read_done:
        mov     x20, #0                // x20 = line_start
        mov     x21, #0                // x21 = i (current index)

proc_loop:
        cmp     x21, x19
        bge     final_check            // if i >= len, go to final check

        adr     x2, inbuf
        add     x2, x2, x21            // address of inbuf[i]
        ldrb    w3, [x2]               // w3 = byte
        cmp     w3, #0x0a
        bne     not_nl

        // newline found: line_len = i - line_start
        sub     x0, x21, x20           // x0 = line_len
        mov     x1, x20                // x1 = line_start
        bl      write_rev_line

        // advance past newline
        add     x21, x21, #1
        add     x20, x20, #1
        b       proc_loop

not_nl:
        add     x21, x21, #1
        b       proc_loop

final_check:
        cmp     x20, x19
        beq     exit_prog              // no remaining bytes

        // final line (no terminating newline)
        sub     x0, x19, x20           // x0 = line_len
        mov     x1, x20                // x1 = line_start
        bl      write_rev_line

exit_prog:
        mov     x0, #0
        mov     x8, #93                // sys_exit
        svc     #0

// ---------------------------------------------------------------------
// write_rev_line: x0 = line_len, x1 = line_start
// Reverses the line into outbuf, appends '\n', writes it.
// Clobbers x0‑x10 (caller‑saved), preserves callee‑saved regs.
write_rev_line:
        mov     x2, x0                 // x2 = line_len
        mov     x3, x1                 // x3 = line_start
        adr     x4, outbuf             // x4 = outbuf base
        adr     x5, inbuf              // x5 = inbuf base
        add     x5, x5, x3             // x5 = source pointer

        mov     x6, #0                 // counter = 0

copy_loop:
        cmp     x6, x2
        bge     copy_done
        // src_offset = line_len - 1 - counter
        sub     x7, x2, #1
        sub     x7, x7, x6
        add     x8, x5, x7
        ldrb    w9, [x8]
        add     x10, x4, x6
        strb    w9, [x10]
        add     x6, x6, #1
        b       copy_loop

copy_done:
        // store newline after reversed line
        add     x8, x4, x2
        mov     w9, #0x0a
        strb    w9, [x8]

        // write outbuf (line_len + 1)
        mov     x0, #1                 // fd = stdout
        mov     x1, x4                 // buf = outbuf
        add     x2, x2, #1             // count = line_len + 1
        mov     x8, #64                // sys_write
        svc     #0
        ret

        .section .bss
        .align 3
inbuf:
        .zero   BUF_SIZE
outbuf:
        .zero   BUF_SIZE+1
