        .global _start
        .text
        .align 2

BUF_SIZE = 65536

_start:
        // read entire input into inbuf
        adrp    x9, inbuf               // x9 = base of input buffer (page)
        add     x9, x9, :lo12:inbuf
        mov     x10, #0                 // pos = 0

read_loop:
        // x0 = fd = 0
        mov     x0, #0
        // x1 = buf + pos
        add     x1, x9, x10
        // x2 = remaining space = BUF_SIZE - pos
        mov     x2, #BUF_SIZE
        sub     x2, x2, x10
        // syscall read
        mov     x8, #63
        svc     #0
        // x0 = nread
        cbz     x0, read_done           // if 0 -> EOF
        // pos += nread
        add     x10, x10, x0
        b       read_loop

read_done:
        // x10 = total length (len)
        // set up processing indexes
        mov     x4, #0                  // line_start = 0
        mov     x5, #0                  // i = 0

proc_loop:
        cmp     x5, x10
        bge     final_check             // i >= len

        // load byte at inbuf + i
        add     x6, x9, x5
        ldrb    w7, [x6]
        cmp     w7, #0x0a               // newline?
        bne     not_nl

        // newline found
        sub     x8, x5, x4              // line_len = i - line_start
        bl      write_rev_line
        // advance past newline
        add     x5, x5, #1
        add     x4, x4, #1
        b       proc_loop

not_nl:
        add     x5, x5, #1
        b       proc_loop

final_check:
        cmp     x4, x10
        beq     exit_prog               // no remaining bytes
        // final line without terminating newline
        sub     x8, x10, x4             // line_len = len - line_start
        bl      write_rev_line
        b       exit_prog

// x8 = line_len, x4 = line_start
write_rev_line:
        // copy reversed line to outbuf and append newline, then write
        adrp    x9, outbuf
        add     x9, x9, :lo12:outbuf    // x9 = outbuf base
        adrp    x10, inbuf
        add     x10, x10, :lo12:inbuf   // x10 = inbuf base
        add     x10, x10, x4            // x10 = source start (inbuf + line_start)

        mov     x11, x8                 // x11 = line_len
        mov     x12, #0                 // counter = 0

copy_loop:
        cmp     x12, x11
        bge     copy_done
        // src_offset = line_len - 1 - counter
        sub     x13, x11, #1
        sub     x13, x13, x12
        add     x14, x10, x13
        ldrb    w15, [x14]
        // dest = outbuf + counter
        add     x16, x9, x12
        strb    w15, [x16]
        add     x12, x12, #1
        b       copy_loop

copy_done:
        // store newline after reversed line
        add     x17, x9, x11
        mov     w18, #0x0a
        strb    w18, [x17]

        // write outbuf (line_len + 1)
        add     x0, x11, #1            // count
        mov     x1, x9                  // buf
        mov     x2, x0                  // count
        mov     x0, #1                  // fd = stdout
        mov     x8, #64                 // sys_write
        svc     #0
        ret

exit_prog:
        mov     x0, #0
        mov     x8, #93                  // sys_exit
        svc     #0

        .section .bss
        .align 3
inbuf:
        .zero BUF_SIZE
outbuf:
        .zero BUF_SIZE+1
