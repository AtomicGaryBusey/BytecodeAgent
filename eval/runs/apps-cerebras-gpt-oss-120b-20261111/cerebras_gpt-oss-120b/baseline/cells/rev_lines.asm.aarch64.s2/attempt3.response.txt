.global _start
    .text
    .align 2

.equ BUF_SIZE, 65536
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // x19 = total bytes read
    mov     x19, #0
    adr     x20, inbuf          // x20 = base of input buffer

read_loop:
    mov     x0, #0              // fd = stdin
    add     x1, x20, x19        // buf = inbuf + len
    mov     x2, #BUF_SIZE
    sub     x2, x2, x19         // count = BUF_SIZE - len
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done
    add     x19, x19, x0        // len += nread
    b       read_loop

read_done:
    mov     x21, #0             // x21 = i (current index)
    mov     x22, #0             // x22 = line_start

proc_loop:
    cmp     x21, x19
    bge     final_check

    add     x0, x20, x21        // addr = inbuf + i
    ldrb    w1, [x0]            // w1 = byte
    cmp     w1, #0x0a
    bne     not_nl

    // newline found
    sub     x3, x21, x22        // x3 = line_len
    mov     x4, x22             // x4 = line_start
    bl      write_rev_line

    // advance past newline
    add     x21, x21, #1
    add     x22, x22, #1
    b       proc_loop

not_nl:
    add     x21, x21, #1
    b       proc_loop

final_check:
    cmp     x22, x19
    beq     exit_prog          // no remaining bytes

    // final line (no terminating newline)
    sub     x3, x19, x22        // x3 = line_len
    mov     x4, x22             // x4 = line_start
    bl      write_rev_line

exit_prog:
    mov     x0, #0
    mov     x8, #SYS_EXIT
    svc     #0

// ------------------------------------------------------------
// write_rev_line: x3 = line_len, x4 = line_start
// Reverses the line into outbuf, appends '\n', writes it.
// Caller-saved registers may be clobbered; callee-saved regs preserved.
write_rev_line:
    // src = inbuf + line_start
    add     x5, x20, x4         // x5 = source pointer
    // dest = outbuf
    adr     x6, outbuf          // x6 = dest pointer
    // i = 0
    mov     x7, #0

rev_copy_loop:
    cmp     x7, x3
    bge     rev_copy_done
    // src_offset = line_len - 1 - i
    sub     x8, x3, #1
    sub     x8, x8, x7
    add     x9, x5, x8
    ldrb    w10, [x9]
    add     x11, x6, x7
    strb    w10, [x11]
    add     x7, x7, #1
    b       rev_copy_loop

rev_copy_done:
    // write newline after reversed line
    add     x12, x6, x3
    mov     w13, #0x0a
    strb    w13, [x12]

    // write syscall: fd=1, buf=outbuf, count=line_len+1
    mov     x0, #1
    mov     x1, x6
    add     x2, x3, #1
    mov     x8, #SYS_WRITE
    svc     #0
    ret

    .section .bss
    .align 3
inbuf:
    .zero   BUF_SIZE
outbuf:
    .zero   BUF_SIZE+1
