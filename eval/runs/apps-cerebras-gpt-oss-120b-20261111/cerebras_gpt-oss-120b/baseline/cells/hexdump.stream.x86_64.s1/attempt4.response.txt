.globl _start

# System call numbers
.equ SYS_READ, 0
.equ SYS_WRITE, 1
.equ SYS_EXIT, 60

# File descriptors
.equ FD_STDIN, 0
.equ FD_STDOUT, 1

# Constants
.equ BUF_SIZE, 65536
.equ LINEBUF_SIZE, 80

.section .bss
    .align 16
buffer:
    .zero BUF_SIZE
linebuf:
    .zero LINEBUF_SIZE
total_len:
    .zero 8                # 8‑byte storage for the total length

.text
_start:
    # ----- read entire input into buffer -----
    xor     rcx, rcx                # rcx = current length = 0
    lea     rbx, [rip + buffer]    # rbx = buffer base

read_loop:
    mov     rax, SYS_READ
    mov     rdi, FD_STDIN
    lea     rsi, [rbx + rcx]       # destination = buffer + current length
    mov     rdx, BUF_SIZE
    sub     rdx, rcx                # remaining space
    syscall
    test    rax, rax
    je      after_read
    add     rcx, rax                # rcx += bytes read
    jmp     read_loop

after_read:
    mov     [rip + total_len], rcx  # store total length

    # ----- dump buffer in hex -----
    mov     rdx, rcx                # rdx = total_len
    xor     rcx, rcx                # rcx = offset (loop variable)
    lea     rbx, [rip + buffer]    # rbx = buffer base

dump_loop:
    cmp     rcx, rdx
    jae     dump_done               # all bytes processed

    # line_len = min(16, total_len - rcx)
    mov     r8, rdx
    sub     r8, rcx
    cmp     r8, 16
    jl      line_len_ok
    mov     r8, 16
line_len_ok:

    # rdi = linebuf start, rsi = write pointer
    lea     rdi, [rip + linebuf]
    mov     rsi, rdi

    # ---- write offset (8 hex digits) ----
    mov     rax, rcx                # value = current offset
    mov     r9, 8
offset_digit_loop:
    mov     r10, rax
    and     r10, 0xF
    cmp     r10, 9
    jbe     offset_digit0
    add     r10, ('a' - 10)
    jmp     offset_store
offset_digit0:
    add     r10, '0'
offset_store:
    mov     byte ptr [rdi + 8 - r9], r10b
    shr     rax, 4
    dec     r9
