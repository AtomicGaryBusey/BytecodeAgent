.globl _start
.equ SYS_READ,   0
.equ SYS_WRITE,  1
.equ SYS_EXIT,   60
.equ STDIN_FD,   0
.equ STDOUT_FD,  1
.equ BUF_SIZE,   65536

.section .bss
    .align 8
buf:        .space BUF_SIZE
lines_cnt:  .zero 8
words_cnt:  .zero 8
bytes_cnt:  .zero 8

.section .rodata
space_char: .byte 0x20
newline_char: .byte 0x0A

.text
_start:
    xor rdi, rdi            # lines = 0
    xor rsi, rsi            # words = 0
    xor rdx, rdx            # bytes = 0
    xor r8, r8              # in_word = 0

read_loop:
    mov eax, SYS_READ
    mov edi, STDIN_FD
    lea rsi, [rip + buf]
    mov edx, BUF_SIZE
    syscall
    test rax, rax
    jz  finish_counts       # EOF
    mov rcx, rax            # rcx = bytes read
    lea rbx, [rip + buf]    # rbx = ptr

proc_loop:
    mov al, byte ptr [rbx]
    inc rdx                 # total bytes++
    cmp al, 0x0A
    jne not_newline
    inc rdi                 # lines++
not_newline:
    ; whitespace test
