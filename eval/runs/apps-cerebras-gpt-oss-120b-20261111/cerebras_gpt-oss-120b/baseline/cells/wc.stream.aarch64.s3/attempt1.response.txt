	.global _start
	.text
_start:
	// initialize counters
	mov	x19, #0          // line count
	mov	x20, #0          // word count
	mov	x21, #0          // byte count
	mov	w22, #1          // prev_is_whitespace = true

read_loop:
	// sys_read(0, buf, BUF_SIZE)
	mov	x0, #0
	adr	x1, buf
	mov	x2, #65536
	mov	x8, #63
	svc	#0
	cbnz	x0, process_chunk
	// x0 == 0 => EOF
	b	done_output

process_chunk:
	// x0 = nread
	mov	x24, x0          // n = bytes read
	mov	x25, #0          // offset = 0

process_byte:
	cmp	x25, x24
	bge	read_loop        // processed all bytes in this chunk

	ldrb	w6, [x1, x25]    // load byte

	// increment byte count
	add	x21, x21, #1

	// check for newline
	cmp	w6, #10
	bne	check_ws
	add	x19, x19, #1     // line count++
	mov	w7, #1           // is_whitespace = true
	b	ws_done

check_ws:
	// whitespace check: space(32), tab(9), VT(11), FF(12), CR(13)
	cmp	w6, #32
	beq	is_ws
	cmp	w6, #9
	beq	is_ws
	cmp	w6, #11
	beq	is_ws
	cmp	w6, #12
	beq	is_ws
	cmp	w6, #13
	beq	is_ws
	mov	w7, #0           // not whitespace
	b	ws_done
is_ws:
	mov	w7, #1           // whitespace

ws_done:
	// word start detection: non-whitespace && prev_whitespace
	cmp	w7, #0
	bne	not_word_start   // if whitespace, skip
	cmp	w22, #1
	bne	not_word_start
	add	x20, x20, #1     // word count++

not_word_start:
	// update prev_is_whitespace
	mov	w22, w7

	// next byte
	add	x25, x25, #1
	b	process_byte

done_output:
	// Build output string "<lines> <words> <bytes>\n"
	adr	x26, outbuf        // x26 = out_ptr

	// ----- lines -----
	adr	x5, numbuf
	add	x5, x5, #22          // x5 = end of temp buffer
	mov	x0, x19
	bl	convert_number

	// write space
	mov	w8, #' '
	strb	w8, [x26], #1

	// ----- words -----
	adr	x5, numbuf
	add	x5, x5, #22
	mov	x0, x20
	bl	convert_number

	// write space
	mov	w8, #' '
	strb	w8, [x26], #1

	// ----- bytes -----
	adr	x5, numbuf
	add	x5, x5, #22
	mov	x0, x21
	bl	convert_number

	// write newline
	mov	w8, #10
	strb	w8, [x26], #1

	// write output
	adr	x0, outbuf
	adr	x1, outbuf
	sub	x2, x26, x1          // length = out_ptr - outbuf
	mov	x8, #64
	svc	#0

	// exit(0)
	mov	x0, #0
	mov	x8, #93
	svc	#0

// ------------------------------------------------------------
// Convert unsigned integer in x0 to decimal ASCII.
// Uses x5 as end pointer of temporary buffer (size 22 bytes).
// Writes result to buffer pointed by x26, advancing x26.
// Preserves x5, x26, and returns via bl (no registers needed).
// ------------------------------------------------------------
convert_number:
	// x0 = value, x5 = temp_end, x26 = out_ptr
	mov	x6, #10                // divisor
	mov	x7, #0                // digit length

	cmp	x0, #0
	bne	conv_loop_start
	// value is zero -> output '0'
	mov	w8, #'0'
	strb	w8, [x26], #1
	b	conv_done

conv_loop_start:
	// loop: while x0 != 0
	udiv	x9, x0, x6           // quotient
	mul	x10, x9, x6
	sub	x11, x0, x10          // remainder
	add	x5, x5, #-1           // move back
	add	w12, w11, #'0'
	strb	w12, [x5]            // store digit
	add	x7, x7, #1           // length++
	mov	x0, x9
	cmp	x0, #0
	bne	conv_loop_start

	// copy stored digits (reverse order) to output buffer
	mov	x8, #0                // copy index
conv_copy:
	cmp	x8, x7
	bge	conv_done
	ldrb	w9, [x5], #1
	strb	w9, [x26], #1
	add	x8, x8, #1
	b	conv_copy

conv_done:
	ret

	.section .bss
	.align 8
buf:	.zero	65536
numbuf:	.space	22
outbuf:	.space	64
