	.global _start
	.text
_start:
	// counters: lines (x19), words (x20), bytes (x21), prev_is_ws (x22)
	mov	x19, #0
	mov	x20, #0
	mov	x21, #0
	mov	w22, #1          // prev_is_ws = true

read_loop:
	// sys_read(0, buf, 65536)
	mov	x0, #0
	adr	x1, buf
	mov	x2, #65536
	mov	x8, #63
	svc	#0
	cbnz	x0, process_chunk
	// x0 == 0 => EOF
	b	done_output

process_chunk:
	// x0 = nread
	mov	x24, x0          // n = bytes read
	mov	x25, #0          // offset = 0

process_byte:
	cmp	x25, x24
	bge	read_loop        // all bytes processed

	ldrb	w6, [x1, x25]    // load byte

	// byte count
	add	x21, x21, #1

	// line count
	cmp	w6, #10
	bne	check_ws
	add	x19, x19, #1
	mov	w7, #1           // is_ws = true
	b	ws_done

check_ws:
	// whitespace test
	cmp	w6, #32
	beq	is_ws
	cmp	w6, #9
	beq	is_ws
	cmp	w6, #11
	beq	is_ws
	cmp	w6, #12
	beq	is_ws
	cmp	w6, #13
	beq	is_ws
	mov	w7, #0           // not whitespace
	b	ws_done
is_ws:
	mov	w7, #1           // whitespace
ws_done:
	// word start detection
	cmp	w7, #0
	bne	not_word_start
	cmp	w22, #1
	bne	not_word_start
	add	x20, x20, #1     // word count++
not_word_start:
	mov	w22, w7          // prev_is_ws = is_ws

	// next byte
	add	x25, x25, #1
	b	process_byte

done_output:
	// output buffer pointer
	adr	x26, outbuf        // x26 = out_ptr

	// ----- lines -----
	adr	x5, numbuf
	add	x5, x5, #22          // x5 = temp_end
	mov	x0, x19
	bl	convert_number

	// space
	mov	w8, #' '
	strb	w8, [x26], #1

	// ----- words -----
	adr	x5, numbuf
	add	x5, x5, #22
	mov	x0, x20
	bl	convert_number

	// space
	mov	w8, #' '
	strb	w8, [x26], #1

	// ----- bytes -----
	adr	x5, numbuf
	add	x5, x5, #22
	mov	x0, x21
	bl	convert_number

	// newline
	mov	w8, #10
	strb	w8, [x26], #1

	// write output
	mov	x0, #1            // fd = stdout
	adr	x1, outbuf       // buf
	sub	x2, x26, x1      // count = out_ptr - outbuf
	mov	x8, #64
	svc	#0

	// exit(0)
	mov	x0, #0
	mov	x8, #93
	svc	#0

// ------------------------------------------------------------
// Convert unsigned integer in x0 to decimal ASCII.
// Uses x5 as end pointer of temporary buffer (size 22 bytes).
// Writes result to buffer pointed by x26, advancing x26.
// Preserves x5, x26, returns via bl.
// ------------------------------------------------------------
convert_number:
	// x0 = value, x5 = temp_end, x26 = out_ptr
	mov	x6, #10                // divisor
	mov	x7, #0                // digit length

	cmp	x0, #0
	bne	conv_loop_start
	// value is zero -> output '0'
	mov	w8, #'0'
	strb	w8, [x26], #1
	b	conv_done

conv_loop_start:
	udiv	x9, x0, x6           // x9 = quotient
	mul	x10, x9, x6
	sub	x11, x0, x10          // x11 = remainder
	add	x5, x5, #-1           // move back
	add	w12, w11, #'0'
	strb	w12, [x5]            // store digit
	add	x7, x7, #1           // length++
	mov	x0, x9
	cmp	x0, #0
	bne	conv_loop_start

	// copy stored digits to output buffer
	mov	x8, #0                // copy index
conv_copy:
	cmp	x8, x7
	bge	conv_done
	ldrb	w9, [x5], #1
	strb	w9, [x26], #1
	add	x8, x8, #1
	b	conv_copy

conv_done:
	ret

	.section .bss
	.align 8
buf:	.zero	65536
numbuf:	.space	22
outbuf:	.space	64
