	.global	_start
	.text

	.equ	SYSCALL_READ,	63
	.equ	SYSCALL_WRITE,	64
	.equ	SYSCALL_EXIT,	93

_start:
	// x10 = total length
	mov	x10, #0

read_loop:
	// read up to 65536 bytes into input_buf
	mov	x0, #0				// fd = stdin
	adrp	x1, input_buf
	add	x1, x1, :lo12:input_buf	// buf
	mov	x2, #65536			// count
	mov	x8, #SYSCALL_READ
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x10, x10, x0			// total += bytes_read
	b	read_loop

read_done:
	// if total == 0, exit silently
	cmp	x10, #0
	bne	encode_start
	mov	x0, #0
	mov	x8, #SYSCALL_EXIT
	svc	#0

encode_start:
	// x11 = input pointer, x14 = remaining, x13 = line_len
	adrp	x11, input_buf
	add	x11, x11, :lo12:input_buf
	mov	x14, x10
	mov	x13, #0
	adrp	x15, b64_table
	add	x15, x15, :lo12:b64_table

encode_main:
	cmp	x14, #3
	bge	encode_triplet
	cmp	x14, #2
	beq	encode_two
	cmp	x14, #1
	beq	encode_one
	b	encode_finish

// --------- encode a triplet (3 bytes) ----------
encode_triplet:
	ldrb	w4, [x11], #1		// b0
	ldrb	w5, [x11], #1		// b1
	ldrb	w6, [x11], #1		// b2
	subs	x14, x14, #3

	// idx0 = b0 >> 2
	lsr	w7, w4, #2

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	w8, w4, #3
	lsl	w8, w8, #4
	lsr	w9, w5, #4
	orr	w8, w8, w9

	// idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	and	w9, w5, #0xF
	lsl	w9, w9, #2
	lsr	w10, w6, #6
	orr	w9, w9, w10

	// idx3 = b2 & 0x3F
	and	w10, w6, #0x3F

	// output idx0
	ldrb	w0, [x15, w7]
