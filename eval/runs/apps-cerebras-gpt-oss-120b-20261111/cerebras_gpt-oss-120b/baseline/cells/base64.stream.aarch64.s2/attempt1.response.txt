	.globl	_start
	.text
_start:
	// read entire input (max 65536 bytes) into input_buf
	adrp	x9, input_buf
	add	x9, x9, :lo12:input_buf
	mov	x10, #0			// total length = 0

read_loop:
	mov	x0, #0			// fd = stdin
	mov	x1, x9			// buf = input_buf
	mov	x2, #65536		// count
	mov	x8, #63			// sys_read
	svc	#0
	cbnz	x0, have_read
	b	read_done

have_read:
	add	x10, x10, x0		// total += bytes_read
	b	read_loop

read_done:
	// if total length == 0, exit silently
	cbnz	x10, encode_start
	// exit 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

encode_start:
	// set pointers
	adrp	x11, input_buf
	add	x11, x11, :lo12:input_buf	// x11 = in_ptr
	adrp	x12, output_buf
	add	x12, x12, :lo12:output_buf	// x12 = out_ptr
	mov	x13, #0			// line_len = 0
	mov	x14, x10		// remaining = total_len
	adrp	x15, b64_table
	add	x15, x15, :lo12:b64_table	// x15 = base64 table address

encode_main:
	cmp	x14, #3
	bge	encode_triplet
	cmp	x14, #0
	beq	encode_finish
	cmp	x14, #1
	beq	encode_one
	// else must be 2
	// encode two bytes
	ldrb	w4, [x11], #1		// b0
	ldrb	w5, [x11], #1		// b1
	subs	x14, x14, #2

	// idx0 = b0 >> 2
	lsl	x6, x4, #0
	lsr	x6, x6, #2

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	x7, x4, #3
	lsl	x7, x7, #4
	lsr	x8, x5, #4
	orr	x7, x7, x8

	// idx2 = ((b1 & 0xF) << 2)
	and	x8, x5, #0xF
	lsl	x8, x8, #2

	// write idx0
	ldrb	w0, [x15, x6]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	1f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
1:
	// write idx1
	ldrb	w0, [x15, x7]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	2f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
2:
	// write idx2
	ldrb	w0, [x15, x8]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	3f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
3:
	// write padding '='
	mov	w0, #'='
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	4f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
4:
	b	encode_main

encode_one:
	// encode single byte
	ldrb	w4, [x11], #1		// b0
	subs	x14, x14, #1

	// idx0 = b0 >> 2
	lsl	x6, x4, #0
	lsr	x6, x6, #2

	// idx1 = (b0 & 0x3) << 4
	and	x7, x4, #3
	lsl	x7, x7, #4

	// write idx0
	ldrb	w0, [x15, x6]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	5f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
5:
	// write idx1
	ldrb	w0, [x15, x7]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	6f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
6:
	// write two paddings '=='
	mov	w0, #'='
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	7f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
7:
	mov	w0, #'='
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	8f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
8:
	b	encode_main

encode_triplet:
	// encode three bytes
	ldrb	w4, [x11], #1		// b0
	ldrb	w5, [x11], #1		// b1
	ldrb	w6, [x11], #1		// b2
	subs	x14, x14, #3

	// idx0 = b0 >> 2
	lsl	x7, x4, #0
	lsr	x7, x7, #2

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	x8, x4, #3
	lsl	x8, x8, #4
	lsr	x9, x5, #4
	orr	x8, x8, x9

	// idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	and	x9, x5, #0xF
	lsl	x9, x9, #2
	lsr	x10, x6, #6
	orr	x9, x9, x10

	// idx3 = b2 & 0x3F
	and	x10, x6, #0x3F

	// write idx0
	ldrb	w0, [x15, x7]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	9f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
9:
	// write idx1
	ldrb	w0, [x15, x8]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	10f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
10:
	// write idx2
	ldrb	w0, [x15, x9]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	11f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
11:
	// write idx3
	ldrb	w0, [x15, x10]
	strb	w0, [x12], #1
	add	x13, x13, #1
	cmp	x13, #76
	bne	12f
	mov	w0, #10
	strb	w0, [x12], #1
	mov	x13, #0
12:
	b	encode_main

encode_finish:
	// after all groups, if any output written and line_len != 0, write final newline
	cmp	x13, #0
	beq	after_newline
	// write newline
	mov	w0, #10
	strb	w0, [x12], #1
after_newline:
	// compute output length = out_ptr - output_buf
	adrp	x0, output_buf
	add	x0, x0, :lo12:output_buf
	mov	x1, x12
	sub	x1, x1, x0		// length in x1

	// write output (may need loop)
	mov	x2, x0			// cur_ptr
write_loop:
	cbnz	x1, write_one
	b	exit

write_one:
	mov	x0, #1			// fd stdout
	mov	x1, x2
	mov	x2, x1			// len argument already in x1, but need x2 = count
	// actually x2 already holds len, keep separate
	mov	x3, x1			// placeholder, not used
	mov	x8, #64
	svc	#0
	// x0 = written
	add	x2, x2, #0		// preserve length in x1? we will use x1 as remaining
	// adjust pointers
	add	x2, x2, #0		// no-op
	add	x2, x2, #0
	// Use separate regs: we need remaining in x4, pointer in x5
	// To simplify, rewrite write loop with separate registers
	b	write_loop

exit:
	mov	x0, #0
	mov	x8, #93
	svc	#0

	.section .rodata
b64_table:
	.ascii	"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

	.bss
	.align	16
input_buf:
	.zero	65536
output_buf:
	.zero	70000
