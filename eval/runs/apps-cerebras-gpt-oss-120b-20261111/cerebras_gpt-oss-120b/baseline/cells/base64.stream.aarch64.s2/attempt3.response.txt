	.global	_start
	.text
	.align	4

	.equ	SYSCALL_READ,	63
	.equ	SYSCALL_WRITE,	64
	.equ	SYSCALL_EXIT,	93

_start:
	// x9 = base of input_buf
	adrp	x9, input_buf
	add	x9, x9, :lo12:input_buf
	mov	x11, x9			// keep base for later
	mov	x10, #0			// total bytes read = 0

read_loop:
	// x1 = buf = base + total
	add	x1, x9, x10
	mov	x0, #0			// fd = stdin
	mov	x2, #65536
	mov	x8, #SYSCALL_READ
	svc	#0
	cbz	x0, read_done
	add	x10, x10, x0
	b	read_loop

read_done:
	// if no input, exit silently
	cbz	x10, exit_success

	// x11 = current input pointer
	mov	x11, x9
	mov	x14, x10			// remaining bytes
	// line buffer pointer
	adrp	x12, line_buf
	add	x12, x12, :lo12:line_buf
	mov	w13, #0			// line_len = 0
	// base64 table pointer
	adrp	x15, b64_table
	add	x15, x15, :lo12:b64_table

encode_loop:
	cmp	x14, #3
	b.ge	encode_triplet
	cmp	x14, #2
	beq	encode_two
	cmp	x14, #1
	beq	encode_one
	b	encode_finish

// ----- encode 3 bytes -----
encode_triplet:
	ldrb	w4, [x11], #1		// b0
	ldrb	w5, [x11], #1		// b1
	ldrb	w6, [x11], #1		// b2
	sub	x14, x14, #3

	// idx0 = b0 >> 2
	lsr	w7, w4, #2

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	w8, w4, #3
	lsl	w8, w8, #4
	lsr	w9, w5, #4
	orr	w8, w8, w9

	// idx2 = ((b1 & 0xF) << 2) | (b2 >> 6)
	and	w9, w5, #0xF
	lsl	w9, w9, #2
	lsr	w10, w6, #6
	orr	w9, w9, w10

	// idx3 = b2 & 0x3F
	and	w10, w6, #0x3F

	// output idx0
	uxtw	x7, w7
	ldrb	w0, [x15, x7]
	bl	store_char
	// output idx1
	uxtw	x8, w8
	ldrb	w0, [x15, x8]
	bl	store_char
	// output idx2
	uxtw	x9, w9
	ldrb	w0, [x15, x9]
	bl	store_char
	// output idx3
	uxtw	x10, w10
	ldrb	w0, [x15, x10]
	bl	store_char
	b	encode_loop

// ----- encode 2 bytes -----
encode_two:
	ldrb	w4, [x11], #1		// b0
	ldrb	w5, [x11], #1		// b1
	sub	x14, x14, #2

	// idx0 = b0 >> 2
	lsr	w7, w4, #2

	// idx1 = ((b0 & 3) << 4) | (b1 >> 4)
	and	w8, w4, #3
	lsl	w8, w8, #4
	lsr	w9, w5, #4
	orr	w8, w8, w9

	// idx2 = ((b1 & 0xF) << 2)
	and	w9, w5, #0xF
	lsl	w9, w9, #2

	// output idx0
	uxtw	x7, w7
	ldrb	w0, [x15, x7]
	bl	store_char
	// output idx1
	uxtw	x8, w8
	ldrb	w0, [x15, x8]
	bl	store_char
	// output idx2
	uxtw	x9, w9
	ldrb	w0, [x15, x9]
	bl	store_char
	// padding '='
	mov	w0, #'='
	bl	store_char
	b	encode_loop

// ----- encode 1 byte -----
encode_one:
	ldrb	w4, [x11], #1		// b0
	sub	x14, x14, #1

	// idx0 = b0 >> 2
	lsr	w7, w4, #2

	// idx1 = ((b0 & 3) << 4)
	and	w8, w4, #3
	lsl	w8, w8, #4

	// output idx0
	uxtw	x7, w7
	ldrb	w0, [x15, x7]
	bl	store_char
	// output idx1
	uxtw	x8, w8
	ldrb	w0, [x15, x8]
	bl	store_char
	// padding '=' twice
	mov	w0, #'='
	bl	store_char
	bl	store_char
	b	encode_loop

// ----- finish output -----
encode_finish:
	cmp	w13, #0
	beq	exit_success

	// store final newline
	mov	w0, #10
	strb	w0, [x12], #1

	// write remaining line (len = line_len + 1)
	mov	x0, #1
	add	w2, w13, #1
	mov	x2, w2
	sub	x1, x
