	.globl	_start
	.text
_start:
	// read entire stdin into buf (max 65536 bytes)
	adrp	x1, buf@PAGE
	add	x1, x1, :lo12:buf       // x1 = ptr to buffer start
	mov	x2, #65536              // x2 = remaining space
	mov	x9, #0                  // x9 = total bytes read

read_loop:
	mov	x0, #0                  // fd = stdin
	mov	x8, #63                 // sys_read
	svc	#0
	cmp	x0, #0
	beq	read_done               // EOF
	cmp	x0, #0
	blt	error_exit              // error
	add	x9, x9, x0              // total += ret
	add	x1, x1, x0              // buf += ret
	subs	x2, x2, x0              // remaining -= ret
	beq	read_done
	b	read_loop

read_done:
	// output "ok\n"
	adrp	x0, ok_msg@PAGE
	add	x0, x0, :lo12:ok_msg    // x0 = pointer to message
	adrp	x1, ok_len@PAGE
	ldr	w1, [x1, :lo12:ok_len]  // w1 = length
	mov	x8, #64                 // sys_write
	mov	x2, w1                  // x2 = length (zero‑extend)
	mov	x1, x0
