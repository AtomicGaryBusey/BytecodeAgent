	.global	_start
	.text
_start:
	// buffer pointer and counters
	adrp	x1, buf@PAGE
	add	x1, x1, :lo12:buf          // x1 = buf start
	mov	x2, #65536                // x2 = remaining space
	mov	x9, #0                    // x9 = total bytes read

read_loop:
	mov	x0, #0                    // fd = stdin
	mov	x8, #63                   // sys_read
	svc	#0
	cmp	x0, #0
	beq	read_done                  // EOF
	blt	error_exit                 // error (negative)
	add	x9, x9, x0                // total += ret
	add	x1, x1, x0                // buf += ret
	subs	x2, x2, x0                // remaining -= ret
	beq	read_done
	b	read_loop

read_done:
	// output "ok\n"
	adrp	x0, ok_msg@PAGE
	add	x0, x0, :lo12:ok_msg      // x0 = pointer to message
	adrp	x1, ok_len@PAGE
	ldr	w1, [x1, :lo12:ok_len]    // w1 = length
	uxtw	x2, w1                    // x2 = length (zero‑extend)
	mov	x8, #64                   // sys_write
	svc	#0

	// exit with status 0
	mov	x0, #0
	mov	x8, #93                   // sys_exit
	svc	#0

error_exit:
	// output "error at 0\n"
	adrp	x0, err_msg@PAGE
	add	x0, x0, :lo12:err_msg
	adrp	x1, err_len@PAGE
	ldr	w1, [x1, :lo12:err_len]
	uxtw	x2, w1
	mov	x8, #64
	svc	#0
	mov	x0, #1
	mov	x8, #93
	svc	#0

	.section	.rodata
ok_msg:
	.asciz	"ok\n"
ok_len:
	// calculate length at assembly time
	.equ ok_len_val, . - ok_msg
	.word	ok_len_val

err_msg:
	.asciz	"error at 0\n"
err_len:
	.equ err_len_val, . - err_msg
	.word	err_len_val

	.bss
	.align	4
buf:
	.zero	65536
