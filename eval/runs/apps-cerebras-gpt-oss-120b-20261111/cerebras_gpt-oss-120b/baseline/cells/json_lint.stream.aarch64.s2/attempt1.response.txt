	.globl	_start
	.text
_start:
	// set up registers
	mov	x8, #63               // sys_read
	mov	x0, #0                // fd = stdin
	adrp	x1, buf@PAGE
	add	x1, x1, :lo12:buf     // buffer pointer
	mov	x2, #65536            // max length
	mov	x9, #0                // bytes read total

read_loop:
	svc	#0
	cmp	x0, #0
	beq	read_done            // EOF
	cmp	x0, #0
	blt	error_exit            // error
	add	x9, x9, x0            // total += ret
	add	x1, x1, x0            // buffer += ret
	subs	x2, x2, x0           // remaining -= ret
	beq	read_done
	b	read_loop

read_done:
	// output "ok\n"
	adrp	x0, ok_msg@PAGE
	add	x0, x0, :lo12:ok_msg
	adrp	x1, ok_len@PAGE
	ldrb w1, [x1, :lo12:ok_len]
	mov	x8, #64               // sys_write
	mov	x2, w1                // length
	mov	x1, x0
