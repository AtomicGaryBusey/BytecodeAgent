.global _start
    .text

// system call numbers
    .equ SYS_READ, 63
    .equ SYS_WRITE, 64
    .equ SYS_EXIT, 93

// constants
    .equ INPUT_BUF_SIZE, 65536
    .equ STACK_MAX, 1024
    .equ OUT_BUF_SIZE, 32

_start:
    // x20 = input buffer address
    adr     x20, input_buf
    mov     x19, #0                     // total bytes read = 0

read_loop:
    mov     x0, #0                      // fd = stdin
    add     x1, x20, x19                // buf = input_buf + total_len
    mov     x2, #INPUT_BUF_SIZE
    sub     x2, x2, x19                 // count = remaining space
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done               // EOF
    add     x19, x19, x0                // total_len += bytes_read
    cmp     x19, #INPUT_BUF_SIZE
    b.lt    read_loop
read_done:
    // x22 = end pointer (input_buf + total_len)
    add     x22, x20, x19
    mov     x21, x20                    // x21 = current pointer

line_loop:
    cmp     x21, x22
    b.ge    done_processing

    // line_start = x21
    mov     x23, x21

    // find newline
    mov     x24, x21
find_nl:
    cmp     x24, x22
    b.ge    no_nl
    ldrb    w0, [x24]
    cmp     w0, #0x0a
    b.eq    have_nl
    add     x24, x24, #1
    b       find_nl
have_nl:
    // line_end = x24 (exclude newline)
    mov     x25, x24
    b       line_process
no_nl:
    // line_end = end
    mov     x25, x22
    b       line_process

line_process:
    // stack base in x26, stack count x27 = 0, error flag x28 = 0
    adr     x26, stack_buf
    mov     x27, #0
    mov     x28, #0

    // token pointer in x5
    mov     x5, x23

token_loop:
    cmp     x5, x25
    b.ge    token_done

    // skip separators (space 0x20, tab 0x09)
skip_sep:
    cmp     x5, x25
    b.ge    token_done
    ldrb    w0, [x5]
    cmp     w0, #0x20
    beq     inc_ptr
    cmp     w0, #0x09
    beq     inc_ptr
    b       token_start
inc_ptr:
    add     x5, x5, #1
    b       skip_sep

token_start:
    // token_start = x5
    mov     x9, x5
    // token_end = x6
    mov     x6, x5
token_end:
    cmp     x6, x25
    b.ge    token_end_done
    ldrb    w0, [x6]
    cmp     w0, #0x20
    beq     token_end_done
    cmp     w0, #0x09
    beq     token_end_done
    add     x6, x6, #1
    b       token_end
token_end_done:
    // token length = x7
    sub     x7, x6, x9

    // load first byte of token
    ldrb    w0, [x9]

    // check if token is single-char operator
    cmp     x7, #1
    b.ne    parse_number
    cmp     w0, #'+'
    beq     op_add
    cmp     w0, #'-'
    beq     op_sub
    cmp     w0, #'*'
    beq     op_mul
    cmp     w0, #'/'
    beq     op_div
    cmp     w0, #'%'
    beq     op_mod
    // not operator, treat as number
    b       parse_number

parse_number:
    // x12 = sign flag (0 pos, 1 neg)
    mov     x12, #0
    // x13 = digit count
    mov     x13, #0
    // x14 = accumulator
    mov     x14, #0

    // handle optional leading '-'
    cmp     w0, #'-'
    b.ne    num_digits_loop
    // leading minus
    mov     x12, #1
    add     x9, x9, #1
    sub     x7, x7, #1
    cmp     x7, #0
    beq     token_error

num_digits_loop:
    // pointer x15 = current digit ptr
    mov     x15, x9
    // digit count in x13, accumulator in x14
num_digit_iter:
    cmp     x15, x6
    b.ge    num_digits_done
    ldrb    w0, [x15]
    cmp     w0, #'0'
    blt     token_error
    cmp     w0, #'9'
    bgt     token_error
    // accumulator = accumulator*10 + (w0-'0')
    mov     x10, #10
    mul     x14, x14, x10
    sub     w0, w0, #'0'
    add     x14, x14, x0
    add     x15, x15, #1
    add     x13, x13, #1
    b       num_digit_iter
num_digits_done:
    // check digit count 1..18
    cmp     x13, #0
    beq     token_error
    cmp     x13, #19
    b.ge    token_error
    // apply sign if needed
    cmp     x12, #0
    beq     push_number
    neg     x14, x14
push_number:
    // push x14 onto stack
    cmp     x27, #STACK_MAX
    b.ge    token_error
    str     x14, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       next_token

op_add:
    // need two operands
    cmp     x27, #2
    b.lt    token_error
    sub     x27, x27, #1
    ldr     x5, [x26, x27, lsl #3]        // b
    sub     x27, x27, #1
    ldr     x4, [x26, x27, lsl #3]        // a
    add     x4, x4, x5
    str     x4, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       next_token

op_sub:
    cmp     x27, #2
    b.lt    token_error
    sub     x27, x27, #1
    ldr     x5, [x26, x27, lsl #3]
    sub     x27, x27, #1
    ldr     x4, [x26, x27, lsl #3]
    sub     x4, x4, x5
    str     x4, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       next_token

op_mul:
    cmp     x27, #2
    b.lt    token_error
    sub     x27, x27, #1
    ldr     x5, [x26, x27, lsl #3]
    sub     x27, x27, #1
    ldr     x4, [x26, x27, lsl #3]
    mul     x4, x4, x5
    str     x4, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       next_token

op_div:
    cmp     x27, #2
    b.lt    token_error
    sub     x27, x27, #1
    ldr     x5, [x26, x27, lsl #3]        // b
    sub     x27, x27, #1
    ldr     x4, [x26, x27, lsl #3]        // a
    // b == 0 ?
    cbz     x5, div_error
    // load INT64_MIN into x6, -1 into x7
    movz    x6, #0x8000, lsl #48        // x6 = 0x8000000000000000
    mov     x7, #0
    sub     x7, x7, #1                  // x7 = -1
    cmp     x4, x6
    b.ne    not_overflow_div
    cmp     x5, x7
    b.ne    not_overflow_div
    // overflow case
    b       div_error
not_overflow_div:
    sdiv    x4, x4, x5
    str     x4, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       next_token
div_error:
    mov     x28, #1
    b       after_error

op_mod:
    cmp     x27, #2
    b.lt    token_error
    sub     x27, x27, #1
    ldr     x5, [x26, x27, lsl #3]        // b
    sub     x27, x27, #1
    ldr     x4, [x26, x27, lsl #3]        // a
    // b == 0 ?
    cbz     x5, mod_error
    // load constants
    movz    x6, #0x8000, lsl #48        // INT64_MIN
    mov     x7, #0
    sub     x7, x7, #1                  // -1
    cmp     x4, x6
    b.ne    not_overflow_mod
    cmp     x5, x7
    b.ne    not_overflow_mod
    b       mod_error
not_overflow_mod:
    sdiv    x8, x4, x5
    mul     x9, x8, x5
    sub     x4, x4, x9
    str     x4, [x26, x27, lsl #3]
    add     x27, x27, #1
    b       next_token
mod_error:
    mov     x28, #1
    b       after_error

token_error:
    mov     x28, #1
    b       after_error

next_token:
    // advance token pointer to token end (x6)
    mov     x5, x6
    b       token_loop

after_error:
    // skip rest of line
    mov     x5, x25
    b       token_done

token_done:
    // decide output
    cmp     x28, #0
    b.ne    print_error
    cmp     x27, #1
    b.ne    print_error
    // good result
    sub     x27, x27, #1
    ldr     x0, [x26, x27, lsl #3]       // value to print
    bl      int_to_ascii
    b       after_output

print_error:
    adr     x0, error_msg
    mov     x1, #6
    bl      write_str

after_output:
    // advance to next line
    cmp     x25, x22
    beq     done_processing
    add     x21, x25, #1
    b       line_loop

done_processing:
    // exit with status 0
    mov     x0, #0
    mov     x8, #SYS_EXIT
    svc     #0

// -----------------------------------------------------------------------------
// write_str: x0 = ptr, x1 = len
write_str:
    mov     x2, x1
    mov     x1, x0
    mov     x0, #1          // stdout
    mov     x8, #SYS_WRITE
    svc     #0
    ret

// -----------------------------------------------------------------------------
// int_to_ascii: input x0 = signed 64-bit integer, uses out_buf, writes via write_str
int_to_ascii:
    adr     x9, out_buf
    add     x9, x9, #OUT_BUF_SIZE     // x9 = end pointer
    mov     x10, x9                  // x10 = cur pointer
    // sign flag in x11
    cmp     x0, #0
    b.ge    .Lpos
    // negative
    mov     x11, #1
    neg     x0, x0
    b       .Lconvert
.Lpos:
    mov     x11, #0
.Lconvert:
    // constant 10 in x12
    mov     x12, #10
    // do-while loop (always at least one digit)
.Lconv_loop:
    udiv    x4, x0, x12
    mul     x5, x4, x12
    sub     x6, x0, x5               // remainder
    add     x6, x6, #'0'
    sub     x10, x10, #1
    strb    w6, [x10]
    mov     x0, x4
    cmp     x0, #0
    bne     .Lconv_loop
    // sign
    cmp     x11, #0
    beq     .Lno_sign
    sub     x10, x10, #1
    mov     w6, #'-'
    strb    w6, [x10]
.Lno_sign:
    // length = end - cur
    sub     x11, x9, x10               // x11 = length
    mov     x0, x10                   // ptr
    mov     x1, x11                   // len
    bl      write_str
    ret

    .section .rodata
error_msg:
    .asciz "error\n"

    .bss
    .align 8
input_buf:
    .zero INPUT_BUF_SIZE
stack_buf:
    .zero STACK_MAX*8
out_buf:
    .zero OUT_BUF_SIZE
