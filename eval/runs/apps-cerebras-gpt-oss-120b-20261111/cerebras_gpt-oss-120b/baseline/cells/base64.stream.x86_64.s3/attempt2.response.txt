.globl _start
.equ IN_BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 66400
.equ LINE_MAX, 76

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
in_buf:
    .zero IN_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE

.text
_start:
    xor rbx, rbx                # total bytes read = 0
.read_loop:
    mov rdi, 0                 # fd = stdin
    lea rsi, [rip + in_buf]
    add rsi, rbx               # buffer + bytes already read
    mov rdx, IN_BUF_SIZE
    sub rdx, rbx               # remaining buffer space
    mov rax, 0                 # sys_read
    syscall
    test rax, rax
    je .encode                 # EOF
    add rbx, rax               # accumulate count
    jmp .read_loop

.encode:
    xor rcx, rcx               # input index i = 0
    xor r8, r8                 # output index out_i = 0
    xor rdx, rdx               # line length counter = 0
    lea r10, [rip + b64_table] # table pointer
    lea r11, [rip + out_buf]   # output buffer base
