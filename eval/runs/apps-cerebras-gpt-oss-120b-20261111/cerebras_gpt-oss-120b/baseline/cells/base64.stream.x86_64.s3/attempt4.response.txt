.globl _start
.equ IN_BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 66400
.equ LINE_MAX, 76

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
in_buf:
    .zero IN_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE

.text
_start:
    xor rbx, rbx                # total_read = 0

read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + in_buf]
    add rsi, rbx                # dest = in_buf + total_read
    mov rdx, IN_BUF_SIZE
    sub rdx, rbx                # remaining space
    test rdx, rdx
    jz encode_start
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    je encode_start             # EOF
    add rbx, rax                # total_read += bytes
    jmp read_loop

encode_start:
    test rbx, rbx
    je exit_success             # empty input

    lea rsi, [rip + in_buf]     # input pointer
    mov rcx, rbx                # remaining input length
    lea rdi, [rip + out_buf]    # output pointer
    xor rdx, rdx                # line_len = 0
    lea r8, [rip + b64_table]   # base64 table

encode_loop:
    cmp rcx, 3
    jl encode_tail

    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]

    # idx0 = b1 >> 2
    mov r9d, eax
    shr r9d, 2
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl0
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl0:

    # idx1 = ((b1 & 3) << 4) | (b2 >> 4)
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ebx
    shr r10d, 4
    or r9d, r10d
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl1:

    # idx2 = ((b2 & 0xF) << 2) | (b3 >> 6)
    mov r9d, ebx
    and r9d, 0xF
    shl r9d, 2
    mov r10d, ecx
    shr r10d, 6
    or r9d, r10d
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl2:

    # idx3 = b3 & 0x3F
    mov r9d, ecx
    and r9d, 0x3F
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl3:

    add rsi, 3
    sub rcx, 3
    jmp encode_loop

encode_tail:
    cmp rcx, 0
    je finish_newline
    cmp rcx, 1
    je tail_one
    # rcx == 2
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]

    # idx0
    mov r9d, eax
    shr r9d, 2
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl4:

    # idx1 = ((b1 & 3) << 4) | (b2 >> 4)
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov r10d, ebx
    shr r10d, 4
    or r9d, r10d
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl5
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl5:

    # idx2 = ((b2 & 0xF) << 2)
    mov r9d, ebx
    and r9d, 0xF
    shl r9d, 2
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl6
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl6:

    # padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl7
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl7:
    jmp finish_newline

tail_one:
    movzx eax, byte ptr [rsi]

    # idx0
    mov r9d, eax
    shr r9d, 2
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl8
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl8:

    # idx1 = (b1 & 3) << 4
    mov r9d, eax
    and r9d, 3
    shl r9d, 4
    mov al, byte ptr [r8 + r9]
    mov byte ptr [rdi], al
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl9
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl9:

    # padding '=' twice
    mov byte ptr [rdi], '='
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl10
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl10:
    mov byte ptr [rdi], '='
    inc rdi
    inc rdx
    cmp rdx, LINE_MAX
    jne .Lno_nl11
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
.Lno_nl11:
    jmp finish_newline

finish_newline:
    cmp rdx, 0
    je write_output
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor rdx, rdx
    jmp write_output

write_output:
    lea rsi, [rip + out_buf]
    mov rbx, rdi
    sub rbx, rsi               # rbx = output length
    test rbx, rbx
    je exit_success

    mov rdx, rbx               # bytes remaining
write_loop:
    mov rdi, 1                 # fd = stdout
    mov rax, 1                 # sys_write
    syscall
    test rax, rax
    jle exit_success
    add rsi, rax
    sub rdx, rax
    test rdx, rdx
    jne write_loop

exit_success:
    mov rax, 60                # sys_exit
    xor rdi, rdi
    syscall
