.globl _start
.equ IN_BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 66400
.equ LINE_MAX, 76

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
.align 16
in_buf:
    .zero IN_BUF_SIZE
out_buf:
    .zero OUT_BUF_SIZE

.text
_start:
    xor rbx, rbx                # total_bytes_read = 0

read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + in_buf]
    add rsi, rbx                # buffer + bytes already read
    mov rdx, IN_BUF_SIZE
    sub rdx, rbx                # remaining space
    test rdx, rdx
    jz encode_start             # buffer full (should not happen)
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    je encode_start             # EOF
    add rbx, rax                # accumulate count
    jmp read_loop

encode_start:
    lea rsi, [rip + in_buf]     # input pointer
    mov rcx, rbx                # remaining input length
    lea rdi, [rip + out_buf]    # output pointer
    xor rdx, rdx                # line_len = 0
    lea r8, [rip + b64_table]   # table pointer

encode_loop:
    cmp rcx, 3
    jl handle_tail

    # load three bytes into rax (24 bits)
    movzx rax, byte [rsi]       # high byte
    shl rax, 16
