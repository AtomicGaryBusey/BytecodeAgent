.globl _start
.bss
.align 16
buf: .zero 65536
outbuf: .zero 64
lines_counter: .zero 8
words_counter: .zero 8
bytes_counter: .zero 8
in_word: .zero 1

.text
_start:
    # read loop
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buf]        # buffer
    mov rdx, 65536              # count
    syscall
    test rax, rax
    je after_read
    add qword ptr [bytes_counter], rax
    lea rsi, [rip + buf]
    xor rbx, rbx                # index = 0
proc_loop:
    cmp rbx, rax
    jae read_loop
    movzx edx, byte ptr [rsi + rbx]
    inc rbx
    cmp dl, 0x0A                # newline
    je is_newline
    cmp dl, 0x20                # space
    je is_ws
    cmp dl, 0x09                # tab
    je is_ws
    cmp dl, 0x0B                # vertical tab
    je is_ws
    cmp dl, 0x0C                # form feed
    je is_ws
    cmp dl, 0x0D                # carriage return
    je is_ws
    # non‑whitespace
    mov al, byte ptr [in_word]
    test al, al
    jnz proc_continue
    inc qword ptr [words_counter]
    mov byte ptr [in_word], 1
    jmp proc_continue
is_newline:
    inc qword ptr [lines_counter]
    mov byte ptr [in_word], 0
    jmp proc_continue
is_ws:
    mov byte ptr [in_word], 0
proc_continue:
    jmp proc_loop

after_read:
    lea rdi, [rip + outbuf]     # destination pointer
    mov rax, qword ptr [lines_counter]
    call conv_uint64
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, qword ptr [words_counter]
    call conv_uint64
    mov byte ptr [rdi], ' '
    inc rdi
    mov rax, qword ptr [bytes_counter]
    call conv_uint64
    mov byte ptr [rdi], 0x0A
    inc rdi
    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                # length
    mov rax, 1                  # sys_write
    mov rdi, 1                  # stdout
    syscall
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

# Convert unsigned 64‑bit integer in RAX to decimal ASCII.
# RDI = destination pointer (advanced past written digits)
conv_uint64:
    sub rsp, 32                 # temporary buffer (max 20 digits)
    mov rbx, rsp
    xor rcx, rcx                # digit count
    cmp rax, 0
    jne .cvt_loop
    mov byte ptr [rbx], '0'
    inc rcx
    jmp .cvt_copy
.cvt_loop:
    xor rdx, rdx
    mov r8, 10
    div r8                      # RAX = quotient, RDX = remainder
    add dl, '0'
    mov byte ptr [rbx + rcx], dl
    inc rcx
    test rax, rax
    jnz .cvt_loop
.cvt_copy:
    mov rsi, rbx
    mov r8, rcx
    dec r8
.cpy:
    cmp r8, -1
    jl .cvt_done
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rdi], al
    inc rdi
    dec r8
    jmp .cpy
.cvt_done:
    add rsp, 32
    ret
