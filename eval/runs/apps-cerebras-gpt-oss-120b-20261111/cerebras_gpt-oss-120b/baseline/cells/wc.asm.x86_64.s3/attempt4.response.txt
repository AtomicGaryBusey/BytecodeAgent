.globl _start
.bss
.align 16
buf:    .zero 65536
outbuf: .zero 64

.text
_start:
    xor     r8, r8          # lines
    xor     r9, r9          # words
    xor     r10, r10        # bytes
    xor     r11, r11        # in_word flag

read_loop:
    mov     rax, 0          # sys_read
    mov     rdi, 0          # fd 0
    lea     rsi, [rip + buf]
    mov     rdx, 65536
    syscall
    test    rax, rax
    je      after_read
    add     r10, rax        # bytes += count
    mov     rcx, rax        # rcx = nbytes
    xor     rbx, rbx        # index = 0

proc_loop:
    cmp     rbx, rcx
    jae     read_loop
    mov     al, byte ptr [buf + rbx]
    inc     rbx
    cmp     al, 0x0A        # newline
    je      is_newline
    cmp     al, 0x20        # space
    je      is_ws
    cmp     al, 0x09        # tab
    je      is_ws
    cmp     al, 0x0B        # vertical tab
    je      is_ws
    cmp     al, 0x0C        # form feed
    je      is_ws
    cmp     al, 0x0D        # carriage return
    je      is_ws
    # non‑whitespace
    test    r11, r11
    jnz     proc_continue
    inc     r9              # words++
    mov     r11, 1
    jmp     proc_continue

is_newline:
    inc     r8              # lines++
    xor     r11, r11        # in_word = 0
    jmp     proc_continue

is_ws:
    xor     r11, r11        # in_word = 0

proc_continue:
    jmp     proc_loop

after_read:
    lea     rdi, [rip + outbuf]

    mov     rax, r8
    call    uint_to_ascii
    mov     byte ptr [rdi], ' '
    inc     rdi

    mov     rax, r9
    call    uint_to_ascii
    mov     byte ptr [rdi], ' '
    inc     rdi

    mov     rax, r10
    call    uint_to_ascii
    mov     byte ptr [rdi], 0x0A
    inc     rdi

    lea     rsi, [rip + outbuf]
    mov     rdx, rdi
    sub     rdx, rsi            # length
    mov     rax, 1              # sys_write
    mov     rdi, 1              # stdout
    syscall

    mov     rax, 60             # sys_exit
    xor     rdi, rdi
    syscall

# ------------------------------------------------------------
# Convert unsigned integer in RAX to decimal ASCII.
# RDI = destination pointer, advanced past written digits.
# Clobbers: RAX, RCX, RDX, RBX, RBP (via push/pop)
uint_to_ascii:
    push    rbx
    sub     rsp, 32             # temp buffer (max 20 bytes)
    mov     rbx, rsp            # rbx = temp buffer
    xor     rcx, rcx            # digit count

    cmp     rax, 0
    jne     .conv_loop
    mov     byte ptr [rbx], '0'
    inc     rcx
    jmp     .copy_back

.conv_loop:
    xor     rdx, rdx
    mov     rbp, 10
    div     rbp                 # rax = quotient, rdx = remainder
    add     dl, '0'
    mov     byte ptr [rbx + rcx], dl
    inc     rcx
    test    rax, rax
    jnz     .conv_loop

.copy_back:
    mov     rdx, rcx
    dec     rdx                 # rdx = most‑significant digit index
.copy_loop:
    cmp     rdx, -1
    jl      .done_copy
    mov     al, byte ptr [rbx + rdx]
    mov     byte ptr [rdi], al
    inc     rdi
    dec     rdx
    jmp     .copy_loop

.done_copy:
    add     rsp, 32
    pop     rbx
    ret
