.globl _start

.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

.section .bss
buffer:          .zero BUF_SIZE
line_off:        .zero MAX_LINES*8
line_len:        .zero MAX_LINES*8
line_cnt:        .zero 8

.section .rodata
nl: .byte 0x0A

.text
_start:
    # read all input into buffer
    xor     rax, rax                # sys_read = 0
    xor     rdi, rdi                # fd = 0
    lea     rsi, [rip + buffer]
    mov     rdx, BUF_SIZE
.read_loop:
    syscall
    test    rax, rax
    jz      .parse_input
    add     qword ptr [rip + total_len], rax
    jmp     .read_loop

.total_len:
    .quad   0

.parse_input:
    lea     rsi, [rip
