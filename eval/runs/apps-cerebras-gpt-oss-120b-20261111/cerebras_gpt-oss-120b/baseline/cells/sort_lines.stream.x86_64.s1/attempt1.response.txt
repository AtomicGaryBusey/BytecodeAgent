.globl _start
.section .rodata
nl: .byte 0x0A

.section .bss
buffer: .zero 32768
line_off: .zero 8000       # 8 * 1000
line_len: .zero 8000

.section .text
_start:
    xor rbx, rbx                # total_len = 0
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buffer]
    mov rdx, 32768
    syscall
    test rax, rax
    je parse_input
    add rbx, rax                # total_len += bytes_read
    jmp read_loop

parse_input:
    mov r8, rbx                 # r8 = total_len
    xor rcx, rcx                # line index = 0
    xor rdx, rdx                # start = 0
    xor rsi, rsi                # i = 0
parse_loop:
    cmp rsi, r8
    je parse_end_no_newline
    mov al, byte ptr [buffer + rsi]
    cmp al, 0x0A
    jne next_char
    mov rax, rsi
    sub rax, rdx                # length = i - start
    lea rdi, [rip + line_off]
    mov [rdi + rcx*8], rdx
    lea rdi, [rip + line_len]
    mov [rdi + rcx*8], rax
    inc rcx
    lea rdx, [rsi + 1]          # start = i + 1
next_char:
    inc rsi
    jmp parse_loop

parse_end_no_newline:
    cmp rdx, r8
    jae after_parse
    mov rax, r8
    sub rax, rdx                # length = total_len - start
    lea rdi, [rip + line_off]
    mov [rdi + rcx*8], rdx
    lea rdi, [rip + line_len]
    mov [rdi + rcx*8], rax
    inc rcx
after_parse:
    mov r13, rcx                # line_count

    cmp r13, 1
    jle sort_done

    mov rdx, 1                  # i = 1
outer_loop:
    cmp rdx, r13
    jae sort_done
    mov rbx, rdx                # key = i
    mov rax, rdx
    dec rax                     # j = i-1
inner_loop:
    cmp rax, -1
    jl outer_continue
    lea rdi, [rip + line_off]
    mov r8,  [rdi + rax*8]      # off_j
    lea rdi, [rip + line_len]
    mov r10, [rdi + rax*8]      # len_j
    lea rdi, [rip + line_off]
    mov r11, [rdi + rbx*8]      # off_key
    lea rdi, [rip + line_len]
    mov r12, [rdi + rbx*8]      # len_key
    mov rdx, r8
    mov rcx, r10
    mov rsi, r11
    mov rdi, r12
    call compare_lines
    cmp rax, 0
    jle inner_done
    ; swap j and key
    lea rdi, [rip + line_off]
    mov r8, [rdi + rax*8]
