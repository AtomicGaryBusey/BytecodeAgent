	.global _start
	.text
	.align 2

// syscall numbers
	.equ SYSCALL_READ, 63
	.equ SYSCALL_WRITE, 64
	.equ SYSCALL_EXIT, 93

// buffer sizes
	.equ BUFFER_SIZE, 65536

_start:
	// x5 = total_len = 0
	mov	x5, #0

read_loop:
	// x0 = fd 0
	mov	x0, #0
	// x1 = input_buf + total_len
	adrp	x1, input_buf
	add	x1, x1, :lo12:input_buf
	add	x1, x1, x5
	// x2 = remaining space = BUFFER_SIZE - total_len
	mov	x2, #BUFFER_SIZE
	sub	x2, x2, x5
	// x8 = read
	mov	x8, #SYSCALL_READ
	svc	#0
	cbnz	x0, read_continue
	// x0 == 0 -> EOF
	b	read_done
read_continue:
	// add bytes read to total_len
	add	x5, x5, x0
	// if total_len >= BUFFER_SIZE stop reading
	cmp	x5, #BUFFER_SIZE
	bge	read_done
	b	read_loop
read_done:
	// x6 = base address of input_buf
	adrp	x6, input_buf
	add	x6, x6, :lo12:input_buf
	// x8 = offset = 0
	mov	x8, #0

line_loop:
	// if offset >= total_len -> finish
	cmp	x8, x5
	bge	finish

	// search for newline
	mov	x9, x8				// x9 = current index
search_newline:
	cmp	x9, x5
	bge	no_newline_found
	add	x10, x6, x9		// addr of byte
	ldrb	w11, [x10]
	cmp	w11, #10			// '\n'
	beq	newline_found
	add	x9, x9, #1
	b	search_newline

newline_found:
	// line_len = x9 - x8
	sub	x12, x9, x8		// x12 = line_len

	// copy reversed line into rev_buf
	adrp	x13, rev_buf
	add	x13, x13, :lo12:rev_buf	// dest base
	// src_ptr = x6 + x8 + line_len - 1
	add	x14, x6, x8
	add	x14, x14, x12
	sub	x14, x14, #1

	mov	x15, #0			// dest index i = 0
copy_rev_loop:
	cmp	x15, x12
	bge	copy_rev_done
	ldrb	w0, [x14]
	strb	w0, [x13, x15]
	sub	x14, x14, #1
	add	x15, x15, #1
	b	copy_rev_loop
copy_rev_done:
	// write reversed bytes
	mov	x0, #1			// fd 1
	adrp	x1, rev_buf
	add	x1, x1, :lo12:rev_buf
	mov	x2, x12
	mov	x8, #SYSCALL_WRITE
	svc	#0

	// write newline
	mov	x0, #1
	adrp	x1, newline_char
	add	x1, x1, :lo12:newline_char
	mov	x2, #1
	mov	x8, #SYSCALL_WRITE
	svc	#0

	// advance offset past newline
	add	x8, x9, #1
	b	line_loop

no_newline_found:
	// final line_len = total_len - offset
	sub	x12, x5, x8
	// copy reversed final line
	adrp	x13, rev_buf
	add	x13, x13, :lo12:rev_buf
	// src_ptr = x6 + total_len - 1
	add	x14, x6, x5
	sub	x14, x14, #1

	mov	x15, #0
copy_rev2_loop:
	cmp	x15, x12
	bge	write_rev2
	ldrb	w0, [x14]
	strb	w0, [x13, x15]
	sub	x14, x14, #1
	add	x15, x15, #1
	b	copy_rev2_loop
write_rev2:
	// write reversed bytes
	mov	x0, #1
	adrp	x1, rev_buf
	add	x1, x1, :lo12:rev_buf
	mov	x2, x12
	mov	x8, #SYSCALL_WRITE
	svc	#0

	// write newline
	mov	x0, #1
	adrp	x1, newline_char
	add	x1, x1, :lo12:newline_char
	mov	x2, #1
	mov	x8, #SYSCALL_WRITE
	svc	#0

	b	finish

finish:
	// exit with status 0
	mov	x0, #0
	mov	x8, #SYSCALL_EXIT
	svc	#0

	.section .bss
	.align 3
input_buf:
	.zero	BUFFER_SIZE
rev_buf:
	.zero	BUFFER_SIZE

	.section .rodata
	.align 1
newline_char:
	.byte	0x0A
