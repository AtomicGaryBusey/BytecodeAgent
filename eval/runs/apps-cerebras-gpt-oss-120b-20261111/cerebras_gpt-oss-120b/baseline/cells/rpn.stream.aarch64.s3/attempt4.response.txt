	.global	_start
	.text
	.align	2

	.equ	SYS_READ,	63
	.equ	SYS_WRITE,	64
	.equ	SYS_EXIT,	93

	.equ	INPUT_MAX,	65536
	.equ	STACK_MAX,	1024
	.equ	OUT_BUF_SIZE,	22

_start:
	// build constant INT64_MIN in x5
	movz	x5, #0x0000
	movk	x5, #0x8000, lsl #48		// x5 = 0x8000000000000000

	// read entire input into input_buf
	adrp	x1, input_buf
	add	x1, x1, :lo12:input_buf
	mov	x19, #0				// total bytes read

read_loop:
	adrp	x0, input_buf
	add	x0, x0, :lo12:input_buf
	add	x0, x0, x19			// dest = input_buf + total
	mov	x2, #INPUT_MAX
	sub	x2, x2, x19			// remaining capacity
	mov	x0, #0				// fd stdin
	mov	x8, #SYS_READ
	svc	#0
	cbnz	x0, have_bytes
	b	after_read
have_bytes:
	add	x19, x19, x0
	b	read_loop
after_read:
	// x19 = total input length
	adrp	x20, input_buf
	add	x20, x20, :lo12:input_buf	// current pointer
	mov	x21, x19			// bytes remaining
	adrp	x9, stack_buf
	add	x9, x9, :lo12:stack_buf	// stack base

main_loop:
	cbnz	x21, process_line
	b	exit_prog

process_line:
	// initialise per‑line state
	mov	x22, #0			// stack size
	mov	x24, #0			// error flag

skip_sep_line:
	cbnz	x21, check_sep
	b	line_end_no_newline
check_sep:
	ldrb	w0, [x20]
	cmp	w0, #0x20
	beq	advance_skip
	cmp	w0, #0x09
	beq	advance_skip
	b	after_skip_sep
advance_skip:
	add	x20, x20, #1
	subs	x21, x21, #1
	b	skip_sep_line
after_skip_sep:
	// empty line?
	cbnz	x21, check_nl
	b	line_end_no_newline
check_nl:
	ldrb	w0, [x20]
	cmp	w0, #0x0A
	beq	line_end_newline

	// token start at x20, remaining in x21
	// find token end
	mov	x25, x20
	mov	x26, x21
token_scan:
	cbnz	x26, token_scan_cont
	b	token_scan_done
token_scan_cont:
	ldrb	w0, [x25]
	cmp	w0, #0x20
	beq	token_scan_done
	cmp	w0, #0x09
	beq	token_scan_done
	cmp	w0, #0x0A
	beq	token_scan_done
	add	x25, x25, #1
	subs	x26, x26, #1
	b	token_scan
token_scan_done:
	sub	x27, x25, x20		// token length

	ldrb	w2, [x20]		// first char

	// single‑char operator?
	cmp	x27, #1
	bne	not_operator
	cmp	w2, #'+'
	beq	op_add
	cmp	w2, #'-'
	beq	op_sub
	cmp	w2, #'*'
	beq	op_mul
	cmp	w2, #'/'
	beq	op_div
	cmp	w2, #'%'
	beq	op_mod
	// not an operator
not_operator:
	// parse integer literal
	mov	x28, #0			// sign flag
	mov	x31, x20		// digit pointer
	cmp
