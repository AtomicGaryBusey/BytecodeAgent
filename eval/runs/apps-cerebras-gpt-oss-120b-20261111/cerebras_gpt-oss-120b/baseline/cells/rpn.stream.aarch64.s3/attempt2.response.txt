	.globl	_start
	.text
	.align	2

// System call numbers
.equ	SYS_READ,	63
.equ	SYS_WRITE,	64
.equ	SYS_EXIT,	93

// Constants
.equ	INPUT_MAX,	65536
.equ	STACK_MAX,	1024          // max values per line
.equ	OUT_BUF_SIZE,	32

// Data sections
	.section	.rodata
error_msg:
	.asciz	"error\n"
error_len = . - error_msg

	.section	.bss
	.align	3
input_buf:
	.zero	INPUT_MAX

stack:
	.zero	STACK_MAX*8

outbuf:
	.zero	OUT_BUF_SIZE

	.text
_start:
	// x19 = stack base
	adrp	x19, stack
	add	x19, x19, :lo12:stack

	// x20 = stack count (0)
	mov	x20, #0

	// x21 = input buffer base
	adrp	x21, input_buf
	add	x21, x21, :lo12:input_buf

	// x22 = total bytes read = 0
	mov	x22, #0

	// -------------------------------------------------
	// Read entire input into input_buf
read_loop:
	// x0 = fd = 0
	mov	x0, #0
	// x1 = buffer = input_buf + total_read
	add	x1, x21, x22
	// x2 = remaining space
	mov	x2, #INPUT_MAX
	sub	x2, x2, x22
	cmp	x2, #0
	beq	read_done          // should not happen (input too big)

	// syscall read
	mov	x8, #SYS_READ
	svc	#0
	// x0 = bytes read, 0 = EOF
	cbnz	x0, have_read
	b	read_done
have_read:
	add	x22, x22, x0
	b	read_loop
read_done:
	// x23 = end pointer = input_buf + total_len
	add	x23, x21, x22

	// -------------------------------------------------
	// Initialise outbuf end newline
	adrp	x24, outbuf
	add	x24, x24, :lo12:outbuf
	add	x25, x24, #OUT_BUF_SIZE-1   // x25 = outbuf + 31
	mov	w0, #0x0A                     // '\n'
	strb	w0, [x25]

	// -------------------------------------------------
	// Main line processing loop
	line_loop:
	// x26 = current line start pointer (initially x21)
	mov	x26, x21
line_loop_check:
	cmp	x26, x23
	bge	exit_prog                 // no more data

	// Find line end (newline or end)
	mov	x27, x26                  // x27 = line_end pointer (exclusive)

find_nl:
	cmp	x27, x23
	beq	line_end_found
	ldrb	w0, [x27]
	cmp	w0, #0x0A                // '\n'
	beq	line_end_found
	add	x27, x27, #1
	b	find_nl
line_end_found:
	// x27 now points to newline or end (exclusive)

	// Reset stack for this line
	mov	x20, #0
	// w28 = error flag = 0
	mov	w28, #0

	// x29 = token pointer = line start (x26)
	mov	x29, x26

token_loop:
	// if error flag set, skip to line_finish
	tnz	w28, token_finish
