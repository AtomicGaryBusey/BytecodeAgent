	.global _start
	.text
	.align 2

.equ BUF_SIZE, 65536

_start:
	// zero counters: lines(x22), words(x23), bytes(x24), in_word(x25)
	mov	x22, #0
	mov	x23, #0
	mov	x24, #0
	mov	x25, #0

	// buffer address in x26, size in x27
	adrp	x26, buffer
	add	x26, x26, :lo12:buffer
	mov	x27, #BUF_SIZE

read_loop:
	// read(fd=0, buf, count)
	mov	x8, #63
	mov	x0, #0
	mov	x1, x26
	mov	x2, x27
	svc	#0
	// result in x0 = nread
	cbnz	x0, process_input
	// nread == 0 => EOF
	b	done_output

process_input:
	// add nread to total bytes counter
	add	x24, x24, x0

	// x19 = ptr, x20 = remaining
	mov	x19, x26
	mov	x20, x0

process_loop:
	ldrb	w1, [x19], #1
	// newline?
	cmp	w1, #0x0A
	bne	not_newline
	add	x22, x22, #1
not_newline:
	// whitespace test
	cmp	w1, #0x20
	beq	whitespace
	cmp	w1, #0x09
	beq	whitespace
	cmp	w1, #0x0A
	beq	whitespace
	cmp	w1, #0x0B
	beq	whitespace
	cmp	w1, #0x0C
	beq	whitespace
	cmp	w1, #0x0D
	beq	whitespace
	// non‑whitespace
	cbnz	x25, continue_process
	add	x23, x23, #1
	mov	x25, #1
	b	continue_process
whitespace:
	mov	x25, #0
continue_process:
	subs	x20, x20, #1
	bne	process_loop

	// read more
	b	read_loop

done_output:
	// output buffer pointer in x28
	adrp	x28, outbuf
	add	x28, x28, :lo12:outbuf

	// lines
	mov	x0, x22
	mov	x1, x28
	bl	uint_to_ascii
	add	x28, x28, x0
	// space
	mov	w2, #' '
	strb	w2, [x28], #1

	// words
	mov	x0, x23
	mov	x1, x28
	bl	uint_to_ascii
	add	x28, x28, x0
	// space
	mov	w2, #' '
	strb	w2, [x28], #1

	// bytes
	mov	x0, x24
	mov	x1, x28
	bl	uint_to_ascii
	add	x28, x28, x0
	// newline
	mov	w2, #10
	strb	w2, [x28], #1

	// length = x28 - outbuf
	adrp	x0, outbuf
	add	x0, x0, :lo12:outbuf
	sub	x1, x28, x0		// x1 = length

	// write(fd=1, buf, len)
	mov	x8, #64
	mov	x0, #1
	mov	x2, x1
	svc	#0

	// exit(0)
	mov	x8, #93
	mov	x0, #0
	svc	#0

// ------------------------------------------------------------
// uint_to_ascii: x0 = value, x1 = dest pointer
// returns length in w0
// ------------------------------------------------------------
uint_to_ascii:
	// preserve original dest for length calculation (not needed)
	// zero length counter
	mov	x6, #0          // digit count

	cmp	x0, #0
	bne	.Lnonzero
	// value == 0
	mov	w2, #'0'
	strb	w2, [x1], #1
	mov	w0, #1
	ret

.Lnonzero:
	// save original sp in x2
	mov	x2, sp
	// divisor 10
	mov	x3, #10

.Ldigit_loop:
	udiv	x4, x0, x3          // quotient
	// remainder = x0 - x4*10
	mul	x5, x4, x3
	sub	x5, x0, x5          // remainder
	add	w5, w5, #'0'        // convert to ASCII
	// push onto stack
	sub	sp, sp, #1
	strb	w5, [sp]
	// increment digit count
	add	x6, x6, #1
	// next iteration
	mov	x0, x4
	cbnz	x0, .Ldigit_loop

	// pop digits to destination in correct order
.Lcopy_loop:
	ldrb	w5, [sp], #1
	strb	w5, [x1], #1
	cmp	sp, x2
	bne	.Lcopy_loop

	// length in w0
	mov	w0, w6
	ret

	.section .bss
	.align 4
buffer:
	.zero	BUF_SIZE

outbuf:
	.zero	64

	.end
