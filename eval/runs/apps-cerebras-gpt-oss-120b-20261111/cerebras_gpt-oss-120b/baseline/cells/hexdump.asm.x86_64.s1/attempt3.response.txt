.globl _start

.section .bss
.align 16
input_buf: .zero 65536
line_buf: .zero 80

.section .rodata
hex_nibble: .ascii "0123456789abcdef"

.text
_start:
    xor     rbx, rbx                # total_len = 0

read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd 0
    lea     rsi, [rip + input_buf]
    add     rsi, rbx                # buf + total_len
    mov     rdx, 65536
    sub     rdx, rbx                # remaining space
    syscall
    test    rax, rax
    je      read_done
    add     rbx, rax                # total_len += bytes_read
    cmp     rbx, 65536
    jae     read_done
    jmp     read_loop

read_done:
    xor     rcx, rcx                # offset = 0

dump_loop:
    cmp     rcx, rbx
    jae     dump_done

    mov     rdx, rbx
    sub     rdx, rcx                # remaining = total_len - offset
    mov     r8, 16
    cmp     rdx, r8
    cmova   rdx, r8                 # n = min(16, remaining)

    lea     r8, [rip + line_buf]    # r8 = line start
    lea     rdi, [rip + line_buf]   # rdi = dest pointer

    # ---- print offset (8 hex digits) ----
    mov     rax, rcx
    mov     r9, 8
offset_hex:
    mov     r10, rax
    shr     r10, 28
    and     r10, 0xF
    lea     r11, [rip + hex_nibble]
    mov     bl, byte ptr [r11 + r10]
    mov     byte ptr [rdi], bl
    inc     rdi
    shl     rax, 4
    dec     r9
    jne     offset_hex

    # two spaces after offset
    mov     byte ptr [rdi], ' '
    inc     rdi
    mov     byte ptr [rdi], ' '
    inc     rdi

    # source pointer for input bytes
    lea     rsi, [rip + input_buf]
    add     rsi, rcx                # rsi = input_buf + offset

    xor     r9, r9                  # i = 0
byte_loop:
    cmp     r9, 16
    je      after_bytes

    cmp     r9, rdx
    jb      have_byte
    # three spaces for missing byte
    mov     byte ptr [rdi], ' '
    inc     rdi
    mov     byte ptr [rdi], ' '
    inc     rdi
    mov     byte ptr [rdi], ' '
    inc     rdi
    jmp     inc_i

have_byte:
    mov     al, [rsi + r9]

    # high nibble
    mov     bl, al
    shr     bl, 4
    and     bl, 0xF
    lea     r10, [rip + hex_nibble]
    movzx   r11, bl
    mov     dl, byte ptr [r10 + r11]
    mov     byte ptr [rdi], dl
    inc     rdi

    # low nibble
    mov     bl, al
    and     bl, 0xF
    movzx   r11, bl
    mov     dl, byte ptr [r10 + r11]
    mov     byte ptr [rdi], dl
    inc     rdi

    # space after byte
    mov     byte ptr [rdi], ' '
    inc     rdi

inc_i:
    inc     r9
    cmp     r9, 8
    jne     byte_loop
    # extra space after eighth byte
    mov     byte ptr [rdi], ' '
    inc     rdi
    jmp     byte_loop

after_bytes:
    # space before ASCII column
    mov     byte ptr [rdi], ' '
    inc     rdi
    mov     byte ptr [rdi], '|'
    inc     rdi

    xor     r9, r9                  # ascii index
ascii_loop:
    cmp     r9, rdx
    je      after_ascii
    mov     al, [rsi + r9]
    cmp     al, 0x20
    jb      ascii_dot
    cmp     al, 0x7e
    ja      ascii_dot
    mov     byte ptr [rdi], al
    inc     rdi
    jmp     ascii_next
ascii_dot:
    mov     byte ptr [rdi], '.'
    inc     rdi
ascii_next:
    inc     r9
    jmp     ascii_loop

after_ascii:
    mov     byte ptr [rdi], '|'
    inc     rdi
    mov     byte ptr [rdi], 0x0a   # newline
    inc     rdi

    # ---- write the line ----
    mov     rdx, rdi
    sub     rdx, r8                 # length = rdi - line_buf
    mov     rax, 1                  # sys_write
    mov     rdi, 1                  # fd 1
    mov     rsi, r8                 # buf = line_buf
    syscall

    add     rcx, 16                 # offset += 16
    jmp     dump_loop

dump_done:
    # ---- final offset line ----
    lea     r8, [rip + line_buf]    # r8 = line start
    lea     rdi, [rip + line_buf]   # rdi = dest pointer

    mov     rax, rbx                # total_len
    mov     r9, 8
offset_final_hex:
    mov     r10, rax
    shr     r10, 28
    and     r10, 0xF
    lea     r11, [rip + hex_nibble]
    mov     bl, byte ptr [r11 + r10]
    mov     byte ptr [rdi], bl
    inc     rdi
    shl     rax, 4
    dec     r9
    jne     offset_final_hex

    mov     byte ptr [rdi], 0x0a   # newline
    inc     rdi

    # write final offset line
    mov     rdx, rdi
    sub     rdx, r8                 # length
    mov     rax, 1
    mov     rdi, 1
    mov     rsi, r8
    syscall

    mov     rax, 60                 # sys_exit
    xor     rdi, rdi
    syscall
