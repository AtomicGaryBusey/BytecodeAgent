.globl _start

.equ BUF_SIZE, 65536
.equ NUMBUF_SIZE, 32
.equ OUTBUF_SIZE, 96

.section .bss
.align 8
lines:         .zero 8
words:         .zero 8
total_bytes:   .zero 8
buf:           .zero BUF_SIZE
numbuf:        .zero NUMBUF_SIZE
outbuf:        .zero OUTBUF_SIZE

.section .text
_start:
    # previous character is whitespace
    mov     r8b, 1

read_loop:
    mov     rax, 0          # sys_read
    mov     rdi, 0          # fd = stdin
    lea     rsi, [buf]
    mov     rdx, BUF_SIZE
    syscall
    test    rax, rax
    je      finish_read     # EOF

    mov     rcx, rax        # remaining bytes
    lea     rbx, [buf]      # pointer to current byte

process_byte:
    test    rcx, rcx
    je      read_loop

    mov     al, [rbx]
    inc     rbx
    dec     rcx

    inc     qword [total_bytes]

    cmp     al, 0x0A
    jne     check_ws
    inc     qword [lines]
