.globl _start

.equ BUF_SIZE, 65536
.equ NUMBUF_SIZE, 24
.equ OUTBUF_SIZE, 64

.section .bss
.align 8
lines:          .zero 8
words:          .zero 8
total_bytes:    .zero 8
buf:            .zero BUF_SIZE
numbuf:         .zero NUMBUF_SIZE
outbuf:         .zero OUTBUF_SIZE

.section .text
_start:
    # start with previous char being whitespace
    mov     r8b, 1

read_loop:
    # syscall read(0, buf, BUF_SIZE)
    mov     rax, 0          # sys_read
    mov     rdi, 0          # fd = stdin
    lea     rsi, [buf]
    mov     rdx, BUF_SIZE
    syscall
    test    rax, rax
    je      output          # EOF
    mov     rcx, rax        # remaining count
    lea     rbx, [buf]      # pointer to current byte

process_byte:
    cmp     rcx, 0
    je      read_loop

    mov     al, byte ptr [rbx]
    inc     rbx
    dec     rcx

    # total_bytes++
    inc     qword ptr [total_bytes]

    # lines count (newline = 0x0A)
    cmp     al, 0x0A
    jne     check_ws
    inc     qword ptr [lines]

check_ws:
    # whitespace test
    cmp     al, 0x20
    je      is_ws
    cmp     al, 0x09
    je      is_ws
    cmp     al, 0x0A
    je      is_ws
    cmp     al, 0x0B
    je      is_ws
    cmp     al, 0x0C
    je      is_ws
    cmp     al, 0x0D
    je      is_ws
    # not whitespace
    cmp     r8b, 1
    jne     not_word_start
    inc     qword ptr [words]
not_word_start:
    mov     r8b, 0
    jmp     process_byte

is_ws:
    mov     r8b, 1
    jmp     process_byte

output:
    # rsi = outbuf, rdx = length accumulator
    lea     rsi, [outbuf]
    xor     rdx, rdx        # outlen = 0

    # ---- convert lines ----
    mov     rax, qword ptr [lines]
    lea     rdi, [numbuf + NUMBUF_SIZE]   # end of numbuf
    xor     rcx, rcx
    cmp     rax, 0
    jne     .conv_loop_lines
    mov     byte ptr [rdi-1], '0'
    lea     rdi, [rdi-1]
    inc     rcx
    jmp     .copy_lines
.conv_loop_lines:
    xor     rdx, rdx
    mov     rbx, 10
    div     rbx
    add     dl, '0'
    dec     rdi
    mov     byte ptr [rdi], dl
    inc     rcx
    cmp     rax, 0
    jne     .conv_loop_lines
.copy_lines:
    mov     r8, rcx          # length
.copy_loop_lines:
    cmp     r8, 0
    je      .after_copy_lines
    mov     al, byte ptr [rdi]
    mov     byte ptr [rsi], al
    inc     rdi
    inc     rsi
    dec     r8
    inc     rdx
    jmp     .copy_loop_lines
.after_copy_lines:
    mov     byte ptr [rsi], ' '
    inc     rsi
    inc     rdx

    # ---- convert words ----
    mov     rax, qword ptr [words]
    lea     rdi, [numbuf + NUMBUF_SIZE]
    xor     rcx, rcx
    cmp     rax, 0
    jne     .conv_loop_words
    mov     byte ptr [rdi-1], '0'
    lea     rdi, [rdi-1]
    inc     rcx
    jmp     .copy_words
.conv_loop_words:
    xor     rdx, rdx
    mov     rbx, 10
    div     rbx
    add     dl, '0'
    dec     rdi
    mov     byte ptr [rdi], dl
    inc     rcx
    cmp     rax, 0
    jne     .conv_loop_words
.copy_words:
    mov     r8, rcx
.copy_loop_words:
    cmp     r8, 0
    je      .after_copy_words
    mov     al, byte ptr [rdi]
    mov     byte ptr [rsi], al
    inc     rdi
    inc     rsi
    dec     r8
    inc     rdx
    jmp     .copy_loop_words
.after_copy_words:
    mov     byte ptr [rsi], ' '
    inc     rsi
    inc     rdx

    # ---- convert total_bytes ----
    mov     rax, qword ptr [total_bytes]
    lea     rdi, [numbuf + NUMBUF_SIZE]
    xor     rcx, rcx
    cmp     rax, 0
    jne     .conv_loop_bytes
    mov     byte ptr [rdi-1], '0'
    lea     rdi, [rdi-1]
    inc     rcx
    jmp     .copy_bytes
.conv_loop_bytes:
    xor     rdx, rdx
    mov     rbx, 10
    div     rbx
    add     dl, '0'
    dec     rdi
    mov     byte ptr [rdi], dl
    inc     rcx
    cmp     rax, 0
    jne     .conv_loop_bytes
.copy_bytes:
    mov     r8, rcx
.copy_loop_bytes:
    cmp     r8, 0
    je      .after_copy_bytes
    mov     al, byte ptr [rdi]
    mov     byte ptr [rsi], al
    inc     rdi
    inc     rsi
    dec     r8
    inc     rdx
    jmp     .copy_loop_bytes
.after_copy_bytes:
    mov     byte ptr [rsi], 0x0A   # newline
    inc     rsi
    inc     rdx

    # write output
    mov     rax, 1          # sys_write
    mov     rdi, 1          # fd = stdout
    lea     rsi, [outbuf]
    syscall

    # exit
    mov     rax, 60         # sys_exit
    xor     rdi, rdi
    syscall
