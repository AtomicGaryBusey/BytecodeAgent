.globl _start

.section .bss
.align 16
buf: .zero 65536

.section .rodata
newline: .byte 0x0A

.section .text
.globl _start
_start:
    # r15 = base address of buffer
    lea r15, [rip + buf]

    # rbx = total bytes read so far
    xor rbx, rbx
read_loop:
    mov rdx, 65536
    sub rdx, rbx                 # remaining space
    jle done_read                # no space left
    mov rdi, 0                   # fd = stdin
    lea rsi, [r15 + rbx]         # buf + offset
    xor rax, rax                 # sys_read = 0
    syscall
    test rax, rax
    jz done_read                 # EOF
    add rbx, rax                 # increase total length
    jmp read_loop
done_read:
    # rcx = current offset in buffer
    xor rcx, rcx

process_loop:
    cmp rcx, rbx
    jge exit_prog                # all bytes processed

    # find next newline or end of input
    mov rdx, rcx                 # rdx = scan position
find_loop:
    cmp rdx, rbx
    jge no_newline               # reached end of input
    mov al, byte ptr [r15 + rdx]
    cmp al, 0x0A
    je have_newline
    inc rdx
    jmp find_loop
have_newline:
    mov r10, 1                   # flag: newline present
    mov r9, rdx                  # r9 = position of newline
    sub r9, rcx                  # r9 = line length
    jmp after_scan
no_newline:
    xor r10, r10                 # flag: no newline
    mov r9, rbx                  # r9 = total length
    sub r9, rcx                  # r9 = line length (remaining)
    # continue to after_scan
after_scan:
    # write reversed line byte by byte
    mov r8, rcx
    add r8, r9
    dec r8                       # r8 = index of last byte of the line
rev_loop:
    cmp r8, rcx
    jl rev_done
    mov al, byte ptr [r15 + r8]
    mov byte ptr [rsp], al      # temporary buffer on stack
    mov rax, 1                   # sys_write
    mov rdi, 1                   # fd = stdout
    lea rsi, [rsp]
    mov rdx, 1
    syscall
    dec r8
    jmp rev_loop
rev_done:
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    # advance offset
    cmp r10, 1
    jne no_newline_adv
    add rcx, r9                  # skip line bytes
    inc rcx                      # skip newline byte
    jmp process_loop
no_newline_adv:
    add rcx, r9                  # consume remaining bytes
    jmp exit_prog

exit_prog:
    mov rax, 60                  # sys_exit
    xor rdi, rdi                 # status 0
    syscall
