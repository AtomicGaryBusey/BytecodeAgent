.global _start

.section .bss
.align 16
buf: .zero 65536

.section .rodata
newline: .byte 0x0A

.section .text
.global _start
_start:
    # read all input into buf
    xor rbx, rbx                # rbx = offset (bytes read so far)
read_loop:
    mov rdx, 65536
    sub rdx, rbx                # rdx = remaining space
    jle done_read               # if no space left, stop reading
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buf]
    add rsi, rbx                # rsi = buf + offset
    xor rax, rax                # sys_read = 0
    syscall
    test rax, rax
    jz done_read                # read returns 0 -> EOF
    add rbx, rax                # increase offset
    jmp read_loop
done_read:
    mov r8, rbx                 # r8 = total_len
    xor rbx, rbx                # rbx = current offset for processing

process_loop:
    cmp rbx, r8
    jge exit_prog               # all bytes processed

    lea rsi, [rip + buf]
    add rsi, rbx                # rsi = start pointer of current line
    xor rdx, rdx                # rdx = line_len = 0

scan_loop:
    mov rax, rbx
    add rax, rdx                # rax = offset + line_len
    cmp rax, r8
    jge no_newline              # reached end of input
    mov al, byte ptr [rsi + rdx]
    cmp al, 0x0A
    je have_newline
    inc rdx
    jmp scan_loop
have_newline:
    mov r9, 1                   # flag: line ended with newline
    jmp after_scan
no_newline:
    mov r9, 0                   # flag: final line without newline
    jmp after_scan
after_scan:
    # reverse line in place (length = rdx)
    xor rcx, rcx                # left index = 0
    mov r10, rdx
    dec r10                     # right index = len-1
rev_loop:
    cmp rcx, r10
    jge rev_done
    mov al, byte ptr [rsi + rcx]
    mov bl, byte ptr [rsi + r10]
    mov byte ptr [rsi + rcx], bl
    mov byte ptr [rsi + r10], al
    inc rcx
    dec r10
    jmp rev_loop
rev_done:
    # write reversed line
    mov rdi, 1                  # fd = stdout
    mov rdx, rdx                # length
    mov rax, 1                  # sys_write = 1
    syscall
    mov r12, rdx                # save line length

    # write newline
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    mov rax, 1
    syscall

    # advance offset
    cmp r9, 1
    jne no_newline_flag
    add rbx, r12                # skip line bytes
    inc rbx                     # skip the newline byte
    jmp process_loop
no_newline_flag:
    add rbx, r12                # consume remaining bytes
    jmp exit_prog

exit_prog:
    mov rdi, 0                  # exit status 0
    mov rax, 60                 # sys_exit = 60
    syscall
