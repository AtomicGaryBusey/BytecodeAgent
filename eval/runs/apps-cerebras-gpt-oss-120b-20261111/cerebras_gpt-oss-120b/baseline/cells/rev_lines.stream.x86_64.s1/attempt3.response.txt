.globl _start

.section .bss
.align 16
buf:    .zero 65536
outbuf: .zero 1

.section .rodata
newline: .byte 0x0A

.section .text
.globl _start
_start:
    # rsi = buffer base
    lea rsi, [rip + buf]

    # rbx = total bytes read so far
    xor rbx, rbx

read_loop:
    mov rdx, 65536
    sub rdx, rbx                 # remaining space
    jle done_read                # no space left
    mov rdi, 0                   # fd = stdin
    lea rdi, [rip + buf]         # not needed, keep rsi as base
    # keep rsi as base, compute address via rsi+rbx
    lea rdx, [rsi + rbx]         # rdx = buf + offset
    mov rdi, 0                   # fd = stdin
    mov rdx, 65536
    sub rdx, rbx
    mov rdi, 0
    mov rax, 0                   # sys_read
    mov rsi, rdx                 # will be overwritten, fix: need proper args:
    # Actually set args: rdi=0, rsi=buf+rbx, rdx=remaining
    lea rsi, [rsi + rbx]         # rsi = buf+offset
    mov rdx, 65536
    sub rdx, rbx
    syscall
    test rax, rax
    jz done_read
    add rbx, rax
    jmp read_loop

done_read:
    xor rcx, rcx                 # current offset = 0

process_loop:
    cmp rcx, rbx
    jge exit_prog                # all processed

    # rdx = scan position
    mov rdx, rcx

find_newline:
    cmp rdx, rbx
    je no_newline                # reached end
    mov al, byte ptr [rsi + rdx]
    cmp al, 0x0A
    je have_newline
    inc rdx
    jmp find_newline

have_newline:
    # line_len = rdx - rcx
    mov r9, rdx
    sub r9, rcx                  # r9 = line length
    # set line_start in r12
    mov r12, rcx
    # set last index in r8 = rdx - 1
    mov r8, rdx
    dec r8
    jmp rev_loop_start

no_newline:
    # line_len = rbx - rcx
    mov r9, rbx
    sub r9, rcx
    mov r12, rcx
    mov r8, rbx
    dec r8                       # last index = total-1
    # fall through to rev_loop_start

rev_loop_start:
    # reverse write loop
rev_loop:
    cmp r8, r12
    jl rev_done
    mov al, byte ptr [rsi + r8]
    mov byte ptr [rip + outbuf], al
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = stdout
    lea rsi, [rip + outbuf]
    mov rdx, 1
    syscall
    dec r8
    jmp rev_loop

rev_done:
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall

    # advance rcx
    cmp rdx, rbx                # rdx is scan position (at newline or end)
    je advance_end
    # newline case: rcx = rdx + 1
    mov rcx, rdx
    inc rcx
    jmp process_loop

advance_end:
    # no newline: consume remaining bytes and exit
    mov rcx, rbx
    jmp exit_prog

exit_prog:
    mov rax, 60                 # sys_exit
    xor rdi, rdi
    syscall
