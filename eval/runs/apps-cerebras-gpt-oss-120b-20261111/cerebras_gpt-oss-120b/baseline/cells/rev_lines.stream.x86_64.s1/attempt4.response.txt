.globl _start

.section .bss
.align 16
buf:    .zero 65536
outbuf: .zero 1

.section .rodata
newline: .byte 0x0A

.section .text
.globl _start
_start:
    xor     rbx, rbx                # total bytes read = 0
    lea     r12, [rip + buf]        # r12 = base of input buffer

read_loop:
    mov     rdx, 65536
    sub     rdx, rbx                # remaining space
    jle     read_done

    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [r12 + rbx]        # buf + offset
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                # update total
    jmp     read_loop

read_done:
    xor     rcx, rcx                # current offset = 0

process_loop:
    cmp     rcx, rbx
    jge     exit_prog

    mov     rdx, rcx                # scan position

find_nl:
    cmp     rdx, rbx
    je      no_newline
    mov     al, byte ptr [r12 + rdx]
    cmp     al, 0x0A
    je      have_newline
    inc     rdx
    jmp     find_nl

have_newline:
    mov     r9, rdx                 # newline position
    mov     r8, rdx
    dec     r8                      # last byte of the line

rev_loop:
    cmp     r8, rcx
    jl      rev_done
    mov     al, byte ptr [r12 + r8]
    mov     byte ptr [rip + outbuf], al
    mov     rax, 1                  # sys_write
    mov     rdi, 1                  # fd = stdout
    lea     rsi, [rip + outbuf]
    mov     rdx, 1
    syscall
    dec     r8
    jmp     rev_loop

rev_done:
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + newline]
    mov     rdx, 1
    syscall
    mov     rcx, r9
    inc     rcx                     # skip newline
    jmp     process_loop

no_newline:
    mov     r8, rbx
    dec     r8                      # last byte of final line

rev_loop2:
    cmp     r8, rcx
    jl      rev_done2
    mov     al, byte ptr [r12 + r8]
    mov     byte ptr [rip + outbuf], al
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + outbuf]
    mov     rdx, 1
    syscall
    dec     r8
    jmp     rev_loop2

rev_done2:
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + newline]
    mov     rdx, 1
    syscall
    jmp     exit_prog

exit_prog:
    mov     rax, 60                 # sys_exit
    xor     rdi, rdi
    syscall
