.globl _start
.text

# --------------------------------------------------------------
# Constants
# --------------------------------------------------------------
BUF_SIZE   = 65536
TAPE_SIZE  = 30000
STACK_MAX  = 1000          # bracket stack depth
# --------------------------------------------------------------

_start:
    #---------------------------------------------------------
    # r12 = buffer address
    # r13 = total bytes read
    #---------------------------------------------------------
    lea r12, [rip + buf]          # buffer pointer
    xor r13, r13                  # bytes read = 0

read_loop:
    mov rax, 0                    # sys_read
    xor rdi, rdi                  # fd = 0
    lea rsi, [r12 + r13]          # dst = buf + bytes_read
    mov rdx, BUF_SIZE
    sub rdx, r13                  # remaining space
    syscall
    test rax, rax
    je read_done
    add r13, rax                  # bytes_read += ret
    cmp r13, BUF_SIZE
    jae read_done
    jmp read_loop

read_done:
    #---------------------------------------------------------
    # Find first '!' separator
    # rcx = index while scanning
    # r14 = program length
    # r15 = input pointer
    # r10 = input remaining length
    #---------------------------------------------------------
    xor rcx, rcx
scan_sep:
    cmp rcx, r13
    jae no_excl
    mov al, byte ptr [r12 + rcx]
    cmp al, 0x21                  # '!'
    je have_excl
    inc rcx
    jmp scan_sep
have_excl:
    mov r14, rcx                 # program length = index
    mov r10, r13                 # total bytes
    sub r10, rcx                 # total - idx
    dec r10                      # input length = total - idx - 1
    lea r15, [r12 + rcx + 1]     # input pointer = buf + idx + 1
    jmp after_split
no_excl:
    mov r14, r13                 # program length = total
    xor r10, r10                 # input length = 0
    lea r15, [r12 + r13]         # input pointer (unused)
after_split:

    #---------------------------------------------------------
    # Bracket matching check and build jump table
    # rsi = address of bracket_stack
    # rdx = stack depth (uint32)
    # rcx = program counter (index)
    #---------------------------------------------------------
    lea rsi, [rip + bracket_stack]
    xor rdx, rdx                 # stack depth = 0
    xor rcx, rcx                 # pc = 0

bracket_loop:
    cmp rcx, r14
    jae bracket_done
    mov al, byte ptr [r12 + rcx]
    cmp al, '['
    je bracket_push
    cmp al, ']'
    je bracket_pop
    inc rcx
    jmp bracket_loop

bracket_push:
    mov [rsi + rdx*4], ecx       # push index (4-byte per entry)
    inc rdx
    inc rcx
    jmp bracket_loop

bracket_pop:
    dec rdx
    js mismatch                  # stack underflow -> unmatched ]
    mov ebx, [rsi + rdx*4]       # matching '[' index
    # store forward jump ( '[' -> ']' )
    mov [rip + jump_table + rbx*4], rcx
