.globl _start

.section .bss
.align 8
buffer:          .zero 32768
ptrs:           .zero 1000*8
lens:           .zero 1000*8
line_cnt:       .zero 8
input_len:      .zero 8

.section .rodata
newline:        .byte 0x0A

.section .text
_start:
    # read all input
read_loop:
    mov rax, 0                 # sys_read
    mov rdi, 0                 # fd 0
    lea rsi, [rip + buffer]
    mov rdx, 32768
    syscall
    test rax, rax
    je parse_input
    # add to input_len
    mov rbx, rax
    lea rcx, [rip + input_len]
    mov rdx, [rcx]
    add rdx, rbx
    mov [rcx], rdx
    jmp read_loop

parse_input:
    # parse lines
    xor rcx, rcx               # i = 0 offset
    xor r8, r8                 # start_offset = 0
    lea rbx, [rip + buffer]    # base pointer
    mov rdx, [rip + input_len] # total length
parse_loop:
    cmp rcx, rdx
    je parse_end
    mov al, byte ptr [rbx + rcx]
    cmp al, 0x0A
    jne inc_offset

    mov rax, rcx
    sub rax, r8                # length = offset - start
