.globl _start

.section .bss
.align 8
buffer:          .zero 32768
ptrs:           .zero 1000*8
lens:           .zero 1000*8
line_cnt:       .zero 8
input_len:      .zero 8

.section .rodata
newline:        .byte 0x0A

.section .text
_start:
    # read all input into buffer
    xor rbx, rbx                # rbx = total bytes read
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + buffer]     # base address
    add rsi, rbx                # offset = bytes already read
    mov rdx, 32768
    sub rdx, rbx                # remaining capacity
    test rdx, rdx
    jz parse_input              # buffer full
    syscall
    test rax, rax
    je parse_input
    add rbx, rax                # accumulate length
    jmp read_loop

parse_input:
    mov [rip + input_len], rbx
    # parse lines
    xor r9, r9                  # line count = 0
    xor rcx, rcx                # i = 0
    xor r8, r8                  # start_offset = 0
    lea rbx, [rip + buffer]     # base pointer
    mov rdx, rbx
    mov rdx, [rip + input_len]  # total length
parse_loop:
    cmp rcx, rdx
    jae parse_done
    mov al, byte ptr [rbx + rcx]
    cmp al, 0x0A
    jne inc_i
    # newline found, store line
    mov rax, rcx
    sub rax, r8                 # length = i - start
    mov rsi, r8                 # start_offset
    add rsi, rbx                # pointer = base + start
    # store pointer
    lea rdi, [rip + ptrs]
    mov r10, r9
    shl r10, 3                  # r10 = line_idx*8
    add rdi, r10
    mov [rdi], rsi
    # store length
    lea rdi, [rip + lens]
    add rdi, r10
    mov [rdi], rax
    inc r9                      # line count++
    mov r8, rcx
    inc r8                      # start = i+1
inc_i:
    inc rcx
    jmp parse_loop

parse_done:
    cmp r8, rdx
    jae store_cnt
    # final line without trailing newline
    mov rax, rdx
    sub rax, r8                 # length = total - start
    mov rsi, r8
    add rsi, rbx                # pointer
    lea rdi, [rip + ptrs]
    mov r10, r9
    shl r10, 3
    add rdi, r10
    mov [rdi], rsi
    lea rdi, [rip + lens]
    add rdi, r10
    mov [rdi], rax
    inc r9
store_cnt:
    mov [rip + line_cnt], r9

    # sort lines (bubble sort)
    mov r13, [rip + line_cnt]   # r13 = line_cnt
    cmp r13, 2
    jl sort_done
outer_sort:
    xor r14, r14                # swapped = 0
    xor r15, r15                # i = 0
inner_loop:
    mov rax, r13
    dec rax                     # line_cnt-1
    cmp r15, rax
    jge after_inner
    # addresses of line i
    mov rax, r15
    shl rax, 3
    lea rdi, [rip + ptrs]       # base ptrs
    add rdi, rax                # addr_i_ptr
    mov r8, [rdi]               # ptr_i
    lea rsi, [rip + lens]
    add rsi, rax                # addr_i_len
    mov r9, [rsi]               # len_i
    # addresses of line i+1
    mov rbx, r15
    inc rbx
    shl rbx, 3
    lea rdx, [rip + ptrs]       # base ptrs
    add rdx, rbx                # addr_next_ptr
    mov r10, [rdx]              # ptr_next
    lea rcx, [rip + lens]
    add rcx, rbx                # addr_next_len
    mov r11, [rcx]              # len_next
    # compare line i (r8,len r9) with line i+1 (r10,len r11)
    mov rdi, r8                 # ptr1
    mov rcx, r9                 # len1
    mov rsi, r10                # ptr2
    mov rdx, r11                # len2
    call compare_lines
    test rax, rax
    je no_swap
    # swap pointers
    mov [rdi], r10              # *addr_i_ptr = ptr_next
    mov [rdx], r8               # *addr_next_ptr = ptr_i
    # swap lengths
    mov [rsi], r11              # *addr_i_len = len_next
    mov [rcx], r9               # *addr_next_len = len_i
    inc r14                     # swapped = 1
no_swap:
    inc r15
    jmp inner_loop
after_inner:
    test r14, r14
    jnz outer_sort
sort_done:

    # output sorted lines
    mov r13, [rip + line_cnt]   # line count
    xor r15, r15                # index = 0
out_loop:
    cmp r15, r13
    jge out_done
    mov rax, r15
    shl rax, 3
    lea rdi, [rip + ptrs]
    add rdi, rax
    mov rsi, [rdi]              # ptr to line
    lea rdx, [rip + lens]
    add rdx, rax
    mov rcx, [rdx]              # length
    # write line content (handle partial writes)
write_loop:
    test rcx, rcx
    jz write_newline
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd 1
    mov rdx, rcx
    syscall
    test rax, rax
    jz write_newline
    sub rcx, rax
    add rsi, rax
    jmp write_loop
write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r15
    jmp out_loop
out_done:
    # exit
    mov rax, 60
    xor rdi, rdi
    syscall

# int compare_lines(const char *p1, size_t len1, const char *p2, size_t len2)
# returns rax = 1 if line1 > line2, else 0
compare_lines:
    # rdi = p1, rcx = len1, rsi = p2, rdx = len2
    mov r8, rcx                # r8 = len1
    cmp rcx, rdx
    cmova r8, rdx              # r8 = min(len1,len2)
    xor r9, r9                 # index = 0
cmp_byte_loop:
    cmp r9, r8
    jae after_cmp_bytes
    mov al, byte ptr [rdi + r9]
    mov bl, byte ptr [rsi + r9]
    cmp al, bl
    ja greater
    jb less
    inc r9
    jmp cmp_byte_loop
greater:
    mov rax, 1
    ret
less:
    xor rax, rax
    ret
after_cmp_bytes:
    cmp rcx, rdx
    ja greater
    xor rax, rax
    ret
