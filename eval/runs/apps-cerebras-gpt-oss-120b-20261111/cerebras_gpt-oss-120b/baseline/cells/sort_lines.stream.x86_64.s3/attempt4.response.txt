.globl _start

.section .bss
.align 8
buffer:          .zero 32768
ptrs:            .zero 1000*8
lens:            .zero 1000*8
line_cnt:        .zero 8
input_len:       .zero 8

.section .rodata
newline:         .byte 0x0A

.section .text
_start:
    # read all input into buffer
    xor rbx, rbx                # total bytes read
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + buffer]
    add rsi, rbx                # offset
    mov rdx, 32768
    sub rdx, rbx                # remaining space
    test rdx, rdx
    jz parse_input
    syscall
    test rax, rax
    je parse_input
    add rbx, rax
    jmp read_loop

parse_input:
    mov [rip + input_len], rbx
    xor r9, r9                  # line count
    xor rcx, rcx                # i = 0
    xor r8, r8                  # start_offset = 0
    lea rbx, [rip + buffer]     # base pointer
    mov rdx, [rip + input_len]

parse_loop:
    cmp rcx, rdx
    jae parse_done
    mov al, byte ptr [rbx + rcx]
    cmp al, 0x0A
    jne inc_i
    # newline found
    mov rax, rcx
    sub rax, r8                 # length = i - start
    mov rsi, r8
    add rsi, rbx                # pointer = base+start
    # store pointer
    lea rdi, [rip + ptrs]
    mov r10, r9
    shl r10, 3
    add rdi, r10
    mov [rdi], rsi
    # store length
    lea rdi, [rip + lens]
    add rdi, r10
    mov [rdi], rax
    inc r9
    mov r8, rcx
    inc r8
inc_i:
    inc rcx
    jmp parse_loop

parse_done:
    cmp r8, rdx
    jae store_cnt
    # final line without trailing newline
    mov rax, rdx
    sub rax, r8
    mov rsi, r8
    add rsi, rbx
    lea rdi, [rip + ptrs]
    mov r10, r9
    shl r10, 3
    add rdi, r10
    mov [rdi], rsi
    lea rdi, [rip + lens]
    add rdi, r10
    mov [rdi], rax
    inc r9
store_cnt:
    mov [rip + line_cnt], r9

    # sort lines (bubble sort)
    mov r13, [rip + line_cnt]   # r13 = line_cnt
    cmp r13, 2
    jl sort_done
outer_sort:
    xor r14, r14                # swapped = 0
    xor r15, r15                # i = 0
inner_loop:
    mov rax, r13
    dec rax                     # line_cnt-1
    cmp r15, rax
    jge after_inner

    mov rax, r15
    shl rax, 3
    lea rdi, [rip + ptrs]
    add rdi, rax                # addr_i_ptr
    mov r8, [rdi]               # ptr_i
    lea rsi, [rip + lens]
    add rsi, rax                # addr_i_len
    mov r9, [rsi]               # len_i

    mov rbx, r15
    inc rbx
    shl rbx, 3
    lea rdx, [rip + ptrs]
    add rdx, rbx                # addr_next_ptr
    mov r10, [rdx]              # ptr_next
    lea rcx, [rip + lens]
    add rcx, rbx                # addr_next_len
    mov r11, [rcx]              # len_next

    # compare lines
    mov rax, r9
    cmp rax, r11
    cmova rax, r11              # rax = minlen
    xor rbx, rbx                # idx = 0
cmp_byte_loop:
    cmp rbx, rax
    jae cmp_done
    mov al, byte ptr [r8 + rbx]
    mov dl, byte ptr [r10 + rbx]
    cmp al, dl
    ja line_greater
    jb line_less
    inc rbx
    jmp cmp_byte_loop
line_greater:
    mov r12, 1                  # need_swap = 1
    jmp after_cmp
line_less:
    xor r12, r12                # need_swap = 0
    jmp after_cmp
cmp_done:
    cmp r9, r11
    ja line_greater
    xor r12, r12                # need_swap = 0
after_cmp:
    test r12, r12
    je no_swap
    # swap pointers
    mov [rdi], r10
    mov [rdx], r8
    # swap lengths
    mov [rsi], r11
    mov [rcx], r9
    inc r14                     # swapped = 1
no_swap:
    inc r15
    jmp inner_loop
after_inner:
    test r14, r14
    jnz outer_sort
sort_done:

    # output sorted lines
    mov r13, [rip + line_cnt]
    xor r15, r15                # index = 0
out_loop:
    cmp r15, r13
    jge out_done
    mov rax, r15
    shl rax, 3
    lea rdi, [rip + ptrs]
    add rdi, rax
    mov rsi, [rdi]              # ptr to line
    lea rdx, [rip + lens]
    add rdx, rax
    mov rcx, [rdx]              # length

write_loop:
    test rcx, rcx
    jz write_newline
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd 1
    mov rdx, rcx
    syscall
    test rax, rax
    jz write_newline
    sub rcx, rax
    add rsi, rax
    jmp write_loop

write_newline:
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc r15
    jmp out_loop
out_done:
    mov rax, 60
    xor rdi, rdi
    syscall
