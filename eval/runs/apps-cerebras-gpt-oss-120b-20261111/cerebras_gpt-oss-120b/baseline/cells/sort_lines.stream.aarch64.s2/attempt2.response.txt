.equ    MAX_INPUT, 32768
.equ    MAX_LINES, 1000

.section .bss
    .align 8
buffer:         .space  MAX_INPUT
line_offs:      .space  (8*MAX_LINES)   // uint64 offsets
line_len:       .space  (4*MAX_LINES)   // uint32 lengths

.section .rodata
    .align 1
newline_char:   .byte   0x0A

.section .text
    .global _start
_start:
    // -------------------------------------------------
    // read all input into buffer
    // -------------------------------------------------
    mov     x0, #0                // fd = stdin
    adr     x5, buffer            // x5 = buffer base address
    mov     x1, x5                // buf
    mov     x2, #MAX_INPUT
    mov     x8, #63               // sys_read
    svc     #0
    mov     x9, x0                // total_len = bytes read
    cbz     x9, exit_ok           // empty input -> exit

read_loop:
    mov     x0, #0
    add     x1, x5, x9            // buf + total_len
    mov     x2, #MAX_INPUT
    sub     x2, x2, x9            // remaining space
    cbz     x2, read_done
    mov     x8, #63
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0            // total_len += bytes_read
    b       read_loop
read_done:
    // -------------------------------------------------
    // parse lines, store offsets and lengths
    // -------------------------------------------------
    mov     x10, #0               // cur_offset = 0
    mov     x11, #0               // line_count = 0

parse_loop:
    cmp     x10, x9
    bge     parse_done

    mov     x12, x10              // line_start = cur_offset

find_nl:
    cmp     x10, x9
    bge     end_of_buf
    add     x13, x5, x10          // &buffer[cur_offset]
    ldrb    w14, [x13]
    cmp     w14, #0x0A
    beq     nl_found
    add     x10, x10, #1
    b       find_nl

nl_found:
    sub     x15, x10, x12         // len = cur_offset - line_start
    // store offset
    mov     x16, x11
    lsl     x16, x16, #3          // *8
    adr     x17, line_offs
    add     x17, x17, x16
    str     x12, [x17]
    // store length
    mov     x18, x11
    lsl     x18, x18, #2          // *4
    adr     x19, line_len
    add     x19, x19, x18
    str     w15, [x19]
    // inc line count
    add     x11, x11, #1
    // skip newline
    add     x10, x10, #1
    b       parse_loop

end_of_buf:
    // final line (may be zero length)
    sub     x15, x10, x12
    mov     x16, x11
    lsl     x16, x16, #3
    adr     x17, line_offs
    add     x17, x17, x16
    str     x12, [x17]
    mov     x18, x11
    lsl     x18, x18, #2
    adr     x19, line_len
    add     x19, x19, x18
    str     w15, [x19]
    add     x11, x11, #1

parse_done:
    cmp     x11, #0
    beq     exit_ok

    // -------------------------------------------------
    // sort lines with bubble sort (ascending)
    // -------------------------------------------------
    mov     x20, #0               // i = 0

outer_loop:
    sub     x21, x11, #1
    cmp     x20, x21
    bge     sort_done

    mov     x22, x11
    sub     x22, x22, #1          // j = line_count-1

inner_loop:
    cmp     x22, x20
    ble     inc_i                 // if j <= i, done inner

    // indices a = j-1, b = j
    sub     x23, x22, #1          // a index
    // load offset a
    mov     x24, x23
    lsl     x24, x24, #3
    adr     x25, line_offs
    add     x25, x25, x24
    ldr     x26, [x25]            // off_a

    // load len a
    mov     x27, x23
    lsl     x27, x27, #2
    adr     x28, line_len
    add     x28, x28, x27
    ldr     w29, [x28]            // len_a

    // load offset b
    mov     x30, x22
    lsl     x30, x30, #3
    adr     x31, line_offs
    add     x31, x
