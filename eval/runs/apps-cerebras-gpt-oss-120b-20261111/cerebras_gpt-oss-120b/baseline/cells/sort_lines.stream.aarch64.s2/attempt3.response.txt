.equ    MAX_INPUT, 32768
.equ    MAX_LINES, 1000
.equ    SYSCALL_READ, 63
.equ    SYSCALL_WRITE, 64
.equ    SYSCALL_EXIT, 93
.equ    FD_STDIN, 0
.equ    FD_STDOUT, 1

.section .bss
    .align 8
buffer:         .space  MAX_INPUT
line_offs:      .space  (8*MAX_LINES)   // uint64 offsets
line_len:       .space  (4*MAX_LINES)   // uint32 lengths

.section .rodata
    .align 1
newline_char:   .byte   0x0A

.section .text
    .global _start
_start:
    // -------------------------------------------------
    // read all input into buffer
    // -------------------------------------------------
    adr     x5, buffer                // x5 = buffer base
    mov     x0, #FD_STDIN            // fd = stdin
    mov     x1, x5                   // buf
    mov     x2, #MAX_INPUT
    mov     x8, #SYSCALL_READ
    svc     #0
    mov     x9, x0                   // total_len = bytes read
    cbz     x9, exit_ok              // empty input -> exit

read_loop:
    mov     x0, #FD_STDIN
    add     x1, x5, x9               // buf + total_len
    mov     x2, #MAX_INPUT
    sub     x2, x2, x9               // remaining space
    cbz     x2, read_done
    mov     x8, #SYSCALL_READ
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0               // total_len += bytes_read
    b       read_loop

read_done:
    // -------------------------------------------------
    // parse lines, store offsets and lengths
    // -------------------------------------------------
    mov     x10, #0                  // cur_offset = 0
    mov     x11, #0                  // line_count = 0

parse_loop:
    cmp     x10, x9
    bge     parse_done

    mov     x12, x10                 // line_start = cur_offset

find_nl:
    cmp     x10, x9
    bge     end_of_buf
    add     x13, x5, x10
    ldrb    w14, [x13]
    cmp     w14, #0x0A
    beq     nl_found
    add     x10, x10, #1
    b       find_nl

nl_found:
    sub     x15, x10, x12           // len = cur_offset - line_start
    // store offset
    mov     x16, x11
    lsl     x16, x16, #3            // *8
    adr     x17, line_offs
    add     x17, x17, x16
    str     x12, [x17]
    // store length
    mov     x18, x11
    lsl     x18, x18, #2            // *4
    adr     x19, line_len
    add     x19, x19, x18
    str     w15, [x19]
    // inc line count
    add     x11, x11, #1
    // skip newline
    add     x10, x10, #1
    b       parse_loop

end_of_buf:
    // final line (may be zero length)
    sub     x15, x10, x12
    mov     x16, x11
    lsl     x16, x16, #3
    adr     x17, line_offs
    add     x17, x17, x16
    str     x12, [x17]
    mov     x18, x11
    lsl     x18, x18, #2
    adr     x19, line_len
    add     x19, x19, x18
    str     w15, [x19]
    add     x11, x11, #1

parse_done:
    cbz     x11, exit_ok            // no lines -> exit

    // -------------------------------------------------
    // sort lines with insertion sort (ascending)
    // -------------------------------------------------
    mov     x20, #0                 // i = 0

outer_sort:
    cmp     x20, x11
    bge     sort_done               // i >= line_count

    mov     x21, x20                // key = i

inner_sort:
    cmp     x21, #0
    beq     next_i

    // a = key-1, b = key
    sub     x22, x21, #1            // a index
    // load offset a
    mov     x23, x22
    lsl     x23, x23, #3
    adr     x24, line_offs
    add     x24, x24, x23
    ldr     x25, [x24]              // off_a

    // load length a
    mov     x26, x22
    lsl     x26, x26, #2
    adr     x27, line_len
    add     x27, x27, x26
    ldr     w28, [x27]              // len_a

    lsl     x29, x29, #3
