.equ MAX_INPUT, 32768
.equ MAX_LINES, 1000

.section .bss
.align 8
buffer:          .space MAX_INPUT
line_offs:       .space (8*MAX_LINES)
line_len:        .space (4*MAX_LINES)

.section .rodata
.align 1
newline_char:    .byte 0x0A

.section .text
.global _start
_start:
    // ---------- read entire input ----------
    mov     x0, #0                      // fd = stdin
    adr     x1, buffer
    mov     x2, #MAX_INPUT
    mov     x8, #63                     // sys_read
    svc     #0
    cbz     x0, no_input                // no bytes read -> empty input
    mov     x9, x0                      // total_read = bytes read

read_loop:
    mov     x0, #0
    adr     x1, buffer
    add     x1, x1, x9                  // buf + total_read
    mov     x2, #MAX_INPUT
    sub     x2, x2, x9                  // remaining space
    cbz     x2, read_done
    mov     x8, #63
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0                  // total_read += bytes_read
    b       read_loop
read_done:
    // x9 = total input size

    // ---------- parse lines ----------
    mov     x10, #0                     // ptr offset
    mov     x11, #0                     // line count

parse_loop:
    cmp     x10, x9
    bge     parse_done
    mov     x12, x10                    // line start offset
find_nl:
    cmp     x10, x9
    bge     end_of_buf
    adr     x13, buffer
    add     x13, x13, x10
    ldrb    w14, [x13]
    cmp     w14, #0x0A
    beq     nl_found
    add     x10, x10, #1
    b       find_nl
nl_found:
    sub     x15, x10, x12               // line length
    // store offset
    mov     x16, x11
    lsl     x16, x16, #3                // *8
    adr     x17, line_offs
    add     x17, x17, x16
    str     x12, [x17]
    // store length (4-byte)
    mov     x18, x11
    lsl     x18, x18, #2                // *4
    adr     x19, line_len
    add     x19, x19, x18
    str     w15, [x19]
    // increment line count
    add     x11, x11, #1
    // skip newline
    add     x10, x10, #1
    b       parse_loop
end_of_buf:
    // store final line (no terminating newline)
    sub     x15, x10, x12               // length (may be 0)
    mov     x16, x11
    lsl     x16, x16, #3
    adr     x17, line_offs
    add     x17, x17, x16
    str     x12, [x17]
    mov     x18, x11
    lsl     x18, x18, #2
    adr     x19, line_len
    add     x19, x19, x18
    str     w15, [x19]
    add     x11, x11, #1
parse_done:
    // if no lines, exit
    cmp     x11, #0
    beq     exit_ok

    // ---------- sort lines (insertion sort) ----------
    mov     x20, #1                     // i = 1
sort_outer:
    cmp     x20, x11
    bge     sort_done
    mov     x21, x20                    // j = i
sort_inner:
    cmp     x21, #0
    beq     inc_i
    // idx_a = j-1
    sub     x22, x21, #1
    // load offset_a
    mov     x23, x22
    lsl     x23, x23, #3
    adr     x24, line_offs
    add     x24, x24, x23
    ldr     x25, [x24]                 // off_a
    // load len_a
    mov     x26, x22
    lsl     x26, x26, #2
    adr     x27, line_len
    add     x27, x27, x26
    ldr     w28, [x27]                 // len_a
    // load offset_b (j)
    mov     x29, x21
    lsl     x29, x29, #3
    adr     x30, line_offs
    add     x30, x30, x29
    ldr     x31, [x30]                 // off_b
    // load len_b
    mov     x0, x21
    l
