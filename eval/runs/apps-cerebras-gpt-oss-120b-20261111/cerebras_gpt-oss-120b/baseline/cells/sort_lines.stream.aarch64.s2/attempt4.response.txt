.equ    MAX_INPUT, 32768
.equ    MAX_LINES, 1000
.equ    SYSCALL_READ, 63
.equ    SYSCALL_WRITE, 64
.equ    SYSCALL_EXIT, 93
.equ    FD_STDIN, 0
.equ    FD_STDOUT, 1
.equ    NL_CHAR, 0x0A

.section .bss
    .align 8
buffer:         .space  MAX_INPUT
line_offs:      .space  (8*MAX_LINES)       // uint64 offsets
line_len:       .space  (4*MAX_LINES)       // uint32 lengths

.section .rodata
    .align 1
newline_char:  .byte   NL_CHAR

.section .text
    .global _start
_start:
    // -------------------------------------------------
    // Read all input into buffer
    // -------------------------------------------------
    adr     x5, buffer                // x5 = buffer base
    mov     x0, #FD_STDIN
    mov     x1, x5
    mov     x2, #MAX_INPUT
    mov     x8, #SYSCALL_READ
    svc     #0
    mov     x9, x0                    // x9 = total_len
    cbz     x9, exit_ok               // empty input -> exit

read_loop:
    // compute remaining space
    mov     x0, #FD_STDIN
    add     x1, x5, x9               // buf + total_len
    mov     x2, #MAX_INPUT
    sub     x2, x2, x9               // remaining
    cbz     x2, read_done
    mov     x8, #SYSCALL_READ
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0               // total_len += bytes_read
    b       read_loop

read_done:
    // -------------------------------------------------
    // Parse lines, store offsets and lengths
    // -------------------------------------------------
    adr     x6, line_offs            // x6 = line_offs base
    adr     x7, line_len             // x7 = line_len base
    mov     x10, #0                  // cur_offset = 0
    mov     x11, #0                  // line_count = 0

parse_loop:
    cmp     x10, x9
    bge     parse_done               // done when cur_offset >= total_len

    mov     x12, x10                 // line_start = cur_offset

find_nl:
    cmp     x10, x9
    bge     final_line               // reached end without newline
    add     x13, x5, x10
    ldrb    w14, [x13]
    cmp     w14, #0x0A
    beq     nl_found
    add     x10, x10, #1
    b       find_nl

nl_found:
    sub     x15, x10, x12           // length = cur_offset - line_start

    // store offset
    mov     x16, x11
    lsl     x16, x16, #3            // *8
    add     x17, x6, x16
    str     x12, [x17]

    // store length
    mov     x18, x11
    lsl     x18, x18, #2            // *4
    add     x19, x7, x18
    str     w15, [x19]

    // inc line count
    add     x11, x11, #1

    // skip newline
    add     x10, x10, #1
    b       parse_loop

final_line:
    // handle final line (may be zero length)
    sub     x15, x10, x12           // length = cur_offset - line_start
    // store offset
    mov     x16, x11
    lsl     x16, x16, #3
    add     x17, x6, x16
    str     x12, [x17]
    // store length
    mov     x18, x11
    lsl     x18, x18, #2
    add     x19, x7, x18
    str     w15, [x19]
    add     x11, x11, #1

parse_done:
    cbz     x11, exit_ok            // no lines -> exit

    // -------------------------------------------------
    // Insertion sort of lines (ascending)
    // -------------------------------------------------
    mov     x20, #1                 // i = 1

sort_outer:
    cmp     x20, x11
    bge     sort_done               // i >= line_count

    // load key offset and length
    mov     x21, x20                // key_index = i
    mov     x22, x21
    lsl     x22, x22, #3            // *8
    add     x23, x6, x22
    ldr     x24, [x23]              // key_offset
    mov     x25, x21
    lsl     x25, x25, #2            // *4
    add     x26, x7, x25
    ldr     w27, [x26]              // key_len (w27)

    // j = i
    mov     x28, x21                // j

    beq     place_key
