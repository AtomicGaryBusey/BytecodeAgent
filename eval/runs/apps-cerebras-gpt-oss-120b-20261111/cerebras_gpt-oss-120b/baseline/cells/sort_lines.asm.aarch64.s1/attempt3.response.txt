.text
.global _start
_start:
    // ---------- read all input ----------
    adrp    x9, input_buf            // base of input buffer (page aligned)
    add     x9, x9, :lo12:input_buf
    mov     x10, #0                  // total bytes read

read_loop:
    // remaining = 32768 - total
    mov     x11, #32768
    sub     x11, x11, x10
    cbz     x11, read_done

    mov     x0, #0                   // fd = stdin
    add     x1, x9, x10              // buf = input_buf + total
    mov     x2, x11                  // count = remaining
    mov     x8, #63                  // sys_read
    svc     #0
    cbz     x0, read_done            // EOF
    add     x10, x10, x0             // total += bytes_read
    b       read_loop

read_done:
    // store total length
    adrp    x12, total_len
    add     x12, x12, :lo12:total_len
    str     x10, [x12]

    // ---------- parse lines ----------
    // line_count = 0
    adrp    x13, line_cnt
    add     x13, x13, :lo12:line_cnt
    mov     x0, #0
    str     x0, [x13]

    // cur = input_buf
    mov     x14, x9
    // end = input_buf + total_len
    ldr     x15, [x12]               // total_len
    add     x16, x9, x15             // end address

parse_loop:
    cmp     x14, x16
    bge     parse_done

    // line_start = cur
    mov     x17, x14                 // save start pointer
    mov     x18, #0                  // length = 0

scan_loop:
    cmp     x14, x16
    bge     scan_end
    ldrb    w19, [x14], #1           // load byte, advance cur
    cmp     w19, #0x0A
    beq     scan_end
    add     x18, x18, #1             // length++
    b       scan_loop

scan_end:
    // store pointer
    adrp    x20, line_ptrs
    add     x20, x20, :lo12:line_ptrs
    ldr     x21, [x13]               // line_cnt
    lsl     x22, x21, #3             // offset = cnt*8
    add     x23, x20, x22
    str     x17, [x23]

    // store length
    adrp    x24, line_lens
    add     x24, x24, :lo12:line_lens
    lsl     x25, x21, #2             // offset = cnt*4
    add     x26, x24, x25
    str     w18, [x26]

    // line_cnt++
    add     x21, x21, #1
    str     x21, [x13]

    b       parse_loop

parse_done:
    // ---------- sort lines (selection sort) ----------
    ldr     x27, [x13]               // line_cnt
    subs    x28, x27, #2
    blt     sort_done

    mov     x29, #0                  // i = 0

outer_loop:
    ldr     x27, [x13]               // line_cnt
    sub     x30, x27, #1
    cmp     x29, x30
    bge     sort_done

    mov     x31, x29                 // min = i
    add     x5, x29, #1              // j = i+1

inner_loop:
    ldr     x27, [x13]               // line_cnt
    cmp     x5, x27
    bge     inner_done

    // load line j pointer and length
    adrp    x6, line_ptrs
    add     x6, x6, :lo12:line_ptrs
    lsl     x7, x5, #3
    add     x8, x6, x7
    ldr     x9, [x8]                 // ptr_j
    adrp    x10, line_lens
    add     x10, x10, :lo12:line_lens
    lsl     x11, x5, #2
    add     x12, x10, x11
    ldr     w13, [x12]               // len_j

    // load line min pointer and length
    adrp    x14, line_ptrs
    add     x14, x14, :lo12:line_ptrs
    lsl     x15, x31, #3
    add     x16, x14, x15
    ldr     x17, [x16]               // ptr_min
    adrp    x18, line_lens
    add     x18, x18, :lo12:line_lens
    lsl     x19, x31, #2
    add     x20, x18, x19
    ldr     w21, [x20]               // len_min

    // compare j vs min, result in w22 (1 if j < min)
    mov     x23, x9                  // ptr_a = ptr_j
    mov     x24, x17                 // ptr_b = ptr_min
    mov     w25, w13                 // len_a
    mov     w26, w21                 // len_b
    mov     w27, #0                  // index
    mov     w22, #0                  // result flag

cmp_loop:
    cmp     w27, w25
    bcs     len_check_a
    cmp     w27, w26
    bcs     len_check_a
    ldrb    w28, [x23, w27]
    ldrb    w29, [x24, w27]
    cmp     w28, w29
    bne     diff_found
    add     w27, w27, #1
    b       cmp_loop

diff_found:
    cmp     w28, w29
    bhs     not_less
    mov     w22, #1
    b       after_cmp

not_less:
    // w22 stays 0
    b       after_cmp

len_check_a:
    // reached end of a
    cmp     w25, w26
    bcs     equal_or_longer
    // a shorter -> less
    mov     w22, #1
    b       after_cmp

equal_or_longer:
    // not less
    // w22 stays 0
    b       after_cmp

after_cmp:
    cbz     w22, no_update_min
    mov     x31, x5                  // min = j

no_update_min:
    add     x5, x5, #1
    b       inner_loop

inner_done:
    // swap i and min if needed
    cmp     x31, x29
    beq     no_swap

    // load i pointer and length
    adrp    x6, line_ptrs
    add     x6, x6, :lo12:line_ptrs
    lsl     x7, x29, #3
    add     x8, x6, x7
    ldr     x9, [x8]                 // ptr_i
    adrp    x10, line_lens
    add     x10, x10, :lo12:line_lens
    lsl     x11, x29, #2
    add     x12, x10, x11
    ldr     w13, [x12]               // len_i

    // load min pointer and length (already in x17, w21)
    // store min into i slot
    str     x17, [x8]
    str     w21, [x12]

    // store i into min slot
    adrp    x14, line_ptrs
    add     x14, x14, :lo12:line_ptrs
    lsl     x15, x31, #3
    add     x16, x14, x15
    str     x9, [x16]

    adrp    x18, line_lens
    add     x18, x18, :lo12:line_lens
    lsl     x19, x31, #2
    add     x20, x18, x19
    str     w13, [x20]

no_swap:
    add     x29, x29, #1
    b       outer_loop

sort_done:
    // ---------- output sorted lines ----------
    mov     x21, #0                  // i = 0

output_loop:
    ldr     x27, [x13]               // line_cnt
    cmp     x21, x27
    bge     output_done

    // load pointer
    adrp    x28, line_ptrs
    add     x28, x28, :lo12:line_ptrs
    lsl     x29, x21, #3
    add     x30, x28, x29
    ldr     x0, [x30]                // line address in x0

    // load length
    adrp    x1, line_lens
    add     x1, x1, :lo12:line_lens
    lsl     x2, x21, #2
    add     x3, x1, x2
    ldr     w4, [x3]                 // length in w4

write_line:
    cbz     w4, write_newline
    mov     x5, #1                  // fd = stdout
    mov     x6, x0
    mov     x7, x4
    mov     x8, #64                 // sys_write
    svc     #0
    mov     x9, x0                  // bytes_written
    sub     w4, w4, w9
    add     x0, x0, x9
    b       write_line

write_newline:
    adrp    x5, newline
    add     x5, x5, :lo12:newline
    mov     x6, #1
    mov     x0, #1
    mov     x8, #64
    svc     #0

    add     x21, x21, #1
    b       output_loop

output_done:
    // ---------- exit ----------
    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
newline:    .byte 0x0A

.align 8
.section .bss
input_buf:   .space 32768
line_ptrs:   .space 8000          // 1000 * 8 bytes
line_lens:   .space 4000          // 1000 * 4 bytes
total_len:   .zero 8
line_cnt:    .zero 8
