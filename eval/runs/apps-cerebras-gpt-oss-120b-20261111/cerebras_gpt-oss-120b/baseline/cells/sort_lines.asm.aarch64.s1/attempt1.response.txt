.text
.global _start
_start:
    // ---------- read all input ----------
    adr     x5, input_buf          // base address of input buffer
    mov     x3, #0                 // offset into buffer (bytes read so far)

read_loop:
    // remaining space = 32768 - offset
    mov     x4, #32768
    sub     x4, x4, x3
    cbz     x4, read_done

    mov     x0, #0                 // fd = stdin
    add     x1, x5, x3             // buf = input_buf + offset
    mov     x2, x4                 // count = remaining space
    mov     x8, #63                // sys_read
    svc     #0
    cbz     x0, read_done          // 0 bytes read -> EOF

    add     x3, x3, x0             // offset += bytes_read
    b       read_loop

read_done:
    // store total length
    adr     x6, total_len
    str     x3, [x6]

    // ---------- parse lines ----------
    // line_count = 0
    adr     x7, line_count
    mov     x0, #0
    str     x0, [x7]

    adr     x8, input_buf          // start of buffer
    ldr     x9, [x6]               // total_len
    add     x10, x8, x9            // end address (buf + total_len)

parse_loop:
    cmp     x8, x10
    bge     parse_done

    // line_start = x8
    mov     x11, x8                // save start pointer
    mov     x12, #0                // len = 0

scan_loop:
    cmp     x8, x10
    bge     scan_end
    ldrb    w13, [x8], #1          // load byte, advance pointer
    cmp     w13, #0x0A
    beq     scan_end
    add     x12, x12, #1
    b       scan_loop

scan_end:
    // store line (pointer and length)
    ldr     x14, [x7]              // line_count
    // store pointer
    adr     x15, line_ptrs
    lsl     x16, x14, #3           // *8
    add     x17, x15, x16
    str     x11, [x17]

    // store length
    adr     x18, line_lens
    lsl     x19, x14, #2           // *4
    add     x20, x18, x19
    str     w12, [x20]

    // line_count++
    add     x14, x14, #1
    str     x14, [x7]

    // continue parsing
    b       parse_loop

parse_done:
    // ---------- sort lines (selection sort) ----------
    ldr     x21, [x7]              // line_count
    cmp     x21, #2
    blt     sort_done

    mov     x22, #0                // i = 0

outer_loop:
    ldr     x21, [x7]              // line_count
    sub     x23, x21, #1
    cmp     x22, x23
    bge     sort_done

    mov     x24, x22               // min = i
    add     x25, x22, #1           // j = i+1

inner_loop:
    ldr     x21, [x7]              // line_count
    cmp     x25, x21
    bge     inner_done

    // load line j
    adr     x26, line_ptrs
    lsl     x27, x25, #3
    add     x28, x26, x27
    ldr     x29, [x28]             // ptr_j
    adr     x30, line_lens
    add     x31, x30, x25, lsl #2
    ldr     w0, [x31]              // len_j

    // load line min
    adr     x1, line_ptrs
    lsl     x2, x24, #3
    add     x3, x1, x2
    ldr     x4, [x3]               // ptr_min
    adr     x5, line_lens
    add     x6, x5, x24, lsl #2
    ldr     w1, [x6]               // len_min

    // compare j vs min
    // x29 = ptr_j, w0 = len_j, x4 = ptr_min, w1 = len_min
    mov     x7, #0                 // result flag (1 if j < min)
    mov     x8, #0                 // index
cmp_loop:
    cmp     x8, x0
    bcs     cmp_len_check
    cmp     x8, x1
    bcs     cmp_len_check
    ldrb    w9, [x29, x8]
    ldrb    w10, [x4, x8]
    cmp     w9, w10
    bne     cmp_diff
    add     x8, x8, #1
    b       cmp_loop

cmp_diff:
    cmp     w9, w10
    bhs     set_not_less
    mov     x7, #1
    b       cmp_len_check
set_not_less:
    // x7 stays 0
    b       cmp_len_check

cmp_len_check:
    cbz     x7, len_compare
    // already less, skip length compare
    b       set_min

len_compare:
    cmp     x0, x1
    bhs     set_min
    mov     x7, #1                 // j is shorter => less

set_min:
    cbz     x7, no_update_min
    mov     x24, x25               // min = j

no_update_min:
    add     x25, x25, #1
    b       inner_loop

inner_done:
    // swap i and min if needed
    cmp     x24, x22
    beq     no_swap

    // load i values
    adr     x9, line_ptrs
    lsl     x10, x22, #3
    add     x11, x9, x10
    ldr     x12, [x11]             // ptr_i
    adr     x13, line_lens
    add     x14, x13, x22, lsl #2
    ldr     w15, [x14]             // len_i

    // load min values
    adr     x16, line_ptrs
    lsl     x17, x24, #3
    add     x18, x16, x17
    ldr     x19, [x18]             // ptr_min
    adr     x20, line_lens
    add     x21, x20, x24, lsl #2
    ldr     w22, [x21]             // len_min

    // store min into i slot
    str     x19, [x11]
    str     w22, [x14]

    // store i into min slot
    str     x12, [x18]
    str     w15, [x21]

no_swap:
    add     x22, x22, #1
    b       outer_loop

sort_done:
    // ---------- output sorted lines ----------
    mov     x22, #0                // i = 0

output_loop:
    ldr     x23, [x7]              // line_count
    cmp     x22, x23
    bge     output_done

    // load pointer
    adr     x24, line_ptrs
    lsl     x25, x22, #3
    add     x26, x24, x25
    ldr     x27, [x26]             // line address

    // load length
    adr     x28, line_lens
    add     x28, x28, x22, lsl #2
    ldr     w29, [x28]             // length

    // write line (handle partial writes)
write_line:
    cbz     w29, write_newline
    mov     x0, #1                // fd = stdout
    mov     x1, x27
    mov     x2, x29
    mov     x8, #64               // sys_write
    svc     #0
    mov     x30, x0               // bytes_written
    sub     x29, x29, x30
    add     x27, x27, x30
    b       write_line

write_newline:
    // write '\n'
    adr     x1, newline
    mov     x2, #1
    mov     x0, #1
    mov     x8, #64
    svc     #0
    // (partial write loop omitted for brevity – single byte will write fully)

    add     x22, x22, #1
    b       output_loop

output_done:
    // ---------- exit ----------
    mov     x0, #0
    mov     x8, #93               // sys_exit
    svc     #0

// ---------- data ----------
.section .rodata
newline:    .byte 0x0A

.section .bss
.align 8
input_buf:   .space 32768
line_ptrs:   .space 8000          // 1000 * 8 bytes
line_lens:   .space 4000          // 1000 * 4 bytes
total_len:   .quad 0
line_count:  .quad 0
