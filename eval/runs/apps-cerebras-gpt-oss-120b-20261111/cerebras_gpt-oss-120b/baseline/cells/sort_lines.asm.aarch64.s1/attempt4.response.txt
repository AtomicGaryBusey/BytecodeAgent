.text
.global _start
_start:
    // ------------------- read all input -------------------
    adrp    x9, input_buf                // x9 = base of input buffer
    add     x9, x9, :lo12:input_buf
    mov     x10, #0                     // x10 = total bytes read

read_loop:
    mov     x11, #32768                 // x11 = remaining space
    sub     x11, x11, x10
    cbz     x11, read_done

    mov     x0, #0                      // fd = stdin
    add     x1, x9, x10                 // buf = input_buf + total
    mov     x2, x11
    mov     x8, #63                     // sys_read
    svc     #0
    cbz     x0, read_done               // EOF
    add     x10, x10, x0                // total += bytes_read
    b       read_loop

read_done:
    // store total length
    adrp    x12, total_len
    add     x12, x12, :lo12:total_len
    str     x10, [x12]

    // line_cnt = 0
    adrp    x13, line_cnt
    add     x13, x13, :lo12:line_cnt
    mov     x0, #0
    str     x0, [x13]

    // ------------------- parse lines -------------------
    mov     x14, x9                     // x14 = cur pointer
    ldr     x15, [x12]                  // x15 = total_len
    add     x15, x9, x15                // x15 = end pointer

parse_loop:
    cmp     x14, x15
    bge     parse_done

    mov     x16, x14                    // line start
    mov     x17, #0                     // line length

scan_loop:
    cmp     x14, x15
    bge     scan_end
    ldrb    w18, [x14], #1
    cmp     w18, #0x0A
    beq     scan_end
    add     x17, x17, #1
    b       scan_loop

scan_end:
    // store pointer
    adrp    x19, line_ptrs
    add     x19, x19, :lo12:line_ptrs
    ldr     x20, [x13]                  // line_cnt
    lsl     x21, x20, #3                // offset = cnt*8
    add     x22, x19, x21
    str     x16, [x22]

    // store length
    adrp    x23, line_lens
    add     x23, x23, :lo12:line_lens
    lsl     x24, x20, #2                // offset = cnt*4
    add     x25, x23, x24
    str     w17, [x25]

    // line_cnt++
    add     x20, x20, #1
    str     x20, [x13]

    b       parse_loop

parse_done:
    // load line count
    ldr     x18, [x13]                  // x18 = line_cnt

    // ------------------- sort lines (selection sort) -------------------
    mov     x19, #0                     // i = 0

outer_loop:
    cmp     x19, x18
    bge     sort_done

    mov     x20, x19                    // min = i
    add     x21, x19, #1                // j = i+1

inner_loop:
    cmp     x21, x18
    bge     inner_done

    // ----- load line j -----
    lsl     x30, x21, #3                // offset = j*8
    add     x26, x19, #0                // placeholder to keep x19 unchanged
    add     x26, x19, #0                // no-op
    add     x26, x19, #0                // no-op
    add     x26, x19, #0                // no-op
    add     x26, x19, #0                // no-op
    add     x26, x19, #0                // no-op
    // compute address
    adrp    x27, line_ptrs
    add     x27, x27, :lo12:line_ptrs
    add     x28, x27, x30
    ldr     x24, [x28]                  // x24 = ptr_j

    lsl     x30, x21, #2                // offset = j*4
    adrp    x27, line_lens
    add     x27, x27, :lo12:line_lens
    add     x28, x27, x30
    ldr     w25, [x28]                  // w25 = len_j

    // ----- load line min -----
    lsl     x30, x20, #3                // offset = min*8
    add     x28, x27, #0                // placeholder (reuse)
    adrp    x27, line_ptrs
    add     x27, x27, :lo12:line_ptrs
    add     x28, x27, x30
    ldr     x26, [x28]                  // x26 = ptr_min

    lsl     x30, x20, #2                // offset = min*4
    adrp    x27, line_lens
    add     x27, x27, :lo12:line_lens
    add     x28, x27, x30
    ldr     w27, [x28]                  // w27 = len_min

    // ----- compare j vs min -----
    mov     x31, #0                     // result flag (0 = not less, 1 = less)
    mov     x32, #0                     // index

cmp_loop:
    cmp     x32, x25
    bge     len_check
    cmp     x32, x27
    bge     len_check
    add     x33, x24, x32
    ldrb    w34, [x33]
    add     x35, x26, x32
    ldrb    w36, [x35]
    cmp     w34, w36
    bne     diff_found
    add     x32, x32, #1
    b       cmp_loop

diff_found:
    cmp     w34, w36
    bhs     not_less
    mov     x31, #1
    b       after_cmp

len_check:
    cmp     x25, x27
    bge     not_less
    mov     x31, #1

after_cmp:
    cbz     x31, no_update_min
    mov     x20, x21                    // min = j
no_update_min:
    add     x21, x21, #1
    b       inner_loop

inner_done:
    // ----- swap i and min if needed -----
    cmp     x20, x19
    beq     no_swap

    // load i pointer and length
    lsl     x30, x19, #3
    adrp    x27, line_ptrs
    add     x27, x27, :lo12:line_ptrs
    add     x28, x27, x30
    ldr     x38, [x28]                  // x38 = ptr_i

    lsl     x30, x19, #2
    adrp    x27, line_lens
    add     x27, x27, :lo12:line_lens
    add     x28, x27, x30
    ldr     w39, [x28]                  // w39 = len_i

    // store min into i slot
    // ptr_min in x26, len_min in w27
    str     x26, [x28]                  // store ptr_min at i pointer slot (reuse x28)
    str     w27, [x28]                  // store len_min at i length slot (reuse same address, okay because length slot is separate address)

    // store i into min slot
    lsl     x30, x20, #3
    adrp    x27, line_ptrs
    add     x27, x27, :lo12:line_ptrs
    add     x28, x27, x30
    str     x38, [x28]                  // ptr_i into min pointer slot

    lsl     x30, x20, #2
    adrp    x27, line_lens
    add     x27, x27, :lo12:line_lens
    add     x28, x27, x30
    str     w39, [x28]                  // len_i into min length slot

no_swap:
    add     x19, x19, #1
    b       outer_loop

sort_done:

    // ------------------- output sorted lines -------------------
    mov     x23, #0                     // i = 0

output_loop:
    cmp     x23, x18
    bge     output_done

    // load pointer
    adrp    x24, line_ptrs
    add     x24, x24, :lo12:line_ptrs
    lsl     x25, x23, #3
    add     x26, x24, x25
    ldr     x27, [x26]                  // x27 = line address

    // load length
    adrp    x28, line_lens
    add     x28, x28, :lo12:line_lens
    lsl     x29, x23, #2
    add     x30, x28, x29
    ldr     w28, [x30]                  // w28 = length

write_loop:
    cbz     w28, write_newline
    mov     x0, #1                      // fd = stdout
    mov     x1, x27                     // buf
    mov     x2, w28                     // count
    mov     x8, #64                     // sys_write
    svc     #0
    mov     x5, x0                      // bytes_written
    sub     w28, w28, w5
    add     x27, x27, x5
    b       write_loop

write_newline:
    adrp    x0, newline
    add     x0, x0, :lo12:newline
    mov     x1, #1
    mov     x2, #1
    mov     x8, #64
    svc     #0

    add     x23, x23, #1
    b       output_loop

output_done:
    // ------------------- exit -------------------
    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
.align 1
newline: .byte 0x0A

.align 8
.bss
.align 8
input_buf: .space 32768
.align 8
line_ptrs: .space 8000          // 1000 * 8 bytes
.align 8
line_lens: .space 4000          // 1000 * 4 bytes
.align 8
total_len: .zero 8
line_cnt: .zero 8
