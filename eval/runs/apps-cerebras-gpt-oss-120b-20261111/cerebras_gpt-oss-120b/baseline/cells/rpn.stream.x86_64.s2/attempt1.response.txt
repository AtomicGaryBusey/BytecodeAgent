.globl _start

.equ INPUT_BUF_SIZE, 65536
.equ STACK_MAX, 1024
.equ NUM_BUF_SIZE, 32
.equ TOKEN_BUF_SIZE, 32

.section .bss
input_buf:
    .zero INPUT_BUF_SIZE
stack:
    .zero STACK_MAX*8
num_buf:
    .zero NUM_BUF_SIZE
token_buf:
    .zero TOKEN_BUF_SIZE

.section .rodata
err_str:
    .ascii "error\n"
err_len = . - err_str
newline_chr:
    .byte 0x0A

.section .text
_start:
    xor r12, r12                # total read length = 0

read_loop:
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, r12                # buffer + offset
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r12                # remaining space
    mov rax, 0                  # sys_read
    syscall
    test rax, rax
    jz read_done
    add r12, rax                # increase total length
    cmp r12, INPUT_BUF_SIZE
    jl read_loop

read_done:
    lea rsi, [rip + input_buf]  # current pointer
    lea rdi, [rsi + r12]        # end pointer
    lea r13, [rip + stack]      # stack base
    jmp line_start

line_start:
    cmp rsi, rdi
    je exit_program

    xor rbx, rbx                # stack index = 0
    xor r8b, r8b                # error flag = 0
    xor rcx, rcx                # token length = 0

line_loop:
    cmp rsi, rdi
    je line_end_no_newline

    mov al, byte ptr [rsi]
    inc rsi
    cmp al, 0x0A                # newline?
    je line_finish_newline
    cmp al, 0x20                # space
    je token_delim
    cmp al, 0x09                # tab
    je token_delim

    # normal character, store in token_buf
    lea rdx, [rip + token_buf]
    mov byte ptr [rdx + rcx], al
    inc rcx
    cmp rcx, TOKEN_BUF_SIZE
    ja token_invalid_set
    jmp line_loop

token_delim:
    test rcx, rcx
    je line_loop                # no token
    # process token (non-final)
process_token:
    mov r9, rcx                 # token length
    lea r10, [rip + token_buf] # token pointer
    cmp r9, 1
    jne check_literal
    mov bl, byte ptr [r10]
    cmp bl, '+'
    je op_add
    cmp bl, '-'
    je op_sub
    cmp bl, '*'
    je op_mul
    cmp bl, '/'
    je op_div
    cmp bl, '%'
    je op_mod
    # not operator, fall through to literal
check_literal:
    xor r11d, r11d              # sign flag 0 = positive
    mov al, byte ptr [r10]
    cmp al, '-'
    jne parse_digits_start
    mov r11b, 1                 # negative sign
    lea r10, [r10 + 1]
    dec r9
parse_digits_start:
    cmp r9, 0
    je token_invalid_set
    xor rdx, rdx                # accumulator
    xor rax, rax                # digit index
digit_loop:
    cmp rax, r9
    je digits_done
    cmp rax, 18
    ja token_invalid_set
    mov bl, byte ptr [r10 + rax]
    sub bl, '0'
    cmp bl, 9
    ja token_invalid_set
    imul rdx, rdx, 10
    add rdx, rbx
    inc rax
    jmp digit_loop
digits_done:
    test r11b, r11b
    je literal_ok
    neg rdx
literal_ok:
    mov [r13 + rbx*8], rdx
    inc rbx
    jmp token_done

token_invalid_set:
    mov r8b, 1
    jmp token_done

token_done:
    xor rcx, rcx                # reset token length
    jmp line_loop

# operator implementations
op_add:
    cmp rbx, 2
    jb token_invalid_set
    mov rax, [r13 + (rbx-1)*8]  # b
    mov rdx, [r13 + (rbx-2)*8]  # a
    add rdx, rax
    mov [r13 + (rbx-2)*8], rdx
    dec rbx
    jmp token_done

op_sub:
    cmp rbx, 2
    jb token_invalid_set
    mov rax, [r13 + (rbx-1)*8]  # b
    mov rdx, [r13 + (rbx-2)*8]  # a
    sub rdx, rax
    mov [r13 + (rbx-2)*8], rdx
    dec rbx
    jmp token_done

op_mul:
    cmp rbx, 2
    jb token_invalid_set
    mov rax, [r13 + (rbx-1)*8]  # b
    mov rdx, [r13 + (rbx-2)*8]  # a
    imul rdx, rax
    mov [r13 + (rbx-2)*8], rdx
    dec rbx
    jmp token_done

op_div:
    cmp rbx, 2
    jb token_invalid_set
    mov rcx, [r13 + (rbx-1)*8]  # b (divisor)
    mov rax, [r13 + (rbx-2)*8]  # a (dividend)
    test rcx, rcx
    je token_invalid_set
    mov rdx, rax
    cmp rax, 0x8000000000000000
    jne .ok_div
    cmp rc
