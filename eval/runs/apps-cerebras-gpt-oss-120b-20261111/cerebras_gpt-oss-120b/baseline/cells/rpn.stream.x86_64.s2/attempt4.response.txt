.globl _start

.equ INPUT_BUF_SIZE, 65536
.equ STACK_MAX, 1024
.equ TOKEN_MAX, 32
.equ OUT_BUF_SIZE, 32

.section .bss
input_buf:
    .zero INPUT_BUF_SIZE
stack:
    .zero STACK_MAX*8
token_buf:
    .zero TOKEN_MAX
out_buf:
    .zero OUT_BUF_SIZE

.section .rodata
err_str:
    .ascii "error\n"
err_len = . - err_str

.section .text
_start:
    xor r12, r12                    # total bytes read

read_loop:
    mov rdi, 0                      # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, r12                    # buffer + offset
    mov rdx, INPUT_BUF_SIZE
    sub rdx, r12                    # remaining space
    mov rax, 0                      # sys_read
    syscall
    test rax, rax
    jz read_done
    add r12, rax                    # increase total length
    cmp r12, INPUT_BUF_SIZE
    jl read_loop

read_done:
    lea rsi, [rip + input_buf]      # current pointer
    lea rdi, [rsi + r12]            # end pointer
    lea r13, [rip + stack]          # stack base
    xor r15b, r15b                  # line_has_data flag = 0
    jmp line_start

line_start:
    xor rbx, rbx                    # stack size = 0
    xor r8b, r8b                    # error flag = 0
    xor rcx, rcx                    # token length = 0
    jmp line_loop

line_loop:
    cmp rsi, rdi
    je line_end                     # end of input
    mov al, byte ptr [rsi]
    inc rsi
    inc r15b                        # any byte means line has data
    cmp al, 0x0A                    # newline ?
    je line_finish_newline
    cmp al, 0x20                    # space
    je token_delim
    cmp al, 0x09                    # tab
    je token_delim

    # normal character, store in token_buf
    lea rdx, [rip + token_buf]
    mov byte ptr [rdx + rcx], al
    inc rcx
    cmp rcx, TOKEN_MAX
    ja token_invalid_set
    jmp line_loop

token_delim:
    test rcx, rcx
    je line_loop                # no token accumulated
    # process token
process_token:
    test r8b, r8b
    jne token_done              # already in error, skip processing

    mov r9, rcx                 # token length
    lea r10, [rip + token_buf]  # token pointer

    cmp r9, 1
    jne maybe_literal
    mov al, byte ptr [r10]
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    # not an operator, fall through to literal

maybe_literal:
    ; parse integer literal
