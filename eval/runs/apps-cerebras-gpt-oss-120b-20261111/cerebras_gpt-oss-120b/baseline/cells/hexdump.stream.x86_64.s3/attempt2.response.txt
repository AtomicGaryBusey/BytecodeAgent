.globl _start
.equ MAX_BUF, 65536

.text
_start:
    # addresses
    lea r14, [rip + buf]          # r14 = input buffer base
    lea r9,  [rip + hex_digits]   # r9 = hex table

    xor r12, r12                  # total_len = 0
    xor r13, r13                  # row_offset = 0
    mov rbx, r14                  # rbx = write pointer for read

read_loop:
    mov rax, 0
    mov rdi, 0
    mov rsi, rbx
    mov rdx, MAX_BUF
    sub rdx, r12                  # remaining space
    syscall
    test rax, rax
    jz read_done
    add r12, rax                  # total_len += bytes_read
    add rbx, rax                  # advance write pointer
    jmp read_loop
read_done:

row_loop:
    cmp r13, r12
    jae after_rows                 # if offset >= total_len, done

    mov rax, r12
    sub rax, r13                  # remaining = total_len - offset
    mov rcx, 16
    cmp rax, rcx
    cmova rcx, rax                # rcx = row_len = min(16, remaining)
    mov r10, rcx                  # preserve row_len

    lea rbx, [rip + line_buf]    # rbx = line buffer start
    mov rsi, rbx                 # rsi = current pointer

    # write offset (8 hex digits)
    mov rax, r13
    mov r8, 8
hex_off_loop:
    mov rdx, rax
    shr rdx, 28
    and rdx, 0xF
    movzx edx, dl
    mov al, byte ptr [r9 + rdx]
    mov byte ptr [rsi], al
    inc rsi
    shl rax, 4
    dec r8
    jne hex_off_loop

    # two spaces after offset
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi

    # byte area (16 slots)
    xor r8, r8                   # i = 0
byte_area_loop:
    cmp r8, 16
    je after_byte_area
    cmp r8, r10
    jl byte_present
    # pad three spaces
    mov byte ptr [rsi], ' '
    mov byte ptr [rsi+1], ' '
    mov byte ptr [rsi+2], ' '
    add rsi, 3
    jmp after_byte_store
byte_present:
    mov al, byte ptr [r14 + r13 + r8]
    movzx ebx, al
    # high
