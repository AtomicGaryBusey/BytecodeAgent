	.global _start
	.section .bss
	.align 8
buf:	.space 65536
outbuf: .space 32

	.text
	.align 2
_start:
	// x20 = lines, x21 = words, x22 = in_word flag, x28 = total bytes
	mov	x20, #0
	mov	x21, #0
	mov	x22, #0
	mov	x28, #0
	adr	x23, buf          // buffer address

read_loop:
	// read(fd=0, buf, 65536)
	mov	x0, #0
	mov	x1, x23
	mov	x2, #65536
	mov	x8, #63
	svc	#0
	cbz	x0, read_done      // EOF
	// accumulate total bytes
	add	x28, x28, x0
	// remaining count
	mov	x25, x0
	// processing pointer
	mov	x24, x23

proc_loop:
	subs	x25, x25, #1
	blt	proc_done
	ldrb	w5, [x24], #1

	// newline?
	cmp	w5, #0x0A
	bne	not_nl
	add	x20, x20, #1
not_nl:
	// whitespace check
	cmp	w5, #0x09
	beq	is_ws
	cmp	w5, #0x0A
	beq	is_ws
	cmp	w5, #0x0B
	beq	is_ws
	cmp	w5, #0x0C
	beq	is_ws
	cmp	w5, #0x0D
	beq	is_ws
	cmp	w5, #0x20
	beq	is_ws
	// non‑whitespace
	mov	x22, #1
	b	proc_cont
is_ws:
	// end of word?
	tst	x22, #1
	beq	proc_cont
	add	x21, x21, #1
	mov	x22, #0
proc_cont:
	b	proc_loop
proc_done:
	b	read_loop

read_done:
	// final word if input ended inside a word
	tst	x22, #1
	beq	skip_last_word
	add	x21, x21, #1
skip_last_word:

	// output buffer start
	adr	x26, outbuf        // outbuf start
	mov	x27, x26           // current output pointer

	// output lines count
	mov	x0, x20
	mov	x1, x27
	bl	uint_to_dec
	mov	x27, x0

	// space
	mov	w5, #' '
	strb	w5, [x27], #1

	// output words count
	mov	x0, x21
	mov	x1, x27
	bl	uint_to_dec
	mov	x27, x0

	// space
	mov	w5, #' '
	strb	w5, [x27], #1

	// output bytes count
	mov	x0, x28
	mov	x1, x27
	bl	uint_to_dec
	mov	x27, x0

	// newline
	mov	w5, #10
	strb	w5, [x27], #1

	// length = x27 - x26
	sub	x2, x27, x26

	// write(fd=1, outbuf, length)
	mov	x0, #1
	mov	x1, x26
	mov	x8, #64
	svc	#0

	// exit(0)
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// ----------------------------------------------------------------
	// uint_to_dec: x0 = number, x1 = output pointer
	// returns updated pointer in x0
	.align 2
uint_to_dec:
	// save original sp
	mov	x7, sp
	cmp	x0, #0
	bne	ud_div_loop
	// zero case
	mov	w5, #'0'
	strb	w5, [x1], #1
	mov	x0, x1
	ret

ud_div_loop:
	mov	x2, #10            // divisor
ud_div:
	udiv	x3, x0, x2        // quotient
	mul	x5, x3, x2
	sub	x4, x0, x5        // remainder
	add	w4, w4, #'0'      // digit character
	sub	sp, sp, #1
	strb	w4, [sp]          // push digit
	mov	x0, x3
	cbz	x0, ud_copy
	b	ud_div

ud_copy:
ud_pop:
	cmp	sp, x7
	beq	ud_done
	ldrb	w5, [sp], #1
	strb	w5, [x1], #1
	b	ud_pop
ud_done:
	mov	x0, x1
	ret
