    .global _start
    .section .bss
    .align 8
buf:        .space 65536
outbuf:     .space 64

    .section .text
    .align 2
_start:
    // registers:
    // x10 = total_bytes
    // x11 = lines
    // x12 = words
    // x13 = in_word flag (0 or 1)
    // x14 = buffer pointer
    // x15 = current pointer in buffer processing
    // x9  = remaining count in processing loop
    // x0-x8 used for syscalls etc.

    mov     x10, #0          // total bytes
    mov     x11, #0          // lines
    mov     x12, #0          // words
    mov     x13, #0          // in_word flag

    adr     x14, buf         // buffer address

read_loop:
    // read(fd=0, buf, 65536)
    mov     x0, #0
    mov     x1, x14
    mov     x2, #65536
    mov     x8, #63
    svc     #0
    // x0 = number of bytes read
    cbz     x0, read_done    // if 0, EOF

    mov     x9, x0           // remaining bytes count
    add     x10, x10, x0     // total bytes accumulator

    mov     x15, x14         // pointer to current byte

proc_loop:
    subs    x9, x9, #1
    blt     proc_done

    ldrb    w5, [x15], #1    // load byte into w5, advance pointer

    // check newline (0x0A)
    cmp     w5, #0x0A
    b.ne    not_newline
    add     x11, x11, #1    // lines++
not_newline:

    // whitespace detection (0x09,0x0A,0x0B,0x0C,0x0D,0x20)
    // compare sequentially
    cmp     w5, #0x09
    b.eq    is_ws
    cmp     w5, #0x0A
    b.eq    is_ws
    cmp     w5, #0x0B
    b.eq    is_ws
    cmp     w5, #0x0C
    b.eq    is_ws
    cmp     w5, #0x0D
    b.eq    is_ws
    cmp     w5, #0x20
    b.eq    is_ws
    // non‑whitespace
    mov     x13, #1
    b       proc_continue

is_ws:
    // if we were inside a word, finish it
    tst     x13, #1
    b.eq    proc_continue
    add     x12, x12, #1    // words++
    mov     x13, #0

proc_continue:
    b       proc_loop

proc_done:
    b       read_loop

read_done:
    // final word if input ended inside a word
    tst     x13, #1
    b.eq    skip_last_word
    add     x12, x12, #1
skip_last_word:

    // ----- output -----
    // x4 = start of outbuf
    adr     x4, outbuf
    mov     x1, x4          // current output pointer

    // output lines count
    mov     x0, x11
    bl      uint_to_dec

    // space
    mov     w2, #' '
    strb    w2, [x1], #1

    // output words count
    mov     x0, x12
    bl      uint_to_dec

    // space
    mov     w2, #' '
    strb    w2, [x1], #1

    // output bytes count
    mov     x0, x10
    bl      uint_to_dec

    // newline
    mov     w2, #10
    strb    w2, [x1], #1

    // length = x1 - x4
    sub     x2, x1, x4      // x2 = length

    // write(fd=1, outbuf, length)
    mov     x0, #1
    mov     x1, x4
    mov     x8, #64
    svc     #0

    // exit(0)
    mov     x0, #0
    mov     x8, #93
    svc     #0
    // should never reach here

// ------------------------------------------------------------
// uint_to_dec: convert unsigned integer in x0 to decimal string.
// Input:  x0 = number, x1 = pointer to output buffer.
// Output: x1 advanced past written characters.
// Clobbers: x2‑x7, w2‑w6, may use stack.
    .align 2
uint_to_dec:
    // save original sp in x7
    mov     x7, sp

    // quick zero case
    cmp     x0, #0
    b.ne    utd_loop_start
    mov     w2, #'0'
    strb    w2, [x1], #1
    b       utd_done

utd_loop_start:
    // x4 will hold constant 10
    mov     x4, #10
utd_div:
    udiv    x3, x0, x4          // x3 = quotient
    mul     x5, x3, x4
    sub     x6, x0, x5          // x6 = remainder
    add     w6, w6, #'0'        // convert digit to ASCII
    sub     sp, sp, #1
    strb    w6, [sp]            // push digit onto stack
    mov     x0, x3
    cbz     x0, utd_copy
    b       utd_div

utd_copy:
    // now pop digits back to output buffer
utd_pop:
    cmp     sp, x7
    b.eq    utd_done
    ldrb    w2, [sp], #1
    strb    w2, [x1], #1
    b       utd_pop

utd_done:
    ret
