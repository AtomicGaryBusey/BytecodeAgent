.global _start
.text
_start:
    // base addresses
    adrp    x20, line_ptrs
    add     x20, x20, :lo12:line_ptrs       // x20 = line_ptrs base
    adrp    x21, line_lens
    add     x21, x21, :lo12:line_lens       // x21 = line_lens base
    adrp    x22, buf
    add     x22, x22, :lo12:buf             // x22 = buf base

    // read input
    mov     x9, #0                          // total_len = 0
read_loop:
    mov     x2, #32768
    sub     x2, x2, x9                      // remaining space
    cbz     x2, read_done
    add     x1, x22, x9                     // buf + total_len
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0                      // total_len += bytes_read
    b       read_loop
read_done:
    // parse lines
    mov     x10, #0                         // offset i
    mov     x11, #0                         // line_start
    mov     w12, #0                         // line count

parse_loop:
    cmp     x10, x9
    beq     parse_finish_check
    add     x23, x22, x10
    ldrb    w24, [x23]                      // current byte
    cmp     w24, #10
    bne     not_newline
    // newline found
    sub     x4, x10, x11                    // length
    add     x5, x22, x11                    // pointer = buf + line_start
    // store pointer
    uxtw    x6, w12
    lsl     x6, x6, #3
    add     x7, x20, x6
    str     x5, [x7]
    // store length
    uxtw    x6, w12
    lsl     x6, x6, #2
    add     x7, x21, x6
    str     w4, [x7]
    // line count++
    add     w12, w12, #1
    // advance line_start past newline
    add     x11, x10, #1
    // advance offset past newline
    add     x10, x10, #1
    b       parse_loop
not_newline:
    add     x10, x10, #1
    b       parse_loop

parse_finish_check:
    cmp     x11, x9
    bge     after_parse
    sub     x4, x9, x11                     // length
    add     x5, x22, x11                    // pointer
    // store pointer
    uxtw    x6, w12
    lsl     x6, x6, #3
    add     x7, x20, x6
    str     x5, [x7]
    // store length
    uxtw    x6, w12
    lsl     x6, x6, #2
    add     x7, x21, x6
    str     w4, [x7]
    add     w12, w12, #1
after_parse:
    cbz     w12, exit_program                // no lines

    // selection sort
    mov     w24, #0                         // i
outer_loop:
    cmp     w24, w12
    bge     sorted_done
    mov     w25, w24                       // min = i
    add     w26, w24, #1                   // j = i+1
inner_loop:
    cmp     w26, w12
    bge     after_inner
    // ----- compare line j (w26) with line min (w25) -----
    // ptr_j
    uxtw    x0, w26
    lsl     x0, x0, #3
    add     x1, x20, x0
    ldr     x2, [x1]                       // x2 = ptr_j
    // len_j
    uxtw    x3, w26
    lsl     x3, x3, #2
    add     x4, x21, x3
    ldr     w5, [x4]                       // w5 = len_j
    // ptr_min
    uxtw    x6, w25
    lsl     x6, x6, #3
    add     x7, x20, x6
    ldr     x8, [x7]                       // x8 = ptr_min
    // len_min
    uxtw    x9, w25
    lsl     x9, x9, #2
    add     x10, x21, x9
    ldr     w11, [x10]                     // w11 = len_min
    // compare loop
    mov     x12, #0                        // offset counter
compare_loop:
    cmp     x12, x5
    bge     j_exhausted
    cmp     x12, x11
    bge     min_exhausted
    add     x13, x2, x12
    ldrb    w13, [x13]
    add     x14, x8, x12
    ldrb    w14, [x14]
    cmp     w13, w14
    bne     diff_bytes
    add     x12, x12, #1
    b       compare_loop
diff_bytes:
    cmp     w13, w14
    bhs     not_less
    mov     w0, #1                         // j < min
    b       after_compare
not_less:
    mov     w0, #0
    b       after_compare
j_exhausted:
    mov     w0, #1                         // j shorter
    b       after_compare
min_exhausted:
    mov     w0, #0                         // min shorter
after_compare:
    // if w0 == 1 then min = j
    cmp     w0, #0
    beq     skip_min_update
    mov     w25, w26
skip_min_update:
    add     w26, w26, #1
    b       inner_loop
after_inner:
    // swap if min != i
    cmp     w25, w24
    beq     continue_outer
    // swap pointers
    uxtw    x0, w24
    lsl     x0, x0, #3
    add     x1, x20, x0
    ldr     x2, [x1]                       // ptr_i
    uxtw    x3, w25
    lsl     x3, x3, #3
    add     x4, x20, x3
    ldr     x5, [x4]                       // ptr_min
    str     x5, [x1]
    str     x2, [x4]
    // swap lengths
    uxtw    x0, w24
    lsl     x0, x0, #2
    add     x1, x21, x0
    ldr     w2, [x1]                       // len_i
    uxtw    x3, w25
    lsl     x3, x3, #2
    add     x4, x21, x3
    ldr     w5, [x4]                       // len_min
    str     w5, [x1]
    str     w2, [x4]
continue_outer:
    add     w24, w24, #1
    b       outer_loop
sorted_done:
    // reset index for output
    mov     w24, #0

output_loop:
    cmp     w24, w12
    bge     exit_program
    // load pointer and length
    uxtw    x0, w24
    lsl     x0, x0, #3
    add     x1, x20, x0
    ldr     x2, [x1]                       // x2 = ptr_line
    uxtw    x3, w24
    lsl     x3, x3, #2
    add     x4, x21, x3
    ldr     w5, [x4]                       // w5 = len_line
write_line_loop:
    cbz     w5, write_newline
    mov     x0, #1                         // fd = stdout
    mov     x1, x2
    mov     w2, w5
    mov     x8, #64
    svc     #0
    cbz     x0, write_newline
    add     x2, x2, x0
    sub     w5, w5, w0
    b       write_line_loop
write_newline:
    adrp    x0, newline
    add     x0, x0, :lo12:newline
    mov     x1, x0
    mov     w2, #1
    mov     x8, #64
    svc     #0
    add     w24, w24, #1
    b       output_loop

exit_program:
    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
newline:
    .byte   0x0A

.bss
.align 4
buf:
    .space 32768
.align 8
line_ptrs:
    .space 8000          // 1000 * 8
.align 4
line_lens:
    .space 4000          // 1000 * 4
