    .global _start
    .text
_start:
    // ----- read entire input into buf -----
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    mov     x2, #32768
    mov     x0, #0          // fd = stdin
    mov     x8, #63         // sys_read
    svc     #0
    cbz     x0, read_done
    mov     x9, x0          // total_len = bytes read so far
read_loop:
    cmp     x9, #32768
    bge     read_done
    adrp    x1, buf
    add     x1, x1, :lo12:buf
    add     x1, x1, x9      // buf + offset
    mov     x2, #32768
    sub     x2, x2, x9      // remaining space
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cbz     x0, read_done
    add     x9, x9, x0      // total_len += bytes_read
    b       read_loop
read_done:
    // x9 = total_len

    // ----- parse lines into arrays -----
    mov     x10, #0         // current offset i
    mov     x11, #0         // line_start_offset
    mov     w12, #0         // line count (in w12)

parse_loop:
    cmp     x10, x9
    beq     parse_finish_check
    adrp    x2, buf
    add     x2, x2, :lo12:buf
    add     x2, x2, x10
    ldrb    w3, [x2]        // current byte
    cmp     w3, #10         // newline?
    bne     not_newline
    // ---- newline found ----
    sub     x4, x10, x11    // length = offset - line_start
    // ptr = buf + line_start
    adrp    x5, buf
    add     x5, x5, :lo12:buf
    add     x5, x5, x11

    // store pointer
    mov     x6, x12
    lsl     x6, x6, #3
    adrp    x7, line_ptrs
    add     x7, x7, :lo12:line_ptrs
    add     x7, x7, x6
    str     x5, [x7]

    // store length
    mov     x8, x12
    lsl     x8, x8, #2
    adrp    x9, line_lens
    add     x9, x9, :lo12:line_lens
    add     x9, x9, x8
    str     w4, [x9]

    // line count++
    add     w12, w12, #1

    // start next line after newline
    add     x11, x10, #1
    add     x10, x10, #1
    b       parse_loop

not_newline:
    add     x10, x10, #1
    b       parse_loop

parse_finish_check:
    // handle possible final line without terminating newline
    cmp     x11, x9
    bge     after_parse
    sub     x4, x9, x11    // length = total_len - line_start
    adrp    x5, buf
    add     x5, x5, :lo12:buf
    add     x5, x5, x11

    // store pointer
    mov     x6, x12
    lsl     x6, x6, #3
    adrp    x7, line_ptrs
    add     x7, x7, :lo12:line_ptrs
    add     x7, x7, x6
    str     x5, [x7]

    // store length
    mov     x8, x12
    lsl     x8, x8, #2
    adrp    x9, line_lens
    add     x9, x9, :lo12:line_lens
    add     x9, x9, x8
    str     w4, [x9]

    add     w12, w12, #1

after_parse:
    // ----- selection sort (ascending) -----
    cmp     w12, #1
    ble     output_lines
    mov     w13, #0            // i = 0
outer_loop:
    cmp     w13, w12
    bge     output_lines
    mov     w14, w13           // min = i
    add     w15, w13, #1       // j = i+1
inner_loop:
    cmp     w15, w12
    bge     after_inner

    // load ptr_j and len_j
    mov     w0, w15
    uxtw    x0, w0
    lsl     x0, x0, #3
    adrp    x1, line_ptrs
    add     x1, x1, :lo12:line_ptrs
    add     x1, x1, x0
    ldr     x2, [x1]           // ptr_j
    mov     x3, w15
    uxtw    x3, w
