.globl _start

.equ INPUT_BUF_SIZE, 49152
.equ READ_CHUNK, 4096
.equ OUT_BUF_SIZE, 70000
.equ LINE_MAX, 76

.section .bss
.align 16
input_buf:
    .space INPUT_BUF_SIZE
out_buf:
    .space OUT_BUF_SIZE

.section .rodata
.align 64
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # rbx = base of input buffer
    lea rbx, [rip + input_buf]
    xor r12, r12                 # total input length = 0

read_loop:
    mov rax, 0                   # sys_read
    mov rdi, 0                   # fd = stdin
    mov rcx, INPUT_BUF_SIZE
    sub rcx, r12                 # remaining space
    cmp rcx, READ_CHUNK
    jl .read_small
    mov rdx, READ_CHUNK
    jmp .do_read
.read_small:
    mov rdx, rcx
.do_read:
    lea rsi, [rbx + r12]         # destination address
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success

    # rsi = input pointer
    lea rsi, [rbx]
    # rdi = output pointer
    lea rdi, [rip + out_buf]
    # r8 = alphabet base
    lea r8, [rip + alphabet]
    xor r13, r13                 # line_len = 0
    mov rcx, r12                 # remaining input bytes

encode_main:
    cmp rcx, 3
    jb encode_remainder

    # load three input bytes
    movzx rax, byte ptr [rsi]        # b0
    movzx rbx, byte ptr [rsi+1]      # b1
    movzx rdx, byte ptr [rsi+2]      # b2

    # s0 in r9
    mov r9, rax
    shr r9, 2

    # s1 in r10
    mov r10, rax
    and r10, 3
    shl r10, 4
    mov r11, rbx
    shr r11, 4
    or r10, r11

    # s2 in r14
    mov r14, rbx
    and r14, 0xF
    shl r14, 2
    mov r15, rdx
    shr r15, 6
    or r14, r15

    # s3 in r15
    mov r15, rdx
    and r15, 0x3F

    # output s0
    mov al, [r8 + r9]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl0
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl0:
    # output s1
    mov al, [r8 + r10]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl1:
    # output s2
    mov al, [r8 + r14]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl2:
    # output s3
    mov al, [r8 + r15]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl3:

    add rsi, 3
    sub rcx, 3
    jmp encode_main

encode_remainder:
    cmp rcx, 0
    je encode_finish
    cmp rcx, 1
    je encode_one
    # rcx == 2
    movzx rax, byte ptr [rsi]        # b0
    movzx rbx, byte ptr [rsi+1]      # b1

    # s0 in r9
    mov r9, rax
    shr r9, 2

    # s1 in r10
    mov r10, rax
    and r10, 3
    shl r10, 4
    mov r11, rbx
    shr r11, 4
    or r10, r11

    # s2 in r14
    mov r14, rbx
    and r14, 0xF
    shl r14, 2

    # output s0
    mov al, [r8 + r9]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl4:
    # output s1
    mov al, [r8 + r10]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl5
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl5:
    # output s2
    mov al, [r8 + r14]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl6
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl6:
    # padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl7
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl7:
    jmp encode_finish

encode_one:
    movzx rax, byte ptr [rsi]        # b0

    # s0 in r9
    mov r9, rax
    shr r9, 2

    # s1 in r10
    mov r10, rax
    and r10, 3
    shl r10, 4

    # output s0
    mov al, [r8 + r9]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl8
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl8:
    # output s1
    mov al, [r8 + r10]
    mov [rdi], al
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl9
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl9:
    # first padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl10
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl10:
    # second padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r13
    cmp r13, LINE_MAX
    jne .no_nl11
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13
.no_nl11:
    jmp encode_finish

encode_finish:
    cmp r13, 0
    je write_output
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r13, r13

write_output:
    # rdi = end pointer, rsi = start of out_buf
    lea rsi, [rip + out_buf]
    mov rdx, rdi
    sub rdx, rsi               # output length in rdx
    mov rbx, rsi               # current pointer

write_loop:
    test rdx, rdx
    jz exit_success
    mov rax, 1                 # sys_write
    mov rdi, 1                 # stdout
    mov rsi, rbx
    syscall
    test rax, rax
    jz exit_success
    add rbx, rax
    sub rdx, rax
    jmp write_loop

exit_success:
    mov rax, 60                # sys_exit
    xor rdi, rdi
    syscall
