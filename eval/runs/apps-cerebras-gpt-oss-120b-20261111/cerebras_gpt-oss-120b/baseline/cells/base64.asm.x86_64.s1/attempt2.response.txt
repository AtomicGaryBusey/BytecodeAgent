.global _start

.equ INPUT_BUF_SIZE, 49152
.equ READ_CHUNK, 4096

.section .bss
.align 16
input_buf:
    .space INPUT_BUF_SIZE
out_buf:
    .space 70000

.section .rodata
.align 64
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # rbx = input buffer base
    lea rbx, [rip + input_buf]
    xor r12, r12                # total input length = 0

read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd = stdin
    mov rcx, INPUT_BUF_SIZE
    sub rcx, r12                # remaining space
    cmp rcx, READ_CHUNK
    jl .read_small
    mov rdx, READ_CHUNK
    jmp .do_read
.read_small:
    mov rdx, rcx
.do_read:
    lea rsi, [rbx + r12]        # buffer offset
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success

    # set up pointers
    lea rsi, [rbx]                      # input pointer
    lea r9,  [rip + out_buf]            # output base
    mov rdi, r9                         # output pointer
    xor r14, r14                        # line_len = 0
    lea r8,  [rip + alphabet]           # alphabet base
    mov rcx, r12                        # remaining input bytes

encode_loop:
    cmp rcx, 3
    jb  encode_remainder

    # load three bytes
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]
    movzx ecx, byte ptr [rsi+2]

    # s0 = b0 >> 2
    mov edx, eax
    shr edx, 2

    # s1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ebx
    shr edi, 4
    or  esi, edi

    # s2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    mov edi, ebx
    and edi, 0xF
    shl edi, 2
    mov ebp, ecx
    shr ebp, 6
    or  edi, ebp

    # s3 = b2 & 0x3F
    mov ebp, ecx
    and ebp, 0x3F

    # output s0
    mov al, [r8 + rdx]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl1
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl1:
    # output s1
    mov al, [r8 + rsi]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl2
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl2:
    # output s2
    mov al, [r8 + rdi]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl3
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl3:
    # output s3
    mov al, [r8 + rbp]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl4
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl4:

    add rsi, 3
    sub rcx, 3
    jmp encode_loop

encode_remainder:
    cmp rcx, 0
    je encode_finish
    cmp rcx, 1
    je encode_one
    # rcx == 2
    movzx eax, byte ptr [rsi]
    movzx ebx, byte ptr [rsi+1]

    # s0 = b0 >> 2
    mov edx, eax
    shr edx, 2

    # s1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov edi, ebx
    shr edi, 4
    or  esi, edi

    # s2 = ((b1 & 0xF) << 2)
    mov edi, ebx
    and edi, 0xF
    shl edi, 2

    # output s0
    mov al, [r8 + rdx]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl5
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl5:
    # output s1
    mov al, [r8 + rsi]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl6
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl6:
    # output s2
    mov al, [r8 + rdi]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl7
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl7:
    # padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl8
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl8:
    jmp encode_finish

encode_one:
    movzx eax, byte ptr [rsi]

    # s0 = b0 >> 2
    mov edx, eax
    shr edx, 2

    # s1 = ((b0 & 3) << 4)
    mov ebx, eax
    and ebx, 3
    shl ebx, 4

    # output s0
    mov al, [r8 + rdx]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl9
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl9:
    # output s1
    mov al, [r8 + rbx]
    mov [rdi], al
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl10
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl10:
    # two padding '='
    mov byte ptr [rdi], '='
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl11
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl11:
    mov byte ptr [rdi], '='
    inc rdi
    inc r14
    cmp r14, 76
    jne .no_nl12
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14
.no_nl12:
    jmp encode_finish

encode_finish:
    cmp r14, 0
    je write_output
    mov byte ptr [rdi], 0x0A
    inc rdi
    xor r14, r14

write_output:
    # r9 = out_buf base, rdi = end pointer
    mov rbx, r9
    mov rcx, rdi
    sub rcx, r9          # rcx = output length

write_loop:
    test rcx, rcx
    jz exit_success
    mov rax, 1           # sys_write
    mov rdi, 1           # stdout
    mov rsi, rbx
    mov rdx, rcx
    syscall
    test rax, rax
    jz exit_success
    add rbx, rax
    sub rcx, rax
    jmp write_loop

exit_success:
    mov rax, 60          # sys_exit
    xor rdi, rdi
    syscall
