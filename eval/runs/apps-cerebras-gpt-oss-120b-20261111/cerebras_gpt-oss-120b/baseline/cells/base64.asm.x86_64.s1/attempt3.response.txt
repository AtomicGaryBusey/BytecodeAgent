.global _start

.equ INPUT_BUF_SIZE, 49152
.equ READ_CHUNK, 4096
.equ OUT_BUF_SIZE, 70000
.equ LINE_MAX, 76

.section .bss
.align 16
input_buf:
    .space INPUT_BUF_SIZE
out_buf:
    .space OUT_BUF_SIZE

.section .rodata
.align 64
alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # rbx = base of input buffer
    lea rbx, [rip + input_buf]
    xor r12, r12                 # total input length = 0

read_loop:
    mov rax, 0                   # sys_read
    mov rdi, 0                   # fd = stdin
    mov rcx, INPUT_BUF_SIZE
    sub rcx, r12                 # remaining space
    cmp rcx, READ_CHUNK
    jl .read_small
    mov rdx, READ_CHUNK
    jmp .do_read
.read_small:
    mov rdx, rcx
.do_read:
    lea rsi, [rbx + r12]         # destination address
    syscall
    test rax, rax
    jz read_done
    add r12, rax
    jmp read_loop

read_done:
    test r12, r12
    jz exit_success

    # rsi = input pointer
    lea rsi, [rbx]
    # r9 = base of output buffer
    lea r9, [rip + out_buf]
    mov r10, r9                 # r10 = output pointer
    # r8 = alphabet base
    lea r8, [rip + alphabet]
    xor r14, r14                # line_len = 0
    mov rcx, r12                # remaining input bytes

encode_main:
    cmp rcx, 3
    jb encode_remainder

    # load three input bytes
    movzx eax, byte ptr [rsi]          # b0
    movzx ebx, byte ptr [rsi+1]        # b1
    movzx edx, byte ptr [rsi+2]        # b2

    # s0 = b0 >> 2
    mov edi, eax
    shr edi, 2

    # s1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov ebp, ebx
    shr ebp, 4
    or  esi, ebp

    # s2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    mov ebp, ebx
    and ebp, 0xF
    shl ebp, 2
    mov rax, rdx
    shr rax, 6
    or  ebp, eax

    # s3 = b2 & 0x3F
    mov r15d, edx
    and r15d, 0x3F

    # output s0
    mov al, [r8 + rdi]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl0
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl0:
    # output s1
    mov al, [r8 + rsi]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl1
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl1:
    # output s2
    mov al, [r8 + ebp]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl2
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl2:
    # output s3
    mov al, [r8 + r15d]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl3
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl3:

    add rsi, 3
    sub rcx, 3
    jmp encode_main

encode_remainder:
    cmp rcx, 0
    je encode_finish
    cmp rcx, 1
    je encode_one
    # rcx == 2
    movzx eax, byte ptr [rsi]          # b0
    movzx ebx, byte ptr [rsi+1]        # b1

    # s0 = b0 >> 2
    mov edi, eax
    shr edi, 2

    # s1 = ((b0 & 3) << 4) | (b1 >> 4)
    mov esi, eax
    and esi, 3
    shl esi, 4
    mov ebp, ebx
    shr ebp, 4
    or  esi, ebp

    # s2 = ((b1 & 0xF) << 2)
    mov ebp, ebx
    and ebp, 0xF
    shl ebp, 2

    # output s0
    mov al, [r8 + rdi]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl4
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl4:
    # output s1
    mov al, [r8 + rsi]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl5
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl5:
    # output s2
    mov al, [r8 + ebp]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl6
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl6:
    # padding '='
    mov byte ptr [r10], '='
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl7
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl7:
    jmp encode_finish

encode_one:
    movzx eax, byte ptr [rsi]          # b0

    # s0 = b0 >> 2
    mov edi, eax
    shr edi, 2

    # s1 = ((b0 & 3) << 4)
    mov ebx, eax
    and ebx, 3
    shl ebx, 4

    # output s0
    mov al, [r8 + rdi]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl8
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl8:
    # output s1
    mov al, [r8 + rbx]
    mov [r10], al
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl9
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl9:
    # padding '=' (first)
    mov byte ptr [r10], '='
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl10
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl10:
    # padding '=' (second)
    mov byte ptr [r10], '='
    inc r10
    inc r14
    cmp r14, LINE_MAX
    jne .no_nl11
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14
.no_nl11:
    jmp encode_finish

encode_finish:
    cmp r14, 0
    je write_output
    mov byte ptr [r10], 0x0A
    inc r10
    xor r14, r14

write_output:
    # r9 = out_buf base, r10 = end pointer
    mov rbx, r9
    mov rdx, r10
    sub rdx, r9               # rdx = output length

write_loop:
    test rdx, rdx
    jz exit_success
    mov rax, 1                # sys_write
    mov rdi, 1                # stdout
    mov rsi, rbx
    syscall
    test rax, rax
    jz exit_success
    add rbx, rax
    sub rdx, rax
    jmp write_loop

exit_success:
    mov rax, 60               # sys_exit
    xor rdi, rdi
    syscall
