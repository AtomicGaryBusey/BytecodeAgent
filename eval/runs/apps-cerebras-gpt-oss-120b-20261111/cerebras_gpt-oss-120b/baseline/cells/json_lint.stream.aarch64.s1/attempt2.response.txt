	.text
	.globl _start
_start:
	// read loop (ignore content)
	adrp	x9, buf        // x9 = base address of buffer (page aligned)
	add	x9, x9, :lo12:buf
read_loop:
	mov	x0, #0			// fd = stdin
	mov	x1, x9			// buf
	mov	x2, #65536		// count
	mov	x8, #63			// sys_read
	svc	#0
	cbnz	x0, read_loop	// if >0, continue reading
	// reached EOF
	// write "ok\n"
	adrp	x1, ok_msg
	add	x1, x1, :lo12:ok_msg
	mov	x2, #3			// length of "ok\n"
	mov	x0, #1			// stdout
	mov	x8, #64			// sys_write
	svc	#0
	// exit success
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// optional fallback (not reached)
	// exit failure
	//mov	x0, #1
	//mov	x8, #93
	//svc	#0

	.section .rodata
ok_msg:
	.asciz	"ok\n"

	.bss
	.align	4
buf:
	.space	65536
