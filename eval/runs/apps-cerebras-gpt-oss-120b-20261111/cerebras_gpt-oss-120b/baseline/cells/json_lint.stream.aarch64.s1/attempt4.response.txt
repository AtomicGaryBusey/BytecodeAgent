	.text
	.global	_start
_start:
	// ------------------------------------------------------------
	// Read entire stdin into buf (max 65536 bytes)
	// ------------------------------------------------------------
	adrp	x9, buf                 // x9 = base address of buffer (page aligned)
	add	x9, x9, :lo12:buf
	mov	x10, #0                 // total bytes read = 0
	mov	x3, #65536              // max input size
read_loop:
	// x0 = fd = 0 (stdin)
	mov	x0, #0
	// x1 = buf + total_read
	add	x1, x9, x10
	// x2 = remaining space = max_input - total_read
	sub	x2, x3, x10
	// sys_read
	mov	x8, #63
	svc	#0
	cmp	x0, #0
	beq	read_done               // EOF
	cmp	x0, #0
	blt	error_exit              // negative = error
	add	x10, x10, x0            // total += read
	cmp	x10, #65536
	bge	read_done               // buffer full
	b	read_loop
read_done:
	// x10 = input length
	// ------------------------------------------------------------
	// If length == 0 -> error at 0
	// ------------------------------------------------------------
	cmp	x10, #0
	bne	parse_input
	// empty input error
	mov	x12, #0
	b	output_error

	// ------------------------------------------------------------
	// Parse input
	// ------------------------------------------------------------
parse_input:
	mov	x11, #0                 // current offset
	bl	skip_ws
	bl	parse_value
	cbnz	w0, parse_success      // w0==0 -> error
	// error already in x12
	b	output_error

parse_success:
	// After value, skip trailing whitespace
	bl	skip_ws
	cmp	x11, x10
	beq	ok_success
	// extra data after valid document
	mov	x12, x11
	b	output_error

ok_success:
	adrp	x0, ok_msg
	add	x0, x0, :lo12:ok_msg
	mov	x1, #3
	bl	write_string
	mov	x0, #0
	b	exit

output_error:
	adrp	x0, err_prefix
	add	x0, x0, :lo12:err_prefix
	mov	x1, #9
	bl	write_string
	mov	x0, x12                 // error offset
	bl	decimal_to_string       // returns ptr in x0, len in x1
	bl	write_string
	adrp	x0, err_suffix
	add	x0, x0, :lo12:err_suffix
	mov	x1, #1
	bl	write_string
	mov	x0, #1
	b	exit

	// ------------------------------------------------------------
	// System call exit
	// ------------------------------------------------------------
exit:
	mov	x8, #93                  // sys_exit
	svc	#0

error_exit:
	// unexpected read error -> exit 1
	mov	x0, #1
	b	exit

	// ------------------------------------------------------------
	// Helper: write_string (x0 = pointer, x1 = length)
	// ------------------------------------------------------------
write_string:
	// x0 = addr, x1 = len
	mov	x2, x1
	mov	x0, #1                  // fd = stdout
	mov	x8, #64                 // sys_write
	svc	#0
	ret

	// ------------------------------------------------------------
	// Helper: decimal_to_string
	// Input: x0 = unsigned integer (0..65535)
	// Output: x0 = pointer to string, x1 = length
	// Uses buffer dec_buf (6 bytes)
	// ------------------------------------------------------------
decimal_to_string:
	adrp	x1, dec_buf
	add	x1, x1, :lo12:dec_buf   // x1 = base of buffer
	add	x2, x1, #5              // x2 = pointer to last byte
	mov	x3, #0                  // length = 0
	cmp	x0, #0
	bne	.dec_nonzero
	// value is 0
	mov	w4, #'0'
	strb	w4, [x2]
	mov	x3, #1
	b	.dec_finish
.dec_nonzero:
	mov	x4, x0                  // work value
.dec_loop:
	// divide by 10
	udiv	x5, x4, #10            // quotient
	mul	x6,
