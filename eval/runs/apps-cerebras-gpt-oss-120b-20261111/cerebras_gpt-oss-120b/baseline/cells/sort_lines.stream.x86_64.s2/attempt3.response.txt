.globl _start
.text
_start:
    xor    rbx, rbx                         # total bytes read = 0
read_loop:
    mov    rax, 0                           # sys_read
    mov    rdi, 0                           # fd = stdin
    lea    rsi, [rip + buf]
    add    rsi, rbx                         # buf + already read
    mov    rdx, 32768
    sub    rdx, rbx                         # remaining buffer space
    syscall
    test   rax, rax
    je     parse_input
    add    rbx, rax
    jmp    read_loop

parse_input:
    lea    rdx, [rip + buf]                 # rdx = base of buffer
    xor    r13, r13                         # start offset of current line
    xor    r14, r14                         # line count

    cmp    rbx, 0
    je     after_parse

outer_parse:
    cmp    r13, rbx
    jae    after_parse

    mov    r15, r13                         # scan position

find_nl:
    cmp    r15, rbx
    je     final_line
    mov    al, [rdx + r15]                  # read byte
    cmp    al, 0x0A
    je     store_line_nl
    inc    r15
    jmp    find_nl

store_line_nl:
    mov    rax, r15
    sub    rax, r13                         # length = r15 - r13
    mov    rsi, r14
    shl    rsi, 3
    lea    rdi, [rip + lines]
    add    rdi, rsi
    mov    [rdi], r13d
    mov    [rdi+4], eax
    inc    r14
    inc    r15
    mov    r13, r15
    jmp    outer_parse

final_line:
    cmp    r13, rbx
    jae    after_parse
    mov    rax, rbx
    sub    rax, r13                         # length = total - start
    mov    rsi, r14
    shl    rsi, 3
    lea    rdi, [rip + lines]
    add    rdi, rsi
    mov    [rdi], r13d
    mov    [rdi+4], eax
    inc    r14

after_parse:
    mov    rbx, r14                         # rbx = line count
    cmp    rbx, 1
    jbe    output_lines

    ; buffer base already in rdx

outer_sort:
    xor    r8b, r8
