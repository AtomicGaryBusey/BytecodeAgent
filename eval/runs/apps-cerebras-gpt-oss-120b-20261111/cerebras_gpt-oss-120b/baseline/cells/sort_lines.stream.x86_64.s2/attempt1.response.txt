.globl _start
.text
_start:
    # read all input into buf
    xor    rbx, rbx                # rbx = total bytes read
read_loop:
    mov    rax, 0                  # sys_read
    mov    rdi, 0                  # fd = stdin
    lea    rsi, [rip + buf]        # base address of buf
    add    rsi, rbx                # buf + already read
    mov    rdx, 32768
    sub    rdx, rbx                # remaining space
    syscall
    test   rax, rax
    je     parse_input
    add    rbx, rax                # increase total bytes
    jmp    read_loop

parse_input:
    mov    r12, rbx                # r12 = total_bytes (keep for later)
    xor    r13, r13                # r13 = start offset
    xor    r14, r14                # r14 = line count

    cmp    r12, 0
    je     sort_lines

parse_outer:
    cmp    r13, r12
    jae    sort_lines

    mov    r15, r13                # r15 = scan position
find_nl:
    cmp    r15, r12
    je     final_line
    lea    rax, [rip + buf]
    add    rax, r15
    mov    al, byte ptr [rax]
    cmp    al, 0x0A
    je     found_nl
    inc    r15
    jmp    find_nl

found_nl:
    mov    rax, r15
    sub    rax, r13                # length = r15 - r13
    # store line entry
    mov    rdx, r14
    shl    rdx, 3                  # *8
    lea    rsi, [rip + lines]
    add    rsi, rdx
    mov    ebx, r13d
    mov    [rsi], ebx
    mov    ebx, eax
    mov    [rsi+4], ebx
    inc    r14                     # line count++

    inc    r15                     # skip newline
    mov    r13, r15                # new start
    jmp    parse_outer

final_line:
    cmp    r13, r12
    jae    sort_lines
    mov    rax, r12
    sub    rax, r13                # length = total - start
    mov    rdx, r14
    shl    rdx, 3
    lea    rsi, [rip + lines]
    add    rsi, rdx
    mov    ebx, r13d
    mov    [rsi], ebx
    mov    ebx, eax
    mov    [rsi+4], ebx
    inc    r14                     # line count++

sort_lines:
    mov    r12, r14                # r12 = line_cnt
    cmp    r12, 0
    je     output_lines

    mov    r10, 1                  # i = 1
outer_sort:
    cmp    r10, r12
    jae    output_lines

    mov    r11, r10                # j = i
inner_sort:
    cmp    r11, 0
    je     inc_i

    # addresses of entries j-1 and j
    mov    rax, r11
    dec    rax
    shl    rax, 3
    lea    rsi, [rip + lines]
    add    rsi, rax                # rsi = &lines[(j-1)*8]

    mov    rdx, [rsi]              # offset_a
    mov    ecx, [rsi+4]            # len_a  (in ecx)

    mov    rax, r11
    shl    rax, 3
    lea    rdi, [rip + lines]
    add    rdi, rax                # rdi = &lines[j*8]

    mov    r8d, [rdi]              # offset_b
    mov    r9d, [rdi+4]            # len_b

    # set up pointers to strings
    lea    rsi, [rip + buf]
    add    rsi, rdx                # rsi = buf + offset_a
    lea    rdi, [rip + buf]
    add    rdi, r8                 # rdi = buf + offset_b

    xor    rbx, rbx                # idx = 0
cmp_loop:
    cmp    rbx, rcx
    jae    len_a_done
    cmp    rbx, r9
    jae    len_b_done
    mov    al, byte ptr [rsi + rbx]
    mov    dl, byte ptr [rdi + rbx]
    cmp    al, dl
    ja     a_greater
    jb     a_less
    inc    rbx
    jmp    cmp_loop

a_greater:
    mov    eax, 1
    jmp    cmp_result
a_less:
    mov    eax, -1
    jmp    cmp_result

len_a_done:
    cmp    ecx, r9d
    je     equal_len
    mov    eax, -1                 # a shorter
    jmp    cmp_result
equal_len:
    xor    eax, eax
    jmp    cmp_result

len_b_done:
    mov    eax, 1                  # b shorter, a longer

cmp_result:
    cmp    eax, 0
    jle    inc_i                  # if <=0, order ok, break inner loop
    # swap entries at (j-1) and j
    mov    rax, [rsi]              # offset_a
    mov    rdx, [rsi+4]            # len_a
    mov    rcx, [rdi]              # offset_b
    mov    r8, [rdi+4]             # len_b
    mov    [rsi], rcx
    mov    [rsi+4], r8
    mov    [rdi], rax
    mov    [rdi+4], rdx
    dec    r11
    jmp    inner_sort

inc_i:
    inc    r10
    jmp    outer_sort

output_lines:
    xor    r10, r10                # i = 0
output_loop:
    cmp    r10, r12
    jae    exit_prog

    mov    rdx, r10
    shl    rdx, 3
    lea    rsi, [rip + lines]
    add    rsi, rdx                # rsi = &lines[i]

    mov    ebx, [rsi]              # offset
    mov    ecx, [rsi+4]            # length

    # write line
    mov    rax, 1                  # sys_write
    mov    rdi, 1                  # fd = stdout
    lea    rsi, [rip + buf]
    add    rsi, rbx                # rsi = buf + offset
    mov    rdx, rcx                # length
    syscall

    # write newline
    mov    rax, 1
    mov    rdi, 1
    lea    rsi, [rip + nl]
    mov    rdx, 1
    syscall

    inc    r10
    jmp    output_loop

exit_prog:
    mov    rax, 60                 # sys_exit
    xor    rdi, rdi
    syscall

.section .rodata
nl: .byte 0x0A

.bss
buf:    .space 32768
lines:  .space 8000   # 1000 * 8 bytes
