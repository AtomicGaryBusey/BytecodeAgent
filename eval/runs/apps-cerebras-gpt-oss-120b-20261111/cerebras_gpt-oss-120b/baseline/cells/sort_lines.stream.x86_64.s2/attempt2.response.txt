.globl _start
.text
_start:
    xor    rbx, rbx                # total bytes read = 0
read_loop:
    mov    rax, 0                  # sys_read
    mov    rdi, 0                  # fd = stdin
    lea    rsi, [rip + buf]
    add    rsi, rbx                # buf + already read
    mov    rdx, 32768
    sub    rdx, rbx                # remaining space
    syscall
    test   rax, rax
    je     parse_input
    add    rbx, rax                # total += bytes read
    jmp    read_loop

parse_input:
    mov    r12, rbx                # r12 = total_bytes
    xor    r13, r13                # r13 = start offset of current line
    xor    r14, r14                # r14 = line count
    cmp    r12, 0
    je     sort_lines

outer_parse:
    cmp    r13, r12
    jae    sort_lines
    mov    r15, r13                # scan position
find_nl:
    cmp    r15, r12
    je     final_line
    lea    rax, [rip + buf]
    add    rax, r15
    mov    al, byte ptr [rax]
    cmp    al, 0x0A
    je     store_line_nl
    inc    r15
    jmp    find_nl

store_line_nl:
    mov    rax, r15                # rax = newline position
    sub    rax, r13                # length = r15 - r13
    mov    rdx, r14
    shl    rdx, 3                  # *8
    lea    rsi, [rip + lines]
    add    rsi, rdx
    mov    ebx, r13d
    mov    [rsi], ebx
    mov    ebx, eax
    mov    [rsi+4], ebx
    inc    r14                     # line count++
    inc    r15                     # skip newline
    mov    r13, r15                # next line start
    jmp    outer_parse

final_line:
    cmp    r13, r12
    jae    sort_lines
    mov    rax, r12
    sub    rax, r13                # length = total - start
    mov    rdx, r14
    shl    rdx, 3
    lea    rsi, [rip + lines]
    add    rsi, rdx
    mov    ebx, r13d
    mov    [rsi], ebx
    mov    ebx, eax
    mov    [rsi+4], ebx
    inc    r14                     # line count++

sort_lines:
    mov    r12, r14                # r12 = line_cnt
    cmp    r12, 1
    jbe    output_lines

    xor    r10, r10                # i = 0
i_loop:
    inc    r10                     # i = i+1
    cmp    r10, r12
    jae    output_lines
    mov    r11, r10                # j = i
j_loop:
    cmp    r11, 0
    je     i_loop

    # compute pointers to entries
    mov    rax, r11
    dec    rax
    shl    rax, 3
    lea    rdi, [rip + lines]      # rdi = base lines
    add    rdi, rax                # rdi = &lines[j-1]

    mov    rax, r11
    shl    rax, 3
    lea    rsi, [rip + lines]
    add    rsi, rax                # rsi = &lines[j]

    # compare lines at rdi (A) and rsi (B)
    # load offsets/lengths
    mov    eax, [rdi]              # offset_a
    mov    ecx, [rdi+4]            # len_a
    mov    ebx, [rsi]              # offset_b
    mov    edx, [rsi+4]            # len_b

    lea    r8, [rip + buf]         # base of buf
    lea    r9, [rip + buf]

    add    r8, rax                 # r8 = buf + offset_a
    add    r9, rbx                 # r9 = buf + offset_b

    xor    rdx, rdx                # idx = 0
compare_loop:
    cmp    rdx, rcx
    jae    len_a_done
    cmp    rdx, rdx                # placeholder, will be replaced by len_b compare
