.globl _start
.equ BUF_SIZE, 65536

.section .bss
.align 16
buf:
    .space BUF_SIZE

.section .rodata
nl:
    .byte 0x0A

.section .text
_start:
    # read entire input into buf
    lea rsi, [rip + buf]          # buffer pointer
    mov rdx, BUF_SIZE             # max bytes to read
    xor rbx, rbx                  # total length = 0
read_loop:
    mov rdi, 0                    # fd = stdin
    mov rax, 0                    # sys_read
    syscall
    test rax, rax
    je read_done
    add rbx, rax                  # total += bytes read
    add rsi, rax                  # buffer += bytes read
    sub rdx, rax                  # remaining space
    cmp rdx, 0
    je read_done
    jmp read_loop
read_done:
    # set up for processing
    lea rsi, [rip + buf]          # start of data
    mov rcx, rbx                  # remaining byte count

line_loop:
    cmp rcx, 0
    je exit_prog

    lea rdx, [rsi]                # scan pointer
    xor r9, r9                    # line length = 0

scan_loop:
    cmp rcx, 0
    je no_newline                 # end of input, no newline
    mov al, byte ptr [rdx]
    cmp al, 0x0A
    je have_newline               # found newline
    inc rdx
    inc r9
    dec rcx
    jmp scan_loop

have_newline:
    xor r10, r10                  # flag: newline present (0)
    jmp after_scan

no_newline:
    mov r10, 1                    # flag: no newline in input (1)

after_scan:
    mov r11, rdx                  # save newline or end address

    test r9, r9
    je rev_done

    lea rbx, [rsi + r9 - 1]       # pointer to last byte of line
rev_byte_loop:
    mov al, byte ptr [rbx]
    mov rdi, 1                    # fd = stdout
    mov rdx, 1                    # length = 1
    mov rsi, rbx                  # address of byte
    mov rax, 1                    # sys_write
    syscall
    dec rbx
    dec r9
    jne rev_byte_loop

rev_done:
    # write newline
    mov rdi, 1
    mov rdx, 1
    lea rsi, [rip + nl]
    mov rax, 1
    syscall

    cmp r10, 0
    jne final_no_newline          # no newline in input, we're done

    # there was a newline, advance past it
    lea rsi, [r11 + 1]            # next line start
    dec rcx                       # consume the newline byte
    jmp line_loop

final_no_newline:
    jmp exit_prog

exit_prog:
    mov rax, 60                   # sys_exit
    xor rdi, rdi                  # status = 0
    syscall
