.globl _start

.equ BUF_SIZE, 65536

.section .bss
.align 16
buf:
    .space BUF_SIZE

.section .rodata
nl:
    .byte 0x0A

.section .text
_start:
    # read entire input into buf
    lea rsi, [rip + buf]          # buffer pointer
    mov rdx, BUF_SIZE             # max bytes to read
    xor rbx, rbx                  # total length = 0
read_loop:
    mov rdi, 0                    # fd = stdin
    mov rax, 0                    # sys_read
    syscall
    test rax, rax
    je read_done
    add rbx, rax                  # total += bytes read
    add rsi, rax                  # buffer += bytes read
    sub rdx, rax                  # remaining space
    cmp rdx, 0
    je read_done
    jmp read_loop
read_done:
    # set up for processing
    lea rsi, [rip + buf]          # current pointer to start of data
    mov rcx, rbx                  # remaining byte count

process_loop:
    cmp rcx, 0
    je exit_prog

    mov rdx, rsi                  # scan pointer
    xor r9, r9                    # line length = 0

scan_loop:
    cmp rcx, 0
    je no_newline                 # reached end without newline
    mov al, byte ptr [rdx]
    cmp al, 0x0A
    je newline_found
    inc rdx
    inc r9
    dec rcx
    jmp scan_loop

newline_found:
    # r9 = line length, rdx points at newline
    test r9, r9
    je write_nl
    lea rbx, [rsi + r9 - 1]       # pointer to last byte of line
rev_byte_loop:
    mov al, byte ptr [rbx]
    mov rdi, 1                    # fd = stdout
    mov rdx, 1                    # count = 1
    mov rsi, rbx                  # address of byte
    mov rax, 1                    # sys_write
    syscall
    dec rbx
    dec r9
    jne rev_byte_loop
write_nl:
    lea rsi, [rip + nl]           # address of newline character
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall
    # advance past newline
    inc rdx                       # rdx was address of newline, now after it
    dec rcx                       # consume newline
    mov rsi, rdx                  # next line start
    jmp process_loop

no_newline:
    # rcx == 0, final line without terminating newline
    test r9, r9
    je write_nl_final
    lea rbx, [rsi + r9 - 1]
rev_last_loop:
    mov al, byte ptr [rbx]
    mov rdi, 1
    mov rdx, 1
    mov rsi, rbx
    mov rax, 1
    syscall
    dec rbx
    dec r9
    jne rev_last_loop
write_nl_final:
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rdi, 1
    mov rax, 1
    syscall
    jmp exit_prog

exit_prog:
    mov rax, 60                   # sys_exit
    xor rdi, rdi                  # status = 0
    syscall
