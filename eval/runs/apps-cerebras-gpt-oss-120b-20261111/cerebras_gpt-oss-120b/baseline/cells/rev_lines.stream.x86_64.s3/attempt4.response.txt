.globl _start
.equ BUF_SIZE, 65536

.section .bss
.align 16
buf:
    .space BUF_SIZE

.section .rodata
nl:
    .byte 0x0A

.section .text
_start:
    # read all input into buf
    lea rsi, [rip + buf]          # destination pointer
    mov rdx, BUF_SIZE             # remaining space
    xor rbx, rbx                  # total bytes read = 0

read_loop:
    mov rdi, 0                    # fd = stdin
    mov rax, 0                    # sys_read
    syscall
    test rax, rax
    je read_done
    add rbx, rax                  # total += bytes read
    add rsi, rax                  # advance destination pointer
    sub rdx, rax                  # decrease remaining space
    cmp rdx, 0
    jne read_loop

read_done:
    # set up for processing
    lea rsi, [rip + buf]          # start of data
    mov rcx, rbx                  # remaining byte count

line_loop:
    cmp rcx, 0
    je exit_prog

    mov rdx, rsi                  # scan pointer
    xor r9, r9                    # line length = 0
    xor r10, r10                  # flag = 0 (newline present)

scan_loop:
    cmp rcx, 0
    je no_newline
    mov al, byte ptr [rdx]
    cmp al, 0x0A
    je have_newline
    inc rdx
    inc r9
    dec rcx
    jmp scan_loop

have_newline:
    jmp after_scan

no_newline:
    mov r10, 1                    # flag = 1 (no newline)

after_scan:
    mov r12, rdx                  # save address of newline or end (callee‑saved)

    test r9, r9
    je rev_done

    lea rbx, [rsi + r9 - 1]       # pointer to last byte of line

rev_byte_loop:
    mov al, byte ptr [rbx]
    mov rdi, 1                    # fd = stdout
    mov rsi, rbx                  # address of byte
    mov rdx, 1                    # length = 1
    mov rax, 1                    # sys_write
    syscall
    dec rbx
    dec r9
    jne rev_byte_loop

rev_done:
    # write newline
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    mov rax, 1
    syscall

    cmp r10, 1
    je exit_prog                  # no newline in input, finished

    # advance past the newline we just processed
    lea rsi, [r12 + 1]
    dec rcx                       # consume the newline byte
    jmp line_loop

exit_prog:
    mov rax, 60                   # sys_exit
    xor rdi, rdi                  # status = 0
    syscall
