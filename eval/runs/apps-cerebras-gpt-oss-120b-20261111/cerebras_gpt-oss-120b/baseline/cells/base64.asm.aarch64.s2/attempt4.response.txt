.globl _start
.text
_start:
    // ---------- read all input ----------
    adr x9, in_len           // x9 = &in_len
    mov x10, #0
    str x10, [x9]            // in_len = 0

read_loop:
    mov x0, #0               // fd = stdin
    adr x1, in_buf
    ldr x2, [x9]             // x2 = in_len
    add x1, x1, x2           // buf = in_buf + in_len
    mov x3, #49152
    sub x2, x3, x2           // count = remaining space
    mov x8, #63              // sys_read
    svc #0
    cbz x0, read_done        // EOF
    ldr x2, [x9]
    add x2, x2, x0
    str x2, [x9]             // in_len += bytes read
    b read_loop

read_done:
    // ---------- check empty input ----------
    ldr x0, [x9]             // x0 = in_len
    cbz x0, exit_program

    // ---------- encode ----------
    adr x1, in_buf          // src_ptr
    adr x2, out_buf         // dst_ptr
    ldr x3, [x9]            // remaining = in_len
    mov w4, #0              // line_len = 0
    adr x5, base64_table    // base64 table

full_loop:
    cmp x3, #3
    blt tail_handle

    ldrb w6, [x1], #1       // b0
    ldrb w7, [x1], #1       // b1
    ldrb w8, [x1], #1       // b2

    // indexes
    lsr w9, w6, #2                          // i0
    and w10, w6, #3
    lsl w10, w10, #4
    lsr w11, w7, #4
    orr w10, w10, w11                       // i1
    and w11, w7, #0xF
    lsl w11, w11, #2
    lsr w12, w8, #6
    orr w11, w11, w12                       // i2
    and w12, w8, #0x3F                      // i3

    // output 4 chars with line breaks
    // char0
    add x13, x5, w9, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl0
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl0:
    // char1
    add x13, x5, w10, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl1
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl1:
    // char2
    add x13, x5, w11, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl2
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl2:
    // char3
    add x13, x5, w12, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl3
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl3:

    sub x3, x3, #3
    b full_loop

tail_handle:
    cbz x3, finish_encode

    // handle 1 or 2 remaining bytes
    cmp x3, #1
    beq handle_one

    // ----- handle two bytes -----
handle_two:
    ldrb w6, [x1], #1
    ldrb w7, [x1], #1

    // i0 = b0>>2
    lsr w9, w6, #2
    // i1 = ((b0&3)<<4)|(b1>>4)
    and w10, w6, #3
    lsl w10, w10, #4
    lsr w11, w7, #4
    orr w10, w10, w11
    // i2 = ((b1&0xF)<<2)
    and w11, w7, #0xF
    lsl w11, w11, #2

    // char0
    add x13, x5, w9, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl4
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl4:
    // char1
    add x13, x5, w10, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl5
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl5:
    // char2
    add x13, x5, w11, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl6
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl6:
    // '=' padding
    mov w14, #'='
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl7
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl7:
    b finish_encode

    // ----- handle one byte -----
handle_one:
    ldrb w6, [x1], #1

    // i0 = b0>>2
    lsr w9, w6, #2
    // i1 = ((b0&3)<<4)
    and w10, w6, #3
    lsl w10, w10, #4

    // char0
    add x13, x5, w9, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl8
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl8:
    // char1
    add x13, x5, w10, uxtw #0
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl9
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl9:
    // two '=' paddings
    mov w14, #'='
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl10
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl10:
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl11
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl11:

finish_encode:
    // ensure final newline if line not empty
    cmp w4, #0
    beq compute_len
    mov w14, #10
    strb w14, [x2], #1

compute_len:
    // out_len = dst_ptr - out_buf
    adr x6, out_buf
    sub x7, x2, x6            // x7 = out_len
    // ---------- write output ----------
    cbz x7, exit_program
    mov x0, #1                // fd = stdout
    adr x1, out_buf
    mov x2, x7                // count
    mov x8, #64               // sys_write
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93               // sys_exit
    svc #0

// ---------- data ----------
.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 8
in_len:
    .zero 8
out_len:
    .zero 8
    .align 8
in_buf:
    .space 49152
out_buf:
    .space 67000
