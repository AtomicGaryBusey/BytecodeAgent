.globl _start
.text
_start:
    // ---------- read all input ----------
    // total_len = 0
    adr x9, in_len
    mov x10, #0
    str x10, [x9]

read_loop:
    // fd = 0
    mov x0, #0
    // buf = in_buf + total_len
    adr x1, in_buf
    ldr x2, [x9]               // total_len
    add x1, x1, x2
    // count = 49152 - total_len
    mov x3, #49152
    sub x2, x3, x2             // x2 = remaining space
    // syscall read
    mov x8, #63
    svc #0
    // x0 = bytes read
    cbz x0, read_done          // EOF
    // total_len += x0
    ldr x2, [x9]
    add x2, x2, x0
    str x2, [x9]
    b read_loop

read_done:
    // ---------- check empty input ----------
    ldr x0, [x9]                // x0 = total_len
    cbz x0, exit_program

    // ---------- encode ----------
    // src_ptr = in_buf
    adr x1, in_buf
    // dst_ptr = out_buf
    adr x2, out_buf
    // remaining = total_len
    ldr x3, [x9]                // x3 = remaining bytes
    // line_len = 0  (w4)
    mov w4, #0
    // base64 table address
    adr x5, base64_table

full_loop:
    cmp x3, #3
    blt tail_handle

    // load three bytes, post-increment src ptr (x1)
    ldrb w6, [x1], #1           // b0
    ldrb w7, [x1], #1           // b1
    ldrb w8, [x1], #1           // b2

    // i0 = b0 >> 2
    lsr w9, w6, #2

    // i1 = ((b0 & 3) << 4) | (b1 >> 4)
    and w10, w6, #3
    lsl w10, w10, #4
    lsr w11, w7, #4
    orr w10, w10, w11

    // i2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and w11, w7, #0xF
    lsl w11, w11, #2
    lsr w12, w8, #6
    orr w11, w11, w12

    // i3 = b2 & 0x3F
    and w12, w8, #0x3F

    // store char0
    add x13, x5, w9
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl0
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl0:

    // store char1
    add x13, x5, w10
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl1
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl1:

    // store char2
    add x13, x5, w11
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl2
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl2:

    // store char3
    add x13, x5, w12
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl3
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl3:

    // decrement remaining
    sub x3, x3, #3
    b full_loop

tail_handle:
    // x3 = remaining bytes (0,1,2)
    cbz x3, finish_encode

    // handle 1 byte
    cmp x3, #1
    beq handle_one

    // handle 2 bytes
handle_two:
    ldrb w6, [x1], #1
    ldrb w7, [x1], #1

    // i0 = b0 >> 2
    lsr w9, w6, #2
    // i1 = ((b0 &3)<<4) | (b1>>4)
    and w10, w6, #3
    lsl w10, w10, #4
    lsr w11, w7, #4
    orr w10, w10, w11
    // i2 = ((b1 &0xF)<<2)
    and w11, w7, #0xF
    lsl w11, w11, #2

    // char0
    add x13, x5, w9
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl4
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl4:

    // char1
    add x13, x5, w10
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl5
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl5:

    // char2
    add x13, x5, w11
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl6
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl6:

    // char3 = '='
    mov w14, #'='
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl7
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl7:
    b finish_encode

handle_one:
    ldrb w6, [x1], #1

    // i0 = b0 >> 2
    lsr w9, w6, #2
    // i1 = ((b0 &3)<<4)
    and w10, w6, #3
    lsl w10, w10, #4

    // char0
    add x13, x5, w9
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl8
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl8:

    // char1
    add x13, x5, w10
    ldrb w14, [x13]
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl9
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl9:

    // char2 = '='
    mov w14, #'='
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl10
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl10:

    // char3 = '='
    mov w14, #'='
    strb w14, [x2], #1
    add w4, w4, #1
    cmp w4, #76
    bne no_nl11
    mov w14, #10
    strb w14, [x2], #1
    mov w4, #0
no_nl11:

finish_encode:
    // ensure final newline if line not empty
    cmp w4, #0
    beq compute_len
    mov w14, #10
    strb w14, [x2], #1

compute_len:
    // out_len = dst_ptr - out_buf
    adr x6, out_buf
    sub x7, x2, x6
    adr x8, out_len
    str x7, [x8]

    // ---------- write output ----------
    ldr x0, [x8]          // length
    cbz x0, exit_program
    mov x1, #1           // fd = stdout
    adr x2, out_buf
    mov x8, #64
    svc #0

exit_program:
    mov x0, #0
    mov x8, #93
    svc #0

// ---------- data ----------
.section .rodata
base64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
    .align 8
in_len:
    .quad 0
out_len:
    .quad 0
    .align 8
in_buf:
    .space 49152
out_buf:
    .space 66000
