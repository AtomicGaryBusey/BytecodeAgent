.equ SYSCALL_READ,   63
.equ SYSCALL_WRITE,  64
.equ SYSCALL_EXIT,   93
.equ BUF_SIZE,       65536

.section .bss
.align 3
buffer:     .space BUF_SIZE
.align 3
totlen:     .zero 8
cur_off:    .zero 8
linebuf:    .zero 80

.section .rodata
hex_table:  .ascii "0123456789abcdef"

.text
.global _start
_start:
    // read all input into buffer
    adr    x19, buffer          // x19 = buf_ptr
    mov    x20, #BUF_SIZE       // x20 = remaining
    mov    x21, #0              // x21 = total_len

read_loop:
    mov    x0, #0               // fd = stdin
    mov    x1, x19              // buf
    mov    x2, x20              // count
    mov    x8, #SYSCALL_READ
    svc    #0
    cbz    x0, read_done        // x0 == 0 : EOF
    // x0 > 0 : bytes read
    add    x21, x21, x0         // total_len += bytes_read
    add    x19, x19, x0         // buf_ptr += bytes_read
    sub    x20, x20, x0         // remaining -= bytes_read
    cbnz   x20, read_loop

read_done:
    // store total length
    adr    x28, totlen
    str    x21, [x28]

    // initialize cur_off = 0
    adr    x28, cur_off
    mov    x0, #0
    str    x0, [x28]

dump_loop:
    // load offset and total length
    adr    x22, cur_off
    ldr    x22, [x22]          // x22 = offset
    adr    x21, totlen
    ldr    x21, [x21]          // x21 = total_len
    cmp    x22, x21
    bge    dump_final

    // row_len = min(16, total_len - offset)
    sub    x24, x21, x22        // remaining
    mov    x25, #16
    cmp    x24, x25
    csel   x25, x24, x25, lt   // x25 = row_len

    // x23 = buffer base
    adr    x23, buffer

    // x5 = linebuf pointer
    adr    x5, linebuf

    // ----- offset hex (8 chars) -----
    mov    x6, x22            // offset value
    adr    x7, hex_table

    mov    w8, #8
hex_loop:
    sub    w8, w8, #1
    lsl    w9, w8, #2          // shift = nibble*4
    lsr    x10, x6, x9
    and    w10, w10, #0xF
    add    x11, x7, x10
    ldrb   w12, [x11]
    strb   w12, [x5], #1
    cbnz   w8, hex_loop

    // two spaces after offset
    mov    w12, #' '
    strb   w12, [x5], #1
    strb   w12, [x5], #1

    // ----- byte hex area -----
    add    x15, x23, x22        // x15 = buffer + offset

    // first half (bytes 0..7)
    mov    w13, #0
first_half:
    cmp    w13, #8
    bge    second_half
    cmp    w13, w25
    bge    space_three

    add    x14, x15, w13
    ldrb   w16, [x14]          // w16 = byte

    // high nibble
    lsr    w17, w16, #4
    and    w17, w17, #0xF
    add    x18, x7, w17
    ldrb   w19, [x18]
    strb   w19, [x5], #1

    // low nibble
    and    w17, w16, #0xF
    add    x18, x7, w17
    ldrb   w19, [x18]
    strb   w19, [x5], #1

    // space after byte
    mov    w19, #' '
    strb   w19, [x5], #1
    b      inc_first

space_three:
    // three spaces for missing byte
    mov    w19, #' '
    strb   w19, [x5], #1
    strb   w19, [x5], #1
    strb   w19, [x5], #1

inc_first:
    add    w13, w13, #1
    b      first_half

second_half:
    // extra space after 8 bytes
    mov    w19, #' '
    strb   w19, [x5], #1

    // bytes 8..15
    mov    w13, #8
second_half_loop:
    cmp    w13, #16
    bge    after_bytes_hex
    cmp    w13, w25
    bge    space_three2

    add    x14, x15, w13
    ldrb   w16, [x14]

    // high nibble
    lsr    w17, w16, #4
    and    w17, w17, #0xF
    add    x18, x7, w17
    ldrb   w19, [x18]
    strb   w19, [x5], #1

    // low nibble
    and    w17, w16, #0xF
    add    x18, x7, w17
    ldrb   w19, [x18]
    strb   w19, [x5], #1

    // space after byte
    mov    w19, #' '
    strb   w19, [x5], #1
    b      inc_second

space_three2:
    // three spaces for missing byte
    mov    w19, #' '
    strb   w19, [x5], #1
    strb   w19, [x5], #1
    strb   w19, [x5], #1

inc_second:
    add    w13, w13, #1
    b      second_half_loop

after_bytes_hex:
    // ----- ASCII part -----
    // space and '|'
    mov    w19, #' '
    strb   w19, [x5], #1
    mov    w19, #'|'
    strb   w19, [x5], #1

    // ascii for each byte in row_len
    mov    w13, #0
ascii_loop:
    cmp    w13, w25
    bge    ascii_done

    add    x14, x15, w13
    ldrb   w16, [x14]

    // printable?
    cmp    w16, #0x20
    blt    ascii_dot
    cmp    w16, #0x7e
    bgt    ascii_dot
    // printable
    strb   w16, [x5], #1
    b      ascii_next

ascii_dot:
    mov    w16, #'.'
    strb   w16, [x5], #1

ascii_next:
    add    w13, w13, #1
    b      ascii_loop

ascii_done:
    // final '|' and newline
    mov    w19, #'|'
    strb   w19, [x5], #1
    mov    w19, #'\n'
    strb   w19, [x5], #1

    // write line
    adr    x9, linebuf
    sub    x0, x5, x9        // x0 = length
    mov    x1, #1           // fd = stdout
    mov    x2, x9           // buf
    mov    x8, #SYSCALL_WRITE
    svc    #0

    // update cur_off
    adr    x28, cur_off
    add    x22, x22, x25
    str    x22, [x28]

    b      dump_loop

dump_final:
    // ----- final total length line -----
    adr    x5, linebuf
    adr    x7, hex_table
    ldr    x6, [x21]          // total length (already in x21)

    mov    w8, #8
final_hex_loop:
    sub    w8, w8, #1
    lsl    w9, w8, #2
    lsr    x10, x6, x9
    and    w10, w10, #0xF
    add    x11, x7, x10
    ldrb   w12, [x11]
    strb   w12, [x5], #1
    cbnz   w8, final_hex_loop

    // newline
    mov    w12, #'\n'
    strb   w12, [x5], #1

    // write final line
    adr    x9, linebuf
    sub    x0, x5, x9        // length
    mov    x1, #1
    mov    x2, x9
    mov    x8, #SYSCALL_WRITE
    svc    #0

    // exit success
    mov    x0, #0
    mov    x8, #SYSCALL_EXIT
    svc    #0
