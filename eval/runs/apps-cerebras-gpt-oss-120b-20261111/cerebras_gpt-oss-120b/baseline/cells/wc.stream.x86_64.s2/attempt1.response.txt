.globl _start

.section .bss
.align 16
buf:
    .zero 65536
tmpbuf:
    .zero 32
tmpbuf_end:
outbuf:
    .zero 64

.section .text

_start:
    xor rbx, rbx          # lines = 0
    xor r10, r10         # words = 0
    xor r13, r13         # bytes = 0
    xor r8, r8           # in_word = 0

read_loop:
    mov rax, 0            # sys_read
    xor rdi, rdi         # fd = 0
    lea rsi, [rip + buf]
    mov rdx, 65536
    syscall
    test rax, rax
    je done_read
    mov r14, rax          # r14 = bytes read
process_loop:
    mov al, byte ptr [rsi]
    inc r13               # total bytes++
    cmp al, 0x0A
    jne not_newline
    inc rbx               # lines++
    xor r8, r8            # in_word = 0
    jmp after_ws
not_newline:
    cmp al, 0x09
    je ws
    cmp al, 0x0B
    je ws
    cmp al, 0x0C
    je ws
    cmp al, 0x0D
    je ws
    cmp al, 0x20
    je ws
    # non‑whitespace
    test r8, r8
    jne after_ws
    inc r10               # words++
    mov r8, 1
    jmp after_ws
ws:
    xor r8, r8            # in_word = 0
after_ws:
    inc rsi
    dec r14
    jne process_loop
    jmp read_loop

done_read:
    lea rdi, [rip + outbuf]    # destination pointer

    mov rax, rbx
    call uint_to_ascii

    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r10
    call uint_to_ascii

    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, r13
    call uint_to_ascii

    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi               # length

    mov rax, 1                 # sys_write
    mov rdi, 1                 # fd = 1
    syscall

    mov rax, 60                # sys_exit
    xor rdi, rdi
    syscall

# -------------------------------------------------
# rax = number, rdi = destination pointer
# returns with rdi advanced past the printed digits
uint_to_ascii:
    push rbx
    push r10
    push r13
    push r8
    push rcx

    mov rbx, rax               # rbx = value
    lea rsi, [rip + tmpbuf_end]

.convert_loop:
    test rbx, rbx
    jne .do_div
    dec rsi
    mov byte ptr [rsi], '0'
    jmp .copy_start
.do_div:
    xor rdx, rdx
    mov rcx, 10
    div rcx                    # rax = quot, rdx = rem
    add dl, '0'
    dec rsi
    mov [rsi], dl
    mov rbx, rax
    test rax, rax
    jne .convert_loop

.copy_start:
    lea rdx, [rip + tmpbuf_end]
    sub rdx, rsi               # rdx = length
.copy_loop:
    mov al, [rsi]
    mov [rdi], al
    inc rsi
    inc rdi
    dec rdx
    jne .copy_loop

    pop rcx
    pop r8
    pop r13
    pop r10
    pop rbx
    ret
