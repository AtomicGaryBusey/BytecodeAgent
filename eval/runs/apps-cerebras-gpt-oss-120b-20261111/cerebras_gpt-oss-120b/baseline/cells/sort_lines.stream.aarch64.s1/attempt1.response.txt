	.globl	_start
	.text
	.align	2
_start:
	// load addresses of buffers
	adr	x19, input_buf          // x19 = input buffer base
	adr	x24, line_ptrs         // x24 = pointers array
	adr	x25, line_lens         // x25 = lengths array

	// x20 = total_len = 0
	mov	x20, #0

	// -----------------
	// read input loop
read_loop:
	mov	x0, #0                     // fd = 0 (stdin)
	add	x1, x19, x20              // buf = input_buf + total_len
	mov	x2, #32768                // max buffer size
	sub	x2, x2, x20               // remaining space
	mov	x8, #63                   // sys_read
	svc	#0
	cbnz	x0, read_continue
	b	read_done
read_continue:
	add	x20, x20, x0              // total_len += bytes_read
	cmp	x20, #32768
	bhs	read_done                 // if total_len >= max, stop
	b	read_loop
read_done:

	// -----------------
	// parse lines
	// x26 = offset = 0
	mov	x26, #0
	// x23 = line_count = 0
	mov	x23, #0

parse_loop:
	cmp	x26, x20
	bge	parse_done

	// x27 = scan offset = x26
	mov	x27, x26

scan_newline:
	cmp	x27, x20
	bge	scan_end                  // reached end of input
	ldrb	w0, [x19, x27]           // load byte
	cmp	w0, #10                  // '\n'
	beq	found_newline
	add	x27, x27, #1
	b	scan_newline

found_newline:
	// length = x27 - x26
	sub	x28, x27, x26            // x28 = length
	// store pointer = input_buf + x26
	add	x29, x19, x26            // x29 = line pointer
	// store pointer
	mov	x0, x23
	lsl	x0, x0, #3
	str	x29, [x24, x0]
	// store length
	mov	x0, x23
	lsl	x0, x0, #2
	str	w28, [x25, x0]
	// line_count++
	add	x23, x23, #1
	// skip newline
	add	x27, x27, #1
	mov	x26, x27
	b	parse_loop

scan_end:
	// final line without trailing newline
	sub	x28, x20, x26            // length = total_len - offset
	cbnz	x28, add_final_line
	b	parse_done

add_final_line:
	// store pointer
	add	x29, x19, x26
	mov	x0, x23
	lsl	x0, x0, #3
	str	x29, [x24, x0]
	// store length
	mov	x0, x23
	lsl	x0, x0, #2
	str	w28, [x25, x0]
	// line_count++
	add	x23, x23, #1

parse_done:

	// -----------------
	// sort lines (simple bubble sort)
	// outer index i = 0 -> x26
	mov	x26, #0

outer_loop:
	cmp	x26, x23
	bge	sort_done

	// inner index j = i + 1 -> x27
	add	x27, x26, #1

inner_loop:
	cmp	x27, x23
	bge	inner_done

	// load ptr_i
	mov	x0, x26
	lsl	x0, x0, #3
	ldr	x4, [x24, x0]          // x4 = ptr_i
	// load len_i
	mov	x0, x26
	lsl	x0, x0, #2
	ldr	w5, [x25, x0]          // w5 = len_i

	// load ptr_j
	mov	x0, x27
	lsl	x0, x0, #3
	ldr	x6, [x24, x0]          // x6 = ptr_j
	// load len_j
	mov	x0, x27
	lsl	x0, x0, #2
	ldr	w7, [x25, x0]          // w7 = len_j

	// compare lines: result in w0 (-1,0,1)
	// x8 = ptr_i, x9 = len_i, x10 = ptr_j, x11 = len_j
	mov	x8, x4
	mov	x9, x5
	mov	x10, x6
	mov	x11, x7
	mov	x12, #0                // index

cmp_loop:
	cmp	x12, x9
	bhs	cmp_lenb
	cmp	x12, x11
	bhs	cmp_lena
	ldrb	w0, [x8, x12]
	ldrb	w1, [x10, x12]
	cmp	w0, w1
	bne	cmp_diff
	add	x12, x12, #1
	b	cmp_loop

cmp_diff:
	cmp	w0, w1
	b.lt	cmp_minus_one
	mov	w0, #1
	b	cmp_done
cmp_minus_one:
	mov	w0, #-1
	b	cmp_done

cmp_lenb:
	cmp	x9, x11
	beq	cmp_eq
	// len_i < len_j
	mov	w0, #-1
	b	cmp_done

cmp_lena:
	cmp	x9, x11
	beq	cmp_eq
	// len_i > len_j
	mov	w0, #1
	b	cmp_done

cmp_eq:
	mov	w0, #0

cmp_done:
	// if w0 > 0, swap
	cmp	w0, #0
	ble	inner_continue

	// swap pointers
	// store ptr_i into j slot
	mov	x0, x27
	lsl	x0, x0, #3
	str	x4, [x24, x0]
	// store ptr_j into i slot
	mov	x0, x26
	lsl	x0, x0, #3
	str	x6, [x24, x0]

	// swap lengths
	mov	x0, x27
	lsl	x0, x0, #2
	str	w5, [x25, x0]
	mov	x0, x26
	lsl	x0, x0, #2
	str	w7, [x25, x0]

inner_continue:
	add	x27, x27, #1
	b	inner_loop

inner_done:
	add	x26, x26, #1
	b	outer_loop

sort_done:

	// -----------------
	// output sorted lines
	// index k = 0 -> x26
	mov	x26, #0

output_loop:
	cmp	x26, x23
	bge	output_done

	// load pointer
	mov	x0, x26
	lsl	x0, x0, #3
	ldr	x4, [x24, x0]          // x4 = line pointer
	// load length
	mov	x0, x26
	lsl	x0, x0, #2
	ldr	w5, [x25, x0]          // w5 = length

	// write line (may need multiple syscalls)
	mov	x6, x4                  // current pointer
	mov	x7, x5                  // remaining

write_loop:
	cbnz	x7, write_one
	b	write_newline

write_one:
	mov	x0, #1                 // fd = 1
	mov	x1, x6
	mov	x2, x7
	mov	x8, #64               // sys_write
	svc	#0
	// x0 = bytes written
	sub	x7, x7, x0
	add	x6, x6, x0
	b	write_loop

write_newline:
	adr	x0, newline
	mov	x1, x0
	mov	x2, #1
	mov	x0, #1
	mov	x8, #64
	svc	#0

	add	x26, x26, #1
	b	output_loop

output_done:
	// exit with status 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

	// -----------------
	// data sections
	.section .bss
	.align	4
input_buf:
	.zero	32768
line_ptrs:
	.zero	8000          // 1000 * 8
line_lens:
	.zero	4000          // 1000 * 4

	.section .rodata
	.align	1
newline:
	.byte	0x0A
