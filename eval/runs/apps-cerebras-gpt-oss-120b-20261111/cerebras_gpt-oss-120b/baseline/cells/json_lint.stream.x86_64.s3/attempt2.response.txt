.globl _start
.section .rodata
msg_err_prefix:
    .ascii "error at "
msg_err_prefix_len = . - msg_err_prefix
msg_ok:
    .ascii "ok\n"
msg_ok_len = . - msg_ok

.bss
buffer:
    .space 65536
dec_buf:
    .space 12

.text
_start:
    # read stdin into buffer
    xor     rbx, rbx                # total length = 0
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + buffer]     # buffer address
    mov     rdx, 65536              # max bytes
    mov     r8, rsi                 # r8 = base address of buffer
read_loop:
    xor     rax, rax                # sys_read
    syscall
    cmp     rax, 0
    je      after_read
    add     rbx, rax                # length += bytes read
    add     rsi, rax                # advance buffer pointer
    sub     rdx, rax
    jne     read_loop
after_read:
    xor     rcx, rcx                # position = 0

    call    skip_ws
    call    parse_value
    call    skip_ws
    cmp     rcx, rbx
    jne     error_at_pos

    # success: write "ok\n"
    mov     rdx, msg_ok_len
    lea     rsi, [rip + msg_ok]
    mov     rdi, 1
    mov     rax, 1
    syscall
    xor     edi, edi                # status 0
    mov     rax, 60
    syscall

# -----------------------------
# Helper routines
# -----------------------------
skip_ws:
    cmp     rcx, rbx
    jae     .skip_ws_done
    mov     al, byte ptr [r8 + rcx]
    cmp     al, 0x20
    je      .inc_pos
    cmp     al, 0x09
    je      .inc_pos
    cmp     al, 0x0A
    je      .inc_pos
    cmp     al, 0x0D
    je      .inc_pos
    jmp     .skip_ws_done
.inc_pos:
    inc     rcx
    jmp     skip_ws
.skip_ws_done:
    ret

parse_value:
    call    skip_ws
    cmp     rcx, rbx
    jae     .set_eof_error
    mov     al, byte ptr [r8 + rcx]
    cmp     al, '"'
    je      parse_string
    cmp     al, '{'
    je      parse_object
    cmp     al, '['
    je      parse_array
    cmp     al, 't'
    je      parse_true
    cmp     al, 'f'
    je      parse_false
    cmp     al, 'n'
    je      parse_null
    cmp     al, '-'
    je      parse_number
    cmp     al, '0'
    je      parse_number
    cmp     al, '1'
    jae     .digit_ge_1
    jmp     error
.digit_ge_1:
    cmp     al, '9'
    jbe     parse_number
    jmp     error
.set_eof_error:
    mov     rcx, rbx
    jmp     error

parse_string:
    inc     rcx                     # skip opening quote
.str_loop:
    cmp     rcx, rbx
    jae     .set_eof_error
    mov     al, byte ptr [r8 + rcx]
    cmp     al, '"'
    je      .after_string
    cmp     al, '\\'
    je      .escape_seq
    cmp     al, 0x20
    jl      error
    inc     rcx
    jmp     .str_loop
.escape_seq:
    inc     rcx
    cmp     rcx, rbx
    jae     .set_eof_error
    mov     al, byte ptr [r8 + rcx]
    cmp     al, '"'
    je      .inc_and_continue
    cmp     al, '\\'
    je      .inc_and_continue
    cmp     al, '/'
    je      .inc_and_continue
    cmp     al, 'b'
    je      .inc_and_continue
    cmp     al, 'f'
    je      .inc_and_continue
    cmp     al, 'n'
    je      .inc_and_continue
    cmp     al, 'r'
    je      .inc_and_continue
    cmp     al, 't'
    je      .inc_and_continue
    cmp     al, 'u'
    je      .unicode_escape
    jmp     error
.inc_and_continue:
    inc     rcx
    jmp     .str_loop
.unicode_escape:
    inc     rcx                     # first hex digit
    mov     rdx, 4
.unicode_loop:
    cmp     rcx, rbx
    jae     .set_eof_error
    mov     al, byte ptr [r8 + rcx]
    ; check 0-9
    cmp     al, '
