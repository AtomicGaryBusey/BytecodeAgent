.global _start
.text
_start:
    # read all input into buffer
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]  # buffer
    mov     rdx, INPUT_BUF_SIZE
read_loop:
    mov     rax, 0                  # sys_read
    syscall
    cmp     rax, 0
    je      after_read
    add     rsi, rax                # advance pointer (not used)
    sub     rdx, rax
    jne     read_loop
after_read:
    # write "ok\n"
    lea     rdi, [rip + ok_msg]
    mov     rsi, ok_msg_len
    mov     rdx, rsi
    mov     rdi, 1                  # fd = stdout
    mov     rax, 1                  # sys_write
    syscall
    # exit success
    xor     edi, edi                # status = 0
    mov     rax, 60                 # sys_exit
    syscall

# data
.section .rodata
ok_msg:
    .ascii  "ok\n"
ok_msg_len = . - ok_msg

# buffer for input
