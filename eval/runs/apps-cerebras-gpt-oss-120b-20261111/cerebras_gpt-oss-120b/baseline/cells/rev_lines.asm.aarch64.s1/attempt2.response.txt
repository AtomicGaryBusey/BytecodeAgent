.section .bss
.align 8
inbuf:
    .space 65536
outbuf:
    .space 65537

.text
.global _start
_start:
    adr x5, inbuf          // x5 = input buffer base
    adr x6, outbuf         // x6 = output buffer base
    movz x4, #1, lsl #16   // x4 = 65536 (buffer size)
    mov x3, #0             // x3 = total_len = 0

read_loop:
    sub x2, x4, x3         // remaining space in buffer
    cbz x2, read_done
    add x1, x5, x3         // dest = inbuf + total_len
    mov x0, #0
    mov x8, #63            // sys_read
    svc #0
    cbz x0, read_done
    add x3, x3, x0         // total_len += bytes_read
    b read_loop

read_done:
    mov x10, #0            // current input position

proc_loop:
    cmp x10, x3
    bge exit_prog

    mov x20, x10          // line_start
    mov x11, x10          // scan pointer

scan_loop:
    cmp x11, x3
    beq no_newline
    add x0, x5, x11
    ldrb w0, [x0]
    cmp w0, #10
    beq newline_found
    add x11, x11, #1
    b scan_loop

newline_found:
    sub x12, x11, x20     // line_len = scan - line_start
    b have_len

no_newline:
    sub x12, x3, x20      // line_len = total_len - line_start

have_len:
    mov x15, x12          // line_len copy
    mov x14, x6           // dest = outbuf

    cbz x15, skip_copy

    // src = inbuf + line_start + line_len - 1
    add x13, x5, x20
    add x13, x13, x15
    sub x13, x13, #1

    mov x12, x15          // copy counter
copy_loop:
    ldrb w0, [x13]
    strb w0, [x14]
    sub x13, x13, #1
    add x14, x14, #1
    subs x12, x12, #1
    bne copy_loop

skip_copy:
    // store newline
    mov w0, #10
    strb w0, [x14]

    // write (line_len + 1) bytes
    add x2, x15, #1
    mov x0, #1
    mov x1, x6
    mov x8, #64            // sys_write
    svc #0

    // advance to next line
    cmp x11, x3
    beq set_end
    add x10, x11, #1
    b proc_loop

set_end:
    mov x10, x3
    b proc_loop

exit_prog:
    mov x0, #0
    mov x8, #93            // sys_exit
    svc #0
