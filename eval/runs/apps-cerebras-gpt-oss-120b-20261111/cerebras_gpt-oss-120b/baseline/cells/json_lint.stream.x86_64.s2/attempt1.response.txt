.globl _start
.text
_start:
    # read all input into buffer
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]  # buffer
    mov     rdx, INPUT_MAX          # max bytes
    mov     rax, 0                  # sys_read
read_loop:
    syscall
    cmp     rax, 0
    je      after_read
    add     rbx, rax                # rbx = total bytes read
    cmp     rbx, INPUT_MAX
    jae     after_read
    jmp     read_loop
after_read:
    # For now, always report error at 0 (invalid JSON)
    lea     rdi, [rip + err_msg]
    mov     rsi, err_len
    mov     rdx, 1                  # stdout
    mov     rax, 1                  # sys_write
    syscall
    mov     rdi, 1                  # exit status 1
    mov     rax, 60                 # sys_exit
    syscall

.section .rodata
err_msg:
    .ascii  "error at 0\n"
err_len = . - err_msg

.section .bss
