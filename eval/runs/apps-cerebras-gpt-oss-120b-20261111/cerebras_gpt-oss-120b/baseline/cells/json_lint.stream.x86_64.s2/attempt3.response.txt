.globl _start
.equ INPUT_MAX, 65536
.equ MAX_NESTING, 1000

.section .bss
.align 8
input_buf: .skip INPUT_MAX
input_len: .skip 8          # qword
cursor:    .skip 8          # qword
nest_depth: .skip 4         # dword
error_offset: .skip 8       # qword, -1 if no error
num_buf:   .skip 32          # buffer for decimal conversion

.section .rodata
msg_ok:      .ascii "ok\n"
msg_ok_len = . - msg_ok
msg_err_pre: .ascii "error at "
msg_err_pre_len = . - msg_err_pre

.text
_start:
    # read all input into input_buf
    xor r8, r8                # total length = 0
read_loop:
    mov rdi, 0                # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, r8               # offset into buffer
    mov rdx, INPUT_MAX
    sub rdx, r8               # remaining space
    mov rax, 0                # sys_read
    syscall
    test rax, rax
    jz read_done
    add r8, rax
    cmp r8, INPUT_MAX
    jae read_done
    jmp read_loop
read_done:
    mov [rip + input_len], r8
    mov qword ptr [rip + error_offset], -1
    mov qword ptr [rip + cursor], 0
    mov dword ptr [rip + nest_depth], 0

    # start parsing document
    lea rdi, [rip + input_buf]    # rdi = base pointer
    mov rdx, r8                   # rdx = input_len
    call parse_document
    mov rax, 0                    # success code (will be overwritten if error)

    mov rax, [rip + error_offset]
    cmp rax, -1
    je print_ok

#--------------------------------------------------------------
# Parsing helpers
# All parsing functions expect:
#   rdi = base pointer
#   rdx = input length
#   rcx = current cursor offset (global variable 'cursor' is kept in sync)
# They return:
#   ZF=1 on success, ZF=0 on failure (error already recorded)
#   rcx updated to new position on success
#--------------------------------------------------------------

# parse_document: ws value ws, then check EOF
parse_document:
    push rbx
    push rsi
    # ws
    call skip_ws
    test eax, eax
    jz parse_error
    # value
    call parse_value
    test eax, eax
    jz parse_error
    # ws
    call skip_ws
    test eax, eax
    jz parse_error
    # ensure at EOF
    mov rax, rcx
    cmp rax, rdx
    jne parse_error
    pop rsi
    pop rbx
    xor eax, eax        # success
    ret

parse_error:
    # set error_offset if not set
    mov rax, [rip + error_offset]
    cmp rax, -1
    jne already_error
    mov [rip + error_offset], rcx
already_error:
    pop rsi
    pop rbx
    mov eax, 1          # failure
    ret

# skip_ws: consumes whitespace, always succeeds
skip_ws:
    push rbx
skip_ws_loop:
    cmp rcx, rdx
    jae skip_ws_done
    mov al, byte ptr [rdi + rcx]
    cmp al, ' '
    je ws_inc
    cmp al, 9
    je ws_inc
    cmp al, 0x0a
    je ws_inc
    cmp al, 0x0d
    je ws_inc
    jmp skip_ws_done
ws_inc:
    inc rcx
    jmp skip_ws_loop
skip_ws_done:
    pop rbx
    xor eax, eax
    ret

# parse_value: dispatch based on first non-ws byte
parse_value:
    push rbx
    # ensure we have at least one byte
    cmp rcx, rdx
    jae value_error
    mov al, byte ptr [rdi + rcx]
    ; check literals
    cmp al
