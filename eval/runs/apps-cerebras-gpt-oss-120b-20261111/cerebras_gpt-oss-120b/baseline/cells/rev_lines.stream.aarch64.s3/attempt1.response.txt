.globl _start
.text
_start:
    // load addresses of buffers
    adr     x20, out_byte          // x20 = address of single-byte output buffer
    adr     x12, buf              // x12 = buffer start address

    // prepare read loop
    mov     x9, x12               // x9 = current write pointer in buffer
    mov     x10, #65536           // x10 = remaining capacity (max 65536)

read_loop:
    mov     x0, #0                // fd = stdin
    mov     x1, x9                // buffer pointer
    mov     x2, x10               // count
    mov     x8, #63               // sys_read
    svc     #0
    cbz     x0, read_done         // if read returns 0 -> EOF

    // x0 = number of bytes read
    add     x9, x9, x0            // advance write pointer
    sub     x10, x10, x0          // decrease remaining capacity
    b       read_loop

read_done:
    // x12 = start of data, x9 = end of data (after last byte)
    mov     x13, x9               // x13 = end pointer
    // processing pointer
    mov     x15, x12              // x15 = current position (p)

process_loop:
    cmp     x15, x13
    bge     finish                // p >= end -> done

    // line_start = p
    mov     x16, x15              // x16 = line_start

    // find newline or end
    mov     x17, x15              // x17 = scanning pointer

find_nl:
    cmp     x17, x13
    bge     no_newline            // reached end
    ldrb    w0, [x17]
    cmp     w0, #0x0a
    beq     have_newline
    add     x17, x17, #1
    b       find_nl

have_newline:
    // line_end = address of newline
    mov     x18, x17
    b       process_line

no_newline:
    // line_end = end pointer (no newline)
    mov     x18, x13

process_line:
    // reverse bytes from line_end-1 down to line_start
    sub     x19, x18, #1          // x19 = src pointer (last byte of line)

rev_loop:
    cmp     x19, x16
    blt     rev_done
    ldrb    w0, [x19]             // load byte
    strb    w0, [x20]             // store to output buffer
    // write one byte
    mov     x0, #1                // fd = stdout
    mov     x1, x20
    mov     x2, #1
    mov     x8, #64               // sys_write
    svc     #0
    sub     x19, x19, #1
    b       rev_loop

rev_done:
    // write newline after the line
    mov     w0, #0x0a
    strb    w0, [x20]
    mov     x0, #1
    mov     x1, x20
    mov     x2, #1
    mov     x8, #64
    svc     #0

    // advance p past the newline if it existed
    cmp     x18, x13
    beq     advance_end           // no newline (end reached)
    add     x15, x18, #1          // skip newline
    b       process_loop

advance_end:
    mov     x15, x13
    b       process_loop

finish:
    // exit with status 0
    mov     x0, #0
    mov     x8, #93               // sys_exit
    svc     #0

.section .bss
.align 8
buf:
    .zero   65536
.align 1
out_byte:
    .zero   1
