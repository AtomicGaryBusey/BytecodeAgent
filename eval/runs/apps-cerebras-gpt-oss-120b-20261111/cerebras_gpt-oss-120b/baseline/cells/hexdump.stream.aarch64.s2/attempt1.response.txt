	.global _start
	.text
_start:
	// load addresses of data sections
	adrp	x19, buffer
	add	x19, x19, :lo12:buffer		// x19 = buffer address
	adrp	x20, linebuf
	add	x20, x20, :lo12:linebuf		// x20 = line buffer address
	adrp	x21, hex_digits
	add	x21, x21, :lo12:hex_digits		// x21 = hex table address

	// x22 = total length read so far
	mov	x22, #0

read_loop:
	// compute remaining space = 65536 - x22
	mov	x3, #65536
	sub	x2, x3, x22				// x2 = count
	cmp	x2, #0
	beq	read_done				// no space left (should not happen)

	// destination = buffer + x22
	add	x1, x19, x22

	// sys_read(fd=0, buf, count)
	mov	x0, #0
	mov	x8, #63
	svc	#0
	cbnz	x0, read_continue
	// x0 == 0 -> EOF
	b	read_done
read_continue:
	// add bytes read to total
	add	x22, x22, x0
	b	read_loop

read_done:
	// x23 = current offset for dumping
	mov	x23, #0

dump_loop:
	cmp	x23, x22
	bge	dump_done

	// remaining = total_len - offset
	sub	x24, x22, x23			// x24 = remaining
	mov	x25, #16
	cmp	x24, x25
	csel	x24, x24, x25, hi		// x24 = len = min(16, remaining)

	// x9 = line buffer pointer
	mov	x9, x20

	// ---- print offset (8 hex digits) ----
	mov	x10, #8				// counter
hex_off_loop:
	sub	x10, x10, #1
	lsl	x12, x10, #2			// shift = x10*4
	lsr	x13, x23, x12
	and	x13, x13, #0xF
	add	x14, x21, x13
	ldrb	w15, [x14]
	strb	w15, [x9], #1
	cbnz	x10, hex_off_loop

	// two spaces after offset
	mov	w15, #' '
	strb	w15, [x9], #1
	strb	w15, [x9], #1

	// ---- hex byte fields ----
	mov	x26, #0				// i = 0
hex_bytes_loop:
	cmp	x26, #16
	bge	hex_bytes_done

	cmp	x26, x24
	bge	hex_blank

	// load byte
	add	x27, x19, x23
	add	x27, x27, x26
	ldrb	w0, [x27]

	// high nibble
	and	x28, w0, #0xF0
	lsr	x
