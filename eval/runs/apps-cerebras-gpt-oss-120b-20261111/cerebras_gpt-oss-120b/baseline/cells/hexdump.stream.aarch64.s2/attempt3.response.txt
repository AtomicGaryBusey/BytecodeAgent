.globl _start
.text
_start:
    // Pointers to sections
    adrp    x19, buffer
    add     x19, x19, :lo12:buffer          // x19 = buffer base
    adrp    x20, linebuf
    add     x20, x20, :lo12:linebuf          // x20 = line buffer base
    adrp    x21, hex_digits
    add     x21, x21, :lo12:hex_digits       // x21 = hex table base

    mov     x22, #0                         // total_len = 0

read_loop:
    // count = 65536 - total_len
    mov     x3, #65536
    sub     x2, x3, x22
    cbz     x2, read_done

    // dest = buffer + total_len
    add     x1, x19, x22

    // sys_read(fd=0, buf, count)
    mov     x0, #0
    mov     x8, #63
    svc     #0
    // result in x0
    cbz     x0, read_done
    add     x22, x22, x0                    // total_len += bytes_read
    b       read_loop

read_done:
    // offset = 0
    mov     x23, #0

dump_loop:
    cmp     x23, x22
    bge     dump_done

    // remaining = total_len - offset
    sub     x24, x22, x23
    // len = min(16, remaining)
    mov     x25, #16
    cmp     x24, x25
    csel    x24, x25, x24, hi              // if remaining > 16 then x24=16 else remaining

    // line buffer pointer
    mov     x26, x20

    // ---- offset hex (8 digits) ----
    mov     x27, #8
offset_hex_loop:
    sub     x27, x27, #1
    lsl     x28, x27, #2
    lsr     x29, x23, x28
    and     x29, x29, #0xF
    ldrb    w30, [x21, x29]
    strb    w30, [x26], #1
    cbnz    x27, offset_hex_loop

    // two spaces after offset
    mov     w0, #32
    strb    w0, [x26], #1
    strb    w0, [x26], #1

    // ---- hex bytes (16 slots) ----
    mov     x28, #0
hex_bytes_loop:
    cmp     x28, #16
    bge     hex_bytes_done

    cmp     x28, x24
    bge     hex_byte_space          // i >= len -> spaces

    // load byte from buffer
    add     x9, x19, x23
    add     x9, x9, x28
    ldrb    w0, [x9]

    // high nibble
    lsr     x10, x0, #4
    and     x10, x10, #0xF
    ldrb    w11, [x21, x10]
    strb    w11, [x26], #1
    // low nibble
    and     x10, x0, #0xF
    ldrb    w11, [x21, x10]
    strb    w11, [x26], #1
    // space after byte
    mov     w0, #32
    strb    w0, [x26], #1
    b       hex_byte_next

hex_byte_space:
    // three spaces
    mov     w0, #32
    strb    w0, [x26], #1
    strb    w0, [x26], #1
    strb    w0, [x26], #1

hex_byte_next:
    // extra space after eighth byte (i == 7)
    cmp     x28, #7
    bne     hex_byte_skip_extra
    mov     w0, #32
    strb    w0, [x26], #1
hex_byte_skip_extra:
    add     x28, x28, #1
    b       hex_bytes_loop

hex_bytes_done:
    // space then '|'
    mov     w0, #32
    strb    w0, [x26], #1
    mov     w0, #124
    strb    w0, [x26], #1

    // ---- ascii column ----
    mov     x28, #0
ascii_loop:
    cmp     x28, x24
    bge     ascii_done
    add     x9, x19, x23
    add     x9, x9, x28
    ldrb    w0, [x9]

    // printable ? 0x20 <= w0 <= 0x7e
    cmp     w0, #32
    b.lt    ascii_dot
    cmp     w0, #126
    b.gt    ascii_dot
    // printable
    strb    w0, [x26], #1
    b       ascii_next

ascii_dot:
    mov     w0, #46
    strb    w0, [x26], #1

ascii_next:
    add     x28, x28, #1
    b       ascii_loop

ascii_done:
    // '|', newline
    mov     w0, #124
    strb    w0, [x26], #1
    mov     w0, #10
    strb    w0, [x26], #1

    // write the line
    mov     x0, #1
    mov     x1, x20
    sub     x2, x26, x20
    mov     x8, #64
    svc     #0

    // advance offset
    add     x23, x23, x24
    b       dump_loop

dump_done:
    // ---- final total length line ----
    mov     x26, x20
    mov     x27, #8
len_hex_loop:
    sub     x27, x27, #1
    lsl     x28, x27, #2
    lsr     x29, x22, x28
    and     x29, x29, #0xF
    ldrb    w30, [x21, x29]
    strb    w30, [x26], #1
    cbnz    x27, len_hex_loop

    // newline
    mov     w0, #10
    strb    w0, [x26], #1

    // write final line
    mov     x0, #1
    mov     x1, x20
    sub     x2, x26, x20
    mov     x8, #64
    svc     #0

    // exit
    mov     x0, #0
    mov     x8, #93
    svc     #0

.bss
.align 8
buffer:
    .zero   65536
linebuf:
    .space  80

.section .rodata
.align 1
hex_digits:
    .ascii  "0123456789abcdef"
