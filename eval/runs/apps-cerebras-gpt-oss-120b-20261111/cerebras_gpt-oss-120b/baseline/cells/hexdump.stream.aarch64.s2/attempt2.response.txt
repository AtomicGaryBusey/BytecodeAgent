.globl _start
.text
_start:
    // addresses of sections
    adrp    x19, buffer
    add     x19, x19, :lo12:buffer          // x19 = buffer base
    adrp    x20, linebuf
    add     x20, x20, :lo12:linebuf          // x20 = line buffer base
    adrp    x21, hex_digits
    add     x21, x21, :lo12:hex_digits       // x21 = hex table base

    mov     x22, #0                          // total_len = 0

read_loop:
    // remaining space = 65536 - total_len
    mov     x3, #65536
    sub     x2, x3, x22                      // x2 = count
    cbz     x2, read_done

    // dest = buffer + total_len
    add     x1, x19, x22

    // sys_read(fd=0, buf, count)
    mov     x0, #0
    mov     x8, #63
    svc     #0
    cbz     x0, read_done                    // EOF
    add     x22, x22, x0                     // total_len += bytes_read
    b       read_loop

read_done:
    // offset = 0
    mov     x23, #0

dump_loop:
    cmp     x23, x22
    bge     dump_done

    // remaining = total_len - offset
    sub     x24, x22, x23                    // x24 = remaining
    mov     x25, #16
    cmp     x24, x25
    csel    x24, x24, x25, hi                // x24 = len = min(16, remaining)

    // line pointer = linebuf
    mov     x26, x20

    // ---- offset hex (8 digits) ----
    mov     x28, #8
offset_hex_loop:
    sub     x28, x28, #1
    lsl     x29, x28, #2
    lsr     x30, x23, x29
    and     x30, x30, #0xF
    add     x31, x21, x30, uxtw
    ldrb    w0, [x31]
    strb    w0, [
