	.global _start
	.text
	.align 2

// constants
BUF_SIZE = 49152
MAX_OUT = 67000

_start:
	// ----- read input loop -----
	adr	x4, input_buf               // x4 = base address of buffer
	adr	x5, total_len               // x5 = address of total_len
	mov	x6, #0                      // x6 = current length (will store later)

read_loop:
	// load current length
	ldr	x7, [x5]                    // x7 = total_len
	// destination = buffer + total_len
	add	x1, x4, x7                  // x1 = dst ptr
	// remaining = BUF_SIZE - total_len
	mov	x2, #BUF_SIZE
	sub	x2, x2, x7                  // x2 = count
	// syscall read(fd=0, buf, count)
	mov	x0, #0
	mov	x8, #63
	svc	#0
	// x0 = bytes read
	cbnz	x0, have_read
	// read returned 0 -> end of input
	b	done_read
have_read:
	// add bytes read to total_len
	ldr	x7, [x5]
	add	x7, x7, x0
	str	x7, [x5]
	// continue reading
	b	read_loop

done_read:
	// check if total_len == 0 -> exit with no output
	ldr	x7, [x5]
	cbnz	x7, encode_start
	// empty input, exit 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

// ----- encode -----
encode_start:
	adr	x1, input_buf               // x1 = src ptr
	adr	x2, output_buf              // x2 = dst ptr
	adr	x3, total_len               // x3 = address of total_len
	ldr	x3, [x3]                    // x3 = remaining bytes
	mov	x4, #0                      // x4 = line_len counter
	adr	x5, b64_alphabet            // x5 = table pointer

encode_full:
	cmp	x3, #3
	blt	encode_tail

	// load three bytes
	ldrb	w0, [x1], #1
	ldrb	w1, [x1], #1
	ldrb	w2, [x1], #1
	sub	x3
