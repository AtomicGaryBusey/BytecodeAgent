.global _start
.text
.align 3

// constants
.equ BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 67000
.equ SYSCALL_READ, 63
.equ SYSCALL_WRITE, 64
.equ SYSCALL_EXIT, 93

_start:
    // total_len = 0
    adr x5, total_len          // x5 = &total_len
    str xzr, [x5]              // *total_len = 0

    // input buffer base
    adr x4, input_buf          // x4 = input_buf

read_loop:
    ldr x7, [x5]               // x7 = total_len
    add x1, x4, x7            // x1 = dest = input_buf + total_len
    mov x2, #BUF_SIZE
    sub x2, x2, x7            // x2 = remaining space
    mov x0, #0                // fd = stdin
    mov x8, #SYSCALL_READ
    svc #0
    cbz x0, read_done         // x0 == 0 -> EOF
    // add bytes read to total_len
    ldr x7, [x5]
    add x7, x7, x0
    str x7, [x5]
    b read_loop

read_done:
    ldr x7, [x5]
    cbz x7, exit_zero         // empty input -> exit

    // setup pointers for encoding
    adr x1, input_buf         // src pointer
    adr x2, output_buf        // dst pointer
    ldr x3, [x5]              // remaining bytes = total_len
    mov x4, #0                // line length counter
    adr x5, b64_alphabet       // alphabet base

encode_full:
    cmp x3, #3
    blt encode_tail

    ldrb w0, [x1], #1
    ldrb w1, [x1], #1
