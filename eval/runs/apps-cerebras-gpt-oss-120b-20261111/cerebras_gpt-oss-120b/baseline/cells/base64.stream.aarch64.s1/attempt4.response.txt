.global _start
.text
.align 3

// syscall numbers
.equ SYSCALL_READ,   63
.equ SYSCALL_WRITE, 64
.equ SYSCALL_EXIT,  93

// buffer sizes
.equ BUF_SIZE,      49152
.equ OUT_BUF_SIZE,  66000

_start:
    // total_len = 0
    adr   x5, total_len          // x5 = &total_len
    str   xzr, [x5]

    // input buffer address
    adr   x6, input_buf          // x6 = input_buf

read_loop:
    ldr   x7, [x5]               // x7 = total_len
    add   x1, x6, x7             // x1 = dest = input_buf + total_len
    mov   x0, #0                 // fd = stdin
    mov   x2, #BUF_SIZE
    sub   x2, x2, x7             // count = BUF_SIZE - total_len
    mov   x8, #SYSCALL_READ
    svc   #0
    cbz   x0, read_done          // x0 == 0 -> EOF
    // add bytes read to total_len
    ldr   x7, [x5]
    add   x7, x7, x0
    str   x7, [x5]
    b     read_loop

read_done:
    ldr   x7, [x5]
    cbz   x7, exit_zero          // empty input -> exit 0

    // set up registers for encoding
    adr   x0, input_buf          // src pointer
    adr   x1, output_buf         // dst pointer
    mov   x2, x7                 // remaining bytes = total_len
    mov   x3, #0                 // line length counter
    adr   x4, b64_alphabet       // alphabet base

encode_loop:
    cmp   x2, #3
    blt   encode_tail

    // load three bytes
    ldrb  w8,  [x0], #1
    ldrb  w9,  [x0], #1
    ldrb  w10, [x0], #1

    // idx0 = w8 >> 2
    lsr   w11, w8, #2

    // idx1 = ((w8 & 0x3) << 4) | (w9 >> 4)
    and   w12, w8, #0x3
    lsl   w12, w12, #4
    lsr   w13, w9, #4
    orr   w12, w12, w13

    // idx2 = ((w9 & 0xF) << 2) | (w10 >> 6)
    and   w13, w9, #0xF
    lsl   w13, w13, #2
    lsr   w14, w10, #6
    orr   w13, w13, w14

    // idx3 = w10 & 0x3F
    and   w14, w10, #0x3F

    // store idx0
    add   x9, x4, w11
