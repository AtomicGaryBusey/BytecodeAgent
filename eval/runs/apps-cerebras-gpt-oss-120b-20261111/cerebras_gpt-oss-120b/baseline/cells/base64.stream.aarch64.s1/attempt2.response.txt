.global _start
.text
.align 3

// constants
.equ BUF_SIZE, 49152
.equ OUT_BUF_SIZE, 67000

_start:
    // ----- initialize total_len to 0 -----
    adr x5, total_len          // x5 = address of total_len
    str xzr, [x5]              // *total_len = 0

    // ----- read input loop -----
    adr x4, input_buf         // x4 = base address of input buffer

read_loop:
    ldr x7, [x5]               // x7 = current total_len
    add x1, x4, x7            // x1 = destination pointer = input_buf + total_len
    mov x2, #BUF_SIZE
    sub x2, x2, x7            // x2 = remaining space = BUF_SIZE - total_len
    mov x0, #0                // fd = 0 (stdin)
    mov x8, #63               // sys_read
    svc #0
    cbz x0, read_done         // if read returned 0 -> end of input
    // add bytes read to total_len
    ldr x7, [x5]
    add x7, x7, x0
    str x7, [x5]
    b read_loop

read_done:
    ldr x7, [x5]
    cbz x7, exit_zero         // if total_len == 0, exit without output

    // ----- encode -----
    adr x1, input_buf        // x1 = src pointer
    adr x2, output_buf       // x2 = dst pointer
    ldr x3, [x5]             // x3 = remaining bytes (total_len)
    mov x4, #0               // x4 = line length counter
    adr x5, b64_alphabet     // x5 = alphabet base

encode_full:
    cmp x3, #3
    blt encode_tail

    // load three bytes
    ldrb w0, [x1]
    add x1, x1, #1
    ldrb w1, [x1]
    add x1, x1, #1
    ldrb w2, [x1]
    add x1, x1, #1
    sub x3, x3, #3

    // w3 = w0 >> 2
    lsr w3, w0, #2
    // w4 = ((w0 & 3) << 4) | (w1 >> 4)
    and w6, w0, #3
    lsl w6, w6, #4
    lsr w7, w1, #4
    orr w4, w6, w7
    // w5 = ((w1 & 0xF) << 2) | (w2 >> 6)
    and w6, w1, #0xF
    lsl w6, w6, #2
    lsr w7, w2, #6
    orr w5, w6, w7
    // w6 = w2 & 0x3F
    and w6, w2, #0x3F

    // output w3
    uxtw x10, w3
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 1f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
1:
    // output w4
    uxtw x10, w4
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 2f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
2:
    // output w5
    uxtw x10, w5
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 3f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
3:
    // output w6
    uxtw x10, w6
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 4f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
4:
    b encode_full

encode_tail:
    // x3 = remaining bytes (0,1,2)
    cmp x3, #0
    beq encode_done
    cmp x3, #1
    beq tail_one
    // must be 2 bytes
    // ----- tail with two bytes -----
    ldrb w0, [x1]
    ldrb w1, [x1, #1]

    // w3 = w0 >> 2
    lsr w3, w0, #2
    // w4 = ((w0 & 3) << 4) | (w1 >> 4)
    and w6, w0, #3
    lsl w6, w6, #4
    lsr w7, w1, #4
    orr w4, w6, w7
    // w5 = (w1 & 0xF) << 2   (no further bits)
    and w5, w1, #0xF
    lsl w5, w5, #2

    // output w3
    uxtw x10, w3
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 5f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
5:
    // output w4
    uxtw x10, w4
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 6f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
6:
    // output w5
    uxtw x10, w5
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 7f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
7:
    // output '=' padding
    mov w7, #0x3d
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne encode_done
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
    b encode_done

tail_one:
    // ----- tail with one byte -----
    ldrb w0, [x1]

    // w3 = w0 >> 2
    lsr w3, w0, #2
    // w4 = (w0 & 3) << 4
    and w4, w0, #3
    lsl w4, w4, #4

    // output w3
    uxtw x10, w3
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 8f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
8:
    // output w4
    uxtw x10, w4
    ldrb w7, [x5, x10]
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 9f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
9:
    // output first '='
    mov w7, #0x3d
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne 10f
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
10:
    // output second '='
    mov w7, #0x3d
    strb w7, [x2], #1
    add x4, x4, #1
    cmp x4, #76
    bne encode_done
    mov w7, #10
    strb w7, [x2], #1
    mov x4, #0
    b encode_done

encode_done:
    // if line not empty, terminate with newline
    cmp x4, #0
    beq final_write
    mov w7, #10
    strb w7, [x2], #1

final_write:
    // write output buffer
    adr x9, output_buf
    sub x0, x2, x9          // x0 = length
    mov x1, x9
    mov x2, x0
    mov x8, #64
    svc #0

    // exit with status 0
    mov x0, #0
    mov x8, #93
    svc #0

exit_zero:
    mov x0, #0
    mov x8, #93
    svc #0

.section .bss
.align 3
input_buf:
    .zero BUF_SIZE
total_len:
    .quad 0
output_buf:
    .zero OUT_BUF_SIZE

.section
