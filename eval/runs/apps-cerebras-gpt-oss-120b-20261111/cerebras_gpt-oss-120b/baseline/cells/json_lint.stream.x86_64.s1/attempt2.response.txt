.text
.globl _start
_start:
    xor     rdi, rdi                       # fd 0
    lea     rsi, [rip + buffer]
    mov     rdx, 65536
read_loop:
    mov     rax, 0                         # sys_read
    syscall
    test    rax, rax
    je      read_done
    cmp     rax, -1
    jl      read_error
    mov     rbx, rax
    mov     rax, [rip + input_len]
    add     rax, rbx
    mov     [rip + input_len], rax
    cmp     rax, 65536
    jae     read_done
    jmp     read_loop
read_done:
    mov     rcx, 0                         # offset
    lea     rsi, [rip + buffer]            # pointer
    mov     rdx, [rip + input_len]         # length
    call    skip_ws
    call    parse_value
    cmp     al, 0
    je      fail
    call    skip_ws
    mov     rax, rcx
    cmp     rax, rdx
    jne     extra_data
    mov     rdi, 1
    lea     rsi, [rip + ok_msg]
    mov     rdx, 3
    mov     rax, 1
    syscall
    mov     rdi, 0
    mov     rax, 60
    syscall
extra_data:
    mov     rax, rcx
    mov     [rip + error_offset], rax
    jmp     report_error
fail:
    mov     rax, [rip + error_offset]
    test    rax, rax
    jne     report_error
    mov     rax, rcx
    mov     [rip + error_offset], rax
report_error:
    ; write "error at "
    mov     rdi, 1
    lea
