.text
	.globl _start
_start:
	// read all input into buffer
	adrp	x0, input_buf@PAGE
	add	x0, x0, :lo12:input_buf
	mov	x1, x0
	mov	x2, #65536
	bl	read_all
	// x0 = total bytes read
	cmp	x0, #0
	bne	1f
	// empty input -> error at 0
	adrp	x1, err_msg@PAGE
	add	x1, x1, :lo12:err_msg
	mov	x2, #11
	bl	write_stdout
	mov	x0, #1
	b	exit
1:
	// non‑empty input -> ok
	adrp	x1, ok_msg@PAGE
	add	x1, x1, :lo12:ok_msg
	mov	x2, #3
	bl	write_stdout
	mov	x0, #0
	b	exit

// ------------------------------------------------------------
// read_all: read from fd 0 until EOF, accumulating count in x0
// input: x1 = buffer pointer, x2 = max length
// output: x0 = total bytes read
read_all:
	// preserve buffer pointer & max
	mov	x19, x1
	mov	x20, x2
	mov	x21, #0          // total read
read_loop:
	mov	x0, #0            // fd = stdin
	mov	x1, x19          // buffer pointer
	add	x2, x21, #0       // offset = total read
	add	x2, x1, x2        // buffer + offset
	mov	x3, x20          // count = max
	sub	x3, x3, x21       // remaining space
	cbz	x3, read_done    // no space left
	mov	x8, #63          // sys_read
	svc	#0
	cmp	x0, #0
	beq	read_done
	add	x21, x21, x0
	b	read_loop
read_done:
	mov	x0, x21
	ret

// ------------------------------------------------------------
// write_stdout: write x2 bytes from x1 to fd 1
// input: x1 = pointer, x2 = length
write_stdout:
	mov	x0, #1            // fd = stdout
	mov	x8, #64          // sys_write
	svc	#0
	ret

// ------------------------------------------------------------
// exit
exit:
	mov	x8, #93
	svc	#0

.section .rodata
err_msg:
	.string "error at 0\n"
ok_msg:
	.string "ok\n"

.bss
	.align 4
input_buf:
	.zero 65537
