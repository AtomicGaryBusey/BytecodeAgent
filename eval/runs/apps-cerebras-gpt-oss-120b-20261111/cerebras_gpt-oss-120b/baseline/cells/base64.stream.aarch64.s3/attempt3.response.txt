.global _start
.text
.p2align 2

.equ MAX_INPUT, 49152
.equ LINE_MAX, 76
.equ NEWLINE, 10
.equ EXIT_SYSCALL, 93
.equ WRITE_SYSCALL, 64
.equ READ_SYSCALL, 63

_start:
    // input buffer pointer in x5
    adrp    x5, input_buf
    add     x5, x5, :lo12:input_buf
    mov     x1, x5                 // write pointer
    mov     x4, #0                 // total bytes read

read_loop:
    mov     x0, #0                 // fd = stdin
    mov     x2, #MAX_INPUT
    mov     x8, #READ_SYSCALL
    svc     #0
    cbz     x0, read_done
    add     x4, x4, x0            // total += bytes read
    add     x1, x1, x0            // advance write ptr
    b       read_loop

read_done:
    cbz     x4, exit_success       // empty input -> exit

    // set up pointers
    adrp    x6, input_buf
    add     x6, x6, :lo12:input_buf   // src pointer
    adrp    x7, output_buf
    add     x7, x7, :lo12:output_buf  // dst pointer
    adrp    x12, base64_alphabet
    add     x12, x12, :lo12:base64_alphabet // alphabet base

    mov     w15, #0               // line character count

    mov     x9, x4                // remaining input bytes

encode_loop:
    cmp     x9, #3
    b.lt    tail_encode

    // load three bytes
    ldrb    w0, [x6], #1
    ldrb    w1, [x6], #1
    ldrb    w2, [x6], #1

    // s0 = b0 >> 2
    mov     w3, w0, LSR #2

    // s1 = ((b0 & 3) << 4) | (b1 >> 4)
    and     w4, w0, #3
    lsl     w4, w4, #4
    mov     w5, w1, LSR #4
    orr     w4, w4, w5

    // s2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and     w5, w1, #0xF
    lsl     w5, w5, #2
    mov     w6, w2, LSR #6
    orr     w5, w5, w6

    // s3 = b2 & 0x3F
    and     w6, w2, #0x3F

    // ---- output s0 ----
    uxtw   x8, w3
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_char0
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_char0:

    // ---- output s1 ----
    uxtw   x8, w4
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_char1
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_char1:

    // ---- output s2 ----
    uxtw   x8, w5
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_char2
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_char2:

    // ---- output s3 ----
    uxtw   x8, w6
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_char3
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_char3:

    sub    x9, x9, #0            // placeholder to keep macro alignment
    sub    x9, x9, #0            // (no op)

    sub    x9, x9, #0            // (no op)

    sub    x9, x9, #0            // (no op)

    sub    x9, x9, #0            // (no op)

    sub    x9, x9, #0            // (no op)

    // decrement remaining count
    sub    x9, x9, #3
    b       encode_loop

tail_encode:
    cmp     x9, #0
    beq     finish_output
    cmp     x9, #1
    beq     tail_one
    // tail two bytes
    // load two bytes
    ldrb    w0, [x6], #1
    ldrb    w1, [x6], #1

    // s0 = b0 >> 2
    mov     w3, w0, LSR #2

    // s1 = ((b0 & 3) << 4) | (b1 >> 4)
    and     w4, w0, #3
    lsl     w4, w4, #4
    mov     w5, w1, LSR #4
    orr     w4, w4, w5

    // s2 = ((b1 & 0xF) << 2)
    and     w5, w1, #0xF
    lsl     w5, w5, #2

    // output s0
    uxtw   x8, w3
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t2_0
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t2_0:

    // output s1
    uxtw   x8, w4
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t2_1
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t2_1:

    // output s2
    uxtw   x8, w5
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t2_2
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t2_2:

    // output padding '='
    mov    w8, #'='
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t2_pad
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t2_pad:

    b       finish_output

tail_one:
    // load one byte
    ldrb    w0, [x6], #1

    // s0 = b0 >> 2
    mov     w3, w0, LSR #2

    // s1 = (b0 & 3) << 4
    and     w4, w0, #3
    lsl     w4, w4, #4

    // output s0
    uxtw   x8, w3
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t1_0
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t1_0:

    // output s1
    uxtw   x8, w4
    add    x9, x12, x8
    ldrb   w8, [x9]
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t1_1
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t1_1:

    // output two padding '='
    mov    w8, #'='
    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t1_pad1
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t1_pad1:

    strb   w8, [x7], #1
    add    w15, w15, #1
    cmp    w15, #LINE_MAX
    b.ne   after_t1_pad2
    mov    w8, #NEWLINE
    strb   w8, [x7], #1
    mov    w15, #0
after_t1_pad2:

    b       finish_output

finish_output:
    // add final newline if needed
    cmp     w15, #0
    beq     do_write
    mov     w8, #NEWLINE
    strb    w8, [x7], #1

do_write:
    // compute output length
    adrp   x0, output_buf
    add    x0, x0, :lo12:output_buf
    mov    x1, x0
    sub    x2, x7, x0      // length

    mov    x8, #WRITE_SYSCALL
    mov    x0, #1          // stdout
    svc    #0

exit_success:
    mov    x0, #0
    mov    x8, #EXIT_SYSCALL
    svc    #0

// -------------------------------------------------
.section .rodata
.base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
input_buf:
    .zero MAX_INPUT
output_buf:
    .zero 66000

)
