.global _start
.text
.align 2

_start:
    // Read input into input_buf
    adrp    x5, input_buf          // x5 = base address of input_buf (high bits)
    add     x5, x5, :lo12:input_buf
    mov     x4, #0                 // total length = 0
    mov     x1, x5                 // current write pointer = input_buf

read_loop:
    mov     x0, #0                 // fd = stdin
    mov     x2, #49152             // max read size
    mov     x8, #63                // sys_read
    svc     #0
    cbz     x0, read_done          // if read returned 0, done
    // x0 = bytes read
    add     x4, x4, x0             // total length += bytes read
    add     x1, x1, x0             // advance write pointer
    b       read_loop

read_done:
    // If no input, exit silently
    cbz     x4, exit_success

    // Prepare pointers for encoding
    adrp    x6, input_buf
    add     x6, x6, :lo12:input_buf // x6 = src pointer
    adrp    x7, output_buf
    add     x7, x7, :lo12:output_buf // x7 = dst pointer
    mov     w8, #0                 // line counter = 0
    mov     x9, x4                 // remaining = total length

encode_loop:
    cmp     x9, #3
    blt     tail_encode

    // Load three bytes
    ldrb    w10, [x6], #1          // b0
    ldrb    w11, [x6], #1          // b1
    ldrb    w12, [x6], #1          // b2

    // Compute sextets
    lsr     w13, w10, #2           // sextet0
    and     w14, w10, #0x3
    lsl     w14, w14, #4
    lsr     w15, w11, #4
    orr     w14, w14, w15          // sextet1
    and     w15, w11, #0xF
    lsl     w15, w15, #2
    lsr     w16, w12, #6
    orr     w15, w15, w16          // sextet2
    and     w16, w12, #0x3F        // sextet3

    // Output sextet0
    adrp    x0, base64_alphabet
    add     x0, x0, :lo12:base64_alphabet
    add     x1, x0, w13
    ldrb    w2, [x1]
    strb
