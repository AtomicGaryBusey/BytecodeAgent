.global _start
.text
.p2align 2

.equ MAX_INPUT, 49152
.equ LINE_MAX, 76
.equ NEWLINE, 10
.equ SYS_READ, 63
.equ SYS_WRITE, 64
.equ SYS_EXIT, 93

_start:
    // load pointers to buffers
    adrp    x9, input_buf
    add     x9, x9, :lo12:input_buf          // x9 = input_ptr (write pointer)
    mov     x10, x9                          // x10 = start of input buffer
    mov     x11, #0                          // total bytes read

read_loop:
    mov     x0, #0                           // fd = stdin
    mov     x1, x9                           // buf = current write ptr
    mov     x2, #MAX_INPUT
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done                    // if 0 bytes read, done
    add     x11, x11, x0                     // total += bytes read
    add     x9, x9, x0                       // advance write ptr
    b       read_loop

read_done:
    cbz     x11, exit_success                // empty input -> exit

    // set up src and dst pointers
    mov     x12, x10                         // x12 = src ptr (input start)
    adrp    x13, output_buf
    add     x13, x13, :lo12:output_buf       // x13 = dst ptr (output)
    adrp    x14, base64_alphabet
    add     x14, x14, :lo12:base64_alphabet   // x14 = alphabet base

    mov     w15, #0                          // line character count

    mov     x16, x11                         // remaining bytes count

encode_loop:
    cmp     x16, #3
    b.lt    tail_encode

    // load three bytes
    ldrb    w0, [x12], #1
    ldrb    w1, [x12], #1
    ldrb    w2, [x12], #1

    // s0 = b0 >> 2
    mov     w3, w0, LSR #2

    // s1 = ((b0 & 3) << 4) | (b1 >> 4)
    and     w4, w0, #3
    lsl     w4, w4, #4
    mov     w5, w1, LSR #4
    orr     w4, w4, w5

    // s2 = ((b1 & 0xF) << 2) | (b2 >> 6)
    and     w5, w1, #0xF
    lsl     w5, w5, #2
    mov     w6, w2, LSR #6
    orr     w5, w5, w6

    // s3 = b2 & 0x3F
    and     w6, w2, #0x3F

    // output s0..s3 with line handling
    // macro-like inline for each index
    // index 0
    uxtw    x0, w3
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after0
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after0:

    // index 1
    uxtw    x0, w4
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after1
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after1:

    // index 2
    uxtw    x0, w5
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after2
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after2:

    // index 3
    uxtw    x0, w6
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after3
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after3:

    sub     x16, x16, #3
    b       encode_loop

tail_encode:
    cmp     x16, #0
    beq     finish_output
    cmp     x16, #1
    beq     tail_one
    // tail two bytes
    ldrb    w0, [x12], #1
    ldrb    w1, [x12], #1

    // s0 = b0 >> 2
    mov     w3, w0, LSR #2

    // s1 = ((b0 & 3) << 4) | (b1 >> 4)
    and     w4, w0, #3
    lsl     w4, w4, #4
    mov     w5, w1, LSR #4
    orr     w4, w4, w5

    // s2 = ((b1 & 0xF) << 2)
    and     w5, w1, #0xF
    lsl     w5, w5, #2

    // output s0
    uxtw    x0, w3
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t2_0
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t2_0:

    // output s1
    uxtw    x0, w4
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t2_1
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t2_1:

    // output s2
    uxtw    x0, w5
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t2_2
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t2_2:

    // padding '='
    mov     w0, #'='
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t2_pad
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t2_pad:
    b       finish_output

tail_one:
    // tail one byte
    ldrb    w0, [x12], #1

    // s0 = b0 >> 2
    mov     w3, w0, LSR #2

    // s1 = (b0 & 3) << 4
    and     w4, w0, #3
    lsl     w4, w4, #4

    // output s0
    uxtw    x0, w3
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t1_0
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t1_0:

    // output s1
    uxtw    x0, w4
    add     x0, x14, x0
    ldrb    w0, [x0]
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t1_1
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t1_1:

    // padding '=' twice
    mov     w0, #'='
    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t1_pad1
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t1_pad1:

    strb    w0, [x13], #1
    add     w15, w15, #1
    cmp     w15, #LINE_MAX
    b.ne    after_t1_pad2
    mov     w0, #NEWLINE
    strb    w0, [x13], #1
    mov     w15, #0
after_t1_pad2:
    b       finish_output

finish_output:
    // ensure final newline if line not empty
    cmp     w15, #0
    beq     do_write
    mov     w0, #NEWLINE
    strb    w0, [x13], #1

do_write:
    // compute length = x13 - output_buf
    adrp   x0, output_buf
    add    x0, x0, :lo12:output_buf
    mov    x1, x0
    sub    x2, x13, x0          // length

    mov    x8, #SYS_WRITE
    mov    x0, #1              // stdout
    svc    #0

exit_success:
    mov    x0, #0
    mov    x8, #SYS_EXIT
    svc    #0

.section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 4
input_buf:
    .zero MAX_INPUT
output_buf:
    .zero 66000
