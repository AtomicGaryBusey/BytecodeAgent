.global _start
.text
.p2align 2

_start:
    // Load address of input buffer
    adrp    x5, input_buf
    add     x5, x5, :lo12:input_buf      // x5 = input_buf base
    mov     x4, #0                      // x4 = total bytes read
    mov     x1, x5                      // x1 = current write pointer

read_loop:
    mov     x0, #0                      // fd = stdin
    mov     x2, #49152                  // max read size
    mov     x8, #63                     // sys_read
    svc     #0
    cbz     x0, read_done               // if read returned 0, done
    // x0 = bytes read
    add     x4, x4, x0                  // total length += bytes read
    add     x1, x1, x0                  // advance write pointer
    b       read_loop

read_done:
    // If no input, exit silently
    cbz     x4, exit_success

    // Prepare pointers
    adrp    x6, input_buf
    add     x6, x6, :lo12:input_buf      // x6 = src pointer
    adrp    x7, output_buf
    add     x7, x7, :lo12:output_buf     // x7 = dst pointer (current)
    adrp    x12, base64_alphabet
    add     x12, x12, :lo12:base64_alphabet // x12 = alphabet base

    mov     w15, #0                     // line character count
    mov     x10, #0                     // total output length

    mov     x9, x4                      // x9 = remaining input bytes

encode_loop:
    cmp     x9, #3
    blt     tail_encode

    // Load three input bytes
    ldrb    w0, [x6], #1                 // b0
    ldrb    w1, [x6], #1                 // b1
    ldrb    w2, [x6], #1                 // b2

    // Compute sextets
    lsr     w3, w0, #2                  // s0
    and     w4, w0, #0x3
    lsl     w4, w4, #4
    lsr     w5, w1, #4
    orr     w4, w4, w5                  // s1
    and     w5, w1, #0xF
    lsl     w5, w5, #2
    lsr     w6, w2, #6
    orr     w5, w5, w6                  // s2
    and     w6, w2, #0x3F               // s3

    // Output s0
    ldrb    w8, [x12, w3]
    strb    w8,
