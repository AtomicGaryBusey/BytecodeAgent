.global _start
.section .bss
.align 8
buffer_in: .zero 65536
buffer_out: .zero 131072
.section .text
_start:
    xor rbx, rbx          # in_len = 0
    xor r12, r12          # out_len = 0
read_loop:
    mov rax, 0            # sys_read
    mov rdi, 0            # fd 0
    lea rsi, [rip + buffer_in]
    add rsi, rbx
    mov rdx, 65536
    sub rdx, rbx          # remaining space
    syscall
    test rax, rax
    jz read_done
    add rbx, rax
    cmp rbx, 65536
    jl read_loop
read_done:
    xor rdx, rdx          # pos = 0
process_loop:
    cmp rdx, rbx
    jge process_done
    mov rcx, rdx          # scan pointer
scan_newline:
    cmp rcx, rbx
    je no_newline
    mov al, byte ptr [rip + buffer_in + rcx]
    cmp al, 10
    je have_newline
    inc rcx
    jmp scan_newline
have_newline:
    mov r9, rcx           # line_end = rcx
    jmp copy_line
no_newline:
    mov r9, rbx           # line_end = in_len
copy_line:
    mov r8, r9
    sub r8, rdx           # len = line_end - line_start
    xor rcx, rcx          # i = 0
copy_rev_loop:
    cmp rcx, r8
    jae copy_rev_done
    mov rax, r9
    dec rax
    sub rax, rcx          # src offset
    mov al, byte ptr [rip + buffer_in + rax]
    mov rsi, r12
    mov byte ptr [rip + buffer_out + rsi], al
    inc r12
    inc rcx
    jmp copy_rev_loop
copy_rev_done:
    mov byte ptr [rip + buffer_out + r12], 10
    inc r12
    mov rdx, r9
    inc rdx
    jmp process_loop
process_done:
    mov rax, 1            # sys_write
    mov rdi, 1            # fd 1
    lea rsi, [rip + buffer_out]
    mov rdx, r12
    syscall
    mov rax, 60           # sys_exit
    xor rdi, rdi
    syscall
