    .section .bss
    .align 8
buf:
    .space 65536
linebuf:
    .space 80

    .section .text
    .global _start
_start:
    // total_len = 0
    mov     x19, #0
    // buf pointer
    adr     x21, buf
    // write pointer = buf
    mov     x22, x21

read_loop:
    mov     x0, #0          // fd = stdin
    mov     x1, x22         // buffer
    // remaining = 65536 - total_len
    mov     x2, #65536
    sub     x2, x2, x19
    mov     x8, #63         // sys_read
    svc     #0
    cbz     x0, read_done   // if 0 bytes read, finished
    // x0 = bytes read
    add     x19, x19, x0    // total_len += bytes read
    add     x22, x22, x0    // advance write pointer
    b       read_loop

read_done:
    // offset = 0
    mov     x20, #0
    // line buffer pointer
    adr     x6, linebuf

dump_loop:
    cmp     x20, x19
    b.ge    dump_done

    // n = min(16, total_len - offset)
    sub     x4, x19, x20        // remaining bytes
    mov     x5, #16
    cmp     x4, #16
    b.lt    set_n_remaining
    b       have_n
set_n_remaining:
    mov     x5, x4
have_n:
    // src_ptr = buf + offset
    add     x10, x21, x20
    mov     x15, x10            // save for ascii column
    // line buffer pointer in x1
    mov     x1, x6

    // ---- print offset (8 hex digits) ----
    mov     x0, x20             // copy offset
    mov     x7, #8
offset_hex_loop:
    // top nibble
    mov     x8, x0, lsr #28
    and     x8, x8, #0xf
    // numeric case
    add     w9, w8, #48
    cmp     w8, #9
    b.gt    offset_alpha
    b       offset_store
offset_alpha:
    add     w9, w8, #87
offset_store:
    strb    w9, [x1], #1
    // shift left 4
    lsl     x0, x0, #4
    subs    x7, x7, #1
    b.ne    offset_hex_loop

    // two spaces after offset
    mov     w9, #' '
    strb    w9, [x1], #1
    strb    w9, [x1], #1

    // ---- hex byte column ----
    mov     x11, #0
hex_byte_loop:
    cmp     x11, #16
    b.ge    after_hex_bytes
    cmp     x11, x5
    b.ge    missing_byte

    // have a real byte
    ldrb    w12, [x10], #1
    // high nibble
    mov     w13, w12, lsr #4
    and     w13, w13, #0xf
    add     w14, w13, #48
    cmp     w13, #9
    b.gt    hex_high_alpha
    b       hex_high_store
hex_high_alpha:
    add     w14, w13, #87
hex_high_store:
    strb    w14, [x1], #1
    // low nibble
    and     w13, w12, #0xf
    add     w14, w13, #48
    cmp     w13, #9
    b.gt    hex_low_alpha
    b       hex_low_store
hex_low_alpha:
    add     w14, w13, #87
hex_low_store:
    strb    w14, [x1], #1
    // space after byte
    mov     w14, #' '
    strb    w14, [x1], #1
    b       inc_hex_i

missing_byte:
    // three spaces for missing byte
    mov     w14, #' '
    strb    w14, [x1], #1
    strb    w14, [x1], #1
    strb    w14, [x1], #1
    b       inc_hex_i

inc_hex_i:
    // extra space after the eighth byte (i == 7)
    cmp     x11, #7
    b.ne    inc_i_done
    mov     w14, #' '
    strb    w14, [x1], #1
inc_i_done:
    add     x11, x11, #1
    b       hex_byte_loop

after_hex_bytes:
    // space then '|'
    mov     w14, #' '
    strb    w14, [x1], #1
    mov     w14, #'|'
    strb    w14, [x1], #1

    // ---- ASCII column ----
    mov     x11, #0
ascii_loop:
    cmp     x11, x5
    b.ge    after_ascii
    ldrb    w12, [x15], #1
    cmp     w12, #0x20
    b.lt    ascii_dot
    cmp     w12, #0x7e
    b.gt    ascii_dot
    // printable
    strb    w12, [x1], #1
    b       inc_ascii
ascii_dot:
    mov     w12, #'.'
    strb    w12, [x1], #1
inc_ascii:
    add     x11, x11, #1
    b       ascii_loop

after_ascii:
    // closing pipe and newline
    mov     w14, #'|'
    strb    w14, [x1], #1
    mov     w14, #10
    strb    w14, [x1], #1

    // write the line
    sub     x8, x1, x6          // length = current - linebuf
    mov     x0, #1
    mov     x1, x6
    mov     x2, x8
    mov     x8, #64
    svc     #0

    // advance offset by n
    add     x20, x20, x5
    b       dump_loop

dump_done:
    // ---- final total length line ----
    mov     x1, x6               // linebuf start
    // convert total_len (x19) to 8 hex digits
    mov     x0, x19
    mov     x7, #8
total_hex_loop:
    mov     x8, x0, lsr #28
    and     x8, x8, #0xf
    add     w9, w8, #48
    cmp     w8, #9
    b.gt    total_alpha
    b       total_store
total_alpha:
    add     w9, w8, #87
total_store:
    strb    w9, [x1], #1
    lsl     x0, x0, #4
    subs    x7, x7, #1
    b.ne    total_hex_loop

    // newline
    mov     w9, #10
    strb    w9, [x1], #1

    sub     x8, x1, x6
    mov     x0, #1
    mov     x1, x6
    mov     x2, x8
    mov     x8, #64
    svc     #0

    // exit
    mov     x0, #0
    mov     x8, #93
    svc     #0
