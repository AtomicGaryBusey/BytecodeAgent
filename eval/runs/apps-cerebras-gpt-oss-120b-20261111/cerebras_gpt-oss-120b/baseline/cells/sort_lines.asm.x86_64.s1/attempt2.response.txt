.globl _start
.equ BUF_SIZE, 32768
.equ MAX_LINES, 1000

.section .bss
.align 8
buf:        .zero BUF_SIZE
.align 8
line_ptrs:  .zero MAX_LINES*8
.align 4
line_lens:  .zero MAX_LINES*4
.align 8
line_cnt:   .zero 8

.section .rodata
newline:    .byte 0x0A

.text
_start:
    # read entire input into buf
    xor rbx, rbx                 # rbx = total bytes read
read_loop:
    mov rax, 0                  # sys_read
    mov rdi, 0                  # fd 0
    lea rsi, [rip + buf]
    add rsi, rbx                # buf + bytes already read
    mov rdx, BUF_SIZE
    sub rdx, rbx                # remaining space
    syscall
    test rax, rax
    je read_done
    add rbx, rax                # update total
    cmp rbx, BUF_SIZE
    jl read_loop
read_done:
    mov rdx, rbx                # rdx = total_len

    # parse lines
    xor r9, r9                  # r9 = line count
    xor r8, r8                  # r8 = line start offset
    xor rcx, rcx                # rcx = index i
parse_loop:
    cmp rcx, rdx
    je parse_done
    mov al, byte ptr [rip + buf + rcx]
    cmp al, 0x0A
    je newline_found
    inc rcx
    jmp parse_loop
newline_found:
    mov rax, rcx
    sub rax, r8                 # length = i - line_start
    lea rsi, [rip + buf + r8]   # pointer to line start
    mov [rel line_ptrs + r9*8], rsi
    mov [rel line_lens + r9*4], eax
    inc r9
    inc rcx
    mov r8, rcx                 # next line start = i+1
    jmp parse_loop
parse_done:
    cmp r8, rdx
    je no_final_line
    mov rax, rdx
    sub rax, r8                 # length = total_len - line_start
    lea rsi, [rip + buf + r8]
    mov [rel line_ptrs + r9*8], rsi
    mov [rel line_lens + r9*4], eax
    inc r9
no_final_line:
    mov [rel line_cnt], r9

    # sort lines (insertion sort)
    mov rdx, qword ptr [rel line_cnt]        # rdx = number of lines
    cmp rdx, 1
    jle sorted_done

    xor rsi, rsi               # i = 0
    inc rsi                    # i = 1
outer_ins:
    mov r10, qword ptr [rel line_ptrs + rsi*8]   # key pointer
    mov r11d, dword ptr [rel line_lens + rsi*4]  # key length
    mov rbx, rsi
    dec rbx                    # j = i-1
inner_loop:
    cmp rbx, -1
    jl insert_place
    mov rdi, qword ptr [rel line_ptrs + rbx*8]    # ptr_j
    mov rsi, dword ptr [rel line_lens + rbx*4]    # len_j
    mov rdx, r10                    # key ptr
    mov rcx, r11                    # key len
    call cmp_lines
    cmp rax, 0
    jle insert_place                # if line_j <= key, stop shifting
    # shift line_j up
    mov r8, qword ptr [rel line_ptrs + rbx*8]
    lea rax, [rbx+1]
    mov [rel line_ptrs + rax*8], r8
    mov r8d, dword ptr [rel line_lens + rbx*4]
    mov [rel line_lens + rax*4], r8d
    dec rbx
    jmp inner_loop
insert_place:
    lea rax, [rbx+1]
    mov [rel line_ptrs + rax*8], r10
    mov [rel line_lens + rax*4], r11d
    inc rsi
    cmp rsi, rdx
    jl outer_ins
sorted_done:
    # output sorted lines
    xor rbx, rbx                 # index = 0
out_loop:
    cmp rbx, qword ptr [rel line_cnt]
    jge out_done
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd stdout
    mov rsi, qword ptr [rel line_ptrs + rbx*8]
    mov rdx, dword ptr [rel line_lens + rbx*4]
    syscall
    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + newline]
    mov rdx, 1
    syscall
    inc rbx
    jmp out_loop
out_done:
    mov rax, 60                 # sys_exit
    xor rdi, rdi                # status 0
    syscall

# ------------------------------------------------------------
# cmp_lines:
#   rdi = ptr_a, rsi = len_a, rdx = ptr_b, rcx = len_b
#   returns rax = -1 if a<b, 0 if equal, 1 if a>b
# ------------------------------------------------------------
cmp_lines:
    push rbx
    xor rax, rax
    xor r8, r8                  # index = 0
cmp_loop:
    cmp r8, rsi
    jae len_done
    cmp r8, rcx
    jae len_done
    mov al, byte ptr [rdi + r8]
    mov bl, byte ptr [rdx + r8]
    cmp al, bl
    jne diff
    inc r8
    jmp cmp_loop
diff:
    ja  greater
    mov rax, -1
    pop rbx
    ret
greater:
    mov rax, 1
    pop rbx
    ret
len_done:
    cmp rsi, rcx
    je equal
    jl shorter
    mov rax, 1
    pop rbx
    ret
shorter:
    mov rax, -1
    pop rbx
    ret
equal:
    xor rax, rax
    pop rbx
    ret
