.global _start

.section .rodata
hex_table:
    .ascii "0123456789abcdef"

.section .bss
.align 16
buf:        .zero 65536
line_buf:   .zero 80

.section .text
_start:
    # pointers
    lea r15, [rip + buf]          # r15 = base of input buffer
    lea r13, [rip + line_buf]     # r13 = line buffer
    lea r14, [rip + hex_table]    # r14 = hex table

    xor r8, r8                    # r8 = total_len
    mov rbx, r15                  # rbx = write pointer

read_loop:
    mov rax, 0                    # sys_read
    xor rdi, rdi                  # fd = 0
    mov rsi, rbx                  # buf ptr
    mov rdx, 65536                # max count
    syscall
    test rax, rax
    jz read_done
    add r8, rax                   # total_len += bytes_read
    add rbx, rax                  # advance write pointer
    jmp read_loop
read_done:

    xor r12, r12                  # r12 = current offset

line_loop:
    cmp r12, r8
    jae final_offset_line

    # remaining = total_len - offset
    mov rcx, r8
    sub rcx, r12                  # rcx = remaining
    mov r11, rcx                  # r11 = row_len (may be >16)
    mov rcx, 16
    cmp r11, rcx
    jl rowlen_small
    mov r11, rcx                  # row_len = 16
rowlen_small:

    # rbx = buf + offset
    lea rbx, [r15 + r12]

    # line pointer
    mov rsi, r13

    # ----- write offset (8 hex digits) -----
    mov eax, r12d
    shr eax, 28
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    shr eax, 24
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    shr eax, 20
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    shr eax, 16
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    shr eax, 12
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    shr eax, 8
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    shr eax, 4
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r12d
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    # two spaces after offset
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi

    # ----- hex bytes -----
    xor rcx, rcx                  # i = 0
hex_loop:
    cmp rcx, 16
    je hex_done
    cmp rcx, r11
    jl have_byte
    # three spaces for missing byte
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], ' '
    inc rsi
    jmp after_hex_byte
have_byte:
    movzx eax, byte ptr [rbx + rcx]
    # high nibble
    mov edx, eax
    shr edx, 4
    and edx, 0xF
    movzx r10d, byte ptr [r14 + rdx]
    mov [rsi], r10b
    inc rsi
    # low nibble
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi
    # space
    mov byte ptr [rsi], ' '
    inc rsi
    # extra space after 8th byte
    cmp rcx, 7
    jne after_extra_space
    mov byte ptr [rsi], ' '
    inc rsi
after_extra_space:
after_hex_byte:
    inc rcx
    jmp hex_loop
hex_done:

    # space and '|'
    mov byte ptr [rsi], ' '
    inc rsi
    mov byte ptr [rsi], '|'
    inc rsi

    # ----- ASCII part -----
    xor rcx, rcx                  # i = 0
ascii_loop:
    cmp rcx, r11
    je ascii_done
    movzx eax, byte ptr [rbx + rcx]
    mov ebp, eax
    cmp al, 0x20
    jl dot_char
    cmp al, 0x7e
    jg dot_char
    # printable
    mov [rsi], al
    inc rsi
    jmp after_ascii_char
dot_char:
    mov byte ptr [rsi], '.'
    inc rsi
after_ascii_char:
    inc rcx
    jmp ascii_loop
ascii_done:

    # closing '|' and newline
    mov byte ptr [rsi], '|'
    inc rsi
    mov byte ptr [rsi], 0x0a
    inc rsi

    # write the line
    mov rcx, rsi
    sub rcx, r13                  # rcx = line length
    mov rax, 1                    # sys_write
    mov rdi, 1                    # fd = stdout
    mov rsi, r13
    mov rdx, rcx
    syscall

    # advance offset
    add r12, r11
    jmp line_loop

final_offset_line:
    # print total length (r8) as 8 hex digits + newline
    mov rsi, r13

    mov eax, r8d
    shr eax, 28
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    shr eax, 24
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    shr eax, 20
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    shr eax, 16
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    shr eax, 12
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    shr eax, 8
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    shr eax, 4
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov eax, r8d
    and eax, 0xF
    movzx r10d, byte ptr [r14 + rax]
    mov [rsi], r10b
    inc rsi

    mov byte ptr [rsi], 0x0a   # newline
    inc rsi

    mov rcx, rsi
    sub rcx, r13                # rcx = line length (9)
    mov rax, 1                  # sys_write
    mov rdi, 1
    mov rsi, r13
    mov rdx, rcx
    syscall

    # exit
    mov rax, 60                 # sys_exit
    xor rdi, rdi               # status 0
    syscall
