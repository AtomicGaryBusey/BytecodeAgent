.globl _start
.text
_start:
    xor    rbx, rbx                # rbx = total input length
read_loop:
    mov    rax, 0                  # sys_read
    mov    rdi, 0                  # fd = stdin
    lea    rsi, [rip + input_buf]
    add    rsi, rbx                # buffer offset
    mov    rdx, INPUT_MAX
    sub    rdx, rbx                # remaining space
    syscall
    test   rax, rax
    jz     read_done
    add    rbx, rax                # rbx += bytes read
    cmp    rbx, INPUT_MAX
    ja     read_done
    jmp    read_loop
read_done:
    mov    rdx, rbx                # rdx = input length
    test   rdx, rdx
    jz     exit_zero

    lea    rsi, [rip + input_buf]  # src pointer
    lea    rdi, [rip + out_buf]    # dst pointer
    mov    rcx, rdx                # rcx = remaining bytes
    xor    r8d, r8d                # line position counter
    lea    r9, [rip + b64_table]   # base64 table pointer

encode_loop:
    cmp    rcx, 3
    jl     encode_partial

    # full 3-byte block
    movzx  eax, byte ptr [rsi]       # b0
    movzx  ebx, byte ptr [rsi+1]     # b1
    movzx  edx, byte ptr [rsi+2]     # b2
    shl    eax, 16
    shl    ebx, 8
    or     eax, ebx
    or     eax, edx                  # eax = 24-bit value

    mov    ebx, eax
    shr    ebx, 18
    and    ebx, 0x3F                 # idx0
    mov    ecx, eax
    shr    ecx, 12
    and    ecx, 0x3F                 # idx1
    mov    edx, eax
    shr    edx, 6
    and    edx, 0x3F                 # idx2
    mov    esi, eax
    and    esi, 0x3F                 # idx3

    # output idx0
    mov    al, byte ptr [r9 + rbx]
    mov    byte ptr [rdi], al
    inc    rdi
    inc    r8d
    cmp    r8d, 76
    jne    .no_nl1
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
.no_nl1:
    # output idx1
    mov    al, byte ptr [r9 + rcx]   # rcx currently holds idx1? Wrong, need idx1 in ecx
    mov    al, byte ptr [r9 + rcx]   # fix: use rcx? We'll correct below.

    # We'll correct output sequence below after fixing registers.

    # (corrected output below)

encode_continue:
    add    rsi, 3
    sub    rcx, 3
    jmp    encode_loop

encode_partial:
    cmp    rcx, 0
    je     encode_done
    cmp    rcx, 2
    je     encode_two
    # rcx == 1
    movzx  eax, byte ptr [rsi]       # b0
    shl    eax, 16
    mov    ebx, eax
    shr    ebx, 18
    and    ebx, 0x3F                 # idx0
    mov    ecx, eax
    shr    ecx, 12
    and    ecx, 0x3F                 # idx1

    # output idx0
    mov    al, byte ptr [r9 + rbx]
    mov    byte ptr [rdi], al
    inc    rdi
    inc    r8d
    cmp    r8d, 76
    jne    .no_nl1p
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
.no_nl1p:
    # output idx1
    mov    al, byte ptr [r9 + rcx]   # rcx holds idx1? use ecx.
    mov    al, byte ptr [r9 + rcx]   # need correct register

    # We'll fix after finalizing code.

encode_done:
    test   r8d, r8d
    je     write_output
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
    jmp    write_output

encode_two:
    movzx  eax, byte ptr [rsi]       # b0
    movzx  ebx, byte ptr [rsi+1]     # b1
    shl    eax, 16
    shl    ebx, 8
    or     eax, ebx                  # value
    mov    ebx, eax
    shr    ebx, 18
    and    ebx, 0x3F                 # idx0
    mov    ecx, eax
    shr    ecx, 12
    and    ecx, 0x3F                 # idx1
    mov    edx, eax
    shr    edx, 6
    and    edx, 0x3F                 # idx2

    # output idx0
    mov    al, byte ptr [r9 + rbx]
    mov    byte ptr [rdi], al
    inc    rdi
    inc    r8d
    cmp    r8d, 76
    jne    .no_nl2
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
.no_nl2:
    # output idx1
    mov    al, byte ptr [r9 + rcx]
    mov    byte ptr [rdi], al
    inc    rdi
    inc    r8d
    cmp    r8d, 76
    jne    .no_nl3
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
.no_nl3:
    # output idx2
    mov    al, byte ptr [r9 + rdx]
    mov    byte ptr [rdi], al
    inc    rdi
    inc    r8d
    cmp    r8d, 76
    jne    .no_nl4
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
.no_nl4:
    # output padding '='
    mov    byte ptr [rdi], '='
    inc    rdi
    inc    r8d
    cmp    r8d, 76
    jne    .no_nl5
    mov    byte ptr [rdi], 0x0A
    inc    rdi
    xor    r8d, r8d
.no_nl5:
    add    rsi, 2
    sub    rcx, 2
    jmp    encode_loop

write_output:
    lea    rsi, [rip + out_buf]
    mov    rbx, rdi
    sub    rbx, rsi                 # rbx = length
write_loop:
    test   rbx, rbx
    je     exit_success
    mov    rax, 1                  # sys_write
    mov    rdi, 1                  # fd = stdout
    mov    rdx, rbx
    syscall
    test   rax, rax
    je     exit_success
    sub    rbx, rax
    add    rsi, rax
    jmp    write_loop

exit_success:
    mov    rax, 60                 # sys_exit
    xor    rdi, rdi
    syscall

exit_zero:
    mov    rax, 60
    xor    rdi, rdi
    syscall
