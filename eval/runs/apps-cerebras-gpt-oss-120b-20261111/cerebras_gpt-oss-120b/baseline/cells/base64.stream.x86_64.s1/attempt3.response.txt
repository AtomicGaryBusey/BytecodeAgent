.globl _start
.text
_start:
    xor     rbx, rbx                # total bytes read = 0
read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0                  # fd = stdin
    lea     rsi, [rip + input_buf]
    add     rsi, rbx                # buffer offset
    mov     rdx, INPUT_MAX
    sub     rdx, rbx                # remaining space
    syscall
    test    rax, rax
    jz      read_done
    add     rbx, rax                # accumulate length
    cmp     rbx, INPUT_MAX
    ja      read_done
    jmp     read_loop
read_done:
    test    rbx, rbx
    jz      exit_success            # empty input, no output

    lea     rsi, [rip + input_buf]  # src pointer
    lea     rdi, [rip + out_buf]    # dst pointer
    mov     rcx, rbx                # remaining input bytes
    xor     r8d, r8d                # column counter
    lea     r9, [rip + b64_table]   # base64 table

encode_loop:
    cmp     rcx, 3
    jl      encode_partial

    movzx   eax, byte ptr [rsi]          # b0
    shl     eax, 16
    movzx   ebx, byte ptr [rsi+1]        # b1
    shl     ebx, 8
    or      eax, ebx
    movzx   ebx, byte ptr [rsi+2]        # b2
    or      eax, ebx                      # eax = 24-bit value

    mov     ebx, eax
    shr     ebx, 18
    and     ebx, 0x3F                     # idx0
    mov     ecx, eax
    shr     ecx, 12
    and     ecx, 0x3F                     # idx1
    mov     edx, eax
    shr     edx, 6
    and     edx, 0x3F                     # idx2
    mov     esi, eax
    and     esi, 0x3F                     # idx3

    # output idx0
    mov     al, byte ptr [r9 + rbx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl0
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl0:
    # output idx1
    mov     al, byte ptr [r9 + rcx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl1
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl1:
    # output idx2
    mov     al, byte ptr [r9 + rdx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl2
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl2:
    # output idx3
    mov     al, byte ptr [r9 + rsi]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl3
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl3:
    add     rsi, 3
    sub     rcx, 3
    jmp     encode_loop

encode_partial:
    cmp     rcx, 2
    je      encode_two
    cmp     rcx, 1
    je      encode_one
    jmp     encode_done               # rcx == 0

encode_one:
    movzx   eax, byte ptr [rsi]          # b0
    shl     eax, 16
    mov     ebx, eax
    shr     ebx, 18
    and     ebx, 0x3F                     # idx0
    mov     ecx, eax
    shr     ecx, 12
    and     ecx, 0x3F                     # idx1

    # output idx0
    mov     al, byte ptr [r9 + rbx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl4
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl4:
    # output idx1
    mov     al, byte ptr [r9 + rcx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl5
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl5:
    # output padding '='
    mov     byte ptr [rdi], '='
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl6
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl6:
    jmp     encode_done

encode_two:
    movzx   eax, byte ptr [rsi]          # b0
    movzx   ebx, byte ptr [rsi+1]        # b1
    shl     eax, 16
    shl     ebx, 8
    or      eax, ebx                      # 16+8 bits value

    mov     ebx, eax
    shr     ebx, 18
    and     ebx, 0x3F                     # idx0
    mov     ecx, eax
    shr     ecx, 12
    and     ecx, 0x3F                     # idx1
    mov     edx, eax
    shr     edx, 6
    and     edx, 0x3F                     # idx2

    # output idx0
    mov     al, byte ptr [r9 + rbx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl7
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl7:
    # output idx1
    mov     al, byte ptr [r9 + rcx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl8
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl8:
    # output idx2
    mov     al, byte ptr [r9 + rdx]
    mov     byte ptr [rdi], al
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl9
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl9:
    # output padding '='
    mov     byte ptr [rdi], '='
    inc     rdi
    inc     r8d
    cmp     r8d, 76
    jne     .no_nl10
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d
.no_nl10:
    jmp     encode_done

encode_done:
    test    r8d, r8d
    je      write_output
    mov     byte ptr [rdi], 0x0A
    inc     rdi
    xor     r8d, r8d

write_output:
    lea     rsi, [rip + out_buf]
    mov     rbx, rdi
    sub     rbx, rsi                     # rbx = total output length
write_loop:
    test    rbx, rbx
    je      exit_success
    mov     rax, 1                       # sys_write
    mov     rdi, 1                       # fd = stdout
    mov     rdx, rbx
    syscall
    test    rax, rax
    je      exit_success
    sub     rbx, rax
    add     rsi, rax
    jmp     write_loop

exit_success:
    mov     rax, 60                      # sys_exit
    xor     rdi, rdi
    syscall

.section .rodata
b64_table:
    .ascii  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
.input_buf:
    .zero INPUT_MAX
.out_buf:
    .zero OUTPUT_MAX

.equ INPUT_MAX
