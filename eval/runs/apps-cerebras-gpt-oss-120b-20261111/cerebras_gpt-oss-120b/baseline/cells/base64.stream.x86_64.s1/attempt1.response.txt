.globl _start
.text
_start:
    xor     rbx, rbx                # total bytes read = 0
read_loop:
    mov     rax, 0                  # sys_read
    mov     rdi, 0
    lea     rsi, [rip + input_buf]
    add     rsi, rbx
    mov     rdx, MAX_IN
    sub     rdx, rbx
    syscall
    test    rax, rax
    je      read_done
    add     rbx, rax
    cmp     rbx, MAX_IN
    jae     read_done
    jmp     read_loop
read_done:
    mov     rdx, rbx                # rdx = input length
    test    rdx, rdx
    je      exit_zero

    lea     rbx, [rip + input_buf] # src pointer
    lea     rdi, [rip + output_buf]# dst pointer
    mov     r14, rdx                # remaining input bytes
    xor     r11, r11                # line position counter
    lea     r10, [rip + b64_table] # base64 table address

encode_loop:
    cmp     r14, 3
    jae     encode_full
    cmp     r14, 2
    je      encode_two
    cmp     r14, 1
    je      encode_one
    jmp     encode_done
