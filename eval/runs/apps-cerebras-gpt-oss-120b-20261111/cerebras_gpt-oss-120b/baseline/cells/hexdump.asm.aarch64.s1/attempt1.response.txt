        .global _start
        .text
_start:
        // x19 = total_len, x20 = write_ptr, x21 = buffer_base
        mov     x19, #0                     // total_len = 0
        adr     x21, inbuf                  // buffer base address
        mov     x20, x21                    // write_ptr = buffer base

read_loop:
        mov     x0, #0                      // fd = stdin
        mov     x1, x20                     // buf = write_ptr
        mov     x2, #65536                  // count
        mov     x8, #63                     // sys_read
        svc     #0
        cbz     x0, read_done               // if read returns 0 -> done
        // x0 = bytes read (positive)
        add     x19, x19, x0                // total_len += bytes_read
        add     x20, x20, x0                // write_ptr += bytes_read
        b       read_loop

read_done:
        // x21 = buffer base, x19 = total_len
        mov     x22, #0                     // offset = 0

process_loop:
        cmp     x22, x19
        bge     finish_dump                 // if offset >= total_len, done

        // remaining = total_len - offset
        sub     x23, x19, x22
        // n = min(16, remaining)
        mov     x25, #16                    // default 16
        cmp     x23, #16
        csel    x25, x23, x25, lt          // if remaining < 16, n = remaining

        // x10 = linebuf pointer
        adr     x10, linebuf

        // ---- print offset (8 hex digits) ----
        mov     x30, #8                     // digit count
        mov     x27, #28                    // shift amount
offset_hex_loop:
        lsr     x28, x22, x27
        and     x28, x28, #0xf
        adr     x9, hex_table
        add     x28, x9, x28
        ldrb    w0, [x28]
        strb    w0, [x10], #1
        sub     x27, x27, #4
        subs    x30, x30, #1
        bne     offset_hex_loop

        // two spaces after offset
        mov     w0, #' '
        strb    w0, [x10], #1
        strb    w0, [x10], #1

        // ---- bytes area (16 slots) ----
        mov     x31, #0                     // i = 0
byte_loop:
        cmp     x31, #16
        bge     after_bytes

        cmp     x31, x25
        bge     missing_byte

        // present byte
        add     x28, x21, x22               // base + offset
        add     x28, x28, x31               // + i
        ldrb    w0, [x28]                   // byte value

        // high nibble
        ubfx    w1, w0, #4, #4
        // low nibble
        and     w2, w0, #0xf

        adr     x9, hex_table
        add     x1, x9, x1
        ldrb    w3, [x1]
        strb    w3, [x10], #1

        add     x2, x9, x2
        ldrb    w3, [x2]
        strb    w3, [x10], #1

        // space after byte
        mov     w3, #' '
        strb    w3, [x10], #1
        b       after_byte_common

missing_byte:
        // three spaces for missing slot
        mov     w0, #' '
        strb    w0, [x10], #1
        strb    w0, [x10], #1
        strb    w0, [x10], #1

after_byte_common:
        // extra space after the 8th slot (i==7)
        cmp     x31, #7
        bne     continue_bytes
        mov     w0, #' '
        strb    w0, [x10], #1
continue_bytes:
        add     x31, x31, #1
        b       byte_loop

after_bytes:
        // space and '|'
        mov     w0, #' '
        strb    w0, [x10], #1
        mov     w0, #'|'
        strb    w0, [x10], #1

        // ---- ASCII representation ----
        mov     x31, #0                     // i = 0
ascii_loop:
        cmp     x31, x25
        bge     after_ascii
        add     x28, x21, x22
        add     x28, x28, x31
        ldrb    w0, [x28]

        // printable check 0x20..0x7e
        cmp     w0, #0x20
        blt     ascii_dot
        cmp     w0, #0x7e
        bgt     ascii_dot
        // printable
        strb    w0, [x10], #1
        b       ascii_next

ascii_dot:
        mov     w0, #'.'
        strb    w0, [x10], #1

ascii_next:
        add     x31, x31, #1
        b       ascii_loop

after_ascii:
        // trailing '|' and newline
        mov     w0, #'|'
        strb    w0, [x10], #1
        mov     w0, #10                     // '\n'
        strb    w0, [x10], #1

        // line length = 63 + n
        mov     x0, #63
        add     x0, x0, x25

        // write line
        mov     x1, linebuf
        mov     x2, x0
        mov     x8, #64                    // sys_write
        mov     x0, #1                     // fd = stdout
        svc     #0

        // advance offset by 16
        add     x22, x22, #16
        b       process_loop

finish_dump:
        // ---- print total length (8 hex digits + newline) ----
        adr     x10, linebuf
        // same offset hex loop but with x19 (total_len)
        mov     x30, #8
        mov     x27, #28
offset_hex_total_loop:
        lsr     x28, x19, x27
        and     x28, x28, #0xf
        adr     x9, hex_table
        add     x28, x9, x28
        ldrb    w0, [x28]
        strb    w0, [x10], #1
        sub     x27, x27, #4
        subs    x30, x30, #1
        bne     offset_hex_total_loop

        // newline
        mov     w0, #10
        strb    w0, [x10], #1

        // write total line (9 bytes)
        mov     x0, #9                      // length
        mov     x1, linebuf
        mov     x2, x0
        mov     x8, #64
        mov     x0, #1
        svc     #0

        // exit
        mov     x0, #0
        mov     x8, #93                     // sys_exit
        svc     #0

        .section .rodata
hex_table:
        .ascii  "0123456789abcdef"

        .bss
        .align 8
inbuf:
        .space 65536
linebuf:
        .space 80
