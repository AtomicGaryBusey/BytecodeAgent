.global _start
.text
_start:
    // initialize buffer pointers and total length
    adr     x21, inbuf          // buffer base
    mov     x20, x21            // write_ptr
    mov     x19, #0            // total_len = 0

read_loop:
    mov     x0, #0             // fd = stdin
    mov     x1, x20            // buf = write_ptr
    mov     x2, #65536        // count
    mov     x8, #63            // sys_read
    svc     #0
    cbz     x0, read_done
    add     x19, x19, x0       // total_len += bytes_read
    add     x20, x20, x0       // write_ptr += bytes_read
    b       read_loop

read_done:
    // offset = 0
    mov     x22, #0

process_loop:
    cmp     x22, x19
    bge     finish_dump

    sub     x23, x19, x22       // remaining = total_len - offset
    mov     x24, #16            // n = 16
    cmp     x23, #16
    csel    x24, x23, x24, lt   // if remaining < 16, n = remaining

    // line buffer pointer
    adr     x10, linebuf

    // ---- print offset (8 hex digits) ----
    mov     x11, #8
    mov     x12, #28
offset_hex_loop:
    lsr     x13, x22, x12
    and     x13, x13, #0xf
    adr     x14, hex_table
    add     x13, x14, x13
    ldrb    w0, [x13]
    strb    w0, [x10], #1
    sub     x12, x12, #4
    subs    x11, x11, #1
    bne     offset_hex_loop

    // two spaces after offset
    mov     w0, #' '
    strb    w0, [x10], #1
    strb    w0, [x10], #1

    // ---- bytes area (16 slots) ----
    mov     x18, #0               // i = 0
byte_loop:
    cmp     x18, #16
    bge     after_bytes
    cmp     x18, x24
    bge     missing_byte

    // present byte
    add     x13, x21, x22
    add     x13, x13, x18
    ldrb    w0, [x13]

    // high nibble
    ubfx    w1, w0, #4, #4
    adr     x14, hex_table
    add     x15, x14, x1
    ldrb    w1, [x15]
    strb    w1, [x10], #1

    // low nibble
    and     w1, w0, #0xf
    add     x15, x14, x1
    ldrb    w1, [x15]
    strb    w1, [x10], #1

    // space after byte
    mov     w1, #' '
    strb    w1, [x10], #1
    b       after_byte_common

missing_byte:
    // three spaces for missing slot
    mov     w1, #' '
    strb    w1, [x10], #1
    strb    w1, [x10], #1
    strb    w1, [x10], #1

after_byte_common:
    // extra space after the 8th slot (i == 7)
    cmp     x18, #7
    bne     continue_bytes
    mov     w1, #' '
    strb    w1, [x10], #1
continue_bytes:
    add     x18, x18, #1
    b       byte_loop

after_bytes:
    // space and '|'
    mov     w1, #' '
    strb    w1, [x10], #1
    mov     w1, #'|'
    strb    w1, [x10], #1

    // ---- ASCII representation ----
    mov     x18, #0
ascii_loop:
    cmp     x18, x24
    bge     after_ascii
    add     x13, x21, x22
    add     x13, x13, x18
    ldrb    w0, [x13]

    // printable check 0x20..0x7e
    cmp     w0, #0x20
    blt     ascii_dot
    cmp     w0, #0x7e
    bgt     ascii_dot
    // printable
    strb    w0, [x10], #1
    b       ascii_next

ascii_dot:
    mov     w0, #'.'
    strb    w0, [x10], #1

ascii_next:
    add     x18, x18, #1
    b       ascii_loop

after_ascii:
    // trailing '|' and newline
    mov     w0, #'|'
    strb    w0, [x10], #1
    mov     w0, #10
    strb    w0, [x10], #1

    // write line: length = 63 + n
    mov     x0, #63
    add     x0, x0, x24        // x0 = length including newline
    mov     x1, linebuf
    mov     x2, x0
    mov     x8, #64           // sys_write
    mov     x0, #1            // fd = stdout
    svc     #0

    // advance offset by 16
    add     x22, x22, #16
    b       process_loop

finish_dump:
    // ---- print total length (8 hex digits + newline) ----
    adr     x10, linebuf
    mov     x11, #8
    mov     x12, #28
total_hex_loop:
    lsr     x13, x19, x12
    and     x13, x13, #0xf
    adr     x14, hex_table
    add     x13, x14, x13
    ldrb    w0, [x13]
    strb    w0, [x10], #1
    sub     x12, x12, #4
    subs    x11, x11, #1
    bne     total_hex_loop

    mov     w0, #10
    strb    w0, [x10], #1

    // write total line (9 bytes)
    mov     x0, #9
    mov     x1, linebuf
    mov     x2, x0
    mov     x8, #64
    mov     x0, #1
    svc     #0

    // exit
    mov     x0, #0
    mov     x8, #93
    svc     #0

.section .rodata
hex_table:
    .ascii  "0123456789abcdef"

.bss
    .align 8
inbuf:
    .space 65536
linebuf:
    .space 80
