.globl _start
.text
_start:
    # ---------- read all stdin ----------
    lea r8, [rip + buf]          # r8 = buffer base
    xor rbx, rbx                 # total_len = 0
read_loop:
    mov rdx, 65536
    sub rdx, rbx                 # remaining space
    test rdx, rdx
    je read_done
    mov rdi, 0                   # fd = stdin
    mov rsi, r8                  # buf
    add rsi, rbx                 # buf + total_len
    xor rax, rax                 # sys_read = 0
    syscall
    test rax, rax
    je read_done
    add rbx, rax                 # total_len += bytes_read
    jmp read_loop
read_done:
    # rbx = total_len
    mov rdx, rbx                 # rdx = total_len for later use

    # ---------- split program / input ----------
    xor rcx, rcx                 # i = 0
find_excl:
    cmp rcx, rdx
    jae no_excl
    mov al, byte ptr [r8 + rcx]
    cmp al, 0x21                 # '!'
    je found_excl
    inc rcx
    jmp find_excl
found_excl:
    mov r10, rcx                 # prog_len = rcx
    lea r12, [r8 + rcx + 1]      # input_base = buf + rcx + 1
    mov r13, rdx                 # r13 = total_len
    sub r13, rcx
    dec r13                     # input_len = total_len - (rcx+1)
    jmp after_split
no_excl:
    mov r10, rdx                 # prog_len = total_len
    lea r12, [r8 + rdx]          # input_base = end of buffer
    xor r13, r13                 # input_len = 0
after_split:

    # ---------- set up pointers ----------
    lea r9, [rip + tape]         # tape base
    mov rsi, r9                  # data pointer = tape base
    xor r14, r14                 # input_pos = 0

    # ---------- bracket balance check ----------
    xor rcx, rcx                 # i = 0
    xor r15, r15                 # depth = 0
bal_check:
    cmp rcx, r10
    jge bal_done
    mov al, byte ptr [r8 + rcx]
    cmp al, '['
    je bal_inc
    cmp al, ']'
    je bal_dec
    inc rcx
    jmp bal_check
bal_inc:
    inc r15
    inc rcx
    jmp bal_check
bal_dec:
    test r15, r15
    je unbalanced_error
    dec r15
    inc rcx
    jmp bal_check
bal_done:
    test r15, r15
    jne unbalanced_error

    # ---------- execution ----------
    xor rbx, rbx                 # pc = 0
exec_loop:
    cmp rbx, r10
    jge exec_done

    mov al, byte ptr [r8 + rbx]

    cmp al, '>'
    je inc_ptr
    cmp al, '<'
    je dec_ptr
    cmp al, '+'
    je inc_cell
    cmp al, '-'
    je dec_cell
    cmp al, '.'
    je out_cell
    cmp al, ','
    je in_cell
    cmp al, '['
    je open_bracket
    cmp al, ']'
    je close_bracket

    # ignore any other character
    inc rbx
    jmp exec_loop

inc_ptr:
    inc rsi
    inc rbx
    jmp exec_loop

dec_ptr:
    dec rsi
    inc rbx
    jmp exec_loop

inc_cell:
    inc byte ptr [rsi]
    inc rbx
    jmp exec_loop

dec_cell:
    dec byte ptr [rsi]
    inc rbx
    jmp exec_loop

out_cell:
    mov rax, 1                  # sys_write
    mov rdi, 1                  # fd = stdout
    mov rdx, 1                  # count = 1
    syscall
    inc rbx
    jmp exec_loop

in_cell:
    cmp r14, r13
    jae in_eof
    mov al, byte ptr [r12 + r14]
    mov byte ptr [rsi], al
    inc r14
    inc rbx
    jmp exec_loop
in_eof:
    mov byte ptr [rsi], 0
    inc rbx
    jmp exec_loop

open_bracket:
    mov al, byte ptr [rsi]
    test al, al
    jne ob_not_zero
    # cell == 0, skip forward
    mov rcx, 1
ob_forward:
    inc rbx
    cmp rbx, r10
    jge unbalanced_error
    mov al, byte ptr [r8 + rbx]
    cmp al, '['
    je ob_inc_nest
    cmp al, ']'
    je ob_dec_nest
    jmp ob_check
ob_inc_nest:
    inc rcx
    jmp ob_check
ob_dec_nest:
    dec rcx
ob_check:
    cmp rcx, 0
    jne ob_forward
    inc rbx
    jmp exec_loop
ob_not_zero:
    inc rbx
    jmp exec_loop

close_bracket:
    mov al, byte ptr [rsi]
    test al, al
    je cb_zero
    # cell != 0, jump back
    mov rcx, 1
cb_backward:
    dec rbx
    cmp rbx, 0
    jl unbalanced_error
    mov al, byte ptr [r8 + rbx]
    cmp al, ']'
    je cb_inc_nest
    cmp al, '['
    je cb_dec_nest
    jmp cb_check
cb_inc_nest:
    inc rcx
    jmp cb_check
cb_dec_nest:
    dec rcx
cb_check:
    cmp rcx, 0
    jne cb_backward
    inc rbx
    jmp exec_loop
cb_zero:
    inc rbx
    jmp exec_loop

unbalanced_error:
    mov edi, 1
    mov eax, 60                 # sys_exit
    syscall

exec_done:
    mov edi, 0
    mov eax, 60                 # sys_exit
    syscall

.section .bss
buf:        .zero 65536
tape:       .zero 30000
