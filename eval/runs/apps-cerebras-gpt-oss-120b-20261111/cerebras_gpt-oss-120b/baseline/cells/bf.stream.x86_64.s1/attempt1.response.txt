.globl _start
.text
_start:
    # ---------- read all stdin ----------
    lea rsi, [rip + buf]          # buffer pointer
    xor rbx, rbx                  # total_len = 0
read_loop:
    mov rdx, 65536
    sub rdx, rbx                  # remaining space
    test rdx, rdx
    je read_done
    mov rdi, 0                    # fd = stdin
    xor rax, rax                  # sys_read = 0
    syscall
    test rax, rax
    je read_done
    add rbx, rax                  # total_len += bytes_read
    jmp read_loop
read_done:
    mov [rip + total_len], rbx

    # ---------- split program / input ----------
    lea rsi, [rip + buf]          # base pointer
    xor rcx, rcx                  # i = 0
    mov rdx, rbx                  # total_len
find_excl:
    cmp rcx, rdx
    jae no_excl
    mov al, byte ptr [rsi + rcx]
    cmp al, 0x21                  # '!'
    je found_excl
    inc rcx
    jmp find_excl
found_excl:
    mov [rip + prog_len], rcx
    lea rax, [rcx + 1]
    mov [rip + input_offset], rax
    mov rax, [rip + total_len]
    sub rax, rax                  # clear rax, will be overwritten
    mov rax, [rip + total_len]
    sub rax, [rip + input_offset]
    mov [rip + input_len], rax
    jmp after_split
no_excl:
    mov [rip + prog_len], rbx
    mov [rip + input_offset], rbx
    mov qword ptr [rip + input_len], 0
after_split:

    # ---------- set up pointers ----------
    lea r12, [rip + buf]          # program base
    mov r13, [rip + prog_len]     # prog_len
    mov r14, [rip + input_offset] # input_offset
    lea r15, [rip + buf]          # temporary for input base
    add r15, r14                   # r15 = input_base
    mov rdx, [rip + input_len]    # rdx = input_len
    mov qword ptr [rip + input_pos], 0

    # ---------- bracket matching ----------
    mov qword ptr [rip + stack_ptr], 0
    xor rcx, rcx                  # pc index
bracket_loop:
    cmp rcx, r13
    jae bracket_done
    mov al, byte ptr [r12 + rcx]
    cmp al, '['
    je bracket_push
    cmp al, ']'
    je bracket_pop
    inc rcx
    jmp bracket_loop
bracket_push:
    mov rax, [rip + stack_ptr]
    mov [rip + stack + rax*8], rcx
    inc rax
    mov [rip + stack
