.global _start
.text
_start:
    // x19 = input pointer
    // x20 = output buffer pointer (single byte)
    // x21 = alphabet pointer
    // w22 = line character count (0..75)
    // x23 = total bytes read

    // initialize pointers
    adrp    x19, in_buf
    add     x19, x19, :lo12:in_buf

    adrp    x20, out_buf
    add     x20, x20, :lo12:out_buf

    adrp    x21, base64_alphabet
    add     x21, x21, :lo12:base64_alphabet

    mov     w22, #0          // line counter = 0
    mov     x23, #0          // total bytes read = 0

read_loop:
    // read into buffer at in_buf + total_read
    mov     x0, #0          // fd = stdin
    mov     x8, #63         // sys_read
    add     x1, x19, x23    // buf = in_buf + total_read
    mov     x2, #49152
    sub     x2, x2, x23     // count = remaining space
    cbz     x2, read_done   // buffer full (should not happen)
    svc     #0
    cbz     x0, read_done   // EOF
    add     x23, x23, x0    // total_read += bytes_read
    b       read_loop

read_done:
    // if no input, exit silently
    cbz     x23, exit_ok

    // set up for encoding
    adrp    x19, in_buf
    add     x19, x19, :lo12:in_buf

encode_loop:
    cmp     x23, #3
    blt     encode_remainder

    // load three bytes
    ldrb    w4, [x19], #1
    ldrb    w5, [x19], #1
    ldrb    w6, [x19], #1

    // sextet0
    mov     w7, w4, lsr #2
    // sextet1
    and     w8, w4, #0x3
    lsl     w8, w8, #4
    mov     w9, w5, lsr #4
    orr     w8, w8, w9
    // sextet2
    and     w9, w5, #0xF
    lsl     w9, w9, #2
    mov     w10, w6, lsr #6
    orr     w9, w9, w10
    // sextet3
    and     w10, w6, #0x3F

    // preserve offsets in callee‑saved registers
    mov     x24, w7
    mov     x25, w8
    mov     x26, w9
    mov     x27, w10

    // output four characters
    add     x12, x21, x24, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    add     x12, x21, x25, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    add     x12, x21, x26, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    add     x12, x21, x27, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    sub     x23, x23, #3
    b       encode_loop

encode_remainder:
    cmp     x23, #0
    beq     final_newline_check

    cmp     x23, #1
    beq     encode_one

    // remainder == 2
    ldrb    w4, [x19], #1
    ldrb    w5, [x19], #1

    // sextet0
    mov     w7, w4, lsr #2
    // sextet1
    and     w8, w4, #0x3
    lsl     w8, w8, #4
    mov     w9, w5, lsr #4
    orr     w8, w8, w9
    // sextet2
    and     w9, w5, #0xF
    lsl     w9, w9, #2

    // preserve offsets
    mov     x24, w7
    mov     x25, w8
    mov     x26, w9

    // output three characters
    add     x12, x21, x24, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    add     x12, x21, x25, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    add     x12, x21, x26, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    // padding '='
    mov     w0, #61
    bl      put_char

    b       final_newline_check

encode_one:
    ldrb    w4, [x19], #1

    // sextet0
    mov     w7, w4, lsr #2
    // sextet1
    and     w8, w4, #0x3
    lsl     w8, w8, #4

    // preserve offsets
    mov     x24, w7
    mov     x25, w8

    // output two characters
    add     x12, x21, x24, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    add     x12, x21, x25, uxtw #0
    ldrb    w0, [x12]
    bl      put_char

    // two padding '='
    mov     w0, #61
    bl      put_char
    bl      put_char

    b       final_newline_check

final_newline_check:
    // if line not already terminated, output final newline
    cmp     w22, #0
    beq     exit_ok

    // write newline
    mov     w0, #10
    strb    w0, [x20]
    mov     x0, #1
    mov     x1, x20
    mov     x2, #1
    mov     x8, #64
    svc     #0
    b       exit_ok

exit_ok:
    mov     x0, #0
    mov     x8, #93
    svc     #0

// ------------------------------------------------------------
put_char:
    // w0 = ASCII character to output
    strb    w0, [x20]
    mov     x0, #1
    mov     x1, x20
    mov     x2, #1
    mov     x8, #64
    svc     #0

    // increment line counter
    add     w22, w22, #1
    cmp     w22, #76
    bne     1f

    // line full: output newline
    mov     w0, #10
    strb    w0, [x20]
    mov     x0, #1
    mov     x1, x20
    mov     x2, #1
    mov     x8, #64
    svc     #0
    mov     w22, #0
1:
    ret

.section .rodata
base64_alphabet:
    .ascii  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.bss
.align 16
in_buf:
    .space 49152
out_buf:
    .space 1
