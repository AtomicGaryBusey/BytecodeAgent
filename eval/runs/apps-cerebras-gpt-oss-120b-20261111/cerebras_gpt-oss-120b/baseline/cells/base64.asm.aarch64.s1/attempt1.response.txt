    .global _start
    .text
_start:
    // registers:
    // x19 = pointer to input buffer (in_buf)
    // x20 = pointer to output char buffer (out_char)
    // x21 = pointer to base64 alphabet
    // x22 = line counter (bytes in current line)
    // x23 = total length read
    // x24 = temporary

    // initialize pointers and counters
    adrp    x19, in_buf          // x19 = base of input buffer
    add     x19, x19, :lo12:in_buf

    adrp    x20, out_char
    add     x20, x20, :lo12:out_char

    adrp    x21, base64_alphabet
    add     x21, x21, :lo12:base64_alphabet

    mov     w22, #0              // line counter = 0
    mov     x23, #0              // total length = 0

read_loop:
    // calculate remaining space = BUF_SIZE - total_len
    mov     x0, #0              // fd = stdin
    mov     x8, #63             // sys_read
    add     x1, x19, x23        // buf = in_buf + total_len
    mov     x2, #49152
    sub     x2, x2, x23         // count = BUF_SIZE - total_len
    cbz     x2, read_done       // no space left (should not happen)

    svc     #0
    cbz     x0, read_done       // read returned 0 -> EOF

    // x0 = bytes read
    add     x23, x23, x0        // total_len += bytes_read
    b       read_loop

read_done:
    // if total_len == 0, exit silently
    cbz     x23, exit_ok

    // set src pointer = in_buf, length = total_len
    mov     x24, x23            // x24 = remaining length
    mov     x2, x19             // x2 = src pointer

encode_loop:
    cmp     x24, #3
    blt     encode_remainder

    // load three bytes
    ldrb    w4, [x2], #1
    ldrb    w5, [x2], #1
    ldrb    w6, [x2], #1

    // compute sextets
    // sextet0 = w4 >> 2
    mov     w7, w4, lsr #2

    // sextet1 = ((w4 & 0x3) << 4) | (w5 >> 4)
    and     w8, w4, #0x3
    lsl     w8, w8, #4
    mov     w9, w5, lsr #4
    orr     w8, w8, w9

    // sextet2 = ((w5 & 0xF) << 2) | (w6 >> 6)
    and     w9, w5, #0xF
    lsl     w9, w9, #2
    mov     w10, w6, lsr #6
    orr     w9, w9, w10

    // sextet3 = w6 & 0x3F
    and     w10, w6, #0x3F

    // output the four chars
    mov     w11, w7
    bl      output_char
    mov     w11, w8
    bl      output_char
    mov     w11, w9
    bl      output_char
    mov     w11, w10
    bl      output_char

    sub     x24, x24, #3
    b       encode_loop

encode_remainder:
    cmp     x24, #0
    beq     finish_output

    cmp     x24, #1
    beq     encode_one

    // case len == 2
    // load two bytes
    ldrb    w4, [x2], #1
    ldrb    w5, [x2], #1

    // sextet0 = w4 >> 2
    mov     w7, w4, lsr #2

    // sextet1 = ((w4 & 0x3) << 4) | (w5 >> 4)
    and     w8, w4, #0x3
    lsl     w8, w8, #4
    mov     w9, w5, lsr #4
    orr     w8, w8, w9

    // sextet2 = ((w5 & 0xF) << 2)
    and     w9, w5, #0xF
    lsl     w9, w9, #2

    // output three chars + padding '='
    mov     w11, w7
    bl      output_char
    mov     w11, w8
    bl      output_char
    mov     w11, w9
    bl      output_char
    // padding
    mov     w11, #61            // '=' ASCII 61
    bl      output_char

    b       finish_output

encode_one:
    // len == 1
    ldrb    w4, [x2], #1

    // sextet0 = w4 >> 2
    mov     w7, w4, lsr #2

    // sextet1 = ((w4 & 0x3) << 4)
    and     w8, w4, #0x3
    lsl     w8, w8, #4

    // output two chars + padding '=='
    mov     w11, w7
    bl      output_char
    mov     w11, w8
    bl      output_char
    // padding
    mov     w11, #61            // '='
    bl      output_char
    mov     w11, #61
    bl      output_char

    b       finish_output

// ------------------------------------------------------------
output_char:
    // w11 = sextet value (0..63) or ASCII for padding
    // if w11 <= 63 we need to look up alphabet, else it's ASCII of '=', already set
    cmp     w11, #63
    bhi     output_raw_char      // if >63 (should be only padding case)

    // lookup character from alphabet
    add     x12, x21, w11, uxtw #0
    ldrb    w13, [x12]

    // store to out_char buffer
    strb    w13, [x20]

    b       do_write

output_raw_char:
    // w11 already contains ASCII value (e.g., '=' or '\n')
    strb    w11, [x20]

do_write:
    // write single byte
    mov     x8, #64            // sys_write
    mov     x0, #1             // fd = stdout
    mov     x1, x20            // buf = out_char
    mov     x2, #1
    svc     #0
    // increment line counter
    add     w22, w22, #1
    cmp     w22, #76
    bne     ret_output_char
    // line full, output newline
    mov     w11, #10           // '\n'
    bl      output_raw_char
    mov     w22, #0
ret_output_char:
    ret

finish_output:
    // if line counter != 0, output final newline
    cmp     w22, #0
    beq     exit_ok
    // write newline
    mov     w11, #10
    bl      output_raw_char
    b       exit_ok

exit_ok:
    mov     x0, #0
    mov     x8, #93            // sys_exit
    svc     #0

    // -----------------------------------------------------------------
    .section .rodata
base64_alphabet:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

    .bss
    .align 16
in_buf:
    .space 49152
out_char:
    .space 1
