.globl _start
.text
_start:
    # read loop, discard input
    mov     rdi, 0                # fd = stdin
    lea     rsi, [rip + read_buf] # buffer
    mov     rdx, read_buf_len
read_loop:
    mov     rax, 0                # sys_read
    syscall
    test    rax, rax
    je      after_read            # EOF
    # ignore the data, continue reading
    jmp     read_loop

after_read:
    # output "ok\n"
    mov     rdi, 1                # fd = stdout
    lea     rsi, [rip + ok_msg]
    mov     rdx, ok_len
    mov     rax, 1                # sys_write
    syscall

    # exit with status 0
    xor     rdi, rdi              # status = 0
    mov     rax, 60               # sys_exit
    syscall

# ----------------------------------------------------------------
.section .rodata
ok_msg:
    .ascii  "ok\n"
ok_len = . - ok_msg

# ----------------------------------------------------------------
.section .bss
.align 16
read_buf_len = 4096
read_buf:
    .zero   read_buf_len
