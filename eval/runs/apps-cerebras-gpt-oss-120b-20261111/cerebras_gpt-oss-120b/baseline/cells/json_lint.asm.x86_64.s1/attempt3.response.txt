.globl _start
.equ READ_BUF_SIZE, 65536
.equ MAX_NUM_BUF, 20

.section .rodata
ok_msg:
    .ascii "ok\n"
ok_len = . - ok_msg
err_prefix:
    .ascii "error at "
err_prefix_len = . - err_prefix
newline:
    .ascii "\n"
newline_len = . - newline

.section .bss
.align 16
read_buf:
    .zero READ_BUF_SIZE
error_num_buf:
    .zero MAX_NUM_BUF

.text
_start:
    ; read entire stdin into read_buf
    lea rbx, [rip + read_buf]      # buffer pointer
    xor rcx, rcx                   # total length = 0
read_loop:
    mov rdi, 0                     # fd = stdin
    lea rsi, [rbx + rcx]           # destination = buffer + total_len
    mov rdx, READ_BUF_SIZE
    sub rdx, rcx                   # remaining space
    mov rax, 0                     # sys_read
    syscall
    test rax, rax
    je after_read
    add rcx, rax                   # total_len += bytes_read
    cmp rcx, READ_BUF_SIZE
    jae after_read
    jmp read_loop
after_read:
    mov rdx, rcx                   # rdx = input length
    xor rcx, rcx                   # rcx = parse position
    xor r8, r8                     # r8 = error flag = 0
    ; rbx already points to buffer start
    call parse_document
    test r8, r8
    jne error_output
    ; ok output
    mov rdi, 1
    lea rsi, [rip + ok_msg]
    mov rdx, ok_len
    mov rax, 1                     # sys_write
    syscall
    xor rdi, rdi
    mov rax, 60                    # sys_exit
    syscall

error_output:
    ; write "error at "
    mov rdi, 1
    lea rsi, [rip + err_prefix]
    mov rdx, err_prefix_len
    mov rax, 1
    syscall
    ; convert error position (r9) to decimal string
    mov rax, r9
    lea rdi, [rip + error_num_buf + MAX_NUM_BUF]   # point past buffer end
    xor rcx, rcx                                   # digit count = 0
conv_loop:
    xor rdx, rdx
    mov rbx, 10
    div rbx
    dec rdi
    add dl, '0'
    mov byte ptr [rdi], dl
    inc rcx
    test rax, rax
    jne conv_loop
    ; write digits
    mov rdi, 1
    mov rsi, rdi            # will be overwritten
    mov rsi, rdi            # placeholder to keep assembler happy

    mov rsi, rdi

    jae skip_ws_done
    mov al, byte ptr [rbx + rcx]
    cmp al, ' '
    je inc_ws
    cmp al, 9
    je inc_ws
    cmp al, 10
    je inc_ws
    cmp al, 13
    je inc_ws
    jmp skip_ws_done

    inc rcx
    cmp rcx, rdx
    jb skip_ws

    ret

    call skip_ws
    call parse_value
    test r8, r8
    jne pd_ret

    mov r9, rcx

    cmp rcx, rdx
    jae set_error_eof
    mov al, byte ptr [rbx + rcx]

    cmp al, '['
    je parse_array
    cmp al, '"'
    je parse_string
    cmp al, 't'
    je parse_true
    cmp al, 'f'
    je parse_false
    cmp al, 'n'
    je parse_null
    cmp al, '-'
    je parse_number
    cmp al, '0'
    je parse_number
    cmp al, '1'
    jb set_error_invalid
    cmp al, '9'
    jbe parse_number
    jmp set_error_invalid

    mov r9, rcx
