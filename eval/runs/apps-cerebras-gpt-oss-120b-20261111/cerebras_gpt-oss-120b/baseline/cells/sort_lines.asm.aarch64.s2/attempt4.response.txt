.globl	_start
	.text
_start:
	// buffer start address
	adr	x5, buffer               // x5 = write pointer (current end)
	adr	x6, total_bytes          // x6 = address of total_bytes
	// total = 0
	mov	x0, #0
	str	x0, [x6]

read_loop:
	// remaining = 32768 - total
	ldr	x3, [x6]                // x3 = total
	mov	x4, #32768
	sub	x2, x4, x3              // x2 = remaining
	cbnz	x2, read_continue
	b	read_done
read_continue:
	// read(fd=0, buf=x5, count=remaining)
	mov	x8, #63                 // sys_read
	mov	x0, #0                 // fd = 0
	mov	x1, x5                 // buf
	// x2 already set = remaining
	svc	#0
	// result in x0 = bytes read
	cbnz	x0, read_got
	b	read_done
read_got:
	// total += bytes_read
	ldr	x3, [x6]
	add	x3, x3, x0
	str	x3, [x6]
	// buffer pointer += bytes_read
	add	x5, x5, x0
	b	read_loop

read_done:
	// load total bytes read
	ldr	x3, [x6]                // x3 = total
	cbnz	x3, parse_input
	b	output_done               // empty input, exit

parse_input:
	adr	x7, buffer              // x7 = ptr = buffer start
	add	x10, x7, x3            // x10 = buffer_end (ptr + total)
	adr	x12, lines             // x12 = lines array base
	mov	x15, #0                // x15 = line count

parse_loop:
	cmp	x7, x10
	bge	parse_done

	// line_start = x7
	mov	x11, x7

scan_newline:
	cmp	x7, x10
	bge	scan_done
	ldrb	w14, [x7]
	cmp	w14, #10               // '\n'
	beq	scan_done
	add	x7, x7, #1
	b	scan_newline

scan_done:
	// length = x7 - line_start
	sub	x13, x7, x11           // x13 = length

	// store line (ptr, len)
	lsl	x14, x15, #4           // x14 = line_count * 16
	add	x14, x12, x14          // x14 = &lines[line_count]
	str	x11, [x14]             // ptr
	str	x13, [x14, #8]         // len

	add	x15, x15, #1           // line_count++

	// if we stopped at newline, skip it
	cmp	x7, x10
	bge	parse_loop
	ldrb	w14, [x7]
	cmp	w14, #10
	bne	parse_loop
	add	x7, x7, #1             // skip newline
	b	parse_loop

parse_done:
	// Insertion sort
	// i = 1
	mov	x9, #1
sort_outer:
	cmp	x9, x15
	bge	sort_done
	// j = i
	mov	x10, x9
sort_inner:
	cmp	x10, #0
	beq	inc_i
	// addr_jm1 = base + (j-1)*16
	sub	x11, x10, #1
	lsl	x11, x11, #4          // x11 = (j-1)*16
	add	x11, x12, x11         // x11 = &lines[j-1]
	// addr_j = base + j*16
	lsl	x14, x10, #4          // x14 = j*16
	add	x14, x12, x14         // x14 = &lines[j]

	// load lines[j-1] into x0(ptr), x1(len)
	ldr	x0, [x11]
	ldr	x1, [x11, #8]
	// load lines[j] into x2(ptr), x3(len)
	ldr	x2, [x14]
	ldr	x3, [x14, #8]

	bl	compare                // returns result in w0
	cmp	w0, #0
	ble	inc_i                  // if <=0, already in order

	// swap entries
	str	x2, [x11]
	str	x3, [x11, #8]
	str	x0, [x14]
	str	x1, [x14, #8]
	sub	x10, x10, #1
	b	sort_inner

inc_i:
	add	x9, x9, #1
	b	sort_outer

sort_done:
	// output sorted lines
	mov	x9, #0
output_loop:
	cmp	x9, x15
	bge	output_done
	lsl	x10, x9, #4
	add	x10, x12, x10
	ldr	x0, [x10]          // ptr
	ldr	x1, [x10, #8]      // len

	// write line data (may need multiple writes)
	mov	x4, x0             // buffer ptr
	mov	x5, x1             // remaining count
write_data_loop:
	cbnz	x5, write_data_continue
	b	write_newline
write_data_continue:
	mov	x0, #1            // fd = 1
	mov	x1, x4
	mov	x2, x5
	mov	x8, #64          // sys_write
	svc	#0
	// x0 = bytes_written
	cmp	x0, #0
	beq	write_newline
	add	x4, x4, x0
	sub	x5, x5, x0
	b	write_data_loop

write_newline:
	adr	x1, newline
	mov	x2, #1
	mov	x0, #1
	mov	x8, #64
	svc	#0

	add	x9, x9, #1
	b	output_loop

output_done:
	// exit status 0
	mov	x0, #0
	mov	x8, #93            // sys_exit
	svc	#0

// ------------------------------------------------------------
// compare: returns -1,0,1 in w0
// x0 = ptr_a, x1 = len_a, x2 = ptr_b, x3 = len_b
compare:
	// x4 = min(len_a,len_b)
	mov	x4, x1
	cmp	x1, x3
	csel	x4, x1, x3, lt
	mov	x5, #0            // index
cmp_loop:
	cmp	x5, x4
	bge	after_cmp_loop
	ldrb	w6, [x0, x5]
	ldrb	w7, [x2, x5]
	cmp	w6, w7
	bne	diff_found
	add	x5, x5, #1
	b	cmp_loop

diff_found:
	cmp	w6, w7
	blt	ret_minus_one
	mov	w0, #1
	ret

after_cmp_loop:
	cmp	x1, x3
	beq	equal_len
	blt	ret_minus_one
	mov	w0, #1
	ret

equal_len:
	mov	w0, #0
	ret

ret_minus_one:
	mov	w0, #-1
	ret

	.section	.rodata
newline:
	.byte	0x0A

	.bss
	.p2align 3
buffer:
	.zero	32768
total_bytes:
	.zero	8
lines:
	.zero	16000
