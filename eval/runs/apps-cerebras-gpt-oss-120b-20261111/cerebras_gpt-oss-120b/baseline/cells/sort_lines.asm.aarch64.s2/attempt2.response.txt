.globl	_start
	.text
_start:
	// buffer pointer = buffer start
	adr	x5, buffer
	// total_bytes address
	adr	x6, total_bytes
	// total = 0
	mov	x0, #0
	str	x0, [x6]

read_loop:
	// remaining = 32768 - total
	ldr	x3, [x6]                // total
	mov	x4, #32768
	sub	x2, x4, x3              // x2 = remaining
	cbnz	x2, read_continue
	b	read_done
read_continue:
	// read(fd=0, buf=x5, count=remaining)
	mov	x8, #63
	mov	x0, #0
	mov	x1, x5
	// x2 already set
	svc	#0
	// result in x0 (bytes read)
	cbnz	x0, read_got
	b	read_done
read_got:
	// total += bytes_read
	ldr	x3, [x6]
	add	x3, x3, x0
	str	x3, [x6]
	// buffer pointer += bytes_read
	add	x5, x5, x0
	b	read_loop

read_done:
	// x6 still points to total_bytes, load total
	ldr	x3, [x6]                // total bytes read -> x3
	// if total == 0, skip parsing and go straight to exit
	cbnz	x3, parse_input
	b	output_done               // no lines, go to exit

parse_input:
	adr	x7, buffer             // ptr = buffer start
	add	x10, x7, x3            // x10 = buffer_end (ptr + total)
	adr	x12, lines             // lines array base
	mov	x15, #0                // line count

parse_loop:
	cmp	x7, x10
	bge	parse_done

	// line_start = x7
	mov	x11, x7

scan_newline:
	cmp	x7, x10
	bge	scan_done
	ldrb	w14, [x7]
	cmp	w14, #10
	beq	scan_done
	add	x7, x7, #1
	b	scan_newline

scan_done:
	// length = x7 - line_start
	sub	x13, x7, x11           // length in x13

	// store line (ptr, len)
	lsl	x14, x15, #4           // offset = line_count * 16
	add	x14, x12, x14          // entry address
	str	x11, [x14]             // ptr
	str	x13, [x14, #8]         // len

	add	x15, x15, #1           // line_count++

	// if we stopped at newline, skip it
	cmp	x7, x10
	bge	parse_loop
	ldrb	w14, [x7]
	cmp	w14, #10
	bne	parse_loop
	add	x7, x7, #1             // skip newline
	b	parse_loop

parse_done:
	// sorting (insertion sort)
	// i = 1
	mov	x9, #1
sort_outer:
	cmp	x9, x15
	bge	sort_done
	// j = i
	mov	x10, x9
sort_inner:
	cmp	x10, #0
	beq	inc_i
	// addr_jm1 = base + (j-1)*16
	sub	x11, x10, #1
	lsl	x12a, x11, #4
	add	x12a, x12, x12a
	// addr_j = base + j*16
	lsl	x13a, x10, #4
	add	x13a, x12, x13a
	// load line[j-1] into x0,x1 ; line[j] into x2,x3
	ldr	x0, [x12a]
	ldr	x1, [x12a, #8]
	ldr	x2, [x13a]
	ldr	x3, [x13a, #8]
	bl	compare
	cmp	w0, #0
	bge	inc_i
	// swap entries
	str	x2, [x12a]
	str	x3, [x12a, #8]
	str	x0, [x13a]
	str	x1, [x13a, #8]
	sub	x10, x10, #1
	b	sort_inner

inc_i:
	add	x9, x9, #1
	b	sort_outer

sort_done:
	// output sorted lines
	mov	x9, #0
output_loop:
	cmp	x9, x15
	bge	output_done
	lsl	x10, x9, #4
	add	x10, x12, x10
	ldr	x0, [x10]          // ptr
	ldr	x1, [x10, #8]      // len

	// write line data
	mov	x4, x0             // buffer ptr
	mov	x5, x1             // remaining count
write_data_loop:
	cbnz	x5, write_data_continue
	b	write_newline
write_data_continue:
	mov	x0, #1            // fd
	mov	x1, x4
	mov	x2, x5
	mov	x8, #64
	svc	#0
	// x0 = bytes_written
	cmp	x0, #0
	beq	write_newline
	add	x4, x4, x0
	sub	x5, x5, x0
	b	write_data_loop

write_newline:
	adr	x1, newline
	mov	x2, #1
	mov	x0, #1
	mov	x8, #64
	svc	#0

	add	x9, x9, #1
	b	output_loop

output_done:
	// exit status 0
	mov	x0, #0
	mov	x8, #93
	svc	#0

// compare function: returns -1,0,1 in w0
// x0 = ptr_a, x1 = len_a, x2 = ptr_b, x3 = len_b
compare:
	// x4 = min(len_a,len_b)
	mov	x4, x1
	cmp	x1, x3
	csel	x4, x1, x3, lt
	mov	x5, #0            // index
cmp_loop:
	cmp	x5, x4
	bge	after_cmp_loop
	ldrb	w6, [x0, x5]
	ldrb	w7, [x2, x5]
	cmp	w6, w7
	bne	diff_found
	add	x5, x5, #1
	b	cmp_loop

diff_found:
	cmp	w6, w7
	blt	ret_minus_one
	mov	w0, #1
	ret

after_cmp_loop:
	cmp	x1, x3
	beq	equal_len
	blt	ret_minus_one
	mov	w0, #1
	ret

equal_len:
	mov	w0, #0
	ret

ret_minus_one:
	mov	w0, #-1
	ret

	.section	.rodata
newline:
	.byte	0x0A

	.bss
	.p2align 3
buffer:
	.zero	32768
total_bytes:
	.zero	8
lines:
	.zero	16000
