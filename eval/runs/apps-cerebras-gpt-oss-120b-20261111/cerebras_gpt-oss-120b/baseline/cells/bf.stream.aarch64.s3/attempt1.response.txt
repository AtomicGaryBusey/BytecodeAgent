.global _start
.equ SYSCALL_READ, 63
.equ SYSCALL_WRITE, 64
.equ SYSCALL_EXIT, 93

.section .bss
.p2align 3
prog_buf:      .zero 65536                // whole input (program + '!' + program input)
.bracket_stack: .zero 8000                // 1000 entries * 8 bytes
.jump_table:   .zero 524288               // 65536 entries * 8 bytes
.tape:         .zero 30000                // brainfuck tape

.text
_start:
    // -----------------------
    // Read entire stdin
    // x5 = total bytes read
    mov     x5, #0                     // total length = 0
read_loop:
    mov     x0, #0                    // fd = stdin
    adr     x1, prog_buf
    add     x1, x1, x5                // buf = prog_buf + total
    mov     x2, #65536
    sub     x2, x2, x5                // count = remaining space
    mov     x8, #SYSCALL_READ
    svc     #0
    cbz     x0, read_done
    add     x5, x5, x0                // total += bytes read
    b       read_loop
read_done:
    // -----------------------
    // Find first '!' separating program from input
    // x6 = program length (code_len)
    // x7 = input length
    mov     x6, #0                     // index
    mov     x7, #0                     // input_len (default)
    mov     x9, #0                     // flag: found_excl (0 = not found)
find_excl:
    cmp     x6, x5
    beq     no_excl                    // reached end, no '!'
    adr     x10, prog_buf
    add     x11, x10, x6              // address of current byte
    ldrb    w12, [x11]
    cmp     w12, #'!'
    beq     found_excl
    add     x6, x6, #1
    b       find_excl
found_excl:
    // x6 is position of '!'
    // program code length = x6
    // input start = prog_buf + x6 + 1
    // input length = total - x6 - 1
    mov     x9, #1                     // flag: found
    add     x13, x10, x6              // prog_buf + x6
    add     x13, x13, #1              // input_ptr = prog_buf + x6 + 1
    sub     x7, x5, x6
    sub     x7, x7, #1                // input_len = total - x6 - 1
    b       split_done
no_excl:
    // whole input is program, no input
    // program length = total, input length = 0
    mov     x6, x5
    mov     x13, x5                   // input_ptr same as end (empty)
    mov     x7, #0
split_done:
    // -----------------------
    // Validate brackets and build jump table
    // x10 = prog_buf (base)
    // x11 = program end address = prog_buf + program_len
    // x12 = loop index (position)
    // x14 = bracket stack base
    // x15 = stack depth
    adr     x10, prog_buf
    add     x11, x10, x6              // program end
    mov     x12, #0                   // position i
    adr     x14, .bracket_stack
    mov     x15, #0                   // stack depth = 0
bracket_loop:
    cmp     x12, x6
    beq     brackets_done
    add     x16, x10, x12            // addr of current command
    ldrb    w17, [x16]
    cmp     w17, #'['
    beq     bracket_open
    cmp     w17, #']'
    beq     bracket_close
    // other: ignore
    add     x12, x12, #1
    b       bracket_loop
bracket_open:
    // push address of '[' onto stack
    mul     x18, x15, #8
    add
