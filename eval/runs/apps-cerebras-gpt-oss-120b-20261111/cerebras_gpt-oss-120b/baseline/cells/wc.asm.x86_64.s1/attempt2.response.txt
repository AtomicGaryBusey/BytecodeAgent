.globl _start

.section .bss
    .align 8
buf:        .zero 65536
lines:      .zero 8
words:      .zero 8
bytes:      .zero 8
outbuf:     .zero 64

.section .text

_start:
    xor r8, r8                     # in_word = 0

read_loop:
    mov rax, 0                     # sys_read
    xor rdi, rdi                   # fd = 0 (stdin)
    lea rsi, [rip + buf]           # buffer
    mov rdx, 65536                 # count
    syscall
    test rax, rax
    je after_read

    mov rcx, rax                   # bytes remaining in buffer
    lea rbx, [rip + buf]           # pointer to current byte

proc_loop:
    mov al, byte ptr [rbx]
    inc rbx

    inc qword ptr [rip + bytes]

    cmp al, 0x0A
    jne not_newline
    inc qword ptr [rip + lines]
not_newline:

    cmp al, 0x20
    je whitespace
    cmp al, 0x09
    je whitespace
    cmp al, 0x0A
    je whitespace
    cmp al, 0x0B
    je whitespace
    cmp al, 0x0C
    je whitespace
    cmp al, 0x0D
    je whitespace
    # non‑whitespace
    test r8, r8
    jne proc_continue
    inc qword ptr [rip + words]
    mov r8, 1
    jmp proc_continue

whitespace:
    xor r8, r8

proc_continue:
    dec rcx
    jne proc_loop
    jmp read_loop

after_read:
    lea rdi, [rip + outbuf]        # output pointer

    mov rax, qword ptr [rip + lines]
    call print_number
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, qword ptr [rip + words]
    call print_number
    mov byte ptr [rdi], ' '
    inc rdi

    mov rax, qword ptr [rip + bytes]
    call print_number
    mov byte ptr [rdi], 0x0A
    inc rdi

    lea rsi, [rip + outbuf]
    mov rdx, rdi
    sub rdx, rsi                  # length

    mov rax, 1                    # sys_write
    mov rdi, 1                    # fd = stdout
    syscall

    mov rax, 60                   # sys_exit
    xor rdi, rdi
    syscall

# ---------------------------------------------------------------
# print_number: rax = number, rdi = output pointer
# returns with rdi advanced past the printed characters
print_number:
    mov rcx, 10
    test rax, rax
    jne .num_loop
    mov byte ptr [rdi], '0'
    inc rdi
    ret
.num_loop:
    xor r9, r9                     # digit count
.num_div:
    xor rdx, rdx
    div rcx                        # rax = rax/10, rdx = remainder
    add dl, '0'
    push rdx
    inc r9
    test rax, rax
    jne .num_div
.pop_loop:
    pop rdx
    mov byte ptr [rdi], dl
    inc rdi
    dec r9
    jne .pop_loop
    ret
