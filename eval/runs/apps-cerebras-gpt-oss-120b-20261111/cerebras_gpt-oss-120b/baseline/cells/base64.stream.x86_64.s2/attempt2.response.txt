.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
input_buf:
    .zero 49152
out_buf:
    .zero 70000

.section .text
_start:
    # -------------------------------------------------
    # Read full input into input_buf
    # -------------------------------------------------
    lea rsi, [rip + input_buf]          # buffer pointer
    xor rcx, rcx                        # total length = 0
read_loop:
    mov rdx, 49152
    sub rdx, rcx                        # space left
    test rdx, rdx
    jz read_done                        # should not happen
    mov rdi, 0                          # fd = stdin
    xor rax, rax                        # sys_read = 0
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                        # total += bytes read
    jmp read_loop
read_done:
    mov r12, rcx                        # r12 = total_len
    test r12, r12
    je exit_program                     # empty input: exit without output

    # -------------------------------------------------
    # Initialize pointers and counters
    # -------------------------------------------------
    lea rsi, [rip + input_buf]          # src pointer
    lea r15, [rip + out_buf]            # dst pointer
    lea rbx, [rip + b64_table]          # base64 table pointer
    xor rcx, rcx                        # line_len = 0
    xor r13, r13                        # out_len = 0

    mov r14, r12                        # r14 = remaining bytes

process_main:
    cmp r14, 3
    jb process_tail

    # load three bytes
    movzx eax, byte ptr [rsi]           # b0
    movzx edx, byte ptr [rsi+1]         # b1
    movzx ecx, byte ptr [rsi+2]         # b2
    shl eax, 16
    shl edx, 8
    or  eax, edx
    or  eax, ecx                         # eax = 24-bit value

    # index 0
    mov ebp, eax
    shr ebp, 18
    and ebp, 0x3F
    movzx ebp, byte ptr [rbx + rbp]     # ebp = char
    mov [r15], ebp
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx1
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx1:
    # index 1
    mov ebp, eax
    shr ebp, 12
    and ebp, 0x3F
    movzx ebp, byte ptr [rbx + rbp]
    mov [r15], ebp
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx2:
    # index 2
    mov ebp, eax
    shr ebp, 6
    and ebp, 0x3F
    movzx ebp, byte ptr [rbx + ebp]
    mov [r15], ebp
    inc r15
