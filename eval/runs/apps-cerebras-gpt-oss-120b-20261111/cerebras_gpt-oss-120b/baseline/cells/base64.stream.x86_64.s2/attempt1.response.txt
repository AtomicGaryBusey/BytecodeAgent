.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
input_buf:
    .zero 49152
out_buf:
    .zero 70000

.section .text
_start:
    # read all input into input_buf
    lea rsi, [rip + input_buf]          # buffer pointer
    xor rcx, rcx                        # total length = 0
read_loop:
    mov rdx, 49152
    sub rdx, rcx                        # space left
    test rdx, rdx
    jz done_read                        # should not happen for given limits
    mov rdi, 0                          # fd = stdin
    xor rax, rax                        # sys_read
    syscall
    test rax, rax
    jz done_read
    add rcx, rax                        # total += bytes read
    jmp read_loop
done_read:
    mov r12, rcx                        # r12 = total_len
    test r12, r12
    je exit_program                     # empty input: exit without output

    # pointers
    lea rsi, [rip + input_buf]          # src pointer
    lea rdi, [rip + out_buf]            # dst pointer
    xor r8d, r8d                        # line count (chars on current line)

    mov rbx, r12                        # remaining bytes counter

process_main:
    cmp rbx, 3
    jb process_tail

    # load three bytes
    movzx eax, byte ptr [rsi]           # byte0
    movzx edx, byte ptr [rsi+1]         # byte1
    movzx ecx, byte ptr [rsi+2]         # byte2
    shl eax, 16
    shl edx, 8
    or  eax, edx
    or  eax, ecx                        # eax = 24-bit value

    # index 0
    mov ebp, eax
    shr ebp, 18
    and ebp, 0x3F
    movzx r9d, byte ptr [rel b64_table + rbp]
    mov
