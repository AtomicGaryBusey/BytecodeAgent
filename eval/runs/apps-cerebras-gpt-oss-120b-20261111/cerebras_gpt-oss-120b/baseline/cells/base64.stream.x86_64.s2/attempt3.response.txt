.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
input_buf:
    .zero 49152
out_buf:
    .zero 70000

.text
_start:
    # -------------------------------------------------
    # Read full input into input_buf
    # -------------------------------------------------
    lea rsi, [rip + input_buf]          # buffer pointer
    xor rcx, rcx                        # total length = 0
read_loop:
    mov rdx, 49152
    sub rdx, rcx                        # space left
    test rdx, rdx
    jz read_done                        # should not happen
    mov rdi, 0                          # fd = stdin
    xor rax, rax                        # sys_read = 0
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                        # total += bytes read
    jmp read_loop
read_done:
    mov r12, rcx                        # r12 = total_len
    test r12, r12
    je exit_program                     # empty input: exit without output

    # -------------------------------------------------
    # Initialize pointers and counters
    # -------------------------------------------------
    lea rsi, [rip + input_buf]          # src pointer
    lea r15, [rip + out_buf]            # dst pointer
    lea rbx, [rip + b64_table]          # base64 table pointer
    xor rcx, rcx                        # line_len = 0
    xor r13, r13                        # out_len = 0

    mov r14, r12                        # r14 = remaining bytes

process_main:
    cmp r14, 3
    jae process_full
    jmp process_tail

process_full:
    # load three bytes
    movzx eax, byte ptr [rsi]           # b0
    shl eax, 16
    movzx edx, byte ptr [rsi+1]         # b1
    shl edx, 8
    or   eax, edx
    movzx edx, byte ptr [rsi+2]         # b2
    or   eax, edx                        # eax = 24-bit value

    # index 0
    mov r8d, eax
    shr r8d, 18
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx1_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx1_done:
    # index 1
    mov r8d, eax
    shr r8d, 12
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx2_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx2_done:
    # index 2
    mov r8d, eax
    shr r8d, 6
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx3_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx3_done:
    # index 3
    mov r8d, eax
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne advance_src
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
advance_src:
    add rsi, 3
    sub r14, 3
    jmp process_main

process_tail:
    cmp r14, 2
    ja  process_tail_one   # r14 == 3 handled earlier, so now r14 could be 2 or 1 or 0
    je  process_tail_two
    cmp r14, 1
    je  process_tail_one
    jmp finish_output        # r14 == 0

process_tail_one:
    # one remaining byte
    movzx eax, byte ptr [rsi]          # b0
    # index 0
    mov r8d, eax
    shr r8d, 2
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    # index 1
    mov r8d, eax
    shl r8d, 4
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    # padding '==' 
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc rcx
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc rcx
    jmp check_line_end

process_tail_two:
    # two remaining bytes
    movzx eax, byte ptr [rsi]          # b0
    shl eax, 8
    movzx edx, byte ptr [rsi+1]        # b1
    or   eax, edx                       # eax = 16-bit value
    # index 0
    mov r8d, eax
    shr r8d, 10
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    # index 1
    mov r8d, eax
    shr r8d, 4
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    # index 2
    mov r8d, eax
    shl r8d, 2
    and r8d, 0x3F
    movzx edx, byte ptr [rbx + r8*1]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    # padding '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc rcx
    jmp check_line_end

check_line_end:
    cmp rcx, 76
    jne finish_output
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
    jmp finish_output

finish_output:
    # ensure final newline if needed (but not if line_len ==0)
    cmp rcx, 0
    je write_output
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx

write_output:
    # write out_buf of length r13 to stdout
    mov rax, 1          # sys_write
    mov rdi, 1          # fd = stdout
    lea rsi, [rip + out_buf]
    mov rdx, r13
    syscall

exit_program:
    mov rax, 60         # sys_exit
    xor rdi, rdi
    syscall
