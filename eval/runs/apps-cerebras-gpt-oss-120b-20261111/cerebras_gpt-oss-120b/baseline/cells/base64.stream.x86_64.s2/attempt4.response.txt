.globl _start

.section .rodata
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.section .bss
    .align 16
input_buf:
    .zero 49152

.text
_start:
    # allocate output buffer on stack (70,000 bytes)
    sub rsp, 70000
    mov rbx, rsp            # rbx = out_base
    mov r15, rbx            # r15 = out_ptr
    xor rcx, rcx            # line_len = 0
    xor r13, r13            # out_len = 0

    # read input
    xor r10, r10            # total_len = 0
read_loop:
    lea rsi, [rip + input_buf]
    add rsi, r10            # buf + total_len
    mov rdx, 49152
    sub rdx, r10            # space left
    test rdx, rdx
    jz read_done
    mov rax, 0              # sys_read
    mov rdi, 0              # fd = stdin
    syscall
    test rax, rax
    jz read_done
    add r10, rax            # total_len += bytes read
    jmp read_loop

read_done:
    test r10, r10
    je exit_prog            # empty input: no output

    lea rsi, [rip + input_buf]   # src pointer
    mov r14, r10                  # remaining bytes
    lea r9, [rip + b64_table]     # table pointer

process_loop:
    cmp r14, 3
    jae full_group

    cmp r14, 2
    je tail_two
    cmp r14, 1
    je tail_one
    jmp finish_output

full_group:
    movzx eax, byte ptr [rsi]       # b0
    movzx edx, byte ptr [rsi+1]     # b1
    movzx ebp, byte ptr [rsi+2]     # b2
    shl eax, 16
    shl edx, 8
    or   eax, edx
    or   eax, ebp                    # eax = 24-bit value

    # index 0
    mov r8d, eax
    shr r8d, 18
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx1_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx1_done:
    # index 1
    mov r8d, eax
    shr r8d, 12
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx2_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx2_done:
    # index 2
    mov r8d, eax
    shr r8d, 6
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne idx3_done
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
idx3_done:
    # index 3
    mov r8d, eax
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne advance_src
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
advance_src:
    add rsi, 3
    sub r14, 3
    jmp process_loop

tail_one:
    movzx eax, byte ptr [rsi]       # b0
    # index 0
    mov r8d, eax
    shr r8d, 2
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne tail1_mid
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
tail1_mid:
    # index 1
    mov r8d, eax
    shl r8d, 4
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne tail1_pad1
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
tail1_pad1:
    # padding '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne tail1_pad2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
tail1_pad2:
    # padding '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne finish_output
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
    jmp finish_output

tail_two:
    movzx eax, byte ptr [rsi]       # b0
    movzx edx, byte ptr [rsi+1]     # b1
    shl eax, 8
    or eax, edx                     # 16-bit value in eax

    # index 0
    mov r8d, eax
    shr r8d, 10
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne tail2_mid
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
tail2_mid:
    # index 1
    mov r8d, eax
    shr r8d, 4
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne tail2_mid2
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
tail2_mid2:
    # index 2
    mov r8d, eax
    shl r8d, 2
    and r8d, 0x3F
    movzx edx, byte ptr [r9 + r8]
    mov [r15], dl
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne tail2_pad
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
tail2_pad:
    # padding '='
    mov byte ptr [r15], '='
    inc r15
    inc r13
    inc rcx
    cmp rcx, 76
    jne finish_output
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx
    jmp finish_output

finish_output:
    cmp rcx, 0
    je write_output
    mov byte ptr [r15], 0x0A
    inc r15
    inc r13
    xor rcx, rcx

write_output:
    test r13, r13
    je exit_prog
    mov rax, 1          # sys_write
    mov rdi, 1          # fd = stdout
    mov rsi, rbx        # buffer start
    mov rdx, r13
    syscall

exit_prog:
    mov rax, 60         # sys_exit
    xor rdi, rdi
    syscall
