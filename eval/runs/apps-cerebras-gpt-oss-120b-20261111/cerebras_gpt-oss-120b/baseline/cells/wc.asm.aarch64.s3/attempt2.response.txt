.global _start
.text
_start:
    // counters: x9=lines, x10=words, x11=bytes, x12=in_word
    mov     x9, #0
    mov     x10, #0
    mov     x11, #0
    mov     x12, #0

read_loop:
    adrp    x7, buf
    add     x7, x7, :lo12:buf

    // read(fd=0, buf, 4096)
    mov     x8, #63
    mov     x0, #0
    mov     x1, x7
    mov     x2, #4096
    svc     #0
    cbz     x0, output_counts          // EOF

    mov     x4, x0                     // bytes remaining in this chunk

process_chunk:
    cbz     x4, read_loop

    ldrb    w13, [x7], #1
    subs    x4, x4, #1

    // total bytes
    add     x11, x11, #1

    // newline?
    cmp     w13, #10
    bne     not_newline
    add     x9, x9, #1
not_newline:

    // whitespace test
    cmp     w13, #0x20
    beq     ws
    cmp     w13, #0x09
    beq     ws
    cmp     w13, #0x0A
    beq     ws
    cmp     w13, #0x0B
    beq     ws
    cmp     w13, #0x0C
    beq     ws
    cmp     w13, #0x0D
    beq     ws
    b       non_ws
ws:
    cmp     x12, #0
    beq     continue_loop
    mov     x12, #0
    b       continue_loop
non_ws:
    cmp     x12, #0
    bne     continue_loop
    add     x10, x10, #1
    mov     x12, #1
continue_loop:
    b       process_chunk

output_counts:
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    mov     x0, x1                     // dest pointer

    // lines
    mov     x2, x9
    bl      itoa

    // space
    mov     w3, #32
    strb    w3, [x0], #1

    // words
    mov     x2, x10
    bl      itoa

    // space
    mov     w3, #32
    strb    w3, [x0], #1

    // bytes
    mov     x2, x11
    bl      itoa

    // newline
    mov     w3, #10
    strb    w3, [x0], #1

    // length = x0 - x1
    sub     x2, x0, x1

    // write(fd=1, buf=x1, count=x2)
    mov     x8, #64
    mov     x0, #1
    mov     x1, x1
    svc     #0

    // exit(0)
    mov     x8, #93
    mov     x0, #0
    svc     #0

// ------------------------------------------------------------
// itoa: convert unsigned integer in x2 to decimal string.
// x0 = destination pointer (updated on return)
// x2 = number to convert
itoa:
    cmp     x2, #0
    bne     itoa_nonzero
    mov     w3, #48                     // '0'
    strb    w3, [x0], #1
    ret

itoa_nonzero:
    sub     sp, sp, #64                // temporary buffer
    mov     x9, #10                    // divisor
    mov     x4, #0                     // digit count

itoa_loop:
    udiv    x5, x2, x9                // quotient
    mul     x6, x5, x9               // quotient * 10
    sub     x6, x2, x6               // remainder
    add     x6, x6, #48              // ascii digit

    add     x7, sp, x4
    strb    w6, [x7]

    add     x4, x4, #1
    mov     x2, x5
    cbz     x2, itoa_copy_digits
    b       itoa_loop

itoa_copy_digits:
    subs    x4, x4, #1                // last index
itoa_copy_loop:
    add     x7, sp, x4
    ldrb    w6, [x7]
    strb    w6, [x0], #1
    subs    x4, x4, #1
    bge     itoa_copy_loop

    add     sp, sp, #64
    ret

    .bss
    .align 8
buf:        .zero   65536
outbuf:     .zero   64
