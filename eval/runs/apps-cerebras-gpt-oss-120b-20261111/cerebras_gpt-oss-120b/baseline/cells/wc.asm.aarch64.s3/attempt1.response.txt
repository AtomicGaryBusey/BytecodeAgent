    .global _start
    .text
_start:
    // zero counters: x9=lines, x10=words, x11=bytes, x12=in_word flag
    mov     x9, #0
    mov     x10, #0
    mov     x11, #0
    mov     x12, #0

read_loop:
    // set buffer pointer in x7
    adrp    x7, buf
    add     x7, x7, :lo12:buf

    // read(fd=0, buf, count=4096)
    mov     x8, #63          // syscall number
    mov     x0, #0           // fd stdin
    mov     x1, x7           // buffer address
    mov     x2, #4096        // max bytes per read
    svc     #0
    // result in x0 (bytes read)
    cbz     x0, output_counts    // if 0, end of input

    // x4 = remaining bytes in this chunk
    mov     x4, x0

process_chunk:
    cbz     x4, read_loop         // if no more bytes, read next chunk

    // load next byte and advance pointer
    ldrb    w13, [x7], #1
    subs    x4, x4, #1

    // count total bytes
    add     x11, x11, #1

    // check for newline (0x0A)
    cmp     w13, #10
    bne     not_newline
    add     x9, x9, #1            // increment lines
not_newline:
    // whitespace test: space 0x20, tab 0x09, newline 0x0A, VT 0x0B, FF 0x0C, CR 0x0D
    cmp     w13, #0x20
    beq     is_whitespace
    cmp     w13, #0x09
    beq     is_whitespace
    cmp     w13, #0x0A
    beq     is_whitespace
    cmp     w13, #0x0B
    beq     is_whitespace
    cmp     w13, #0x0C
    beq     is_whitespace
    cmp     w13, #0x0D
    beq     is_whitespace
    b       non_whitespace

is_whitespace:
    // if we were inside a word, leave the word
    cmp     x12, #0
    beq     continue_loop
    mov     x12, #0
    b       continue_loop

non_whitespace:
    // non-whitespace character
    cmp     x12, #0
    bne     continue_loop
    // start of a new word
    add     x10, x10, #1
    mov     x12, #1

continue_loop:
    b       process_chunk

output_counts:
    // prepare output buffer pointer in x1
    adrp    x1, outbuf
    add     x1, x1, :lo12:outbuf
    mov     x0, x1               // x0 = dest pointer for itoa

    // convert lines (x9)
    mov     x2, x9
    bl      itoa

    // store space
    mov     w3, #' '
    strb    w3, [x0], #1

    // convert words (x10)
    mov     x2, x10
    bl      itoa

    // store space
    mov     w3, #' '
    strb    w3, [x0], #1

    // convert bytes (x11)
    mov     x2, x11
    bl      itoa

    // store newline
    mov     w3, #10
    strb    w3, [x0], #1

    // compute length = end_ptr - start_ptr
    sub     x2, x0, x1          // x2 = length

    // write(fd=1, buf=outbuf, count=length)
    mov     x8, #64
    mov     x0, #1
    mov     x1, x1
    svc     #0

    // exit(0)
    mov     x8, #93
    mov     x0, #0
    svc     #0

// ------------------------------------------------------------
// itoa: convert unsigned integer in x2 to decimal string.
// x0 = destination pointer (updated on return)
// x2 = number to convert
// Returns with x0 pointing after the written string.
itoa:
    // if number is zero, output "0"
    cmp     x2, #0
    bne     itoa_nonzero
    mov     w3, #'0'
    strb    w3, [x0], #1
    ret

itoa_nonzero:
    // reserve temporary space on stack (32 bytes)
    sub     sp, sp, #32
    mov     x4, #0              // digit count

itoa_loop:
    // quotient = x2 / 10
    udiv    x5, x2, #10
    // remainder = x2 - quotient*10
    mul     x6, x5, #10
    sub     x6, x2, x6
    // convert digit to ASCII
    add     x6, x6, #'0'
    // store digit at temporary buffer (sp + x4)
    add     x7, sp, x4
    strb    w6, [x7]
    // increment digit count
    add     x4, x4, #1
    // prepare next iteration
    mov     x2, x5
    cbz     x2, itoa_copy_digits
    b       itoa_loop

itoa_copy_digits:
    // x4 = number of digits
    // copy digits in reverse order to destination
    subs    x4, x4, #1          // x4 = last index
itoa_copy_loop:
    add     x7, sp, x4
    ldrb    w6, [x7]
    strb    w6, [x0], #1
    subs    x4, x4, #1
    bge     itoa_copy_loop

    // restore stack pointer
    add     sp, sp, #32
    ret

    .bss
    .align 8
buf:    .zero   4096
outbuf: .zero   64
