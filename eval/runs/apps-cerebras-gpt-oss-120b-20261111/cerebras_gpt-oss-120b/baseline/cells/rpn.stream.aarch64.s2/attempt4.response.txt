.text
.global _start
_start:
    // x19 = input buffer base
    adr     x19, input_buf

    // x20 = total bytes read = 0
    mov     x20, #0

read_loop:
    // remaining = BUF_SIZE - total
    mov     x1, #65536
    sub     x1, x1, x20
    cbz     x1, read_done

    mov     x0, #0                     // fd = stdin
    add     x2, x19, x20               // buf = input_buf + total
    mov     x8, #63                    // sys_read
    svc     #0
    cmp     x0, #0
    beq     read_done
    blt     exit_error
    add     x20, x20, x0               // total += bytes read
    b       read_loop

read_done:
    // x22 = stack base
    adr     x22, stack
    // x23 = stack size = 0
    mov     x23, #0
    // x24 = error flag = 0
    mov     x24, #0

    // cur pointer = input_buf
    mov     x25, x19
    // end pointer = input_buf + total
    add     x26, x19, x20

line_loop:
    cmp     x25, x26
    beq     exit_prog                  // no more data

    // reset per-line state
    mov     x23, #0
    mov     x24, #0

token_skip_separators:
    cmp     x25, x26
    beq     line_end
    ldrb    w0, [x25]
    cmp     w0, #' '
    beq     inc_cur_skip
    cmp     w0, #9                    // tab
    beq     inc_cur_skip
    cmp     w0, #10                   // newline
    beq     line_end
    b       token_start

inc_cur_skip:
    add     x25, x25, #1
    b       token_skip_separators

token_start:
    // x27 = token start pointer
    mov     x27, x25

    // find token end
    mov     x28, x25                  // x28 = ptr
token_find_end:
    cmp     x28, x26
    beq     token_end_found
    ldrb    w0, [x28]
    cmp     w0, #' '
    beq     token_end_found
    cmp     w0, #9
    beq     token_end_found
    cmp     w0, #10
    beq     token_end_found
    add     x28, x28, #1
    b       token_find_end
token_end_found:
    // token_len = x28 - x27
    sub     x9, x28, x27

    // load first byte
    ldrb    w0, [x27]

    // check if token is a single-character operator
    cmp     x9, #1
    bne     token_maybe_number

    // single char; possible operator
    cmp     w0, #'+'
    beq     do_op_plus
    cmp     w0, #'-'
    beq     do_op_minus
    cmp     w0, #'*'
    beq     do_op_mul
    cmp     w0, #'/'
    beq     do_op_div
    cmp     w0, #'%'
    beq     do_op_mod

    // not operator, must be a single digit number
    cmp     w0, #'0'
    blt     token_invalid
    cmp     w0, #'9'
    bgt     token_invalid
    // convert digit
    sub     w0, w0, #'0'
    uxtw    x10, w0
    b       push_number

token_maybe_number:
    // token length >1 ; could be negative number or positive number
    // check optional leading '-'
    cmp     w0, #'-'
    bne     token_positive_number

    // leading '-', ensure at least 2 chars
    cmp     x9, #2
    blt     token_invalid
    // check second char is digit
    ldrb    w1, [x27, #1]
    cmp     w1, #'0'
    blt     token_invalid
    cmp     w1, #'9'
    bgt     token_invalid
    // parse unsigned magnitude
    mov     x10, #0
    mov     x11, #0                     // digit count
    add     x12, x27, #1                // ptr to first digit
parse_neg_digits:
    cmp     x12, x28
    beq     end_parse_neg
    ldrb    w0, [x12]
    sub     w0, w0, #'0'
    uxtw    x0, w0
    mov     x13, #10
    mul     x10, x10, x13
    add     x10, x10, x0
    add     x11, x11, #1
    cmp     x11, #18
    bgt     token_invalid
    add     x12, x12, #1
    b       parse_neg_digits
end_parse_neg:
    neg     x10, x10
    b       push_number

token_positive_number:
    // first char must be digit
    cmp     w0, #'0'
    blt     token_invalid
    cmp     w0, #'9'
    bgt     token_invalid
    // parse digits
    mov     x10, #0
    mov     x11, #0
    mov     x12, x27                    // ptr
parse_pos_digits:
    cmp     x12, x28
    beq     end_parse_pos
    ldrb    w0, [x12]
    sub     w0, w0, #'0'
    uxtw    x0, w0
    mov     x13, #10
    mul     x10, x10, x13
    add     x10, x10, x0
    add     x11, x11, #1
    cmp     x11, #18
    bgt     token_invalid
    add     x12, x12, #1
    b       parse_pos_digits
end_parse_pos:
    // number parsed in x10
    b       push_number

token_invalid:
    mov     x24, #1                     // set error flag
    b       skip_to_line_end

push_number:
    // push x10 onto stack
    lsl     x13, x23, #3                // offset = index*8
    str     x10, [x22, x13]
    add     x23, x23, #1
    b       after_token

do_op_plus:
    // need at least two values
    cmp     x23, #2
    blt     op_error
    // b = stack[--size]
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x11, [x22, x13]
    // a = stack[--size]
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x12, [x22, x13]
    add     x10, x12, x11               // result
    // push result
    lsl     x13, x23, #3
    str     x10, [x22, x13]
    add     x23, x23, #1
    b       after_token

do_op_minus:
    cmp     x23, #2
    blt     op_error
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x11, [x22, x13]             // b
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x12, [x22, x13]             // a
    sub     x10, x12, x11
    lsl     x13, x23, #3
    str     x10, [x22, x13]
    add     x23, x23, #1
    b       after_token

do_op_mul:
    cmp     x23, #2
    blt     op_error
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x11, [x22, x13]
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x12, [x22, x13]
    mul     x10, x12, x11
    lsl     x13, x23, #3
    str     x10, [x22, x13]
    add     x23, x23, #1
    b       after_token

do_op_div:
    cmp     x23, #2
    blt     op_error
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x11, [x22, x13]             // b
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x12, [x22, x13]             // a
    // division checks
    cmp     x11, #0
    beq     op_error
    // INT64_MIN = 0x8000000000000000
    mov     x15, #0x8000
    movk    x15, #0x0000, lsl #16
    movk    x15, #0x0000, lsl #32
    movk    x15, #0x0000, lsl #48
    // now x15 = 0x8000000000000000
    cmp     x12, x15
    bne     do_div_compute
    // a == INT64_MIN, check b == -1
    cmp     x11, #-1
    beq     op_error
do_div_compute:
    sdiv    x10, x12, x11
    lsl     x13, x23, #3
    str     x10, [x22, x13]
    add     x23, x23, #1
    b       after_token

do_op_mod:
    cmp     x23, #2
    blt     op_error
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x11, [x22, x13]            // b
    sub     x23, x23, #1
    lsl     x13, x23, #3
    ldr     x12, [x22, x13]            // a
    cmp     x11, #0
    beq     op_error
    // INT64_MIN check same as div
    mov     x15, #0x8000
    movk    x15, #0x0000, lsl #16
    movk    x15, #0x0000, lsl #32
    movk    x15, #0x0000, lsl #48
    cmp     x12, x15
    bne     do_mod_compute
    cmp     x11, #-1
    beq     op_error
do_mod_compute:
    sdiv    x9, x12, x11               // q = a / b
    mul     x10, x9, x11               // q*b
    sub     x10, x12, x10              // a - q*b
    lsl     x13, x23, #3
    str     x10, [x22, x13]
    add     x23, x23, #1
    b       after_token

op_error:
    mov     x24, #1
    b       skip_to_line_end

after_token:
    // advance cur to token end
    mov     x25, x28
    b       token_skip_separators

skip_to_line_end:
    // skip until newline or end
skip_line:
    cmp     x25, x26
    beq     line_end
    ldrb    w0, [x25]
    cmp     w0, #10
    beq     line_end
    add     x25, x25, #1
    b       skip_line

line_end:
    // at newline or end, possibly consume newline
    cmp     x25, x26
    beq     after_line_cleanup
    ldrb    w0, [x25]
    cmp     w0, #10
    bne     after_line_cleanup
    add     x25, x25, #1

after_line_cleanup:
    // decide output
    cmp     x24, #0
    bne     output_error
    cmp     x23, #1
    bne     output_error
    // get result
    ldr     x0, [x22]                  // result in x0
    bl      print_number
    b       line_loop

output_error:
    // write "error\n"
    mov     x0, #1                     // fd stdout
    adr     x1, error_msg
    mov     x2, #6
    mov     x8, #64                    // sys_write
    svc     #0
    b       line_loop

print_number:                           // x0 = signed 64-bit value
    // buffer at out_buf (32 bytes)
    adr     x1, out_buf
    add     x1, x1, #32                // ptr to end
    mov     x5, x1                     // end pointer
    mov     x6, #0                     // sign flag

    cmp     x0, #0
    bge     pn_positive
    mov     x6, #1
    neg     x0, x0                     // two's complement works for INT64_MIN

pn_positive:
    mov     x7, #0                     // digit count

pn_loop:
    cmp     x0, #0
    beq     pn_no_digits
    mov     x8, #10
    udiv    x9, x0, x8                // quotient
    mul     x10, x9, x8
    sub     x11, x0, x10              // remainder
    add     x11, x11, #'0'
    sub     x5, x5, #1
    strb    w11, [x5]
    add     x7, x7, #1
    mov     x0, x9
    b       pn_loop

pn_no_digits:
    cmp     x7, #0
    bne     pn_sign
    // number is zero
    sub     x5, x5, #1
    mov     w11, #'0'
    strb    w11, [x5]
    add     x7, x7, #1

pn_sign:
    cmp     x6, #0
    beq     pn_prepare_write
    sub     x5, x5, #1
    mov     w11, #'-'
    strb    w11, [x5]
    add     x7, x7, #1

pn_prepare_write:
    // newline
    sub     x5, x5, #1
    mov     w11, #10
    strb    w11, [x5]
    add     x7, x7, #1

    // write loop (fd=1, buf=x5, len=x7)
    mov     x0, #1
    mov     x1, x5
    mov     x2, x7
    mov     x8, #64

pn_write_loop:
    svc     #0
    cmp     x0, x2
    beq     pn_done
    sub     x2, x2, x0
    add     x1, x1, x0
    b       pn_write_loop

pn_done:
    ret

exit_prog:
    mov     x0, #0
    mov     x8, #93
    svc     #0

exit_error:
    mov     x0, #1
    mov     x8, #93
    svc     #0

.section .rodata
error_msg:
    .ascii  "error\n"

.bss
    .align 8
input_buf:
    .space 65536
stack:
    .space 8192
out_buf:
    .space 32
