.text
.global _start
_start:
    // Read entire input (max 65536 bytes)
    mov     x9, #0                     // total bytes read
read_loop:
    mov     x0, #0                     // fd = stdin
    adr     x1, input_buf
    add     x1, x1, x9                 // buffer offset
    mov     x2, #65536
    sub     x2, x2, x9                 // remaining space
    cmp     x2, #0
    ble     read_done
    mov     x8, #63                    // sys_read
    svc     #0
    cmp     x0, #0
    beq     read_done
    blt     exit_error                 // error
    add     x9, x9, x0
    b       read_loop
read_done:
    adr     x0, input_len
    str     x9, [x0]

    // Set pointers for processing
    adr     x2, input_buf              // cur ptr
    adr     x1, input_len
    ldr     x3, [x1]                   // total_len
    add     x4, x2, x3                 // end ptr (x4)

line_loop:
    cmp     x2, x4
    beq     exit_prog                  // no more data

    // start new line
    adr     x0, sptr
    mov     x1, #0
    str     x1, [x0]                   // stack pointer = 0
    mov     x27, #0                    // error flag = 0

token_loop:
    cmp     x2, x4
    beq     line_finish
    ldrb    w0, [x2]
    cmp     w0, #'\n'
    beq     line_finish
    cmp     w0, #' '
    beq     skip_sep
    cmp     w0, #9
    beq     skip_sep
    // token start
    b       token_start

skip_sep:
    add     x2, x2, #1
    b       token_loop

token_start:
    // w0 = first char of token
    cmp     w0, #'0'
    blt     not_digit
    cmp     w0, #'9'
    bgt     not_digit
    // positive integer literal
    mov     x10, #0            // value
    mov     x11, #0            // digit count
pos_digit_loop:
    sub     w0, w0, #'0'
    uxtw    x0, w0
    mul     x10, x10, #10
