.text
.global _start
_start:
    // ----- read whole input into input_buf (max 65536 bytes) -----
    adr     x20, input_buf          // base of input buffer
    mov     x9, #0                  // total bytes read = 0

read_loop:
    // remaining space = BUF_SIZE - x9
    adr     x2, buf_size
    ldr     x2, [x2]                // x2 = 65536
    sub     x2, x2, x9
    cbz     x2, read_done

    mov     x0, #0                  // fd = stdin
    add     x1, x20, x9             // buf = input_buf + x9
    mov     x8, #63                 // sys_read
    svc     #0
    cmp     x0, #0
    beq     read_done
    blt     exit_error
    add     x9, x9, x0              // total += bytes read
    b       read_loop

read_done:
    // x9 = total length of input
    // pointers for processing
    adr     x2, input_buf           // cur pointer
    add     x4, x2, x9              // end pointer = cur + len

line_loop:
    cmp     x2, x4
    beq     exit_prog               // no more data

    // reset evaluation stack and error flag for this line
    mov     x13, #0                 // stack index (size)
    mov     x27, #0                 // error flag = 0

token_scan:
    // skip separators (space or tab)
    cmp     x2, x4
    beq     line_finish
    ldrb    w0, [x2]
    cmp     w0, #0x20               // space
    beq     skip_sep
    cmp     w0, #9                  // tab
    beq     skip_sep
    cmp     w0, #10                 // newline
    beq     line_finish
    // token start at x2, find token end
    mov     x6, x2                  // token_end pointer

find_end:
    cmp     x6, x4
    beq     token_end_found
    ldrb    w0, [x6]
    cmp     w0, #0x20
    beq     token_end_found
    cmp     w0, #9
    beq     token_end_found
    cmp     w0, #10
    beq     token_end_found
    add     x6, x6, #1
    b       find_end
token_end_found:
    sub     x8, x6, x2              // token_len in x8

    // classify token
    ldrb    w0, [x2]                // first char
    cmp     x8, #1
    beq     maybe_operator

    // token_len > 1, could be number (optional leading '-')
    cmp     w0, #'-'
    beq     token_negative_number
    // must start with digit
    cmp     w0, #'0'
    blt     token_invalid
    cmp     w0, #'9'
    bgt     token_invalid
    // positive number
    mov     x10, #0                 // accumulator
    mov     x11, #0                 // digit count
    mov     x9, x2                  // pointer p

parse_pos_digits:
    cmp     x9, x6
    beq     parse_done
    ldrb    w0, [x9]
    sub     w0, w0, #'0'
    uxtw    x0, w0
    mov     x12, #10
    mul     x10, x10, x12
    add     x10, x10, x0
    add     x11, x11, #1
    cmp     x11, #18
    bgt     token_invalid
    add     x9, x9, #1
    b       parse_pos_digits
parse_done:
    b       push_number

token_negative_number:
    // ensure second char is digit
    cmp     x8, #2
    blt     token_invalid
    ldrb    w1, [x2, #1]
    cmp     w1, #'0'
    blt     token_invalid
    cmp     w1, #'9'
    bgt     token_invalid
    // parse number after '-'
    mov     x10, #0
    mov     x11, #0
    add     x9, x2, #1            // start after '-'

parse_neg_digits:
    cmp     x9, x6
    beq     parse_neg_done
    ldrb    w0, [x9]
    sub     w0, w0, #'0'
    uxtw    x0, w0
    mov     x12, #10
    mul     x10, x10, x12
    add     x10, x10, x0
    add     x11, x11, #1
    cmp     x11, #18
    bgt     token_invalid
    add     x9, x9, #1
    b       parse_neg_digits
parse_neg_done:
    neg     x10, x10
    b       push_number

maybe_operator:
    // token_len == 1, could be operator or single digit number
    cmp     w0, #'+'
    beq     op_plus
    cmp     w0, #'-'
    beq     op_minus
    cmp     w0, #'*'
    beq     op_mul
    cmp     w0, #'/'
    beq     op_div
    cmp     w0, #'%'
    beq     op_mod
    // not an operator, must be a digit (single digit number)
    cmp     w0, #'0'
    blt     token_invalid
    cmp     w0, #'9'
    bgt     token_invalid
    sub     w0, w0, #'0'
    uxtw   x10, w0
    b       push_number

op_plus:
    // need at least two values on stack
    cmp     x13, #2
    blt     op_error
    // pop b
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x11, [x20, x14]
    // pop a
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x12, [x20, x14]
    // compute a+b
    add     x10, x12, x11
    // push result
    lsl     x14, x13, #3
    str     x10, [x20, x14]
    add     x13, x13, #1
    b       after_token

op_minus:
    cmp     x13, #2
    blt     op_error
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x11, [x20, x14]       // b
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x12, [x20, x14]       // a
    sub     x10, x12, x11
    lsl     x14, x13, #3
    str     x10, [x20, x14]
    add     x13, x13, #1
    b       after_token

op_mul:
    cmp     x13, #2
    blt     op_error
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x11, [x20, x14]
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x12, [x20, x14]
    mul     x10, x12, x11
    lsl     x14, x13, #3
    str     x10, [x20, x14]
    add     x13, x13, #1
    b       after_token

op_div:
    cmp     x13, #2
    blt     op_error
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x11, [x20, x14]       // b
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x12, [x20, x14]       // a
    // check divide by zero
    cmp     x11, #0
    beq     op_error
    // check overflow a == INT64_MIN && b == -1
    movz    x15, #0x8000, lsl #48 // x15 = INT64_MIN
    mov     x16, #-1               // x16 = -1
    cmp     x12, x15
    bne     do_div
    cmp     x11, x16
    beq     op_error
do_div:
    sdiv    x10, x12, x11
    lsl     x14, x13, #3
    str     x10, [x20, x14]
    add     x13, x13, #1
    b       after_token

op_mod:
    cmp     x13, #2
    blt     op_error
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x11, [x20, x14]       // b
    sub     x13, x13, #1
    lsl     x14, x13, #3
    ldr     x12, [x20, x14]       // a
    // check divide by zero
    cmp     x11, #0
    beq     op_error
    // check overflow a == INT64_MIN && b == -1
    movz    x15, #0x8000, lsl #48 // x15 = INT64_MIN
    mov     x16, #-1               // x16 = -1
    cmp     x12, x15
    bne     do_mod
    cmp     x11, x16
    beq     op_error
do_mod:
    sdiv    x9, x12, x11          // q = a / b
    mul     x10, x9, x11         // q*b
    sub     x10, x12, x10        // a - q*b
    lsl     x14, x13, #3
    str     x10, [x20, x14]
    add     x13, x13, #1
    b       after_token

op_error:
    mov     x27, #1
    b       skip_to_line_end

token_invalid:
    mov     x27, #1                 // set error flag
    b       skip_to_line_end

push_number:
    // push x10 onto the stack
    adr     x20, stack
    lsl     x14, x13, #3
    str     x10, [x20, x14]
    add     x13, x13, #1
    b       after_token

skip_sep:
    add     x2, x2, #1
    b       token_scan

skip_to_line_end:
    // skip rest of this line
skip_line_loop:
    cmp     x2, x4
    beq     line_finish
    ldrb    w0, [x2]
    cmp     w0, #10
    beq     line_finish
    add     x2, x2, #1
    b       skip_line_loop

after_token:
    // advance cur to token end and continue scanning
    mov     x2, x6
    b       token_scan

line_finish:
    // at newline or end, consume newline if present
    cmp     x2, x4
    beq     after_line_done
    ldrb    w0, [x2]
    cmp     w0, #10
    bne     after_line_done
    add     x2, x2, #1

after_line_done:
    // decide output
    cmp     x27, #0
    bne     output_error
    cmp     x13, #1
    bne     output_error
    // get the single result
    adr     x20, stack
    ldr     x0, [x20]               // result in x0
    bl      print_number
    b       line_loop

output_error:
    // write "error\n"
    mov     x0, #1                  // fd = stdout
    adr     x1, error_msg
    mov     x2, #6                  // length
    mov     x8, #64                 // sys_write
    svc     #0
    b       line_loop

// -------------------------------------------------------------
print_number:                         // x0 = signed value to print
    // buffer at out_buf (32 bytes)
    adr     x1, out_buf
    add     x1, x1, #32
    mov     x5, x1                  // end pointer
    mov     x6, #0                  // sign flag =0

    cmp     x0, #0
    bge     num_positive
    mov     x6, #1                  // negative
    neg     x0, x0                  // two's complement works for INT_MIN
num_positive:
    mov     x7, #0                  // digit count

num_loop:
    cmp     x0, #0
    beq     num_done
    mov     x8, #10
    udiv    x9, x0, x8              // quotient
    mul     x10, x9, x8
    sub     x11, x0, x10            // remainder
    add     x11, x11, #'0'
    sub     x5, x5, #1
    strb    w11, [x5]
    add     x7, x7, #1
    mov     x0, x9
    b       num_loop
num_done:
    // handle zero specially (no digits emitted)
    cmp     x7, #0
    bne     num_sign
    sub     x5, x5, #1
    mov     w11, #'0'
    strb    w11, [x5]
    add     x7, x7, #1
num_sign:
    cmp     x6, #0
    beq     num_prepare_write
    sub     x5, x5, #1
    mov     w11, #'-'
    strb    w11, [x5]
    add     x7, x7, #1
num_prepare_write:
    // append newline
    sub     x5, x5, #1
    mov     w11, #10
    strb    w11, [x5]
    add     x7, x7, #1

    // write the string (fd=1)
    mov     x0, #1
    mov     x1, x5
    mov     x2, x7
    mov     x8, #64
write_loop:
    svc     #0
    cmp     x0, x2
    beq     write_done
    sub     x2, x2, x0
    add     x1, x1, x0
    b       write_loop
write_done:
    ret

exit_prog:
    mov     x0, #0
    mov     x8, #93                  // exit
    svc     #0

exit_error:
    mov     x0, #1
    mov     x8, #93
    svc     #0

.section .rodata
error_msg:
    .ascii  "error\n"

.data
buf_size:
    .quad   65536

.bss
    .align 8
input_buf:
    .space 65536
stack:
    .space 8192        // 1024 * 8 bytes
out_buf:
    .space 32
