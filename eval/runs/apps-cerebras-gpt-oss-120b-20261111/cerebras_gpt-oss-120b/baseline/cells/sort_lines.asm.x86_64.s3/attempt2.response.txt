.globl _start
.text
_start:
    xor rax, rax                # sys_read = 0
    mov rdi, 0                  # fd = stdin
    lea rsi, [rip + buffer]
    mov rdx, 32768
read_loop:
    mov rax, 0
    syscall
    test rax, rax
    je parse_input
    mov rbx, rax                # bytes read
    mov rcx, [rip + total_len]
    add rcx, rbx
    mov [rip + total_len], rcx
    jmp read_loop

parse_input:
    mov rcx, [rip + total_len]   # total_len
    lea rsi, [rip + buffer]      # base pointer
    xor rdx, rdx                 # i = 0
    xor r8d, r8d                 # start = 0
parse_loop:
    cmp rdx, rcx
    je parse_finish_no_newline
    mov al, [rsi + rdx]
    cmp al, 10                   # newline?
    jne not_newline
    # newline at i
    mov rax, rdx
    sub rax, r8                  # len = i - start
    mov rbx, [rip + line_cnt]
    lea rdi, [rip + lines]
    mov r9, rbx
    shl r9, 3
    add rdi, r9
    mov dword ptr [rdi], r8d
    mov dword ptr [rdi + 4], eax
    inc qword ptr [rip + line_cnt]
    lea r8, [rdx + 1]            # start = i+1
not_newline:
    inc rdx
    jmp parse_loop

parse_finish_no_newline:
    cmp r8, rcx
    jae parse_done
    mov rax, rcx
    sub rax, r8
    mov rbx, [rip + line_cnt]
    lea rdi, [rip + lines]
    mov r9, rbx
    shl r9, 3
    add rdi, r9
    mov dword ptr [rdi], r8d
    mov dword ptr [rdi + 4], eax
    inc qword ptr [rip + line_cnt]
parse_done:
    # bubble sort
    xor rbx, rbx                # pass = 0
outer_pass:
    mov rcx, [rip + line_cnt]
    sub rcx, rbx
    dec rcx                     # rcx = line_cnt - pass - 1
    cmp rcx, 0
    jle sort_done
    xor rdx, rdx                # j = 0
inner_pass:
    cmp rdx, rcx
    jge inner_done
    # addresses of lines[j] and lines[j+1]
    mov r8, rdx
    shl r8, 3
    lea r9, [rip + lines]
    add r9, r8                  # r9 = &lines[j]
    mov r10, rdx
    inc r10
    shl r10, 3
    lea r11, [rip + lines]
    add r11, r10                # r11 = &lines[j+1]

    push rdx                    # save j
    # load start/len for line j
    mov eax, dword ptr [r9]          # a_start
    mov ebx, dword ptr [r9 + 4]      # a_len
    # load start/len for line j+1
    mov ecx, dword ptr [r11]         # b_start
    mov edx, dword ptr [r11 + 4]     # b_len
    mov r8, rdx                      # b_len -> r8
    lea rdi, [rip + buffer]
    add rdi, rax                     # a_ptr
    lea rdx, [rip + buffer]
    add rdx, rcx                     # b_ptr
    mov rsi, rbx                     # a_len
    call line_cmp
    pop rdx                         # restore j
    cmp eax, 0
    jle no_swap
    # swap entries
    mov rax, qword ptr [r9]
    mov qword ptr [r9], qword ptr [r11]
    mov qword ptr [r11], rax
no_swap:
    inc rdx
    jmp inner_pass
inner_done:
    inc rbx
    jmp outer_pass

sort_done:
    xor rdx, rdx                # index = 0
    mov rcx, [rip + line_cnt]
output_loop:
    cmp rdx, rcx
    jge output_done
    mov r8, rdx
    shl r8, 3
    lea r9, [rip + lines]
    add r9, r8
    mov esi, dword ptr [r9]        # start offset
    mov ebp, dword ptr [r9 + 4]    # length

    # write line bytes
    mov rax, 1
    mov rdi, 1
    lea rax, [rip + buffer]
    add rax, rsi
    mov rsi, rax
    mov rdx, rbp
    syscall

    # write newline
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + nl]
    mov rdx, 1
    syscall

    inc rdx
    jmp output_loop
output_done:
    mov rax, 60
    xor rdi, rdi
    syscall

# line_cmp: compare two lines
# rdi = a_ptr, rsi = a_len, rdx = b_ptr, r8 = b_len
# returns eax = 1 if a > b, 0 if equal, -1 if a < b
line_cmp:
    xor rax, rax
cmp_loop:
    test rsi, rsi
    jz len_check_a
    test r8, r8
    jz len_check_b
    mov al, byte ptr [rdi]
    mov bl, byte ptr [rdx]
    cmp al, bl
    jne diff
    inc rdi
    inc rdx
    dec rsi
    dec r8
    jmp cmp_loop
diff:
    movzx eax, al
    movzx ebx, bl
    cmp eax, ebx
    ja gt
    mov eax, -1
    ret
gt:
    mov eax, 1
    ret
len_check_a:
    test r8, r8
    jz equal
    mov eax, -1
    ret
len_check_b:
    mov eax, 1
    ret
equal:
    xor eax, eax
    ret

.section .rodata
nl: .byte 0x0A

.bss
.align 8
buffer: .zero 32768
lines: .zero 8000        # 1000 * 8 bytes
total_len: .zero 8
line_cnt: .zero 8
