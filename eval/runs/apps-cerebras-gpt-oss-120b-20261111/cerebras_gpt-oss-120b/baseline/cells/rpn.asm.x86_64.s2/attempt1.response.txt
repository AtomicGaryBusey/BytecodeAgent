.globl _start
_start:
    # r8 = total input length (starts 0)
    xor     r8, r8
read_loop:
    mov     rax, 0                # sys_read
    mov     rdi, 0                # fd stdin
    lea     rsi, [rip + input_buf]
    mov     rdx, INPUT_BUF_SIZE
    syscall
    test    rax, rax
    je      after_read
    add     r8, rax
    jmp     read_loop
after_read:
    # rsi = current pointer, rcx = remaining bytes
    lea     rsi, [rip + input_buf]
    mov     rcx, r8

line_loop:
    cmp     rcx, 0
    je      exit_prog

    # Find end of line (newline or end of input)
    mov     rbx, rsi          # rbx = line start
    mov     rdx, rcx          # rdx = remaining
    mov     rdi, rsi          # rdi = scan pointer
scan_loop:
    test    rdx, rdx
    je      no_newline
    cmp     byte ptr [rdi], 0x0A
    je      have_newline
    inc     rdi
    dec     rdx
    jmp     scan_loop
have_newline:
    # rdi points at newline, rdx = bytes after newline
    mov     r9, rdi          # r9 = line_end (exclude newline)
    sub     rdx, 1           # one byte for newline
    jmp     line_ready
no_newline:
    # No newline, line ends at input end
    mov     r9, rdi          # r9 = line_end (input end)
    mov     rdx, 0
line_ready:
    # rbx = line_start, r9 = line_end (exclusive), rcx = remaining total data
    # rdx = bytes after line (including newline if present)
    # Process this line
    xor     r12, r12         # stack size = 0
    mov     r13, 0           # error flag = 0

    mov     r14, rbx         # token pointer = line_start
token_loop:
    # skip separators (space=0x20, tab=0x09)
skip_sep:
    cmp     r14, r9
    je      after_tokens
    mov     al, byte ptr [r14]
    cmp     al, 0x20
    je      inc_sep
    cmp     al, 0x09
    je      inc_sep
    jmp     token_start
inc_sep:
    inc     r14
    jmp     skip_sep

token_start:
    mov     r15, r14         # r15 = token start
    # find token end
find_end:
    cmp     r14, r9
    je      token_found
    mov     al, byte ptr [r14]
    cmp     al, 0x20
    je      token_found
    cmp     al, 0x09
    je      token_found
    inc     r14
    jmp     find_end
token_found:
    mov     rax, r14
    sub     rax, r15         # rax = token_len
    mov     rdx, rax         # token_len in rdx
    test    rdx, rdx
    je      token_loop       # empty token (should not happen)

    cmp     r13, 0
    jne     token_loop       # if error already set, ignore token

    # check if operator (single char in + - * / %)
    mov     rcx, rdx
    cmp     rcx, 1
    jne     parse_number
    mov     al, byte ptr [r15]
    cmp     al, '+'
    je      op_add
    cmp     al, '-'
    je      op_sub
    cmp     al, '*'
    je      op_mul
    cmp     al, '/'
    je      op_div
    cmp     al, '%'
    je      op_mod
    # not an operator, fall through to number parsing
    jmp     parse_number

# ---- Operators ----
op_add:
    cmp     r12, 2
    jl      set_error
    # pop b
    dec     r12
    mov     rcx, [rip + stack_base + r12*8]
    # pop a
    dec     r12
    mov     rdx, [rip + stack_base + r12*8]
    add     rdx, rcx
    # push result
    mov     [rip + stack_base + r12*8], rdx
    inc     r12
    jmp     token_loop

op_sub:
    cmp     r12, 2
    jl      set_error
    dec     r12
    mov     rcx, [rip + stack_base + r12*8]   # b
    dec     r12
    mov     rdx, [rip + stack_base + r12*8]   # a
    sub     rdx, rcx
    mov     [rip + stack_base + r12*8], rdx
    inc     r12
    jmp     token_loop

op_mul:
    cmp     r12, 2
    jl      set_error
    dec     r12
    mov     rcx, [rip + stack_base + r12*8]   # b
    dec     r12
    mov     rdx, [rip + stack_base + r12*8]   # a
    mov     rax, rdx
    imul    rax, rcx
    mov     [rip + stack_base + r12*8], rax
    inc     r12
    jmp     token_loop

op_div:
    cmp     r12, 2
    jl      set_error
    dec     r12
    mov     rcx, [rip + stack_base + r12*8]   # b
    dec     r12
    mov     rdx, [rip + stack_base + r12*8]   # a
    test    rcx, rcx
    je      set_error
    mov     r8, rdx
    mov     r9, rcx
    mov     rax, rdx
    mov     rdx, rdx
    cqo
    mov     rbx, rcx
    mov     rsi, rdx        # save sign-extended high for overflow check
    mov     rdi, rdx        # not used
    # check overflow a==INT64_MIN && b==-1
    cmp     rax, 0x8000000000000000
    jne     do_div
    cmp     rcx, -1
    jne     do_div
    jmp     set_error
do_div:
    idiv    rcx
    mov     [rip + stack_base + r12*8], rax
    inc     r12
    jmp     token_loop

op_mod:
    cmp     r12, 2
    jl      set_error
    dec     r12
    mov     rcx, [rip + stack_base + r12*8]   # b
    dec     r12
    mov     rdx, [rip + stack_base + r12*8]   # a
    test    rcx, rcx
    je      set_error
    mov     rax, rdx
    cqo
    cmp     rax, 0x8000000000000000
    jne     do_mod
    cmp     rcx, -1
    jne     do_mod
    jmp     set_error
do_mod:
    idiv    rcx
    mov     [rip + stack_base + r12*8], rdx   # remainder
    inc     r12
    jmp     token_loop

set_error:
    mov     r13, 1
    jmp     token_loop

# ---- Number parsing ----
parse_number:
    # r15 = token start, rdx = token_len
    mov     rax, rdx          # token_len
    mov     rbx, r15          # token ptr
    mov     rsi, 0            # sign flag 0=positive,1=negative
    # check optional leading '-'
    mov     al, byte ptr [rbx]
    cmp     al, '-'
    jne     num_no_sign
    mov     rsi, 1
    inc     rbx
    dec     rax                # decrement length
num_no_sign:
    # now rbx points at first digit, rax = digit_len
    cmp     rax, 1
    jl      set_error
    cmp     rax, 18
    jg      set_error
    xor     rdx, rdx          # value accumulator
num_digit_loop:
    cmp     rax, 0
    je      num_digits_done
    mov     al, byte ptr [rbx]
    sub     al, '0'
    cmp     al, 9
    ja      set_error
    mov     rcx, rdx
    imul    rcx, 10
    add     rcx, rax
    mov     rdx, rcx
    inc     rbx
    dec     rax
    jmp     num_digit_loop
num_digits_done:
    test    rsi, rsi
    je      push_number
    neg     rdx
push_number:
    # push rdx onto stack
    mov     [rip + stack_base + r12*8], rdx
    inc     r12
    jmp     token_loop

after_tokens:
    # line processing finished
    cmp     r13, 1
    je      output_error
    cmp     r12, 1
    jne     output_error

    # correct result in stack[0]
    mov     rax, [rip + stack_base]
    jmp     output_number
output_error:
    # write "error\n"
    mov     rax, 1                # sys_write
    mov     rdi, 1                # fd stdout
    lea     rsi, [rip + err_msg]
    mov     rdx, err_len
    syscall
    jmp     advance_line

output_number:
    # rax = signed value to print
    # Determine sign
    mov     rcx, rax
    sar     rcx, 63               # rcx = sign mask (0 or -1)
    mov     rdx, rcx
    xor     rax, rcx
    sub     rax, rcx              # rax = absolute unsigned value
    # rdx = sign mask
    lea     rdi, [rip + out_buf + OUT_BUF_SIZE]  # point to end
    mov     rbx, rdi
    mov     r9, rax
    cmp     r9, 0
    jne     num_convert_loop
    dec     rdi
    mov     byte ptr [rdi], '0'
    jmp     num_convert_done
num_convert_loop:
    xor     rdx, rdx
    mov     rax, r9
    mov     rcx, 10
    div     rcx
    add     dl, '0'
    dec     rdi
    mov     byte ptr [rdi], dl
    mov     r9, rax
    test    rax, rax
    jne     num_convert_loop
num_convert_done:
    cmp     rcx, 0
    je      no_sign
    # add '-'
    dec     rdi
    mov     byte ptr [rdi], '-'
no_sign:
    # compute length
    lea     rsi, [rip + out_buf]
    mov     rdx, rbx
    sub     rdx, rdi          # rdx = number length
    # copy number to out_buf start
    mov     r8, rdx
    mov     rcx, r8
    rep movsb               # copy from rdi to rsi
    # append newline
    mov     byte ptr [rsi + rcx], 0x0A
    inc     rcx                # length+1
    # write result
    mov     rax, 1
    mov     rdi, 1
    lea     rsi, [rip + out_buf]
    mov     rdx, rcx
    syscall
    jmp     advance_line

advance_line:
    # advance to next line
    mov     rsi, r9            # r9 was line_end (newline or end)
    cmp     byte ptr [rsi], 0x0A
    jne     set_remaining
    inc     rsi
set_remaining:
    # rcx = remaining bytes after this line
    mov     rdx, rcx
    sub     rdx, rsi
    mov     rcx, rdx
    jmp     line_loop

exit_prog:
    mov     rax, 60
    xor     rdi, rdi
    syscall

# ----------------- Data -----------------
.section .rodata
err_msg:
    .ascii  "error\n"
err_len = . - err_msg

# ----------------- BSS -----------------
.section .bss
.align 16
input_buf:
    .zero INPUT_BUF_SIZE
INPUT_BUF_SIZE = 65536

.align 16
stack_base:
    .zero STACK_SIZE
STACK_SIZE = 8192        # 1024 entries * 8 bytes

.align 16
out_buf:
    .zero OUT_BUF_SIZE
OUT_BUF_SIZE = 32
