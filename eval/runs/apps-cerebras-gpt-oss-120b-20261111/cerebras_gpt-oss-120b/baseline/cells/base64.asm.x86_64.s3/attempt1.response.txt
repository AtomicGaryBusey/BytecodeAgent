.global _start
.bss
    .align 16
input_buf:
    .zero 49152
line_buf:
    .zero 77

.rodata
    .align 16
b64_table:
    .ascii "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

.text
_start:
    # read all input
    lea rbx, [rip + input_buf]          # rbx = write pointer
read_loop:
    mov rdi, 0                          # fd = stdin
    mov rsi, rbx                        # buffer
    mov rdx, 4096                       # max read size
    mov rax, 0                          # sys_read
    syscall
    test rax, rax
    je read_done
    add rbx, rax                        # advance write pointer
    jmp read_loop
read_done:
    # set up pointers for encoding
    lea rsi, [rip + input_buf]          # rsi = read pointer (current)
    mov rdx, rbx                        # rdx = end pointer (input_buf + len)
    lea r9, [rip + line_buf]            # r9 = line buffer address
    xor r8d, r8d                        # line_pos = 0
    lea r10, [rip + b64_table]          # r10 = base64 table

encode_loop:
    cmp rsi, rdx
    jge encode_done

    mov rax, rdx
    sub rax, rsi                        # remaining bytes
    cmp rax, 3
    jl partial_case

    # full 3-byte group
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rsi+1]
    mov cl, byte ptr [rsi+2]

    movzx eax, al
    shl eax, 16
    movzx ebx, bl
    shl ebx, 8
    or eax, ebx
    movzx ebx, cl
    or eax, ebx                         # eax = 24-bit value

    # first output char
    mov edx, eax
    shr edx, 18
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # second
    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # third
    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # fourth
    mov edx, eax
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char

    add rsi, 3
    jmp encode_loop

partial_case:
    cmp rax, 2
    je two_bytes
    # one byte remaining
    mov al, byte ptr [rsi]
    movzx eax, al
    shl eax, 16

    # first char
    mov edx, eax
    shr edx, 18
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # second char
    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # padding
    mov dl, '='
    call put_char
    call put_char
    jmp encode_done

two_bytes:
    # load two bytes
    mov al, byte ptr [rsi]
    mov bl, byte ptr [rsi+1]
    movzx eax, al
    shl eax, 16
    movzx ebx, bl
    shl ebx, 8
    or eax, ebx                         # eax = 16+8 bits

    # first char
    mov edx, eax
    shr edx, 18
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # second char
    mov edx, eax
    shr edx, 12
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # third char
    mov edx, eax
    shr edx, 6
    and edx, 0x3F
    mov dl, byte ptr [r10 + rdx]
    call put_char
    # padding
    mov dl, '='
    call put_char
    jmp encode_done

encode_done:
    cmp r8d, 0
    je exit_program
    mov byte ptr [r9 + r8], 10          # newline
    inc r8
    mov rdi, 1
    mov rsi, r9
    mov rdx, r8
    mov rax, 1                          # sys_write
    syscall
    jmp exit_program

# ------------------------------------------------------------
# put_char: output character stored in dl to line buffer,
# handling line length of 76 characters.
put_char:
    mov byte ptr [r9 + r8], dl
    inc r8
    cmp r8, 76
    jne .pc_ret
    # line full: add newline and write
    mov byte ptr [r9 + 76], 10
    mov rdi, 1
    mov rsi, r9
    mov rdx, 77
    mov rax, 1                          # sys_write
    syscall
    xor r8d, r8d
.pc_ret:
    ret

exit_program:
    mov rdi, 0
    mov rax, 60                         # sys_exit
    syscall
