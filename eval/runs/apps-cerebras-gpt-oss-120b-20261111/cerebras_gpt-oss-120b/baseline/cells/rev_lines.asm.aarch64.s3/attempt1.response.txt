	.section .bss
input_buf:	.space 65536
output_buf:	.space 65537

	.text
	.globl _start
_start:
	// ---- read all input -------------------------------------------------
	// x9 = base of input buffer
	adr	x9, input_buf
	// x10 = total length read so far
	mov	x10, #0

read_loop:
	// x1 = buffer address = base + total_len
	add	x1, x9, x10
	// x2 = remaining space = 65536 - total_len
	mov	x3, #65536
	sub	x2, x3, x10
	// syscall parameters
	mov	x0, #0          // fd = stdin
	mov	x8, #63         // sys_read
	svc	#0
	// x0 = bytes read
	cbz	x0, read_done
	add	x10, x10, x0
	b	read_loop

read_done:
	// x9 = input base, x10 = total_len
	// x11 = end pointer (input base + total_len)
	add	x11, x9, x10

	// ---- process lines --------------------------------------------------
	// x12 = output pointer (output base)
	adr	x12, output_buf
	// x13 = current source pointer
	mov	x13, x9

line_loop:
	cmp	x13, x11
	bge	output_finish          // src_ptr >= end -> done

	// x14 = scan pointer, start at current src
	mov	x14, x13

scan_loop:
	cmp	x14, x11
	bge	no_newline              // reached end
	ldrb	w0, [x14]
	cmp	w0, #10                 // newline?
	beq	found_newline
	add	x14, x14, #1
	b	scan_loop

found_newline:
	// x14 points to newline byte
	b	process_line

no_newline:
	// No newline found, line ends at end of input
	mov	x14, x11

process_line:
	// length = line_end - src_ptr
	sub	x15, x14, x13          // x15 = len
	cbz	x15, rev_skip_copy    // if len == 0 skip copy

rev_copy_loop:
	// addr = src_ptr + len - 1
	add	x16, x13, x15
	sub	x16, x16, #1
	ldrb	w0, [x16]
	strb	w0, [x12], #1
	sub	x15, x15, #1
	cbnz	x15, rev_copy_loop

rev_skip_copy:
	// write newline to output
	mov	w0, #10
	strb	w0, [x12], #1

	// advance src pointer past newline if present
	cmp	x14, x11
	beq	src_at_end
	// newline present
	add	x13, x14, #1
	b	line_loop

src_at_end:
	// reached end of input
	mov	x13, x14
	b	line_loop

output_finish:
	// x9 = output base again
	adr	x9, output_buf
	// x0 = output length = dst_ptr - output_base
	sub	x0, x12, x9

	// ---- write output ---------------------------------------------------
	// x1 = current write pointer
	mov	x1, x9
	// x2 = remaining length
	mov	x2, x0

write_loop:
	cbz	x2, exit_program
	mov	x0, #1          // fd = stdout
	mov	x8, #64         // sys_write
	svc	#0
	// x0 = bytes written
	cbz	x0, exit_program
	add	x1, x1, x0
	sub	x2, x2, x0
	b	write_loop

exit_program:
	mov	x8, #93         // sys_exit
	mov	x0, #0
	svc	#0
