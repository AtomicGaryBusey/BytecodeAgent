.globl _start
.text
_start:
    # ---------- read whole stdin into input_buf ----------
    xor rcx, rcx                     # rcx = total bytes read
read_loop:
    mov rax, 0                       # sys_read
    mov rdi, 0                       # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, rcx                     # buffer pointer = input_buf + rcx
    mov rdx, 65536
    sub rdx, rcx                     # remaining space
    test rdx, rdx
    je read_done
    syscall
    test rax, rax
    je read_done
    add rcx, rax                     # rcx += bytes read
    cmp rcx, 65536
    jb read_loop
read_done:
    mov rdx, 0                       # rdx = cursor offset

line_loop:
    cmp rdx, rcx
    jge exit_program                  # no more input

    xor rbx, rbx                      # stack_len = 0
    xor r8, r8                        # error_flag = 0

token_loop:
    # skip separators; detect line end
    cmp rdx, rcx
    jge line_end
    mov al, byte ptr [rip + input_buf + rdx]
    cmp al, 0x0A                      # newline ?
    je line_end
    cmp al, 0x20                      # space
    je inc_cursor
    cmp al, 0x09                      # tab
    je inc_cursor
