.globl _start
.text
_start:
    # read entire stdin into input_buf
    lea rbx, [rip + input_buf]        # rbx = input buffer base
    xor rcx, rcx                      # rcx = total bytes read
read_loop:
    mov rax, 0                        # sys_read
    mov rdi, 0                        # fd = stdin
    lea rsi, [rbx + rcx]              # buffer pointer
    mov rdx, 65536
    sub rdx, rcx                      # remaining space
    test rdx, rdx
    jz read_done
    syscall
    test rax, rax
    jz read_done
    add rcx, rax                      # increase total length
    cmp rcx, 65536
    jb read_loop
read_done:
    # rcx now holds input_len
    xor rdx, rdx                      # rdx = current offset (cursor)

line_start:
    cmp rdx, rcx
    jae exit_program                  # done when cursor >= input_len

    # find end of line (newline or end of input)
    mov r8, rdx                       # r8 = scan pointer
find_nl:
    cmp r8, rcx
    jae no_nl
    mov al, byte ptr [rbx + r8]
    cmp al, 0x0a
    je found_nl
    inc r8
    jmp find_nl
found_nl:
    mov r9, r8                        # r9 = line_end (points at newline)
    jmp process_line
no_nl:
    mov r9, rcx                       # line_end = end of input

process_line:
    # rdx = line_start, r9 = line_end (exclusive)
    xor r10, r10                      # r10 = stack count
    xor r11, r11                      # r11 = error flag (0 = no error)

    mov r12, rdx                      # r12 = token pointer

token_loop:
    cmp r12, r9
    jae token_done

    # skip separators (space 0x20, tab 0x09)
    mov al, byte ptr [rbx + r12]
    cmp al, 0x20
    je inc_token
    cmp al, 0x09
    je inc_token
    # token starts at r12
    mov r13, r12                      # r13 = token_start
    mov r14, r12                      # r14 = scan for token end
tok_end:
    cmp r14, r9
    jae tok_end_done
    mov al, byte ptr [rbx + r14]
    cmp al, 0x20
    je tok_end_done
    cmp al, 0x09
    je tok_end_done
    inc r14
    jmp tok_end
tok_end_done:
    # token = [r13, r14), length = r15
    mov r15, r14
    sub r15, r13

    test r11, r11
    jne skip_rest

    mov al, byte ptr [rbx + r13]      # first char of token
    cmp r15, 1
    jne check_literal
    # possible operator (single char)
    cmp al, '+'
    je op_add
    cmp al, '-'
    je op_sub
    cmp al, '*'
    je op_mul
    cmp al, '/'
    je op_div
    cmp al, '%'
    je op_mod
    # not an operator, treat as literal
check_literal:
    # literals: optional leading '-', then 1-18 digits, total length ≤19
    cmp r15, 19
    ja set_error_invalid
    cmp al, '-'
    jne parse_pos_lit
    # negative literal, need at least one digit after '-'
    cmp r15, 2
    jb set_error_invalid
    mov rcx, r13
    inc rcx                           # point after '-'
    xor rax, rax                      # value = 0
parse_neg_digits:
    cmp rcx, r14
    jae parsed_neg
    mov dl, byte ptr [rbx + rcx]
    cmp dl, '0'
    jb set_error_invalid
    cmp dl, '9'
    ja set_error_invalid
    imul rax, rax, 10
    sub dl, '0'
    add rax, rdx
    inc rcx
    jmp parse_neg_digits
parsed_neg:
    neg rax
    jmp push_number
parse_pos_lit:
    # positive literal, need at least one digit
    cmp r15, 1
    jb set_error_invalid
    mov rcx, r13
    xor rax, rax
parse_pos_digits:
    cmp rcx, r14
    jae parsed_pos
    mov dl, byte ptr [rbx + rcx]
    cmp dl, '0'
    jb set_error_invalid
    cmp dl, '9'
    ja set_error_invalid
    imul rax, rax, 10
    sub dl, '0'
    add rax, rdx
    inc rcx
    jmp parse_pos_digits
parsed_pos:
    # rax has parsed value
    jmp push_number

set_error_invalid:
    mov r11, 1
    jmp skip_rest

push_number:
    mov [stack_buf + r10*8], rax
    inc r10
    jmp after_token

op_add:
    cmp r10, 2
    jb set_error_invalid
    dec r10
    mov rbx, [stack_buf + r10*8]      # b
    dec r10
    mov rax, [stack_buf + r10*8]      # a
    add rax, rbx
    mov [stack_buf + r10*8], rax
    inc r10
    jmp after_token

op_sub:
    cmp r10, 2
    jb set_error_invalid
    dec r10
    mov rbx, [stack_buf + r10*8]
    dec r10
    mov rax, [stack_buf + r10*8]
    sub rax, rbx
    mov [stack_buf + r10*8], rax
    inc r10
    jmp after_token

op_mul:
    cmp r10, 2
    jb set_error_invalid
    dec r10
    mov rbx, [stack_buf + r10*8]
    dec r10
    mov rax, [stack_buf + r10*8]
    mul rbx                           # unsigned multiply, result in rdx:rax
    mov [stack_buf + r10*8], rax
    inc r10
    jmp after_token

op_div:
    cmp r10, 2
    jb set_error_invalid
    dec r10
    mov rbx, [stack_buf + r10*8]      # divisor b
    dec r10
    mov rax, [stack_buf + r10*8]      # dividend a
    cmp rbx, 0
    je set_error_invalid
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne .no_div_over
    cmp rbx, -1
    je set_error_invalid
.no_div_over:
    cqo
    idiv rbx
    mov [stack_buf + r10*8], rax
    inc r10
    jmp after_token

op_mod:
    cmp r10, 2
    jb set_error_invalid
    dec r10
    mov rbx, [stack_buf + r10*8]      # divisor b
    dec r10
    mov rax, [stack_buf + r10*8]      # dividend a
    cmp rbx, 0
    je set_error_invalid
    mov rcx, 0x8000000000000000
    cmp rax, rcx
    jne .no_mod_over
    cmp rbx, -1
    je set_error_invalid
.no_mod_over:
    cqo
    idiv rbx
    mov [stack_buf + r10*8], rdx
    inc r10
    jmp after_token

after_token:
    mov r12, r14                      # advance token pointer
    jmp token_loop

skip_rest:
    mov r12, r9                       # skip to end of line
    jmp token_loop

token_done:
    # line finished, check for errors or result count
    test r11, r11
    jne line_error
    cmp r10, 1
    jne line_error

    # success: output the single value
    mov rax, [stack_buf]              # value to print
    # convert to decimal string
    lea rdi, [rip + num_buf]
    mov rdx, 32
    add rdi, rdx                      # point to end of buffer
    xor rcx, rcx                      # digit count = 0
    mov rbx, rax                      # signed copy
    cmp rbx, 0
    jne .nonzero
    dec rdi
    mov byte ptr [rdi], '0'
    inc rcx
    jmp .write_number
.nonzero:
    xor r8, r8                        # sign flag 0 = positive
    test rbx, rbx
    jge .positive
    mov r8, 1                         # sign = negative
    neg rbx
.positive:
    mov r9, 10
div_loop:
    xor rdx, rdx
    div r9                            # unsigned divide rbx / 10
    add dl, '0'
    dec rdi
    mov byte ptr [rdi], dl
    inc rcx
    test rax, rax
    jne div_loop
    cmp r8, 0
    je .write_number
    dec rdi
    mov byte ptr [rdi], '-'
    inc rcx
.write_number:
    # write rcx bytes from rdi to stdout
    mov rax, 1                        # sys_write
    mov rdi, 1                        # fd = stdout
    mov rsi, rdi                      # placeholder, will be overwritten
    mov rsi, rdi                      # we need start address in rsi, so:
    mov rsi, rdi                      # (this line will be corrected below)

    # correct register usage:
    mov rsi, rdi          # wrong, we need start address, not fd.
    # Fix: use a temporary register for fd.
    mov r8, 1                     # r8 = fd
    mov rdi, r8
    mov rsi, rdi          # still wrong, we lost start pointer. Let's store start pointer in rbx before syscall.

    # Revised: store start pointer in rbx before clobbering rdi
    mov rbx, rdi          # rbx = start address
    mov rdx, rcx
    mov rax, 1
    mov rdi, 1
    mov rsi, rbx
    syscall

    jmp line_next

line_error:
    # write "error\n"
    mov rax, 1
    mov rdi, 1
    lea rsi, [rip + error_msg]
    mov rdx, error_len
    syscall

line_next:
    # advance cursor to next line
    mov rdx, r9
    cmp r9, rcx
    je after_last_line
    inc rdx
after_last_line:
    jmp line_start

exit_program:
    mov rax, 60
    xor rdi, rdi
    syscall

.section .rodata
error_msg:
    .ascii "error\n"
error_len = . - error_msg

.section .bss
