.globl _start
.text
_start:
    # ------------------------------------------------------------
    # Read entire stdin into input_buf (max 65536 bytes)
    # rcx = total bytes read
    xor rcx, rcx
read_loop:
    mov rax, 0                    # sys_read
    mov rdi, 0                    # fd = stdin
    lea rsi, [rip + input_buf]
    add rsi, rcx                  # buffer pointer = input_buf + rcx
    mov rdx, INPUT_BUF_SIZE
    sub rdx, rcx                  # remaining space
    test rdx, rdx
    je read_done
    syscall
    test rax, rax
    je read_done
    add rcx, rax                  # rcx += bytes read
    cmp rcx, INPUT_BUF_SIZE
    jb read_loop
read_done:
    # rcx = input_len, rdx will be current offset (cursor)
    xor rdx, rdx

line_start:
    cmp rdx, rcx
    jge exit_program                 # cursor >= input_len -> done

    # Find end of current line (newline or EOF)
    mov r8, rdx                      # r8 = scan pointer
find_nl:
    cmp r8, rcx
    jge no_nl
    mov al, byte ptr [rip + input_buf + r8]
    cmp al, 10                       #
