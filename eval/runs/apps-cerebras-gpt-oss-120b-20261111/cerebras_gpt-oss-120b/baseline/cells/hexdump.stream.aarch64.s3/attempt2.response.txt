    .globl _start
    .text
    .align 4

    // system call numbers
    SYS_READ   = 63
    SYS_WRITE  = 64
    SYS_EXIT   = 93

    // file descriptors
    STDIN  = 0
    STDOUT = 1

    // limits
    MAX_INPUT = 65536

    // hex digit table
    .section .rodata
hex_digits:
    .ascii "0123456789abcdef"

    // buffers
    .section .bss
    .align 4
input_buf:
    .zero MAX_INPUT
linebuf:
    .zero 80

    .section .text
_start:
    // x19 = base address of input buffer
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf

    // x20 = total length = 0
    mov     x20, #0

read_loop:
    // read(fd=STDIN, buf = input_buf + total_len, count = MAX_INPUT - total_len)
    mov     x0, #STDIN
    add     x1, x19, x20
    mov     x2, #MAX_INPUT
    sub     x2, x2, x20
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done          // EOF
    add     x20, x20, x0           // total_len += bytes read
    cmp     x20, #MAX_INPUT
    bge     read_done
    b       read_loop

read_done:
    // if total_len == 0, go directly to final offset line
    cbz     x20, final_offset_line

    // x21 = current offset = 0
    mov     x21, #0

dump_loop:
    // x22 = bytes left = total_len - offset
    sub     x22, x20, x21
    // limit to 16
    mov     x23, #16
    cmp     x22, x23
    csel    x22, x22, x23, hi      // if more than 16, keep 16 else keep remaining

    // x24 = pointer to line buffer start
    adrp    x24, linebuf
    add     x24, x24, :lo12:linebuf
    mov     x25, x24               // x25 = current write pointer

    // ---- write offset (8 hex digits) ----
    mov     x26, #7                // nibble index
offset_nib_loop:
    // shift = nibble*4
    mov     x27, x26
    lsl     x27, x27, #2
    lsr     x28, x21, x27          // offset >> shift
    and     x28, x28, #0xf
    adrp    x29, hex_digits
    add     x29, x29, :lo12:hex_digits
    add     x29, x29, x28, uxtw #0
    ldrb    w30, [x29]
    strb    w30, [x25], #1
    subs    x26, x26, #1
    bge     offset_nib_loop

    // two spaces after offset
    mov     w30, #' '
    strb    w30, [x25], #1
    strb    w30, [x25], #1

    // ---- hex byte area (16 slots) ----
    mov     x26, #0                // byte index
hex_slot_loop:
    cmp     x26, #16
    beq     hex_slot_done

    cmp     x26, x22
    bge     hex_slot_spaces        // past actual bytes

    // load byte from input buffer
    add     x27, x19, x21
    add     x27, x27, x26
    ldrb    w28, [x27]

    // high nibble
    lsr     w29, w28, #4
    and     w29, w29, #0xf
    adrp    x30, hex_digits
    add     x30, x30, :lo12:hex_digits
    add     x30, x30, x29, uxtw #0
    ldrb    w31, [x30]
    strb    w31, [x25], #1

    // low nibble
