.globl _start
.text
.align 4

.equ SYS_READ,   63
.equ SYS_WRITE,  64
.equ SYS_EXIT,   93
.equ STDIN,      0
.equ STDOUT,     1
.equ MAX_INPUT,  65536

.section .rodata
hex_digits:
    .ascii "0123456789abcdef"

.section .bss
.align 8
input_buf:
    .zero MAX_INPUT
linebuf:
    .zero 80

.section .text
_start:
    // addresses
    adrp    x19, input_buf
    add     x19, x19, :lo12:input_buf      // x19 = input_buf
    adrp    x22, linebuf
    add     x22, x22, :lo12:linebuf       // x22 = linebuf
    adrp    x23, hex_digits
    add     x23, x23, :lo12:hex_digits    // x23 = hex_digits

    mov     x20, #0                        // total_len = 0

read_loop:
    mov     x0, #STDIN
    add     x1, x19, x20                  // buf = input_buf + total_len
    mov     x2, #MAX_INPUT
    sub     x2, x2, x20                   // count = MAX_INPUT - total_len
    mov     x8, #SYS_READ
    svc     #0
    cbz     x0, read_done
    add     x20, x20, x0                  // total_len += bytes read
    cmp     x20, #MAX_INPUT
    bge     read_done
    b       read_loop

read_done:
    cbz     x20, final_offset             // empty input

    mov     x21, #0                       // offset = 0

dump_loop:
    sub     x24, x20, x21                 // remaining = total_len - offset
    mov     x25, #16
    cmp     x24, x25
    csel    x26, x25, x24, hi            // n = min(16, remaining)

    // build line in linebuf, pos in x27
    mov     x27, #0

    // offset hex (8 nibbles)
    mov     x28, #7
offset_nib_loop:
    mov     x29, x28
    lsl     x29, x29, #2                  // shift = nibble*4
    lsr     x30, x21, x29                 // offset >> shift
    and     x30, x30, #0xf                // nibble
    add     x12, x23, x30                 // addr = hex_digits + nibble
    ldrb    w0, [x12]
    add     x1, x22, x27
    strb    w0, [x1]
    add     x27, x27, #1
    subs    x28, x28, #1
    bge     offset_nib_loop

    // two spaces after offset
    mov     w0, #' '
    add     x1, x22, x27
    strb    w0, [x1]
    add     x27, x27, #1
    add     x1, x22, x27
    strb    w0, [x1]
    add     x27, x27, #1

    // hex byte slots 0..15
    mov     x28, #0                       // i = 0
hex_loop:
    cmp     x28, #16
    beq     hex_done

    cmp     x28, x26
    bge     hex_spaces                    // i >= n

    // load byte from input buffer
    add     x4, x19, x21
    add     x4, x4, x28
    ldrb    w1, [x4]

    // high nibble
    mov     w2, w1, lsr #4
    and     w2, w2, #0xf
    add     x12, x23, x2
    ldrb    w2, [x12]
    add     x1, x22, x27
    strb    w2, [x1]
    add     x27, x27, #1

    // low nibble
    and     w2, w1, #0xf
    add     x12, x23, w2
    ldrb    w2, [x12]
    add
