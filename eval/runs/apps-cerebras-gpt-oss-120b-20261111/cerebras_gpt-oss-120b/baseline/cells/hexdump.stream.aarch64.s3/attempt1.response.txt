	.global _start
	.text
	.align 4

// system call numbers
SYS_READ   = 63
SYS_WRITE  = 64
SYS_EXIT   = 93

// file descriptors
STDIN  = 0
STDOUT = 1

// constants
MAX_INPUT   = 65536
LINE_MAX    = 80   // enough for a line + newline

_start:
	// x19 = pointer to input buffer
	adr	x19, input_buf

	// x20 = total length = 0
	mov	x20, #0

read_loop:
	// x0 = fd (stdin)
	mov	x0, #STDIN
	// x1 = buffer pointer = input_buf + total_len
	add	x1, x19, x20
	// x2 = remaining space = MAX_INPUT - total_len
	mov	x2, #MAX_INPUT
	sub	x2, x2, x20
	// syscall read
	mov	x8, #SYS_READ
	svc	#0
	cbz	x0, read_done          // EOF
	add	x20, x20, x0           // total_len += bytes_read
	cmp	x20, #MAX_INPUT
	bge	read_done
	b	read_loop

read_done:
	// if total_len == 0, just print final offset line
	cmp	x20, #0
	bne	dump_loop_start

	// print "00000000\n"
	adr	x1, linebuf
	adr	x2, hex_digits
	// write 8 zero hex digits
	mov	x3, #8
	mov	x4, #0
write_zero_loop:
	ldrb	w5, [x2]          // '0'
	strb	w5, [x1], #1
	subs	x3, x3, #1
	bne	write_zero_loop
	// newline
	mov	w5, #0x0a
	strb	w5, [x1], #1
	// length = 9
	mov	x2, #9
	mov	x0, #STDOUT
	mov	x8, #SYS_WRITE
	svc	#0
	b	exit

dump_loop_start:
	// x21 = offset = 0
	mov	x21, #0

line_loop:
	// x22 = bytes_in_this_line = min(16, total_len - offset)
	mov	x22, x20
	sub	x22, x22, x21
	cmp	x22, #16
	bgt	over_16
	// x22 already <=16
	b	continue_bytes_calc
over_16:
	mov	x22, #16
continue_bytes_calc:

	// x7 = ptr = linebuf start
	adr	x7, linebuf

	// ---- print offset (8 hex digits) ----
	// i = 7 downto 0
	mov	x5, #7
offset_nibble_loop:
	lsl	x8, x5, #2               // shift = i*4
	lsr	x9, x21, x8
	and	x9, x9, #0xf
	adr	x10, hex_digits
	add	x10, x10, x9
	ldrb	w11, [x10]
	strb	w11, [x7], #1
	subs	x5, x5, #1
	bge	offset_nibble_loop

	// two spaces after offset
	mov	w12, #' '
	strb	w12, [x7], #1
	strb	w12, [x7], #1

	// ---- hex byte area ----
	// i = 0 .. 15
	mov	x5, #0
hex_byte_loop:
	cmp	x5, #16
	beq	hex_byte_done

	cmp	x5, x22
	bge	hex_byte_spaces

	// byte is present
	add	x8, x19, x21
	add	x8, x8, x5
	ldrb	w9, [x8]

	// high nibble
	lsr	w10, w9, #4
	and	w10, w10, #0xf
	adr	x11, hex_digits
	add	x11, x11, x10
	ldrb	w12, [x11]
	strb	w12, [x7], #1

	// low nibble
	and	w10, w9, #0xf
	add	x11, x11, w10
	ldrb
