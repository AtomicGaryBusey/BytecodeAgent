# 1. Assemble (bca.image.assemble_program: refuse file inclusion, macros and the prompt's forbidden
#    directives, then GNU as on stdin under resource limits; AArch64 gets no header)
/usr/bin/aarch64-linux-gnu-as -o edit.o -- < edit.s   # plus -al=edit.lst for the listing in 03-object

# 2. Link: static, no libc, no start files, entry _start (run in 04-executable/)
/usr/bin/aarch64-linux-gnu-ld -static -nostdlib -e _start -Map=edit.map -o edit ../03-object/edit.o
